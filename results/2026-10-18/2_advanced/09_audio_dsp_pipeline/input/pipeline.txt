# Integer audio pipeline: test signals, hum removal, voice, drums, guitar, mixdown
SET fs 8000
SET neg_third $fs / -3
SET low_cut -7 / 2
SET len 3200

# test signals and filter responses
IMPULSE imp $fs 400 0 16384
STEP stp $fs 400 100 -12000
NOISE hiss $fs 3200 20261018
GAIN hiss_q hiss 1 60
ANALYZE hiss 100 2000
FIR binom imp 64 1 6 15 20 15 6 1
FIR diff stp 2 1 0 -1
BIQUAD lp_imp imp 1888 3777 1888 -18236 9405 1451 2903 1451 -14015 3436
RESAMPLE lp_up lp_imp 12000
EXCERPT lp_imp 0 24
EXCERPT lp_up 0 20
EXCERPT diff 96 12
MOVAVG stp_avg stp 7
EXCERPT stp_avg 96 16
EXCERPT stp_avg 390 16
REVERB wet imp 3
COMPARE binom lp_imp
GAIN silence imp 0 1
NORMALIZE silence_n silence 20000
ANALYZE silence_n 100 2000
DELAY late_imp imp 007

# hum removal
DCBLOCK hum_dc hum 32440 32768
BIQUAD hum_clean hum_dc 16225 -32424 16225 -32424 16066 15916 -31612 15916 -31612 15449
ANALYZE hum 20 200
ANALYZE hum_clean 20 200
PLOT hum 0 2400 60 5 4096 64
PLOT hum_clean 0 2400 60 5 4096 64
DCBLOCK hum_bad hum 32768 32768

# voice chain
MIX noisy 1 voice 1 hiss_q 1
GATE gated noisy 900 160
BIQUAD voice_hp gated 14661 -29323 14661 -29141 13120
COMPRESS voice_c voice_hp 6000 4 1 8192 96
GAIN voice_up voice_c 3 2
GAIN voice_up voice_c 3
ECHO voice_echo voice_up 1200 -9830 32768 2000
ECHO voice_ring voice_up 100 40000 32768 0
ENVELOPE voice_env voice 6554 160
EXCERPT voice_env 400 16
ANALYZE voice 60 400
ANALYZE voice_echo 60 400
ANALYZE voice 3000 3999
EXPECT voice samples eq $len

PITCHTRACK voice 100 50 60 400 1000
PLOT voice 0 3200 80 8 6554 160
PLOT voice_echo 1000 1600 64 6 6554 160
METER voice 700
EXCERPT voice_c 640 24
GAIN Loud voice 3 2
GAIN loud vocals 3 2

# drums
COMPRESS drums_c drums 9000 3 1 16384 48
NORMALIZE drums_n drums_c 26000
FADE drums_f drums_n 200 800
FADE drums_g drums_n 9000 0
BIQUAD drums_eq drums 20647 0 3554 0 7817
ANALYZE drums 40 200
ANALYZE drums_f 40 200
METER drums_f 800
ONSETS drums 16384 66 8000 800
ONSETS drums_f 16384 66 30000 800
HISTOGRAM drums 16
HISTOGRAM drums_c 16
DELAY drums_late drums 17000

# guitar
RESAMPLE g8 guitar $fs
ANALYZE guitar 80 600
ANALYZE g8 80 600
TRIM g_head g8 0 2400
DELAY g_late g_head 800
EXCERPT g8 100 16
MIX bad 2 voice 1 guitar 1
MIX bad 2 voice 1 drums
GAIN g_soft g8 1.5 2

# mixdown
MIX mixdown 4 voice_echo 3 drums_f 2 g_late 2
GAIN hot mixdown 5 2
ANALYZE hot 60 400
NORMALIZE master hot 30000
TRIM voice_a voice 400 1100
CONCAT both voice_a drums_f g_head
FADE g_head g_head 100 100
COMPARE mixdown master
COMPARE master master
EXPECT master peak eq 30000
EXPECT hot clipped gt 0
EXPECT master samples lt 5000
EXPECT master loudness ge 3
EXPECT master peak >= 3
GAIN boost master $gain 1
ANALYZE imp 10 400
SET zero $fs / 0
SET big 100000 * 100000
METER master 800
PLOT master 0 8000 100 7 3277 64
TRACE master
SAVE master
SAVE hum_clean
SAVE g_head
