# Classical cipher jobs

# Round trips with the substitution ciphers
ENCRYPT caesar 3 note AS note_c
DECRYPT caesar 3 note_c AS note_back
COMPARE note note_back
ENCRYPT affine 5,8 note AS note_a
DECRYPT affine 05,08 note_a
ENCRYPT vigenere lemon note AS note_v
DECRYPT vigenere LEMON note_v
ENCRYPT autokey Queen note AS note_k
DECRYPT autokey QUEEN note_k
ENCRYPT autokey Swallow verse AS verse_k
DECRYPT autokey SWALLOW verse_k

# Transposition and Playfair
ENCRYPT railfence 3 note AS note_r
DECRYPT railfence 03 note_r
ENCRYPT columnar zebras note AS note_z
DECRYPT columnar 632415 note_z
ENCRYPT columnar Balloon9 note AS note_b
DECRYPT columnar BALLOON9 note_b AS note_b2
COMPARE note note_b2
ENCRYPT playfair Monarchy jazz AS jazz_p
DECRYPT playfair MONARCHY jazz_p
ENCRYPT playfair Jackdaw note
ENCRYPT playfair NIGHTINGALE garden AS garden_p
COMPARE garden_p intercept_h

# Statistics
ANALYZE lighthouse
ANALYZE intercept_c MAXLEN=12

# Breaking intercepted messages
BREAK caesar intercept_a AS sol_a
COMPARE market sol_a
BREAK affine intercept_b TOP=8 AS sol_b
COMPARE bread sol_b
BREAK vigenere intercept_c MAXLEN=12 AS sol_c
COMPARE lighthouse sol_c
BREAK vigenere intercept_d KASISKI=80
BREAK vigenere intercept_d IOC=7000 TOP=3
BREAK vigenere intercept_i MAXLEN=16 AS sol_i
BREAK vigenere intercept_i REFINE=0 MAXLEN=16 TOP=2
COMPARE castle sol_i
BREAK autokey intercept_e AS sol_e
COMPARE desert sol_e
BREAK railfence intercept_f TOP=3
BREAK columnar intercept_g AS sol_g
COMPARE castle sol_g
BREAK playfair intercept_h KEYS=keylist TOP=4 AS sol_h
COMPARE garden sol_h

# Automatic identification
IDENTIFY intercept_a
IDENTIFY intercept_b
IDENTIFY intercept_d
IDENTIFY intercept_e
IDENTIFY intercept_f
IDENTIFY intercept_g
IDENTIFY intercept_h
IDENTIFY intercept_h KEYS=keylist
IDENTIFY intercept_c KEYS=keylist
