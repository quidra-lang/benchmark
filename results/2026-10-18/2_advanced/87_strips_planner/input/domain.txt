; courier domain: trucks drive along roads, drones fly between any two places
; a drone needs a charge for every flight and recharges only at a station
(define (domain courier)
  (:requirements :strips :typing :negative-preconditions :equality)
  (:types truck drone - vehicle
          vehicle parcel - mobile
          place)
  (:predicates
    (at ?m - mobile ?p - place)
    (in ?c - parcel ?v - vehicle)
    (loaded ?d - drone)
    (charged ?d - drone)
    (road ?a ?b - place)
    (station ?p - place)
    (heavy ?c - parcel)
    (closed ?p - place)
    (fragile ?c - parcel))
  (:action drive
    :parameters (?t - truck ?from ?to - place)
    :precondition (and (at ?t ?from) (road ?from ?to) (not (closed ?to)))
    :effect (and (not (at ?t ?from)) (at ?t ?to) (increase (total-cost) 2)))
  (:action load
    :parameters (?c - parcel ?t - truck ?p - place)
    :precondition (and (at ?c ?p) (at ?t ?p))
    :effect (and (not (at ?c ?p)) (in ?c ?t)))
  (:action unload
    :parameters (?c - parcel ?t - truck ?p - place)
    :precondition (and (in ?c ?t) (at ?t ?p))
    :effect (and (not (in ?c ?t)) (at ?c ?p)))
  (:action fly
    :parameters (?d - drone ?from ?to - place)
    :precondition (and (at ?d ?from) (charged ?d) (not (= ?from ?to)) (not (closed ?to)))
    :effect (and (not (at ?d ?from)) (at ?d ?to) (not (charged ?d)) (increase (total-cost) 3)))
  (:action recharge
    :parameters (?d - drone ?p - place)
    :precondition (and (at ?d ?p) (station ?p) (not (charged ?d)))
    :effect (and (charged ?d) (increase (total-cost) 2)))
  (:action wrap
    :parameters (?c - parcel ?p - depot)
    :precondition (and (at ?c ?p) (not (fragile ?c)) (at ?x ?p))
    :effect (and (fragile ?c) (increase (total-cost) 1)))
  (:action tow
    :parameters (?t - truck ?d - drone ?from ?to - place)
    :precondition (and (at ?t ?from) (at ?d ?from) (road ?from ?to) (not (charged ?t)))
    :effect (and (not (at ?t ?from)) (not (at ?d ?from)) (at ?t ?to) (at ?d ?to)
                 (increase (total-cost) 4)))
  (:action pick
    :parameters (?c - parcel ?d - drone ?p - place)
    :precondition (and (at ?c ?p) (at ?d ?p) (not (heavy ?c)) (not (loaded ?d)))
    :effect (and (not (at ?c ?p)) (in ?c ?d) (loaded ?d)))
  (:action drop
    :parameters (?c - parcel ?d - drone ?p - place)
    :precondition (and (in ?c ?d) (at ?d ?p))
    :effect (and (not (in ?c ?d)) (not (loaded ?d)) (at ?c ?p)))
)
