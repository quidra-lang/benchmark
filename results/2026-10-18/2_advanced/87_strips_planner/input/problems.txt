; p1: one truck carries a heavy parcel along a chain of roads
(define (problem p1)
  (:domain courier)
  (:objects t1 - truck d1 - drone c1 c2 - parcel h1 s1 s2 s3 - place)
  (:init (at t1 h1) (at d1 h1) (charged d1) (at c1 h1) (at c2 s1) (heavy c1)
         (station h1)
         (road h1 s1) (road s1 h1) (road s1 s2) (road s2 s1) (road s2 s3) (road s3 s2))
  (:goal (and (at c1 s3) (at c2 h1))))

; p2: two stations, so the drone can help the truck
(define (problem p2)
  (:domain courier)
  (:objects t1 - truck d1 - drone c1 c2 c3 - parcel h1 h2 s1 s2 s3 - place)
  (:init (at t1 h1) (at d1 h1) (charged d1)
         (at c1 h1) (at c2 s1) (at c3 h2) (heavy c1) (heavy t1)
         (station h1) (station h2)
         (road h1 s1) (road s1 h1) (road s1 s2) (road s2 s1) (road s2 h2) (road h2 s2)
         (road h2 s3) (road s3 h2) (road s1 s2))
  (:goal (and (at c1 s3) (at c2 h2) (at c3 s1))))

; p3: the drone is stranded after its first flight
(define (problem p3)
  (:domain courier)
  (:objects d1 - drone c1 c2 - parcel h1 s1 s2 - place)
  (:init (at d1 h1) (charged d1) (at c1 s1) (at c2 s2) (station h1))
  (:goal (and (at c1 h1) (at c2 h1))))

; p4: the heavy parcel cannot enter the closed place
(define (problem p4)
  (:domain courier)
  (:objects t1 - truck d1 - drone c1 c2 - parcel h1 s1 s2 - place)
  (:init (at t1 h1) (at d1 h1) (charged d1) (at c1 h1) (at c2 h1) (heavy c2)
         (station h1) (closed s2) (fuel t1) (at c3 s1)
         (road h1 s1) (road s1 h1) (road s1 s2) (road s2 s1) (road s2))
  (:goal (and (at c1 s1) (at c2 s2) (at c3 h1))))
