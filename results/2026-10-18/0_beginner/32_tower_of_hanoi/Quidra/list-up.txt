int hanoi(int n, string source, string target, string auxiliary)
    if n == 0
        return 0
    int first = hanoi(n - 1, source, auxiliary, target)
    print("move disk {n} from {source} to {target}{NL}")
    int last = hanoi(n - 1, auxiliary, target, source)
    return first + 1 + last
