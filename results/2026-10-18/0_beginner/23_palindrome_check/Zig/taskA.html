1const std = @import("std");
2
3pub fn main(init: std.process.Init) !void {
4 var output_buffer: [8192]u8 = undefined;
5 var writer = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
6 const out = &writer.interface;
7 const phrases = [_][]const u8{
8 "Never odd or even",
9 "Race a car",
10 "Was it a car or a cat I saw?",
11 "Hello, World!",
12 "Step on no pets.",
13 "No lemon, no melon",
14 };
15 for (phrases) |phrase| {
16 var clean: [100]u8 = undefined;
17 var len: usize = 0;
18 for (phrase) |c| {
19 const lower = std.ascii.toLower(c);
20 if (lower >= 'a' and lower <= 'z') {
21 clean[len] = lower;
22 len += 1;
23 }
24 }
25 var palindrome = true;
26 for (0..len / 2) |i| {
27 if (clean[i] != clean[len - 1 - i]) {
28 palindrome = false;
29 break;
30 }
31 }
32 try out.print("{s} -> {s}\n", .{ phrase, if (palindrome) @as([]const u8, "yes") else "no" });
33 }
34 try out.flush();
35}
