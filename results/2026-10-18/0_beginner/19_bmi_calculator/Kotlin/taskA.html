1import java.util.Locale
2
3fun main() {
4 val names = listOf("Aiko", "Kenji", "Mika", "Ryo", "Yuki")
5 val heights = intArrayOf(158, 172, 165, 180, 150)
6 val weights = intArrayOf(45, 68, 70, 101, 52)
7 for (i in names.indices) {
8 val bmi = (weights[i] * 10000).toDouble() / (heights[i] * heights[i])
9 val category = when {
10 bmi < 18.5 -> "underweight"
11 bmi < 25 -> "normal"
12 bmi < 30 -> "overweight"
13 else -> "obese"
14 }
15 println("%s: BMI %.1f (%s)".format(Locale.ROOT, names[i], bmi, category))
16 }
17}
