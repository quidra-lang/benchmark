1import java.net.URLDecoder;
2import java.nio.charset.StandardCharsets;
3import java.util.ArrayList;
4import java.util.List;
5import java.util.Map;
6import java.util.TreeMap;
7
8public class Main {
9 public static void main(String[] args) {
10 String[] queries = {"name=Alice+Smith&age=30&city=New%20York",
11 "q=c%2B%2B+tutorial&page=2&sort=desc&page=3",
12 "msg=50%25+off%21&debug&lang=en&ref=&lang=ja",
13 "redirect=https%3a%2f%2fexample.com%2Fhome&token=abc==&item10=pen&item2=fish%26chips"};
14 for (int index = 0; index < queries.length; index++) {
15 Map<String, List<String>> values = new TreeMap<>();
16 for (String item : queries[index].split("&")) {
17 int equals = item.indexOf('=');
18 String key = URLDecoder.decode(equals < 0 ? item : item.substring(0, equals), StandardCharsets.UTF_8);
19 String value = URLDecoder.decode(equals < 0 ? "" : item.substring(equals + 1), StandardCharsets.UTF_8);
20 values.computeIfAbsent(key, k -> new ArrayList<>()).add(value);
21 }
22 System.out.println("query " + (index + 1) + ":");
23 for (var entry : values.entrySet()) {
24 List<String> quoted = entry.getValue().stream().map(v -> "\"" + v + "\"").toList();
25 System.out.println(" " + entry.getKey() + ": " + String.join(", ", quoted));
26 }
27 }
28 }
29}
