3fn decode(s: &str) -> String {
4 let b = s.as_bytes();
5 let (mut out, mut i) = (String::new(), 0);
6 while i < b.len() {
7 if b[i] == b'+' {
8 out.push(' ');
9 i += 1;
10 } else if b[i] == b'%' {
11 let value = u8::from_str_radix(&s[i + 1..i + 3], 16).unwrap();
12 out.push(value as char);
13 i += 3;
14 } else {
15 out.push(b[i] as char);
16 i += 1;
17 }
18 }
19 out
20}
