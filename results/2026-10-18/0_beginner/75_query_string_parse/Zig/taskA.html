1const std = @import("std");
2
3const Entry = struct { key: [256]u8, klen: usize, value: [256]u8, vlen: usize };
4
5fn less(_: void, a: Entry, b: Entry) bool {
6 return std.mem.order(u8, a.key[0..a.klen], b.key[0..b.klen]) == .lt;
7}
8
9fn decode(s: []const u8, buffer: *[256]u8) usize {
10 const component = buffer[0..s.len];
11 @memcpy(component, s);
12 for (component) |*ch| {
13 if (ch.* == '+') ch.* = ' ';
14 }
15 const decoded = std.Uri.percentDecodeInPlace(component);
16 std.mem.copyForwards(u8, buffer[0..decoded.len], decoded);
17 return decoded.len;
18}
19
20pub fn main(init: std.process.Init) !void {
21 var output_buffer: [8192]u8 = undefined;
22 var writer = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
23 const out = &writer.interface;
24 const queries = [_][]const u8{
25 "name=Alice+Smith&age=30&city=New%20York",
26 "q=c%2B%2B+tutorial&page=2&sort=desc&page=3",
27 "msg=50%25+off%21&debug&lang=en&ref=&lang=ja",
28 "redirect=https%3a%2f%2fexample.com%2Fhome&token=abc==&item10=pen&item2=fish%26chips",
29 };
30 for (queries, 0..) |query, k| {
31 var entries: [16]Entry = undefined;
32 var len: usize = 0;
33 var items = std.mem.splitScalar(u8, query, '&');
34 while (items.next()) |item| {
35 const split = std.mem.findScalar(u8, item, '=') orelse item.len;
36 entries[len].klen = decode(item[0..split], &entries[len].key);
37 entries[len].vlen = decode(if (split < item.len) item[split + 1 ..] else "", &entries[len].value);
38 len += 1;
39 }
40 std.mem.sort(Entry, entries[0..len], {}, less);
41 try out.print("query {d}:\n", .{k + 1});
42 for (entries[0..len], 0..) |entry, i| {
43 const same = i > 0 and
44 std.mem.eql(u8, entry.key[0..entry.klen], entries[i - 1].key[0..entries[i - 1].klen]);
45 if (same) {
46 try out.writeAll(", ");
47 } else {
48 if (i > 0)
49 try out.writeByte('\n');
50 try out.print(" {s}: ", .{entry.key[0..entry.klen]});
51 }
52 try out.print("\"{s}\"", .{entry.value[0..entry.vlen]});
53 }
54 try out.writeByte('\n');
55 }
56 try out.flush();
57}
