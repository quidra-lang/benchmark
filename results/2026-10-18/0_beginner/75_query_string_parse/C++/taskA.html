8std::string decode(const std::string &s) {
9 std::string out;
10 for (std::size_t i = 0; i < s.size(); i++) {
11 if (s[i] == '+') {
12 out += ' ';
13 } else if (s[i] == '%') {
14 out += static_cast<char>(std::stoi(s.substr(i + 1, 2), nullptr, 16));
15 i += 2;
16 } else {
17 out += s[i];
18 }
19 }
20 return out;
21}

23int main() {
24 std::array<std::string, 4> queries = {
25 "name=Alice+Smith&age=30&city=New%20York",
26 "q=c%2B%2B+tutorial&page=2&sort=desc&page=3",

28 "redirect=https%3a%2f%2fexample.com%2Fhome&token=abc==&item10=pen&item2=fish%26chips",
29 };
30 for (int k = 0; k < 4; k++) {
31 std::map<std::string, std::vector<std::string>> values;
32 std::istringstream input(queries[k]);
33 for (std::string item; std::getline(input, item, '&');) {
34 std::size_t at = item.find('=');
35 std::string key = decode(item.substr(0, at));
36 std::string value = at == std::string::npos ? "" : decode(item.substr(at + 1));
37 values[key].push_back(value);
38 }
39 std::cout << "query " << k + 1 << ":\n";
40 for (const auto &[key, items] : values) {
41 std::cout << " " << key << ": ";
42 for (std::size_t i = 0; i < items.size(); i++) {
43 if (i)
44 std::cout << ", ";
45 std::cout << '"' << items[i] << '"';
46 }
