1using System;
2using System.Collections.Generic;
3using System.Linq;
4using System.Net;
5
6string[] queries = { "name=Alice+Smith&age=30&city=New%20York",
7 "q=c%2B%2B+tutorial&page=2&sort=desc&page=3",
8 "msg=50%25+off%21&debug&lang=en&ref=&lang=ja",
9 "redirect=https%3a%2f%2fexample.com%2Fhome&token=abc==&item10=pen&item2=fish%26chips" };
10for (int index = 0; index < queries.Length; index++)
11{
12 var values = new SortedDictionary<string, List<string>>(StringComparer.Ordinal);
13 foreach (string item in queries[index].Split('&'))
14 {
15 int equals = item.IndexOf('=');
16 string key = WebUtility.UrlDecode(equals < 0 ? item : item[..equals]);
17 string value = WebUtility.UrlDecode(equals < 0 ? "" : item[(equals + 1)..]);
18 if (!values.ContainsKey(key)) values[key] = new List<string>();
19 values[key].Add(value);
20 }
21 Console.WriteLine($"query {index + 1}:");
22 foreach (var pair in values)
23 {
24 var quoted = pair.Value.Select(v => "\"" + v + "\"");
25 Console.WriteLine($" {pair.Key}: {string.Join(", ", quoted)}");
26 }
27}
