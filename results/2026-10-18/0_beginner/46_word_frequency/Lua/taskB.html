1local text = "The cat sat on the mat. The dog sat on the rug, and the cat watched the dog. A bird sang on the fence, and the Dog barked at the bird. The cat slept in the sun, the bird flew away, and the dog sat by the door."
2local frequencies = {}
3for word in text:lower():gmatch("[a-z]+") do frequencies[word] = (frequencies[word] or 0) + 1 end
4local words = {}
5for word in pairs(frequencies) do words[#words + 1] = word end
6table.sort(words, function(a, b)
7 if frequencies[a] ~= frequencies[b] then return frequencies[a] > frequencies[b] end
8 return a < b
9end)
10for i = 1, 5 do io.write(string.format("%s: %d\n", words[i], frequencies[words[i]])) end
