1package main
2
3import (
4 "fmt"
5 "sort"
6 "strings"
7)
8
9func main() {
10 text := "The cat sat on the mat. The dog sat on the rug, and the cat watched the dog. A bird sang on the fence, and the Dog barked at the bird. The cat slept in the sun, the bird flew away, and the dog sat by the door."
11 words := strings.FieldsFunc(strings.ToLower(text), func(c rune) bool {
12 return c < 'a' || c > 'z'
13 })
14 counts := map[string]int{}
15 for _, w := range words {
16 counts[w]++
17 }
18 entries := []string{}
19 for w := range counts {
20 entries = append(entries, w)
21 }
22 sort.Slice(entries, func(i, j int) bool {
23 a, b := entries[i], entries[j]
24 if counts[a] != counts[b] {
25 return counts[a] > counts[b]
26 }
27 return a < b
28 })
29 for _, w := range entries[:5] {
30 fmt.Printf("%s: %d\n", w, counts[w])
31 }
32 fmt.Printf("distinct words: %d\n", len(entries))
33}
