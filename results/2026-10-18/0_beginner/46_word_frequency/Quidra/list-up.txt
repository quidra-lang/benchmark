string text = "The cat sat on the mat. The dog sat on the rug, and the cat watched the dog. A bird sang on the fence, and the Dog barked at the bird. The cat slept in the sun, the bird flew away, and the dog sat by the door."
string alphabet = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ"
string normalized = ""
for nat i < len(text)
    match alphabet.find(text[i])
        nat position
            normalized += alphabet[position % 26]
        none
            normalized += " "
string[] tokens = normalized.split(" ")
set.Set<string> distinct = set.Set<string>()
for string word = tokens[]
    if word != ""
        .distinct.add(word)
string[] words = distinct.values().sorted()
int[] counts = array(len(words), fill = 0)
for nat i < len(words)
    for string word = tokens[]
        if word == words[i]
            counts[i] += 1
bool[] used = array(len(words), fill = false)
for 0 <= int rank < 5
    int best = -1
    for nat i < len(words)
        if not used[i]
            if best == -1 or counts[i] > counts[best]
                best = int(i)
    used[best] = true
    print("{words[best]}: {counts[best]}{NL}")
