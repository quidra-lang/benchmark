1import java.util.ArrayList;
2import java.util.Comparator;
3import java.util.HashMap;
4import java.util.List;
5import java.util.Locale;
6import java.util.Map;
7
8public class Main {
9 public static void main(String[] args) {
10 String text = "The cat sat on the mat. The dog sat on the rug, and the cat watched the dog. A bird sang on the fence, and the Dog barked at the bird. The cat slept in the sun, the bird flew away, and the dog sat by the door.";
11 Map<String, Integer> counts = new HashMap<>();
12 for (String word : text.toLowerCase(Locale.ROOT).split("[^a-z]+")) {
13 if (!word.isEmpty()) {
14 counts.merge(word, 1, Integer::sum);
15 }
16 }
17 List<String> words = new ArrayList<>(counts.keySet());
18 words.sort(Comparator.<String>comparingInt(counts::get).reversed().thenComparing(Comparator.naturalOrder()));
19 for (String word : words.subList(0, 5)) {
20 System.out.println(word + ": " + counts.get(word));
21 }
22 System.out.println("distinct words: " + counts.size());
23 }
24}
