1const std = @import("std");

3const Entry = struct { word: []const u8, count: u32 };
4
5fn less(_: void, a: Entry, b: Entry) bool {
6 return if (a.count != b.count) a.count > b.count else std.mem.order(u8, a.word, b.word) == .lt;
7}
8
9pub fn main(init: std.process.Init) !void {
10 var output_buffer: [8192]u8 = undefined;
11 var writer = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
12 const out = &writer.interface;
13 const text = "The cat sat on the mat. The dog sat on the rug, and the cat watched the dog. A bird sang on the fence, and the Dog barked at the bird. The cat slept in the sun, the bird flew away, and the dog sat by the door.";
14 var lower: [text.len]u8 = undefined;
15 _ = std.ascii.lowerString(&lower, text);
16 var counts: std.StringHashMapUnmanaged(u32) = .empty;
17 defer counts.deinit(std.heap.page_allocator);
18 var words = std.mem.tokenizeAny(u8, &lower, " ,.");
19 while (words.next()) |word| {
20 const entry = try counts.getOrPut(std.heap.page_allocator, word);
21 if (!entry.found_existing)
22 entry.value_ptr.* = 0;
23 entry.value_ptr.* += 1;
24 }
25 var entries: [100]Entry = undefined;
26 const len = counts.count();
27 var iterator = counts.iterator();
28 var i: usize = 0;
29 while (iterator.next()) |entry| : (i += 1) {
30 entries[i] = .{ .word = entry.key_ptr.*, .count = entry.value_ptr.* };
31 }
32 std.mem.sort(Entry, entries[0..len], {}, less);
33 for (entries[0..5]) |entry|
34 try out.print("{s}: {d}\n", .{ entry.word, entry.count });
35 try out.print("distinct words: {d}\n", .{len});
36 try out.flush();
37}
