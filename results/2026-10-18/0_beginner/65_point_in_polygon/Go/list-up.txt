		inside := false
			if (y1 > py) != (y2 > py) {
				intersection := float64(x1) + float64(x2-x1)*float64(py-y1)/float64(y2-y1)
				if intersection > float64(px) {
					inside = !inside
				}
			}
		result := "outside"
		if inside {
			result = "inside"
		}
		fmt.Printf("(%d, %d): %s\n", px, py, result)
