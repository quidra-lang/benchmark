    int crossings = 0
        if (y1 > py) != (y2 > py)
            real64 intersection = real64(x1) + real64(x2 - x1) * real64(py - y1) / real64(y2 - y1)
            if intersection > real64(px)
                crossings += 1
    string result = if crossings % 2 == 1 then "inside" else "outside"
    print("({px}, {py}): {result}{NL}")
