    point_x := real64(px)
    int crossings := 0
        if (y1 > py) != (y2 > py)
            intersection := real64(x2 - x1)
            intersection *= real64(py - y1)
            intersection /= real64(y2 - y1)
            intersection += real64(x1)
            if intersection > point_x
                crossings += 1
    result := if crossings % 2 == 1 then "inside" else "outside"
    println("({px}, {py}): {result}")
