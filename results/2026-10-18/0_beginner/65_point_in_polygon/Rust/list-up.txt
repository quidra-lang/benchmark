        let mut inside = false;
            if (y1 > py) != (y2 > py) {
                let intersection =
                    f64::from(x1) + f64::from(x2 - x1) * f64::from(py - y1) / f64::from(y2 - y1);
                if intersection > f64::from(px) {
                    inside = !inside;
                }
            }
        println!(
            "({}, {}): {}",
            px,
            py,
            if inside { "inside" } else { "outside" }
        );
