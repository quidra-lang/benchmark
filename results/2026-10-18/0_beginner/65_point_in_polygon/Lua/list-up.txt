    local crossings = 0
        if (y1 > py) ~= (y2 > py) then
            local intersection = x1 + (x2 - x1) * (py - y1) / (y2 - y1)
            if intersection > px then crossings = crossings + 1 end
        end
    io.write(string.format("(%d, %d): %s\n", px, py, crossings % 2 == 1 and "inside" or "outside"))
