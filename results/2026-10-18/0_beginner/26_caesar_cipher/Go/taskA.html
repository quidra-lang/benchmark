5func cipher(s string, shift byte) string {
6 out := []byte(s)
7 for i, c := range out {
8 if c >= 'a' && c <= 'z' {
9 out[i] = 'a' + (c-'a'+shift)%26
10 } else if c >= 'A' && c <= 'Z' {
11 out[i] = 'A' + (c-'A'+shift)%26
12 }
13 }
14 return string(out)
15}
