local function caesar(message, shift)
    return (message:gsub("[A-Za-z]", function(letter)
        local code = letter:byte()
        local base = code <= 90 and 65 or 97
        return string.char(base + (code - base + shift) % 26)
local ciphertext = caesar("Zoe, meet Max by the Old Bridge at six!", 3)
