4void cipher(std::string &s, int shift) {
5 for (char &c : s) {
6 if (c >= 'a' && c <= 'z')
7 c = static_cast<char>('a' + (c - 'a' + shift) % 26);
8 else if (c >= 'A' && c <= 'Z')
9 c = static_cast<char>('A' + (c - 'A' + shift) % 26);
10 }
11}
