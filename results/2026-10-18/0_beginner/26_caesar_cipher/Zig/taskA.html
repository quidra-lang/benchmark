3fn shift(c: u8, amount: u8) u8 {
4 const base: u8 = if (c >= 'a' and c <= 'z') 'a' else if (c >= 'A' and c <= 'Z') 'A' else return c;
5 return base + (c - base + amount) % 26;
6}

8pub fn main(init: std.process.Init) !void {
9 var output_buffer: [8192]u8 = undefined;
10 var writer = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
11 const out = &writer.interface;
12 const message = "Zoe, meet Max by the Old Bridge at six!";
13 var cipher: [message.len]u8 = undefined;
14 var plain: [message.len]u8 = undefined;
15 for (message, 0..) |c, i|
16 cipher[i] = shift(c, 3);
17 for (cipher, 0..) |c, i|
18 plain[i] = shift(c, 23);
19 try out.print("{s}\n{s}\n", .{ cipher, plain });
20 try out.flush();
21}
