string caesar(string source, int shift)
    int offset = (shift % 26 + 26) % 26
    string lower = "abcdefghijklmnopqrstuvwxyz"
    string upper = "ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    string shifted = lower[offset:] + lower[:<offset] + upper[offset:] + upper[:<offset]
    return source.translate(lower + upper, shifted)
