string caesar(string source, int shift)
    offset := (shift % 26 + 26) % 26
    lower := "abcdefghijklmnopqrstuvwxyz"
    upper := "ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    shifted := lower[offset:] + lower[:<offset] + upper[offset:] + upper[:<offset]
    return source.translate(lower + upper, shifted)
encrypted := caesar(message, 3)
print.line("{encrypted}{NL}{caesar(encrypted, -3)}")
