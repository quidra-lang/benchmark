const numbers = [
  "688126340316", "0128469655154", "92181914514156", "177332430060552", "7982377590420309",
for (const number of numbers) {
  const total = [...number].reverse().reduce((sum, char, i) => {
    const digit = Number(char) * (i % 2 === 1 ? 2 : 1);
    return sum + (digit > 9 ? digit - 9 : digit);
  }, 0);
  const expected = (10 - (total - Number(number.at(-1))) % 10) % 10;
  const result = total % 10 === 0 ? "valid" : `invalid (expected check digit ${expected})`;
  console.log(`${number}: ${result}`);
