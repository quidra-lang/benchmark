1using System;
2using System.Collections.Generic;
3using System.Globalization;
4
5CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
6
7var cache = new List<char>();
8int hits = 0, misses = 0;
9foreach (char key in "BACBDABEADDCAEB")
10{
11 string status;
12 if (cache.Remove(key))
13 {
14 hits++;
15 status = "hit";
16 }
17 else
18 {
19 misses++;
20 status = "miss";
21 if (cache.Count == 3)
22 {
23 status += $", evict {cache[^1]}";
24 cache.RemoveAt(cache.Count - 1);
25 }
26 }
27 cache.Insert(0, key);
28 Console.WriteLine($"{key}: {status} [{string.Join(" ", cache)}]");
29}
