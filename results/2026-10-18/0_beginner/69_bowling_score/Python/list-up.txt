    index = total = 0
    cumulative = []
    for _ in range(9):
        if rolls[index] == 10:
            total += 10 + rolls[index + 1] + rolls[index + 2]
            index += 1
        else:
            score = rolls[index] + rolls[index + 1]
            total += score + (rolls[index + 2] if score == 10 else 0)
            index += 2
        cumulative.append(total)
    total += sum(rolls[index:])
    cumulative.append(total)
    print(f"game {number}:", *cumulative)
