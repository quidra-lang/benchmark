let commands = ["undo", "type hello world", "delete 5", "type there", "undo", "undo", "redo",
    "type friend", "redo", "delete 20", "undo", "redo", "redo", "type see you"]
var undo = [String](), redo = [String](), buffer = ""
for command in commands {
    var special: String? = nil
    if command == "undo" {
        if let previous = undo.popLast() {
            redo.append(buffer)
            buffer = previous
        } else {
            special = "nothing to undo"
        }
    } else if command == "redo" {
        if let next = redo.popLast() {
            undo.append(buffer)
            buffer = next
        } else {
            special = "nothing to redo"
        }
    } else {
        undo.append(buffer)
        redo.removeAll()
        if command.hasPrefix("type ") {
            buffer += command.dropFirst(5)
        } else {
            buffer = String(buffer.prefix(max(0, buffer.count - Int(command.dropFirst(7))!)))
        }
    }
}
