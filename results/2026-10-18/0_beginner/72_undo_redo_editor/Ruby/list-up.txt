commands = [
  'undo', 'type hello world', 'delete 5', 'type there', 'undo', 'undo', 'redo',
  'type friend', 'redo', 'delete 20', 'undo', 'redo', 'redo', 'type see you'
]
buffer = ''
undo_stack, redo_stack = [], []
commands.each do |command|
  if command == 'undo'
    if undo_stack.empty?
      puts 'undo => nothing to undo'
      next
    end
    redo_stack << buffer
    buffer = undo_stack.pop
  elsif command == 'redo'
    if redo_stack.empty?
      puts 'redo => nothing to redo'
      next
    end
    undo_stack << buffer
    buffer = redo_stack.pop
  else
    undo_stack << buffer
    redo_stack.clear
    if command.start_with?('type ')
      buffer += command[5..]
    else
      buffer = buffer[0, [0, buffer.length - command[7..].to_i].max]
    end
  end
end
