import collections
import math
commands := [
    "undo", "type hello world", "delete 5", "type there", "undo", "undo", "redo", "type friend",
    "redo", "delete 20", "undo", "redo", "redo", "type see you",
]
buffer := ""
string[] undo := []
string[] redo := []
for command := commands[]
    if command == "undo"
        if undo.len() == 0
            continue
        .redo.push(buffer)
        buffer = .undo.take()
    elif command == "redo"
        if redo.len() == 0
            continue
        .undo.push(buffer)
        buffer = .redo.take()
    else
        .undo.push(buffer)
        redo = []
        if command.starts_with("type ")
            buffer += command[5:]
        else
            count := int(command[7:])
            keep := math.max(0, int(buffer.len()) - count)
            buffer = buffer[:<keep]
