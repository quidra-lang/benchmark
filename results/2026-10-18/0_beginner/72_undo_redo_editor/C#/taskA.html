1using System;
2using System.Collections.Generic;
3using System.Globalization;
4
5CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
6
7string[] commands = { "undo", "type hello world", "delete 5", "type there", "undo", "undo", "redo",
8 "type friend", "redo", "delete 20", "undo", "redo", "redo", "type see you" };
9var undo = new Stack<string>();
10var redo = new Stack<string>();
11string buffer = "";
12foreach (string command in commands)
13{
14 string? special = null;
15 if (command == "undo")
16 {
17 if (undo.Count == 0)
18 {
19 special = "nothing to undo";
20 }
21 else
22 {
23 redo.Push(buffer);
24 buffer = undo.Pop();
25 }
26 }
27 else if (command == "redo")
28 {
29 if (redo.Count == 0)
30 {
31 special = "nothing to redo";
32 }
33 else
34 {
35 undo.Push(buffer);
36 buffer = redo.Pop();
37 }
38 }
39 else
40 {
41 undo.Push(buffer);
42 redo.Clear();
43 if (command.StartsWith("type ", StringComparison.Ordinal))
44 buffer += command[5..];
45 else
46 buffer = buffer[..Math.Max(0, buffer.Length - int.Parse(command[7..], CultureInfo.InvariantCulture))];
47 }
48 Console.WriteLine(command + " => " + (special ?? "\"" + buffer + "\""));
49}
