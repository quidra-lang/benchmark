paths = lambda do |blocked|
  dp = Array.new(grid[0].length, 0)
  dp[0] = 1
  grid.each do |row|
    row.each_char.with_index do |cell, column|
      if blocked && cell == '#'
        dp[column] = 0
      elsif column.positive?
        dp[column] += dp[column - 1]
      end
    end
  end
end
puts "without obstacles: #{paths.call(false)}"
puts "with obstacles: #{paths.call(true)}"
