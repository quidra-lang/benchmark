    for (0..2) |mode| {
        var dp: [6][7]u64 = @splat(@splat(0));
        for (0..6) |r| {
            for (0..7) |c| {
                if (mode == 1 and grid[r][c] == '#')
                    continue;
                if (r == 0 and c == 0) {
                    dp[r][c] = 1;
                    continue;
                }
                if (r > 0)
                    dp[r][c] += dp[r - 1][c];
                if (c > 0)
                    dp[r][c] += dp[r][c - 1];
            }
        }
    }
