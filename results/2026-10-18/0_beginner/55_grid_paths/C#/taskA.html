1using System;
2using System.Globalization;
3
4CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
5
6string[] grid = { ".......", "..#....", "....#..", ".#.....", "...#.#.", "......." };
7foreach (bool obstacles in new[] { false, true })
8{
9 int[,] dp = new int[6, 7];
10 dp[0, 0] = 1;
11 for (int r = 0; r < 6; r++)
12 {
13 for (int c = 0; c < 7; c++)
14 {
15 if (obstacles && grid[r][c] == '#')
16 {
17 dp[r, c] = 0;
18 continue;
19 }
20 if (r > 0) dp[r, c] += dp[r - 1, c];
21 if (c > 0) dp[r, c] += dp[r, c - 1];
22 }
23 }
24 Console.WriteLine($"{(obstacles ? "with" : "without")} obstacles: {dp[5, 6]}");
25}
