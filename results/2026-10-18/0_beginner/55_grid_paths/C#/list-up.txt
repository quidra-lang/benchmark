foreach (bool obstacles in new[] { false, true })
{
    int[,] dp = new int[6, 7];
    dp[0, 0] = 1;
    for (int r = 0; r < 6; r++)
    {
        for (int c = 0; c < 7; c++)
        {
            if (obstacles && grid[r][c] == '#')
            {
                dp[r, c] = 0;
                continue;
            }
            if (r > 0) dp[r, c] += dp[r - 1, c];
            if (c > 0) dp[r, c] += dp[r, c - 1];
        }
    }
}
