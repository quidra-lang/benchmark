1package main
2
3import (
4 "fmt"
5 "strconv"
6 "strings"
7)
8
9func main() {
10 expressions := []string{"3 4 + 2 *", "-17 5 / 4 -3 * -", "1 2 + * 3", "15 7 1 1 + - / 3 * 2 1 1 + + -",
11 "4 2 5 * + 1 3 2 * - 5 + /", "2 3 4 + 5", "46341 46339 * -2 /"}
12 for _, expression := range expressions {
13 stack := []int64{}
14 message := ""
15 for _, token := range strings.Fields(expression) {
16 if len(token) == 1 && strings.Contains("+-*/", token) {
17 if len(stack) < 2 {
18 message = "stack underflow"
19 break
20 }
21 b, a := stack[len(stack)-1], stack[len(stack)-2]
22 stack = stack[:len(stack)-2]
23 if token == "/" && b == 0 {
24 message = "division by zero"
25 break
26 }
27 value := int64(0)
28 switch token {
29 case "+":
30 value = a + b
31 case "-":
32 value = a - b
33 case "*":
34 value = a * b
35 case "/":
36 value = a / b
37 }
38 stack = append(stack, value)
39 } else {
40 value, _ := strconv.ParseInt(token, 10, 64)
41 stack = append(stack, value)
42 }
43 }
44 if message == "" && len(stack) != 1 {
45 message = "leftover operands"
46 }
47 if message != "" {
48 fmt.Printf("%s => error: %s\n", expression, message)
49 } else {
50 fmt.Printf("%s => %d\n", expression, stack[0])
51 }
52 }
53}
