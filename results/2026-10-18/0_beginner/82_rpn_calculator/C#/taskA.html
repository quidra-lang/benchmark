1using System;
2using System.Collections.Generic;
3using System.Globalization;
4
5CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
6
7string[] expressions = { "3 4 + 2 *", "-17 5 / 4 -3 * -", "1 2 + * 3",
8 "15 7 1 1 + - / 3 * 2 1 1 + + -", "4 2 5 * + 1 3 2 * - 5 + /",
9 "2 3 4 + 5", "46341 46339 * -2 /" };
10foreach (string expression in expressions)
11{
12 var stack = new Stack<int>();
13 string? error = null;
14 foreach (string token in expression.Split(' '))
15 {
16 if (token.Length == 1 && "+-*/".Contains(token, StringComparison.Ordinal))
17 {
18 if (stack.Count < 2)
19 {
20 error = "stack underflow";
21 break;
22 }
23 int b = stack.Pop(), a = stack.Pop();
24 if (token == "/" && b == 0)
25 {
26 error = "division by zero";
27 break;
28 }
29 int value = token switch
30 {
31 "+" => a + b,
32 "-" => a - b,
33 "*" => a * b,
34 _ => a / b
35 };
36 stack.Push(value);
37 }
38 else
39 {
40 stack.Push(int.Parse(token, CultureInfo.InvariantCulture));
41 }
42 }
43 if (error is null && stack.Count != 1) error = "leftover operands";
44 string result = error is null ? stack.Peek().ToString(CultureInfo.InvariantCulture) : "error: " + error;
45 Console.WriteLine($"{expression} => {result}");
46}
