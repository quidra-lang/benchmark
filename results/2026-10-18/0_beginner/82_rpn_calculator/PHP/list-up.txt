function evaluate(string $expression): string
{
    $stack = [];
    foreach (explode(' ', $expression) as $token) {
        if (in_array($token, ['+', '-', '*', '/'], true)) {
            if (count($stack) < 2) {
                return 'error: stack underflow';
            }
            $b = array_pop($stack);
            $a = array_pop($stack);
            $stack[] = match ($token) {
                '+' => $a + $b,
                '-' => $a - $b,
                '*' => $a * $b,
                '/' => intdiv($a, $b),
            };
        } else {
            $stack[] = (int) $token;
        }
    }
}
