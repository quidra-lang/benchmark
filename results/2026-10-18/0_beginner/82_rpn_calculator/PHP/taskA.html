11 $b = array_pop($stack);
12 $a = array_pop($stack);
13 if ($token === '/' && $b === 0) {

16 $stack[] = match ($token) {
17 '+' => $a + $b,
18 '-' => $a - $b,
19 '*' => $a * $b,
20 '/' => intdiv($a, $b),

22 } else {

24 }
25 }
