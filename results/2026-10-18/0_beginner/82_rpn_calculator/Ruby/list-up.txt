expressions = [
  '3 4 + 2 *', '-17 5 / 4 -3 * -', '1 2 + * 3',
]
expressions.each do |expression|
  stack = []
  expression.split.each do |token|
    unless %w[+ - * /].include?(token)
      stack << token.to_i
      next
    end
    if stack.length < 2
      error = 'stack underflow'
      break
    end
    b, a = stack.pop, stack.pop
    result = case token
             when '+' then a + b
             when '-' then a - b
             when '*' then a * b
             else
               a.quo(b).truncate
             end
    stack << result
