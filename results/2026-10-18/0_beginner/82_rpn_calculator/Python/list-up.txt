expressions = [
    "-17 5 / 4 -3 * -",
    "1 2 + * 3",
]
for expression in expressions:
    stack = []
    for token in expression.split():
        if token not in {"+", "-", "*", "/"}:
            stack.append(int(token))
            continue
        if len(stack) < 2:
            error = "stack underflow"
            break
        b, a = stack.pop(), stack.pop()
        if token == "+":
            result = a + b
        elif token == "-":
            result = a - b
        elif token == "*":
            result = a * b
        else:
            result = int(a / b)
        stack.append(result)
