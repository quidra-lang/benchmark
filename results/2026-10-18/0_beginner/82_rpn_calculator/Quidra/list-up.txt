string[] expressions = [
    "3 4 + 2 *", "-17 5 / 4 -3 * -", "1 2 + * 3", "15 7 1 1 + - / 3 * 2 1 1 + + -",
]
for string expression = expressions[]
    int[] stack = []
    string[] tokens = expression.split(" ")
    for string token = tokens[]
        if len(token) == 1 and "+-*/".contains(token)
            if len(stack) < 2
                failure = "stack underflow"
                break
            int b = stack[len(stack) - 1]
            int a = stack[len(stack) - 2]
            stack = stack[:<len(stack) - 2]
            int value = 0
            if token == "+"
                value = a + b
            elif token == "-"
                value = a - b
            elif token == "*"
                value = a * b
            else
                value = a / b
            stack = stack.append(value)
        else
            int value = int.parse(token)
            stack = stack.append(value)
