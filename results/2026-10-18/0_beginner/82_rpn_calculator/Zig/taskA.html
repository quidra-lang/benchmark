1const std = @import("std");
2
3pub fn main(init: std.process.Init) !void {
4 var output_buffer: [8192]u8 = undefined;
5 var writer = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
6 const out = &writer.interface;
7 const expressions = [_][]const u8{
8 "3 4 + 2 *",
9 "-17 5 / 4 -3 * -",
10 "1 2 + * 3",
11 "15 7 1 1 + - / 3 * 2 1 1 + + -",
12 "4 2 5 * + 1 3 2 * - 5 + /",
13 "2 3 4 + 5",
14 "46341 46339 * -2 /",
15 };
16 for (expressions) |expression| {
17 var stack: [32]i64 = undefined;
18 var len: usize = 0;
19 var err: ?[]const u8 = null;
20 var tokens = std.mem.splitScalar(u8, expression, ' ');
21 while (tokens.next()) |token| {
22 const operator = token.len == 1 and
23 (token[0] == '+' or token[0] == '-' or token[0] == '*' or token[0] == '/');
24 if (!operator) {
25 stack[len] = try std.fmt.parseInt(i64, token, 10);
26 len += 1;
27 continue;
28 }
29 if (len < 2) {
30 err = "stack underflow";
31 break;
32 }
33 const b = stack[len - 1];
34 const a = stack[len - 2];
35 len -= 2;
36 if (token[0] == '/' and b == 0) {
37 err = "division by zero";
38 break;
39 }
40 stack[len] = switch (token[0]) {
41 '+' => a + b,
42 '-' => a - b,
43 '*' => a * b,
44 '/' => @divTrunc(a, b),
45 else => unreachable,
46 };
47 len += 1;
48 }
49 if (err == null and len != 1)
50 err = "leftover operands";
51 try out.print("{s} => ", .{expression});
52 if (err) |message| {
53 try out.print("error: {s}\n", .{message});
54 } else {
55 try out.print("{d}\n", .{stack[0]});
56 }
57 }
58 try out.flush();
59}
