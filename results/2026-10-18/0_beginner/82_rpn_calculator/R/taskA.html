1evaluate <- function(expression) {
2 stack <- numeric(0)
3 for (token in strsplit(expression, " ", fixed = TRUE)[[1]]) {
4 if (token %in% c("+", "-", "*", "/")) {
5 if (length(stack) < 2) {
6 return("error: stack underflow")
7 }
8 b <- tail(stack, 1)
9 a <- stack[length(stack) - 1]
10 stack <- head(stack, -2)
11 if (token == "/" && b == 0) {
12 return("error: division by zero")
13 }
14 value <- switch(token,
15 "+" = a + b,
16 "-" = a - b,
17 "*" = a * b,
18 "/" = trunc(a / b)
19 )
20 stack <- c(stack, value)
21 } else {
22 stack <- c(stack, as.numeric(token))
23 }
24 }
25 if (length(stack) == 1) sprintf("%d", stack[1]) else "error: leftover operands"
26}
27
28expressions <- c(
29 "3 4 + 2 *", "-17 5 / 4 -3 * -", "1 2 + * 3", "15 7 1 1 + - / 3 * 2 1 1 + + -",
30 "4 2 5 * + 1 3 2 * - 5 + /", "2 3 4 + 5", "46341 46339 * -2 /"
31)
32for (expression in expressions) cat(sprintf("%s => %s\n", expression, evaluate(expression)))
