function evaluate(expression)
    stack = Int[]
    for token in split(expression)
        if token in ("+", "-", "*", "/")
            length(stack) < 2 && return "error: stack underflow"
            b = pop!(stack)
            a = pop!(stack)
            if token == "+"
                value = a+b
            elseif token == "-"
                value = a-b
            elseif token == "*"
                value = a*b
            else
                value = div(a, b)
            end
            push!(stack, value)
        else
            push!(stack, parse(Int, token))
        end
    end
end
for expression in [
    "-17 5 / 4 -3 * -",
    "1 2 + * 3",
]
    println(expression, " => ", evaluate(expression))
