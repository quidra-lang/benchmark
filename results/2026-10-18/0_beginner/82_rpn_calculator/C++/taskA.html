9 "15 7 1 1 + - / 3 * 2 1 1 + + -", "4 2 5 * + 1 3 2 * - 5 + /", "2 3 4 + 5", "46341 46339 * -2 /"};
10 for (const std::string &expression : expressions) {
11 std::istringstream input(expression);
12 std::vector<long long> stack;
13 std::string error;
14 for (std::string token; input >> token;) {
15 if (token.size() == 1 && std::string("+-*/").find(token[0]) != std::string::npos) {
16 if (stack.size() < 2) {
17 error = "stack underflow";
18 break;
19 }
20 long long b = stack.back();
21 stack.pop_back();
22 long long a = stack.back();
23 stack.pop_back();
24 if (token == "/" && b == 0) {
25 error = "division by zero";
26 break;
27 }
28 stack.push_back(token == "+" ? a + b : token == "-" ? a - b : token == "*" ? a * b : a / b);
29 } else {
30 stack.push_back(std::stoll(token));
31 }
32 }
33 if (error.empty() && stack.size() != 1)
34 error = "leftover operands";
35 std::cout << expression << " => ";
36 if (error.empty())
37 std::cout << stack[0] << '\n';
38 else
39 std::cout << "error: " << error << '\n';
40 }
