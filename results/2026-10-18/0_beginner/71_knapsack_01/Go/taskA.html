3import "fmt"

5func main() {
6 names := []string{"tent", "stove", "lantern", "rope", "camera", "blanket", "water"}
7 weights, values := []int{10, 3, 2, 3, 5, 5, 6}, []int{120, 65, 45, 20, 90, 85, 100}
8 dp := [8][21]int{}
9 for i := 1; i <= 7; i++ {
10 for w := 0; w <= 20; w++ {
11 dp[i][w] = dp[i-1][w]
12 if weights[i-1] <= w {
13 dp[i][w] = max(dp[i][w], dp[i-1][w-weights[i-1]]+values[i-1])
14 }
15 }
16 }
17 chosen := [7]bool{}
18 capacity, totalWeight := 20, 0
19 for i := 7; i >= 1; i-- {
20 if dp[i][capacity] != dp[i-1][capacity] {
21 chosen[i-1] = true
22 capacity -= weights[i-1]
23 totalWeight += weights[i-1]
24 }
25 }
26 fmt.Printf("max value: %d\ntotal weight: %d\nitems:", dp[7][20], totalWeight)
27 for i, name := range names {
28 if chosen[i] {
29 fmt.Printf(" %s", name)
30 }
31 }

33}
