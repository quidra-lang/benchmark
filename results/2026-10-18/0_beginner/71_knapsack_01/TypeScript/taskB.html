1const capacity = 20;
2const dp = Array.from({length: items.length + 1}, () => Array<number>(capacity + 1).fill(0));
3items.forEach(({weight, value}, index) => {
4 const i = index + 1;
5 for (let limit = 0; limit <= capacity; limit++) {
6 dp[i][limit] = dp[i - 1][limit];
7 if (weight <= limit) dp[i][limit] = Math.max(dp[i][limit], dp[i - 1][limit - weight] + value);
8 }
9});
10const chosen: typeof items = [];
11let remaining = capacity;
12for (let i = items.length; i > 0; i--) {

14 chosen.push(items[i - 1]);
15 remaining -= items[i - 1].weight;
16 }
17}
18chosen.reverse();
