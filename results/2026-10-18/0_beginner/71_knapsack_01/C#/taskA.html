1using System;

4
5CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;

7string[] names = { "tent", "stove", "lantern", "rope", "camera", "blanket", "water" };
8int[] weights = { 10, 3, 2, 3, 5, 5, 6 }, values = { 120, 65, 45, 20, 90, 85, 100 };
9int[,] dp = new int[8, 21];
10for (int i = 1; i <= 7; i++)
11{
12 for (int capacity = 0; capacity <= 20; capacity++)
13 {
14 dp[i, capacity] = dp[i - 1, capacity];
15 if (weights[i - 1] <= capacity)
16 {
17 dp[i, capacity] = Math.Max(dp[i, capacity], dp[i - 1, capacity - weights[i - 1]] + values[i - 1]);
18 }
19 }
20}
21bool[] chosen = new bool[7];
22int remaining = 20;
23for (int i = 7; i >= 1; i--)
24{
25 if (dp[i, remaining] != dp[i - 1, remaining])
26 {
27 chosen[i - 1] = true;
28 remaining -= weights[i - 1];
29 }
30}
31var items = names.Where((_, i) => chosen[i]);
32Console.WriteLine($"max value: {dp[7, 20]}");
33Console.WriteLine($"total weight: {20 - remaining}");
34Console.WriteLine($"items: {string.Join(" ", items)}");
