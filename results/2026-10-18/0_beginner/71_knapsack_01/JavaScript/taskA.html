1const items = [
2 {name: "tent", weight: 10, value: 120}, {name: "stove", weight: 3, value: 65},
3 {name: "lantern", weight: 2, value: 45}, {name: "rope", weight: 3, value: 20},
4 {name: "camera", weight: 5, value: 90}, {name: "blanket", weight: 5, value: 85},
5 {name: "water", weight: 6, value: 100}
6];
7const capacity = 20;
8const dp = Array.from({length: items.length + 1}, () => Array(capacity + 1).fill(0));
9items.forEach(({weight, value}, index) => {
10 const i = index + 1;
11 for (let limit = 0; limit <= capacity; limit++) {
12 dp[i][limit] = dp[i - 1][limit];
13 if (weight <= limit) dp[i][limit] = Math.max(dp[i][limit], dp[i - 1][limit - weight] + value);
14 }
15});
16const chosen = [];
17let remaining = capacity;
18for (let i = items.length; i > 0; i--) {
19 if (dp[i][remaining] !== dp[i - 1][remaining]) {
20 chosen.push(items[i - 1]);
21 remaining -= items[i - 1].weight;
22 }
23}
24chosen.reverse();
25console.log(`max value: ${dp[items.length][capacity]}`);
