1import java.util.Locale
2
3data class Point(val x: Int, val y: Int)
4
5fun cross(o: Point, a: Point, b: Point): Int =
6 (a.x - o.x) * (b.y - o.y) - (a.y - o.y) * (b.x - o.x)
7
8fun chain(points: List<Point>): MutableList<Point> {
9 val result = mutableListOf<Point>()
10 for (point in points) {
11 while (result.size >= 2 && cross(result[result.size - 2], result.last(), point) <= 0) {
12 result.removeLast()
13 }
14 result.add(point)
15 }
16 return result
17}

19fun main() {
20 val points = listOf(
21 Point(3, 2), Point(-6, 1), Point(8, 4), Point(0, 0), Point(-3, 7),
22 Point(4, -3), Point(1, -5), Point(-2, 1), Point(5, 6), Point(-6, -2),
23 Point(2, -2), Point(7, -1), Point(-3, 4), Point(2, 8), Point(-6, 3),
24 ).sortedWith(compareBy<Point> { it.x }.thenBy { it.y })
25 val lower = chain(points)
26 val upper = chain(points.reversed())
27 lower.removeLast()
28 upper.removeLast()
29 lower.addAll(upper)
30 var crossSum = 0
31 println("hull: ${lower.size} points")
32 for (i in lower.indices) {
33 val a = lower[i]
34 val b = lower[(i + 1) % lower.size]
35 println("(${a.x}, ${a.y})")
36 crossSum += a.x * b.y - b.x * a.y
37 }
