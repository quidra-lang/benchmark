3int cross(int[] o, int[] a, int[] b)
4 return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

6int[][] chain(int[][] points)
7 int[][] result := []
8 for int[] point := points[]
9 while len(result) >= 2 and cross(result[len(result) - 2], result[len(result) - 1], point) <= 0
10 .result.take()
11 .result.push(point)
12 return result
