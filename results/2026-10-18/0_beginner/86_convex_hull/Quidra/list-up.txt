import collections
int cross(int[] o, int[] a, int[] b)
    return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])
int[][] chain(int[][] points)
    int[][] result := []
    for point := points[]
        while len(result) >= 2 and cross(result[len(result) - 2], result[len(result) - 1], point) <= 0
            .result.take()
        .result.push(point)
    return result
int[][] points := [
    [3, 2], [-6, 1], [8, 4], [0, 0], [-3, 7], [4, -3], [1, -5], [-2, 1], [5, 6], [-6, -2], [2, -2],
    [7, -1], [-3, 4], [2, 8], [-6, 3],
points = points[collections.lex_order(points)]
lower := chain(points)
upper := chain(points[::-1])
.lower.take()
.upper.take()
hull := lower.concat(upper)
