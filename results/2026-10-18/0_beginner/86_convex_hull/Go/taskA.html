4 "fmt"
5 "sort"

8func cross(o, a, b [2]int) int {
9 return (a[0]-o[0])*(b[1]-o[1]) - (a[1]-o[1])*(b[0]-o[0])
10}
11
12func main() {
13 points := [][2]int{{3, 2}, {-6, 1}, {8, 4}, {0, 0}, {-3, 7}, {4, -3}, {1, -5}, {-2, 1}, {5, 6},
14 {-6, -2}, {2, -2}, {7, -1}, {-3, 4}, {2, 8}, {-6, 3}}
15 sort.Slice(points, func(i, j int) bool {
16 if points[i][0] != points[j][0] {
17 return points[i][0] < points[j][0]
18 }
19 return points[i][1] < points[j][1]
20 })
21 lower, upper := [][2]int{}, [][2]int{}
22 for _, p := range points {
23 for len(lower) >= 2 && cross(lower[len(lower)-2], lower[len(lower)-1], p) <= 0 {
24 lower = lower[:len(lower)-1]
25 }
26 lower = append(lower, p)
27 }
28 for i := len(points) - 1; i >= 0; i-- {
29 p := points[i]
30 for len(upper) >= 2 && cross(upper[len(upper)-2], upper[len(upper)-1], p) <= 0 {
31 upper = upper[:len(upper)-1]
32 }
33 upper = append(upper, p)
34 }
35 hull := append(lower[:len(lower)-1], upper[:len(upper)-1]...)
36 twice := 0
37 fmt.Printf("hull: %d points\n", len(hull))
38 for i, p := range hull {
39 q := hull[(i+1)%len(hull)]
40 fmt.Printf("(%d, %d)\n", p[0], p[1])
41 twice += p[0]*q[1] - q[0]*p[1]
42 }
43 fmt.Printf("area: %d.%d\n", twice/2, twice%2*5)
44}
