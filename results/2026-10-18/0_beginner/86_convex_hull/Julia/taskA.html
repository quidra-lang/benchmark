3cross_product(o, a, b) = (a[1]-o[1])*(b[2]-o[2])-(a[2]-o[2])*(b[1]-o[1])
4
5function chain(points)
6 result = Tuple{Int,Int}[]
7 for point in points
8 while length(result) >= 2 && cross_product(result[end-1], result[end], point) <= 0
9 pop!(result)
10 end
11 push!(result, point)
12 end
13 return result
14end
