1#include <stdio.h>
2#include <stdlib.h>
3
4typedef struct {
5 int x, y;
6} Point;
7
8int compare(const void *a, const void *b) {
9 const Point *x = a, *y = b;
10 return x->x != y->x ? x->x - y->x : x->y - y->y;
11}
12
13int cross(Point o, Point a, Point b) {
14 return (a.x - o.x) * (b.y - o.y) - (a.y - o.y) * (b.x - o.x);
15}
16
17int main(void) {
18 Point points[] = {{3, 2}, {-6, 1}, {8, 4}, {0, 0}, {-3, 7}, {4, -3}, {1, -5}, {-2, 1}, {5, 6},
19 {-6, -2}, {2, -2}, {7, -1}, {-3, 4}, {2, 8}, {-6, 3}};
20 qsort(points, 15, sizeof(Point), compare);
21 Point lower[15], upper[15], hull[30];
22 int nl = 0, nu = 0, nh = 0;
23 for (int i = 0; i < 15; i++) {
24 while (nl >= 2 && cross(lower[nl - 2], lower[nl - 1], points[i]) <= 0)
25 nl--;
26 lower[nl++] = points[i];
27 }
28 for (int i = 14; i >= 0; i--) {
29 while (nu >= 2 && cross(upper[nu - 2], upper[nu - 1], points[i]) <= 0)
30 nu--;
31 upper[nu++] = points[i];
32 }
33 for (int i = 0; i < nl - 1; i++)
34 hull[nh++] = lower[i];
35 for (int i = 0; i < nu - 1; i++)
36 hull[nh++] = upper[i];
37 int twice = 0;
38 printf("hull: %d points\n", nh);
39 for (int i = 0; i < nh; i++) {
40 Point p = hull[i], q = hull[(i + 1) % nh];
41 printf("(%d, %d)\n", p.x, p.y);
42 twice += p.x * q.y - q.x * p.y;
43 }
44 printf("area: %d.%d\n", twice / 2, twice % 2 * 5);
45 return 0;
46}
