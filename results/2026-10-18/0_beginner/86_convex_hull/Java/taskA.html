1import java.util.ArrayList;
2import java.util.Comparator;
3import java.util.List;
4import java.util.Locale;
5
6public class Main {
7 record Point(int x, int y) {}
8
9 static int cross(Point o, Point a, Point b) {
10 return (a.x() - o.x()) * (b.y() - o.y()) - (a.y() - o.y()) * (b.x() - o.x());
11 }
12
13 static List<Point> chain(List<Point> points) {
14 List<Point> result = new ArrayList<>();
15 for (Point point : points) {
16 while (result.size() >= 2 && cross(result.get(result.size() - 2), result.getLast(), point) <= 0) {
17 result.removeLast();
18 }
19 result.add(point);
20 }
21 return result;
22 }
23
24 public static void main(String[] args) {
25 List<Point> points = new ArrayList<>(List.of(
26 new Point(3, 2), new Point(-6, 1), new Point(8, 4), new Point(0, 0), new Point(-3, 7),
27 new Point(4, -3), new Point(1, -5), new Point(-2, 1), new Point(5, 6), new Point(-6, -2),
28 new Point(2, -2), new Point(7, -1), new Point(-3, 4), new Point(2, 8), new Point(-6, 3)));
29 points.sort(Comparator.comparingInt(Point::x).thenComparingInt(Point::y));
30 List<Point> lower = chain(points);
31 List<Point> upper = chain(points.reversed());
32 lower.removeLast();
33 upper.removeLast();
34 lower.addAll(upper);
35 int crossSum = 0;
36 System.out.println("hull: " + lower.size() + " points");
37 for (int i = 0; i < lower.size(); i++) {
38 Point a = lower.get(i), b = lower.get((i + 1) % lower.size());
39 System.out.println("(" + a.x() + ", " + a.y() + ")");
40 crossSum += a.x() * b.y() - b.x() * a.y();
41 }
42 System.out.printf(Locale.ROOT, "area: %.1f%n", crossSum / 2.0);
43 }
44}
