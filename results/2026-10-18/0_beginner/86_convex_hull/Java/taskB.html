1 record Point(int x, int y) {}
2 static int cross(Point o, Point a, Point b) {
3 return (a.x() - o.x()) * (b.y() - o.y()) - (a.y() - o.y()) * (b.x() - o.x());
4 }
5 static List<Point> chain(List<Point> points) {
6 List<Point> result = new ArrayList<>();
7 for (Point point : points) {
8 while (result.size() >= 2 && cross(result.get(result.size() - 2), result.getLast(), point) <= 0) {
9 result.removeLast();
10 }
11 result.add(point);
12 }
13 return result;
14 }
