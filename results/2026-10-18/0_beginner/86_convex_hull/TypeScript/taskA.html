6function cross(o: number[], a: number[], b: number[]): number {
7 return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0]);
8}

10function chain(ordered: number[][]): number[][] {
11 const result: number[][] = [];
12 for (const point of ordered) {
13 while (result.length >= 2 && cross(result.at(-2)!, result.at(-1)!, point) <= 0) result.pop();
14 result.push(point);
15 }
16 return result;
17}
