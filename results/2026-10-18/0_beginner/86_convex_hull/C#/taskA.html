8static int Cross((int x, int y) o, (int x, int y) a, (int x, int y) b)
9{
10 return (a.x - o.x) * (b.y - o.y) - (a.y - o.y) * (b.x - o.x);
11}
12
13static List<(int x, int y)> Chain(IEnumerable<(int x, int y)> points)
14{
15 var result = new List<(int x, int y)>();
16 foreach (var point in points)
17 {
18 while (result.Count >= 2 && Cross(result[^2], result[^1], point) <= 0)
19 {
20 result.RemoveAt(result.Count - 1);
21 }
22 result.Add(point);
23 }
24 return result;
25}

27(int x, int y)[] input =
28{
29 (3, 2), (-6, 1), (8, 4), (0, 0), (-3, 7), (4, -3), (1, -5), (-2, 1), (5, 6), (-6, -2), (2, -2),
30 (7, -1), (-3, 4), (2, 8), (-6, 3),
31};
32var points = input.OrderBy(p => p.x).ThenBy(p => p.y).ToList();
33var lower = Chain(points);
34var upper = Chain(points.AsEnumerable().Reverse());
35lower.RemoveAt(lower.Count - 1);
36upper.RemoveAt(upper.Count - 1);
37lower.AddRange(upper);
38int crossSum = 0;
39Console.WriteLine($"hull: {lower.Count} points");
40for (int i = 0; i < lower.Count; i++)
41{
42 var a = lower[i];
43 var b = lower[(i + 1) % lower.Count];
44 Console.WriteLine($"({a.x}, {a.y})");
45 crossSum += a.x * b.y - b.x * a.y;
46}
