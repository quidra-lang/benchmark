1static int Cross((int x, int y) o, (int x, int y) a, (int x, int y) b)
2{
3 return (a.x - o.x) * (b.y - o.y) - (a.y - o.y) * (b.x - o.x);
4}
5static List<(int x, int y)> Chain(IEnumerable<(int x, int y)> points)
6{
7 var result = new List<(int x, int y)>();
8 foreach (var point in points)
9 {
10 while (result.Count >= 2 && Cross(result[^2], result[^1], point) <= 0)
11 {
12 result.RemoveAt(result.Count - 1);
13 }
14 result.Add(point);
15 }
16 return result;
17}
