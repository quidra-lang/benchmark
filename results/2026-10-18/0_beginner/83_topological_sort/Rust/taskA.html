1fn solve(names: &[&str], edges: &[(&str, &str)], show_cycle: bool) {
2 let n = names.len();
3 let mut graph = vec![vec![false; n]; n];
4 let mut indegree = vec![0; n];
5 let mut taken = vec![false; n];
6 for (a, b) in edges {
7 let a = names.iter().position(|name| name == a).unwrap();
8 let b = names.iter().position(|name| name == b).unwrap();
9 graph[a][b] = true;
10 indegree[b] += 1;
11 }
12 let mut order = Vec::new();
13 loop {
14 let best = (0..n)
15 .filter(|&v| !taken[v] && indegree[v] == 0)
16 .min_by_key(|&v| names[v]);
17 let Some(v) = best else {
18 break;
19 };
20 order.push(names[v]);
21 taken[v] = true;
22 for w in 0..n {
23 if graph[v][w] {
24 indegree[w] -= 1;
25 }
26 }
27 }
28 println!("order: {}", order.join(" -> "));
29 if show_cycle {
30 let mut remaining: Vec<&str> = (0..n).filter(|&v| !taken[v]).map(|v| names[v]).collect();
31 remaining.sort();
32 println!("cycle: {}", remaining.join(" "));
33 }
34}
