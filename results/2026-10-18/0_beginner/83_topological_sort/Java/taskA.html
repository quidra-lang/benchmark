1import java.util.ArrayList;
2import java.util.Arrays;
3import java.util.List;
4import java.util.stream.IntStream;

6public class Main {
7 static void courses(String[] original, String[][] edges, boolean showCycle) {
8 String[] names = Arrays.stream(original).sorted().toArray(String[]::new);
9 int n = names.length;
10 int[] degree = new int[n];
11 boolean[][] graph = new boolean[n][n];
12 for (String[] edge : edges) {
13 int a = Arrays.asList(names).indexOf(edge[0]), b = Arrays.asList(names).indexOf(edge[1]);
14 graph[a][b] = true;
15 degree[b]++;
16 }
17 boolean[] taken = new boolean[n];
18 List<String> order = new ArrayList<>();
19 while (true) {
20 int next = IntStream.range(0, n).filter(i -> !taken[i] && degree[i] == 0).findFirst().orElse(-1);
21 if (next < 0) {
22 break;
23 }
24 taken[next] = true;
25 order.add(names[next]);
26 for (int i = 0; i < n; i++) {
27 if (graph[next][i]) {
28 degree[i]--;
29 }
30 }
31 }
32 System.out.println("order: " + String.join(" -> ", order));
33 if (showCycle) {
34 var remaining = IntStream.range(0, n).filter(i -> !taken[i]).mapToObj(i -> names[i]).toList();
35 System.out.println("cycle: " + String.join(" ", remaining));
36 }
37 }
