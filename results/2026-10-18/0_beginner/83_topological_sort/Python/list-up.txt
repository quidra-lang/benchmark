    ready = sorted(course for course in courses if indegree[course] == 0)
    while ready:
        course = ready.pop(0)
        order.append(course)
        for target in graph[course]:
            indegree[target] -= 1
            if indegree[target] == 0:
                ready.append(target)
        ready.sort()
    return order, sorted(set(courses) - set(order))
