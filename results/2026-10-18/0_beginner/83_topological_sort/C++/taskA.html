7void solve(const std::vector<std::string> &names, const std::vector<std::array<std::string, 2>> &edges,
8 bool show_cycle) {
9 int n = static_cast<int>(names.size());
10 std::vector<std::vector<bool>> graph(n, std::vector<bool>(n));
11 std::vector<int> indegree(n);
12 std::vector<bool> taken(n);
13 for (const std::array<std::string, 2> &e : edges) {
14 int a = static_cast<int>(std::find(names.begin(), names.end(), e[0]) - names.begin());
15 int b = static_cast<int>(std::find(names.begin(), names.end(), e[1]) - names.begin());
16 graph[a][b] = true;
17 ++indegree[b];
18 }
19 std::cout << "order: ";
20 int count = 0;
21 while (true) {
22 int best = -1;
23 for (int v = 0; v < n; v++)
24 if (!taken[v] && !indegree[v] && (best < 0 || names[v] < names[best]))
25 best = v;
26 if (best < 0)
27 break;
28 if (count++)
29 std::cout << " -> ";
30 std::cout << names[best];
31 taken[best] = true;
32 for (int v = 0; v < n; v++)
33 if (graph[best][v])
34 --indegree[v];
35 }
36 std::cout << '\n';
37 if (show_cycle) {
38 std::vector<std::string> remaining;
39 for (int v = 0; v < n; v++)
40 if (!taken[v])
41 remaining.push_back(names[v]);
42 std::sort(remaining.begin(), remaining.end());
43 std::cout << "cycle:";
44 for (const std::string &name : remaining)
45 std::cout << ' ' << name;
46 std::cout << '\n';
47 }
48}
