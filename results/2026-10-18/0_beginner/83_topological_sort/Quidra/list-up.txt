    ordered := names.sorted()
    n := len(ordered)
    while true
        selected := n
        for nat i < n
            if not taken[i] and indegree[i] == 0
                selected = i
                break
        if selected == n
            break
        taken[selected] = true
        for target := graph.outgoing(selected)[]
            indegree[target] -= 1
    if show_cycle
        remaining := [for i := ordered.indices()[] if not taken[i] then ordered[i]]
        println("cycle: {remaining.join(" ")}")
