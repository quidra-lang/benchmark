1topological <- function(courses, edges) {
2 indegree <- setNames(rep(0, length(courses)), courses)
3 adjacency <- setNames(replicate(length(courses), character(0), simplify = FALSE), courses)
4 for (edge in edges) {
5 adjacency[[edge[1]]] <- c(adjacency[[edge[1]]], edge[2])
6 indegree[edge[2]] <- indegree[edge[2]] + 1
7 }
8 ordered <- character(0)
9 alphabetical <- sort(courses, method = "radix")
10 repeat {
11 ready <- alphabetical[!alphabetical %in% ordered & indegree[alphabetical] == 0]
12 if (length(ready) == 0) break
13 course <- ready[1]
14 ordered <- c(ordered, course)
15 for (target in adjacency[[course]]) indegree[target] <- indegree[target] - 1
16 }
17 list(order = ordered, remaining = alphabetical[!alphabetical %in% ordered])
18}
19
20courses1 <- c("physics", "algebra", "graphics", "programming", "statistics", "calculus", "databases", "algorithms")
21edges1 <- list(
22 c("calculus", "physics"), c("algebra", "calculus"), c("programming", "algorithms"),
23 c("algebra", "statistics"), c("algorithms", "databases"), c("statistics", "databases"),
24 c("calculus", "graphics"), c("algorithms", "graphics"), c("physics", "graphics")
25)
26courses2 <- c("music", "ecology", "art", "geology", "biology", "chemistry")
27edges2 <- list(
28 c("music", "art"), c("art", "ecology"), c("biology", "chemistry"), c("chemistry", "ecology"),
29 c("ecology", "geology"), c("geology", "chemistry")
30)
31first <- topological(courses1, edges1)
32second <- topological(courses2, edges2)
33cat("order: ", paste(first$order, collapse = " -> "), "\n", sep = "")
34cat("order: ", paste(second$order, collapse = " -> "), "\n", sep = "")
35cat("cycle: ", paste(second$remaining, collapse = " "), "\n", sep = "")
