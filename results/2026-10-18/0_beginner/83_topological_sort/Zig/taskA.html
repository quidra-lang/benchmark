3fn topological(
4 out: *std.Io.Writer,
5 names: []const []const u8,
6 edges: []const [2]usize,
7 cycle: bool,
8) std.Io.Writer.Error!void {
9 var indegree: [8]usize = @splat(0);
10 var taken: [8]bool = @splat(false);
11 for (edges) |e|
12 indegree[e[1]] += 1;
13 try out.writeAll("order: ");
14 var count: usize = 0;
15 while (true) {
16 var best: ?usize = null;
17 for (names, 0..) |name, i| {
18 if (!taken[i] and indegree[i] == 0) {
19 if (best == null or std.mem.order(u8, name, names[best.?]) == .lt)
20 best = i;
21 }
22 }
23 const v = best orelse break;
24 if (count > 0)
25 try out.writeAll(" -> ");
26 try out.writeAll(names[v]);
27 count += 1;
28 taken[v] = true;
29 for (edges) |e| {
30 if (e[0] == v)
31 indegree[e[1]] -= 1;
32 }
33 }
34 try out.writeByte('\n');
35 if (cycle) {
36 try out.writeAll("cycle:");
37 var remaining: [8][]const u8 = undefined;
38 var len: usize = 0;
39 for (names, 0..) |name, i| {
40 if (!taken[i]) {
41 remaining[len] = name;
42 len += 1;
43 }
44 }
45 std.mem.sort([]const u8, remaining[0..len], {}, nameLess);
46 for (remaining[0..len]) |name|
47 try out.print(" {s}", .{name});
48 try out.writeByte('\n');
49 }
50}
