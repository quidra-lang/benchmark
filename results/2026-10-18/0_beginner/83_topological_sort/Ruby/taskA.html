1def kahn(courses, pairs)
2 graph = courses.to_h { |course| [course, []] }
3 indegree = courses.to_h { |course| [course, 0] }
4 pairs.each do |source, target|
5 graph[source] << target
6 indegree[target] += 1
7 end
8 ready = courses.select { |course| indegree[course].zero? }.sort
9 order = []
10 until ready.empty?
11 course = ready.shift
12 order << course
13 graph[course].each do |target|
14 indegree[target] -= 1
15 ready << target if indegree[target].zero?
16 end
17 ready.sort!
18 end
19 [order, (courses - order).sort]
20end
