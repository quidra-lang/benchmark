9func solve(names []string, edges [][2]string, showCycle bool) {
10 n := len(names)
11 graph := make([][]bool, n)
12 indegree := make([]int, n)
13 taken := make([]bool, n)
14 index := map[string]int{}
15 for i, name := range names {
16 graph[i] = make([]bool, n)
17 index[name] = i
18 }
19 for _, e := range edges {
20 a, b := index[e[0]], index[e[1]]
21 graph[a][b] = true
22 indegree[b]++
23 }
24 order := []string{}
25 for {
26 best := -1
27 for v := 0; v < n; v++ {
28 if !taken[v] && indegree[v] == 0 && (best < 0 || names[v] < names[best]) {
29 best = v
30 }
31 }
32 if best < 0 {
33 break
34 }
35 order = append(order, names[best])
36 taken[best] = true
37 for v := 0; v < n; v++ {
38 if graph[best][v] {
39 indegree[v]--
40 }
41 }
42 }
43 fmt.Printf("order: %s\n", strings.Join(order, " -> "))
44 if showCycle {
45 remaining := []string{}
46 for i, name := range names {
47 if !taken[i] {
48 remaining = append(remaining, name)
49 }
50 }
51 sort.Strings(remaining)
52 fmt.Printf("cycle: %s\n", strings.Join(remaining, " "))
53 }
54}
