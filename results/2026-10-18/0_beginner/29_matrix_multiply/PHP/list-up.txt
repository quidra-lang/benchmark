$a = [[3, -2, 5, 1], [-4, 7, 0, 6], [2, 9, -8, -3]];
$b = [[1, 4], [-5, 2], [6, -1], [0, 7]];
for ($i = 0; $i < 3; $i++) {
    for ($j = 0; $j < 2; $j++) {
        $value = 0;
        for ($k = 0; $k < 4; $k++) {
            $value += $a[$i][$k] * $b[$k][$j];
        }
        printf('%5d', $value);
    }
    echo "\n";
