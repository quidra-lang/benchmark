    val matrix = Array(5) { IntArray(6) }
    val dr = intArrayOf(0, 1, 0, -1)
    val dc = intArrayOf(1, 0, -1, 0)
    var r = 0
    var c = 0
    var direction = 0
    for (value in 1..30) {
        matrix[r][c] = value
        if (value == 30) break
        val nr = r + dr[direction]
        val nc = c + dc[direction]
        if (nr !in 0 until 5 || nc !in 0 until 6 || matrix[nr][nc] != 0) direction = (direction + 1) % 4
        r += dr[direction]
        c += dc[direction]
    }
