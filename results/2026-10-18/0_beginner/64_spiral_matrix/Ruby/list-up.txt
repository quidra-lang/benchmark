rows, columns = 5, 6
matrix = Array.new(rows) { Array.new(columns, 0) }
directions = [[0, 1], [1, 0], [0, -1], [-1, 0]]
row = column = direction = 0
(1..rows * columns).each do |value|
  matrix[row][column] = value
  break if value == rows * columns
  dr, dc = directions[direction]
  r, c = row + dr, column + dc
  if !r.between?(0, rows - 1) || !c.between?(0, columns - 1) || !matrix[r][c].zero?
    direction = (direction + 1) % 4
    dr, dc = directions[direction]
  end
  row += dr
  column += dc
end
