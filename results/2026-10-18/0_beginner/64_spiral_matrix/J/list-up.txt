  matrix =. 5 6$0
  steps =. 4 2$0 1 1 0 0 _1 _1 0
  row =. column =. direction =. 0
  for_value. >:i.30 do.
    matrix =. value (<row,column)}matrix
    if. value=30 do. break. end.
    'dr dc' =. direction{steps
    nextrow =. row+dr
    nextcolumn =. column+dc
    blocked =. 1
    if. (nextrow>:0)*.(nextrow<5)*.(nextcolumn>:0)*.nextcolumn<6 do.
      blocked =. 0~:(<nextrow,nextcolumn){matrix
    end.
    if. blocked do.
      direction =. 4|direction+1
      'dr dc' =. direction{steps
      nextrow =. row+dr
      nextcolumn =. column+dc
    end.
    row =. nextrow
    column =. nextcolumn
  end.
