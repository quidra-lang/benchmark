1const std = @import("std");
2
3const names = [_][]const u8{ "Nina", "Leo", "Emma" };
4
5fn account(name: []const u8) usize {
6 for (names, 0..) |n, i| {
7 if (std.mem.eql(u8, n, name))
8 return i;
9 }
10 unreachable;
11}
12
13fn lessAccount(_: void, a: usize, b: usize) bool {
14 return std.mem.order(u8, names[a], names[b]) == .lt;
15}
16
17pub fn main(init: std.process.Init) !void {
18 var output_buffer: [8192]u8 = undefined;
19 var writer = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
20 const out = &writer.interface;
21 var balances = [_]u32{ 12000, 3500, 800 };
22 const commands = [_][]const u8{
23 "deposit Emma 1200",
24 "withdraw Leo 4000",
25 "transfer Nina Leo 5000",
26 "withdraw Leo 4000",
27 "transfer Emma Nina 2500",
28 "transfer Emma Nina 2000",
29 "withdraw Emma 300",
30 "deposit Nina 750",
31 "transfer Leo Emma 3200",
32 "withdraw Nina 9800",
33 };
34 for (commands, 0..) |command, k| {
35 var parts = std.mem.splitScalar(u8, command, ' ');
36 const op = parts.next().?;
37 const source = account(parts.next().?);
38 var destination: usize = 0;
39 const transfer = std.mem.eql(u8, op, "transfer");
40 if (transfer)
41 destination = account(parts.next().?);
42 const amount = try std.fmt.parseInt(u32, parts.next().?, 10);
43 var success = true;
44 if (std.mem.eql(u8, op, "deposit")) {
45 balances[source] += amount;
46 } else if (balances[source] < amount) {
47 success = false;
48 } else {
49 balances[source] -= amount;
50 if (transfer)
51 balances[destination] += amount;
52 }
53 try out.print("op {d}: {s} {s}\n", .{
54 k + 1,
55 command,
56 if (success) @as([]const u8, "ok") else "failed (insufficient funds)",
57 });
58 }
59 var order = [_]usize{ 0, 1, 2 };
60 std.mem.sort(usize, &order, {}, lessAccount);
61 for (order) |i|
62 try out.print("{s}: {d}\n", .{ names[i], balances[i] });
63 try out.flush();
64}
