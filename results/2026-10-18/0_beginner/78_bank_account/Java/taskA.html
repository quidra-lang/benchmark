4public class Main {

16 if (parts[0].equals("deposit")) {
17 balances.put(name, balances.get(name) + amount);
18 } else {
19 balances.put(name, balances.get(name) - amount);

21 balances.put(parts[2], balances.get(parts[2]) + amount);
22 }
23 }
24 }
25 System.out.println("op " + (i + 1) + ": " + transactions[i]
26 + (success ? " ok" : " failed (insufficient funds)"));
