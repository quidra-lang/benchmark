int account(string name, string[] names)
    for 0 <= int i < int(len(names))
        if names[i] == name
            return i
string[] names = ["Nina", "Leo", "Emma"]
int[] balances = [12000, 3500, 800]
string[] commands = [
    "deposit Emma 1200", "withdraw Leo 4000", "transfer Nina Leo 5000", "withdraw Leo 4000",
    "transfer Emma Nina 2500", "transfer Emma Nina 2000", "withdraw Emma 300", "deposit Nina 750",
for 0 <= int i < 10
    string[] fields = commands[i].split(" ")
    int source = account(fields[1], names)
    int amount = int.parse(fields[len(fields) - 1])
    bool success = true
    if fields[0] == "deposit"
        balances[source] += amount
    elif amount > balances[source]
        success = false
    else
        balances[source] -= amount
        if fields[0] == "transfer"
            int target = account(fields[2], names)
            balances[target] += amount
