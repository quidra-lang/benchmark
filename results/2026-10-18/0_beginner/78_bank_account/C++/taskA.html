1#include <iostream>

3#include <array>
4#include <map>

7int main() {
8 std::map<std::string, int> balances = {{"Nina", 12000}, {"Leo", 3500}, {"Emma", 800}};
9 std::array<std::string, 10> transactions = {"deposit Emma 1200", "withdraw Leo 4000",
10 "transfer Nina Leo 5000", "withdraw Leo 4000", "transfer Emma Nina 2500", "transfer Emma Nina 2000",
11 "withdraw Emma 300", "deposit Nina 750", "transfer Leo Emma 3200", "withdraw Nina 9800"};
12 for (int k = 0; k < 10; k++) {
13 std::istringstream input(transactions[k]);
14 std::string op, from, to;
15 int amount;
16 input >> op >> from;
17 if (op == "transfer")
18 input >> to >> amount;
19 else
20 input >> amount;
21 bool ok = op == "deposit" || balances[from] >= amount;
22 if (ok) {
23 if (op == "deposit") {
24 balances[from] += amount;
25 } else {
26 balances[from] -= amount;
27 if (op == "transfer")
28 balances[to] += amount;
29 }
30 }
31 std::cout << "op " << k + 1 << ": " << transactions[k] << ' ' << (ok ? "ok" : "failed (insufficient funds)")
32 << '\n';
33 }
34 for (const auto &[name, balance] : balances)
35 std::cout << name << ": " << balance << '\n';
36}
