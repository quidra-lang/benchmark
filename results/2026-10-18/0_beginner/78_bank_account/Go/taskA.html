1package main
2
3import (
4 "fmt"
5 "sort"
6 "strconv"
7 "strings"
8)
9
10func main() {
11 balances := map[string]int{"Nina": 12000, "Leo": 3500, "Emma": 800}
12 transactions := []string{"deposit Emma 1200", "withdraw Leo 4000", "transfer Nina Leo 5000",
13 "withdraw Leo 4000", "transfer Emma Nina 2500", "transfer Emma Nina 2000", "withdraw Emma 300",
14 "deposit Nina 750", "transfer Leo Emma 3200", "withdraw Nina 9800"}
15 for k, transaction := range transactions {
16 parts := strings.Fields(transaction)
17 op, from := parts[0], parts[1]
18 amount, _ := strconv.Atoi(parts[len(parts)-1])
19 ok := op == "deposit" || balances[from] >= amount
20 if ok {
21 if op == "deposit" {
22 balances[from] += amount
23 } else {
24 balances[from] -= amount
25 if op == "transfer" {
26 balances[parts[2]] += amount
27 }
28 }
29 }
30 result := "failed (insufficient funds)"
31 if ok {
32 result = "ok"
33 }
34 fmt.Printf("op %d: %s %s\n", k+1, transaction, result)
35 }
36 names := []string{}
37 for name := range balances {
38 names = append(names, name)
39 }
40 sort.Strings(names)
41 for _, name := range names {
42 fmt.Printf("%s: %d\n", name, balances[name])
43 }
44}
