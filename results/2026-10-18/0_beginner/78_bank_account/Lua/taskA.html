1local balances = {Nina = 12000, Leo = 3500, Emma = 800}
2local transactions = {
3 "deposit Emma 1200", "withdraw Leo 4000", "transfer Nina Leo 5000", "withdraw Leo 4000",
4 "transfer Emma Nina 2500", "transfer Emma Nina 2000", "withdraw Emma 300", "deposit Nina 750",
5 "transfer Leo Emma 3200", "withdraw Nina 9800",
6}
7for number, transaction in ipairs(transactions) do
8 local parts = {}
9 for part in transaction:gmatch("%S+") do parts[#parts + 1] = part end
10 local operation, name, amount = parts[1], parts[2], tonumber(parts[#parts])
11 local success = operation == "deposit" or balances[name] >= amount
12 if success then
13 if operation == "deposit" then
14 balances[name] = balances[name] + amount
15 else
16 balances[name] = balances[name] - amount
17 if operation == "transfer" then balances[parts[3]] = balances[parts[3]] + amount end
18 end
19 end
20 io.write(string.format("op %d: %s %s\n", number, transaction, success and "ok" or "failed (insufficient funds)"))
21end
22local names = {}
23for name in pairs(balances) do names[#names + 1] = name end
24table.sort(names)
25for _, name in ipairs(names) do io.write(string.format("%s: %d\n", name, balances[name])) end
