1balances <- c(Nina = 12000, Leo = 3500, Emma = 800)
2transactions <- c(
3 "deposit Emma 1200", "withdraw Leo 4000", "transfer Nina Leo 5000", "withdraw Leo 4000",
4 "transfer Emma Nina 2500", "transfer Emma Nina 2000", "withdraw Emma 300", "deposit Nina 750",
5 "transfer Leo Emma 3200", "withdraw Nina 9800"
6)
7for (number in seq_along(transactions)) {
8 parts <- strsplit(transactions[number], " ", fixed = TRUE)[[1]]
9 operation <- parts[1]
10 name <- parts[2]
11 amount <- as.integer(tail(parts, 1))
12 success <- operation == "deposit" || balances[name] >= amount
13 if (success) {
14 if (operation == "deposit") {
15 balances[name] <- balances[name] + amount
16 } else {
17 balances[name] <- balances[name] - amount
18 if (operation == "transfer") balances[parts[3]] <- balances[parts[3]] + amount
19 }
20 }
21 result <- if (success) "ok" else "failed (insufficient funds)"
22 cat(sprintf("op %d: %s %s\n", number, transactions[number], result))
23}
