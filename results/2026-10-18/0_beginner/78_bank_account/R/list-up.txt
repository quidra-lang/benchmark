  "transfer Emma Nina 2500", "transfer Emma Nina 2000", "withdraw Emma 300", "deposit Nina 750",

  operation <- parts[1]

  amount <- as.integer(tail(parts, 1))
  success <- operation == "deposit" || balances[name] >= amount
