1import java.util.Arrays;
2import java.util.Comparator;
3import java.util.stream.IntStream;
4
5public class Main {
6 static int find(int[] parent, int v) {
7 if (parent[v] != v) {
8 parent[v] = find(parent, parent[v]);
9 }
10 return parent[v];
11 }
12
13 public static void main(String[] args) {
14 int[][] edges = {{0, 1, 7}, {2, 0, 4}, {1, 2, 9}, {1, 3, 14}, {3, 2, 6}, {2, 4, 3}, {4, 3, 5},
15 {5, 3, 11}, {4, 6, 16}, {6, 5, 8}, {7, 5, 12}, {6, 7, 2}, {7, 4, 10}};
16 Arrays.sort(edges, Comparator.comparingInt(e -> e[2]));
17 int[] parent = IntStream.range(0, 8).toArray();
18 int accepted = 0, skipped = 0, total = 0;
19 for (int[] edge : edges) {
20 int a = find(parent, edge[0]), b = find(parent, edge[1]);
21 if (a == b) {
22 skipped++;
23 continue;
24 }
25 parent[a] = b;
26 accepted++;
27 total += edge[2];
28 char first = (char)('A' + Math.min(edge[0], edge[1]));
29 char second = (char)('A' + Math.max(edge[0], edge[1]));
30 System.out.println(first + "-" + second + " " + edge[2]);
31 if (accepted == 7) {
32 break;
33 }
34 }
35 System.out.println("total weight: " + total);
36 System.out.println("skipped: " + skipped);
37 }
38}
