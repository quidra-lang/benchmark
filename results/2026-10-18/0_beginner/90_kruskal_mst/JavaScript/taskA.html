1const vertices = "ABCDEFGH";
2const edges = [
3 {a: "A", b: "B", weight: 7}, {a: "C", b: "A", weight: 4},
4 {a: "B", b: "C", weight: 9}, {a: "B", b: "D", weight: 14},
5 {a: "D", b: "C", weight: 6}, {a: "C", b: "E", weight: 3},
6 {a: "E", b: "D", weight: 5}, {a: "F", b: "D", weight: 11},
7 {a: "E", b: "G", weight: 16}, {a: "G", b: "F", weight: 8},
8 {a: "H", b: "F", weight: 12}, {a: "G", b: "H", weight: 2},
9 {a: "H", b: "E", weight: 10}
10];
11const parent = new Map([...vertices].map(vertex => [vertex, vertex]));
12
13function find(vertex) {
14 while (parent.get(vertex) !== vertex) {
15 parent.set(vertex, parent.get(parent.get(vertex)));
16 vertex = parent.get(vertex);
17 }
18 return vertex;
19}
20
21let accepted = 0, skipped = 0, total = 0;
22for (const {a, b, weight} of edges.sort((a, b) => a.weight - b.weight)) {
23 const rootA = find(a), rootB = find(b);
24 if (rootA === rootB) {
25 skipped++;
26 continue;
27 }
28 parent.set(rootB, rootA);
29 accepted++;
30 total += weight;
31 console.log(`${[a, b].sort().join("-")} ${weight}`);
32 if (accepted === vertices.length - 1) break;
33}
34console.log(`total weight: ${total}`);
35console.log(`skipped: ${skipped}`);
