1const vertices = "ABCDEFGH";
2const edges = [
3 {a: "A", b: "B", weight: 7}, {a: "C", b: "A", weight: 4},
4 {a: "B", b: "C", weight: 9}, {a: "B", b: "D", weight: 14},
5 {a: "D", b: "C", weight: 6}, {a: "C", b: "E", weight: 3},

9 {a: "H", b: "E", weight: 10}
10];
11const parents = new Map([...vertices].map(vertex => [vertex, vertex]));
12
13function find(vertex: string): string {
14 while (parents.get(vertex) !== vertex) {
15 parents.set(vertex, parents.get(parents.get(vertex)!)!);
16 vertex = parents.get(vertex)!;
17 }
18 return vertex;
19}

21let accepted = 0, skipped = 0, total = 0;
22for (const {a, b, weight} of edges.sort((a, b) => a.weight - b.weight)) {
23 const rootA = find(a), rootB = find(b);
24 if (rootA === rootB) {
25 skipped++;
26 continue;
27 }
28 parents.set(rootB, rootA);

30 total += weight;

33}
