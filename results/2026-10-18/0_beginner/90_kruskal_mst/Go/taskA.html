1package main
2
3import (
4 "fmt"
5 "sort"
6)
7
8type Edge struct {
9 a, b rune
10 weight int
11}
12
13func find(parent *[8]int, v int) int {
14 if parent[v] != v {
15 parent[v] = find(parent, parent[v])
16 }
17 return parent[v]
18}
19
20func main() {
21 edges := []Edge{{'A', 'B', 7}, {'C', 'A', 4}, {'B', 'C', 9}, {'B', 'D', 14}, {'D', 'C', 6},
22 {'C', 'E', 3}, {'E', 'D', 5}, {'F', 'D', 11}, {'E', 'G', 16}, {'G', 'F', 8}, {'H', 'F', 12},
23 {'G', 'H', 2}, {'H', 'E', 10}}
24 sort.Slice(edges, func(i, j int) bool {
25 return edges[i].weight < edges[j].weight
26 })
27 parent := [8]int{}
28 for i := range parent {
29 parent[i] = i
30 }
31 accepted, total, skipped := 0, 0, 0
32 for _, e := range edges {
33 if accepted == 7 {
34 break
35 }
36 a, b := find(&parent, int(e.a-'A')), find(&parent, int(e.b-'A'))
37 if a == b {
38 skipped++
39 continue
40 }
41 parent[a] = b
42 accepted++
43 total += e.weight
44 fmt.Printf("%c-%c %d\n", min(e.a, e.b), max(e.a, e.b), e.weight)
45 }
46 fmt.Printf("total weight: %d\nskipped: %d\n", total, skipped)
47}
