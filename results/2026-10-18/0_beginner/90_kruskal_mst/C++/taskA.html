7struct Edge {
8 char a, b;
9 int weight;
10};
11
12int find(std::array<int, 8> &parent, int v) {
13 if (parent[v] != v)
14 parent[v] = find(parent, parent[v]);
15 return parent[v];
16}
17
18int main() {
19 std::vector<Edge> edges = {{'A', 'B', 7}, {'C', 'A', 4}, {'B', 'C', 9}, {'B', 'D', 14}, {'D', 'C', 6},
20 {'C', 'E', 3}, {'E', 'D', 5}, {'F', 'D', 11}, {'E', 'G', 16}, {'G', 'F', 8}, {'H', 'F', 12},
21 {'G', 'H', 2}, {'H', 'E', 10}};
22 std::sort(edges.begin(), edges.end(), [](const Edge &a, const Edge &b) {
23 return a.weight < b.weight;
24 });
25 std::array<int, 8> parent;
26 std::iota(parent.begin(), parent.end(), 0);
27 int accepted = 0, total = 0, skipped = 0;
28 for (const Edge &e : edges) {
29 if (accepted == 7)
30 break;
31 int a = find(parent, e.a - 'A'), b = find(parent, e.b - 'A');
32 if (a == b) {
33 ++skipped;
34 continue;
35 }
36 parent[a] = b;
37 ++accepted;
38 total += e.weight;
39 std::cout << std::min(e.a, e.b) << '-' << std::max(e.a, e.b) << ' ' << e.weight << '\n';
40 }
41 std::cout << "total weight: " << total << "\nskipped: " << skipped << '\n';
42}
