        std::string text = std::to_string(n);
        std::size_t start = text[0] == '-' ? 1 : 0;
        if (start)
            std::cout << '-';
        for (std::size_t i = start; i < text.size(); i++) {

                std::cout << ',';
            std::cout << text[i];
