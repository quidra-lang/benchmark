        std::size_t start = text[0] == '-' ? 1 : 0;
        if (start)
            std::cout << '-';
        for (std::size_t i = start; i < text.size(); i++) {
            if (i > start && (text.size() - i) % 3 == 0)
                std::cout << ',';
            std::cout << text[i];
        }
