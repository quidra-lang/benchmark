1local function outcome(board)
2 local lines = {board[1], board[2], board[3]}
3 for column = 1, 3 do
4 lines[#lines + 1] = board[1]:sub(column, column) .. board[2]:sub(column, column) .. board[3]:sub(column, column)
5 end
6 lines[#lines + 1] = board[1]:sub(1, 1) .. board[2]:sub(2, 2) .. board[3]:sub(3, 3)
7 lines[#lines + 1] = board[1]:sub(3, 3) .. board[2]:sub(2, 2) .. board[3]:sub(1, 1)
8 for _, line in ipairs(lines) do
9 local first = line:sub(1, 1)
10 if first ~= "." and line == string.rep(first, 3) then return first .. " wins" end
11 end
12 return table.concat(board):find(".", 1, true) and "in progress" or "draw"
13end
