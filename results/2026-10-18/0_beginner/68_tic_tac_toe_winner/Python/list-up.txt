boards = [
    ["XXX", "OO.", "O.X"],
    ["X.O", ".X.", "O.."],
    ["XO.", ".OX", "XO."],
    ["XOX", "XOO", "OXX"],
    ["X.O", "XO.", "O.X"],
    ["XOX", "OXO", "OXX"],
]
for number, board in enumerate(boards, 1):
    lines = list(board)
    lines += ["".join(column) for column in zip(*board)]
    lines += [
        "".join(board[i][i] for i in range(3)),
        "".join(board[i][2 - i] for i in range(3)),
    ]
    winner = next(
        (line[0] for line in lines if line[0] != "." and len(set(line)) == 1), None
    )
    if winner:
        result = f"{winner} wins"
    else:
        result = "in progress" if any("." in row for row in board) else "draw"
