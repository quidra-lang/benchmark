7foreach ($frequencies as $symbol => $frequency) {

9 $trees->insert($node, [-$frequency, -ord($symbol)]);
10}
11while (count($trees) > 1) {
12 $left = $trees->extract();
13 $right = $trees->extract();

15 'weight' => $left['weight'] + $right['weight'],

17 'left' => $left,
18 'right' => $right,
19 ];
20 $trees->insert($node, [-$node['weight'], -$node['smallest']]);
