1import java.util.Comparator;
2import java.util.HashMap;
3import java.util.Map;
4import java.util.PriorityQueue;
5import java.util.TreeMap;
6
7public class Main {
8 record Node(int weight, int minimum, Character symbol, Node left, Node right) {}
9
10 static void codes(Node node, String prefix, Map<Character, String> result) {
11 if (node.symbol != null) {
12 result.put(node.symbol, prefix);
13 } else {
14 codes(node.left, prefix + "0", result);
15 codes(node.right, prefix + "1", result);
16 }
17 }
18
19 public static void main(String[] args) {
20 String text = "she sells sea shells by the sea shore";
21 Map<Character, Integer> frequencies = new TreeMap<>();
22 for (char ch : text.toCharArray()) {
23 frequencies.merge(ch, 1, Integer::sum);
24 }
25 PriorityQueue<Node> queue = new PriorityQueue<>(
26 Comparator.comparingInt((Node n) -> n.weight).thenComparingInt(n -> n.minimum));
27 for (var entry : frequencies.entrySet()) {
28 queue.add(new Node(entry.getValue(), entry.getKey(), entry.getKey(), null, null));
29 }
30 while (queue.size() > 1) {
31 Node left = queue.remove(), right = queue.remove();
32 queue.add(new Node(left.weight + right.weight, Math.min(left.minimum, right.minimum), null, left, right));
33 }
34 Map<Character, String> codes = new HashMap<>();
35 codes(queue.remove(), "", codes);
36 for (var entry : frequencies.entrySet()) {
37 char ch = entry.getKey();
38 String code = codes.get(ch);
39 System.out.println("'" + ch + "' freq=" + entry.getValue() + " code=" + code);
40 }
41 int bits = frequencies.entrySet().stream()
42 .mapToInt(entry -> entry.getValue() * codes.get(entry.getKey()).length())
43 .sum();
44 System.out.println("encoded bits: " + bits);
45 System.out.println("fixed 8-bit: " + text.length() * 8);
46 }
47}
