4typedef struct {
5 int weight, minimum, left, right, symbol;
6} Node;

8int pop(int *roots, int *length, const Node *nodes) {
9 int best = 0;
10 for (int i = 1; i < *length; i++) {
11 Node a = nodes[roots[i]], b = nodes[roots[best]];
12 if (a.weight < b.weight || (a.weight == b.weight && a.minimum < b.minimum))
13 best = i;
14 }
15 int result = roots[best];
16 roots[best] = roots[--*length];
17 return result;
18}

20void assign(const Node *nodes, int root, char *prefix, int depth, char codes[128][128]) {
21 if (nodes[root].symbol >= 0) {
22 prefix[depth] = 0;
23 strcpy(codes[nodes[root].symbol], prefix);
24 return;
25 }
26 prefix[depth] = '0';
27 assign(nodes, nodes[root].left, prefix, depth + 1, codes);
28 prefix[depth] = '1';
29 assign(nodes, nodes[root].right, prefix, depth + 1, codes);
30}
