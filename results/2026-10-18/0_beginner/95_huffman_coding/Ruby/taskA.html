1text = 'she sells sea shells by the sea shore'
2frequencies = text.chars.tally
3trees = frequencies.map do |symbol, weight|
4 {weight: weight, minimum: symbol, symbol: symbol, left: nil, right: nil}
5end
6while trees.length > 1
7 trees.sort_by! { |tree| [tree[:weight], tree[:minimum]] }
8 left, right = trees.shift, trees.shift
9 trees << {
10 weight: left[:weight] + right[:weight],
11 minimum: [left[:minimum], right[:minimum]].min,
12 symbol: nil, left: left, right: right
13 }
14end
15codes = {}
16visit = lambda do |tree, prefix|
17 if tree[:symbol]
18 codes[tree[:symbol]] = prefix
19 else
20 visit.call(tree[:left], prefix + '0')
21 visit.call(tree[:right], prefix + '1')
22 end
23end
24visit.call(trees[0], '')
25frequencies.keys.sort.each do |symbol|

27end
