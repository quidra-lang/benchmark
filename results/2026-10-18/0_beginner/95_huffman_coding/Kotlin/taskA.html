1import java.util.PriorityQueue
2
3data class Node(
4 val weight: Int,
5 val minimum: Int,
6 val symbol: Char? = null,
7 val left: Node? = null,
8 val right: Node? = null,
9)
10
11fun codes(node: Node, prefix: String, result: MutableMap<Char, String>) {
12 if (node.symbol != null) {
13 result[node.symbol] = prefix
14 } else {
15 codes(node.left!!, prefix + "0", result)
16 codes(node.right!!, prefix + "1", result)
17 }
18}
19
20fun main() {
21 val text = "she sells sea shells by the sea shore"
22 val frequencies = text.groupingBy { it }.eachCount().toSortedMap()
23 val queue = PriorityQueue<Node>(compareBy<Node> { it.weight }.thenBy { it.minimum })
24 for ((ch, count) in frequencies) queue.add(Node(count, ch.code, ch))
25 while (queue.size > 1) {
26 val left = queue.remove()
27 val right = queue.remove()
28 queue.add(Node(left.weight + right.weight, minOf(left.minimum, right.minimum), null, left, right))
29 }
30 val mapping = mutableMapOf<Char, String>()
31 codes(queue.remove(), "", mapping)
32 for ((ch, count) in frequencies) {
33 val code = mapping.getValue(ch)
34 println("'$ch' freq=$count code=$code")
35 }
36 println("encoded bits: ${frequencies.entries.sumOf { it.value * mapping.getValue(it.key).length }}")
37 println("fixed 8-bit: ${text.length * 8}")
38}
