1local text = "she sells sea shells by the sea shore"
2local frequencies = {}
3for symbol in text:gmatch(".") do frequencies[symbol] = (frequencies[symbol] or 0) + 1 end
4local symbols, trees = {}, {}
5for symbol, frequency in pairs(frequencies) do
6 symbols[#symbols + 1] = symbol
7 trees[#trees + 1] = {weight = frequency, smallest = symbol:byte(), symbol = symbol}
8end
9table.sort(symbols)
10while #trees > 1 do
11 table.sort(trees, function(a, b)
12 return a.weight < b.weight or (a.weight == b.weight and a.smallest < b.smallest)
13 end)
14 local left = table.remove(trees, 1)
15 local right = table.remove(trees, 1)
16 trees[#trees + 1] = {
17 weight = left.weight + right.weight,
18 smallest = math.min(left.smallest, right.smallest),
19 left = left,
20 right = right,
21 }
22end
23local codes = {}
24
25local function visit(node, prefix)
26 if node.symbol then
27 codes[node.symbol] = prefix
28 else
29 visit(node.left, prefix .. "0")
30 visit(node.right, prefix .. "1")
31 end
32end
33
34visit(trees[1], "")
35local bits = 0
36for _, symbol in ipairs(symbols) do
37 io.write(string.format("'%s' freq=%d code=%s\n", symbol, frequencies[symbol], codes[symbol]))
38 bits = bits + frequencies[symbol] * #codes[symbol]
39end
40io.write(string.format("encoded bits: %d\nfixed 8-bit: %d\n", bits, #text * 8))
