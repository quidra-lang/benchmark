1import 'dart:math' as math;
2
3class HNode {
4 final int weight, minimum;
5 final int? symbol;
6 final HNode? left, right;
7 HNode(this.weight, this.minimum, {this.symbol, this.left, this.right});
8}
9
10void main() {
11 const text = 'she sells sea shells by the sea shore';
12 final frequency = <int, int>{};
13 for (final c in text.codeUnits) {
14 frequency[c] = (frequency[c] ?? 0) + 1;
15 }
16 final active = [
17 for (final e in frequency.entries) HNode(e.value, e.key, symbol: e.key),
18 ];
19
20 int compare(HNode a, HNode b) {
21 final c = a.weight.compareTo(b.weight);
22 return c != 0 ? c : a.minimum.compareTo(b.minimum);
23 }
24
25 while (active.length > 1) {
26 active.sort(compare);
27 final left = active.removeAt(0);
28 final right = active.removeAt(0);
29 active.add(
30 HNode(
31 left.weight + right.weight,
32 math.min(left.minimum, right.minimum),
33 left: left,
34 right: right,
35 ),
36 );
37 }
38 final codes = <int, String>{};
39 void assign(HNode node, String path) {
40 if (node.symbol != null) {
41 codes[node.symbol!] = path;
42 return;
43 }
44 assign(node.left!, '${path}0');
45 assign(node.right!, '${path}1');
46 }
47
48 assign(active.single, '');
49 final symbols = frequency.keys.toList()..sort();
50 var bits = 0;
51 for (final c in symbols) {
52 print("'${String.fromCharCode(c)}' freq=${frequency[c]} code=${codes[c]}");
53 bits += frequency[c]! * codes[c]!.length;
54 }
55 print('encoded bits: $bits\nfixed 8-bit: ${text.length * 8}');
56}
