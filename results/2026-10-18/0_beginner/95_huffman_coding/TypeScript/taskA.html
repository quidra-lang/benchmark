11for (const symbol of text) frequencies.set(symbol, (frequencies.get(symbol) ?? 0) + 1);

13 weight, minimum: symbol, symbol, left: null, right: null

15while (trees.length > 1) {

18 trees.push({
19 weight: left.weight + right.weight,

21 symbol: null, left, right

26function visit(tree: HuffmanNode, prefix: string): void {
27 if (tree.symbol !== null) {
28 codes.set(tree.symbol, prefix);
29 } else {
30 visit(tree.left!, prefix + "0");

32 }
