8static void Codes(Node node, string prefix, Dictionary<char, string> result)
9{
10 if (node.Symbol is char ch)
11 {
12 result[ch] = prefix;
13 }
14 else
15 {
16 Codes(node.Left!, prefix + "0", result);
17 Codes(node.Right!, prefix + "1", result);
18 }
19}
20
21string text = "she sells sea shells by the sea shore";
22var frequencies = text.GroupBy(ch => ch).ToDictionary(g => g.Key, g => g.Count());
23var queue = new PriorityQueue<Node, (int weight, int minimum)>();
24foreach (var pair in frequencies)
25{
26 var leaf = new Node(pair.Value, pair.Key, pair.Key);
27 queue.Enqueue(leaf, (leaf.Weight, leaf.Minimum));
28}
29while (queue.Count > 1)
30{
31 Node left = queue.Dequeue(), right = queue.Dequeue();
32 var parent = new Node(left.Weight + right.Weight, Math.Min(left.Minimum, right.Minimum), null, left, right);
33 queue.Enqueue(parent, (parent.Weight, parent.Minimum));
34}
35var codes = new Dictionary<char, string>();
36Codes(queue.Dequeue(), "", codes);
37foreach (char ch in frequencies.Keys.Order())
38{
39 string code = codes[ch];
40 Console.WriteLine($"'{ch}' freq={frequencies[ch]} code={code}");
41}
42Console.WriteLine($"encoded bits: {frequencies.Sum(pair => pair.Value * codes[pair.Key].Length)}");
43Console.WriteLine($"fixed 8-bit: {text.Length * 8}");

45record Node(int Weight, int Minimum, char? Symbol, Node? Left = null, Node? Right = null);
