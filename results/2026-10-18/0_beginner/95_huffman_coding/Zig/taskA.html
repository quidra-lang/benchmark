1const std = @import("std");
2
3const Node = struct { weight: u32, minimum: u8, symbol: ?u8 = null, left: ?usize = null, right: ?usize = null };
4
5fn compare(nodes: *const [64]Node, a: usize, b: usize) std.math.Order {
6 return if (nodes[a].weight != nodes[b].weight)
7 std.math.order(nodes[a].weight, nodes[b].weight)
8 else
9 std.math.order(nodes[a].minimum, nodes[b].minimum);
10}
11
12fn assign(
13 nodes: *const [64]Node,
14 index: usize,
15 path: *[64]u8,
16 depth: usize,
17 codes: *[128][64]u8,
18 lengths: *[128]usize,
19) void {
20 const node = nodes[index];
21 if (node.symbol) |c| {
22 @memcpy(codes[c][0..depth], path[0..depth]);
23 lengths[c] = depth;
24 return;
25 }
26 path[depth] = '0';
27 assign(nodes, node.left.?, path, depth + 1, codes, lengths);
28 path[depth] = '1';
29 assign(nodes, node.right.?, path, depth + 1, codes, lengths);
30}
31
32pub fn main(init: std.process.Init) !void {
33 var output_buffer: [8192]u8 = undefined;
34 var writer = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
35 const out = &writer.interface;
36 const text = "she sells sea shells by the sea shore";
37 var frequency: [128]u32 = @splat(0);
38 for (text) |c|
39 frequency[c] += 1;
40 var nodes: [64]Node = undefined;
41 var nlen: usize = 0;
42 var active = std.PriorityQueue(usize, *const [64]Node, compare).initContext(&nodes);
43 defer active.deinit(std.heap.page_allocator);
44 for (frequency, 0..) |freq, c| {
45 if (freq > 0) {
46 nodes[nlen] = .{ .weight = freq, .minimum = @intCast(c), .symbol = @intCast(c) };
47 try active.push(std.heap.page_allocator, nlen);
48 nlen += 1;
49 }
50 }
51 while (active.count() > 1) {
52 const left = active.pop().?;
53 const right = active.pop().?;
54 nodes[nlen] = .{
55 .weight = nodes[left].weight + nodes[right].weight,
56 .minimum = @min(nodes[left].minimum, nodes[right].minimum),
57 .left = left,
58 .right = right,
59 };
60 try active.push(std.heap.page_allocator, nlen);
61 nlen += 1;
62 }
63 var codes: [128][64]u8 = undefined;
64 var lengths: [128]usize = @splat(0);
65 var path: [64]u8 = undefined;
66 assign(&nodes, active.pop().?, &path, 0, &codes, &lengths);
67 var bits: usize = 0;
68 for (frequency, 0..) |freq, c| {
69 if (freq > 0) {
70 try out.print("'{c}' freq={d} code={s}\n", .{ @as(u8, @intCast(c)), freq, codes[c][0..lengths[c]] });
71 bits += @as(usize, freq) * lengths[c];
72 }
73 }
74 try out.print("encoded bits: {d}\nfixed 8-bit: {d}\n", .{ bits, text.len * 8 });
75 try out.flush();
76}
