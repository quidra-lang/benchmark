3import "fmt"
4
5type Node struct {
6 weight, minimum, left, right, symbol int
7}
8
9func pop(roots *[]int, nodes []Node) int {
10 best := 0
11 for i := 1; i < len(*roots); i++ {
12 a, b := nodes[(*roots)[i]], nodes[(*roots)[best]]
13 if a.weight < b.weight || a.weight == b.weight && a.minimum < b.minimum {
14 best = i
15 }
16 }
17 result := (*roots)[best]
18 (*roots)[best] = (*roots)[len(*roots)-1]
19 *roots = (*roots)[:len(*roots)-1]
20 return result
21}
22
23func assign(nodes []Node, root int, prefix string, codes *[128]string) {
24 if nodes[root].symbol >= 0 {
25 codes[nodes[root].symbol] = prefix
26 return
27 }
28 assign(nodes, nodes[root].left, prefix+"0", codes)
29 assign(nodes, nodes[root].right, prefix+"1", codes)
30}
31
32func main() {
33 text := "she sells sea shells by the sea shore"
34 freq := [128]int{}
35 for i := range text {
36 freq[text[i]]++
37 }
38 nodes := []Node{}
39 roots := []int{}
40 for c, f := range freq {
41 if f > 0 {
42 roots = append(roots, len(nodes))
43 nodes = append(nodes, Node{f, c, -1, -1, c})
44 }
45 }
46 for len(roots) > 1 {
47 left, right := pop(&roots, nodes), pop(&roots, nodes)
48 roots = append(roots, len(nodes))
49 weight := nodes[left].weight + nodes[right].weight
50 nodes = append(nodes, Node{weight, min(nodes[left].minimum, nodes[right].minimum), left, right, -1})
51 }
52 codes := [128]string{}
53 assign(nodes, roots[0], "", &codes)
54 bits := 0
55 for c, f := range freq {
56 if f > 0 {
57 fmt.Printf("'%c' freq=%d code=%s\n", c, f, codes[c])
58 bits += f * len(codes[c])
59 }
60 }
61 fmt.Printf("encoded bits: %d\nfixed 8-bit: %d\n", bits, len(text)*8)
62}
