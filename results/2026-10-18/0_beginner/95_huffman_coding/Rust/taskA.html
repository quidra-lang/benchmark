1struct Node {
2 weight: usize,
3 minimum: usize,
4 left: Option<usize>,
5 right: Option<usize>,
6 symbol: Option<usize>,
7}
8
9fn pop(roots: &mut Vec<usize>, nodes: &[Node]) -> usize {
10 let position = roots
11 .iter()
12 .enumerate()
13 .min_by_key(|pair| {
14 let v = *pair.1;
15 (nodes[v].weight, nodes[v].minimum)
16 })
17 .unwrap()
18 .0;
19 roots.remove(position)
20}
21
22fn assign(nodes: &[Node], root: usize, prefix: String, codes: &mut [String]) {
23 if let Some(symbol) = nodes[root].symbol {
24 codes[symbol] = prefix;
25 return;
26 }
27 assign(
28 nodes,
29 nodes[root].left.unwrap(),
30 prefix.clone() + "0",
31 codes,
32 );
33 assign(nodes, nodes[root].right.unwrap(), prefix + "1", codes);
34}

36fn main() {
37 let text = "she sells sea shells by the sea shore";
38 let mut freq = [0; 128];
39 for c in text.bytes() {
40 freq[c as usize] += 1;
41 }
42 let mut nodes = Vec::new();
43 let mut roots = Vec::new();
44 for c in 0..128 {
45 if freq[c] > 0 {
46 roots.push(nodes.len());
47 nodes.push(Node {
48 weight: freq[c],
49 minimum: c,
50 left: None,
51 right: None,
52 symbol: Some(c),
53 });
54 }
55 }
56 while roots.len() > 1 {
57 let left = pop(&mut roots, &nodes);
58 let right = pop(&mut roots, &nodes);
59 roots.push(nodes.len());
60 nodes.push(Node {
61 weight: nodes[left].weight + nodes[right].weight,
62 minimum: nodes[left].minimum.min(nodes[right].minimum),
63 left: Some(left),
64 right: Some(right),
65 symbol: None,
66 });
67 }
68 let mut codes = vec![String::new(); 128];
69 assign(&nodes, roots[0], String::new(), &mut codes);
70 let mut bits = 0;
71 for c in 0..128 {
72 if freq[c] > 0 {
73 println!("'{}' freq={} code={}", c as u8 as char, freq[c], codes[c]);
74 bits += freq[c] * codes[c].len();
75 }
76 }
77 println!("encoded bits: {}\nfixed 8-bit: {}", bits, text.len() * 8);
78}
