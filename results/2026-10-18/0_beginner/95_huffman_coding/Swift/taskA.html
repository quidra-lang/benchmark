1class Node {
2 let weight: Int, minimum: Int
3 let symbol: Character?
4 let left: Node?, right: Node?
5 init(_ weight: Int, _ minimum: Int, _ symbol: Character?, _ left: Node? = nil, _ right: Node? = nil) {
6 self.weight = weight
7 self.minimum = minimum
8 self.symbol = symbol
9 self.left = left
10 self.right = right
11 }
12}
13
14func codes(_ node: Node, _ prefix: String, _ result: inout [Character: String]) {
15 if let symbol = node.symbol {
16 result[symbol] = prefix
17 } else {
18 codes(node.left!, prefix + "0", &result)
19 codes(node.right!, prefix + "1", &result)
20 }
21}
22
23let text = "she sells sea shells by the sea shore"
24var frequencies = [Character: Int]()
25for ch in text { frequencies[ch, default: 0] += 1 }
26var trees = frequencies.map { ch, count in Node(count, Int(ch.asciiValue!), ch) }
27while trees.count > 1 {
28 trees.sort { $0.weight == $1.weight ? $0.minimum < $1.minimum : $0.weight < $1.weight }
29 let left = trees.removeFirst(), right = trees.removeFirst()
30 trees.append(Node(left.weight + right.weight, min(left.minimum, right.minimum), nil, left, right))
31}
32var mapping = [Character: String]()
33codes(trees[0], "", &mapping)
34for ch in frequencies.keys.sorted() {
35 let code = mapping[ch]!, count = frequencies[ch]!
36 print("'\(ch)' freq=\(count) code=\(code)")
37}
38print("encoded bits: \(frequencies.reduce(0) { $0 + $1.value * mapping[$1.key]!.count })")
39print("fixed 8-bit: \(text.count * 8)")
