int[][] arrays = [
    [3, -5, 8, -2, 6, -9, 4, 1, -7, 5, -3, 2],
    [-4, 7, -1, 3, -8, 10, -2, 5, -6, -2, 9, -12, 4, -1],
]
for int[] values = arrays[]
    int current = values[0]
    int best = current
    int start = 0
    int best_start = 0
    int best_end = 0
    for 1 <= int i < int(len(values))
        if values[i] > current + values[i]
            current = values[i]
            start = i
        else
            current += values[i]
        if current > best
            best = current
            best_start = start
            best_end = i
