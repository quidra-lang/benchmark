    const arrays = [_][]const i32{
        &.{ 3, -5, 8, -2, 6, -9, 4, 1, -7, 5, -3, 2 },
        &.{ -4, 7, -1, 3, -8, 10, -2, 5, -6, -2, 9, -12, 4, -1 },
    };
    for (arrays) |a| {
        var current = a[0];
        var best = current;
        var start: usize = 0;
        var bi: usize = 0;
        var bj: usize = 0;
        for (1..a.len) |i| {
            if (current + a[i] < a[i]) {
                current = a[i];
                start = i;
            } else {
                current += a[i];
            }
            if (current > best) {
                best = current;
                bi = start;
                bj = i;
            }
        }
    }
