1using System;

7int[] coins = { 1, 5, 6, 9 };

11 int[] ways = new int[amount + 1];

13 ways[0] = 1;

15 {
16 foreach (int coin in coins)

19 }
20 }
21
22 foreach (int coin in coins)
23 {
24 for (int value = coin; value <= amount; value++) ways[value] += ways[value - coin];
25 }

28 {
29 while (remaining >= coins[i])
30 {
31 remaining -= coins[i];
32 greedy++;
33 }
34 }
