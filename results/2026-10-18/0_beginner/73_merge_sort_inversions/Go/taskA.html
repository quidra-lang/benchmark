5func mergeSort(a []int, l, r int) int {
6 if l == r {
7 return 0
8 }
9 m := (l + r) / 2
10 count := mergeSort(a, l, m)
11 count += mergeSort(a, m+1, r)
12 temp := []int{}
13 i, j := l, m+1
14 for i <= m && j <= r {
15 if a[i] < a[j] {
16 temp = append(temp, a[i])
17 i++
18 } else {
19 temp = append(temp, a[j])
20 j++
21 count += m - i + 1
22 }
23 }
24 temp = append(temp, a[i:m+1]...)
25 temp = append(temp, a[j:r+1]...)
26 copy(a[l:r+1], temp)
27 fmt.Printf("merge [%d..%d]:", l, r)
28 for _, n := range a[l : r+1] {
29 fmt.Printf(" %d", n)
30 }
31 fmt.Println()
32 return count
33}
