7 }
8 int m = (l + r) / 2;
9 int inversions = mergeSort(values, l, m);
10 inversions += mergeSort(values, m + 1, r);
11 int[] merged = new int[r - l + 1];
12 int i = l, j = m + 1, k = 0;
13 while (i <= m && j <= r) {
14 if (values[i] < values[j]) {
15 merged[k++] = values[i++];
16 } else {
17 merged[k++] = values[j++];
18 inversions += m - i + 1;
19 }
20 }
21 while (i <= m) {
22 merged[k++] = values[i++];
23 }
24 while (j <= r) {
25 merged[k++] = values[j++];
26 }
