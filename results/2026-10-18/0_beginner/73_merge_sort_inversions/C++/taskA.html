5int merge_sort(std::vector<int> &a, int l, int r) {
6 if (l == r)
7 return 0;
8 int m = (l + r) / 2, count = merge_sort(a, l, m);
9 count += merge_sort(a, m + 1, r);
10 std::vector<int> temp;
11 int i = l, j = m + 1;
12 while (i <= m && j <= r) {
13 if (a[i] < a[j]) {
14 temp.push_back(a[i++]);
15 } else {
16 temp.push_back(a[j++]);
17 count += m - i + 1;
18 }
19 }
20 temp.insert(temp.end(), a.begin() + i, a.begin() + m + 1);
21 temp.insert(temp.end(), a.begin() + j, a.begin() + r + 1);
22 std::copy(temp.begin(), temp.end(), a.begin() + l);
23 std::cout << "merge [" << l << ".." << r << "]:";
24 for (int k = l; k <= r; k++)
25 std::cout << ' ' << a[k];
26 std::cout << '\n';
27 return count;
28}
