import collections
class Team(string name)
    int wins := 0
    int draws := 0
    int losses := 0
    int scored := 0
    int conceded := 0
    int points()
        return 3 * wins + draws
    int difference()
        return scored - conceded
    void .record(int goals, int against)
        scored += goals
        conceded += against
        if goals == against
            draws += 1
        elif goals > against
            wins += 1
        else
            losses += 1
names := ["Lions", "Tigers", "Eagles", "Sharks", "Wolves", "Bears"]
matches := [
    "Tigers 2-2 Wolves",
    "Wolves 2-1 Eagles",
    "Lions 0-1 Tigers",
    "Tigers 3-3 Bears",
    "Sharks 2-2 Wolves",
    "Bears 3-1 Wolves",
    "Tigers 2-1 Eagles",
    "Sharks 2-0 Tigers",
    "Wolves 2-1 Lions",
teams := [for string name := names[] then Team(name)]
for match_text := matches[]
    fields := match_text.split(" ")
    home := names.index(fields[0])
    away := names.index(fields[2])
    score := fields[1].split("-")
    first := int.parse(score[0])
    second := int.parse(score[1])
    .teams[home].record(first, second)
    .teams[away].record(second, first)
points := [for team := teams[] then team.points()]
differences := [for team := teams[] then team.difference()]
goals := [for team := teams[] then team.scored]
teams = teams.sorted(by = (points, differences, goals, names), descending = (true, true, true, false))
