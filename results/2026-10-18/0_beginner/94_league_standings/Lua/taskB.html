1local teams = {"Lions", "Tigers", "Eagles", "Sharks", "Wolves", "Bears"}
2local records = {}
3for _, team in ipairs(teams) do
4 records[team] = {played = 0, wins = 0, draws = 0, losses = 0, gf = 0, ga = 0, points = 0}
5end
6local matches = {
7 "Lions 0-3 Bears", "Tigers 2-2 Wolves", "Eagles 4-0 Sharks", "Bears 0-1 Sharks",
8 "Wolves 2-1 Eagles", "Lions 0-1 Tigers", "Tigers 3-3 Bears", "Eagles 2-2 Lions",
9 "Sharks 2-2 Wolves", "Bears 3-1 Wolves", "Lions 2-0 Sharks", "Tigers 2-1 Eagles",
10 "Eagles 1-0 Bears", "Sharks 2-0 Tigers", "Wolves 2-1 Lions",
11}
12for _, match in ipairs(matches) do
13 local home, h, a, away = match:match("(%a+) (%d+)%-(%d+) (%a+)")
14 h, a = tonumber(h), tonumber(a)
15 for _, side in ipairs({{home, h, a}, {away, a, h}}) do
16 local record = records[side[1]]
17 local gf, ga = side[2], side[3]
18 record.played = record.played + 1
19 record.gf, record.ga = record.gf + gf, record.ga + ga
20 if gf > ga then
21 record.wins = record.wins + 1
22 record.points = record.points + 3
23 elseif gf == ga then
24 record.draws = record.draws + 1
25 record.points = record.points + 1
26 else
27 record.losses = record.losses + 1
28 end
29 end
30end
31table.sort(teams, function(a, b)
32 local x, y = records[a], records[b]
33 if x.points ~= y.points then return x.points > y.points end
34 if x.gf - x.ga ~= y.gf - y.ga then return x.gf - x.ga > y.gf - y.ga end
35 if x.gf ~= y.gf then return x.gf > y.gf end
36 return a < b
37end)
