1class Team {
2 final String name;
3 int played = 0, wins = 0, draws = 0, losses = 0, gf = 0, ga = 0, points = 0;
4 Team(this.name);
5 int get gd => gf - ga;

7
8void main() {
9 final teams = {
10 for (final name in [
11 'Lions',
12 'Tigers',
13 'Eagles',
14 'Sharks',
15 'Wolves',
16 'Bears',
17 ])
18 name: Team(name),
19 };
20 const matches = [
21 'Lions 0-3 Bears',
22 'Tigers 2-2 Wolves',
23 'Eagles 4-0 Sharks',
24 'Bears 0-1 Sharks',
25 'Wolves 2-1 Eagles',
26 'Lions 0-1 Tigers',
27 'Tigers 3-3 Bears',
28 'Eagles 2-2 Lions',
29 'Sharks 2-2 Wolves',
30 'Bears 3-1 Wolves',
31 'Lions 2-0 Sharks',
32 'Tigers 2-1 Eagles',
33 'Eagles 1-0 Bears',
34 'Sharks 2-0 Tigers',
35 'Wolves 2-1 Lions',
36 ];
37 for (final match in matches) {

39 final goals = f[1].split('-');
40 final h = int.parse(goals[0]);
41 final a = int.parse(goals[1]);
42 final home = teams[f[0]]!;
43 final away = teams[f[2]]!;

48 away.gf += a;

59 home.draws++;
60 away.draws++;
61 home.points++;
62 away.points++;
63 }

66 ..sort((a, b) {
67 var c = b.points.compareTo(a.points);
68 if (c != 0) return c;
69 c = b.gd.compareTo(a.gd);
70 if (c != 0) return c;
71 c = b.gf.compareTo(a.gf);
72 return c != 0 ? c : a.name.compareTo(b.name);
73 });
74 for (var i = 0; i < order.length; i++) {
75 final t = order[i];
76 print(
77 '${i + 1}. ${t.name} P=${t.played} W=${t.wins} D=${t.draws} L=${t.losses} GF=${t.gf} GA=${t.ga} GD=${t.gd > 0 ? '+' : ''}${t.gd} Pts=${t.points}',
78 );
79 }
