class Team {
  final String name;
  int played = 0, wins = 0, draws = 0, losses = 0, gf = 0, ga = 0, points = 0;
  Team(this.name);
  int get gd => gf - ga;
}
  final teams = {
    for (final name in [
      'Lions',
      'Tigers',
      'Eagles',
      'Sharks',
      'Wolves',
      'Bears',
    ])
      name: Team(name),
  };
  const matches = [
    'Tigers 2-2 Wolves',
    'Wolves 2-1 Eagles',
    'Lions 0-1 Tigers',
    'Tigers 3-3 Bears',
    'Sharks 2-2 Wolves',
    'Bears 3-1 Wolves',
    'Tigers 2-1 Eagles',
    'Sharks 2-0 Tigers',
    'Wolves 2-1 Lions',
  ];
  for (final match in matches) {
    final f = match.split(' ');
    final goals = f[1].split('-');
    final h = int.parse(goals[0]);
    final a = int.parse(goals[1]);
    final home = teams[f[0]]!;
    final away = teams[f[2]]!;
    home.played++;
    away.played++;
    home.gf += h;
    home.ga += a;
    away.gf += a;
    away.ga += h;
    if (h > a) {
      home.wins++;
      away.losses++;
      home.points += 3;
    } else if (h < a) {
      away.wins++;
      home.losses++;
      away.points += 3;
    } else {
      home.draws++;
      away.draws++;
      home.points++;
      away.points++;
    }
  }
  final order = teams.values.toList()
    ..sort((a, b) {
      var c = b.points.compareTo(a.points);
      if (c != 0) return c;
      c = b.gd.compareTo(a.gd);
      if (c != 0) return c;
      c = b.gf.compareTo(a.gf);
      return c != 0 ? c : a.name.compareTo(b.name);
    });
