const Team = struct {
    name: []const u8,
    played: u32 = 0,
    wins: u32 = 0,
    draws: u32 = 0,
    losses: u32 = 0,
    gf: i32 = 0,
    ga: i32 = 0,
    points: u32 = 0,
};
fn find(teams: *const [6]Team, name: []const u8) usize {
    for (teams, 0..) |t, i| {
        if (std.mem.eql(u8, t.name, name))
            return i;
    }
}
fn less(_: void, a: Team, b: Team) bool {
    if (a.points != b.points)
        return a.points > b.points;
    const ag = a.gf - a.ga;
    const bg = b.gf - b.ga;
    if (ag != bg)
        return ag > bg;
    if (a.gf != b.gf)
        return a.gf > b.gf;
    return std.mem.order(u8, a.name, b.name) == .lt;
}
    var teams = [_]Team{
        .{ .name = "Lions" },  .{ .name = "Tigers" }, .{ .name = "Eagles" }, .{ .name = "Sharks" },
        .{ .name = "Wolves" }, .{ .name = "Bears" },
    };
    const matches = [_][]const u8{
        "Lions 0-3 Bears",   "Tigers 2-2 Wolves", "Eagles 4-0 Sharks", "Bears 0-1 Sharks",
        "Wolves 2-1 Eagles", "Lions 0-1 Tigers",  "Tigers 3-3 Bears",  "Eagles 2-2 Lions",
        "Sharks 2-2 Wolves", "Bears 3-1 Wolves",  "Lions 2-0 Sharks",  "Tigers 2-1 Eagles",
        "Eagles 1-0 Bears",  "Sharks 2-0 Tigers", "Wolves 2-1 Lions",
    };
    for (matches) |match| {
        var fields = std.mem.splitScalar(u8, match, ' ');
        const home = find(&teams, fields.next().?);
        var goals = std.mem.splitScalar(u8, fields.next().?, '-');
        const h = try std.fmt.parseInt(i32, goals.next().?, 10);
        const a = try std.fmt.parseInt(i32, goals.next().?, 10);
        const away = find(&teams, fields.next().?);
        teams[home].played += 1;
        teams[away].played += 1;
        teams[home].gf += h;
        teams[home].ga += a;
        teams[away].gf += a;
        teams[away].ga += h;
        if (h > a) {
            teams[home].wins += 1;
            teams[away].losses += 1;
            teams[home].points += 3;
        } else if (h < a) {
            teams[away].wins += 1;
            teams[home].losses += 1;
            teams[away].points += 3;
        } else {
            teams[home].draws += 1;
            teams[away].draws += 1;
            teams[home].points += 1;
            teams[away].points += 1;
        }
    }
    std.mem.sort(Team, &teams, {}, less);
