1const std = @import("std");
2
3const Team = struct {
4 name: []const u8,
5 played: u32 = 0,
6 wins: u32 = 0,
7 draws: u32 = 0,
8 losses: u32 = 0,
9 gf: i32 = 0,
10 ga: i32 = 0,
11 points: u32 = 0,
12};
13
14fn find(teams: *const [6]Team, name: []const u8) usize {
15 for (teams, 0..) |t, i| {
16 if (std.mem.eql(u8, t.name, name))
17 return i;
18 }
19 unreachable;
20}
21
22fn less(_: void, a: Team, b: Team) bool {
23 if (a.points != b.points)
24 return a.points > b.points;
25 const ag = a.gf - a.ga;
26 const bg = b.gf - b.ga;
27 if (ag != bg)
28 return ag > bg;
29 if (a.gf != b.gf)
30 return a.gf > b.gf;
31 return std.mem.order(u8, a.name, b.name) == .lt;
32}

34pub fn main(init: std.process.Init) !void {
35 var output_buffer: [8192]u8 = undefined;
36 var writer = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
37 const out = &writer.interface;
38 var teams = [_]Team{
39 .{ .name = "Lions" }, .{ .name = "Tigers" }, .{ .name = "Eagles" }, .{ .name = "Sharks" },
40 .{ .name = "Wolves" }, .{ .name = "Bears" },
41 };
42 const matches = [_][]const u8{
43 "Lions 0-3 Bears", "Tigers 2-2 Wolves", "Eagles 4-0 Sharks", "Bears 0-1 Sharks",
44 "Wolves 2-1 Eagles", "Lions 0-1 Tigers", "Tigers 3-3 Bears", "Eagles 2-2 Lions",
45 "Sharks 2-2 Wolves", "Bears 3-1 Wolves", "Lions 2-0 Sharks", "Tigers 2-1 Eagles",
46 "Eagles 1-0 Bears", "Sharks 2-0 Tigers", "Wolves 2-1 Lions",
47 };
48 for (matches) |match| {
49 var fields = std.mem.splitScalar(u8, match, ' ');
50 const home = find(&teams, fields.next().?);
51 var goals = std.mem.splitScalar(u8, fields.next().?, '-');
52 const h = try std.fmt.parseInt(i32, goals.next().?, 10);
53 const a = try std.fmt.parseInt(i32, goals.next().?, 10);
54 const away = find(&teams, fields.next().?);
55 teams[home].played += 1;
56 teams[away].played += 1;
57 teams[home].gf += h;
58 teams[home].ga += a;
59 teams[away].gf += a;
60 teams[away].ga += h;
61 if (h > a) {
62 teams[home].wins += 1;
63 teams[away].losses += 1;
64 teams[home].points += 3;
65 } else if (h < a) {
66 teams[away].wins += 1;
67 teams[home].losses += 1;
68 teams[away].points += 3;
69 } else {
70 teams[home].draws += 1;
71 teams[away].draws += 1;
72 teams[home].points += 1;
73 teams[away].points += 1;
74 }
75 }
76 std.mem.sort(Team, &teams, {}, less);
77 for (teams, 0..) |t, i| {
78 const gd = t.gf - t.ga;
79 try out.print("{d}. {s} P={d} W={d} D={d} L={d} GF={d} GA={d} GD={s}{d} Pts={d}\n", .{
80 i + 1, t.name, t.played, t.wins, t.draws, t.losses, t.gf, t.ga,
81 if (gd > 0) @as([]const u8, "+") else "", gd, t.points,
82 });
83 }
84 try out.flush();
85}
