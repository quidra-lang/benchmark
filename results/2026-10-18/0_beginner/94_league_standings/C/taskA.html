1#include <stdio.h>
2#include <stdlib.h>
3#include <string.h>
4
5typedef struct {
6 const char *name;
7 int played, wins, draws, losses, gf, ga, points;
8} Team;
9
10void record(Team *t, int gf, int ga) {
11 t->played++;
12 t->gf += gf;
13 t->ga += ga;
14 if (gf > ga) {
15 t->wins++;
16 t->points += 3;
17 } else if (gf == ga) {
18 t->draws++;
19 t->points++;
20 } else {
21 t->losses++;
22 }
23}
24
25int compare(const void *a, const void *b) {
26 const Team *x = a, *y = b;
27 if (x->points != y->points)
28 return y->points - x->points;
29 if (x->gf - x->ga != y->gf - y->ga)
30 return (y->gf - y->ga) - (x->gf - x->ga);
31 if (x->gf != y->gf)
32 return y->gf - x->gf;
33 return strcmp(x->name, y->name);
34}

36int main(void) {
37 const char *names[] = {"Lions", "Tigers", "Eagles", "Sharks", "Wolves", "Bears"};
38 const char *matches[] = {

41 "Lions 2-0 Sharks", "Tigers 2-1 Eagles", "Eagles 1-0 Bears", "Sharks 2-0 Tigers", "Wolves 2-1 Lions"};
42 Team teams[6];
43 for (int i = 0; i < 6; i++)
44 teams[i] = (Team){.name = names[i]};

46 char home[32], away[32];

51 x++;
52 while (strcmp(teams[y].name, away))
53 y++;
54 record(&teams[x], h, a);
55 record(&teams[y], a, h);
56 }
57 qsort(teams, 6, sizeof(Team), compare);
58 for (int i = 0; i < 6; i++) {
59 Team t = teams[i];
