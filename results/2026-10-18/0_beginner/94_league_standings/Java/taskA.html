1import java.util.ArrayList;
2import java.util.Arrays;
3import java.util.Comparator;
4import java.util.List;
5
6public class Main {
7 public static void main(String[] args) {
8 String[] matches = {
9 "Lions 0-3 Bears", "Tigers 2-2 Wolves", "Eagles 4-0 Sharks", "Bears 0-1 Sharks", "Wolves 2-1 Eagles",
10 "Lions 0-1 Tigers", "Tigers 3-3 Bears", "Eagles 2-2 Lions", "Sharks 2-2 Wolves", "Bears 3-1 Wolves",
11 "Lions 2-0 Sharks", "Tigers 2-1 Eagles", "Eagles 1-0 Bears", "Sharks 2-0 Tigers", "Wolves 2-1 Lions"};
12 String[] names = {"Lions", "Tigers", "Eagles", "Sharks", "Wolves", "Bears"};
13 int[][] stats = new int[6][6];
14 for (String match : matches) {
15 String[] parts = match.split(" "), goals = parts[1].split("-");
16 int home = Arrays.asList(names).indexOf(parts[0]), away = Arrays.asList(names).indexOf(parts[2]);
17 int h = Integer.parseInt(goals[0]), a = Integer.parseInt(goals[1]);
18 stats[home][0]++;
19 stats[away][0]++;
20 stats[home][4] += h;
21 stats[home][5] += a;
22 stats[away][4] += a;
23 stats[away][5] += h;
24 if (h == a) {
25 stats[home][2]++;
26 stats[away][2]++;
27 } else {
28 stats[h > a ? home : away][1]++;
29 stats[h > a ? away : home][3]++;
30 }
31 }
32 List<Integer> order = new ArrayList<>(List.of(0, 1, 2, 3, 4, 5));
33 order.sort(Comparator.<Integer>comparingInt(i -> 3 * stats[i][1] + stats[i][2]).reversed()
34 .thenComparing(Comparator.<Integer>comparingInt(i -> stats[i][4] - stats[i][5]).reversed())
35 .thenComparing(Comparator.<Integer>comparingInt(i -> stats[i][4]).reversed())
36 .thenComparing(i -> names[i]));
37 for (int position = 0; position < 6; position++) {
38 int i = order.get(position), gd = stats[i][4] - stats[i][5];
39 String difference = (gd > 0 ? "+" : "") + gd;
40 System.out.println((position + 1) + ". " + names[i] + " P=" + stats[i][0] + " W=" + stats[i][1]
41 + " D=" + stats[i][2] + " L=" + stats[i][3] + " GF=" + stats[i][4] + " GA=" + stats[i][5]
42 + " GD=" + difference + " Pts=" + (3 * stats[i][1] + stats[i][2]));
43 }
44 }
45}
