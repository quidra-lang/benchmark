1#[derive(Default)]
2struct Team {
3 name: String,
4 played: i32,
5 wins: i32,
6 draws: i32,
7 losses: i32,
8 gf: i32,
9 ga: i32,
10 points: i32,
11}
12
13fn record(t: &mut Team, gf: i32, ga: i32) {
14 t.played += 1;
15 t.gf += gf;
16 t.ga += ga;
17 if gf > ga {
18 t.wins += 1;
19 t.points += 3;
20 } else if gf == ga {
21 t.draws += 1;
22 t.points += 1;
23 } else {
24 t.losses += 1;
25 }
26}
27
28fn main() {
29 let mut teams: Vec<Team> = ["Lions", "Tigers", "Eagles", "Sharks", "Wolves", "Bears"]
30 .iter()
31 .map(|&name| Team {
32 name: name.to_string(),
33 ..Team::default()
34 })
35 .collect();
36 for game in [
37 "Lions 0-3 Bears",
38 "Tigers 2-2 Wolves",
39 "Eagles 4-0 Sharks",
40 "Bears 0-1 Sharks",
41 "Wolves 2-1 Eagles",
42 "Lions 0-1 Tigers",
43 "Tigers 3-3 Bears",
44 "Eagles 2-2 Lions",
45 "Sharks 2-2 Wolves",
46 "Bears 3-1 Wolves",
47 "Lions 2-0 Sharks",
48 "Tigers 2-1 Eagles",
49 "Eagles 1-0 Bears",
50 "Sharks 2-0 Tigers",
51 "Wolves 2-1 Lions",
52 ] {
53 let parts: Vec<&str> = game.split(' ').collect();
54 let goals: Vec<i32> = parts[1].split('-').map(|n| n.parse().unwrap()).collect();
55 let x = teams.iter().position(|t| t.name == parts[0]).unwrap();
56 let y = teams.iter().position(|t| t.name == parts[2]).unwrap();
57 record(&mut teams[x], goals[0], goals[1]);
58 record(&mut teams[y], goals[1], goals[0]);
59 }
60 teams.sort_by(|a, b| {
61 b.points
62 .cmp(&a.points)
63 .then((b.gf - b.ga).cmp(&(a.gf - a.ga)))
64 .then(b.gf.cmp(&a.gf))
65 .then(a.name.cmp(&b.name))
66 });
67 for (i, t) in teams.iter().enumerate() {
68 let gd = t.gf - t.ga;
69 println!(
70 "{}. {} P={} W={} D={} L={} GF={} GA={} GD={}{} Pts={}",
71 i + 1,
72 t.name,
73 t.played,
74 t.wins,
75 t.draws,
76 t.losses,
77 t.gf,
78 t.ga,
79 if gd > 0 { "+" } else { "" },
80 gd,
81 t.points
82 );
83 }
84}
