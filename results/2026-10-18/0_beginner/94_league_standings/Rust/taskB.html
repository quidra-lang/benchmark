1#[derive(Default)]
2struct Team {
3 name: String,
4 played: i32,
5 wins: i32,
6 draws: i32,
7 losses: i32,

9 ga: i32,

12fn record(t: &mut Team, gf: i32, ga: i32) {
13 t.played += 1;
14 t.gf += gf;
15 t.ga += ga;
16 if gf > ga {
17 t.wins += 1;
18 t.points += 3;
19 } else if gf == ga {
20 t.draws += 1;
21 t.points += 1;
22 } else {
23 t.losses += 1;
24 }
25}
