teams = ["Lions", "Tigers", "Eagles", "Sharks", "Wolves", "Bears"]
matches = [
    "Tigers 2-2 Wolves",
    "Wolves 2-1 Eagles",
    "Lions 0-1 Tigers",
    "Tigers 3-3 Bears",
    "Sharks 2-2 Wolves",
    "Bears 3-1 Wolves",
    "Tigers 2-1 Eagles",
    "Sharks 2-0 Tigers",
    "Wolves 2-1 Lions",
]
fields = ["played", "wins", "draws", "losses", "gf", "ga", "points"]
stats = {name: dict.fromkeys(fields, 0) for name in teams}
for match in matches:
    home, score, away = match.split()
    h, a = map(int, score.split("-"))
    for name, goals, conceded in [(home, h, a), (away, a, h)]:
        record = stats[name]
        record["played"] += 1
        record["gf"] += goals
        record["ga"] += conceded
        if goals > conceded:
            record["wins"] += 1
            record["points"] += 3
        elif goals == conceded:
            record["draws"] += 1
            record["points"] += 1
        else:
            record["losses"] += 1
def ranking_key(name):
    s = stats[name]
    return (-s["points"], s["ga"] - s["gf"], -s["gf"], name)
ordered = sorted(teams, key=ranking_key)
