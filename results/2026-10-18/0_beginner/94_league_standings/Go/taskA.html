1package main
2
3import (
4 "fmt"
5 "sort"
6)
7
8type Team struct {
9 name string
10 played, wins, draws, losses, gf, ga, points int
11}
12
13func record(t *Team, gf, ga int) {
14 t.played++
15 t.gf += gf
16 t.ga += ga
17 if gf > ga {
18 t.wins++
19 t.points += 3
20 } else if gf == ga {
21 t.draws++
22 t.points++
23 } else {
24 t.losses++
25 }
26}
27
28func main() {
29 teams := []Team{}
30 for _, name := range []string{"Lions", "Tigers", "Eagles", "Sharks", "Wolves", "Bears"} {
31 teams = append(teams, Team{name: name})
32 }
33 matches := []string{
34 "Lions 0-3 Bears", "Tigers 2-2 Wolves", "Eagles 4-0 Sharks", "Bears 0-1 Sharks", "Wolves 2-1 Eagles",
35 "Lions 0-1 Tigers", "Tigers 3-3 Bears", "Eagles 2-2 Lions", "Sharks 2-2 Wolves", "Bears 3-1 Wolves",
36 "Lions 2-0 Sharks", "Tigers 2-1 Eagles", "Eagles 1-0 Bears", "Sharks 2-0 Tigers", "Wolves 2-1 Lions",
37 }
38 for _, game := range matches {
39 var home, away string
40 var h, a int
41 fmt.Sscanf(game, "%s %d-%d %s", &home, &h, &a, &away)
42 x, y := 0, 0
43 for teams[x].name != home {
44 x++
45 }
46 for teams[y].name != away {
47 y++
48 }
49 record(&teams[x], h, a)
50 record(&teams[y], a, h)
51 }
52 sort.Slice(teams, func(i, j int) bool {
53 a, b := teams[i], teams[j]
54 if a.points != b.points {
55 return a.points > b.points
56 }
57 if a.gf-a.ga != b.gf-b.ga {
58 return a.gf-a.ga > b.gf-b.ga
59 }
60 if a.gf != b.gf {
61 return a.gf > b.gf
62 }
63 return a.name < b.name
64 })
65 for i, t := range teams {
66 gd := t.gf - t.ga
67 sign := ""
68 if gd > 0 {
69 sign = "+"
70 }
71 fmt.Printf("%d. %s P=%d W=%d D=%d L=%d GF=%d GA=%d GD=%s%d Pts=%d\n", i+1, t.name, t.played, t.wins,
72 t.draws, t.losses, t.gf, t.ga, sign, gd, t.points)
73 }
74}
