1struct TeamStats {
2 var played = 0, wins = 0, draws = 0, losses = 0
3 var goalsFor = 0, goalsAgainst = 0
4 var points: Int { 3 * wins + draws }
5 var goalDifference: Int { goalsFor - goalsAgainst }
6}
7
8let matches = [
9 "Lions 0-3 Bears", "Tigers 2-2 Wolves", "Eagles 4-0 Sharks", "Bears 0-1 Sharks", "Wolves 2-1 Eagles",
10 "Lions 0-1 Tigers", "Tigers 3-3 Bears", "Eagles 2-2 Lions", "Sharks 2-2 Wolves", "Bears 3-1 Wolves",
11 "Lions 2-0 Sharks", "Tigers 2-1 Eagles", "Eagles 1-0 Bears", "Sharks 2-0 Tigers", "Wolves 2-1 Lions",
12]
13let names = ["Lions", "Tigers", "Eagles", "Sharks", "Wolves", "Bears"]
14var stats = [TeamStats](repeating: TeamStats(), count: names.count)
15for match in matches {
16 let parts = match.split(separator: " ").map(String.init)
17 let goals = parts[1].split(separator: "-").map { Int($0)! }
18 let home = names.firstIndex(of: parts[0])!, away = names.firstIndex(of: parts[2])!
19 let h = goals[0], a = goals[1]
20 stats[home].played += 1
21 stats[away].played += 1
22 stats[home].goalsFor += h
23 stats[home].goalsAgainst += a
24 stats[away].goalsFor += a
25 stats[away].goalsAgainst += h
26 if h == a {
27 stats[home].draws += 1
28 stats[away].draws += 1
29 } else {
30 stats[h > a ? home : away].wins += 1
31 stats[h > a ? away : home].losses += 1
32 }
33}
34let order = names.indices.sorted { a, b in
35 if stats[a].points != stats[b].points { return stats[a].points > stats[b].points }
36 if stats[a].goalDifference != stats[b].goalDifference { return stats[a].goalDifference > stats[b].goalDifference }
37 if stats[a].goalsFor != stats[b].goalsFor { return stats[a].goalsFor > stats[b].goalsFor }
38 return names[a] < names[b]
39}
40for (position, i) in order.enumerated() {
41 let record = stats[i], difference = (record.goalDifference > 0 ? "+" : "") + String(record.goalDifference)
42 print("\(position + 1). \(names[i]) P=\(record.played) W=\(record.wins) D=\(record.draws) " +
43 "L=\(record.losses) GF=\(record.goalsFor) GA=\(record.goalsAgainst) GD=\(difference) Pts=\(record.points)")
44}
