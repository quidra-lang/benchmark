	items := []struct {
		name                      string
		price, quantity, category int
	}{
		{"milk", 198, 2, 0},
		{"notebook", 165, 3, 1},
		{"bread", 248, 1, 0},
		{"apple", 128, 4, 0},
		{"detergent", 437, 1, 1},
		{"eggs", 268, 1, 0},
	}
	totals := [2]int{}
	for _, item := range items {
		subtotal := item.price * item.quantity
		totals[item.category] += subtotal
		fmt.Printf("%s %d x %d = %d\n", item.name, item.price, item.quantity, subtotal)
	}
	foodTax, generalTax := totals[0]*8/100, totals[1]*10/100
	fmt.Printf("food: %d yen, tax %d yen\ngeneral: %d yen, tax %d yen\ntotal: %d yen\n", totals[0],
		foodTax, totals[1], generalTax, totals[0]+totals[1]+foodTax+generalTax)
