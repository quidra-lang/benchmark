    const char *names[] = {"milk", "notebook", "bread", "apple", "detergent", "eggs"};
    int prices[] = {198, 165, 248, 128, 437, 268}, quantities[] = {2, 3, 1, 4, 1, 1};
    int category[] = {0, 1, 0, 0, 1, 0}, totals[2] = {0}, taxes[2];
    for (int i = 0; i < 6; i++) {
        int subtotal = prices[i] * quantities[i];
        totals[category[i]] += subtotal;
        printf("%s %d x %d = %d\n", names[i], prices[i], quantities[i], subtotal);
    }
    taxes[0] = totals[0] * 8 / 100;
    taxes[1] = totals[1] * 10 / 100;
    printf("food: %d yen, tax %d yen\ngeneral: %d yen, tax %d yen\ntotal: %d yen\n", totals[0], taxes[0],
        totals[1], taxes[1], totals[0] + totals[1] + taxes[0] + taxes[1]);
