1using System;
2using System.Collections.Generic;
3using System.Globalization;
4
5CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
6
7int[] values = { 3, 8, 11, 15, 19, 24, 27, 31, 36, 42, 47, 53, 58, 64, 70, 77 };
8foreach (int target in new[] { 42, 3, 77, 31, 50, 1 })
9{
10 int lo = 0, hi = 15, found = -1;
11 var probes = new List<int>();
12 while (lo <= hi)
13 {
14 int mid = (lo + hi) / 2;
15 probes.Add(mid);
16 if (values[mid] == target)
17 {
18 found = mid;
19 break;
20 }
21 if (values[mid] < target)
22 lo = mid + 1;
23 else
24 hi = mid - 1;
25 }
26 string result = found >= 0 ? "index " + found : "not found";
27 Console.WriteLine($"{target}: probes {string.Join(" ", probes)} -> {result}");
28}
