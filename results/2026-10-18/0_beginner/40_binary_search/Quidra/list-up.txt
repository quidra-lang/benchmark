int[] values = [3, 8, 11, 15, 19, 24, 27, 31, 36, 42, 47, 53, 58, 64, 70, 77]
int[] targets = [42, 3, 77, 31, 50, 1]
for int target = targets[]
    int lo = 0
    int hi = 15
    int found = -1
    print("{target}: probes")
    while lo <= hi
        int mid = (lo + hi) / 2
        print(" {mid}")
        if values[mid] == target
            found = mid
        elif values[mid] < target
            lo = mid + 1
        else
            hi = mid - 1
    if found >= 0
    else
        print(" -> not found{NL}")
