    lo, hi, found = 0, len(values) - 1, -1
    probes = []
    while lo <= hi:
        mid = (lo + hi) // 2
        probes.append(mid)

        if values[mid] < target:
            lo = mid + 1

            hi = mid - 1
    result = f"index {found}" if found >= 0 else "not found"
    print(f"{target}: probes {' '.join(map(str, probes))} -> {result}")
