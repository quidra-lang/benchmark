3fn solve(n: usize, row: usize, queens: *[8]usize, count: *u32, first: *[8]usize) void {
4 if (row == n) {
5 count.* += 1;
6 if (n == 8 and count.* == 1)
7 first.* = queens.*;
8 return;
9 }
10 for (0..n) |col| {
11 var safe = true;
12 for (0..row) |r| {
13 const delta = if (queens[r] > col) queens[r] - col else col - queens[r];
14 if (queens[r] == col or delta == row - r) {
15 safe = false;
16 break;
17 }
18 }
19 if (safe) {
20 queens[row] = col;
21 solve(n, row + 1, queens, count, first);
22 }
23 }
24}
