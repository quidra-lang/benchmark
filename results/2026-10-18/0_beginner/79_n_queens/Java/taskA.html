4 static int queens(int[] positions, int row, int[] first) {
5 if (row == positions.length) {
6 if (first[0] == -1) {
7 System.arraycopy(positions, 0, first, 0, positions.length);
8 }
9 return 1;
10 }
11 if (row == 0) {
12 Arrays.fill(first, -1);
13 }
14 int count = 0;
15 for (int column = 0; column < positions.length; column++) {
16 boolean safe = true;
17 for (int previous = 0; previous < row; previous++) {
18 if (positions[previous] == column || Math.abs(positions[previous] - column) == row - previous) {
19 safe = false;
20 break;
21 }
22 }
23 if (safe) {
24 positions[row] = column;
25 count += queens(positions, row + 1, first);
26 }
27 }
28 return count;
29 }
