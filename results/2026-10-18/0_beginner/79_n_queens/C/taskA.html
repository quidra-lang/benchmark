5void search(int n, int row, int *positions, int *count, int *first) {
6 if (row == n) {
7 if (n == 8 && *count == 0)
8 memcpy(first, positions, 8 * sizeof(int));
9 (*count)++;
10 return;
11 }
12 for (int col = 0; col < n; col++) {
13 int safe = 1;
14 for (int r = 0; r < row; r++)
15 if (positions[r] == col || abs(positions[r] - col) == row - r) {
16 safe = 0;
17 break;
18 }
19 if (safe) {
20 positions[row] = col;
21 search(n, row + 1, positions, count, first);
22 }
23 }
24}
