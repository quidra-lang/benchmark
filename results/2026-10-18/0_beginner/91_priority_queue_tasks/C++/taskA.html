1#include <iostream>
2#include <string>
3#include <vector>
4#include <array>
5#include <queue>
6#include <sstream>
7
8struct Task {
9 std::string name;
10 int priority, order;
11};
12
13struct Later {
14 bool operator()(const Task &a, const Task &b) const {
15 return a.priority != b.priority ? a.priority > b.priority : a.order > b.order;
16 }
17};
18
19int main() {
20 std::priority_queue<Task, std::vector<Task>, Later> queue;
21 int sequence = 0;
22 const std::array<std::string, 16> operations = {"add email 3", "add backup 5", "add deploy 1", "run",
23 "add report 3", "add test 2", "add lint 3", "run", "run", "add cleanup 5", "add review 2", "run",
24 "add build 1", "add docs 4", "run", "run"};
25 for (const std::string &operation : operations) {
26 if (operation == "run") {
27 Task task = queue.top();
28 queue.pop();
29 std::cout << "run: " << task.name << " (priority " << task.priority << ")\n";
30 } else {
31 std::istringstream input(operation);
32 std::string op;
33 Task task;
34 input >> op >> task.name >> task.priority;
35 task.order = sequence++;
36 queue.push(task);
37 }
38 }
39 std::cout << "remaining:";
40 while (!queue.empty()) {
41 std::cout << ' ' << queue.top().name;
42 queue.pop();
43 }
44 std::cout << '\n';
45}
