7
8type Task struct {
9 name string
10 priority, order int
11}
12
13type TaskQueue []Task
14
15func (q TaskQueue) Len() int { return len(q) }
16
17func (q TaskQueue) Less(i, j int) bool {
18 if q[i].priority != q[j].priority {
19 return q[i].priority < q[j].priority
20 }
21 return q[i].order < q[j].order
22}
23
24func (q TaskQueue) Swap(i, j int) { q[i], q[j] = q[j], q[i] }
25
26func (q *TaskQueue) Push(value any) {
27 *q = append(*q, value.(Task))
28}
29
30func (q *TaskQueue) Pop() any {
31 last := (*q)[len(*q)-1]
32 *q = (*q)[:len(*q)-1]
33 return last
34}
35
36func main() {
37 queue := TaskQueue{}
38 sequence := 0
39 operations := []string{"add email 3", "add backup 5", "add deploy 1", "run", "add report 3",
40 "add test 2", "add lint 3", "run", "run", "add cleanup 5", "add review 2", "run", "add build 1",
41 "add docs 4", "run", "run"}
42 for _, operation := range operations {
43 if operation == "run" {
44 task := heap.Pop(&queue).(Task)
45 fmt.Printf("run: %s (priority %d)\n", task.name, task.priority)
46 } else {
47 task := Task{order: sequence}
48 fmt.Sscanf(operation, "add %s %d", &task.name, &task.priority)
49 sequence++
50 heap.Push(&queue, task)
51 }
52 }
53 fmt.Print("remaining:")
54 for len(queue) > 0 {
55 task := heap.Pop(&queue).(Task)
56 fmt.Printf(" %s", task.name)
57 }
58 fmt.Println()
59}
