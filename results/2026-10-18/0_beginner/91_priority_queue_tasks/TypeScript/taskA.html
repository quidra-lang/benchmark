4const less = (a: Task, b: Task): boolean =>

7function push(task: Task): void {
8 heap.push(task);
9 let index = heap.length - 1;
10 while (index > 0) {
11 const parent = Math.floor((index - 1) / 2);
12 if (!less(heap[index], heap[parent])) break;
13 [heap[index], heap[parent]] = [heap[parent], heap[index]];
14 index = parent;
15 }
16}

18function pop(): Task {
19 const first = heap[0], last = heap.pop()!;
20 if (heap.length > 0) {
21 heap[0] = last;
22 let index = 0;
23 while (2 * index + 1 < heap.length) {
24 let child = 2 * index + 1;
25 if (child + 1 < heap.length && less(heap[child + 1], heap[child])) child++;
26 if (!less(heap[child], heap[index])) break;
27 [heap[index], heap[child]] = [heap[child], heap[index]];
28 index = child;
29 }
30 }
31 return first;
32}
