1import java.util.PriorityQueue
2
3data class Task(val name: String, val priority: Int, val sequence: Int)
4
5fun main() {
6 val operations = listOf("add email 3", "add backup 5", "add deploy 1", "run", "add report 3", "add test 2",
7 "add lint 3", "run", "run", "add cleanup 5", "add review 2", "run", "add build 1",
8 "add docs 4", "run", "run")
9 val queue = PriorityQueue<Task>(compareBy<Task> { it.priority }.thenBy { it.sequence })
10 var sequence = 0
11 for (operation in operations) {
12 val parts = operation.split(' ')
13 if (parts[0] == "add") {
14 queue.add(Task(parts[1], parts[2].toInt(), sequence++))
15 } else {
16 val task = queue.remove()
17 println("run: ${task.name} (priority ${task.priority})")
18 }
19 }
20 println("remaining: ${queue.sortedWith(queue.comparator()).joinToString(" ") { it.name }}")
21}
