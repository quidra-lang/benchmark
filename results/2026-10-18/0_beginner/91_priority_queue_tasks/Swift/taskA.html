1struct Task {
2 let name: String
3 let priority: Int
4 let sequence: Int
5
6 func precedes(_ other: Task) -> Bool {
7 priority == other.priority ? sequence < other.sequence : priority < other.priority
8 }
9}
10
11struct Heap {
12 var tasks = [Task]()
13
14 mutating func push(_ task: Task) {
15 tasks.append(task)
16 var i = tasks.count - 1
17 while i > 0 {
18 let parent = (i - 1) / 2
19 if !tasks[i].precedes(tasks[parent]) { break }
20 tasks.swapAt(i, parent)
21 i = parent
22 }
23 }
24
25 mutating func pop() -> Task {
26 let result = tasks[0], last = tasks.removeLast()
27 if tasks.isEmpty { return result }
28 tasks[0] = last
29 var i = 0
30 while 2 * i + 1 < tasks.count {
31 var child = 2 * i + 1
32 if child + 1 < tasks.count && tasks[child + 1].precedes(tasks[child]) { child += 1 }
33 if !tasks[child].precedes(tasks[i]) { break }
34 tasks.swapAt(i, child)
35 i = child
36 }
37 return result
38 }
39}
