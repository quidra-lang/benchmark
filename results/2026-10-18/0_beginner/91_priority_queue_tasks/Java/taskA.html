1import java.util.Comparator;
2import java.util.List;
3import java.util.PriorityQueue;
4
5public class Main {
6 record Task(String name, int priority, int sequence) {}
7
8 public static void main(String[] args) {
9 String[] operations = {"add email 3", "add backup 5", "add deploy 1", "run", "add report 3", "add test 2",
10 "add lint 3", "run", "run", "add cleanup 5", "add review 2", "run", "add build 1",
11 "add docs 4", "run", "run"};
12 PriorityQueue<Task> queue = new PriorityQueue<>(
13 Comparator.comparingInt(Task::priority).thenComparingInt(Task::sequence));
14 int sequence = 0;
15 for (String operation : operations) {
16 String[] parts = operation.split(" ");
17 if (parts[0].equals("add")) {
18 queue.add(new Task(parts[1], Integer.parseInt(parts[2]), sequence++));
19 } else {
20 Task task = queue.remove();
21 System.out.println("run: " + task.name() + " (priority " + task.priority() + ")");
22 }
23 }
24 List<String> remaining = queue.stream().sorted(queue.comparator()).map(Task::name).toList();
25 System.out.println("remaining: " + String.join(" ", remaining));
26 }
27}
