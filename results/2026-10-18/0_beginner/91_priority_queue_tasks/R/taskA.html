6heap_push <- function(task) {
7 items <- heap$items
8 items[[length(items) + 1]] <- task
9 child <- length(items)
10 while (child > 1) {
11 parent <- child %/% 2
12 if (!before(items[[child]], items[[parent]])) break
13 items[c(child, parent)] <- items[c(parent, child)]
14 child <- parent
15 }
16 heap$items <- items
17}

19heap_pop <- function() {
20 items <- heap$items
21 task <- items[[1]]
22 last <- tail(items, 1)[[1]]
23 items <- head(items, -1)
24 if (length(items) > 0) {
25 items[[1]] <- last
26 parent <- 1
27 while (2 * parent <= length(items)) {
28 child <- 2 * parent
29 if (child < length(items) && before(items[[child + 1]], items[[child]])) child <- child + 1
30 if (!before(items[[child]], items[[parent]])) break
31 items[c(child, parent)] <- items[c(parent, child)]
32 parent <- child
33 }
34 }
35 heap$items <- items
36 task
37}
