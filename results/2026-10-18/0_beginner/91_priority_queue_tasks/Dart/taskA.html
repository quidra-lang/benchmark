1class Task {
2 final String name;
3 final int priority, sequence;
4 Task(this.name, this.priority, this.sequence);
5}
6
7bool before(Task a, Task b) => a.priority != b.priority
8 ? a.priority < b.priority
9 : a.sequence < b.sequence;
10
11class Heap {
12 final items = <Task>[];
13 void push(Task task) {
14 var i = items.length;
15 items.add(task);
16 while (i > 0) {
17 final p = (i - 1) ~/ 2;
18 if (!before(items[i], items[p])) break;
19 final current = items[i];
20 items[i] = items[p];
21 items[p] = current;
22 i = p;
23 }
24 }
25
26 Task pop() {
27 final result = items.first, last = items.removeLast();
28 if (items.isNotEmpty) {
29 items[0] = last;
30 var i = 0;
31 while (2 * i + 1 < items.length) {
32 var child = 2 * i + 1;
33 if (child + 1 < items.length &&
34 before(items[child + 1], items[child])) {
35 child++;
36 }
37 if (!before(items[child], items[i])) break;
38 final current = items[i];
39 items[i] = items[child];
40 items[child] = current;
41 i = child;
42 }
43 }
44 return result;
45 }
46}

48void main() {
49 const operations = [

64 'run',
65 'run',
66 ];
67 final heap = Heap();
68 var sequence = 0;
69 for (final operation in operations) {
70 final tokens = operation.split(' ');
71 if (tokens[0] == 'add') {
72 heap.push(Task(tokens[1], int.parse(tokens[2]), sequence++));
73 } else {
74 final task = heap.pop();
75 print('run: ${task.name} (priority ${task.priority})');
76 }
77 }
78 final remaining = <String>[];
79 while (heap.items.isNotEmpty) {
80 remaining.add(heap.pop().name);
81 }
82 print('remaining: ${remaining.join(' ')}');
83}
