4typedef struct {
5 char name[32];
6 int priority, order;
7} Task;
8
9int before(Task a, Task b) {
10 return a.priority != b.priority ? a.priority < b.priority : a.order < b.order;
11}
12
13void push(Task *heap, int *size, Task task) {
14 int i = (*size)++;
15 while (i > 0) {
16 int p = (i - 1) / 2;
17 if (!before(task, heap[p]))
18 break;
19 heap[i] = heap[p];
20 i = p;
21 }
22 heap[i] = task;
23}
24
25Task pop(Task *heap, int *size) {
26 Task result = heap[0], last = heap[--*size];
27 if (*size) {
28 int i = 0;
29 while (2 * i + 1 < *size) {
30 int c = 2 * i + 1;
31 if (c + 1 < *size && before(heap[c + 1], heap[c]))
32 c++;
33 if (!before(heap[c], last))
34 break;
35 heap[i] = heap[c];
36 i = c;
37 }
38 heap[i] = last;
39 }
40 return result;
41}
