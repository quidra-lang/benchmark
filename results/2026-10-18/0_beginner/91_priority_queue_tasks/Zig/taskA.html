1const std = @import("std");
2
3const Task = struct { name: []const u8, priority: u32, sequence: usize };
4
5fn compare(_: void, a: Task, b: Task) std.math.Order {
6 return if (a.priority != b.priority)
7 std.math.order(a.priority, b.priority)
8 else
9 std.math.order(a.sequence, b.sequence);
10}
11
12pub fn main(init: std.process.Init) !void {
13 var output_buffer: [8192]u8 = undefined;
14 var writer = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
15 const out = &writer.interface;
16 const operations = [_][]const u8{
17 "add email 3",
18 "add backup 5",
19 "add deploy 1",
20 "run",
21 "add report 3",
22 "add test 2",
23 "add lint 3",
24 "run",
25 "run",
26 "add cleanup 5",
27 "add review 2",
28 "run",
29 "add build 1",
30 "add docs 4",
31 "run",
32 "run",
33 };
34 var heap: std.PriorityQueue(Task, void, compare) = .empty;
35 defer heap.deinit(std.heap.page_allocator);
36 var sequence: usize = 0;
37 for (operations) |operation| {
38 var tokens = std.mem.splitScalar(u8, operation, ' ');
39 const op = tokens.next().?;
40 if (std.mem.eql(u8, op, "add")) {
41 const name = tokens.next().?;
42 const priority = try std.fmt.parseInt(u32, tokens.next().?, 10);
43 try heap.push(std.heap.page_allocator, .{ .name = name, .priority = priority, .sequence = sequence });
44 sequence += 1;
45 } else {
46 const task = heap.pop().?;
47 try out.print("run: {s} (priority {d})\n", .{ task.name, task.priority });
48 }
49 }
50 try out.writeAll("remaining:");
51 while (heap.pop()) |task| {
52 try out.print(" {s}", .{task.name});
53 }
54 try out.writeByte('\n');
55 try out.flush();
56}
