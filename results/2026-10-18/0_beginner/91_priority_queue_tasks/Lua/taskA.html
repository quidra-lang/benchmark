1local function before(a, b)
2 return a.priority < b.priority or (a.priority == b.priority and a.sequence < b.sequence)
3end

5local heap = {}

7local function push(task)
8 heap[#heap + 1] = task
9 local child = #heap
10 while child > 1 do
11 local parent = child // 2
12 if not before(heap[child], heap[parent]) then break end
13 heap[child], heap[parent] = heap[parent], heap[child]
14 child = parent
15 end
16end
17
18local function pop()
19 local task = heap[1]
20 local last = table.remove(heap)
21 if #heap > 0 then
22 heap[1] = last
23 local parent = 1
24 while 2 * parent <= #heap do
25 local child = 2 * parent
26 if child < #heap and before(heap[child + 1], heap[child]) then child = child + 1 end
27 if not before(heap[child], heap[parent]) then break end
28 heap[child], heap[parent] = heap[parent], heap[child]
29 parent = child
30 end
31 end
32 return task
33end
