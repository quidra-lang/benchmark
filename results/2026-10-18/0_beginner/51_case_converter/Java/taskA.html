1import java.util.Arrays;
2import java.util.List;
3import java.util.Locale;
4
5public class Main {
6 public static void main(String[] args) {
7 String[] ids = {
8 "user_name", "maxRetryCount", "ShoppingCartItem", "background-color",
9 "get_last_login_time", "isEmailAddressValid"};
10 for (String id : ids) {
11 List<String> words = Arrays.stream(id.split("[_-]|(?=[A-Z])"))
12 .filter(w -> !w.isEmpty())
13 .map(w -> w.toLowerCase(Locale.ROOT))
14 .toList();
15 List<String> capitals = words.stream()
16 .map(w -> w.substring(0, 1).toUpperCase(Locale.ROOT) + w.substring(1))
17 .toList();
18 String camel = words.get(0) + String.join("", capitals.subList(1, capitals.size()));
19 System.out.println(id + ": snake=" + String.join("_", words) + ", camel=" + camel
20 + ", pascal=" + String.join("", capitals) + ", kebab=" + String.join("-", words));
21 }
22 }
23}
