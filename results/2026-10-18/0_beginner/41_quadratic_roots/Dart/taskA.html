1import 'dart:io';
2import 'dart:math' as math;
3
4void main() {
5 final out = StringBuffer();
6 for (final s in [
7 [1, -3, -10],
8 [-2, 5, 4],
9 [3, 1, -2],
10 [4, -12, 9],
11 [1, 2, 5],
12 [-2, 3, -4],
13 ]) {
14 final a = s[0], b = s[1], c = s[2], disc = b * b - 4 * a * c;
15 final numerator = -b.toDouble(), denominator = 2.0 * a;
16 out.write('$a $b $c: ');
17 if (disc > 0) {
18 final r1 = (numerator + math.sqrt(disc)) / denominator,
19 r2 = (numerator - math.sqrt(disc)) / denominator;
20 out.write(
21 'x1 = ${math.max(r1, r2).toStringAsFixed(4)}, x2 = ${math.min(r1, r2).toStringAsFixed(4)}',
22 );
23 } else if (disc == 0) {
24 out.write('x = ${(numerator / denominator).toStringAsFixed(4)}');
25 } else {
26 final p = numerator / denominator, q = math.sqrt(-disc) / (2 * a.abs());
27 out.write(
28 'x = ${p.toStringAsFixed(4)} + ${q.toStringAsFixed(4)}i, ${p.toStringAsFixed(4)} - ${q.toStringAsFixed(4)}i',
29 );
30 }
31 out.writeln();
32 }
33 stdout.write(out);
34}
