import math
for (int a, int b, int c) := coefficients[]
    int d := b * b - 4 * a * c
    real64 denominator := real64(2 * a)
    real64 numerator := real64(-b)
    if d > 0
        real64 root := math.sqrt(real64(d))
        real64 first := (numerator + root) / denominator
        real64 second := (numerator - root) / denominator
        if first < second
            (first, second) = (second, first)
        print.line("x1 = {first:frac=4}, x2 = {second:frac=4}")
