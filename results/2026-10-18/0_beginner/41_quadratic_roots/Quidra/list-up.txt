int[][] coefficients = [[1, -3, -10], [-2, 5, 4], [3, 1, -2], [4, -12, 9], [1, 2, 5], [-2, 3, -4]]
for int[] row = coefficients[]
    int a = row[0]
    int b = row[1]
    int c = row[2]
    int d = b * b - 4 * a * c
    real64 denominator = real64(2 * a)
    if d > 0
        real64 root = math.sqrt(real64(d))
        real64 first = (real64(-b) + root) / denominator
        real64 second = (real64(-b) - root) / denominator
        if first < second
            (first, second) = (second, first)
        print("x1 = {first:frac=4}, x2 = {second:frac=4}{NL}")
