1import java.util.Locale
2import kotlin.math.abs
3import kotlin.math.sqrt
4
5fun main() {
6 val equations = listOf(
7 Triple(1, -3, -10), Triple(-2, 5, 4), Triple(3, 1, -2), Triple(4, -12, 9), Triple(1, 2, 5),
8 Triple(-2, 3, -4),
9 )
10 for ((a, b, c) in equations) {
11 val d = b * b - 4 * a * c
12 val prefix = "$a $b $c: "
13 if (d > 0) {
14 val first = (-b + sqrt(d.toDouble())) / (2 * a)
15 val second = (-b - sqrt(d.toDouble())) / (2 * a)
16 println((prefix + "x1 = %.4f, x2 = %.4f").format(Locale.ROOT, maxOf(first, second), minOf(first, second)))
17 } else if (d == 0) {
18 println((prefix + "x = %.4f").format(Locale.ROOT, -b / (2.0 * a)))
19 } else {
20 val p = -b / (2.0 * a)
21 val q = sqrt(-d.toDouble()) / (2 * abs(a))
22 println((prefix + "x = %.4f + %.4fi, %.4f - %.4fi").format(Locale.ROOT, p, q, p, q))
23 }
24 }
25}
