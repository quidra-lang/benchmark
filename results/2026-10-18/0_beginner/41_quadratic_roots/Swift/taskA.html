1import Foundation
2
3let posix = Locale(identifier: "en_US_POSIX")
4let equations = [(1, -3, -10), (-2, 5, 4), (3, 1, -2), (4, -12, 9), (1, 2, 5), (-2, 3, -4)]
5for (a, b, c) in equations {
6 let d = b * b - 4 * a * c, prefix = "\(a) \(b) \(c): "
7 if d > 0 {
8 let first = (Double(-b) + sqrt(Double(d))) / Double(2 * a)
9 let second = (Double(-b) - sqrt(Double(d))) / Double(2 * a)
10 print(String(format: prefix + "x1 = %.4f, x2 = %.4f", locale: posix, max(first, second), min(first, second)))
11 } else if d == 0 {
12 print(String(format: prefix + "x = %.4f", locale: posix, Double(-b) / Double(2 * a)))
13 } else {
14 let p = Double(-b) / Double(2 * a), q = sqrt(Double(-d)) / Double(2 * abs(a))
15 print(String(format: prefix + "x = %.4f + %.4fi, %.4f - %.4fi", locale: posix, p, q, p, q))
16 }
17}
