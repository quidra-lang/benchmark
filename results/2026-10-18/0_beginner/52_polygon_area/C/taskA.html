1#include <stdio.h>
2#include <stdlib.h>
3#include <math.h>
4
5int main(void) {
6 int polygons[3][7][2] = {{{0, 0}, {5, 0}, {6, 4}, {1, 3}},
7 {{2, 1}, {0, 4}, {3, 7}, {7, 6}, {8, 2}},
8 {{0, 0}, {8, 0}, {8, 6}, {5, 6}, {4, 3}, {3, 6}, {0, 6}}};
9 int sizes[] = {4, 5, 7};
10 for (int k = 0; k < 3; k++) {
11 int twice = 0;
12 double perimeter = 0;
13 for (int i = 0; i < sizes[k]; i++) {
14 int j = (i + 1) % sizes[k];
15 int x = polygons[k][i][0], y = polygons[k][i][1];
16 int nx = polygons[k][j][0], ny = polygons[k][j][1];
17 int dx = x - nx, dy = y - ny;
18 twice += x * ny - nx * y;
19 perimeter += sqrt((double)(dx * dx + dy * dy));
20 }
21 twice = abs(twice);
22 printf("polygon %d: area=%.1f perimeter=%.3f\n", k + 1, twice / 2.0, perimeter);
23 }
24 return 0;
25}
