import math
(int, int)[][] polygons = [
    [(2, 1), (0, 4), (3, 7), (7, 6), (8, 2)],
]
for (int, int)[] polygon = polygons[nat k]
    int cross_sum = 0
    real64 perimeter = 0.0
    for (int x, int y) = polygon[nat i]
        (int next_x, int next_y) = polygon[(i + 1) % len(polygon)]
        cross_sum += x * next_y - next_x * y
        int dx = x - next_x
        int dy = y - next_y
        perimeter += math.sqrt(real64(dx * dx + dy * dy))
    int magnitude = cross_sum.abs()
    real64 area = real64(magnitude)
    area /= 2.0
