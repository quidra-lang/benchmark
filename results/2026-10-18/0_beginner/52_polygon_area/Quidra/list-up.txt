int[][][] polygons = [
    [[0, 0], [5, 0], [6, 4], [1, 3]],
    [[2, 1], [0, 4], [3, 7], [7, 6], [8, 2]],
]
for 0 <= int k < int(len(polygons))
    int[][] polygon = polygons[k]
    int cross_sum = 0
    real64 perimeter = 0.0
    for 0 <= int i < int(len(polygon))
        int j = (i + 1) % int(len(polygon))
        cross_sum += polygon[i][0] * polygon[j][1] - polygon[j][0] * polygon[i][1]
        int dx = polygon[i][0] - polygon[j][0]
        int dy = polygon[i][1] - polygon[j][1]
        perimeter += math.sqrt(real64(dx * dx + dy * dy))
    int magnitude = if cross_sum < 0 then -cross_sum else cross_sum
    real64 area = real64(magnitude) / 2.0
