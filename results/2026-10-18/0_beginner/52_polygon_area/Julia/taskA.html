3function main()
4 polygons = [
5 [(0, 0), (5, 0), (6, 4), (1, 3)],
6 [(2, 1), (0, 4), (3, 7), (7, 6), (8, 2)],
7 [(0, 0), (8, 0), (8, 6), (5, 6), (4, 3), (3, 6), (0, 6)],
8 ]
9 for (number, vertices) in enumerate(polygons)
10 twice_area, perimeter = 0, 0.0
11 for i in eachindex(vertices)
12 x, y = vertices[i]
13 next_x, next_y = vertices[mod1(i+1, length(vertices))]
14 twice_area += x*next_y-next_x*y
15 dx, dy = x-next_x, y-next_y
16 perimeter += sqrt(Float64(dx*dx+dy*dy))
17 end
18 @printf(
19 "polygon %d: area=%.1f perimeter=%.3f\n",
20 number,
21 abs(twice_area)/2,
22 perimeter
23 )
24 end
25end
