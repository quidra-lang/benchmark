1import java.util.Locale
2import kotlin.math.abs
3import kotlin.math.sqrt
4
5fun main() {
6 val polygons = listOf(
7 listOf(0 to 0, 5 to 0, 6 to 4, 1 to 3),
8 listOf(2 to 1, 0 to 4, 3 to 7, 7 to 6, 8 to 2),
9 listOf(0 to 0, 8 to 0, 8 to 6, 5 to 6, 4 to 3, 3 to 6, 0 to 6),
10 )
11 for ((k, vertices) in polygons.withIndex()) {
12 var cross = 0
13 var perimeter = 0.0
14 for (i in vertices.indices) {
15 val (x, y) = vertices[i]
16 val (nx, ny) = vertices[(i + 1) % vertices.size]
17 cross += x * ny - nx * y
18 val dx = x - nx
19 val dy = y - ny
20 perimeter += sqrt((dx * dx + dy * dy).toDouble())
21 }
22 println("polygon %d: area=%.1f perimeter=%.3f".format(Locale.ROOT, k + 1, abs(cross) / 2.0, perimeter))
23 }
24}
