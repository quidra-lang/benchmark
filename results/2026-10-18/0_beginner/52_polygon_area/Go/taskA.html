1package main
2
3import (
4 "fmt"
5 "math"
6)
7
8func main() {
9 polygons := [][][2]int{{{0, 0}, {5, 0}, {6, 4}, {1, 3}}, {{2, 1}, {0, 4}, {3, 7}, {7, 6}, {8, 2}},
10 {{0, 0}, {8, 0}, {8, 6}, {5, 6}, {4, 3}, {3, 6}, {0, 6}}}
11 for k, p := range polygons {
12 twice, perimeter := 0, 0.0
13 for i, v := range p {
14 w := p[(i+1)%len(p)]
15 dx, dy := v[0]-w[0], v[1]-w[1]
16 twice += v[0]*w[1] - w[0]*v[1]
17 perimeter += math.Sqrt(float64(dx*dx + dy*dy))
18 }
19 fmt.Printf("polygon %d: area=%.1f perimeter=%.3f\n",
20 k+1, math.Abs(float64(twice))/2, perimeter)
21 }
22}
