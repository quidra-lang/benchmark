1#include <iostream>
2#include <iomanip>
3#include <vector>
4#include <array>
5#include <cmath>
6
7int main() {
8 std::vector<std::vector<std::array<int, 2>>> polygons = {{{0, 0}, {5, 0}, {6, 4}, {1, 3}},
9 {{2, 1}, {0, 4}, {3, 7}, {7, 6}, {8, 2}}, {{0, 0}, {8, 0}, {8, 6}, {5, 6}, {4, 3}, {3, 6}, {0, 6}}};
10 std::cout << std::fixed;
11 for (std::size_t k = 0; k < polygons.size(); k++) {
12 int twice = 0;
13 double perimeter = 0;
14 const std::vector<std::array<int, 2>> &p = polygons[k];
15 for (std::size_t i = 0; i < p.size(); i++) {
16 std::size_t j = (i + 1) % p.size();
17 int dx = p[i][0] - p[j][0], dy = p[i][1] - p[j][1];
18 twice += p[i][0] * p[j][1] - p[j][0] * p[i][1];
19 perimeter += std::sqrt(dx * dx + dy * dy);
20 }
21 twice = std::abs(twice);
22 std::cout << "polygon " << k + 1 << ": area=" << std::setprecision(1) << twice / 2.0
23 << " perimeter=" << std::setprecision(3) << perimeter << '\n';
24 }
25}
