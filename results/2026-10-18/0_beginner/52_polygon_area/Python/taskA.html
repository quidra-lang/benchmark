3polygons = [
4 [(0, 0), (5, 0), (6, 4), (1, 3)],
5 [(2, 1), (0, 4), (3, 7), (7, 6), (8, 2)],
6 [(0, 0), (8, 0), (8, 6), (5, 6), (4, 3), (3, 6), (0, 6)],

8for index, vertices in enumerate(polygons, 1):
9 cross_sum = 0
10 perimeter = 0.0
11 for i, (x, y) in enumerate(vertices):
12 next_x, next_y = vertices[(i + 1) % len(vertices)]
13 cross_sum += x * next_y - next_x * y
14 dx, dy = x - next_x, y - next_y
15 perimeter += sqrt(dx * dx + dy * dy)
16 print(f"polygon {index}: area={abs(cross_sum) / 2:.1f} perimeter={perimeter:.3f}")
