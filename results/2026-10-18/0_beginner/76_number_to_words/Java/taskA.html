1import java.util.ArrayList;
2import java.util.List;
3
4public class Main {
5 static String underThousand(int n) {
6 String[] ones = {
7 "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine", "ten",
8 "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen", "seventeen",
9 "eighteen", "nineteen"};
10 String[] tens = {"", "", "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety"};
11 List<String> parts = new ArrayList<>();
12 if (n >= 100) {
13 parts.add(ones[n / 100] + " hundred");
14 }
15 int remainder = n % 100;
16 if (remainder >= 20) {
17 parts.add(tens[remainder / 10] + (remainder % 10 > 0 ? "-" + ones[remainder % 10] : ""));
18 } else if (remainder > 0) {
19 parts.add(ones[remainder]);
20 }
21 return String.join(" ", parts);
22 }
23
24 public static void main(String[] args) {
25 for (int n : new int[]{0, 13, 40, 100, 568, 71000, 300019, 999999}) {
26 List<String> parts = new ArrayList<>();
27 if (n == 0) {
28 parts.add("zero");
29 }
30 if (n >= 1000) {
31 parts.add(underThousand(n / 1000) + " thousand");
32 }
33 if (n % 1000 > 0) {
34 parts.add(underThousand(n % 1000));
35 }
36 System.out.println(n + ": " + String.join(" ", parts));
37 }
38 }
