1#include <iostream>
2#include <string>
3#include <array>
4
5const std::array<std::string, 20> small = {"zero", "one", "two", "three", "four", "five", "six",
6 "seven", "eight", "nine", "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
7 "seventeen", "eighteen", "nineteen"};
8const std::array<std::string, 10> tens = {"", "", "twenty", "thirty", "forty", "fifty", "sixty",
9 "seventy", "eighty", "ninety"};
10
11std::string under(int n) {
12 std::string out;
13 if (n >= 100) {
14 out = small[n / 100] + " hundred";
15 n %= 100;
16 if (n)
17 out += ' ';
18 }
19 if (n >= 20) {
20 out += tens[n / 10];
21 if (n % 10)
22 out += "-" + small[n % 10];
23 } else if (n) {
24 out += small[n];
25 }
26 return out;
27}
28
29int main() {
30 for (int n : std::array<int, 8>{0, 13, 40, 100, 568, 71000, 300019, 999999}) {
31 std::string result;
32 if (n == 0)
33 result = "zero";
34 if (n >= 1000)
35 result = under(n / 1000) + " thousand";
36 if (n % 1000) {
37 if (!result.empty())
38 result += ' ';
39 result += under(n % 1000);
40 }
41 std::cout << n << ": " << result << '\n';
42 }
