1function small_words(number)
2 units = split(
3 "zero one two three four five six seven eight nine ten eleven twelve thirteen fourteen fifteen sixteen seventeen eighteen nineteen",
4 )
5 tens = split("zero ten twenty thirty forty fifty sixty seventy eighty ninety")
6 parts = String[]
7 hundreds, remainder = divrem(number, 100)
8 hundreds > 0 && push!(parts, string(units[hundreds+1], " hundred"))
9 if remainder >= 20
10 phrase = String(tens[div(remainder, 10)+1])
11 remainder%10 > 0 && (phrase *= string("-", units[remainder%10+1]))
12 push!(parts, phrase)
13 elseif remainder > 0
14 push!(parts, units[remainder+1])
15 end
16 return join(parts, " ")
17end

19function number_words(number)
20 number == 0 && return "zero"
21 thousands, remainder = divrem(number, 1000)
22 parts = String[]
23 thousands > 0 && push!(parts, string(small_words(thousands), " thousand"))
24 remainder > 0 && push!(parts, small_words(remainder))
25 return join(parts, " ")
26end
