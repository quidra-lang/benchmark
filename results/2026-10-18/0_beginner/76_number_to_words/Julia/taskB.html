1function small_words(number)
2 units = split(
3 "zero one two three four five six seven eight nine ten eleven twelve thirteen fourteen fifteen sixteen seventeen eighteen nineteen",
4 tens = split("zero ten twenty thirty forty fifty sixty seventy eighty ninety")
5 parts = String[]
6 hundreds, remainder = divrem(number, 100)
7 hundreds > 0 && push!(parts, string(units[hundreds+1], " hundred"))
8 if remainder >= 20
9 phrase = String(tens[div(remainder, 10)+1])
10 remainder%10 > 0 && (phrase *= string("-", units[remainder%10+1]))
11 push!(parts, phrase)
12 elseif remainder > 0
13 push!(parts, units[remainder+1])
14 end
15 return join(parts, " ")
16end
17function number_words(number)
18 number == 0 && return "zero"
19 thousands, remainder = divrem(number, 1000)
20 parts = String[]
21 thousands > 0 && push!(parts, string(small_words(thousands), " thousand"))
22 remainder > 0 && push!(parts, small_words(remainder))
23 return join(parts, " ")
