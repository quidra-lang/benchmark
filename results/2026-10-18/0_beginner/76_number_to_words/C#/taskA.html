1using System;
2using System.Collections.Generic;
3using System.Globalization;
4
5CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
6
7static string UnderThousand(int n)
8{
9 string[] ones = { "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
10 "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen", "seventeen", "eighteen", "nineteen" };
11 string[] tens = { "", "", "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety" };
12 var parts = new List<string>();
13 if (n >= 100) parts.Add(ones[n / 100] + " hundred");
14 int remainder = n % 100;
15 if (remainder >= 20) parts.Add(tens[remainder / 10] + (remainder % 10 > 0 ? "-" + ones[remainder % 10] : ""));
16 else if (remainder > 0) parts.Add(ones[remainder]);
17 return string.Join(" ", parts);
18}
19
20foreach (int n in new[] { 0, 13, 40, 100, 568, 71000, 300019, 999999 })
21{
22 var parts = new List<string>();
23 if (n == 0) parts.Add("zero");
24 if (n >= 1000) parts.Add(UnderThousand(n / 1000) + " thousand");
25 if (n % 1000 > 0) parts.Add(UnderThousand(n % 1000));
26 Console.WriteLine($"{n}: {string.Join(" ", parts)}");
27}
