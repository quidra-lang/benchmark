7def under_thousand(value)
8 parts = []
9 parts << SMALL[value / 100] + ' hundred' if value >= 100
10 remainder = value % 100
11 if remainder >= 20
12 word = TENS[remainder / 10]
13 word += '-' + SMALL[remainder % 10] unless (remainder % 10).zero?
14 parts << word
15 elsif remainder.positive?
16 parts << SMALL[remainder]
17 end
18 parts.join(' ')
19end
