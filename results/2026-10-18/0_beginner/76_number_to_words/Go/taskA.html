8func under(n int) string {
9 small := []string{"zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
10 "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen", "seventeen", "eighteen",
11 "nineteen"}
12 tens := []string{"", "", "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety"}
13 out := ""
14 if n >= 100 {
15 out = small[n/100] + " hundred"
16 n %= 100
17 if n > 0 {
18 out += " "
19 }
20 }
21 if n >= 20 {
22 out += tens[n/10]
23 if n%10 > 0 {
24 out += "-" + small[n%10]
25 }
26 } else if n > 0 {
27 out += small[n]
28 }
29 return out
30}

32func main() {

44 fmt.Printf("%d: %s\n", n, strings.Join(parts, " "))
