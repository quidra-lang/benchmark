3const ones = [_][]const u8{
4 "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
5 "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen", "seventeen", "eighteen", "nineteen",
6};
7const tens = [_][]const u8{ "", "", "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety" };

9fn under(out: *std.Io.Writer, n: u32) std.Io.Writer.Error!void {
10 const h = n / 100;
11 const y = n % 100;
12 if (h > 0)
13 try out.print("{s} hundred", .{ones[h]});
14 if (y > 0) {
15 if (h > 0)
16 try out.writeByte(' ');
17 if (y < 20) {
18 try out.writeAll(ones[y]);
19 } else {
20 try out.writeAll(tens[y / 10]);
21 if (y % 10 > 0)
22 try out.print("-{s}", .{ones[y % 10]});
23 }
24 }
25}

27pub fn main(init: std.process.Init) !void {

32 try out.print("{d}: ", .{n});
33 if (n == 0) {
34 try out.writeAll("zero");
35 } else {
36 const thousands = n / 1000;
37 const rest = n % 1000;
38 if (thousands > 0) {
39 try under(out, thousands);
40 try out.writeAll(" thousand");
41 }
42 if (rest > 0) {
43 if (thousands > 0)
44 try out.writeByte(' ');
45 try under(out, rest);
46 }
47 }

49 }
