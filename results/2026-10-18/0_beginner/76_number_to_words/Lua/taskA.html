1local units = {
2 "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine", "ten", "eleven",
3 "twelve", "thirteen", "fourteen", "fifteen", "sixteen", "seventeen", "eighteen", "nineteen",
4}
5local tens = {"zero", "ten", "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety"}
6
7local function small_words(number)
8 local parts = {}
9 local hundreds, remainder = number // 100, number % 100
10 if hundreds > 0 then parts[#parts + 1] = units[hundreds + 1] .. " hundred" end
11 if remainder >= 20 then
12 local phrase = tens[remainder // 10 + 1]
13 if remainder % 10 > 0 then phrase = phrase .. "-" .. units[remainder % 10 + 1] end
14 parts[#parts + 1] = phrase
15 elseif remainder > 0 then
16 parts[#parts + 1] = units[remainder + 1]
17 end
18 return table.concat(parts, " ")
19end
20
21local function number_words(number)
22 if number == 0 then return "zero" end
23 local parts = {}
24 local thousands, remainder = number // 1000, number % 1000
25 if thousands > 0 then parts[#parts + 1] = small_words(thousands) .. " thousand" end
26 if remainder > 0 then parts[#parts + 1] = small_words(remainder) end
27 return table.concat(parts, " ")
28end
29
30for _, number in ipairs({0, 13, 40, 100, 568, 71000, 300019, 999999}) do
31 io.write(string.format("%d: %s\n", number, number_words(number)))
32end
