board := [".#......", "..#.....", "###.....", "........", "........", "........", ".....###", "........"]
    string[] next := []
    for int r < 8
        row := ""
        for int c < 8
            int neighbors := 0
            for -1 <= int dr <= 1
                for -1 <= int dc <= 1
                    nr := r + dr
                    nc := c + dc
                    if (dr != 0 or dc != 0) and 0 <= nr < 8 and 0 <= nc < 8
                        if board[nr][nc] == "#"
                            neighbors += 1
            live := neighbors == 3 or (board[r][c] == "#" and neighbors == 2)
            row += if live then "#" else "."
        next = next.append(row)
    board = next
