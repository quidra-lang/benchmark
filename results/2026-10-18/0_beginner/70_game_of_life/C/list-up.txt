    char board[8][9] = {".#......", "..#.....", "###.....", "........", "........", "........", ".....###",
        "........"};
        char next[8][9];
        for (int r = 0; r < 8; r++) {
            for (int c = 0; c < 8; c++) {
                int neighbors = 0;
                for (int dr = -1; dr <= 1; dr++)
                    for (int dc = -1; dc <= 1; dc++) {
                        int rr = r + dr, cc = c + dc;
                        if ((dr || dc) && rr >= 0 && rr < 8 && cc >= 0 && cc < 8 && board[rr][cc] == '#')
                            neighbors++;
                    }
                next[r][c] = neighbors == 3 || (board[r][c] == '#' && neighbors == 2) ? '#' : '.';
            }
        }
        memcpy(board, next, sizeof board);
