1import java.util.Locale
2import kotlin.math.sqrt
3
4fun main() {
5 val values = intArrayOf(72, 85, 64, 90, 85, 58, 77, 85, 69, 93, 71)
6 val mean = values.average()
7 var squared = 0.0
8 for (value in values) {
9 val difference = value - mean
10 squared += difference * difference
11 }
12 val sorted = values.sorted()
13 val counts = values.toList().groupingBy { it }.eachCount()
14 val mode = counts.keys.minWith(compareByDescending<Int> { counts[it]!! }.thenBy { it })
15 println("mean: %.3f".format(Locale.ROOT, mean))
16 println("median: ${sorted[values.size / 2]}")
17 println("mode: $mode")
18 println("range: ${sorted.last() - sorted.first()}")
19 println("stdev: %.3f".format(Locale.ROOT, sqrt(squared / values.size)))
20}
