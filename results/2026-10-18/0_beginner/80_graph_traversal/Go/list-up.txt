func dfs(v int, graph *[9][9]bool, seen *[9]bool, order *[]int) {
	seen[v] = true
	*order = append(*order, v)
	for w := 0; w < 9; w++ {
		if graph[v][w] && !seen[w] {
			dfs(w, graph, seen, order)
		}
	}
}

		graph[e[0]][e[1]], graph[e[1]][e[0]] = true, true
	}

			if graph[v][w] && !seen[w] {
				seen[w] = true
				queue = append(queue, w)
			}
		}
	}
