edges = [(5, 0), (3, 8), (7, 2), (1, 6), (0, 2), (5, 3), (7, 5), (6, 4), (8, 7)]
graph = [[] for _ in range(9)]
for a, b in edges:
    graph[a].append(b)
    graph[b].append(a)
for neighbors in graph:
    neighbors.sort()
def dfs(vertex, seen, order):
    seen.add(vertex)
    order.append(vertex)
    for neighbor in graph[vertex]:
        if neighbor not in seen:
            dfs(neighbor, seen, order)
