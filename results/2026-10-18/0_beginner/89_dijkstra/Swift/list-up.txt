var distance = [Int](repeating: Int.max, count: 7), previous = [Int](repeating: -1, count: 7)
distance[0] = 0
let names = Array("ABCDEFG")
for v in 0..<7 {
    var path = [String](), u = v
    while u >= 0 {
        path.append(String(names[u]))
        u = previous[u]
    }
    print("\(names[v]): distance=\(distance[v]) path=\(path.reversed().joined(separator: "-"))")
