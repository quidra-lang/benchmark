1using System;
2using System.Collections.Generic;
3using System.Globalization;
4using System.Linq;
5
6CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
7
8var edges = new[]
9{
10 (0, 1, 7), (0, 2, 3), (0, 3, 9), (1, 2, 2), (2, 4, 6), (1, 3, 3), (3, 4, 4), (3, 5, 5),
11 (4, 5, 3), (4, 6, 8), (5, 6, 4),
12};
13int[,] weight = new int[7, 7];
14foreach (var (a, b, w) in edges)
15{
16 weight[a, b] = w;
17 weight[b, a] = w;
18}
19int[] distance = Enumerable.Repeat(int.MaxValue, 7).ToArray();
20int[] previous = Enumerable.Repeat(-1, 7).ToArray();
21distance[0] = 0;
22bool[] done = new bool[7];
23for (int count = 0; count < 7; count++)
24{
25 int u = Enumerable.Range(0, 7).Where(v => !done[v]).MinBy(v => distance[v]);
26 done[u] = true;
27 for (int v = 0; v < 7; v++)
28 {
29 if (weight[u, v] > 0 && distance[u] + weight[u, v] < distance[v])
30 {
31 distance[v] = distance[u] + weight[u, v];
32 previous[v] = u;
33 }
34 }
35}
36for (int v = 0; v < 7; v++)
37{
38 var path = new List<char>();
39 for (int u = v; u >= 0; u = previous[u]) path.Add((char)('A' + u));
40 path.Reverse();
41 Console.WriteLine($"{(char)('A' + v)}: distance={distance[v]} path={string.Join("-", path)}");
42}
