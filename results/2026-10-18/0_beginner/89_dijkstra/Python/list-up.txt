names = "ABCDEFG"
distances = [float("inf")] * len(names)
distances[0] = 0
previous = [-1] * len(names)
for target, name in enumerate(names):
    vertex = target
    while vertex >= 0:
        path.append(names[vertex])
        vertex = previous[vertex]
    print(f"{name}: distance={distances[target]} path={'-'.join(reversed(path))}")
