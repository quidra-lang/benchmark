3fn solve(out: *std.Io.Writer, comptime n: usize, system: usize, matrix: [n][n + 1]f64) !void {
4 var a = matrix;
5 for (0..n) |k| {
6 var pivot = k;
7 for (k..n) |i| {
8 if (@abs(a[i][k]) > @abs(a[pivot][k]))
9 pivot = i;
10 }
11 if (pivot != k) {
12 std.mem.swap([n + 1]f64, &a[k], &a[pivot]);
13 }
14 for (k + 1..n) |i| {
15 const f = a[i][k] / a[k][k];
16 for (k..n + 1) |j|
17 a[i][j] = a[i][j] - f * a[k][j];
18 }
