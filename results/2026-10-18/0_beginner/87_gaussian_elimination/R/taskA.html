1solve_system <- function(values) {
2 n <- nrow(values)
3 for (k in seq_len(n)) {
4 pivot <- k - 1 + which.max(abs(values[k:n, k]))
5 if (pivot != k) values[c(k, pivot), ] <- values[c(pivot, k), ]
6 if (k < n) {
7 for (i in (k + 1):n) {
8 factor <- values[i, k] / values[k, k]
9 for (j in k:(n + 1)) {
10 values[i, j] <- values[i, j] - factor * values[k, j]
11 }
12 }
13 }
14 }
15 solution <- numeric(n)
16 for (i in n:1) {
17 subtotal <- values[i, n + 1]
18 if (i < n) {
19 for (j in (i + 1):n) {
20 subtotal <- subtotal - values[i, j] * solution[j]
21 }
22 }
23 solution[i] <- subtotal / values[i, i]
24 }
25 solution
26}
