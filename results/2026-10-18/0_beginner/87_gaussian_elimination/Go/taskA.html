8func main() {
9 systems := [][][]float64{
10 {{2, -1, 3, 7}, {4, 1, -2, 3}, {-3, 5, 1, -4}},
11 {{1, 3, -2, 4, 6}, {5, -2, 1, 3, -7}, {-2, 4, 6, -1, 9}, {3, 1, -4, 2, 5}},
12 }
13 for s, a := range systems {
14 n := len(a)
15 x := make([]float64, n)
16 for k := 0; k < n; k++ {
17 pivot := k
18 for i := k + 1; i < n; i++ {
19 if math.Abs(a[i][k]) > math.Abs(a[pivot][k]) {
20 pivot = i
21 }
22 }
23 a[k], a[pivot] = a[pivot], a[k]
24 for i := k + 1; i < n; i++ {
25 f := a[i][k] / a[k][k]
26 for j := k; j <= n; j++ {
27 a[i][j] = a[i][j] - f*a[k][j]
28 }
29 }
30 }
31 for i := n - 1; i >= 0; i-- {
32 sum := a[i][n]
33 for j := i + 1; j < n; j++ {
34 sum = sum - a[i][j]*x[j]
35 }
36 x[i] = sum / a[i][i]
37 }
38 fmt.Printf("system %d:\n", s+1)
39 for i, value := range x {
40 fmt.Printf("x%d = %.4f\n", i+1, value)
41 }
42 }
