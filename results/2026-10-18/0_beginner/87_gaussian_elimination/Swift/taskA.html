3let systems: [[[Double]]] = [[[2, -1, 3, 7], [4, 1, -2, 3], [-3, 5, 1, -4]],
4 [[1, 3, -2, 4, 6], [5, -2, 1, 3, -7], [-2, 4, 6, -1, 9], [3, 1, -4, 2, 5]]]
5for (system, original) in systems.enumerated() {
6 var a = original
7 let n = a.count
8 for k in 0..<n {
9 let pivot = (k..<n).max { abs(a[$0][k]) < abs(a[$1][k]) }!
10 a.swapAt(k, pivot)
11 for i in (k + 1)..<n {
12 let factor = a[i][k] / a[k][k]
13 for j in k...n { a[i][j] = a[i][j] - factor * a[k][j] }
14 }
15 }
16 var x = [Double](repeating: 0, count: n)
17 for i in stride(from: n - 1, through: 0, by: -1) {
18 var sum = a[i][n]
19 for j in (i + 1)..<n { sum = sum - a[i][j] * x[j] }
20 x[i] = sum / a[i][i]
21 }
22 print("system \(system + 1):")
23 for i in 0..<n {
