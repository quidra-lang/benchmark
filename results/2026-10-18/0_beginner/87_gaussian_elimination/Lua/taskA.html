1local function solve_system(matrix)
2 local n = #matrix
3 for k = 1, n do
4 local pivot = k
5 for i = k + 1, n do
6 if math.abs(matrix[i][k]) > math.abs(matrix[pivot][k]) then pivot = i end
7 end
8 matrix[k], matrix[pivot] = matrix[pivot], matrix[k]
9 for i = k + 1, n do
10 local factor = matrix[i][k] / matrix[k][k]
11 for j = k, n + 1 do matrix[i][j] = matrix[i][j] - factor * matrix[k][j] end
12 end
13 end
14 local solution = {}
15 for i = n, 1, -1 do
16 local subtotal = matrix[i][n + 1]
17 for j = i + 1, n do subtotal = subtotal - matrix[i][j] * solution[j] end
18 solution[i] = subtotal / matrix[i][i]
19 end
20 return solution
21end

23local systems = {
24 {{2.0, -1.0, 3.0, 7.0}, {4.0, 1.0, -2.0, 3.0}, {-3.0, 5.0, 1.0, -4.0}},
25 {{1.0, 3.0, -2.0, 4.0, 6.0}, {5.0, -2.0, 1.0, 3.0, -7.0}, {-2.0, 4.0, 6.0, -1.0, 9.0}, {3.0, 1.0, -4.0, 2.0, 5.0}},
26}
27for number, matrix in ipairs(systems) do
28 io.write(string.format("system %d:\n", number))
29 for i, value in ipairs(solve_system(matrix)) do
30 io.write(string.format("x%d = %.4f\n", i, value))
31 end
