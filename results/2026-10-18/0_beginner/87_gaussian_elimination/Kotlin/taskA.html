1import java.util.Locale
2import kotlin.math.abs
3
4fun main() {
5 val systems = listOf(
6 arrayOf(
7 doubleArrayOf(2.0, -1.0, 3.0, 7.0),
8 doubleArrayOf(4.0, 1.0, -2.0, 3.0),
9 doubleArrayOf(-3.0, 5.0, 1.0, -4.0),
10 ),
11 arrayOf(
12 doubleArrayOf(1.0, 3.0, -2.0, 4.0, 6.0),
13 doubleArrayOf(5.0, -2.0, 1.0, 3.0, -7.0),
14 doubleArrayOf(-2.0, 4.0, 6.0, -1.0, 9.0),
15 doubleArrayOf(3.0, 1.0, -4.0, 2.0, 5.0),
16 ),
17 )
18 for ((system, a) in systems.withIndex()) {
19 val n = a.size
20 for (k in 0 until n) {
21 val pivot = (k until n).maxBy { abs(a[it][k]) }
22 val temporary = a[k]
23 a[k] = a[pivot]
24 a[pivot] = temporary
25 for (i in k + 1 until n) {
26 val factor = a[i][k] / a[k][k]
27 for (j in k..n) a[i][j] = a[i][j] - factor * a[k][j]
28 }
29 }
30 val x = DoubleArray(n)
31 for (i in n - 1 downTo 0) {
32 var sum = a[i][n]
33 for (j in i + 1 until n) sum = sum - a[i][j] * x[j]
34 x[i] = sum / a[i][i]
35 }
36 println("system ${system + 1}:")
37 for (i in 0 until n) println("x%d = %.4f".format(Locale.ROOT, i + 1, x[i]))
38 }
39}
