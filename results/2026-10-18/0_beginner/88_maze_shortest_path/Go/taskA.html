3import "fmt"
4
5func main() {
6 rows := []string{"#############", "#S..#.......#", "#.#.#.###.#.#", "#.#...#...#.#", "#.###.#.###.#",
7 "#...#...#...#", "###.#####.#.#", "#.......#..G#", "#############"}
8 maze := make([][]byte, 9)
9 parent, distance := [117]int{}, [117]int{}
10 start, goal := 0, 0
11 for r, row := range rows {
12 maze[r] = []byte(row)
13 }
14 for i := 0; i < 117; i++ {
15 parent[i] = -1
16 if maze[i/13][i%13] == 'S' {
17 start = i
18 }
19 if maze[i/13][i%13] == 'G' {
20 goal = i
21 }
22 }
23 queue := []int{start}
24 parent[start] = start
25 directions := [][2]int{{-1, 0}, {1, 0}, {0, -1}, {0, 1}}
26 for head := 0; head < len(queue); head++ {
27 v := queue[head]
28 r, c := v/13, v%13
29 for _, d := range directions {
30 rr, cc := r+d[0], c+d[1]
31 if rr < 0 || rr >= 9 || cc < 0 || cc >= 13 || maze[rr][cc] == '#' {
32 continue
33 }
34 w := rr*13 + cc
35 if parent[w] < 0 {
36 parent[w] = v
37 distance[w] = distance[v] + 1
38 queue = append(queue, w)
39 }
40 }
41 }
42 for v := goal; v != start; v = parent[v] {
43 if maze[v/13][v%13] == '.' {
44 maze[v/13][v%13] = '*'
45 }
46 }
47 fmt.Printf("steps: %d\n", distance[goal])
48 for _, row := range maze {
49 fmt.Println(string(row))
50 }
51}
