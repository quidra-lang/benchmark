1import java.util.ArrayDeque;
2import java.util.Arrays;
3import java.util.Deque;
4
5public class Main {
6 public static void main(String[] args) {
7 String[] rows = {
8 "#############",
9 "#S..#.......#",
10 "#.#.#.###.#.#",
11 "#.#...#...#.#",
12 "#.###.#.###.#",
13 "#...#...#...#",
14 "###.#####.#.#",
15 "#.......#..G#",
16 "#############"};
17 char[][] maze = Arrays.stream(rows).map(String::toCharArray).toArray(char[][]::new);
18 int start = -1, goal = -1;
19 for (int r = 0; r < 9; r++) {
20 for (int c = 0; c < 13; c++) {
21 if (maze[r][c] == 'S') {
22 start = r * 13 + c;
23 }
24 if (maze[r][c] == 'G') {
25 goal = r * 13 + c;
26 }
27 }
28 }
29 int[] parent = new int[117];
30 Arrays.fill(parent, -1);
31 parent[start] = start;
32 Deque<Integer> queue = new ArrayDeque<>();
33 queue.add(start);
34 int[] dr = {-1, 1, 0, 0}, dc = {0, 0, -1, 1};
35 while (!queue.isEmpty()) {
36 int cell = queue.remove(), r = cell / 13, c = cell % 13;
37 if (cell == goal) {
38 break;
39 }
40 for (int d = 0; d < 4; d++) {
41 int nr = r + dr[d], nc = c + dc[d];
42 if (nr < 0 || nr >= 9 || nc < 0 || nc >= 13) {
43 continue;
44 }
45 int next = nr * 13 + nc;
46 if (maze[nr][nc] != '#' && parent[next] < 0) {
47 parent[next] = cell;
48 queue.add(next);
49 }
50 }
51 }
52 int steps = 0;
53 for (int cell = goal; cell != start; cell = parent[cell]) {
54 if (cell != goal) {
55 maze[cell / 13][cell % 13] = '*';
56 }
57 steps++;
58 }
59 System.out.println("steps: " + steps);
60 for (char[] row : maze) {
61 System.out.println(new String(row));
62 }
63 }
