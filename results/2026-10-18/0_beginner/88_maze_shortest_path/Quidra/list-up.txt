for 0 <= int r < 9
    for 0 <= int c < 13
        if maze[r][c] == "S"
            start = r * 13 + c
        if maze[r][c] == "G"
            goal = r * 13 + c
int[] parent = array(117, fill = -1)
parent[start] = start
int[] queue = [start]
int head = 0
int[] dr = [-1, 1, 0, 0]
int[] dc = [0, 0, -1, 1]
while head < int(len(queue)) and parent[goal] == -1
    int current = queue[head]
    head += 1
    int r = current / 13
    int c = current % 13
    for 0 <= int direction < 4
        int nr = r + dr[direction]
        int nc = c + dc[direction]
        if 0 <= nr < 9 and 0 <= nc < 13
            int neighbor = nr * 13 + nc
            if maze[nr][nc] != "#" and parent[neighbor] == -1
                parent[neighbor] = current
                queue = queue.append(neighbor)
bool[] path = array(117, fill = false)
int steps = 0
int current = goal
while current != start
    path[current] = true
    current = parent[current]
    steps += 1
print("steps: {steps}{NL}")
for 0 <= int r < 9
    string line = ""
    for 0 <= int c < 13
        string character = maze[r][c]
        if path[r * 13 + c] and character == "."
            character = "*"
        line += character
    print(line + NL)
