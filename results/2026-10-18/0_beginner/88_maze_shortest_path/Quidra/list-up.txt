    for (int dr, int dc) = directions[]
        int nr = r + dr
        int nc = c + dc
        if 0 <= nr < 9 and 0 <= nc < 13
            int neighbor = nr * 13 + nc
            if maze[neighbor] != "#" and parent[neighbor] == -1
