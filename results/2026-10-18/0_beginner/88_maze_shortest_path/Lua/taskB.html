18 head = head + 1

20 for _, direction in ipairs({{-1, 0}, {1, 0}, {0, -1}, {0, 1}}) do

24 parents[position] = cell

28end
29local cell, steps = goal, 0
30while cell[1] ~= start[1] or cell[2] ~= start[2] do
31 local r, c = table.unpack(cell)
32 if maze[r][c] == "." then maze[r][c] = "*" end
33 cell = parents[key(r, c)]
34 steps = steps + 1
35end
36io.write(string.format("steps: %d\n", steps))
37for _, row in ipairs(maze) do io.write(table.concat(row), "\n") end
