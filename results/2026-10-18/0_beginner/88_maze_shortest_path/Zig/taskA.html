40 const u: usize = @intCast(r * 13 + c);
41 if (maze[u] != '#' and parent[u] == -1) {
42 parent[u] = @intCast(v);

44 tail += 1;
45 }
46 }
47 }

49 var steps: usize = 0;

54 steps += 1;
55 }

57 for (0..9) |r|
58 try out.print("{s}\n", .{maze[r * 13 .. r * 13 + 13]});
59 try out.flush();
60}
