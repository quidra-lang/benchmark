1using System;
2using System.Collections.Generic;
3using System.Globalization;
4using System.Linq;
5
6CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
7
8string[] rows =
9{
10 "#############",
11 "#S..#.......#",
12 "#.#.#.###.#.#",
13 "#.#...#...#.#",
14 "#.###.#.###.#",
15 "#...#...#...#",
16 "###.#####.#.#",
17 "#.......#..G#",
18 "#############",
19};
20char[][] maze = rows.Select(row => row.ToCharArray()).ToArray();
21int start = -1, goal = -1;
22for (int r = 0; r < 9; r++)
23{
24 for (int c = 0; c < 13; c++)
25 {
26 if (maze[r][c] == 'S') start = r * 13 + c;
27 if (maze[r][c] == 'G') goal = r * 13 + c;
28 }
29}
30int[] parent = Enumerable.Repeat(-1, 117).ToArray();
31parent[start] = start;
32var queue = new Queue<int>();
33queue.Enqueue(start);
34int[] dr = { -1, 1, 0, 0 }, dc = { 0, 0, -1, 1 };
35while (queue.Count > 0)
36{
37 int cell = queue.Dequeue(), r = cell / 13, c = cell % 13;
38 if (cell == goal) break;
39 for (int d = 0; d < 4; d++)
40 {
41 int nr = r + dr[d], nc = c + dc[d];
42 if (nr < 0 || nr >= 9 || nc < 0 || nc >= 13) continue;
43 int next = nr * 13 + nc;
44 if (maze[nr][nc] != '#' && parent[next] < 0)
45 {
46 parent[next] = cell;
47 queue.Enqueue(next);
48 }
49 }
50}
51int steps = 0;
52for (int cell = goal; cell != start; cell = parent[cell])
53{
54 if (cell != goal) maze[cell / 13][cell % 13] = '*';
55 steps++;
56}
57Console.WriteLine($"steps: {steps}");
58foreach (char[] row in maze) Console.WriteLine(new string(row));
