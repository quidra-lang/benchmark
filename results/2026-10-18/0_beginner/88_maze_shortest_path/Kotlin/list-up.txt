    for (r in 0 until 9) {
        for (c in 0 until 13) {
            if (maze[r][c] == 'S') start = r * 13 + c
            if (maze[r][c] == 'G') goal = r * 13 + c
        }
    }
    val parent = IntArray(117) { -1 }
    parent[start] = start
    val queue = ArrayDeque<Int>()
    queue.addLast(start)
    while (queue.isNotEmpty()) {
        val cell = queue.removeFirst()
        if (cell == goal) break
            val next = nr * 13 + nc
            if (maze[nr][nc] != '#' && parent[next] < 0) {
                parent[next] = cell
                queue.addLast(next)
            }
    }
    var steps = 0
    var cell = goal
    while (cell != start) {
        if (cell != goal) maze[cell / 13][cell % 13] = '*'
        steps++
        cell = parent[cell]
    }
    println("steps: $steps")
