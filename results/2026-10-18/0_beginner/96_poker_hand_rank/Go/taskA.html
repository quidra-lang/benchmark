10type Hand struct {
11 text string
12 category int
13 key []int
14}
15
16func evaluate(text string) Hand {
17 ranks := "23456789TJQKA"
18 counts := [15]int{}
19 flush := true
20 for i := 0; i < 5; i++ {
21 rank := strings.IndexByte(ranks, text[i*3]) + 2
22 counts[rank]++
23 if text[i*3+1] != text[1] {
24 flush = false
25 }
26 }
27 key := []int{}
28 maximum, pairs := 0, 0
29 for r := 14; r >= 2; r-- {
30 if counts[r] > 0 {
31 key = append(key, r)
32 maximum = max(maximum, counts[r])
33 if counts[r] == 2 {
34 pairs++
35 }
36 }
37 }
38 wheel := len(key) == 5 && counts[14] > 0 && counts[2] > 0 && counts[3] > 0 && counts[4] > 0 && counts[5] > 0
39 straight := len(key) == 5 && (key[0]-key[4] == 4 || wheel)
40 category := 0
41 switch {
42 case straight && flush:
43 category = 8
44 case maximum == 4:
45 category = 7
46 case maximum == 3 && pairs == 1:
47 category = 6
48 case flush:
49 category = 5
50 case straight:
51 category = 4
52 case maximum == 3:
53 category = 3
54 case pairs == 2:
55 category = 2
56 case pairs == 1:
57 category = 1
58 }
59 sort.Slice(key, func(i, j int) bool {
60 a, b := key[i], key[j]
61 if counts[a] != counts[b] {
62 return counts[a] > counts[b]
63 }
64 return a > b
65 })
66 if wheel && straight {
67 for i := range key {
68 key[i] = 5 - i
69 }
70 }
71 return Hand{text, category, key}
72}

74func main() {
75 hands := []Hand{}
76 input := []string{"9C 9D 5H QH 2C", "4D AH 3C 5S 2H", "KS 8S 2S JS 6S", "7H 7D KC 7S KD",
77 "3H 6C 4S 2D 5C", "JD TC 4H 8C AS", "QC 6D QS 3S 6H", "9H KH 9S 8H 3D"}
78 for _, text := range input {
79 hands = append(hands, evaluate(text))
80 }
81 sort.Slice(hands, func(i, j int) bool {
82 a, b := hands[i], hands[j]
83 if a.category != b.category {
84 return a.category > b.category
85 }
86 return slices.Compare(a.key, b.key) > 0
87 })
88 categories := []string{"high card", "one pair", "two pair", "three of a kind", "straight", "flush",
89 "full house", "four of a kind", "straight flush"}
90 for i, h := range hands {
91 fmt.Printf("%d. %s: %s\n", i+1, h.text, categories[h.category])
92 }
93}
