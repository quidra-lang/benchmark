1struct Hand {
2 let text: String
3 let category: Int
4 let sequence: [Int]
5}

7func evaluate(_ text: String) -> Hand {
8 let cards = text.split(separator: " ").map { Array($0) }, alphabet = Array("23456789TJQKA")
9 var counts = [Int: Int]()
10 for card in cards { counts[alphabet.firstIndex(of: card[0])! + 2, default: 0] += 1 }
11 let ranks = counts.keys.sorted(), wheel = ranks == [2, 3, 4, 5, 14]
12 let straight = wheel || (ranks.count == 5 && ranks.last! - ranks.first! == 4)
13 let flush = cards.allSatisfy { $0[1] == cards[0][1] }
14 let largest = counts.values.max()!, pairs = counts.values.filter { $0 == 2 }.count
15 let category: Int
16 if straight && flush {
17 category = 8
18 } else if largest == 4 {
19 category = 7
20 } else if largest == 3 && pairs == 1 {
21 category = 6
22 } else if flush {
23 category = 5
24 } else if straight {
25 category = 4
26 } else if largest == 3 {
27 category = 3
28 } else if pairs == 2 {
29 category = 2
30 } else if pairs == 1 {
31 category = 1
32 } else {
33 category = 0
34 }
35 var sequence = counts.keys.sorted { counts[$0]! == counts[$1]! ? $0 > $1 : counts[$0]! > counts[$1]! }
36 if wheel && straight { sequence = [5, 4, 3, 2, 1] }
37 return Hand(text: text, category: category, sequence: sequence)
38}

42let categories = ["high card", "one pair", "two pair", "three of a kind", "straight",
