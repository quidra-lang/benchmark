1describe_hand <- function(original) {
2 cards <- strsplit(original, " ", fixed = TRUE)[[1]]
3 alphabet <- strsplit("23456789TJQKA", "", fixed = TRUE)[[1]]
4 ranks <- match(substr(cards, 1, 1), alphabet) + 1
5 frequencies <- tabulate(ranks, nbins = 14)
6 distinct <- which(frequencies > 0)
7 wheel <- identical(distinct, c(2L, 3L, 4L, 5L, 14L))
8 straight <- length(distinct) == 5 && (max(distinct) - min(distinct) == 4 || wheel)
9 flush <- length(unique(substr(cards, 2, 2))) == 1
10 counts <- sort(frequencies[distinct], decreasing = TRUE)
11 if (straight && flush) {
12 strength <- 9
13 } else if (counts[1] == 4) {
14 strength <- 8
15 } else if (identical(counts, c(3L, 2L))) {
16 strength <- 7
17 } else if (flush) {
18 strength <- 6
19 } else if (straight) {
20 strength <- 5
21 } else if (counts[1] == 3) {
22 strength <- 4
23 } else if (identical(counts, c(2L, 2L, 1L))) {
24 strength <- 3
25 } else if (counts[1] == 2) {
26 strength <- 2
27 } else {
28 strength <- 1
29 }
30 sequence <- if (wheel && straight) c(5, 4, 3, 2, 1) else distinct[order(-frequencies[distinct], -distinct)]
31 categories <- c(
32 "high card", "one pair", "two pair", "three of a kind", "straight", "flush", "full house",
33 "four of a kind", "straight flush"
34 )
35 list(
36 original = original, strength = strength, sequence = c(sequence, rep(0, 5 - length(sequence))),
37 category = categories[strength]
38 )
39}

45hands <- lapply(originals, describe_hand)
46strengths <- vapply(hands, function(hand) hand$strength, numeric(1))
47ties <- do.call(rbind, lapply(hands, function(hand) hand$sequence))
48ordered <- order(-strengths, -ties[, 1], -ties[, 2], -ties[, 3], -ties[, 4], -ties[, 5])
49for (position in seq_along(ordered)) {
50 hand <- hands[[ordered[position]]]
51 cat(sprintf("%d. %s: %s\n", position, hand$original, hand$category))
52}
