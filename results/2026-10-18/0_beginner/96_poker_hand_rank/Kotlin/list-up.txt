data class Hand(val text: String, val category: Int, val sequence: List<Int>)
fun evaluate(text: String): Hand {
    val cards = text.split(' ')
    val counts = cards.groupingBy { "23456789TJQKA".indexOf(it[0]) + 2 }.eachCount()
    val ranks = counts.keys.sorted()
    val wheel = ranks == listOf(2, 3, 4, 5, 14)
    val straight = wheel || (ranks.size == 5 && ranks.last() - ranks.first() == 4)
    val flush = cards.all { it[1] == cards[0][1] }
    val largest = counts.values.max()
    val pairs = counts.values.count { it == 2 }
    val category = when {
        straight && flush -> 8
        largest == 4 -> 7
        largest == 3 && pairs == 1 -> 6
        flush -> 5
        straight -> 4
    }
    var sequence = counts.keys.sortedWith(compareByDescending<Int> { counts.getValue(it) }.thenByDescending { it })
    if (wheel && straight) sequence = listOf(5, 4, 3, 2, 1)
    return Hand(text, category, sequence)
