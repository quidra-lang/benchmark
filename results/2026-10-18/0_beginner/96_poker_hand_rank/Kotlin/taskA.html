1data class Hand(val text: String, val category: Int, val sequence: List<Int>)
2
3fun evaluate(text: String): Hand {
4 val cards = text.split(' ')
5 val counts = cards.groupingBy { "23456789TJQKA".indexOf(it[0]) + 2 }.eachCount()
6 val ranks = counts.keys.sorted()
7 val wheel = ranks == listOf(2, 3, 4, 5, 14)
8 val straight = wheel || (ranks.size == 5 && ranks.last() - ranks.first() == 4)
9 val flush = cards.all { it[1] == cards[0][1] }
10 val largest = counts.values.max()
11 val pairs = counts.values.count { it == 2 }
12 val category = when {
13 straight && flush -> 8
14 largest == 4 -> 7
15 largest == 3 && pairs == 1 -> 6
16 flush -> 5
17 straight -> 4
18 largest == 3 -> 3
19 pairs == 2 -> 2
20 pairs == 1 -> 1
21 else -> 0
22 }
23 var sequence = counts.keys.sortedWith(compareByDescending<Int> { counts.getValue(it) }.thenByDescending { it })
24 if (wheel && straight) sequence = listOf(5, 4, 3, 2, 1)
25 return Hand(text, category, sequence)
26}

31 val categories = listOf("high card", "one pair", "two pair", "three of a kind", "straight",
32 "flush", "full house", "four of a kind", "straight flush")
33 val hands = input.map(::evaluate).sortedWith { a, b ->
34 var comparison = b.category.compareTo(a.category)
35 if (comparison == 0) {
36 for (i in a.sequence.indices) {
37 comparison = b.sequence[i].compareTo(a.sequence[i])
38 if (comparison != 0) break
39 }
40 }
41 comparison
42 }
43 for ((i, hand) in hands.withIndex()) println("${i + 1}. ${hand.text}: ${categories[hand.category]}")
44}
