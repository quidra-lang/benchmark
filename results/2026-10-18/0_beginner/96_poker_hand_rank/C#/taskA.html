7static (string text, int category, int[] sequence) Evaluate(string text)
8{
9 string[] cards = text.Split(' ');
10 var counts = cards.GroupBy(card => "23456789TJQKA".IndexOf(card[0]) + 2)
11 .ToDictionary(group => group.Key, group => group.Count());
12 int[] ranks = counts.Keys.Order().ToArray();
13 bool wheel = ranks.SequenceEqual(new[] { 2, 3, 4, 5, 14 });
14 bool straight = wheel || (ranks.Length == 5 && ranks[^1] - ranks[0] == 4);
15 bool flush = cards.All(card => card[1] == cards[0][1]);
16 int largest = counts.Values.Max(), pairs = counts.Values.Count(count => count == 2);
17 int category;
18 if (straight && flush) category = 8;
19 else if (largest == 4) category = 7;
20 else if (largest == 3 && pairs == 1) category = 6;
21 else if (flush) category = 5;
22 else if (straight) category = 4;
23 else if (largest == 3) category = 3;
24 else if (pairs == 2) category = 2;
25 else if (pairs == 1) category = 1;
26 else category = 0;
27 var sequence = counts.Keys.OrderByDescending(rank => counts[rank]).ThenByDescending(rank => rank).ToArray();
28 if (wheel && straight) sequence = new[] { 5, 4, 3, 2, 1 };
29 return (text, category, sequence);
30}
