1function describe_hand(original)
2 cards = split(original)
3 alphabet = "23456789TJQKA"
4 ranks = [findfirst(==(card[1]), alphabet)+1 for card in cards]
5 frequencies = [count(==(rank), ranks) for rank = 2:14]
6 distinct = [rank for rank = 2:14 if frequencies[rank-1] > 0]
7 wheel = distinct == [2, 3, 4, 5, 14]
8 straight = length(distinct) == 5 && (distinct[end]-distinct[1] == 4 || wheel)
9 flush = all(card[2] == cards[1][2] for card in cards)
10 counts = sort([frequencies[rank-1] for rank in distinct]; rev = true)
11 if straight && flush
12 strength = 9
13 elseif counts[1] == 4
14 strength = 8
15 elseif counts == [3, 2]
16 strength = 7
17 elseif flush
18 strength = 6
19 elseif straight
20 strength = 5
21 elseif counts[1] == 3
22 strength = 4
23 elseif counts == [2, 2, 1]
24 strength = 3
25 elseif counts[1] == 2
26 strength = 2
27 else
28 strength = 1
29 end
30 categories = [
31 "high card",
32 "one pair",
33 "two pair",
34 "three of a kind",
35 "straight",
36 "flush",
37 "full house",
38 "four of a kind",
39 "straight flush",
40 ]
41 sequence =
42 wheel && straight ? [5, 4, 3, 2, 1] :
43 sort(distinct; by = rank -> (-frequencies[rank-1], -rank))
44 return (
45 original = original,
46 strength = strength,
47 sequence = sequence,
48 category = categories[strength],
49 )
50end

62sort!(hands; by = hand -> (-hand.strength, Tuple(-rank for rank in hand.sequence)))
63for (position, hand) in enumerate(hands)
64 println(position, ". ", hand.original, ": ", hand.category)
