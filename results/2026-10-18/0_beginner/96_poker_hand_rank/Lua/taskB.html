1local categories = {
2 "high card", "one pair", "two pair", "three of a kind", "straight", "flush", "full house",
3 "four of a kind", "straight flush",
4}
5local function describe(original)
6 local alphabet, frequencies, suits = "23456789TJQKA", {}, {}
7 for card in original:gmatch("%S+") do
8 local rank = alphabet:find(card:sub(1, 1), 1, true) + 1
9 frequencies[rank] = (frequencies[rank] or 0) + 1
10 suits[card:sub(2, 2)] = true
11 end
12 local distinct, counts, suit_count = {}, {}, 0
13 for rank = 2, 14 do
14 if frequencies[rank] then
15 distinct[#distinct + 1] = rank
16 counts[#counts + 1] = frequencies[rank]
17 end
18 end
19 for _ in pairs(suits) do suit_count = suit_count + 1 end
20 local wheel = #distinct == 5 and distinct[1] == 2 and distinct[2] == 3
21 and distinct[3] == 4 and distinct[4] == 5 and distinct[5] == 14
22 local straight = #distinct == 5 and (distinct[5] - distinct[1] == 4 or wheel)
23 local flush = suit_count == 1
24 table.sort(counts, function(a, b) return a > b end)
25 local strength
26 if straight and flush then
27 strength = 9
28 elseif counts[1] == 4 then
29 strength = 8
30 elseif counts[1] == 3 and counts[2] == 2 then
31 strength = 7
32 elseif flush then
33 strength = 6
34 elseif straight then
35 strength = 5
36 end
37 table.sort(distinct, function(a, b)
38 if frequencies[a] ~= frequencies[b] then return frequencies[a] > frequencies[b] end
39 return a > b
40 end)
41 if wheel and straight then distinct = {5, 4, 3, 2, 1} end
42 return {original = original, strength = strength, sequence = distinct, category = categories[strength]}
43end
44local originals = {
45 "9C 9D 5H QH 2C", "4D AH 3C 5S 2H", "KS 8S 2S JS 6S", "7H 7D KC 7S KD", "3H 6C 4S 2D 5C",
46}
47for _, original in ipairs(originals) do
48 hands[#hands + 1] = describe(original)
49end
50table.sort(hands, function(a, b)
51 if a.strength ~= b.strength then return a.strength > b.strength end
52 for i = 1, #a.sequence do
53 if a.sequence[i] ~= b.sequence[i] then return a.sequence[i] > b.sequence[i] end
54 end
55 return false
56end)
57for position, hand in ipairs(hands) do
58 io.write(string.format("%d. %s: %s\n", position, hand.original, hand.category))
59end
