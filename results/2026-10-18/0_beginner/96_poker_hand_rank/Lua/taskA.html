1local categories = {
2 "high card", "one pair", "two pair", "three of a kind", "straight", "flush", "full house",
3 "four of a kind", "straight flush",
4}
5
6local function describe(original)
7 local alphabet, frequencies, suits = "23456789TJQKA", {}, {}
8 for card in original:gmatch("%S+") do
9 local rank = alphabet:find(card:sub(1, 1), 1, true) + 1
10 frequencies[rank] = (frequencies[rank] or 0) + 1
11 suits[card:sub(2, 2)] = true
12 end
13 local distinct, counts, suit_count = {}, {}, 0
14 for rank = 2, 14 do
15 if frequencies[rank] then
16 distinct[#distinct + 1] = rank
17 counts[#counts + 1] = frequencies[rank]
18 end
19 end
20 for _ in pairs(suits) do suit_count = suit_count + 1 end
21 local wheel = #distinct == 5 and distinct[1] == 2 and distinct[2] == 3
22 and distinct[3] == 4 and distinct[4] == 5 and distinct[5] == 14
23 local straight = #distinct == 5 and (distinct[5] - distinct[1] == 4 or wheel)
24 local flush = suit_count == 1
25 table.sort(counts, function(a, b) return a > b end)
26 local strength
27 if straight and flush then
28 strength = 9
29 elseif counts[1] == 4 then
30 strength = 8
31 elseif counts[1] == 3 and counts[2] == 2 then
32 strength = 7
33 elseif flush then
34 strength = 6
35 elseif straight then
36 strength = 5
37 elseif counts[1] == 3 then
38 strength = 4
39 elseif counts[1] == 2 and counts[2] == 2 then
40 strength = 3
41 elseif counts[1] == 2 then
42 strength = 2
43 else
44 strength = 1
45 end
46 table.sort(distinct, function(a, b)
47 if frequencies[a] ~= frequencies[b] then return frequencies[a] > frequencies[b] end
48 return a > b
49 end)
50 if wheel and straight then distinct = {5, 4, 3, 2, 1} end
51 return {original = original, strength = strength, sequence = distinct, category = categories[strength]}
52end

54local hands = {}
55local originals = {
56 "9C 9D 5H QH 2C", "4D AH 3C 5S 2H", "KS 8S 2S JS 6S", "7H 7D KC 7S KD", "3H 6C 4S 2D 5C",
57 "JD TC 4H 8C AS", "QC 6D QS 3S 6H", "9H KH 9S 8H 3D",
58}
59for _, original in ipairs(originals) do
60 hands[#hands + 1] = describe(original)
61end
62table.sort(hands, function(a, b)
63 if a.strength ~= b.strength then return a.strength > b.strength end
64 for i = 1, #a.sequence do
65 if a.sequence[i] ~= b.sequence[i] then return a.sequence[i] > b.sequence[i] end
66 end
67 return false
68end)
69for position, hand in ipairs(hands) do
70 io.write(string.format("%d. %s: %s\n", position, hand.original, hand.category))
71end
