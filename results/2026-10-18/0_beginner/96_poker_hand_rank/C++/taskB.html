6Hand evaluate(const std::string &text) {

8 std::array<int, 15> counts{};
9 bool flush = true;
10 for (int i = 0; i < 5; i++) {
11 int rank = static_cast<int>(ranks.find(text[i * 3])) + 2;
12 ++counts[rank];
13 if (text[i * 3 + 1] != text[1])
14 flush = false;
15 }
16 Hand h{text, 0, {}};
17 int maximum = 0, pairs = 0;
18 for (int r = 14; r >= 2; r--)
19 if (counts[r]) {
20 h.key.push_back(r);
21 maximum = std::max(maximum, counts[r]);
22 if (counts[r] == 2)
23 ++pairs;
24 }
25 bool wheel = h.key.size() == 5 && counts[14] && counts[2] && counts[3] && counts[4] && counts[5];
26 bool straight = h.key.size() == 5 && (h.key[0] - h.key[4] == 4 || wheel);
27 h.category = straight && flush ? 8
28 : maximum == 4 ? 7
29 : maximum == 3 && pairs == 1 ? 6
30 : flush ? 5
31 : straight ? 4
32 : pairs == 1 ? 1 : 0;
33 std::sort(h.key.begin(), h.key.end(), [&counts](int a, int b) {
34 return counts[a] != counts[b] ? counts[a] > counts[b] : a > b;
35 });
36 if (wheel && straight)
37 for (int i = 0; i < 5; i++)
38 h.key[i] = 5 - i;
39 return h;
40}
