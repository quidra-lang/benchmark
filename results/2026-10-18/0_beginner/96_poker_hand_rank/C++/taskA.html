7struct Hand {
8 std::string text;
9 int category;
10 std::vector<int> key;
11};

13Hand evaluate(const std::string &text) {
14 std::string ranks = "23456789TJQKA";
15 std::array<int, 15> counts{};
16 bool flush = true;
17 for (int i = 0; i < 5; i++) {
18 int rank = static_cast<int>(ranks.find(text[i * 3])) + 2;
19 ++counts[rank];
20 if (text[i * 3 + 1] != text[1])
21 flush = false;
22 }
23 Hand h{text, 0, {}};
24 int maximum = 0, pairs = 0;
25 for (int r = 14; r >= 2; r--)
26 if (counts[r]) {
27 h.key.push_back(r);
28 maximum = std::max(maximum, counts[r]);
29 if (counts[r] == 2)
30 ++pairs;
31 }
32 bool wheel = h.key.size() == 5 && counts[14] && counts[2] && counts[3] && counts[4] && counts[5];
33 bool straight = h.key.size() == 5 && (h.key[0] - h.key[4] == 4 || wheel);
34 h.category = straight && flush ? 8
35 : maximum == 4 ? 7
36 : maximum == 3 && pairs == 1 ? 6
37 : flush ? 5
38 : straight ? 4
39 : maximum == 3 ? 3
40 : pairs == 2 ? 2
41 : pairs == 1 ? 1 : 0;
42 std::sort(h.key.begin(), h.key.end(), [&counts](int a, int b) {
43 return counts[a] != counts[b] ? counts[a] > counts[b] : a > b;
44 });
45 if (wheel && straight)
46 for (int i = 0; i < 5; i++)
47 h.key[i] = 5 - i;
48 return h;
49}

51int main() {
52 std::vector<Hand> hands;
53 const std::array<std::string, 8> input = {"9C 9D 5H QH 2C", "4D AH 3C 5S 2H", "KS 8S 2S JS 6S", "7H 7D KC 7S KD",
54 "3H 6C 4S 2D 5C", "JD TC 4H 8C AS", "QC 6D QS 3S 6H", "9H KH 9S 8H 3D"};
55 for (const std::string &text : input)
56 hands.push_back(evaluate(text));
57 std::sort(hands.begin(), hands.end(), [](const Hand &a, const Hand &b) {
58 return a.category != b.category ? a.category > b.category : a.key > b.key;
59 });
60 std::array<std::string, 9> categories = {"high card", "one pair", "two pair", "three of a kind",
61 "straight", "flush", "full house", "four of a kind", "straight flush"};
62 for (std::size_t i = 0; i < hands.size(); i++)
63 std::cout << i + 1 << ". " << hands[i].text << ": " << categories[hands[i].category] << '\n';
64}
