const hands = [
  "9C 9D 5H QH 2C", "4D AH 3C 5S 2H", "KS 8S 2S JS 6S", "7H 7D KC 7S KD",
  "3H 6C 4S 2D 5C", "JD TC 4H 8C AS", "QC 6D QS 3S 6H", "9H KH 9S 8H 3D"
];
const categories = [
  "high card", "one pair", "two pair", "three of a kind", "straight",
];
const ranked = hands.map(hand => {
  const cards = hand.split(" "), counts = new Map();
  for (const card of cards) {
    const rank = "23456789TJQKA".indexOf(card[0]) + 2;
    counts.set(rank, (counts.get(rank) ?? 0) + 1);
  }
  const ranks = [...counts.keys()].sort((a, b) => a - b);
  const wheel = ranks.join(",") === "2,3,4,5,14";
  const straight = ranks.length === 5 && (ranks.at(-1) - ranks[0] === 4 || wheel);
  const flush = new Set(cards.map(card => card[1])).size === 1;
  const pattern = [...counts.values()].sort((a, b) => b - a);
  let strength;
  if (straight && flush) strength = 8;
  else if (pattern[0] === 4) strength = 7;
  else if (pattern[0] === 3 && pattern[1] === 2) strength = 6;
  else if (flush) strength = 5;
  else if (straight) strength = 4;
  let sequence = [...counts.keys()].sort((a, b) => counts.get(b) - counts.get(a) || b - a);
  if (wheel && straight) sequence = [5, 4, 3, 2, 1];
  return {strength, sequence, hand};
});
ranked.sort((a, b) => {
  if (a.strength !== b.strength) return b.strength - a.strength;
  for (let i = 0; i < a.sequence.length; i++) {
    if (a.sequence[i] !== b.sequence[i]) return b.sequence[i] - a.sequence[i];
  }
  return 0;
});
ranked.forEach(({strength, hand}, i) => console.log(`${i + 1}. ${hand}: ${categories[strength]}`));
