9public class Main {
10 record Hand(String text, int category, int[] sequence) {}
11
12 static Hand evaluate(String text) {
13 String[] cards = text.split(" ");
14 Map<Integer, Integer> counts = new HashMap<>();
15 for (String card : cards) {
16 counts.merge("23456789TJQKA".indexOf(card.charAt(0)) + 2, 1, Integer::sum);
17 }
18 List<Integer> ranks = new ArrayList<>(counts.keySet());
19 Collections.sort(ranks);
20 boolean wheel = ranks.equals(List.of(2, 3, 4, 5, 14));
21 boolean straight = wheel || (ranks.size() == 5 && ranks.getLast() - ranks.getFirst() == 4);
22 boolean flush = Arrays.stream(cards).allMatch(card -> card.charAt(1) == cards[0].charAt(1));
23 int largest = Collections.max(counts.values());
24 long pairs = counts.values().stream().filter(count -> count == 2).count();
25 int category;
26 if (straight && flush) {
27 category = 8;
28 } else if (largest == 4) {
29 category = 7;
30 } else if (largest == 3 && pairs == 1) {
31 category = 6;
32 } else if (flush) {
33 category = 5;
34 } else if (straight) {
35 category = 4;
36 } else if (largest == 3) {
37 category = 3;
38 } else if (pairs == 2) {
39 category = 2;
40 } else if (pairs == 1) {
41 category = 1;
42 } else {
43 category = 0;
44 }
45 ranks.sort(Comparator.<Integer>comparingInt(counts::get).reversed()
46 .thenComparing(Comparator.reverseOrder()));
47 if (wheel && straight) {
48 ranks = new ArrayList<>(List.of(5, 4, 3, 2, 1));
49 }
50 return new Hand(text, category, ranks.stream().mapToInt(Integer::intValue).toArray());
51 }
52
53 public static void main(String[] args) {
54 String[] input = {"9C 9D 5H QH 2C", "4D AH 3C 5S 2H", "KS 8S 2S JS 6S", "7H 7D KC 7S KD",
55 "3H 6C 4S 2D 5C", "JD TC 4H 8C AS", "QC 6D QS 3S 6H", "9H KH 9S 8H 3D"};
56 String[] categories = {"high card", "one pair", "two pair", "three of a kind", "straight",
57 "flush", "full house", "four of a kind", "straight flush"};
58 List<Hand> hands = new ArrayList<>();
59 for (String text : input) {
60 hands.add(evaluate(text));
61 }
62 hands.sort(Comparator.comparingInt(Hand::category).reversed()
63 .thenComparing(Hand::sequence, (a, b) -> Arrays.compare(b, a)));
64 for (int i = 0; i < hands.size(); i++) {
65 Hand hand = hands.get(i);
66 System.out.println((i + 1) + ". " + hand.text() + ": " + categories[hand.category()]);
67 }
68 }
