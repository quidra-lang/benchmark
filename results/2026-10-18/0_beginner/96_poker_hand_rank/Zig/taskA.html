3const Hand = struct { original: []const u8, category: u8, sequence: [5]u8, len: usize };

5fn rankLess(counts: *const [15]u8, a: u8, b: u8) bool {
6 return if (counts[a] != counts[b]) counts[a] > counts[b] else a > b;
7}

9fn evaluate(original: []const u8) Hand {
10 var counts: [15]u8 = @splat(0);
11 var flush = true;
12 var suit: u8 = 0;
13 var tokens = std.mem.splitScalar(u8, original, ' ');
14 while (tokens.next()) |card| {
15 const rank = std.mem.findScalar(u8, "23456789TJQKA", card[0]).? + 2;
16 counts[rank] += 1;
17 if (suit == 0)
18 suit = card[1];
19 if (card[1] != suit)
20 flush = false;
21 }
22 var sequence: [5]u8 = undefined;
23 var len: usize = 0;
24 var pairs: u8 = 0;
25 var triple = false;
26 var four = false;
27 for (2..15) |r| {
28 if (counts[r] > 0) {
29 sequence[len] = @intCast(r);
30 len += 1;
31 }
32 if (counts[r] == 2)
33 pairs += 1;
34 if (counts[r] == 3)
35 triple = true;
36 if (counts[r] == 4)
37 four = true;
38 }
39 const wheel = len == 5 and counts[14] == 1 and
40 counts[2] == 1 and counts[3] == 1 and counts[4] == 1 and counts[5] == 1;
41 const straight = len == 5 and (wheel or sequence[4] - sequence[0] == 4);
42 std.mem.sort(u8, sequence[0..len], @as(*const [15]u8, &counts), rankLess);
43 if (wheel)
44 sequence = .{ 5, 4, 3, 2, 1 };
45 const category: u8 = if (straight and flush)
46 8
47 else if (four)
48 7
49 else if (triple and pairs == 1)
50 6
51 else if (flush)
52 5
53 else if (straight)
54 4
55 else if (triple)
56 3
57 else if (pairs == 2)
58 2
59 else if (pairs == 1)
60 1
61 else
62 0;
63 return .{ .original = original, .category = category, .sequence = sequence, .len = len };
64}

66fn less(_: void, a: Hand, b: Hand) bool {
67 if (a.category != b.category)
68 return a.category > b.category;
69 for (0..a.len) |i| {
70 if (a.sequence[i] != b.sequence[i])
71 return a.sequence[i] > b.sequence[i];
72 }
73 return false;
74}

77 var output_buffer: [8192]u8 = undefined;
78 var writer = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
79 const out = &writer.interface;
80 const originals = [_][]const u8{

84 var hands: [8]Hand = undefined;
85 for (originals, 0..) |s, i|

89 "high card",

101 try out.flush();
102}
