1struct Hand {
2 text: String,
3 category: usize,
4 key: Vec<usize>,
5}
6fn evaluate(text: &str) -> Hand {
7 let ranks = b"23456789TJQKA";
8 let bytes = text.as_bytes();
9 let mut counts = [0; 15];
10 let mut flush = true;
11 for i in 0..5 {
12 let rank = ranks.iter().position(|&c| c == bytes[i * 3]).unwrap() + 2;
13 counts[rank] += 1;
14 if bytes[i * 3 + 1] != bytes[1] {
15 flush = false;
16 }
17 }
18 let mut key: Vec<usize> = (2..=14).filter(|&r| counts[r] > 0).rev().collect();
19 let maximum = *counts.iter().max().unwrap();
20 let pairs = counts.iter().filter(|&&c| c == 2).count();
21 let wheel = key.len() == 5 && [14, 2, 3, 4, 5].iter().all(|&r| counts[r] > 0);
22 let straight = key.len() == 5 && (key[0] - key[4] == 4 || wheel);
23 let category = if straight && flush {
24 8
25 } else if maximum == 4 {
26 7
27 } else if maximum == 3 && pairs == 1 {
28 6
29 } else if flush {
30 5
31 } else if straight {
32 4
33 } else if maximum == 3 {
34 } else if pairs == 2 {
35 } else if pairs == 1 {
36 } else {
37 };
38 key.sort_by(|a, b| counts[*b].cmp(&counts[*a]).then(b.cmp(a)));
39 if wheel && straight {
40 key = (1..=5).rev().collect();
41 }
42 Hand {
43 text: text.to_string(),
44 category,
45 key,
46 }
47}
48 let mut hands: Vec<Hand> = [
49 "4D AH 3C 5S 2H",
50 "3H 6C 4S 2D 5C",
51 ]
52 .iter()
53 .map(|s| evaluate(s))
54 .collect();
55 hands.sort_by(|a, b| b.category.cmp(&a.category).then(b.key.cmp(&a.key)));
56 let categories = [
57 "high card",
58 "one pair",
59 "two pair",
60 "three of a kind",
61 "straight",
62 ];
63 for (i, h) in hands.iter().enumerate() {
64 println!("{}. {}: {}", i + 1, h.text, categories[h.category]);
65 }
