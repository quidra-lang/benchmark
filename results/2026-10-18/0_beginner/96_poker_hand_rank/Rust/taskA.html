1struct Hand {
2 text: String,
3 category: usize,
4 key: Vec<usize>,
5}
6
7fn evaluate(text: &str) -> Hand {
8 let ranks = b"23456789TJQKA";
9 let bytes = text.as_bytes();
10 let mut counts = [0; 15];
11 let mut flush = true;
12 for i in 0..5 {
13 let rank = ranks.iter().position(|&c| c == bytes[i * 3]).unwrap() + 2;
14 counts[rank] += 1;
15 if bytes[i * 3 + 1] != bytes[1] {
16 flush = false;
17 }
18 }
19 let mut key: Vec<usize> = (2..=14).filter(|&r| counts[r] > 0).rev().collect();
20 let maximum = *counts.iter().max().unwrap();
21 let pairs = counts.iter().filter(|&&c| c == 2).count();
22 let wheel = key.len() == 5 && [14, 2, 3, 4, 5].iter().all(|&r| counts[r] > 0);
23 let straight = key.len() == 5 && (key[0] - key[4] == 4 || wheel);
24 let category = if straight && flush {
25 8
26 } else if maximum == 4 {
27 7
28 } else if maximum == 3 && pairs == 1 {
29 6
30 } else if flush {
31 5
32 } else if straight {
33 4
34 } else if maximum == 3 {
35 3
36 } else if pairs == 2 {
37 2
38 } else if pairs == 1 {
39 1
40 } else {
41 0
42 };
43 key.sort_by(|a, b| counts[*b].cmp(&counts[*a]).then(b.cmp(a)));
44 if wheel && straight {
45 key = (1..=5).rev().collect();
46 }
47 Hand {
48 text: text.to_string(),
49 category,
50 key,
51 }
52}

54fn main() {
55 let mut hands: Vec<Hand> = [
56 "9C 9D 5H QH 2C",
57 "4D AH 3C 5S 2H",
58 "KS 8S 2S JS 6S",
59 "7H 7D KC 7S KD",

62 "QC 6D QS 3S 6H",
63 "9H KH 9S 8H 3D",
64 ]
65 .iter()
66 .map(|s| evaluate(s))
67 .collect();
68 hands.sort_by(|a, b| b.category.cmp(&a.category).then(b.key.cmp(&a.key)));
69 let categories = [
70 "high card",
71 "one pair",
72 "two pair",
73 "three of a kind",
74 "straight",
75 "flush",
76 "full house",
77 "four of a kind",
78 "straight flush",
79 ];
80 for (i, h) in hands.iter().enumerate() {
81 println!("{}. {}: {}", i + 1, h.text, categories[h.category]);
82 }
83}
