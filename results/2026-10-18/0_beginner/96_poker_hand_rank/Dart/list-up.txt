class Hand {
  final String original;
  final int category;
  final List<int> sequence;
  Hand(this.original, this.category, this.sequence);
}
Hand evaluate(String original) {
  final cards = original.split(' ');
  final counts = List.filled(15, 0);
  int rank(String c) => '23456789TJQKA'.indexOf(c) + 2;
  for (final card in cards) {
    counts[rank(card[0])]++;
  }
  final flush = cards.every((c) => c[1] == cards.first[1]);
  final ascending = [
    for (var r = 2; r <= 14; r++)
      if (counts[r] > 0) r,
  ];
  final wheel =
      ascending.length == 5 && [14, 2, 3, 4, 5].every((r) => counts[r] == 1);
  final straight =
      ascending.length == 5 && (wheel || ascending.last - ascending.first == 4);
  final pairs = counts.where((c) => c == 2).length;
  final triple = counts.contains(3);
  final four = counts.contains(4);
  final sequence = ascending
    ..sort((a, b) {
      final c = counts[b].compareTo(counts[a]);
      return c != 0 ? c : b.compareTo(a);
    });
  if (wheel) {
    sequence.clear();
    sequence.addAll([5, 4, 3, 2, 1]);
  }
  final category = straight && flush
      ? 8
      : four
      ? 7
      : triple && pairs == 1
      ? 6
      : flush
      ? 5
      : straight
      ? 4
      : 0;
  return Hand(original, category, sequence);
}
