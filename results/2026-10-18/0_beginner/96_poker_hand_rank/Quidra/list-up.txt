import collections
import text
int[] score(string hand)
    ranks := text.characters("23456789TJQKA")
    string[] cards := hand.split(" ")
    int[] counts := array(15, fill = 0)
    bool is_flush := true
    for string card := cards[]
        nat rank_index := ranks.index(card[0])
        counts[rank_index + 2] += 1
        if card[1] != cards[0][1]
            is_flush = false
    int[] present := [for nat rank := counts.indices()[] if counts[rank] > 0 then int(rank)]
    int[] pattern := counts[present].sorted(descending = true)
    bool wheel := present == [2, 3, 4, 5, 14]
    bool straight := len(present) == 5 and (present[4] - present[0] == 4 or wheel)
    int category := 0
    if straight and is_flush
        category = 8
    elif pattern[0] == 4
        category = 7
    elif pattern == [3, 2]
        category = 6
    elif is_flush
        category = 5
    elif straight
        category = 4
    int[] sequence := present.sorted(by = (counts[present], present), descending = true)
    if straight and wheel
        sequence = [5, 4, 3, 2, 1]
    return [category].concat(sequence)
string[] hands := [
    "4D AH 3C 5S 2H",
    "3H 6C 4S 2D 5C",
int[][] keys := [for string hand := hands[] then score(hand)]
nat[] order := collections.lex_order(keys, descending = true)
for nat best := order[nat position]
    print.line("{position + 1}. {hands[best]}: {categories[keys[best][0]]}")
