import collections
import text
int[] score(string hand)
    ranks := text.characters("23456789TJQKA")
    cards := hand.split(" ")
    counts := array(15, fill = 0)
    is_flush := true
    for card := cards[]
        rank_index := ranks.index(card[0])
        counts[rank_index + 2] += 1
        if card[1] != cards[0][1]
            is_flush = false
    present := [for rank := counts.indices()[] if counts[rank] > 0 then int(rank)]
    pattern := counts[present].sorted(descending = true)
    wheel := present == [2, 3, 4, 5, 14]
    straight := present.len() == 5 and (present[4] - present[0] == 4 or wheel)
    category := 0
    if straight and is_flush
        category = 8
    elif pattern[0] == 4
        category = 7
    elif pattern == [3, 2]
        category = 6
    elif is_flush
        category = 5
    elif straight
        category = 4
    sequence := present.sorted(by = (counts[present], present), descending = true)
    if straight and wheel
        sequence = [5, 4, 3, 2, 1]
    return [category].concat(sequence)
