    int[] bytes = text.codepoints()
    string result = ""
    for 0 <= int i < int(len(bytes)) step 3
        int remaining = int(len(bytes)) - i
        result += alphabet[value / 262144]
        result += alphabet[value / 4096 % 64]
        result += if remaining >= 2 then alphabet[value / 64 % 64] else "="
        result += if remaining >= 3 then alphabet[value % 64] else "="
    return result
