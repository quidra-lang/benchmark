    int[] bytes := text.codepoints()
    string result := ""
    for nat i < len(bytes) step 3
        nat remaining := len(bytes) - i
        int[] divisors := [262144, 4096, 64, 1]
        for int divisor := divisors[nat digit]
            result += if digit <= remaining then alphabet[value / divisor % 64] else "="
    return result
