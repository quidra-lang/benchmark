1using System;
2using System.Text;
3
4string alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
5foreach (string text in new[] { "Man", "Hello, World!", "", "<b>bold</b>", "a+b=c?" })
6{
7 var encoded = new StringBuilder();
8 for (int i = 0; i < text.Length; i += 3)
9 {
10 int remaining = text.Length - i;
11 int bits = text[i] << 16;
12 if (remaining > 1) bits |= text[i + 1] << 8;
13 if (remaining > 2) bits |= text[i + 2];
14 encoded.Append(alphabet[(bits >> 18) & 63]);
15 encoded.Append(alphabet[(bits >> 12) & 63]);
16 encoded.Append(remaining > 1 ? alphabet[(bits >> 6) & 63] : '=');
17 encoded.Append(remaining > 2 ? alphabet[bits & 63] : '=');
18 }
19 Console.WriteLine($"\"{text}\" -> \"{encoded}\"");
20}
