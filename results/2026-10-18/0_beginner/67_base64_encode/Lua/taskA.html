1local function encode64(source)
2 local alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
3 local output = {}
4 for first = 1, #source, 3 do
5 local remaining = math.min(3, #source - first + 1)
6 local value = 0
7 for offset = 0, 2 do value = value * 256 + (offset < remaining and source:byte(first + offset) or 0) end
8 for i, divisor in ipairs({262144, 4096, 64, 1}) do
9 local index = (value // divisor) % 64 + 1
10 output[#output + 1] = i <= remaining + 1 and alphabet:sub(index, index) or "="
11 end
12 end
13 return table.concat(output)
14end

16for _, source in ipairs({"Man", "Hello, World!", "", "<b>bold</b>", "a+b=c?"}) do
17 io.write(string.format('"%s" -> "%s"\n', source, encode64(source)))
18end
