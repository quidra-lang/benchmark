local limit, prime = 200, {}
for n = 0, limit do prime[n] = n >= 2 end
for p = 2, limit do
    if prime[p] then
        for multiple = 2 * p, limit, p do prime[multiple] = false end
    end
end
for n = 2, limit do
    if prime[n] then primes[#primes + 1] = n end
end
for first = 1, #primes, 10 do
    io.write(table.concat(primes, " ", first, math.min(first + 9, #primes)), "\n")
