import collections
import text
prime := array(201, fill = true)
for 2 <= int p <= 200
    if prime[p]
        for 2 * p <= int multiple <= 200 step p
            prime[multiple] = false
int[] primes := []
for 2 <= int n <= 200
    if prime[n]
        .primes.push(n)
groups := primes.chunks(10)
for int[] group := groups[]
    print.line(text.join(group, " "))
