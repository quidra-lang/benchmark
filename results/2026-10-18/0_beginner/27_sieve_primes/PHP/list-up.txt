$prime = array_fill(0, $limit + 1, true);
for ($p = 2; $p <= $limit; $p++) {
    if ($prime[$p]) {
        for ($multiple = 2 * $p; $multiple <= $limit; $multiple += $p) {
            $prime[$multiple] = false;
        }
    }
}
for ($n = 2; $n <= $limit; $n++) {
    if ($prime[$n]) {
        $primes[] = $n;
    }
}
