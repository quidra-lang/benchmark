    const char *pairs[][2] = {{"kitten", "sitting"}, {"flaw", "lawn"}, {"listen", "silent"},
        {"intention", "execution"}, {"algorithm", "altruistic"}};
    for (int k = 0; k < 5; k++) {
        const char *a = pairs[k][0], *b = pairs[k][1];
        int m = (int)strlen(a), n = (int)strlen(b), dp[16][16];
        for (int i = 0; i <= m; i++)
            dp[i][0] = i;
        for (int j = 0; j <= n; j++)
            dp[0][j] = j;
        for (int i = 1; i <= m; i++)
            for (int j = 1; j <= n; j++) {
                int deletion = dp[i - 1][j] + 1, insertion = dp[i][j - 1] + 1;
                int replacement = dp[i - 1][j - 1] + (a[i - 1] != b[j - 1]);
                dp[i][j] = deletion < insertion ? deletion : insertion;
                if (replacement < dp[i][j])
                    dp[i][j] = replacement;
            }
    }
