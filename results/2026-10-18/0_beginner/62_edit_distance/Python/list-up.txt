pairs = [
    ("flaw", "lawn"),
]
for a, b in pairs:
    dp = [[i] + [0] * len(b) for i in range(len(a) + 1)]
    dp[0] = list(range(len(b) + 1))
    for i in range(1, len(a) + 1):
        for j in range(1, len(b) + 1):
            cost = int(a[i - 1] != b[j - 1])
            dp[i][j] = min(dp[i - 1][j] + 1, dp[i][j - 1] + 1, dp[i - 1][j - 1] + cost)
