1function edit_distance(a, b)
2 table = zeros(Int, length(a)+1, length(b)+1)
3 table[:, 1] = 0:length(a)
4 table[1, :] = 0:length(b)
5 for i = 1:length(a), j = 1:length(b)
6 cost = a[i] == b[j] ? 0 : 1
7 table[i+1, j+1] = min(table[i, j+1]+1, table[i+1, j]+1, table[i, j]+cost)
8 end
9 return table[end, end]
10end
