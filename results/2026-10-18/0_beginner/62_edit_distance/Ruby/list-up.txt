pairs = [
  ['kitten', 'sitting'], ['flaw', 'lawn'], ['listen', 'silent'],
pairs.each do |a, b|
  dp = Array.new(a.length + 1) { Array.new(b.length + 1, 0) }
  (0..a.length).each { |i| dp[i][0] = i }
  (0..b.length).each { |j| dp[0][j] = j }
  (1..a.length).each do |i|
    (1..b.length).each do |j|
      cost = a[i - 1] == b[j - 1] ? 0 : 1
      dp[i][j] = [dp[i - 1][j] + 1, dp[i][j - 1] + 1, dp[i - 1][j - 1] + cost].min
    end
  end
end
