string[][] pairs = [
    ["kitten", "sitting"], ["flaw", "lawn"], ["listen", "silent"], ["intention", "execution"],
]
for string[] pair = pairs[]
    string a = pair[0]
    string b = pair[1]
    int rows = int(len(a))
    int columns = int(len(b))
    int[][] dp = array(nat(rows + 1), fill = array(nat(columns + 1), fill = 0))
    for 0 <= int i <= rows
        dp[i][0] = i
    for 0 <= int j <= columns
        dp[0][j] = j
    for 1 <= int i <= rows
        for 1 <= int j <= columns
            int cost = if a[i - 1] == b[j - 1] then 0 else 1
            dp[i][j] = math.min(math.min(dp[i - 1][j] + 1, dp[i][j - 1] + 1), dp[i - 1][j - 1] + cost)
