import math
(string, string)[] pairs := [
    ("flaw", "lawn"),
]
for (string a, string b) := pairs[]
    int rows := int(len(a))
    int columns := int(len(b))
    int[][] dp := [for int i := (0:rows)[] then [i].concat(array(nat(columns), fill = 0))]
    dp[0] = (0:columns)
    for 1 <= int i <= rows
        for 1 <= int j <= columns
            int cost := if a[i - 1] == b[j - 1] then 0 else 1
            dp[i][j] = [dp[i - 1][j] + 1, dp[i][j - 1] + 1, dp[i - 1][j - 1] + cost].min_all()
