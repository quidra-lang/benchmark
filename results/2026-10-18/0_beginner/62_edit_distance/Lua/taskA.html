1local function edit_distance(a, b)
2 local table_values = {}
3 for i = 0, #a do table_values[i] = {[0] = i} end
4 for j = 0, #b do table_values[0][j] = j end
5 for i = 1, #a do
6 for j = 1, #b do
7 local cost = a:sub(i, i) == b:sub(j, j) and 0 or 1
8 table_values[i][j] = math.min(table_values[i - 1][j] + 1, table_values[i][j - 1] + 1,
9 table_values[i - 1][j - 1] + cost)
10 end
11 end
12 return table_values[#a][#b]
13end
