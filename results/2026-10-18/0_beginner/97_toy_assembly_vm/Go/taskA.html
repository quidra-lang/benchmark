9func value(token string, registers [4]int64) int64 {
10 if token[0] >= 'A' && token[0] <= 'D' {
11 return registers[token[0]-'A']
12 }
13 n, _ := strconv.ParseInt(token, 10, 64)
14 return n
15}

17func main() {
18 program := []string{"SET A 1", "SET B 1", "SET D 8", "MUL A B", "PRINT A", "ADD B 1", "SUB D 1",
19 "JNZ D -4", "SET A 1071", "SET B 462", "SET C A", "MOD C B", "SET A B", "SET B C", "JNZ B -4", "PRINT A",
20 "HALT"}

35 pc++
36 }

38 }
39 if op == "PRINT" {
40 fmt.Println(value(parts[1], registers))
41 } else {
42 r := int(parts[1][0] - 'A')
43 x := value(parts[2], registers)
44 switch op {
45 case "SET":
46 registers[r] = x
47 case "ADD":
48 registers[r] += x

50 registers[r] -= x
