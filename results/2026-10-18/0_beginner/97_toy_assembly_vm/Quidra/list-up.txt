while 0 <= pc < int(len(program))
    fields := program[pc].split(" ")
    operation := fields[0]
    executed += 1
    if operation == "HALT"
        break
    if operation == "PRINT"
        println(value_of(fields[1], registers))
    elif operation == "JNZ"
        pc += if value_of(fields[1], registers) != 0 then offset else 1
        continue
println("executed: {executed}")
