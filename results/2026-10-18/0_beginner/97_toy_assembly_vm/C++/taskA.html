1#include <iostream>
2#include <string>
3#include <array>
4#include <sstream>
5
6int value(const std::string &token, const std::array<int, 4> &registers) {
7 return token[0] >= 'A' && token[0] <= 'D' ? registers[token[0] - 'A'] : std::stoi(token);
8}
9
10int main() {
11 std::array<std::string, 17> program = {"SET A 1", "SET B 1", "SET D 8", "MUL A B", "PRINT A",
12 "ADD B 1", "SUB D 1", "JNZ D -4", "SET A 1071", "SET B 462", "SET C A", "MOD C B", "SET A B", "SET B C",
13 "JNZ B -4", "PRINT A", "HALT"};
14 std::array<int, 4> registers{};
15 int pc = 0, executed = 0;
16 while (pc >= 0 && pc < 17) {
17 std::istringstream input(program[pc]);
18 std::string op, a, b;
19 input >> op >> a >> b;
20 ++executed;
21 if (op == "HALT")
22 break;
23 if (op == "JNZ") {
24 pc += value(a, registers) != 0 ? std::stoi(b) : 1;
25 continue;
26 }
27 if (op == "PRINT") {
28 std::cout << value(a, registers) << '\n';
29 } else {
30 int r = a[0] - 'A', x = value(b, registers);
31 if (op == "SET")
32 registers[r] = x;
33 else if (op == "ADD")
34 registers[r] += x;
35 else if (op == "SUB")
36 registers[r] -= x;
37 else if (op == "MUL")
38 registers[r] *= x;
39 else if (op == "MOD")
40 registers[r] %= x;
41 }
42 ++pc;
43 }
44 std::cout << "executed: " << executed << '\n';
45}
