1using System;
2using System.Globalization;
3
4CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
5
6static int Value(string token, int[] registers)
7{
8 char first = token[0];
9 return first >= 'A' && first <= 'D' ? registers[first - 'A'] : int.Parse(token, CultureInfo.InvariantCulture);
10}
11
12string[] program = { "SET A 1", "SET B 1", "SET D 8", "MUL A B", "PRINT A", "ADD B 1", "SUB D 1",
13 "JNZ D -4", "SET A 1071", "SET B 462", "SET C A", "MOD C B", "SET A B",
14 "SET B C", "JNZ B -4", "PRINT A", "HALT" };
15int[] registers = new int[4];
16int pc = 0, executed = 0;
17while (pc >= 0 && pc < program.Length)
18{
19 string[] parts = program[pc].Split(' ');
20 executed++;
21 if (parts[0] == "HALT") break;
22 if (parts[0] == "JNZ")
23 {
24 pc += Value(parts[1], registers) != 0 ? int.Parse(parts[2], CultureInfo.InvariantCulture) : 1;
25 continue;
26 }
27 if (parts[0] == "PRINT")
28 {
29 Console.WriteLine(Value(parts[1], registers));
30 }
31 else
32 {
33 int target = parts[1][0] - 'A', operand = Value(parts[2], registers);
34 switch (parts[0])
35 {
36 case "SET":
37 registers[target] = operand;
38 break;
39 case "ADD":
40 registers[target] += operand;
41 break;
42 case "SUB":
43 registers[target] -= operand;
44 break;
45 case "MUL":
46 registers[target] *= operand;
47 break;
48 case "MOD":
49 registers[target] %= operand;
50 break;
51 }
52 }
53 pc++;
54}
55Console.WriteLine($"executed: {executed}");
