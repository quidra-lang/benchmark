while 0 <= pc < len(program):
    parts = program[pc].split()
    operation = parts[0]
    executed += 1
    if operation == "HALT":
        break
    if operation == "JNZ":
        pc += int(parts[2]) if value(parts[1]) != 0 else 1
        continue
    if operation == "PRINT":
        print(value(parts[1]))
print(f"executed: {executed}")
