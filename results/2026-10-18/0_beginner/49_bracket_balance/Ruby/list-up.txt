  stack = []
  error = nil
  text.each_char.with_index do |char, i|
    if '([{'.include?(char)
      stack << char
    elsif matching.key?(char)
      if stack.empty? || stack[-1] != matching[char]
        error = i
        break
      end
      stack.pop
    end
  end
  error = text.length if error.nil? && !stack.empty?
