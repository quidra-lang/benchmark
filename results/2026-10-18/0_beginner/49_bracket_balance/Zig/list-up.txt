        var stack: [100]u8 = undefined;
        var len: usize = 0;
        var err: ?usize = null;
        for (s, 0..) |c, i| {
            if (c == '(' or c == '[' or c == '{') {
                stack[len] = c;
                len += 1;
            } else if (c == ')' or c == ']' or c == '}') {
                const expected: u8 = if (c == ')') '(' else if (c == ']') '[' else '{';
                if (len == 0 or stack[len - 1] != expected) {
                    err = i;
                    break;
                }
                len -= 1;
            }
        }
        if (err == null and len > 0)
            err = s.len;
