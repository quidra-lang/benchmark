        val stack = ArrayDeque<Char>()
        var error = -1
        for ((i, ch) in text.withIndex()) {
            val close = ")]}".indexOf(ch)
            if (ch in "([{") {
                stack.add(ch)
            } else if (close >= 0) {
                if (stack.isEmpty() || stack.last() != "([{"[close]) {
                    error = i
                    break
                }
                stack.removeLast()
            }
        }
        if (error < 0 && stack.isNotEmpty()) error = text.length
