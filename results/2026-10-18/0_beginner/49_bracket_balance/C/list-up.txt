        char stack[128];
        int top = 0, error = -1, len = (int)strlen(s);
        for (int i = 0; i < len; i++) {
            char c = s[i];
            if (c == '(' || c == '[' || c == '{') {
                stack[top++] = c;
            } else if (c == ')' || c == ']' || c == '}') {
                char expected = c == ')' ? '(' : c == ']' ? '[' : '{';
                if (!top || stack[top - 1] != expected) {
                    error = i;
                    break;
                }
                top--;
            }
        }
        if (error < 0 && top)
            error = len;
