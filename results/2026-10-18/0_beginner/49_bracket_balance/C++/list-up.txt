        std::vector<char> stack;
        int error = -1;
        for (int i = 0; i < static_cast<int>(s.size()); i++) {
            char c = s[i];
            if (c == '(' || c == '[' || c == '{') {
                stack.push_back(c);
            } else if (c == ')' || c == ']' || c == '}') {
                char expected = c == ')' ? '(' : c == ']' ? '[' : '{';
                if (stack.empty() || stack.back() != expected) {
                    error = i;
                    break;
                }
                stack.pop_back();
            }
        }
        if (error < 0 && !stack.empty())
            error = static_cast<int>(s.size());
