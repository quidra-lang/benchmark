    var stack = [Character](), error = -1
    for (i, ch) in text.enumerated() {
        if openings.contains(ch) {
            stack.append(ch)
        } else if let close = closings.firstIndex(of: ch) {
            if stack.last != openings[close] {
                error = i
                break
            }
            stack.removeLast()
        }
    }
    if error < 0 && !stack.isEmpty { error = text.count }
