  const stack: string[] = [];
  let error = -1;
  for (let i = 0; i < text.length; i++) {
    const char = text[i];
    if ("([{".includes(char)) {
      stack.push(char);
    } else if (matching.has(char)) {
      if (stack.length === 0 || stack.at(-1) !== matching.get(char)) {
        error = i;
        break;
      }
      stack.pop();
    }
  }
