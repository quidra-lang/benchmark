    string stack = ""
    int failure = -1
    for nat i < len(text)
        string character = text[i]
        if openings.contains(character)
            stack += character
        else
            match closings.find(character)
                nat position
                    if len(stack) == 0 or stack[len(stack) - 1] != openings[position]
                        failure = int(i)
                        break
                    stack = stack[:<len(stack) - 1]
    if failure == -1 and len(stack) > 0
        failure = int(len(text))
