closings := ")]}}"
    stack := ""
    failure := -1
    for: i < #text
        character := text[i]
        if openings.contains(character)
            stack += character
        else
            match closings.find(character)
                nat position
                    if #stack == 0 or stack[#stack - 1] != openings[position]
                        failure = int(i)
                        break
                    stack = stack[:<#stack - 1]
    if failure == -1 and #stack > 0
        failure = int(#text)
