openings := "([{{"
closings := ")]}}"
    stack := ""
    failure := -1
    for: i < text.len()
        character := text[i]
        if openings.contains(character)
            stack += character
        else
            match closings.find(character)
                nat position
                    if stack.len() == 0 or stack[stack.len() - 1] != openings[position]
                        failure = int(i)
                        break
                    stack = stack[:<stack.len() - 1]
    if failure == -1 and stack.len() > 0
        failure = int(text.len())
