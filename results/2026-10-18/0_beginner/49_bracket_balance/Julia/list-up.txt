    stack = Char[]
    error_index = -1
    for (position, c) in enumerate(text)
        if c in ('(', '[', '{')
            push!(stack, c)
        elseif haskey(matching, c)
            if isempty(stack) || stack[end] != matching[c]
                error_index = position-1
                break
            end
            pop!(stack)
        end
    end
    if error_index < 0 && !isempty(stack)
        error_index = length(text)
