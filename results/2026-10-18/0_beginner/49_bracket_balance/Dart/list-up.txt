    final stack = <String>[];
    int? error;
    for (var i = 0; i < s.length; i++) {
      final c = s[i];
      if ('([{'.contains(c)) {
        stack.add(c);
      } else if (')]}'.contains(c)) {
        final expected = '([{'[')]}'.indexOf(c)];
        if (stack.isEmpty || stack.last != expected) {
          error = i;
          break;
        }
        stack.removeLast();
      }
    }
    if (error == null && stack.isNotEmpty) error = s.length;
