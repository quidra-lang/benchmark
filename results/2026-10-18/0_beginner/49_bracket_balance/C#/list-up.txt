    var stack = new Stack<char>();
    int error = -1;
    for (int i = 0; i < text.Length; i++)
    {
        char ch = text[i];
        int close = ")]}".IndexOf(ch);
        if ("([{".IndexOf(ch) >= 0)
        {
            stack.Push(ch);
        }
        else if (close >= 0)
        {
            if (stack.Count == 0 || stack.Peek() != "([{"[close])
            {
                error = i;
                break;
            }
            stack.Pop();
        }
    }
    if (error < 0 && stack.Count > 0) error = text.Length;
