3fn solve(board: *[81]u8, placements: *u64) bool {
4 const cell = std.mem.findScalar(u8, board, 0) orelse return true;
5 const r = cell / 9;
6 const c = cell % 9;
7 var digit: u8 = 1;
8 while (digit <= 9) : (digit += 1) {
9 var valid = true;
10 for (0..9) |k| {
11 if (board[r * 9 + k] == digit or board[k * 9 + c] == digit or
12 board[(r / 3 * 3 + k / 3) * 9 + c / 3 * 3 + k % 3] == digit)
13 {
14 valid = false;
15 break;
16 }
17 }
18 if (valid) {
19 board[cell] = digit;
20 placements.* += 1;
21 if (solve(board, placements))
22 return true;
23 board[cell] = 0;
24 }
25 }
26 return false;
27}
