5func solve(board *[9][9]int, placements *uint64) bool {
6 row, col := -1, -1
7 for r := 0; r < 9 && row < 0; r++ {
8 for c := 0; c < 9; c++ {
9 if board[r][c] == 0 {
10 row, col = r, c
11 break
12 }
13 }
14 }
15 if row < 0 {
16 return true
17 }
18 for d := 1; d <= 9; d++ {
19 safe := true
20 for k := 0; k < 9; k++ {
21 if board[row][k] == d || board[k][col] == d {
22 safe = false
23 }
24 }
25 for r := row / 3 * 3; r < row/3*3+3; r++ {
26 for c := col / 3 * 3; c < col/3*3+3; c++ {
27 if board[r][c] == d {
28 safe = false
29 }
30 }
31 }
32 if safe {
33 board[row][col] = d
34 (*placements)++
35 if solve(board, placements) {
36 return true
37 }
38 board[row][col] = 0
39 }
40 }
41 return false
42}
