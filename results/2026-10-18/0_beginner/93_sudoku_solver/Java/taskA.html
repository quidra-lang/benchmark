4 static boolean solve(int[][] board, long[] placements) {
5 for (int r = 0; r < 9; r++) {
6 for (int c = 0; c < 9; c++) {
7 if (board[r][c] != 0) {
8 continue;
9 }
10 for (int digit = 1; digit <= 9; digit++) {
11 boolean valid = true;
12 for (int i = 0; i < 9; i++) {
13 if (board[r][i] == digit || board[i][c] == digit
14 || board[r / 3 * 3 + i / 3][c / 3 * 3 + i % 3] == digit) {
15 valid = false;
16 }
17 }
18 if (!valid) {
19 continue;
20 }
21 board[r][c] = digit;
22 placements[0]++;
23 if (solve(board, placements)) {
24 return true;
25 }
26 board[r][c] = 0;
27 }
28 return false;
29 }
30 }
31 return true;
32 }
