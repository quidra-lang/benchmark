import text
string encode(string source)
    string result = ""
    for (string character, nat count) = text.runs(source)[]
        result += "{count}{character}"
    return result
string decode(string encoded)
    string result = ""
    text.Cursor input(encoded)
    while not input.done()
        nat count = .input.take_nat()
        string character = .input.take_char()
        result += string.repeat(character, count)
    return result
string[] values = [
    "AAABBBCCCD", "WWWWWWWWWWWWBWWWWWWWWWWWWBBB", "ALGORITHM", "MISSISSIPPI",
for string value = values[]
    string encoded = encode(value)
    print.line("{value} -> {encoded} -> {decode(encoded)}")
