1#include <iostream>
2#include <string>
3#include <vector>
4#include <array>
5#include <algorithm>
6#include <map>
7#include <sstream>
8
9int main() {
10 std::array<std::string, 13> lines = {"date,region,product,quantity,unit_price", "2026-04-01,north,notebook,10,200",
11 "2026-04-01,east,marker,6,150", "2026-04-01,south,eraser,25,80", "2026-04-02,west,stapler,4,600",
12 "2026-04-02,north,marker,8,150", "2026-04-02,east,notebook,9,200", "2026-04-03,south,stapler,3,600",
13 "2026-04-03,west,eraser,15,80", "2026-04-03,north,eraser,20,80", "2026-04-04,east,stapler,5,600",
14 "2026-04-04,south,notebook,6,200", "2026-04-04,west,marker,8,150"};
15 std::map<std::string, int> regions, products, days;
16 int total = 0;
17 for (int i = 1; i < 13; i++) {
18 std::istringstream input(lines[i]);
19 std::array<std::string, 5> fields;
20 for (std::string &field : fields)
21 std::getline(input, field, ',');
22 int quantity = std::stoi(fields[3]), amount = quantity * std::stoi(fields[4]);
23 total += amount;
24 regions[fields[1]] += amount;
25 products[fields[2]] += quantity;
26 days[fields[0]] += amount;
27 }
28 std::vector<std::pair<std::string, int>> ranked(regions.begin(), regions.end());
29 std::sort(ranked.begin(), ranked.end(), [](const auto &a, const auto &b) {
30 return a.second != b.second ? a.second > b.second : a.first < b.first;
31 });
32 std::cout << "total: " << total << '\n';
33 for (const auto &[name, amount] : ranked)
34 std::cout << name << ": " << amount << '\n';
35 for (const auto &[name, quantity] : products)
36 std::cout << name << ": " << quantity << '\n';
37 auto best = std::max_element(days.begin(), days.end(), [](const auto &a, const auto &b) {
38 return a.second < b.second;
39 });
40 std::cout << "best day: " << best->first << " (" << best->second << ")\n";
41}
