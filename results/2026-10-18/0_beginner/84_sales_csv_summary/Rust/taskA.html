1use std::collections::BTreeMap;
2
3fn main() {
4 let lines = [
5 "date,region,product,quantity,unit_price",
6 "2026-04-01,north,notebook,10,200",
7 "2026-04-01,east,marker,6,150",
8 "2026-04-01,south,eraser,25,80",
9 "2026-04-02,west,stapler,4,600",
10 "2026-04-02,north,marker,8,150",
11 "2026-04-02,east,notebook,9,200",
12 "2026-04-03,south,stapler,3,600",
13 "2026-04-03,west,eraser,15,80",
14 "2026-04-03,north,eraser,20,80",
15 "2026-04-04,east,stapler,5,600",
16 "2026-04-04,south,notebook,6,200",
17 "2026-04-04,west,marker,8,150",
18 ];
19 let (mut regions, mut products, mut days) = (BTreeMap::new(), BTreeMap::new(), BTreeMap::new());
20 let mut total = 0;
21 for line in &lines[1..] {
22 let f: Vec<&str> = line.split(',').collect();
23 let quantity: i32 = f[3].parse().unwrap();
24 let price: i32 = f[4].parse().unwrap();
25 let amount = quantity * price;
26 total += amount;
27 *regions.entry(f[1]).or_insert(0) += amount;
28 *products.entry(f[2]).or_insert(0) += quantity;
29 *days.entry(f[0]).or_insert(0) += amount;
30 }
31 let mut ranked: Vec<(&str, i32)> = regions.into_iter().collect();
32 ranked.sort_by(|a, b| b.1.cmp(&a.1).then(a.0.cmp(b.0)));
33 println!("total: {}", total);
34 for (name, amount) in ranked {
35 println!("{}: {}", name, amount);
36 }
37 for (name, quantity) in products {
38 println!("{}: {}", name, quantity);
39 }
40 let (best, amount) = days
41 .into_iter()
42 .rev()
43 .max_by_key(|(_, sales)| *sales)
44 .unwrap();
45 println!("best day: {} ({})", best, amount);
46}
