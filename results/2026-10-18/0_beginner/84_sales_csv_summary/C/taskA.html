5typedef struct {
6 char name[32];
7 int total;
8} Entry;
9
10void add(Entry *entries, int *count, const char *name, int amount) {
11 int i = 0;
12 while (i < *count && strcmp(entries[i].name, name))
13 i++;
14 if (i == *count) {
15 strcpy(entries[i].name, name);
16 entries[i].total = 0;
17 (*count)++;
18 }
19 entries[i].total += amount;
20}
21
22int by_total(const void *a, const void *b) {
23 const Entry *x = a, *y = b;
24 return x->total != y->total ? y->total - x->total : strcmp(x->name, y->name);
25}
26
27int by_name(const void *a, const void *b) {
28 return strcmp(((const Entry *)a)->name, ((const Entry *)b)->name);
29}
