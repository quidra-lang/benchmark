int grouped_total(string key, string[] labels, int[] amounts)
    int[] selected = [for nat i < len(labels) if labels[i] == key then amounts[i]]
    return selected.sum()
string[] csv = [
    "2026-04-01,north,notebook,10,200",
    "2026-04-02,north,marker,8,150",
    "2026-04-02,east,notebook,9,200",
    "2026-04-03,north,eraser,20,80",
    "2026-04-04,south,notebook,6,200",
]
string[] regions = []
string[] products = []
int[] quantities = []
int[] amounts = []
set.Set<string> region_set = set.Set<string>()
set.Set<string> product_set = set.Set<string>()
for 1 <= int i < int(len(csv))
    string[] fields = csv[i].split(",")
    int quantity = int.parse(fields[3])
    int price = int.parse(fields[4])
    int amount = quantity * price
    regions = regions.append(fields[1])
    products = products.append(fields[2])
    quantities = quantities.append(quantity)
    amounts = amounts.append(amount)
    .region_set.add(fields[1])
    .product_set.add(fields[2])
string[] region_names = region_set.values().sorted()
int[] region_totals = [for string name = region_names[] then grouped_total(name, regions, amounts)]
bool[] used = array(len(region_names), fill = false)
for nat rank < len(region_names)
    int best = -1
    for nat i < len(region_names)
        if not used[i]
            if best == -1 or region_totals[i] > region_totals[best]
                best = int(i)
    used[best] = true
    print("{region_names[best]}: {region_totals[best]}{NL}")
string[] product_names = product_set.values().sorted()
for string name = product_names[]
    int quantity = grouped_total(name, products, quantities)
