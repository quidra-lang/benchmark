import collections
string[] csv = [
    "date,region,product,quantity,unit_price",
    "2026-04-01,north,notebook,10,200",
    "2026-04-02,north,marker,8,150",
    "2026-04-02,east,notebook,9,200",
    "2026-04-03,north,eraser,20,80",
    "2026-04-04,south,notebook,6,200",
map<string, int> regional
map<string, int> product_quantities
for string row = csv[1:][]
    string[] fields = row.split(",")
    int quantity = int.parse(fields[3])
    int price = int.parse(fields[4])
    int amount = quantity * price
    .regional.increment(fields[1], by = amount)
    .product_quantities.increment(fields[2], by = quantity)
auto regions = collections.ranked_entries(regional)
for (string region, int revenue) = regions[]
auto products = collections.sorted_entries(product_quantities)
for (string product, int quantity) = products[]
