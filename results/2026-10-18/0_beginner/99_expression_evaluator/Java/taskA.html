1import java.util.HashMap;
2import java.util.List;
3import java.util.Map;
4import java.util.regex.MatchResult;
5import java.util.regex.Pattern;
6
7public class Main {
8 static class Parser {
9 final List<String> tokens;
10 final Map<String, Integer> variables;
11 int position;
12
13 Parser(List<String> tokens, int position, Map<String, Integer> variables) {
14 this.tokens = tokens;
15 this.position = position;
16 this.variables = variables;
17 }
18
19 boolean take(String token) {
20 if (position >= tokens.size() || !tokens.get(position).equals(token)) {
21 return false;
22 }
23 position++;
24 return true;
25 }
26
27 int expression() {
28 int result = term();
29 while (position < tokens.size()
30 && (tokens.get(position).equals("+") || tokens.get(position).equals("-"))) {
31 String operation = tokens.get(position++);
32 int right = term();
33 result = operation.equals("+") ? result + right : result - right;
34 }
35 return result;
36 }
37
38 int term() {
39 int result = unary();
40 while (position < tokens.size()
41 && (tokens.get(position).equals("*") || tokens.get(position).equals("/"))) {
42 String operation = tokens.get(position++);
43 int right = unary();
44 if (operation.equals("/") && right == 0) {
45 throw new IllegalStateException("division by zero");
46 }
47 result = operation.equals("*") ? result * right : result / right;
48 }
49 return result;
50 }
51
52 int unary() {
53 if (take("-")) {
54 return -unary();
55 }
56 if (take("(")) {
57 int result = expression();
58 take(")");
59 return result;
60 }
61 String token = tokens.get(position++);
62 if (token.charAt(0) >= '0' && token.charAt(0) <= '9') {
63 return Integer.parseInt(token);
64 }
65 Integer value = variables.get(token);
66 if (value == null) {
67 throw new IllegalStateException("undefined variable " + token);
68 }
69 return value;
70 }
71 }
72
73 public static void main(String[] args) {
74 String[] lines = {"width = 12", "height = width - 5", "area = width * height", "(area + 6) / 4",
75 "shift = -area / 8 + 3", "height = 100 / (height - 7) + bonus", "volume = area * depth",
76 "width = (width+3)*-(2-4)", "width * shift - height"};
77 Map<String, Integer> variables = new HashMap<>();
78 var tokenPattern = Pattern.compile("[0-9]+|[a-z]+|[^ ]");
79 for (String line : lines) {
80 List<String> tokens = tokenPattern.matcher(line).results().map(MatchResult::group).toList();
81 boolean assignment = tokens.size() > 1 && tokens.get(1).equals("=");
82 String name = assignment ? tokens.get(0) : null;
83 try {
84 int result = new Parser(tokens, assignment ? 2 : 0, variables).expression();
85 if (assignment) {
86 variables.put(name, result);
87 System.out.println(name + " = " + result);
88 } else {
89 System.out.println(result);
90 }
91 } catch (IllegalStateException error) {
92 System.out.println("error: " + error.getMessage());
93 }
94 }
95 }
96}
