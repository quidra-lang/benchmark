1evaluate_tokens <- function(tokens, variables, start) {
2 position <- start
3 factor <- function() {
4 token <- tokens[position]
5 position <<- position + 1
6 if (token == "-") {
7 return(-factor())
8 }
9 if (token == "(") {
10 value <- expression()
11 position <<- position + 1
12 return(value)
13 }
14 if (grepl("^[0-9]", token)) {
15 return(as.numeric(token))
16 }
17 if (!token %in% names(variables)) stop(paste("undefined variable", token), call. = FALSE)
18 unname(variables[token])
19 }
20
21 term <- function() {
22 value <- factor()
23 while (position <= length(tokens) && tokens[position] %in% c("*", "/")) {
24 operation <- tokens[position]
25 position <<- position + 1
26 right <- factor()
27 if (operation == "/" && right == 0) stop("division by zero", call. = FALSE)
28 value <- if (operation == "*") value * right else trunc(value / right)
29 }
30 value
31 }
32
33 expression <- function() {
34 value <- term()
35 while (position <= length(tokens) && tokens[position] %in% c("+", "-")) {
36 operation <- tokens[position]
37 position <<- position + 1
38 right <- term()
39 value <- if (operation == "+") value + right else value - right
40 }
41 value
42 }
43 expression()
44}
45
46lines <- c(
47 "width = 12", "height = width - 5", "area = width * height", "(area + 6) / 4",
48 "shift = -area / 8 + 3", "height = 100 / (height - 7) + bonus", "volume = area * depth",
49 "width = (width+3)*-(2-4)", "width * shift - height"
50)
51variables <- numeric(0)
52for (line in lines) {
53 tokens <- regmatches(line, gregexpr("[0-9]+|[a-z]+|[()+*/=-]", line, perl = TRUE))[[1]]
54 assignment <- length(tokens) >= 2 && tokens[2] == "="
55 result <- tryCatch(
56 list(ok = TRUE, value = evaluate_tokens(tokens, variables, if (assignment) 3 else 1)),
57 error = function(e) list(ok = FALSE, error = conditionMessage(e))
58 )
59 if (!result$ok) {
60 cat("error: ", result$error, "\n", sep = "")
61 } else if (assignment) {
62 variables[tokens[1]] <- result$value
63 cat(sprintf("%s = %d\n", tokens[1], result$value))
64 } else {
65 cat(sprintf("%d\n", result$value))
66 }
67}
