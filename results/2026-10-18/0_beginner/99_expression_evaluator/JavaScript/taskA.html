1function evaluate(tokens, variables) {
2 let index = 0;
3 const peek = () => tokens[index] ?? "";
4 function factor() {
5 const token = tokens[index++];
6 if (token === "-") return -factor();
7 if (token === "(") {
8 const value = expression();
9 index++;
10 return value;
11 }
12 if (/^\d+$/.test(token)) return Number(token);
13 if (!variables.has(token)) throw new Error(`undefined variable ${token}`);
14 return variables.get(token);
15 }
16
17 function term() {
18 let value = factor();
19 while (peek() === "*" || peek() === "/") {
20 const operation = tokens[index];
21 index++;
22 const right = factor();
23 if (operation === "*") {
24 value *= right;
25 } else {
26 if (right === 0) throw new Error("division by zero");
27 value = Math.trunc(value / right);
28 }
29 }
30 return value;
31 }
32
33 function expression() {
34 let value = term();
35 while (peek() === "+" || peek() === "-") {
36 const operation = tokens[index];
37 index++;
38 const right = term();
39 value = operation === "+" ? value + right : value - right;
40 }
41 return value;
42 }
43 return expression();
44}
