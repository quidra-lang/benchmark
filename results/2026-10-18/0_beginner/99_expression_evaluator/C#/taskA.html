1using System;
2using System.Collections.Generic;
3using System.Globalization;
4using System.Linq;
5using System.Text.RegularExpressions;
6
7CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
8
9string[] lines = { "width = 12", "height = width - 5", "area = width * height", "(area + 6) / 4",
10 "shift = -area / 8 + 3", "height = 100 / (height - 7) + bonus", "volume = area * depth",
11 "width = (width+3)*-(2-4)", "width * shift - height" };
12var variables = new Dictionary<string, int>();
13foreach (string line in lines)
14{
15 var tokens = Regex.Matches(line, "[0-9]+|[a-z]+|[^ ]")
16 .Select(match => match.Value).ToList();
17 bool assignment = tokens.Count > 1 && tokens[1] == "=";
18 string? name = assignment ? tokens[0] : null;
19 try
20 {
21 int result = new Parser(tokens, assignment ? 2 : 0, variables).Expression();
22 if (assignment)
23 {
24 variables[name!] = result;
25 Console.WriteLine($"{name} = {result}");
26 }
27 else
28 {
29 Console.WriteLine(result);
30 }
31 }
32 catch (InvalidOperationException error)
33 {
34 Console.WriteLine($"error: {error.Message}");
35 }
36}
37
38class Parser
39{
40 readonly List<string> tokens;
41 readonly Dictionary<string, int> variables;
42 int position;
43
44 public Parser(List<string> tokens, int position, Dictionary<string, int> variables)
45 {
46 this.tokens = tokens;
47 this.position = position;
48 this.variables = variables;
49 }
50
51 bool Take(string token)
52 {
53 if (position >= tokens.Count || tokens[position] != token) return false;
54 position++;
55 return true;
56 }
57
58 public int Expression()
59 {
60 int result = Term();
61 while (position < tokens.Count && (tokens[position] == "+" || tokens[position] == "-"))
62 {
63 string operation = tokens[position++];
64 int right = Term();
65 result = operation == "+" ? result + right : result - right;
66 }
67 return result;
68 }
69
70 int Term()
71 {
72 int result = Unary();
73 while (position < tokens.Count && (tokens[position] == "*" || tokens[position] == "/"))
74 {
75 string operation = tokens[position++];
76 int right = Unary();
77 if (operation == "/" && right == 0) throw new InvalidOperationException("division by zero");
78 result = operation == "*" ? result * right : result / right;
79 }
80 return result;
81 }
82
83 int Unary()
84 {
85 if (Take("-")) return -Unary();
86 if (Take("("))
87 {
88 int result = Expression();
89 Take(")");
90 return result;
91 }
92 string token = tokens[position++];
93 if (char.IsAsciiDigit(token[0])) return int.Parse(token, CultureInfo.InvariantCulture);
94 if (!variables.TryGetValue(token, out int value))
95 throw new InvalidOperationException("undefined variable " + token);
96 return value;
97 }
98}
