1use std::collections::BTreeMap;
2
3struct Parser<'a> {
4 tokens: Vec<String>,
5 position: usize,
6 variables: &'a BTreeMap<String, i64>,
7}
8
9impl Parser<'_> {
10 fn unary(&mut self) -> Result<i64, String> {
11 let token = self.tokens[self.position].clone();
12 self.position += 1;
13 if token == "-" {
14 return Ok(-self.unary()?);
15 }
16 if token == "(" {
17 let value = self.sum()?;
18 self.position += 1;
19 return Ok(value);
20 }
21 if token.as_bytes()[0].is_ascii_digit() {
22 return Ok(token.parse().unwrap());
23 }
24 self.variables
25 .get(&token)
26 .copied()
27 .ok_or_else(|| format!("undefined variable {}", token))
28 }
29
30 fn product(&mut self) -> Result<i64, String> {
31 let mut value = self.unary()?;
32 while self.position < self.tokens.len()
33 && (self.tokens[self.position] == "*" || self.tokens[self.position] == "/")
34 {
35 let operation = self.tokens[self.position].clone();
36 self.position += 1;
37 let right = self.unary()?;
38 if operation == "/" && right == 0 {
39 return Err("division by zero".to_string());
40 }
41 value = if operation == "*" {
42 value * right
43 } else {
44 value / right
45 };
46 }
47 Ok(value)
48 }
49
50 fn sum(&mut self) -> Result<i64, String> {
51 let mut value = self.product()?;
52 while self.position < self.tokens.len()
53 && (self.tokens[self.position] == "+" || self.tokens[self.position] == "-")
54 {
55 let operation = self.tokens[self.position].clone();
56 self.position += 1;
57 let right = self.product()?;
58 value = if operation == "+" {
59 value + right
60 } else {
61 value - right
62 };
63 }
64 Ok(value)
65 }
66}
67
68fn tokenize(line: &str) -> Vec<String> {
69 let bytes = line.as_bytes();
70 let (mut tokens, mut i) = (Vec::new(), 0);
71 while i < bytes.len() {
72 if bytes[i] == b' ' {
73 i += 1;
74 continue;
75 }
76 let start = i;
77 if bytes[i].is_ascii_digit() {
78 while i < bytes.len() && bytes[i].is_ascii_digit() {
79 i += 1;
80 }
81 } else if bytes[i].is_ascii_lowercase() {
82 while i < bytes.len() && bytes[i].is_ascii_lowercase() {
83 i += 1;
84 }
85 } else {
86 i += 1;
87 }
88 tokens.push(line[start..i].to_string());
89 }
90 tokens
91}
92
93fn main() {
94 let mut variables = BTreeMap::new();
95 for line in [
96 "width = 12",
97 "height = width - 5",
98 "area = width * height",
99 "(area + 6) / 4",
100 "shift = -area / 8 + 3",
101 "height = 100 / (height - 7) + bonus",
102 "volume = area * depth",
103 "width = (width+3)*-(2-4)",
104 "width * shift - height",
105 ] {
106 let tokens = tokenize(line);
107 let assignment = tokens.len() > 1 && tokens[1] == "=";
108 let mut parser = Parser {
109 tokens,
110 position: if assignment { 2 } else { 0 },
111 variables: &variables,
112 };
113 match parser.sum() {
114 Err(error) => println!("error: {}", error),
115 Ok(value) => {
116 if assignment {
117 let name = parser.tokens[0].clone();
118 variables.insert(name.clone(), value);
119 println!("{} = {}", name, value);
120 } else {
121 println!("{}", value);
122 }
123 }
124 }
125 }
126}
