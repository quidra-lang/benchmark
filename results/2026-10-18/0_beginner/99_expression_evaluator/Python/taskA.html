4class Parser:
5 def __init__(self, tokens, variables):
6 self.tokens = tokens
7 self.variables = variables
8 self.index = 0
9
10 def peek(self):
11 return self.tokens[self.index] if self.index < len(self.tokens) else ""
12
13 def factor(self):
14 token = self.peek()
15 self.index += 1
16 if token == "-":
17 return -self.factor()
18 if token == "(":
19 value = self.expression()
20 self.index += 1
21 return value
22 if token.isdigit():
23 return int(token)
24 if token not in self.variables:
25 raise ValueError(f"undefined variable {token}")
26 return self.variables[token]
27
28 def term(self):
29 value = self.factor()
30 while self.peek() in ("*", "/"):
31 operation = self.peek()
32 self.index += 1
33 right = self.factor()
34 if operation == "*":
35 value *= right
36 else:
37 if right == 0:
38 raise ValueError("division by zero")
39 value = int(value / right)
40 return value
41
42 def expression(self):
43 value = self.term()
44 while self.peek() in ("+", "-"):
45 operation = self.peek()
46 self.index += 1
47 right = self.term()
48 value = value + right if operation == "+" else value - right
49 return value
