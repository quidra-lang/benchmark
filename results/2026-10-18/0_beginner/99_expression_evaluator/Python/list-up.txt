    def term(self):
        value = self.factor()
        while self.peek() in ("*", "/"):
            operation = self.peek()
            self.index += 1
            right = self.factor()
            if operation == "*":
                value *= right
            else:
                if right == 0:
                    raise ValueError("division by zero")
                value = int(value / right)
        return value
