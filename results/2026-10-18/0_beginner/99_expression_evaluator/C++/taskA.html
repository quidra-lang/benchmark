7std::vector<std::string> tokenize(const std::string &line) {
8 std::vector<std::string> tokens;
9 for (std::size_t i = 0; i < line.size();) {
10 if (line[i] == ' ') {
11 ++i;
12 continue;
13 }
14 std::size_t start = i;
15 if (line[i] >= '0' && line[i] <= '9')
16 while (i < line.size() && line[i] >= '0' && line[i] <= '9')
17 ++i;
18 else if (line[i] >= 'a' && line[i] <= 'z')
19 while (i < line.size() && line[i] >= 'a' && line[i] <= 'z')
20 ++i;
21 else
22 ++i;
23 tokens.push_back(line.substr(start, i - start));
24 }
25 return tokens;
26}

28struct Parser {
29 std::vector<std::string> tokens;
30 std::size_t position = 0;
31 const std::map<std::string, long long> &variables;
32 std::string error;
33
34 long long unary() {
35 std::string token = tokens[position++];
36 if (token == "-")
37 return -unary();
38 if (token == "(") {
39 long long value = sum();
40 if (error.empty())
41 ++position;
42 return value;
43 }
44 if (token[0] >= '0' && token[0] <= '9')
45 return std::stoll(token);
46 auto found = variables.find(token);
47 if (found != variables.end())
48 return found->second;
49 error = "undefined variable " + token;
50 return 0;
51 }
52
53 long long product() {
54 long long value = unary();
55 while (error.empty() && position < tokens.size() && (tokens[position] == "*" || tokens[position] == "/")) {
56 std::string operation = tokens[position++];
57 long long right = unary();
58 if (!error.empty())
59 return 0;
60 if (operation == "/" && right == 0) {
61 error = "division by zero";
62 return 0;
63 }
64 value = operation == "*" ? value * right : value / right;
65 }
66 return value;
67 }
68
69 long long sum() {
70 long long value = product();
71 while (error.empty() && position < tokens.size() && (tokens[position] == "+" || tokens[position] == "-")) {
72 std::string operation = tokens[position++];
73 long long right = product();
74 if (!error.empty())
75 return 0;
76 value = operation == "+" ? value + right : value - right;
77 }
78 return value;
79 }
80};

82int main() {
83 std::map<std::string, long long> variables;
84 const std::array<std::string, 9> lines = {"width = 12", "height = width - 5", "area = width * height",
85 "(area + 6) / 4", "shift = -area / 8 + 3", "height = 100 / (height - 7) + bonus", "volume = area * depth",
86 "width = (width+3)*-(2-4)", "width * shift - height"};
87 for (const std::string &line : lines) {
88 Parser parser{tokenize(line), 0, variables, {}};
89 bool assignment = parser.tokens.size() > 1 && parser.tokens[1] == "=";
90 if (assignment)
91 parser.position = 2;
92 long long value = parser.sum();
93 if (!parser.error.empty()) {
94 std::cout << "error: " << parser.error << '\n';
95 } else if (assignment) {
96 variables[parser.tokens[0]] = value;
97 std::cout << parser.tokens[0] << " = " << value << '\n';
98 } else {
99 std::cout << value << '\n';
100 }
101 }
