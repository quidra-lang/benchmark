3class Parser
4{
5 public function __construct(private array $tokens, private array $variables, private int $position) {}
6
7 private function factor(): int
8 {
9 $token = $this->tokens[$this->position++];
10 if ($token === '-') {
11 return -$this->factor();
12 }
13 if ($token === '(') {
14 $value = $this->expression();
15 $this->position++;
16 return $value;
17 }
18 if (ctype_digit($token)) {
19 return (int) $token;
20 }
21 if (!array_key_exists($token, $this->variables)) {
22 throw new RuntimeException('undefined variable ' . $token);
23 }
24 return $this->variables[$token];
25 }
26
27 private function term(): int
28 {
29 $value = $this->factor();
30 while (isset($this->tokens[$this->position]) && in_array($this->tokens[$this->position], ['*', '/'], true)) {
31 $operation = $this->tokens[$this->position++];
32 $right = $this->factor();
33 if ($operation === '/' && $right === 0) {
34 throw new RuntimeException('division by zero');
35 }
36 $value = $operation === '*' ? $value * $right : intdiv($value, $right);
37 }
38 return $value;
39 }
40
41 public function expression(): int
42 {
43 $value = $this->term();
44 while (isset($this->tokens[$this->position]) && in_array($this->tokens[$this->position], ['+', '-'], true)) {
45 $operation = $this->tokens[$this->position++];
46 $right = $this->term();
47 $value = $operation === '+' ? $value + $right : $value - $right;
48 }
49 return $value;
50 }
51}
