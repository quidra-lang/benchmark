8type Parser struct {
9 tokens []string
10 position int
11 variables map[string]int64
12 err string
13}
14
15func tokenize(line string) []string {
16 tokens := []string{}
17 for i := 0; i < len(line); {
18 if line[i] == ' ' {
19 i++
20 continue
21 }
22 start := i
23 if line[i] >= '0' && line[i] <= '9' {
24 for i < len(line) && line[i] >= '0' && line[i] <= '9' {
25 i++
26 }
27 } else if line[i] >= 'a' && line[i] <= 'z' {
28 for i < len(line) && line[i] >= 'a' && line[i] <= 'z' {
29 i++
30 }
31 } else {
32 i++
33 }
34 tokens = append(tokens, line[start:i])
35 }
36 return tokens
37}

39func (p *Parser) unary() int64 {
40 token := p.tokens[p.position]
41 p.position++
42 if token == "-" {
43 return -p.unary()
44 }
45 if token == "(" {
46 value := p.sum()
47 if p.err == "" {
48 p.position++
49 }
50 return value
51 }
52 if token[0] >= '0' && token[0] <= '9' {
53 value, _ := strconv.ParseInt(token, 10, 64)
54 return value
55 }
56 value, ok := p.variables[token]
57 if !ok {
58 p.err = "undefined variable " + token
59 }
60 return value
61}
62
63func (p *Parser) product() int64 {
64 value := p.unary()
65 for p.err == "" && p.position < len(p.tokens) && (p.tokens[p.position] == "*" || p.tokens[p.position] == "/") {
66 operation := p.tokens[p.position]
67 p.position++
68 right := p.unary()
69 if p.err != "" {
70 return 0
71 }
72 if operation == "/" && right == 0 {
73 p.err = "division by zero"
74 return 0
75 }
76 if operation == "*" {
77 value *= right
78 } else {
79 value /= right
80 }
81 }
82 return value
83}
84
85func (p *Parser) sum() int64 {
86 value := p.product()
87 for p.err == "" && p.position < len(p.tokens) && (p.tokens[p.position] == "+" || p.tokens[p.position] == "-") {
88 operation := p.tokens[p.position]
89 p.position++
90 right := p.product()
91 if p.err != "" {
92 return 0
93 }
94 if operation == "+" {
95 value += right
96 } else {
97 value -= right
98 }
99 }
100 return value
101}

103func main() {
104 variables := map[string]int64{}
105 lines := []string{"width = 12", "height = width - 5", "area = width * height", "(area + 6) / 4",
106 "shift = -area / 8 + 3", "height = 100 / (height - 7) + bonus", "volume = area * depth",
107 "width = (width+3)*-(2-4)", "width * shift - height"}
108 for _, line := range lines {
109 tokens := tokenize(line)
110 assignment := len(tokens) > 1 && tokens[1] == "="
111 parser := Parser{tokens: tokens, variables: variables}
112 if assignment {
113 parser.position = 2
114 }
115 value := parser.sum()
116 if parser.err != "" {
117 fmt.Printf("error: %s\n", parser.err)
118 } else if assignment {
119 variables[tokens[0]] = value
120 fmt.Printf("%s = %d\n", tokens[0], value)
121 } else {
122 fmt.Println(value)
123 }
124 }
