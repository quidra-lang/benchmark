1local function tokenize(line)
2 local tokens, i = {}, 1
3 while i <= #line do
4 local c = line:sub(i, i)
5 if c == " " then
6 i = i + 1
7 elseif c:match("[0-9a-z]") then
8 local pattern = c:match("%d") and "%d" or "[a-z]"
9 local first = i
10 repeat i = i + 1 until i > #line or not line:sub(i, i):match(pattern)
11 tokens[#tokens + 1] = line:sub(first, i - 1)
12 else
13 tokens[#tokens + 1] = c
14 i = i + 1
15 end
16 end
17 return tokens
18end

20local function evaluate(tokens, variables, position)
21 local expression, term, factor
22 factor = function()
23 local token = tokens[position]
24 position = position + 1
25 if token == "-" then return -factor() end
26 if token == "(" then
27 local value = expression()
28 position = position + 1
29 return value
30 end
31 local number = tonumber(token)
32 if number then return number end
33 if variables[token] == nil then error("undefined variable " .. token, 0) end
34 return variables[token]
35 end
36 term = function()
37 local value = factor()
38 while tokens[position] == "*" or tokens[position] == "/" do
39 local operation = tokens[position]
40 position = position + 1
41 local right = factor()
42 if operation == "/" and right == 0 then error("division by zero", 0) end
43 if operation == "*" then
44 value = value * right
45 else
46 local quotient = math.abs(value) // math.abs(right)
47 value = (value < 0) ~= (right < 0) and -quotient or quotient
48 end
49 end
50 return value
51 end
52 expression = function()
53 local value = term()
54 while tokens[position] == "+" or tokens[position] == "-" do
55 local operation = tokens[position]
56 position = position + 1
57 local right = term()
58 value = operation == "+" and value + right or value - right
59 end
60 return value
61 end
62 return expression()
63end

65local lines = {
66 "width = 12", "height = width - 5", "area = width * height", "(area + 6) / 4",
67 "shift = -area / 8 + 3", "height = 100 / (height - 7) + bonus", "volume = area * depth",
68 "width = (width+3)*-(2-4)", "width * shift - height",
69}
70local variables = {}
71for _, line in ipairs(lines) do
72 local tokens = tokenize(line)
73 local assignment = tokens[2] == "="
74 local success, value = pcall(evaluate, tokens, variables, assignment and 3 or 1)
75 if not success then
76 io.write("error: ", value, "\n")
77 elseif assignment then
78 variables[tokens[1]] = value
79 io.write(string.format("%s = %d\n", tokens[1], value))
80 else
81 io.write(string.format("%d\n", value))
82 end
83end
