int | error product(const string[] tokens, nat &position, const map<string, int> variables)
    value := try unary(tokens, &position, variables)
    while position < len(tokens)
        operation := tokens[position]
        if operation != "*" and operation != "/"
            break
        position += 1
        right := try unary(tokens, &position, variables)
        if operation == "*"
            value *= right
        else
            if right == 0
                return error("division by zero")
            value /= right
    return value
