5typedef struct {
6 char names[32][32];
7 long long values[32];
8 int count;
9} Variables;

11typedef struct {
12 char tokens[128][32];
13 int count, position;
14 char error[128];
15 Variables *variables;
16} Parser;

18long long sum(Parser *p);

20void tokenize(Parser *p, const char *line) {
21 p->count = 0;
22 p->position = 0;
23 p->error[0] = 0;
24 for (int i = 0; line[i];) {
25 if (line[i] == ' ') {
26 i++;
27 continue;
28 }
29 int length = 0;
30 if (line[i] >= '0' && line[i] <= '9') {
31 while (line[i] >= '0' && line[i] <= '9')
32 p->tokens[p->count][length++] = line[i++];
33 } else if (line[i] >= 'a' && line[i] <= 'z') {
34 while (line[i] >= 'a' && line[i] <= 'z')
35 p->tokens[p->count][length++] = line[i++];
36 } else {
37 p->tokens[p->count][length++] = line[i++];
38 }
39 p->tokens[p->count++][length] = 0;
40 }
41}

43long long unary(Parser *p) {
44 const char *token = p->tokens[p->position++];
45 if (strcmp(token, "-") == 0)
46 return -unary(p);
47 if (strcmp(token, "(") == 0) {
48 long long value = sum(p);
49 if (!p->error[0])
50 p->position++;
51 return value;
52 }
53 if (token[0] >= '0' && token[0] <= '9')
54 return strtoll(token, NULL, 10);
55 for (int i = 0; i < p->variables->count; i++)
56 if (strcmp(p->variables->names[i], token) == 0)
57 return p->variables->values[i];
58 snprintf(p->error, sizeof p->error, "undefined variable %s", token);
59 return 0;
60}
61
62long long product(Parser *p) {
63 long long value = unary(p);
64 while (!p->error[0] && p->position < p->count &&
65 (strcmp(p->tokens[p->position], "*") == 0 || strcmp(p->tokens[p->position], "/") == 0)) {
66 char operation = p->tokens[p->position++][0];
67 long long right = unary(p);
68 if (p->error[0])
69 return 0;
70 if (operation == '/' && right == 0) {
71 strcpy(p->error, "division by zero");
72 return 0;
73 }
74 value = operation == '*' ? value * right : value / right;
75 }
76 return value;
77}
78
79long long sum(Parser *p) {
80 long long value = product(p);
81 while (!p->error[0] && p->position < p->count &&
82 (strcmp(p->tokens[p->position], "+") == 0 || strcmp(p->tokens[p->position], "-") == 0)) {
83 char operation = p->tokens[p->position++][0];
84 long long right = product(p);
85 if (p->error[0])
86 return 0;
87 value = operation == '+' ? value + right : value - right;
88 }
89 return value;
90}

92int main(void) {

94 "shift = -area / 8 + 3", "height = 100 / (height - 7) + bonus", "volume = area * depth",
95 "width = (width+3)*-(2-4)", "width * shift - height"};
96 Variables variables = {0};
97 for (int line = 0; line < 9; line++) {
98 Parser p;
99 p.variables = &variables;
100 tokenize(&p, lines[line]);
101 int assignment = p.count > 1 && strcmp(p.tokens[1], "=") == 0;
102 if (assignment)
103 p.position = 2;
104 long long value = sum(&p);
105 if (p.error[0]) {
106 printf("error: %s\n", p.error);
107 } else if (assignment) {
108 int i = 0;
109 while (i < variables.count && strcmp(variables.names[i], p.tokens[0]))
110 i++;
111 if (i == variables.count) {
112 strcpy(variables.names[i], p.tokens[0]);
113 variables.count++;
114 }
115 variables.values[i] = value;
116 printf("%s = %lld\n", p.tokens[0], value);
117 } else {
118 printf("%lld\n", value);
119 }
120 }
