1class Parser
2 def initialize(tokens, variables)
3 @tokens = tokens
4 @variables = variables
5 @index = 0
6 end
7
8 def peek
9 @tokens[@index]
10 end

12 def factor
13 token = peek
14 @index += 1
15 return -factor if token == '-'
16 if token == '('
17 value = expression
18 @index += 1
19 return value
20 end
21 return token.to_i if token.match?(/\A\d+\z/)
22 raise "undefined variable #{token}" unless @variables.key?(token)
23 @variables[token]
24 end
25
26 def term
27 value = factor
28 while %w[* /].include?(peek)
29 operation = peek
30 @index += 1
31 right = factor
32 if operation == '*'
33 value *= right
34 else
35 raise 'division by zero' if right.zero?
36 value = value.quo(right).truncate
37 end
38 end
39 value
40 end
41
42 def expression
43 value = term
44 while %w[+ -].include?(peek)
45 operation = peek
46 @index += 1
47 right = term
48 value = operation == '+' ? value + right : value - right
49 end
50 value
51 end
52end
