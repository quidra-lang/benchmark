            squared = dx*dx+dy*dy
                nearest = (i-1, j-1, squared)
                farthest = (i-1, j-1, squared)
    for (label, pair) in [("closest", nearest), ("farthest", farthest)]
        distance = sqrt(Float64(pair[3]))
        @printf("%s: P%d P%d %.4f\n", label, pair[1], pair[2], distance)
    end
