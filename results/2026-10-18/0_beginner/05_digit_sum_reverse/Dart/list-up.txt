    var remaining = original;
    var digits = 0;
    var sum = 0;
    var reversed = 0;
    while (remaining > 0) {
      final digit = remaining % 10;
      digits++;
      sum += digit;
      reversed = reversed * 10 + digit;
      remaining ~/= 10;
    }
    print('$original: digits=$digits sum=$sum reversed=$reversed');
