        int remaining = value, digits = 0, sum = 0, reversed = 0;
        while (remaining > 0) {
            int digit = remaining % 10;
            ++digits;
            sum += digit;
            reversed = reversed * 10 + digit;
            remaining /= 10;
        }
        std::cout << value << ": digits=" << digits << " sum=" << sum
                  << " reversed=" << reversed << '\n';
