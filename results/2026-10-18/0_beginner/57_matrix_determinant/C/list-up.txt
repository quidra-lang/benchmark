int det(int a[5][5], int n) {
    if (n == 1)
        return a[0][0];
    int result = 0;
    for (int j = 0; j < n; j++) {
        int minor[5][5];
        for (int r = 1; r < n; r++) {
            int col = 0;
            for (int c = 0; c < n; c++)
                if (c != j)
                    minor[r - 1][col++] = a[r][c];
        }
        int term = a[0][j] * det(minor, n - 1);
        result += j % 2 ? -term : term;
    }
}
