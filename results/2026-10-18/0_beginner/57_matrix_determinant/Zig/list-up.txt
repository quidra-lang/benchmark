fn det(comptime n: usize, matrix: [n][n]i64) i64 {
    if (n == 1) {
        return matrix[0][0];
    } else {
        var result: i64 = 0;
        for (0..n) |j| {
            var minor: [n - 1][n - 1]i64 = undefined;
            for (1..n) |r| {
                var col: usize = 0;
                for (0..n) |c| {
                    if (c != j) {
                        minor[r - 1][col] = matrix[r][c];
                        col += 1;
                    }
                }
            }
            const term = matrix[0][j] * det(n - 1, minor);
            result += if (j % 2 == 0) term else -term;
        }
    }
}
