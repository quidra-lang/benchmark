3fn det(comptime n: usize, matrix: [n][n]i64) i64 {
4 if (n == 1) {
5 return matrix[0][0];
6 } else {
7 var result: i64 = 0;
8 for (0..n) |j| {
9 var minor: [n - 1][n - 1]i64 = undefined;
10 for (1..n) |r| {
11 var col: usize = 0;
12 for (0..n) |c| {
13 if (c != j) {
14 minor[r - 1][col] = matrix[r][c];
15 col += 1;
16 }
17 }
18 }
19 const term = matrix[0][j] * det(n - 1, minor);
20 result += if (j % 2 == 0) term else -term;
21 }
22 return result;
23 }
24}
