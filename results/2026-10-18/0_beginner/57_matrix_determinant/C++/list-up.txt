int det(const std::vector<std::vector<int>> &a) {
    int n = static_cast<int>(a.size());
    if (n == 1)
        return a[0][0];
    int result = 0;
    for (int j = 0; j < n; j++) {
        std::vector<std::vector<int>> minor(n - 1);
        for (int r = 1; r < n; r++) {
            for (int c = 0; c < n; c++)
                if (c != j)
                    minor[r - 1].push_back(a[r][c]);
        }
        int term = a[0][j] * det(minor);
        result += j % 2 ? -term : term;
    }
}
