7static int Determinant(int[][] matrix)
8{
9 int n = matrix.Length;
10 if (n == 1) return matrix[0][0];
11 int result = 0;
12 for (int column = 0; column < n; column++)
13 {
14 int[][] minor = matrix.Skip(1).Select(row => row.Where((_, c) => c != column).ToArray()).ToArray();
15 int term = matrix[0][column] * Determinant(minor);
16 result += column % 2 == 0 ? term : -term;
17 }
18 return result;
19}
