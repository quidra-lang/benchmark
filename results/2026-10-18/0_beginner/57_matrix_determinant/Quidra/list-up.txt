int determinant(int[][] matrix)
    nat n = len(matrix)
    if n == 1
        return matrix[0][0]
    int total = 0
    for nat j < n
        int[][] minor = [for int[] row = matrix[1:][] then [for nat c = row.indices()[] if c != j then row[c]]]
        int term = matrix[0][j] * determinant(minor)
        total += if j % 2 == 0 then term else -term
