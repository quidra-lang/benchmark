int determinant(int[][] matrix)
    int n = int(len(matrix))
    if n == 1
        return matrix[0][0]
    int total = 0
    for 0 <= int j < n
        int[][] minor = [for int[] row = matrix[1:][] then [for nat c < len(row) if int(c) != j then row[c]]]
        int term = matrix[0][j] * determinant(minor)
        total += if j % 2 == 0 then term else -term
