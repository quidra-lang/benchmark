function laplace(array $matrix): int
{
    $n = count($matrix);
    if ($n === 1) {
        return $matrix[0][0];
    }
    $total = 0;
    for ($j = 0; $j < $n; $j++) {
        $minor = [];
        for ($row = 1; $row < $n; $row++) {
            $values = $matrix[$row];
            array_splice($values, $j, 1);
            $minor[] = $values;
        }
        $term = $matrix[0][$j] * laplace($minor);
        $total += $j % 2 === 0 ? $term : -$term;
    }
}
