    static int determinant(int[][] matrix) {
        int n = matrix.length;
        if (n == 1) {
            return matrix[0][0];
        }
        int result = 0;
        for (int column = 0; column < n; column++) {
            int[][] minor = new int[n - 1][n - 1];
            for (int r = 1; r < n; r++) {
                int index = 0;
                for (int c = 0; c < n; c++) {
                    if (c != column) {
                        minor[r - 1][index++] = matrix[r][c];
                    }
                }
            }
            int term = matrix[0][column] * determinant(minor);
            result += column % 2 == 0 ? term : -term;
        }
    }
