func det(a [][]int) int {
	if len(a) == 1 {
		return a[0][0]
	}
	result := 0
	for j := range a {
		minor := [][]int{}
		for _, r := range a[1:] {
			row := []int{}
			for c, v := range r {
				if c != j {
					row = append(row, v)
				}
			}
			minor = append(minor, row)
		}
		term := a[0][j] * det(minor)
		if j%2 == 0 {
			result += term
		} else {
			result -= term
		}
	}
}
