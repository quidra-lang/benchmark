1import java.util.ArrayList;
2import java.util.Arrays;
3import java.util.LinkedHashMap;
4import java.util.List;
5import java.util.Map;
6
7public class Main {
8 public static void main(String[] args) {
9 Map<String, List<String>> groups = new LinkedHashMap<>();
10 String[] words = {
11 "listen", "stone", "night", "silent", "notes", "apple", "thing", "enlist", "earth",
12 "onset", "heart", "tones"};
13 for (String word : words) {
14 char[] letters = word.toCharArray();
15 Arrays.sort(letters);
16 groups.computeIfAbsent(new String(letters), key -> new ArrayList<>()).add(word);
17 }
18 for (List<String> group : groups.values()) {
19 System.out.println(String.join(" ", group));
20 }
21 }
22}
