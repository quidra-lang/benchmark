    var remaining = n;
    final factors = <String>[];
    for (var d = 2; d * d <= remaining; d++) {
      var e = 0;
      while (remaining % d == 0) {
        remaining ~/= d;
        e++;
      }
      if (e > 0) factors.add(e == 1 ? '$d' : '$d^$e');
    }
    if (remaining > 1) factors.add('$remaining');
    print('$n = ${factors.join(' * ')}');
