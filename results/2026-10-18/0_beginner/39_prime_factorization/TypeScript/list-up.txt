for (const number of [84, 9973, 15625, 1001, 123456, 48000, 999958, 999999]) {
  let remaining = number, divisor = 2;
  const factors: string[] = [];
  while (divisor * divisor <= remaining) {
    let exponent = 0;
    while (remaining % divisor === 0) {
      remaining /= divisor;
      exponent++;
    }
    if (exponent > 0) factors.push(exponent > 1 ? `${divisor}^${exponent}` : String(divisor));
    divisor++;
  }
  if (remaining > 1) factors.push(String(remaining));
  console.log(`${number} = ${factors.join(" * ")}`);
