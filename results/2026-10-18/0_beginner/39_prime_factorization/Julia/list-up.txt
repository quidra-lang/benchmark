    remaining, divisor = original, 2
    factors = String[]
    while divisor * divisor <= remaining
        factor_power = 0
        while remaining % divisor == 0
            remaining = div(remaining, divisor)
            factor_power += 1
        end
        if factor_power > 0
            push!(
                factors,
        end
        divisor += 1
    end
    if remaining > 1
        push!(factors, string(remaining))
    println(original, " = ", join(factors, " * "))
