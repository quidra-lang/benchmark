1using System;
2using System.Collections.Generic;
3using System.Globalization;
4
5CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
6
7foreach (int n in new[] { 84, 9973, 15625, 1001, 123456, 48000, 999958, 999999 })
8{
9 int remaining = n;
10 var factors = new List<string>();
11 for (int d = 2; d * d <= remaining; d++)
12 {
13 int exponent = 0;
14 while (remaining % d == 0)
15 {
16 remaining /= d;
17 exponent++;
18 }
19 if (exponent > 0) factors.Add(d + (exponent > 1 ? "^" + exponent : ""));
20 }
21 if (remaining > 1) factors.Add(remaining.ToString(CultureInfo.InvariantCulture));
22 Console.WriteLine($"{n} = {string.Join(" * ", factors)}");
23}
