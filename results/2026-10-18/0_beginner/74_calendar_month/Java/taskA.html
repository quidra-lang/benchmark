6public class Main {

11 int[] offsets = {0, 3, 2, 5, 0, 3, 5, 1, 4, 6, 2, 4};

13 if (index > 0) {
14 System.out.println();

22 }

24 System.out.println("Su Mo Tu We Th Fr Sa");
25 List<String> week = new ArrayList<>(Collections.nCopies(weekday, " "));
26 for (int day = 1; day <= lengths[month - 1]; day++) {
27 week.add(String.format(Locale.ROOT, "%2d", day));
