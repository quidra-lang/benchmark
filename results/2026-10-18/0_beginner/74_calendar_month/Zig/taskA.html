1const std = @import("std");
2
3fn leap(y: u32) bool {
4 return y % 4 == 0 and (y % 100 != 0 or y % 400 == 0);
5}
6
7pub fn main(init: std.process.Init) !void {
8 var output_buffer: [8192]u8 = undefined;
9 var writer = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
10 const out = &writer.interface;
11 const names = [_][]const u8{
12 "January", "February", "March", "April", "May", "June",
13 "July", "August", "September", "October", "November", "December",
14 };
15 const lengths = [_]u32{ 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31 };
16 for ([_][2]u32{ .{ 2024, 2 }, .{ 2026, 8 }, .{ 2026, 11 } }, 0..) |p, k| {
17 const year = p[0];
18 const month = p[1];
19 const prior_year = year - 1;
20 var days: u32 = 365 * prior_year + prior_year / 4 - prior_year / 100 + prior_year / 400;
21 for (1..month) |m|
22 days += if (m == 2 and leap(year)) @as(u32, 29) else lengths[m - 1];
23 const offset = (days + 1) % 7;
24 const last = if (month == 2 and leap(year)) @as(u32, 29) else lengths[month - 1];
25 if (k > 0)
26 try out.writeByte('\n');
27 try out.print("{s} {d}\nSu Mo Tu We Th Fr Sa\n", .{ names[month - 1], year });
28 try out.splatByteAll(' ', offset * 3);
29 var day: u32 = 1;
30 while (day <= last) : (day += 1) {
31 try out.print("{d:>2}", .{day});
32 if ((offset + day - 1) % 7 == 6 or day == last) {
33 try out.writeByte('\n');
34 } else {
35 try out.writeByte(' ');
36 }
37 }
38 }
39 try out.flush();
40}
