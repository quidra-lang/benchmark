function days_in_month(year, month)
    leap = year % 400 == 0 || (year % 4 == 0 && year % 100 != 0)
    return month == 2 && leap ? 29 : [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31][month]
end
