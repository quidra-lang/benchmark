let lengths = [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31]
    var year = initialYear, month = initialMonth, day = initialDay
    let original = date(year, month, day)
    for _ in 0..<days {
        let monthLength = lengths[month - 1] +
            (month == 2 && year % 4 == 0 && (year % 100 != 0 || year % 400 == 0) ? 1 : 0)
        day += 1
        if day > monthLength {
            day = 1
            month += 1
            if month > 12 {
                month = 1
                year += 1
            }
        }
    }
    print("\(original) + \(days) days = \(date(year, month, day))")
