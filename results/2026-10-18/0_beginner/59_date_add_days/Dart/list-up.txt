  const lengths = [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31];
    var y = c[0], m = c[1], d = c[2];
    for (var i = 0; i < c[3]; i++) {
      var days = lengths[m - 1];
      if (m == 2 && y % 4 == 0 && (y % 100 != 0 || y % 400 == 0)) days = 29;
      d++;
      if (d > days) {
        d = 1;
        m++;
        if (m > 12) {
          m = 1;
          y++;
        }
      }
    }
    print('${date(c[0], c[1], c[2])} + ${c[3]} days = ${date(y, m, d)}');
