    const char *symbols = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789", *codes[] = {".-", "-...", "-.-.",
        "-..", ".", "..-.", "--.", "....", "..", ".---", "-.-", ".-..", "--", "-.", "---", ".--.", "--.-",
        ".-.", "...", "-", "..-", "...-", ".--", "-..-", "-.--", "--..", "-----", ".----", "..---", "...--",
        "....-", ".....", "-....", "--...", "---..", "----."};
    const char *message = "MEET ME AT PIER 9 AT 1745";
    char encoded[512] = "";
    for (int i = 0; message[i]; i++) {
        if (message[i] == ' ') {
            strcat(encoded, " / ");
            continue;
        }
        if (i && message[i - 1] != ' ')
            strcat(encoded, " ");
        const char *p = strchr(symbols, message[i]);
        strcat(encoded, codes[p - symbols]);
    }
    char decoded[128];
    int d = 0;
    for (char *token = strtok(encoded, " "); token; token = strtok(NULL, " ")) {
        if (strcmp(token, "/") == 0)
            decoded[d++] = ' ';
        else
            for (int i = 0; i < 36; i++)
                if (strcmp(token, codes[i]) == 0) {
                    decoded[d++] = symbols[i];
                    break;
                }
    }
    decoded[d] = 0;
    puts(decoded);
