3import (
4 "fmt"
5 "slices"

9func main() {
10 symbols := "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"
11 codes := []string{".-", "-...", "-.-.", "-..", ".", "..-.", "--.", "....", "..", ".---", "-.-",
12 ".-..", "--", "-.", "---", ".--.", "--.-", ".-.", "...", "-", "..-", "...-", ".--", "-..-", "-.--",
13 "--..", "-----", ".----", "..---", "...--", "....-", ".....", "-....", "--...", "---..", "----."}
14 message := "MEET ME AT PIER 9 AT 1745"
15 words := []string{}
16 for _, w := range strings.Split(message, " ") {
17 parts := []string{}
18 for _, c := range w {
19 parts = append(parts, codes[strings.IndexRune(symbols, c)])
20 }
21 words = append(words, strings.Join(parts, " "))
22 }
23 encoded := strings.Join(words, " / ")
24 fmt.Println(encoded)
25 decoded := []string{}
26 for _, w := range strings.Split(encoded, " / ") {
27 word := ""
28 for _, token := range strings.Split(w, " ") {
29 word += string(symbols[slices.Index(codes, token)])
30 }
31 decoded = append(decoded, word)
32 }
33 fmt.Println(strings.Join(decoded, " "))
34}
