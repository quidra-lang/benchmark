    var writer = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
    const out = &writer.interface;
    const codes = [_][]const u8{
        ".-",    "-...",  "-.-.",  "-..",   ".",     "..-.",  "--.",   "....",  "..",    ".---",
        "-.-",   ".-..",  "--",    "-.",    "---",   ".--.",  "--.-",  ".-.",   "...",   "-",
        "..-",   "...-",  ".--",   "-..-",  "-.--",  "--..",  "-----", ".----", "..---", "...--",
        "....-", ".....", "-....", "--...", "---..", "----.",
    };
    const alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789";
    const message = "MEET ME AT PIER 9 AT 1745";
    var encoded: [512]u8 = undefined;
    var len: usize = 0;
    var need_space = false;
    for (message) |c| {
        if (c == ' ') {
            @memcpy(encoded[len..][0..3], " / ");
            len += 3;
            need_space = false;
            continue;
        }
        if (need_space) {
            encoded[len] = ' ';
            len += 1;
        }
        const index: usize = if (c >= 'A' and c <= 'Z') c - 'A' else 26 + c - '0';
        @memcpy(encoded[len..][0..codes[index].len], codes[index]);
        len += codes[index].len;
        need_space = true;
    }
    var tokens = std.mem.splitScalar(u8, encoded[0..len], ' ');
    while (tokens.next()) |token| {
        if (std.mem.eql(u8, token, "/")) {
            try out.writeByte(' ');
        } else {
            for (codes, 0..) |code, i| {
                if (std.mem.eql(u8, code, token)) {
                    try out.writeByte(alphabet[i]);
                    break;
                }
            }
        }
    }
