3pub fn main(init: std.process.Init) !void {
4 var output_buffer: [8192]u8 = undefined;
5 var writer = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
6 const out = &writer.interface;
7 const codes = [_][]const u8{
8 ".-", "-...", "-.-.", "-..", ".", "..-.", "--.", "....", "..", ".---",
9 "-.-", ".-..", "--", "-.", "---", ".--.", "--.-", ".-.", "...", "-",
10 "..-", "...-", ".--", "-..-", "-.--", "--..", "-----", ".----", "..---", "...--",
11 "....-", ".....", "-....", "--...", "---..", "----.",
12 };
13 const alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789";
14 const message = "MEET ME AT PIER 9 AT 1745";
15 var encoded: [512]u8 = undefined;
16 var len: usize = 0;
17 var need_space = false;
18 for (message) |c| {
19 if (c == ' ') {
20 @memcpy(encoded[len..][0..3], " / ");
21 len += 3;
22 need_space = false;
23 continue;
24 }
25 if (need_space) {
26 encoded[len] = ' ';
27 len += 1;
28 }
29 const index: usize = if (c >= 'A' and c <= 'Z') c - 'A' else 26 + c - '0';
30 @memcpy(encoded[len..][0..codes[index].len], codes[index]);
31 len += codes[index].len;
32 need_space = true;
33 }
34 try out.print("{s}\n", .{encoded[0..len]});
35 var tokens = std.mem.splitScalar(u8, encoded[0..len], ' ');
36 while (tokens.next()) |token| {
37 if (std.mem.eql(u8, token, "/")) {
38 try out.writeByte(' ');
39 } else {
40 for (codes, 0..) |code, i| {
41 if (std.mem.eql(u8, code, token)) {
42 try out.writeByte(alphabet[i]);
43 break;
44 }
45 }
46 }
47 }
48 try out.writeByte('\n');
49 try out.flush();
50}
