string alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"
string[] codes = [
    ".-", "-...", "-.-.", "-..", ".", "..-.", "--.", "....", "..", ".---", "-.-", ".-..", "--",
    "-.", "---", ".--.", "--.-", ".-.", "...", "-", "..-", "...-", ".--", "-..-", "-.--", "--..",
    "-----", ".----", "..---", "...--", "....-", ".....", "-....", "--...", "---..", "----.",
]
string message = "MEET ME AT PIER 9 AT 1745"
string[] words = message.split(" ")
string[] encoded_words = []
for string word = words[]
    string[] encoded_letters = []
    for nat i < len(word)
        match alphabet.find(word[i])
            nat index
                encoded_letters = encoded_letters.append(codes[index])
    encoded_words = encoded_words.append(encoded_letters.join(" "))
string encoded = encoded_words.join(" / ")
string[] decoded_words = []
string[] morse_words = encoded.split(" / ")
for string word = morse_words[]
    string decoded_word = ""
    string[] symbols = word.split(" ")
    for string symbol = symbols[]
        for nat i < len(codes)
            if codes[i] == symbol
                decoded_word += alphabet[i]
                break
    decoded_words = decoded_words.append(decoded_word)
print(encoded + NL + decoded_words.join(" ") + NL)
