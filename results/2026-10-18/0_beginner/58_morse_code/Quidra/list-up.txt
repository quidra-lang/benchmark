import collections
nat index_of<T: equatable>(T[] values, T needle)
    return collections.index(values, needle)
string alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"
string[] codes = [
    ".-", "-...", "-.-.", "-..", ".", "..-.", "--.", "....", "..", ".---", "-.-", ".-..", "--",
    "-.", "---", ".--.", "--.-", ".-.", "...", "-", "..-", "...-", ".--", "-..-", "-.--", "--..",
    "-----", ".----", "..---", "...--", "....-", ".....", "-....", "--...", "---..", "----.",
]
string message = "MEET ME AT PIER 9 AT 1745"
int[] letters = alphabet.codepoints()
string[] encoded_words = [
    for string word = message.split(" ")[]
    then [for int character = word.codepoints()[] then codes[index_of(letters, character)]].join(" ")
]
string encoded = encoded_words.join(" / ")
string[] decoded_words = [
    for string word = encoded.split(" / ")[]
    then [for string symbol = word.split(" ")[] then alphabet[index_of(codes, symbol)]].join("")
]
