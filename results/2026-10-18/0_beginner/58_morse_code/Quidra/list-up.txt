import collections
nat index_of<T: equatable>(T[] values, T needle)
    return collections.index(values, needle)
alphabet := "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"
codes := [
    ".-", "-...", "-.-.", "-..", ".", "..-.", "--.", "....", "..", ".---", "-.-", ".-..", "--",
    "-.", "---", ".--.", "--.-", ".-.", "...", "-", "..-", "...-", ".--", "-..-", "-.--", "--..",
    "-----", ".----", "..---", "...--", "....-", ".....", "-....", "--...", "---..", "----.",
]
message := "MEET ME AT PIER 9 AT 1745"
letters := alphabet.codepoints()
encoded_words := [
    for word := message.split(" ")[]
    then [for character := word.codepoints()[] then codes[index_of<int>(letters, character)]].join(" ")
]
encoded := encoded_words.join(" / ")
decoded_words := [
    for word := encoded.split(" / ")[]
    then [for symbol := word.split(" ")[] then alphabet[index_of<string>(codes, symbol)]].join("")
]
