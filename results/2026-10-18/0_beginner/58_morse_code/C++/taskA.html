1#include <iostream>
2#include <string>
3#include <array>
4#include <algorithm>
5#include <sstream>
6
7int main() {
8 const std::string symbols = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789", message = "MEET ME AT PIER 9 AT 1745";
9 const std::array<std::string, 36> codes = {".-", "-...", "-.-.", "-..", ".", "..-.", "--.", "....",
10 "..", ".---", "-.-", ".-..", "--", "-.", "---", ".--.", "--.-", ".-.", "...", "-", "..-", "...-",
11 ".--", "-..-", "-.--", "--..", "-----", ".----", "..---", "...--", "....-", ".....", "-....", "--...",
12 "---..", "----."};
13 std::string encoded;
14 for (std::size_t i = 0; i < message.size(); i++) {
15 if (message[i] == ' ') {
16 encoded += " / ";
17 continue;
18 }
19 if (i && message[i - 1] != ' ')
20 encoded += ' ';
21 encoded += codes[symbols.find(message[i])];
22 }
23 std::cout << encoded << '\n';
24 std::istringstream input(encoded);
25 std::string decoded;
26 for (std::string token; input >> token;) {
27 if (token == "/")
28 decoded += ' ';
29 else
30 decoded += symbols[std::find(codes.begin(), codes.end(), token) - codes.begin()];
31 }
32 std::cout << decoded << '\n';
33}
