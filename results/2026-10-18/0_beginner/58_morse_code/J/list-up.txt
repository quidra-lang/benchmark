  result =. 0$<''
  start =. 0
  for_position. I.x E.y do.
    result =. result,< (position-start){.start}.y
    start =. position+#x
  end.
  result,<start}.y

join =: 4 : 0
  (#x) }. ; (<x) ,&.> y
)
  alphabet =. 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789'
  codes =. ' ' split '.- -... -.-. -.. . ..-. --. .... .. .--- -.- .-.. -- -. --- .--. --.- .-. ... - ..- ...- .-- -..- -.-- --.. ----- .---- ..--- ...-- ....- ..... -.... --... ---.. ----.'
  words =. ' ' split 'MEET ME AT PIER 9 AT 1745'
  encodedwords =. 0$<''
  for_box. words do.
    word =. >box
    encodedwords =. encodedwords,<' ' join (alphabet i.word){codes
  end.
  encoded =. ' / ' join encodedwords
  decodedwords =. 0$<''
  for_box. ' / ' split encoded do.
    tokens =. ' ' split >box
    word =. ''
    for_token. tokens do.
      index =. codes i.token
      word =. word,index{alphabet
    end.
    decodedwords =. decodedwords,<word
  end.
  smoutput ' ' join decodedwords
