symbols = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789'
codes = '.- -... -.-. -.. . ..-. --. .... .. .--- -.- .-.. -- -. --- .--. --.- .-. ... - ..- ...- .-- -..- -.-- --.. ----- .---- ..--- ...-- ....- ..... -.... --... ---.. ----.'.split(' ')
table = symbols.chars.zip(codes).to_h
reverse = table.invert
message = 'MEET ME AT PIER 9 AT 1745'
encoded = message.split(' ').map { |word| word.chars.map { |char| table[char] }.join(' ') }.join(' / ')
decoded = encoded.split(' / ').map { |word| word.split(' ').map { |code| reverse[code] }.join }.join(' ')
