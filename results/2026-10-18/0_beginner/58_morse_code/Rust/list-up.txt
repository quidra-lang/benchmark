    let symbols = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789";
    let codes = [
        ".-", "-...", "-.-.", "-..", ".", "..-.", "--.", "....", "..", ".---", "-.-", ".-..", "--",
        "-.", "---", ".--.", "--.-", ".-.", "...", "-", "..-", "...-", ".--", "-..-", "-.--",
        "--..", "-----", ".----", "..---", "...--", "....-", ".....", "-....", "--...", "---..",
        "----.",
    ];
    let message = "MEET ME AT PIER 9 AT 1745";
    let encoded = message
        .split(' ')
        .map(|w| {
            w.bytes()
                .map(|c| codes[symbols.bytes().position(|s| s == c).unwrap()])
                .collect::<Vec<_>>()
                .join(" ")
        })
        .collect::<Vec<_>>()
        .join(" / ");
    let decoded = encoded
        .split(" / ")
        .map(|w| {
            w.split(' ')
                .map(|token| {
                    symbols.as_bytes()[codes.iter().position(|&c| c == token).unwrap()] as char
                })
                .collect::<String>()
        })
        .collect::<Vec<_>>()
        .join(" ");
