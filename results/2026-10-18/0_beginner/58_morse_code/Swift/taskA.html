3let alphabet = Array("ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789")

6let reverse = Dictionary(uniqueKeysWithValues: zip(codes, alphabet))

8let encoded = message.split(separator: " ")
9 .map { word in word.map { forward[$0]! }.joined(separator: " ") }

12 .map { word in String(word.split(separator: " ").map { reverse[String($0)]! }) }
