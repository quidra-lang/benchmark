1local alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"
2local codes = ".- -... -.-. -.. . ..-. --. .... .. .--- -.- .-.. -- -. --- .--. --.- .-. ... - ..- ...- .-- -..- -.-- --.. ----- .---- ..--- ...-- ....- ..... -.... --... ---.. ----."
3local mapping, reverse_mapping, index = {}, {}, 1
4for code in codes:gmatch("%S+") do
5 local letter = alphabet:sub(index, index)
6 mapping[letter] = code
7 reverse_mapping[code] = letter
8 index = index + 1
9end
10local words = {}
11for word in ("MEET ME AT PIER 9 AT 1745"):gmatch("%S+") do
12 local letters = {}
13 for letter in word:gmatch(".") do letters[#letters + 1] = mapping[letter] end
14 words[#words + 1] = table.concat(letters, " ")
15end
16local encoded = table.concat(words, " / ")
17local decoded_words = {}
18for word in encoded:gmatch("[^/]+") do
19 local letters = {}
20 for code in word:gmatch("%S+") do letters[#letters + 1] = reverse_mapping[code] end
21 decoded_words[#decoded_words + 1] = table.concat(letters)
22end
23io.write(encoded, "\n", table.concat(decoded_words, " "), "\n")
