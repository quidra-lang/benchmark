const symbols = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789";
const codes = ".- -... -.-. -.. . ..-. --. .... .. .--- -.- .-.. -- -. --- .--. --.- .-. ... - ..- ...- .-- -..- -.-- --.. ----- .---- ..--- ...-- ....- ..... -.... --... ---.. ----.".split(" ");
const table = new Map([...symbols].map((char, i) => [char, codes[i]]));
const reverse = new Map(codes.map((code, i) => [code, symbols[i]]));
const message = "MEET ME AT PIER 9 AT 1745";
const encoded = message.split(" ").map(word => [...word].map(char => table.get(char)).join(" ")).join(" / ");
const decoded = encoded.split(" / ").map(word => word.split(" ").map(code => reverse.get(code)).join("")).join(" ");
