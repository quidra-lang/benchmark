  final columns = List.filled(5, 0);
  var total = 0;
  for (final row in matrix) {
    var sum = 0;
    for (var j = 0; j < 5; j++) {
      final v = row[j];
      sum += v;
      columns[j] += v;
    }
    total += sum;
    out.writeln(sum.toString().padLeft(6));
  }
  for (final v in columns) {
    out.write(v.toString().padLeft(6));
  }
  out.writeln(total.toString().padLeft(6));
