3let words = "we planted beans and carrots in the garden behind our old house".split(separator: " ")
4let longest = words.max { $0.count < $1.count }!
5let shortest = words.min { $0.count < $1.count }!
6let average = Double(words.reduce(0) { $0 + $1.count }) / Double(words.count)
7print("words: \(words.count)")
8print("longest: \(longest)")
9print("shortest: \(shortest)")
10print(String(format: "average length: %.2f", locale: Locale(identifier: "en_US_POSIX"), average))
