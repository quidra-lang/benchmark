3
4int main(void) {
5 char sentence[] = "we planted beans and carrots in the garden behind our old house";
6 char *longest = NULL, *shortest = NULL;
7 int count = 0, total = 0;
8 for (char *w = strtok(sentence, " "); w; w = strtok(NULL, " ")) {
9 int len = (int)strlen(w);
10 count++;
11 total += len;
12 if (!longest || len > (int)strlen(longest))
13 longest = w;
14 if (!shortest || len < (int)strlen(shortest))
15 shortest = w;
16 }
17 printf("words: %d\nlongest: %s\nshortest: %s\naverage length: %.2f\n",
18 count, longest, shortest, (double)total / count);
19 return 0;
20}
