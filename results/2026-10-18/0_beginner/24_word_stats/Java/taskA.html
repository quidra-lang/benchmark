1import java.util.Arrays;
2import java.util.Comparator;

4
5public class Main {

8 var byLength = Comparator.comparingInt(String::length);
9 String longest = Arrays.stream(words).sorted(byLength.reversed()).findFirst().orElseThrow();
10 String shortest = Arrays.stream(words).sorted(byLength).findFirst().orElseThrow();
11 double average = Arrays.stream(words).mapToInt(String::length).average().orElseThrow();

13 System.out.println("longest: " + longest);
