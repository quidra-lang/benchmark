3fn romanValue(c: u8) i32 {
4 return switch (c) {
5 'I' => 1,
6 'V' => 5,
7 'X' => 10,
8 'L' => 50,
9 'C' => 100,
10 'D' => 500,
11 'M' => 1000,
12 else => unreachable,
13 };
14}

16pub fn main(init: std.process.Init) !void {
17 var output_buffer: [8192]u8 = undefined;
18 var writer = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
19 const out = &writer.interface;
20 const values = [_]i32{ 1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1 };
21 const symbols = [_][]const u8{ "M", "CM", "D", "CD", "C", "XC", "L", "XL", "X", "IX", "V", "IV", "I" };
22 for ([_]i32{ 9, 14, 49, 94, 444, 1994, 2768, 3999 }) |n| {
23 var remaining = n;
24 var roman: [32]u8 = undefined;
25 var len: usize = 0;
26 for (values, symbols) |value, symbol| {
27 while (remaining >= value) {
28 @memcpy(roman[len..][0..symbol.len], symbol);
29 len += symbol.len;
30 remaining -= value;
31 }
32 }
33 var decoded: i32 = 0;
34 for (roman[0..len], 0..) |c, i| {
35 const v = romanValue(c);
36 decoded += if (i + 1 < len and v < romanValue(roman[i + 1])) -v else v;
37 }
38 try out.print("{d} -> {s} -> {d}\n", .{ n, roman[0..len], decoded });
39 }
40 try out.flush();
41}
