1local function to_roman(number)
2 local result = ""
3 local numerals = {
4 {1000, "M"}, {900, "CM"}, {500, "D"}, {400, "CD"}, {100, "C"}, {90, "XC"}, {50, "L"},
5 {40, "XL"}, {10, "X"}, {9, "IX"}, {5, "V"}, {4, "IV"}, {1, "I"},
6 }
7 for _, pair in ipairs(numerals) do
8 while number >= pair[1] do
9 result = result .. pair[2]
10 number = number - pair[1]
11 end
12 end
13 return result
14end

16local function from_roman(roman)

18 for i = 1, #roman do
19 local value = values[roman:sub(i, i)]
20 total = total + (i < #roman and value < values[roman:sub(i + 1, i + 1)] and -value or value)
21 end
22 return total
23end
