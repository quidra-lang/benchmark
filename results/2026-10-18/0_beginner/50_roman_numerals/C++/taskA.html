5int value(char c) {
6 switch (c) {
7 case 'I': return 1;
8 case 'V': return 5;
9 case 'X': return 10;
10 case 'L': return 50;
11 case 'C': return 100;
12 case 'D': return 500;
13 case 'M': return 1000;
14 default: return 0;
15 }
16}
17
18int main() {
19 const std::array<int, 13> values = {1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1};
20 const std::array<std::string, 13> symbols = {"M", "CM", "D", "CD", "C", "XC", "L", "XL", "X", "IX",
21 "V", "IV", "I"};
22 for (int n : std::array<int, 8>{9, 14, 49, 94, 444, 1994, 2768, 3999}) {
23 int remaining = n;
24 std::string roman;
25 for (int i = 0; i < 13; i++)
26 while (remaining >= values[i]) {
27 roman += symbols[i];
28 remaining -= values[i];
29 }
30 int result = 0;
31 for (std::size_t i = 0; i < roman.size(); i++) {
32 int v = value(roman[i]), next = i + 1 < roman.size() ? value(roman[i + 1]) : 0;
33 result += v < next ? -v : v;
34 }
35 std::cout << n << " -> " << roman << " -> " << result << '\n';
36 }
37}
