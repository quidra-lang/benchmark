denominations = [10000, 5000, 1000, 500, 100, 50, 10, 5, 1]
  remaining, pieces = amount, 0
  parts = []
  denominations.each do |face|
    count, remaining = remaining.divmod(face)
    if count.positive?
      parts << "#{face}x#{count}"
      pieces += count
    end
  end
  puts "#{amount} yen: #{parts.join(' ')} (#{pieces} pieces)"
