    std::array<int, 9> denominations = {10000, 5000, 1000, 500, 100, 50, 10, 5, 1};
        int remaining = amount, total = 0;
        std::cout << amount << " yen:";
        for (int d : denominations) {
            int count = remaining / d;
            remaining %= d;
            if (count) {
                std::cout << ' ' << d << 'x' << count;
                total += count;
            }
        }
