int[] amounts := [3780, 15420, 99999, 506, 48263]
int[] denominations := [10000, 5000, 1000, 500, 100, 50, 10, 5, 1]
for amount := amounts[]
    remaining := amount
    int pieces := 0
    print("{amount} yen:")
    for face := denominations[]
        count := remaining / face
        remaining %= face
        pieces += count
        if count > 0
            print(" {face}x{count}")
    print.line(" ({pieces} pieces)")
