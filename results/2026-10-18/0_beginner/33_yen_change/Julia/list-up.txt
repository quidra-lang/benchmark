    remaining, total = amount, 0
    pieces = String[]
    for face in [10000, 5000, 1000, 500, 100, 50, 10, 5, 1]
        quantity, remaining = divrem(remaining, face)
        if quantity > 0
            push!(pieces, string(face, "x", quantity))
            total += quantity
        end
    end
    println(amount, " yen: ", join(pieces, " "), " (", total, " pieces)")
