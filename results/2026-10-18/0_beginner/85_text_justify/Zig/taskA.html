3pub fn main(init: std.process.Init) !void {
4 var output_buffer: [8192]u8 = undefined;

6 const out = &writer.interface;
7 const paragraph = "Good software is written for people first and for machines second. A clear program explains its purpose through careful names, small functions and honest tests. When a team agrees on shared rules for internationalization, documentation and code review, every newcomer can read the code quickly, change it safely and ship new improvements with confidence instead of fear, week after week.";
8 var words: [100][]const u8 = undefined;
9 var count: usize = 0;
10 var tokens = std.mem.splitScalar(u8, paragraph, ' ');
11 while (tokens.next()) |word| {
12 words[count] = word;
13 count += 1;
14 }
15 var start: usize = 0;
16 while (start < count) {
17 var end = start;
18 var letters: usize = 0;
19 while (end < count and letters + words[end].len + (end - start) <= 30) {
20 letters += words[end].len;
21 end += 1;
22 }
23 const n = end - start;
24 try out.writeByte('|');
25 if (end == count or n == 1) {
26 for (words[start..end], 0..) |word, i| {
27 if (i > 0)
28 try out.writeByte(' ');
29 try out.writeAll(word);
30 }
31 try out.splatByteAll(' ', 30 - (letters + n - 1));
32 } else {
33 const spaces = 30 - letters;
34 const gaps = n - 1;
35 for (words[start..end], 0..) |word, i| {
36 try out.writeAll(word);
37 if (i < gaps) {
38 const width = spaces / gaps + @as(usize, if (i < spaces % gaps) 1 else 0);
39 try out.splatByteAll(' ', width);
40 }
41 }
42 }
43 try out.writeAll("|\n");
44 start = end;
45 }
46 try out.flush();
