1using System;
2using System.Collections.Generic;
3using System.Linq;
4using System.Text;
5
6string paragraph = "Good software is written for people first and for machines second. A clear program explains its purpose " +
7 "through careful names, small functions and honest tests. When a team agrees on shared rules for " +
8 "internationalization, documentation and code review, every newcomer can read the code quickly, change it " +
9 "safely and ship new improvements with confidence instead of fear, week after week.";
10var lines = new List<List<string>>();
11var current = new List<string>();
12int letters = 0;
13foreach (string word in paragraph.Split(' '))
14{
15 if (current.Count > 0 && letters + current.Count + word.Length > 30)
16 {
17 lines.Add(current);
18 current = new List<string>();
19 letters = 0;
20 }
21 current.Add(word);
22 letters += word.Length;
23}
24if (current.Count > 0) lines.Add(current);
25for (int i = 0; i < lines.Count; i++)
26{
27 var words = lines[i];
28 string text;
29 if (i == lines.Count - 1 || words.Count == 1)
30 {
31 text = string.Join(" ", words).PadRight(30);
32 }
33 else
34 {
35 int spaces = 30 - words.Sum(w => w.Length), gaps = words.Count - 1;
36 var line = new StringBuilder(words[0]);
37 for (int j = 1; j < words.Count; j++)
38 {
39 line.Append(' ', spaces / gaps + (j - 1 < spaces % gaps ? 1 : 0));
40 line.Append(words[j]);
41 }
42 text = line.ToString();
43 }
44 Console.WriteLine($"|{text}|");
45}
