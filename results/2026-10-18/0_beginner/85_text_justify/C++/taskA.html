15 letters += words[end].size();
16 end++;
17 }
18 std::size_t gaps = end - start - 1;

23 if (i + 1 < end) {
24 std::size_t spaces =
25 last || gaps == 0 ? 1 : (30 - letters) / gaps + (i - start < (30 - letters) % gaps);
26 line += std::string(spaces, ' ');
27 }
28 }
29 line.resize(30, ' ');
30 std::cout << '|' << line << "|\n";
