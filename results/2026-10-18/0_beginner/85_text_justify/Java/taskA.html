1import java.util.ArrayList;
2import java.util.List;
3import java.util.Locale;
4
5public class Main {
6 public static void main(String[] args) {
7 String paragraph = "Good software is written for people first and for machines second. "
8 + "A clear program explains its purpose through careful names, small functions and honest tests. "
9 + "When a team agrees on shared rules for internationalization, documentation and code review, "
10 + "every newcomer can read the code quickly, change it safely and ship new improvements "
11 + "with confidence instead of fear, week after week.";
12 List<List<String>> lines = new ArrayList<>();
13 List<String> current = new ArrayList<>();
14 int letters = 0;
15 for (String word : paragraph.split(" ")) {
16 if (!current.isEmpty() && letters + current.size() + word.length() > 30) {
17 lines.add(current);
18 current = new ArrayList<>();
19 letters = 0;
20 }
21 current.add(word);
22 letters += word.length();
23 }
24 if (!current.isEmpty()) {
25 lines.add(current);
26 }
27 for (int i = 0; i < lines.size(); i++) {
28 List<String> words = lines.get(i);
29 String text;
30 if (i == lines.size() - 1 || words.size() == 1) {
31 text = String.format(Locale.ROOT, "%-30s", String.join(" ", words));
32 } else {
33 int spaces = 30 - words.stream().mapToInt(String::length).sum();
34 int gaps = words.size() - 1;
35 StringBuilder line = new StringBuilder(words.get(0));
36 for (int j = 1; j < words.size(); j++) {
37 line.append(" ".repeat(spaces / gaps + (j - 1 < spaces % gaps ? 1 : 0)));
38 line.append(words.get(j));
39 }
40 text = line.toString();
41 }
42 System.out.println("|" + text + "|");
43 }
44 }
45}
