1local paragraph = "Good software is written for people first and for machines second. A clear program explains its purpose through careful names, small functions and honest tests. When a team agrees on shared rules for internationalization, documentation and code review, every newcomer can read the code quickly, change it safely and ship new improvements with confidence instead of fear, week after week."
2local lines, current, letters = {}, {}, 0
3for word in paragraph:gmatch("%S+") do
4 if #current > 0 and letters + #current + #word > 30 then
5 lines[#lines + 1] = current
6 current, letters = {}, 0
7 end
8 current[#current + 1] = word
9 letters = letters + #word
10end
11lines[#lines + 1] = current
12for number, words in ipairs(lines) do
13 local line
14 if number == #lines or #words == 1 then
15 line = table.concat(words, " ")
16 line = line .. string.rep(" ", 30 - #line)
17 else
18 local total = 0
19 for _, word in ipairs(words) do total = total + #word end
20 local spaces, gaps = 30 - total, #words - 1
21 local base, extra = spaces // gaps, spaces % gaps
22 line = words[1]
23 for i = 1, gaps do
24 line = line .. string.rep(" ", base + (i <= extra and 1 or 0)) .. words[i + 1]
25 end
26 end
27 io.write("|", line, "|\n")
28end
