1import Foundation
2
3let posix = Locale(identifier: "en_US_POSIX")
4let names = ["Tanaka", "Suzuki", "Yamada", "Ito", "Kato", "Mori"]
5let scores = [[78, 85, 94], [88, 74, 82], [65, 90, 71], [80, 79, 85], [92, 88, 95], [70, 62, 78]]
6let totals = scores.map { $0.reduce(0, +) }
7let order = names.indices.sorted { totals[$0] == totals[$1] ? names[$0] < names[$1] : totals[$0] > totals[$1] }
8var rank = 0, previous = -1
9for (position, i) in order.enumerated() {
10 if totals[i] != previous { rank = position + 1 }
11 previous = totals[i]
12 let average = String(format: "%.2f", locale: posix, Double(totals[i]) / 3)
13 print("\(rank). \(names[i]) total=\(totals[i]) average=\(average)")
14}
