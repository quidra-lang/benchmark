1import java.util.Arrays;
2import java.util.Comparator;
3import java.util.List;
4import java.util.Locale;
5import java.util.stream.IntStream;
6
7public class Main {
8 public static void main(String[] args) {
9 String[] names = {"Tanaka", "Suzuki", "Yamada", "Ito", "Kato", "Mori"};
10 int[][] scores = {{78, 85, 94}, {88, 74, 82}, {65, 90, 71}, {80, 79, 85}, {92, 88, 95}, {70, 62, 78}};
11 int[] totals = Arrays.stream(scores).mapToInt(row -> Arrays.stream(row).sum()).toArray();
12 List<Integer> order = IntStream.range(0, names.length).boxed()
13 .sorted(Comparator.<Integer>comparingInt(i -> totals[i]).reversed().thenComparing(i -> names[i]))
14 .toList();
15 int rank = 0, previous = -1;
16 for (int position = 0; position < order.size(); position++) {
17 int i = order.get(position);
18 if (totals[i] != previous) {
19 rank = position + 1;
20 }
21 previous = totals[i];
22 System.out.printf(Locale.ROOT, "%d. %s total=%d average=%.2f%n",
23 rank, names[i], totals[i], totals[i] / 3.0);
24 }
25 }
26}
