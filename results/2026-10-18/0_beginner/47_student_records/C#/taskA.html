1using System;
2using System.Globalization;
3using System.Linq;
4
5CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
6string[] names = { "Tanaka", "Suzuki", "Yamada", "Ito", "Kato", "Mori" };
7int[][] scores =
8[
9 [78, 85, 94], [88, 74, 82], [65, 90, 71], [80, 79, 85], [92, 88, 95], [70, 62, 78],
10];
11int[] totals = scores.Select(row => row.Sum()).ToArray();
12int[] order = Enumerable.Range(0, names.Length)
13 .OrderByDescending(i => totals[i])
14 .ThenBy(i => names[i], StringComparer.Ordinal)
15 .ToArray();
16int rank = 0, previous = -1;
17for (int position = 0; position < order.Length; position++)
18{
19 int i = order[position];
20 if (totals[i] != previous) rank = position + 1;
21 previous = totals[i];
22 Console.WriteLine($"{rank}. {names[i]} total={totals[i]} average={totals[i] / 3.0:F2}");
23}
