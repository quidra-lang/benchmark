1import java.util.Locale
2
3fun main() {
4 val names = listOf("Tanaka", "Suzuki", "Yamada", "Ito", "Kato", "Mori")
5 val scores = listOf(
6 listOf(78, 85, 94), listOf(88, 74, 82), listOf(65, 90, 71), listOf(80, 79, 85),
7 listOf(92, 88, 95), listOf(70, 62, 78),
8 )
9 val totals = scores.map { it.sum() }
10 val order = names.indices.sortedWith(compareByDescending<Int> { totals[it] }.thenBy { names[it] })
11 var rank = 0
12 var previous = -1
13 for ((position, i) in order.withIndex()) {
14 if (totals[i] != previous) rank = position + 1
15 previous = totals[i]
16 println("%d. %s total=%d average=%.2f".format(Locale.ROOT, rank, names[i], totals[i], totals[i] / 3.0))
17 }
18}
