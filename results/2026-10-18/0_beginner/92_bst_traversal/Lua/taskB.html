1local function insert_node(node, value)

10local function delete_node(node, value)

12 if value < node.value then
13 node.left = delete_node(node.left, value)

15 node.right = delete_node(node.right, value)

18 elseif not node.right then

27end
28local function traversal(node, mode, result)
