1local function insert_node(node, value)
2 if not node then return {value = value} end
3 if value < node.value then

5 else

8 return node
9end
10local function delete_node(node, value)
11 if value < node.value then
12 node.left = delete_node(node.left, value)
13 elseif value > node.value then
14 node.right = delete_node(node.right, value)
15 elseif not node.left then
16 return node.right
17 elseif not node.right then
18 return node.left
19 else
20 local successor = node.right
21 while successor.left do successor = successor.left end
22 node.value = successor.value

24 end
25 return node
26end
27local function traversal(node, mode, result)

32end
33local function height(node)
