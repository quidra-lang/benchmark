Node = Struct.new(:key, :left, :right)
def insert(node, key)
  return Node.new(key) if node.nil?
  if key < node.key
    node.left = insert(node.left, key)
  else
    node.right = insert(node.right, key)
  end
  node
end
def delete(node, key)
  if key < node.key
    node.left = delete(node.left, key)
  elsif key > node.key
    node.right = delete(node.right, key)
  else
    return node.right if node.left.nil?
    return node.left if node.right.nil?
    successor = node.right
    successor = successor.left until successor.left.nil?
    node.key = successor.key
    node.right = delete(node.right, successor.key)
  end
  node
end
def traverse(node, order)
  return [] if node.nil?
  left, right = traverse(node.left, order), traverse(node.right, order)
  return [node.key] + left + right if order == 'preorder'
end
def height(node)
  node.nil? ? 0 : 1 + [height(node.left), height(node.right)].max
end
root = nil
[50, 27, 74, 13, 39, 62, 88, 6, 31, 45, 70, 34].each { |key| root = insert(root, key) }
root = delete(root, 6)
root = delete(root, 27)
puts "preorder after deletions: #{traverse(root, 'preorder').join(' ')}"
puts "height after deletions: #{height(root)}"
