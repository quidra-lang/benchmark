1class Node {

7func insert(_ node: Node?, _ key: Int) -> Node {
8 guard let node else { return Node(key) }
9 if key < node.key {
10 node.left = insert(node.left, key)
11 } else {
12 node.right = insert(node.right, key)
13 }
14 return node
15}
16
17func delete(_ node: Node?, _ key: Int) -> Node? {
18 guard let node else { return nil }
19 if key < node.key {
20 node.left = delete(node.left, key)
21 } else if key > node.key {
22 node.right = delete(node.right, key)
23 } else {
24 if node.left == nil { return node.right }
25 if node.right == nil { return node.left }
26 var successor = node.right!
27 while let left = successor.left { successor = left }
28 node.key = successor.key
29 node.right = delete(node.right, successor.key)
30 }
31 return node
32}

34func traverse(_ node: Node?, _ mode: Int, _ order: inout [Int]) {
35 guard let node else { return }
36 if mode == 1 { order.append(node.key) }
37 traverse(node.left, mode, &order)
38 if mode == 0 { order.append(node.key) }
39 traverse(node.right, mode, &order)
40 if mode == 2 { order.append(node.key) }
41}

43func height(_ node: Node?) -> Int {
44 guard let node else { return 0 }
45 return 1 + max(height(node.left), height(node.right))
46}
