1class Node {
2 var key: Int
3 var left: Node?, right: Node?
4 init(_ key: Int) { self.key = key }
5}
6func insert(_ node: Node?, _ key: Int) -> Node {
7 guard let node else { return Node(key) }
8 if key < node.key {
9 node.left = insert(node.left, key)
10 } else {
11 node.right = insert(node.right, key)
12 }
13 return node
14}
15func delete(_ node: Node?, _ key: Int) -> Node? {
16 if key < node.key {
17 node.left = delete(node.left, key)
18 } else if key > node.key {
19 node.right = delete(node.right, key)
20 } else {
21 if node.left == nil { return node.right }
22 if node.right == nil { return node.left }
23 var successor = node.right!
24 while let left = successor.left { successor = left }
25 node.key = successor.key
26 node.right = delete(node.right, successor.key)
27 }
28 return node
29}
30func traverse(_ node: Node?, _ mode: Int, _ order: inout [Int]) {
31 guard let node else { return }
32 if mode == 1 { order.append(node.key) }
33 traverse(node.left, mode, &order)
34 traverse(node.right, mode, &order)
35}
36func height(_ node: Node?) -> Int {
37 guard let node else { return 0 }
38 return 1 + max(height(node.left), height(node.right))
39}
