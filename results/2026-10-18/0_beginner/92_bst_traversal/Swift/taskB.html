1class Node {
2 var key: Int
3 var left: Node?, right: Node?
4 init(_ key: Int) { self.key = key }
5}
6func insert(_ node: Node?, _ key: Int) -> Node {
7 guard let node else { return Node(key) }
8 if key < node.key {
9 node.left = insert(node.left, key)
10 } else {
11 node.right = insert(node.right, key)
12 }
13 return node
14}
15func delete(_ node: Node?, _ key: Int) -> Node? {
16 guard let node else { return nil }
17 if key < node.key {
18 node.left = delete(node.left, key)
19 } else if key > node.key {
20 node.right = delete(node.right, key)
21 } else {
22 if node.left == nil { return node.right }
23 if node.right == nil { return node.left }
24 var successor = node.right!
25 while let left = successor.left { successor = left }
26 node.key = successor.key
27 node.right = delete(node.right, successor.key)
28 }
29 return node
30}
31func traverse(_ node: Node?, _ mode: Int, _ order: inout [Int]) {
32 guard let node else { return }
33 if mode == 1 { order.append(node.key) }
34 traverse(node.left, mode, &order)
35 traverse(node.right, mode, &order)
36}
37func height(_ node: Node?) -> Int {
38 guard let node else { return 0 }
39 return 1 + max(height(node.left), height(node.right))
40}
