3const Node = struct { key: i32, left: ?usize = null, right: ?usize = null };
4
5fn traverse(out: *std.Io.Writer, nodes: *const [12]Node, root: ?usize, mode: usize) std.Io.Writer.Error!void {
6 const i = root orelse return;
7 if (mode == 1)
8 try out.print(" {d}", .{nodes[i].key});
9 try traverse(out, nodes, nodes[i].left, mode);
10 if (mode == 0)
11 try out.print(" {d}", .{nodes[i].key});
12 try traverse(out, nodes, nodes[i].right, mode);
13 if (mode == 2)
14 try out.print(" {d}", .{nodes[i].key});
15}

17fn height(nodes: *const [12]Node, root: ?usize) usize {
18 const i = root orelse return 0;
19 return 1 + @max(height(nodes, nodes[i].left), height(nodes, nodes[i].right));
20}

22fn remove(nodes: *[12]Node, root: ?usize, key: i32) ?usize {
23 const i = root orelse return null;
24 if (key < nodes[i].key) {
25 nodes[i].left = remove(nodes, nodes[i].left, key);
26 } else if (key > nodes[i].key) {
27 nodes[i].right = remove(nodes, nodes[i].right, key);
28 } else {
29 if (nodes[i].left == null)
30 return nodes[i].right;
31 if (nodes[i].right == null)
32 return nodes[i].left;
33 var successor = nodes[i].right.?;
34 while (nodes[successor].left) |next|
35 successor = next;
36 nodes[i].key = nodes[successor].key;
37 nodes[i].right = remove(nodes, nodes[i].right, nodes[successor].key);
38 }
39 return i;
40}

42pub fn main(init: std.process.Init) !void {

45 const out = &writer.interface;
46 var nodes: [12]Node = undefined;
47 var len: usize = 0;
48 var root: ?usize = null;
49 for ([_]i32{ 50, 27, 74, 13, 39, 62, 88, 6, 31, 45, 70, 34 }) |key| {
50 var link = &root;

52 link = if (key < nodes[i].key) &nodes[i].left else &nodes[i].right;
53 }
54 nodes[len] = .{ .key = key };
55 link.* = len;
56 len += 1;
57 }
58 const labels = [_][]const u8{ "inorder:", "preorder:", "postorder:" };

70 const i = queue[head];
71 head += 1;

79 tail += 1;
80 }
