3function insert(node: TreeNode | null, key: number): TreeNode {
4 if (node === null) return {key, left: null, right: null};
5 if (key < node.key) node.left = insert(node.left, key);
6 else node.right = insert(node.right, key);
7 return node;
8}

10function remove(node: TreeNode, key: number): TreeNode | null {
11 if (key < node.key) {
12 node.left = remove(node.left!, key);
13 } else if (key > node.key) {
14 node.right = remove(node.right!, key);
15 } else {

17 if (node.right === null) return node.left;
18 let successor = node.right;
19 while (successor.left !== null) successor = successor.left;
20 node.key = successor.key;
21 node.right = remove(node.right, successor.key);
22 }
23 return node;
24}

26function traverse(node: TreeNode | null, order: string): number[] {
27 if (node === null) return [];
28 const left = traverse(node.left, order), right = traverse(node.right, order);
29 if (order === "inorder") return [...left, node.key, ...right];
30 if (order === "preorder") return [node.key, ...left, ...right];
31 return [...left, ...right, node.key];
32}
