import c = collections
class Tree()
    c.BinaryNodes<int> nodes = c.BinaryNodes<int>()
    c.NodeId | none root = none
    private c.NodeId .insert(c.NodeId | none id, int key)
        match id
            none
                return .nodes.add(key)
            c.NodeId
                auto node = nodes.get(id)
                if key < node.value
                    node.left = .insert(node.left, key)
                else
                    node.right = .insert(node.right, key)
                .nodes.set(id, node)
                return id
    void .add(int key)
        root = .insert(root, key)
    private c.NodeId | none .erase(c.NodeId | none id, int key)
        match id
            none
                return none
            c.NodeId
                auto node = nodes.get(id)
                if key < node.value
                    node.left = .erase(node.left, key)
                elif key > node.value
                    node.right = .erase(node.right, key)
                else
                    match node.right
                        none
                            return node.left
                        c.NodeId right
                            if not nodes.present(node.left)
                                return right
                            (auto successor, auto remaining) = .nodes.detach_leftmost(right)
                            node.value = successor
                            node.right = remaining
                .nodes.set(id, node)
                return id
    void .remove(int key)
        root = .erase(root, key)
