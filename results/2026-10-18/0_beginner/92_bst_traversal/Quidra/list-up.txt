import collections
import math
import text
class Node(int key, int left = -1, int right = -1)
void visit(int index, int mode, Node[] nodes, int[] &order)
    if index == -1
        return
    Node node = nodes[index]
    if mode == 1
        .order.push(node.key)
    visit(node.left, mode, nodes, &order)
    visit(node.right, mode, nodes, &order)
int insert(int index, int child, Node[] &nodes)
    if index == -1
        return child
    if nodes[child].key < nodes[index].key
        nodes[index].left = insert(nodes[index].left, child, &nodes)
    else
        nodes[index].right = insert(nodes[index].right, child, &nodes)
    return index
int height(int index, Node[] nodes)
    if index == -1
        return 0
    return 1 + math.max(height(nodes[index].left, nodes), height(nodes[index].right, nodes))
int erase(int index, int key, Node[] &nodes)
    if index == -1
        return -1
    Node node = nodes[index]
    if key < node.key
        nodes[index].left = erase(node.left, key, &nodes)
    elif key > node.key
        nodes[index].right = erase(node.right, key, &nodes)
    else
        if node.left == -1
            return node.right
        if node.right == -1
            return node.left
        int successor = node.right
        while nodes[successor].left != -1
            successor = nodes[successor].left
        nodes[index].key = nodes[successor].key
        nodes[index].right = erase(node.right, nodes[successor].key, &nodes)
    return index
void show(string label, int[] order)
    print.line("{label}: {text.join(order, " ")}")
int[] keys = [50, 27, 74, 13, 39, 62, 88, 6, 31, 45, 70, 34]
Node[] nodes = [for int key = keys[] then Node(key)]
int root = -1
for int i < 12
    root = insert(root, i, &nodes)
root = erase(root, 6, &nodes)
root = erase(root, 27, &nodes)
int[] after = []
visit(root, 1, nodes, &after)
show("preorder after deletions", after)
print.line("height after deletions: {height(root, nodes)}")
