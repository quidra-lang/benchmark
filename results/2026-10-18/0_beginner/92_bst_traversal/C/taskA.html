3typedef struct {
4 int key, left, right;
5} Node;
6
7Node nodes[12];
8int used = 0;
9
10int insert(int root, int key) {
11 if (root < 0) {
12 int v = used++;
13 nodes[v] = (Node){key, -1, -1};
14 return v;
15 }
16 if (key < nodes[root].key)
17 nodes[root].left = insert(nodes[root].left, key);
18 else
19 nodes[root].right = insert(nodes[root].right, key);
20 return root;
21}

23void traverse(int root, int mode) {
24 if (root < 0)
25 return;
26 if (mode == 1)
27 printf(" %d", nodes[root].key);
28 traverse(nodes[root].left, mode);
29 if (mode == 0)
30 printf(" %d", nodes[root].key);
31 traverse(nodes[root].right, mode);
32 if (mode == 2)
33 printf(" %d", nodes[root].key);
34}

36int height(int root) {
37 if (root < 0)
38 return 0;
39 int l = height(nodes[root].left), r = height(nodes[root].right);
40 return 1 + (l > r ? l : r);
41}

43int erase(int root, int key) {
44 if (root < 0)
45 return root;
46 if (key < nodes[root].key) {
47 nodes[root].left = erase(nodes[root].left, key);
48 } else if (key > nodes[root].key) {
49 nodes[root].right = erase(nodes[root].right, key);
50 } else {
51 if (nodes[root].left < 0)
52 return nodes[root].right;
53 if (nodes[root].right < 0)
54 return nodes[root].left;
55 int successor = nodes[root].right;
56 while (nodes[successor].left >= 0)
57 successor = nodes[successor].left;
58 nodes[root].key = nodes[successor].key;
59 nodes[root].right = erase(nodes[root].right, nodes[successor].key);
60 }
61 return root;
62}
