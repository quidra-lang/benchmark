17function delete_node(node, value)
18 node === nothing && return nothing
19 if value < node.value
20 node.left = delete_node(node.left, value)
21 elseif value > node.value
22 node.right = delete_node(node.right, value)
23 elseif node.left === nothing
24 return node.right
25 elseif node.right === nothing

38function traversal(node, mode)
39 node === nothing && return Int[]
40 left, right = traversal(node.left, mode), traversal(node.right, mode)
41 if mode == "inorder"

43 elseif mode == "preorder"
44 return vcat(node.value, left, right)
45 else
46 return vcat(left, right, node.value)
47 end
