7static Node Insert(Node? node, int key)
8{
9 if (node is null) return new Node(key);
10 if (key < node.Key) node.Left = Insert(node.Left, key);
11 else node.Right = Insert(node.Right, key);
12 return node;
13}
14
15static Node? Delete(Node? node, int key)
16{
17 if (node is null) return null;
18 if (key < node.Key)
19 {
20 node.Left = Delete(node.Left, key);
21 }
22 else if (key > node.Key)
23 {
24 node.Right = Delete(node.Right, key);
25 }
26 else
27 {
28 if (node.Left is null) return node.Right;
29 if (node.Right is null) return node.Left;
30 Node successor = node.Right;
31 while (successor.Left is not null) successor = successor.Left;
32 node.Key = successor.Key;
33 node.Right = Delete(node.Right, successor.Key);
34 }
35 return node;
36}

38static void Traverse(Node? node, int mode, List<int> order)
39{
40 if (node is null) return;
41 if (mode == 1) order.Add(node.Key);
42 Traverse(node.Left, mode, order);
43 if (mode == 0) order.Add(node.Key);
44 Traverse(node.Right, mode, order);
45 if (mode == 2) order.Add(node.Key);
46}

48static int Height(Node? node) => node is null ? 0 : 1 + Math.Max(Height(node.Left), Height(node.Right));
