2fun insert(node: Node?, key: Int): Node {

4 if (key < node.key) node.left = insert(node.left, key) else node.right = insert(node.right, key)

6}
7fun delete(node: Node?, key: Int): Node? {
8 if (key < node.key) {
9 node.left = delete(node.left, key)
10 } else if (key > node.key) {
11 node.right = delete(node.right, key)
12 } else {
13 if (node.left == null) return node.right
14 if (node.right == null) return node.left

16 while (successor.left != null) successor = successor.left!!

21}
