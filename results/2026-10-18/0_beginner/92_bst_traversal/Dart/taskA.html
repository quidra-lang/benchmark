3class Node {
4 int key;
5 Node? left, right;
6 Node(this.key);

9Node insert(Node? node, int key) {
10 if (node == null) return Node(key);
11 if (key < node.key) {
12 node.left = insert(node.left, key);
13 } else {
14 node.right = insert(node.right, key);
15 }
16 return node;
17}
18
19int height(Node? node) =>
20 node == null ? 0 : 1 + math.max(height(node.left), height(node.right));
21
22Node? remove(Node? node, int key) {
23 if (node == null) return null;
24 if (key < node.key) {
25 node.left = remove(node.left, key);
26 } else if (key > node.key) {
27 node.right = remove(node.right, key);
28 } else {
29 if (node.left == null) return node.right;
30 if (node.right == null) return node.left;
31 var successor = node.right!;
32 while (successor.left != null) {
33 successor = successor.left!;
34 }
35 node.key = successor.key;
36 node.right = remove(node.right, successor.key);
37 }
38 return node;
39}
