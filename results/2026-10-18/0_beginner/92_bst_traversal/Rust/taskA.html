1struct Node {
2 key: i32,
3 left: Option<usize>,
4 right: Option<usize>,
5}

7fn insert(nodes: &mut Vec<Node>, root: Option<usize>, key: i32) -> usize {
8 let Some(v) = root else {
9 nodes.push(Node {
10 key,
11 left: None,
12 right: None,
13 });
14 return nodes.len() - 1;
15 };
16 if key < nodes[v].key {
17 let left = nodes[v].left;
18 let child = insert(nodes, left, key);
19 nodes[v].left = Some(child);
20 } else {
21 let right = nodes[v].right;
22 let child = insert(nodes, right, key);
23 nodes[v].right = Some(child);
24 }
25 v
26}

28fn traverse(nodes: &[Node], root: Option<usize>, mode: usize) {
29 if let Some(v) = root {
30 if mode == 1 {
31 print!(" {}", nodes[v].key);
32 }
33 traverse(nodes, nodes[v].left, mode);
34 if mode == 0 {
35 print!(" {}", nodes[v].key);
36 }
37 traverse(nodes, nodes[v].right, mode);
38 if mode == 2 {
39 print!(" {}", nodes[v].key);
40 }
41 }
42}

44fn height(nodes: &[Node], root: Option<usize>) -> usize {
45 root.map_or(0, |v| {
46 1 + height(nodes, nodes[v].left).max(height(nodes, nodes[v].right))
47 })
48}

50fn erase(nodes: &mut [Node], root: Option<usize>, key: i32) -> Option<usize> {
51 let v = root?;
52 if key < nodes[v].key {
53 let left = nodes[v].left;
54 nodes[v].left = erase(nodes, left, key);
55 } else if key > nodes[v].key {
56 let right = nodes[v].right;
57 nodes[v].right = erase(nodes, right, key);
58 } else {
59 if nodes[v].left.is_none() {
60 return nodes[v].right;
61 }
62 if nodes[v].right.is_none() {
63 return nodes[v].left;
64 }
65 let mut successor = nodes[v].right.unwrap();
66 while let Some(left) = nodes[successor].left {
67 successor = left;
68 }
69 nodes[v].key = nodes[successor].key;
70 let right = nodes[v].right;
71 let successor_key = nodes[successor].key;
72 nodes[v].right = erase(nodes, right, successor_key);
73 }
74 Some(v)
75}
