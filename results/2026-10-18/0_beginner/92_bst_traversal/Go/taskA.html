5type Node struct {
6 key, left, right int
7}
8
9func insert(nodes *[]Node, root, key int) int {
10 if root < 0 {
11 *nodes = append(*nodes, Node{key, -1, -1})
12 return len(*nodes) - 1
13 }
14 if key < (*nodes)[root].key {
15 child := insert(nodes, (*nodes)[root].left, key)
16 (*nodes)[root].left = child
17 } else {
18 child := insert(nodes, (*nodes)[root].right, key)
19 (*nodes)[root].right = child
20 }
21 return root
22}
23
24func traverse(nodes []Node, root, mode int) {
25 if root < 0 {
26 return
27 }
28 if mode == 1 {
29 fmt.Printf(" %d", nodes[root].key)
30 }
31 traverse(nodes, nodes[root].left, mode)
32 if mode == 0 {
33 fmt.Printf(" %d", nodes[root].key)
34 }
35 traverse(nodes, nodes[root].right, mode)
36 if mode == 2 {
37 fmt.Printf(" %d", nodes[root].key)
38 }
39}

41func height(nodes []Node, root int) int {
42 if root < 0 {
43 return 0
44 }
45 return 1 + max(height(nodes, nodes[root].left), height(nodes, nodes[root].right))
46}
47
48func erase(nodes []Node, root, key int) int {
49 if root < 0 {
50 return root
51 }
52 if key < nodes[root].key {
53 nodes[root].left = erase(nodes, nodes[root].left, key)
54 } else if key > nodes[root].key {
55 nodes[root].right = erase(nodes, nodes[root].right, key)
56 } else {
57 if nodes[root].left < 0 {
58 return nodes[root].right
59 }
60 if nodes[root].right < 0 {
61 return nodes[root].left
62 }
63 successor := nodes[root].right
64 for nodes[successor].left >= 0 {
65 successor = nodes[successor].left
66 }
67 nodes[root].key = nodes[successor].key
68 nodes[root].right = erase(nodes, nodes[root].right, nodes[successor].key)
69 }
70 return root
71}
