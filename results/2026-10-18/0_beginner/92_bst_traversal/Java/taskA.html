1import java.util.ArrayDeque;
2import java.util.ArrayList;
3import java.util.Deque;
4import java.util.List;
5
6public class Main {
7 static class Node {
8 int key;
9 Node left, right;
10
11 Node(int key) {
12 this.key = key;
13 }
14 }
15
16 static Node insert(Node node, int key) {
17 if (node == null) {
18 return new Node(key);
19 }
20 if (key < node.key) {
21 node.left = insert(node.left, key);
22 } else {
23 node.right = insert(node.right, key);
24 }
25 return node;
26 }
27
28 static Node delete(Node node, int key) {
29 if (node == null) {
30 return null;
31 }
32 if (key < node.key) {
33 node.left = delete(node.left, key);
34 } else if (key > node.key) {
35 node.right = delete(node.right, key);
36 } else {
37 if (node.left == null) {
38 return node.right;
39 }
40 if (node.right == null) {
41 return node.left;
42 }
43 Node successor = node.right;
44 while (successor.left != null) {
45 successor = successor.left;
46 }
47 node.key = successor.key;
48 node.right = delete(node.right, successor.key);
49 }
50 return node;
51 }
52
53 static void traverse(Node node, int mode, List<Integer> order) {
54 if (node == null) {
55 return;
56 }
57 if (mode == 1) {
58 order.add(node.key);
59 }
60 traverse(node.left, mode, order);
61 if (mode == 0) {
62 order.add(node.key);
63 }
64 traverse(node.right, mode, order);
65 if (mode == 2) {
66 order.add(node.key);
67 }
68 }
69
70 static int height(Node node) {
71 return node == null ? 0 : 1 + Math.max(height(node.left), height(node.right));
72 }
73
74 static String joined(List<Integer> values) {
75 return String.join(" ", values.stream().map(String::valueOf).toList());
76 }
77
78 public static void main(String[] args) {
79 Node root = null;
80 for (int key : new int[]{50, 27, 74, 13, 39, 62, 88, 6, 31, 45, 70, 34}) {
81 root = insert(root, key);
82 }
83 String[] labels = {"inorder", "preorder", "postorder"};
84 for (int mode = 0; mode < 3; mode++) {
85 List<Integer> order = new ArrayList<>();
86 traverse(root, mode, order);
87 System.out.println(labels[mode] + ": " + joined(order));
88 }
89 Deque<Node> queue = new ArrayDeque<>();
90 List<Integer> level = new ArrayList<>();
91 queue.add(root);
92 while (!queue.isEmpty()) {
93 Node node = queue.remove();
94 level.add(node.key);
95 if (node.left != null) {
96 queue.add(node.left);
97 }
98 if (node.right != null) {
99 queue.add(node.right);
100 }
101 }
102 System.out.println("levelorder: " + joined(level));
103 System.out.println("height: " + height(root));
104 root = delete(root, 6);
105 root = delete(root, 27);
106 List<Integer> after = new ArrayList<>();
107 traverse(root, 1, after);
108 System.out.println("preorder after deletions: " + joined(after));
109 System.out.println("height after deletions: " + height(root));
110 }
