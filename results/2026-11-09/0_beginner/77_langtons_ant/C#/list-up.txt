bool[,] black = new bool[11, 11];
int r = 5, c = 5, direction = 0;
int[] dr = { -1, 0, 1, 0 }, dc = { 0, 1, 0, -1 };
for (int step = 0; step < 300; step++)
{
    direction = (direction + (black[r, c] ? 3 : 1)) % 4;
    black[r, c] = !black[r, c];
    r = (r + dr[direction] + 11) % 11;
    c = (c + dc[direction] + 11) % 11;
}
