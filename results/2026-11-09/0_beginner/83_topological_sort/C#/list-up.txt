    int[] degree = new int[n];
    bool[] taken = new bool[n];
    while (true)
    {
        int next = Enumerable.Range(0, n).FirstOrDefault(i => !taken[i] && degree[i] == 0, -1);
        if (next < 0) break;
        taken[next] = true;
        order.Add(names[next]);
        for (int i = 0; i < n; i++)
        {
            if (graph[next, i]) degree[i]--;
        }
    }

    {
        var remaining = names.Where((_, i) => !taken[i]);
