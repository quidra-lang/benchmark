    let values = [72, 85, 64, 90, 85, 58, 77, 85, 69, 93, 71];
    let mut sorted = values;
    sorted.sort();
    let (mut mode, mut best, mut count) = (sorted[0], 0, 0);
    for i in 0..sorted.len() {
        count = if i > 0 && sorted[i] == sorted[i - 1] {
            count + 1
        } else {
            1
        };
        if count > best {
            best = count;
            mode = sorted[i];
        }
    }
