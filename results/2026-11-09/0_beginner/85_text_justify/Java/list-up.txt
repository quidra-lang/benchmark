        int width = 30;
        List<List<String>> lines = new ArrayList<>();
        List<String> current = new ArrayList<>();
        int letters = 0;
        for (String word : paragraph.split(" ")) {
            if (!current.isEmpty() && letters + current.size() + word.length() > width) {
                lines.add(current);
                current = new ArrayList<>();
                letters = 0;
            }
            current.add(word);
            letters += word.length();
        }
        if (!current.isEmpty()) {
            lines.add(current);
        }
        for (int i = 0; i < lines.size(); i++) {
            List<String> words = lines.get(i);
            String text;
            if (i == lines.size() - 1 || words.size() == 1) {
                text = String.format(Locale.ROOT, "%-" + width + "s", String.join(" ", words));
            } else {
                int spaces = width - words.stream().mapToInt(String::length).sum();
                int gaps = words.size() - 1;
                StringBuilder line = new StringBuilder(words.get(0));
                for (int j = 1; j < words.size(); j++) {
                    line.append(" ".repeat(spaces / gaps + (j - 1 < spaces % gaps ? 1 : 0)));
                    line.append(words.get(j));
                }
                text = line.toString();
            }
            System.out.println("|" + text + "|");
        }
