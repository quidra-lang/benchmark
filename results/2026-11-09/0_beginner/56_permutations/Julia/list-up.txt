    characters = collect("ABCD")
    used = falses(4)
    counter = 0
    function visit(prefix)
        if length(prefix) == 4
            counter += 1
            println(counter, ": ", prefix)
            return
        end
        for i = 1:4
            if !used[i]
                used[i] = true
                visit(string(prefix, characters[i]))
                used[i] = false
            end
        end
    end
    visit("")
