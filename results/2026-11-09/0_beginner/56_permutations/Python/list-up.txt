letters = "ABCD"
used = [False] * len(letters)
count = 0
def visit(prefix):
    global count
    if len(prefix) == len(letters):
        count += 1
        print(f"{count}: {prefix}")
        return
    for i, letter in enumerate(letters):
        if not used[i]:
            used[i] = True
            visit(prefix + letter)
            used[i] = False
visit("")
