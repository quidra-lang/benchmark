function outcome(board)
    lines = [collect(row) for row in board]
    append!(lines, [[board[row][column] for row = 1:3] for column = 1:3])
    push!(lines, [board[i][i] for i = 1:3])
    push!(lines, [board[i][4-i] for i = 1:3])
    for line in lines
        if line[1] != '.' && all(==(line[1]), line)
            return string(line[1], " wins")
        end
    end
