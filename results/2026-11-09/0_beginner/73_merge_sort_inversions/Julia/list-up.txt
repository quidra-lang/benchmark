function merge_sort!(numbers, lo, hi, inversions = 0)
    lo == hi && return inversions
    mid = div(lo+hi, 2)
    inversions = merge_sort!(numbers, lo, mid, inversions)
    inversions = merge_sort!(numbers, mid+1, hi, inversions)
    left, right = numbers[lo:mid], numbers[(mid+1):hi]
    i, j = 1, 1
    while i <= length(left) && j <= length(right)
        if left[i] <= right[j]
            push!(merged, left[i])
            i += 1
        else
            push!(merged, right[j])
            j += 1
            inversions += length(left)-i+1
        end
    end
    println("merge [", lo-1, "..", hi-1, "]: ", join(merged, " "))
    return inversions
end
inversions = merge_sort!(numbers, 1, length(numbers))
