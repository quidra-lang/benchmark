    static int inversions = 0;
    static void mergeSort(int[] values, int l, int r) {
        int m = (l + r) / 2;
        mergeSort(values, l, m);
        mergeSort(values, m + 1, r);
        int i = l, j = m + 1, k = 0;
        while (i <= m && j <= r) {
            if (values[i] < values[j]) {
                merged[k++] = values[i++];
            } else {
                merged[k++] = values[j++];
                inversions += m - i + 1;
            }
        }
        System.out.println("merge [" + l + ".." + r + "]: "
                + String.join(" ", Arrays.stream(merged).mapToObj(String::valueOf).toList()));
