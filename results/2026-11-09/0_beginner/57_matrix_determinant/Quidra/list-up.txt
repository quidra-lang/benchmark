int determinant(int[][] matrix)
    n := #matrix
    if n == 1
        return matrix[0][0]
    total := 0
    for: j < n
        minor := [for row := matrix[1:][] then [for c := row.indices()[] if c != j then row[c]]]
        term := matrix[0][j] * determinant(minor)
        total += if j % 2 == 0 then term else -term
    return total
