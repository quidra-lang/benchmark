        let (mut current, mut best, mut start, mut left, mut right) = (a[0], a[0], 0, 0, 0);
        for i in 1..a.len() {
            if current < 0 {
                current = a[i];
                start = i;
            } else {
                current += a[i];
            }
            if current > best {
                best = current;
                left = start;
                right = i;
            }
        }
        println!("max sum {} at [{}, {}]", best, left, right);
