    current := values[0]
    best := current
    nat start := 0
    nat best_start := 0
    nat best_end := 0
    for: 1 <= i < #values
        if current < 0
            current = values[i]
            start = i
        else
            current += values[i]
        if current > best
            best = current
            best_start = start
            best_end = i
