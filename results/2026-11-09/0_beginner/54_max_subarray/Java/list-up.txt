            int current = values[0], best = current, start = 0, bestStart = 0, bestEnd = 0;
            for (int i = 1; i < values.length; i++) {
                if (current < 0) {
                    current = values[i];
                    start = i;
                } else {
                    current += values[i];
                }
                if (current > best) {
                    best = current;
                    bestStart = start;
                    bestEnd = i;
                }
            }
            System.out.println("max sum " + best + " at [" + bestStart + ", " + bestEnd + "]");
