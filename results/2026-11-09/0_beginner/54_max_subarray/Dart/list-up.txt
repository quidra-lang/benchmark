    var current = a[0], best = current, start = 0, bi = 0, bj = 0;
    for (var i = 1; i < a.length; i++) {
      if (current + a[i] < a[i]) {
        current = a[i];
        start = i;
      } else {
        current += a[i];
      }
      if (current > best) {
        best = current;
        bi = start;
        bj = i;
      }
    print('max sum $best at [$bi, $bj]');
