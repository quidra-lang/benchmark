    for i in 0..10 {
        for j in i + 1..10 {
            let dx = p[i][0] - p[j][0];
            let dy = p[i][1] - p[j][1];
            let d = dx * dx + dy * dy;
            for k in 0..2 {
                if (i == 0 && j == 1) || (if k == 0 { d < best[k] } else { d > best[k] }) {
                    best[k] = d;
                    bi[k] = i;
                    bj[k] = j;
                }
            }
    for k in 0..2 {
        println!(
            "{}: P{} P{} {:.4}",
            if k == 0 { "closest" } else { "farthest" },
            bi[k],
            bj[k],
            f64::from(best[k]).sqrt()
        );
