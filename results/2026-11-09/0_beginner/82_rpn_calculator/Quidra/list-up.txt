expressions := [
    "-17 5 / 4 -3 * -",
]
for expression := expressions[]
    int[] stack := []
    failure := ""
    tokens := expression.split(" ")
    for token := tokens[]
        if #token == 1 and "+-*/".contains(token)
            if #stack < 2
                failure = "stack underflow"
                break
            b := .stack.take()
            a := .stack.take()
            value := 0
            if token == "+"
                value = a + b
            elif token == "-"
                value = a - b
            elif token == "*"
                value = a * b
            elif b == 0
                failure = "division by zero"
                break
            else
                value = a / b
            .stack.push(value)
        else
            value := int(token)
            .stack.push(value)
    if failure == "" and #stack != 1
        failure = "leftover operands"
    if failure != ""
        println("{expression} => error: {failure}")
    else
        println("{expression} => {stack[0]}")
