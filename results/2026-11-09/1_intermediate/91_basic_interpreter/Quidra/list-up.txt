class Value(string type, number := 0, text := "")
class Token(string kind, string text, value := Value("S"))
class Expression(string kind, string type, text := "", number := 0, int[] args := [])
class PrintItem(string kind, expression := -1)
class Clause(line := -1, int[] body := [])
class Statement(string kind)
    line := 0
    next := -1
    PrintItem[] items := []
class Engine()
    source := map<int, string>()
    Expression[] expressions := []
    Statement[] statements := []
    random_state := 1
    buffer := ""
    string[] output := []
string variable_type(string name)
    return if name.ends_with("$") then "S" else "I"
bool contains_word(string[] options, string word)
    for option := options[]
        if option == word
            return true
    return false

bool valid_function(string word)
    return contains_word(["ABS", "SGN", "INT", "SQR", "RND", "LEN", "ASC", "VAL", "CHR$", "STR$", "LEFT$", "RIGHT$", "MID$", "INSTR"], word)

bool signature(string word, string types)
    if contains_word(["ABS", "SGN", "INT", "SQR", "RND", "CHR$", "STR$"], word)
        return types == "I"
int record_expression(Engine &engine, Expression expression)
    id := int(#engine.expressions)
    engine.expressions = engine.expressions.append(expression)
    return id

int record_statement(Engine &engine, Statement statement)
    id := int(#engine.statements)
    engine.statements = engine.statements.append(statement)
    return id

int make_integer(Engine &engine, int value)
    return record_expression(&engine, Expression("constant", "I", number = value))

int make_string(Engine &engine, string value)
    return record_expression(&engine, Expression("constant", "S", text = value))

int node(Engine &engine, string kind, string type, string text, int[] args)
    return record_expression(&engine, Expression(kind, type, text = text, args = args))
void | error tokenize(Engine &engine, string text)
    Token[] tokens := []
    p := 0
    while p < int(#text)
        c := text[p]
        if c == " "
            p += 1
            continue
        start := p
        if c == DQ
            p += 1
            while p < int(#text) and text[p] != DQ
                p += 1
            if p == int(#text)
                return error("SYNTAX ERROR")
            value := text[start + 1:<p]
            tokens = tokens.append(Token("S", text[start:<p + 1], Value("S", text = value)))
            p += 1
        elif digit(c)
            while p < int(#text) and digit(text[p])
                p += 1
            word := text[start:<p]
            value := int(word)
            if value > 32767
                return error("SYNTAX ERROR")
            tokens = tokens.append(Token("I", word, Value("I", number = value)))
        elif upper(c)
            while p < int(#text) and (upper(text[p]) or digit(text[p]))
                p += 1
            if p < int(#text) and text[p] == "$"
                p += 1
            word := text[start:<p]
            tokens = tokens.append(Token("W", word))
        else
            if not "+-*/^=<>() ,;:".contains(c)
                return error("SYNTAX ERROR")
            word := c
            if p + 1 < int(#text) and contains_word(["<>", "<=", ">="], text[p:<p + 2])
                word = text[p:<p + 2]
            tokens = tokens.append(Token("P", word))
            p += int(#word)
    engine.tokens = tokens.append(Token("E", ""))
    engine.position = 0
    return void

Token peek(Engine engine)
    return engine.tokens[engine.position]

bool at(Engine engine, string word)
    token := peek(engine)
    return (token.kind == "W" or token.kind == "P") and token.text == word

bool take(Engine &engine, string word)
    if not at(engine, word)
        return false
    engine.position += 1
    return true

void | error expect(Engine &engine, string word)
    if not take(&engine, word)
        return error("SYNTAX ERROR")
    return void
int | error integer(Engine &engine, level := 0)
    result := try parse_expression(&engine, level)
    if engine.expressions[result].type != "I"
        return error("SYNTAX ERROR")
    return result
int | error primary(Engine &engine)
    token := peek(engine)
    if token.kind == "I"
        engine.position += 1
        return make_integer(&engine, token.value.number)
    if token.kind == "S"
        engine.position += 1
        return make_string(&engine, token.value.text)
    if take(&engine, "(")
        value := try parse_expression(&engine)
        try expect(&engine, ")")
        return value
    if token.kind == "W" and valid_function(token.text)
        engine.position += 1
        try expect(&engine, "(")
        args := [try parse_expression(&engine)]
        while take(&engine, ",")
            args = args.append(try parse_expression(&engine))
        try expect(&engine, ")")
        types := ""
        for expression := args[]
            types += engine.expressions[expression].type
        if not signature(token.text, types)
            return error("SYNTAX ERROR")
        return node(&engine, "call", variable_type(token.text), token.text, args)
    return error("SYNTAX ERROR")

int | error parse_expression(Engine &engine, int level)
    if level == 2 and take(&engine, "NOT")
        return node(&engine, "unary", "I", "NOT", [try integer(&engine, 2)])
    if level == 6 and take(&engine, "-")
        return node(&engine, "unary", "I", "-", [try integer(&engine, 6)])
    if level == 8
        return primary(&engine)
    left := try parse_expression(&engine, level + 1)
    string[][] operators := [
        ["OR"], ["AND"], [], ["=", "<>", "<", ">", "<=", ">="],
        ["+", "-"], ["*", "/", "MOD"], [], ["^"]]
    while contains_word(operators[level], peek(engine).text)
        op := peek(engine).text
        engine.position += 1
        right := try parse_expression(&engine, if level == 7 then 6 else level + 1)
        t := engine.expressions[left].type
        if t != engine.expressions[right].type or (t == "S" and level != 3 and op != "+")
            return error("SYNTAX ERROR")
        left = node(&engine, "binary", if level == 3 then "I" else t, op, [left, right])
        if level == 7
            break
    return left
int | error parse_statement(Engine &engine, bool in_clause)
    word := peek(engine).text
    s := Statement(word)
    if take(&engine, "PRINT")
        last_value := false
        while peek(engine).kind != "E" and not at(engine, ":") and not at(engine, "ELSE")
            if take(&engine, ";")
                s.items = s.items.append(PrintItem(";"))
                last_value = false
            elif take(&engine, ",")
                s.items = s.items.append(PrintItem(","))
                last_value = false
            else
                kind := if at(engine, "TAB") then "TAB" elif at(engine, "SPC") then "SPC" else "value"
                if kind == "value" and last_value
                    return error("SYNTAX ERROR")
                expression := -1
                if kind == "value"
                    expression = try parse_expression(&engine)
                else
                s.items = s.items.append(PrintItem(kind, expression))
                last_value = kind == "value"
        return record_statement(&engine, s)
int[] | error parse_list(Engine &engine, bool in_clause)
    result := [try parse_statement(&engine, in_clause)]
    while take(&engine, ":")
        result = result.append(try parse_statement(&engine, in_clause))
    return result

int[] | error parse(Engine &engine, string source)
    try tokenize(&engine, source)
    body := try parse_list(&engine)
    if peek(engine).kind != "E"
        return error("SYNTAX ERROR")
    return body

void connect(Engine &engine, int[] body, int after, int line)
    following := after
    for: #body > i >= 0
        id := body[i]
        engine.statements[id].line = line
        engine.statements[id].next = following
        following = id
void end_line(Engine &engine)
    value := engine.buffer
    while #value > 0 and value[#value - 1] == " "
        value = value[0:<#value - 1]
    engine.output = engine.output.append(value)
    engine.buffer = ""

void message(Engine &engine, string text)
    if engine.buffer != ""
        end_line(&engine)
    engine.output = engine.output.append(text)

int | error checked(int n)
    if n < -32768 or n > 32767
        return error("OVERFLOW")
    return n
Value[] | error values_of(Engine &engine, int[] expressions)
    Value[] result := []
    for id := expressions[]
        result = result.append(try evaluate(&engine, id))
    return result

Value | error evaluate(Engine &engine, int id)
    e := engine.expressions[id]
    if e.kind == "constant"
        return if e.type == "I" then Value("I", number = e.number) else Value("S", text = e.text)
    if e.kind == "unary"
        a := (try evaluate(&engine, e.args[0])).number
        if e.text == "NOT"
            return Value("I", number = int(NOT int16(a)))
        return Value("I", number = try checked(-a))
    args := try values_of(&engine, e.args)
    if e.kind == "call"
        return apply_function(&engine, e.text, args)
    a := args[0]
    b := args[1]
    x := a.number
    y := b.number
    if e.text == "+"
        return Value("I", number = try checked(x + y))
    if e.text == "-"
        return Value("I", number = try checked(x - y))
    if e.text == "*"
        return Value("I", number = try checked(x * y))
    if e.text == "/" or e.text == "MOD"
        if y == 0
            return error("DIVISION BY ZERO")
        result := if e.text == "MOD" then x % y else try checked(x / y)
        return Value("I", number = result)
    if e.text == "AND"
        return Value("I", number = int(int16(x) AND int16(y)))
    return Value("I", number = int(int16(x) OR int16(y)))

Value | error apply_function(Engine &engine, string name, Value[] args)
    a := args[0]
    n := a.number
    s := a.text
    if name == "RND"
        if n < 1
            return error("ILLEGAL FUNCTION CALL")
        engine.random_state = engine.random_state * 48271 % 2147483647
        return Value("I", number = engine.random_state % n + 1)
int | error execute(Engine &engine, int id)
    s := engine.statements[id]
    next := s.next
    elif s.kind == "PRINT"
        for item := s.items[]
            if item.kind == ","
                engine.buffer += " " * nat(10 - #engine.buffer % 10)
            elif item.kind != ";"
                value := try evaluate(&engine, item.expression)
                if item.kind == "value"
                    engine.buffer += if value.type == "S" then value.text else string(value.number)
                else
        if #s.items == 0 or not contains_word([";", ","], s.items[-1].kind)
            end_line(&engine)
    return next
void run(Engine &engine)
    int[][] parsed := []
    int[] line_numbers := []
    for source_line := engine.source.keys().sorted()[]
        content := ""
        match engine.source.get(source_line)
            string found
                content = found
            none
                continue
        match parse(&engine, content)
            int[] statements
                parsed = parsed.append(statements)
                line_numbers = line_numbers.append(source_line)
            error problem
                message(&engine, "ERROR IN LINE {source_line}: SYNTAX ERROR")
                engine.output = engine.output.append("READY")
                return
    after := -1
    for: #parsed > i >= 0
        body := parsed[i]
        connect(&engine, body, after, line_numbers[i])
        .engine.entries.set(line_numbers[i], body[0])
        after = body[0]
    position := after
    while position >= 0
        match execute(&engine, position)
            int next
                position = next
            error problem
                if engine.handler != 0 and not engine.handling
                else
                    message(&engine, "ERROR IN LINE {engine.statements[position].line}: {problem.message}")
                    position = -1
    if engine.buffer != ""
        end_line(&engine)
    engine.output = engine.output.append("READY")
