function decimal(int $numerator, int $denominator, int $places): string
{
    $scale = 10 ** $places;
    $value = intdiv($numerator * $scale, $denominator);
    return intdiv($value, $scale) . '.' . str_pad((string)($value % $scale), $places, '0', STR_PAD_LEFT);
}
