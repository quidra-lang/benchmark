def decimal(numerator, denominator, places)
  scale = 10**places
  whole, fraction = ((numerator * scale) / denominator).divmod(scale)
  "#{whole}.#{fraction.to_s.rjust(places, '0')}"
end

def percentage(part, whole)
  decimal(part * 100, whole, 2) + '%'
