fixed <- function(numerator, denominator, places) {
    scale <- 10^places
    value <- (numerator * scale) %/% denominator
    paste0(sprintf("%.0f", value %/% scale), ".",
           sprintf(paste0("%0", places, "d"), as.integer(value %% scale)))
}
