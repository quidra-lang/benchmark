int child(const FileSystem source, int parent, string name)
    match get_node(source, parent).entries().get(name)
        int id
            return id
        none
            return -1
class Resolution(node := -1, parent := -1, name := "", problem := "")
Resolution walk(FileSystem &source, string path, int base, bool follow, bool original, bool parents, int &expansions)
    id := if path.starts_with("/") then 1 else base
    parts := [for part := path.split("/")[] if part != "" then part]
    for position := parts.indices()[]
        node := get_node(source, id)
        if not node.directory()
            return Resolution(problem = "Not a directory")
        name := parts[position]
        if name == "."
            continue
        if name == ".."
            id = node.parent
            continue
        last := position + 1 == #parts
        next := child(source, id, name)
        if next == -1
            if original and parents and not last
                next = create_node(&source, id, name, NodeData.Directory(map<string, int>()))
            else
                return Resolution(parent = if original and last then id else -1, name = name, problem = "No such file or directory")
        candidate := get_node(source, next)
        if candidate.tag() == "l" and (not last or follow)
            target := walk(&source, candidate.target(), id, true, false, false, &expansions)
            if target.node == -1
                return target
            next = target.node
        id = next
    return Resolution(node = id)
Resolution resolve(FileSystem &source, string path, follow := true, parents := false)
    return walk(&source, path, source.current, follow, true, parents, &expansions)
