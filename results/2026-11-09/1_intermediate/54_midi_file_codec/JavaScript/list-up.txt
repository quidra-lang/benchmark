const hex = (n, width = 2) => n.toString(16).toUpperCase().padStart(width, "0");
const be = (n, size) =>
    Array.from({ length: size }, (_, i) => Math.floor(n / 256 ** (size - 1 - i)) % 256);

const line = (e, track) => `${e.tick} ${track} ${e.type}${fields(e) ? " " + fields(e) : ""}`;
function encode(s) {
    function vlq(n) {
        const bytes = [n % 128];
        n = Math.floor(n / 128);
        while (n) {
            bytes.unshift((n % 128) + 128);
            n = Math.floor(n / 128);
        }
        return bytes;
    }
    const bytes = [77, 84, 104, 100, 0, 0, 0, 6].concat(
        be(s.format, 2),
        be(s.tracks.length, 2),
        be(s.division, 2),
    );
    for (const track of s.tracks) {
        let data = [],
            previousTick = 0,
            status = 0;

            data = data.concat(vlq(e.tick - previousTick));

const songs = new Map();
function command(words) {
    const [cmd, name, ...args] = words;
    const known = [
        "DECODE",
        "ENCODE",
        "TRANSPOSE",
    const fail = (reason) => {
        throw new Error(reason);
    };
    if (!known.includes(cmd)) fail("unknown command");
    const song = songs.get(name);
    if (!song) fail("no song " + name);
    const s = song,
        a = args.map(Number);
    const range = (i, lo, hi) => a[i] >= lo && a[i] <= hi;
    if (
        (cmd === "TRANSPOSE" &&
            (!range(0, -127, 127) || (args.length === 2 && !range(1, 1, 16)))) ||
    )
        fail("bad argument");
    switch (cmd) {
        case "DECODE":
            emit(`HEADER ${s.format} ${s.tracks.length} ${s.division}`);
            s.tracks.forEach((t, i) => t.forEach((e) => emit(line(e, i))));
            break;
        case "ENCODE": {
            const bytes = encode(s);
            for (let i = 0; i < bytes.length; i += 16)
                emit(
                    hex(i, 4) +
                        ": " +
                        bytes
                            .slice(i, i + 16)
                            .map((n) => hex(n))
                            .join(" "),
                );
            emit(bytes.length + " bytes");
            break;
        }
        case "TRANSPOSE": {
            let kept = 0;
            const dropped = [];
            s.tracks = s.tracks.map((track, t) =>
                track.filter((e) => {
                    if (
                        !["NOTE_ON", "NOTE_OFF", "POLY_PRESSURE"].includes(e.type) ||
                        (args.length === 2 && e.ch !== a[1])
                    )
                        return true;
                    const n = e.data[0] + a[0];
                    if (n < 0 || n > 127) {
                        dropped.push(
                            `dropped ${e.tick} ${t} ${e.type} ${e.ch} ${noteNames[e.data[0]]}`,
                        );
                        return false;
                    }
                    e.data[0] = n;
                    kept++;
                    return true;
                }),
            );
            emit(`transposed ${kept} dropped ${dropped.length}`);
            dropped.forEach(emit);
            break;
        }
