    static int Compare(int[] left, int[] right)
    {
        for (int i = 0; i < Math.Min(left.Length, right.Length); i++)
        {
            int order = left[i].CompareTo(right[i]);
            if (order != 0) return order;
        }
        return left.Length.CompareTo(right.Length);
    }
