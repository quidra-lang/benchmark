function rankkeys(keys)
    ordered=sort!(unique(keys))
    positions=Dict(key=>i-1 for (i, key) in enumerate(ordered))
    [positions[key] for key in keys]
end
degreecolors(g) = rankkeys(vec(sum(g.adj; dims = 2)))
function refine(g, colors)
    rounds=0
    while true
        signatures=[
            (colors[v], Tuple(sort!([colors[w] for w in 1:vertices(g) if g.adj[v, w]]))) for
            v in 1:vertices(g)
        ]
        next=rankkeys(signatures)
        maximum(next)==maximum(colors) && return colors, rounds
        colors=next
        rounds+=1
    end
end
stable(g) = refine(g, degreecolors(g))
