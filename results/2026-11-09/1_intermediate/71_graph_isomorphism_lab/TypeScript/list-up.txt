    adjacent(u: number, v: number): boolean { return (this.rows[u] & (1 << v)) !== 0; }
    neighbors(u: number): number[] { return Array.from({ length: this.n }, (_, v) => v).filter(v => this.adjacent(u, v)); }
function lex(a: number[], b: number[]): number {
    for (let i = 0; i < Math.min(a.length, b.length); i++) if (a[i] !== b[i]) return a[i] - b[i];
    return a.length - b.length;
}
function popcount(mask: number): number {
    let count = 0;
    while (mask) { mask &= mask - 1; count++; }
    return count;
}
function cells(colors: number[]): number[][] {
    const result = Array.from({ length: Math.max(...colors) + 1 }, () => [] as number[]);
    colors.forEach((color, vertex) => result[color].push(vertex));
    return result;
}
function refine(graph: Graph, initial: number[]): Refinement {
    let colors = [...initial], rounds = 0;
    while (true) {
        const signatures = colors.map((color, vertex) => [color, ...graph.neighbors(vertex).map(v => colors[v]).sort((a, b) => a - b)]);
        const order = Array.from({ length: graph.n }, (_, v) => v).sort((a, b) => lex(signatures[a], signatures[b]));
        const next = Array<number>(graph.n).fill(0);
        let color = 0;
        for (let i = 0; i < order.length; i++) {
            if (i && lex(signatures[order[i - 1]], signatures[order[i]]) !== 0) color++;
            next[order[i]] = color;
        }
        if (color + 1 === Math.max(...colors) + 1) return { colors, rounds };
        colors = next; rounds++;
    }
}
