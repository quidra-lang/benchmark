class Item(string key, int value)
class Slot(Item | none item := none, tombstone := false)
string content(const Slot slot)
    match slot.item
        Item item
            return item.key
        none
            return if slot.tombstone then "D" else "-"
string pair_text(const Item item)
    return "{item.key}={item.value}"
Mutation put(Table &table, string key, int value, string command, int maximum, tracing := false)
    found := search(table, key, tracing)
    match found.item
        Item previous
            updated := if command == "ADD" then previous.value + value else value
            replacement := Item(key, updated)
            if table.mode == Mode.Chain
                table.buckets[found.position][found.chain_index] = replacement
            else
                table.slots[found.position] = Slot(replacement)
            row := "{command} {key}={updated} UPDATED old={previous.value} at=" + position_text(table, found.position) + " probes={found.probes}"
            return Mutation(false, [], [row].concat(trace))
        none
            inserted := Item(key, value)
            placed := place(&table, inserted, found, tracing)
            row := "{command} {key}={value} NEW"
string[] get_item(Table &table, string key, bool tracing)
    found := search(table, key, tracing)
    string row
    match found.item
        Item item
            row = "GET " + pair_text(item) + " at=" + position_text(table, found.position) + " probes={found.probes}"
        none
            row = "GET {key} NOT FOUND probes={found.probes}"
    return [row].concat(trace_path(found, tracing))
string[] dump(const Table table)
    if table.mode == Mode.Chain
        for position := table.buckets.indices()[]
            string[] entries := []
            for item := table.buckets[position][]
                entries = entries.append(pair_text(item))
            rows = rows.append("{position} " + (if #entries == 0 then "-" else entries.join(" ")))
    else
        for position := table.slots.indices()[]
            prefix := string(position)
            if table.mode == Mode.Cuckoo
                prefix = "T{int(position) / table.cap + 1} {int(position) % table.cap}"
            slot := table.slots[position]
            string text
            match slot.item
                Item item
                    text = pair_text(item)
                    if table.mode == Mode.RobinHood
                        text += " d=" + string(distance(table, item.key, int(position)))
                none
                    text = content(slot)
            rows = rows.append(prefix + " " + text)
    return rows
string[] | error execute_table(Table &table, const Simulator settings, const string[] words)
    command := words[0]
    if command == "PUT" or command == "ADD"
        return put(&table, words[1], int(words[2]), command, settings.max_load, settings.tracing).rows
