function normalized(string $line): array
    return array_values(
        array_filter(
            explode(" ", strtolower($line)),
            fn(string $word): bool => $word !== "" && !in_array($word, ["the", "a", "an"], true),
        ),
    );
    case Use = "use";
    case Light = "light";
    case Score = "score";
    case Restore = "restore";
    public static function parse(string $word): ?self
        $canonical = match ($word) {
            "use", "apply" => "use",
            "light", "ignite" => "light",
            "score" => "score",
            default => null,
        return $canonical === null ? null : self::from($canonical);
    public function noObjects(): bool

    private function define(string $line): void
        $words = explode(" ", $line);
        $kind = $words[0];
        $id = $words[1];
        $text = fn(int $index): string => implode(" ", array_slice($words, $index));
        switch ($kind) {
            case "ROOM":
                $dark = $words[2] === "DARK";
                $this->rooms[$id] = new Room($text($dark ? 3 : 2), $dark);
                break;
            case "ITEM":
                $item = new Item($id, (int) $words[3]);
                $this->state->places[$id] = match ($words[2]) {
                    "PLAYER" => new Place(PlaceKind::Player),
                    "NOWHERE" => new Place(PlaceKind::Nowhere),
                    default => new Place(PlaceKind::Room, $words[2]),
                $this->state->fuel[$id] = 0;
                $this->state->lit[$id] = false;
                $this->state->containers[$id] = ContainerState::Open;
                $index = 4;
                if ($words[$index] === "TREASURE") {
                    $item->points = (int) $words[$index + 1];
                    $index += 2;
                if ($words[$index] === "LIGHT") {
                    $item->light = true;
                    $this->state->fuel[$id] = (int) $words[$index + 1];
                    $index += 2;
                $item->name = $text($index);
                $item->nameWords = array_fill_keys(array_slice($words, $index), true);
                $this->items[$id] = $item;
                break;
            case "RULE":
            case "ENTER":
                $index = $kind === "RULE" ? 3 : 2;
                $flag = null;
                $whenOn = true;
                if ($words[$index] === "IF" || $words[$index] === "UNLESS") {
                    $whenOn = $words[$index] === "IF";
                    $flag = $words[$index + 1];
                    $index += 2;
                $effectName = $words[$index++];
                $effect = match ($effectName) {
                    "MESSAGE" => new MessageEffect($text($index)),
                    "ADD" => new MoveEffect(
                        $words[$index],
                        new Place(PlaceKind::Room, $words[$index + 1]),
                    ),
                $rule = new Rule($id, $kind === "RULE" ? $words[2] : "", $flag, $whenOn, $effect);
                if ($kind === "RULE") {
                    $this->rules[] = $rule;
                } else {
                    $this->enterRules[] = $rule;
                break;
            case "TROPHY":
                $this->trophy = $id;
                break;
            case "START":
                $this->state->room = $id;
                break;
    private function children(): array
        $result = [];
        foreach ($this->ids as $id) {
            $result[$this->state->places[$id]->key()][] = $id;
        return $result;
    private function isOpen(string $id): bool
        return $this->items[$id]->capacity !== null &&
            $this->state->containers[$id] === ContainerState::Open;
    private function reachable(Place $place): array
        $children = $this->children();
        $result = [];
        $visit = function (Place $place) use ($children, &$result, &$visit): void {
            foreach ($children[$place->key()] ?? [] as $id) {
                $result[$id] = true;
                if ($this->isOpen($id)) {
                    $visit(new Place(PlaceKind::Inside, $id));
        $visit($place);
        return $result;
    private function carried(): array
        return $this->reachable(new Place(PlaceKind::Player));
    private function isCarried(string $id): bool
        $carried = $this->carried();
        return isset($carried[$id]);
    private function roomItems(): array
        return $this->reachable(new Place(PlaceKind::Room, $this->state->room));
    private function isLit(): bool
        if (!$this->rooms[$this->state->room]->dark) {
            return true;
        foreach ($this->carried() + $this->roomItems() as $id => $_) {
            if ($this->state->lit[$id]) {
                return true;
        return false;
    private function visible(): array
        return $this->isLit() ? $this->carried() + $this->roomItems() : $this->carried();
    private function score(): Score
        $stored = 0;
        $treasures = 0;
        $points = 0;
        foreach ($this->items as $id => $item) {
            if ($item->points > 0) {
                $treasures++;
                $place = $this->state->places[$id];
                while ($place->kind === PlaceKind::Inside) {
                    $place = $this->state->places[$place->id];
                if ($place->kind === PlaceKind::Room && $place->id === $this->trophy) {
                    $stored++;
                    $points += $item->points;
        return new Score($stored, $treasures, $points, $points + $this->state->bonus);
    private function resolve(array $phrase): string
        $visible = $this->visible();
        foreach ($phrase as $word) {
            if (!isset($this->vocabulary[$word])) {
                fail('I don\'t know the word "' . $word . '".');
        $matches = [];
        $last = $phrase[count($phrase) - 1];
        foreach ($this->items as $id => $item) {
            if (!isset($visible[$id]) || !isset($item->nouns[$last])) {
                continue;
            foreach (array_slice($phrase, 0, -1) as $word) {
                if (!isset($item->nameWords[$word])) {
                    continue 2;
            $matches[] = $id;
        if ($matches === []) {
            fail("You can't see any such thing.");
        if (count($matches) > 1) {
            fail(
                "Which do you mean: " .
                    implode(
                        ", ",
                        array_map(fn(string $id): string => $this->items[$id]->name, $matches),
                    ) .
                    "?",
            );
        return $this->pronoun = $matches[0];
    private function runRules(array $rules, string $first, string $second = ""): array
        $lines = [];
        $executed = 0;
        foreach ($rules as $rule) {
            if (
                $rule->first !== $first ||
                $rule->second !== $second ||
                ($rule->flag !== null && isset($this->state->flags[$rule->flag]) !== $rule->whenOn)
            ) {
                continue;
            $executed++;
            $effect = $rule->effect;
            if ($effect instanceof MessageEffect) {
                $lines[] = $effect->text;
            } elseif ($effect instanceof OpenEffect) {
            } elseif ($effect instanceof MoveEffect) {
                $this->state->places[$effect->item] = $effect->place;
        return [$lines, $executed];
    private function execute(Verb $verb, array $first, ?array $second): array
        switch ($verb) {
            case Verb::Score:
                $score = $this->score();
                return [
                    "You have scored {$score->total} points (treasure {$score->points}, bonus {$this->state->bonus}) in {$this->state->moves} moves.",
                    "Treasures stored: {$score->stored} of {$score->treasures}.",
                ];
            case Verb::Use:
                $id = $this->resolve($first);
                $target = $this->resolve($second);
                if (!$this->isCarried($id)) {
                    fail("You aren't carrying the " . $this->items[$id]->name . ".");
                [$lines, $executed] = $this->runRules($this->rules, $id, $target);
                if ($executed === 0) {
                    fail("Nothing happens.");
                return $lines === [] ? ["Done."] : $lines;
    private function endTurn(bool $baseline): array
        $this->state->moves++;
        $nearby = $this->carried() + $this->roomItems();
        $lines = [];
        foreach ($this->ids as $id) {
            if ($this->state->lit[$id]) {
                $this->state->fuel[$id]--;
                if ($this->state->fuel[$id] === 0) {
                    $this->state->lit[$id] = false;
                    if (isset($nearby[$id])) {
                        $lines[] = "The " . $this->items[$id]->name . " goes out.";
        $lit = $this->isLit();
        if ($baseline && !$lit) {
            $lines[] = "It is now pitch dark.";
        } elseif (!$baseline && $lit) {
            array_push($lines, ...$this->describe());
        $this->state->darkness = $lit ? 0 : $this->state->darkness + 1;
        if ($this->state->darkness === 2) {
            $lines[] = "You hear something moving in the dark.";
        if ($this->state->darkness === 3) {
            $this->state->ended = EndStatus::Dead;
        $score = $this->score();
        if (
            $this->state->ended === EndStatus::None &&
            $score->treasures > 0 &&
            $score->stored === $score->treasures
        ) {
            $this->state->ended = EndStatus::Won;
        if ($this->state->ended !== EndStatus::None) {
            $lines[] =
                $this->state->ended === EndStatus::Dead
                    ? "*** You have died ***"
                    : "*** You have won ***";
            $lines[] = "You scored {$score->total} points in {$this->state->moves} moves.";
        return $lines;
    private function interpret(string $line): array
        $words = normalized($line);
        if ($words === []) {
            fail("I beg your pardon?");
        $verb = Verb::parse($words[0]);
        if ($verb === null) {
            fail('I don\'t know the verb "' . $words[0] . '".');
        if ($this->state->ended !== EndStatus::None && $verb !== Verb::Restore) {
            fail("The game is over.");
        $rest = array_slice($words, 1);
        $first = $rest;
        $second = null;
        if ($verb->noObjects()) {
            if ($rest !== []) {
                fail("I don't understand that.");
        } else {
            if ($rest === []) {
                fail(
                    $verb === Verb::Go
                        ? "Which way do you want to go?"
                        : "What do you want to " . $verb->value . "?",
                );
            if ($verb === Verb::Go) {
            } else {
                $prepositions = [];
                if ($verb === Verb::Put) {
                } elseif ($verb === Verb::Use) {
                    $prepositions = ["on", "with"];
                if ($prepositions !== []) {
                    $index = 0;
                    while ($index < count($rest) && !in_array($rest[$index], $prepositions, true)) {
                        $index++;
                    if ($index === 0 || $index + 1 >= count($rest)) {
                        fail("I don't understand that.");
                    $first = array_slice($rest, 0, $index);
                    $second = array_slice($rest, $index + 1);
        $baseline = $this->isLit();
        if ($verb === Verb::Go) {
        } else {
            $lines = $this->execute($verb, $first, $second);
        if ($verb->action()) {
            array_push($lines, ...$this->endTurn($baseline));
        return $lines;
