int position_of<T>(const T[] values, T value)
    for index := values.indices()[]
        if values[index] == value
            return int(index)
    return -1
string[] directions()
    return ["north", "south", "east", "west", "up", "down"]
int direction_index(string word)
    names := directions()
    for index := names.indices()[]
        if word == names[index] or word == names[index][0]
            return int(index)
    return -1
string[] command_words(string line)
    lowered := ""
    int[] points := line.codepoints()
    for index := points.indices()[]
        point := points[index]
        lowered += if 65 <= point <= 90 then "abcdefghijklmnopqrstuvwxyz"[point - 65] else line[index]
    return [for word := lowered.split(" ")[] if word != "" and word != "the" and word != "a" and word != "an" then word]

Rule read_rule(const string[] words, int start, Trigger trigger, const string[] room_ids, const string[] item_ids)
    at := start
    Condition | none condition := none
    if words[at] == "IF" or words[at] == "UNLESS"
        condition = Condition(words[at + 1], words[at] == "IF")
        at += 2
    MessageEffect | OpenEffect | MoveEffect | FlagEffect | PointsEffect effect
    if words[at] == "MESSAGE"
        effect = MessageEffect(words[at + 1:].join(" "))
    elif words[at] == "ADD"
        effect = MoveEffect(position_of(item_ids, words[at + 1]), Place(PlaceKind.Floor, position_of(room_ids, words[at + 2])))
    return Rule(trigger, condition, effect)
Adventure read_world(const string[][] definitions)
    room_ids := [for words := definitions[] if words[0] == "ROOM" then words[1]]
    item_ids := [for words := definitions[] if words[0] == "ITEM" then words[1]].sorted()
    rooms := array(#room_ids, fill = Room())
    items := array(#item_ids, fill = Item())
    for index := item_ids.indices()[]
        items[index].nouns = [item_ids[index]]
    state := State(0, array(#item_ids, fill = Place(PlaceKind.Nowhere)), array(#item_ids, fill = ContainerState.Open), array(#item_ids, fill = false), array(#item_ids, fill = 0), array(#room_ids, fill = array(6, fill = Exit())), array(#room_ids, fill = false))
    Rule[] rules := []
    trophy := 0
    max_weight := 0
    for words := definitions[]
        command := words[0]
        if command == "ROOM"
            index := position_of(room_ids, words[1])
            dark := words[2] == "DARK"
            start := if dark then 3 else 2
            rooms[index].dark = dark
            rooms[index].name = words[start:].join(" ")
        elif command == "ITEM"
            index := position_of(item_ids, words[1])
            if words[2] == "PLAYER"
                state.places[index] = Place(PlaceKind.Player)
            elif words[2] == "NOWHERE"
                state.places[index] = Place(PlaceKind.Nowhere)
            else
                state.places[index] = Place(PlaceKind.Floor, position_of(room_ids, words[2]))
            item := items[index]
            item.weight = int(words[3])
            at := 4
            if words[at] == "TREASURE"
                item.treasure = int(words[at + 1])
                at += 2
            if words[at] == "LIGHT"
                item.light = true
                state.fuel[index] = int(words[at + 1])
                at += 2
            item.name = words[at:].join(" ")
            item.name_words = words[at:]
            items[index] = item
        elif command == "RULE"
            rules = rules.append(read_rule(words, 3, Trigger(TriggerKind.Use, position_of(item_ids, words[1]), position_of(item_ids, words[2])), room_ids, item_ids))
        elif command == "TROPHY"
            trophy = position_of(room_ids, words[1])
        elif command == "START"
            state.room = position_of(room_ids, words[1])
    distinct<string> vocabulary
    for item := items[]
        for word := item.nouns.concat(item.name_words)[]
            .vocabulary.add(word)
    return Adventure(World(rooms, items, rules, vocabulary, trophy, max_weight), state)
void say(Adventure &game, string text)
    game.output = game.output.append(text)
Place room_floor(const Adventure game)
    return Place(PlaceKind.Floor, game.state.room)
bool reachable(const Adventure game, int item, Place root)
    place := game.state.places[item]
    while place.kind == PlaceKind.Inside
        if game.state.containers[place.index] != ContainerState.Open
            return false
        place = game.state.places[place.index]
    return place == root
bool carried(const Adventure game, int item)
    return reachable(game, item, Place(PlaceKind.Player))
bool room_item(const Adventure game, int item)
    return reachable(game, item, room_floor(game))
bool is_lit(const Adventure game)
    if not game.world.rooms[game.state.room].dark
        return true
    for index := game.world.items.indices()[]
        if game.state.lit[index] and (carried(game, int(index)) or room_item(game, int(index)))
            return true
    return false
bool stored(const Adventure game, int item)
    place := game.state.places[item]
    while place.kind == PlaceKind.Inside
        place = game.state.places[place.index]
    return place == Place(PlaceKind.Floor, game.world.trophy)
(int, int, int) treasures(const Adventure game)
    all := [for index := game.world.items.indices()[] if game.world.items[index].treasure > 0 then int(index)]
    present := [for item := all[] if stored(game, item) then item]
    return ([for item := present[] then game.world.items[item].treasure].sum(), int(#present), int(#all))
int score_value(const Adventure game)
    points, _, _ := treasures(game)
    return points + game.state.bonus
int | error resolve(Adventure &game, const string[] phrase)
    lit := is_lit(game)
    for word := phrase[]
        if not game.world.vocabulary.has(word)
            return error("I don't know the word {DQ}{word}{DQ}.")
    int[] matches := []
    for index := game.world.items.indices()[]
        item := int(index)
        if not carried(game, item) and not (lit and room_item(game, item))
            continue
        definition := game.world.items[item]
        if position_of(definition.nouns, phrase[-1]) < 0
            continue
        matches_all := true
        for adjective := phrase[0:<int(#phrase) - 1][]
            if position_of(definition.name_words, adjective) < 0
                matches_all = false
        if matches_all
            matches = matches.append(item)
    if #matches == 0
        return error("You can't see any such thing.")
    if #matches > 1
        names := [for item := matches[] then game.world.items[item].name].join(", ")
        return error("Which do you mean: " + names + "?")
    game.pronoun = matches[0]
    return matches[0]
bool condition_holds(const State state, Condition | none condition)
    match condition
        Condition value
            return state.flags.has(value.flag) == value.expected
        none
            return true
(int, bool) run_rules(Adventure &game, Trigger trigger)
    executed := 0
    message := false
    for rule := game.world.rules[]
        if rule.trigger != trigger or not condition_holds(game.state, rule.condition)
            continue
        executed += 1
        match rule.effect
            MessageEffect effect
                say(&game, effect.text)
                message = true
            MoveEffect effect
                game.state.places[effect.item] = effect.place
    return (executed, message)
void pass_turn(Adventure &game, bool baseline)
    game.state.moves += 1
    nearby := [for index := game.world.items.indices()[] then carried(game, int(index)) or room_item(game, int(index))]
    for index := game.world.items.indices()[]
        if game.state.lit[index]
            game.state.fuel[index] -= 1
            if game.state.fuel[index] == 0
                game.state.lit[index] = false
                if nearby[index]
                    say(&game, "The {game.world.items[index].name} goes out.")
    lit := is_lit(game)
    if baseline and not lit
        say(&game, "It is now pitch dark.")
    elif not baseline and lit
        describe(&game, true)
    game.state.darkness = if lit then 0 else game.state.darkness + 1
    if game.state.darkness == 2
        say(&game, "You hear something moving in the dark.")
    if game.state.darkness == 3
        game.state.ended = EndStatus.Dead
    points, count, total := treasures(game)
    if game.state.ended == EndStatus.Playing and total > 0 and count == total
        game.state.ended = EndStatus.Won
    if game.state.ended != EndStatus.Playing
        say(&game, if game.state.ended == EndStatus.Dead then "*** You have died ***" else "*** You have won ***")
        say(&game, "You scored {points + game.state.bonus} points in {game.state.moves} moves.")

Verb[] verb_list()
    return [Verb.Go, Verb.Look, Verb.Examine, Verb.Take, Verb.Drop, Verb.Put, Verb.Inventory, Verb.Open, Verb.Close, Verb.Unlock, Verb.Lock, Verb.Use, Verb.Light, Verb.Extinguish, Verb.Wait, Verb.Score, Verb.Save, Verb.Restore, Verb.Verbose, Verb.Brief]
string verb_name(Verb kind)
    names := "go look examine take drop put inventory open close unlock lock use light extinguish wait score save restore verbose brief".split(" ")
    return names[position_of(verb_list(), kind)]
Verb | none parse_verb(string word)
    groups := ["go walk run", "look l", "examine x inspect read", "take get grab", "drop discard", "put insert place", "inventory inv i", "open", "close shut", "unlock", "lock", "use apply", "light ignite", "extinguish douse snuff", "wait z", "score", "save", "restore load", "verbose", "brief"]
    for index := groups.indices()[]
        if position_of(groups[index].split(" "), word) >= 0
            return verb_list()[index]
    return none
bool no_arguments(Verb kind)
    return position_of([Verb.Look, Verb.Inventory, Verb.Wait, Verb.Score, Verb.Save, Verb.Restore, Verb.Verbose, Verb.Brief], kind) >= 0
bool time_action(Verb kind)
    return position_of([Verb.Go, Verb.Take, Verb.Drop, Verb.Put, Verb.Open, Verb.Close, Verb.Unlock, Verb.Lock, Verb.Use, Verb.Light, Verb.Extinguish, Verb.Wait], kind) >= 0
bool | error perform(Adventure &game, Verb kind, const string[] first, const string[] second, int direction, bool &baseline)
    match kind
        Verb.Use
            a := try resolve(&game, first)
            b := try resolve(&game, second)
            if not carried(game, a)
                return error("You aren't carrying the {game.world.items[a].name}.")
            executed, message := run_rules(&game, Trigger(TriggerKind.Use, a, b))
            if executed == 0
                return error("Nothing happens.")
            if not message
                say(&game, "Done.")
        Verb.Score
            points, count, total := treasures(game)
            say(&game, "You have scored {points + game.state.bonus} points (treasure {points}, bonus {game.state.bonus}) in {game.state.moves} moves.")
            say(&game, "Treasures stored: {count} of {total}.")
    return true
void | error interpret(Adventure &game, string line)
    words := command_words(line)
    if #words == 0
        return error("I beg your pardon?")
    if direction_index(words[0]) >= 0
        words = ["go"].concat(words)
    Verb kind
    match parse_verb(words[0])
        Verb value
            kind = value
        none
            return error("I don't know the verb {DQ}{words[0]}{DQ}.")
    if game.state.ended != EndStatus.Playing and kind != Verb.Restore
        return error("The game is over.")
    first := words[1:]
    string[] second := []
    direction := -1
    if no_arguments(kind)
        if #first > 0
            return error("I don't understand that.")
    else
        if #first == 0
            return error(if kind == Verb.Go then "Which way do you want to go?" else "What do you want to " + verb_name(kind) + "?")
        if kind == Verb.Go
        else
            string[] prepositions := []
            if kind == Verb.Put
            elif kind == Verb.Use
                prepositions = ["on", "with"]
            split := -1
            for index := first.indices()[]
                if position_of(prepositions, first[index]) >= 0
                    split = int(index)
                    break
            if split >= 0
                if split == 0 or split + 1 == int(#first)
                    return error("I don't understand that.")
                second = first[split + 1:]
                first = first[0:<split]
            elif kind == Verb.Put or kind == Verb.Use
                return error("I don't understand that.")
    baseline := is_lit(game)
    succeeded := try perform(&game, kind, first, second, direction, &baseline)
    if succeeded and time_action(kind)
        pass_turn(&game, baseline)
    return void
void | error process_line(Adventure &game, string line)
    words := command_words(line)
    if #words > 0 and (words[0] == "again" or words[0] == "g")
    else
        game.previous = line
        try interpret(&game, line)
    return void
