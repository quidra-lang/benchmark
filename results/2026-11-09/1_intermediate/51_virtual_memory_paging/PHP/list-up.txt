const POLICIES = [
    "FIFO",
    "LRU",
    "LFU",
    "NRU",
    "AGING",
    "CLOCK",
    "ECLOCK",
    "WSCLOCK",
    "RANDOM",
    "OPT",
];
const DEFAULTS = [
    "page" => 256,
    "frames" => 4,
    "tlb" => 4,
    "sets" => 1,
    "policy" => "FIFO",
    "scope" => "GLOBAL",
    "asid" => "OFF",
    "tick" => 0,
    "tau" => 8,
    "seed" => 1,
];

function settings(array $w): array
{
    $c = DEFAULTS;
    $seen = [];
    foreach (array_slice($w, 1) as $pair) {
        [$key, $value] = explode("=", $pair);
        if (!array_key_exists($key, $c)) {
            fail("unknown key $key");
        }
        if (isset($seen[$key])) {
            fail("duplicate key $key");
        }
        $seen[$key] = true;
        $n = (int) $value;
        $ranges = [
            "frames" => [1, 16],
            "tlb" => [1, 8],
            "sets" => [1, 8],
            "tick" => [0, 1000],
            "tau" => [1, 1000],
            "seed" => [1, 2147483646],
        ];
        if ($key === "policy") {
            $valid = in_array($value, POLICIES, true);
        } elseif ($key === "scope") {
            $valid = in_array($value, ["GLOBAL", "LOCAL"], true);
        } elseif ($key === "asid") {
            $valid = in_array($value, ["ON", "OFF"], true);
        } elseif ($key === "page") {
            $valid = $n >= 16 && $n <= 4096 && ($n & ($n - 1)) === 0;
        } else {
            $valid = $n >= $ranges[$key][0] && $n <= $ranges[$key][1];
        }
        if (!$valid) {
            fail("invalid $key");
        }
        $c[$key] = in_array($key, ["policy", "scope", "asid"], true) ? $value : $n;
    }
    if ($c["tlb"] % $c["sets"] !== 0) {
        fail("invalid sets");
    }
    return $c;
}
