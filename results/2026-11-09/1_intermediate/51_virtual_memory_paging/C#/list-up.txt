    static Config Settings(string[] tokens) {
        Config c = new();
        HashSet<string> seen = new();
        string[] keys = { "page",  "frames", "tlb",  "sets", "policy",
                          "scope", "asid",   "tick", "tau",  "seed" };
        foreach (string token in tokens.Skip(1)) {
            string[] pair = token.Split('=');
            string key = pair[0], value = pair[1];
            if (!keys.Contains(key))
                throw new CommandError("unknown key " + key);
            if (!seen.Add(key))
                throw new CommandError("duplicate key " + key);
            bool valid;
            if (key == "policy") {
                valid = Policies.Contains(value);
                if (valid)
                    c.Policy = value;
            } else if (key == "scope") {
                valid = value is "GLOBAL" or "LOCAL";
                if (valid)
                    c.Scope = value;
            } else if (key == "asid") {
                valid = value is "ON" or "OFF";
                if (valid)
                    c.Asid = value;
            } else {
                long n = long.Parse(value, CultureInfo.InvariantCulture);
                valid = key switch { "page" => n >= 16 && n <= 4096 && (n & (n - 1)) == 0,
                                     "frames" => n >= 1 && n <= 16,
                                     "tlb" or "sets" => n >= 1 && n <= 8,
                                     "tick" => n >= 0 && n <= 1000,
                                     "tau" => n >= 1 && n <= 1000,
                                     _ => n >= 1 && n <= 2147483646 };
                if (valid)
                    switch (key) {
                    case "page":
                        c.Page = (int)n;
                        break;
                    case "frames":
                        c.Frames = (int)n;
                        break;
                    case "tlb":
                        c.Tlb = (int)n;
                        break;
                    case "sets":
                        c.Sets = (int)n;
                        break;
                    case "tick":
                        c.Tick = (int)n;
                        break;
                    case "tau":
                        c.Tau = (int)n;
                        break;
                    case "seed":
                        c.Seed = n;
                        break;
                    }
            }
            if (!valid)
                throw new CommandError("invalid " + key);
        }
        if (c.Tlb % c.Sets != 0)
            throw new CommandError("invalid sets");
        return c;
    }
