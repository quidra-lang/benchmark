final class Table
{
    public array $rows = [];
    public function __construct(
        public string $name,
        public array $columns,
        public ?int $primary = null,
    ) {}
    public function column(string $name): ?int
    {
        foreach ($this->columns as $i => $c) {
            if ($c["name"] === $name) {
                return $i;
            }
        }
        return null;
    }
}
final class Scope
{
    public array $entries = [];
    public int $width = 0;
    public function add(string $name, Table $table): void
    {
        foreach ($this->entries as $entry) {
            if ($entry["name"] === $name) {
                fail("duplicate table name " . $name);
            }
        }
        $this->entries[] = ["name" => $name, "table" => $table, "offset" => $this->width];
        $this->width += count($table->columns);
    }
    public function resolve(Expr $e): string
    {
        $found = [];
        $seen = false;
        foreach ($this->entries as $entry) {
            if ($e->table === null || $e->table === $entry["name"]) {
                $seen = true;
                $i = $entry["table"]->column($e->column);
                if ($i !== null) {
                    $found[] = [$entry["offset"] + $i, $entry["table"]->columns[$i]["type"]];
                }
            }
        }
        if ($e->table !== null && !$seen) {
            fail("unknown table " . $e->table);
        }
        if (!$found) {
            fail("unknown column " . $e->text);
        }
        if (count($found) > 1) {
            fail("ambiguous column " . $e->column);
        }
        $e->index = $found[0][0];
        return $found[0][1];
    }
}
function compatible(string $a, string $b): bool
{
    return $a === $b || $a === "NULL" || $b === "NULL";
}
function requireType(string $a, string $b, string $op): void
{
    if (!compatible($a, $b)) {
        fail("type mismatch for " . $op);
    }
}
function check(
    Expr $e,
    Scope $scope,
    bool $allow = false,
    ?array $grouped = null,
    bool $inside = false,
): string {
    if ($e->kind === "literal") {
        return $e->value === null ? "NULL" : (is_int($e->value) ? "INT" : "TEXT");
    }
    if ($e->kind === "ref") {
        $t = $scope->resolve($e);
        if ($grouped !== null && !$inside && !in_array($e->index, $grouped, true)) {
            fail("column " . $e->text . " is not grouped");
        }
        return $t;
    }
    if ($e->kind === "aggregate") {
        if (!$allow || $inside) {
            fail("misuse of aggregate " . $e->op);
        }
function condition(
    ?Expr $e,
    Scope $s,
    string $name,
    bool $allow = false,
    ?array $grouped = null,
): void {
    if ($e !== null) {
        requireType(check($e, $s, $allow, $grouped), "INT", $name);
    }
}
function truth(int|string|null $v): bool
{
    return $v !== null && $v !== 0;
}
function evaluate(Expr $e, array $row, ?array $group = null): int|string|null
{
    if ($e->kind === "literal") {
        return $e->value;
    }
    if ($e->kind === "ref") {
        return $row[$e->index];
    }
function satisfies(?Expr $e, array $row, ?array $group = null): bool
{
    return $e === null || truth(evaluate($e, $row, $group));
}
function rowCount(int $n): string
{
    return $n . ($n === 1 ? " row" : " rows");
}
function written(int|string|null $v): string
{
    return $v === null ? "NULL" : (string) $v;
}
