    sealed record Entry(string Name, Table Table, int Offset);
    sealed class Scope
    {
        public readonly List<Entry>Entries = new();
        int width;
        public void Add(string n, Table t) {
            if (Entries.Any(e => e.Name == n)) throw Error("duplicate table name "+n);
            Entries.Add(new(n, t, width));
            width += t.Columns.Count;
        }
        public string Resolve(Expr e) {
            List<(int Index, string Type)>found = new();
            bool seen = false;
            foreach (Entry p in Entries) if (e.Table == null || e.Table == p.Name) {
                seen = true;
                int i = p.Table.Column(e.Column);
                if (i >= 0) found.Add((p.Offset+i, p.Table.Columns[i].Type));
            }
            if (e.Table != null && !seen) throw Error("unknown table "+e.Table);
            if (found.Count == 0) throw Error("unknown column "+e.Text);
            if (found.Count>1) throw Error("ambiguous column "+e.Column);
            e.Index = found[0].Index;
            return found[0].Type;
        }
    }
