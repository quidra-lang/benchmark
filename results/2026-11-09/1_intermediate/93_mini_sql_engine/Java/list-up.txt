    static class Scope {
        final List<Entry>entries = new ArrayList<>();
        int width;
        void add(String n, Table t) {
            if (entries.stream().anyMatch(e -> e.name.equals(n))) throw error("duplicate table name "+n);
            entries.add(new Entry(n, t, width));
            width += t.columns.size();
        }
        String resolve(Expr ref) {
            List<Map.Entry<Integer, String>>found = new ArrayList<>();
            boolean seen = false;
            for (Entry e:entries) if (ref.table == null || e.name.equals(ref.table)) {
                seen = true;
                int i = e.table.column(ref.column);
                if (i >= 0) found.add(Map.entry(e.offset+i, e.table.columns.get(i).type));
            }
            if (ref.table != null && !seen) throw error("unknown table "+ref.table);
            if (found.isEmpty()) throw error("unknown column "+ref.text);
            if (found.size()>1) throw error("ambiguous column "+ref.column);
            ref.index = found.get(0).getKey();
            return found.get(0).getValue();
        }
    }
