function resolve(e: Expr, scope: ScopePart[]): number {
    if (e.table !== null) {
        const part = scope.find((p) => p.name === e.table);
        if (!part) fail("unknown table " + e.table);
        const index = part.table.columns.findIndex((c) => c.name === e.column);
        if (index < 0) fail("unknown column " + e.text);
        return (e.index = part.offset + index);
    }
    const matches = scope.flatMap((p) =>
        p.table.columns.flatMap((c, i) => (c.name === e.column ? [p.offset + i] : [])),
    );
    if (matches.length === 0) fail("unknown column " + e.column);
    if (matches.length > 1) fail("ambiguous column " + e.column);
    return (e.index = matches[0]);
}
function check(
    e: Expr,
    scope: ScopePart[],
    allow: boolean,
    grouping: Set<number> | null,
    inside = false,
): SqlType {
    if (e.kind === "literal")
        return e.value === null ? "NULL" : typeof e.value === "number" ? "INT" : "TEXT";
    if (e.kind === "reference") {
        const index = resolve(e, scope);

                return part.table.columns[index - part.offset].type;
