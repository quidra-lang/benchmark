static int resting(int i) {

static int best(int side) {
    int b = -1;
    for (int i = 0; i < no; i++)
        if (resting(i) && orders[i].side == side &&
            (b < 0 ||
             (side ? orders[i].price < orders[b].price : orders[i].price > orders[b].price) ||
             (orders[i].price == orders[b].price && orders[i].priority < orders[b].priority)))
            b = i;
    return b;
static int matchable(int side, I price, I other) {
    return price < 0 || (side ? other >= price : other <= price);

static void execute(int a) {
    Order *x = &orders[a];
    while (x->remaining) {
        int b = best(1 - x->side);
        if (b < 0 || !matchable(x->side, x->price, orders[b].price))
            break;
        Order *y = &orders[b];
        if (x->trader == y->trader) {
            printf("STP %s %lld\n", y->id, y->remaining);
            y->remaining = y->visible = 0;
            y->status = 4;
            continue;
        }
        I q = min(x->remaining, y->visible), p = y->price;
        int bi = x->side ? b : a, si = x->side ? a : b;
        I bf = q * (bi == a ? taker : maker), sf = q * (si == a ? taker : maker);
        trades[nx++] = (Trade){p, q, bf, sf, bi, si, a};
        x->remaining -= q;
        y->remaining -= q;
        y->visible -= q;
        Trader *bt = &traders[orders[bi].trader], *st = &traders[orders[si].trader];
        bt->fees += bf;
        st->fees += sf;
        trade_line(nx - 1);
        if (!y->remaining)
            y->status = 3;
        else {
            y->status = 2;

                y->priority = ++priority;
