void fail(string reason)
    emit("ERROR " + reason)
    if action == "CANCEL" or action == "RETURN" or action == "ORDER"
        order := integer_value(words[1]) - 1
        if order < 0 or order >= state.order_used
            fail("unknown order " + words[1])
            return
        entry := state.orders[order]
        if action == "CANCEL"
            if entry.backorder == 0
                fail("no backorder " + words[1])
            else
                state.orders[order].cancelled += entry.backorder
