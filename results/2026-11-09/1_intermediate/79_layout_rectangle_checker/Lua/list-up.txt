local function keys(t)
    local list = {}
    for key in pairs(t) do
        list[#list + 1] = key
    end
    table.sort(list)
    return list
end
local function compare(a, b)
    local r = {
        x1 = math.max(a.x1, b.x1),
        y1 = math.max(a.y1, b.y1),
        x2 = math.min(a.x2, b.x2),
        y2 = math.min(a.y2, b.y2)
    }
    r.ox, r.oy = r.x2 - r.x1, r.y2 - r.y1
    r.gap = math.max(0, -r.ox, -r.oy)
    if r.ox > 0 and r.oy > 0 then
        r.kind = "overlap"
    elseif r.ox == 0 and r.oy == 0 then
        r.kind = "corner"
    elseif r.ox >= 0 and r.oy >= 0 then
        r.kind = "edge"
    else
        r.kind = "separate"
        r.direction = r.ox < 0 and r.oy < 0 and "diagonal" or r.ox < 0 and "x" or "y"
    end
    return r
end
local function spacing(list, s)
    local result = {}
    for i, a in ipairs(list) do
        for j = i + 1, #list do
            local b, r = list[j], compare(a, list[j])
            if r.kind == "corner" or r.kind == "separate" and r.gap < s then
                result[#result + 1] = {
                    a = a.name,
                    b = b.name,
                    gap = r.gap,
                    direction = r.kind == "corner" and "corner" or r.direction
                }
            end
        end
    end
    table.sort(
        result,
        function(a, b)
            return a.gap < b.gap or a.gap == b.gap and (a.a < b.a or a.a == b.a and a.b < b.b)
        end
    )
    return result
end
local C = {rectangles = {}, rules = {}, history = {}, output = {}}
function C:emit(format, ...)
    self.output[#self.output + 1] = string.format(format, ...)
end
function C:list(layer)
    local result = {}
    for _, name in ipairs(keys(self.rectangles)) do
        local r = self.rectangles[name]
        if not layer or r.layer == layer then
            result[#result + 1] = r
        end
    end
    return result
end
function C:require(layer)
    local list = self:list(layer)
    if #list == 0 then
        self:fail("no rectangles on %s", layer)
    end
    return list
end
function C:violationLines(list)
    for _, v in ipairs(list) do
        self:emit("%s %s gap %d %s", v.a, v.b, v.gap, v.direction)
    end
end
function C:check()
    if not next(self.rules) then
        self:fail("no rules")
    end
    local rules = {}
    for _, r in pairs(self.rules) do
        rules[#rules + 1] = r
    end
    table.sort(
        rules,
        function(a, b)
            return a.kind < b.kind or
                a.kind == b.kind and (a.inner < b.inner or a.inner == b.inner and a.outer < b.outer)
        end
    )
    local total = 0
    for _, r in ipairs(rules) do
        local list, lines = self:list(r.inner), {}
        if r.kind == "spacing" then
            for _, v in ipairs(spacing(list, r.value)) do
                lines[#lines + 1] = string.format("%s %s gap %d %s", v.a, v.b, v.gap, v.direction)
            end
        if r.kind == "enclose" then
            self:emit("enclose %s %s %d: %d violations", r.inner, r.outer, r.value, #lines)
        else
            self:emit("%s %s %d: %d violations", r.kind, r.inner, r.value, #lines)
        end
        for _, line in ipairs(lines) do
            self:emit("%s", line)
        end
        total = total + #lines
    end
    self:emit("check: %d violations in %d rules", total, #rules)
function C:execute(line)
    local args = words(line)
    local command = table.remove(args, 1)
    elseif command == "RULE" then
        local r = {kind = args[1]:lower(), inner = args[2], outer = ""}
        if r.kind == "enclose" then
        else
            r.value = tonumber(args[3])
        end
        self.rules[r.kind .. ":" .. r.inner .. ":" .. r.outer] = r
        return
    elseif command == "CHECK" then
        self:check()
        return
    end
    local layer = command == "COVER" and args[2] or args[1]
    local list = self:require(layer)
    elseif command == "DRC" then
        local s = tonumber(args[2])
        local violations = spacing(list, s)
        self:emit("drc %s spacing %d: %d violations", layer, s, #violations)
        self:violationLines(violations)
