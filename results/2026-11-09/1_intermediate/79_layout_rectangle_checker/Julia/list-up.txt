struct Rectangle
    name::String
    layer::String
    box::NTuple{4,Int}
end
rectwidth(r) = r.box[3]-r.box[1]
rectheight(r) = r.box[4]-r.box[2]
function relation(a, b)
    x=max(a.box[1], b.box[1])
    y=max(a.box[2], b.box[2])
    u=min(a.box[3], b.box[3])
    v=min(a.box[4], b.box[4])
    ox=u-x
    oy=v-y
    kind=ox>0 && oy>0 ? 0 : ox<0 || oy<0 ? 3 : ox==0 && oy==0 ? 2 : 1
    (
        kind = kind,
        box = (x, y, u, v),
        ox = ox,
        oy = oy,
        gap = max(0, -ox, -oy),
        direction = ox<0 && oy<0 ? "diagonal" : ox<0 ? "x" : "y",
    )
end
function spacinglines(rects, s)
    found=Tuple{Int,String,String,String}[]
    for i in 1:(length(rects)-1), j in (i+1):length(rects)
        a=rects[i]
        b=rects[j]
        r=relation(a, b)
        r.kind==2 && push!(found, (0, a.name, b.name, "corner"))
        r.kind==3 && r.gap<s && push!(found, (r.gap, a.name, b.name, r.direction))
    end
    ["$a $b gap $gap $direction" for (gap, a, b, direction) in sort!(found)]
end
    rectangles::Dict{String,Rectangle}
    rules::Dict{NTuple{3,String},Int}
    history::Vector{Tuple{Dict{String,Rectangle},String}}
sortedrects(w) = sort!(collect(values(w.rectangles)); by = r->r.name)
onlayer(w, layer) = [r for r in sortedrects(w) if r.layer==layer]
function requirelayer(w, layer)
    result=onlayer(w, layer)
    check(!isempty(result), "no rectangles on $layer")
    result
end
function report(heading, lines)
    println(heading)
    foreach(println, lines)
end
function ruleviolations(w, kind, layer, other, value)
    kind=="spacing" && return spacinglines(onlayer(w, layer), value)
function execute!(w, line)
    parts=String.(split(line, ' '))
    command=parts[1]
    args=parts[2:end]
    elseif command=="RULE"
        kind=lowercase(args[1])
        layer=args[2]
        other=kind=="enclose" ? args[3] : ""
        value=parse(Int, args[end])
        check(kind!="enclose" || layer!=other, "same layer $layer")
        w.rules[(kind, layer, other)]=value
        println("rule ", join([kind; args[2:end]], ' '))
    elseif command=="CHECK"
        check(!isempty(w.rules), "no rules")
        total=0
        for (kind, layer, other) in sort!(collect(keys(w.rules)))
            value=w.rules[(kind, layer, other)]
            lines=ruleviolations(w, kind, layer, other, value)
            heading=kind=="enclose" ? "$kind $layer $other $value" : "$kind $layer $value"
            report("$heading: $(length(lines)) violations", lines)
            total+=length(lines)
        end
        println("check: $total violations in $(length(w.rules)) rules")
    else
        minimumcount=command=="COVER" ? parse(Int, args[1]) : 1
        layer=args[command=="COVER" ? 2 : 1]
        rects=requirelayer(w, layer)
        if command=="DRC"
            s=parse(Int, args[2])
            lines=spacinglines(rects, s)
            report("drc $layer spacing $s: $(length(lines)) violations", lines)
    command in ("ADD", "COPY", "MOVE", "RESIZE", "SPLIT", "LAYER", "DELETE") &&
        push!(w.history, (before, line))
