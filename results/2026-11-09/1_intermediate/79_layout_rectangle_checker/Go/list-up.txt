type geometry struct{ x1, y1, x2, y2 int64 }
type rectangle struct {
	name, layer string
	geometry
}
type relation struct {
	geometry
	ox, oy, gap     int64
	kind, direction string
}
type rule struct {
	kind, inner, outer string
	value              int64
}
func number(s string) int64 { n, _ := strconv.ParseInt(s, 10, 64); return n }
func min(a, b int64) int64 {
	if a < b {
		return a
	}
	return b
}
func max(a, b int64) int64 {
	if a > b {
		return a
	}
	return b
}
func compare(a, b geometry) relation {
	r := relation{geometry: geometry{max(a.x1, b.x1), max(a.y1, b.y1), min(a.x2, b.x2), min(a.y2, b.y2)}}
	r.ox, r.oy = r.x2-r.x1, r.y2-r.y1
	r.gap = max(0, max(-r.ox, -r.oy))
	if r.ox > 0 && r.oy > 0 {
		r.kind = "overlap"
	} else if r.ox == 0 && r.oy == 0 {
		r.kind = "corner"
	} else if r.ox >= 0 && r.oy >= 0 {
		r.kind = "edge"
	} else {
		r.kind = "separate"
		if r.ox < 0 && r.oy < 0 {
			r.direction = "diagonal"
		} else if r.ox < 0 {
			r.direction = "x"
		} else {
			r.direction = "y"
		}
	}
	return r
}
func (c *checker) emit(format string, args ...any) {
	c.output = append(c.output, fmt.Sprintf(format, args...))
}
func (c *checker) list(layer string) []rectangle {
	result := []rectangle{}
	for _, r := range c.rectangles {
		if layer == "" || r.layer == layer {
			result = append(result, r)
		}
	}
	sort.Slice(result, func(i, j int) bool { return result[i].name < result[j].name })
	return result
}
func (c *checker) require(layer string) []rectangle {
	list := c.list(layer)
	if len(list) == 0 {
		fail("no rectangles on %s", layer)
	}
	return list
}
func spacing(list []rectangle, s int64) []violation {
	result := []violation{}
	for i, a := range list {
		for _, b := range list[i+1:] {
			r := compare(a.geometry, b.geometry)
			if r.kind == "corner" || r.kind == "separate" && r.gap < s {
				direction := r.direction
				if r.kind == "corner" {
					direction = "corner"
				}
				result = append(result, violation{a.name, b.name, r.gap, direction})
			}
		}
	}
	sort.Slice(result, func(i, j int) bool {
		a, b := result[i], result[j]
		return a.gap < b.gap || a.gap == b.gap && (a.a < b.a || a.a == b.a && a.b < b.b)
	})
	return result
}

		c.emit("%s %s gap %d %s", v.a, v.b, v.gap, v.direction)

func (c *checker) check() {
	if len(c.rules) == 0 {
		fail("no rules")
	}
	rules := []rule{}
	for _, r := range c.rules {
		rules = append(rules, r)
	}
	sort.Slice(rules, func(i, j int) bool {
		a, b := rules[i], rules[j]
		return a.kind < b.kind || a.kind == b.kind && (a.inner < b.inner || a.inner == b.inner && a.outer < b.outer)
	})
	total := 0
	for _, r := range rules {
		list := c.list(r.inner)
		lines := []string{}
		switch r.kind {
		case "spacing":
			for _, v := range spacing(list, r.value) {
				lines = append(lines, fmt.Sprintf("%s %s gap %d %s", v.a, v.b, v.gap, v.direction))
			}
		if r.kind == "enclose" {
			c.emit("enclose %s %s %d: %d violations", r.inner, r.outer, r.value, len(lines))
		} else {
			c.emit("%s %s %d: %d violations", r.kind, r.inner, r.value, len(lines))
		}
		for _, line := range lines {
			c.emit("%s", line)
		}
		total += len(lines)
	}
	c.emit("check: %d violations in %d rules", total, len(rules))
}
