def value_key(value):
    return (isinstance(value, str), value)


def row_key(row):
    return tuple(value_key(value) for value in row)


def value_notation(value):
    if isinstance(value, int):
        return str(value)
    if PLAIN_SYMBOL.fullmatch(value) and value not in RESERVED:
        return value
    return f'"{value}"'
