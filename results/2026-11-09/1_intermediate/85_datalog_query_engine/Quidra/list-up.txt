bool text_before(string first, string second)
    int[] a := first.codepoints()
    int[] b := second.codepoints()
    for: i < #a
        if i == #b
            return false
        if a[i] != b[i]
            return a[i] < b[i]
    return #a < #b
bool bare_symbol(string text)
    if text == "" or not "abcdefghijklmnopqrstuvwxyz".contains(text[0]) or contains_text(reserved_words(), text)
        return false
    for: i < #text
        if not word_character(text[i])
            return false
    return true

class Datum(symbol := false, number := 0, text := "")
    string notation()
        return if not symbol then string(number) else (if bare_symbol(text) then text else DQ + text + DQ)

    string encoded()
        return if symbol then "s{#text}:" + text else "i{number};"

    bool before(Datum other)
        if symbol != other.symbol
            return not symbol
        return if symbol then text_before(text, other.text) else number < other.number

class Row(Datum[] values, nat round_number := 0, nat rule_number := 0)
    string encoded()
        result := ""
        for value := values[]
            result += value.encoded()
        return result
    string[] variables(nat[] term_ids)
        string[] result := []
        for node_id := term_ids[]
            if nodes[node_id].kind == 2
                result = result.append(nodes[node_id].text)
        return result

    string[] body_variables(Literal[] body)
        string[] result := []
        for item := body[]
            for name := variables(literal_terms(item))[]
                if not contains_text(result, name)
                    result = result.append(name)
        return result
bool row_before(Row a, Row b)
    for: i < #a.values
        if a.values[i] != b.values[i]
            return a.values[i].before(b.values[i])
    return false

Row[] sorted_rows(Row[] items)
    if #items < 2
        return items
    middle := #items / 2
    a := sorted_rows(items[:<middle])
    b := sorted_rows(items[middle:])
    Row[] result := []
    nat i := 0
    nat j := 0
    while i < #a and j < #b
        if row_before(b[j], a[i])
            result = result.append(b[j])
            j += 1
        else
            result = result.append(a[i])
            i += 1
    return result.concat(a[i:]).concat(b[j:])
string answer_text(string[] names, Row row)
    string[] parts := []
    for: i < #names
        parts = parts.append(names[i] + " = " + row.values[i].notation())
    return parts.join(", ")
class Predicate(string name, nat arity)
    Row[] tuples := []
    membership := map<string, bool>()

    bool has_row(Row row)
        match membership.get(row.encoded())
            bool present
                return present
            none
                return false

    void .insert(Row row)
        if not has_row(row)
            tuples = tuples.append(row)
            .membership.set(row.encoded(), true)
class Answers(string[] names, Row[] rows)
Answers find_answers(Syntax syntax, Literal[] body, Model model)
    names := syntax.body_variables(body)
    unique_rows := Predicate("", #names)
    for binding := solve_body(syntax, body, model.relations, model.relations)[]
        Datum[] values := []
        for name := names[]
            match binding.assignments.get(name)
                Datum value
                    values = values.append(value)
                none
                    void
        .unique_rows.insert(Row(values))
    return Answers(names, sorted_rows(unique_rows.tuples))
    string[] | error .query(Statement statement)
        try check_known(body_atoms(statement.body))
        string[] first_variables := if statement.variable == "" then [] else [statement.variable]
        try check_safety(statement.syntax, Literal(), statement.body, first_variables)
        answers := find_answers(statement.syntax, statement.body, .get_model())
        if statement.aggregate == "count"
            return [string(#answers.rows)]
        if statement.aggregate == ""
            if #answers.names == 0
                return [if #answers.rows == 0 then "false." else "true."]
            output := [for row := answers.rows[] then answer_text(answers.names, row)]
            return output.append(counted(#answers.rows, "answer", "answers"))
