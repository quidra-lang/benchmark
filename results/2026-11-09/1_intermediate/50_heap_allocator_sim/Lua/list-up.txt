function Heap:required(n)
    local s = ((n + 7) // 8) * 8 + 8
    if self.mode == "BUDDY" then
        local p = 32
        while p < s do
            p = p * 2
        end
        return p
    end
    return s
end
function Heap:allocate(s, al)
    local choices = {}
    local start = 1
    for i, b in ipairs(self.blocks) do
        if b.a <= self.rover and self.rover < b.a + b.size then
            start = i
        end
        if not b.used then
            local p = ((b.a + 8 + al - 1) // al) * al - 8
            if p - b.a > 0 and p - b.a < 16 then
                p = p + al
            end
            if p + s <= b.a + b.size then
                choices[#choices + 1] = {b = b, p = p, i = i}
            end
        end
    end
    if #choices == 0 then
        return nil
    end
    local mode = self.mode
    table.sort(
        choices,
        function(x, y)
            local a, b = x.b, y.b
            if mode == "NEXT" then
                return (x.i - start + #self.blocks) % #self.blocks <
                    (y.i - start + #self.blocks) % #self.blocks
            elseif mode == "BEST" or mode == "BUDDY" then
                if a.size ~= b.size then
                    return a.size < b.size
                end
            elseif mode == "WORST" then
                if a.size ~= b.size then
                    return a.size > b.size
                end
            elseif mode == "SEGFIT" then
                if class(a.size) ~= class(b.size) then
                    return class(a.size) < class(b.size)
                end
                return a.age > b.age
            end
            return a.a < b.a
        end
    )
    local c = choices[1]
    local b, p = c.b, c.p
    table.remove(self.blocks, c.i)
    local used
    if mode == "BUDDY" then
        local z = b.size
        while z > s do
            z = z // 2
            self.blocks[#self.blocks + 1] = self:freeblock(b.a + z, z)
        end
        used = {a = b.a, size = s, used = true}
    else
        if p > b.a then
            self.blocks[#self.blocks + 1] = self:freeblock(b.a, p - b.a)
        end
        local z = b.a + b.size - p
        used = {a = p, size = (z - s >= 16 and s or z), used = true}
        if z - s >= 16 then
            self.blocks[#self.blocks + 1] = self:freeblock(p + s, z - s)
        end
    end
    self.blocks[#self.blocks + 1] = used
    self:sort()
    if mode == "NEXT" then
        self.rover = (used.a + used.size) % self.size
    end
    return used
    local b = self:allocate(self:required(n), al)
