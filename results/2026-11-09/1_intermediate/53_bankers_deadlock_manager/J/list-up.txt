        waits =. i.0
            for_e. History do.
                if. (>kind)eq 1{::>e do.
                    n =. >:n
                    if. 'unblocked'eq >kind do. waits =. waits,4{::>e end.
                end.
            end.
        if. #waits do.
            emit 'waits: ',(num #waits),' unblocked, total ',(num +/waits),', average ',((+/waits)fixed (#waits),2)
        else. emit 'waits: none' end.
