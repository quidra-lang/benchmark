        for (Event e : state.history)
            if (e.kind.equals("unblocked")) {
                waits++;
                totalWait += e.waited;
            }
        if (waits == 0)
            emit("waits: none");

            int average = totalWait * 100 / waits;
            emit("waits: " + waits + " unblocked, total " + totalWait + ", average " +
                 average / 100 + "." + String.format(Locale.ROOT, "%02d", average % 100));
