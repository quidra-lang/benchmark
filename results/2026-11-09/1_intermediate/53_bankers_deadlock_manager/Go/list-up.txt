		counts := map[string]int{}
		waits := 0
		for _, e := range m.History {
			counts[e.Kind]++
			if e.Kind == "unblocked" {
				waits += e.Wait
			}
		}
		if counts["unblocked"] == 0 {
			emit("waits: none")
		} else {
			avg := waits * 100 / counts["unblocked"]
			emit(fmt.Sprintf("waits: %d unblocked, total %d, average %d.%02d", counts["unblocked"], waits, avg/100, avg%100))
