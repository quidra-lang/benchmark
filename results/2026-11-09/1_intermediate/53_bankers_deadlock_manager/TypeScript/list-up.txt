            const waits = state.history.filter((e) => e.kind === "unblocked"),
                total = waits.reduce((s, e) => s + e.wait, 0);
            const average = waits.length ? Math.floor((total * 100) / waits.length) : 0;
            emit(
                waits.length
                    ? `waits: ${waits.length} unblocked, total ${total}, average ${Math.floor(average / 100)}.${String(average % 100).padStart(2, "0")}`
                    : "waits: none",
            );
