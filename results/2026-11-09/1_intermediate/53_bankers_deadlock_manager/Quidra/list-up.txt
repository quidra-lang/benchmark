            kinds := ["granted", "blocked", "unblocked", "released", "finished", "aborted", "cancelled"]
            counts := array(7, fill = 0)
            waited_sum := 0
            for event := data.events[]
                for: j < #kinds
                    if event.kind == kinds[j]
                        counts[j] += 1
                if event.kind == "unblocked"
                    waited_sum += event.waited
            if counts[2] == 0
                output = output.append("waits: none")
            else
                average := waited_sum * 100 / counts[2]
                fraction := padded(string(average % 100), 2, true).split(" ").join("0")
                output = output.append("waits: {counts[2]} unblocked, total {waited_sum}, average {average / 100}.{fraction}")
