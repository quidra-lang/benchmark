        string[] es=c.Keys.OrderBy(e=>e,StringComparer.Ordinal).ToArray();

            "C"
        }.Concat(c.ContainsKey("H")?new[]{
            "H"
        }:Array.Empty<string>()).Concat(es.Where(e=>e!="C"&&e!="H")).ToArray():es;
