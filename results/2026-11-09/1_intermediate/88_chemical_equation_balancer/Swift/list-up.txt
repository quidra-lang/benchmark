func order(_ c: Composition) -> [String] {
    let es = c.keys.sorted()
    return c["C"] == nil ? es : ["C"] + (c["H"] == nil ? [] : ["H"]) + es.filter {
        $0 != "C" && $0 != "H"
    }
func hill(_ c: Composition, _ q: Int = 0) -> String {
    order(c).map {
        $0 + (c[$0] == 1 ? "" : String(c[$0]!))
    }.joined() + (q == 0 ? "" : "^" + (abs(q) == 1 ? "" : String(abs(q))) + (q > 0 ? "+" : "-"))

            throw ChemError.message("syntax")
        }
        return Formula(c: c, q: q)
