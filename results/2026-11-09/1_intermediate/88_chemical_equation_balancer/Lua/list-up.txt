local function order(c)
    local es={};
    for e in pairs(c) do
        es[#es+1]=e
    end;
    table.sort(es);
    if not c.C then
        return es
    end;
    local out={'C'};
    if c.H then
        out[#out+1]='H'
    end;
    for _,e in ipairs(es) do
        if e~='C' and e~='H' then
            out[#out+1]=e
        end
    end;
    return out
