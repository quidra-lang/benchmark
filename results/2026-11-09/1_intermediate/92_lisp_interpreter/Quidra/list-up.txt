class Frame(int parent)
    variables := map<string, int>()
int store(Engine &engine, Value item)
    id := int(#engine.values)
    engine.values = engine.values.append(item)
    return id
int extend_frame(Engine &engine, int parent)
    id := int(#engine.frames)
    engine.frames = engine.frames.append(Frame(parent))
    return id
void bind(Engine &engine, int frame, string name, int id)
    .engine.frames[frame].variables.set(name, id)
int | error find_frame(const Engine engine, int frame, string name)
    current := frame
    while current >= 0
        match engine.frames[current].variables.get(name)
            int stored
                return current
            none
                current = engine.frames[current].parent
int | error lookup(const Engine engine, int frame, string name)
    found := try find_frame(engine, frame, name)
    match engine.frames[found].variables.get(name)
        int stored
            if stored == 2
                return error("uninitialized variable: {name}")
            return stored
int make_closure(Engine &engine, Parameters formals, int[] expressions, int env_id, name := "")
    item := Value("procedure", text = name)
    item.body = expressions
    item.env_id = env_id
    return store(&engine, item)
int | error evaluate_body(Engine &engine, int[] expressions, int env_id)
    result := 1
    for expression := expressions[]
        result = try evaluate(&engine, expression, env_id)
    return result
