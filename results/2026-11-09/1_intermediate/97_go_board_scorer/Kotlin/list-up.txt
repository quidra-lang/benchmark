data class Reading(val success:Boolean,val move:Int=-1)
fun ladder(b:String,n:Int,p:Int):LadderResult {
    val defender=b[p];
    val attacker=other(defender);
    fun read(current:String,depth:Int,defending:Boolean):Reading {
        if(depth>=40)return Reading(defending);
        val g=group(current,n,p);
        val candidates=g.liberties.toMutableSet();
        val seen=mutableSetOf<Int>()
        if(defending)for(q in g.stones)for(t in adjacent(q,n))if(current[t]==attacker&&t !in seen){
            val e=group(current,n,t);
            seen.addAll(e.stones);
            if(e.liberties.size==1)candidates.addAll(e.liberties)
        }
        for(q in candidates.sorted()){
            val next=placement(current,n,if(defending)defender else attacker,q);
            if(next.reason.isNotEmpty())continue;
            val success=if(defending){
                val count=group(next.board,n,p).liberties.size;
                count>=3||(count==2&&!read(next.board,depth+1,false).success)
            }else !read(next.board,depth+1,true).success;
            if(success){
                val r=Reading(true,q);
                return r
            }
        }
        val r=Reading(false);
        return r
