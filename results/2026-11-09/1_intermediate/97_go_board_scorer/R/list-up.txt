    if (depth >= 40) {
      return(list(success = defending, move = 0L))
    }
    g <- group(current, n, p)
    candidates <- g$liberties
    seen <- rep(FALSE, length(current))
    if (defending) {
      for (q in g$stones) {
        for (t in adjacent(q, n)) {
          if (current[t] == attacker && !seen[t]) {
