function ladder(b, n, p)
    defender=b[p]
    attacker=other(defender)
    function read(current, depth, defending)
        depth>=40 && return (defending, 0)
        g=group(current, n, p)
        candidates=copy(g.liberties)
        seen=Set{Int}()
        if defending
            for q in g.stones, t in adjacent(q, n)
                if current[t]==attacker && !(t in seen)
                    enemy=group(current, n, t)
                    union!(seen, enemy.stones)
                    length(enemy.liberties)==1 && union!(candidates, enemy.liberties)
                end
            end
        end
        for q in sort!(collect(candidates))
            next, _, reason=placement(current, n, defending ? defender : attacker, q)
            !isempty(reason) && continue
            if defending
                m=length(group(next, n, p).liberties)
                success=m>=3 || (m==2 && !read(next, depth+1, false)[1])
            else
                success=!read(next, depth+1, true)[1]
            end
            if success
                memo[key]=(true, q)
                return memo[key]
            end
        end
        memo[key]=(false, 0)
