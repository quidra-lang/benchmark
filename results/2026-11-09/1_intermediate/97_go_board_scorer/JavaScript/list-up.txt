const sorted = (s) => [...s].sort((a, b) => a - b);
function ladder(b, n, p) {
    const defender = b[p],
        attacker = other(defender),
    function read(current, depth, defending) {
        if (depth >= 40)
            return {
                success: defending,
                move: -1,
            };
        const g = group(current, n, p),
            candidates = new Set(g.liberties),
            seen = new Set();
        if (defending)
            for (const q of g.stones)
                for (const t of adjacent(q, n))
                    if (current[t] === attacker && !seen.has(t)) {
                        const e = group(current, n, t);
                        for (const s of e.stones) seen.add(s);
                        if (e.liberties.size === 1) for (const s of e.liberties) candidates.add(s);
                    }
        for (const q of sorted(candidates)) {
            const next = placement(current, n, defending ? defender : attacker, q);
            if (next.reason) continue;
            let success;
            if (defending) {
                const m = group(next.board, n, p).liberties.size;
                success = m >= 3 || (m === 2 && !read(next.board, depth + 1, false).success);
            } else success = !read(next.board, depth + 1, true).success;
            if (success) {
                const result = {
                    success: true,
                    move: q,
                };
                return result;
            }
        }
        const result = {
            success: false,
            move: -1,
        };
        return result;
