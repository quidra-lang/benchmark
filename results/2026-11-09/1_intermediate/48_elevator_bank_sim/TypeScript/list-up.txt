function configuration(lines: string[][]): Config {
    const entries = new Map<string, string[]>();
    for (const [key, ...values] of lines) {
        ensure(!keys.includes(key), `unknown key ${key}`);
        ensure(entries.has(key), `duplicate key ${key}`);
        entries.set(key, values);
    }
    for (const key of keys.slice(0, 5)) ensure(!entries.has(key), `missing key ${key}`);
    const c: Config = {
        floors: 0,
        cars: 0,
        capacity: 0,
        door: 0,
        starts: [],
        policy: "COST",
        zones: [],
        park: -1,
        homes: [],
        patience: -1,
        bypass: 100,
        period: -1,
        limit: 1000,
        compare: [],
    for (const key of keys) {
        const a = entries.get(key);
        if (!a) continue;
        const n = key === "START" || key === "ZONES" ? c.cars : key === "PARK" ? c.cars + 1 : 1;
        ensure(
            key === "COMPARE" ? a.length < 1 || a.length > 4 : a.length !== n,
            `wrong number of values for ${key}`,
        );
        const bounded = (text: string, lo: number, hi: number) => {
            const n = Number(text);
            ensure(n < lo || n > hi, `bad value for ${key}`);
            return n;
        };
        switch (key) {
            case "FLOORS":
                c.floors = bounded(a[0], 2, 50);
                break;
            case "CARS":
                c.cars = bounded(a[0], 1, 8);
                break;
            case "CAPACITY":
                c.capacity = bounded(a[0], 1, 20);
                break;
            case "DOOR":
                c.door = bounded(a[0], 1, 9);
                break;
            case "START":
                c.starts = a.map((x) => bounded(x, 1, c.floors));
                break;
            case "POLICY":
                ensure(!policies.includes(a[0]), "bad value for POLICY");
                c.policy = a[0];
                break;
            case "PATIENCE":
                c.patience = bounded(a[0], 1, 1000);
                break;
        }
    }
    if (!c.zones.length) c.zones = c.starts.map(() => [1, c.floors]);
    for (let i = 0; i < c.cars; ++i) {
        const [lo, hi] = c.zones[i];
        ensure(c.starts[i] < lo || c.starts[i] > hi, `car ${i + 1} starts outside its zone`);
        ensure(
            c.park >= 0 && (c.homes[i] < lo || c.homes[i] > hi),
            `car ${i + 1} home outside its zone`,
        );
    }
    return c;
const serves = (c: Config, i: number, a: number, b: number) =>
    c.zones[i][0] <= Math.min(a, b) && Math.max(a, b) <= c.zones[i][1];
function events(
    c: Config,
    lines: {
        number: number;
        words: string[];
    }[],
    output: string[],
) {
    const accepted: Event[] = [];
    let last = 0,
        groups = 0,
        fire = false;
    for (const line of lines) {
        const [time, command, ...args] = line.words;
        const t = Number(time),
            v = args.map(Number),
            [a, b, n] = v;
        let error = "";
        if (!(command in counts)) error = `unknown command ${command}`;
        else if (args.length !== counts[command])
            error = `wrong number of arguments for ${command}`;
        else if (t < last) error = "time goes backwards";
        else if (command === "CALL") {
            if (a < 1 || a > c.floors || b < 1 || b > c.floors) error = "bad floor";
            else if (a === b) error = "same floor";
            else if (n < 1 || n > 99) error = "bad count";
            else if (!c.starts.some((_, i) => serves(c, i, a, b)))
                error = `no car serves ${a}->${b}`;
            else if (fire) error = "fire service active";
        } else if (command === "CANCEL" || command === "WHERE") {
            if (a < 1 || a > groups) error = "unknown group";
        } else if (["OUT", "IN", "HOLD"].includes(command)) {
        } else if (command === "FIRE") {
        } else if (command === "RESET" && !fire) error = "fire not active";
        if (error) {
            output.push(`line ${line.number}: ${error}`);
            continue;
        }
        accepted.push({
            time: t,
            command,
            values: v,
        });
        last = t;
        if (command === "CALL") ++groups;
    }
    return accepted;
function simulate(c: Config, input: Event[], policy: string, logging = true): Result {

        else {

            candidates = r.cars

        chosen.car.calls.push(g);

        group.lost += group.waiting;

    const choose = (car: Car) => {

            s = targets(car),

                load(car) * 100 >= c.capacity * c.bypass

function statistics(c: Config, r: Result, out: string[]) {
    r.groups.forEach((g, i) =>
        out.push(
            `G${i + 1} t=${g.time} ${g.a}->${g.b} n=${g.n} delivered=${g.delivered} lost=${g.lost} wait=${g.boarded === g.n ? g.lastBoard - g.time : "-"} travel=${g.delivered === g.n ? g.lastAlight - g.time : "-"} cars=${join([...g.cars].sort((a, b) => a - b))}`,
        ),
    );
    r.cars.forEach((car, i) => {
        const u = Math.floor((car.busy * 1000) / (r.end + 1));
        out.push(
            `E${i + 1} distance=${car.distance} stops=${car.stops} boarded=${car.boarded} busy=${car.busy} util=${Math.floor(u / 10)}.${u % 10}% energy=${car.energy}`,
        );
    });
    const aggregate = (label: string, gs: Group[]) =>
        out.push(
            `${label}: groups=${gs.length} passengers=${gs.reduce((n, g) => n + g.n, 0)} delivered=${gs.reduce((n, g) => n + g.delivered, 0)} avg_wait=${avg(
                gs.reduce((n, g) => n + g.wait, 0),
                gs.reduce((n, g) => n + g.delivered, 0),
            )}`,
        );

        `summary: groups=${r.groups.length} passengers=${r.groups.reduce((n, g) => n + g.n, 0)} delivered=${r.delivered} abandoned=${r.abandoned} evacuated=${r.evacuated} avg_wait=${avg(r.wait, r.delivered)} avg_travel=${avg(r.travel, r.delivered)} max_wait=${r.delivered ? r.maxWait : "-"} end=${r.end}`,
