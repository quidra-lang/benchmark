function initial()
    b=fill('.', 64)
    b[28]=b[37]='W'
    b[29]=b[36]='B'
    String(b)
Othello() = Othello(initial(), 'B', false, Move[], Move[])
function restore!(e, p)
    e.board=p.board
    e.side=p.side
    e.over=p.over
function reset!(e)
    restore!(e, Position(initial(), 'B', false))
    empty!(e.history)
    empty!(e.redo)
    if cmd=="NEW"
        reset!(e)
        emit("OK")
