local function initial()
    local b={};
    for i=1,64 do
        b[i]='.'
    end;
    b[28]='W';
    b[37]='W';
    b[29]='B';
    b[36]='B';
    return table.concat(b)
function Engine.new()
    local e=setmetatable({},Engine);
    e:reset();
    return e
function Engine:reset()
    self.board=initial();
    self.side='B';
    self.over=false;
    self.history={};
    self.redo={}
    if cmd=='NEW' then
        self:reset();
        emit('OK')
