type automaton struct {
	kind        string
	states      []string
	alphabet    string
	start       int
	final       []bool
	transitions [][][]int
}
func makeAutomaton(kind string, states []string, alphabet string, start int) *automaton {
	a := &automaton{kind: kind, states: states, alphabet: alphabet, start: start, final: make([]bool, len(states)), transitions: make([][][]int, len(states))}
	for i := range states {
		a.transitions[i] = make([][]int, len(alphabet)+1)
	}
	return a
}
func series(prefix string, n int) []string {
	names := make([]string, n)
	for i := range names {
		names[i] = prefix + strconv.Itoa(i)
	}
	return names
}
func (a *automaton) add(p, c, q int) {
	for _, old := range a.transitions[p][c] {
		if old == q {
			return
		}
	}
	a.transitions[p][c] = append(a.transitions[p][c], q)
	sort.Ints(a.transitions[p][c])
}
func orderedSet(set []bool) []int {
	result := []int{}
	for i, present := range set {
		if present {
			result = append(result, i)
		}
	}
	return result
}
func (a *automaton) closure(states []int) []int {
	seen := make([]bool, len(a.states))
	queue := []int{}
	for _, state := range states {
		if !seen[state] {
			seen[state] = true
			queue = append(queue, state)
		}
	}
	for i := 0; i < len(queue); i++ {
		for _, next := range a.transitions[queue[i]][len(a.alphabet)] {
			if !seen[next] {
				seen[next] = true
				queue = append(queue, next)
			}
		}
	}
	return orderedSet(seen)
}
func (a *automaton) destination(states []int, c int) []int {
	next := []int{}
	for _, state := range states {
		next = append(next, a.transitions[state][c]...)
	}
	return a.closure(next)
}
func (a *automaton) accepts(states []int) bool {
	for _, state := range states {
		if a.final[state] {
			return true
		}
	}
	return false
}
func setKey(states []int) string {
	parts := []string{}
	for _, state := range states {
		parts = append(parts, strconv.Itoa(state))
	}
	return strings.Join(parts, ",")
}
func determinize(a *automaton) (*automaton, [][]int) {
	sets := [][]int{a.closure([]int{a.start})}
	indices := map[string]int{setKey(sets[0]): 0}
	rows := [][]int{}
	for i := 0; i < len(sets); i++ {
		row := []int{}
		for c := range a.alphabet {
			next := a.destination(sets[i], c)
			key := setKey(next)
			index, exists := indices[key]
			if !exists {
				index = len(sets)
				indices[key] = index
				sets = append(sets, next)
			}
			row = append(row, index)
		}
		rows = append(rows, row)
	}
	d := makeAutomaton("DFA", series("D", len(sets)), a.alphabet, 0)
	for i, set := range sets {
		d.final[i] = a.accepts(set)
		for c, next := range rows[i] {
			d.add(i, c, next)
		}
	}
	return d, sets
func countLength(a *automaton, n int) int64 {
	counts := make([]int64, len(a.states))
	counts[a.start] = 1
	for step := 0; step < n; step++ {
		next := make([]int64, len(a.states))
		for state, count := range counts {
			for c := range a.alphabet {
				next[a.transitions[state][c][0]] += count
			}
		}
		counts = next
	}
	total := int64(0)
	for state, count := range counts {
		if a.final[state] {
			total += count
		}
	}
	return total
func fail(format string, args ...any) { panic(problem(fmt.Sprintf(format, args...))) }
func (t *toolkit) find(name string) *automaton {
	for _, entry := range t.table {
		if entry.name == name {
			return entry.automaton
		}
	}
	return nil
}
func (t *toolkit) command(words []string) []string {
	command, args := words[0], words[1:]
	counts := map[string]int{"ACCEPT": 2, "CLOSURE": 2, "TODFA": 2, "EPSFREE": 2, "MINIMIZE": 2, "EQUIV": 2, "UNION": 3, "INTERSECT": 3, "DIFF": 3, "COMPLEMENT": 2, "REVERSE": 2, "CONCAT": 3, "STAR": 2, "ENUM": 2, "COUNT": 2, "EMPTY": 1, "FINITE": 1, "INFO": 1, "TABLE": 1, "SHOW": 1, "SYNC": 1, "LIST": 0, "DELETE": 1}
	count, known := counts[command]
	if !known {
		fail("unknown command %s", command)
	}
	if len(args) != count {
		fail("wrong number of arguments")
	}
	references := 1
	if command == "EQUIV" || command == "UNION" || command == "INTERSECT" || command == "DIFF" || command == "CONCAT" {
	}
	autos := []*automaton{}
	for i := 0; i < references; i++ {
		a := t.find(args[i])
		if a == nil {
			fail("unknown automaton %s", args[i])
		}
		autos = append(autos, a)
	}
	a := autos[0]
	switch command {
	case "COUNT":
		n, _ := strconv.Atoi(args[1])
		d, _ := determinize(a)
		return []string{strconv.FormatInt(countLength(d, n), 10)}
