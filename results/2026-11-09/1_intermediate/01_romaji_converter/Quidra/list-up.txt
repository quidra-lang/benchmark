int locate(string needle, string[] haystack)
    for: k < #haystack
        if haystack[k] == needle
            return int(k)
    return -1

string lower_ascii(string value)

    target := "abcdefghijklmnopqrstuvwxyz"
    result := ""

            nat position
                result += target[position]
            none
                result += value[k]
    return result

int case_pattern(string value)
    all_lower := true
    all_upper := true
    title := contains("ABCDEFGHIJKLMNOPQRSTUVWXYZ", value[0])
    letters := 0
    for: k < #value
        character := value[k]
        if contains("ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz", character)
            letters += 1
            if contains("ABCDEFGHIJKLMNOPQRSTUVWXYZ", character)
                all_lower = false
                if k > 0
                    title = false
            else
                all_upper = false
    if all_lower
        return 0
Syllable syllable(const Inventory db, string system, string value, int position)
    for width := [3, 2, 1][]
        if position + width <= int(#value)
            match db.readings.get(system + ":" + value[position:<position + width])
                int mora
                    return Syllable(mora, width)
                none
                    continue
    return Syllable(-1, 0)
Reading read_word(const Inventory db, string system, string original, int start)
    value := lower_ascii(original)
    int[] morae := []
    int[] columns := []
    p := 0
    while p < int(#value)
        c := value[p]
        if c == "-"
        elif c == "'"
        else
            matched := syllable(db, system, value, p)
            if matched.mora >= 0
                mora = matched.mora
                width = matched.width
        morae = morae.append(mora)
        columns = columns.append(start + p)
        p += width
    return Reading("", morae, columns)
int[][] long_flags(const Inventory db, int[] morae)
    pairs := array(#morae, fill = 0)
    elements := array(#morae, fill = 0)
    nat p := 0
    while p < #morae
        m := morae[p]
        if m == db.L
            elements[p] = 1
        if m < db.N and p + 1 < #morae and morae[p + 1] < 5
        p += 1
    return [pairs, elements]

string apply_case(string value, int pattern, string case_value)
    actual := pattern
    if case_value == "LOWER"
        actual = 0
    elif case_value == "TITLE"
        actual = 1
    elif case_value == "UPPER"
        actual = 2
    if actual == 1
        return upper_ascii(value[0]) + value[1:]
    if actual == 2
        return upper_ascii(value)
    return value
string execute(const Inventory db, string input_line, string[] &settings)
    fields := input_line.split(" ")
    command := fields[0]
    names := "SET TABLE CONVERT NORMALIZE REVERSE VALID DETECT DIFF SAME MORAE SYLLABLES ROW STATS SORT INDEX FIND HAIKU KAIBUN SHIRITORI".split(" ")
    if locate(command, names) < 0
        return "ERROR unknown command {command}"
    argument_count := if command == "NORMALIZE" or command == "DETECT" then 0 elif command == "CONVERT" or command == "DIFF" or command == "SAME" or command == "FIND" then 2 else 1
    systems := system_names()
    source := "NORMALIZE"
    target := "NORMALIZE"
    if argument_count > 0
        source = fields[1]
        if locate(source, systems) < 0
            return "ERROR unknown system {source}"
        target = source
    if command == "CONVERT" or command == "DIFF" or command == "SAME"
        target = fields[2]
        if locate(target, systems) < 0
            return "ERROR unknown system {target}"
    text_start := int(#command) + 1
    for: 1 <= k <= argument_count
        text_start += int(#fields[k]) + 1
    text := input_line[text_start:]
    letters := "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ'-"
    Word[] words := []
    int[] word_lines := []
    poem_line := 0
    p := 0
    while p < int(#text)
        if contains(letters, text[p])
            start := p
            while p < int(#text) and contains(letters, text[p])
                p += 1
            original := text[start:<p]
            words = words.append(Word(original, start + 1, case_pattern(original), [], []))
    for: k < #words
        system := if command == "SAME" and word_lines[k] == 1 then target else source
        reading := read_word(db, system, words[k].original, words[k].start)
        if reading.problem != ""
            return "ERROR " + reading.problem
        words[k].morae = reading.morae
        words[k].columns = reading.columns
    if command == "CONVERT" or command == "NORMALIZE" or command == "REVERSE"
        result := ""
        cursor := 0
        for word := words[]
            result += text[cursor:<word.start - 1]
            writable := word
            if command == "REVERSE"
            written := write_parts(db, target, writable, settings[0])
            if written.problem != ""
                return "ERROR " + written.problem
            result += apply_case(written.parts.join(""), word.pattern, settings[1])
            cursor = word.start - 1 + int(#word.original)
        return result + text[cursor:]
