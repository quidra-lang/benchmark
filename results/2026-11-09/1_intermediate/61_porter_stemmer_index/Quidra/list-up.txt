bool includes(string[] values, string value)
    for candidate := values[]
        if candidate == value
            return true
    return false

bool is_letter(string character)
    return "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz".contains(character)

bool is_word(string text)
    if not (1 <= #text <= 30)
        return false
    for: index < #text
        if not is_letter(text[index])
            return false
    return true

string lowercase(string text)
    uppercase_letters := "ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    lowercase_letters := "abcdefghijklmnopqrstuvwxyz"
    result := ""
    for: index < #text
        character := text[index]
        match uppercase_letters.find(character)
            nat position
                result += lowercase_letters[position]
            none
                result += character
    return result
class Stemming(string stem, StepChange[] changes)
Stemming stemming_steps(string word)
    StepChange[] changes := []
    current := word
    if #word > 2
    return Stemming(current, changes)
string porter_stem(string word)
    return stemming_steps(word).stem
string | error word_argument(string text)
    if not is_word(text)
        return error("BAD WORD {text}")
    return lowercase(text)
string[] split_arguments(string name, string rest)
    return rest.split(" ")
void emit(string head, string[] items := [])
    if #items == 0
        println("{head}")
    else
        joined := items.join(" ")
        println("{head} {joined}")
class Token(string text, string form, string stem, nat line, nat position, nat start)
class Document(nat number)
    string[] lines := []
    Token[] tokens := []

    Token[] .add_line(string text)
        lines = lines.append(text)
        Token[] added := []
        nat index := 0
        while index < #text
            if not is_letter(text[index])
                index += 1
                continue
            start := index
            while index < #text and is_letter(text[index])
                index += 1
            word := text[start:<index]
            form := lowercase(word)
            token := Token(word, form, porter_stem(form), #lines, #tokens + 1, start)
            tokens = tokens.append(token)
            added = added.append(token)
        return added
    string reference(Token token)
        return "{number}:{token.line}:{token.position}"

class Occurrence(nat index, Token token)
class Command(string name, nat minimum, nat maximum)
Command[] commands()
    return [
        Command("PHRASE", 2, 10),
        Command("STOP", 1, 1),
        Command("UNSTOP", 1, 1),
Command | error find_command(string name)
    known := commands()
    for command := known[]
        if command.name == name
            return command
    return error("UNKNOWN COMMAND {name}")

class SearchEngine(string[] stopwords)
    Document[] documents := []
    nat .add_document(nat number)
        nat index := 0
        while index < #documents and documents[index].number < number
            index += 1
        before := documents[0:<index]
        documents = before.append(Document(number)).concat(documents[index:<#documents])
        return index

    Token[] .add_line(nat index, string text)
        return .documents[index].add_line(text)
    bool is_indexed(Token token)
        return not includes(stopwords, token.form)
    map<string, Occurrence[]> occurrence_map()
        map<string, Occurrence[]> occurrences
        for: index < #documents
            for token := documents[index].tokens[]
                if is_indexed(token)
                    occurrence := Occurrence(index, token)
                    match occurrences.get(token.stem)
                        Occurrence[] found
                            .occurrences.set(token.stem, found.append(occurrence))
                        none
                            .occurrences.set(token.stem, [occurrence])
        return occurrences
    nat form_count(string form)
        nat count := 0
        for document := documents[]
            for token := document.tokens[]
                if token.form == form
                    count += 1
        return count

    bool token_matches(Token token, string target)
        return target == "?" or (is_indexed(token) and token.stem == target)

    bool phrase_matches(Token[] tokens, nat start, string[] targets)
        for: target := targets[offset]
            if not token_matches(tokens[start + offset], target)
                return false
        return true
    void | error phrase(string[] arguments)
        string[] words := []
        for argument := arguments[]
            words = words.append(try word_argument(argument))
        only_stopwords := true
        for word := words[]
            if not includes(stopwords, word)
                only_stopwords = false
        if only_stopwords
            return error("STOPWORDS ONLY")
        string[] targets := []
        for word := words[]
            targets = targets.append(if includes(stopwords, word) then "?" else porter_stem(word))
        string[] matches := []
        for document := documents[]
            if #targets <= #document.tokens
                for: 0 <= start <= #document.tokens - #targets
                    if phrase_matches(document.tokens, start, targets)
                        matches = matches.append(document.reference(document.tokens[start]))
        phrase_text := targets.join(" ")
        emit("PHRASE {DQ}{phrase_text}{DQ} matches={#matches}", matches)
    void | error .stop_word(string[] arguments)
        word := try word_argument(arguments[0])
        if includes(stopwords, word)
            return error("ALREADY STOPWORD {word}")
        stopwords = stopwords.append(word)
        emit("STOP {word} unindexed={form_count(word)}")
        return void

    void | error .unstop_word(string[] arguments)
        word := try word_argument(arguments[0])
        if not includes(stopwords, word)
            return error("NOT STOPWORD {word}")
        stopwords = [for stopword := stopwords[] if stopword != word then stopword]
        emit("UNSTOP {word} indexed={form_count(word)}")
        return void
    void | error .run(string line)
        name := line
        string[] arguments := []
        match line.find(" ")
            nat space
                name = line[0:<space]
                arguments = split_arguments(name, line[space + 1:<#line])
            none
                void
        command := try find_command(name)
        if not (command.minimum <= #arguments <= command.maximum)
            return error("BAD ARGUMENTS {name}")
        elif name == "PHRASE"
            return phrase(arguments)
        elif name == "STOP"
            return .stop_word(arguments)
        elif name == "UNSTOP"
            return .unstop_word(arguments)
    void .execute(string line)
        match .run(line)
            void
                void
            error problem
                println("ERROR {problem.message}")

string[] read_lines()
    string[] lines := []
    line := ""
    reading := true
    while reading
        match scan(&line)
            void
                lines = lines.append(line)
            error
                reading = false
    return lines

input_lines := read_lines()
header_words := input_lines[0].split(" ")
engine := SearchEngine(header_words[1:<#header_words])
nat line_index := 1
while input_lines[line_index].starts_with("DOC ")
    document_line := input_lines[line_index]
    document_index := .engine.add_document(nat(document_line[4:<#document_line]))
    line_index += 1
    while input_lines[line_index] != "ENDDOC"
        .engine.add_line(document_index, input_lines[line_index])
        line_index += 1
    line_index += 1
while input_lines[line_index] != "END"
    .engine.execute(input_lines[line_index])
    line_index += 1
