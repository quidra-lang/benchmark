string letters()
    return "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"
bool all_chars(string text, string alphabet)
    for: i < #text
        if not alphabet.contains(text[i])
            return false
    return true
string lower(string text)
    result := ""
    for: i < #text
        match "ABCDEFGHIJKLMNOPQRSTUVWXYZ".find(text[i])
            nat index
                result += "abcdefghijklmnopqrstuvwxyz"[index]
            none
                result += text[i]
    return result
nat find_at(string text, string needle, nat start := 0)
    if start > #text
        return #text
    match text[start:].find(needle)
        nat index
            return start + index
        none
            return #text
string trim_spaces(string text)
    nat start := 0
    end := #text
    while start < end and text[start] == " "
        start += 1
    while end > start and text[end - 1] == " "
        end -= 1
    return text[start:<end]
string left_trim(string text)
    nat start := 0
    while start < #text and text[start] == " "
        start += 1
    return text[start:]
string escape_html(string text)
    html := ""
    for: i < #text
        ch := text[i]
        if ch == "&"
        else
            html += ch
    return html
string strip_tags(string text)
    result := ""
    nat at := 0
    while at < #text
        if text[at] == "<"
            close := find_at(text, ">", at + 1)
            if close < #text
                at = close + 1
                continue
        result += text[at]
        at += 1
    return result
nat run_end(string text, nat start)
    end := start + 1
    while end < #text and text[end] == text[start]
        end += 1
    return end
class Block(string kind, text := "", nat number := 0, ordered := false)
    nat[] children := []
Block | none atx_heading(string line)
    if line == "" or line[0] != "#"
        return none
    size := run_end(line, 0)
    if size > 6 or (size < #line and line[size] != " ")
        return none
    text := trim_spaces(line[size:])
    end := #text
    while end > 0 and text[end - 1] == "#"
        end -= 1
    if end == 0 or text[end - 1] == " "
        text = trim_spaces(text[:<end])
    return Block("heading", text, size)
class Parser()
    Block[] blocks := []
    nat .add(Block block)
        blocks = blocks.append(block)
        return #blocks - 1
    nat[] .parse(string[] lines)
        nat[] result := []
        nat at := 0
        while at < #lines
            line := lines[at]
            if line == ""
                at += 1
                continue
            if not line.starts_with(" ")
                match atx_heading(line)
                    Block block
                        result = result.append(.add(block))
                        at += 1
                        continue
                    none
                        void
                if line.starts_with(">")
                    string[] content := []
                    while at < #lines and lines[at].starts_with(">")
                        row := lines[at][1:]
                        content = content.append(if row.starts_with(" ") then row[1:] else row)
                        at += 1
                    block := Block("quote")
                    block.children = .parse(content)
                    result = result.append(.add(block))
                    continue
            end := at + 1
            nat level := 0
            while end < #lines and lines[end] != ""
            text := [for row := lines[at:<end][] then left_trim(row)].join(NL)
            kind := if level > 0 then "heading" elif text == "[TOC]" then "toc" else "paragraph"
            result = result.append(.add(Block(kind, text, level)))
            at = end + (if level > 0 then 1 else 0)
        return result
class Piece(string html, character := "", nat count := 0, opens := false, closes := false)
string emphasis(Piece[] pieces, nat start, nat end)
    html := ""
    at := start
    while at < end
        piece := pieces[at]
        close := end
        if piece.opens
        if close < end
        else
            html += if piece.count == 0 then piece.html else piece.character * piece.count
            at += 1
    return html
class Heading(nat level, string identifier, string label)
string toc_list(Heading[] headings, int[] parents, int parent)
    html := "<ul>{NL}"
    for: entry := headings[i]
        if parents[i] != parent
            continue
        html += "<li><a href={DQ}#{entry.identifier}{DQ}>{entry.label}</a>"
        children := false
        for owner := parents[]
            if owner == int(i)
                children = true
        if children
            html += NL + toc_list(headings, parents, int(i)) + NL
        html += "</li>{NL}"
    return html + "</ul>"
class Renderer(local Parser parser)
    Block[] blocks := parser.blocks
    used_ids := distinct<string>()
    Heading[] headings := []
    string[] notes := []
    string .inline_html(string text, links := true, footnotes := true)
        Piece[] pieces := []
        nat at := 0
        while at < #text
            ch := text[at]
            pieces = pieces.append(Piece(escape_html(ch)))
            at += 1
        return emphasis(pieces, 0, #pieces)
    string .heading_html(Block block)
        text := block.text
        explicit_at := #text
        nat at := 0
        while at < #text
            found := find_at(text, " {{#", at)
            if found == #text
                break
            explicit_at = found
            at = found + 1
        id := ""
        if text.ends_with("}}") and explicit_at < #text
            candidate := text[explicit_at + 3:<#text - 1]
            if candidate != "" and all_chars(candidate, letters() + "-_")
                id = candidate
                text = trim_spaces(text[:<explicit_at])
        if id == ""
            base := ""
            lowered := lower(text)
            for: i < #lowered
                ch := lowered[i]
                if "abcdefghijklmnopqrstuvwxyz0123456789-_".contains(ch)
                    base += ch
                elif ch == " " or ch == NL
                    base += "-"
            if base == ""
                base = "heading"
            id = base
            nat suffix := 1
            while used_ids.has(id)
                id = "{base}-{suffix}"
                suffix += 1
        .used_ids.add(id)
        string html := .inline_html(text)
        headings = headings.append(Heading(block.number, id, strip_tags(html)))
        return "<h{block.number} id={DQ}{id}{DQ}>{html}</h{block.number}>"
    string .sequence(nat[] indices)
        html := ""
        for index := indices[]
            html += .render(index) + NL
        return html
    string .render(nat index)
        block := blocks[index]
        if block.kind == "heading"
            return .heading_html(block)
        if block.kind == "toc"
            return "{HT}toc{HT}"
        if block.kind == "quote"
            string content := .sequence(block.children)
            return "<blockquote>{NL}{content}</blockquote>"
    string contents()
        int[] parents := []
        nat[] stack := []
        for: entry := headings[i]
            while #stack > 0 and headings[stack[-1]].level >= entry.level
                stack = stack[:<#stack - 1]
            parents = parents.append(if #stack == 0 then -1 else int(stack[-1]))
            stack = stack.append(i)
        string body := if #headings == 0 then "" else toc_list(headings, parents, -1) + NL
        return "<nav class={DQ}toc{DQ}>{NL}{body}</nav>"
    string .footnote_list()
        if #notes == 0
            return ""
