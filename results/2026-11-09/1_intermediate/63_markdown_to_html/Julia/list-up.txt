const TOC = "\x01toc\x02"
part(s, a, b = lastindex(s)) = a > b ? "" : String(s[a:b])
starts(s, prefix, at = 1) = startswith(part(s, at), prefix)
escapehtml(s) = replace(s, '&'=>"&amp;", '<'=>"&lt;", '>'=>"&gt;", '"'=>"&quot;")
striptags(s) = replace(s, r"<[^>]*>"=>"")
trim(s) = String(strip(s, ' '))
ltrim(s) = String(lstrip(s, ' '))
mutable struct Block
    kind::Symbol
    text::String
    number::Int
    ordered::Bool
    lines::Vector{String}
    children::Vector{Block}
    items::Vector{Vector{Block}}
    alignments::Vector{String}
    rows::Vector{Vector{String}}
end
Block(kind, text = "", number = 0) =
    Block(kind, text, number, false, String[], Block[], Vector{Block}[], String[], Vector{String}[])
mutable struct Parser
    references::Dict{String,Reference}
    footnotes::Dict{String,String}
end
Parser() = Parser(Dict{String,Reference}(), Dict{String,String}())
function atx(line)
    m = match(r"^(#{1,6})(?: |$)", line)
    m === nothing && return nothing
    n = length(m.captures[1])
    text = trim(part(line, n+1))
    shorter = String(rstrip(text, '#'))
    if isempty(shorter) || endswith(shorter, " ")
        text = trim(shorter)
    end
    Block(:heading, text, n)
end
function parseblocks!(p, lines)
    blocks = Block[]
    at = 1
    while at <= length(lines)
        line = lines[at]
        if isempty(line)
            at += 1
            continue
        end
        if !startswith(line, " ")
            h = atx(line)
            if h !== nothing
                push!(blocks, h)
                at += 1
                continue
            end
            if startswith(line, ">")
                content = String[]
                while at <= length(lines) && startswith(lines[at], ">")
                    row = part(lines[at], 2)
                    push!(content, starts(row, " ") ? part(row, 2) : row)
                    at += 1
                end
                alert = match(r"^\[!(NOTE|TIP|IMPORTANT|WARNING|CAUTION)\]$", content[1])
                b = alert === nothing ? Block(:quote) : Block(:alert, String(alert.captures[1]))
                b.children = parseblocks!(p, alert === nothing ? content : content[2:end])
                push!(blocks, b)
                continue
            end
        end
        finish = at+1
        level = 0
        while finish<=length(lines) && !isempty(lines[finish])
        end
        text = join(ltrim.(lines[at:(finish-1)]), "\n")
        push!(blocks, Block(level>0 ? :heading : text=="[TOC]" ? :toc : :paragraph, text, level))
        at = finish+(level>0)
    end
    blocks
end
struct Piece
    html::String
    char::Char
    count::Int
    opens::Bool
    closes::Bool
end
Piece(html::String) = Piece(html, ' ', 0, false, false)
function emphasis(pieces, first = 1, last = length(pieces))
    out = IOBuffer()
    at = first
    while at<=last
        p = pieces[at]
        close = 0

        end

        else
            print(out, p.count==0 ? p.html : repeat(string(p.char), p.count))

end
mutable struct Renderer
    references::Dict{String,Reference}
    definitions::Dict{String,String}
    ids::Set{String}
    headings::Vector{Tuple{Int,String,String}}
    notes::Vector{String}
    counts::Vector{Int}
end
Renderer(p::Parser) =
    Renderer(p.references, p.footnotes, Set{String}(), Tuple{Int,String,String}[], String[], Int[])
function inline!(r, s; links = true, notes = true)
    pieces=Piece[]
    at=1
    while at<=length(s)
        ch=s[at]
        push!(pieces, Piece(escapehtml(string(ch))))
        at+=1
    end
    emphasis(pieces)
end
function renderheading!(r, b)
    text=b.text
    m=match(r"^(.*) \{#([A-Za-z0-9_-]+)\}$"s, text)
    if m!==nothing
        text=String(rstrip(m.captures[1], ' '))
        id=String(m.captures[2])
    else
        base=replace(replace(lowercase(text), r"[^a-z0-9_ \n-]"=>""), r"[ \n]"=>"-")
        isempty(base) && (base="heading")
        id=base
        suffix=1
        while id in r.ids
            id="$base-$suffix"
            suffix+=1
        end
    end
    push!(r.ids, id)
    html=inline!(r, text)
    push!(r.headings, (b.number, id, striptags(html)))
    "<h$(b.number) id=\"$id\">$html</h$(b.number)>"
end
renderblocks!(r, blocks) = join((renderblock!(r, b)*"\n" for b in blocks))
function renderblock!(r, b)
    if b.kind==:heading
        return renderheading!(r, b)
    elseif b.kind==:toc
        return TOC
    elseif b.kind==:quote
        return "<blockquote>\n"*renderblocks!(r, b.children)*"</blockquote>"
    end
end
function toc(r)
    parents=Int[]
    stack=Int[]
    for (i, (level, id, label)) in enumerate(r.headings)
        while !isempty(stack) && r.headings[last(stack)][1]>=level
            pop!(stack)
        end
        push!(parents, isempty(stack) ? 0 : last(stack))
        push!(stack, i)
    end
    function listing(parent)
        html="<ul>\n"
        for i in findall(==(parent), parents)
            _, id, label=r.headings[i]
            html*="<li><a href=\"#$id\">$label</a>"
            if i in parents
                html*="\n"*listing(i)*"\n"
            end
            html*="</li>\n"
        end
        html*"</ul>"
    end
    "<nav class=\"toc\">\n"*(isempty(parents) ? "" : listing(0)*"\n")*"</nav>"
end
function footnotes!(r)
    isempty(r.notes) && return ""
end
function main()
    lines=String[]
    for line in eachline(stdin)
        line=="@@END@@" && break
        push!(lines, line)
    end
    parser=Parser()
    blocks=parseblocks!(parser, lines)
    renderer=Renderer(parser)
    html=renderblocks!(renderer, blocks)
    print(replace(html, TOC=>toc(renderer)), footnotes!(renderer))
end
main()
