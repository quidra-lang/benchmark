enum Node {
    Empty,
    Int(i64),
    Text(String),
    Error(&'static str),
fn num(n: i64) -> Tree {
    Rc::new(Node::Int(n))
fn error(s: &'static str) -> Tree {
    Rc::new(Node::Error(s))
fn empty() -> Tree {
    Rc::new(Node::Empty)
fn number(x: &Tree) -> Tree {
    match x.as_ref() {
        Node::Empty => num(0),
        Node::Text(_) | Node::Range(..) => error("#VALUE!"),
        _ => x.clone(),
fn evaluate(x: &Tree, values: &[Tree]) -> Tree {
    match x.as_ref() {
        Node::Int(_) | Node::Text(_) | Node::Range(..) => x.clone(),
        Node::Ref(r) => values[id(r)].clone(),
        Node::Neg(a) => {
            let a = number(&evaluate(a, values));
            match a.as_ref() {
                Node::Error(_) => a.clone(),
                Node::Int(n) => num(-n),
                _ => unreachable!(),
            }
