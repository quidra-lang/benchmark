const Value = struct { k: enum { empty, num, str, err, range } = .empty, n: i64 = 0, s: S = "", a: Ref = .{ .c = 0, .r = 0 }, b: Ref = .{ .c = 0, .r = 0 } };
fn num(n: i64) Value {
    return .{ .k = .num, .n = n };
fn err(s: S) Value {
    return .{ .k = .err, .s = s };
fn numeric(v: Value) Value {
    return switch (v.k) {
        .empty => num(0),
        .num, .err => v,
        else => err("#VALUE!"),
    };
