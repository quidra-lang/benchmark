bool text_less(string a, string b)
    int[] first := a.codepoints()
    int[] second := b.codepoints()
    count := if #first < #second then #first else #second
    for: i < count
        if first[i] != second[i]
            return first[i] < second[i]
    return #first < #second

void emit(string text)
    println(text)

bool contains_word(string[] items, string value)
    for item := items[]
        if item == value
            return true
    return false
nat position(string[] items, string value)
    for: item := items[i]
        if item == value
            return i
    return #items

int parse_time(string text)
    return int(text[:<2]) * 60 + int(text[3:])
int half_price(int value)
    return ((value + 19) / 20) * 10

string segment_key(string first, string second)
    return if text_less(first, second) then first + "-" + second else second + "-" + first

class Station(string id, nat zone, string name)
    transfer := 0
    closed := false
    walks := map<string, int>()

class TransitLine(string id, int first, int last, int headway, string[] stops, int[] offsets)
    delay := 0
    suspended := false

    int trips()
        return (last - first) / headway + 1

class Direction(TransitLine service, string[] stops, int[] offsets)
    int train_time(int train, nat at)
        return service.first + service.delay + train * service.headway + offsets[at]

    int first_train(nat at, int ready)
        int delta := ready - train_time(0, at)
        train := if delta <= 0 then 0 else (delta + service.headway - 1) / service.headway
        return if train < service.trips() then train else -1
Direction[] directions(TransitLine service)
    total := service.offsets[-1]
    return [Direction(service, service.stops, service.offsets), Direction(service, service.stops[::-1], [for offset := service.offsets[::-1][] then total - offset])]

class Leg(string kind, string line_id, string terminal, string origin, int start, string destination, int finish, nat stops)
class Label(timestamp := -1, Leg[] journey := [])
class Labels()
    data := map<string, Label>()

    Label get_label(string name)
        match data.get(name)
            Label value
                return value
            none
                return Label()

    bool improves(string name, int arrival)
        Label old := get_label(name)
        return old.timestamp < 0 or arrival < old.timestamp

    void .put_label(string name, Label value)
        .data.set(name, value)

class WalkResult(Labels labels, string[] changed)
class Ticket(Leg leg, nat zones, int price, int discount)
Leg make_ride(Direction route, int train, nat first, nat last)
    return Leg("ride", route.service.id, route.stops[-1], route.stops[first], route.train_time(train, first), route.stops[last], route.train_time(train, last), last - first)
class Planner()
    stations := map<string, Station>()
    services := map<string, TransitLine>()
    closed_segments := map<string, bool>()
    discount := 0
    window := 0
    int[] prices := []

    Station get_station(string id)
        match stations.get(id)
            Station value
                return value
            none
                return Station(id, 1, "")

    TransitLine get_service(string id)
        match services.get(id)
            TransitLine value
                return value
            none
                return TransitLine(id, 0, 0, 1, [], [])

    void | error check_station(string id)
        if not stations.has(id)
            return error("unknown station {id}")
        return void
    void | error check_open(string id)
        if get_station(id).closed
            return error("station {id} is closed")
        return void

    void | error endpoints(string origin, string destination)
        try check_station(origin)
        try check_station(destination)
        if origin == destination
            return error("origin and destination are the same")
        try check_open(origin)
        try check_open(destination)
        return void

    Direction | error toward(TransitLine service, string terminal)
        if terminal != service.stops[0] and terminal != service.stops[-1]
            return error("{terminal} is not a terminal of line {service.id}")
        return directions(service)[if terminal == service.stops[-1] then 0 else 1]

    bool blocked(string first, string second)
        return closed_segments.has(segment_key(first, second))

    TransitLine[] ordered_services(active := false)
        return [for id := services.keys().sorted()[] if not active or not get_service(id).suspended then get_service(id)]

    int ready_time(string origin, string name, int arrival)
        return arrival + (if origin == name then 0 else get_station(name).transfer)

    WalkResult walk_from(Labels initial, string name, Label source)
        result := initial
        string[] changed := []
        Station station := get_station(name)
        for neighbor := station.walks.keys().sorted()[]
            minutes := 0
            match station.walks.get(neighbor)
                int value
                    minutes = value
                none
                    void
            arrival := source.timestamp + minutes
            if not get_station(neighbor).closed and arrival <= 1440 and result.improves(neighbor, arrival)
                leg := Leg("walk", "", "", name, source.timestamp, neighbor, arrival, 0)
                .result.put_label(neighbor, Label(arrival, source.journey.append(leg)))
                changed = changed.append(neighbor)
        return WalkResult(result, changed)

    Labels ride_from(Labels initial, Direction route, string name, int ready, Leg[] journey)
        result := initial
        at := position(route.stops, name)
        if at >= #route.stops - 1
            return result
        train := route.first_train(at, ready)
        if train < 0
            return result
        for: at + 1 <= j < #route.stops
            stop := route.stops[j]
            if blocked(route.stops[j - 1], stop)
                break
            if get_station(stop).closed
                continue
            arrival := route.train_time(train, j)
            if result.improves(stop, arrival)
                .result.put_label(stop, Label(arrival, journey.append(make_ride(route, train, at, j))))
        return result

    Labels earliest(string origin, int departure)
        result := Labels()
        .result.put_label(origin, Label(departure))
        string[] settled := []
        TransitLine[] active := ordered_services(true)
        while true
            selected := ""
            timestamp := -1
            for name := result.data.keys().sorted()[]
                if not contains_word(settled, name)
                    candidate := result.get_label(name).timestamp
                    if timestamp < 0 or candidate < timestamp
                        selected = name
                        timestamp = candidate
            if selected == ""
                return result
            settled = settled.append(selected)
            source := result.get_label(selected)
            int ready := ready_time(origin, selected, timestamp)
            for service := active[]
                for route := directions(service)[]
                    result = ride_from(result, route, selected, ready, source.journey)
            result = walk_from(result, selected, source).labels
    Ticket[] fare_items(Leg[] journey, bool child)
        cap := if child then half_price(discount) else discount
        previous := -1
        Ticket[] result := []
        for leg := journey[]
            if leg.kind != "ride"
                continue
            Direction route := toward(get_service(leg.line_id), leg.terminal)
            at := position(route.stops, leg.origin)
            nat[] zones := []
            for: at <= j <= at + leg.stops
                nat zone := get_station(route.stops[j]).zone
                found := false
                for value := zones[]
                    if value == zone
                        found = true
                if not found
                    zones = zones.append(zone)
            price_index := if #zones <= #prices then #zones - 1 else #prices - 1
            price := if child then half_price(prices[price_index]) else prices[price_index]
            reduction := if previous >= 0 and leg.start - previous <= window then (if cap < price then cap else price) else 0
            result = result.append(Ticket(leg, #zones, price, reduction))
            previous = leg.finish
        return result
    void | error .execute(string text)
        words := text.split(" ")
        command := words[0]
        args := words[1:]
        if command == "STATION"
            id := args[0]
            if stations.has(id)
                return error("station {id} already exists")
            .stations.set(id, Station(id, nat(args[1]), args[2]))
        elif command == "LINE"
            id := args[0]
            if services.has(id)
                return error("line {id} already exists")
            stops := args[4::2]
            for name := stops[]
                try check_station(name)
            string[] seen := []
            for name := stops[]
                if contains_word(seen, name)
                    return error("station {name} appears twice on line {id}")
                seen = seen.append(name)
            first := parse_time(args[1])
            last := parse_time(args[2])
            if first > last
                return error("line {id} first departure is after last departure")
            offsets := [0]
            for word := args[5::2][]
                offsets = offsets.append(offsets[-1] + int(word))
            total := offsets[-1]
            if last + total > 1440
                return error("line {id} runs past 24:00")
            value := TransitLine(id, first, last, int(args[3]), stops, offsets)
            .services.set(id, value)
        elif command == "WALK"
            try check_station(args[0])
            try check_station(args[1])
            if args[0] == args[1]
                return error("walk from {args[0]} to itself")
            Station first := get_station(args[0])
            Station second := get_station(args[1])
            minutes := int(args[2])
            .first.walks.set(args[1], minutes)
            .second.walks.set(args[0], minutes)
            .stations.set(args[0], first)
            .stations.set(args[1], second)
        elif command == "TARIFF"
            int[] new_prices := [for word := args[2:][] then int(word)]
            for: 1 <= i < #new_prices
                if new_prices[i] < new_prices[i - 1]
                    return error("tariff prices must not decrease")
            discount = int(args[0])
            window = int(args[1])
            prices = new_prices
        elif command == "ROUTE" or command == "FEWEST" or command == "PARETO" or command == "FARE" or command == "LASTTRAIN" or command == "ARRIVEBY" or command == "PROFILE"
            origin := args[0]
            destination := args[1]
            try endpoints(origin, destination)
            if command == "FARE"
                if args[3] != "ADULT" and args[3] != "CHILD"
                    return error("unknown fare class {args[3]}")
                if #prices == 0
                    return error("no tariff")
            departure := parse_time(args[2])
            heading := "{command} {origin} {destination}"
            else
                Label value := earliest(origin, departure).get_label(destination)
                if value.timestamp < 0
                    emit("{heading} {args[2]}" + (if command == "FARE" then " {args[3]}" else "") + " unreachable")
                else
                    Ticket[] tickets := fare_items(value.journey, args[3] == "CHILD")
                    total := 0
                    for ticket := tickets[]
                        total += ticket.price - ticket.discount
                    emit("{heading} {args[2]} {args[3]} rides {#tickets} total {total}")
                    for ticket := tickets[]
                        emit("TICKET {ticket.leg.line_id} {ticket.leg.origin} {ticket.leg.destination} zones {ticket.zones} price {ticket.price} discount {ticket.discount} pay {ticket.price - ticket.discount}")
        return void

planner := Planner()
nat count
scan(&count)
for: i < count
    string line
    scan(&line)
    match .planner.execute(line)
        void
            void
        error issue
            emit("ERROR {issue.message}")
