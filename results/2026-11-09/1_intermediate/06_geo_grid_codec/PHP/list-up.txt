function ancestor(string $sys, array $cell, int $q): array
{
    [$p, $x, $y] = $cell;
    [$w, $h] = grid($sys, $p);
    [$v, $u] = grid($sys, $q);
    return [$q, intdiv($x * $v, $w), intdiv($y * $u, $h)];
