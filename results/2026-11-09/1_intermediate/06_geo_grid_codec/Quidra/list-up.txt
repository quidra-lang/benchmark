Cell ancestor(const Cell cell, int p)
    old_dim := dimensions(cell.sys, cell.p)
    new_dim := dimensions(cell.sys, p)
    return Cell(cell.sys, p, cell.x * new_dim[0] / old_dim[0], cell.y * new_dim[1] / old_dim[1])
bool same_cell(const Cell a, const Cell b)
    return a.p == b.p and a.x == b.x and a.y == b.y
string compact(Cell[] cells)
    Cell[] current := []
    for cell := cells[]
        covered := false
        for other := cells[]
            if other.p < cell.p and same_cell(ancestor(cell, other.p), other)
                covered = true
        for other := current[]
            if same_cell(cell, other)
                covered = true
        if not covered
            current = current.append(cell)
    changed := true
    while changed
        changed = false
        for cell := current[]
            index := precision_index(cell.sys, cell.p)
            if index == 0
                continue
            parent := ancestor(cell, precisions(cell.sys)[index - 1])
            child_dim := dimensions(cell.sys, cell.p)
            parent_dim := dimensions(cell.sys, parent.p)
            required := (child_dim[0] / parent_dim[0]) * (child_dim[1] / parent_dim[1])
            count := 0
            for other := current[]
                if other.p == cell.p and same_cell(ancestor(other, parent.p), parent)
                    count += 1
            if count != required
                continue
            remaining := [for other := current[] if other.p != cell.p or not same_cell(ancestor(other, parent.p), parent) then other]
            current = remaining.append(parent)
            changed = true
            break
    for: 1 <= k < int(#current)
        cell := current[k]
        p := k
        while p > 0
            other := current[p - 1]
            if other.p < cell.p or (other.p == cell.p and (other.y > cell.y or (other.y == cell.y and other.x <= cell.x)))
                break
            current[p] = other
            p -= 1
        current[p] = cell
    result := [for cell := current[] then encode(cell)]
    return result.join(" ")
