function point(id: number): Point {
    return id === 0 ? depot! : getStop(id);
}
function distance(a: number, b: number): number {
    const x = point(a),
        y = point(b);
    return Math.abs(x.x - y.x) + Math.abs(x.y - y.y);
}
function length(route: number[]): number {
    const path = [0, ...route, 0];
    return path.slice(1).reduce((sum, id, i) => sum + distance(path[i], id), 0);
