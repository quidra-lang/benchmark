const Int = i64;
fn abs(n: Int) Int {
    return if (n < 0) -n else n;
    fn distance(self: *Workbench, a: usize, b: usize) Int {
        const p = if (a == 0) self.depot.? else self.stops.get(a).?;
        const q = if (b == 0) self.depot.? else self.stops.get(b).?;
        return abs(p.x - q.x) + abs(p.y - q.y);
    }
    fn length(self: *Workbench, p: []const usize) Int {
        var total: Int = 0;
        var last: usize = 0;
        for (p) |id| {
            total += self.distance(last, id);
            last = id;
        }
        return total + self.distance(last, 0);
