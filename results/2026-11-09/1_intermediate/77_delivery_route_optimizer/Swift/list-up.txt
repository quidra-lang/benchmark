    func point(_ n: Int) -> (Int,Int) {
        if n == 0 {
            return depot!
        }
        let s = stops[n]!
        return (s.x,s.y)
    }
    func d(_ a: Int,_ b: Int) -> Int {
        let x = point(a),y = point(b)
        return abs(x.0-y.0)+abs(x.1-y.1)
    }
    func length(_ p: [Int]) -> Int {
        var at = 0
        var sum = 0
        for n in p {
            sum += d(at,n)
            at = n
        }
        return sum+d(at,0)
