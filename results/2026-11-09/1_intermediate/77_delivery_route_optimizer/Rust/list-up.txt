    fn point(&self, n: usize) -> (i64, i64) {
        if n == 0 {
            self.depot.unwrap()
        } else {
            let s = &self.stops[&n];
            (s.x, s.y)
        }
    }
    fn d(&self, a: usize, b: usize) -> i64 {
        let x = self.point(a);
        let y = self.point(b);
        (x.0 - y.0).abs() + (x.1 - y.1).abs()
    }
    fn length(&self, p: &[usize]) -> i64 {
        let mut at = 0;
        let mut sum = 0;
        for &n in p {
            sum += self.d(at, n);
            at = n;
        }
        sum + self.d(at, 0)
