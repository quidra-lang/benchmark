local function copy(v)
    if type(v) ~= "table" then
        return v
    end
    local r = {}
    for k, x in pairs(v) do
        r[k] = copy(x)
    end
    return r
end
local function color(x)
    return x == 1 and "W" or "B"
end
local function positionKey(pos)
    local a = {}
    for x = 1, 2 do
        for p = 0, 25 do
            a[#a + 1] = pos[x][p]
        end
    end
    return table.concat(a, ",")
end
local function pairLess(a, b)
    return a[1] > b[1] or a[1] == b[1] and a[2] > b[2]
end
local function notation(moves, hits)
    local key = copy(moves)
    table.sort(key, pairLess)
    while true do
        local ai, bi
        for i, a in ipairs(key) do
            if a[2] > 0 and not hits[a[2]] then
                for j, b in ipairs(key) do
                    if b[1] == a[2] then
                        ai, bi = i, j
                        break
                    end
                end
            end
            if ai then
                break
            end
        end
        if not ai then
            break
        end
        local a, b = key[ai], key[bi]
        table.remove(key, math.max(ai, bi))
        table.remove(key, math.min(ai, bi))
        key[#key + 1] = {a[1], b[2]}
        table.sort(key, pairLess)
    end
    local parts, marked, i = {}, {}, 1
    while i <= #key do
        local s, t = key[i][1], key[i][2]
        local j = i + 1
        while j <= #key and key[j][1] == s and key[j][2] == t do
            j = j + 1
        end
        local text = (s == 25 and "bar" or s) .. "/" .. (t == 0 and "off" or t)
        if hits[t] and not marked[t] then
            text = text .. "*"
            marked[t] = true
        end
        if j - i > 1 then
            text = text .. "(" .. j - i .. ")"
        end
        parts[#parts + 1] = text
        i = j
    end
    return key, #parts == 0 and "none" or table.concat(parts, " ")
end
local function validPoint(s)
    return s and s:match("^[1-9]%d?$") and tonumber(s) <= 24
end
local function parsePlay(tokens)
    if #tokens == 1 and tokens[1] == "none" then
        return {}
    end
    local moves = {}
    local function syntax()
        error("ILLEGAL syntax", 0)
    end
    for _, token in ipairs(tokens) do
        local path, k = token:match("^(.-)%(([234])%)$")
        if not path then
            path, k = token, 1
        else
            k = tonumber(k)
        end
        local fields = {}
        for field in (path .. "/"):gmatch("(.-)/") do
            fields[#fields + 1] = field
        end
        if #fields < 2 then
            syntax()
        end
        local values = {}
        for i, word in ipairs(fields) do
            if i == 1 then
                if word == "bar" then
                    values[i] = 25
                elseif validPoint(word) then
                    values[i] = tonumber(word)
                else
                    syntax()
                end
            else
                if word == "off" then
                    values[i] = 0
                else
                    local point = word:gsub("%*$", "")
                    if not validPoint(point) then
                        syntax()
                    end
                    values[i] = tonumber(point)
                end
                if values[i - 1] <= values[i] then
                    syntax()
                end
            end
        end
        for i = 1, #values - 1 do
            for _ = 1, k do
                moves[#moves + 1] = {values[i], values[i + 1]}
            end
        end
    end
    return moves
