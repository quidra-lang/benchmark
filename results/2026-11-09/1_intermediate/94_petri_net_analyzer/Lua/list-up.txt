function Net:enabled(m)
    local a={}
    for t=1,#self.t do
        if not self:reason(m,t) then
            a[#a+1]=t
        end
    end
    return a
function Net:commit(t)
    self.history[#self.history+1]=self.current;
    self.current=self:fire(self.current,t)
    elseif cmd=='SIMULATE' then
        local steps,x=number(w[2],1,1000),number(w[3],1,2147483646);
        if not steps then
            bad('invalid steps '..w[2])
        end
        if not x then
            bad('invalid seed '..w[3])
        end
        if w[4]~='PRIORITY' and w[4]~='RANDOM' then
            bad('invalid mode '..w[4])
        end
        local counts={}
        for t=1,#self.t do
            counts[t]=0
        end
        local done=0
        while done<steps do
            local e=self:enabled(self.current);
            if #e==0 then print('dead (steps='..done..')');break end
            local t=e[1];
            if w[4]=='RANDOM' then
                x=x*48271%2147483647;
                t=e[x%#e+1]
            else
                for _,u in ipairs(e) do
                    if self.priority[u]>self.priority[t] then
                        t=u
                    end
                end
            end
            self:commit(t);
            counts[t]=counts[t]+1;
            done=done+1;
            print('step '..done..': '..self.t[t]..' -> '..mark(self.current))
        end
        if done==steps then
            print('completed (steps='..steps..')')
        end
        local a={}
        for t,name in ipairs(self.t) do
            a[t]=name..'='..counts[t]
        end
        print('counts: '..table.concat(a,' '))
