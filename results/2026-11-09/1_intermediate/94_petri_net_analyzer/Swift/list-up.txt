    func enabled(_ m: [Int]) -> [Int] {
        transitions.indices.filter { reason(m, $0) == nil }
    func commit(_ t: Int) {
        history.append(current)
        current = fire(current, t)
        case "SIMULATE":
            guard let steps = number(w[1], 1, 1000) else {
                try bad("invalid steps \(w[1])")
            }
            guard var x = number(w[2], 1, 2147483646) else {
                try bad("invalid seed \(w[2])")
            }
            if !["PRIORITY", "RANDOM"].contains(w[3]) {
                try bad("invalid mode \(w[3])")
            }
            var counts = Array(repeating: 0, count: transitions.count), done = 0
            while done < steps {
                let es = enabled(current)
                if es.isEmpty {
                    print("dead (steps=\(done))")
                    break
                }
                var t = es[0]
                if w[3] == "RANDOM" {
                    x = x * 48271 % 2147483647
                    t = es[x % es.count]
                } else {
                    for u in es where priority[u] > priority[t] {
                        t = u
                    }
                }
                commit(t)
                counts[t] += 1
                done += 1
                print("step \(done): \(transitions[t]) -> \(mark(current))")
            }
            if done == steps {
                print("completed (steps=\(steps))")
            }
            print("counts: " + transitions.indices.map { "\(transitions[$0])=\(counts[$0])" }.joined(separator: " "))
