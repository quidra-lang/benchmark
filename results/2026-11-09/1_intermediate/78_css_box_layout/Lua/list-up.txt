function Element:frame(horizontal)
    if horizontal then
        return self:n("padding-left") + self:n("padding-right") + self:n("border-left") +
            self:n("border-right")
    end
    return self:n("padding-top") + self:n("padding-bottom") + self:n("border-top") +
        self:n("border-bottom")
end
function Element:border()
    return {self.x, self.y, self.cw + self:frame(true), self.ch + self:frame(false)}
end
function Element:content()
    return {
        self.x + self:n("border-left") + self:n("padding-left"),
        self.y + self:n("border-top") + self:n("padding-top"),
        self.cw,
        self.ch
    }
end
