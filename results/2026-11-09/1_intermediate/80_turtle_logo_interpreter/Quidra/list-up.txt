class Flow(nat kind := 0, value := 0)
class OutputValue(bool has_value, int value)
class LogoEnvironment(string procedure_name)
    variables := map<string, int>()
    environments := [LogoEnvironment("")]
    int | error lookup(Node node)
        for: #environments > i >= 0
            match environments[i].variables.get(node.text)
                int value
                    return value
                none
                    void
        return logo_error(node.line, node.text + " has no value")

    void .make_variable(string name, int value)
        for: #environments > i >= 1
            match environments[i].variables.get(name)
                int old
                    .environments[i].variables.set(name, value)
                    return
                none
                    void
        .environments[0].variables.set(name, value)
    OutputValue | error .call_procedure(Node call)
        index := procedure_index(program.definitions, call.text)
        procedure := program.definitions[index]
        int[] values := []
        for argument := call.children[]
            values = values.append(try .evaluate(argument))
        scope := LogoEnvironment(procedure.name)
        for: i < #values
            .scope.variables.set(procedure.parameters[i], values[i])
        environments = environments.append(scope)
        Flow outcome := try .run_list(procedure.body)
        environments = environments[:<#environments - 1]
        return OutputValue(outcome.kind == 2, outcome.value)
    int | error .evaluate(nat index)
        node := program.pool[index]
        if node.kind == 1
            return lookup(node)
    Flow | error .run_list(nat index)
        for command := program.pool[index].children[]
            Flow signal := try .execute(command)
            if signal.kind != 0
                return signal
        return Flow()
    Flow | error .execute(nat index)
        command := program.pool[index]
        name := command.text
        inputs := command.children
        if procedure_index(program.definitions, name) >= 0
            OutputValue result := try .call_procedure(command)
        elif name == "WHILE"
            while (try .evaluate(inputs[0])) != 0
                Flow signal := try .run_list(inputs[1])
                if signal.kind != 0
                    return signal
        elif name == "OUTPUT"
            return Flow(2, try .evaluate(inputs[0]))
        elif name == "MAKE" or name == "LOCALMAKE" or name == "LOCAL"
            variable := program.pool[inputs[0]].text
            int value := if name == "LOCAL" then 0 else try .evaluate(inputs[1])
            if name == "MAKE"
                .make_variable(variable, value)
            else
                .environments[-1].variables.set(variable, value)
