import math
bool digits(string value, nat limit := 9)
    if #value == 0 or #value > limit
        return false
    for: k < #value
        if not "0123456789".contains(value[k])
            return false
    return true
int time_value(string value)
    if #value != 12
        return -1
    if value[2] != ":" or value[5] != ":" or value[8] != ","
        return -1
    raw := value[:<2] + value[3:<5] + value[6:<8] + value[9:]
    if not digits(raw)
        return -1
    hours := int(value[:<2])
    minutes := int(value[3:<5])
    seconds := int(value[6:<8])
    if minutes > 59 or seconds > 59
        return -1
    return hours * 3600000 + minutes * 60000 + seconds * 1000 + int(value[9:])
string time_text(int value, style := "SRT")
    separator := if style == "VTT" then "." else ","
    return "{value / 3600000:02.}:{value / 60000 % 60:02.}:{value / 1000 % 60:02.}{separator}{value % 1000:03.}"
class Cue(int index, int start, int end, string[] lines)
bool dialogue(const Cue cue)
    for line := cue.lines[]
        if line.starts_with("- ")
            return true
    return false
string[] words_of(string[] lines)
    return [for word := lines.join(" ").split(" ")[] if word != "" then word]
string[] wrap(string[] lines, nat width)
    string[] result := []
    for word := words_of(lines)[]
        if #result > 0 and #result[-1] + 1 + #word <= width
            result[-1] += " " + word
        else
            result = result.append(word)
    return result
string[] schema(string command)
    if command == "GAPS" or command == "FIXOVERLAP" or command == "EXTEND" or command == "MERGE" or command == "SPLIT"
        return ["unsigned"]
    return []
string[] execute(string input_line, Cue[] &cues, Cue[][] &history, string[] &history_names)
    fields := input_line.split(" ")
    command := fields[0]
    known := " LIST VALIDATE SEARCH AT GAPS STATS PRINT TOVTT TOASS FRAMES SHIFT SHIFTFROM SCALE SYNC SNAP FIXOVERLAP EXTEND TIME SORT RENUMBER MERGE SPLIT INSERT DELETE TEXT BALANCE STRIP CLEANHI REPLACE UNDO "
    if not known.contains(" {command} ")
        return ["ERROR unknown command"]
    kinds := schema(command)
    string_tail := command == "SEARCH" or command == "INSERT" or command == "TEXT" or command == "REPLACE"
    if #fields < #kinds + 1 or (not string_tail and #fields != #kinds + 1)
        return ["ERROR bad arguments"]
    int[] values := []
    for: k < #kinds
        argument := if string_tail and k + 1 == #kinds then fields[k + 1:].join(" ") else fields[k + 1]
        kind := kinds[k]
        value := 0
        if kind == "time"
        elif kind == "fps"
        elif kind == "text"
        elif kind != "string"
            magnitude := if kind == "signed" and argument.starts_with("-") then argument[1:] else argument
            if not digits(magnitude)
                return ["ERROR bad arguments"]
            value = int(magnitude)
            if argument.starts_with("-")
                value = -value
        values = values.append(value)
    if command == "GAPS" or command == "FIXOVERLAP" or command == "MERGE"
        if values[0] > 60000
            return ["ERROR bad arguments"]
    string[] output := []
    if command == "PRINT" or command == "TOVTT" or command == "TOASS"
        for cue := cues[]
            lines := cue.lines
            if command != "PRINT"
            if command == "TOASS"
            elif #lines > 0
                settings := ""
                style := if command == "TOVTT" then "VTT" else "SRT"
                output = output.append("{cue.index}").append("{time_text(cue.start, style)} --> {time_text(cue.end, style)}{settings}").concat(lines).append("")
        return output
    if command == "MERGE"
        Cue[] merged := []
        joins := 0
        for cue := cues[]
            if #merged > 0
                &previous := &merged[-1]
                if cue.start - previous.end < values[0] and not dialogue(previous) and not dialogue(cue)
                    lines := wrap(previous.lines.concat(cue.lines), 42)
                    if #lines <= 2
                        previous.end = math.max(previous.end, cue.end)
                        previous.lines = lines
                        joins += 1
                        continue
            merged = merged.append(cue)
        cues = merged
        return ["MERGE merged={joins} cues={#cues}"]
void load_block(string[] block, int number, Cue[] &cues, int &bad)
    problem := ""
    if not digits(block[0], 6)
        problem = "index"
    elif #block < 2
        problem = "time"
    else
        value := block[1]
        if #value != 29
            problem = "time"
        elif value[12:<17] != " --> " or time_value(value[:<12]) < 0 or time_value(value[17:]) < 0
            problem = "time"
        elif #block < 3
            problem = "no text"
    if problem != ""
        bad += 1
        println("BAD CUE {number}: {problem}")
    else
        cues = cues.append(Cue(int(block[0]), time_value(block[1][:<12]), time_value(block[1][17:]), block[2:]))
Cue[] cue_list := []
string[] pending_block := []
block_number := 0
bad_count := 0
while true
    string input_line
    scan(&input_line)
    if input_line == "" or input_line == "###"
        if #pending_block > 0
            block_number += 1
            load_block(pending_block, block_number, &cue_list, &bad_count)
            pending_block = []
        if input_line == "###"
            break
    else
        pending_block = pending_block.append(input_line)
println("LOAD cues={#cue_list} bad={bad_count}")
