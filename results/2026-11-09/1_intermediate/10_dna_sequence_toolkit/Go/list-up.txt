func fixed(value, places int) string {
	scale := 1
	for i := 0; i < places; i++ {
		scale *= 10
	}
	return fmt.Sprintf("%d.%0*d", value/scale, places, value%scale)
func pct(part, whole int) string {
	return fixed(10000*part/whole, 2)
