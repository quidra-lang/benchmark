function Cron:matchDay(date)
    local last, day, weekday = monthLength(date.year, date.month), date.day, date.weekday
    local monthDay, weekDay = self.fields[3].set[day] == true, self.fields[5].set[weekday] == true
    for _, special in ipairs(self.fields[3].specials) do
        local matches
        if special.kind == "last" then
            matches = day == last
        elseif special.kind == "lastWeekday" then
            matches = day == nearest(date, last)
        elseif special.kind == "beforeLast" then
            matches = day == last - special.value
        else
            matches = special.value <= last and day == nearest(date, special.value)
        end
        monthDay = monthDay or matches
    end
    for _, special in ipairs(self.fields[5].specials) do
        local matches
        if special.kind == "nthWeekday" then
            matches = weekday == special.value and (day - 1) // 7 + 1 == special.occurrence
        else
            matches = weekday == special.value and day + 7 > last
        end
        weekDay = weekDay or matches
    end
    local condition
    if self:orRule() then
        condition = monthDay or weekDay
    else
        condition = monthDay and weekDay
    end
    return monthDay, weekDay, condition
end
function Cron:scheduled(date)
    local _, _, condition = self:matchDay(date)
    return self.fields[4].set[date.month] == true and condition
