int point(int card)
    return math.min(card % 13 + 1, 10)

string card_text(int card)
    return "A23456789TJQK"[card % 13] + "CDHS"[card / 13]

Total hand_total(const Hand hand)
    value := [for card := hand.cards[] then point(card)].sum()
    soft := contains([for card := hand.cards[] then card % 13], 0) and value + 10 <= 21
    return Total(value + (if soft then 10 else 0), soft)

bool blackjack(const Hand hand)
    return not hand.split and #hand.cards == 2 and hand_total(hand).value == 21

string hand_text(const Hand hand)
    score := hand_total(hand)
    status := string(score.value)
    if blackjack(hand)
        status = "blackjack"
    elif score.value > 21
        status = "bust {score.value}"
    elif score.soft
        status = "soft {score.value}"
    cards := [for card := hand.cards[] then card_text(card)].join(" ")
    return "{cards} ({status})"
