fn money(n: Integer) -> String {
    format!("{}.{:02}", n / 100, n % 100)
}
fn signed_money(n: Integer) -> String {
    if n > 0 {
        format!("+{}", money(n))
    } else if n < 0 {
        format!("-{}", money(-n))
    } else {
        "0.00".to_owned()
    }
