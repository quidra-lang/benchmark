using Integer = std::int64_t;
struct Params {
    Integer refractory_ms, twave_ms, searchback_percent, premature_percent, qrs_width_ms, match_ms, pause_ms,
        brady_ms, tachy_ms, excerpt_ms, segment_ms, strip_start_ms, strip_ms, strip_rows, chart_rows,
        poincare_size;
};
struct Annotation {
    int sample;
    char label;
    int beat = -1;
};
struct Input {
    Values ecg;
    std::vector<Annotation> annotations;
    Params params;
};
Input read_input() {
    Input out;
    std::ifstream ecg("input/ecg.csv");
    std::string line;
    std::getline(ecg, line);
    while (std::getline(ecg, line))
        out.ecg.push_back(std::stoll(line.substr(line.find(',') + 1)));
    std::ifstream annotations("input/annotations.csv");
    std::getline(annotations, line);
    while (std::getline(annotations, line)) {
        const std::size_t comma = line.find(',');
        out.annotations.push_back({std::stoi(line.substr(0, comma)), line[comma + 1], -1});
    }
    std::ifstream params("input/params.txt");
    Integer *fields[] = {
        &out.params.refractory_ms,     &out.params.twave_ms,     &out.params.searchback_percent,
        &out.params.premature_percent, &out.params.qrs_width_ms, &out.params.match_ms,
        &out.params.pause_ms,          &out.params.brady_ms,     &out.params.tachy_ms,
        &out.params.excerpt_ms,        &out.params.segment_ms,   &out.params.strip_start_ms,
        &out.params.strip_ms,          &out.params.strip_rows,   &out.params.chart_rows,
        &out.params.poincare_size};
    std::string key;
    for (Integer *field : fields)
        params >> key >> *field;
    return out;
}
struct Beat {
    int peak = 0, sample = 0, reference = -1;
    Integer rr = 0, width = 0;
    char label = 'N';
};
static Detection detect(const Signals &signals, const Params &params) {
    Detection out;
    out.peaks = find_peaks(signals);
    out.initial = out.levels = initial_levels(signals);
    int last = -1;
    for (std::size_t i = 0; i < out.beats.size(); ++i) {
        Beat &beat = out.beats[i];
        const Peak &peak = out.peaks[beat.peak];
        beat.sample = std::max(0, peak.r - 21);
        beat.width = qrs_width(signals, peak.r);
        if (i)
            beat.rr = 5 * (beat.sample - out.beats[i - 1].sample);
        if (i >= 9) {
            Integer preceding = 0;
            for (std::size_t j = i - 8; j < i; ++j)
                preceding += out.beats[j].rr;
            if (800 * beat.rr < params.premature_percent * preceding)
                beat.label = beat.width > params.qrs_width_ms ? 'V' : 'A';
        }
    }
    return out;
}
static void match(std::vector<Beat> &beats, std::vector<Annotation> &annotations, Integer tolerance) {
    for (std::size_t k = 0; k < beats.size(); ++k) {
        int selected = -1;
        Integer distance = 0;
        for (std::size_t i = 0; i < annotations.size(); ++i) {
            const Integer difference = std::abs(Integer(annotations[i].sample) - beats[k].sample);
            if (annotations[i].beat < 0 && difference <= tolerance &&
                (selected < 0 || difference < distance)) {
                selected = i;
                distance = difference;
            }
        }
        if (selected >= 0) {
            beats[k].reference = selected;
            annotations[selected].beat = k;
        }
    }
}
static Lines rhythm(const Input &input, const Detection &detection) {
    const Params &p = input.params;
    const int first = p.strip_start_ms / 5, length = p.strip_ms / 5, columns = length / 10;
    std::string beat_marks(columns, ' '), ref_marks(columns, ' ');
    for (const Beat &beat : detection.beats)
        if (beat.sample >= first && beat.sample < first + length)
            beat_marks[(beat.sample - first) / 10] = beat.label;
    for (const Annotation &annotation : input.annotations)
        if (annotation.sample >= first && annotation.sample < first + length)
            ref_marks[(annotation.sample - first) / 10] = annotation.label;
    out.push_back(" beats |" + beat_marks + "|");
    out.push_back("  refs |" + ref_marks + "|");
    return out;

Integer floor_div(Integer a, Integer b) {
    Integer quotient = a / b;
    if (a % b < 0)
        --quotient;
    return quotient;
}
Integer rdiv(Integer a, Integer b) {
    return floor_div(2 * a + b, 2 * b);
}
std::string decimal2(Integer value) {
    const Integer magnitude = std::abs(value);
    std::ostringstream out;
    if (value < 0)
        out << '-';
    out << magnitude / 100 << '.' << std::setw(2) << std::setfill('0') << magnitude % 100;
    return out.str();
}
std::string reduced(Integer a, Integer b) {
    const Integer divisor = std::gcd(a, b);
    return std::to_string(a / divisor) + "/" + std::to_string(b / divisor);
}
std::string ratio(Integer a, Integer b) {
    const std::string prefix = std::to_string(a) + "/" + std::to_string(b);
    return b ? prefix + " reduced=" + reduced(a, b) + " percent=" + decimal2(rdiv(10000 * a, b))
             : prefix + " none";
}
