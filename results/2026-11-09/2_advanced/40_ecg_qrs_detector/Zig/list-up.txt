pub const Params = struct { refractory_ms: i64 = 0, twave_ms: i64 = 0, searchback_percent: i64 = 0,
     premature_percent: i64 = 0, qrs_width_ms: i64 = 0, match_ms: i64 = 0, pause_ms: i64 = 0, brady_ms: i64 = 0,
     tachy_ms: i64 = 0, excerpt_ms: i64 = 0, segment_ms: i64 = 0, strip_start_ms: i64 = 0, strip_ms: i64 = 0,
     strip_rows: i64 = 0, chart_rows: i64 = 0, poincare_size: i64 = 0 };
pub const Annotation = struct { sample: usize, label: u8, beat: ?usize = null };
pub const Input = struct { ecg: []const i64, annotations: []Annotation, params: Params };
pub fn readInput(io: std.Io) !Input {
    var ecg: std.ArrayList(i64) = .empty;
    const ecg_lines = try readLines(io, "input/ecg.csv");
    for (ecg_lines[1..]) |line| append(i64, &ecg, number(split(line, ',')[1]));
    var annotations: std.ArrayList(Annotation) = .empty;
    const annotation_lines = try readLines(io, "input/annotations.csv");
    for (annotation_lines[1..]) |line| {
        const fields = split(line, ',');
        append(Annotation, &annotations, .{ .sample = u(number(fields[0])), .label = fields[1][0] });
    }
    var params: Params = .{};
    for (try readLines(io, "input/params.txt")) |line| {
        const fields = split(line, ' ');
        inline for (std.meta.fields(Params)) |field| if (eq(fields[0], field.name)) {
            @field(params, field.name) = number(fields[1]);
        };
    }
    return .{ .ecg = ecg.items, .annotations = annotations.items, .params = params };
}

fn match(beats: []Beat, annotations: []c.Annotation, tolerance: i64) void {
    for (beats, 0..) |*beat, k| {
        var selected: ?usize = null;
        var distance: i64 = 0;
        for (annotations, 0..) |annotation, i| {
            const difference = detector.abs(c.n(annotation.sample) - c.n(beat.sample));
            if (annotation.beat == null and difference <= tolerance and (selected == null or difference < distance)) {
                selected = i;
                distance = difference;
            }
        }
        if (selected) |index| {
            beat.reference = index;
            annotations[index].beat = k;
        }
    }
}
fn rhythm(input: c.Input, detection: Detection) c.Lines {
    const p = input.params;
    const first = c.u(@divFloor(p.strip_start_ms, 5));
    const length = c.u(@divFloor(p.strip_ms, 5));
    const columns = length / 10;
    const beat_marks = repeat(' ', columns);
    const ref_marks = repeat(' ', columns);
    for (detection.beats.items) |beat| if (beat.sample >= first and beat.sample < first + length) {
        beat_marks[(beat.sample - first) / 10] = beat.label;
    };
    for (input.annotations) |annotation| if (annotation.sample >= first and annotation.sample < first + length) {
        ref_marks[(annotation.sample - first) / 10] = annotation.label;
    };
    out.add(" beats |{s}|", .{beat_marks});
    out.add("  refs |{s}|", .{ref_marks});
    return out;
const OrderedLine = struct { sample: usize, priority: usize, line: []const u8 };
fn orderedLess(_: void, a: OrderedLine, b: OrderedLine) bool {
    return if (a.sample != b.sample) a.sample < b.sample else a.priority < b.priority;
}

pub fn rdiv(a: i64, b: i64) i64 {
    return @divFloor(2 * a + b, 2 * b);
}
pub fn decimal2(value: i64) []const u8 {
    const magnitude = @abs(value);
    return c.fmt("{s}{d}.{d:0>2}", .{ if (value < 0) "-" else "", magnitude / 100, magnitude % 100 });
}
pub fn reduced(a: i64, b: i64) []const u8 {
    var x = a;
    var y = b;
    while (y != 0) {
        const remainder = @mod(x, y);
        x = y;
        y = remainder;
    }
    return c.fmt("{d}/{d}", .{ @divFloor(a, x), @divFloor(b, x) });
}
pub fn ratio(a: i64, b: i64) []const u8 {
    return if (b == 0) c.fmt("{d}/{d} none", .{ a, b }) else c.fmt("{d}/{d} reduced={s} percent={s}",
         .{ a, b, reduced(a, b), decimal2(rdiv(10000 * a, b)) });
}
