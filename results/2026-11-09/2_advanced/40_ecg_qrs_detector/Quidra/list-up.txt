class Beat(int number, int sample, int rr, int hr, int width, string label, string method)
    annotation := -1
    offset := 0

Beat[] beat_list(int[] registered, detector.Peak[] peaks, int[] band, int premature_percent, int width_limit)
    Beat[] beats := []
    for peak_index := registered[]
        peak := peaks[peak_index]
        number := int(#beats) + 1
        sample := if peak.position < 21 then 0 else peak.position - 21
        rr := if number > 1 then 5 * (sample - beats[-1].sample) else -1
        heart_rate := if rr > 0 then report.rdiv(6000000, rr) else -1
        width := detector.width(band, peak.position)
        label := "N"
        if number >= 10
            preceding := 0
            for: number - 8 <= previous < number
                preceding += beats[previous - 1].rr
            if 800 * rr < premature_percent * preceding
                label = if width > width_limit then "V" else "A"
        beats = beats.append(Beat(number, sample, rr, heart_rate, width, label, if peak.category == "qrs" then "threshold" else "searchback"))
    return beats

void match_annotations(Beat[] &beats, io.Annotation[] &refs, int tolerance)
    for: index < int(#beats)
        beat := beats[index]
        chosen := -1
        distance := 0
        for: reference < int(#refs)
            delta := absolute(refs[reference].sample - beat.sample)
            if refs[reference].beat < 0 and delta <= tolerance and (chosen < 0 or delta < distance)
                chosen = reference
                distance = delta
        if chosen >= 0
            beat.annotation = chosen
            beat.offset = 5 * (beat.sample - refs[chosen].sample)
            refs[chosen].beat = index
            beats[index] = beat
class Segment(int start, int stop)
    refs := 0
    tp := 0
    fp := 0
    false_negative := 0
    noise := 0
    beats := 0
    int[] rr := []

Segment[] segments(Beat[] beats, io.Annotation[] refs, detector.Peak[] peaks, int count, int segment_ms)
    width := segment_ms / 5
    Segment[] result := []
    for: start < count step width
        stop := if start + width < count then start + width else count
        segment := Segment(5 * start, 5 * stop)
        for reference := refs[]
            if reference.sample >= start and reference.sample < stop
                segment.refs += 1
                if reference.beat >= 0
                    segment.tp += 1
                else
                    segment.false_negative += 1
        for beat := beats[]
            if beat.sample >= start and beat.sample < stop
                segment.beats += 1
                if beat.annotation < 0
                    segment.fp += 1
                if beat.rr >= 0
                    segment.rr = segment.rr.append(beat.rr)
        for peak := peaks[]
            if peak.moment >= start and peak.moment < stop and peak.category == "noise"
                segment.noise += 1
        result = result.append(segment)
    return result

string[] beat_csv(Beat[] beats, io.Annotation[] refs)
    result := ["beat,sample,time_ms,rr_ms,hr_bpm,width_ms,label,method,ref_sample,ref_label"]
    for beat := beats[]
        ref_sample := if beat.annotation < 0 then "-" else "{refs[beat.annotation].sample}"
        ref_label := if beat.annotation < 0 then "-" else refs[beat.annotation].label
        heart_rate := if beat.hr < 0 then "-" else report.two(beat.hr)
        fields := ["{beat.number}", "{beat.sample}", "{5 * beat.sample}", report.missing(beat.rr), heart_rate, "{beat.width}", beat.label, beat.method, ref_sample, ref_label]
        result = result.append(fields.join(","))
    return result
class OrderedLine(int sample, int priority, string text)

string[] ordered_lines(OrderedLine[] records)
    OrderedLine[] ordered := []
    for record := records[]
        position := int(#ordered)
        ordered = ordered.append(record)
        while position > 0 and (record.sample < ordered[position - 1].sample or (record.sample == ordered[position - 1].sample and record.priority < ordered[position - 1].priority))
            ordered[position] = ordered[position - 1]
            position -= 1
        ordered[position] = record
    return [for record := ordered[] then record.text]

string[] matching_csv(Beat[] beats, io.Annotation[] refs)
    OrderedLine[] records := []
    for reference := refs[]
        line := "{reference.sample},{reference.label},-,-,-,-,FN,-"
        if reference.beat >= 0
            beat := beats[reference.beat]
            line = "{reference.sample},{reference.label},{beat.number},{beat.sample},{beat.label},{beat.offset},TP," + yes(reference.label == beat.label)
        records = records.append(OrderedLine(reference.sample, 0, line))
    for beat := beats[]
        if beat.annotation < 0
            records = records.append(OrderedLine(beat.sample, 1, "-,-,{beat.number},{beat.sample},{beat.label},-,FP,-"))
    return ["ref_sample,ref_label,beat,beat_sample,beat_label,offset_ms,status,agree"].concat(ordered_lines(records))
string[] label_names()
    return ["N", "V", "A"]

int label_index(string label)
    return if label == "N" then 0 elif label == "V" then 1 else 2

    detected := [0, 0, 0]
    int[][] confusion := array(3, fill = array(4, fill = 0))
    extras := [0, 0, 0]
    method_total := [0, 0]
    method_tp := [0, 0]
    int[] offsets := []
    OrderedLine[] errors := []
    tp := 0
    fp := 0
    for reference := refs[]
        category := label_index(reference.label)
        types[category] += 1
        if reference.beat >= 0
            tp += 1

        else
            confusion[category][3] += 1
            errors = errors.append(OrderedLine(reference.sample, 1, "false_negative sample={reference.sample} time_ms={5 * reference.sample} label={reference.label}"))
    for beat := beats[]
        category := label_index(beat.label)
        method := if beat.method == "threshold" then 0 else 1
        detected[category] += 1
        method_total[method] += 1
        if beat.annotation >= 0
            method_tp[method] += 1
            offsets = offsets.append(beat.offset)
        else
            fp += 1
            extras[category] += 1
            errors = errors.append(OrderedLine(beat.sample, 0, "false_positive beat={beat.number} sample={beat.sample} time_ms={5 * beat.sample} label={beat.label}"))
    false_negative := int(#refs) - tp
        "MATCHING",
        "tolerance_ms {params.match_ms}",
        "references {#refs} N={types[0]} V={types[1]} A={types[2]}",
        "detections {#beats} N={detected[0]} V={detected[1]} A={detected[2]}",
        "true_positives {tp}",

        "positive_predictivity " + report.ratio(tp, int(#beats)),
        "detection_error_rate " + report.ratio(fp + false_negative, int(#refs)),

    else

                absolute_max = absolute(offset)
        result = result.append("offset_ms mean=" + report.two(report.rdiv(100 * hrv.sum_values(offsets), int(#offsets))) + " mean_abs=" + report.two(report.rdiv(100 * absolute_sum, int(#offsets))) + " max_abs={absolute_max}")
    result = result.concat(["CONFUSION", "ref  " + pad("N", 6) + pad("V", 6) + pad("A", 6) + "  miss"])
    names := label_names()
    for: category < 3

        for value := confusion[category][]
            line += pad("{value}", 6)
        result = result.append(line)
    line := "extra"
    for value := extras[]
        line += pad("{value}", 6)
    result = result.append(line)
    for: 1 <= category < 3
        prefix := if category == 1 then "veb" else "sveb"
        result = result.append(prefix + "_sensitivity " + report.ratio(confusion[category][category], types[category]))
        result = result.append(prefix + "_positive_predictivity " + report.ratio(confusion[category][category], detected[category]))

    for part := parts[]

int floor_div(int numerator, int denominator)
    quotient := numerator / denominator
    return if numerator % denominator < 0 then quotient - 1 else quotient

int rdiv(int numerator, int denominator)
    return floor_div(2 * numerator + denominator, 2 * denominator)

string two(int value)
    magnitude := if value < 0 then -value else value
    whole := magnitude / 100
    decimal := magnitude % 100
    return (if value < 0 then "-" else "") + "{whole}.{decimal:02.}"
string fraction(int numerator, int denominator)
    first := numerator
    second := denominator
    while second != 0
        first, second = second, first % second
    return "{numerator / first}/{denominator / first}"

string ratio(int numerator, int denominator)
    if denominator == 0
        return "{numerator}/0 none"
    return "{numerator}/{denominator} reduced=" + fraction(numerator, denominator) + " percent=" + two(rdiv(10000 * numerator, denominator))
