update_levels =: 4 : 0
    peak =. y{Peaks

        values =. ((2 3{peak)+7*1 3{Levels)intdiv 8
        Levels =: values 1 3}Levels

search_back =: 3 : 0
    while. 1 do.
        last =. 0
        if. #Registered do.
            last =. (<({:Registered),0){Peaks
        end.
        if. (y-last)<:(RA2*SearchbackPercent)intdiv 100 do.
            break.
        end.
        current =. thresholds Levels
        chosen =. _1
        best =. _1
        for_index. i.#Peaks do.
            'moment position peak_i peak_f slope thr_i thr_f category' =. index{Peaks
            if. (category=4)*.(moment>last)*.(moment<y)*.(peak_i>2{current)*.peak_f>3{current do.
                if. peak_i>best do.
                    chosen =. index
                    best =. peak_i
                end.
            end.
        end.
        if. chosen<0 do.
            break.
        end.
        2 update_levels chosen
        Peaks =: 1 (<chosen,7)}Peaks
        register_peak chosen
    end.
    0
)
detect =: 3 : 0
    InitialLevels =: initial_levels Signals
    Levels =: InitialLevels
    Registered =: i.0
    Intervals =: i.0
    RA1 =: 200
    RA2 =: 200
    for_index. i.#Peaks do.
        moment =. (<index,0){Peaks
        search_back moment
        current =. thresholds Levels
        Peaks =: (0{current)(<index,5)}Peaks
        Peaks =: (1{current)(<index,6)}Peaks
        peak =. index{Peaks
        last =. _1
        if. #Registered do.
            last =. {:Registered
        end.
        category =. 4
        update =. 1
        register =. 0
        if. last>:0 do.
            if. (moment-(<last,0){Peaks)<Refractory do.
                category =. 3
                update =. _1
            end.
        end.
        if. category~:3 do.
            if. ((2{peak)>0{current)*.(3{peak)>1{current do.
                twave =. 0
                if. last>:0 do.
                    twave =. ((moment-(<last,0){Peaks)<Twave)*.(2*4{peak)<(<last,4){Peaks
                end.
                if. twave do.
                    category =. 2
                    update =. 1
                else.
                    category =. 0
                    update =. 0
                    register =. 1
                end.
            end.
        end.
        Peaks =: category (<index,7)}Peaks
        if. update>:0 do.
            update update_levels index
        end.
        if. register do.
            register_peak index
        end.
    end.
    search_back #Samples
    0
)
build_beats =: 3 : 0
    Beats =: 0 9$0x
    for_peak_index. Registered do.
        peak =. peak_index{Peaks
        number =. 1+#Beats
        sample =. 0>.(1{peak)-21
        rr =. _1
        hr =. _1
        if. number>1 do.
            rr =. 5*sample-(<((#Beats)-1),1){Beats
            hr =. 6000000 rdiv rr
        end.
        width =. (1{"1 Signals)width_ms 1{peak
        label =. 0
        if. number>:10 do.
            previous =. (number-9)+i.8
            preceding =. +/2{"1 previous{Beats
            if. (800*rr)<PrematurePercent*preceding do.
                label =. 2
                if. width>WidthMs do.
                    label =. 1
                end.
            end.
        end.

    end.
    0

match_annotations =: 3 : 0
    for_index. i.#Beats do.
        sample =. (<index,1){Beats
        chosen =. _1
        distance =. 0
        for_reference. i.#Annotations do.
            delta =. |((<reference,0){Annotations)-sample

                if. delta<:MatchMs intdiv 5 do.

            end.

            Beats =: chosen (<index,7)}Beats
            Beats =: (5*sample-(<chosen,0){Annotations)(<index,8)}Beats

    end.
    0

filter_ecg =: 3 : 0
    count =. #y
    cumulative =. count$0x
    moving =. count$0x
    low =. count$0x
    band =. count$0x
    derivative =. count$0x
    squared =. count$0x
    integrated =. count$0x
    window =. 0x
    for_index. i.count do.
        value =. ((2*cumulative at index-1)-cumulative at index-2)+index{y
        value =. (value-2*y at index-6)+y at index-12
        cumulative =. value index}cumulative
        low =. (value intdiv 32)index}low
        value =. ((moving at index-1)+index{low)-low at index-32
        moving =. value index}moving
        value =. (low at index-16)-value intdiv 32
        band =. value index}band
        value =. ((((2*value)+band at index-1)-band at index-3)-2*band at index-4)intdiv 8
        derivative =. value index}derivative
        squared =. (value*value)index}squared
        window =. (window+index{squared)-squared at index-30
        integrated =. (window intdiv 30)index}integrated
    end.
    low,.band,.derivative,.squared,.integrated
)
find_peaks =: 3 : 0
    band =. 1{"1 y
    derivative =. 2{"1 y
    integrated =. 4{"1 y
    result =. 0 8$0x
    count =. #integrated
    for_moment. i.count do.
        value =. moment{integrated
        if. value<:0 do.
            continue.
        end.
        first =. 0>.moment-20
        previous =. (first+i.moment-first){integrated
        last =. (count-1)<.moment+20
        following =. ((moment+1)+i.last-moment){integrated
        if. (+./previous>:value)+.+./following>value do.
            continue.
        end.
        position =. 0>.moment-30
        for_index. (position+1)+i.moment-position do.
            if. (|index{band)>|position{band do.
                position =. index
            end.
        end.
        first =. 0>.moment-29
        slope =. >./|(first+i.(moment-first)+1){derivative
        result =. result,moment,position,value,(|position{band),slope,0 0 _1
    end.
    result
)
