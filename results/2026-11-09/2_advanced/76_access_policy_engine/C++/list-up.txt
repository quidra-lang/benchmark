        if (coverage)
            (*coverage)[element->index] = result->decision == Decision::Permit ? 'P'
                                          : result->decision == Decision::Deny ? 'D'
                                          : result->decision == Decision::NA   ? '-'
                                                                               : 'I';
        return result;
