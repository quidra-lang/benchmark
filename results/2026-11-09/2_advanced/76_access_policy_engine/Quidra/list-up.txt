        for: index < int(#engine.elements)
            if engine.elements[index].kind != "rule"
                continue
            symbol := "."
            if checked.error_message == "" and checked.evaluations[index].state == 2
                string decision := checked.evaluations[index].decision.name
                symbol = if decision == "PERMIT" then "P" elif decision == "DENY" then "D" elif decision == "NOT_APPLICABLE" then "-" else "I"
            engine.coverage[index] += symbol
