const na = "NOT_APPLICABLE"
func evaluate(e *Element, attributes map[string]Value, coverage map[*Element]string) *Evaluation {
	var r *Evaluation
	if e.kind == "rule" {
		r = ruleResult(e,
			attributes)
		attachments(r, attributes)
		if coverage != nil {
			symbol := "I"
			switch r.decision {
			case "PERMIT":
				symbol = "P"
			case "DENY":
				symbol = "D"
			case na:
				symbol = "-"
			}
			coverage[e] = symbol
		}
		return r
	}
