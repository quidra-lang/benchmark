    fun flow(s: Heat.Thermal,x: DoubleArray,p: Int,first: Int,last: Int): Double {
        var sum=0.0
        for(f in first..last) {
            val face=s.faces[p][f];
            if(face.neighbor>=0)sum+=face.g*(x[face.neighbor]-x[p]) else if(face.boundary)sum+=face.g*(face.theta-x[p])
