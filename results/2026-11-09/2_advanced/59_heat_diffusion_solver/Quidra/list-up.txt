real64 flow(int[][] neighbors, real64[][] conductance, real64[][] theta, bool[][] boundary, real64[] field, int point, int first, int last)
    real64[] terms := []
    for: first <= direction <= last
        neighbor := neighbors[point][direction]
        g := conductance[point][direction]
        if neighbor >= 0
            terms = terms.append(g * (field[neighbor] - field[point]))
real64[][] advance(string method, int width, int height, int[][] neighbors, real64[][] conductance, real64[][] theta, bool[][] boundary, real64[] capacity, real64[] field, real64[] source, real64 dt)
    count := width * height
    real64[] rhs := []
    if method == "ftcs"
        for: p < count
            rhs = rhs.append(field[p] + dt * (flow(neighbors, conductance, theta, boundary, field, p, 0, 3) + source[p]) / capacity[p])
        return [rhs, []]
