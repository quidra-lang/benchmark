function ftcs(object $model, array $old, array $source, float $dt): object
{
    $field = [];
    for ($p = 0; $p < $model->count; $p++) $field[] = $old[$p] + $dt * (totalFlow($model, $old, $p) + $source[$p]) / $model->capacities[$p];
