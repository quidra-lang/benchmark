func step(plate *Plate, old, source []float64, dt float64, method string) ([]float64, []float64) {
	n := len(old)
	rhs := make([]float64, n)
	if method == "ftcs" {
		for p := range old {
			rhs[p] = old[p] + dt*(flows(plate, old, p, 0, 3)+source[p])/plate.C[p]
		}
		return rhs, nil
