export function ftcs(model: ThermalModel, old: number[], source: number[], dt: number): Step {
  return {
    field: old.map(
      (value, p) => value + (dt * (totalFlow(model, old, p) + source[p])) / model.capacities[p],
    ),
    intermediate: null,
  };
