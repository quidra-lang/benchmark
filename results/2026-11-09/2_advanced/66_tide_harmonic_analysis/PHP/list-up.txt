
function level($model, $time) {
    $value = $model->mean;
    foreach ($model->constituents as $constituent) {
        $theta = angle($constituent->speed, $time);
        $value += $constituent->a * cos($theta) + $constituent->b * sin($theta);
    }
    return $value;
}
