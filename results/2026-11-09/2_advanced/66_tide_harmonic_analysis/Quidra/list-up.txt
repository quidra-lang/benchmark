real64 height(real64[] speeds, real64[] coefficients, real64 hour)
    result := coefficients[0]
    for: k < int(#speeds)
        theta := harmonic_fit.angle(speeds[k], hour)
        result += coefficients[2 * k + 1] * math.cos(theta) + coefficients[2 * k + 2] * math.sin(theta)
    return result

real64[] hours(real64[] speeds, real64[] coefficients, int start, int count)
    return [for i := (0:<(int(count) + 1))[] then height(speeds, coefficients, real64(start + i))]

Extremum[] grid_extrema(real64[] speeds, real64[] coefficients, int start, int duration, int spacing)
    count := duration * 60 / spacing
    minutes := [for i := (0:<(int(count) + 1))[] then 60 * start + i * spacing]
    heights := [for minute := minutes[] then height(speeds, coefficients, real64(minute) / 60.0)]
    Extremum[] extrema := []
    for: 1 <= j < count
        left := heights[j - 1]
        center := heights[j]
        right := heights[j + 1]
        high := left < center and center > right
        low := left > center and center < right
        if not high and not low
            continue
        delta := (left - right) / (2.0 * (left - 2.0 * center + right))
        minute := real64(minutes[j]) + delta * real64(spacing)
        level := center - (left - right) * delta / 4.0
        extrema = extrema.append(Extremum(high, minute, level))
    return extrema
