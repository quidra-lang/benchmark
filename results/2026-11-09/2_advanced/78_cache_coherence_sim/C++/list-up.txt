void Cache::touch_way(int row, int way, bool fill) {
    CacheSet &set = sets[row];
    if (replacement == "LRU" || (replacement == "FIFO" && fill))
        set.touched[way] = ++clock;
    else if (replacement == "PLRU") {
        int lo = 0, hi = ways, node = 1;
        while (hi - lo > 1) {
            int mid = (lo + hi) / 2;
            if (way < mid) {
                set.bits[node] = 1;
                hi = mid;
                node *= 2;
            } else {
                set.bits[node] = 0;
                lo = mid;
                node = 2 * node + 1;
            }
        }
    }
}
void Cache::touch(Number block) {
    int way = locate(block);
    if (way >= 0)
        touch_way(int(block % sets.size()), way, false);
}
RequestTransition requester(const std::string &protocol, char old, char op, bool other) {
    using T = TransactionKind;
    if (protocol == "VI") {
        if (op == 'R')
            return {old == 'I' ? T::BusRd : T::None, 'V'};
        return {T::BusWr, old == 'I' ? 'I' : 'V'};
    }
    if (old == 'I') {
        if (op == 'W')
            return {T::BusRdX, 'M'};
        return {T::BusRd, protocol == "MSI" || other ? 'S' : 'E'};
    }
    if (op == 'R')
        return {T::None, old};
    if (old == 'M' || old == 'E')
        return {T::None, 'M'};
    return {T::BusUpgr, 'M'};
}
