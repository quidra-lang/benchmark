func (c *Cache) touch(set, way int, fill bool) {
	if c.policy == "FIFO" && !fill {
		return
	}
	c.clock++
	c.sets[set].ways[way].stamp = c.clock
	if c.policy == "PLRU" {
		node,
			lo, hi := 1, 0, len(c.sets[set].ways)
		for hi-lo > 1 {
			mid := (lo + hi) / 2
			if way < mid {
				c.sets[set].bits[node] = 1
				node *= 2
				hi = mid
			} else {
				c.sets[set].bits[node] = 0
				node = node*2 + 1
				lo = mid
			}
		}
	}
}
func (c *Cache) hit(b int) { set, way := c.find(b); c.touch(set, way, false) }
func requestTransition(protocol, state, operation string) (string, string) {
	if protocol == "VI" {
		if operation == "W" {
			return "BusWr",
				state
		}
		if state == "I" {
			return "BusRd", "V"
		}
		return "", state
	}
	if state == "I" {
		if operation == "W" {
			return "BusRdX",
				"M"
		}
		if protocol == "MSI" {
			return "BusRd", "S"
		}
		return "BusRd", "E/S"
	}
	if operation == "R" {
		return "",
			state
	}
	if state == "S" || state == "O" {
		return "BusUpgr", "M"
	}
	return "", "M"
}
