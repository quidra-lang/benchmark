void handle_access(Simulation &sim, Event event, int issue)
    core := event.core
    block := event.block
    before := [for watched := sim.config.watch[] then line_states(sim, watched)]
    old := cache.state(sim.l1[core], block)
    hit := old != "I"
    stats.add(&sim.cores[core], "accesses")
    stats.add(&sim.cores[core], if event.operation == "R" then "reads" else "writes")
    stats.add(&sim.cores[core], if hit then "hits" else "misses")
    others := false
    for: k < sim.config.cores
        if k != core and cache.state(sim.l1[k], block) != "I"
            others = true
    transition := protocol.requester(sim.scenario.protocol, old, event.operation, others)
    miss := ""
    sharing := ""
    if hit
        cache.touch(&sim.l1[core], block)
    completion := Completion(issue + sim.config.t1, "l1")
    if transition.transaction != ""
        completion = run_transaction(&sim, event, transition.transaction, issue + sim.config.t1)
    if old != "I"
        cache.change(&sim.l1[core], block, transition.state)
    latency := completion.tick - issue
    stats.add(&sim.cores[core], "latency", latency)
    stats.add(&sim.cores[core], if event.operation == "R" then "read_latency" else "write_latency", latency)
    stats.add(&sim.cores[core], completion.service)
    sim.cores[core].finish = math.max(sim.cores[core].finish, completion.tick)
    if event.operation == "R"
        sim.ready[core] = completion.tick
    sim.accesses = sim.accesses.append(Access(event, issue, latency, completion.service, hit, miss))
void touch_way(Cache &cache, int row, int way, bool fill)
    if cache.replacement == "LRU" or (cache.replacement == "FIFO" and fill)
        cache.clock += 1
        cache.touched[row][way] = cache.clock
    elif cache.replacement == "PLRU"
        low := 0
        high := cache.ways
        node := 1
        while high - low > 1
            middle := (low + high) / 2
            if way < middle
                cache.bits[row][node] = 1
                high = middle
                node *= 2
            else
                cache.bits[row][node] = 0
                low = middle
                node = 2 * node + 1
void touch(Cache &cache, int block)
    way := locate(cache, block)
    if way >= 0
        touch_way(&cache, block % cache.sets, way, false)
Request requester(string protocol, string old, string operation, bool others)
    if protocol == "VI"
        if operation == "R"
            return Request(if old == "I" then "BusRd" else "", "V")
        return Request("BusWr", if old == "I" then "I" else "V")
    if old == "I"
        if operation == "W"
            return Request("BusRdX", "M")
        return Request("BusRd", if protocol != "MSI" and not others then "E" else "S")
    if operation == "R"
        return Request("", old)
    if old == "M" or old == "E"
        return Request("", "M")
    return Request("BusUpgr", "M")
