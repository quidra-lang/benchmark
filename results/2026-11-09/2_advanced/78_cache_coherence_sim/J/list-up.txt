cache_touch_way =: 4 : 0
    'row way fill' =. y
    policy =. x{::CachePolicies
    ways =. x{CacheWays
    position =. (row*ways)+way
    if. (policy-:'LRU')+.((policy-:'FIFO')*.fill) do.
        clock =. 1+x{CacheClock
        CacheClock =: clock x}CacheClock
        touched =. x{::CacheTouched
        CacheTouched =: (<clock position}touched)x}CacheTouched
    elseif. policy-:'PLRU' do.
        lo =. 0
        hi =. ways
        node =. 1
        bits =. x{::CacheBits
        while. 1<hi-lo do.
            middle =. (lo+hi)intdiv 2
            if. way<middle do.
                bits =. 1((row*ways)+node)}bits
                hi =. middle
                node =. 2*node
            else.
                bits =. 0((row*ways)+node)}bits
                lo =. middle
                node =. 1+2*node
            end.
        end.
        CacheBits =: (<bits)x}CacheBits
    end.
)
cache_touch =: 4 : 0
    position =. x cache_locate y
    if. position~:_1 do.
        ways =. x{CacheWays
        x cache_touch_way(position intdiv ways),(ways|position),0
    end.
)
requester =: 4 : 0

            new =. 'I'
        end.
        (<3),<new return.
    end.
    if. old='I' do.
        if. operation=1 do.
            (<1),<'M' return.
        end.
        new =. 'S'
        if. ((<x)e.;:'MESI MOESI')*.-.other_copy do.
            new =. 'E'
        end.
        (<0),<new return.
    end.
    if. operation=0 do.
        (<_1),<old return.
    end.
    if. old e.'ME' do.
        (<_1),<'M'
    else.
        (<2),<'M'
    end.
)
