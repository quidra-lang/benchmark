    private function touchWay(int $row, int $way, bool $fill): void
    {
        if ($this->policy === 'LRU' || ($this->policy === 'FIFO' && $fill)) {
            $this->times[$row][$way] = ++$this->clock;
        } elseif ($this->policy === 'PLRU') {
            $lo = 0;
            $hi = $this->ways;
            $node = 1;
            while ($hi - $lo > 1) {
                $mid = intdiv($lo + $hi, 2);
                if ($way < $mid) {
                    $this->bits[$row][$node] = 1;
                    $hi = $mid;
                    $node *= 2;
                } else {
                    $this->bits[$row][$node] = 0;
                    $lo = $mid;
                    $node = 2 * $node + 1;
                }
            }
        }
    }
    public function touch(int $block): void
    {
        $place = $this->locate($block);
        if ($place !== null) $this->touchWay($place[0], $place[1], false);
    }
function requestTransition(string $protocol, string $old, string $operation): array
{
    if ($protocol === 'VI') {
        if ($operation === 'R') return $old === 'I' ? ['BusRd', 'V'] : [null, 'V'];
        return ['BusWr', $old === 'I' ? 'I' : 'V'];
    }
    if ($old === 'I') {
        if ($operation === 'W') return ['BusRdX', 'M'];
        return ['BusRd', $protocol === 'MSI' ? 'S' : 'ES'];
    }
    if ($operation === 'R') return [null, $old];
    if ($old === 'M' || $old === 'E') return [null, 'M'];
    return ['BusUpgr', 'M'];
}
