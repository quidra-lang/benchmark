  void touch(int set, int way, bool fill) {
    if (policy == 'FIFO' && !fill) return;
    sets[set].ways[way].stamp = ++clock;
    if (policy == 'PLRU') {
      var node = 1, lo = 0, hi = sets[set].ways.length;
      while (hi - lo > 1) {
        final mid = (lo + hi) ~/ 2;
        if (way < mid) {
          sets[set].bits[node] = 1;
          node *= 2;
          hi = mid;
        } else {
          sets[set].bits[node] = 0;
          node = node * 2 + 1;
          lo = mid;
        }
      }
    }
  }
  void hit(int b) {
    final (set, way) = find(b);
    touch(set, way, false);

(String, String) requestTransition(
  String protocol,
  String state,
  String operation,
) {
  if (protocol == 'VI') {
    if (operation == 'W') return ('BusWr', state);
    return state == 'I' ? ('BusRd', 'V') : ('', state);
  }
  if (state == 'I') {
    if (operation == 'W') return ('BusRdX', 'M');
    return ('BusRd', protocol == 'MSI' ? 'S' : 'E/S');
  }
  if (operation == 'R') return ('', state);
  return state == 'S' || state == 'O' ? ('BusUpgr', 'M') : ('', 'M');
}
