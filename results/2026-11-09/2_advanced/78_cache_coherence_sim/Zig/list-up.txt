    fn touchWay(self: *Cache, row_index: usize, way: usize, fill_value: bool) void {
        const set = &self.sets[row_index];
        if (self.replacement == .lru or (self.replacement == .fifo and fill_value)) {
            self.clock += 1;
            set.touched[way] = self.clock;
        } else if (self.replacement == .plru) {
            var lo: usize = 0;
            var hi = self.ways;
            var node: usize = 1;
            while (hi - lo > 1) {
                const mid = (lo + hi) / 2;
                if (way < mid) {
                    set.bits[node] = 1;
                    hi = mid;
                    node *= 2;
                } else {
                    set.bits[node] = 0;
                    lo = mid;
                    node = 2 * node + 1;
                }
            }
        }
    }
    pub fn touch(self: *Cache, block: i64) void {
        if (self.locate(block)) |way| self.touchWay(self.row(block), way, false);
    }
pub fn requester(protocol: b.Protocol, old: u8, operation: u8, other: bool) RequestTransition {
    if (protocol == .vi) {
        if (operation == 'R') return .{ .transaction = if (old == 'I') .rd else null, .state = 'V' };
        return .{ .transaction = .wr, .state = if (old == 'I') 'I' else 'V' };
    }
    if (old == 'I') {
        if (operation == 'W') return .{ .transaction = .rdx, .state = 'M' };
        return .{ .transaction = .rd, .state = if (protocol == .msi or other) 'S' else 'E' };
    }
    if (operation == 'R') return .{ .transaction = null, .state = old };
    if (old == 'M' or old == 'E') return .{ .transaction = null, .state = 'M' };
    return .{ .transaction = .upgr, .state = 'M' };
}
