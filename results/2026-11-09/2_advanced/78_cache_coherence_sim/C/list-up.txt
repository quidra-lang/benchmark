static void touch_way(Cache *cache, int row, int way, bool fill) {
    CacheSet *set = &cache->sets[row];
    if (cache->replacement == LRU || (cache->replacement == FIFO && fill))
        set->touched[way] = ++cache->clock;
    else if (cache->replacement == PLRU) {
        int lo = 0, hi = cache->ways, node = 1;
        while (hi - lo > 1) {
            int mid = (lo + hi) / 2;
            if (way < mid) {
                set->bits[node] = 1;
                hi = mid;
                node *= 2;
            } else {
                set->bits[node] = 0;
                lo = mid;
                node = node * 2 + 1;
            }
        }
    }
}
void cache_touch(Cache *cache, Number block) {
    int way = cache_locate(cache, block);
    if (way >= 0)
        touch_way(cache, (int)(block % cache->set_count), way, false);
}
RequestTransition requester(Protocol protocol, char old, char op, bool other) {
    if (protocol == P_VI) {
        if (op == 'R')
            return (RequestTransition){old == 'I' ? BUS_RD : TX_NONE, 'V'};
        return (RequestTransition){BUS_WR, old == 'I' ? 'I' : 'V'};
    }
    if (old == 'I') {
        if (op == 'W')
            return (RequestTransition){BUS_RDX, 'M'};
        return (RequestTransition){BUS_RD, protocol == P_MSI || other ? 'S' : 'E'};
    }
    if (op == 'R')
        return (RequestTransition){TX_NONE, old};
    if (old == 'M' || old == 'E')
        return (RequestTransition){TX_NONE, 'M'};
    return (RequestTransition){BUS_UPGR, 'M'};
}
