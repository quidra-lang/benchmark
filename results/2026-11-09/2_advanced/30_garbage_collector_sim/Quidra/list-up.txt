    int[] roots := []
    int value(int source)
        return if source < 0 then 0 else roots[source]
    void .push(int number)
        .retain(number)
        roots = roots.append(number)
        name := command.name
        if name == "PUSH"
            .push(value(command.source))
        elif name == "LOAD"
            .push(collector.objects[roots[command.slot] - 1].fields[command.field])
        elif name == "POP"
            for _ < command.count
                previous := roots[-1]
                roots = roots[0:<#roots - 1]
                .release(previous)
