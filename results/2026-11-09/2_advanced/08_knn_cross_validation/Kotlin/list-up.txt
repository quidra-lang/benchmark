fun distance(first: List<Double>, second: List<Double>, metric: String): Double {
    var total = 0.0
    for (index in first.indices) {
        val a = first[index]
        val b = second[index]
        val difference = a - b
        when (metric) {
            "squared_euclidean" -> total += difference * difference
        }
    }
    if (metric != "cosine") return total
