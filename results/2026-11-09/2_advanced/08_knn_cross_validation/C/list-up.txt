double distance(const double *a,const double *b,int count,Metric metric){double total=0,dot=0,aa=0,bb=0;for(int j=0;j<count;j++){double x=a[j],y=b[j],d=x-y;if(metric==SQUARED_EUCLIDEAN)total+=d*d;else if(metric==MANHATTAN)total+=fabs(d);else if(metric==CHEBYSHEV)total=fmax(total,fabs(d));else{dot+=x*y;aa+=x*x;bb+=y*y;}}if(metric==COSINE){double na=sqrt(aa),nb=sqrt(bb);if(na==0||nb==0)return 1;return 1-dot/(na*nb);}return total;}
