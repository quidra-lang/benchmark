        fields = split(lines[position])
        position += 1
        if fields[1]=="VARIABLE"
            push!(names,fields[2])
            index[fields[2]] = length(names)
            push!(states,String.(fields[3:end]))
function parse_evidence(a,fields)
    evidence = Pair{Int,Int}[]
    for field in fields
        variable,state = split(field,'=')
        v = findfirst(==(variable),a.names)
        push!(evidence,v=>findfirst(==(state),a.states[v]))
    end
    evidence
end
