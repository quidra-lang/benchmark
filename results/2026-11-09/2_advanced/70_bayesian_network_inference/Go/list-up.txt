		case "VARIABLE":
			n.Names[a[1]] = len(n.Variables)
			n.Variables = append(n.Variables,
				Variable{Name: a[1], States: a[2:]})
func evidence(token string, n Network) Evidence {
	a := strings.Split(token, "=")
	v := n.Names[a[0]]
	for s, name := range n.Variables[v].States {
		if name == a[1] {
			return Evidence{v, s}
		}
	}
	panic("unknown evidence")
}
