function parseEvidence($network, $tokens)
{
    $evidence = [];
    foreach ($tokens as $token) {
        [$name, $state] = explode('=', $token);
        $variable = $network->names[$name];
        $evidence[$variable] = $network->stateNames[$variable][$state];
    }
    return $evidence;
}
