        local fields = words(lines[position])
        position = position + 1
        if fields[1] == 'VARIABLE' then
            local v = #network.names + 1
            network.index[fields[2]], network.names[v], network.states[v], network.parents[v] = v, fields[2], {}, {}
            for i = 3, #fields do network.states[v][#network.states[v] + 1] = fields[i] end
local function parse_evidence(fields, first, last)
    local result = {map = {}, order = {}}
    for i = first or 1, last or #fields do
        local name, state = fields[i]:match('^([^=]+)=(.+)$')
        local v = network.index[name]
        local s
        for index, candidate in ipairs(states[v]) do if candidate == state then s = index end end
        result.map[v] = s
        result.order[#result.order + 1] = {v, s}
    end
    return result
end
