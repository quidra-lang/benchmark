        List<List<Rational.Point>> traverse(boolean invertFirst, boolean invertSecond) {
            boolean[] inverted = {invertFirst, invertSecond};
            Set<Crossing> visited = new HashSet<>();
            List<List<Rational.Point>> result = new ArrayList<>();
            for (Crossing start : crossings) {
                if (visited.contains(start))
                    continue;
                List<Rational.Point> points = new ArrayList<>();
                Node current = start.first;
                while (true) {
                    visited.add(current.crossing);
                    points.add(current.point);
                    boolean forward = current.entering != inverted[current.side];
                    Node stopped = forward ? current.next : current.previous;
                    while (stopped.crossing == null) {
                        points.add(stopped.point);
                        stopped = forward ? stopped.next : stopped.previous;
                    }
                    if (stopped.crossing == start)
                        break;
                    current = stopped.other;
                }
                result.add(points);
            }
            for (int side = 0; side < 2; side++)
                for (int index = 0; index < rings.get(side).size(); index++) {
                    boolean hasCrossing = false;
                    for (Node node : nodes.get(side).get(index))
                        if (node.crossing != null)
                            hasCrossing = true;
                    if (!hasCrossing && initialInside.get(side).get(index) != inverted[side])
                        result.add(new ArrayList<>(rings.get(side).get(index)));
                }
            return result;
        }
    static List<Polygon> assemble(List<List<Rational.Point>> resultRings) {
        List<List<Rational.Point>> rotated = new ArrayList<>();
        for (List<Rational.Point> ring : resultRings)
            rotated.add(rotate(ring));
        int[] depths = new int[rotated.size()];
        for (int i = 0; i < rotated.size(); i++)
            for (int j = 0; j < rotated.size(); j++)
                if (i != j && inside(rotated.get(j), rotated.get(i).getFirst()))
                    depths[i]++;
        List<List<Rational.Point>> oriented = new ArrayList<>();
        List<Integer> outerIndices = new ArrayList<>();
        for (int i = 0; i < rotated.size(); i++) {
            List<Rational.Point> ring = rotated.get(i);
            boolean outer = depths[i] % 2 == 0;
            oriented.add(counterclockwise(ring) == outer ? ring : reverse(ring));
            if (outer)
                outerIndices.add(i);
        }
        outerIndices.sort(Comparator.comparing(i -> oriented.get(i).getFirst()));
        List<Polygon> result = new ArrayList<>();
        for (int index : outerIndices) {
            List<Rational.Point> outer = oriented.get(index);
            List<List<Rational.Point>> holes = new ArrayList<>();
            for (int j = 0; j < oriented.size(); j++)
                if (depths[j] == depths[index] + 1 && inside(outer, oriented.get(j).getFirst()))
                    holes.add(oriented.get(j));
            holes.sort(Comparator.comparing(List::getFirst));
            List<List<Rational.Point>> polygon = new ArrayList<>();
            polygon.add(outer);
            polygon.addAll(holes);
            result.add(new Polygon(polygon));
        }
        return result;
    }
