    def traverse(invert_first, invert_second)
      inverted, visited, result = [invert_first, invert_second], Set.new, []
      @crossings.each do |start|
        next if visited.include?(start)
        points, current = [], start.first
        loop do
          visited.add(current.crossing)
          points << current.point
          forward = current.entering != inverted[current.side]
          stopped = forward ? current.next : current.previous
          until stopped.crossing
            points << stopped.point
            stopped = forward ? stopped.next : stopped.previous
          end
          break if stopped.crossing.equal?(start)
          current = stopped.other
        end
        result << points
      end
      @rings.each_with_index do |sequence, side|
        sequence.each_with_index do |ring, index|
          next if @nodes[side][index].any?(&:crossing)
          result << ring.dup if @initial_inside[side][index] != inverted[side]
        end
      end
      result
    end
  def assemble(result_rings)
    rotated = result_rings.map { |ring| rotate(ring) }
    depths = rotated.each_with_index.map do |ring, index|
      rotated.each_with_index.count { |other, other_index| other_index != index && inside?(other, ring.first) }
    end
    oriented = rotated.each_with_index.map do |ring, index|
      counterclockwise?(ring) == depths[index].even? ? ring : reverse(ring)
    end
    outers = depths.each_index.select { |index| depths[index].even? }
    outers.sort_by! { |index| RationalGeometry.point_key(oriented[index].first) }
    outers.map do |index|
      outer = oriented[index]
      holes = oriented.each_with_index.filter_map do |ring, hole_index|
        ring if depths[hole_index] == depths[index] + 1 && inside?(outer, ring.first)
      end
      [outer, *holes.sort_by { |ring| RationalGeometry.point_key(ring.first) }]
    end
  end
  def binary_result(kind, first, second)
    graph = GreinerHormann::Graph.new(first.rings, second.rings)
    intersection = Rings.assemble(graph.traverse(false, false))
    union = Rings.assemble(graph.traverse(true, true))
