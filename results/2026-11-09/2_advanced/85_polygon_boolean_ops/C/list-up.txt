Rings traverse(const GhGraph *g, bool invert_first, bool invert_second) {
    bool inverted[2] = {invert_first, invert_second};
    bool *visited = allocate(g->crossing_count * sizeof(bool));
    memset(visited, 0, g->crossing_count * sizeof(bool));
    Rings result = {0};
    for (size_t i = 0; i < g->order_count; ++i) {
        Crossing *start = g->crossings[i];
        if (visited[start->index])
            continue;
        Ring *ring = ring_new();
        GhNode *current = start->first;
        for (;;) {
            visited[current->crossing->index] = true;
            ring_point(ring, current->point);
            bool forward = current->entering != inverted[current->side];
            GhNode *moving = current;
            for (;;) {
                GhNode *next = forward ? moving->next : moving->previous;
                ring_edge(ring, forward ? moving->outgoing : reverse_line(next->outgoing));
                moving = next;
                if (moving->crossing)
                    break;
                ring_point(ring, moving->point);
            }
            if (moving->crossing == start)
                break;
            current = moving->other;
        }
        rings_add(&result, ring);
    }
    for (int side = 0; side < 2; ++side)
        for (size_t i = 0; i < g->rings[side].count; ++i) {
            bool crossed = false;
            for (size_t j = 0; j < g->nodes[side][i].count; ++j)
                crossed = crossed || g->nodes[side][i].items[j]->crossing != NULL;
            if (!crossed && g->initial_inside[side][i] != inverted[side])
                rings_add(&result, ring_retain(g->rings[side].items[i]));
        }
    free(visited);
    return result;
}
PolygonList assemble(const Rings *input) {
    Rings rings = {0};
    int *depth = allocate(input->count * sizeof(int));
    memset(depth, 0, input->count * sizeof(int));
    for (size_t i = 0; i < input->count; ++i)
        rings_add(&rings, rotate_ring(input->items[i]));
    for (size_t i = 0; i < rings.count; ++i) {
        for (size_t j = 0; j < rings.count; ++j)
            if (i != j && inside_ring(rings.items[i]->points[0], rings.items[j]))
                ++depth[i];
        if (counterclockwise(rings.items[i]) != (depth[i] % 2 == 0)) {
            Ring *reversed = reverse_ring(rings.items[i]);
            ring_release(rings.items[i]);
            rings.items[i] = reversed;
        }
    }
    PolygonList result = {0};
    for (size_t i = 0; i < rings.count; ++i)
        if (depth[i] % 2 == 0) {
            Polygon *polygon = allocate(sizeof(*polygon));
            *polygon = (Polygon){.references = 1};
            rings_add(&polygon->rings, ring_retain(rings.items[i]));
            Rings holes = {0};
            for (size_t j = 0; j < rings.count; ++j)
                if (depth[j] == depth[i] + 1 &&
                    inside_ring(rings.items[j]->points[0], rings.items[i]))
                    rings_add(&holes, ring_retain(rings.items[j]));
            if (holes.count)
                qsort(holes.items, holes.count, sizeof(Ring *), ring_compare);
            for (size_t j = 0; j < holes.count; ++j)
                rings_add(&polygon->rings, holes.items[j]);
            free(holes.items);
            polygon_list_add(&result, polygon);
        }
    if (result.count)
        qsort(result.items, result.count, sizeof(Polygon *), polygon_compare);
    return result;
}
