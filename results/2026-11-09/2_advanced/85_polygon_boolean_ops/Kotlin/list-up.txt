    fun result(kind: String): Region {
        val inverted = if (kind == "UNION") listOf(true, true) else if (kind == "DIFFERENCE") listOf(true, false) else listOf(false, false)
        val visited = BooleanArray(locations.size)
        val rings = mutableListOf<Ring>()
        for (start in order) if (!visited[start]) {
            var side = 0
            var id = start
            val points = mutableListOf<Point>()
            while (true) {
                val (ri, index) = locations[id][side]
                val list = nodes[side][ri]
                val node = list[index]
                visited[id] = true
                points.add(node.point)
                val forward = node.enter != inverted[side]
                var next = index
                do {
                    next = if (forward) (next + 1) % list.size else (next + list.size - 1) % list.size
                    if (list[next].crossing == null) points.add(list[next].point)
                } while (list[next].crossing == null)
                val found = list[next].crossing!!
                if (found == start) break
                id = found
                side = 1 - side
            }
            rings.add(points)
        }
        for (side in 0..1) {
            val other = nodes[1 - side].map {
                list -> list.map {
                    it.point
                }
            }
            for (list in nodes[side]) if (list.none {
                it.crossing != null
            } && RingOps.interior(list[0].point, other) != inverted[side]) {
                rings.add(list.map {
                    it.point
                })
            }
        }
        return RingOps.assemble(rings)
    }
    fun assemble(input: Rings): Region {
        val rings = input.map(::rotate).toMutableList()
        val depths = rings.indices.map {
            i -> rings.indices.count {
                it != i && inside(rings[i][0], rings[it])
            }
        }
        for (i in rings.indices) if (ccw(rings[i]) != (depths[i] % 2 == 0)) rings[i] = reverse(rings[i])
        return rings.indices.filter {
            depths[it] % 2 == 0
        }.sortedBy {
            rings[it][0]
        }.map {
            i ->
            listOf(rings[i]) + rings.indices.filter {
                depths[it] == depths[i] + 1 && inside(rings[it][0], rings[i])
            }
            .sortedBy {
                rings[it][0]
            }.map {
                rings[it]
            }
        }
    }
