function Graph:traverse(invert_first, invert_second)
    local inverted, visited, result = {invert_first, invert_second}, {}, {}
    for _, start in ipairs(self.crossings) do
        if not visited[start] then
            local ring, current = {lines = {}}, start.first
            while true do
                visited[current.crossing] = true
                local forward = current.entering ~= inverted[current.side]
                append_node(ring, current, forward)
                local stopped = forward and current.next or current.previous
                while not stopped.crossing do
                    append_node(ring, stopped, forward)
                    stopped = forward and stopped.next or stopped.previous
                end
                if stopped.crossing == start then break end
                current = stopped.other
            end
            result[#result + 1] = ring
        end
    end
    for side, sequence in ipairs(self.rings) do
        for r, ring in ipairs(sequence) do
            local crossed = false
            for _, node in ipairs(self.nodes[side][r]) do if node.crossing then crossed = true; break end end
            if not crossed and self.initial_inside[side][r] ~= inverted[side] then result[#result + 1] = ring end
        end
    end
    return result
end
function Rings.assemble(result_rings)
    local rotated, depths, oriented, outer_indices = {}, {}, {}, {}
    for i, ring in ipairs(result_rings) do rotated[i] = Rings.rotate(ring) end
    for i, ring in ipairs(rotated) do
        local depth = 0
        for j, other in ipairs(rotated) do if i ~= j and Rings.inside(other, ring[1]) then depth = depth + 1 end end
        depths[i] = depth
        oriented[i] = Rings.ccw(ring) == (depth % 2 == 0) and ring or Rings.reverse(ring)
        if depth % 2 == 0 then outer_indices[#outer_indices + 1] = i end
    end
    table.sort(outer_indices, function(a, b) return Rational.less(oriented[a][1], oriented[b][1]) end)
    local result = {}
    for _, index in ipairs(outer_indices) do
        local outer, holes = oriented[index], {}
        for i, ring in ipairs(oriented) do
            if depths[i] == depths[index] + 1 and Rings.inside(outer, ring[1]) then holes[#holes + 1] = ring end
        end
        table.sort(holes, function(a, b) return Rational.less(a[1], b[1]) end)
        local polygon = {outer}
        for _, hole in ipairs(holes) do polygon[#polygon + 1] = hole end
        result[#result + 1] = polygon
    end
    return result
end
local function binary_result(kind, first, second)
    local graph = Graph.new(first.rings, second.rings)
    local intersection, union = Rings.assemble(graph:traverse(false, false)), Rings.assemble(graph:traverse(true, true))
