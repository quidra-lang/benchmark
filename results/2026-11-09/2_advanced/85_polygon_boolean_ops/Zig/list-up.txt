pub const Crossing = struct { point: r.Point, first: *Node, second: *Node, visited: bool = false };
    pub fn traverse(self: Graph, invert_first: bool, invert_second: bool) rings.Sequence {
        const inverted = [2]bool{ invert_first, invert_second };
        for (self.crossings) |record| record.visited = false;
        var result: std.ArrayList(rings.Ring) = .empty;
        for (self.crossings) |start| {
            if (start.visited) continue;
            var points: std.ArrayList(r.Point) = .empty;
            var current = start.first;
            while (true) {
                current.crossing.?.visited = true;
                r.append(r.Point, &points, current.point);
                const forward = current.entering != inverted[current.side];
                var stopped = if (forward) current.next else current.previous;
                while (stopped.crossing == null) {
                    r.append(r.Point, &points, stopped.point);
                    stopped = if (forward) stopped.next else stopped.previous;
                }
                if (stopped.crossing.? == start) break;
                current = stopped.other.?;
            }
            r.append(rings.Ring, &result, points.items);
        }
        for (self.rings, 0..) |sequence, side| for (sequence, 0..) |ring, index| {
            var crossed = false;
            for (self.nodes[side][index]) |node| crossed = crossed or node.crossing != null;
            if (!crossed and self.initial_inside[side][index] != inverted[side]) r.append(rings.Ring, &result, ring);
        };
        return result.items;
    }
pub fn assemble(result_rings: Sequence) Polygons {
    const rotated = r.alloc(Ring, result_rings.len);
    for (result_rings, rotated) |ring, *out| out.* = rotate(ring);
    const depths = r.zero(usize, rotated.len, 0);
    for (rotated, 0..) |ring, index| for (rotated, 0..) |other, j| if (index != j and inside(other, ring[0])) {
        depths[index] += 1;
    };
    for (rotated, depths) |*ring, depth| if (counterclockwise(ring.*) != (depth % 2 == 0)) {
        ring.* = reverse(ring.*);
    };
    var outers: std.ArrayList(usize) = .empty;
    for (depths, 0..) |depth, index| if (depth % 2 == 0) r.append(usize, &outers, index);
    std.mem.sort(usize, outers.items, rotated, struct {
        fn less(sequence: []Ring, a: usize, b: usize) bool {
            return ringLess({}, sequence[a], sequence[b]);
        }
    }.less);
    var polygons: std.ArrayList(Sequence) = .empty;
    for (outers.items) |index| {
        var holes: std.ArrayList(Ring) = .empty;
        for (rotated, depths) |ring, depth| if (depth == depths[index] + 1 and inside(rotated[index],
             ring[0])) r.append(Ring, &holes, ring);
        std.mem.sort(Ring, holes.items, {}, ringLess);
        const sequence = r.alloc(Ring, holes.items.len + 1);
        sequence[0] = rotated[index];
        @memcpy(sequence[1..], holes.items);
        r.append(Sequence, &polygons, sequence);
    }
    return polygons.items;
}
