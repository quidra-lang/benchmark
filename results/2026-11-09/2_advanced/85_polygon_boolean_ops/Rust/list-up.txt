    pub fn result(&self,kind:&str)->Region{
        let inverted=match kind{
            "UNION"=>[true,true],"DIFFERENCE"=>[true,false],_=>[false,false]
        };
        let mut visited=vec![false;self.locations.len()];
        let mut rings=Vec::new();
        for &start in &self.order{
            if visited[start]{
                continue;
            }
            let(mut side,mut id)=(0,start);
            let mut points=Vec::new();
            loop{
                let(ri,index)=self.locations[id][side];
                let list=&self.nodes[side][ri];
                let node=&list[index];
                visited[id]=true;
                points.push(node.point);
                let forward=node.enter!=inverted[side];
                let mut next=index;
                loop{
                    next=if forward{
                        (next+1)%list.len()
                    }else{
                        (next+list.len()-1)%list.len()
                    };
                    if list[next].crossing.is_some(){
                        break;
                    }
                    points.push(list[next].point);
                }
                let found=list[next].crossing.unwrap();
                if found==start{
                    break;
                }
                id=found;
                side=1-side;
            }
            rings.push(points);
        }
        for side in 0..2{
            for list in &self.nodes[side]{
                if list.iter().any(|n|n.crossing.is_some()){
                    continue;
                }
                let other:Vec<_>=self.nodes[1-side].iter().map(|r|r.iter().map(|n|n.point).collect()).collect();
                let inside=rings::interior(list[0].point,&other);
                if inside!=inverted[side]{
                    rings.push(list.iter().map(|n|n.point).collect());
                }
            }
        }
        rings::assemble(rings)
    }
pub fn assemble(mut rings:Rings)->Region{
    for r in &mut rings{
        rotate(r);
    }
    let depth:Vec<_>=rings.iter().enumerate().map(|(i,r)|rings.iter().enumerate().filter(|(j,q)|i!=*j&&inside(r[0],q)).count()).collect();
    for(i,r)in rings.iter_mut().enumerate(){
        if ccw(r)!=(depth[i]%2==0){
            reverse(r);
        }
    }
    let mut outers:Vec<_>=(0..rings.len()).filter(|&i|depth[i]%2==0).collect();
    outers.sort_by_key(|&i|rings[i][0]);
    outers.iter().map(|&i|{
        let mut holes:Vec<_>=(0..rings.len()).filter(|&j|depth[j]==depth[i]+1&&inside(rings[j][0],&rings[i])).collect();
        holes.sort_by_key(|&j|rings[j][0]);
        let mut polygon=vec![rings[i].clone()];
        polygon.extend(holes.into_iter().map(|j|rings[j].clone()));
        polygon
    }).collect()
