    public function traverse(bool $invertFirst, bool $invertSecond): array
    {
        $inverted = [$invertFirst, $invertSecond];
        $visited = $result = [];
        foreach ($this->crossings as $start) {
            if (isset($visited[$start->crossing]))
                continue;
            $ring = [];
            $current = $start;
            while (true) {
                $visited[$current->crossing] = true;
                $forward = $current->entering !== $inverted[$current->side];
                $ring[] = new Vertex($current->point, $forward ? $current->line : $current->previous->line);
                $stopped = $forward ? $current->next : $current->previous;
                while ($stopped->crossing === null) {
                    $ring[] = new Vertex(
                        $stopped->point,
                        $forward ? $stopped->line : $stopped->previous->line,
                    );
                    $stopped = $forward ? $stopped->next : $stopped->previous;
                }
                if ($stopped->crossing === $start->crossing)
                    break;
                $current = $stopped->other;
            }
            $result[] = $ring;
        }
        foreach ([$this->first, $this->second] as $side => $sequence)
            foreach ($sequence as $i => $ring)
                if (
                    !array_any($this->nodes[$side][$i], fn($n) => $n->crossing !== null)
                    && $this->initialInside[$side][$i] !== $inverted[$side]
                )
                    $result[] = $ring;
        return $result;
    }
function assemble(array $rings): array
{
    $rotated = array_map(rotateRing(...), $rings);
    $depths = $oriented = [];
    foreach ($rotated as $i => $ring) {
        $depth = 0;
        foreach ($rotated as $j => $other)
            if ($i !== $j && inside($other, $ring[0]->point))
                ++$depth;
        $depths[$i] = $depth;
        $oriented[$i] = counterclockwise($ring) === (($depth % 2) === 0) ? $ring : reverseRing($ring);
    }
    $outer = array_values(array_filter(array_keys($rotated), fn($i) => ($depths[$i] % 2) === 0));
    usort($outer, fn($a, $b) => pointCompare($oriented[$a][0]->point, $oriented[$b][0]->point));
    $result = [];
    foreach ($outer as $i) {
        $holes = [];
        foreach ($oriented as $j => $r)
            if ($depths[$j] === ($depths[$i] + 1) && inside($oriented[$i], $r[0]->point))
                $holes[] = $r;
        usort($holes, fn($a, $b) => pointCompare($a[0]->point, $b[0]->point));
        $result[] = [$oriented[$i], ...$holes];
    }
    return $result;
}
