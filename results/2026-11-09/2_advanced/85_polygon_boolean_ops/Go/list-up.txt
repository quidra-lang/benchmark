func (g *GHGraph) Traverse(invertFirst, invertSecond bool) []Ring {
	inverted := [2]bool{invertFirst, invertSecond}
	visited := map[*Crossing]bool{}
	out := []Ring{}
	for _, start := range g.Crossings {
		if visited[start] {
			continue
		}
		points := Ring{}
		current := start.First
		for {
			visited[current.Crossing] = true
			points = append(points, current.Point)
			forward := current.Entering != inverted[current.Side]
			stopped := current.Previous
			if forward {
				stopped = current.Next
			}
			for stopped.Crossing == nil {
				points = append(points, stopped.Point)
				if forward {
					stopped = stopped.Next
				} else {
					stopped = stopped.Previous
				}
			}
			if stopped.Crossing == start {
				break
			}
			current = stopped.Other
		}
		out = append(out, points)
	}
	for side, sequence := range g.Rings {
		for i, ring := range sequence {
			hasCrossing := false
			for _, node := range g.Nodes[side][i] {
				if node.Crossing != nil {
					hasCrossing = true
					break
				}
			}
			if !hasCrossing && g.InitialInside[side][i] != inverted[side] {
				out = append(out, ring)
			}
		}
	}
	return out
}
func assemble(input []Ring) []*Polygon {
	rings := make([]Ring, len(input))
	depth := make([]int, len(input))
	for i, r := range input {
		rings[i] = rotate(r)
	}
	for i, r := range rings {
		for j, other := range rings {
			if i != j && insideRing(r[0], other) {
				depth[i]++
			}
		}
		if ccw(r) != (depth[i]%2 == 0) {
			rings[i] = reverse(r)
		}
	}
	outer := []int{}
	for i := range rings {
		if depth[i]%2 == 0 {
			outer = append(outer, i)
		}
	}
	sort.Slice(outer, func(i, j int) bool { return pointLess(rings[outer[i]][0], rings[outer[j]][0]) })
	out := []*Polygon{}
	for _, i := range outer {
		holes := []Ring{}
		for j, r := range rings {
			if depth[j] == depth[i]+1 && insideRing(r[0], rings[i]) {
				holes = append(holes, r)
			}
		}
		sort.Slice(holes, func(i, j int) bool { return pointLess(holes[i][0], holes[j][0]) })
		out = append(out, &Polygon{Rings: append([]Ring{rings[i]}, holes...)})
	}
	return out
