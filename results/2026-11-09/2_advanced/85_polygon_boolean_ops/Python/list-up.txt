    def traverse(self, invert_first, invert_second):
        inverted = invert_first, invert_second
        visited = set()
        result = []
        for start in self.crossings:
            if start in visited:
                continue
            points = []
            current = start.first
            while True:
                visited.add(current.crossing)
                points.append(current.point)
                forward = current.entering != inverted[current.side]
                stopped = current.next if forward else current.previous
                while stopped.crossing is None:
                    points.append(stopped.point)
                    stopped = stopped.next if forward else stopped.previous
                if stopped.crossing is start:
                    break
                current = stopped.other
            result.append(points)
        for side, sequence in enumerate(self.rings):
            for index, ring in enumerate(sequence):
                if any(node.crossing is not None for node in self.nodes[side][index]):
                    continue
                if self.initial_inside[side][index] != inverted[side]:
                    result.append(list(ring))
        return result
def assemble(result_rings):
    rotated = [rotate(ring) for ring in result_rings]
    depths = [sum(inside(other, ring[0]) for other_index, other in enumerate(rotated)
                  if other_index != index) for index, ring in enumerate(rotated)]
    oriented = [ring if counterclockwise(ring) == (depth % 2 == 0) else reverse(ring)
                for ring, depth in zip(rotated, depths)]
    outer_indices = [index for index, depth in enumerate(depths) if depth % 2 == 0]
    outer_indices.sort(key=lambda index: point_key(oriented[index][0]))
    result = []
    for index in outer_indices:
        outer = oriented[index]
        holes = [ring for ring, depth in zip(oriented, depths)
                 if depth == depths[index] + 1 and inside(outer, ring[0])]
        holes.sort(key=lambda ring: point_key(ring[0]))
        result.append([outer, *holes])
    return result
