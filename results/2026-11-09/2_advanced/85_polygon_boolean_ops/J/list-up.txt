traverse_graph =: 4 : 0
    'points sides ids next previous other entering order ring_nodes initial_inside sequences' =. x
    inverted =. y
    visited =. (#order)$0
    result =. 0$a:
    for_start. order do.
        start_id =. start{ids
        if. start_id{visited do.
            continue.
        end.
        ring =. 0 2$0x
        current =. start
        while. 1 do.
            visited =. 1(current{ids)}visited
            ring =. ring,current{points
            forward =. (current{entering)~:(current{sides){inverted
            if. forward do.
                stopped =. current{next
            else.
                stopped =. current{previous
            end.
            while. _1=stopped{ids do.
                ring =. ring,stopped{points
                if. forward do.
                    stopped =. stopped{next
                else.
                    stopped =. stopped{previous
                end.
            end.
            if. start_id=stopped{ids do.
                break.
            end.
            current =. stopped{other
        end.
        result =. result,<ring
    end.
    for_side. i.2 do.
        sequence =. side{::sequences
        for_index. i.#sequence do.
            nodes =. index{::side{::ring_nodes
            if. +./_1~:nodes{ids do.
                continue.
            end.
            state =. index{side{::initial_inside
            if. state~:side{inverted do.
                result =. result,index{sequence
            end.
        end.
    end.
    result
)
assemble =: 3 : 0
    rotated =. rotate_ring&.>y
    depths =. (#y)$0
    for_index. i.#y do.
        point =. 0{index{::rotated
        depth =. 0
        for_other. (i.#y)-.index do.
            depth =. depth+(other{::rotated)inside point
        end.
        depths =. depth index}depths
    end.
    oriented =. 0$a:
    for_index. i.#y do.
        ring =. index{::rotated
        if. (counterclockwise ring)~:0=2|index{depths do.
            ring =. reverse_ring ring
        end.
        oriented =. oriented,<ring
    end.
    outer_indices =. I.0=2|depths
    keys =. 0 2$0x
    for_index. outer_indices do.
        keys =. keys,point_key 0{index{::oriented
    end.
    result =. 0$a:
    for_index. (/:keys){outer_indices do.
        outer =. index{::oriented
        holes =. 0$a:
        for_other. I.depths=1+index{depths do.
            ring =. other{::oriented
            if. outer inside 0{ring do.
                holes =. holes,<ring
            end.
        end.
        result =. result,<(<outer),sort_rings holes
    end.
    result
)
    'kind first second' =. y
    graph =. (2{::first)make_graph 2{::second
    intersection =. assemble graph traverse_graph 0 0
    union =. assemble graph traverse_graph 1 1
