rational.Point[][][] assemble(const rational.Point[][] source)
    rational.Point[][] normalized := []
    for ring := source[]
        normalized = normalized.append(rotate(ring))
    depths := array(#normalized, fill = 0)
    for: index < int(#normalized)
        for: other < int(#normalized)
            if index != other and inside(normalized[index][0], normalized[other])
                depths[index] += 1
        if ccw(normalized[index]) != (depths[index] % 2 == 0)
            normalized[index] = reverse(normalized[index])
    rational.Point[][][] result := []
    for: index < int(#normalized)
        if depths[index] % 2 == 0
            rational.Point[][] holes := []
            for: other < int(#normalized)
                if depths[other] == depths[index] + 1 and inside(normalized[other][0], normalized[index])
                    holes = holes.append(normalized[other])
            result = result.append([normalized[index]].concat(sorted_rings(holes)))
    return sorted_polygons(result)
