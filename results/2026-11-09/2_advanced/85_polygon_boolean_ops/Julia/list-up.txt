function traverse_graph(graph,invert_first,invert_second)
    inverted = [invert_first,invert_second]
    visited = Set{Int}()
    result = Vector{Point}[]
    for start in graph.crossings
        graph.nodes[start].crossing in visited && continue
        points = Point[]
        current = start
        while true
            node = graph.nodes[current]
            push!(visited,node.crossing)
            push!(points,node.point)
            forward = node.entering != inverted[node.side]
            stopped = forward ? node.next : node.previous
            while graph.nodes[stopped].crossing == 0
                push!(points,graph.nodes[stopped].point)
                stopped = forward ? graph.nodes[stopped].next : graph.nodes[stopped].previous
            end
            graph.nodes[stopped].crossing == graph.nodes[start].crossing && break
            current = graph.nodes[stopped].other
        end
        push!(result,points)
    end
    for side in 1:2, (i,ring) in enumerate(graph.sequences[side])
        any(graph.nodes[id].crossing != 0 for id in graph.ring_nodes[side][i]) && continue
        graph.initial[side][i] != inverted[side] && push!(result,copy(ring))
    end
    return result
end
function assemble_rings(rings)
    rotated = rotate_ring.(rings)
    depths = [count(
        j -> j != i && inside_ring(rotated[j],rotated[i][1]),
        eachindex(rotated)
    ) for i in eachindex(rotated)]
    oriented = [counterclockwise(r) == iseven(d) ? r : reverse_ring(r) for (r,d) in zip(rotated,depths)]
    indices = sort([i for i in eachindex(depths) if iseven(depths[i])]; by=i -> point_key(oriented[i][1]))
    result = Vector{Vector{Point}}[]
    for i in indices
        holes = [oriented[j] for j in eachindex(oriented) if depths[j] == depths[i]+1 && inside_ring(
            oriented[i],
            oriented[j][1]
        )]
        sort!(holes; by=r -> point_key(r[1]))
        push!(result, [oriented[i:i];holes])
    end
    return result
end
function binary_result(kind,first,second)
    graph = build_graph(first.rings,second.rings)
    intersection = assemble_rings(traverse_graph(graph,false,false))
    union_result = assemble_rings(traverse_graph(graph,true,true))
