const BUILTINS = new Set('length substr index split sprintf toupper tolower close int'.split(' '));
const PAIRS = new Set('+= -= *= /= %= ^= == <= >= != ++ -- && || >>'.split(' '));

function tokenize(source) {
  const tokens = [];
  let index = 0, line = 1, column = 1;
  const advance = () => { if (source[index++] === '\n') { line++; column = 1; } else column++; };
  const append = (kind, text, position, value = text) => tokens.push({ kind, text, position, value });
  while (index < source.length) {
    const position = { line, column }, start = index, character = source[index], pair = source.slice(index, index + 2);
    if (character === ' ') { advance(); continue; }
    if (character === '\n') { append('newline', '\n', position); advance(); continue; }
    if (digit(character)) {
      while (digit(source[index])) advance();
      const text = source.slice(start, index); append('number', text, position, Number(text)); continue;
    }
    if (letter(character)) {
      while (letter(source[index]) || digit(source[index])) advance();
      const text = source.slice(start, index);
      const kind = KEYWORDS.has(text) ? 'keyword' : BUILTINS.has(text) ? 'builtin' : source[index] === '(' ? 'function' : 'name';
      append(kind, text, position); continue;
    }
    if (character === '"') {
      advance(); let value = '';
      while (index < source.length && source[index] !== '"') {
        if (source[index] === '\n') throw new SyntaxFault(position);
        if (source[index] === '\\') {
        } else { value += source[index]; advance(); }
      }
      if (index === source.length) throw new SyntaxFault(position);
      advance(); append('string', source.slice(start, index), position, value); continue;
    }
    if (PAIRS.has(pair)) { append('symbol', pair, position); advance(); advance(); continue; }
    if ('+-*/%^!><?:=()[]{} ,;$'.includes(character)) { append('symbol', character, position); advance(); continue; }
  }
  append('eof', '', { line, column: 1 }); return tokens;
}

  statement() {

  simple() {

  expressionList(greater = true) {

  }
  conditional(greater) {

  }
  membership(greater) {
    let result = this.comparison(greater);

  }
  comparison(greater) {

  }
  concat(greater) {
    let left = this.additive(greater);

  }
  additive(greater) {
    let left = this.multiplicative(greater);
    return left;
  }
  multiplicative(greater) {
    let left = this.unary(greater);
    return left;
  }
  unary(greater) {

  }
  postfix(greater) {
    if (this.is('++') || this.is('--')) return this.increment(greater);
    const target = this.atom(greater);

  }
  location(greater) {

function toNumber(value) {
  if (value.kind === 'number' || value.kind === 'strnum') return value.number;
  if (value.kind === 'uninitialized') return 0;
  const found = /^[ \t\n]*([+-]?\d+)/.exec(value.string);
  return found === null ? 0 : Number(found[1]);
}

function toString(value) {
  if (value.kind === 'uninitialized') return '';
  return value.kind === 'number' ? String(value.number) : value.string;
}

function inputValue(value) {
  return /^[ \t\n]*[+-]?\d+[ \t\n]*$/.test(value) ? { kind: 'strnum', string: value, number: toNumber(string(value)) } : string(value);
}

function splitString(value, separator) {
  if (separator === ' ') return value.match(/[^ \t\n]+/g) || [];
  if (separator === '') return Array.from(value);
  return value === '' ? [] : value.split(separator);
}

class Fields {
  declare count: any;
  declare fail: any;
  declare record: any;
  declare setting: any;
  declare values: any;
  constructor(setting, fail) { this.setting = setting; this.fail = fail; this.record = string(''); this.values = new Map(); this.count = 0; }
  setRecord(value) {
    this.record = inputValue(value);
    const pieces = splitString(value, toString(this.setting('FS')));
    this.values = new Map(pieces.map((piece, i) => [i + 1, inputValue(piece)])); this.count = pieces.length;
  }
  read(index) {
    if (index === 0) return this.record;
    return index > this.count ? uninitialized : this.values.get(index) || string('');
  }
  rebuild() {
    const separator = toString(this.setting('OFS')), parts = [];
    let previous = 0;
    for (const [index, value] of Array.from(this.values).sort((a, b) => a[0] - b[0])) {
      if (separator !== '') parts.push(separator.repeat(index - previous - Number(previous === 0)));
      parts.push(toString(value)); previous = index;
    }
    if (separator !== '' && this.count > 0) parts.push(separator.repeat(this.count - previous - Number(previous === 0)));
    this.record = inputValue(parts.join(''));
  }
  write(index, value) {
    if (index === 0) { this.setRecord(toString(value)); return; }
    this.count = Math.max(this.count, index); this.values.set(index, value); this.rebuild();
  }
  resize(value, position) {
    const count = toNumber(value);
    if (count < 0) this.fail(position, 'negative NF');
    if (count < this.count) for (const index of this.values.keys()) if (index > count) this.values.delete(index);
    this.count = count;
    this.rebuild();
  }
}
function display(value) {
  if (value.kind === 'uninitialized') return '(uninitialized)';
  if (value.kind === 'number') return toString(value);
  let result = '"';
  for (const character of value.string) {
    if (character === '"') result += '\\"';
    else if (character === '\\') result += '\\\\';
    else if (character === '\n') result += '\\n';
    else if (character === '\t') result += '\\t';
    else {
      const code = character.charCodeAt(0);
      result += code >= 32 && code <= 126 ? character : '\\' + code.toString(8).padStart(3, '0');
    }
  }
  return result + '"' + (value.kind === 'strnum' ? ' (strnum)' : '');

function argumentsList(lines) {

  }
  return { options, operands };
}

function resolve(iterator) {
