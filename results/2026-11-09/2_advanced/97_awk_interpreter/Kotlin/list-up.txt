data class Position(val line: Int, val column: Int) : Comparable<Position> {
    override fun compareTo(other: Position): Int = compareValuesBy(this, other, Position::line, Position::column)
    override fun toString(): String = "$line:$column"
}
class SyntaxError(val position: Position) : Exception()
data class Token(val kind: String, val text: String, val value: String, val position: Position)
object Lexer {
    fun letter(c: Char): Boolean = c in 'A'..'Z' || c in 'a'..'z'
    fun digit(c: Char): Boolean = c in '0'..'9'
    fun lex(text: String): List<Token> {
        val positions = mutableListOf<Position>();
        var line = 1;
        var column = 1
        for (c in text) {
            positions.add(Position(line, column));
            if (c.code == 10) {
                line++;
                column = 1
            } else column++
        }
        positions.add(Position(line, column))
        val result = mutableListOf<Token>();
        var i = 0
        val keywords = setOf("BEGIN", "END", "function", "if", "else", "while", "for", "do", "break", "continue", "next", "exit", "return", "delete", "in", "getline", "print", "printf")
        val builtins = setOf("length", "substr", "index", "split", "sprintf", "toupper", "tolower", "close", "int")
        val symbols = setOf("+=", "-=", "*=", "/=", "%=", "^=", "==", "<=", ">=", "!=", "++", "--", "&&", "||", ">>")
        while (i < text.length) {
            if (text[i] == ' ') {
                i++;
                continue
            }
            if (text[i].code == 92 && text.getOrNull(i + 1)?.code == 10) {
                i += 2;
                continue
            }
            if (text[i] == '#') {
                while (i < text.length && text[i].code != 10) i++;
                continue
            }
            val start = i;
            var kind = "symbol";
            var value = ""
            if (text[i].code == 10) {
                i++;
                kind = "newline"
            }
            else if (digit(text[i])) {
                while (i < text.length && digit(text[i])) i++;
                kind = "number"
            }
            else if (letter(text[i]) || text[i] == '_') {
                while (i < text.length && (letter(text[i]) || digit(text[i]) || text[i] == '_')) i++
                val name = text.substring(start, i)
                kind = if (name in keywords) "keyword" else if (name in builtins) "builtin" else if (text.getOrNull(i) == '(') "function" else "name"
            } else if (text[i].code == 34) {
                i++;
                var closed = false;
                val decoded = StringBuilder()
                while (i < text.length) {
                    if (text[i].code == 10) throw SyntaxError(positions[start])
                    if (text[i].code == 34) {
                        i++;
                        closed = true;
                        break
                    }
                    if (text[i].code == 92) {
                        val slash = i++
                        if (i == text.length) throw SyntaxError(positions[start])
                        decoded.append(when (text[i].code) {
                            110 -> 10.toChar();
                            116 -> 9.toChar();
                            92, 34, 47 -> text[i];
                            else -> throw SyntaxError(positions[slash])
                        });
                        i++
                    } else decoded.append(text[i++])
                }
                if (!closed) throw SyntaxError(positions[start]);
                kind = "string";
                value = decoded.toString()
            } else if (i + 1 < text.length && text.substring(i, i + 2) in symbols) i += 2
            else if (text[i] in "+-*/%^!><?:=()[]{},;$") i++
            else throw SyntaxError(positions[i])
            result.add(Token(kind, text.substring(start, i), value, positions[start]))
        }
        result.add(Token("end", "", "", positions.last()));
        return result
}

    data class Number(val value: Int) : ExprKind
    data class Text(val value: String) : ExprKind

    data class Index(val name: String, val indices: List<Expr>) : ExprKind
    data class Field(val index: Expr) : ExprKind
    data class Parentheses(val body: Expr) : ExprKind
    data class Unary(val op: String, val body: Expr) : ExprKind
    data class Binary(val op: String, val left: Expr, val right: Expr) : ExprKind
    data class Assignment(val op: String, val target: Expr, val right: Expr) : ExprKind
    data class Increment(val delta: Int, val prefix: Boolean, val target: Expr) : ExprKind
    data class Conditional(val condition: Expr, val yes: Expr, val no: Expr) : ExprKind
    data class Member(val indices: List<Expr>, val name: String, val namePosition: Position) : ExprKind
    data class Call(val name: String, val builtin: Boolean, val arguments: List<Expr>) : ExprKind
    data class Getline(val target: Expr?, val file: Expr?) : ExprKind
}

sealed interface StmtKind {
    var index = 0;
    var function = false;
    var special = false;
    var loops = 0
    val peek: Token get() = tokens[index]
    val text: String get() = peek.text
    fun take(): Token {
        val token = peek;
        if (token.kind != "end") index++;
        return token
    }
    fun expect(text: String): Token {
        if (this.text != text || peek.kind == "end") throw SyntaxError(peek.position);
        return take()
    }
    fun name(): Token {
        if (peek.kind != "name") throw SyntaxError(peek.position);
        return take()
    }
    fun newlines() {
        while (peek.kind == "newline") take()
    }
    fun terminator() {
        if (text == ";" || peek.kind == "newline") take() else if (text != "}") throw SyntaxError(peek.position)
    }
    fun expressions(suppress: Boolean): List<Expr> {
        val result = mutableListOf(expression(suppress));
        while (text == ",") {
            take();
            newlines();
            result.add(expression(suppress))
        };
        return result
    }
    fun expression(suppress: Boolean = false): Expr {
        val left = conditional(suppress);
        if (left.lvalue && text in listOf("=", "+=", "-=", "*=", "/=", "%=", "^=")) {
            val token = take();
            return Expr(token.position, ExprKind.Assignment(token.text, left, expression(suppress)))
        };
        return left
    }
    fun conditional(suppress: Boolean): Expr {
        val condition = binary(1, suppress);
        if (text == "?") {
        };
        return condition
    }
    fun binary(minimum: Int, suppress: Boolean): Expr {
        var left = unary(suppress)
        while (true) {
            val explicit = when (text) {
                "||" -> 1;
                "&&" -> 2;
                "in" -> 3;
                "<", "<=", "==", "!=", ">=" -> 4;
                ">" -> if (suppress) 0 else 4;
                "+", "-" -> 6;
                "*", "/", "%" -> 7;
                else -> 0
            }
            val concat = explicit == 0 && (peek.kind in listOf("number", "string", "name", "function", "builtin") || text in listOf("$", "(", "++", "--"));
            val priority = if (concat) 5 else explicit
            if (priority == 0 || priority < minimum) return left
    fun unary(suppress: Boolean): Expr {
        if (text in listOf("!", "-", "+")) {
        val left = postfix(suppress);
        if (text == "^") {
            val token = take();
            return Expr(token.position, ExprKind.Binary(token.text, left, unary(suppress)))
        };
        return left
    }
    fun postfix(suppress: Boolean): Expr {
        if (text == "++" || text == "--") return prefix(suppress);
        val body = atom(suppress);
        if (body.lvalue && (text == "++" || text == "--")) {
            val token = take();
            return Expr(token.position, ExprKind.Increment(if (token.text == "++") 1 else -1, false, body))
        };
        return body
    }
    fun lvalue(suppress: Boolean): Expr {
        if (peek.kind != "name" && text != "$") throw SyntaxError(peek.position);
        return atom(suppress)
    }
    fun atom(suppress: Boolean): Expr {
        val token = peek
        val kind: ExprKind = when (token.kind) {
            "number" -> {
                take();
                ExprKind.Number(token.text.toInt())
            }
            "string" -> {
                take();
                ExprKind.Text(token.value)
            }
            "name" -> {
                take();
                if (text == "[") {
                } else ExprKind.Name(token.text)
            }
            else -> when (token.text) {
                "$" -> {
                    take();
                    ExprKind.Field(if (text == "++" || text == "--") prefix(suppress) else atom(suppress))
                }
            }
        }
        return Expr(token.position, kind)
    }
    fun block(): Stmt {
        val position = expect("{").position;
        val statements = mutableListOf<Stmt>();
        while (text != "}") {
            if (peek.kind == "newline" || text == ";") take() else statements.add(statement())
        };
        expect("}");
        return Stmt(position, StmtKind.Block(statements))
    }
    fun statement(): Stmt {
        val token = peek;
        val position = token.position
        val kind: StmtKind = when (text) {
            "{" -> return block()
            ";" -> {
                take();
                return Stmt(position, StmtKind.Empty)
            }
            "print", "printf" -> {
                take();
                val printf = token.text == "printf";
                val empty = text in listOf(";", 10.toChar().toString(), "}", ">", ">>");
                var parenthesized = false
                if (text == "(") {
                if (empty && printf) throw SyntaxError(peek.position)
                val arguments = if (empty) emptyList() else if (parenthesized) {
                    take();
                    val values = expressions(false);
                    expect(")");
                    values
                } else expressions(true)
                val redirect = if (text == ">" || text == ">>") {
                    val op = take();
                    Redirect(op.text == ">>", binary(5, false), op.position)
                } else null
                StmtKind.Output(printf, arguments, redirect)
            }
            else -> StmtKind.Expression(expression())
        }
        terminator();
        return Stmt(position, kind)
    }
    fun program(): Program {
        val rules = mutableListOf<Rule>();
        val functions = mutableListOf<Function>()
        while (peek.kind != "end") {
            if (text == ";" || peek.kind == "newline") {
                take();
                continue
            };
            val position = peek.position
            } else if (text == "BEGIN" || text == "END") {
                val kind = take().text;
                special = true;
                val action = block();
                special = false;
                rules.add(Rule(kind, position, null, null, action))
            }
            else if (text == "{") rules.add(Rule("all", position, null, null, block()))
        }
        return Program(rules, functions)
    }
sealed interface Value {
    data class Number(val value: Int) : Value
    data class Text(val value: String) : Value
    data class Numeric(val value: String, override val number: Int) : Value
    data object Uninitialized : Value
    val number: Int get() = when (this) {
        is Number -> value;
        is Numeric -> number;
        Uninitialized -> 0;
        is Text -> {
    }
    val text: String get() = when (this) {
        is Number -> value.toString();
        is Text -> value;
        is Numeric -> value;
        Uninitialized -> ""
    }
    val display: String get() = when (this) {
        is Number -> value.toString();
        is Text -> quote(value);
        is Numeric -> quote(value) + " (strnum)";
        Uninitialized -> "(uninitialized)"
    }
    companion object {
        fun whitespace(c: Char): Boolean = c == ' ' || c.code == 9 || c.code == 10
        fun input(text: String): Value {
            val raw = text.trim(::whitespace);
            val digits = if (raw.firstOrNull() in listOf('+', '-')) raw.drop(1) else raw;
            return if (digits.isNotEmpty() && digits.all(Lexer::digit)) Numeric(text, Text(text).number) else Text(text)
        }
    }
}
fun quote(text: String): String = buildString {
    append(34.toChar());
    for (c in text) {
        when (c.code) {
            34, 92 -> {
                append(92.toChar());
                append(c)
            };
            10 -> {
                append(92.toChar());
                append('n')
            };
            9 -> {
                append(92.toChar());
                append('t')
            };
            in 32..126 -> append(c);
            else -> {
                append(92.toChar());
                append(c.code.toString(8).padStart(3, '0'))
            }
        }
    };
    append(34.toChar())
}
fun splitText(text: String, separator: String): List<String> {
    if (separator.isEmpty()) return text.map(Char::toString)
    return if (text.isEmpty()) emptyList() else text.split(separator)
}
class Fields {
    var record: Value = Value.Text("");
    val values = mutableListOf<Value>()
    fun set(text: String, separator: String) {
        record = Value.input(text);
        values.clear();
        values.addAll(splitText(text, separator).map {
            Value.input(it)
        })
    }
    fun get(index: Int): Value = if (index == 0) record else values.getOrNull(index - 1) ?: Value.Uninitialized
    fun rebuild(separator: String) {
        record = Value.input(values.joinToString(separator) {
            it.text
        })
    }
    fun assign(index: Int, value: Value, fs: String, ofs: String) {
        if (index == 0) set(value.text, fs) else {
            while (values.size < index) values.add(Value.Text(""));
            values[index - 1] = value;
            rebuild(ofs)
        }
    }
    fun nf(size: Int, ofs: String) {
        while (values.size > size) values.removeAt(values.lastIndex);
        while (values.size < size) values.add(Value.Text(""));
        rebuild(ofs)
    }
}

fun validFile(text: String): Boolean = text.isNotEmpty() && !text.startsWith(".") && text.all {
    Lexer.letter(it) || Lexer.digit(it) || it in "._-"
}
fun readFile(name: String): List<String>? = if (validFile(name) && File("input/$name").isFile) File("input/$name").readLines() else null
fun arguments(text: String): Arguments {
    val result = Arguments()
    fun assignment(text: String): Pair<String, String>? {
        val i = text.indexOf('=');
        return if (i < 0) null else text.substring(0, i) to text.substring(i + 1)
    }
    fun variable(text: String): Boolean = text.isNotEmpty() && (Lexer.letter(text[0]) || text[0] == '_') && text.all {
        Lexer.letter(it) || Lexer.digit(it) || it == '_'
    }
    for (line in text.lineSequence().filter {
        it.isNotEmpty()
    }) {
        if (line.startsWith("-F ")) result.options.add("FS" to line.drop(3))
        else if (line.startsWith("-v ")) result.options.add(assignment(line.drop(3))!!)
        else {
            val pair = assignment(line);
            if (pair != null && variable(pair.first)) result.operands.add(Operand.Assignment(pair.first, pair.second)) else {
                result.operands.add(Operand.Input(line, result.files.size));
                result.files.add(FileCount(line))
            }
        }
    }
    return result
}

class Runtime(val program: Program, val preparation: Preparation, val arguments: Arguments) {

    }

    fun cell(frame: Int?, name: String): Cell? = if (frame == null) globals[name] else locals[frame][name]

    fun location(expression: Expr): Location = when (val kind = expression.kind) {
        is ExprKind.Name -> Location.Variable(frame(kind.name), kind.name)
        is ExprKind.Field -> {
            val number = this.expression(kind.index).number;
            if (number < 0) throw RuntimeError(expression.position, "negative field index");
            Location.Field(number)
        }
        else -> error("not an lvalue")
    }
    fun read(location: Location): Value = when (location) {
        is Location.Variable -> value(location.frame, location.name);
        is Location.Field -> fields.get(location.index)
    }
    fun store(location: Location, value: Value, position: Position): Value {
        when (location) {
            is Location.Variable -> if (location.frame == null && location.name == "NF") {
                val n = value.number;
                if (n < 0) throw RuntimeError(position, "negative NF");
                fields.nf(n, global("OFS").text)
            } else if (location.frame != null) locals[location.frame][location.name] = ScalarCell(value) else {
                preparation.globals.add(location.name);
                globals[location.name] = ScalarCell(value)
            }
            is Location.Element -> location.array.elements[location.key] = value
            is Location.Field -> fields.assign(location.index, value, global("FS").text, global("OFS").text)
        }
        return read(location)
    }
    fun expression(expression: Expr): Value {
        val position = expression.position
        return when (val kind = expression.kind) {
            is ExprKind.Number -> Value.Number(kind.value)
            is ExprKind.Text -> Value.Text(kind.value)
            is ExprKind.Name, is ExprKind.Index, is ExprKind.Field -> read(location(expression))
            is ExprKind.Parentheses -> this.expression(kind.body)
            is ExprKind.Assignment -> {
                val location = location(kind.target);
                val b = this.expression(kind.right);
                val value = if (kind.op == "=") b else arithmetic(kind.op.take(1), read(location).number, b.number, position);
                store(location, value, position)
            }

        }

                }
            }

                    if (rule.kind in listOf("BEGIN", "END")) continue
                    try {
                        if (testRule(rule, i)) {
                            rules[i]++;
                            if (rule.action != null) statement(rule.action) else streams.output(null, false, fields.record.text + global("ORS").text)
                        }
                    }
    fun report(status: String) {
        streams.save()
        val lines = mutableListOf(status, "[inputs]")
        for (file in arguments.files) lines.add("${file.name} records=${file.records}")
        lines.add("[rules]")
        for ((i, rule) in program.rules.withIndex()) lines.add("${rule.position} ${rule.kind} count=${rules[i]}")
        lines.addAll(listOf("[outputs]", "stdout.txt writes=${streams.writes} bytes=${streams.stdout.length}"))
        lines.add("[lines]")
        for (line in preparation.lines.keys.sorted()) lines.add("$line count=${preparation.lines.getValue(line)}")
        lines.add("[globals]")
        for (name in preparation.globals.sorted()) {
            val cell = globals[name];
            val display = if (cell is ArrayCell) "array (${cell.array.elements.size})" else global(name).display;
            lines.add("$name = $display")
        }
        saveStats(lines)
    }
fun saveStats(lines: List<String>) {
    File("output").mkdirs();
    File("output/stats.txt").writeText(lines.joinToString("\n") + "\n")
}
fun main() {
    val program = try {
        Parser(Lexer.lex(File("input/program.awk").readText())).program()
    val preparation = Preparation(program)
    try {
        preparation.analyze()
    val runtime = Runtime(program, preparation, arguments(File("input/args.txt").readText()))
    val status = try {
        runtime.execute();
        "status: exit ${runtime.status}"
    runtime.report(status)
