tokenize <- function(source) {
    characters <- strsplit(source, "", fixed = TRUE)[[1L]]
    tokens <- list()
    index <- line <- column <- 1L
    while (index <= length(characters)) {
        character <- characters[index]
        position <- c(line, column)
        if (character == " ") {
            index <- index + 1L
            column <- column + 1L
            next
        }
        if (character == "\\" && index < length(characters) && characters[index + 1L] == "\n") {
            index <- index + 2L
            line <- line + 1L
            column <- 1L
            next
        }
        if (character == "#") {
            while (index <= length(characters) && characters[index] != "\n") {
                index <- index + 1L
                column <- column + 1L
            }
            next
        }
        if (character == "\n") {
            tokens[[length(tokens) + 1L]] <- list(kind = "newline", spelling = "\n", value = NULL, position = position)
            index <- index + 1L
            line <- line + 1L
            column <- 1L
            next
        }
        finish <- index + 1L
        value <- NULL
        if (digit(character)) {
            while (finish <= length(characters) && digit(characters[finish])) finish <- finish + 1L
            kind <- "number"
            value <- as.integer(paste0(characters[seq.int(index, finish - 1L)], collapse = ""))
        } else if (letter(character) || character == "_") {
            while (finish <= length(characters) && name_character(characters[finish])) finish <- finish + 1L
            word <- paste0(characters[seq.int(index, finish - 1L)], collapse = "")
            kind <- if (word %in% KEYWORDS) "keyword" else if (word %in% BUILTINS) "builtin" else if (finish <= length(characters) && characters[finish] == "(") "function_name" else "name"
        } else if (character == "\"") {
            pieces <- character()
            while (finish <= length(characters) && characters[finish] != "\"") {
                if (characters[finish] == "\n") syntax_issue(position)
                if (characters[finish] == "\\") {
                    if (finish == length(characters) || !characters[finish + 1L] %in% names(ESCAPES)) syntax_issue(c(line, column + finish - index))
                    pieces <- c(pieces, ESCAPES[[characters[finish + 1L]]])
                    finish <- finish + 2L
                } else {
                    pieces <- c(pieces, characters[finish])
                    finish <- finish + 1L
                }
            }
            if (finish > length(characters)) syntax_issue(position)
            finish <- finish + 1L
            kind <- "string"
            value <- paste0(pieces, collapse = "")
        } else if (finish <= length(characters) && paste0(character, characters[finish]) %in% PAIRS) {
            finish <- finish + 1L
            kind <- "symbol"
        } else if (character %in% SYMBOLS) kind <- "symbol" else syntax_issue(position)
        spelling <- paste0(characters[seq.int(index, finish - 1L)], collapse = "")
        tokens[[length(tokens) + 1L]] <- list(kind = kind, spelling = spelling, value = value, position = position)
        column <- column + finish - index
        index <- finish
    }
    tokens[[length(tokens) + 1L]] <- list(kind = "end", spelling = "", value = NULL, position = c(sum(characters == "\n") + 1L, 1L))
    tokens
}

parse_program <- function(parser) {
    items <- list()
    while (current_token(parser)$kind != "end") {
        if (current_token(parser)$spelling %in% c("\n", ";")) {
            take_token(parser)
            next
        }
        opening <- current_token(parser)
        if (!is.null(accept_token(parser, "function"))) {
        }
        patterns <- list()
        if (opening$spelling %in% c("BEGIN", "END")) flavor <- take_token(parser)$spelling else if (opening$spelling == "{") flavor <- "all" else {
        }
        parser$phase <- flavor
        if (flavor %in% c("BEGIN", "END", "all") || current_token(parser)$spelling == "{") {
            action <- parse_block(parser)
        parser$phase <- NULL
        items[[length(items) + 1L]] <- awk_node("rule", opening$position, c(patterns, list(action)), flavor = flavor, patterns = patterns, action = action)
    }
    items
}
parse_block <- function(parser) {
    opening <- expect_token(parser, "{")
    statements <- list()
    while (current_token(parser)$spelling != "}") {
        if (current_token(parser)$spelling %in% c("\n", ";")) take_token(parser) else statements[[length(statements) + 1L]] <- parse_statement(parser)
    }
    expect_token(parser, "}")
    awk_node("block", opening$position, statements, statements = statements)
}
parse_terminator <- function(parser) {
    if (current_token(parser)$spelling %in% c(";", "\n")) take_token(parser) else if (current_token(parser)$spelling != "}") syntax_error(parser)
}
parse_statement <- function(parser) {
    token <- current_token(parser)
    spelling <- token$spelling
    if (spelling == "{") return(parse_block(parser))
    if (!is.null(accept_token(parser, ";"))) return(awk_node("empty", token$position))
        expect_token(parser, ")")
        body <- parse_loop_body(parser)
        return(awk_node("for", token$position, list(initial, condition, update, body), initial = initial, condition = condition, update = update, body = body))
    }
    } else {
        expression <- parse_expression(parser)
        result <- awk_node("expression_statement", token$position, list(expression), expression = expression)
    }
    parse_terminator(parser)
    result
}

parse_expression_list <- function(parser, greater = TRUE) {
    result <- list(parse_expression(parser, greater))
    while (!is.null(accept_token(parser, ","))) {
        skip_newlines(parser)
        result[[length(result) + 1L]] <- parse_expression(parser, greater)
    }
    result
}
parse_expression <- function(parser, greater = TRUE) {
    left <- parse_conditional(parser, greater)
    if (left$kind %in% LVALUE_KINDS && current_token(parser)$spelling %in% ASSIGNMENTS) {
        operator <- take_token(parser)
        right <- parse_expression(parser, greater)
        return(awk_node("assignment", operator$position, list(left, right), operator = operator$spelling, location = left, value = right))
    }
    left
}

parse_lvalue <- function(parser) {
    if (current_token(parser)$kind == "name") {
        node <- parse_name(parser)
        if (!is.null(accept_token(parser, "["))) {
        }
        return(node)
    }
    dollar <- accept_token(parser, "$")
    if (!is.null(dollar)) {
        index <- if (current_token(parser)$spelling %in% c("++", "--")) parse_prefix(parser) else parse_atom(parser)
        return(awk_node("field", dollar$position, list(index), index = index))
    }
    syntax_error(parser)
}

input_value <- function(text) {
    characters <- strsplit(text, "", fixed = TRUE)[[1L]]
    beginning <- 1L
    finish <- length(characters)
    while (beginning <= finish && characters[beginning] %in% WHITESPACE) beginning <- beginning + 1L
    while (beginning <= finish && characters[finish] %in% WHITESPACE) finish <- finish - 1L
    if (beginning <= finish && characters[beginning] %in% c("+", "-")) beginning <- beginning + 1L
    numeric <- beginning <= finish && all(vapply(characters[seq.int(beginning, finish)], digit, logical(1L)))
    if (numeric) list(kind = "strnum", text = text, numeric = numeric_prefix(text)) else string_value(text)
}

literal_index <- function(text, sought, start = 1L) {
    if (sought == "" || nchar(sought) > nchar(text) - start + 1L) return(0L)
    for (index in seq.int(start, nchar(text) - nchar(sought) + 1L)) {
        if (substr(text, index, index + nchar(sought) - 1L) == sought) return(index)
    }
    0L
}
split_pieces <- function(text, separator) {
    if (text == "") return(character())
    if (separator == "") return(strsplit(text, "", fixed = TRUE)[[1L]])
    pieces <- character()
    if (separator != " ") {
        beginning <- 1L
        repeat {
            found <- literal_index(text, separator, beginning)
            if (found == 0L) break
            pieces <- c(pieces, if (found == beginning) "" else substr(text, beginning, found - 1L))
            beginning <- found + nchar(separator)
        }
        return(c(pieces, if (beginning > nchar(text)) "" else substring(text, beginning)))
    }

quoted_value <- function(text) {
    pieces <- vapply(strsplit(text, "", fixed = TRUE)[[1L]], function(character) {
        if (character == "\"") "\\\"" else if (character == "\\") "\\\\" else if (character == "\n") "\\n" else if (character == "\t") "\\t" else {
            code <- utf8ToInt(character)
            if (code >= 32L && code <= 126L) character else paste0("\\", sprintf("%03o", code))
        }
    }, character(1L))
    paste0("\"", paste0(pieces, collapse = ""), "\"")
}

}

parse_arguments <- function(lines) {
    options <- operands <- list()
    for (line in lines) {
        if (startsWith(line, "-F ")) {
            options[[length(options) + 1L]] <- list(name = "FS", value = substring(line, 4L))
        } else if (startsWith(line, "-v ")) {
            assignment <- substring(line, 4L)
            equals <- literal_index(assignment, "=")
            options[[length(options) + 1L]] <- list(name = substr(assignment, 1L, equals - 1L), value = substring(assignment, equals + 1L))
        } else {
            equals <- literal_index(line, "=")
            name <- if (equals > 0L) substr(line, 1L, equals - 1L) else ""
            operand <- if (equals > 0L && valid_name(name)) {
                record(kind = "assignment", name = name, value = substring(line, equals + 1L), records = 0L)
            } else record(kind = "file", name = line, value = "", records = 0L)
            operands[[length(operands) + 1L]] <- operand
        }
    }
    list(options = options, operands = operands)
}

    if (place$kind == "field") return(read_field(engine$record, place$index))
store <- function(engine, place, value, position) {
    force(value)
    if (place$kind == "field") write_field(engine$record, place$index, value) else if (place$kind == "array") {
    } else if (place$binding$scope != 0L) {
        engine$local_frames[[length(engine$local_frames)]][[place$binding$name]] <- value
    } else if (place$binding$name == "NF") {
        size <- to_number(value)
        if (size < 0L) runtime_issue(position, "negative NF")
        resize_fields(engine$record, size)
    } else engine$globals[[place$binding$name]] <- value
}
evaluate <- function(engine, node) {
    kind <- node$kind
    if (kind == "number") return(number(node$value))
    if (kind == "string") return(string_value(node$value))
    if (kind %in% LVALUE_KINDS) return(read_location(engine, location(engine, node)))
    if (kind == "paren") return(evaluate(engine, node$expression))
    if (kind == "assignment") {
        place <- location(engine, node$location)
        value <- evaluate(engine, node$value)
        if (node$operator != "=") value <- arithmetic(substr(node$operator, 1L, nchar(node$operator) - 1L), read_location(engine, place), value, node$position)
        store(engine, place, value, node$position)
        return(read_location(engine, place))
    }
execute_output <- function(engine, node) {
    arguments <- lapply(node$arguments, function(argument) evaluate(engine, argument))
    if (node$kind == "print") {
        if (!length(arguments)) arguments <- list(engine$record$zero)
        values <- vapply(arguments, to_string, character(1L))
        text <- paste0(paste0(values, collapse = to_string(global_value(engine, "OFS"))), to_string(global_value(engine, "ORS")))
    } else text <- format_output(arguments, node$position)
    filename <- if (is.null(node$filename)) NULL else to_string(evaluate(engine, node$filename))
    write_output(engine$streams, text, filename, node$redirect, node$redirect_position)
}
execute_statement <- function(engine, node) {
    kind <- node$kind
    if (kind == "block") {
        for (child in node$statements) execute_statement(engine, child)
        return(invisible(NULL))
    }
    if (kind == "empty") return(invisible(NULL))
    line <- as.character(node$position[1L])
    engine$lines[[line]] <- engine$lines[[line]] + 1L
    if (kind == "expression_statement") evaluate(engine, node$expression) else if (kind %in% c("print", "printf")) {
        execute_output(engine, node)
next_record <- function(engine) {
    repeat {
        if (engine$input_position <= length(engine$input_lines)) {
            line <- engine$input_lines[engine$input_position]
            engine$input_position <- engine$input_position + 1L
            engine$current_operand$records <- engine$current_operand$records + 1L
            engine$globals[["NR"]] <- number(to_number(global_value(engine, "NR")) + 1L)
            engine$globals[["FNR"]] <- number(to_number(global_value(engine, "FNR")) + 1L)
            return(line)
        }
        if (engine$operand_position > length(engine$operands)) return(NULL)
        operand <- engine$operands[[engine$operand_position]]
        engine$operand_position <- engine$operand_position + 1L
        if (operand$kind == "assignment") assign_global_input(engine, operand$name, operand$value) else {
            engine$current_operand <- operand
            engine$input_lines <- read_records(operand$name)
            engine$input_position <- 1L
            engine$globals[["FILENAME"]] <- string_value(operand$name)
            engine$globals[["FNR"]] <- number(0L)
        }
    }
}
run_actions <- function(engine, phase) {
    engine$phase <- phase
    for (rule in engine$analysis$rules) {
        if (rule$flavor == phase) {
            rule$count <- rule$count + 1L
            execute_statement(engine, rule$action)
        }
    }
}

run_records <- function(engine) {
    if (!any(vapply(engine$analysis$rules, function(rule) rule$flavor != "BEGIN", logical(1L)))) return(invisible(NULL))
    engine$phase <- "records"
    rules <- Filter(function(rule) !rule$flavor %in% c("BEGIN", "END"), engine$analysis$rules)
    repeat {
        line <- next_record(engine)
        if (is.null(line)) return(invisible(NULL))
        set_record(engine$record, line)
        tryCatch({
            for (rule in rules) {
                if (rule_matches(engine, rule)) {
                    rule$count <- rule$count + 1L
                    if (is.null(rule$action)) {
                        write_output(engine$streams, paste0(to_string(engine$record$zero), to_string(global_value(engine, "ORS"))))
                    } else execute_statement(engine, rule$action)
                }
            }
        }, awk_transfer = function(transfer) {
            if (transfer$kind != "next") stop(transfer)
        })
    }
}
execute <- function(engine) {
    tryCatch({
        for (option in engine$options) assign_global_input(engine, option$name, option$value)
        tryCatch({
            run_actions(engine, "BEGIN")
            run_records(engine)
        }, awk_transfer = function(transfer) {
            if (transfer$kind != "exit") stop(transfer)
        })
        tryCatch(run_actions(engine, "END"), awk_transfer = function(transfer) {
            if (transfer$kind != "exit") stop(transfer)
        })
    }, runtime_issue = function(error) engine$error <- error)
}
statistics <- function(engine) {
    status <- if (is.null(engine$error)) paste("status: exit", engine$exit_status) else {
        paste0("status: runtime error at ", engine$error$position[1L], ":", engine$error$position[2L], ": ", engine$error$message)
    }
    lines <- c(status, "[inputs]")
    for (operand in engine$operands) if (operand$kind == "file") lines <- c(lines, paste0(operand$name, " records=", operand$records))
    lines <- c(lines, "[rules]")
    for (rule in engine$analysis$rules) lines <- c(lines, paste0(rule$position[1L], ":", rule$position[2L], " ", rule$flavor, " count=", rule$count))
    lines <- c(lines, "[outputs]", paste0("stdout.txt writes=", engine$streams$stdout$writes, " bytes=", engine$streams$stdout$bytes))
    lines <- c(lines, "[lines]")
    for (line in sort(as.integer(ls(engine$lines)))) lines <- c(lines, paste0(line, " count=", engine$lines[[as.character(line)]]))
    lines <- c(lines, "[globals]")
    for (name in sort(ls(engine$global_names, all.names = TRUE), method = "radix")) {
        value <- global_value(engine, name)
        text <- if (exists(paste0("0:", name), engine$analysis$array_names, inherits = FALSE)) paste0("array (", value$size, ")") else describe_value(value)
        lines <- c(lines, paste(name, "=", text))
    }
    lines
}
main <- function() {
    options(expressions = 100000L)
    arguments <- parse_arguments(readLines("input/args.txt", warn = FALSE))
    source_lines <- readLines("input/program.awk", warn = FALSE)
    source <- if (length(source_lines)) paste0(paste0(source_lines, collapse = "\n"), "\n") else ""
    result <- tryCatch({
        items <- parse_program(new_parser(tokenize(source)))
        analysis <- prepare(new_analysis(items))
        engine <- new_engine(analysis, arguments$options, arguments$operands)
        execute(engine)
        list(lines = statistics(engine), streams = engine$streams)
    save_streams(result$streams)
    writeLines(result$lines, "output/stats.txt", useBytes = TRUE)
}
main()
