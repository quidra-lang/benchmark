        SyntaxIssue(Position pos) { this.pos = pos; }

    static final Set<String> KEYWORDS =
                                 Set.of("BEGIN", "END", "function", "if", "else", "while", "for",
                                        "do", "break", "continue", "next", "exit", "return",
                                        "delete", "in", "getline", "print", "printf"),
                             BUILTINS = Set.of("length", "substr", "index", "split", "sprintf",
                                               "toupper", "tolower", "close", "int"),
                             PAIRS = Set.of("+=", "-=", "*=", "/=", "%=", "^=", "==", "<=", ">=",
                                            "!=", "++", "--", "&&", "||", ">>");
    static boolean letter(char c) {
        return c >= 'a' && c <= 'z' || c >= 'A' && c <= 'Z' || c == '_';
    }
    static boolean digit(char c) { return c >= '0' && c <= '9'; }
    static List<Token> tokenize(String source) {
        List<Token> result = new ArrayList<>();
        int index = 0, line = 1, column = 1;
        while (index < source.length()) {
            char c = source.charAt(index);
            Position start = new Position(line, column);
            if (c == ' ') {
                index++;
                column++;
                continue;
            }
            if (c == '\\' && index + 1 < source.length() && source.charAt(index + 1) == '\n') {
                index += 2;
                line++;
                column = 1;
                continue;
            }
            if (c == '#') {
                int end = source.indexOf('\n', index);
                if (end < 0)
                    end = source.length();
                column += end - index;
                index = end;
                continue;
            }
            if (c == '\n') {
                result.add(new Token(Kind.NEWLINE, "\n", null, start));
                index++;
                line++;
                column = 1;
                continue;
            }
            int end = index + 1;
            Kind kind;
            String value = null;
            if (digit(c)) {
                while (end < source.length() && digit(source.charAt(end)))
                    end++;
                kind = Kind.NUMBER;
                value = source.substring(index, end);
            } else if (letter(c)) {
                while (end < source.length() &&
                       (letter(source.charAt(end)) || digit(source.charAt(end))))
                    end++;
                String word = source.substring(index, end);
                kind = KEYWORDS.contains(word)                              ? Kind.KEYWORD
                       : BUILTINS.contains(word)                            ? Kind.BUILTIN
                       : end < source.length() && source.charAt(end) == '(' ? Kind.FUNCNAME
                                                                            : Kind.NAME;
            } else if (c == '"') {
                StringBuilder text = new StringBuilder();
                while (end < source.length() && source.charAt(end) != '"') {
                    char next = source.charAt(end);
                    if (next == '\n')
                        throw new SyntaxIssue(start);
                    if (next == '\\') {
                        if (end + 1 == source.length())
                            throw new SyntaxIssue(new Position(line, column + end - index));
                        char escaped = source.charAt(end + 1);
                        if ("nt\\\"/".indexOf(escaped) < 0)
                            throw new SyntaxIssue(new Position(line, column + end - index));
                        text.append(switch (escaped) {
                            case 'n' -> '\n';
                            case 't' -> '\t';
                            default -> escaped;
                        });
                        end += 2;
                    } else {
                        text.append(next);
                        end++;
                    }
                }
                if (end == source.length())
                    throw new SyntaxIssue(start);
                end++;
                kind = Kind.STRING;
                value = text.toString();
            } else if (end < source.length() && PAIRS.contains(source.substring(index, end + 1))) {
                end++;
                kind = Kind.SYMBOL;
            } else if ("+-*/%^!><?:=()[]{},;$".indexOf(c) >= 0)
                kind = Kind.SYMBOL;
            else
                throw new SyntaxIssue(start);
            result.add(new Token(kind, source.substring(index, end), value, start));
            column += end - index;
            index = end;
        }
        result.add(new Token(Kind.END, "", null, new Position(line, 1)));
        return result;
    }
}

    static boolean whitespace(char c) { return c == ' ' || c == '\t' || c == '\n'; }
    static Value input(String text) {
        int start = 0, end = text.length();
        while (start < end && whitespace(text.charAt(start)))
            start++;
        while (end > start && whitespace(text.charAt(end - 1)))
            end--;
        int digits = start;
        if (digits < end && (text.charAt(digits) == '+' || text.charAt(digits) == '-'))
            digits++;
        boolean numeric = digits < end;
        for (int i = digits; i < end; i++)
            if (!Lexer.digit(text.charAt(i)))
                numeric = false;
        return numeric ? new Value(Kind.STRNUM, text, numericPrefix(text)) : string(text);
    }

        return value.kind() == Kind.STRING ? numericPrefix(value.text()) : value.numeric();
    }

        return switch (value.kind()) {
            case NUMBER -> Integer.toString(value.numeric());
            case UNINITIALIZED -> "";
            default -> value.text();
        };
    }
    static List<String> split(String text, String separator) {
        List<String> result = new ArrayList<>();
        if (text.isEmpty())
            return result;
        if (separator.isEmpty()) {
            for (int i = 0; i < text.length(); i++)
                result.add(text.substring(i, i + 1));
        } else if (separator.equals(" ")) {
                result.add(text.substring(start));
        } else {
            int start = 0, next;
            while ((next = text.indexOf(separator, start)) >= 0) {
                result.add(text.substring(start, next));
                start = next + separator.length();
            }
            result.add(text.substring(start));
        }
        return result;
    }
    static final class Record {
        final Function<String, Value> special;
        Value zero = string("");
        final List<Value> fields = new ArrayList<>();
        Record(Function<String, Value> special) { this.special = special; }
        void set(String text) {
            zero = input(text);
            fields.clear();
            for (String piece : split(text, text(special.apply("FS"))))
                fields.add(input(piece));
        }
        Value read(int index) {
            return index == 0        ? zero
            : index <= fields.size() ? fields.get(index - 1)
                                     : UNINITIALIZED;
        }
        void rebuild() {
            StringJoiner joined = new StringJoiner(text(special.apply("OFS")));
            for (Value value : fields)
                joined.add(text(value));
            zero = input(joined.toString());
        }
        void write(int index, Value value) {
            if (index == 0) {
                set(text(value));
                return;
            }
            while (fields.size() < index)
                fields.add(string(""));
            fields.set(index - 1, value);
            rebuild();
        }
        void resize(int size) {
            while (fields.size() > size)
                fields.removeLast();
            while (fields.size() < size)
                fields.add(string(""));
            rebuild();
        }
    }
    static String quoted(String text) {
        StringBuilder result = new StringBuilder("\"");
        for (int i = 0; i < text.length(); i++) {
            char c = text.charAt(i);
            result.append(switch (c) {
                case '"' -> "\\\"";
                case '\\' -> "\\\\";
                case '\n' -> "\\n";
                case '\t' -> "\\t";
                default ->
                    c >= 32 && c <= 126 ? String.valueOf(c)
                                        : "\\" + String.format(Locale.ROOT, "%03o", (int)c);
            });
        }
        return result.append('"').toString();
    }

            case UNINITIALIZED -> "(uninitialized)";

        };

        ValuesFields.Value read(Location location) {
            return switch (location) {
                case VariableLocation v -> variable(v.binding());
                case FieldLocation f -> record.read(f.index()) ;
        void store(Location location, ValuesFields.Value value, Lexer.Position pos) {
            switch (location) {
            case FieldLocation f -> record.write(f.index(), value) ;
            case VariableLocation v -> {
                Binding binding = v.binding();
                if (binding.scope() == null && binding.name().equals("NF")) {
                    int size = numeric(value);
                    if (size < 0)
                        throw new Builtins.RuntimeIssue(pos, "negative NF");
                    record.resize(size)
                    ;
                } else
                    cell(binding).scalar = value;
            }
            }
        }
        ValuesFields.Value evaluate(Parser.Expr expression) throws IOException {
            return switch (expression) {
                case Parser.Literal literal -> literal.value();
                case Parser.Variable v -> read(location(v));
                case Parser.ArrayAccess a -> read(location(a));
                case Parser.Field f -> read(location(f));
                case Parser.Paren p -> evaluate(p.expression());
                case Parser.Assignment a -> {
                    Location location = location(a.location());
                    ValuesFields.Value value = evaluate(a.value());
                    if (!a.operator().equals("="))
                        value = arithmetic(a.operator().substring(0, 1), read(location), value,
                                           a.pos());
                    store(location, value, a.pos());
                    yield read(location);
                }
        void output(Parser.Output output) throws IOException {
            List<ValuesFields.Value> values = new ArrayList<>();
            for (Parser.Expr expression : output.arguments())
                values.add(evaluate(expression));
            String content;
            if (output.formatted())
                content = Builtins.format(values, output.pos());
            else {
                if (values.isEmpty())
                    values.add(record.zero);
                StringJoiner joined = new StringJoiner(text(global("OFS")));
                for (ValuesFields.Value value : values)
                    joined.add(text(value));
                content = joined + text(global("ORS"));
            String filename = output.filename() == null ? null : text(evaluate(output.filename()));
            streams.write(content, filename, output.redirect(), output.redirectPos());
        }
        void statement(Parser.Stmt statement) throws IOException {
            if (statement instanceof Parser.Block block) {
                for (Parser.Stmt child : block.statements())
                    statement(child);
                return;
            }
            if (statement instanceof Parser.Empty)
                return;
            lines.merge(statement.pos().line(), 1, Integer::sum);
            switch (statement) {
            case Parser.ExpressionStatement expression -> evaluate(expression.expression());
            case Parser.Output output -> output(output);
        String nextRecord() throws IOException {
            while (true) {
                if (inputIndex < inputLines.size()) {
                    String line = inputLines.get(inputIndex++);
                    current.records++;
                    globals.get("NR").scalar = number(numeric(global("NR")) + 1);
                    globals.get("FNR").scalar = number(numeric(global("FNR")) + 1);
                    return line;
                }
                if (operandIndex == arguments.operands().size())
                    return null;
                Streams.Operand operand = arguments.operands().get(operandIndex++);
                if (!operand.file)
                    assignInput(operand.name, operand.value);
                else {
                    current = operand;
                    inputLines = Streams.records(operand.name);
                    inputIndex = 0;
                    globals.get("FILENAME").scalar = string(operand.name);
                    globals.get("FNR").scalar = number(0);
                }
            }
        }
        void actions(String phase) throws IOException {
            this.phase = phase;
            for (Parser.Rule rule : analysis.rules)
                if (rule.flavor.equals(phase)) {
                    rule.count++;
                    statement(rule.action);
                }
        }
        boolean matches(Parser.Rule rule) throws IOException {
            if (rule.flavor.equals("all"))
                return true;
        void records() throws IOException {
            if (analysis.rules.stream().noneMatch(rule -> !rule.flavor.equals("BEGIN")))
                return;
            phase = "records";
            String line;
            while ((line = nextRecord()) != null) {
                record.set(line)
                ;
                try {
                    for (Parser.Rule rule : analysis.rules)
                        if (!rule.flavor.equals("BEGIN") && !rule.flavor.equals("END") &&
                            matches(rule)) {
                            rule.count++;
                            if (rule.action == null)
                                streams.write(text(record.zero) + text(global("ORS")), null, null,
                                              null);
                            else
                                statement(rule.action);
                        }
                } catch (Transfer transfer) {
                    if (transfer.flow != Flow.NEXT)
                        throw transfer;
                }
            }
        }

            try {
                for (Streams.Option option : arguments.options())
                    assignInput(option.name(), option.value());
                try {
                    actions("BEGIN");
                    records();
        List<String> statistics() {
            List<String> result = new ArrayList<>();
            result.add(error == null
                           ? "status: exit " + exitStatus
                           : "status: runtime error at " + error.pos + ": " + error.getMessage());
            result.add("[inputs]");
            for (Streams.Operand operand : arguments.operands())
                if (operand.file)
                    result.add(operand.name + " records=" + operand.records);
            result.add("[rules]");
            for (Parser.Rule rule : analysis.rules)
                result.add(rule.pos + " " + rule.flavor + " count=" + rule.count);
            result.add("[outputs]");
            result.add("stdout.txt writes=" + streams.stdout.writes +
                       " bytes=" + streams.stdout.content.length());
            result.add("[lines]");
            for (var entry : lines.entrySet())
                result.add(entry.getKey() + " count=" + entry.getValue());
            result.add("[globals]");
            for (String name : globalNames) {
                Binding binding = new Binding(null, name);
                result.add(name + " = " +
                           (analysis.arrays.contains(binding)
                                ? "array (" + globals.get(name).array.size() + ")"
                                : ValuesFields.describe(variable(binding))));
            }
            return result;
        }
    }
    public static void main(String[] args) throws IOException {
        Streams.Arguments arguments =
            Streams.parse(Files.readString(Path.of("input/args.txt"), StandardCharsets.US_ASCII));
        String source = Files.readString(Path.of("input/program.awk"), StandardCharsets.US_ASCII);
        Streams streams;
        List<String> statistics;
        try {
            Analysis analysis = new Analysis(new Parser(Lexer.tokenize(source)).parse()).prepare();
            Engine engine = new Engine(analysis, arguments);
            engine.execute();
            streams = engine.streams;
            statistics = engine.statistics();
        streams.save();
        Files.writeString(Path.of("output/stats.txt"), String.join("\n", statistics) + "\n",
                          StandardCharsets.US_ASCII);
    }
