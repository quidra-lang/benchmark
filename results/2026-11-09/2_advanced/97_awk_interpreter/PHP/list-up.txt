function tokenize(string $source): array
{
    $tokens = [];
    $i = 0;
    $line = $column = 1;
    $n = strlen($source);
    while ($i < $n) {
        $ch = $source[$i];
        $start = [$line, $column];
        if ($ch === ' ') {
            ++$i;
            ++$column;
            continue;
        }
        if (substr($source, $i, 2) === "\\\n") {
            $i += 2;
            ++$line;
            $column = 1;
            continue;
        }
        if ($ch === '#') {
            $end = strpos($source, "\n", $i);
            if ($end === false)
                $end = $n;
            $column += $end - $i;
            $i = $end;
            continue;
        }
        if ($ch === "\n") {
            $tokens[] = new Token('newline', "\n", null, $start);
            ++$i;
            ++$line;
            $column = 1;
            continue;
        }
        $end = $i + 1;
        $value = null;
        if (ctype_digit($ch)) {
            while ($end < $n && ctype_digit($source[$end]))
                ++$end;
            $kind = 'number';
            $value = (int) substr($source, $i, $end - $i);
        } elseif (ctype_alpha($ch) || $ch === '_') {
            while ($end < $n && (ctype_alnum($source[$end]) || $source[$end] === '_'))
                ++$end;
            $word = substr($source, $i, $end - $i);
            $kind = in_array($word, KEYWORDS, true)
                ? 'keyword'
                : (
                    in_array($word, BUILTINS, true)
                        ? 'builtin'
                        : ($end < $n && $source[$end] === '(' ? 'function_name' : 'name')
                );
        } elseif ($ch === '"') {
            $value = '';
            while ($end < $n && $source[$end] !== '"') {
                if ($source[$end] === "\n")
                    throw new SyntaxIssue($start);
                if ($source[$end] === '\\') {
                    if (($end + 1) === $n || !isset(ESCAPES[$source[$end + 1]]))
                        throw new SyntaxIssue([$line, $column + $end - $i]);
                    $value .= ESCAPES[$source[$end + 1]];
                    $end += 2;
                } else
                    $value .= $source[$end++];
            }
            if ($end === $n)
                throw new SyntaxIssue($start);
            ++$end;
            $kind = 'string';
        } elseif (in_array(substr($source, $i, 2), PAIRS, true)) {
            ++$end;
            $kind = 'symbol';
        } elseif (str_contains('+-*/%^!><?:=()[]{},;$', $ch))
            $kind = 'symbol';
        else
            throw new SyntaxIssue($start);
        $tokens[] = new Token($kind, substr($source, $i, $end - $i), $value, $start);
        $column += $end - $i;
        $i = $end;
    }
    $tokens[] = new Token('end', '', null, [substr_count($source, "\n") + 1, 1]);
    return $tokens;
}

final class AwkParser
{
    private int $index = 0, $loopDepth = 0;
    private bool $function = false;
    private ?string $phase = null;

    public function __construct(
        private array $tokens,
    ) {}

    private function peek(): Token
    {
        return $this->tokens[$this->index];
    }

    private function take(): Token
    {
        return $this->tokens[$this->index++];
    }

    private function fail(?array $p = null): never
    {
        throw new SyntaxIssue($p ?? $this->peek()->position);
    }

    private function accept(string $s): ?Token
    {
        return $this->peek()->spelling === $s ? $this->take() : null;
    }

    private function expect(string $s): Token
    {
        if ($this->peek()->spelling !== $s)
            $this->fail();
        return $this->take();
    }

    private function name(): object
    {
        if ($this->peek()->kind !== 'name')
            $this->fail();
        $t = $this->take();
        return node('name', $t->position, fields: ['name' => $t->spelling]);
    }

    private function newlines(): void
    {
        while ($this->accept("\n") !== null) {
        }
    }

    public function parse(): array
    {
        $items = [];
        while ($this->peek()->kind !== 'end') {
            if (in_array($this->peek()->spelling, ["\n", ';'], true)) {
                $this->take();
                continue;
            }
            $start = $this->peek();
            }
            $patterns = [];
            if ($start->spelling === 'BEGIN' || $start->spelling === 'END')
                $flavor = $this->take()->spelling;
            elseif ($start->spelling === '{')
                $flavor = 'all';
            }
            $this->phase = $flavor;
            if (in_array($flavor, ['BEGIN', 'END', 'all'], true) || $this->peek()->spelling === '{')
                $action = $this->block();
            $this->phase = null;
            $items[] = node(
                'rule',
                $start->position,
                [...$patterns, $action],
                compact('flavor', 'patterns', 'action'),
            );
        }
        return $items;
    }

    private function block(): object
    {
        $open = $this->expect('{');
        $statements = [];
        while ($this->peek()->spelling !== '}') {
            if (in_array($this->peek()->spelling, ["\n", ';'], true))
                $this->take();
            else
                $statements[] = $this->statement();
        }
        $this->expect('}');
        return node('block', $open->position, $statements, compact('statements'));
    }
    private function terminator(): void
    {
        if (in_array($this->peek()->spelling, [';', "\n"], true))
            $this->take();
        elseif ($this->peek()->spelling !== '}')
            $this->fail();
    }

    private function statement(): object
    {
        $t = $this->peek();
        $s = $t->spelling;
        if ($s === '{')
            return $this->block();
        if ($this->accept(';') !== null)
            return node('empty', $t->position);
        if ($this->accept('if') !== null) {
            $condition = $this->condition();
            $this->newlines();
        if ($s === 'print' || $s === 'printf') {
            $this->take();
            $result = $this->outputStatement($t);
        } else {
            $expression = $this->expression();
            $result = node('expression_statement', $t->position, [$expression], compact('expression'));
        }
        $this->terminator();
        return $result;
    }

    private function parenthesizedOutput(): bool
    {
        if ($this->peek()->spelling !== '(')
            return false;
        $depth = 0;
    private function outputStatement(Token $t): object
    {
        $arguments = [];
        if ($this->parenthesizedOutput()) {
            $this->expect('(');
            $arguments = $this->expressionList();
            $this->expect(')');
        } elseif (!in_array($this->peek()->spelling, [';', "\n", '}', '>', '>>'], true))
            $arguments = $this->expressionList(false);
        if ($t->spelling === 'printf' && !$arguments)
            $this->fail();
        $redirect = in_array($this->peek()->spelling, ['>', '>>'], true) ? $this->take() : null;
        $filename = $redirect === null ? null : $this->concatenation();
        return node(
            $t->spelling,
            $t->position,
            [...$arguments, $filename],
            [
                'arguments' => $arguments,
                'redirect' => $redirect?->spelling,
                'redirectPosition' => $redirect?->position,
                'filename' => $filename,
            ],
        );
    }

    private function expressionList(bool $greater = true): array
    {
        $r = [$this->expression($greater)];
        while ($this->accept(',') !== null) {
            $this->newlines();
            $r[] = $this->expression($greater);
        }
        return $r;
    }

    private function expression(bool $greater = true): object
    {
        $left = $this->conditional($greater);
        if (
            in_array($left->kind, LVALUE_KINDS, true) && in_array($this->peek()->spelling, ASSIGNMENTS, true)
        ) {
            $op = $this->take();
            $right = $this->expression($greater);
            return node(
                'assignment',
                $op->position,
                [$left, $right],
                ['operator' => $op->spelling, 'location' => $left, 'value' => $right],
            );
        }
        return $left;
    }

    private function conditional(bool $greater): object
    {
        $condition = $this->logical(0, $greater);
        $op = $this->accept('?');
        if ($op === null)
            return $condition;
    }
    private function logical(int $level, bool $greater): object
    {
        if ($level === 2)
            return $this->membership($greater);
        $op = $level === 0 ? '||' : '&&';
        $left = $this->logical($level + 1, $greater);
        while ($this->peek()->spelling === $op) {
        }
        return $left;
    }

    private function membership(bool $greater): object
    {
        $left = $this->comparison($greater);
        while ($this->peek()->spelling === 'in') {
        }
        return $left;
    }

    private function comparison(bool $greater): object
    {
        $left = $this->concatenation();
        while (
            in_array($this->peek()->spelling, COMPARISONS, true)
            && ($greater || $this->peek()->spelling !== '>')
        ) {
        }
        return $left;
    }

    private function concatenation(): object
    {
        $left = $this->additive();
        while (
            in_array($this->peek()->kind, ['number', 'string', 'name', 'function_name', 'builtin'], true)
            || in_array($this->peek()->spelling, ['$', '(', '++', '--'], true)
        ) {
        }
        return $left;
    }

    private function additive(): object
    {
        $left = $this->multiplicative();
        while (in_array($this->peek()->spelling, ['+', '-'], true)) {
        }
        return $left;
    }

    private function multiplicative(): object
    {
        $left = $this->unary();
        while (in_array($this->peek()->spelling, ['*', '/', '%'], true)) {
        }
        return $left;
    }

    private function unary(): object
    {
        if (in_array($this->peek()->spelling, ['!', '-', '+'], true)) {
        }
        $left = $this->postfix();
        if ($this->peek()->spelling === '^') {
        }
        return $left;
    }
    private function postfix(): object
    {
        if (in_array($this->peek()->spelling, ['++', '--'], true))
            return $this->prefix();
        $left = $this->atom();
        if (
            in_array($left->kind, LVALUE_KINDS, true) && in_array($this->peek()->spelling, ['++', '--'], true)
        ) {
        }
        return $left;
    }

    private function lvalue(): object
    {
        if ($this->peek()->kind === 'name') {
            $n = $this->name();
            if ($this->accept('[') !== null) {
            }
            return $n;
        }
        $dollar = $this->accept('$');
        if ($dollar !== null) {
            $index = in_array($this->peek()->spelling, ['++', '--'], true) ? $this->prefix() : $this->atom();
            return node('field', $dollar->position, [$index], compact('index'));
        }
        $this->fail();
    }

    private function atom(): object
    {
        $t = $this->peek();
        if ($t->kind === 'number' || $t->kind === 'string') {
            $this->take();
            return node($t->kind, $t->position, fields: ['value' => $t->value]);
        }
        if ($t->kind === 'name' || $t->spelling === '$')
            return $this->lvalue();

function numericPrefix(string $s): int
{
    return preg_match('/^[ \t\n]*([+-]?[0-9]+)/D', $s, $m) ? (int) $m[1] : 0;
}

function inputValue(string $s): AwkValue
{
    return preg_match('/^[ \t\n]*[+-]?[0-9]+[ \t\n]*$/D', $s)
        ? new AwkValue('strnum', $s, numericPrefix($s))
        : vString($s);
}

function toNumber(AwkValue $v): int
{
    return $v->kind === 'string' ? numericPrefix($v->text) : $v->numeric;
}

function toString(AwkValue $v): string
{
    return $v->kind === 'number' ? (string) $v->numeric : $v->text ?? '';
}

function splitPieces(string $text, string $separator): array
{
    if ($text === '')
        return [];
    if ($separator === '')
        return str_split($text);
    if ($separator !== ' ')
        return explode($separator, $text);
    return preg_split('/[ \t\n]+/', trim($text, " \t\n"), -1, PREG_SPLIT_NO_EMPTY);
}

final class Record
{
    public AwkValue $zero;
    public array $fields = [];

    public function __construct(
        private Closure $special,
    ) {
        $this->zero = vString('');
    }

    public function set(string $text): void
    {
        $this->zero = inputValue($text);
        $this->fields = array_map(inputValue(...), splitPieces($text, toString(($this->special)('FS'))));
    }

    public function read(int $index): AwkValue
    {
        return $index === 0 ? $this->zero : $this->fields[$index - 1] ?? uninitialized();
    }

    private function rebuild(): void
    {
        $this->zero = inputValue(implode(
            toString(($this->special)('OFS')),
            array_map(toString(...), $this->fields),
        ));
    }

    public function write(int $index, AwkValue $v): void
    {
        if ($index === 0) {
            $this->set(toString($v));
            return;
        }
        while (count($this->fields) < $index)
            $this->fields[] = vString('');
        $this->fields[$index - 1] = $v;
        $this->rebuild();
    }

    public function resize(int $size): void
    {
        $this->fields = array_slice($this->fields, 0, $size);
        while (count($this->fields) < $size)
            $this->fields[] = vString('');
        $this->rebuild();
    }
}
function quoted(string $s): string
{
    $out = '"';
    $escapes = ['"' => '\\"', '\\' => '\\\\', "\n" => '\\n', "\t" => '\\t'];
    foreach (str_split($s) as $c) {
        $code = ord($c);
        $out .=
            $escapes[$c]
            ?? ($code >= 32 && $code <= 126 ? $c : '\\' . str_pad(decoct($code), 3, '0', STR_PAD_LEFT));
    }
    return $out . '"';
}

    };

function parseArguments(array $lines): array
{
    $options = $operands = [];
    foreach ($lines as $line) {
        if (str_starts_with($line, '-F '))
            $options[] = ['FS', substr($line, 3)];
        elseif (str_starts_with($line, '-v '))
            $options[] = explode('=', substr($line, 3), 2);
        else {
            $equal = strpos($line, '=');
            if ($equal !== false && validName(substr($line, 0, $equal)))
                $operands[] = new Operand('assignment', substr($line, 0, $equal), substr($line, $equal + 1));
            else
                $operands[] = new Operand('file', $line);
        }
    }
    return [$options, $operands];
}

            elseif ($name === 'NF') {
                $size = toNumber($value);
                if ($size < 0)
                    throw new RuntimeIssue($position, 'negative NF');
                $this->record->resize($size);
            } else
                $this->globals[$name] = $value;
        }
    }

    private function evaluate(object $n): AwkValue
    {
        switch ($n->kind) {
            case 'number':
                return vNumber($n->value);
            case 'string':
                return vString($n->value);
            case 'name':
            case 'array':
            case 'field':
                return $this->read($this->location($n));
            case 'paren':
                return $this->evaluate($n->expression);
            case 'assignment':
                $location = $this->location($n->location);
                $value = $this->evaluate($n->value);
                if ($n->operator !== '=')
                    $value = arithmetic(
                        substr($n->operator, 0, -1),
                        $this->read($location),
                        $value,
                        $n->position,
                    );
                $this->store($location, $value, $n->position);
                return $this->read($location);
    private function output(object $n): void
    {
        $arguments = [];
        foreach ($n->arguments as $arg)
            $arguments[] = $this->evaluate($arg);
        if ($n->kind === 'print') {
            if (!$arguments)
                $arguments = [$this->record->zero];
            $text =
                implode(toString($this->globalValue('OFS')), array_map(toString(...), $arguments))
                . toString($this->globalValue('ORS'));
        $filename = $n->filename === null ? null : toString($this->evaluate($n->filename));
        $this->streams->write($text, $filename, $n->redirect, $n->redirectPosition);
    }
    private function statement(object $n): void
    {
        if ($n->kind === 'block') {
            foreach ($n->statements as $s)
                $this->statement($s);
            return;
        }
        if ($n->kind === 'empty')
            return;
        ++$this->lines[$n->position[0]];
        switch ($n->kind) {
            case 'expression_statement':
                $this->evaluate($n->expression);
                break;
            case 'print':
            case 'printf':
                $this->output($n);
                break;
    private function nextRecord(): ?string
    {
        while (true) {
            if ($this->inputPosition < count($this->inputLines)) {
                $line = $this->inputLines[$this->inputPosition++];
                ++$this->currentOperand->records;
                $this->globals['NR'] = vNumber(toNumber($this->globalValue('NR')) + 1);
                $this->globals['FNR'] = vNumber(toNumber($this->globalValue('FNR')) + 1);
                return $line;
            }
            if ($this->operandPosition === count($this->operands))
                return null;
            $operand = $this->operands[$this->operandPosition++];
            if ($operand->kind === 'assignment')
                $this->assignGlobalInput($operand->name, $operand->value);
            else {
                $this->currentOperand = $operand;
                $this->inputLines = readRecords($operand->name);
                $this->inputPosition = 0;
                $this->globals['FILENAME'] = vString($operand->name);
                $this->globals['FNR'] = vNumber(0);
            }
        }
    }

    private function runActions(string $phase): void
    {
        $this->phase = $phase;
        foreach ($this->analysis->rules as $r)
            if ($r->flavor === $phase) {
                ++$r->count;
                $this->statement($r->action);
            }
    }

    private function matches(object $r): bool
    {
        if ($r->flavor === 'all')
            return true;
    private function runRecords(): void
    {
        if (!array_any($this->analysis->rules, fn($r) => $r->flavor !== 'BEGIN'))
            return;
        $this->phase = 'records';
        $rules = array_values(array_filter(
            $this->analysis->rules,
            fn($r) => !in_array($r->flavor, ['BEGIN', 'END'], true),
        ));
        while (($line = $this->nextRecord()) !== null) {
            $this->record->set($line);
            try {
                foreach ($rules as $r)
                    if ($this->matches($r)) {
                        ++$r->count;
                        if ($r->action === null)
                            $this->streams->write(
                                toString($this->record->zero) . toString($this->globalValue('ORS')),
                            );
                        else
                            $this->statement($r->action);
                    }
            } catch (Transfer $t) {
                if ($t->kind !== 'next')
                    throw $t;
            }
        }
    }

    public function execute(): void
    {
        try {
            foreach ($this->options as [$name, $text])
                $this->assignGlobalInput($name, $text);
            try {
                $this->runActions('BEGIN');
                $this->runRecords();
    public function statistics(): array
    {
        $out = [
            $this->error === null
                ? 'status: exit ' . $this->exitStatus
                : 'status: runtime error at '
                . implode(':', $this->error->position)
                . ': '
                . $this->error->getMessage(),
            '[inputs]',
        ];
        foreach ($this->operands as $o)
            if ($o->kind === 'file')
                $out[] = "{$o->name} records={$o->records}";
        $out[] = '[rules]';
        foreach ($this->analysis->rules as $r)
            $out[] = implode(':', $r->position) . " {$r->flavor} count={$r->count}";
        $out[] = '[outputs]';
        $s = $this->streams->stdout;
        $out[] = "stdout.txt writes={$s->writes} bytes=" . strlen($s->content);
        $out[] = '[lines]';
        ksort($this->lines, SORT_NUMERIC);
        foreach ($this->lines as $line => $count)
            $out[] = "$line count=$count";
        $out[] = '[globals]';
        $names = array_keys($this->globalNames);
        sort($names, SORT_STRING);
        foreach ($names as $name) {
            $v = $this->globalValue($name);
            $text = isset($this->analysis->arrayNames['g:' . $name])
                ? 'array (' . count($v->elements) . ')'
                : describe($v);
            $out[] = "$name = $text";
        }
        return $out;
    }
}
