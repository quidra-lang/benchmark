bool letter(int c) => c >= 65 && c <= 90 || c >= 97 && c <= 122;
bool digit(int c) => c >= 48 && c <= 57;
bool wordIn(String words, String value) => words.split(' ').contains(value);
int decimal(String text) => int.parse(text);
List<Token> tokenize(String source) {
  final tokens = <Token>[];
  var i = 0, line = 1, column = 1;
  while (i < source.length) {
    var c = source.codeUnitAt(i);
    final position = Position(line, column);
    if (c == 32) {
      i++;
      column++;
      continue;
    }
    if (c == 92 && i + 1 < source.length && source.codeUnitAt(i + 1) == 10) {
      i += 2;
      line++;
      column = 1;
      continue;
    }
    if (c == 35) {
      while (i < source.length && source.codeUnitAt(i) != 10) {
        i++;
        column++;
      }
      continue;
    }
    if (c == 10) {
      tokens.add(Token('newline', '\n', position));
      i++;
      line++;
      column = 1;
      continue;
    }
    if (digit(c)) {
      final start = i;
      while (i < source.length && digit(source.codeUnitAt(i))) {
        i++;
        column++;
      }
      final text = source.substring(start, i);
      tokens.add(Token('number', text, position, number: decimal(text)));
      continue;
    }
    if (letter(c) || c == 95) {
      final start = i;
      while (i < source.length &&
          (letter(source.codeUnitAt(i)) ||
              digit(source.codeUnitAt(i)) ||
              source.codeUnitAt(i) == 95)) {
        i++;
        column++;
      }
      final text = source.substring(start, i);
      var kind = 'name';
      if (wordIn(
        'BEGIN END function if else while for do break continue next exit return delete in getline print printf',
        text,
      )) {
        kind = 'keyword';
      } else if (wordIn(
        'length substr index split sprintf toupper tolower close int',
        text,
      )) {
        kind = 'builtin';
      } else if (i < source.length && source[i] == '(') {
        kind = 'function_name';
      }
      tokens.add(Token(kind, text, position));
      continue;
    }
    if (c == 34) {
      final start = i;
      i++;
      column++;
      final value = StringBuffer();
      while (true) {
        if (i == source.length || source.codeUnitAt(i) == 10)
          throw SyntaxIssue(position);
        c = source.codeUnitAt(i++);
        column++;
        if (c == 34) break;
        if (c == 92) {
          final slash = Position(line, column - 1);
          if (i == source.length) throw SyntaxIssue(slash);
          c = source.codeUnitAt(i++);
          column++;
          c = switch (c) {
            110 => 10,
            116 => 9,
            92 || 34 || 47 => c,
            _ => throw SyntaxIssue(slash),
          };
        }
        value.writeCharCode(c);
      }
      tokens.add(
        Token(
          'string',
          source.substring(start, i),
          position,
          text: value.toString(),
        ),
      );
      continue;
    }
    var symbol = '';
    if (i + 1 < source.length &&
        wordIn(
          '+= -= *= /= %= ^= == <= >= != ++ -- && || >>',
          source.substring(i, i + 2),
        )) {
      symbol = source.substring(i, i + 2);
    } else if ('+-*/%^!><?:=()[]{},;\$'.contains(source[i])) {
      symbol = source[i];
    }
    if (symbol.isEmpty) throw SyntaxIssue(position);
    tokens.add(Token('symbol', symbol, position));
    i += symbol.length;
    column += symbol.length;
  }
  tokens.add(
    Token(
      'end',
      '',
      Position(source.codeUnits.where((c) => c == 10).length + 1, 1),
    ),
  );
  return tokens;
}

class Value {

Value numeric(int n) => Value('number', n);

bool whitespace(int c) => c == 32 || c == 9 || c == 10;
(int, int, bool) numericPrefix(String text) {
  var i = 0, sign = 1;
  while (i < text.length && whitespace(text.codeUnitAt(i))) i++;
  if (i < text.length && (text[i] == '+' || text[i] == '-')) {
    if (text[i] == '-') sign = -1;
    i++;
  }
  final start = i;
  while (i < text.length && digit(text.codeUnitAt(i))) i++;
  if (i == start) return (0, i, false);
  return (sign * decimal(text.substring(start, i)), i, true);
}

Value inputValue(String text) {
  final (n, end, valid) = numericPrefix(text);
  var last = end;
  while (last < text.length && whitespace(text.codeUnitAt(last))) last++;
  return valid && last == text.length
      ? Value('strnum', n, text)
      : stringValue(text);
}

class AwkRecord {
  Value zero = stringValue('');
  final List<Value> fields = [];
  final Value Function(String) global;
  AwkRecord(this.global);
  void set(String text) {
    zero = inputValue(text);
    fields.clear();
    fields.addAll(splitPieces(text, global('FS').text).map(inputValue));
  }

  Value read(int index) => index == 0
      ? zero
      : index > fields.length
      ? const Value()
      : fields[index - 1];
  void rebuild() {
    zero = inputValue(fields.map((v) => v.text).join(global('OFS').text));
  }

  void resize(int n) {
    if (n < fields.length) fields.removeRange(n, fields.length);
    while (fields.length < n) fields.add(stringValue(''));
    rebuild();
  }

  void write(int index, Value value) {
    if (index == 0) {
      set(value.text);
      return;
    }
    while (fields.length < index) fields.add(stringValue(''));
    fields[index - 1] = value;
    rebuild();
  }
}
String escapedString(String text) {
  final out = StringBuffer('"');
  for (final c in text.codeUnits) {
    out.write(switch (c) {
      92 => r'\\',
      34 => r'\"',
      10 => r'\n',
      9 => r'\t',
      _ =>
        c >= 32 && c <= 126
            ? String.fromCharCode(c)
            : '\\${c.toRadixString(8).padLeft(3, '0')}',
    });
  }
  out.write('"');
  return out.toString();
}

(List<Option>, List<Operand>) parseArguments(List<String> lines) {
  final options = <Option>[];
  final operands = <Operand>[];
  for (final line in lines) {
    if (line.startsWith('-F ')) {
      options.add(Option('FS', line.substring(3)));
      continue;
    }
    if (line.startsWith('-v ')) {
      final equal = line.indexOf('=', 3);
      options.add(Option(line.substring(3, equal), line.substring(equal + 1)));
      continue;
    }
    final equal = line.indexOf('=');
    if (equal >= 0 && validName(line.substring(0, equal))) {
      operands.add(
        Operand(
          'assignment',
          line.substring(0, equal),
          line.substring(equal + 1),
        ),
      );
    } else {
      operands.add(Operand('file', line));
    }
  }
  return (options, operands);
}

class Analysis {

  final List<Item> functions = [], rules = [];

    for (final item in items)
      (item.kind == 'function' ? functions : rules).add(item);
  }
  void expression(Expr? node, Item? function) {
    if (node == null) return;
    expressions.add(node);
    if (node.kind == 'name' ||
        node.kind == 'array' ||
        node.kind == 'parameter') {
      if (function != null && function.parameterNames.containsKey(node.name)) {
        node.binding = (function, node.name);
      } else {
        node.binding = (null, node.name);
        globals.add(node.name);
      }
      if (functionTable.containsKey(node.name))
        issue(node.position, 2, 'function ${node.name} used as variable');
    } else if (node.kind == 'call') {
    expression(node.left, function);
    expression(node.right, function);
    expression(node.third, function);
    for (final arg in node.args) expression(arg, function);
  }

  void statement(Statement? node, Item? function) {
    if (node == null) return;
    statements.add(node);
    if (node.kind != 'block' && node.kind != 'empty')
      lines.add(node.position.line);
    for (final expr in node.exprs) expression(expr, function);
    expression(node.variable, function);
    expression(node.array, function);
    expression(node.redirect, function);
    for (final child in node.statements) statement(child, function);
    statement(node.body, function);
    statement(node.otherwise, function);
  }
  void prepare() {
    for (final item in items) {
      final function = item.kind == 'function' ? item : null;
      for (final parameter in item.parameters) expression(parameter, function);
      for (final pattern in item.patterns) expression(pattern, function);
      statement(item.body, function);
    }
    if (errors.isNotEmpty) {
      errors.sort((a, b) {
        if (a.position.line != b.position.line)
          return a.position.line.compareTo(b.position.line);
        if (a.position.column != b.position.column)
          return a.position.column.compareTo(b.position.column);
        return a.priority.compareTo(b.priority);
      });
      throw errors.first;
    }

    try {
      for (final option in options) assignInput(option.name, option.value);
      try {
        actions('BEGIN');
        records();
  List<String> statistics() {
    final failure = error;
    final result = [
      failure == null
          ? 'status: exit $exitStatus'
          : 'status: runtime error at ${failure.position}: ${failure.message}',
      '[inputs]',
    ];
    for (final operand in operands) {
      if (operand.kind == 'file')
        result.add('${operand.name} records=${operand.records}');
    }
    result.add('[rules]');
    for (final rule in analysis.rules)
      result.add('${rule.position} ${rule.flavor} count=${rule.count}');
    result.add('[outputs]');
    result.add(
      'stdout.txt writes=${streams.stdout.writes} bytes=${streams.stdout.content.length}',
    );
    result.add('[lines]');
    final lineNumbers = lines.keys.toList()..sort();
    for (final line in lineNumbers) result.add('$line count=${lines[line]}');
    result.add('[globals]');
    final globalNames = globals.keys.toList()..sort();
    for (final name in globalNames)
      result.add(
        '$name = ${analysis.arrays.contains((null, name)) ? 'array (${globals[name]!.array!.length})' : describe(global(name))}',
      );
    return result;
  }
}

void main() {
  final (options, operands) = parseArguments(
    textLines(readText('input/args.txt')),
  );
  var streams = Streams();
  List<String> statistics;
  try {
    final parser = Parser(tokenize(readText('input/program.awk')));
    final analysis = Analysis(parser.parse());
    analysis.prepare();
    final engine = Engine(analysis, options, operands);
    engine.execute();
    streams = engine.streams;
    statistics = engine.statistics();
  streams.save();
  File('output/stats.txt').writeAsStringSync('${statistics.join('\n')}\n');
}
