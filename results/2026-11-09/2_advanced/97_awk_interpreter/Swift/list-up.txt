struct Position: Error, Hashable, Comparable, CustomStringConvertible {
    let line: Int, column: Int
    static func < (a: Self, b: Self) -> Bool {
        (a.line, a.column) < (b.line, b.column)
    };
    var description: String {
        "\(line):\(column)"
    }
}

enum Lexer {
    static func letter(_ c: UInt8) -> Bool {
        (65...90).contains(c) || (97...122).contains(c)
    }
    static func digit(_ c: UInt8) -> Bool {
        (48...57).contains(c)
    }
    static func lex(_ text: String) throws -> [Token] {
        let bytes = Array(text.utf8);
        var positions: [Position] = [], line = 1, column = 1
        for c in bytes {
            positions.append(Position(line: line, column: column));
            if c == 10 {
                line += 1;
                column = 1
            } else {
                column += 1
            }
        };
        positions.append(Position(line: line, column: column))
        var result: [Token] = [], i = 0
        func word(_ a: Int, _ b: Int) -> String {
            String(decoding: bytes[a..<b], as: UTF8.self)
        }
        while i < bytes.count {
            if bytes[i] == 32 {
                i += 1;
                continue
            }
            let start = i;
            var kind = "symbol", value = ""
            if bytes[i] == 10 {
                i += 1;
                kind = "newline"
            }
            else if digit(bytes[i]) {
                i += 1;
                while i < bytes.count && digit(bytes[i]) {
                    i += 1
                };
                kind = "number"
            }
            else if letter(bytes[i]) || bytes[i] == 95 {
                i += 1;
                while i < bytes.count && (letter(bytes[i]) || digit(bytes[i]) || bytes[i] == 95) {
                    i += 1
                };
                let name = word(start, i)
                if ["BEGIN", "END", "function", "if", "else", "while", "for", "do", "break", "continue", "next", "exit", "return", "delete", "in", "getline", "print", "printf"].contains(name) {
                    kind = "keyword"
                }
                else if ["length", "substr", "index", "split", "sprintf", "toupper", "tolower", "close", "int"].contains(name) {
                    kind = "builtin"
                }
                else {
                    kind = i < bytes.count && bytes[i] == 40 ? "function" : "name"
                }
            } else if bytes[i] == 34 {
                i += 1;
                var closed = false, decoded: [UInt8] = []
                while i < bytes.count {
                    if bytes[i] == 10 {
                        throw positions[start]
                    };
                    if bytes[i] == 34 {
                        i += 1;
                        closed = true;
                        break
                    }
                    if bytes[i] == 92 {
                    } else {
                        decoded.append(bytes[i]);
                        i += 1
                    }
                };
                if !closed {
                    throw positions[start]
                };
                kind = "string";
                value = String(decoding: decoded, as: UTF8.self)
            } else if i + 1 < bytes.count && ["+=", "-=", "*=", "/=", "%=", "^=", "==", "<=", ">=", "!=", "++", "--", "&&", "||", ">>"].contains(word(i, i + 2)) {
                i += 2
            }
            else if Array("+-*/%^!><?:=()[]{},;$".utf8).contains(bytes[i]) {
                i += 1
            }
            else {
                throw positions[i]
            }
            result.append(Token(kind: kind, text: word(start, i), value: value, position: positions[start]))
        };
        result.append(Token(kind: "end", text: "", value: "", position: positions[bytes.count]));
        return result
    }

            }

    func block() throws -> Stmt {
        let position = try expect("{").position;
        var statements: [Stmt] = []
        while text != "}" {
            if peek.kind == "newline" || text == ";" {
                take()
            } else {
                statements.append(try statement())
            }
        };
        try expect("}");
        return Stmt(position, .block(statements))
    }

        let token = peek, position = peek.position, kind: StmtKind
        switch text {
            case "{": return try block()
            case ";": take();

            case "print", "printf":
            take();

            var parenthesized = false
            if text == "(" {
            if empty && printf {
                throw peek.position
            }
            let arguments: [Expr]
            if empty {
                arguments = []
            } else if parenthesized {
                take();
                arguments = try expressions(false);

            } else {
                arguments = try expressions(true)
            }

            if text == ">" || text == ">>" {

            };
            kind = .output(printf, arguments, redirect)
            default: kind = .expression(try expression())
        };
        try terminator();
        return Stmt(position, kind)
    }
    func program() throws -> Program {
        var rules: [Rule] = [], functions: [Function] = []

            }
            else if text == "{" {
                rules.append(Rule(kind: "all", position: position, start: nil, end: nil, action: try block()))
            }
        };
        return Program(rules: rules, functions: functions)
    }
enum Value {
    case number(Int), string(String), numeric(String, Int), uninitialized
    static func whitespace(_ c: Character) -> Bool {
        c == " " || c == "\t" || c == "\n"
    }
    static func input(_ text: String) -> Self {
        let trimmed = text.drop(while: whitespace).reversed().drop(while: whitespace).reversed(), raw = String(trimmed), digits = raw.first == "+" || raw.first == "-" ? String(raw.dropFirst()) : raw
        return !digits.isEmpty && digits.utf8.allSatisfy({
            (48...57).contains($0)
        }) ? .numeric(text, Value.string(text).number) : .string(text)
    }
    var number: Int {
        switch self {
            case .number(let n), .numeric(_, let n): return n
            case .uninitialized: return 0
            case .string(let s): let raw = String(s.drop(while: Self.whitespace)), negative = raw.first == "-", rest = raw.first == "+" || raw.first == "-" ? raw.dropFirst() : raw[...], digits = rest.prefix {
        }
    }
    var text: String {
        switch self {
            case .number(let n): return String(n);
            case .string(let s), .numeric(let s, _): return s;
            case .uninitialized: return ""
        }
    }
    var display: String {
        switch self {
            case .number(let n): return String(n);
            case .string(let s): return quote(s);
            case .numeric(let s, _): return quote(s) + " (strnum)";
            case .uninitialized: return "(uninitialized)"
        }
    }
}
func quote(_ text: String) -> String {
    var result = "\""
    for c in text.utf8 {
        switch c {
            case 34: result += "\\\"";
            case 92: result += "\\\\";
            case 10: result += "\\n";
            case 9: result += "\\t";
            case 32...126: result.append(Character(UnicodeScalar(c)));
            default: let octal = String(c, radix: 8);
            result += "\\" + String(repeating: "0", count: 3 - octal.count) + octal
        }
    };
    return result + "\""
}
func splitText(_ text: String, _ separator: String) -> [String] {
    if separator == " " {
        return text.split(whereSeparator: Value.whitespace).map(String.init)
    };
    if separator.isEmpty {
        return text.map(String.init)
    };
    return text.isEmpty ? [] : text.components(separatedBy: separator)
}
final class Fields {
    var record = Value.string(""), values: [Value] = []
    func set(_ text: String, _ separator: String) {
        record = .input(text);
        values = splitText(text, separator).map(Value.input)
    }
    func get(_ index: Int) -> Value {
        index == 0 ? record : index <= values.count ? values[index - 1] : .uninitialized
    }
    func rebuild(_ separator: String) {
        record = .input(values.map(\.text).joined(separator: separator))
    }
    func assign(_ index: Int, _ value: Value, _ fs: String, _ ofs: String) {
        if index == 0 {
            set(value.text, fs)
        } else {
            if index > values.count {
                values += Array(repeating: .string(""), count: index - values.count)
            };
            values[index - 1] = value;
            rebuild(ofs)
        }
    }
    func nf(_ size: Int, _ ofs: String) {
        if size < values.count {
            values.removeLast(values.count - size)
        } else {
            values += Array(repeating: .string(""), count: size - values.count)
        };
        rebuild(ofs)
    }
}

func validFile(_ text: String) -> Bool {
    !text.isEmpty && text.first != "." && text.utf8.allSatisfy {
        Lexer.letter($0) || Lexer.digit($0) || [UInt8(46), 95, 45].contains($0)
    }
}
func readFile(_ name: String) -> [String]? {
    guard validFile(name), let text = try? String(contentsOfFile: "input/" + name, encoding: .ascii) else {
        return nil
    };
    return text.isEmpty ? [] : Array(text.components(separatedBy: "\n").dropLast())
}
func arguments(_ text: String) -> Arguments {
    var result = Arguments()
    func assignment(_ text: String) -> (String, String)? {
        guard let i = text.firstIndex(of: "=") else {
            return nil
        };
        return (String(text[..<i]), String(text[text.index(after: i)...]))
    }
    func variable(_ text: String) -> Bool {
        guard let first = text.utf8.first, Lexer.letter(first) || first == 95 else {
            return false
        };
        return text.utf8.allSatisfy {
            Lexer.letter($0) || Lexer.digit($0) || $0 == 95
        }
    }
    for line in text.split(separator: "\n").map(String.init) {
        if line.hasPrefix("-F ") {
            result.options.append(("FS", String(line.dropFirst(3))))
        }
        else if line.hasPrefix("-v ") {
            result.options.append(assignment(String(line.dropFirst(3)))!)
        }
        else if let (name, value) = assignment(line), variable(name) {
            result.operands.append(.assignment(name, value))
        }
        else {
            result.operands.append(.file(line, result.files.count));
            result.files.append((line, 0))
        }
    };
    return result
}

            switch next {
                case .assignment(let name, let value): preparation.globals.insert(name);
                globals[name] = .scalar(.input(value))
                case .file(let name, let index): globals["FILENAME"] = .scalar(.string(name));
                globals["FNR"] = .scalar(.number(0));
                current = MainInput(file: index, lines: readFile(name)!, position: 0)
            }
        }
    }
    func testRule(_ rule: Rule, _ index: Int) throws -> Bool {
        if rule.kind == "all" {
            return true
        }
    func execute() throws {
        for (name, value) in arguments.options {
            preparation.globals.insert(name);
            globals[name] = .scalar(.input(value))
        }
        var exiting = false
        for (i, rule) in program.rules.enumerated() where rule.kind == "BEGIN" {
            rules[i] += 1
            do {
                try statement(rule.action!)
            } catch Flow.exit {
                exiting = true;
                break
            }
        }
        phase = "rules"
        if !exiting && program.rules.contains(where: {
            $0.kind != "BEGIN"
        }) {
            records: while let record = nextRecord() {
                fields.set(record, global("FS").text)
                for (i, rule) in program.rules.enumerated() where !["BEGIN", "END"].contains(rule.kind) {
                    do {
                        if try testRule(rule, i) {
                            rules[i] += 1
                            if let action = rule.action {
                                try statement(action)
                            } else {
                                streams.output(nil, false, fields.record.text + global("ORS").text)
                            }
                        }
    func report(_ status: String) {
        streams.save();
        var lines = [status, "[inputs]"]
        for (name, count) in arguments.files {
            lines.append("\(name) records=\(count)")
        };
        lines.append("[rules]")
        for (i, rule) in program.rules.enumerated() {
            lines.append("\(rule.position) \(rule.kind) count=\(rules[i])")
        };
        lines += ["[outputs]", "stdout.txt writes=\(streams.writes) bytes=\(streams.stdout.utf8.count)"]
        lines.append("[lines]")
        for line in preparation.lines.keys.sorted() {
            lines.append("\(line) count=\(preparation.lines[line]!)")
        };
        lines.append("[globals]")
        for name in preparation.globals.sorted() {
            let value: String;
            if case .array(let array)? = globals[name] {
                value = "array (\(array.elements.count))"
            } else {
                value = global(name).display
            };
            lines.append("\(name) = \(value)")
        };
        saveStats(lines)
    }
}
func saveStats(_ lines: [String]) {
    try! FileManager.default.createDirectory(atPath: "output", withIntermediateDirectories: true)
    try! (lines.joined(separator: "\n") + "\n").write(toFile: "output/stats.txt", atomically: true, encoding: .utf8)
}
func run() {
    let source = try! String(contentsOfFile: "input/program.awk", encoding: .ascii), program: Program
    do {
        program = try Parser(Lexer.lex(source)).program()
    let preparation = Preparation(program)
    do {
        try preparation.analyze()
    let args = arguments(try! String(contentsOfFile: "input/args.txt", encoding: .ascii)), runtime = Runtime(program, preparation, args), status: String
    do {
        try runtime.execute();
        status = "status: exit \(runtime.status)"
    runtime.report(status)
run()
