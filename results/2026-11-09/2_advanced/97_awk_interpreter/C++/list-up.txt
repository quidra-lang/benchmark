    std::vector<Token> tokenize(const std::string& source) {
        const std::set<std::string> keywords = {"BEGIN", "END", "function", "if", "else", "while", "for", "do", "break", "continue", "next", "exit", "return", "delete", "in", "getline", "print", "printf"};
        const std::set<std::string> builtins = {"length", "substr", "index", "split", "sprintf", "toupper", "tolower", "close", "int"};
        const std::set<std::string> pairs = {"+=", "-=", "*=", "/=", "%=", "^=", "==", "<=", ">=", "!=", "++", "--", "&&", "||", ">>"};
        const std::map<char, char> escapes = { {'n', '\n'}, {'t', '\t'}, {'\\', '\\'}, {'"', '"'}, {'/', '/'} };
        std::vector<Token> result;
        std::size_t i = 0;
        Position position;
        while (i < source.size()) {
            char c = source[i];
            Position start = position;
            if (c == ' ') {
                ++i;
                ++position.column;
                continue;
            }
            if (c == '\\' && i + 1 < source.size() && source[i + 1] == '\n') {
                i += 2;
                ++position.line;
                position.column = 1;
                continue;
            }
            if (c == '#') {
                while (i < source.size() && source[i] != '\n') {
                    ++i;
                    ++position.column;
                }
                continue;
            }
            if (c == '\n') {
                result.push_back({"newline", "\n", "", start});
                ++i;
                ++position.line;
                position.column = 1;
                continue;
            }
            std::size_t end = i + 1;
            std::string kind, value;
            if (digit(c)) {
                while (end < source.size() && digit(source[end]))++end;
                kind = "number";
                value = source.substr(i, end - i);
            } else if (letter(c)) {
                while (end < source.size() &&(letter(source[end]) || digit(source[end])))++end;
                std::string word = source.substr(i, end - i);
                kind = keywords.count(word) ? "keyword" : builtins.count(word) ? "builtin" : end < source.size() && source[end] == '(' ? "function_name" : "name";
            } else if (c == '"') {
                while (end < source.size() && source[end] != '"') {
                    if (source[end] == '\n') throw Issue {start, "", 0};
                    if (source[end] == '\\') {
                        if (end + 1 >= source.size() || !escapes.count(source[end + 1])) throw Issue { {position.line, position.column + static_cast<int>(end - i)}, "", 0};
                        value.push_back(escapes.at(source[end + 1]));
                        end += 2;
                    } else value.push_back(source[end++]);
                }
                if (end == source.size()) throw Issue {start, "", 0};
                ++end;
                kind = "string";
            } else if (pairs.count(source.substr(i, 2))) {
                ++end;
                kind = "symbol";
            } else if (std::string("+-*/%^!><?:=()[]{},;$").find(c) != std::string::npos) kind = "symbol";
            else throw Issue {start, "", 0};
            result.push_back({kind, source.substr(i, end - i), value, start});
            position.column += static_cast<int>(end - i);
            i = end;
        }
        result.push_back({"end", "", "", {static_cast<int>(std::count(source.begin(), source.end(), '\n')) + 1, 1} });
        return result;
    }

        const std::string spelling = token.text;
        if (spelling == "{") return block();

            result->a = condition();
            newlines();

            take();
            result = output(token);
        } else {
            result->kind = "expression_statement";
            result->a = expression();
        }
        terminator();
        return result;
    }

    std::vector<std::string> split_pieces(const std::string& text, const std::string& separator) {
        std::vector<std::string> result;
        if (text.empty()) return result;
        if (separator.empty()) {
            for (char c : text) result.emplace_back(1, c);
            return result;
        }
        if (separator == " ") {
        std::size_t start = 0;
        while (true) {
            std::size_t next = text.find(separator, start);
            if (next == std::string::npos) {
                result.push_back(text.substr(start));
                break;
            }
            result.push_back(text.substr(start, next - start));
            start = next + separator.size();
        }
        return result;
    }

    }

                if (analysis.arrays.count({-1, name})) value->array = std::make_shared < Array >();
            }
            for (const std::string& name : std::vector<std::string> {"FNR", "NR"}) cell({-1, name})->scalar = Value::number(0);
            cell({-1, "FILENAME"})->scalar = Value::string("");
            cell({-1, "FS"})->scalar = Value::string(" ");
            cell({-1, "OFS"})->scalar = Value::string(" ");
            cell({-1, "ORS"})->scalar = Value::string("\n");
            cell({-1, "SUBSEP"})->scalar = Value::string(std::string(1, 28));
            for (int line : analysis.lines) line_counts[line] = 0;
        }
        void execute() {
            try {
                for (const auto& option : arguments.options) assign_input(option.first, option.second);
                try {
                    actions("BEGIN");
                    records();
                } catch (const Transfer& t) {
                    if (t.kind != "exit") throw;
                }
                try {
                    actions("END");
                } catch (const Transfer& t) {
                    if (t.kind != "exit") throw;
                }
            } catch (const Issue& issue) {
                error = issue;
            }
        }
        void save() {
            streams.save();
            std::ofstream out("output/stats.txt");
            if (error) out << "status: runtime error at " << error->position.text() << ": " << error->message << '\n';
            else out << "status: exit " << exit_status << '\n';
            out << "[inputs]\n";
            for (const InputOperand& operand : arguments.operands) if (operand.file) out << operand.name << " records=" << operand.records << '\n';
            out << "[rules]\n";
            for (const NodePtr& rule : analysis.rules) out << rule->position.text() << ' ' << rule->flavor << " count=" << rule->count << '\n';
            out << "[outputs]\nstdout.txt writes=" << streams.standard.writes << " bytes=" << streams.standard.content.size() << '\n';
            out << "[lines]\n";
            for (const auto& entry : line_counts) out << entry.first << " count=" << entry.second << '\n';
            out << "[globals]\n";
            for (const std::string& name : analysis.globals) {
                out << name << " = ";
                if (analysis.arrays.count({-1, name})) out << "array (" << array({-1, name})->size() << ')';
                else out << global(name).describe();
                out << '\n';
            }
        }
