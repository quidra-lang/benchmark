int power10(int places)
    result := 1
    for: j < places
        result *= 10
    return result

int scaled(real64 value, places := 4)
    absolute := if value < 0.0 then -value else value
    rounded := math.floor(absolute * real64(power10(places)) + 0.5)
    result := int(rounded)
    return if value < 0.0 then -result else result
real64 criterion_value(Fitted model, int criterion)
    return if criterion == 0 then model.bic elif criterion == 1 then model.aic else model.icl

bool model_before(Fitted first, Fitted second, int criterion)
    a := report.scaled(criterion_value(first, criterion))
    b := report.scaled(criterion_value(second, criterion))
    if a != b
        return a < b
    if first.p != second.p
        return first.p < second.p
    if first.count != second.count
        return first.count < second.count
    return first.type_index < second.type_index
Fitted[] ranking(Fitted[] source, int criterion)
    Fitted[] result := []
    for model := source[]
        index := 0
        while index < int(#result) and not model_before(model, result[index], criterion)
            index += 1
        Fitted[] updated := []
        for: j < int(#result)
            if j == index
                updated = updated.append(model)
            updated = updated.append(result[j])
        if index == int(#result)
            updated = updated.append(model)
        result = updated
    return result

    return [for model := source[] if model.count == target then model]
