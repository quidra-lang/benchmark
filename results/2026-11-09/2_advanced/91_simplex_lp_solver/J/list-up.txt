solve_lp =: 3 : 0
    'lower upper number level' =. y
    LogLevel =: level
    form =. standard_form(<lower),<upper
    load_form form
    restore_tableau(<InitialMatrix),(<InitialRhs),<InitialBasis
    LogLines =: 0$a:
    0 emit_log emit 'SOLVE node';number;'rows';(#RowNames);'columns';(#ColumnNames);'artificials';+/Artificial
    for_j. i.#Mappings do.
        'kind offset columns signs upper_row' =. j{::Mappings
        text =. 'MAP ',(j{::ModelVariables),' ',kind
        if. -.kind-:'split' do. text =. text,' ',fraction offset end.
        if. (kind-:'shift')*.finite j{::upper do. text =. text,' upper ',fraction j{::upper end.
        1 emit_log text
    end.
    for_i. I.RowSigns=_1 do. 1 emit_log 'FLIP ',i{::RowNames end.
    p1 =. p2 =. drives =. redundant =. 0
    w =. 0x
    status =. 'OPTIMAL'
    unbounded =. _1
    if. +./Artificial do.
        1 emit_log 'PHASE 1'
        'p1 unused logs' =. run_phase(<Phase1Costs),(<i.#ColumnNames),<1
        if. LogLevel>:1 do. LogLines =: LogLines,logs end.
        w =. phase_value Phase1Costs
        1 emit_log emit 'PHASE 1 END objective';w;'iterations';p1
        if. w>0 do.
            status =. 'INFEASIBLE'
        else.
            for_i. i.#TBeta do.
                basic =. i{TBasis
                if. -.basic{Artificial do. continue. end.
                candidates =. I.(-.Artificial)*.0~:i{TMatrix
                if. 0=#candidates do.
                    redundant =. redundant+1
                    1 emit_log 'REDUNDANT row ',i{::RowNames
                else.
                    entering =. {.candidates
                    leaving =. basic{::ColumnNames
                    pivot i,entering
                    drives =. drives+1
                    1 emit_log emit 'DRIVE row';(i{::RowNames);'enter';(entering{::ColumnNames);'leave';leaving
                end.
            end.
        end.
    end.
    if. -.status-:'INFEASIBLE' do.
        1 emit_log 'PHASE 2'
        'p2 unbounded logs' =. run_phase(<Phase2Costs),(<I.-.Artificial),<2
        if. LogLevel>:1 do. LogLines =: LogLines,logs end.
        if. unbounded~:_1 do.
            status =. 'UNBOUNDED'
            1 emit_log 'UNBOUNDED enter ',unbounded{::ColumnNames
        end.
        value =. 2 display_value phase_value Phase2Costs
        1 emit_log emit 'PHASE 2 END objective';value;'iterations';p2
    end.
    objective =. ''
    if. status-:'OPTIMAL' do. objective =. 2 display_value phase_value Phase2Costs end.
    text =. emit 'RESULT';status;'objective';(fraction objective);'phase1';p1;'phase2';p2;'drives';drives;'redundant';redundant
    0 emit_log text
    solution =. tableau_solution ''
    tableau =. (<TMatrix),(<TBeta),<TBasis
    counts =. p1;p2;drives;redundant;(p1+p2+drives)
    (<status),(<objective),(<solution),(<form),(<tableau),(<unbounded),(<w),counts,<LogLines
)

parse_expression =: 3 : 0

        if. (0{::y)-:,'-' do.
            sign =. _1x
            position =. 1
        end.
    end.
    while. position<#y do.
        text =. position{::y
        value =. 0 read_number text
        if. finite value do.
            if. (position+1)=#y do. 0$a: return. end.
            name =. (position+1){::y

            position =. position+2
        elseif. is_name text do.
            value =. 1x
            name =. text
            position =. position+1
        else.
            0$a: return.
        end.

        if. position=#y do. terms return. end.
        operator =. position{::y
        if. -.(<operator)e.(<,'+'),<,'-' do. 0$a: return. end.

parse_block =: 3 : 0
    'name body end_line' =. y
    error =. 0$a:
    direction =. ''
    objective =. constraints =. bounds =. integers =. binaries =. 0$a:
    state =. 0
    for_record. body do.
        'line text' =. >record
        words =. split_words text
        accepted =. 0
        select. state
        case. 0 do.
            if. (<text)e.'maximize';'minimize' do.
                direction =. text
                state =. 1
                accepted =. 1
            end.
        case. 1 do.
            named =. named_expression words
            if. #named do.
                objective =. 1{::named
                state =. 2
                accepted =. 1
            end.
        case. 2 do.
            if. text-:'subject to' do.
                state =. 3
                accepted =. 1
            end.
        case. 3;4 do.
            if. (state=4)*.(<text)e.;:'bounds integer binary' do.
                state =. 5+(;:'bounds integer binary')i.<text
                accepted =. 1
            else.
                item =. line parse_constraint words
                if. #item do.
                    constraints =. constraints,<item
                    state =. 4
                    accepted =. 1
                end.
            end.
            break.
        end.
    end.
    if. (0=#error)*.state<4 do. error =. (<end_line),<'syntax error' end.
    (<name),(<error),(<direction),(<objective),(<constraints),(<bounds),(<integers),<binaries

run_phase =: 3 : 0
    'costs candidates phase' =. y
    iterations =. 0
    lines =. 0$a:
    candidates =. /:~candidates
    while. 1 do.
        reduced =. reduced_costs costs
        negative =. candidates#~(candidates{reduced)<0
        if. 0=#negative do. (<iterations),(<_1),<lines return. end.
        column =. {.negative
        choices =. 0 3$0x
        for_row. i.#TBeta do.
            coefficient =. (<row,column){TMatrix
            if. coefficient>0 do.
                choices =. choices,((row{TBeta)%coefficient),(row{TBasis),row
            end.
        end.
        if. 0=#choices do. (<iterations),(<column),<lines return. end.
        'ratio leaving row' =. ({./:choices){choices
        leaving =. <.leaving
        row =. <.row
        before =. column{reduced
        pivot row,column
        iterations =. iterations+1
        value =. phase display_value phase_value costs
        line =. 'PIVOT ',(integer iterations),' enter ',(column{::ColumnNames),' reduced ',fraction before
        line =. line,' leave ',(leaving{::ColumnNames),' row ',(row{::RowNames),' ratio ',fraction ratio

    end.
