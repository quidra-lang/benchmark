    let chosen: Slot | undefined;
    let words: WordEntry[] | undefined;
    for (const slot of slots) {
      if (state.assigned[slot.index] !== null) continue;
      const current = candidates(slot);
      if (!words || current.length < words.length) {
        chosen = slot;
        words = current;
      }
    }
    if (!chosen || !words) return 'complete';
    if (!words.length) {
      result.deadEnds++;
      result.failures[chosen.index]++;
      return 'fail';
    }
    for (const word of words) {
      if (result.nodes === budget) return 'abort';
      result.nodes++;
      state.assigned[chosen.index] = word;
      state.used.add(word.word);
      const changed: number[] = [];
      chosen.cells.forEach((cell, position) => {
        if (state.cells[cell] === '.') {
          changed.push(cell);
          state.cells[cell] = word.word[position];
        }
      });
      result.order.push(chosen.index);
      const wiped = chosen.crossing.find(
        (i) => state.assigned[i] === null && candidates(slots[i]).length === 0,
      );
      let outcome: 'complete' | 'fail' | 'abort';
      if (wiped !== undefined) {
        result.prunes++;
        result.failures[wiped]++;
        outcome = 'fail';
      } else {
        outcome = search();
      }
      if (outcome !== 'fail') return outcome;
      state.assigned[chosen.index] = null;
      state.used.delete(word.word);
      for (const cell of changed) state.cells[cell] = '.';

    ].sort((a, b) => a - b);
  }

    const key = `${minimum}:${pattern}`;
    let matching = this.matches.get(key);
    if (!matching) {
      let pool = this.byLength.get(pattern.length) ?? [];
      for (let position = 0; position < pattern.length; position++) {
        if (pattern[position] !== '.') {
          const entries =
            this.byPosition.get(`${pattern.length}:${position}:${pattern[position]}`) ?? [];
          if (entries.length < pool.length) pool = entries;
        }
      }

        (entry) =>
          entry.score >= minimum &&
          [...pattern].every((letter, i) => letter === '.' || letter === entry.word[i]),
      );

    }

function fillTemplate(template: Template, index: WordIndex, relaxStep: number): FilledTemplate {
  const attempts: Attempt[] = [];
  let minimum = template.settings.min_score;
  while (true) {
    const search = attempt(
      template.initial,
      template.slots,
      index,
      minimum,
      template.settings.node_budget,
    );
    attempts.push(search);
    if (search.outcome === 'filled' || relaxStep === 0 || minimum === 1) {
      return { ...template, attempts, finalAttempt: search, final: search.final };
    }
    minimum = Math.max(minimum - relaxStep, 1);
  }
