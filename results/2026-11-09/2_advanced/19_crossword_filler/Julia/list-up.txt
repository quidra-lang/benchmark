    state.words[slot.index] = id
    state.words[slot.index] = 0
attempt_candidates(attempt, slot, index) = candidates(index, pattern(attempt.state, slot), attempt.minimum, attempt.state.used)
function search!(attempt, layout, index)
    chosen, words = 0, Int[]
    for slot in layout.slots
        attempt.state.words[slot.index] != 0 && continue
        options = attempt_candidates(attempt, slot, index)
        if chosen == 0 || length(options) < length(words)
            chosen, words = slot.index, options
        end
    end
    chosen == 0 && return "complete"
    if isempty(words)
        attempt.dead_ends += 1
        attempt.failures[chosen] += 1
        return "fail"
    end
    slot = layout.slots[chosen]
    for word in words
        attempt.nodes == attempt.budget && return "abort"
        attempt.nodes += 1
        filled = assign!(attempt.state, slot, word, index)
        push!(attempt.sequence, chosen)
        wiped = findfirst(other -> attempt.state.words[other] == 0 && isempty(attempt_candidates(attempt, layout.slots[other], index)), slot.crossings)
        if !isnothing(wiped)
            attempt.prunes += 1
            attempt.failures[slot.crossings[wiped]] += 1
            result = "fail"
        else
            result = search!(attempt, layout, index)
        end
        result != "fail" && return result
        unassign!(attempt.state, slot, filled)
    for slot in slots
        owners = slot.across ? down_slot : across_slot
        slot.crossings = sort([owners[index] for index in slot.cells if owners[index] != 0])
    ranked = sort(collect(keys(scores)); by=word -> (-scores[word], word))
    words, lookup = Word[], Dict{String,Int}()
    by_length, by_letter = Dict{Int,Vector{Int}}(), Dict{Tuple{Int,Int,Char},Vector{Int}}()
    for (id, text) in enumerate(ranked)
        letters = collect(text)
        lookup[text] = id
        push!(get!(by_length, length(letters), Int[]), id)
        for (position, letter) in enumerate(letters)
            push!(get!(by_letter, (length(letters), position, letter), Int[]), id)
        end
        push!(words, Word(text, letters, scores[text]))
function candidates(index, pattern, minimum, used)
    pool = get(index.by_length, length(pattern), Int[])
    for (position, mark) in enumerate(pattern)
        mark == '.' && continue
        bucket = get(index.by_letter, (length(pattern), position, mark), Int[])
        length(bucket) < length(pool) && (pool = bucket)
    end
    result = Int[]
    for id in pool
        index.words[id].score < minimum && break
        !(id in used) && matches(index.words[id].letters, pattern) && push!(result, id)
    end
    result
