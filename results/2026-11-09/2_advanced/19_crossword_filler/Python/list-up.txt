        self.words[slot.index] = word
        self.words[slot.index] = None
    def candidates(self, slot):
        return self.index.candidates(self.state.pattern(slot), self.min_score, self.state.used)
    def choose_slot(self):
        chosen = None
        chosen_words = None
        for slot in self.layout.slots:
            if self.state.words[slot.index] is None:
                words = self.candidates(slot)
                if chosen is None or len(words) < len(chosen_words):
                    chosen = slot
                    chosen_words = words
        return chosen, chosen_words
    def wiped_out_slot(self, slot):
        for other in slot.crossings:
            crossing = self.layout.slots[other]
            if self.state.words[other] is None and not self.candidates(crossing):
                return crossing
        return None
    def search(self):
        slot, words = self.choose_slot()
        if slot is None:
            return COMPLETE
        if not words:
            self.dead_ends += 1
            self.failures[slot.index] += 1
            return FAIL
        for word in words:
            if self.nodes == self.budget:
                return ABORT
            self.nodes += 1
            filled = self.state.assign(slot, word)
            self.sequence.append(slot)
            wiped = self.wiped_out_slot(slot)
            if wiped is not None:
                self.prunes += 1
                self.failures[wiped.index] += 1
                result = FAIL
            else:
                result = self.search()
            if result != FAIL:
                return result
            self.state.unassign(slot, filled)
        for slot in self.slots:
            owners = self.down_slot if slot.across else self.across_slot
            crossing = [owners[cell] for cell in slot.cells if owners[cell] is not None]
            slot.crossings = sorted(crossing)

    return sorted(scores, key=lambda word: (-scores[word], word))
        self.ranked = rank_order(scores)
        self.by_length = {}
        self.by_letter = {}
        for word in self.ranked:
            self.by_length.setdefault(len(word), []).append(word)
            for position, letter in enumerate(word):
                self.by_letter.setdefault((len(word), position, letter), []).append(word)
    def pool(self, pattern):
        length = len(pattern)
        pool = self.by_length.get(length, [])
        for position, mark in enumerate(pattern):
            if mark != EMPTY:
                bucket = self.by_letter.get((length, position, mark), [])
                if len(bucket) < len(pool):
                    pool = bucket
        return pool
    def candidates(self, pattern, min_score, used):
        result = []
        for word in self.pool(pattern):
            if self.scores[word] < min_score:
                break
            if word not in used and matches(word, pattern):
                result.append(word)
        return result
    def run_attempts(self, index, relax_step):
        min_score = self.min_score
        while True:
            attempt = Attempt(self.layout, index, self.initial, min_score, self.node_budget)
            attempt.run()
            self.attempts.append(attempt)
            if attempt.outcome == "filled" or relax_step < 1 or min_score <= 1:
                break
            min_score = max(min_score - relax_step, 1)
