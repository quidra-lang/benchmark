        assigned[slot.index] = id
        assigned[slot.index] = -1
    int[] candidates(slots.Slot slot)
        return index.candidates(state.pattern(slot), min_score, state.used)
    (int, int[]) choose_slot()
        chosen := -1
        int[] choices := []
        for slot := layout.entries[]
            if state.assigned[slot.index] >= 0
                continue
            int[] available := candidates(slot)
            if chosen < 0 or #available < #choices
                chosen = slot.index
                choices = available
        return (chosen, choices)
    int wiped_out_slot(slots.Slot slot)
        for crossing := slot.crossings[]
            if state.assigned[crossing] < 0 and #candidates(layout.entries[crossing]) == 0
                return crossing
        return -1
    string .search()
        chosen, choices := choose_slot()
        if chosen < 0
            return "complete"
        slots.Slot slot := layout.entries[chosen]
        if #choices == 0
            dead_ends += 1
            failures[chosen] += 1
            return "fail"
        for id := choices[]
            if nodes == budget
                return "abort"
            nodes += 1
            int[] changed := .state.assign(slot, id, index.ranked[id].word)
            sequence = sequence.append(chosen)
            int wiped := wiped_out_slot(slot)
            result := "fail"
            if wiped >= 0
                prunes += 1
                failures[wiped] += 1
            else
                result = .search()
            if result != "fail"
                return result
            .state.unassign(slot, changed)
    for: index < int(#entries)
        int[] crossing := []
        for position := entries[index].cells[]
            int other := if entries[index].across then down_slot[position] else across_slot[position]
            if other >= 0
                crossing = crossing.append(other)
        entries[index].crossings = crossing.sorted()
    local names := scores.keys().sorted()
    for: 100 >= candidate_score >= 1
        for word := names[]
            if score_of(scores, word) != candidate_score
                continue
            id := int(#ranked)
            ranked = ranked.append(Entry(word, candidate_score))
            .by_word.set(word, id)
            length := int(#word)
            int[] length_words := words_of_length(length)
            .by_length.set(length, length_words.append(id))
            for: position < length
                (int, int, string) key := (length, position, word[position])
                int[] letter_words := words_of_letter(length, position, word[position])
                .by_letter.set(key, letter_words.append(id))
    int[] candidates(string pattern, int min_score, bool[] used)
        length := int(#pattern)
        int[] pool := words_of_length(length)
        for: position < length
            if pattern[position] != "."
                int[] bucket := words_of_letter(length, position, pattern[position])
                if #bucket < #pool
                    pool = bucket
        int[] result := []
        for id := pool[]
            if ranked[id].score < min_score
                break
            if not used[id] and matches(ranked[id].word, pattern)
                result = result.append(id)
        return result
    attempt_minimum := minimum
    while true
        attempt := filler.Attempt(layout, index, initial, attempt_minimum, budget)
        .attempt.run()
        attempts = attempts.append(attempt)
        if attempt.outcome == "filled" or settings.relax_step < 1 or attempt_minimum <= 1
            break
        attempt_minimum = math.max(attempt_minimum - settings.relax_step, 1)
