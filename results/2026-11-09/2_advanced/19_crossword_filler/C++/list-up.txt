Attempt::SearchResult Attempt::search() {
    int chosen = -1;
    Lines candidates;
    for (const Slot& slot : layout.slots) if (state.words[slot.index].empty()) {
        Lines words = index.candidates(state.pattern(slot), minimum, state.used);
        if (chosen < 0 || words.size() < candidates.size()) { chosen = slot.index; candidates = std::move(words); }
    }
    if (chosen < 0) return SearchResult::complete;
    if (candidates.empty()) { ++deadEnds; ++failures[chosen]; return SearchResult::fail; }
    const Slot& slot = layout.slots[chosen];
    for (const std::string& word : candidates) {
        if (nodes == budget) return SearchResult::abort;
        ++nodes;
        Indices filled = state.assign(slot, word); sequence.push_back(chosen);
        int wiped = -1;
        for (int other : slot.crossings) if (state.words[other].empty() && index.candidates(state.pattern(layout.slots[other]), minimum, state.used).empty()) { wiped = other; break; }
        SearchResult result;
        if (wiped >= 0) { ++prunes; ++failures[wiped]; result = SearchResult::fail; }
        else { if (static_cast<int>(sequence.size()) > maxDepth) { maxDepth = static_cast<int>(sequence.size()); best = state; } result = search(); }
        if (result != SearchResult::fail) return result;
        state.unassign(slot, filled); sequence.pop_back(); ++backtracks;
    for (Slot& slot : slots) { for (int position : slot.cells) { int crossing = (slot.across ? downSlot : acrossSlot)[position]; if (crossing >= 0) slot.crossings.push_back(crossing); } std::sort(slot.crossings.begin(), slot.crossings.end()); }
    std::sort(ranked.begin(), ranked.end(), [&](const std::string& a, const std::string& b) { return this->scores.at(a) != this->scores.at(b) ? this->scores.at(a) > this->scores.at(b) : a < b; });
    for (const std::string& word : ranked) { int length = static_cast<int>(word.size()); byLength[length].push_back(word); for (int position = 0; position < length; ++position) byLetter[{length, position, word[position]}].push_back(word); }
Lines WordIndex::candidates(const std::string& pattern, int minimum, const std::set<std::string>& used) const {
    int length = static_cast<int>(pattern.size());
    std::map<int, Lines>::const_iterator lengthEntry = byLength.find(length);
    if (lengthEntry == byLength.end()) return {};
    const Lines* pool = &lengthEntry->second;
    for (int position = 0; position < length; ++position) if (pattern[position] != '.') {
        std::map<std::tuple<int, int, char>, Lines>::const_iterator bucket = byLetter.find({length, position, pattern[position]});
        if (bucket == byLetter.end()) return {};
        if (bucket->second.size() < pool->size()) pool = &bucket->second;
    }
    Lines result;
    for (const std::string& word : *pool) { if (scores.at(word) < minimum) break; if (!used.count(word) && matches(word, pattern)) result.push_back(word); }
    return result;
}
