    private function chooseSlot(): array
    {
        $chosen = null;
        $chosenWords = null;
        foreach ($this->layout->slots as $slot) {
            if ($this->state->words[$slot->index] !== null) {
                continue;
            }
            $words = $this->candidates($slot);
            if ($chosen === null || count($words) < count($chosenWords)) {
                $chosen = $slot;
                $chosenWords = $words;
            }
        }
        return [$chosen, $chosenWords];
    }

    {

                return $crossing;
            }
        }
        return null;
    }
