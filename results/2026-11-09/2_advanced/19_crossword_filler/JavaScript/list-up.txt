    function search() {
        let chosen = null;
        let words = null;
        for (const slot of slots) {
            if (state.assigned[slot.index] !== null) continue;
            const current = candidates(slot);
            if (words === null || current.length < words.length) {
                chosen = slot;
                words = current;
            }
        }
        if (chosen === null) return 'complete';
        if (words.length === 0) {
            result.deadEnds++;
            result.failures[chosen.index]++;
            return 'fail';
        }
        for (const word of words) {
            if (result.nodes === budget) return 'abort';
            result.nodes++;
            state.assigned[chosen.index] = word;
            state.used.add(word.word);
            const changed = [];
            for (let position = 0; position < chosen.length; position++) {
                const cell = chosen.cells[position];
                if (state.cells[cell] !== '.') continue;
                changed.push(cell);
                state.cells[cell] = word.word[position];
            }
            result.order.push(chosen.index);
            let wiped = null;
            for (const crossing of chosen.crossing) {
                if (state.assigned[crossing] === null && candidates(slots[crossing]).length === 0) {
                    wiped = crossing;
                    break;
                }
            }
            let outcome;
            if (wiped !== null) {
                result.prunes++;
                result.failures[wiped]++;
                outcome = 'fail';
            } else {
                outcome = search();
            }
            if (outcome !== 'fail') return outcome;
            state.assigned[chosen.index] = null;
            state.used.delete(word.word);
            for (const cell of changed) state.cells[cell] = '.';

        }

    }

const rankOrder = (a, b) => b.score - a.score || compareText(a.word, b.word);
        this.words = words.slice().sort(rankOrder);

            this.byLength.get(length).push(entry);
            for (let position = 0; position < length; position++) {

            }

    candidates(pattern, minimum, used) {
        const key = minimum + ':' + pattern;
        if (!this.matches.has(key)) {
            let pool = this.byLength.get(pattern.length) ?? [];
            for (let position = 0; position < pattern.length; position++) {
                if (pattern[position] === '.') continue;
                const collection = this.byPosition.get(`${pattern.length}:${position}:${pattern[position]}`) ?? [];
                if (collection.length < pool.length) pool = collection;
            }
            const matching = [];
            for (const entry of pool) {
                if (entry.score < minimum) break;
                let agrees = true;
                for (let position = 0; position < pattern.length; position++) {
                    if (pattern[position] !== '.' && pattern[position] !== entry.word[position]) { agrees = false; break; }
                }
                if (agrees) matching.push(entry);
            }
            this.matches.set(key, matching);
        }
        return this.matches.get(key).filter(entry => !used.has(entry.word));
    }

        if (search.outcome === 'filled' || relaxStep === 0 || minimum === 1) break;
        minimum = Math.max(minimum - relaxStep, 1);
    }
