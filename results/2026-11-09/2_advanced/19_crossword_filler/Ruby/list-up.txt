  def candidates(slot)
    @index.candidates(@state.pattern(slot), @minimum, @used)
  end
  def search
    chosen = nil
    choices = nil
    @slots.each do |slot|
      next if @state.assigned[slot.id]
      available = candidates(slot)
      if chosen.nil? || available.length < choices.length
        chosen = slot
        choices = available
      end
    end
    return :complete if chosen.nil?
    if choices.empty?
      @dead_ends += 1
      @failures[chosen.id] += 1
      return :fail
    end
    choices.each do |word|
      return :abort if @nodes == @budget
      @nodes += 1
      @state.assigned[chosen.id] = word
      @used.add(word)
      wiped = chosen.crossings.find do |id|
        @state.assigned[id].nil? && candidates(@slots[id]).empty?
      end
      if wiped
        @prunes += 1
        @failures[wiped] += 1
        result = :fail
      else
        result = search
      end
      return result if result == :complete || result == :abort
      @state.assigned[chosen.id] = nil
      @used.delete(word)
      changed.each { |cell| @state.cells[cell] = '.' }
    slots.each do |slot|
      slot.crossings = slot.cells.flat_map { |cell| cell_slots[cell] }
                           .reject { |id| id == slot.id }.uniq.sort
    end
    @ranked = scores.keys.sort_by { |word| [-scores.fetch(word), word] }
    @by_length = Hash.new { |hash, length| hash[length] = [] }
    @by_position = Hash.new { |hash, key| hash[key] = [] }
    @ranked.each do |word|
      @by_length[word.length] << word
      word.each_char.with_index do |letter, position|
        @by_position[[word.length, position, letter]] << word
      end
    end
  def candidates(pattern, minimum, used)
    pool = @by_length.fetch(pattern.length, [])
    known = []
    pattern.each_char.with_index do |letter, position|
      next if letter == '.'
      known << [position, letter]
      matching = @by_position.fetch([pattern.length, position, letter], [])
      pool = matching if matching.length < pool.length
    end
    result = []
    pool.each do |word|
      break if @scores.fetch(word) < minimum
      next if used.include?(word)
      next unless known.all? { |position, letter| word[position] == letter }
      result << word
    end
    result
  end
