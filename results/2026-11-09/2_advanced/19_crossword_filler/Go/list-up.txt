	state.Words[slot.Index] = id
	state.Words[slot.Index] = -1
func (a *Attempt) candidates(slot Slot) []int {
	return a.Index.Candidates(a.State.Pattern(slot), a.Minimum, a.State.Used)
}
func (a *Attempt) search() string {
	chosen := -1
	var choices []int
	for _, slot := range a.Layout.Slots {
		if a.State.Words[slot.Index] >= 0 {
			continue
		}
		available := a.candidates(slot)
		if chosen < 0 || len(available) < len(choices) {
			chosen, choices = slot.Index, available
		}
	}
	if chosen < 0 {
		return "complete"
	}
	if len(choices) == 0 {
		a.DeadEnds++
		a.Failures[chosen]++
		return "fail"
	}
	slot := a.Layout.Slots[chosen]
	for _, word := range choices {
		if a.Nodes == a.Budget {
			return "abort"
		}
		a.Nodes++
		changed := a.State.Assign(slot, word, a.Index)
		a.Sequence = append(a.Sequence, chosen)
		wiped := -1
		for _, other := range slot.Crossings {
			if a.State.Words[other] < 0 && len(a.candidates(a.Layout.Slots[other])) == 0 {
				wiped = other
				break
			}
		}
		result := "fail"
		if wiped >= 0 {
			a.Prunes++
			a.Failures[wiped]++
		} else {
			result = a.search()
		}
		if result != "fail" {
			return result
		}
		a.State.Unassign(slot, changed)

		owners := layout.AcrossSlot

			owners = layout.DownSlot
		}
		for _, cell := range slot.Cells {
			if owners[cell] >= 0 {
				slot.Crossings = append(slot.Crossings, owners[cell])
			}
		}
		sort.Ints(slot.Crossings)
	sort.Slice(names, func(i, j int) bool {
		if scores[names[i]] == scores[names[j]] {
			return names[i] < names[j]
		}
		return scores[names[i]] > scores[names[j]]
	})
	index := &WordIndex{Lookup: make(map[string]int), ByLength: make(map[int][]int), ByLetter: make(map[LetterKey][]int)}
	for id, text := range names {
		letters := []byte(text)
		index.Words = append(index.Words, Word{text, letters, scores[text]})
		index.Lookup[text] = id
		index.ByLength[len(letters)] = append(index.ByLength[len(letters)], id)

func (index *WordIndex) Candidates(pattern []byte, minimum int, used []bool) []int {
	pool := index.ByLength[len(pattern)]
	for position, mark := range pattern {
		if mark == '.' {
			continue
		}
		bucket := index.ByLetter[LetterKey{len(pattern), position, mark}]
		if len(bucket) < len(pool) {
			pool = bucket
		}
	}
	var result []int
	for _, id := range pool {
		if index.Words[id].Score < minimum {
			break
		}
		if !used[id] && matches(index.Words[id].Letters, pattern) {
			result = append(result, id)
		}
	}
	return result
