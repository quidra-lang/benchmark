    public function stateAllowed(int $cell, int $time): bool
    {
        return (
            !isset($this->states[$time][$cell])
            && (!isset($this->permanent[$cell]) || $time < $this->permanent[$cell])
        );
    }
    public function actionAllowed(int $source, int $destination, int $time): bool
    {
        return (
            $this->stateAllowed($destination, $time)
            && ($source === $destination || !isset($this->moves[$time][$source . ':' . $destination]))
        );
    }
    public function arrivalAllowed(int $goal): int
    {
        $r = 0;
        foreach ($this->states as $time => $cells)
            if (isset($cells[$goal]))
                $r = max($r, $time + 1);
        return $r;
    }
    public function reserve(array $path, int $start = 0, bool $permanent = false): void
    {
        foreach ($path as $step => $cell) {
            $time = $start + $step;
            $this->states[$time][$cell] = true;
            if ($step && $cell !== $path[$step - 1])
                $this->moves[$time][$cell . ':' . $path[$step - 1]] = [$cell, $path[$step - 1]];
        }
        if ($permanent) {
            $goal = $path[count($path) - 1];
            $time = $start + count($path) - 1;
            $this->permanent[$goal] = min($this->permanent[$goal] ?? $time, $time);
        }
    }

    private function solvePP(array &$run): void
    {
        $budget = $run['budget'];
        foreach ($this->settings['orders'] as $index => $order) {
            $number = $index + 1;
            ++$run['high'];
            $before = $budget->expansions;
            [$paths, $attempts, $failure] = tryOrder(
                $this->grid,
                $this->agents,
                $this->distances,
                $this->settings['horizon'],
                $order,
                $budget,
            );
            foreach ($attempts as $a) {
                $text = "order $number agent " . agentName($a['agent']);
                $text .= $a['path'] === null ? ' failed' : ' cost ' . (count($a['path']) - 1);
                $run['trace'][] = $text . ' expansions ' . $a['expansions'];
            }
            if ($failure === 'll_budget') {
                $run['reason'] = 'll_budget';
                break;
            }
            $names = implode('', array_map(agentName(...), $order));
            $planned = count($attempts) - ($failure === null ? 0 : 1);
            $expanded = $budget->expansions - $before;
            $text = "order $number $names planned $planned expansions $expanded";
            if ($failure === null) {
                $run['specific'][] = $text . ' ok';
                $run['paths'] = $paths;
                $run['status'] = 'OK';
                $run['end'] = "end OK order $number";
                return;
            }
            $run['specific'][] = $text . ' failed ' . agentName($failure);
        }
        $run['reason'] ??= 'orders';
        $run['end'] = 'end FAIL ' . $run['reason'];
    }
    private function validate(array $paths): array
    {
        $start = $goal = $moves = $follows = 0;
        foreach ($paths as $i => $p) {
            $start += (int) ($p[0] !== $this->agents[$i]['start']);
            $goal += (int) ($p[count($p) - 1] !== $this->agents[$i]['goal']);
            for ($t = 1; $t < count($p); ++$t)
                if (!in_array($p[$t], $this->grid->adjacent[$p[$t - 1]], true))
                    ++$moves;
        }
        $collisions = conflicts($paths);
        $vertex = count(array_filter($collisions, fn($c) => $c['kind'] === 'vertex'));
        $edge = count($collisions) - $vertex;
        $horizon = max(array_map(fn($p) => count($p) - 1, $paths));
        for ($t = 1; $t <= $horizon; ++$t)
            foreach ($paths as $a => $pa) {
                $old = position($pa, $t - 1);
                $next = position($pa, $t);
                if ($next === $old)
                    continue;
                foreach ($paths as $b => $pb)
                    if ($a !== $b && $next === position($pb, $t - 1) && position($pb, $t) !== $old)
                        ++$follows;
            }
        return [
            'verdict' => ($start + $goal + $moves + $vertex + $edge) === 0 ? 'VALID' : 'INVALID',
            'start' => $start,
            'goal' => $goal,
            'moves' => $moves,
            'vertex' => $vertex,
            'edge' => $edge,
            'follows' => $follows,
            'conflicts' => $collisions,
        ];
    }
    private function metrics(array $paths, string $function): array
    {
        $records = $traffic = $cellWaits = [];
        $lengths = array_map(fn($p) => count($p) - 1, $paths);
        $soc = array_sum($lengths);
        $makespan = max($lengths);
        $cost = $function === 'SOC' ? $soc : $makespan;
        foreach ($paths as $i => $p) {
            $actions = actionString($p, $this->grid);
            $moves = str_replace('.', '', $actions);
            $turns = $leave = 0;
            for ($j = 1; $j < strlen($moves); ++$j)
                if ($moves[$j] !== $moves[$j - 1])
                    ++$turns;
            for ($j = 1; $j < count($p); ++$j) {
                if ($p[$j - 1] === $this->agents[$i]['goal'] && $p[$j] !== $this->agents[$i]['goal'])
                    ++$leave;
                if ($p[$j] === $p[$j - 1])
                    $cellWaits[$p[$j]] = ($cellWaits[$p[$j]] ?? 0) + 1;
            }
            foreach (array_unique($p) as $c)
                $traffic[$c] = ($traffic[$c] ?? 0) + 1;
            $records[] = [
                'cost' => $lengths[$i],
                'extra' => $lengths[$i] - $this->overview['agents'][$i]['shortest'],
                'waits' => substr_count($actions, '.'),
                'turns' => $turns,
                'arrive' => array_search($this->agents[$i]['goal'], $p, true),
                'leave' => $leave,
                'actions' => $actions,
            ];
        }
        $arrivals = [];
        for ($t = 0; $t <= $makespan; ++$t)
            $arrivals[] = count(array_filter($lengths, fn($l) => $l <= $t));
        $hotspots = array_keys($traffic);
        usort($hotspots, fn($a, $b) => $traffic[$b] <=> $traffic[$a] ?: $a <=> $b);
        $hotspots = array_slice($hotspots, 0, 5);
        return [
            'soc' => $soc,
            'makespan' => $makespan,
            'cost' => $cost,
            'ratio' => fraction($cost, $this->overview[$function === 'SOC' ? 'soc' : 'makespan']),
            'agents' => $records,
            'sum_extra' => array_sum(array_column($records, 'extra')),
            'max_extra' => max(array_column($records, 'extra')),
            'waits' => array_sum(array_column($records, 'waits')),
            'turns' => array_sum(array_column($records, 'turns')),
            'arrivals' => $arrivals,
            'traffic' => $traffic,
            'cell_waits' => $cellWaits,
            'hotspots' => $hotspots,
        ];
    }

function position(array $path, int $time): int
{
    return $path[min($time, count($path) - 1)];
}
function conflicts(array $paths): array
{
    $result = [];
    $horizon = max(array_map(fn($p) => count($p) - 1, $paths));
    $n = count($paths);
    for ($t = 1; $t <= $horizon; ++$t)
        for ($a = 0; $a < $n; ++$a)
            for ($b = $a + 1; $b < $n; ++$b) {
                $ca = position($paths[$a], $t);
                $cb = position($paths[$b], $t);
                if ($ca === $cb)
                    $result[] = ['kind' => 'vertex', 'time' => $t, 'a' => $a, 'b' => $b, 'cell' => $ca];
                elseif ($ca === position($paths[$b], $t - 1) && $cb === position($paths[$a], $t - 1))
                    $result[] = [
                        'kind' => 'edge',
                        'time' => $t,
                        'a' => $a,
                        'b' => $b,
                        'u' => position($paths[$a], $t - 1),
                        'v' => $ca,
                    ];
            }
    return $result;
}
function planCost(array $paths, string $function): int
{
    $lengths = array_map(fn($p) => count($p) - 1, $paths);
    return $function === 'SOC' ? array_sum($lengths) : max($lengths);
}
function tryOrder(
    Grid $grid,
    array $agents,
    array $distances,
    int $horizon,
    array $order,
    Budget $budget,
): array {
    $p = new Prohibitions();
    $paths = $attempts = [];
    foreach ($order as $i) {
        try {
            [$path, $expanded] = lowSearch(
                $grid,
                $agents[$i],
                $distances[$i],
                $agents[$i]['start'],
                0,
                $p,
                $horizon,
                $budget,
            );
        } catch (BudgetExceeded) {
            return [null, $attempts, 'll_budget'];
        }
        $attempts[] = ['agent' => $i, 'path' => $path, 'expansions' => $expanded];
        if ($path === null)
            return [null, $attempts, $i];
        $paths[$i] = $path;
        $p->reserve($path, 0, true);
    }
    ksort($paths, SORT_NUMERIC);
    return [array_values($paths), $attempts, null];
}
