        boolean stateAllowed(GridMap.Cell cell, int time) {
            if (states.contains(new State(cell, time)))
                return false;
            for (State state : permanent)
                if (state.cell().equals(cell) && state.time() <= time)
                    return false;
            return true;
        }
        boolean actionAllowed(GridMap.Cell source, GridMap.Cell destination, int time) {
            return stateAllowed(destination, time) &&
                (source.equals(destination) ||
                 !moves.contains(new Move(source, destination, time)));
        }
        int arrivalAllowed(GridMap.Cell goal) {
            int arrival = 0;
            for (State state : states)
                if (state.cell().equals(goal))
                    arrival = Math.max(arrival, state.time() + 1);
            return arrival;
        }
    static void reserve(Prohibitions prohibitions, List<GridMap.Cell> path, int startTime,
                        boolean permanent) {
        for (int k = 0; k < path.size(); k++) {
            GridMap.Cell cell = path.get(k);
            int time = startTime + k;
            prohibitions.states.add(new State(cell, time));
            if (k > 0 && !cell.equals(path.get(k - 1)))
                prohibitions.moves.add(new Move(cell, path.get(k - 1), time));
        }
        if (permanent)
            prohibitions.permanent.add(new State(path.getLast(), startTime + path.size() - 1));
    }
    static Search search(GridMap grid, GridMap.Agent agent, Map<GridMap.Cell, Integer> distances,
                         GridMap.Cell start, int t0, Prohibitions prohibitions, int tmax,
                         Budget budget, boolean window) {
        budget.searches++;
        int beginning = budget.expansions, arrival = prohibitions.arrivalAllowed(agent.goal()),
            insertion = 0;
        PriorityQueue<Entry> open =
            new PriorityQueue<>(Comparator.comparingInt(Entry::f)
                                    .thenComparing(Comparator.comparingInt(Entry::g).reversed())
                                    .thenComparingInt(Entry::insertion));
        State startState = new State(start, t0);
        Map<State, State> parents = new HashMap<>();
        parents.put(startState, null);
        open.add(new Entry(distances.get(start), 0, 0, startState));
        while (!open.isEmpty()) {
            State state = open.remove().state();
            budget.expand();
            GridMap.Cell cell = state.cell();
            int time = state.time();
            if (cell.equals(agent.goal()) && time >= arrival || window && time == tmax) {
                List<GridMap.Cell> path = new ArrayList<>();
                for (State current = state; current != null; current = parents.get(current))
                    path.add(current.cell());
                Collections.reverse(path);
                return new Search(path, budget.expansions - beginning);
            }
            if (time == tmax)
                continue;
            for (GridMap.Cell destination : grid.destinations(cell)) {
                State next = new State(destination, time + 1);
                if (!distances.containsKey(destination) || parents.containsKey(next) ||
                    !prohibitions.actionAllowed(cell, destination, time + 1))
                    continue;
                parents.put(next, state);
                int g = time + 1 - t0;
                open.add(new Entry(g + distances.get(destination), g, ++insertion, next));
            }
        }
        return new Search(null, budget.expansions - beginning);
    }
    GridMap() throws Exception {
        List<String> lines =
            Files.readAllLines(Path.of("input/map.txt"), StandardCharsets.US_ASCII);
        String[] f = lines.getFirst().split(" ");
        width = Integer.parseInt(f[0]);
        height = Integer.parseInt(f[1]);
        rows = lines.subList(1, lines.size());
        for (int y = 0; y < height; y++)
            for (int x = 0; x < width; x++)
                if (rows.get(y).charAt(x) == '.')
                    free.add(new Cell(x, y));
    }
    static List<Agent> agents() throws Exception {
        List<Agent> agents = new ArrayList<>();
        List<String> lines =
            Files.readAllLines(Path.of("input/agents.csv"), StandardCharsets.US_ASCII);
        for (String line : lines.subList(1, lines.size())) {
            String[] f = line.split(",");
            agents.add(new Agent(Integer.parseInt(f[0]),
                                 new Cell(Integer.parseInt(f[1]), Integer.parseInt(f[2])),
                                 new Cell(Integer.parseInt(f[3]), Integer.parseInt(f[4]))));
        }
        return agents;
    }
        Settings(int count) throws Exception {
            List<Integer> numberOrder = new ArrayList<>();
            for (int i = 0; i < count; i++)
                numberOrder.add(i);
            orders.add(numberOrder);
            for (String line :
                 Files.readAllLines(Path.of("input/params.txt"), StandardCharsets.US_ASCII)) {
                String[] f = line.split(" ");
                switch (f[0]) {
                case "HORIZON":
                    horizon = Integer.parseInt(f[1]);
                    break;
                case "HL_BUDGET":
                    hlBudget = Integer.parseInt(f[1]);
                    break;
                case "LL_BUDGET":
                    llBudget = Integer.parseInt(f[1]);
                    break;
                case "WINDOW":
                    window = Integer.parseInt(f[1]);
                    step = Integer.parseInt(f[2]);
                    break;
                case "ORDER": {
                    List<Integer> order = new ArrayList<>();
                    for (int i = 1; i < f.length; i++)
                        order.add(Integer.parseInt(f[i]));
                    orders.add(order);
                    break;
                }
                case "RUN":
                    runs.add(new Definition(f[1], f[2]));
                    break;
                case "ANIMATE":
                    animate = f[1];
                    frames = Integer.parseInt(f[2]);
                    break;
                case "DELAY":
                    delays.add(new Delay(Integer.parseInt(f[1]), Integer.parseInt(f[2]),
                                         Integer.parseInt(f[3])));
                    break;
                }
            }
        }

        void solvePp(Run run) {
            int answer = 0;
            for (int index = 0; index < settings.orders.size(); index++) {
                int orderNumber = index + 1;
                List<Integer> order = settings.orders.get(index);
                run.highLevel++;
                int before = run.budget.expansions;
                Prioritized.Result result = Prioritized.tryOrder(
                    grid, agents, distances, settings.horizon, order, run.budget);
                for (Prioritized.Attempt attempt : result.attempts()) {
                    LowLevel.Search search = attempt.search();
                    String prefix =
                        "order " + orderNumber + " agent " + Report.name(attempt.agent());
                    run.trace.add(prefix +
                                  (search.path() == null ? " failed"
                                                         : " cost " + (search.path().size() - 1)) +
                                  " expansions " + search.expansions());
                }
                if (result.aborted()) {
                    run.reason = "ll_budget";
                    break;
                }
                StringBuilder names = new StringBuilder();
                for (int id : order)
                    names.append(Report.name(id));
                int planned = result.attempts().size() - (result.failed() == null ? 0 : 1);
                String summary = "order " + orderNumber + " " + names + " planned " + planned +
                                 " expansions " + (run.budget.expansions - before);
                if (result.failed() == null) {
                    run.specific.add(summary + " ok");
                    run.paths = result.paths();
                    run.status = "OK";
                    answer = orderNumber;
                    break;
                }
                run.specific.add(summary + " failed " + Report.name(result.failed()));
            }
            if (run.status.equals("OK"))
                run.end = "end OK order " + answer;
            else {
                if (run.reason == null)
                    run.reason = "orders";
                run.end = "end FAIL " + run.reason;
            }
        }
        Validation validate(List<List<GridMap.Cell>> paths) {
            int start = 0, goal = 0, moves = 0;
            for (int id = 0; id < agents.size(); id++) {
                List<GridMap.Cell> path = paths.get(id);
                if (!path.getFirst().equals(agents.get(id).start()))
                    start++;
                if (!path.getLast().equals(agents.get(id).goal()))
                    goal++;
                for (int t = 1; t < path.size(); t++)
                    if (!grid.destinations(path.get(t - 1)).contains(path.get(t)))
                        moves++;
            }
            List<Cbs.Conflict> collisions = Cbs.conflicts(paths);
            int vertex = (int)collisions.stream().filter(c -> c.kind().equals("vertex")).count(),
                edge = collisions.size() - vertex, follows = 0;
            int horizon = Cbs.cost(paths, "MAKESPAN");
            for (int time = 1; time <= horizon; time++)
                for (int a = 0; a < paths.size(); a++) {
                    GridMap.Cell pastA = Cbs.position(paths.get(a), time - 1),
                                 nextA = Cbs.position(paths.get(a), time);
                    if (nextA.equals(pastA))
                        continue;
                    for (int b = 0; b < paths.size(); b++)
                        if (a != b && nextA.equals(Cbs.position(paths.get(b), time - 1)) &&
                            !Cbs.position(paths.get(b), time).equals(pastA))
                            follows++;
                }
            return new Validation(start, goal, moves, vertex, edge, follows, collisions);
        }
        Metrics metrics(List<List<GridMap.Cell>> paths, String function) {
            Metrics result = new Metrics();
            result.soc = Cbs.cost(paths, "SOC");
            result.makespan = Cbs.cost(paths, "MAKESPAN");
            result.cost = function.equals("SOC") ? result.soc : result.makespan;
            result.ratio =
                Report.fraction(result.cost, function.equals("SOC") ? socLower : makespanLower);
            for (int id = 0; id < agents.size(); id++) {
                List<GridMap.Cell> path = paths.get(id);
                PathStats stats = pathStats(path, id);
                result.agents.add(stats);
                result.sumExtra += stats.extra();
                result.maxExtra = Math.max(result.maxExtra, stats.extra());
                result.waits += stats.waits();
                result.turns += stats.turns();
                for (GridMap.Cell cell : new HashSet<>(path))
                    result.traffic.merge(cell, 1, Integer::sum);
                for (int t = 1; t < path.size(); t++)
                    if (path.get(t).equals(path.get(t - 1)))
                        result.cellWaits.merge(path.get(t), 1, Integer::sum);
            }
            for (int time = 0; time <= result.makespan; time++) {
                int arrived = 0;
                for (List<GridMap.Cell> path : paths)
                    if (path.size() - 1 <= time)
                        arrived++;
                result.arrivals.add(arrived);
            }
            result.hotspots = new ArrayList<>(result.traffic.keySet());
            result.hotspots.sort(
                Comparator.comparingInt((GridMap.Cell cell) -> - result.traffic.get(cell))
                    .thenComparingInt(GridMap.Cell::y)
                    .thenComparingInt(GridMap.Cell::x));
            if (result.hotspots.size() > 5)
                result.hotspots = new ArrayList<>(result.hotspots.subList(0, 5));
            return result;
        }
        void solve() {
            for (Definition definition : settings.runs) {
                Run run = new Run(definition, settings.llBudget);
                switch (definition.solver()) {
                case "PP":
                    solvePp(run);
                    break;
                case "CBS":
                case "ICBS":
                    solveCbs(run);
                    break;
                default:
                    solveWhca(run);
                    break;
                }
                if (run.status.equals("OK")) {
                    run.metrics = metrics(run.paths, definition.function());
                    run.validation = validate(run.paths);
                    run.execution = delayedExecution(run.paths);
                }
                results.add(run);
            }
        }

            Metrics m = run.metrics;
            Execution execution = run.execution;
            List<String> lines = new ArrayList<>();
            lines.add("run " + run.name() + " solver " + run.definition.solver() +
                      " cost_function " + run.definition.function() + " result OK cost " + m.cost +
                      " soc " + m.soc + " makespan " + m.makespan + " ratio " + m.ratio);
            lines.addAll(run.specific);
            for (int id = 0; id < agents.size(); id++) {
                PathStats p = m.agents.get(id);
                lines.add("agent " + Report.name(id) + " cost " + p.cost() + " extra " + p.extra() +
                          " waits " + p.waits() + " turns " + p.turns() + " arrive " + p.arrive() +
                          " leave " + p.leave() + " moves " + p.actions());
            }
            lines.add("arrivals " + joinNumbers(m.arrivals));
            List<String> hot = new ArrayList<>();
            for (GridMap.Cell cell : m.hotspots)
                hot.add(cell + ":" + m.traffic.get(cell));
            lines.add("hotspots " + String.join(" ", hot));
            if (execution.deadlock() != null)
                lines.add("execution deadlock time " + execution.deadlock());
            else
                lines.add("execution soc " + execution.soc() + " makespan " + execution.makespan() +
                          " blocked " + execution.blocked() + " finish " +
                          joinNumbers(execution.finish()));
            return lines;
        }

        List<String> animationLines() {
            Run selected = results.stream()
                               .filter(run -> run.name().equals(settings.animate))
                               .findFirst()
                               .orElseThrow();

                return List.of("run " + selected.name() + " FAIL");
            List<String> lines = new ArrayList<>();
            lines.add("run " + selected.name() + " frames " + settings.frames);
            for (int time = 0; time < settings.frames; time++) {
                int done = 0, moving = 0, waiting = 0;
                Map<GridMap.Cell, Character> glyphs = new HashMap<>();
                for (GridMap.Agent agent : agents)
                    glyphs.put(agent.goal(), Character.toLowerCase(agent.name()));
                for (int id = 0; id < agents.size(); id++) {
                    List<GridMap.Cell> path = selected.paths.get(id);
                    GridMap.Cell cell = Cbs.position(path, time);
                    glyphs.put(cell, Report.name(id));
                    if (time >= path.size() - 1)
                        done++;
                    else if (time >= 1 && !cell.equals(Cbs.position(path, time - 1)))
                        moving++;
                    else
                        waiting++;
                }
                lines.add("");
                lines.add("frame " + time + " done " + done + " moving " + moving + " waiting " +
                          waiting);
                lines.addAll(picture(glyphs));
            }
            lines.add("");
            lines.add("traffic");
            lines.addAll(countPicture(selected.metrics.traffic));
            lines.add("");
            lines.add("waits");
            lines.addAll(countPicture(selected.metrics.cellWaits));
            return lines;
        }

    static GridMap.Cell position(List<GridMap.Cell> path, int time) {
        return path.get(Math.min(time, path.size() - 1));
    }
    static List<Conflict> conflicts(List<List<GridMap.Cell>> paths) {
        List<Conflict> result = new ArrayList<>();
        int horizon = paths.stream().mapToInt(path -> path.size() - 1).max().orElse(0);
        for (int time = 1; time <= horizon; time++)
            for (int a = 0; a < paths.size(); a++)
                for (int b = a + 1; b < paths.size(); b++) {
                    GridMap.Cell currentA = position(paths.get(a), time),
                                 currentB = position(paths.get(b), time);
                    if (currentA.equals(currentB))
                        result.add(new Conflict("vertex", time, a, b, currentA, currentA));
                    else if (currentA.equals(position(paths.get(b), time - 1)) &&
                             currentB.equals(position(paths.get(a), time - 1)))
                        result.add(new Conflict("edge", time, a, b,
                                                position(paths.get(a), time - 1), currentA));
                }
        return result;
    }
    static int cost(List<List<GridMap.Cell>> paths, String function) {
        return function.equals("SOC")
            ? paths.stream().mapToInt(path -> path.size() - 1).sum()
            : paths.stream().mapToInt(path -> path.size() - 1).max().orElse(0);
    }
    static Result tryOrder(GridMap grid, List<GridMap.Agent> agents,
                           List<Map<GridMap.Cell, Integer>> distances, int horizon,
                           List<Integer> order, LowLevel.Budget budget) {
        LowLevel.Prohibitions prohibitions = new LowLevel.Prohibitions();
        List<List<GridMap.Cell>> paths = new ArrayList<>(Collections.nCopies(agents.size(), null));
        List<Attempt> attempts = new ArrayList<>();
        for (int id : order) {
            LowLevel.Search search;
            try {
                search =
                    LowLevel.search(grid, agents.get(id), distances.get(id), agents.get(id).start(),
                                    0, prohibitions, horizon, budget, false);
            } catch (LowLevel.BudgetExceeded exception) {
                return new Result(null, attempts, null, true);
            }
            attempts.add(new Attempt(id, search));
            if (search.path() == null)
                return new Result(null, attempts, id, false);
            paths.set(id, search.path());
            LowLevel.reserve(prohibitions, search.path(), 0, true);
        }
        return new Result(paths, attempts, null, false);
    }
