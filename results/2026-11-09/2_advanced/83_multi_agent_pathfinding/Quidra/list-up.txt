class Prohibitions(int cell_count)
    states := distinct<int>()
    moves := distinct<int>()
    permanent := map<int, int>()

bool state_allowed(Prohibitions restrictions, int cell, int tick)
    if restrictions.states.has(tick * restrictions.cell_count + cell)
        return false
    match restrictions.permanent.get(cell)
        int start
            return tick < start
        none
            return true
bool action_allowed(Prohibitions restrictions, int source, int destination, int tick)
    if not state_allowed(restrictions, destination, tick)
        return false
    key := (tick * restrictions.cell_count + source) * restrictions.cell_count + destination
    return source == destination or not restrictions.moves.has(key)
int arrival_allowed(Prohibitions restrictions, int goal)
    result := 0
    for key := restrictions.states.values()[]
        if key % restrictions.cell_count == goal
            next := key / restrictions.cell_count + 1
            if next > result
                result = next
    return result
void reserve_path(Prohibitions &restrictions, int[] path, start_tick := 0, permanent := false)
    cells := restrictions.cell_count
    for: step < int(#path)
        tick := start_tick + step
        .restrictions.states.add(tick * cells + path[step])
        if step > 0 and path[step] != path[step - 1]
            .restrictions.moves.add((tick * cells + path[step]) * cells + path[step - 1])
    if permanent
        goal := path[-1]
        start := start_tick + int(#path) - 1
        match restrictions.permanent.get(goal)
            int previous
                if previous < start
                    start = previous
            none
                void
        .restrictions.permanent.set(goal, start)
SearchResult search_path(grid_map.Grid grid, grid_map.Agent agent, int[] distances, int start, int start_tick, Prohibitions restrictions, int maximum_tick, Budget &budget, window := false)
    budget.searches += 1
    beginning := budget.expansions
    arrival := arrival_allowed(restrictions, agent.goal)
    root := SearchNode(start, start_tick, distances[start], 0, 0, -1)
    heap := [root]
    generated := [root]
    distinct<int> seen
    cells := int(#grid.free)
    .seen.add(start_tick * cells + start)
    while #heap != 0
        current := pop_open(&heap)
        budget.expansions += 1
        expanded := budget.expansions - beginning
        if budget.limit >= 0 and budget.expansions > budget.limit
            return SearchResult("aborted", [], expanded)
        if (current.cell == agent.goal and current.tick >= arrival) or (window and current.tick == maximum_tick)
            int[] reversed := []
            cursor := current.number
            while cursor >= 0
                reversed = reversed.append(generated[cursor].cell)
                cursor = generated[cursor].parent
            return SearchResult("ok", reversed[::-1], expanded)
        if current.tick == maximum_tick
            continue
        for destination := grid.destinations[current.cell][]
            tick := current.tick + 1
            key := tick * cells + destination
            if distances[destination] < 0 or seen.has(key) or not action_allowed(restrictions, current.cell, destination, tick)
                continue
            .seen.add(key)
            number := int(#generated)
            g := tick - start_tick
            node := SearchNode(destination, tick, g + distances[destination], g, number, current.number)
            generated = generated.append(node)
            push_open(&heap, node)
    return SearchResult("no_path", [], budget.expansions - beginning)
Grid read_grid()
    rows := input_lines("map.txt")
    dimensions := rows[0].split(" ")
    width := int(dimensions[0])
    height := int(dimensions[1])
    rows = rows[1:]
    bool[] free := []
    for: y < height
        for: x < width
            free = free.append(rows[y][x] == ".")
    int[][] destinations := []
    dx := [0, 0, 1, 0, -1]
    dy := [0, -1, 0, 1, 0]
    for: cell < width * height
        int[] adjacent := []
        if free[cell]
            for: action < 5
                x := cell % width + dx[action]
                y := cell / width + dy[action]
                if x >= 0 and x < width and y >= 0 and y < height and free[y * width + x]
                    adjacent = adjacent.append(y * width + x)
        destinations = destinations.append(adjacent)
    return Grid(width, height, rows, free, destinations)
Agent[] read_agents(Grid grid)
    rows := input_lines("agents.csv")
    Agent[] result := []
    names := "ABCDEFGHIJKLMNOPQRST"
    for: 1 <= i < int(#rows)
        fields := rows[i].split(",")
        id := int(fields[0])
        int start := int(fields[2]) * grid.width + int(fields[1])
        int goal := int(fields[4]) * grid.width + int(fields[3])
        result = result.append(Agent(id, start, goal, names[id]))
    return result
Parameters read_parameters(int count)
    horizon := 0
    high_budget := 0
    low_budget := 0
    window := 0
    step := 0
    int[][] orders := []
    int[] number_order := []
    for: i < count
        number_order = number_order.append(i)
    orders = orders.append(number_order)
    RunDefinition[] runs := []
    animate := ""
    frames := 0
    Delay[] delays := []
    for row := grid_map.input_lines("params.txt")[]
        fields := row.split(" ")
        kind := fields[0]
        if kind == "HORIZON"
            horizon = int(fields[1])
        elif kind == "HL_BUDGET"
            high_budget = int(fields[1])
        elif kind == "LL_BUDGET"
            low_budget = int(fields[1])
        elif kind == "WINDOW"
            window = int(fields[1])
            step = int(fields[2])
        elif kind == "ORDER"
            orders = orders.append([for word := fields[1:][] then int(word)])
        elif kind == "RUN"
            runs = runs.append(RunDefinition(fields[1], fields[2]))
        elif kind == "ANIMATE"
            animate = fields[1]
            frames = int(fields[2])
        else
            delays = delays.append(Delay(int(fields[1]), int(fields[2]), int(fields[3])))
    return Parameters(horizon, high_budget, low_budget, window, step, orders, runs, animate, frames, delays)
void solve_pp(Run &run, grid_map.Grid grid, grid_map.Agent[] agents, int[][] distances, Parameters settings)
    for: index < int(#settings.orders)
        number := index + 1
        run.high_level += 1
        int beginning := run.budget.expansions
        result := prioritized.try_order(grid, agents, distances, settings.horizon, settings.orders[index], &run.budget)
        planned := 0
        for attempt := result.attempts[]
            prefix := "order {number} agent " + report.agent_name(attempt.agent)
            if attempt.result.status == "ok"
                planned += 1
                run.trace = run.trace.append(prefix + " cost {int(#attempt.result.path) - 1} expansions {attempt.result.expansions}")
            else
                run.trace = run.trace.append(prefix + " failed expansions {attempt.result.expansions}")
        if result.status == "ll_budget"
            run.reason = "ll_budget"
            return
        suffix := if result.status == "ok" then "ok" else "failed " + report.agent_name(result.failed_agent)
        run.specific = run.specific.append("order {number} " + agent_order(settings.orders[index]) + " planned {planned} expansions {run.budget.expansions - beginning} {suffix}")
        if result.status == "ok"
            run.success = true
            run.answer_order = number
            run.paths = result.paths
            return
    run.reason = "orders"
PlanMetrics plan_metrics(int[][] paths, grid_map.Grid grid, grid_map.Agent[] agents, Overview overview, string cost_function)
    result := PlanMetrics()
    result.soc = cbs.plan_soc(paths)
    result.makespan = cbs.plan_makespan(paths)
    result.cost = if cost_function == "SOC" then result.soc else result.makespan
    result.traffic = array(#grid.free, fill = 0)
    result.cell_waits = array(#grid.free, fill = 0)
    for: i < int(#agents)
        metrics := path_metrics(paths[i], agents[i].goal, overview.distances[i], grid.width)
        result.agents = result.agents.append(metrics)
        result.sum_extra += metrics.extra
        if metrics.extra > result.max_extra
            result.max_extra = metrics.extra
        result.waits += metrics.waits
        result.turns += metrics.turns
        distinct<int> visited
        for cell := paths[i][]
            .visited.add(cell)
        for cell := visited.values()[]
            result.traffic[cell] += 1
        for: 1 <= tick < int(#paths[i])
            if paths[i][tick] == paths[i][tick - 1]
                result.cell_waits[paths[i][tick]] += 1
    for: tick <= result.makespan
        count := 0
        for path := paths[]
            if int(#path) - 1 <= tick
                count += 1
        result.arrivals = result.arrivals.append(count)
    selected := array(#grid.free, fill = false)
    for: rank < 5
        best := -1
        for: cell < int(#grid.free)
            if not selected[cell] and result.traffic[cell] > 0 and (best < 0 or result.traffic[cell] > result.traffic[best])
                best = cell
        if best < 0
            break
        selected[best] = true
        result.hotspots = result.hotspots.append(best)
    return result
Run[] run_solvers(grid_map.Grid grid, grid_map.Agent[] agents, int[][] distances, Parameters settings, Overview overview)
    Run[] results := []
    for definition := settings.runs[]
        run := Run(definition, settings.low_budget)
        if definition.solver == "PP"
            solve_pp(&run, grid, agents, distances, settings)
        elif definition.solver == "WHCA"
            solve_whca(&run, grid, agents, distances, settings)
        else
            solve_cbs(&run, grid, agents, distances, settings)
        if run.success
            run.metrics = plan_metrics(run.paths, grid, agents, overview, definition.cost_function)
            run.execution = delayed_execution(run.paths, settings.delays)
            string ending
            if definition.solver == "PP"
                ending = "end OK order {run.answer_order}"
            elif definition.solver == "WHCA"
                ending = "end OK rounds {run.high_level}"
            else
                ending = "end OK node {run.answer_number} cost {run.metrics.cost}"
            run.trace = run.trace.append(ending)
        else
            run.trace = run.trace.append("end FAIL " + run.reason)
        results = results.append(run)
    return results
string[] successful_lines(Run run, grid_map.Grid grid, grid_map.Agent[] agents)
    string[] lines := []
    for: i < int(#agents)
        AgentMetrics metrics := run.metrics.agents[i]
        lines = lines.append("agent {agents[i].name} cost {metrics.cost} extra {metrics.extra} waits {metrics.waits} turns {metrics.turns} arrive {metrics.arrival} leave {metrics.leaves} moves {metrics.actions}")
    lines = lines.append("arrivals " + [for count := run.metrics.arrivals[] then string(count)].join(" "))
    string[] hotspots := []
    for cell := run.metrics.hotspots[]
        hotspots = hotspots.append(report.cell_text(cell, grid.width) + ":{run.metrics.traffic[cell]}")
    lines = lines.append("hotspots " + hotspots.join(" "))
    if run.execution.deadlock
        lines = lines.append("execution deadlock time {run.execution.cutoff}")
    else
        lines = lines.append("execution soc {run.execution.soc} makespan {run.execution.makespan} blocked {run.execution.blocked} finish " + [for tick := run.execution.finish[] then string(tick)].join(" "))
    return lines
string[] validate_plan(string name, int[][] paths, grid_map.Grid grid, grid_map.Agent[] agents, int[] classes)
    starts := 0
    goals := 0
    moves := 0
    for: i < int(#agents)
        if paths[i][0] != agents[i].start
            starts += 1
        if paths[i][-1] != agents[i].goal
            goals += 1
        for: 1 <= tick < int(#paths[i])
            if not contains_cell(grid.destinations[paths[i][tick - 1]], paths[i][tick])
                moves += 1
    cbs.Conflict[] conflicts := cbs.find_conflicts(paths)
    vertices := 0
    edges := 0
    for conflict := conflicts[]
        if conflict.kind == "vertex"
            vertices += 1
        else
            edges += 1
    follows := 0
    horizon := cbs.plan_makespan(paths)
    for: 1 <= tick <= horizon
        for: first < int(#agents)
            current := cbs.path_position(paths[first], tick)
            previous := cbs.path_position(paths[first], tick - 1)
            if current == previous
                continue
            for: second < int(#agents)
                if first != second and current == cbs.path_position(paths[second], tick - 1) and cbs.path_position(paths[second], tick) != previous
                    follows += 1
    verdict := if starts + goals + moves + vertices + edges == 0 then "VALID" else "INVALID"
    lines := ["plan {name} verdict {verdict} start {starts} goal {goals} moves {moves} vertex {vertices} edge {edges} follows {follows}"]
    for: i < int(#conflicts)
        line := "  " + report.conflict_text(conflicts[i], grid.width)
        if #classes > 0
            line += " " + class_name(classes[i])
        lines = lines.append(line)
    return lines
string[] validation_lines(Run[] results, grid_map.Grid grid, grid_map.Agent[] agents, Overview overview)
    lines := validate_plan("INDEPENDENT", overview.independent, grid, agents, overview.conflict_classes)
    for run := results[]
        if run.success
            lines = lines.concat(validate_plan(run.name, run.paths, grid, agents, []))
        else
            lines = lines.append("plan {run.name} verdict NONE")
    return lines
string[] occupancy_picture(grid_map.Grid grid, grid_map.Agent[] agents, int[][] paths, int tick)
    string[] cells := []
    for: cell < int(#grid.free)
        cells = cells.append(if grid.free[cell] then "." else "@")
    lower_names := "abcdefghijklmnopqrst"
    for: i < int(#agents)
        cells[agents[i].goal] = lower_names[i]
    for: i < int(#agents)
        cells[cbs.path_position(paths[i], tick)] = agents[i].name
    string[] rows := []
    for: y < grid.height
        rows = rows.append(cells[y * grid.width:<(y + 1) * grid.width].join(""))
    return rows
string[] animation_lines(Run[] results, grid_map.Grid grid, grid_map.Agent[] agents, Parameters settings)
    selected := 0
    for: i < int(#results)
        if results[i].name == settings.animate
            selected = i
    run := results[selected]
    if not run.success
        return ["run {run.name} FAIL"]
    lines := ["run {run.name} frames {settings.frames}"]
    for: tick < settings.frames
        done := 0
        moving := 0
        waiting := 0
        for path := run.paths[]
            length := int(#path) - 1
            if tick >= length
                done += 1
            elif tick >= 1 and cbs.path_position(path, tick) != cbs.path_position(path, tick - 1)
                moving += 1
            else
                waiting += 1
        lines = lines.concat(["", "frame {tick} done {done} moving {moving} waiting {waiting}"]).concat(occupancy_picture(grid, agents, run.paths, tick))
    return lines.concat(["", "traffic"]).concat(count_picture(grid, run.metrics.traffic)).concat(["", "waits"]).concat(count_picture(grid, run.metrics.cell_waits))

int path_position(int[] path, int tick)
    index := if tick < int(#path) then tick else int(#path) - 1
    return path[index]
int plan_soc(int[][] paths)
    result := 0
    for path := paths[]
        result += int(#path) - 1
    return result
int plan_makespan(int[][] paths)
    result := 0
    for path := paths[]
        length := int(#path) - 1
        if length > result
            result = length
    return result
int plan_cost(int[][] paths, string cost_function)
    return if cost_function == "SOC" then plan_soc(paths) else plan_makespan(paths)
Conflict[] find_conflicts(int[][] paths)
    Conflict[] result := []
    horizon := plan_makespan(paths)
    for: 1 <= tick <= horizon
        for: first < int(#paths)
            for: first + 1 <= second < int(#paths)
                a := path_position(paths[first], tick)
                b := path_position(paths[second], tick)
                if a == b
                    result = result.append(Conflict("vertex", tick, first, second, a, a))
                elif a == path_position(paths[second], tick - 1) and b == path_position(paths[first], tick - 1)
                    result = result.append(Conflict("edge", tick, first, second, path_position(paths[first], tick - 1), a))
    return result
OrderResult try_order(grid_map.Grid grid, grid_map.Agent[] agents, int[][] distances, int horizon, int[] order, low_level.Budget &budget)
    restrictions := low_level.Prohibitions(int(#grid.free))
    int[][] paths := []
    for agent := agents[]
        paths = paths.append([])
    Attempt[] attempts := []
    for agent_id := order[]
        result := low_level.search_path(grid, agents[agent_id], distances[agent_id], agents[agent_id].start, 0, restrictions, horizon, &budget)
        if result.status == "aborted"
            return OrderResult("ll_budget", agent_id, [], attempts)
        attempts = attempts.append(Attempt(agent_id, result))
        if result.status != "ok"
            return OrderResult("failed", agent_id, [], attempts)
        paths[agent_id] = result.path
        low_level.reserve_path(&restrictions, result.path, 0, true)
    return OrderResult("ok", -1, paths, attempts)
