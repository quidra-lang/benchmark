class DictionaryTrie {
    class Node {
        val children=arrayOfNulls<Node>(26);
        var terminal=false
    }
    val root=Node();
    var nodes=1;
    var words=0
    fun add(word: String): Boolean {
        var node=root;
        for(letter in word) {
            val i=letter-'A';
            var child=node.children[i];
            if(child==null) {
                child=Node();
                node.children[i]=child;
                nodes++
            };
            node=child
        };
        if(node.terminal)return false;
        node.terminal=true;
        words++;
        return true
    }
    fun prefix(word: String): Node? {
        var node=root;
        for(letter in word)node=node.children[letter-'A']?:return null;
        return node
    }
object Game {
    const val TILE_ORDER="ABCDEFGHIJKLMNOPQRSTUVWXYZ?"
    class Minstd(seed: Int) {
        var state=seed.toLong();
        fun next(): Int {
            state=state*48271%2147483647;
            return state.toInt()
        }
    }
    fun shuffle(bag: MutableList<Char>,random: Minstd) {
        for(i in bag.lastIndex downTo 1) {
            val j=random.next()%(i+1);
            val tile=bag[i];
            bag[i]=bag[j];
            bag[j]=tile
        }
    }
    fun bag(counts: IntArray,random: Minstd): MutableList<Char> {
        val result=mutableListOf<Char>();
        for(tile in TILE_ORDER)repeat(counts[tile.code]) {
            result.add(tile)
        };
        shuffle(result,random);
        return result
    }
    fun draw(bag: MutableList<Char>,count: Int): String {
        val result=StringBuilder();
        repeat(minOf(count,bag.size)) {
            result.append(bag.removeAt(bag.lastIndex))
        };
        return result.toString()
    }
object MoveGenerator {
    const val LETTERS="ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    fun anchors(board: Array<CharArray>): Array<BooleanArray> {
        val result=Array(15) {
            BooleanArray(15)
        };
        if(Board.tiles(board).isEmpty()) {
            result[7][7]=true;
            return result
        };
        for(r in 0..14)for(c in 0..14)if(board[r][c]=='.'&&Board.neighbors(r,c).any {
            board[it.row][it.column]!='.'
        })result[r][c]=true;
        return result
    }
    fun checks(board: Array<CharArray>,trie: DictionaryTrie): Array<IntArray> {
        val result=Array(15) {
            IntArray(15) {
                -1
            }
        };
        for(r in 0..14)for(c in 0..14)if(board[r][c]=='.') {
            var first=r;
            var last=r;
            while(first>0&&board[first-1][c]!='.')first--;
            while(last+1<15&&board[last+1][c]!='.')last++;
            if(first==last)continue;
            val prefix=(first until r).map {
                board[it][c].uppercaseChar()
            }.joinToString("");
            val suffix=(r+1..last).map {
                board[it][c].uppercaseChar()
            }.joinToString("");
            var mask=0;
            for(i in 0..25)if(trie.contains(prefix+('A'+i)+suffix))mask=mask or (1 shl i);
            result[r][c]=mask
        };
        return result
    }
    class Search(val board: Array<CharArray>,rack: String,val trie: DictionaryTrie,val anchors: Array<BooleanArray>,
        val checks: Array<IntArray>) {
        val counts=IntArray(128);
        val result=mutableListOf<Board.Move>();
        var row=0;
        var anchor=0
        init {
            for(c in rack)counts[c.code]++
        }
        fun extend(node: DictionaryTrie.Node,column: Int,word: String,start: Int) {
            val empty=column==15||board[row][column]=='.';
            if(empty&&column>anchor&&node.terminal)result.add(Board.Move(row,start,0,word));
            if(column==15)return;
            if(!empty) {
                val letter=board[row][column];
                val child=node.children[letter.uppercaseChar()-'A'];
                if(child!=null)extend(child,column+1,word+letter,start);
                return
            };
            for(i in 0..25) {
                val child=node.children[i]?:continue;
                if(checks[row][column]!=-1&&(checks[row][column] and (1 shl i))==0)continue;
                val letter='A'+i;
                for(blank in 0..1) {
                    val tile=if(blank==0)letter else '?';
                    val written=if(blank==0)letter else letter.lowercaseChar();
                    if(counts[tile.code]>0) {
                        counts[tile.code]--;
                        extend(child,column+1,word+written,start);
                        counts[tile.code]++
                    }
                }
            }
        }
        fun left(node: DictionaryTrie.Node,prefix: String,remaining: Int) {
            extend(node,anchor,prefix,anchor-prefix.length);
            if(remaining==0)return;
            for(i in 0..25) {
                val child=node.children[i]?:continue;
                val letter='A'+i;
                for(blank in 0..1) {
                    val tile=if(blank==0)letter else '?';
                    val written=if(blank==0)letter else letter.lowercaseChar();
                    if(counts[tile.code]>0) {
                        counts[tile.code]--;
                        left(child,prefix+written,remaining-1);
                        counts[tile.code]++
                    }
                }
            }
        }
        fun generate(): List<Board.Move> {
            for(r in 0..14)for(a in 0..14)if(anchors[r][a]) {
                row=r;
                anchor=a;
                if(anchor>0&&board[row][anchor-1]!='.') {
                    var first=anchor-1;
                    while(first>0&&board[row][first-1]!='.')first--;
                    val prefix=String(board[row],first,anchor-first);
                    val node=trie.prefix(prefix.uppercase());
                    if(node!=null)extend(node,anchor,prefix,first)
                } else {
                    var first=anchor-1;
                    while(first>=0&&board[row][first]=='.'&&!anchors[row][first])first--;
                    left(trie.root,"",anchor-first-1)
                }
            };
            return result
        }
    }
    fun generate(board: Array<CharArray>,rack: String,trie: DictionaryTrie): List<Board.Move> {
        val anchors=anchors(board);
        val across=Search(board,rack,trie,anchors,checks(board,trie)).generate();
        val transposed=Board.transpose(board);
        val downAnchors=Array(15) {
            r->BooleanArray(15) {
                c->anchors[c][r]
            }
        };
        val down=Search(transposed,rack,trie,downAnchors,checks(transposed,trie)).generate().map {
            Board.Move(it.column,it.row,1,it.word)
        };
        val distinct=linkedMapOf<List<Board.Tile>,Board.Move>();
        for(move in across+down) {
            val tiles=Scoring.placement(board,move);
            if(tiles.size==1&&move.direction==1) {
                val tile=tiles.first();
                if(Board.run(Scoring.apply(board,tiles),tile.row,tile.column,0).word.length>=2)continue
            };
            distinct[tiles.sortedWith(compareBy<Board.Tile> {
                it.row
            }.thenBy {
                it.column
            })]=move
        };
        return distinct.values.toList()
    }
}
object Scoring {
    const val RACK_SIZE=7;
    const val BINGO_BONUS=50
    data class WordScore(val word: Board.Move,val score: Int)
    data class Result(val score: Int,val tiles: List<Board.Tile>,val words: List<WordScore>)
    fun squares(move: Board.Move)=move.word.mapIndexed {
        i,letter->Board.Tile(move.row+move.direction*i,move.column+(1-move.direction)*i,letter)
    }
    fun placement(board: Array<CharArray>,move: Board.Move)=squares(move).filter {
        board[it.row][it.column]=='.'
    }
    fun apply(board: Array<CharArray>,tiles: List<Board.Tile>): Array<CharArray> {
        val result=Array(board.size) {
            board[it].clone()
        };
        for(tile in tiles)result[tile.row][tile.column]=tile.letter;
        return result
    }
    fun crosses(board: Array<CharArray>,move: Board.Move): List<Board.Move> {
        val tiles=placement(board,move);
        val after=apply(board,tiles);
        return tiles.map {
            Board.run(after,it.row,it.column,1-move.direction)
        }.filter {
            it.word.length>=2
        }
    }
    fun wordScore(board: Array<CharArray>,word: Board.Move,values: IntArray): Int {
        var score=0;
        var multiplier=1;
        for(tile in squares(word)) {
            var value=if(tile.letter in 'A'..'Z')values[tile.letter.code] else 0;
            if(board[tile.row][tile.column]=='.') {
                val premium=Board.premiums[tile.row][tile.column];
                value*=if(premium=='+')2 else if(premium=='^')3 else 1;
                multiplier*=if(premium=='*')2 else if(premium=='#')3 else 1
            };
            score+=value
        };
        return score*multiplier
    }
    fun evaluate(board: Array<CharArray>,move: Board.Move,values: IntArray): Result {
        val words=(listOf(move)+crosses(board,move)).map {
            WordScore(it,wordScore(board,it,values))
        };
        val tiles=placement(board,move);
        return Result(words.sumOf {
            it.score
        }+(if(tiles.size==RACK_SIZE)BINGO_BONUS else 0),tiles,words)
    }
}
object Board {
    const val SIZE=15;
    const val COLUMNS="ABCDEFGHIJKLMNO"
    val premiums=listOf("#..+...#...+..#",".*...^...^...*.","..*...+.+...*..","+..*...+...*..+","....*.....*....",
        ".^...^...^...^.","..+...+.+...+..","#..+...*...+..#","..+...+.+...+..",".^...^...^...^.","....*.....*....",
        "+..*...+...*..+","..*...+.+...*..",".*...^...^...*.","#..+...#...+..#")
    data class Move(val row: Int,val column: Int,val direction: Int,val word: String)
    data class Tile(val row: Int,val column: Int,val letter: Char)
    data class Cell(val row: Int,val column: Int)
    fun inside(r: Int,c: Int)=r in 0 until SIZE&&c in 0 until SIZE
    fun square(r: Int,c: Int)="${COLUMNS[c]}${r+1}"
    fun position(move: Move)=if(move.direction==0)"${move.row+1}${COLUMNS[move.column]}" else "${COLUMNS[move.column]}${move.row+1}"
    fun parse(text: String,word: String): Move? {
        if(text.matches(Regex("([1-9]|1[0-5])[A-O]")))return Move(text.dropLast(1).toInt()-1,COLUMNS.indexOf(text.last()),
            0,word);
        if(text.matches(Regex("[A-O]([1-9]|1[0-5])")))return Move(text.drop(1).toInt()-1,COLUMNS.indexOf(text.first()),
            1,word);
        return null
    }
    fun empty()=Array(SIZE) {
        CharArray(SIZE) {
            '.'
        }
    }
    fun transpose(board: Array<CharArray>)=Array(SIZE) {
        r->CharArray(SIZE) {
            c->board[c][r]
        }
    }
    fun transposeTiles(tiles: List<Tile>)=tiles.map {
        Tile(it.column,it.row,it.letter)
    }
    fun tiles(board: Array<CharArray>): List<Tile> {
        val result=mutableListOf<Tile>();
        for(r in 0 until SIZE)for(c in 0 until SIZE)if(board[r][c]!='.')result.add(Tile(r,c,board[r][c]));
        return result
    }
    fun neighbors(r: Int,c: Int)=listOf(Cell(r-1,c),Cell(r+1,c),Cell(r,c-1),Cell(r,c+1)).filter {
        inside(it.row,it.column)
    }
    fun run(board: Array<CharArray>,row: Int,column: Int,direction: Int): Move {
        val dr=direction;
        val dc=1-direction;
        var r=row;
        var c=column;
        while(inside(r-dr,c-dc)&&board[r-dr][c-dc]!='.') {
            r-=dr;
            c-=dc
        };
        val firstRow=r;
        val firstColumn=c;
        val word=StringBuilder();
        while(inside(r,c)&&board[r][c]!='.') {
            word.append(board[r][c]);
            r+=dr;
            c+=dc
        };
        return Move(firstRow,firstColumn,direction,word.toString())
    }
    val standard=compareByDescending<Evaluated> {
        it.score
    }.thenBy {
        it.move.row
    }.thenBy {
        it.move.column
    }.thenBy {
        it.move.direction
    }.thenBy {
        it.move.word
    }
    fun validName(text: String)=text.matches(Regex("[a-z0-9_]{1,20}"))
    fun validNumber(text: String,low: Int,high: Int)=text.length<=10&&text.matches(Regex("0|[1-9][0-9]*"))&&text.toLong() in low.toLong()..high.toLong()
    fun validRack(text: String)=text.matches(Regex("[A-Z?]{1,7}"))
    fun collection(text: String)=text.toList().sorted().joinToString("").ifEmpty {
        "-"
    }
    fun removeTiles(rack: String,removed: String): String {
        val remaining=IntArray(128);
        for(c in rack)remaining[c.code]++;
        for(c in removed)remaining[c.code]--;
        return buildString {
            for(c in remaining.indices)repeat(remaining[c]) {
                append(c.toChar())
            }
        }
    }
    fun notation(move: Evaluated)=Board.position(move.move)+" "+move.move.word
    fun strategy(name: String)=when(name) {
        "tiles"->compareByDescending<Evaluated> {
            it.tiles.size
        }.then(standard);
        "equity"->compareByDescending<Evaluated> {
            it.score+it.value
        }.then(standard);
        else->standard
    }
    fun jobError(job: List<String>,seen: Set<String>): String? {
        val fields=job.first().split(' ');
        val kind=fields[0];
        val name=fields[1];
        if(!validName(name))return "bad name";
        if(name in seen)return "duplicate name";
        if(kind=="TOPN") {
            if(!validNumber(fields[2],1,99))return "bad count";
            if(fields[3] !in strategies)return "bad order";
            val rack=job[1].split(' ')[1];
            if(!validRack(rack))return "bad rack";
            return boardError(board(job.drop(3)),rack)
        };
        if(kind=="VALIDATE")return boardError(board(job.drop(2).takeWhile {
            !it.startsWith("MOVE")
        }),"");
        if(!validNumber(fields[2],1,2147483646))return "bad seed";
        if(job.size-1 !in 2..4)return "bad player count ${job.size-1}";
        val names=mutableSetOf<String>();
        for(i in 1 until job.size) {
            val player=job[i].split(' ');
            if(!validName(player[1])||player[2] !in strategies||kind=="GAME"&&!validNumber(player[3],
                0,99))return "bad player $i";
            if(!names.add(player[1]))return "duplicate player $i"
        };
        return null
    }
    fun leaveValue(rack: String): Int {
        val counts=IntArray(128);
        var value=0;
        var vowels=0;
        var consonants=0;
        for(c in rack) {
            counts[c.code]++;
            value+=weights[c.code];
            if(c in VOWELS)vowels++ else if(c!='?')consonants++
        };
        var duplicates=0;
        for(c in MoveGenerator.LETTERS)duplicates+=maxOf(0,counts[c.code]-1);
        return value-4*duplicates-3*maxOf(0,abs(vowels-consonants)-1)
    }
    fun evaluate(board: Array<CharArray>,rack: String,move: Board.Move): Evaluated {
        val score=Scoring.evaluate(board,move,values);
        val leave=removeTiles(rack,score.tiles.map {
            tileKind(it.letter)
        }.joinToString(""));
        val words=score.words.joinToString(",") {
            "${it.word.word}:${it.score}"
        };
        return Evaluated(move,score.score,score.tiles,words,leave,leaveValue(leave))
    }
    fun legal(board: Array<CharArray>,rack: String)=MoveGenerator.generate(board,rack,trie).map {
        evaluate(board,rack,it)
    }.sortedWith(standard)
    fun duplicate(job: List<String>): List<String> {
        val header=job.first().split(' ');
        val name=header[1];
        val seed=header[2].toInt();
        val random=Game.Minstd(seed);
        val bag=Game.bag(counts,random);
        var board=Board.empty();
        var rack=Game.draw(bag,7);
        val players=mutableListOf<Player>();
        val lines=mutableListOf("DUPLICATE $name seed=$seed players=${job.size-1}");
        for(i in 1 until job.size) {
            val fields=job[i].split(' ');
            players.add(Player(fields[1],fields[2]));
            lines.add("player $i ${fields[1]} strategy=${fields[2]}")
        };
        var rounds=0;
        var masterTotal=0;
        var reason="stuck";
        while(true) {
            val legal=legal(board,rack);
            if(legal.isEmpty())break;
            rounds++;
            val master=legal.first();
            val picks=players.map {
                player->val move=legal.minWith(strategy(player.strategy));
                player.score+=move.score;
                move.score
            };
            masterTotal+=master.score;
            lines.add("round $rounds rack=${collection(rack)} moves=${legal.size} master ${notation(master)} score=${master.score} picks=${picks.joinToString(",")}");
            board=Scoring.apply(board,master.tiles);
            rack=master.leave+Game.draw(bag,7-master.leave.length);
            if(rack.isEmpty()) {
                reason="out";
                break
            }
        };
        lines.add("end reason=$reason rounds=$rounds"+if(reason=="stuck")" rack=${collection(rack)}" else "");
        lines.add("master total=$masterTotal");
        for(player in players) {
            val percent=if(masterTotal>=1)100*player.score/masterTotal else 0;
            lines.add("final ${player.name} total=${player.score} percent=$percent")
        };
        val maximum=players.maxOf {
            it.score
        };
        lines.add("winner "+players.filter {
            it.score==maximum
        }.joinToString(",") {
            it.name
        });
        lines.add(positionSummary(board));
        lines.addAll(Board.display(board));
        add(gameTotals,"rounds",rounds);
        return lines
    }
    fun run() {
        for(line in File("input/tiles.txt").readLines())if(line.isNotEmpty()&&!line.startsWith('#')) {
            val fields=line.split(' ');
            val tile=fields[0][0].code;
            values[tile]=fields[1].toInt();
            counts[tile]=fields[2].toInt();
            weights[tile]=fields[3].toInt()
        };
        var candidates=0;
        var skipped=0;
        var duplicates=0;
        val lengths=IntArray(16);
        for(line in File("input/dictionary.txt").readLines()) {
            if(line.isEmpty()||line.startsWith('#'))continue;
            candidates++;
            if(!line.matches(Regex("[A-Z]{2,15}")))skipped++ else if(trie.add(line))lengths[line.length]++ else duplicates++
        };
        val histogram="lengths"+(2..15).filter {
            lengths[it]>0
        }.joinToString("") {
            " $it=${lengths[it]}"
        };
        val dictionary=listOf("DICTIONARY lines=$candidates words=${trie.words} skipped=$skipped duplicates=$duplicates nodes=${trie.nodes}",
            histogram);
        val jobs=mutableListOf<List<String>>();
        var current: MutableList<String>?=null;
        for(line in File("input/jobs.txt").readLines()) {
            val block=current;
            if(block==null) {
                if(line.isNotEmpty()&&!line.startsWith('#'))current=mutableListOf(line)
            } else if(line=="END") {
                jobs.add(block);
                current=null
            } else block.add(line)
        };
        val moves=mutableListOf(dictionary);
        val validation=mutableListOf<List<String>>();
        val games=mutableListOf<List<String>>();
        val seen=mutableSetOf<String>();
        for(job in jobs) {
            val header=job.first().split(' ');
            val kind=header[0];
            val name=header[1];
            val destination=when(kind) {
                "TOPN"->moves;
                "VALIDATE"->validation;
                else->games
            };
            val totals=when(kind) {
                "TOPN"->topTotals;
                "VALIDATE"->validationTotals;
                else->gameTotals
            };
            add(totals,when(kind) {
                "GAME"->"games";
                "DUPLICATE"->"duplicates";
                else->"jobs"
            },1);
            val error=jobError(job,seen);
            seen.add(name);
            if(error!=null) {
                destination.add(listOf("$kind $name","error $error"));
                add(totals,"errors",1)
            } else destination.add(when(kind) {
                "TOPN"->topn(job);
                "VALIDATE"->validate(job);
                "GAME"->ordinary(job);
                else->duplicate(job)
            })
        };
        moves.add(listOf(summary(topTotals,"jobs","errors","moves","bingos")));
        validation.add(listOf(summary(validationTotals,"jobs","errors","moves","valid","invalid","loss")));
        games.add(listOf(summary(gameTotals,"games","duplicates","errors","turns","rounds","plays","exchanges",
            "passes","bingos")));
        File("output").mkdirs();
        write("moves.txt",moves);
        write("validation.txt",validation);
        write("game.txt",games)
    }
