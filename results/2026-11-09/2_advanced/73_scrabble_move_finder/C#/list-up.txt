public sealed class DictionaryTrie {
    public sealed class Node {
        public Node[] children = new Node[26];
        public bool terminal;
    }
    public readonly Node root = new Node();
    public int nodes = 1, words;
    public bool Add(string word) {
        Node node = root;
        foreach (char letter in word.ToCharArray()) {
            int i = letter - 'A';
            if (node.children[i] == null) {
                node.children[i] = new Node();
                nodes++;
            }
            node = node.children[i];
        }
        if (node.terminal)
            return false;
        node.terminal = true;
        words++;
        return true;
    }
    public Node Prefix(string word) {
        Node node = root;
        foreach (char letter in word.ToCharArray()) {
            node = node.children[letter - 'A'];
            if (node == null)
                return null;
        }
        return node;
    }
    public bool Contains(string word) {
        Node node = Prefix(word);
        return node != null && node.terminal;
    }

public static class Game {
    public static readonly string TILE_ORDER = "ABCDEFGHIJKLMNOPQRSTUVWXYZ?";
    public sealed class Minstd {
        public long state;
        public Minstd(int seed) { state = seed; }
        public int Next() {
            state = state * 48271 % 2147483647;
            return (int)state;
        }
    }
    public static void Shuffle(List<char> bag, Minstd random) {
        for (int i = bag.Count - 1; i >= 1; i--) {
            int j = random.Next() % (i + 1);
            char tile = bag[i];
            bag[i] = bag[j];
            bag[j] = tile;
        }
    }
    public static List<char> Bag(int[] counts, Minstd random) {
        List<char> result = new();
        foreach (char tile in TILE_ORDER.ToCharArray())
            for (int i = 0; i < counts[tile]; i++)
                result.Add(tile);
        Shuffle(result, random);
        return result;
    }
    public static char RemoveLast(List<char> bag) {
        char value = bag[^1];
        bag.RemoveAt(bag.Count - 1);
        return value;
    }
    public static string Draw(List<char> bag, int count) {
        StringBuilder tiles = new StringBuilder();
        while (count-- > 0 && bag.Count != 0)
            tiles.Append(RemoveLast(bag));
        return tiles.ToString();
    }
public static class MoveGenerator {
    public static readonly string LETTERS = "ABCDEFGHIJKLMNOPQRSTUVWXYZ";
    public static bool[][] Anchors(char[][] board) {
        bool[][] result = Enumerable.Range(0, 15)
                              .Select(
                                  _ => new bool[15])
                              .ToArray();
        if (Board.Tiles(board).Count == 0) {
            result[7][7] = true;
            return result;
        }
        for (int r = 0; r < 15; r++)
            for (int c = 0; c < 15; c++)
                if (board[r][c] == '.')
                    foreach (Board.Cell neighbor in Board.Neighbors(r, c))
                        if (board[neighbor.row][neighbor.column] != '.')
                            result[r][c] = true;
        return result;
    }
    public static int[][] Checks(char[][] board, DictionaryTrie trie) {
        int[][] result = Enumerable.Range(0, 15)
                             .Select(
                                 _ => new int[15])
                             .ToArray();
        foreach (int[] row in result)
            Array.Fill(row, -1);
        for (int r = 0; r < 15; r++)
            for (int c = 0; c < 15; c++)
                if (board[r][c] == '.') {
                    int first = r, last = r;
                    while (first > 0 && board[first - 1][c] != '.')
                        first--;
                    while (last + 1 < 15 && board[last + 1][c] != '.')
                        last++;
                    if (first == last)
                        continue;
                    StringBuilder prefix = new StringBuilder(), suffix = new StringBuilder();
                    for (int i = first; i < r; i++)
                        prefix.Append(char.ToUpperInvariant(board[i][c]));
                    for (int i = r + 1; i <= last; i++)
                        suffix.Append(char.ToUpperInvariant(board[i][c]));
                    int mask = 0;
                    for (int i = 0; i < 26; i++)
                        if (trie.Contains(prefix.ToString() + (char)('A' + i) + suffix))
                            mask |= 1 << i;
                    result[r][c] = mask;
                }
        return result;
    }
    public sealed class Search {
        public char[][] board;
        public DictionaryTrie trie;
        public bool[][] anchors;
        public int[][] checks;
        public int[] counts = new int[128];
        public List<Board.Move> result = new();
        public int row, anchor;
        public Search(char[][] board, string rack, DictionaryTrie trie, bool[][] anchors,
                      int[][] checks) {
            this.board = board;
            this.trie = trie;
            this.anchors = anchors;
            this.checks = checks;
            foreach (char c in rack.ToCharArray())
                counts[c]++;
        }
        public void Extend(DictionaryTrie.Node node, int column, string word, int start) {
            bool empty = column == 15 || board[row][column] == '.';
            if (empty && column > anchor && node.terminal)
                result.Add(new Board.Move(row, start, 0, word));
            if (column == 15)
                return;
            if (!empty) {
                char letter = board[row][column];
                DictionaryTrie.Node child = node.children[char.ToUpperInvariant(letter) - 'A'];
                if (child != null)
                    Extend(child, column + 1, word + letter, start);
                return;
            }
            for (int i = 0; i < 26; i++)
                if (node.children[i] != null &&
                    (checks[row][column] == -1 || (checks[row][column] & (1 << i)) != 0)) {
                    char letter = (char)('A' + i);
                    for (int blank = 0; blank < 2; blank++) {
                        char tile = blank == 0 ? letter : '?',
                             written = blank == 0 ? letter : char.ToLowerInvariant(letter);
                        if (counts[tile] > 0) {
                            counts[tile]--;
                            Extend(node.children[i], column + 1, word + written, start);
                            counts[tile]++;
                        }
                    }
                }
        }
        public void Left(DictionaryTrie.Node node, string prefix, int remaining) {
            Extend(node, anchor, prefix, anchor - prefix.Length);
            if (remaining == 0)
                return;
            for (int i = 0; i < 26; i++)
                if (node.children[i] != null) {
                    char letter = (char)('A' + i);
                    for (int blank = 0; blank < 2; blank++) {
                        char tile = blank == 0 ? letter : '?',
                             written = blank == 0 ? letter : char.ToLowerInvariant(letter);
                        if (counts[tile] > 0) {
                            counts[tile]--;
                            Left(node.children[i], prefix + written, remaining - 1);
                            counts[tile]++;
                        }
                    }
                }
        }
        public List<Board.Move> Generate() {
            for (row = 0; row < 15; row++)
                for (anchor = 0; anchor < 15; anchor++)
                    if (anchors[row][anchor]) {
                        if (anchor > 0 && board[row][anchor - 1] != '.') {
                            int first = anchor - 1;
                            while (first > 0 && board[row][first - 1] != '.')
                                first--;
                            string prefix = new string(board[row], first, anchor - first);
                            DictionaryTrie.Node node = trie.Prefix(prefix.ToUpperInvariant());
                            if (node != null)
                                Extend(node, anchor, prefix, first);
                        } else {
                            int first = anchor - 1;
                            while (first >= 0 && board[row][first] == '.' && !anchors[row][first])
                                first--;
                            Left(trie.root, "", anchor - first - 1);
                        }
                    }
            return result;
        }
    }
    public static List<Board.Move> Generate(char[][] board, string rack, DictionaryTrie trie) {
        bool[][] anchors = Anchors(board);
        List<Board.Move> across =
            new Search(board, rack, trie, anchors, Checks(board, trie)).Generate();
        char[][] transposed = Board.Transpose(board);
        bool[][] downAnchors = Enumerable.Range(0, 15)
                                   .Select(
                                       _ => new bool[15])
                                   .ToArray();
        for (int r = 0; r < 15; r++)
            for (int c = 0; c < 15; c++)
                downAnchors[r][c] = anchors[c][r];
        List<Board.Move> down =
            new Search(transposed, rack, trie, downAnchors, Checks(transposed, trie)).Generate();
        foreach (Board.Move move in down)
            across.Add(new Board.Move(move.column, move.row, 1, move.word));
        Dictionary<string, Board.Move> distinct = new(StringComparer.Ordinal);
        foreach (Board.Move move in across) {
            List<Board.Tile> tiles = Scoring.Placement(board, move);
            if (tiles.Count == 1 && move.direction == 1) {
                Board.Tile tile = tiles[0];
                if (Board.Run(Scoring.Apply(board, tiles), tile.row, tile.column, 0).word.Length >=
                    2)
                    continue;
            }
            tiles.Sort((a, b) => {
                int c = a.row.CompareTo(b.row);
                return c != 0 ? c : a.column.CompareTo(b.column);
            });
            distinct[string.Join(";", tiles.Select(tile => tile.row + "," + tile.column + "," +
                                                           tile.letter))] = move;
        }
        return new(distinct.Values);
    }

public static class Scoring {
    public static readonly int RACK_SIZE = 7, BINGO_BONUS = 50;
    public record WordScore(Board.Move word, int score) {}
    public record Result(int score, List<Board.Tile> tiles, List<WordScore> words) {}
    public static List<Board.Tile> Squares(Board.Move move) {
        List<Board.Tile> result = new();
        for (int i = 0; i < move.word.Length; i++)
            result.Add(new Board.Tile(move.row + move.direction * i,
                                      move.column + (1 - move.direction) * i, move.word[i]));
        return result;
    }
    public static List<Board.Tile> Placement(char[][] board, Board.Move move) {
        List<Board.Tile> result = new();
        foreach (Board.Tile tile in Squares(move))
            if (board[tile.row][tile.column] == '.')
                result.Add(tile);
        return result;
    }
    public static char[][] Apply(char[][] board, List<Board.Tile> tiles) {
        char[][] result = new char [board.Length][];
        for (int r = 0; r < board.Length; r++)
            result[r] = board[r].ToArray();
        foreach (Board.Tile tile in tiles)
            result[tile.row][tile.column] = tile.letter;
        return result;
    }
    public static List<Board.Move> Crosses(char[][] board, Board.Move move) {
        List<Board.Tile> tiles = Placement(board, move);
        char[][] after = Apply(board, tiles);
        List<Board.Move> result = new();
        foreach (Board.Tile tile in tiles) {
            Board.Move word = Board.Run(after, tile.row, tile.column, 1 - move.direction);
            if (word.word.Length >= 2)
                result.Add(word);
        }
        return result;
    }
    public static int ComputeWordScore(char[][] board, Board.Move word, int[] values) {
        int score = 0, multiplier = 1;
        foreach (Board.Tile tile in Squares(word)) {
            char letter = tile.letter;
            int value = letter >= 'A' && letter <= 'Z' ? values[letter] : 0;
            if (board[tile.row][tile.column] == '.') {
                char premium = Board.PREMIUMS[tile.row][tile.column];
                value *= premium == '+' ? 2 : premium == '^' ? 3 : 1;
                multiplier *= premium == '*' ? 2 : premium == '#' ? 3 : 1;
            }
            score += value;
        }
        return score * multiplier;
    }
    public static Result Evaluate(char[][] board, Board.Move move, int[] values) {
        List<Board.Move> words = new();
        words.Add(move);
        words.AddRange(Crosses(board, move));
        List<WordScore> scores = new();
        int score = 0;
        foreach (Board.Move word in words) {
            int value = ComputeWordScore(board, word, values);
            score += value;
            scores.Add(new WordScore(word, value));
        }
        List<Board.Tile> tiles = Placement(board, move);
        if (tiles.Count == RACK_SIZE)
            score += BINGO_BONUS;
        return new Result(score, tiles, scores);
    }

    public static readonly int SIZE = 15;
    public static readonly string COLUMNS = "ABCDEFGHIJKLMNO";
    public static readonly string[] PREMIUMS = {
        "#..+...#...+..#", ".*...^...^...*.", "..*...+.+...*..", "+..*...+...*..+",
        "....*.....*....", ".^...^...^...^.", "..+...+.+...+..", "#..+...*...+..#",
        "..+...+.+...+..", ".^...^...^...^.", "....*.....*....", "+..*...+...*..+",
        "..*...+.+...*..", ".*...^...^...*.", "#..+...#...+..#"
    };
    public record Move(int row, int column, int direction, string word) {}
    public record Tile(int row, int column, char letter) {}
    public record Cell(int row, int column) {}
    public static bool Inside(int r, int c) { return 0 <= r && r < SIZE && 0 <= c && c < SIZE; }
    public static string Square(int r, int c) { return "" + COLUMNS[c] + (r + 1); }
    public static string Position(Move move) {
        return move.direction == 0 ? "" + (move.row + 1) + COLUMNS[move.column]
                                   : "" + COLUMNS[move.column] + (move.row + 1);
    }
    public static char[][] Empty() {
        char[][] board = Enumerable.Range(0, SIZE)
                             .Select(
                                 _ => new char[SIZE])
                             .ToArray();
        foreach (char[] row in board)
            Array.Fill(row, '.');
        return board;
    }
    public static char[][] Transpose(char[][] board) {
        char[][] result = Enumerable.Range(0, SIZE)
                              .Select(
                                  _ => new char[SIZE])
                              .ToArray();
        for (int r = 0; r < SIZE; r++)
            for (int c = 0; c < SIZE; c++)
                result[r][c] = board[c][r];
        return result;
    }
    public static List<Tile> TransposeTiles(List<Tile> tiles) {
        List<Tile> result = new();
        foreach (Tile tile in tiles)
            result.Add(new Tile(tile.column, tile.row, tile.letter));
        return result;
    }
    public static List<Tile> Tiles(char[][] board) {
        List<Tile> result = new();
        for (int r = 0; r < SIZE; r++)
            for (int c = 0; c < SIZE; c++)
                if (board[r][c] != '.')
                    result.Add(new Tile(r, c, board[r][c]));
        return result;
    }
    public static List<Cell> Neighbors(int r, int c) {
        List<Cell> result = new();
        foreach ((int dr, int dc) in new[] { (-1, 0), (1, 0), (0, -1), (0, 1) })
            if (Inside(r + dr, c + dc))
                result.Add(new(r + dr, c + dc));
        return result;
    }
    public static Move Run(char[][] board, int r, int c, int direction) {
        int dr = direction, dc = 1 - direction;
        while (Inside(r - dr, c - dc) && board[r - dr][c - dc] != '.') {
            r -= dr;
            c -= dc;
        }
        int firstRow = r, firstColumn = c;
        StringBuilder word = new StringBuilder();
        while (Inside(r, c) && board[r][c] != '.') {
            word.Append(board[r][c]);
            r += dr;
            c += dc;
        }
        return new Move(firstRow, firstColumn, direction, word.ToString());
    }
    public static readonly string VOWELS = "AEIOU";
    public static readonly HashSet<string> STRATEGIES =
        new HashSet<string> { "score", "tiles", "equity" };
    public record Evaluated(Board.Move move, int score, List<Board.Tile> tiles, string words,
                            string leave, int value) {}
    public static readonly Comparison<Evaluated> STANDARD = (a, b) => {
        int c = b.score.CompareTo(a.score);
        if (c == 0)
            c = a.move.row.CompareTo(b.move.row);
        if (c == 0)
            c = a.move.column.CompareTo(b.move.column);
        if (c == 0)
            c = a.move.direction.CompareTo(b.move.direction);
        return c != 0 ? c : string.CompareOrdinal(a.move.word, b.move.word);
    };
    public static bool ValidName(string text) { return text.Matches("[a-z0-9_]{1,20}"); }
    public static bool ValidNumber(string text, int low, int high) {
        if (!text.Matches("0|[1-9][0-9]*") || text.Length > 10)
            return false;
        long value = long.Parse(text);
        return low <= value && value <= high;
    }
    public static string Collection(string text) {
        char[] tiles = text.ToCharArray();
        tiles.AsSpan().Sort();
        return tiles.Length == 0 ? "-" : new string(tiles);
    }
    public static char TileKind(char letter) {
        return letter >= 'A' && letter <= 'Z' ? letter : '?';
    }
    public static string RemoveTiles(string rack, string removed) {
        int[] remaining = new int[128];
        foreach (char c in rack.ToCharArray())
            remaining[c]++;
        foreach (char c in removed.ToCharArray())
            remaining[c]--;
        StringBuilder result = new StringBuilder();
        for (int c = 0; c < 128; c++)
            for (int i = 0; i < remaining[c]; i++)
                result.Append((char)c);
        return result.ToString();
    }
    public static string Notation(Evaluated move) {
        return Board.Position(move.move) + " " + move.move.word;
    }
    public static Comparison<Evaluated> Strategy(string name) => (a, b) => {
        int c = name == "tiles"    ? b.tiles.Count.CompareTo(a.tiles.Count)
                : name == "equity" ? (b.score + b.value).CompareTo(a.score + a.value)
                                   : 0;
        return c != 0 ? c : STANDARD(a, b);
    };
    public static string JobError(List<string> job, HashSet<string> seen) {
        string[] fields = job[0].Split(" ");
        string kind = fields[0], name = fields[1];
        if (!ValidName(name))
            return "bad name";
        if (seen.Contains(name))
            return "duplicate name";
        if (kind.Equals("TOPN")) {
            if (!ValidNumber(fields[2], 1, 99))
                return "bad count";
            if (!STRATEGIES.Contains(fields[3]))
                return "bad order";
            string rack = job[1].Split(" ")[1];
            if (!ValidRack(rack))
                return "bad rack";
            return BoardError(ParseBoard(job.GetRange(3, job.Count - 3)), rack);
        }
        if (kind.Equals("VALIDATE")) {
            List<string> rows = new();
            for (int i = 2; i < job.Count && !job[i].StartsWith("MOVE"); i++)
                rows.Add(job[i]);
            return BoardError(ParseBoard(rows), "");
        }
        if (!ValidNumber(fields[2], 1, 2147483646))
            return "bad seed";
        if (job.Count - 1 < 2 || job.Count - 1 > 4)
            return "bad player count " + (job.Count - 1);
        HashSet<string> names = new();
        for (int i = 1; i < job.Count; i++) {
            string[] player = job[i].Split(" ");
            if (!ValidName(player[1]) || !STRATEGIES.Contains(player[2]) ||
                kind.Equals("GAME") && !ValidNumber(player[3], 0, 99))
                return "bad player " + i;
            if (!names.Add(player[1]))
                return "duplicate player " + i;
        }
        return null;
    }
    public static int LeaveValue(string rack) {
        int[] counts = new int[128];
        int value = 0, duplicates = 0, vowels = 0, consonants = 0;
        foreach (char c in rack.ToCharArray()) {
            counts[c]++;
            value += weights[c];
            if (VOWELS.IndexOf(c) >= 0)
                vowels++;
            else if (c != '?')
                consonants++;
        }
        foreach (char c in MoveGenerator.LETTERS.ToCharArray())
            duplicates += Math.Max(0, counts[c] - 1);
        return value - 4 * duplicates - 3 * Math.Max(0, Math.Abs(vowels - consonants) - 1);
    }
    public static Evaluated Evaluate(char[][] board, string rack, Board.Move move) {
        Scoring.Result score = Scoring.Evaluate(board, move, values);
        StringBuilder removed = new StringBuilder();
        foreach (Board.Tile tile in score.tiles)
            removed.Append(TileKind(tile.letter));
        string leave = RemoveTiles(rack, removed.ToString());
        List<string> words = new();
        foreach (Scoring.WordScore word in score.words)
            words.Add(word.word.word + ":" + word.score);
        return new Evaluated(move, score.score, score.tiles, string.Join(",", words), leave,
                             LeaveValue(leave));
    }
    public static List<Evaluated> Legal(char[][] board, string rack) {
        List<Evaluated> result = new();
        foreach (Board.Move move in MoveGenerator.Generate(board, rack, trie))
            result.Add(Evaluate(board, rack, move));
        result.Sort(STANDARD);
        return result;
    }
    public static List<string> Duplicate(List<string> job) {
        string[] header = job[0].Split(" ");
        string name = header[1];
        int seed = int.Parse(header[2]);
        Game.Minstd random = new Game.Minstd(seed);
        List<char> bag = Game.Bag(counts, random);
        char[][] board = Board.Empty();
        string rack = Game.Draw(bag, 7);
        List<Player> players = new();
        List<string> lines = new();
        lines.Add("DUPLICATE " + name + " seed=" + seed + " players=" + (job.Count - 1));
        for (int i = 1; i < job.Count; i++) {
            string[] fields = job[i].Split(" ");
            players.Add(new Player(fields[1], fields[2], 0, ""));
            lines.Add("player " + i + " " + fields[1] + " strategy=" + fields[2]);
        }
        int rounds = 0, masterTotal = 0;
        string reason = "stuck";
        while (true) {
            List<Evaluated> legal = Legal(board, rack);
            if (legal.Count == 0)
                break;
            rounds++;
            Evaluated master = legal[0];
            List<string> picks = new();
            foreach (Player player in players) {
                Evaluated move = legal
                                     .OrderBy(move => move,
                                              Comparer<Evaluated>.Create(Strategy(player.strategy)))
                                     .First();
                player.score += move.score;
                picks.Add(Convert.ToString(move.score));
            }
            masterTotal += master.score;
            lines.Add("round " + rounds + " rack=" + Collection(rack) + " moves=" + legal.Count +
                      " master " + Notation(master) + " score=" + master.score +
                      " picks=" + string.Join(",", picks));
            board = Scoring.Apply(board, master.tiles);
            rack = master.leave + Game.Draw(bag, 7 - master.leave.Length);
            if (rack.Length == 0) {
                reason = "out";
                break;
            }
        }
        lines.Add("end reason=" + reason + " rounds=" + rounds +
                  (reason.Equals("stuck") ? " rack=" + Collection(rack) : ""));
        lines.Add("master total=" + masterTotal);
        int maximum = int.MinValue;
        foreach (Player player in players) {
            int percent = masterTotal >= 1 ? 100 * player.score / masterTotal : 0;
            lines.Add("final " + player.name + " total=" + player.score + " percent=" + percent);
            maximum = Math.Max(maximum, player.score);
        }
        List<string> winners = new();
        foreach (Player player in players)
            if (player.score == maximum)
                winners.Add(player.name);
        lines.Add("winner " + string.Join(",", winners));
        lines.Add(PositionSummary(board));
        lines.AddRange(Board.Display(board));
        Add(gameTotals, "rounds", rounds);
        return lines;
    }

                skipped++;

                lengths[line.Length]++;
            else
                duplicates++;
        }
        StringBuilder histogram = new StringBuilder("lengths");
        for (int i = 2; i <= 15; i++)
            if (lengths[i] > 0)
                histogram.Append(" ").Append(i).Append("=").Append(lengths[i]);
        List<string> dictionary =
            new List<string> { "DICTIONARY lines=" + candidates + " words=" + trie.words +
                                   " skipped=" + skipped + " duplicates=" + duplicates +
                                   " nodes=" + trie.nodes,
                               histogram.ToString() };
        List<List<string>> jobs = new();
        List<string> current = null;
        foreach (string line in File.ReadAllLines(Path.Combine("input", "jobs.txt")).ToList()) {
            if (current == null) {
                if (line.Length != 0 && !line.StartsWith("#")) {
                    current = new();
                    current.Add(line);
                }
            } else if (line.Equals("END")) {
                jobs.Add(current);
                current = null;
            } else
                current.Add(line);
        }
        List<List<string>> moves = new(), validation = new(), games = new();
        moves.Add(dictionary);
        HashSet<string> seen = new();
        foreach (List<string> job in jobs) {
            string[] header = job[0].Split(" ");
            string kind = header[0], name = header[1];
            List<List<string>> destination = kind.Equals("TOPN")       ? moves
                                             : kind.Equals("VALIDATE") ? validation
                                                                       : games;
            Dictionary<string, int> totals = kind.Equals("TOPN")       ? topTotals
                                             : kind.Equals("VALIDATE") ? validationTotals
                                                                       : gameTotals;
            Add(totals,
                kind.Equals("GAME")        ? "games"
                : kind.Equals("DUPLICATE") ? "duplicates"
                                           : "jobs",
                1);
            string error = JobError(job, seen);
            seen.Add(name);
            if (error != null) {
                destination.Add(new List<string> { kind + " " + name, "error " + error });
                Add(totals, "errors", 1);
            } else
                destination.Add(kind switch { "TOPN" => Topn(job), "VALIDATE" => Validate(job),
                                              "GAME" => Ordinary(job),
                                              _ => Duplicate(job) });
        }
        moves.Add(new List<string> { Summary(topTotals, "jobs", "errors", "moves", "bingos") });
        validation.Add(new List<string> { Summary(validationTotals, "jobs", "errors", "moves",
                                                  "valid", "invalid", "loss") });
        games.Add(new List<string> { Summary(gameTotals, "games", "duplicates", "errors", "turns",
                                             "rounds", "plays", "exchanges", "passes", "bingos") });
        Directory.CreateDirectory(Path.Combine("output"));
        Write("moves.txt", moves);
        Write("validation.txt", validation);
        Write("game.txt", games);
    }
