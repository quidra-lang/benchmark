'use strict';

class Trie {
    constructor() {
        this.root = { children: new Map(), terminal: false };
        this.nodes = 1;
        this.words = 0;
    }

    add(word) {
        let node = this.root;
        for (const letter of word) {
            if (!node.children.has(letter)) {
                node.children.set(letter, { children: new Map(), terminal: false });
                this.nodes++;
            }
            node = node.children.get(letter);
        }
        if (node.terminal) return false;
        node.terminal = true;
        this.words++;
        return true;
    }

    find(prefix) {
        let node = this.root;
        for (const letter of prefix) {
            node = node.children.get(letter);
            if (!node) return null;
        }
        return node;
    }

    has(word) {
        const node = this.find(word);
        return node !== null && node.terminal;
    }

class Minstd {
    constructor(seed) { this.state = seed; }
    next() {
        this.state = (48271 * this.state) % 2147483647;
        return this.state;
    }
}

function shuffle(bag, random) {
    for (let i = bag.length - 1; i >= 1; i--) {
        const j = random.next() % (i + 1);
        [bag[i], bag[j]] = [bag[j], bag[i]];
    }
}

function makeBag(table, random) {
    const bag = [];
    for (const [letter, tile] of table) for (let i = 0; i < tile.count; i++) bag.push(letter);
    shuffle(bag, random);
    return bag;
}

function draw(bag, count) {
    let tiles = '';
    for (let i = 0; i < count && bag.length > 0; i++) tiles += bag.pop();
    return tiles;
}

function anchors(grid) {
    const result = new Set();
    if (board.tiles(grid).length === 0) {
        result.add(board.CENTER * board.SIZE + board.CENTER);
        return result;
    }
    for (let row = 0; row < board.SIZE; row++) for (let column = 0; column < board.SIZE; column++) {
        if (grid[row][column] === '.' && board.adjacent(grid, row, column)) result.add(row * board.SIZE + column);
    }
    return result;
}

function crossChecks(grid, trie) {
    const allowed = Array.from({ length: board.SIZE }, () => Array(board.SIZE).fill(null));
    let checked = 0;
    let dead = 0;
    let total = 0;
    for (let row = 0; row < board.SIZE; row++) for (let column = 0; column < board.SIZE; column++) {
        if (grid[row][column] !== '.') continue;
        const letters = new Set();
        if (!board.occupied(grid, row - 1, column) && !board.occupied(grid, row + 1, column)) {
            for (const letter of ALPHABET) letters.add(letter);
        } else {
            checked++;
            for (const letter of ALPHABET) {
                const word = crossWord(grid, { row, column, letter }, 'down').word.toUpperCase();
                if (trie.has(word)) letters.add(letter);
            }
            if (letters.size === 0) dead++;
            total += letters.size;
        }
        allowed[row][column] = letters;
    }
    return { allowed, stats: [checked, dead, total] };
}

function acrossMoves(grid, rack, trie, anchorSet, checks) {
    const result = [];
    const counts = new Map();
    for (const tile of rack) counts.set(tile, (counts.get(tile) || 0) + 1);
    for (const anchor of anchorSet) {
        const row = Math.floor(anchor / board.SIZE);
        const anchorColumn = anchor % board.SIZE;
        function choices(node, allowed, use) {
            for (const letter of Array.from(node.children.keys()).sort()) {
                if (allowed !== null && !allowed.has(letter)) continue;
                for (const tile of [letter, '?']) {
                    if ((counts.get(tile) || 0) === 0) continue;
                    counts.set(tile, counts.get(tile) - 1);
                    use(node.children.get(letter), tile === '?' ? letter.toLowerCase() : letter);
                    counts.set(tile, counts.get(tile) + 1);
                }
            }
        }
        function extend(column, node, word, start, filled) {
            const occupied = column < board.SIZE && grid[row][column] !== '.';
            if (!occupied && filled && node.terminal) {
                const tiles = [];
                for (let i = 0; i < word.length; i++) {
                    if (grid[row][start + i] === '.') tiles.push({ row, column: start + i, letter: word[i] });
                }
                result.push({ row, column: start, direction: 'across', word, tiles });
            }
            if (column >= board.SIZE) return;
            if (occupied) {
                const letter = grid[row][column];
                const child = node.children.get(letter.toUpperCase());
                if (child) extend(column + 1, child, word + letter, start, filled);
            } else {
                choices(node, checks.allowed[row][column], (child, letter) =>
                    extend(column + 1, child, word + letter, start, filled || column === anchorColumn));
            }
        }
        if (board.occupied(grid, row, anchorColumn - 1)) {
            let start = anchorColumn - 1;
            while (board.occupied(grid, row, start - 1)) start--;
            const prefix = grid[row].slice(start, anchorColumn).join('');
            const node = trie.find(prefix.toUpperCase());
            if (node) extend(anchorColumn, node, prefix, start, false);
        } else {
            let limit = 0;
            while (anchorColumn - limit - 1 >= 0 && grid[row][anchorColumn - limit - 1] === '.' &&
                !anchorSet.has(row * board.SIZE + anchorColumn - limit - 1)) limit++;
            function leftPart(node, word) {
                extend(anchorColumn, node, word, anchorColumn - word.length, false);
                if (word.length < limit) choices(node, null, (child, letter) => leftPart(child, word + letter));
            }
            leftPart(trie.root, '');
        }
    }
    return result;
}

function generate(grid, rack, trie) {
    const anchorSet = anchors(grid);
    const across = crossChecks(grid, trie);
    const transposed = board.transpose(grid);
    const down = crossChecks(transposed, trie);
    const moves = acrossMoves(grid, rack, trie, anchorSet, across);
    for (const move of acrossMoves(transposed, rack, trie, anchors(transposed), down)) {
        const tiles = board.transposeTiles(move.tiles);
        if (tiles.length === 1 && (board.occupied(grid, tiles[0].row, tiles[0].column - 1) ||
            board.occupied(grid, tiles[0].row, tiles[0].column + 1))) continue;
        moves.push({ ...move, row: move.column, column: move.row, direction: 'down', tiles });
    }
    return { moves, anchors: anchorSet.size, across: across.stats, down: down.stats };
}

const board = require('./board');
const RACK_SIZE = 7;
const BINGO = 50;
const tileKind = letter => letter >= 'a' && letter <= 'z' ? '?' : letter;

function squares(move) {
    return Array.from(move.word, (letter, i) => ({
        row: move.row + (move.direction === 'down' ? i : 0),
        column: move.column + (move.direction === 'across' ? i : 0), letter,
    }));
}

function crossWord(grid, tile, direction) {
    const dr = direction === 'down' ? 1 : 0;
    const dc = direction === 'across' ? 1 : 0;
    let row = tile.row;
    let column = tile.column;
    while (board.occupied(grid, row - dr, column - dc)) { row -= dr; column -= dc; }
    const start = { row, column, direction };
    let word = '';
    while (board.inside(row, column) && ((row === tile.row && column === tile.column) || board.occupied(grid, row, column))) {
        word += row === tile.row && column === tile.column ? tile.letter : grid[row][column];
        row += dr;
        column += dc;
    }
    return { ...start, word };
}

function wordScore(grid, word, table) {
    let total = 0;
    let multiplier = 1;
    for (const tile of squares(word)) {
        let points = table.get(tileKind(tile.letter)).value;
        if (grid[tile.row][tile.column] === '.') {
            const premium = board.PREMIUMS[tile.row][tile.column];
            if (premium === '+') points *= 2;
            if (premium === '^') points *= 3;
            if (premium === '*') multiplier *= 2;
            if (premium === '#') multiplier *= 3;
        }
        total += points;
    }
    return total * multiplier;
}

function score(grid, move, table) {
    const tiles = squares(move).filter(tile => grid[tile.row][tile.column] === '.');
    const words = [{ row: move.row, column: move.column, direction: move.direction, word: move.word }];
    const perpendicular = move.direction === 'across' ? 'down' : 'across';
    for (const tile of tiles) {
        const cross = crossWord(grid, tile, perpendicular);
        if (cross.word.length >= 2) words.push(cross);
    }
    let total = tiles.length === RACK_SIZE ? BINGO : 0;
    for (const word of words) {
        word.score = wordScore(grid, word, table);
        total += word.score;
    }
    return { ...move, tiles, words, score: total };
}

function tiles(grid) {
    const result = [];
    for (let row = 0; row < SIZE; row++) for (let column = 0; column < SIZE; column++) {
        if (grid[row][column] !== '.') result.push({ row, column, letter: grid[row][column] });
    }
    return result;
}

function adjacent(grid, row, column) {
    return [[-1, 0], [1, 0], [0, -1], [0, 1]].some(([dr, dc]) => occupied(grid, row + dr, column + dc));
}

function connected(grid) {
    const placed = tiles(grid);
    if (placed.length === 0) return true;
    const queue = [placed[0]];
    const seen = new Set([placed[0].row * SIZE + placed[0].column]);
    for (let i = 0; i < queue.length; i++) {
        const { row, column } = queue[i];
        for (const [dr, dc] of [[-1, 0], [1, 0], [0, -1], [0, 1]]) {
            const r = row + dr;
            const c = column + dc;
            const key = r * SIZE + c;
            if (occupied(grid, r, c) && !seen.has(key)) {
                seen.add(key);
                queue.push({ row: r, column: c });
            }
        }
    }
    return seen.size === placed.length;
}

function readInput() {
    const table = new Map();
    for (const line of readLines('input/tiles.txt')) {
        if (line === '' || line.startsWith('#')) continue;
        const [letter, value, count, weight] = line.split(' ');
        table.set(letter, { value: Number(value), count: Number(count), weight: Number(weight) });
    }
    const trie = new Trie();
    const stats = { lines: 0, skipped: 0, duplicates: 0, lengths: new Map() };
    for (const line of readLines('input/dictionary.txt')) {
        if (line === '' || line.startsWith('#')) continue;
        stats.lines++;
        if (!/^[A-Z]{2,15}$/.test(line)) stats.skipped++;
        else if (!trie.add(line)) stats.duplicates++;
        else stats.lengths.set(line.length, (stats.lengths.get(line.length) || 0) + 1);
    }
    const lines = readLines('input/jobs.txt');
    const jobs = [];
    for (let i = 0; i < lines.length; i++) {
        if (lines[i] === '' || lines[i].startsWith('#')) continue;
        const [kind, name, ...args] = lines[i].split(' ');
        const body = [];
        while (lines[++i] !== 'END') body.push(lines[i]);
        jobs.push({ kind, name, args, body });
    }
    return { table, trie, stats, jobs };
}
function checkJob(job, duplicate, table, trie) {
    if (!nameValid(job.name)) return { error: 'bad name' };
    if (duplicate) return { error: 'duplicate name' };
    if (job.kind === 'TOPN') {
        if (!numberValid(job.args[0], 1, 99)) return { error: 'bad count' };
        if (!STRATEGIES.includes(job.args[1])) return { error: 'bad order' };
        const rack = job.body[0].split(' ')[1];
        if (!rackValid(rack)) return { error: 'bad rack' };
        return { ...checkBoard(job.body.slice(2), rack, table, trie), rack,
            count: Number(job.args[0]), strategy: job.args[1] };
    }
    if (job.kind === 'VALIDATE') {
        let at = 1;
        while (at < job.body.length && !job.body[at].startsWith('MOVE')) at++;
        return { ...checkBoard(job.body.slice(1, at), '', table, trie), records: job.body.slice(at) };
    }
    if (!numberValid(job.args[0], 1, 2147483646)) return { error: 'bad seed' };
    if (job.body.length < 2 || job.body.length > 4) return { error: `bad player count ${job.body.length}` };
    const players = [];
    const names = new Set();
    for (let i = 0; i < job.body.length; i++) {
        const [, name, strategy, threshold] = job.body[i].split(' ');
        if (!nameValid(name) || !STRATEGIES.includes(strategy) ||
            (job.kind === 'GAME' && !numberValid(threshold, 0, 99))) return { error: `bad player ${i + 1}` };
        if (names.has(name)) return { error: `duplicate player ${i + 1}` };
        names.add(name);
        players.push({ name, strategy, threshold: job.kind === 'GAME' ? Number(threshold) : 0, index: i });
    }
    return { players, seed: Number(job.args[0]) };
}
function leaveValue(tiles, table) {
    const counts = frequencies(tiles);
    let value = 0;
    let duplicates = 0;
    let vowels = 0;
    let consonants = 0;
    for (const tile of tiles) {
        value += table.get(tile).weight;
        if (VOWELS.includes(tile)) vowels++;
        else if (tile !== '?') consonants++;
    }
    for (const [tile, count] of counts) if (tile !== '?') duplicates += count - 1;
    return value - 4 * duplicates - 3 * Math.max(0, Math.abs(vowels - consonants) - 1);
}
function removeTiles(rack, used) {
    const remaining = Array.from(rack);
    for (const tile of used) remaining.splice(remaining.indexOf(tile), 1);
    return remaining.join('');
}
function standardOrder(a, b) {
    return b.score - a.score || a.row - b.row || a.column - b.column ||
        (a.direction === b.direction ? 0 : a.direction === 'across' ? -1 : 1) || ascii(a.word, b.word);
}
function strategyOrder(moves, strategy) {
    return moves.slice().sort((a, b) => (strategy === 'tiles' ? b.tiles.length - a.tiles.length :
        strategy === 'equity' ? (b.score + b.leaveValue) - (a.score + a.leaveValue) : 0) || standardOrder(a, b));
}

}
function duplicateJob(job, settings, trie, table) {
    const random = new game.Minstd(settings.seed);
    const bag = game.makeBag(table, random);
    const grid = board.empty();
    const players = settings.players.map(player => ({ ...player, total: 0 }));
    let rack = game.draw(bag, scoring.RACK_SIZE);
    let masterTotal = 0;
    let rounds = 0;
    let reason;
    const lines = [`DUPLICATE ${job.name} seed=${settings.seed} players=${players.length}`];
    players.forEach((player, i) => lines.push(`player ${i + 1} ${player.name} strategy=${player.strategy}`));
    while (true) {
        const moves = legalMoves(grid, rack, trie, table).moves;
        if (moves.length === 0) { reason = 'stuck'; break; }
        rounds++;
        const master = moves[0];
        const picks = [];
        for (const player of players) {
            const move = strategyOrder(moves, player.strategy)[0];
            player.total += move.score;
            picks.push(move.score);
        }
        masterTotal += master.score;
        lines.push(`round ${rounds} rack=${collection(rack)} moves=${moves.length} master ${notation(master)} score=${master.score} picks=${picks.join(',')}`);
        place(grid, master);
        rack = master.leave;
        rack += game.draw(bag, scoring.RACK_SIZE - rack.length);
        if (rack.length === 0) { reason = 'out'; break; }
    }
    lines.push(reason === 'out' ? `end reason=out rounds=${rounds}` : `end reason=stuck rounds=${rounds} rack=${collection(rack)}`,
        `master total=${masterTotal}`);
    for (const player of players) {
        const percent = masterTotal > 0 ? Math.floor(100 * player.total / masterTotal) : 0;
        lines.push(`final ${player.name} total=${player.total} percent=${percent}`);
    }
    const maximum = Math.max(...players.map(player => player.total));
    lines.push('winner ' + players.filter(player => player.total === maximum).map(player => player.name).join(','),
        positionSummary(grid), ...board.display(grid));
    return { lines, rounds };
}
function main() {
    const { table, trie, stats, jobs } = readInput();
    const lengths = Array.from(stats.lengths.keys()).sort((a, b) => a - b);
    const moves = [[`DICTIONARY lines=${stats.lines} words=${trie.words} skipped=${stats.skipped} duplicates=${stats.duplicates} nodes=${trie.nodes}`,
        'lengths ' + lengths.map(length => `${length}=${stats.lengths.get(length)}`).join(' ')]];
    const validation = [];
    const games = [];
    const topTotal = { jobs: 0, errors: 0, moves: 0, bingos: 0 };
    const validationTotal = { jobs: 0, errors: 0, moves: 0, valid: 0, invalid: 0, loss: 0 };
    const gameTotal = { games: 0, duplicates: 0, errors: 0, turns: 0, rounds: 0, plays: 0, exchanges: 0, passes: 0, bingos: 0 };
    const seen = new Set();
    for (const job of jobs) {
        const duplicate = seen.has(job.name);
        seen.add(job.name);
        const settings = checkJob(job, duplicate, table, trie);
        const paragraphs = job.kind === 'TOPN' ? moves : job.kind === 'VALIDATE' ? validation : games;
        const totals = job.kind === 'TOPN' ? topTotal : job.kind === 'VALIDATE' ? validationTotal : gameTotal;
        if (job.kind === 'GAME') gameTotal.games++;
        else if (job.kind === 'DUPLICATE') gameTotal.duplicates++;
        else totals.jobs++;
        if (settings.error) {
            totals.errors++;
            paragraphs.push([`${job.kind} ${job.name}`, 'error ' + settings.error]);
            continue;
        }
        if (job.kind === 'TOPN') {
            const result = topJob(job, settings, trie, table);
            moves.push(result.lines);
            topTotal.moves += result.moves;
            topTotal.bingos += result.bingos;
        } else if (job.kind === 'VALIDATE') {
            const result = validationJob(job, settings, trie, table);
            validation.push(result.lines);
            for (const key of ['moves', 'valid', 'invalid', 'loss']) validationTotal[key] += result[key];
        } else if (job.kind === 'GAME') {
            const result = gameJob(job, settings, trie, table);
            games.push(result.lines);
            for (const key of ['turns', 'plays', 'exchanges', 'passes', 'bingos']) gameTotal[key] += result[key];
        } else {
            const result = duplicateJob(job, settings, trie, table);
            games.push(result.lines);
            gameTotal.rounds += result.rounds;
        }
    }
    moves.push([`SUMMARY jobs=${topTotal.jobs} errors=${topTotal.errors} moves=${topTotal.moves} bingos=${topTotal.bingos}`]);
    validation.push([`SUMMARY jobs=${validationTotal.jobs} errors=${validationTotal.errors} moves=${validationTotal.moves} valid=${validationTotal.valid} invalid=${validationTotal.invalid} loss=${validationTotal.loss}`]);
    games.push([`SUMMARY games=${gameTotal.games} duplicates=${gameTotal.duplicates} errors=${gameTotal.errors} turns=${gameTotal.turns} rounds=${gameTotal.rounds} plays=${gameTotal.plays} exchanges=${gameTotal.exchanges} passes=${gameTotal.passes} bingos=${gameTotal.bingos}`]);
    fs.mkdirSync('output', { recursive: true });
    writeParagraphs('moves.txt', moves);
    writeParagraphs('validation.txt', validation);
    writeParagraphs('game.txt', games);
}
