class Trie()
    int[][] children := [[for i := (0:<26)[] then -1]]
    terminal := [false]
    word_count := 0

    bool .add(string word)
        node := 0
        for: i < int(#word)
            letter := board.character_index(board.letters, word[i])
            if children[node][letter] < 0
                created := int(#children)
                children[node][letter] = created
                children = children.append([for j := (0:<26)[] then -1])
                terminal = terminal.append(false)
            node = children[node][letter]
        if terminal[node]
            return false
        terminal[node] = true
        word_count += 1
        return true

    int prefix(string word)
        node := 0
        for: i < int(#word)
            letter := board.character_index(board.letters, word[i])
            if letter < 0 or children[node][letter] < 0
                return -1
            node = children[node][letter]
        return node

    bool has(string word)
        int node := prefix(word)
        return node >= 0 and terminal[node]

const tile_order := "ABCDEFGHIJKLMNOPQRSTUVWXYZ?"

int tile_index(string tile)
    return if tile == "?" then 26 else board.character_index(board.letters, tile)

int points(string collection, int[] values)
    result := 0
    for: i < int(#collection)
        result += values[tile_index(collection[i])]
    return result

class Bag(int state, int[] counts)
    string[] tiles := []
    for: i < 27
        for: j < counts[i]
            tiles = tiles.append(tile_order[i])

    int .next_random()
        state = (48271 * state) % 2147483647
        return state

    void .shuffle()
        i := int(#tiles) - 1
        while i >= 1
            int j := .next_random() % (i + 1)
            temporary := tiles[i]
            tiles[i] = tiles[j]
            tiles[j] = temporary
            i -= 1

    string .draw(int amount)
        drawn := ""
        for: i < amount
            if #tiles == 0
                break
            last := int(#tiles) - 1
            drawn += tiles[last]
            tiles = tiles[0:<last]
        return drawn

bool[][] anchors(string[][] grid)
    result := [for row := (0:<15)[] then [for column := (0:<15)[] then false]]
    if #board.tiles(grid) == 0
        result[7][7] = true
        return result
    for: row < 15
        for: column < 15
            if grid[row][column] == "."
                for point := board.neighbours(row, column)[]
                    if grid[point[0]][point[1]] != "."
                        result[row][column] = true
    return result

class Checks(bool[][] needed, bool[][][] allowed)

Checks cross_checks(string[][] grid, dictionary_trie.Trie trie)
    needed := [for row := (0:<15)[] then [for column := (0:<15)[] then false]]
    allowed := [for row := (0:<15)[] then [for column := (0:<15)[] then [for letter := (0:<26)[] then true]]]
    for: row < 15
        for: column < 15
            if grid[row][column] != "."
                continue
            first := row
            last := row
            while first > 0 and grid[first - 1][column] != "."
                first -= 1
            while last + 1 < 15 and grid[last + 1][column] != "."
                last += 1
            if first == last
                continue
            needed[row][column] = true
            prefix := [for r := (int(first):<int(row))[] then grid[r][column]].join("")
            suffix := [for r := (int(row + 1):<(int(last) + 1))[] then grid[r][column]].join("")
            for: letter < 26
                allowed[row][column][letter] = trie.has(board.uppercase(prefix) + board.letters[letter] + board.uppercase(suffix))
    return Checks(needed, allowed)

class Search(string[][] grid, string rack, dictionary_trie.Trie trie, bool[][] anchor_squares, Checks checks)
    int[] counts := [for i := (0:<27)[] then 0]
    for: i < int(#rack)
        tile := if rack[i] == "?" then 26 else board.character_index(board.letters, rack[i])
        counts[tile] += 1
    board.Move[] result := []

    void .extend_right(int row, int anchor, int node, int column, string word, int start, int[] remaining)
        empty := column == 15 or grid[row][column] == "."
        if empty and column > anchor and trie.terminal[node]
            result = result.append(board.Move(row, start, 0, word))
        if column == 15
            return
        if not empty
            character := grid[row][column]
            letter := board.character_index(board.letters, board.uppercase(character))
            child := trie.children[node][letter]
            if child >= 0
                .extend_right(row, anchor, child, column + 1, word + character, start, remaining)
            return
        for: letter < 26
            child := trie.children[node][letter]
            if child < 0 or not checks.allowed[row][column][letter]
                continue
            for: alternative < 2
                tile := if alternative == 0 then letter else 26
                if remaining[tile] > 0
                    next := remaining
                    next[tile] -= 1
                    string character := if alternative == 0 then board.letters[letter] else board.lowercase_letters[letter]
                    .extend_right(row, anchor, child, column + 1, word + character, start, next)

    void .left_part(int row, int anchor, int node, string prefix, int limit, int[] remaining)
        .extend_right(row, anchor, node, anchor, prefix, anchor - int(#prefix), remaining)
        if limit == 0
            return
        for: letter < 26
            child := trie.children[node][letter]
            if child < 0
                continue
            for: alternative < 2
                tile := if alternative == 0 then letter else 26
                if remaining[tile] > 0
                    next := remaining
                    next[tile] -= 1
                    string character := if alternative == 0 then board.letters[letter] else board.lowercase_letters[letter]
                    .left_part(row, anchor, child, prefix + character, limit - 1, next)

    void .across_moves()
        for: row < 15
            for: anchor < 15
                if not anchor_squares[row][anchor]
                    continue
                if anchor > 0 and grid[row][anchor - 1] != "."
                    first := anchor - 1
                    while first > 0 and grid[row][first - 1] != "."
                        first -= 1
                    prefix := grid[row][first:<anchor].join("")
                    node := trie.prefix(board.uppercase(prefix))
                    if node >= 0
                        .extend_right(row, anchor, node, anchor, prefix, first, counts)
                else
                    first := anchor - 1
                    while first >= 0 and grid[row][first] == "." and not anchor_squares[row][first]
                        first -= 1
                    .left_part(row, anchor, 0, "", anchor - first - 1, counts)

board.Move[] generate(string[][] grid, string rack, dictionary_trie.Trie trie)
    anchor_squares := anchors(grid)
    across := Search(grid, rack, trie, anchor_squares, cross_checks(grid, trie))
    .across.across_moves()
    transposed := board.transpose(grid)
    swapped := [for row := (0:<15)[] then [for column := (0:<15)[] then anchor_squares[column][row]]]
    down := Search(transposed, rack, trie, swapped, cross_checks(transposed, trie))
    .down.across_moves()
    result := across.result
    for candidate := down.result[]
        move := board.Move(candidate.column, candidate.row, 1, candidate.word)
        placed := scoring.placement(grid, move)
        if #placed == 1
            after := scoring.apply(grid, placed)
            if #board.run_at(after, placed[0].row, placed[0].column, 0).word >= 2
                continue
        result = result.append(move)
    return result
const rack_size := 7
const bingo_bonus := 50

board.Tile[] squares(board.Move move)
    return [for i := (0:<int(#move.word))[] then board.Tile(move.row + move.direction * i, move.column + (1 - move.direction) * i, move.word[i])]

board.Tile[] placement(string[][] grid, board.Move move)
    return [for tile := squares(move)[] if grid[tile.row][tile.column] == "." then tile]

string[][] apply(string[][] grid, board.Tile[] placed)
    result := grid
    for tile := placed[]
        result[tile.row][tile.column] = tile.letter
    return result

board.Move[] cross_words(string[][] grid, board.Move move)
    placed := placement(grid, move)
    after := apply(grid, placed)
    board.Move[] result := []
    for tile := placed[]
        word := board.run_at(after, tile.row, tile.column, 1 - move.direction)
        if #word.word >= 2
            result = result.append(word)
    return result

int word_score(string[][] grid, board.Move word, int[] values)
    score := 0
    multiplier := 1
    layout := board.premiums()
    for tile := squares(word)[]
        letter := board.character_index(board.letters, tile.letter)
        value := if letter < 0 then 0 else values[letter]
        if grid[tile.row][tile.column] == "."
            premium := layout[tile.row][tile.column]
            value *= if premium == "+" then 2 else (if premium == "^" then 3 else 1)
            multiplier *= if premium == "*" then 2 else (if premium == "#" then 3 else 1)
        score += value
    return score * multiplier

class Scored(int score, board.Tile[] placed, board.Move[] words, int[] word_scores)

Scored score_move(string[][] grid, board.Move move, int[] values)
    board.Move[] words := [move].concat(cross_words(grid, move))
    word_scores := [for word := words[] then word_score(grid, word, values)]
    score := 0
    for value := word_scores[]
        score += value
    placed := placement(grid, move)
    if #placed == rack_size
        score += bingo_bonus
    return Scored(score, placed, words, word_scores)

int character_index(string alphabet, string character)
    for: i < int(#alphabet)
        if alphabet[i] == character
            return i
    return -1
string uppercase(string text)
    result := ""
    for: i < int(#text)
        at := character_index(lowercase_letters, text[i])
        result += if at >= 0 then letters[at] else text[i]
    return result
string lower_letter(string character)
    return lowercase_letters[character_index(letters, character)]
bool inside(int row, int column)
    return 0 <= row and row < 15 and 0 <= column and column < 15
string square_name(int row, int column)

string position(Move move)
    return if move.direction == 0 then string(move.row + 1) + columns[move.column] else square_name(move.row, move.column)
string[][] premiums()
    rows := ["#..+...#...+..#", ".*...^...^...*.", "..*...+.+...*..", "+..*...+...*..+", "....*.....*....", ".^...^...^...^.", "..+...+.+...+..", "#..+...*...+..#", "..+...+.+...+..", ".^...^...^...^.", "....*.....*....", "+..*...+...*..+", "..*...+.+...*..", ".*...^...^...*.", "#..+...#...+..#"]
    return [for row := rows[] then [for c := (0:<15)[] then row[c]]]

int[][] neighbours(int row, int column)
    int[][] candidates := [[row - 1, column], [row + 1, column], [row, column - 1], [row, column + 1]]
    return [for point := candidates[] if inside(point[0], point[1]) then point]
Move run_at(string[][] grid, int row, int column, int direction)
    dr := direction
    dc := 1 - direction
    while inside(row - dr, column - dc) and grid[row - dr][column - dc] != "."
        row -= dr
        column -= dc
    first_row := row
    first_column := column
    word := ""
    while inside(row, column) and grid[row][column] != "."
        word += grid[row][column]
        row += dr
        column += dc
    return Move(first_row, first_column, direction, word)
int integer(string text)
    result := int(text)
    return result
bool valid_name(string text)
    if #text < 1 or #text > 20
        return false
    for: i < int(#text)
        if board.character_index("abcdefghijklmnopqrstuvwxyz0123456789_", text[i]) < 0
            return false
    return true
bool valid_number(string text, int minimum, int maximum)
    if #text == 0 or (#text > 1 and text[0] == "0")
        return false
    for: i < int(#text)
        if board.character_index("0123456789", text[i]) < 0
            return false
    value := integer(text)
    return minimum <= value and value <= maximum
bool valid_strategy(string strategy)
    return strategy == "score" or strategy == "tiles" or strategy == "equity"
bool valid_word(string word, bool uppercase_only)
    if #word < 2 or #word > 15
        return false
    string alphabet := if uppercase_only then board.letters else board.letters + board.lowercase_letters
    for: i < int(#word)
        if board.character_index(alphabet, word[i]) < 0
            return false
    return true
bool lex_less(string first, string second)
    return first != second and [first, second].sorted()[0] == first
string sorted_tiles(string text)
    return [for i := (0:<int(#text))[] then text[i]].sorted().join("")
string collection(string text)
    return if #text == 0 then "-" else sorted_tiles(text)
int[] tile_counts(string text)
    result := [for i := (0:<27)[] then 0]
    for: i < int(#text)
        result[game.tile_index(text[i])] += 1
    return result
string kind_of(string character)
    return if board.character_index(board.letters, character) >= 0 then character else "?"
string remove_tiles(string rack, string removed)
    counts := tile_counts(rack)
    for: i < int(#removed)
        counts[game.tile_index(removed[i])] -= 1
    result := ""
    for: i < 27
        result += game.tile_order[i] * nat(counts[i])
    return sorted_tiles(result)
int string_index(string[] values, string wanted)
    for: i < int(#values)
        if values[i] == wanted
            return i
    return -1
string notation(Evaluated evaluated)
    return board.position(evaluated.move) + " " + evaluated.move.word
bool before(Evaluated first, Evaluated second, string strategy)
    primary_first := if strategy == "tiles" then int(#first.placed) else (if strategy == "equity" then first.score + first.value else 0)
    primary_second := if strategy == "tiles" then int(#second.placed) else (if strategy == "equity" then second.score + second.value else 0)
    if primary_first != primary_second
        return primary_first > primary_second
    if first.score != second.score
        return first.score > second.score
    if first.move.row != second.move.row
        return first.move.row < second.move.row
    if first.move.column != second.move.column
        return first.move.column < second.move.column
    if first.move.direction != second.move.direction
        return first.move.direction < second.move.direction
    return lex_less(first.move.word, second.move.word)
Evaluated[] sorted_moves(Evaluated[] moves, string strategy)
    if #moves < 2
        return moves
    middle := int(#moves) / 2
    left := sorted_moves(moves[0:<middle], strategy)
    right := sorted_moves(moves[middle:<int(#moves)], strategy)
    Evaluated[] result := []
    a := 0
    b := 0
    while a < int(#left) and b < int(#right)
        if before(right[b], left[a], strategy)
            result = result.append(right[b])
            b += 1
        else
            result = result.append(left[a])
            a += 1
    return result.concat(left[a:<int(#left)]).concat(right[b:<int(#right)])
int strategy_choice(Evaluated[] moves, string strategy)
    result := 0
    for: 1 <= i < int(#moves)
        if before(moves[i], moves[result], strategy)
            result = i
    return result
class Inputs()

        if #current == 0
            if line != "" and line[0] != "#"
                current = [line]
        elif line == "END"
            jobs = jobs.append(current)
            current = []
        else
            current = current.append(line)

    string[] dictionary_lines := ["DICTIONARY lines={candidate_lines} words={trie.word_count} skipped={skipped} duplicates={duplicates} nodes={#trie.children}", "lengths " + length_parts.join(" ")]
class Scrabble(Inputs input_data)
    int[] values := input_data.values
    int[] counts := input_data.counts
    int[] weights := input_data.weights
    dictionary_trie.Trie trie := input_data.trie
    top_totals := [0, 0, 0, 0]
    validation_totals := [0, 0, 0, 0, 0, 0]
    game_totals := [0, 0, 0, 0, 0, 0, 0, 0, 0]
    string[][] move_blocks := [input_data.dictionary_lines]
    string[][] validation_blocks := []
    string[][] game_blocks := []
    string job_error(string[] job, string[] seen)
        fields := job[0].split(" ")
        kind := fields[0]
        name := fields[1]
        if not valid_name(name)
            return "bad name"
        if string_index(seen, name) >= 0
            return "duplicate name"
        if kind == "TOPN"
            if not valid_number(fields[2], 1, 99)
                return "bad count"
            if not valid_strategy(fields[3])
                return "bad order"
            rack := job[1].split(" ")[1]
            if not valid_rack(rack)
                return "bad rack"
            return board_error(job[3:<int(#job)], rack)
        if kind == "VALIDATE"
            string[] rows := []
            for line := job[2:<int(#job)][]
                if line.starts_with("MOVE")
                    break
                rows = rows.append(line)
            return board_error(rows, "")
        if not valid_number(fields[2], 1, 2147483646)
            return "bad seed"
        player_count := int(#job) - 1
        if player_count < 2 or player_count > 4
            return "bad player count {player_count}"
        string[] names := []
        for: 1 <= i < int(#job)
            player := job[i].split(" ")
            if not valid_name(player[1]) or not valid_strategy(player[2])
                return "bad player {i}"
            if kind == "GAME" and not valid_number(player[3], 0, 99)
                return "bad player {i}"
            if string_index(names, player[1]) >= 0
                return "duplicate player {i}"
            names = names.append(player[1])
        return ""
    int leave_value(string rack)
        quantities := tile_counts(rack)
        duplicates := 0
        vowels := 0
        consonants := 0
        result := 0
        for: i < 27
            result += quantities[i] * weights[i]
            if i < 26
                if quantities[i] > 1
                    duplicates += quantities[i] - 1
                if board.character_index("AEIOU", game.tile_order[i]) >= 0
                    vowels += quantities[i]
                else
                    consonants += quantities[i]
        imbalance := vowels - consonants
        if imbalance < 0
            imbalance = -imbalance
        penalty := if imbalance > 1 then imbalance - 1 else 0
        return result - 4 * duplicates - 3 * penalty
    Evaluated evaluate(string[][] grid, string rack, board.Move move)
        scored := scoring.score_move(grid, move, values)
        removed := [for tile := scored.placed[] then kind_of(tile.letter)].join("")
        leave := remove_tiles(rack, removed)
        words := [for i := (0:<int(#scored.words))[] then scored.words[i].word + ":" + string(scored.word_scores[i])].join(",")
        return Evaluated(move, scored.score, scored.placed, words, leave, leave_value(leave))
    Evaluated[] legal_moves(string[][] grid, string rack)
        Evaluated[] result := [for move := move_generator.generate(grid, rack, trie)[] then evaluate(grid, rack, move)]
        return sorted_moves(result, "score")
    string[] .duplicate_game(string[] job)
        header := job[0].split(" ")
        name := header[1]
        seed := integer(header[2])
        player_count := int(#job) - 1
        bag := game.Bag(seed, counts)
        .bag.shuffle()
        grid := board.empty_board()
        rack := .bag.draw(7)
        string[] names := []
        string[] strategies := []
        totals := [for p := (0:<int(player_count))[] then 0]
        lines := ["DUPLICATE {name} seed={seed} players={player_count}"]
        for: 1 <= i < int(#job)
            fields := job[i].split(" ")
            names = names.append(fields[1])
            strategies = strategies.append(fields[2])
            lines = lines.append("player {i} {fields[1]} strategy={fields[2]}")
        round_number := 0
        master_total := 0
        reason := "stuck"
        while true
            Evaluated[] legal := legal_moves(grid, rack)
            if #legal == 0
                break
            round_number += 1
            master := legal[0]
            int[] picks := []
            for: p < player_count
                score := legal[strategy_choice(legal, strategies[p])].score
                totals[p] += score
                picks = picks.append(score)
            master_total += master.score
            lines = lines.append("round {round_number} rack=" + collection(rack) + " moves={#legal} master " + notation(master) + " score={master.score} picks=" + [for score := picks[] then string(score)].join(","))
            grid = scoring.apply(grid, master.placed)
            drawn := .bag.draw(7 - int(#master.leave))
            rack = master.leave + drawn
            if rack == ""
                reason = "out"
                break
        suffix := if reason == "stuck" then " rack=" + collection(rack) else ""
        lines = lines.concat(["end reason={reason} rounds={round_number}" + suffix, "master total={master_total}"])
        maximum := totals[0]
        for: p < player_count
            if totals[p] > maximum
                maximum = totals[p]
            percent := if master_total >= 1 then 100 * totals[p] / master_total else 0
            lines = lines.append("final {names[p]} total={totals[p]} percent={percent}")
        winners := [for p := (0:<int(player_count))[] if totals[p] == maximum then names[p]]
        lines = lines.append("winner " + winners.join(","))
        game_totals[4] += round_number
        return lines.append(position_summary(grid)).concat(board.display(grid))
void write_blocks(string name, string[][] blocks)
    writer1 := file.write("output/" + name)
    .writer1([for block := blocks[] then block.join(NL)].join(NL + NL) + NL)

    program := Scrabble(Inputs())
    .program.handle_command()

    write_blocks("validation.txt", program.validation_blocks)
