new_trie <- function() {
  trie <- new.env(parent=emptyenv())
  trie$children <- list(integer(26L))
  trie$terminal <- FALSE
  trie$words <- 0L
  trie
}
tile_index <- function(letter) ifelse(letter == "?",27L,match(letter,LETTERS))
rack_counts <- function(rack) tabulate(tile_index(strsplit(rack,"",fixed=TRUE)[[1L]]),nbins=27L)
add_word <- function(trie,word) {
  node <- 1L
  for (letter in strsplit(word,"",fixed=TRUE)[[1L]]) {
    i <- tile_index(letter)
    if (trie$children[[node]][i] == 0L) {
      trie$children[[length(trie$children)+1L]] <- integer(26L)
      trie$terminal <- c(trie$terminal,FALSE)
      trie$children[[node]][i] <- length(trie$children)
    }
    node <- trie$children[[node]][i]
  }
  if (trie$terminal[node]) return(FALSE)
  trie$terminal[node] <- TRUE
  trie$words <- trie$words+1L
  TRUE
}
prefix_node <- function(trie,word) {
  node <- 1L
  for (letter in strsplit(word,"",fixed=TRUE)[[1L]]) {
    node <- trie$children[[node]][tile_index(letter)]
    if (node == 0L) break
  }
  node
}
has_word <- function(trie,word) {
  node <- prefix_node(trie,word)
  node != 0L && trie$terminal[node]
}
TILE_ORDER <- c(LETTERS,"?")
new_minstd <- function(seed) {
  random <- new.env(parent=emptyenv())
  random$state <- seed
  random
}
next_random <- function(random) {
  next_state <- 48271L*(random$state %% 44488L)-3399L*(random$state %/% 44488L)
  if (next_state <= 0L) next_state <- next_state+2147483647L
  random$state <- next_state
  next_state
}
shuffle_bag <- function(bag,random) {
  if (length(bag$tiles) > 1L) for (i in seq.int(length(bag$tiles),2L)) {
    j <- next_random(random) %% i+1L
    saved <- bag$tiles[i]
    bag$tiles[i] <- bag$tiles[j]
    bag$tiles[j] <- saved
  }
  invisible(NULL)
}
make_bag <- function(counts,random) {
  bag <- new.env(parent=emptyenv())
  bag$tiles <- rep(TILE_ORDER,counts)
  shuffle_bag(bag,random)
  bag
}
draw_tiles <- function(bag,n) {
  drawn <- character()
  for (i in seq_len(min(n,length(bag$tiles)))) {
    drawn <- c(drawn,tail(bag$tiles,1L))
    bag$tiles <- bag$tiles[-length(bag$tiles)]
  }
  paste(drawn,collapse="")
}
anchors <- function(board) {
  if (!any(board != ".")) return(list(CENTER))
  result <- list()
  for (r in seq_len(BOARD_SIZE)) for (c in seq_len(BOARD_SIZE)) if (board[r,c] == "." && any(vapply(
      neighbors(r,c),
      function(p) board[p[1L],p[2L]] != ".",
      logical(1)
  ))) result[[length(result)+1L]] <- c(r,c)
  result
}
cross_checks <- function(board,trie) {
  result <- list()
  for (r in seq_len(BOARD_SIZE)) for (c in seq_len(BOARD_SIZE)) {
    if (board[r,c] != ".") next
    first <- last <- r
    while (first > 1L && board[first-1L,c] != ".") first <- first-1L
    while (last < BOARD_SIZE && board[last+1L,c] != ".") last <- last+1L
    if (first == last) next
    prefix <- if (first < r) toupper(paste(board[seq.int(first,r-1L),c],collapse="")) else ""
    suffix <- if (r < last) toupper(paste(board[seq.int(r+1L,last),c],collapse="")) else ""
    allowed <- LETTERS[vapply(LETTERS,function(l) has_word(trie,paste0(prefix,l,suffix)),logical(1))]
    result[[paste(r,c,sep=",")]] <- allowed
  }
  result
}
across_moves <- function(board,rack,trie,anchor_squares,checks) {
  result <- list()
  counts <- rack_counts(rack)
  anchor_keys <- vapply(anchor_squares,function(p) paste(p,collapse=","),character(1))
  anchor_squares <- anchor_squares[order(
      vapply(anchor_squares,function(p) p[1L],integer(1)),
      vapply(anchor_squares,function(p) p[2L],integer(1))
  )]
  for (point in anchor_squares) {
    row <- point[1L]
    anchor <- point[2L]
    extend <- function(node,column,word,start) {
      empty <- column > BOARD_SIZE || board[row,column] == "."
      if (empty && column > anchor && trie$terminal[node]) result[[length(result)+1L]] <<- list(
          row=row,
          column=start,
          direction=0L,
          word=word
      )
      if (column > BOARD_SIZE) return(invisible(NULL))
      if (!empty) {
        letter <- board[row,column]
        child <- trie$children[[node]][tile_index(toupper(letter))]
        if (child != 0L) extend(child,column+1L,paste0(word,letter),start)
        return(invisible(NULL))
      }
      key <- paste(row,column,sep=",")
      constrained <- key %in% names(checks)
      allowed <- checks[[key]]
      for (i in which(trie$children[[node]] != 0L)) {
        letter <- LETTERS[i]
        if (constrained && !(letter %in% allowed)) next
        child <- trie$children[[node]][i]
        for (tile in c(i,27L)) if (counts[tile] > 0L) {
          counts[tile] <<- counts[tile]-1L
          written <- if (tile == 27L) tolower(letter) else letter
          extend(child,column+1L,paste0(word,written),start)
          counts[tile] <<- counts[tile]+1L
        }
      }
      invisible(NULL)
    }
    left_part <- function(node,prefix,remaining) {
      extend(node,anchor,prefix,anchor-nchar(prefix))
      if (remaining == 0L) return(invisible(NULL))
      for (i in which(trie$children[[node]] != 0L)) for (tile in c(i,27L)) if (counts[tile] > 0L) {
        counts[tile] <<- counts[tile]-1L
        written <- if (tile == 27L) tolower(LETTERS[i]) else LETTERS[i]
        left_part(trie$children[[node]][i],paste0(prefix,written),remaining-1L)
        counts[tile] <<- counts[tile]+1L
      }
      invisible(NULL)
    }
    if (anchor > 1L && board[row,anchor-1L] != ".") {
      first <- anchor-1L
      while (first > 1L && board[row,first-1L] != ".") first <- first-1L
      prefix <- paste(board[row,seq.int(first,anchor-1L)],collapse="")
      node <- prefix_node(trie,toupper(prefix))
      if (node != 0L) extend(node,anchor,prefix,first)
    } else {
      first <- anchor-1L
      while (first >= 1L && board[row,first] == "." && !(paste(
          row,
          first,
          sep=","
      ) %in% anchor_keys)) first <- first-1L
      left_part(1L,"",anchor-first-1L)
    }
  }
  result
}
placement_key <- function(new_tiles) {
  indices <- order(
      vapply(new_tiles,function(t) t$row,integer(1)),
      vapply(new_tiles,function(t) t$column,integer(1)),
      vapply(new_tiles,function(t) t$letter,character(1)),
      method="radix"
  )
  paste(
      vapply(new_tiles[indices],function(t) paste(t$row,t$column,t$letter,sep=":"),character(1)),
      collapse=";"
  )
}
generate_moves <- function(board,rack,trie) {
  anchor_squares <- anchors(board)
  across <- across_moves(board,rack,trie,anchor_squares,cross_checks(board,trie))
  transposed <- transpose_board(board)
  down <- across_moves(transposed,rack,trie,lapply(anchor_squares,rev),cross_checks(transposed,trie))
  candidates <- c(across,lapply(down,function(m) list(row=m$column,column=m$row,direction=1L,word=m$word)))
  result <- list()
  keys <- character()
  for (move in candidates) {
    new_tiles <- placement(board,move)
    if (length(new_tiles) == 1L && move$direction == 1L) {
      tile <- new_tiles[[1L]]
      if (nchar(run_at(apply_move(board,new_tiles),tile$row,tile$column,0L)$word) >= 2L) next
    }
    key <- placement_key(new_tiles)
    index <- match(key,keys)
    if (is.na(index)) {
      keys <- c(keys,key)
      result[[length(result)+1L]] <- move
    } else result[[index]] <- move
  }
  result
}
RACK_SIZE <- 7L
BINGO_BONUS <- 50L
move_cells <- function(move) {
  letters <- strsplit(move$word,"",fixed=TRUE)[[1L]]
  lapply(
      seq_along(letters),
      function(i) list(
          row=move$row+move$direction*(i-1L),
          column=move$column+(1L-move$direction)*(i-1L),
          letter=letters[i]
      )
  )
}
placement <- function(board,move) Filter(function(t) board[t$row,t$column] == ".",move_cells(move))
apply_move <- function(board,new_tiles) {
  result <- board
  for (tile in new_tiles) result[tile$row,tile$column] <- tile$letter
  result
}
cross_words <- function(board,move) {
  new_tiles <- placement(board,move)
  after <- apply_move(board,new_tiles)
  result <- list()
  for (tile in new_tiles) {
    word <- run_at(after,tile$row,tile$column,1L-move$direction)
    if (nchar(word$word) >= 2L) result[[length(result)+1L]] <- word
  }
  result
}
word_score <- function(board,word,values) {
  score <- 0L
  multiplier <- 1L
  for (tile in move_cells(word)) {
    r <- tile$row
    c <- tile$column
    l <- tile$letter
    value <- if (l %in% LETTERS) values[tile_index(l)] else 0L
    if (board[r,c] == ".") {
      premium <- substr(PREMIUMS[r],c,c)
      value <- value*(if (premium == "+") 2L else if (premium == "^") 3L else 1L)
      multiplier <- multiplier*(if (premium == "*") 2L else if (premium == "#") 3L else 1L)
    }
    score <- score+value
  }
  score*multiplier
}
score_move <- function(board,move,values) {
  words <- c(list(move),cross_words(board,move))
  scores <- vapply(words,function(word) word_score(board,word,values),integer(1))
  new_tiles <- placement(board,move)
  list(
      score=sum(scores)+if (length(new_tiles) == RACK_SIZE) BINGO_BONUS else 0L,
      new_tiles=new_tiles,
      words=lapply(seq_along(words),function(i) list(word=words[[i]],score=scores[i]))
  )
}

BOARD_SIZE <- 15L
CENTER <- c(8L,8L)
COLUMNS <- LETTERS[seq_len(15L)]
inside <- function(row,column) row >= 1L && row <= BOARD_SIZE && column >= 1L && column <= BOARD_SIZE
square_name <- function(row,column) paste0(COLUMNS[column],row)
move_position <- function(move) if (move$direction == 0L) paste0(move$row,COLUMNS[move$column]) else paste0(
    COLUMNS[move$column],
    move$row
)
parse_position <- function(text) {
  if (grepl("^([1-9]|1[0-5])[A-O]$",text)) return(list(
      row=as.integer(substr(text,1L,nchar(text)-1L)),
      column=match(substr(text,nchar(text),nchar(text)),COLUMNS),
      direction=0L
  ))
  if (grepl("^[A-O]([1-9]|1[0-5])$",text)) return(list(
      row=as.integer(substring(text,2L)),
      column=match(substr(text,1L,1L),COLUMNS),
      direction=1L
  ))
  NULL
}
transpose_board <- function(board) t(board)
board_tiles <- function(board) {
  result <- list()
  for (r in seq_len(BOARD_SIZE)) for (c in seq_len(BOARD_SIZE)) if (board[
      r,
      c
  ] != ".") result[[length(result)+1L]] <- list(
      row=r,
      column=c,
      letter=board[r,c]
  )
  result
}
neighbors <- function(row,column) {
  points <- list(c(row-1L,column),c(row+1L,column),c(row,column-1L),c(row,column+1L))
  Filter(function(p) inside(p[1L],p[2L]),points)
}
run_at <- function(board,row,column,direction) {
  dr <- direction
  dc <- 1L-direction
  while (inside(row-dr,column-dc) && board[row-dr,column-dc] != ".") {
    row <- row-dr
    column <- column-dc
  }
  first_row <- row
  first_column <- column
  letters <- character()
  while (inside(row,column) && board[row,column] != ".") {
    letters <- c(letters,board[row,column])
    row <- row+dr
    column <- column+dc
  }
  list(row=first_row,column=first_column,direction=direction,word=paste(letters,collapse=""))
}
STRATEGIES <- c("score","tiles","equity")
words <- function(...) paste(...,collapse=" ")
attributes_text <- function(...) {
  items <- list(...)
  paste(paste0(names(items),"=",vapply(items,as.character,character(1))),collapse=" ")
}
valid_name <- function(value) grepl("^[a-z0-9_]{1,20}$",value)
valid_number <- function(value,minimum,maximum) {
  if (!grepl("^(0|[1-9][0-9]*)$",value)) return(FALSE)
  upper <- as.character(maximum)
  if (nchar(value) > nchar(upper) || (nchar(value) == nchar(upper) && value > upper)) return(FALSE)
  as.integer(value) >= minimum
}
collection <- function(value) if (nzchar(value)) paste(
    sort(characters(value),method="radix"),
    collapse=""
) else "-"
tile_kind <- function(letter) ifelse(letter %in% LETTERS,letter,"?")
remove_tiles <- function(rack,removed) {
  counts <- rack_counts(rack)
  for (letter in characters(removed)) counts[tile_index(letter)] <- counts[tile_index(letter)]-1L
  paste(sort(rep(TILE_ORDER,counts),method="radix"),collapse="")
}
strategy_sort <- function(moves,strategy="score") {
  if (!length(moves)) return(moves)
  prefix <- if (strategy == "tiles") -vapply(
      moves,
      function(e) length(e$new_tiles),
      integer(1)
  ) else if (strategy == "equity") -vapply(
      moves,
      function(e) e$score+e$value,
      integer(1)
  ) else rep(0L,length(moves))
  indices <- order(
      prefix,
      -vapply(moves,function(e) e$score,integer(1)),
      vapply(moves,function(e) e$move$row,integer(1)),
      vapply(moves,function(e) e$move$column,integer(1)),
      vapply(moves,function(e) e$move$direction,integer(1)),
      vapply(moves,function(e) e$move$word,character(1)),
      method="radix"
  )
  moves[indices]
}
read_inputs <- function() {
  values <- counts <- weights <- integer(27L)
  for (line in readLines("input/tiles.txt",warn=FALSE)) {
    if (!nzchar(line) || startsWith(line,"#")) next
    f <- strsplit(line," ",fixed=TRUE)[[1L]]
    i <- tile_index(f[1L])
    values[i] <- as.integer(f[2L])
    counts[i] <- as.integer(f[3L])
    weights[i] <- as.integer(f[4L])
  }
  trie <- new_trie()
  statistics <- new_totals()
  lengths <- integer(15L)
  for (line in readLines("input/dictionary.txt",warn=FALSE)) {
    if (!nzchar(line) || startsWith(line,"#")) next
    bump(statistics,"lines")
    if (!grepl("^[A-Z]{2,15}$",line)) bump(
        statistics,
        "skipped"
    ) else if (add_word(trie,line)) lengths[nchar(line)] <- lengths[nchar(line)]+1L else bump(
        statistics,
        "duplicates"
    )
  }
  jobs <- list()
  current <- NULL
  for (line in readLines("input/jobs.txt",warn=FALSE)) {
    if (is.null(current)) {
      if (nzchar(line) && !startsWith(line,"#")) current <- line
    } else if (line == "END") {
      jobs[[length(jobs)+1L]] <- current
      current <- NULL
    } else current <- c(current,line)
  }
  dictionary_lines <- c(
      paste(
          "DICTIONARY",
          attributes_text(
              lines=counter(statistics,"lines"),
              words=trie$words,
              skipped=counter(statistics,"skipped"),
              duplicates=counter(statistics,"duplicates"),
              nodes=length(trie$children)
          )
      ),
      paste0("lengths",paste0(" ",which(lengths > 0L),"=",lengths[lengths > 0L],collapse=""))
  )
  a <- list(
      values=values,
      counts=counts,
      weights=weights,
      trie=trie,
      topn_totals=new_totals(),
      validation_totals=new_totals(),
      game_totals=new_totals()
  )
  list(program=a,jobs=jobs,dictionary_lines=dictionary_lines)
}
job_error <- function(a,job,seen) {
  f <- strsplit(job[1L]," ",fixed=TRUE)[[1L]]
  kind <- f[1L]
  name <- f[2L]
  if (!valid_name(name)) return("bad name")
  if (name %in% seen) return("duplicate name")
  if (kind == "TOPN") {
    if (!valid_number(f[3L],1L,99L)) return("bad count")
    if (!(f[4L] %in% STRATEGIES)) return("bad order")
    rack <- strsplit(job[2L]," ",fixed=TRUE)[[1L]][2L]
    if (!valid_rack(rack)) return("bad rack")
    return(board_error(a,job[-seq_len(3L)],rack))
  }
  if (kind == "VALIDATE") {
    rows <- character()
    for (line in job[-seq_len(2L)]) {
      if (startsWith(line,"MOVE")) break
      rows <- c(rows,line)
    }
    return(board_error(a,rows,""))
  }
  if (!valid_number(f[3L],1L,2147483646L)) return("bad seed")
  if (length(job)-1L < 2L || length(job)-1L > 4L) return(paste("bad player count",length(job)-1L))
  names <- character()
  for (i in seq_len(length(job)-1L)) {
    player <- strsplit(job[i+1L]," ",fixed=TRUE)[[1L]]
    if (!valid_name(player[2L]) || !(player[3L] %in% STRATEGIES) || (kind == "GAME" && !valid_number(
        player[4L],
        0L,
        99L
    ))) return(paste("bad player",i))
    if (player[2L] %in% names) return(paste("duplicate player",i))
    names <- c(names,player[2L])
  }
  NULL
}
leave_value <- function(a,rack) {
  counts <- rack_counts(rack)
  duplicates <- sum(pmax(0L,counts[seq_len(26L)]-1L))
  vowels <- sum(counts[tile_index(VOWELS)])
  consonants <- sum(counts[which(!(LETTERS %in% VOWELS))])
  sum(a$weights[tile_index(characters(rack))])-4L*duplicates-3L*max(0L,abs(vowels-consonants)-1L)
}
evaluate_move <- function(a,board,rack,move) {
  scored <- score_move(board,move,a$values)
  removed <- paste(vapply(scored$new_tiles,function(t) tile_kind(t$letter),character(1)),collapse="")
  leave <- remove_tiles(rack,removed)
  word_list <- paste(
      vapply(scored$words,function(w) paste0(w$word$word,":",w$score),character(1)),
      collapse=","
  )
  list(
      move=move,
      score=scored$score,
      new_tiles=scored$new_tiles,
      words=word_list,
      leave=leave,
      value=leave_value(a,leave)
  )
}
legal_moves <- function(a,board,rack) strategy_sort(lapply(
    generate_moves(board,rack,a$trie),
    function(m) evaluate_move(a,board,rack,m)
))
duplicate_game <- function(a,job) {
  f <- strsplit(job[1L]," ",fixed=TRUE)[[1L]]
  name <- f[2L]
  seed <- as.integer(f[3L])
  random <- new_minstd(seed)
  bag <- make_bag(a$counts,random)
  board <- matrix(".",BOARD_SIZE,BOARD_SIZE)
  rack <- draw_tiles(bag,RACK_SIZE)
  players <- lapply(strsplit(job[-1L]," ",fixed=TRUE),function(f) f[-1L])
  totals <- integer(length(players))
  lines <- words("DUPLICATE",name,attributes_text(seed=seed,players=length(players)))
  for (i in seq_along(players)) lines <- c(
      lines,
      words("player",i,players[[i]][1L],paste0("strategy=",players[[i]][2L]))
  )
  round_number <- master_total <- 0L
  reason <- "stuck"
  repeat {
    legal <- legal_moves(a,board,rack)
    if (!length(legal)) break
    round_number <- round_number+1L
    master <- legal[[1L]]
    choices <- lapply(players,function(p) strategy_sort(legal,p[2L])[[1L]])
    totals <- totals+vapply(choices,function(m) m$score,integer(1))
    master_total <- master_total+master$score
    picks <- paste(vapply(choices,function(m) m$score,integer(1)),collapse=",")
    lines <- c(
        lines,
        words(
            "round",
            round_number,
            attributes_text(rack=collection(rack),moves=length(legal)),
            "master",
            notation(master),
            attributes_text(score=master$score,picks=picks)
        )
    )
    board <- apply_move(board,master$new_tiles)
    rack <- paste0(master$leave,draw_tiles(bag,RACK_SIZE-nchar(master$leave)))
    if (!nzchar(rack)) {
      reason <- "out"
      break
    }
  }
  ending <- attributes_text(reason=reason,rounds=round_number)
  if (reason == "stuck") ending <- paste0(ending," rack=",collection(rack))
  lines <- c(lines,paste("end",ending),paste0("master total=",master_total))
  for (i in seq_along(players)) {
    percent <- if (master_total >= 1L) (100L*totals[i]) %/% master_total else 0L
    lines <- c(lines,words("final",players[[i]][1L],attributes_text(total=totals[i],percent=percent)))
  }
  winners <- vapply(players[which(totals == max(totals))],function(p) p[1L],character(1))
  lines <- c(lines,paste("winner",paste(winners,collapse=",")),position_summary(board),board_display(board))
  bump(a$game_totals,"rounds",round_number)
  lines
}
main <- function() {
  invisible(Sys.setlocale("LC_COLLATE","C"))
  input <- read_inputs()
  a <- input$program
  move_blocks <- list(input$dictionary_lines)
  validation_blocks <- game_blocks <- list()
  seen <- character()
  for (job in input$jobs) {
    f <- strsplit(job[1L]," ",fixed=TRUE)[[1L]]
    kind <- f[1L]
    name <- f[2L]
    totals <- if (kind == "TOPN") a$topn_totals else if (kind == "VALIDATE") a$validation_totals else a$game_totals
    key <- if (kind %in% c("TOPN","VALIDATE")) "jobs" else if (kind == "GAME") "games" else "duplicates"
    bump(totals,key)
    error <- job_error(a,job,seen)
    seen <- union(seen,name)
    if (!is.null(error)) {
      block <- c(paste(kind,name),paste("error",error))
      bump(totals,"errors")
    } else block <- switch(
        kind,
        TOPN=topn(a,job),
        VALIDATE=validate_job(a,job),
        GAME=ordinary_game(a,job),
        DUPLICATE=duplicate_game(a,job)
    )
    if (kind == "TOPN") {
      move_blocks[[length(move_blocks)+1L]] <- block
    } else if (kind == "VALIDATE") {
      validation_blocks[[length(validation_blocks)+1L]] <- block
    } else {
      game_blocks[[length(game_blocks)+1L]] <- block
    }
  }
  move_blocks[[length(move_blocks)+1L]] <- paste(
      "SUMMARY",
      counter_text(a$topn_totals,c("jobs","errors","moves","bingos"))
  )
  validation_blocks[[length(validation_blocks)+1L]] <- paste(
      "SUMMARY",
      counter_text(a$validation_totals,c("jobs","errors","moves","valid","invalid","loss"))
  )
  game_blocks[[length(game_blocks)+1L]] <- paste(
      "SUMMARY",
      counter_text(
          a$game_totals,
          c("games","duplicates","errors","turns","rounds","plays","exchanges","passes","bingos")
      )
  )
  dir.create("output",showWarnings=FALSE)
  outputs <- list("moves.txt"=move_blocks,"validation.txt"=validation_blocks,"game.txt"=game_blocks)
  for (filename in names(outputs)) {
    text <- paste(
        vapply(outputs[[filename]],function(block) paste(block,collapse="\n"),character(1)),
        collapse="\n\n"
    )
    connection <- file(paste0("output/",filename),"wb")
    writeChar(paste0(text,"\n"),connection,eos=NULL,useBytes=TRUE)
    close(connection)
  }
}
