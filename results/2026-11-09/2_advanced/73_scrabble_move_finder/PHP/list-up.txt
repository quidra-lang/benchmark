<?php

class TrieNode
{
    public $children = [];
    public $terminal = false;
}

class Trie
{
    public $root;
    public $nodes = 1;
    public $words = 0;

    public function __construct()
    {
        $this->root = new TrieNode();
    }

    public function add($word)
    {
        $node = $this->root;
        foreach (str_split($word) as $letter) {
            if (!isset($node->children[$letter])) {
                $node->children[$letter] = new TrieNode();
                $this->nodes++;
            }
            $node = $node->children[$letter];
        }
        if ($node->terminal) {
            return false;
        }
        $node->terminal = true;
        $this->words++;
        return true;
    }

    public function find($prefix)
    {
        $node = $this->root;
        foreach (str_split($prefix) as $letter) {
            $node = $node->children[$letter] ?? null;
            if ($node === null) {
                return null;
            }
        }
        return $node;
    }

    public function has($word)
    {
        $node = $this->find($word);
        return $node !== null && $node->terminal;
    }

class Minstd
{
    public $state;

    public function __construct($seed)
    {
        $this->state = $seed;
    }

    public function next()
    {
        $this->state = (48271 * $this->state) % 2147483647;
        return $this->state;
    }
}

function shuffleBag(&$bag, $random)
{
    for ($i = count($bag) - 1; $i >= 1; $i--) {
        $j = $random->next() % ($i + 1);
        [$bag[$i], $bag[$j]] = [$bag[$j], $bag[$i]];
    }
}

function makeBag($table, $random)
{
    $bag = [];
    foreach ($table as $letter => $tile) {
        for ($i = 0; $i < $tile['count']; $i++) {
            $bag[] = $letter;
        }
    }
    shuffleBag($bag, $random);
    return $bag;
}

function draw(&$bag, $count)
{
    $tiles = '';
    for ($i = 0; $i < $count && $bag; $i++) {
        $tiles .= array_pop($bag);
    }
    return $tiles;
}

const ALPHABET = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ';

function anchors($grid)
{
    $result = [];
    if (!boardTiles($grid)) {
        return [CENTER * BOARD_SIZE + CENTER => true];
    }
    for ($row = 0; $row < BOARD_SIZE; $row++) {
        for ($column = 0; $column < BOARD_SIZE; $column++) {
            if ($grid[$row][$column] === '.' && adjacent($grid, $row, $column)) {
                $result[$row * BOARD_SIZE + $column] = true;
            }
        }
    }
    return $result;
}

function crossChecks($grid, $trie)
{
    $allowed = array_fill(0, BOARD_SIZE, array_fill(0, BOARD_SIZE, null));
    $checked = $dead = $total = 0;
    for ($row = 0; $row < BOARD_SIZE; $row++) {
        for ($column = 0; $column < BOARD_SIZE; $column++) {
            if ($grid[$row][$column] !== '.') {
                continue;
            }
            $letters = [];
            if (!occupied($grid, $row - 1, $column) && !occupied($grid, $row + 1, $column)) {
                foreach (str_split(ALPHABET) as $letter) {
                    $letters[$letter] = true;
                }
            } else {
                $checked++;
                foreach (str_split(ALPHABET) as $letter) {
                    $word = crossWord($grid, ['row' => $row, 'column' => $column, 'letter' => $letter], 'down');
                    if ($trie->has(strtoupper($word['word']))) {
                        $letters[$letter] = true;
                    }
                }
                if (!$letters) {
                    $dead++;
                }
                $total += count($letters);
            }
            $allowed[$row][$column] = $letters;
        }
    }
    return ['allowed' => $allowed, 'stats' => [$checked, $dead, $total]];
}

function acrossMoves($grid, $rack, $trie, $anchorSet, $checks)
{
    $result = [];
    $counts = array_count_values(str_split($rack));
    foreach ($anchorSet as $anchor => $unused) {
        $row = intdiv($anchor, BOARD_SIZE);
        $anchorColumn = $anchor % BOARD_SIZE;
        $choices = function ($node, $allowed, $use) use (&$counts) {
            $letters = array_keys($node->children);
            sort($letters, SORT_STRING);
            foreach ($letters as $letter) {
                if ($allowed !== null && !isset($allowed[$letter])) {
                    continue;
                }
                foreach ([$letter, '?'] as $tile) {
                    if (($counts[$tile] ?? 0) === 0) {
                        continue;
                    }
                    $counts[$tile]--;
                    $use($node->children[$letter], $tile === '?' ? strtolower($letter) : $letter);
                    $counts[$tile]++;
                }
            }
        };
        $extend = function ($column, $node, $word, $start, $filled) use (&$extend, &$result, $grid, $row, $anchorColumn, $checks, $choices) {
            $isOccupied = $column < BOARD_SIZE && $grid[$row][$column] !== '.';
            if (!$isOccupied && $filled && $node->terminal) {
                $tiles = [];
                for ($i = 0; $i < strlen($word); $i++) {
                    if ($grid[$row][$start + $i] === '.') {
                        $tiles[] = ['row' => $row, 'column' => $start + $i, 'letter' => $word[$i]];
                    }
                }
                $result[] = ['row' => $row, 'column' => $start, 'direction' => 'across', 'word' => $word, 'tiles' => $tiles];
            }
            if ($column >= BOARD_SIZE) {
                return;
            }
            if ($isOccupied) {
                $letter = $grid[$row][$column];
                $child = $node->children[strtoupper($letter)] ?? null;
                if ($child !== null) {
                    $extend($column + 1, $child, $word . $letter, $start, $filled);
                }
            } else {
                $choices($node, $checks['allowed'][$row][$column], function ($child, $letter) use (&$extend, $column, $word, $start, $filled, $anchorColumn) {
                    $extend($column + 1, $child, $word . $letter, $start, $filled || $column === $anchorColumn);
                });
            }
        };
        if (occupied($grid, $row, $anchorColumn - 1)) {
            $start = $anchorColumn - 1;
            while (occupied($grid, $row, $start - 1)) {
                $start--;
            }
            $prefix = implode('', array_slice($grid[$row], $start, $anchorColumn - $start));
            $node = $trie->find(strtoupper($prefix));
            if ($node !== null) {
                $extend($anchorColumn, $node, $prefix, $start, false);
            }
        } else {
            $limit = 0;
            while ($anchorColumn - $limit - 1 >= 0 && $grid[$row][$anchorColumn - $limit - 1] === '.' &&
                !isset($anchorSet[$row * BOARD_SIZE + $anchorColumn - $limit - 1])) {
                $limit++;
            }
            $leftPart = function ($node, $word) use (&$leftPart, $extend, $choices, $anchorColumn, $limit) {
                $extend($anchorColumn, $node, $word, $anchorColumn - strlen($word), false);
                if (strlen($word) < $limit) {
                    $choices($node, null, function ($child, $letter) use (&$leftPart, $word) {
                        $leftPart($child, $word . $letter);
                    });
                }
            };
            $leftPart($trie->root, '');
        }
    }
    return $result;
}

function generateMoves($grid, $rack, $trie)
{
    $anchorSet = anchors($grid);
    $across = crossChecks($grid, $trie);
    $transposed = transposeBoard($grid);
    $down = crossChecks($transposed, $trie);
    $moves = acrossMoves($grid, $rack, $trie, $anchorSet, $across);
    foreach (acrossMoves($transposed, $rack, $trie, anchors($transposed), $down) as $move) {
        $tiles = transposeTiles($move['tiles']);
        if (count($tiles) === 1 && (occupied($grid, $tiles[0]['row'], $tiles[0]['column'] - 1) ||
            occupied($grid, $tiles[0]['row'], $tiles[0]['column'] + 1))) {
            continue;
        }
        $moves[] = array_merge($move, ['row' => $move['column'], 'column' => $move['row'], 'direction' => 'down', 'tiles' => $tiles]);
    }
    return ['moves' => $moves, 'anchors' => count($anchorSet), 'across' => $across['stats'], 'down' => $down['stats']];
}

const RACK_SIZE = 7;
const BINGO = 50;

function tileKind($letter)
{
    return $letter >= 'a' && $letter <= 'z' ? '?' : $letter;
}

function moveSquares($move)
{
    $result = [];
    foreach (str_split($move['word']) as $i => $letter) {
        $result[] = ['row' => $move['row'] + ($move['direction'] === 'down' ? $i : 0),
            'column' => $move['column'] + ($move['direction'] === 'across' ? $i : 0), 'letter' => $letter];
    }
    return $result;
}

function crossWord($grid, $tile, $direction)
{
    $dr = $direction === 'down' ? 1 : 0;
    $dc = $direction === 'across' ? 1 : 0;
    $row = $tile['row'];
    $column = $tile['column'];
    while (occupied($grid, $row - $dr, $column - $dc)) {
        $row -= $dr;
        $column -= $dc;
    }
    $start = ['row' => $row, 'column' => $column, 'direction' => $direction];
    $word = '';
    while (inside($row, $column) && (($row === $tile['row'] && $column === $tile['column']) || occupied($grid, $row, $column))) {
        $word .= (($row === $tile['row'] && $column === $tile['column']) ? $tile['letter'] : $grid[$row][$column]);
        $row += $dr;
        $column += $dc;
    }
    $start['word'] = $word;
    return $start;
}

function wordScore($grid, $word, $table)
{
    $total = 0;
    $multiplier = 1;
    foreach (moveSquares($word) as $tile) {
        $points = $table[tileKind($tile['letter'])]['value'];
        if ($grid[$tile['row']][$tile['column']] === '.') {
            $premium = premium($tile['row'], $tile['column']);
            if ($premium === '+') {
                $points *= 2;
            } elseif ($premium === '^') {
                $points *= 3;
            } elseif ($premium === '*') {
                $multiplier *= 2;
            } elseif ($premium === '#') {
                $multiplier *= 3;
            }
        }
        $total += $points;
    }
    return $total * $multiplier;
}

function scoreMove($grid, $move, $table)
{
    $tiles = array_values(array_filter(moveSquares($move), fn($tile) => $grid[$tile['row']][$tile['column']] === '.'));
    $words = [['row' => $move['row'], 'column' => $move['column'], 'direction' => $move['direction'], 'word' => $move['word']]];
    $perpendicular = $move['direction'] === 'across' ? 'down' : 'across';
    foreach ($tiles as $tile) {
        $cross = crossWord($grid, $tile, $perpendicular);
        if (strlen($cross['word']) >= 2) {
            $words[] = $cross;
        }
    }
    $total = count($tiles) === RACK_SIZE ? BINGO : 0;
    foreach ($words as &$word) {
        $word['score'] = wordScore($grid, $word, $table);
        $total += $word['score'];
    }
    unset($word);
    return array_merge($move, ['tiles' => $tiles, 'words' => $words, 'score' => $total]);
}

const BOARD_SIZE = 15;
const CENTER = 7;
const PREMIUM_ROWS = [
    '# . . + . . . # . . . + . . #',
    '. * . . . ^ . . . ^ . . . * .',
    '. . * . . . + . + . . . * . .',
    '+ . . * . . . + . . . * . . +',
    '. . . . * . . . . . * . . . .',
    '. ^ . . . ^ . . . ^ . . . ^ .',
    '. . + . . . + . + . . . + . .',
    '# . . + . . . * . . . + . . #',
    '. . + . . . + . + . . . + . .',
    '. ^ . . . ^ . . . ^ . . . ^ .',
    '. . . . * . . . . . * . . . .',
    '+ . . * . . . + . . . * . . +',
    '. . * . . . + . + . . . * . .',
    '. * . . . ^ . . . ^ . . . * .',
    '# . . + . . . # . . . + . . #'
];

function premium($row, $column)
{
    return PREMIUM_ROWS[$row][2 * $column];
}

function inside($row, $column)
{
    return $row >= 0 && $row < BOARD_SIZE && $column >= 0 && $column < BOARD_SIZE;
}

function occupied($grid, $row, $column)
{
    return inside($row, $column) && $grid[$row][$column] !== '.';
}

function squareName($row, $column)
{
    return chr(65 + $column) . ($row + 1);
}

function position($row, $column, $direction)
{
    return $direction === 'across' ? ($row + 1) . chr(65 + $column) : squareName($row, $column);
}

function emptyBoard()
{
    return array_fill(0, BOARD_SIZE, array_fill(0, BOARD_SIZE, '.'));
}

function transposeBoard($grid)
{
    $result = emptyBoard();
    for ($row = 0; $row < BOARD_SIZE; $row++) {
        for ($column = 0; $column < BOARD_SIZE; $column++) {
            $result[$row][$column] = $grid[$column][$row];
        }
    }
    return $result;
}

function transposeTiles($tiles)
{
    return array_map(fn($tile) => ['row' => $tile['column'], 'column' => $tile['row'], 'letter' => $tile['letter']], $tiles);
}


function boardTiles($grid)
{
    $result = [];
    for ($row = 0; $row < BOARD_SIZE; $row++) {
        for ($column = 0; $column < BOARD_SIZE; $column++) {
            if ($grid[$row][$column] !== '.') {
                $result[] = ['row' => $row, 'column' => $column, 'letter' => $grid[$row][$column]];
            }
        }
    }
    return $result;
}

function adjacent($grid, $row, $column)
{
    foreach ([[-1, 0], [1, 0], [0, -1], [0, 1]] as [$dr, $dc]) {
        if (occupied($grid, $row + $dr, $column + $dc)) {
            return true;
        }
    }
    return false;
}

function connected($grid)
{
    $placed = boardTiles($grid);
    if (!$placed) {
        return true;
    }
    $queue = [$placed[0]];
    $seen = [$placed[0]['row'] * BOARD_SIZE + $placed[0]['column'] => true];
    for ($i = 0; $i < count($queue); $i++) {
        $row = $queue[$i]['row'];
        $column = $queue[$i]['column'];
        foreach ([[-1, 0], [1, 0], [0, -1], [0, 1]] as [$dr, $dc]) {
            $r = $row + $dr;
            $c = $column + $dc;
            $key = $r * BOARD_SIZE + $c;
            if (occupied($grid, $r, $c) && !isset($seen[$key])) {
                $seen[$key] = true;
                $queue[] = ['row' => $r, 'column' => $c];
            }
        }
    }
    return count($seen) === count($placed);
}

function sortedTiles($tiles)
{
    $letters = str_split($tiles);
    sort($letters, SORT_STRING);
    return implode('', $letters);
}
function collection($tiles)
{
    $result = sortedTiles($tiles);
    return $result === '' ? '-' : $result;
}
function nameValid($name)
{
    return (bool)preg_match('/^[a-z0-9_]{1,20}$/D', $name);
}
function numberValid($text, $low, $high)
{
    return (bool)preg_match('/^(0|[1-9][0-9]*)$/D', $text) && (int)$text >= $low && (int)$text <= $high;
}
function notation($move)
{
    return position($move['row'], $move['column'], $move['direction']) . ' ' . $move['word'];
}
function readInput()
{
    $table = [];
    foreach (file('input/tiles.txt', FILE_IGNORE_NEW_LINES) as $line) {
        if ($line === '' || $line[0] === '#') {
            continue;
        }
        [$letter, $value, $count, $weight] = explode(' ', $line);
        $table[$letter] = ['value' => (int)$value, 'count' => (int)$count, 'weight' => (int)$weight];
    }
    $trie = new Trie();
    $stats = ['lines' => 0, 'skipped' => 0, 'duplicates' => 0, 'lengths' => []];
    foreach (file('input/dictionary.txt', FILE_IGNORE_NEW_LINES) as $line) {
        if ($line === '' || $line[0] === '#') {
            continue;
        }
        $stats['lines']++;
        if (!preg_match('/^[A-Z]{2,15}$/D', $line)) {
            $stats['skipped']++;
        } elseif (!$trie->add($line)) {
            $stats['duplicates']++;
        } else {
            $length = strlen($line);
            $stats['lengths'][$length] = ($stats['lengths'][$length] ?? 0) + 1;
        }
    }
    $lines = file('input/jobs.txt', FILE_IGNORE_NEW_LINES);
    $jobs = [];
    for ($i = 0; $i < count($lines); $i++) {
        if ($lines[$i] === '' || $lines[$i][0] === '#') {
            continue;
        }
        $fields = explode(' ', $lines[$i]);
        $kind = array_shift($fields);
        $name = array_shift($fields);
        $body = [];
        while ($lines[++$i] !== 'END') {
            $body[] = $lines[$i];
        }
        $jobs[] = ['kind' => $kind, 'name' => $name, 'args' => $fields, 'body' => $body];
    }
    return [$table, $trie, $stats, $jobs];
}
function checkJob($job, $duplicate, $table, $trie)
{
    if (!nameValid($job['name'])) {
        return ['error' => 'bad name'];
    }
    if ($duplicate) {
        return ['error' => 'duplicate name'];
    }
    if ($job['kind'] === 'TOPN') {
        if (!numberValid($job['args'][0], 1, 99)) {
            return ['error' => 'bad count'];
        }
        if (!in_array($job['args'][1], STRATEGIES, true)) {
            return ['error' => 'bad order'];
        }
        $rack = explode(' ', $job['body'][0])[1];
        if (!rackValid($rack)) {
            return ['error' => 'bad rack'];
        }
        return array_merge(checkBoard(array_slice($job['body'], 2), $rack, $table, $trie),
            ['rack' => $rack, 'count' => (int)$job['args'][0], 'strategy' => $job['args'][1]]);
    }
    if ($job['kind'] === 'VALIDATE') {
        $at = 1;
        while ($at < count($job['body']) && !str_starts_with($job['body'][$at], 'MOVE')) {
            $at++;
        }
        return array_merge(checkBoard(array_slice($job['body'], 1, $at - 1), '', $table, $trie), ['records' => array_slice($job['body'], $at)]);
    }
    if (!numberValid($job['args'][0], 1, 2147483646)) {
        return ['error' => 'bad seed'];
    }
    if (count($job['body']) < 2 || count($job['body']) > 4) {
        return ['error' => 'bad player count ' . count($job['body'])];
    }
    $players = $names = [];
    foreach ($job['body'] as $i => $line) {
        $fields = explode(' ', $line);
        $name = $fields[1];
        $strategy = $fields[2];
        $threshold = $fields[3] ?? null;
        if (!nameValid($name) || !in_array($strategy, STRATEGIES, true) ||
            ($job['kind'] === 'GAME' && !numberValid($threshold, 0, 99))) {
            return ['error' => 'bad player ' . ($i + 1)];
        }
        if (isset($names[$name])) {
            return ['error' => 'duplicate player ' . ($i + 1)];
        }
        $names[$name] = true;
        $players[] = ['name' => $name, 'strategy' => $strategy, 'threshold' => $job['kind'] === 'GAME' ? (int)$threshold : 0, 'index' => $i];
    }
    return ['players' => $players, 'seed' => (int)$job['args'][0]];
}
function leaveValue($tiles, $table)
{
    $counts = frequencies($tiles);
    $value = $duplicates = $vowels = $consonants = 0;
    foreach (str_split($tiles) as $tile) {
        $value += $table[$tile]['weight'];
        if (str_contains(VOWELS, $tile)) {
            $vowels++;
        } elseif ($tile !== '?') {
            $consonants++;
        }
    }
    foreach ($counts as $tile => $count) {
        if ($tile !== '?') {
            $duplicates += $count - 1;
        }
    }
    return $value - 4 * $duplicates - 3 * max(0, abs($vowels - $consonants) - 1);
}
function removeTiles($rack, $used)
{
    foreach ($used as $tile) {
        $index = strpos($rack, $tile);
        $rack = substr_replace($rack, '', $index, 1);
    }
    return $rack;
}
function standardOrder($a, $b)
{
    return ($b['score'] <=> $a['score']) ?: (($a['row'] <=> $b['row']) ?: (($a['column'] <=> $b['column']) ?:
        (($a['direction'] === $b['direction'] ? 0 : ($a['direction'] === 'across' ? -1 : 1)) ?: strcmp($a['word'], $b['word']))));
}
function strategyOrder($moves, $strategy)
{
    usort($moves, function ($a, $b) use ($strategy) {
        $order = $strategy === 'tiles' ? count($b['tiles']) - count($a['tiles']) :
            ($strategy === 'equity' ? ($b['score'] + $b['leaveValue']) - ($a['score'] + $a['leaveValue']) : 0);
        return $order ?: standardOrder($a, $b);
    });
    return $moves;
}
function legalMoves($grid, $rack, $trie, $table)
{
    $generated = generateMoves($grid, $rack, $trie);
    $moves = [];
    foreach ($generated['moves'] as $move) {
        $result = scoreMove($grid, $move, $table);
        $used = array_map(fn($tile) => tileKind($tile['letter']), $result['tiles']);
        $result['leave'] = removeTiles($rack, $used);
        $result['leaveValue'] = leaveValue($result['leave'], $table);
        $moves[] = $result;
    }
    usort($moves, 'standardOrder');
    foreach ($moves as $i => &$move) {
        $move['rank'] = $i + 1;
    }
    unset($move);
    $generated['moves'] = $moves;
    return $generated;
}
function duplicateJob($job, $settings, $trie, $table)
{
    $random = new Minstd($settings['seed']);
    $bag = makeBag($table, $random);
    $grid = emptyBoard();
    $players = [];
    foreach ($settings['players'] as $config) {
        $player = (object)$config;
        $player->total = 0;
        $players[] = $player;
    }
    $rack = draw($bag, RACK_SIZE);
    $masterTotal = $rounds = 0;
    $lines = ['DUPLICATE ' . $job['name'] . ' seed=' . $settings['seed'] . ' players=' . count($players)];
    foreach ($players as $i => $player) {
        $lines[] = 'player ' . ($i + 1) . ' ' . $player->name . ' strategy=' . $player->strategy;
    }
    while (true) {
        $moves = legalMoves($grid, $rack, $trie, $table)['moves'];
        if (!$moves) {
            $reason = 'stuck';
            break;
        }
        $rounds++;
        $master = $moves[0];
        $picks = [];
        foreach ($players as $player) {
            $chosen = strategyOrder($moves, $player->strategy)[0];
            $player->total += $chosen['score'];
            $picks[] = $chosen['score'];
        }
        $masterTotal += $master['score'];
        $lines[] = 'round ' . $rounds . ' rack=' . collection($rack) . ' moves=' . count($moves) . ' master ' . notation($master) .
            ' score=' . $master['score'] . ' picks=' . implode(',', $picks);
        place($grid, $master);
        $rack = $master['leave'];
        $rack .= draw($bag, RACK_SIZE - strlen($rack));
        if ($rack === '') {
            $reason = 'out';
            break;
        }
    }
    $lines[] = $reason === 'out' ? 'end reason=out rounds=' . $rounds : 'end reason=stuck rounds=' . $rounds . ' rack=' . collection($rack);
    $lines[] = 'master total=' . $masterTotal;
    $maximum = null;
    foreach ($players as $player) {
        $percent = $masterTotal > 0 ? intdiv(100 * $player->total, $masterTotal) : 0;
        $lines[] = 'final ' . $player->name . ' total=' . $player->total . ' percent=' . $percent;
        $maximum = $maximum === null ? $player->total : max($maximum, $player->total);
    }
    $winners = [];
    foreach ($players as $player) {
        if ($player->total === $maximum) {
            $winners[] = $player->name;
        }
    }
    $lines[] = 'winner ' . implode(',', $winners);
    $lines[] = positionSummary($grid);
    array_push($lines, ...displayBoard($grid));
    return ['lines' => $lines, 'rounds' => $rounds];
}
function main()
{
    [$table, $trie, $stats, $jobs] = readInput();
    $lengths = $stats['lengths'];
    ksort($lengths, SORT_NUMERIC);
    $fields = [];
    foreach ($lengths as $length => $count) {
        $fields[] = $length . '=' . $count;
    }
    $moves = [['DICTIONARY lines=' . $stats['lines'] . ' words=' . $trie->words . ' skipped=' . $stats['skipped'] .
        ' duplicates=' . $stats['duplicates'] . ' nodes=' . $trie->nodes, 'lengths ' . implode(' ', $fields)]];
    $validation = $games = [];
    $topTotal = ['jobs' => 0, 'errors' => 0, 'moves' => 0, 'bingos' => 0];
    $validationTotal = ['jobs' => 0, 'errors' => 0, 'moves' => 0, 'valid' => 0, 'invalid' => 0, 'loss' => 0];
    $gameTotal = ['games' => 0, 'duplicates' => 0, 'errors' => 0, 'turns' => 0, 'rounds' => 0, 'plays' => 0, 'exchanges' => 0, 'passes' => 0, 'bingos' => 0];
    $seen = [];
    foreach ($jobs as $job) {
        $duplicate = isset($seen[$job['name']]);
        $seen[$job['name']] = true;
        $settings = checkJob($job, $duplicate, $table, $trie);
        if ($job['kind'] === 'GAME') {
            $gameTotal['games']++;
        } elseif ($job['kind'] === 'DUPLICATE') {
            $gameTotal['duplicates']++;
        } elseif ($job['kind'] === 'TOPN') {
            $topTotal['jobs']++;
        } else {
            $validationTotal['jobs']++;
        }
        if (isset($settings['error'])) {
            $paragraph = [$job['kind'] . ' ' . $job['name'], 'error ' . $settings['error']];
            if ($job['kind'] === 'TOPN') {
                $topTotal['errors']++;
                $moves[] = $paragraph;
            } elseif ($job['kind'] === 'VALIDATE') {
                $validationTotal['errors']++;
                $validation[] = $paragraph;
            } else {
                $gameTotal['errors']++;
                $games[] = $paragraph;
            }
            continue;
        }
        if ($job['kind'] === 'TOPN') {
            $result = topJob($job, $settings, $trie, $table);
            $moves[] = $result['lines'];
            $topTotal['moves'] += $result['moves'];
            $topTotal['bingos'] += $result['bingos'];
        } elseif ($job['kind'] === 'VALIDATE') {
            $result = validationJob($job, $settings, $trie, $table);
            $validation[] = $result['lines'];
            foreach (['moves', 'valid', 'invalid', 'loss'] as $key) {
                $validationTotal[$key] += $result[$key];
            }
        } elseif ($job['kind'] === 'GAME') {
            $result = gameJob($job, $settings, $trie, $table);
            $games[] = $result['lines'];
            foreach (['turns', 'plays', 'exchanges', 'passes', 'bingos'] as $key) {
                $gameTotal[$key] += $result[$key];
            }
        } else {
            $result = duplicateJob($job, $settings, $trie, $table);
            $games[] = $result['lines'];
            $gameTotal['rounds'] += $result['rounds'];
        }
    }
    $moves[] = ['SUMMARY jobs=' . $topTotal['jobs'] . ' errors=' . $topTotal['errors'] . ' moves=' . $topTotal['moves'] . ' bingos=' . $topTotal['bingos']];
    $validation[] = ['SUMMARY jobs=' . $validationTotal['jobs'] . ' errors=' . $validationTotal['errors'] . ' moves=' . $validationTotal['moves'] .
        ' valid=' . $validationTotal['valid'] . ' invalid=' . $validationTotal['invalid'] . ' loss=' . $validationTotal['loss']];
    $games[] = ['SUMMARY games=' . $gameTotal['games'] . ' duplicates=' . $gameTotal['duplicates'] . ' errors=' . $gameTotal['errors'] .
        ' turns=' . $gameTotal['turns'] . ' rounds=' . $gameTotal['rounds'] . ' plays=' . $gameTotal['plays'] .
        ' exchanges=' . $gameTotal['exchanges'] . ' passes=' . $gameTotal['passes'] . ' bingos=' . $gameTotal['bingos']];
    if (!is_dir('output')) {
        mkdir('output', 0777, true);
    }
    writeParagraphs('moves.txt', $moves);
    writeParagraphs('validation.txt', $validation);
    writeParagraphs('game.txt', $games);
}
