pub struct Node {
    pub children: [Option<usize>;26],
    pub terminal: bool
}
pub struct Trie {
    pub nodes: Vec<Node>,
    pub words: usize
}
impl Trie {
    pub fn new() -> Self {
        Self {
            nodes: vec![Node {
                children: [None;26],terminal: false
            }],
            words: 0
        }
    }
    pub fn insert(&mut self, word: &[u8]) -> bool {
        let mut node = 0;
        for &letter in word {
            let index = (letter-b'A') as usize;
            node = if let Some(child) = self.nodes[node].children[index] {
                child
            } else {
                let child = self.nodes.len();
                self.nodes.push(Node {
                    children: [None;26],
                    terminal: false
                });
                self.nodes[node].children[index] = Some(child);
                child
            };
        }
        if self.nodes[node].terminal {
            false
        } else {
            self.nodes[node].terminal = true;
            self.words += 1;
            true
        }
    }
    pub fn prefix(&self, word: &[u8]) -> Option<usize> {
        let mut node = 0;
        for &letter in word {
            node = self.nodes[node].children[(letter.to_ascii_uppercase()-b'A') as usize]?;
        }
        Some(node)
    }
    pub fn contains(&self, word: &[u8]) -> bool {
        self.prefix(word).map_or(false,|n| self.nodes[n].terminal)
    }

pub fn index(tile: u8) -> usize {
    if tile == b'?' || tile.is_ascii_lowercase() {
        26
    } else {
        (tile-b'A') as usize
    }
}
pub fn shuffle(bag: &mut [u8], generator: &mut Generator) {
    for i in (1..bag.len()).rev() {
        let j = (generator.next()%(i+1) as i64) as usize;
        bag.swap(i,j);
    }
}
pub fn bag(counts: &[usize;27], generator: &mut Generator) -> Vec<u8> {
    let mut bag = Vec::new();
    for (i,&count) in counts.iter().enumerate() {
        bag.extend(std::iter::repeat(if i == 26 {
            b'?'
        } else {
            b'A'+i as u8
        }).take(count));
    }
    shuffle(&mut bag,generator);
    bag
}
pub fn draw(bag: &mut Vec<u8>, count: usize) -> Vec<u8> {
    let mut drawn = Vec::new();
    for _ in 0..count {
        if let Some(tile) = bag.pop() {
            drawn.push(tile);
        } else {
            break;
        }
    }
    drawn
}
#[derive(Clone)] pub struct Move {
    pub row: usize,
    pub column: usize,
    pub down: bool,
    pub word: Vec<u8>
}
pub fn anchors(board: &Board) -> [[bool;SIZE];SIZE] {
    let mut anchors = [[false;SIZE];SIZE];
    if board::tiles(board).is_empty() {
        anchors[7][7] = true;
    } else {
        for r in 0..SIZE {
            for c in 0..SIZE {
                anchors[r][c] = board[r][c] == b'.' && board::neighbors(r,c).iter().any(|&(a,b)| board[a][b] != b'.');
            }
        }
    }
    anchors
}
pub fn checks(board: &Board, trie: &Trie) -> ([[u32;SIZE];SIZE],[usize;3]) {
    let mut checks = [[0u32;SIZE];SIZE];
    let mut statistics = [0;3];
    for r in 0..SIZE {
        for c in 0..SIZE {
            if board[r][c] != b'.' {
                continue;
            }
            let mut first = r;
            while first > 0 && board[first-1][c] != b'.' {
                first -= 1;
            }
            let mut last = r;
            while last+1 < SIZE && board[last+1][c] != b'.' {
                last += 1;
            }
            if first == last {
                checks[r][c] = (1<<26)-1;
                continue;
            }
            statistics[0] += 1;
            for letter in b'A'..=b'Z' {
                let word: Vec<_> = (first..=last).map(|row| if row == r {
                    letter
                } else {
                    board[row][c].to_ascii_uppercase()
                }).collect();
                if trie.contains(&word) {
                    checks[r][c] |= 1<<(letter-b'A');
                }
            }
            statistics[1] += usize::from(checks[r][c] == 0);
            statistics[2] += checks[r][c].count_ones() as usize;
        }
    }
    (checks,statistics)
}
struct Generator<'a> {
    board: &'a Board,
    trie: &'a Trie,
    anchors: [[bool;SIZE];SIZE],
    checks: [[u32;SIZE];SIZE],
    row: usize,
    anchor: usize,
    result: Vec<Move>
}
impl<'a> Generator<'a> {
    fn extend(&mut self, column: usize, start: usize, node: usize, word: &mut Vec<u8>, counts: &mut [usize;27], placed: usize) {
        if column > self.anchor && placed > 0 && word.len() >= 2 && self.trie.nodes[node].terminal && (column == SIZE || self.board[self.row][column] == b'.') {
            self.result.push(Move {
                row: self.row,
                column: start,
                down: false,
                word: word.clone()
            });
        }
        if column == SIZE {
            return;
        }
        let existing = self.board[self.row][column];
        if existing != b'.' {
            if let Some(next) = self.trie.nodes[node].children[(existing.to_ascii_uppercase()-b'A') as usize] {
                word.push(existing);
                self.extend(column+1,start,next,word,counts,placed);
                word.pop();
            }
        } else {
            for letter in 0..26 {
                if self.checks[self.row][column]&(1<<letter) == 0 {
                    continue;
                }
                if let Some(next) = self.trie.nodes[node].children[letter] {
                    for tile in [letter,26].iter().copied() {
                        if counts[tile] > 0 {
                            counts[tile] -= 1;
                            word.push(if tile == 26 {
                                b'a'+letter as u8
                            } else {
                                b'A'+letter as u8
                            });
                            self.extend(column+1,start,next,word,counts,placed+1);
                            word.pop();
                            counts[tile] += 1;
                        }
                    }
                }
            }
        }
    }
    fn left(&mut self, limit: usize, node: usize, word: &mut Vec<u8>, counts: &mut [usize;27]) {
        let length = word.len();
        self.extend(self.anchor,self.anchor-length,node,word,counts,length);
        if word.len() == limit {
            return;
        }
        for letter in 0..26 {
            if let Some(next) = self.trie.nodes[node].children[letter] {
                for tile in [letter,26].iter().copied() {
                    if counts[tile] > 0 {
                        counts[tile] -= 1;
                        word.push(if tile == 26 {
                            b'a'+letter as u8
                        } else {
                            b'A'+letter as u8
                        });
                        self.left(limit,next,word,counts);
                        word.pop();
                        counts[tile] += 1;
                    }
                }
            }
        }
    }
    fn generate(mut self, rack: &[u8]) -> Vec<Move> {
        let mut counts = [0;27];
        for &tile in rack {
            counts[if tile == b'?' {
                26
            } else {
                (tile-b'A') as usize
            }] += 1;
        }
        for r in 0..SIZE {
            for c in 0..SIZE {
                if !self.anchors[r][c] {
                    continue;
                }
                self.row = r;
                self.anchor = c;
                if c > 0 && self.board[r][c-1] != b'.' {
                    let mut first = c-1;
                    while first > 0 && self.board[r][first-1] != b'.' {
                        first -= 1;
                    }
                    let mut prefix = self.board[r][first..c].to_vec();
                    if let Some(node) = self.trie.prefix(&prefix) {
                        self.extend(c,first,node,&mut prefix,&mut counts,0);
                    }
                } else {
                    let mut limit = 0;
                    while c > limit && self.board[r][c-limit-1] == b'.' && !self.anchors[r][c-limit-1] {
                        limit += 1;
                    }
                    self.left(limit,0,&mut Vec::new(),&mut counts);
                }
            }
        }
        self.result
    }
}
pub fn generate(board: &Board, rack: &[u8], trie: &Trie) -> Vec<Move> {
    fn across(board: &Board, rack: &[u8], trie: &Trie) -> Vec<Move> {
        let (checks,_) = checks(board,trie);
        Generator {
            board,
            trie,
            anchors: anchors(board),
            checks,
            row: 0,
            anchor: 0,
            result: Vec::new()
        }.generate(rack)
    }
    let mut result = across(board,rack,trie);
    let transposed = board::transpose(board);
    for mut movement in across(&transposed,rack,trie) {
        std::mem::swap(&mut movement.row,&mut movement.column);
        movement.down = true;
        let new: Vec<_> = movement.word.iter().enumerate().filter(|(i,_)| board[movement.row+*i][movement.column] == b'.').map(|(i,_)| (movement.row+i,movement.column)).collect();
        if new.len() == 1 {
            let (r,c) = new[0];
            if (c > 0 && board[r][c-1] != b'.') || (c+1 < SIZE && board[r][c+1] != b'.') {
                continue;
            }
        }
        result.push(movement);
    }
    result
}
pub const RACK_SIZE: usize = 7;
pub const BINGO: i64 = 50;
pub fn squares(m: &Move) -> Vec<(usize,usize,u8)> {
    m.word.iter().enumerate().map(|(i,&tile)| (m.row+if m.down {
        i
    } else {
        0
    },m.column+if m.down {
        0
    } else {
        i
    },tile)).collect()
}
pub fn placement(board: &Board, m: &Move) -> Vec<(usize,usize,u8)> {
    let mut result: Vec<_> = squares(m).into_iter().filter(|&(r,c,_)| board[r][c] == b'.').collect();
    result.sort();
    result
}
pub fn apply(board: &Board, m: &Move) -> Board {
    let mut result = *board;
    for (r,c,tile) in placement(board,m) {
        result[r][c] = tile;
    }
    result
}
pub fn words(board: &Board, m: &Move) -> Vec<Word> {
    let after = apply(board,m);
    let mut result = vec![Word {
        row: m.row,column: m.column,down: m.down,text: m.word.clone()
    }];
    for (r,c,_) in squares(m) {
        if board[r][c] != b'.' {
            continue;
        }
        let down = !m.down;
        let (mut first_r,mut first_c) = (r,c);
        while if down {
            first_r > 0 && after[first_r-1][first_c] != b'.'
        } else {
            first_c > 0 && after[first_r][first_c-1] != b'.'
        }
        {
            if down {
                first_r -= 1;
            } else {
                first_c -= 1;
            }
        }
        let (mut row,mut column) = (first_r,first_c);
        let mut text = Vec::new();
        while row < 15 && column < 15 && after[row][column] != b'.' {
            text.push(after[row][column]);
            if down {
                row += 1;
            } else {
                column += 1;
            }
        }
        if text.len() >= 2 {
            result.push(Word {
                row: first_r,
                column: first_c,
                down,
                text
            });
        }
    }
    result
}
pub fn word_score(board: &Board, word: &Word, values: &[i64;27]) -> i64 {
    let mut sum = 0;
    let mut multiplier = 1;
    for (i,&tile) in word.text.iter().enumerate() {
        let row = word.row+if word.down {
            i
        } else {
            0
        };
        let column = word.column+if word.down {
            0
        } else {
            i
        };
        let mut value = if tile.is_ascii_lowercase() {
            0
        } else {
            values[(tile-b'A') as usize]
        };
        if board[row][column] == b'.' {
            match board::PREMIUM[row].as_bytes()[column] {
                b'+' => value *= 2,b'^' => value *= 3,b'*' => multiplier *= 2,b'#' => multiplier *= 3,_ => ()
            }
        }
        sum += value;
    }
    sum*multiplier
}
pub fn score(board: &Board, movement: &Move, values: &[i64;27]) -> (i64,usize,String) {
    let tiles = placement(board,movement).len();
    let words = words(board,movement);
    let scores: Vec<_> = words.iter().map(|word| word_score(board,word,values)).collect();
    let mut score: i64 = scores.iter().sum();
    if tiles == RACK_SIZE {
        score += BINGO;
    }
    let notation = words.iter().zip(scores).map(|(w,s)| format!("{}:{}",String::from_utf8(w.text.clone()).unwrap(),s)).collect::<Vec<_>>().join(",");
    (score,tiles,notation)
}

pub fn number(text: &str, low: i64, high: i64) -> Option<i64> {
    if text.is_empty() || !text.bytes().all(|b| b.is_ascii_digit()) || (text.len() > 1 && text.starts_with('0')) {
        return None;
    }
    text.parse::<i64>().ok().filter(|&v| low <= v && v <= high)
}
pub fn transpose(board: &Board) -> Board {
    let mut result = empty();
    for r in 0..SIZE {
        for c in 0..SIZE {
            result[r][c] = board[c][r];
        }
    }
    result
}
pub fn tiles(board: &Board) -> Vec<(usize,usize,u8)> {
    let mut result = Vec::new();
    for r in 0..SIZE {
        for c in 0..SIZE {
            if board[r][c] != b'.' {
                result.push((r,c,board[r][c]));
            }
        }
    }
    result
}
pub fn neighbors(row: usize, column: usize) -> Vec<(usize,usize)> {
    let mut result = Vec::new();
    for (dr,dc) in [(-1,0),(1,0),(0,-1),(0,1)].iter().copied() {
        let r = row as i32+dr;
        let c = column as i32+dc;
        if r >= 0 && c >= 0 && r < SIZE as i32 && c < SIZE as i32 {
            result.push((r as usize,c as usize));
        }
    }
    result
}
pub fn connected(board: &Board) -> bool {
    let occupied = tiles(board);
    let Some(&(r,c,_)) = occupied.first() else {
        return true;
    };
    let mut seen = [[false;SIZE];SIZE];
    seen[r][c] = true;
    let mut queue = VecDeque::from([(r,c)]);
    let mut count = 0;
    while let Some((r,c)) = queue.pop_front() {
        count += 1;
        for (a,b) in neighbors(r,c) {
            if board[a][b] != b'.' && !seen[a][b] {
                seen[a][b] = true;
                queue.push_back((a,b));
            }
        }
    }
    count == occupied.len()
}
fn collection(tiles: &[u8]) -> String {
    let mut sorted = tiles.to_vec();

    if sorted.is_empty() {
        "-".to_owned()
    } else {
        text(&sorted)
    }
}

    }

        } else {
            None
        };
        names.insert(name.to_owned());
        let mut board = board::empty();
        let mut records = Vec::new();
        let mut rack = String::new();
        let mut players = Vec::new();
        let mut count = 0;
        let mut seed = 0;
        if kind == "TOPN" || kind == "VALIDATE" {
            if kind == "TOPN" {
                rack = block[1].split_whitespace().nth(1).unwrap().to_owned();
                if reason.is_none() {
                    if let Some(n) = board::number(header[2],1,99) {
                        count = n as usize;
                    } else {
                        reason = Some("bad count".to_owned());
                    }
                }
                if reason.is_none() && !valid_strategy(header[3]) {
                    reason = Some("bad order".to_owned());
                }
                if reason.is_none() && !valid_rack(&rack) {
                    reason = Some("bad rack".to_owned());
                }
            }
            let first = if kind == "TOPN" {
                3
            } else {
                2
            };
            let end = (first..block.len()).find(|&i| block[i].starts_with("MOVE")).unwrap_or(block.len());
            records = block[end..].to_vec();
            if reason.is_none() {
                match context.check_board(&block[first..end],rack.as_bytes()) {
                    Ok(value) => board = value,Err(error) => reason = Some(error)
                }
            }
        } else {
            players = block[1..].iter().map(|line| line.split_whitespace().map(str::to_owned).collect::<Vec<_>>()).collect::<Vec<_>>();
            if reason.is_none() {
                if let Some(value) = board::number(header[2],1,2147483646) {
                    seed = value;
                } else {
                    reason = Some("bad seed".to_owned());
                }
            }
            if reason.is_none() && !(2..=4).contains(&players.len()) {
                reason = Some(format!("bad player count {}",players.len()));
            }
            let mut player_names = BTreeSet::new();
            if reason.is_none() {
                for (i,w) in players.iter().enumerate() {
                    if !valid_name(&w[1]) || !valid_strategy(&w[2]) || (kind == "GAME" && board::number(&w[3],0,99).is_none()) {
                        reason = Some(format!("bad player {}",i+1));
                        break;
                    }
                    if !player_names.insert(w[1].clone()) {
                        reason = Some(format!("duplicate player {}",i+1));
                        break;
                    }
                }
            }
        }
        let error = reason.map(|reason| vec![format!("{} {}",kind,name),format!("error {}",reason)]);
        match kind {
            "TOPN" => {
                top_totals[0] += 1;
                if let Some(error) = error {
                    top_totals[1] += 1;
                    moves.push(error);
                } else {
                    let (lines,total,bingos) = context.topn(name,count,header[3],rack.as_bytes(),&board);
                    top_totals[2] += total;
                    top_totals[3] += bingos;
                    moves.push(lines);
                }
            }
            "VALIDATE" => {
                validation_totals[0] += 1;
                if let Some(error) = error {
                    validation_totals[1] += 1;
                    validation.push(error);
                } else {
                    let (lines,totals) = context.validate(name,board,&records);
                    validation_totals[2] += totals[0];
                    validation_totals[3] += totals[1];
                    validation_totals[4] += totals[2];
                    validation_totals[5] += totals[5];
                    validation.push(lines);
                }
            }
            _ => {
                game_totals[usize::from(kind == "DUPLICATE")] += 1;
                if let Some(error) = error {
                    game_totals[2] += 1;
                    games.push(error);
                } else {
                    let (lines,totals) = context.play_game(name,seed,&players,kind == "DUPLICATE");
                    game_totals[3] += totals[0];
                    game_totals[4] += totals[1];
                    game_totals[5] += totals[2];
                    game_totals[6] += totals[3];
                    game_totals[7] += totals[4];
                    game_totals[8] += totals[5];
                    games.push(lines);
                }
            }
        }
    }
    moves.push(vec![format!("SUMMARY jobs={} errors={} moves={} bingos={}",top_totals[0],top_totals[1],top_totals[2],top_totals[3])]);
    validation.push(vec![format!("SUMMARY jobs={} errors={} moves={} valid={} invalid={} loss={}",validation_totals[0],validation_totals[1],validation_totals[2],validation_totals[3],validation_totals[4],validation_totals[5])]);
    games.push(vec![format!("SUMMARY games={} duplicates={} errors={} turns={} rounds={} plays={} exchanges={} passes={} bingos={}",game_totals[0],game_totals[1],game_totals[2],game_totals[3],game_totals[4],game_totals[5],game_totals[6],game_totals[7],game_totals[8])]);
    fs::create_dir_all("output").unwrap();
    for (name,paragraphs) in [("moves.txt",moves),("validation.txt",validation),("game.txt",games)] {
        fs::write(format!("output/{}",name),paragraphs.iter().map(|p| p.join("\n")).collect::<Vec<_>>().join("\n\n")+"\n").unwrap();
    }
}
