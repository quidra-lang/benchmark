mutable struct WordTrie
    children::Vector{Dict{Char,Int}}
    terminal::BitVector
    words::Int
end
WordTrie() = WordTrie([Dict{Char,Int}()],falses(1),0)
tile_index(letter) = letter=='?' ? 27 : Int(letter)-Int('A')+1
function rack_counts(rack)
    counts = zeros(Int,27)
    for letter in rack
        counts[tile_index(letter)] += 1
    end
    counts
end
function add_word!(trie,word)
    node = 1
    for letter in word
        if !haskey(trie.children[node],letter)
            push!(trie.children,Dict{Char,Int}())
            push!(trie.terminal,false)
            trie.children[node][letter] = length(trie.children)
        end
        node = trie.children[node][letter]
    end
    trie.terminal[node] && return false
    trie.terminal[node] = true
    trie.words += 1
    true
end
function prefix_node(trie,word)
    node = 1
    for letter in word
        node = get(trie.children[node],letter,0)
        node==0 && break
    end
    node
end
function has_word(trie,word)
    node = prefix_node(trie,word)
    node!=0 && trie.terminal[node]
end
const TILE_ORDER = "ABCDEFGHIJKLMNOPQRSTUVWXYZ?"
mutable struct Minstd
    state::Int64
end
function next_random!(random)
    random.state = (48271*random.state)%2147483647
end
function shuffle_bag!(bag,random)
    for i in length(bag):-1:2
        j = next_random!(random)%i+1
        bag[i],bag[j] = bag[j],bag[i]
    end
    bag
end
function make_bag(counts,random)
    bag = [letter for letter in TILE_ORDER for _ in 1:counts[tile_index(letter)]]
    shuffle_bag!(bag,random)
end
function draw_tiles!(bag,n)
    join([pop!(bag) for _ in 1:min(n,length(bag))])
end
function anchors(board)
    isempty(board_tiles(board)) && return Set([CENTER])
    Set((r,c) for r in 1:BOARD_SIZE for c in 1:BOARD_SIZE if board[r,c]=='.' && any(
        p->board[p...]!='.',
        neighbors(r,c)
    ))
end
function cross_checks(board,trie)
    result = Dict{Tuple{Int,Int},Set{Char}}()
    for r in 1:BOARD_SIZE,c in 1:BOARD_SIZE
        board[r,c]!='.' && continue
        first,last = r,r
        while first>1 && board[first-1,c]!='.'
            first -= 1
        end
        while last<BOARD_SIZE && board[last+1,c]!='.'
            last += 1
        end
        first==last && continue
        prefix,suffix = uppercase(join(board[first:r-1,c])),uppercase(join(board[r+1:last,c]))
        result[(r,c)] = Set(l for l in LETTERS if has_word(trie,prefix*l*suffix))
    end
    result
end
function across_moves(board,rack,trie,anchor_squares,checks)
    result,counts = NamedTuple[],rack_counts(rack)
    for (row,anchor) in sort!(collect(anchor_squares))
        function extend(node,column,word,start)
            empty = column>BOARD_SIZE || board[row,column]=='.'
            empty && column>anchor && trie.terminal[node] && push!(
                result,
                (row=row,column=start,direction=0,word=word)
            )
            column>BOARD_SIZE && return
            if !empty
                letter = board[row,column]
                child = get(trie.children[node],uppercase(letter),0)
                child!=0 && extend(child,column+1,word*letter,start)
                return
            end
            allowed = get(checks,(row,column),nothing)
            for letter in sort!(collect(keys(trie.children[node])))
                !isnothing(allowed) && !(letter in allowed) && continue
                child = trie.children[node][letter]
                for (tile,written) in ((tile_index(letter),letter),(27,lowercase(letter)))
                    if counts[tile]>0
                        counts[tile] -= 1
                        extend(child,column+1,word*written,start)
                        counts[tile] += 1
                    end
                end
            end
        end
        function left_part(node,prefix,remaining)
            extend(node,anchor,prefix,anchor-length(prefix))
            remaining==0 && return
            for letter in sort!(collect(keys(trie.children[node])))
                for (tile,written) in ((tile_index(letter),letter),(27,lowercase(letter)))
                    if counts[tile]>0
                        counts[tile] -= 1
                        left_part(trie.children[node][letter],prefix*written,remaining-1)
                        counts[tile] += 1
                    end
                end
            end
        end
        if anchor>1 && board[row,anchor-1]!='.'
            first = anchor-1
            while first>1 && board[row,first-1]!='.'
                first -= 1
            end
            prefix = join(board[row,first:anchor-1])
            node = prefix_node(trie,uppercase(prefix))
            node!=0 && extend(node,anchor,prefix,first)
        else
            first = anchor-1
            while first>=1 && board[row,first]=='.' && !((row,first) in anchor_squares)
                first -= 1
            end
            left_part(1,"",anchor-first-1)
        end
    end
    result
end
placement_key(new_tiles) = Tuple(sort(collect(new_tiles)))
function generate_moves(board,rack,trie)
    anchor_squares = anchors(board)
    across = across_moves(board,rack,trie,anchor_squares,cross_checks(board,trie))
    transposed = transpose_board(board)
    down = across_moves(
        transposed,
        rack,
        trie,
        Set((c,r) for (r,c) in anchor_squares),
        cross_checks(transposed,trie)
    )
    candidates = [across;[(row=m.column,column=m.row,direction=1,word=m.word) for m in down]]
    result = Dict{Tuple,NamedTuple}()
    for move in candidates
        new_tiles = placement(board,move)
        if length(new_tiles)==1 && move.direction==1
            r,c,_ = new_tiles[1]
            length(run_at(apply_move(board,new_tiles),r,c,0).word)>=2 && continue
        end
        result[placement_key(new_tiles)] = move
    end
    collect(values(result))
end
const RACK_SIZE = 7
const BINGO_BONUS = 50
function move_cells(move)
    [(
        move.row+move.direction*(i-1),
        move.column+(1-move.direction)*(i-1),
        letter
    ) for (i,letter) in enumerate(move.word)]
end
placement(board,move) = [(r,c,l) for (r,c,l) in move_cells(move) if board[r,c]=='.']
function apply_move(board,new_tiles)
    result = copy(board)
    for (r,c,l) in new_tiles
        result[r,c] = l
    end
    result
end
function cross_words(board,move)
    new_tiles = placement(board,move)
    after = apply_move(board,new_tiles)
    result = NamedTuple[]
    for (r,c,_) in new_tiles
        word = run_at(after,r,c,1-move.direction)
        length(word.word)>=2 && push!(result,word)
    end
    result
end
function word_score(board,word,values)
    score,multiplier = 0,1
    for (r,c,l) in move_cells(word)
        value = isuppercase(l) ? values[tile_index(l)] : 0
        if board[r,c]=='.'
            premium = PREMIUMS[r][c]
            value *= premium=='+' ? 2 : premium=='^' ? 3 : 1
            multiplier *= premium=='*' ? 2 : premium=='#' ? 3 : 1
        end
        score += value
    end
    score*multiplier
end
function score_move(board,move,values)
    words = [move]
    append!(words,cross_words(board,move))
    scores = [word_score(board,w,values) for w in words]
    new_tiles = placement(board,move)
    (
        score=sum(scores)+(length(new_tiles)==RACK_SIZE ? BINGO_BONUS : 0),
        new_tiles=new_tiles,
        words=collect(zip(words,scores))
    )
end
const PREMIUMS = [
    "#..+...#...+..#",
    ".*...^...^...*.",
    "..*...+.+...*..",
    "+..*...+...*..+",
    "....*.....*....",
    ".^...^...^...^.",
    "..+...+.+...+..",
    "#..+...*...+..#",
    "..+...+.+...+..",
    ".^...^...^...^.",
    "....*.....*....",
    "+..*...+...*..+",
    "..*...+.+...*..",
    ".*...^...^...*.",
    "#..+...#...+..#"
]

const BOARD_SIZE = 15
const CENTER = (8,8)
const LETTERS = "ABCDEFGHIJKLMNOPQRSTUVWXYZ"
const COLUMNS = "ABCDEFGHIJKLMNO"
inside(row,column) = 1<=row<=BOARD_SIZE && 1<=column<=BOARD_SIZE
square_name(row,column) = "$(COLUMNS[column])$row"
move_position(move) = move.direction==0 ? "$(move.row)$(COLUMNS[move.column])" : "$(COLUMNS[move.column])$(move.row)"
transpose_board(board) = permutedims(board)
board_tiles(board) = [(r,c,board[r,c]) for r in 1:BOARD_SIZE for c in 1:BOARD_SIZE if board[r,c]!='.']
neighbors(row,column) = [(r,c) for (r,c) in (
    (row-1,column),
    (row+1,column),
    (row,column-1),
    (row,column+1)
) if inside(r,c)]
function run_at(board,row,column,direction)
    dr,dc = direction,1-direction
    while inside(row-dr,column-dc) && board[row-dr,column-dc]!='.'
        row,column = row-dr,column-dc
    end
    first_row,first_column,letters = row,column,Char[]
    while inside(row,column) && board[row,column]!='.'
        push!(letters,board[row,column])
        row,column = row+dr,column+dc
    end
    (row=first_row,column=first_column,direction=direction,word=join(letters))
end
const STRATEGIES = ("score","tiles","equity")
words(parts...) = join(string.(parts)," ")
attributes(items::Pair...) = join(["$(first(p))=$(last(p))" for p in items]," ")
valid_name(value) = occursin(r"^[a-z0-9_]{1,20}$",value)
function valid_number(value,minimum,maximum)
    occursin(r"^(0|[1-9][0-9]*)$",value) || return false
    n = tryparse(Int,value)
    !isnothing(n) && minimum<=n<=maximum
end
function remove_tiles(rack,removed)
    counts = rack_counts(rack)
    for letter in removed
        counts[tile_index(letter)] -= 1
    end
    join([repeat(string(l),counts[tile_index(l)]) for l in sort!(collect(TILE_ORDER))])
end
notation(e) = move_position(e.move)*" "*e.move.word
standard_key(e) = (-e.score,e.move.row,e.move.column,e.move.direction,e.move.word)
function strategy_key(e,strategy)
    prefix = strategy=="score" ? () : strategy=="tiles" ? (-length(e.new_tiles),) : (-e.score-e.value,)
    (prefix...,standard_key(e)...)
end
function read_inputs()
    values,counts,weights = zeros(Int,27),zeros(Int,27),zeros(Int,27)
    for line in readlines("input/tiles.txt")
        (isempty(line) || startswith(line,"#")) && continue
        f = split(line)
        i = tile_index(f[1][1])
        values[i],counts[i],weights[i] = parse.(Int,f[2:end])
    end
    trie,statistics,lengths = WordTrie(),Dict{String,Int}(),Dict{Int,Int}()
    for line in readlines("input/dictionary.txt")
        (isempty(line) || startswith(line,"#")) && continue
        bump!(statistics,"lines")
        if !occursin(r"^[A-Z]{2,15}$",line)
            bump!(statistics,"skipped")
        elseif add_word!(trie,line)
            lengths[length(line)] = get(lengths,length(line),0)+1
        else
            bump!(statistics,"duplicates")
        end
    end
    jobs,current = Vector{String}[],nothing
    for line in readlines("input/jobs.txt")
        if isnothing(current)
            !isempty(line) && !startswith(line,"#") && (current=[line])
        elseif line=="END"
            push!(jobs,current)
            current = nothing
        else
            push!(current,line)
        end
    end
    dictionary_lines = ["DICTIONARY "*attributes("lines"=>get(statistics,"lines",0),"words"=>trie.words,"skipped"=>get(statistics,"skipped",0),
        "duplicates"=>get(statistics,"duplicates",0),"nodes"=>length(trie.children)),
        "lengths"*join([" $n=$(lengths[n])" for n in sort!(collect(keys(lengths)))])]
    a = (
        ;values,
        counts,
        weights,
        trie,
        topn_totals=Dict{String,Int}(),
        validation_totals=Dict{String,Int}(),
        game_totals=Dict{String,Int}()
    )
    a,jobs,dictionary_lines
end
function job_error(a,job,seen)
    f = split(job[1])
    kind,name = f[1:2]
    !valid_name(name) && return "bad name"
    name in seen && return "duplicate name"
    if kind=="TOPN"
        !valid_number(f[3],1,99) && return "bad count"
        !(f[4] in STRATEGIES) && return "bad order"
        rack = split(job[2])[2]
        !valid_rack(rack) && return "bad rack"
        return board_error(a,job[4:end],rack)
    elseif kind=="VALIDATE"
        rows = String[]
        for line in job[3:end]
            startswith(line,"MOVE") && break
            push!(rows,line)
        end
        return board_error(a,rows,"")
    end
    !valid_number(f[3],1,2147483646) && return "bad seed"
    !(2<=length(job)-1<=4) && return "bad player count $(length(job)-1)"
    names = Set{String}()
    for (i,line) in enumerate(job[2:end])
        player = split(line)
        (!valid_name(player[2]) || !(player[3] in STRATEGIES) || (kind=="GAME" && !valid_number(
            player[4],
            0,
            99
        ))) && return "bad player $i"
        player[2] in names && return "duplicate player $i"
        push!(names,player[2])
    end
    nothing
end
function leave_value(a,rack)
    counts = rack_counts(rack)
    duplicates = sum(max(0,n-1) for n in counts[1:26])
    vowels = sum(counts[tile_index(l)] for l in VOWELS)
    consonants = sum(counts[i] for (i,l) in enumerate(LETTERS) if !(l in VOWELS))
    sum((a.weights[tile_index(l)] for l in rack);init=0)-4duplicates-3max(0,abs(vowels-consonants)-1)
end
function evaluate_move(a,board,rack,move)
    scored = score_move(board,move,a.values)
    leave = remove_tiles(rack,[tile_kind(l) for (_,_,l) in scored.new_tiles])
    word_list = join(["$(w.word):$s" for (w,s) in scored.words],',')
    (
        move=move,
        score=scored.score,
        new_tiles=scored.new_tiles,
        words=word_list,
        leave=leave,
        value=leave_value(a,leave)
    )
end
function legal_moves(a,board,rack)
    sort!([evaluate_move(a,board,rack,m) for m in generate_moves(board,rack,a.trie)];by=standard_key)
end
function duplicate_game(a,job)
    _,name,seed_text = split(job[1])
    seed = parse(Int,seed_text)
    random = Minstd(seed)
    bag,board = make_bag(a.counts,random),fill('.',BOARD_SIZE,BOARD_SIZE)
    rack = draw_tiles!(bag,RACK_SIZE)
    players = [String.(split(line)[2:end]) for line in job[2:end]]
    totals = zeros(Int,length(players))
    lines = [words("DUPLICATE",name,attributes("seed"=>seed,"players"=>length(players)))]
    for (i,p) in enumerate(players)
        push!(lines,words("player",i,p[1],"strategy="*p[2]))
    end
    round_number,master_total,reason = 0,0,"stuck"
    while true
        legal = legal_moves(a,board,rack)
        isempty(legal) && break
        round_number += 1
        master = legal[1]
        choices = [argmin(m->strategy_key(m,p[2]),legal) for p in players]
        for (i,m) in enumerate(choices)
            totals[i] += m.score
        end
        master_total += master.score
        push!(
            lines,
            words(
                "round",
                round_number,
                attributes("rack"=>collection(rack),"moves"=>length(legal)),
                "master",
                notation(master),
                attributes("score"=>master.score,"picks"=>join([m.score for m in choices],','))
            )
        )
        board = apply_move(board,master.new_tiles)
        rack = master.leave*draw_tiles!(bag,RACK_SIZE-length(master.leave))
        if isempty(rack)
            reason = "out"
            break
        end
    end
    push!(
        lines,
        "end "*attributes(
            "reason"=>reason,
            "rounds"=>round_number
        )*(reason=="stuck" ? " rack="*collection(rack) : "")
    )
    push!(lines,"master total=$master_total")
    for (p,total) in zip(players,totals)
        percent = master_total>=1 ? 100total÷master_total : 0
        push!(lines,words("final",p[1],attributes("total"=>total,"percent"=>percent)))
    end
    push!(lines,"winner "*join([p[1] for (p,total) in zip(players,totals) if total==maximum(totals)],','))
    push!(lines,position_summary(board))
    append!(lines,board_display(board))
    bump!(a.game_totals,"rounds",round_number)
    lines
end
function main()
    a,jobs,dictionary_lines = read_inputs()
    move_blocks,validation_blocks,game_blocks = [dictionary_lines],Vector{String}[],Vector{String}[]
    seen = Set{String}()
    for job in jobs
        kind,name = split(job[1])[1:2]
        if kind=="TOPN"
            destination,totals,key = move_blocks,a.topn_totals,"jobs"
        elseif kind=="VALIDATE"
            destination,totals,key = validation_blocks,a.validation_totals,"jobs"
        else
            destination,totals,key = game_blocks,a.game_totals,kind=="GAME" ? "games" : "duplicates"
        end
        bump!(totals,key)
        error = job_error(a,job,seen)
        push!(seen,name)
        if !isnothing(error)
            push!(destination,["$kind $name","error "*error])
            bump!(totals,"errors")
        elseif kind=="TOPN"
            push!(destination,topn(a,job))
        elseif kind=="VALIDATE"
            push!(destination,validate_job(a,job))
        elseif kind=="GAME"
            push!(destination,ordinary_game(a,job))
        else
            push!(destination,duplicate_game(a,job))
        end
    end
    summaries = [(move_blocks,a.topn_totals,("jobs","errors","moves","bingos")),
        (validation_blocks,a.validation_totals,("jobs","errors","moves","valid","invalid","loss")),
        (
            game_blocks,
            a.game_totals,
            ("games","duplicates","errors","turns","rounds","plays","exchanges","passes","bingos")
        )]
    mkpath("output")
    for (filename,(blocks,totals,keys)) in zip(("moves.txt","validation.txt","game.txt"),summaries)
        push!(blocks,["SUMMARY "*attributes((key=>get(totals,key,0) for key in keys)...)])
        write(joinpath("output",filename),join([join(block,"\n") for block in blocks],"\n\n")*"\n")
    end
end
