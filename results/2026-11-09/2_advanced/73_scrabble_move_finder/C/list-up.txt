static int add_node(Trie *trie) {
    if (trie->count == trie->capacity) {
        trie->capacity = trie->capacity ? 2 * trie->capacity : 128;
        trie->nodes = realloc(trie->nodes, (size_t)trie->capacity * sizeof *trie->nodes);
    }
    int index = trie->count++;
    for (int i = 0; i < 26; ++i)
        trie->nodes[index].children[i] = -1;
    trie->nodes[index].terminal = false;
    return index;
}
void trie_init(Trie *trie) {
    memset(trie, 0, sizeof *trie);
    add_node(trie);
}
bool trie_add(Trie *trie, const char *word) {
    int node = 0;
    for (const char *p = word; *p; ++p) {
        int letter = *p - 'A', child = trie->nodes[node].children[letter];
        if (child < 0) {
            child = add_node(trie);
            trie->nodes[node].children[letter] = child;
        }
        node = child;
    }
    if (trie->nodes[node].terminal)
        return false;
    trie->nodes[node].terminal = true;
    ++trie->words;
    return true;
}
int trie_prefix(const Trie *trie, const char *word) {
    int node = 0;
    while (*word) {
        node = trie->nodes[node].children[*word++ - 'A'];
        if (node < 0)
            break;
    }
    return node;
}
bool trie_contains(const Trie *trie, const char *word) {
    int node = trie_prefix(trie, word);
    return node >= 0 && trie->nodes[node].terminal;
}

int64_t random_next(Minstd *random) {
    random->state = 48271 * random->state % 2147483647;
    return random->state;
}
void shuffle(Bag *bag, Minstd *random) {
    for (int i = bag->count - 1; i >= 1; --i) {
        int j = (int)(random_next(random) % (i + 1));
        char tile = bag->tiles[i];
        bag->tiles[i] = bag->tiles[j];
        bag->tiles[j] = tile;
    }
}
Bag make_bag(const Counts counts, Minstd *random) {
    Bag out = {0};
    for (int tile = 0; tile < 27; ++tile)
        for (int i = 0; i < counts[tile]; ++i)
            out.tiles[out.count++] = tile_order[tile];
    shuffle(&out, random);
    return out;
}

    while (count-- > 0 && bag->count > 0)
        out[n++] = bag->tiles[--bag->count];
    out[n] = 0;
}

Anchors anchors(const Board *board) {
    Anchors out = {0};
    if (board_tiles(board).count == 0) {
        out.cells[7][7] = true;
        out.count = 1;
        return out;
    }
    for (int r = 0; r < 15; ++r)
        for (int c = 0; c < 15; ++c)
            if (board->cells[r][c] == '.') {
                int adjacent[4][2], n = neighbors(r, c, adjacent);
                for (int i = 0; i < n; ++i)
                    if (board->cells[adjacent[i][0]][adjacent[i][1]] != '.') {
                        out.cells[r][c] = true;
                        ++out.count;
                        break;
                    }
            }
    return out;
}
Checks cross_checks(const Board *board, const Trie *trie) {
    Checks out = {0};
    for (int r = 0; r < 15; ++r)
        for (int c = 0; c < 15; ++c) {
            if (board->cells[r][c] != '.')
                continue;
            int first = r, last = r;
            while (first > 0 && board->cells[first - 1][c] != '.')
                --first;
            while (last + 1 < 15 && board->cells[last + 1][c] != '.')
                ++last;
            if (first == last)
                continue;
            out.checked[r][c] = true;
            char word[16], capital[16];
            for (int letter = 0; letter < 26; ++letter) {
                int p = 0;
                for (int row = first; row <= last; ++row)
                    word[p++] = row == r ? (char)('A' + letter) : board->cells[row][c];
                word[p] = 0;
                uppercase(word, capital);
                if (trie_contains(trie, capital))
                    out.mask[r][c] |= UINT32_C(1) << letter;
            }
        }
    return out;
}
typedef struct {
    const Board *board;
    const Trie *trie;
    Anchors anchors;
    Checks checks;
    Counts counts;
    Moves *out;
    int row, anchor;
} Generator;
static void extend(Generator *g, int node, int column, const char *word, int start) {
    bool empty = column == 15 || g->board->cells[g->row][column] == '.';
    if (empty && column > g->anchor && g->trie->nodes[node].terminal) {
        Move move = {g->row, start, 0, {0}};
        strcpy(move.word, word);
        Moves_append(g->out, move);
    }
    if (column == 15)
        return;
    size_t length = strlen(word);
    if (!empty) {
        char written = g->board->cells[g->row][column],
             capital = lower(written) ? (char)(written - 'a' + 'A') : written;
        int child = g->trie->nodes[node].children[capital - 'A'];
        if (child >= 0) {
            char next[16];
            strcpy(next, word);
            next[length] = written;
            next[length + 1] = 0;
            extend(g, child, column + 1, next, start);
        }
        return;
    }
    for (int letter = 0; letter < 26; ++letter) {
        int child = g->trie->nodes[node].children[letter];
        if (child < 0 || (g->checks.checked[g->row][column] &&
                          !(g->checks.mask[g->row][column] & (UINT32_C(1) << letter))))
            continue;
        for (int blank = 0; blank < 2; ++blank) {
            int tile = blank ? 26 : letter;
            if (g->counts[tile]) {
                --g->counts[tile];
                char next[16];
                strcpy(next, word);
                next[length] = (char)((blank ? 'a' : 'A') + letter);
                next[length + 1] = 0;
                extend(g, child, column + 1, next, start);
                ++g->counts[tile];
            }
        }
    }
}
static void left_part(Generator *g, int node, const char *prefix, int remaining) {
    size_t length = strlen(prefix);
    extend(g, node, g->anchor, prefix, g->anchor - (int)length);
    if (!remaining)
        return;
    for (int letter = 0; letter < 26; ++letter) {
        int child = g->trie->nodes[node].children[letter];
        if (child < 0)
            continue;
        for (int blank = 0; blank < 2; ++blank) {
            int tile = blank ? 26 : letter;
            if (g->counts[tile]) {
                --g->counts[tile];
                char next[16];
                strcpy(next, prefix);
                next[length] = (char)((blank ? 'a' : 'A') + letter);
                next[length + 1] = 0;
                left_part(g, child, next, remaining - 1);
                ++g->counts[tile];
            }
        }
    }
}
static Moves across_moves(const Board *board, const char *rack, const Trie *trie) {
    Moves out = {0};
    Generator g = {0};
    g.board = board;
    g.trie = trie;
    g.anchors = anchors(board);
    g.checks = cross_checks(board, trie);
    g.out = &out;
    for (const char *p = rack; *p; ++p)
        ++g.counts[tile_index(*p)];
    for (int r = 0; r < 15; ++r)
        for (int anchor = 0; anchor < 15; ++anchor) {
            if (!g.anchors.cells[r][anchor])
                continue;
            g.row = r;
            g.anchor = anchor;
            if (anchor > 0 && board->cells[r][anchor - 1] != '.') {
                int first = anchor - 1;
                while (first > 0 && board->cells[r][first - 1] != '.')
                    --first;
                char prefix[16], capital[16];
                int p = 0;
                for (int c = first; c < anchor; ++c)
                    prefix[p++] = board->cells[r][c];
                prefix[p] = 0;
                uppercase(prefix, capital);
                int node = trie_prefix(trie, capital);
                if (node >= 0)
                    extend(&g, node, anchor, prefix, first);
            } else {
                int first = anchor - 1;
                while (first >= 0 && board->cells[r][first] == '.' && !g.anchors.cells[r][first])
                    --first;
                left_part(&g, 0, "", anchor - first - 1);
            }
        }
    return out;
}
Moves generate(const Board *board, const char *rack, const Trie *trie) {
    Moves out = across_moves(board, rack, trie);
    Board transposed = transpose(board);
    Moves down = across_moves(&transposed, rack, trie);
    for (size_t i = 0; i < down.count; ++i) {
        Move move = down.items[i];
        int r = move.column;
        move.column = move.row;
        move.row = r;
        move.direction = 1;
        Placement tiles = placement(board, &move);
        if (tiles.count == 1) {
            Board after = apply_tiles(board, &tiles);
            Move across = run_at(&after, tiles.items[0].row, tiles.items[0].column, 0);
            if (strlen(across.word) >= 2)
                continue;
        }
        Moves_append(&out, move);
    }
    free(down.items);
    return out;
}
Placement squares(const Move *move) {
    Placement out = {0};
    for (int i = 0; move->word[i]; ++i)
        out.items[out.count++] = (Tile){move->row + move->direction * i,
                                        move->column + (1 - move->direction) * i, move->word[i]};
    return out;
}
Placement placement(const Board *board, const Move *move) {
    Placement cells = squares(move), out = {0};
    for (int i = 0; i < cells.count; ++i)
        if (board->cells[cells.items[i].row][cells.items[i].column] == '.')
            out.items[out.count++] = cells.items[i];
    return out;
}
Board apply_tiles(const Board *board, const Placement *tiles) {
    Board out = *board;
    for (int i = 0; i < tiles->count; ++i) {
        const Tile *tile = &tiles->items[i];
        out.cells[tile->row][tile->column] = tile->letter;
    }
    return out;
}
Words cross_words(const Board *board, const Move *move) {
    Placement tiles = placement(board, move);
    Board after = apply_tiles(board, &tiles);
    Words out = {0};
    for (int i = 0; i < tiles.count; ++i) {
        Move word = run_at(&after, tiles.items[i].row, tiles.items[i].column, 1 - move->direction);
        if (strlen(word.word) >= 2)
            out.items[out.count++] = word;
    }
    return out;
}
int word_score(const Board *board, const Move *word, const Counts values) {
    Placement cells = squares(word);
    int score = 0, multiplier = 1;
    for (int i = 0; i < cells.count; ++i) {
        Tile tile = cells.items[i];
        int value = upper(tile.letter) ? values[tile.letter - 'A'] : 0;
        if (board->cells[tile.row][tile.column] == '.') {
            char premium = premiums[tile.row][tile.column];
            value *= premium == '+' ? 2 : premium == '^' ? 3 : 1;
            multiplier *= premium == '*' ? 2 : premium == '#' ? 3 : 1;
        }
        score += value;
    }
    return score * multiplier;
}
Scored score_move(const Board *board, const Move *move, const Counts values) {
    Scored out = {0};
    out.tiles = placement(board, move);
    out.words[out.count++] = (WordScore){*move, word_score(board, move, values)};
    Words crosses = cross_words(board, move);
    for (int i = 0; i < crosses.count; ++i)
        out.words[out.count++] =
            (WordScore){crosses.items[i], word_score(board, &crosses.items[i], values)};
    for (int i = 0; i < out.count; ++i)
        out.score += out.words[i].score;
    if (out.tiles.count == RACK_SIZE)
        out.score += BINGO_BONUS;
    return out;
}

const char tile_order[28] = "ABCDEFGHIJKLMNOPQRSTUVWXYZ?";
bool upper(char c) { return c >= 'A' && c <= 'Z'; }
bool lower(char c) { return c >= 'a' && c <= 'z'; }
int tile_index(char c) { return upper(c) ? c - 'A' : 26; }
void uppercase(const char *word, char *out) {
    while (*word) {
        char c = *word++;
        *out++ = lower(c) ? (char)(c - 'a' + 'A') : c;
    }
    *out = 0;
}
bool inside(int r, int c) { return r >= 0 && r < 15 && c >= 0 && c < 15; }
const char *square_name(int r, int c) {
    static char buffers[16][8];
    static unsigned index;
    char *out = buffers[index++ % 16];
    snprintf(out, 8, "%c%d", 'A' + c, r + 1);
    return out;
}
const char *position(const Move *m) {
    static char buffers[16][8];
    static unsigned index;
    char *out = buffers[index++ % 16];
    if (m->direction)
        snprintf(out, 8, "%c%d", 'A' + m->column, m->row + 1);
    else
        snprintf(out, 8, "%d%c", m->row + 1, 'A' + m->column);
    return out;
}
Board empty_board(void) {
    Board out;
    memset(out.cells, '.', sizeof out.cells);
    return out;
}
Board transpose(const Board *board) {
    Board out;
    for (int r = 0; r < 15; ++r)
        for (int c = 0; c < 15; ++c)
            out.cells[r][c] = board->cells[c][r];
    return out;
}
Placement transpose_tiles(const Placement *tiles) {
    Placement out = {0};
    for (int i = 0; i < tiles->count; ++i)
        out.items[out.count++] =
            (Tile){tiles->items[i].column, tiles->items[i].row, tiles->items[i].letter};
    return out;
}
BoardTiles board_tiles(const Board *board) {
    BoardTiles out = {0};
    for (int r = 0; r < 15; ++r)
        for (int c = 0; c < 15; ++c)
            if (board->cells[r][c] != '.')
                out.items[out.count++] = (Tile){r, c, board->cells[r][c]};
    return out;
}
int neighbors(int r, int c, int out[4][2]) {
    const int directions[4][2] = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}};
    int count = 0;
    for (int i = 0; i < 4; ++i)
        if (inside(r + directions[i][0], c + directions[i][1])) {
            out[count][0] = r + directions[i][0];
            out[count++][1] = c + directions[i][1];
        }
    return count;
}
Move run_at(const Board *board, int r, int c, int direction) {
    int dr = direction, dc = 1 - direction;
    while (inside(r - dr, c - dc) && board->cells[r - dr][c - dc] != '.') {
        r -= dr;
        c -= dc;
    }
    Move out = {r, c, direction, {0}};
    int count = 0;
    while (inside(r, c) && board->cells[r][c] != '.') {
        out.word[count++] = board->cells[r][c];
        r += dr;
        c += dc;
    }
    out.word[count] = 0;
    return out;
}
static Tokens split(const char *text) {
    Tokens out = {0};
    while (*text) {
        while (*text == ' ')
            ++text;
        if (!*text)
            break;
        int p = 0;
        while (*text && *text != ' ')
            out.items[out.count][p++] = *text++;
        out.items[out.count++][p] = 0;
    }
    return out;
}
static bool valid_name(const char *text) {
    size_t n = strlen(text);
    if (n == 0 || n > 20)
        return false;
    for (const char *p = text; *p; ++p)
        if (!lower(*p) && !(*p >= '0' && *p <= '9') && *p != '_')
            return false;
    return true;
}
static bool valid_number(const char *text, int minimum, int maximum) {
    if (!*text || (text[0] == '0' && text[1]))
        return false;
    int64_t number = 0;
    for (const char *p = text; *p; ++p) {
        if (*p < '0' || *p > '9')
            return false;
        number = number * 10 + *p - '0';
        if (number > maximum)
            return false;
    }
    return number >= minimum;
}
static bool valid_strategy(const char *text) {
    return !strcmp(text, "score") || !strcmp(text, "tiles") || !strcmp(text, "equity");
}
static Strategy strategy(const char *text) {
    return !strcmp(text, "tiles") ? TILES : !strcmp(text, "equity") ? EQUITY : SCORE;
}
static const char *collection(const char *text) {
    static char buffers[64][201];
    static unsigned index;
    char *out = buffers[index++ % 64];
    int counts[128] = {0}, n = 0;
    while (*text)
        ++counts[(unsigned char)*text++];
    for (int c = 0; c < 128; ++c)
        for (int i = 0; i < counts[c]; ++i)
            out[n++] = (char)c;
    if (n == 0)
        out[n++] = '-';
    out[n] = 0;
    return out;
}
static void remove_tiles(const char *rack, const char *removed, char *out) {
    int counts[128] = {0};
    while (*rack)
        ++counts[(unsigned char)*rack++];
    while (*removed)
        --counts[(unsigned char)*removed++];
    for (int c = 0; c < 128; ++c)
        for (int i = 0; i < counts[c]; ++i)
            *out++ = (char)c;
    *out = 0;
}
static const char *notation(const Evaluated *move) {
    static char buffers[16][64];
    static unsigned index;
    char *out = buffers[index++ % 16];
    snprintf(out, 64, "%s %s", position(&move->move), move->move.word);
    return out;
}
static int standard_compare(const void *left, const void *right) {
    const Evaluated *a = left, *b = right;
    if (a->score != b->score)
        return a->score > b->score ? -1 : 1;
    if (a->move.row != b->move.row)
        return a->move.row - b->move.row;
    if (a->move.column != b->move.column)
        return a->move.column - b->move.column;
    if (a->move.direction != b->move.direction)
        return a->move.direction - b->move.direction;
    return strcmp(a->move.word, b->move.word);
}
static bool strategy_before(const Evaluated *a, const Evaluated *b, Strategy order) {
    if (order == TILES && a->tiles.count != b->tiles.count)
        return a->tiles.count > b->tiles.count;
    if (order == EQUITY && a->score + a->value != b->score + b->value)
        return a->score + a->value > b->score + b->value;
    return standard_compare(a, b) < 0;
}
static int choose(const Evaluations *moves, Strategy order) {
    int best = 0;
    for (size_t i = 1; i < moves->count; ++i)
        if (strategy_before(&moves->items[i], &moves->items[best], order))
            best = (int)i;
    return best;
}
static bool job_error(const Program *p, const Lines *job, const Lines *seen, char *reason) {
    Tokens header = split(job->items[0]);
    const char *kind = header.items[0], *name = header.items[1];
    if (!valid_name(name)) {
        strcpy(reason, "bad name");
        return true;
    }
    for (size_t i = 0; i < seen->count; ++i)
        if (!strcmp(name, seen->items[i])) {
            strcpy(reason, "duplicate name");
            return true;
        }
    if (!strcmp(kind, "TOPN")) {
        if (!valid_number(header.items[2], 1, 99)) {
            strcpy(reason, "bad count");
            return true;
        }
        if (!valid_strategy(header.items[3])) {
            strcpy(reason, "bad order");
            return true;
        }
        Tokens row = split(job->items[1]);
        if (!valid_rack(row.items[1])) {
            strcpy(reason, "bad rack");
            return true;
        }
        return board_error(p, job, row.items[1], reason);
    }
    if (!strcmp(kind, "VALIDATE"))
        return board_error(p, job, "", reason);
    if (!valid_number(header.items[2], 1, 2147483646)) {
        strcpy(reason, "bad seed");
        return true;
    }
    if (job->count - 1 < 2 || job->count - 1 > 4) {
        snprintf(reason, 128, "bad player count %zu", job->count - 1);
        return true;
    }
    char names[4][21];
    for (size_t i = 1; i < job->count; ++i) {
        Tokens row = split(job->items[i]);
        if (!valid_name(row.items[1]) || !valid_strategy(row.items[2]) ||
            (!strcmp(kind, "GAME") && !valid_number(row.items[3], 0, 99))) {
            snprintf(reason, 128, "bad player %zu", i);
            return true;
        }
        for (size_t j = 0; j + 1 < i; ++j)
            if (!strcmp(names[j], row.items[1])) {
                snprintf(reason, 128, "duplicate player %zu", i);
                return true;
            }
        strcpy(names[i - 1], row.items[1]);
    }
    return false;
}
static int leave_value(const Program *p, const char *rack) {
    Counts counts = {0};
    int total = 0, vowels = 0, consonants = 0, duplicates = 0;
    for (const char *c = rack; *c; ++c) {
        int tile = tile_index(*c);
        ++counts[tile];
        total += p->weights[tile];
        if (*c != '?') {
            if (strchr("AEIOU", *c))
                ++vowels;
            else
                ++consonants;
        }
    }
    for (int i = 0; i < 26; ++i)
        if (counts[i] > 1)
            duplicates += counts[i] - 1;
    int imbalance = abs(vowels - consonants) - 1;
    if (imbalance < 0)
        imbalance = 0;
    return total - 4 * duplicates - 3 * imbalance;
}
static Evaluated evaluate(const Program *p, const Board *board, const char *rack,
                          const Move *move) {
    Scored scored = score_move(board, move, p->values);
    Evaluated out = {0};
    out.move = *move;
    out.score = scored.score;
    out.tiles = scored.tiles;
    char removed[16];
    for (int i = 0; i < scored.tiles.count; ++i)
        removed[i] = upper(scored.tiles.items[i].letter) ? scored.tiles.items[i].letter : '?';
    removed[scored.tiles.count] = 0;
    remove_tiles(rack, removed, out.leave);
    out.value = leave_value(p, out.leave);
    size_t offset = 0;
    for (int i = 0; i < scored.count; ++i) {
        offset += (size_t)snprintf(out.words + offset, sizeof out.words - offset, "%s%s:%d",
                                   i ? "," : "", scored.words[i].word.word, scored.words[i].score);
    }
    return out;
}
static Evaluations legal_moves(const Program *p, const Board *board, const char *rack) {
    Moves generated = generate(board, rack, &p->trie);
    Evaluations out = {0};
    for (size_t i = 0; i < generated.count; ++i)
        Evaluations_append(&out, evaluate(p, board, rack, &generated.items[i]));
    free(generated.items);
    if (out.count > 1)
        qsort(out.items, out.count, sizeof *out.items, standard_compare);
    return out;
}
static Lines duplicate_game(Program *p, const Lines *job) {
    Tokens header = split(job->items[0]);
    Minstd random = {strtoll(header.items[2], NULL, 10)};
    Bag bag = make_bag(p->counts, &random);
    Board board = empty_board();
    char rack[8];
    draw(&bag, 7, rack);
    GamePlayer players[4] = {0};
    int count = (int)job->count - 1;
    Lines out = {0};
    add_line(&out, "DUPLICATE %s seed=%s players=%d", header.items[1], header.items[2], count);
    for (int i = 0; i < count; ++i) {
        Tokens row = split(job->items[i + 1]);
        strcpy(players[i].name, row.items[1]);
        strcpy(players[i].strategy_name, row.items[2]);
        players[i].strategy = strategy(row.items[2]);
        add_line(&out, "player %d %s strategy=%s", i + 1, players[i].name,
                 players[i].strategy_name);
    }
    int round = 0, master_total = 0;
    bool went_out = false;
    for (;;) {
        Evaluations legal = legal_moves(p, &board, rack);
        if (legal.count == 0) {
            free(legal.items);
            break;
        }
        ++round;
        Evaluated master = legal.items[0];
        Text text = {0};
        text_add(&text, "round %d rack=%s moves=%zu master %s score=%d picks=", round,
                 collection(rack), legal.count, notation(&master), master.score);
        for (int i = 0; i < count; ++i) {
            int score = legal.items[choose(&legal, players[i].strategy)].score;
            players[i].score += score;
            text_add(&text, "%s%d", i ? "," : "", score);
        }
        finish(&out, &text);
        master_total += master.score;
        board = apply_tiles(&board, &master.tiles);
        char drawn[8];
        draw(&bag, 7 - (int)strlen(master.leave), drawn);
        snprintf(rack, sizeof rack, "%s%s", master.leave, drawn);
        free(legal.items);
        if (rack[0] == 0) {
            went_out = true;
            break;
        }
    }
    if (went_out)
        add_line(&out, "end reason=out rounds=%d", round);
    else
        add_line(&out, "end reason=stuck rounds=%d rack=%s", round, collection(rack));
    add_line(&out, "master total=%d", master_total);
    int maximum = players[0].score;
    for (int i = 0; i < count; ++i) {
        int percent = master_total >= 1 ? 100 * players[i].score / master_total : 0;
        add_line(&out, "final %s total=%d percent=%d", players[i].name, players[i].score, percent);
        if (players[i].score > maximum)
            maximum = players[i].score;
    }
    Text text = {0};
    text_add(&text, "winner ");
    bool separator = false;
    for (int i = 0; i < count; ++i)
        if (players[i].score == maximum) {
            text_add(&text, "%s%s", separator ? "," : "", players[i].name);
            separator = true;
        }
    finish(&out, &text);
    append_board(&out, &board);
    p->game.rounds += round;
    return out;
}
static void read_inputs(Program *p) {
    trie_init(&p->trie);
    FILE *file = fopen("input/tiles.txt", "r");
    char *text;
    while ((text = read_line(file))) {
        if (text[0] && text[0] != '#') {
            Tokens row = split(text);
            int tile = tile_index(row.items[0][0]);
            p->values[tile] = atoi(row.items[1]);
            p->counts[tile] = atoi(row.items[2]);
            p->weights[tile] = atoi(row.items[3]);
        }
        free(text);
    }
    fclose(file);
    file = fopen("input/dictionary.txt", "r");
    int candidates = 0, skipped = 0, duplicates = 0, lengths[16] = {0};
    while ((text = read_line(file))) {
        if (text[0] && text[0] != '#') {
            ++candidates;
            size_t length = strlen(text);
            bool good = length >= 2 && length <= 15;
            for (const char *c = text; *c; ++c)
                if (!upper(*c))
                    good = false;
            if (!good)
                ++skipped;
            else if (trie_add(&p->trie, text))
                ++lengths[length];
            else
                ++duplicates;
        }
        free(text);
    }
    fclose(file);
    add_line(&p->dictionary_lines, "DICTIONARY lines=%d words=%d skipped=%d duplicates=%d nodes=%d",
             candidates, p->trie.words, skipped, duplicates, p->trie.count);
    Text description = {0};
    text_add(&description, "lengths");
    for (int length = 2; length <= 15; ++length)
        if (lengths[length])
            text_add(&description, " %d=%d", length, lengths[length]);
    finish(&p->dictionary_lines, &description);
    file = fopen("input/jobs.txt", "r");
    Lines current = {0};
    while ((text = read_line(file))) {
        if (current.count == 0) {
            if (text[0] && text[0] != '#')
                Lines_append(&current, text);
            else
                free(text);
        } else if (!strcmp(text, "END")) {
            Blocks_append(&p->jobs, current);
            current = (Lines){0};
            free(text);
        } else
            Lines_append(&current, text);
    }
    fclose(file);
}
static void write_blocks(const char *name, const Blocks *blocks) {
    char path[128];
    snprintf(path, sizeof path, "output/%s", name);
    FILE *file = fopen(path, "w");
    for (size_t i = 0; i < blocks->count; ++i) {
        if (i)
            fputc('\n', file);
        for (size_t j = 0; j < blocks->items[i].count; ++j)
            fprintf(file, "%s\n", blocks->items[i].items[j]);
    }
    fclose(file);
}
int main(void) {
    Program *p = calloc(1, sizeof *p);
    read_inputs(p);
    Blocks move_blocks = {0}, validation_blocks = {0}, game_blocks = {0};
    Blocks_append(&move_blocks, p->dictionary_lines);
    p->dictionary_lines = (Lines){0};
    Lines seen = {0};
    for (size_t i = 0; i < p->jobs.count; ++i) {
        const Lines *job = &p->jobs.items[i];
        Tokens header = split(job->items[0]);
        const char *kind = header.items[0], *name = header.items[1];
        Blocks *destination;
        if (!strcmp(kind, "TOPN")) {
            destination = &move_blocks;
            ++p->topn.jobs;
        } else if (!strcmp(kind, "VALIDATE")) {
            destination = &validation_blocks;
            ++p->validation.jobs;
        } else {
            destination = &game_blocks;
            if (!strcmp(kind, "GAME"))
                ++p->game.games;
            else
                ++p->game.duplicates;
        }
        char reason[128];
        bool bad = job_error(p, job, &seen, reason);
        add_line(&seen, "%s", name);
        Lines result = {0};
        if (bad) {
            add_line(&result, "%s %s", kind, name);
            add_line(&result, "error %s", reason);
            if (!strcmp(kind, "TOPN"))
                ++p->topn.errors;
            else if (!strcmp(kind, "VALIDATE"))
                ++p->validation.errors;
            else
                ++p->game.errors;
        } else if (!strcmp(kind, "TOPN"))
            result = topn(p, job);
        else if (!strcmp(kind, "VALIDATE"))
            result = validate(p, job);
        else if (!strcmp(kind, "GAME"))
            result = ordinary_game(p, job);
        else
            result = duplicate_game(p, job);
        Blocks_append(destination, result);
    }
    Lines summary = {0};
    add_line(&summary, "SUMMARY jobs=%d errors=%d moves=%d bingos=%d", p->topn.jobs, p->topn.errors,
             p->topn.moves, p->topn.bingos);
    Blocks_append(&move_blocks, summary);
    summary = (Lines){0};
    add_line(&summary, "SUMMARY jobs=%d errors=%d moves=%d valid=%d invalid=%d loss=%d",
             p->validation.jobs, p->validation.errors, p->validation.moves, p->validation.valid,
             p->validation.invalid, p->validation.loss);
    Blocks_append(&validation_blocks, summary);
    summary = (Lines){0};
    add_line(
        &summary,
        "SUMMARY games=%d duplicates=%d errors=%d turns=%d rounds=%d plays=%d exchanges=%d passes=%d bingos=%d",
        p->game.games, p->game.duplicates, p->game.errors, p->game.turns, p->game.rounds,
        p->game.plays, p->game.exchanges, p->game.passes, p->game.bingos);
    Blocks_append(&game_blocks, summary);
    mkdir("output", 0777);
    write_blocks("moves.txt", &move_blocks);
    write_blocks("validation.txt", &validation_blocks);
    write_blocks("game.txt", &game_blocks);
    Blocks *reports[3] = {&move_blocks, &validation_blocks, &game_blocks};
    for (int i = 0; i < 3; ++i) {
        for (size_t j = 0; j < reports[i]->count; ++j)
            free_lines(&reports[i]->items[j]);
        free(reports[i]->items);
    }
    free_lines(&seen);
    for (size_t i = 0; i < p->jobs.count; ++i)
        free_lines(&p->jobs.items[i]);
    free(p->jobs.items);
    free(p->trie.nodes);
    free(p);
    return 0;
}
