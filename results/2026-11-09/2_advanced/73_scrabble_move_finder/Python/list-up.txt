
from collections import Counter


class Node:
    def __init__(self):
        self.children = {}
        self.terminal = False


class Trie:
    def __init__(self):
        self.root = Node()
        self.nodes, self.words = 1, 0

    def add(self, word):
        node = self.root
        for letter in word:
            if letter not in node.children:
                node.children[letter] = Node()
                self.nodes += 1
            node = node.children[letter]
        if node.terminal:
            return False
        node.terminal = True
        self.words += 1
        return True

    def prefix(self, word):
        node = self.root
        for letter in word:
            node = node.children.get(letter)
            if node is None:
                break
        return node

    def __contains__(self, word):
        node = self.prefix(word)
        return node is not None and node.terminal



TILE_ORDER = "ABCDEFGHIJKLMNOPQRSTUVWXYZ?"


class Minstd:
    def __init__(self, seed):
        self.state = seed

    def next(self):
        self.state = self.state * 48271 % 2147483647
        return self.state


def shuffle(bag, random):
    for position in range(len(bag) - 1, 0, -1):
        other = random.next() % (position + 1)
        bag[position], bag[other] = bag[other], bag[position]


def make_bag(counts, random):
    bag = [letter for letter in TILE_ORDER for _ in range(counts[letter])]
    shuffle(bag, random)
    return bag


def draw(bag, count):
    return "".join(bag.pop() for _ in range(min(count, len(bag))))


LETTERS = "ABCDEFGHIJKLMNOPQRSTUVWXYZ"


def anchors(board):
    if not tiles(board):
        return {CENTER}
    return {(row, column) for row in range(SIZE) for column in range(SIZE)
            if board[row][column] == "." and any(board[r][c] != "." for r, c in neighbors(row, column))}


def cross_checks(board, trie):
    result = {}
    for row in range(SIZE):
        for column in range(SIZE):
            if board[row][column] != ".":
                continue
            first, last = row, row
            while first > 0 and board[first - 1][column] != ".":
                first -= 1
            while last + 1 < SIZE and board[last + 1][column] != ".":
                last += 1
            if first == last:
                continue
            prefix = "".join(board[r][column] for r in range(first, row)).upper()
            suffix = "".join(board[r][column] for r in range(row + 1, last + 1)).upper()
            result[row, column] = {letter for letter in LETTERS if prefix + letter + suffix in trie}
    return result


def across_moves(board, rack, trie, anchor_squares, checks):
    result, counts = [], Counter(rack)
    for row, anchor in sorted(anchor_squares):
        def extend(node, column, word, start):
            empty = column == SIZE or board[row][column] == "."
            if empty and column > anchor and node.terminal:
                result.append(Move(row, start, 0, word))
            if column == SIZE:
                return
            if not empty:
                letter = board[row][column]
                child = node.children.get(letter.upper())
                if child is not None:
                    extend(child, column + 1, word + letter, start)
                return
            allowed = checks.get((row, column))
            for letter in sorted(node.children):
                if allowed is not None and letter not in allowed:
                    continue
                child = node.children[letter]
                for tile, written in ((letter, letter), ("?", letter.lower())):
                    if counts[tile]:
                        counts[tile] -= 1
                        extend(child, column + 1, word + written, start)
                        counts[tile] += 1

        def left_part(node, prefix, remaining):
            extend(node, anchor, prefix, anchor - len(prefix))
            if remaining == 0:
                return
            for letter in sorted(node.children):
                for tile, written in ((letter, letter), ("?", letter.lower())):
                    if counts[tile]:
                        counts[tile] -= 1
                        left_part(node.children[letter], prefix + written, remaining - 1)
                        counts[tile] += 1

        if anchor > 0 and board[row][anchor - 1] != ".":
            first = anchor - 1
            while first > 0 and board[row][first - 1] != ".":
                first -= 1
            prefix = "".join(board[row][first:anchor])
            node = trie.prefix(prefix.upper())
            if node is not None:
                extend(node, anchor, prefix, first)
        else:
            first = anchor - 1
            while first >= 0 and board[row][first] == "." and (row, first) not in anchor_squares:
                first -= 1
            left_part(trie.root, "", anchor - first - 1)
    return result


def generate(board, rack, trie):
    anchor_squares = anchors(board)
    across = across_moves(board, rack, trie, anchor_squares, cross_checks(board, trie))
    transposed = transpose(board)
    down = across_moves(transposed, rack, trie, {(column, row) for row, column in anchor_squares},
                        cross_checks(transposed, trie))
    result = {}
    for move in across + [Move(move.column, move.row, 1, move.word) for move in down]:
        new_tiles = placement(board, move)
        if len(new_tiles) == 1 and move.direction == 1:
            row, column, _ = new_tiles[0]
            if len(run_at(apply(board, new_tiles), row, column, 0).word) >= 2:
                continue
        result[tuple(sorted(new_tiles))] = move
    return list(result.values())
RACK_SIZE, BINGO_BONUS = 7, 50


def squares(move):
    return [(move.row + move.direction * offset, move.column + (1 - move.direction) * offset, letter)
            for offset, letter in enumerate(move.word)]


def placement(board, move):
    return [(row, column, letter) for row, column, letter in squares(move) if board[row][column] == "."]


def apply(board, new_tiles):
    result = [row[:] for row in board]
    for row, column, letter in new_tiles:
        result[row][column] = letter
    return result


def cross_words(board, move):
    new_tiles = placement(board, move)
    after = apply(board, new_tiles)
    result = []
    for row, column, _ in new_tiles:
        word = run_at(after, row, column, 1 - move.direction)
        if len(word.word) >= 2:
            result.append(word)
    return result


def word_score(board, word, values):
    score, multiplier = 0, 1
    for row, column, letter in squares(word):
        value = values[letter] if letter.isupper() else 0
        if board[row][column] == ".":
            premium = PREMIUMS[row][column]
            value *= 2 if premium == "+" else 3 if premium == "^" else 1
            multiplier *= 2 if premium == "*" else 3 if premium == "#" else 1
        score += value
    return score * multiplier


def score_move(board, move, values):
    words = [move] + cross_words(board, move)
    scores = [word_score(board, word, values) for word in words]
    new_tiles = placement(board, move)
    score = sum(scores) + (BINGO_BONUS if len(new_tiles) == RACK_SIZE else 0)
    return score, new_tiles, list(zip(words, scores))

def inside(row, column):
    return 0 <= row < SIZE and 0 <= column < SIZE
def square_name(row, column):
    return f"{COLUMNS[column]}{row + 1}"
def position(move):
    return f"{COLUMNS[move.column]}{move.row + 1}" if move.direction else f"{move.row + 1}{COLUMNS[move.column]}"
def transpose(board):
    return [list(row) for row in zip(*board)]
def transpose_tiles(tiles):
    return [(column, row, letter) for row, column, letter in tiles]
def tiles(board):
    return [(row, column, letter) for row, values in enumerate(board)
            for column, letter in enumerate(values) if letter != "."]
def neighbors(row, column):
    return [(r, c) for r, c in ((row - 1, column), (row + 1, column), (row, column - 1), (row, column + 1))
            if inside(r, c)]
def run_at(board, row, column, direction):
    dr, dc = direction, 1 - direction
    while inside(row - dr, column - dc) and board[row - dr][column - dc] != ".":
        row, column = row - dr, column - dc
    first_row, first_column, letters = row, column, []
    while inside(row, column) and board[row][column] != ".":
        letters.append(board[row][column])
        row, column = row + dr, column + dc
    return Move(first_row, first_column, direction, "".join(letters))

def valid_name(value):

def valid_number(value, minimum, maximum):
    return re.fullmatch(r"0|[1-9][0-9]*", value) is not None and minimum <= int(value) <= maximum
def collection(value):
    return "".join(sorted(value)) or "-"
def tile_kind(letter):
    return letter if letter.isupper() else "?"
def remove_tiles(rack, removed):
    counts = Counter(rack)
    counts.subtract(removed)
    return "".join(letter * counts[letter] for letter in sorted(counts))
def notation(evaluated):
    return position(evaluated.move) + " " + evaluated.move.word
def standard_key(evaluated):
    move = evaluated.move
    return -evaluated.score, move.row, move.column, move.direction, move.word
def strategy_key(evaluated, strategy):
    prefix = () if strategy == "score" else (-len(evaluated.new_tiles),) if strategy == "tiles" else (-evaluated.score - evaluated.value,)
    return prefix + standard_key(evaluated)
def read_inputs():

    for line in Path("input/tiles.txt").read_text(encoding="ascii").splitlines():

            letter, value, count, weight = line.split()

    trie, dictionary_counts, lengths = Trie(), Counter(), Counter()

        if not line or line.startswith("#"):

            dictionary_counts["duplicates"] += 1
    jobs, current = [], None

        elif line == "END":
            jobs.append(current)
            current = None
        else:

    dictionary_lines = [f"DICTIONARY lines={dictionary_counts['lines']} words={trie.words} "
                        f"skipped={dictionary_counts['skipped']} duplicates={dictionary_counts['duplicates']} nodes={trie.nodes}",
                        "lengths" + "".join(f" {length}={lengths[length]}" for length in sorted(lengths))]
    return values, counts, weights, trie, jobs, dictionary_lines
    def __init__(self, values, counts, weights, trie):
        self.values, self.counts, self.weights, self.trie = values, counts, weights, trie
        self.topn_totals, self.validation_totals, self.game_totals = Counter(), Counter(), Counter()
    def job_error(self, job, seen):
        fields = job[0].split()
        kind, name = fields[:2]
        if not valid_name(name):
            return "bad name"
        if name in seen:
            return "duplicate name"
        if kind == "TOPN":
            if not valid_number(fields[2], 1, 99):
                return "bad count"
            if fields[3] not in STRATEGIES:
                return "bad order"
            rack = job[1].split()[1]
            if not valid_rack(rack):
                return "bad rack"
            return self.board_error([list(row) for row in job[3:]], rack)
        if kind == "VALIDATE":
            rows = []
            for line in job[2:]:
                if line.startswith("MOVE"):
                    break
                rows.append(list(line))
            return self.board_error(rows, "")
        if not valid_number(fields[2], 1, 2147483646):
            return "bad seed"
        if not 2 <= len(job) - 1 <= 4:
            return f"bad player count {len(job) - 1}"
        names = set()
        for number, line in enumerate(job[1:], 1):
            player = line.split()
            if not valid_name(player[1]) or player[2] not in STRATEGIES:
                return f"bad player {number}"
            if kind == "GAME" and not valid_number(player[3], 0, 99):
                return f"bad player {number}"
            if player[1] in names:
                return f"duplicate player {number}"
            names.add(player[1])
        return None
    def leave_value(self, rack):
        counts = Counter(rack)
        duplicates = sum(max(0, counts[letter] - 1) for letter in LETTERS)
        vowels = sum(counts[letter] for letter in VOWELS)
        consonants = sum(counts[letter] for letter in LETTERS if letter not in VOWELS)
        return sum(self.weights[letter] for letter in rack) - 4 * duplicates - 3 * max(0, abs(vowels - consonants) - 1)
    def evaluate(self, board, rack, move):
        score, new_tiles, word_scores = score_move(board, move, self.values)
        leave = remove_tiles(rack, (tile_kind(letter) for _, _, letter in new_tiles))
        word_list = ",".join(f"{word.word}:{value}" for word, value in word_scores)
        return Evaluated(move, score, new_tiles, word_list, leave, self.leave_value(leave))
    def legal_moves(self, board, rack):
        return sorted((self.evaluate(board, rack, move) for move in generate(board, rack, self.trie)), key=standard_key)
    def duplicate_game(self, job):
        _, name, seed = job[0].split()
        random = Minstd(int(seed))
        bag = make_bag(self.counts, random)
        board = [["."] * SIZE for _ in range(SIZE)]
        rack = draw(bag, RACK_SIZE)
        players = [line.split()[1:] for line in job[1:]]
        totals = [0] * len(players)
        lines = [f"DUPLICATE {name} seed={int(seed)} players={len(players)}"]
        for number, (player, strategy) in enumerate(players, 1):
            lines.append(f"player {number} {player} strategy={strategy}")
        round_number, master_total, reason = 0, 0, "stuck"
        while True:
            legal = self.legal_moves(board, rack)
            if not legal:
                break
            round_number += 1
            master = legal[0]
            choices = [min(legal, key=lambda move: strategy_key(move, strategy)) for _, strategy in players]
            for index, move in enumerate(choices):
                totals[index] += move.score
            master_total += master.score
            picks = ",".join(str(move.score) for move in choices)
            lines.append(f"round {round_number} rack={collection(rack)} moves={len(legal)} "
                         f"master {notation(master)} score={master.score} picks={picks}")
            board = apply(board, master.new_tiles)
            rack = master.leave + draw(bag, RACK_SIZE - len(master.leave))
            if not rack:
                reason = "out"
                break
        suffix = f" rack={collection(rack)}" if reason == "stuck" else ""
        lines.extend([f"end reason={reason} rounds={round_number}{suffix}", f"master total={master_total}"])
        for (player, _), total in zip(players, totals):
            percent = 100 * total // master_total if master_total >= 1 else 0
            lines.append(f"final {player} total={total} percent={percent}")
        winners = [player for (player, _), total in zip(players, totals) if total == max(totals)]
        lines.append("winner " + ",".join(winners))
        lines.append(self.position_summary(board))
        lines.extend(display(board))
        self.game_totals["rounds"] += round_number
        return lines
    def process(self, jobs, dictionary_lines):
        move_blocks, validation_blocks, game_blocks = [dictionary_lines], [], []
        seen = set()
        for job in jobs:
            kind, name = job[0].split()[:2]
            if kind == "TOPN":
                destination, totals, key = move_blocks, self.topn_totals, "jobs"
            elif kind == "VALIDATE":
                destination, totals, key = validation_blocks, self.validation_totals, "jobs"
            else:
                destination, totals = game_blocks, self.game_totals
                key = "games" if kind == "GAME" else "duplicates"
            totals[key] += 1
            error = self.job_error(job, seen)
            seen.add(name)
            if error:
                destination.append([f"{kind} {name}", "error " + error])
                totals["errors"] += 1
            elif kind == "TOPN":
                destination.append(self.topn(job))
            elif kind == "VALIDATE":
                destination.append(self.validate(job))
            elif kind == "GAME":
                destination.append(self.ordinary_game(job))
            else:
                destination.append(self.duplicate_game(job))
        summaries = ((move_blocks, self.topn_totals, ("jobs", "errors", "moves", "bingos")),
                     (validation_blocks, self.validation_totals, ("jobs", "errors", "moves", "valid", "invalid", "loss")),
                     (game_blocks, self.game_totals, ("games", "duplicates", "errors", "turns", "rounds", "plays", "exchanges", "passes", "bingos")))
        for blocks, totals, keys in summaries:
            blocks.append(["SUMMARY " + " ".join(f"{key}={totals[key]}" for key in keys)])
        return move_blocks, validation_blocks, game_blocks
def main():
    values, counts, weights, trie, jobs, dictionary_lines = read_inputs()
