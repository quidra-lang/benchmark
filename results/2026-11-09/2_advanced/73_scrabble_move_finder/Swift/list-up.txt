struct TrieNode { var children = Array(repeating: -1, count: 26); var terminal = false }
final class Trie {
    private(set) var nodes = [TrieNode()]
    private(set) var words = 0
    @discardableResult func add(_ word: [UInt8]) -> Bool {
        var node = 0
        for letter in word { let index = Int(letter - 65); if nodes[node].children[index] == -1 { let child = nodes.count; nodes.append(TrieNode()); nodes[node].children[index] = child }; node = nodes[node].children[index] }
        if nodes[node].terminal { return false }
        nodes[node].terminal = true; words += 1
        return true
    }
    func prefix(_ word: [UInt8]) -> Int? { var node = 0; for letter in word { node = nodes[node].children[Int(letter - 65)]; if node == -1 { return nil } }; return node }
    func contains(_ word: [UInt8]) -> Bool { prefix(word).map { nodes[$0].terminal } ?? false }

let tileOrder: [UInt8] = Array("ABCDEFGHIJKLMNOPQRSTUVWXYZ?".utf8)
final class Minstd { private var state: Int64; init(_ seed: Int64) { state = seed }; func next() -> Int64 { state = state * 48271 % 2147483647; return state } }
func shuffle(_ bag: inout [UInt8], _ random: Minstd) { if bag.count > 1 { for position in stride(from: bag.count - 1, through: 1, by: -1) { let other = Int(random.next() % Int64(position + 1)); bag.swapAt(position, other) } } }
func makeBag(_ counts: [Int], _ random: Minstd) -> [UInt8] { var bag = tileOrder.indices.flatMap { Array(repeating: tileOrder[$0], count: counts[$0]) }; shuffle(&bag, random); return bag }
func draw(_ bag: inout [UInt8], _ count: Int) -> [UInt8] { var drawn: [UInt8] = []; for _ in 0..<min(count, bag.count) { drawn.append(bag.removeLast()) }; return drawn }
let letters: [UInt8] = Array("ABCDEFGHIJKLMNOPQRSTUVWXYZ".utf8)
func anchors(_ board: Board) -> Set<Square> {
    if boardTiles(board).isEmpty { return [center] }
    var result: Set<Square> = []
    for row in 0..<boardSize { for column in 0..<boardSize where board[row][column] == 46 {
        let square = Square(row: row, column: column)
        if neighbors(square).contains(where: { board[$0.row][$0.column] != 46 }) { result.insert(square) }
    } }
    return result
}
func crossChecks(_ board: Board, _ trie: Trie) -> [Square: Set<UInt8>] {
    var result: [Square: Set<UInt8>] = [:]
    for row in 0..<boardSize { for column in 0..<boardSize where board[row][column] == 46 {
        var first = row, last = row
        while first > 0 && board[first - 1][column] != 46 { first -= 1 }
        while last + 1 < boardSize && board[last + 1][column] != 46 { last += 1 }
        if first == last { continue }
        let prefix = (first..<row).map { upper(board[$0][column]) }, suffix = ((row + 1)..<(last + 1)).map { upper(board[$0][column]) }
        result[Square(row: row, column: column)] = Set(letters.filter { trie.contains(prefix + [$0] + suffix) })
    } }
    return result
}
func acrossMoves(_ board: Board, _ rack: [UInt8], _ trie: Trie, _ anchorSquares: Set<Square>, _ checks: [Square: Set<UInt8>]) -> [Move] {
    var result: [Move] = [], counts = Array(repeating: 0, count: 27)
    for letter in rack { counts[letter == 63 ? 26 : Int(letter - 65)] += 1 }
    for square in anchorSquares.sorted(by: { $0.row == $1.row ? $0.column < $1.column : $0.row < $1.row }) {
        let row = square.row, anchor = square.column
        func extend(_ node: Int, _ column: Int, _ word: [UInt8], _ start: Int) {
            let empty = column == boardSize || board[row][column] == 46
            if empty && column > anchor && trie.nodes[node].terminal { result.append(Move(row: row, column: start, direction: .across, word: word)) }
            if column == boardSize { return }
            if !empty {
                let letter = board[row][column], child = trie.nodes[node].children[Int(upper(letter) - 65)]
                if child != -1 { extend(child, column + 1, word + [letter], start) }
                return
            }
            let allowed = checks[Square(row: row, column: column)]
            for index in 0..<26 where trie.nodes[node].children[index] != -1 {
                let letter = UInt8(65 + index), child = trie.nodes[node].children[index]
                if let allowed, !allowed.contains(letter) { continue }
                for (tile, written) in [(index, letter), (26, letter + 32)] where counts[tile] > 0 { counts[tile] -= 1; extend(child, column + 1, word + [written], start); counts[tile] += 1 }
            }
        }
        func leftPart(_ node: Int, _ prefix: [UInt8], _ remaining: Int) {
            extend(node, anchor, prefix, anchor - prefix.count)
            if remaining == 0 { return }
            for index in 0..<26 where trie.nodes[node].children[index] != -1 {
                let letter = UInt8(65 + index)
                for (tile, written) in [(index, letter), (26, letter + 32)] where counts[tile] > 0 { counts[tile] -= 1; leftPart(trie.nodes[node].children[index], prefix + [written], remaining - 1); counts[tile] += 1 }
            }
        }
        if anchor > 0 && board[row][anchor - 1] != 46 {
            var first = anchor - 1
            while first > 0 && board[row][first - 1] != 46 { first -= 1 }
            let prefix = Array(board[row][first..<anchor])
            if let node = trie.prefix(prefix.map(upper)) { extend(node, anchor, prefix, first) }
        } else {
            var first = anchor - 1
            while first >= 0 && board[row][first] == 46 && !anchorSquares.contains(Square(row: row, column: first)) { first -= 1 }
            leftPart(0, [], anchor - first - 1)
        }
    }
    return result
}
struct PlacementKey: Hashable { let tiles: [PlacedTile] }
func generate(_ board: Board, _ rack: [UInt8], _ trie: Trie) -> [Move] {
    let anchorSquares = anchors(board), across = acrossMoves(board, rack, trie, anchorSquares, crossChecks(board, trie)), transposed = transpose(board)
    let downAnchors = Set(anchorSquares.map { Square(row: $0.column, column: $0.row) }), down = acrossMoves(transposed, rack, trie, downAnchors, crossChecks(transposed, trie)).map { Move(row: $0.column, column: $0.row, direction: .down, word: $0.word) }
    var seen: Set<PlacementKey> = [], result: [Move] = []
    for move in across + down {
        let newTiles = placement(board, move)
        if newTiles.count == 1 && move.direction == .down && runAt(applying(board, newTiles), newTiles[0].square, .across).word.count >= 2 { continue }
        if seen.insert(PlacementKey(tiles: newTiles.sorted())).inserted { result.append(move) }
    }
    return result
}
let rackSize = 7
let bingoBonus = 50
func squares(_ move: Move) -> [PlacedTile] { move.word.enumerated().map { PlacedTile(row: move.row + move.direction.dr * $0.offset, column: move.column + move.direction.dc * $0.offset, letter: $0.element) } }
func placement(_ board: Board, _ move: Move) -> [PlacedTile] { squares(move).filter { board[$0.row][$0.column] == 46 } }
func applying(_ board: Board, _ tiles: [PlacedTile]) -> Board { var result = board; for tile in tiles { result[tile.row][tile.column] = tile.letter }; return result }
func crossWords(_ board: Board, _ move: Move) -> [Move] { let newTiles = placement(board, move), after = applying(board, newTiles); return newTiles.map { runAt(after, $0.square, move.direction.perpendicular) }.filter { $0.word.count >= 2 } }
func wordScore(_ board: Board, _ word: Move, _ values: [Int]) -> Int {
    var score = 0, multiplier = 1
    for tile in squares(word) {
        var value = lowercase(tile.letter) ? 0 : values[Int(tile.letter - 65)]
        if board[tile.row][tile.column] == 46 {
            let premium = premiums[tile.row][tile.column]
            value *= premium == 43 ? 2 : premium == 94 ? 3 : 1
            multiplier *= premium == 42 ? 2 : premium == 35 ? 3 : 1
        }
        score += value
    }
    return score * multiplier
}
struct ScoredWord { let move: Move; let score: Int }
struct MoveScore { let score: Int; let tiles: [PlacedTile]; let words: [ScoredWord] }
func scoreMove(_ board: Board, _ move: Move, _ values: [Int]) -> MoveScore {
    let words = ([move] + crossWords(board, move)).map { ScoredWord(move: $0, score: wordScore(board, $0, values)) }, tiles = placement(board, move)
    return MoveScore(score: words.map { $0.score }.reduce(0, +) + (tiles.count == rackSize ? bingoBonus : 0), tiles: tiles, words: words)
}

enum Direction: Int {

    var dc: Int { 1 - rawValue }

struct Square: Hashable { let row: Int; let column: Int }
let center = Square(row: 7, column: 7)

    static func < (_ first: PlacedTile, _ second: PlacedTile) -> Bool { if first.row != second.row { return first.row < second.row }; if first.column != second.column { return first.column < second.column }; return first.letter < second.letter }

func text(_ bytes: [UInt8]) -> String { String(decoding: bytes, as: UTF8.self) }
func upper(_ letter: UInt8) -> UInt8 { letter >= 97 && letter <= 122 ? letter - 32 : letter }
func lowercase(_ letter: UInt8) -> Bool { letter >= 97 && letter <= 122 }
func inside(_ row: Int, _ column: Int) -> Bool { row >= 0 && row < boardSize && column >= 0 && column < boardSize }
func squareName(_ row: Int, _ column: Int) -> String { String(UnicodeScalar(UInt8(65 + column))) + String(row + 1) }
func position(_ move: Move) -> String { let column = String(UnicodeScalar(UInt8(65 + move.column))); return move.direction == .across ? String(move.row + 1) + column : column + String(move.row + 1) }
func transpose(_ board: Board) -> Board { (0..<boardSize).map { column in (0..<boardSize).map { board[$0][column] } } }
func transposeTiles(_ tiles: [PlacedTile]) -> [PlacedTile] { tiles.map { PlacedTile(row: $0.column, column: $0.row, letter: $0.letter) } }
func boardTiles(_ board: Board) -> [PlacedTile] { board.indices.flatMap { row in board[row].indices.filter { board[row][$0] != 46 }.map { PlacedTile(row: row, column: $0, letter: board[row][$0]) } } }
func neighbors(_ square: Square) -> [Square] { [Square(row: square.row - 1, column: square.column), Square(row: square.row + 1, column: square.column), Square(row: square.row, column: square.column - 1), Square(row: square.row, column: square.column + 1)].filter { inside($0.row, $0.column) } }
func runAt(_ board: Board, _ square: Square, _ direction: Direction) -> Move {
    var row = square.row, column = square.column
    while inside(row - direction.dr, column - direction.dc) && board[row - direction.dr][column - direction.dc] != 46 { row -= direction.dr; column -= direction.dc }
    let firstRow = row, firstColumn = column
    var letters: [UInt8] = []
    while inside(row, column) && board[row][column] != 46 { letters.append(board[row][column]); row += direction.dr; column += direction.dc }
    return Move(row: firstRow, column: firstColumn, direction: direction, word: letters)
}

enum Strategy: String { case score, tiles, equity }
let vowels: Set<UInt8> = Set("AEIOU".utf8)
struct TileTable { let values: [Int]; let counts: [Int]; let weights: [Int] }
struct Inputs { let table: TileTable; let trie: Trie; let jobs: [[String]]; let dictionaryLines: [String] }
struct Evaluated {
    let move: Move; let score: Int; let newTiles: [PlacedTile]; let wordList: String; let leave: [UInt8]; let value: Int
    var key: PlacementKey { PlacementKey(tiles: newTiles.sorted()) }
}
func validName(_ value: String) -> Bool { let bytes = Array(value.utf8); return !bytes.isEmpty && bytes.count <= 20 && bytes.allSatisfy { ($0 >= 97 && $0 <= 122) || ($0 >= 48 && $0 <= 57) || $0 == 95 } }
func validNumber(_ value: String, _ minimum: Int, _ maximum: Int) -> Bool { let bytes = Array(value.utf8); if bytes.isEmpty || (bytes.count > 1 && bytes[0] == 48) || !bytes.allSatisfy({ $0 >= 48 && $0 <= 57 }) { return false }; guard let number = Int(value) else { return false }; return number >= minimum && number <= maximum }
func validWord(_ value: [UInt8]) -> Bool { value.count >= 2 && value.count <= boardSize && value.allSatisfy { let letter = upper($0); return letter >= 65 && letter <= 90 } }
func collection(_ value: [UInt8]) -> String { value.isEmpty ? "-" : text(value.sorted()) }
func tileKind(_ letter: UInt8) -> UInt8 { lowercase(letter) ? 63 : letter }
func tileIndex(_ letter: UInt8) -> Int { letter == 63 || lowercase(letter) ? 26 : Int(letter - 65) }
func countsOf(_ rack: [UInt8]) -> [Int] { var counts = Array(repeating: 0, count: 27); for letter in rack { counts[tileIndex(letter)] += 1 }; return counts }
func tileBytes(_ counts: [Int]) -> [UInt8] { ([26] + Array(0..<26)).flatMap { Array(repeating: tileOrder[$0], count: counts[$0]) } }
func removeTiles(_ rack: [UInt8], _ removed: [UInt8]) -> [UInt8] { var counts = countsOf(rack); for letter in removed { counts[tileIndex(letter)] -= 1 }; return tileBytes(counts) }
func notation(_ evaluated: Evaluated) -> String { position(evaluated.move) + " " + text(evaluated.move.word) }
func standardPrecedes(_ first: Evaluated, _ second: Evaluated) -> Bool {
    if first.score != second.score { return first.score > second.score }
    let a = first.move, b = second.move
    if a.row != b.row { return a.row < b.row }; if a.column != b.column { return a.column < b.column }
    if a.direction != b.direction { return a.direction.rawValue < b.direction.rawValue }
    return a.word.lexicographicallyPrecedes(b.word)
}
func strategyPrecedes(_ first: Evaluated, _ second: Evaluated, _ strategy: Strategy) -> Bool {
    if strategy == .tiles && first.newTiles.count != second.newTiles.count { return first.newTiles.count > second.newTiles.count }
    if strategy == .equity && first.score + first.value != second.score + second.value { return first.score + first.value > second.score + second.value }
    return standardPrecedes(first, second)
}
func inputLines(_ name: String) throws -> [String] { try String(contentsOfFile: "input/" + name, encoding: .ascii).split(separator: "\n", omittingEmptySubsequences: false).dropLast().map(String.init) }
func fields(_ line: String) -> [String] { line.split(separator: " ").map(String.init) }
func readInputs() throws -> Inputs {
    var values = Array(repeating: 0, count: 27), counts = values, weights = values
    for line in try inputLines("tiles.txt") where !line.isEmpty && line.first != "#" {
        let parts = fields(line), index = tileIndex(parts[0].utf8.first!)
        values[index] = Int(parts[1])!; counts[index] = Int(parts[2])!; weights[index] = Int(parts[3])!
    }
    let trie = Trie()
    var candidates = 0, skipped = 0, duplicates = 0, lengths: [Int: Int] = [:]
    for line in try inputLines("dictionary.txt") where !line.isEmpty && line.first != "#" {
        candidates += 1
        let bytes = Array(line.utf8)
        if bytes.count < 2 || bytes.count > 15 || !bytes.allSatisfy({ $0 >= 65 && $0 <= 90 }) { skipped += 1 }
        else if trie.add(bytes) { lengths[bytes.count, default: 0] += 1 } else { duplicates += 1 }
    }
    var jobs: [[String]] = [], current: [String] = [], active = false
    for line in try inputLines("jobs.txt") {
        if !active { if !line.isEmpty && line.first != "#" { current = [line]; active = true } }
        else if line == "END" { jobs.append(current); current = []; active = false } else { current.append(line) }
    }
    let dictionaryLines = ["DICTIONARY lines=\(candidates) words=\(trie.words) skipped=\(skipped) duplicates=\(duplicates) nodes=\(trie.nodes.count)", "lengths" + lengths.keys.sorted().map { " \($0)=\(lengths[$0]!)" }.joined()]
    return Inputs(table: TileTable(values: values, counts: counts, weights: weights), trie: trie, jobs: jobs, dictionaryLines: dictionaryLines)
}
    func jobError(_ job: [String], _ seen: Set<String>) -> String? {
        let header = fields(job[0]), kind = header[0], name = header[1]
        if !validName(name) { return "bad name" }; if seen.contains(name) { return "duplicate name" }
        if kind == "TOPN" {
            if !validNumber(header[2], 1, 99) { return "bad count" }; if Strategy(rawValue: header[3]) == nil { return "bad order" }
            let rack = Array(fields(job[1])[1].utf8)
            if !validRack(rack) { return "bad rack" }
            return boardError(job.dropFirst(3).map { Array($0.utf8) }, rack)
        }
        if kind == "VALIDATE" { return boardError(job.dropFirst(2).prefix { !$0.hasPrefix("MOVE") }.map { Array($0.utf8) }, []) }
        if !validNumber(header[2], 1, 2147483646) { return "bad seed" }
        if job.count - 1 < 2 || job.count - 1 > 4 { return "bad player count \(job.count - 1)" }
        var names: Set<String> = []
        for (index, line) in job.dropFirst().enumerated() {
            let parts = fields(line)
            if !validName(parts[1]) || Strategy(rawValue: parts[2]) == nil || (kind == "GAME" && !validNumber(parts[3], 0, 99)) { return "bad player \(index + 1)" }
            if !names.insert(parts[1]).inserted { return "duplicate player \(index + 1)" }
        }
        return nil
    }
    func leaveValue(_ rack: [UInt8]) -> Int {
        let counts = countsOf(rack), duplicates = (0..<26).map { max(0, counts[$0] - 1) }.reduce(0, +)
        let vowelCount = rack.filter { vowels.contains($0) }.count, consonants = rack.filter { $0 != 63 && !vowels.contains($0) }.count
        return rack.map { table.weights[tileIndex($0)] }.reduce(0, +) - 4 * duplicates - 3 * max(0, abs(vowelCount - consonants) - 1)
    }
    func evaluate(_ board: Board, _ rack: [UInt8], _ move: Move) -> Evaluated {
        let scored = scoreMove(board, move, table.values), leave = removeTiles(rack, scored.tiles.map { tileKind($0.letter) })
        let words = scored.words.map { "\(text($0.move.word)):\($0.score)" }.joined(separator: ",")
        return Evaluated(move: move, score: scored.score, newTiles: scored.tiles, wordList: words, leave: leave, value: leaveValue(leave))
    }
    func legalMoves(_ board: Board, _ rack: [UInt8]) -> [Evaluated] { generate(board, rack, trie).map { evaluate(board, rack, $0) }.sorted(by: standardPrecedes) }
    func duplicateGame(_ job: [String]) -> [String] {
        let header = fields(job[0]), name = header[1], seed = Int64(header[2])!, random = Minstd(seed), players = job.dropFirst().map { line -> (name: String, strategy: Strategy) in let parts = fields(line); return (parts[1], Strategy(rawValue: parts[2])!) }
        var bag = makeBag(table.counts, random), board = Array(repeating: Array(repeating: UInt8(46), count: boardSize), count: boardSize), rack = draw(&bag, rackSize), totals = Array(repeating: 0, count: players.count), lines = ["DUPLICATE \(name) seed=\(seed) players=\(players.count)"]
        for (index, player) in players.enumerated() { lines.append("player \(index + 1) \(player.name) strategy=\(player.strategy.rawValue)") }
        var round = 0, masterTotal = 0, reason = "stuck"
        while true {
            let legal = legalMoves(board, rack)
            guard let master = legal.first else { break }
            round += 1
            let choices = players.map { player in legal.min { strategyPrecedes($0, $1, player.strategy) }! }
            for index in players.indices { totals[index] += choices[index].score }
            masterTotal += master.score
            lines.append("round \(round) rack=\(collection(rack)) moves=\(legal.count) master \(notation(master)) score=\(master.score) picks=\(choices.map { String($0.score) }.joined(separator: ","))")
            board = applying(board, master.newTiles); rack = master.leave + draw(&bag, rackSize - master.leave.count)
            if rack.isEmpty { reason = "out"; break }
        }
        lines += ["end reason=\(reason) rounds=\(round)" + (reason == "stuck" ? " rack=" + collection(rack) : ""), "master total=\(masterTotal)"]
        for index in players.indices { let percent = masterTotal >= 1 ? 100 * totals[index] / masterTotal : 0; lines.append("final \(players[index].name) total=\(totals[index]) percent=\(percent)") }
        lines.append("winner " + players.indices.filter { totals[$0] == totals.max()! }.map { players[$0].name }.joined(separator: ","))
        lines.append(positionSummary(board)); lines += display(board); gameTotals["rounds", default: 0] += round
        return lines
    }
    func process(_ jobs: [[String]], _ dictionaryLines: [String]) -> [[[String]]] {
        var moveBlocks = [dictionaryLines], validationBlocks: [[String]] = [], gameBlocks: [[String]] = [], seen: Set<String> = []
        for job in jobs {
            let header = fields(job[0]), kind = header[0], name = header[1]
            if kind == "TOPN" { topTotals["jobs", default: 0] += 1 } else if kind == "VALIDATE" { validationTotals["jobs", default: 0] += 1 } else { gameTotals[kind == "GAME" ? "games" : "duplicates", default: 0] += 1 }
            let error = jobError(job, seen)
            seen.insert(name)
            if let error {
                let block = [kind + " " + name, "error " + error]
                if kind == "TOPN" { moveBlocks.append(block); topTotals["errors", default: 0] += 1 } else if kind == "VALIDATE" { validationBlocks.append(block); validationTotals["errors", default: 0] += 1 } else { gameBlocks.append(block); gameTotals["errors", default: 0] += 1 }
            } else if kind == "TOPN" { moveBlocks.append(topn(job)) } else if kind == "VALIDATE" { validationBlocks.append(validate(job)) } else if kind == "GAME" { gameBlocks.append(ordinaryGame(job)) } else { gameBlocks.append(duplicateGame(job)) }
        }
        moveBlocks.append(["SUMMARY " + ["jobs", "errors", "moves", "bingos"].map { "\($0)=\(topTotals[$0, default: 0])" }.joined(separator: " ")])
        validationBlocks.append(["SUMMARY " + ["jobs", "errors", "moves", "valid", "invalid", "loss"].map { "\($0)=\(validationTotals[$0, default: 0])" }.joined(separator: " ")])
        gameBlocks.append(["SUMMARY " + ["games", "duplicates", "errors", "turns", "rounds", "plays", "exchanges", "passes", "bingos"].map { "\($0)=\(gameTotals[$0, default: 0])" }.joined(separator: " ")])
        return [moveBlocks, validationBlocks, gameBlocks]
    }
}
