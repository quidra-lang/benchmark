class TrieNode {
  final List<int> children = List.filled(26, -1);
  bool terminal = false;
}

class Trie {
  final List<TrieNode> nodes = [TrieNode()];
  int words = 0;
  bool add(String word) {
    var node = 0;
    for (final code in word.codeUnits) {
      final letter = code - 65;
      var next = nodes[node].children[letter];
      if (next < 0) {
        next = nodes.length;
        nodes.add(TrieNode());
        nodes[node].children[letter] = next;
      }
      node = next;
    }
    if (nodes[node].terminal) return false;
    nodes[node].terminal = true;
    words++;
    return true;
  }

  int find(String prefix) {
    var node = 0;
    for (final code in prefix.toUpperCase().codeUnits) {
      node = nodes[node].children[code - 65];
      if (node < 0) return -1;
    }
    return node;
  }

  bool contains(String word) {
    final node = find(word);
    return node >= 0 && nodes[node].terminal;
  }

class Generator {
  int state;
  Generator(this.state);
  int next() {
    state = 48271 * state % 2147483647;
    return state;
  }

  void shuffle(List<String> bag) {
    for (var i = bag.length - 1; i >= 1; i--) {
      final j = next() % (i + 1), temporary = bag[i];
      bag[i] = bag[j];
      bag[j] = temporary;
    }
  }
}

List<String> newBag(Tiles tiles, Generator g) {
  final bag = <String>[];
  for (var i = 0; i < 27; i++) {
    final letter = i == 26 ? '?' : String.fromCharCode(65 + i);
    for (var k = 0; k < tiles.count[i]; k++) {
      bag.add(letter);
    }
  }
  g.shuffle(bag);
  return bag;
}

String draw(List<String> bag, int k) {
  final drawn = StringBuffer();
  for (var i = 0; i < k && bag.isNotEmpty; i++) {
    drawn.write(bag.removeLast());
  }
  return drawn.toString();
}

class CrossStats {
  int squares = 0, empty = 0, letters = 0;
}

List<List<bool>> anchors(Board b) {
  final empty = tileCount(b) == 0;
  return [
    for (var r = 0; r < 15; r++)
      [
        for (var c = 0; c < 15; c++)
          b.cells[r][c] == '.' &&
              ((empty && r == center && c == center) ||
                  (!empty && adjacent(b, r, c))),
      ],
  ];
}

(List<List<Check>>, CrossStats) crossChecks(Board b, int direction, Trie trie) {
  final checks = List.generate(
        15,
        (_) => List.generate(15, (_) => Check(false, 0)),
      ),
      stats = CrossStats(),
      perpendicular = 1 - direction;
  final (dr, dc) = directions(perpendicular);
  for (var r = 0; r < 15; r++) {
    for (var c = 0; c < 15; c++) {
      if (b.cells[r][c] != '.') continue;
      final has =
          (inside(r - dr, c - dc) && b.cells[r - dr][c - dc] != '.') ||
          (inside(r + dr, c + dc) && b.cells[r + dr][c + dc] != '.');
      checks[r][c] = Check(has, has ? 0 : (1 << 26) - 1);
      if (!has) continue;
      stats.squares++;
      for (var letter = 0; letter < 26; letter++) {
        final after = b.copy();
        after.cells[r][c] = String.fromCharCode(65 + letter);
        if (trie.contains(wordAt(after, r, c, perpendicular).word)) {
          checks[r][c].letters |= 1 << letter;
          stats.letters++;
        }
      }
      if (checks[r][c].letters == 0) stats.empty++;
    }
  }
  return (checks, stats);
}

String placementKey(Board b, RawMove move) {
  final (dr, dc) = directions(move.direction);
  final s = StringBuffer();
  for (var i = 0; i < move.word.length; i++) {
    final r = move.row + i * dr, c = move.col + i * dc;
    if (b.cells[r][c] == '.')
      s.write('${(r * 15 + c).toString().padLeft(3, '0')}${move.word[i]}');
  }
  return s.toString();
}

List<RawMove> generateMoves(Board board, String rack, Trie trie) {
  final moves = <RawMove>[], seen = <String>{};
  for (var direction = 0; direction < 2; direction++) {
    final b = direction == 0 ? board : board.transpose(), anchor = anchors(b);
    final (checks, _) = crossChecks(b, 0, trie);
    for (var row = 0; row < 15; row++) {
      for (var col = 0; col < 15; col++) {
        if (!anchor[row][col]) continue;
        final counts = collectionCounts(rack);
        void record(int start, String word, int placed) {
          if (placed == 0 || word.length < 2) return;
          final move = direction == 0
              ? RawMove(row, start, 0, word)
              : RawMove(start, row, 1, word);
          if (placed == 1 && direction == 1) {
            final after = board.copy();
            final (dr, dc) = directions(direction);
            var newRow = 0, newCol = 0;
            for (var i = 0; i < word.length; i++) {
              final r = move.row + i * dr, c = move.col + i * dc;
              if (board.cells[r][c] == '.') {
                after.cells[r][c] = word[i];
                newRow = r;
                newCol = c;
              }
            }
            if (wordAt(after, newRow, newCol, 0).word.length >= 2) return;
          }
          final key = placementKey(board, move);
          if (seen.add(key)) moves.add(move);
        }

        void extend(
          int at,
          int start,
          int node,
          String word,
          int placed,
          bool filled,
        ) {
          if (at >= 15) {
            if (trie.nodes[node].terminal && filled)
              record(start, word, placed);
            return;
          }
          final tile = b.cells[row][at];
          if (tile != '.') {
            final next = trie
                .nodes[node]
                .children[tile.toUpperCase().codeUnitAt(0) - 65];
            if (next >= 0)
              extend(at + 1, start, next, word + tile, placed, filled);
            return;
          }
          if (trie.nodes[node].terminal && filled) record(start, word, placed);
          for (var letter = 0; letter < 26; letter++) {
            final next = trie.nodes[node].children[letter];
            if (next < 0 || ((checks[row][at].letters >> letter) & 1) == 0)
              continue;
            if (counts[letter] > 0) {
              counts[letter]--;
              extend(
                at + 1,
                start,
                next,
                word + String.fromCharCode(65 + letter),
                placed + 1,
                filled || at == col,
              );
              counts[letter]++;
            }
            if (counts[26] > 0) {
              counts[26]--;
              extend(
                at + 1,
                start,
                next,
                word + String.fromCharCode(97 + letter),
                placed + 1,
                filled || at == col,
              );
              counts[26]++;
            }
          }
        }

        if (col > 0 && b.cells[row][col - 1] != '.') {
          var start = col - 1;
          while (start > 0 && b.cells[row][start - 1] != '.') {
            start--;
          }
          final prefix = b.cells[row].sublist(start, col).join(),
              node = trie.find(prefix);
          if (node >= 0) extend(col, start, node, prefix, 0, false);
        } else {
          var limit = 0;
          for (
            var c = col - 1;
            c >= 0 && b.cells[row][c] == '.' && !anchor[row][c];
            c--
          ) {
            limit++;
          }
          void left(String prefix, int node, int length) {
            extend(col, col - length, node, prefix, length, false);
            if (length == limit) return;
            for (var letter = 0; letter < 26; letter++) {
              final next = trie.nodes[node].children[letter];
              if (next < 0) continue;
              if (counts[letter] > 0) {
                counts[letter]--;
                left(
                  prefix + String.fromCharCode(65 + letter),
                  next,
                  length + 1,
                );
                counts[letter]++;
              }
              if (counts[26] > 0) {
                counts[26]--;
                left(
                  prefix + String.fromCharCode(97 + letter),
                  next,
                  length + 1,
                );
                counts[26]++;
              }
            }
          }

          left('', 0, 0);
        }
      }
    }
  }
  return moves;
}

List<(int, int)> moveSquares(RawMove move) {
  final (dr, dc) = directions(move.direction);
  return [
    for (var i = 0; i < move.word.length; i++)
      (move.row + i * dr, move.col + i * dc),
  ];
}

int wordScore(BoardWord word, Set<(int, int)> newTiles, Tiles tiles) {
  final (dr, dc) = directions(word.direction);
  var sum = 0, multiplier = 1;
  for (var i = 0; i < word.word.length; i++) {
    final r = word.row + i * dr,
        c = word.col + i * dc,
        code = word.word.codeUnitAt(i);
    var value = code >= 65 && code <= 90 ? tiles.value[code - 65] : 0;
    if (newTiles.contains((r, c))) {
      switch (premiums[r][c]) {
        case '+':
          value *= 2;
          break;
        case '^':
          value *= 3;
          break;
        case '*':
          multiplier *= 2;
          break;
        case '#':
          multiplier *= 3;
          break;
      }
    }
    sum += value;
  }
  return sum * multiplier;
}

Move evaluateMove(Board board, String rack, RawMove raw, Tiles tiles) {
  final m = Move(raw),
      after = board.copy(),
      newTiles = <(int, int)>{},
      counts = collectionCounts(rack),
      squares = moveSquares(raw);
  for (var i = 0; i < squares.length; i++) {
    final (r, c) = squares[i];
    if (board.cells[r][c] == '.') {
      final letter = raw.word[i];
      m.placed.add(PlacedTile(r, c, letter));
      newTiles.add((r, c));
      after.cells[r][c] = letter;
      counts[tileIndex(letter)]--;
    }
  }
  final words = [BoardWord(raw.row, raw.col, raw.direction, raw.word)];
  for (final tile in m.placed) {
    final word = wordAt(after, tile.row, tile.col, 1 - raw.direction);
    if (word.word.length >= 2) words.add(word);
  }
  for (final word in words) {
    final score = wordScore(word, newTiles, tiles);
    m.score += score;
    m.words.add(ScoredWord(word, score));
  }
  m.tiles = m.placed.length;
  if (m.tiles == rackSize) m.score += bingoBonus;
  m.leave = countsString(counts);
  return m;
}

  for (final tile in move.placed) {
    board.cells[tile.row][tile.col] = tile.letter;
  }
}

    r -= dr;
    c -= dc;

    if (!validName(a[1]) || !strategy(a[2]) || threshold == null)

String reportDuplicate(Job job, Tiles tiles, Trie trie, GameTotals total) {
  final out = StringBuffer(
    'DUPLICATE ${job.header[1]} seed=${job.seed} players=${job.players.length}\n',
  );
  for (var i = 0; i < job.players.length; i++) {
    final p = job.players[i];
    out.writeln('player ${i + 1} ${p.name} strategy=${p.strategy}');
  }
  final generator = Generator(job.seed), bag = newBag(tiles, generator);
  var rack = draw(bag, rackSize);
  final board = Board.empty(), totals = List.filled(job.players.length, 0);
  var masterTotal = 0, round = 0, reason = 'stuck';
  while (true) {
    final moves = legalMoves(board, rack, tiles, trie);
    if (moves.isEmpty) {
      reason = 'stuck';
      break;
    }
    round++;
    final master = moves[0], picks = <String>[];
    for (var i = 0; i < job.players.length; i++) {
      final choice = orderedMoves(moves, job.players[i].strategy)[0];
      totals[i] += choice.score;
      picks.add('${choice.score}');
    }
    masterTotal += master.score;
    out.writeln(
      'round $round rack=${notation(rack)} moves=${moves.length} master ${moveName(master)} score=${master.score} picks=${picks.join(',')}',
    );
    play(board, master);
    rack = master.leave + draw(bag, rackSize - master.leave.length);
    if (rack.isEmpty) {
      reason = 'out';
      break;
    }
  }
  out.writeln(
    reason == 'out'
        ? 'end reason=out rounds=$round'
        : 'end reason=stuck rounds=$round rack=${notation(rack)}',
  );
  out.writeln('master total=$masterTotal');
  var best = totals[0];
  for (final value in totals) {
    best = max(best, value);
  }
  final winners = <String>[];
  for (var i = 0; i < job.players.length; i++) {
    final p = job.players[i],
        percentage = masterTotal >= 1 ? 100 * totals[i] ~/ masterTotal : 0;
    out.writeln('final ${p.name} total=${totals[i]} percent=$percentage');
    if (totals[i] == best) winners.add(p.name);
  }
  out.writeln('winner ${winners.join(',')}');
  boardReport(out, board);
  total.rounds += round;
  return out.toString();
}

    'DICTIONARY lines=${stats.lines} words=${trie.words} skipped=${stats.skipped} duplicates=${stats.duplicates} nodes=${trie.nodes.length}\n',
  );
  dictionary.writeln(
    [
      'lengths',
      for (var length = 0; length < stats.lengths.length; length++)
        if (stats.lengths[length] > 0) '$length=${stats.lengths[length]}',
    ].join(' '),
  );
  final movesParagraphs = [paragraph(dictionary.toString())],
      validationParagraphs = <String>[],
      gameParagraphs = <String>[],
      topTotals = TopTotals(),
      validationTotals = ValidationTotals(),
      gameTotals = GameTotals(),
      seen = <String>{};
  for (final job in jobs) {
    final kind = job.header[0], name = h[1];
    if (kind == 'TOPN') {
      topTotals.jobs++;
    } else if (kind == 'VALIDATE') {
      validationTotals.jobs++;
    } else if (kind == 'GAME') {
      gameTotals.games++;
    } else {
      gameTotals.duplicates++;
    }
    final reason = checkJob(job, seen, tiles, trie);
    if (reason.isNotEmpty) {
      final error = '$kind $name\nerror $reason';
      if (kind == 'TOPN') {
        topTotals.errors++;
        movesParagraphs.add(error);
      } else if (kind == 'VALIDATE') {
        validationTotals.errors++;
        validationParagraphs.add(error);
      } else {
        gameTotals.errors++;
        gameParagraphs.add(error);
      }
      continue;
    }
    if (kind == 'TOPN') {
      movesParagraphs.add(paragraph(reportTop(job, tiles, trie, topTotals)));
    } else if (kind == 'VALIDATE') {
      validationParagraphs.add(
        paragraph(reportValidation(job, tiles, trie, validationTotals)),
      );
    } else if (kind == 'GAME') {
      gameParagraphs.add(paragraph(reportGame(job, tiles, trie, gameTotals)));
    } else {
      gameParagraphs.add(
        paragraph(reportDuplicate(job, tiles, trie, gameTotals)),
      );
    }
  }
  movesParagraphs.add(
    'SUMMARY jobs=${topTotals.jobs} errors=${topTotals.errors} moves=${topTotals.moves} bingos=${topTotals.bingos}',
  );
  validationParagraphs.add(
    'SUMMARY jobs=${validationTotals.jobs} errors=${validationTotals.errors} moves=${validationTotals.moves} valid=${validationTotals.valid} invalid=${validationTotals.invalid} loss=${validationTotals.loss}',
  );
  gameParagraphs.add(
    'SUMMARY games=${gameTotals.games} duplicates=${gameTotals.duplicates} errors=${gameTotals.errors} turns=${gameTotals.turns} rounds=${gameTotals.rounds} plays=${gameTotals.plays} exchanges=${gameTotals.exchanges} passes=${gameTotals.passes} bingos=${gameTotals.bingos}',
  );
  Directory('output').createSync(recursive: true);
  writeParagraphs('output/moves.txt', movesParagraphs);
  writeParagraphs('output/validation.txt', validationParagraphs);
  writeParagraphs('output/game.txt', gameParagraphs);
}
