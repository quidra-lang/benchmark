package main

type TrieNode struct {
	Children [26]int
	Terminal bool
}
type Trie struct {
	Nodes []TrieNode
	Words int
}

func newTrieNode() TrieNode {
	var n TrieNode
	for i := range n.Children {
		n.Children[i] = -1
	}
	return n
}
func newTrie() Trie { return Trie{Nodes: []TrieNode{newTrieNode()}} }
func (t *Trie) add(word string) bool {
	node := 0
	for i := range word {
		letter := int(word[i] - 'A')
		next := t.Nodes[node].Children[letter]
		if next < 0 {
			next = len(t.Nodes)
			t.Nodes = append(t.Nodes,
				newTrieNode())
			t.Nodes[node].Children[letter] = next
		}
		node = next
	}
	if t.Nodes[node].Terminal {
		return false
	}
	t.Nodes[node].Terminal = true
	t.Words++
	return true
}
func (t Trie) find(prefix string) int {
	node := 0
	for i := range prefix {
		c := upper(prefix[i])
		node = t.Nodes[node].Children[c-'A']
		if node < 0 {
			return -1
		}
	}
	return node
}
func (t Trie) contains(word string) bool {
	node := t.find(word)
	return node >= 0 && t.Nodes[node].Terminal
}

type Generator struct{ State int64 }

func (g *Generator) next() int64 { g.State = 48271 * g.State % 2147483647; return g.State }
func (g *Generator) shuffle(bag []byte) {
	for i := len(bag) - 1; i >= 1; i-- {
		j := int(g.next() % int64(i+1))
		bag[i], bag[j] = bag[j], bag[i]
	}
}
func newBag(tiles Tiles, g *Generator) []byte {
	var bag []byte
	for i, n := range tiles.Count {
		letter := byte('A' + i)
		if i == 26 {
			letter = '?'
		}
		for k := 0; k < n; k++ {
			bag = append(bag,
				letter)
		}
	}
	g.shuffle(bag)
	return bag
}
func draw(bag *[]byte, k int) string {
	var drawn []byte
	for i := 0; i < k && len(*bag) > 0; i++ {
		last := len(*bag) - 1
		drawn = append(drawn,
			(*bag)[last])
		*bag = (*bag)[:last]
	}
	return string(drawn)
}

type Check struct {
	IsCheck bool
	Letters uint32
}
type CrossStats struct{ Squares, Empty, Letters int }

func anchors(b Board) [15][15]bool {
	var a [15][15]bool
	empty := tileCount(b) == 0
	for r := 0; r < 15; r++ {
		for c := 0; c < 15; c++ {
			a[r][c] = b[r][c] == '.' && ((empty && r == center && c == center) || (!empty && adjacent(b,
				r, c)))
		}
	}
	return a
}
func crossChecks(b Board, direction int, trie Trie) ([15][15]Check, CrossStats) {
	var checks [15][15]Check
	var stats CrossStats
	perpendicular := 1 - direction
	dr,
		dc := directions(perpendicular)
	for r := 0; r < 15; r++ {
		for c := 0; c < 15; c++ {
			if b[r][c] != '.' {
				continue
			}
			has := (inside(r-dr,
				c-dc) && b[r-dr][c-dc] != '.') || (inside(r+dr, c+dc) && b[r+dr][c+dc] != '.')
			checks[r][c].IsCheck = has
			if !has {
				checks[r][c].Letters = (1 << 26) - 1
				continue
			}
			stats.Squares++
			for letter := 0; letter < 26; letter++ {
				after := b
				after[r][c] = byte('A' + letter)
				if trie.contains(wordAt(after,
					r, c, perpendicular).Word) {
					checks[r][c].Letters |= 1 << letter
					stats.Letters++
				}
			}
			if checks[r][c].Letters == 0 {
				stats.Empty++
			}
		}
	}
	return checks,
		stats
}
func placementKey(b Board, move RawMove) string {
	dr, dc := directions(move.Direction)
	var s strings.Builder
	for i := range move.Word {
		r,
			c := move.Row+i*dr, move.Col+i*dc
		if b[r][c] == '.' {
			fmt.Fprintf(&s, "%03d%c", r*15+c, move.Word[i])
		}
	}
	return s.String()
}
func generateMoves(board Board, rack string, trie Trie) []RawMove {
	var moves []RawMove
	seen := map[string]bool{}
	for direction := 0; direction < 2; direction++ {
		b := board
		if direction == 1 {
			b = b.transpose()
		}
		anchor := anchors(b)
		checks,
			_ := crossChecks(b, 0, trie)
		for row := 0; row < 15; row++ {
			for col := 0; col < 15; col++ {
				if !anchor[row][col] {
					continue
				}
				counts := collectionCounts(rack)
				record := func(start int,
					word string, placed int) {
					if placed == 0 || len(word) < 2 {
						return
					}
					move := RawMove{row, start, direction,
						word}
					if direction == 1 {
						move.Row, move.Col = start, row
					}
					if placed == 1 && direction == 1 {
						after := board
						dr,
							dc := directions(direction)
						newRow, newCol := 0, 0
						for i := range word {
							r, c := move.Row+i*dr, move.Col+i*dc
							if board[r][c] == '.' {
								after[r][c] = word[i]
								newRow,
									newCol = r, c
							}
						}
						if len(wordAt(after, newRow, newCol, 0).Word) >= 2 {
							return
						}
					}
					key := placementKey(board,
						move)
					if !seen[key] {
						seen[key] = true
						moves = append(moves, move)
					}
				}
				var extend func(int, int, int, string,
					int, bool)
				extend = func(at, start, node int, word string, placed int, filled bool) {
					if at >= 15 {
						if trie.Nodes[node].Terminal && filled {
							record(start,
								word, placed)
						}
						return
					}
					tile := b[row][at]
					if tile != '.' {
						next := trie.Nodes[node].Children[upper(tile)-'A']
						if next >= 0 {
							extend(at+1,
								start, next, word+string(tile), placed, filled)
						}
						return
					}
					if trie.Nodes[node].Terminal && filled {
						record(start,
							word, placed)
					}
					for letter, next := range trie.Nodes[node].Children {
						if next < 0 || checks[row][at].Letters>>letter&1 == 0 {
							continue
						}
						if counts[letter] > 0 {
							counts[letter]--
							extend(at+1,
								start, next, word+string(byte('A'+letter)), placed+1, filled || at == col)
							counts[letter]++
						}
						if counts[26] > 0 {
							counts[26]--
							extend(at+1,
								start, next, word+string(byte('a'+letter)), placed+1, filled || at == col)
							counts[26]++
						}
					}
				}
				if col > 0 && b[row][col-1] != '.' {
					start := col - 1
					for start > 0 && b[row][start-1] != '.' {
						start--
					}
					prefix := string(b[row][start:col])
					node := trie.find(prefix)
					if node >= 0 {
						extend(col,
							start, node, prefix, 0, false)
					}
				} else {
					limit := 0
					for c := col - 1; c >= 0 && b[row][c] == '.' && !anchor[row][c]; c-- {
						limit++
					}
					var left func(string,
						int, int)
					left = func(prefix string, node, length int) {
						extend(col, col-length, node, prefix, length, false)
						if length == limit {
							return
						}
						for letter, next := range trie.Nodes[node].Children {
							if next < 0 {
								continue
							}
							if counts[letter] > 0 {
								counts[letter]--
								left(prefix+string(byte('A'+letter)),
									next, length+1)
								counts[letter]++
							}
							if counts[26] > 0 {
								counts[26]--
								left(prefix+string(byte('a'+letter)),
									next, length+1)
								counts[26]++
							}
						}
					}
					left("", 0, 0)
				}
			}
		}
	}
	return moves
}

type PlacedTile struct {

}
type ScoredWord struct {
	BoardWord BoardWord
	Score     int

type Move struct {
	Raw                            RawMove

func moveSquares(move RawMove) []Square {
	a := make([]Square, len(move.Word))
	dr, dc := directions(move.Direction)
	for i := range a {
		a[i] = Square{move.Row + i*dr,
			move.Col + i*dc}
	}
	return a
}
func wordScore(word BoardWord, newTiles map[Square]bool, tiles Tiles) int {
	dr, dc := directions(word.Direction)
	sum,
		multiplier := 0, 1
	for i := range word.Word {
		r, c := word.Row+i*dr, word.Col+i*dc
		value := 0
		letter := word.Word[i]
		if letter >= 'A' && letter <= 'Z' {
			value = tiles.Value[letter-'A']
		}
		if newTiles[Square{r,
			c}] {
			switch premiums[r][c] {
			case '+':
				value *= 2
			case '^':
				value *= 3
			case '*':
				multiplier *= 2
			case '#':
				multiplier *= 3
			}
		}
		sum += value
	}
	return sum * multiplier
}
func evaluateMove(board Board, rack string, raw RawMove, tiles Tiles) Move {
	m := Move{Raw: raw}
	after := board
	newTiles := map[Square]bool{}
	counts := collectionCounts(rack)
	for i, p := range moveSquares(raw) {
		if board[p.Row][p.Col] == '.' {
			letter := raw.Word[i]
			m.Placed = append(m.Placed,
				PlacedTile{p.Row, p.Col, letter})
			newTiles[p] = true
			after[p.Row][p.Col] = letter
			counts[tileIndex(letter)]--
		}
	}
	words := []BoardWord{{raw.Row,
		raw.Col, raw.Direction, raw.Word}}
	for _, tile := range m.Placed {
		word := wordAt(after, tile.Row, tile.Col,
			1-raw.Direction)
		if len(word.Word) >= 2 {
			words = append(words, word)
		}
	}
	for _, word := range words {
		score := wordScore(word,
			newTiles, tiles)
		m.Score += score
		m.Words = append(m.Words, ScoredWord{word, score})
	}
	m.Tiles = len(m.Placed)
	if m.Tiles == rackSize {
		m.Score += bingoBonus
	}
	m.Leave = countsString(counts)
	return m
}
func play(board *Board, move Move) {
	for _, tile := range move.Placed {
		board[tile.Row][tile.Col] = tile.Letter
	}
}
func wordList(move Move) string {
	a := make([]string, len(move.Words))
	for i, w := range move.Words {
		a[i] = fmt.Sprintf("%s:%d",
			w.BoardWord.Word, w.Score)
	}
	return strings.Join(a, ",")
}

var premiums = [15]string{"#..+...#...+..#", ".*...^...^...*.", "..*...+.+...*..", "+..*...+...*..+", "....*.....*....",

func newBoard() Board {
	var b Board
	for r := 0; r < 15; r++ {
		for c := 0; c < 15; c++ {
			b[r][c] = '.'
		}
	}
	return b
}
func inside(r, c int) bool       { return r >= 0 && r < 15 && c >= 0 && c < 15 }
func squareName(r, c int) string { return fmt.Sprintf("%c%d", 'A'+c, r+1) }
func positionName(r, c, direction int) string {
	if direction == 0 {
		return fmt.Sprintf("%d%c", r+1, 'A'+c)
	}
	return squareName(r, c)
}
func (b Board) transpose() Board {
	var a Board
	for r := 0; r < 15; r++ {
		for c := 0; c < 15; c++ {
			a[r][c] = b[c][r]
		}
	}
	return a
}
func transposeSquares(a []Square) []Square {
	b := make([]Square, len(a))
	for i, p := range a {
		b[i] = Square{p.Col, p.Row}
	}
	return b
}
func directions(direction int) (int, int) {
	if direction == 0 {
		return 0, 1
	}
	return 1, 0
}
func wordAt(b Board, r, c, direction int) BoardWord {
	dr, dc := directions(direction)
	for inside(r-dr, c-dc) && b[r-dr][c-dc] != '.' {
		r -= dr
		c -= dc
	}
	word := BoardWord{Row: r,
		Col: c, Direction: direction}
	var s strings.Builder
	for inside(r, c) && b[r][c] != '.' {
		s.WriteByte(b[r][c])
		r += dr
		c += dc
	}
	word.Word = s.String()
	return word
}
func adjacent(b Board, r, c int) bool {
	for _, d := range [][2]int{{-1, 0}, {1, 0}, {0, -1}, {0, 1}} {
		rr, cc := r+d[0],
			c+d[1]
		if inside(rr, cc) && b[rr][cc] != '.' {
			return true
		}
	}
	return false
}
func tileCount(b Board) int {
	n := 0
	for _, row := range b {
		for _, c := range row {
			if c != '.' {
				n++
			}
		}
	}
	return n
}

func upper(c byte) byte {
	if c >= 'a' && c <= 'z' {
		return c - 'a' + 'A'
	}
	return c
}
func tileIndex(c byte) int {
	if c == '?' || c >= 'a' && c <= 'z' {
		return 26
	}
	return int(c - 'A')
}
func collectionCounts(s string) [27]int {
	var a [27]int
	for i := range s {
		a[tileIndex(s[i])]++
	}
	return a
}
func countsString(a [27]int) string {
	var s strings.Builder
	for k := 0; k < a[26]; k++ {
		s.WriteByte('?')
	}
	for i := 0; i < 26; i++ {
		for k := 0; k < a[i]; k++ {
			s.WriteByte(byte('A' + i))
		}
	}
	return s.String()
}
func sortedCollection(s string) string { return countsString(collectionCounts(s)) }
func notation(s string) string {
	if s == "" {
		return "-"
	}
	return sortedCollection(s)
}
func validName(s string) bool {
	if len(s) < 1 || len(s) > 20 {
		return false
	}
	for i := range s {
		c := s[i]
		if !(c >= 'a' && c <= 'z' || c >= '0' && c <= '9' || c == '_') {
			return false
		}
	}
	return true
}
func validNumber(s string, lo, hi int) (int, bool) {
	if len(s) == 0 || len(s) > 1 && s[0] == '0' {
		return 0, false
	}
	n := 0
	for i := range s {
		c := s[i]
		if c < '0' || c > '9' {
			return 0,
				false
		}
		digit := int(c - '0')
		if n > (hi-digit)/10 {
			return 0, false
		}
		n = 10*n + digit
		if n > hi {
			return 0, false
		}
	}
	return n,
		n >= lo && n <= hi
}
func validWord(s string, uppercaseOnly bool) bool {
	if len(s) < 2 || len(s) > 15 {
		return false
	}
	for i := range s {
		c := s[i]
		if !(c >= 'A' && c <= 'Z' || !uppercaseOnly && c >= 'a' && c <= 'z') {
			return false
		}
	}
	return true
}
func strategy(s string) bool { return s == "score" || s == "tiles" || s == "equity" }
func readData() (Tiles, Trie, DictionaryStats, []Job) {
	var tiles Tiles
	for _, line := range strings.Split(strings.TrimSuffix(text("input/tiles.txt"),
		"\n"), "\n") {
		if line == "" || line[0] == '#' {
			continue
		}
		a := strings.Fields(line)
		i := tileIndex(a[0][0])
		tiles.Value[i],
			tiles.Count[i], tiles.Weight[i] = integer(a[1]), integer(a[2]), integer(a[3])
	}
	trie := newTrie()
	var stats DictionaryStats
	for _, word := range strings.Split(strings.TrimSuffix(text("input/dictionary.txt"), "\n"), "\n") {
		if word == "" || word[0] == '#' {
			continue
		}
		stats.Lines++
		if !validWord(word,
			true) {
			stats.Skipped++
			continue
		}
		if !trie.add(word) {
			stats.Duplicates++
			continue
		}
		stats.Lengths[len(word)]++
	}
	lines := strings.Split(strings.TrimSuffix(text("input/jobs.txt"),
		"\n"), "\n")
	var jobs []Job
	for pos := 0; pos < len(lines); pos++ {
		line := lines[pos]
		if line == "" || line[0] == '#' {
			continue
		}
		job := Job{Header: strings.Fields(line),
			Board: newBoard()}
		for pos++; pos < len(lines) && lines[pos] != "END"; pos++ {
			job.Lines = append(job.Lines,
				lines[pos])
		}
		jobs = append(jobs, job)
	}
	return tiles, trie, stats, jobs
}
func checkJob(job *Job, seen map[string]bool, tiles Tiles, trie Trie) string {
	h := job.Header
	name := h[1]
	reason := ""
	if !validName(name) {
		reason = "bad name"
	} else if seen[name] {
		reason = "duplicate name"
	}
	seen[name] = true
	if reason != "" {
		return reason
	}
	switch h[0] {
	case "TOPN":
		count,
			valid := validNumber(h[2], 1, 99)
		if !valid {
			return "bad count"
		}
		if !strategy(h[3]) {
			return "bad order"
		}
		rack := strings.Fields(job.Lines[0])[1]
		if !validRack(rack) {
			return "bad rack"
		}
		job.Count,
			job.Order, job.Rack = count, h[3], rack
		board, error := checkBoard(job.Lines[2:], rack, tiles, trie)
		job.Board = board
		return error
	case "VALIDATE":
		var boardLines []string
		records := false
		for _, line := range job.Lines[1:] {
			if strings.HasPrefix(line, "MOVE") {
				records = true
			}
			if records {
				job.Records = append(job.Records,
					strings.Fields(line))
			} else {
				boardLines = append(boardLines, line)
			}
		}
		board, error := checkBoard(boardLines,
			"", tiles, trie)
		job.Board = board
		return error
	default:
		seed, valid := validNumber(h[2], 1, 2147483646)
		if !valid {
			return "bad seed"
		}
		job.Seed = seed
		if len(job.Lines) < 2 || len(job.Lines) > 4 {
			return fmt.Sprintf("bad player count %d",
				len(job.Lines))
		}
		names := map[string]bool{}
		for i, line := range job.Lines {
			a := strings.Fields(line)
			threshold := 0
			validThreshold := true
			if h[0] == "GAME" {
				threshold,
					validThreshold = validNumber(a[3], 0, 99)
			}
			if !validName(a[1]) || !strategy(a[2]) || !validThreshold {
				return fmt.Sprintf("bad player %d",
					i+1)
			}
			if names[a[1]] {
				return fmt.Sprintf("duplicate player %d", i+1)
			}
			names[a[1]] = true
			job.Players = append(job.Players,
				PlayerSpec{a[1], a[2], threshold})
		}
		return ""
	}
}
func leaveValue(rack string, tiles Tiles) int {
	counts := collectionCounts(rack)
	value, duplicates := 0,
		0
	for i, n := range counts {
		value += n * tiles.Weight[i]
		if i < 26 && n > 0 {
			duplicates += n - 1
		}
	}
	v, c, _ := rackBreakdown(rack)
	difference := v - c
	if difference < 0 {
		difference = -difference
	}
	return value - 4*duplicates - 3*max(0,
		difference-1)
}
func standardLess(a, b Move) bool {
	if a.Score != b.Score {
		return a.Score > b.Score
	}
	if a.Raw.Row != b.Raw.Row {
		return a.Raw.Row < b.Raw.Row
	}
	if a.Raw.Col != b.Raw.Col {
		return a.Raw.Col < b.Raw.Col
	}
	if a.Raw.Direction != b.Raw.Direction {
		return a.Raw.Direction < b.Raw.Direction
	}
	return a.Raw.Word < b.Raw.Word
}
func legalMoves(board Board, rack string, tiles Tiles, trie Trie) []Move {
	raw := generateMoves(board, rack,
		trie)
	moves := make([]Move, len(raw))
	for i, r := range raw {
		moves[i] = evaluateMove(board, rack, r, tiles)
		moves[i].LeaveValue = leaveValue(moves[i].Leave,
			tiles)
	}
	sort.Slice(moves, func(i, j int) bool { return standardLess(moves[i], moves[j]) })
	for i := range moves {
		moves[i].Rank = i + 1
	}
	return moves
}
func orderedMoves(moves []Move, strategy string) []Move {
	a := append([]Move(nil), moves...)
	sort.SliceStable(a,
		func(i, j int) bool {
			if strategy == "tiles" {
				return a[i].Tiles > a[j].Tiles
			}
			if strategy == "equity" {
				return a[i].Score+a[i].LeaveValue > a[j].Score+a[j].LeaveValue
			}
			return false
		})
	return a
}
func reportDuplicate(job Job, tiles Tiles, trie Trie, total *GameTotals) string {
	var out strings.Builder
	line(&out,
		"DUPLICATE %s seed=%d players=%d", job.Header[1], job.Seed, len(job.Players))
	for i, p := range job.Players {
		line(&out,
			"player %d %s strategy=%s", i+1, p.Name, p.Strategy)
	}
	generator := Generator{int64(job.Seed)}
	bag := newBag(tiles,
		&generator)
	rack := draw(&bag, rackSize)
	board := newBoard()
	totals := make([]int, len(job.Players))
	masterTotal,
		round := 0, 0
	reason := "stuck"
	for {
		moves := legalMoves(board, rack, tiles, trie)
		if len(moves) == 0 {
			reason = "stuck"
			break
		}
		round++
		master := moves[0]
		picks := make([]string,
			len(job.Players))
		for i, p := range job.Players {
			choice := orderedMoves(moves, p.Strategy)[0]
			totals[i] += choice.Score
			picks[i] = fmt.Sprint(choice.Score)
		}
		masterTotal += master.Score
		line(&out,
			"round %d rack=%s moves=%d master %s score=%d picks=%s", round, notation(rack), len(moves), moveName(master),
			master.Score, strings.Join(picks, ","))
		play(&board, master)
		rack = master.Leave + draw(&bag, rackSize-len(master.Leave))
		if rack == "" {
			reason = "out"
			break
		}
	}
	if reason == "out" {
		line(&out,
			"end reason=out rounds=%d", round)
	} else {
		line(&out, "end reason=stuck rounds=%d rack=%s", round, notation(rack))
	}
	line(&out,
		"master total=%d", masterTotal)
	best := totals[0]
	for _, value := range totals {
		best = max(best, value)
	}
	var winners []string
	for i, p := range job.Players {
		percentage := 0
		if masterTotal >= 1 {
			percentage = 100 * totals[i] / masterTotal
		}
		line(&out,
			"final %s total=%d percent=%d", p.Name, totals[i], percentage)
		if totals[i] == best {
			winners = append(winners,
				p.Name)
		}
	}
	line(&out, "winner %s", strings.Join(winners, ","))
	boardReport(&out, board)
	total.Rounds += round
	return out.String()
}
func main() {
	tiles, trie, stats, jobs := readData()
	var dictionary strings.Builder
	line(&dictionary, "DICTIONARY lines=%d words=%d skipped=%d duplicates=%d nodes=%d",
		stats.Lines, trie.Words, stats.Skipped, stats.Duplicates, len(trie.Nodes))
	parts := []string{"lengths"}
	for length, count := range stats.Lengths {
		if count > 0 {
			parts = append(parts, fmt.Sprintf("%d=%d", length, count))
		}
	}
	line(&dictionary,
		strings.Join(parts, " "))
	movesParagraphs := []string{paragraph(dictionary.String())}
	var validationParagraphs,
		gameParagraphs []string
	var topTotals TopTotals
	var validationTotals ValidationTotals
	var gameTotals GameTotals
	seen := map[string]bool{}
	for _, inputJob := range jobs {
		job := inputJob
		kind, name := job.Header[0], job.Header[1]
		switch kind {
		case "TOPN":
			topTotals.Jobs++
		case "VALIDATE":
			validationTotals.Jobs++
		case "GAME":
			gameTotals.Games++
		case "DUPLICATE":
			gameTotals.Duplicates++
		}
		reason := checkJob(&job,
			seen, tiles, trie)
		if reason != "" {
			error := kind + " " + name + "\nerror " + reason
			switch kind {
			case "TOPN":
				topTotals.Errors++
				movesParagraphs = append(movesParagraphs,
					error)
			case "VALIDATE":
				validationTotals.Errors++
				validationParagraphs = append(validationParagraphs,
					error)
			default:
				gameTotals.Errors++
				gameParagraphs = append(gameParagraphs, error)
			}
			continue
		}
		switch kind {
		case "TOPN":
			movesParagraphs = append(movesParagraphs,
				paragraph(reportTop(job, tiles, trie, &topTotals)))
		case "VALIDATE":
			validationParagraphs = append(validationParagraphs,
				paragraph(reportValidation(job, tiles, trie, &validationTotals)))
		case "GAME":
			gameParagraphs = append(gameParagraphs,
				paragraph(reportGame(job, tiles, trie, &gameTotals)))
		case "DUPLICATE":
			gameParagraphs = append(gameParagraphs,
				paragraph(reportDuplicate(job, tiles, trie, &gameTotals)))
		}
	}
	movesParagraphs = append(movesParagraphs,
		fmt.Sprintf("SUMMARY jobs=%d errors=%d moves=%d bingos=%d", topTotals.Jobs, topTotals.Errors, topTotals.Moves,
			topTotals.Bingos))
	validationParagraphs = append(validationParagraphs, fmt.Sprintf("SUMMARY jobs=%d errors=%d moves=%d valid=%d invalid=%d loss=%d",
		validationTotals.Jobs, validationTotals.Errors, validationTotals.Moves, validationTotals.Valid, validationTotals.Invalid,
		validationTotals.Loss))
	gameParagraphs = append(gameParagraphs, fmt.Sprintf("SUMMARY games=%d duplicates=%d errors=%d turns=%d rounds=%d plays=%d exchanges=%d passes=%d bingos=%d",
		gameTotals.Games, gameTotals.Duplicates, gameTotals.Errors, gameTotals.Turns, gameTotals.Rounds, gameTotals.Plays,
		gameTotals.Exchanges, gameTotals.Passes, gameTotals.Bingos))
	must(os.MkdirAll("output", 0755))
	writeParagraphs("output/moves.txt",
		movesParagraphs)
	writeParagraphs("output/validation.txt", validationParagraphs)
	writeParagraphs("output/game.txt",
		gameParagraphs)
}
