M.__index = M
local LETTERS = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ'
function M.new() return setmetatable({root = {children = {}, terminal = false}, nodes = 1, words = 0}, M) end
function M:add(word)
    local node = self.root
    for i = 1, #word do
        local letter = word:sub(i, i)
        if not node.children[letter] then node.children[letter] = {children = {}, terminal = false}; self.nodes = self.nodes + 1 end
        node = node.children[letter]
    end
    if node.terminal then return false end
    node.terminal, self.words = true, self.words + 1
    return true
end
function M:prefix(word)
    local node = self.root
    for i = 1, #word do node = node.children[word:sub(i, i)]; if not node then return nil end end
    return node
end
function M:contains(word)
    local node = self:prefix(word)
    return node ~= nil and node.terminal
end
local M = {TILE_ORDER = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ?'}
function M.generator(seed) return {state = seed} end
function M.next_random(random) random.state = random.state * 48271 % 2147483647; return random.state end
function M.shuffle(bag, random)
    for position = #bag, 2, -1 do local other = M.next_random(random) % position + 1; bag[position], bag[other] = bag[other], bag[position] end
end
function M.make_bag(counts, random)
    local bag = {}
    for i = 1, 27 do local letter = M.TILE_ORDER:sub(i, i); for _ = 1, counts[letter] do bag[#bag + 1] = letter end end
    M.shuffle(bag, random)
    return bag
end
function M.draw(bag, count)
    local drawn = {}
    for _ = 1, math.min(count, #bag) do drawn[#drawn + 1] = table.remove(bag) end
    return table.concat(drawn)
end

local M = {LETTERS = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ'}
local function cell_key(row, column) return (row - 1) * 15 + column end
function M.anchors(position)
    if #board.tiles(position) == 0 then return {{8, 8}} end
    local result = {}
    for r = 1, 15 do
        for c = 1, 15 do
            if position[r][c] == '.' then
                for _, neighbor in ipairs(board.neighbors(r, c)) do if position[neighbor[1]][neighbor[2]] ~= '.' then result[#result + 1] = {r, c}; break end end
            end
        end
    end
    return result
end
function M.cross_checks(position, trie)
    local result = {}
    for r = 1, 15 do
        for c = 1, 15 do
            if position[r][c] == '.' then
                local first, last = r, r
                while first > 1 and position[first - 1][c] ~= '.' do first = first - 1 end
                while last < 15 and position[last + 1][c] ~= '.' do last = last + 1 end
                if first ~= last then
                    local prefix, suffix = {}, {}
                    for row = first, r - 1 do prefix[#prefix + 1] = position[row][c] end
                    for row = r + 1, last do suffix[#suffix + 1] = position[row][c] end
                    local before, after, allowed = table.concat(prefix):upper(), table.concat(suffix):upper(), {}
                    for i = 1, 26 do local letter = M.LETTERS:sub(i, i); if trie:contains(before .. letter .. after) then allowed[letter] = true end end
                    result[cell_key(r, c)] = allowed
                end
            end
        end
    end
    return result
end
function M.across_moves(position, rack, trie, anchors, checks)
    local result, counts, anchor_set = {}, {}, {}
    for i = 1, #rack do local letter = rack:sub(i, i); counts[letter] = (counts[letter] or 0) + 1 end
    for _, anchor in ipairs(anchors) do anchor_set[cell_key(anchor[1], anchor[2])] = true end
    for _, square in ipairs(anchors) do
        local row, anchor = square[1], square[2]
        local function extend(node, column, word, start)
            local empty = column == 16 or position[row][column] == '.'
            if empty and column > anchor and node.terminal then result[#result + 1] = {row = row, column = start, direction = 0, word = word} end
            if column == 16 then return end
            if not empty then
                local letter = position[row][column]
                local child = node.children[letter:upper()]
                if child then extend(child, column + 1, word .. letter, start) end
                return
            end
            local allowed = checks[cell_key(row, column)]
            for i = 1, 26 do
                local letter = M.LETTERS:sub(i, i)
                local child = node.children[letter]
                if child and (not allowed or allowed[letter]) then
                    for _, option in ipairs({{letter, letter}, {'?', letter:lower()}}) do
                        local tile, written = option[1], option[2]
                        if (counts[tile] or 0) > 0 then counts[tile] = counts[tile] - 1; extend(child, column + 1, word .. written, start); counts[tile] = counts[tile] + 1 end
                    end
                end
            end
        end
        local function left_part(node, prefix, remaining)
            extend(node, anchor, prefix, anchor - #prefix)
            if remaining == 0 then return end
            for i = 1, 26 do
                local letter = M.LETTERS:sub(i, i)
                local child = node.children[letter]
                if child then
                    for _, option in ipairs({{letter, letter}, {'?', letter:lower()}}) do
                        local tile, written = option[1], option[2]
                        if (counts[tile] or 0) > 0 then counts[tile] = counts[tile] - 1; left_part(child, prefix .. written, remaining - 1); counts[tile] = counts[tile] + 1 end
                    end
                end
            end
        end
        if anchor > 1 and position[row][anchor - 1] ~= '.' then
            local first = anchor - 1
            while first > 1 and position[row][first - 1] ~= '.' do first = first - 1 end
            local letters = {}
            for c = first, anchor - 1 do letters[#letters + 1] = position[row][c] end
            local prefix = table.concat(letters)
            local node = trie:prefix(prefix:upper())
            if node then extend(node, anchor, prefix, first) end
        else
            local first = anchor - 1
            while first >= 1 and position[row][first] == '.' and not anchor_set[cell_key(row, first)] do first = first - 1 end
            left_part(trie.root, '', anchor - first - 1)
        end
    end
    return result
end
function M.placement_key(tiles)
    local parts = {}
    for _, tile in ipairs(tiles) do parts[#parts + 1] = {tile[1], tile[2], tile[3]} end
    table.sort(parts, function(a, b) return a[1] < b[1] or a[1] == b[1] and (a[2] < b[2] or a[2] == b[2] and a[3] < b[3]) end)
    local result = {}
    for _, tile in ipairs(parts) do result[#result + 1] = tile[1] .. ':' .. tile[2] .. ':' .. tile[3] end
    return table.concat(result, ';')
end
function M.generate(position, rack, trie)
    local anchors = M.anchors(position)
    local moves = M.across_moves(position, rack, trie, anchors, M.cross_checks(position, trie))
    local transposed, swapped = board.transpose(position), {}
    for _, anchor in ipairs(anchors) do swapped[#swapped + 1] = {anchor[2], anchor[1]} end
    table.sort(swapped, function(a, b) return a[1] < b[1] or a[1] == b[1] and a[2] < b[2] end)
    for _, move in ipairs(M.across_moves(transposed, rack, trie, swapped, M.cross_checks(transposed, trie))) do moves[#moves + 1] = {row = move.column, column = move.row, direction = 1, word = move.word} end
    local unique, result = {}, {}
    for _, move in ipairs(moves) do
        local tiles, keep = scoring.placement(position, move), true
        if #tiles == 1 and move.direction == 1 then
            local tile = tiles[1]
            if #board.run_at(scoring.apply(position, tiles), tile[1], tile[2], 0).word >= 2 then keep = false end
        end
        if keep then
            local key = M.placement_key(tiles)
            if not unique[key] then unique[key] = true; result[#result + 1] = move end
        end
    end
    return result
end
return M

local M = {RACK_SIZE = 7, BINGO_BONUS = 50}
function M.squares(move)
    local result = {}
    for i = 1, #move.word do result[#result + 1] = {move.row + move.direction * (i - 1), move.column + (1 - move.direction) * (i - 1), move.word:sub(i, i)} end
    return result
end
function M.placement(position, move)
    local result = {}
    for _, cell in ipairs(M.squares(move)) do if position[cell[1]][cell[2]] == '.' then result[#result + 1] = cell end end
    return result
end
function M.apply(position, tiles)
    local result = {}
    for r = 1, 15 do result[r] = {}; for c = 1, 15 do result[r][c] = position[r][c] end end
    for _, tile in ipairs(tiles) do result[tile[1]][tile[2]] = tile[3] end
    return result
end
function M.cross_words(position, move)
    local tiles = M.placement(position, move)
    local after, result = M.apply(position, tiles), {}
    for _, tile in ipairs(tiles) do local word = board.run_at(after, tile[1], tile[2], 1 - move.direction); if #word.word >= 2 then result[#result + 1] = word end end
    return result
end
function M.word_score(position, word, values)
    local score, multiplier = 0, 1
    for _, cell in ipairs(M.squares(word)) do
        local r, c, letter = table.unpack(cell)
        local value = letter:match('^[A-Z]$') and values[letter] or 0
        if position[r][c] == '.' then
            local premium = board.PREMIUMS[r][c]
            value = value * (premium == '+' and 2 or premium == '^' and 3 or 1)
            multiplier = multiplier * (premium == '*' and 2 or premium == '#' and 3 or 1)
        end
        score = score + value
    end
    return score * multiplier
end
function M.score_move(position, move, values)
    local words, scores, score = {move}, {}, 0
    for _, word in ipairs(M.cross_words(position, move)) do words[#words + 1] = word end
    for _, word in ipairs(words) do local value = M.word_score(position, word, values); scores[#scores + 1] = {word, value}; score = score + value end
    local tiles = M.placement(position, move)
    if #tiles == 7 then score = score + 50 end
    return score, tiles, scores
end
return M
local M = {SIZE = 15, CENTER = 8, COLUMNS = 'ABCDEFGHIJKLMNO'}
local premium_rows = {'#..+...#...+..#', '.*...^...^...*.', '..*...+.+...*..', '+..*...+...*..+', '....*.....*....', '.^...^...^...^.', '..+...+.+...+..', '#..+...*...+..#', '..+...+.+...+..', '.^...^...^...^.', '....*.....*....', '+..*...+...*..+', '..*...+.+...*..', '.*...^...^...*.', '#..+...#...+..#'}
M.PREMIUMS = {}
for r, row in ipairs(premium_rows) do M.PREMIUMS[r] = {}; for c = 1, 15 do M.PREMIUMS[r][c] = row:sub(c, c) end end
function M.inside(row, column) return row >= 1 and row <= 15 and column >= 1 and column <= 15 end
function M.square_name(row, column) return M.COLUMNS:sub(column, column) .. row end
function M.position(move)
    local column = M.COLUMNS:sub(move.column, move.column)
    return move.direction == 0 and move.row .. column or column .. move.row
end
function M.empty()
    local result = {}
    for r = 1, 15 do result[r] = {}; for c = 1, 15 do result[r][c] = '.' end end
    return result
end
function M.transpose(board)
    local result = M.empty()
    for r = 1, 15 do for c = 1, 15 do result[r][c] = board[c][r] end end
    return result
end
function M.tiles(board)
    local result = {}
    for r = 1, 15 do for c = 1, 15 do if board[r][c] ~= '.' then result[#result + 1] = {r, c, board[r][c]} end end end
    return result
end
function M.neighbors(row, column)
    local result = {}
    for _, cell in ipairs({{row - 1, column}, {row + 1, column}, {row, column - 1}, {row, column + 1}}) do if M.inside(cell[1], cell[2]) then result[#result + 1] = cell end end
    return result
end
function M.run_at(board, row, column, direction)
    local dr, dc = direction, 1 - direction
    while M.inside(row - dr, column - dc) and board[row - dr][column - dc] ~= '.' do row, column = row - dr, column - dc end
    local first_row, first_column, letters = row, column, {}
    while M.inside(row, column) and board[row][column] ~= '.' do letters[#letters + 1] = board[row][column]; row, column = row + dr, column + dc end
    return {row = first_row, column = first_column, direction = direction, word = table.concat(letters)}
end

    local result = {}

    local letters = {}
    for i = 1, #text do letters[i] = text:sub(i, i) end

    return table.concat(letters)

local function valid_number(text, minimum, maximum)
    if not text:match('^%d+$') or #text > 1 and text:sub(1, 1) == '0' then return false end
    local value = tonumber(text)
    return value ~= nil and value >= minimum and value <= maximum
end

    local result = {}
    for _, letter in ipairs(sorted_keys(counts)) do result[#result + 1] = string.rep(letter, counts[letter]) end
    return table.concat(result)
end

local function strategy_valid(strategy) return strategy == 'score' or strategy == 'tiles' or strategy == 'equity' end
local function job_error(job, seen)
    local header = fields(job[1])
    local kind, name = header[1], header[2]
    if not valid_name(name) then return 'bad name' end
    if seen[name] then return 'duplicate name' end
    if kind == 'TOPN' then
        if not valid_number(header[3], 1, 99) then return 'bad count' end
        if not strategy_valid(header[4]) then return 'bad order' end
        local rack = fields(job[2])[2]
        if not valid_rack(rack) then return 'bad rack' end
        return board_error(read_board(job, 4, #job), rack)
    elseif kind == 'VALIDATE' then
        local last = #job
        for i = 3, #job do if job[i]:sub(1, 4) == 'MOVE' then last = i - 1; break end end
        return board_error(read_board(job, 3, last), '')
    end
    if not valid_number(header[3], 1, 2147483646) then return 'bad seed' end
    if #job - 1 < 2 or #job - 1 > 4 then return 'bad player count ' .. (#job - 1) end
    local player_names = {}
    for i = 2, #job do
        local player = fields(job[i])
        if not valid_name(player[2]) or not strategy_valid(player[3]) or kind == 'GAME' and not valid_number(player[4], 0, 99) then return 'bad player ' .. (i - 1) end
        if player_names[player[2]] then return 'duplicate player ' .. (i - 1) end
        player_names[player[2]] = true
    end
    return nil
end
local function leave_value(rack)
    local counts, duplicates, vowels, consonants, total = count_chars(rack), 0, 0, 0, 0
    for i = 1, #rack do total = total + weights[rack:sub(i, i)] end
    for i = 1, 26 do
        local letter = LETTERS:sub(i, i)
        local count = counts[letter] or 0
        duplicates = duplicates + math.max(0, count - 1)
        if VOWELS:find(letter, 1, true) then vowels = vowels + count else consonants = consonants + count end
    end
    return total - 4 * duplicates - 3 * math.max(0, math.abs(vowels - consonants) - 1)
end
local function evaluate(position, rack, move)
    local score, tiles, word_scores = scoring.score_move(position, move, values)
    local removed, words = {}, {}
    for _, tile in ipairs(tiles) do removed[#removed + 1] = tile_kind(tile[3]) end
    for _, entry in ipairs(word_scores) do words[#words + 1] = entry[1].word .. ':' .. entry[2] end
    local leave = remove_tiles(rack, table.concat(removed))
    return {move = move, score = score, new_tiles = tiles, words = table.concat(words, ','), leave = leave, value = leave_value(leave)}
end
local function standard_less(a, b)
    if a.score ~= b.score then return a.score > b.score end
    local x, y = a.move, b.move
    if x.row ~= y.row then return x.row < y.row end
    if x.column ~= y.column then return x.column < y.column end
    if x.direction ~= y.direction then return x.direction < y.direction end
    return x.word < y.word
end
local function strategy_less(a, b, strategy)
    if strategy == 'tiles' and #a.new_tiles ~= #b.new_tiles then return #a.new_tiles > #b.new_tiles end
    if strategy == 'equity' and a.score + a.value ~= b.score + b.value then return a.score + a.value > b.score + b.value end
    return standard_less(a, b)
end
local function chosen_move(legal, strategy)
    local best = legal[1]
    for i = 2, #legal do if strategy_less(legal[i], best, strategy) then best = legal[i] end end
    return best
end
local function notation(evaluated) return board.position(evaluated.move) .. ' ' .. evaluated.move.word end
local function legal_moves(position, rack)
    local moves = {}
    for _, move in ipairs(generator.generate(position, rack, trie)) do moves[#moves + 1] = evaluate(position, rack, move) end
    table.sort(moves, standard_less)
    return moves
end
local function duplicate_game(job)
    local header = fields(job[1])
    local name, seed = header[2], tonumber(header[3])
    local random = game.generator(seed)
    local bag = game.make_bag(tile_counts, random)
    local position, rack, players, totals = board.empty(), game.draw(bag, 7), {}, {}
    local lines = {'DUPLICATE ' .. name .. ' seed=' .. seed .. ' players=' .. (#job - 1)}
    for i = 2, #job do
        local player = fields(job[i])
        players[#players + 1], totals[#totals + 1] = {player[2], player[3]}, 0
        lines[#lines + 1] = 'player ' .. (i - 1) .. ' ' .. player[2] .. ' strategy=' .. player[3]
    end
    local round, master_total, reason = 0, 0, 'stuck'
    while true do
        local legal = legal_moves(position, rack)
        if #legal == 0 then break end
        round = round + 1
        local master, picks = legal[1], {}
        for i, player in ipairs(players) do local move = chosen_move(legal, player[2]); totals[i] = totals[i] + move.score; picks[i] = move.score end
        master_total = master_total + master.score
        lines[#lines + 1] = 'round ' .. round .. ' rack=' .. collection(rack) .. ' moves=' .. #legal .. ' master ' .. notation(master) .. ' score=' .. master.score .. ' picks=' .. table.concat(picks, ',')
        position = scoring.apply(position, master.new_tiles)
        rack = master.leave .. game.draw(bag, 7 - #master.leave)
        if #rack == 0 then reason = 'out'; break end
    end
    lines[#lines + 1] = 'end reason=' .. reason .. ' rounds=' .. round .. (reason == 'stuck' and ' rack=' .. collection(rack) or '')
    lines[#lines + 1] = 'master total=' .. master_total
    local maximum = totals[1]
    for _, total in ipairs(totals) do maximum = math.max(maximum, total) end
    local winners = {}
    for i, player in ipairs(players) do
        local percent = master_total >= 1 and 100 * totals[i] // master_total or 0
        lines[#lines + 1] = 'final ' .. player[1] .. ' total=' .. totals[i] .. ' percent=' .. percent
        if totals[i] == maximum then winners[#winners + 1] = player[1] end
    end
    lines[#lines + 1] = 'winner ' .. table.concat(winners, ',')
    lines[#lines + 1] = position_summary(position)
    append(lines, board.display(position))
    add_total(game_totals, 'rounds', round)
    return lines
end
