export interface Node {
  children: Map<string, Node>;
  terminal: boolean;
}
const makeNode = (): Node => ({ children: new Map<string, Node>(), terminal: false });
export class Trie {
  root = makeNode();
  nodes = 1;
  words = 0;
  add(word: string) {
    let node = this.root;
    for (const letter of word) {
      let child = node.children.get(letter);
      if (!child) {
        child = makeNode();
        node.children.set(letter, child);
        this.nodes++;
      }
      node = child;
    }
    if (node.terminal) return false;
    node.terminal = true;
    this.words++;
    return true;
  }
  find(prefix: string): Node | null {
    let node = this.root;
    for (const letter of prefix) {
      const child = node.children.get(letter);
      if (!child) return null;
      node = child;
    }
    return node;
  }
  has(word: string) {
    return this.find(word)?.terminal ?? false;
  }

export class Minstd {
  constructor(private state: number) {}
  next() {
    this.state = (48271 * this.state) % 2147483647;
    return this.state;
  }
}
export function shuffle(bag: string[], random: Minstd) {
  for (let i = bag.length - 1; i >= 1; i--) {
    const j = random.next() % (i + 1);
    [bag[i], bag[j]] = [bag[j], bag[i]];
  }
}
export function makeBag(table: Table, random: Minstd) {
  const bag: string[] = [];
  for (const [letter, tile] of table) for (let i = 0; i < tile.count; i++) bag.push(letter);
  shuffle(bag, random);
  return bag;
}
export function draw(bag: string[], count: number) {
  let tiles = '';
  for (let i = 0; i < count && bag.length; i++) tiles += bag.pop()!;
  return tiles;
}

const ALPHABET = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ';
export function anchors(grid: Grid) {
  const result = new Set<number>();
  if (!board.tiles(grid).length) result.add(board.CENTER * board.SIZE + board.CENTER);
  else {
    for (let r = 0; r < board.SIZE; r++) {
      for (let c = 0; c < board.SIZE; c++) {
        if (grid[r][c] === '.' && board.adjacent(grid, r, c)) result.add(r * board.SIZE + c);
      }
    }
  }
  return result;
}
export function crossChecks(grid: Grid, trie: Trie) {
  const allowed: (Set<string> | null)[][] = Array.from({ length: board.SIZE }, () =>
    Array(board.SIZE).fill(null),
  );
  let checked = 0;
  let dead = 0;
  let total = 0;
  for (let row = 0; row < board.SIZE; row++) {
    for (let column = 0; column < board.SIZE; column++) {
      if (grid[row][column] !== '.') continue;
      const letters = new Set<string>();
      if (!board.occupied(grid, row - 1, column) && !board.occupied(grid, row + 1, column)) {
        for (const letter of ALPHABET) letters.add(letter);
      } else {
        checked++;
        for (const letter of ALPHABET) {
          if (trie.has(crossWord(grid, { row, column, letter }, 'down').word.toUpperCase())) {
            letters.add(letter);
          }
        }
        if (!letters.size) dead++;
        total += letters.size;
      }
      allowed[row][column] = letters;
    }
  }
  return { allowed, stats: [checked, dead, total] };
}
function acrossMoves(
  grid: Grid,
  rack: string,
  trie: Trie,
  anchorSet: Set<number>,
  checks: ReturnType<typeof crossChecks>,
) {
  const result: Move[] = [];
  const counts = new Map<string, number>();
  for (const tile of rack) counts.set(tile, (counts.get(tile) ?? 0) + 1);
  for (const anchor of anchorSet) {
    const row = Math.floor(anchor / board.SIZE);
    const anchorColumn = anchor % board.SIZE;
    function choices(
      node: Node,
      allowed: Set<string> | null,
      use: (node: Node, letter: string) => void,
    ) {
      for (const letter of [...node.children.keys()].sort()) {
        if (allowed !== null && !allowed.has(letter)) continue;
        for (const tile of [letter, '?']) {
          if (!(counts.get(tile) ?? 0)) continue;
          counts.set(tile, counts.get(tile)! - 1);
          use(node.children.get(letter)!, tile === '?' ? letter.toLowerCase() : letter);
          counts.set(tile, counts.get(tile)! + 1);
        }
      }
    }
    function extend(column: number, node: Node, word: string, start: number, filled: boolean) {
      const occupied = column < board.SIZE && grid[row][column] !== '.';
      if (!occupied && filled && node.terminal) {
        const tiles: Tile[] = [];
        for (let i = 0; i < word.length; i++) {
          if (grid[row][start + i] === '.') tiles.push({ row, column: start + i, letter: word[i] });
        }
        result.push({ row, column: start, direction: 'across', word, tiles });
      }
      if (column >= board.SIZE) return;
      if (occupied) {
        const letter = grid[row][column];
        const child = node.children.get(letter.toUpperCase());
        if (child) extend(column + 1, child, word + letter, start, filled);
      } else {
        choices(node, checks.allowed[row][column], (child, letter) =>
          extend(column + 1, child, word + letter, start, filled || column === anchorColumn),
        );
      }
    }
    if (board.occupied(grid, row, anchorColumn - 1)) {
      let start = anchorColumn - 1;
      while (board.occupied(grid, row, start - 1)) start--;
      const prefix = grid[row].slice(start, anchorColumn).join('');
      const node = trie.find(prefix.toUpperCase());
      if (node) extend(anchorColumn, node, prefix, start, false);
    } else {
      let limit = 0;
      while (
        anchorColumn - limit - 1 >= 0 &&
        grid[row][anchorColumn - limit - 1] === '.' &&
        !anchorSet.has(row * board.SIZE + anchorColumn - limit - 1)
      ) {
        limit++;
      }
      const leftPart = (node: Node, word: string) => {
        extend(anchorColumn, node, word, anchorColumn - word.length, false);
        if (word.length < limit) {
          choices(node, null, (child, letter) => leftPart(child, word + letter));
        }
      };
      leftPart(trie.root, '');
    }
  }
  return result;
}
export function generate(grid: Grid, rack: string, trie: Trie) {
  const anchorSet = anchors(grid);
  const across = crossChecks(grid, trie);
  const transposed = board.transpose(grid);
  const down = crossChecks(transposed, trie);
  const moves = acrossMoves(grid, rack, trie, anchorSet, across);
  for (const move of acrossMoves(transposed, rack, trie, anchors(transposed), down)) {
    const tiles = board.transposeTiles(move.tiles);
    if (
      tiles.length === 1 &&
      (board.occupied(grid, tiles[0].row, tiles[0].column - 1) ||
        board.occupied(grid, tiles[0].row, tiles[0].column + 1))
    ) {
      continue;
    }
    moves.push({ ...move, row: move.column, column: move.row, direction: 'down', tiles });
  }
  return { moves, anchors: anchorSet.size, across: across.stats, down: down.stats };
}

export const tileKind = (letter: string) => (letter >= 'a' && letter <= 'z' ? '?' : letter);
export const squares = (move: Word): Tile[] =>
  Array.from(move.word, (letter, i) => ({
    row: move.row + (move.direction === 'down' ? i : 0),
    column: move.column + (move.direction === 'across' ? i : 0),
    letter,
  }));
export function crossWord(grid: Grid, tile: Tile, direction: Direction): Word {
  const dr = direction === 'down' ? 1 : 0;
  const dc = direction === 'across' ? 1 : 0;
  let row = tile.row;
  let column = tile.column;
  while (board.occupied(grid, row - dr, column - dc)) {
    row -= dr;
    column -= dc;
  }
  const start = { row, column, direction };
  let word = '';
  while (
    board.inside(row, column) &&
    ((row === tile.row && column === tile.column) || board.occupied(grid, row, column))
  ) {
    word += row === tile.row && column === tile.column ? tile.letter : grid[row][column];
    row += dr;
    column += dc;
  }
  return { ...start, word };
}
export function wordScore(grid: Grid, word: Word, table: Table) {
  let total = 0;
  let multiplier = 1;
  for (const tile of squares(word)) {
    let points = table.get(tileKind(tile.letter))!.value;
    if (grid[tile.row][tile.column] === '.') {
      const premium = board.PREMIUMS[tile.row][tile.column];
      if (premium === '+') points *= 2;
      if (premium === '^') points *= 3;
      if (premium === '*') multiplier *= 2;
      if (premium === '#') multiplier *= 3;
    }
    total += points;
  }
  return total * multiplier;
}
export function score(grid: Grid, move: Word, table: Table): ScoredMove {
  const tiles = squares(move).filter((t) => grid[t.row][t.column] === '.');
  const words: Word[] = [
    { row: move.row, column: move.column, direction: move.direction, word: move.word },
  ];
  const perpendicular = move.direction === 'across' ? 'down' : 'across';
  for (const tile of tiles) {
    const cross = crossWord(grid, tile, perpendicular);
    if (cross.word.length >= 2) words.push(cross);
  }
  let total = tiles.length === RACK_SIZE ? BINGO : 0;
  const scored = words.map((word) => {
    const points = wordScore(grid, word, table);
    total += points;
    return { ...word, score: points };
  });
  return { ...move, tiles, words: scored, score: total };
}

export function connected(grid: Grid) {
  const placed = tiles(grid);
  if (!placed.length) return true;
  const queue: Point[] = [placed[0]];
  const seen = new Set([placed[0].row * SIZE + placed[0].column]);
  for (let i = 0; i < queue.length; i++) {
    for (const [dr, dc] of [
      [-1, 0],
      [1, 0],
      [0, -1],
      [0, 1],
    ]) {
      const row = queue[i].row + dr;
      const column = queue[i].column + dc;
      const key = row * SIZE + column;
      if (occupied(grid, row, column) && !seen.has(key)) {
        seen.add(key);
        queue.push({ row, column });
      }
    }
  }
  return seen.size === placed.length;

function readInput() {
  const table: Table = new Map();
  for (const line of readLines('input/tiles.txt')) {
    if (!line || line.startsWith('#')) continue;
    const [letter, value, count, weight] = line.split(' ');
    table.set(letter, { value: Number(value), count: Number(count), weight: Number(weight) });
  }
  const trie = new Trie();
  const stats = { lines: 0, skipped: 0, duplicates: 0, lengths: new Map<number, number>() };
  for (const line of readLines('input/dictionary.txt')) {
    if (!line || line.startsWith('#')) continue;
    stats.lines++;
    if (!/^[A-Z]{2,15}$/.test(line)) stats.skipped++;
    else if (!trie.add(line)) stats.duplicates++;
    else stats.lengths.set(line.length, (stats.lengths.get(line.length) ?? 0) + 1);
  }
  const lines = readLines('input/jobs.txt');
  const jobs: Job[] = [];
  for (let i = 0; i < lines.length; i++) {
    if (!lines[i] || lines[i].startsWith('#')) continue;
    const [raw, name, ...args] = lines[i].split(' ');
    const kind: JobKind =
      raw === 'TOPN' ? raw : raw === 'VALIDATE' ? raw : raw === 'GAME' ? raw : 'DUPLICATE';
    const body: string[] = [];
    while (lines[++i] !== 'END') body.push(lines[i]);
    jobs.push({ kind, name, args, body });
  }
  return { table, trie, stats, jobs };
}
function checkJob(
  job: Job,
  duplicate: boolean,
  table: Table,
  trie: Trie,
): Checked<TopSettings | ValidationSettings | GameSettings> {
  if (!nameValid(job.name)) return { error: 'bad name' };
  if (duplicate) return { error: 'duplicate name' };
  if (job.kind === 'TOPN') {
    if (!numberValid(job.args[0], 1, 99)) return { error: 'bad count' };
    const strategy = job.args[1];
    if (!strategyValid(strategy)) return { error: 'bad order' };
    const rack = job.body[0].split(' ')[1];
    if (!rackValid(rack)) return { error: 'bad rack' };
    const checked = checkBoard(job.body.slice(2), rack, table, trie);
    return 'error' in checked
      ? checked
      : { kind: 'TOPN', ...checked, rack, count: Number(job.args[0]), strategy };
  }
  if (job.kind === 'VALIDATE') {
    let at = 1;
    while (at < job.body.length && !job.body[at].startsWith('MOVE')) at++;
    const checked = checkBoard(job.body.slice(1, at), '', table, trie);
    return 'error' in checked
      ? checked
      : { kind: 'VALIDATE', ...checked, records: job.body.slice(at) };
  }
  if (!numberValid(job.args[0], 1, 2147483646)) return { error: 'bad seed' };
  if (job.body.length < 2 || job.body.length > 4) {
    return { error: `bad player count ${job.body.length}` };
  }
  const players: Player[] = [];
  const names = new Set<string>();
  for (let i = 0; i < job.body.length; i++) {
    const [, name, strategy, threshold] = job.body[i].split(' ');
    if (
      !nameValid(name) ||
      !strategyValid(strategy) ||
      (job.kind === 'GAME' && !numberValid(threshold, 0, 99))
    ) {
      return { error: `bad player ${i + 1}` };
    }
    if (names.has(name)) return { error: `duplicate player ${i + 1}` };
    names.add(name);
    players.push({
      name,
      strategy,
      threshold: job.kind === 'GAME' ? Number(threshold) : 0,
      index: i,
    });
  }
  return { kind: job.kind, players, seed: Number(job.args[0]) };
}
function leaveValue(tiles: string, table: Table) {
  const counts = frequencies(tiles);
  let value = 0;
  let duplicates = 0;
  let vowels = 0;
  let consonants = 0;
  for (const tile of tiles) {
    value += table.get(tile)!.weight;
    if (VOWELS.includes(tile)) vowels++;
    else if (tile !== '?') consonants++;
  }
  for (const [tile, count] of counts) if (tile !== '?') duplicates += count - 1;
  return value - 4 * duplicates - 3 * Math.max(0, Math.abs(vowels - consonants) - 1);
}
function removeTiles(rack: string, used: Iterable<string>) {
  const left = Array.from(rack);
  for (const tile of used) left.splice(left.indexOf(tile), 1);
  return left.join('');
}
function standardOrder(a: LegalMove, b: LegalMove) {
  return (
    b.score - a.score ||
    a.row - b.row ||
    a.column - b.column ||
    (a.direction === b.direction ? 0 : a.direction === 'across' ? -1 : 1) ||
    ascii(a.word, b.word)
  );
}
function strategyOrder(moves: LegalMove[], strategy: Strategy) {
  return moves
    .slice()
    .sort(
      (a, b) =>
        (strategy === 'tiles'
          ? b.tiles.length - a.tiles.length
          : strategy === 'equity'
            ? b.score + b.leaveValue - (a.score + a.leaveValue)
            : 0) || standardOrder(a, b),
    );
}
function legalMoves(grid: Grid, rack: string, trie: Trie, table: Table) {
  const generated = generate(grid, rack, trie);
  const moves = generated.moves
    .map((move): LegalMove => {
      const result = scoring.score(grid, move, table);
      const leave = removeTiles(
        rack,
        result.tiles.map((t) => scoring.tileKind(t.letter)),
      );
      return { ...result, leave, leaveValue: leaveValue(leave, table), rank: 0 };
    })
    .sort(standardOrder);
  moves.forEach((move, i) => (move.rank = i + 1));
  return { ...generated, moves };
}
function place(grid: Grid, move: Pick<ScoredMove, 'tiles'>) {
  for (const tile of move.tiles) grid[tile.row][tile.column] = tile.letter;
}
function duplicateJob(job: Job, settings: GameSettings, trie: Trie, table: Table) {
  const random = new game.Minstd(settings.seed);
  const bag = game.makeBag(table, random);
  const grid = board.empty();
  const players = settings.players.map((p) => ({ ...p, total: 0 }));
  const lines = [`DUPLICATE ${job.name} seed=${settings.seed} players=${players.length}`];
  players.forEach((p, i) => lines.push(`player ${i + 1} ${p.name} strategy=${p.strategy}`));
  let rack = game.draw(bag, scoring.RACK_SIZE);
  let masterTotal = 0;
  let rounds = 0;
  let reason: 'out' | 'stuck';
  while (true) {
    const moves = legalMoves(grid, rack, trie, table).moves;
    if (!moves.length) {
      reason = 'stuck';
      break;
    }
    rounds++;
    const master = moves[0];
    const picks: number[] = [];
    for (const player of players) {
      const move = strategyOrder(moves, player.strategy)[0];
      player.total += move.score;
      picks.push(move.score);
    }
    masterTotal += master.score;
    lines.push(
      `round ${rounds} rack=${collection(rack)} moves=${moves.length} master ${notation(master)} score=${master.score} picks=${picks.join(',')}`,
    );
    place(grid, master);
    rack = master.leave;
    rack += game.draw(bag, scoring.RACK_SIZE - rack.length);
    if (!rack.length) {
      reason = 'out';
      break;
    }
  }
  lines.push(
    reason === 'out'
      ? `end reason=out rounds=${rounds}`
      : `end reason=stuck rounds=${rounds} rack=${collection(rack)}`,
    `master total=${masterTotal}`,
  );
  for (const player of players) {
    lines.push(
      `final ${player.name} total=${player.total} percent=${masterTotal > 0 ? Math.floor((100 * player.total) / masterTotal) : 0}`,
    );
  }
  const maximum = Math.max(...players.map((p) => p.total));
  lines.push(
    'winner ' +
      players
        .filter((p) => p.total === maximum)
        .map((p) => p.name)
        .join(','),
    positionSummary(grid),
    ...board.display(grid),
  );
  return { lines, rounds };
}

    if ('error' in settings) {
      if (job.kind === 'TOPN') topTotal.errors++;
      else if (job.kind === 'VALIDATE') validationTotal.errors++;
      else gameTotal.errors++;
      paragraphs.push([`${job.kind} ${job.name}`, 'error ' + settings.error]);
      continue;
    }
    if (settings.kind === 'TOPN') {
      const result = topJob(job, settings, trie, table);
      moves.push(result.lines);
      topTotal.moves += result.moves;
      topTotal.bingos += result.bingos;
    } else if (settings.kind === 'VALIDATE') {
      const result = validationJob(job, settings, trie, table);
      validation.push(result.lines);
      validationTotal.moves += result.moves;
      validationTotal.valid += result.valid;
      validationTotal.invalid += result.invalid;
      validationTotal.loss += result.loss;
    } else if (settings.kind === 'GAME') {
      const result = gameJob(job, settings, trie, table);
      games.push(result.lines);
      gameTotal.turns += result.turns;
      gameTotal.plays += result.plays;
      gameTotal.exchanges += result.exchanges;
      gameTotal.passes += result.passes;
      gameTotal.bingos += result.bingos;
    } else {
      const result = duplicateJob(job, settings, trie, table);
      games.push(result.lines);
      gameTotal.rounds += result.rounds;
    }
  }
  moves.push([
    `SUMMARY jobs=${topTotal.jobs} errors=${topTotal.errors} moves=${topTotal.moves} bingos=${topTotal.bingos}`,
  ]);
  validation.push([
    `SUMMARY jobs=${validationTotal.jobs} errors=${validationTotal.errors} moves=${validationTotal.moves} valid=${validationTotal.valid} invalid=${validationTotal.invalid} loss=${validationTotal.loss}`,
  ]);
  games.push([
    `SUMMARY games=${gameTotal.games} duplicates=${gameTotal.duplicates} errors=${gameTotal.errors} turns=${gameTotal.turns} rounds=${gameTotal.rounds} plays=${gameTotal.plays} exchanges=${gameTotal.exchanges} passes=${gameTotal.passes} bingos=${gameTotal.bingos}`,
  ]);
  mkdirSync('output', { recursive: true });
  writeParagraphs('moves.txt', moves);
  writeParagraphs('validation.txt', validation);
  writeParagraphs('game.txt', games);
}
