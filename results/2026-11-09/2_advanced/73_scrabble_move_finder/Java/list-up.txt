import java.util.*;
final class DictionaryTrie {
    static final class Node {
        Node[] children = new Node[26];
        boolean terminal;
    }
    final Node root = new Node();
    int nodes = 1, words;
    boolean add(String word) {
        Node node = root;
        for (char letter : word.toCharArray()) {
            int i = letter - 'A';
            if (node.children[i] == null) {
                node.children[i] = new Node();
                nodes++;
            }
            node = node.children[i];
        }
        if (node.terminal)
            return false;
        node.terminal = true;
        words++;
        return true;
    }
    Node prefix(String word) {
        Node node = root;
        for (char letter : word.toCharArray()) {
            node = node.children[letter - 'A'];
            if (node == null)
                return null;
        }
        return node;
    }
    boolean contains(String word) {
        Node node = prefix(word);
        return node != null && node.terminal;
    }
import java.util.*;
final class Game {
    static final String TILE_ORDER = "ABCDEFGHIJKLMNOPQRSTUVWXYZ?";
    static final class Minstd {
        long state;
        Minstd(int seed) { state = seed; }
        int next() {
            state = state * 48271 % 2147483647;
            return (int)state;
        }
    }
    static void shuffle(List<Character> bag, Minstd random) {
        for (int i = bag.size() - 1; i >= 1; i--) {
            int j = random.next() % (i + 1);
            char tile = bag.get(i);
            bag.set(i, bag.get(j));
            bag.set(j, tile);
        }
    }
    static List<Character> bag(int[] counts, Minstd random) {
        List<Character> result = new ArrayList<>();
        for (char tile : TILE_ORDER.toCharArray())
            for (int i = 0; i < counts[tile]; i++)
                result.add(tile);
        shuffle(result, random);
        return result;
    }
    static String draw(List<Character> bag, int count) {
        StringBuilder tiles = new StringBuilder();
        while (count-- > 0 && !bag.isEmpty())
            tiles.append(bag.removeLast());
        return tiles.toString();
    }
final class MoveGenerator {
    static final String LETTERS = "ABCDEFGHIJKLMNOPQRSTUVWXYZ";
    static boolean[][] anchors(char[][] board) {
        boolean[][] result = new boolean[15][15];
        if (Board.tiles(board).isEmpty()) {
            result[7][7] = true;
            return result;
        }
        for (int r = 0; r < 15; r++)
            for (int c = 0; c < 15; c++)
                if (board[r][c] == '.')
                    for (Board.Cell neighbor : Board.neighbors(r, c))
                        if (board[neighbor.row()][neighbor.column()] != '.')
                            result[r][c] = true;
        return result;
    }
    static int[][] checks(char[][] board, DictionaryTrie trie) {
        int[][] result = new int[15][15];
        for (int[] row : result)
            Arrays.fill(row, -1);
        for (int r = 0; r < 15; r++)
            for (int c = 0; c < 15; c++)
                if (board[r][c] == '.') {
                    int first = r, last = r;
                    while (first > 0 && board[first - 1][c] != '.')
                        first--;
                    while (last + 1 < 15 && board[last + 1][c] != '.')
                        last++;
                    if (first == last)
                        continue;
                    StringBuilder prefix = new StringBuilder(), suffix = new StringBuilder();
                    for (int i = first; i < r; i++)
                        prefix.append(Character.toUpperCase(board[i][c]));
                    for (int i = r + 1; i <= last; i++)
                        suffix.append(Character.toUpperCase(board[i][c]));
                    int mask = 0;
                    for (int i = 0; i < 26; i++)
                        if (trie.contains(prefix.toString() + (char)('A' + i) + suffix))
                            mask |= 1 << i;
                    result[r][c] = mask;
                }
        return result;
    }
    static final class Search {
        char[][] board;
        DictionaryTrie trie;
        boolean[][] anchors;
        int[][] checks;
        int[] counts = new int[128];
        List<Board.Move> result = new ArrayList<>();
        int row, anchor;
        Search(char[][] board, String rack, DictionaryTrie trie, boolean[][] anchors,
               int[][] checks) {
            this.board = board;
            this.trie = trie;
            this.anchors = anchors;
            this.checks = checks;
            for (char c : rack.toCharArray())
                counts[c]++;
        }
        void extend(DictionaryTrie.Node node, int column, String word, int start) {
            boolean empty = column == 15 || board[row][column] == '.';
            if (empty && column > anchor && node.terminal)
                result.add(new Board.Move(row, start, 0, word));
            if (column == 15)
                return;
            if (!empty) {
                char letter = board[row][column];
                DictionaryTrie.Node child = node.children[Character.toUpperCase(letter) - 'A'];
                if (child != null)
                    extend(child, column + 1, word + letter, start);
                return;
            }
            for (int i = 0; i < 26; i++)
                if (node.children[i] != null &&
                    (checks[row][column] == -1 || (checks[row][column] & (1 << i)) != 0)) {
                    char letter = (char)('A' + i);
                    for (int blank = 0; blank < 2; blank++) {
                        char tile = blank == 0 ? letter : '?',
                             written = blank == 0 ? letter : Character.toLowerCase(letter);
                        if (counts[tile] > 0) {
                            counts[tile]--;
                            extend(node.children[i], column + 1, word + written, start);
                            counts[tile]++;
                        }
                    }
                }
        }
        void left(DictionaryTrie.Node node, String prefix, int remaining) {
            extend(node, anchor, prefix, anchor - prefix.length());
            if (remaining == 0)
                return;
            for (int i = 0; i < 26; i++)
                if (node.children[i] != null) {
                    char letter = (char)('A' + i);
                    for (int blank = 0; blank < 2; blank++) {
                        char tile = blank == 0 ? letter : '?',
                             written = blank == 0 ? letter : Character.toLowerCase(letter);
                        if (counts[tile] > 0) {
                            counts[tile]--;
                            left(node.children[i], prefix + written, remaining - 1);
                            counts[tile]++;
                        }
                    }
                }
        }
        List<Board.Move> generate() {
            for (row = 0; row < 15; row++)
                for (anchor = 0; anchor < 15; anchor++)
                    if (anchors[row][anchor]) {
                        if (anchor > 0 && board[row][anchor - 1] != '.') {
                            int first = anchor - 1;
                            while (first > 0 && board[row][first - 1] != '.')
                                first--;
                            String prefix = new String(board[row], first, anchor - first);
                            DictionaryTrie.Node node = trie.prefix(prefix.toUpperCase(Locale.ROOT));
                            if (node != null)
                                extend(node, anchor, prefix, first);
                        } else {
                            int first = anchor - 1;
                            while (first >= 0 && board[row][first] == '.' && !anchors[row][first])
                                first--;
                            left(trie.root, "", anchor - first - 1);
                        }
                    }
            return result;
        }
    }
    static List<Board.Move> generate(char[][] board, String rack, DictionaryTrie trie) {
        boolean[][] anchors = anchors(board);
        List<Board.Move> across =
            new Search(board, rack, trie, anchors, checks(board, trie)).generate();
        char[][] transposed = Board.transpose(board);
        boolean[][] downAnchors = new boolean[15][15];
        for (int r = 0; r < 15; r++)
            for (int c = 0; c < 15; c++)
                downAnchors[r][c] = anchors[c][r];
        List<Board.Move> down =
            new Search(transposed, rack, trie, downAnchors, checks(transposed, trie)).generate();
        for (Board.Move move : down)
            across.add(new Board.Move(move.column(), move.row(), 1, move.word()));
        Map<List<Board.Tile>, Board.Move> distinct = new LinkedHashMap<>();
        for (Board.Move move : across) {
            List<Board.Tile> tiles = Scoring.placement(board, move);
            if (tiles.size() == 1 && move.direction() == 1) {
                Board.Tile tile = tiles.getFirst();
                if (Board.run(Scoring.apply(board, tiles), tile.row(), tile.column(), 0)
                        .word()
                        .length() >= 2)
                    continue;
            }
            tiles.sort(
                Comparator.comparingInt(Board.Tile::row).thenComparingInt(Board.Tile::column));
            distinct.put(List.copyOf(tiles), move);
        }
        return new ArrayList<>(distinct.values());
    }
}
final class Scoring {
    static final int RACK_SIZE = 7, BINGO_BONUS = 50;
    record WordScore(Board.Move word, int score) {}
    record Result(int score, List<Board.Tile> tiles, List<WordScore> words) {}
    static List<Board.Tile> squares(Board.Move move) {
        List<Board.Tile> result = new ArrayList<>();
        for (int i = 0; i < move.word().length(); i++)
            result.add(new Board.Tile(move.row() + move.direction() * i,
                                      move.column() + (1 - move.direction()) * i,
                                      move.word().charAt(i)));
        return result;
    }
    static List<Board.Tile> placement(char[][] board, Board.Move move) {
        List<Board.Tile> result = new ArrayList<>();
        for (Board.Tile tile : squares(move))
            if (board[tile.row()][tile.column()] == '.')
                result.add(tile);
        return result;
    }
    static char[][] apply(char[][] board, List<Board.Tile> tiles) {
        char[][] result = new char[board.length][];
        for (int r = 0; r < board.length; r++)
            result[r] = board[r].clone();
        for (Board.Tile tile : tiles)
            result[tile.row()][tile.column()] = tile.letter();
        return result;
    }
    static List<Board.Move> crosses(char[][] board, Board.Move move) {
        List<Board.Tile> tiles = placement(board, move);
        char[][] after = apply(board, tiles);
        List<Board.Move> result = new ArrayList<>();
        for (Board.Tile tile : tiles) {
            Board.Move word = Board.run(after, tile.row(), tile.column(), 1 - move.direction());
            if (word.word().length() >= 2)
                result.add(word);
        }
        return result;
    }
    static int wordScore(char[][] board, Board.Move word, int[] values) {
        int score = 0, multiplier = 1;
        for (Board.Tile tile : squares(word)) {
            char letter = tile.letter();
            int value = letter >= 'A' && letter <= 'Z' ? values[letter] : 0;
            if (board[tile.row()][tile.column()] == '.') {
                char premium = Board.PREMIUMS[tile.row()].charAt(tile.column());
                value *= premium == '+' ? 2 : premium == '^' ? 3 : 1;
                multiplier *= premium == '*' ? 2 : premium == '#' ? 3 : 1;
            }
            score += value;
        }
        return score * multiplier;
    }
    static Result evaluate(char[][] board, Board.Move move, int[] values) {
        List<Board.Move> words = new ArrayList<>();
        words.add(move);
        words.addAll(crosses(board, move));
        List<WordScore> scores = new ArrayList<>();
        int score = 0;
        for (Board.Move word : words) {
            int value = wordScore(board, word, values);
            score += value;
            scores.add(new WordScore(word, value));
        }
        List<Board.Tile> tiles = placement(board, move);
        if (tiles.size() == RACK_SIZE)
            score += BINGO_BONUS;
        return new Result(score, tiles, scores);
    }
}
import java.util.*;
final class Board {
    static final int SIZE = 15;
    static final String COLUMNS = "ABCDEFGHIJKLMNO";
    static final String[] PREMIUMS = {"#..+...#...+..#", ".*...^...^...*.", "..*...+.+...*..",
                                      "+..*...+...*..+", "....*.....*....", ".^...^...^...^.",
                                      "..+...+.+...+..", "#..+...*...+..#", "..+...+.+...+..",
                                      ".^...^...^...^.", "....*.....*....", "+..*...+...*..+",
                                      "..*...+.+...*..", ".*...^...^...*.", "#..+...#...+..#"};
    record Move(int row, int column, int direction, String word) {}
    record Tile(int row, int column, char letter) {}
    record Cell(int row, int column) {}
    static boolean inside(int r, int c) { return 0 <= r && r < SIZE && 0 <= c && c < SIZE; }
    static String square(int r, int c) { return "" + COLUMNS.charAt(c) + (r + 1); }
    static String position(Move move) {
        return move.direction() == 0 ? "" + (move.row() + 1) + COLUMNS.charAt(move.column())
                                     : "" + COLUMNS.charAt(move.column()) + (move.row() + 1);
    }
    static Move parse(String text, String word) {
        if (text.matches("([1-9]|1[0-5])[A-O]"))
            return new Move(Integer.parseInt(text.substring(0, text.length() - 1)) - 1,
                            COLUMNS.indexOf(text.charAt(text.length() - 1)), 0, word);
        if (text.matches("[A-O]([1-9]|1[0-5])"))
            return new Move(Integer.parseInt(text.substring(1)) - 1,
                            COLUMNS.indexOf(text.charAt(0)), 1, word);
        return null;
    }
    static char[][] empty() {
        char[][] board = new char[SIZE][SIZE];
        for (char[] row : board)
            Arrays.fill(row, '.');
        return board;
    }
    static char[][] transpose(char[][] board) {
        char[][] result = new char[SIZE][SIZE];
        for (int r = 0; r < SIZE; r++)
            for (int c = 0; c < SIZE; c++)
                result[r][c] = board[c][r];
        return result;
    }
    static List<Tile> transposeTiles(List<Tile> tiles) {
        List<Tile> result = new ArrayList<>();
        for (Tile tile : tiles)
            result.add(new Tile(tile.column(), tile.row(), tile.letter()));
        return result;
    }
    static List<Tile> tiles(char[][] board) {
        List<Tile> result = new ArrayList<>();
        for (int r = 0; r < SIZE; r++)
            for (int c = 0; c < SIZE; c++)
                if (board[r][c] != '.')
                    result.add(new Tile(r, c, board[r][c]));
        return result;
    }
    static List<Cell> neighbors(int r, int c) {
        List<Cell> result = new ArrayList<>();
        for (int[] delta : new int[][] {{-1, 0}, {1, 0}, {0, -1}, {0, 1}})
            if (inside(r + delta[0], c + delta[1]))
                result.add(new Cell(r + delta[0], c + delta[1]));
        return result;
    }
    static Move run(char[][] board, int r, int c, int direction) {
        int dr = direction, dc = 1 - direction;
        while (inside(r - dr, c - dc) && board[r - dr][c - dc] != '.') {
            r -= dr;
            c -= dc;
        }
        int firstRow = r, firstColumn = c;
        StringBuilder word = new StringBuilder();
        while (inside(r, c) && board[r][c] != '.') {
            word.append(board[r][c]);
            r += dr;
            c += dc;
        }
        return new Move(firstRow, firstColumn, direction, word.toString());
    }
    static final Comparator<Evaluated> STANDARD =
        Comparator.comparingInt((Evaluated e) -> - e.score())
            .thenComparingInt(e -> e.move().row())
            .thenComparingInt(e -> e.move().column())
            .thenComparingInt(e -> e.move().direction())
            .thenComparing(e -> e.move().word());
    static boolean validName(String text) { return text.matches("[a-z0-9_]{1,20}"); }
    static boolean validNumber(String text, int low, int high) {
        if (!text.matches("0|[1-9][0-9]*") || text.length() > 10)
            return false;
        long value = Long.parseLong(text);
        return low <= value && value <= high;
    }
    static String collection(String text) {
        char[] tiles = text.toCharArray();
        Arrays.sort(tiles);
        return tiles.length == 0 ? "-" : new String(tiles);
    }
    static String removeTiles(String rack, String removed) {
        int[] remaining = new int[128];
        for (char c : rack.toCharArray())
            remaining[c]++;
        for (char c : removed.toCharArray())
            remaining[c]--;
        StringBuilder result = new StringBuilder();
        for (int c = 0; c < 128; c++)
            for (int i = 0; i < remaining[c]; i++)
                result.append((char)c);
        return result.toString();
    }
    static String notation(Evaluated move) {
        return Board.position(move.move()) + " " + move.move().word();
    }
    static Comparator<Evaluated> strategy(String name) {
        return name.equals("tiles")
            ? Comparator.comparingInt((Evaluated move) -> - move.tiles().size())
                  .thenComparing(STANDARD)
        : name.equals("equity")
            ? Comparator.comparingInt((Evaluated move) -> - move.score() - move.value())
                  .thenComparing(STANDARD)
            : STANDARD;
    }
    static String jobError(List<String> job, Set<String> seen) {
        String[] fields = job.getFirst().split(" ");
        String kind = fields[0], name = fields[1];
        if (!validName(name))
            return "bad name";
        if (seen.contains(name))
            return "duplicate name";
        if (kind.equals("TOPN")) {
            if (!validNumber(fields[2], 1, 99))
                return "bad count";
            if (!STRATEGIES.contains(fields[3]))
                return "bad order";
            String rack = job.get(1).split(" ")[1];
            if (!validRack(rack))
                return "bad rack";
            return boardError(board(job.subList(3, job.size())), rack);
        }
        if (kind.equals("VALIDATE")) {
            List<String> rows = new ArrayList<>();
            for (int i = 2; i < job.size() && !job.get(i).startsWith("MOVE"); i++)
                rows.add(job.get(i));
            return boardError(board(rows), "");
        }
        if (!validNumber(fields[2], 1, 2147483646))
            return "bad seed";
        if (job.size() - 1 < 2 || job.size() - 1 > 4)
            return "bad player count " + (job.size() - 1);
        Set<String> names = new HashSet<>();
        for (int i = 1; i < job.size(); i++) {
            String[] player = job.get(i).split(" ");
            if (!validName(player[1]) || !STRATEGIES.contains(player[2]) ||
                kind.equals("GAME") && !validNumber(player[3], 0, 99))
                return "bad player " + i;
            if (!names.add(player[1]))
                return "duplicate player " + i;
        }
        return null;
    }
    static int leaveValue(String rack) {
        int[] counts = new int[128];
        int value = 0, duplicates = 0, vowels = 0, consonants = 0;
        for (char c : rack.toCharArray()) {
            counts[c]++;
            value += weights[c];
            if (VOWELS.indexOf(c) >= 0)
                vowels++;
            else if (c != '?')
                consonants++;
        }
        for (char c : MoveGenerator.LETTERS.toCharArray())
            duplicates += Math.max(0, counts[c] - 1);
        return value - 4 * duplicates - 3 * Math.max(0, Math.abs(vowels - consonants) - 1);
    }
    static Evaluated evaluate(char[][] board, String rack, Board.Move move) {
        Scoring.Result score = Scoring.evaluate(board, move, values);
        StringBuilder removed = new StringBuilder();
        for (Board.Tile tile : score.tiles())
            removed.append(tileKind(tile.letter()));
        String leave = removeTiles(rack, removed.toString());
        List<String> words = new ArrayList<>();
        for (Scoring.WordScore word : score.words())
            words.add(word.word().word() + ":" + word.score());
        return new Evaluated(move, score.score(), score.tiles(), String.join(",", words), leave,
                             leaveValue(leave));
    }
    static List<Evaluated> legal(char[][] board, String rack) {
        List<Evaluated> result = new ArrayList<>();
        for (Board.Move move : MoveGenerator.generate(board, rack, trie))
            result.add(evaluate(board, rack, move));
        result.sort(STANDARD);
        return result;
    }
    static List<String> duplicate(List<String> job) {
        String[] header = job.getFirst().split(" ");
        String name = header[1];
        int seed = Integer.parseInt(header[2]);
        Game.Minstd random = new Game.Minstd(seed);
        List<Character> bag = Game.bag(counts, random);
        char[][] board = Board.empty();
        String rack = Game.draw(bag, 7);
        List<Player> players = new ArrayList<>();
        List<String> lines = new ArrayList<>();
        lines.add("DUPLICATE " + name + " seed=" + seed + " players=" + (job.size() - 1));
        for (int i = 1; i < job.size(); i++) {
            String[] fields = job.get(i).split(" ");
            players.add(new Player(fields[1], fields[2], 0, ""));
            lines.add("player " + i + " " + fields[1] + " strategy=" + fields[2]);
        }
        int rounds = 0, masterTotal = 0;
        String reason = "stuck";
        while (true) {
            List<Evaluated> legal = legal(board, rack);
            if (legal.isEmpty())
                break;
            rounds++;
            Evaluated master = legal.getFirst();
            List<String> picks = new ArrayList<>();
            for (Player player : players) {
                Evaluated move = Collections.min(legal, strategy(player.strategy));
                player.score += move.score();
                picks.add(Integer.toString(move.score()));
            }
            masterTotal += master.score();
            lines.add("round " + rounds + " rack=" + collection(rack) + " moves=" + legal.size() +
                      " master " + notation(master) + " score=" + master.score() +
                      " picks=" + String.join(",", picks));
            board = Scoring.apply(board, master.tiles());
            rack = master.leave() + Game.draw(bag, 7 - master.leave().length());
            if (rack.isEmpty()) {
                reason = "out";
                break;
            }
        }
        lines.add("end reason=" + reason + " rounds=" + rounds +
                  (reason.equals("stuck") ? " rack=" + collection(rack) : ""));
        lines.add("master total=" + masterTotal);
        int maximum = Integer.MIN_VALUE;
        for (Player player : players) {
            int percent = masterTotal >= 1 ? 100 * player.score / masterTotal : 0;
            lines.add("final " + player.name + " total=" + player.score + " percent=" + percent);
            maximum = Math.max(maximum, player.score);
        }
        List<String> winners = new ArrayList<>();
        for (Player player : players)
            if (player.score == maximum)
                winners.add(player.name);
        lines.add("winner " + String.join(",", winners));
        lines.add(positionSummary(board));
        lines.addAll(Board.display(board));
        add(gameTotals, "rounds", rounds);
        return lines;
    }
    public static void main(String[] args) throws Exception {
        for (String line : Files.readAllLines(Path.of("input", "tiles.txt")))

                String[] fields = line.split(" ");

                values[tile] = Integer.parseInt(fields[1]);
                counts[tile] = Integer.parseInt(fields[2]);
                weights[tile] = Integer.parseInt(fields[3]);
            }
        int candidates = 0, skipped = 0, duplicates = 0;
        int[] lengths = new int[16];
        for (String line : Files.readAllLines(Path.of("input", "dictionary.txt"))) {
            if (line.isEmpty() || line.startsWith("#"))
                continue;
            candidates++;
            if (!line.matches("[A-Z]{2,15}"))
                skipped++;
            else if (trie.add(line))
                lengths[line.length()]++;
            else
                duplicates++;
        }
        StringBuilder histogram = new StringBuilder("lengths");
        for (int i = 2; i <= 15; i++)
            if (lengths[i] > 0)
                histogram.append(" ").append(i).append("=").append(lengths[i]);
        List<String> dictionary = List.of("DICTIONARY lines=" + candidates +
                                              " words=" + trie.words + " skipped=" + skipped +
                                              " duplicates=" + duplicates + " nodes=" + trie.nodes,
                                          histogram.toString());
        List<List<String>> jobs = new ArrayList<>();
        List<String> current = null;
        for (String line : Files.readAllLines(Path.of("input", "jobs.txt"))) {
            if (current == null) {
                if (!line.isEmpty() && !line.startsWith("#")) {
                    current = new ArrayList<>();
                    current.add(line);
                }
            } else if (line.equals("END")) {
                jobs.add(current);
                current = null;
            } else
                current.add(line);
        }
        List<List<String>> moves = new ArrayList<>(), validation = new ArrayList<>(),
                           games = new ArrayList<>();
        moves.add(dictionary);
        Set<String> seen = new HashSet<>();
        for (List<String> job : jobs) {
            String[] header = job.getFirst().split(" ");
            String kind = header[0], name = header[1];
            List<List<String>> destination = kind.equals("TOPN")       ? moves
                                             : kind.equals("VALIDATE") ? validation
                                                                       : games;
            Map<String, Integer> totals = kind.equals("TOPN")       ? topTotals
                                          : kind.equals("VALIDATE") ? validationTotals
                                                                    : gameTotals;
            add(totals,
                kind.equals("GAME")        ? "games"
                : kind.equals("DUPLICATE") ? "duplicates"
                                           : "jobs",
                1);
            String error = jobError(job, seen);
            seen.add(name);
            if (error != null) {
                destination.add(List.of(kind + " " + name, "error " + error));
                add(totals, "errors", 1);
            } else
                destination.add(switch (kind) {
                    case "TOPN" -> topn(job);
                    case "VALIDATE" -> validate(job);
                    case "GAME" -> ordinary(job);
                    default -> duplicate(job);
                });
        }
        moves.add(List.of(summary(topTotals, "jobs", "errors", "moves", "bingos")));
        validation.add(List.of(
            summary(validationTotals, "jobs", "errors", "moves", "valid", "invalid", "loss")));
        games.add(List.of(summary(gameTotals, "games", "duplicates", "errors", "turns", "rounds",
                                  "plays", "exchanges", "passes", "bingos")));
        Files.createDirectories(Path.of("output"));
        write("moves.txt", moves);
        write("validation.txt", validation);
        write("game.txt", games);
    }
