function kernel(string $name, float $u): float
{
    $t = 1 - $u * $u;
    if ($name === 'epanechnikov') {
        return 0.75 * $t;
    }
    if ($name === 'biweight') {
        return 0.9375 * $t * $t;
    }
    return 1.09375 * $t * $t * $t;
}
