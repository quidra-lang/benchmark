int modulo(int value)
    result := value % 26
    return if result < 0 then result + 26 else result
string transform(string text, string cipher, string key, bool decrypt)
    multiplier := 1
    int[] plain := []
    result := ""
    for: position < #text
        character := text[position]
        value := scoring.letter_value(character)
        if value < 0
            result += character
            continue
        if cipher == "vigenere"
            shift = scoring.letter_value(key[#plain % #key])
        elif cipher == "autokey"
            shift = if #plain < #key then scoring.letter_value(key[#plain]) else plain[#plain - #key]
        transformed := modulo(if decrypt then inverse(multiplier) * (value - shift) else multiplier * value + shift)
        plain = plain.append(if decrypt then transformed else value)
        result += if scoring.alphabet.contains(character) then scoring.alphabet[transformed] else scoring.lowercase[transformed]
    return result
const alphabet := "ABCDEFGHIJKLMNOPQRSTUVWXYZ"
const lowercase := "abcdefghijklmnopqrstuvwxyz"
int letter_value(string character)
    for: 0 <= index < 26
        if character == alphabet[index] or character == lowercase[index]
            return index
    return -1
