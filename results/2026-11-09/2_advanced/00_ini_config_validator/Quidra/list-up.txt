class Scalar(string notation, measure := 0, truth := false)
string lowercase(string text)
    result := ""
    for: index < #text
        character := text[index]
        match "ABCDEFGHIJKLMNOPQRSTUVWXYZ".find(character)
            nat offset
                result += "abcdefghijklmnopqrstuvwxyz"[offset]
            none
                result += character
    return result
bool characters_in(string text, string allowed)
    for: index < #text
        if not allowed.contains(text[index])
            return false
    return true
bool digits(string text, nat minimum, nat maximum)
    return minimum <= #text <= maximum and characters_in(text, "0123456789")
int | none size_value(string text)
    nat position := 0
    while position < #text and digits(text[position], 1, 1)
        position += 1
    whole := text[0:<position]
    if not digits(whole, 1, 6)
        return none
    fraction := ""
    if position < #text and text[position] == "."
        position += 1
        start := position
        while position < #text and digits(text[position], 1, 1)
            position += 1
        fraction = text[start:<position]
        if not digits(fraction, 1, 3)
            return none
    unit := lowercase(text[position:<#text])
    factor := 1
    if unit == "kb"
        factor = 1024
    elif unit == "mb"
        factor = 1048576
    elif unit == "gb"
        factor = 1073741824
    elif unit != "" and unit != "b"
        return none
    int result := int(whole) * factor
    if fraction != ""
        divisor := 1
        for _ < #fraction
            divisor *= 10
        int scaled := int(fraction) * factor
        if scaled % divisor != 0
            return none
        result += scaled / divisor
    return result
Scalar | none scalar_value(string kind, string text, string[] allowed)
    if kind == "int" or kind == "size" or kind == "duration" or kind == "date"
        int | none number := none
        if kind == "int"
            number = integer_value(text)
        elif kind == "size"
            number = size_value(text)
        match number
            int value
                return Scalar(emitter.scalar_notation(kind, text, value), value)
            none
                return none
void report_invalid(Profile &profile, string key, string kind, string[] allowed,
                    string text, bool secret, bool item, lexer.Origin origin)
    display := if secret then "********" else emitter.string_notation(text)
    if kind == "enum"
        .profile.report(origin, "E_ENUM", key + ": " + display + " is not one of " + allowed.join(","))
    else
        .profile.report(origin, "E_TYPE", key + ": " + display + " is not a valid " + kind)
Scalar | none normalize(Profile &profile, string key, schema.Spec spec,
                        string text, lexer.Origin origin)
    secret := spec.has("secret")
    string[] allowed := if spec.has("values") then spec.attribute("values").split(",") else []
    Scalar normalized
    if spec.kind == "list"
    else
        match scalar_value(spec.kind, text, allowed)
            Scalar accepted
                normalized = accepted
            none
                report_invalid(&profile, key, spec.kind, allowed, text, secret, false, origin)
                return none
    if spec.has("range")
        bounds := spec.attribute("range").split("..")
        string[] displayed_bounds := []
        outside := false
        for: bound := bounds[index]
            if bound == ""
                displayed_bounds = displayed_bounds.append("")
                continue
            measure := 0
            notation := ""
            if spec.kind == "string" or spec.kind == "list"
                measure = int(bound)
                notation = string(measure)
            else
                match scalar_value(spec.kind, bound, [])
                    Scalar valid_bound
                        measure = valid_bound.measure
                        notation = valid_bound.notation
                    none
                        return none
            displayed_bounds = displayed_bounds.append(notation)
            if (index == 0 and normalized.measure < measure) or (index == 1 and normalized.measure > measure)
                outside = true
        if outside
            display := if secret then "********" else normalized.notation
            .profile.report(origin, "E_RANGE", key + ": " + display + " is out of range " + displayed_bounds.join(".."))
            return none
    return normalized
string scalar_notation(string kind, string text, number := 0, truth := false)
    if kind == "int" or kind == "size" or kind == "duration"
        return string(number)
