export type Scalar = string | number | boolean;
export type Value = Scalar | Scalar[];
function normalizeSize(text: string): number | null {
  const match = /^([0-9]{1,6})(?:\.([0-9]{1,3}))?(B|KB|MB|GB)?$/i.exec(text);
  if (!match) return null;
  const factors: Record<string, number> = { B: 1, KB: 1024, MB: 1048576, GB: 1073741824 };
  const factor = factors[(match[3] || 'B').toUpperCase()];
  let value = Number(match[1]) * factor;
  if (match[2] !== undefined) {
    const numerator = Number(match[2]) * factor;
    const denominator = 10 ** match[2].length;
    if (numerator % denominator !== 0) return null;
    value += numerator / denominator;
  }
  return value;
}

function checkedValue(
  key: string,
  text: string,
  spec: Spec,
  origin: Origin,
  report: Reporter,
): CheckedValue | null {
  const attributes = spec.attributes;
  const secret = attributes.has('secret');
  const typeError = (type: ScalarType, part: string, item: boolean) => {
    const display = (item ? 'item ' : '') + (secret ? MASK : stringNotation(part));
    if (type === 'enum') {
      report(origin, 'E_ENUM', `${key}: ${display} is not one of ${attributes.get('values')}`);
    } else report(origin, 'E_TYPE', `${key}: ${display} is not a valid ${type}`);
  };
  let value: Value;
  if (spec.type === 'list') {
  } else {
    const normalized = normalizeScalar(spec.type, text, attributes);
    if (normalized === null) {
      typeError(spec.type, text, false);
      return null;
    }
    value = normalized;
  }
  const range = attributes.get('range');
  if (range !== undefined) {
    const count = spec.type === 'string' || spec.type === 'list';
    const bounds = range
      .split('..')
      .map((bound) =>
        bound === ''
          ? null
          : count
            ? Number(bound)
            : normalizeScalar(spec.type as ScalarType, bound, attributes),
      );
    const measured =
      Array.isArray(value) || (typeof value === 'string' && count) ? value.length : value;
    const outside = bounds.some((bound, i) => {
      if (bound === null) return false;
      if (typeof measured === 'number' && typeof bound === 'number') {
        return i === 0 ? measured < bound : measured > bound;
      }
      return false;
    });
    if (outside) {
      const display = count
        ? `${spec.type === 'string' ? 'length' : 'count'} ${measured}`
        : secret
          ? MASK
          : valueNotation(spec.type, value, attributes);
      report(
        origin,
        'E_RANGE',
        `${key}: ${display} is out of range ${bounds.map((bound) => (bound === null ? '' : String(bound))).join('..')}`,
      );
      return null;
    }
  }
  return { value, notation: valueNotation(spec.type, value, attributes) };
}
