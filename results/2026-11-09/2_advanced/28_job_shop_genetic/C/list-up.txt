static char *percent(Integer a, Integer b) {
    Integer value = (2000 * a + b) / (2 * b);
    char *text = malloc(64);
    snprintf(text, 64, "%" PRId64 ".%" PRId64 "%%", value / 10, value % 10);
    return text;
}
