acceptance =: 3 : 0
    transaction =. y
    if. is_coinbase transaction do.
        ('coinbase';0),<0$a: return.
    end.
    if. ((<tx_id transaction)e.Confirmed)+.(find_entry tx_id transaction)<#Mempool do.
        ('duplicate';0),<0$a: return.
    end.
    'reason fee' =. check_transaction (<transaction),(<current_utxo 0),(<0$a:),(<next_height 0),(<Wallets),<Maturity
    if. #reason do.
        (reason;0),<0$a: return.
    end.
    if. (fee*1000)<MinFeeRate*tx_size transaction do.
        ('low-fee';0),<0$a: return.
    end.
    conflicts =. 0$a:
    total =. 0
    higher =. 1
    inputs =. tx_inputs transaction
    for_entry. Mempool do.
        pending =. >entry
        other =. m_transaction pending
        if. +./(tx_inputs other)e.inputs do.
            conflicts =. conflicts,<tx_id other
            total =. total+m_fee pending
            higher =. higher*.(fee*tx_size other)>(m_fee pending)*tx_size transaction
        end.
    end.
    if. (#conflicts)*.((fee<:total)+.-.higher) do.
        ('conflict';0),<0$a: return.
    end.

    record =. parent{::Records

    if. (b_height block)~:1+r_height record do.
        (<'bad-height'),(<utxo),<fees return.
    end.

        (<'bad-difficulty'),(<utxo),<fees return.
    end.
    if. (b_time block)<:Records median_time parent do.
        (<'time-too-old'),(<utxo),<fees return.
    end.

        (<'time-too-new'),(<utxo),<fees return.
    end.

            ((<reason,' tx ',":i+1),<utxo),<fees return.

    end.
    reason =. MaxBlock context_free_error block

        reason =. 'bad-parent'

        i.0 return.
    end.

        Orphans =: Orphans,<(<block),<peer

            log 'EVICT ',(short b_hash o_block 0{::Orphans),' orphan-limit'

    end.

check_transaction =: 3 : 0
    'transaction utxo spent height owners maturity' =. y
    addresses =. tx_addresses transaction
    amounts =. tx_amounts transaction
    known =. wallet_address&.>owners
    if. (0 e.amounts)+.-.*./addresses e.known do.
        'bad-output';0 return.
    end.
    'references coins' =. utxo
    seen =. 0$a:
    total =. 0
    inputs =. tx_inputs transaction
    signatures =. tx_signatures transaction
    for_i. i.#inputs do.
        reference =. i{inputs
        if. (reference e.spent)+.reference e.seen do.
            'double-spend';0 return.
        end.
        if. -.reference e.references do.
            'missing-input';0 return.
        end.
        coin =. (references i.reference){::coins
        if. -.coin is_mature height,maturity do.
            'immature';0 return.
        end.
        owner =. (known i.<coin_address coin){::owners
        expected =. sign (<wallet_secret owner),(<i),<tx_body transaction
        if. -.expected-:i{::signatures do.
            'bad-sig';0 return.
        end.
        seen =. seen,reference
        total =. total+coin_amount coin
    end.
    amount =. tx_total transaction
    if. total<amount do.
        'overspend';0 return.
    end.
    '';total-amount
