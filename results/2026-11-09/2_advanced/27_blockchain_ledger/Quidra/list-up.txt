Check check_transaction(Transaction transaction, map<string, Coin> utxo, distinct<string> spent, int height, map<string, wallets.Wallet> owners, int maturity)
    for output := transaction.outputs[]
        if not owners.has(output.address) or output.amount == 0
            return Check("bad-output")
    distinct<string> seen
    total := 0
    for: index < int(#transaction.inputs)
        reference := transaction.inputs[index]
        if spent.has(reference) or seen.has(reference)
            return Check("double-spend")
        if not utxo.has(reference)
            return Check("missing-input")
        coin := entry(utxo, reference)
        if not coin.mature(height, maturity)
            return Check("immature")
        owner := entry(owners, coin.address)
        if transaction.signatures[index] != wallets.sign(owner.secret, index, transaction.body())
            return Check("bad-sig")
        .seen.add(reference)
        total += coin.amount
    amount := transaction.output_total()
    if total < amount
        return Check("overspend")
    return Check("", total - amount)
