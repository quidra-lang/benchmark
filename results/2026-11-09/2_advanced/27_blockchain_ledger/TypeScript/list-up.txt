export class StoredBlock extends Block {
  readonly work: number;
  constructor(
    block: Block,
    readonly parent: StoredBlock | null,
    readonly utxo: Utxo,
    readonly fees: number,
    readonly transactionFees: number[],
    readonly source: string,
  ) {
    super(
      block.height,
      block.previous,
      block.root,
      block.timestamp,
      block.difficulty,
      block.nonce,
      block.transactions,
    );
    this.work = parent === null ? 1 : parent.work + 16 ** block.difficulty;
  }
}
export function ancestor(block: StoredBlock, height: number): StoredBlock {
  while (block.height > height) block = block.parent!;
  return block;
}
export function sequence(tip: StoredBlock): StoredBlock[] {
  const blocks: StoredBlock[] = [];
  for (let current: StoredBlock | null = tip; current !== null; current = current.parent) {
    blocks.push(current);
  }
  return blocks.reverse();
}
export function forkPoint(first: StoredBlock, second: StoredBlock): StoredBlock {
  first = ancestor(first, Math.min(first.height, second.height));
  second = ancestor(second, first.height);
  while (first.hash !== second.hash) {
    first = first.parent!;
    second = second.parent!;
  }
  return first;
}
export class Block {
  hash: string;
  readonly size: number;
  constructor(
    readonly height: number,
    readonly previous: string,
    readonly root: string,
    readonly timestamp: number,
    readonly difficulty: number,
    public nonce: number,
    readonly transactions: Transaction[],
  ) {
    this.hash = hash(header(this));
    this.size = 0;
    for (const transaction of transactions.slice(1)) this.size += transaction.size;
  }
}
export const header = (block: Block): string =>
  [block.height, block.previous, block.root, block.timestamp, block.difficulty, block.nonce].join(
    '|',
  );
export const proofOfWork = (block: Block): boolean =>
  block.hash.startsWith('0'.repeat(block.difficulty));
export function parseBlock(fields: string[], transactionLines: string[]): Block {
  return new Block(
    Number(fields[3]),
    fields[4],
    fields[5],
    Number(fields[6]),
    Number(fields[7]),
    Number(fields[8]),
    transactionLines.map((text) => new Transaction(text)),
  );
}
export function contextFreeCheck(block: Block, maximumSize: number): ContextReason | null {
  if (!proofOfWork(block)) return 'bad-pow';
  if (merkle(block.transactions.map((t) => t.txid)) !== block.root) return 'bad-merkle';
  if (
    !block.transactions[0].coinbase ||
    block.transactions[0].height !== block.height ||
    block.transactions.slice(1).some((t) => t.coinbase)
  ) {
    return 'bad-structure';
  }
  if (block.size > maximumSize) return 'too-large';
  return null;
}
export type Utxo = Map<string, Coin>;
export const mature = (coin: Coin, height: number, maturity: number): boolean =>
  !coin.coinbase || coin.height === 0 || height >= coin.height + maturity;
export function outputTotal(transaction: Transaction): number {
  let total = 0;
  for (const output of transaction.outputs) total += output.amount;
  return total;
}
export function check(
  transaction: Transaction,
  utxo: Utxo,
  spent: Set<string>,
  height: number,
  maturity: number,
  owners: Map<string, Wallet>,
): CheckResult {
  if (transaction.outputs.some((o) => !owners.has(o.address) || o.amount === 0)) {
    return { ok: false, reason: 'bad-output' };
  }
  const seen = new Set<string>();
  let inputTotal = 0;
  for (let i = 0; i < transaction.inputs.length; i++) {
    const input = transaction.inputs[i];
    if (spent.has(input.key) || seen.has(input.key)) return { ok: false, reason: 'double-spend' };
    seen.add(input.key);
    const coin = utxo.get(input.key);
    if (!coin) return { ok: false, reason: 'missing-input' };
    if (!mature(coin, height, maturity)) return { ok: false, reason: 'immature' };
    if (transaction.signatures[i] !== sign(owners.get(coin.address)!.secret, i, transaction.body)) {
      return { ok: false, reason: 'bad-sig' };
    }
    inputTotal += coin.amount;
  }
  const outputs = outputTotal(transaction);
  return inputTotal < outputs
    ? { ok: false, reason: 'overspend' }
    : { ok: true, fee: inputTotal - outputs };
}
export function apply(transaction: Transaction, utxo: Utxo, height: number): void {
  for (const input of transaction.inputs) utxo.delete(input.key);
  transaction.outputs.forEach((output, index) =>
    utxo.set(`${transaction.txid}:${index}`, {
      ...output,
      txid: transaction.txid,
      index,
      height,
      coinbase: transaction.coinbase,
    }),
  );
}
