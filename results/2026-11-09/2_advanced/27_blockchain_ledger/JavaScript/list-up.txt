class StoredBlock {
    constructor(block, parent, utxo, fees, transactionFees, source) {

function ancestor(block, height) {
    while (block.height > height) block = block.parent;
    return block;
}
function sequence(tip) {
    const blocks = [];
    for (let current = tip; current !== null; current = current.parent) blocks.push(current);
    return blocks.reverse();
}
function forkPoint(first, second) {
    first = ancestor(first, Math.min(first.height, second.height));
    second = ancestor(second, first.height);
    while (first.hash !== second.hash) { first = first.parent; second = second.parent; }
    return first;
}
const header = block => [block.height, block.previous, block.root, block.timestamp, block.difficulty, block.nonce].join('|');
const proofOfWork = block => block.hash.startsWith('0'.repeat(block.difficulty));
class Block {
    constructor(height, previous, root, timestamp, difficulty, nonce, transactions) {
        Object.assign(this, { height, previous, root, timestamp, difficulty, nonce, transactions });
        this.hash = hash(header(this));
        this.size = 0;
        for (const transaction of transactions.slice(1)) this.size += transaction.size;
    }
}
function parseBlock(fields, transactionLines) {
    return new Block(Number(fields[3]), fields[4], fields[5], Number(fields[6]), Number(fields[7]), Number(fields[8]),
        transactionLines.map(text => new Transaction(text)));
}
function contextFreeCheck(block, maximumSize) {
    if (!proofOfWork(block)) return 'bad-pow';
    if (merkle(block.transactions.map(transaction => transaction.txid)) !== block.root) return 'bad-merkle';
    if (!block.transactions[0].coinbase || block.transactions[0].height !== block.height ||
        block.transactions.slice(1).some(transaction => transaction.coinbase)) return 'bad-structure';
    if (block.size > maximumSize) return 'too-large';
    return null;
}
const mature = (coin, height, maturity) => !coin.coinbase || coin.height === 0 || height >= coin.height + maturity;
function outputTotal(transaction) {
    let total = 0;
    for (const output of transaction.outputs) total += output.amount;
    return total;
}
function check(transaction, utxo, spent, height, maturity, owners) {
    if (transaction.outputs.some(output => !owners.has(output.address) || output.amount === 0)) return { reason: 'bad-output' };
    const seen = new Set();
    let inputTotal = 0;
    for (let i = 0; i < transaction.inputs.length; i++) {
        const input = transaction.inputs[i];
        if (spent.has(input.key) || seen.has(input.key)) return { reason: 'double-spend' };
        seen.add(input.key);
        const coin = utxo.get(input.key);
        if (coin === undefined) return { reason: 'missing-input' };
        if (!mature(coin, height, maturity)) return { reason: 'immature' };
        if (transaction.signatures[i] !== sign(owners.get(coin.address).secret, i, transaction.body)) return { reason: 'bad-sig' };
        inputTotal += coin.amount;
    }
    const outputs = outputTotal(transaction);
    return inputTotal < outputs ? { reason: 'overspend' } : { fee: inputTotal - outputs };
}
function apply(transaction, utxo, height) {
    for (const input of transaction.inputs) utxo.delete(input.key);
    for (let index = 0; index < transaction.outputs.length; index++) {
        utxo.set(transaction.txid + ':' + index, { ...transaction.outputs[index], txid: transaction.txid, index,
            height, coinbase: transaction.coinbase });
    }
