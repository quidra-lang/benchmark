class Point(int x, int y)
class Pattern(int version)
    size := 17 + 4 * version
    int[][] cells := [for y := (0:<size)[] then array(nat(size), fill = 0)]
    int[][] areas := [for y := (0:<size)[] then array(nat(size), fill = 0)]
Point[] reading_order(Pattern pattern)
    Point[] positions := []
    right := pattern.size - 1
    pair := 0
    while right >= 1
        if right == 6
            right = 5
        for: offset < pattern.size
            y := if pair % 2 == 0 then pattern.size - 1 - offset else offset
            for: column < 2
                x := right - column
                if pattern.areas[y][x] == 0
                    positions = positions.append(Point(x, y))
        pair += 1
        right -= 2
