def reading_order(areas):
    size = len(areas)
    right_columns = list(range(size - 1, 7, -2)) + [5, 3, 1]
    positions = []
    for pair, right in enumerate(right_columns):
        rows = range(size - 1, -1, -1) if pair % 2 == 0 else range(size)
        for y in rows:
            for x in (right, right - 1):
                if areas[y][x] == DATA:
                    positions.append((x, y))
    return positions
