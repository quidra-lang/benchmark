    fun readingOrder(areas: Array<IntArray>): List<Point> {
        val size=areas.size;
        val order=mutableListOf<Point>();
        var right=size-1;
        var pair=0;
        while(right>=1) {
            if(right==6)right=5;
            for(i in 0 until size) {
                val y=if(pair%2==0)size-1-i else i;
                for(x in right downTo right-1)if(areas[y][x]==DATA)order.add(Point(x,y))
            };
            pair++;
            right-=2
        };
        return order
    }
