MIN_INT = -2147483648
MAX_INT = 2147483647
BINARY = frozenset(("ADD", "SUB", "MUL", "DIV", "MOD", "LT", "LE", "GT", "GE", "EQ", "NE"))
def calculate(op, a, b=None):
    if op == "ADD":
        result = a + b
    if not MIN_INT <= result <= MAX_INT:
        raise RuntimeFault("integer overflow")
    return result
