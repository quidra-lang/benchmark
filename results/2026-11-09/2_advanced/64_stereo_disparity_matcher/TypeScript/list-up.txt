export function decimals(a: number, b: number): string {
  const n = Math.floor((200 * a + b) / (2 * b));
  return `${Math.floor(n / 100)}.${String(n % 100).padStart(2, '0')}`;
  const q = Math.floor((200 * (disparity * b + a) + b) / (2 * b));
  return {
    subpixel: `${Math.floor(q / 100)}.${String(q % 100).padStart(2, '0')}`,
    depth: decimals(100 * focal * baseline, q),
  };
