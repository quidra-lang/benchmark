string uppercase(string text)
    result := ""
    lower := "acgt"
    upper := "ACGT"
    for: i < int(#text)
        character := text[i]
        for: j < 4
            if character == lower[j]
                character = upper[j]
                break
        result += character
    return result
string reverse_complement(string text)
    bases := "ACGTN"
    complements := "TGCAN"
    result := ""
    i := int(#text) - 1
    while i >= 0
        for: j < 5
            if text[i] == bases[j]
                result += complements[j]
                break
        i -= 1
    return result

class Index(string text, int[] suffixes, int occ_interval, int sample_interval, bool sampled)
    size := int(#text)
    string bwt := [for position := suffixes[] then text[if position == 0 then size - 1 else position - 1]].join("")
    int[] totals := [for symbol := (0:<6)[] then 0]
    for: i < size
        totals[suffix_array.symbol_index(text[i])] += 1
    int[] cumulative := [for symbol := (0:<6)[] then 0]
    for: 1 <= symbol < 6
        cumulative[symbol] = cumulative[symbol - 1] + totals[symbol - 1]
    int[][] checkpoints := []
    int[] running := [for symbol := (0:<6)[] then 0]
    for: i <= size
        if i % occ_interval == 0
            checkpoints = checkpoints.append(running)
        if i < size
            running[suffix_array.symbol_index(bwt[i])] += 1
    int[] samples := [for row := (0:<int(size))[] then (if sampled and suffixes[row] % sample_interval == 0 then suffixes[row] else -1)]

    int occurrence(int symbol, int end)
        checkpoint := end / occ_interval
        result := checkpoints[checkpoint][symbol]
        for: checkpoint * occ_interval <= i < end
            if suffix_array.symbol_index(bwt[i]) == symbol
                result += 1
        return result

    int lf(int row)
        symbol := suffix_array.symbol_index(bwt[row])
        return cumulative[symbol] + occurrence(symbol, row)

    int[] extend(int lower, int upper, string base)
        symbol := suffix_array.symbol_index(base)
        if symbol < 2
            return [0, 0]
        return [cumulative[symbol] + occurrence(symbol, lower), cumulative[symbol] + occurrence(symbol, upper)]

    int[] locate(int row)
        current := row
        steps := 0
        while samples[current] < 0
            current = lf(current)
            steps += 1
        return [samples[current] + steps, steps]
    int[] backward_search(string query)
        interval := [0, size]
        i := int(#query) - 1
        while i >= 0
            interval = extend(interval[0], interval[1], query[i])
            if interval[0] >= interval[1]
                return interval
            i -= 1
        return interval
int[] lower_bounds(string sequence, fm_index.Index reverse)
    z := 0
    interval := [0, reverse.size]
    int[] bounds := []
    for: i < int(#sequence)
        interval = reverse.extend(interval[0], interval[1], sequence[i])
        if interval[0] >= interval[1]
            z += 1
            interval = [0, reverse.size]
        bounds = bounds.append(z)
    return bounds
Result backtrack(string sequence, int maximum, fm_index.Index forward, fm_index.Index reverse)
    bounds := lower_bounds(sequence, reverse)
    int[][] stack := [[int(#sequence) - 1, maximum, 0, forward.size]]
    nodes := 0
    int[][] leaves := []
    positions := 0
    bases := "ACGT"
    while #stack != 0
        state := stack[-1]
        stack = stack[0:<int(#stack) - 1]
        nodes += 1
        i := state[0]
        z := state[1]
        lower_bound := if i < 0 then 0 else bounds[i]
        if z < lower_bound
            continue
        if i == -1
            leaves = leaves.append([state[2], state[3], maximum - z])
            positions += state[3] - state[2]
            continue
        base := 3
        while base >= 0
            interval := forward.extend(state[2], state[3], bases[base])
            if interval[0] < interval[1]
                if bases[base] == sequence[i]
                    stack = stack.append([i - 1, z, interval[0], interval[1]])
                elif z >= 1
                    stack = stack.append([i - 1, z - 1, interval[0], interval[1]])
            base -= 1
    return Result(nodes, leaves, positions)
class ReferenceText(sequence_io.Reference[] references)
    int[] offsets := []
    offset := 0
    for reference := references[]
        offsets = offsets.append(offset)
        offset += int(#reference.bases) + 1
    string text := [for reference := references[] then reference.bases].join("#") + "$"
    string reverse := text[0:<int(#text) - 1][::-1] + "$"

    int[] position(int absolute_position)
        for: r < int(#references)
            if offsets[r] <= absolute_position and absolute_position < offsets[r] + int(#references[r].bases)
                return [r, absolute_position - offsets[r]]
        return [-1, -1]
int[][] operation_runs(string operations)
    int[][] runs := []
    alphabet := "MID"
    for: i < int(#operations)
        operation := 0
        for: j < 3
            if operations[i] == alphabet[j]
                operation = j
        last := int(#runs) - 1
        if last >= 0 and runs[last][0] == operation
            runs[last][1] += 1
        else
            runs = runs.append([operation, 1])
    return runs
Alignment describe(int reference, int position, int strand, string operations, int score, string core, int trimmed, sequence_io.Reference[] references)
    oriented := if strand == 0 then core else sequence_io.reverse_complement(core)
    sequence := references[reference].bases
    string[] cigar_parts := []
    string[] md_parts := []
    int[][] pairs := []
    i := 0
    j := position
    matches := 0
    nm := 0
    alphabet := "MID"
    for run := operation_runs(operations)[]
        operation := run[0]
        count := run[1]
        cigar_parts = cigar_parts.append(string(count) + alphabet[operation])
        if operation == 1
            i += count
            nm += count
        elif operation == 2
            md_parts = md_parts.append(string(matches) + "^" + sequence[j:<j + count])
            matches = 0
            j += count
            nm += count
        else
            for: k < count
                pairs = pairs.append([i, j])
                if oriented[i] == sequence[j]
                    matches += 1
                else
                    md_parts = md_parts.append(string(matches) + sequence[j])
                    matches = 0
                    nm += 1
                i += 1
                j += 1
    md_parts = md_parts.append(string(matches))
    if trimmed > 0
        if strand == 0
            cigar_parts = cigar_parts.append("{trimmed}S")
        else
            cigar_parts = ["{trimmed}S"].concat(cigar_parts)
    return Alignment(reference, position, strand, operations, score, oriented, pairs, cigar_parts.join(""), nm, md_parts.join(""))
class Mapper(ReferenceText reference_text, fm_index.Index forward, fm_index.Index reverse, sequence_io.Params params)
    search_counts := [0, 0, 0, 0]
    seed_counts := [0, 0, 0, 0, 0, 0, 0]
    Alignment[] hit_alignments(int[][] hits, Result result)
        size := int(#reference_text.references)
        base := forward.size + 1
        keys := [for i := (0:<int(#hits))[] then (((hits[i][0] * size + hits[i][1]) * base + hits[i][2]) * 2 + hits[i][3]) * int(#hits) + i]
        order := [for key := keys.sorted()[] then key % int(#hits)]
        best := hits[order[0]][0]
        Alignment[] alignments := []
        for index := order[]
            hit := hits[index]
            if hit[0] <= best + 1
                score := (result.length - hit[0]) * params.scores[0] - hit[0] * params.scores[1]
                alignments = alignments.append(describe(hit[1], hit[2], hit[3], "M" * nat(result.length), score, result.core, result.trimmed, reference_text.references))
        return alignments
    Result .map_read(sequence_io.Read read)
        int[] qualities := [for code := read.quality.codepoints()[] then code - 33]
        length := int(#read.bases)
        while length > 0 and qualities[length - 1] < params.trim_quality
            length -= 1
        trimmed := int(#read.bases) - length
        core := read.bases[0:<length]
        result := Result(read, core, qualities, length, trimmed, false, "SHORT", "", 0, 0, 0, 0, [], [])
        if length < params.min_length
            return result
        oriented := [core, sequence_io.reverse_complement(core)]
        plus_exact := forward.backward_search(oriented[0])
        minus_exact := forward.backward_search(oriented[1])
        allowed := if plus_exact[0] < plus_exact[1] or minus_exact[0] < minus_exact[1] then 1 else params.mismatches
        int[][] hits := []
        for: strand < 2
            found := search.backtrack(oriented[strand], allowed, forward, reverse)
            search_counts[0] += 1
            search_counts[1] += found.nodes
            search_counts[2] += int(#found.leaves)
            search_counts[3] += found.positions
            for leaf := found.leaves[]
                for: leaf[0] <= row < leaf[1]
                    located := reference_text.position(forward.locate(row)[0])
                    hits = hits.append([leaf[2], located[0], located[1], strand])
        if #hits != 0
            result.alignments = hit_alignments(hits, result)
            result.mapped = true
            best := length
            for hit := hits[]
                if hit[0] < best
                    best = hit[0]
            result.x0 = int(#[for hit := hits[] if hit[0] == best then hit])
            result.x1 = int(#[for hit := hits[] if hit[0] == best + 1 then hit])
            result.method = if best == 0 then "EXACT" else "MISMATCH"
            result.score = (length - best) * params.scores[0] - best * params.scores[1]
            result.mapq = if result.x0 >= 2 then 0 elif result.x1 == 0 then 60 - 12 * best else maximum(30 - 12 * best - 3 * result.x1, 1)
            for: r < int(#reference_text.references)
                if #[for hit := hits[] if hit[0] == best and hit[1] == r then hit] != 0
                    result.targets = result.targets.append(r)
            return result
string sam_record(Result result, sequence_io.Reference[] references, int maximum_alternatives)
    read := result.read
    if not result.mapped
        return sam_writer.unmapped(read.name, read.bases, read.quality, result.reason)
    primary := result.alignments[0]
    reverse := primary.strand == 1
    bases := if reverse then sequence_io.reverse_complement(read.bases) else read.bases
    quality := if reverse then read.quality[::-1] else read.quality
    alternatives := ""
    others := int(#result.alignments) - 1
    if others >= 1 and others <= maximum_alternatives
        for: 1 <= i < int(#result.alignments)
            item := result.alignments[i]
            strand := if item.strand == 0 then "+" else "-"
            alternatives += references[item.reference].name + ",{strand}{item.position + 1},{item.cigar},{item.nm};"
    return sam_writer.mapped(read.name, if reverse then 16 else 0, references[primary.reference].name, primary.position, result.mapq, primary.cigar, bases, quality, primary.nm, primary.md, result.score, result.x0, result.x1, result.method, alternatives)
class Study(sequence_io.Input data)
    sequence_io.Params params := data.params
    sequence_io.Reference[] references := data.references
    reference_text := ReferenceText(references)
    string text := reference_text.text
    size := int(#text)
    suffix_array.Built forward_sa := suffix_array.build(text)
    suffix_array.Built reverse_sa := suffix_array.build(reference_text.reverse)
    forward := fm_index.Index(text, forward_sa.suffixes, params.occ_interval, params.sample_interval, true)
    reverse := fm_index.Index(reference_text.reverse, reverse_sa.suffixes, params.occ_interval, params.sample_interval, false)
    mapper := Mapper(reference_text, forward, reverse, params)
    Result[] results := []
    for read := data.reads[]
        results = results.append(.mapper.map_read(read))
    int[][] depths := [for reference := references[] then [for p := (0:<int(#reference.bases))[] then 0]]
    mapq0_starts := depths
    int[][][][] pileup := [for reference := references[] then [for p := (0:<int(#reference.bases))[] then [for strand := (0:<2)[] then [for base := (0:<4)[] then 0]]]]
    for result := results[]
        if not result.mapped
            continue
        local primary := result.alignments[0]
        if result.mapq == 0
            mapq0_starts[primary.reference][primary.position] += 1
        if result.mapq < params.coverage_mapq
            continue
        for pair := primary.pairs[]
            depths[primary.reference][pair[1]] += 1
            local base := suffix_array.symbol_index(primary.oriented[pair[0]]) - 2
            local cycle := if primary.strand == 0 then pair[0] + 1 else result.length - pair[0]
            if base >= 0 and result.qualities[cycle - 1] >= params.variants[1]
                pileup[primary.reference][pair[1]][primary.strand][base] += 1
    string[] sam_lines()
        names := [for reference := references[] then reference.name]
        lengths := [for reference := references[] then int(#reference.bases)]
        lines := sam_writer.header(names, lengths)
        for result := results[]
            lines = lines.append(sam_record(result, references, params.max_alternatives))
        return lines
    string[] coverage_lines()
        lines := ["window {params.coverage_window} min_mapq {params.coverage_mapq}"]
        for: r < int(#references)
            reference := references[r]
            length := int(#reference.bases)
            values := depths[r]
            covered := int(#[for value := values[] if value > 0 then value])
            depth_sum := total(values)
            max_depth := values.sorted()[length - 1]
            lines = lines.append("reference {reference.name} length {length} covered {covered} breadth " + sam_writer.ratio(covered, length) + " mean_depth " + sam_writer.ratio(depth_sum, length) + " max_depth {max_depth}")
            string[] histogram := []
            for: depth <= max_depth
                count := int(#[for value := values[] if value == depth then value])
                histogram = histogram.append("{depth}:{count}")
            lines = lines.append("depth_histogram " + histogram.join(" "))
            for: 0 <= first < length step params.coverage_window
                last := minimum(first + params.coverage_window, length)
                count := last - first
                window := values[first:<last]
                fragment := reference.bases[first:<last]
                gc := count_character(fragment, "G") + count_character(fragment, "C")
                sum := total(window)
                maximum_depth := window.sorted()[count - 1]
                zero := int(#[for value := window[] if value == 0 then value])
                mapq0 := total(mapq0_starts[r][first:<last])
                lines = lines.append("{reference.name} {first + 1} {last} gc " + sam_writer.ratio(gc, count) + " sum {sum} mean " + sam_writer.ratio(sum, count) + " max {maximum_depth} zero {zero} mapq0 {mapq0} |" + "#" * nat(10 * sum / count))
        return lines
    int[] distinct_values(int[] values)
        int[] result := []
        for value := values.sorted()[]
            if #result == 0 or value != result[-1]
                result = result.append(value)
        return result
    string[] summary_lines()
        read_count := int(#results)
        mapped := [for i := (0:<int(read_count))[] if results[i].mapped then i]
        unique := [for i := mapped[] if results[i].x0 == 1 then i]
        multi := [for i := mapped[] if results[i].x0 >= 2 then i]
        short := int(#[for result := results[] if not result.mapped and result.reason == "SHORT" then result])
        trimmed_reads := int(#[for result := results[] if result.trimmed > 0 then result])
        trimmed_bases := total([for result := results[] then result.trimmed])
        lines := ["reads total {read_count} trimmed {trimmed_reads} trimmed_bases {trimmed_bases} short {short}",
            "mapped {#mapped} rate " + sam_writer.ratio(int(#mapped), read_count) + " unique {#unique} multi {#multi} unmapped {read_count - int(#mapped)}",
            "unmapped short {short} nohit {read_count - int(#mapped) - short}"]
        method_names := ["EXACT", "MISMATCH", "SEED"]
        method_counts := [for method := method_names[] then int(#[for i := mapped[] if results[i].method == method then i])]
        forward_count := int(#[for i := mapped[] if results[i].alignments[0].strand == 0 then i])
        lines = lines.concat(["method exact {method_counts[0]} mismatch {method_counts[1]} seed {method_counts[2]}", "strand forward {forward_count} reverse {int(#mapped) - forward_count}"])
        for: r < int(#references)
            indices := [for i := mapped[] if results[i].alignments[0].reference == r then i]
            unique_count := int(#[for i := indices[] if results[i].x0 == 1 then i])
            plus_count := int(#[for i := indices[] if results[i].alignments[0].strand == 0 then i])
            lines = lines.append("reference {references[r].name} mapped {#indices} unique {unique_count} forward {plus_count} reverse {int(#indices) - plus_count}")
        lengths := [for result := results[] then result.length]
        ordered_lengths := lengths.sorted()
        first_bin := ordered_lengths[0] / 10 * 10
        last_bin := ordered_lengths[read_count - 1] / 10 * 10
        lines = lines.append("length_bins")
        for: first_bin <= lower <= last_bin step 10
            indices := [for i := (0:<int(read_count))[] if lower <= lengths[i] and lengths[i] <= lower + 9 then i]
            mapped_count := int(#[for i := indices[] if results[i].mapped then i])
            unique_count := int(#[for i := indices[] if results[i].mapped and results[i].x0 == 1 then i])
            lines = lines.append("{lower}-{lower + 9} reads {#indices} mapped {mapped_count} unique {unique_count}")
        n_reads := [for i := (0:<int(read_count))[] if includes(results[i].read.bases, "N") then i]
        n_bases := total([for i := n_reads[] then count_character(results[i].read.bases, "N")])
        mapped_n := int(#[for i := n_reads[] if results[i].mapped then i])
        lines = lines.concat(["n_reads {#n_reads} n_bases {n_bases} mapped {mapped_n}", "multi_targets"])
        map<string, int> target_counts
        for i := multi[]
            name_set := [for r := results[i].targets[] then references[r].name].join("+")
            old := 0
            match target_counts.get(name_set)
                int found
                    old = found
                none
                    old = 0
            .target_counts.set(name_set, old + 1)
        if target_counts.size() == 0
            lines = lines.append("none")
        else
            for key := target_counts.keys().sorted()[]
                match target_counts.get(key)
                    int count
                        lines = lines.append("{key} {count}")
                    none
                        continue
        lines = lines.append("mapq_histogram")
        int[] mapqs := distinct_values([for i := mapped[] then results[i].mapq]).sorted(descending = true)
        for mapq := mapqs[]
            count := int(#[for i := mapped[] if results[i].mapq == mapq then i])
            lines = lines.append("{mapq} {count}")
        lines = lines.append("nm_histogram")
        int[] nm_values := distinct_values([for i := mapped[] then results[i].alignments[0].nm])
        for nm := nm_values[]
            count := int(#[for i := mapped[] if results[i].alignments[0].nm == nm then i])
            lines = lines.append("{nm} {count}")
        max_length := [for result := results[] then int(#result.read.bases)].sorted()[read_count - 1]
        aligned := 0
        mismatches := 0
        substitutions := [for base := (0:<4)[] then [for other := (0:<5)[] then 0]]
        cycle_aligned := [for cycle := (0:<int(max_length))[] then 0]
        cycle_mismatches := cycle_aligned
        quality_counts := [0, 0]
        quality_sums := [0, 0]
        quality_aligned := [0, 0, 0, 0, 0]
        quality_mismatches := quality_aligned
        insertion_events := 0
        inserted_bases := 0
        deletion_events := 0
        deleted_bases := 0
        for index := unique[]
            result := results[index]
            primary := result.alignments[0]
            reference := references[primary.reference].bases
            for pair := primary.pairs[]
                i := pair[0]
                j := pair[1]
                cycle := if primary.strand == 0 then i + 1 else result.length - i
                quality := result.qualities[cycle - 1]
                reference_base := suffix_array.symbol_index(reference[j]) - 2
                oriented_base := suffix_array.symbol_index(primary.oriented[i]) - 2
                if oriented_base < 0
                    oriented_base = 4
                mismatch := primary.oriented[i] != reference[j]
                error_index := if mismatch then 1 else 0
                quality_bin := minimum(quality / 10, 4)
                aligned += 1
                substitutions[reference_base][oriented_base] += 1
                cycle_aligned[cycle - 1] += 1
                quality_counts[error_index] += 1
                quality_sums[error_index] += quality
                quality_aligned[quality_bin] += 1
                if mismatch
                    mismatches += 1
                    cycle_mismatches[cycle - 1] += 1
                    quality_mismatches[quality_bin] += 1
            for run := operation_runs(primary.operations)[]
                if run[0] == 1
                    insertion_events += 1
                    inserted_bases += run[1]
                elif run[0] == 2
                    deletion_events += 1
                    deleted_bases += run[1]
        lines = lines.concat(["error_profile aligned_bases {aligned} mismatches {mismatches} rate " + sam_writer.ratio(mismatches, aligned), "substitutions", "ref A C G T N"])
        for: base < 4
            lines = lines.append(suffix_array.bases[base] + " " + [for count := substitutions[base][] then string(count)].join(" "))
        lines = lines.append("mismatch_by_cycle")
        for: cycle < max_length
            lines = lines.append("cycle {cycle + 1} aligned {cycle_aligned[cycle]} mismatches {cycle_mismatches[cycle]} rate " + sam_writer.ratio(cycle_mismatches[cycle], cycle_aligned[cycle]))
        lines = lines.append("quality matched {quality_counts[0]} mean " + sam_writer.ratio(quality_sums[0], quality_counts[0]) + " mismatched {quality_counts[1]} mean " + sam_writer.ratio(quality_sums[1], quality_counts[1]))
        lines = lines.append("quality_bins")
        bins := ["0-9", "10-19", "20-29", "30-39", "40-"]
        for: bin_index < 5
            lines = lines.append("{bins[bin_index]} aligned {quality_aligned[bin_index]} mismatches {quality_mismatches[bin_index]} rate " + sam_writer.ratio(quality_mismatches[bin_index], quality_aligned[bin_index]))
        cigar := [0, 0, 0, 0]
        for index := mapped[]
            result := results[index]
            for run := operation_runs(result.alignments[0].operations)[]
                cigar[run[0]] += run[1]
            cigar[3] += result.trimmed
        int[] searches := mapper.search_counts
        int[] seeds := mapper.seed_counts
        return lines.concat(["cigar M {cigar[0]} I {cigar[1]} D {cigar[2]} S {cigar[3]}",
            "indels insertions {insertion_events} {inserted_bases} deletions {deletion_events} {deleted_bases}",
            "search dfs_calls {searches[0]} dfs_nodes {searches[1]} dfs_leaves {searches[2]} dfs_positions {searches[3]}",
            "seeds reads {seeds[0]} seeds {seeds[1]} with_n {seeds[2]} repetitive {seeds[3]} without_hit {seeds[4]} candidates {seeds[5]} passing {seeds[6]}"])

string mapped(string name, int flag, string reference, int position, int mapq, string cigar, string bases, string quality, int nm, string md, int score, int x0, int x1, string method, string alternatives)
    fields := [name, string(flag), reference, string(position + 1), string(mapq), cigar, "*", "0", "0", bases, quality, "NM:i:{nm}", "MD:Z:" + md, "AS:i:{score}", "X0:i:{x0}", "X1:i:{x1}", "YM:Z:" + method]
    if alternatives != ""
        fields = fields.append("XA:Z:" + alternatives)
    return fields.join(HT)
