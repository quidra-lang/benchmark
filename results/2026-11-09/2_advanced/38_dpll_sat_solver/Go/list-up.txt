type NumberedClause struct {
	Number   int
	Literals []int
}
type Preparation struct {
	Clauses     [][]int
	Fixed       map[int]bool
	Active      []int
	Counts      map[string]int
	Records     map[string][]int
	EmptyClause int
	Pure        bool
}

	}
	for _, c := range working {
		p.Clauses = append(p.Clauses, c.Literals)
