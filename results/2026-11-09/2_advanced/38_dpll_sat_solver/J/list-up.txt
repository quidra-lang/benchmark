preprocess =: 3 : 0
    'variables original use_pure' =. y
    working =. 0$a:
    fixed =. variables$_1
    counts =. 8$0
    records =. 4$<i.0
    empty_clause =. _1
    while. 1 do.
        unit =. 0
        for_box. working do.
            'number clause' =. >box
            if. 0=#clause do.
                empty_clause =. number
                break.
            end.
        end.
        if. empty_clause>:0 do.
            break.
        end.
        for_box. working do.
            clause =. 1{::>box
            if. 1=#clause do.
                unit =. {.clause
                break.
            end.
        end.
        if. unit=0 do.
            break.
        end.
        fixed =. (unit>0)((|unit)-1)}fixed
        counts =. (1+2{counts)2}counts
        surviving =. 0$a:
        for_box. working do.
            'number clause' =. >box
            if. unit e.clause do.
                counts =. (1+3{counts)3}counts
                records =. (<(1{::records),number)1}records
            else.
                reduced =. clause#~clause~:-unit
                surviving =. surviving,<(<number),<reduced
            end.
        end.
        working =. surviving
    end.
    if. empty_clause<0 do.
        removed =. (#working)$0
        sets =. ((#working),(1+2*variables))$0
        for_index. i.#working do.
            entry =. index{::working
            if. index{removed do.
                counts =. (1+4{counts)4}counts
                records =. (<(2{::records),0{::entry)2}records
        if. use_pure do.
            while. 1 do.
            end.
        end.
    end.
    counts =. (+/0 3 4 6{counts)7}counts
    records =. /:~&.>records
    clauses =. 1&{::&.>working
    active =. 1+I.fixed<0
    (<clauses),(<fixed),(<active),(<counts),(<records),(<empty_clause),<use_pure
)
preprocessing_report =: 3 : 0
    'clauses fixed active counts records empty pure' =. y
    if. empty>:0 do.
        lines =. lines,<'simplified empty_clause=',text empty
