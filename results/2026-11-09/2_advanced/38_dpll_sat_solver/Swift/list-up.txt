    let number: Int
    let literals: [Int]
}

    var unitFixed = 0

    var subsumed = 0
    var pureFixed = 0
    var pureRemoved = 0
    var tautologyRecords: [Int] = []
    var unitRecords: [Int] = []
    var subsumedRecords: [Int] = []
    var pureRecords: [Int] = []
    var emptyClause: Int?
    let pure: Bool
    var removed: Int { tautologies + unitRemoved + subsumed + pureRemoved }
}
