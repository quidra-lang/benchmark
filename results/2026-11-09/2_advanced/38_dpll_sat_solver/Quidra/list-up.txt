int absolute(int value)
    return if value < 0 then -value else value

bool contains(int[] values, int target)
    for value := values[]
        if value == target
            return true
    return false

class Clause(int number, int[] literals)
class Preparation(int[][] clauses, int[] fixed, int[] active, int[] counts, int[][] records, int empty_clause, bool pure)

Preparation run(int variables, int[][] original, bool use_pure)
    Clause[] working := []
    fixed := array(nat(variables), fill = -1)
    counts := array(8, fill = 0)
    int[][] records := array(4, fill = [])
    empty_clause := -1
    applying := true
    while applying
        unit := 0
        for clause := working[]
            if #clause.literals == 0
                empty_clause = clause.number
                break
        if empty_clause >= 0
            break
        for clause := working[]
            if #clause.literals == 1
                unit = clause.literals[0]
                break
        if unit == 0
            break
        fixed[absolute(unit) - 1] = if unit > 0 then 1 else 0
        counts[2] += 1
        Clause[] surviving := []
        for clause := working[]
            if contains(clause.literals, unit)
                counts[3] += 1
                records[1] = records[1].append(clause.number)
            else
                surviving = surviving.append(Clause(clause.number, [for literal := clause.literals[] if literal != -unit then literal]))
        working = surviving
    if empty_clause < 0
        removed := array(#working, fill = false)
        sets := array(#working, fill = array(nat(2 * variables + 1), fill = false))
        for: index < int(#working)
            if removed[index]
                counts[4] += 1
                records[2] = records[2].append(working[index].number)
        if use_pure
            eliminating := true
            while eliminating
    counts[7] = counts[0] + counts[3] + counts[4] + counts[6]
    for: index < 4
        records[index] = records[index].sorted()
    clauses := [for clause := working[] then clause.literals]
    int[] active := []
    for: index < variables
        if fixed[index] < 0
            active = active.append(index + 1)
