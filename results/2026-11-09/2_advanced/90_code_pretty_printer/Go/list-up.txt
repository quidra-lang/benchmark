func trailingPart(t *Token) *Doc {
	if len(t.Trailing) == 0 {
		return text("")
	}
	parts := []string{}
	for _, c := range t.Trailing {
		parts = append(parts, c.Text)
	}
	return suffix(strings.Join(parts, " "))
}
func forceBroken(d *Doc) bool {
	if d.Kind == "hardline" || d.Kind == "break_parent" {
		return true
	}
	for _, child := range d.Children {
		if forceBroken(child) {
			return true
		}
	}
	return false
}
func group(docs ...*Doc) *Doc {
	child := cat(docs...)
	return &Doc{Kind: "group", Children: []*Doc{child}, Forced: forceBroken(child)}
}
func fits(remaining int, first Command, original []Command) bool {
	if remaining < 0 {
		return false
	}
	stack := []Command{first}
	unused := len(original) - 1
	for len(stack) > 0 || unused >= 0 {
		if len(stack) == 0 {
			stack = append(stack, original[unused])
			unused--
		}
		c := stack[len(stack)-1]
		stack = stack[:len(stack)-1]
		d := c.Doc
		switch d.Kind {
		case "text":
			remaining -= len(d.Text)
		case "if_break":
			value := d.Text
			if c.Flat {
				value = d.Other
			}
			remaining -= len(value)
		case "line":
			if !c.Flat {
				return true
			}
			remaining--
		case "softline":
			if !c.Flat {
				return true
			}
		case "hardline":
			return true
		case "concat":
			stack = pushChildren(stack, c)
		case "group":
			flat := c.Flat
			if d.Forced {
				flat = false
			}
			stack = append(stack, Command{c.Indent, flat, d.Children[0]})
		case "nest", "align":
			stack = append(stack, Command{c.Indent, c.Flat, d.Children[0]})
		}
		if remaining < 0 {
			return false
		}
	}
	return true
}
