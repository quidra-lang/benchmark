static Doc *trailing_part(Builder *builder, const Token *token) {
    if (!token->trailing.count)
        return doc_text(builder->pool, "");
    TextLines parts = {0};
    for (int i = 0; i < token->trailing.count; ++i)
        lines_add(&parts, token->trailing.data[i]->value);
    char *joined = join_texts(&parts, " ");
    Doc *out = doc_suffix(builder->pool, joined);
    free(joined);
    lines_free(&parts);
    return out;
}
static bool force_broken(const Doc *doc) {
    if (doc->kind == D_HARDLINE || doc->kind == D_BREAK_PARENT)
        return true;
    for (int i = 0; i < doc->children.count; ++i)
        if (force_broken(doc->children.data[i]))
            return true;
    return false;
}
Doc *doc_group(DocPool *pool, Doc *child) {
    Doc *out = doc_marker(pool, D_GROUP);
    docs_push(&out->children, child);
    out->forced = force_broken(child);
    return out;
static bool fits(int remaining, Command first, const Commands *original) {
    if (remaining < 0)
        return false;
    Commands stack = {0};
    command_push(&stack, first);
    int unused = original->count - 1;
    bool result = true;
    while (stack.count || unused >= 0) {
        if (!stack.count)
            command_push(&stack, original->data[unused--]);
        Command command = stack.data[--stack.count];
        Doc *doc = command.doc;
        bool done = false;
        switch (doc->kind) {
        case D_TEXT:
            remaining -= (int)strlen(doc->value);
            break;
        case D_IF_BREAK:
            remaining -= (int)strlen(command.flat ? doc->other : doc->value);
            break;
        case D_LINE:
            if (!command.flat)
                done = true;
            else
                --remaining;
            break;
        case D_SOFTLINE:
            if (!command.flat)
                done = true;
            break;
        case D_HARDLINE:
            done = true;
            break;
        case D_CONCAT:
            push_children(&stack, command);
            break;
        case D_GROUP:
            command_push(&stack, (Command){command.indent, doc->forced ? false : command.flat,
                                           doc->children.data[0]});
            break;
        case D_NEST:
        case D_ALIGN:
            command_push(&stack, (Command){command.indent, command.flat, doc->children.data[0]});
            break;
        default:
            break;
        }
        if (done)
            break;
        if (remaining < 0) {
            result = false;
            break;
        }
    }
    free(stack.data);
    return result;
}
static void finalize(TextLines *lines, const LineBuffer *current, TextLines *pending, int width,
                     int comment_column) {
    char *line = trim_right(copy_text(current->data ? current->data : ""));
    if (pending->count) {
        char *comments = join_texts(pending, " ");
        size_t length = strlen(line);
        char *spaces = comment_column > 0 && (int)length < comment_column - 1 &&
                               comment_column - 1 + (int)strlen(comments) <= width
                           ? repeated(' ', comment_column - 1 - (int)length)
                           : copy_text(" ");
        char *with_comments = format_text("%s%s%s", line, spaces, comments);
        free(line);
        free(spaces);
        free(comments);
        line = with_comments;
    }
    lines_free(pending);
    lines_push(lines, line);
}
