static DocPtr trailingPart(Token *token) {
    if (token->trailing.empty())
        return stringDoc("");
    Lines parts;
    for (Comment *comment : token->trailing)
        parts.push_back(comment->value);
    return suffix(join(parts));
}
static bool forceBroken(const DocPtr &doc) {
    if (doc->kind == DocKind::Hardline || doc->kind == DocKind::BreakParent)
        return true;
    for (const DocPtr &child : doc->children)
        if (forceBroken(child))
            return true;
    return false;
}
DocPtr group(const Docs &docs) {
    DocPtr out = marker(DocKind::Group);
    out->children = {cat(docs)};
    out->forced = forceBroken(out->children[0]);
    return out;
static bool fits(int remaining, Command first, const std::vector<Command> &original) {
    if (remaining < 0)
        return false;
    std::vector<Command> stack{std::move(first)};
    int unused = static_cast<int>(original.size()) - 1;
    while (!stack.empty() || unused >= 0) {
        if (stack.empty())
            stack.push_back(original[unused--]);
        Command command = std::move(stack.back());
        stack.pop_back();
        const DocPtr &doc = command.doc;
        switch (doc->kind) {
        case DocKind::Text:
            remaining -= static_cast<int>(doc->value.size());
            break;
        case DocKind::IfBreak:
            remaining -= static_cast<int>((command.flat ? doc->other : doc->value).size());
            break;
        case DocKind::Line:
            if (!command.flat)
                return true;
            --remaining;
            break;
        case DocKind::Softline:
            if (!command.flat)
                return true;
            break;
        case DocKind::Hardline:
            return true;
        case DocKind::Concat:
            pushChildren(stack, command);
            break;
        case DocKind::Group:
            stack.push_back({command.indent, doc->forced ? false : command.flat, doc->children[0]});
            break;
        case DocKind::Nest:
        case DocKind::Align:
            stack.push_back({command.indent, command.flat, doc->children[0]});
            break;
        default:
            break;
        }
        if (remaining < 0)
            return false;
    }
    return true;
}
