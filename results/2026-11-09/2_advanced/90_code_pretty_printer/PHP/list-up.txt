        return concat(
            $this->comments($leading ?? $t->leading, $t),
            $t->text,
            $t->trailing ? suffix(implode(' ', array_map(fn($c) => $c->text, $t->trailing))) : null,
        );
function forceBroken(Doc $d): bool
{
    if (in_array($d->kind, ['hardline', 'break_parent'], true))
        return true;
    if ($d->kind === 'concat')
        return array_any($d->items, forceBroken(...));
    return $d->child !== null && forceBroken($d->child);
}

function group(mixed $child): Doc
{
    $d = new Doc('group', child: concat($child));
    $d->forced = forceBroken($d->child);
    return $d;
}
function fits(int $remaining, array $candidate, array $rest): bool
{
    if ($remaining < 0)
        return false;
    $check = [$candidate];
    $next = count($rest) - 1;
    while (true) {
        if (!$check) {
            if ($next < 0)
                return true;
            $check[] = $rest[$next--];
        }
        [$indent, $mode, $d] = array_pop($check);
        switch ($d->kind) {
            case 'text':
                $remaining -= strlen($d->value);
                break;
            case 'if_break':
                $remaining -= strlen($mode === 'break' ? $d->value : $d->flat);
                break;
            case 'line':
                if ($mode === 'break')
                    return true;
                --$remaining;
                break;
            case 'softline':
                if ($mode === 'break')
                    return true;
                break;
            case 'hardline':
                return true;
            case 'concat':
                foreach (array_reverse($d->items) as $child)
                    $check[] = [$indent, $mode, $child];
                break;
            case 'group':
                $check[] = [$indent, $d->forced ? 'break' : $mode, $d->child];
                break;
            case 'nest':
            case 'align':
                $check[] = [$indent, $mode, $d->child];
                break;
        }
        if ($remaining < 0)
            return false;
    }
}

function layout(Doc $document, array $profile): array
{
    $result = [];
    $current = '';
    $pending = [];
    $stack = [[0, 'break', $document]];
    $finalize = function () use (&$result, &$current, &$pending, $profile): void {
        $out = rtrim($current, ' ');
        if ($pending) {
            $s = implode(' ', $pending);
            $q = $profile['comment_column'];
            if ($q > 0 && strlen($out) < ($q - 1) && ($q - 1 + strlen($s)) <= $profile['width'])
                $out .= str_repeat(' ', $q - 1 - strlen($out)) . $s;
            else
                $out .= ' ' . $s;
            $pending = [];
        }
        $result[] = $out;
    };
            case 'hardline':
                $finalize();
                $current = str_repeat(' ', $i);
                break;
            case 'suffix':
                $pending[] = $d->value;
                break;
            case 'group':
                $d->mode = $mode === 'flat'
                    ? 'flat'
                    : (
                        $d->forced
                            ? 'forced'
                            : (
                                fits($profile['width'] - strlen($current), [$i, 'flat', $d->child], $stack)
                                    ? 'flat'
                                    : 'break'
                            )
                    );
                $stack[] = [$i, $d->mode === 'flat' ? 'flat' : 'break', $d->child];

    $finalize();
