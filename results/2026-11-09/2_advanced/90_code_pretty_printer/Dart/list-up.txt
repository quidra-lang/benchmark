
  Doc trailingPart(Token t) => t.trailing.isEmpty
      ? text('')
      : suffix(t.trailing.map((c) => c.text).join(' '));
bool forceBroken(Doc d) =>
    d.kind == 'hardline' ||
    d.kind == 'break_parent' ||
    d.children.any(forceBroken);
Doc group(List<Doc> docs) {
  final child = cat(docs);
  return Doc('group', children: [child], forced: forceBroken(child));
}
bool fits(int remaining, Command first, List<Command> original) {
  if (remaining < 0) return false;
  final stack = [first];
  var unused = original.length - 1;
  while (stack.isNotEmpty || unused >= 0) {
    if (stack.isEmpty) stack.add(original[unused--]);
    final c = stack.removeLast(), d = c.doc;
    switch (d.kind) {
      case 'text':
        remaining -= d.text.length;
        break;
      case 'if_break':
        remaining -= (c.flat ? d.other : d.text).length;
        break;
      case 'line':
        if (!c.flat) return true;
        remaining--;
        break;
      case 'softline':
        if (!c.flat) return true;
        break;
      case 'hardline':
        return true;
      case 'concat':
        pushChildren(stack, c);
        break;
      case 'group':
        stack.add(
          Command(c.indent, d.forced ? false : c.flat, d.children.first),
        );
        break;
      case 'nest':
      case 'align':
        stack.add(Command(c.indent, c.flat, d.children.first));
        break;
    }
    if (remaining < 0) return false;
  }
  return true;
}
