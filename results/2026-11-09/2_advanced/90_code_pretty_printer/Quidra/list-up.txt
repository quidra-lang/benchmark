int trailing_document(Construction &context, int token)
    string[] comments := []
    for comment := context.tokens[token].trailing[]
        comments = comments.append(comment.text)
    if #comments == 0
        return text_document(&context, "")
    return doc.suffix(&context.documents, comments.join(" "))
bool forced(const Store store, int index)
    document := store.items[index]
    if document.kind == "hardline" or document.kind == "break_parent"
        return true
    for child := document.children[]
        if forced(store, child)
            return true
    return false
class Output(string[] lines := [], current := "", string[] pending := [])
bool fits(const doc.Store store, int width, const Command first, const Command[] stack)
    if width < 0
        return false
    remaining := width
    check := stack.append(first)
    while #check != 0
        command := check[-1]
        check = check[0:<int(#check) - 1]
        doc.Document document := store.items[command.document]
        if document.kind == "text"
            remaining -= int(#document.value)
        elif document.kind == "if_break"
            remaining -= int(#(if command.flat then document.flat_value else document.value))
        elif document.kind == "line"
            if not command.flat
                return true
            remaining -= 1
        elif document.kind == "softline"
            if not command.flat
                return true
        elif document.kind == "hardline"
            return true
        elif document.kind == "concat"
            backwards := document.children[::-1]
            for child := backwards[]
                check = check.append(Command(command.indent, command.flat, child))
        elif document.kind == "group"
            check = check.append(Command(command.indent, if doc.forced(store, command.document) then false else command.flat, document.children[0]))
        elif document.kind == "nest" or document.kind == "align"
            check = check.append(Command(command.indent, command.flat, document.children[0]))
        if remaining < 0
            return false
    return true

void finalize(Output &output, int width, int column)
    line := lexer.trim_end(output.current)
    if #output.pending > 0
        comment := output.pending.join(" ")
        if column > 0 and int(#line) < column - 1 and column - 1 + int(#comment) <= width
            line += lexer.repeat(" ", column - 1 - int(#line))
        else
            line += " "
        line += comment
        output.pending = []
    output.lines = output.lines.append(line)

                finalize(&output, width, column)
                output.current = lexer.repeat(" ", command.indent)
                current_column = command.indent

            output.pending = output.pending.append(document.value)
        elif document.kind == "group"
            mode := "flat"
            if not command.flat
                if doc.forced(store, command.document)
                    mode = "forced"
                elif not fits(store, width - current_column, Command(command.indent, true, document.children[0]), stack)
                    mode = "break"
