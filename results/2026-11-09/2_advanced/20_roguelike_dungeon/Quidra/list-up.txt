Command[] expand_commands(string[][] rows)
    Command[] result := []
    for tokens := rows[]
        name := tokens[0]
        argument := ""
        count := 1
        if name == "wait"
            if #tokens == 2
                count = int(tokens[1])
        else
            if #tokens > 1
                argument = tokens[1]
            if #tokens == 3
                count = int(tokens[2])
        for: index < count
            result = result.append(Command(name, argument))
    return result

        result = outcome

    void .run()
        .update_view()
        for command := commands[]
            if result != ""
                unused += 1
            elif not .execute(command)
                failed += 1
            else
                succeeded += 1
                if result == ""
                    .advance_turn()
        if result == ""
            .finish("alive", "commands_exhausted")

        .update_view()
        .monsters_act()
        if result != ""
            return
        .end_turn()
        if result != ""
            return
        for checkpoint := config.checkpoints[]
            if turn == checkpoint
                .write_checkpoint()
                break
        turn += 1
    bool .execute(Command command)
        if command.name == "move"
            offset := fov.direction(command.argument)
            return .move_player(offset[0], offset[1])
        if command.name == "wait"
            return true
        if command.name == "pickup"
            return .pickup()
