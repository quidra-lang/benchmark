        if symbol.definition.kind == "undefined"
            if contains(context.wraps, symbol.name)
                symbol.name = "__wrap_" + symbol.name
            elif symbol.name.starts_with("__real_") and contains(context.wraps, symbol.name[7:<int(#symbol.name)])
                symbol.name = symbol.name[7:<int(#symbol.name)]
        object_file.symbols[index] = symbol
    object_index := int(#context.loaded)
    for: index < int(#object_file.symbols)
        objects.Symbol symbol := object_file.symbols[index]
        if symbol.binding != "local"
            entries := entries_for(context, symbol.name)
            entries = entries.append(objects.Reference(object_index, index))
            .context.entries.set(symbol.name, entries)
