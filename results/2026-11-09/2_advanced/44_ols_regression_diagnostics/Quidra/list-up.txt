    n := fitted.n
    rank := fitted.rank
    leverage_limit := params[3] * inference.as_real(rank) / inference.as_real(n)
    cook_limit := params[5] / inference.as_real(n)
    dffits_limit := params[6] * inference.root(inference.as_real(rank) / inference.as_real(n))
        h := fitted.leverage[i]
        flags := ""
        if h > leverage_limit
            flags += "H"
        if inference.absolute(studentized) > params[4]
            flags += "O"
        if cooks > cook_limit
            flags += "C"
        if inference.absolute(dffits) > dffits_limit
            flags += "F"
        rows = rows.append(Row(z, studentized, cooks, dffits, if flags == "" then "-" else flags, dfbetas))
    return rows
