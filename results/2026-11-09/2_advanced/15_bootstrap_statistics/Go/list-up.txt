func integer(text string) int64 {
	value, err := strconv.ParseInt(text, 10, 64)
	if err != nil {
		panic(err)
	}
	return value
}
func hundredths(text string) int64 {
	parts := strings.SplitN(text, ".", 2)
	fraction := "00"
	if len(parts) > 1 {
		fraction = (parts[1] + "00")[:2]
	}
	return integer(parts[0])*100 + integer(fraction)
