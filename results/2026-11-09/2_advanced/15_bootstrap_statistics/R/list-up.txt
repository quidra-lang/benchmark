hundredths <- function(text) {
  parts <- strsplit(text, ".", fixed = TRUE)[[1L]]
  fraction <- if (length(parts) == 1L) "00" else paste0(parts[2L], strrep("0", 2L - nchar(parts[2L])))
  as.numeric(parts[1L]) * 100 + as.numeric(fraction)
