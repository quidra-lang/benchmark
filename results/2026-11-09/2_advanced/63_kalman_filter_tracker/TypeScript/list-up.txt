  for (const scan of scans) {
    const k = scan.number;
    const live = tracks.filter((t) => t.status !== 'deleted');
    const innovations = live.map((t) =>
      scan.points.map((point) => kalman.innovation(t.estimate, point, p.sigma)),
    );
    const costs: Costs = innovations.map((row) => row.map((v) => (v.d2 <= p.gate ? v.d2 : null)));
    const assigned = optimal(costs);
    const byTrack = new Map(assigned);
    const used = new Set(assigned.map((pair) => pair[1]));
    for (let i = 0; i < live.length; i++) {
      const track = live[i];
      const j = byTrack.get(i);
      if (j !== undefined) {
        track.hits++;
        track.consecutive = 0;
        track.end = k;
      } else {
        track.consecutive++;
      }
    for (const track of live) {
      if (track.status === 'confirmed') {
        if (track.consecutive >= p.delete) {
          track.status = 'deleted';
          track.deletion = k;
        }
      } else if (track.hits >= p.confirm[0]) {
        track.status = 'confirmed';
        track.confirmation = k;
      } else if (k - track.birth + 1 >= p.confirm[1]) {
        track.status = 'deleted';
        track.deletion = k;
      }
    }
    for (let j = 0; j < scan.points.length; j++) {
      if (used.has(j)) continue;
      const measurement = scan.points[j];
      const track: Track = {
        number: tracks.length + 1,
        birth: k,
        end: k,
        confirmation: null,
        deletion: null,
        status: 'tentative',
        hits: 1,
        consecutive: 0,
        estimate: kalman.initialize(p.model, measurement, p),
        records: [{ measurement, nis: null }],
      };
      tracks.push(track);
