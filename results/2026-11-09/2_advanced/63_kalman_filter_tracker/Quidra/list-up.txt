        for: scan_index < scans
            live := [for i := (0:<int(#tracks))[] if tracks[i].status != 2 then i]
            measurements := data.measurements[scan_index]
            rows := int(#live)
            columns := int(#measurements)
            for: i < rows
                for: j < columns
                    value := tracks[live[i]].primary_filter.innovation(measurements[j])
                    costs[i][j] = value.distance2
                    if value.distance2 <= p.gate
                        usable[i][j] = true
            assigned := association.optimal_assignment(costs, usable)
            matches := [for i := (0:<int(rows))[] then -1]
            used_columns = [for j := (0:<int(columns))[] then false]
            for pair := assigned[]
                matches[pair[0]] = pair[1]
                used_columns[pair[1]] = true
            for: i < rows
                index := live[i]
                j := matches[i]
                if j >= 0
                    tracks[index].hits += 1
                    tracks[index].consecutive_misses = 0
                else
                    tracks[index].consecutive_misses += 1
                if tracks[index].status == 1
                    if tracks[index].consecutive_misses >= p.delete_misses
                        tracks[index].status = 2
                        tracks[index].deletion = scan_index + 1
                elif tracks[index].hits >= p.confirm_hits
                    tracks[index].status = 1
                    tracks[index].confirmation = scan_index + 1
                elif scan_index - tracks[index].birth + 1 >= p.confirm_window
                    tracks[index].status = 2
                    tracks[index].deletion = scan_index + 1
            for: j < columns
                if not used_columns[j]
                    index := int(#tracks)
                    tracks = tracks.append(Track(index + 1, scan_index, measurements[j], make_filter(p.dimension, measurements[j]), scans))
