    assigned <- optimal_assignment(length(live),length(measurements),usable)

      track <- live[[row]]
      column <- matches[row]
      if (column != 0L) {

        track$consecutive_misses <- 0L

          track$deleted <- scan

        track$status <- "confirmed"
        track$confirmed <- scan
      } else if (scan-track$birth+1L >= window) {
        track$status <- "deleted"
        track$deleted <- scan
      }
