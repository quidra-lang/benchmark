function characterizeEvents(
  signals: Signal[],
  data: Data,
  estimates: Map<Family, Estimation>,
  baseline: Baseline,
): Event[] {
  const ranges: { start: number; end: number }[] = [];
  for (const number of [...new Set(signals.map((s) => s.subgroup))].sort((a, b) => a - b)) {
    const previous = ranges.at(-1);
    if (previous && number - previous.end - 1 <= data.spec.event_gap) previous.end = number;
    else ranges.push({ start: number, end: number });
  }
  return ranges.map(({ start, end }, index) => {
    const found = signals.filter((s) => start <= s.subgroup && s.subgroup <= end);
    let total = 0;
    let deviations = 0;
    let outOfSpec = 0;
    for (let i = start; i <= end; i++) {
      const group = data.groups[i - 1];
      total += group.total;
      deviations += group.sd;
      for (const sample of group.samples) {
        if (sample.value < data.spec.lsl || sample.value > data.spec.usl) outOfSpec++;
      }
    }
    const length = end - start + 1;
    return {
      number: index + 1,
      start,
      end,
      length,
      signals: found,
      outOfSpec,
      shift: (total / (data.spec.subgroup_size * length) - baseline.center) / baseline.sigma,
      spread: deviations / length / estimates.get('XBAR_S')!.model.limits[1].cl,
    };
  });
