int[] consecutive(int first, int last)
    int[] result := []
    for: first <= number <= last
        result = result.append(number)
    return result
bool contains_number(int[] numbers, int value)
    for number := numbers[]
        if number == value
            return true
    return false
class Event(int number, int start, int end)
    rules.Signal[] signals := []
    int length()
        return end - start + 1
    bool contains(int subgroup)
        return start <= subgroup <= end
    int size := settings.size
    int phase1 := settings.phase1
    variable.Subgroup[] subgroups := variable.subgroup_statistics(values, size)
    int signal_subgroup(rules.Signal signal)
        if signal.chart == "I" or signal.chart == "MR"
            return (signal.end - 1) / size + 1
        return signal.end
    int event_number(int subgroup)
        for event := events[]
            if event.contains(subgroup)
                return event.number
        return 0
    void characterize(Event &event)
        for signal := signals[]
            int subgroup := signal_subgroup(signal)
            if not event.contains(subgroup)
                continue
            event.signals = event.signals.append(signal)
        total := 0
        real64 sd_sum := 0.0
        for: event.start <= number <= event.end
            group := subgroups[number - 1]
            total += group.total
            sd_sum += group.sd
            for value := group.values[]
                if value < settings.lsl or value > settings.usl
                    event.out_of_spec += 1
        mean := real64(total) / real64(size * event.length())
        event.shift = (mean - mu0) / sigma
        event.spread = (sd_sum / real64(event.length())) / sbar
    void .build_events()
        int[] groups := []
        for signal := signals[]
            int subgroup := signal_subgroup(signal)
            if not contains_number(groups, subgroup)
                groups = groups.append(subgroup)
        for subgroup := groups.sorted()[]
            last := int(#events) - 1
            if last >= 0 and subgroup - events[last].end - 1 <= settings.event_gap
                events[last].end = subgroup
            else
                events = events.append(Event(int(#events) + 1, subgroup, subgroup))
        for: index < #events
            characterize(&events[index])
    void .propose_limits()

                if family == 2
                    points = points.concat(consecutive((subgroup - 1) * size + 1, subgroup * size))
                else
                    points = points.append(subgroup)
            proposals = proposals.append(Proposal(family, int(#points), family_limits(family, points)))
