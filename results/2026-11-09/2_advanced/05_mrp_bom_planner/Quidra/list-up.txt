Item[] read_items(string[] &errors)
    Item[] accepted := []
    for row := read_rows("items.csv")[]
        fields := row.fields
        id := fields[0]
        rule := fields[2]
        parameter := fields[3]
        failure := ""
        int[] values := []
        if index_of(accepted, id) >= 0
            failure = "duplicate item {id}"
        elif rule != "LFL" and rule != "FOQ" and rule != "POQ" and rule != "MIN"
            failure = "unknown lot rule {rule}"
        else
            for part := parameter.split("/")[]
                values = values.append(int(part))
            valid := false
            if rule == "MIN"
                valid = #values == 2 and values[0] >= 1 and values[1] >= 1
            elif rule == "LFL"
                valid = #values == 1 and values[0] == 0
            else
                valid = #values == 1 and values[0] >= 1
            if not valid
                failure = "bad lot parameter {parameter} for {rule}"
        if failure != ""
            errors = errors.append("items.csv line {row.line}: {failure}")
        else
            accepted = accepted.append(Item(id, int(fields[1]), rule, values[0], if #values == 2 then values[1] else 0, int(fields[4]), int(fields[5]), int(fields[6]), int(fields[7]), fields[8]))
    Item[] result := []
    for id := ids(accepted)[]
        result = result.append(accepted[index_of(accepted, id)])
    return result

int need(BomRow row, int quantity)
    return (quantity * row.qty_per * (100 + row.scrap) + 99) / 100
class Bom(items.Item[] inventory)
    BomRow[] rows := []
    BomRow[] children(string id)
        string[] names := []
        for row := rows[]
            if row.parent == id
                names = names.append(row.child)
        BomRow[] result := []
        for name := names.sorted()[]
            for row := rows[]
                if row.parent == id and row.child == name
                    result = result.append(row)
                    break
        return result
    BomRow[] parents(string id)
        string[] names := []
        for row := rows[]
            if row.child == id
                names = names.append(row.parent)
        BomRow[] result := []
        for name := names.sorted()[]
            for row := rows[]
                if row.child == id and row.parent == name
                    result = result.append(row)
                    break
        return result
    bool manufactured(string id)
        return #children(id) > 0
    bool contains(string parent, string child)
        for row := rows[]
            if row.parent == parent and row.child == child
                return true
        return false
    int low_level(string id, int[] &codes)
        index := items.index_of(inventory, id)
        if codes[index] >= 0
            return codes[index]
        result := 0
        for row := parents(id)[]
            candidate := low_level(row.parent, &codes) + 1
            if candidate > result
                result = candidate
        codes[index] = result
        return result
    int[] planning_order(int[] codes)
        int[] result := []
        for id := items.ids(inventory)[]
            result = result.append(items.index_of(inventory, id))
        for: 1 <= offset < int(#result)
            current := result[offset]
            cursor := offset
            while cursor > 0 and codes[result[cursor - 1]] > codes[current]
                result[cursor] = result[cursor - 1]
                cursor -= 1
            result[cursor] = current
        return result
Bom read_bom(items.Item[] inventory, string[] &errors)
    graph := Bom(inventory)
    for entry := items.read_rows("bom.csv")[]
        fields := entry.fields
        parent := fields[0]
        child := fields[1]
        quantity := int(fields[2])
        scrap := int(fields[3])
        failure := ""
        if items.index_of(inventory, parent) < 0
            failure = "unknown item {parent}"
        elif items.index_of(inventory, child) < 0
            failure = "unknown item {child}"
        elif quantity < 1
            failure = "bad quantity {quantity}"
        elif scrap > 50
            failure = "bad scrap {scrap}"
        elif parent == child
            failure = "cycle {parent} -> {parent}"
        elif graph.contains(parent, child)
            failure = "duplicate {parent} -> {child}"
        if failure != ""
            errors = errors.append("bom.csv line {entry.line}: {failure}")
        else
            graph.rows = graph.rows.append(BomRow(parent, child, quantity, scrap))
    return graph
class WorkCenter(string id, int minutes, int efficiency, string description)

class Operation(string item, int number, string center, int setup, int run)

int center_index(WorkCenter[] centers, string id)
    for: index < #centers
        if centers[index].id == id

WorkCenter[] read_workcenters(string[] &errors)
    WorkCenter[] accepted := []
    for row := items.read_rows("workcenters.csv")[]
        fields := row.fields
        efficiency := int(fields[2])
        failure := ""
        if center_index(accepted, fields[0]) >= 0
            failure = "duplicate center {fields[0]}"
        elif efficiency < 50 or efficiency > 150
            failure = "bad efficiency {efficiency}"
        if failure != ""
            errors = errors.append("workcenters.csv line {row.line}: {failure}")
        else
            accepted = accepted.append(WorkCenter(fields[0], int(fields[1]), efficiency, fields[3]))

    for center := accepted[]
        names = names.append(center.id)
    WorkCenter[] result := []
    for id := names.sorted()[]
        result = result.append(accepted[center_index(accepted, id)])
    return result
Operation[] routing(Operation[] operations, string item)
    Operation[] result := []
    for operation := operations[]
        if operation.item == item
            result = result.append(operation)
    for: 1 <= offset < int(#result)
        current := result[offset]
        cursor := offset
        while cursor > 0 and result[cursor - 1].number > current.number
            result[cursor] = result[cursor - 1]
            cursor -= 1
        result[cursor] = current
    return result
Operation[] read_routings(items.Item[] inventory, WorkCenter[] centers, bom.Bom graph, string[] &errors)
    Operation[] result := []
    for row := items.read_rows("routings.csv")[]
        fields := row.fields
        id := fields[0]
        number := int(fields[1])
        center := fields[2]
        run := int(fields[4])
        failure := ""
        if items.index_of(inventory, id) < 0
            failure = "unknown item {id}"
        elif center_index(centers, center) < 0
            failure = "unknown center {center}"
        elif not graph.manufactured(id)
            failure = "purchased item {id}"
        elif run < 1
            failure = "bad run time {run}"
        else
            for operation := routing(result, id)[]
                if operation.number == number
                    failure = "duplicate {id} operation {number}"
                    break
        if failure != ""
            errors = errors.append("routings.csv line {row.line}: {failure}")
        else
            result = result.append(Operation(id, number, center, int(fields[3]), run))
    return result
int operation_week(int receipt, int lead_time, int index, int count)
    result := receipt - lead_time + index * lead_time / count
    return if result < 1 then 1 else result
int lot_size(items.Item item, int week, int projected, int[] receipts, int[] gross, int horizon)
    net := item.safety_stock - projected
    if item.rule == "LFL"
        return net
int sum_values(int[] values)

ScheduleRow[] read_schedule(string name, items.Item[] inventory, int horizon, bool unique, string[] &errors)
    ScheduleRow[] result := []
    for row := items.read_rows(name)[]
        string id := row.fields[0]
        index := items.index_of(inventory, id)
        week := int(row.fields[1])
        quantity := int(row.fields[2])
        failure := ""
        if index < 0
            failure = "unknown item {id}"
        elif week < 1 or week > horizon
            failure = "week {week} outside horizon"
        elif quantity < 1
            failure = "bad quantity {quantity}"
        elif unique
            for previous := result[]
                if previous.item == index and previous.week == week
                    failure = "duplicate {id} week {week}"
                    break
        if failure != ""
            errors = errors.append("{name} line {row.line}: {failure}")
        else
            result = result.append(ScheduleRow(index, week, quantity, row.line))
    return result

class PlanningData(Params params, items.Item[] inventory, bom.Bom graph, capacity.WorkCenter[] centers, capacity.Operation[] operations, ScheduleRow[] mps, ScheduleRow[] receipts, Inquiry[] inquiries)
    codes := array(#inventory, fill = -1)
    cumulative := array(#inventory, fill = -1)
    for: index < #inventory
        codes[index] = graph.low_level(inventory[index].id, &codes)
        cumulative[index] = graph.cumulative(inventory[index].id, &cumulative)
    int[] planning_order := graph.planning_order(codes)
    int[] weeks := []
    for: 1 <= week <= params.horizon
        weeks = weeks.append(week)
    int[][] scheduled := []
    for: index < #inventory
        row := array(nat(params.horizon + 1), fill = 0)
        for receipt := receipts[]
            if receipt.item == int(index)
                row[receipt.week] += receipt.quantity
        scheduled = scheduled.append(row)
void plan_item(ItemPlan &plan, int horizon, int item_index)
    for demand := plan.demands[]
        plan.gross[demand.week] += demand.quantity
    projected := plan.item.on_hand
    plan.on_hand[0] = projected
    for: 1 <= week <= horizon
        int available := projected + plan.receipts[week] - plan.gross[week]
        projected = available
        if available < plan.item.safety_stock
            quantity := lot.lot_size(plan.item, week, available, plan.receipts, plan.gross, horizon)
            order := PlannedOrder(item_index, int(#plan.orders) + 1, week, quantity, plan.item.lead_time)
            plan.orders = plan.orders.append(order)
            plan.net[week] = plan.item.safety_stock - available
            plan.planned_receipts[week] = quantity
            plan.planned_releases[order.release] += quantity
            projected = available + quantity
        plan.on_hand[week] = projected
Plan run_mrp(PlanningData data, Demand[] order_demands)
    ItemPlan[] plans := []
    for: index < #data.inventory
        plans = plans.append(ItemPlan(data.inventory[index], data.params.horizon, data.scheduled[index]))
    for row := data.mps[]
        plans[row.item].demands = plans[row.item].demands.append(Demand(row.item, "MPS", row.week, row.quantity))
    for demand := order_demands[]
        plans[demand.item].demands = plans[demand.item].demands.append(demand)
    for index := data.planning_order[]
        plan_item(&plans[index], data.params.horizon, index)
        for order := plans[index].orders[]
            for row := data.graph.children(data.inventory[index].id)[]
                child := items.index_of(data.inventory, row.child)
                plans[child].demands = plans[child].demands.append(Demand(child, "DEPENDENT", order.release, bom.need(row, order.quantity), parent = index, parent_number = order.number))
    int[][] loads := []
    for center := data.centers[]
        loads = loads.append(array(nat(data.params.horizon + 1), fill = 0))
    for index := data.planning_order[]
        capacity.Operation[] routing := capacity.routing(data.operations, data.inventory[index].id)
        for order := plans[index].orders[]
            for: operation < #routing
                week := capacity.operation_week(order.receipt, data.inventory[index].lead_time, int(operation), int(#routing))
                loads[capacity.center_index(data.centers, routing[operation].center)][week] += routing[operation].minutes(order.quantity)
    int[][] overloads := []
    for: center < #data.centers
        int[] excess := []
        for load := loads[center][]
            excess = excess.append(if load > data.centers[center].available then load - data.centers[center].available else 0)
        overloads = overloads.append(excess)
    return Plan(plans, loads, overloads)
class PromiseResult(Promise[] promises, Plan final_plan)
PromiseResult promise_inquiries(PlanningData data, Plan base)
    current := base
    Demand[] promised := []
    Promise[] promises := []
    for inquiry := data.inquiries[]
        promise := Promise(inquiry)
        for: inquiry.week <= week <= data.params.horizon
            demand := Demand(inquiry.item, "ORDER", week, inquiry.quantity, inquiry.id, inquiry.line)
            trial := run_mrp(data, promised.append(demand))
            reason := ""
            for index := data.planning_order[]
                if trial.entries[index].past_due_quantity() > current.entries[index].past_due_quantity()
                    reason = data.inventory[index].id
                    break
            if reason == ""
                if promise.material_week == 0
                    promise.material_week = week
                for: center < #data.centers
                    for period := data.weeks[]
                        if trial.overloads[center][period] > current.overloads[center][period]
                            reason = "{data.centers[center].id} week {period}"
                            break
                    if reason != ""
                        break
            if reason == ""
                promise.week = week
                promised = promised.append(demand)
                current = trial
                break
            if week == inquiry.week
                promise.reason = reason
        promises = promises.append(promise)
    return PromiseResult(promises, current)
string[] grid_lines(PlanningData data, Plan plan)
    string[] result := []
    for index := data.planning_order[]
        entry := plan.entries[index]
        item := entry.item
        result = result.append("ITEM {item.id} LLC {data.codes[index]} LT {item.lead_time} LOT {item.lot_label()} SS {item.safety_stock} OH {item.on_hand} {item.description}")
        result = result.concat([report.table_row("WEEK", data.weeks, 6), report.table_row("GROSS", entry.gross[1:<#entry.gross], 6), report.table_row("RECEIPTS", entry.receipts[1:<#entry.receipts], 6), report.table_row("ON_HAND", entry.on_hand[1:<#entry.on_hand], 6), report.table_row("NET", entry.net[1:<#entry.net], 6), report.table_row("PLAN_RCPT", entry.planned_receipts[1:<#entry.planned_receipts], 6), report.table_row("PLAN_REL", entry.planned_releases[1:<#entry.planned_releases], 6)])
        result = result.append("TOTAL gross {sum_values(entry.gross)} receipts {sum_values(entry.receipts)} planned {sum_values(entry.planned_receipts)} orders {#entry.orders} ending {entry.on_hand[data.params.horizon]}")
    return result
string[] capacity_lines(PlanningData data, Plan plan)
    string[] result := []
    int[] totals := []
    for: index < #data.centers
        center := data.centers[index]
        loads := plan.loads[index]
        excess := plan.overloads[index]
        leveled := capacity.level_loads(loads, center.available)
        int[] utilization := []
        overloaded := 0
        peak := 0
        largest := 0
        for week := data.weeks[]
            utilization = utilization.append(report.percent(loads[week], center.available))
            if excess[week] > 0
                overloaded += 1
            if loads[week] > largest
                largest = loads[week]
                peak = week
        total_load := sum_values(loads)
        totals = totals.append(total_load)
        backlog := if leveled[1] > center.available then leveled[1] - center.available else 0
        result = result.concat(["CENTER {center.id} minutes {center.minutes} efficiency {center.efficiency} available {center.available} {center.description}", report.table_row("WEEK", data.weeks, 7), report.table_row("LOAD", loads[1:<#loads], 7), report.table_row("UTIL", utilization, 7), report.table_row("OVER", excess[1:<#excess], 7), report.table_row("LEVELED", leveled[1:<#leveled], 7), "TOTAL load {total_load} available {center.available * data.params.horizon} utilization {report.percent(total_load, center.available * data.params.horizon)} overloaded {overloaded} peak {peak} backlog {backlog}"])
    return result
string[] supply_lines(PlanningData data, Plan final_plan, string order_id)
    string[] result := []
    for index := data.planning_order[]
        entry := final_plan.entries[index]
        for supply_index := entry.order_supplies[]
            supply := entry.supplies[supply_index]
            serves := false
            for root := supply.roots[]
                if root.kind == "ORDER" and root.order_id == order_id
                    serves = true
                    break
            if serves
                order := entry.orders[supply.number - 1]
                result = result.append("  SUPPLY {entry.item.id} order {order.number} receipt {order.receipt} release {order.release} quantity {order.quantity}")
    return result
string[] promise_lines(PlanningData data, Promise[] promises, Plan base, Plan final_plan)
    string[] result := []
    on_time := 0
    late := 0
    rejected := 0
    for promise := promises[]
        inquiry := promise.inquiry
        header := "ORDER {inquiry.id} {data.inventory[inquiry.item].id} quantity {inquiry.quantity} requested {inquiry.week}"
        material := if promise.material_week == 0 then "none" else string(promise.material_week)
        if promise.week == 0
            result = result.append(header + " rejected material {material} reason {promise.reason}")
            rejected += 1
            continue
        if promise.week == inquiry.week
            result = result.append(header + " promised {promise.week}")
            on_time += 1
        else
            result = result.append(header + " promised {promise.week} late {promise.week - inquiry.week} material {material} reason {promise.reason}")
            late += 1
        result = result.concat(supply_lines(data, final_plan, inquiry.id))
    for index := data.planning_order[]
        before := base.entries[index]
        after := final_plan.entries[index]
        first := 0
        for week := data.weeks[]
            if before.planned_receipts[week] != after.planned_receipts[week]
                first = week
                break
        if first > 0
            result = result.append("CHANGE {data.inventory[index].id} from {first} planned {sum_values(before.planned_receipts)} -> {sum_values(after.planned_receipts)} orders {#before.orders} -> {#after.orders}")
    return result.append("SUMMARY orders {#promises} on_time {on_time} late {late} rejected {rejected}")

string table_row(string label, int[] values, int width)
    result := label
    while #result < 10
        result += " "
    for value := values[]
        cell := string(value)
        while int(#cell) < width
            cell = " " + cell
        result += cell
    return result
