class Item {
  final String id, rule, description;
  final int lead, safety, onHand, material, labor;
  final List<int> lot;
  Item(List<String> fields, this.lot)
      : id = fields[0], lead = int.parse(fields[1]), rule = fields[2], safety = int.parse(fields[4]),
        onHand = int.parse(fields[5]), material = int.parse(fields[6]), labor = int.parse(fields[7]), description = fields[8];
  String get lotLabel => rule == 'LFL' ? rule : '$rule ${lot.join(rule == 'MIN' ? '/' : '')}';
}
List<int>? lotValues(String rule, String text) {
  final values = text.split('/').map(int.parse).toList();
  final valid = rule == 'MIN' ? values.length == 2 && values.every((v) => v >= 1) : values.length == 1 && (rule == 'LFL' ? values[0] == 0 : values[0] >= 1);
  return valid ? values : null;
}
Map<String, Item> readItems(List<String> errors) {
  final items = <String, Item>{};
  for (final row in readRows('items.csv')) {
    final f = row.fields, id = f[0], rule = f[2], lot = f[3];
    final values = lotValues(rule, lot);
    final reason = items.containsKey(id) ? 'duplicate item $id' : !['LFL', 'FOQ', 'POQ', 'MIN'].contains(rule) ? 'unknown lot rule $rule' : values == null ? 'bad lot parameter $lot for $rule' : null;
    if (reason != null) { errors.add('items.csv line ${row.line}: $reason'); } else { items[id] = Item(f, values!); }
  }
  return items;
}
class BomRow {
  final String parent, child;
  final int quantity, scrap;
  BomRow(List<String> fields) : parent = fields[0], child = fields[1], quantity = int.parse(fields[2]), scrap = int.parse(fields[3]);
}
int need(BomRow row, int quantity) => (quantity * row.quantity * (100 + row.scrap) + 99) ~/ 100;
class Bom {
  final Map<String, List<BomRow>> children, parents;
  Bom(Map<String, Item> items) : children = {for (final id in items.keys) id: []}, parents = {for (final id in items.keys) id: []};
  bool manufactured(String id) => children[id]!.isNotEmpty;
  int code(String id, Map<String, int> codes) {
    if (codes.containsKey(id)) return codes[id]!;
    var value = 0;
    for (final row in parents[id]!) { value = max(value, code(row.parent, codes) + 1); }
    return codes[id] = value;
  }
Bom readBom(Map<String, Item> items, List<String> errors) {
  final bom = Bom(items);
  for (final source in readRows('bom.csv')) {
    final row = BomRow(source.fields), parent = row.parent, child = row.child;
    String? reason;
    if (!items.containsKey(parent)) { reason = 'unknown item $parent'; }
    else if (!items.containsKey(child)) { reason = 'unknown item $child'; }
    else if (row.quantity < 1) { reason = 'bad quantity ${row.quantity}'; }
    else if (row.scrap > 50) { reason = 'bad scrap ${row.scrap}'; }
    else if (parent == child) { reason = 'cycle $parent -> $parent'; }
    else if (bom.children[parent]!.any((other) => other.child == child)) { reason = 'duplicate $parent -> $child'; }
    else {
      final path = bom.path(child, parent, {});
      if (path != null) reason = 'cycle $parent -> ${path.join(' -> ')}';
    }
    if (reason != null) { errors.add('bom.csv line ${source.line}: $reason'); continue; }
    bom.children[parent]!.add(row); bom.parents[child]!.add(row);
    bom.children[parent]!.sort((a, b) => a.child.compareTo(b.child));
    bom.parents[child]!.sort((a, b) => a.parent.compareTo(b.parent));
  }
  return bom;
}
class WorkCenter {
  final String id, description;
  final int minutes, efficiency, available;
  WorkCenter(List<String> f) : id = f[0], minutes = int.parse(f[1]), efficiency = int.parse(f[2]), description = f[3], available = int.parse(f[1]) * int.parse(f[2]) ~/ 100;
}
class Operation {
  final String item, center;
  final int number, setup, run;
  Operation(List<String> f) : item = f[0], number = int.parse(f[1]), center = f[2], setup = int.parse(f[3]), run = int.parse(f[4]);
  int minutes(int quantity) => setup + quantity * run;
}
Map<String, WorkCenter> readCenters(List<String> errors) {
  final centers = <String, WorkCenter>{};
  for (final row in readRows('workcenters.csv')) {
    final id = row.fields[0], efficiency = int.parse(row.fields[2]);
    final reason = centers.containsKey(id) ? 'duplicate center $id' : efficiency < 50 || efficiency > 150 ? 'bad efficiency $efficiency' : null;
    if (reason != null) { errors.add('workcenters.csv line ${row.line}: $reason'); } else { centers[id] = WorkCenter(row.fields); }
  }
  return centers;
}
Map<String, List<Operation>> readRoutings(Map<String, Item> items, Map<String, WorkCenter> centers, Bom bom, List<String> errors) {
  final routings = <String, List<Operation>>{};
  for (final row in readRows('routings.csv')) {
    final op = Operation(row.fields), id = op.item, center = op.center;
    String? reason;
    if (!items.containsKey(id)) { reason = 'unknown item $id'; }
    else if (!centers.containsKey(center)) { reason = 'unknown center $center'; }
    else if (!bom.manufactured(id)) { reason = 'purchased item $id'; }
    else if (op.run < 1) { reason = 'bad run time ${op.run}'; }
    else if ((routings[id] ?? []).any((other) => other.number == op.number)) { reason = 'duplicate $id operation ${op.number}'; }
    if (reason != null) { errors.add('routings.csv line ${row.line}: $reason'); } else { (routings[id] ??= []).add(op); }
  }
  for (final operations in routings.values) { operations.sort((a, b) => a.number.compareTo(b.number)); }
  return routings;
}

String? scheduleError(String item, int week, int quantity, Map<String, Item> items, int horizon) {
  if (!items.containsKey(item)) return 'unknown item $item';
  if (week < 1 || week > horizon) return 'week $week outside horizon';
  if (quantity < 1) return 'bad quantity $quantity';
  return null;
}

Plan runMrp(PlanningData data, List<Demand> orders) {
  final plans = {for (final id in data.order) id: ItemPlan(data.items[id]!, data.horizon, data.scheduled[id]!)};
  for (final demand in [...data.mps, ...orders]) { plans[demand.item]!.demands.add(demand); }
  for (final id in data.order) {
    final plan = plans[id]!, item = plan.item;
    for (final demand in plan.demands) { plan.gross[demand.week] += demand.quantity; }
    var projected = item.onHand;
    for (final week in data.weeks) {
      projected += plan.receipts[week] - plan.gross[week];
      if (projected < item.safety) {
        final quantity = lotSize(item, week, projected, plan.receipts, plan.gross, data.horizon);
        final order = PlannedOrder(id, plan.orders.length + 1, week, quantity, item.lead); plan.orders.add(order);
        plan.net[week] = item.safety - projected; plan.plannedReceipts[week] = quantity; plan.plannedReleases[order.release] += quantity; projected += quantity;
      }
      plan.onHand[week] = projected;
    }
    for (final order in plan.orders) {
      for (final row in data.bom.children[id]!) { plans[row.child]!.demands.add(Demand(row.child, 'DEPENDENT', order.release, need(row, order.quantity), parent: id, parentNumber: order.number)); }
    }
  }
  final loads = {for (final id in data.centers.keys) id: List.filled(data.horizon + 1, 0)};
  for (final entry in data.routings.entries) {
    for (final order in plans[entry.key]!.orders) {
      for (final (index, op) in entry.value.indexed) { loads[op.center]![operationWeek(order.receipt, data.items[entry.key]!.lead, index, entry.value.length)] += op.minutes(order.quantity); }
    }
  }
  return Plan(plans, loads, data.centers);
}
({List<Promise> promises, Plan finalPlan}) promiseInquiries(PlanningData data, Plan base) {
  var current = base;
  final promised = <Demand>[], promises = <Promise>[];
  for (final inquiry in data.inquiries) {
    final promise = Promise(inquiry);
    for (var week = inquiry.week; week <= data.horizon; week++) {
      final demand = Demand(inquiry.item, 'ORDER', week, inquiry.quantity, orderId: inquiry.id, line: inquiry.line);
      final trial = runMrp(data, [...promised, demand]); String? reason;
      for (final id in data.order) { if (trial.items[id]!.pastDue > current.items[id]!.pastDue) { reason = id; break; } }
      if (reason == null) {
        if (promise.materialWeek == 0) promise.materialWeek = week;
        capacity: for (final id in data.centerOrder) {
          for (final w in data.weeks) { if (trial.overloads[id]![w] > current.overloads[id]![w]) { reason = '$id week $w'; break capacity; } }
        }
      }
      if (reason == null) { promise.week = week; promised.add(demand); current = trial; break; }
      if (week == inquiry.week) promise.reason = reason;
    }
    promises.add(promise);
  }
  return (promises: promises, finalPlan: current);
}
