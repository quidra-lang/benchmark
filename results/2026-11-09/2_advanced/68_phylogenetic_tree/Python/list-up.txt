def consensus_tree(size, taxa, outgroup, clades):
    tree = tree_building.Tree(size, taxa)
    root = tree.internal()
    nodes = {mask: tree.internal() for mask in sorted(clades)}
    for mask, node in nodes.items():
        containers = [other for other in nodes if other != mask and other & mask == mask]
        parent = nodes[min(containers, key=int.bit_count)] if containers else root
        tree.connect(parent, node)
    for taxon in taxa:
        containers = [mask for mask in nodes if mask & (1 << taxon)]
        parent = nodes[min(containers, key=int.bit_count)] if containers else root
        tree.connect(parent, taxon)
    return tree

    def internal(self):
        node = self.next_node
        self.next_node += 1
        self.adjacent[node] = {}
        return node
    def connect(self, first, second, length=0.0):
        self.adjacent[first][second] = length
        self.adjacent[second][first] = length
    def __init__(self, tree, outgroup):
        self.tree = tree
        self.outgroup = outgroup
        self.root = next(iter(tree.adjacent[outgroup]))
        self.parents = {}
        self.children = {}
        self.masks = {}
        self.order = []

        def visit(node, parent):
            self.parents[node] = parent
            self.order.append(node)
            children = [neighbor for neighbor in tree.adjacent[node] if neighbor != parent]
            self.children[node] = children
            mask = 1 << node if node < tree.size else 0
            for child in children:
                visit(child, node)
                mask |= self.masks[child]
            self.masks[node] = mask

        visit(self.root, None)
        self.ingroup = self.masks[self.root] ^ (1 << outgroup)
    def splits(self):
        return {self.masks[node] for node in self.tree.internal_nodes() if node != self.root}
