void populate_consensus(tree_building.Tree &tree, int parent, int mask, int[] clades)
    size := tree.size
    int[] maximal := []
    for clade := clades[]
        if clade == mask or not subset(clade, mask, size)
            continue
        contained := false
        for other := clades[]
            if other != clade and other != mask and subset(clade, other, size) and subset(other, mask, size)
                contained = true
        if not contained
            maximal = maximal.append(clade)
    covered := 0
    for clade := maximal[]
        node := .tree.internal()
        .tree.connect(parent, node)
        populate_consensus(&tree, node, clade, clades)
        covered += clade
    for: taxon < size
        if tree_io.contains_taxon(mask, taxon) and not tree_io.contains_taxon(covered, taxon)
            .tree.connect(parent, taxon)
tree_building.Tree consensus_tree(int size, int[] taxa, int outgroup, int[] clades)
    tree := tree_building.empty_tree(size)
    root := .tree.internal()
    .tree.connect(root, outgroup)
    ingroup := sum_integers([for taxon := taxa[] if taxon != outgroup then 2 ^ taxon])
    populate_consensus(&tree, root, ingroup, clades)
    return tree
class Generator(int state)
    int .draw()
        state = (48271 * state) % 2147483647
        return state
Bootstrap bootstrap_trees(string[] sequences, int[] taxa, int outgroup, Params params, int[] main_splits)
    size := int(#taxa)
    length := int(#sequences[0])
    generator := Generator(params.seed)
    supports := [for mask := (0:<int(2 ^ size))[] then 0]
    for: 1 <= replicate <= params.replicates
        weights := [for c := (0:<int(length))[] then 0]
        for: draw < length
            value := .generator.draw()
            weights[value % length] += 1
        matrix := distances.build_matrix(sequences, weights, params.model, params.saturation).values
        tree := tree_building.neighbor_joining(matrix, taxa)
        splits := tree_building.split_set(tree, outgroup)
        for mask := splits[]
            supports[mask] += 1
    appeared := [for mask := (0:<int(#supports))[] if supports[mask] > 0 then mask]
    majority := [for mask := appeared[] if 2 * supports[mask] > params.replicates then mask]
    return Bootstrap(supports, lines, identical, consensus_tree(size, taxa, outgroup, majority), appeared)
Rooted root_tree(Tree tree, int outgroup)
    size := int(#tree.adjacent)
    root := tree.adjacent[outgroup][0]
    parents := [for i := (0:<int(size))[] then -2]
    int[][] children := [for i := (0:<int(size))[] then []]
    masks := [for i := (0:<int(size))[] then 0]
    int[] order := []
    visit_nodes(tree, root, -1, &parents, &children, &masks, &order)
    return Rooted(root, parents, children, masks, order, masks[root] - 2 ^ outgroup)
int[] split_set(Tree tree, int outgroup)
    orientation := root_tree(tree, outgroup)
    return [for node := tree.internal_nodes()[] if node != orientation.root then orientation.masks[node]].sorted()
