 'leaves clades' =. y
 cap =. 2*N
 root =. N
 adj =. (cap,cap)$0
 len =. (cap,cap)$0.0
 all =. +/2^leaves-.Outgroup
 for_i. i.#clades do.
  mask =. i{clades
  candidates =. I.(clades~:mask)*.(mask&subset)"0 clades
  parent =. root
  if. #candidates do.
   sizes =. bitcount"0 candidates{clades
   parent =. N+1+(sizes i.<./sizes){candidates
  end.
  child =. N+1+i
  adj =. 1(<parent,child)}adj
  adj =. 1(<child,parent)}adj
 end.
 for_leaf. leaves do.
  parent =. root
  if. leaf~:Outgroup do.
   candidates =. I.(2^leaf)&subset"0 clades
   if. #candidates do.
    counts =. bitcount"0 candidates{clades
    parent =. N+1+(counts i.<./counts){candidates
   end.
  end.
  adj =. 1(<parent,leaf)}adj
  adj =. 1(<leaf,parent)}adj
 end.
 adj;len;leaves;0

 Supports =: (2^N)$0

  weights =. L$0
  for_draw. i.L do.
   rng =. 2147483647|48271x*rng
   col =. L|rng
   weights =. (1+col{weights)col}weights
  end.
  matrix =. 0{::MainModel distance_matrix weights
  tree =. matrix neighbor_joining i.N
  ss =. splits tree
  Supports =: (1+ss{Supports)ss}Supports
 end.
