int select_target(Tower tower, TowerType kind, enemies.Enemy[] population, int[] living, pathing.DistanceField field)
    best := -1
    best_key := 0
    for position := living[]
        enemy := population[position]
        if can_aim(kind.targets, enemy) and tower.in_range(enemy.position(), kind.levels[tower.level - 1].range)
            key := tower.target_key(enemy, field)
            if best < 0 or key < best_key
                best = position
                best_key = key
    return best
