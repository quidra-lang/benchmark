export function chooseTarget(
  tower: Tower,
  enemies: Iterable<Enemy>,
  map: GridMap,
  distances: number[],
) {
  let best: Enemy | null = null;
  let bestValue = 0;
  for (const enemy of enemies) {
    if (!enemy.alive || !canAim(tower.type.targets, enemy) || !withinRange(tower, enemy, map)) {
      continue;
    }
    if (!best || value < bestValue || (value === bestValue && enemy.number < best.number)) {
      best = enemy;
      bestValue = value;
    }
  }
  return best;
