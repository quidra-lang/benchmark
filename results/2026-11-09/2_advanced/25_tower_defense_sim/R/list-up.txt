select_target <- function(tower, enemies, field) {
  best <- NULL
  best_key <- 0
  for (enemy in enemies) {
    if (!can_aim(tower$kind$targets, enemy) || !in_range(tower, enemy_position(enemy))) next
    key <- target_key(tower, enemy, field)
    if (is.null(best) || key < best_key) {
      best <- enemy
      best_key <- key
    }
  }
  best
