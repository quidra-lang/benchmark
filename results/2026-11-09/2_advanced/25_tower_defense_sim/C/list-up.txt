    for (int i = 0; i < count; i++) {
        int index = living[i];
        const Enemy *enemy = &enemies[index];
        if (!enemy->alive || !can_aim(tower->kind->targets, enemy) || !in_range(tower, grid, enemy_position(enemy, grid))) continue;
