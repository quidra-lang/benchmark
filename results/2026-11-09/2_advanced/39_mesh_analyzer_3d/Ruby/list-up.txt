        rounded = point.map { |coordinate| spacing * (2 * coordinate + spacing).div(2 * spacing) }
        points << rounded
        corners = triangle.vertices.map { |vertex| representatives[vertex] }
        face = triangle.dup
        face.vertices = corners
      @renumber = present.each_with_index.to_h { |original, index| [original, index + 1] }
      @points = [nil] + present.map { |original| @snapped[original] }
      @faces.each { |face| face.vertices = face.vertices.map { |vertex| @renumber.fetch(vertex) } }
    def normal(face, points)
      a, b, c = face.vertices.map { |vertex| points[vertex] }
      cross(subtract(b, a), subtract(c, a))
    end
