        local a, b, c = representative[triangle.vertices[1]], representative[triangle.vertices[2]], representative[triangle.vertices[3]]
            faces[#faces + 1] = {triangle = triangle.triangle, polygon = triangle.polygon, line = triangle.line, group = triangle.group, vertices = {a, b, c}}
    for _, original in ipairs(keys(present)) do
        d.points[#d.points + 1] = d.snapped[original]
        d.renumber[original] = #d.points
    end
    local initial = {}
    for _, face in ipairs(kept) do
        local v = face.vertices
        local vertices = {d.renumber[v[1]], d.renumber[v[2]], d.renumber[v[3]]}
        d.faces[#d.faces + 1] = {triangle = face.triangle, polygon = face.polygon, line = face.line, group = face.group, vertices = vertices}
function M.normal(face, points)
    local v = face.vertices
    return M.cross(M.subtract(points[v[2]], points[v[1]]), M.subtract(points[v[3]], points[v[1]]))
end
