        const position = vertices[original].map(value => spacing * Math.floor((2 * value + spacing) / (2 * spacing)));
        const vertices = triangle.vertices.map(vertex => representatives[vertex]);
        originalToNew.set(original, vertices.length);
        vertices.push(snapped.positions[original]);
    }
    const faces = removal.faces.map((face, i) => {
        const sequence = face.vertices.map(vertex => originalToNew.get(vertex));
        return { ...face, number: i + 1, vertices: sequence, initial: sequence.slice() };
    });
function normal(face, vertices) {
    const [a, b, c] = face.vertices.map(vertex => vertices[vertex]);
    return cross(difference(b, a), difference(c, a));
}
