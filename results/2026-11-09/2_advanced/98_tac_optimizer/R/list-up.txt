calculate <- function(operation, first, second) {
    result <- switch(operation, "<" = first < second, "<=" = first <= second, ">" = first > second,
        ">=" = first >= second, "==" = first == second, "!=" = first != second)
    gmp::as.bigz(as.integer(result))
}
