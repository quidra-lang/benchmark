int operation(string operator_text, int first, int second)
    result := if operator_text == "<" then first < second elif operator_text == "<=" then first <= second elif operator_text == ">" then first > second elif operator_text == ">=" then first >= second elif operator_text == "==" then first == second else first != second
    return if result then 1 else 0
