int prepend(Engine &engine, int continuation, Entry[] entries)
    result := continuation
    position := int(#entries) - 1
    while position >= 0
        index := int(#engine.goal_nodes)
        engine.goal_nodes = engine.goal_nodes.append(GoalNode(entries[position], result))
        result = index
        position -= 1
    return result
bool try_clause(Engine &engine, terms.Store &store, Profiles &profiles, const database.Database db, int predicate, int candidate, int goal, int barrier, int context, int continuation)
    database.Predicate definition := db.predicates[predicate]
    profile := profile_index(&profiles, definition.name, definition.arity)
    profiles.rows[profile].tries += 1
    clause := database.copy_clause(&store, definition.clauses[candidate])
    if not unify.unify(&store, &engine.trail, clause.head, goal)
        return false
    engine.goals = if clause.body < 0 then continuation else prepend(&engine, continuation, [Entry(clause.body, barrier, context)])
    return true

bool user_call(Engine &engine, terms.Store &store, Profiles &profiles, const database.Database db, int goal, int context)
    predicate := database.goal_predicate(db, store, goal)
    if predicate < 0
        procedure := terms.atom(&store, "procedure")
        indicator := terms.term_indicator(&store, goal)
        missing := terms.compound(&store, "existence_error", [procedure, indicator])
        fault(&engine, &store, missing)
        return false
    candidates := database.candidates(db, store, predicate, goal)
    if #candidates == 0
        return false
    height := int(#engine.stack)
    if #candidates > 1
        index := create_choice(&engine, "CLAUSES", engine.goals)
        engine.choices[index].goal = goal
        engine.choices[index].predicate = predicate
        engine.choices[index].candidates = candidates
        engine.choices[index].next_candidate = 1
        engine.choices[index].barrier = height
        engine.choices[index].context = context

bool execute_step(Engine &engine, terms.Store &store, Profiles &profiles, const database.Database db)
    first := engine.goal_nodes[engine.goals]
    engine.goals = first.next
    entry := first.entry
    engine.context = entry.context
    if #entry.marker > 0
        return execute_marker(&engine, &store, entry)
    goal := terms.dereference(store, entry.goal)
    terms.Node node := store.nodes[goal]
    arity := terms.arity(store, goal)
    engine.current_name = node.name
    engine.current_arity = arity
    if control(node.name, arity)
        return control_goal(&engine, &store, goal, entry.barrier, entry.context)
    return user_call(&engine, &store, &profiles, db, goal, entry.context)
