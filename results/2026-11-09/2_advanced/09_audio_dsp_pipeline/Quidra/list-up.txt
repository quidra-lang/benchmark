class Signal(string name, int rate, int[] samples, origin := "input", int[] ingredients := [])
    string .register_signal(int number, string command, Parsed arguments, int rate, int[] samples, string extras)
        name := arguments.tokens[0]
        version := int(#archive)
        archive = archive.append(signals.Signal(name, rate, samples, "line {number} {command}", arguments.sources))
        .signal_table.set(name, version)
    void trace_lines(int version, int depth, string[] &lines)
        signals.Signal source := archive[version]
        string indent := " " * nat(2 * depth)
        lines = lines.append("{indent}{source.name} <- {source.origin}")
        for ingredient := source.ingredients[]
            trace_lines(ingredient, depth + 1, &lines)
        if name == "TRACE"
            lines := [source.name]
            trace_lines(arguments.sources[0], 1, &lines)
            return lines
