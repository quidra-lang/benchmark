  Signal = Struct.new(:name, :rate, :samples, :origin, :ingredients, keyword_init: true) do
  def create(name, rate, samples, origin, ingredients = [])
    Signal.new(name: name, rate: rate, samples: samples.freeze,
               origin: origin, ingredients: ingredients.freeze).freeze
  end
    signal = Signals.create(name, rate, samples, "line #{line} #{command}", arguments.grep(Signals::Signal))
    @signals[name] = signal
  def trace(source)
    lines = []
    stack = [[source, 1]]
    until stack.empty?
      signal, depth = stack.pop
      lines << ' ' * (2 * depth) + "#{signal.name} <- #{signal.origin}"
      signal.ingredients.reverse_each { |ingredient| stack << [ingredient, depth + 1] }
    end
    lines
  end
    when 'TRACE' then [source.name, trace(source)]
