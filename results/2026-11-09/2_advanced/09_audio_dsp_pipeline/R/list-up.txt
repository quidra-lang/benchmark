trace_lines <- function(signal, depth) {
  lines <- paste0(strrep("  ", depth), signal$name, " <- ", signal$origin)
  for (ingredient in signal$ingredients) lines <- c(lines, trace_lines(ingredient, depth + 1L))
  lines
