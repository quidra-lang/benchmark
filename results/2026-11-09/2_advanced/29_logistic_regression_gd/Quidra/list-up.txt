class Selection(local CrossValidation[] results)
    best := results[0]
    for result := results[]
        if result.loss <= best.loss
            best = result
    local mean := total(best.fold_losses) / 5.0
    local spread := total([for value := best.fold_losses[] then (value - mean) * (value - mean)])
    se := square_root(spread / 4.0) / square_root(5.0)
    real64 one_se := best.lambda
    for result := results[]
        if result.loss <= best.loss + se
            one_se = result.lambda

string fixed(real64 value, int places)
    text := if places == 4 then "{value:.4}" else "{value:.6}"
    parsed := real64(text)
    if text.starts_with("-") and parsed == 0.0
        return text[1:]
    return text
