char *real(double value, int digits) {
    if (fabs(value) < 0.5 / pow(10.0, digits))
        value = 0;
    int length = snprintf(NULL, 0, "%.*f", digits, value);
    char *result = malloc((size_t)length + 1);
    snprintf(result, (size_t)length + 1, "%.*f", digits, value);
    return result;
}
