fn f4(v: f64) -> String {
    fixed(v, 4)
}
fn f6(v: f64) -> String {
    fixed(v, 6)
}
fn sum(xs: impl IntoIterator<Item = f64>) -> f64 {
    xs.into_iter().fold(0.0, |s, x|s+x)
}

            total += losses[i];
            if usize::from(p[i] >= d.threshold()) != d.y[i] {
                errors += 1
            }
        }
        folds.push(total/check.len()as f64);
    }

pub fn fixed(v: f64, p: usize) -> String {
    let s = format!("{v:.p$}");
    if s.starts_with('-') && s.parse::<f64>().unwrap() == 0.0 {
        s[1..].into()
    }else {
        s
    }
}
