    public static string Fixed(double value, int places) {
        string text = value.ToString("F" + places, CultureInfo.InvariantCulture);
        return text.StartsWith("-") && double.Parse(text, CultureInfo.InvariantCulture) == 0
                   ? text.Substring(1)
                   : text;
    }
