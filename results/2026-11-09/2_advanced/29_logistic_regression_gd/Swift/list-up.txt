func f4(_ v: Double) -> String {
    fixed(v, 4)
}
func f6(_ v: Double) -> String {
    fixed(v, 6)
}

func fixed(_ value: Double, _ places: Int) -> String {
    let s = String(format: "%.*f", places, value);
    return s.hasPrefix("-") && Double(s) == 0 ? String(s.dropFirst()) : s
}
