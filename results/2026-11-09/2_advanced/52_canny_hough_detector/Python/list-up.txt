def read_trig(filename="input/trig.txt"):
    lines = Path(filename).read_text(encoding="ascii").splitlines()
    values = [tuple(map(int, line.split(" "))) for line in lines]
    if len(values) != 180 or any(index != row[0] for index, row in enumerate(values)):
        raise ValueError("invalid trigonometry table")
    return [(row[1], row[2]) for row in values]

def clamped(image, x, y):
    height = len(image)
    width = len(image[0])
    x = min(max(x, 0), width - 1)
    y = min(max(y, 0), height - 1)
    return image[y][x]


def smoothing(image, params):
    height = len(image)
    width = len(image[0])
    size = params["gauss_size"]
    half = size // 2
    weights = params["gauss_kernel"]
    divisor = params["gauss_divisor"]
    output = []
    for y in range(height):
        row = []
        for x in range(width):
            total = 0
            for j in range(size):
                for i in range(size):
                    total += weights[j * size + i] * clamped(
                        image, x + i - half, y + j - half
                    )
            row.append(total // divisor)
        output.append(row)
    return output


def gradient(smoothed):
    height = len(smoothed)
    width = len(smoothed[0])
    gx = [[0] * width for _ in range(height)]
    gy = [[0] * width for _ in range(height)]
    magnitude = [[0] * width for _ in range(height)]
    for y in range(height):
        for x in range(width):
            q1 = clamped(smoothed, x - 1, y - 1)
            q2 = clamped(smoothed, x, y - 1)
            q3 = clamped(smoothed, x + 1, y - 1)
            q4 = clamped(smoothed, x - 1, y)
            q6 = clamped(smoothed, x + 1, y)
            q7 = clamped(smoothed, x - 1, y + 1)
            q8 = clamped(smoothed, x, y + 1)
            q9 = clamped(smoothed, x + 1, y + 1)
            sx = (q3 + 2 * q6 + q9) - (q1 + 2 * q4 + q7)
            sy = (q7 + 2 * q8 + q9) - (q1 + 2 * q2 + q3)
            gx[y][x] = sx
            gy[y][x] = sy
            magnitude[y][x] = abs(sx) + abs(sy)
    return gx, gy, magnitude

def directions(gx, gy):
    image = []
    for rowx, rowy in zip(gx, gy):
        row = []
        for dx, dy in zip(rowx, rowy):
            ax, ay = abs(dx), abs(dy)
            if 29 * ay <= 12 * ax:
                row.append(0)
            elif 29 * ay >= 70 * ax:
                row.append(90)
            else:
                row.append(45 if dx * dy < 0 else 135)
        image.append(row)
    return image


def suppress(magnitude, direction):
    height = len(magnitude)
    width = len(magnitude[0])
    image = [[0] * width for _ in range(height)]
    for y in range(1, height - 1):
        for x in range(1, width - 1):
            before, after = OFFSETS[direction[y][x]]
            value = magnitude[y][x]
            if (value > magnitude[y + before[1]][x + before[0]]
                    and value >= magnitude[y + after[1]][x + after[0]]):
                image[y][x] = value
    return image

def overlay_image(original, segments, circles, corners, divisor):
    image = [[value // divisor for value in row] for row in original]
    height = len(image)
    width = len(image[0])

    def mark(point):
        x, y = point
        if 0 <= x < width and 0 <= y < height:
            image[y][x] = 255

    for segment in segments:
        for point in bresenham(segment["start"], segment["end"]):
            mark(point)
    for circle in circles:
        cx, cy = circle["center"]
        for dx, dy in circle["outline"]:
            mark((cx + dx, cy + dy))
    for corner in corners:
        for offset in range(-2, 3):
            mark((corner["x"] + offset, corner["y"]))
            mark((corner["x"], corner["y"] + offset))
    return image
def report_lines(image, magnitude, direction, nms, strong, weak, edges,
                 components_list, candidates, circle_peaks_list, circles,
                 circle_edge, line_edge, max_votes, line_peaks_list, lines,
                 segments, used, junctions, polygons, corner_candidates,
                 corners, vertices_with_corner):
    height = len(image)
    width = len(image[0])
    kinds = Counter(polygon["kind"] for polygon in polygons)
    directions_count = Counter(direction[y][x] for x, y in edges)
    match_count = Counter(corner["match_kind"] for corner in corners)
    values = [
        ("image", f"{width} {height}"),
        ("gradient_nonzero", sum(value > 0 for row in magnitude for value in row)),
        ("gradient_max", max(value for row in magnitude for value in row)),
        ("nms_kept", sum(value > 0 for row in nms for value in row)),
        ("strong", len(strong)), ("weak", len(weak)),
        ("edges", len(edges)),
        ("weak_kept", len(weak & edges)),
        ("weak_dropped", len(weak - edges)),
        ("edge_components", len(components_list)),
    ]
    for angle in (0, 45, 90, 135):
        values.append((f"direction_{angle}", directions_count[angle]))
    values.extend([
        ("circle_candidates", len(candidates)),
        ("circle_peaks", len(circle_peaks_list)),
        ("circles", len(circles)),
        ("circle_edge_pixels", len(circle_edge)),
        ("line_edge_pixels", len(line_edge)),
        ("hough_max_votes", max_votes),
        ("hough_peaks", len(line_peaks_list)),
        ("lines", len(lines)),
        ("segments", len(segments)),
        ("claimed_pixels", len(used)),
        ("junctions", len(junctions)),
        ("polygons", len(polygons)),
    ])
    for kind in POLYGON_KINDS:
        values.append((kind, kinds[kind]))
    values.extend([
        ("corner_candidates", len(corner_candidates)),
        ("corners", len(corners)),
        ("corners_on_vertices", match_count["vertex"]),
        ("corners_on_segments", match_count["segment"]),
        ("corners_unmatched", match_count["none"]),
        ("vertices_with_corner", vertices_with_corner),
    ])
    return [f"{name} {value}" for name, value in values]

def rho(point, angle, trig, step):
    x, y = point
    cosine, sine = trig[angle]
    return (x * cosine + y * sine) // (16384 * step)
