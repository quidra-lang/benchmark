    fun read(): Data {
        val input=lines("image.pgm");
        val size=input[1].split(' ').map(String::toInt);
        val image=Array(size[1]) {
            y->input[y+3].split(' ').map(String::toInt).toIntArray()
        };
        val trig=lines("trig.txt").map {
            it.split(' ').drop(1).map(String::toInt).toIntArray()
        }.toTypedArray();
        val values=lines("params.txt").associate {
            val w=it.split(' ');
            w[0] to w.drop(1).map(String::toLong).toLongArray()
        };
        return Data(image,trig,Params(values))
    }

    data class Gradient(val gx: Array<IntArray>,val gy: Array<IntArray>,val magnitude: Array<IntArray>)
    fun clamped(image: Array<IntArray>,x: Int,y: Int)=image[y.coerceIn(0,image.lastIndex)][x.coerceIn(0,image[0].lastIndex)]
    fun smooth(image: Array<IntArray>,p: PgmIo.Params): Array<IntArray> {
        val height=image.size;
        val width=image[0].size;
        val k=p.get("gauss_size");
        val half=k/2;
        val kernel=p.row("gauss_kernel");
        val result=Array(height) {
            IntArray(width)
        };
        for(y in 0 until height)for(x in 0 until width) {
            var sum=0L;
            for(j in 0 until k)for(i in 0 until k)sum+=kernel[j*k+i]*clamped(image,x+i-half,y+j-half);
            result[y][x]=(sum/p.get("gauss_divisor")).toInt()
        };
        return result
    }
    fun gradient(image: Array<IntArray>): Gradient {
        val height=image.size;
        val width=image[0].size;
        val gx=Array(height) {
            IntArray(width)
        };
        val gy=Array(height) {
            IntArray(width)
        };
        val m=Array(height) {
            IntArray(width)
        };
        for(y in 0 until height)for(x in 0 until width) {
            val a=clamped(image,x-1,y-1);
            val b=clamped(image,x,y-1);
            val c=clamped(image,x+1,y-1);
            val d=clamped(image,x-1,y);
            val f=clamped(image,x+1,y);
            val g=clamped(image,x-1,y+1);
            val h=clamped(image,x,y+1);
            val i=clamped(image,x+1,y+1);
            gx[y][x]=c+2*f+i-(a+2*d+g);
            gy[y][x]=g+2*h+i-(a+2*b+c);
            m[y][x]=abs(gx[y][x])+abs(gy[y][x])
        };
        return Gradient(gx,gy,m)
    }

    fun inside(x: Int,y: Int,width: Int,height: Int)=x>=0&&x<width&&y>=0&&y<height
    fun neighbors(p: PgmIo.Point,width: Int,height: Int): List<PgmIo.Point> {
        val result=mutableListOf<PgmIo.Point>();
        for(dy in -1..1)for(dx in -1..1)if((dx!=0||dy!=0)&&inside(p.x+dx,p.y+dy,width,height))result+=PgmIo.Point(p.x+dx,p.y+dy);
        return result
    }
    fun process(g: Filters.Gradient,low: Int,high: Int): Result {
        val m=g.magnitude;
        val height=m.size;
        val width=m[0].size;
        val result=Result(height,width);
        val queue=ArrayDeque<PgmIo.Point>();
        for(y in 0 until height)for(x in 0 until width) {
            val gx=g.gx[y][x];
            val gy=g.gy[y][x];
            val ax=abs(gx);
            val ay=abs(gy);
            val direction=if(29*ay<=12*ax)0 else if(29*ay>=70*ax)90 else if(gx*gy<0)45 else 135;
            result.direction[y][x]=direction;
            if(x==0||y==0||x==width-1||y==height-1)continue;
            val dx=if(direction==90)0 else if(direction==45)1 else -1;
            val dy=if(direction==0)0 else -1;
            if(m[y][x]>m[y+dy][x+dx]&&m[y][x]>=m[y-dy][x-dx])result.nms[y][x]=m[y][x];
            if(result.nms[y][x]>=high) {
                result.strong++;
                result.edge[y][x]=true;
                queue.addLast(PgmIo.Point(x,y))
            } else if(result.nms[y][x]>=low)result.weak++
        };
        while(queue.isNotEmpty()) {
            val point=queue.removeFirst();
            for(next in neighbors(point,width,height))if(!result.edge[next.y][next.x]&&result.nms[next.y][next.x]>=low) {
                result.edge[next.y][next.x]=true;
                queue.addLast(next)
            }
        };
        val seen=Array(height) {
            BooleanArray(width)
        };
        for(y in 0 until height)for(x in 0 until width) {
            if(result.edge[y][x]&&result.nms[y][x]<high)result.weakKept++;
            if(!result.edge[y][x]||seen[y][x])continue;
            val component=mutableListOf<PgmIo.Point>();
            queue.addLast(PgmIo.Point(x,y));
            seen[y][x]=true;
            while(queue.isNotEmpty()) {
                val point=queue.removeFirst();
                component+=point;
                for(next in neighbors(point,width,height))if(result.edge[next.y][next.x]&&!seen[next.y][next.x]) {
                    seen[next.y][next.x]=true;
                    queue.addLast(next)
                }
            };
            result.components.add(component)
        };
        return result
    }

    fun images() {
        val edgeImage=Array(height) {
            IntArray(width)
        };
        val overlay=Array(height) {
            IntArray(width)
        };
        for(y in 0 until height)for(x in 0 until width) {

            overlay[y][x]=image[y][x]/params.get("overlay_divisor")
        };

        for(circle in circles)for(point in circleVotes.outlines.getValue(circle.cell.r))mark(overlay,translate(point,circle.cell.x,circle.cell.y));
        for(c in cornerResult.corners)for(e in -2..2) {
            mark(overlay,Point(c.x+e,c.y));
            mark(overlay,Point(c.x,c.y+e))
        };
        val votes=Array(lineVotes.high-lineVotes.low+1) {
            IntArray(180)
        };
        for(r in lineVotes.low..lineVotes.high)for(t in 0..179)votes[r-lineVotes.low][t]=if(lineVotes.maximum==0)0 else 255*lineVotes.count(t,
            r)/lineVotes.maximum;
        PgmIo.pgm("edges.pgm",edgeImage);
        PgmIo.pgm("hough.pgm",votes);
        PgmIo.pgm("overlay.pgm",overlay)
    }
