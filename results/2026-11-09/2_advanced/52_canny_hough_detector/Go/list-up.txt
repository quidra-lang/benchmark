func parseInt(s string) int {
	v, e := strconv.Atoi(s)
	return v
}
func readInputs() (Image, Trig, Params) {
	lines := readLines("input/image.pgm")
	image := Image{}
	for _, line := range lines[3:] {
		row := []int{}
		for _, word := range strings.Split(line, " ") {
			row = append(row, parseInt(word))
		}
		image = append(image, row)
	}
	trig := Trig{}
	for _, line := range readLines("input/trig.txt") {
		f := strings.Split(line, " ")
		trig[parseInt(f[0])] = Point{parseInt(f[1]), parseInt(f[2])}
	}
	params := Params{}
	for _, line := range readLines("input/params.txt") {
		f := strings.Split(line, " ")
		for _, word := range f[1:] {
			v, e := strconv.ParseInt(word, 10, 64)
			params[f[0]] = append(params[f[0]], v)
		}
	}
	return image, trig, params
}
func writePGM(name string, image Image) {
	lines := []string{"P2", fmt.Sprintf("%d %d", len(image[0]), len(image)), "255"}
	for _, row := range image {
		fields := make([]string, len(row))
		for i, v := range row {
			fields[i] = strconv.Itoa(v)
		}
		lines = append(lines, strings.Join(fields, " "))
	}
	writeLines(name, lines)
}
func newImage(width, height int) Image {
	out := make(Image, height)
	for y := range out {
		out[y] = make([]int, width)
	}
	return out
}
func clamp(v, low, high int) int { return min(max(v, low), high) }
func reference(image Image, x, y int) int {
	return image[clamp(y, 0, len(image)-1)][clamp(x, 0, len(image[0])-1)]
}
func abs(v int) int {
	if v < 0 {
		return -v
	}
	return v
}
func floorDiv(a, b int) int {
	q := a / b
	if a%b < 0 {
		q--
	}
	return q
}
func smooth(image Image, p Params) Image {
	width, height := len(image[0]), len(image)
	out := newImage(width, height)
	size := p.integer("gauss_size")
	half := size / 2
	kernel := p["gauss_kernel"]
	for y := 0; y < height; y++ {
		for x := 0; x < width; x++ {
			total := 0
			for j := 0; j < size; j++ {
				for i := 0; i < size; i++ {
					total += int(kernel[j*size+i]) * reference(image, x+i-half, y+j-half)
				}
			}
			out[y][x] = floorDiv(total, p.integer("gauss_divisor"))
		}
	}
	return out
}
func gradient(image Image) (Image, Image, Image) {
	width, height := len(image[0]), len(image)
	gx, gy, magnitude := newImage(width, height), newImage(width, height), newImage(width, height)
	for y := 0; y < height; y++ {
		for x := 0; x < width; x++ {
			q1, q2, q3 := reference(image, x-1, y-1), reference(image, x, y-1), reference(image, x+1, y-1)
			q4, q6 := reference(image, x-1, y), reference(image, x+1, y)
			q7, q8, q9 := reference(image, x-1, y+1), reference(image, x, y+1), reference(image, x+1, y+1)
			gx[y][x] = (q3 + 2*q6 + q9) - (q1 + 2*q4 + q7)
			gy[y][x] = (q7 + 2*q8 + q9) - (q1 + 2*q2 + q3)
			magnitude[y][x] = abs(gx[y][x]) + abs(gy[y][x])
		}
	}
	return gx, gy, magnitude
}
type Mask [][]bool

func newMask(width, height int) Mask {
	out := make(Mask, height)
	for y := range out {
		out[y] = make([]bool, width)
	}
	return out
}
func inside(p Point, width, height int) bool {
	return p.X >= 0 && p.X < width && p.Y >= 0 && p.Y < height
}
func neighbors(p Point, width, height int) []Point {
	out := []Point{}
	for dy := -1; dy <= 1; dy++ {
		for dx := -1; dx <= 1; dx++ {
			q := Point{p.X + dx, p.Y + dy}
			if (dx != 0 || dy != 0) && inside(q, width, height) {
				out = append(out, q)
			}
		}
	}
	return out
}
func directions(gx, gy Image) Image {
	out := newImage(len(gx[0]), len(gx))
	for y, row := range gx {
		for x, v := range row {
			ax, ay := abs(v), abs(gy[y][x])
			direction := 135
			if 29*ay <= 12*ax {
				direction = 0
			} else if 29*ay >= 70*ax {
				direction = 90
			} else if v*gy[y][x] < 0 {
				direction = 45
			}
			out[y][x] = direction
		}
	}
	return out
}
func suppress(magnitude, direction Image) Image {
	width, height := len(magnitude[0]), len(magnitude)
	out := newImage(width, height)
	offsets := map[int]Point{0: {-1, 0}, 45: {1, -1}, 90: {0, -1}, 135: {-1, -1}}
	for y := 1; y < height-1; y++ {
		for x := 1; x < width-1; x++ {
			d := offsets[direction[y][x]]
			v := magnitude[y][x]
			if v > magnitude[y+d.Y][x+d.X] && v >= magnitude[y-d.Y][x-d.X] {
				out[y][x] = v
			}
		}
	}
	return out
}
func hysteresis(nms Image, low, high int) (Mask, int, int, int) {
	width, height := len(nms[0]), len(nms)
	edge := newMask(width, height)
	queue := []Point{}
	strong, weak := 0, 0
	for y, row := range nms {
		for x, v := range row {
			if v >= high {
				strong++
				edge[y][x] = true
				queue = append(queue, Point{x, y})
			} else if v >= low {
				weak++
			}
		}
	}
	for i := 0; i < len(queue); i++ {
		for _, q := range neighbors(queue[i], width, height) {
			if !edge[q.Y][q.X] && nms[q.Y][q.X] >= low {
				edge[q.Y][q.X] = true
				queue = append(queue, q)
			}
		}
	}
	return edge, strong, weak, len(queue) - strong
}
func components(edge Mask) [][]Point {
	width, height := len(edge[0]), len(edge)
	seen := newMask(width, height)
	groups := [][]Point{}
	for y, row := range edge {
		for x, present := range row {
			if !present || seen[y][x] {
				continue
			}
			seen[y][x] = true
			queue := []Point{{x, y}}
			for i := 0; i < len(queue); i++ {
				for _, q := range neighbors(queue[i], width, height) {
					if edge[q.Y][q.X] && !seen[q.Y][q.X] {
						seen[q.Y][q.X] = true
						queue = append(queue, q)
					}
				}
			}
			groups = append(groups, queue)
		}
	}
	return groups
}

func overlay(image Image, segments []*Segment, circles []*Circle, corners []*Corner, divisor int) Image {
	width, height := len(image[0]), len(image)
	out := newImage(width, height)
	for y, row := range image {
		for x, v := range row {
			out[y][x] = v / divisor
		}
	}
	mark := func(q Point) {
		if inside(q, width, height) {
			out[q.Y][q.X] = 255
		}
	}
	for _, s := range segments {
		for _, q := range bresenham(s.Start, s.End) {
			mark(q)
		}
	}
	for _, c := range circles {
		for _, o := range c.Offsets {
			mark(Point{c.Cell.X + o.X, c.Cell.Y + o.Y})
		}
	}
	for _, c := range corners {
		for e := -2; e <= 2; e++ {
			mark(Point{c.Point.X + e, c.Point.Y})
			mark(Point{c.Point.X, c.Point.Y + e})
		}
	}
	return out
}
func main() {
	image, trig, p := readInputs()
	width, height := len(image[0]), len(image)
	gx, gy, magnitude := gradient(smooth(image, p))
	direction := directions(gx, gy)
	nms := suppress(magnitude, direction)
	edge, strong, weak, weakKept := hysteresis(nms, p.integer("canny_low"), p.integer("canny_high"))
	groups := components(edge)
	circleVoting := circleVote(edge, p)
	circles := chooseCircles(circleVoting, edge, p.integer("circle_max_count"))
	linePoints := []Point{}
	edgePixels, circleEdgePixels := 0, 0
	edgeImage := newImage(width, height)
	counts := map[string]int{}
	for y, row := range magnitude {
		for x, value := range row {
			if value > 0 {
				counts["gradient_nonzero"]++
			}
			counts["gradient_max"] = max(counts["gradient_max"], value)
			if nms[y][x] > 0 {
				counts["nms_kept"]++
			}
			if !edge[y][x] {
				continue
			}
			edgePixels++
			edgeImage[y][x] = 255
			counts[fmt.Sprintf("direction_%d", direction[y][x])]++
			circleEdge := false
			for _, circle := range circles {
				if circumference(circle, Point{x, y}) {
					circleEdge = true
					break
				}
			}
			if circleEdge {
				circleEdgePixels++
			} else {
				linePoints = append(linePoints, Point{x, y})
			}
		}
	}
	lineVoting := lineVote(linePoints, width, height, trig, p.integer("hough_rho_step"), p.integer("hough_threshold"))
	lines, segments, used := extractSegments(linePoints, lineVoting, trig, p, width, height)
	joins, adjacent := junctions(segments, p)
	polygons := findPolygons(segments, joins, adjacent)
	for _, polygon := range polygons {
		polygonAttributes(polygon, segments, edge, p)
		counts[polygon.Kind]++
	}
	shapes := measureShapes(polygons, circles, image, p.integer("contrast_margin"))
	corners, cornerCandidates := harris(gx, gy, p)
	onVertices, onSegments, verticesWithCorner := matchCorners(corners, polygons, segments, p.integer("corner_match_distance"))
	voteImage := newImage(180, lineVoting.High-lineVoting.Low+1)
	if lineVoting.Maximum > 0 {
		for t, row := range lineVoting.Votes {
			for r, v := range row {
				voteImage[r-lineVoting.Low][t] = 255 * v / lineVoting.Maximum
			}
		}
	}
	writePGM("edges.pgm", edgeImage)
	writePGM("hough.pgm", voteImage)
	writePGM("overlay.pgm", overlay(image, segments, circles, corners, p.integer("overlay_divisor")))
	writeLines("lines.txt", lineLines(lines))
	writeLines("circles.txt", circleLines(circles))
	writeLines("polygons.txt", polygonLines(polygons))
	writeLines("corners.txt", cornerLines(corners))
	writeLines("components.txt", componentLines(groups, edge, magnitude, direction, shapes))
	values := map[string]int{"strong": strong, "weak": weak, "edges": edgePixels, "weak_kept": weakKept,
		"weak_dropped": weak - weakKept, "edge_components": len(groups), "circle_candidates": circleVoting.Candidates,
		"circle_peaks": len(circleVoting.Peaks), "circles": len(circles), "circle_edge_pixels": circleEdgePixels,
		"line_edge_pixels": len(linePoints), "hough_max_votes": lineVoting.Maximum, "hough_peaks": len(lineVoting.Peaks),
		"lines": len(lines), "segments": len(segments), "claimed_pixels": claimed, "junctions": len(joins),
		"polygons": len(polygons), "corner_candidates": cornerCandidates, "corners": len(corners),
		"corners_on_vertices": onVertices, "corners_on_segments": onSegments, "corners_unmatched": len(corners) - onVertices - onSegments,
		"vertices_with_corner": verticesWithCorner}
	for key, value := range values {
		counts[key] = value
	}
	report := []string{fmt.Sprintf("image %d %d", width, height)}
	for _, key := range []string{"gradient_nonzero", "gradient_max", "nms_kept", "strong", "weak", "edges",
		"weak_kept", "weak_dropped", "edge_components", "direction_0", "direction_45", "direction_90",
		"direction_135", "circle_candidates", "circle_peaks", "circles", "circle_edge_pixels", "line_edge_pixels",
		"hough_max_votes", "hough_peaks", "lines", "segments", "claimed_pixels", "junctions", "polygons",
		"equilateral_triangle", "right_triangle", "triangle", "square", "rectangle", "parallelogram",
		"trapezoid", "quadrilateral", "corner_candidates", "corners", "corners_on_vertices", "corners_on_segments",
		"corners_unmatched", "vertices_with_corner"} {
		report = append(report, fmt.Sprintf("%s %d", key, counts[key]))
	}
	writeLines("report.txt", report)
}
func rho(point Point, t int, trig Trig, step int) int {
	return floorDiv(point.X*trig[t].X+point.Y*trig[t].Y, 16384*step)
}
