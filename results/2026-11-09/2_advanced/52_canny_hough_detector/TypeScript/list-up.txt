const lines = (path: string) => readFileSync(path, 'ascii').trimEnd().split('\n');
export function readImage(path: string): Image {
  const a = lines(path);
  const [width, height] = a[1].split(' ').map(Number);
  return { width, height, pixels: a.slice(3).flatMap((r) => r.split(' ').map(Number)) };
}
export function readTrig(path: string): Trig {
  const cos: number[] = [];
  const sin: number[] = [];
  for (const line of lines(path)) {
    const [t, c, s] = line.split(' ').map(Number);
    cos[t] = c;
    sin[t] = s;
  }
  return { cos, sin };
}
export function readParams(path: string): Parameters {
  const rows = lines(path).map((r) => r.split(' ').slice(1).map(Number));
  return {
    gauss_size: rows[0][0],
    gauss_kernel: rows[1],
    gauss_divisor: rows[2][0],
    canny_low: rows[3][0],
    canny_high: rows[4][0],
    circle_min_radius: rows[5][0],
    circle_max_radius: rows[6][0],
    circle_min_percent: rows[7][0],
    circle_max_count: rows[8][0],
    hough_rho_step: rows[9][0],
    hough_threshold: rows[10][0],
    hough_max_lines: rows[11][0],
    segment_tolerance: rows[12][0],
    segment_max_gap: rows[13][0],
    segment_min_length: rows[14][0],
    junction_distance: rows[15][0],
    junction_min_angle: rows[16][0],
    angle_tolerance: rows[17][0],
    square_tolerance: rows[18][0],
    contrast_margin: rows[19][0],
    harris_window: rows[20][0],
    harris_threshold: rows[21][0],
    harris_max_count: rows[22][0],
    corner_match_distance: rows[23][0],
    overlay_divisor: rows[24][0],
  };
}
export function writeImage(path: string, width: number, height: number, pixels: ArrayLike<number>) {
  const a = ['P2', `${width} ${height}`, '255'];
  for (let y = 0; y < height; y++) {
    a.push(Array.from({ length: width }, (_, x) => pixels[y * width + x]).join(' '));
  }
  writeFileSync(path, a.join('\n') + '\n', 'ascii');
}
export function at(values: ArrayLike<number>, x: number, y: number, width: number, height: number) {
  return values[Math.min(Math.max(y, 0), height - 1) * width + Math.min(Math.max(x, 0), width - 1)];
}
export function smooth(image: Image, p: Parameters) {
  const { width: w, height: h, pixels } = image;
  const size = p.gauss_size;
  const half = (size - 1) / 2;
  const result: number[] = Array(w * h).fill(0);
  for (let y = 0; y < h; y++) {
    for (let x = 0; x < w; x++) {
      let sum = 0;
      for (let j = 0; j < size; j++) {
        for (let i = 0; i < size; i++) {
          sum += p.gauss_kernel[j * size + i] * at(pixels, x + i - half, y + j - half, w, h);
        }
      }
      result[y * w + x] = Math.floor(sum / p.gauss_divisor);
    }
  }
  return result;
}
export function gradient(values: number[], w: number, h: number) {
  const gx: number[] = Array(w * h).fill(0);
  const gy: number[] = Array(w * h).fill(0);
  const magnitude: number[] = Array(w * h).fill(0);
  for (let y = 0; y < h; y++) {
    for (let x = 0; x < w; x++) {
      const q: number[] = [];
      for (let j = -1; j <= 1; j++) {
        for (let i = -1; i <= 1; i++) q.push(at(values, x + i, y + j, w, h));
      }
      const k = y * w + x;
      gx[k] = q[2] + 2 * q[5] + q[8] - (q[0] + 2 * q[3] + q[6]);
      gy[k] = q[6] + 2 * q[7] + q[8] - (q[0] + 2 * q[1] + q[2]);
      magnitude[k] = Math.abs(gx[k]) + Math.abs(gy[k]);
    }
  }
  return { gx, gy, magnitude };
}
export function neighbors(index: number, width: number, height: number) {
  const x = index % width;
  const y = Math.floor(index / width);
  const result: number[] = [];
  for (let j = -1; j <= 1; j++) {
    for (let i = -1; i <= 1; i++) {
      if (!i && !j) continue;
      const nx = x + i;
      const ny = y + j;
      if (nx >= 0 && nx < width && ny >= 0 && ny < height) result.push(ny * width + nx);
    }
  }
  return result;
}
export function directions(gx: number[], gy: number[]) {
  return gx.map((v, i) => {
    const ax = Math.abs(v);
    const ay = Math.abs(gy[i]);
    return 29 * ay <= 12 * ax ? 0 : 29 * ay >= 70 * ax ? 90 : v * gy[i] < 0 ? 45 : 135;
  });
}
export function suppress(m: number[], dir: number[], w: number, h: number) {
  const result: number[] = Array(w * h).fill(0);
  const offsets: Record<number, number> = { 0: -1, 45: 1 - w, 90: -w, 135: -1 - w };
  for (let y = 1; y < h - 1; y++) {
    for (let x = 1; x < w - 1; x++) {
      const i = y * w + x;
      const d = offsets[dir[i]];
      if (m[i] > m[i + d] && m[i] >= m[i - d]) result[i] = m[i];
    }
  }
  return result;
}
export function hysteresis(nms: number[], w: number, h: number, low: number, high: number) {
  const edges = new Uint8Array(nms.length);
  const strong = new Uint8Array(nms.length);
  const weak = new Uint8Array(nms.length);
  const queue: number[] = [];
  for (let i = 0; i < nms.length; i++) {
    if (nms[i] >= high) {
      strong[i] = edges[i] = 1;
      queue.push(i);
    } else if (nms[i] >= low) weak[i] = 1;
  }
  for (let k = 0; k < queue.length; k++) {
    for (const next of neighbors(queue[k], w, h)) {
      if (!edges[next] && nms[next] >= low) {
        edges[next] = 1;
        queue.push(next);
      }
    }
  }
  return { edges, strong, weak };
}
export function components(edges: Uint8Array, w: number, h: number) {
  const visited = new Uint8Array(edges.length);
  const groups: number[][] = [];
  for (let i = 0; i < edges.length; i++) {
    if (!edges[i] || visited[i]) continue;
    const group = [i];
    visited[i] = 1;
    for (let k = 0; k < group.length; k++) {
      for (const next of neighbors(group[k], w, h)) {
        if (edges[next] && !visited[next]) {
          visited[next] = 1;
          group.push(next);
        }
      }
    }
    groups.push(group);
  }
  return groups;
}

function main() {
  const image = readImage('input/image.pgm');
  const trig = readTrig('input/trig.txt');
  const p = readParams('input/params.txt');
  const { width: w, height: h } = image;
  const { gx, gy, magnitude } = gradient(smooth(image, p), w, h);
  const dir = directions(gx, gy);
  const nms = suppress(magnitude, dir, w, h);
  const { edges, strong, weak } = hysteresis(nms, w, h, p.canny_low, p.canny_high);
  const groups = components(edges, w, h);
  const circleDetection = circlePeaks(
    circleVotes(edges, w, h, p.circle_min_radius, p.circle_max_radius),
    w,
    h,
    p.circle_min_percent,
  );
  const circles = selectCircles(circleDetection.peaks, edges, w, h, p.circle_max_count);
  const { removed, pixels } = removeCircles(edges, circles, w);
  const votes = lineVotes(pixels, trig, p.hough_rho_step);
  const peaks = linePeaks(votes, p.hough_threshold);
  const vote = voteImage(votes, trig, w, h, p.hough_rho_step);
  const { lines, segments, used } = extractSegments(peaks, pixels, trig, p, edges.length);
  const junctions = findJunctions(segments, p);
  const polygons = findPolygons(segments, junctions.adjacency, edges.length);
  for (const polygon of polygons) polygonAttributes(polygon, edges, w, h, p);
  const shapes: Shape[] = [...polygons, ...circles];
  measureShapes(shapes, image, p.contrast_margin);
  const cornerDetection = detect(
    gx,
    gy,
    w,
    h,
    p.harris_window,
    p.harris_threshold,
    p.harris_max_count,
  );
  const corners = matchCorners(
    cornerDetection.corners,
    polygons,
    segments,
    p.corner_match_distance,
  );
  mkdirSync('output', { recursive: true });
  const write = (file: string, rows: string[]) =>
    writeFileSync(`output/${file}.txt`, rows.join('\n') + '\n', 'ascii');
  writeImage(
    'output/edges.pgm',
    w,
    h,
    edges.map((e) => e * 255),
  );
  writeImage('output/hough.pgm', 180, vote.height, vote.pixels);
  const overlay = image.pixels.map((value) => Math.floor(value / p.overlay_divisor));
  const mark = (q: Point) => {
    if (insideImage(q, w, h)) overlay[q.y * w + q.x] = 255;
  };
  for (const segment of segments) for (const q of bresenham(segment.start, segment.end)) mark(q);
  for (const circle of circles) {
    for (const q of circle.offsets) mark({ x: circle.x + q.x, y: circle.y + q.y });
  }
  for (const corner of corners) {
    for (let e = -2; e <= 2; e++) {
      mark({ x: corner.x + e, y: corner.y });
      mark({ x: corner.x, y: corner.y + e });
    }
  }
  writeImage('output/overlay.pgm', w, h, overlay);
  write('components', [
    `components ${groups.length}`,
    ...componentRows(groups, edges, magnitude, dir, shapes, w, h),
  ]);
  const weakKept = count(weak.map((v, i) => (v && edges[i] ? 1 : 0)));
  const report = [`image ${w} ${h}`];
  const metrics: [string, number][] = [
    ['gradient_nonzero', magnitude.filter((n) => n > 0).length],
    ['gradient_max', Math.max(...magnitude)],
    ['nms_kept', nms.filter((n) => n > 0).length],
    ['strong', count(strong)],
    ['weak', count(weak)],
    ['edges', count(edges)],
    ['weak_kept', weakKept],
    ['weak_dropped', count(weak) - weakKept],
    ['edge_components', groups.length],
    ...[0, 45, 90, 135].map((d): [string, number] => [
      `direction_${d}`,
      count(edges.map((e, i) => (e && dir[i] === d ? 1 : 0))),
    ]),
    ['circle_candidates', circleDetection.candidates],
    ['circle_peaks', circleDetection.peaks.length],
    ['circles', circles.length],
    ['circle_edge_pixels', count(removed)],
    ['line_edge_pixels', pixels.length],
    ['hough_max_votes', vote.maximum],
    ['hough_peaks', peaks.length],
    ['lines', lines.length],
    ['segments', segments.length],
    ['claimed_pixels', count(used)],
    ['junctions', junctions.total],
    ['polygons', polygons.length],
  ];
  report.push(...metrics.map(([name, value]) => `${name} ${value}`));
  write('report', report);
}
export const rho = (p: Point, t: number, trig: Trig, step: number) =>
  Math.floor((p.x * trig.cos[t] + p.y * trig.sin[t]) / (16384 * step));
