typedef int64_t I;
void *allocate(size_t count, size_t size) {
    void *value = calloc(count ? count : 1, size);
    return value;
}
Image image_new(int width, int height) {
    return (Image){width, height, allocate((size_t)width * height, sizeof(I))};
}
I image_at(Image image, Point point) {
    return image.data[point.y * image.width + point.x];
}
void image_set(Image image, Point point, I value) {
    image.data[point.y * image.width + point.x] = value;
}
bool image_inside(Image image, Point point) {
    return point.x >= 0 && point.x < image.width && point.y >= 0 && point.y < image.height;
}
Input load(void) {
    Input out = {0};
    FILE *input = fopen("input/image.pgm", "r");
    char key[64];
    int width, height, maximum;
    fscanf(input, "%63s %d %d %d", key, &width, &height, &maximum);
    out.image = image_new(width, height);
    for (int i = 0; i < width * height; ++i)
        fscanf(input, "%" SCNd64, &out.image.data[i]);
    fclose(input);
    input = fopen("input/trig.txt", "r");
    for (int i = 0; i < 180; ++i) {
        int angle;
        fscanf(input, "%d %" SCNd64 " %" SCNd64, &angle, &out.trig.data[i][0], &out.trig.data[i][1]);
    }
    fclose(input);
    input = fopen("input/params.txt", "r");
    Params *p = &out.params;
    fscanf(input, "%63s %" SCNd64 " %63s", key, &p->ks, key);
    p->kernel = allocate(p->ks * p->ks, sizeof(I));
    for (I i = 0; i < p->ks * p->ks; ++i)
        fscanf(input, "%" SCNd64, &p->kernel[i]);
    I *fields[] = {&p->divisor,
                   &p->low,
                   &p->high,
                   &p->r_min,
                   &p->r_max,
                   &p->circle_percent,
                   &p->circle_count,
                   &p->rho_step,
                   &p->vote_threshold,
                   &p->line_count,
                   &p->tolerance,
                   &p->max_gap,
                   &p->min_length,
                   &p->junction_distance,
                   &p->junction_angle,
                   &p->angle_tolerance,
                   &p->square_tolerance,
                   &p->margin,
                   &p->harris_window,
                   &p->harris_threshold,
                   &p->corner_count,
                   &p->match_distance,
                   &p->overlay_divisor};
    for (int i = 0; i < 23; ++i)
        fscanf(input, "%63s %" SCNd64, key, fields[i]);
    fclose(input);
    return out;
}
void write_pgm(const char *name, Image image) {
    FILE *out = fopen(format("output/%s", name), "w");
    fprintf(out, "P2\n%d %d\n255\n", image.width, image.height);
    for (int y = 0; y < image.height; ++y) {
        for (int x = 0; x < image.width; ++x)
            fprintf(out, "%s%" PRId64, x ? " " : "", image_at(image, (Point){x, y}));
        fputc('\n', out);
    }
    fclose(out);
}
I absolute(I value) {
    return value < 0 ? -value : value;
}
I floor_div(I a, I b) {
    return a / b - (a % b < 0);
}
I clamped(Image image, I x, I y) {
    if (x < 0)
        x = 0;
    if (x >= image.width)
        x = image.width - 1;
    if (y < 0)
        y = 0;
    if (y >= image.height)
        y = image.height - 1;
    return image_at(image, (Point){x, y});
}
Image smooth(Image image, const Params *p) {
    Image out = image_new(image.width, image.height);
    I half = (p->ks - 1) / 2;
    for (int y = 0; y < image.height; ++y)
        for (int x = 0; x < image.width; ++x) {
            I sum = 0;
            for (I j = 0; j < p->ks; ++j)
                for (I i = 0; i < p->ks; ++i)
                    sum += p->kernel[j * p->ks + i] * clamped(image, x + i - half, y + j - half);
            image_set(out, (Point){x, y}, floor_div(sum, p->divisor));
        }
    return out;
}
Gradient gradient(Image image) {
    Gradient out = {image_new(image.width, image.height), image_new(image.width, image.height),
                    image_new(image.width, image.height)};
    for (int y = 0; y < image.height; ++y)
        for (int x = 0; x < image.width; ++x) {
            I q[9];
            for (int j = -1; j <= 1; ++j)
                for (int i = -1; i <= 1; ++i)
                    q[(j + 1) * 3 + i + 1] = clamped(image, x + i, y + j);
            I gx = q[2] + 2 * q[5] + q[8] - q[0] - 2 * q[3] - q[6],
              gy = q[6] + 2 * q[7] + q[8] - q[0] - 2 * q[1] - q[2];
            image_set(out.gx, (Point){x, y}, gx);
            image_set(out.gy, (Point){x, y}, gy);
            image_set(out.magnitude, (Point){x, y}, absolute(gx) + absolute(gy));
        }
    return out;
}
Canny canny(Gradient g, const Params *p) {
    int w = g.gx.width, h = g.gx.height;
    Canny out = {.direction = image_new(w, h),
                 .nms = image_new(w, h),
                 .edges = image_new(w, h),
                 .components = allocate(w * h, sizeof(Points))};
    for (int y = 0; y < h; ++y)
        for (int x = 0; x < w; ++x) {
            Point point = {x, y};
            I gx = image_at(g.gx, point), gy = image_at(g.gy, point), ax = absolute(gx), ay = absolute(gy);
            I direction = 29 * ay <= 12 * ax ? 0 : 29 * ay >= 70 * ax ? 90 : gx * gy < 0 ? 45 : 135;
            image_set(out.direction, point, direction);
            if (x == 0 || y == 0 || x == w - 1 || y == h - 1)
                continue;
            Point a, b;
            if (direction == 0) {
                a = (Point){x - 1, y};
                b = (Point){x + 1, y};
            } else if (direction == 45) {
                a = (Point){x + 1, y - 1};
                b = (Point){x - 1, y + 1};
            } else if (direction == 90) {
                a = (Point){x, y - 1};
                b = (Point){x, y + 1};
            } else {
                a = (Point){x - 1, y - 1};
                b = (Point){x + 1, y + 1};
            }
            I m = image_at(g.magnitude, point);
            if (m > image_at(g.magnitude, a) && m >= image_at(g.magnitude, b))
                image_set(out.nms, point, m);
        }
    Points queue = {0};
    for (int y = 0; y < h; ++y)
        for (int x = 0; x < w; ++x) {
            Point point = {x, y};
            I value = image_at(out.nms, point);
            if (value >= p->high) {
                ++out.strong;
                image_set(out.edges, point, 1);
                point_push(&queue, point);
            } else if (value >= p->low)
                ++out.weak;
        }
    for (int head = 0; head < queue.count; ++head) {
        Point point = queue.data[head];
        for (int dy = -1; dy <= 1; ++dy)
            for (int dx = -1; dx <= 1; ++dx) {
                Point next = {point.x + dx, point.y + dy};
                if (image_inside(out.edges, next) && !image_at(out.edges, next) &&
                    image_at(out.nms, next) >= p->low) {
                    image_set(out.edges, next, 1);
                    point_push(&queue, next);
                }
            }
    }
    Image visited = image_new(w, h);
    for (int y = 0; y < h; ++y)
        for (int x = 0; x < w; ++x) {
            Point first = {x, y};
            if (!image_at(out.edges, first) || image_at(visited, first))
                continue;
            Points component = {0};
            point_push(&component, first);
            image_set(visited, first, 1);
            for (int head = 0; head < component.count; ++head) {
                Point point = component.data[head];
                for (int dy = -1; dy <= 1; ++dy)
                    for (int dx = -1; dx <= 1; ++dx) {
                        Point next = {point.x + dx, point.y + dy};
                        if (image_inside(out.edges, next) && image_at(out.edges, next) &&
                            !image_at(visited, next)) {
                            image_set(visited, next, 1);
                            point_push(&component, next);
                        }
                    }
            }
            out.components[out.count++] = component;
        }
    return out;
}
Corners corners(Gradient g, const Params *p) {
    Corners out = {.values = allocate((size_t)response.width * response.height, sizeof(Corner))};
    for (int y = 0; y < response.height; ++y)
        for (int x = 0; x < response.width; ++x) {
            Point point = {x, y};
            I value = image_at(response, point);
            if (value < p->harris_threshold)
                continue;
            bool peak = true;
            for (I dy = -1; dy <= 1 && peak; ++dy)
                for (I dx = -1; dx <= 1; ++dx) {
                    Point other = {x + dx, y + dy};
                    if ((dx == 0 && dy == 0) || !image_inside(response, other))
                        continue;
                    I comparison = image_at(response, other);
                    bool earlier = other.y < y || (other.y == y && other.x < x);
                    if (value < comparison || (value == comparison && earlier)) {
                        peak = false;
                        break;
                    }
                }
            if (peak)
                out.values[out.count++] = (Corner){point, value};
        }
static void paint(Image overlay, Point point) {
    if (image_inside(overlay, point))
        image_set(overlay, point, 255);
    Input input = load();
    Image image = input.image;
    Params p = input.params;
    Gradient g = gradient(smooth(image, &p));
    Canny edges = canny(g, &p);
    CircleVotes circle_data = circle_votes(edges.edges, &p);
    int circle_count = (int)minimum(p.circle_count, circle_data.peak_count);
    Image line_edges = image_new(image.width, image.height);
    memcpy(line_edges.data, edges.edges.data, (size_t)image.width * image.height * sizeof(I));
    I circle_edge_pixels = 0;
    for (int y = 0; y < image.height; ++y)
        for (int x = 0; x < image.width; ++x)
            if (image_at(edges.edges, (Point){x, y}))
                for (int i = 0; i < circle_count; ++i)
                    if (circumference((Point){x, y}, circles[i])) {
                        image_set(line_edges, (Point){x, y}, 0);
                        ++circle_edge_pixels;
                        break;
                    }
    LineVotes line_data = line_votes(line_edges, &input.trig, &p);
    Line *lines = allocate(p.line_count, sizeof(Line));
    int line_count = 0;
    Segments segments = {0};
    Image claimed = image_new(image.width, image.height);
    Projected *selected = allocate((size_t)image.width * image.height, sizeof(Projected));
    for (int pi = 0; pi < line_data.peak_count; ++pi) {
        if (line_count >= p.line_count)
            break;
        LinePeak peak = line_data.peaks[pi];
        int count = 0;
        for (int y = 0; y < image.height; ++y)
            for (int x = 0; x < image.width; ++x) {
                Point point = {x, y};
                if (image_at(line_edges, point) && !image_at(claimed, point) &&
                    absolute(rho(point, peak.theta, &input.trig, &p) - peak.rho) <= p.tolerance)
                    selected[count++] = (Projected){point, point.y * input.trig.data[peak.theta][0] -
                                                               point.x * input.trig.data[peak.theta][1]};
    Image overlay = image_new(image.width, image.height), edge_image = image_new(image.width, image.height),
          hough_image = image_new(line_data.votes.width, line_data.votes.height);
    for (int i = 0; i < image.width * image.height; ++i) {
        overlay.data[i] = floor_div(image.data[i], p.overlay_divisor);
        edge_image.data[i] = 255 * edges.edges.data[i];
    }
    for (int i = 0; i < hough_image.width * hough_image.height; ++i)
        hough_image.data[i] =
            line_data.maximum ? floor_div(255 * line_data.votes.data[i], line_data.maximum) : 0;
    for (int i = 0; i < segments.count; ++i) {
        Points points = bresenham(segments.data[i].start, segments.data[i].end);
        for (int j = 0; j < points.count; ++j)
            paint(overlay, points.data[j]);
    }
    for (int i = 0; i < circle_count; ++i) {
        Circle circle = circles[i];
        Points offsets = circle_data.offsets[circle.peak.radius - p.r_min];
        for (int j = 0; j < offsets.count; ++j)
            paint(overlay, (Point){circle.peak.center.x + offsets.data[j].x,
                                   circle.peak.center.y + offsets.data[j].y});
    }
    for (int i = 0; i < corner_data.count; ++i) {
        Corner corner = corner_data.values[i];
        for (I e = -2; e <= 2; ++e) {
            paint(overlay, (Point){corner.point.x + e, corner.point.y});
            paint(overlay, (Point){corner.point.x, corner.point.y + e});
        }
    }
    write_pgm("edges.pgm", edge_image);
    write_pgm("hough.pgm", hough_image);
    write_pgm("overlay.pgm", overlay);
    FILE *report = fopen("output/report.txt", "w");
    for (int i = 0; i < edges.count; ++i) {
        Points component = edges.components[i];
        I x0 = image.width - 1, y0 = image.height - 1, x1 = 0, y1 = 0, ends = 0, strength = 0,
          directions[4] = {0};
        for (int j = 0; j < component.count; ++j) {
            Point point = component.data[j];
            I neighbors = 0;
            for (I dy = -1; dy <= 1; ++dy)
                for (I dx = -1; dx <= 1; ++dx) {
                    Point next = {point.x + dx, point.y + dy};
                    if ((dx || dy) && image_inside(image, next) && image_at(edges.edges, next))
                        ++neighbors;
                }
            ends += neighbors == 1;
            strength += image_at(g.magnitude, point);
            ++directions[image_at(edges.direction, point) / 45];
        }
    const char *names[] = {"gradient_nonzero",
                           "gradient_max",
                           "nms_kept",
                           "strong",
                           "weak",
                           "edges",
                           "weak_kept",
                           "weak_dropped",
                           "edge_components",
                           "direction_0",
                           "direction_45",
                           "direction_90",
                           "direction_135",
                           "circle_candidates",
                           "circle_peaks",
                           "circles",
                           "circle_edge_pixels",
                           "line_edge_pixels",
                           "hough_max_votes",
                           "hough_peaks",
                           "lines",
                           "segments",
                           "claimed_pixels",
                           "junctions",
                           "polygons",
                           "equilateral_triangle",
                           "right_triangle",
                           "triangle",
                           "square",
                           "rectangle",
                           "parallelogram",
                           "trapezoid",
                           "quadrilateral",
                           "corner_candidates",
                           "corners",
                           "corners_on_vertices",
                           "corners_on_segments",
                           "corners_unmatched",
                           "vertices_with_corner"};
    I counts[39] = {0};
    for (int i = 0; i < image.width * image.height; ++i) {
        counts[0] += g.magnitude.data[i] > 0;
        counts[1] = maximum(counts[1], g.magnitude.data[i]);
        counts[2] += edges.nms.data[i] > 0;
        counts[5] += edges.edges.data[i];
        counts[6] += edges.edges.data[i] && edges.nms.data[i] >= p.low && edges.nms.data[i] < p.high;
        counts[22] += claimed.data[i];
        if (edges.edges.data[i])
            ++counts[9 + edges.direction.data[i] / 45];
    }
    counts[3] = edges.strong;
    counts[4] = edges.weak;
    counts[7] = edges.weak - counts[6];
    counts[8] = edges.count;
    counts[13] = circle_data.candidates;
    counts[14] = circle_data.peak_count;
    counts[15] = circle_count;
    counts[16] = circle_edge_pixels;
    counts[17] = counts[5] - circle_edge_pixels;
    counts[18] = line_data.maximum;
    counts[19] = line_data.peak_count;
    counts[20] = line_count;
    counts[21] = segments.count;
    counts[23] = junctions.count;
    counts[24] = polygons.count;
    fprintf(report, "image %d %d\n", image.width, image.height);
    for (int i = 0; i < 39; ++i)
        fprintf(report, "%s %" PRId64 "\n", names[i], counts[i]);
I rho(Point point, I theta, const Trig *trig, const Params *p) {
    return floor_div(point.x * trig->data[theta][0] + point.y * trig->data[theta][1], 16384 * p->rho_step);
}
