pub fn readLines(io: std.Io, path: []const u8) ![]const []const u8 {
    const value = try std.Io.Dir.cwd().readFileAlloc(io, path, allocator, .unlimited);
    const lines = split(value, '\n');
    return lines[0 .. lines.len - 1];
}
pub const Point = struct { x: i64, y: i64 };
pub const offsets = [_]i64{ -1, 0, 1 };
pub const Image = struct {
    width: usize,
    height: usize,
    data: []i64,
    pub fn init(width: usize, height: usize) Image {
        const data = alloc(i64, width * height);
        @memset(data, 0);
        return .{ .width = width, .height = height, .data = data };
    }
    pub fn inside(self: Image, point: Point) bool {
        return point.x >= 0 and point.x < n(self.width) and point.y >= 0 and point.y < n(self.height);
    }
    pub fn at(self: Image, point: Point) i64 {
        return self.data[u(point.y) * self.width + u(point.x)];
    }
    pub fn set(self: Image, point: Point, value: i64) void {
        self.data[u(point.y) * self.width + u(point.x)] = value;
    }
fn value(line: []const u8) i64 {
    return number(words(line)[1]);
}
pub fn load(io: std.Io) !Input {
    const lines = try readLines(io, "input/image.pgm");
    const dimensions = words(lines[1]);
    const image = Image.init(u(number(dimensions[0])), u(number(dimensions[1])));
    for (lines[3..], 0..) |line, y| for (words(line), 0..) |field, x| {
        image.data[y * image.width + x] = number(field);
    };
    var trig: Trig = undefined;
    for (try readLines(io, "input/trig.txt"), 0..) |line, t| {
        const fields = words(line);
        trig[t] = .{ number(fields[1]), number(fields[2]) };
    }
    const settings = try readLines(io, "input/params.txt");
    const coefficients = words(settings[1]);
    const kernel = alloc(i64, coefficients.len - 1);
    for (coefficients[1..], kernel) |field, *coefficient| coefficient.* = number(field);
    return .{ .image = image, .trig = trig, .params = .{ .ks = value(settings[0]), .kernel = kernel,
         .divisor = value(settings[2]), .low = value(settings[3]), .high = value(settings[4]), .r_min = value(settings[5]),
         .r_max = value(settings[6]), .circle_percent = value(settings[7]), .circle_count = value(settings[8]),
         .rho_step = value(settings[9]), .vote_threshold = value(settings[10]), .line_count = value(settings[11]),
         .tolerance = value(settings[12]), .max_gap = value(settings[13]), .min_length = value(settings[14]),
         .junction_distance = value(settings[15]), .junction_angle = value(settings[16]), .angle_tolerance = value(settings[17]),
         .square_tolerance = value(settings[18]), .margin = value(settings[19]), .harris_window = value(settings[20]),
         .harris_threshold = value(settings[21]), .corner_count = value(settings[22]), .match_distance = value(settings[23]),
         .overlay_divisor = value(settings[24]) } };
}
pub fn writePgm(io: std.Io, name: []const u8, image: Image) !void {
    var out: Text = .{};
    out.add("P2\n{d} {d}\n255\n", .{ image.width, image.height });
    for (0..image.height) |y| {
        for (0..image.width) |x| out.add("{s}{d}", .{ if (x == 0) "" else " ", image.data[y * image.width + x] });
        out.add("\n", .{});
    }
    try std.Io.Dir.cwd().writeFile(io, .{ .sub_path = fmt("output/{s}", .{name}), .data = out.value() });
}
pub fn absolute(value: i64) i64 {
    return if (value < 0) -value else value;
}
pub fn clamped(image: c.Image, x: i64, y: i64) i64 {
    return image.at(.{ .x = @max(0, @min(c.n(image.width) - 1, x)), .y = @max(0, @min(c.n(image.height) - 1, y)) });
}
pub fn smooth(image: c.Image, p: c.Params) c.Image {
    const out = c.Image.init(image.width, image.height);
    const half = @divTrunc(p.ks - 1, 2);
    for (0..image.height) |yy| for (0..image.width) |xx| {
        const x = c.n(xx);
        const y = c.n(yy);
        var sum: i64 = 0;
        for (0..c.u(p.ks)) |j| for (0..c.u(p.ks)) |i| {
            sum += p.kernel[j * c.u(p.ks) + i] * clamped(image, x + c.n(i) - half, y + c.n(j) - half);
        };
        out.set(.{ .x = x, .y = y }, @divFloor(sum, p.divisor));
    };
    return out;
}
pub const Gradient = struct { gx: c.Image, gy: c.Image, magnitude: c.Image };
pub fn gradient(image: c.Image) Gradient {
    const out: Gradient = .{ .gx = c.Image.init(image.width, image.height), .gy = c.Image.init(image.width,
         image.height), .magnitude = c.Image.init(image.width, image.height) };
    for (0..image.height) |yy| for (0..image.width) |xx| {
        const x = c.n(xx);
        const y = c.n(yy);
        var q: [9]i64 = undefined;
        for (c.offsets, 0..) |dy, j| for (c.offsets, 0..) |dx, i| {
            q[j * 3 + i] = clamped(image, x + dx, y + dy);
        };
        const gx = q[2] + 2 * q[5] + q[8] - q[0] - 2 * q[3] - q[6];
        const gy = q[6] + 2 * q[7] + q[8] - q[0] - 2 * q[1] - q[2];
        const point: c.Point = .{ .x = x, .y = y };
        out.gx.set(point, gx);
        out.gy.set(point, gy);
        out.magnitude.set(point, absolute(gx) + absolute(gy));
    };
    return out;
}
pub const Result = struct { direction: c.Image, nms: c.Image, edges: c.Image, components: []const []const c.Point,
     strong: i64 = 0, weak: i64 = 0 };
pub fn detect(g: filters.Gradient, p: c.Params) Result {
    const width = g.gx.width;
    const height = g.gx.height;
    var out: Result = .{ .direction = c.Image.init(width, height), .nms = c.Image.init(width, height),
         .edges = c.Image.init(width, height), .components = &.{} };
    for (0..height) |yy| for (0..width) |xx| {
        const x = c.n(xx);
        const y = c.n(yy);
        const point: c.Point = .{ .x = x, .y = y };
        const gx = g.gx.at(point);
        const gy = g.gy.at(point);
        const ax = filters.absolute(gx);
        const ay = filters.absolute(gy);
        const direction: i64 = if (29 * ay <= 12 * ax) 0 else if (29 * ay >= 70 * ax) 90 else if (gx * gy < 0) 45 else 135;
        out.direction.set(point, direction);
        if (xx == 0 or yy == 0 or xx == width - 1 or yy == height - 1) continue;
        const previous: c.Point = switch (direction) {
            0 => .{ .x = x - 1, .y = y },
            45 => .{ .x = x + 1, .y = y - 1 },
            90 => .{ .x = x, .y = y - 1 },
            else => .{ .x = x - 1, .y = y - 1 },
        };
        const next: c.Point = switch (direction) {
            0 => .{ .x = x + 1, .y = y },
            45 => .{ .x = x - 1, .y = y + 1 },
            90 => .{ .x = x, .y = y + 1 },
            else => .{ .x = x + 1, .y = y + 1 },
        };
        const value = g.magnitude.at(point);
        if (value > g.magnitude.at(previous) and value >= g.magnitude.at(next)) out.nms.set(point, value);
    };
    var queue: std.ArrayList(c.Point) = .empty;
    for (0..height) |yy| for (0..width) |xx| {
        const point: c.Point = .{ .x = c.n(xx), .y = c.n(yy) };
        const value = out.nms.at(point);
        if (value >= p.high) {
            out.strong += 1;
            out.edges.set(point, 1);
            c.append(c.Point, &queue, point);
        } else if (value >= p.low) out.weak += 1;
    };
    var head: usize = 0;
    while (head < queue.items.len) : (head += 1) {
        const point = queue.items[head];
        for (c.offsets) |dy| for (c.offsets) |dx| {
            const next: c.Point = .{ .x = point.x + dx, .y = point.y + dy };
            if (out.edges.inside(next) and out.edges.at(next) == 0 and out.nms.at(next) >= p.low) {
                out.edges.set(next, 1);
                c.append(c.Point, &queue, next);
            }
        };
    }
    const visited = c.Image.init(width, height);
    var components: std.ArrayList([]const c.Point) = .empty;
    for (0..height) |yy| for (0..width) |xx| {
        const first: c.Point = .{ .x = c.n(xx), .y = c.n(yy) };
        if (out.edges.at(first) == 0 or visited.at(first) != 0) continue;
        var component: std.ArrayList(c.Point) = .empty;
        c.append(c.Point, &component, first);
        visited.set(first, 1);
        head = 0;
        while (head < component.items.len) : (head += 1) {
            const point = component.items[head];
            for (c.offsets) |dy| for (c.offsets) |dx| {
                const next: c.Point = .{ .x = point.x + dx, .y = point.y + dy };
                if (out.edges.inside(next) and out.edges.at(next) != 0 and visited.at(next) == 0) {
                    visited.set(next, 1);
                    c.append(c.Point, &component, next);
                }
            };
        }
        c.append([]const c.Point, &components, component.items);
    };
    out.components = components.items;
    return out;
}
pub fn detect(g: filters.Gradient, p: c.Params) Result {
    var values: std.ArrayList(Corner) = .empty;
    for (0..response.height) |y| for (0..response.width) |x| {
        const point: c.Point = .{ .x = c.n(x), .y = c.n(y) };
        const value = response.at(point);
        if (value < p.harris_threshold) continue;
        var peak = true;
        for (c.offsets) |dy| {
            for (c.offsets) |dx| {
                const other: c.Point = .{ .x = point.x + dx, .y = point.y + dy };
                if ((dx == 0 and dy == 0) or !response.inside(other)) continue;
                const comparison = response.at(other);
                const earlier = other.y < point.y or (other.y == point.y and other.x < point.x);
                if (value < comparison or (value == comparison and earlier)) {
                    peak = false;
                    break;
                }
            }
            if (!peak) break;
        }
        if (peak) c.append(Corner, &values, .{ .point = point, .response = value });
    };

pub fn rho(point: c.Point, theta: i64, trig: *const c.Trig, p: c.Params) i64 {
    return @divFloor(point.x * trig[c.u(theta)][0] + point.y * trig[c.u(theta)][1], 16384 * p.rho_step);
}
