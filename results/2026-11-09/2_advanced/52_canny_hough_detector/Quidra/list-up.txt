int absolute(int value)
    return if value < 0 then -value else value

int clamp(int value, int low, int high)
    return if value < low then low elif value > high then high else value

int clamped(int[][] image, int x, int y)
    return image[clamp(y, 0, int(#image) - 1)][clamp(x, 0, int(#image[0]) - 1)]

int[][] smoothing(int[][] image, int size, int[] kernel, int divisor)
    half := size / 2
    int[][] result := []
    for: y < int(#image)
        int[] row := []
        for: x < int(#image[0])
            total := 0
            for: j < size
                for: i < size
                    total += kernel[j * size + i] * clamped(image, x + i - half, y + j - half)
            row = row.append(total / divisor)
        result = result.append(row)
    return result

class Gradient(int[][] gx, int[][] gy, int[][] magnitude)

Gradient gradient(int[][] image)
    int[][] gx := []
    int[][] gy := []
    int[][] magnitude := []
    for: y < int(#image)
        int[] row_x := []
        int[] row_y := []
        int[] strengths := []
        for: x < int(#image[0])
            int[] q := []
            for: -1 <= dy < 2
                for: -1 <= dx < 2
                    q = q.append(clamped(image, x + dx, y + dy))
            horizontal := (q[2] + 2 * q[5] + q[8]) - (q[0] + 2 * q[3] + q[6])
            vertical := (q[6] + 2 * q[7] + q[8]) - (q[0] + 2 * q[1] + q[2])
            row_x = row_x.append(horizontal)
            row_y = row_y.append(vertical)
            strengths = strengths.append(absolute(horizontal) + absolute(vertical))
        gx = gx.append(row_x)
        gy = gy.append(row_y)
        magnitude = magnitude.append(strengths)
    return Gradient(gx, gy, magnitude)
int[][] neighbors(int width, int height, int x, int y)
    int[][] result := []
    for: -1 <= dy < 2
        for: -1 <= dx < 2
            nx := x + dx
            ny := y + dy
            if (dx != 0 or dy != 0) and 0 <= nx < width and 0 <= ny < height
                result = result.append([nx, ny])
    return result

int[][] directions(int[][] gx, int[][] gy)
    int[][] result := []
    for: y < int(#gx)
        int[] row := []
        for: x < int(#gx[0])
            horizontal := gx[y][x]
            vertical := gy[y][x]
            ax := filters.absolute(horizontal)
            ay := filters.absolute(vertical)
            direction := if 29 * ay <= 12 * ax then 0 elif 29 * ay >= 70 * ax then 90 elif horizontal * vertical < 0 then 45 else 135
            row = row.append(direction)
        result = result.append(row)
    return result

int[][] suppress(int[][] magnitude, int[][] direction)
    height := int(#magnitude)
    width := int(#magnitude[0])
    result := [for y := (0:<int(height))[] then [for x := (0:<int(width))[] then 0]]
    for: 1 <= y < height - 1
        for: 1 <= x < width - 1
            angle := direction[y][x]
            dx := if angle == 0 or angle == 135 then -1 elif angle == 45 then 1 else 0
            dy := if angle == 0 then 0 else -1
            value := magnitude[y][x]
            if value > magnitude[y + dy][x + dx] and value >= magnitude[y - dy][x - dx]
                result[y][x] = value
    return result

class Hysteresis(bool[][] edges, bool[][] strong, bool[][] weak)

Hysteresis hysteresis(int[][] nonmax, int low, int high)
    height := int(#nonmax)
    width := int(#nonmax[0])
    edges := [for y := (0:<int(height))[] then [for x := (0:<int(width))[] then false]]
    strong := edges
    weak := edges
    int[][] queue := []
    for: y < height
        for: x < width
            value := nonmax[y][x]
            if value >= high
                strong[y][x] = true
                edges[y][x] = true
                queue = queue.append([x, y])
            elif value >= low
                weak[y][x] = true
    cursor := 0
    while cursor < int(#queue)
        point := queue[cursor]
        cursor += 1
        for next := neighbors(width, height, point[0], point[1])[]
            x := next[0]
            y := next[1]
            if not edges[y][x] and (strong[y][x] or weak[y][x])
                edges[y][x] = true
                queue = queue.append(next)
    return Hysteresis(edges, strong, weak)

int[][][] components(bool[][] edges)
    height := int(#edges)
    width := int(#edges[0])
    visited := [for y := (0:<int(height))[] then [for x := (0:<int(width))[] then false]]
    int[][][] result := []
    for: y < height
        for: x < width
            if not edges[y][x] or visited[y][x]
                continue
            int[][] queue := [[x, y]]
            visited[y][x] = true
            cursor := 0
            while cursor < int(#queue)
                point := queue[cursor]
                cursor += 1
                for next := neighbors(width, height, point[0], point[1])[]
                    nx := next[0]
                    ny := next[1]
                    if edges[ny][nx] and not visited[ny][nx]
                        visited[ny][nx] = true
                        queue = queue.append(next)
            result = result.append(queue)
    return result

    int[][] candidates := []
    maximum := 0
    for: y < height
        for: x < width
            value := response[y][x]
            if value < threshold
                continue
            peak := true
            for: -1 <= dy < 2
                for: -1 <= dx < 2
                    nx := x + dx
                    ny := y + dy
                    if (dx == 0 and dy == 0) or nx < 0 or nx >= width or ny < 0 or ny >= height
                        continue
                    other := response[ny][nx]

            if peak
                candidates = candidates.append([x, y, value])
class Lab self(pgm_io.Input data)
    int[][] image := data.image
    int[][] trig := data.trig
    pgm_io.Params settings := data.params
    height := int(#data.image)
    width := int(#data.image[0])
    filters.Gradient gradient := filters.gradient(filters.smoothing(data.image, data.params.gauss_size, data.params.gauss_kernel, data.params.gauss_divisor))
    int[][] directions := canny.directions(gradient.gx, gradient.gy)
    int[][] nonmax := canny.suppress(gradient.magnitude, directions)
    canny.Hysteresis thresholded := canny.hysteresis(nonmax, data.params.canny_low, data.params.canny_high)
    bool[][] edges := thresholded.edges
    int[][] edge_points := []
    Component[] components := []
    Circle[] circles := []
    bool[][] circle_edge := [for y := (0:<int(#data.image))[] then [for x := (0:<int(#data.image[0]))[] then false]]
    int[][] line_edge := []
    void .prepare_components()
        for: y < height
            for: x < width
                if edges[y][x]
                    edge_points = edge_points.append([x, y])
        for pixels := canny.components(edges)[]
            ends := 0
            strength := 0
            counts := [0, 0, 0, 0]
            for point := pixels[]
                x := point[0]
                y := point[1]
                adjacent := 0
                for next := canny.neighbors(width, height, x, y)[]
                    if edges[next[1]][next[0]]
                        adjacent += 1
                if adjacent == 1
                    ends += 1
    void .detect_circles()
        votes := hough.circle_votes(edge_points, width, height, settings.circle_min_radius, settings.circle_max_radius)
        result := hough.circle_peaks(votes, width, height, settings.circle_min_percent)
        peaks := sorted_circle_peaks(result.peaks)
        circle_candidate_count = result.candidate_count
        circle_peak_count = int(#peaks)
        for: i < min_value(settings.circle_max_count, int(#peaks))
            peak := peaks[i]
            outline := votes.outlines[peak.radius - settings.circle_min_radius]
            circle := Circle([peak.x, peak.y], peak.radius, peak.votes, peak.point_count, outline, 0, 0)
            gaps := circle_gaps(circle, edges, width, height)
            circle.gap = gaps[0]
            circle.arcs = gaps[1]
            circles = circles.append(circle)
        for point := edge_points[]
            on_circle := false
            for circle := circles[]
                if circumference(circle, point)
                    on_circle = true
                    break
            if on_circle
                circle_edge[point[1]][point[0]] = true
            else
                line_edge = line_edge.append(point)
    void .detect_lines()
        int[][] image_corners := [[0, 0], [width - 1, 0], [0, height - 1], [width - 1, height - 1]]
        for: angle < 180
            for point := image_corners[]
                value := hough.rho(point, angle, trig, settings.hough_rho_step)
                rho_low = min_value(rho_low, value)
                rho_high = max_value(rho_high, value)
        line_votes = hough.line_votes(line_edge, trig, settings.hough_rho_step, rho_low, rho_high)
        for row := line_votes[]
            for value := row[]
                maximum_votes = max_value(maximum_votes, value)
        peaks := hough.line_peaks(line_votes, rho_low, settings.hough_threshold)
        line_peak_count = int(#peaks)
    int[][] overlay_image()
        overlay := [for row := image[] then [for value := row[] then value / settings.overlay_divisor]]
        for segment := segments[]
            for point := bresenham(segment.first, segment.last)[]
                self.mark_pixel(&overlay, point)
        for circle := circles[]
            for offset := circle.outline[]
                self.mark_pixel(&overlay, [circle.center[0] + offset[0], circle.center[1] + offset[1]])
        for corner := detected_corners[]
            for: -2 <= offset < 3
                self.mark_pixel(&overlay, [corner.point[0] + offset, corner.point[1]])
                self.mark_pixel(&overlay, [corner.point[0], corner.point[1] + offset])
        return overlay
    string[] report_lines()

        nms_kept := 0
        strong_count := 0
        weak_count := 0
        weak_kept := 0
        circle_edge_count := 0
        claimed_count := 0
        direction_counts := [0, 0, 0, 0]
        for: y < height
            for: x < width
                value := gradient.magnitude[y][x]
                if value > 0
                    gradient_nonzero += 1
                gradient_maximum = max_value(gradient_maximum, value)
                if nonmax[y][x] > 0
                    nms_kept += 1
                if thresholded.strong[y][x]
                    strong_count += 1
                if thresholded.weak[y][x]
                    weak_count += 1
                    if edges[y][x]
                        weak_kept += 1
                if edges[y][x]

                    claimed_count += 1
        kinds := polygon_kinds()
        kind_counts := [for i := (0:<int(#kinds))[] then 0]
        for polygon := polygons[]

            "strong {strong_count}",
            "weak {weak_count}",
            "edges {#edge_points}",
            "weak_kept {weak_kept}",

            "edge_components {#components}",
        ]
        for: i < 4
            result = result.append("direction_{45 * i} {direction_counts[i]}")
        result = result.concat([
            "circle_candidates {circle_candidate_count}",
            "circle_peaks {circle_peak_count}",
            "circles {#circles}",
            "circle_edge_pixels {circle_edge_count}",
            "line_edge_pixels {#line_edge}",
            "hough_max_votes {maximum_votes}",
            "hough_peaks {line_peak_count}",
            "lines {#lines}",
            "segments {#segments}",
            "claimed_pixels {claimed_count}",
            "junctions {#junctions}",
            "polygons {#polygons}",
        ])
        for: i < int(#kinds)
            result = result.append("{kinds[i]} {kind_counts[i]}")
        return result.concat([
            "corner_candidates {corner_candidate_count}",
            "corners {#detected_corners}",

            "corners_unmatched {matches[0]}",

    void write_reports()

int floor_divide(int numerator, int denominator)
    return if numerator >= 0 then numerator / denominator else -((-numerator + denominator - 1) / denominator)

int rho(int[] point, int angle, int[][] trig, int step)
    row := trig[angle]
    return floor_divide(point[0] * row[0] + point[1] * row[1], 16384 * step)
