struct Pixel: Hashable, Comparable {
    let x: Int
    let y: Int
    static func < (a: Pixel, b: Pixel) -> Bool { a.y == b.y ? a.x < b.x : a.y < b.y }
    static func + (a: Pixel, b: Pixel) -> Pixel { Pixel(x: a.x + b.x, y: a.y + b.y) }
func readLines(_ path: String) throws -> [String] { try String(contentsOfFile: path, encoding: .ascii).split(separator: "\n").map(String.init) }
func readPGM() throws -> Image { try readLines("input/image.pgm").dropFirst(3).map { $0.components(separatedBy: " ").map { Int($0)! } } }
func readTrig() throws -> [TrigValue] {
    try readLines("input/trig.txt").map { line in let fields = line.components(separatedBy: " "); return TrigValue(cosine: Int(fields[1])!, sine: Int(fields[2])!) }
}

func readParameters() throws -> Parameters {
    let values = Dictionary(uniqueKeysWithValues: try readLines("input/params.txt").map { line in
        let fields = line.components(separatedBy: " ")
        return (fields[0], fields.dropFirst().map { Int($0)! })
    })
    func value(_ key: String) -> Int { values[key]![0] }
    return Parameters(gaussSize: value("gauss_size"), gaussKernel: values["gauss_kernel"]!, gaussDivisor: value("gauss_divisor"), cannyLow: value("canny_low"), cannyHigh: value("canny_high"),

                      houghRhoStep: value("hough_rho_step"), houghThreshold: value("hough_threshold"), houghMaxLines: value("hough_max_lines"), segmentTolerance: value("segment_tolerance"),
                      segmentMaxGap: value("segment_max_gap"), segmentMinLength: value("segment_min_length"), junctionDistance: value("junction_distance"), junctionMinAngle: value("junction_min_angle"),
                      angleTolerance: value("angle_tolerance"), squareTolerance: value("square_tolerance"), contrastMargin: value("contrast_margin"), harrisWindow: value("harris_window"),
                      harrisThreshold: value("harris_threshold"), harrisMaxCount: value("harris_max_count"), cornerMatchDistance: value("corner_match_distance"), overlayDivisor: value("overlay_divisor"))
}
func writePGM(_ name: String, _ image: Image) throws { try writeLines(name, ["P2", "\(image[0].count) \(image.count)", "255"] + image.map { $0.map(String.init).joined(separator: " ") }) }
func clamped(_ image: Image, _ x: Int, _ y: Int) -> Int { image[min(max(y, 0), image.count - 1)][min(max(x, 0), image[0].count - 1)] }

func smoothing(_ image: Image, _ params: Parameters) -> Image {
    let size = params.gaussSize, half = size / 2
    return image.indices.map { y in image[0].indices.map { x in
        var total = 0
        for j in 0..<size { for i in 0..<size { total += params.gaussKernel[j * size + i] * clamped(image, x + i - half, y + j - half) } }
        return total / params.gaussDivisor
    } }
}

struct Gradient { let x: Image; let y: Image; let magnitude: Image }

func gradient(_ image: Image) -> Gradient {
    let height = image.count, width = image[0].count
    var gx = Array(repeating: Array(repeating: 0, count: width), count: height), gy = gx, magnitude = gx
    for y in 0..<height {
        for x in 0..<width {
            let q1 = clamped(image, x - 1, y - 1), q2 = clamped(image, x, y - 1), q3 = clamped(image, x + 1, y - 1)
            let q4 = clamped(image, x - 1, y), q6 = clamped(image, x + 1, y)
            let q7 = clamped(image, x - 1, y + 1), q8 = clamped(image, x, y + 1), q9 = clamped(image, x + 1, y + 1)
            gx[y][x] = (q3 + 2 * q6 + q9) - (q1 + 2 * q4 + q7)
            gy[y][x] = (q7 + 2 * q8 + q9) - (q1 + 2 * q2 + q3)
            magnitude[y][x] = abs(gx[y][x]) + abs(gy[y][x])
        }
    }
    return Gradient(x: gx, y: gy, magnitude: magnitude)
}
let neighborOffsets = [Pixel(x: -1, y: -1), Pixel(x: 0, y: -1), Pixel(x: 1, y: -1), Pixel(x: -1, y: 0), Pixel(x: 1, y: 0), Pixel(x: -1, y: 1), Pixel(x: 0, y: 1), Pixel(x: 1, y: 1)]

func directions(_ gradient: Gradient) -> Image {
    gradient.x.indices.map { y in gradient.x[0].indices.map { x in
        let dx = gradient.x[y][x], dy = gradient.y[y][x], ax = abs(dx), ay = abs(dy)
        return 29 * ay <= 12 * ax ? 0 : 29 * ay >= 70 * ax ? 90 : dx * dy < 0 ? 45 : 135
    } }
}

func suppress(_ magnitude: Image, _ direction: Image) -> Image {
    let height = magnitude.count, width = magnitude[0].count
    var output = Array(repeating: Array(repeating: 0, count: width), count: height)
    let offsets = [0: (Pixel(x: -1, y: 0), Pixel(x: 1, y: 0)), 45: (Pixel(x: 1, y: -1), Pixel(x: -1, y: 1)), 90: (Pixel(x: 0, y: -1), Pixel(x: 0, y: 1)), 135: (Pixel(x: -1, y: -1), Pixel(x: 1, y: 1))]
    for y in 1..<(height - 1) {
        for x in 1..<(width - 1) {
            let (before, after) = offsets[direction[y][x]]!, value = magnitude[y][x]
            if value > magnitude[y + before.y][x + before.x] && value >= magnitude[y + after.y][x + after.x] { output[y][x] = value }
        }
    }
    return output
}

struct Edges { let pixels: Set<Pixel>; let strongPixels: Set<Pixel>; let weakPixels: Set<Pixel> }

func hysteresis(_ nonmax: Image, _ low: Int, _ high: Int) -> Edges {
    var strong = Set<Pixel>(), weakPixels = Set<Pixel>()
    for y in nonmax.indices {
        for x in nonmax[0].indices {
            if nonmax[y][x] >= high { strong.insert(Pixel(x: x, y: y)) } else if nonmax[y][x] >= low { weakPixels.insert(Pixel(x: x, y: y)) }
        }
    }
    let allowed = strong.union(weakPixels)
    var edges = strong, queue = strong.sorted(), cursor = 0
    while cursor < queue.count {
        let point = queue[cursor]
        cursor += 1
        for offset in neighborOffsets { let next = point + offset; if allowed.contains(next) && edges.insert(next).inserted { queue.append(next) } }
    }
    return Edges(pixels: edges, strongPixels: strong, weakPixels: weakPixels)
}

func edgeComponents(_ edges: Set<Pixel>) -> [[Pixel]] {
    var waiting = edges, components: [[Pixel]] = []
    while let first = waiting.min() {
        waiting.remove(first)
        var queue = [first], cursor = 0
        while cursor < queue.count {
            let point = queue[cursor]
            cursor += 1
            for offset in neighborOffsets { let next = point + offset; if waiting.remove(next) != nil { queue.append(next) } }
        }
        components.append(queue)
    }
    return components
}

                let other = point + offset

            }
            if keep { candidates.append(CornerCandidate(point: point, response: value)) }
        }
    }

func overlayImage(_ original: Image, _ segments: [Segment], _ circles: [Circle], _ corners: [Corner], _ divisor: Int) -> Image {
    var image = original.map { $0.map { $0 / divisor } }
    let height = image.count, width = image[0].count
    func mark(_ point: Pixel) { if point.x >= 0 && point.x < width && point.y >= 0 && point.y < height { image[point.y][point.x] = 255 } }
    for segment in segments { for point in bresenham(segment.start, segment.end) { mark(point) } }
    for circle in circles { for offset in circle.outline { mark(circle.center + offset) } }
    for corner in corners { for offset in -2...2 { mark(Pixel(x: corner.point.x + offset, y: corner.point.y)); mark(Pixel(x: corner.point.x, y: corner.point.y + offset)) } }
    return image
}

func voteImage(_ votes: [LineCell: Int], _ width: Int, _ height: Int, _ trig: [TrigValue], _ step: Int) -> (image: Image, maximum: Int) {
    let corners = [Pixel(x: 0, y: 0), Pixel(x: width - 1, y: 0), Pixel(x: 0, y: height - 1), Pixel(x: width - 1, y: height - 1)]
    let limits = (0..<180).flatMap { angle in corners.map { rho($0, angle, trig, step) } }, maximum = votes.values.max() ?? 0
    let image = (limits.min()!...limits.max()!).map { radial in (0..<180).map { angle in maximum > 0 ? 255 * (votes[LineCell(angle: angle, rho: radial)] ?? 0) / maximum : 0 } }
    return (image, maximum)
}

func floorDivide(_ numerator: Int, _ denominator: Int) -> Int { numerator / denominator - (numerator % denominator < 0 ? 1 : 0) }
func rho(_ point: Pixel, _ angle: Int, _ trig: [TrigValue], _ step: Int) -> Int { floorDivide(point.x * trig[angle].cosine + point.y * trig[angle].sine, 16384 * step) }
