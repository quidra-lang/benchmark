typedef Image = List<List<int>>;
typedef Point = (int, int);

class Params {
  final Map<String, List<int>> values;
  Params(this.values);
  int integer(String key) => values[key]![0];
  List<int> operator [](String key) => values[key]!;
}

List<String> readLines(String path) => File(path).readAsLinesSync();
(Image, List<Point>, Params) readInputs() {
  final image = [
        for (final line in readLines('input/image.pgm').skip(3))
          line.split(' ').map(int.parse).toList(),
      ],
      trig = List<Point>.filled(180, (0, 0));
  for (final line in readLines('input/trig.txt')) {
    final v = line.split(' ').map(int.parse).toList();
    trig[v[0]] = (v[1], v[2]);
  }
  final params = <String, List<int>>{};
  for (final line in readLines('input/params.txt')) {
    final f = line.split(' ');
    params[f.first] = f.skip(1).map(int.parse).toList();
  }
  return (image, trig, Params(params));
}
void writePGM(String name, Image image) {
  writeLines(name, [
    'P2',
    '${image.first.length} ${image.length}',
    '255',
    for (final row in image) row.join(' '),
  ]);
}
Image newImage(int width, int height) =>
    List.generate(height, (_) => List.filled(width, 0));
int reference(Image image, int x, int y) =>
    image[y.clamp(0, image.length - 1).toInt()][x
        .clamp(0, image.first.length - 1)
        .toInt()];
int floorDiv(int a, int b) => a ~/ b - (a < 0 && a % b != 0 ? 1 : 0);
Image smooth(Image image, Params p) {
  final width = image.first.length,
      height = image.length,
      result = newImage(width, height),
      size = p.integer('gauss_size'),
      half = size ~/ 2,
      kernel = p['gauss_kernel'];
  for (var y = 0; y < height; y++) {
    for (var x = 0; x < width; x++) {
      var total = 0;
      for (var j = 0; j < size; j++) {
        for (var i = 0; i < size; i++) {
          total +=
              kernel[j * size + i] *
              reference(image, x + i - half, y + j - half);
        }
      }
      result[y][x] = floorDiv(total, p.integer('gauss_divisor'));
    }
  }
  return result;
}

(Image, Image, Image) gradient(Image image) {
  final width = image.first.length,
      height = image.length,
      gx = newImage(width, height),
      gy = newImage(width, height),
      magnitude = newImage(width, height);
  for (var y = 0; y < height; y++) {
    for (var x = 0; x < width; x++) {
      final q1 = reference(image, x - 1, y - 1),
          q2 = reference(image, x, y - 1),
          q3 = reference(image, x + 1, y - 1),
          q4 = reference(image, x - 1, y),
          q6 = reference(image, x + 1, y),
          q7 = reference(image, x - 1, y + 1),
          q8 = reference(image, x, y + 1),
          q9 = reference(image, x + 1, y + 1);
      gx[y][x] = (q3 + 2 * q6 + q9) - (q1 + 2 * q4 + q7);
      gy[y][x] = (q7 + 2 * q8 + q9) - (q1 + 2 * q2 + q3);
      magnitude[y][x] = gx[y][x].abs() + gy[y][x].abs();
    }
  }
  return (gx, gy, magnitude);
}
typedef Mask = List<List<bool>>;
Mask newMask(int width, int height) =>
    List.generate(height, (_) => List.filled(width, false));
bool inside(Point p, int width, int height) =>
    p.$1 >= 0 && p.$1 < width && p.$2 >= 0 && p.$2 < height;
List<Point> neighbors(Point p, int width, int height) => [
  for (var dy = -1; dy <= 1; dy++)
    for (var dx = -1; dx <= 1; dx++)
      if ((dx != 0 || dy != 0) && inside((p.$1 + dx, p.$2 + dy), width, height))
        (p.$1 + dx, p.$2 + dy),
];
Image directions(Image gx, Image gy) {
  final result = newImage(gx.first.length, gx.length);
  for (var y = 0; y < gx.length; y++) {
    for (var x = 0; x < gx[y].length; x++) {
      final ax = gx[y][x].abs(), ay = gy[y][x].abs();
      result[y][x] = 29 * ay <= 12 * ax
          ? 0
          : 29 * ay >= 70 * ax
          ? 90
          : gx[y][x] * gy[y][x] < 0
          ? 45
          : 135;
    }
  }
  return result;
}

Image suppress(Image magnitude, Image direction) {
  final width = magnitude.first.length,
      height = magnitude.length,
      result = newImage(width, height);
  const offsets = <int, Point>{
    0: (-1, 0),
    45: (1, -1),
    90: (0, -1),
    135: (-1, -1),
  };
  for (var y = 1; y < height - 1; y++) {
    for (var x = 1; x < width - 1; x++) {
      final d = offsets[direction[y][x]]!, value = magnitude[y][x];
      if (value > magnitude[y + d.$2][x + d.$1] &&
          value >= magnitude[y - d.$2][x - d.$1]) {
        result[y][x] = value;
      }
    }
  }
  return result;
}

(Mask, int, int, int) hysteresis(Image nms, int low, int high) {
  final width = nms.first.length,
      height = nms.length,
      edge = newMask(width, height),
      queue = <Point>[];
  var strong = 0, weak = 0;
  for (var y = 0; y < height; y++) {
    for (var x = 0; x < width; x++) {
      final value = nms[y][x];
      if (value >= high) {
        strong++;
        edge[y][x] = true;
        queue.add((x, y));
      } else if (value >= low) {
        weak++;
      }
    }
  }
  for (var i = 0; i < queue.length; i++) {
    for (final q in neighbors(queue[i], width, height)) {
      if (!edge[q.$2][q.$1] && nms[q.$2][q.$1] >= low) {
        edge[q.$2][q.$1] = true;
        queue.add(q);
      }
    }
  }
  return (edge, strong, weak, queue.length - strong);
}

List<List<Point>> components(Mask edge) {
  final width = edge.first.length,
      height = edge.length,
      seen = newMask(width, height),
      groups = <List<Point>>[];
  for (var y = 0; y < height; y++) {
    for (var x = 0; x < width; x++) {
      if (!edge[y][x] || seen[y][x]) {
        continue;
      }
      seen[y][x] = true;
      final queue = <Point>[(x, y)];
      for (var i = 0; i < queue.length; i++) {
        for (final q in neighbors(queue[i], width, height)) {
          if (edge[q.$2][q.$1] && !seen[q.$2][q.$1]) {
            seen[q.$2][q.$1] = true;
            queue.add(q);
          }
        }
      }
      groups.add(queue);
    }
  }
  return groups;
}

Image overlay(
  Image image,
  List<Segment> segments,
  List<Circle> circles,
  List<Corner> corners,
  int divisor,
) {
  final width = image.first.length,
      height = image.length,
      result = [
        for (final row in image) [for (final v in row) v ~/ divisor],
      ];
  void mark(Point q) {
    if (inside(q, width, height)) {
      result[q.$2][q.$1] = 255;
    }
  }

  for (final segment in segments) {
    for (final q in bresenham(segment.start, segment.end)) {
      mark(q);
    }
  }
  for (final c in circles) {
    for (final o in c.offsets) {
      mark((c.cell.x + o.$1, c.cell.y + o.$2));
    }
  }
  for (final c in corners) {
    for (var e = -2; e <= 2; e++) {
      mark((c.point.$1 + e, c.point.$2));
      mark((c.point.$1, c.point.$2 + e));
    }
  }
  return result;
}
void main() {
  final (image, trig, p) = readInputs();
  final width = image.first.length, height = image.length;
  final (gx, gy, magnitude) = gradient(smooth(image, p));
  final direction = directions(gx, gy), nms = suppress(magnitude, direction);
  final (edge, strong, weak, weakKept) = hysteresis(
    nms,
    p.integer('canny_low'),
    p.integer('canny_high'),
  );
  final groups = components(edge),
      circleVoting = circleVote(edge, p),
      circles = chooseCircles(
        circleVoting,
        edge,
        p.integer('circle_max_count'),
      ),
      linePoints = <Point>[],
      edgeImage = newImage(width, height),
      counts = <String, int>{};
  void increment(String key) {
    counts[key] = (counts[key] ?? 0) + 1;
  }

  var edgePixels = 0, circleEdgePixels = 0;
  for (var y = 0; y < height; y++) {
    for (var x = 0; x < width; x++) {
      final value = magnitude[y][x];
      if (value > 0) {
        increment('gradient_nonzero');
      }
      counts['gradient_max'] = max(counts['gradient_max'] ?? 0, value);
      if (nms[y][x] > 0) {
        increment('nms_kept');
      }
      if (!edge[y][x]) {
        continue;
      }
      edgePixels++;
      edgeImage[y][x] = 255;
      increment('direction_${direction[y][x]}');
      if (circles.any((c) => circumference(c, (x, y)))) {
        circleEdgePixels++;
      } else {
        linePoints.add((x, y));
      }
    }
  }
  final lineVoting = lineVote(
    linePoints,
    width,
    height,
    trig,
    p.integer('hough_rho_step'),
    p.integer('hough_threshold'),
  );
  final (lines, segments, used) = extractSegments(
    linePoints,
    lineVoting,
    trig,
    p,
    width,
    height,
  );
  final (joins, adjacent) = junctions(segments, p);
  final polygons = findPolygons(segments, joins, adjacent);
  for (final polygon in polygons) {
    polygonAttributes(polygon, segments, edge, p);
    increment(polygon.kind);
  }
  final shapes = measureShapes(
    polygons,
    circles,
    image,
    p.integer('contrast_margin'),
  );
  final (corners, cornerCandidates) = harris(gx, gy, p);
  final (onVertices, onSegments, verticesWithCorner) = matchCorners(
    corners,
    polygons,
    segments,
    p.integer('corner_match_distance'),
  );
  final claimed = used.expand((row) => row).where((v) => v).length,
      voteImage = newImage(180, lineVoting.high - lineVoting.low + 1);
  if (lineVoting.maximum > 0) {
    for (var t = 0; t < 180; t++) {
      for (final entry in lineVoting.votes[t].entries) {
        voteImage[entry.key - lineVoting.low][t] =
            255 * entry.value ~/ lineVoting.maximum;
      }
    }
  }
  writePGM('edges.pgm', edgeImage);
  writePGM('hough.pgm', voteImage);
  writePGM(
    'overlay.pgm',
    overlay(image, segments, circles, corners, p.integer('overlay_divisor')),
  );
  writeLines('lines.txt', lineLines(lines));
  writeLines('circles.txt', circleLines(circles));
  writeLines('polygons.txt', polygonLines(polygons));
  writeLines('corners.txt', cornerLines(corners));
  writeLines(
    'components.txt',
    componentLines(groups, edge, magnitude, direction, shapes),
  );
  counts.addAll({
    'strong': strong,
    'weak': weak,
    'edges': edgePixels,
    'weak_kept': weakKept,
    'weak_dropped': weak - weakKept,
    'edge_components': groups.length,

int rho(Point p, int t, List<Point> trig, int step) =>
    floorDiv(p.$1 * trig[t].$1 + p.$2 * trig[t].$2, 16384 * step);
