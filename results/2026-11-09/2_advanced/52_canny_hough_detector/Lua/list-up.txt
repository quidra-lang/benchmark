local function lines(path)
    local result = {}
    for line in io.lines(path) do
        result[#result + 1] = line
    end
    return result
end
local function integers(line)
    local result = {}
    for value in line:gmatch("[^ ]+") do
        result[#result + 1] = assert(math.tointeger(tonumber(value)))
    end
    return result
end
function M.read()
    local rows, image = lines("input/image.pgm"), {}
    for index = 4, #rows do
        image[#image + 1] = integers(rows[index])
    end
    local trig = {}
    for _, line in ipairs(lines("input/trig.txt")) do
        local values = integers(line)
        trig[values[1]] = {values[2], values[3]}
    end
    local params = {}
    for _, line in ipairs(lines("input/params.txt")) do
        local name, tail = line:match("([^ ]+) (.+)")
        local values = integers(tail)
        params[name] = name == "gauss_kernel" and values or values[1]
    end
    return image, trig, params
end
function M.write_pgm(name, image)
    local rows = {"P2", #image[1] .. " " .. #image, "255"}
    for _, row in ipairs(image) do
        local values = {}
        for _, value in ipairs(row) do
            values[#values + 1] = tostring(value)
        end
        rows[#rows + 1] = table.concat(values, " ")
    end
    M.write_lines(name, rows)
end
function M.clamped(image, x, y)
    x, y = math.min(math.max(x, 0), #image[1] - 1), math.min(math.max(y, 0), #image - 1)
    return image[y + 1][x + 1]
end
function M.smoothing(image, size, kernel, divisor)
    local result, half = {}, size // 2
    for y = 0, #image - 1 do
        local row = {}
        for x = 0, #image[1] - 1 do
            local total = 0
            for j = 0, size - 1 do
                for i = 0, size - 1 do
                    total = total + kernel[j * size + i + 1] * M.clamped(image, x + i - half, y + j - half)
                end
            end
            row[#row + 1] = total // divisor
        end
        result[#result + 1] = row
    end
    return result
end
function M.gradient(image)
    local gx, gy, magnitude = {}, {}, {}
    for y = 0, #image - 1 do
        gx[y + 1], gy[y + 1], magnitude[y + 1] = {}, {}, {}
        for x = 0, #image[1] - 1 do
            local q = {}
            for dy = -1, 1 do
                for dx = -1, 1 do
                    q[#q + 1] = M.clamped(image, x + dx, y + dy)
                end
            end
            local horizontal = (q[3] + 2 * q[6] + q[9]) - (q[1] + 2 * q[4] + q[7])
            local vertical = (q[7] + 2 * q[8] + q[9]) - (q[1] + 2 * q[2] + q[3])
            gx[y + 1][x + 1], gy[y + 1][x + 1] = horizontal, vertical
            magnitude[y + 1][x + 1] = math.abs(horizontal) + math.abs(vertical)
        end
    end
    return gx, gy, magnitude
end
function M.neighbors(width, height, x, y)
    local result = {}
    for dy = -1, 1 do
        for dx = -1, 1 do
            local nx, ny = x + dx, y + dy
            if (dx ~= 0 or dy ~= 0) and nx >= 0 and nx < width and ny >= 0 and ny < height then
                result[#result + 1] = {nx, ny}
            end
        end
    end
    return result
end
function M.directions(gx, gy)
    local result = {}
    for y, row in ipairs(gx) do
        result[y] = {}
        for x, horizontal in ipairs(row) do
            local vertical = gy[y][x]
            local ax, ay = math.abs(horizontal), math.abs(vertical)
            result[y][x] = 29 * ay <= 12 * ax and 0 or 29 * ay >= 70 * ax and 90 or horizontal * vertical < 0 and 45 or 135
        end
    end
    return result
end
function M.suppress(magnitude, directions)
    local offsets = {[0] = {-1, 0}, [45] = {1, -1}, [90] = {0, -1}, [135] = {-1, -1}}
    local result, height, width = {}, #magnitude, #magnitude[1]
    for y = 0, height - 1 do
        result[y + 1] = {}
        for x = 0, width - 1 do
            local value = 0
            if x > 0 and x < width - 1 and y > 0 and y < height - 1 then
                local offset = offsets[directions[y + 1][x + 1]]
                local dx, dy = offset[1], offset[2]
                local current = magnitude[y + 1][x + 1]
                if current > magnitude[y + dy + 1][x + dx + 1] and current >= magnitude[y - dy + 1][x - dx + 1] then
                    value = current
                end
            end
            result[y + 1][x + 1] = value
        end
    end
    return result
end
function M.hysteresis(nonmax, low, high)
    local height, width = #nonmax, #nonmax[1]
    local edges, strong, weak, queue = {}, {}, {}, {}
    for y = 0, height - 1 do
        for x = 0, width - 1 do
            local key, value = y * width + x + 1, nonmax[y + 1][x + 1]
            if value >= high then
                strong[key], edges[key], queue[#queue + 1] = true, true, {x, y}
            elseif value >= low then
                weak[key] = true
            end
        end
    end
    local cursor = 1
    while cursor <= #queue do
        local point = queue[cursor]
        cursor = cursor + 1
        for _, next_point in ipairs(M.neighbors(width, height, point[1], point[2])) do
            local key = next_point[2] * width + next_point[1] + 1
            if not edges[key] and (strong[key] or weak[key]) then
                edges[key], queue[#queue + 1] = true, next_point
            end
        end
    end
    return edges, strong, weak
end
function M.components(edges, width, height)
    local visited, result = {}, {}
    for y = 0, height - 1 do
        for x = 0, width - 1 do
            local key = y * width + x + 1
            if edges[key] and not visited[key] then
                local queue, cursor = {{x, y}}, 1
                visited[key] = true
                while cursor <= #queue do
                    local point = queue[cursor]
                    cursor = cursor + 1
                    for _, next_point in ipairs(M.neighbors(width, height, point[1], point[2])) do
                        local next_key = next_point[2] * width + next_point[1] + 1
                        if edges[next_key] and not visited[next_key] then
                            visited[next_key], queue[#queue + 1] = true, next_point
                        end
                    end
                end
                result[#result + 1] = {number = #result + 1, pixels = queue}
            end
        end
    end
    return result
end

                        if (dx ~= 0 or dy ~= 0) and nx >= 0 and nx < width and ny >= 0 and ny < height then
                            local other = response[ny + 1][nx + 1]
                            if other > value or other == value and (ny < y or ny == y and nx < x) then
                                peak = false
                            end
                        end
                    end
                end
                if peak then
                    candidates[#candidates + 1] = {x = x, y = y, response = value}
                end
            end
        end
    end

local function report_lines()
    local gradient_nonzero, gradient_maximum, nms_kept, weak_kept = 0, 0, 0, 0
    local direction_counts, kind_counts, match_counts = {[0] = 0, [45] = 0, [90] = 0, [135] = 0}, {}, {vertex = 0, segment = 0, none = 0}
    for y = 1, height do
        for x = 1, width do
            gradient_nonzero = gradient_nonzero + (magnitude[y][x] > 0 and 1 or 0)
            gradient_maximum = math.max(gradient_maximum, magnitude[y][x])
            nms_kept = nms_kept + (nonmax[y][x] > 0 and 1 or 0)
            local pixel = (y - 1) * width + x
            if edges[pixel] then
                direction_counts[directions[y][x]] = direction_counts[directions[y][x]] + 1
                weak_kept = weak_kept + (weak[pixel] and 1 or 0)
            end
        end
    end
    for _, polygon in ipairs(polygons) do
        kind_counts[polygon.kind] = (kind_counts[polygon.kind] or 0) + 1
    end
    for _, corner in ipairs(corners) do
        match_counts[corner.match_kind] = match_counts[corner.match_kind] + 1
    end
    local result = {"image " .. width .. " " .. height, "gradient_nonzero " .. gradient_nonzero, "gradient_max " .. gradient_maximum,
        "nms_kept " .. nms_kept, "strong " .. set_count(strong), "weak " .. set_count(weak), "edges " .. #edge_points,
        "weak_kept " .. weak_kept, "weak_dropped " .. set_count(weak) - weak_kept, "edge_components " .. #components}
    for _, direction in ipairs({0, 45, 90, 135}) do
        result[#result + 1] = "direction_" .. direction .. " " .. direction_counts[direction]
    end
    local values = {{"circle_candidates", circle_candidate_count}, {"circle_peaks", #circle_peaks}, {"circles", #circles},
        {"circle_edge_pixels", set_count(circle_edge)}, {"line_edge_pixels", #line_edge}, {"hough_max_votes", maximum_votes},
        {"hough_peaks", #line_peaks}, {"lines", #lines}, {"segments", #segments}, {"claimed_pixels", set_count(used)},
        {"junctions", junction_count}, {"polygons", #polygons}}
    for _, value in ipairs(values) do
        result[#result + 1] = value[1] .. " " .. value[2]
    end
    for _, kind in ipairs(polygon_kinds) do
        result[#result + 1] = kind .. " " .. (kind_counts[kind] or 0)
    end
    for _, value in ipairs({{"corner_candidates", #corner_candidates}, {"corners", #corners}, {"corners_on_vertices", match_counts.vertex},
        {"corners_on_segments", match_counts.segment}, {"corners_unmatched", match_counts.none}, {"vertices_with_corner", vertices_with_corner}}) do
        result[#result + 1] = value[1] .. " " .. value[2]
    end
    return result
end

function M.rho(point, angle, trig, step)
    local row = trig[angle]
    return (point[1] * row[1] + point[2] * row[2]) // (16384 * step)
end
