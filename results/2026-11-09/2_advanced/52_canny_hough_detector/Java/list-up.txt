    static Data read() throws IOException {
        List<String> input = lines("image.pgm");
        String[] size = input.get(1).split(" ");
        int width = Integer.parseInt(size[0]), height = Integer.parseInt(size[1]);
        int[][] image = new int[height][width], trig = new int[180][2];
        for (int y = 0; y < height; y++) {
            String[] words = input.get(y + 3).split(" ");
            for (int x = 0; x < width; x++)
                image[y][x] = Integer.parseInt(words[x]);
        }
        List<String> table = lines("trig.txt");
        for (int t = 0; t < 180; t++) {
            String[] w = table.get(t).split(" ");
            trig[t][0] = Integer.parseInt(w[1]);
            trig[t][1] = Integer.parseInt(w[2]);
        }
        Params params = new Params();
        for (String line : lines("params.txt")) {
            String[] w = line.split(" ");
            long[] values = new long[w.length - 1];
            for (int i = 1; i < w.length; i++)
                values[i - 1] = Long.parseLong(w[i]);
            params.values.put(w[0], values);
        }
        return new Data(image, trig, params);
    }
    static void pgm(String name, int[][] image) throws IOException {
        List<String> out = new ArrayList<>(List.of("P2", image[0].length + " " + image.length, "255"));
        for (int[] row : image) {
            List<String> words = new ArrayList<>();
            for (int value : row)
                words.add(Integer.toString(value));
            out.add(String.join(" ", words));
        }
        write(name, out);
    }

    static int clamped(int[][] image, int x, int y) {
        return image[Math.min(Math.max(y, 0), image.length - 1)]
                    [Math.min(Math.max(x, 0), image[0].length - 1)];
    }
    static int[][] smooth(int[][] image, PgmIo.Params params) {
        int height = image.length, width = image[0].length, k = params.get("gauss_size"), half = k / 2;
        long[] kernel = params.row("gauss_kernel");
        int[][] result = new int[height][width];
        for (int y = 0; y < height; y++)
            for (int x = 0; x < width; x++) {
                long sum = 0;
                for (int j = 0; j < k; j++)
                    for (int i = 0; i < k; i++)
                        sum += kernel[j * k + i] * clamped(image, x + i - half, y + j - half);
                result[y][x] = (int)(sum / params.get("gauss_divisor"));
            }
        return result;
    }
    static Gradient gradient(int[][] image) {
        int height = image.length, width = image[0].length;
        int[][] gx = new int[height][width], gy = new int[height][width], m = new int[height][width];
        for (int y = 0; y < height; y++)
            for (int x = 0; x < width; x++) {
                int a = clamped(image, x - 1, y - 1), b = clamped(image, x, y - 1),
                    c = clamped(image, x + 1, y - 1), d = clamped(image, x - 1, y),
                    f = clamped(image, x + 1, y), g = clamped(image, x - 1, y + 1),
                    h = clamped(image, x, y + 1), i = clamped(image, x + 1, y + 1);
                gx[y][x] = c + 2 * f + i - (a + 2 * d + g);
                gy[y][x] = g + 2 * h + i - (a + 2 * b + c);
                m[y][x] = Math.abs(gx[y][x]) + Math.abs(gy[y][x]);
            }
        return new Gradient(gx, gy, m);
    }

    static boolean inside(int x, int y, int width, int height) {

    static List<PgmIo.Point> neighbors(PgmIo.Point p, int width, int height) {
        List<PgmIo.Point> result = new ArrayList<>();
        for (int dy = -1; dy <= 1; dy++)
            for (int dx = -1; dx <= 1; dx++)
                if ((dx != 0 || dy != 0) && inside(p.x() + dx, p.y() + dy, width, height))
                    result.add(new PgmIo.Point(p.x() + dx, p.y() + dy));
        return result;
    }
    static Result process(Filters.Gradient gradient, int low, int high) {
        int[][] m = gradient.magnitude();
        int height = m.length, width = m[0].length;
        Result result = new Result();
        result.direction = new int[height][width];
        result.nms = new int[height][width];
        result.edge = new boolean[height][width];
        Deque<PgmIo.Point> queue = new ArrayDeque<>();
        for (int y = 0; y < height; y++)
            for (int x = 0; x < width; x++) {
                int gx = gradient.gx()[y][x], gy = gradient.gy()[y][x], ax = Math.abs(gx), ay = Math.abs(gy);
                int direction = 29 * ay <= 12 * ax ? 0 : 29 * ay >= 70 * ax ? 90 : gx * gy < 0 ? 45 : 135;
                result.direction[y][x] = direction;
                if (x == 0 || y == 0 || x == width - 1 || y == height - 1)
                    continue;
                int dx = direction == 90 ? 0 : direction == 45 ? 1 : -1, dy = direction == 0 ? 0 : -1;
                if (m[y][x] > m[y + dy][x + dx] && m[y][x] >= m[y - dy][x - dx])
                    result.nms[y][x] = m[y][x];
                if (result.nms[y][x] >= high) {
                    result.strong++;
                    result.edge[y][x] = true;
                    queue.add(new PgmIo.Point(x, y));
                } else if (result.nms[y][x] >= low)
                    result.weak++;
            }
        while (!queue.isEmpty()) {
            PgmIo.Point point = queue.removeFirst();
            for (PgmIo.Point next : neighbors(point, width, height))
                if (!result.edge[next.y()][next.x()] && result.nms[next.y()][next.x()] >= low) {
                    result.edge[next.y()][next.x()] = true;
                    queue.add(next);
                }
        }
        boolean[][] seen = new boolean[height][width];
        for (int y = 0; y < height; y++)
            for (int x = 0; x < width; x++) {
                if (result.edge[y][x] && result.nms[y][x] < high)
                    result.weakKept++;
                if (!result.edge[y][x] || seen[y][x])
                    continue;
                List<PgmIo.Point> component = new ArrayList<>();
                queue.add(new PgmIo.Point(x, y));
                seen[y][x] = true;
                while (!queue.isEmpty()) {
                    PgmIo.Point point = queue.removeFirst();
                    component.add(point);
                    for (PgmIo.Point next : neighbors(point, width, height))
                        if (result.edge[next.y()][next.x()] && !seen[next.y()][next.x()]) {
                            seen[next.y()][next.x()] = true;
                            queue.add(next);
                        }
                }
                result.components.add(component);
            }
        return result;
    }

            }
        int[][] votes = new int[lineVotes.high - lineVotes.low + 1][180];
