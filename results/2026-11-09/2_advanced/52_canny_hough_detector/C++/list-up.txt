struct Point {
    I x, y;
    bool operator==(const Point &b) const {
        return x == b.x && y == b.y;
    }
    bool operator<(const Point &b) const {
        return std::tie(y, x) < std::tie(b.y, b.x);
    }
};
struct Image {
    int width = 0, height = 0;
    std::vector<I> data;
    Image() = default;
    Image(int w, int h) : width(w), height(h), data(w * h) {
    }
    bool inside(Point p) const {
        return p.x >= 0 && p.x < width && p.y >= 0 && p.y < height;
    }
    I &at(Point p) {
        return data[p.y * width + p.x];
    }
    I at(Point p) const {
        return data[p.y * width + p.x];
    }
Input load() {
    Input out;
    std::ifstream image("input/image.pgm");
    std::string marker;
    int width, height, maximum;
    image >> marker >> width >> height >> maximum;
    out.image = Image(width, height);
    for (I &value : out.image.data)
        image >> value;
    std::ifstream trig("input/trig.txt");
    for (auto &row : out.trig) {
        int angle;
        trig >> angle >> row[0] >> row[1];
    }
    std::ifstream parameters("input/params.txt");
    Params &p = out.params;
    parameters >> marker >> p.ks >> marker;
    p.kernel.resize(p.ks * p.ks);
    for (I &value : p.kernel)
        parameters >> value;
    for (I *value : {&p.divisor,
                     &p.low,
                     &p.high,
                     &p.r_min,
                     &p.r_max,
                     &p.circle_percent,
                     &p.circle_count,
                     &p.rho_step,
                     &p.vote_threshold,
                     &p.line_count,
                     &p.tolerance,
                     &p.max_gap,
                     &p.min_length,
                     &p.junction_distance,
                     &p.junction_angle,
                     &p.angle_tolerance,
                     &p.square_tolerance,
                     &p.margin,
                     &p.harris_window,
                     &p.harris_threshold,
                     &p.corner_count,
                     &p.match_distance,
                     &p.overlay_divisor})
        parameters >> marker >> *value;
    return out;
}
void write_pgm(const std::string &name, const Image &image) {
    std::ofstream out("output/" + name);
    out << "P2\n" << image.width << ' ' << image.height << "\n255\n";
    for (int y = 0; y < image.height; ++y) {
        for (int x = 0; x < image.width; ++x)
            out << (x ? " " : "") << image.at({x, y});
        out << '\n';
    }
}
I floor_div(I a, I b) {
    return a / b - (a % b < 0);
}
I clamped(const Image &image, I x, I y) {
    x = std::max<I>(0, std::min<I>(image.width - 1, x));
    y = std::max<I>(0, std::min<I>(image.height - 1, y));
    return image.at({x, y});
}
Image smooth(const Image &image, const Params &p) {
    Image out(image.width, image.height);
    const I half = (p.ks - 1) / 2;
    for (int y = 0; y < image.height; ++y)
        for (int x = 0; x < image.width; ++x) {
            I sum = 0;
            for (I j = 0; j < p.ks; ++j)
                for (I i = 0; i < p.ks; ++i)
                    sum += p.kernel[j * p.ks + i] * clamped(image, x + i - half, y + j - half);
            out.at({x, y}) = floor_div(sum, p.divisor);
        }
    return out;
}
Gradient gradient(const Image &image) {
    Gradient out{Image(image.width, image.height), Image(image.width, image.height),
                 Image(image.width, image.height)};
    for (int y = 0; y < image.height; ++y)
        for (int x = 0; x < image.width; ++x) {
            I q[9];
            for (int j = -1; j <= 1; ++j)
                for (int i = -1; i <= 1; ++i)
                    q[(j + 1) * 3 + i + 1] = clamped(image, x + i, y + j);
            const I gx = q[2] + 2 * q[5] + q[8] - q[0] - 2 * q[3] - q[6],
                    gy = q[6] + 2 * q[7] + q[8] - q[0] - 2 * q[1] - q[2];
            out.gx.at({x, y}) = gx;
            out.gy.at({x, y}) = gy;
            out.magnitude.at({x, y}) = std::abs(gx) + std::abs(gy);
        }
    return out;
}
Canny canny(const Gradient &g, const Params &p) {
    const int w = g.gx.width, h = g.gx.height;
    Canny out{Image(w, h), Image(w, h), Image(w, h), {}};
    for (int y = 0; y < h; ++y)
        for (int x = 0; x < w; ++x) {
            const Point point{x, y};
            const I gx = g.gx.at(point), gy = g.gy.at(point), ax = std::abs(gx), ay = std::abs(gy);
            const I direction = 29 * ay <= 12 * ax ? 0 : 29 * ay >= 70 * ax ? 90 : gx * gy < 0 ? 45 : 135;
            out.direction.at(point) = direction;
            if (x == 0 || y == 0 || x == w - 1 || y == h - 1)
                continue;
            Point a, b;
            if (direction == 0) {
                a = {x - 1, y};
                b = {x + 1, y};
            } else if (direction == 45) {
                a = {x + 1, y - 1};
                b = {x - 1, y + 1};
            } else if (direction == 90) {
                a = {x, y - 1};
                b = {x, y + 1};
            } else {
                a = {x - 1, y - 1};
                b = {x + 1, y + 1};
            }
            const I m = g.magnitude.at(point);
            if (m > g.magnitude.at(a) && m >= g.magnitude.at(b))
                out.nms.at(point) = m;
        }
    Points queue;
    for (int y = 0; y < h; ++y)
        for (int x = 0; x < w; ++x) {
            Point point{x, y};
            const I value = out.nms.at(point);
            if (value >= p.high) {
                ++out.strong;
                out.edges.at(point) = 1;
                queue.push_back(point);
            } else if (value >= p.low)
                ++out.weak;
        }
    for (std::size_t head = 0; head < queue.size(); ++head) {
        const Point point = queue[head];
        for (int dy = -1; dy <= 1; ++dy)
            for (int dx = -1; dx <= 1; ++dx) {
                const Point next{point.x + dx, point.y + dy};
                if (out.edges.inside(next) && !out.edges.at(next) && out.nms.at(next) >= p.low) {
                    out.edges.at(next) = 1;
                    queue.push_back(next);
                }
            }
    }
    Image visited(w, h);
    for (int y = 0; y < h; ++y)
        for (int x = 0; x < w; ++x) {
            const Point first{x, y};
            if (!out.edges.at(first) || visited.at(first))
                continue;
            Points component{first};
            visited.at(first) = 1;
            for (std::size_t head = 0; head < component.size(); ++head) {
                const Point point = component[head];
                for (int dy = -1; dy <= 1; ++dy)
                    for (int dx = -1; dx <= 1; ++dx) {
                        const Point next{point.x + dx, point.y + dy};
                        if (out.edges.inside(next) && out.edges.at(next) && !visited.at(next)) {
                            visited.at(next) = 1;
                            component.push_back(next);
                        }
                    }
            }
            out.components.push_back(std::move(component));
        }
    return out;
}

I rho(Point point, I theta, const Trig &trig, const Params &p) {
    return floor_div(point.x * trig[theta][0] + point.y * trig[theta][1], 16384 * p.rho_step);
}
