function readImage(string $path): array
{
    $lines = file($path, FILE_IGNORE_NEW_LINES);
    [$width, $height] = array_map('intval', explode(' ', $lines[1]));
    $pixels = [];
    foreach (array_slice($lines, 3) as $line) {
        foreach (explode(' ', $line) as $value) $pixels[] = (int)$value;
    }
    return compact('width', 'height', 'pixels');
}

function readTrig(string $path): array
{
    $cos = $sin = [];
    foreach (file($path, FILE_IGNORE_NEW_LINES) as $line) {
        [$theta, $c, $s] = array_map('intval', explode(' ', $line));
        $cos[$theta] = $c;
        $sin[$theta] = $s;
    }
    return compact('cos', 'sin');
}

function readParams(string $path): array
{
    $params = [];
    foreach (file($path, FILE_IGNORE_NEW_LINES) as $line) {
        $tokens = explode(' ', $line);
        $name = array_shift($tokens);
        $values = array_map('intval', $tokens);
        $params[$name] = $name === 'gauss_kernel' ? $values : $values[0];
    }
    return $params;
}

function writeImage(string $path, int $width, int $height, array $pixels): void
{
    $lines = ['P2', $width . ' ' . $height, '255'];
    for ($y = 0; $y < $height; $y++) $lines[] = implode(' ', array_slice($pixels, $y * $width, $width));
    file_put_contents($path, implode("\n", $lines) . "\n");
}
function clamped(array $values, int $x, int $y, int $width, int $height): int
{
    $x = min(max($x, 0), $width - 1);
    $y = min(max($y, 0), $height - 1);
    return $values[$y * $width + $x];
}

function smooth(array $image, array $params): array
{
    $width = $image['width'];
    $height = $image['height'];
    $size = $params['gauss_size'];
    $half = intdiv($size - 1, 2);
    $result = array_fill(0, $width * $height, 0);
    for ($y = 0; $y < $height; $y++) {
        for ($x = 0; $x < $width; $x++) {
            $sum = 0;
            for ($j = 0; $j < $size; $j++) {
                for ($i = 0; $i < $size; $i++) {
                    $sum += $params['gauss_kernel'][$j * $size + $i] * clamped($image['pixels'], $x + $i - $half, $y + $j - $half, $width, $height);
                }
            }
            $result[$y * $width + $x] = intdiv($sum, $params['gauss_divisor']);
        }
    }
    return $result;
}

function gradient(array $values, int $width, int $height): array
{
    $gx = $gy = $magnitude = array_fill(0, $width * $height, 0);
    for ($y = 0; $y < $height; $y++) {
        for ($x = 0; $x < $width; $x++) {
            $q = [];
            for ($dy = -1; $dy <= 1; $dy++) for ($dx = -1; $dx <= 1; $dx++) $q[] = clamped($values, $x + $dx, $y + $dy, $width, $height);
            $index = $y * $width + $x;
            $gx[$index] = ($q[2] + 2 * $q[5] + $q[8]) - ($q[0] + 2 * $q[3] + $q[6]);
            $gy[$index] = ($q[6] + 2 * $q[7] + $q[8]) - ($q[0] + 2 * $q[1] + $q[2]);
            $magnitude[$index] = abs($gx[$index]) + abs($gy[$index]);
        }
    }
    return compact('gx', 'gy', 'magnitude');
}
function neighbors(int $index, int $width, int $height): array
{
    $x = $index % $width;
    $y = intdiv($index, $width);
    $result = [];
    for ($dy = -1; $dy <= 1; $dy++) {
        for ($dx = -1; $dx <= 1; $dx++) {
            if ($dx === 0 && $dy === 0) continue;
            $nx = $x + $dx;
            $ny = $y + $dy;
            if ($nx >= 0 && $nx < $width && $ny >= 0 && $ny < $height) $result[] = $ny * $width + $nx;
        }
    }
    return $result;
}

function directions(array $gx, array $gy): array
{
    $result = [];
    foreach ($gx as $index => $value) {
        $ax = abs($value);
        $ay = abs($gy[$index]);
        if (29 * $ay <= 12 * $ax) $result[] = 0;
        elseif (29 * $ay >= 70 * $ax) $result[] = 90;
        else $result[] = $value * $gy[$index] < 0 ? 45 : 135;
    }
    return $result;
}

function suppress(array $magnitude, array $direction, int $width, int $height): array
{
    $result = array_fill(0, $width * $height, 0);
    $offset = [0 => -1, 45 => 1 - $width, 90 => -$width, 135 => -1 - $width];
    for ($y = 1; $y < $height - 1; $y++) {
        for ($x = 1; $x < $width - 1; $x++) {
            $index = $y * $width + $x;
            $delta = $offset[$direction[$index]];
            $m = $magnitude[$index];
            if ($m > $magnitude[$index + $delta] && $m >= $magnitude[$index - $delta]) $result[$index] = $m;
        }
    }
    return $result;
}

function hysteresis(array $nms, int $width, int $height, int $low, int $high): array
{
    $edges = $strong = $weak = array_fill(0, count($nms), 0);
    $queue = [];
    foreach ($nms as $i => $value) {
        if ($value >= $high) {
            $strong[$i] = $edges[$i] = 1;
            $queue[] = $i;
        } elseif ($value >= $low) {
            $weak[$i] = 1;
        }
    }
    for ($head = 0; $head < count($queue); $head++) {
        foreach (neighbors($queue[$head], $width, $height) as $next) {
            if (!$edges[$next] && $nms[$next] >= $low) {
                $edges[$next] = 1;
                $queue[] = $next;
            }
        }
    }
    return compact('edges', 'strong', 'weak');
}

function edgeComponents(array $edges, int $width, int $height): array
{
    $visited = array_fill(0, count($edges), 0);
    $groups = [];
    foreach ($edges as $index => $value) {
        if (!$value || $visited[$index]) continue;
        $group = [$index];
        $visited[$index] = 1;
        for ($head = 0; $head < count($group); $head++) {
            foreach (neighbors($group[$head], $width, $height) as $next) {
                if ($edges[$next] && !$visited[$next]) {
                    $visited[$next] = 1;
                    $group[] = $next;
                }
            }
        }
        $groups[] = $group;
    }
    return $groups;
}

function overlayImage(array $image, array $segments, array $circles, array $corners, int $divisor): array
{
    $width = $image['width'];
    $height = $image['height'];
    $pixels = array_map(fn($value) => intdiv($value, $divisor), $image['pixels']);
    $mark = function (array $point) use (&$pixels, $width, $height): void {
        if (insideImage($point, $width, $height)) $pixels[$point['y'] * $width + $point['x']] = 255;
    };
    foreach ($segments as $segment) foreach (bresenham($segment['start'], $segment['end']) as $point) $mark($point);
    foreach ($circles as $circle) {
        foreach ($circle['offsets'] as $offset) $mark(['x' => $circle['x'] + $offset['x'], 'y' => $circle['y'] + $offset['y']]);
    }
    foreach ($corners as $corner) {
        for ($e = -2; $e <= 2; $e++) {
            $mark(['x' => $corner['x'] + $e, 'y' => $corner['y']]);
            $mark(['x' => $corner['x'], 'y' => $corner['y'] + $e]);
        }
    }
    return $pixels;
}

function rho(array $point, int $theta, array $trig, int $step): int
{
    return floorDiv($point['x'] * $trig['cos'][$theta] + $point['y'] * $trig['sin'][$theta], 16384 * $step);
}
