    public static Data Read() {
        List<string> input = Lines("image.pgm");
        string[] size = input[1].Split(' ');
        int width = int.Parse(size[0]), height = int.Parse(size[1]);
        int[][] image = Enumerable.Range(0, height)
                            .Select(
                                _ => new int[width])
                            .ToArray(),
                trig = Enumerable.Range(0, 180)
                           .Select(
                               _ => new int[2])
                           .ToArray();
        for (int y = 0; y < height; y++) {
            string[] words = input[y + 3].Split(' ');
            for (int x = 0; x < width; x++)
                image[y][x] = int.Parse(words[x]);
        }
        List<string> table = Lines("trig.txt");
        for (int t = 0; t < 180; t++) {
            string[] w = table[t].Split(' ');
            trig[t][0] = int.Parse(w[1]);
            trig[t][1] = int.Parse(w[2]);
        }
        Params settings = new Params();
        foreach (string line in Lines("params.txt")) {
            string[] w = line.Split(' ');
            long[] values = new long[w.Length - 1];
            for (int i = 1; i < w.Length; i++)
                values[i - 1] = long.Parse(w[i]);
            settings.values[w[0]] = values;
        }
        return new Data(image, trig, settings);
    }
    public static void Pgm(string name, int[][] image) {
        List<string> output = new(["P2", image[0].Length + " " + image.Length, "255"]);
        foreach (int[] row in image) {
            List<string> words = new();
            foreach (int value in row)
                words.Add(Convert.ToString(value));
            output.Add(string.Join(" ", words));
        }
        Write(name, output);
    }

    public static int Clamped(int[][] image, int x, int y) {
        return image[Math.Min(Math.Max(y, 0), image.Length - 1)]
                    [Math.Min(Math.Max(x, 0), image[0].Length - 1)];
    }
    public static int[][] Smooth(int[][] image, PgmIo.Params settings) {
        int height = image.Length, width = image[0].Length, k = settings.Get("gauss_size"), half = k / 2;
        long[] kernel = settings.Row("gauss_kernel");
        int[][] result = Enumerable.Range(0, height)
                             .Select(
                                 _ => new int[width])
                             .ToArray();
        for (int y = 0; y < height; y++)
            for (int x = 0; x < width; x++) {
                long sum = 0;
                for (int j = 0; j < k; j++)
                    for (int i = 0; i < k; i++)
                        sum += kernel[j * k + i] * Clamped(image, x + i - half, y + j - half);
                result[y][x] = (int)(sum / settings.Get("gauss_divisor"));
            }
        return result;
    }
    public static Gradient ComputeGradient(int[][] image) {
        int height = image.Length, width = image[0].Length;
        int[][] gx = Enumerable.Range(0, height)
                         .Select(
                             _ => new int[width])
                         .ToArray(),
                gy = Enumerable.Range(0, height)
                         .Select(
                             _ => new int[width])
                         .ToArray(),
                m = Enumerable.Range(0, height)
                        .Select(
                            _ => new int[width])
                        .ToArray();
        for (int y = 0; y < height; y++)
            for (int x = 0; x < width; x++) {
                int a = Clamped(image, x - 1, y - 1), b = Clamped(image, x, y - 1),
                    c = Clamped(image, x + 1, y - 1), d = Clamped(image, x - 1, y),
                    f = Clamped(image, x + 1, y), g = Clamped(image, x - 1, y + 1),
                    h = Clamped(image, x, y + 1), i = Clamped(image, x + 1, y + 1);
                gx[y][x] = c + 2 * f + i - (a + 2 * d + g);
                gy[y][x] = g + 2 * h + i - (a + 2 * b + c);
                m[y][x] = Math.Abs(gx[y][x]) + Math.Abs(gy[y][x]);
            }
        return new Gradient(gx, gy, m);
    }

    public static bool Inside(int x, int y, int width, int height) {
        return x >= 0 && x < width && y >= 0 && y < height;
    }
    public static List<PgmIo.Point> Neighbors(PgmIo.Point p, int width, int height) {
        List<PgmIo.Point> result = new();
        for (int dy = -1; dy <= 1; dy++)
            for (int dx = -1; dx <= 1; dx++)
                if ((dx != 0 || dy != 0) && Inside(p.x + dx, p.y + dy, width, height))
                    result.Add(new PgmIo.Point(p.x + dx, p.y + dy));
        return result;
    }
    public static Result Process(Filters.Gradient gradient, int low, int high) {
        int[][] m = gradient.magnitude;
        int height = m.Length, width = m[0].Length;
        Result result = new Result();
        result.direction = Enumerable.Range(0, height)
                               .Select(
                                   _ => new int[width])
                               .ToArray();
        result.nms = Enumerable.Range(0, height)
                         .Select(
                             _ => new int[width])
                         .ToArray();
        result.edge = Enumerable.Range(0, height)
                          .Select(
                              _ => new bool[width])
                          .ToArray();
        Queue<PgmIo.Point> queue = new();
        for (int y = 0; y < height; y++)
            for (int x = 0; x < width; x++) {
                int gx = gradient.gx[y][x], gy = gradient.gy[y][x], ax = Math.Abs(gx), ay = Math.Abs(gy);
                int direction = 29 * ay <= 12 * ax ? 0 : 29 * ay >= 70 * ax ? 90 : gx * gy < 0 ? 45 : 135;
                result.direction[y][x] = direction;
                if (x == 0 || y == 0 || x == width - 1 || y == height - 1)
                    continue;
                int dx = direction == 90 ? 0 : direction == 45 ? 1 : -1, dy = direction == 0 ? 0 : -1;
                if (m[y][x] > m[y + dy][x + dx] && m[y][x] >= m[y - dy][x - dx])
                    result.nms[y][x] = m[y][x];
                if (result.nms[y][x] >= high) {
                    result.strong++;
                    result.edge[y][x] = true;
                    queue.Enqueue(new PgmIo.Point(x, y));
                } else if (result.nms[y][x] >= low)
                    result.weak++;
            }
        while (queue.Count != 0) {
            PgmIo.Point point = queue.Dequeue();
            foreach (PgmIo.Point next in Neighbors(point, width, height))
                if (!result.edge[next.y][next.x] && result.nms[next.y][next.x] >= low) {
                    result.edge[next.y][next.x] = true;
                    queue.Enqueue(next);
                }
        }
        bool[][] seen = Enumerable.Range(0, height)
                            .Select(
                                _ => new bool[width])
                            .ToArray();
        for (int y = 0; y < height; y++)
            for (int x = 0; x < width; x++) {
                if (result.edge[y][x] && result.nms[y][x] < high)
                    result.weakKept++;
                if (!result.edge[y][x] || seen[y][x])
                    continue;
                List<PgmIo.Point> component = new();
                queue.Enqueue(new PgmIo.Point(x, y));
                seen[y][x] = true;
                while (queue.Count != 0) {
                    PgmIo.Point point = queue.Dequeue();
                    component.Add(point);
                    foreach (PgmIo.Point next in Neighbors(point, width, height))
                        if (result.edge[next.y][next.x] && !seen[next.y][next.x]) {
                            seen[next.y][next.x] = true;
                            queue.Enqueue(next);
                        }
                }
                result.components.Add(component);
            }
        return result;
    }

    public static int FloorDivision(int a, int b) {
        int q = a / b;
        return a % b < 0 ? q - 1 : q;
    }
    public static int Rho(PgmIo.Point p, int t, int[][] trig, int step) {
        return Hough.FloorDivision(p.x * trig[t][0] + p.y * trig[t][1], 16384 * step);
    }
