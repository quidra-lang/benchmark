func parseValue(s, kind string) Value {
	if s == "" {
		return Value{null: true}
	}
	switch kind {
	case "STRING":
		return Value{text: s}
	case "BOOL":
		v := int64(0)
		if s == "true" {
			v = 1
		}
		return Value{number: v}
	case "DATE":
		return Value{number: dateNumber(s)}
	default:
		return Value{number: integer(s)}
	}
}
func valueText(v Value, kind string, csv bool) string {
	if v.unknown {
		return "?"
	}
	if v.null {
		if csv {
			return ""
		}
		return "NULL"
	}
	switch kind {
	case "STRING":
		return v.text
	case "BOOL":
		if v.number == 1 {
			return "true"
		}
		return "false"
	case "DATE":
		return dateText(v.number)
	default:
		return strconv.FormatInt(v.number,
			10)
	}
