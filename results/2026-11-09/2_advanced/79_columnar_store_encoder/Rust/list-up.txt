pub fn field(source: &str, kind: usize) -> Value {
    if source.is_empty() {
        Value::Null
    } else {
        match kind {
            0 => Value::Number(source.parse().unwrap()),1 => Value::Text(source.to_owned()),2 => Value::Boolean(source == "true"),_ => Value::Number(date(source))
        }
    }
}
