Cell parse_cell(string kind, string text)
    if text == ""
        return Cell(kind = kind)
    if kind == "STRING"
        return Cell(false, text = text, kind = kind)
    if kind == "BOOL"
        return Cell(false, if text == "true" then 1 else 0, kind = kind)
    int number := if kind == "DATE" then day_number(text) else int(text)
    return Cell(false, number, kind = kind)
