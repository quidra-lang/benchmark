            tokens = String.(split(line))
            kind, fields = tokens[1], tokens[2:end]
            elseif kind == "CLASS"
                push!(sim.classes, VehicleClass(fields[1], parse(Int, fields[2]), parse(Int, fields[3]), fields[4] == "Y", fields[5] == "Y"))
struct VehicleClass
    name::String
    vmax::Int
    weight::Int
    overtakes::Bool
    stops::Bool
end
