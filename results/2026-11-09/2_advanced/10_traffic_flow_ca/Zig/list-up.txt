        for (texts) |text| { for (S.lines(text)) |line| { const tokens = S.split(line, ' '); const kind = tokens[0]; const fields = tokens[1..];
            } else if (S.eq(kind, "CLASS")) { S.append(*const V.Class, &self.classes, S.create(V.Class, .{ .name = fields[0], .vmax = S.parse(fields[1]), .weight = S.parse(fields[2]), .overtakes = S.eq(fields[3], "Y"), .stops = S.eq(fields[4], "Y") }));
pub fn eq(a: []const u8, b: []const u8) bool { return std.mem.eql(u8, a, b); }
pub fn split(text: []const u8, delimiter: u8) []const []const u8 { var result: std.ArrayList([]const u8) = .empty; var it = std.mem.splitScalar(u8, text, delimiter); while (it.next()) |part| append([]const u8, &result, part); return result.items; }
