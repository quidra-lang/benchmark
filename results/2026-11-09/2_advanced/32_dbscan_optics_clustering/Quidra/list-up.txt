Result cluster(int[][] neighborhoods, int minimum)
    core := [for row := neighborhoods[] then int(#row) >= minimum]
    labels := array(#neighborhoods, fill = -2)
    number := 0
    for: index < int(#neighborhoods)
        if labels[index] != -2
            continue
        if not core[index]
            labels[index] = -1
            continue
        labels[index] = number
        queue := neighborhoods[index]
        head := 0
        while head < int(#queue)
            other := queue[head]
            head += 1
            if labels[other] == -1
                labels[other] = number
            elif labels[other] == -2
                labels[other] = number
                if core[other]
                    queue = queue.concat(neighborhoods[other])
        number += 1
    return Result(labels, core)
