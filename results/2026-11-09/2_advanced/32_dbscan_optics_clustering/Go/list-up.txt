func dbscan(neighborhoods [][]int, minPts int) ([]int, []bool) {
	labels, core := make([]int, len(neighborhoods)), make([]bool, len(neighborhoods))
	for i, n := range neighborhoods {
		labels[i] = -2
		core[i] = len(n) >= minPts
	}
	cluster := 0
	for p := range labels {
		if labels[p] != -2 {
			continue
		}
		if !core[p] {
			labels[p] = -1
			continue
		}
		labels[p] = cluster
		queue := append([]int(nil), neighborhoods[p]...)
		for head := 0; head < len(queue); head++ {
			q := queue[head]
			if labels[q] == -1 {
				labels[q] = cluster
			} else if labels[q] == -2 {
				labels[q] = cluster
				if core[q] {
					queue = append(queue, neighborhoods[q]...)
				}
			}
		}
		cluster++
	}
	return labels, core
func extract(order []int, reach, core []int64, radius int) ([]int, []bool) {
	labels, isCore := make([]int, len(order)), make([]bool, len(order))
	for i := range labels {
		labels[i] = -1
	}
	cluster := -1
	limit := int64(radius) * int64(radius)
	for _, p := range order {
		if reach[p] > limit {
			if core[p] <= limit {
				cluster++
				labels[p] = cluster
			}
		} else {
			labels[p] = cluster
		}
		isCore[p] = core[p] <= limit
	}
	return labels, isCore
