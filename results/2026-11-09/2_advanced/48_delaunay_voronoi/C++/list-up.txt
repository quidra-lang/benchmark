void Triangulation::add(int a, int b, int c) {
    directed[{a, b}] = c;
    directed[{b, c}] = a;
    directed[{c, a}] = b;
}
void Triangulation::remove(int a, int b, int c) {
    directed.erase({a, b});
    directed.erase({b, c});
    directed.erase({c, a});
}
