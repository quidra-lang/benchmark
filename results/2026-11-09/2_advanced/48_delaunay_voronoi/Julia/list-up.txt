function incircle(a,b,c,d)
    ax,ay = a[1]-d[1],a[2]-d[2]
    bx,by = b[1]-d[1],b[2]-d[2]
    cx,cy = c[1]-d[1],c[2]-d[2]
    aa,bb,cc = ax*ax+ay*ay,bx*bx+by*by,cx*cx+cy*cy
    ax*(by*cc-bb*cy)-ay*(bx*cc-bb*cx)+aa*(bx*cy-by*cx)
end
function add_triangle!(mesh,a,b,c)
    mesh.directed[(a,b)] = c
    mesh.directed[(b,c)] = a
    mesh.directed[(c,a)] = b
end
function remove_triangle!(mesh,a,b,c)

    delete!(mesh.directed,(b,c))
    delete!(mesh.directed,(c,a))
