void push_pending(distinct<int> &pending, int[] &stack, int key)
    if not pending.has(key)
        .pending.add(key)
        stack = stack.append(key)
class Mesh self(int[][] points, map<int, int> directed, int[] initial)
    int edge_key(int a, int b)
        return (a - 1) * int(#points) + b - 1

    int remaining(int a, int b)
        match directed.get(edge_key(a, b))
            int value
                return value
            none
                return 0

    void .add_triangle(int a, int b, int c)
        .directed.set(edge_key(a, b), c)
        .directed.set(edge_key(b, c), a)
        .directed.set(edge_key(c, a), b)

    void .remove_triangle(int a, int b, int c)
        .directed.remove(edge_key(a, b))
        .directed.remove(edge_key(b, c))
        .directed.remove(edge_key(c, a))
        while #stack != 0
            key := stack[-1]
            stack = stack[0:<int(#stack) - 1]
            .pending.remove(key)
            a := key / int(#points) + 1
            b := key % int(#points) + 1
            int q := remaining(b, a)
            if remaining(a, b) == site and q > 0 and predicates.incircle(points[a - 1], points[b - 1], points[site - 1], points[q - 1]) > 0
                .self.remove_triangle(a, b, site)
                .self.remove_triangle(b, a, q)
                .self.add_triangle(a, q, site)
                .self.add_triangle(q, b, site)
                push_pending(&pending, &stack, edge_key(a, q))
                push_pending(&pending, &stack, edge_key(q, b))
                flips += 1
