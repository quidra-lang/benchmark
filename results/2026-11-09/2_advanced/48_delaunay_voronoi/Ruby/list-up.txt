  def incircle(a, b, c, d)
    ax, ay = a[0] - d[0], a[1] - d[1]
    bx, by = b[0] - d[0], b[1] - d[1]
    cx, cy = c[0] - d[0], c[1] - d[1]
    aa, bb, cc = ax * ax + ay * ay, bx * bx + by * by, cx * cx + cy * cy
    ax * (by * cc - bb * cy) - ay * (bx * cc - bb * cx) + aa * (bx * cy - by * cx)
  end

      a, b = check.first
      check.delete([a, b])
      q = @directed[[b, a]]
      next unless q && incircle(@points[a], @points[b], @points[site], @points[q]).positive?

      remove(a, b, site)
      remove(b, a, q)
      add(a, q, site)
      add(q, b, site)
      check.add([a, q])
      check.add([q, b])
      flips += 1
    end
  def add(a, b, c)
    @directed[[a, b]] = c
    @directed[[b, c]] = a
    @directed[[c, a]] = b
  end

  def remove(a, b, c)
    @directed.delete([a, b])
    @directed.delete([b, c])
    @directed.delete([c, a])
  end
