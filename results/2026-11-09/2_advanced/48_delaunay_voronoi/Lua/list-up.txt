function M.incircle(a, b, c, d)
    local ax, ay, bx, by = a[1] - d[1], a[2] - d[2], b[1] - d[1], b[2] - d[2]
    local cx, cy = c[1] - d[1], c[2] - d[2]
    local aa, bb, cc = ax * ax + ay * ay, bx * bx + by * by, cx * cx + cy * cy
    return ax * (by * cc - bb * cy) - ay * (bx * cc - bb * cx) + aa * (bx * cy - by * cx)
end
function Mesh:add(a, b, c)
    self.directed[a][b], self.directed[b][c], self.directed[c][a] = c, a, b
end
function Mesh:remove(a, b, c)
    self.directed[a][b], self.directed[b][c], self.directed[c][a] = nil, nil, nil
end

            stack[#stack + 1] = {a, b, key}
        end
