Collected | none collect(Context &context, const defs.Macro definition, const Pending[] pending, int head, const lexer.Token invocation, string display_name, int offset)
    lexer.Token[][] arguments := [[]]
    lexer.Token[] separators := []
    depth := 0
    cursor := pending[head].next
    closing_index := -1
    while cursor != -1
        current := pending[cursor].token
        if current.text == ")" and depth == 0
            closing_index = cursor
            break
        if current.text == "," and depth == 0
            separators = separators.append(current)
            arguments = arguments.append([])
        else
            if current.text == "("
                depth += 1
            elif current.text == ")"
                depth -= 1
            last := int(#arguments) - 1
            arguments[last] = arguments[last].append(current)
        cursor = pending[cursor].next
    if closing_index == -1
        expansion_error(&context, invocation, display_name, offset, "unterminated argument list invoking macro {DQ}{definition.name}{DQ}")
        return none
    count := int(#arguments)
    required := int(#definition.parameters) - (if definition.variadic then 1 else 0)
    empty_zero := not definition.variadic and required == 0 and count == 1 and #arguments[0] == 0
    message := ""
    if not definition.variadic and required == 0 and not empty_zero
        message = "macro {DQ}{definition.name}{DQ} passed {count} arguments, but takes just 0"
    elif count < required
        message = "macro {DQ}{definition.name}{DQ} requires {required} arguments, but only {count} given"
    elif count > required and not definition.variadic
        message = "macro {DQ}{definition.name}{DQ} passed {count} arguments, but takes just {required}"
    if message != ""
        expansion_error(&context, invocation, display_name, offset, message)
        return none
    if definition.variadic
        lexer.Token[] variable := []
        for: required <= index < count
            if index > required
                variable = variable.append(separators[index - 1])
            variable = variable.concat(arguments[index])
        arguments = arguments[0:<required].append(variable)
    elif empty_zero
        arguments = []
    return Collected(arguments, pending[closing_index].next, pending[closing_index].token)
