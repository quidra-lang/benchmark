is_word =: 3 : 0
    if. 0=#y do. 0 return. end.
    (*./y e.Lowercase,Digits,'''' )*.(-.({.y)='''')*.-.({:y)=''''
)
is_reference =: 3 : 0
    if. 3~:#y do. 0 return. end.
    ('('=0{y)*.(')'=2{y)*.(1{y)e.Digits
)
is_number =: 3 : '((#y)e.1 2)*.*./y e.Digits'
is_element =: 3 : 0
    if. (<y)e.;:'* #' do. 1 return. end.
    if. '@'={.y do. is_word }.y else. is_word y end.
)
well_formed =: 4 : 0
    count =. #y
    if. (<x)e.TextDirectives,<'REASM' do. count>0 return. end.
    select. x
    case. 'QUIT';'STOP' do. (count>0)*.*./is_word&>y
    case. 'KEY' do.
        if. count~:2 do. 0 return. end.
        (is_word 0{::y)*.is_number 1{::y
    case. 'DECOMP';'MEMORY' do. (count>0)*.*./is_element&>y
    end.
)
load_script =: 3 : 0
    ScriptLines =: read_lines y
    Texts =: empty_table ''
    for_box. TextDirectives do. Texts =: (box,<0$a:) put_key Texts end.
    QuitWords =: StopWords =: Persist =: 0$a:
    Capacity =: 3
    Pres =: Posts =: Groups =: Keywords =: Rules =: Items =: Diagnostics =: 0$a:
    PreTable =: PostTable =: GroupWords =: FirstDefinitions =: empty_table ''
    KeywordWords =: 0$a:
    currentkeyword =. currentrule =. _1
    ignoring =. 0
    for_index. i.#ScriptLines do.
        line =. index{::ScriptLines
        if. 0=#line do. continue. end.
        if. '#'={.line do. continue. end.
        parts =. ' ' split line
        name =. 0{::parts
        args =. }.parts
        text =. ' ' join args
        if. (<name)e.GlobalDirectives,<'KEY' do.
            currentkeyword =. currentrule =. _1
            ignoring =. 0
        elseif. (<name)e.RuleDirectives,ItemDirectives do.
            if. ignoring do. continue. end.
            if. (<name)e.RuleDirectives do. currentrule =. _1 end.
        end.
        code =. message =. ''
        unique =. ''
        if. -.(<name)e.Directives do.
            code =. 'SYNTAX'
            message =. 'unknown directive ',name
        elseif. -.name well_formed args do.
            code =. 'SYNTAX'
            message =. 'bad arguments for ',name
        elseif. ((<name)e.RuleDirectives)*.currentkeyword=_1 do.
            code =. 'ORPHAN'
            message =. name,' without KEY'
        elseif. (name-:'REASM')*.currentrule=_1 do.
            code =. 'ORPHAN'
            message =. 'REASM without DECOMP or MEMORY'
        elseif. (<name)e.MoveDirectives do.
            orphan =. currentrule=_1
            if. -.orphan do. orphan =. 'MEMORY'-:0{::currentrule{::Rules end.
            if. orphan do.
                code =. 'ORPHAN'
                message =. name,' without DECOMP'
            end.
        end.
        if. 0=#code do.
            if. name-:'MEMSIZE' do. unique =. name
            elseif. (<name)e.;:'KEY SYNON PRE POST PERSIST' do. unique =. name,' ',0{::args end.
            if. #unique do.
                if. unique has_key FirstDefinitions do.
                    code =. 'DUPLICATE'
                    message =. unique,' first defined at line ',number unique get_key FirstDefinitions
                    if. name-:'KEY' do. ignoring =. 1 end.
                end.
            end.
        end.
        if. #code do.
            add_diagnostic (<index+1),(<code),<message
            continue.
        end.
        if. #unique do. FirstDefinitions =: ((<unique),<index+1) put_key FirstDefinitions end.
        if. (<name)e.TextDirectives do.
            Texts =: ((<name),<(name get_key Texts),<text) put_key Texts
        elseif. name-:'QUIT' do. QuitWords =: ~.QuitWords,args
        elseif. name-:'STOP' do. StopWords =: ~.StopWords,args
        elseif. name-:'KEY' do.
            currentkeyword =. #Keywords
            KeywordWords =: KeywordWords,0{args
            Keywords =: Keywords,<(0{args),(<num 1{::args),(<index+1),(<0$0),<0$0
        elseif. (<name)e.RuleDirectives do.
            currentrule =. #Rules
            Rules =: Rules,<(<name),(<args),(<index+1),<0$0
            field =. 4-'MEMORY'-:name
            keyword =. currentkeyword{::Keywords
            keyword =. (<(field{::keyword),currentrule) field}keyword
            Keywords =: (<keyword) currentkeyword}Keywords
        else.
            itemid =. #Items
            Items =: Items,<(<name),(<text),(<args),<index+1
            rule =. currentrule{::Rules
            rule =. (<(3{::rule),itemid) 3}rule
            Rules =: (<rule) currentrule}Rules
        end.
    end.
    i.0
)
word_fits =: 4 : 0
    if. x-:'#' do. 1 return. end.
    if. '@'={.x do.
        group =. }.x
        if. -.group has_key GroupWords do. 0 return. end.
        (<y)e.group get_key GroupWords
    else. x-:y end.

match_search =: 3 : 0
    'pattern words position start bindings' =. y
    if. position=#pattern do. (<start=#words),<bindings return. end.
    element =. position{::pattern
    if. element-:'*' do.
    elseif. start<#words do.
        if. element word_fits start{::words do.
            match_search (<pattern),(<words),(<position+1),(<start+1),<bindings return.
        end.
    end.
    (<0),<0$a:
)
match_pattern =: 4 : 'match_search (<x),(<y),(<0),(<0),<0$a:'
build_reply =: 4 : 0
    bindings =. x
    text =. ''
    position =. 0
    while. position<#y do.
        chunk =. (3<.(#y)-position){.position}.y
        if. is_reference chunk do.
            value =. Digits i.1{chunk
            text =. text,' ' join reflect_words value binding bindings
            position =. position+3
        else.
            text =. text,position{y
            position =. position+1
        end.
    end.
    parts =. ' ' split text
    text =. ' ' join parts#~0<#&>parts
    if. #text do. text =. (toupper {.text),}.text end.
    text
)

            lines =. lines,<line

        end.
    end.
    lines,<line

normalize =: 3 : 0
    text =. tolower y
    text =. LF (I.text e.',.?!;')}text
    clauses =. 0$a:
    for_box. LF split text do.
        part =. >box
        part =. ' ' (I.-.part e.Lowercase,Digits,'''')}part
        words =. 0$a:
        for_wordbox. words_of part do.
            word =. >wordbox
            while. #word do.
                if. ''''~:{.word do. break. end.
                word =. }.word
            end.
            while. #word do.
                if. ''''~:{:word do. break. end.
                word =. }:word
            end.
            if. #word do. words =. words,<word end.
        end.
        if. #words do. clauses =. clauses,<words end.
    end.
    clauses
)

keyword_reply =: 3 : 0
    selected =. select_clause y
    if. selected=_1 do.
        DecisionTrace =: DecisionTrace,<'CLAUSE none'
        (<0),<'' return.
    end.
    original =. selected{::y
    keys =. keyword_list original
    labels =. 0$a:
    for_id. keys do. labels =. labels,0{id{::Keywords end.
    DecisionTrace =: DecisionTrace,<'CLAUSE ',(number selected+1),' ',' ' join original
    DecisionTrace =: DecisionTrace,<'KEYS ',' ' join labels
    keyword =. ({.keys){::Keywords
    found =. (3{::keyword) find_rule original
    if. 0{::found do.
        'matched rulenum ruleid bindings' =. found
        DecisionTrace =: DecisionTrace,<'MEMORY ',(0{::keyword),' ',number rulenum
        RuleHits =: (1+ruleid{RuleHits) ruleid}RuleHits
        if. #3{::ruleid{::Rules do.
            'itemnum item' =. take_item ruleid
            (0{::keyword) memory_store bindings build_reply 1{::item
        end.
    end.
    entered =. 0$0
    for_id. keys do.
        if. id e.entered do. continue. end.
        current =. id
        words =. original
        DecisionTrace =: DecisionTrace,<'ENTER ',0{::current{::Keywords
        while. 1 do.
            entered =. entered,current
            KeywordUses =: (1+current{KeywordUses) current}KeywordUses
            keyword =. current{::Keywords
            word =. 0{::keyword
            found =. (4{::keyword) find_rule words
            if. -.0{::found do.
                DecisionTrace =: DecisionTrace,<'NOMATCH ',word
                break.
            end.
            'matched rulenum ruleid bindings' =. found
            RuleHits =: (1+ruleid{RuleHits) ruleid}RuleHits
            if. 0=#3{::ruleid{::Rules do.
                DecisionTrace =: DecisionTrace,<'DECOMP ',word,' ',(number rulenum),' 0'
                break.
            end.
            'itemnum item' =. take_item ruleid
            DecisionTrace =: DecisionTrace,<'DECOMP ',word,' ',(number rulenum),' ',number itemnum
            'kind text args itemline' =. item
            if. kind-:'REASM' do.
                KeywordReplies =: (1+current{KeywordReplies) current}KeywordReplies
                Topics =: word increment_count Topics
                (<1),<bindings build_reply text return.
            end.
        end.
    end.
    (<0),<''
)
decide =: 3 : 0
    clauses =. normalize y
    DecisionTrace =: 0$a:
    repeated =. (0<#'REPEAT' get_key Texts)*.(0<#clauses)*.clauses-:Previous
    if. repeated do.
        DecisionTrace =: <'REPEAT'
        reply =. take_text 'REPEAT'
        source =. 4
    else.
        result =. keyword_reply pre_substitute clauses
        source =. 0
        if. 0{::result do. reply =. 1{::result
        elseif. #Memory do.
            reply =. memory_recall ''
            source =. 1
        elseif. ('?'={:y)*.0<#'QUESTION' get_key Texts do.
            reply =. take_text 'QUESTION'
            source =. 2
        else.
            reply =. take_text 'DEFAULT'
            source =. 3
        end.
    end.
    Previous =: clauses
    SourceCounts =: (1+source{SourceCounts) source}SourceCounts
    if. source e.2 3 do. Unrecognized =: Unrecognized,Turns end.
    DecisionTrace =: DecisionTrace,<'REPLY ',source{::Sources
    (<reply),<DecisionTrace
)

debug_command =: 3 : 0
    parts =. ' ' split y
    name =. 0{::parts
    args =. }.parts
    text =. ' ' join args
    if. -.(<name)e.Commands do. <'unknown command ',name return. end.
    if. -.name arguments_fit (<text),<args do. <'bad arguments' return. end.
    if. (<name)e.KeywordCommands do.
        if. -.(0{args)e.KeywordWords do. <'unknown keyword ',0{::args return. end.
    end.
    select. name
    case. ':undo' do.
        if. 0=#UndoSnapshot do. <'nothing to undo' return. end.
        restore_state UndoSnapshot
        <'undone turn ',number Turns+1
    case. ':say';':explain' do.
        if. is_quit text do. <'would quit' return. end.
        saved =. save_state 1
        result =. decide text
        restore_state saved
        if. name-:':say' do. <0{::result else. 1{::result end.

run_session =: 4 : 0
    'name lines' =. y
    new_session name
    greetings =. 'INITIAL' get_key Texts
    Transcript =: Transcript,(<'SESSION ',name),eliza_lines ((#greetings)|x){::greetings
    for_box. lines do.
        line =. >box
        if. ':'={.line do.
            Transcript =: Transcript,<line
            output =. debug_command line
            Transcript =: Transcript,indent_lines output
            continue.
        end.
        Transcript =: Transcript,<'YOU: ',line
        if. is_quit line do.
            farewells =. 'FINAL' get_key Texts
            Transcript =: Transcript,eliza_lines ((#farewells)|x){::farewells
            break.
        end.
        UndoSnapshot =: save_state 0
        Turns =: Turns+1
        for_wordbox. all_words normalize line do.
            if. -.wordbox e.StopWords do. WordCounts =: (>wordbox) increment_count WordCounts end.
        end.
        History =: History,<(number Turns),': ',line
        result =. decide line
        Transcript =: Transcript,eliza_lines 0{::result
        LastTrace =: 1{::result
        if. TraceOn do. Transcript =: Transcript,indent_lines LastTrace end.
    end.
    write_notes ''
)
