bool has_word(string[] words, string target)
    for word := words[]
        if word == target
            return true
    return false
bool is_word(string text)
    if #text == 0 or text[0] == "'" or text[#text - 1] == "'"
        return false
    for: index < #text
        if not "abcdefghijklmnopqrstuvwxyz0123456789'".contains(text[index])
            return false
    return true
bool is_number(string text)
    if #text < 1 or #text > 2
        return false
    for: index < #text
        if not "0123456789".contains(text[index])
            return false
    return true
bool is_reference(string text)
    return #text == 3 and text[0] == "(" and "0123456789".contains(text[1]) and text[2] == ")"
bool is_element(string text)
    if text == "*" or text == "#"
        return true
    if text.starts_with("@")
        return is_word(text[1:<#text])
    return is_word(text)
bool all_words(string[] words)
    for word := words[]
        if not is_word(word)
            return false
    return true
bool well_formed(string name, string[] arguments)
    count := int(#arguments)
    if has_word(["INITIAL", "FINAL", "DEFAULT", "QUESTION", "REPEAT", "REASM"], name)
        return count >= 1
    if name == "QUIT" or name == "STOP"
        return count >= 1 and all_words(arguments)
    if name == "MEMSIZE"
        return count == 1 and is_number(arguments[0]) and 1 <= int(arguments[0]) <= 20
    if name == "PERSIST"
        return count == 1 and has_word(["cycles", "memory"], arguments[0])
    if name == "PRE" or name == "SYNON"
        return count >= 2 and all_words(arguments)
    if name == "POST"
        return count >= 2 and is_word(arguments[0])
    if name == "KEY"
        return count == 2 and is_word(arguments[0]) and is_number(arguments[1])
    if name == "DECOMP" or name == "MEMORY"
        if count == 0
            return false
        for argument := arguments[]
            if not is_element(argument)
                return false
        return true

string definition_key(string name, string[] arguments)

Script load_script()
    string source := string.from_utf8(file.read_bin("input/script.txt"))
    lines := source.split(NL)
    lines = lines[0:<#lines - 1]
    result := Script(lines)
    map<string, int> first_lines
    keyword := -1
    rule := -1
    ignoring := false
    globals := ["INITIAL", "FINAL", "QUIT", "STOP", "DEFAULT", "QUESTION", "REPEAT", "MEMSIZE", "PERSIST", "PRE", "POST", "SYNON"]
    string[] known := globals.concat(["KEY", "DECOMP", "MEMORY", "REASM", "GOTO", "NEWKEY", "RESTATE"])
    for: offset < #lines
        line := lines[offset]
        if line == "" or line.starts_with("#")
            continue
        words := line.split(" ")
        name := words[0]
        arguments := words[1:<#words]
        text := arguments.join(" ")
        number := int(offset) + 1
        rule_line := name == "DECOMP" or name == "MEMORY"
        item_line := has_word(["REASM", "GOTO", "NEWKEY", "RESTATE"], name)
        if has_word(globals, name) or name == "KEY"
            keyword = -1
            rule = -1
            ignoring = false
        elif rule_line or item_line
            if ignoring
                continue
            if rule_line
                rule = -1
        code := ""
        message := ""
        unique := ""
        if not has_word(known, name)
            code = "SYNTAX"
            message = "unknown directive {name}"
        elif not well_formed(name, arguments)
            code = "SYNTAX"
            message = "bad arguments for {name}"
        elif rule_line and keyword < 0
            code = "ORPHAN"
            message = "{name} without KEY"
        elif name == "REASM" and rule < 0
            code = "ORPHAN"
            message = "REASM without DECOMP or MEMORY"
        elif has_word(["GOTO", "NEWKEY", "RESTATE"], name) and (rule < 0 or result.rules[rule].kind == "MEMORY")
            code = "ORPHAN"
            message = "{name} without DECOMP"
        else
            unique = definition_key(name, arguments)
            if unique != ""
                match first_lines.get(unique)
                    int first
                        code = "DUPLICATE"
                        message = "{unique} first defined at line {first}"
                    none
                        message = ""
        if code != ""
            result.diagnostics = result.diagnostics.append(Diagnostic(number, code, message))
            if code == "DUPLICATE" and name == "KEY"
                ignoring = true
            continue
        if unique != ""
            .first_lines.set(unique, number)
        if has_word(["INITIAL", "FINAL", "DEFAULT", "QUESTION", "REPEAT"], name)
            .result.texts.set(name, result.text_list(name).append(text))
        elif name == "QUIT"
            result.quit_words = result.quit_words.concat(arguments)
        elif name == "STOP"
            result.stop_words = result.stop_words.concat(arguments)
        elif name == "MEMSIZE"
            result.memsize = int(arguments[0])
        elif name == "PERSIST"
            result.persist = result.persist.append(arguments[0])
        elif name == "KEY"
            keyword = int(#result.keywords)
            result.keywords = result.keywords.append(Keyword(arguments[0], int(arguments[1]), number))
        elif rule_line
            rule = int(#result.rules)
            result.rules = result.rules.append(Rule(name, arguments, number))
            if name == "MEMORY"
                result.keywords[keyword].memories = result.keywords[keyword].memories.append(rule)
            else
                result.keywords[keyword].decomps = result.keywords[keyword].decomps.append(rule)
        else
            item := int(#result.items)
            result.items = result.items.append(Item(name, text, arguments, number))
            result.rules[rule].items = result.rules[rule].items.append(item)
    return result
bool is_wildcard(string element)

class MatchResult(bool found, string[][] bindings := [])
MatchResult match_from(string[] pattern, string[] words, loader.Script script, int position, int start, string[][] bindings)
    if position == int(#pattern)
        return MatchResult(start == int(#words), bindings)
    element := pattern[position]
    if element == "*"
        for: start <= finish <= int(#words)
            result := match_from(pattern, words, script, position + 1, finish, bindings.append(words[start:<finish]))
            if result.found
                return result
    elif start < int(#words) and fits(element, words[start], script)
        next := if is_wildcard(element) then bindings.append([words[start]]) else bindings
        return match_from(pattern, words, script, position + 1, start + 1, next)
    return MatchResult(false)
MatchResult match_pattern(string[] pattern, string[] words, loader.Script script)
    return match_from(pattern, words, script, 0, 0, [])
string build_reply(string template, string[][] bindings, loader.Script script)
    text := ""
    position := 0
    while position < int(#template)
        if position + 3 <= int(#template) and loader.is_reference(template[position:<position + 3])
            position += 3
        else
            text += template[position]
            position += 1
    string[] words := []
    for word := text.split(" ")[]
        if word != ""
            words = words.append(word)
    text = words.join(" ")
    if #text > 0
        for: 0 <= index < 26
            if text[0] == "abcdefghijklmnopqrstuvwxyz"[index]
                text = "ABCDEFGHIJKLMNOPQRSTUVWXYZ"[index] + text[1:<#text]
                break
    return text

class MemoryQueue(int capacity)
    string[] texts := []
    string[] log := []
    bool .store(string prefix, string keyword, string text)
        for previous := texts[]
            if previous == text
                log = log.append("{prefix} SKIP {keyword} {text}")
                return false
        if int(#texts) == capacity
            log = log.append("{prefix} DROP {texts[0]}")
            texts = texts[1:<#texts]
        texts = texts.append(text)
        log = log.append("{prefix} STORE {keyword} {text}")
        return true
    void .clear(string prefix)
        if #texts > 0
            log = log.append("{prefix} CLEAR {#texts}")
            texts = []
string[] normalized_part(string text)
    cleaned := ""
    for: index < #text
        character := text[index]
        cleaned += if "abcdefghijklmnopqrstuvwxyz0123456789'".contains(character) then character else " "
    string[] result := []
    for token := cleaned.split(" ")[]
        word := token
        while #word > 0 and word[0] == "'"
            word = word[1:<#word]
        while #word > 0 and word[#word - 1] == "'"
            word = word[0:<#word - 1]
        if word != ""
            result = result.append(word)
    return result
string[][] normalize(string text)
    string[][] result := []
    part := ""
    for: index < #text
        character := text[index]
        int[] codes := character.codepoints()
        if 65 <= codes[0] <= 90
            character = "abcdefghijklmnopqrstuvwxyz"[codes[0] - 65]
        if ",.?!;".contains(character)
            words := normalized_part(part)
            if #words > 0
                result = result.append(words)
            part = ""
        else
            part += character
    words := normalized_part(part)
    if #words > 0
        result = result.append(words)
    return result

class Selection(number := 0, string[] words := [], int[] keywords := [])
Selection select_clause(string[][] clauses, loader.Script script)
    for: clause_index < #clauses
        int[] ids := []
        for word := clauses[clause_index][]
            id := script.keyword_index(word)
            if id >= 0 and not has_id(ids, id)
                ids = ids.append(id)
        if #ids == 0
            continue
        for: 1 <= offset < int(#ids)
            current := ids[offset]
            cursor := offset
            while cursor > 0 and script.keywords[ids[cursor - 1]].rank < script.keywords[current].rank
                ids[cursor] = ids[cursor - 1]
                cursor -= 1
            ids[cursor] = current
        return Selection(int(clause_index) + 1, clauses[clause_index], ids)
    return Selection()

class Chatbot(loader.Script script)
    state := State(script)
    State saved := state
    undo_available := false
    string[] transcript := []
    sessions := 0
    total_turns := 0
    total_sources := array(5, fill = 0)
    string prefix()
        return "{state.session.name} {state.session.turns}"
    void .cycle_reset()
        state.positions = array(#script.rules, fill = 0)
        state.list_positions = array(3, fill = 0)
    string .take_text(string kind)
        index := if kind == "DEFAULT" then 0 elif kind == "QUESTION" then 1 else 2
        texts := script.text_list(kind)
        int position := state.list_positions[index]
        state.list_positions[index] = (position + 1) % int(#texts)
        return texts[position]
    Taken .take_item(int rule_id)
        loader.Rule rule := script.rules[rule_id]
        int position := state.positions[rule_id]
        item := rule.items[position]
        state.positions[rule_id] = (position + 1) % int(#rule.items)
        state.item_uses[item] += 1
        return Taken(position + 1, item)
    string .keyword_reply(string[][] clauses, string[] &trace)
        selected := select_clause(clauses, script)
        if selected.number == 0
            trace = trace.append("CLAUSE none")
            return ""
        trace = trace.append("CLAUSE {selected.number} {selected.words.join(" ")}")
        string[] names := []
        for id := selected.keywords[]
            names = names.append(script.keywords[id].word)
        trace = trace.append("KEYS " + names.join(" "))
        loader.Keyword first := script.keywords[selected.keywords[0]]
        for: offset < #first.memories
            rule_id := first.memories[offset]
            loader.Rule rule := script.rules[rule_id]
            matched := matcher.match_pattern(rule.pattern, selected.words, script)
            if not matched.found
                continue
            trace = trace.append("MEMORY {first.word} {offset + 1}")
            state.hits[rule_id] += 1
            if #rule.items > 0
                Taken taken := .take_item(rule_id)
                text := responder.build_reply(script.items[taken.item].text, matched.bindings, script)
                if .state.queue.store(prefix(), first.word, text)
                    state.session.stored += 1
            break
        entered := array(#script.keywords, fill = false)
        for initial := selected.keywords[]
            if entered[initial]
                continue
            current := initial
            words := selected.words
            trace = trace.append("ENTER {script.keywords[current].word}")
            while true
                entered[current] = true
                state.keyword_uses[current] += 1
                loader.Keyword keyword := script.keywords[current]
                rule_id := -1
                rule_number := 0
                matched := matcher.MatchResult(false)
                for: offset < #keyword.decomps
                    candidate := keyword.decomps[offset]
                    matched = matcher.match_pattern(script.rules[candidate].pattern, words, script)
                    if matched.found
                        rule_id = candidate
                        rule_number = int(offset) + 1
                        break
                if rule_id < 0
                    trace = trace.append("NOMATCH {keyword.word}")
                    break
                state.hits[rule_id] += 1
                if #script.rules[rule_id].items == 0
                    trace = trace.append("DECOMP {keyword.word} {rule_number} 0")
                    break
                Taken taken := .take_item(rule_id)
                loader.Item item := script.items[taken.item]
                trace = trace.append("DECOMP {keyword.word} {rule_number} {taken.number}")
                if item.kind == "REASM"
                    state.keyword_replies[current] += 1
                    increment(&state.session.topics, keyword.word)
                    return responder.build_reply(item.text, matched.bindings, script)
        return ""
    Reply .respond(string text)
        clauses := normalize(text)
        string[] trace := []
        answer := ""
        source := 0
        if #script.text_list("REPEAT") > 0 and #clauses > 0 and state.session.has_previous and clauses == state.session.previous
            trace = ["REPEAT"]
            answer = .take_text("REPEAT")
            source = 4
        else
            answer = .keyword_reply(prepare(text, script), &trace)
        sources := ["keyword", "memory", "question", "default", "repeat"]
        trace = trace.append("REPLY {sources[source]}")
        state.session.previous = clauses
        state.session.has_previous = true
        state.session.sources[source] += 1
        if source == 2 or source == 3
            state.session.unrecognized = state.session.unrecognized.append(state.session.turns)
        return Reply(answer, trace)
    void .start_session(string name)
        if sessions > 0
            transcript = transcript.append("")
        if not loader.has_word(script.persist, "cycles")
            .cycle_reset()
        if not loader.has_word(script.persist, "memory")
            .state.queue.clear(name + " 0")
        state.session = Session(name)
        undo_available = false
        sessions += 1
        transcript = transcript.append("SESSION {name}")
        initial := script.text_list("INITIAL")
        transcript = transcript.concat(responder.eliza_lines(initial[(sessions - 1) % int(#initial)]))
    void .end_session()
        Session current := state.session
        missing := if #current.unrecognized == 0 then "none" else number_list(current.unrecognized)
        transcript = transcript.concat(["NOTES {current.name}", "  " + source_line(current.turns, current.sources), "  topics: " + ranked_counts(current.topics), "  words: " + ranked_counts(current.words), "  memory: stored={current.stored} recalled={current.recalled} pending={#state.queue.texts}", "  unrecognized: " + missing])
        total_turns += current.turns
        for: index < #total_sources
            total_sources[index] += current.sources[index]
    void .utterance(string text)
        transcript = transcript.append("YOU: " + text)
        if is_quit(text, script)
            finals := script.text_list("FINAL")
            transcript = transcript.concat(responder.eliza_lines(finals[(sessions - 1) % int(#finals)]))
            .end_session()
            return
        saved = state
        undo_available = true
        state.session.turns += 1
        for word := flatten(normalize(text))[]
            if not loader.has_word(script.stop_words, word)
                increment(&state.session.words, word)
        state.session.history = state.session.history.append("{state.session.turns}: {text}")
        Reply answer := .respond(text)
        state.session.last_trace = answer.trace
        transcript = transcript.concat(responder.eliza_lines(answer.text))
        if state.session.trace_on
            for line := answer.trace[]
                transcript = transcript.append("  " + line)
    int criterion(int id, bool usage)
        return if usage then state.keyword_uses[id] else script.keywords[id].rank
    int[] ordered_keywords(bool usage)
        string[] names := []
        for keyword := script.keywords[]
            names = names.append(keyword.word)
        names = names.sorted()
        int[] ids := []
        for word := names[]
            ids = ids.append(script.keyword_index(word))
        for: 1 <= offset < int(#ids)
            current := ids[offset]
            cursor := offset
            while cursor > 0 and criterion(ids[cursor - 1], usage) < criterion(current, usage)
                ids[cursor] = ids[cursor - 1]
                cursor -= 1
            ids[cursor] = current
        return ids
    string[] .command(string line)
        words := line.split(" ")
        name := words[0]
        arguments := words[1:<#words]
        text := arguments.join(" ")
        no_arguments := [":keys", ":why", ":history", ":memory", ":forget", ":reset", ":unused", ":stats", ":undo"]
        text_commands := [":prepare", ":reflect", ":say", ":explain"]
        single := [":groups", ":find", ":script", ":cycles", ":coverage"]
        string[] known := no_arguments.concat(text_commands).concat(single).concat([":trace", ":test"])
        if not loader.has_word(known, name)
            return ["unknown command {name}"]
        count := int(#arguments)
        if (loader.has_word(no_arguments, name) and count != 0) or (loader.has_word(text_commands, name) and text == "") or (loader.has_word(single, name) and count != 1)
            return ["bad arguments"]
        if name == ":trace" and (count != 1 or not loader.has_word(["on", "off"], arguments[0]))
            return ["bad arguments"]
        if name == ":test" and count < 2
            return ["bad arguments"]
        keyword_id := -1
        if loader.has_word([":script", ":cycles", ":coverage", ":test"], name)
            keyword_id = script.keyword_index(arguments[0])
            if keyword_id < 0
                return ["unknown keyword {arguments[0]}"]
        if name == ":undo"
            if not undo_available
                return ["nothing to undo"]
            state = saved
            undo_available = false
            return ["undone turn {state.session.turns + 1}"]
        if is_quit(text, script)
            return ["would quit"]
        State before := state
        Reply answer := .respond(text)
        state = before
        return if name == ":say" then [answer.text] else answer.trace
    void .run_command(string line)
        transcript = transcript.append(line)
        string[] result := .command(line)
        for output := result[]
            transcript = transcript.append("  " + output)
    string[] keyword_stats()
        result := ["keyword,rank,uses,replies,decomposition_hits"]
        for id := ordered_keywords(true)[]
            loader.Keyword keyword := script.keywords[id]
            int[] hits := []
            for rule_id := keyword.decomps[]
                hits = hits.append(state.hits[rule_id])
            counts := if #hits == 0 then "-" else number_list(hits, "/")
            result = result.append("{keyword.word},{keyword.rank},{state.keyword_uses[id]},{state.keyword_replies[id]},{counts}")
        return result
    void .finish()
        transcript = transcript.concat(["", "SUMMARY sessions={sessions} " + source_line(total_turns, total_sources)])
