    public function cmp(int|self $x): int
    {
        $x = self::of($x);
        $g = gcd($this->d, $x->d);
        return ($this->n * intdiv($x->d, $g)) <=> ($x->n * intdiv($this->d, $g));
    }
function lexCompare(array $a, array $b): int
{
    for ($i = 0; $i < min(count($a), count($b)); ++$i)
        if ($a[$i] !== $b[$i])
            return $a[$i] <=> $b[$i];
    return count($a) <=> count($b);
}
