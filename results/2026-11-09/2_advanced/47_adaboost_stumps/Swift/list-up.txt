func trainDiscrete(_ rows: [Row], _ maxRounds: Int, weights initialWeights: [Double]? = nil, shrinkage: Double = 1, space suppliedSpace: SearchSpace? = nil) -> Model {
    let space = suppliedSpace ?? SearchSpace(rows), labels = rows.map { $0.label }
    var weights = initialWeights ?? Array(repeating: 1 / Double(rows.count), count: rows.count)
    var rounds: [BoostRound] = [], bound = 1.0, stop = "completed"
    for _ in 0..<maxRounds {
        let candidate = discreteSearch(space, labels, weights), e = candidate.error
        if e >= 0.5 { stop = "weak"; break }
        let alpha = e <= 1e-10 ? 1 : shrinkage * (0.5 * log((1 - e) / e))
        let stump = Stump(feature: candidate.stump.feature, thresholds: candidate.stump.thresholds, values: [-Double(candidate.polarity) * alpha, Double(candidate.polarity) * alpha])
        let updated = rows.indices.map { i -> Double in
            if weights[i] == 0 { return 0 }
            let decision = rows[i].values[stump.feature] > stump.thresholds[0] ? candidate.polarity : -candidate.polarity
            return weights[i] * (decision == rows[i].label ? exp(-alpha) : exp(alpha))
        }
        let normalizer = orderedSum(updated)
        weights = updated.map { $0 / normalizer }
        rounds.append(BoostRound(stump: stump, weights: before, error: e, alpha: alpha, polarity: candidate.polarity, z: normalizer, bound: bound))
        if e <= 1e-10 { stop = "perfect"; break }
    }
    return Model(rounds: rounds, stop: stop)
}

func leftCumulative(_ groups: [[Int]], _ values: [Double]) -> [Double] {
    var total = 0.0, result = [0.0]
    for group in groups.dropLast() {
        for index in group { total = total + values[index] }
        result.append(total)
    }
    return result
}
func discreteSearch(_ space: SearchSpace, _ labels: [Int], _ weights: [Double]) -> StumpCandidate {
    let positives = labels.indices.map { labels[$0] == 1 ? weights[$0] : 0 }, negatives = labels.indices.map { labels[$0] == -1 ? weights[$0] : 0 }
    let positiveTotal = orderedSum(positives), negativeTotal = orderedSum(negatives)
    var candidates: [StumpCandidate] = []
    for feature in space.thresholds.indices {
        let leftPositive = leftCumulative(space.groups[feature], positives), leftNegative = leftCumulative(space.groups[feature], negatives)
        for (index, threshold) in space.thresholds[feature].enumerated() {
            candidates.append(StumpCandidate(stump: Stump(feature: feature, thresholds: [threshold], values: [-1, 1]), error: leftPositive[index] + (negativeTotal - leftNegative[index]), polarity: 1))
            candidates.append(StumpCandidate(stump: Stump(feature: feature, thresholds: [threshold], values: [1, -1]), error: leftNegative[index] + (positiveTotal - leftPositive[index]), polarity: -1))
        }
    }
    return choose(candidates) { $0.error }
}
func prediction(_ score: Double) -> Int { score > 0 ? 1 : -1 }
func rowScore(_ rounds: [BoostRound], _ values: [Double]) -> Double { orderedSum(rounds.map { $0.stump.output(values) }) }
func scoreRows(_ rounds: [BoostRound], _ rows: [Row]) -> [Double] { rows.map { rowScore(rounds, $0.values) } }
func addRoundScores(_ round: BoostRound, _ rows: [Row], _ scores: inout [Double]) {
    for i in rows.indices { scores[i] = scores[i] + round.stump.output(rows[i].values) }
}
func countErrors(_ rows: [Row], _ scores: [Double]) -> Int { rows.indices.filter { prediction(scores[$0]) != rows[$0].label }.count }
func trainingCurve(_ rounds: [BoostRound], _ training: [Row], _ testing: [Row], _ variant: String) -> [CurvePoint] {
    var trainScores = Array(repeating: 0.0, count: training.count), testScores = Array(repeating: 0.0, count: testing.count)
    var scale = 0.0, curve: [CurvePoint] = []
    for (index, round) in rounds.enumerated() {
        addRoundScores(round, training, &trainScores)
        addRoundScores(round, testing, &testScores)
        scale = scale + magnitude(round)
        let minimum = training.indices.map { Double(training[$0].label) * trainScores[$0] }.min()!
        curve.append(CurvePoint(round: index + 1, trainErrors: countErrors(training, trainScores), testErrors: countErrors(testing, testScores),
                                trainLoss: loss(training, trainScores, variant), testLoss: loss(testing, testScores, variant), margin: minimum / scale))
    }
    return curve
}
