static double sum(const Vector &values) {
    double out = 0;
    for (double value : values)
        out += value;
    return out;
}
Model discrete(const Rows &rows, const Params &p, int maximum, double shrinkage, Vector weights) {
    if (weights.empty())
        weights = Vector(rows.size(), 1.0 / rows.size());
    const Search search = prepare(rows, 0);
    Model out;
    out.variant = "discrete";
    double bound = 1;
    for (int t = 0; t < maximum; ++t) {
        const Candidate candidate = discrete_stump(rows, search, weights);
        const double e = candidate.error;
        if (e >= 0.5) {
            out.stop = "weak";
            break;
        }
        const double alpha = e <= 0.0000000001 ? 1 : shrinkage * (0.5 * std::log((1 - e) / e));
        Round round;
        round.stump = candidate.stump;
        for (double &value : round.stump.values)
            value *= alpha;
        for (std::size_t i = 0; i < rows.size(); ++i) {
            const int decision = candidate.stump.output(rows[i].values);
            weights[i] = weights[i] * std::exp(decision == rows[i].label ? -alpha : alpha);
        }
        round.z = sum(weights);
        for (double &weight : weights)
            weight /= round.z;
        out.rounds.push_back(std::move(round));
        if (e <= 0.0000000001) {
            out.stop = "perfect";
            break;
        }
    }
    return out;
}
Search prepare(const Rows &rows, int bins) {
    Search out;
    for (std::size_t j = 0; j < rows[0].values.size(); ++j) {
        Feature feature;
        feature.order.resize(rows.size());
        std::iota(feature.order.begin(), feature.order.end(), 0);
        std::stable_sort(feature.order.begin(), feature.order.end(),
                         [&](int a, int b) { return rows[a].values[j] < rows[b].values[j]; });
        feature.thresholds.push_back(rows[feature.order[0]].values[j] - 1);
        feature.ends.push_back(0);
        for (std::size_t i = 1; i < feature.order.size(); ++i) {
            const double left = rows[feature.order[i - 1]].values[j],
                         right = rows[feature.order[i]].values[j];
            if (left != right) {
                feature.thresholds.push_back((left + right) / 2);
                feature.ends.push_back(i);
            }
        }
        out.features.push_back(std::move(feature));
    }
    return out;
}
Candidate discrete_stump(const Rows &rows, const Search &search, const Vector &weights) {
    double positive = 0, negative = 0;
    for (std::size_t i = 0; i < rows.size(); ++i)
        if (rows[i].label == 1)
            positive += weights[i];
        else
            negative += weights[i];
    std::vector<Candidate> candidates;
    for (std::size_t j = 0; j < search.features.size(); ++j) {
        const Feature &feature = search.features[j];
        int position = 0;
        double lp = 0, ln = 0;
        for (std::size_t k = 0; k < feature.thresholds.size(); ++k) {
            while (position < feature.ends[k]) {
                const int i = feature.order[position++];
                if (rows[i].label == 1)
                    lp += weights[i];
                else
                    ln += weights[i];
            }
            for (int polarity : Ints{1, -1})
                candidates.push_back(
                    {{int(j), {feature.thresholds[k]}, {double(-polarity), double(polarity)}},
                     polarity == 1 ? lp + (negative - ln) : ln + (positive - lp),
                     polarity});
        }
    }
    return choose(candidates);
}
int prediction(double score) {
    return score > 0 ? 1 : -1;
}
static Vector scores(const Model &model, const Rows &rows, int count = -1) {
    if (count < 0)
        count = model.rounds.size();
    count = std::min(count, int(model.rounds.size()));
    Vector out(rows.size());
    for (std::size_t i = 0; i < rows.size(); ++i)
        for (int t = 0; t < count; ++t)
            out[i] += model.rounds[t].stump.output(rows[i].values);
    return out;
}
static Curve curve(const Model &model, const Rows &train, const Rows &test) {
    Curve out;
    for (std::size_t t = 1; t <= model.rounds.size(); ++t) {
        const Vector a = scores(model, train, t), b = scores(model, test, t);
        out.train.push_back(error_count(train, a));
        out.test.push_back(error_count(test, b));
    }
    return out;
}
