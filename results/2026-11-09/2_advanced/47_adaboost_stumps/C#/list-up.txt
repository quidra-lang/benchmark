    public record Round(Stumps.Stump stump, int polarity, double error, double alpha, double z, double bound,
                        double loss, double[] weights) {
    }
    public sealed class Model {
        public List<Round> rounds = new();
        public string stop = "completed";
        public double[][] cuts;
    }
    public static Model Discrete(List<Dataset.Row> rows, double[] initial, int maximum, double shrinkage) {
        Stumps.Search search = new Stumps.Search(rows);
        double[] weights = initial.ToArray();
        for (int t = 0; t < maximum; t++) {
            Stumps.Candidate candidate = Stumps.Discrete(search, weights);
            double e = candidate.error;
            if (e >= 0.5) {
                model.stop = "weak";
                break;
            }
            double alpha = e <= 0.0000000001 ? 1 : shrinkage * (0.5 * Math.Log((1 - e) / e));
            Stumps.Stump stump =
                new Stumps.Stump(candidate.stump.feature, candidate.stump.thresholds,
                                 new double[] { -candidate.polarity * alpha, candidate.polarity * alpha });
            for (int i = 0; i < weights.Length; i++) {
                int decision =
                    rows[i].x[stump.feature] > stump.thresholds[0] ? candidate.polarity : -candidate.polarity;
                weights[i] *= Math.Exp(decision == rows[i].label ? -alpha : alpha);
                z += weights[i];
            }
            for (int i = 0; i < weights.Length; i++)
                weights[i] /= z;
            model.rounds.Add(new Round(stump, candidate.polarity, e, alpha, z, bound, double.NaN, before));
            if (e <= 0.0000000001) {
                model.stop = "perfect";
                break;
            }
        }
        return model;
    }
        public Search(List<Dataset.Row> rows) {
            this.rows = rows;
            int d = rows[0].x.Length, n = rows.Count;
            sorted = Enumerable.Range(0, d)
                         .Select(
                             _ => new int[n])
                         .ToArray();
            ends = new int [d][];
            thresholds = new double [d][];
            for (int j = 0; j < d; j++) {
                int feature = j;
                List<int> order = new();
                for (int i = 0; i < n; i++)
                    order.Add(i);
                order.Sort((a, b) => {
                    int comparison = rows[a].x[feature].CompareTo(rows[b].x[feature]);
                    return comparison != 0 ? comparison : a.CompareTo(b);
                });
                List<double> distinct = new();
                List<int> starts = new();
                for (int i = 0; i < n; i++) {
                    sorted[j][i] = order[i];
                    double value = rows[order[i]].x[j];
                    if (i == 0 || value != rows[order[i - 1]].x[j]) {
                        distinct.Add(value);
                        starts.Add(i);
                    }
                }
                int q = distinct.Count;
                ends[j] = new int[q];
                thresholds[j] = new double[q];
                for (int c = 0; c < q; c++) {
                    ends[j][c] = starts[c];
                    thresholds[j][c] = c == 0 ? distinct[0] - 1 : (distinct[c - 1] + distinct[c]) / 2;
                }
            }
        }
    public static Candidate Discrete(Search search, double[] weights) {
        double plus = 0, minus = 0;
        for (int i = 0; i < weights.Length; i++)
            if (search.rows[i].label == 1)
                plus += weights[i];
            else
                minus += weights[i];
        List<Candidate> candidates = new();
        for (int j = 0; j < search.sorted.Length; j++) {
            int position = 0;
            double leftPlus = 0, leftMinus = 0;
            for (int c = 0; c < search.thresholds[j].Length; c++) {
                while (position < search.ends[j][c]) {
                    int i = search.sorted[j][position++];
                    if (search.rows[i].label == 1)
                        leftPlus += weights[i];
                    else
                        leftMinus += weights[i];
                }
                double threshold = search.thresholds[j][c];
                candidates.Add(new Candidate(new Stump(j, new double[] { threshold }, new double[] { -1, 1 }),
                                             leftPlus + (minus - leftMinus), 1));
                candidates.Add(new Candidate(new Stump(j, new double[] { threshold }, new double[] { 1, -1 }),
                                             leftMinus + (plus - leftPlus), -1));
            }
        }
        return Choose(candidates);
    }
    public static int Prediction(double score) {
        return score > 0 ? 1 : -1;
    }
    public static int Earliest(int[] values) {
        int index = 0;
        for (int i = 1; i < values.Length; i++)
            if (values[i] < values[index])
                index = i;
        return index;
    }
    public static double Score(Adaboost.Model model, double[] x, int limit) {
        double score = 0;
        for (int t = 0; t < Math.Min(limit, model.rounds.Count); t++)
            score += model.rounds[t].stump.Output(x);
        return score;
    }
    public static double[] Scores(Adaboost.Model model, List<Dataset.Row> rows, int limit) {
        double[] scores = new double[rows.Count];
        for (int i = 0; i < scores.Length; i++)
            scores[i] = Score(model, rows[i].x, limit);
        return scores;
    }
    public static Adaboost.Model Prefix(Adaboost.Model model, int count) {
        Adaboost.Model result = new Adaboost.Model();
        result.rounds.AddRange(model.rounds.GetRange(0, Math.Min(count, model.rounds.Count) - (0)));
        result.stop = model.stop;
        result.cuts = model.cuts;
        return result;
    }
    public sealed class Curve {
        public int[] trainErrors, testErrors;
        public double[] trainLoss, testLoss, margin;
        public Curve(Adaboost.Model model, List<Dataset.Row> training, List<Dataset.Row> testing,
                     bool logit) {
            int r = model.rounds.Count;
            trainErrors = new int[r];
            testErrors = new int[r];
            trainLoss = new double[r];
            testLoss = new double[r];
            margin = new double[r];
            for (int t = 0; t < r; t++) {
                double[] a = Scores(model, training, t + 1), b = Scores(model, testing, t + 1);
                trainErrors[t] = Errors(training, a);
                testErrors[t] = Errors(testing, b);
            }
        }
    }
    public static void CrossValidate() {
        int count = settings.Integer("FOLDS");
        cvErrors = new int[maximum];
        for (int fold = 0; fold < count; fold++) {
            int validation = 0;
            for (int i = 0; i < n; i++)
                if (i % count == fold)
                    validation++;
            double[] w = new double[n];
            for (int i = 0; i < n; i++)
                if (i % count != fold)
                    w[i] = 1.0 / (n - validation);
            Adaboost.Model model = Adaboost.Discrete(train, w, maximum, 1);
            folds.Add(model);
            for (int t = 1; t <= maximum; t++)
                for (int i = 0; i < n; i++)
                    if (i % count == fold &&
                        Evaluation.Prediction(Score(model, train[i].x, t)) != train[i].label)
                        cvErrors[t - 1]++;
        }
        selected = Earliest(cvErrors) + 1;
        crossScores = new double[n];
        for (int i = 0; i < n; i++)
            crossScores[i] = Score(folds[i % count], train[i].x, selected);
    }
    public static void TrainModels() {
        trainScores = new double [variants.Count][];
        testScores = new double [variants.Count][];
        for (int v = 0; v < variants.Count; v++) {
            string variant = variants[v];
            Adaboost.Model training = Variants.Train(variant, train, settings),
                           model = variant.Equals("discrete") ? Prefix(training, selected) : training;
            full.Add(training);
            models.Add(model);
            trainScores[v] = Scores(model, train, maximum);
            testScores[v] = Scores(model, test, maximum);
            curves.Add(new Curve(training, train, test, variant.Equals("logit")));
        }
        for (int v = 0; v < variants.Count; v++) {
            string variant = variants[v];
            Adaboost.Model model = models[v];
            rounds += model.rounds.Count;
            output.Add(Metrics("train", variant, model.rounds.Count, train, trainScores[v],
                               variant.Equals("logit")));
            output.Add(
                Metrics("test", variant, model.rounds.Count, test, testScores[v], variant.Equals("logit")));
