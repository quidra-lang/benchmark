export function normalize(weights: number[]): number {
  let total = 0;
  for (const w of weights) total += w;
  for (let i = 0; i < weights.length; i++) weights[i] /= total;
  return total;
}
export function discrete(
  rows: BinaryRow[],
  maximumRounds: number,
  shrinkage = 1,
  initialWeights: number[] | null = null,
): Training {
  const plans = prepare(rows);
  const weights =
    initialWeights === null
      ? Array<number>(rows.length).fill(1 / rows.length)
      : initialWeights.slice();
  const rounds: DiscreteRound[] = [];
  let bound = 1;
  let stop: Stop = 'completed';
  for (let t = 0; t < maximumRounds; t++) {
    const candidate = discreteSearch(rows, weights, plans);
    const error = candidate.error;
    if (error >= 0.5) {
      stop = 'weak';
      break;
    }
    const alpha = error <= 1e-10 ? 1 : shrinkage * (0.5 * Math.log((1 - error) / error));
    const polarity = candidate.polarity;
    const stump = new Stump(
      candidate.feature,
      [candidate.threshold],
      [-polarity * alpha, polarity * alpha],
    );
    const before = weights.slice();
    rows.forEach((row, i) => {
      const decision = row.values[candidate.feature] > candidate.threshold ? polarity : -polarity;
      weights[i] *= decision === row.label ? Math.exp(-alpha) : Math.exp(alpha);
    });
    const z = normalize(weights);
    bound *= z;
    rounds.push({ kind: 'discrete', stump, polarity, error, alpha, z, bound, weights: before });
    if (error <= 1e-10) {
      stop = 'perfect';
      break;
    }
  }
  return { rounds, stop };
}
export function prepare(rows: ValuesRow[]) {
  return rows[0].values.map((_, feature) => {
    const order = rows
      .map((_, i) => i)
      .sort((a, b) => rows[a].values[feature] - rows[b].values[feature] || a - b);
    const groups: number[][] = [];
    const distinct: number[] = [];
    for (const i of order) {
      const value = rows[i].values[feature];
      if (!distinct.length || distinct[distinct.length - 1] !== value) {
        distinct.push(value);
        groups.push([]);
      }
      groups[groups.length - 1].push(i);
    }
    const thresholds = [distinct[0] - 1];
    for (let i = 1; i < distinct.length; i++) thresholds.push((distinct[i - 1] + distinct[i]) / 2);
    return { feature, order, groups, thresholds };
  });
}
export function discreteSearch(rows: BinaryRow[], weights: number[], plans: Plan[]) {
  let positive = 0;
  let negative = 0;
  rows.forEach((row, i) => {
    positive += row.label === 1 ? weights[i] : 0;
    negative += row.label === -1 ? weights[i] : 0;
  });
  const candidates: { feature: number; threshold: number; polarity: number; error: number }[] = [];
  for (const plan of plans) {
    let lp = 0;
    let ln = 0;
    for (let k = 0; k < plan.thresholds.length; k++) {
      if (k > 0) {
        for (const i of plan.groups[k - 1]) {
          lp += rows[i].label === 1 ? weights[i] : 0;
          ln += rows[i].label === -1 ? weights[i] : 0;
        }
      }
      candidates.push(
        {
          feature: plan.feature,
          threshold: plan.thresholds[k],
          polarity: 1,
          error: lp + (negative - ln),
        },
        {
          feature: plan.feature,
          threshold: plan.thresholds[k],
          polarity: -1,
          error: ln + (positive - lp),
        },
      );
    }
  }
  return choose(candidates);
}
export const prediction = (score: number): number => (score > 0 ? 1 : -1);
function minimumPosition(values: number[]): number {
  let best = 0;
  for (let i = 1; i < values.length; i++) if (values[i] < values[best]) best = i;
  return best;
}
function curve(rounds: Round[], train: BinaryRow[], test: BinaryRow[], variant: Variant) {
  const trainScores = train.map(() => 0);
  const testScores = test.map(() => 0);
  let scale = 0;
  return rounds.map((round) => {
    train.forEach((row, i) => {
      trainScores[i] += round.stump.output(row.values);
    });
    test.forEach((row, i) => {
      testScores[i] += round.stump.output(row.values);
    });
    scale += magnitude(round);
    return {
      trainErrors: errorCount(train, trainScores),
      testErrors: errorCount(test, testScores),
      trainLoss: loss(train, trainScores, variant),
      testLoss: loss(test, testScores, variant),
      minimum: Math.min(...train.map((row, i) => row.label * trainScores[i])) / scale,
    };
  });
}
function crossValidation(rows: BinaryRow[], params: Parameters) {
  const folds = Array.from({ length: params.folds }, (_, fold) => {
    const validation = rows.map((_, i) => i).filter((i) => i % params.folds === fold);
    const count = rows.length - validation.length;
    const model = discrete(
      rows,
      params.rounds,
      1,
      rows.map((_, i) => (i % params.folds === fold ? 0 : 1 / count)),
    );
    const scores = validation.map(() => 0);
    const errors: number[] = [];
    for (let t = 0; t < params.rounds; t++) {
      if (t < model.rounds.length) {
        validation.forEach((row, i) => {
          scores[i] += model.rounds[t].stump.output(rows[row].values);
        });
      }
      errors.push(validation.filter((row, i) => prediction(scores[i]) !== rows[row].label).length);
    }
    return { model, validation, count, errors };
  });
  const errors = Array<number>(params.rounds).fill(0);
  for (const fold of folds) for (let t = 0; t < params.rounds; t++) errors[t] += fold.errors[t];
  const selected = minimumPosition(errors) + 1;
  const scores = rows.map(() => 0);
  for (const fold of folds) {
    for (const i of fold.validation) {
      scores[i] = score(fold.model.rounds.slice(0, selected), rows[i].values);
    }
  }
  return { folds, errors, selected, scores };
}
function analyzeModels(data: Data, cv: CV) {
  return data.params.variants.map((name) => {
    const full = trainVariant(name, data.train, data.params);
    const rounds = name === 'discrete' ? full.rounds.slice(0, cv.selected) : full.rounds.slice();
    const scale = sum(rounds.map(magnitude));
    const trainScores = scoreRows(rounds, data.train);
    const testScores = scoreRows(rounds, data.test);
    return {
      name,
      full,
      rounds,
      scale,
      trainScores,
      testScores,
      trainMargins: data.train.map((r, i) => (r.label * trainScores[i]) / scale),
      testMargins: data.test.map((r, i) => (r.label * testScores[i]) / scale),
      trainPredictions: trainScores.map(prediction),
      testPredictions: testScores.map(prediction),
      curve: curve(full.rounds, data.train, data.test, name),
      importance: importance(rounds, data.features.length),
    };
  });
