real64 normalize(real64[] &weights)
    real64 sum := 0.0
    for value := weights[]
        sum += value
    for: i < int(#weights)
        weights[i] /= sum
    return sum

class Round(int feature, real64[] thresholds, real64[] values, int polarity, real64 error_value, real64 alpha, real64 z, real64 bound, real64[] weights, real64 loss)
class Model(Round[] rounds, string reason, real64[][] cuts)
Model discrete(real64[][] rows, int[] labels, int maximum, real64 shrinkage, real64[] initial_weights := [])
    weights := initial_weights
    if #weights == 0
        weights = array(#rows, fill = 1.0 / stumps.as_real(int(#rows)))
    stumps.Prepared[] prepared := stumps.prepare(rows)
    Round[] rounds := []
    real64 bound := 1.0
    for _ < maximum
        stumps.Stump candidate := stumps.discrete(rows, labels, weights, prepared)
        real64 error_value := candidate.error_value
        if error_value >= 0.5
            return Model(rounds, "weak", [])
        alpha := if error_value <= 0.0000000001 then 1.0 else shrinkage * (0.5 * logarithm((1.0 - error_value) / error_value))
        real64 polarity := stumps.as_real(candidate.polarity)
        outputs := [-polarity * alpha, polarity * alpha]
        for: i < int(#rows)
            int decision := if rows[i][candidate.feature] > candidate.thresholds[0] then candidate.polarity else -candidate.polarity
            weights[i] *= exponential(if decision == labels[i] then -alpha else alpha)
        z := normalize(&weights)
        rounds = rounds.append(Round(candidate.feature, candidate.thresholds, outputs, candidate.polarity, error_value, alpha, z, bound, previous, 0.0))
        if error_value <= 0.0000000001
            return Model(rounds, "perfect", [])
    return Model(rounds, "completed", [])
Prepared[] prepare(real64[][] rows)
    Prepared[] result := []
    for: j < int(#rows[0])
        int[] order := []
        for: i < int(#rows)
            place := int(#order)
            for: k < int(#order)
                if rows[i][j] < rows[order[k]][j]
                    place = k
                    break
            order = order[0:<place].append(i).concat(order[place:<int(#order)])
        real64[] unique_threshold_values := []
        int[][] batches := []
        for i := order[]
            value := rows[i][j]
            if #unique_threshold_values == 0 or value != unique_threshold_values[-1]
                unique_threshold_values = unique_threshold_values.append(value)
                batches = batches.append([])
            last := int(#batches) - 1
            batches[last] = batches[last].append(i)
        thresholds := [unique_threshold_values[0] - 1.0]
        for: i < int(#unique_threshold_values) - 1
            thresholds = thresholds.append((unique_threshold_values[i] + unique_threshold_values[i + 1]) / 2.0)
        result = result.append(Prepared(order, unique_threshold_values, batches, thresholds))
    return result
Stump discrete(real64[][] rows, int[] labels, real64[] weights, Prepared[] prepared)
    real64 positive := 0.0
    real64 negative := 0.0
    for: i < int(#rows)
        if labels[i] == 1
            positive += weights[i]
        else
            negative += weights[i]
    Stump[] candidates := []
    for: j < int(#prepared)
        feature := prepared[j]
        real64 left_positive := 0.0
        real64 left_negative := 0.0
        for: k < int(#feature.thresholds)
            if k > 0
                for i := feature.batches[k - 1][]
                    if labels[i] == 1
                        left_positive += weights[i]
                    else
                        left_negative += weights[i]
            threshold := feature.thresholds[k]
            candidates = candidates.append(Stump(j, [threshold], [], 1, left_positive + (negative - left_negative), [], []))
            candidates = candidates.append(Stump(j, [threshold], [], -1, left_negative + (positive - left_positive), [], []))
    return choose(candidates)
real64 row_score(boost.adaboost.Round[] rounds, real64[] values, limit := -1)
    count := if limit < 0 or limit > int(#rounds) then int(#rounds) else limit
    real64 result := 0.0
    for: t < count
        record := rounds[t]
        result += boost.adaboost.stumps.output(record.feature, record.thresholds, record.values, values)
    return result

real64[] calc_scores(boost.adaboost.Round[] rounds, real64[][] rows, limit := -1)
    return [for row := rows[] then row_score(rounds, row, limit)]
CrossValidation cross_validation(real64[][] rows, int[] labels, boost.dataset.Params params)
    Fold[] folds := []
    errors := array(nat(params.rounds), fill = 0)
    for: int fold < params.folds
        int[] validation := []
        for: i < int(#rows)
            if i % params.folds == fold
                validation = validation.append(i)
        train_count := int(#rows) - int(#validation)
        weights := [for i := (0:<int(#rows))[] then if i % params.folds == fold then 0.0 else 1.0 / evaluation.as_real(train_count)]
        boost.adaboost.Model model := boost.adaboost.discrete(rows, labels, params.rounds, 1.0, weights)
        int[] fold_errors := []
        for: 1 <= int t <= params.rounds
            count := 0
            for i := validation[]
                if evaluation.sign(row_score(model.rounds, rows[i], t)) != labels[i]
                    count += 1
            fold_errors = fold_errors.append(count)
            errors[t - 1] += count
        folds = folds.append(Fold(model.rounds, model.reason, train_count, validation, fold_errors))
    selected := 0
    for: 1 <= int t < params.rounds
        if errors[t] < errors[selected]
            selected = t
    real64[] out_of_fold := []
    for: i < int(#rows)
        out_of_fold = out_of_fold.append(row_score(folds[i % params.folds].rounds, rows[i], selected + 1))
    return CrossValidation(folds, errors, selected + 1, out_of_fold)
Curve[] calc_curve(boost.adaboost.Round[] rounds, real64[][] training, int[] train_labels, real64[][] testing, int[] test_labels, bool logit)
    real64[] train_scores := array(#training, fill = 0.0)
    real64[] test_scores := array(#testing, fill = 0.0)
    real64 scale := 0.0
    Curve[] result := []
    for record := rounds[]
        for: i < int(#training)
            train_scores[i] += boost.adaboost.stumps.output(record.feature, record.thresholds, record.values, training[i])
        for: i < int(#testing)
            test_scores[i] += boost.adaboost.stumps.output(record.feature, record.thresholds, record.values, testing[i])
        result = result.append(Curve(count_errors(train_labels, train_scores), count_errors(test_labels, test_scores), calc_loss(train_labels, train_scores, logit), calc_loss(test_labels, test_scores, logit), minimum / scale))
    return result
