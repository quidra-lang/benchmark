pub fn discrete(rows: []const c.Row, p: c.Params, maximum: usize, shrinkage: f64, initial: ?[]const f64) Model {
    _ = p;
    const weights = if (initial) |values| c.dupe(f64, values) else c.vector(rows.len);
    if (initial == null) @memset(weights, 1 / c.f(rows.len));
    const search = stumps.prepare(rows, 0);
    var rounds: std.ArrayList(Round) = .empty;
    var reason: []const u8 = "completed";
    var bound: f64 = 1;
    for (0..maximum) |_| {
        const candidate = stumps.discrete(rows, search, weights);
        const e = candidate.error_;
        if (e >= 0.5) {
            reason = "weak";
            break;
        }
        const alpha = if (e <= 0.0000000001) @as(f64, 1) else shrinkage * (0.5 * @log((1 - e) / e));
        const values = c.dupe(f64, candidate.stump.values);
        for (values) |*value| value.* = value.* * alpha;
        var round: Round = .{ .stump = .{ .feature = candidate.stump.feature, .thresholds = candidate.stump.thresholds,
             .values = values }, .weights = c.dupe(f64, weights), .error_ = e, .alpha = alpha, .polarity = candidate.polarity };
        for (rows, weights) |row, *weight| {
            const decision = candidate.stump.output(row.values);
            weight.* = weight.* * @exp(if (decision == c.f(row.label)) -alpha else alpha);
        }
        round.z = sum(weights);
        for (weights) |*weight| weight.* = weight.* / round.z;
        c.append(Round, &rounds, round);
        if (e <= 0.0000000001) {
            reason = "perfect";
            break;
        }
    }
    return .{ .variant = "discrete", .stop = reason, .rounds = rounds.items };
}
fn rowLess(context: SortContext, a: usize, b: usize) bool {
    const x = context.rows[a].values[context.feature];
    const y = context.rows[b].values[context.feature];
    return if (x != y) x < y else a < b;
}
pub fn prepare(rows: []const c.Row, bins: usize) Search {
    const features = c.alloc(Feature, rows[0].values.len);
    for (features, 0..) |*feature, j| {
        const order = c.alloc(usize, rows.len);
        for (order, 0..) |*value, i| value.* = i;
        std.mem.sort(usize, order, SortContext{ .rows = rows, .feature = j }, rowLess);
        var thresholds: std.ArrayList(f64) = .empty;
        var ends: std.ArrayList(usize) = .empty;
        c.append(f64, &thresholds, rows[order[0]].values[j] - 1);
        c.append(usize, &ends, 0);
        for (1..rows.len) |i| {
            const left = rows[order[i - 1]].values[j];
            const right = rows[order[i]].values[j];
            if (left != right) {
                c.append(f64, &thresholds, (left + right) / 2);
                c.append(usize, &ends, i);
            }
        }
        feature.* = .{ .order = order, .ends = ends.items, .thresholds = thresholds.items, .cuts = cuts };
    }
    return .{ .features = features };
}
pub fn discrete(rows: []const c.Row, search: Search, weights: []const f64) Result {
    var positive: f64 = 0;
    var negative: f64 = 0;
    for (rows, weights) |row, weight| if (row.label == 1) {
        positive += weight;
    } else {
        negative += weight;
    };
    var candidates: [8000]Candidate = undefined;
    var count: usize = 0;
    for (search.features, 0..) |feature, j| {
        var position: usize = 0;
        var lp: f64 = 0;
        var ln: f64 = 0;
        for (feature.thresholds, feature.ends) |threshold, end| {
            while (position < end) {
                const i = feature.order[position];
                position += 1;
                if (rows[i].label == 1) lp += weights[i] else ln += weights[i];
            }
            for ([_]i64{ 1, -1 }) |polarity| {
                candidates[count] = .{ .feature = j, .threshold = threshold, .left = -c.f(polarity),
                     .right = c.f(polarity), .error_ = if (polarity == 1) lp + (negative - ln) else ln + (positive - lp),
                     .polarity = polarity };
                count += 1;
            }
        }
    }
    return choose(candidates[0..count]);
}
pub fn prediction(score: f64) i64 {
    return if (score > 0) 1 else -1;
}
fn scores(model: adaboost.Model, rows: []const c.Row, limit: ?usize) c.Vector {
    const count = @min(limit orelse model.rounds.len, model.rounds.len);
    const out = c.vector(rows.len);
    for (rows, 0..) |row, i| for (model.rounds[0..count]) |round| {
        out[i] += round.stump.output(row.values);
    };
    return out;
}
fn curve(model: adaboost.Model, train: []const c.Row, test_rows: []const c.Row) Curve {
    const count = model.rounds.len;
    const out: Curve = .{ .train = c.alloc(usize, count), .test_rows = c.alloc(usize, count), .train_loss = c.vector(count),
         .test_loss = c.vector(count), .minimum = c.vector(count) };
    for (1..count + 1) |t| {
        const a = scores(model, train, t);
        const b = scores(model, test_rows, t);
        out.train[t - 1] = errorCount(train, a);
        out.test_rows[t - 1] = errorCount(test_rows, b);
    }
    return out;
}
