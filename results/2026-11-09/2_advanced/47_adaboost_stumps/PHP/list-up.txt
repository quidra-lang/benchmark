function normalize(&$weights)
{
    $total = 0.0;
    foreach ($weights as $weight) $total += $weight;
    foreach ($weights as &$weight) $weight /= $total;
    unset($weight);
    return $total;
}
function discrete($rows, $maximumRounds, $shrinkage = 1, $initialWeights = null)
{
    $plans = prepare($rows);
    $weights = $initialWeights ?? array_fill(0, count($rows), 1 / count($rows));
    $rounds = [];
    $bound = 1.0;
    $stop = 'completed';
    for ($t = 0; $t < $maximumRounds; $t++) {
        $candidate = discrete_search($rows, $weights, $plans);
        $error = $candidate['error'];
        if ($error >= 0.5) { $stop = 'weak'; break; }
        $alpha = $error <= 1e-10 ? 1 : $shrinkage * (0.5 * log((1 - $error) / $error));
        $polarity = $candidate['polarity'];
        $stump = new Stump($candidate['feature'], [$candidate['threshold']], [-$polarity * $alpha, $polarity * $alpha]);
        $before = $weights;
        foreach ($rows as $i => $row) {
            $decision = $row['values'][$candidate['feature']] > $candidate['threshold'] ? $polarity : -$polarity;
            $weights[$i] *= $decision === $row['label'] ? exp(-$alpha) : exp($alpha);
        }
        $z = normalize($weights);
        $bound *= $z;
        $rounds[] = ['stump' => $stump, 'polarity' => $polarity, 'error' => $error, 'alpha' => $alpha, 'z' => $z, 'bound' => $bound, 'weights' => $before];
        if ($error <= 1e-10) { $stop = 'perfect'; break; }
    }
    return compact('rounds', 'stop');
}
function prepare($rows)
{
    $plans = [];
    foreach ($rows[0]['values'] as $feature => $value) {
        $order = range(0, count($rows) - 1);
        usort($order, fn($a, $b) => ($rows[$a]['values'][$feature] <=> $rows[$b]['values'][$feature]) ?: ($a <=> $b));
        $groups = [];
        $distinct = [];
        foreach ($order as $i) {
            $current = $rows[$i]['values'][$feature];
            if (count($distinct) === 0 || $distinct[count($distinct) - 1] != $current) {
                $distinct[] = $current;
                $groups[] = [];
            }
            $groups[count($groups) - 1][] = $i;
        }
        $thresholds = [$distinct[0] - 1];
        for ($i = 1; $i < count($distinct); $i++) $thresholds[] = ($distinct[$i - 1] + $distinct[$i]) / 2;
        $plans[] = compact('feature', 'order', 'groups', 'thresholds');
    }
    return $plans;
}
function discrete_search($rows, $weights, $plans)
{
    $positive = 0.0;
    $negative = 0.0;
    foreach ($rows as $i => $row) {
        $positive += $row['label'] === 1 ? $weights[$i] : 0;
        $negative += $row['label'] === -1 ? $weights[$i] : 0;
    }
    $candidates = [];
    foreach ($plans as $plan) {
        $leftPositive = 0.0;
        $leftNegative = 0.0;
        foreach ($plan['thresholds'] as $k => $threshold) {
            if ($k > 0) foreach ($plan['groups'][$k - 1] as $i) {
                $leftPositive += $rows[$i]['label'] === 1 ? $weights[$i] : 0;
                $leftNegative += $rows[$i]['label'] === -1 ? $weights[$i] : 0;
            }
            $candidates[] = ['feature' => $plan['feature'], 'threshold' => $threshold, 'polarity' => 1, 'error' => $leftPositive + ($negative - $leftNegative)];
            $candidates[] = ['feature' => $plan['feature'], 'threshold' => $threshold, 'polarity' => -1, 'error' => $leftNegative + ($positive - $leftPositive)];
        }
    }
    return choose($candidates);
}

function prediction($score)
{
    return $score > 0 ? 1 : -1;
}
function minimum_position($values)
{
    $best = 0;
    for ($i = 1; $i < count($values); $i++) if ($values[$i] < $values[$best]) $best = $i;
    return $best;
}

function curve($rounds, $train, $test, $variant)
{
    $trainScores = array_fill(0, count($train), 0.0);
    $testScores = array_fill(0, count($test), 0.0);
    $scale = 0.0;
    $result = [];
    foreach ($rounds as $round) {
        foreach ($train as $i => $row) $trainScores[$i] += $round['stump']->output($row['values']);
        foreach ($test as $i => $row) $testScores[$i] += $round['stump']->output($row['values']);
        $scale += magnitude($round);
        $margins = [];
        foreach ($train as $i => $row) $margins[] = $row['label'] * $trainScores[$i];
        $result[] = ['trainErrors' => error_count($train, $trainScores), 'testErrors' => error_count($test, $testScores),
            'trainLoss' => loss($train, $trainScores, $variant), 'testLoss' => loss($test, $testScores, $variant), 'minimum' => min($margins) / $scale];
    }
    return $result;
}
function cross_validation($rows, $params)
{
    $folds = [];
    for ($fold = 0; $fold < $params['folds']; $fold++) {
        $validation = [];
        foreach ($rows as $i => $row) if ($i % $params['folds'] === $fold) $validation[] = $i;
        $count = count($rows) - count($validation);
        $weights = [];
        foreach ($rows as $i => $row) $weights[] = $i % $params['folds'] === $fold ? 0.0 : 1 / $count;
        $model = discrete($rows, $params['rounds'], 1, $weights);
        $scores = array_fill(0, count($validation), 0.0);
        $errors = [];
        for ($t = 0; $t < $params['rounds']; $t++) {
            if ($t < count($model['rounds'])) foreach ($validation as $i => $index) {
                $scores[$i] += $model['rounds'][$t]['stump']->output($rows[$index]['values']);
            }
            $countErrors = 0;
            foreach ($validation as $i => $index) if (prediction($scores[$i]) !== $rows[$index]['label']) $countErrors++;
            $errors[] = $countErrors;
        }
        $folds[] = compact('model', 'validation', 'count', 'errors');
    }
    $errors = array_fill(0, $params['rounds'], 0);
    foreach ($folds as $fold) for ($t = 0; $t < $params['rounds']; $t++) $errors[$t] += $fold['errors'][$t];
    $selected = minimum_position($errors) + 1;
    $scores = array_fill(0, count($rows), 0.0);
    foreach ($folds as $fold) {
        $rounds = array_slice($fold['model']['rounds'], 0, $selected);
        foreach ($fold['validation'] as $i) $scores[$i] = score($rounds, $rows[$i]['values']);
    }
    return compact('folds', 'errors', 'selected', 'scores');
}
