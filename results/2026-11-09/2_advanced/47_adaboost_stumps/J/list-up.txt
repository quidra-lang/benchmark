train_discrete =: 3 : 0
    'rows labels maximum shrinkage initial_weights' =. y
    weights =. initial_weights
    if. 0=#weights do.
        weights =. (#rows)$1.0%#rows
    end.
    prepared =. prepare rows
    rounds =. 0$a:
    bound =. 1.0
    reason =. 'completed'
    for_t. i.maximum do.
        candidate =. discrete_search (<rows),(<labels),(<weights),<prepared
        error =. 1{::candidate
        if. error>:0.5 do.
            reason =. 'weak'
            break.
        end.
        if. error<:0.0000000001 do.
            alpha =. 1.0
        else.
            alpha =. shrinkage*0.5*^.(1.0-error)%error
        end.
        polarity =. 2{::candidate
        'feature thresholds unused' =. 0{::candidate
        stump =. (<feature),(<thresholds),<(-polarity*alpha),polarity*alpha
        for_i. i.#rows do.
            decision =. polarity
            if. ((<i,feature){rows)<:0{thresholds do.
                decision =. -polarity
            end.
            exponent =. alpha
            if. decision=i{labels do.
                exponent =. -alpha
            end.
            weights =. ((i{weights)*^exponent)i}weights
        end.
        z =. ordered_sum weights
        weights =. weights%z
        record =. (<stump),(<polarity),(<error),(<alpha),(<z),(<bound),(<previous),<0.0
        rounds =. rounds,<record
        if. error<:0.0000000001 do.
            reason =. 'perfect'
            break.
        end.
    end.
    (<rounds),(<reason),<0$a:
)

prepare =: 3 : 0
    result =. 0$a:
    for_j. i.{:$y do.
        column =. j{"1 y
        order =. /:column
        distinct =. ~.order{column
        batches =. 0$a:
        for_value. distinct do.
            batches =. batches,<I.column=value
        end.
        thresholds =. ((0{distinct)-1.0),((}:distinct)+}.distinct)%2.0
        result =. result,<(<order),(<distinct),(<batches),<thresholds
    end.

)
discrete_search =: 3 : 0
    'rows labels weights prepared' =. y
    positive =. ordered_sum (labels=1)#weights
    negative =. ordered_sum (labels=_1)#weights
    candidates =. 0$a:
    for_j. i.#prepared do.
        'order distinct batches thresholds' =. j{::prepared
        left_positive =. left_negative =. 0.0
        for_k. i.#thresholds do.
            if. k do.
                for_i. (k-1){::batches do.
                    if. (i{labels)=1 do.
                        left_positive =. left_positive+i{weights
                    else.
                        left_negative =. left_negative+i{weights
                    end.
                end.
            end.
            threshold =. k{thresholds
            stump =. (<j),(<,threshold),<i.0
            candidates =. candidates,<(<stump),(<left_positive+(negative-left_negative)),(<1),(<i.0),<i.0
            candidates =. candidates,<(<stump),(<left_negative+(positive-left_positive)),(<_1),(<i.0),<i.0
        end.
    end.
    choose candidates
)

    folds =. 0$a:
    errors =. Rounds$0
    for_f. i.Folds do.
        validation =. I.f=Folds|i.#TrainX
        train_count =. (#TrainX)-#validation
        weights =. (#TrainX)$1.0%train_count
        weights =. 0.0 validation}weights
        model =. train_discrete (<TrainX),(<TrainY),(<Rounds),(<1.0),<weights
        rounds =. model_rounds model
        fold_errors =. i.0
        for_t. 1+i.Rounds do.
            selected =. t first_rounds rounds
            values =. selected score_rows validation{TrainX
            fold_errors =. fold_errors,(validation{TrainY)error_count values
        end.
        errors =. errors+fold_errors
        folds =. folds,<(<model),(<train_count),(<validation),<fold_errors
    end.
    selected =. 1+{.I.errors=<./errors
    out_of_fold =. (#TrainX)$0.0
    for_fold. folds do.
        'model train_count validation unused' =. >fold
        rounds =. selected first_rounds model_rounds model
        for_i. validation do.
            out_of_fold =. (rounds row_score i{TrainX)i}out_of_fold
        end.
    end.
    (<folds),(<errors),(<selected),<out_of_fold
)

    'rounds training train_labels testing test_labels logit' =. y
    train_scores =. (#training)$0.0
    test_scores =. (#testing)$0.0
    scale =. 0.0
    result =. 0 5$0.0
    for_box. rounds do.
        record =. >box
        stump =. round_stump record
        train_scores =. train_scores+stump stump_outputs training
        test_scores =. test_scores+stump stump_outputs testing
        scale =. scale+magnitude record

        test_error =. test_labels error_count test_scores

        minimum =. (<./train_labels*train_scores)%scale
        result =. result,train_error,test_error,train_loss,test_loss,minimum
    end.
    result
)
