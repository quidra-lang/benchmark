real match_result(int home_score, int away_score)
    return if home_score > away_score then real(1) elif home_score == away_score then real(1) / real(2) else real(0)
class Match(int round, int home, int away, int home_score, int away_score)
    margin := home_score - away_score
    y := match_result(home_score, away_score)

real64 expected_score(real64 home_rating, real64 away_rating, real64 home_advantage)
    return 1.0 / (1.0 + 10.0 ^ ((away_rating - home_rating - home_advantage) / 400.0))
