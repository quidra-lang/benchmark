        var total_wait: i64 = 0;
        var max_wait: i64 = 0;
        for (state.delivered.items, 0..) |index, i| {
            const copy = s.copies.items[index];
            const packet = s.packets.items[copy.packet];
            const wait = copy.delivery - packet.arrival;
            total_wait += wait;
            max_wait = @max(max_wait, wait);
        out.add("TOTAL delivered {d} direct {d} fifo {d} causal {d} both {d} wait {d} max_wait {d} max_held {d}",
             .{ state.delivered.items.len, counts[0], counts[1], counts[2], counts[3], total_wait,
             max_wait, state.max_held });
