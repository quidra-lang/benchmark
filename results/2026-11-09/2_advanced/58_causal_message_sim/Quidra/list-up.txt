        total_wait := 0
        max_wait := 0
        for: number < int(#state.delivered)
            packet := sim.packets[state.delivered[number]]
            wait := packet.delivered_at - packet.arrives_at
            total_wait += wait
            max_wait = maximum(max_wait, wait)
        lines = lines.append("TOTAL delivered {#state.delivered} direct {modes[0]} fifo {modes[1]} causal {modes[2]} both {modes[3]} wait {total_wait} max_wait {max_wait} max_held {state.max_held}")
