const lines = (path: string) => readFileSync(path, 'ascii').split('\n').slice(0, -1);
function examples(path: string): Example[] {
  return lines(path)
    .slice(1)
    .map((line) => {
      const [id, x, y, label] = line.split(',');
      return { id, point: [Number(x), Number(y)], label: Number(label) };
    });
}
export function read() {
  const rows = lines('input/params.txt').map((line) => line.split(' '));
  const cs = rows[0].slice(1).map(Number);
  const kernels: Kernel[] = [];
  let i = 1;
  while (rows[i][0] === 'KERNEL') {
    const [, name, kind, a, b, c] = rows[i++];
    const base = { name, index: kernels.length };
    if (kind === 'linear') kernels.push({ ...base, kind });
    else if (kind === 'poly') {
      kernels.push({ ...base, kind, degree: Number(a), gamma: Number(b), coef0: Number(c) });
    } else kernels.push({ ...base, kind: 'rbf', gamma: Number(a) });
  }
  const r = rows.slice(i);
  const nums = r.map((row) => row.slice(1).map(Number));
  const params: Parameters = {
    cs,
    kernels,
    tau: nums[0][0],
    eps: nums[1][0],
    maxPasses: nums[2][0],
    folds: nums[3][0],
  };
  return {
    train: examples('input/train.csv'),
    test: examples('input/test.csv'),
    multiTrain: multi.filter((row) => row.role === 'train'),
    multiTest: multi.filter((row) => row.role === 'test'),
    params,
  };
}
export function value(kernel: Kernel, a: Point, b: Point) {
  if (kernel.kind === 'linear') return a[0] * b[0] + a[1] * b[1];
  if (kernel.kind === 'poly') {
    const s = kernel.gamma * (a[0] * b[0] + a[1] * b[1]) + kernel.coef0;
    let result = 1;
    for (let i = 0; i < kernel.degree; i++) result *= s;
    return result;
  }
  const e1 = a[0] - b[0];
  const e2 = a[1] - b[1];
  return Math.exp(-kernel.gamma * (e1 * e1 + e2 * e2));
}
export const matrix = (kernel: Kernel, points: Point[]) =>
  points.map((a) => points.map((b) => value(kernel, a, b)));
export const subset = (matrix: number[][], indices: number[]) =>
  indices.map((i) => indices.map((j) => matrix[i][j]));
export const sign = (value: number) => (value > 0 ? 1 : -1);
export function folds(count: number, k: number) {
  return Array.from({ length: k }, (_, fold) => {
    const train: number[] = [];
    const test: number[] = [];
    for (let i = 0; i < count; i++) (i % k === fold ? test : train).push(i);
    return { train, test };
  });
}
export const train = (
  points: Point[],
  labels: number[],

  C: number,
  kernel: Kernel,
  params: Parameters,

export function decision(model: Trainer, point: Point) {
  let sum = 0;
  for (let j = 0; j < model.alpha.length; j++) {
    if (model.alpha[j] > 0) {
      sum += model.alpha[j] * model.labels[j] * value(model.kernel, model.points[j], point);
    }
  }
  return sum - model.b;
}
function support(model: Trainer): Support {
  let total = 0;
  let bound = 0;
  let free = 0;
  for (const a of model.alpha) {
    if (a > 0) {
      total++;
      if (a === model.C) bound++;
      else free++;
    }
  }
  return { total, bound, free };
}

function crossValidate(train: Example[], p: Parameters, matrices: number[][][]): CrossValidation {
  const partition = folds(train.length, p.folds);
  const combinations: Candidate[] = [];
  const best: Candidate[] = [];
  for (const kernel of p.kernels) {
    const candidates: Candidate[] = [];
    for (const C of p.cs) {
      const result: Candidate = {
        kernel,
        C,
        correct: 0,
        support: 0,
        passes: 0,
        steps: 0,
        values: Array(train.length).fill(0),
        folds: [],
      };
      for (const fold of partition) {
        const model = smo.train(
          fold.train.map((i) => train[i].point),
          fold.train.map((i) => train[i].label),
          kernels.subset(matrices[kernel.index], fold.train),
          C,
          kernel,
          p,
        );
        let correct = 0;
        for (const i of fold.test) {
          result.values[i] = smo.decision(model, train[i].point);
          if (sign(result.values[i]) === train[i].label) correct++;
        }
        const sv = support(model);
        const passes = model.full + model.nonbound;
        result.folds.push({
          train: fold.train.length,
          test: fold.test.length,
          correct,
          support: sv.total,
          passes,
          steps: model.steps,
          converged: model.converged,
        });
        result.correct += correct;
        result.support += sv.total;
        result.passes += passes;
        result.steps += model.steps;
      }
      candidates.push(result);
      combinations.push(result);
    }
    best.push(candidates.slice().sort((a, b) => b.correct - a.correct || a.C - b.C)[0]);
  }
  const ranking = combinations
    .slice()
    .sort((a, b) => b.correct - a.correct || a.C - b.C || a.kernel.index - b.kernel.index);
  return { combinations, best, ranking, selected: ranking[0] };
}
function crossValidationLines(cv: CrossValidation, n: number, p: Parameters) {
  const lines = [`cross_validation folds ${p.folds} examples ${n}`];
  for (const best of cv.best) {
    lines.push(
      `best ${best.kernel.name} C ${fixed(best.C, 4)} correct ${best.correct}/${n} accuracy ${ratio(best.correct, n)}`,
    );
  }
  lines.push(`selected ${cv.selected.kernel.name} C ${fixed(cv.selected.C, 4)}`, 'ranking');
  cv.ranking.forEach((e, i) =>
    lines.push(`${i + 1} ${e.kernel.name} C ${fixed(e.C, 4)} correct ${e.correct}/${n}`),
  );
  lines.push(`cv_detail ${cv.selected.kernel.name} C ${fixed(cv.selected.C, 4)}`);
  cv.selected.folds.forEach((f, i) =>
    lines.push(
      `fold ${i + 1} train ${f.train} test ${f.test} correct ${f.correct} support_vectors ${f.support} ${runInfo(f)}`,
    ),
  );
  return lines;
}
