List<String> lines(String path) => File(path).readAsLinesSync();
Example parseExample(List<String> a) =>
    Example(a[0], (double.parse(a[1]), double.parse(a[2])), int.parse(a[3]));
List<Example> readExamples(String path) =>
    lines(path).skip(1).map((s) => parseExample(s.split(','))).toList();
Params readParams() {
  final p = Params();
  for (final line in lines('input/params.txt')) {
    final a = line.split(' ');
    switch (a[0]) {
      case 'C_GRID':
        p.cs.addAll(a.sublist(1).map(double.parse));
      case 'KERNEL':
        p.kernels.add(
          Kernel(
            a[1],
            a[2],
            a[2] == 'poly' ? int.parse(a[3]) : 0,
            a[2] == 'poly'
                ? double.parse(a[4])
                : a[2] == 'rbf'
                ? double.parse(a[3])
                : 0,
            a[2] == 'poly' ? double.parse(a[5]) : 0,
          ),
        );
      case 'TOLERANCE':
        p.tolerance = double.parse(a[1]);
      case 'EPS':
        p.epsilon = double.parse(a[1]);
      case 'MAX_PASSES':
        p.maxPasses = int.parse(a[1]);
      case 'FOLDS':
        p.folds = int.parse(a[1]);
    }
  }
  return p;
}
class Kernel {
  final String name, kind;
  final int degree;
  final double gamma, coef;
  Kernel(
    this.name,
    this.kind, [
    this.degree = 0,
    this.gamma = 0,
    this.coef = 0,
  ]);
  double value(Point a, Point b) {
    if (kind == 'rbf') {
      final e1 = a.$1 - b.$1, e2 = a.$2 - b.$2;
      return exp(-gamma * (e1 * e1 + e2 * e2));
    }
    var v = a.$1 * b.$1 + a.$2 * b.$2;
    if (kind == 'linear') return v;
    final s = gamma * v + coef;
    v = 1;
    for (var d = 0; d < degree; d++) {
      v *= s;
    }
    return v;
  }
List<List<double>> kernelMatrix(List<Example> a, Kernel k) => [
  for (final x in a) [for (final y in a) k.value(x.point, y.point)],
];
(List<Example>, List<List<double>>) subset(
  List<Example> a,
  List<List<double>> q,
  List<int> indices,
) => (
  [for (final i in indices) a[i]],
  [
    for (final i in indices) [for (final j in indices) q[i][j]],
  ],
);
int sign(double v) => v > 0 ? 1 : -1;
(List<int>, List<int>) folds(int n, int k, int fold) => (
  [
    for (var i = 0; i < n; i++)
      if (i % k != fold) i,
  ],
  [
    for (var i = 0; i < n; i++)
      if (i % k == fold) i,
  ],
);

    for (var j = 0; j < alpha.length; j++) {
      if (alpha[j] > 0)
        s += alpha[j] * data[j].label * kernel.value(data[j].point, z);
    }

    var total = 0, bound = 0;
    for (final a in alpha) {
      if (a > 0) {
        total++;

      }
    }
    return (total, bound, total - bound);
  }
Model train(
  List<Example> data,
  List<List<double>> q,
  Kernel k,
  double c,
  Params p,
) {
  final m = Model(data, q, k, c, p);
  return m;

class FoldResult {
  final int training, test, correct, support, passes, steps;
  final bool converged;
  FoldResult(
    this.training,
    this.test,
    this.correct,
    this.support,
    this.passes,
    this.steps,
    this.converged,
  );
}

class CV {
  final int kernel, cIndex;
  final double c;
  int correct = 0, support = 0, passes = 0, steps = 0;
  final List<double> values;
  final folds = <FoldResult>[];
  CV(this.kernel, this.cIndex, this.c, int n) : values = List.filled(n, 0);
}

List<double> decisions(Model m, List<Example> data) =>
    data.map((e) => m.decision(e.point)).toList();
class Application {
  final List<Example> trainData, testData;
  final Params params;
  final matrices = <List<List<double>>>[],
      grid = <List<CV>>[],
      models = <Model>[],
      trainValues = <List<double>>[],
      testValues = <List<double>>[];
  late final List<int> best;
  int selected = 0;
  Application(this.trainData, this.testData, this.params);
  void prepare() {
    final n = trainData.length, p = params;
    best = List.filled(p.kernels.length, 0);
    for (var ki = 0; ki < p.kernels.length; ki++) {
      final k = p.kernels[ki], q = kernelMatrix(trainData, k), row = <CV>[];
      matrices.add(q);
      for (var ci = 0; ci < p.cs.length; ci++) {
        final c = p.cs[ci], cv = CV(ki, ci, c, n);
        for (var fold = 0; fold < p.folds; fold++) {
          final (training, validation) = folds(n, p.folds, fold);
          final (data, matrix) = subset(trainData, q, training);
          final m = train(data, matrix, k, c, p);
          var count = 0;
          for (final i in validation) {
            final v = m.decision(trainData[i].point);
            cv.values[i] = v;
            if (sign(v) == trainData[i].label) count++;
          }
          final (support, _, _) = m.counts();
          cv.folds.add(
            FoldResult(
              training.length,
              validation.length,
              count,
              support,
              m.passes.length,
              m.steps,
              m.converged,
            ),
          );
          cv.correct += count;
          cv.support += support;
          cv.passes += m.passes.length;
          cv.steps += m.steps;
        }
        row.add(cv);
        if (ci == 0 || cv.correct > row[best[ki]].correct) best[ki] = ci;
      }
      grid.add(row);
      final chosen = grid[ki][best[ki]],
          current = grid[selected][best[selected]];
      if (chosen.correct > current.correct ||
          (chosen.correct == current.correct && chosen.c < current.c))
        selected = ki;
      final m = train(trainData, q, k, chosen.c, p);
      models.add(m);
      trainValues.add(m.trainingValues());
      testValues.add(decisions(m, testData));
    }
  }
  List<String> cvOutput() {
    final n = trainData.length,
        p = params,
        out = ['cross_validation folds ${p.folds} examples $n'],
        ranked = <CV>[],
        header = [
          'kernel',
          'C',
          for (var i = 1; i <= p.folds; i++) 'fold$i',
          'correct',
          'accuracy',
          'support_vectors',
          'passes',
          'steps',
        ],
        csv = <String>[];
    csv.add(header.join(','));
    for (var ki = 0; ki < p.kernels.length; ki++) {
      final k = p.kernels[ki], chosen = grid[ki][best[ki]];
      out.add(
        'best ${k.name} C ${fixed(chosen.c, 4)} correct ${chosen.correct}/$n accuracy ${ratio(chosen.correct, n)}',
      );
      for (final cv in grid[ki]) {
        ranked.add(cv);
        csv.add(
          [
            k.name,
            fixed(cv.c, 4),
            ...cv.folds.map((f) => '${f.correct}'),
            '${cv.correct}',
            ratio(cv.correct, n),
            '${cv.support}',
            '${cv.passes}',
            '${cv.steps}',
          ].join(','),
        );
      }
    }
    final chosen = grid[selected][best[selected]],
        name = p.kernels[selected].name;
    out.addAll(['selected $name C ${fixed(chosen.c, 4)}', 'ranking']);
    ranked.sort((x, y) {
      var c = y.correct.compareTo(x.correct);
      if (c != 0) return c;
      c = x.c.compareTo(y.c);
      return c != 0 ? c : x.kernel.compareTo(y.kernel);
    });
    for (var i = 0; i < ranked.length; i++) {
      final cv = ranked[i];
      out.add(
        '${i + 1} ${p.kernels[cv.kernel].name} C ${fixed(cv.c, 4)} correct ${cv.correct}/$n',
      );
    }
    out.add('cv_detail $name C ${fixed(chosen.c, 4)}');
    for (var i = 0; i < chosen.folds.length; i++) {
      final f = chosen.folds[i];
      out.add(
        'fold ${i + 1} train ${f.training} test ${f.test} correct ${f.correct} support_vectors ${f.support} passes ${f.passes} steps ${f.steps} converged ${yes(f.converged)}',
      );
    }
    writeLines('cv_grid.csv', csv);
    return out;
  }

void main() {
  final a = Application(
    readExamples('input/train.csv'),
    readExamples('input/test.csv'),
    readParams(),
  );
  a.prepare();
  final parts = [
        a.cvOutput(),
        for (var i = 0; i < a.params.kernels.length; i++) a.modelOutput(i),
        a.progressOutput(),
        a.pathOutput(),
        a.learningOutput(),
      ],
      cal = a.calibrate(),
  final out = <String>[];
  for (var i = 0; i < parts.length; i++) {
    if (i > 0) out.add('');
    out.addAll(parts[i]);
  }
  writeLines('model.txt', out);
  a.predictionsOutput(cal, peg);
