    KernelDefs =: 0$a:
    for_line. <;._2 (1!:1 <'input/params.txt') do.
        fields =. split_words >line
        select. 0{::fields
        case. 'C_GRID' do. CGrid =: 0.0+parse_number&>}.fields
        case. 'KERNEL' do.
            name =. 1{::fields
            kind =. 2{::fields
            degree =. 0
            gamma =. coef0 =. 0.0
            if. kind-:'poly' do.
                degree =. parse_number 3{::fields
                gamma =. 0.0+parse_number 4{::fields

            elseif. kind-:'rbf' do. gamma =. 0.0+parse_number 3{::fields end.
            KernelDefs =: KernelDefs,<(<name),(<kind),(<degree),(<gamma),<coef0
        case. 'TOLERANCE' do. Tolerance =: 0.0+parse_number 1{::fields

kernel_value =: 4 : 0
    'first second' =. y
    'name kind degree gamma coef0' =. x
    if. kind-:'rbf' do.
        'e1 e2' =. first-second
        ^-gamma*((e1*e1)+e2*e2) return.
    end.
    dot =. ((0{first)*0{second)+(1{first)*1{second
    if. kind-:'linear' do. dot return. end.
    factor =. (gamma*dot)+coef0
    value =. 1.0
    for_iteration. i.degree do. value =. value*factor end.
    value
)
kernel_matrix =: 4 : 0
    matrix =. ((#y),#y)$0.0
    for_i. i.#y do.
        for_j. i.#y do.
            value =. x kernel_value (<i{y),<j{y
            matrix =. value(<i,j)}matrix
        end.
    end.
    matrix
)

smo_decision =: 4 : 0
    points =. 0{::x
    labels =. 1{::x
    kernel =. 2{::x
    alpha =. 5{::x
    total =. 0.0
    for_i. I.alpha>0 do.
        term =. ((i{alpha)*i{labels)*(kernel kernel_value (<i{points),<y)
        total =. total+term
    end.
    total-6{::x

    bound =. +/(alpha>0)*.alpha=coefficient

cross_validate =: 3 : 0
    TrainIds =: 0{::TrainData
    TrainX =: 1{::TrainData
    TrainY =: 2{::TrainData
    TestIds =: 0{::TestData
    TestX =: 1{::TestData
    TestY =: 2{::TestData
    partitions =. (#TrainY)make_folds FoldCount
    Matrices =: Combinations =: Best =: 0$a:
    keys =. 0 3$0.0
    for_k. i.#KernelDefs do.
        kernel =. k{::KernelDefs
        matrix =. kernel kernel_matrix TrainX
        Matrices =: Matrices,<matrix
        best =. a:
        for_coefficient. CGrid do.
            values =. (#TrainY)$0.0
            foldrecords =. 0$a:
            correct =. support =. passes =. steps =. 0
            for_partition. partitions do.
                'train_indices test_indices' =. >partition
                points =. train_indices{TrainX
                labels =. train_indices{TrainY
                submatrix =. train_indices kernel_submatrix matrix
                model =. smo_train (<points),(<labels),(<kernel),(<submatrix),<coefficient
                predicted =. model smo_decisions test_indices{TrainX
                values =. predicted test_indices}values
                foldcorrect =. (test_indices{TrainY)correct_count predicted
                foldsupport =. {.support_counts model
                counters =. 7{::model
                foldrecords =. foldrecords,<(<#train_indices),(<#test_indices),(<foldcorrect),(<foldsupport),<model
                correct =. correct+foldcorrect
                support =. support+foldsupport
                passes =. passes++/2{.counters
                steps =. steps+3{counters
            end.
            result =. (<k),(<coefficient),(<values),(<foldrecords),(<correct),(<support),(<passes),<steps
            Combinations =: Combinations,<result
            keys =. keys,(-correct),coefficient,k
            if. best-:a: do. best =. result
            elseif. correct>4{::best do. best =. result end.
        end.
        Best =: Best,<best
    end.
    Ranking =: (/:keys){Combinations
    Selected =: 0{::Ranking
    i.0 0
)
