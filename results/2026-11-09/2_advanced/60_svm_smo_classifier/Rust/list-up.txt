pub fn binary(file: &str) -> Vec<Example> {
    fs::read_to_string(file).unwrap().lines().skip(1).map(|line| {
        let w: Vec<_> = line.split(',').collect();
        Example { id: w[0].to_owned(), point: [w[1].parse().unwrap(), w[2].parse().unwrap()], label: w[3].parse().unwrap() }
    }).collect()
}
pub fn settings() -> Settings {
    let mut p = Settings { candidates: Vec::new(), kernels: Vec::new(), tolerance: 0.0, eps: 0.0,
        max_passes: 0, folds: 0, fractions: Vec::new(), newton: 0, multi_kernel: String::new(), multi_c: 0.0,
        lambda: 0.0, iterations: 0, seed: 0, x: [0.0; 2], y: [0.0; 2], width: 0, height: 0 };
    for line in fs::read_to_string("input/params.txt").unwrap().lines() {
        let w: Vec<_> = line.split_whitespace().collect();
        match w[0] {
            "C_GRID" => p.candidates = w[1..].iter().map(|v| v.parse().unwrap()).collect(),
            "KERNEL" => p.kernels.push(Kernel { name: w[1].to_owned(), kind: match w[2] {
                "linear" => Kind::Linear,
                "poly" => Kind::Polynomial { degree: w[3].parse().unwrap(), gamma: w[4].parse().unwrap(), constant: w[5].parse().unwrap() },
                _ => Kind::Rbf(w[3].parse().unwrap()),
            } }),
            "TOLERANCE" => p.tolerance = w[1].parse().unwrap(),
            "EPS" => p.eps = w[1].parse().unwrap(),
            "MAX_PASSES" => p.max_passes = w[1].parse().unwrap(),
            "FOLDS" => p.folds = w[1].parse().unwrap(),
        }
    }
    p
}
pub enum Kind { Linear, Polynomial { degree: usize, gamma: f64, constant: f64 }, Rbf(f64) }

#[derive(Clone)]
pub struct Kernel { pub name: String, pub kind: Kind }

impl Kernel {
    pub fn value(&self, a: [f64; 2], b: [f64; 2]) -> f64 {
        match self.kind {
            Kind::Linear => a[0]*b[0]+a[1]*b[1],
            Kind::Polynomial { degree, gamma, constant } => {
                let s = gamma*(a[0]*b[0]+a[1]*b[1])+constant;
                let mut value = 1.0;
                for _ in 0..degree { value *= s; }
                value
            }
            Kind::Rbf(gamma) => {
                let (e1, e2) = (a[0]-b[0], a[1]-b[1]);
                (-gamma*(e1*e1+e2*e2)).exp()
            }
        }
    }
    pub fn matrix(&self, points: &[[f64; 2]]) -> Vec<Vec<f64>> {
        points.iter().map(|&a| points.iter().map(|&b| self.value(a, b)).collect()).collect()
    }
pub fn sign(value: f64) -> i32 { if value > 0.0 { 1 } else { -1 } }
pub fn folds(n: usize, k: usize, fold: usize) -> (Vec<usize>, Vec<usize>) {
    ((0..n).filter(|i| i%k != fold).collect(), (0..n).filter(|i| i%k == fold).collect())
}
pub fn train(points: &[[f64; 2]], labels: &[f64], matrix: &[Vec<f64>], c: f64, params: &Settings) -> Model {
    let model = Model { points: points.to_vec(), labels: labels.to_vec(), alpha: vec![0.0; points.len()], b: 0.0, c,
        full: 0, nonbound: 0, examined: 0, steps: 0, choices: [0; 3], clipping: [0; 3], converged: false, passes: Vec::new() };
    let mut t = Trainer { model, errors: labels.iter().map(|y| -y).collect(), matrix, params };
    t.model
}
    pub fn decision(&self, kernel: &Kernel, point: [f64; 2]) -> f64 {
        let mut sum = 0.0;
        for i in 0..self.alpha.len() {
            if self.alpha[i] > 0.0 { sum += self.alpha[i]*self.labels[i]*kernel.value(self.points[i], point); }
        }
        sum-self.b
    }

    pub fn support(&self) -> (usize, usize, usize) {
        let total = self.alpha.iter().filter(|&&a| a > 0.0).count();
        let bound = self.alpha.iter().filter(|&&a| a == self.c).count();
        (total, bound, total-bound)
    }
struct FoldResult { training: usize, test: usize, correct: usize, support: usize, passes: usize, steps: usize, converged: bool }
struct CrossValidation { folds: Vec<FoldResult>, values: Vec<f64>, correct: usize }

fn cross_validation(data: &[Example], kernel: &Kernel, matrix: &[Vec<f64>], c: f64, params: &Settings) -> CrossValidation {
    let mut result = CrossValidation { folds: Vec::new(), values: vec![0.0; data.len()], correct: 0 };
    for fold in 0..params.folds {
        let (training, test) = evaluation::folds(data.len(), params.folds, fold);
        let points: Vec<_> = training.iter().map(|&i| data[i].point).collect();
        let labels: Vec<_> = training.iter().map(|&i| data[i].label as f64).collect();
        let q: Vec<Vec<_>> = training.iter().map(|&i| training.iter().map(|&j| matrix[i][j]).collect()).collect();
        let model = smo::train(&points, &labels, &q, c, params);
        let mut correct = 0;
        for &i in &test {
            result.values[i] = model.decision(kernel, data[i].point);
            correct += usize::from(sign(result.values[i]) == data[i].label);
        }
        result.correct += correct;
        result.folds.push(FoldResult { training: training.len(), test: test.len(), correct, support: model.support().0,
            passes: model.passes.len(), steps: model.steps, converged: model.converged });
    }
    result
}
