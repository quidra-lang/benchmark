    public static List<string> Lines(string name) {
        return File.ReadAllLines(Path.Combine("input", name)).ToList();

    public static double[] Reals(string[] w, int first) {
        double[] a = new double[w.Length - first];
        for (int i = 0; i < a.Length; i++)
            a[i] = double.Parse(w[i + first]);
        return a;
    }
    public static Example ParseExample(string[] w) {
        return new Example(w[0], new Kernels.Point(double.Parse(w[1]), double.Parse(w[2])), int.Parse(w[3]));
    }
    public static List<Example> Binary(string name) {
        List<string> lines = Lines(name);
        List<Example> result = new();
        for (int i = 1; i < lines.Count; i++)
            result.Add(ParseExample(lines[i].Split(',')));
        return result;
    }
    public static Data Read() {
        List<Example> train = Binary("train.csv"), test = Binary("test.csv"), multiTrain = new(),
                      multiTest = new();
        Settings s = new Settings();
        foreach (string line in Lines("params.txt")) {
            string[] w = line.Split(' ');
            switch (w[0]) {
            case "C_GRID":
                s.coefficients = Reals(w, 1);
                break;
            case "KERNEL": {
                string kind = w[2];
                int degree = kind.Equals("poly") ? int.Parse(w[3]) : 0;
                double gamma = kind.Equals("poly")  ? double.Parse(w[4])
                               : kind.Equals("rbf") ? double.Parse(w[3])
                                                    : 0,
                       offset = kind.Equals("poly") ? double.Parse(w[5]) : 0;
                s.kernels.Add(new Kernels.Kernel(w[1], kind, degree, gamma, offset));
                break;
            }
            case "TOLERANCE":
                s.tolerance = double.Parse(w[1]);
                break;
            case "EPS":
                s.eps = double.Parse(w[1]);
                break;
            case "MAX_PASSES":
                s.maximum = int.Parse(w[1]);
                break;
            case "FOLDS":
                s.folds = int.Parse(w[1]);
                break;
            }
        }
        return new Data(train, test, multiTrain, multiTest, s);
    }

    public record Kernel(string name, string kind, int degree, double gamma, double offset) {
        public double Value(Point a, Point b) {
            if (kind.Equals("linear"))
                return a.x * b.x + a.y * b.y;
            if (kind.Equals("poly")) {
                double s = gamma * (a.x * b.x + a.y * b.y) + offset, value = 1;
                for (int i = 0; i < degree; i++)
                    value *= s;
                return value;
            }
            double e1 = a.x - b.x, e2 = a.y - b.y;
            return Math.Exp(-gamma * (e1 * e1 + e2 * e2));
        }
        public double[][] Matrix(List<Point> points) {
            int n = points.Count;
            double[][] matrix = Enumerable.Range(0, n)
                                    .Select(
                                        _ => new double[n])
                                    .ToArray();
            for (int i = 0; i < n; i++)
                for (int j = 0; j < n; j++)
                    matrix[i][j] = Value(points[i], points[j]);
            return matrix;
        }
    }
    public static double[][] Submatrix(double[][] matrix, List<int> indices) {
        int n = indices.Count;
        double[][] result = Enumerable.Range(0, n)
                                .Select(
                                    _ => new double[n])
                                .ToArray();
        for (int i = 0; i < n; i++)
            for (int j = 0; j < n; j++)
                result[i][j] = matrix[indices[i]][indices[j]];
        return result;
    }

    public static int[] Support(Smo.Model model) {
        int[] counts = new int[3];
        foreach (double a in model.alpha)
            if (a > 0) {
                counts[0]++;

            }
        return counts;
    }

    public static CrossValidation CrossValidate(int kernel, double c) {
        CrossValidation result = new CrossValidation();
        result.kernel = kernel;
        result.c = c;
        result.decisions = new double[points.Count];
        foreach (Evaluation.Fold fold in Evaluation.Folds(points.Count, settings.folds)) {
            Smo.Model model = Smo.Train(SelectPoints(fold.train), SelectLabels(trainLabels, fold.train),
                                        settings.kernels[kernel],
                                        Kernels.Submatrix(matrices[kernel], fold.train), c, settings);
            int correct = 0;
            foreach (int i in fold.test) {
                double value = model.Decision(points[i]);
                result.decisions[i] = value;
                if (Evaluation.Sign(value) == trainLabels[i])
                    correct++;
            }
            result.folds.Add(new FoldResult(model, fold.train.Count, fold.test.Count, correct));
            result.correct += correct;
            result.sv += Support(model)[0];
            result.passes += model.Passes();
            result.steps += model.steps;
        }
        return result;
    }
