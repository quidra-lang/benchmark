    static List<String> lines(String name) throws Exception {
        return Files.readAllLines(Path.of("input", name));
    }
    static double[] reals(String[] w, int first) {
        double[] a = new double[w.length - first];
        for (int i = 0; i < a.length; i++)
            a[i] = Double.parseDouble(w[i + first]);
        return a;
    }
    static Example example(String[] w) {
        return new Example(w[0], new Kernels.Point(Double.parseDouble(w[1]), Double.parseDouble(w[2])),
                           Integer.parseInt(w[3]));
    }
    static List<Example> binary(String name) throws Exception {
        List<String> lines = lines(name);
        List<Example> result = new ArrayList<>();
        for (int i = 1; i < lines.size(); i++)
            result.add(example(lines.get(i).split(",")));
        return result;
    }
    static Data read() throws Exception {
        List<Example> train = binary("train.csv"), test = binary("test.csv"), multiTrain = new ArrayList<>(),
                      multiTest = new ArrayList<>();
        Settings s = new Settings();
        for (String line : lines("params.txt")) {
            String[] w = line.split(" ");
            switch (w[0]) {
            case "C_GRID" -> s.coefficients = reals(w, 1);
            case "KERNEL" -> {
                String kind = w[2];
                int degree = kind.equals("poly") ? Integer.parseInt(w[3]) : 0;
                double gamma = kind.equals("poly")  ? Double.parseDouble(w[4])
                               : kind.equals("rbf") ? Double.parseDouble(w[3])
                                                    : 0,
                       offset = kind.equals("poly") ? Double.parseDouble(w[5]) : 0;
                s.kernels.add(new Kernels.Kernel(w[1], kind, degree, gamma, offset));
            }
            case "TOLERANCE" -> s.tolerance = Double.parseDouble(w[1]);
            case "EPS" -> s.eps = Double.parseDouble(w[1]);
            case "MAX_PASSES" -> s.maximum = Integer.parseInt(w[1]);
            case "FOLDS" -> s.folds = Integer.parseInt(w[1]);
            }
        }
        return new Data(train, test, multiTrain, multiTest, s);
    }

    record Point(double x, double y) {}
    record Kernel(String name, String kind, int degree, double gamma, double offset) {
        double value(Point a, Point b) {
            if (kind.equals("linear"))
                return a.x() * b.x() + a.y() * b.y();
            if (kind.equals("poly")) {
                double s = gamma * (a.x() * b.x() + a.y() * b.y()) + offset, value = 1;
                for (int i = 0; i < degree; i++)
                    value *= s;
                return value;
            }
            double e1 = a.x() - b.x(), e2 = a.y() - b.y();
            return Math.exp(-gamma * (e1 * e1 + e2 * e2));
        }
        double[][] matrix(List<Point> points) {
            int n = points.size();
            double[][] matrix = new double[n][n];
            for (int i = 0; i < n; i++)
                for (int j = 0; j < n; j++)
                    matrix[i][j] = value(points.get(i), points.get(j));
            return matrix;
        }
    }
    static double[][] submatrix(double[][] matrix, List<Integer> indices) {
        int n = indices.size();
        double[][] result = new double[n][n];
        for (int i = 0; i < n; i++)
            for (int j = 0; j < n; j++)
                result[i][j] = matrix[indices.get(i)][indices.get(j)];
        return result;
    }
    static int sign(double value) {
        return value > 0 ? 1 : -1;
    }
    static List<Fold> folds(int size, int count) {
        List<Fold> folds = new ArrayList<>();
        for (int k = 0; k < count; k++) {
            List<Integer> train = new ArrayList<>(), test = new ArrayList<>();
            for (int i = 0; i < size; i++)
                (i % count == k ? test : train).add(i);
            folds.add(new Fold(train, test));
        }
        return folds;
    }

    static int[] support(Smo.Model model) {
        int[] counts = new int[3];
        for (double a : model.alpha)
            if (a > 0) {
                counts[0]++;
                counts[a == model.c ? 1 : 2]++;
            }
        return counts;
    }

    static final class CrossValidation {
        int kernel, correct, sv, passes, steps;
        double c;
        double[] decisions;
        List<FoldResult> folds = new ArrayList<>();

    static CrossValidation crossValidation(int kernel, double c) {
        CrossValidation result = new CrossValidation();
        result.kernel = kernel;
        result.c = c;
        result.decisions = new double[points.size()];
        for (Evaluation.Fold fold : Evaluation.folds(points.size(), settings.folds)) {
            Smo.Model model = Smo.train(selectPoints(fold.train()), selectLabels(trainLabels, fold.train()),
                                        settings.kernels.get(kernel),
                                        Kernels.submatrix(matrices.get(kernel), fold.train()), c, settings);
            int correct = 0;
            for (int i : fold.test()) {
                double value = model.decision(points.get(i));
                result.decisions[i] = value;
                if (Evaluation.sign(value) == trainLabels[i])
                    correct++;
            }
            result.folds.add(new FoldResult(model, fold.train().size(), fold.test().size(), correct));
            result.correct += correct;
            result.sv += support(model)[0];
            result.passes += model.passes();
            result.steps += model.steps;
        }
        return result;
    }

        int order = Integer.compare(b.correct, a.correct);

    public static void main(String[] args) throws Exception {
        data = Dataset.read();
        settings = data.settings();
        points = Dataset.points(data.train());
        trainLabels = Dataset.labels(data.train());
        testLabels = Dataset.labels(data.test());
        List<CrossValidation> grid = new ArrayList<>(), best = new ArrayList<>();
        int kernelCount = settings.kernels.size();
        for (int k = 0; k < kernelCount; k++) {
            matrices.add(settings.kernels.get(k).matrix(points));
            CrossValidation chosen = null;
            for (double c : settings.coefficients) {
                CrossValidation cv = crossValidation(k, c);
                grid.add(cv);
                if (chosen == null || cv.correct > chosen.correct)
                    chosen = cv;
            }
            best.add(chosen);
        }
        CrossValidation selected = best.get(0);
        for (CrossValidation cv : best)
            if (compare(cv, selected) < 0)
                selected = cv;
        List<CrossValidation> ranked = new ArrayList<>(grid);
        ranked.sort(Main::compare);
        List<Final> models = new ArrayList<>();
        for (int k = 0; k < kernelCount; k++) {
            Kernels.Kernel kernel = settings.kernels.get(k);
            models.add(
                evaluate(Smo.train(points, trainLabels, kernel, matrices.get(k), best.get(k).c, settings)));
        }
        Final selectedModel = models.get(selected.kernel);
        Calibration calibration = calibrate(selected, selectedModel);
        List<String> modelLines = new ArrayList<>(
            List.of("cross_validation folds " + settings.folds + " examples " + points.size()));
        for (CrossValidation cv : best)
            modelLines.add("best " + settings.kernels.get(cv.kernel).name() + " C " + f(cv.c) + " correct " +
                           cv.correct + "/" + points.size() + " accuracy " +
                           Evaluation.ratio(cv.correct, points.size()));
        modelLines.add("selected " + selectedModel.model.kernel.name() + " C " + f(selected.c));
        modelLines.add("ranking");

        for (int i = 0; i < selected.folds.size(); i++) {

                           " correct " + fold.correct() + " support_vectors " + support(fold.model())[0] +
