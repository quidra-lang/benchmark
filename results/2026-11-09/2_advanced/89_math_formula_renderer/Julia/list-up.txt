const ATOM_KINDS = Set(["identifier","number","symbol","command","text"])
const ARGUMENT_KINDS = union(ATOM_KINDS,Set(["eqref","{","frac","binom","overset","underset","sqrt","accent"]))
function argument!(parser,name)
    peek_token(parser).kind in ARGUMENT_KINDS || formula_error(peek_token(parser).column,"missing argument for " * name)
    nucleus!(parser)
end

        if peek_token(parser).kind == "symbol" && peek_token(parser).value == "["
            opening = take_token!(parser)
            elements = element_list!(parser;index=true)
            (peek_token(parser).kind == "symbol" && peek_token(parser).value == "]") || formula_error(opening.column,"missing ]")
            take_token!(parser)
            index = collapse(elements)
        end
        body = argument!(parser,"\\sqrt")
        return index === nothing ? node("sqrt",[body]) : node("root",[index,body])
function radical_box(body,index=nothing)
    height = body.height
    extra = index === nothing ? 0 : max(0,index.width-1)
    top = index === nothing ? 0 : max(0,index.height-height)
    result = new_box(extra+height+1+body.width,height+1+top,body.baseline+1+top,index === nothing ? "sqrt" : "root")
    draw_string!(result,repeat("_",body.width),extra+height+1,top)
    draw_string!(result,"\\",extra,height+top)
    for row in 1:height
        draw_string!(result,"/",extra+height-row+1,row+top)
    end
    if index === nothing
        add_child!(result,body,extra+height+1,1)
    else
        add_child!(result,index,0,height-index.height+top,"index: ")
        add_child!(result,body,extra+height+1,1+top,"body: ")
    end
    result

        index = typeset(node.children[1],parameters,references,"script")
        body = typeset(node.children[2],parameters,references,style)
        return radical_box(body,index)

    width::Int
    height::Int
    baseline::Int
    description::String
    children::Vector{Tuple{FormulaBox,Int,Int,String}}
    strings::Vector{Tuple{String,Int,Int}}
end
