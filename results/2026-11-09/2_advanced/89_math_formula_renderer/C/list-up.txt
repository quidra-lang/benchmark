        if (!contains_word("frac binom overset underset sqrt left right", name)) {
            *error = (Issue){column, format_text("unknown command \\%s", name)};
            free(name);
            return out;
        }
        Kind kind = !strcmp(name, "frac")       ? K_FRAC
                    : !strcmp(name, "binom")    ? K_BINOM
                    : !strcmp(name, "overset")  ? K_OVERSET
                    : !strcmp(name, "underset") ? K_UNDERSET
                    : !strcmp(name, "sqrt")     ? K_SQRT
                    : !strcmp(name, "left")     ? K_LEFT
                                                : K_RIGHT;
        token_push(&out, kind, copy_text(""), column);
static Node *argument(Parser *parser, const char *name) {
    if (parser->error.message)
        return node_new(parser->pool, K_ROW);
    Token token = parser_peek(parser);
    Kind kind = token.kind;
    if (kind == K_IDENTIFIER || kind == K_NUMBER || kind == K_SYMBOL || kind == K_COMMAND ||
        kind == K_TEXT || kind == K_EQREF || kind == K_OPEN_BRACE || kind == K_FRAC ||
        kind == K_BINOM || kind == K_OVERSET || kind == K_UNDERSET || kind == K_SQRT ||
        kind == K_ACCENT)
        return nucleus(parser);
    fail(parser, token.column, format_text("missing argument for %s", name));
    return node_new(parser->pool, K_ROW);
}
    case K_SQRT: {
        Token next = parser_peek(parser);
        if (next.kind == K_SYMBOL && !strcmp(next.value, "[")) {
            Token open = parser_read(parser);
            NodeList nodes = parse_elements(parser, false, true);
            next = parser_peek(parser);
            if (!parser->error.message) {
                if (next.kind != K_SYMBOL || strcmp(next.value, "]"))
                    fail(parser, open.column, copy_text("missing ]"));
                else
                    parser_read(parser);
            }
            out->kind = K_ROOT;
            node_push(&out->items, collapse(parser, nodes));
        }
        node_push(&out->items, argument(parser, "\\sqrt"));
        return out;
    }
    if (kind == K_SQRT || kind == K_ROOT) {
        Box *body = typeset(typesetter, node->items.data[node->items.count - 1], script),
            *index = NULL;
        int extra = 0, top = 0;
        if (kind == K_ROOT) {
            index = typeset(typesetter, node->items.data[0], true);
            extra = maximum(0, index->width - 1);
            top = maximum(0, index->height - body->height);
        }
        int height = body->height;
        Box *out = new_box(pool, extra + height + 1 + body->width, height + 1 + top,
                           body->baseline + 1 + top, copy_text(index ? "root" : "sqrt"));
        box_draw_owned(out, repeated('_', body->width), extra + height + 1, top);
        box_draw(out, "\\", extra, height + top);
        for (int r = 1; r <= height; ++r)
            box_draw(out, "/", extra + height - r + 1, r + top);
        if (index) {
            box_add(out, index, 0, height - index->height + top, "index: ");
            box_add(out, body, extra + height + 1, 1 + top, "body: ");
        } else
            box_add(out, body, extra + height + 1, 1 + top, "");
        return out;
    }
Box *new_box(BoxPool *pool, int width, int height, int baseline, char *description) {
    Box *box = allocate(sizeof *box);
    box->width = width;
    box->height = height;
    box->baseline = baseline;
    box->description = description;
    box_push(&pool->boxes, box);
    return box;
