        const std::map<std::string, Kind> constructs{
            {"frac", Kind::Fraction},     {"binom", Kind::Binomial},  {"overset", Kind::Overset},
            {"underset", Kind::Underset}, {"sqrt", Kind::SquareRoot}, {"left", Kind::Left},
            {"right", Kind::Right}};
        auto construct = constructs.find(name);
        if (construct == constructs.end()) {
            fail(column, "unknown command \\" + name);
            return out;
        }
        out.tokens.push_back({construct->second, "", column});
NodePtr Parser::argument(const std::string &name) {
    Kind kind = peek().kind;
    if (kind == Kind::Identifier || kind == Kind::Number || kind == Kind::Symbol ||
        kind == Kind::Command || kind == Kind::Text || kind == Kind::Eqref ||
        kind == Kind::OpenBrace || kind == Kind::Fraction || kind == Kind::Binomial ||
        kind == Kind::Overset || kind == Kind::Underset || kind == Kind::SquareRoot ||
        kind == Kind::Accent)
        return nucleus();
    throw Issue{peek().column, "missing argument for " + name};
}
    case Kind::SquareRoot: {
        if (peek().kind == Kind::Symbol && peek().value == "[") {
            Token open = read();
            Nodes index = elements(false, true);
            if (peek().kind != Kind::Symbol || peek().value != "]")
                throw Issue{open.column, "missing ]"};
            read();
            out->kind = Kind::Root;
            out->items.push_back(collapse(std::move(index)));
        }
        out->items.push_back(argument("\\sqrt"));
        return out;
    }
    if (kind == Kind::SquareRoot || kind == Kind::Root) {
        BoxPtr body = typeset(*node.items.back(), script), index;
        int extra = 0, top = 0;
        if (kind == Kind::Root) {
            index = typeset(*node.items[0], true);
            extra = std::max(0, index->width - 1);
            top = std::max(0, index->height - body->height);
        }
        int height = body->height;
        BoxPtr out = newBox(extra + height + 1 + body->width, height + 1 + top,
                            body->baseline + 1 + top, index ? "root" : "sqrt");
        out->draw(std::string(body->width, '_'), extra + height + 1, top);
        out->draw("\\", extra, height + top);
        for (int r = 1; r <= height; ++r)
            out->draw("/", extra + height - r + 1, r + top);
        if (index) {
            out->add(index, 0, height - index->height + top, "index: ");
            out->add(body, extra + height + 1, 1 + top, "body: ");
        } else
            out->add(body, extra + height + 1, 1 + top);
        return out;
    }
BoxPtr newBox(int width, int height, int baseline, std::string description) {
    BoxPtr out = std::make_shared<Box>();
    out->width = width;
    out->height = height;
    out->baseline = baseline;
    out->description = std::move(description);
    return out;
}
