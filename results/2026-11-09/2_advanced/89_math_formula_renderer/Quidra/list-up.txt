                elif has(["frac", "binom", "overset", "underset", "sqrt", "left", "right"], name)
                    kind = name
                else
                    return Problem(column, "unknown command \{name}")
        else
            return Problem(column, "invalid character {character}")
        result = result.append(Token(kind, value, column))
int argument(Parsing &context, string name)
    token := peek(context)
    if lexer.has(["identifier", "number", "symbol", "command", "text", "eqref", "{{", "frac", "binom", "overset", "underset", "sqrt", "accent"], token.kind)
        return nucleus(&context)
    return fail(&context, token.column, "missing argument for {name}")
    if token.kind == "sqrt"
        int | none index := none
        next := peek(context)
        if next.kind == "symbol" and next.value == "["
            opening := take(&context)
            nodes := parse_list(&context, false, true)
            if failed(context)
                return -1
            next = peek(context)
            if next.kind != "symbol" or next.value != "]"
                return fail(&context, opening.column, "missing ]")
            ignored := take(&context)
            index = collapse(&context, nodes)
        body := argument(&context, "\sqrt")
        if failed(context)
            return -1
        match index
            int found
                return add(&context, Node("root", children = [found, body]))
            none
                return add(&context, Node("sqrt", children = [body]))
    if node.kind == "sqrt" or node.kind == "root"
        body_index := typeset(nodes, node.children[-1], script, params, references, &store)
        boxes.Box body := store.items[body_index]
        int | none root_index := none
        extra := 0
        top := 0
        if node.kind == "root"
            built := typeset(nodes, node.children[0], true, params, references, &store)
            root_index = built
            extra = boxes.maximum(0, store.items[built].width - 1)
            top = boxes.maximum(0, store.items[built].height - body.height)
        height := body.height
        result := boxes.Box(extra + height + 1 + body.width, height + 1 + top, body.baseline + 1 + top, node.kind)
        boxes.ink(&result, extra + height + 1, top, boxes.repeat("_", body.width))
        boxes.ink(&result, extra, height + top, "\")
        for: 1 <= row < height + 1
            boxes.ink(&result, extra + height - row + 1, row + top, "/")
        match root_index
            int built
                boxes.child(&result, built, 0, height + top - store.items[built].height, "index: ")
                boxes.child(&result, body_index, extra + height + 1, 1 + top, "body: ")
            none
                boxes.child(&result, body_index, extra + height + 1, 1 + top)
        return boxes.add(&store, result)
class Box(int width, int height, int baseline, string description, Child[] children := [], Ink[] strings := [])
int add(Store &store, const Box box)
    index := int(#store.items)
    store.items = store.items.append(box)
    return index
