class Ratio(int numerator, int denominator)
    int compare(Ratio other)
        first_product := numerator * other.denominator
        second_product := other.numerator * denominator
        return if first_product < second_product then -1 elif first_product > second_product then 1 else 0

    string text()
        left := numerator
        right := denominator
        while right != 0
            left, right = right, left % right
        reduced_numerator := numerator / left
        reduced_denominator := denominator / left
        return "{reduced_numerator}/{reduced_denominator}"

Ratio similarity(const Fingerprint first, const Fingerprint second)
    common := 0
    values := first.bits.values()
    for value := values[]
        if second.bits.has(value)
            common += 1
    return Ratio(common, int(first.bits.size()) + int(second.bits.size()) - common)
