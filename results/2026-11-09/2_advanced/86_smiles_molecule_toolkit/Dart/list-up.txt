class Fraction {
  final int numerator, denominator;
  Fraction(this.numerator, this.denominator);
  int compare(Fraction b) =>
      (numerator * b.denominator).compareTo(b.numerator * denominator);
  @override
  String toString() => '$numerator/$denominator';
}

  final g = gcd(n, d);

}
