function tanimoto(array $a, array $b): array
{
    $common = count(array_intersect_key($a, $b));
    return [$common, count($a) + count($b) - $common];
}
function fractionText(array $f): string
{
    [$a, $b] = $f;
    $x = $a;
    $y = $b;
    while ($y)
        [$x, $y] = [$y, $x % $y];
    return intdiv($a, $x) . '/' . intdiv($b, $x);
}

function fractionCompare(array $a, array $b): int
{
    return ($a[0] * $b[1]) <=> ($b[0] * $a[1]);
}
