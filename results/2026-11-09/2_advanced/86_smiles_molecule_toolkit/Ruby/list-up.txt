  def self.tanimoto(first, second)
    common = (first & second).length
    [common, first.length + second.length - common]
  end
  def self.fraction_text(value)
    p, q = value
    divisor = p.gcd(q)
    "#{p / divisor}/#{q / divisor}"
  end
