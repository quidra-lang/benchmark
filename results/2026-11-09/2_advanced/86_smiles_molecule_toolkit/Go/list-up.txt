func fraction(n, d int) Fraction {
	g := gcd(n, d)
	return Fraction{n / g, d / g}
}
func (a Fraction) Cmp(b Fraction) int {
	left, right := a.N*b.D, b.N*a.D
	if left < right {
		return -1
	}
	if left > right {
		return 1
	}
	return 0
}
