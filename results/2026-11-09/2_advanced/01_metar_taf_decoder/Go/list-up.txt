func stateOf(s *Section) *Section {
	return &Section{wind: s.wind, visibility: s.visibility, weather: s.weather, cloud: s.cloud}
}
func overlay(s, c *Section) *Section {
	r := stateOf(s)
	if c.wind != nil {
		r.wind = c.wind
	}
	if c.visibility != nil {
		r.visibility = c.visibility
	}
	if c.weather != nil {
		r.weather = c.weather
	}
	if c.cloud != nil {
		r.cloud = c.cloud
	}
	return r
