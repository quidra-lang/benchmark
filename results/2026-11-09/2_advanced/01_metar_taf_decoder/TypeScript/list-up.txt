function stateOf(state: WeatherState): WeatherState {
  return {
    wind: state.wind,
    visibility: state.visibility,
    weather: state.weather,
    cloud: state.cloud,
  };
}
function overlay(state: WeatherState, change: WeatherState): WeatherState {
  return {
    wind: change.wind ?? state.wind,
    visibility: change.visibility ?? state.visibility,
    weather: change.weather ?? state.weather,
    cloud: change.cloud ?? state.cloud,
  };
