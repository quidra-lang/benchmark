overlay(s, c) = State((isnothing(getfield(c, k)) ? getfield(s, k) : getfield(c, k) for k in fieldnames(State))...)
function prevailing(f, h)
    s = f.base
    for c in f.changes
        if c.kind == "FM" && c.start <= h
            s = c.state
        elseif c.kind == "BECMG" && c.stop <= h
            s = overlay(s, c.state)
        end
    end
    s
