  def self.match_lines(text, c, groups)
    lines = ["match #{c[0]}-#{c[1]} #{Formatter.quote(capture(text, c, 0))}"]
    (1..groups).each do |k|
      value = capture(text, c, k)
      lines << (value ? "group #{k} #{c[2 * k]}-#{c[2 * k + 1]} #{Formatter.quote(value)}" : "group #{k} -")
    end
    lines
  end
