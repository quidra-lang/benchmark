string span(const int[] capture, int number, const int[] subject)
    start := capture[2 * number]
    finish := capture[2 * number + 1]
    return "{start}-{finish} " + format.quote(subject[start:<finish])

string[] match_lines(const int[] capture, const int[] subject, int groups)
    lines := ["match " + span(capture, 0, subject)]
    for: 1 <= number <= groups
        lines = lines.append("group {number} " + (if group_set(capture, number) then span(capture, number, subject) else "-"))
    return lines
