    public sealed class Model {
        public string name;
        public char[] states, symbols;
        public char missing;
        public double[] pi;
        public double[][] a, b;
        public Model(string name, char[] states, char[] symbols, char missing, double[] pi, double[][] a,
                     double[][] b) {
            this.name = name;
            this.states = states;
            this.symbols = symbols;
            this.missing = missing;
            this.pi = Normalize(pi);
            this.a = new double [a.Length][];
            this.b = new double [b.Length][];
            for (int i = 0; i < a.Length; i++)
                this.a[i] = Normalize(a[i]);
            for (int i = 0; i < b.Length; i++)
                this.b[i] = Normalize(b[i]);
        }
    public static double[] Normalize(double[] values) {
        double sum = 0;
        foreach (double value in values)
            sum += value;
        double[] result = new double[values.Length];
        for (int i = 0; i < values.Length; i++)
            result[i] = values[i] / sum;
        return result;
    }
    public static List<string> Lines(string name) {
        return File.ReadAllLines(Path.Combine("input", name), Encoding.ASCII).ToList();
    }
    public static double[] Numbers(string[] words, int start) {
        double[] result = new double[words.Length - start];
        for (int i = start; i < words.Length; i++)
            result[i - start] = double.Parse(words[i]);
        return result;
    }
    public static char[] Letters(string line) {
        string[] w = line.Split(' ');
        char[] result = new char[w.Length - 1];
        for (int i = 1; i < w.Length; i++)
            result[i - 1] = w[i][0];
        return result;
    }
    public static Data Read() {
        List<string> input = Lines("model.txt");
        char[] states = Letters(input[0]), symbols = Letters(input[1]);
        char missing = input[2].Split(' ') [1][0];
        int k = states.Length;
        double[] pi = Numbers(input[3].Split(' '), 1);
        double[][] a = new double [k][], b = new double [k][];
        for (int i = 0; i < k; i++) {
            a[i] = Numbers(input[4 + i].Split(' '), 2);
            b[i] = Numbers(input[4 + k + i].Split(' '), 2);
        }
        Dictionary<string, string[]> settings = new();
        foreach (string line in Lines("params.txt")) {
            string[] words = line.Split(' ');
            settings[words[0]] = words.Skip(1).Take(words.Length - (1)).ToArray();
        }
        return new Data(new Model("INITIAL", states, symbols, missing, pi, a, b), Lines("sequences.txt"),
                        Lines("labeled.txt"), settings);
    }
    public static void Write(string name, List<string> lines) {
        File.WriteAllText(Path.Combine("output", name), string.Join("\n", lines) + "\n", Encoding.ASCII);
    }

    public static int Maximum(double[] values) {
        int best = 0;
        for (int i = 1; i < values.Length; i++)
            if (values[i] > values[best])
                best = i;
        return best;
    }
    public static double Emission(ModelIo.Model model, int observed, int state) {
        return observed < 0 ? 1 : model.b[state][observed];
    }
    public static Forward ScaledForward(ModelIo.Model model, int[] observations) {
        int t = observations.Length, k = model.states.Length;
        double[][] values = Enumerable.Range(0, t)
                                .Select(
                                    _ => new double[k])
                                .ToArray(),
                   emissions = Enumerable.Range(0, t)
                                   .Select(
                                       _ => new double[k])
                                   .ToArray();
        double[] scales = new double[t];
        double loglik = 0;
        for (int p = 0; p < t; p++) {
            for (int j = 0; j < k; j++) {
                emissions[p][j] = Emission(model, observations[p], j);
                double value = 0;
                if (p == 0)
                    value = model.pi[j];
                else
                    for (int i = 0; i < k; i++)
                        value += values[p - 1][i] * model.a[i][j];
                values[p][j] = value * emissions[p][j];
                scales[p] += values[p][j];
            }
            for (int j = 0; j < k; j++)
                values[p][j] /= scales[p];
            loglik += Math.Log(scales[p]);
        }
        return new Forward(values, scales, emissions, loglik);
    }
    public static Fb Backward(ModelIo.Model model, int[] observations) {
        Forward f = ScaledForward(model, observations);
        int t = observations.Length, k = model.states.Length;
        double[][] g = Enumerable.Range(0, t)
                           .Select(
                               _ => new double[k])
                           .ToArray(),
                   gamma = Enumerable.Range(0, t)
                               .Select(
                                   _ => new double[k])
                               .ToArray();
        Array.Fill(g[t - 1], 1);
        for (int p = t - 2; p >= 0; p--)
            for (int i = 0; i < k; i++) {
                double sum = 0;
                for (int j = 0; j < k; j++)
                    sum += model.a[i][j] * f.emissions[p + 1][j] * g[p + 1][j];
                g[p][i] = sum / f.scales[p + 1];
            }
        for (int p = 0; p < t; p++)
            for (int i = 0; i < k; i++)
                gamma[p][i] = f.values[p][i] * g[p][i];
        return new Fb(f.values, g, f.scales, f.emissions, gamma, f.loglik);
    }
    public static Path Viterbi(ModelIo.Model model, int[] observations) {
        int t = observations.Length, k = model.states.Length;
        double[][] logA = Enumerable.Range(0, k)
                              .Select(
                                  _ => new double[k])
                              .ToArray();
        for (int i = 0; i < k; i++)
            for (int j = 0; j < k; j++)
                logA[i][j] = Math.Log(model.a[i][j]);
        double[] scores = new double[k];
        int[][] previous = Enumerable.Range(0, t)
                               .Select(
                                   _ => new int[k])
                               .ToArray();
        for (int j = 0; j < k; j++)
            scores[j] =
                Math.Log(model.pi[j]) + (observations[0] < 0 ? 0 : Math.Log(model.b[j][observations[0]]));
        for (int p = 1; p < t; p++) {
            double[] next = new double[k];
            for (int j = 0; j < k; j++) {
                double[] candidates = new double[k];
                for (int i = 0; i < k; i++)
                    candidates[i] = scores[i] + logA[i][j];
                int best = Maximum(candidates);
                previous[p][j] = best;
                next[j] =
                    candidates[best] + (observations[p] < 0 ? 0 : Math.Log(model.b[j][observations[p]]));
            }
            scores = next;
        }
        int[] path = new int[t];
        path[t - 1] = Maximum(scores);
        double logprob = scores[path[t - 1]];
        for (int p = t - 1; p > 0; p--)
            path[p - 1] = previous[p][path[p]];
        return new Path(path, logprob);
    }
    public static Decoding Decode(ModelIo.Model model, int[] observations) {
        Fb fb = Backward(model, observations);
        Path path = Viterbi(model, observations);
        int[] posterior = new int[observations.Length];
        double[] confidence = new double[observations.Length];
        for (int t = 0; t < observations.Length; t++) {
            posterior[t] = Maximum(fb.gamma[t]);
            confidence[t] = fb.gamma[t][posterior[t]];
        }
        return new Decoding(path, posterior, confidence, fb);
    }
    public static void Accumulate(int[][] matrix, int[] truth, int[] prediction) {
        for (int i = 0; i < truth.Length; i++)
            matrix[truth[i]][prediction[i]]++;
    }
    public static int Correct(int[] a, int[] b) {
        int count = 0;
        for (int i = 0; i < a.Length; i++)
            if (a[i] == b[i])
                count++;
        return count;
    }

    public static Inference.Decoding[][] Decode(List<Sequence> records) {
        Inference.Decoding[][] result = Enumerable.Range(0, 4)
                                            .Select(
                                                _ => new Inference.Decoding[records.Count])
                                            .ToArray();
        for (int m = 0; m < 4; m++)
            for (int i = 0; i < records.Count; i++)
                result[m][i] = Inference.Decode(models[m], records[i].observations);
        return result;
    }
    public static int[] Path(Inference.Decoding decoding, int decoder) {
        return decoder == 0 ? decoding.viterbi.states : decoding.posterior;
    }
    public static List<string> EvaluationReport() {
        List<string> output = new();
        int k = initial.states.Length;
        for (int m = 0; m < 4; m++)
            for (int decoder = 0; decoder < 2; decoder++) {
                Block(output,
                      "EVALUATION " + models[m].name + " " + (decoder == 0 ? "VITERBI" : "POSTERIOR"));
                int[][] matrix = Enumerable.Range(0, k)
                                     .Select(
                                         _ => new int[k])
                                     .ToArray();
                int[] regions = new int[4], boundaries = new int[4], bins = new int[10],
                      binCorrect = new int[10], recordCorrect = new int[test.Count];
                for (int r = 0; r < test.Count; r++) {
                    Sequence record = test[r];
                    Inference.Decoding decoding = testDecoded[m][r];
                    int[] prediction = Path(decoding, decoder);
                    recordCorrect[r] = Evaluation.Correct(record.truth, prediction);
                    testCorrect[m][decoder] += recordCorrect[r];
                    Evaluation.Accumulate(matrix, record.truth, prediction);
                    int[] rc = Evaluation.Regions(Segments(prediction), TruthRegions(record.truth)),
                          bc = Evaluation.BoundaryMatches(prediction, record.truth,
                                                          Integer("boundary_tolerance"));
                    for (int j = 0; j < 4; j++) {
                        regions[j] += rc[j];
                        boundaries[j] += bc[j];
                    }
                    if (decoder == 1)
                        for (int t = 0; t < prediction.Length; t++) {
                            int bin = Math.Min(9, (int)Math.Floor(10 * decoding.confidence[t]));
                            bins[bin]++;
                            if (prediction[t] == record.truth[t])
                                binCorrect[bin]++;
                        }
                }
                output.AddRange(Evaluation.Matrix(initial.states, matrix));
                output.AddRange(Evaluation.Metrics(initial.states, matrix));
                output.Add("REGIONS " + initial.states[segmentState] + " predicted " + regions[0] + " true " +
                           regions[1] + " precision " + Evaluation.Fraction(regions[2], regions[0]) +
                           " recall " + Evaluation.Fraction(regions[3], regions[1]));
                output.Add("BOUNDARIES predicted " + boundaries[0] + " true " + boundaries[1] +
                           " precision " + Evaluation.Fraction(boundaries[2], boundaries[0]) + " recall " +
                           Evaluation.Fraction(boundaries[3], boundaries[1]));
                if (decoder == 1)
                    for (int bin = 0; bin < 10; bin++)
                        if (bins[bin] > 0)
                            output.Add("CALIBRATION 0." + bin + "-" + (bin == 9 ? "1.0" : "0." + (bin + 1)) +
                                       " positions " + bins[bin] + " correct " + binCorrect[bin] +
                                       " accuracy " + Evaluation.Fraction(binCorrect[bin], bins[bin]));
                for (int r = 0; r < test.Count; r++)
                    output.Add("RECORD " + test[r].name + " length " + test[r].observations.Length +
                               " correct " + recordCorrect[r]);
            }

        ModelIo.Data data = ModelIo.Read();

        ModelIo.Write("evaluation.txt", evaluation);
