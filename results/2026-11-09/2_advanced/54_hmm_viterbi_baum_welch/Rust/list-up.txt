pub fn normalize(values: &mut [f64]) {
    let sum: f64 = values.iter().sum();
    for value in values { *value /= sum; }
}

pub fn read() -> io::Result<Input> {
    let text = fs::read_to_string("input/model.txt")?;
    let q: Vec<Vec<&str>> = text.lines().map(|s| s.split_whitespace().collect()).collect();
    let states: Vec<char> = q[0][1..].iter().map(|s| s.chars().next().unwrap()).collect();
    let symbols = q[1][1..].iter().map(|s| s.chars().next().unwrap()).collect();
    let missing = q[2][1].chars().next().unwrap();
    let k = states.len();
    let row = |values: &[&str]| {
        let mut values: Vec<f64> = values.iter().map(|s| s.parse().unwrap()).collect();
        normalize(&mut values);
        values
    };
    let model = Model { name: "INITIAL".into(), initial: row(&q[3][1..]), transition: (0..k).map(|i| row(&q[4 + i][2..])).collect(),
        emission: (0..k).map(|i| row(&q[4 + k + i][2..])).collect() };
    let unlabeled = fs::read_to_string("input/sequences.txt")?.lines().enumerate().map(|(i, l)| {
        let v: Vec<&str> = l.split_whitespace().collect();
        RawRecord { name: v[0].into(), role: String::new(), text: v[1].into(), labels: None, line: i + 1 }
    }).collect();
    let content = fs::read_to_string("input/labeled.txt")?;
    let lines: Vec<&str> = content.lines().collect();
    let labeled = lines.chunks(2).enumerate().map(|(i, lines)| {
        let v: Vec<&str> = lines[0].split_whitespace().collect();
        RawRecord { name: v[0].into(), role: v[1].into(), text: v[2].into(), labels: Some(lines[1].into()), line: 2 * i + 1 }
    }).collect();
    let params = fs::read_to_string("input/params.txt")?.lines().map(|s| s.split_whitespace().skip(1).map(str::to_owned).collect()).collect();
    Ok(Input { model, states, symbols, missing, unlabeled, labeled, params })
}
pub fn maximum(values: &[f64]) -> usize {
    let mut best = 0;
    for i in 1..values.len() { if values[i] > values[best] { best = i; } }
    best
}

pub fn emission(model: &Model, observation: Option<usize>, state: usize) -> f64 {
    observation.map_or(1.0, |c| model.emission[state][c])
}

pub struct ForwardBackward {
    pub forward: Vec<Vec<f64>>,
    pub backward: Vec<Vec<f64>>,
    pub gamma: Vec<Vec<f64>>,
    pub scales: Vec<f64>,
    pub loglik: f64,
}

pub fn forward_backward(model: &Model, observation: &Observation) -> ForwardBackward {
    let t = observation.len();
    let k = model.initial.len();
    let mut forward = vec![vec![0.0; k]; t];
    let mut scales = vec![0.0; t];
    for position in 0..t {
        for j in 0..k {
            let sum = if position == 0 { model.initial[j] } else { (0..k).map(|i| forward[position - 1][i] * model.transition[i][j]).sum() };
            forward[position][j] = sum * emission(model, observation[position], j);
        }
        scales[position] = forward[position].iter().sum();
        for value in &mut forward[position] { *value /= scales[position]; }
    }
    let mut backward = vec![vec![1.0; k]; t];
    for position in (0..t - 1).rev() {
        for i in 0..k {
            let sum: f64 = (0..k).map(|j| model.transition[i][j] * emission(model, observation[position + 1], j) * backward[position + 1][j]).sum();
            backward[position][i] = sum / scales[position + 1];
        }
    }
    let gamma = (0..t).map(|position| (0..k).map(|i| forward[position][i] * backward[position][i]).collect()).collect();
    let loglik = scales.iter().map(|s| s.ln()).sum();
    ForwardBackward { forward, backward, gamma, scales, loglik }
}

pub struct Viterbi {
    pub path: Vec<usize>,
    pub logprob: f64,
}

pub fn viterbi(model: &Model, observation: &Observation) -> Viterbi {
    let t = observation.len();
    let k = model.initial.len();
    let log_emission = |p: usize, j: usize| observation[p].map_or(0.0, |c| model.emission[j][c].ln());
    let mut delta: Vec<f64> = (0..k).map(|j| model.initial[j].ln() + log_emission(0, j)).collect();
    let mut previous = vec![vec![0; k]; t];
    for position in 1..t {
        let mut next = vec![0.0; k];
        for j in 0..k {
            let values: Vec<f64> = (0..k).map(|i| delta[i] + model.transition[i][j].ln()).collect();
            let best = maximum(&values);
            previous[position][j] = best;
            next[j] = values[best] + log_emission(position, j);
        }
        delta = next;
    }
    let mut path = vec![0; t];
    path[t - 1] = maximum(&delta);
    let logprob = delta[path[t - 1]];
    for position in (1..t).rev() { path[position - 1] = previous[position][path[position]]; }
    Viterbi { path, logprob }
}

pub struct Decoding {
    pub fb: ForwardBackward,
    pub viterbi: Viterbi,
    pub posterior: Vec<usize>,
    pub confidence: Vec<f64>,
}

pub fn decode(model: &Model, observation: &Observation) -> Decoding {
    let fb = forward_backward(model, observation);
    let posterior: Vec<usize> = fb.gamma.iter().map(|g| maximum(g)).collect();
    let confidence = fb.gamma.iter().zip(&posterior).map(|(g, &j)| g[j]).collect();
    Decoding { fb, viterbi: viterbi(model, observation), posterior, confidence }
}
pub fn accumulate(table: &mut Confusion, truth: &[usize], predicted: &[usize]) {
    for (&a, &b) in truth.iter().zip(predicted) { table[a][b] += 1; }
}
fn validate(raw: &[model_io::RawRecord], file: &str, symbols: &[char], states: &[char], missing: char, params: &Params) -> (Vec<Record>, Vec<String>) {
    let mut names = BTreeSet::new();
    let mut records = Vec::new();
    let mut rejected = Vec::new();
    for r in raw {
        let text: Vec<char> = r.text.to_ascii_uppercase().chars().collect();
        let labels: Vec<char> = r.labels.as_ref().map_or(Vec::new(), |s| s.chars().collect());
        let error = if !names.insert(r.name.clone()) { Some("duplicate name".into()) }
            else if let Some(i) = text.iter().position(|&c| c != missing && !symbols.contains(&c)) { Some(format!("invalid symbol {} at position {}", text[i], i + 1)) }
            else if let Some(i) = labels.iter().position(|c| !states.contains(c)) { Some(format!("invalid label {} at position {}", labels[i], i + 1)) }
            else if r.labels.is_some() && text.len() != labels.len() { Some(format!("length mismatch {} {}", text.len(), labels.len())) }
            else if text.len() < params.minimum_sequence { Some(format!("too short {}", text.len())) }
            else {
                let count = text.iter().filter(|&&c| c == missing).count();
                if 100 * count > params.missing_percent * text.len() { Some(format!("too many missing {}", count)) } else { None }
            };
        if let Some(error) = error {
            rejected.push(format!("REJECT {} line {} {} {}", file, r.line, r.name, error));
        } else {
            records.push(Record { name: r.name.clone(), role: r.role.clone(), observation: text.iter().map(|c| symbols.iter().position(|s| s == c)).collect(),
                truth: labels.iter().map(|c| states.iter().position(|s| s == c).unwrap()).collect() });
        }
    }
    (records, rejected)
}
fn path_text(path: &[usize], names: &[char]) -> String {
    path.iter().map(|&i| names[i]).collect()
}

fn correct(first: &[usize], second: &[usize]) -> usize {
    first.iter().zip(second).filter(|(a, b)| a == b).count()
}
fn paths<'a>(decode: &'a Decoding, decoder: usize) -> &'a [usize] {
    if decoder == 0 { &decode.viterbi.path } else { &decode.posterior }
}
