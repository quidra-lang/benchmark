  def ordered_sum(values)
    total = 0
    values.each { |value| total += value }
    total
  end

  def normalize(values)
    total = ordered_sum(values)
    values.map { |value| value.to_f / total }
  end

  class Model
    attr_reader :name, :states, :symbols, :missing, :initial, :transition, :emission

    def initialize(name, states, symbols, missing, initial, transition, emission)
      @name, @states, @symbols, @missing = name, states, symbols, missing
      @initial = HMM.normalize(initial)
      @transition = transition.map { |row| HMM.normalize(row) }
      @emission = emission.map { |row| HMM.normalize(row) }
    end
  end

  def read_tokens(filename)
    File.readlines(File.join('input', filename), chomp: true).map { |line| line.split(' ') }
  end

  def read_inputs
    rows = read_tokens('model.txt')
    states, symbols, missing = rows[0].drop(1), rows[1].drop(1), rows[2][1]
    k = states.length
    model = Model.new('INITIAL', states, symbols, missing, rows[3].drop(1).map(&:to_f),
                      rows[4, k].map { |row| row.drop(2).map(&:to_f) },
                      rows[4 + k, k].map { |row| row.drop(2).map(&:to_f) })
    [model, read_tokens('sequences.txt'), read_tokens('labeled.txt'), read_tokens('params.txt')]
  end
  Forward = Struct.new(:values, :scales, :emissions, :loglik)
  ForwardBackward = Struct.new(:forward, :backward, :scales, :emissions, :posterior, :loglik)
  Decoding = Struct.new(:viterbi, :viterbi_logprob, :posterior, :confidence, :fb)

  def maximum_index(values)
    best = 0
    (1...values.length).each { |i| best = i if values[i] > values[best] }
    best
  end

  def emission_factors(model, observation)
    model.emission.map { |row| observation == -1 ? 1.0 : row[observation] }
  end

  def log_emission_factors(model, observation)
    model.emission.map { |row| observation == -1 ? 0.0 : Math.log(row[observation]) }
  end

  def scaled_forward(model, observations)
    k = model.states.length
    emissions = observations.map { |c| emission_factors(model, c) }
    values, scales, loglik = [], [], 0.0
    emissions.each_with_index do |factors, t|
      unscaled = Array.new(k) do |j|
        if t.zero?
          model.initial[j] * factors[j]
        else
          total = 0.0
          k.times { |i| total += values[t - 1][i] * model.transition[i][j] }
          total * factors[j]
        end
      end
      scale = ordered_sum(unscaled)
      scales << scale
      values << unscaled.map { |value| value / scale }
      loglik += Math.log(scale)
    end
    Forward.new(values, scales, emissions, loglik)
  end

  def forward_backward(model, observations)
    f = scaled_forward(model, observations)
    k, length = model.states.length, observations.length
    backward = Array.new(length) { Array.new(k, 0.0) }
    backward[-1] = Array.new(k, 1.0)
    (length - 2).downto(0) do |t|
      k.times do |i|
        total = 0.0
        k.times { |j| total += model.transition[i][j] * f.emissions[t + 1][j] * backward[t + 1][j] }
        backward[t][i] = total / f.scales[t + 1]
      end
    end
    posterior = Array.new(length) { |t| Array.new(k) { |i| f.values[t][i] * backward[t][i] } }
    ForwardBackward.new(f.values, backward, f.scales, f.emissions, posterior, f.loglik)
  end
  def viterbi(model, observations)
    k = model.states.length
    transitions = model.transition.map { |row| row.map { |v| Math.log(v) } }
    factors = log_emission_factors(model, observations[0])
    scores = Array.new(k) { |i| Math.log(model.initial[i]) + factors[i] }
    predecessors = [Array.new(k, 0)]
    observations.drop(1).each do |c|
      factors = log_emission_factors(model, c)
      previous_states = []
      next_scores = Array.new(k) do |j|
        candidates = Array.new(k) { |i| scores[i] + transitions[i][j] }
        previous = maximum_index(candidates)
        previous_states << previous
        candidates[previous] + factors[j]
      end
      predecessors << previous_states
      scores = next_scores
    end
    path = Array.new(observations.length, 0)
    path[-1] = maximum_index(scores)
    logprob = scores[path[-1]]
    (path.length - 1).downto(1) { |t| path[t - 1] = predecessors[t][path[t]] }
    [path, logprob]
  end

  def decode(model, observations)
    fb = forward_backward(model, observations)
    path, logprob = viterbi(model, observations)
    posterior = fb.posterior.map { |row| maximum_index(row) }
    confidence = fb.posterior.each_with_index.map { |row, t| row[posterior[t]] }
    Decoding.new(path, logprob, posterior, confidence, fb)
  end
  def confusion_matrix(k)
    Array.new(k) { Array.new(k, 0) }
  end

  def accumulate_confusion(matrix, truth, prediction)
    truth.zip(prediction).each { |actual, predicted| matrix[actual][predicted] += 1 }
  end

  def correct_count(truth, prediction)
    truth.zip(prediction).count { |actual, predicted| actual == predicted }
  end

  def evaluate_model(model, records, p)
    decoded = records.map { |r| decode(model, r.observations) }
    blocks, totals = [], []
    %w[VITERBI POSTERIOR].each do |decoder|
      matrix = confusion_matrix(model.states.length)
      regions, bounds = Array.new(4, 0), Array.new(4, 0)
      calibration = Array.new(10) { [0, 0] }
      record_lines, total = [], 0
      records.zip(decoded).each do |record, result|
        path = decoder == 'VITERBI' ? result.viterbi : result.posterior
        correct = correct_count(record.path, path)
        total += correct
        accumulate_confusion(matrix, record.path, path)
        truth = runs(record.path).select { |state, _, _| state == p[:segment_state] }.map { |_, s, e| [s, e] }
        rc = region_counts(extract_segments(path, p), truth)
        bc = boundary_counts(path, record.path, p[:boundary_tolerance])
        4.times { |i| regions[i] += rc[i]; bounds[i] += bc[i] }
        if decoder == 'POSTERIOR'
          result.confidence.each_with_index do |confidence, t|
            b = [(10 * confidence).floor, 9].min
            calibration[b][0] += 1
            calibration[b][1] += 1 if path[t] == record.path[t]
          end
        end
        record_lines << "RECORD #{record.name} length #{path.length} correct #{correct}"
      end
      lines = ["EVALUATION #{model.name} #{decoder}"] + matrix_lines(model.states, matrix) + metric_lines(model.states, matrix)
      lines << "REGIONS #{model.states[p[:segment_state]]} predicted #{regions[0]} true #{regions[1]} precision #{fraction(regions[2], regions[0])} recall #{fraction(regions[3], regions[1])}"
      lines << "BOUNDARIES predicted #{bounds[0]} true #{bounds[1]} precision #{fraction(bounds[2], bounds[0])} recall #{fraction(bounds[3], bounds[1])}"
      if decoder == 'POSTERIOR'
        calibration.each_with_index do |(n, correct), b|
          next if n.zero?
          interval = b < 9 ? "0.#{b}-0.#{b + 1}" : '0.9-1.0'
          lines << "CALIBRATION #{interval} positions #{n} correct #{correct} accuracy #{fraction(correct, n)}"
        end
      end
      blocks << lines + record_lines
      totals << total
    end
    [blocks, totals]
  end

  def state_string(model, path)
    path.map { |i| model.states[i] }.join
  end
