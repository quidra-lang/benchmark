    static final class Model {
        String name;
        char[] states, symbols;
        char missing;
        double[] pi;
        double[][] a, b;
        Model(String name, char[] states, char[] symbols, char missing, double[] pi, double[][] a,
              double[][] b) {
            this.name = name;
            this.states = states;
            this.symbols = symbols;
            this.missing = missing;
            this.pi = normalize(pi);
            this.a = new double[a.length][];
            this.b = new double[b.length][];
            for (int i = 0; i < a.length; i++)
                this.a[i] = normalize(a[i]);
            for (int i = 0; i < b.length; i++)
                this.b[i] = normalize(b[i]);
        }
    }
    static double[] normalize(double[] values) {
        double sum = 0;
        for (double value : values)
            sum += value;
        double[] result = new double[values.length];
        for (int i = 0; i < values.length; i++)
            result[i] = values[i] / sum;
        return result;
    }
    static List<String> lines(String name) throws IOException {
        return Files.readAllLines(Path.of("input", name), StandardCharsets.US_ASCII);
    }
    static double[] numbers(String[] words, int start) {
        double[] result = new double[words.length - start];
        for (int i = start; i < words.length; i++)
            result[i - start] = Double.parseDouble(words[i]);
        return result;
    }
    static char[] letters(String line) {
        String[] w = line.split(" ");
        char[] result = new char[w.length - 1];
        for (int i = 1; i < w.length; i++)
            result[i - 1] = w[i].charAt(0);
        return result;
    }
    static Data read() throws IOException {
        List<String> input = lines("model.txt");
        char[] states = letters(input.get(0)), symbols = letters(input.get(1));
        char missing = input.get(2).split(" ")[1].charAt(0);
        int k = states.length;
        double[] pi = numbers(input.get(3).split(" "), 1);
        double[][] a = new double[k][], b = new double[k][];
        for (int i = 0; i < k; i++) {
            a[i] = numbers(input.get(4 + i).split(" "), 2);
            b[i] = numbers(input.get(4 + k + i).split(" "), 2);
        }
        Map<String, String[]> params = new HashMap<>();
        for (String line : lines("params.txt")) {
            String[] words = line.split(" ");
            params.put(words[0], Arrays.copyOfRange(words, 1, words.length));
        }
        return new Data(new Model("INITIAL", states, symbols, missing, pi, a, b), lines("sequences.txt"),
                        lines("labeled.txt"), params);
    }
    record Forward(double[][] values, double[] scales, double[][] emissions, double loglik) {}
    record Fb(double[][] forward, double[][] backward, double[] scales, double[][] emissions,
              double[][] gamma, double loglik) {}
    record Path(int[] states, double logprob) {}
    record Decoding(Path viterbi, int[] posterior, double[] confidence, Fb fb) {}
    static int maximum(double[] values) {
        int best = 0;
        for (int i = 1; i < values.length; i++)
            if (values[i] > values[best])
                best = i;
        return best;
    }
    static double emission(ModelIo.Model model, int observed, int state) {
        return observed < 0 ? 1 : model.b[state][observed];
    }
    static Forward forward(ModelIo.Model model, int[] observations) {
        int t = observations.length, k = model.states.length;
        double[][] values = new double[t][k], emissions = new double[t][k];
        double[] scales = new double[t];
        double loglik = 0;
        for (int p = 0; p < t; p++) {
            for (int j = 0; j < k; j++) {
                emissions[p][j] = emission(model, observations[p], j);
                double value = 0;
                if (p == 0)
                    value = model.pi[j];
                else
                    for (int i = 0; i < k; i++)
                        value += values[p - 1][i] * model.a[i][j];
                values[p][j] = value * emissions[p][j];
                scales[p] += values[p][j];
            }
            for (int j = 0; j < k; j++)
                values[p][j] /= scales[p];
            loglik += Math.log(scales[p]);
        }
        return new Forward(values, scales, emissions, loglik);
    }
    static Fb backward(ModelIo.Model model, int[] observations) {
        Forward f = forward(model, observations);
        int t = observations.length, k = model.states.length;
        double[][] g = new double[t][k], gamma = new double[t][k];
        Arrays.fill(g[t - 1], 1);
        for (int p = t - 2; p >= 0; p--)
            for (int i = 0; i < k; i++) {
                double sum = 0;
                for (int j = 0; j < k; j++)
                    sum += model.a[i][j] * f.emissions()[p + 1][j] * g[p + 1][j];
                g[p][i] = sum / f.scales()[p + 1];
            }
        for (int p = 0; p < t; p++)
            for (int i = 0; i < k; i++)
                gamma[p][i] = f.values()[p][i] * g[p][i];
        return new Fb(f.values(), g, f.scales(), f.emissions(), gamma, f.loglik());
    }
    static Path viterbi(ModelIo.Model model, int[] observations) {
        int t = observations.length, k = model.states.length;
        double[][] logA = new double[k][k];
        for (int i = 0; i < k; i++)
            for (int j = 0; j < k; j++)
                logA[i][j] = Math.log(model.a[i][j]);
        double[] scores = new double[k];
        int[][] previous = new int[t][k];
        for (int j = 0; j < k; j++)
            scores[j] =
                Math.log(model.pi[j]) + (observations[0] < 0 ? 0 : Math.log(model.b[j][observations[0]]));
        for (int p = 1; p < t; p++) {
            double[] next = new double[k];
            for (int j = 0; j < k; j++) {
                double[] candidates = new double[k];
                for (int i = 0; i < k; i++)
                    candidates[i] = scores[i] + logA[i][j];
                int best = maximum(candidates);
                previous[p][j] = best;
                next[j] =
                    candidates[best] + (observations[p] < 0 ? 0 : Math.log(model.b[j][observations[p]]));
            }
            scores = next;
        }
        int[] path = new int[t];
        path[t - 1] = maximum(scores);
        double logprob = scores[path[t - 1]];
        for (int p = t - 1; p > 0; p--)
            path[p - 1] = previous[p][path[p]];
        return new Path(path, logprob);
    }
    static Decoding decode(ModelIo.Model model, int[] observations) {
        Fb fb = backward(model, observations);
        Path path = viterbi(model, observations);
        int[] posterior = new int[observations.length];
        double[] confidence = new double[observations.length];
        for (int t = 0; t < observations.length; t++) {
            posterior[t] = maximum(fb.gamma()[t]);
            confidence[t] = fb.gamma()[t][posterior[t]];
        }
        return new Decoding(path, posterior, confidence, fb);
    }
    static void accumulate(int[][] matrix, int[] truth, int[] prediction) {
        for (int i = 0; i < truth.length; i++)
            matrix[truth[i]][prediction[i]]++;
    }
    static int correct(int[] a, int[] b) {
        int count = 0;
        for (int i = 0; i < a.length; i++)
            if (a[i] == b[i])
                count++;
        return count;
    }
    static int index(char[] values, char c) {
        for (int i = 0; i < values.length; i++)
            if (values[i] == c)
                return i;
        return -1;
    }

    static Inference.Decoding[][] decode(List<Sequence> records) {
        Inference.Decoding[][] result = new Inference.Decoding[4][records.size()];
        for (int m = 0; m < 4; m++)
            for (int i = 0; i < records.size(); i++)
                result[m][i] = Inference.decode(models[m], records.get(i).observations());
        return result;
    }
    static int[] path(Inference.Decoding decoding, int decoder) {
        return decoder == 0 ? decoding.viterbi().states() : decoding.posterior();
    }
    static String states(int[] path) {
        StringBuilder text = new StringBuilder();
        for (int state : path)
            text.append(initial.states[state]);
        return text.toString();
    }
    static List<String> evaluationReport() {
        List<String> out = new ArrayList<>();
        int k = initial.states.length;
        for (int m = 0; m < 4; m++)
            for (int decoder = 0; decoder < 2; decoder++) {
                block(out, "EVALUATION " + models[m].name + " " + (decoder == 0 ? "VITERBI" : "POSTERIOR"));
                int[][] matrix = new int[k][k];
                int[] regions = new int[4], boundaries = new int[4], bins = new int[10],
                      binCorrect = new int[10], recordCorrect = new int[test.size()];
                for (int r = 0; r < test.size(); r++) {
                    Sequence record = test.get(r);
                    Inference.Decoding decoding = testDecoded[m][r];
                    int[] prediction = path(decoding, decoder);
                    recordCorrect[r] = Evaluation.correct(record.truth(), prediction);
                    testCorrect[m][decoder] += recordCorrect[r];
                    Evaluation.accumulate(matrix, record.truth(), prediction);
                    int[] rc = Evaluation.regions(segments(prediction), truthRegions(record.truth())),
                          bc = Evaluation.boundaryMatches(prediction, record.truth(),
                                                          integer("boundary_tolerance"));
                    for (int j = 0; j < 4; j++) {
                        regions[j] += rc[j];
                        boundaries[j] += bc[j];
                    }
                    if (decoder == 1)
                        for (int t = 0; t < prediction.length; t++) {
                            int bin = Math.min(9, (int)Math.floor(10 * decoding.confidence()[t]));
                            bins[bin]++;
                            if (prediction[t] == record.truth()[t])
                                binCorrect[bin]++;
                        }
                }
                out.addAll(Evaluation.matrix(initial.states, matrix));
                out.addAll(Evaluation.metrics(initial.states, matrix));
                out.add("REGIONS " + initial.states[segmentState] + " predicted " + regions[0] + " true " +
                        regions[1] + " precision " + Evaluation.fraction(regions[2], regions[0]) +
                        " recall " + Evaluation.fraction(regions[3], regions[1]));
                out.add("BOUNDARIES predicted " + boundaries[0] + " true " + boundaries[1] + " precision " +
                        Evaluation.fraction(boundaries[2], boundaries[0]) + " recall " +
                        Evaluation.fraction(boundaries[3], boundaries[1]));
                if (decoder == 1)
                    for (int bin = 0; bin < 10; bin++)
                        if (bins[bin] > 0)
                            out.add("CALIBRATION 0." + bin + "-" + (bin == 9 ? "1.0" : "0." + (bin + 1)) +
                                    " positions " + bins[bin] + " correct " + binCorrect[bin] + " accuracy " +
                                    Evaluation.fraction(binCorrect[bin], bins[bin]));
                for (int r = 0; r < test.size(); r++)
                    out.add("RECORD " + test.get(r).name() + " length " + test.get(r).observations().length +
                            " correct " + recordCorrect[r]);
            }
