class Model {
  final String name;
  final List<String> states, symbols;
  final String missing;
  final List<double> initial;
  final List<List<double>> transition, emission;
  Model(
    this.name,
    this.states,
    this.symbols,
    this.missing,
    List<double> initial,
    List<List<double>> transition,
    List<List<double>> emission,
  ) : initial = normalize(initial),
      transition = transition.map(normalize).toList(),
      emission = emission.map(normalize).toList();
}
double sum(Iterable<double> values) {
  var total = 0.0;
  for (final value in values) {
    total += value;
  }
  return total;
}

List<double> normalize(List<double> values) {
  final total = sum(values);
  return [for (final value in values) value / total];
}

List<List<double>> matrix(int rows, int columns) =>
    List.generate(rows, (_) => List.filled(columns, 0.0));
List<String> readLines(String path) => File(path).readAsLinesSync();
RawInputs readInputs() {
  var lines = readLines('input/model.txt');
  final states = lines[0].split(' ').sublist(1),
      symbols = lines[1].split(' ').sublist(1),
      missing = lines[2].split(' ')[1],
      initial = lines[3].split(' ').skip(1).map(double.parse).toList(),
      transition = <List<double>>[],
      emission = <List<double>>[];
  for (var i = 0; i < states.length; i++) {
    transition.add(lines[4 + i].split(' ').skip(2).map(double.parse).toList());
    emission.add(
      lines[4 + states.length + i]
          .split(' ')
          .skip(2)
          .map(double.parse)
          .toList(),
    );
  }
  final model = Model(
        'INITIAL',
        states,
        symbols,
        missing,
        initial,
        transition,
        emission,
      ),
      sequences = <RawRecord>[],
      labeled = <RawRecord>[],
      params = <String, List<String>>{};
  lines = readLines('input/sequences.txt');
  for (var i = 0; i < lines.length; i++) {
    final f = lines[i].split(' ');
    sequences.add(RawRecord(f[0], f[1], i + 1));
  }
  lines = readLines('input/labeled.txt');
  for (var i = 0; i < lines.length; i += 2) {
    final f = lines[i].split(' ');
    labeled.add(RawRecord(f[0], f[2], i + 1, role: f[1], labels: lines[i + 1]));
  }
  for (final line in readLines('input/params.txt')) {
    final f = line.split(' ');
    params[f.first] = f.sublist(1);
  }
  return RawInputs(model, sequences, labeled, params);
}
int maximumIndex(List<double> values) {
  var best = 0;
  for (var i = 1; i < values.length; i++) {
    if (values[i] > values[best]) {
      best = i;
    }
  }
  return best;
}

List<double> emissionFactors(Model model, int observation, bool logarithm) => [
  for (var i = 0; i < model.states.length; i++)
    observation < 0
        ? (logarithm ? 0.0 : 1.0)
        : logarithm
        ? log(model.emission[i][observation])
        : model.emission[i][observation],
];

class Forward {
  final values = <List<double>>[],
      emissions = <List<double>>[],
      scales = <double>[];
  double loglik = 0;
}

Forward scaledForward(Model model, List<int> observations) {
  final result = Forward(), k = model.states.length;
  for (var t = 0; t < observations.length; t++) {
    final factors = emissionFactors(model, observations[t], false),
        values = List.filled(k, 0.0);
    result.emissions.add(factors);
    for (var state = 0; state < k; state++) {
      if (t == 0) {
        values[state] = model.initial[state] * factors[state];
      } else {
        var total = 0.0;
        for (var previous = 0; previous < k; previous++) {
          total +=
              result.values[t - 1][previous] *
              model.transition[previous][state];
        }
        values[state] = total * factors[state];
      }
    }
    final scale = sum(values);
    result.scales.add(scale);
    for (var i = 0; i < k; i++) {
      values[i] /= scale;
    }
    result.values.add(values);
    result.loglik += log(scale);
  }
  return result;
}

class ForwardBackward {
  final Forward forward;
  final List<List<double>> backward, posterior;
  ForwardBackward(this.forward, this.backward, this.posterior);
}

ForwardBackward forwardBackward(Model model, List<int> observations) {
  final forward = scaledForward(model, observations),
      k = model.states.length,
      length = observations.length,
      backward = matrix(length, k);
  for (var i = 0; i < k; i++) {
    backward[length - 1][i] = 1;
  }
  for (var t = length - 2; t >= 0; t--) {
    for (var i = 0; i < k; i++) {
      var total = 0.0;
      for (var j = 0; j < k; j++) {
        total +=
            model.transition[i][j] *
            forward.emissions[t + 1][j] *
            backward[t + 1][j];
      }
      backward[t][i] = total / forward.scales[t + 1];
    }
  }
  final posterior = [
    for (var t = 0; t < length; t++)
      [for (var i = 0; i < k; i++) forward.values[t][i] * backward[t][i]],
  ];
  return ForwardBackward(forward, backward, posterior);
}
(List<int>, double) viterbi(Model model, List<int> observations) {
  final k = model.states.length,
      transition = [for (final row in model.transition) row.map(log).toList()],
      predecessors = List.generate(
        observations.length,
        (_) => List.filled(model.states.length, 0),
      );
  var factors = emissionFactors(model, observations.first, true),
      scores = [for (var i = 0; i < k; i++) log(model.initial[i]) + factors[i]];
  for (var t = 1; t < observations.length; t++) {
    factors = emissionFactors(model, observations[t], true);
    final next = List.filled(k, 0.0);
    for (var state = 0; state < k; state++) {
      final candidates = [
            for (var previous = 0; previous < k; previous++)
              scores[previous] + transition[previous][state],
          ],
          previous = maximumIndex(candidates);
      predecessors[t][state] = previous;
      next[state] = candidates[previous] + factors[state];
    }
    scores = next;
  }
  final path = List.filled(observations.length, 0);
  path[path.length - 1] = maximumIndex(scores);
  final logprob = scores[path.last];
  for (var t = path.length - 1; t > 0; t--) {
    path[t - 1] = predecessors[t][path[t]];
  }
  return (path, logprob);
}

class Decoding {
  final List<int> viterbi, posterior;
  final double viterbiLogprob;
  final List<double> confidence;
  final ForwardBackward fb;
  Decoding(
    this.viterbi,
    this.posterior,
    this.viterbiLogprob,
    this.confidence,
    this.fb,
  );
}

Decoding decode(Model model, List<int> observations) {
  final fb = forwardBackward(model, observations);
  final (path, logprob) = viterbi(model, observations);
  final posterior = [for (final row in fb.posterior) maximumIndex(row)],
      confidence = [
        for (var t = 0; t < observations.length; t++)
          fb.posterior[t][posterior[t]],
      ];
  return Decoding(path, posterior, logprob, confidence, fb);
}
List<List<int>> intMatrix(int rows, int columns) =>
    List.generate(rows, (_) => List.filled(columns, 0));
int correctCount(List<int> a, List<int> b) {
  var count = 0;
  for (var i = 0; i < a.length; i++) {
    if (a[i] == b[i]) {
      count++;
    }
  }
  return count;
}

void addConfusion(
  List<List<int>> table,
  List<int> truth,
  List<int> prediction,
) {
  for (var i = 0; i < truth.length; i++) {
    table[truth[i]][prediction[i]]++;
  }
}
String encode(List<int> indices, List<String> names) =>
    indices.map((i) => names[i]).join();
(List<SequenceRecord>, List<String>) validate(
  List<RawRecord> raws,
  bool labeled,
  Model model,
  Params p,
) {
  final seen = <String>{}, accepted = <SequenceRecord>[], rejected = <String>[];
  for (final raw in raws) {
    var reason = seen.add(raw.name) ? '' : 'duplicate name';
    final upper = raw.observation.toUpperCase(),
        observations = <int>[],
        path = <int>[];
    var missing = 0;
    if (reason.isEmpty) {
      for (var i = 0; i < upper.length; i++) {
        final letter = upper[i], index = model.symbols.indexOf(letter);
        if (index < 0 && letter != model.missing) {
          reason = 'invalid symbol $letter at position ${i + 1}';
          break;
        }
        observations.add(index);
        if (index < 0) {
          missing++;
        }
      }
    }
    if (reason.isEmpty && labeled) {
      for (var i = 0; i < raw.labels.length; i++) {
        final letter = raw.labels[i], index = model.states.indexOf(letter);
        if (index < 0) {
          reason = 'invalid label $letter at position ${i + 1}';
          break;
        }
        path.add(index);
      }
    }
    if (reason.isEmpty && labeled && upper.length != raw.labels.length) {
      reason = 'length mismatch ${upper.length} ${raw.labels.length}';
    }
    if (reason.isEmpty && upper.length < p.minLength) {
      reason = 'too short ${upper.length}';
    }
    if (reason.isEmpty && 100 * missing > p.maxMissing * upper.length) {
      reason = 'too many missing $missing';
    }
    if (reason.isNotEmpty) {
      rejected.add(
        'REJECT ${labeled ? 'labeled.txt' : 'sequences.txt'} line ${raw.line} ${raw.name} $reason',
      );
    } else {
      accepted.add(
        SequenceRecord(raw.name, raw.role, upper, observations, path, missing),
      );
    }
  }
  return (accepted, rejected);
}

Validated validateInputs(RawInputs raw, Params p) {
  final (unlabeled, rejected1) = validate(raw.sequences, false, raw.initial, p);
  final (labeled, rejected2) = validate(raw.labeled, true, raw.initial, p);
  return Validated(
    unlabeled,
    labeled.where((r) => r.role == 'TRAIN').toList(),
    labeled.where((r) => r.role == 'TEST').toList(),
    [...rejected1, ...rejected2],
    rejected1.length,
    rejected2.length,
  );
}
class ModelEvaluation {
  final blocks = <List<String>>[],
      correct = <int>[0, 0],
      decodings = <Decoding>[];
}

ModelEvaluation evaluateModel(
  Model model,
  List<SequenceRecord> testing,
  Params p,
) {
  final result = ModelEvaluation();
  for (final record in testing) {
    result.decodings.add(decode(model, record.observations));
  }
  for (var decoder = 0; decoder < 2; decoder++) {
    final name = decoder == 0 ? 'VITERBI' : 'POSTERIOR',
        table = intMatrix(model.states.length, model.states.length),
        calibration = List.filled(10, 0),
        calibrationCorrect = List.filled(10, 0),
        recordLines = <String>[];
    var predictedRegions = 0,
        trueRegionCount = 0,
        matchedPredictedRegions = 0,
        matchedTrueRegions = 0,
        predictedBoundaries = 0,
        trueBoundaryCount = 0,
        matchedPredictedBoundaries = 0,
        matchedTrueBoundaries = 0;
    for (var i = 0; i < testing.length; i++) {
      final record = testing[i],
          d = result.decodings[i],
          path = decoder == 0 ? d.viterbi : d.posterior,
          correct = correctCount(record.path, path);
      result.correct[decoder] += correct;
      addConfusion(table, record.path, path);
      final predicted = segments(path, p),
          truth = trueRegions(record.path, p.target);
      final (a, b) = regionMatches(predicted, truth);
      predictedRegions += predicted.length;
      trueRegionCount += truth.length;
      matchedPredictedRegions += a;
      matchedTrueRegions += b;
      final predictedB = boundaries(path), truthB = boundaries(record.path);
      final (ba, bb) = boundaryMatches(predictedB, truthB, p.boundaryTolerance);
      predictedBoundaries += predictedB.length;
      trueBoundaryCount += truthB.length;
      matchedPredictedBoundaries += ba;
      matchedTrueBoundaries += bb;
      if (decoder == 1) {
        for (var t = 0; t < d.confidence.length; t++) {
          final bin = min(9, (10 * d.confidence[t]).floor());
          calibration[bin]++;
          if (path[t] == record.path[t]) {
            calibrationCorrect[bin]++;
          }
        }
      }
      recordLines.add(
        'RECORD ${record.name} length ${record.path.length} correct $correct',
      );
    }
    final lines = <String>[
      'EVALUATION ${model.name} $name',
      ...confusionLines(table, model.states),
      ...metricsLines(table, model.states),
      'REGIONS ${model.states[p.target]} predicted $predictedRegions true $trueRegionCount precision ${fraction(matchedPredictedRegions, predictedRegions)} recall ${fraction(matchedTrueRegions, trueRegionCount)}',
      'BOUNDARIES predicted $predictedBoundaries true $trueBoundaryCount precision ${fraction(matchedPredictedBoundaries, predictedBoundaries)} recall ${fraction(matchedTrueBoundaries, trueBoundaryCount)}',
    ];
    if (decoder == 1) {
      for (var bin = 0; bin < 10; bin++) {
        final count = calibration[bin];
        if (count == 0) {
          continue;
        }
        final interval = bin == 9 ? '0.9-1.0' : '0.$bin-0.${bin + 1}';
        lines.add(
          'CALIBRATION $interval positions $count correct ${calibrationCorrect[bin]} accuracy ${fraction(calibrationCorrect[bin], count)}',
        );
      }
    }
    lines.addAll(recordLines);
    result.blocks.add(lines);
  }
  return result;
}
(List<String>, List<String>) annotationFiles(
  List<SequenceRecord> records,
  Model model,
  List<Decoding> decodings,
  Params p,
) {
  final decodingLines = <String>[],
      segmentLines = <String>[
        'sequence,decoder,start,end,length,mean_posterior,support,rich_fraction,motif_ratio,island',
      ],
      rng = Minstd(p.posteriorSeed);
  for (var i = 0; i < records.length; i++) {
    final record = records[i], d = decodings[i], samplePaths = <List<int>>[];
    decodingLines.addAll([
      'SEQUENCE ${record.name} length ${record.observations.length} loglik ${number(d.fb.forward.loglik)} viterbi ${number(d.viterbiLogprob)} differences $differences uncertain $uncertain expected_segments ${number(totalSegments / p.posteriorSamples)}',
      'VITERBI ${encode(d.viterbi, model.states)}',
      'POSTERIOR ${encode(d.posterior, model.states)}',
      'CONFIDENT ${confident.join()}',
      'IMPUTED ${imputed.join()}',
    ]);
  final raw = readInputs(),
      p = Params(raw.params, raw.initial),
      input = validateInputs(raw, p),
      models = <Model>[raw.initial, bw.model, vt.model, supervised.model],
      evaluations = <ModelEvaluation>[],
      allDecodings = <List<Decoding>>[];
  for (final model in models) {
    evaluations.add(evaluateModel(model, input.test, p));
    allDecodings.add([
      for (final record in input.unlabeled) decode(model, record.observations),
    ]);
  }
  final (decodingLines, segmentLines) = annotationFiles(
    input.unlabeled,
    bw.model,
    allDecodings[1],
    p,
  );
  writeLines('decodings.txt', decodingLines);
  writeLines(
    'evaluation.txt',
    evaluationFile(models, evaluations, supervised, input.train, input.test, p),
  );
