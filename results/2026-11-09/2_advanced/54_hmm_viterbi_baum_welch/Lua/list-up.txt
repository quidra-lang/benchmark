function M.normalize(values)
    local total, result = 0.0, {}
    for _, value in ipairs(values) do total = total + value end
    for i, value in ipairs(values) do result[i] = value / total end
    return result
end
function M.model(name, template, initial, transition, emission)
    local result = {name = name, states = template.states, symbols = template.symbols, missing = template.missing,
        initial = M.normalize(initial), transition = {}, emission = {}}
    for i, row in ipairs(transition) do result.transition[i] = M.normalize(row) end
    for i, row in ipairs(emission) do result.emission[i] = M.normalize(row) end
    return result
end
local function read_tokens(name)
    local input = assert(io.open("input/" .. name, "r"))
    local result = {}
    for line in input:lines() do
        local fields = {}
        for word in line:gmatch("%S+") do fields[#fields + 1] = word end
        result[#result + 1] = fields
    end
    input:close()
    return result
end
local function numbers(row, start)
    local result = {}
    for i = start, #row do result[#result + 1] = tonumber(row[i]) end
    return result
end
function M.read()
    local rows = read_tokens("model.txt")
    local states, symbols = {}, {}
    for i = 2, #rows[1] do states[#states + 1] = rows[1][i] end
    for i = 2, #rows[2] do symbols[#symbols + 1] = rows[2][i] end
    local template = {states = states, symbols = symbols, missing = rows[3][2]}
    local transition, emission = {}, {}
    for i = 1, #states do
        transition[i] = numbers(rows[4 + i], 3)
        emission[i] = numbers(rows[4 + #states + i], 3)
    end
    return M.model("INITIAL", template, numbers(rows[4], 2), transition, emission),
        read_tokens("sequences.txt"), read_tokens("labeled.txt"), read_tokens("params.txt")
end
function M.maximum(values)
    local best = 1
    for i = 2, #values do if values[i] > values[best] then best = i end end
    return best
end
function M.emissions(model, observation, logarithms)
    local result = {}
    for state, row in ipairs(model.emission) do
        local value = observation == -1 and 1.0 or row[observation]
        result[state] = logarithms and math.log(value) or value
    end
    return result
end
function M.forward(model, observations)
    local result = {forward = {}, scales = {}, emissions = {}, loglik = 0.0}
    for t, observation in ipairs(observations) do
        local factors = M.emissions(model, observation, false)
        local row, scale = {}, 0.0
        for j = 1, #model.states do
            local value
            if t == 1 then value = model.initial[j] * factors[j]
            else
                local total = 0.0
                for i = 1, #model.states do total = total + result.forward[t - 1][i] * model.transition[i][j] end
                value = total * factors[j]
            end
            row[j], scale = value, scale + value
        end
        result.scales[t], result.emissions[t] = scale, factors
        for j = 1, #row do row[j] = row[j] / scale end
        result.forward[t] = row
        result.loglik = result.loglik + math.log(scale)
    end
    return result
end
function M.forward_backward(model, observations)
    local result = M.forward(model, observations)
    result.backward, result.posterior = {}, {}
    for t = 1, #observations do result.backward[t] = {} end
    for i = 1, #model.states do result.backward[#observations][i] = 1.0 end
    for t = #observations - 1, 1, -1 do
        for i = 1, #model.states do
            local total = 0.0
            for j = 1, #model.states do
                total = total + model.transition[i][j] * result.emissions[t + 1][j] * result.backward[t + 1][j]
            end
            result.backward[t][i] = total / result.scales[t + 1]
        end
    end
    for t = 1, #observations do
        result.posterior[t] = {}
        for i = 1, #model.states do result.posterior[t][i] = result.forward[t][i] * result.backward[t][i] end
    end
    return result
end
function M.viterbi(model, observations)
    local scores, predecessors, log_transition = {}, {{}}, {}
    for i, row in ipairs(model.transition) do
        log_transition[i] = {}
        for j, value in ipairs(row) do log_transition[i][j] = math.log(value) end
    end
    local factors = M.emissions(model, observations[1], true)
    for j = 1, #model.states do scores[j] = math.log(model.initial[j]) + factors[j] end
    for t = 2, #observations do
        local next_scores, previous_states = {}, {}
        factors = M.emissions(model, observations[t], true)
        for j = 1, #model.states do
            local candidates = {}
            for i = 1, #model.states do candidates[i] = scores[i] + log_transition[i][j] end
            local previous = M.maximum(candidates)
            previous_states[j], next_scores[j] = previous, candidates[previous] + factors[j]
        end
        scores, predecessors[t] = next_scores, previous_states
    end
    local path = {}
    path[#observations] = M.maximum(scores)
    local logprob = scores[path[#observations]]
    for t = #observations, 2, -1 do path[t - 1] = predecessors[t][path[t]] end
    return path, logprob
end
function M.decode(model, observations)
    local fb = M.forward_backward(model, observations)
    local path, logprob = M.viterbi(model, observations)
    local posterior, confidence = {}, {}
    for t, row in ipairs(fb.posterior) do posterior[t] = M.maximum(row); confidence[t] = row[posterior[t]] end
    return {viterbi = path, logprob = logprob, posterior = posterior, confidence = confidence, fb = fb}
end
function M.matrix(count)
    local result = {}
    for i = 1, count do
        result[i] = {}
        for j = 1, count do result[i][j] = 0 end
    end
    return result
end
function M.accumulate(matrix, truth, prediction)
    for t, actual in ipairs(truth) do matrix[actual][prediction[t]] = matrix[actual][prediction[t]] + 1 end
end
function M.correct(truth, prediction)
    local count = 0
    for t, actual in ipairs(truth) do if actual == prediction[t] then count = count + 1 end end
    return count
end

local function index_of(values, target)
    for i, value in ipairs(values) do if value == target then return i end end
end
local function encode(text, alphabet, missing)
    local result = {}
    for t = 1, #text do
        local character = text:sub(t, t)
        result[t] = character == missing and -1 or index_of(alphabet, character)
    end
    return result
end
seen = {}
for line = 1, #labeled_rows, 2 do
    local row, labels = labeled_rows[line], labeled_rows[line + 1][1]
    local name, text = row[1], row[3]:upper()
    local reason = rejection(name, text, labels, seen)
    if reason then rejections[#rejections + 1] = "REJECT labeled.txt line " .. line .. " " .. name .. " " .. reason
    else
        local target = row[2] == "TRAIN" and train_records or test_records
        target[#target + 1] = {name = name, observations = encode(text, initial.symbols, initial.missing), path = encode(labels, initial.states)}
    end
end
local models = {initial, baum, viterbi_model, supervised}
local evaluation_blocks, scores, decoded = {}, {}, {}
for model_index, model in ipairs(models) do
    local results = {}
    for i, record in ipairs(test_records) do results[i] = inference.decode(model, record.observations) end
    scores[model_index] = {}
    for decoder = 1, 2 do
        local name = decoder == 1 and "VITERBI" or "POSTERIOR"
        local matrix, regions, boundaries, calibration, record_lines = evaluation.matrix(#model.states), {0, 0, 0, 0}, {0, 0, 0, 0}, {}, {}
        for bucket = 1, 10 do calibration[bucket] = {0, 0} end
        local total_correct = 0
        for i, record in ipairs(test_records) do
            local result = results[i]
            local path = decoder == 1 and result.viterbi or result.posterior
            local correct = evaluation.correct(record.path, path)
            total_correct = total_correct + correct
            evaluation.accumulate(matrix, record.path, path)
            local true_regions = {}
            for _, run in ipairs(evaluation.runs(record.path)) do if run[1] == segment_state then true_regions[#true_regions + 1] = {run[2], run[3]} end end
            local rc = evaluation.region_counts(segments(path), true_regions)
            local bc = evaluation.boundary_counts(path, record.path, integer("boundary_tolerance"))
            for j = 1, 4 do regions[j], boundaries[j] = regions[j] + rc[j], boundaries[j] + bc[j] end
            if decoder == 2 then
                for t, confidence in ipairs(result.confidence) do
                    local bucket = math.min(9, math.floor(10 * confidence)) + 1
                    calibration[bucket][1] = calibration[bucket][1] + 1
                    if path[t] == record.path[t] then calibration[bucket][2] = calibration[bucket][2] + 1 end
                end
            end
            record_lines[#record_lines + 1] = "RECORD " .. record.name .. " length " .. #path .. " correct " .. correct
        end
        local block = {"EVALUATION " .. model.name .. " " .. name}
        append(block, evaluation.matrix_lines(model.states, matrix))
        append(block, evaluation.metrics(model.states, matrix))
        block[#block + 1] = "REGIONS " .. model.states[segment_state] .. " predicted " .. regions[1] .. " true " .. regions[2]
            .. " precision " .. fraction(regions[3], regions[1]) .. " recall " .. fraction(regions[4], regions[2])
        block[#block + 1] = "BOUNDARIES predicted " .. boundaries[1] .. " true " .. boundaries[2] .. " precision "
            .. fraction(boundaries[3], boundaries[1]) .. " recall " .. fraction(boundaries[4], boundaries[2])
        if decoder == 2 then
            for bucket = 0, 9 do
                local row = calibration[bucket + 1]
                if row[1] > 0 then
                    local range = bucket == 9 and "0.9-1.0" or "0." .. bucket .. "-0." .. (bucket + 1)
                    block[#block + 1] = "CALIBRATION " .. range .. " positions " .. row[1] .. " correct " .. row[2] .. " accuracy " .. fraction(row[2], row[1])
                end
            end
        end
        append(block, record_lines)
        evaluation_blocks[#evaluation_blocks + 1], scores[model_index][decoder] = block, total_correct
    end
    decoded[model_index] = {}
    for i, record in ipairs(sequences) do decoded[model_index][i] = inference.decode(model, record.observations) end
end
local function text(path, alphabet)
    local result = {}
    for i, index in ipairs(path) do result[i] = alphabet[index] end
    return table.concat(result)
end
