        index_by_id[data.pages[i].id] = i
    Ignored[] ignored := []
    local graph_io.Link[] accepted_links := []
    local bool[][] accepted := array(#data.pages, fill = array(#data.pages, fill = false))
    for record := data.records[]
        int source := index_by_id[record.source]
        int target := index_by_id[record.target]
        reason := ""
        if source < 0 or target < 0
            reason = "unknown"
        elif source == target
            reason = "self"
        elif accepted[source][target]
            reason = "duplicate"
        else
            accepted[source][target] = true
            accepted_links = accepted_links.append(graph_io.Link(source, target))
        if reason != ""
            ignored = ignored.append(Ignored(record, reason))
