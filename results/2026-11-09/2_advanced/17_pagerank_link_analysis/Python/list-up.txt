class LinkRecord:
    def __init__(self, line, source, target):
        self.line = line
        self.source = source
        self.target = target
def read_links(path):
    records = []
    for number, text in enumerate(read_lines(path)[1:], 2):
        source, target = text.split(",")
        records.append(LinkRecord(number, int(source), int(target)))
    return records
def classify_links(records, index_of):
    accepted = set()
    ignored = []
    for record in records:
        if record.source not in index_of or record.target not in index_of:
            ignored.append((record, "unknown"))
            continue
        link = index_of[record.source], index_of[record.target]
        if link[0] == link[1]:
            ignored.append((record, "self"))
        elif link in accepted:
            ignored.append((record, "duplicate"))
        else:
            accepted.add(link)
