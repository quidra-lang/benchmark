const ( Unknown Cell=iota; Filled; Empty )
type Grid [][]Cell
func isClueLine(text string) bool { if text=="" { return false }; for _,c:=range []byte(text) { if c!=' ' && (c<'0' || c>'9') { return false } }; return true }
func parseClue(text string) ([]int,bool) {
 if text=="0" { return []int{},true }; clue:=[]int{}
 for _,token:=range strings.Split(text," ") { if len(token)<1 || len(token)>2 || token[0]=='0' { return nil,false }; for _,c:=range []byte(token) { if c<'0' || c>'9' { return nil,false } }; n,_:=strconv.Atoi(token); clue=append(clue,n) }; return clue,true
}
func clueSpan(clue []int) int { sum:=len(clue)-1; for _,n:=range clue { sum+=n }; return sum }
func charState(c byte) Cell { if c=='#' { return Filled }; if c=='.' { return Empty }; return Unknown }
func readBlocks(path string) [][]string {
 data,e:=os.ReadFile(path); if e!=nil { panic(e) }; blocks:=[][]string{}; var block []string
 for _,line:=range strings.Split(strings.TrimSuffix(string(data),"\n"),"\n") { if block==nil { if line!="" && !strings.HasPrefix(line,"#") { block=[]string{line} } } else if line=="END" { blocks=append(blocks,block); block=nil } else { block=append(block,line) } }; return blocks
}
func validName(name string) bool { if len(name)<1 || len(name)>20 { return false }; for _,c:=range []byte(name) { if !(c>='a' && c<='z' || c>='0' && c<='9' || c=='_') { return false } }; return true }
func parsePuzzle(block []string) *Puzzle {
 size:=strings.Split(block[1]," "); p:=&Puzzle{name:strings.TrimPrefix(block[0],"PUZZLE "),width:integer(size[1]),height:integer(size[2]),answerResult:"-"}; at:=3; for block[at]!="COLUMNS" { p.rawRows=append(p.rawRows,block[at]); at++ }; at++
 for at<len(block) && isClueLine(block[at]) { p.rawColumns=append(p.rawColumns,block[at]); at++ }; for at<len(block) && strings.HasPrefix(block[at],"GIVEN ") { fields:=strings.Split(block[at]," "); p.givens=append(p.givens,Given{integer(fields[1])-1,integer(fields[2])-1,charState(fields[3][0])}); at++ }; if at<len(block) { p.hasAnswer=true; p.answerLines=append([]string{},block[at+1:]...) }; return p
}
func (p *Puzzle) Validate(seen map[string]bool) string {
 if !validName(p.name) { return "bad name" }; if seen[p.name] { return "duplicate name" }; if p.width<1 || p.width>30 || p.height<1 || p.height>30 { return "bad size" }; if len(p.rawRows)!=p.height { return fmt.Sprintf("row count %d expected %d",len(p.rawRows),p.height) }; if len(p.rawColumns)!=p.width { return fmt.Sprintf("column count %d expected %d",len(p.rawColumns),p.width) }
 totals:=[2]int{}; for kind,texts:=range [][]string{p.rawRows,p.rawColumns} { label,length:="row",p.width; if kind==1 { label,length="column",p.height }; clues:=[][]int{}; for i,text:=range texts { clue,valid:=parseClue(text); if !valid { return fmt.Sprintf("bad clue %s %d",label,i+1) }; if clueSpan(clue)>length { return fmt.Sprintf("%s %d too long",label,i+1) }; clues=append(clues,clue); for _,n:=range clue { totals[kind]+=n } }; if kind==0 { p.rows=clues } else { p.columns=clues } }; if totals[0]!=totals[1] { return fmt.Sprintf("total mismatch rows=%d columns=%d",totals[0],totals[1]) }
 occupied:=map[[2]int]bool{}; for k,g:=range p.givens { if g.row<0 || g.row>=p.height || g.column<0 || g.column>=p.width || g.state==Unknown { return fmt.Sprintf("bad given %d",k+1) }; cell:=[2]int{g.row,g.column}; if occupied[cell] { return fmt.Sprintf("duplicate given %d",k+1) }; occupied[cell]=true }
 if p.hasAnswer { if len(p.answerLines)!=p.height { return "bad answer" }; for _,line:=range p.answerLines { if len(line)!=p.width { return "bad answer" }; for _,c:=range []byte(line) { if c!='#' && c!='.' && c!='?' { return "bad answer" } } }; p.answer=stateGrid(p.answerLines) }; return ""
}
func logical(rows,columns [][]int,givens []Given,precise bool) Logical {
 grid:=make(Grid,len(rows)); for r:=range grid { grid[r]=make([]Cell,len(columns)) }; for _,g:=range givens { grid[g.row][g.column]=g.state }; rowMarks,columnMarks:=make([]bool,len(rows)),make([]bool,len(columns)); for i:=range rowMarks { rowMarks[i]=true }; for i:=range columnMarks { columnMarks[i]=true }; p:=propagate(grid,rows,columns,precise,rowMarks,columnMarks); result:="stuck"; if p.conflict>=0 { result="contradiction at="+p.steps[p.conflict].Label() } else if determined(grid)==len(rows)*len(columns) { result="solved" }; return Logical{grid,p,result}
}
func analyze(rows,columns [][]int,givens []Given) *Analysis {
 a:=&Analysis{simple:logical(rows,columns,givens,false),logic:logical(rows,columns,givens,true),difficulty:"-"}; a.lineSolves=len(a.logic.propagation.steps)
 if a.logic.propagation.conflict>=0 { a.status="contradiction" } else if a.logic.result=="solved" { a.status="unique"; a.solutions=[]Grid{a.logic.grid}; a.difficulty="medium"; if a.simple.result=="solved" { a.difficulty="easy" } } else { a.search=&Search{rows:rows,columns:columns}; a.search.Run(a.logic.grid,1); a.solutions=a.search.solutions; a.status=[]string{"none","unique","multiple"}[len(a.solutions)]; if a.status=="unique" { a.difficulty="hard" }; a.lineSolves+=a.search.lineSolves; a.guesses=len(a.search.guesses); a.maxDepth=a.search.maxDepth }; return a
}
func puzzleParagraph(p *Puzzle) []string {
 lines:=[]string{"PUZZLE "+p.name}; if p.error!="" { return append(lines,"error "+p.error) }; a:=p.analysis; lines=append(lines,fmt.Sprintf("size=%dx%d givens=%d status=%s difficulty=%s",p.width,p.height,len(p.givens),a.status,a.difficulty),"clues "+clueStats(p.rows,p.columns),fmt.Sprintf("simple known=%d line_solves=%d result=%s",determined(a.simple.grid),len(a.simple.propagation.steps),a.simple.result),fmt.Sprintf("logic known=%d line_solves=%d result=%s",determined(a.logic.grid),len(a.logic.propagation.steps),a.logic.result))
 if a.search!=nil { s:=a.search; lines=append(lines,fmt.Sprintf("search guesses=%d max_depth=%d backtracks=%d solutions=%d line_solves=%d",len(s.guesses),s.maxDepth,s.backtracks,len(s.solutions),s.lineSolves)); for _,g:=range s.guesses { lines=append(lines,fmt.Sprintf("guess %d depth=%d cell=%d,%d value=%c solves=%d known=%d result=%s",g.number,g.depth,g.row+1,g.column+1,stateChar(g.value),g.solves,g.known,g.result)) } }
 if a.status=="multiple" { cells:=differing(a.solutions[0],a.solutions[1]); lines=append(lines,fmt.Sprintf("differ count=%d cells=%s",len(cells),cellList(cells))) }
 names,grids:=[]string{},[]Grid{}; if a.status=="contradiction" { names=append(names,"partial"); grids=append(grids,a.logic.grid) } else { if a.simple.result=="stuck" { names=append(names,"simple"); grids=append(grids,a.simple.grid) }; if a.logic.result=="stuck" { names=append(names,"logic"); grids=append(grids,a.logic.grid) }; if a.status=="multiple" { names=append(names,"first","second"); grids=append(grids,a.solutions...) } }; if len(names)>0 { lines=append(lines,gridDisplay(names,grids)...) }; if a.status=="unique" { lines=append(lines,"sheet"); lines=append(lines,answerSheet(p.rows,p.columns,a.solutions[0])...); lines=append(lines,"picture "+features(a.solutions[0])) }
 if p.hasAnswer { result,answerLines:=grade(p); p.answerResult=result; lines=append(lines,answerLines...) }; return lines
func processPuzzles() ([]*Puzzle,[]string,[]string) {
 puzzles:=[]*Puzzle{}; paragraphs:=[][]string{}; stats:=[]string{"puzzle,status,difficulty,width,height,givens,simple_known,logic_known,line_solves,guesses,max_depth,solutions,answer"}; seen:=map[string]bool{}; counts:=map[string]int{}; lineSolves,guesses:=0,0; ranking:=[]*Puzzle{}
 for _,block:=range readBlocks("input/puzzles.txt") { p:=parsePuzzle(block); p.error=p.Validate(seen); seen[p.name]=true; puzzles=append(puzzles,p); if p.error!="" { counts["invalid"]++; stats=append(stats,p.name+",invalid"+strings.Repeat(",-",11)) } else { p.analysis=analyze(p.rows,p.columns,p.givens); a:=p.analysis; counts[a.status]++; counts[a.difficulty]++; lineSolves+=a.lineSolves; guesses+=a.guesses; if a.status=="unique" { ranking=append(ranking,p) } }; paragraphs=append(paragraphs,puzzleParagraph(p)); if p.error=="" { a:=p.analysis; stats=append(stats,fmt.Sprintf("%s,%s,%s,%d,%d,%d,%d,%d,%d,%d,%d,%d,%s",p.name,a.status,a.difficulty,p.width,p.height,len(p.givens),determined(a.simple.grid),determined(a.logic.grid),a.lineSolves,a.guesses,a.maxDepth,len(a.solutions),p.answerResult)) } }
 sort.Slice(ranking,func(i,j int)bool { a,b:=ranking[i],ranking[j]; if difficultyRank(a.analysis.difficulty)!=difficultyRank(b.analysis.difficulty) { return difficultyRank(a.analysis.difficulty)<difficultyRank(b.analysis.difficulty) }; x,y:=a.analysis.lineSolves+25*a.analysis.guesses,b.analysis.lineSolves+25*b.analysis.guesses; if x!=y { return x<y }; return a.name<b.name }); ranks:=[]string{"RANKING"}; for i,p:=range ranking { a:=p.analysis; ranks=append(ranks,fmt.Sprintf("rank %d %s %s score=%d size=%dx%d",i+1,p.name,a.difficulty,a.lineSolves+25*a.guesses,p.width,p.height)) }; paragraphs=append(paragraphs,ranks,[]string{fmt.Sprintf("SUMMARY puzzles=%d invalid=%d contradiction=%d none=%d unique=%d multiple=%d easy=%d medium=%d hard=%d line_solves=%d guesses=%d",len(puzzles),counts["invalid"],counts["contradiction"],counts["none"],counts["unique"],counts["multiple"],counts["easy"],counts["medium"],counts["hard"],lineSolves,guesses)}); return puzzles,joinParagraphs(paragraphs),stats
func main() { puzzles,solutions,stats:=processPuzzles(); generated:=processPictures(puzzles); if e:=os.MkdirAll("output",0755); e!=nil { panic(e) }; for name,lines:=range map[string][]string{"solutions.txt":solutions,"stats.csv":stats,"generated.txt":generated} { if e:=os.WriteFile("output/"+name,[]byte(strings.Join(lines,"\n")+"\n"),0644); e!=nil { panic(e) } } }
func lineCells(grid Grid,row bool,index int) []Cell { if row { return append([]Cell(nil),grid[index]...) }; cells:=make([]Cell,len(grid)); for r:=range grid { cells[r]=grid[r][index] }; return cells }
func determined(grid Grid) int { n:=0; for _,row:=range grid { for _,c:=range row { if c!=Unknown { n++ } } }; return n }
func cloneGrid(grid Grid) Grid { copy:=make(Grid,len(grid)); for r:=range grid { copy[r]=append([]Cell(nil),grid[r]...) }; return copy }
func propagate(grid Grid,rows,columns [][]int,precise bool,rowMarks,columnMarks []bool) Propagation {
 result:=Propagation{conflict:-1}
 for { isRow,index:=true,-1; for i,mark:=range rowMarks { if mark { index=i; break } }; if index<0 { isRow=false; for i,mark:=range columnMarks { if mark { index=i; break } } }; if index<0 { return result }
  var clue []int; if isRow { rowMarks[index]=false; clue=rows[index] } else { columnMarks[index]=false; clue=columns[index] }; cells:=lineCells(grid,isRow,index); solved:=solveLine(cells,clue,precise); step:=Step{row:isRow,index:index}
  if solved==nil { step.contradiction=true; result.steps=append(result.steps,step); result.conflict=len(result.steps)-1; return result }
  for position,state:=range solved { if cells[position]!=Unknown || state==Unknown { continue }; step.gain++; if isRow { grid[index][position]=state; columnMarks[position]=true } else { grid[position][index]=state; rowMarks[position]=true } }; result.steps=append(result.steps,step)
 }
}
type Search struct { rows,columns [][]int; guesses []Guess; solutions []Grid; maxDepth,backtracks,lineSolves int }
func (s *Search) Run(grid Grid,depth int) bool {
 r,c:=guessCell(grid)
 for _,value:=range []Cell{Filled,Empty} { child:=cloneGrid(grid); child[r][c]=value; rowMarks,columnMarks:=make([]bool,len(s.rows)),make([]bool,len(s.columns)); rowMarks[r]=true; columnMarks[c]=true; p:=propagate(child,s.rows,s.columns,true,rowMarks,columnMarks); known:=determined(child); result:="open"
  if p.conflict>=0 { result="contradiction at="+p.steps[p.conflict].Label(); s.backtracks++ } else if known==len(s.rows)*len(s.columns) { s.solutions=append(s.solutions,child); result=fmt.Sprintf("solution%d",len(s.solutions)) }; s.maxDepth=max(s.maxDepth,depth); s.lineSolves+=len(p.steps); s.guesses=append(s.guesses,Guess{len(s.guesses)+1,depth,r,c,value,len(p.steps),known,result})
  if result=="open" { if s.Run(child,depth+1) { return true } } else if len(s.solutions)==2 { return true }
 }; return false
}
