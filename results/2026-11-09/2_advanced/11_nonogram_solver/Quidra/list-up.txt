bool is_clue_line(string text)
    if text == ""
        return false
    for point := text.codepoints()[]
        if point != 32 and not (48 <= point <= 57)
            return false
    return true
int[] | none parse_clue(string text)
    if text == "0"
        return []
    int[] result := []
    for token := text.split(" ")[]
        if #token < 1 or #token > 2 or token[0] == "0"
            return none
        for point := token.codepoints()[]
            if point < 48 or point > 57
                return none
        result = result.append(int(token))
    return result
int total(int[] clue)
    result := 0
    for length := clue[]
        result += length
    return result
bool fits(int[] clue, int length)
    return total(clue) + int(#clue) - 1 <= length
int char_state(string character)
    return if character == "#" then line.FILLED elif character == "." then line.EMPTY else line.UNKNOWN
string[][] read_blocks(string path)
    string text := string.from_utf8(file.read_bin(path))
    string[][] result := []
    string[] current := []
    inside := false
    for row := text.split(NL)[]
        if not inside
            if row != "" and row[0] != "#"
                current = [row]
                inside = true
        elif row == "END"
            result = result.append(current)
            current = []
            inside = false
        else
            current = current.append(row)
    return result
bool valid_name(string name)
    if #name < 1 or #name > 20
        return false
    for point := name.codepoints()[]
        if not (97 <= point <= 122 or 48 <= point <= 57 or point == 95)
            return false
    return true
CheckedClues check_clues(string[] texts, string kind, int length)
    int[][] result := []
    for: index < #texts
        match clues.parse_clue(texts[index])
            int[] clue
                if not clues.fits(clue, length)
                    return CheckedClues([], "{kind} {index + 1} too long")

class Puzzle(local string[] texts)
    string name := texts[0][7:<#texts[0]]
    local dimensions := texts[1].split(" ")
    width := int(dimensions[1])
    height := int(dimensions[2])
    local columns_at := 3
    while texts[columns_at] != "COLUMNS"
        columns_at += 1
    string[] row_lines := texts[3:<columns_at]
    string[] column_lines := []
    RawGiven[] given_lines := []
    local position := columns_at + 1
    while position < int(#texts) and clues.is_clue_line(texts[position])
        column_lines = column_lines.append(texts[position])
        position += 1
    while position < int(#texts) and texts[position].starts_with("GIVEN ")
        parts := texts[position].split(" ")
        given_lines = given_lines.append(RawGiven(int(parts[1]), int(parts[2]), parts[3]))
        position += 1
    answer_present := position < int(#texts)
    string[] answer_lines := if answer_present then texts[position + 1:<#texts] else []
    int[][] row_clues := []
    int[][] column_clues := []
    Given[] givens := []
    int[][] answer := []
    string .validate(map<string, int> earlier)
        if not valid_name(name)
            return "bad name"
        if earlier.has(name)
            return "duplicate name"
        if width < 1 or width > 30 or height < 1 or height > 30
            return "bad size"
        if int(#row_lines) != height
            return "row count {#row_lines} expected {height}"
        if int(#column_lines) != width
            return "column count {#column_lines} expected {width}"
        rows := check_clues(row_lines, "row", width)
        if rows.problem != ""
            return rows.problem
        row_clues = rows.values
        columns := check_clues(column_lines, "column", height)
        if columns.problem != ""
            return columns.problem
        column_clues = columns.values
        row_total := 0
        column_total := 0
        for clue := row_clues[]
            row_total += clues.total(clue)
        for clue := column_clues[]
            column_total += clues.total(clue)
        if row_total != column_total
            return "total mismatch rows={row_total} columns={column_total}"
        map<string, int> taken
        for: index < #given_lines
            given := given_lines[index]
            if given.row < 1 or given.row > height or given.column < 1 or given.column > width or (given.value != "#" and given.value != ".")
                return "bad given {index + 1}"
            key := "{given.row},{given.column}"
            if taken.has(key)
                return "duplicate given {index + 1}"
            .taken.set(key, 1)
            givens = givens.append(Given(given.row - 1, given.column - 1, render.char_state(given.value)))
        if answer_present
            if int(#answer_lines) != height
                return "bad answer"
            for text := answer_lines[]
                if int(#text) != width
                    return "bad answer"
                for: index < #text
                    if text[index] != "#" and text[index] != "." and text[index] != "?"
                        return "bad answer"
            answer = state_grid(answer_lines)
        return ""

Logical logical_solve(int[][] row_clues, int[][] column_clues, Given[] givens, bool precise)
    int[][] grid := []
    for: row < #row_clues
        grid = grid.append(array(#column_clues, fill = line.UNKNOWN))
    for given := givens[]
        grid[given.row][given.column] = given.state
    details := propagation.propagate(&grid, row_clues, column_clues, precise, array(#row_clues, fill = true), array(#column_clues, fill = true))
    string result := if details.contradiction then "contradiction at=" + details.conflict.label() elif propagation.determined_count(grid) == int(#row_clues * #column_clues) then "solved" else "stuck"
    return Logical(grid, details, result)

int[] line_cells(int[][] grid, bool is_row, int index)
    if is_row
        return grid[index]
    int[] result := []
    for row := grid[]
        result = result.append(row[index])
    return result
int determined_count(int[][] grid)
    result := 0
    for row := grid[]
        for state := row[]
            if state != line.UNKNOWN
                result += 1
    return result
class Result()
    Step[] steps := []
    contradiction := false
    conflict := Step(true, 0, 0, false)
Result propagate(int[][] &grid, int[][] row_clues, int[][] column_clues, bool precise, bool[] row_marks, bool[] column_marks)
    result := Result()
    while true
        is_row := true
        chosen := -1
        for: index < #row_marks
            if row_marks[index]
                chosen = int(index)
                break
        if chosen < 0
            is_row = false
            for: index < #column_marks
                if column_marks[index]
                    chosen = int(index)
                    break
        if chosen < 0
            return result
        int[] clue
        if is_row
            row_marks[chosen] = false
            clue = row_clues[chosen]
        else
            column_marks[chosen] = false
            clue = column_clues[chosen]
        cells := line_cells(grid, is_row, chosen)
        solved := line.solve_line(cells, clue, precise)
        if solved.contradiction
            result.contradiction = true
            result.conflict = Step(is_row, chosen, 0, true)
            result.steps = result.steps.append(result.conflict)
            return result
        gain := 0
        for: position < #cells
            if cells[position] == line.UNKNOWN and solved.cells[position] != line.UNKNOWN
                gain += 1
                if is_row
                    grid[chosen][position] = solved.cells[position]
                    column_marks[position] = true
                else
                    grid[position][chosen] = solved.cells[position]
                    row_marks[position] = true
        result.steps = result.steps.append(Step(is_row, chosen, gain, false))
class Search(int[][] row_clues, int[][] column_clues)
    Guess[] guesses := []
    int[][][] solutions := []
    backtracks := 0
    max_depth := 0
    line_solves := 0
    bool .run(int[][] grid, int depth)
        chosen := guess_cell(grid)
        for value := [line.FILLED, line.EMPTY][]
            number := int(#guesses) + 1
            if depth > max_depth
                max_depth = depth
            child := grid
            child[chosen.row][chosen.column] = value
            row_marks := array(#row_clues, fill = false)
            column_marks := array(#column_clues, fill = false)
            row_marks[chosen.row] = true
            column_marks[chosen.column] = true
            propagated := propagation.propagate(&child, row_clues, column_clues, true, row_marks, column_marks)
            known := propagation.determined_count(child)
            result := "open"
            if propagated.contradiction
                result = "contradiction at=" + propagated.conflict.label()
                backtracks += 1
            elif known == int(#row_clues * #column_clues)
                solutions = solutions.append(child)
                result = "solution{#solutions}"
            solves := int(#propagated.steps)
            line_solves += solves
            guesses = guesses.append(Guess(number, depth, chosen.row, chosen.column, value, solves, known, result))
            if result == "open"
                if .run(child, depth + 1)
                    return true
            elif #solutions == 2
                return true
        return false
