let unknown = 0, filled = 1, empty = 2
func isClueLine(_ text: String) -> Bool { !text.isEmpty && text.utf8.allSatisfy { $0 == 32 || (48...57).contains($0) } }
func parseClue(_ text: String) -> [Int]? {
    if text == "0" { return [] }
    let tokens = text.split(separator: " ", omittingEmptySubsequences: false)
    var result: [Int] = []
    for token in tokens {
        let bytes = Array(token.utf8)
        if !(1...2).contains(bytes.count) || bytes[0] == 48 || !bytes.allSatisfy({ (48...57).contains($0) }) { return nil }
        result.append(Int(token)!)
    }
    return result
}
func clueFits(_ clue: [Int], _ length: Int) -> Bool { clue.reduce(0, +) + clue.count - 1 <= length }
func charState(_ character: Character) -> Int { character == "#" ? filled : character == "." ? empty : unknown }
func readBlocks(_ path: String) throws -> [[String]] {
    var blocks: [[String]] = [], current: [String] = [], inside = false
    for line in try String(contentsOfFile: path, encoding: .utf8).components(separatedBy: "\n") {
        if !inside { if !line.isEmpty && !line.hasPrefix("#") { current = [line]; inside = true } }
        else if line == "END" { blocks.append(current); current = []; inside = false }
        else { current.append(line) }
    }
    return blocks
}

func propagate(_ grid: inout [[Int]], _ rowClues: [[Int]], _ columnClues: [[Int]], _ precise: Bool, _ rowMarks: [Bool], _ columnMarks: [Bool]) -> Propagation {
    var rows = rowMarks, columns = columnMarks, result = Propagation()
    while true {
        let isRow: Bool, index: Int
        if let row = rows.firstIndex(of: true) { isRow = true; index = row; rows[row] = false }
        else if let column = columns.firstIndex(of: true) { isRow = false; index = column; columns[column] = false }
        else { return result }

        var gain = 0
        for position in cells.indices where cells[position] == unknown && solved[position] != unknown {
            gain += 1
            if isRow { grid[index][position] = solved[position]; columns[position] = true }
            else { grid[position][index] = solved[position]; rows[position] = true }
        }
        result.steps.append(Step(isRow: isRow, index: index, gain: gain, contradiction: false))
    }
}
