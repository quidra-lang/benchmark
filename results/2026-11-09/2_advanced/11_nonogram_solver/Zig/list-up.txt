pub const Cell = enum(u8) { unknown, filled, empty };
pub fn eq(a: []const u8, b: []const u8) bool { return std.mem.eql(u8, a, b); }
pub fn number(text: []const u8) usize { return std.fmt.parseInt(usize, text, 10) catch unreachable; }
pub fn split(text: []const u8, delimiter: u8) Strings { var result: std.ArrayList([]const u8) = .empty; var it = std.mem.splitScalar(u8, text, delimiter); while (it.next()) |part| append([]const u8, &result, part); return result.items; }
pub fn lines(text: []const u8) Strings { const result = split(text, '\n'); return result[0..result.len - 1]; }
pub fn isClueLine(text: []const u8) bool { if (text.len == 0) return false; for (text) |ch| { if (ch != ' ' and (ch < '0' or ch > '9')) return false; } return true; }
pub fn parse(text: []const u8) ?Clue { if (eq(text, "0")) return &.{}; const tokens = split(text, ' '); const result = alloc(usize, tokens.len); for (tokens, result) |token, *value| { if (token.len == 0 or token.len > 2 or token[0] < '1' or token[0] > '9') return null; for (token) |ch| { if (ch < '0' or ch > '9') return null; } value.* = number(token); } return result; }
pub fn total(clue: Clue) usize { var result: usize = 0; for (clue) |length| result += length; return result; }
pub fn span(clue: Clue) i64 { return int(total(clue)) + int(clue.len) - 1; }
pub fn fits(clue: Clue, length: usize) bool { return span(clue) <= int(length); }
pub fn charState(ch: u8) L.Cell { return switch (ch) { '?' => .unknown, '#' => .filled, else => .empty }; }
const Given = struct { row: usize, column: usize, value: L.Cell };
const RawGiven = struct { row: usize, column: usize, value: u8 };
const Puzzle = struct { name: []const u8, width: usize, height: usize, row_lines: C.Strings, column_lines: C.Strings, given_lines: []const RawGiven, answer_lines: ?C.Strings, rows: C.Clues = &.{}, columns: C.Clues = &.{}, givens: []const Given = &.{}, answer: ?P.Grid = null };
const Status = enum { contradiction, none, unique, multiple };
const Difficulty = enum { easy, medium, hard, unavailable };
fn difficultyText(value: Difficulty) []const u8 { return if (value == .unavailable) "-" else @tagName(value); }
fn readBlocks(text: []const u8) []const C.Strings { var blocks: std.ArrayList(C.Strings) = .empty; var block: std.ArrayList([]const u8) = .empty; var inside = false;
    for (C.lines(text)) |line| { if (!inside) { if (line.len > 0 and line[0] != '#') { inside = true; block = .empty; C.append([]const u8, &block, line); } } else if (C.eq(line, "END")) { C.append(C.Strings, &blocks, block.items); inside = false; } else C.append([]const u8, &block, line); } return blocks.items;
}
fn parsePuzzle(block: C.Strings) *Puzzle { const size = C.split(block[1], ' '); var column_at: usize = 3; while (!C.eq(block[column_at], "COLUMNS")) column_at += 1; var position = column_at + 1; const first = position; while (position < block.len and C.isClueLine(block[position])) position += 1; const column_lines = block[first..position]; var givens: std.ArrayList(RawGiven) = .empty;
    while (position < block.len and std.mem.startsWith(u8, block[position], "GIVEN ")) : (position += 1) { const fields = C.split(block[position], ' '); C.append(RawGiven, &givens, .{ .row = C.number(fields[1]), .column = C.number(fields[2]), .value = fields[3][0] }); }
    return C.create(Puzzle, .{ .name = block[0][7..], .width = C.number(size[1]), .height = C.number(size[2]), .row_lines = block[3..column_at], .column_lines = column_lines, .given_lines = givens.items, .answer_lines = if (position < block.len) block[position + 1..] else null });
}
fn validName(name: []const u8) bool { if (name.len == 0 or name.len > 20) return false; for (name) |ch| { if (!((ch >= 'a' and ch <= 'z') or (ch >= '0' and ch <= '9') or ch == '_')) return false; } return true; }
fn validate(puzzle: *Puzzle, earlier: std.StringHashMapUnmanaged(void)) ?[]const u8 {
    if (!validName(puzzle.name)) return "bad name"; if (earlier.contains(puzzle.name)) return "duplicate name"; if (puzzle.width < 1 or puzzle.width > 30 or puzzle.height < 1 or puzzle.height > 30) return "bad size";
    if (puzzle.row_lines.len != puzzle.height) return C.fmt("row count {d} expected {d}", .{ puzzle.row_lines.len, puzzle.height }); if (puzzle.column_lines.len != puzzle.width) return C.fmt("column count {d} expected {d}", .{ puzzle.column_lines.len, puzzle.width });
    const rows = C.alloc(C.Clue, puzzle.height); const columns = C.alloc(C.Clue, puzzle.width);
    for (puzzle.row_lines, rows, 1..) |text, *clue, i| { clue.* = C.parse(text) orelse return C.fmt("bad clue row {d}", .{i}); if (!C.fits(clue.*, puzzle.width)) return C.fmt("row {d} too long", .{i}); }
    for (puzzle.column_lines, columns, 1..) |text, *clue, i| { clue.* = C.parse(text) orelse return C.fmt("bad clue column {d}", .{i}); if (!C.fits(clue.*, puzzle.height)) return C.fmt("column {d} too long", .{i}); }
    var row_total: usize = 0; var column_total: usize = 0; for (rows) |clue| row_total += C.total(clue); for (columns) |clue| column_total += C.total(clue); if (row_total != column_total) return C.fmt("total mismatch rows={d} columns={d}", .{ row_total, column_total }); puzzle.rows = rows; puzzle.columns = columns;
    var taken = [_]bool{false} ** 900; const givens = C.alloc(Given, puzzle.given_lines.len); for (puzzle.given_lines, givens, 1..) |raw, *given, k| { if (raw.row < 1 or raw.row > puzzle.height or raw.column < 1 or raw.column > puzzle.width or (raw.value != '#' and raw.value != '.')) return C.fmt("bad given {d}", .{k}); const slot = (raw.row - 1) * puzzle.width + raw.column - 1; if (taken[slot]) return C.fmt("duplicate given {d}", .{k}); taken[slot] = true; given.* = .{ .row = raw.row - 1, .column = raw.column - 1, .value = R.charState(raw.value) }; } puzzle.givens = givens;
    if (puzzle.answer_lines) |texts| { if (texts.len != puzzle.height) return "bad answer"; for (texts) |text| { if (text.len != puzzle.width or !validStates(text, true)) return "bad answer"; } puzzle.answer = gridFromText(texts); } return null;
}
const Logical = struct { grid: P.Grid, propagation: P.Result };
fn logicalSolve(allocator: std.mem.Allocator, rows: C.Clues, columns: C.Clues, givens: []const Given, precise: bool) Logical { var grid: P.Grid = .{ .width = columns.len, .height = rows.len }; for (givens) |given| grid.set(given.row, given.column, given.value); var row_marks = [_]bool{false} ** 30; var column_marks = [_]bool{false} ** 30; @memset(row_marks[0..rows.len], true); @memset(column_marks[0..columns.len], true); const propagation = P.propagate(allocator, &grid, rows, columns, precise, &row_marks, &column_marks); return .{ .grid = grid, .propagation = propagation }; }
fn solvingResult(grid: *const P.Grid, propagation: P.Result) []const u8 { if (propagation.conflict) |step| return C.fmt("contradiction at={s}", .{step.label()}); return if (grid.known() == grid.width * grid.height) "solved" else "stuck"; }
fn analyze(rows: C.Clues, columns: C.Clues, givens: []const Given) Analysis {
    var arena = std.heap.ArenaAllocator.init(C.backing); const allocator = arena.allocator(); const simple = logicalSolve(allocator, rows, columns, givens, false); const logic = logicalSolve(allocator, rows, columns, givens, true); var search: ?Q.Search = null; var solutions: []const P.Grid = &.{}; var status: Status = undefined; var difficulty: Difficulty = .unavailable;
    if (logic.propagation.conflict != null) status = .contradiction else if (logic.grid.known() == rows.len * columns.len) { status = .unique; const one = allocator.alloc(P.Grid, 1) catch @panic("allocation failed"); one[0] = logic.grid; solutions = one; difficulty = if (simple.propagation.conflict == null and simple.grid.known() == rows.len * columns.len) .easy else .medium; } else { var result: Q.Search = .{}; _ = result.run(allocator, &logic.grid, rows, columns, 1); solutions = result.solutions.items; status = switch (solutions.len) { 0 => .none, 1 => .unique, else => .multiple }; if (status == .unique) difficulty = .hard; search = result; }
    return .{ .arena = arena, .simple_grid = simple.grid, .logic_grid = logic.grid, .simple = simple.propagation, .logic = logic.propagation, .search = search, .solutions = solutions, .status = status, .difficulty = difficulty, .line_solves = logic.propagation.steps.items.len + (if (search) |result| result.line_solves else @as(usize, 0)), .guesses = if (search) |result| result.guesses.items.len else 0, .max_depth = if (search) |result| result.max_depth else 0 };
}
fn buildReport(puzzle: *Puzzle) BuiltReport {
    var analysis = analyze(puzzle.rows, puzzle.columns, puzzle.givens); defer analysis.deinit(); const simple_result = solvingResult(&analysis.simple_grid, analysis.simple); const logic_result = solvingResult(&analysis.logic_grid, analysis.logic); var lines: C.Lines = .{};
    lines.add("PUZZLE {s}", .{puzzle.name}); lines.add("size={d}x{d} givens={d} status={s} difficulty={s}", .{ puzzle.width, puzzle.height, puzzle.givens.len, @tagName(analysis.status), difficultyText(analysis.difficulty) }); lines.add("clues {s}", .{clueStatistics(puzzle.rows, puzzle.columns)}); lines.add("simple known={d} line_solves={d} result={s}", .{ analysis.simple_grid.known(), analysis.simple.steps.items.len, simple_result }); lines.add("logic known={d} line_solves={d} result={s}", .{ analysis.logic_grid.known(), analysis.logic.steps.items.len, logic_result }); lines.add("steps {s}", .{stepStatistics(analysis.logic)});
    if (analysis.search) |search| { lines.add("search guesses={d} max_depth={d} backtracks={d} solutions={d} line_solves={d}", .{ search.guesses.items.len, search.max_depth, search.backtracks, search.solutions.items.len, search.line_solves }); for (search.guesses.items) |guess| { const outcome = if (guess.outcome == .contradiction) C.fmt("contradiction at={s}", .{guess.conflict.?.label()}) else @tagName(guess.outcome); lines.add("guess {d} depth={d} cell={d},{d} value={c} solves={d} known={d} result={s}", .{ guess.number, guess.depth, guess.position.row + 1, guess.position.column + 1, R.stateChar(guess.value), guess.solves, guess.known, outcome }); } }
    if (analysis.status == .multiple) { const positions = differing(&analysis.solutions[0], &analysis.solutions[1]); const parts = C.alloc([]const u8, positions.len); for (positions, parts) |position, *part| part.* = C.fmt("{d},{d}", .{ position.row + 1, position.column + 1 }); lines.add("differ count={d} cells={s}", .{ positions.len, C.join(parts, " ") }); }
    var names: std.ArrayList([]const u8) = .empty; var grids: std.ArrayList(*const P.Grid) = .empty;
    if (analysis.status == .contradiction) { C.append([]const u8, &names, "partial"); C.append(*const P.Grid, &grids, &analysis.logic_grid); } else { if (C.eq(simple_result, "stuck")) { C.append([]const u8, &names, "simple"); C.append(*const P.Grid, &grids, &analysis.simple_grid); } if (C.eq(logic_result, "stuck")) { C.append([]const u8, &names, "logic"); C.append(*const P.Grid, &grids, &analysis.logic_grid); } if (analysis.status == .multiple) { C.append([]const u8, &names, "first"); C.append([]const u8, &names, "second"); C.append(*const P.Grid, &grids, &analysis.solutions[0]); C.append(*const P.Grid, &grids, &analysis.solutions[1]); } }
    if (names.items.len > 0) lines.extend(R.gridDisplay(names.items, grids.items)); if (analysis.status == .unique) { lines.write("sheet"); lines.extend(R.answerSheet(puzzle.rows, puzzle.columns, &analysis.solutions[0])); lines.add("picture {s}", .{featuresText(&analysis.solutions[0])}); }
    const report = C.create(Report, .{ .puzzle = puzzle, .status = analysis.status, .difficulty = analysis.difficulty, .line_solves = analysis.line_solves, .guesses = analysis.guesses, .max_depth = analysis.max_depth, .solutions = analysis.solutions.len, .simple_known = analysis.simple_grid.known(), .logic_known = analysis.logic_grid.known(), .grading = grading }); return .{ .report = report, .paragraph = lines };
fn processPuzzles(blocks: []const C.Strings) PuzzleOutputs {
    var earlier: std.StringHashMapUnmanaged(void) = .empty; var reports: std.ArrayList(*Report) = .empty; var solutions: C.Lines = .{}; var stats: C.Lines = .{}; stats.write("puzzle,status,difficulty,width,height,givens,simple_known,logic_known,line_solves,guesses,max_depth,solutions,answer"); var invalid: usize = 0;
    for (blocks) |block| { const puzzle = parsePuzzle(block); const reason = validate(puzzle, earlier); earlier.put(C.allocator, puzzle.name, {}) catch @panic("allocation failed");
        if (reason) |error_text| { invalid += 1; var paragraph: C.Lines = .{}; paragraph.add("PUZZLE {s}", .{puzzle.name}); paragraph.add("error {s}", .{error_text}); solutions.paragraph(paragraph); var text: C.Text = .{}; text.add("{s},invalid", .{puzzle.name}); for (0..11) |_| text.write(",-"); stats.write(text.value()); } else { const built = buildReport(puzzle); C.append(*Report, &reports, built.report); solutions.paragraph(built.paragraph); const report = built.report; stats.add("{s},{s},{s},{d},{d},{d},{d},{d},{d},{d},{d},{d},{s}", .{ puzzle.name, @tagName(report.status), difficultyText(report.difficulty), puzzle.width, puzzle.height, puzzle.givens.len, report.simple_known, report.logic_known, report.line_solves, report.guesses, report.max_depth, report.solutions, report.grading }); }
    }
    var arena = std.heap.ArenaAllocator.init(init.gpa); defer arena.deinit(); C.allocator = arena.allocator(); C.backing = init.gpa; const dir = std.Io.Dir.cwd(); const puzzles_text = try dir.readFileAlloc(init.io, "input/puzzles.txt", C.allocator, .unlimited); const pictures_text = try dir.readFileAlloc(init.io, "input/pictures.txt", C.allocator, .unlimited); const outputs = processPuzzles(readBlocks(puzzles_text)); const generated = processPictures(readBlocks(pictures_text), outputs.reports); try dir.createDirPath(init.io, "output"); const names = [_][]const u8{ "solutions.txt", "stats.csv", "generated.txt" }; const reports = [_]C.Lines{ outputs.solutions, outputs.stats, generated }; for (names, reports) |name, report| try dir.writeFile(init.io, .{ .sub_path = C.fmt("output/{s}", .{name}), .data = report.content() });
pub const Grid = struct {
    width: usize, height: usize, cells: [900]L.Cell = [_]L.Cell{.unknown} ** 900,
    pub fn get(self: *const Grid, row: usize, column: usize) L.Cell { return self.cells[row * self.width + column]; }
    pub fn set(self: *Grid, row: usize, column: usize, value: L.Cell) void { self.cells[row * self.width + column] = value; }
    pub fn line(self: *const Grid, is_row: bool, index: usize) [30]L.Cell { var result: [30]L.Cell = undefined; const count = if (is_row) self.width else self.height; for (0..count) |position| result[position] = if (is_row) self.get(index, position) else self.get(position, index); return result; }
    pub fn known(self: *const Grid) usize { var count: usize = 0; for (self.cells[0..self.width * self.height]) |state| { if (state != .unknown) count += 1; } return count; }
};
pub const Result = struct { steps: std.ArrayList(Step) = .empty, conflict: ?Step = null };
pub fn propagate(allocator: std.mem.Allocator, grid: *Grid, rows: C.Clues, columns: C.Clues, precise: bool, row_marks: *[30]bool, column_marks: *[30]bool) Result {
    var result: Result = .{};
    while (true) { var chosen: ?usize = null; var is_row = true; for (row_marks[0..grid.height], 0..) |mark, i| { if (mark) { chosen = i; break; } } if (chosen == null) { is_row = false; for (column_marks[0..grid.width], 0..) |mark, i| { if (mark) { chosen = i; break; } } } const index = chosen orelse return result;
        if (is_row) row_marks[index] = false else column_marks[index] = false; const clue = if (is_row) rows[index] else columns[index]; const count = if (is_row) grid.width else grid.height; const cells = grid.line(is_row, index); const solved = L.solve(cells[0..count], clue, precise);
        if (!solved.consistent) { const step: Step = .{ .is_row = is_row, .index = index, .gain = 0, .contradiction = true }; result.steps.append(allocator, step) catch @panic("allocation failed"); result.conflict = step; return result; }
        var gain: usize = 0; for (cells[0..count], solved.cells[0..count], 0..) |old, value, position| { if (old != .unknown or value == .unknown) continue; gain += 1; if (is_row) { grid.set(index, position, value); column_marks[position] = true; } else { grid.set(position, index, value); row_marks[position] = true; } }
        result.steps.append(allocator, .{ .is_row = is_row, .index = index, .gain = gain }) catch @panic("allocation failed");
    }
}
pub const Search = struct {
    guesses: std.ArrayList(Guess) = .empty, solutions: std.ArrayList(P.Grid) = .empty, backtracks: usize = 0, max_depth: usize = 0, line_solves: usize = 0,
    pub fn run(self: *Search, allocator: std.mem.Allocator, grid: *const P.Grid, rows: C.Clues, columns: C.Clues, depth: usize) bool {
        const position = guessCell(grid); for ([_]L.Cell{ .filled, .empty }) |value| { var child = grid.*; child.set(position.row, position.column, value); var row_marks = [_]bool{false} ** 30; var column_marks = [_]bool{false} ** 30; row_marks[position.row] = true; column_marks[position.column] = true; const propagation = P.propagate(allocator, &child, rows, columns, true, &row_marks, &column_marks); const known = child.known(); var guess: Guess = .{ .number = self.guesses.items.len + 1, .depth = depth, .position = position, .value = value, .solves = propagation.steps.items.len, .known = known, .outcome = .open, .conflict = propagation.conflict };
            if (propagation.conflict != null) { guess.outcome = .contradiction; self.backtracks += 1; } else if (known == grid.width * grid.height) { self.solutions.append(allocator, child) catch @panic("allocation failed"); guess.outcome = if (self.solutions.items.len == 1) .solution1 else .solution2; }
            self.guesses.append(allocator, guess) catch @panic("allocation failed"); self.max_depth = @max(self.max_depth, depth); self.line_solves += guess.solves;
            if (guess.outcome == .open) { if (self.run(allocator, &child, rows, columns, depth + 1)) return true; } else if (self.solutions.items.len == 2) return true;
        } return false;
    }
};
