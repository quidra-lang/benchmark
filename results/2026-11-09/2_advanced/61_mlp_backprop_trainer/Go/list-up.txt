func fixed(v float64, digits int) string {
	s := fmt.Sprintf("%.*f", digits, v)
	if strings.HasPrefix(s, "-") {
		zero := true
		for _, c := range s[1:] {
			if c != '0' && c != '.' {
				zero = false
			}
		}
		if zero {
			s = s[1:]
		}
	}
	return s
