pub fn real(value: f64, digits: usize) []const u8 {
    const text = switch (digits) {
        4 => c.fmt("{d:.4}", .{value}),
    };
    if (text.len > 0 and text[0] == '-') {
        var zero = true;
        for (text[1..]) |ch| {
            if (ch != '0' and ch != '.') zero = false;
        }
        if (zero) return text[1..];
    }
    return text;
