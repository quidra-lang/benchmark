string padded(int value, int width)
    text := string(value)
    return "0" * nat(width - int(#text)) + text
string real_text(real64 value, int places)
    factor := 10 ^ places
    magnitude := if value < 0.0 then -value else value
    rounded := math.floor(magnitude * real64(factor) + 0.5)
    sign_text := if value < 0.0 and rounded != 0 then "-" else ""
    return sign_text + string(rounded / factor) + "." + padded(rounded % factor, places)
