function satisfies(region: Region, condition: Condition): boolean {
  const comparison =
    typeof condition.value === 'string'
      ? Number(region.orientation !== condition.value)
      : compare(region.features[condition.feature], condition.value);
  switch (condition.operator) {
    case '<':
      return comparison < 0;
    case '<=':
      return comparison <= 0;
    case '>':
      return comparison > 0;
    case '>=':
      return comparison >= 0;
    case '=':
      return comparison === 0;
    case '!=':
      return comparison !== 0;
    default:
      throw new Error('Unknown operator');
function applyRules(regions: Region[], rules: Rules): void {
  for (const r of regions) {
    const dropped = rules.filters.findIndex((condition) => !satisfies(r, condition));
    r.kept = dropped < 0;
    r.droppedBy = dropped < 0 ? null : dropped;
    if (!r.kept) continue;
    r.className =
      rules.classes.find((rule) => rule.conditions.every((condition) => satisfies(r, condition)))
        ?.name ?? 'other';
    r.flags = rules.flags
      .filter((rule) => rule.conditions.every((condition) => satisfies(r, condition)))
      .map((rule) => rule.name);
  }
