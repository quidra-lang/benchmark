bool condition_holds(Condition condition, Region region)
    if condition.feature == "orientation"
        equal := region.orientation == condition.text
        return if condition.comparison == "=" then equal else not equal
    first := value_of(region, condition.feature)
    second := parse_ratio(condition.text)
    left := first.numerator * second.denominator
    right := second.numerator * first.denominator
    op := condition.comparison
    return if op == "<" then left < right elif op == "<=" then left <= right elif op == ">" then left > right elif op == ">=" then left >= right elif op == "=" then left == right else left != right
bool rule_matches(Rule rule, Region region)
    for condition := rule.conditions[]
        if not condition_holds(condition, region)
            return false
    return true
void apply_rules(ImageAnalysis &analysis, Rules rules)
    for: index < int(#analysis.regions)
        &region := &analysis.regions[index]
        for: filter < int(#rules.filters)
            if not condition_holds(rules.filters[filter], region)
                region.dropped_by = filter
                break
        if region.dropped_by >= 0
            continue
        region.kept = true
        for rule := rules.classes[]
            if rule_matches(rule, region)
                region.class_name = rule.name
                break
        region.flags = [for rule := rules.flags[] if rule_matches(rule, region) then rule.name]
        count := [for region := regions[] if region.dropped_by == index then 1].sum()
