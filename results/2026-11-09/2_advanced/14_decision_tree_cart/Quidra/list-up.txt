class Fraction(int numerator, int denominator)
    int compare(Fraction other)
        first := numerator * other.denominator
        second := other.numerator * denominator
        return if first < second then -1 elif first > second then 1 else 0
int[] class_counts(dataset.Row[] rows, int classes)
    counts := array(nat(classes), fill = 0)
    for row := rows[]
        counts[row.label] += 1
    return counts
int square_sum(int[] counts)
    return (counts ^ 2).sum()
int majority(int[] counts)
    chosen := 0
    for: 1 <= i < int(#counts)
        if counts[i] > counts[chosen]
            chosen = i
    return chosen
int error_count(int[] counts)
    return counts.sum() - counts[majority(counts)]
Fraction quality(int[] left, int[] right)
    left_size := left.sum()
    right_size := right.sum()
    return Fraction(square_sum(left) * right_size + square_sum(right) * left_size, left_size * right_size)
bool improves(Fraction value, int[] counts)
    return value.numerator * counts.sum() > square_sum(counts) * value.denominator
class Split(dataset.Attribute attribute, int point)

class Surrogate(Split split, bool left, int score)

int[] counts_at(map<int, int[]> groups, int value, int classes)
    match groups.get(value)
        int[] counts
            return counts
        none
            return array(nat(classes), fill = 0)
Candidate[] candidates(dataset.Attribute attribute, dataset.Row[] rows, int[] labels, int classes)
    map<int, int[]> groups
    total := array(nat(classes), fill = 0)
    for: i < int(#rows)
        value := rows[i].values[attribute.index]
        counts := counts_at(groups, value, classes)
        counts[labels[i]] += 1
        .groups.set(value, counts)
        total[labels[i]] += 1
    ordered := groups.keys().sorted()
    Candidate[] result := []
    if attribute.is_integer
        left := array(nat(classes), fill = 0)
        for: i < int(#ordered) - 1
            left = left + counts_at(groups, ordered[i], classes)
            result = result.append(Candidate(Split(attribute, ordered[i] + ordered[i + 1]), left, total - left))
    elif #ordered > 1
        for value := ordered[]
            left := counts_at(groups, value, classes)
            result = result.append(Candidate(Split(attribute, value), left, total - left))
    return result
Candidate | none best_split(dataset.Row[] rows, dataset.Attribute[] attributes, int classes, int min_leaf)
    labels := [for row := rows[] then row.label]
    Candidate | none best := none
    for attribute := attributes[]
        for candidate := candidates(attribute, rows, labels, classes)[]
            if candidate.left_size < min_leaf or candidate.right_size < min_leaf
                continue
            replace := false
            match best
                Candidate previous
                    replace = candidate.quality().compare(previous.quality()) > 0
                none
                    replace = true
            if replace
                best = candidate
    return best
Surrogate[] find_surrogates(dataset.Row[] rows, Split primary, dataset.Attribute[] attributes, int limit)
    directions := [for row := rows[] then if primary.goes_left(row.values[primary.attribute.index]) then 0 else 1]
    count := int(#rows)
    left_size := count - directions.sum()
    baseline := math.max(left_size, count - left_size)
    Surrogate[] result := []
    for attribute := attributes[]
        if attribute.index == primary.attribute.index
            continue
        Surrogate | none best := none
        for candidate := candidates(attribute, rows, directions, 2)[]
            agreement := candidate.left_counts[0] + candidate.right_counts[1]
            value := Surrogate(candidate.split, agreement >= count - agreement, math.max(agreement, count - agreement))
            replace := false
            match best
                Surrogate previous
                    replace = value.score > previous.score
                none
                    replace = true
            if replace
                best = value
        match best
            Surrogate value
                if value.score > baseline
                    position := int(#result)
                    result = result.append(value)
                    while position > 0 and result[position - 1].score < value.score
                        result[position] = result[position - 1]
                        position -= 1
                    result[position] = value
            none
                continue
    return result[0:<math.min(limit, int(#result))]
class Builder(dataset.Attribute[] attributes, int classes, dataset.Settings settings)
    Node[] nodes := []
    int .grow_node(dataset.Row[] rows, int number, int depth)
        node := Node(number, depth, impurity.class_counts(rows, classes))
        index := int(#nodes)
        nodes = nodes.append(node)
        if node.errors == 0 or depth == settings.max_depth or node.size < settings.min_split
            return index
        match best_split(rows, attributes, classes, settings.min_leaf)
            Candidate candidate
                if not impurity.improves(candidate.quality(), node.counts)
                    return index
                split := candidate.split
                left_rows := [for row := rows[] if split.goes_left(row.values[split.attribute.index]) then row]
                right_rows := [for row := rows[] if not split.goes_left(row.values[split.attribute.index]) then row]
                node.split = split
                node.surrogates = find_surrogates(rows, split, attributes, settings.max_surrogates)
                node.default_left = #left_rows >= #right_rows
                node.left = .grow_node(left_rows, 2 * number, depth + 1)
                node.right = .grow_node(right_rows, 2 * number + 1, depth + 1)
                nodes[index] = node
            none
                return index
        return index
Tree grow(dataset.Row[] rows, dataset.Attribute[] attributes, int classes, dataset.Settings settings)
    builder := Builder(attributes, classes, settings)
    .builder.grow_node(rows, 1, 0)
    return Tree(builder.nodes)

class Route(int[] path, int[] methods)
Direction direction(tree_builder.Node node, dataset.Row row)
    value := row.values[node.split.attribute.index]
    if value >= 0
        return Direction(node.split.goes_left(value), 0)
    for surrogate := node.surrogates[]
        value = row.values[surrogate.split.attribute.index]
        if value >= 0
            return Direction(surrogate.goes_left(value), 1)
    return Direction(node.default_left, 2)
class View(tree_builder.Tree tree, bool[] cut, depth_limit := -1)
    bool is_leaf(int index)
        tree_builder.Node node := tree.nodes[index]
        return node.left < 0 or cut[index] or (depth_limit >= 0 and node.depth >= depth_limit)
    Route route(dataset.Row row)
        index := 0
        path := [index]
        int[] methods := []
        while not is_leaf(index)
            tree_builder.Node node := tree.nodes[index]
            choice := direction(node, row)
            index = if choice.left then node.left else node.right
            path = path.append(index)
            methods = methods.append(choice.method)
        return Route(path, methods)
    tree_builder.Node leaf_of(dataset.Row row)
        Route result := route(row)
        return tree.nodes[result.path[-1]]
    int predict(dataset.Row row)
        return leaf_of(row).prediction
    int correct_count(dataset.Row[] rows)
        return [for row := rows[] if predict(row) == row.label then 1].sum()
View full_tree(tree_builder.Tree tree)
    return View(tree, array(#tree.nodes, fill = false))
int[][] confusion_matrix(int[] actual, int[] predicted, int classes)
    int[][] matrix := array(nat(classes), fill = array(nat(classes), fill = 0))
    for: i < int(#actual)
        matrix[actual[i]][predicted[i]] += 1
    return matrix
string[] classification_lines(string title, string[] classes, int[] actual, int[] predicted)
    count := int(#classes)
    matrix := confusion_matrix(actual, predicted, count)
    correct := [for i := matrix.indices()[] then matrix[i][i]].sum()
    total := int(#actual)
    lines := ["confusion {title} correct {correct}/{total} accuracy {fraction_text(correct, total)}", "actual\predicted " + classes.join(" ")]
    for: i < count
        lines = lines.append(classes[i] + " " + counts_text(matrix[i]))
    real64[] f1 := []
    for: k < count
        hits := matrix[k][k]
        predicted_total := [for row := matrix[] then row[k]].sum()
        support := matrix[k].sum()
        if predicted_total + support > 0
            f1 = f1.append(real64(2 * hits) / real64(predicted_total + support))
        lines = lines.append("class {classes[k]} precision {fraction_text(hits, predicted_total)} recall {fraction_text(hits, support)} f1 {fraction_text(2 * hits, predicted_total + support)} support {support}")
    return lines.append("macro_f1 {real_text(average(f1))}")

int decide(tree_builder.Tree tree, int index, dataset.Row[][] reached, Decision[] &decisions, bool[] &cut)
    tree_builder.Node node := tree.nodes[index]
    leaf_error := [for row := reached[index][] if row.label != node.prediction then 1].sum()
    if node.left < 0
        return leaf_error
    subtree_error := decide(tree, node.left, reached, &decisions, &cut)
    subtree_error += decide(tree, node.right, reached, &decisions, &cut)
    pruned := leaf_error <= subtree_error
    decisions = decisions.append(Decision(index, int(#reached[index]), leaf_error, subtree_error, pruned))
    if pruned
        cut[index] = true
        return leaf_error
    return subtree_error
Reduced reduced_error(tree_builder.Tree tree, dataset.Row[] rows)
    grown := evaluation.full_tree(tree)
    reached := reached_rows(grown, rows)
    Decision[] decisions := []
    cut := grown.cut
    decide(tree, 0, reached, &decisions, &cut)
    return Reduced(evaluation.View(tree, cut), decisions)
