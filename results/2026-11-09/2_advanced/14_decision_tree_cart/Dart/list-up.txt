}

List<String> readLines(String path) => File(path).readAsLinesSync();
List<Row> readRows(String path, List<Attribute> attributes, List<String> classes) {
  final rows = <Row>[];
  for (final line in readLines(path).skip(1)) {
    final fields = line.split(',');
    final values = <int?>[];
    for (final attribute in attributes) {
      final text = fields[attribute.index + 1];
      values.add(text == '?' ? null : attribute.integer ? int.parse(text) : attribute.values.indexOf(text));
    }
    rows.add(Row(fields[0], values, classes.indexOf(fields.last)));
  }
  return rows;
}
Dataset loadDataset() {
  final attributes = <Attribute>[];
  late String target;
  late List<String> classes;
  for (final line in readLines('input/schema.txt')) {
    final fields = line.split(' ');
    if (fields[0] == 'ATTRIBUTE') {
      attributes.add(Attribute(attributes.length, fields[1], fields[2] == 'INT', fields.skip(3).toList()));
    } else {
      target = fields[1];
      classes = fields.skip(2).toList();
    }
  }
  final entries = <String, List<int>>{};
  for (final line in readLines('input/params.txt')) {
    final fields = line.split(' ');
    entries[fields[0]] = fields.skip(1).map(int.parse).toList();
  }
  return Dataset(attributes, target, classes,
      readRows('input/train.csv', attributes, classes),
      readRows('input/validation.csv', attributes, classes),
      readRows('input/test.csv', attributes, classes), Settings(entries));
}
typedef Fraction = (int, int);
class Statistics {
  final int size;
  final List<int> counts;
  final int squares, majority, errors;
  Statistics(this.size, this.counts, this.squares, this.majority, this.errors);
}
int sum(Iterable<int> values) => values.fold(0, (total, value) => total + value);
int majority(List<int> counts) {
  var selected = 0;
  for (var index = 1; index < counts.length; index++) {
    if (counts[index] > counts[selected]) selected = index;
  }
  return selected;
}
int squareSum(List<int> counts) => counts.fold(0, (total, count) => total + count * count);
Statistics statistics(List<Row> rows, int classes) {
  final counts = List.filled(classes, 0);
  for (final row in rows) { counts[row.label]++; }
  final selected = majority(counts);
  return Statistics(rows.length, counts, squareSum(counts), selected, rows.length - counts[selected]);
}
int compareFractions(Fraction first, Fraction second) =>
    (first.$1 * second.$2).compareTo(second.$1 * first.$2);
Fraction quality(Statistics left, Statistics right) =>
    (left.squares * right.size + right.squares * left.size, left.size * right.size);
bool improves(Fraction value, Statistics parent) => value.$1 * parent.size > parent.squares * value.$2;
class Split {
  final Attribute attribute;
  final int point;
  Split(this.attribute, this.point);
  bool goesLeft(int value) => attribute.integer ? 2 * value <= point : value == point;
class Surrogate {
  final Split split;
  final bool left;
  final int score;
  Surrogate(this.split, this.left, this.score);
}
class Candidate {
  final Split split;
  final List<Row> left, right;
  final Fraction value;
  Candidate(this.split, this.left, this.right, this.value);
}
class Node {
  final int number, depth;
  final Statistics stats;
  Split? split;
  List<Surrogate> surrogates = [];
  bool defaultLeft = true;
  Node? left, right;
  Node(this.number, this.depth, this.stats);
}
List<Split> splitCandidates(List<Row> rows, Attribute attribute) {
  final values = rows.map((row) => row.values[attribute.index]!).toSet().toList()..sort();
  if (values.length < 2) return [];
  return attribute.integer
      ? [for (var index = 1; index < values.length; index++) Split(attribute, values[index - 1] + values[index])]
      : [for (final value in values) Split(attribute, value)];
}
class TreeBuilder {
  final List<Attribute> attributes;
  final int classes;
  final Settings settings;
  TreeBuilder(this.attributes, this.classes, this.settings);
  Candidate? bestForAttribute(List<Row> rows, Attribute attribute) {
    Candidate? best;
    for (final split in splitCandidates(rows, attribute)) {
      final left = <Row>[], right = <Row>[];
      for (final row in rows) {
        (split.goesLeft(row.values[attribute.index]!) ? left : right).add(row);
      }
      if (left.length < settings.minLeaf || right.length < settings.minLeaf) continue;
      final value = quality(statistics(left, classes), statistics(right, classes));
      if (best == null || compareFractions(value, best.value) > 0) {
        best = Candidate(split, left, right, value);
      }
    }
    return best;
  }
  Candidate? bestSplit(List<Row> rows) {
    Candidate? best;
    for (final attribute in attributes) {
      final candidate = bestForAttribute(rows, attribute);
      if (candidate != null && (best == null || compareFractions(candidate.value, best.value) > 0)) best = candidate;
    }
    return best;
  }
  List<Surrogate> findSurrogates(List<Row> rows, Split primary, int leftCount, int rightCount) {
    final retained = <Surrogate>[];
    for (final attribute in attributes) {
      if (attribute.index == primary.attribute.index) continue;
      Surrogate? best;
      for (final split in splitCandidates(rows, attribute)) {
        var agreement = 0;
        for (final row in rows) {
          if (split.goesLeft(row.values[attribute.index]!) == primary.goesLeft(row.values[primary.attribute.index]!)) agreement++;
        }
        final left = agreement >= rows.length - agreement;
        final score = math.max(agreement, rows.length - agreement);
        if (best == null || score > best.score) best = Surrogate(split, left, score);
      }
      if (best != null && best.score > math.max(leftCount, rightCount)) retained.add(best);
    }
    retained.sort((a, b) {
      final order = b.score.compareTo(a.score);
      return order != 0 ? order : a.split.attribute.index.compareTo(b.split.attribute.index);
    });
    return retained.take(settings.maxSurrogates).toList();
  }
  Node grow(List<Row> rows) => growNode(rows, 1, 0);
  Node growNode(List<Row> rows, int number, int depth) {
    final node = Node(number, depth, statistics(rows, classes));
    if (node.stats.errors == 0 || depth == settings.maxDepth || rows.length < settings.minSplit) return node;
    final best = bestSplit(rows);
    if (best == null || !improves(best.value, node.stats)) return node;
    node.split = best.split;
    node.defaultLeft = best.left.length >= best.right.length;
    node.surrogates = findSurrogates(rows, best.split, best.left.length, best.right.length);
    node.left = growNode(best.left, 2 * number, depth + 1);
    node.right = growNode(best.right, 2 * number + 1, depth + 1);
    return node;
  }
}
class RouteStep {
  final Node node;
  final bool left;
  final String method;
  RouteStep(this.node, this.left, this.method);
}
class Route {
  final Node leaf;
  final List<Node> path;
  final List<RouteStep> steps;
  Route(this.leaf, this.path, this.steps);
}
class Tree {
  final Node root;
  final Set<int> cuts;
  Tree(this.root, [Set<int>? cuts]) : cuts = cuts ?? {};
  bool isLeaf(Node node) => node.split == null || cuts.contains(node.number);
  Route route(Row row) {
    var node = root;
    final path = <Node>[], steps = <RouteStep>[];
    while (true) {
      path.add(node);
      if (isLeaf(node)) return Route(node, path, steps);
      var left = node.defaultLeft, method = 'default';
      final split = node.split!, value = row.values[split.attribute.index];
      if (value != null) {
        left = split.goesLeft(value);
        method = 'primary';
      } else {
        for (final surrogate in node.surrogates) {
          final observed = row.values[surrogate.split.attribute.index];
          if (observed == null) continue;
          left = surrogate.split.goesLeft(observed) == surrogate.left;
          method = 'surrogate';
          break;
        }
      }
      steps.add(RouteStep(node, left, method));
      node = left ? node.left! : node.right!;
    }
  }
  int predict(Row row) => route(row).leaf.stats.majority;
  int correct(List<Row> rows) => rows.where((row) => predict(row) == row.label).length;
}
List<String> classificationLines(String title, List<String> classes, List<Row> rows, List<int> predicted) {
  final matrix = List.generate(classes.length, (_) => List.filled(classes.length, 0));
  var correct = 0;
  for (var index = 0; index < rows.length; index++) {
    matrix[rows[index].label][predicted[index]]++;
    if (rows[index].label == predicted[index]) correct++;
  }
  final lines = [
    'confusion $title correct $correct/${rows.length} accuracy ${fractionText(correct, rows.length)}',
    'actual\\predicted ${classes.join(' ')}',
    for (var index = 0; index < classes.length; index++) '${classes[index]} ${matrix[index].join(' ')}',
  ];
  var macro = 0.0, defined = 0;
  for (var index = 0; index < classes.length; index++) {
    final hits = matrix[index][index], predictedTotal = sum(matrix.map((counts) => counts[index])), support = sum(matrix[index]);
    final denominator = predictedTotal + support;
    if (denominator > 0) { macro += 2 * hits / denominator; defined++; }
    lines.add('class ${classes[index]} precision ${fractionText(hits, predictedTotal)} recall ${fractionText(hits, support)} f1 ${fractionText(2 * hits, denominator)} support $support');
  }
  lines.add('macro_f1 ${defined == 0 ? '-' : realText(macro / defined)}');
  return lines;
}
(Tree, List<Decision>) reducedErrorPruning(Node root, List<Row> rows) {
  final original = Tree(root), reached = <int, List<Row>>{};
  for (final row in rows) {
    for (final node in original.route(row).path) { reached.putIfAbsent(node.number, () => []).add(row); }
  }
  final cuts = <int>{}, decisions = <Decision>[];
  int decide(Node node) {
    final visits = reached[node.number] ?? <Row>[];
    final leafError = visits.where((row) => row.label != node.stats.majority).length;
    if (node.split == null) return leafError;
    final subtreeError = decide(node.left!) + decide(node.right!);
    final pruned = leafError <= subtreeError;
    decisions.add(Decision(node, visits.length, leafError, subtreeError, pruned));
    if (pruned) { cuts.add(node.number); return leafError; }
    return subtreeError;
  }
  decide(root);
  return (Tree(root, cuts), decisions);
}
