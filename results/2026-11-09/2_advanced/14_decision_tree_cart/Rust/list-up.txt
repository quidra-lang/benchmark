pub struct Attribute {
    pub name: String,
    pub integer: bool,
    pub values: Vec<String>,
}
#[derive(Clone)]
pub struct Row {
    pub id: String,
    pub values: Vec<Option<i64>>,
    pub label: usize,
}
impl Row {
    pub fn missing_count(&self) -> usize { self.values.iter().filter(|value| value.is_none()).count() }
pub struct Settings {
    pub max_depth: usize,
    pub min_split: usize,
    pub min_leaf: usize,
    pub max_surrogates: usize,
pub fn read_lines(path: &str) -> io::Result<Vec<String>> {
    Ok(fs::read_to_string(path)?.lines().map(str::to_owned).collect())
}
fn read_rows(path: &str, attributes: &[Attribute], classes: &[String]) -> io::Result<Vec<Row>> {
    let mut rows = Vec::new();
    for line in read_lines(path)?.into_iter().skip(1) {
        let fields: Vec<_> = line.split(',').collect();
        let values = attributes.iter().enumerate().map(|(index, attribute)| {
            let text = fields[index + 1];
            if text == "?" { None }
            else if attribute.integer { Some(text.parse().unwrap()) }
            else { Some(attribute.values.iter().position(|value| value == text).unwrap() as i64) }
        }).collect();
        let label = classes.iter().position(|name| name == fields[fields.len() - 1]).unwrap();
        rows.push(Row { id: fields[0].to_owned(), values, label });
    }
    Ok(rows)
}
pub fn load_dataset(directory: &str) -> io::Result<Dataset> {
    let mut attributes = Vec::new();
    let mut target = String::new();
    let mut classes = Vec::new();
    for line in read_lines(&format!("{directory}/schema.txt"))? {
        let fields: Vec<_> = line.split(' ').collect();
        if fields[0] == "ATTRIBUTE" {
            attributes.push(Attribute { name: fields[1].to_owned(), integer: fields[2] == "INT",
                values: fields[3..].iter().map(|value| (*value).to_owned()).collect() });
        } else {
            target = fields[1].to_owned();
            classes = fields[2..].iter().map(|value| (*value).to_owned()).collect();
        }
    }
    let parameters: Vec<Vec<i64>> = read_lines(&format!("{directory}/params.txt"))?.iter()
        .map(|line| line.split(' ').skip(1).map(|value| value.parse().unwrap()).collect()).collect();
    let settings = Settings {
        max_depth: parameters[0][0] as usize, min_split: parameters[1][0] as usize,
        min_leaf: parameters[2][0] as usize, max_surrogates: parameters[3][0] as usize,
        folds: parameters[4][0] as usize, curve: parameters[5].iter().map(|&value| value as usize).collect(),
        bag_trees: parameters[6][0] as usize, bag_seed: parameters[7][0],
        repeats: parameters[8][0] as usize, permutation_seed: parameters[9][0],
    };
    let train = read_rows(&format!("{directory}/train.csv"), &attributes, &classes)?;
    let validation = read_rows(&format!("{directory}/validation.csv"), &attributes, &classes)?;
    let test = read_rows(&format!("{directory}/test.csv"), &attributes, &classes)?;
    Ok(Dataset { attributes, target, classes, train, validation, test, settings })
}
pub struct Fraction { pub numerator: i64, pub denominator: i64 }
impl Fraction {
    pub fn compare(self, other: Self) -> Ordering {
        (self.numerator * other.denominator).cmp(&(other.numerator * self.denominator))
    }
pub fn class_counts(rows: &[Row], classes: usize) -> Vec<i64> {
    let mut counts = vec![0; classes];
    for row in rows { counts[row.label] += 1; }
    counts
}
pub fn square_sum(counts: &[i64]) -> i64 { counts.iter().map(|count| count * count).sum() }
pub fn majority(counts: &[i64]) -> usize {
    let mut best = 0;
    for index in 1..counts.len() { if counts[index] > counts[best] { best = index; } }
    best
}
pub fn error_count(counts: &[i64]) -> i64 { counts.iter().sum::<i64>() - counts[majority(counts)] }
pub fn evaluation_value(left: &[i64], right: &[i64]) -> Fraction {
    let left_size: i64 = left.iter().sum();
    let right_size: i64 = right.iter().sum();
    Fraction { numerator: square_sum(left) * right_size + square_sum(right) * left_size,
        denominator: left_size * right_size }
}
pub fn improves(value: Fraction, counts: &[i64]) -> bool {
    value.numerator * counts.iter().sum::<i64>() > square_sum(counts) * value.denominator
}
pub struct Split { pub attribute: usize, pub point: i64 }
impl Split {
    pub fn goes_left(self, value: i64, attributes: &[Attribute]) -> bool {
        if attributes[self.attribute].integer { 2 * value <= self.point } else { value == self.point }
    }
#[derive(Clone)]
pub struct Surrogate { pub split: Split, pub forward: bool, pub score: i64 }
pub struct Candidate { pub split: Split, pub left: Vec<i64>, pub right: Vec<i64> }
impl Candidate {
    pub fn value(&self) -> Fraction { evaluation_value(&self.left, &self.right) }
    pub fn left_size(&self) -> i64 { self.left.iter().sum() }
    pub fn right_size(&self) -> i64 { self.right.iter().sum() }
}
pub struct Node {
    pub number: usize,
    pub depth: usize,
    pub counts: Vec<i64>,
    pub size: i64,
    pub prediction: usize,
    pub errors: i64,
    pub split: Option<Split>,
    pub surrogates: Vec<Surrogate>,
    pub default_left: bool,
    pub children: Option<[usize; 2]>,
}
pub struct Tree { pub nodes: Vec<Node> }
pub fn candidates(attribute: usize, attributes: &[Attribute], rows: &[Row], labels: &[usize], classes: usize) -> Vec<Candidate> {
    let values: BTreeSet<_> = rows.iter().map(|row| row.values[attribute].unwrap()).collect();
    if values.len() < 2 { return Vec::new(); }
    let ordered: Vec<_> = values.into_iter().collect();
    let points = if attributes[attribute].integer {
        ordered.windows(2).map(|pair| pair[0] + pair[1]).collect::<Vec<_>>()
    } else { ordered };
    let mut result = Vec::new();
    for point in points {
        let split = Split { attribute, point };
        let mut left = vec![0; classes];
        let mut right = vec![0; classes];
        for (row, &label) in rows.iter().zip(labels) {
            if split.goes_left(row.values[attribute].unwrap(), attributes) { left[label] += 1; }
            else { right[label] += 1; }
        }
        result.push(Candidate { split, left, right });
    }
    result
}
pub struct TreeBuilder<'a> { pub attributes: &'a [Attribute], pub classes: usize, pub settings: &'a Settings }
impl TreeBuilder<'_> {
    pub fn best_split(&self, rows: &[Row], attributes: &[usize]) -> Option<Candidate> {
        let labels: Vec<_> = rows.iter().map(|row| row.label).collect();
        let mut best: Option<Candidate> = None;
        for &attribute in attributes {
            for candidate in candidates(attribute, self.attributes, rows, &labels, self.classes) {
                if candidate.left_size() < self.settings.min_leaf as i64 || candidate.right_size() < self.settings.min_leaf as i64 { continue; }
                if best.as_ref().is_none_or(|current| candidate.value().compare(current.value()).is_gt()) { best = Some(candidate); }
            }
        }
        best
    }
    fn surrogates(&self, rows: &[Row], primary: Split) -> Vec<Surrogate> {
        let labels: Vec<_> = rows.iter().map(|row| if primary.goes_left(row.values[primary.attribute].unwrap(), self.attributes) { 0 } else { 1 }).collect();
        let left = labels.iter().filter(|&&label| label == 0).count() as i64;
        let size = rows.len() as i64;
        let baseline = left.max(size - left);
        let mut selected = Vec::new();
        for attribute in 0..self.attributes.len() {
            if attribute == primary.attribute { continue; }
            let mut best: Option<Surrogate> = None;
            for candidate in candidates(attribute, self.attributes, rows, &labels, 2) {
                let agreement = candidate.left[0] + candidate.right[1];
                let forward = agreement >= size - agreement;
                let score = agreement.max(size - agreement);
                if best.as_ref().is_none_or(|current| score > current.score) {
                    best = Some(Surrogate { split: candidate.split, forward, score });
                }
            }
            if let Some(best) = best { if best.score > baseline { selected.push(best); } }
        }
        selected.sort_by_key(|surrogate| (std::cmp::Reverse(surrogate.score), surrogate.split.attribute));
        selected.truncate(self.settings.max_surrogates);
        selected
    }
    pub fn grow(&self, rows: &[Row]) -> Rc<Tree> {
        let mut tree = Tree { nodes: Vec::new() };
        self.grow_node(rows, 1, 0, &mut tree);
        Rc::new(tree)
    }
    fn grow_node(&self, rows: &[Row], number: usize, depth: usize, tree: &mut Tree) -> usize {
        let counts = class_counts(rows, self.classes);
        let prediction = majority(&counts);
        let index = tree.nodes.len();
        let pure = counts[prediction] == rows.len() as i64;
        let errors = error_count(&counts);
        tree.nodes.push(Node { number, depth, counts, size: rows.len() as i64, prediction, errors,
            split: None, surrogates: Vec::new(), default_left: true, children: None });
        if pure || depth == self.settings.max_depth || rows.len() < self.settings.min_split { return index; }
        let attributes: Vec<_> = (0..self.attributes.len()).collect();
        let Some(best) = self.best_split(rows, &attributes) else { return index; };
        if !improves(best.value(), &tree.nodes[index].counts) { return index; }
        let split = best.split;
        let (left, right): (Vec<_>, Vec<_>) = rows.iter().cloned().partition(|row| split.goes_left(row.values[split.attribute].unwrap(), self.attributes));
        tree.nodes[index].split = Some(split);
        tree.nodes[index].surrogates = self.surrogates(rows, split);
        tree.nodes[index].default_left = left.len() >= right.len();
        let left_child = self.grow_node(&left, number * 2, depth + 1, tree);
        let right_child = self.grow_node(&right, number * 2 + 1, depth + 1, tree);
        tree.nodes[index].children = Some([left_child, right_child]);
        index
    }
}
#[derive(Clone, Copy, PartialEq)]
pub enum Method { Primary, Surrogate, Default }
impl Method {
    pub fn index(self) -> usize { match self { Self::Primary => 0, Self::Surrogate => 1, Self::Default => 2 } }
pub struct Route { pub path: Vec<usize>, pub methods: Vec<Method> }
#[derive(Clone)]
pub struct PrunedTree { pub tree: Rc<Tree>, pub cut: BTreeSet<usize> }
impl PrunedTree {
    pub fn new(tree: Rc<Tree>) -> Self { Self { tree, cut: BTreeSet::new() } }
    pub fn is_leaf(&self, index: usize) -> bool { self.tree.nodes[index].split.is_none() || self.cut.contains(&index) }
    pub fn route(&self, row: &Row, attributes: &[Attribute]) -> Route {
        let mut index = 0;
        let mut path = vec![index];
        let mut methods = Vec::new();
        while !self.is_leaf(index) {
            let node = &self.tree.nodes[index];
            let split = node.split.unwrap();
            let mut direction = (node.default_left, Method::Default);
            if let Some(value) = row.values[split.attribute] { direction = (split.goes_left(value, attributes), Method::Primary); }
            else {
                for surrogate in &node.surrogates {
                    if let Some(value) = row.values[surrogate.split.attribute] {
                        direction = (surrogate.split.goes_left(value, attributes) == surrogate.forward, Method::Surrogate);
                        break;
                    }
                }
            }
            index = node.children.unwrap()[if direction.0 { 0 } else { 1 }];
            path.push(index);
            methods.push(direction.1);
        }
        Route { path, methods }
    }
    pub fn leaf_of(&self, row: &Row, attributes: &[Attribute]) -> usize { *self.route(row, attributes).path.last().unwrap() }
    pub fn predict(&self, row: &Row, attributes: &[Attribute]) -> usize { self.tree.nodes[self.leaf_of(row, attributes)].prediction }
    pub fn correct_count(&self, rows: &[Row], attributes: &[Attribute]) -> i64 { rows.iter().filter(|row| self.predict(row, attributes) == row.label).count() as i64 }
pub fn classification_lines(title: &str, classes: &[String], rows: &[Row], predictions: &[usize]) -> Vec<String> {
    let mut matrix = vec![vec![0i64; classes.len()]; classes.len()];
    for (row, &predicted) in rows.iter().zip(predictions) { matrix[row.label][predicted] += 1; }
    let correct: i64 = (0..classes.len()).map(|i| matrix[i][i]).sum();
    let total = rows.len() as i64;
    let mut lines = vec![format!("confusion {title} correct {correct}/{total} accuracy {}", fraction_text(correct, total, 4)),
        format!("actual\\predicted {}", classes.join(" "))];
    for (name, counts) in classes.iter().zip(&matrix) { lines.push(format!("{name} {}", counts.iter().map(i64::to_string).collect::<Vec<_>>().join(" "))); }
    let mut f1_sum = 0.0;
    let mut f1_count = 0;
    for (index, name) in classes.iter().enumerate() {
        let hits = matrix[index][index];
        let predicted: i64 = matrix.iter().map(|counts| counts[index]).sum();
        let support: i64 = matrix[index].iter().sum();
        if predicted + support > 0 { f1_sum += 2.0 * hits as f64 / (predicted + support) as f64; f1_count += 1; }
        lines.push(format!("class {name} precision {} recall {} f1 {} support {support}",
            fraction_text(hits, predicted, 4), fraction_text(hits, support, 4), fraction_text(2 * hits, predicted + support, 4)));
    }
    lines.push(format!("macro_f1 {}", real_text(f1_sum / f1_count as f64, 4)));
    lines
}
fn decide(index: usize, tree: &Tree, reached: &[Vec<usize>], rows: &[Row], decisions: &mut Vec<Decision>, cut: &mut BTreeSet<usize>) -> i64 {
    let node = &tree.nodes[index];
    let leaf_error = reached[index].iter().filter(|&&row| rows[row].label != node.prediction).count() as i64;
    let Some([left, right]) = node.children else { return leaf_error; };
    let subtree_error = decide(left, tree, reached, rows, decisions, cut) + decide(right, tree, reached, rows, decisions, cut);
    let pruned = leaf_error <= subtree_error;
    decisions.push(Decision { node: index, rows: reached[index].len(), leaf_error, subtree_error, pruned });
    if pruned { cut.insert(index); leaf_error } else { subtree_error }
}
pub fn reduced_error_pruning(tree: Rc<Tree>, rows: &[Row], attributes: &[Attribute]) -> (PrunedTree, Vec<Decision>) {
    let mut pruned = PrunedTree::new(tree);
    let mut reached = vec![Vec::new(); pruned.tree.nodes.len()];
    for (index, row) in rows.iter().enumerate() { for node in pruned.route(row, attributes).path { reached[node].push(index); } }
    let mut decisions = Vec::new();
    decide(0, &pruned.tree, &reached, rows, &mut decisions, &mut pruned.cut);
    (pruned, decisions)
}
    fn new(data: &'a Dataset) -> Self {
        let builder = TreeBuilder { attributes: &data.attributes, classes: data.classes.len(), settings: &data.settings };
        let grown = PrunedTree::new(builder.grow(&data.train));
        let (pruned, decisions) = reduced_error_pruning(grown.tree.clone(), &data.validation, &data.attributes);
        let mut result = Self { data, builder, grown, pruned, decisions, rules: Vec::new(), bounds: Vec::new(), fold_trees: Vec::new(),
            depths: Vec::new(), selected_depth: 0, sequence: Vec::new(), selected_index: 0, bags: Vec::new(),
            oob_votes: vec![vec![0; data.classes.len()]; data.train.len()], test_votes: vec![vec![0; data.classes.len()]; data.test.len()], predictions: BTreeMap::new() };
        for (name, tree) in [("grown", &result.grown), ("pruned", &result.pruned), ("depth", &result.depths[result.selected_depth].tree), ("cost_complexity", &result.sequence[result.selected_index].tree)] {
            result.predictions.insert(name, data.test.iter().map(|row| tree.predict(row, &data.attributes)).collect());
        }
        result
    fn accuracy(&self, tree: &PrunedTree, rows: &[Row]) -> String { ratio_text(tree.correct_count(rows, &self.data.attributes), rows.len() as i64) }
    fn predictions_file(&self) -> Vec<String> {
        let mut lines = vec!["id,predicted,actual,grown,leaf,path,missing".to_owned()];
        for row in &self.data.test {
            let route = self.pruned.route(row, &self.data.attributes);
            let leaf = &self.pruned.tree.nodes[*route.path.last().unwrap()];
            let path = route.path.iter().map(|&index| self.pruned.tree.nodes[index].number.to_string()).collect::<Vec<_>>().join("-");
            lines.push(format!("{},{},{},{},{},{path},{}", row.id, self.data.classes[leaf.prediction], self.data.classes[row.label], self.data.classes[self.grown.predict(row, &self.data.attributes)], leaf.number, route.methods.iter().filter(|&&method| method != Method::Primary).count()));
        }
        lines
    }
    fn model_evaluation(&self, name: &str) -> Vec<String> { classification_lines(&format!("{name} test"), &self.data.classes, &self.data.test, &self.predictions[name]) }
    fn routing_lines(&self) -> Vec<String> {
        let mut tallies = vec![[0usize; 3]; self.pruned.tree.nodes.len()];
        for row in &self.data.test {
            let route = self.pruned.route(row, &self.data.attributes);
            for (&index, method) in route.path.iter().zip(route.methods) { tallies[index][method.index()] += 1; }
        }
        let mut lines = vec!["routing pruned test".to_owned()];
        for index in self.pruned.internal_nodes() {
            let counts = tallies[index];
            lines.push(format!("node {} rows {} primary {} surrogate {} default {}", self.pruned.tree.nodes[index].number, counts.iter().sum::<usize>(), counts[0], counts[1], counts[2]));
        }
        for (name, complete) in [("complete", true), ("incomplete", false)] {
            let rows: Vec<_> = self.data.test.iter().filter(|row| (row.missing_count() == 0) == complete).cloned().collect();
            lines.push(format!("{name} rows {} correct {}", rows.len(), self.accuracy(&self.pruned, &rows)));
        }
        lines
    }
    fn evaluation_file(&self) -> Vec<String> {
        join_sections(vec![self.data_lines(), vec![self.summary_line("grown", &self.grown), self.summary_line("pruned", &self.pruned)], self.model_evaluation("grown"), self.model_evaluation("pruned"), self.routing_lines(), self.cross_validation_lines(), self.depth_lines(), self.cost_lines(), self.bagging_lines(), self.mcnemar_lines(), self.auc_lines(), self.learning_lines()])
    }
    let data = load_dataset("input")?;
    let analysis = Analysis::new(&data);
    fs::create_dir_all("output")?;
    for (name, lines) in [("tree.txt", analysis.tree_file()), ("pruned_tree.txt", analysis.pruned_tree_file()), ("predictions.csv", analysis.predictions_file()), ("rules.txt", analysis.rules_file()), ("evaluation.txt", analysis.evaluation_file()), ("importance.txt", analysis.importance_file())] {
        fs::write(format!("output/{name}"), lines.join("\n") + "\n")?;
    }
