function readLines(path) {
    return fs.readFileSync(path, 'ascii').slice(0, -1).split('\n');
}
function readDataset() {
    const schema = readLines('input/schema.txt').map(line => line.split(' '));
    const attributes = schema.slice(0, -1).map(fields => ({
        name: fields[1],
        kind: fields[2],
        values: fields.slice(3),
    }));
    const classFields = schema.at(-1);
    const target = classFields[1];
    const classes = classFields.slice(2);
    const classNumbers = new Map(classes.map((name, index) => [name, index]));
    const valueNumbers = attributes.map(attribute => new Map(
        attribute.values.map((name, index) => [name, index])
    ));
    function readRows(path) {
        return readLines(path).slice(1).map(line => {
            const fields = line.split(',');
            const values = attributes.map((attribute, index) => {
                const text = fields[index + 1];
                if (text === '?') return null;
                return attribute.kind === 'INT' ? Number(text) : valueNumbers[index].get(text);
            });
            return { id: fields[0], values, label: classNumbers.get(fields.at(-1)) };
        });
    }
    const params = new Map(readLines('input/params.txt').map(line => {
        const [name, ...values] = line.split(' ');
        return [name, values.map(Number)];
    }));
    const settings = {
        maxDepth: params.get('MAX_DEPTH')[0],
        minSamplesSplit: params.get('MIN_SAMPLES_SPLIT')[0],
        minSamplesLeaf: params.get('MIN_SAMPLES_LEAF')[0],
        maxSurrogates: params.get('MAX_SURROGATES')[0],
    };
    return {
        attributes, classes, target, settings,
        train: readRows('input/train.csv'),
        validation: readRows('input/validation.csv'),
        test: readRows('input/test.csv'),
    };
}
function majority(counts) {
    let selected = 0;
    for (let index = 1; index < counts.length; index++) {
        if (counts[index] > counts[selected]) selected = index;
    }
    return selected;
}
function statistics(rows, classCount) {
    const counts = Array(classCount).fill(0);
    for (const row of rows) counts[row.label]++;
    let sumSquares = 0;
    for (const count of counts) sumSquares += count * count;
    const predicted = majority(counts);
    return {
        n: rows.length,
        counts,
        sumSquares,
        majority: predicted,
        error: rows.length - counts[predicted],
    };
}
function evaluationValue(left, right) {
    return {
        numerator: left.sumSquares * right.n + right.sumSquares * left.n,
        denominator: left.n * right.n,
    };
}
function compareFractions(left, right) {
    const a = left.numerator * right.denominator;
    const b = right.numerator * left.denominator;
    return a < b ? -1 : a > b ? 1 : 0;
}
function improves(value, stats) {
    return value.numerator * stats.n > stats.sumSquares * value.denominator;
}
function candidates(rows, attribute, index) {
    const values = [...new Set(rows.map(row => row.values[index]))].sort((a, b) => a - b);
    if (values.length < 2) return [];
    if (attribute.kind === 'CAT') {
        return values.map(value => ({ attribute: index, kind: 'CAT', value }));
    }
    const result = [];
    for (let position = 0; position + 1 < values.length; position++) {
        result.push({
            attribute: index,
            kind: 'INT',
            threshold: values[position] + values[position + 1],
        });
    }
    return result;
}
function goesLeft(split, row) {
    const value = row.values[split.attribute];
    return split.kind === 'INT' ? 2 * value <= split.threshold : value === split.value;
}
function partition(rows, split) {
    const left = [];
    const right = [];
    for (const row of rows) (goesLeft(split, row) ? left : right).push(row);
    return { left, right };
}
function bestForAttribute(rows, attribute, index, minLeaf, classCount) {
    let best = null;
    for (const split of candidates(rows, attribute, index)) {
        const divided = partition(rows, split);
        if (divided.left.length < minLeaf || divided.right.length < minLeaf) continue;
        const leftStats = statistics(divided.left, classCount);
        const rightStats = statistics(divided.right, classCount);
        const value = evaluationValue(leftStats, rightStats);
        if (best === null || compareFractions(value, best.value) > 0) {
            best = { split, left: divided.left, right: divided.right, value };
        }
    }
    return best;
}
function bestSplit(rows, attributes, minLeaf, classCount) {
    let best = null;
    for (let index = 0; index < attributes.length; index++) {
        const candidate = bestForAttribute(rows, attributes[index], index, minLeaf, classCount);
        if (candidate !== null && (best === null || compareFractions(candidate.value, best.value) > 0)) {
            best = candidate;
        }
    }
    return best;
}
function findSurrogates(rows, primary, attributes, limit, leftCount, rightCount) {
    const retained = [];
    const primaryLeft = rows.map(row => goesLeft(primary, row));
    for (let index = 0; index < attributes.length; index++) {
        if (index === primary.attribute) continue;
        let best = null;
        for (const split of candidates(rows, attributes[index], index)) {
            let agreement = 0;
            for (let rowIndex = 0; rowIndex < rows.length; rowIndex++) {
                if (goesLeft(split, rows[rowIndex]) === primaryLeft[rowIndex]) agreement++;
            }
            const reversed = rows.length - agreement;
            const direction = agreement >= reversed ? 'left' : 'right';
            const score = Math.max(agreement, reversed);
            if (best === null || score > best.score) best = { split, direction, score };
        }
        if (best !== null && best.score > Math.max(leftCount, rightCount)) retained.push(best);
    }
    retained.sort((a, b) => b.score - a.score || a.split.attribute - b.split.attribute);
    return retained.slice(0, limit);
}
function growTree(rows, attributes, classCount, settings) {
    function grow(sequence, number, depth) {
        const stats = statistics(sequence, classCount);
        const node = {
            ...stats,
            rows: sequence,
            number,
            depth,
            split: null,
            left: null,
            right: null,
            surrogates: [],
            defaultDirection: 'left',
        };
        if (stats.error === 0 || depth === settings.maxDepth || sequence.length < settings.minSamplesSplit) {
            return node;
        }
        const best = bestSplit(sequence, attributes, settings.minSamplesLeaf, classCount);
        if (best === null || !improves(best.value, stats)) return node;
        node.split = best.split;
        node.defaultDirection = best.left.length >= best.right.length ? 'left' : 'right';
        node.surrogates = findSurrogates(
            sequence, best.split, attributes, settings.maxSurrogates,
            best.left.length, best.right.length
        );
        node.left = grow(best.left, 2 * number, depth + 1);
        node.right = grow(best.right, 2 * number + 1, depth + 1);
        return node;
    }
    return grow(rows, 1, 0);
}
function isLeaf(node, view) {
    return node.split === null || view.cuts.has(node.number);
}
function route(view, row) {
    const path = [];
    const steps = [];
    let node = view.root;
    while (true) {
        path.push(node);
        if (isLeaf(node, view)) return { leaf: node, prediction: node.majority, path, steps };
        let left;
        let method;
        if (row.values[node.split.attribute] !== null) {
            left = goesLeft(node.split, row);
            method = 'primary';
        } else {
            const surrogate = node.surrogates.find(item => row.values[item.split.attribute] !== null);
            if (surrogate !== undefined) {
                left = goesLeft(surrogate.split, row) === (surrogate.direction === 'left');
                method = 'surrogate';
            } else {
                left = node.defaultDirection === 'left';
                method = 'default';
            }
        }
        steps.push({ node, left, method });
        node = left ? node.left : node.right;
    }
}
function predict(view, row) {
    return route(view, row).prediction;
}
function correctCount(view, rows) {
    let correct = 0;
    for (const row of rows) if (predict(view, row) === row.label) correct++;
    return correct;
}
function classificationLines(title, rows, predictions, classes) {
    const count = classes.length;
    const matrix = Array.from({ length: count }, () => Array(count).fill(0));
    let correct = 0;
    for (let index = 0; index < rows.length; index++) {
        matrix[rows[index].label][predictions[index]]++;
        if (rows[index].label === predictions[index]) correct++;
    }
    const lines = [
        `confusion ${title} correct ${correct}/${rows.length} accuracy ${reducedFraction(correct, rows.length)}`,
        `actual\\predicted ${classes.join(' ')}`,
    ];
    for (let index = 0; index < count; index++) lines.push(`${classes[index]} ${matrix[index].join(' ')}`);
    let macro = 0;
    let defined = 0;
    for (let index = 0; index < count; index++) {
        const truePositive = matrix[index][index];
        let predicted = 0;
        let actual = 0;
        for (let other = 0; other < count; other++) {
            predicted += matrix[other][index];
            actual += matrix[index][other];
        }
        const precision = reducedFraction(truePositive, predicted);
        const recall = reducedFraction(truePositive, actual);
        const f1 = reducedFraction(2 * truePositive, predicted + actual);
        lines.push(`class ${classes[index]} precision ${precision} recall ${recall} f1 ${f1} support ${actual}`);
        if (predicted + actual > 0) {
            macro += (2 * truePositive) / (predicted + actual);
            defined++;
        }
    }
    lines.push(`macro_f1 ${defined === 0 ? '-' : formatReal(macro / defined)}`);
    return lines;
}
function reducedError(root, validation) {
    const original = { root, cuts: new Set() };
    const visits = new Map(preorder(original).map(node => [node.number, []]));
    for (const row of validation) {
        for (const node of route(original, row).path) visits.get(node.number).push(row);
    }
    const cuts = new Set();
    const decisions = [];
    function visit(node) {
        const rows = visits.get(node.number);
        let leafError = 0;
        for (const row of rows) if (row.label !== node.majority) leafError++;
        if (node.split === null) return leafError;
        const leftError = visit(node.left);
        const rightError = visit(node.right);
        const subtreeError = leftError + rightError;
        const pruned = leafError <= subtreeError;
        if (pruned) cuts.add(node.number);
        decisions.push({ node, rows: rows.length, leafError, subtreeError, pruned });
        return pruned ? leafError : subtreeError;
    }
    visit(root);
    return { view: { root, cuts }, decisions };
}
function predictionLines(data, grownRoutes, prunedRoutes) {
    const lines = ['id,predicted,actual,grown,leaf,path,missing'];
    for (let index = 0; index < data.test.length; index++) {
        const row = data.test[index];
        const result = prunedRoutes[index];
        let missing = 0;
        for (const step of result.steps) if (step.method !== 'primary') missing++;
        lines.push([
            row.id, data.classes[result.prediction], data.classes[row.label],
            data.classes[grownRoutes[index].prediction], result.leaf.number,
            result.path.map(node => node.number).join('-'), missing,
        ].join(','));
    }
    return lines;
}
function routingLines(data, pruned, routes) {
    const nodes = preorder(pruned).filter(node => !isLeaf(node, pruned));
    const tallies = new Map(nodes.map(node => [node.number, { rows: 0, primary: 0, surrogate: 0, default: 0 }]));
    const groups = [{ rows: 0, correct: 0 }, { rows: 0, correct: 0 }];
    for (let index = 0; index < routes.length; index++) {
        const result = routes[index];
        for (const step of result.steps) {
            const tally = tallies.get(step.node.number);
            tally.rows++;
            tally[step.method]++;
        }
        const group = groups[data.test[index].values.includes(null) ? 1 : 0];
        group.rows++;
        if (result.prediction === data.test[index].label) group.correct++;
    }
    const lines = ['routing pruned test'];
    for (const node of nodes) {
        const tally = tallies.get(node.number);
        lines.push(`node ${node.number} rows ${tally.rows} primary ${tally.primary} surrogate ${tally.surrogate} default ${tally.default}`);
    }
    for (let index = 0; index < groups.length; index++) {
        const group = groups[index];
        lines.push(`${index === 0 ? 'complete' : 'incomplete'} rows ${group.rows} correct ${ratio(group.correct, group.rows)}`);
    }
    return lines;
}
function writeBlocks(name, blocks) {
    const text = blocks.map(lines => lines.join('\n')).join('\n\n') + '\n';
    fs.writeFileSync(`output/${name}`, text, 'ascii');
}

    const root = growTree(data.train, data.attributes, data.classes.length, data.settings);
    const grown = { root, cuts: new Set() };

    const pruned = pruning.view;
