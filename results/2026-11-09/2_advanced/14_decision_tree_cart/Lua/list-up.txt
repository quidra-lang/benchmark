function M.read_lines(path)
    local lines = {}
    for line in io.lines(path) do lines[#lines + 1] = line end
    return lines
end
local function fields(line, separator)
    local result = {}
    for value in (line .. separator):gmatch("(.-)" .. separator) do result[#result + 1] = value end
    return result
end
function M.load(directory)
    local attributes, classes, target = {}, {}
    for _, line in ipairs(M.read_lines(directory .. "/schema.txt")) do
        local parts = fields(line, " ")
        if parts[1] == "ATTRIBUTE" then
            local attribute = {index = #attributes + 1, name = parts[2], integer = parts[3] == "INT", values = {}, positions = {}}
            for index = 4, #parts do
                attribute.values[#attribute.values + 1] = parts[index]
                attribute.positions[parts[index]] = #attribute.values
            end
            attributes[#attributes + 1] = attribute
        else
            target = parts[2]
            for index = 3, #parts do classes[#classes + 1] = parts[index] end
        end
    end
    local class_positions = {}
    for index, name in ipairs(classes) do class_positions[name] = index end
    local function read_rows(name)
        local result = {}
        for index, line in ipairs(M.read_lines(directory .. "/" .. name .. ".csv")) do
            if index > 1 then
                local parts, values = fields(line, ","), {}
                for _, attribute in ipairs(attributes) do
                    local text = parts[attribute.index + 1]
                    local value = false
                    if text ~= "?" then
                        value = attribute.integer and tonumber(text) or attribute.positions[text]
                    end
                    values[attribute.index] = value
                end
                result[#result + 1] = {id = parts[1], values = values, label = class_positions[parts[#parts]]}
            end
        end
        return result
    end
    local settings = {}
    for _, line in ipairs(M.read_lines(directory .. "/params.txt")) do
        local parts = fields(line, " ")
        local key = parts[1]:lower()
        local values = {}
        for index = 2, #parts do values[#values + 1] = tonumber(parts[index]) end
        settings[key] = key == "curve" and values or values[1]
    end
    return {attributes = attributes, target = target, classes = classes, settings = settings,
        train = read_rows("train"), validation = read_rows("validation"), test = read_rows("test")}
end
function M.class_counts(rows, size)
    local counts = {}
    for index = 1, size do counts[index] = 0 end
    for _, row in ipairs(rows) do counts[row.label] = counts[row.label] + 1 end
    return counts
end
function M.sum(values)
    local total = 0
    for _, value in ipairs(values) do total = total + value end
    return total
end
function M.square_sum(counts)
    local total = 0
    for _, value in ipairs(counts) do total = total + value * value end
    return total
end
function M.majority(counts)
    local best = 1
    for index = 2, #counts do
        if counts[index] > counts[best] then best = index end
    end
    return best
end
function M.evaluation_value(left, right)
    local nl, nr = M.sum(left), M.sum(right)
    return {M.square_sum(left) * nr + M.square_sum(right) * nl, nl * nr}
end
function M.compare(first, second)
    local a, b = first[1] * second[2], second[1] * first[2]
    return a < b and -1 or (a > b and 1 or 0)
end
function M.improves(value, counts)
    return value[1] * M.sum(counts) > M.square_sum(counts) * value[2]
end
local Split = {}
Split.__index = Split
function Split:goes_left(value)
    if self.attribute.integer then return 2 * value <= self.point end
    return value == self.point
end
function M.candidates(attribute, rows)
    local present, values = {}, {}
    for _, row in ipairs(rows) do
        local value = row.values[attribute.index]
        if not present[value] then present[value] = true; values[#values + 1] = value end
    end
    table.sort(values)
    local result = {}
    if #values < 2 then return result end
    for index, value in ipairs(values) do
        if not attribute.integer or index < #values then
            local point = attribute.integer and (value + values[index + 1]) or value
            result[#result + 1] = setmetatable({attribute = attribute, point = point}, Split)
        end
    end
    return result
end
local Builder = {}
Builder.__index = Builder
function M.new(attributes, class_count, settings)
    return setmetatable({attributes = attributes, class_count = class_count, settings = settings}, Builder)
end
function Builder:best_split(rows, attributes)
    local best
    for _, attribute in ipairs(attributes) do
        for _, split in ipairs(M.candidates(attribute, rows)) do
            local left, right = {}, {}
            for _, row in ipairs(rows) do
                local group = split:goes_left(row.values[attribute.index]) and left or right
                group[#group + 1] = row
            end
            if #left >= self.settings.min_samples_leaf and #right >= self.settings.min_samples_leaf then
                local lc = impurity.class_counts(left, self.class_count)
                local rc = impurity.class_counts(right, self.class_count)
                local value = impurity.evaluation_value(lc, rc)
                if not best or impurity.compare(value, best.value) > 0 then
                    best = {split = split, left = left, right = right, value = value}
                end
            end
        end
    end
    return best
end
function Builder:find_surrogates(rows, primary, left_size, right_size)
    local result = {}
    for _, attribute in ipairs(self.attributes) do
        if attribute ~= primary.attribute then
            local best
            for _, split in ipairs(M.candidates(attribute, rows)) do
                local agreement = 0
                for _, row in ipairs(rows) do
                    if split:goes_left(row.values[attribute.index]) == primary:goes_left(row.values[primary.attribute.index]) then
                        agreement = agreement + 1
                    end
                end
                local score = math.max(agreement, #rows - agreement)
                if not best or score > best.score then
                    best = {split = split, direction = agreement >= #rows - agreement and "left" or "right", score = score}
                end
            end
            if best and best.score > math.max(left_size, right_size) then result[#result + 1] = best end
        end
    end
    table.sort(result, function(a, b)
        if a.score ~= b.score then return a.score > b.score end
        return a.split.attribute.index < b.split.attribute.index
    end)
    while #result > self.settings.max_surrogates do table.remove(result) end
    return result
end
function Builder:grow(rows, number, depth)
    number, depth = number or 1, depth or 0
    local counts = impurity.class_counts(rows, self.class_count)
    local prediction = impurity.majority(counts)
    local node = {number = number, depth = depth, counts = counts, size = #rows,
        prediction = prediction, errors = #rows - counts[prediction], surrogates = {}}
    if node.errors == 0 or depth == self.settings.max_depth or #rows < self.settings.min_samples_split then return node end
    local best = self:best_split(rows, self.attributes)
    if not best or not impurity.improves(best.value, counts) then return node end
    node.split = best.split
    node.default = #best.left >= #best.right and "left" or "right"
    node.surrogates = self:find_surrogates(rows, best.split, #best.left, #best.right)
    node.left = self:grow(best.left, 2 * number, depth + 1)
    node.right = self:grow(best.right, 2 * number + 1, depth + 1)
    return node
end
function M.view(root, cut)
    return setmetatable({root = root, cut = cut or {}}, View)
end
function View:is_leaf(node)
    return not node.split or self.cut[node.number] == true
end
function View:route(row)
    local node, path, steps = self.root, {}, {}
    while true do
        path[#path + 1] = node
        if self:is_leaf(node) then return {leaf = node, path = path, steps = steps} end
        local value = row.values[node.split.attribute.index]
        local left, method
        if value ~= false then
            left, method = node.split:goes_left(value), "primary"
        else
            for _, surrogate in ipairs(node.surrogates) do
                value = row.values[surrogate.split.attribute.index]
                if value ~= false then
                    left = surrogate.split:goes_left(value) == (surrogate.direction == "left")
                    method = "surrogate"
                    break
                end
            end
            if not method then left, method = node.default == "left", "default" end
        end
        steps[#steps + 1] = {node = node, left = left, method = method}
        node = left and node.left or node.right
    end
end
function View:predict(row)
    return self:route(row).leaf.prediction
end
function View:correct_count(rows)
    local correct = 0
    for _, row in ipairs(rows) do
        if self:predict(row) == row.label then correct = correct + 1 end
    end
    return correct
end
function M.classification(title, classes, rows, predictions)
    local matrix, correct = {}, 0
    for a = 1, #classes do
        matrix[a] = {}
        for b = 1, #classes do matrix[a][b] = 0 end
    end
    for index, row in ipairs(rows) do
        matrix[row.label][predictions[index]] = matrix[row.label][predictions[index]] + 1
        if row.label == predictions[index] then correct = correct + 1 end
    end
    local lines = {string.format("confusion %s correct %d/%d accuracy %s", title, correct, #rows, M.fraction(correct, #rows)),
        "actual\\predicted " .. table.concat(classes, " ")}
    for index, name in ipairs(classes) do lines[#lines + 1] = name .. " " .. table.concat(matrix[index], " ") end
    local total, available = 0.0, 0
    for index, name in ipairs(classes) do
        local predicted, support, hits = 0, 0, matrix[index][index]
        for other = 1, #classes do
            predicted = predicted + matrix[other][index]
            support = support + matrix[index][other]
        end
        lines[#lines + 1] = string.format("class %s precision %s recall %s f1 %s support %d", name,
            M.fraction(hits, predicted), M.fraction(hits, support), M.fraction(2 * hits, predicted + support), support)
        if predicted + support > 0 then total, available = total + 2 * hits / (predicted + support), available + 1 end
    end
    lines[#lines + 1] = "macro_f1 " .. (available > 0 and M.real(total / available) or "-")
    return lines
end
function M.reduced_error(root, rows)
    local original, reached, errors = evaluation.view(root), {}, {}
    for _, row in ipairs(rows) do
        for _, node in ipairs(original:route(row).path) do
            reached[node.number] = (reached[node.number] or 0) + 1
            if row.label ~= node.prediction then errors[node.number] = (errors[node.number] or 0) + 1 end
        end
    end
    local cut, decisions = {}, {}
    local function visit(node)
        local leaf_error = errors[node.number] or 0
        if not node.split then return leaf_error end
        local left_error = visit(node.left)
        local subtree_error = left_error + visit(node.right)
        local pruned = leaf_error <= subtree_error
        if pruned then cut[node.number] = true end
        decisions[#decisions + 1] = {node = node, rows = reached[node.number] or 0,
            leaf_error = leaf_error, subtree_error = subtree_error, pruned = pruned}
        return pruned and leaf_error or subtree_error
    end
    visit(root)
    return evaluation.view(root, cut), decisions
end
local function missing_count(row)
    local count = 0
    for _, value in ipairs(row.values) do if value == false then count = count + 1 end end
    return count
end
local Study = {}
Study.__index = Study
function Study.new(data)
    local self = setmetatable({data = data, attributes = data.attributes, classes = data.classes,
        train = data.train, validation = data.validation, test = data.test, settings = data.settings}, Study)
    self.builder = builder_module.new(self.attributes, #self.classes, self.settings)
    self.grown = evaluation.view(self.builder:grow(self.train))
    self.pruned, self.decisions = pruning.reduced_error(self.grown.root, self.validation)
    self.predictions = {}
    local models = {grown = self.grown, pruned = self.pruned, depth = self.selected_depth.tree,
        cost_complexity = self.sequence[self.selected_index].tree}
    for name, tree in pairs(models) do
        local predictions = {}
        for _, row in ipairs(self.test) do predictions[#predictions + 1] = tree:predict(row) end
        self.predictions[name] = predictions
    end
    return self
function Study:accuracy(tree, rows)
    return evaluation.ratio(tree:correct_count(rows), #rows)
end
function Study:predictions_file()
    local lines = {"id,predicted,actual,grown,leaf,path,missing"}
    for index, row in ipairs(self.test) do
        local route, path, missing = self.pruned:route(row), {}, 0
        for _, node in ipairs(route.path) do path[#path + 1] = node.number end
        for _, step in ipairs(route.steps) do if step.method ~= "primary" then missing = missing + 1 end end
        lines[#lines + 1] = table.concat({row.id, self.classes[route.leaf.prediction], self.classes[row.label],
            self.classes[self.predictions.grown[index]], route.leaf.number, table.concat(path, "-"), missing}, ",")
    end
    return lines
end
function Study:routing_lines()
    local tallies, lines = {}, {"routing pruned test"}
    for _, node in ipairs(self.pruned:internal_nodes()) do tallies[node.number] = {primary = 0, surrogate = 0, default = 0} end
    local complete, incomplete = {}, {}
    for _, row in ipairs(self.test) do
        for _, step in ipairs(self.pruned:route(row).steps) do
            local tally = tallies[step.node.number]
            tally[step.method] = tally[step.method] + 1
        end
        local group = missing_count(row) == 0 and complete or incomplete
        group[#group + 1] = row
    end
    for _, node in ipairs(self.pruned:internal_nodes()) do
        local tally = tallies[node.number]
        lines[#lines + 1] = format("node %d rows %d primary %d surrogate %d default %d", node.number,
            tally.primary + tally.surrogate + tally.default, tally.primary, tally.surrogate, tally.default)
    end
    lines[#lines + 1] = format("complete rows %d correct %s", #complete, self:accuracy(self.pruned, complete))
    lines[#lines + 1] = format("incomplete rows %d correct %s", #incomplete, self:accuracy(self.pruned, incomplete))
    return lines
end
function Study:model_evaluation(model)
    return evaluation.classification(model .. " test", self.classes, self.test, self.predictions[model])
end

end
