struct Attribute {
    let name: String
    let integer: Bool
    let values: [String]
}
struct Row {
    let id: String
    var values: [Int64?]
    let label: Int
    var missingCount: Int { values.filter { $0 == nil }.count }
}
struct Settings {
    let maxDepth: Int
    let minSplit: Int
    let minLeaf: Int
    let maxSurrogates: Int
func readLines(_ path: String) throws -> [String] {
    try String(contentsOfFile: path, encoding: .utf8).split(separator: "\n").map(String.init)
}
func readRows(_ path: String, _ attributes: [Attribute], _ classes: [String]) throws -> [Row] {
    try readLines(path).dropFirst().map { line in
        let fields = line.split(separator: ",").map(String.init)
        let values: [Int64?] = attributes.enumerated().map { index, attribute in
            let text = fields[index + 1]
            if text == "?" { return nil }
            return attribute.integer ? Int64(text)! : Int64(attribute.values.firstIndex(of: text)!)
        }
        return Row(id: fields[0], values: values, label: classes.firstIndex(of: fields.last!)!)
    }
}
func loadDataset(_ directory: String) throws -> Dataset {
    var attributes: [Attribute] = []
    var target = ""
    var classes: [String] = []
    for line in try readLines("\(directory)/schema.txt") {
        let fields = line.split(separator: " ").map(String.init)
        if fields[0] == "ATTRIBUTE" {
            attributes.append(Attribute(name: fields[1], integer: fields[2] == "INT", values: Array(fields.dropFirst(3))))
        } else {
            target = fields[1]
            classes = Array(fields.dropFirst(2))
        }
    }
    let parameters = try readLines("\(directory)/params.txt").map { line in
        line.split(separator: " ").dropFirst().map { Int64($0)! }
    }
    let settings = Settings(maxDepth: Int(parameters[0][0]), minSplit: Int(parameters[1][0]),
        minLeaf: Int(parameters[2][0]), maxSurrogates: Int(parameters[3][0]), folds: Int(parameters[4][0]),
        curve: parameters[5].map(Int.init), bagTrees: Int(parameters[6][0]), bagSeed: parameters[7][0],
        repeats: Int(parameters[8][0]), permutationSeed: parameters[9][0])
    return Dataset(attributes: attributes, target: target, classes: classes,
        train: try readRows("\(directory)/train.csv", attributes, classes),
        validation: try readRows("\(directory)/validation.csv", attributes, classes),
        test: try readRows("\(directory)/test.csv", attributes, classes), settings: settings)
}
struct Fraction {
    let numerator: Int64
    let denominator: Int64
    func compare(_ other: Fraction) -> Int {
        let first = numerator * other.denominator
        let second = other.numerator * denominator
        return first == second ? 0 : first < second ? -1 : 1
    }
func classCounts(_ rows: [Row], _ classCount: Int) -> [Int64] {
    var counts = Array<Int64>(repeating: 0, count: classCount)
    for row in rows { counts[row.label] += 1 }
    return counts
}
func squareSum(_ counts: [Int64]) -> Int64 { counts.reduce(0) { $0 + $1 * $1 } }
func majority(_ counts: [Int64]) -> Int {
    var best = 0
    for index in counts.indices.dropFirst() where counts[index] > counts[best] { best = index }
    return best
}
func errorCount(_ counts: [Int64]) -> Int64 { counts.reduce(0, +) - counts[majority(counts)] }
func evaluationValue(_ left: [Int64], _ right: [Int64]) -> Fraction {
    let leftSize = left.reduce(0, +)
    let rightSize = right.reduce(0, +)
    return Fraction(numerator: squareSum(left) * rightSize + squareSum(right) * leftSize,
        denominator: leftSize * rightSize)
}
func improves(_ value: Fraction, _ counts: [Int64]) -> Bool {
    value.numerator * counts.reduce(0, +) > squareSum(counts) * value.denominator
}
struct Split {
    let attribute: Int
    let point: Int64
    func goesLeft(_ value: Int64, _ attributes: [Attribute]) -> Bool {
        attributes[attribute].integer ? 2 * value <= point : value == point
    }
struct Surrogate {
    let split: Split
    let forward: Bool
    let score: Int64
}
struct Candidate {
    let split: Split
    let left: [Int64]
    let right: [Int64]
    var value: Fraction { evaluationValue(left, right) }
    var leftSize: Int64 { left.reduce(0, +) }
    var rightSize: Int64 { right.reduce(0, +) }
}
struct Node {
    let number: Int
    let depth: Int
    let counts: [Int64]
    let size: Int64
    let prediction: Int
    let errors: Int64
    var split: Split? = nil
    var surrogates: [Surrogate] = []
    var defaultLeft = true
    var children: (left: Int, right: Int)? = nil
}
func candidates(_ attribute: Int, _ attributes: [Attribute], _ rows: [Row], _ labels: [Int], _ classes: Int) -> [Candidate] {
    let values = Set(rows.map { $0.values[attribute]! }).sorted()
    guard values.count > 1 else { return [] }
    let points = attributes[attribute].integer ? zip(values, values.dropFirst()).map { first, second in first + second } : values
    return points.map { point in
        let split = Split(attribute: attribute, point: point)
        var left = Array<Int64>(repeating: 0, count: classes)
        var right = Array<Int64>(repeating: 0, count: classes)
        for (row, label) in zip(rows, labels) {
            if split.goesLeft(row.values[attribute]!, attributes) { left[label] += 1 }
            else { right[label] += 1 }
        }
        return Candidate(split: split, left: left, right: right)
    }
}
struct TreeBuilder {
    let attributes: [Attribute]
    let classes: Int
    let settings: Settings
    func bestSplit(_ rows: [Row], _ selectedAttributes: [Int]) -> Candidate? {
        let labels = rows.map(\.label)
        var best: Candidate?
        for attribute in selectedAttributes {
            for candidate in candidates(attribute, attributes, rows, labels, classes) {
                if candidate.leftSize < Int64(settings.minLeaf) || candidate.rightSize < Int64(settings.minLeaf) { continue }
                if best == nil || candidate.value.compare(best!.value) > 0 { best = candidate }
            }
        }
        return best
    }
    func findSurrogates(_ rows: [Row], _ primary: Split) -> [Surrogate] {
        let labels = rows.map { primary.goesLeft($0.values[primary.attribute]!, attributes) ? 0 : 1 }
        let left = Int64(labels.filter { $0 == 0 }.count)
        let size = Int64(rows.count)
        let baseline = max(left, size - left)
        var selected: [Surrogate] = []
        for attribute in attributes.indices where attribute != primary.attribute {
            var best: Surrogate?
            for candidate in candidates(attribute, attributes, rows, labels, 2) {
                let agreement = candidate.left[0] + candidate.right[1]
                let forward = agreement >= size - agreement
                let score = max(agreement, size - agreement)
                if best == nil || score > best!.score { best = Surrogate(split: candidate.split, forward: forward, score: score) }
            }
            if let best, best.score > baseline { selected.append(best) }
        }
        selected.sort { $0.score == $1.score ? $0.split.attribute < $1.split.attribute : $0.score > $1.score }
        return Array(selected.prefix(settings.maxSurrogates))
    }
    func grow(_ rows: [Row]) -> [Node] {
        var nodes: [Node] = []
        growNode(rows, 1, 0, &nodes)
        return nodes
    }
    @discardableResult
    private func growNode(_ rows: [Row], _ number: Int, _ depth: Int, _ nodes: inout [Node]) -> Int {
        let counts = classCounts(rows, classes)
        let predicted = majority(counts)
        let index = nodes.count
        nodes.append(Node(number: number, depth: depth, counts: counts, size: Int64(rows.count),
            prediction: predicted, errors: errorCount(counts)))
        if counts[predicted] == Int64(rows.count) || depth == settings.maxDepth || rows.count < settings.minSplit { return index }
        guard let best = bestSplit(rows, Array(attributes.indices)), improves(best.value, counts) else { return index }
        let split = best.split
        let left = rows.filter { split.goesLeft($0.values[split.attribute]!, attributes) }
        let right = rows.filter { !split.goesLeft($0.values[split.attribute]!, attributes) }
        nodes[index].split = split
        nodes[index].surrogates = findSurrogates(rows, split)
        nodes[index].defaultLeft = left.count >= right.count
        let leftChild = growNode(left, number * 2, depth + 1, &nodes)
        let rightChild = growNode(right, number * 2 + 1, depth + 1, &nodes)
        nodes[index].children = (leftChild, rightChild)
        return index
    }
}
enum Method: Int {
    case primary, surrogate, `default`
struct Route {
    let path: [Int]
    let methods: [Method]
}
struct PrunedTree {
    let nodes: [Node]
    var cut: Set<Int> = []
    func isLeaf(_ index: Int) -> Bool { nodes[index].split == nil || cut.contains(index) }
    func route(_ row: Row, _ attributes: [Attribute]) -> Route {
        var index = 0
        var path = [index]
        var methods: [Method] = []
        while !isLeaf(index) {
            let node = nodes[index]
            let split = node.split!
            var left = node.defaultLeft
            var method = Method.default
            if let value = row.values[split.attribute] {
                left = split.goesLeft(value, attributes)
                method = .primary
            } else {
                for surrogate in node.surrogates {
                    if let value = row.values[surrogate.split.attribute] {
                        left = surrogate.split.goesLeft(value, attributes) == surrogate.forward
                        method = .surrogate
                        break
                    }
                }
            }
            index = left ? node.children!.left : node.children!.right
            path.append(index)
            methods.append(method)
        }
        return Route(path: path, methods: methods)
    }
    func leafOf(_ row: Row, _ attributes: [Attribute]) -> Int { route(row, attributes).path.last! }
    func predict(_ row: Row, _ attributes: [Attribute]) -> Int { nodes[leafOf(row, attributes)].prediction }
    func correctCount(_ rows: [Row], _ attributes: [Attribute]) -> Int64 {
        Int64(rows.filter { predict($0, attributes) == $0.label }.count)
    }
func classificationLines(_ title: String, _ classes: [String], _ rows: [Row], _ predictions: [Int]) -> [String] {
    var matrix = Array(repeating: Array<Int64>(repeating: 0, count: classes.count), count: classes.count)
    for (row, predicted) in zip(rows, predictions) { matrix[row.label][predicted] += 1 }
    let correct = classes.indices.reduce(Int64(0)) { $0 + matrix[$1][$1] }
    let total = Int64(rows.count)
    var lines = ["confusion \(title) correct \(correct)/\(total) accuracy \(fractionText(correct, total))",
        "actual\\predicted " + classes.joined(separator: " ")]
    for (name, counts) in zip(classes, matrix) { lines.append(name + " " + counts.map(String.init).joined(separator: " ")) }
    var f1Sum = 0.0
    var f1Count = 0
    for (index, name) in classes.enumerated() {
        let hits = matrix[index][index]
        let predicted = matrix.reduce(Int64(0)) { $0 + $1[index] }
        let support = matrix[index].reduce(0, +)
        if predicted + support > 0 {
            f1Sum += Double(2 * hits) / Double(predicted + support)
            f1Count += 1
        }
        lines.append("class \(name) precision \(fractionText(hits, predicted)) recall \(fractionText(hits, support)) f1 \(fractionText(2 * hits, predicted + support)) support \(support)")
    }
    lines.append("macro_f1 \(realText(f1Sum / Double(f1Count)))")
    return lines
}
private func decide(_ index: Int, _ nodes: [Node], _ reached: [[Int]], _ rows: [Row], _ decisions: inout [Decision], _ cut: inout Set<Int>) -> Int64 {
    let node = nodes[index]
    let leafError = Int64(reached[index].filter { rows[$0].label != node.prediction }.count)
    guard let children = node.children else { return leafError }
    let leftError = decide(children.left, nodes, reached, rows, &decisions, &cut)
    let rightError = decide(children.right, nodes, reached, rows, &decisions, &cut)
    let subtreeError = leftError + rightError
    let pruned = leafError <= subtreeError
    decisions.append(Decision(node: index, rows: reached[index].count, leafError: leafError, subtreeError: subtreeError, pruned: pruned))
    if pruned { cut.insert(index) }
    return pruned ? leafError : subtreeError
}
func reducedErrorPruning(_ nodes: [Node], _ rows: [Row], _ attributes: [Attribute]) -> (tree: PrunedTree, decisions: [Decision]) {
    let grown = PrunedTree(nodes: nodes)
    var reached = Array(repeating: [Int](), count: nodes.count)
    for (index, row) in rows.enumerated() { for node in grown.route(row, attributes).path { reached[node].append(index) } }
    var decisions: [Decision] = []
    var cut: Set<Int> = []
    _ = decide(0, nodes, reached, rows, &decisions, &cut)
    return (PrunedTree(nodes: nodes, cut: cut), decisions)
}
    init(_ data: Dataset) {
        self.data = data

    func accuracy(_ tree: PrunedTree, _ rows: [Row]) -> String {

    }
    func predictionsFile() -> [String] {
        var lines = ["id,predicted,actual,grown,leaf,path,missing"]
        for row in data.test {
            let route = pruned.route(row, data.attributes)
            let leaf = pruned.nodes[route.path.last!]
            let path = route.path.map { String(pruned.nodes[$0].number) }.joined(separator: "-")
            let missing = route.methods.filter { $0 != .primary }.count
            lines.append("\(row.id),\(data.classes[leaf.prediction]),\(data.classes[row.label]),\(data.classes[grown.predict(row, data.attributes)]),\(leaf.number),\(path),\(missing)")
        }
        return lines
    }
    func modelEvaluation(_ name: String) -> [String] {
        classificationLines("\(name) test", data.classes, data.test, predictions[name]!)
    }
    func routingLines() -> [String] {
        var tallies = Array(repeating: [0, 0, 0], count: pruned.nodes.count)
        for row in data.test {
            let route = pruned.route(row, data.attributes)
            for (index, method) in zip(route.path, route.methods) { tallies[index][method.rawValue] += 1 }
        }
        var lines = ["routing pruned test"]
        for index in pruned.internalNodes() {
            let counts = tallies[index]
            lines.append("node \(pruned.nodes[index].number) rows \(counts.reduce(0, +)) primary \(counts[0]) surrogate \(counts[1]) default \(counts[2])")
        }
        for (name, complete) in [("complete", true), ("incomplete", false)] {
            let rows = data.test.filter { ($0.missingCount == 0) == complete }
            lines.append("\(name) rows \(rows.count) correct \(accuracy(pruned, rows))")
        }
        return lines
    }
