int register(string text)
    if #text < 2 or text[0] != "r" or #text > 3
        return -1
    digits := text[1:<int(#text)]
    if #digits > 1 and digits[0] == "0"
        return -1
    result := 0
    for: position < int(#digits)
        character := digits[position]
        digit := -1
        for: i < 10
            if string(i) == character
                digit = i
        if digit < 0
            return -1
        result = 10 * result + digit
    return if result <= 31 then result else -1

int[][][] dependencies(int[][] instructions)
    last := [for i := (0:<32)[] then -1]
    int[][][] result := []
    for: i < int(#instructions)
        int[][] operands := []
        for: 3 <= field <= 4 step 1
            source := instructions[i][field]
            if source > 0
                operands = operands.append([source, last[source]])
        result = result.append(operands)
        destination := instructions[i][2]
        if destination > 0
            last[destination] = i
    return result

bool id_resolved(int[] instruction, string stage)
    return (instruction[1] == 5 and instruction[3] < 0) or ((instruction[1] == 4 or instruction[1] == 5) and stage == "ID")

int ready(int[] instruction, int[] producer, int[] timing, bool forwarding, string stage)
    if not forwarding
        return timing[4] + 1
    result := if producer[1] == 2 then timing[4] else timing[3]
    return result + (if id_resolved(instruction, stage) then 1 else 0)
int maximum(int a, int b)
    return if a > b then a else b

class Result(int[][] timings, int[][] records, int[] delays, int cycles)

Result simulate(int[][] instructions, int[][][] sources, bool forwarding, string stage, int latency, bool[] missed)
    int[][] timings := []
    int[][] records := []
    int[] delays := []
    for: i < int(#instructions)
        instruction := instructions[i]
        length := if instruction[1] == 1 then latency else 1
        fetch := 1
        decode := 2
        start := 3
        if i > 0
            previous := timings[i - 1]
            fetch = previous[1]
            if missed[i - 1]
                fetch = previous[2] + (if hazards.id_resolved(instructions[i - 1], stage) then 0 else 1)
            decode = maximum(fetch + 1, previous[2])
            ideal := previous[2] + 1
            available := [ideal, ideal]
            producers := [-1, -1]
            for source := sources[i][]
                p := source[1]
                if p >= 0
                    cycle := hazards.ready(instruction, instructions[p], timings[p], forwarding, stage)
                    slot := if instructions[p][1] == 2 then 0 else 1
                    if cycle > available[slot]
                        available[slot] = cycle
                        producers[slot] = p
            bounds := [decode + 1, previous[3], available[0], available[1]]
            start = ideal
            for: cause < 4
                if bounds[cause] > start
                    records = records.append([cause, bounds[cause] - start, (if cause < 2 then i - 1 else i), i, (if cause < 2 then -1 else producers[cause - 2])])
                    start = bounds[cause]
            delays = delays.append(start - ideal)
        timings = timings.append([fetch, decode, start, start + length, start + length + 1])
    last := int(#instructions) - 1
    if instructions[last][1] == 1 and latency >= 2
        records = records.append([1, latency - 1, last, -1, -1])
    return Result(timings, records, delays, timings[last][4])

Trace validate(string[] lines)
    int[][] instructions := []
    string[] rejected := []
    string[] gaps := []
    map<int, int> first
    for: i < int(#lines)
        result := check(lines[i], first, instructions)
        if result.reason != ""
            rejected = rejected.append("reject line {i + 1} {result.reason}")
        else
            instruction := result.instruction
            if not first.has(instruction[0])
                .first.set(instruction[0], int(#instructions))
            instructions = instructions.append(instruction)
    return Trace(int(#lines), instructions, rejected, gaps, first)
int[][] totals(int[][] records)
    int[][] result := [[0, 0, 0, 0], [0, 0, 0, 0]]
    for record := records[]
        result[0][record[0]] += record[1]
        result[1][record[0]] += 1
    return result
