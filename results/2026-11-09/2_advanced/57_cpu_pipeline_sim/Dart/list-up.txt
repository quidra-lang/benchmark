void dependences(List<Instruction> a) {
  final last = List.filled(32, -1);
  for (var i = 0; i < a.length; i++) {
    for (final r in [a[i].rs1, a[i].rs2]) {
      if (r > 0) a[i].sources.add(Source(r, last[r]));
    }
    if (a[i].rd > 0) last[a[i].rd] = i;
  }
}

bool idResolved(Instruction i, Variant v) =>
    i.control && (v.stage == 'ID' || (i.kind == 'JUMP' && i.rs1 < 0));
int ready(
  Instruction i,
  Source source,
  List<Instruction> a,
  List<Timing> t,
  Variant v,
) {
  final p = source.producer;
  if (!v.forward) return t[p].w + 1;
  var r = a[p].kind == 'LOAD' ? t[p].w : t[p].m;
  if (idResolved(i, v)) r++;
  return r;
class Timing {
  int f, d, x, m = 0, w = 0;
  Timing(this.f, this.d, this.x);
}

class Stall {

Run simulate(List<Instruction> a, Variant v, List<Prediction> p) {
  final out = Run();
  for (var i = 0; i < a.length; i++) {
    final ins = a[i], t = Timing(1, 2, 3);
    if (i > 0) {
      final previous = out.timings[i - 1];
      t.f = previous.d;
      if (p[i - 1].wrong) t.f = previous.x + (idResolved(a[i - 1], v) ? 0 : 1);
      t.d = max(t.f + 1, previous.x);
      final bounds = [t.d + 1, previous.m, previous.x + 1, previous.x + 1],
          producer = List.filled(4, -1);
      for (final source in ins.sources) {
        if (source.producer < 0) continue;
        final cause = a[source.producer].kind == 'LOAD' ? 2 : 3,
            r = ready(ins, source, a, out.timings, v);
        if (r > bounds[cause]) {
          bounds[cause] = r;
          producer[cause] = source.producer;
        }
      }
      var current = previous.x + 1;
      for (var cause = 0; cause < 4; cause++) {
        final b = bounds[cause];
        if (b > current) {
          out.stalls.add(
            Stall(
              cause,
              b - current,
              cause < 2 ? i - 1 : i,
              i,
              producer[cause],
            ),
          );
          out.totals[cause] += b - current;
          out.events[cause]++;
          current = b;
        }
      }
      t.x = current;
    }
    final e = ins.kind == 'MUL' ? v.latency : 1;
    t.m = t.x + e;
    t.w = t.m + 1;
    out.timings.add(t);
  out.cycles = out.timings.last.w;
  return out;
Trace traceCheck() {
  final lines = readLines('input/trace.txt'),
      out = Trace(lines.length),
      first = <int, Instruction>{};
  for (var line = 0; line < lines.length; line++) {
    final a = lines[line].split(' ');
      pc = hexValue(a[0]) ?? -1;
      if (pc < 0 || pc % 4 != 0) reason = 'pc';
    }
    if (reason.isEmpty) {
      kind = a[1];
      if (!classes.contains(kind)) reason = 'class';
    }
    if (reason.isEmpty) {
      final r = a.sublist(2, 5).map(register).toList();
      if (r.any((v) => v == null)) {
        reason = 'register';
      } else {
        rd = r[0]!;
        rs1 = r[1]!;
        rs2 = r[2]!;
    if (reason.isNotEmpty) {
      out.rejected.add((line + 1, reason));
      continue;
    }
    final ins = Instruction(
      pc,
      kind,
      rd,
      rs1,
      rs2,
      taken,
      target,
      address,
      line + 1,
    );
    final n = out.instructions.length;
    if (n > 0 && pc != out.instructions.last.nextPC)
      out.gaps.add((line + 1, out.instructions.last.nextPC, pc));
    out.staticInstructions.putIfAbsent(pc, () => []).add(n);
    out.instructions.add(ins);
  }
  out.pcs.sort();
  dependences(out.instructions);
  return out;
