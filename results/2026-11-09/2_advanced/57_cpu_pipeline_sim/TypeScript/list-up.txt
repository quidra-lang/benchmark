export const control = (instruction: Instruction) =>
  instruction.kind === 'BRANCH' || instruction.kind === 'JUMP';
export const idResolved = (instruction: Instruction, variant: Variant) =>
  control(instruction) &&
  (variant.stage === 'ID' || (instruction.kind === 'JUMP' && instruction.rs1 === undefined));
export function dependencies(instructions: Instruction[]): Source[][] {
  const latest = new Map<number, number>();
  return instructions.map((instruction, i) => {
    const sources: Source[] = [];
    for (const position of ['rs1', 'rs2'] as const) {
      const register = instruction[position];
      if (register !== undefined && register !== 0) {
        sources.push({ register, position, producer: latest.get(register) });
      }
    }
    if (instruction.rd !== undefined && instruction.rd !== 0) latest.set(instruction.rd, i);
    return sources;
  });
}
export function ready(
  source: Source,
  instruction: Instruction,
  variant: Variant,
  instructions: Instruction[],
  timings: Timing[],
): number | undefined {
  const p = source.producer;
  if (p === undefined) return undefined;
  return variant.forwarding
    ? (instructions[p].kind === 'LOAD' ? timings[p].writeback : timings[p].memory) +
        (idResolved(instruction, variant) ? 1 : 0)
    : timings[p].writeback + 1;

export function simulate(
  instructions: Instruction[],
  sources: Source[][],
  variant: Variant,
  mispredicted: boolean[],
): Run {
  const timings: Timing[] = [];
  const stalls: Stall[] = [];
  instructions.forEach((instruction, i) => {
    let fetch = 1;
    let decode = 2;
    let execute = 3;
    if (i) {
      const previous = timings[i - 1];
      const ideal = previous.execute + 1;
      fetch = mispredicted[i - 1]
        ? previous.execute + (idResolved(instructions[i - 1], variant) ? 0 : 1)
        : previous.decode;
      decode = Math.max(fetch + 1, previous.execute);
      const bounds: Record<Cause, number> = {
        control: decode + 1,
        structural: previous.memory,
        load_use: ideal,
        raw: ideal,
      };
      const producers: Partial<Record<Cause, number>> = {};
      for (const source of sources[i]) {
        const cycle = ready(source, instruction, variant, instructions, timings);
        if (cycle === undefined || source.producer === undefined) continue;
        const cause = instructions[source.producer].kind === 'LOAD' ? 'load_use' : 'raw';
        if (cycle > bounds[cause]) {
          bounds[cause] = cycle;
          producers[cause] = source.producer;
        }
      }
      execute = ideal;
      for (const cause of causes) {
        if (bounds[cause] > execute) {
          stalls.push({
            cause,
            cycles: bounds[cause] - execute,
            responsible: cause === 'control' || cause === 'structural' ? i - 1 : i,
            waiting: i,
            producer: producers[cause],
          });
          execute = bounds[cause];
        }
      }
    }
    const memory = execute + (instruction.kind === 'MUL' ? variant.latency : 1);
    timings.push({ fetch, decode, execute, memory, writeback: memory + 1 });
