int parse_register(const char *text) {
    if (strcmp(text, "-") == 0)
        return -1;
    const size_t length = strlen(text);
    if (length < 2 || length > 3 || text[0] != 'r' || (length == 3 && text[1] == '0'))
        return -2;
    int out = 0;
    for (size_t i = 1; i < length; ++i) {
        if (text[i] < '0' || text[i] > '9')
            return -2;
        out = 10 * out + text[i] - '0';
    }
    return out <= 31 ? out : -2;
}
bool control(Instruction instruction) {
    return instruction.kind == BRANCH || instruction.kind == JUMP;
}
Sources *dependencies(const Instruction *instructions, int count) {
    Sources *out = allocate((size_t)count, sizeof(*out));
    int last[32];
    for (int i = 0; i < 32; ++i)
        last[i] = -1;
    for (int i = 0; i < count; ++i) {
        const Instruction instruction = instructions[i];
        out[i].producer[0] = instruction.rs1 > 0 ? last[instruction.rs1] : -1;
        out[i].producer[1] = instruction.rs2 > 0 ? last[instruction.rs2] : -1;
        if (instruction.rd > 0)
            last[instruction.rd] = i;
    }
    return out;
}
bool id_resolved(Instruction instruction, Variant variant) {
    return (instruction.kind == JUMP && instruction.rs1 < 0) ||
           (control(instruction) && variant.id);
}
int ready_cycle(Instruction consumer, Instruction producer, Timing time, Variant variant) {
    return !variant.forwarding
               ? time.w + 1
               : (producer.kind == LOAD ? time.w : time.m) + id_resolved(consumer, variant);
Run simulate(const Instruction *instructions, const Sources *sources, int count, Variant variant,
             const bool *flags) {
    Run out = {0};
    out.timings = allocate((size_t)count, sizeof(*out.timings));
    out.records = allocate((size_t)count * 4 + 1, sizeof(*out.records));
    for (int i = 0; i < count; ++i) {
        Timing *time = &out.timings[i];
        int execute = instructions[i].kind == MUL ? variant.latency : 1;
        if (i == 0) {
            time->f = 1;
            time->d = 2;
            time->x = 3;
        } else {
            Timing previous = out.timings[i - 1];
            time->f =
                flags[i - 1] ? previous.x + !id_resolved(instructions[i - 1], variant) : previous.d;
            time->d = time->f + 1 > previous.x ? time->f + 1 : previous.x;
            int ideal = previous.x + 1;
            int bound[4] = {time->d + 1, previous.m, ideal, ideal}, producer[4] = {-1, -1, -1, -1};
            for (int j = 0; j < 2; ++j) {
                int p = sources[i].producer[j];
                if (p < 0)
                    continue;
                int cause = instructions[p].kind == LOAD ? 2 : 3;
                int ready = ready_cycle(instructions[i], instructions[p], out.timings[p], variant);
                if (ready > bound[cause]) {
                    bound[cause] = ready;
                    producer[cause] = p;
                }
            }
            time->x = ideal;
            for (int cause = 0; cause < 4; ++cause)
                if (bound[cause] > time->x) {
                    out.records[out.count++] = (Record){cause, bound[cause] - time->x,
                                                        cause < 2 ? i - 1 : i, i, producer[cause]};
                    time->x = bound[cause];
                }
        }
        time->m = time->x + execute;
        time->w = time->m + 1;
    }
    for (int i = 0; i < out.count; ++i) {
        Record record = out.records[i];
        out.stalls[record.cause] += record.cycles;
        ++out.events[record.cause];
    }
    out.cycles = out.timings[count - 1].w;
    return out;
