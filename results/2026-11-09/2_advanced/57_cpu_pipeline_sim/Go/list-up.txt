func register(s string) (int, bool) {
	if s == "-" {
		return -1, true
	}
	if len(s) < 2 || s[0] != 'r' {
		return 0, false
	}
	digits := s[1:]
	if len(digits) > 1 && digits[0] == '0' {
		return 0, false
	}
	v := 0
	for _, c := range digits {
		if c < '0' || c > '9' {
			return 0, false
		}
		v = v*10 + int(c-'0')
		if v > 31 {
			return 0, false
		}
	c := Config{}
	for _, line := range readLines("input/config.txt") {
		a := strings.Split(line, " ")
		case "VARIANT":
			c.Variants = append(c.Variants, Variant{a[1], a[2] == "ON", a[3], integer(a[4])})
		case "PRIMARY":
			c.PrimaryVariant, c.PrimaryPredictor = a[1], a[2]
		case "WINDOW":
			c.Start, c.End = integer(a[1]), integer(a[2])
func (i Instruction) control() bool { return i.Class == "BRANCH" || i.Class == "JUMP" }
func dependences(a []Instruction) {
	last := [32]int{}
	for r := range last {
		last[r] = -1
	}
	for i := range a {
		for _, r := range []int{a[i].RS1, a[i].RS2} {
			if r > 0 {
				a[i].Sources = append(a[i].Sources, Source{r, last[r]})
			}
		}
		if a[i].RD > 0 {
			last[a[i].RD] = i
		}
	}
}
func idResolved(i Instruction, v Variant) bool {
	return i.control() && (v.Stage == "ID" || (i.Class == "JUMP" && i.RS1 < 0))
}
func ready(i Instruction, source Source, a []Instruction, t []Timing, v Variant) int {
	p := source.Producer
	if !v.Forward {
		return t[p].W + 1
	}
	r := t[p].M
	if a[p].Class == "LOAD" {
		r = t[p].W
	}
	if idResolved(i, v) {
		r++
	}
	return r
type Timing struct{ F, D, X, M, W int }
type Stall struct{ Cause, Cycles, Responsible, Waiting, Producer int }
func simulate(a []Instruction, v Variant, p []Prediction) Run {
	out := Run{Timings: make([]Timing, len(a))}
	for i, ins := range a {
		t := Timing{F: 1, D: 2, X: 3}
		if i > 0 {
			previous := out.Timings[i-1]
			t.F = previous.D
			if p[i-1].Wrong {
				t.F = previous.X
				if !idResolved(a[i-1], v) {
					t.F++
				}
			}
			t.D = max(t.F+1, previous.X)
			bounds := [4]int{t.D + 1, previous.M, previous.X + 1, previous.X + 1}
			producer := [4]int{-1, -1, -1, -1}
			for _, source := range ins.Sources {
				if source.Producer < 0 {
					continue
				}
				cause := 3
				if a[source.Producer].Class == "LOAD" {
					cause = 2
				}
				r := ready(ins, source, a, out.Timings, v)
				if r > bounds[cause] {
					bounds[cause], producer[cause] = r, source.Producer
				}
			}
			current := previous.X + 1
			for cause, b := range bounds {
				if b > current {
					responsible := i
					if cause < 2 {
						responsible = i - 1
					}
					out.Stalls = append(out.Stalls, Stall{cause, b - current, responsible, i, producer[cause]})
					out.Totals[cause] += b - current
					out.Events[cause]++
					current = b
				}
			}
			t.X = current
		}
		e := 1
		if ins.Class == "MUL" {
			e = v.Latency
		}
		t.M, t.W = t.X+e, t.X+e+1
		out.Timings[i] = t
	}
	last := len(a) - 1
	out.Cycles = out.Timings[last].W
	return out
}
func traceCheck() Trace {
	lines := readLines("input/trace.txt")
	out := Trace{Lines: len(lines), Static: map[int][]int{}}
	first := map[int]Instruction{}
	classSet := map[string]bool{}
	for _, c := range classes {
		classSet[c] = true
	}
	for line, text := range lines {
		a := strings.Split(text, " ")
		reason := ""
		ins := Instruction{Line: line + 1}
		if reason == "" {
			ins.Class = a[1]
			if !classSet[ins.Class] {
				reason = "class"
			}
		}
		if reason == "" {
			rd, rdOK := register(a[2])
			rs1, s1OK := register(a[3])
			rs2, s2OK := register(a[4])
			ins.RD, ins.RS1, ins.RS2 = rd, rs1, rs2
		if reason != "" {
			out.Rejected = append(out.Rejected, Rejection{line + 1, reason})
			continue
		}
		out.Static[ins.PC] = append(out.Static[ins.PC], n)
		out.Instructions = append(out.Instructions, ins)
	}
	sort.Ints(out.PCs)
	dependences(out.Instructions)
	return out

		}

		t := run.Timings[i]
		annotation := annotations[i]
		out = append(out, DiagramRow{i, ins.PC, ins.Class, false, []Span{{t.F, t.F, 'F', 0}, {t.F + 1, t.D - 1,

			t.M, 'M', 3}, {t.W, t.W, 'W', 4}}, annotation})
	writeLines("pipeline.txt", lines)

		fmt.Sprintf("cycle_check %d + 4 + %d = %d", n, sum, n+4+sum), fmt.Sprintf("cpi_stack base 1.0000 fill %s control %s structural %s load_use %s raw %s",
	writeParts("stats.txt", parts)
	stallPC := map[int]int{}
	for _, s := range run.Stalls {
		stallPC[a[s.Responsible].PC] += s.Cycles
	}
	for _, pc := range trace.PCs {
		executions := trace.Static[pc]
		ins := a[executions[0]]

		if ins.control() || i == len(a)-1 {
			pc := a[lo].PC
			if groups[pc] == nil {
				groups[pc] = &Segment{PC: pc}
