int parse_register(const std::string &text) {
    if (text == "-")
        return -1;
    if (text.size() < 2 || text.size() > 3 || text[0] != 'r' ||
        (text.size() == 3 && text[1] == '0'))
        return -2;
    int out = 0;
    for (std::size_t i = 1; i < text.size(); ++i) {
        if (text[i] < '0' || text[i] > '9')
            return -2;
        out = 10 * out + text[i] - '0';
    }
    return out <= 31 ? out : -2;
}
Sources dependencies(const std::vector<Instruction> &instructions) {
    std::array<int, 32> last;
    last.fill(-1);
    Sources out;
    for (int i = 0; i < int(instructions.size()); ++i) {
        const Instruction &instruction = instructions[i];
        out.push_back({instruction.rs1 > 0 ? last[instruction.rs1] : -1,
                       instruction.rs2 > 0 ? last[instruction.rs2] : -1});
        if (instruction.rd > 0)
            last[instruction.rd] = i;
    }
    return out;
}
bool id_resolved(const Instruction &instruction, const Variant &variant) {
    return (instruction.kind == jump && instruction.rs1 < 0) ||
           (instruction.control() && variant.id);
}
int ready_cycle(const Instruction &consumer, const Instruction &producer, const Timing &time,
                const Variant &variant) {
    if (!variant.forwarding)
        return time.w + 1;
    return (producer.kind == load ? time.w : time.m) + int(id_resolved(consumer, variant));
Run simulate(const std::vector<Instruction> &instructions, const Sources &sources,
             const Variant &variant, const std::vector<bool> &mispredicted) {
    Run out;
    out.timings.resize(instructions.size());
    for (int i = 0; i < int(instructions.size()); ++i) {
        Timing &time = out.timings[i];
        const int execute = instructions[i].kind == multiply ? variant.latency : 1;
        if (i == 0) {
            time.f = 1;
            time.d = 2;
            time.x = 3;
        } else {
            const Timing &previous = out.timings[i - 1];
            time.f = mispredicted[i - 1] ? previous.x + !id_resolved(instructions[i - 1], variant)
                                         : previous.d;
            time.d = std::max(time.f + 1, previous.x);
            const int ideal = previous.x + 1;
            std::array<int, 4> bound{time.d + 1, previous.m, ideal, ideal},
                producer{-1, -1, -1, -1};
            for (int p : sources[i])
                if (p >= 0) {
                    const int cause = instructions[p].kind == load ? 2 : 3;
                    const int ready =
                        ready_cycle(instructions[i], instructions[p], out.timings[p], variant);
                    if (ready > bound[cause]) {
                        bound[cause] = ready;
                        producer[cause] = p;
                    }
                }
            time.x = ideal;
            for (int cause = 0; cause < 4; ++cause)
                if (bound[cause] > time.x) {
                    out.records.push_back(
                        {cause, bound[cause] - time.x, cause < 2 ? i - 1 : i, i, producer[cause]});
                    time.x = bound[cause];
                }
        }
        time.m = time.x + execute;
        time.w = time.m + 1;
    }
    for (const Record &record : out.records) {
        out.stalls[record.cause] += record.cycles;
        ++out.events[record.cause];
    }
    out.cycles = out.timings.back().w;
    return out;
static std::optional<Instruction> check(const std::string &text, const Trace &trace,
                                        std::string &error) {
    std::vector<std::string> fields;
    std::size_t start = 0;
    for (std::size_t i = 0; i <= text.size(); ++i)
        if (i == text.size() || text[i] == ' ') {
            fields.push_back(text.substr(start, i - start));
            start = i + 1;
        }
    const auto pc = parse_hex(fields[0]);
    if (!pc || *pc % 4)
        return reject("pc");
    const auto kind = std::find(classes.begin(), classes.end(), fields[1]);
    if (kind == classes.end())
        return reject("class");
    Instruction out;
    out.pc = *pc;
    out.kind = Class(kind - classes.begin());
    const std::array<int, 3> registers{parse_register(fields[2]), parse_register(fields[3]),
                                       parse_register(fields[4])};
    out.rd = registers[0];
    out.rs1 = registers[1];
    out.rs2 = registers[2];
    out.next = out.taken ? out.target : out.pc + 4;
    return out;
}
static Trace validate(const Lines &lines) {
    Trace trace;
    for (int i = 0; i < int(lines.size()); ++i) {
        std::string error;
        auto instruction = check(lines[i], trace, error);
        if (!instruction) {
            trace.rejected.push_back({i + 1, error});
            continue;
        }
        instruction->line = i + 1;
        if (!trace.instructions.empty() && trace.instructions.back().next != instruction->pc)
            trace.gaps.push_back({i + 1, trace.instructions.back().next, instruction->pc});
        trace.first.emplace(instruction->pc, int(trace.instructions.size()));
        trace.instructions.push_back(*instruction);
    }
    return trace;
