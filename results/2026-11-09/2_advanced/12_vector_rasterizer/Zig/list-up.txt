fn penCommand(tag: Tag) bool { return switch (tag) { .LINE, .RECT, .CIRCLE, .ELLIPSE, .POLYLINE, .POLYGON, .PATH => true, else => false }; }
fn brushCommand(tag: Tag) bool { return switch (tag) { .FILLRECT, .FILLCIRCLE, .FILLELLIPSE, .FILLPOLYGON, .FILLPATH, .FLOOD, .TEXT => true, else => false }; }
    fn check(self: *const Renderer, command: *const Command) ?ErrorName { const tag = enumValue(Tag, command.keyword) orelse return .unknown_command; const args = command.args; if (!validForm(tag, args)) return .bad_arguments; if (penCommand(tag) and !self.styles.pens.contains(args[0])) return .unknown_style; if (brushCommand(tag) and !self.styles.brushes.contains(args[0])) return .unknown_style;
        if (penCommand(tag)) { const pen = self.styles.pens.get(args[0]).?; self.penCoverage(mask, tag, pen, args[1..]); const color = self.styles.colors.get(pen.color).?; for (mask.bits, 0..) |value, pixel| { if (value) self.layer().paint(pixel, color, command.line); } } else { const brush = self.styles.brushes.get(args[0]).?; self.brushCoverage(mask, tag, brush, args[1..]); const first = self.styles.colors.get(brush.colors[0]).?; const box = mask.box(); for (mask.bits, 0..) |value, pixel| { if (!value) continue; const color = if (brush.axis) |axis| gradient(first, self.styles.colors.get(brush.colors[1]).?, axis, box.?, .{ .x = C.int(pixel % mask.width), .y = C.int(pixel / mask.width) }) else first; self.layer().paint(pixel, color, command.line); } } return mask;
