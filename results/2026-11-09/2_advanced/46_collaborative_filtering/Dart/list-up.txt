double rmse(List<double> errors) {
  if (errors.isEmpty) return double.nan;
  var sum = 0.0;
  for (final e in errors) {
    sum += e * e;
  }
  return math.sqrt(sum / errors.length);
}

  return s == '-0.0000' ? '0.0000' : s;
