    static double rmse(List<Double> errors) {
        if (errors.isEmpty())
            return Double.NaN;
        double sum = 0;
        for (double e : errors)
            sum += e * e;
        return Math.sqrt(sum / errors.size());
    }
    static String real(double value) {
        if (Double.isNaN(value))
            return "-";
        String text = String.format(Locale.ROOT, "%.4f", value);
        return text.equals("-0.0000") ? "0.0000" : text;
    }
