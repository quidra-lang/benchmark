function M.rmse(errors)
    local sum = 0.0
    for _, error in ipairs(errors) do
        sum = sum + error * error
    end
    return math.sqrt(sum / #errors)
end

    if value == nil then
        return "-"
    end
    local result = string.format("%.4f", value)
    return result == "-0.0000" and "0.0000" or result
