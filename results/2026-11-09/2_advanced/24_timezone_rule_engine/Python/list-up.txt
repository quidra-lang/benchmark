def chosen_candidate(candidates, option):
    if len(candidates) == 1:
        return candidates[0]
    if len(candidates) == 2 and option == "EARLIER":
        return candidates[0]
    if len(candidates) == 2 and option == "LATER":
        return candidates[1]
    return None
