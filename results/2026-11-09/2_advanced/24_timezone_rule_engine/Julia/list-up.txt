function chosen_candidate(found::Vector{Int}, option::Union{Nothing,String})
    length(found) == 1 && return found[1]
    if length(found) == 2
        option == "EARLIER" && return found[1]
        option == "LATER" && return found[2]
    end
    nothing
end
