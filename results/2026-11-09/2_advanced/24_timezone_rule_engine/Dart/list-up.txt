int? chosenCandidate(List<int> candidates, String? option) {
  if (candidates.length == 1 || candidates.length == 2 && option == 'EARLIER') return candidates[0];
  if (candidates.length == 2 && option == 'LATER') return candidates[1];
  return null;
}
