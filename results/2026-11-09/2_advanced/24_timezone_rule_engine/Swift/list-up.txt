func chosenCandidate(_ candidates: [Int], _ option: String?) -> Int? {
    if candidates.count == 1 { return candidates[0] }
    if candidates.count == 2 { if option == "EARLIER" { return candidates[0] }; if option == "LATER" { return candidates[1] } }
    return nil
}
