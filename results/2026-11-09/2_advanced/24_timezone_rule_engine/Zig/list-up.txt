fn chosenCandidate(found: resolver.Candidates, option: ?[]const u8) ?usize {
    if (found.count == 1) return found.indices[0];
    if (found.count == 2) {
        if (option) |choice| {
            if (r.eq(choice, "EARLIER")) return found.indices[0];
            if (r.eq(choice, "LATER")) return found.indices[1];
        }
    }
    return null;
}
