func chosenCandidate(candidates []int, option string) int {
	if len(candidates) == 1 || len(candidates) == 2 && option == "EARLIER" {
		return candidates[0]
	}
	if len(candidates) == 2 && option == "LATER" {
		return candidates[1]
	}
	return -1
}
