int chosen_candidate(int[] candidates, string option)
    return if #candidates == 1 then candidates[0] elif #candidates == 2 and option == "EARLIER" then candidates[0] elif #candidates == 2 and option == "LATER" then candidates[1] else -1
string unresolved_text(resolver.Timeline timeline, int local_time, int[] candidates)
    if #candidates > 0
        return "AMBIGUOUS " + [for index := candidates[] then candidate_text(timeline, local_time, index)].join(" or ")
    index := timeline.skipping_transition(local_time)
    timestamp := timeline.starts[index]
    local_before := report.local_text(timestamp + timeline.periods[index - 1].offset)
    local_after := report.local_text(timestamp + timeline.periods[index].offset)
    return "NONEXISTENT gap {local_before} to {local_after} at {report.ut_text(timestamp)}"
    string to_utc(Zone zone, int local_time, string option)
        int[] candidates := zone.timeline.candidates(local_time)
        index := chosen_candidate(candidates, option)
        return if index < 0 then unresolved_text(zone.timeline, local_time, candidates) else candidate_text(zone.timeline, local_time, index)
