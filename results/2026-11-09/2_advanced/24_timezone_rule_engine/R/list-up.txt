chosen_candidate <- function(candidates, option) {
  if (length(candidates) == 1 || length(candidates) == 2 && identical(option, "EARLIER")) return(candidates[1])
  if (length(candidates) == 2 && identical(option, "LATER")) return(candidates[2])
  0L
}
