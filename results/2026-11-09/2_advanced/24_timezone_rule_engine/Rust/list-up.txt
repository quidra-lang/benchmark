fn chosen_candidate(candidates: &[usize], option: Option<&str>) -> Option<usize> {
    match (candidates.len(), option) { (1, _) | (2, Some("EARLIER")) => Some(candidates[0]), (2, Some("LATER")) => Some(candidates[1]), _ => None }
}
