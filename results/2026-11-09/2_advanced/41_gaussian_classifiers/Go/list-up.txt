func argmax(v []float64) int {
	best := 0
	for i := 1; i < len(v); i++ {
		if v[i] > v[best] {
			best = i
		}
	}
	return best
}
func argmin(v []float64) int {
	best := 0
	for i := 1; i < len(v); i++ {
		if v[i] < v[best] {
			best = i
		}
	}
	return best
}
func (m *GaussianModel) scores(v []float64) []float64 {
	q := m.squaredDistances(v)
	g := make([]float64, len(q))
	for k := range g {
		g[k] = m.LogPriors[k] - m.Normalizer - m.LogDets[k]/2 - q[k]/2
	}
	return g
}
type Prediction struct {
	Predicted, Decision     int
	Posterior, LogPosterior []float64
	Confidence              float64
}

func (m *GaussianModel) inferVector(v []float64, costs [][]float64) Prediction {
	g := m.scores(v)
	pred := argmax(g)
	ex := make([]float64, len(g))
	total := 0.0
	for k, x := range g {
		ex[k] = math.Exp(x - g[pred])
		total += ex[k]
	}
	probs := make([]float64, len(g))
	logs := make([]float64, len(g))
	lt := math.Log(total)
	for k := range g {
		probs[k] = ex[k] / total
		logs[k] = (g[k] - g[pred]) - lt
	}
	decision := pred
	if costs != nil {
		risk := make([]float64, len(g))
		for b := range risk {
			for k, p := range probs {
				risk[b] += p * costs[k][b]
			}
		}
		decision = argmin(risk)
	}
	return Prediction{pred, decision, probs, logs, probs[argmax(probs)]}
}
func (m *GaussianModel) infer(values []float64, costs [][]float64) Prediction {
	return m.inferVector(m.vector(values), costs)
}

		}
	}

		count, correct := 0, 0
		for i, r := range results {
			if r.Confidence >= threshold {
				count++
				if r.Predicted == a.TestActual[i] {
					correct++
				}
			}
		}
		lines = append(lines, fmt.Sprintf("reject %s accepted %d coverage %s accuracy %s", real(threshold), count, ratio(count, a.M), ratio(correct, count)))
	}
func (a *Analysis) ensemble() []int {
	pred := make([]int, a.M)
	for i := range pred {
		p := make([]float64, a.C)
		for k := range p {
			for m := 0; m < 3; m++ {
				p[k] += a.Results[m][i].Posterior[k]
			}
			p[k] /= 3
		}
		pred[i] = argmax(p)
	}
	return pred
}
func (a *Analysis) costLines() []string {
	lines := []string{"costs"}
	for k, name := range modelNames {
		total, decision := 0.0, 0.0
		changed, correct := 0, 0
		for i, r := range a.Results[k] {
			y := a.TestActual[i]
			total += a.Data.Costs[y][r.Predicted]
			decision += a.Data.Costs[y][r.Decision]
			if r.Decision != r.Predicted {
				changed++
			}
			if r.Decision == y {
				correct++
			}
		}
		lines = append(lines, fmt.Sprintf("%s argmax_total %s decision_total %s changed %d decision_accuracy %s",
			name, real(total), real(decision), changed, ratio(correct, a.M)))
	}
