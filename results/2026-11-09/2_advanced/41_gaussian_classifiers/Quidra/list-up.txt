int argmax(real64[] values)
    best := 0
    for: 1 <= index < int(#values)
        if values[index] > values[best]
            best = index
    return best

int argmin(real64[] values)
    best := 0
    for: 1 <= index < int(#values)
        if values[index] < values[best]
            best = index
    return best
