export function predict(model: Model, vector: number[], costs: Matrix | null = null): Prediction {
  const h = (model.features.length * Math.log(2 * Math.PI)) / 2;
  const distances: number[] = [];
  const scores: number[] = [];
  for (const g of model.classes) {
    const q = quadratic(
      g.lower,
      vector.map((v, j) => v - g.mean[j]),
    );
    distances.push(q);
    scores.push(Math.log(g.prior) - h - g.logDeterminant / 2 - q / 2);
  }
  let predicted = 0;
  for (let k = 1; k < scores.length; k++) if (scores[k] > scores[predicted]) predicted = k;
  const maximum = scores[predicted];
  const exponentials = scores.map((g) => Math.exp(g - maximum));
  let sum = 0;
  for (const e of exponentials) sum += e;
  const posterior = exponentials.map((e) => e / sum);
  const logPosterior = scores.map((g) => g - maximum - Math.log(sum));
  let decision = predicted;
  if (costs) {
    let least = Infinity;
    for (let b = 0; b < scores.length; b++) {
      let risk = 0;
      for (let k = 0; k < scores.length; k++) risk += posterior[k] * costs[k][b];
      if (risk < least) {
        least = risk;
        decision = b;
      }
    }
  }
  return {
    predicted,
    posterior,
    logPosterior,
    confidence: Math.max(...posterior),
    decision,
    distances,
  };
}
export const predictRow = (model: Model, row: Row, costs: Matrix | null = null) =>
  predict(
    model,
    model.features.map((f) => row.values[f]),
    costs,
  );

      lines.push(
        fields(
          'error',
          r.id,
          'actual',
          r.label,
          'predicted',
          d.classes[predicted[i]],
          'confidence',
          real(results[i].confidence),
        ),
      );
    }
  });
  for (const t of d.settings.reject) {
    let accepted = 0;
    let correct = 0;
    d.test.forEach((r, i) => {
      if (results[i].confidence >= t) {
        accepted++;
        correct += Number(r.class === predicted[i]);
      }
    });
    lines.push(
      fields(
        'reject',
        real(t),
        'accepted',
        accepted,
        'coverage',
        ratio(accepted, n),
        'accuracy',
        ratio(correct, accepted),
      ),
    );
  }
  const ensemble = d.test.map((_, r) => {
    const p = d.classes.map((__, k) => {
      let sum = 0;
      for (let m = 0; m < 3; m++) sum += results[m][r].posterior[k];
      return sum / 3;
    });
    let best = 0;
    for (let k = 1; k < p.length; k++) if (p[k] > p[best]) best = k;
    return best;
  });

      arg += d.costs[r.class][p.predicted];
      decision += d.costs[r.class][p.decision];

      correct += Number(p.decision === r.class);
    });
