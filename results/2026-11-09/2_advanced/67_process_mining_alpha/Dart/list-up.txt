
String relation(Map<Pair, int> d, String a, String b) {
  final ab = count(d, a, b) > 0, ba = count(d, b, a) > 0;
  return ab && ba
      ? '||'
      : ab
      ? '->'
      : ba
      ? '<-'
      : '#';
