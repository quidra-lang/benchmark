function basic_relation($counts, $first, $second)
{
    $forward = follows_count($counts, $first, $second) > 0;
    $reverse = follows_count($counts, $second, $first) > 0;
    return $forward ? ($reverse ? '||' : '->') : ($reverse ? '<-' : '#');
