func relation(d map[Pair]int, a, b string) string {
	ab, ba := d[Pair{a, b}] > 0, d[Pair{b, a}] > 0
	if ab && ba {
		return "||"
	}
	if ab {
		return "->"
	}
	if ba {
		return "<-"
	}
	return "#"
