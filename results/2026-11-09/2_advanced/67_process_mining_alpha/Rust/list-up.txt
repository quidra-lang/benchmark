pub fn relation(counts: &Counts, a: &str, b: &str) -> &'static str {
    match (get(counts, a, b) > 0, get(counts, b, a) > 0) {
        (true, false) => "->", (false, true) => "<-", (true, true) => "||", _ => "#",
    }
