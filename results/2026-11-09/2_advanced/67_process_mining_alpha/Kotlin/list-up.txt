    fun basic(counts: Map<Pair,Long>,a: String,b: String): String {
        val forward=count(counts,a,b)>0;
        val reverse=count(counts,b,a)>0;
        return if(forward&&!reverse)"->" else if(reverse&&!forward)"<-" else if(forward)"||" else "#"
