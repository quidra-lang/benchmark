int replace(Store &store, int original, const map<int, int> replacements)
    index := resolve(store, original)
    type := store.nodes[index]
    if type.kind == "variable"
        match replacements.get(index)
            int found
                return found
            none
                return index
    if type.kind == "list"
        element := replace(&store, type.first, replacements)
        return add(&store, "list", element)
    if type.kind == "pair" or type.kind == "function"
        first := replace(&store, type.first, replacements)
        second := replace(&store, type.second, replacements)
        return add(&store, type.kind, first, second)
    return index

int instantiate(Store &store, const Scheme scheme)
    map<int, int> replacements
    ordered := scheme.quantified.sorted()
    for variable := ordered[]
        created := fresh(&store)
        .replacements.set(variable, created)
    return replace(&store, scheme.type, replacements)
