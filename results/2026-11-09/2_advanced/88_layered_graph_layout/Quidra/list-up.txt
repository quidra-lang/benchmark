    return false
int[] unique(const int[] items)
    int[] result := []
    for item := items[]

int compare(const int[] first, const int[] second)
    count := if #first < #second then int(#first) else int(#second)
    for: position < count
        if first[position] != second[position]
            return if first[position] < second[position] then -1 else 1
    return if #first < #second then -1 elif #first > #second then 1 else 0

bool change_setting(Settings &settings, string key, string value)
    names := setting_names()
    index := -1
    for: position < int(#names)
        if names[position] == key
            index = position
    if index < 0
        return false
    string[] choices := []
    if key == "layering"
        choices = ["longest", "coffman"]
    elif key == "promotion" or key == "sifting" or key == "reduce"
        choices = ["on", "off"]
    elif key == "heuristic"
        choices = ["barycenter", "median"]
    elif key == "direction"
        choices = ["TB", "BT", "LR", "RL"]
    if #choices > 0
        if not has(choices, value)
            return false
        settings.values[index] = value
        return true
    if #value < 1 or #value > 10
        return false
    for: position < int(#value)
        if value[position].codepoints()[0] < 48 or value[position].codepoints()[0] > 57
            return false
    number := int(value)
    low := if key == "sweeps" or key == "restarts" then 0 elif key == "layer_gap" then 2 else 1
    high := if key == "width" then 40 elif key == "sweeps" then 50 elif key == "seed" then 2147483646 else 9
    if number < low or number > high
        return false
    settings.values[index] = string(number)
    return true

Graph[] read_graphs()
    defaults := Settings()
    string params := string.from_utf8(file.read_bin("input/params.txt"))
    settings_lines := params.split(NL)
    for line := settings_lines[]
        if line != "" and not line.starts_with("#")
            words := line.split(" ")
            accepted := change_setting(&defaults, words[0], words[1])
    string source := string.from_utf8(file.read_bin("input/graphs.txt"))
    lines := source.split(NL)
    Graph[] result := []
    current := Graph("", defaults)
    for: line_index < int(#lines)
        line := lines[line_index]
        if line == "" or line.starts_with("#")
            continue
        words := line.split(" ")
        if words[0] == "GRAPH"
            current = Graph(words[1], defaults)
            continue
        if words[0] == "END"
            match current.problem
                Problem ignored
                    current.problem = ignored
                none
                    if #current.nodes == 0
                        current.problem = Problem("no_nodes", line_index + 1)
            for: position < int(#current.nodes)
                cursor := position
                while cursor > 0 and current.nodes[cursor].id < current.nodes[cursor - 1].id
                    saved := current.nodes[cursor]
                    current.nodes[cursor] = current.nodes[cursor - 1]
                    current.nodes[cursor - 1] = saved
                    cursor -= 1
            match current.problem
                Problem ignored
                    current.problem = ignored
                none
                    for: position < int(#current.edges)
                        original := current.edges[position]
                        first := required_index(current.nodes, original.first)
                        second := required_index(current.nodes, original.second)
                        current.edges[position] = Edge(first, second)
            result = result.append(current)
            continue
        skip := false
        match current.problem
            Problem ignored
                skip = true
            none
                skip = false
        if skip
            continue
        string | none reason := none
        if words[0] == "OPTION"
            if not change_setting(&current.settings, words[1], words[2])
                reason = "bad_option"
        elif words[0] == "NODE"
            id := int(words[1])
            if known(current.nodes, id)
                reason = "duplicate_node"
            elif #current.nodes == 40
                reason = "too_many_nodes"
            else
                current.nodes = current.nodes.append(Node(id, words[2]))
        else
            first := int(words[1])
            second := int(words[2])
            edge := Edge(first, second)
            if not known(current.nodes, first) or not known(current.nodes, second)
                reason = "unknown_node"
            elif first == second
                current.loops += 1
            elif has(current.edges, edge)
                current.duplicates += 1
            else
                current.edges = current.edges.append(edge)
        match reason
            string kind
                current.problem = Problem(kind, line_index + 1)
            none
                current.problem = none
    return result

void visit(int vertex, const int[] layer, const int[][] lower, bool[] &visited, int[][] &order)
    if visited[vertex]
        return
    visited[vertex] = true
    order[layer[vertex]] = order[layer[vertex]].append(vertex)
    for successor := lower[vertex][]
        visit(successor, layer, lower, &visited, &order)
Model build_model(const io.Graph graph, const int[] heights, const io.Edge[] dag, const bool[] removed)
    nodes := int(#graph.nodes)
    highest := heights[0]
    for value := heights[]
        highest = io.maximum(highest, value)
    int[] raw := []
    for value := heights[]
        raw = raw.append(highest - value)
    used := io.unique(raw)
    for: edge_index < int(#dag)
        if not removed[edge_index]
            edge := dag[edge_index]
            for: raw[edge.first] + 1 <= level < raw[edge.second]
                used = io.unique(used.append(level))
    used = used.sorted()
    int[] layer := []
    for value := raw[]
        for: position < int(#used)
            if used[position] == value
                layer = layer.append(position)
                break
    int[] width := []
    for node := graph.nodes[]
        width = width.append(int(#node.label) + 2)
    int[][] chains := array(#graph.edges, fill = [])
    for: edge_index < int(#dag)
        if removed[edge_index]
            continue
        edge := dag[edge_index]
        chain := [edge.first]
        for: layer[edge.first] + 1 <= level < layer[edge.second]
            chain = chain.append(int(#layer))
            layer = layer.append(level)
            width = width.append(1)
        chains[edge_index] = chain.append(edge.second)
    crossing.Segment[][] gaps := array(nat(int(#used) - 1), fill = [])
    int[][] upper := array(#layer, fill = [])
    lower := upper
    for: edge_index < int(#chains)
        chain := chains[edge_index]
        for: 1 <= position < int(#chain)
            first := chain[position - 1]
            second := chain[position]
            gaps[layer[first]] = gaps[layer[first]].append(crossing.Segment(first, second, edge_index))
            upper[second] = upper[second].append(first)
            lower[first] = lower[first].append(second)
    int[][] initial := array(#used, fill = [])
    visited := array(#layer, fill = false)
    for: vertex < nodes
        visit(vertex, layer, lower, &visited, &initial)
    return Model(layer, width, chains, gaps, upper, lower, initial)
Key reorder_key(int vertex, const int[][] reference, const int[] places, bool median)
    int[] values := []
    for neighbor := reference[vertex][]
        values = values.append(places[neighbor])
    values = values.sorted()
    if median
        return Key(vertex, values[(int(#values) - 1) / 2] + values[int(#values) / 2], 1)
int[] insert_vertex(const int[] source, int position, int vertex)
    int[] result := []
    for: cursor < int(#source) + 1
        if cursor == position
            result = result.append(vertex)
        if cursor < int(#source)
            result = result.append(source[cursor])
    return result
Reduction reduce_crossings(const io.Graph graph, const Model model)
    vertices := int(#model.layer)
    int random_state := graph.settings.integer("seed")
    string[] records := []
    string[] csv := []
    adopted := model.initial
    best_count := 2147483647
    best_run := 0
    best_sweep := 0
    initial := 0
    for: int run < graph.settings.integer("restarts") + 1
        order := model.initial
        if run > 0
            for: row < int(#order)
                cursor := int(#order[row]) - 1
                while cursor > 0
                    random_state = 48271 * random_state % 2147483647
                    selected := random_state % (cursor + 1)
                    saved := order[row][cursor]
                    order[row][cursor] = order[row][selected]
                    order[row][selected] = saved
                    cursor -= 1
        current_count := crossing.total(model.gaps, order, vertices)
        if run == 0
            initial = current_count
        run_count := current_count
        run_sweep := 0
        run_order := order
        history := [order]
        counts := [current_count]
        records = records.append("sweep {run} 0 initial crossings {current_count}")
        csv = csv.append("{graph.name},{run},0,{current_count}")
        for: 1 <= int sweep < graph.settings.integer("sweeps") + 1
            last := sweep - 1
            if counts[last] == 0 or (last >= 2 and history[last] == history[last - 2])
                break
            old := crossing.positions(order, vertices)
            down := sweep % 2 == 1
            int[] row_order := []
            if down
                for: 1 <= row < int(#order)
                    row_order = row_order.append(row)
            else
                row := int(#order) - 2
                while row >= 0
                    row_order = row_order.append(row)
                    row -= 1
            reference := if down then model.upper else model.lower
            for row := row_order[]
                places := crossing.positions(order, vertices)
                Key[] moving := []
                for vertex := order[row][]
                    if #reference[vertex] > 0
                        moving = moving.append(reorder_key(vertex, reference, places, graph.settings.text("heuristic") == "median"))
                for: position < int(#moving)
                    cursor := position
                    while cursor > 0
                        left := moving[cursor]
                        right := moving[cursor - 1]
                        if left.numerator * right.denominator >= right.numerator * left.denominator
                            break
                        moving[cursor] = right
                        moving[cursor - 1] = left
                        cursor -= 1
                next := 0
                for: position < int(#order[row])
                    if #reference[order[row][position]] > 0
                        order[row][position] = moving[next].vertex
                        next += 1
            places := crossing.positions(order, vertices)
            moved := 0
            for: vertex < vertices
                if places[vertex] != old[vertex]
                    moved += 1
            swaps := 0
            while true
                changed := false
                for: row < int(#order)
                    for: position < int(#order[row]) - 1
                        first := order[row][position]
                        second := order[row][position + 1]
                        if crossing.adjacent(first, second, model.upper, model.lower, places) > crossing.adjacent(second, first, model.upper, model.lower, places)
                            order[row][position] = second
                            order[row][position + 1] = first
                            places[first] = position + 1
                            places[second] = position
                            swaps += 1
                            changed = true
                if not changed
                    break
            current_count = crossing.total(model.gaps, order, vertices)
            history = history.append(order)
            counts = counts.append(current_count)
            direction := if down then "down" else "up"
            records = records.append("sweep {run} {sweep} {direction} moved {moved} swaps {swaps} crossings {current_count}")
            csv = csv.append("{graph.name},{run},{sweep},{current_count}")
            if current_count < run_count
                run_count = current_count
                run_sweep = sweep
                run_order = order
        if run_count < best_count
            best_count = run_count
            best_run = run
            best_sweep = run_sweep
            adopted = run_order
    records = records.append("best run {best_run} sweep {best_sweep} crossings {best_count}")
    if graph.settings.on("sifting")
        round := 0
        while true
            round += 1
            moves := 0
            for: vertex < vertices
                row := model.layer[vertex]
                old := 0
                int[] base := []
                for: position < int(#adopted[row])
                    if adopted[row][position] == vertex
                        old = position
                    else
                        base = base.append(adopted[row][position])
                int[] costs := []
                minimum_cost := 2147483647
                for: position < int(#base) + 1
                    adopted[row] = insert_vertex(base, position, vertex)
                    value := crossing.total(model.gaps, adopted, vertices)
                    costs = costs.append(value)
                    minimum_cost = io.minimum(minimum_cost, value)
                chosen := old
                if costs[old] != minimum_cost
                    for: position < int(#costs)
                        if costs[position] == minimum_cost
                            chosen = position
                            break
                adopted[row] = insert_vertex(base, chosen, vertex)
                if chosen != old
                    moves += 1
            records = records.append("sift {round} moves {moves} crossings {crossing.total(model.gaps, adopted, vertices)}")
            if moves == 0
                break
    return Reduction(adopted, records, csv, initial, best_count, best_run, best_sweep)
Geometry geometry(const io.Graph graph, const Model model, const int[][] order, const coordinates.Coordinates placement)
    int[] tracks := []
    int[] gaps := []
    int[][] offsets := []
    for segments := model.gaps[]
        bent_counts := array(#model.layer, fill = 0)
        for segment := segments[]
            if placement.x[segment.upper] != placement.x[segment.lower]
                bent_counts[segment.upper] += 1
        Bus[] buses := []
        for: segment_index < int(#segments)
            segment := segments[segment_index]
            first_x := placement.x[segment.upper]
            second_x := placement.x[segment.lower]
            if first_x == second_x
                continue
            side := if bent_counts[segment.upper] >= 2 then 0 else 1
            vertex := if side == 0 then segment.upper else segment.lower
            low := io.minimum(first_x, second_x)
            high := io.maximum(first_x, second_x)
            bus_index := -1
            for: position < int(#buses)
                if buses[position].side == side and buses[position].vertex == vertex
                    bus_index = position
                    break
            if bus_index < 0
                buses = buses.append(Bus(low, high, side, vertex, placement.x[vertex], [segment_index]))
            else
                buses[bus_index].low = io.minimum(buses[bus_index].low, low)
                buses[bus_index].high = io.maximum(buses[bus_index].high, high)
                buses[bus_index].members = buses[bus_index].members.append(segment_index)
        for: position < int(#buses)
            cursor := position
            while cursor > 0
                first := buses[cursor]
                second := buses[cursor - 1]
                if io.compare([first.low, first.high, first.side, first.x], [second.low, second.high, second.side, second.x]) >= 0
                    break
                buses[cursor] = second
                buses[cursor - 1] = first
                cursor -= 1
        int[] ends := []
        assignments := array(#segments, fill = -1)
        for bus := buses[]
            track := -1
            for: position < int(#ends)
                if ends[position] + 1 < bus.low
                    track = position
                    break
            if track < 0
                track = int(#ends)
                ends = ends.append(bus.high)
            else
                ends[track] = bus.high
            for member := bus.members[]
                assignments[member] = track + 1
        gap := io.maximum(graph.settings.integer("layer_gap"), int(#ends) + 2)
        for: position < int(#assignments)
            if assignments[position] >= 0
                assignments[position] += (gap - 2 - int(#ends)) / 2
        tracks = tracks.append(int(#ends))
        gaps = gaps.append(gap)
        offsets = offsets.append(assignments)
    string direction := graph.settings.text("direction")
    vertical := direction == "TB" or direction == "BT"
    int[] thickness := []
    for row := order[]
        value := 1
        if not vertical
            for vertex := row[]
                value = io.maximum(value, model.width[vertex])
        thickness = thickness.append(value)
    starts := array(#order, fill = 0)
    for: 1 <= row < int(#order)
        starts[row] = starts[row - 1] + thickness[row - 1] + gaps[row - 1]
    main_length := starts[-1] + thickness[-1]
    secondary_length := 0
    for: vertex < int(#model.layer)
        secondary_length = io.maximum(secondary_length, placement.x[vertex] + placement.right[vertex] + 1)
    return Geometry(tracks, gaps, offsets, thickness, starts, main_length, secondary_length)

Drawing draw(const io.Graph graph, const Model model, const coordinates.Coordinates placement, const Geometry shape, const bool[] reversed, const bool[] removed, const int[] | none path)
    string direction := graph.settings.text("direction")
    vertical := direction == "TB" or direction == "BT"
    picture := canvas.Canvas(shape.main_length, shape.secondary_length, direction)
    for: vertex < int(#model.layer)
        row := model.layer[vertex]
        if vertex >= int(#graph.nodes)
            for: position < shape.thickness[row]
                canvas.line(&picture, shape.starts[row] + position, placement.x[vertex], true)
        elif not vertical and #model.lower[vertex] > 0
            for: model.width[vertex] <= position < shape.thickness[row]
                canvas.line(&picture, shape.starts[row] + position, placement.x[vertex], true)
    for: gap_index < int(#model.gaps)
        start := shape.starts[gap_index] + shape.thickness[gap_index]
        gap := shape.gaps[gap_index]
        segments := model.gaps[gap_index]
        for: segment_index < int(#segments)
            segment := segments[segment_index]
            first_x := placement.x[segment.upper]
            second_x := placement.x[segment.lower]
            offset := shape.offsets[gap_index][segment_index]
            if offset >= 0
                for: position < offset
                    canvas.line(&picture, start + position, first_x, true)
                canvas.corner(&picture, start + offset, first_x)
                canvas.corner(&picture, start + offset, second_x)
                for: io.minimum(first_x, second_x) + 1 <= secondary < io.maximum(first_x, second_x)
                    canvas.line(&picture, start + offset, secondary, false)
                for: offset + 1 <= position < gap
                    canvas.line(&picture, start + position, second_x, true)
            else
                for: position < gap
                    canvas.line(&picture, start + position, first_x, true)
    canvas.Cell[][] routes := array(#graph.edges, fill = [])
    lengths := array(#graph.edges, fill = 0)
    bends := lengths
    for: edge_index < int(#model.chains)
        if removed[edge_index]
            continue
        chain := model.chains[edge_index]
        first := chain[0]
        second := chain[-1]
        upper_main := shape.starts[model.layer[first]] + (if vertical then 1 else model.width[first])
        lower_main := shape.starts[model.layer[second]] - 1
        points := [canvas.map_cell(picture, upper_main, placement.x[first])]
        for: 1 <= position < int(#chain)
            upper_vertex := chain[position - 1]
            lower_vertex := chain[position]
            gap_index := model.layer[upper_vertex]
            offset := -1
            for: segment_index < int(#model.gaps[gap_index])
                segment := model.gaps[gap_index][segment_index]
                if segment.upper == upper_vertex and segment.lower == lower_vertex
                    offset = shape.offsets[gap_index][segment_index]
                    break
            if offset >= 0
                main_coordinate := shape.starts[gap_index] + shape.thickness[gap_index] + offset
                points = points.append(canvas.map_cell(picture, main_coordinate, placement.x[upper_vertex]))
                points = points.append(canvas.map_cell(picture, main_coordinate, placement.x[lower_vertex]))
                bends[edge_index] += 2
        points = points.append(canvas.map_cell(picture, lower_main, placement.x[second]))
        if reversed[edge_index]
            points = points[::-1]
            canvas.arrow(&picture, upper_main, placement.x[first], true)
        else
            canvas.arrow(&picture, lower_main, placement.x[second], false)
        length := 1
        for: 1 <= position < int(#points)
            length += io.absolute(points[position].row - points[position - 1].row) + io.absolute(points[position].column - points[position - 1].column)
        lengths[edge_index] = length
        routes[edge_index] = points
    int[] on_path := []
    match path
        none
            on_path = []
    for: vertex < int(#graph.nodes)
        row := model.layer[vertex]
        highlighted := io.has(on_path, vertex)
        string text := (if highlighted then "{{" else "[") + graph.nodes[vertex].label + (if highlighted then "}}" else "]")
        cell := if vertical then canvas.map_cell(picture, shape.starts[row], placement.x[vertex] - placement.left[vertex]) elif direction == "LR" then canvas.Cell(placement.x[vertex], shape.starts[row]) else canvas.Cell(placement.x[vertex], shape.main_length - shape.starts[row] - model.width[vertex])
        canvas.label(&picture, cell.row, cell.column, text)
    return Drawing(picture, routes, lengths, bends)

Processed process_graph(const io.Graph graph)
    nodes := int(#graph.nodes)
    analysis := analyze(graph)
    sequence := layering.sequence(nodes, graph.edges)
    places := array(nat(nodes), fill = 0)
    for: position < int(#sequence)
        places[sequence[position]] = position
    bool[] reversed := []
    io.Edge[] dag := []
    for edge := graph.edges[]
        reverse := places[edge.first] > places[edge.second]
        reversed = reversed.append(reverse)
        dag = dag.append(if reverse then io.Edge(edge.second, edge.first) else edge)
    removed := array(#dag, fill = false)
    io.Edge[] drawn := []
    for: edge_index < int(#dag)
        edge := dag[edge_index]
        if graph.settings.on("reduce") and not reversed[edge_index]
        if not removed[edge_index]
            drawn = drawn.append(edge)
    levels := layering.assign(nodes, drawn, graph.settings.text("layering") == "coffman", graph.settings.integer("width"), graph.settings.on("promotion"))

    reduced := reduce_crossings(graph, model)

    positions := crossing.positions(reduced.order, vertices)

    for: iteration < count
        outgoing := array(nat(count), fill = 0)
        incoming := array(nat(count), fill = 0)
        for edge := edges[]
            if remaining[edge.first] and remaining[edge.second]
                outgoing[edge.first] += 1
                incoming[edge.second] += 1
        sink := -1
        source := -1
        greedy := -1
        for: vertex < count
            if not remaining[vertex]
                continue
            if sink < 0 and outgoing[vertex] == 0
                sink = vertex
            if source < 0 and incoming[vertex] == 0
                source = vertex
            if greedy < 0 or outgoing[vertex] - incoming[vertex] > outgoing[greedy] - incoming[greedy]
                greedy = vertex
        chosen := if sink >= 0 then sink elif source >= 0 then source else greedy
        if sink >= 0
            tail = [chosen].concat(tail)
        else
            head = head.append(chosen)
        remaining[chosen] = false
    return head.concat(tail)

int count(const Segment[] segments, const int[] places, int width)
    Pair[] pairs := []
    for segment := segments[]
        pairs = pairs.append(Pair(places[segment.upper], places[segment.lower]))
    for: position < int(#pairs)
        cursor := position
        while cursor > 0
            left := pairs[cursor]
            right := pairs[cursor - 1]
            if left.first > right.first or (left.first == right.first and left.second >= right.second)
                break
            pairs[cursor] = right
            pairs[cursor - 1] = left
            cursor -= 1
    capacity := 1
    while capacity < width
        capacity *= 2
    tree := array(nat(capacity * 2 - 1), fill = 0)
    result := 0
    for entry := pairs[]
        position := entry.second + capacity - 1
        tree[position] += 1
        while position > 0
            if position % 2 == 1
                result += tree[position + 1]
            position = (position - 1) / 2
            tree[position] += 1
    return result
