pub fn acyclic(graph: *g.Graph) []const usize {
    const n = graph.nodes; const remaining = g.zero(bool, n, true); var head: std.ArrayList(usize) = .empty; var tail: std.ArrayList(usize) = .empty;
    for (0..n) |_| {
        const indeg = g.zero(i64, n, 0); const outdeg = g.zero(i64, n, 0); for (graph.edges.items) |e| if (remaining[e.from] and remaining[e.to]) { outdeg[e.from] += 1; indeg[e.to] += 1; };
        var chosen: ?usize = null; var sink = false;
        for (0..n) |v| if (remaining[v] and outdeg[v] == 0) { chosen = v; sink = true; break; };
        if (chosen == null) for (0..n) |v| if (remaining[v] and indeg[v] == 0) { chosen = v; break; };
        if (chosen == null) { var best: i64 = std.math.minInt(i64); for (0..n) |v| if (remaining[v] and outdeg[v] - indeg[v] > best) { chosen = v; best = outdeg[v] - indeg[v]; }; }
        const v = chosen.?; remaining[v] = false; g.append(usize, if (sink) &tail else &head, v);
    }
    std.mem.reverse(usize, tail.items); for (tail.items) |v| g.append(usize, &head, v); const positions = g.alloc(usize, n); for (head.items, 0..) |v, i| positions[v] = i;
    for (graph.edges.items) |*e| { e.reversed = positions[e.from] > positions[e.to]; e.a = if (e.reversed) e.to else e.from; e.b = if (e.reversed) e.from else e.to; }
    if (graph.settings.reduce) { const reach = g.zero(bool, n * n, false); for (graph.edges.items) |e| reach[e.a * n + e.b] = true; for (0..n) |k| for (0..n) |i| for (0..n) |j| { reach[i * n + j] = reach[i * n + j] or (reach[i * n + k] and reach[k * n + j]); };
        for (graph.edges.items) |*e| { if (e.reversed) continue; for (graph.edges.items) |other| if (other.a == e.a and other.b != e.b and reach[other.b * n + e.b]) { e.transitive = true; break; }; }
    }
    return head.items;
}
pub const Promotion = struct { round: usize, vertex: usize, difference: i64 };
pub const Result = struct { heights: []i64, labels: []usize, promotions: []const Promotion, rounds: usize = 0, dummies_before: i64 = 0 };
fn promote(graph: *const g.Graph, h: []i64, v: usize) i64 { var difference: i64 = 0; for (graph.edges.items) |e| if (!e.transitive and e.b == v and h[e.a] == h[v] + 1) { difference += promote(graph, h, e.a); }; h[v] += 1; for (graph.edges.items) |e| if (!e.transitive) { if (e.a == v) difference += 1; if (e.b == v) difference -= 1; }; return difference; }
pub fn layers(graph: *g.Graph, order: []const usize) Result {
    const n = graph.nodes; const h = g.zero(i64, n, -1); const labels = g.zero(usize, n, 0);
    if (g.eq(graph.settings.layering, "longest")) { var i = n; while (i > 0) { i -= 1; const v = order[i]; h[v] = 0; for (graph.edges.items) |e| if (!e.transitive and e.a == v) { h[v] = @max(h[v], h[e.b] + 1); }; } }
    else {
    }
    var result = Result{ .heights = h, .labels = labels, .promotions = &.{} }; for (graph.edges.items) |e| if (!e.transitive) { result.dummies_before += h[e.a] - h[e.b] - 1; };
    var records: std.ArrayList(Promotion) = .empty;
    if (graph.settings.promotion) while (true) { result.rounds += 1; var changed = false; for (0..n) |v| { var parent = false; for (graph.edges.items) |e| if (!e.transitive and e.b == v) { parent = true; break; }; if (!parent) continue; const saved = g.copy(i64, h); const difference = promote(graph, h, v); if (difference < 0) { changed = true; g.append(Promotion, &records, .{ .round = result.rounds, .vertex = v, .difference = difference }); } else @memcpy(h, saved); } if (!changed) break; };
    result.promotions = records.items; var maximum: i64 = 0; for (h) |x| maximum = @max(maximum, x); const used = g.zero(bool, @intCast(maximum + 1), false); for (h) |x| used[@intCast(maximum - x)] = true;
    for (graph.edges.items) |e| if (!e.transitive) { var level = maximum - h[e.a] + 1; while (level < maximum - h[e.b]) : (level += 1) used[@intCast(level)] = true; };
    const normalized = g.alloc(usize, used.len); var next: usize = 0; for (used, 0..) |yes, level| { normalized[level] = next; next += @intFromBool(yes); } for (graph.vertices.items, h) |*v, x| v.layer = normalized[@intCast(maximum - x)]; return result;
}
const Cell = struct { lines: u8 = 0, arrow: u8 = 0, box: u8 = 0, highlighted: bool = false };
pub const Canvas = struct {
    rows: usize, columns: usize, main_size: i64, settings: g.Settings, cells: []Cell,
    pub fn init(settings: g.Settings, main_size: i64, secondary_size: i64) Canvas { const rows: usize = @intCast(if (settings.vertical()) main_size else secondary_size); const columns: usize = @intCast(if (settings.vertical()) secondary_size else main_size); return .{ .rows = rows, .columns = columns, .main_size = main_size, .settings = settings, .cells = g.zero(Cell, rows * columns, .{}) }; }
    pub fn map(self: Canvas, main: i64, secondary: i64) g.Point { const m = if (g.eq(self.settings.direction, "BT") or g.eq(self.settings.direction, "RL")) self.main_size - 1 - main else main; return if (self.settings.vertical()) .{ .row = m, .column = secondary } else .{ .row = secondary, .column = m }; }
    fn cell(self: *Canvas, p: g.Point) *Cell { return &self.cells[@as(usize, @intCast(p.row)) * self.columns + @as(usize, @intCast(p.column))]; }
    pub fn line(self: *Canvas, main: i64, secondary: i64, main_direction: bool) void { const bit: u8 = if (self.settings.vertical() == main_direction) 1 else 2; self.cell(self.map(main, secondary)).lines |= bit; }
    pub fn corner(self: *Canvas, main: i64, secondary: i64) void { self.cell(self.map(main, secondary)).lines |= 4; }
    pub fn arrow(self: *Canvas, point: g.Point, reversed: bool) void { const direction = self.settings.direction; const value: u8 = if (g.eq(direction, "TB")) (if (reversed) @as(u8, '^') else 'v') else if (g.eq(direction, "BT")) (if (reversed) @as(u8, 'v') else '^') else if (g.eq(direction, "LR")) (if (reversed) @as(u8, '<') else '>') else (if (reversed) @as(u8, '>') else '<'); self.cell(point).arrow = value; }
    pub fn box(self: *Canvas, point: g.Point, label: []const u8, highlighted: bool) void { const value = g.fmt("{s}{s}{s}", .{if (highlighted) "{" else "[", label, if (highlighted) "}" else "]"}); for (value, 0..) |c, i| self.cell(.{ .row = point.row, .column = point.column + @as(i64, @intCast(i)) }).box = c; }
    pub fn crossings(self: Canvas) usize { var count: usize = 0; for (self.cells) |c| count += @intFromBool(c.lines & 3 == 3); return count; }
    pub fn display(self: Canvas, name: []const u8, out: *g.Text) void {
        out.line("graph {s} {d}x{d}", .{name, self.columns, self.rows}); out.put("    ", .{}); for (0..self.columns) |i| out.put("{d}", .{i % 10}); out.put("\n", .{});
        for (0..self.rows) |row| { var text: g.Text = .{}; text.put("{d: >3} ", .{row}); for (0..self.columns) |column| { const c = self.cells[row * self.columns + column]; const value: u8 = if (c.box != 0) c.box else if (c.arrow != 0) c.arrow else if (c.highlighted) '#' else switch (c.lines) { 0 => ' ', 1 => '|', 2 => '-', else => '+' }; text.put("{c}", .{value}); } out.line("{s}", .{std.mem.trimEnd(u8, text.data.items, " ")}); }
    }
};

pub fn positions(orders: Orders, n: usize) []usize { const pos = g.alloc(usize, n); for (orders) |row| for (row, 0..) |v,i| { pos[v] = i; }; return pos; }
pub fn same(a: Orders, b: Orders) bool { for (a,b) |x,y| if (!std.mem.eql(usize,x,y)) return false; return true; }
pub fn crosses(a: g.Segment, b: g.Segment, pos: []const usize) bool { return (pos[a.upper] < pos[b.upper] and pos[a.lower] > pos[b.lower]) or (pos[a.upper] > pos[b.upper] and pos[a.lower] < pos[b.lower]); }
pub fn gap(graph: *const g.Graph, orders: Orders, layer: usize) usize {
    const pos = positions(orders, graph.vertices.items.len); const Pair = struct { a: usize, b: usize }; var pairs: std.ArrayList(Pair) = .empty;
    for (graph.segments.items) |s| if (graph.vertices.items[s.upper].layer == layer) g.append(Pair,&pairs,.{ .a=pos[s.upper], .b=pos[s.lower] });
    std.mem.sort(Pair,pairs.items,{},struct { fn less(_:void,a:Pair,b:Pair) bool { return a.a < b.a or (a.a == b.a and a.b < b.b); } }.less);
    var c: usize = 1; while (c < orders[layer + 1].len) c *= 2; const tree = g.zero(usize,2*c-1,0); var count: usize = 0;
    for (pairs.items) |pair| { var k = pair.b + c - 1; tree[k] += 1; while (k > 0) { if (k % 2 == 1) count += tree[k + 1]; k = (k - 1) / 2; tree[k] += 1; } } return count;
}
pub fn total(graph: *const g.Graph, orders: Orders) usize { var count: usize = 0; for (0..orders.len-1) |i| count += gap(graph,orders,i); return count; }
pub fn brute(graph: *const g.Graph, orders: Orders) usize { const pos=positions(orders,graph.vertices.items.len); var count:usize=0; for(graph.segments.items,0..) |a,i| for(graph.segments.items[i+1..]) |b| { if(graph.vertices.items[a.upper].layer == graph.vertices.items[b.upper].layer and crosses(a,b,pos)) count+=1; }; return count; }
pub fn neighbors(graph: *const g.Graph, vertex:usize, upper:bool) []usize { var out:std.ArrayList(usize)=.empty; for(graph.segments.items) |s| { if(upper and s.lower==vertex) g.append(usize,&out,s.upper); if(!upper and s.upper==vertex) g.append(usize,&out,s.lower); } return out.items; }
pub fn cost(graph:*const g.Graph, v:usize,w:usize,pos:[]const usize) usize { var count:usize=0; for([_]bool{true,false}) |up| for(neighbors(graph,v,up)) |a| for(neighbors(graph,w,up)) |b| { count+=@intFromBool(pos[a]>pos[b]); }; return count; }

pub fn option(s: *Settings, name: []const u8, value: []const u8) bool {
    if (eq(name, "layering") or eq(name, "heuristic") or eq(name, "direction") or eq(name, "promotion") or eq(name, "sifting") or eq(name, "reduce")) {
        if (eq(name, "layering")) { if (!eq(value, "longest") and !eq(value, "coffman")) return false; s.layering = value; }
        else if (eq(name, "heuristic")) { if (!eq(value, "barycenter") and !eq(value, "median")) return false; s.heuristic = value; }
        else if (eq(name, "direction")) { if (!eq(value, "TB") and !eq(value, "BT") and !eq(value, "LR") and !eq(value, "RL")) return false; s.direction = value; }
        else { if (!eq(value, "on") and !eq(value, "off")) return false; const flag = eq(value, "on"); if (eq(name, "promotion")) s.promotion = flag else if (eq(name, "sifting")) s.sifting = flag else s.reduce = flag; } return true;
    }
    if (value.len == 0 or value.len > 10) return false; for (value) |c| if (!std.ascii.isDigit(c)) return false;
    const n = std.fmt.parseInt(i64, value, 10) catch return false;
    if (eq(name, "width")) { if (n < 1 or n > 40) return false; s.width = @intCast(n); }
    else if (eq(name, "sweeps")) { if (n > 50) return false; s.sweeps = @intCast(n); }
    else if (eq(name, "node_sep") or eq(name, "dummy_sep")) { if (n < 1 or n > 9) return false; if (eq(name, "node_sep")) s.node_sep = n else s.dummy_sep = n; }
    else if (eq(name, "layer_gap")) { if (n < 2 or n > 9) return false; s.layer_gap = n; }
    else if (eq(name, "restarts")) { if (n > 9) return false; s.restarts = @intCast(n); }
    else if (eq(name, "seed")) { if (n < 1 or n > 2147483646) return false; s.seed = n; }
    else return false; return true;
}

pub fn parse(params: []const u8, text: []const u8) []Graph {
    var defaults: Settings = .{}; var lines = std.mem.splitScalar(u8, params, '\n'); while (lines.next()) |line| { if (line.len == 0 or line[0] == '#') continue; const w = words(line); _ = option(&defaults, w[0], w[1]); }
    var graphs: std.ArrayList(Graph) = .empty; lines = std.mem.splitScalar(u8, text, '\n'); var line_number: usize = 0; var g: Graph = undefined;
    while (lines.next()) |line| {
        line_number += 1; if (line.len == 0 or line[0] == '#') continue; const w = words(line);
        if (eq(w[0], "GRAPH")) { g = .{ .name = w[1], .settings = defaults }; continue; }
        if (eq(w[0], "END")) {
            if (g.error_kind == null and g.vertices.items.len == 0) g.fail("no_nodes", line_number);
            if (g.error_kind == null) {
                const ids = alloc(usize, g.vertices.items.len); for (g.vertices.items, ids) |v, *id| id.* = v.id;
                std.mem.sort(Vertex, g.vertices.items, {}, struct { fn less(_: void, a: Vertex, b: Vertex) bool { return a.id < b.id; } }.less);
                for (g.edges.items) |*e| { e.from = find(g, ids[e.from]).?; e.to = find(g, ids[e.to]).?; }
                if (g.path) |path| g.path = .{find(g, ids[path[0]]).?, find(g, ids[path[1]]).?}; g.nodes = g.vertices.items.len;
                for (g.vertices.items) |*v| { v.width = @intCast(v.label.len + 2); if (g.settings.vertical()) { v.left = @divFloor(v.width - 1, 2); v.right = v.width - 1 - v.left; } }
            }
            append(Graph, &graphs, g); continue;
        }
        if (g.error_kind != null) continue;
        if (eq(w[0], "OPTION")) { if (!option(&g.settings, w[1], w[2])) g.fail("bad_option", line_number); }
        else if (eq(w[0], "NODE")) { const id = number(w[1]); if (find(g, id) != null) g.fail("duplicate_node", line_number) else if (g.vertices.items.len == 40) g.fail("too_many_nodes", line_number) else append(Vertex, &g.vertices, .{ .name = w[1], .label = w[2], .id = id }); }
        else { const a = find(g, number(w[1])); const c = find(g, number(w[2])); if (a == null or c == null) { g.fail("unknown_node", line_number); continue; }
            if (eq(w[0], "PATH")) { g.path = .{a.?, c.?}; continue; }
            if (a.? == c.?) { g.loops += 1; continue; }
            var duplicate = false; for (g.edges.items) |e| if (e.from == a.? and e.to == c.?) { duplicate = true; break; };
            if (duplicate) g.duplicates += 1 else append(Edge, &g.edges, .{ .from = a.?, .to = c.? });
        }
    }
    return graphs.items;
}
fn visit(graph: *const g.Graph, v: usize, seen: []bool, orders: []std.ArrayList(usize)) void {
    if (seen[v]) return; seen[v] = true; g.append(usize, &orders[graph.vertices.items[v].layer], v);
    for (graph.segments.items) |s| if (s.upper == v) visit(graph, s.lower, seen, orders);
}
fn initialOrders(graph: *g.Graph) c.Orders {
    const n = graph.nodes;
    for (graph.edges.items, 0..) |*e, ei| {
        if (e.transitive) continue; var chain: std.ArrayList(usize) = .empty; g.append(usize, &chain, e.a);
        for (graph.vertices.items[e.a].layer + 1 .. graph.vertices.items[e.b].layer) |layer| { const v = graph.vertices.items.len; g.append(g.Vertex, &graph.vertices, .{ .name = g.fmt("d{d}", .{v - n + 1}), .label = "", .layer = layer, .dummy = true }); g.append(usize, &chain, v); }
        g.append(usize, &chain, e.b); e.chain = chain.items; for (1..chain.items.len) |i| g.append(g.Segment, &graph.segments, .{ .upper = chain.items[i - 1], .lower = chain.items[i], .edge = ei });
    }
    var count: usize = 0; for (graph.vertices.items) |v| count = @max(count, v.layer + 1); const lists = g.alloc(std.ArrayList(usize), count); @memset(lists, .empty); const seen = g.zero(bool, graph.vertices.items.len, false);
    for (0..n) |v| visit(graph, v, seen, lists); const orders = g.alloc([]usize, count); for (lists, orders) |list, *row| row.* = list.items; return orders;
}
fn reorder(graph: *const g.Graph, orders: c.Orders, layer: usize, upper: bool) void {
    const pos = c.positions(orders, graph.vertices.items.len); const Key = struct { vertex: usize, previous: usize, numerator: i64, denominator: i64 }; var movable: std.ArrayList(Key) = .empty; var positions: std.ArrayList(usize) = .empty;
    for (orders[layer], 0..) |v, old| {
        const refs = c.neighbors(graph, v, upper); if (refs.len == 0) continue; const values = g.alloc(usize, refs.len); for (refs, values) |u, *p| p.* = pos[u]; var sum: i64 = 0; for (values) |p| sum += @intCast(p); var denominator: i64 = @intCast(values.len);
        if (g.eq(graph.settings.heuristic, "median")) { std.mem.sort(usize, values, {}, std.sort.asc(usize)); sum = @intCast(values[(values.len - 1) / 2] + values[values.len / 2]); denominator = 1; }
        g.append(Key, &movable, .{ .vertex = v, .previous = old, .numerator = sum, .denominator = denominator }); g.append(usize, &positions, old);
    }
    std.mem.sort(Key, movable.items, {}, struct { fn less(_: void, x: Key, y: Key) bool { const left = x.numerator * y.denominator; const right = y.numerator * x.denominator; return left < right or (left == right and x.previous < y.previous); } }.less);
    for (positions.items, movable.items) |p, key| orders[layer][p] = key.vertex;
}
fn exchanges(graph: *const g.Graph, orders: c.Orders) usize {
    var total: usize = 0;
    while (true) { var changed = false; for (orders) |row| { if (row.len < 2) continue; for (0..row.len - 1) |j| { const pos = c.positions(orders, graph.vertices.items.len); const v = row[j]; const w = row[j + 1]; if (c.cost(graph, v, w, pos) > c.cost(graph, w, v, pos)) { std.mem.swap(usize, &row[j], &row[j + 1]); total += 1; changed = true; } } } if (!changed) break; }
    return total;
}
const Reduction = struct { orders: c.Orders, initial: usize, swept: usize, run: usize, sweep: usize };
fn reduceCrossings(graph: *const g.Graph, initial: c.Orders, layout: *g.Text, csv: *g.Text) Reduction {
    var result = Reduction{ .orders = c.clone(initial), .initial = c.total(graph, initial), .swept = std.math.maxInt(usize), .run = 0, .sweep = 0 }; var random = graph.settings.seed;
    for (0..graph.settings.restarts + 1) |run| {
        const orders = c.clone(initial);
        if (run > 0) for (orders) |row| { var j = row.len; while (j > 1) { j -= 1; random = @mod(48271 * random, 2147483647); const k: usize = @intCast(@mod(random, @as(i64, @intCast(j + 1)))); std.mem.swap(usize, &row[j], &row[k]); } };
        var history: std.ArrayList(c.Orders) = .empty; g.append(c.Orders, &history, c.clone(orders)); var count = c.total(graph, orders); var best = count; var best_sweep: usize = 0; var best_order = c.clone(orders);
        layout.line("sweep {d} 0 initial crossings {d}", .{run, count}); csv.line("{s},{d},0,{d}", .{graph.name, run, count});
        var sweep: usize = 1;
        while (count != 0 and sweep <= graph.settings.sweeps) : (sweep += 1) {
            if (sweep >= 3 and c.same(history.items[sweep - 1], history.items[sweep - 3])) break;
            const old = c.positions(orders, graph.vertices.items.len); const downward = sweep % 2 == 1;
            for (1..orders.len) |step| reorder(graph, orders, if (downward) step else orders.len - 1 - step, downward);
            const pos = c.positions(orders, graph.vertices.items.len); var moved: usize = 0; for (old, pos) |before, after| moved += @intFromBool(before != after); const swaps = exchanges(graph, orders); count = c.total(graph, orders);
            layout.line("sweep {d} {d} {s} moved {d} swaps {d} crossings {d}", .{run, sweep, if (downward) "down" else "up", moved, swaps, count}); csv.line("{s},{d},{d},{d}", .{graph.name, run, sweep, count}); g.append(c.Orders, &history, c.clone(orders));
            if (count < best) { best = count; best_sweep = sweep; best_order = c.clone(orders); }
        }
        if (best < result.swept) { result.orders = best_order; result.swept = best; result.run = run; result.sweep = best_sweep; }
    }
    layout.line("best run {d} sweep {d} crossings {d}", .{result.run, result.sweep, result.swept});
    if (graph.settings.sifting) {
        var round: usize = 0; while (true) { round += 1; var moves: usize = 0;
            for (graph.vertices.items, 0..) |vertex, v| {
                const row = result.orders[vertex.layer]; var old: usize = 0; for (row, 0..) |u, j| if (u == v) { old = j; break; };
                const others = g.alloc(usize, row.len - 1); var j: usize = 0; for (row) |u| if (u != v) { others[j] = u; j += 1; };
                const costs = g.alloc(usize, row.len); var minimum = std.math.maxInt(usize);
                for (0..row.len) |p| { @memcpy(row[0..p], others[0..p]); row[p] = v; @memcpy(row[p + 1 ..], others[p..]); costs[p] = c.total(graph, result.orders); minimum = @min(minimum, costs[p]); }
                var chosen = old; if (costs[old] != minimum) for (costs, 0..) |cost, p| if (cost == minimum) { chosen = p; break; };
                @memcpy(row[0..chosen], others[0..chosen]); row[chosen] = v; @memcpy(row[chosen + 1 ..], others[chosen..]); moves += @intFromBool(chosen != old);
            }
            layout.line("sift {d} moves {d} crossings {d}", .{round, moves, c.total(graph, result.orders)}); if (moves == 0) break;
        }
    }
    return result;
}
const Bus = struct { upper: bool, vertex: usize, lo: i64, hi: i64, segments: std.ArrayList(usize) = .empty };
const Geometry = struct { starts: []i64, thickness: []i64, gaps: []i64, tracks: []usize, main_size: i64, secondary_size: i64 };
fn geometry(graph: *g.Graph, orders: c.Orders, x: []const i64) Geometry {
    const layers = orders.len; const starts = g.zero(i64, layers, 0); const thickness = g.zero(i64, layers, 1); const gaps = g.zero(i64, layers - 1, 0); const tracks = g.zero(usize, layers - 1, 0);
    for (0..layers - 1) |layer| {
        const upper_count = g.zero(usize, x.len, 0); for (graph.segments.items) |s| if (graph.vertices.items[s.upper].layer == layer and x[s.upper] != x[s.lower]) { upper_count[s.upper] += 1; };
        var buses: std.ArrayList(Bus) = .empty;
        for (graph.segments.items, 0..) |s, si| {
            if (graph.vertices.items[s.upper].layer != layer or x[s.upper] == x[s.lower]) continue; const upper = upper_count[s.upper] >= 2; const vertex = if (upper) s.upper else s.lower; var existing: ?usize = null;
            for (buses.items, 0..) |bus, j| if (bus.upper == upper and bus.vertex == vertex) { existing = j; break; };
            const j = existing orelse blk: { g.append(Bus, &buses, .{ .upper = upper, .vertex = vertex, .lo = @min(x[s.upper], x[s.lower]), .hi = @max(x[s.upper], x[s.lower]) }); break :blk buses.items.len - 1; };
            const bus = &buses.items[j]; bus.lo = @min(bus.lo, @min(x[s.upper], x[s.lower])); bus.hi = @max(bus.hi, @max(x[s.upper], x[s.lower])); g.append(usize, &bus.segments, si);
        }
        std.mem.sort(Bus, buses.items, x, struct { fn less(coords: []const i64, first: Bus, second: Bus) bool { if (first.lo != second.lo) return first.lo < second.lo; if (first.hi != second.hi) return first.hi < second.hi; if (first.upper != second.upper) return first.upper; return coords[first.vertex] < coords[second.vertex]; } }.less);
        var ends: std.ArrayList(i64) = .empty;
        for (buses.items) |bus| { var choice: ?usize = null; for (ends.items, 0..) |end, j| if (end + 1 < bus.lo) { choice = j; break; }; const track = choice orelse blk: { g.append(i64, &ends, bus.hi); break :blk ends.items.len - 1; }; ends.items[track] = bus.hi; for (bus.segments.items) |si| graph.segments.items[si].track = @intCast(track + 1); }
        tracks[layer] = ends.items.len; gaps[layer] = @max(graph.settings.layer_gap, @as(i64, @intCast(ends.items.len)) + 2); const offset = @divFloor(gaps[layer] - 2 - @as(i64, @intCast(ends.items.len)), 2);
        for (graph.segments.items) |*s| if (graph.vertices.items[s.upper].layer == layer and s.track > 0) { s.track += offset; };
    }
    if (!graph.settings.vertical()) for (orders, 0..) |row, i| for (row) |v| { thickness[i] = @max(thickness[i], graph.vertices.items[v].width); };
    for (1..layers) |i| starts[i] = starts[i - 1] + thickness[i - 1] + gaps[i - 1]; var secondary: i64 = 0; for (graph.vertices.items, x) |v, coord| secondary = @max(secondary, coord + v.right + 1);
    return .{ .starts = starts, .thickness = thickness, .gaps = gaps, .tracks = tracks, .main_size = starts[layers - 1] + thickness[layers - 1], .secondary_size = secondary };
}

    const path = graph.path orelse return &.{}; const parent = g.zero(?usize, graph.nodes, null); parent[path[0]] = path[0]; var queue: std.ArrayList(usize) = .empty; g.append(usize, &queue, path[0]); var at: usize = 0;
fn paint(graph: *g.Graph, geometry_data: Geometry, x: []const i64, path: []const usize) drawing.Canvas {
    const geo = geometry_data; var canvas = drawing.Canvas.init(graph.settings, geo.main_size, geo.secondary_size);
    for (graph.vertices.items, 0..) |v, id| { const band = geo.starts[v.layer]; if (v.dummy) { var j: i64 = 0; while (j < geo.thickness[v.layer]) : (j += 1) canvas.line(band + j, x[id], true); } else if (!graph.settings.vertical() and c.neighbors(graph, id, false).len > 0) { var j = v.width; while (j < geo.thickness[v.layer]) : (j += 1) canvas.line(band + j, x[id], true); } }
    for (graph.segments.items) |s| {
        const layer = graph.vertices.items[s.upper].layer; const start = geo.starts[layer] + geo.thickness[layer]; const gap = geo.gaps[layer]; var j: i64 = 0;
        if (x[s.upper] == x[s.lower]) { while (j < gap) : (j += 1) canvas.line(start + j, x[s.upper], true); }
        else { while (j < s.track) : (j += 1) canvas.line(start + j, x[s.upper], true); canvas.corner(start + s.track, x[s.upper]); canvas.corner(start + s.track, x[s.lower]); var coord = @min(x[s.upper], x[s.lower]) + 1; while (coord < @max(x[s.upper], x[s.lower])) : (coord += 1) canvas.line(start + s.track, coord, false); j = s.track + 1; while (j < gap) : (j += 1) canvas.line(start + j, x[s.lower], true); }
    }
    for (graph.edges.items, 0..) |*edge, ei| {
        if (edge.transitive) continue; var route: std.ArrayList(g.Point) = .empty; const va = graph.vertices.items[edge.a]; const vb = graph.vertices.items[edge.b]; const size: i64 = if (graph.settings.vertical()) 1 else va.width;
        g.append(g.Point, &route, canvas.map(geo.starts[va.layer] + size, x[edge.a]));
        for (graph.segments.items) |s| if (s.edge == ei and x[s.upper] != x[s.lower]) { const layer = graph.vertices.items[s.upper].layer; const main_coord = geo.starts[layer] + geo.thickness[layer] + s.track; g.append(g.Point, &route, canvas.map(main_coord, x[s.upper])); g.append(g.Point, &route, canvas.map(main_coord, x[s.lower])); edge.bends += 2; };
        g.append(g.Point, &route, canvas.map(geo.starts[vb.layer] - 1, x[edge.b])); canvas.arrow(if (edge.reversed) route.items[0] else route.items[route.items.len - 1], edge.reversed);
        if (edge.reversed) std.mem.reverse(g.Point, route.items); edge.route = route.items; edge.length = 1; for (1..route.items.len) |i| edge.length += @intCast(@abs(route.items[i].row - route.items[i - 1].row) + @abs(route.items[i].column - route.items[i - 1].column));
        var highlighted = false; if (path.len > 1) for (1..path.len) |i| if (path[i - 1] == edge.from and path[i] == edge.to) { highlighted = true; break; };
        if (highlighted) for (1..route.items.len) |i| canvas.highlight(route.items[i - 1], route.items[i]);
    }
    for (graph.vertices.items[0..graph.nodes], 0..) |v, id| { var highlighted = false; for (path) |u| if (id == u) { highlighted = true; break; }; const point = if (graph.settings.vertical()) canvas.map(geo.starts[v.layer], x[id] - v.left) else g.Point{ .row = x[id], .column = if (g.eq(graph.settings.direction, "LR")) geo.starts[v.layer] else geo.main_size - geo.starts[v.layer] - v.width }; canvas.box(point, v.label, highlighted); }
    return canvas;
}
