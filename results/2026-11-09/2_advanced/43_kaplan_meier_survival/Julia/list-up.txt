function risk_sets(subjects)
    times = sort(unique(s.time for s in subjects))
    risk = length(subjects)
    rows = NamedTuple[]
    for time in times
        events = count(s->s.time==time && s.event==1,subjects)
        censored = count(s->s.time==time && s.event==0,subjects)
        push!(rows,(time=time,risk=risk,events=events,censored=censored))
        risk -= events+censored
    end
    rows
end
function kaplan_meier(subjects)
    events = [r for r in risk_sets(subjects) if r.events>0]
    survival = 1.0
    greenwood = hazard = variance = 0.0
    rows = NamedTuple[]
    for (i,event) in enumerate(events)
        time,risk,deaths = event.time,event.risk,event.events
        survival *= (risk-deaths)/risk
        hazard += deaths/risk
        variance += deaths/(risk*risk)
        next_time = i<length(events) ? events[i+1].time : nothing
        censored = count(s->s.event==0 && s.time>=time && (isnothing(next_time) || s.time<next_time),subjects)
        push!(rows,(time=time,risk=risk,events=deaths,censored=censored,survival=survival,
                    se=error,lower=lower,upper=upper,hazard=hazard,hazard_se=sqrt(variance)))
    end
    rows
end
function row_at(rows,time)
    for row in Iterators.reverse(rows)
        row.time<=time && return row
    end
    (survival=1.0,se=0.0,lower=nothing,upper=nothing,hazard=0.0,hazard_se=0.0)
end
function number(value,digits=4)
    isnothing(value) && return "NA"
    text = Printf.format(Printf.Format("%.$(digits)f"),value)
function collection_summary(subjects,params)
    rows = kaplan_meier(subjects)
    (rows=rows,life=life,subjects=length(subjects),events=events,censored=length(subjects)-events,
     maximum=maximum(s.time for s in subjects),person_days=person_days,rate=rate,

end
function km_table_lines(name,subjects,summary,params)
    for time in params["REPORT_TIMES"]
        row = row_at(summary.rows,time)
        risk = count(s->s.time>=time,subjects)
        events = count(s->s.time<=time && s.event==1,subjects)
        push!(
            lines,
            "at $time at_risk $risk events $events survival $(number(row.survival)) std_err $(number(row.se)) lower $(number(row.lower)) upper $(number(row.upper)) cum_hazard $(number(row.hazard)) hazard_se $(number(row.hazard_se))"
        )
    end
    for row in summary.rows
        fields = vcat(string.([row.time,row.risk,row.events,row.censored]),
          number.([row.survival,row.se,row.lower,row.upper,row.hazard,row.hazard_se]))
        push!(lines,join(lpad(value,9) for value in fields))
    end
