int risk_size(const Subjects *subjects, int time) {
    int count = 0;
    for (int i = 0; i < subjects->count; ++i)
        count += subjects->data[i].time >= time;
    return count;
}
int event_count(const Subjects *subjects, int time) {
    int count = 0;
    for (int i = 0; i < subjects->count; ++i)
        count += subjects->data[i].time == time && subjects->data[i].event == 1;
    return count;
}
IntList event_times(const Subjects *subjects) {
    IntList out = {0};
    for (int i = 0; i < subjects->count; ++i)
        if (subjects->data[i].event)
            int_push(&out, subjects->data[i].time);
    int_sort(&out);
    int count = 0;
    for (int i = 0; i < out.count; ++i)
        if (!count || out.data[i] != out.data[count - 1])
            out.data[count++] = out.data[i];
    out.count = count;
    return out;
}
KMRows kaplan_meier(const Subjects *subjects) {
    IntList times = event_times(subjects);
    KMRows out = {.count = times.count};
    out.data = calloc((size_t)times.count, sizeof(KMRow));
    double survival = 1, greenwood = 0, hazard = 0, variance = 0;
    bool defined = true;
    for (int i = 0; i < times.count; ++i) {
        KMRow row = {.time = times.data[i], .error_defined = true};
        row.risk = risk_size(subjects, row.time);
        row.events = event_count(subjects, row.time);
        for (int j = 0; j < subjects->count; ++j) {
            const Subject *subject = &subjects->data[j];
            if (!subject->event && subject->time >= row.time &&
                (i + 1 == times.count || subject->time < times.data[i + 1]))
                ++row.censored;
        }
        survival = survival * ((double)(row.risk - row.events) / row.risk);
        row.survival = survival;
        hazard += (double)row.events / row.risk;
        variance += (double)row.events / (row.risk * row.risk);
        row.hazard = hazard;
        row.hazard_se = sqrt(variance);
        out.data[i] = row;
    }
    free(times.data);
    return out;
}
