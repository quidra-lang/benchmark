int adler(int[] data)
    first := 1
    second := 0
    for value := data[]
        first = (first + value) % 65521
        second = (second + first) % 65521
    return second * 65536 + first

int modular_power(int base, int exponent)
    result := 1
    while exponent > 0
        if exponent % 2 == 1
            result = result * base % 65521
        base = base * base % 65521
        exponent /= 2
    return result

int modulo(int value, int divisor)
    result := value % divisor
    return if result < 0 then result + divisor else result

class Repaired(int[] data, int[] record)

Repaired[] repair_byte(int[] data, int stored, int computed)
    difference_a := modulo(stored % 65536 - computed % 65536, 65521)
    difference_b := modulo(stored / 65536 - computed / 65536, 65521)
    delta := 0
    if difference_a >= 1 and difference_a <= 255
        delta = difference_a
    elif difference_a >= 65266 and difference_a <= 65520
        delta = difference_a - 65521
    else
        return []
    inverse := modular_power(modulo(delta, 65521), 65519)
    distance := difference_b * inverse % 65521
    if distance < 1 or distance > int(#data)
        return []
    position := int(#data) - distance
    before := data[position]
    after := before + delta
    if after < 0 or after > 255
        return []
    repaired := data
    repaired[position] = after
    if adler(repaired) != stored
        return []
    return [Repaired(repaired, [position, before, after])]

DecodeResult decode(int method, int[] payload, int original, int offset_bits, int length_bits)
    if method == 0
        if int(#payload) < original
            return DecodeResult([], "end_of_data", int(#payload))
        return DecodeResult(payload[0:<original])
    if method == 1
        result := lzw.decode(payload, original)
        if #result.problems > 0
            return DecodeResult([], result.problems[0].kind, result.problems[0].position)
        return DecodeResult(result.data)
    if method == 2
        return decode_lz(payload, original, offset_bits, length_bits)
    if method == 3
        return decode_lzh(payload, original, offset_bits, length_bits)
    result := arithmetic.decode(payload, original)
    if #result.problems > 0
        return DecodeResult([], result.problems[0].kind, result.problems[0].position)
    return DecodeResult(result.data)

class Encoding(string name, int[] data, int[][] payloads, int best, int[] lzw_stats, int[] lz_stats, int[] lzh_stats, int[] slots, int[] arith_stats)
    payload := payloads[best]

Encoding encode_file(string name, int[] data, Params params)
    tokens := lz77.tokenize(data, params.offset_bits, params.length_bits)
    zw := lzw.encode(data)
    lz_payload := encode_lz(tokens, params.offset_bits, params.length_bits)
    zh := encode_lzh(tokens, params)
    arith := arithmetic.encode(data)
    payloads := [data, zw.data, lz_payload, zh.data, arith.data]
    best := 0
    for: 1 <= method < 5
        if #payloads[method] < #payloads[best]
            best = method
    return Encoding(name, data, payloads, best, zw.stats, stats, zh.stats, zh.slots, arith.stats)
int[] uint_bytes(int value, int count)
    result := array(nat(count), fill = 0)
    for: count - 1 >= index >= 0
        result[index] = value % 256
        value /= 256
    return result

int read_uint(int[] data, int position, int count)
    result := 0
    for: index < count
        result = result * 256 + data[position + index]
    return result

int[] build_archive(Encoding[] encodings, Params params)
    result := [84, 88, 65, 82, params.offset_bits, params.length_bits]
    result = result.concat(uint_bytes(int(#encodings), 2))
    for encoded := encodings[]
        header := [int(#encoded.name)]
        header = header.concat(encoded.name.codepoints()).append(encoded.best)
        header = header.concat(uint_bytes(int(#encoded.data), 4))
        header = header.concat(uint_bytes(int(#encoded.payload), 4)).concat(uint_bytes(adler(encoded.data), 4))
        check := 0
        for value := header[]
            check = (check + value) % 256
        result = result.concat(header).append(check).concat(encoded.payload)
    return result

bool valid_name(int[] name)
    if #name < 1 or #name > 32
        return false
    allowed := "abcdefghijklmnopqrstuvwxyz0123456789._-".codepoints()
    for value := name[]
        found := false
        for candidate := allowed[]
            if candidate == value
                found = true
        if not found
            return false
    return true
class Entry(int number, int offset)
    status := ""
    valid_header := false
    name := ""
    method := 0
    original := 0
    compressed := 0
    checksum := 0
    start := 0
    finish := 0
    int[] data := []
    computed := 0
    failure_kind := ""
    failure_position := 0
    int[] repair := []

class Parsed(int size)
    offset_bits := 0
    length_bits := 0
    declared := 0
    Entry[] entries := []
    Entry[] intact := []
    reason := ""
    stopped := false
    trailing := 0

Parsed parse_archive(int[] archive)
    parsed := Parsed(int(#archive))
    magic := [84, 88, 65, 82]
    if #archive < 8 or archive[0:<4] != magic
        parsed.reason = "bad_magic"
        return parsed
    parsed.offset_bits = archive[4]
    parsed.length_bits = archive[5]
    if parsed.offset_bits < 6 or parsed.offset_bits > 12 or parsed.length_bits < 2 or parsed.length_bits > 5
        parsed.reason = "bad_params"
        return parsed
    parsed.declared = read_uint(archive, 6, 2)
    position := 8
    for: 1 <= number <= parsed.declared
        entry := Entry(number, position)
        if position >= int(#archive) or position + 15 + archive[position] > int(#archive)
            entry.status = "truncated"
            parsed.entries = parsed.entries.append(entry)
            parsed.stopped = true
            break
        length := archive[position]
        name := archive[position + 1:<position + 1 + length]
        cursor := position + 1 + length
        start := position + 15 + length
        entry.method = archive[cursor]
        entry.original = read_uint(archive, cursor + 1, 4)
        entry.compressed = read_uint(archive, cursor + 5, 4)
        entry.checksum = read_uint(archive, cursor + 9, 4)
        entry.start = start
        entry.finish = start + entry.compressed
        check := 0
        for: position <= index < start - 1
            check = (check + archive[index]) % 256
        entry.valid_header = valid_name(name) and check == archive[start - 1]
        if not entry.valid_header
            entry.status = "bad_header"
            parsed.entries = parsed.entries.append(entry)
            position = entry.finish
            if position > int(#archive)
                parsed.stopped = true
                break
            continue
        entry.name = byte_text(name)
        if entry.finish > int(#archive)
            entry.status = "truncated"
            parsed.entries = parsed.entries.append(entry)
            parsed.stopped = true
            break
        if entry.method >= 5
            entry.status = "unsupported"
        else
            decoded := decode(entry.method, archive[start:<entry.finish], entry.original, parsed.offset_bits, parsed.length_bits)
            if decoded.kind != ""
                entry.status = "error"
                entry.failure_kind = decoded.kind
                entry.failure_position = decoded.position
            else
                entry.data = decoded.data
                entry.computed = adler(decoded.data)
                if entry.computed == entry.checksum
                    entry.status = "ok"
                else
                    repairs := repair_byte(decoded.data, entry.checksum, entry.computed)
                    if #repairs > 0
                        repaired := repairs[0]
                        entry.status = "repaired"
                        entry.data = repaired.data
                        entry.repair = repaired.record
                    else
                        entry.status = "checksum"
        parsed.entries = parsed.entries.append(entry)
        if entry.status == "ok" or entry.status == "repaired"
            parsed.intact = parsed.intact.append(entry)
        position = entry.finish
    if not parsed.stopped
        parsed.trailing = int(#archive) - position
    return parsed

int[] verify(int[] archive, Encoding[] encodings)
    parsed := parse_archive(archive)
    verified := 0
    for: index < int(#parsed.entries)
        entry := parsed.entries[index]
        if index < int(#encodings) and entry.status == "ok" and entry.data == encodings[index].data
            verified += 1
    return [int(#parsed.entries), verified]
string hex_value(int value, int width)
    if width <= 0
        return ""
    low := nat(value % 16 ^ width)
    hex := low.b16
    return "0" * (width - #hex) + hex
string entry_test_line(Entry entry, int archive_size)
    prefix := "entry {entry.number} offset {entry.offset}"
    if not entry.valid_header
        return prefix + " status " + (if entry.status == "bad_header" then "bad_header" else "truncated header")
    line := prefix + " name {entry.name} method " + method_name(entry.method) + " status {entry.status}"
    if entry.status == "repaired"
        line += " byte {entry.repair[0]} from {entry.repair[1]} to {entry.repair[2]}"
    elif entry.status == "checksum"
        line += " stored " + hex_value(entry.checksum, 8) + " computed " + hex_value(entry.computed, 8)
    elif entry.status == "error"
        line += " {entry.failure_kind} at {entry.failure_position}"
    elif entry.status == "truncated"
        line += " payload have {archive_size - entry.start} need {entry.compressed}"
    return line

string[] test_report(Parsed parsed, Encoding[] encodings, int[] salvaged, int verified)
    string[] lines := []
    if parsed.reason != ""
        lines = ["archive error {parsed.reason}"]
    else
        names := status_names()
        counts := array(7, fill = 0)
        for entry := parsed.entries[]
            lines = lines.append(entry_test_line(entry, parsed.size))
            for: status < 7
                if entry.status == names[status]
                    counts[status] += 1
        line := "summary entries {parsed.declared} checked {#parsed.entries}"
        for: status < 7
            line += " " + names[status] + " {counts[status]}"
        lines = lines.append(line + " missing {parsed.declared - int(#parsed.entries)} trailing {parsed.trailing}")
    original := 0
    for: index < int(#encodings)
        encoded := encodings[index]
        entry := parsed.intact[index]
        original += int(#encoded.data)
        lines = lines.append("salvage {entry.name} method " + method_name(encoded.best) + " compressed {#encoded.payload} was " + method_name(entry.method) + " {entry.compressed}")
    return lines.append("salvaged entries {#encodings} bytes {#salvaged} original {original} verified {verified}")

string[] text_lines(int[] data)
    text := byte_text(data)
    lines := text.split(NL)
    return lines[0:<int(#lines) - 1]

string[] extraction_report(Entry[] entries)
    string[] lines := []
    for entry := entries[]
        contents := text_lines(entry.data)
        words := 0
        longest := 0
        for line := contents[]
            if int(#line) > longest
                longest = int(#line)
            for word := line.split(" ")[]
                if word != ""
                    words += 1
        lines = lines.append("file {entry.name} bytes {#entry.data} lines {#contents} words {words} longest {longest}")
        lines = lines.concat(contents)
    return lines
string[] hex_lines(int[] data)
    string[] lines := []
    for: start < int(#data) step 32
        end := if start + 32 < int(#data) then start + 32 else int(#data)
        line := ""
        for: start <= index < end
            line += hex_value(data[index], 2)
        lines = lines.append(line)
    return lines

int[] parse_hex(string text)
    digits := text.codepoints()
    int[] values := []
    for: index < int(#digits) step 2
        first := if digits[index] <= 57 then digits[index] - 48 else digits[index] - 87
        second := if digits[index + 1] <= 57 then digits[index + 1] - 48 else digits[index + 1] - 87
        values = values.append(16 * first + second)
    return values
