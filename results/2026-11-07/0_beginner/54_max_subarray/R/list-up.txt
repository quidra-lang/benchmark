for (numbers in arrays) {
  current <- best <- numbers[1]
  start <- best_start <- best_end <- 1
  for (i in 2:length(numbers)) {
    if (numbers[i] > current + numbers[i]) {
      current <- numbers[i]
      start <- i
    } else {
      current <- current + numbers[i]
    }
    if (current > best) {
      best <- current
      best_start <- start
      best_end <- i
    }
  }
  cat(sprintf("max sum %d at [%d, %d]\n", best, best_start - 1, best_end - 1))
