        var current = values[0]
        var best = current
        var start = 0
        var bestStart = 0
        var bestEnd = 0
        for (i in 1 until values.size) {
            if (current < 0) {
                current = values[i]
                start = i
            } else {
                current += values[i]
            }
            if (current > best) {
                best = current
                bestStart = start
                bestEnd = i
            }
        }
        println("max sum $best at [$bestStart, $bestEnd]")
