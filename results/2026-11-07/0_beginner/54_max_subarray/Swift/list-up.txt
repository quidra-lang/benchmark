    var current = values[0], best = current, start = 0, bestStart = 0, bestEnd = 0
    for i in 1..<values.count {
        if current < 0 {
            current = values[i]
            start = i
        } else {
            current += values[i]
        }
        if current > best {
            best = current
            bestStart = start
            bestEnd = i
        }
    }
    print("max sum \(best) at [\(bestStart), \(bestEnd)]")
