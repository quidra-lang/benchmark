1class EvaluationError implements Exception {
2 final String message;
3 EvaluationError(this.message);
4}
5
6class Parser {
7 final List<String> tokens;
8 final Map<String, int> variables;
9 int position;
10 Parser(this.tokens, this.variables, this.position);
11
12 bool at(String s) => position < tokens.length && tokens[position] == s;
13
14 int expression() {
15 var result = term();
16 while (at('+') || at('-')) {
17 final op = tokens[position++], right = term();
18 result = op == '+' ? result + right : result - right;
19 }
20 return result;
21 }
22
23 int term() {
24 var result = unary();
25 while (at('*') || at('/')) {
26 final op = tokens[position++], right = unary();
27 if (op == '/' && right == 0) throw EvaluationError('division by zero');
28 result = op == '*' ? result * right : result ~/ right;
29 }
30 return result;
31 }
32
33 int unary() {
34 if (at('-')) {
35 position++;
36 return -unary();
37 }
38 return primary();
39 }
40
41 int primary() {
42 if (at('(')) {
43 position++;
44 final result = expression();
45 position++;
46 return result;
47 }
48 final token = tokens[position++];
49 if (RegExp(r'^\d').hasMatch(token)) return int.parse(token);
50 return variables[token] ??
51 (throw EvaluationError('undefined variable $token'));
52 }
53}
54
55void main() {
56 const lines = [
57 'width = 12',
58 'height = width - 5',
59 'area = width * height',
60 '(area + 6) / 4',
61 'shift = -area / 8 + 3',
62 'height = 100 / (height - 7) + bonus',
63 'volume = area * depth',
64 'width = (width+3)*-(2-4)',
65 'width * shift - height',
66 ];
67 final variables = <String, int>{};
68 for (final line in lines) {
69 final tokens = RegExp(r'\d+|[a-z]+|[()+*/=\-]')
70 .allMatches(line)
71 .map((m) => m.group(0)!)
72 .toList();
73 final assignment = tokens.length > 1 && tokens[1] == '=';
74 final parser = Parser(tokens, variables, assignment ? 2 : 0);
75 try {
76 final result = parser.expression();
77 if (assignment) {
78 variables[tokens[0]] = result;
79 print('${tokens[0]} = $result');
80 } else {
81 print(result);
82 }
83 } on EvaluationError catch (error) {
84 print('error: ${error.message}');
85 }
86 }
87}
