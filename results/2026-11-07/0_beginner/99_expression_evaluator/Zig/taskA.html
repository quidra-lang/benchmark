1const std = @import("std");
2
3const EvalError = error{ DivisionByZero, UndefinedVariable };
4const Table = std.StringHashMapUnmanaged(i64);
5
6const Parser = struct {
7 tokens: []const []const u8,
8 position: usize = 0,
9 variables: *Table,
10 missing: []const u8 = "",
11
12 fn at(self: *Parser, s: []const u8) bool {
13 return self.position < self.tokens.len and std.mem.eql(u8, self.tokens[self.position], s);
14 }
15
16 fn expression(self: *Parser) EvalError!i64 {
17 var result = try self.term();
18 while (self.at("+") or self.at("-")) {
19 const op = self.tokens[self.position][0];
20 self.position += 1;
21 const right = try self.term();
22 result = if (op == '+') result + right else result - right;
23 }
24 return result;
25 }
26
27 fn term(self: *Parser) EvalError!i64 {
28 var result = try self.unary();
29 while (self.at("*") or self.at("/")) {
30 const op = self.tokens[self.position][0];
31 self.position += 1;
32 const right = try self.unary();
33 if (op == '/' and right == 0)
34 return error.DivisionByZero;
35 result = if (op == '*') result * right else @divTrunc(result, right);
36 }
37 return result;
38 }
39
40 fn unary(self: *Parser) EvalError!i64 {
41 if (self.at("-")) {
42 self.position += 1;
43 return -(try self.unary());
44 }
45 return self.primary();
46 }
47
48 fn primary(self: *Parser) EvalError!i64 {
49 if (self.at("(")) {
50 self.position += 1;
51 const result = try self.expression();
52 self.position += 1;
53 return result;
54 }
55 const token = self.tokens[self.position];
56 self.position += 1;
57 if (std.ascii.isDigit(token[0]))
58 return std.fmt.parseInt(i64, token, 10) catch unreachable;
59 if (self.variables.get(token)) |value|
60 return value;
61 self.missing = token;
62 return error.UndefinedVariable;
63 }
64};
65
66pub fn main(init: std.process.Init) !void {
67 var output_buffer: [8192]u8 = undefined;
68 var writer = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
69 const out = &writer.interface;
70 const lines = [_][]const u8{
71 "width = 12",
72 "height = width - 5",
73 "area = width * height",
74 "(area + 6) / 4",
75 "shift = -area / 8 + 3",
76 "height = 100 / (height - 7) + bonus",
77 "volume = area * depth",
78 "width = (width+3)*-(2-4)",
79 "width * shift - height",
80 };
81 var variables: Table = .empty;
82 defer variables.deinit(std.heap.page_allocator);
83 for (lines) |line| {
84 var tokens: [128][]const u8 = undefined;
85 var len: usize = 0;
86 var i: usize = 0;
87 while (i < line.len) {
88 if (line[i] == ' ') {
89 i += 1;
90 continue;
91 }
92 const start = i;
93 const digit = std.ascii.isDigit(line[i]);
94 const letter = std.ascii.isLower(line[i]);
95 i += 1;
96 if (digit) {
97 while (i < line.len and std.ascii.isDigit(line[i]))
98 i += 1;
99 } else if (letter) {
100 while (i < line.len and std.ascii.isLower(line[i]))
101 i += 1;
102 }
103 tokens[len] = line[start..i];
104 len += 1;
105 }
106 const assignment = len > 1 and std.mem.eql(u8, tokens[1], "=");
107 var parser = Parser{ .tokens = tokens[0..len], .position = if (assignment) 2 else 0, .variables = &variables };
108 const result = parser.expression() catch |err| {
109 switch (err) {
110 error.DivisionByZero => try out.writeAll("error: division by zero\n"),
111 error.UndefinedVariable => try out.print("error: undefined variable {s}\n", .{parser.missing}),
112 }
113 continue;
114 };
115 if (assignment) {
116 try variables.put(std.heap.page_allocator, tokens[0], result);
117 try out.print("{s} = {d}\n", .{ tokens[0], result });
118 } else {
119 try out.print("{d}\n", .{result});
120 }
121 }
122 try out.flush();
123}
