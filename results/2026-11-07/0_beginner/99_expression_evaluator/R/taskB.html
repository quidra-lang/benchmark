1evaluate_tokens <- function(tokens, variables, start) {
2 position <- start
3 factor <- function() {
4 token <- tokens[position]
5 position <<- position + 1
6 if (token == "-") {
7 return(-factor())
8 }
9 if (token == "(") {
10 value <- expression()
11 position <<- position + 1
12 return(value)
13 }
14 if (grepl("^[0-9]", token)) {
15 return(as.numeric(token))
16 }
17 if (!token %in% names(variables)) stop(paste("undefined variable", token), call. = FALSE)
18 unname(variables[token])
19 }
20 term <- function() {
21 value <- factor()
22 while (position <= length(tokens) && tokens[position] %in% c("*", "/")) {
23 operation <- tokens[position]
24 position <<- position + 1
25 right <- factor()
26 if (operation == "/" && right == 0) stop("division by zero", call. = FALSE)
27 value <- if (operation == "*") value * right else trunc(value / right)
28 }
29 value
30 }
31 expression <- function() {
32 value <- term()
33 while (position <= length(tokens) && tokens[position] %in% c("+", "-")) {
34 operation <- tokens[position]
35 position <<- position + 1
36 right <- term()
37 value <- if (operation == "+") value + right else value - right
38 }
39 value
40 }
41 expression()
42}
43lines <- c(
44 "width = 12", "height = width - 5", "area = width * height", "(area + 6) / 4",
45 "shift = -area / 8 + 3", "height = 100 / (height - 7) + bonus", "volume = area * depth",
46)
47variables <- numeric(0)
48for (line in lines) {
49 tokens <- regmatches(line, gregexpr("[0-9]+|[a-z]+|[()+*/=-]", line, perl = TRUE))[[1]]
50 assignment <- length(tokens) >= 2 && tokens[2] == "="
51 result <- tryCatch(
52 list(ok = TRUE, value = evaluate_tokens(tokens, variables, if (assignment) 3 else 1)),
53 error = function(e) list(ok = FALSE, error = conditionMessage(e))
54 )
55 if (!result$ok) {
56 cat("error: ", result$error, "\n", sep = "")
57 } else if (assignment) {
58 variables[tokens[1]] <- result$value
59 } else {
60 cat(sprintf("%d\n", result$value))
61 }
