1import Foundation
2
3enum EvaluationError: Error { case failure(String) }
4
5class Parser {
6 let tokens: [String], variables: [String: Int]
7 var position: Int
8
9 init(_ tokens: [String], _ position: Int, _ variables: [String: Int]) {
10 self.tokens = tokens
11 self.position = position
12 self.variables = variables
13 }
14
15 func take(_ token: String) -> Bool {
16 if position >= tokens.count || tokens[position] != token { return false }
17 position += 1
18 return true
19 }
20
21 func expression() throws -> Int {
22 var result = try term()
23 while position < tokens.count && (tokens[position] == "+" || tokens[position] == "-") {
24 let operation = tokens[position]
25 position += 1
26 let right = try term()
27 result = operation == "+" ? result + right : result - right
28 }
29 return result
30 }
31
32 func term() throws -> Int {
33 var result = try unary()
34 while position < tokens.count && (tokens[position] == "*" || tokens[position] == "/") {
35 let operation = tokens[position]
36 position += 1
37 let right = try unary()
38 if operation == "/" && right == 0 { throw EvaluationError.failure("division by zero") }
39 result = operation == "*" ? result * right : result / right
40 }
41 return result
42 }
43
44 func unary() throws -> Int {
45 if take("-") { return -(try unary()) }
46 if take("(") {
47 let result = try expression()
48 _ = take(")")
49 return result
50 }
51 let token = tokens[position]
52 position += 1
53 if let number = Int(token) { return number }
54 guard let value = variables[token] else { throw EvaluationError.failure("undefined variable " + token) }
55 return value
56 }
57}
58
59let lines = ["width = 12", "height = width - 5", "area = width * height", "(area + 6) / 4",
60 "shift = -area / 8 + 3", "height = 100 / (height - 7) + bonus", "volume = area * depth",
61 "width = (width+3)*-(2-4)", "width * shift - height"]
62var variables = [String: Int]()
63for line in lines {
64 let tokens = line.matches(of: #/[0-9]+|[a-z]+|[^ ]/#).map { String($0.output) }
65 let assignment = tokens.count > 1 && tokens[1] == "="
66 let name = assignment ? tokens[0] : nil
67 do {
68 let result = try Parser(tokens, assignment ? 2 : 0, variables).expression()
69 if let name {
70 variables[name] = result
71 print("\(name) = \(result)")
72 } else {
73 print(result)
74 }
75 } catch EvaluationError.failure(let message) {
76 print("error: \(message)")
77 } catch { fatalError("Unexpected evaluation error") }
78}
