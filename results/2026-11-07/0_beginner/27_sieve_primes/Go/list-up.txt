	limit := 200
	prime := make([]bool, limit+1)
	for n := 2; n <= limit; n++ {
		prime[n] = true
	}
	for p := 2; p <= limit; p++ {
		if prime[p] {
			for j := p * 2; j <= limit; j += p {
				prime[j] = false
			}
		}
	}
	primes := []int{}
	for n := 2; n <= limit; n++ {
		if prime[n] {
			primes = append(primes, n)
		}
	}
