5func cross(_ o: Point, _ a: Point, _ b: Point) -> Int {
6 (a.x - o.x) * (b.y - o.y) - (a.y - o.y) * (b.x - o.x)

9func chain(_ points: [Point]) -> [Point] {
10 var result = [Point]()
11 for point in points {
12 while result.count >= 2 && cross(result[result.count - 2], result.last!, point) <= 0 {
13 result.removeLast()
14 }
15 result.append(point)
16 }
17 return result
18}
