6using Point = std::pair<int, int>;
7
8int cross(Point o, Point a, Point b) {
9 return (a.first - o.first) * (b.second - o.second) - (a.second - o.second) * (b.first - o.first);
10}
11
12int main() {
13 std::vector<Point> points = {{3, 2}, {-6, 1}, {8, 4}, {0, 0}, {-3, 7}, {4, -3}, {1, -5}, {-2, 1},
14 {5, 6}, {-6, -2}, {2, -2}, {7, -1}, {-3, 4}, {2, 8}, {-6, 3}};
15 std::sort(points.begin(), points.end());
16 std::vector<Point> lower, upper;
17 for (Point p : points) {
18 while (lower.size() >= 2 && cross(lower[lower.size() - 2], lower.back(), p) <= 0)
19 lower.pop_back();
20 lower.push_back(p);
21 }
22 for (auto i = points.rbegin(); i != points.rend(); ++i) {
23 Point p = *i;
24 while (upper.size() >= 2 && cross(upper[upper.size() - 2], upper.back(), p) <= 0)
25 upper.pop_back();
26 upper.push_back(p);
27 }
28 lower.pop_back();
29 upper.pop_back();
30 lower.insert(lower.end(), upper.begin(), upper.end());
31 int twice = 0;
32 std::cout << "hull: " << lower.size() << " points\n";
33 for (std::size_t i = 0; i < lower.size(); i++) {
34 Point p = lower[i], q = lower[(i + 1) % lower.size()];
35 std::cout << '(' << p.first << ", " << p.second << ")\n";
36 twice += p.first * q.second - q.first * p.second;
37 }
38 std::cout << "area: " << twice / 2 << '.' << twice % 2 * 5 << '\n';
39}
