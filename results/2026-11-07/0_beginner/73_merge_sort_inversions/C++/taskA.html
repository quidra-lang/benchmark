5void merge_sort(std::vector<int> &a, int l, int r, int &count) {
6 if (l == r)
7 return;
8 int m = (l + r) / 2;
9 merge_sort(a, l, m, count);
10 merge_sort(a, m + 1, r, count);
11 std::vector<int> temp;
12 int i = l, j = m + 1;
13 while (i <= m && j <= r) {
14 if (a[i] < a[j]) {
15 temp.push_back(a[i++]);
16 } else {
17 temp.push_back(a[j++]);
18 count += m - i + 1;
19 }
20 }
21 temp.insert(temp.end(), a.begin() + i, a.begin() + m + 1);
22 temp.insert(temp.end(), a.begin() + j, a.begin() + r + 1);
23 std::copy(temp.begin(), temp.end(), a.begin() + l);
24 std::cout << "merge [" << l << ".." << r << "]:";
25 for (int k = l; k <= r; k++)
26 std::cout << ' ' << a[k];
27 std::cout << '\n';
28}
