void merge_sort(int[] &values, int left, int right, int &inversions)
    middle := (left + right) / 2
    merge_sort(&values, left, middle, &inversions)
    merge_sort(&values, middle + 1, right, &inversions)
    first := values[left:middle]
    second := values[(middle + 1):right]
    i := 0
    j := 0
    while i < int(#first) and j < int(#second)
        if first[i] <= second[j]
            merged = merged.append(first[i])
            i += 1
        else
            merged = merged.append(second[j])
            j += 1
            inversions += int(#first) - i
    println("merge [{left}..{right}]: {text.join(merged, " ")}")
inversions := 0
merge_sort(&values, 0, 9, &inversions)
