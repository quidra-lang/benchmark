6static void MergeSort(int[] values, int l, int r, ref int inversions)
7{
8 if (l == r) return;
9 int m = (l + r) / 2;
10 MergeSort(values, l, m, ref inversions);
11 MergeSort(values, m + 1, r, ref inversions);
12 int[] merged = new int[r - l + 1];
13 int i = l, j = m + 1, k = 0;
14 while (i <= m && j <= r)
15 {
16 if (values[i] < values[j])
17 {
18 merged[k++] = values[i++];
19 }
20 else
21 {
22 merged[k++] = values[j++];
23 inversions += m - i + 1;
24 }
25 }
26 while (i <= m) merged[k++] = values[i++];
27 while (j <= r) merged[k++] = values[j++];
28 Array.Copy(merged, 0, values, l, merged.Length);
29 Console.WriteLine($"merge [{l}..{r}]: {string.Join(" ", merged)}");
30}

33int inversions = 0;
34MergeSort(values, 0, values.Length - 1, ref inversions);
35Console.WriteLine($"sorted: {string.Join(" ", values)}");
36Console.WriteLine($"inversions: {inversions}");
