void merge_sort(int *a, int l, int r, int *count) {
    int m = (l + r) / 2;
    merge_sort(a, l, m, count);
    merge_sort(a, m + 1, r, count);
    int temp[10], i = l, j = m + 1, k = 0;
    while (i <= m && j <= r) {
        if (a[i] < a[j]) {
            temp[k++] = a[i++];
        } else {
            temp[k++] = a[j++];
            *count += m - i + 1;
        }
    }
    printf("merge [%d..%d]:", l, r);
    for (k = l; k <= r; k++)
        printf(" %d", a[k]);
    putchar('\n');
}
