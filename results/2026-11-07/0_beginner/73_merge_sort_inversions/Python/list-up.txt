inversions = 0
def merge_sort(left, right):
    global inversions
    middle = (left + right) // 2
    merge_sort(left, middle)
    merge_sort(middle + 1, right)
    a, b = values[left : middle + 1], values[middle + 1 : right + 1]
    i = j = 0
    while i < len(a) and j < len(b):
        if a[i] < b[j]:
            merged.append(a[i])
            i += 1
        else:
            merged.append(b[j])
            j += 1
            inversions += len(a) - i
    print(f"merge [{left}..{right}]:", *merged)
