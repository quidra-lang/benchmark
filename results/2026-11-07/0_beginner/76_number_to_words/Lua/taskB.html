1local units = {
2 "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine", "ten", "eleven",
3 "twelve", "thirteen", "fourteen", "fifteen", "sixteen", "seventeen", "eighteen", "nineteen",

5local tens = {"zero", "ten", "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety"}
6local function small_words(number)
7 local parts = {}
8 local hundreds, remainder = number // 100, number % 100
9 if hundreds > 0 then parts[#parts + 1] = units[hundreds + 1] .. " hundred" end
10 if remainder >= 20 then
11 local phrase = tens[remainder // 10 + 1]
12 if remainder % 10 > 0 then phrase = phrase .. "-" .. units[remainder % 10 + 1] end
13 parts[#parts + 1] = phrase
14 elseif remainder > 0 then
15 parts[#parts + 1] = units[remainder + 1]
16 end
17 return table.concat(parts, " ")
18local function number_words(number)
19 if number == 0 then return "zero" end
20 local parts = {}
21 local thousands, remainder = number // 1000, number % 1000
22 if thousands > 0 then parts[#parts + 1] = small_words(thousands) .. " thousand" end
23 if remainder > 0 then parts[#parts + 1] = small_words(remainder) end
24 return table.concat(parts, " ")
