	sales := []int{120, 135, 98, 143, 160, 152, 171, 149, 188, 176}
	windowSize := 3
	for i := windowSize - 1; i < len(sales); i++ {
		sum := 0
		for _, sale := range sales[i-windowSize+1 : i+1] {
			sum += sale
		}
		avg := float64(sum) / float64(windowSize)
		fmt.Printf("day %d: %.2f\n", i+1, avg)
	}
