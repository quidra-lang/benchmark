14local function find_word(word)
15 local directions = {
16 {"E", 0, 1}, {"SE", 1, 1}, {"S", 1, 0}, {"SW", 1, -1}, {"W", 0, -1}, {"NW", -1, -1},
17 {"N", -1, 0}, {"NE", -1, 1},
18 }
19 for row = 0, 9 do
20 for column = 0, 9 do
21 for _, direction in ipairs(directions) do
22 local matched = true
23 for offset = 0, #word - 1 do
24 local r, c = row + offset * direction[2], column + offset * direction[3]
25 if r < 0 or r >= 10 or c < 0 or c >= 10
26 or board[r + 1]:sub(c + 1, c + 1) ~= word:sub(offset + 1, offset + 1) then
27 matched = false
28 break
29 end
30 end
31 if matched then return string.format("(%d, %d) %s", row, column, direction[1]) end
32 end
33 end
34 end
35 return "not found"
36end
