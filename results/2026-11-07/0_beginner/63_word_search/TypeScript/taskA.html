12function find(word: string): string {

14 for (let column = 0; column < board[0].length; column++) {
15 for (const {name, dr, dc} of directions) {
16 const matches = [...word].every((char, i) => {
17 const r = row + dr * i, c = column + dc * i;
18 return r >= 0 && r < board.length && c >= 0 && c < board[0].length && board[r][c] === char;
19 });
20 if (matches) return `(${row}, ${column}) ${name}`;
21 }
22 }
23 }
