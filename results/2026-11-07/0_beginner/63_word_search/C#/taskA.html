7string[] board = { "MMONKENBIR", "PMTIGERJPL", "TGREJEDUZI", "SPYUTEJDEC", "WKAPSIDLBN",

10int[] dr = { 0, 1, 1, 1, 0, -1, -1, -1 }, dc = { 1, 1, 0, -1, -1, -1, 0, 1 };

16 {
17 for (int c = 0; c < 10 && !found; c++)
18 {
19 for (int d = 0; d < 8; d++)
20 {

22 {
23 int nr = r + k * dr[d], nc = c + k * dc[d];
24 return nr >= 0 && nr < 10 && nc >= 0 && nc < 10 && board[nr][nc] == word[k];
25 });
26 if (match)
27 {
28 result = $"({r}, {c}) {directions[d]}";
29 found = true;
30 break;
31 }
32 }
33 }
34 }
