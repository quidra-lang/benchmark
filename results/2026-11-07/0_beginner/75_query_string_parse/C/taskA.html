5typedef struct {
6 char key[128], value[256];
7 int order;
8} Entry;
9
10int compare(const void *a, const void *b) {
11 const Entry *x = a, *y = b;
12 int c = strcmp(x->key, y->key);
13 return c ? c : x->order - y->order;
14}
15
16void decode(const char *s, char *out) {
17 while (*s) {
18 if (*s == '+') {
19 *out++ = ' ';
20 s++;
21 } else if (*s == '%') {
22 unsigned byte;
23 sscanf(s + 1, "%2x", &byte);
24 *out++ = (char)byte;
25 s += 3;
26 } else {
27 *out++ = *s++;
28 }
29 }
30 *out = 0;
31}

53 }

55 printf("query %d:\n", k + 1);

57 if (i == 0 || strcmp(entries[i].key, entries[i - 1].key)) {
