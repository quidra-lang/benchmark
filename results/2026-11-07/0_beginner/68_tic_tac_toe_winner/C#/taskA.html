7string[][] boards = { new[] { "XXX", "OO.", "O.X" }, new[] { "X.O", ".X.", "O.." },
8 new[] { "XO.", ".OX", "XO." }, new[] { "XOX", "XOO", "OXX" },

13{
14 string cells = string.Concat(boards[i]);

16 cells[line[0]] == cells[line[1]] && cells[line[0]] == cells[line[2]]);
