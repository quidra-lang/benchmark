        pair = (squared, i, j)
        if closest is None or squared < closest[0]:
            closest = pair
        if farthest is None or squared > farthest[0]:
            farthest = pair
for label, (squared, i, j) in [("closest", closest), ("farthest", farthest)]:
    distance = sqrt(squared)
    print(f"{label}: P{i} P{j} {distance:.4f}")
