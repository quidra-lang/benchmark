1class Node(var key: Int, var left: Node? = null, var right: Node? = null)
2
3fun insert(node: Node?, key: Int): Node {
4 if (node == null) return Node(key)
5 if (key < node.key) node.left = insert(node.left, key) else node.right = insert(node.right, key)
6 return node
7}
8
9fun delete(node: Node?, key: Int): Node? {
10 if (node == null) return null
11 if (key < node.key) {
12 node.left = delete(node.left, key)
13 } else if (key > node.key) {
14 node.right = delete(node.right, key)
15 } else {
16 if (node.left == null) return node.right
17 if (node.right == null) return node.left
18 var successor = node.right!!
19 while (successor.left != null) successor = successor.left!!
20 node.key = successor.key
21 node.right = delete(node.right, successor.key)
22 }
23 return node
24}

26fun traverse(node: Node?, mode: Int, order: MutableList<Int>) {
27 if (node == null) return
28 if (mode == 1) order.add(node.key)
29 traverse(node.left, mode, order)
30 if (mode == 0) order.add(node.key)
31 traverse(node.right, mode, order)
32 if (mode == 2) order.add(node.key)
33}
34
35fun height(node: Node?): Int = if (node == null) 0 else 1 + maxOf(height(node.left), height(node.right))
