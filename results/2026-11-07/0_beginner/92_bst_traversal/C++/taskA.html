7struct Node {
8 int key, left = -1, right = -1;
9};
10
11std::vector<Node> nodes;
12
13int insert(int root, int key) {
14 if (root < 0) {
15 nodes.push_back({key});
16 return static_cast<int>(nodes.size()) - 1;
17 }
18 if (key < nodes[root].key) {
19 int child = insert(nodes[root].left, key);
20 nodes[root].left = child;
21 } else {
22 int child = insert(nodes[root].right, key);
23 nodes[root].right = child;
24 }
25 return root;
26}
27
28void traverse(int root, int mode) {
29 if (root < 0)
30 return;
31 if (mode == 1)
32 std::cout << ' ' << nodes[root].key;
33 traverse(nodes[root].left, mode);
34 if (mode == 0)
35 std::cout << ' ' << nodes[root].key;
36 traverse(nodes[root].right, mode);
37 if (mode == 2)
38 std::cout << ' ' << nodes[root].key;
39}
40
41int height(int root) {
42 return root < 0 ? 0 : 1 + std::max(height(nodes[root].left), height(nodes[root].right));
43}
44
45int erase(int root, int key) {
46 if (root < 0)
47 return root;
48 if (key < nodes[root].key) {
49 nodes[root].left = erase(nodes[root].left, key);
50 } else if (key > nodes[root].key) {
51 nodes[root].right = erase(nodes[root].right, key);
52 } else {
53 if (nodes[root].left < 0)
54 return nodes[root].right;
55 if (nodes[root].right < 0)
56 return nodes[root].left;
57 int successor = nodes[root].right;
58 while (nodes[successor].left >= 0)
59 successor = nodes[successor].left;
60 nodes[root].key = nodes[successor].key;
61 nodes[root].right = erase(nodes[root].right, nodes[successor].key);
62 }
63 return root;
64}
