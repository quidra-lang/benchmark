13delete_node <- function(node, value) {
14 if (is.null(node)) {
15 return(NULL)
16 }
17 if (value < node$value) {
18 node$left <- delete_node(node$left, value)
19 } else if (value > node$value) {
20 node$right <- delete_node(node$right, value)
21 } else if (is.null(node$left)) {
22 return(node$right)
23 } else if (is.null(node$right)) {
24 return(node$left)
25 } else {
26 successor <- node$right
27 while (!is.null(successor$left)) successor <- successor$left
28 node$value <- successor$value
29 node$right <- delete_node(node$right, successor$value)
30 }
31 node
32}

34traversal <- function(node, mode) {
35 if (is.null(node)) {
36 return(integer(0))
37 }
38 left <- traversal(node$left, mode)
39 right <- traversal(node$right, mode)
40 switch(mode,
41 inorder = c(left, node$value, right),
42 preorder = c(node$value, left, right),
43 postorder = c(left, right, node$value)
44 )
45}
