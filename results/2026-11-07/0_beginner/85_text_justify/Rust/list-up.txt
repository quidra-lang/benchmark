    let width = 30;
    let words: Vec<&str> = paragraph.split(' ').collect();
    let mut start = 0;
    while start < words.len() {
        let (mut end, mut letters) = (start, 0);
        while end < words.len() && letters + words[end].len() + end - start <= width {
            letters += words[end].len();
            end += 1;
        }
        let gaps = end - start - 1;
        let last = end == words.len();
        let mut line = String::new();
        for i in start..end {
            line.push_str(words[i]);
            if i + 1 < end {
                let spaces = if last || gaps == 0 {
                    1
                } else {
                    (width - letters) / gaps + usize::from(i - start < (width - letters) % gaps)
                };
                line.push_str(&" ".repeat(spaces));
            }
        }
        line.push_str(&" ".repeat(width - line.len()));
        println!("|{}|", line);
        start = end;
