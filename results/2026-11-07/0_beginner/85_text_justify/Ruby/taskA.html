1paragraph =
2 'Good software is written for people first and for machines second. ' \
3 'A clear program explains its purpose through careful names, small functions and honest tests. ' \
4 'When a team agrees on shared rules for internationalization, documentation and code review, ' \
5 'every newcomer can read the code quickly, change it safely and ship new improvements ' \
6 'with confidence instead of fear, week after week.'
7width = 30
8lines = []
9line = []
10letters = 0
11paragraph.split(' ').each do |word|
12 if !line.empty? && letters + line.length + word.length > width
13 lines << line
14 line = []
15 letters = 0
16 end
17 line << word
18 letters += word.length
19end
20lines << line
21lines.each_with_index do |words, index|
22 if index == lines.length - 1 || words.length == 1
23 text = words.join(' ').ljust(width)
24 else
25 base, extra = (width - words.sum(&:length)).divmod(words.length - 1)
26 text = words[0]
27 words.drop(1).each_with_index do |word, gap|
28 text += ' ' * (base + (gap < extra ? 1 : 0)) + word
29 end
30 end
31 puts "|#{text}|"
32end
