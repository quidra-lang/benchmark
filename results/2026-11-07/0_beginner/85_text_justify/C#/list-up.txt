{
    if (current.Count > 0 && letters + current.Count + word.Length > width)
    {
        lines.Add(current);
        current = new List<string>();
        letters = 0;
    }
    current.Add(word);
    letters += word.Length;
}
if (current.Count > 0) lines.Add(current);
for (int i = 0; i < lines.Count; i++)
{
    var words = lines[i];
    string text;
    if (i == lines.Count - 1 || words.Count == 1)
    {
        text = string.Join(" ", words).PadRight(width);
    }
    else
    {
        int spaces = width - words.Sum(w => w.Length), gaps = words.Count - 1;
        var line = new StringBuilder(words[0]);
        for (int j = 1; j < words.Count; j++)
        {
            line.Append(' ', spaces / gaps + (j - 1 < spaces % gaps ? 1 : 0));
            line.Append(words[j]);
        }
        text = line.ToString();
    }
    Console.WriteLine($"|{text}|");
}
