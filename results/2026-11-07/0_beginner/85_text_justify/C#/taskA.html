1using System;
2using System.Collections.Generic;
3using System.Linq;
4using System.Text;
5
6string paragraph = "Good software is written for people first and for machines second. A clear program explains its purpose " +
7 "through careful names, small functions and honest tests. When a team agrees on shared rules for " +
8 "internationalization, documentation and code review, every newcomer can read the code quickly, change it " +
9 "safely and ship new improvements with confidence instead of fear, week after week.";
10int width = 30;
11var lines = new List<List<string>>();
12var current = new List<string>();
13int letters = 0;
14foreach (string word in paragraph.Split(' '))
15{
16 if (current.Count > 0 && letters + current.Count + word.Length > width)
17 {
18 lines.Add(current);
19 current = new List<string>();
20 letters = 0;
21 }
22 current.Add(word);
23 letters += word.Length;
24}
25if (current.Count > 0) lines.Add(current);
26for (int i = 0; i < lines.Count; i++)
27{
28 var words = lines[i];
29 string text;
30 if (i == lines.Count - 1 || words.Count == 1)
31 {
32 text = string.Join(" ", words).PadRight(width);
33 }
34 else
35 {
36 int spaces = width - words.Sum(w => w.Length), gaps = words.Count - 1;
37 var line = new StringBuilder(words[0]);
38 for (int j = 1; j < words.Count; j++)
39 {
40 line.Append(' ', spaces / gaps + (j - 1 < spaces % gaps ? 1 : 0));
41 line.Append(words[j]);
42 }
43 text = line.ToString();
44 }
45 Console.WriteLine($"|{text}|");
46}
