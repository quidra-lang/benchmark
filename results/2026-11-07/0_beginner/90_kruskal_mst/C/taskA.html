4typedef struct {
5 char a, b;
6 int weight;
7} Edge;
8
9int compare(const void *a, const void *b) {
10 return ((const Edge *)a)->weight - ((const Edge *)b)->weight;
11}
12
13int find(int *parent, int v) {
14 if (parent[v] != v)
15 parent[v] = find(parent, parent[v]);
16 return parent[v];
17}
18
19int main(void) {
20 Edge edges[] = {{'A', 'B', 7}, {'C', 'A', 4}, {'B', 'C', 9}, {'B', 'D', 14}, {'D', 'C', 6},
21 {'C', 'E', 3}, {'E', 'D', 5}, {'F', 'D', 11}, {'E', 'G', 16}, {'G', 'F', 8}, {'H', 'F', 12},
22 {'G', 'H', 2}, {'H', 'E', 10}};
23 qsort(edges, 13, sizeof(Edge), compare);
24 int parent[8];
25 for (int i = 0; i < 8; i++)
26 parent[i] = i;
27 int accepted = 0, total = 0, skipped = 0;
28 for (int i = 0; i < 13 && accepted < 7; i++) {
29 int a = find(parent, edges[i].a - 'A'), b = find(parent, edges[i].b - 'A');
30 if (a == b) {
31 skipped++;
32 continue;
33 }
34 parent[a] = b;
35 accepted++;
36 total += edges[i].weight;
37 char u = edges[i].a, v = edges[i].b;
38 if (u > v) {
39 char t = u;
40 u = v;
41 v = t;
42 }
43 printf("%c-%c %d\n", u, v, edges[i].weight);
44 }
45 printf("total weight: %d\nskipped: %d\n", total, skipped);
