1local items = {
2 {"tent", 10, 120}, {"stove", 3, 65}, {"lantern", 2, 45}, {"rope", 3, 20}, {"camera", 5, 90},
3 {"blanket", 5, 85}, {"water", 6, 100},
4}
5local capacity, dp = 20, {[0] = {}}
6for weight = 0, capacity do dp[0][weight] = 0 end
7for i, item in ipairs(items) do
8 dp[i] = {}
9 for weight = 0, capacity do
10 dp[i][weight] = dp[i - 1][weight]
11 if item[2] <= weight then dp[i][weight] = math.max(dp[i][weight], dp[i - 1][weight - item[2]] + item[3]) end
12 end
13end
14local chosen, remaining = {}, capacity
15for i = #items, 1, -1 do
16 if dp[i][remaining] ~= dp[i - 1][remaining] then
17 chosen[i] = true
18 remaining = remaining - items[i][2]
19 end
20end
21local names = {}
22for i, item in ipairs(items) do
23 if chosen[i] then names[#names + 1] = item[1] end
24end
25io.write(string.format("max value: %d\ntotal weight: %d\nitems: %s\n",
26 dp[#items][capacity], capacity - remaining, table.concat(names, " ")))
