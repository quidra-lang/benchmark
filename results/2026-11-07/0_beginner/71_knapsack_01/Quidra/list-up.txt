names := ["tent", "stove", "lantern", "rope", "camera", "blanket", "water"]
weights := [10, 3, 2, 3, 5, 5, 6]
values := [120, 65, 45, 20, 90, 85, 100]
limit := 20
int[][] dp := array(8, fill = array(limit + 1, fill = 0))
for: 1 <= i <= 7
    for: capacity <= limit
        dp[i][capacity] = dp[i - 1][capacity]
        if weights[i - 1] <= capacity
            candidate := dp[i - 1][capacity - weights[i - 1]] + values[i - 1]
            if candidate > dp[i][capacity]
                dp[i][capacity] = candidate
chosen := array(7, fill = false)
capacity := limit
for: 7 >= i >= 1
    if dp[i][capacity] != dp[i - 1][capacity]
        chosen[i - 1] = true
        capacity -= weights[i - 1]
total_weight := weights[chosen].sum()
items := names[chosen]
println("max value: {dp[7][limit]}{NL}total weight: {total_weight}{NL}items: {items.join(" ")}")
