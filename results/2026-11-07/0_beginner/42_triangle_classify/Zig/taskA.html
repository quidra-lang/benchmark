3pub fn main(init: std.process.Init) !void {
4 var output_buffer: [8192]u8 = undefined;
5 var writer = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
6 const out = &writer.interface;
7 const triangles = [_][3]u32{
8 .{ 3, 8, 5 }, .{ 6, 6, 6 }, .{ 5, 8, 5 }, .{ 13, 5, 12 },
9 .{ 2, 9, 4 }, .{ 7, 4, 7 }, .{ 8, 6, 7 }, .{ 4, 9, 6 },
10 };
11 for (triangles) |original| {
12 var s = original;
13 std.mem.sort(u32, &s, {}, std.sort.asc(u32));
14 try out.print("{d} {d} {d}: ", .{ original[0], original[1], original[2] });
15 if (s[0] + s[1] <= s[2]) {
16 try out.writeAll("not a triangle\n");
17 continue;
18 }
19 const side: []const u8 = if (s[0] == s[2])
20 "equilateral"
21 else if (s[0] == s[1] or s[1] == s[2])
22 "isosceles"
23 else
24 "scalene";
25 const lhs = s[2] * s[2];
26 const rhs = s[0] * s[0] + s[1] * s[1];
27 const angle: []const u8 = if (lhs < rhs) "acute" else if (lhs == rhs) "right" else "obtuse";
28 try out.print("{s} {s}\n", .{ side, angle });
29 }
30 try out.flush();
31}
