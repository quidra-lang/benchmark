1local text = {
2 "#############",
3 "#S..#.......#",
4 "#.#.#.###.#.#",
5 "#.#...#...#.#",
6 "#.###.#.###.#",
7 "#...#...#...#",
8 "###.#####.#.#",
9 "#.......#..G#",
10 "#############",
11}
12local maze, start, goal = {}, nil, nil
13for row, line in ipairs(text) do
14 maze[row] = {}
15 for column = 1, #line do
16 local c = line:sub(column, column)
17 maze[row][column] = c
18 if c == "S" then
19 start = {row, column}
20 elseif c == "G" then
21 goal = {row, column}
22 end
23 end
24end
25
26local function key(row, column) return row .. "," .. column end
27
28local queue, parents, head = {start}, {}, 1
29parents[key(table.unpack(start))] = start
30while head <= #queue do
31 local cell = queue[head]
32 head = head + 1
33 if cell[1] == goal[1] and cell[2] == goal[2] then break end
34 for _, direction in ipairs({{-1, 0}, {1, 0}, {0, -1}, {0, 1}}) do
35 local r, c = cell[1] + direction[1], cell[2] + direction[2]
36 local position = key(r, c)
37 if r >= 1 and r <= 9 and c >= 1 and c <= 13 and maze[r][c] ~= "#" and not parents[position] then
38 parents[position] = cell
39 queue[#queue + 1] = {r, c}
40 end
41 end
42end
43local cell, steps = goal, 0
44while cell[1] ~= start[1] or cell[2] ~= start[2] do
45 local r, c = table.unpack(cell)
46 if maze[r][c] == "." then maze[r][c] = "*" end
47 cell = parents[key(r, c)]
48 steps = steps + 1
49end
50io.write(string.format("steps: %d\n", steps))
51for _, row in ipairs(maze) do io.write(table.concat(row), "\n") end
