1using System;
2using System.Collections.Generic;
3using System.Globalization;
4using System.Linq;
5
6CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
7
8string[] operations = { "add email 3", "add backup 5", "add deploy 1", "run", "add report 3", "add test 2",
9 "add lint 3", "run", "run", "add cleanup 5", "add review 2", "run", "add build 1",
10 "add docs 4", "run", "run" };
11var queue = new PriorityQueue<string, (int priority, int sequence)>();
12int sequence = 0;
13foreach (string operation in operations)
14{
15 string[] parts = operation.Split(' ');
16 if (parts[0] == "add")
17 {
18 queue.Enqueue(parts[1], (int.Parse(parts[2], CultureInfo.InvariantCulture), sequence++));
19 }
20 else
21 {
22 queue.TryDequeue(out string? name, out var priority);
23 Console.WriteLine($"run: {name} (priority {priority.priority})");
24 }
25}
26var remaining = queue.UnorderedItems.OrderBy(item => item.Priority).Select(item => item.Element);
27Console.WriteLine($"remaining: {string.Join(" ", remaining)}");
