5 items <- heap$items

7 child <- length(items)
8 while (child > 1) {
9 parent <- child %/% 2

13 }

17 items <- heap$items
18 task <- items[[1]]
19 last <- tail(items, 1)[[1]]
20 items <- head(items, -1)

22 items[[1]] <- last
23 parent <- 1
24 while (2 * parent <= length(items)) {
25 child <- 2 * parent
26 if (child < length(items) && before(items[[child + 1]], items[[child]])) child <- child + 1
27 if (!before(items[[child]], items[[parent]])) break
28 items[c(child, parent)] <- items[c(parent, child)]
29 parent <- child
30 }
31 }
32 heap$items <- items

34}
