1#include <iostream>
2#include <string>
3#include <vector>
4#include <array>
5#include <algorithm>
6#include <sstream>
7
8struct Team {
9 std::string name;
10 int played = 0, wins = 0, draws = 0, losses = 0, gf = 0, ga = 0, points = 0;
11};
12
13void record(Team &t, int gf, int ga) {
14 ++t.played;
15 t.gf += gf;
16 t.ga += ga;
17 if (gf > ga) {
18 ++t.wins;
19 t.points += 3;
20 } else if (gf == ga) {
21 ++t.draws;
22 ++t.points;
23 } else {
24 ++t.losses;
25 }
26}
27
28int main() {
29 std::vector<Team> teams;
30 const std::array<std::string, 6> names = {"Lions", "Tigers", "Eagles", "Sharks", "Wolves", "Bears"};
31 const std::array<std::string, 15> matches = {
32 "Lions 0-3 Bears", "Tigers 2-2 Wolves", "Eagles 4-0 Sharks", "Bears 0-1 Sharks", "Wolves 2-1 Eagles",
33 "Lions 0-1 Tigers", "Tigers 3-3 Bears", "Eagles 2-2 Lions", "Sharks 2-2 Wolves", "Bears 3-1 Wolves",
34 "Lions 2-0 Sharks", "Tigers 2-1 Eagles", "Eagles 1-0 Bears", "Sharks 2-0 Tigers", "Wolves 2-1 Lions"};
35 for (const std::string &name : names)
36 teams.push_back({name});
37 for (const std::string &match : matches) {
38 std::istringstream input(match);
39 std::string home, away;
40 char dash;
41 int h, a;
42 input >> home >> h >> dash >> a >> away;
43 int x = 0, y = 0;
44 while (teams[x].name != home)
45 ++x;
46 while (teams[y].name != away)
47 ++y;
48 record(teams[x], h, a);
49 record(teams[y], a, h);
50 }
51 std::sort(teams.begin(), teams.end(), [](const Team &a, const Team &b) {
52 if (a.points != b.points)
53 return a.points > b.points;
54 if (a.gf - a.ga != b.gf - b.ga)
55 return a.gf - a.ga > b.gf - b.ga;
56 if (a.gf != b.gf)
57 return a.gf > b.gf;
58 return a.name < b.name;
59 });
60 for (std::size_t i = 0; i < teams.size(); i++) {
61 const Team &t = teams[i];
62 int gd = t.gf - t.ga;
63 std::cout << i + 1 << ". " << t.name << " P=" << t.played << " W=" << t.wins << " D=" << t.draws
64 << " L=" << t.losses << " GF=" << t.gf << " GA=" << t.ga << " GD=" << (gd > 0 ? "+" : "") << gd
65 << " Pts=" << t.points << '\n';
66 }
67}
