3function main()

7 "Lions 0-3 Bears",
8 "Tigers 2-2 Wolves",
9 "Eagles 4-0 Sharks",
10 "Bears 0-1 Sharks",
11 "Wolves 2-1 Eagles",
12 "Lions 0-1 Tigers",
13 "Tigers 3-3 Bears",
14 "Eagles 2-2 Lions",
15 "Sharks 2-2 Wolves",
16 "Bears 3-1 Wolves",
17 "Lions 2-0 Sharks",
18 "Tigers 2-1 Eagles",
19 "Eagles 1-0 Bears",
20 "Sharks 2-0 Tigers",
21 "Wolves 2-1 Lions",
22 ]
23 for match in matches
24 home, score, away = split(match)
25 h, a = parse.(Int, split(score, '-'))
26 for (team, gf, ga) in [(home, h, a), (away, a, h)]

35 record[3] += 1
36 record[7] += 1
37 else
38 record[4] += 1
39 end
40 end
41 end
42 sort!(
43 teams;
44 by = team -> (
45 -records[team][7],
46 -(records[team][5]-records[team][6]),
47 -records[team][5],
48 team,
49 ),
50 )
51 for (position, team) in enumerate(teams)
52 played, wins, draws, losses, gf, ga, points = records[team]
53 difference = gf-ga
54 signed = difference > 0 ? string("+", difference) : string(difference)
55 @printf(
56 "%d. %s P=%d W=%d D=%d L=%d GF=%d GA=%d GD=%s Pts=%d\n",

63 gf,
64 ga,
