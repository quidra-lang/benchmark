5int main(void) {
6 const char *expressions[] = {"3 4 + 2 *", "-17 5 / 4 -3 * -", "1 2 + * 3", "15 7 1 1 + - / 3 * 2 1 1 + + -",
7 "4 2 5 * + 1 3 2 * - 5 + /", "2 3 4 + 5", "46341 46339 * -2 /"};
8 for (int k = 0; k < 7; k++) {
9 char text[128];
10 strcpy(text, expressions[k]);
11 long long stack[64];
12 int top = 0;
13 const char *error = NULL;
14 for (char *token = strtok(text, " "); token; token = strtok(NULL, " ")) {
15 if (strlen(token) == 1 && strchr("+-*/", token[0])) {
16 if (top < 2) {
17 error = "stack underflow";
18 break;
19 }
20 long long b = stack[--top], a = stack[--top], v = 0;
21 if (token[0] == '/' && b == 0) {
22 error = "division by zero";
23 break;
24 }
25 switch (token[0]) {
26 case '+':
27 v = a + b;
28 break;
29 case '-':
30 v = a - b;
31 break;
32 case '*':
33 v = a * b;
34 break;
35 case '/':
36 v = a / b;
37 break;
38 }
39 stack[top++] = v;
40 } else {
41 stack[top++] = strtoll(token, NULL, 10);
42 }
43 }
44 if (!error && top != 1)
45 error = "leftover operands";
46 if (error)
47 printf("%s => error: %s\n", expressions[k], error);
48 else
49 printf("%s => %lld\n", expressions[k], stack[0]);
50 }
51 return 0;
52}
