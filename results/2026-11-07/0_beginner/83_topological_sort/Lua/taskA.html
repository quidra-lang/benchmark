1local function topological(courses, edges)
2 local indegree, adjacency = {}, {}
3 for _, course in ipairs(courses) do
4 indegree[course] = 0
5 adjacency[course] = {}
6 end
7 for _, edge in ipairs(edges) do
8 local source, target = table.unpack(edge)
9 adjacency[source][#adjacency[source] + 1] = target
10 indegree[target] = indegree[target] + 1
11 end
12 table.sort(courses)
13 local order, taken = {}, {}
14 while true do
15 local ready
16 for _, course in ipairs(courses) do
17 if not taken[course] and indegree[course] == 0 then
18 ready = course
19 break
20 end
21 end
22 if not ready then break end
23 order[#order + 1] = ready
24 taken[ready] = true
25 for _, target in ipairs(adjacency[ready]) do indegree[target] = indegree[target] - 1 end
26 end
27 local remaining = {}
28 for _, course in ipairs(courses) do
29 if not taken[course] then remaining[#remaining + 1] = course end
30 end
31 return order, remaining
32end
