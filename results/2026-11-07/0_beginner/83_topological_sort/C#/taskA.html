5static void Courses(string[] original, (string, string)[] edges, bool showCycle)
6{
7 string[] names = original.Order(StringComparer.Ordinal).ToArray();
8 int n = names.Length;
9 int[] degree = new int[n];
10 bool[,] graph = new bool[n, n];
11 foreach (var (source, target) in edges)
12 {
13 int a = Array.IndexOf(names, source), b = Array.IndexOf(names, target);
14 graph[a, b] = true;
15 degree[b]++;
16 }
17 bool[] taken = new bool[n];
18 var order = new List<string>();
19 while (true)
20 {
21 int next = Enumerable.Range(0, n).FirstOrDefault(i => !taken[i] && degree[i] == 0, -1);
22 if (next < 0) break;
23 taken[next] = true;
24 order.Add(names[next]);
25 for (int i = 0; i < n; i++)
26 {
27 if (graph[next, i]) degree[i]--;
28 }
29 }
30 Console.WriteLine($"order: {string.Join(" -> ", order)}");
31 if (showCycle)
32 {
33 var remaining = names.Where((_, i) => !taken[i]);
34 Console.WriteLine($"cycle: {string.Join(" ", remaining)}");
35 }
36}
