1import java.util.ArrayDeque;
2import java.util.ArrayList;
3import java.util.Arrays;
4import java.util.Collections;
5import java.util.Deque;
6import java.util.List;
7
8public class Main {
9 static void dfs(int v, boolean[][] graph, boolean[] seen, List<Integer> order) {
10 seen[v] = true;
11 order.add(v);
12 for (int w = 0; w < graph.length; w++) {
13 if (graph[v][w] && !seen[w]) {
14 dfs(w, graph, seen, order);
15 }
16 }
17 }
18
19 static String joined(List<Integer> values) {
20 return String.join(" ", values.stream().map(String::valueOf).toList());
21 }
22
23 public static void main(String[] args) {
24 boolean[][] graph = new boolean[9][9];
25 int[][] edges = {{5, 0}, {3, 8}, {7, 2}, {1, 6}, {0, 2}, {5, 3}, {7, 5}, {6, 4}, {8, 7}};
26 for (int[] edge : edges) {
27 graph[edge[0]][edge[1]] = true;
28 graph[edge[1]][edge[0]] = true;
29 }
30 List<Integer> depth = new ArrayList<>();
31 dfs(0, graph, new boolean[9], depth);
32 boolean[] seen = new boolean[9];
33 Deque<Integer> queue = new ArrayDeque<>();
34 List<Integer> breadth = new ArrayList<>();
35 seen[0] = true;
36 queue.add(0);
37 while (!queue.isEmpty()) {
38 int v = queue.remove();
39 breadth.add(v);
40 for (int w = 0; w < 9; w++) {
41 if (graph[v][w] && !seen[w]) {
42 seen[w] = true;
43 queue.add(w);
44 }
45 }
46 }
47 List<List<Integer>> components = new ArrayList<>();
48 Arrays.fill(seen, false);
49 for (int v = 0; v < 9; v++) {
50 if (!seen[v]) {
51 List<Integer> component = new ArrayList<>();
52 dfs(v, graph, seen, component);
53 Collections.sort(component);
54 components.add(component);
55 }
56 }
57 System.out.println("DFS: " + joined(depth));
58 System.out.println("BFS: " + joined(breadth));
59 System.out.println("components: " + components.size());
60 for (int i = 0; i < components.size(); i++) {
61 System.out.println("component " + (i + 1) + ": " + joined(components.get(i)));
62 }
63 }
64}
