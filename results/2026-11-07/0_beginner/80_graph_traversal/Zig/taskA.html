3fn dfs(graph: *const [9][9]bool, v: usize, visited: *[9]bool, order: *[9]usize, len: *usize) void {
4 visited[v] = true;
5 order[len.*] = v;
6 len.* += 1;
7 for (0..9) |u| {
8 if (graph[v][u] and !visited[u])
9 dfs(graph, u, visited, order, len);
10 }
11}
