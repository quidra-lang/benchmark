7static void Dfs(int v, bool[,] graph, bool[] seen, List<int> order)
8{
9 seen[v] = true;
10 order.Add(v);
11 for (int w = 0; w < seen.Length; w++)
12 {
13 if (graph[v, w] && !seen[w]) Dfs(w, graph, seen, order);
14 }
15}
16
17bool[,] graph = new bool[9, 9];
18var edges = new[] { (5, 0), (3, 8), (7, 2), (1, 6), (0, 2), (5, 3), (7, 5), (6, 4), (8, 7) };
19foreach (var (a, b) in edges)
20{
21 graph[a, b] = true;
22 graph[b, a] = true;
23}
24var depth = new List<int>();
25Dfs(0, graph, new bool[9], depth);
26bool[] seen = new bool[9];
27var queue = new Queue<int>();
28var breadth = new List<int>();
29seen[0] = true;
30queue.Enqueue(0);
31while (queue.Count > 0)
32{
33 int v = queue.Dequeue();
34 breadth.Add(v);
35 for (int w = 0; w < 9; w++)
36 {
37 if (graph[v, w] && !seen[w])
38 {
39 seen[w] = true;
40 queue.Enqueue(w);
41 }
42 }
43}
44var components = new List<List<int>>();
45Array.Fill(seen, false);
46for (int v = 0; v < 9; v++)
47{
48 if (!seen[v])
49 {
50 var component = new List<int>();
51 Dfs(v, graph, seen, component);
52 component.Sort();
53 components.Add(component);
54 }
55}
