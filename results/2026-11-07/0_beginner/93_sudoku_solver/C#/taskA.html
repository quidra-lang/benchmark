7static bool Solve(int[,] board, ref long placements)
8{
9 for (int r = 0; r < 9; r++)
10 {
11 for (int c = 0; c < 9; c++)
12 {
13 if (board[r, c] != 0) continue;
14 for (int digit = 1; digit <= 9; digit++)
15 {
16 bool valid = Enumerable.Range(0, 9).All(i => board[r, i] != digit && board[i, c] != digit &&
17 board[r / 3 * 3 + i / 3, c / 3 * 3 + i % 3] != digit);
18 if (!valid) continue;
19 board[r, c] = digit;
20 placements++;
21 if (Solve(board, ref placements)) return true;
22 board[r, c] = 0;
23 }
24 return false;
25 }
26 }
27 return true;
28}
