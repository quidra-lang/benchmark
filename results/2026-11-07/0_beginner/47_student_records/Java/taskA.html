1import java.util.Arrays;
2import java.util.Comparator;
3import java.util.List;
4import java.util.Locale;
5import java.util.stream.IntStream;
6
7public class Main {
8 record Student(String name, int total) {}
9
10 public static void main(String[] args) {
11 String[] names = {"Tanaka", "Suzuki", "Yamada", "Ito", "Kato", "Mori"};
12 int[][] scores = {{78, 85, 94}, {88, 74, 82}, {65, 90, 71}, {80, 79, 85}, {92, 88, 95}, {70, 62, 78}};
13 List<Student> students = IntStream.range(0, names.length)
14 .mapToObj(i -> new Student(names[i], Arrays.stream(scores[i]).sum()))
15 .sorted(Comparator.comparingInt(Student::total).reversed().thenComparing(Student::name))
16 .toList();
17 int rank = 0, previous = -1;
18 for (int position = 0; position < students.size(); position++) {
19 Student student = students.get(position);
20 if (student.total() != previous) {
21 rank = position + 1;
22 }
23 previous = student.total();
24 System.out.printf(Locale.ROOT, "%d. %s total=%d average=%.2f%n",
25 rank, student.name(), student.total(), student.total() / 3.0);
26 }
27 }
28}
