1const std = @import("std");
2
3const Student = struct { name: []const u8, scores: [3]u32 };
4
5fn total(s: Student) u32 {
6 return s.scores[0] + s.scores[1] + s.scores[2];
7}
8
9fn less(_: void, a: Student, b: Student) bool {
10 return if (total(a) != total(b)) total(a) > total(b) else std.mem.order(u8, a.name, b.name) == .lt;
11}
12
13pub fn main(init: std.process.Init) !void {
14 var output_buffer: [8192]u8 = undefined;
15 var writer = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
16 const out = &writer.interface;
17 var students = [_]Student{
18 .{ .name = "Tanaka", .scores = .{ 78, 85, 94 } },
19 .{ .name = "Suzuki", .scores = .{ 88, 74, 82 } },
20 .{ .name = "Yamada", .scores = .{ 65, 90, 71 } },
21 .{ .name = "Ito", .scores = .{ 80, 79, 85 } },
22 .{ .name = "Kato", .scores = .{ 92, 88, 95 } },
23 .{ .name = "Mori", .scores = .{ 70, 62, 78 } },
24 };
25 std.mem.sort(Student, &students, {}, less);
26 var rank: usize = 0;
27 var previous: u32 = 0;
28 for (students, 0..) |s, i| {
29 const t = total(s);
30 if (i == 0 or t != previous)
31 rank = i + 1;
32 previous = t;
33 try out.print("{d}. {s} total={d} average={d:.2}\n", .{ rank, s.name, t, @as(f64, @floatFromInt(t)) / 3.0 });
34 }
35 try out.flush();
36}
