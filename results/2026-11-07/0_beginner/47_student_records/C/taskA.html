5typedef struct {
6 const char *name;
7 int scores[3];
8 int total;
9} Student;
10
11int compare(const void *a, const void *b) {
12 const Student *x = a, *y = b;
13 if (x->total != y->total)
14 return y->total - x->total;
15 return strcmp(x->name, y->name);
16}
17
18int main(void) {
19 Student students[] = {{"Tanaka", {78, 85, 94}, 0}, {"Suzuki", {88, 74, 82}, 0}, {"Yamada", {65, 90, 71}, 0},
20 {"Ito", {80, 79, 85}, 0}, {"Kato", {92, 88, 95}, 0}, {"Mori", {70, 62, 78}, 0}};
21 for (int i = 0; i < 6; i++)
22 for (int j = 0; j < 3; j++)
23 students[i].total += students[i].scores[j];
24 qsort(students, 6, sizeof(Student), compare);
25 int rank = 0;
26 for (int i = 0; i < 6; i++) {
27 if (i == 0 || students[i].total != students[i - 1].total)
28 rank = i + 1;
29 printf("%d. %s total=%d average=%.2f\n",
30 rank, students[i].name, students[i].total, students[i].total / 3.0);
31 }
32 return 0;
33}
