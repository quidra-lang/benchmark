5typedef struct {

8} Hand;

10Hand evaluate(const char *text) {
11 const char *ranks = "23456789TJQKA";
12 int counts[15] = {0}, flush = 1;
13 for (int i = 0; i < 5; i++) {
14 int rank = (int)(strchr(ranks, text[i * 3]) - ranks) + 2;
15 counts[rank]++;
16 if (text[i * 3 + 1] != text[1])
17 flush = 0;
18 }
19 Hand h = {text, 0, {0}, 0};
20 int maximum = 0, pairs = 0;
21 for (int r = 14; r >= 2; r--)
22 if (counts[r]) {
23 h.key[h.length++] = r;
24 if (counts[r] > maximum)
25 maximum = counts[r];
26 if (counts[r] == 2)
27 pairs++;
28 }
29 int wheel = h.length == 5 && counts[14] && counts[2] && counts[3] && counts[4] && counts[5];
30 int straight = h.length == 5 && (h.key[0] - h.key[4] == 4 || wheel);
31 h.category = straight && flush ? 8
32 : maximum == 4 ? 7
33 : maximum == 3 && pairs == 1 ? 6
34 : flush ? 5
35 : straight ? 4
36 : maximum == 3 ? 3
37 : pairs == 2 ? 2
38 : pairs == 1 ? 1 : 0;
39 h.length = 0;
40 for (int count = 4; count >= 1; count--)
41 for (int rank = 14; rank >= 2; rank--)
42 if (counts[rank] == count)
43 h.key[h.length++] = rank;
44 if (wheel && straight)
45 for (int i = 0; i < 5; i++)
46 h.key[i] = 5 - i;
47 return h;
48}

51 const Hand *x = a, *y = b;
52 if (x->category != y->category)
53 return y->category - x->category;
54 for (int i = 0; i < 5; i++)
55 if (x->key[i] != y->key[i])
56 return y->key[i] - x->key[i];
57 return 0;
58}
