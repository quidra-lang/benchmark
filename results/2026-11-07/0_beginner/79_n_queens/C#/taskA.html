7static int Queens(int[] positions, int row, int[] first, ref bool found)
8{
9 if (row == positions.Length)
10 {
11 if (!found) Array.Copy(positions, first, positions.Length);
12 found = true;
13 return 1;
14 }
15 int count = 0;
16 for (int column = 0; column < positions.Length; column++)
17 {
18 bool safe = Enumerable.Range(0, row).All(previous =>
19 positions[previous] != column && Math.Abs(positions[previous] - column) != row - previous);
20 if (safe)
21 {
22 positions[row] = column;
23 count += Queens(positions, row + 1, first, ref found);
24 }
25 }
26 return count;
27}

31{
32 int[] positions = new int[n], first = new int[n];
33 bool found = false;
34 int solutions = Queens(positions, 0, first, ref found);
35 Console.WriteLine($"N={n}: {solutions} solutions");

37}
