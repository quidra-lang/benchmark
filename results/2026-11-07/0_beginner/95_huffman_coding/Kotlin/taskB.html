1data class Node(
2 val weight: Int,
3 val minimum: Int,
4 val symbol: Char? = null,
5 val left: Node? = null,
6 val right: Node? = null,
7)
8fun codes(node: Node, prefix: String, result: MutableMap<Char, String>) {
9 if (node.symbol != null) {
10 result[node.symbol] = prefix
11 } else {
12 codes(node.left!!, prefix + "0", result)
13 codes(node.right!!, prefix + "1", result)
14 }
15}
16 val text = "she sells sea shells by the sea shore"
17 val frequencies = text.groupingBy { it }.eachCount().toSortedMap()
18 val queue = PriorityQueue<Node>(compareBy<Node> { it.weight }.thenBy { it.minimum })
19 for ((ch, count) in frequencies) queue.add(Node(count, ch.code, ch))
20 while (queue.size > 1) {
21 val left = queue.remove()
22 val right = queue.remove()
23 queue.add(Node(left.weight + right.weight, minOf(left.minimum, right.minimum), null, left, right))
24 }
25 val mapping = mutableMapOf<Char, String>()
26 codes(queue.remove(), "", mapping)
27 for ((ch, count) in frequencies) {
28 val code = mapping.getValue(ch)
29 println("'$ch' freq=$count code=$code")
30 }
31 println("encoded bits: ${frequencies.entries.sumOf { it.value * mapping.getValue(it.key).length }}")
32 println("fixed 8-bit: ${text.length * 8}")
