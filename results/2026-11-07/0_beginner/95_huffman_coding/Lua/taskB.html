1local text = "she sells sea shells by the sea shore"
2local frequencies = {}
3for symbol in text:gmatch(".") do frequencies[symbol] = (frequencies[symbol] or 0) + 1 end
4local symbols, trees = {}, {}
5for symbol, frequency in pairs(frequencies) do
6 symbols[#symbols + 1] = symbol
7 trees[#trees + 1] = {weight = frequency, smallest = symbol:byte(), symbol = symbol}
8table.sort(symbols)
9while #trees > 1 do
10 table.sort(trees, function(a, b)
11 return a.weight < b.weight or (a.weight == b.weight and a.smallest < b.smallest)
12 end)
13 local left = table.remove(trees, 1)
14 local right = table.remove(trees, 1)
15 trees[#trees + 1] = {
16 weight = left.weight + right.weight,
17 smallest = math.min(left.smallest, right.smallest),
18 left = left,
19 right = right,
20 }
21local codes = {}
22local function visit(node, prefix)
23 if node.symbol then
24 codes[node.symbol] = prefix
25 else
26 visit(node.left, prefix .. "0")
27 visit(node.right, prefix .. "1")
28visit(trees[1], "")
29local bits = 0
30for _, symbol in ipairs(symbols) do
31 io.write(string.format("'%s' freq=%d code=%s\n", symbol, frequencies[symbol], codes[symbol]))

33io.write(string.format("encoded bits: %d\nfixed 8-bit: %d\n", bits, #text * 8))
