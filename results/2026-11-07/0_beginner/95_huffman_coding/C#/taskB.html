1static void Codes(Node node, string prefix, Dictionary<char, string> result)
2{
3 if (node.Symbol is char ch)
4 {
5 result[ch] = prefix;
6 }
7 else
8 {
9 Codes(node.Left!, prefix + "0", result);
10 Codes(node.Right!, prefix + "1", result);
11 }
12}
13string text = "she sells sea shells by the sea shore";
14var frequencies = text.GroupBy(ch => ch).ToDictionary(g => g.Key, g => g.Count());
15var queue = new PriorityQueue<Node, (int weight, int minimum)>();
16foreach (var pair in frequencies)
17{
18 var leaf = new Node(pair.Value, pair.Key, pair.Key);
19 queue.Enqueue(leaf, (leaf.Weight, leaf.Minimum));
20}
21while (queue.Count > 1)
22{
23 Node left = queue.Dequeue(), right = queue.Dequeue();
24 var parent = new Node(left.Weight + right.Weight, Math.Min(left.Minimum, right.Minimum), null, left, right);
25 queue.Enqueue(parent, (parent.Weight, parent.Minimum));
26}
27var codes = new Dictionary<char, string>();
28Codes(queue.Dequeue(), "", codes);
29foreach (char ch in frequencies.Keys.Order())
30{
31 string code = codes[ch];
32 Console.WriteLine($"'{ch}' freq={frequencies[ch]} code={code}");
33}
34Console.WriteLine($"encoded bits: {frequencies.Sum(pair => pair.Value * codes[pair.Key].Length)}");
35Console.WriteLine($"fixed 8-bit: {text.Length * 8}");
