1#include <algorithm>
2#include <array>
3#include <iostream>
4#include <string>

7struct Node {
8 int weight, minimum, left, right, symbol;
9};
10
11int pop(std::vector<int> &roots, const std::vector<Node> &nodes) {
12 auto best = std::min_element(roots.begin(), roots.end(), [&nodes](int x, int y) {
13 const Node &a = nodes[x], &b = nodes[y];
14 return a.weight < b.weight || (a.weight == b.weight && a.minimum < b.minimum);
15 });
16 int result = *best;
17 roots.erase(best);
18 return result;
19}
20
21void assign(const std::vector<Node> &nodes, int root, const std::string &prefix,
22 std::array<std::string, 128> &codes) {
23 if (nodes[root].symbol >= 0) {
24 codes[nodes[root].symbol] = prefix;
25 return;
26 }
27 assign(nodes, nodes[root].left, prefix + "0", codes);
28 assign(nodes, nodes[root].right, prefix + "1", codes);
29}
30
31int main() {
32 std::string text = "she sells sea shells by the sea shore";
33 std::array<int, 128> freq{};
34 for (unsigned char c : text)
35 ++freq[c];
36 std::vector<Node> nodes;
37 std::vector<int> roots;
38 for (int c = 0; c < 128; ++c) {
39 if (freq[c]) {
40 roots.push_back(static_cast<int>(nodes.size()));
41 nodes.push_back({freq[c], c, -1, -1, c});
42 }
43 }
44 while (roots.size() > 1) {
45 int left = pop(roots, nodes);
46 int right = pop(roots, nodes);
47 roots.push_back(static_cast<int>(nodes.size()));
48 nodes.push_back({nodes[left].weight + nodes[right].weight,
49 std::min(nodes[left].minimum, nodes[right].minimum), left, right, -1});
50 }
51 std::array<std::string, 128> codes;
52 assign(nodes, roots[0], "", codes);
53 int bits = 0;
54 for (int c = 0; c < 128; ++c) {
55 if (freq[c]) {
56 std::cout << "'" << static_cast<char>(c) << "' freq=" << freq[c]
57 << " code=" << codes[c] << '\n';
58 bits += freq[c] * static_cast<int>(codes[c].size());
59 }
60 }
61 std::cout << "encoded bits: " << bits << "\nfixed 8-bit: " << text.size() * 8 << '\n';
62}
