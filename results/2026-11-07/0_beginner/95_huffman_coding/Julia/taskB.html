1 text = "she sells sea shells by the sea shore"
2 frequencies = Dict{Char,Int}()
3 for symbol in text
4 frequencies[symbol] = get(frequencies, symbol, 0)+1
5 end
6 nodes = NamedTuple[]
7 for symbol in sort(collect(keys(frequencies)))
8 push!(
9 nodes,
10 (
11 weight = frequencies[symbol],
12 smallest = Int(symbol),
13 symbol = symbol,
14 left = 0,
15 right = 0,
16 ),
17 )
18 end
19 trees = collect(eachindex(nodes))
20 while length(trees) > 1
21 sort!(trees; by = index -> (nodes[index].weight, nodes[index].smallest))
22 left = popfirst!(trees)
23 right = popfirst!(trees)
24 push!(
25 nodes,
26 (
27 weight = nodes[left].weight+nodes[right].weight,
28 smallest = min(nodes[left].smallest, nodes[right].smallest),
29 symbol = nothing,
30 left = left,
31 right = right,
32 ),
33 )
34 push!(trees, length(nodes))
35 end
36 codes = Dict{Char,String}()
37 function visit(index, prefix)
38 node = nodes[index]
39 if node.symbol !== nothing
40 codes[node.symbol] = prefix
41 else
42 visit(node.left, prefix*"0")
43 visit(node.right, prefix*"1")
44 end
45 end
46 visit(trees[1], "")
47 bits = 0
48 for symbol in sort(collect(keys(frequencies)))
49 println("'", symbol, "' freq=", frequencies[symbol], " code=", codes[symbol])
50 bits += frequencies[symbol]*length(codes[symbol])
51 end
