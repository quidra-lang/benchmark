1 record Node(int weight, int minimum, Character symbol, Node left, Node right) {}
2 static void codes(Node node, String prefix, Map<Character, String> result) {
3 if (node.symbol != null) {
4 result.put(node.symbol, prefix);
5 } else {
6 codes(node.left, prefix + "0", result);
7 codes(node.right, prefix + "1", result);
8 }
9 }

19 }
