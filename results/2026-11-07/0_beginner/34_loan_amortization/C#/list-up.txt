int balance = 1000000, month = 0, totalInterest = 0, totalPaid = 0;
while (balance > 0)
{
    month++;
    int interest = balance / 100;
    int payment = Math.Min(50000, balance + interest);
    int principal = payment - interest;
    balance -= principal;
    totalInterest += interest;
    totalPaid += payment;
    Console.WriteLine($"month {month}: interest={interest} principal={principal} balance={balance}");
}
