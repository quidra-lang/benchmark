    local following = {}
    for row = 1, 8 do
        following[row] = {}
        for column = 1, 8 do
            local neighbors = 0
            for dr = -1, 1 do
                for dc = -1, 1 do
                    local r, c = row + dr, column + dc
                    if (dr ~= 0 or dc ~= 0) and r >= 1 and r <= 8 and c >= 1 and c <= 8 and board[r][c] then
                        neighbors = neighbors + 1
            following[row][column] = neighbors == 3 or (board[row][column] and neighbors == 2)
    board = following
