1using System;
2using System.Collections.Generic;
3using System.Globalization;
4
5CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
6
7string[] transactions = { "deposit Emma 1200", "withdraw Leo 4000", "transfer Nina Leo 5000",
8 "withdraw Leo 4000", "transfer Emma Nina 2500", "transfer Emma Nina 2000",
9 "withdraw Emma 300", "deposit Nina 750", "transfer Leo Emma 3200", "withdraw Nina 9800" };
10var balances = new SortedDictionary<string, int>(StringComparer.Ordinal)
11{
12 ["Nina"] = 12000,
13 ["Leo"] = 3500,
14 ["Emma"] = 800
15};
16for (int i = 0; i < transactions.Length; i++)
17{
18 string[] parts = transactions[i].Split(' ');
19 string name = parts[1];
20 int amount = int.Parse(parts[^1], CultureInfo.InvariantCulture);
21 bool success = parts[0] == "deposit" || balances[name] >= amount;
22 if (success)
23 {
24 if (parts[0] == "deposit")
25 {
26 balances[name] += amount;
27 }
28 else
29 {
30 balances[name] -= amount;
31 if (parts[0] == "transfer") balances[parts[2]] += amount;
32 }
33 }
34 Console.WriteLine($"op {i + 1}: {transactions[i]} {(success ? "ok" : "failed (insufficient funds)")}");
35}
36foreach (var pair in balances) Console.WriteLine($"{pair.Key}: {pair.Value}");
