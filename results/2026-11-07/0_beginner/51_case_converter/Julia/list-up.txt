for identifier in [
    "ShoppingCartItem",
]
    separated = replace(identifier, r"([A-Z])" => s" \1")
    separated = replace(separated, '_'=>' ', '-'=>' ')
    words = split(lowercase(separated))
    capitals = [uppercasefirst(word) for word in words]
    snake = join(words, "_")
    camel = words[1] * join(capitals[2:end])
    pascal = join(capitals)
    kebab = join(words, "-")
