1fn value(token: &str, registers: &[i64; 4]) -> i64 {
2 let c = token.as_bytes()[0];
3 if (b'A'..=b'D').contains(&c) {
4 registers[(c - b'A') as usize]
5 } else {
6 token.parse().unwrap()
7 }
8}
