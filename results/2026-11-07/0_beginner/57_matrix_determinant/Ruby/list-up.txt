def determinant(matrix)
  return matrix[0][0] if matrix.length == 1
  total = matrix[0].each_with_index.sum do |value, j|
    minor = matrix.drop(1).map { |row| row[0...j] + row.drop(j + 1) }
    term = value * determinant(minor)
    j.even? ? term : -term
  end
  total
