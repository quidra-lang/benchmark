laplace <- function(values) {
  n <- nrow(values)
  if (n == 1) {
    return(values[1, 1])
  }
  total <- 0L
  for (j in seq_len(n)) {
    term <- values[1, j] * laplace(values[-1, -j, drop = FALSE])
    total <- total + if (j %% 2 == 1) term else -term
  }
  total
