fn det(a: &[Vec<i32>]) -> i32 {
    if a.len() == 1 {
        return a[0][0];
    }
    let mut result = 0;
    for j in 0..a.len() {
        let minor: Vec<Vec<i32>> = a[1..]
            .iter()
            .map(|row| {
                row.iter()
                    .enumerate()
                    .filter_map(|(c, &v)| (c != j).then_some(v))
                    .collect()
            })
            .collect();
        let term = a[0][j] * det(&minor);
        result += if j % 2 == 0 { term } else { -term };
    }
