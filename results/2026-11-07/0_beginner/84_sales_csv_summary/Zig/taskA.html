1const std = @import("std");
2
3const Entry = struct { name: []const u8, value: u32 };
4
5fn add(entries: *[12]Entry, len: *usize, name: []const u8, value: u32) void {
6 for (entries[0..len.*]) |*e| {
7 if (std.mem.eql(u8, e.name, name)) {
8 e.value += value;
9 return;
10 }
11 }
12 entries[len.*] = .{ .name = name, .value = value };
13 len.* += 1;
14}
15
16fn nameLess(_: void, a: Entry, b: Entry) bool {
17 return std.mem.order(u8, a.name, b.name) == .lt;
18}
19
20fn salesLess(_: void, a: Entry, b: Entry) bool {
21 return if (a.value != b.value) a.value > b.value else nameLess({}, a, b);
22}
23
24pub fn main(init: std.process.Init) !void {
25 var output_buffer: [8192]u8 = undefined;
26 var writer = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
27 const out = &writer.interface;
28 const lines = [_][]const u8{
29 "date,region,product,quantity,unit_price",
30 "2026-04-01,north,notebook,10,200",
31 "2026-04-01,east,marker,6,150",
32 "2026-04-01,south,eraser,25,80",
33 "2026-04-02,west,stapler,4,600",
34 "2026-04-02,north,marker,8,150",
35 "2026-04-02,east,notebook,9,200",
36 "2026-04-03,south,stapler,3,600",
37 "2026-04-03,west,eraser,15,80",
38 "2026-04-03,north,eraser,20,80",
39 "2026-04-04,east,stapler,5,600",
40 "2026-04-04,south,notebook,6,200",
41 "2026-04-04,west,marker,8,150",
42 };
43 var regions: [12]Entry = undefined;
44 var products: [12]Entry = undefined;
45 var dates: [12]Entry = undefined;
46 var rlen: usize = 0;
47 var plen: usize = 0;
48 var dlen: usize = 0;
49 var total: u32 = 0;
50 for (lines[1..]) |line| {
51 var fields = std.mem.splitScalar(u8, line, ',');
52 const date = fields.next().?;
53 const region = fields.next().?;
54 const product = fields.next().?;
55 const quantity = try std.fmt.parseInt(u32, fields.next().?, 10);
56 const price = try std.fmt.parseInt(u32, fields.next().?, 10);

58 total += sales;
59 add(&regions, &rlen, region, sales);
60 add(&products, &plen, product, quantity);
61 add(&dates, &dlen, date, sales);
62 }
63 std.mem.sort(Entry, regions[0..rlen], {}, salesLess);
64 std.mem.sort(Entry, products[0..plen], {}, nameLess);
65 std.mem.sort(Entry, dates[0..dlen], {}, salesLess);
66 try out.print("total: {d}\n", .{total});
67 for (regions[0..rlen]) |e|
68 try out.print("{s}: {d}\n", .{ e.name, e.value });
69 for (products[0..plen]) |e|
70 try out.print("{s}: {d}\n", .{ e.name, e.value });
71 const best = dates[0];
72 try out.print("best day: {s} ({d})\n", .{ best.name, best.value });
73 try out.flush();
74}
