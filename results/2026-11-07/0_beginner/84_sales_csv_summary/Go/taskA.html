1package main
2
3import (
4 "fmt"
5 "sort"
6 "strconv"
7 "strings"
8)
9
10func main() {
11 lines := []string{"date,region,product,quantity,unit_price", "2026-04-01,north,notebook,10,200",
12 "2026-04-01,east,marker,6,150", "2026-04-01,south,eraser,25,80", "2026-04-02,west,stapler,4,600",
13 "2026-04-02,north,marker,8,150", "2026-04-02,east,notebook,9,200", "2026-04-03,south,stapler,3,600",
14 "2026-04-03,west,eraser,15,80", "2026-04-03,north,eraser,20,80", "2026-04-04,east,stapler,5,600",
15 "2026-04-04,south,notebook,6,200", "2026-04-04,west,marker,8,150"}
16 regions, products, days := map[string]int{}, map[string]int{}, map[string]int{}
17 total := 0
18 for _, line := range lines[1:] {
19 fields := strings.Split(line, ",")
20 quantity, _ := strconv.Atoi(fields[3])
21 price, _ := strconv.Atoi(fields[4])
22 amount := quantity * price
23 total += amount
24 regions[fields[1]] += amount
25 products[fields[2]] += quantity
26 days[fields[0]] += amount
27 }
28 regionNames := []string{}
29 for name := range regions {
30 regionNames = append(regionNames, name)
31 }
32 sort.Slice(regionNames, func(i, j int) bool {
33 a, b := regionNames[i], regionNames[j]
34 if regions[a] != regions[b] {
35 return regions[a] > regions[b]
36 }
37 return a < b
38 })
39 fmt.Printf("total: %d\n", total)
40 for _, name := range regionNames {
41 fmt.Printf("%s: %d\n", name, regions[name])
42 }
43 productNames := []string{}
44 for name := range products {
45 productNames = append(productNames, name)
46 }
47 sort.Strings(productNames)
48 for _, name := range productNames {
49 fmt.Printf("%s: %d\n", name, products[name])
50 }
51 best, bestTotal := "", -1
52 for date, sales := range days {
53 if sales > bestTotal || sales == bestTotal && (best == "" || date < best) {
54 best, bestTotal = date, sales
55 }
56 }
57 fmt.Printf("best day: %s (%d)\n", best, bestTotal)
58}
