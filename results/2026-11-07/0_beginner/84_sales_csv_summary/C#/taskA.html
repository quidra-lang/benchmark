1using System;
2using System.Collections.Generic;
3using System.Globalization;
4using System.Linq;
5
6CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
7
8string[] lines = {
9 "date,region,product,quantity,unit_price",
10 "2026-04-01,north,notebook,10,200",
11 "2026-04-01,east,marker,6,150",
12 "2026-04-01,south,eraser,25,80",
13 "2026-04-02,west,stapler,4,600",
14 "2026-04-02,north,marker,8,150",
15 "2026-04-02,east,notebook,9,200",
16 "2026-04-03,south,stapler,3,600",
17 "2026-04-03,west,eraser,15,80",
18 "2026-04-03,north,eraser,20,80",
19 "2026-04-04,east,stapler,5,600",
20 "2026-04-04,south,notebook,6,200",
21 "2026-04-04,west,marker,8,150" };
22var regions = new Dictionary<string, int>();
23var products = new SortedDictionary<string, int>(StringComparer.Ordinal);
24var days = new SortedDictionary<string, int>(StringComparer.Ordinal);
25foreach (string line in lines.Skip(1))
26{
27 string[] fields = line.Split(',');
28 int quantity = int.Parse(fields[3], CultureInfo.InvariantCulture);
29 int amount = quantity * int.Parse(fields[4], CultureInfo.InvariantCulture);
30 regions[fields[1]] = regions.GetValueOrDefault(fields[1]) + amount;
31 products[fields[2]] = products.GetValueOrDefault(fields[2]) + quantity;
32 days[fields[0]] = days.GetValueOrDefault(fields[0]) + amount;
33}
34Console.WriteLine($"total: {days.Values.Sum()}");
35foreach (var pair in regions.OrderByDescending(p => p.Value).ThenBy(p => p.Key, StringComparer.Ordinal))
36{
37 Console.WriteLine($"{pair.Key}: {pair.Value}");
38}
39foreach (var pair in products) Console.WriteLine($"{pair.Key}: {pair.Value}");
40var best = days.OrderByDescending(p => p.Value).ThenBy(p => p.Key, StringComparer.Ordinal).First();
41Console.WriteLine($"best day: {best.Key} ({best.Value})");
