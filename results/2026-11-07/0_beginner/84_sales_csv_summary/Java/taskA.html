1import java.util.ArrayList;
2import java.util.Comparator;
3import java.util.HashMap;
4import java.util.List;
5import java.util.Map;
6import java.util.TreeMap;
7
8public class Main {
9 public static void main(String[] args) {
10 String[] lines = {
11 "date,region,product,quantity,unit_price",
12 "2026-04-01,north,notebook,10,200",
13 "2026-04-01,east,marker,6,150",
14 "2026-04-01,south,eraser,25,80",
15 "2026-04-02,west,stapler,4,600",
16 "2026-04-02,north,marker,8,150",
17 "2026-04-02,east,notebook,9,200",
18 "2026-04-03,south,stapler,3,600",
19 "2026-04-03,west,eraser,15,80",
20 "2026-04-03,north,eraser,20,80",
21 "2026-04-04,east,stapler,5,600",
22 "2026-04-04,south,notebook,6,200",
23 "2026-04-04,west,marker,8,150"};
24 Map<String, Integer> regions = new HashMap<>(), products = new TreeMap<>(), days = new TreeMap<>();
25 for (int i = 1; i < lines.length; i++) {
26 String[] fields = lines[i].split(",");
27 int quantity = Integer.parseInt(fields[3]), amount = quantity * Integer.parseInt(fields[4]);
28 regions.merge(fields[1], amount, Integer::sum);
29 products.merge(fields[2], quantity, Integer::sum);
30 days.merge(fields[0], amount, Integer::sum);
31 }
32 System.out.println("total: " + days.values().stream().mapToInt(Integer::intValue).sum());
33 List<String> names = new ArrayList<>(regions.keySet());
34 names.sort(Comparator.<String>comparingInt(regions::get).reversed()
35 .thenComparing(Comparator.naturalOrder()));
36 for (String name : names) {
37 System.out.println(name + ": " + regions.get(name));
38 }
39 for (var entry : products.entrySet()) {
40 System.out.println(entry.getKey() + ": " + entry.getValue());
41 }
42 var best = days.entrySet().stream()
43 .min(Comparator.<Map.Entry<String, Integer>>comparingInt(Map.Entry::getValue).reversed()
44 .thenComparing(Map.Entry::getKey))
45 .orElseThrow();
46 System.out.println("best day: " + best.getKey() + " (" + best.getValue() + ")");
47 }
48}
