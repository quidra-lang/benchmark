6private fun leap(year: Int) = year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
7private fun months(year: Int) = listOf(
8 31, if (leap(year)) 29 else 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31
9)
10
11private fun dateNumber(text: String): Int {
12 val year = text.substring(0, 4).toInt()
13 val month = text.substring(5, 7).toInt()
14 val day = text.substring(8, 10).toInt()
15 if (year !in 2000..2099 || month !in 1..12) throw Fault("invalid date $text")
16 val lengths = months(year)
17 if (day !in 1..lengths[month - 1]) throw Fault("invalid date $text")
18 var result = 0
19 for (y in 2000 until year) result += if (leap(y)) 366 else 365
20 for (m in 0 until month - 1) result += lengths[m]
21 return result + day - 1
22}
23
24private fun dateText(value: Int): String {
25 var number = value
26 var year = 2000
27 while (number >= (if (leap(year)) 366 else 365)) {
28 number -= if (leap(year)) 366 else 365
29 year++
30 }
31 var month = 1
32 for (days in months(year)) {
33 if (number < days) break
34 number -= days
35 month++
36 }
37 return "%04d-%02d-%02d".format(year, month, number + 1)
38}

40private fun dated(day: Int) = "${dateText(day)} ${weekdays[day % 7]}"
41private fun cents(text: String) = text.substringBefore('.').toLong() * 100 +

103 if (type !in roomTypes) throw Fault("invalid type $type")

110 private fun reservation(id: String, status: String? = null): Reservation {
111 val value = reservations[id] ?: throw Fault("unknown reservation $id")
112 if (status != null && value.status != status)
113 throw Fault("invalid status $id ${value.status}")
114 return value

117 private fun stay(first: String, last: String): Pair<Int, Int> {
118 val arrival = dateNumber(first)
119 val departure = dateNumber(last)
120 if (arrival < today) throw Fault("past date $first")
121 if (departure <= arrival) throw Fault("invalid range $first $last")
122 if (departure - arrival > 30) throw Fault("too long ${departure - arrival}")
123 return arrival to departure
124 }
125
126 private fun period(
127 first: String, last: String, past: Boolean = false, limited: Boolean = true
128 ): Pair<Int, Int> {
129 val start = dateNumber(first)
130 val end = dateNumber(last)
131 if (past && start < today) throw Fault("past date $first")
132 if (end < start) throw Fault("invalid range $first $last")
133 if (limited && end - start + 1 > 31)
134 throw Fault("too long ${end - start + 1}")
135 return start to end
136 }

138 private fun nightRate(room: Room, day: Int): Long {
139 val percentage = seasons.asReversed().firstOrNull {
140 it.first <= day && day <= it.last
141 }?.percentage ?: 100
142 return room.rate * percentage / 100
143 }
144
145 private fun price(room: Room, arrival: Int, departure: Int) =
146 (arrival until departure).map {
147 Night(it, room.number, nightRate(room, it))
148 }.toMutableList()
149
150 private fun unavailable(
151 room: Room, arrival: Int, departure: Int, exceptFor: Reservation? = null
152 ): Int? {
153 for (day in arrival until departure) {
154 val slot = day to room.number
155 if (slot in blocks) return day
156 val occupant = occupied[slot]
157 if (occupant != null && occupant.reservation !== exceptFor) return day
158 }
159 return null
160 }

162 private fun freeRoom(
163 type: String, arrival: Int, departure: Int, exceptFor: Reservation? = null
164 ) = rooms.values.firstOrNull {
165 it.type == type && unavailable(it, arrival, departure, exceptFor) == null
166 }
167
168 private fun inhouse(room: Room) = reservations.values.firstOrNull {
169 it.room === room && it.status == "inhouse"
170 }
171
172 private fun occupy(reservation: Reservation, nights: List<Night>) {
173 for (night in nights) occupied[night.day to night.room] =
174 Occupied(reservation, night.rate)
175 }
176
177 private fun release(reservation: Reservation, from: Int? = null) {
178 for (night in reservation.nights)
179 if (from == null || night.day >= from)
180 occupied.remove(night.day to night.room)
181 }

183 private fun create(
184 id: String, guest: String, room: Room,
185 arrival: Int, departure: Int, indent: Int = 0
186 ) {
187 val r = Reservation(id, guest, room, arrival, departure, price(room, arrival, departure))
188 reservations[id] = r
189 used += id
190 occupy(r, r.nights)
191 emit(indent, "BOOK", id, guest, "room", room.number, room.type,
192 dateText(arrival), "->", dateText(departure), "nights",
193 departure - arrival, "total", money(r.lodging()))
194 }
195
196 private fun guestTotal(guest: String) =
197 reservations.values.filter { it.guest == guest }.sumOf { it.paid }
198
199 private fun tier(guest: String) = when {
200 guestTotal(guest) / 100 >= 2000 -> "GOLD"
201 guestTotal(guest) / 100 >= 500 -> "SILVER"
202 else -> "BRONZE"
203 }
204
205 private fun bill(r: Reservation): Bill {
206 val lodging = r.lodging()
207 val percentage = when (tier(r.guest)) {
208 "GOLD" -> 10
209 "SILVER" -> 5
210 else -> 0
211 }
212 val discount = lodging * percentage / 100
213 val extras = r.extras.sumOf { it.amount }
214 val subtotal = lodging - discount + r.penaltyRate + extras
215 val tax = subtotal / 10
216 return Bill(lodging, discount, r.penaltyRate, extras, tax, subtotal + tax)
217 }
218
219 private fun deposit(r: Reservation, amount: Long) = if (amount >= r.deposit) {
220 "deposit ${money(r.deposit)} due ${money(amount - r.deposit)}"
221 } else {
222 "deposit ${money(r.deposit)} refund ${money(r.deposit - amount)}"
223 }
224
225 private fun checkout(r: Reservation, day: Int, indent: Int = 0) {
226 if (day < r.departure) {
227 val penalty = r.nights.first { it.day == day }
228 r.penaltyDay = day
229 r.penaltyRate = penalty.rate
230 release(r, day)
231 r.nights.removeAll { it.day >= day }
232 r.departure = day
233 }
234 val result = bill(r)
235 r.bill = result
236 r.paid = result.total
237 r.status = "checkedout"
238 r.posting = day
239 r.room.clean = false
240 emit(indent, "CHECKOUT", r.id, "nights", r.nights.size,
241 "lodging", money(result.lodging), "discount", money(result.discount),
242 "penalty", money(result.penalty), "extras", money(result.extras),
243 "tax", money(result.tax), "total", money(result.total), deposit(r, result.total))
244 }

246 private fun processWaitlist() {
247 val remaining = mutableListOf<Waiting>()
248 for (entry in waitlist) {
249 if (entry.expiry < today) {
250 emit(2, "WAITLIST", entry.id, "expired")
251 } else {
252 val room = freeRoom(entry.type, entry.arrival, entry.departure)
253 if (room == null)
254 remaining += entry
255 else
256 create(entry.id, entry.guest, room, entry.arrival, entry.departure, 2)
257 }
258 }
259 waitlist.clear()
260 waitlist.addAll(remaining)
261 }

263 private fun book(t: List<String>, specific: Boolean, waiting: Boolean) {
264 checkId(t[1])
265 val fixedRoom = if (specific) room(t[3]) else null
266 val type = fixedRoom?.type ?: t[3]
267 if (!specific) checkType(type)
268 val (arrival, departure) = stay(t[4], t[5])
269 val selected = if (specific) {
270 val found = fixedRoom!!
271 val invalid = unavailable(found, arrival, departure)
272 if (invalid != null)
273 throw Fault("room unavailable ${t[3]} ${dateText(invalid)}")
274 found
275 } else freeRoom(type, arrival, departure)
276 if (selected == null && !waiting)
277 throw Fault("no availability $type ${t[4]} ${t[5]}")
278 if (selected == null) {
279 waitlist += Waiting(t[1], t[2], type, arrival, departure)
280 used += t[1]
281 emit(0, "WAITLIST", t[1], "queued position", waitlist.size)
282 } else create(t[1], t[2], selected, arrival, departure)
283 }

285 private fun changeBlock(t: List<String>, blocking: Boolean) {
286 val room = room(t[1])
287 val (first, last) = period(t[2], t[3], past = true)
288 if (blocking) {
289 for (day in first..last) {
290 if ((day to room.number) in occupied)
291 throw Fault("room unavailable ${t[1]} ${dateText(day)}")
292 }
293 }
294 var changed = 0
295 for (day in first..last) {
296 val slot = day to room.number
297 if (if (blocking) blocks.add(slot) else blocks.remove(slot)) changed++
298 }
299 emit(0, if (blocking) "BLOCK" else "UNBLOCK",
300 t[1], t[2], "->", t[3], if (blocking) "added" else "removed", changed)
301 }

303 private fun advance(t: List<String>) {
304 val day = dateNumber(t[1])
305 if (day < today) throw Fault("past date ${t[1]}")
306 emit(0, "TODAY", dated(day))
307 val departing = reservations.values
308 .filter { it.status == "inhouse" && it.departure < day }
309 .sortedWith(compareBy<Reservation> { it.departure }.thenBy { it.id })
310 for (r in departing) checkout(r, r.departure, 2)
311 val absent = reservations.values
312 .filter { it.status == "booked" && it.arrival < day }
313 .sortedWith(compareBy<Reservation> { it.arrival }.thenBy { it.id })
314 for (r in absent) {
315 release(r)
316 r.fee = r.firstRate()
317 r.paid = r.fee
318 r.status = "noshow"
319 r.posting = r.arrival
320 emit(2, "NOSHOW", r.id, r.guest, "fee", money(r.fee), deposit(r, r.fee))
321 }
322 today = day
323 }

325 private fun modify(t: List<String>) {
326 val r = reservation(t[1], "booked")
327 val (arrival, departure) = stay(t[2], t[3])
328 var room = r.room
329 var changed = "kept"
330 if (unavailable(room, arrival, departure, r) != null) {
331 room = freeRoom(room.type, arrival, departure, r)
332 ?: throw Fault("no availability ${r.room.type} ${t[2]} ${t[3]}")
333 changed = "reassigned"
334 }
335 val nights = price(room, arrival, departure)
336 release(r)
337 r.room = room
338 r.arrival = arrival
339 r.departure = departure
340 r.nights = nights
341 occupy(r, nights)
342 emit(0, "MODIFY", t[1], "room", room.number, changed, t[2], "->", t[3],
343 "nights", departure - arrival, "total", money(r.lodging()))
344 }

346 private fun cancel(t: List<String>) {
347 val index = waitlist.indexOfFirst { it.id == t[1] }
348 if (index >= 0) {
349 waitlist.removeAt(index)
350 emit(0, "CANCEL", t[1], "waitlist removed")
351 return
352 }
353 val r = reservation(t[1], "booked")
354 val days = r.arrival - today
355 val fee = when {
356 days >= 7 -> 0L
357 days >= 1 -> r.firstRate() / 2
358 else -> r.firstRate()
359 }
360 release(r)
361 r.fee = fee
362 r.paid = fee
363 r.status = "cancelled"
364 r.posting = today
365 emit(0, "CANCEL", t[1], "days", days, "fee", money(fee), deposit(r, fee))
366 }

368 private fun checkin(t: List<String>) {
369 val r = reservation(t[1], "booked")
370 if (r.arrival != today)
371 throw Fault("not arrival date ${t[1]} ${dateText(r.arrival)}")
372 val occupant = inhouse(r.room)
373 if (occupant != null)
374 throw Fault("room occupied ${r.room.number} ${occupant.id}")
375 if (!r.room.clean) throw Fault("room dirty ${r.room.number}")
376 r.status = "inhouse"
377 emit(0, "CHECKIN", t[1], r.guest, "room", r.room.number, "until", dateText(r.departure))
378 }

380 private fun move(t: List<String>) {
381 val r = reservation(t[1], "inhouse")
382 val room = room(t[2])
383 val previous = r.room
384 if (room === previous) throw Fault("same room ${t[2]}")
385 val remaining = r.nights.filter { it.day >= today }
386 if (remaining.isEmpty()) throw Fault("no remaining nights ${t[1]}")
387 val invalid = unavailable(room, remaining.first().day, r.departure, r)
388 if (invalid != null) throw Fault("room unavailable ${t[2]} ${dateText(invalid)}")
389 val occupant = inhouse(room)
390 if (occupant != null)
391 throw Fault("room occupied ${t[2]} ${occupant.id}")
392 if (!room.clean) throw Fault("room dirty ${t[2]}")
393 release(r, today)
394 r.nights = r.nights.map { n ->
395 if (n.day >= today) n.copy(room = room.number) else n
396 }.toMutableList()
397 occupy(r, r.nights.filter { it.day >= today })
398 previous.clean = false
399 r.room = room
400 emit(0, "MOVE", t[1], previous.number, "->", room.number, "nights", remaining.size)
401 }

403 private fun extend(t: List<String>) {
404 val r = reservation(t[1], "inhouse")
405 val departure = dateNumber(t[2])
406 if (departure <= r.departure)
407 throw Fault("invalid range ${dateText(r.departure)} ${t[2]}")
408 if (departure - r.arrival > 30) throw Fault("too long ${departure - r.arrival}")
409 val invalid = unavailable(r.room, r.departure, departure, r)
410 if (invalid != null)
411 throw Fault("room unavailable ${r.room.number} ${dateText(invalid)}")
412 val nights = price(r.room, r.departure, departure)
413 r.nights.addAll(nights)
414 occupy(r, nights)
415 r.departure = departure
416 emit(0, "EXTEND", t[1], "until", t[2], "nights", r.nights.size,
417 "total", money(r.lodging()))
418 }

439 private fun invoice(t: List<String>) {
440 val r = reservation(t[1])
441 emit(0, "INVOICE", t[1], r.guest, "room", r.room.number, r.room.type, r.status)
442 if (r.status == "cancelled" || r.status == "noshow") {
443 emit(2, if (r.status == "cancelled") "cancellation" else "noshow",
444 "fee", money(r.fee))
445 emit(2, "total", money(r.fee))
446 emit(2, deposit(r, r.fee))
447 return
448 }
449 for (night in r.nights)
450 emit(2, dated(night.day), "room", night.room, money(night.rate))
451 val bill = if (r.status == "checkedout") r.bill!! else bill(r)
452 emit(2, "lodging", money(bill.lodging))
453 emit(2, "discount", money(bill.discount))
454 if (r.penaltyDay != null)
455 emit(2, "penalty", dated(r.penaltyDay!!), money(bill.penalty))
456 for (extra in r.extras) emit(2, "extra", extra.item, money(extra.amount))
457 emit(2, "tax", money(bill.tax))
458 emit(2, "total", money(bill.total))
459 emit(2, deposit(r, bill.total))
460 }

462 private fun roomNumbers(type: String? = null): List<Int> =
463 rooms.values.filter { type == null || it.type == type }.map { it.number }

465 private fun availability(t: List<String>) {
466 checkType(t[1])
467 val (start, end) = period(t[2], t[3])
468 val numbers = roomNumbers(t[1])
469 emit(0, "AVAIL", t[1], t[2], "->", t[3])
470 for (day in start..end) {
471 val free = numbers.filter { n ->
472 (day to n) !in blocks && (day to n) !in occupied
473 }
474 emit(2, dated(day), "free", free.size, *free.map { it as Any }.toTypedArray())
475 }
476 }

478 private fun occupancyParts(numbers: List<Int>, day: Int): String {
479 val blocked = numbers.count { (day to it) in blocks }
480 val sold = numbers.count { (day to it) in occupied }
481 val available = numbers.size - blocked
482 val rate = if (available == 0) "n/a" else "${sold * 100 / available}%"
483 return "rooms ${numbers.size} blocked $blocked occupied $sold rate $rate"
484 }

486 private fun occupancy(t: List<String>) {
487 val day = dateNumber(t[1])
488 emit(0, "OCCUPANCY", dated(day), occupancyParts(roomNumbers(), day))
489 for (type in roomTypes) {
490 val numbers = roomNumbers(type)
491 if (numbers.isNotEmpty()) emit(2, type, occupancyParts(numbers, day))
492 }
493 }

495 private fun daily(day: Int): Daily {
496 var available = 0L
497 var sold = 0L
498 var revenue = 0L
499 for (room in rooms.values) {
500 val slot = day to room.number
501 if (slot !in blocks) available++
502 val night = occupied[slot]
503 if (night != null) {
504 sold++
505 revenue += night.rate
506 }
507 }
508 return Daily(available, sold, revenue)
509 }

511 private fun metrics(day: Daily): String {
512 val rate = if (day.available == 0L) "n/a" else "${day.occupied * 100 / day.available}%"
513 val adr = if (day.occupied == 0L) "n/a" else money(day.revenue / day.occupied)
514 val revpar = if (day.available == 0L) "n/a" else money(day.revenue / day.available)
515 return "sold ${day.occupied}/${day.available} rate $rate revenue ${money(day.revenue)} adr $adr revpar $revpar"
516 }

597 private fun revenue(t: List<String>) {
598 val (first, last) = period(t[1], t[2], limited = false)
599 val posted = reservations.values.filter { it.posting != null && it.posting!! in first..last }
600 val done = posted.filter { it.status == "checkedout" }
601 val cancelled = posted.filter { it.status == "cancelled" }
602 val absent = posted.filter { it.status == "noshow" }
603 val lodging = done.sumOf { it.bill!!.lodging }
604 val discount = done.sumOf { it.bill!!.discount }
605 val penalty = done.sumOf { it.bill!!.penalty }
606 val extras = done.sumOf { it.bill!!.extras }
607 val tax = done.sumOf { it.bill!!.tax }
608 val cancellations = cancelled.sumOf { it.fee }
609 val noshows = absent.sumOf { it.fee }
610 emit(0, "REVENUE", t[1], "->", t[2])
611 for ((label, amount) in listOf(
612 "lodging" to lodging, "discount" to discount, "penalty" to penalty,
613 "extras" to extras, "tax" to tax, "cancellation" to cancellations,
614 "noshow" to noshows
615 )) emit(2, label, money(amount))
616 emit(2, "total", money(lodging - discount + penalty + extras + tax + cancellations + noshows))
617 emit(2, "checkouts", done.size, "cancellations", cancelled.size, "noshows", absent.size)
618 }
619
620 private fun run(t: List<String>) {
621 when (t[0]) {
622 "ROOM" -> {
623 val number = t[1].toInt()
624 if (number in rooms) throw Fault("duplicate room ${t[1]}")
625 checkType(t[2])
626 rooms[number] = Room(number, t[2], cents(t[3]))
627 emit(0, "ROOM", t[1], t[2], t[3])
628 }
629 "RATE" -> {
630 val room = room(t[1])
631 val previous = room.rate
632 room.rate = cents(t[2])
633 emit(0, "RATE", t[1], money(previous), "->", t[2])
634 }
635 "SEASON" -> {
636 val (first, last) = period(t[1], t[2], limited = false)
637 seasons += Season(first, last, t[3].toInt())
638 emit(0, "SEASON", seasons.size, t[1], "->", t[2], "${t[3]}%")
639 }
640 "BLOCK" -> changeBlock(t, true)
641 "UNBLOCK" -> changeBlock(t, false)
642 "CLEAN" -> {
643 val room = room(t[1])
644 if (room.clean) throw Fault("already clean ${t[1]}")
645 room.clean = true
646 emit(0, "CLEAN", t[1])
647 }
648 "TODAY" -> advance(t)
649 "BOOK" -> book(t, false, false)
650 "BOOKROOM" -> book(t, true, false)
651 "WAITLIST" -> book(t, false, true)
652 "MODIFY" -> modify(t)
653 "CANCEL" -> cancel(t)
654 "CHECKIN" -> checkin(t)
655 "MOVE" -> move(t)
656 "EXTEND" -> extend(t)
657 "CHARGE" -> charge(t)
658 "PAY" -> pay(t)
659 "CHECKOUT" -> checkout(reservation(t[1], "inhouse"), today)
660 "INVOICE" -> invoice(t)
661 "AVAIL" -> availability(t)
662 "OCCUPANCY" -> occupancy(t)
663 "STATS" -> stats(t)
664 "ROOMS" -> showRooms()
665 "LIST" -> listReservations(t)
666 "QUEUE" -> showQueue()
667 "SCHEDULE" -> schedule(t)
668 "GUEST" -> guest(t)
669 "REVENUE" -> revenue(t)
670 }
671 }
672
673 fun process(line: String) {
674 try {
675 run(line.split(' '))
676 } catch (fault: Fault) {
677 emit(0, "ERROR", fault.message ?: "")
678 }
679 processWaitlist()
680 }
681}
