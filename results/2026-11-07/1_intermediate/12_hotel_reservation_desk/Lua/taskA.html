1local ROOM_TYPES = {"SINGLE", "DOUBLE", "SUITE"}
2local WEEKDAYS = {"SAT", "SUN", "MON", "TUE", "WED", "THU", "FRI"}

4local function fail(reason)
5 error({message = reason}, 0)
6end
7
8local function leap(year)
9 return year % 4 == 0 and (year % 100 ~= 0 or year % 400 == 0)
10end
11
12local function month_lengths(year)
13 return {31, leap(year) and 29 or 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31}
14end
15
16local function day_number(text)
17 local year, month, day = text:match("^(%d+)%-(%d+)%-(%d+)$")
18 year, month, day = tonumber(year), tonumber(month), tonumber(day)
19 if year < 2000 or year > 2099 or month < 1 or month > 12 then
20 fail("invalid date " .. text)
21 end
22 local lengths = month_lengths(year)
23 if day < 1 or day > lengths[month] then fail("invalid date " .. text) end
24 local number = day - 1
25 for previous = 2000, year - 1 do
26 number = number + (leap(previous) and 366 or 365)
27 end
28 for previous = 1, month - 1 do
29 number = number + lengths[previous]
30 end
31 return number
32end
33
34local function date_text(number)
35 local year = 2000
36 while true do
37 local days = leap(year) and 366 or 365
38 if number < days then break end
39 number, year = number - days, year + 1
40 end
41 local month = 1
42 for _, days in ipairs(month_lengths(year)) do
43 if number < days then break end
44 number, month = number - days, month + 1
45 end
46 return string.format("%04d-%02d-%02d", year, month, number + 1)
47end
48
49local function date_weekday(day)
50 return date_text(day) .. " " .. WEEKDAYS[day % 7 + 1]
51end

53local function cents(text)
54 local whole, fraction = text:match("^(%d+)%.(%d%d)$")
55 return tonumber(whole) * 100 + tonumber(fraction)
56end
57
58local function money(value)
59 return string.format("%d.%02d", value // 100, value % 100)

103local function reservation(id, status)
104 local item = reservations[id]
105 if not item then fail("unknown reservation " .. id) end
106 if status and item.status ~= status then
107 fail("invalid status " .. id .. " " .. item.status)
108 end
109 return item

112local function stay(first, last)
113 local arrival, departure = day_number(first), day_number(last)
114 if arrival < today then fail("past date " .. first) end
115 if departure <= arrival then fail("invalid range " .. first .. " " .. last) end
116 if departure - arrival > 30 then fail("too long " .. (departure - arrival)) end
117 return arrival, departure
118end

120local function period(first, last, past_restricted, limited)
121 local start, finish = day_number(first), day_number(last)
122 if past_restricted and start < today then fail("past date " .. first) end
123 if finish < start then fail("invalid range " .. first .. " " .. last) end
124 if limited ~= false and finish - start + 1 > 31 then
125 fail("too long " .. (finish - start + 1))
126 end
127 return start, finish
128end

130local function room_numbers()
131 local result = {}
132 for number in pairs(rooms) do result[#result + 1] = number end
133 table.sort(result)
134 return result
135end

137local function night_rate(room, day)
138 local percentage = 100
139 for i = #seasons, 1, -1 do
140 local entry = seasons[i]
141 if entry[1] <= day and day <= entry[2] then
142 percentage = entry[3]
143 break
144 end
145 end
146 return room.rate * percentage // 100
147end

149local function price(room, arrival, departure)
150 local nights = {}
151 for day = arrival, departure - 1 do
152 nights[#nights + 1] = {day, room.number, night_rate(room, day)}
153 end
154 return nights
155end

157local function night_key(day, number)
158 return day .. ":" .. number
159end

161local function first_unavailable(room, arrival, departure, except_for)
162 for day = arrival, departure - 1 do
163 local key = night_key(day, room.number)
164 if blocks[key] then return day end
165 local existing = occupied[key]
166 if existing and existing[1] ~= except_for then return day end
167 end
168 return nil
169end

171local function free_room(kind, arrival, departure, except_for)
172 for _, number in ipairs(room_numbers()) do
173 local room = rooms[number]
174 if room.kind == kind and
175 not first_unavailable(room, arrival, departure, except_for) then
176 return room
177 end
178 end
179 return nil
180end

182local function inhouse(room)
183 for _, id in ipairs(reservation_order) do
184 local item = reservations[id]
185 if item.status == "inhouse" and item.room == room then return item end
186 end
187 return nil

190local function occupy(item, nights)
191 for _, night in ipairs(nights) do
192 occupied[night_key(night[1], night[2])] = {item, night[3]}
193 end
194end

196local function release(item, first)
197 for _, night in ipairs(item.nights) do
198 if not first or night[1] >= first then
199 occupied[night_key(night[1], night[2])] = nil
200 end
201 end
202end

204local function lodging(item)
205 local amount = 0
206 for _, night in ipairs(item.nights) do amount = amount + night[3] end
207 return amount
208end

210local function first_rate(item)
211 return item.nights[1][3]
212end

214local function create_reservation(id, guest, room, arrival, departure, indent)
215 local nights = price(room, arrival, departure)
216 local item = {
217 id = id, guest = guest, room = room, arrival = arrival, departure = departure,
218 nights = nights, status = "booked", extras = {}, deposit = 0, paid = 0,
219 posting = nil, fee = 0, penalty = nil, bill = nil
220 }
221 reservations[id] = item
222 reservation_order[#reservation_order + 1] = id
223 used_ids[id] = true
224 occupy(item, nights)
225 local parts = {
226 "BOOK", id, guest, "room", room.number, room.kind,
227 date_text(arrival), "->", date_text(departure),
228 "nights", departure - arrival, "total", money(lodging(item))
229 }
230 if indent then emit_indented(table.unpack(parts)) else emit(table.unpack(parts)) end
231end

233local function guest_total(guest)
234 local amount = 0
235 for _, id in ipairs(reservation_order) do
236 local item = reservations[id]
237 if item.guest == guest then amount = amount + item.paid end
238 end
239 return amount
240end

242local function tier(guest)
243 local points = guest_total(guest) // 100
244 if points >= 2000 then return "GOLD" end
245 if points >= 500 then return "SILVER" end
246 return "BRONZE"
247end
248
249local function calculate_bill(item)
250 local charge = lodging(item)
251 local level = tier(item.guest)
252 local percentage = level == "GOLD" and 10 or level == "SILVER" and 5 or 0
253 local discount = charge * percentage // 100
254 local penalty = item.penalty and item.penalty[3] or 0
255 local extras = 0
256 for _, extra in ipairs(item.extras) do extras = extras + extra[2] end
257 local subtotal = charge - discount + penalty + extras
258 local tax = subtotal // 10
259 return {charge, discount, penalty, extras, tax, subtotal + tax}
260end

262local function deposit_parts(item, amount)
263 if amount >= item.deposit then
264 return {"deposit", money(item.deposit), "due", money(amount - item.deposit)}
265 end
266 return {"deposit", money(item.deposit), "refund", money(item.deposit - amount)}
267end

269local function check_out(item, day, indent)
270 if day < item.departure then
271 for _, night in ipairs(item.nights) do
272 if night[1] == day then
273 item.penalty = night
274 break
275 end
276 end
277 release(item, day)
278 local remaining = {}
279 for _, night in ipairs(item.nights) do
280 if night[1] < day then remaining[#remaining + 1] = night end
281 end
282 item.nights = remaining
283 item.departure = day
284 end
285 local bill = calculate_bill(item)
286 item.bill = bill
287 item.paid = bill[6]
288 item.status = "checkedout"
289 item.posting = day
290 item.room.clean = false
291 local parts = {
292 "CHECKOUT", item.id, "nights", #item.nights,
293 "lodging", money(bill[1]), "discount", money(bill[2]),
294 "penalty", money(bill[3]), "extras", money(bill[4]),
295 "tax", money(bill[5]), "total", money(bill[6])
296 }
297 for _, part in ipairs(deposit_parts(item, bill[6])) do
298 parts[#parts + 1] = part
299 end
300 if indent then emit_indented(table.unpack(parts)) else emit(table.unpack(parts)) end
301end

303local function process_waitlist()
304 local remaining = {}
305 for _, entry in ipairs(waitlist) do
306 if entry.expiry < today then
307 emit_indented("WAITLIST", entry.id, "expired")
308 else
309 local room = free_room(entry.kind, entry.arrival, entry.departure)
310 if not room then
311 remaining[#remaining + 1] = entry
312 else
313 create_reservation(entry.id, entry.guest, room, entry.arrival, entry.departure, true)
314 end
315 end
316 end
317 waitlist = remaining
318end

341local function cmd_block(a)
342 local room = require_room(a[1])
343 local start, finish = period(a[2], a[3], true)
344 for day = start, finish do
345 if occupied[night_key(day, room.number)] then
346 fail("room unavailable " .. a[1] .. " " .. date_text(day))
347 end
348 end

350 for day = start, finish do

352 if not blocks[key] then
353 blocks[key] = true
354 added = added + 1

357 emit("BLOCK", a[1], a[2], "->", a[3], "added", added)

381local function cmd_today(a)
382 local day = day_number(a[1])
383 if day < today then fail("past date " .. a[1]) end
384 emit("TODAY", date_weekday(day))
385 local departing, absent = {}, {}
386 for _, id in ipairs(reservation_order) do
387 local item = reservations[id]
388 if item.status == "inhouse" and item.departure < day then
389 departing[#departing + 1] = item
390 end
391 end
392 table.sort(departing, function(x, y)
393 if x.departure ~= y.departure then return x.departure < y.departure end
394 return x.id < y.id
395 end)
396 for _, item in ipairs(departing) do
397 check_out(item, item.departure, true)
398 end
399 for _, id in ipairs(reservation_order) do
400 local item = reservations[id]
401 if item.status == "booked" and item.arrival < day then
402 absent[#absent + 1] = item
403 end
404 end
405 table.sort(absent, function(x, y)
406 if x.arrival ~= y.arrival then return x.arrival < y.arrival end
407 return x.id < y.id
408 end)
409 for _, item in ipairs(absent) do
410 release(item)
411 item.fee = first_rate(item)
412 item.paid = item.fee
413 item.status = "noshow"
414 item.posting = item.arrival
415 local parts = {"NOSHOW", item.id, item.guest, "fee", money(item.fee)}
416 for _, part in ipairs(deposit_parts(item, item.fee)) do
417 parts[#parts + 1] = part
418 end
419 emit_indented(table.unpack(parts))
420 end
421 today = day
422end

460local function cmd_modify(a)
461 local item = reservation(a[1], "booked")
462 local arrival, departure = stay(a[2], a[3])
463 local room = item.room
464 local choice = "kept"
465 if first_unavailable(room, arrival, departure, item) then
466 room = free_room(room.kind, arrival, departure, item)
467 if not room then
468 fail("no availability " .. item.room.kind .. " " .. a[2] .. " " .. a[3])
469 end
470 choice = "reassigned"
471 end
472 local nights = price(room, arrival, departure)
473 release(item)
474 item.room = room
475 item.arrival = arrival
476 item.departure = departure
477 item.nights = nights
478 occupy(item, nights)
479 emit("MODIFY", a[1], "room", room.number, choice, a[2], "->", a[3],
480 "nights", departure - arrival, "total", money(lodging(item)))
481end

483local function cmd_cancel(a)
484 for i, entry in ipairs(waitlist) do
485 if entry.id == a[1] then
486 table.remove(waitlist, i)
487 emit("CANCEL", a[1], "waitlist removed")
488 return
489 end
490 end
491 local item = reservation(a[1], "booked")
492 local days = item.arrival - today
493 local first = first_rate(item)
494 local fee = days >= 7 and 0 or days >= 1 and first // 2 or first
495 release(item)
496 item.fee = fee
497 item.paid = fee
498 item.status = "cancelled"
499 item.posting = today
500 local parts = {"CANCEL", a[1], "days", days, "fee", money(fee)}
501 for _, part in ipairs(deposit_parts(item, fee)) do parts[#parts + 1] = part end
502 emit(table.unpack(parts))
503end

505local function cmd_checkin(a)
506 local item = reservation(a[1], "booked")
507 if item.arrival ~= today then
508 fail("not arrival date " .. a[1] .. " " .. date_text(item.arrival))
509 end
510 local occupant = inhouse(item.room)
511 if occupant then
512 fail("room occupied " .. item.room.number .. " " .. occupant.id)

514 if not item.room.clean then fail("room dirty " .. item.room.number) end
515 item.status = "inhouse"
516 emit("CHECKIN", a[1], item.guest, "room", item.room.number,
517 "until", date_text(item.departure))

520local function cmd_move(a)
521 local item = reservation(a[1], "inhouse")
522 local room = require_room(a[2])
523 local previous = item.room
524 if room == previous then fail("same room " .. a[2]) end
525 local remaining = {}
526 for _, night in ipairs(item.nights) do
527 if night[1] >= today then remaining[#remaining + 1] = night end
528 end
529 if #remaining == 0 then fail("no remaining nights " .. a[1]) end
530 local unavailable = first_unavailable(room, remaining[1][1], item.departure, item)
531 if unavailable then fail("room unavailable " .. a[2] .. " " .. date_text(unavailable)) end
532 local occupant = inhouse(room)
533 if occupant then fail("room occupied " .. a[2] .. " " .. occupant.id) end
534 if not room.clean then fail("room dirty " .. a[2]) end
535 release(item, today)
536 local future = {}
537 for _, night in ipairs(item.nights) do
538 if night[1] >= today then
539 night[2] = room.number
540 future[#future + 1] = night
541 end
542 end
543 occupy(item, future)
544 previous.clean = false
545 item.room = room
546 emit("MOVE", a[1], previous.number, "->", a[2], "nights", #remaining)
547end

594 local item = reservation(a[1])
595 local room = item.room
596 emit("INVOICE", a[1], item.guest, "room", room.number, room.kind, item.status)
597 if item.status == "cancelled" or item.status == "noshow" then

611 emit_indented("penalty", date_weekday(item.penalty[1]), money(bill[3]))
612 end
613 for _, extra in ipairs(item.extras) do
614 emit_indented("extra", extra[1], money(extra[2]))
615 end

621local function cmd_avail(a)
622 require_type(a[1])
623 local start, finish = period(a[2], a[3])
624 local numbers = {}
625 for _, number in ipairs(room_numbers()) do
626 if rooms[number].kind == a[1] then numbers[#numbers + 1] = number end
627 end
628 emit("AVAIL", a[1], a[2], "->", a[3])
629 for day = start, finish do
630 local free = {}
631 for _, number in ipairs(numbers) do
632 local key = night_key(day, number)
633 if not blocks[key] and not occupied[key] then free[#free + 1] = number end
634 end
635 local parts = {date_weekday(day), "free", #free}
636 for _, number in ipairs(free) do parts[#parts + 1] = number end
637 emit_indented(table.unpack(parts))
638 end
639end

641local function occupancy_parts(numbers, day)
642 local blocked, used = 0, 0
643 for _, number in ipairs(numbers) do
644 local key = night_key(day, number)
645 if blocks[key] then blocked = blocked + 1 end
646 if occupied[key] then used = used + 1 end
647 end
648 local available = #numbers - blocked
649 local rate = available > 0 and used * 100 // available .. "%" or "n/a"
650 return {"rooms", #numbers, "blocked", blocked, "occupied", used, "rate", rate}
651end

653local function cmd_occupancy(a)
654 local day = day_number(a[1])
655 local parts = {"OCCUPANCY", date_weekday(day)}
656 for _, value in ipairs(occupancy_parts(room_numbers(), day)) do
657 parts[#parts + 1] = value
658 end
659 emit(table.unpack(parts))
660 for _, kind in ipairs(ROOM_TYPES) do
661 local numbers = {}
662 for number, room in pairs(rooms) do
663 if room.kind == kind then numbers[#numbers + 1] = number end
664 end
665 if #numbers > 0 then
666 local row = {kind}
667 for _, value in ipairs(occupancy_parts(numbers, day)) do
668 row[#row + 1] = value
669 end
670 emit_indented(table.unpack(row))
671 end
672 end

675local function daily_values(day)
676 local available = 0
677 local used = 0
678 local revenue = 0
679 for number in pairs(rooms) do
680 local key = night_key(day, number)
681 if not blocks[key] then available = available + 1 end
682 if occupied[key] then
683 used = used + 1
684 revenue = revenue + occupied[key][2]
685 end
686 end
687 return available, used, revenue
688end

690local function metrics_parts(available, used, revenue)
691 local rate = available > 0 and used * 100 // available .. "%" or "n/a"
692 local average = used > 0 and money(revenue // used) or "n/a"
693 local revpar = available > 0 and money(revenue // available) or "n/a"
694 return {"sold", used .. "/" .. available, "rate", rate, "revenue", money(revenue),
695 "adr", average, "revpar", revpar}
696end

720local function cmd_rooms(a)
721 local numbers = room_numbers()
722 emit("ROOMS", date_weekday(today), "rooms", #numbers)
723 for _, number in ipairs(numbers) do
724 local room = rooms[number]
725 local parts = {
726 number, room.kind, "base", money(room.rate), "tonight", money(night_rate(room, today)),
727 room.clean and "clean" or "dirty"
728 }
729 local occupant = inhouse(room)
730 if occupant then
731 parts[#parts + 1] = "occupied"
732 parts[#parts + 1] = occupant.id
733 elseif blocks[night_key(today, number)] then
734 parts[#parts + 1] = "blocked"
735 else
736 parts[#parts + 1] = "vacant"
737 end
738 local upcoming
739 for _, id in ipairs(reservation_order) do
740 local item = reservations[id]
741 if item.status == "booked" and item.room == room and
742 (not upcoming or item.arrival < upcoming.arrival) then
743 upcoming = item
744 end
745 end
746 if upcoming then
747 parts[#parts + 1] = "next"
748 parts[#parts + 1] = upcoming.id
749 parts[#parts + 1] = date_text(upcoming.arrival)
750 end
751 emit_indented(table.unpack(parts))
752 end
753end

780local function cmd_schedule(a)
781 local day = day_number(a[1])
782 local arrivals, departures = {}, {}
783 for _, id in ipairs(reservation_order) do
784 local item = reservations[id]
785 if item.arrival == day and item.status ~= "cancelled" then
786 arrivals[#arrivals + 1] = item
787 end
788 if item.departure == day and
789 (item.status == "booked" or item.status == "inhouse" or
790 item.status == "checkedout") then
791 departures[#departures + 1] = item
792 end
793 end
794 table.sort(arrivals, function(x, y) return x.id < y.id end)
795 table.sort(departures, function(x, y) return x.id < y.id end)
796 emit("SCHEDULE", date_weekday(day), "arrivals", #arrivals, "departures", #departures)
797 for _, item in ipairs(arrivals) do
798 emit_indented("arrival", item.id, item.guest, "room", item.room.number, item.status)
799 end
800 for _, item in ipairs(departures) do
801 emit_indented("departure", item.id, item.guest, "room", item.room.number, item.status)
802 end
803end

816 date_text(item.arrival), "->", date_text(item.departure), item.status,
817 deposit and "deposit" or "paid", money(deposit and item.deposit or item.paid))

823local function cmd_revenue(a)
824 local start, finish = period(a[1], a[2], false, false)
825 local lodging, discount, penalty, extras, tax = 0, 0, 0, 0, 0
826 local cancellations, noshows, checkouts, cancelled, absent = 0, 0, 0, 0, 0
827 for _, id in ipairs(reservation_order) do
828 local item = reservations[id]
829 if item.posting and start <= item.posting and item.posting <= finish then
830 if item.status == "checkedout" then
831 checkouts = checkouts + 1
832 lodging = lodging + item.bill[1]
833 discount = discount + item.bill[2]
834 penalty = penalty + item.bill[3]
835 extras = extras + item.bill[4]
836 tax = tax + item.bill[5]
837 elseif item.status == "cancelled" then
838 cancelled = cancelled + 1
839 cancellations = cancellations + item.fee
840 elseif item.status == "noshow" then
841 absent = absent + 1
842 noshows = noshows + item.fee
843 end
844 end
845 end
846 emit("REVENUE", a[1], "->", a[2])
847 emit_indented("lodging", money(lodging))
848 emit_indented("discount", money(discount))
849 emit_indented("penalty", money(penalty))
850 emit_indented("extras", money(extras))
851 emit_indented("tax", money(tax))
852 emit_indented("cancellation", money(cancellations))
853 emit_indented("noshow", money(noshows))
854 emit_indented("total", money(lodging - discount + penalty + extras + tax + cancellations + noshows))
855 emit_indented("checkouts", checkouts, "cancellations", cancelled, "noshows", absent)
856end

871local function execute(line)
872 local fields = {}
873 for token in line:gmatch("[^ ]+") do fields[#fields + 1] = token end
874 local command = fields[1]
875 local arguments = {}
876 for i = 2, #fields do arguments[#arguments + 1] = fields[i] end
877 local ok, reason = pcall(function()
878 handlers[command](arguments)
879 end)
880 if not ok then
881 if type(reason) == "table" and reason.message then
882 emit("ERROR", reason.message)
883 else
884 error(reason)
885 end
886 end
887 process_waitlist()
888end
