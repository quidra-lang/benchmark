12month_lengths <- function(year) {
13 c(31, if (leap(year)) 29 else 28, 31, 30, 31, 30,
14 31, 31, 30, 31, 30, 31)

17day_number <- function(text) {
18 year <- as.numeric(substr(text, 1, 4))
19 month <- as.numeric(substr(text, 6, 7))
20 day <- as.numeric(substr(text, 9, 10))
21 if (year < 2000 || year > 2099 || month < 1 || month > 12)
22 fail(paste("invalid date", text))
23 lengths <- month_lengths(year)
24 if (day < 1 || day > lengths[month])
25 fail(paste("invalid date", text))
26 count <- 0
27 if (year > 2000) {
28 for (previous in 2000:(year - 1))
29 count <- count + (if (leap(previous)) 366 else 365)
30 }
31 count + sum(head(lengths, month - 1)) + day - 1
32}

34date_text <- function(number) {
35 year <- 2000
36 repeat {
37 length <- if (leap(year)) 366 else 365
38 if (number < length) break
39 number <- number - length
40 year <- year + 1
41 }
42 month <- 1
43 for (length in month_lengths(year)) {
44 if (number < length) break
45 number <- number - length
46 month <- month + 1
47 }
48 sprintf("%04d-%02d-%02d", year, month, number + 1)
49}

110stay <- function(first, last) {
111 arrival <- day_number(first)
112 departure <- day_number(last)
113 if (arrival < state$today) fail(paste("past date", first))
114 if (departure <= arrival) fail(paste("invalid range", first, last))
115 if (departure - arrival > 30) fail(paste("too long", departure - arrival))
116 c(arrival, departure)

119period <- function(first, last, past_restricted = FALSE, limited = TRUE) {
120 start <- day_number(first)
121 end <- day_number(last)
122 if (past_restricted && start < state$today) fail(paste("past date", first))
123 if (end < start) fail(paste("invalid range", first, last))
124 if (limited && end - start + 1 > 31) fail(paste("too long", end - start + 1))
125 c(start, end)

130night_rate <- function(room, day) {
131 percentage <- 100
132 if (length(state$seasons)) {
133 for (i in rev(seq_along(state$seasons))) {
134 entry <- state$seasons[[i]]
135 if (entry[1] <= day && day <= entry[2]) {
136 percentage <- entry[3]
137 break
138 }
139 }
140 }
141 floor(room$rate * percentage / 100)

153first_unavailable <- function(room, arrival, departure, except_for = NULL) {
154 if (arrival >= departure) return(NULL)
155 for (day in arrival:(departure - 1)) {
156 key <- night_key(day, room$number)
157 if (stored(state$blocks, key)) return(day)
158 if (stored(state$occupied, key)) {
159 entry <- lookup(state$occupied, key)
160 if (is.null(except_for) || entry$reservation != except_for$id) return(day)
161 }
162 }
163 NULL
164}

205create_reservation <- function(id, guest, room, arrival, departure, indent = 0) {
206 nights <- price(room, arrival, departure)
207 entry <- new_object(
208 id = id, guest = guest, room = room, arrival = arrival, departure = departure,
209 nights = nights, status = "booked", extras = list(), deposit = 0, paid = 0,

211 )

213 state$reservation_order <- c(state$reservation_order, id)

215 occupy(entry, nights)
216 emit("BOOK", id, guest, "room", room$number, room$kind,
217 date_text(arrival), "->", date_text(departure),
218 "nights", departure - arrival, "total", money(lodging(entry)),

236calculate_bill <- function(entry) {
237 charge <- lodging(entry)
238 percentage <- switch(tier(entry$guest), GOLD = 10, SILVER = 5, BRONZE = 0)
239 discount <- floor(charge * percentage / 100)
240 penalty <- if (is.null(entry$penalty)) 0 else entry$penalty[2]
241 extras <- sum(vapply(entry$extras, function(value) value[[2]], numeric(1)))
242 subtotal <- charge - discount + penalty + extras
243 tax <- floor(subtotal / 10)
244 c(charge, discount, penalty, extras, tax, subtotal + tax)
245}

253check_out <- function(entry, day, indent = 0) {
254 if (day < entry$departure) {
255 penalty <- NULL
256 for (night in entry$nights) {
257 if (night[1] == day) {
258 penalty <- c(day, night[3])
259 break
260 }
261 }
262 entry$penalty <- penalty
263 release(entry, day)
264 entry$nights <- Filter(function(night) night[1] < day, entry$nights)
265 entry$departure <- day
266 }
267 bill <- calculate_bill(entry)
268 entry$bill <- bill
269 entry$paid <- bill[6]
270 entry$status <- "checkedout"
271 entry$posting <- day
272 entry$room$clean <- FALSE
273 emit("CHECKOUT", entry$id, "nights", length(entry$nights),
274 "lodging", money(bill[1]), "discount", money(bill[2]),
275 "penalty", money(bill[3]), "extras", money(bill[4]),
276 "tax", money(bill[5]), "total", money(bill[6]),
277 deposit_parts(entry, bill[6]), indent = indent)
278}

359cmd_today <- function(a) {
360 day <- day_number(a[1])
361 if (day < state$today) fail(paste("past date", a[1]))
362 emit("TODAY", date_weekday(day))
363 departing <- state$reservation_order[vapply(state$reservation_order, function(id) {
364 entry <- reservation(id)
365 entry$status == "inhouse" && entry$departure < day
366 }, logical(1))]
367 if (length(departing)) {
368 departures <- vapply(departing, function(id) reservation(id)$departure, numeric(1))
369 departing <- departing[order(departures, departing)]
370 }
371 for (id in departing) {
372 entry <- reservation(id)
373 check_out(entry, entry$departure, indent = 2)
374 }
375 absent <- state$reservation_order[vapply(state$reservation_order, function(id) {
376 entry <- reservation(id)
377 entry$status == "booked" && entry$arrival < day
378 }, logical(1))]
379 if (length(absent)) {
380 arrivals <- vapply(absent, function(id) reservation(id)$arrival, numeric(1))
381 absent <- absent[order(arrivals, absent)]
382 }
383 for (id in absent) {
384 entry <- reservation(id)
385 release(entry)
386 entry$fee <- first_rate(entry)
387 entry$paid <- entry$fee
388 entry$status <- "noshow"
389 entry$posting <- entry$arrival
390 emit("NOSHOW", entry$id, entry$guest, "fee", money(entry$fee),
391 deposit_parts(entry, entry$fee), indent = 2)
392 }
393 state$today <- day
394}

502 if (night[1] >= state$today) night[2] <- room$number

505 future <- Filter(function(night) night[1] >= state$today, entry$nights)
506 occupy(entry, future)
507 previous$clean <- FALSE
508 entry$room <- room
509 emit("MOVE", a[1], previous$number, "->", a[2], "nights", length(remaining))

591occupancy_parts <- function(numbers, day) {
592 blocked <- sum(vapply(numbers, function(number)
593 stored(state$blocks, night_key(day, number)), logical(1)))
594 occupied <- sum(vapply(numbers, function(number)
595 stored(state$occupied, night_key(day, number)), logical(1)))
596 available <- length(numbers) - blocked
597 rate <- if (available > 0) paste0(floor(occupied * 100 / available), "%") else "n/a"
598 c("rooms", length(numbers), "blocked", blocked, "occupied", occupied, "rate", rate)
599}

628metrics_parts <- function(available, occupied, revenue) {
629 rate <- if (available > 0) paste0(floor(occupied * 100 / available), "%") else "n/a"
630 average <- if (occupied > 0) money(floor(revenue / occupied)) else "n/a"
631 revpar <- if (available > 0) money(floor(revenue / available)) else "n/a"
632 c("sold", paste0(occupied, "/", available), "rate", rate,
633 "revenue", money(revenue), "adr", average, "revpar", revpar)
634}

648cmd_rooms <- function(a) {
649 emit("ROOMS", date_weekday(state$today), "rooms", length(room_numbers()))
650 for (number in room_numbers()) {
651 room <- require_room(as.character(number))
652 parts <- c(
653 number, room$kind, "base", money(room$rate),
654 "tonight", money(night_rate(room, state$today)),
655 if (room$clean) "clean" else "dirty"
656 )
657 occupant <- inhouse(room)
658 if (!is.null(occupant)) {
659 parts <- c(parts, "occupied", occupant$id)
660 } else if (stored(state$blocks, night_key(state$today, number))) {
661 parts <- c(parts, "blocked")
662 } else {
663 parts <- c(parts, "vacant")
664 }
665 upcoming <- NULL
666 for (id in state$reservation_order) {
667 entry <- reservation(id)
668 if (entry$status == "booked" && identical(entry$room, room) &&
669 (is.null(upcoming) || entry$arrival < upcoming$arrival))
670 upcoming <- entry
671 }
672 if (!is.null(upcoming))
673 parts <- c(parts, "next", upcoming$id, date_text(upcoming$arrival))
674 emit(parts, indent = 2)
675 }
676}

702cmd_schedule <- function(a) {
703 day <- day_number(a[1])
704 arrivals <- state$reservation_order[vapply(state$reservation_order, function(id) {
705 entry <- reservation(id)
706 entry$arrival == day && entry$status != "cancelled"
707 }, logical(1))]
708 departures <- state$reservation_order[vapply(state$reservation_order, function(id) {
709 entry <- reservation(id)
710 entry$departure == day && entry$status %in% c("booked", "inhouse", "checkedout")

712 emit("SCHEDULE", date_weekday(day), "arrivals", length(arrivals),
713 "departures", length(departures))
714 for (id in sort(arrivals, method = "radix")) {
715 entry <- reservation(id)
716 emit("arrival", id, entry$guest, "room", entry$room$number, entry$status, indent = 2)
717 }
718 for (id in sort(departures, method = "radix")) {
719 entry <- reservation(id)
720 emit("departure", id, entry$guest, "room", entry$room$number, entry$status, indent = 2)
721 }

742cmd_revenue <- function(a) {
743 dates <- period(a[1], a[2], limited = FALSE)
744 totals <- c(0, 0, 0, 0, 0)
745 cancellations <- 0
746 noshows <- 0
747 checkouts <- 0
748 cancelled <- 0
749 absent <- 0
750 for (id in state$reservation_order) {
751 entry <- reservation(id)
752 if (is.null(entry$posting) ||
753 entry$posting < dates[1] || entry$posting > dates[2]) next
754 if (entry$status == "checkedout") {
755 totals <- totals + entry$bill[1:5]
756 checkouts <- checkouts + 1
757 } else if (entry$status == "cancelled") {
758 cancellations <- cancellations + entry$fee
759 cancelled <- cancelled + 1
760 } else if (entry$status == "noshow") {
761 noshows <- noshows + entry$fee
762 absent <- absent + 1
763 }
764 }
765 emit("REVENUE", a[1], "->", a[2])
766 for (i in seq_along(c("lodging", "discount", "penalty", "extras", "tax")))
767 emit(c("lodging", "discount", "penalty", "extras", "tax")[i],
768 money(totals[i]), indent = 2)
769 emit("cancellation", money(cancellations), indent = 2)
770 emit("noshow", money(noshows), indent = 2)
771 total <- totals[1] - totals[2] + sum(totals[3:5]) + cancellations + noshows
772 emit("total", money(total), indent = 2)
773 emit("checkouts", checkouts, "cancellations", cancelled, "noshows", absent, indent = 2)
774}

789execute <- function(line) {
790 words <- strsplit(line, " ", fixed = TRUE)[[1]]
791 command <- words[1]
792 arguments <- words[-1]
793 tryCatch(
794 handlers[[command]](arguments),
795 error = function(e) {
796 if (!inherits(e, "CommandError")) stop(e)
797 emit("ERROR", conditionMessage(e))
798 }
799 )
800 process_waitlist()
801}
