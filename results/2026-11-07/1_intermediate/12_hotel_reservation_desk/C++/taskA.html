23bool leap(int year) {
24 return year % 4 == 0 && (year % 100 != 0 || year % 400 == 0);
25}
26
27int days_in_month(int year, int month) {
28 return month == 2 && leap(year) ? 29 : normal_months[month - 1];
29}
30
31int year_length(int year) {
32 return leap(year) ? 366 : 365;
33}
34
35int date_number(const string& text) {
36 int year = stoi(text.substr(0, 4));
37 int month = stoi(text.substr(5, 2));
38 int day = stoi(text.substr(8, 2));
39 if (year < 2000 || year > 2099 || month < 1 || month > 12 ||
40 day < 1 || day > days_in_month(year, month))
41 throw runtime_error("invalid date " + text);
42
43 int number = 0;
44 for (int y = 2000; y < year; ++y) number += year_length(y);
45 for (int m = 1; m < month; ++m) number += days_in_month(year, m);
46 return number + day - 1;
47}
48
49string date_text(int number) {
50 int year = 2000;
51 while (number >= year_length(year)) {
52 number -= year_length(year);
53 ++year;
54 }
55 int month = 1;
56 while (number >= days_in_month(year, month)) {
57 number -= days_in_month(year, month);
58 ++month;
59 }
60 ostringstream out;
61 out << setw(4) << setfill('0') << year << '-'
62 << setw(2) << month << '-' << setw(2) << (number + 1);
63 return out.str();
64}
65
66string date_weekday(int day) {
67 return date_text(day) + " " + weekdays[day % 7];
68}
69
70Money cents(const string& text) {
71 size_t point = text.find('.');
72 return stoll(text.substr(0, point)) * 100 + stoll(text.substr(point + 1));
73}
74
75string money(Money cents_amount) {
76 ostringstream out;
77 out << cents_amount / 100 << '.' << setw(2) << setfill('0') << cents_amount % 100;
78 return out.str();
79}

161 template<typename... Args>
162 void emit(bool indented, const Args&... args) {
163 ostringstream out;
164 out << (indented ? " " : "");
165 bool first = true;
166 ((out << (first ? "" : " ") << args, first = false), ...);
167 output.push_back(out.str());
168 }

170 template<typename... Args>
171 void say(const Args&... args) { emit(false, args...); }

173 template<typename... Args>
174 void say_indented(const Args&... args) { emit(true, args...); }

176 static void require_type(const string& kind) {
177 if (find(room_types.begin(), room_types.end(), kind) == room_types.end())
178 throw runtime_error("invalid type " + kind);

181 Room& require_room(const string& text) {
182 auto it = rooms.find(stoi(text));
183 if (it == rooms.end()) throw runtime_error("unknown room " + text);
184 return it->second;
185 }
186
187 void require_unused(const string& id) {
188 if (used_ids.count(id)) throw runtime_error("duplicate reservation " + id);
189 }
190
191 Reservation& require_reservation(const string& id, const string& status = "") {
192 auto it = reservations.find(id);
193 if (it == reservations.end()) throw runtime_error("unknown reservation " + id);
194 if (!status.empty() && it->second.status != status)
195 throw runtime_error("invalid status " + id + " " + it->second.status);
196 return it->second;
197 }

199 pair<int, int> stay(const string& first, const string& last) {
200 int arrival = date_number(first);
201 int departure = date_number(last);
202 if (arrival < today) throw runtime_error("past date " + first);
203 if (departure <= arrival)
204 throw runtime_error("invalid range " + first + " " + last);
205 if (departure - arrival > 30)
206 throw runtime_error("too long " + to_string(departure - arrival));
207 return {arrival, departure};
208 }
209
210 pair<int, int> period(const string& first, const string& last,
211 bool past_restricted = false, bool limited = true) {
212 int start = date_number(first);
213 int end = date_number(last);
214 if (past_restricted && start < today)
215 throw runtime_error("past date " + first);
216 if (end < start) throw runtime_error("invalid range " + first + " " + last);
217 if (limited && end - start + 1 > 31)
218 throw runtime_error("too long " + to_string(end - start + 1));
219 return {start, end};
220 }
221
222 Money night_rate(const Room& room, int day) const {
223 for (auto it = seasons.rbegin(); it != seasons.rend(); ++it)
224 if (it->first <= day && day <= it->last)
225 return room.rate * it->percentage / 100;
226 return room.rate;
227 }
228
229 vector<Night> price(const Room& room, int first, int last) const {
230 vector<Night> nights;
231 for (int day = first; day < last; ++day)
232 nights.push_back({day, room.number, night_rate(room, day)});
233 return nights;
234 }
235
236 optional<int> first_unavailable(const Room& room, int first, int last,
237 const Reservation* except_for = nullptr) const {
238 for (int day = first; day < last; ++day) {
239 pair<int, int> key{day, room.number};
240 if (blocked.count(key)) return day;
241 auto holder = occupied.find(key);
242 if (holder != occupied.end() && holder->second.first != except_for)
243 return day;
244 }
245 return nullopt;
246 }
247
248 Room* free_room(const string& kind, int first, int last,
249 const Reservation* except_for = nullptr) {
250 for (auto& [number, room] : rooms)
251 if (room.kind == kind &&
252 !first_unavailable(room, first, last, except_for))
253 return &room;
254 return nullptr;
255 }
256
257 Reservation* inhouse(const Room& room) {
258 for (const string& id : reservation_order) {
259 Reservation& reservation = reservations.at(id);
260 if (reservation.status == "inhouse" && reservation.room == &room)
261 return &reservation;
262 }
263 return nullptr;
264 }
265
266 void occupy(Reservation& reservation, const vector<Night>& nights) {
267 for (const Night& night : nights)
268 occupied[{night.day, night.room}] = {&reservation, night.rate};
269 }
270
271 void release(const Reservation& reservation, optional<int> first = nullopt) {
272 for (const Night& night : reservation.nights)
273 if (!first || night.day >= *first)
274 occupied.erase({night.day, night.room});
275 }

277 void create_reservation(const string& id, const string& guest,
278 Room& room, int arrival, int departure, bool indent = false) {
279 auto nights = price(room, arrival, departure);
280 auto inserted = reservations.emplace(
281 piecewise_construct,
282 forward_as_tuple(id),
283 forward_as_tuple(id, guest, &room, arrival, departure, move(nights))
284 );
285 Reservation& reservation = inserted.first->second;
286 reservation_order.push_back(id);
287 used_ids.insert(id);
288 occupy(reservation, reservation.nights);
289 emit(indent, "BOOK", id, guest, "room", room.number, room.kind,
290 date_text(arrival), "->", date_text(departure), "nights", departure - arrival,
291 "total", money(reservation.lodging()));
292 }
293
294 Money guest_total(const string& guest) const {
295 Money paid = 0;
296 for (const auto& [id, reservation] : reservations)
297 if (reservation.guest == guest) paid += reservation.paid;
298 return paid;
299 }
300
301 string tier(const string& guest) const {
302 Money points = guest_total(guest) / 100;
303 return points >= 2000 ? "GOLD" : points >= 500 ? "SILVER" : "BRONZE";
304 }
305
306 Bill calculate_bill(const Reservation& reservation) const {
307 Money lodging = reservation.lodging();
308 string category = tier(reservation.guest);
309 int percentage = category == "GOLD" ? 10 : category == "SILVER" ? 5 : 0;
310 Money discount = lodging * percentage / 100;
311 Money penalty = reservation.penalty ? reservation.penalty->second : 0;
312 Money extras = 0;
313 for (const auto& entry : reservation.extras) extras += entry.second;
314 Money subtotal = lodging - discount + penalty + extras;
315 Money tax = subtotal / 10;
316 return {lodging, discount, penalty, extras, tax, subtotal + tax};
317 }
318
319 string deposit_parts(const Reservation& reservation, Money total) const {
320 if (total >= reservation.deposit)
321 return "deposit " + money(reservation.deposit) + " due " +
322 money(total - reservation.deposit);
323 return "deposit " + money(reservation.deposit) + " refund " +
324 money(reservation.deposit - total);
325 }
326
327 void check_out(Reservation& reservation, int day, bool indent = false) {
328 if (day < reservation.departure) {
329 auto night = find_if(reservation.nights.begin(), reservation.nights.end(),
330 [day](const Night& item) { return item.day == day; });
331 Money penalty = night->rate;
332 reservation.penalty = make_pair(day, penalty);
333 release(reservation, day);
334 reservation.nights.erase(
335 remove_if(reservation.nights.begin(), reservation.nights.end(),
336 [day](const Night& item) { return item.day >= day; }),
337 reservation.nights.end());
338 reservation.departure = day;
339 }
340 Bill bill = calculate_bill(reservation);
341 reservation.bill = bill;
342 reservation.paid = bill.total;
343 reservation.status = "checkedout";
344 reservation.posting = day;
345 reservation.room->clean = false;
346 emit(indent, "CHECKOUT", reservation.id, "nights", reservation.nights.size(),
347 "lodging", money(bill.lodging), "discount", money(bill.discount),
348 "penalty", money(bill.penalty), "extras", money(bill.extras),
349 "tax", money(bill.tax), "total", money(bill.total),
350 deposit_parts(reservation, bill.total));
351 }

353 void process_waitlist() {
354 vector<WaitingEntry> remaining;
355 for (const WaitingEntry& entry : waitlist) {
356 if (entry.expiry < today) {
357 say_indented("WAITLIST", entry.id, "expired");
358 continue;
359 }
360 Room* room = free_room(entry.kind, entry.arrival, entry.departure);
361 if (!room) remaining.push_back(entry);
362 else create_reservation(entry.id, entry.guest, *room,
363 entry.arrival, entry.departure, true);
364 }
365 waitlist = move(remaining);
366 }

389 void cmd_block(const vector<string>& a) {
390 Room& room = require_room(a[0]);
391 auto [first, last] = period(a[1], a[2], true);
392 for (int day = first; day <= last; ++day)
393 if (occupied.count({day, room.number}))
394 throw runtime_error("room unavailable " + a[0] + " " + date_text(day));
395 int added = 0;
396 for (int day = first; day <= last; ++day)
397 added += blocked.insert({day, room.number}).second;
398 say("BLOCK", a[0], a[1], "->", a[2], "added", added);
399 }

401 void cmd_unblock(const vector<string>& a) {
402 Room& room = require_room(a[0]);
403 auto [first, last] = period(a[1], a[2], true);
404 int removed = 0;
405 for (int day = first; day <= last; ++day)
406 removed += static_cast<int>(blocked.erase({day, room.number}));
407 say("UNBLOCK", a[0], a[1], "->", a[2], "removed", removed);
408 }

413 room.clean = true;

417 void cmd_today(const vector<string>& a) {
418 int day = date_number(a[0]);
419 if (day < today) throw runtime_error("past date " + a[0]);
420 say("TODAY", date_weekday(day));
421 vector<Reservation*> departing, absent;
422 for (auto& [id, reservation] : reservations)
423 if (reservation.status == "inhouse" && reservation.departure < day)
424 departing.push_back(&reservation);
425 sort(departing.begin(), departing.end(), [](const auto* x, const auto* y) {
426 return tie(x->departure, x->id) < tie(y->departure, y->id);
427 });
428 for (Reservation* reservation : departing)
429 check_out(*reservation, reservation->departure, true);
430
431 for (auto& [id, reservation] : reservations)
432 if (reservation.status == "booked" && reservation.arrival < day)
433 absent.push_back(&reservation);
434 sort(absent.begin(), absent.end(), [](const auto* x, const auto* y) {
435 return tie(x->arrival, x->id) < tie(y->arrival, y->id);
436 });
437 for (Reservation* reservation : absent) {
438 release(*reservation);
439 reservation->fee = reservation->first_rate();
440 reservation->paid = reservation->fee;
441 reservation->status = "noshow";
442 reservation->posting = reservation->arrival;
443 say_indented("NOSHOW", reservation->id, reservation->guest, "fee",
444 money(reservation->fee), deposit_parts(*reservation, reservation->fee));
445 }
446 today = day;
447 }

471 require_type(a[2]);

482 void cmd_modify(const vector<string>& a) {
483 Reservation& reservation = require_reservation(a[0], "booked");
484 auto [arrival, departure] = stay(a[1], a[2]);
485 Room* room = reservation.room;
486 string choice = "kept";
487 if (first_unavailable(*room, arrival, departure, &reservation)) {
488 Room* available = free_room(room->kind, arrival, departure, &reservation);
489 if (!available)
490 throw runtime_error("no availability " + room->kind + " " + a[1] + " " + a[2]);
491 room = available;
492 choice = "reassigned";
493 }
494 vector<Night> nights = price(*room, arrival, departure);
495 release(reservation);
496 reservation.room = room;
497 reservation.arrival = arrival;
498 reservation.departure = departure;
499 reservation.nights = move(nights);
500 occupy(reservation, reservation.nights);
501 say("MODIFY", a[0], "room", room->number, choice, a[1], "->", a[2],
502 "nights", departure - arrival, "total", money(reservation.lodging()));
503 }

505 void cmd_cancel(const vector<string>& a) {
506 auto waiting = find_if(waitlist.begin(), waitlist.end(),
507 [&](const WaitingEntry& entry) { return entry.id == a[0]; });
508 if (waiting != waitlist.end()) {
509 waitlist.erase(waiting);
510 say("CANCEL", a[0], "waitlist removed");
511 return;
512 }
513 Reservation& reservation = require_reservation(a[0], "booked");
514 int days = reservation.arrival - today;
515 Money first = reservation.first_rate();
516 Money fee = days >= 7 ? 0 : days >= 1 ? first / 2 : first;
517 release(reservation);
518 reservation.fee = fee;
519 reservation.paid = fee;
520 reservation.status = "cancelled";
521 reservation.posting = today;
522 say("CANCEL", a[0], "days", days, "fee", money(fee),
523 deposit_parts(reservation, fee));
524 }

526 void cmd_checkin(const vector<string>& a) {
527 Reservation& reservation = require_reservation(a[0], "booked");
528 if (reservation.arrival != today)
529 throw runtime_error("not arrival date " + a[0] + " " + date_text(reservation.arrival));
530 Reservation* occupant = inhouse(*reservation.room);
531 if (occupant)
532 throw runtime_error("room occupied " + to_string(reservation.room->number) +
533 " " + occupant->id);
534 if (!reservation.room->clean)
535 throw runtime_error("room dirty " + to_string(reservation.room->number));
536 reservation.status = "inhouse";
537 say("CHECKIN", a[0], reservation.guest, "room", reservation.room->number,
538 "until", date_text(reservation.departure));
539 }

541 void cmd_move(const vector<string>& a) {
542 Reservation& reservation = require_reservation(a[0], "inhouse");
543 Room& room = require_room(a[1]);
544 Room* previous = reservation.room;
545 if (&room == previous) throw runtime_error("same room " + a[1]);
546 vector<Night> remaining;
547 for (const Night& night : reservation.nights)
548 if (night.day >= today) remaining.push_back(night);
549 if (remaining.empty()) throw runtime_error("no remaining nights " + a[0]);
550 auto unavailable = first_unavailable(room, remaining.front().day,
551 reservation.departure, &reservation);
552 if (unavailable)
553 throw runtime_error("room unavailable " + a[1] + " " + date_text(*unavailable));
554 Reservation* occupant = inhouse(room);
555 if (occupant)
556 throw runtime_error("room occupied " + a[1] + " " + occupant->id);
557 if (!room.clean) throw runtime_error("room dirty " + a[1]);
558 release(reservation, today);
559 for (Night& night : reservation.nights)
560 if (night.day >= today) night.room = room.number;
561 vector<Night> changed;
562 for (const Night& night : reservation.nights)
563 if (night.day >= today) changed.push_back(night);
564 occupy(reservation, changed);
565 previous->clean = false;
566 reservation.room = &room;
567 say("MOVE", a[0], previous->number, "->", room.number, "nights", remaining.size());
568 }

570 void cmd_extend(const vector<string>& a) {
571 Reservation& reservation = require_reservation(a[0], "inhouse");
572 int departure = date_number(a[1]);
573 if (departure <= reservation.departure)
574 throw runtime_error("invalid range " + date_text(reservation.departure) + " " + a[1]);
575 if (departure - reservation.arrival > 30)
576 throw runtime_error("too long " + to_string(departure - reservation.arrival));
577 auto unavailable = first_unavailable(*reservation.room,
578 reservation.departure, departure, &reservation);
579 if (unavailable)
580 throw runtime_error("room unavailable " + to_string(reservation.room->number) +
581 " " + date_text(*unavailable));
582 vector<Night> nights = price(*reservation.room, reservation.departure, departure);
583 reservation.nights.insert(reservation.nights.end(), nights.begin(), nights.end());
584 occupy(reservation, nights);
585 reservation.departure = departure;
586 say("EXTEND", a[0], "until", a[1], "nights", reservation.nights.size(),
587 "total", money(reservation.lodging()));
588 }

614 void cmd_invoice(const vector<string>& a) {
615 Reservation& reservation = require_reservation(a[0]);
616 say("INVOICE", reservation.id, reservation.guest, "room",
617 reservation.room->number, reservation.room->kind, reservation.status);
618 if (reservation.status == "cancelled" || reservation.status == "noshow") {
619 say_indented(reservation.status == "cancelled" ? "cancellation" : "noshow",
620 "fee", money(reservation.fee));
621 say_indented("total", money(reservation.fee));
622 say_indented(deposit_parts(reservation, reservation.fee));
623 return;
624 }
625 for (const Night& night : reservation.nights)
626 say_indented(date_weekday(night.day), "room", night.room, money(night.rate));
627 Bill bill = reservation.bill ? *reservation.bill : calculate_bill(reservation);
628 say_indented("lodging", money(bill.lodging));
629 say_indented("discount", money(bill.discount));
630 if (reservation.penalty)
631 say_indented("penalty", date_weekday(reservation.penalty->first),
632 money(bill.penalty));
633 for (const auto& charge : reservation.extras)
634 say_indented("extra", charge.first, money(charge.second));
635 say_indented("tax", money(bill.tax));
636 say_indented("total", money(bill.total));
637 say_indented(deposit_parts(reservation, bill.total));
638 }

640 void cmd_avail(const vector<string>& a) {
641 require_type(a[0]);
642 auto [first, last] = period(a[1], a[2]);
643 say("AVAIL", a[0], a[1], "->", a[2]);
644 for (int day = first; day <= last; ++day) {
645 ostringstream line;
646 vector<int> free;
647 for (const auto& [number, room] : rooms) {
648 if (room.kind != a[0] ||
649 blocked.count({day, number}) || occupied.count({day, number}))
650 continue;
651 free.push_back(number);
652 }
653 line << date_weekday(day) << " free " << free.size();
654 for (int number : free) line << ' ' << number;
655 say_indented(line.str());
656 }

659 string occupancy_parts(const vector<const Room*>& selected, int day) const {
660 int blocked_count = 0, occupied_count = 0;
661 for (const Room* room : selected) {
662 blocked_count += blocked.count({day, room->number}) != 0;
663 occupied_count += occupied.count({day, room->number}) != 0;
664 }
665 int available = static_cast<int>(selected.size()) - blocked_count;
666 string rate = available ? to_string(occupied_count * 100 / available) + "%" : "n/a";
667 return "rooms " + to_string(selected.size()) + " blocked " +
668 to_string(blocked_count) + " occupied " + to_string(occupied_count) +
669 " rate " + rate;
670 }

686 tuple<int, int, Money> daily_values(int day) const {
687 int available = static_cast<int>(rooms.size()), occupied_count = 0;
688 Money revenue = 0;
689 for (const auto& [number, room] : rooms) {
690 available -= blocked.count({day, number}) != 0;
691 auto holder = occupied.find({day, number});
692 if (holder != occupied.end()) {
693 ++occupied_count;
694 revenue += holder->second.second;
695 }
696 }
697 return {available, occupied_count, revenue};
698 }
699
700 static string metrics_parts(int available, int occupied_count, Money revenue) {
701 string rate = available ?
702 to_string(occupied_count * 100 / available) + "%" : "n/a";
703 string adr = occupied_count ? money(revenue / occupied_count) : "n/a";
704 string revpar = available ? money(revenue / available) : "n/a";
705 return "sold " + to_string(occupied_count) + "/" + to_string(available) +
706 " rate " + rate + " revenue " + money(revenue) +
707 " adr " + adr + " revpar " + revpar;
708 }

711 auto [first, last] = period(a[0], a[1]);
712 int total_available = 0, total_occupied = 0;
713 Money total_revenue = 0;
714 say("STATS", a[0], "->", a[1]);

716 auto [available, taken, revenue] = daily_values(day);
717 total_available += available;
718 total_occupied += taken;
719 total_revenue += revenue;
720 say_indented(date_weekday(day), metrics_parts(available, taken, revenue));
721 }
722 say_indented("total", metrics_parts(total_available, total_occupied, total_revenue));
723 }

725 void cmd_rooms() {
726 say("ROOMS", date_weekday(today), "rooms", rooms.size());
727 for (const auto& [number, room] : rooms) {
728 ostringstream line;
729 line << number << ' ' << room.kind << " base " << money(room.rate)
730 << " tonight " << money(night_rate(room, today))
731 << (room.clean ? " clean" : " dirty");
732 Reservation* occupant = inhouse(room);
733 if (occupant) line << " occupied " << occupant->id;
734 else if (blocked.count({today, number})) line << " blocked";
735 else line << " vacant";
736 const Reservation* upcoming = nullptr;
737 for (const auto& [id, reservation] : reservations) {
738 if (reservation.status != "booked" || reservation.room != &room) continue;
739 if (!upcoming || reservation.arrival < upcoming->arrival ||
740 (reservation.arrival == upcoming->arrival && reservation.id < upcoming->id))
741 upcoming = &reservation;
742 }
743 if (upcoming) line << " next " << upcoming->id
744 << ' ' << date_text(upcoming->arrival);
745 say_indented(line.str());
746 }

808 void cmd_revenue(const vector<string>& a) {
809 auto [first, last] = period(a[0], a[1], false, false);
810 Money lodging = 0, discount = 0, penalty = 0, extras = 0, tax = 0;
811 Money cancellations = 0, noshows = 0;
812 int checkouts = 0, cancel_count = 0, no_show_count = 0;
813 for (const auto& [id, r] : reservations) {
814 if (!r.posting || *r.posting < first || *r.posting > last) continue;
815 if (r.status == "checkedout") {
816 ++checkouts;
817 const Bill& bill = *r.bill;
818 lodging += bill.lodging;
819 discount += bill.discount;
820 penalty += bill.penalty;
821 extras += bill.extras;
822 tax += bill.tax;
823 } else if (r.status == "cancelled") {
824 cancellations += r.fee;
825 ++cancel_count;
826 } else if (r.status == "noshow") {
827 noshows += r.fee;
828 ++no_show_count;
829 }
830 }
831 say("REVENUE", a[0], "->", a[1]);
832 say_indented("lodging", money(lodging));
833 say_indented("discount", money(discount));
834 say_indented("penalty", money(penalty));
835 say_indented("extras", money(extras));
836 say_indented("tax", money(tax));
837 say_indented("cancellation", money(cancellations));
838 say_indented("noshow", money(noshows));
839 say_indented("total", money(lodging - discount + penalty + extras + tax +
840 cancellations + noshows));
841 say_indented("checkouts", checkouts, "cancellations", cancel_count,
842 "noshows", no_show_count);
843 }
