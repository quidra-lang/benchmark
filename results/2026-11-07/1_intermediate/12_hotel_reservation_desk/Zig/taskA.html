4const types = [_][]const u8{ "SINGLE", "DOUBLE", "SUITE" };
5const weekdays = [_][]const u8{ "SAT", "SUN", "MON", "TUE", "WED", "THU", "FRI" };
6fn integer(s: []const u8) i64 {
7 return std.fmt.parseInt(i64, s, 10) catch unreachable;
8}
9fn amount(s: []const u8) i64 {
10 const dot = std.mem.findScalar(u8, s, '.').?;
11 return integer(s[0..dot]) * 100 + integer(s[dot + 1 ..]);
12}
13fn leap(y: i64) bool {
14 return @mod(y, 4) == 0 and (@mod(y, 100) != 0 or @mod(y, 400) == 0);
15}
16fn monthDays(y: i64, m: i64) i64 {
17 const days = [_]i64{ 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31 };
18 return if (m == 2 and leap(y)) 29 else days[@intCast(m - 1)];
19}
20fn dayNumber(y: i64, m: i64, d: i64) i64 {
21 var total = d - 1;
22 var year: i64 = 2000;
23 while (year < y) : (year += 1) total += if (leap(year)) @as(i64, 366) else 365;
24 var month: i64 = 1;
25 while (month < m) : (month += 1) total += monthDays(y, month);
26 return total;
27}

89fn free(room: *Room, day: i64, except: ?*Reservation) bool {
90 const occupant = room.occupied.get(day);
91 return !room.blocks.contains(day) and (occupant == null or occupant == except);

93const Hotel = struct {
94 gpa: Allocator,
95 out: *std.Io.Writer,
96 error_message: []const u8 = "",
97 today: i64 = dayNumber(2026, 1, 1),
98 rooms: std.ArrayList(*Room) = .empty,
99 reservations: std.StringArrayHashMapUnmanaged(*Reservation) = .empty,
100 seasons: std.ArrayList(Season) = .empty,
101 queue: std.ArrayList(Waiting) = .empty,
102 used: std.StringHashMapUnmanaged(void) = .empty,
103 fn fail(self: *Hotel, comptime fmt: []const u8, args: anytype) Error {
104 self.error_message = std.fmt.allocPrint(self.gpa, fmt, args) catch return error.OutOfMemory;
105 return error.BadCommand;
106 }
107 fn money(self: *Hotel, c: i64) Allocator.Error![]const u8 {
108 const u: u64 = @intCast(c);
109 return std.fmt.allocPrint(self.gpa, "{d}.{d:0>2}", .{ u / 100, u % 100 });
110 }
111 fn date(self: *Hotel, s: []const u8) Error!i64 {
112 const y = integer(s[0..4]);
113 const m = integer(s[5..7]);
114 const d = integer(s[8..10]);
115 if (y < 2000 or y > 2099 or m < 1 or m > 12 or d < 1 or d > monthDays(y, m)) return self.fail("invalid date {s}", .{s});
116 return dayNumber(y, m, d);
117 }
118 fn dateText(self: *Hotel, day: i64) Allocator.Error![]const u8 {
119 var n = day;
120 var y: i64 = 2000;
121 var m: i64 = 1;
122 while (n >= (if (leap(y)) @as(i64, 366) else 365)) {
123 n -= if (leap(y)) @as(i64, 366) else 365;
124 y += 1;
125 }
126 while (n >= monthDays(y, m)) {
127 n -= monthDays(y, m);
128 m += 1;
129 }
130 return std.fmt.allocPrint(self.gpa, "{d:0>4}-{d:0>2}-{d:0>2}", .{ @as(u64, @intCast(y)), @as(u64, @intCast(m)), @as(u64, @intCast(n + 1)) });
131 }
132 fn dated(self: *Hotel, day: i64) Allocator.Error![]const u8 {
133 return std.fmt.allocPrint(self.gpa, "{s} {s}", .{ try self.dateText(day), weekdays[@intCast(@mod(day, 7))] });
134 }

147 fn typeCheck(self: *Hotel, kind: []const u8) Error!void {
148 for (types) |t| {
149 if (std.mem.eql(u8, kind, t)) return;

151 return self.fail("invalid type {s}", .{kind});

153 fn idCheck(self: *Hotel, id: []const u8) Error!void {
154 if (self.used.contains(id)) return self.fail("duplicate reservation {s}", .{id});
155 }
156 fn statusCheck(self: *Hotel, r: *Reservation, status: Status) Error!void {
157 if (r.status != status) return self.fail("invalid status {s} {s}", .{ r.id, @tagName(r.status) });
158 }
159 fn range(self: *Hotel, a: []const u8, b: []const u8, stay: bool, past: bool, limit: bool) Error!Span {
160 const first = try self.date(a);
161 const last = try self.date(b);
162 if ((stay or past) and first < self.today) return self.fail("past date {s}", .{a});
163 if (if (stay) last <= first else last < first) return self.fail("invalid range {s} {s}", .{ a, b });
164 const count = last - first + (if (stay) @as(i64, 0) else 1);
165 if (limit and count > (if (stay) @as(i64, 30) else 31)) return self.fail("too long {d}", .{count});
166 return .{ .first = first, .last = last };
167 }
168 fn sorted(self: *Hotel, rs: []const *Reservation, departure: bool, ids: bool) Error![]*Reservation {
169 const result = try self.gpa.dupe(*Reservation, rs);
170 const Order = struct {
171 departure: bool,
172 ids: bool,
173 fn less(order: @This(), left: *Reservation, right: *Reservation) bool {
174 const ad = if (order.departure) left.last else left.first;
175 const bd = if (order.departure) right.last else right.first;
176 if (!order.ids and ad != bd) return ad < bd;
177 return std.mem.order(u8, left.id, right.id) == .lt;
178 }
179 };
180 std.mem.sort(*Reservation, result, Order{ .departure = departure, .ids = ids }, Order.less);
181 return result;
182 }
183 fn rate(self: *Hotel, r: *Room, day: i64) i64 {
184 var p: i64 = 100;
185 for (self.seasons.items) |s| {
186 if (s.first <= day and day <= s.last) p = s.percent;
187 }
188 return @divTrunc(r.rate * p, 100);
189 }
190 fn price(self: *Hotel, r: *Room, first: i64, last: i64) Error!std.ArrayList(Night) {
191 var result: std.ArrayList(Night) = .empty;
192 var d = first;
193 while (d < last) : (d += 1) try result.append(self.gpa, .{ .day = d, .room = r, .rate = self.rate(r, d) });
194 return result;
195 }
196 fn unavailable(r: *Room, first: i64, last: i64, except: ?*Reservation) ?i64 {
197 var d = first;
198 while (d < last) : (d += 1) {
199 if (!free(r, d, except)) return d;
200 }
201 return null;
202 }
203 fn search(self: *Hotel, kind: []const u8, first: i64, last: i64, except: ?*Reservation) ?*Room {
204 for (self.rooms.items) |r| {
205 if (std.mem.eql(u8, r.kind, kind) and unavailable(r, first, last, except) == null) return r;
206 }
207 return null;
208 }
209 fn inhouse(self: *Hotel, r: *Room) ?*Reservation {
210 for (self.reservations.values()) |guest| {
211 if (guest.status == .inhouse and guest.room == r) return guest;
212 }
213 return null;
214 }
215 fn release(r: *Reservation) void {
216 for (r.nights.items) |n| {
217 if (n.room.occupied.get(n.day) == r) _ = n.room.occupied.remove(n.day);
218 }
219 }
220 fn occupy(self: *Hotel, r: *Reservation) Error!void {
221 for (r.nights.items) |n| try n.room.occupied.put(self.gpa, n.day, r);
222 }
223 fn create(self: *Hotel, id: []const u8, guest: []const u8, room_ptr: *Room, first: i64, last: i64, indent: []const u8) Error!void {
224 const r = try self.gpa.create(Reservation);
225 r.* = .{ .id = id, .guest = guest, .room = room_ptr, .first = first, .last = last, .nights = try self.price(room_ptr, first, last) };
226 try self.reservations.put(self.gpa, id, r);
227 try self.used.put(self.gpa, id, {});
228 try self.occupy(r);
229 try self.out.print("{s}BOOK {s} {s} room {d} {s} {s} -> {s} nights {d} total {s}\n", .{ indent, id, guest, room_ptr.number, room_ptr.kind, try self.dateText(first), try self.dateText(last), last - first, try self.money(r.lodging()) });
230 }
231 fn paid(self: *Hotel, guest: []const u8) i64 {
232 var total: i64 = 0;
233 for (self.reservations.values()) |r| {
234 if (std.mem.eql(u8, r.guest, guest)) total += r.paid;
235 }
236 return total;
237 }
238 fn tier(self: *Hotel, guest: []const u8) []const u8 {
239 const points = @divTrunc(self.paid(guest), 100);
240 return if (points >= 2000) "GOLD" else if (points >= 500) "SILVER" else "BRONZE";
241 }
242 fn bill(self: *Hotel, r: *Reservation) [6]i64 {
243 const level = self.tier(r.guest);
244 const lodging = r.lodging();
245 const discount = @divTrunc(lodging * @as(i64, if (std.mem.eql(u8, level, "GOLD")) 10 else if (std.mem.eql(u8, level, "SILVER")) 5 else 0), 100);
246 const penalty = if (r.penalty) |n| n.rate else 0;
247 const extras = r.extraTotal();
248 const subtotal = lodging - discount + penalty + extras;
249 const tax = @divTrunc(subtotal, 10);
250 return .{ lodging, discount, penalty, extras, tax, subtotal + tax };
251 }
252 fn deposit(self: *Hotel, r: *Reservation, total: i64) Allocator.Error![]const u8 {
253 return std.fmt.allocPrint(self.gpa, "deposit {s} {s} {s}", .{ try self.money(r.deposit), if (total >= r.deposit) "due" else "refund", try self.money(if (total >= r.deposit) total - r.deposit else r.deposit - total) });
254 }
255 fn checkout(self: *Hotel, r: *Reservation, day: i64, indent: []const u8) Error!void {
256 if (day < r.last) {
257 for (r.nights.items) |n| {
258 if (n.day == day) {
259 r.penalty = n;
260 break;
261 }
262 }
263 release(r);
264 var count: usize = 0;
265 for (r.nights.items) |n| {
266 if (n.day < day) {
267 r.nights.items[count] = n;
268 count += 1;
269 }
270 }
271 r.nights.shrinkRetainingCapacity(count);
272 r.last = day;
273 try self.occupy(r);
274 }
275 const b = self.bill(r);
276 r.final_bill = b;
277 r.paid = b[5];
278 r.status = .checkedout;
279 r.posting = day;
280 r.room.clean = false;
281 try self.out.print("{s}CHECKOUT {s} nights {d} lodging {s} discount {s} penalty {s} extras {s} tax {s} total {s} {s}\n", .{ indent, r.id, r.nights.items.len, try self.money(b[0]), try self.money(b[1]), try self.money(b[2]), try self.money(b[3]), try self.money(b[4]), try self.money(b[5]), try self.deposit(r, b[5]) });
282 }
283 fn processQueue(self: *Hotel) Error!void {
284 var i: usize = 0;
285 while (i < self.queue.items.len) {
286 const w = self.queue.items[i];
287 if (w.expiry < self.today) {
288 _ = self.queue.orderedRemove(i);
289 try self.out.print(" WAITLIST {s} expired\n", .{w.id});
290 } else if (self.search(w.kind, w.first, w.last, null)) |r| {
291 _ = self.queue.orderedRemove(i);
292 try self.create(w.id, w.guest, r, w.first, w.last, " ");
293 } else i += 1;
294 }
295 }
296 fn metrics(self: *Hotel, day: i64) [3]i64 {
297 var result: [3]i64 = @splat(0);
298 for (self.rooms.items) |r| {
299 if (!r.blocks.contains(day)) result[0] += 1;
300 if (r.occupied.get(day)) |guest| {
301 result[1] += 1;
302 for (guest.nights.items) |n| {
303 if (n.day == day and n.room == r) {
304 result[2] += n.rate;
305 break;
306 }
307 }
308 }
309 }
310 return result;
311 }
312 fn metricText(self: *Hotel, s: i64, k: i64, v: i64) Allocator.Error![]const u8 {
313 return std.fmt.allocPrint(self.gpa, "sold {d}/{d} rate {s} revenue {s} adr {s} revpar {s}", .{ k, s, if (s == 0) "n/a" else try std.fmt.allocPrint(self.gpa, "{d}%", .{@divTrunc(k * 100, s)}), try self.money(v), if (k == 0) "n/a" else try self.money(@divTrunc(v, k)), if (s == 0) "n/a" else try self.money(@divTrunc(v, s)) });
314 }
315 fn occupancy(self: *Hotel, kind: ?[]const u8, day: i64) Allocator.Error![]const u8 {
316 var count: i64 = 0;
317 var blocked: i64 = 0;
318 var occupied: i64 = 0;
319 for (self.rooms.items) |r| {
320 if (kind != null and !std.mem.eql(u8, r.kind, kind.?)) continue;
321 count += 1;
322 if (r.blocks.contains(day)) blocked += 1;
323 if (r.occupied.contains(day)) occupied += 1;
324 }
325 const available = count - blocked;
326 return std.fmt.allocPrint(self.gpa, "rooms {d} blocked {d} occupied {d} rate {s}", .{ count, blocked, occupied, if (available == 0) "n/a" else try std.fmt.allocPrint(self.gpa, "{d}%", .{@divTrunc(occupied * 100, available)}) });
327 }
328 fn execute(self: *Hotel, a: []const []const u8) Error!void {
329 const op = a[0];
330 const id: []const u8 = if (a.len > 1) a[1] else "";
331 if (std.mem.eql(u8, op, "ROOM")) {
332 const number = integer(id);
333 if (self.findRoom(number) != null) return self.fail("duplicate room {s}", .{id});
334 try self.typeCheck(a[2]);
335 const r = try self.gpa.create(Room);
336 r.* = .{ .number = number, .kind = a[2], .rate = amount(a[3]) };
337 try self.rooms.append(self.gpa, r);
338 std.mem.sort(*Room, self.rooms.items, {}, struct {
339 fn less(_: void, left: *Room, right: *Room) bool {
340 return left.number < right.number;
341 }
342 }.less);
343 try self.out.print("ROOM {d} {s} {s}\n", .{ number, r.kind, try self.money(r.rate) });
344 } else if (std.mem.eql(u8, op, "RATE")) {
345 const r = try self.room(id);
346 const old = r.rate;
347 r.rate = amount(a[2]);
348 try self.out.print("RATE {s} {s} -> {s}\n", .{ id, try self.money(old), try self.money(r.rate) });
349 } else if (std.mem.eql(u8, op, "SEASON")) {
350 const span = try self.range(id, a[2], false, false, false);
351 try self.seasons.append(self.gpa, .{ .first = span.first, .last = span.last, .percent = integer(a[3]) });
352 try self.out.print("SEASON {d} {s} -> {s} {s}%\n", .{ self.seasons.items.len, id, a[2], a[3] });
353 } else if (std.mem.eql(u8, op, "BLOCK") or std.mem.eql(u8, op, "UNBLOCK")) {
354 const r = try self.room(id);
355 const span = try self.range(a[2], a[3], false, true, true);
356 var d = span.first;
357 if (std.mem.eql(u8, op, "BLOCK")) {
358 while (d <= span.last) : (d += 1) {
359 if (r.occupied.contains(d)) return self.fail("room unavailable {s} {s}", .{ id, try self.dateText(d) });
360 }
361 }
362 var changed: usize = 0;
363 d = span.first;
364 while (d <= span.last) : (d += 1) {
365 if (std.mem.eql(u8, op, "BLOCK")) {
366 if (!r.blocks.contains(d)) {
367 changed += 1;
368 try r.blocks.put(self.gpa, d, {});
369 }
370 } else if (r.blocks.remove(d)) changed += 1;
371 }
372 try self.out.print("{s} {s} {s} -> {s} {s} {d}\n", .{ op, id, a[2], a[3], if (std.mem.eql(u8, op, "BLOCK")) "added" else "removed", changed });
373 } else if (std.mem.eql(u8, op, "CLEAN")) {
374 const r = try self.room(id);
375 if (r.clean) return self.fail("already clean {s}", .{id});
376 r.clean = true;
377 try self.out.print("CLEAN {s}\n", .{id});
378 } else if (std.mem.eql(u8, op, "TODAY")) {
379 const day = try self.date(id);
380 if (day < self.today) return self.fail("past date {s}", .{id});
381 try self.out.print("TODAY {s}\n", .{try self.dated(day)});
382 var departures: std.ArrayList(*Reservation) = .empty;
383 var arrivals: std.ArrayList(*Reservation) = .empty;
384 for (self.reservations.values()) |r| {
385 if (r.status == .inhouse and r.last < day) try departures.append(self.gpa, r);
386 if (r.status == .booked and r.first < day) try arrivals.append(self.gpa, r);
387 }
388 for (try self.sorted(departures.items, true, false)) |r| try self.checkout(r, r.last, " ");
389 for (try self.sorted(arrivals.items, false, false)) |r| {
390 r.fee = r.nights.items[0].rate;
391 r.paid = r.fee;
392 r.status = .noshow;
393 r.posting = r.first;
394 release(r);
395 try self.out.print(" NOSHOW {s} {s} fee {s} {s}\n", .{ r.id, r.guest, try self.money(r.fee), try self.deposit(r, r.fee) });
396 }
397 self.today = day;
398 } else if (std.mem.eql(u8, op, "BOOK") or std.mem.eql(u8, op, "BOOKROOM") or std.mem.eql(u8, op, "WAITLIST")) {
399 try self.idCheck(id);
400 var selected: ?*Room = null;
401 if (std.mem.eql(u8, op, "BOOKROOM")) {
402 selected = try self.room(a[3]);
403 } else try self.typeCheck(a[3]);
404 const span = try self.range(a[4], a[5], true, false, true);
405 if (std.mem.eql(u8, op, "BOOKROOM")) {
406 if (unavailable(selected.?, span.first, span.last, null)) |d| return self.fail("room unavailable {d} {s}", .{ selected.?.number, try self.dateText(d) });
407 } else selected = self.search(a[3], span.first, span.last, null);
408 if (selected) |r| {
409 try self.create(id, a[2], r, span.first, span.last, "");
410 } else if (std.mem.eql(u8, op, "WAITLIST")) {
411 try self.queue.append(self.gpa, .{ .id = id, .guest = a[2], .kind = a[3], .first = span.first, .last = span.last, .expiry = span.first });
412 try self.used.put(self.gpa, id, {});
413 try self.out.print("WAITLIST {s} queued position {d}\n", .{ id, self.queue.items.len });
414 } else return self.fail("no availability {s} {s} {s}", .{ a[3], a[4], a[5] });
415 } else if (std.mem.eql(u8, op, "CANCEL")) {
416 for (self.queue.items, 0..) |w, i| {
417 if (std.mem.eql(u8, w.id, id)) {
418 _ = self.queue.orderedRemove(i);
419 try self.out.print("CANCEL {s} waitlist removed\n", .{id});
420 return;
421 }
422 }
423 const r = try self.reservation(id);
424 try self.statusCheck(r, .booked);
425 const days = r.first - self.today;
426 r.fee = if (days >= 7) 0 else if (days >= 1) @divTrunc(r.nights.items[0].rate, 2) else r.nights.items[0].rate;
427 r.paid = r.fee;
428 r.status = .cancelled;
429 r.posting = self.today;
430 release(r);
431 try self.out.print("CANCEL {s} days {d} fee {s} {s}\n", .{ id, days, try self.money(r.fee), try self.deposit(r, r.fee) });
432 } else if (std.mem.eql(u8, op, "MODIFY") or std.mem.eql(u8, op, "CHECKIN") or std.mem.eql(u8, op, "MOVE") or std.mem.eql(u8, op, "EXTEND") or std.mem.eql(u8, op, "CHARGE") or std.mem.eql(u8, op, "PAY") or std.mem.eql(u8, op, "CHECKOUT") or std.mem.eql(u8, op, "INVOICE")) {
433 const r = try self.reservation(id);
434 if (std.mem.eql(u8, op, "MODIFY")) {
435 try self.statusCheck(r, .booked);
436 const span = try self.range(a[2], a[3], true, false, true);
437 const old = r.room;
438 const selected = (if (unavailable(r.room, span.first, span.last, r) == null) r.room else self.search(r.room.kind, span.first, span.last, r)) orelse return self.fail("no availability {s} {s} {s}", .{ r.room.kind, a[2], a[3] });
439 release(r);
440 r.room = selected;
441 r.first = span.first;
442 r.last = span.last;
443 r.nights = try self.price(selected, span.first, span.last);
444 try self.occupy(r);
445 try self.out.print("MODIFY {s} room {d} {s} {s} -> {s} nights {d} total {s}\n", .{ id, selected.number, if (selected == old) "kept" else "reassigned", a[2], a[3], span.last - span.first, try self.money(r.lodging()) });
446 } else if (std.mem.eql(u8, op, "CHECKIN")) {
447 try self.statusCheck(r, .booked);
448 if (r.first != self.today) return self.fail("not arrival date {s} {s}", .{ id, try self.dateText(r.first) });
449 if (self.inhouse(r.room)) |guest| return self.fail("room occupied {d} {s}", .{ r.room.number, guest.id });
450 if (!r.room.clean) return self.fail("room dirty {d}", .{r.room.number});
451 r.status = .inhouse;
452 try self.out.print("CHECKIN {s} {s} room {d} until {s}\n", .{ id, r.guest, r.room.number, try self.dateText(r.last) });
453 } else if (std.mem.eql(u8, op, "MOVE")) {
454 try self.statusCheck(r, .inhouse);
455 const dest = try self.room(a[2]);
456 if (dest == r.room) return self.fail("same room {d}", .{dest.number});
457 var count: usize = 0;
458 for (r.nights.items) |n| {
459 if (n.day >= self.today) count += 1;
460 }
461 if (count == 0) return self.fail("no remaining nights {s}", .{id});
462 for (r.nights.items) |n| {
463 if (n.day >= self.today and !free(dest, n.day, r)) return self.fail("room unavailable {d} {s}", .{ dest.number, try self.dateText(n.day) });
464 }
465 if (self.inhouse(dest)) |guest| return self.fail("room occupied {d} {s}", .{ dest.number, guest.id });
466 if (!dest.clean) return self.fail("room dirty {d}", .{dest.number});
467 const old = r.room;
468 for (r.nights.items) |*n| {
469 if (n.day >= self.today) {
470 _ = n.room.occupied.remove(n.day);
471 n.room = dest;
472 try dest.occupied.put(self.gpa, n.day, r);
473 }
474 }
475 try self.out.print("MOVE {s} {d} -> {d} nights {d}\n", .{ id, old.number, dest.number, count });
476 old.clean = false;
477 r.room = dest;
478 } else if (std.mem.eql(u8, op, "EXTEND")) {
479 try self.statusCheck(r, .inhouse);
480 const last = try self.date(a[2]);
481 if (last <= r.last) return self.fail("invalid range {s} {s}", .{ try self.dateText(r.last), a[2] });
482 if (last - r.first > 30) return self.fail("too long {d}", .{last - r.first});
483 if (unavailable(r.room, r.last, last, r)) |d| return self.fail("room unavailable {d} {s}", .{ r.room.number, try self.dateText(d) });
484 const nights = try self.price(r.room, r.last, last);
485 try r.nights.appendSlice(self.gpa, nights.items);
486 r.last = last;
487 try self.occupy(r);
488 try self.out.print("EXTEND {s} until {s} nights {d} total {s}\n", .{ id, a[2], r.nights.items.len, try self.money(r.lodging()) });
489 } else if (std.mem.eql(u8, op, "CHARGE")) {
490 try self.statusCheck(r, .inhouse);
491 const m = amount(a[2]);
492 if (m == 0) return self.fail("invalid amount", .{});
493 try r.extras.append(self.gpa, .{ .name = a[3], .amount = m });
494 try self.out.print("CHARGE {s} {s} {s} extras {s}\n", .{ id, a[3], try self.money(m), try self.money(r.extraTotal()) });
495 } else if (std.mem.eql(u8, op, "PAY")) {
496 if (r.status != .booked and r.status != .inhouse) return self.fail("invalid status {s} {s}", .{ id, @tagName(r.status) });
497 const m = amount(a[2]);
498 if (m == 0) return self.fail("invalid amount", .{});
499 r.deposit += m;
500 try self.out.print("PAY {s} {s} deposit {s}\n", .{ id, try self.money(m), try self.money(r.deposit) });
501 } else if (std.mem.eql(u8, op, "CHECKOUT")) {
502 try self.statusCheck(r, .inhouse);
503 try self.checkout(r, self.today, "");
504 } else {
505 try self.out.print("INVOICE {s} {s} room {d} {s} {s}\n", .{ id, r.guest, r.room.number, r.room.kind, @tagName(r.status) });
506 if (r.status == .cancelled or r.status == .noshow) {
507 try self.out.print(" {s} fee {s}\n total {s}\n {s}\n", .{ if (r.status == .cancelled) "cancellation" else "noshow", try self.money(r.fee), try self.money(r.fee), try self.deposit(r, r.fee) });
508 } else {
509 for (r.nights.items) |n| try self.out.print(" {s} room {d} {s}\n", .{ try self.dated(n.day), n.room.number, try self.money(n.rate) });
510 const b = if (r.status == .checkedout) r.final_bill.? else self.bill(r);
511 try self.out.print(" lodging {s}\n discount {s}\n", .{ try self.money(b[0]), try self.money(b[1]) });
512 if (r.penalty) |n| try self.out.print(" penalty {s} {s}\n", .{ try self.dated(n.day), try self.money(b[2]) });
513 for (r.extras.items) |e| try self.out.print(" extra {s} {s}\n", .{ e.name, try self.money(e.amount) });
514 try self.out.print(" tax {s}\n total {s}\n {s}\n", .{ try self.money(b[4]), try self.money(b[5]), try self.deposit(r, b[5]) });
515 }
516 }
517 } else if (std.mem.eql(u8, op, "AVAIL")) {
518 try self.typeCheck(id);
519 const span = try self.range(a[2], a[3], false, false, true);
520 try self.out.print("AVAIL {s} {s} -> {s}\n", .{ id, a[2], a[3] });
521 var d = span.first;
522 while (d <= span.last) : (d += 1) {
523 var count: usize = 0;
524 for (self.rooms.items) |r| {
525 if (std.mem.eql(u8, r.kind, id) and free(r, d, null)) count += 1;
526 }
527 try self.out.print(" {s} free {d}", .{ try self.dated(d), count });
528 for (self.rooms.items) |r| {
529 if (std.mem.eql(u8, r.kind, id) and free(r, d, null)) try self.out.print(" {d}", .{r.number});
530 }
531 try self.out.writeByte('\n');
532 }
533 } else if (std.mem.eql(u8, op, "OCCUPANCY")) {
534 const d = try self.date(id);
535 try self.out.print("OCCUPANCY {s} {s}\n", .{ try self.dated(d), try self.occupancy(null, d) });
536 for (types) |kind| {
537 var count: usize = 0;
538 for (self.rooms.items) |r| {
539 if (std.mem.eql(u8, r.kind, kind)) count += 1;
540 }
541 if (count > 0) try self.out.print(" {s} {s}\n", .{ kind, try self.occupancy(kind, d) });
542 }
543 } else if (std.mem.eql(u8, op, "STATS")) {
544 const span = try self.range(id, a[2], false, false, true);
545 var totals: [3]i64 = @splat(0);
546 var d = span.first;
547 try self.out.print("STATS {s} -> {s}\n", .{ id, a[2] });
548 while (d <= span.last) : (d += 1) {
549 const m = self.metrics(d);
550 for (0..3) |i| totals[i] += m[i];
551 try self.out.print(" {s} {s}\n", .{ try self.dated(d), try self.metricText(m[0], m[1], m[2]) });
552 }
553 try self.out.print(" total {s}\n", .{try self.metricText(totals[0], totals[1], totals[2])});
554 } else if (std.mem.eql(u8, op, "ROOMS")) {
555 try self.out.print("ROOMS {s} rooms {d}\n", .{ try self.dated(self.today), self.rooms.items.len });
556 for (self.rooms.items) |r| {
557 const guest = self.inhouse(r);
558 var next: ?*Reservation = null;
559 for (self.reservations.values()) |x| {
560 if (x.status == .booked and x.room == r and (next == null or x.first < next.?.first)) next = x;
561 }
562 try self.out.print(" {d} {s} base {s} tonight {s} {s} {s}", .{ r.number, r.kind, try self.money(r.rate), try self.money(self.rate(r, self.today)), if (r.clean) "clean" else "dirty", if (guest) |g| try std.fmt.allocPrint(self.gpa, "occupied {s}", .{g.id}) else if (r.blocks.contains(self.today)) "blocked" else "vacant" });
563 if (next) |x| try self.out.print(" next {s} {s}", .{ x.id, try self.dateText(x.first) });
564 try self.out.writeByte('\n');
565 }
566 } else if (std.mem.eql(u8, op, "LIST")) {
567 var selected: std.ArrayList(*Reservation) = .empty;
568 for (self.reservations.values()) |r| {
569 if (std.mem.eql(u8, @tagName(r.status), id)) try selected.append(self.gpa, r);
570 }
571 try self.out.print("LIST {s} {d}\n", .{ id, selected.items.len });
572 for (try self.sorted(selected.items, false, false)) |r| try self.out.print(" {s} {s} room {d} {s} -> {s}\n", .{ r.id, r.guest, r.room.number, try self.dateText(r.first), try self.dateText(r.last) });
573 } else if (std.mem.eql(u8, op, "QUEUE")) {
574 try self.out.print("QUEUE {d}\n", .{self.queue.items.len});
575 for (self.queue.items, 0..) |w, i| try self.out.print(" {d} {s} {s} {s} {s} -> {s}\n", .{ i + 1, w.id, w.guest, w.kind, try self.dateText(w.first), try self.dateText(w.last) });
576 } else if (std.mem.eql(u8, op, "SCHEDULE")) {
577 const d = try self.date(id);
578 var arrivals: std.ArrayList(*Reservation) = .empty;
579 var departures: std.ArrayList(*Reservation) = .empty;
580 for (self.reservations.values()) |r| {
581 if (r.first == d and r.status != .cancelled) try arrivals.append(self.gpa, r);
582 if (r.last == d and (r.status == .booked or r.status == .inhouse or r.status == .checkedout)) try departures.append(self.gpa, r);
583 }
584 try self.out.print("SCHEDULE {s} arrivals {d} departures {d}\n", .{ try self.dated(d), arrivals.items.len, departures.items.len });
585 for (try self.sorted(arrivals.items, false, true)) |r| try self.out.print(" arrival {s} {s} room {d} {s}\n", .{ r.id, r.guest, r.room.number, @tagName(r.status) });
586 for (try self.sorted(departures.items, false, true)) |r| try self.out.print(" departure {s} {s} room {d} {s}\n", .{ r.id, r.guest, r.room.number, @tagName(r.status) });
587 } else if (std.mem.eql(u8, op, "GUEST")) {
588 var count: usize = 0;
589 for (self.reservations.values()) |r| {
590 if (std.mem.eql(u8, r.guest, id)) count += 1;
591 }
592 if (count == 0) return self.fail("unknown guest {s}", .{id});
593 try self.out.print("GUEST {s}\n", .{id});
594 for (self.reservations.values()) |r| {
595 if (!std.mem.eql(u8, r.guest, id)) continue;
596 const active = r.status == .booked or r.status == .inhouse;
597 try self.out.print(" {s} room {d} {s} {s} -> {s} {s} {s} {s}\n", .{ r.id, r.room.number, r.room.kind, try self.dateText(r.first), try self.dateText(r.last), @tagName(r.status), if (active) "deposit" else "paid", try self.money(if (active) r.deposit else r.paid) });
598 }
599 try self.out.print(" total paid {s} points {d} tier {s}\n", .{ try self.money(self.paid(id)), @divTrunc(self.paid(id), 100), self.tier(id) });
600 } else if (std.mem.eql(u8, op, "REVENUE")) {
601 const span = try self.range(id, a[2], false, false, false);
602 var totals: [7]i64 = @splat(0);
603 var counts: [3]usize = @splat(0);
604 for (self.reservations.values()) |r| {
605 const posting = r.posting orelse continue;
606 if (posting < span.first or posting > span.last) continue;
607 if (r.status == .checkedout) {
608 counts[0] += 1;
609 for (0..5) |i| totals[i] += r.final_bill.?[i];
610 } else if (r.status == .cancelled) {
611 counts[1] += 1;
612 totals[5] += r.fee;
613 } else if (r.status == .noshow) {
614 counts[2] += 1;
615 totals[6] += r.fee;
616 }
617 }
618 try self.out.print("REVENUE {s} -> {s}\n", .{ id, a[2] });
619 const labels = [_][]const u8{ "lodging", "discount", "penalty", "extras", "tax", "cancellation", "noshow" };
620 for (0..7) |i| try self.out.print(" {s} {s}\n", .{ labels[i], try self.money(totals[i]) });
621 try self.out.print(" total {s}\n checkouts {d} cancellations {d} noshows {d}\n", .{ try self.money(totals[0] - totals[1] + totals[2] + totals[3] + totals[4] + totals[5] + totals[6]), counts[0], counts[1], counts[2] });
622 }
623 }
624};
625pub fn main(init: std.process.Init) !void {
626 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
627 defer arena.deinit();
628 const gpa = arena.allocator();
629 var ib: [4096]u8 = undefined;
630 var input = std.Io.File.stdin().readerStreaming(init.io, &ib);
631 var ob: [8192]u8 = undefined;
632 var output = std.Io.File.stdout().writerStreaming(init.io, &ob);
633 var hotel = Hotel{ .gpa = gpa, .out = &output.interface };
634 const n: usize = @intCast(integer((try input.interface.takeDelimiter('\n')).?));
635 for (0..n) |_| {
636 const line = try gpa.dupe(u8, (try input.interface.takeDelimiter('\n')).?);
637 var a: std.ArrayList([]const u8) = .empty;
638 var it = std.mem.splitScalar(u8, line, ' ');
639 while (it.next()) |part| try a.append(gpa, part);
640 var draft: std.Io.Writer.Allocating = .init(gpa);
641 hotel.out = &draft.writer;
642 hotel.execute(a.items) catch |err| {
643 if (err == error.BadCommand) {
644 try output.interface.print("ERROR {s}\n", .{hotel.error_message});
645 hotel.out = &output.interface;
646 try hotel.processQueue();
647 continue;
648 }
649 return err;
650 };
651 try output.interface.writeAll(draft.writer.buffered());
652 hotel.out = &output.interface;
653 try hotel.processQueue();
654 }
655 try output.interface.flush();
656}
