3const types = ['SINGLE', 'DOUBLE', 'SUITE'];
4const weekdays = ['SAT', 'SUN', 'MON', 'TUE', 'WED', 'THU', 'FRI'];
5int amount(String s) {
6 final p = s.split('.');
7 return int.parse(p[0]) * 100 + int.parse(p[1]);
8}
9String money(int c) => '${c ~/ 100}.${(c % 100).toString().padLeft(2, '0')}';
10class HotelError implements Exception {
11 final String reason;
12 HotelError(this.reason);
13}
14Never fail(String s) => throw HotelError(s);
15bool leap(int y) => y % 4 == 0 && (y % 100 != 0 || y % 400 == 0);
16int monthDays(int y, int m) => m == 2 && leap(y) ? 29 : [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31][m - 1];
17int date(String s) {
18 final y = int.parse(s.substring(0, 4)), m = int.parse(s.substring(5, 7)), d = int.parse(s.substring(8, 10));
19 if (y < 2000 || y > 2099 || m < 1 || m > 12 || d < 1 || d > monthDays(y, m)) fail('invalid date $s');
20 var result = d - 1;
21 for (var year = 2000; year < y; year++) result += leap(year) ? 366 : 365;
22 for (var month = 1; month < m; month++) result += monthDays(y, month);
23 return result;
24}
25String dateText(int n) {
26 var y = 2000, m = 1;
27 while (n >= (leap(y) ? 366 : 365)) {
28 n -= leap(y) ? 366 : 365;
29 y++;
30 }
31 while (n >= monthDays(y, m)) {
32 n -= monthDays(y, m);
33 m++;
34 }
35 return '$y-${m.toString().padLeft(2, '0')}-${(n + 1).toString().padLeft(2, '0')}';
36}
37String dated(int n) => '${dateText(n)} ${weekdays[n % 7]}';
38class Room {
39 final int number;
40 final String type;
41 int rate;
42 bool clean = true;
43 final blocks = <int>{};
44 final occupied = <int, Reservation>{};
45 Room(this.number, this.type, this.rate);
46 bool free(int day, [Reservation? except]) => !blocks.contains(day) && (occupied[day] == null || occupied[day] == except);
47}
48class Season {
49 final int start, end, percent;
50 Season(this.start, this.end, this.percent);
51}
52class Night {
53 final int day, rate;
54 Room room;
55 Night(this.day, this.room, this.rate);
56}
57class Reservation {
58 final String id, guest;
59 Room room;
60 int start, end;
61 String status = 'booked';
62 List<Night> nights = [];
63 final extras = <(String, int)>[];
64 int deposit = 0, paid = 0, fee = 0;
65 int? posting;
66 Night? penalty;
67 List<int>? finalBill;
68 Reservation(this.id, this.guest, this.room, this.start, this.end);
69 int get lodging => nights.fold(0, (n, night) => n + night.rate);
70 int get extraTotal => extras.fold(0, (n, e) => n + e.$2);
71}
72class Waiting {
73 final String id, guest, type;
74 final int start, end, expiry;
75 Waiting(this.id, this.guest, this.type, this.start, this.end) : expiry = start;
76}
77class Hotel {
78 int today = date('2026-01-01');
79 final rooms = <int, Room>{};
80 final reservations = <String, Reservation>{};
81 final used = <String>{};
82 final seasons = <Season>[];
83 final queue = <Waiting>[];
84 List<Room> roomList() => rooms.values.toList()..sort((a, b) => a.number.compareTo(b.number));
85 List<Reservation> sorted(Iterable<Reservation> rs, {bool departure = false, bool ids = false}) => rs.toList()..sort((a, b) {
86 final c = ids ? 0 : (departure ? a.end : a.start).compareTo(departure ? b.end : b.start);
87 return c != 0 ? c : a.id.compareTo(b.id);
88 });
89 Room room(String token) => rooms[int.parse(token)] ?? fail('unknown room $token');
90 Reservation reservation(String id) => reservations[id] ?? fail('unknown reservation $id');
91 void typeCheck(String type) {
92 if (!types.contains(type)) fail('invalid type $type');
93 }
94 void idCheck(String id) {
95 if (used.contains(id)) fail('duplicate reservation $id');
96 }
97 void statusCheck(Reservation r, String s) {
98 if (r.status != s) fail('invalid status ${r.id} ${r.status}');
99 }
100 (int, int) range(String a, String b, {bool stay = false, bool past = false, bool limit = true}) {
101 final first = date(a), last = date(b);
102 if ((stay || past) && first < today) fail('past date $a');
103 if (stay ? last <= first : last < first) fail('invalid range $a $b');
104 final count = last - first + (stay ? 0 : 1);
105 if (limit && count > (stay ? 30 : 31)) fail('too long $count');
106 return (first, last);
107 }
108 int rate(Room r, int d) {
109 final season = seasons.reversed.where((s) => s.start <= d && d <= s.end).firstOrNull;
110 return r.rate * (season?.percent ?? 100) ~/ 100;
111 }
112 List<Night> price(Room room, int a, int b) => [for (var d = a; d < b; d++) Night(d, room, rate(room, d))];
113 int? unavailable(Room r, int a, int b, [Reservation? except]) {
114 for (var d = a; d < b; d++) {
115 if (!r.free(d, except)) return d;
116 }
117 return null;
118 }
119 Room? search(String type, int a, int b, [Reservation? except]) => roomList().where((r) => r.type == type && unavailable(r, a, b, except) == null).firstOrNull;
120 Reservation? inhouse(Room room) => reservations.values.where((r) => r.status == 'inhouse' && r.room == room).firstOrNull;
121 void release(Reservation r) {
122 for (final n in r.nights)
123 if (n.room.occupied[n.day] == r) n.room.occupied.remove(n.day);
124 }
125 void occupy(Reservation r) {
126 for (final n in r.nights) n.room.occupied[n.day] = r;
127 }
128 Reservation create(String id, String guest, Room room, int a, int b, StringBuffer out, {String indent = ''}) {
129 final r = Reservation(id, guest, room, a, b);
130 r.nights = price(room, a, b);
131 reservations[id] = r;
132 used.add(id);
133 occupy(r);
134 out.writeln('${indent}BOOK $id $guest room ${room.number} ${room.type} ${dateText(a)} -> ${dateText(b)} nights ${b - a} total ${money(r.lodging)}');
135 return r;
136 }
137 int paid(String guest) => reservations.values.where((r) => r.guest == guest).fold(0, (n, r) => n + r.paid);
138 String tier(String guest) {
139 final points = paid(guest) ~/ 100;
140 return points >= 2000 ? 'GOLD' : points >= 500 ? 'SILVER' : 'BRONZE';
141 }
142 List<int> bill(Reservation r) {
143 final level = tier(r.guest), lodging = r.lodging;
144 final discount = lodging * (level == 'GOLD' ? 10 : level == 'SILVER' ? 5 : 0) ~/ 100;
145 final penalty = r.penalty?.rate ?? 0, extras = r.extraTotal;
146 final subtotal = lodging - discount + penalty + extras;
147 final tax = subtotal ~/ 10;
148 return [lodging, discount, penalty, extras, tax, subtotal + tax];
149 }
150 String deposit(Reservation r, int total) => 'deposit ${money(r.deposit)} ${total >= r.deposit ? 'due ${money(total - r.deposit)}' : 'refund ${money(r.deposit - total)}'}';
151 void checkout(Reservation r, int day, StringBuffer out, {String indent = ''}) {
152 if (day < r.end) {
153 r.penalty = r.nights.firstWhere((n) => n.day == day);
154 release(r);
155 r.nights = r.nights.where((n) => n.day < day).toList();
156 r.end = day;
157 occupy(r);
158 }
159 final b = bill(r);
160 r.finalBill = b;
161 r.paid = b[5];
162 r.status = 'checkedout';
163 r.posting = day;
164 r.room.clean = false;
165 out.writeln('${indent}CHECKOUT ${r.id} nights ${r.nights.length} lodging ${money(b[0])} discount ${money(b[1])} penalty ${money(b[2])} extras ${money(b[3])} tax ${money(b[4])} total ${money(b[5])} ${deposit(r, b[5])}');
166 }
167 void processQueue(StringBuffer out) {
168 for (final w in queue.toList()) {
169 if (w.expiry < today) {
170 queue.remove(w);
171 out.writeln(' WAITLIST ${w.id} expired');
172 continue;
173 }
174 final r = search(w.type, w.start, w.end);
175 if (r != null) {
176 queue.remove(w);
177 create(w.id, w.guest, r, w.start, w.end, out, indent: ' ');
178 }
179 }
180 }
181 (int, int, int) metrics(int d) {
182 var available = 0, sold = 0, revenue = 0;
183 for (final r in rooms.values) {
184 if (!r.blocks.contains(d)) available++;
185 final guest = r.occupied[d];
186 if (guest != null) {
187 sold++;
188 revenue += guest.nights.firstWhere((n) => n.day == d && n.room == r).rate;
189 }
190 }
191 return (available, sold, revenue);
192 }
193 String metricText(int s, int k, int v) => 'sold $k/$s rate ${s == 0 ? 'n/a' : '${k * 100 ~/ s}%'} revenue ${money(v)} adr ${k == 0 ? 'n/a' : money(v ~/ k)} revpar ${s == 0 ? 'n/a' : money(v ~/ s)}';
194 String occupancy(Iterable<Room> rs, int d) {
195 final list = rs.toList(), blocked = list.where((r) => r.blocks.contains(d)).length, occupied = list.where((r) => r.occupied.containsKey(d)).length;
196 final available = list.length - blocked;
197 return 'rooms ${list.length} blocked $blocked occupied $occupied rate ${available == 0 ? 'n/a' : '${occupied * 100 ~/ available}%'}';
198 }
199 void execute(List<String> a, StringBuffer out) {
200 final op = a[0], id = a.length > 1 ? a[1] : '';
201 if (op == 'ROOM') {
202 final n = int.parse(id);
203 if (rooms.containsKey(n)) fail('duplicate room $id');
204 typeCheck(a[2]);
205 final r = Room(n, a[2], amount(a[3]));
206 rooms[n] = r;
207 out.writeln('ROOM $n ${r.type} ${money(r.rate)}');
208 } else if (op == 'RATE') {
209 final r = room(id);
210 final old = r.rate;
211 r.rate = amount(a[2]);
212 out.writeln('RATE $id ${money(old)} -> ${money(r.rate)}');
213 } else if (op == 'SEASON') {
214 final (first, last) = range(id, a[2], limit: false);
215 seasons.add(Season(first, last, int.parse(a[3])));
216 out.writeln('SEASON ${seasons.length} $id -> ${a[2]} ${a[3]}%');
217 } else if (op == 'BLOCK' || op == 'UNBLOCK') {
218 final r = room(id);
219 final (first, last) = range(a[2], a[3], past: true);
220 if (op == 'BLOCK') {
221 for (var d = first; d <= last; d++) {
222 if (r.occupied.containsKey(d)) fail('room unavailable $id ${dateText(d)}');
223 }
224 }
225 var changed = 0;
226 for (var d = first; d <= last; d++)
227 if (op == 'BLOCK' ? r.blocks.add(d) : r.blocks.remove(d)) changed++;
228 out.writeln('$op $id ${a[2]} -> ${a[3]} ${op == 'BLOCK' ? 'added' : 'removed'} $changed');
229 } else if (op == 'CLEAN') {
230 final r = room(id);
231 if (r.clean) fail('already clean $id');
232 r.clean = true;
233 out.writeln('CLEAN $id');
234 } else if (op == 'TODAY') {
235 final d = date(id);
236 if (d < today) fail('past date $id');
237 out.writeln('TODAY ${dated(d)}');
238 for (final r in sorted(reservations.values.where((r) => r.status == 'inhouse' && r.end < d), departure: true)) checkout(r, r.end, out, indent: ' ');
239 for (final r in sorted(reservations.values.where((r) => r.status == 'booked' && r.start < d))) {
240 r.fee = r.nights.first.rate;
241 r.paid = r.fee;
242 r.status = 'noshow';
243 r.posting = r.start;
244 release(r);
245 out.writeln(' NOSHOW ${r.id} ${r.guest} fee ${money(r.fee)} ${deposit(r, r.fee)}');
246 }
247 today = d;
248 } else if (op == 'BOOK' || op == 'BOOKROOM' || op == 'WAITLIST') {
249 idCheck(id);
250 Room? selected;
251 if (op == 'BOOKROOM') {
252 selected = room(a[3]);
253 } else {
254 typeCheck(a[3]);
255 }
256 final (first, last) = range(a[4], a[5], stay: true);
257 if (op == 'BOOKROOM') {
258 final d = unavailable(selected!, first, last);
259 if (d != null) fail('room unavailable ${selected.number} ${dateText(d)}');
260 } else {
261 selected = search(a[3], first, last);
262 }
263 if (selected != null) {
264 create(id, a[2], selected, first, last, out);
265 } else if (op == 'WAITLIST') {
266 queue.add(Waiting(id, a[2], a[3], first, last));
267 used.add(id);
268 out.writeln('WAITLIST $id queued position ${queue.length}');
269 } else {
270 fail('no availability ${a[3]} ${a[4]} ${a[5]}');
271 }
272 } else if (op == 'CANCEL') {
273 final index = queue.indexWhere((w) => w.id == id);
274 if (index >= 0) {
275 queue.removeAt(index);
276 out.writeln('CANCEL $id waitlist removed');
277 return;
278 }
279 final r = reservation(id);
280 statusCheck(r, 'booked');
281 final days = r.start - today;
282 r.fee = days >= 7 ? 0 : days >= 1 ? r.nights.first.rate ~/ 2 : r.nights.first.rate;
283 r.paid = r.fee;
284 r.status = 'cancelled';
285 r.posting = today;
286 release(r);
287 out.writeln('CANCEL $id days $days fee ${money(r.fee)} ${deposit(r, r.fee)}');
288 } else if (['MODIFY', 'CHECKIN', 'MOVE', 'EXTEND', 'CHARGE', 'PAY', 'CHECKOUT', 'INVOICE'].contains(op)) {
289 final r = reservation(id);
290 if (op == 'MODIFY') {
291 statusCheck(r, 'booked');
292 final (first, last) = range(a[2], a[3], stay: true);
293 final old = r.room, selected = unavailable(r.room, first, last, r) == null ? r.room : search(r.room.type, first, last, r);
294 if (selected == null) fail('no availability ${r.room.type} ${a[2]} ${a[3]}');
295 release(r);
296 r.room = selected;
297 r.start = first;
298 r.end = last;
299 r.nights = price(selected, first, last);
300 occupy(r);
301 out.writeln('MODIFY $id room ${selected.number} ${selected == old ? 'kept' : 'reassigned'} ${a[2]} -> ${a[3]} nights ${last - first} total ${money(r.lodging)}');
302 } else if (op == 'CHECKIN') {
303 statusCheck(r, 'booked');
304 if (r.start != today) fail('not arrival date $id ${dateText(r.start)}');
305 final occupant = inhouse(r.room);
306 if (occupant != null) fail('room occupied ${r.room.number} ${occupant.id}');
307 if (!r.room.clean) fail('room dirty ${r.room.number}');
308 r.status = 'inhouse';
309 out.writeln('CHECKIN $id ${r.guest} room ${r.room.number} until ${dateText(r.end)}');
310 } else if (op == 'MOVE') {
311 statusCheck(r, 'inhouse');
312 final dest = room(a[2]);
313 if (dest == r.room) fail('same room ${dest.number}');
314 final remaining = r.nights.where((n) => n.day >= today).toList();
315 if (remaining.isEmpty) fail('no remaining nights $id');
316 for (final n in remaining)
317 if (!dest.free(n.day, r)) fail('room unavailable ${dest.number} ${dateText(n.day)}');
318 final occupant = inhouse(dest);
319 if (occupant != null) fail('room occupied ${dest.number} ${occupant.id}');
320 if (!dest.clean) fail('room dirty ${dest.number}');
321 final old = r.room;
322 for (final n in remaining) {
323 n.room.occupied.remove(n.day);
324 n.room = dest;
325 dest.occupied[n.day] = r;
326 }
327 out.writeln('MOVE $id ${old.number} -> ${dest.number} nights ${remaining.length}');
328 old.clean = false;
329 r.room = dest;
330 } else if (op == 'EXTEND') {
331 statusCheck(r, 'inhouse');
332 final end = date(a[2]);
333 if (end <= r.end) fail('invalid range ${dateText(r.end)} ${a[2]}');
334 if (end - r.start > 30) fail('too long ${end - r.start}');
335 final d = unavailable(r.room, r.end, end, r);
336 if (d != null) fail('room unavailable ${r.room.number} ${dateText(d)}');
337 r.nights.addAll(price(r.room, r.end, end));
338 r.end = end;
339 occupy(r);
340 out.writeln('EXTEND $id until ${a[2]} nights ${r.nights.length} total ${money(r.lodging)}');
341 } else if (op == 'CHARGE') {
342 statusCheck(r, 'inhouse');
343 final m = amount(a[2]);
344 if (m == 0) fail('invalid amount');
345 r.extras.add((a[3], m));
346 out.writeln('CHARGE $id ${a[3]} ${money(m)} extras ${money(r.extraTotal)}');
347 } else if (op == 'PAY') {
348 if (r.status != 'booked' && r.status != 'inhouse') fail('invalid status $id ${r.status}');
349 final m = amount(a[2]);
350 if (m == 0) fail('invalid amount');
351 r.deposit += m;
352 out.writeln('PAY $id ${money(m)} deposit ${money(r.deposit)}');
353 } else if (op == 'CHECKOUT') {
354 statusCheck(r, 'inhouse');
355 checkout(r, today, out);
356 } else {
357 out.writeln('INVOICE $id ${r.guest} room ${r.room.number} ${r.room.type} ${r.status}');
358 if (r.status == 'cancelled' || r.status == 'noshow') {
359 out.writeln(' ${r.status == 'cancelled' ? 'cancellation' : 'noshow'} fee ${money(r.fee)}');
360 out.writeln(' total ${money(r.fee)}');
361 out.writeln(' ${deposit(r, r.fee)}');
362 } else {
363 for (final n in r.nights) out.writeln(' ${dated(n.day)} room ${n.room.number} ${money(n.rate)}');
364 final b = r.status == 'checkedout' ? r.finalBill! : bill(r);
365 out.writeln(' lodging ${money(b[0])}');
366 out.writeln(' discount ${money(b[1])}');
367 if (r.penalty != null) out.writeln(' penalty ${dated(r.penalty!.day)} ${money(b[2])}');
368 for (final e in r.extras) out.writeln(' extra ${e.$1} ${money(e.$2)}');
369 out.writeln(' tax ${money(b[4])}');
370 out.writeln(' total ${money(b[5])}');
371 out.writeln(' ${deposit(r, b[5])}');
372 }
373 }
374 } else if (op == 'AVAIL') {
375 typeCheck(id);
376 final (first, last) = range(a[2], a[3]);
377 out.writeln('AVAIL $id ${a[2]} -> ${a[3]}');
378 final rs = roomList();
379 for (var d = first; d <= last; d++) {
380 final available = rs.where((r) => r.type == id && r.free(d)).toList();
381 out.writeln(' ${dated(d)} free ${available.length}${available.isEmpty ? '' : ' ${available.map((r) => r.number).join(' ')}'}');
382 }
383 } else if (op == 'OCCUPANCY') {
384 final d = date(id);
385 out.writeln('OCCUPANCY ${dated(d)} ${occupancy(rooms.values, d)}');
386 for (final type in types) {
387 final rs = rooms.values.where((r) => r.type == type);
388 if (rs.isNotEmpty) out.writeln(' $type ${occupancy(rs, d)}');
389 }
390 } else if (op == 'STATS') {
391 final (first, last) = range(id, a[2]);
392 var s = 0, k = 0, value = 0;
393 out.writeln('STATS $id -> ${a[2]}');
394 for (var d = first; d <= last; d++) {
395 final m = metrics(d);
396 s += m.$1;
397 k += m.$2;
398 value += m.$3;
399 out.writeln(' ${dated(d)} ${metricText(m.$1, m.$2, m.$3)}');
400 }
401 out.writeln(' total ${metricText(s, k, value)}');
402 } else if (op == 'ROOMS') {
403 out.writeln('ROOMS ${dated(today)} rooms ${rooms.length}');
404 for (final r in roomList()) {
405 final occupant = inhouse(r), next = sorted(reservations.values.where((x) => x.status == 'booked' && x.room == r));
406 out.writeln(' ${r.number} ${r.type} base ${money(r.rate)} tonight ${money(rate(r, today))} ${r.clean ? 'clean' : 'dirty'} ${occupant != null ? 'occupied ${occupant.id}' : r.blocks.contains(today) ? 'blocked' : 'vacant'}${next.isEmpty ? '' : ' next ${next.first.id} ${dateText(next.first.start)}'}');
407 }
408 } else if (op == 'LIST') {
409 final rs = sorted(reservations.values.where((r) => r.status == id));
410 out.writeln('LIST $id ${rs.length}');
411 for (final r in rs) out.writeln(' ${r.id} ${r.guest} room ${r.room.number} ${dateText(r.start)} -> ${dateText(r.end)}');
412 } else if (op == 'QUEUE') {
413 out.writeln('QUEUE ${queue.length}');
414 for (var i = 0; i < queue.length; i++) {
415 final w = queue[i];
416 out.writeln(' ${i + 1} ${w.id} ${w.guest} ${w.type} ${dateText(w.start)} -> ${dateText(w.end)}');
417 }
418 } else if (op == 'SCHEDULE') {
419 final d = date(id);
420 final arrivals = sorted(reservations.values.where((r) => r.start == d && r.status != 'cancelled'), ids: true);
421 final departures = sorted(reservations.values.where((r) => r.end == d && ['booked', 'inhouse', 'checkedout'].contains(r.status)), ids: true);
422 out.writeln('SCHEDULE ${dated(d)} arrivals ${arrivals.length} departures ${departures.length}');
423 for (final r in arrivals) out.writeln(' arrival ${r.id} ${r.guest} room ${r.room.number} ${r.status}');
424 for (final r in departures) out.writeln(' departure ${r.id} ${r.guest} room ${r.room.number} ${r.status}');
425 } else if (op == 'GUEST') {
426 final rs = reservations.values.where((r) => r.guest == id).toList();
427 if (rs.isEmpty) fail('unknown guest $id');
428 out.writeln('GUEST $id');
429 for (final r in rs) out.writeln(' ${r.id} room ${r.room.number} ${r.room.type} ${dateText(r.start)} -> ${dateText(r.end)} ${r.status} ${r.status == 'booked' || r.status == 'inhouse' ? 'deposit ${money(r.deposit)}' : 'paid ${money(r.paid)}'}');
430 out.writeln(' total paid ${money(paid(id))} points ${paid(id) ~/ 100} tier ${tier(id)}');
431 } else if (op == 'REVENUE') {
432 final (first, last) = range(id, a[2], limit: false);
433 final totals = List.filled(7, 0), counts = List.filled(3, 0);
434 for (final r in reservations.values) {
435 if (r.posting == null || r.posting! < first || r.posting! > last) continue;
436 if (r.status == 'checkedout') {
437 counts[0]++;
438 for (var i = 0; i < 5; i++) totals[i] += r.finalBill![i];
439 } else if (r.status == 'cancelled') {
440 counts[1]++;
441 totals[5] += r.fee;
442 } else if (r.status == 'noshow') {
443 counts[2]++;
444 totals[6] += r.fee;
445 }
446 }
447 out.writeln('REVENUE $id -> ${a[2]}');
448 final labels = ['lodging', 'discount', 'penalty', 'extras', 'tax', 'cancellation', 'noshow'];
449 for (var i = 0; i < 7; i++) out.writeln(' ${labels[i]} ${money(totals[i])}');
450 out.writeln(' total ${money(totals[0] - totals[1] + totals[2] + totals[3] + totals[4] + totals[5] + totals[6])}');
451 out.writeln(' checkouts ${counts[0]} cancellations ${counts[1]} noshows ${counts[2]}');
452 }
453 }
454}
455void main() {
456 final hotel = Hotel(), n = int.parse(stdin.readLineSync()!);
457 for (var i = 0; i < n; i++) {
458 final draft = StringBuffer(), a = stdin.readLineSync()!.split(' ');
459 try {
460 hotel.execute(a, draft);
461 stdout.write(draft);
462 } on HotelError catch (e) {
463 print('ERROR ${e.reason}');
464 }
465 draft.clear();
466 hotel.processQueue(draft);
467 stdout.write(draft);
468 }
