4const TYPES = ["SINGLE", "DOUBLE", "SUITE"];
5const WEEKDAYS = ["SAT", "SUN", "MON", "TUE", "WED", "THU", "FRI"];
6const DATES = [];
7const DAYS = new Map();
8for (let year = 2000; year <= 2099; year++) {
9 const leap = year % 4 === 0 && (year % 100 !== 0 || year % 400 === 0);
10 const lengths = [31, leap ? 29 : 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31];
11 for (let month = 1; month <= 12; month++) {
12 for (let day = 1; day <= lengths[month - 1]; day++) {
13 const text = year + "-" + String(month).padStart(2, "0") + "-" + String(day).padStart(2, "0");
14 DAYS.set(text, DATES.length);
15 DATES.push(text);
16 }
17 }
18}
19const lex = (a, b) => a < b ? -1 : a > b ? 1 : 0;
20const dateText = (day) => DATES[day];
21const dateWeek = (day) => dateText(day) + " " + WEEKDAYS[day % 7];
22const cents = (s) => BigInt(s.replace(".", ""));
23const money = (v) => v / 100n + "." + String(v % 100n).padStart(2, "0");
24const lodging = (x) => x.nights.reduce((v, n) => v + n.rate, 0n);
25const extras = (x) => x.extras.reduce((v, e) => v + e.amount, 0n);
26const byArrival = (a, b) => a.start - b.start || lex(a.id, b.id);
27class CommandError extends Error {}

29class Hotel {
30 today = DAYS.get("2026-01-01");
31 rooms = new Map();
32 seasons = [];
33 reservations = new Map();
34 queue = [];
35 used = new Set();
36 output = [];
37
38 emit(indent, ...parts) {
39 this.output.push(" ".repeat(indent) + parts.join(" "));
40 }
41
42 date(text) {
43 const day = DAYS.get(text);
44 if (day === undefined) throw new CommandError("invalid date " + text);
45 return day;
46 }
47
48 room(text) {
49 const r = this.rooms.get(Number(text));
50 if (!r) throw new CommandError("unknown room " + text);
51 return r;
52 }
53
54 reservation(id) {
55 const x = this.reservations.get(id);
56 if (!x) throw new CommandError("unknown reservation " + id);
57 return x;
58 }
59
60 status(x, allowed) {
61 if (!allowed.includes(x.status)) throw new CommandError("invalid status " + x.id + " " + x.status);
62 }
63
64 type(kind) {
65 if (!TYPES.includes(kind)) throw new CommandError("invalid type " + kind);
66 }
67
68 unused(id) {
69 if (this.used.has(id)) throw new CommandError("duplicate reservation " + id);
70 }
71
72 stay(a, b) {
73 const start = this.date(a);
74 const end = this.date(b);
75 if (start < this.today) throw new CommandError("past date " + a);
76 if (end <= start) throw new CommandError("invalid range " + a + " " + b);
77 if (end - start > 30) throw new CommandError("too long " + (end - start));
78 return [start, end];
79 }
80
81 period(a, b, future = false, limited = true) {
82 const start = this.date(a);
83 const end = this.date(b);
84 if (future && start < this.today) throw new CommandError("past date " + a);
85 if (end < start) throw new CommandError("invalid range " + a + " " + b);
86 if (limited && end - start + 1 > 31) throw new CommandError("too long " + (end - start + 1));
87 return [start, end];
88 }
89
90 sortedRooms() {
91 return [...this.rooms.values()].sort((a, b) => a.number - b.number);
92 }
93
94 rate(r, day) {
95 const percent = this.seasons.findLast(season => season.start <= day && day <= season.end)?.percent ?? 100n;
96 return r.base * percent / 100n;
97 }
98
99 price(r, start, end) {
100 const nights = [];
101 for (let day = start; day < end; day++) nights.push({day, room: r.number, rate: this.rate(r, day)});
102 return nights;
103 }
104
105 free(r, day, except = null) {
106 const night = r.occupied.get(day);
107 return !r.blocks.has(day) && (!night || night.owner === except);
108 }
109
110 unavailable(r, start, end, except = null) {
111 for (let day = start; day < end; day++) {
112 if (!this.free(r, day, except)) return day;
113 }
114 return null;
115 }
116
117 findRoom(kind, start, end, except = null) {
118 return this.sortedRooms().find(r => r.kind === kind && this.unavailable(r, start, end, except) === null) ?? null;
119 }
120
121 inHouse(room) {
122 return [...this.reservations.values()].find(x => x.status === "inhouse" && x.room === room);
123 }
124
125 release(x) {
126 for (const n of x.nights) this.rooms.get(n.room).occupied.delete(n.day);
127 }
128
129 occupy(x) {
130 for (const n of x.nights) this.rooms.get(n.room).occupied.set(n.day, {owner: x.id, rate: n.rate});
131 }
132
133 create(id, guest, r, start, end, indent = 0) {
134 const x = {id, guest, room: r.number, start, end, nights: this.price(r, start, end),
135 status: "booked", extras: [], deposit: 0n, paid: 0n, posting: null, fee: 0n, penalty: null, bill: null};
136 this.reservations.set(id, x);
137 this.used.add(id);
138 this.occupy(x);
139 this.emit(indent, "BOOK", id, guest, "room", r.number, r.kind, dateText(start), "->", dateText(end),
140 "nights", end - start, "total", money(lodging(x)));
141 }
142
143 guestPaid(guest) {
144 return [...this.reservations.values()].filter(x => x.guest === guest).reduce((v, x) => v + x.paid, 0n);
145 }
146
147 tier(guest) {
148 const points = this.guestPaid(guest) / 100n;
149 return points >= 2000n ? {points, name: "GOLD", percent: 10n} :
150 points >= 500n ? {points, name: "SILVER", percent: 5n} : {points, name: "BRONZE", percent: 0n};
151 }
152
153 bill(x) {
154 const l = lodging(x);
155 const discount = l * this.tier(x.guest).percent / 100n;
156 const penalty = x.penalty?.rate ?? 0n;
157 const e = extras(x);
158 const subtotal = l - discount + penalty + e;
159 const tax = subtotal / 10n;
160 return {lodging: l, discount, penalty, extras: e, tax, total: subtotal + tax};
161 }
162
163 deposit(x, amount) {
164 return "deposit " + money(x.deposit) + (amount >= x.deposit ? " due " + money(amount - x.deposit) : " refund " + money(x.deposit - amount));
165 }
166
167 checkout(x, day, indent = 0) {
168 if (day < x.end) {
169 x.penalty = x.nights.find(n => n.day === day);
170 for (const n of x.nights.filter(n => n.day >= day)) this.rooms.get(n.room).occupied.delete(n.day);
171 x.nights = x.nights.filter(n => n.day < day);
172 x.end = day;
173 }
174 const b = this.bill(x);
175 x.bill = b;
176 x.paid = b.total;
177 x.status = "checkedout";
178 x.posting = day;
179 this.rooms.get(x.room).clean = false;
180 this.emit(indent, "CHECKOUT", x.id, "nights", x.nights.length, "lodging", money(b.lodging),
181 "discount", money(b.discount), "penalty", money(b.penalty), "extras", money(b.extras),
182 "tax", money(b.tax), "total", money(b.total), this.deposit(x, b.total));
183 }
184
185 processQueue() {
186 const remaining = [];
187 for (const w of this.queue) {
188 if (w.expiry < this.today) {
189 this.emit(2, "WAITLIST", w.id, "expired");
190 } else {
191 const r = this.findRoom(w.kind, w.start, w.end);
192 if (r) this.create(w.id, w.guest, r, w.start, w.end, 2);
193 else remaining.push(w);
194 }
195 }
196 this.queue = remaining;
197 }
198
199 metrics(day) {
200 let available = 0;
201 let occupied = 0;
202 let revenue = 0n;
203 for (const r of this.rooms.values()) {
204 if (!r.blocks.has(day)) available++;
205 const night = r.occupied.get(day);
206 if (night) {
207 occupied++;
208 revenue += night.rate;
209 }
210 }
211 return {available, occupied, revenue};
212 }
213
214 metricsText(m) {
215 return "sold " + m.occupied + "/" + m.available + " rate " + (m.available ? Math.floor(m.occupied * 100 / m.available) + "%" : "n/a") +
216 " revenue " + money(m.revenue) + " adr " + (m.occupied ? money(m.revenue / BigInt(m.occupied)) : "n/a") +
217 " revpar " + (m.available ? money(m.revenue / BigInt(m.available)) : "n/a");
218 }
219
220 occupancyText(rooms, day) {
221 const blocked = rooms.filter(r => r.blocks.has(day)).length;
222 const occupied = rooms.filter(r => r.occupied.has(day)).length;
223 const available = rooms.length - blocked;
224 return "rooms " + rooms.length + " blocked " + blocked + " occupied " + occupied +
225 " rate " + (available ? Math.floor(occupied * 100 / available) + "%" : "n/a");
226 }
227
228 invoice(x) {
229 this.emit(0, "INVOICE", x.id, x.guest, "room", x.room, this.rooms.get(x.room).kind, x.status);
230 if (x.status === "cancelled" || x.status === "noshow") {
231 this.emit(2, x.status === "cancelled" ? "cancellation" : "noshow", "fee", money(x.fee));
232 this.emit(2, "total", money(x.fee));
233 this.emit(2, this.deposit(x, x.fee));
234 return;
235 }
236 for (const n of x.nights) this.emit(2, dateWeek(n.day), "room", n.room, money(n.rate));
237 const b = x.status === "checkedout" ? x.bill : this.bill(x);
238 this.emit(2, "lodging", money(b.lodging));
239 this.emit(2, "discount", money(b.discount));
240 if (x.penalty) this.emit(2, "penalty", dateWeek(x.penalty.day), money(b.penalty));
241 for (const e of x.extras) this.emit(2, "extra", e.item, money(e.amount));
242 this.emit(2, "tax", money(b.tax));
243 this.emit(2, "total", money(b.total));
244 this.emit(2, this.deposit(x, b.total));
245 }

247 command(cmd, a) {
248 switch (cmd) {
249 case "ROOM": {
250 const number = Number(a[0]);
251 if (this.rooms.has(number)) throw new CommandError("duplicate room " + a[0]);
252 this.type(a[1]);
253 const base = cents(a[2]);
254 this.rooms.set(number, {number, kind: a[1], base, clean: true, blocks: new Set(), occupied: new Map()});
255 this.emit(0, "ROOM", number, a[1], money(base));
256 break;
257 }
258 case "RATE": {
259 const r = this.room(a[0]);
260 const old = r.base;
261 r.base = cents(a[1]);
262 this.emit(0, "RATE", r.number, money(old), "->", money(r.base));
263 break;
264 }
265 case "SEASON": {
266 const [start, end] = this.period(a[0], a[1], false, false);
267 this.seasons.push({start, end, percent: BigInt(a[2])});
268 this.emit(0, "SEASON", this.seasons.length, a[0], "->", a[1], a[2] + "%");
269 break;
270 }
271 case "BLOCK":
272 case "UNBLOCK": {
273 const r = this.room(a[0]);
274 const [start, end] = this.period(a[1], a[2], true);
275 if (cmd === "BLOCK") {
276 for (let day = start; day <= end; day++) {
277 if (r.occupied.has(day)) throw new CommandError("room unavailable " + r.number + " " + dateText(day));
278 }
279 }
280 const nights = new Set(Array.from({length: end - start + 1}, (_, i) => start + i));
281 const previous = r.blocks.size;
282 r.blocks = cmd === "BLOCK" ? r.blocks.union(nights) : r.blocks.difference(nights);
283 const count = Math.abs(r.blocks.size - previous);
284 this.emit(0, cmd, r.number, a[1], "->", a[2], cmd === "BLOCK" ? "added" : "removed", count);
285 break;
286 }
287 case "CLEAN": {
288 const r = this.room(a[0]);
289 if (r.clean) throw new CommandError("already clean " + r.number);
290 r.clean = true;
291 this.emit(0, "CLEAN", r.number);
292 break;
293 }
294 case "TODAY": {
295 const day = this.date(a[0]);
296 if (day < this.today) throw new CommandError("past date " + a[0]);
297 this.emit(0, "TODAY", dateWeek(day));
298 const checkouts = [...this.reservations.values()].filter(x => x.status === "inhouse" && x.end < day);
299 checkouts.sort((a, b) => a.end - b.end || lex(a.id, b.id));
300 for (const x of checkouts) this.checkout(x, x.end, 2);
301 const noshows = [...this.reservations.values()].filter(x => x.status === "booked" && x.start < day).sort(byArrival);
302 for (const x of noshows) {
303 x.fee = x.nights[0].rate;
304 x.paid = x.fee;
305 x.status = "noshow";
306 x.posting = x.start;
307 this.release(x);
308 this.emit(2, "NOSHOW", x.id, x.guest, "fee", money(x.fee), this.deposit(x, x.fee));
309 }
310 this.today = day;
311 break;
312 }
313 case "BOOK":
314 case "BOOKROOM":
315 case "WAITLIST": {
316 this.unused(a[0]);
317 let specific = null;
318 if (cmd === "BOOKROOM") specific = this.room(a[2]);
319 else this.type(a[2]);
320 const [start, end] = this.stay(a[3], a[4]);
321 const r = specific ?? this.findRoom(a[2], start, end);
322 if (specific) {
323 const day = this.unavailable(specific, start, end);
324 if (day !== null) throw new CommandError("room unavailable " + specific.number + " " + dateText(day));
325 }
326 if (r) {
327 this.create(a[0], a[1], r, start, end);
328 } else if (cmd === "WAITLIST") {
329 this.queue.push({id: a[0], guest: a[1], kind: a[2], start, end, expiry: start});
330 this.used.add(a[0]);
331 this.emit(0, "WAITLIST", a[0], "queued", "position", this.queue.length);
332 } else {
333 throw new CommandError("no availability " + a[2] + " " + a[3] + " " + a[4]);
334 }
335 break;
336 }
337 case "MODIFY": {
338 const x = this.reservation(a[0]);
339 this.status(x, ["booked"]);
340 const [start, end] = this.stay(a[1], a[2]);
341 const old = this.rooms.get(x.room);
342 const kept = this.unavailable(old, start, end, x.id) === null;
343 const r = kept ? old : this.findRoom(old.kind, start, end, x.id);
344 if (!r) throw new CommandError("no availability " + old.kind + " " + a[1] + " " + a[2]);
345 this.release(x);
346 x.room = r.number;
347 x.start = start;
348 x.end = end;
349 x.nights = this.price(r, start, end);
350 this.occupy(x);
351 this.emit(0, "MODIFY", x.id, "room", r.number, kept ? "kept" : "reassigned", a[1], "->", a[2],
352 "nights", end - start, "total", money(lodging(x)));
353 break;
354 }
355 case "CANCEL": {
356 const i = this.queue.findIndex(w => w.id === a[0]);
357 if (i >= 0) {
358 this.queue.splice(i, 1);
359 this.emit(0, "CANCEL", a[0], "waitlist", "removed");
360 break;
361 }
362 const x = this.reservation(a[0]);
363 this.status(x, ["booked"]);
364 const days = x.start - this.today;
365 x.fee = days >= 7 ? 0n : days >= 1 ? x.nights[0].rate / 2n : x.nights[0].rate;
366 x.paid = x.fee;
367 x.status = "cancelled";
368 x.posting = this.today;
369 this.release(x);
370 this.emit(0, "CANCEL", x.id, "days", days, "fee", money(x.fee), this.deposit(x, x.fee));
371 break;
372 }
373 case "CHECKIN": {
374 const x = this.reservation(a[0]);
375 this.status(x, ["booked"]);
376 if (x.start !== this.today) throw new CommandError("not arrival date " + x.id + " " + dateText(x.start));
377 const occupant = this.inHouse(x.room);
378 if (occupant) throw new CommandError("room occupied " + x.room + " " + occupant.id);
379 if (!this.rooms.get(x.room).clean) throw new CommandError("room dirty " + x.room);
380 x.status = "inhouse";
381 this.emit(0, "CHECKIN", x.id, x.guest, "room", x.room, "until", dateText(x.end));
382 break;
383 }
384 case "MOVE": {
385 const x = this.reservation(a[0]);
386 this.status(x, ["inhouse"]);
387 const r = this.room(a[1]);
388 if (r.number === x.room) throw new CommandError("same room " + r.number);
389 const remaining = x.nights.filter(n => n.day >= this.today);
390 if (!remaining.length) throw new CommandError("no remaining nights " + x.id);
391 for (const n of remaining) {
392 if (!this.free(r, n.day, x.id)) throw new CommandError("room unavailable " + r.number + " " + dateText(n.day));
393 }
394 const occupant = this.inHouse(r.number);
395 if (occupant) throw new CommandError("room occupied " + r.number + " " + occupant.id);
396 if (!r.clean) throw new CommandError("room dirty " + r.number);
397 const old = x.room;
398 for (const n of remaining) {
399 this.rooms.get(n.room).occupied.delete(n.day);
400 n.room = r.number;
401 r.occupied.set(n.day, {owner: x.id, rate: n.rate});
402 }
403 this.emit(0, "MOVE", x.id, old, "->", r.number, "nights", remaining.length);
404 this.rooms.get(old).clean = false;
405 x.room = r.number;
406 break;
407 }
408 case "EXTEND": {
409 const x = this.reservation(a[0]);
410 this.status(x, ["inhouse"]);
411 const end = this.date(a[1]);
412 if (end <= x.end) throw new CommandError("invalid range " + dateText(x.end) + " " + a[1]);
413 if (end - x.start > 30) throw new CommandError("too long " + (end - x.start));
414 const r = this.rooms.get(x.room);
415 const unavailable = this.unavailable(r, x.end, end, x.id);
416 if (unavailable !== null) throw new CommandError("room unavailable " + r.number + " " + dateText(unavailable));
417 x.nights.push(...this.price(r, x.end, end));
418 x.end = end;
419 this.occupy(x);
420 this.emit(0, "EXTEND", x.id, "until", a[1], "nights", x.nights.length, "total", money(lodging(x)));
421 break;
422 }
423 case "CHARGE":
424 case "PAY": {
425 const x = this.reservation(a[0]);
426 this.status(x, cmd === "CHARGE" ? ["inhouse"] : ["booked", "inhouse"]);
427 const amount = cents(a[1]);
428 if (amount === 0n) throw new CommandError("invalid amount");
429 if (cmd === "CHARGE") {
430 x.extras.push({item: a[2], amount});
431 this.emit(0, "CHARGE", x.id, a[2], money(amount), "extras", money(extras(x)));
432 } else {
433 x.deposit += amount;
434 this.emit(0, "PAY", x.id, money(amount), "deposit", money(x.deposit));
435 }
436 break;
437 }
438 case "CHECKOUT": {
439 const x = this.reservation(a[0]);
440 this.status(x, ["inhouse"]);
441 this.checkout(x, this.today);
442 break;
443 }
444 case "INVOICE":
445 this.invoice(this.reservation(a[0]));
446 break;
447 case "AVAIL": {
448 this.type(a[0]);
449 const [start, end] = this.period(a[1], a[2]);
450 this.emit(0, "AVAIL", a[0], a[1], "->", a[2]);
451 for (let day = start; day <= end; day++) {
452 const rooms = this.sortedRooms().filter(r => r.kind === a[0] && this.free(r, day));
453 this.emit(2, dateWeek(day), "free", rooms.length, ...rooms.map(r => r.number));
454 }
455 break;
456 }
457 case "OCCUPANCY": {
458 const day = this.date(a[0]);
459 this.emit(0, "OCCUPANCY", dateWeek(day), this.occupancyText([...this.rooms.values()], day));
460 for (const kind of TYPES) {
461 const rooms = [...this.rooms.values()].filter(r => r.kind === kind);
462 if (rooms.length) this.emit(2, kind, this.occupancyText(rooms, day));
463 }
464 break;
465 }
466 case "STATS": {
467 const [start, end] = this.period(a[0], a[1]);
468 this.emit(0, "STATS", a[0], "->", a[1]);
469 const total = {available: 0, occupied: 0, revenue: 0n};
470 for (let day = start; day <= end; day++) {
471 const m = this.metrics(day);
472 this.emit(2, dateWeek(day), this.metricsText(m));
473 total.available += m.available;
474 total.occupied += m.occupied;
475 total.revenue += m.revenue;
476 }
477 this.emit(2, "total", this.metricsText(total));
478 break;
479 }
480 case "ROOMS": {
481 this.emit(0, "ROOMS", dateWeek(this.today), "rooms", this.rooms.size);
482 for (const r of this.sortedRooms()) {
483 const occupant = this.inHouse(r.number);
484 const situation = occupant ? "occupied " + occupant.id : r.blocks.has(this.today) ? "blocked" : "vacant";
485 const next = [...this.reservations.values()].filter(x => x.status === "booked" && x.room === r.number).sort(byArrival)[0];
486 this.emit(2, r.number, r.kind, "base", money(r.base), "tonight", money(this.rate(r, this.today)), r.clean ? "clean" : "dirty", situation,
487 ...(next ? ["next", next.id, dateText(next.start)] : []));
488 }
489 break;
490 }
491 case "LIST": {
492 const targets = [...this.reservations.values()].filter(x => x.status === a[0]).sort(byArrival);
493 this.emit(0, "LIST", a[0], targets.length);
494 for (const x of targets) this.emit(2, x.id, x.guest, "room", x.room, dateText(x.start), "->", dateText(x.end));
495 break;
496 }
497 case "QUEUE":
498 this.emit(0, "QUEUE", this.queue.length);
499 for (let i = 0; i < this.queue.length; i++) {
500 const w = this.queue[i];
501 this.emit(2, i + 1, w.id, w.guest, w.kind, dateText(w.start), "->", dateText(w.end));
502 }
503 break;
504 case "SCHEDULE": {
505 const day = this.date(a[0]);
506 const arrivals = [...this.reservations.values()].filter(x => x.start === day && x.status !== "cancelled").sort((a, b) => lex(a.id, b.id));
507 const departures = [...this.reservations.values()].filter(x => x.end === day && ["booked", "inhouse", "checkedout"].includes(x.status)).sort((a, b) => lex(a.id, b.id));
508 this.emit(0, "SCHEDULE", dateWeek(day), "arrivals", arrivals.length, "departures", departures.length);
509 for (const x of arrivals) this.emit(2, "arrival", x.id, x.guest, "room", x.room, x.status);
510 for (const x of departures) this.emit(2, "departure", x.id, x.guest, "room", x.room, x.status);
511 break;
512 }
513 case "GUEST": {
514 const targets = [...this.reservations.values()].filter(x => x.guest === a[0]);
515 if (!targets.length) throw new CommandError("unknown guest " + a[0]);
516 this.emit(0, "GUEST", a[0]);
517 for (const x of targets) this.emit(2, x.id, "room", x.room, this.rooms.get(x.room).kind, dateText(x.start), "->", dateText(x.end), x.status,
518 ["booked", "inhouse"].includes(x.status) ? "deposit" : "paid", money(["booked", "inhouse"].includes(x.status) ? x.deposit : x.paid));
519 const tier = this.tier(a[0]);
520 this.emit(2, "total", "paid", money(this.guestPaid(a[0])), "points", String(tier.points), "tier", tier.name);
521 break;
522 }
523 case "REVENUE": {
524 const [start, end] = this.period(a[0], a[1], false, false);
525 const targets = [...this.reservations.values()].filter(x => x.posting !== null && x.posting >= start && x.posting <= end);
526 const totals = {lodging: 0n, discount: 0n, penalty: 0n, extras: 0n, tax: 0n, cancellation: 0n, noshow: 0n};
527 let checkouts = 0;
528 let cancellations = 0;
529 let noshows = 0;
530 for (const x of targets) {
531 if (x.status === "checkedout") {
532 checkouts++;
533 for (const key of ["lodging", "discount", "penalty", "extras", "tax"]) totals[key] += x.bill[key];
534 } else if (x.status === "cancelled") {
535 cancellations++;
536 totals.cancellation += x.fee;
537 } else if (x.status === "noshow") {
538 noshows++;
539 totals.noshow += x.fee;
540 }
541 }
542 this.emit(0, "REVENUE", a[0], "->", a[1]);
543 for (const key of ["lodging", "discount", "penalty", "extras", "tax", "cancellation", "noshow"]) this.emit(2, key, money(totals[key]));
544 const total = totals.lodging - totals.discount + totals.penalty + totals.extras + totals.tax + totals.cancellation + totals.noshow;
545 this.emit(2, "total", money(total));
546 this.emit(2, "checkouts", checkouts, "cancellations", cancellations, "noshows", noshows);
547 break;
548 }
549 }
