590 elif action == "CANCEL"

604 elif state.rooms[room].inhouse >= 0

606 elif state.rooms[room].dirty

608 else

610 state.rooms[room].inhouse = index
611 emit("CHECKIN " + id + " " + entry.guest + " room " + string(state.rooms[room].number) + " until " + date_text(entry.departure))
612 elif action == "MOVE"

618 fail("same room " + words[2])
619 return

621 if remaining <= 0

623 return

626 fail("room unavailable " + words[2] + " " + date_text(day))
627 elif state.rooms[room].inhouse >= 0
628 fail("room occupied " + words[2] + " " + state.bookings[state.rooms[room].inhouse].id)
629 elif state.rooms[room].dirty
630 fail("room dirty " + words[2])
631 else

636 state.occupied[room][night] = index

639 state.rooms[entry.room].inhouse = -1
640 state.rooms[room].inhouse = index
641 state.bookings[index].room = room

657 state.bookings[index].nights = last - entry.arrival

663 elif action == "PAY"

672 else
673 state.charges[entry.extra_tail].next = charge

675 state.bookings[index].extras += amount
676 emit("CHARGE " + id + " " + words[3] + " " + money(amount) + " extras " + money(state.bookings[index].extras))
677 elif action == "CHECKOUT"

679 else
680 emit("INVOICE " + id + " " + entry.guest + " room " + string(state.rooms[entry.room].number) + " " + type_name(state.rooms[entry.room].kind) + " " + status_name(entry.status))

685 else

687 emit(" " + date_weekday(entry.arrival + k) + " room " + string(state.rooms[state.night_rooms[index][k]].number) + " " + money(state.night_rates[index][k]))
688 bill := if entry.status == 2 then entry.bill else calculate_bill(state, index)
