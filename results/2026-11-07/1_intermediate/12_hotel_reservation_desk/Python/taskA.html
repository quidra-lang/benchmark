12def leap(year):
13 return year % 4 == 0 and (year % 100 != 0 or year % 400 == 0)
14

16def month_lengths(year):
17 return (31, 29 if leap(year) else 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31)

20def date_number(text):
21 year, month, day = (int(part) for part in text.split("-"))
22 if not (2000 <= year <= 2099 and 1 <= month <= 12):
23 raise CommandError(f"invalid date {text}")
24 lengths = month_lengths(year)
25 if not 1 <= day <= lengths[month - 1]:
26 raise CommandError(f"invalid date {text}")
27 years = sum(366 if leap(previous) else 365 for previous in range(2000, year))
28 return years + sum(lengths[:month - 1]) + day - 1

30
31def date_text(number):
32 year = 2000
33 while number >= (366 if leap(year) else 365):
34 number -= 366 if leap(year) else 365
35 year += 1
36 month = 1
37 for length in month_lengths(year):
38 if number < length:
39 break
40 number -= length
41 month += 1
42 return f"{year:04d}-{month:02d}-{number + 1:02d}"

45def date_weekday(number):
46 return date_text(number) + " " + WEEKDAYS[number % 7]

129 def reservation(self, identifier, status=None):
130 if identifier not in self.reservations:
131 raise CommandError(f"unknown reservation {identifier}")
132 reservation = self.reservations[identifier]
133 if status is not None and reservation.status != status:
134 raise CommandError(f"invalid status {identifier} {reservation.status}")

137 def stay(self, first, last):
138 arrival, departure = date_number(first), date_number(last)
139 if arrival < self.today:
140 raise CommandError(f"past date {first}")
141 if departure <= arrival:
142 raise CommandError(f"invalid range {first} {last}")
143 if departure - arrival > 30:
144 raise CommandError(f"too long {departure - arrival}")
145 return arrival, departure

147 def period(self, first, last, past_restricted=False, limited=True):
148 start, end = date_number(first), date_number(last)
149 if past_restricted and start < self.today:
150 raise CommandError(f"past date {first}")
151 if end < start:
152 raise CommandError(f"invalid range {first} {last}")
153 if limited and end - start + 1 > 31:
154 raise CommandError(f"too long {end - start + 1}")
155 return start, end

157 def night_rate(self, room, day):
158 percentage = next(
159 (value for start, end, value in reversed(self.seasons) if start <= day <= end),
160 100,
161 )
162 return room.rate * percentage // 100

164 def price(self, room, arrival, departure):
165 return [(day, room.number, self.night_rate(room, day)) for day in range(arrival, departure)]

167 def first_unavailable(self, room, arrival, departure, except_for=None):
168 for day in range(arrival, departure):
169 key = day, room.number
170 if key in self.blocks:
171 return day
172 occupied = self.occupied.get(key)
173 if occupied is not None and occupied[0] is not except_for:
174 return day
175 return None

177 def free_room(self, kind, arrival, departure, except_for=None):
178 for number in sorted(self.rooms):
179 room = self.rooms[number]
180 if room.kind == kind and self.first_unavailable(room, arrival, departure, except_for) is None:
181 return room
182 return None

211 def tier(self, guest):
212 points = self.guest_total(guest) // 100
213 return "GOLD" if points >= 2000 else "SILVER" if points >= 500 else "BRONZE"

215 def calculate_bill(self, reservation):
216 lodging = reservation.lodging()
217 percentage = {"GOLD": 10, "SILVER": 5, "BRONZE": 0}[self.tier(reservation.guest)]
218 discount = lodging * percentage // 100
219 penalty = 0 if reservation.penalty is None else reservation.penalty[1]
220 extras = sum(amount for _, amount in reservation.extras)
221 subtotal = lodging - discount + penalty + extras
222 tax = subtotal // 10
223 return lodging, discount, penalty, extras, tax, subtotal + tax

225 def deposit_parts(self, reservation, total):
226 if total >= reservation.deposit:
227 return "deposit", money(reservation.deposit), "due", money(total - reservation.deposit)
228 return "deposit", money(reservation.deposit), "refund", money(reservation.deposit - total)

230 def check_out(self, reservation, day, indent=0):
231 if day < reservation.departure:
232 rate = next(rate for date, _, rate in reservation.nights if date == day)
233 reservation.penalty = day, rate
234 self.release(reservation, day)
235 reservation.nights = [night for night in reservation.nights if night[0] < day]
236 reservation.departure = day
237 lodging, discount, penalty, extras, tax, total = self.calculate_bill(reservation)
238 reservation.bill = lodging, discount, penalty, extras, tax, total
239 reservation.paid = total
240 reservation.status = "checkedout"
241 reservation.posting = day
242 reservation.room.clean = False
243 self.emit(
244 "CHECKOUT", reservation.identifier, "nights", len(reservation.nights),
245 "lodging", money(lodging), "discount", money(discount), "penalty", money(penalty),
246 "extras", money(extras), "tax", money(tax), "total", money(total),
247 *self.deposit_parts(reservation, total), indent=indent,
248 )

282 def cmd_block(self, number, first, last):
283 room = self.require_room(number)
284 start, end = self.period(first, last, past_restricted=True)
285 for day in range(start, end + 1):
286 if (day, room.number) in self.occupied:
287 raise CommandError(f"room unavailable {number} {date_text(day)}")
288 nights = {(day, room.number) for day in range(start, end + 1)}
289 added = len(nights - self.blocks)
290 self.blocks.update(nights)
291 self.emit("BLOCK", number, first, "->", last, "added", added)

293 def cmd_unblock(self, number, first, last):
294 room = self.require_room(number)
295 start, end = self.period(first, last, past_restricted=True)
296 nights = {(day, room.number) for day in range(start, end + 1)}
297 removed = len(nights & self.blocks)
298 self.blocks.difference_update(nights)
299 self.emit("UNBLOCK", number, first, "->", last, "removed", removed)

327 reservation.status = "noshow"
328 reservation.posting = reservation.arrival
329 self.emit(
330 "NOSHOW", reservation.identifier, reservation.guest, "fee", money(reservation.fee),

416 def cmd_move(self, identifier, number):
417 reservation = self.reservation(identifier, "inhouse")
418 room = self.require_room(number)
419 previous = reservation.room
420 if room is previous:
421 raise CommandError(f"same room {number}")
422 remaining = [night for night in reservation.nights if night[0] >= self.today]
423 if not remaining:
424 raise CommandError(f"no remaining nights {identifier}")
425 unavailable = self.first_unavailable(room, remaining[0][0], reservation.departure, reservation)
426 if unavailable is not None:
427 raise CommandError(f"room unavailable {number} {date_text(unavailable)}")
428 occupant = self.inhouse(room)
429 if occupant is not None:
430 raise CommandError(f"room occupied {number} {occupant.identifier}")
431 if not room.clean:
432 raise CommandError(f"room dirty {number}")
433 self.release(reservation, self.today)
434 reservation.nights = [
435 (day, room.number if day >= self.today else old_room, rate)
436 for day, old_room, rate in reservation.nights
437 ]
438 self.occupy(reservation, [night for night in reservation.nights if night[0] >= self.today])
439 previous.clean = False
440 reservation.room = room
441 self.emit("MOVE", identifier, previous.number, "->", number, "nights", len(remaining))

606 self.emit("SCHEDULE", date_weekday(day), "arrivals", len(arrivals), "departures", len(departures))
607 for label, reservations in (("arrival", arrivals), ("departure", departures)):
608 for reservation in reservations:
609 self.emit(label, reservation.identifier, reservation.guest, "room", reservation.room.number, reservation.status, indent=2)

611 def cmd_guest(self, guest):
612 reservations = [r for r in self.reservations.values() if r.guest == guest]
613 if not reservations:
614 raise CommandError(f"unknown guest {guest}")
615 self.emit("GUEST", guest)
616 for reservation in reservations:
617 amount = ("deposit", reservation.deposit) if reservation.status in ("booked", "inhouse") else ("paid", reservation.paid)
618 self.emit(
619 reservation.identifier, "room", reservation.room.number, reservation.room.kind,
620 date_text(reservation.arrival), "->", date_text(reservation.departure), reservation.status,
621 amount[0], money(amount[1]), indent=2,
622 )
623 total = self.guest_total(guest)
624 self.emit("total paid", money(total), "points", total // 100, "tier", self.tier(guest), indent=2)

626 def cmd_revenue(self, first, last):
627 start, end = self.period(first, last, limited=False)
628 posted = [r for r in self.reservations.values() if r.posting is not None and start <= r.posting <= end]
629 checked_out = [r for r in posted if r.status == "checkedout"]
630 cancelled = [r for r in posted if r.status == "cancelled"]
631 absent = [r for r in posted if r.status == "noshow"]
632 lodging, discount, penalty, extras, tax = [sum(r.bill[index] for r in checked_out) for index in range(5)]
633 cancellations = sum(r.fee for r in cancelled)
634 noshows = sum(r.fee for r in absent)
635 self.emit("REVENUE", first, "->", last)
636 for label, amount in (
637 ("lodging", lodging), ("discount", discount), ("penalty", penalty), ("extras", extras),
638 ("tax", tax), ("cancellation", cancellations), ("noshow", noshows),
639 ):
640 self.emit(label, money(amount), indent=2)
641 self.emit("total", money(lodging - discount + penalty + extras + tax + cancellations + noshows), indent=2)
642 self.emit("checkouts", len(checked_out), "cancellations", len(cancelled), "noshows", len(absent), indent=2)

644 def execute(self, line):
645 command, *arguments = line.split()
646 try:
647 getattr(self, "cmd_" + command.lower())(*arguments)
648 except CommandError as error:
649 self.emit("ERROR", str(error))
650 self.process_waitlist()
