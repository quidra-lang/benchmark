99 private static bool Leap(int year)
100 => year % 4 == 0 && (year % 100 != 0 || year % 400 == 0);
101
102 private static int DaysInMonth(int year, int month)
103 => month == 2 && Leap(year) ? 29 : NormalMonths[month - 1];
104
105 private static int DaysInYear(int year) => Leap(year) ? 366 : 365;

107 private static int DateNumber(string text)
108 {
109 int year = int.Parse(text.Substring(0, 4), CultureInfo.InvariantCulture);
110 int month = int.Parse(text.Substring(5, 2), CultureInfo.InvariantCulture);
111 int day = int.Parse(text.Substring(8, 2), CultureInfo.InvariantCulture);
112 if (year < 2000 || year > 2099 || month < 1 || month > 12 ||
113 day < 1 || day > DaysInMonth(year, month))
114 throw new CommandError("invalid date " + text);
115
116 int number = 0;
117 for (int previous = 2000; previous < year; ++previous)
118 number += DaysInYear(previous);
119 for (int priorMonth = 1; priorMonth < month; ++priorMonth)
120 number += DaysInMonth(year, priorMonth);
121 return number + day - 1;
122 }
123
124 private static string DateText(int number)
125 {
126 int year = 2000;
127 while (number >= DaysInYear(year))
128 {
129 number -= DaysInYear(year);
130 ++year;
131 }
132 int month = 1;
133 while (number >= DaysInMonth(year, month))
134 {
135 number -= DaysInMonth(year, month);
136 ++month;
137 }
138 return $"{year:D4}-{month:D2}-{number + 1:D2}";
139 }

189 private (int Arrival, int Departure) Stay(string first, string last)
190 {
191 int arrival = DateNumber(first);
192 int departure = DateNumber(last);
193 if (arrival < today)
194 throw new CommandError("past date " + first);
195 if (departure <= arrival)
196 throw new CommandError($"invalid range {first} {last}");
197 if (departure - arrival > 30)
198 throw new CommandError("too long " + (departure - arrival));
199 return (arrival, departure);

202 private (int Start, int End) Period(string first, string last,
203 bool pastRestricted = false, bool limited = true)
204 {
205 int start = DateNumber(first);
206 int end = DateNumber(last);
207 if (pastRestricted && start < today)
208 throw new CommandError("past date " + first);
209 if (end < start)
210 throw new CommandError($"invalid range {first} {last}");
211 if (limited && end - start + 1 > 31)
212 throw new CommandError("too long " + (end - start + 1));
213 return (start, end);
214 }

216 private long NightRate(Room room, int day)
217 {
218 int percentage = 100;
219 for (int i = seasons.Count - 1; i >= 0; --i)
220 {
221 Season season = seasons[i];
222 if (season.Start <= day && day <= season.End)
223 {
224 percentage = season.Percentage;
225 break;
226 }
227 }
228 return room.Rate * percentage / 100;
229 }

239 private int? FirstUnavailable(Room room, int arrival, int departure,
240 Reservation? exceptFor = null)
241 {
242 for (int day = arrival; day < departure; ++day)
243 {
244 var key = (day, room.Number);
245 if (blocks.Contains(key))
246 return day;
247 if (occupied.TryGetValue(key, out var holder) &&
248 !ReferenceEquals(holder.Reservation, exceptFor))
249 return day;
250 }
251 return null;
252 }

254 private Room? FreeRoom(string kind, int arrival, int departure,
255 Reservation? exceptFor = null)
256 {
257 foreach (Room room in rooms.Values)
258 if (room.Kind == kind &&
259 FirstUnavailable(room, arrival, departure, exceptFor) == null)
260 return room;
261 return null;
262 }

303 private string Tier(string guest)
304 {
305 long points = GuestTotal(guest) / 100;

309 private Bill CalculateBill(Reservation reservation)
310 {
311 long lodging = reservation.Lodging();
312 int percentage = Tier(reservation.Guest) switch
313 {
314 "GOLD" => 10,
315 "SILVER" => 5,
316 _ => 0
317 };
318 long discount = lodging * percentage / 100;
319 long penalty = reservation.Penalty?.Rate ?? 0;
320 long extras = reservation.Extras.Sum(charge => charge.Amount);
321 long subtotal = lodging - discount + penalty + extras;
322 long tax = subtotal / 10;
323 return new Bill(lodging, discount, penalty, extras, tax, subtotal + tax);
324 }

331 private void Checkout(Reservation reservation, int day, bool indented = false)
332 {
333 if (day < reservation.Departure)
334 {
335 long penaltyRate = reservation.Nights.First(night => night.Day == day).Rate;
336 reservation.Penalty = (day, penaltyRate);
337 Release(reservation, day);
338 reservation.Nights = reservation.Nights.Where(night => night.Day < day).ToList();
339 reservation.Departure = day;
340 }
341 Bill bill = CalculateBill(reservation);
342 reservation.FinalBill = bill;
343 reservation.Paid = bill.Total;
344 reservation.Status = "checkedout";
345 reservation.Posting = day;
346 reservation.Room.Clean = false;
347 string message = $"CHECKOUT {reservation.Id} nights {reservation.Nights.Count} " +
348 $"lodging {Money(bill.Lodging)} discount {Money(bill.Discount)} " +
349 $"penalty {Money(bill.Penalty)} extras {Money(bill.Extras)} " +
350 $"tax {Money(bill.Tax)} total {Money(bill.Total)} " +
351 DepositText(reservation, bill.Total);
352 if (indented)
353 EmitIndented(message);
354 else
355 Emit(message);
356 }

435 private void CmdToday(string text)
436 {
437 int day = DateNumber(text);
438 if (day < today)
439 throw new CommandError("past date " + text);
440 Emit("TODAY", DateWeekday(day));
441 foreach (Reservation r in reservations
442 .Where(r => r.Status == "inhouse" && r.Departure < day)
443 .OrderBy(r => r.Departure).ThenBy(r => r.Id, StringComparer.Ordinal)
444 .ToList())

446
447 foreach (Reservation r in reservations
448 .Where(r => r.Status == "booked" && r.Arrival < day)
449 .OrderBy(r => r.Arrival).ThenBy(r => r.Id, StringComparer.Ordinal)
450 .ToList())
451 {
452 Release(r);
453 r.Fee = r.FirstRate();
454 r.Paid = r.Fee;
455 r.Status = "noshow";
456 r.Posting = r.Arrival;
457 EmitIndented($"NOSHOW {r.Id} {r.Guest} fee {Money(r.Fee)} {DepositText(r, r.Fee)}");
458 }
459 today = day;

501 {

562 {
563 Reservation booking = RequireReservation(id, "inhouse");
564 Room room = RequireRoom(number);
565 Room previous = booking.Room;
566 if (ReferenceEquals(room, previous))
567 throw new CommandError("same room " + number);
568 List<Night> remaining = booking.Nights.Where(night => night.Day >= today).ToList();
569 if (remaining.Count == 0)
570 throw new CommandError("no remaining nights " + id);
571 int? unavailable = FirstUnavailable(room, remaining[0].Day, booking.Departure, booking);
572 if (unavailable != null)
573 throw new CommandError($"room unavailable {number} {DateText(unavailable.Value)}");
574 Reservation? occupant = Inhouse(room);
575 if (occupant != null)
576 throw new CommandError($"room occupied {number} {occupant.Id}");
577 if (!room.Clean)
578 throw new CommandError("room dirty " + number);
579 Release(booking, today);
580 foreach (Night night in remaining)
581 night.Room = room.Number;
582 Occupy(booking, remaining);
583 previous.Clean = false;
584 booking.Room = room;
585 Emit("MOVE", id, previous.Number, "->", room.Number, "nights", remaining.Count);
586 }

690 int day = DateNumber(text);
691 Emit($"OCCUPANCY {DateWeekday(day)} {OccupancyParts(rooms.Values.ToList(), day)}");
692 foreach (string kind in RoomTypes)
693 {
694 List<Room> choices = rooms.Values.Where(room => room.Kind == kind).ToList();
695 if (choices.Count > 0)
696 EmitIndented($"{kind} {OccupancyParts(choices, day)}");
697 }
698 }

700 private (int Available, int Occupied, long Revenue) DailyValues(int day)
701 {
702 int available = rooms.Count - rooms.Keys.Count(number => blocks.Contains((day, number)));
703 int taken = 0;
704 long revenue = 0;
705 foreach (int number in rooms.Keys)
706 {
707 if (occupied.TryGetValue((day, number), out var holder))
708 {
709 ++taken;
710 revenue += holder.Rate;
711 }
712 }
713 return (available, taken, revenue);
714 }
715
716 private static string MetricsParts(int available, int taken, long revenue)
717 {
718 string percent = available > 0 ? $"{taken * 100 / available}%" : "n/a";
719 string average = taken > 0 ? Money(revenue / taken) : "n/a";
720 string revpar = available > 0 ? Money(revenue / available) : "n/a";
721 return $"sold {taken}/{available} rate {percent} revenue {Money(revenue)} " +
722 $"adr {average} revpar {revpar}";
723 }

725 private void CmdStats(string first, string last)
726 {
727 var (start, end) = Period(first, last);
728 int sumAvailable = 0, sumTaken = 0;
729 long sumRevenue = 0;
730 Emit("STATS", first, "->", last);
731 for (int day = start; day <= end; ++day)
732 {
733 var (available, taken, revenue) = DailyValues(day);
734 sumAvailable += available;
735 sumTaken += taken;
736 sumRevenue += revenue;
737 EmitIndented($"{DateWeekday(day)} {MetricsParts(available, taken, revenue)}");
738 }
739 EmitIndented($"total {MetricsParts(sumAvailable, sumTaken, sumRevenue)}");
740 }

742 private void CmdRooms()
743 {
744 Emit("ROOMS", DateWeekday(today), "rooms", rooms.Count);
745 foreach (Room room in rooms.Values)
746 {
747 string line = $"{room.Number} {room.Kind} base {Money(room.Rate)} " +
748 $"tonight {Money(NightRate(room, today))} " +
749 (room.Clean ? "clean" : "dirty");
750 Reservation? occupant = Inhouse(room);
751 if (occupant != null)
752 line += " occupied " + occupant.Id;
753 else if (blocks.Contains((today, room.Number)))
754 line += " blocked";
755 else
756 line += " vacant";
757
758 Reservation? upcoming = reservations
759 .Where(r => r.Status == "booked" && ReferenceEquals(r.Room, room))
760 .OrderBy(r => r.Arrival).FirstOrDefault();
761 if (upcoming != null)
762 line += $" next {upcoming.Id} {DateText(upcoming.Arrival)}";
763 EmitIndented(line);
764 }
765 }

788 private void CmdSchedule(string text)
789 {
790 int day = DateNumber(text);
791 var arrivals = reservations
792 .Where(r => r.Arrival == day && r.Status != "cancelled")
793 .OrderBy(r => r.Id, StringComparer.Ordinal).ToList();
794 var departures = reservations
795 .Where(r => r.Departure == day &&
796 (r.Status == "booked" || r.Status == "inhouse" || r.Status == "checkedout"))
797 .OrderBy(r => r.Id, StringComparer.Ordinal).ToList();
798 Emit("SCHEDULE", DateWeekday(day), "arrivals", arrivals.Count,
799 "departures", departures.Count);
800 foreach (Reservation booking in arrivals)
801 EmitIndented("arrival", booking.Id, booking.Guest,
802 "room", booking.Room.Number, booking.Status);
803 foreach (Reservation booking in departures)
804 EmitIndented("departure", booking.Id, booking.Guest,
805 "room", booking.Room.Number, booking.Status);
806 }

808 private void CmdGuest(string guest)
809 {
810 List<Reservation> entries = reservations.Where(r => r.Guest == guest).ToList();
811 if (entries.Count == 0)
812 throw new CommandError("unknown guest " + guest);
813 Emit("GUEST", guest);
814 foreach (Reservation booking in entries)
815 {
816 bool open = booking.Status == "booked" || booking.Status == "inhouse";
817 EmitIndented(booking.Id, "room", booking.Room.Number, booking.Room.Kind,
818 DateText(booking.Arrival), "->", DateText(booking.Departure),
819 booking.Status, open ? "deposit" : "paid",
820 Money(open ? booking.Deposit : booking.Paid));
821 }
822 long total = GuestTotal(guest);
823 EmitIndented("total paid", Money(total), "points", total / 100, "tier", Tier(guest));
824 }

826 private void CmdRevenue(string first, string last)
827 {
828 var (start, end) = Period(first, last, limited: false);
829 List<Reservation> posted = reservations
830 .Where(r => r.Posting != null && r.Posting >= start && r.Posting <= end).ToList();
831 List<Reservation> checkedOut = posted.Where(r => r.Status == "checkedout").ToList();
832 List<Reservation> cancelled = posted.Where(r => r.Status == "cancelled").ToList();
833 List<Reservation> absent = posted.Where(r => r.Status == "noshow").ToList();
834 long lodging = checkedOut.Sum(r => r.FinalBill!.Lodging);
835 long discount = checkedOut.Sum(r => r.FinalBill!.Discount);
836 long penalty = checkedOut.Sum(r => r.FinalBill!.Penalty);
837 long extras = checkedOut.Sum(r => r.FinalBill!.Extras);
838 long tax = checkedOut.Sum(r => r.FinalBill!.Tax);
839 long cancellations = cancelled.Sum(r => r.Fee);
840 long noshows = absent.Sum(r => r.Fee);
841 Emit("REVENUE", first, "->", last);
842 EmitIndented("lodging", Money(lodging));
843 EmitIndented("discount", Money(discount));
844 EmitIndented("penalty", Money(penalty));
845 EmitIndented("extras", Money(extras));
846 EmitIndented("tax", Money(tax));
847 EmitIndented("cancellation", Money(cancellations));
848 EmitIndented("noshow", Money(noshows));
849 EmitIndented("total", Money(lodging - discount + penalty + extras + tax + cancellations + noshows));
850 EmitIndented("checkouts", checkedOut.Count, "cancellations", cancelled.Count,
851 "noshows", absent.Count);
852 }

854 public void Execute(string line)
855 {
856 string[] words = line.Split(' ');
857 string command = words[0];
858 string[] args = words.Skip(1).ToArray();
859 try
860 {
861 switch (command)
862 {
863 case "ROOM": CmdRoom(args[0], args[1], args[2]); break;
864 case "RATE": CmdRate(args[0], args[1]); break;
865 case "SEASON": CmdSeason(args[0], args[1], args[2]); break;
866 case "BLOCK": CmdBlock(args[0], args[1], args[2]); break;
867 case "UNBLOCK": CmdUnblock(args[0], args[1], args[2]); break;
868 case "CLEAN": CmdClean(args[0]); break;
869 case "TODAY": CmdToday(args[0]); break;
870 case "BOOK": CmdBook(args[0], args[1], args[2], args[3], args[4]); break;
871 case "BOOKROOM": CmdBookroom(args[0], args[1], args[2], args[3], args[4]); break;
872 case "WAITLIST": CmdWaitlist(args[0], args[1], args[2], args[3], args[4]); break;
873 case "MODIFY": CmdModify(args[0], args[1], args[2]); break;
874 case "CANCEL": CmdCancel(args[0]); break;
875 case "CHECKIN": CmdCheckin(args[0]); break;
876 case "MOVE": CmdMove(args[0], args[1]); break;
877 case "EXTEND": CmdExtend(args[0], args[1]); break;
878 case "CHARGE": CmdCharge(args[0], args[1], args[2]); break;
879 case "PAY": CmdPay(args[0], args[1]); break;
880 case "CHECKOUT": CmdCheckout(args[0]); break;
881 case "INVOICE": CmdInvoice(args[0]); break;
882 case "AVAIL": CmdAvail(args[0], args[1], args[2]); break;
883 case "OCCUPANCY": CmdOccupancy(args[0]); break;
884 case "STATS": CmdStats(args[0], args[1]); break;
885 case "ROOMS": CmdRooms(); break;
886 case "LIST": CmdList(args[0]); break;
887 case "QUEUE": CmdQueue(); break;
888 case "SCHEDULE": CmdSchedule(args[0]); break;
889 case "GUEST": CmdGuest(args[0]); break;
890 case "REVENUE": CmdRevenue(args[0], args[1]); break;
891 }
892 }
893 catch (CommandError error)
894 {
895 Emit("ERROR", error.Message);
896 }
897 ProcessWaitlist();
898 }
899}
