4const TYPES: [&str; 3] = ["SINGLE", "DOUBLE", "SUITE"];
5const WEEKDAYS: [&str; 7] = ["SAT", "SUN", "MON", "TUE", "WED", "THU", "FRI"];
6
7fn leap(year: i32) -> bool {
8 year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
9}
10
11fn month_lengths(year: i32) -> [i32; 12] {
12 [31, if leap(year) { 29 } else { 28 }, 31, 30, 31, 30,
13 31, 31, 30, 31, 30, 31]
14}
15
16fn date_number(text: &str) -> Result<i32, String> {
17 let year: i32 = text[0..4].parse().unwrap_or(0);
18 let month: i32 = text[5..7].parse().unwrap_or(0);
19 let day: i32 = text[8..10].parse().unwrap_or(0);
20 if !(2000..=2099).contains(&year) || !(1..=12).contains(&month) {
21 return Err(format!("invalid date {text}"));
22 }
23 let lengths = month_lengths(year);
24 if day < 1 || day > lengths[(month - 1) as usize] {
25 return Err(format!("invalid date {text}"));
26 }
27 let mut number = 0;
28 for previous in 2000..year {
29 number += if leap(previous) { 366 } else { 365 };
30 }
31 for previous in 0..(month - 1) as usize {
32 number += lengths[previous];
33 }
34 Ok(number + day - 1)
35}
36
37fn date_text(mut number: i32) -> String {
38 let mut year = 2000;
39 loop {
40 let days = if leap(year) { 366 } else { 365 };
41 if number < days {
42 break;
43 }
44 number -= days;
45 year += 1;
46 }
47 let lengths = month_lengths(year);
48 let mut month = 0;
49 while number >= lengths[month] {
50 number -= lengths[month];
51 month += 1;
52 }
53 format!("{year:04}-{:02}-{:02}", month + 1, number + 1)
54}
55
56fn date_weekday(number: i32) -> String {
57 format!("{} {}", date_text(number), WEEKDAYS[number.rem_euclid(7) as usize])
58}
59
60fn cents(text: &str) -> i64 {
61 let (whole, fractional) = text.split_once('.').unwrap();
62 whole.parse::<i64>().unwrap() * 100 + fractional.parse::<i64>().unwrap()
63}
64
65fn money(amount: i64) -> String {
66 format!("{}.{:02}", amount / 100, amount % 100)
67}

139struct HotelDesk {
140 today: i32,
141 rooms: BTreeMap<i32, Room>,
142 seasons: Vec<Season>,

152impl HotelDesk {

156 rooms: BTreeMap::new(),
157 seasons: Vec::new(),
158 blocks: HashSet::new(),
159 occupied: HashMap::new(),

168 fn emit(&mut self, line: String, indent: bool) {
169 self.output.push(if indent { format!(" {line}") } else { line });

172 fn require_room(&self, number: i32) -> Result<&Room, String> {
173 self.rooms.get(&number).ok_or_else(|| format!("unknown room {number}"))
174 }
175
176 fn require_type(&self, kind: &str) -> Result<(), String> {
177 if TYPES.contains(&kind) {
178 Ok(())
179 } else {
180 Err(format!("invalid type {kind}"))
181 }
182 }
183
184 fn require_unused(&self, id: &str) -> Result<(), String> {
185 if self.used_ids.contains(id) {
186 Err(format!("duplicate reservation {id}"))
187 } else {
188 Ok(())
189 }
190 }
191
192 fn reservation(&self, id: &str, expected: Option<&str>) -> Result<&Reservation, String> {
193 let reservation = self.reservations.get(id)
194 .ok_or_else(|| format!("unknown reservation {id}"))?;
195 if let Some(status) = expected {
196 if reservation.status != status {
197 return Err(format!("invalid status {id} {}", reservation.status));
198 }
199 }
200 Ok(reservation)
201 }
202
203 fn stay(&self, first: &str, last: &str) -> Result<(i32, i32), String> {
204 let arrival = date_number(first)?;
205 let departure = date_number(last)?;
206 if arrival < self.today {
207 return Err(format!("past date {first}"));
208 }
209 if departure <= arrival {
210 return Err(format!("invalid range {first} {last}"));
211 }
212 if departure - arrival > 30 {
213 return Err(format!("too long {}", departure - arrival));
214 }
215 Ok((arrival, departure))
216 }
217
218 fn period(&self, first: &str, last: &str, restrict_past: bool,
219 limit: bool) -> Result<(i32, i32), String> {
220 let start = date_number(first)?;
221 let end = date_number(last)?;
222 if restrict_past && start < self.today {
223 return Err(format!("past date {first}"));
224 }
225 if end < start {
226 return Err(format!("invalid range {first} {last}"));
227 }
228 if limit && end - start + 1 > 31 {
229 return Err(format!("too long {}", end - start + 1));
230 }
231 Ok((start, end))
232 }
233
234 fn night_rate(&self, room: &Room, day: i32) -> i64 {
235 let percentage = self.seasons.iter().rev()
236 .find(|s| s.first <= day && day <= s.last)
237 .map_or(100, |s| s.percentage);
238 room.rate * percentage / 100
239 }
240
241 fn price(&self, room: &Room, arrival: i32, departure: i32) -> Vec<Night> {
242 (arrival..departure).map(|day| Night {
243 day,
244 room: room.number,
245 rate: self.night_rate(room, day),
246 }).collect()
247 }
248
249 fn first_unavailable(&self, room: i32, arrival: i32, departure: i32,
250 except_for: Option<&str>) -> Option<i32> {
251 (arrival..departure).find(|day| {
252 let key = (*day, room);
253 self.blocks.contains(&key) ||
254 self.occupied.get(&key)
255 .is_some_and(|(id, _)| Some(id.as_str()) != except_for)
256 })
257 }
258
259 fn free_room(&self, kind: &str, arrival: i32, departure: i32,
260 except_for: Option<&str>) -> Option<i32> {
261 self.rooms.iter()
262 .find(|(number, room)| room.kind == kind &&
263 self.first_unavailable(**number, arrival, departure, except_for).is_none())
264 .map(|(number, _)| *number)
265 }
266
267 fn inhouse(&self, room: i32) -> Option<&Reservation> {
268 self.reservations.values()
269 .find(|reservation| reservation.status == "inhouse" && reservation.room == room)
270 }
271
272 fn occupy(&mut self, id: &str, nights: &[Night]) {
273 for night in nights {
274 self.occupied.insert((night.day, night.room), (id.to_string(), night.rate));
275 }
276 }
277
278 fn release(&mut self, nights: &[Night], first: Option<i32>) {
279 for night in nights {
280 if first.is_none_or(|day| night.day >= day) {
281 self.occupied.remove(&(night.day, night.room));
282 }
283 }
284 }

286 fn create_reservation(&mut self, id: &str, guest: &str, room_number: i32,
287 arrival: i32, departure: i32, indent: bool) {
288 let room = self.rooms[&room_number].clone();
289 let nights = self.price(&room, arrival, departure);
290 let total: i64 = nights.iter().map(|night| night.rate).sum();
291 let reservation = Reservation {
292 id: id.to_string(),
293 guest: guest.to_string(),
294 room: room_number,
295 arrival,
296 departure,
297 nights: nights.clone(),
298 status: "booked".into(),
299 extras: Vec::new(),
300 deposit: 0,
301 paid: 0,
302 posting: None,
303 fee: 0,
304 penalty: None,
305 bill: None,
306 };
307 self.reservations.insert(id.to_string(), reservation);
308 self.order.push(id.to_string());
309 self.used_ids.insert(id.to_string());
310 self.occupy(id, &nights);
311 self.emit(format!(
312 "BOOK {id} {guest} room {} {} {} -> {} nights {} total {}",
313 room.number, room.kind, date_text(arrival), date_text(departure),
314 departure - arrival, money(total)
315 ), indent);
316 }
317
318 fn guest_total(&self, guest: &str) -> i64 {
319 self.reservations.values()
320 .filter(|r| r.guest == guest)
321 .map(|r| r.paid)
322 .sum()
323 }
324
325 fn tier(&self, guest: &str) -> &'static str {
326 let points = self.guest_total(guest) / 100;
327 if points >= 2000 {
328 "GOLD"
329 } else if points >= 500 {
330 "SILVER"
331 } else {
332 "BRONZE"
333 }
334 }
335
336 fn calculate_bill(&self, reservation: &Reservation) -> Bill {
337 let lodging = reservation.lodging();
338 let percentage = match self.tier(&reservation.guest) {
339 "GOLD" => 10,
340 "SILVER" => 5,
341 _ => 0,
342 };
343 let discount = lodging * percentage / 100;
344 let penalty = reservation.penalty.map_or(0, |(_, rate)| rate);
345 let extras: i64 = reservation.extras.iter().map(|(_, amount)| amount).sum();
346 let subtotal = lodging - discount + penalty + extras;
347 let tax = subtotal / 10;
348 Bill {
349 lodging, discount, penalty, extras, tax, total: subtotal + tax,
350 }
351 }
352
353 fn deposit_parts(reservation: &Reservation, total: i64) -> String {
354 if total >= reservation.deposit {
355 format!("deposit {} due {}", money(reservation.deposit),
356 money(total - reservation.deposit))
357 } else {
358 format!("deposit {} refund {}", money(reservation.deposit),
359 money(reservation.deposit - total))
360 }
361 }
362
363 fn check_out(&mut self, id: &str, day: i32, indent: bool) {
364 let mut reservation = self.reservations[id].clone();
365 if day < reservation.departure {
366 let rate = reservation.nights.iter().find(|night| night.day == day)
367 .expect("early departure must have a priced night").rate;
368 reservation.penalty = Some((day, rate));
369 self.release(&reservation.nights, Some(day));
370 reservation.nights.retain(|night| night.day < day);
371 reservation.departure = day;
372 }
373 let bill = self.calculate_bill(&reservation);
374 reservation.bill = Some(bill);
375 reservation.paid = bill.total;
376 reservation.status = "checkedout".into();
377 reservation.posting = Some(day);
378 self.rooms.get_mut(&reservation.room).unwrap().clean = false;
379 self.emit(format!(
380 "CHECKOUT {id} nights {} lodging {} discount {} penalty {} extras {} tax {} total {} {}",
381 reservation.nights.len(), money(bill.lodging), money(bill.discount),
382 money(bill.penalty), money(bill.extras), money(bill.tax),
383 money(bill.total), Self::deposit_parts(&reservation, bill.total)
384 ), indent);
385 self.reservations.insert(id.to_string(), reservation);
386 }
387
388 fn process_waitlist(&mut self) {
389 let entries = std::mem::take(&mut self.waitlist);
390 let mut remaining = Vec::new();
391 for entry in entries {
392 if entry.expiry < self.today {
393 self.emit(format!("WAITLIST {} expired", entry.id), true);
394 } else if let Some(room) = self.free_room(&entry.kind, entry.arrival,
395 entry.departure, None) {
396 self.create_reservation(&entry.id, &entry.guest, room,
397 entry.arrival, entry.departure, true);
398 } else {
399 remaining.push(entry);
400 }
401 }
402 self.waitlist = remaining;
403 }

405 fn occupancy_parts(&self, room_numbers: &[i32], day: i32) -> String {
406 let blocked = room_numbers.iter()
407 .filter(|number| self.blocks.contains(&(day, **number))).count();
408 let occupied = room_numbers.iter()
409 .filter(|number| self.occupied.contains_key(&(day, **number))).count();
410 let available = room_numbers.len() - blocked;
411 let rate = if available > 0 {
412 format!("{}%", occupied * 100 / available)
413 } else {
414 "n/a".into()
415 };
416 format!("rooms {} blocked {blocked} occupied {occupied} rate {rate}",
417 room_numbers.len())
418 }
419
420 fn daily_values(&self, day: i32) -> (i64, i64, i64) {
421 let available = self.rooms.len() as i64 -
422 self.rooms.keys().filter(|n| self.blocks.contains(&(day, **n))).count() as i64;
423 let mut occupied = 0;
424 let mut revenue = 0;
425 for number in self.rooms.keys() {
426 if let Some((_, rate)) = self.occupied.get(&(day, *number)) {
427 occupied += 1;
428 revenue += rate;
429 }
430 }
431 (available, occupied, revenue)
432 }
433
434 fn metrics_parts(available: i64, occupied: i64, revenue: i64) -> String {
435 let rate = if available > 0 {
436 format!("{}%", occupied * 100 / available)
437 } else { "n/a".into() };
438 let adr = if occupied > 0 {
439 money(revenue / occupied)
440 } else { "n/a".into() };
441 let revpar = if available > 0 {
442 money(revenue / available)
443 } else { "n/a".into() };
444 format!("sold {occupied}/{available} rate {rate} revenue {} adr {adr} revpar {revpar}",
445 money(revenue))
446 }

448 fn command(&mut self, words: &[&str]) -> Result<(), String> {
449 let a = &words[1..];
450 match words[0] {
451 "ROOM" => {
452 let number: i32 = a[0].parse().unwrap();
453 if self.rooms.contains_key(&number) {
454 return Err(format!("duplicate room {number}"));
455 }
456 self.require_type(a[1])?;
457 let rate = cents(a[2]);
458 self.rooms.insert(number, Room {
459 number, kind: a[1].into(), rate, clean: true,
460 });
461 self.emit(format!("ROOM {number} {} {}", a[1], money(rate)), false);
462 }
463 "RATE" => {
464 let number: i32 = a[0].parse().unwrap();
465 let room = self.require_room(number)?.clone();
466 let new_rate = cents(a[1]);
467 self.rooms.get_mut(&number).unwrap().rate = new_rate;
468 self.emit(format!("RATE {number} {} -> {}", money(room.rate),
469 money(new_rate)), false);
470 }
471 "SEASON" => {
472 let (start, end) = self.period(a[0], a[1], false, false)?;
473 let percentage = a[2].parse::<i64>().unwrap();
474 self.seasons.push(Season { first: start, last: end, percentage });
475 self.emit(format!("SEASON {} {} -> {} {}%",
476 self.seasons.len(), a[0], a[1], percentage), false);
477 }
478 "BLOCK" | "UNBLOCK" => {
479 let number: i32 = a[0].parse().unwrap();
480 self.require_room(number)?;
481 let (start, end) = self.period(a[1], a[2], true, true)?;
482 if words[0] == "BLOCK" {
483 for day in start..=end {
484 if self.occupied.contains_key(&(day, number)) {
485 return Err(format!("room unavailable {number} {}", date_text(day)));
486 }
487 }
488 let mut added = 0;
489 for day in start..=end {
490 if self.blocks.insert((day, number)) {
491 added += 1;
492 }
493 }
494 self.emit(format!("BLOCK {number} {} -> {} added {added}",
495 a[1], a[2]), false);
496 } else {
497 let mut removed = 0;
498 for day in start..=end {
499 if self.blocks.remove(&(day, number)) {
500 removed += 1;
501 }
502 }
503 self.emit(format!("UNBLOCK {number} {} -> {} removed {removed}",
504 a[1], a[2]), false);
505 }
506 }
507 "CLEAN" => {
508 let number: i32 = a[0].parse().unwrap();
509 let room = self.require_room(number)?;
510 if room.clean {
511 return Err(format!("already clean {number}"));
512 }
513 self.rooms.get_mut(&number).unwrap().clean = true;
514 self.emit(format!("CLEAN {number}"), false);
515 }
516 "TODAY" => {
517 let day = date_number(a[0])?;
518 if day < self.today {
519 return Err(format!("past date {}", a[0]));
520 }
521 self.emit(format!("TODAY {}", date_weekday(day)), false);
522 let mut departing: Vec<(i32, String)> = self.reservations.values()
523 .filter(|r| r.status == "inhouse" && r.departure < day)
524 .map(|r| (r.departure, r.id.clone())).collect();
525 departing.sort();
526 for (departure, id) in departing {
527 self.check_out(&id, departure, true);
528 }
529 let mut absent: Vec<(i32, String)> = self.reservations.values()
530 .filter(|r| r.status == "booked" && r.arrival < day)
531 .map(|r| (r.arrival, r.id.clone())).collect();
532 absent.sort();
533 for (_, id) in absent {
534 let mut reservation = self.reservations[&id].clone();
535 self.release(&reservation.nights, None);
536 reservation.fee = reservation.first_rate();
537 reservation.paid = reservation.fee;
538 reservation.status = "noshow".into();
539 reservation.posting = Some(reservation.arrival);
540 self.emit(format!(
541 "NOSHOW {} {} fee {} {}",
542 id, reservation.guest, money(reservation.fee),
543 Self::deposit_parts(&reservation, reservation.fee)
544 ), true);
545 self.reservations.insert(id, reservation);
546 }
547 self.today = day;
548 }
549 "BOOK" | "BOOKROOM" | "WAITLIST" => {
550 self.require_unused(a[0])?;
551 let is_room = words[0] == "BOOKROOM";
552 let requested_room = if is_room {
553 Some(self.require_room(a[2].parse().unwrap())?.clone())
554 } else {
555 self.require_type(a[2])?;
556 None
557 };
558 let (arrival, departure) = self.stay(a[3], a[4])?;
559 if let Some(room) = requested_room {
560 if let Some(day) = self.first_unavailable(room.number, arrival, departure, None) {
561 return Err(format!("room unavailable {} {}", room.number, date_text(day)));
562 }
563 self.create_reservation(a[0], a[1], room.number, arrival, departure, false);
564 } else if let Some(room) = self.free_room(a[2], arrival, departure, None) {
565 self.create_reservation(a[0], a[1], room, arrival, departure, false);
566 } else if words[0] == "BOOK" {
567 return Err(format!("no availability {} {} {}", a[2], a[3], a[4]));
568 } else {
569 self.waitlist.push(Waiting {
570 id: a[0].into(), guest: a[1].into(), kind: a[2].into(),
571 arrival, departure, expiry: arrival,
572 });
573 self.used_ids.insert(a[0].to_string());
574 self.emit(format!("WAITLIST {} queued position {}", a[0], self.waitlist.len()),
575 false);
576 }
577 }
578 "MODIFY" => {
579 let old = self.reservation(a[0], Some("booked"))?.clone();
580 let (arrival, departure) = self.stay(a[1], a[2])?;
581 let current = self.require_room(old.room)?.clone();
582 let (room, choice) = if self.first_unavailable(current.number, arrival,
583 departure, Some(a[0])).is_none() {
584 (current.number, "kept")
585 } else if let Some(other) = self.free_room(&current.kind, arrival,
586 departure, Some(a[0])) {
587 (other, "reassigned")
588 } else {
589 return Err(format!("no availability {} {} {}", current.kind, a[1], a[2]));
590 };
591 let nights = self.price(self.require_room(room)?, arrival, departure);
592 self.release(&old.nights, None);
593 let mut updated = old;
594 updated.room = room;
595 updated.arrival = arrival;
596 updated.departure = departure;
597 updated.nights = nights.clone();
598 self.occupy(a[0], &nights);
599 self.emit(format!(
600 "MODIFY {} room {room} {choice} {} -> {} nights {} total {}",
601 a[0], a[1], a[2], departure - arrival, money(updated.lodging())
602 ), false);
603 self.reservations.insert(a[0].to_string(), updated);
604 }
605 "CANCEL" => {
606 if let Some(position) = self.waitlist.iter().position(|e| e.id == a[0]) {
607 self.waitlist.remove(position);
608 self.emit(format!("CANCEL {} waitlist removed", a[0]), false);
609 } else {
610 let mut reservation = self.reservation(a[0], Some("booked"))?.clone();
611 let days = reservation.arrival - self.today;
612 let first_rate = reservation.first_rate();
613 let fee = if days >= 7 {
614 0
615 } else if days >= 1 {
616 first_rate / 2
617 } else {
618 first_rate
619 };
620 self.release(&reservation.nights, None);
621 reservation.fee = fee;
622 reservation.paid = fee;
623 reservation.status = "cancelled".into();
624 reservation.posting = Some(self.today);
625 self.emit(format!(
626 "CANCEL {} days {days} fee {} {}",
627 a[0], money(fee), Self::deposit_parts(&reservation, fee)
628 ), false);
629 self.reservations.insert(a[0].to_string(), reservation);
630 }
631 }
632 "CHECKIN" => {
633 let reservation = self.reservation(a[0], Some("booked"))?.clone();
634 if reservation.arrival != self.today {
635 return Err(format!("not arrival date {} {}",
636 a[0], date_text(reservation.arrival)));
637 }
638 if let Some(occupant) = self.inhouse(reservation.room) {
639 return Err(format!("room occupied {} {}", reservation.room, occupant.id));
640 }
641 if !self.require_room(reservation.room)?.clean {
642 return Err(format!("room dirty {}", reservation.room));
643 }
644 self.reservations.get_mut(a[0]).unwrap().status = "inhouse".into();
645 self.emit(format!(
646 "CHECKIN {} {} room {} until {}",
647 a[0], reservation.guest, reservation.room, date_text(reservation.departure)
648 ), false);
649 }
650 "MOVE" => {
651 let old = self.reservation(a[0], Some("inhouse"))?.clone();
652 let destination: i32 = a[1].parse().unwrap();
653 self.require_room(destination)?;
654 if destination == old.room {
655 return Err(format!("same room {destination}"));
656 }
657 let remaining: Vec<Night> = old.nights.iter()
658 .filter(|n| n.day >= self.today).cloned().collect();
659 if remaining.is_empty() {
660 return Err(format!("no remaining nights {}", a[0]));
661 }
662 if let Some(day) = self.first_unavailable(destination, remaining[0].day,
663 old.departure, Some(a[0])) {
664 return Err(format!("room unavailable {destination} {}", date_text(day)));
665 }
666 if let Some(occupant) = self.inhouse(destination) {
667 return Err(format!("room occupied {destination} {}", occupant.id));
668 }
669 if !self.require_room(destination)?.clean {
670 return Err(format!("room dirty {destination}"));
671 }
672 self.release(&old.nights, Some(self.today));
673 let mut updated = old.clone();
674 for night in &mut updated.nights {
675 if night.day >= self.today {
676 night.room = destination;
677 }
678 }
679 let new_nights: Vec<Night> = updated.nights.iter()
680 .filter(|n| n.day >= self.today).cloned().collect();
681 self.occupy(a[0], &new_nights);
682 self.rooms.get_mut(&old.room).unwrap().clean = false;
683 updated.room = destination;
684 self.emit(format!(
685 "MOVE {} {} -> {destination} nights {}",
686 a[0], old.room, remaining.len()
687 ), false);
688 self.reservations.insert(a[0].to_string(), updated);
689 }
690 "EXTEND" => {
691 let old = self.reservation(a[0], Some("inhouse"))?.clone();
692 let departure = date_number(a[1])?;
693 if departure <= old.departure {
694 return Err(format!("invalid range {} {}", date_text(old.departure), a[1]));
695 }
696 if departure - old.arrival > 30 {
697 return Err(format!("too long {}", departure - old.arrival));
698 }
699 if let Some(day) = self.first_unavailable(old.room, old.departure,
700 departure, Some(a[0])) {
701 return Err(format!("room unavailable {} {}", old.room, date_text(day)));
702 }
703 let nights = self.price(self.require_room(old.room)?, old.departure, departure);
704 let mut updated = old;
705 updated.nights.extend(nights.iter().cloned());
706 updated.departure = departure;
707 self.occupy(a[0], &nights);
708 self.emit(format!(
709 "EXTEND {} until {} nights {} total {}",
710 a[0], a[1], updated.nights.len(), money(updated.lodging())
711 ), false);
712 self.reservations.insert(a[0].to_string(), updated);
713 }
714 "CHARGE" => {
715 let mut reservation = self.reservation(a[0], Some("inhouse"))?.clone();
716 let amount = cents(a[1]);
717 if amount == 0 {
718 return Err("invalid amount".into());
719 }
720 reservation.extras.push((a[2].into(), amount));
721 let total: i64 = reservation.extras.iter().map(|(_, value)| *value).sum();
722 self.emit(format!("CHARGE {} {} {} extras {}",
723 a[0], a[2], a[1], money(total)), false);
724 self.reservations.insert(a[0].to_string(), reservation);
725 }
726 "PAY" => {
727 let mut reservation = self.reservation(a[0], None)?.clone();
728 if reservation.status != "booked" && reservation.status != "inhouse" {
729 return Err(format!("invalid status {} {}", a[0], reservation.status));
730 }
731 let amount = cents(a[1]);
732 if amount == 0 {
733 return Err("invalid amount".into());
734 }
735 reservation.deposit += amount;
736 self.emit(format!("PAY {} {} deposit {}", a[0], a[1],
737 money(reservation.deposit)), false);
738 self.reservations.insert(a[0].to_string(), reservation);
739 }
740 "CHECKOUT" => {
741 self.reservation(a[0], Some("inhouse"))?;
742 self.check_out(a[0], self.today, false);
743 }
744 "INVOICE" => {
745 let reservation = self.reservation(a[0], None)?.clone();
746 let room = self.require_room(reservation.room)?;
747 self.emit(format!("INVOICE {} {} room {} {} {}",
748 a[0], reservation.guest, room.number, room.kind,
749 reservation.status), false);
750 if reservation.status == "cancelled" || reservation.status == "noshow" {
751 let label = if reservation.status == "cancelled" {
752 "cancellation"
753 } else {
754 "noshow"
755 };
756 self.emit(format!("{label} fee {}", money(reservation.fee)), true);
757 self.emit(format!("total {}", money(reservation.fee)), true);
758 self.emit(Self::deposit_parts(&reservation, reservation.fee), true);
759 } else {
760 for night in &reservation.nights {
761 self.emit(format!("{} room {} {}",
762 date_weekday(night.day), night.room, money(night.rate)), true);
763 }
764 let bill = reservation.bill.unwrap_or_else(|| self.calculate_bill(&reservation));
765 self.emit(format!("lodging {}", money(bill.lodging)), true);
766 self.emit(format!("discount {}", money(bill.discount)), true);
767 if let Some((day, _)) = reservation.penalty {
768 self.emit(format!("penalty {} {}", date_weekday(day), money(bill.penalty)),
769 true);
770 }
771 for (item, amount) in &reservation.extras {
772 self.emit(format!("extra {item} {}", money(*amount)), true);
773 }
774 self.emit(format!("tax {}", money(bill.tax)), true);
775 self.emit(format!("total {}", money(bill.total)), true);
776 self.emit(Self::deposit_parts(&reservation, bill.total), true);
777 }
778 }
779 "AVAIL" => {
780 self.require_type(a[0])?;
781 let (start, end) = self.period(a[1], a[2], false, true)?;
782 self.emit(format!("AVAIL {} {} -> {}", a[0], a[1], a[2]), false);
783 for day in start..=end {
784 let free: Vec<i32> = self.rooms.iter()
785 .filter(|(number, room)| room.kind == a[0] &&
786 !self.blocks.contains(&(day, **number)) &&
787 !self.occupied.contains_key(&(day, **number)))
788 .map(|(number, _)| *number).collect();
789 let suffix = free.iter().map(|number| format!(" {number}")).collect::<Vec<_>>().concat();
790 self.emit(format!("{} free {}{suffix}", date_weekday(day), free.len()), true);
791 }
792 }
793 "OCCUPANCY" => {
794 let day = date_number(a[0])?;
795 let numbers: Vec<i32> = self.rooms.keys().copied().collect();
796 self.emit(format!("OCCUPANCY {} {}", date_weekday(day),
797 self.occupancy_parts(&numbers, day)), false);
798 for kind in TYPES {
799 let numbers: Vec<i32> = self.rooms.values()
800 .filter(|room| room.kind == kind).map(|room| room.number).collect();
801 if !numbers.is_empty() {
802 self.emit(format!("{kind} {}", self.occupancy_parts(&numbers, day)), true);
803 }
804 }
805 }
806 "STATS" => {
807 let (start, end) = self.period(a[0], a[1], false, true)?;
808 self.emit(format!("STATS {} -> {}", a[0], a[1]), false);
809 let mut totals = (0, 0, 0);
810 for day in start..=end {
811 let (available, occupied, revenue) = self.daily_values(day);
812 totals.0 += available;
813 totals.1 += occupied;
814 totals.2 += revenue;
815 self.emit(format!("{} {}", date_weekday(day),
816 Self::metrics_parts(available, occupied, revenue)), true);
817 }
818 self.emit(format!("total {}", Self::metrics_parts(totals.0, totals.1, totals.2)), true);
819 }
820 "ROOMS" => {
821 self.emit(format!("ROOMS {} rooms {}", date_weekday(self.today),
822 self.rooms.len()), false);
823 let rooms: Vec<Room> = self.rooms.values().cloned().collect();
824 for room in rooms {
825 let occupancy = if let Some(occupant) = self.inhouse(room.number) {
826 format!("occupied {}", occupant.id)
827 } else if self.blocks.contains(&(self.today, room.number)) {
828 "blocked".into()
829 } else {
830 "vacant".into()
831 };
832 let mut line = format!(
833 "{} {} base {} tonight {} {} {occupancy}",
834 room.number, room.kind, money(room.rate),
835 money(self.night_rate(&room, self.today)),
836 if room.clean { "clean" } else { "dirty" }
837 );
838 if let Some(upcoming) = self.reservations.values()
839 .filter(|r| r.status == "booked" && r.room == room.number)
840 .min_by(|a, b| (a.arrival, &a.id).cmp(&(b.arrival, &b.id))) {
841 line += &format!(" next {} {}", upcoming.id, date_text(upcoming.arrival));
842 }
843 self.emit(line, true);
844 }
845 }
846 "LIST" => {
847 let mut reservations: Vec<Reservation> = self.reservations.values()
848 .filter(|r| r.status == a[0]).cloned().collect();
849 reservations.sort_by(|a, b| (a.arrival, &a.id).cmp(&(b.arrival, &b.id)));
850 self.emit(format!("LIST {} {}", a[0], reservations.len()), false);
851 for reservation in reservations {
852 let line = format!("{} {} room {} {} -> {}",
853 reservation.id, reservation.guest, reservation.room,
854 date_text(reservation.arrival), date_text(reservation.departure));
855 self.emit(line, true);
856 }
857 }
858 "QUEUE" => {
859 self.emit(format!("QUEUE {}", self.waitlist.len()), false);
860 let entries = self.waitlist.clone();
861 for (position, entry) in entries.iter().enumerate() {
862 self.emit(format!("{} {} {} {} {} -> {}",
863 position + 1, entry.id, entry.guest, entry.kind,
864 date_text(entry.arrival), date_text(entry.departure)), true);
865 }
866 }
867 "SCHEDULE" => {
868 let day = date_number(a[0])?;
869 let arrivals: Vec<Reservation> = self.reservations.values()
870 .filter(|r| r.arrival == day && r.status != "cancelled")
871 .cloned().collect();
872 let departures: Vec<Reservation> = self.reservations.values()
873 .filter(|r| r.departure == day &&
874 (r.status == "booked" || r.status == "inhouse" ||
875 r.status == "checkedout"))
876 .cloned().collect();
877 self.emit(format!("SCHEDULE {} arrivals {} departures {}",
878 date_weekday(day), arrivals.len(), departures.len()), false);
879 for reservation in arrivals {
880 self.emit(format!("arrival {} {} room {} {}", reservation.id,
881 reservation.guest, reservation.room, reservation.status), true);
882 }
883 for reservation in departures {
884 self.emit(format!("departure {} {} room {} {}", reservation.id,
885 reservation.guest, reservation.room, reservation.status), true);
886 }
887 }
888 "GUEST" => {
889 let guest = a[0];
890 if !self.reservations.values().any(|r| r.guest == guest) {
891 return Err(format!("unknown guest {guest}"));
892 }
893 self.emit(format!("GUEST {guest}"), false);
894 let matches: Vec<Reservation> = self.order.iter()
895 .filter_map(|id| self.reservations.get(id))
896 .filter(|r| r.guest == guest).cloned().collect();
897 for reservation in matches {
898 let room = self.require_room(reservation.room)?;
899 let part = if reservation.status == "booked" ||
900 reservation.status == "inhouse" {
901 format!("deposit {}", money(reservation.deposit))
902 } else {
903 format!("paid {}", money(reservation.paid))
904 };
905 self.emit(format!("{} room {} {} {} -> {} {} {}",
906 reservation.id, room.number, room.kind,
907 date_text(reservation.arrival), date_text(reservation.departure),
908 reservation.status, part), true);
909 }
910 let total = self.guest_total(guest);
911 self.emit(format!("total paid {} points {} tier {}", money(total),
912 total / 100, self.tier(guest)), true);
913 }
914 "REVENUE" => {
915 let (start, end) = self.period(a[0], a[1], false, false)?;
916 let posted: Vec<&Reservation> = self.reservations.values()
917 .filter(|r| r.posting.is_some_and(|day| start <= day && day <= end)).collect();
918 let mut sums = [0i64; 7];
919 let mut checkouts = 0;
920 let mut cancellations = 0;
921 let mut noshows = 0;
922 for reservation in posted {
923 match reservation.status.as_str() {
924 "checkedout" => {
925 checkouts += 1;
926 let bill = reservation.bill.unwrap();
927 sums[0] += bill.lodging;
928 sums[1] += bill.discount;
929 sums[2] += bill.penalty;
930 sums[3] += bill.extras;
931 sums[4] += bill.tax;
932 }
933 "cancelled" => {
934 cancellations += 1;
935 sums[5] += reservation.fee;
936 }
937 "noshow" => {
938 noshows += 1;
939 sums[6] += reservation.fee;
940 }
941 _ => {}
942 }
943 }
944 self.emit(format!("REVENUE {} -> {}", a[0], a[1]), false);
945 for (label, value) in ["lodging", "discount", "penalty", "extras",
946 "tax", "cancellation", "noshow"].iter().zip(sums) {
947 self.emit(format!("{label} {}", money(value)), true);
948 }
949 let total = sums[0] - sums[1] + sums[2] + sums[3] + sums[4] + sums[5] + sums[6];
950 self.emit(format!("total {}", money(total)), true);
951 self.emit(format!("checkouts {checkouts} cancellations {cancellations} noshows {noshows}"),
952 true);
953 }
954 _ => unreachable!("input command names are guaranteed valid"),
955 }
956 Ok(())
957 }
958
959 fn execute(&mut self, line: &str) {
960 let words: Vec<&str> = line.split_ascii_whitespace().collect();
961 if let Err(reason) = self.command(&words) {
962 self.emit(format!("ERROR {reason}"), false);
963 }
964 self.process_waitlist();
965 }
966}

968fn main() {
969 let mut input = String::new();
970 io::stdin().read_to_string(&mut input).unwrap();
971 let mut lines = input.lines();
972 let commands: usize = lines.next().unwrap().parse().unwrap();
973 let mut hotel = HotelDesk::new();
974 for line in lines.take(commands) {
975 hotel.execute(line);
976 }
977 if !hotel.output.is_empty() {
978 println!("{}", hotel.output.join("\n"));
979 }
980}
