6struct DeskError: Error {
7 let reason: String
8 init(_ reason: String) { self.reason = reason }
9}

11func leap(_ year: Int) -> Bool {
12 year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
13}
14
15func monthLengths(_ year: Int) -> [Int] {
16 [31, leap(year) ? 29 : 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31]
17}
18
19func dateNumber(_ text: String) throws -> Int {
20 let parts = text.split(separator: "-").compactMap { Int($0) }
21 guard parts.count == 3 else { throw DeskError("invalid date \(text)") }
22 let (year, month, day) = (parts[0], parts[1], parts[2])
23 guard (2000...2099).contains(year), (1...12).contains(month),
24 (1...monthLengths(year)[month - 1]).contains(day) else {
25 throw DeskError("invalid date \(text)")
26 }
27 var result = 0
28 if year > 2000 {
29 for previous in 2000..<year {
30 result += leap(previous) ? 366 : 365
31 }
32 }
33 if month > 1 {
34 for length in monthLengths(year).prefix(month - 1) {
35 result += length
36 }
37 }
38 return result + day - 1
39}
40
41func dateText(_ dayNumber: Int) -> String {
42 var offset = dayNumber
43 var year = 2000
44 while offset >= (leap(year) ? 366 : 365) {
45 offset -= leap(year) ? 366 : 365
46 year += 1
47 }
48 var month = 1
49 for days in monthLengths(year) {
50 if offset < days { break }
51 offset -= days
52 month += 1
53 }
54 return String(format: "%04d-%02d-%02d", year, month, offset + 1)
55}
56
57func dateWeekday(_ day: Int) -> String {
58 "\(dateText(day)) \(weekdays[day % 7])"
59}
60
61func cents(_ text: String) -> Int {
62 let parts = text.split(separator: ".")
63 return Int(parts[0])! * 100 + Int(parts[1])!
64}
65
66func money(_ amount: Int) -> String {
67 "\(amount / 100)." + String(format: "%02d", amount % 100)
68}

183 private func room(_ text: String) throws -> Room {
184 guard let result = rooms[Int(text)!] else { throw DeskError("unknown room \(text)") }
185 return result
186 }

188 private func requireType(_ kind: String) throws {
189 guard roomTypes.contains(kind) else { throw DeskError("invalid type \(kind)") }
190 }

192 private func requireUnused(_ id: String) throws {
193 guard !usedIDs.contains(id) else { throw DeskError("duplicate reservation \(id)") }
194 }

196 private func reservation(_ id: String, status: String? = nil) throws -> Reservation {
197 guard let result = reservations[id] else { throw DeskError("unknown reservation \(id)") }
198 if let status, result.status != status {
199 throw DeskError("invalid status \(id) \(result.status)")
200 }
201 return result
202 }

204 private func stay(_ first: String, _ last: String) throws -> (Int, Int) {
205 let arrival = try dateNumber(first)
206 let departure = try dateNumber(last)
207 if arrival < today { throw DeskError("past date \(first)") }
208 if departure <= arrival { throw DeskError("invalid range \(first) \(last)") }
209 if departure - arrival > 30 { throw DeskError("too long \(departure - arrival)") }
210 return (arrival, departure)
211 }
212
213 private func period(_ first: String, _ last: String,
214 pastRestricted: Bool = false, limited: Bool = true) throws -> (Int, Int) {
215 let start = try dateNumber(first)
216 let end = try dateNumber(last)
217 if pastRestricted && start < today { throw DeskError("past date \(first)") }
218 if end < start { throw DeskError("invalid range \(first) \(last)") }
219 if limited && end - start + 1 > 31 {
220 throw DeskError("too long \(end - start + 1)")
221 }
222 return (start, end)
223 }
224
225 private func nightRate(_ room: Room, _ day: Int) -> Int {
226 let percentage = seasons.reversed()
227 .first(where: { $0.start <= day && day <= $0.end })?.percentage ?? 100
228 return room.rate * percentage / 100
229 }
230
231 private func price(_ room: Room, _ arrival: Int, _ departure: Int) -> [Night] {
232 (arrival..<departure).map { Night(day: $0, room: room.number, rate: nightRate(room, $0)) }
233 }
234
235 private func firstUnavailable(_ room: Room, _ arrival: Int, _ departure: Int,
236 except reservation: Reservation? = nil) -> Int? {
237 for day in arrival..<departure {
238 let key = NightKey(day: day, room: room.number)
239 if blocks.contains(key) { return day }
240 if let occupant = occupied[key]?.reservation, occupant !== reservation { return day }
241 }
242 return nil
243 }
244
245 private func findRoom(_ kind: String, _ arrival: Int, _ departure: Int,
246 except reservation: Reservation? = nil) -> Room? {
247 rooms.values.sorted { $0.number < $1.number }
248 .first { $0.kind == kind && firstUnavailable($0, arrival, departure, except: reservation) == nil }
249 }
250
251 private func inHouse(_ room: Room) -> Reservation? {
252 reservations.values.first { $0.status == "inhouse" && $0.room === room }
253 }
254
255 private func occupy(_ reservation: Reservation, _ nights: [Night]) {
256 for night in nights {
257 occupied[NightKey(day: night.day, room: night.room)] =
258 Occupancy(reservation: reservation, rate: night.rate)
259 }
260 }
261
262 private func release(_ reservation: Reservation, from first: Int? = nil) {
263 for night in reservation.nights where first == nil || night.day >= first! {
264 occupied.removeValue(forKey: NightKey(day: night.day, room: night.room))
265 }
266 }
267
268 private func createReservation(id: String, guest: String, room: Room,
269 arrival: Int, departure: Int, indented: Bool = false) {
270 let nights = price(room, arrival, departure)
271 let entry = Reservation(id: id, guest: guest, room: room, arrival: arrival,
272 departure: departure, nights: nights)
273 reservations[id] = entry
274 reservationOrder.append(id)
275 usedIDs.insert(id)
276 occupy(entry, nights)
277 let line = "BOOK \(id) \(guest) room \(room.number) \(room.kind) " +
278 "\(dateText(arrival)) -> \(dateText(departure)) nights \(departure - arrival) " +
279 "total \(money(entry.lodging))"
280 if indented { emitIndented(line) } else { emit(line) }
281 }
282
283 private func guestTotal(_ guest: String) -> Int {
284 reservations.values.filter { $0.guest == guest }.reduce(0) { $0 + $1.paid }
285 }
286
287 private func tier(_ guest: String) -> String {
288 let points = guestTotal(guest) / 100
289 return points >= 2000 ? "GOLD" : points >= 500 ? "SILVER" : "BRONZE"
290 }
291
292 private func bill(_ reservation: Reservation) -> Bill {
293 let lodging = reservation.lodging
294 let percentage = tier(reservation.guest) == "GOLD" ? 10 :
295 tier(reservation.guest) == "SILVER" ? 5 : 0
296 let discount = lodging * percentage / 100
297 let penalty = reservation.penalty?.rate ?? 0
298 let extras = reservation.extras.reduce(0) { $0 + $1.amount }
299 let subtotal = lodging - discount + penalty + extras
300 let tax = subtotal / 10
301 return Bill(lodging: lodging, discount: discount, penalty: penalty,
302 extras: extras, tax: tax, total: subtotal + tax)
303 }
304
305 private func depositText(_ reservation: Reservation, _ total: Int) -> String {
306 if total >= reservation.deposit {
307 return "deposit \(money(reservation.deposit)) due \(money(total - reservation.deposit))"
308 }
309 return "deposit \(money(reservation.deposit)) refund \(money(reservation.deposit - total))"
310 }
311
312 private func checkout(_ reservation: Reservation, on day: Int, indented: Bool = false) {
313 if day < reservation.departure {
314 let rate = reservation.nights.first { $0.day == day }!.rate
315 reservation.penalty = (day, rate)
316 release(reservation, from: day)
317 reservation.nights.removeAll { $0.day >= day }
318 reservation.departure = day
319 }
320 let amount = bill(reservation)
321 reservation.bill = amount
322 reservation.paid = amount.total
323 reservation.status = "checkedout"
324 reservation.posting = day
325 reservation.room.clean = false
326 let line = "CHECKOUT \(reservation.id) nights \(reservation.nights.count) " +
327 "lodging \(money(amount.lodging)) discount \(money(amount.discount)) " +
328 "penalty \(money(amount.penalty)) extras \(money(amount.extras)) " +
329 "tax \(money(amount.tax)) total \(money(amount.total)) " +
330 depositText(reservation, amount.total)
331 if indented { emitIndented(line) } else { emit(line) }
332 }

334 private func processWaitlist() {
335 var remaining: [WaitingEntry] = []
336 for entry in waitlist {
337 if entry.expiry < today {
338 emitIndented("WAITLIST", entry.id, "expired")
339 } else if let room = findRoom(entry.kind, entry.arrival, entry.departure) {
340 createReservation(id: entry.id, guest: entry.guest, room: room,
341 arrival: entry.arrival, departure: entry.departure, indented: true)
342 } else {
343 remaining.append(entry)
344 }
345 }
346 waitlist = remaining
347 }

349 private func occupancyText(_ numbers: [Int], _ day: Int) -> String {
350 let blocked = numbers.filter { blocks.contains(NightKey(day: day, room: $0)) }.count
351 let count = numbers.filter { occupied[NightKey(day: day, room: $0)] != nil }.count
352 let available = numbers.count - blocked
353 let rate = available > 0 ? "\(count * 100 / available)%" : "n/a"
354 return "rooms \(numbers.count) blocked \(blocked) occupied \(count) rate \(rate)"
355 }
356
357 private func dailyMetrics(_ day: Int) -> DailyMetrics {
358 let numbers = Array(rooms.keys)
359 let available = numbers.count - numbers.filter {
360 blocks.contains(NightKey(day: day, room: $0))
361 }.count
362 let nights = numbers.compactMap { occupied[NightKey(day: day, room: $0)] }
363 return DailyMetrics(available: available, occupied: nights.count,
364 revenue: nights.reduce(0) { $0 + $1.rate })
365 }
366
367 private func metricsText(_ metrics: DailyMetrics) -> String {
368 let (available, count, revenue) =
369 (metrics.available, metrics.occupied, metrics.revenue)
370 let rate = available == 0 ? "n/a" : "\(count * 100 / available)%"
371 let adr = count == 0 ? "n/a" : money(revenue / count)
372 let revpar = available == 0 ? "n/a" : money(revenue / available)
373 return "sold \(count)/\(available) rate \(rate) revenue \(money(revenue)) " +
374 "adr \(adr) revpar \(revpar)"
375 }

377 private func invoice(_ reservation: Reservation) {
378 emit("INVOICE", reservation.id, reservation.guest, "room", reservation.room.number,
379 reservation.room.kind, reservation.status)
380 if reservation.status == "cancelled" || reservation.status == "noshow" {
381 emitIndented(reservation.status == "cancelled" ? "cancellation" : "noshow",
382 "fee", money(reservation.fee))
383 emitIndented("total", money(reservation.fee))
384 emitIndented(depositText(reservation, reservation.fee))
385 return
386 }
387 for night in reservation.nights {
388 emitIndented(dateWeekday(night.day), "room", night.room, money(night.rate))
389 }
390 let amount = reservation.status == "checkedout" ? reservation.bill! : bill(reservation)
391 emitIndented("lodging", money(amount.lodging))
392 emitIndented("discount", money(amount.discount))
393 if let penalty = reservation.penalty {
394 emitIndented("penalty", dateWeekday(penalty.day), money(amount.penalty))
395 }
396 for extra in reservation.extras {
397 emitIndented("extra", extra.item, money(extra.amount))
398 }
399 emitIndented("tax", money(amount.tax))
400 emitIndented("total", money(amount.total))
401 emitIndented(depositText(reservation, amount.total))
402 }

404 private func executeCommand(_ command: String, _ a: [String]) throws {
405 switch command {
406 case "ROOM":
407 let number = Int(a[0])!
408 if rooms[number] != nil { throw DeskError("duplicate room \(a[0])") }
409 try requireType(a[1])
410 rooms[number] = Room(number, a[1], cents(a[2]))
411 emit("ROOM", a[0], a[1], a[2])
412
413 case "RATE":
414 let entry = try room(a[0])
415 let previous = entry.rate
416 entry.rate = cents(a[1])
417 emit("RATE", a[0], money(previous), "->", a[1])
418
419 case "SEASON":
420 let (start, end) = try period(a[0], a[1], limited: false)
421 seasons.append(Season(start: start, end: end, percentage: Int(a[2])!))
422 emit("SEASON", seasons.count, a[0], "->", a[1], a[2] + "%")
423
424 case "BLOCK", "UNBLOCK":
425 let room = try self.room(a[0])
426 let (start, end) = try period(a[1], a[2], pastRestricted: true)
427 if command == "BLOCK" {
428 for day in start...end {
429 if occupied[NightKey(day: day, room: room.number)] != nil {
430 throw DeskError("room unavailable \(a[0]) \(dateText(day))")
431 }
432 }
433 }
434 var changed = 0
435 for day in start...end {
436 let key = NightKey(day: day, room: room.number)
437 if command == "BLOCK" {
438 if blocks.insert(key).inserted { changed += 1 }
439 } else if blocks.remove(key) != nil {
440 changed += 1
441 }
442 }
443 if command == "BLOCK" {
444 emit("BLOCK", a[0], a[1], "->", a[2], "added", changed)
445 } else {
446 emit("UNBLOCK", a[0], a[1], "->", a[2], "removed", changed)
447 }
448
449 case "CLEAN":
450 let entry = try room(a[0])
451 if entry.clean { throw DeskError("already clean \(a[0])") }
452 entry.clean = true
453 emit("CLEAN", a[0])
454
455 case "TODAY":
456 let day = try dateNumber(a[0])
457 if day < today { throw DeskError("past date \(a[0])") }
458 emit("TODAY", dateWeekday(day))
459 let departures = reservations.values
460 .filter { $0.status == "inhouse" && $0.departure < day }
461 .sorted { ($0.departure, $0.id) < ($1.departure, $1.id) }
462 for entry in departures {
463 checkout(entry, on: entry.departure, indented: true)
464 }
465 let absences = reservations.values
466 .filter { $0.status == "booked" && $0.arrival < day }
467 .sorted { ($0.arrival, $0.id) < ($1.arrival, $1.id) }
468 for entry in absences {
469 release(entry)
470 entry.fee = entry.firstRate
471 entry.paid = entry.fee
472 entry.status = "noshow"
473 entry.posting = entry.arrival
474 emitIndented("NOSHOW", entry.id, entry.guest, "fee",
475 money(entry.fee), depositText(entry, entry.fee))
476 }
477 today = day
478
479 case "BOOK", "BOOKROOM", "WAITLIST":
480 let id = a[0], guest = a[1]
481 try requireUnused(id)
482 let isSpecific = command == "BOOKROOM"
483 if isSpecific {
484 let selected = try room(a[2])
485 let (arrival, departure) = try stay(a[3], a[4])
486 if let day = firstUnavailable(selected, arrival, departure) {
487 throw DeskError("room unavailable \(a[2]) \(dateText(day))")
488 }
489 createReservation(id: id, guest: guest, room: selected,
490 arrival: arrival, departure: departure)
491 } else {
492 let kind = a[2]
493 try requireType(kind)
494 let (arrival, departure) = try stay(a[3], a[4])
495 if let selected = findRoom(kind, arrival, departure) {
496 createReservation(id: id, guest: guest, room: selected,
497 arrival: arrival, departure: departure)
498 } else if command == "BOOK" {
499 throw DeskError("no availability \(kind) \(a[3]) \(a[4])")
500 } else {
501 waitlist.append(WaitingEntry(expiry: arrival, id: id, guest: guest, kind: kind,
502 arrival: arrival, departure: departure))
503 usedIDs.insert(id)
504 emit("WAITLIST", id, "queued position", waitlist.count)
505 }
506 }
507
508 case "MODIFY":
509 let entry = try reservation(a[0], status: "booked")
510 let (arrival, departure) = try stay(a[1], a[2])
511 var selected = entry.room
512 var choice = "kept"
513 if firstUnavailable(selected, arrival, departure, except: entry) != nil {
514 guard let replacement = findRoom(selected.kind, arrival, departure, except: entry) else {
515 throw DeskError("no availability \(selected.kind) \(a[1]) \(a[2])")
516 }
517 selected = replacement
518 choice = "reassigned"
519 }
520 let nights = price(selected, arrival, departure)
521 release(entry)
522 entry.room = selected
523 entry.arrival = arrival
524 entry.departure = departure
525 entry.nights = nights
526 occupy(entry, nights)
527 emit("MODIFY", entry.id, "room", selected.number, choice,
528 a[1], "->", a[2], "nights", departure - arrival,
529 "total", money(entry.lodging))
530
531 case "CANCEL":
532 if let index = waitlist.firstIndex(where: { $0.id == a[0] }) {
533 waitlist.remove(at: index)
534 emit("CANCEL", a[0], "waitlist removed")
535 } else {
536 let entry = try reservation(a[0], status: "booked")
537 let days = entry.arrival - today
538 let fee = days >= 7 ? 0 : days >= 1 ? entry.firstRate / 2 : entry.firstRate
539 release(entry)
540 entry.fee = fee
541 entry.paid = fee
542 entry.status = "cancelled"
543 entry.posting = today
544 emit("CANCEL", entry.id, "days", days, "fee", money(fee),
545 depositText(entry, fee))
546 }
547
548 case "CHECKIN":
549 let entry = try reservation(a[0], status: "booked")
550 if entry.arrival != today {
551 throw DeskError("not arrival date \(entry.id) \(dateText(entry.arrival))")
552 }
553 if let guest = inHouse(entry.room) {
554 throw DeskError("room occupied \(entry.room.number) \(guest.id)")
555 }
556 if !entry.room.clean {
557 throw DeskError("room dirty \(entry.room.number)")
558 }
559 entry.status = "inhouse"
560 emit("CHECKIN", entry.id, entry.guest, "room", entry.room.number,
561 "until", dateText(entry.departure))
562
563 case "MOVE":
564 let entry = try reservation(a[0], status: "inhouse")
565 let selected = try room(a[1])
566 let previous = entry.room
567 if selected === previous { throw DeskError("same room \(a[1])") }
568 let remaining = entry.nights.filter { $0.day >= today }
569 if remaining.isEmpty { throw DeskError("no remaining nights \(entry.id)") }
570 if let day = firstUnavailable(selected, remaining[0].day, entry.departure, except: entry) {
571 throw DeskError("room unavailable \(a[1]) \(dateText(day))")
572 }
573 if let guest = inHouse(selected) {
574 throw DeskError("room occupied \(selected.number) \(guest.id)")
575 }
576 if !selected.clean { throw DeskError("room dirty \(selected.number)") }
577 release(entry, from: today)
578 entry.nights = entry.nights.map {
579 Night(day: $0.day, room: $0.day >= today ? selected.number : $0.room, rate: $0.rate)
580 }
581 occupy(entry, entry.nights.filter { $0.day >= today })
582 previous.clean = false
583 entry.room = selected
584 emit("MOVE", entry.id, previous.number, "->", selected.number,
585 "nights", remaining.count)
586
587 case "EXTEND":
588 let entry = try reservation(a[0], status: "inhouse")
589 let departure = try dateNumber(a[1])
590 if departure <= entry.departure {
591 throw DeskError("invalid range \(dateText(entry.departure)) \(a[1])")
592 }
593 if departure - entry.arrival > 30 {
594 throw DeskError("too long \(departure - entry.arrival)")
595 }
596 if let day = firstUnavailable(entry.room, entry.departure, departure, except: entry) {
597 throw DeskError("room unavailable \(entry.room.number) \(dateText(day))")
598 }
599 let nights = price(entry.room, entry.departure, departure)
600 entry.nights.append(contentsOf: nights)
601 occupy(entry, nights)
602 entry.departure = departure
603 emit("EXTEND", entry.id, "until", a[1], "nights", entry.nights.count,
604 "total", money(entry.lodging))
605
606 case "CHARGE":
607 let entry = try reservation(a[0], status: "inhouse")
608 let amount = cents(a[1])
609 if amount == 0 { throw DeskError("invalid amount") }
610 entry.extras.append(ExtraCharge(item: a[2], amount: amount))
611 let total = entry.extras.reduce(0) { $0 + $1.amount }
612 emit("CHARGE", entry.id, a[2], a[1], "extras", money(total))
613
614 case "PAY":
615 let entry = try reservation(a[0])
616 if entry.status != "booked" && entry.status != "inhouse" {
617 throw DeskError("invalid status \(entry.id) \(entry.status)")
618 }
619 let amount = cents(a[1])
620 if amount == 0 { throw DeskError("invalid amount") }
621 entry.deposit += amount
622 emit("PAY", entry.id, a[1], "deposit", money(entry.deposit))
623
624 case "CHECKOUT":
625 let entry = try reservation(a[0], status: "inhouse")
626 checkout(entry, on: today)
627
628 case "INVOICE":
629 invoice(try reservation(a[0]))
630
631 case "AVAIL":
632 try requireType(a[0])
633 let (start, end) = try period(a[1], a[2])
634 let numbers = rooms.values.filter { $0.kind == a[0] }.map(\.number).sorted()
635 emit("AVAIL", a[0], a[1], "->", a[2])
636 for day in start...end {
637 let free = numbers.filter { number in
638 let key = NightKey(day: day, room: number)
639 return !blocks.contains(key) && occupied[key] == nil
640 }
641 var line = "\(dateWeekday(day)) free \(free.count)"
642 if !free.isEmpty {
643 line += " " + free.map(String.init).joined(separator: " ")
644 }
645 emitIndented(line)
646 }
647
648 case "OCCUPANCY":
649 let day = try dateNumber(a[0])
650 emit("OCCUPANCY", dateWeekday(day), occupancyText(Array(rooms.keys), day))
651 for kind in roomTypes {
652 let numbers = rooms.values.filter { $0.kind == kind }.map(\.number)
653 if !numbers.isEmpty { emitIndented(kind, occupancyText(numbers, day)) }
654 }
655
656 case "STATS":
657 let (start, end) = try period(a[0], a[1])
658 emit("STATS", a[0], "->", a[1])
659 var sums = DailyMetrics(available: 0, occupied: 0, revenue: 0)
660 for day in start...end {
661 let item = dailyMetrics(day)
662 sums = DailyMetrics(available: sums.available + item.available,
663 occupied: sums.occupied + item.occupied,
664 revenue: sums.revenue + item.revenue)
665 emitIndented(dateWeekday(day), metricsText(item))
666 }
667 emitIndented("total", metricsText(sums))
668
669 case "ROOMS":
670 emit("ROOMS", dateWeekday(today), "rooms", rooms.count)
671 for entry in rooms.values.sorted(by: { $0.number < $1.number }) {
672 var parts = [String(entry.number), entry.kind, "base", money(entry.rate),
673 "tonight", money(nightRate(entry, today)),
674 entry.clean ? "clean" : "dirty"]
675 if let guest = inHouse(entry) {
676 parts += ["occupied", guest.id]
677 } else if blocks.contains(NightKey(day: today, room: entry.number)) {
678 parts.append("blocked")
679 } else {
680 parts.append("vacant")
681 }
682 if let upcoming = reservations.values
683 .filter({ $0.status == "booked" && $0.room === entry })
684 .min(by: { ($0.arrival, $0.id) < ($1.arrival, $1.id) }) {
685 parts += ["next", upcoming.id, dateText(upcoming.arrival)]
686 }
687 emitIndented(parts.joined(separator: " "))
688 }
689
690 case "LIST":
691 let entries = reservations.values.filter { $0.status == a[0] }
692 .sorted { ($0.arrival, $0.id) < ($1.arrival, $1.id) }
693 emit("LIST", a[0], entries.count)
694 for entry in entries {
695 emitIndented(entry.id, entry.guest, "room", entry.room.number,
696 dateText(entry.arrival), "->", dateText(entry.departure))
697 }
698
699 case "QUEUE":
700 emit("QUEUE", waitlist.count)
701 for (index, entry) in waitlist.enumerated() {
702 emitIndented(index + 1, entry.id, entry.guest, entry.kind,
703 dateText(entry.arrival), "->", dateText(entry.departure))
704 }
705
706 case "SCHEDULE":
707 let day = try dateNumber(a[0])
708 let arrivals = reservations.values
709 .filter { $0.arrival == day && $0.status != "cancelled" }
710 .sorted { $0.id < $1.id }
711 let departures = reservations.values
712 .filter {
713 $0.departure == day &&
714 ["booked", "inhouse", "checkedout"].contains($0.status)
715 }
716 .sorted { $0.id < $1.id }
717 emit("SCHEDULE", dateWeekday(day), "arrivals", arrivals.count,
718 "departures", departures.count)
719 for entry in arrivals {
720 emitIndented("arrival", entry.id, entry.guest, "room", entry.room.number,
721 entry.status)
722 }
723 for entry in departures {
724 emitIndented("departure", entry.id, entry.guest, "room", entry.room.number,
725 entry.status)
726 }
727
728 case "GUEST":
729 let entries = reservationOrder.compactMap { reservations[$0] }
730 .filter { $0.guest == a[0] }
731 if entries.isEmpty { throw DeskError("unknown guest \(a[0])") }
732 emit("GUEST", a[0])
733 for entry in entries {
734 let active = entry.status == "booked" || entry.status == "inhouse"
735 emitIndented(entry.id, "room", entry.room.number, entry.room.kind,
736 dateText(entry.arrival), "->", dateText(entry.departure),
737 entry.status, active ? "deposit" : "paid",
738 money(active ? entry.deposit : entry.paid))
739 }
740 let paid = guestTotal(a[0])
741 emitIndented("total paid", money(paid), "points", paid / 100, "tier", tier(a[0]))
742
743 case "REVENUE":
744 let (start, end) = try period(a[0], a[1], limited: false)
745 let posted = reservations.values.filter { entry in
746 guard let day = entry.posting else { return false }
747 return start <= day && day <= end
748 }
749 let checkedOut = posted.filter { $0.status == "checkedout" }
750 let cancelled = posted.filter { $0.status == "cancelled" }
751 let noShows = posted.filter { $0.status == "noshow" }
752 let lodging = checkedOut.reduce(0) { $0 + $1.bill!.lodging }
753 let discount = checkedOut.reduce(0) { $0 + $1.bill!.discount }
754 let penalty = checkedOut.reduce(0) { $0 + $1.bill!.penalty }
755 let extras = checkedOut.reduce(0) { $0 + $1.bill!.extras }
756 let tax = checkedOut.reduce(0) { $0 + $1.bill!.tax }
757 let cancellations = cancelled.reduce(0) { $0 + $1.fee }
758 let noShowFees = noShows.reduce(0) { $0 + $1.fee }
759 emit("REVENUE", a[0], "->", a[1])
760 for (label, amount) in [
761 ("lodging", lodging), ("discount", discount), ("penalty", penalty),
762 ("extras", extras), ("tax", tax), ("cancellation", cancellations),
763 ("noshow", noShowFees)
764 ] {
765 emitIndented(label, money(amount))
766 }
767 let total = lodging - discount + penalty + extras + tax + cancellations + noShowFees
768 emitIndented("total", money(total))
769 emitIndented("checkouts", checkedOut.count, "cancellations",
770 cancelled.count, "noshows", noShows.count)
771 default:
772 break
773 }
774 }

776 func execute(_ line: String) {
777 let fields = line.split(separator: " ").map(String.init)
778 guard let command = fields.first else { return }
779 do {
780 try executeCommand(command, Array(fields.dropFirst()))
781 } catch let error as DeskError {
782 emit("ERROR", error.reason)
783 } catch {
784 fatalError("Unexpected hotel command failure: \(error)")
785 }
786 processWaitlist()
787 }
788}
789
790let count = Int(readLine()!)!
791let desk = HotelDesk()
792for _ in 0..<count {
793 desk.execute(readLine()!)
794}
795if !desk.output.isEmpty {
796 print(desk.output.joined(separator: "\n"))
797}
