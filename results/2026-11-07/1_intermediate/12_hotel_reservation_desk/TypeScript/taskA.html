3type OccupiedNight = {owner: string; rate: bigint};
4type Room = {number: number; kind: string; base: bigint; clean: boolean; blocks: Set<number>; occupied: Map<number, OccupiedNight>};
5type Night = {day: number; room: number; rate: bigint};
6type Bill = {lodging: bigint; discount: bigint; penalty: bigint; extras: bigint; tax: bigint; total: bigint};
7type Reservation = {id: string; guest: string; room: number; start: number; end: number; nights: Night[]; status: string; extras: {item: string; amount: bigint}[]; deposit: bigint; paid: bigint; posting: number | null; fee: bigint; penalty: Night | null; bill: Bill | null};
8type Waiting = {id: string; guest: string; kind: string; start: number; end: number; expiry: number};
9type Season = {start: number; end: number; percent: bigint};
10type Metrics = {available: number; occupied: number; revenue: bigint};
11const TYPES = ["SINGLE", "DOUBLE", "SUITE"];
12const WEEKDAYS = ["SAT", "SUN", "MON", "TUE", "WED", "THU", "FRI"];
13const DATES: string[] = [];
14const DAYS = new Map<string, number>();
15for (let year = 2000; year <= 2099; year++) {
16 const leap = year % 4 === 0 && (year % 100 !== 0 || year % 400 === 0);
17 const lengths = [31, leap ? 29 : 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31];
18 for (let month = 1; month <= 12; month++) {
19 for (let day = 1; day <= lengths[month - 1]; day++) {
20 const text = year + "-" + String(month).padStart(2, "0") + "-" + String(day).padStart(2, "0");
21 DAYS.set(text, DATES.length);
22 DATES.push(text);
23 }
24 }
25}
26const lex = (a: string, b: string): number => a < b ? -1 : a > b ? 1 : 0;
27const dateText = (day: number): string => DATES[day];
28const dateWeek = (day: number): string => dateText(day) + " " + WEEKDAYS[day % 7];
29const cents = (s: string): bigint => BigInt(s.replace(".", ""));
30const money = (v: bigint): string => v / 100n + "." + String(v % 100n).padStart(2, "0");
31const lodging = (x: Reservation): bigint => x.nights.reduce((v, n) => v + n.rate, 0n);
32const extras = (x: Reservation): bigint => x.extras.reduce((v, e) => v + e.amount, 0n);
33const byArrival = (a: Reservation, b: Reservation): number => a.start - b.start || lex(a.id, b.id);
34class CommandError extends Error {}

36class Hotel {
37 today = DAYS.get("2026-01-01")!;
38 rooms = new Map<number, Room>();
39 seasons: Season[] = [];
40 reservations = new Map<string, Reservation>();
41 queue: Waiting[] = [];
42 used = new Set<string>();
43 output: string[] = [];

45 emit(indent: number, ...parts: (string | number)[]): void {
46 this.output.push(" ".repeat(indent) + parts.join(" "));

49 date(text: string): number {
50 const day = DAYS.get(text);
51 if (day === undefined) throw new CommandError("invalid date " + text);
52 return day;
53 }
54
55 room(text: string): Room {
56 const r = this.rooms.get(Number(text));
57 if (!r) throw new CommandError("unknown room " + text);
58 return r;
59 }

61 reservation(id: string): Reservation {
62 const x = this.reservations.get(id);
63 if (!x) throw new CommandError("unknown reservation " + id);
64 return x;
65 }
66
67 status(x: Reservation, allowed: string[]): void {
68 if (!allowed.includes(x.status)) throw new CommandError("invalid status " + x.id + " " + x.status);
69 }
70
71 type(kind: string): void {
72 if (!TYPES.includes(kind)) throw new CommandError("invalid type " + kind);
73 }
74
75 unused(id: string): void {
76 if (this.used.has(id)) throw new CommandError("duplicate reservation " + id);
77 }
78
79 stay(a: string, b: string): number[] {
80 const start = this.date(a);
81 const end = this.date(b);
82 if (start < this.today) throw new CommandError("past date " + a);
83 if (end <= start) throw new CommandError("invalid range " + a + " " + b);
84 if (end - start > 30) throw new CommandError("too long " + (end - start));
85 return [start, end];
86 }
87
88 period(a: string, b: string, future = false, limited = true): number[] {
89 const start = this.date(a);
90 const end = this.date(b);
91 if (future && start < this.today) throw new CommandError("past date " + a);
92 if (end < start) throw new CommandError("invalid range " + a + " " + b);
93 if (limited && end - start + 1 > 31) throw new CommandError("too long " + (end - start + 1));
94 return [start, end];
95 }
96
97 sortedRooms(): Room[] {
98 return [...this.rooms.values()].sort((a, b) => a.number - b.number);
99 }
100
101 rate(r: Room, day: number): bigint {
102 const percent = this.seasons.findLast(season => season.start <= day && day <= season.end)?.percent ?? 100n;
103 return r.base * percent / 100n;
104 }
105
106 price(r: Room, start: number, end: number): Night[] {
107 const nights: Night[] = [];
108 for (let day = start; day < end; day++) nights.push({day, room: r.number, rate: this.rate(r, day)});
109 return nights;
110 }
111
112 free(r: Room, day: number, except: string | null = null): boolean {
113 const night = r.occupied.get(day);
114 return !r.blocks.has(day) && (!night || night.owner === except);
115 }
116
117 unavailable(r: Room, start: number, end: number, except: string | null = null): number | null {
118 for (let day = start; day < end; day++) {
119 if (!this.free(r, day, except)) return day;
120 }
121 return null;
122 }
123
124 findRoom(kind: string, start: number, end: number, except: string | null = null): Room | null {
125 return this.sortedRooms().find(r => r.kind === kind && this.unavailable(r, start, end, except) === null) ?? null;
126 }
127
128 inHouse(room: number): Reservation | undefined {
129 return [...this.reservations.values()].find(x => x.status === "inhouse" && x.room === room);
130 }
131
132 release(x: Reservation): void {
133 for (const n of x.nights) this.rooms.get(n.room)!.occupied.delete(n.day);
134 }
135
136 occupy(x: Reservation): void {
137 for (const n of x.nights) this.rooms.get(n.room)!.occupied.set(n.day, {owner: x.id, rate: n.rate});
138 }

140 create(id: string, guest: string, r: Room, start: number, end: number, indent = 0): void {
141 const x: Reservation = {id, guest, room: r.number, start, end, nights: this.price(r, start, end),
142 status: "booked", extras: [], deposit: 0n, paid: 0n, posting: null, fee: 0n, penalty: null, bill: null};
143 this.reservations.set(id, x);
144 this.used.add(id);
145 this.occupy(x);
146 this.emit(indent, "BOOK", id, guest, "room", r.number, r.kind, dateText(start), "->", dateText(end),
147 "nights", end - start, "total", money(lodging(x)));
148 }
149
150 guestPaid(guest: string): bigint {
151 return [...this.reservations.values()].filter(x => x.guest === guest).reduce((v, x) => v + x.paid, 0n);
152 }
153
154 tier(guest: string): {points: bigint; name: string; percent: bigint} {
155 const points = this.guestPaid(guest) / 100n;
156 return points >= 2000n ? {points, name: "GOLD", percent: 10n} :
157 points >= 500n ? {points, name: "SILVER", percent: 5n} : {points, name: "BRONZE", percent: 0n};
158 }
159
160 bill(x: Reservation): Bill {
161 const l = lodging(x);
162 const discount = l * this.tier(x.guest).percent / 100n;
163 const penalty = x.penalty?.rate ?? 0n;
164 const e = extras(x);
165 const subtotal = l - discount + penalty + e;
166 const tax = subtotal / 10n;
167 return {lodging: l, discount, penalty, extras: e, tax, total: subtotal + tax};
168 }
169
170 deposit(x: Reservation, amount: bigint): string {
171 return "deposit " + money(x.deposit) + (amount >= x.deposit ? " due " + money(amount - x.deposit) : " refund " + money(x.deposit - amount));
172 }
173
174 checkout(x: Reservation, day: number, indent = 0): void {
175 if (day < x.end) {
176 x.penalty = x.nights.find(n => n.day === day)!;
177 for (const n of x.nights.filter(n => n.day >= day)) this.rooms.get(n.room)!.occupied.delete(n.day);
178 x.nights = x.nights.filter(n => n.day < day);
179 x.end = day;
180 }
181 const b = this.bill(x);
182 x.bill = b;
183 x.paid = b.total;
184 x.status = "checkedout";
185 x.posting = day;
186 this.rooms.get(x.room)!.clean = false;
187 this.emit(indent, "CHECKOUT", x.id, "nights", x.nights.length, "lodging", money(b.lodging),
188 "discount", money(b.discount), "penalty", money(b.penalty), "extras", money(b.extras),
189 "tax", money(b.tax), "total", money(b.total), this.deposit(x, b.total));
190 }
191
192 processQueue(): void {
193 const remaining: Waiting[] = [];
194 for (const w of this.queue) {
195 if (w.expiry < this.today) {
196 this.emit(2, "WAITLIST", w.id, "expired");
197 } else {
198 const r = this.findRoom(w.kind, w.start, w.end);
199 if (r) this.create(w.id, w.guest, r, w.start, w.end, 2);
200 else remaining.push(w);
201 }
202 }
203 this.queue = remaining;
204 }
205
206 metrics(day: number): Metrics {
207 let available = 0;
208 let occupied = 0;
209 let revenue = 0n;
210 for (const r of this.rooms.values()) {
211 if (!r.blocks.has(day)) available++;
212 const night = r.occupied.get(day);
213 if (night) {
214 occupied++;
215 revenue += night.rate;
216 }
217 }
218 return {available, occupied, revenue};
219 }
220
221 metricsText(m: Metrics): string {
222 return "sold " + m.occupied + "/" + m.available + " rate " + (m.available ? Math.floor(m.occupied * 100 / m.available) + "%" : "n/a") +
223 " revenue " + money(m.revenue) + " adr " + (m.occupied ? money(m.revenue / BigInt(m.occupied)) : "n/a") +
224 " revpar " + (m.available ? money(m.revenue / BigInt(m.available)) : "n/a");
225 }
226
227 occupancyText(rooms: Room[], day: number): string {
228 const blocked = rooms.filter(r => r.blocks.has(day)).length;
229 const occupied = rooms.filter(r => r.occupied.has(day)).length;
230 const available = rooms.length - blocked;
231 return "rooms " + rooms.length + " blocked " + blocked + " occupied " + occupied +
232 " rate " + (available ? Math.floor(occupied * 100 / available) + "%" : "n/a");
233 }
234
235 invoice(x: Reservation): void {
236 this.emit(0, "INVOICE", x.id, x.guest, "room", x.room, this.rooms.get(x.room)!.kind, x.status);
237 if (x.status === "cancelled" || x.status === "noshow") {
238 this.emit(2, x.status === "cancelled" ? "cancellation" : "noshow", "fee", money(x.fee));
239 this.emit(2, "total", money(x.fee));
240 this.emit(2, this.deposit(x, x.fee));
241 return;
242 }
243 for (const n of x.nights) this.emit(2, dateWeek(n.day), "room", n.room, money(n.rate));
244 const b = x.status === "checkedout" ? x.bill! : this.bill(x);
245 this.emit(2, "lodging", money(b.lodging));
246 this.emit(2, "discount", money(b.discount));
247 if (x.penalty) this.emit(2, "penalty", dateWeek(x.penalty.day), money(b.penalty));
248 for (const e of x.extras) this.emit(2, "extra", e.item, money(e.amount));
249 this.emit(2, "tax", money(b.tax));
250 this.emit(2, "total", money(b.total));
251 this.emit(2, this.deposit(x, b.total));
252 }

254 command(cmd: string, a: string[]): void {
255 switch (cmd) {
256 case "ROOM": {
257 const number = Number(a[0]);
258 if (this.rooms.has(number)) throw new CommandError("duplicate room " + a[0]);
259 this.type(a[1]);
260 const base = cents(a[2]);
261 this.rooms.set(number, {number, kind: a[1], base, clean: true, blocks: new Set(), occupied: new Map()});
262 this.emit(0, "ROOM", number, a[1], money(base));
263 break;
264 }
265 case "RATE": {
266 const r = this.room(a[0]);
267 const old = r.base;
268 r.base = cents(a[1]);
269 this.emit(0, "RATE", r.number, money(old), "->", money(r.base));
270 break;
271 }
272 case "SEASON": {
273 const [start, end] = this.period(a[0], a[1], false, false);
274 this.seasons.push({start, end, percent: BigInt(a[2])});
275 this.emit(0, "SEASON", this.seasons.length, a[0], "->", a[1], a[2] + "%");
276 break;
277 }
278 case "BLOCK":
279 case "UNBLOCK": {
280 const r = this.room(a[0]);
281 const [start, end] = this.period(a[1], a[2], true);
282 if (cmd === "BLOCK") {
283 for (let day = start; day <= end; day++) {
284 if (r.occupied.has(day)) throw new CommandError("room unavailable " + r.number + " " + dateText(day));
285 }
286 }
287 const nights = new Set(Array.from({length: end - start + 1}, (_, i) => start + i));
288 const previous = r.blocks.size;
289 r.blocks = cmd === "BLOCK" ? r.blocks.union(nights) : r.blocks.difference(nights);
290 const count = Math.abs(r.blocks.size - previous);
291 this.emit(0, cmd, r.number, a[1], "->", a[2], cmd === "BLOCK" ? "added" : "removed", count);
292 break;
293 }
294 case "CLEAN": {
295 const r = this.room(a[0]);
296 if (r.clean) throw new CommandError("already clean " + r.number);
297 r.clean = true;
298 this.emit(0, "CLEAN", r.number);
299 break;
300 }
301 case "TODAY": {
302 const day = this.date(a[0]);
303 if (day < this.today) throw new CommandError("past date " + a[0]);
304 this.emit(0, "TODAY", dateWeek(day));
305 const checkouts = [...this.reservations.values()].filter(x => x.status === "inhouse" && x.end < day);
306 checkouts.sort((a, b) => a.end - b.end || lex(a.id, b.id));
307 for (const x of checkouts) this.checkout(x, x.end, 2);
308 const noshows = [...this.reservations.values()].filter(x => x.status === "booked" && x.start < day).sort(byArrival);
309 for (const x of noshows) {
310 x.fee = x.nights[0].rate;
311 x.paid = x.fee;
312 x.status = "noshow";
313 x.posting = x.start;
314 this.release(x);
315 this.emit(2, "NOSHOW", x.id, x.guest, "fee", money(x.fee), this.deposit(x, x.fee));
316 }
317 this.today = day;
318 break;
319 }
320 case "BOOK":
321 case "BOOKROOM":
322 case "WAITLIST": {
323 this.unused(a[0]);
324 let specific: Room | null = null;
325 if (cmd === "BOOKROOM") specific = this.room(a[2]);
326 else this.type(a[2]);
327 const [start, end] = this.stay(a[3], a[4]);
328 const r = specific ?? this.findRoom(a[2], start, end);
329 if (specific) {
330 const day = this.unavailable(specific, start, end);
331 if (day !== null) throw new CommandError("room unavailable " + specific.number + " " + dateText(day));
332 }
333 if (r) {
334 this.create(a[0], a[1], r, start, end);
335 } else if (cmd === "WAITLIST") {
336 this.queue.push({id: a[0], guest: a[1], kind: a[2], start, end, expiry: start});
337 this.used.add(a[0]);
338 this.emit(0, "WAITLIST", a[0], "queued", "position", this.queue.length);
339 } else {
340 throw new CommandError("no availability " + a[2] + " " + a[3] + " " + a[4]);
341 }
342 break;
343 }
344 case "MODIFY": {
345 const x = this.reservation(a[0]);
346 this.status(x, ["booked"]);
347 const [start, end] = this.stay(a[1], a[2]);
348 const old = this.rooms.get(x.room)!;
349 const kept = this.unavailable(old, start, end, x.id) === null;
350 const r = kept ? old : this.findRoom(old.kind, start, end, x.id);
351 if (!r) throw new CommandError("no availability " + old.kind + " " + a[1] + " " + a[2]);
352 this.release(x);
353 x.room = r.number;
354 x.start = start;
355 x.end = end;
356 x.nights = this.price(r, start, end);
357 this.occupy(x);
358 this.emit(0, "MODIFY", x.id, "room", r.number, kept ? "kept" : "reassigned", a[1], "->", a[2],
359 "nights", end - start, "total", money(lodging(x)));
360 break;
361 }
362 case "CANCEL": {
363 const i = this.queue.findIndex(w => w.id === a[0]);
364 if (i >= 0) {
365 this.queue.splice(i, 1);
366 this.emit(0, "CANCEL", a[0], "waitlist", "removed");
367 break;
368 }
369 const x = this.reservation(a[0]);
370 this.status(x, ["booked"]);
371 const days = x.start - this.today;
372 x.fee = days >= 7 ? 0n : days >= 1 ? x.nights[0].rate / 2n : x.nights[0].rate;
373 x.paid = x.fee;
374 x.status = "cancelled";
375 x.posting = this.today;
376 this.release(x);
377 this.emit(0, "CANCEL", x.id, "days", days, "fee", money(x.fee), this.deposit(x, x.fee));
378 break;
379 }
380 case "CHECKIN": {
381 const x = this.reservation(a[0]);
382 this.status(x, ["booked"]);
383 if (x.start !== this.today) throw new CommandError("not arrival date " + x.id + " " + dateText(x.start));
384 const occupant = this.inHouse(x.room);
385 if (occupant) throw new CommandError("room occupied " + x.room + " " + occupant.id);
386 if (!this.rooms.get(x.room)!.clean) throw new CommandError("room dirty " + x.room);
387 x.status = "inhouse";
388 this.emit(0, "CHECKIN", x.id, x.guest, "room", x.room, "until", dateText(x.end));
389 break;
390 }
391 case "MOVE": {
392 const x = this.reservation(a[0]);
393 this.status(x, ["inhouse"]);
394 const r = this.room(a[1]);
395 if (r.number === x.room) throw new CommandError("same room " + r.number);
396 const remaining = x.nights.filter(n => n.day >= this.today);
397 if (!remaining.length) throw new CommandError("no remaining nights " + x.id);
398 for (const n of remaining) {
399 if (!this.free(r, n.day, x.id)) throw new CommandError("room unavailable " + r.number + " " + dateText(n.day));
400 }
401 const occupant = this.inHouse(r.number);
402 if (occupant) throw new CommandError("room occupied " + r.number + " " + occupant.id);
403 if (!r.clean) throw new CommandError("room dirty " + r.number);
404 const old = x.room;
405 for (const n of remaining) {
406 this.rooms.get(n.room)!.occupied.delete(n.day);
407 n.room = r.number;
408 r.occupied.set(n.day, {owner: x.id, rate: n.rate});
409 }
410 this.emit(0, "MOVE", x.id, old, "->", r.number, "nights", remaining.length);
411 this.rooms.get(old)!.clean = false;
412 x.room = r.number;
413 break;
414 }
415 case "EXTEND": {
416 const x = this.reservation(a[0]);
417 this.status(x, ["inhouse"]);
418 const end = this.date(a[1]);
419 if (end <= x.end) throw new CommandError("invalid range " + dateText(x.end) + " " + a[1]);
420 if (end - x.start > 30) throw new CommandError("too long " + (end - x.start));
421 const r = this.rooms.get(x.room)!;
422 const unavailable = this.unavailable(r, x.end, end, x.id);
423 if (unavailable !== null) throw new CommandError("room unavailable " + r.number + " " + dateText(unavailable));
424 x.nights.push(...this.price(r, x.end, end));
425 x.end = end;
426 this.occupy(x);
427 this.emit(0, "EXTEND", x.id, "until", a[1], "nights", x.nights.length, "total", money(lodging(x)));
428 break;
429 }
430 case "CHARGE":
431 case "PAY": {
432 const x = this.reservation(a[0]);
433 this.status(x, cmd === "CHARGE" ? ["inhouse"] : ["booked", "inhouse"]);
434 const amount = cents(a[1]);
435 if (amount === 0n) throw new CommandError("invalid amount");
436 if (cmd === "CHARGE") {
437 x.extras.push({item: a[2], amount});
438 this.emit(0, "CHARGE", x.id, a[2], money(amount), "extras", money(extras(x)));
439 } else {
440 x.deposit += amount;
441 this.emit(0, "PAY", x.id, money(amount), "deposit", money(x.deposit));
442 }
443 break;
444 }
445 case "CHECKOUT": {
446 const x = this.reservation(a[0]);
447 this.status(x, ["inhouse"]);
448 this.checkout(x, this.today);
449 break;
450 }
451 case "INVOICE":
452 this.invoice(this.reservation(a[0]));
453 break;
454 case "AVAIL": {
455 this.type(a[0]);
456 const [start, end] = this.period(a[1], a[2]);
457 this.emit(0, "AVAIL", a[0], a[1], "->", a[2]);
458 for (let day = start; day <= end; day++) {
459 const rooms = this.sortedRooms().filter(r => r.kind === a[0] && this.free(r, day));
460 this.emit(2, dateWeek(day), "free", rooms.length, ...rooms.map(r => r.number));
461 }
462 break;
463 }
464 case "OCCUPANCY": {
465 const day = this.date(a[0]);
466 this.emit(0, "OCCUPANCY", dateWeek(day), this.occupancyText([...this.rooms.values()], day));
467 for (const kind of TYPES) {
468 const rooms = [...this.rooms.values()].filter(r => r.kind === kind);
469 if (rooms.length) this.emit(2, kind, this.occupancyText(rooms, day));
470 }
471 break;
472 }
473 case "STATS": {
474 const [start, end] = this.period(a[0], a[1]);
475 this.emit(0, "STATS", a[0], "->", a[1]);
476 const total = {available: 0, occupied: 0, revenue: 0n};
477 for (let day = start; day <= end; day++) {
478 const m = this.metrics(day);
479 this.emit(2, dateWeek(day), this.metricsText(m));
480 total.available += m.available;
481 total.occupied += m.occupied;
482 total.revenue += m.revenue;
483 }
484 this.emit(2, "total", this.metricsText(total));
485 break;
486 }
487 case "ROOMS": {
488 this.emit(0, "ROOMS", dateWeek(this.today), "rooms", this.rooms.size);
489 for (const r of this.sortedRooms()) {
490 const occupant = this.inHouse(r.number);
491 const situation = occupant ? "occupied " + occupant.id : r.blocks.has(this.today) ? "blocked" : "vacant";
492 const next = [...this.reservations.values()].filter(x => x.status === "booked" && x.room === r.number).sort(byArrival)[0];
493 this.emit(2, r.number, r.kind, "base", money(r.base), "tonight", money(this.rate(r, this.today)), r.clean ? "clean" : "dirty", situation,
494 ...(next ? ["next", next.id, dateText(next.start)] : []));
495 }
496 break;
497 }
498 case "LIST": {
499 const targets = [...this.reservations.values()].filter(x => x.status === a[0]).sort(byArrival);
500 this.emit(0, "LIST", a[0], targets.length);
501 for (const x of targets) this.emit(2, x.id, x.guest, "room", x.room, dateText(x.start), "->", dateText(x.end));
502 break;
503 }
504 case "QUEUE":
505 this.emit(0, "QUEUE", this.queue.length);
506 for (let i = 0; i < this.queue.length; i++) {
507 const w = this.queue[i];

509 }
510 break;
511 case "SCHEDULE": {
512 const day = this.date(a[0]);
513 const arrivals = [...this.reservations.values()].filter(x => x.start === day && x.status !== "cancelled").sort((a, b) => lex(a.id, b.id));
514 const departures = [...this.reservations.values()].filter(x => x.end === day && ["booked", "inhouse", "checkedout"].includes(x.status)).sort((a, b) => lex(a.id, b.id));
515 this.emit(0, "SCHEDULE", dateWeek(day), "arrivals", arrivals.length, "departures", departures.length);
516 for (const x of arrivals) this.emit(2, "arrival", x.id, x.guest, "room", x.room, x.status);
517 for (const x of departures) this.emit(2, "departure", x.id, x.guest, "room", x.room, x.status);
518 break;
519 }
520 case "GUEST": {
521 const targets = [...this.reservations.values()].filter(x => x.guest === a[0]);
522 if (!targets.length) throw new CommandError("unknown guest " + a[0]);
523 this.emit(0, "GUEST", a[0]);
524 for (const x of targets) this.emit(2, x.id, "room", x.room, this.rooms.get(x.room)!.kind, dateText(x.start), "->", dateText(x.end), x.status,
525 ["booked", "inhouse"].includes(x.status) ? "deposit" : "paid", money(["booked", "inhouse"].includes(x.status) ? x.deposit : x.paid));
526 const tier = this.tier(a[0]);
527 this.emit(2, "total", "paid", money(this.guestPaid(a[0])), "points", String(tier.points), "tier", tier.name);
528 break;
529 }
530 case "REVENUE": {
531 const [start, end] = this.period(a[0], a[1], false, false);
532 const targets = [...this.reservations.values()].filter(x => x.posting !== null && x.posting >= start && x.posting <= end);
533 const totals: Record<string, bigint> = {lodging: 0n, discount: 0n, penalty: 0n, extras: 0n, tax: 0n, cancellation: 0n, noshow: 0n};
534 let checkouts = 0;
535 let cancellations = 0;
536 let noshows = 0;
537 for (const x of targets) {
538 if (x.status === "checkedout") {
539 checkouts++;
540 for (const key of ["lodging", "discount", "penalty", "extras", "tax"]) totals[key] += x.bill![key as keyof Bill];
541 } else if (x.status === "cancelled") {
542 cancellations++;
543 totals.cancellation += x.fee;
544 } else if (x.status === "noshow") {
545 noshows++;
546 totals.noshow += x.fee;
547 }
548 }
549 this.emit(0, "REVENUE", a[0], "->", a[1]);
550 for (const key of ["lodging", "discount", "penalty", "extras", "tax", "cancellation", "noshow"]) this.emit(2, key, money(totals[key]));
551 const total = totals.lodging - totals.discount + totals.penalty + totals.extras + totals.tax + totals.cancellation + totals.noshow;
552 this.emit(2, "total", money(total));
553 this.emit(2, "checkouts", checkouts, "cancellations", cancellations, "noshows", noshows);
554 break;
555 }
556 }
