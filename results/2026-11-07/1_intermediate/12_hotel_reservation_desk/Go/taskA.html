15func leap(year int) bool {
16 return year%4 == 0 && (year%100 != 0 || year%400 == 0)
17}
18
19func monthLengths(year int) [12]int {
20 february := 28
21 if leap(year) {
22 february++
23 }
24 return [12]int{31, february, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31}
25}

27func dayNumber(text string) (int, error) {
28 parts := strings.Split(text, "-")
29 year, _ := strconv.Atoi(parts[0])
30 month, _ := strconv.Atoi(parts[1])
31 day, _ := strconv.Atoi(parts[2])
32 if year < 2000 || year > 2099 || month < 1 || month > 12 {
33 return 0, fmt.Errorf("invalid date %s", text)
34 }
35 lengths := monthLengths(year)
36 if day < 1 || day > lengths[month-1] {
37 return 0, fmt.Errorf("invalid date %s", text)
38 }
39 number := day - 1
40 for previous := 2000; previous < year; previous++ {
41 if leap(previous) {
42 number += 366
43 } else {
44 number += 365
45 }
46 }
47 for previous := 0; previous < month-1; previous++ {
48 number += lengths[previous]
49 }
50 return number, nil
51}

53func dateText(number int) string {
54 year := 2000
55 for {
56 days := 365
57 if leap(year) {
58 days++
59 }
60 if number < days {
61 break
62 }
63 number -= days
64 year++
65 }
66 month := 1
67 for _, days := range monthLengths(year) {
68 if number < days {
69 break
70 }
71 number -= days
72 month++
73 }
74 return fmt.Sprintf("%04d-%02d-%02d", year, month, number+1)
75}

77func dateWeekday(day int) string {
78 return dateText(day) + " " + weekdays[day%7]
79}

81func cents(text string) int {
82 parts := strings.Split(text, ".")
83 whole, _ := strconv.Atoi(parts[0])
84 fraction, _ := strconv.Atoi(parts[1])
85 return 100*whole + fraction
86}

88func money(value int) string {
89 return fmt.Sprintf("%d.%02d", value/100, value%100)
90}

92func stringLine(parts ...any) string {
93 texts := make([]string, len(parts))
94 for i, part := range parts {
95 texts[i] = fmt.Sprint(part)
96 }
97 return strings.Join(texts, " ")
98}

250func (desk *HotelDesk) stay(first, last string) (int, int, error) {
251 arrival, err := dayNumber(first)
252 if err != nil {
253 return 0, 0, err
254 }
255 departure, err := dayNumber(last)
256 if err != nil {
257 return 0, 0, err
258 }
259 switch {
260 case arrival < desk.Today:
261 return 0, 0, fmt.Errorf("past date %s", first)
262 case departure <= arrival:
263 return 0, 0, fmt.Errorf("invalid range %s %s", first, last)
264 case departure-arrival > 30:
265 return 0, 0, fmt.Errorf("too long %d", departure-arrival)
266 }
267 return arrival, departure, nil
268}

270func (desk *HotelDesk) period(first, last string, pastRestricted, limited bool) (int, int, error) {
271 start, err := dayNumber(first)
272 if err != nil {
273 return 0, 0, err
274 }
275 end, err := dayNumber(last)
276 if err != nil {
277 return 0, 0, err
278 }
279 switch {
280 case pastRestricted && start < desk.Today:
281 return 0, 0, fmt.Errorf("past date %s", first)
282 case end < start:
283 return 0, 0, fmt.Errorf("invalid range %s %s", first, last)
284 case limited && end-start+1 > 31:
285 return 0, 0, fmt.Errorf("too long %d", end-start+1)
286 }
287 return start, end, nil
288}

290func (desk *HotelDesk) nightRate(room *Room, day int) int {
291 percentage := 100
292 for i := len(desk.Seasons) - 1; i >= 0; i-- {
293 season := desk.Seasons[i]
294 if season.Start <= day && day <= season.End {
295 percentage = season.Percentage
296 break
297 }
298 }
299 return room.Rate * percentage / 100
300}

303 nights := make([]Night, 0, departure-arrival)
304 for day := arrival; day < departure; day++ {
305 nights = append(nights, Night{day, room.Number, desk.nightRate(room, day)})

310func (desk *HotelDesk) firstUnavailable(
311 room *Room, arrival, departure int, exceptFor *Reservation,
312) *int {
313 for day := arrival; day < departure; day++ {
314 key := NightKey{day, room.Number}
315 if desk.Blocks[key] {
316 return &day
317 }
318 if occupied, ok := desk.Occupied[key]; ok && occupied.Reservation != exceptFor {
319 return &day
320 }
321 }
322 return nil
323}

360func (desk *HotelDesk) release(reservation *Reservation, first *int) {

368func (desk *HotelDesk) createReservation(
369 id, guest string, room *Room, arrival, departure int, indent bool,
370) {
371 nights := desk.price(room, arrival, departure)
372 reservation := &Reservation{
373 ID: id, Guest: guest, Room: room,
374 Arrival: arrival, Departure: departure, Nights: nights, Status: "booked",
375 }
376 desk.Reservations[id] = reservation
377 desk.ReservationOrder = append(desk.ReservationOrder, reservation)
378 desk.UsedIDs[id] = true
379 desk.occupy(reservation, nights)
380 parts := []any{
381 "BOOK", id, guest, "room", room.Number, room.Kind,
382 dateText(arrival), "->", dateText(departure), "nights", departure - arrival,
383 "total", money(reservation.lodging()),
384 }
385 if indent {
386 desk.emitIndented(parts...)
387 } else {
388 desk.emit(parts...)
389 }
390}

402func (desk *HotelDesk) tier(guest string) string {
403 points := desk.guestTotal(guest) / 100
404 switch {
405 case points >= 2000:
406 return "GOLD"
407 case points >= 500:
408 return "SILVER"
409 default:
410 return "BRONZE"
411 }
412}

414func (desk *HotelDesk) calculateBill(reservation *Reservation) *Bill {
415 lodging := reservation.lodging()
416 percentage := 0
417 switch desk.tier(reservation.Guest) {
418 case "GOLD":
419 percentage = 10
420 case "SILVER":
421 percentage = 5
422 }
423 discount := lodging * percentage / 100
424 penalty := 0
425 if reservation.Penalty != nil {
426 penalty = reservation.Penalty.Rate
427 }
428 extras := 0
429 for _, extra := range reservation.Extras {
430 extras += extra.Amount
431 }
432 subtotal := lodging - discount + penalty + extras
433 tax := subtotal / 10
434 return &Bill{lodging, discount, penalty, extras, tax, subtotal + tax}
435}

437func depositParts(reservation *Reservation, amount int) []any {
438 if amount >= reservation.Deposit {
439 return []any{"deposit", money(reservation.Deposit), "due", money(amount - reservation.Deposit)}
440 }
441 return []any{"deposit", money(reservation.Deposit), "refund", money(reservation.Deposit - amount)}
442}

444func (desk *HotelDesk) checkOut(reservation *Reservation, day int, indent bool) {
445 if day < reservation.Departure {
446 for _, night := range reservation.Nights {
447 if night.Day == day {
448 penalty := night
449 reservation.Penalty = &penalty
450 break
451 }
452 }
453 desk.release(reservation, &day)
454 past := make([]Night, 0, len(reservation.Nights))
455 for _, night := range reservation.Nights {
456 if night.Day < day {
457 past = append(past, night)
458 }
459 }
460 reservation.Nights = past
461 reservation.Departure = day
462 }
463 bill := desk.calculateBill(reservation)
464 reservation.Bill = bill
465 reservation.Paid = bill.Total
466 reservation.Status = "checkedout"
467 reservation.Posting = &day
468 reservation.Room.Clean = false
469 parts := []any{
470 "CHECKOUT", reservation.ID, "nights", len(reservation.Nights),
471 "lodging", money(bill.Lodging), "discount", money(bill.Discount),
472 "penalty", money(bill.Penalty), "extras", money(bill.Extras),
473 "tax", money(bill.Tax), "total", money(bill.Total),
474 }
475 parts = append(parts, depositParts(reservation, bill.Total)...)
476 if indent {
477 desk.emitIndented(parts...)
478 } else {
479 desk.emit(parts...)
480 }
481}

483func (desk *HotelDesk) processWaitlist() {
484 remaining := make([]WaitingEntry, 0, len(desk.Waitlist))
485 for _, entry := range desk.Waitlist {
486 if entry.Expiry < desk.Today {
487 desk.emitIndented("WAITLIST", entry.ID, "expired")
488 continue
489 }
490 room := desk.freeRoom(entry.Kind, entry.Arrival, entry.Departure, nil)
491 if room == nil {
492 remaining = append(remaining, entry)
493 } else {
494 desk.createReservation(entry.ID, entry.Guest, room, entry.Arrival, entry.Departure, true)
495 }
496 }
497 desk.Waitlist = remaining
498}

535func (desk *HotelDesk) cmdBlock(a []string) error {
536 room, err := desk.requireRoom(a[0])
537 if err != nil {
538 return err
539 }
540 start, end, err := desk.period(a[1], a[2], true, true)
541 if err != nil {
542 return err
543 }
544 for day := start; day <= end; day++ {
545 if _, exists := desk.Occupied[NightKey{day, room.Number}]; exists {
546 return fmt.Errorf("room unavailable %s %s", a[0], dateText(day))
547 }
548 }
549 added := 0
550 for day := start; day <= end; day++ {
551 key := NightKey{day, room.Number}
552 if !desk.Blocks[key] {
553 desk.Blocks[key] = true
554 added++
555 }
556 }
557 desk.emit("BLOCK", a[0], a[1], "->", a[2], "added", added)
558 return nil
559}

595func (desk *HotelDesk) cmdToday(a []string) error {
596 day, err := dayNumber(a[0])
597 if err != nil {
598 return err
599 }
600 if day < desk.Today {
601 return fmt.Errorf("past date %s", a[0])
602 }
603 desk.emit("TODAY", dateWeekday(day))
604 var departing []*Reservation
605 for _, reservation := range desk.ReservationOrder {
606 if reservation.Status == "inhouse" && reservation.Departure < day {
607 departing = append(departing, reservation)
608 }
609 }
610 sort.Slice(departing, func(i, j int) bool {
611 if departing[i].Departure != departing[j].Departure {
612 return departing[i].Departure < departing[j].Departure
613 }
614 return departing[i].ID < departing[j].ID
615 })
616 for _, reservation := range departing {
617 desk.checkOut(reservation, reservation.Departure, true)
618 }
619 var absent []*Reservation
620 for _, reservation := range desk.ReservationOrder {
621 if reservation.Status == "booked" && reservation.Arrival < day {
622 absent = append(absent, reservation)
623 }
624 }
625 sort.Slice(absent, func(i, j int) bool {
626 if absent[i].Arrival != absent[j].Arrival {
627 return absent[i].Arrival < absent[j].Arrival
628 }
629 return absent[i].ID < absent[j].ID
630 })
631 for _, reservation := range absent {
632 desk.release(reservation, nil)
633 reservation.Fee = reservation.firstRate()
634 reservation.Paid = reservation.Fee
635 reservation.Status = "noshow"
636 posting := reservation.Arrival
637 reservation.Posting = &posting
638 parts := []any{"NOSHOW", reservation.ID, reservation.Guest, "fee", money(reservation.Fee)}
639 desk.emitIndented(append(parts, depositParts(reservation, reservation.Fee)...)...)
640 }
641 desk.Today = day
642 return nil
643}

706func (desk *HotelDesk) cmdModify(a []string) error {
707 reservation, err := desk.reservation(a[0], "booked")
708 if err != nil {
709 return err
710 }
711 arrival, departure, err := desk.stay(a[1], a[2])
712 if err != nil {
713 return err
714 }
715 room := reservation.Room
716 choice := "kept"
717 if desk.firstUnavailable(room, arrival, departure, reservation) != nil {
718 room = desk.freeRoom(room.Kind, arrival, departure, reservation)
719 if room == nil {
720 return fmt.Errorf("no availability %s %s %s", reservation.Room.Kind, a[1], a[2])
721 }
722 choice = "reassigned"
723 }
724 nights := desk.price(room, arrival, departure)
725 desk.release(reservation, nil)
726 reservation.Room = room
727 reservation.Arrival = arrival
728 reservation.Departure = departure
729 reservation.Nights = nights
730 desk.occupy(reservation, nights)
731 desk.emit("MODIFY", a[0], "room", room.Number, choice, a[1], "->", a[2],
732 "nights", departure-arrival, "total", money(reservation.lodging()))
733 return nil
734}

736func (desk *HotelDesk) cmdCancel(a []string) error {
737 for position, entry := range desk.Waitlist {
738 if entry.ID == a[0] {
739 desk.Waitlist = append(desk.Waitlist[:position], desk.Waitlist[position+1:]...)
740 desk.emit("CANCEL", a[0], "waitlist removed")
741 return nil
742 }
743 }
744 reservation, err := desk.reservation(a[0], "booked")
745 if err != nil {
746 return err
747 }
748 days := reservation.Arrival - desk.Today
749 firstRate := reservation.firstRate()
750 fee := firstRate
751 if days >= 7 {
752 fee = 0
753 } else if days >= 1 {
754 fee = firstRate / 2
755 }
756 desk.release(reservation, nil)
757 reservation.Fee = fee
758 reservation.Paid = fee
759 reservation.Status = "cancelled"
760 posting := desk.Today
761 reservation.Posting = &posting
762 parts := []any{"CANCEL", a[0], "days", days, "fee", money(fee)}
763 desk.emit(append(parts, depositParts(reservation, fee)...)...)
764 return nil
765}

769 if err != nil {
770 return err
771 }
772 if reservation.Arrival != desk.Today {
773 return fmt.Errorf("not arrival date %s %s", a[0], dateText(reservation.Arrival))
774 }
775 if occupant := desk.inhouse(reservation.Room); occupant != nil {
776 return fmt.Errorf("room occupied %d %s", reservation.Room.Number, occupant.ID)
777 }

779 return fmt.Errorf("room dirty %d", reservation.Room.Number)

782 desk.emit("CHECKIN", a[0], reservation.Guest, "room", reservation.Room.Number,
783 "until", dateText(reservation.Departure))
784 return nil

787func (desk *HotelDesk) cmdMove(a []string) error {
788 reservation, err := desk.reservation(a[0], "inhouse")
789 if err != nil {
790 return err
791 }
792 room, err := desk.requireRoom(a[1])
793 if err != nil {
794 return err
795 }
796 previous := reservation.Room
797 if room == previous {
798 return fmt.Errorf("same room %s", a[1])
799 }
800 var remaining []Night
801 for _, night := range reservation.Nights {
802 if night.Day >= desk.Today {
803 remaining = append(remaining, night)
804 }
805 }
806 if len(remaining) == 0 {
807 return fmt.Errorf("no remaining nights %s", a[0])
808 }
809 if unavailable := desk.firstUnavailable(
810 room, remaining[0].Day, reservation.Departure, reservation,
811 ); unavailable != nil {
812 return fmt.Errorf("room unavailable %s %s", a[1], dateText(*unavailable))
813 }
814 if occupant := desk.inhouse(room); occupant != nil {
815 return fmt.Errorf("room occupied %s %s", a[1], occupant.ID)
816 }
817 if !room.Clean {
818 return fmt.Errorf("room dirty %s", a[1])
819 }
820 desk.release(reservation, &desk.Today)
821 for i, night := range reservation.Nights {
822 if night.Day >= desk.Today {
823 reservation.Nights[i].Room = room.Number
824 }
825 }
826 var future []Night
827 for _, night := range reservation.Nights {
828 if night.Day >= desk.Today {
829 future = append(future, night)
830 }
831 }
832 desk.occupy(reservation, future)
833 previous.Clean = false
834 reservation.Room = room
835 desk.emit("MOVE", a[0], previous.Number, "->", a[1], "nights", len(remaining))
836 return nil
837}

1193func (desk *HotelDesk) cmdRevenue(a []string) error {
1194 start, end, err := desk.period(a[0], a[1], false, false)
1195 if err != nil {
1196 return err
1197 }
1198 var lodging, discount, penalty, extras, tax int
1199 var cancellations, noshows int
1200 var checkouts, cancelled, absent int
1201 for _, reservation := range desk.ReservationOrder {
1202 if reservation.Posting == nil || *reservation.Posting < start || *reservation.Posting > end {
1203 continue
1204 }
1205 switch reservation.Status {
1206 case "checkedout":
1207 checkouts++
1208 lodging += reservation.Bill.Lodging
1209 discount += reservation.Bill.Discount
1210 penalty += reservation.Bill.Penalty
1211 extras += reservation.Bill.Extras
1212 tax += reservation.Bill.Tax
1213 case "cancelled":
1214 cancelled++
1215 cancellations += reservation.Fee
1216 case "noshow":
1217 absent++
1218 noshows += reservation.Fee
1219 }
1220 }
1221 desk.emit("REVENUE", a[0], "->", a[1])
1222 desk.emitIndented("lodging", money(lodging))
1223 desk.emitIndented("discount", money(discount))
1224 desk.emitIndented("penalty", money(penalty))
1225 desk.emitIndented("extras", money(extras))
1226 desk.emitIndented("tax", money(tax))
1227 desk.emitIndented("cancellation", money(cancellations))
1228 desk.emitIndented("noshow", money(noshows))
1229 desk.emitIndented("total", money(lodging-discount+penalty+extras+tax+cancellations+noshows))
1230 desk.emitIndented("checkouts", checkouts, "cancellations", cancelled, "noshows", absent)
