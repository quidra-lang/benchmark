27 private static boolean leap(int year) {
28 return year % 4 == 0 && (year % 100 != 0 || year % 400 == 0);
29 }

31 private static int[] monthLengths(int year) {
32 return new int[]{31, leap(year) ? 29 : 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
33 }

35 private static int dayNumber(String text) {
36 String[] parts = text.split("-");
37 int year = Integer.parseInt(parts[0]);
38 int month = Integer.parseInt(parts[1]);
39 int day = Integer.parseInt(parts[2]);
40 if (year < 2000 || year > 2099 || month < 1 || month > 12) {
41 throw new CommandError("invalid date " + text);
42 }
43 int[] lengths = monthLengths(year);
44 if (day < 1 || day > lengths[month - 1]) {
45 throw new CommandError("invalid date " + text);
46 }
47 int number = day - 1;
48 for (int previous = 2000; previous < year; ++previous) {
49 number += leap(previous) ? 366 : 365;
50 }
51 for (int previous = 0; previous < month - 1; ++previous) {
52 number += lengths[previous];
53 }
54 return number;
55 }
56
57 private static String dateText(int number) {
58 int year = 2000;
59 while (true) {
60 int days = leap(year) ? 366 : 365;
61 if (number < days) {
62 break;
63 }
64 number -= days;
65 ++year;
66 }
67 int month = 1;
68 for (int days : monthLengths(year)) {
69 if (number < days) {
70 break;
71 }
72 number -= days;
73 ++month;
74 }
75 return String.format(Locale.ROOT, "%04d-%02d-%02d", year, month, number + 1);
76 }

78 private static String dateWeekday(int day) {
79 return dateText(day) + " " + WEEKDAYS[day % 7];

200 private int[] stay(String first, String last) {
201 int arrival = dayNumber(first);
202 int departure = dayNumber(last);
203 if (arrival < today) {
204 throw new CommandError("past date " + first);
205 }
206 if (departure <= arrival) {
207 throw new CommandError("invalid range " + first + " " + last);
208 }
209 if (departure - arrival > 30) {
210 throw new CommandError("too long " + (departure - arrival));
211 }
212 return new int[]{arrival, departure};
213 }

215 private int[] period(String first, String last, boolean pastRestricted, boolean limited) {
216 int start = dayNumber(first);
217 int end = dayNumber(last);
218 if (pastRestricted && start < today) {
219 throw new CommandError("past date " + first);
220 }
221 if (end < start) {
222 throw new CommandError("invalid range " + first + " " + last);
223 }
224 if (limited && end - start + 1 > 31) {
225 throw new CommandError("too long " + (end - start + 1));
226 }
227 return new int[]{start, end};
228 }

230 private long nightRate(Room room, int day) {
231 int percentage = 100;
232 for (int i = seasons.size() - 1; i >= 0; --i) {
233 Season season = seasons.get(i);
234 if (season.start() <= day && day <= season.end()) {
235 percentage = season.percentage();
236 break;
237 }
238 }
239 return room.rate * percentage / 100;
240 }

250 private Integer firstUnavailable(
251 Room room, int arrival, int departure, Reservation exceptFor
252 ) {
253 for (int day = arrival; day < departure; ++day) {
254 NightKey key = new NightKey(day, room.number);
255 if (blocks.contains(key)) {
256 return day;
257 }
258 Occupied night = occupied.get(key);
259 if (night != null && night.reservation() != exceptFor) {
260 return day;
261 }
262 }
263 return null;
264 }

270 if (room.kind.equals(kind) &&
271 firstUnavailable(room, arrival, departure, exceptFor) == null) {
272 return room;
273 }
274 }
275 return null;
276 }

328 private String tier(String guest) {
329 long points = guestTotal(guest) / 100;
330 return points >= 2000 ? "GOLD" : points >= 500 ? "SILVER" : "BRONZE";

333 private Bill calculateBill(Reservation reservation) {
334 long lodging = reservation.lodging();
335 int percentage = switch (tier(reservation.guest)) {
336 case "GOLD" -> 10;
337 case "SILVER" -> 5;
338 default -> 0;
339 };
340 long discount = lodging * percentage / 100;
341 long penalty = reservation.penalty == null ? 0 : reservation.penalty.rate();
342 long extras = reservation.extras.stream().mapToLong(Extra::amount).sum();
343 long subtotal = lodging - discount + penalty + extras;
344 long tax = subtotal / 10;
345 return new Bill(lodging, discount, penalty, extras, tax, subtotal + tax);
346 }

361 private void checkOut(Reservation reservation, int day, boolean indent) {
362 if (day < reservation.departure) {
363 for (Night night : reservation.nights) {
364 if (night.day() == day) {
365 reservation.penalty = night;
366 break;
367 }
368 }
369 release(reservation, day);
370 reservation.nights.removeIf(night -> night.day() >= day);
371 reservation.departure = day;
372 }
373 Bill bill = calculateBill(reservation);
374 reservation.bill = bill;
375 reservation.paid = bill.total();
376 reservation.status = "checkedout";
377 reservation.posting = day;
378 reservation.room.clean = false;
379 List<Object> parts = new ArrayList<>(Arrays.asList(
380 "CHECKOUT", reservation.id, "nights", reservation.nights.size(),
381 "lodging", money(bill.lodging()), "discount", money(bill.discount()),
382 "penalty", money(bill.penalty()), "extras", money(bill.extras()),
383 "tax", money(bill.tax()), "total", money(bill.total())
384 ));
385 parts.addAll(Arrays.asList(depositParts(reservation, bill.total())));
386 if (indent) {
387 emitIndented(parts.toArray());
388 } else {
389 emit(parts.toArray());
390 }
391 }

394 List<WaitingEntry> remaining = new ArrayList<>();
395 for (WaitingEntry entry : waitlist) {
396 if (entry.expiry() < today) {
397 emitIndented("WAITLIST", entry.id(), "expired");
398 } else {
399 Room room = freeRoom(entry.kind(), entry.arrival(), entry.departure(), null);
400 if (room == null) {
401 remaining.add(entry);

403 createReservation(

405 );
406 }
407 }
408 }
409 waitlist.clear();
410 waitlist.addAll(remaining);

474 private void cmdToday(String[] a) {
475 int day = dayNumber(a[0]);
476 if (day < today) {
477 throw new CommandError("past date " + a[0]);
478 }
479 emit("TODAY", dateWeekday(day));
480 List<Reservation> departing = reservations.values().stream()
481 .filter(r -> r.status.equals("inhouse") && r.departure < day)
482 .sorted(Comparator.comparingInt((Reservation r) -> r.departure).thenComparing(r -> r.id))
483 .toList();
484 for (Reservation reservation : departing) {
485 checkOut(reservation, reservation.departure, true);
486 }
487 List<Reservation> absent = reservations.values().stream()
488 .filter(r -> r.status.equals("booked") && r.arrival < day)
489 .sorted(Comparator.comparingInt((Reservation r) -> r.arrival).thenComparing(r -> r.id))
490 .toList();
491 for (Reservation reservation : absent) {
492 release(reservation, null);
493 reservation.fee = reservation.firstRate();
494 reservation.paid = reservation.fee;
495 reservation.status = "noshow";
496 reservation.posting = reservation.arrival;
497 List<Object> parts = new ArrayList<>(Arrays.asList(
498 "NOSHOW", reservation.id, reservation.guest, "fee", money(reservation.fee)
499 ));
500 parts.addAll(Arrays.asList(depositParts(reservation, reservation.fee)));
501 emitIndented(parts.toArray());
502 }
503 today = day;
504 }

615 Room previous = reservation.room;
616 if (room == previous) {
617 throw new CommandError("same room " + a[1]);
618 }
619 List<Night> remaining = reservation.nights.stream()
620 .filter(night -> night.day() >= today).toList();
621 if (remaining.isEmpty()) {
622 throw new CommandError("no remaining nights " + a[0]);
623 }

625 room, remaining.get(0).day(), reservation.departure, reservation
626 );
627 if (unavailable != null) {
628 throw new CommandError("room unavailable " + a[1] + " " + dateText(unavailable));
629 }

631 if (occupant != null) {
632 throw new CommandError("room occupied " + a[1] + " " + occupant.id);
633 }
634 if (!room.clean) {

637 release(reservation, today);
638 List<Night> updated = new ArrayList<>();
639 for (Night night : reservation.nights) {
640 updated.add(night.day() >= today
641 ? new Night(night.day(), room.number, night.rate()) : night);
642 }
643 reservation.nights = updated;
644 occupy(reservation, updated.stream().filter(night -> night.day() >= today).toList());
645 previous.clean = false;
646 reservation.room = room;
647 emit("MOVE", a[0], previous.number, "->", a[1], "nights", remaining.size());
648 }

705 private void cmdInvoice(String[] a) {
706 Reservation reservation = reservation(a[0], null);
707 emit("INVOICE", a[0], reservation.guest, "room", reservation.room.number,

935 private void cmdRevenue(String[] a) {
936 int[] dates = period(a[0], a[1], false, false);
937 long lodging = 0, discount = 0, penalty = 0, extras = 0, tax = 0;
938 long cancellations = 0, noshows = 0;
939 int checkouts = 0, cancelled = 0, absent = 0;
940 for (Reservation reservation : reservations.values()) {
941 if (reservation.posting == null ||
942 reservation.posting < dates[0] || reservation.posting > dates[1]) {
943 continue;
944 }
945 switch (reservation.status) {
946 case "checkedout":
947 ++checkouts;
948 lodging += reservation.bill.lodging();
949 discount += reservation.bill.discount();
950 penalty += reservation.bill.penalty();
951 extras += reservation.bill.extras();
952 tax += reservation.bill.tax();
953 break;
954 case "cancelled":
955 ++cancelled;
956 cancellations += reservation.fee;
957 break;
958 case "noshow":
959 ++absent;
960 noshows += reservation.fee;
961 break;
962 default:
963 break;
964 }
965 }
966 emit("REVENUE", a[0], "->", a[1]);
967 emitIndented("lodging", money(lodging));
968 emitIndented("discount", money(discount));
969 emitIndented("penalty", money(penalty));
970 emitIndented("extras", money(extras));
971 emitIndented("tax", money(tax));
972 emitIndented("cancellation", money(cancellations));
973 emitIndented("noshow", money(noshows));
974 emitIndented("total", money(
975 lodging - discount + penalty + extras + tax + cancellations + noshows
976 ));
977 emitIndented("checkouts", checkouts, "cancellations", cancelled, "noshows", absent);
