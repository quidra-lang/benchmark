1TYPES = %w[SINGLE DOUBLE SUITE].freeze
2WEEKDAYS = %w[SAT SUN MON TUE WED THU FRI].freeze
3DATES = []
4DAYS = {}
5(2000..2099).each do |year|
6 leap = (year % 4).zero? && (!(year % 100).zero? || (year % 400).zero?)
7 lengths = [31, leap ? 29 : 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31]
8 (1..12).each do |month|
9 (1..lengths[month - 1]).each do |day|
10 text = format('%04d-%02d-%02d', year, month, day)
11 DAYS[text] = DATES.length
12 DATES << text
13 end
14 end
15end
16Room = Struct.new(:number, :kind, :base, :clean, :blocks, :occupied)
17Night = Struct.new(:day, :room, :rate)
18Reservation = Struct.new(:id, :guest, :room, :start, :finish, :nights, :status,
19 :extras, :deposit, :paid, :posting, :fee, :penalty, :bill)
20Waiting = Struct.new(:id, :guest, :kind, :start, :finish, :expiry)
21class CommandError < StandardError; end

23def date_text(day)
24 DATES[day]
25end
26
27def date_week(day)
28 "#{date_text(day)} #{WEEKDAYS[day % 7]}"
29end

31def cents(text)
32 whole, fraction = text.split('.')
33 whole.to_i * 100 + fraction.to_i
34end
35
36def money(value)
37 format('%d.%02d', value / 100, value % 100)
38end
39
40def lodging(x)
41 x.nights.sum(&:rate)
42end
43
44def extras(x)
45 x.extras.sum { |_, amount| amount }
46end

48class Hotel
49 attr_reader :output
50
51 def initialize
52 @today = DAYS['2026-01-01']
53 @rooms = {}
54 @seasons = []
55 @reservations = {}
56 @queue = []
57 @used = {}
58 @output = []
59 end
60
61 def emit(indent, *parts)
62 @output << ' ' * indent + parts.join(' ')
63 end
64
65 def date(text)
66 raise CommandError, "invalid date #{text}" unless DAYS.key?(text)
67 DAYS[text]
68 end
69
70 def room(text)
71 raise CommandError, "unknown room #{text}" unless @rooms.key?(text.to_i)
72 @rooms[text.to_i]
73 end
74
75 def reservation(id)
76 raise CommandError, "unknown reservation #{id}" unless @reservations.key?(id)
77 @reservations[id]
78 end
79
80 def status(x, allowed)
81 raise CommandError, "invalid status #{x.id} #{x.status}" unless allowed.include?(x.status)
82 end
83
84 def type(kind)
85 raise CommandError, "invalid type #{kind}" unless TYPES.include?(kind)
86 end
87
88 def unused(id)
89 raise CommandError, "duplicate reservation #{id}" if @used.key?(id)
90 end
91
92 def stay(a, b)
93 start = date(a)
94 finish = date(b)
95 raise CommandError, "past date #{a}" if start < @today
96 raise CommandError, "invalid range #{a} #{b}" if finish <= start
97 raise CommandError, "too long #{finish - start}" if finish - start > 30
98 [start, finish]
99 end
100
101 def period(a, b, future = false, limited = true)
102 start = date(a)
103 finish = date(b)
104 raise CommandError, "past date #{a}" if future && start < @today
105 raise CommandError, "invalid range #{a} #{b}" if finish < start
106 raise CommandError, "too long #{finish - start + 1}" if limited && finish - start + 1 > 31
107 [start, finish]
108 end

110 def sorted_rooms
111 @rooms.values.sort_by(&:number)
112 end
113
114 def rate(r, day)
115 season = @seasons.reverse_each.find { |start, finish, _| start <= day && day <= finish }
116 percent = season ? season[2] : 100
117 r.base * percent / 100
118 end
119
120 def price(r, start, finish)
121 (start...finish).map { |day| Night.new(day, r.number, rate(r, day)) }
122 end
123
124 def free?(r, day, except = nil)
125 night = r.occupied[day]
126 !r.blocks.key?(day) && (night.nil? || night[0] == except)
127 end
128
129 def unavailable(r, start, finish, except = nil)
130 (start...finish).find { |day| !free?(r, day, except) }
131 end
132
133 def find_room(kind, start, finish, except = nil)
134 sorted_rooms.find { |r| r.kind == kind && unavailable(r, start, finish, except).nil? }
135 end
136
137 def in_house(number)
138 @reservations.values.find { |x| x.status == 'inhouse' && x.room == number }
139 end
140
141 def release(x)
142 x.nights.each { |n| @rooms[n.room].occupied.delete(n.day) }
143 end
144
145 def occupy(x)
146 x.nights.each { |n| @rooms[n.room].occupied[n.day] = [x.id, n.rate] }
147 end
148
149 def create(id, guest, r, start, finish, indent = 0)
150 x = Reservation.new(id, guest, r.number, start, finish, price(r, start, finish),
151 'booked', [], 0, 0, nil, 0, nil, nil)
152 @reservations[id] = x
153 @used[id] = true
154 occupy(x)
155 emit(indent, 'BOOK', id, guest, 'room', r.number, r.kind, date_text(start), '->',
156 date_text(finish), 'nights', finish - start, 'total', money(lodging(x)))
157 end
158
159 def guest_paid(guest)
160 @reservations.values.select { |x| x.guest == guest }.sum(&:paid)
161 end
162
163 def tier(guest)
164 points = guest_paid(guest) / 100
165 if points >= 2000
166 [points, 'GOLD', 10]
167 elsif points >= 500
168 [points, 'SILVER', 5]
169 else
170 [points, 'BRONZE', 0]
171 end
172 end
173
174 def bill(x)
175 l = lodging(x)
176 discount = l * tier(x.guest)[2] / 100
177 penalty = x.penalty ? x.penalty.rate : 0
178 e = extras(x)
179 subtotal = l - discount + penalty + e
180 tax = subtotal / 10
181 {lodging: l, discount: discount, penalty: penalty, extras: e, tax: tax, total: subtotal + tax}
182 end
183
184 def deposit_text(x, amount)
185 if amount >= x.deposit
186 "deposit #{money(x.deposit)} due #{money(amount - x.deposit)}"
187 else
188 "deposit #{money(x.deposit)} refund #{money(x.deposit - amount)}"
189 end
190 end
191
192 def checkout(x, day, indent = 0)
193 if day < x.finish
194 x.penalty = x.nights.find { |n| n.day == day }
195 x.nights.select { |n| n.day >= day }.each { |n| @rooms[n.room].occupied.delete(n.day) }
196 x.nights = x.nights.select { |n| n.day < day }
197 x.finish = day
198 end
199 b = bill(x)
200 x.bill = b
201 x.paid = b[:total]
202 x.status = 'checkedout'
203 x.posting = day
204 @rooms[x.room].clean = false
205 emit(indent, 'CHECKOUT', x.id, 'nights', x.nights.length, 'lodging', money(b[:lodging]),
206 'discount', money(b[:discount]), 'penalty', money(b[:penalty]), 'extras', money(b[:extras]),
207 'tax', money(b[:tax]), 'total', money(b[:total]), deposit_text(x, b[:total]))
208 end
209
210 def process_queue
211 remaining = []
212 @queue.each do |w|
213 if w.expiry < @today
214 emit(2, 'WAITLIST', w.id, 'expired')
215 else
216 r = find_room(w.kind, w.start, w.finish)
217 if r
218 create(w.id, w.guest, r, w.start, w.finish, 2)
219 else
220 remaining << w
221 end
222 end
223 end
224 @queue = remaining
225 end
226
227 def metrics(day)
228 nights = @rooms.values.filter_map { |r| r.occupied[day] }
229 available = @rooms.values.count { |r| !r.blocks.key?(day) }
230 [available, nights.length, nights.sum(&:last)]
231 end
232
233 def metrics_text(values)
234 available, occupied, revenue = values
235 occupancy = available.positive? ? "#{occupied * 100 / available}%" : 'n/a'
236 adr = occupied.positive? ? money(revenue / occupied) : 'n/a'
237 revpar = available.positive? ? money(revenue / available) : 'n/a'
238 "sold #{occupied}/#{available} rate #{occupancy} revenue #{money(revenue)} adr #{adr} revpar #{revpar}"
239 end
240
241 def occupancy_text(rooms, day)
242 blocked = rooms.count { |r| r.blocks.key?(day) }
243 occupied = rooms.count { |r| r.occupied.key?(day) }
244 available = rooms.length - blocked
245 occupancy = available.positive? ? "#{occupied * 100 / available}%" : 'n/a'
246 "rooms #{rooms.length} blocked #{blocked} occupied #{occupied} rate #{occupancy}"
247 end
248
249 def invoice(x)
250 emit(0, 'INVOICE', x.id, x.guest, 'room', x.room, @rooms[x.room].kind, x.status)
251 if %w[cancelled noshow].include?(x.status)
252 emit(2, x.status == 'cancelled' ? 'cancellation' : 'noshow', 'fee', money(x.fee))
253 emit(2, 'total', money(x.fee))
254 emit(2, deposit_text(x, x.fee))
255 return
256 end
257 x.nights.each { |n| emit(2, date_week(n.day), 'room', n.room, money(n.rate)) }
258 b = x.status == 'checkedout' ? x.bill : bill(x)
259 emit(2, 'lodging', money(b[:lodging]))
260 emit(2, 'discount', money(b[:discount]))
261 emit(2, 'penalty', date_week(x.penalty.day), money(b[:penalty])) if x.penalty
262 x.extras.each { |item, amount| emit(2, 'extra', item, money(amount)) }
263 emit(2, 'tax', money(b[:tax]))
264 emit(2, 'total', money(b[:total]))
265 emit(2, deposit_text(x, b[:total]))
266 end

268 def command(cmd, a)
269 case cmd
270 when 'ROOM'
271 number = a[0].to_i
272 raise CommandError, "duplicate room #{a[0]}" if @rooms.key?(number)
273 type(a[1])
274 base = cents(a[2])
275 @rooms[number] = Room.new(number, a[1], base, true, {}, {})
276 emit(0, 'ROOM', number, a[1], money(base))
277 when 'RATE'
278 r = room(a[0])
279 old = r.base
280 r.base = cents(a[1])
281 emit(0, 'RATE', r.number, money(old), '->', money(r.base))
282 when 'SEASON'
283 start, finish = period(a[0], a[1], false, false)
284 @seasons << [start, finish, a[2].to_i]
285 emit(0, 'SEASON', @seasons.length, a[0], '->', a[1], a[2] + '%')
286 when 'BLOCK', 'UNBLOCK'
287 r = room(a[0])
288 start, finish = period(a[1], a[2], true)
289 if cmd == 'BLOCK'
290 (start..finish).each do |day|
291 raise CommandError, "room unavailable #{r.number} #{date_text(day)}" if r.occupied.key?(day)
292 end
293 end
294 count = 0
295 (start..finish).each do |day|
296 if cmd == 'BLOCK'
297 count += 1 unless r.blocks.key?(day)
298 r.blocks[day] = true
299 elsif r.blocks.delete(day)
300 count += 1
301 end
302 end
303 emit(0, cmd, r.number, a[1], '->', a[2], cmd == 'BLOCK' ? 'added' : 'removed', count)
304 when 'CLEAN'
305 r = room(a[0])
306 raise CommandError, "already clean #{r.number}" if r.clean
307 r.clean = true
308 emit(0, 'CLEAN', r.number)
309 when 'TODAY'
310 day = date(a[0])
311 raise CommandError, "past date #{a[0]}" if day < @today
312 emit(0, 'TODAY', date_week(day))
313 checkouts = @reservations.values.select { |x| x.status == 'inhouse' && x.finish < day }
314 checkouts.sort_by { |x| [x.finish, x.id] }.each { |x| checkout(x, x.finish, 2) }
315 noshows = @reservations.values.select { |x| x.status == 'booked' && x.start < day }
316 noshows.sort_by { |x| [x.start, x.id] }.each do |x|
317 x.fee = x.nights.first.rate
318 x.paid = x.fee
319 x.status = 'noshow'
320 x.posting = x.start
321 release(x)
322 emit(2, 'NOSHOW', x.id, x.guest, 'fee', money(x.fee), deposit_text(x, x.fee))
323 end
324 @today = day
325 when 'BOOK', 'BOOKROOM', 'WAITLIST'
326 unused(a[0])
327 specific = nil
328 if cmd == 'BOOKROOM'
329 specific = room(a[2])
330 else
331 type(a[2])
332 end
333 start, finish = stay(a[3], a[4])
334 r = specific || find_room(a[2], start, finish)
335 if specific
336 day = unavailable(specific, start, finish)
337 raise CommandError, "room unavailable #{specific.number} #{date_text(day)}" unless day.nil?
338 end
339 if r
340 create(a[0], a[1], r, start, finish)
341 elsif cmd == 'WAITLIST'
342 @queue << Waiting.new(a[0], a[1], a[2], start, finish, start)
343 @used[a[0]] = true
344 emit(0, 'WAITLIST', a[0], 'queued', 'position', @queue.length)
345 else
346 raise CommandError, "no availability #{a[2]} #{a[3]} #{a[4]}"
347 end
348 when 'MODIFY'
349 x = reservation(a[0])
350 status(x, ['booked'])
351 start, finish = stay(a[1], a[2])
352 old = @rooms[x.room]
353 kept = unavailable(old, start, finish, x.id).nil?
354 r = kept ? old : find_room(old.kind, start, finish, x.id)
355 raise CommandError, "no availability #{old.kind} #{a[1]} #{a[2]}" unless r
356 release(x)
357 x.room = r.number
358 x.start = start
359 x.finish = finish
360 x.nights = price(r, start, finish)
361 occupy(x)
362 emit(0, 'MODIFY', x.id, 'room', r.number, kept ? 'kept' : 'reassigned',
363 a[1], '->', a[2], 'nights', finish - start, 'total', money(lodging(x)))
364 when 'CANCEL'
365 i = @queue.index { |w| w.id == a[0] }
366 if i
367 @queue.delete_at(i)
368 emit(0, 'CANCEL', a[0], 'waitlist', 'removed')
369 else
370 x = reservation(a[0])
371 status(x, ['booked'])
372 days = x.start - @today
373 x.fee = days >= 7 ? 0 : days >= 1 ? x.nights.first.rate / 2 : x.nights.first.rate
374 x.paid = x.fee
375 x.status = 'cancelled'
376 x.posting = @today
377 release(x)
378 emit(0, 'CANCEL', x.id, 'days', days, 'fee', money(x.fee), deposit_text(x, x.fee))
379 end
380 when 'CHECKIN'
381 x = reservation(a[0])
382 status(x, ['booked'])
383 raise CommandError, "not arrival date #{x.id} #{date_text(x.start)}" if x.start != @today
384 occupant = in_house(x.room)
385 raise CommandError, "room occupied #{x.room} #{occupant.id}" if occupant
386 raise CommandError, "room dirty #{x.room}" unless @rooms[x.room].clean
387 x.status = 'inhouse'
388 emit(0, 'CHECKIN', x.id, x.guest, 'room', x.room, 'until', date_text(x.finish))
389 when 'MOVE'
390 x = reservation(a[0])
391 status(x, ['inhouse'])
392 r = room(a[1])
393 raise CommandError, "same room #{r.number}" if r.number == x.room
394 remaining = x.nights.select { |n| n.day >= @today }
395 raise CommandError, "no remaining nights #{x.id}" if remaining.empty?
396 remaining.each do |n|
397 raise CommandError, "room unavailable #{r.number} #{date_text(n.day)}" unless free?(r, n.day, x.id)
398 end
399 occupant = in_house(r.number)
400 raise CommandError, "room occupied #{r.number} #{occupant.id}" if occupant
401 raise CommandError, "room dirty #{r.number}" unless r.clean
402 old = x.room
403 remaining.each do |n|
404 @rooms[n.room].occupied.delete(n.day)
405 n.room = r.number
406 r.occupied[n.day] = [x.id, n.rate]
407 end
408 emit(0, 'MOVE', x.id, old, '->', r.number, 'nights', remaining.length)
409 @rooms[old].clean = false
410 x.room = r.number
411 when 'EXTEND'
412 x = reservation(a[0])
413 status(x, ['inhouse'])
414 finish = date(a[1])
415 raise CommandError, "invalid range #{date_text(x.finish)} #{a[1]}" if finish <= x.finish
416 raise CommandError, "too long #{finish - x.start}" if finish - x.start > 30
417 r = @rooms[x.room]
418 blocked = unavailable(r, x.finish, finish, x.id)
419 raise CommandError, "room unavailable #{r.number} #{date_text(blocked)}" unless blocked.nil?
420 x.nights.concat(price(r, x.finish, finish))
421 x.finish = finish
422 occupy(x)
423 emit(0, 'EXTEND', x.id, 'until', a[1], 'nights', x.nights.length, 'total', money(lodging(x)))
424 when 'CHARGE', 'PAY'
425 x = reservation(a[0])
426 status(x, cmd == 'CHARGE' ? ['inhouse'] : %w[booked inhouse])
427 amount = cents(a[1])
428 raise CommandError, 'invalid amount' if amount.zero?
429 if cmd == 'CHARGE'
430 x.extras << [a[2], amount]
431 emit(0, 'CHARGE', x.id, a[2], money(amount), 'extras', money(extras(x)))
432 else
433 x.deposit += amount
434 emit(0, 'PAY', x.id, money(amount), 'deposit', money(x.deposit))
435 end
436 when 'CHECKOUT'
437 x = reservation(a[0])
438 status(x, ['inhouse'])
439 checkout(x, @today)
440 when 'INVOICE'
441 invoice(reservation(a[0]))
442 when 'AVAIL'
443 type(a[0])
444 start, finish = period(a[1], a[2])
445 emit(0, 'AVAIL', a[0], a[1], '->', a[2])
446 (start..finish).each do |day|
447 rooms = sorted_rooms.select { |r| r.kind == a[0] && free?(r, day) }
448 emit(2, date_week(day), 'free', rooms.length, *rooms.map(&:number))
449 end
450 when 'OCCUPANCY'
451 day = date(a[0])
452 emit(0, 'OCCUPANCY', date_week(day), occupancy_text(@rooms.values, day))
453 TYPES.each do |kind|
454 rooms = @rooms.values.select { |r| r.kind == kind }
455 emit(2, kind, occupancy_text(rooms, day)) unless rooms.empty?
456 end
457 when 'STATS'
458 start, finish = period(a[0], a[1])
459 emit(0, 'STATS', a[0], '->', a[1])
460 total = [0, 0, 0]
461 (start..finish).each do |day|
462 values = metrics(day)
463 emit(2, date_week(day), metrics_text(values))
464 3.times { |i| total[i] += values[i] }
465 end
466 emit(2, 'total', metrics_text(total))
467 when 'ROOMS'
468 emit(0, 'ROOMS', date_week(@today), 'rooms', @rooms.length)
469 sorted_rooms.each do |r|
470 occupant = in_house(r.number)
471 situation = occupant ? "occupied #{occupant.id}" : r.blocks.key?(@today) ? 'blocked' : 'vacant'
472 following = @reservations.values.select { |x| x.status == 'booked' && x.room == r.number }.min_by { |x| [x.start, x.id] }
473 parts = [r.number, r.kind, 'base', money(r.base), 'tonight', money(rate(r, @today)), r.clean ? 'clean' : 'dirty', situation]
474 parts.push('next', following.id, date_text(following.start)) if following
475 emit(2, *parts)
476 end
477 when 'LIST'
478 targets = @reservations.values.select { |x| x.status == a[0] }.sort_by { |x| [x.start, x.id] }
479 emit(0, 'LIST', a[0], targets.length)
480 targets.each { |x| emit(2, x.id, x.guest, 'room', x.room, date_text(x.start), '->', date_text(x.finish)) }
481 when 'QUEUE'
482 emit(0, 'QUEUE', @queue.length)
483 @queue.each_with_index do |w, i|
484 emit(2, i + 1, w.id, w.guest, w.kind, date_text(w.start), '->', date_text(w.finish))
485 end
486 when 'SCHEDULE'
487 day = date(a[0])
488 arrivals = @reservations.values.select { |x| x.start == day && x.status != 'cancelled' }.sort_by(&:id)
489 departures = @reservations.values.select { |x| x.finish == day && %w[booked inhouse checkedout].include?(x.status) }.sort_by(&:id)
490 emit(0, 'SCHEDULE', date_week(day), 'arrivals', arrivals.length, 'departures', departures.length)
491 arrivals.each { |x| emit(2, 'arrival', x.id, x.guest, 'room', x.room, x.status) }
492 departures.each { |x| emit(2, 'departure', x.id, x.guest, 'room', x.room, x.status) }
493 when 'GUEST'
494 targets = @reservations.values.select { |x| x.guest == a[0] }
495 raise CommandError, "unknown guest #{a[0]}" if targets.empty?
496 emit(0, 'GUEST', a[0])
497 targets.each do |x|
498 pending = %w[booked inhouse].include?(x.status)
499 emit(2, x.id, 'room', x.room, @rooms[x.room].kind, date_text(x.start), '->',
500 date_text(x.finish), x.status, pending ? 'deposit' : 'paid', money(pending ? x.deposit : x.paid))
501 end
502 points, name, = tier(a[0])
503 emit(2, 'total', 'paid', money(guest_paid(a[0])), 'points', points, 'tier', name)
504 when 'REVENUE'
505 start, finish = period(a[0], a[1], false, false)
506 targets = @reservations.values.select { |x| x.posting && start <= x.posting && x.posting <= finish }
507 totals = %i[lodging discount penalty extras tax cancellation noshow].to_h { |key| [key, 0] }
508 checkouts = 0
509 cancellations = 0
510 noshows = 0
511 targets.each do |x|
512 if x.status == 'checkedout'
513 checkouts += 1
514 %i[lodging discount penalty extras tax].each { |key| totals[key] += x.bill[key] }
515 elsif x.status == 'cancelled'
516 cancellations += 1
517 totals[:cancellation] += x.fee
518 elsif x.status == 'noshow'
519 noshows += 1
520 totals[:noshow] += x.fee
521 end
522 end
523 emit(0, 'REVENUE', a[0], '->', a[1])
524 totals.each { |key, value| emit(2, key, money(value)) }
525 total = totals[:lodging] - totals[:discount] + totals[:penalty] + totals[:extras] + totals[:tax] + totals[:cancellation] + totals[:noshow]
526 emit(2, 'total', money(total))
527 emit(2, 'checkouts', checkouts, 'cancellations', cancellations, 'noshows', noshows)
528 end
529 end

531 def execute(line)
532 cmd, *args = line.split
533 begin
534 command(cmd, args)
535 rescue CommandError => error
536 emit(0, 'ERROR', error.message)
537 end
538 process_queue
539 end
