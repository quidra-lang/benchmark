345final shapes = List.generate(7, (kind) {
346 final pattern = frames[kind];
347 final n = pattern.length;
348 var blocks = <Cell>[];
349 for (var row = 0; row < n; row++) {
350 for (var col = 0; col < n; col++)
351 if (pattern[row][col] == '#') blocks.add(Cell(col, row));
352 }
353 final rotations = <List<Cell>>[blocks];
354 for (var i = 0; i < 3; i++) {
355 blocks = blocks.map((b) => Cell(n - 1 - b.y, b.x)).toList();
356 rotations.add(blocks);
357 }
358 return rotations;
359});
360int seedStep(int x) => (48271 * x) % 2147483647;
361Position spawn(int kind) => Position(0, kind == 1 ? 5 : 4, 22);
362List<Cell> cells(int kind, Position p) =>
363 shapes[kind][p.s].map((c) => Cell(p.x + c.x, p.y - c.y)).toList();
364int lowest(int kind, Position p) =>
365 cells(kind, p).map((c) => c.y).reduce(math.min);
366(int, List<int>) bag(int random) {
367 final pieces = List<int>.generate(7, (i) => i);
368 for (var i = 6; i >= 1; i--) {
369 random = seedStep(random);
370 final j = random % (i + 1);
371 final temp = pieces[i];
372 pieces[i] = pieces[j];
373 pieces[j] = temp;
374 }
375 return (random, pieces);
376}

378bool fits(List<int> board, int kind, Position p) {
379 for (final c in cells(kind, p)) {
380 if (c.x < 1 || c.x > 10 || c.y < 1 || c.y > 22) return false;
381 if ((board[c.y - 1] & (1 << (c.x - 1))) != 0) return false;
382 }
383 return true;
384}

386Movement? move(List<int> board, int kind, Position p, int op) {
387 if (op == 6 || op == 7) {
388 final b = (p.s + (op == 6 ? 1 : 3)) % 4;
389 final row = kickStates.indexWhere((s) => s[0] == p.s && s[1] == b);
390 final offsets = kind == 1
391 ? [
392 [0, 0],
393 ]
394 : (kind == 0 ? iKicks : ordinaryKicks)[row];
395 for (var i = 0; i < offsets.length; i++) {
396 final q = Position(b, p.x + offsets[i][0], p.y + offsets[i][1]);
397 if (fits(board, kind, q)) return Movement(q, i + 1);
398 }
399 return null;
400 }
401 final dx = (op == 0 || op == 2)
402 ? -1
403 : (op == 1 || op == 3)
404 ? 1
405 : 0;
406 final dy = dx == 0 ? -1 : 0;
407 final repeatMove = op == 2 || op == 3 || op == 5;
408 var q = p;
409 var changed = false;
410 do {
411 final next = q.translate(dx, dy);
412 if (!fits(board, kind, next)) break;
413 q = next;
414 changed = true;
415 } while (repeatMove);
416 return changed ? Movement(q, 0) : null;
417}

419Position landing(List<int> board, int kind, Position p) =>
420 move(board, kind, p, 5)?.position ?? p;
421(List<int>, int) placed(List<int> board, int kind, Position p) {
422 final result = List<int>.of(board);
423 for (final c in cells(kind, p)) result[c.y - 1] |= 1 << (c.x - 1);
424 final remaining = result.where((r) => r != 1023).toList();
425 final n = 22 - remaining.length;
426 return (remaining..addAll(List.filled(n, 0)), n);
427}

429Metrics evaluate(List<int> board, int cleared) {
430 final heights = List<int>.filled(10, 0);
431 var holes = 0, bump = 0;
432 for (var c = 0; c < 10; c++) {
433 for (var r = 21; r >= 0; r--)
434 if ((board[r] & (1 << c)) != 0) {
435 heights[c] = r + 1;
436 break;
437 }
438 for (var r = 0; r < heights[c]; r++)
439 if ((board[r] & (1 << c)) == 0) holes++;
440 }
441 for (var c = 0; c < 9; c++) bump += (heights[c] - heights[c + 1]).abs();
442 final value =
443 76 * cleared -
444 51 * heights.fold<int>(0, (a, b) => a + b) -
445 36 * holes -
446 18 * bump;
447 return Metrics(heights, holes, bump, value);
448}

450Routes routes(List<int> board, int kind, Position start, [Position? target]) {
451 if (!fits(board, kind, start)) return Routes([], null);
452 final entries = <Route>[Route(start, -1, -1)];
453 final seen = <String>{start.key};
454 for (var i = 0; i < entries.length; i++) {
455 final entry = entries[i];
456 if (target != null &&
457 landing(board, kind, entry.position).key == target.key)
458 return Routes(entries, i);
459 for (var op = 0; op < 8; op++) {
460 final next = move(board, kind, entry.position, op);
461 if (next != null && seen.add(next.position.key))
462 entries.add(Route(next.position, i, op));
463 }
464 }
465 return Routes(target == null ? entries : [], null);
466}

468List<int> shortest(List<Route> entries, int index) {
469 final ops = <int>[];
470 while (entries[index].parent >= 0) {
471 ops.add(entries[index].operation);
472 index = entries[index].parent;
473 }
474 return ops.reversed.toList();
475}

477int tSpin(List<int> board, Position p, int test) {
478 final corners = [
479 Cell(p.x, p.y),
480 Cell(p.x + 2, p.y),
481 Cell(p.x + 2, p.y - 2),
482 Cell(p.x, p.y - 2),
483 ];
484 final blocked = corners
485 .map(
486 (c) =>
487 c.x < 1 ||
488 c.x > 10 ||
489 c.y < 1 ||
490 c.y > 22 ||
491 (board[c.y - 1] & (1 << (c.x - 1))) != 0,
492 )
493 .toList();
494 if (blocked.where((b) => b).length < 3) return 0;
495 return (blocked[p.s] && blocked[(p.s + 1) % 4]) || test == 5 ? 2 : 1;
496}

582 void operation(int op) {
583 if (op < 8) {
584 state.keys++;
585 final next = move(state.board, state.kind, state.position, op);
586 if (next == null) {
587 print('fail ${operations[op]}');
588 return;
589 }
590 if (op == 4 || op == 5) {
591 drops(state.position.y - next.position.y);
592 update(next.position, 1);
593 state.fall = 0;
594 state.ground = 0;
595 } else {
596 update(next.position, op >= 6 ? 2 : 1, next.test);
597 if (state.ground > 0 && state.resets < 15) {
598 state.ground = 0;
599 state.resets++;
600 }
601 }
602 } else if (op == 8) {
603 final p = landing(state.board, state.kind, state.position);
604 final rows = state.position.y - p.y;
605 drops(2 * rows);
606 if (rows > 0) update(p, 1);
607 lock();
608 } else if (op == 9) {
609 if (grounded())
610 lock();
611 else
612 print('fail LOCK');
613 } else {
614 if (state.usedHold) {
615 print('fail HOLD');
616 return;
617 }
618 state.usedHold = true;
619 state.stats.holds++;
620 final previous = state.hold;
621 state.hold = state.kind;
622 spawnPiece(previous ?? takeNext());
623 }
624 }

626 void tick() {
627 if (!grounded()) {
628 state.ground = 0;
629 state.fall++;
630 final interval = state.level >= 13
631 ? 1
632 : [60, 48, 37, 28, 21, 16, 11, 8, 6, 4, 3, 2][state.level - 1];
633 if (state.fall >= interval) {
634 update(state.position.translate(0, -1), 1);
635 state.fall = 0;
636 }
637 } else {
638 state.ground++;
639 if (state.ground >= 30) lock();
640 }
641 }

643 void lock() {
644 final kind = state.kind, p = state.position;
645 final oldCells = cells(kind, p);
646 final route = routes(state.board, kind, spawn(kind), p);
647 final shortestOps = route.index == null
648 ? null
649 : shortest(route.entries, route.index!);
650 final spin = kind == 2 && state.last == 2
651 ? tSpin(state.board, p, state.test)
652 : 0;
653 final placement = placed(state.board, kind, p);
654 final n = placement.$2;
655 state.board = placement.$1;
656 history.add(start.copy());
657 state.pieces++;
658 state.stats.kinds[kind]++;
659 state.stats.operationCount += state.keys;
660 final info = clearInfo(spin, n);
661 final level = state.level;
662 var points = info.points * level, sent = info.sent, back = false;
663 if (n > 0) {
664 state.streak++;
665 if (n == 4 || spin != 0) {
666 state.chain++;
667 back = state.chain >= 2;
668 } else
669 state.chain = 0;
670 } else
671 state.streak = 0;
672 if (back) {
673 points = points * 3 ~/ 2;
674 sent++;
675 }
676 final combo = state.streak - 1;
677 final comboPoints = combo >= 1 ? 50 * combo * level : 0;
678 if (combo >= 0)
679 sent += [0, 1, 1, 2, 2, 3, 3, 4, 4, 4, 5][math.min(combo, 10)];
680 final perfect = n > 0 && state.board.every((r) => r == 0);
681 final perfectPoints = perfect
682 ? (n == 4 && back ? 3200 : [800, 1200, 1800, 2000][n - 1]) * level
683 : 0;
684 if (perfect) {
685 sent += 10;
686 state.stats.perfect++;
687 }
688 final gained = points + comboPoints + perfectPoints;
689 state.score += gained;
690 state.lines += n;
691 state.stats.sent += sent;
692 state.stats.combo = math.max(state.stats.combo, combo);
693 state.stats.chain = math.max(state.stats.chain, state.chain);
694 print(
695 'lock ${state.pieces} ${kinds[kind]} ${states[p.s]} ${p.x} ${p.y} keys ${state.keys} ${shortestOps?.length ?? '-'} drop ${state.drop} lines $n sent $sent +$gained total ${state.score}',
696 );
697 if (shortestOps != null && state.keys > shortestOps.length) {
698 state.stats.faults++;
699 print(
700 'finesse ${shortestOps.isEmpty ? 'none' : operationText(shortestOps)}',
701 );
702 }
703 if (info.name >= 0) {
704 state.stats.clears[info.name]++;
705 var line = 'clear ${clearNames[info.name]} $points';
706 if (back) line += ' b2b';
707 if (combo >= 1) line += ' combo $combo $comboPoints';
708 if (perfect) line += ' perfect $perfectPoints';
709 print(line);
710 }
711 final nextLevel = state.initialLevel + state.lines ~/ 10;
712 if (nextLevel > level) {
713 state.level = nextLevel;
714 print('level ${state.level}');
715 }
716 if (oldCells.every((c) => c.y >= 21)) gameOver('lock out');
717 final cancelled = math.min(
718 sent,
719 state.pending.fold<int>(0, (a, b) => a + b),
720 );
721 var remaining = cancelled;
722 while (remaining > 0) {
723 final used = math.min(remaining, state.pending[0]);
724 remaining -= used;
725 state.pending[0] -= used;
726 if (state.pending[0] == 0) state.pending.removeAt(0);
727 }
728 if (cancelled > 0) {
729 state.stats.cancelled += cancelled;
730 print(
731 'cancel $cancelled pending ${state.pending.fold<int>(0, (a, b) => a + b)}',
732 );
733 }
734 if (n == 0) {
735 var risen = 0;
736 while (state.pending.isNotEmpty && risen < 8) {
737 final count = math.min(state.pending[0], 8 - risen);
738 state.random = seedStep(state.random);
739 final hole = state.random % 10 + 1;
740 insertGarbage(count, hole);
741 state.stats.received += count;
742 risen += count;
743 state.pending[0] -= count;
744 if (state.pending[0] == 0) state.pending.removeAt(0);
745 print(
746 'rise $count hole $hole pending ${state.pending.fold<int>(0, (a, b) => a + b)}',
747 );
748 }
749 }
750 state.usedHold = false;
751 spawnPiece(takeNext());
752 start = state.copy();
753 }

755 List<Candidate> candidates(int kind, Position start, List<int> prefix) {
756 final result = <Candidate>[];
757 final seen = <String>{};
758 for (final turns in <List<int>>[
759 [],
760 [6],
761 [6, 6],
762 [7],
763 ]) {
764 var p = start;
765 var valid = true;
766 for (final op in turns) {
767 final next = move(state.board, kind, p, op);
768 if (next == null) {
769 valid = false;
770 break;
771 }
772 p = next.position;
773 }
774 if (!valid) continue;
775 void add(Position q, List<int> shifts) {
776 final landed = landing(state.board, kind, q);
777 final occupied = cells(
778 kind,
779 landed,
780 ).map((c) => (c.y - 1) * 10 + c.x - 1).toList()..sort();
781 final id = occupied.join(',');
782 if (!seen.add(id)) return;
783 final r = placed(state.board, kind, landed);
784 final value = evaluate(r.$1, r.$2).value;
785 result.add(
786 Candidate(kind, value, [...prefix, ...turns, ...shifts, 8], id),
787 );
788 }
789
790 add(p, []);
791 for (final direction in [0, 1]) {
792 var q = p;
793 final shifts = <int>[];
794 while (true) {
795 final next = move(state.board, kind, q, direction);
796 if (next == null) break;
797 q = next.position;
798 shifts.add(direction);
799 add(q, shifts);
800 }
801 }
802 }
803 return result;
804 }

806 List<Candidate> ranked() {
807 final result = candidates(state.kind, state.position, []);
808 final held = state.hold ?? state.queue[0];
809 if (!state.usedHold && fits(state.board, held, spawn(held)))
810 result.addAll(candidates(held, spawn(held), [10]));
811 for (var i = 0; i < result.length; i++) result[i].order = i;
812 result.sort((a, b) {
813 final diff = b.value.compareTo(a.value);
814 return diff != 0 ? diff : a.order.compareTo(b.order);
815 });
816 return result;
817 }

839 ..sort((a, b) {

843 return dx != 0 ? dx : a.position.s.compareTo(b.position.s);

849 'spin ${states[c.position.s]} ${c.position.x} ${c.position.y} ${clearNames[info.name]}',
850 );
851 }
852 print('spins ${list.length}');

885 void command(String cmd, List<String> args) {
886 switch (cmd) {
887 case 'PRINT':
888 final current = state.ended
889 ? <String>{}
890 : cells(
891 state.kind,
892 state.position,
893 ).map((c) => '${c.x},${c.y}').toSet();
894 final ghost = state.ended
895 ? <String>{}
896 : cells(
897 state.kind,
898 landing(state.board, state.kind, state.position),
899 ).map((c) => '${c.x},${c.y}').toSet();
900 for (var r = 22; r >= 1; r--) {
901 var line = '${r.toString().padLeft(2)} ';
902 for (var c = 1; c <= 10; c++) {
903 line += (state.board[r - 1] & (1 << (c - 1))) != 0
904 ? '#'
905 : current.contains('$c,$r')
906 ? '@'
907 : ghost.contains('$c,$r')
908 ? '+'
909 : '.';
910 }
911 print(line);
912 }
913 break;
914 case 'NEXT':
915 print(
916 'hold ${state.hold == null ? '-' : kinds[state.hold!]} next ${state.queue.take(5).map((i) => kinds[i]).join()}',
917 );
918 break;
919 case 'PREVIEW':
920 final n = int.parse(args[0]);
921 final queue = List<int>.of(state.queue);
922 var random = state.random;
923 while (queue.length < n) {
924 final b = bag(random);
925 random = b.$1;
926 queue.addAll(b.$2);
927 }
928 print('preview ${queue.take(n).map((i) => kinds[i]).join()}');
929 break;
930 case 'STATUS':
931 print(
932 'score ${state.score} level ${state.level} lines ${state.lines} pieces ${state.pieces}',
933 );
934 final piece = state.ended
935 ? '-'
936 : '${kinds[state.kind]} ${states[state.position.s]} ${state.position.x} ${state.position.y}';
937 print(
938 'piece $piece streak ${state.streak} chain ${state.chain} pending ${state.pending.fold<int>(0, (a, b) => a + b)}',
939 );
940 break;
941 case 'ANALYZE':
942 final e = evaluate(state.board, 0);
943 print('heights ${e.heights.join(' ')}');
944 print('holes ${e.holes} bumpiness ${e.bumpiness} value ${e.value}');
945 break;
946 case 'ROW':
947 final row = int.parse(args[0]);
948 final pattern = args[1];
949 if (cells(
950 state.kind,
951 state.position,
952 ).any((c) => c.y == row && pattern[c.x - 1] == '#')) {
953 print('ERROR overlap');
954 break;
955 }
956 var mask = 0;
957 for (var i = 0; i < 10; i++) if (pattern[i] == '#') mask |= 1 << i;
958 state.board[row - 1] = mask;
959 print('OK');
960 break;
961 case 'GARBAGE':
962 final n = int.parse(args[0]), hole = int.parse(args[1]);
963 insertGarbage(n, hole);
964 var success = false;
965 for (var j = 0; j <= n; j++) {
966 final p = state.position.translate(0, j);
967 if (fits(state.board, state.kind, p)) {
968 state.position = p;
969 success = true;
970 print('OK');
971 break;
972 }
973 }
974 if (!success) gameOver('top out');
975 break;
976 case 'ATTACK':
977 state.pending.add(int.parse(args[0]));
978 print('pending ${state.pending.fold<int>(0, (a, b) => a + b)}');
979 break;
980 case 'TICK':
981 for (var i = 0; i < int.parse(args[0]); i++) tick();
982 break;
983 case 'UNDO':
984 if (history.isEmpty)
985 print('ERROR nothing to undo');
986 else {
987 state = history.removeLast().copy();
988 start = state.copy();
989 print('undo ${state.pieces} ${kinds[state.kind]}');
990 }
991 break;
992 case 'SAVE':
993 saves[args[0]] = SavedGame(
994 state.copy(),
995 history.map((s) => s.copy()).toList(),
996 start.copy(),
997 );
998 print('OK');
999 break;
1000 case 'LOAD':
1001 final saved = saves[args[0]];
1002 if (saved == null)
1003 print('ERROR unknown save');
1004 else {
1005 state = saved.state.copy();
1006 history = saved.history.map((s) => s.copy()).toList();
1007 start = saved.start.copy();
1008 print('load ${state.pieces} ${kinds[state.kind]}');
1009 }
1010 break;
1011 case 'SPINS':
1012 spins();
1013 break;
1014 case 'HINT':
1015 final list = ranked().take(int.parse(args[0])).toList();
1016 for (var i = 0; i < list.length; i++)
1017 print(
1018 'hint ${i + 1} ${kinds[list[i].kind]} ${operationText(list[i].operations)} value ${list[i].value}',
1019 );
1020 break;
1021 case 'AUTO':
1022 for (var i = 0; i < int.parse(args[0]); i++) {
1023 final c = ranked()[0];
1024 print('auto ${kinds[c.kind]} ${operationText(c.operations)}');
1025 for (final op in c.operations) operation(op);
1026 }
1027 break;
1028 }
1029 }

1032bool validNumber(String text, int minValue, int maxValue) {
1033 if (text.isEmpty ||
1034 (text.length > 1 && text[0] == '0') ||
1035 !RegExp(r'^[0-9]+$').hasMatch(text))
1036 return false;
1037 final n = int.tryParse(text);
1038 return n != null && n >= minValue && n <= maxValue;
1039}

1041bool validArgs(String command, List<String> args) {
1042 if (command == 'NEW')
1043 return args.length == 2 &&
1044 validNumber(args[0], 1, 2147483646) &&
1045 validNumber(args[1], 1, 15);
1046 if (command == 'ROW')
1047 return args.length == 2 &&
1048 validNumber(args[0], 1, 22) &&
1049 args[1].length == 10 &&
1050 RegExp(r'^[#.]{10}$').hasMatch(args[1]) &&
1051 args[1].contains('.');
1052 if (command == 'GARBAGE')
1053 return args.length == 2 &&
1054 validNumber(args[0], 1, 10) &&
1055 validNumber(args[1], 1, 10);
1056 if (command == 'SAVE' || command == 'LOAD')
1057 return args.length == 1 && RegExp(r'^[a-z]{1,8}$').hasMatch(args[0]);
1058 final limits = {
1059 'PREVIEW': 14,
1060 'ATTACK': 20,
1061 'TICK': 1000,
1062 'HINT': 10,
1063 'AUTO': 50,
1064 };
1065 if (limits.containsKey(command))
1066 return args.length == 1 && validNumber(args[0], 1, limits[command]!);
1067 return args.isEmpty;
1068}
