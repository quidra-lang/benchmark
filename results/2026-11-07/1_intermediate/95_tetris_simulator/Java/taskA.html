249 static int[][] blocks(char kind, Position position) {
250 String[] rows;
251 switch (kind) {
252 case 'I': rows = new String[]{"....", "####", "....", "...."}; break;
253 case 'O': rows = new String[]{"##", "##"}; break;
254 case 'T': rows = new String[]{".#.", "###", "..."}; break;
255 case 'S': rows = new String[]{".##", "##.", "..."}; break;
256 case 'Z': rows = new String[]{"##.", ".##", "..."}; break;
257 case 'J': rows = new String[]{"#..", "###", "..."}; break;
258 default: rows = new String[]{"..#", "###", "..."}; break;
259 }
260 int size = rows.length;
261 int[][] out = new int[4][2];
262 int count = 0;
263 for (int j = 0; j < size; ++j)
264 for (int i = 0; i < size; ++i)
265 if (rows[j].charAt(i) == '#') {
266 int x = i, y = j;
267 for (int r = 0; r < position.rotation; ++r) {
268 int old = x;
269 x = size - 1 - y;
270 y = old;
271 }
272 out[count++] = new int[]{position.x + x, position.y - y};
273 }
274 return out;
275 }

283 static boolean fits(int[] board, char kind, Position position) {
284 for (int[] cell : blocks(kind, position)) {
285 int c = cell[0], r = cell[1];
286 if (c < 1 || c > 10 || r < 1 || r > 22 || (board[r - 1] & (1 << (c - 1))) != 0) return false;
287 }
288 return true;
289 }

291 static Movement move(int[] board, char kind, Position position, String action) {
292 if (action.equals("CW") || action.equals("CCW")) {
293 int rotation = (position.rotation + (action.equals("CW") ? 1 : 3)) % 4;
294 int[][] offsets = kind == 'O' ? new int[][]{{0,0}} : (kind == 'I' ? I_KICKS : KICKS).get("" + position.rotation + rotation);
295 for (int i = 0; i < offsets.length; ++i) {
296 Position next = new Position(rotation, position.x + offsets[i][0], position.y + offsets[i][1]);
297 if (fits(board, kind, next)) return new Movement(next, i + 1);
298 }
299 return null;
300 }
301 int dx = action.equals("L") || action.equals("LL") ? -1 : action.equals("R") || action.equals("RR") ? 1 : 0;
302 int dy = dx == 0 ? -1 : 0;
303 boolean repeat = action.equals("LL") || action.equals("RR") || action.equals("DD");
304 boolean changed = false;
305 Position current = position;
306 do {
307 Position next = new Position(current.rotation, current.x + dx, current.y + dy);
308 if (!fits(board, kind, next)) break;
309 current = next;
310 changed = true;
311 } while (repeat);
312 return changed ? new Movement(current, 0) : null;
313 }

320 static Placement placed(int[] board, char kind, Position position) {
321 int[] filled = board.clone();
322 for (int[] cell : blocks(kind, position)) filled[cell[1] - 1] |= 1 << (cell[0] - 1);
323 int[] filtered = new int[22];
324 int length = 0;
325 for (int row : filled) if (row != 1023) filtered[length++] = row;
326 return new Placement(filtered, 22 - length);
327 }

329 static Evaluation evaluate(int[] board, int cleared) {
330 Evaluation e = new Evaluation();
331 int sum = 0;
332 for (int c = 0; c < 10; ++c) {
333 for (int r = 21; r >= 0; --r) if ((board[r] & (1 << c)) != 0) {
334 e.heights[c] = r + 1;
335 break;
336 }
337 sum += e.heights[c];
338 for (int r = 0; r < e.heights[c]; ++r)
339 if ((board[r] & (1 << c)) == 0) e.holes++;
340 if (c > 0) e.bumpiness += Math.abs(e.heights[c - 1] - e.heights[c]);
341 }
342 e.value = 76 * cleared - 51 * sum - 36 * e.holes - 18 * e.bumpiness;
343 return e;
344 }

346 static ArrayList<Route> routes(int[] board, char kind, Position start, Position target) {
347 ArrayList<Route> queue = new ArrayList<>();
348 if (!fits(board, kind, start)) return queue;
349 Set<Position> visited = new HashSet<>();
350 visited.add(start);
351 queue.add(new Route(start, new ArrayList<>()));
352 for (int index = 0; index < queue.size(); ++index) {
353 Route entry = queue.get(index);
354 if (target != null && landing(board, kind, entry.position).equals(target)) {
355 ArrayList<Route> found = new ArrayList<>();
356 found.add(entry);
357 return found;
358 }
359 for (String action : MOVES) {
360 Movement result = move(board, kind, entry.position, action);
361 if (result != null && visited.add(result.position)) {
362 ArrayList<String> keys = new ArrayList<>(entry.keys);
363 keys.add(action);
364 queue.add(new Route(result.position, keys));
365 }
366 }
367 }
368 return target == null ? queue : new ArrayList<>();
369 }

371 static String spinType(int[] board, Position pos, int test) {
372 int[][] corners = {{pos.x,pos.y},{pos.x+2,pos.y},{pos.x+2,pos.y-2},{pos.x,pos.y-2}};
373 boolean[] occupied = new boolean[4];
374 int count = 0;
375 for (int i = 0; i < 4; ++i) {
376 int col = corners[i][0], row = corners[i][1];
377 occupied[i] = col < 1 || col > 10 || row < 1 || row > 22 || (board[row - 1] & (1 << (col - 1))) != 0;
378 if (occupied[i]) count++;
379 }
380 if (count < 3) return "";
381 return (occupied[pos.rotation] && occupied[(pos.rotation+1)%4]) || test == 5 ? "full" : "mini";
382 }

489 void operation(String action) {
490 State s = state;
491 if (Arrays.asList(MOVES).contains(action)) {
492 s.keys++;
493 Movement result = move(s.board,s.kind,s.position,action);
494 if (result == null) {
495 emit("fail " + action);
496 return;
497 }
498 if (action.equals("SD") || action.equals("DD")) {
499 addDrop(s.position.y-result.position.y);
500 updatePosition(result.position,1,0);
501 s.fall = s.ground = 0;
502 } else {
503 boolean rotation = action.equals("CW") || action.equals("CCW");
504 updatePosition(result.position,rotation?2:1,result.test);
505 if (s.ground > 0 && s.resets < 15) {
506 s.ground = 0;
507 s.resets++;
508 }
509 }
510 } else if (action.equals("HD")) {
511 Position target = landing(s.board,s.kind,s.position);
512 int rows = s.position.y - target.y;
513 addDrop(2*rows);
514 if (rows > 0) updatePosition(target,1,0);
515 lock();
516 } else if (action.equals("LOCK")) {
517 if (grounded()) lock();
518 else emit("fail LOCK");
519 } else if (action.equals("HOLD")) {
520 if (s.usedHold) {
521 emit("fail HOLD");
522 return;
523 }
524 s.usedHold = true;
525 s.stats.holds++;
526 char current = s.kind;
527 char next = s.hold == '-' ? takeNext() : s.hold;
528 s.hold = current;
529 spawnPiece(next);
530 }
531 }

533 void tick() {
534 State s = state;
535 if (!grounded()) {
536 s.ground = 0;
537 s.fall++;
538 int interval = s.level >= 13 ? 1 : FALL_INTERVALS[s.level-1];
539 if (s.fall >= interval) {
540 updatePosition(new Position(s.position.rotation,s.position.x,s.position.y-1),1,0);
541 s.fall = 0;
542 }
543 } else {
544 s.ground++;
545 if (s.ground >= 30) lock();
546 }
547 }

549 void lock() {
550 State s = state;
551 char kind = s.kind;
552 Position pos = s.position;
553 int[][] oldCells = blocks(kind,pos);
554 ArrayList<Route> shortest = routes(s.board,kind,spawn(kind),pos);
555 String spin = kind == 'T' && s.last == 2 ? spinType(s.board,pos,s.test) : "";
556 Placement placement = placed(s.board,kind,pos);
557 s.board = placement.board;
558 int lines = placement.lines;
559 history.add(start.copy());
560 s.pieces++;
561 s.stats.kinds[KINDS.indexOf(kind)]++;
562 s.stats.operations += s.keys;
563 ClearInfo info = clearInfo(spin,lines);
564 int level = s.level;
565 int points = info.points*level, sent = info.sent;
566 boolean backToBack = false;
567 if (lines > 0) {
568 s.streak++;
569 if (lines == 4 || !spin.isEmpty()) {
570 s.chain++;
571 backToBack = s.chain >= 2;
572 } else s.chain = 0;
573 } else s.streak = 0;
574 if (backToBack) {
575 points = points*3/2;
576 sent++;
577 }
578 int combo = s.streak-1;
579 int comboPoints = combo >= 1 ? 50*combo*level : 0;
580 if (combo >= 0) sent += COMBO_SENT[Math.min(combo,10)];
581 boolean perfect = lines > 0;
582 for (int row : s.board) if (row != 0) perfect = false;
583 int perfectPoints = 0;
584 if (perfect) {
585 int[] pointsTable = {800,1200,1800,2000};
586 perfectPoints = (lines == 4 && backToBack ? 3200 : pointsTable[lines-1])*level;
587 sent += 10;
588 s.stats.perfect++;
589 }
590 int gained = points+comboPoints+perfectPoints;
591 s.score += gained;
592 s.lines += lines;
593 s.stats.sent += sent;
594 s.stats.combo = Math.max(s.stats.combo,combo);
595 s.stats.chain = Math.max(s.stats.chain,s.chain);
596 String length = shortest.isEmpty() ? "-" : Integer.toString(shortest.get(0).keys.size());
597 emit("lock " + s.pieces + " " + kind + " " + STATES[pos.rotation] + " " + pos.x + " " + pos.y +
598 " keys " + s.keys + " " + length + " drop " + s.drop + " lines " + lines + " sent " + sent +
599 " +" + gained + " total " + s.score);
600 if (!shortest.isEmpty() && s.keys > shortest.get(0).keys.size()) {
601 s.stats.faults++;
602 ArrayList<String> operations = shortest.get(0).keys;
603 emit("finesse " + (operations.isEmpty() ? "none" : String.join(" ",operations)));
604 }
605 if (info.index >= 0) {
606 s.stats.clears[info.index]++;
607 StringBuilder report = new StringBuilder("clear " + info.name + " " + points);
608 if (backToBack) report.append(" b2b");
609 if (combo >= 1) report.append(" combo ").append(combo).append(' ').append(comboPoints);
610 if (perfect) report.append(" perfect ").append(perfectPoints);
611 emit(report.toString());
612 }
613 int newLevel = s.initialLevel+s.lines/10;
614 if (newLevel > level) {
615 s.level = newLevel;
616 emit("level " + s.level);
617 }
618 boolean lockOut = true;
619 for (int[] cell : oldCells) if (cell[1] < 21) lockOut = false;
620 if (lockOut) gameOver("lock out");
621 int cancelled = Math.min(sent,pendingSum(s.pending));
622 int remaining = cancelled;
623 while (remaining > 0) {
624 int count = Math.min(remaining,s.pending.get(0));
625 s.pending.set(0,s.pending.get(0)-count);
626 remaining -= count;
627 if (s.pending.get(0) == 0) s.pending.remove(0);
628 }
629 if (cancelled > 0) {
630 s.stats.cancelled += cancelled;
631 emit("cancel " + cancelled + " pending " + pendingSum(s.pending));
632 }
633 if (lines == 0) {
634 int risen = 0;
635 while (!s.pending.isEmpty() && risen < 8) {
636 int count = Math.min(s.pending.get(0),8-risen);
637 s.random = nextRandom(s.random);
638 int hole = (int)(s.random%10)+1;
639 garbage(count,hole);
640 s.stats.received += count;
641 risen += count;
642 s.pending.set(0,s.pending.get(0)-count);
643 if (s.pending.get(0) == 0) s.pending.remove(0);
644 emit("rise " + count + " hole " + hole + " pending " + pendingSum(s.pending));
645 }
646 }
647 s.usedHold = false;
648 spawnPiece(takeNext());
649 start = s.copy();
650 }

652 ArrayList<Candidate> candidates(char kind, Position first, List<String> prefix) {
653 ArrayList<Candidate> result = new ArrayList<>();
654 Set<String> visited = new HashSet<>();
655 List<List<String>> rotations = List.of(List.of(),List.of("CW"),List.of("CW","CW"),List.of("CCW"));
656 for (List<String> turns : rotations) {
657 Position current = first;
658 boolean possible = true;
659 for (String turn : turns) {
660 Movement next = move(state.board,kind,current,turn);
661 if (next == null) {
662 possible = false;
663 break;
664 }
665 current = next.position;
666 }
667 if (!possible) continue;
668 addCandidate(result,visited,kind,current,prefix,turns,List.of());
669 for (String direction : List.of("L","R")) {
670 Position shifted = current;
671 ArrayList<String> shifts = new ArrayList<>();
672 while (true) {
673 Movement next = move(state.board,kind,shifted,direction);
674 if (next == null) break;
675 shifted = next.position;
676 shifts.add(direction);
677 addCandidate(result,visited,kind,shifted,prefix,turns,shifts);
678 }
679 }
680 }
681 return result;
682 }

684 void addCandidate(List<Candidate> result, Set<String> visited, char kind, Position position,
685 List<String> prefix, List<String> turns, List<String> shifts) {
686 Position down = landing(state.board,kind,position);
687 int[] squares = new int[4];
688 int index = 0;
689 for (int[] cell : blocks(kind,down)) squares[index++] = (cell[1]-1)*10+cell[0]-1;
690 Arrays.sort(squares);
691 String id = Arrays.toString(squares);
692 if (!visited.add(id)) return;
693 Placement placed = placed(state.board,kind,down);
694 ArrayList<String> keys = new ArrayList<>(prefix);
695 keys.addAll(turns);
696 keys.addAll(shifts);
697 keys.add("HD");
698 result.add(new Candidate(evaluate(placed.board,placed.lines).value,kind,keys));
699 }

749 void command(String name, List<String> args) {
750 State s = state;
751 switch (name) {
752 case "PRINT": {
753 Set<Integer> current = new HashSet<>(), ghost = new HashSet<>();
754 if (!s.ended) {
755 for (int[] cell : blocks(s.kind,s.position)) current.add(cell[1]*10+cell[0]);
756 for (int[] cell : blocks(s.kind,landing(s.board,s.kind,s.position))) ghost.add(cell[1]*10+cell[0]);
757 }
758 for (int r = 22; r >= 1; --r) {
759 StringBuilder line = new StringBuilder((r < 10 ? " " : "") + r + " ");
760 for (int c = 1; c <= 10; ++c) {
761 int id = r*10+c;
762 char ch = (s.board[r-1]&(1<<(c-1))) != 0 ? '#' : current.contains(id) ? '@' : ghost.contains(id) ? '+' : '.';
763 line.append(ch);
764 }
765 emit(line.toString());
766 }
767 break;
768 }
769 case "NEXT": {
770 StringBuilder next = new StringBuilder();
771 for (int i = 0; i < 5; ++i) next.append(s.queue.get(i));
772 emit("hold " + s.hold + " next " + next);
773 break;
774 }
775 case "PREVIEW": {
776 int count = Integer.parseInt(args.get(0));
777 ArrayList<Character> queue = new ArrayList<>(s.queue);
778 long random = s.random;
779 while (queue.size() < count) {
780 char[] bag = KINDS.toCharArray();
781 for (int i = 6; i >= 1; --i) {
782 random = nextRandom(random);
783 int j = (int)(random%(i+1));
784 char temp = bag[i];
785 bag[i] = bag[j];
786 bag[j] = temp;
787 }
788 for (char kind : bag) queue.add(kind);
789 }
790 StringBuilder result = new StringBuilder();
791 for (int i = 0; i < count; ++i) result.append(queue.get(i));
792 emit("preview " + result);
793 break;
794 }
795 case "STATUS": {
796 emit("score " + s.score + " level " + s.level + " lines " + s.lines + " pieces " + s.pieces);
797 String piece = s.ended ? "-" : s.kind + " " + STATES[s.position.rotation] + " " + s.position.x + " " + s.position.y;
798 emit("piece " + piece + " streak " + s.streak + " chain " + s.chain + " pending " + pendingSum(s.pending));
799 break;
800 }
801 case "ANALYZE": {
802 Evaluation e = evaluate(s.board,0);
803 StringBuilder heights = new StringBuilder("heights");
804 for (int value : e.heights) heights.append(' ').append(value);
805 emit(heights.toString());
806 emit("holes " + e.holes + " bumpiness " + e.bumpiness + " value " + e.value);
807 break;
808 }
809 case "ROW": {
810 int row = Integer.parseInt(args.get(0));
811 String pattern = args.get(1);
812 for (int[] cell : blocks(s.kind,s.position)) {
813 if (cell[1] == row && pattern.charAt(cell[0]-1) == '#') {
814 emit("ERROR overlap");
815 return;
816 }
817 }
818 int mask = 0;
819 for (int c = 0; c < 10; ++c) if (pattern.charAt(c) == '#') mask |= 1 << c;
820 s.board[row-1] = mask;
821 emit("OK");
822 break;
823 }
824 case "GARBAGE": {
825 int rows = Integer.parseInt(args.get(0)), hole = Integer.parseInt(args.get(1));
826 garbage(rows,hole);
827 for (int i = 0; i <= rows; ++i) {
828 Position next = new Position(s.position.rotation,s.position.x,s.position.y+i);
829 if (fits(s.board,s.kind,next)) {
830 s.position = next;
831 emit("OK");
832 return;
833 }
834 }
835 gameOver("top out");
836 break;
837 }
838 case "ATTACK": {
839 s.pending.add(Integer.parseInt(args.get(0)));
840 emit("pending " + pendingSum(s.pending));
841 break;
842 }
843 case "TICK": {
844 int count = Integer.parseInt(args.get(0));
845 for (int i = 0; i < count; ++i) tick();
846 break;
847 }
848 case "UNDO": {
849 if (history.isEmpty()) emit("ERROR nothing to undo");
850 else {
851 state = history.remove(history.size()-1).copy();
852 start = state.copy();
853 emit("undo " + state.pieces + " " + state.kind);
854 }
855 break;
856 }
857 case "SAVE": {
858 saves.put(args.get(0),new Snapshot(s,history,start));
859 emit("OK");
860 break;
861 }
862 case "LOAD": {
863 Snapshot snap = saves.get(args.get(0));
864 if (snap == null) emit("ERROR unknown save");
865 else {
866 state = snap.state.copy();
867 history = new ArrayList<>();
868 for (State record : snap.history) history.add(record.copy());
869 start = snap.start.copy();
870 emit("load " + state.pieces + " " + state.kind);
871 }
872 break;
873 }
874 case "SPINS": spins(); break;
875 case "HINT": {
876 ArrayList<Candidate> ranked = ranked();
877 int count = Math.min(Integer.parseInt(args.get(0)),ranked.size());
878 for (int i = 0; i < count; ++i) {
879 Candidate candidate = ranked.get(i);
880 emit("hint " + (i+1) + " " + candidate.kind + " " + String.join(" ",candidate.keys) + " value " + candidate.value);
881 }
882 break;
883 }
884 case "AUTO": {
885 int count = Integer.parseInt(args.get(0));
886 for (int i = 0; i < count; ++i) {
887 ArrayList<Candidate> ranked = ranked();
888 if (ranked.isEmpty()) break;
889 Candidate best = ranked.get(0);
890 emit("auto " + best.kind + " " + String.join(" ",best.keys));
891 for (String op : best.keys) operation(op);
892 }
893 break;
894 }
895 default: break;
896 }
897 }

900 static boolean number(String word, long low, long high) {
901 if (word.isEmpty() || (word.length() > 1 && word.charAt(0) == '0')) return false;
902 long value = 0;
903 for (int i = 0; i < word.length(); ++i) {
904 char c = word.charAt(i);
905 if (c < '0' || c > '9') return false;
906 value = value*10+(c-'0');
907 if (value > high) return false;
908 }
909 return value >= low;
910 }

912 static boolean validArguments(String command, List<String> args) {
913 if (command.equals("NEW")) return args.size() == 2 && number(args.get(0),1,2147483646) && number(args.get(1),1,15);
914 if (command.equals("ROW")) return args.size() == 2 && number(args.get(0),1,22) && args.get(1).matches("[.#]{10}") && args.get(1).contains(".");
915 if (command.equals("GARBAGE")) return args.size() == 2 && number(args.get(0),1,10) && number(args.get(1),1,10);
916 if (command.equals("SAVE") || command.equals("LOAD")) return args.size() == 1 && args.get(0).matches("[a-z]{1,8}");
917 Map<String,Integer> limits = Map.of("PREVIEW",14,"ATTACK",20,"TICK",1000,"HINT",10,"AUTO",50);
918 Integer high = limits.get(command);
919 if (high != null) return args.size() == 1 && number(args.get(0),1,high);
920 return args.isEmpty();
921 }

923 public static void main(String[] args) throws Exception {
924 BufferedReader input = new BufferedReader(new InputStreamReader(System.in));
925 String line = input.readLine();
926 if (line == null) return;
927 int count = Integer.parseInt(line.trim());
928 Game game = null;
929 int serial = 0;
930 for (int i = 0; i < count; ++i) {
931 line = input.readLine();
932 if (line == null) break;
933 String[] words = line.trim().split("\\s+");
934 if (words.length == 0 || words[0].isEmpty()) continue;
935 String command = words[0];
936 List<String> parameters = Arrays.asList(words).subList(1,words.length);
937 try {
938 if (COMMANDS.contains(command)) {
939 if (!validArguments(command,parameters)) System.out.println("ERROR bad arguments");
940 else if (command.equals("NEW")) {
941 if (game != null && !game.state.ended) {
942 game.emit("game stopped");
943 game.summary();
944 }
945 long seed = Long.parseLong(parameters.get(0));
946 int level = Integer.parseInt(parameters.get(1));
947 System.out.println("game " + ++serial + " seed " + seed + " level " + level);
948 game = new Game(serial,seed,level);
949 } else if (game == null) System.out.println("ERROR no game");
950 else if (game.state.ended && !AFTER_END.contains(command)) System.out.println("ERROR game over");
951 else game.command(command,parameters);
952 } else {
953 String invalid = null;
954 for (String word : words) if (!OPERATIONS.contains(word)) { invalid = word; break; }
955 if (invalid != null) System.out.println("ERROR unknown " + invalid);
956 else if (game == null) System.out.println("ERROR no game");
957 else if (game.state.ended) System.out.println("ERROR game over");
958 else for (String action : words) game.operation(action);
959 }
960 } catch (GameEnd ignored) {}
961 }
962 if (game != null && !game.state.ended) {
963 game.emit("game stopped");
964 game.summary();
965 }
966 }
