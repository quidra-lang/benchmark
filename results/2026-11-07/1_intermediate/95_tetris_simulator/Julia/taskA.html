114seed_step(x::Int) = mod(48271*x,2147483647)
115spawn(kind::Int) = (0,kind==2 ? 5 : 4,22)
116piece_cells(kind::Int,p::Tuple{Int,Int,Int}) = [(p[2]+b[1],p[3]-b[2]) for b in SHAPES[kind][p[1]+1]]
117lowest(kind::Int,p::Tuple{Int,Int,Int}) = minimum(b[2] for b in piece_cells(kind,p))
118function bag(random::Int)
119 pieces = collect(1:7)
120 for i in 7:-1:2
121 random = seed_step(random)
122 j = mod(random,i)+1
123 pieces[i],pieces[j] = pieces[j],pieces[i]
124 end
125 random,pieces
126end
127function fits(board::Vector{Int},kind::Int,p::Tuple{Int,Int,Int})
128 for (x,y) in piece_cells(kind,p)
129 if x<1 || x>10 || y<1 || y>22 || (board[y] & (1<<(x-1))) != 0
130 return false
131 end
132 end
133 true
134end
135function move(board::Vector{Int},kind::Int,p::Tuple{Int,Int,Int},op::Int)
136 if op==7 || op==8
137 b = mod(p[1]+(op==7 ? 1 : 3),4)
138 row = findfirst(x->x==(p[1],b),KICK_STATES)
139 offsets = kind==2 ? [(0,0)] : (kind==1 ? I_KICKS : ORDINARY_KICKS)[row]
140 for (i,d) in enumerate(offsets)
141 q = (b,p[2]+d[1],p[3]+d[2])
142 if fits(board,kind,q)
143 return q,i
144 end
145 end
146 return nothing
147 end
148 dx = op==1 || op==3 ? -1 : op==2 || op==4 ? 1 : 0
149 dy = dx==0 ? -1 : 0
150 repeat = op==3 || op==4 || op==6
151 q=p
152 changed=false
153 while true
154 next = (q[1],q[2]+dx,q[3]+dy)
155 fits(board,kind,next) || break
156 q=next
157 changed=true
158 repeat || break
159 end
160 changed ? (q,0) : nothing
161end
162landing(board,kind,p) = let result=move(board,kind,p,6); result === nothing ? p : result[1] end
163function placed(board::Vector{Int},kind::Int,p::Tuple{Int,Int,Int})
164 next=copy(board)
165 for (x,y) in piece_cells(kind,p)
166 next[y] |= 1<<(x-1)
167 end
168 remaining=filter(!=(1023),next)
169 n=22-length(remaining)
170 append!(remaining,zeros(Int,n))
171 remaining,n
172end
173function evaluate(board::Vector{Int},cleared::Int)
174 heights=zeros(Int,10)
175 holes=0
176 for c in 1:10
177 for r in 22:-1:1
178 if (board[r] & (1<<(c-1))) != 0
179 heights[c]=r
180 break
181 end
182 end
183 for r in 1:heights[c]
184 if (board[r] & (1<<(c-1))) == 0
185 holes += 1
186 end
187 end
188 end
189 bump=sum(abs(heights[c]-heights[c+1]) for c in 1:9)
190 (heights,holes,bump,76*cleared-51*sum(heights)-36*holes-18*bump)
191end
192function routes(board::Vector{Int},kind::Int,start::Tuple{Int,Int,Int},target=nothing)
193 fits(board,kind,start) || return Route[],nothing
194 entries=[Route(start,0,0)]
195 seen=Set([start])
196 i=1
197 while i<=length(entries)
198 entry=entries[i]
199 if target !== nothing && landing(board,kind,entry.position)==target
200 return entries,i
201 end
202 for op in 1:8
203 result=move(board,kind,entry.position,op)
204 result === nothing && continue
205 p=result[1]
206 if !(p in seen)
207 push!(seen,p)
208 push!(entries,Route(p,i,op))
209 end
210 end
211 i+=1
212 end
213 target === nothing ? (entries,nothing) : (Route[],nothing)
214end
215function shortest(routes::Vector{Route},index::Int)
216 ops=Int[]
217 while routes[index].parent>0
218 push!(ops,routes[index].operation)
219 index=routes[index].parent
220 end
221 reverse!(ops)
222end
223function t_spin(board::Vector{Int},p::Tuple{Int,Int,Int},test::Int)
224 corners=[(p[2],p[3]),(p[2]+2,p[3]),(p[2]+2,p[3]-2),(p[2],p[3]-2)]
225 blocked=Bool[]
226 for (x,y) in corners
227 push!(blocked,x<1 || x>10 || y<1 || y>22 || (board[y] & (1<<(x-1))) != 0)
228 end
229 count(identity,blocked)<3 && return 0
230 blocked[p[1]+1] && blocked[mod(p[1]+1,4)+1] || test==5 ? 2 : 1
231end

290function operation!(game::Game,op::Int)
291 s=game.s
292 if op<=8
293 s.keys+=1
294 result=move(s.board,s.kind,s.position,op)
295 if result===nothing
296 println("fail ",OPERATIONS[op]);return
297 end
298 p,test=result
299 if op==5 || op==6
300 drops!(s,s.position[3]-p[3])
301 update!(s,p,1)
302 s.fall=0;s.ground=0
303 else
304 update!(s,p,op>=7 ? 2 : 1,test)
305 if s.ground>0 && s.resets<15
306 s.ground=0;s.resets+=1
307 end
308 end
309 elseif op==9
310 p=landing(s.board,s.kind,s.position)
311 rows=s.position[3]-p[3]
312 drops!(s,2*rows)
313 rows>0 && update!(s,p,1)
314 lock!(game)
315 elseif op==10
316 if grounded(s)
317 lock!(game)
318 else
319 println("fail LOCK")
320 end
321 else
322 if s.used_hold
323 println("fail HOLD");return
324 end
325 s.used_hold=true;s.stats.holds+=1
326 previous=s.hold;s.hold=s.kind
327 spawn_piece!(game,previous==0 ? next_piece!(game) : previous)
328 end
329end
330function tick!(game::Game)
331 s=game.s
332 if !grounded(s)
333 s.ground=0;s.fall+=1
334 interval=s.level>=13 ? 1 : [60,48,37,28,21,16,11,8,6,4,3,2][s.level]
335 if s.fall>=interval
336 update!(s,(s.position[1],s.position[2],s.position[3]-1),1)
337 s.fall=0
338 end
339 else
340 s.ground+=1
341 s.ground>=30 && lock!(game)
342 end
343end
344function lock!(game::Game)
345 s=game.s
346 kind=s.kind;p=s.position
347 old_cells=piece_cells(kind,p)
348 route,index=routes(s.board,kind,spawn(kind),p)
349 shortest_ops=index===nothing ? nothing : shortest(route,index)
350 spin=kind==3 && s.last==2 ? t_spin(s.board,p,s.test) : 0
351 s.board,n=placed(s.board,kind,p)
352 push!(game.history,deepcopy(game.start))
353 s.pieces+=1;s.stats.kinds[kind]+=1;s.stats.operations+=s.keys
354 clear_name,points,sent=clear_info(spin,n)
355 level=s.level;points*=level;back=false
356 if n>0
357 s.streak+=1
358 if n==4 || spin!=0
359 s.chain+=1;back=s.chain>=2
360 else
361 s.chain=0
362 end
363 else
364 s.streak=0
365 end
366 if back
367 points=div(3*points,2);sent+=1
368 end
369 combo=s.streak-1
370 combo_points=combo>=1 ? 50*combo*level : 0
371 if combo>=0
372 sent+=[0,1,1,2,2,3,3,4,4,4,5][min(combo,10)+1]
373 end
374 perfect=n>0 && all(==(0),s.board)
375 perfect_points=perfect ? (n==4 && back ? 3200 : [800,1200,1800,2000][n])*level : 0
376 if perfect
377 sent+=10;s.stats.perfect+=1
378 end
379 gained=points+combo_points+perfect_points
380 s.score+=gained;s.lines+=n;s.stats.sent+=sent
381 s.stats.combo=max(s.stats.combo,combo);s.stats.chain=max(s.stats.chain,s.chain)
382 count_text=shortest_ops===nothing ? "-" : string(length(shortest_ops))
383 println("lock $(s.pieces) $(KINDS[kind]) $(STATES[p[1]+1]) $(p[2]) $(p[3]) keys $(s.keys) $count_text drop $(s.drop) lines $n sent $sent +$gained total $(s.score)")
384 if shortest_ops !== nothing && s.keys>length(shortest_ops)
385 s.stats.faults+=1
386 println("finesse ",isempty(shortest_ops) ? "none" : ops_text(shortest_ops))
387 end
388 if clear_name>0
389 s.stats.clears[clear_name]+=1
390 line="clear $(CLEAR_NAMES[clear_name]) $points"
391 back && (line*=" b2b")
392 combo>=1 && (line*=" combo $combo $combo_points")
393 perfect && (line*=" perfect $perfect_points")
394 println(line)
395 end
396 next_level=s.initial_level+div(s.lines,10)
397 if next_level>level
398 s.level=next_level;println("level $(s.level)")
399 end
400 all(c->c[2]>=21,old_cells) && game_over!(game,"lock out")
401 cancelled=min(sent,sum(s.pending))
402 remaining=cancelled
403 while remaining>0
404 used=min(remaining,s.pending[1]);remaining-=used;s.pending[1]-=used
405 s.pending[1]==0 && popfirst!(s.pending)
406 end
407 if cancelled>0
408 s.stats.cancelled+=cancelled;println("cancel $cancelled pending $(sum(s.pending))")
409 end
410 if n==0
411 risen=0
412 while !isempty(s.pending) && risen<8
413 count=min(s.pending[1],8-risen)
414 s.random=seed_step(s.random)
415 hole=mod(s.random,10)+1
416 insert_garbage!(game,count,hole)
417 s.stats.received+=count;risen+=count;s.pending[1]-=count
418 s.pending[1]==0 && popfirst!(s.pending)
419 println("rise $count hole $hole pending $(sum(s.pending))")
420 end
421 end
422 s.used_hold=false
423 spawn_piece!(game,next_piece!(game))
424 game.start=deepcopy(s)
425end
426function candidates(s::State,kind::Int,start,prefix::Vector{Int})
427 result=Candidate[]
428 seen=Set{NTuple{4,Int}}()
429 for turns in [Int[],[7],[7,7],[8]]
430 p=start

435 valid=false;break
436 end

440 function add(q,shifts::Vector{Int})
441 landed=landing(s.board,kind,q)
442 occupied=sort([(c[2]-1)*10+c[1]-1 for c in piece_cells(kind,landed)])
443 key=Tuple(occupied)
444 key in seen && return
445 push!(seen,key)
446 board,n=placed(s.board,kind,landed)
447 value=evaluate(board,n)[4]
448 push!(result,Candidate(kind,value,vcat(prefix,turns,shifts,[9]),0))
449 end
450 add(p,Int[])
451 for direction in [1,2]
452 q=p;shifts=Int[]
453 while true
454 next=move(s.board,kind,q,direction)
455 next===nothing && break
456 q=next[1];push!(shifts,direction);add(q,shifts)
457 end
458 end
459 end
460 result
461end
462function ranked(s::State)
463 result=candidates(s,s.kind,s.position,Int[])
464 held=s.hold==0 ? s.queue[1] : s.hold
465 if !s.used_hold && fits(s.board,held,spawn(held))
466 append!(result,candidates(s,held,spawn(held),[11]))
467 end
468 for (i,c) in enumerate(result)
469 c.order=i
470 end
471 sort!(result,by=c->(-c.value,c.order))
472end
473function spins(s::State)
474 if s.kind!=3
475 println("spins 0");return
476 end
477 found=Dict{NTuple{3,Int},Int}()
478 for entry in routes(s.board,3,s.position)[1]
479 for op in [7,8]
480 result=move(s.board,3,entry.position,op)
481 result===nothing && continue
482 p,test=result

487 end
488 end
489 end
490 rows=sort(collect(found),by=x->(x.first[3],x.first[2],x.first[1]))
491 for (p,spin) in rows
492 _,n=placed(s.board,3,p)
493 name=clear_info(spin,n)[1]
494 println("spin $(STATES[p[1]+1]) $(p[2]) $(p[3]) $(CLEAR_NAMES[name])")
495 end
496 println("spins $(length(rows))")
497end

512function command!(game::Game,command::String,args::Vector{String})
513 s=game.s
514 if command=="PRINT"
515 current=s.ended ? Set{Tuple{Int,Int}}() : Set(piece_cells(s.kind,s.position))
516 ghost=s.ended ? Set{Tuple{Int,Int}}() : Set(piece_cells(s.kind,landing(s.board,s.kind,s.position)))
517 for r in 22:-1:1
518 line=lpad(string(r),2)*" "
519 for c in 1:10
520 ch=(s.board[r]&(1<<(c-1)))!=0 ? '#' : (c,r) in current ? '@' : (c,r) in ghost ? '+' : '.'
521 line*=string(ch)
522 end
523 println(line)
524 end
525 elseif command=="NEXT"
526 println("hold ",s.hold==0 ? '-' : KINDS[s.hold]," next ",join((KINDS[i] for i in s.queue[1:5])))
527 elseif command=="PREVIEW"
528 n=parse(Int,args[1]);queue=copy(s.queue);random=s.random
529 while length(queue)<n
530 random,pieces=bag(random);append!(queue,pieces)
531 end
532 println("preview ",join((KINDS[i] for i in queue[1:n])))
533 elseif command=="STATUS"
534 println("score $(s.score) level $(s.level) lines $(s.lines) pieces $(s.pieces)")
535 piece=s.ended ? "-" : "$(KINDS[s.kind]) $(STATES[s.position[1]+1]) $(s.position[2]) $(s.position[3])"
536 println("piece $piece streak $(s.streak) chain $(s.chain) pending $(sum(s.pending))")
537 elseif command=="ANALYZE"
538 heights,holes,bump,value=evaluate(s.board,0)
539 println("heights ",join(heights," "))
540 println("holes $holes bumpiness $bump value $value")
541 elseif command=="ROW"
542 row=parse(Int,args[1]);pattern=args[2]
543 if any(c->c[2]==row && pattern[c[1]]=='#',piece_cells(s.kind,s.position))
544 println("ERROR overlap");return
545 end
546 mask=0
547 for i in 1:10
548 pattern[i]=='#' && (mask |= 1<<(i-1))
549 end
550 s.board[row]=mask;println("OK")
551 elseif command=="GARBAGE"
552 n=parse(Int,args[1]);hole=parse(Int,args[2]);insert_garbage!(game,n,hole)
553 for j in 0:n
554 p=(s.position[1],s.position[2],s.position[3]+j)
555 if fits(s.board,s.kind,p)
556 s.position=p;println("OK");return
557 end
558 end
559 game_over!(game,"top out")
560 elseif command=="ATTACK"
561 push!(s.pending,parse(Int,args[1]));println("pending $(sum(s.pending))")
562 elseif command=="TICK"
563 for _ in 1:parse(Int,args[1]);tick!(game);end
564 elseif command=="UNDO"
565 if isempty(game.history)
566 println("ERROR nothing to undo")
567 else
568 game.s=pop!(game.history)
569 game.start=deepcopy(game.s)
570 println("undo $(game.s.pieces) $(KINDS[game.s.kind])")
571 end
572 elseif command=="SAVE"
573 game.saves[args[1]]=(deepcopy(s),deepcopy(game.history),deepcopy(game.start))
574 println("OK")
575 elseif command=="LOAD"
576 if !haskey(game.saves,args[1])
577 println("ERROR unknown save")
578 else
579 game.s,game.history,game.start=deepcopy(game.saves[args[1]])
580 println("load $(game.s.pieces) $(KINDS[game.s.kind])")
581 end
582 elseif command=="SPINS"
583 spins(s)
584 elseif command=="HINT"
585 list=ranked(s);n=min(parse(Int,args[1]),length(list))
586 for i in 1:n
587 c=list[i]
588 println("hint $i $(KINDS[c.kind]) $(ops_text(c.operations)) value $(c.value)")
589 end
590 elseif command=="AUTO"
591 for _ in 1:parse(Int,args[1])
592 c=first(ranked(s))
593 println("auto $(KINDS[c.kind]) $(ops_text(c.operations))")
594 for op in c.operations
595 operation!(game,op)
596 end
597 end
598 end
599end
600function valid_number(text::String,min_value::Int,max_value::Int)
601 occursin(r"^(0|[1-9][0-9]*)$",text) || return false
602 n=tryparse(Int,text)
603 n!==nothing && min_value<=n<=max_value
604end
605function valid_args(command::String,args::Vector{String})
606 if command=="NEW"
607 return length(args)==2 && valid_number(args[1],1,2147483646) && valid_number(args[2],1,15)
608 elseif command=="ROW"
609 return length(args)==2 && valid_number(args[1],1,22) && occursin(r"^[#.]{10}$",args[2]) && occursin('.',args[2])
610 elseif command=="GARBAGE"
611 return length(args)==2 && valid_number(args[1],1,10) && valid_number(args[2],1,10)
612 elseif command=="SAVE" || command=="LOAD"
613 return length(args)==1 && occursin(r"^[a-z]{1,8}$",args[1])
614 end
615 limits=Dict("PREVIEW"=>14,"ATTACK"=>20,"TICK"=>1000,"HINT"=>10,"AUTO"=>50)
616 if haskey(limits,command)
617 return length(args)==1 && valid_number(args[1],1,limits[command])
618 end
619 isempty(args)
620end
621function main()
622 n=parse(Int,readline(stdin))
623 game=nothing
624 serial=0
625 for _ in 1:n
626 words=split(readline(stdin),' ')
627 command=String(words[1])
628 args=String.(words[2:end])
629 try
630 if command in COMMANDS
631 if !valid_args(command,args)
632 println("ERROR bad arguments")
633 elseif command=="NEW"
634 if game !== nothing && !game.s.ended
635 println("game stopped");summary(game)
636 end
637 serial+=1
638 println("game $serial seed $(parse(Int,args[1])) level $(parse(Int,args[2]))")
639 game=new_game(serial,parse(Int,args[1]),parse(Int,args[2]))
640 elseif game===nothing
641 println("ERROR no game")
642 elseif game.s.ended && !(command in AFTER_END)
643 println("ERROR game over")
644 else
645 command!(game,command,args)
646 end
647 else
648 invalid=findfirst(w->!(w in OPERATIONS),words)
649 if invalid !== nothing
650 println("ERROR unknown $(words[invalid])")
651 elseif game===nothing
652 println("ERROR no game")
653 elseif game.s.ended
654 println("ERROR game over")
655 else
656 for word in words
657 operation!(game,findfirst(==(word),OPERATIONS))
658 end
659 end
660 end
661 catch e
662 e isa GameOver || rethrow(e)
663 end
664 end
665 if game !== nothing && !game.s.ended
666 println("game stopped");summary(game)
667 end
668end
