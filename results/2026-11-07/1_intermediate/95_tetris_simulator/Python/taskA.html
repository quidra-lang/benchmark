89def rotation_states(rows):
90 size = len(rows)
91 blocks = [(i, j) for j, row in enumerate(rows) for i, mark in enumerate(row) if mark == "#"]
92 states = []
93 for _ in range(4):
94 states.append(tuple(blocks))
95 blocks = [(size - 1 - j, i) for i, j in blocks]
96 return tuple(states)

119def shuffled_bag(value):
120 bag = list(KINDS)
121 for i in range(len(bag) - 1, 0, -1):
122 value = next_random(value)
123 j = value % (i + 1)
124 bag[i], bag[j] = bag[j], bag[i]
125 return bag, value

143class Well:
144 def __init__(self):
145 self.rows = [[False] * WIDTH for _ in range(HEIGHT)]
146
147 def copy(self):
148 well = Well()
149 well.rows = [row[:] for row in self.rows]
150 return well
151
152 def filled(self, column, row):
153 return self.rows[row - 1][column - 1]
154
155 def blocked(self, column, row):
156 return not (1 <= column <= WIDTH and 1 <= row <= HEIGHT) or self.filled(column, row)
157
158 def fits(self, kind, position):
159 return not any(self.blocked(column, row) for column, row in piece_cells(kind, position))
160
161 def grounded(self, kind, position):
162 return not self.fits(kind, offset(position, 0, -1))
163
164 def landing(self, kind, position):
165 while not self.grounded(kind, position):
166 position = offset(position, 0, -1)
167 return position
168
169 def place(self, kind, position):
170 for column, row in piece_cells(kind, position):
171 self.rows[row - 1][column - 1] = True
172 kept = [row for row in self.rows if not all(row)]
173 cleared = HEIGHT - len(kept)
174 self.rows = kept + [[False] * WIDTH for _ in range(cleared)]
175 return cleared
176
177 def is_empty(self):
178 return not any(any(row) for row in self.rows)
179
180 def set_row(self, row, pattern):
181 self.rows[row - 1] = [mark == "#" for mark in pattern]
182
183 def insert_garbage(self, count, hole):
184 if any(any(row) for row in self.rows[HEIGHT - count:]):
185 return False
186 garbage = [[column != hole for column in range(1, WIDTH + 1)] for _ in range(count)]
187 self.rows = garbage + self.rows[:HEIGHT - count]
188 return True
189
190 def metrics(self):
191 heights = []
192 holes = 0
193 for column in range(WIDTH):
194 height = max(
195 (number for number, row in enumerate(self.rows, 1) if row[column]), default=0
196 )
197 heights.append(height)
198 holes += sum(1 for row in self.rows[:height] if not row[column])
199 bumpiness = sum(abs(left - right) for left, right in zip(heights, heights[1:]))
200 return heights, holes, bumpiness

203def evaluation(well, cleared):
204 heights, holes, bumpiness = well.metrics()
205 return 76 * cleared - 51 * sum(heights) - 36 * holes - 18 * bumpiness

208def rotate(well, kind, position, operation):
209 after = (position.state + (1 if operation == "CW" else 3)) % 4
210 if kind == "O":
211 kicks = ((0, 0),)
212 else:
213 kicks = (I_KICKS if kind == "I" else KICKS)[position.state, after]
214 for test, (dx, dy) in enumerate(kicks, 1):
215 target = Position(after, position.x + dx, position.y + dy)
216 if well.fits(kind, target):
217 return target, test
218 return None
219
220
221def move(well, kind, position, operation):
222 if operation in ROTATIONS:
223 return rotate(well, kind, position, operation)
224 dx, dy = STEPS[operation]
225 target = position
226 while well.fits(kind, offset(target, dx, dy)):
227 target = offset(target, dx, dy)
228 if operation not in REPEATED:
229 break
230 return None if target == position else (target, 0)

233def explore(well, kind, start):
234 paths = {start: ()}
235 frontier = deque([start])
236 while frontier:
237 position = frontier.popleft()
238 yield position, paths[position]
239 for operation in MOVEMENTS:
240 moved = move(well, kind, position, operation)
241 if moved is not None and moved[0] not in paths:
242 paths[moved[0]] = paths[position] + (operation,)
243 frontier.append(moved[0])
244

246def shortest_sequence(well, kind, target):
247 start = SPAWNS[kind]
248 if not well.fits(kind, start):
249 return None
250 for position, path in explore(well, kind, start):
251 if well.landing(kind, position) == target:
252 return path
253 return None

256def tspin_type(well, position, test):
257 state, x, y = position
258 corners = [
259 well.blocked(column, row)
260 for column, row in ((x, y), (x + 2, y), (x + 2, y - 2), (x, y - 2))
261 ]
262 if sum(corners) < 3:
263 return None
264 if corners[state] and corners[(state + 1) % 4] or test == 5:
265 return "full"
266 return "mini"

269def rotated(well, kind, position, rotations):
270 for rotation in rotations:
271 moved = move(well, kind, position, rotation)
272 if moved is None:
273 return None
274 position = moved[0]
275 return position

277
278def shift_sequences(well, kind, position):
279 yield (), position
280 for step in ("L", "R"):
281 current = position
282 shifts = ()
283 while (moved := move(well, kind, current, step)) is not None:
284 current = moved[0]
285 shifts += (step,)
286 yield shifts, current

289def placements(well, kind, start, prefix):
290 candidates = []
291 seen = set()
292 for rotations in ROTATION_SEQUENCES:
293 position = rotated(well, kind, start, rotations)
294 if position is None:
295 continue
296 for shifts, shifted in shift_sequences(well, kind, position):
297 landing = well.landing(kind, shifted)
298 blocks = frozenset(piece_cells(kind, landing))
299 if blocks in seen:
300 continue
301 seen.add(blocks)
302 trial = well.copy()
303 cleared = trial.place(kind, landing)
304 operations = prefix + rotations + shifts + ("HD",)
305 candidates.append(Candidate(evaluation(trial, cleared), kind, operations))
306 return candidates

426 def perform(self, operations):
427 for operation in operations:
428 if not self.operate(operation):
429 print(f"fail {operation}")

431 def operate(self, operation):
432 if operation == "HD":
433 self.hard_drop()
434 elif operation == "LOCK":
435 if not self.well.grounded(self.kind, self.position):
436 return False
437 self.lock()
438 elif operation == "HOLD":
439 return self.hold_piece()
440 else:
441 return self.move_piece(operation)
442 return True

444 def add_drop_points(self, points):
445 self.score += points
446 self.drop_points += points
447 self.stats.drop_total += points
448
449 def move_piece(self, operation):
450 self.operations += 1
451 moved = move(self.well, self.kind, self.position, operation)
452 if moved is None:
453 return False
454 target, test = moved
455 if operation in ("SD", "DD"):
456 self.add_drop_points(self.position.y - target.y)
457 self.update_position(target, "move")
458 self.fall_counter = 0
459 self.ground_counter = 0
460 return True
461 if operation in ROTATIONS:
462 self.update_position(target, "rotation", test)
463 else:
464 self.update_position(target, "move")
465 if self.ground_counter > 0 and self.resets < MAX_RESETS:
466 self.ground_counter = 0
467 self.resets += 1
468 return True

470 def hard_drop(self):
471 landing = self.well.landing(self.kind, self.position)
472 rows = self.position.y - landing.y
473 self.add_drop_points(2 * rows)
474 if rows > 0:
475 self.update_position(landing, "move")
476 self.lock()

478 def hold_piece(self):
479 if self.hold_used:
480 return False
481 self.hold_used = True
482 self.stats.hold_count += 1
483 if self.hold is None:
484 self.hold = self.kind

486 else:

492 def tick(self):
493 if not self.well.grounded(self.kind, self.position):
494 self.ground_counter = 0
495 self.fall_counter += 1
496 if self.fall_counter >= fall_interval(self.level):
497 self.update_position(offset(self.position, 0, -1), "move")
498 self.fall_counter = 0
499 else:
500 self.ground_counter += 1
501 if self.ground_counter >= LOCK_DELAY:
502 self.lock()

504 def lock(self):
505 kind, position = self.kind, self.position
506 shortest = shortest_sequence(self.well, kind, position)
507 spin = None
508 if kind == "T" and self.last_motion == "rotation":
509 spin = tspin_type(self.well, position, self.rotation_test)
510 blocks = piece_cells(kind, position)
511 cleared = self.well.place(kind, position)
512 self.history.append(self.start_state)
513 self.pieces += 1
514 self.stats.kind_counts[kind] += 1
515 self.stats.operation_total += self.operations
516 sent = self.score_lock(shortest, spin, cleared)
517 if all(row > VISIBLE_HEIGHT for _, row in blocks):
518 self.game_over("lock out")
519 self.cancel_garbage(sent)
520 if cleared == 0:
521 self.receive_garbage()
522 self.hold_used = False
523 self.spawn(self.take_next())
524 self.record_start()

526 def score_lock(self, shortest, spin, cleared):
527 stats = self.stats
528 level = self.level
529 clear_type = CLEAR_TYPES.get((spin, cleared))
530 name, base_points, sent = clear_type or (None, 0, 0)
531 points = base_points * level
532 back_to_back = False
533 if cleared > 0:
534 self.streak += 1
535 if cleared == 4 or spin is not None:
536 self.chain += 1
537 back_to_back = self.chain >= 2
538 else:
539 self.chain = 0
540 else:
541 self.streak = 0
542 if back_to_back:
543 points = points * 3 // 2
544 sent += 1
545 combo = self.streak - 1
546 combo_points = 50 * combo * level if combo >= 1 else 0
547 if combo >= 0:
548 sent += COMBO_LINES[min(combo, len(COMBO_LINES) - 1)]
549 perfect = cleared > 0 and self.well.is_empty()
550 perfect_points = 0
551 if perfect:
552 if cleared == 4 and back_to_back:
553 perfect_points = B2B_TETRIS_PERFECT_POINTS * level
554 else:
555 perfect_points = PERFECT_POINTS[cleared] * level
556 sent += 10
557 stats.perfects += 1
558 gained = points + combo_points + perfect_points
559 self.score += gained
560 self.lines += cleared
561 stats.sent += sent
562 stats.max_combo = max(stats.max_combo, combo)
563 stats.max_chain = max(stats.max_chain, self.chain)
564
565 state, x, y = self.position
566 keys = "-" if shortest is None else len(shortest)
567 print(
568 f"lock {self.pieces} {self.kind} {STATE_NAMES[state]} {x} {y}"
569 f" keys {self.operations} {keys} drop {self.drop_points}"
570 f" lines {cleared} sent {sent} +{gained} total {self.score}"
571 )
572 if shortest is not None and self.operations > len(shortest):
573 stats.faults += 1
574 print("finesse " + (" ".join(shortest) or "none"))
575 if clear_type is not None:
576 stats.clear_counts[name] += 1
577 line = f"clear {name} {points}"
578 if back_to_back:
579 line += " b2b"
580 if combo >= 1:
581 line += f" combo {combo} {combo_points}"
582 if perfect:
583 line += f" perfect {perfect_points}"
584 print(line)
585 new_level = self.start_level + self.lines // 10
586 if new_level > level:
587 self.level = new_level
588 print(f"level {self.level}")
589 return sent

591 def reduce_pending(self, amount):
592 while amount > 0:
593 used = min(amount, self.pending[0])
594 self.pending[0] -= used
595 amount -= used
596 if self.pending[0] == 0:
597 self.pending.pop(0)

599 def cancel_garbage(self, sent):
600 cancelled = min(sent, sum(self.pending))
601 if cancelled >= 1:
602 self.reduce_pending(cancelled)
603 self.stats.cancelled += cancelled
604 print(f"cancel {cancelled} pending {sum(self.pending)}")

606 def receive_garbage(self):
607 risen = 0
608 while self.pending and risen < MAX_RISE:
609 rows = min(self.pending[0], MAX_RISE - risen)
610 hole = self.draw_random() % WIDTH + 1
611 self.insert_garbage(rows, hole)
612 self.stats.received += rows
613 risen += rows
614 self.reduce_pending(rows)
615 print(f"rise {rows} hole {hole} pending {sum(self.pending)}")

718 def spins(self):
719 if self.kind != "T":
720 print("spins 0")
721 return
722 found = {}
723 for position, _ in explore(self.well, "T", self.position):
724 for rotation in ROTATIONS:
725 moved = move(self.well, "T", position, rotation)
726 if moved is None or not self.well.grounded("T", moved[0]):
727 continue
728 target, test = moved
729 spin = tspin_type(self.well, target, test)
730 if spin == "full" or spin is not None and target not in found:
731 found[target] = spin
732 for target in sorted(found, key=lambda position: (position.y, position.x, position.state)):
733 trial = self.well.copy()
734 cleared = trial.place("T", target)
735 name = CLEAR_TYPES[found[target], cleared][0]
736 print(f"spin {STATE_NAMES[target.state]} {target.x} {target.y} {name}")
737 print(f"spins {len(found)}")

787class Session:
788 def __init__(self):
789 self.game = None
790 self.games = 0
791
792 def stop(self):
793 if self.game is not None and not self.game.ended:
794 print("game stopped")
795 self.game.print_summary()
796
797 def new_game(self, seed, level):
798 self.stop()
799 self.games += 1
800 print(f"game {self.games} seed {seed} level {level}")
801 self.game = Game(self.games, seed, level)
802
803 def run_command(self, name, arguments):
804 command = COMMANDS[name]
805 values = [parse(word) for parse, word in zip(command.parsers, arguments)]
806 if len(arguments) != len(command.parsers) or None in values:
807 print("ERROR bad arguments")
808 elif name == "NEW":
809 self.new_game(*values)
810 elif self.game is None:
811 print("ERROR no game")
812 elif self.game.ended and not command.after_end:
813 print("ERROR game over")
814 else:
815 command.handler(self.game, *values)
816
817 def run_operations(self, words):
818 unknown = next((word for word in words if word not in OPERATIONS), None)
819 if unknown is not None:
820 print(f"ERROR unknown {unknown}")
821 elif self.game is None:
822 print("ERROR no game")
823 elif self.game.ended:
824 print("ERROR game over")
825 else:
826 self.game.perform(words)
827
828 def execute(self, line):
829 words = line.split(" ")
830 try:
831 if words[0] in COMMANDS:
832 self.run_command(words[0], words[1:])
833 else:
834 self.run_operations(words)
835 except GameOver:
836 pass
