107function fits(array $board, string $kind, array $p): bool
108{
109 foreach (cells($kind, $p) as [$c, $r]) {

116function movePiece(array $board, string $kind, array $p, string $operation): ?array
117{
118 global $iKicks, $ordinaryKicks;
119 [$s, $x, $y] = $p;
120 if ($operation === "CW" || $operation === "CCW") {
121 $b = ($s + ($operation === "CW" ? 1 : 3)) % 4;
122 $offsets = $kind === "O" ? [[0, 0]] : ($kind === "I" ? $iKicks : $ordinaryKicks)["$s$b"];
123 foreach ($offsets as $i => $delta) {
124 $q = [$b, $x + $delta[0], $y + $delta[1]];
125 if (fits($board, $kind, $q)) {
126 return ["p" => $q, "test" => $i + 1];
127 }
128 }
129 return null;
130 }
131 $dx =
132 $operation === "L" || $operation === "LL"
133 ? -1
134 : ($operation === "R" || $operation === "RR"
135 ? 1
136 : 0);
137 $dy = $dx === 0 ? -1 : 0;
138 $repeat = in_array($operation, ["LL", "RR", "DD"], true);
139 $position = $p;
140 $changed = false;
141 do {
142 $q = [$s, $position[1] + $dx, $position[2] + $dy];
143 if (!fits($board, $kind, $q)) {
144 break;
145 }
146 $position = $q;
147 $changed = true;
148 } while ($repeat);
149 return $changed ? ["p" => $position, "test" => 0] : null;
150}

155function placed(array $board, string $kind, array $p): array
156{

164function evaluateWell(array $board, int $cleared): array
165{
166 $heights = array_fill(0, 10, 0);
167 $holes = 0;
168 $bumpiness = 0;
169 for ($c = 0; $c < 10; $c++) {
170 for ($r = 21; $r >= 0; $r--) {
171 if (($board[$r] & (1 << $c)) !== 0) {
172 $heights[$c] = $r + 1;
173 break;
174 }
175 }
176 for ($r = 0; $r < $heights[$c]; $r++) {
177 if (($board[$r] & (1 << $c)) === 0) {
178 $holes++;
179 }
180 }
181 }
182 for ($c = 0; $c < 9; $c++) {
183 $bumpiness += abs($heights[$c] - $heights[$c + 1]);
184 }
185 return [
186 "heights" => $heights,
187 "holes" => $holes,
188 "bumpiness" => $bumpiness,
189 "value" => 76 * $cleared - 51 * array_sum($heights) - 36 * $holes - 18 * $bumpiness,
190 ];
191}
192function routes(array $board, string $kind, array $start, ?array $target = null): array
193{
194 global $movements;
195 if (!fits($board, $kind, $start)) {
196 return [];
197 }
198 $entries = [["position" => $start, "keys" => []]];
199 $seen = [positionKey($start) => true];
200 for ($i = 0; $i < count($entries); $i++) {
201 $entry = $entries[$i];
202 if ($target !== null && landing($board, $kind, $entry["position"]) === $target) {
203 return [$entry];
204 }
205 foreach ($movements as $op) {
206 $r = movePiece($board, $kind, $entry["position"], $op);
207 if ($r === null) {
208 continue;
209 }
210 $key = positionKey($r["p"]);
211 if (isset($seen[$key])) {
212 continue;
213 }
214 $seen[$key] = true;
215 $entries[] = ["position" => $r["p"], "keys" => array_merge($entry["keys"], [$op])];
216 }
217 }
218 return $target !== null ? [] : $entries;
219}
220function tSpin(array $board, array $p, int $test): string
221{
222 [$s, $x, $y] = $p;
223 $corners = [[$x, $y], [$x + 2, $y], [$x + 2, $y - 2], [$x, $y - 2]];
224 $blocked = [];
225 foreach ($corners as [$c, $r]) {
226 $blocked[] =
227 $c < 1 || $c > 10 || $r < 1 || $r > 22 || ($board[$r - 1] & (1 << ($c - 1))) !== 0;
228 }
229 if (count(array_filter($blocked)) < 3) {
230 return "";
231 }
232 return ($blocked[$s] && $blocked[($s + 1) % 4]) || $test === 5 ? "full" : "mini";

382 public function operation(string $op): void
383 {
384 global $movements;
385 $s = &$this->s;
386 if (in_array($op, $movements, true)) {
387 $s["keys"]++;
388 $r = movePiece($s["board"], $s["kind"], $s["position"], $op);
389 if ($r === null) {
390 emit("fail " . $op);
391 return;
392 }
393 if ($op === "SD" || $op === "DD") {
394 $this->drops($s["position"][2] - $r["p"][2]);
395 $this->updatePosition($r["p"], "move");
396 $s["fall"] = 0;
397 $s["ground"] = 0;
398 } else {
399 $this->updatePosition(
400 $r["p"],
401 $op === "CW" || $op === "CCW" ? "rotation" : "move",
402 $r["test"],
403 );
404 if ($s["ground"] > 0 && $s["resets"] < 15) {
405 $s["ground"] = 0;
406 $s["resets"]++;
407 }
408 }
409 return;
410 }
411 if ($op === "HD") {
412 $p = landing($s["board"], $s["kind"], $s["position"]);
413 $rows = $s["position"][2] - $p[2];
414 $this->drops(2 * $rows);
415 if ($rows > 0) {
416 $this->updatePosition($p, "move");
417 }
418 $this->lock();
419 return;
420 }
421 if ($op === "LOCK") {
422 if ($this->grounded()) {
423 $this->lock();
424 } else {
425 emit("fail LOCK");
426 }
427 return;
428 }
429 if ($s["usedHold"]) {
430 emit("fail HOLD");
431 return;
432 }
433 $s["usedHold"] = true;
434 $s["stats"]["holds"]++;
435 $previous = $s["hold"];
436 $s["hold"] = $s["kind"];
437 $this->spawnPiece($previous ?? $this->takeNext());
438 }

449 if ($s["fall"] >= $interval) {

453 );

455 }

459 $this->lock();
460 }

463 private function lock(): void
464 {
465 global $states;
466 $s = &$this->s;
467 $kind = $s["kind"];
468 $position = $s["position"];
469 $oldCells = cells($kind, $position);
470 $routes = routes($s["board"], $kind, startPosition($kind), $position);
471 $shortest = $routes[0] ?? null;
472 $spin =
473 $kind === "T" && $s["last"] === "rotation"
474 ? tSpin($s["board"], $position, $s["test"])
475 : "";
476 $placed = placed($s["board"], $kind, $position);
477 $n = $placed["lines"];
478 $s["board"] = $placed["board"];
479 $this->history[] = $this->start;
480 $s["pieces"]++;
481 $s["stats"]["kinds"][$kind]++;
482 $s["stats"]["operations"] += $s["keys"];
483 $info = clearInfo($spin, $n);
484 $level = $s["level"];
485 $points = $info["points"] * $level;
486 $sent = $info["sent"];
487 $back = false;
488 if ($n > 0) {
489 $s["streak"]++;
490 if ($n === 4 || $spin !== "") {
491 $s["chain"]++;
492 $back = $s["chain"] >= 2;
493 } else {
494 $s["chain"] = 0;
495 }
496 } else {
497 $s["streak"] = 0;
498 }
499 if ($back) {
500 $points = intdiv($points * 3, 2);
501 $sent++;
502 }
503 $combo = $s["streak"] - 1;
504 $comboPoints = $combo >= 1 ? 50 * $combo * $level : 0;
505 if ($combo >= 0) {
506 $sent += [0, 1, 1, 2, 2, 3, 3, 4, 4, 4, 5][min($combo, 10)];
507 }
508 $perfect = $n > 0 && count(array_filter($s["board"], fn($row) => $row !== 0)) === 0;
509 $perfectPoints = $perfect
510 ? ($n === 4 && $back ? 3200 : [800, 1200, 1800, 2000][$n - 1]) * $level
511 : 0;
512 if ($perfect) {
513 $sent += 10;
514 $s["stats"]["perfect"]++;
515 }
516 $gained = $points + $comboPoints + $perfectPoints;
517 $s["score"] += $gained;
518 $s["lines"] += $n;
519 $s["stats"]["sent"] += $sent;
520 $s["stats"]["combo"] = max($s["stats"]["combo"], $combo);
521 $s["stats"]["chain"] = max($s["stats"]["chain"], $s["chain"]);
522 emit(
523 "lock " .
524 $s["pieces"] .
525 " " .
526 $kind .
527 " " .
528 $states[$position[0]] .
529 " " .
530 $position[1] .
531 " " .
532 $position[2] .
533 " keys " .
534 $s["keys"] .
535 " " .
536 ($shortest === null ? "-" : count($shortest["keys"])) .
537 " drop " .
538 $s["drop"] .
539 " lines " .
540 $n .
541 " sent " .
542 $sent .
543 " +" .
544 $gained .
545 " total " .
546 $s["score"],
547 );
548 if ($shortest !== null && $s["keys"] > count($shortest["keys"])) {
549 $s["stats"]["faults"]++;
550 emit(
551 "finesse " . (count($shortest["keys"]) ? implode(" ", $shortest["keys"]) : "none"),
552 );
553 }
554 if ($info["name"] !== "") {
555 $s["stats"]["clears"][$info["name"]]++;
556 emit(
557 "clear " .
558 $info["name"] .
559 " " .
560 $points .
561 ($back ? " b2b" : "") .
562 ($combo >= 1 ? " combo " . $combo . " " . $comboPoints : "") .
563 ($perfect ? " perfect " . $perfectPoints : ""),
564 );
565 }
566 $nextLevel = $s["initialLevel"] + intdiv($s["lines"], 10);
567 if ($nextLevel > $level) {
568 $s["level"] = $nextLevel;
569 emit("level " . $s["level"]);
570 }
571 $lockOut = true;
572 foreach ($oldCells as [$c, $r]) {
573 if ($r < 21) {
574 $lockOut = false;
575 break;
576 }
577 }
578 if ($lockOut) {
579 $this->over("lock out");
580 }
581 $cancelled = min($sent, array_sum($s["pending"]));
582 $remaining = $cancelled;
583 while ($remaining > 0) {
584 $count = min($remaining, $s["pending"][0]);
585 $remaining -= $count;
586 $s["pending"][0] -= $count;
587 if ($s["pending"][0] === 0) {
588 array_shift($s["pending"]);
589 }
590 }
591 if ($cancelled > 0) {
592 $s["stats"]["cancelled"] += $cancelled;
593 emit("cancel " . $cancelled . " pending " . array_sum($s["pending"]));
594 }
595 if ($n === 0) {
596 $risen = 0;
597 while (count($s["pending"]) > 0 && $risen < 8) {
598 $count = min($s["pending"][0], 8 - $risen);
599 $s["random"] = seedStep($s["random"]);
600 $hole = ($s["random"] % 10) + 1;
601 $this->insertGarbage($count, $hole);
602 $s["stats"]["received"] += $count;
603 $risen += $count;
604 $s["pending"][0] -= $count;
605 if ($s["pending"][0] === 0) {
606 array_shift($s["pending"]);
607 }
608 emit("rise " . $count . " hole " . $hole . " pending " . array_sum($s["pending"]));
609 }
610 }
611 $s["usedHold"] = false;
612 $this->spawnPiece($this->takeNext());
613 $this->start = $s;
614 }
615 private function candidates(string $kind, array $start, array $prefix): array
616 {
617 $s = &$this->s;
618 $result = [];
619 $seen = [];
620 foreach ([[], ["CW"], ["CW", "CW"], ["CCW"]] as $turns) {
621 $p = $start;
622 $valid = true;
623 foreach ($turns as $turn) {
624 $step = movePiece($s["board"], $kind, $p, $turn);
625 if ($step === null) {
626 $valid = false;
627 break;
628 }
629 $p = $step["p"];
630 }
631 if (!$valid) {
632 continue;
633 }
634 $add = function (array $q, array $shifts) use (
635 &$result,
636 &$seen,
637 $s,
638 $kind,
639 $prefix,
640 $turns,
641 ): void {
642 $landed = landing($s["board"], $kind, $q);
643 $indexes = [];
644 foreach (cells($kind, $landed) as [$c, $r]) {
645 $indexes[] = ($r - 1) * 10 + $c - 1;
646 }
647 sort($indexes);
648 $key = implode(",", $indexes);
649 if (isset($seen[$key])) {
650 return;
651 }
652 $seen[$key] = true;
653 $r = placed($s["board"], $kind, $landed);
654 $result[] = [
655 "value" => evaluateWell($r["board"], $r["lines"])["value"],
656 "kind" => $kind,
657 "keys" => array_merge($prefix, $turns, $shifts, ["HD"]),
658 ];
659 };
660 $add($p, []);
661 foreach (["L", "R"] as $direction) {
662 $q = $p;
663 $shifts = [];
664 while (true) {
665 $step = movePiece($s["board"], $kind, $q, $direction);
666 if ($step === null) {
667 break;
668 }
669 $q = $step["p"];
670 $shifts[] = $direction;
671 $add($q, $shifts);
672 }
673 }
674 }
675 return $result;
676 }
677 private function ranked(): array
678 {
679 $s = &$this->s;
680 $result = $this->candidates($s["kind"], $s["position"], []);
681 $held = $s["hold"] ?? $s["queue"][0];
682 if (!$s["usedHold"] && fits($s["board"], $held, startPosition($held))) {

690 }

693 return $result;

1014function validArgs(string $command, array $args): bool
1015{
1016 $number = fn($text, $low, $high) => preg_match('/^(0|[1-9][0-9]*)$/D', $text) &&
1017 ((int) $text) >= $low &&
1018 ((int) $text) <= $high;
1019 if ($command === "NEW") {
1020 return count($args) === 2 && $number($args[0], 1, 2147483646) && $number($args[1], 1, 15);
1021 }
1022 if ($command === "ROW") {
1023 return count($args) === 2 &&
1024 $number($args[0], 1, 22) &&
1025 preg_match('/^[#.]{10}$/D', $args[1]) &&
1026 str_contains($args[1], ".");
1027 }
1028 if ($command === "GARBAGE") {
1029 return count($args) === 2 && $number($args[0], 1, 10) && $number($args[1], 1, 10);
1030 }
1031 if ($command === "SAVE" || $command === "LOAD") {
1032 return count($args) === 1 && preg_match('/^[a-z]{1,8}$/D', $args[0]);
1033 }
1034 $limits = ["PREVIEW" => 14, "ATTACK" => 20, "TICK" => 1000, "HINT" => 10, "AUTO" => 50];
1035 if (isset($limits[$command])) {
1036 return count($args) === 1 && $number($args[0], 1, $limits[$command]);
1037 }
1038 return count($args) === 0;
1039}
