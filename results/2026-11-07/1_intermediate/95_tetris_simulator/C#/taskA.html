139 internal static Cell[][][] InitializeShapes() {
140 var result = new Cell[7][][];
141 for (int kind = 0; kind < 7; kind++) {
142 int size = Frames[kind].Length;
143 var blocks = new List<Cell>();
144 for (int row = 0; row < size; row++)
145 for (int col = 0; col < size; col++)
146 if (Frames[kind][row][col] == '#')
147 blocks.Add(new Cell(col, row));
148 result[kind] = new Cell[4][];
149 result[kind][0] = blocks.ToArray();
150 for (int rotation = 1; rotation < 4; rotation++) {
151 blocks = blocks.Select(cell => new Cell(size - 1 - cell.Y, cell.X)).ToList();
152 result[kind][rotation] = blocks.ToArray();
153 }
154 }
155 return result;
156 }

158 internal static long SeedStep(long value) => 48271 * value % 2147483647;
159 internal static Cell[] Cells(int kind, Position pos) =>

162 internal static (long, List<int>) Bag(long value) {
163 var pieces = Enumerable.Range(0, 7).ToList();
164 for (int i = 6; i >= 1; i--) {
165 value = SeedStep(value);
166 int j = (int)(value % (i + 1));
167 (pieces[i], pieces[j]) = (pieces[j], pieces[i]);
168 }
169 return (value, pieces);
170 }
171 internal static bool Fits(int[] board, int kind, Position pos) {
172 foreach (var cell in Cells(kind, pos)) {
173 if (cell.X < 1 || cell.X > 10 || cell.Y < 1 || cell.Y > 22)
174 return false;
175 if ((board[cell.Y - 1] & (1 << (cell.X - 1))) != 0)
176 return false;
177 }
178 return true;
179 }
180 internal static Movement? Move(int[] board, int kind, Position pos, int op) {
181 if (op == 6 || op == 7) {
182 int next = (pos.State + (op == 6 ? 1 : 3)) % 4;
183 int row = Array.FindIndex(KickStates, k => k.From == pos.State && k.To == next);
184 var offsets = kind == 1 ? new[] { (0, 0) } : (kind == 0 ? IKicks : OrdinaryKicks)[row];
185 for (int i = 0; i < offsets.Length; i++) {
186 var p = new Position(next, pos.X + offsets[i].Item1, pos.Y + offsets[i].Item2);
187 if (Fits(board, kind, p))
188 return new Movement(p, i + 1);
189 }
190 return null;
191 }
192 int dx = (op == 0 || op == 2) ? -1 : (op == 1 || op == 3) ? 1 : 0;
193 int dy = dx == 0 ? -1 : 0;
194 bool repeated = op == 2 || op == 3 || op == 5;
195 var current = pos;
196 bool changed = false;
197 do {
198 var p = new Position(pos.State, current.X + dx, current.Y + dy);
199 if (!Fits(board, kind, p))
200 break;
201 current = p;
202 changed = true;
203 } while (repeated);
204 return changed ? new Movement(current, 0) : null;
205 }
206 internal static Position Landing(int[] board, int kind,
207 Position pos) => Move(board, kind, pos, 5)?.Position ?? pos;
208 internal static (int[], int) Placed(int[] board, int kind, Position pos) {
209 var result = (int[])board.Clone();
210 foreach (var cell in Cells(kind, pos))
211 result[cell.Y - 1] |= 1 << (cell.X - 1);
212 var remaining = result.Where(row => row != 1023).ToList();
213 int count = 22 - remaining.Count;
214 remaining.AddRange(Enumerable.Repeat(0, count));
215 return (remaining.ToArray(), count);
216 }
217 internal static WellMetrics Evaluate(int[] board, int cleared) {
218 var heights = new int[10];
219 int holes = 0, bump = 0;
220 for (int col = 0; col < 10; col++) {
221 for (int row = 21; row >= 0; row--)
222 if ((board[row] & (1 << col)) != 0) {
223 heights[col] = row + 1;
224 break;
225 }
226 for (int row = 0; row < heights[col]; row++)
227 if ((board[row] & (1 << col)) == 0)
228 holes++;
229 }
230 for (int col = 0; col < 9; col++)
231 bump += Math.Abs(heights[col] - heights[col + 1]);
232 return new WellMetrics(heights, holes, bump,
233 76 * cleared - 51 * heights.Sum() - 36 * holes - 18 * bump);
234 }
235 internal static (List<Route>, int?)
236 Routes(int[] board, int kind, Position start, Position? target = null) {
237 if (!Fits(board, kind, start))
238 return (new List<Route>(), null);
239 var entries = new List<Route> { new(start, -1, -1) };
240 var seen = new HashSet<Position> { start };
241 for (int i = 0; i < entries.Count; i++) {
242 var entry = entries[i];
243 if (target.HasValue && Landing(board, kind, entry.Position) == target.Value)
244 return (entries, i);
245 for (int op = 0; op < 8; op++) {
246 var next = Move(board, kind, entry.Position, op);
247 if (next.HasValue && seen.Add(next.Value.Position))
248 entries.Add(new Route(next.Value.Position, i, op));
249 }
250 }
251 return (target.HasValue ? new List<Route>() : entries, null);
252 }
253 internal static List<int> Shortest(List<Route> nodes, int index) {
254 var result = new List<int>();
255 while (nodes[index].Parent >= 0) {
256 result.Add(nodes[index].Operation);
257 index = nodes[index].Parent;
258 }
259 result.Reverse();
260 return result;
261 }
262 internal static int TSpin(int[] board, Position p, int test) {
263 var corners = new[] { new Cell(p.X, p.Y), new Cell(p.X + 2, p.Y),
264 new Cell(p.X + 2, p.Y - 2), new Cell(p.X, p.Y - 2) };
265 var blocked = corners
266 .Select(c => c.X < 1 || c.X > 10 || c.Y < 1 || c.Y > 22 ||
267 (board[c.Y - 1] & (1 << (c.X - 1))) != 0)
268 .ToArray();
269 if (blocked.Count(x => x) < 3)
270 return 0;
271 return (blocked[p.State] && blocked[(p.State + 1) % 4]) || test == 5 ? 2 : 1;
272 }

283 internal static bool ValidNumber(string text, int min, int max) {
284 if (text.Length == 0 || (text.Length > 1 && text[0] == '0') ||
285 !text.All(ch => ch >= '0' && ch <= '9'))
286 return false;
287 return long.TryParse(text, out long value) && value >= min && value <= max;
288 }
289 internal static bool ValidArguments(string command, string[] args) {
290 if (command == "NEW")
291 return args.Length == 2 && ValidNumber(args[0], 1, 2147483646) &&
292 ValidNumber(args[1], 1, 15);
293 if (command == "ROW")
294 return args.Length == 2 && ValidNumber(args[0], 1, 22) && args[1].Length == 10 &&
295 args[1].All(ch => ch == '#' || ch == '.') && args[1].Contains('.');
296 if (command == "GARBAGE")
297 return args.Length == 2 && ValidNumber(args[0], 1, 10) && ValidNumber(args[1], 1, 10);
298 if (command == "SAVE" || command == "LOAD")
299 return args.Length == 1 && args[0].Length >= 1 && args[0].Length <= 8 &&
300 args[0].All(ch => ch >= 'a' && ch <= 'z');
301 int limit = command switch { "PREVIEW" => 14, "ATTACK" => 20, "TICK" => 1000,
302 "HINT" => 10, "AUTO" => 50,
303 _ => 0 };
304 return limit != 0 ? args.Length == 1 && ValidNumber(args[0], 1, limit) : args.Length == 0;
305 }
306 internal static void Main() {
307 int count = int.Parse(Console.ReadLine()!);
308 Game? game = null;
309 int serial = 0;
310 for (int i = 0; i < count; i++) {
311 var words = Console.ReadLine()!.Split(' ');
312 string command = words[0];
313 var args = words.Skip(1).ToArray();
314 try {
315 if (Commands.Contains(command)) {
316 if (!ValidArguments(command, args)) {
317 Console.WriteLine("ERROR bad arguments");
318 continue;
319 }
320 if (command == "NEW") {
321 if (game != null && !game.State.Ended) {
322 Console.WriteLine("game stopped");
323 game.Summary();
324 }
325 serial++;
326 Console.WriteLine(
327 $"game {serial} seed {int.Parse(args[0])} level {int.Parse(args[1])}");
328 game = new Game(serial, int.Parse(args[0]), int.Parse(args[1]));
329 } else if (game == null)
330 Console.WriteLine("ERROR no game");
331 else if (game.State.Ended && !AfterEnd.Contains(command))
332 Console.WriteLine("ERROR game over");
333 else
334 game.Command(command, args);
335 } else {
336 var invalid = words.FirstOrDefault(word => !Operations.Contains(word));
337 if (invalid != null)
338 Console.WriteLine($"ERROR unknown {invalid}");
339 else if (game == null)
340 Console.WriteLine("ERROR no game");
341 else if (game.State.Ended)
342 Console.WriteLine("ERROR game over");
343 else
344 foreach (var word in words)
345 game.Operation(Array.IndexOf(Operations, word));
346 }
347 } catch (GameOverException) {
348 }
349 }
350 if (game != null && !game.State.Ended) {
351 Console.WriteLine("game stopped");
352 game.Summary();
353 }
354 }

356internal sealed class Game {
357 private readonly int number;
358 public GameState State { get; private set; }
359 private List<GameState> history = new();
360 private GameState start;
361 private readonly Dictionary<string, SavedGame> saves = new();
362 public Game(int number, int seed, int level) {
363 this.number = number;
364 State = new GameState { Random = seed, InitialLevel = level, Level = level };
365 AddBag();
366 SpawnPiece(TakeNext());
367 start = State.Copy();
368 }
369 private void AddBag() {
370 var (next, pieces) = Tetris.Bag(State.Random);
371 State.Random = next;
372 State.Queue.AddRange(pieces);
373 }
374 private int TakeNext() {
375 int kind = State.Queue[0];
376 State.Queue.RemoveAt(0);
377 if (State.Queue.Count < 5)
378 AddBag();
379 return kind;
380 }
381 private void SpawnPiece(int kind) {
382 State.Kind = kind;
383 State.Position = Tetris.Spawn(kind);
384 State.Last = 0;
385 State.Test = 0;
386 State.Keys = 0;
387 State.Drop = 0;
388 State.Fall = 0;
389 State.Ground = 0;
390 State.Resets = 0;
391 State.Lowest = Tetris.Lowest(kind, State.Position);
392 if (!Tetris.Fits(State.Board, kind, State.Position))
393 GameOver("block out");
394 }
395 private void GameOver(string reason) {
396 Console.WriteLine($"game over {reason}");
397 Summary();
398 State.Ended = true;
399 throw new GameOverException();
400 }
401 private void Update(Position p, int last, int test = 0) {
402 State.Position = p;
403 State.Last = last;
404 State.Test = test;
405 int row = Tetris.Lowest(State.Kind, p);
406 if (row < State.Lowest) {
407 State.Lowest = row;
408 State.Resets = 0;
409 }
410 }
411 private bool Grounded() => Tetris.Move(State.Board, State.Kind, State.Position, 4) == null;
412 private void Drops(int points) {
413 State.Score += points;
414 State.Drop += points;
415 State.Stats.Drops += points;
416 }
417 private void InsertGarbage(int count, int hole) {
418 if (State.Board.Skip(22 - count).Any(row => row != 0))
419 GameOver("top out");
420 State.Board = Enumerable.Repeat(1023 ^ (1 << (hole - 1)), count)
421 .Concat(State.Board.Take(22 - count))
422 .ToArray();
423 }
424 public void Operation(int op) {
425 if (op < 8) {
426 State.Keys++;
427 var next = Tetris.Move(State.Board, State.Kind, State.Position, op);
428 if (!next.HasValue) {
429 Console.WriteLine($"fail {Tetris.Operations[op]}");
430 return;
431 }
432 if (op == 4 || op == 5) {
433 Drops(State.Position.Y - next.Value.Position.Y);
434 Update(next.Value.Position, 1);
435 State.Fall = 0;
436 State.Ground = 0;
437 } else {
438 Update(next.Value.Position, op >= 6 ? 2 : 1, next.Value.Test);
439 if (State.Ground > 0 && State.Resets < 15) {
440 State.Ground = 0;
441 State.Resets++;
442 }
443 }
444 } else if (op == 8) {
445 var p = Tetris.Landing(State.Board, State.Kind, State.Position);
446 int rows = State.Position.Y - p.Y;
447 Drops(2 * rows);
448 if (rows > 0)
449 Update(p, 1);
450 Lock();
451 } else if (op == 9) {
452 if (Grounded())
453 Lock();
454 else
455 Console.WriteLine("fail LOCK");
456 } else {
457 if (State.UsedHold) {
458 Console.WriteLine("fail HOLD");
459 return;
460 }
461 State.UsedHold = true;
462 State.Stats.Holds++;
463 int? previous = State.Hold;
464 State.Hold = State.Kind;
465 SpawnPiece(previous ?? TakeNext());
466 }
467 }
468 private void Tick() {
469 if (!Grounded()) {
470 State.Ground = 0;
471 State.Fall++;
472 int interval = State.Level >= 13 ? 1 : new[] { 60, 48, 37, 28, 21, 16,
473 11, 8, 6, 4, 3, 2 }[State.Level - 1];
474 if (State.Fall >= interval) {
475 Update(State.Position with { Y = State.Position.Y - 1 }, 1);
476 State.Fall = 0;
477 }
478 } else {
479 State.Ground++;
480 if (State.Ground >= 30)
481 Lock();
482 }
483 }
484 private void Lock() {
485 int kind = State.Kind;
486 var p = State.Position;
487 var oldCells = Tetris.Cells(kind, p);
488 var route = Tetris.Routes(State.Board, kind, Tetris.Spawn(kind), p);
489 var shortest =
490 route.Item2.HasValue ? Tetris.Shortest(route.Item1, route.Item2.Value) : null;
491 int spin = kind == 2 && State.Last == 2 ? Tetris.TSpin(State.Board, p, State.Test) : 0;
492 var placement = Tetris.Placed(State.Board, kind, p);
493 int n = placement.Item2;
494 State.Board = placement.Item1;
495 history.Add(start.Copy());
496 State.Pieces++;
497 State.Stats.Kinds[kind]++;
498 State.Stats.Operations += State.Keys;
499 var info = Tetris.Clearance(spin, n);
500 int level = State.Level;
501 int points = info.Points * level;
502 int sent = info.Sent;
503 bool back = false;
504 if (n > 0) {
505 State.Streak++;
506 if (n == 4 || spin != 0) {
507 State.Chain++;
508 back = State.Chain >= 2;
509 } else
510 State.Chain = 0;
511 } else
512 State.Streak = 0;
513 if (back) {
514 points = points * 3 / 2;
515 sent++;
516 }
517 int combo = State.Streak - 1;
518 int comboPoints = combo >= 1 ? 50 * combo * level : 0;
519 if (combo >= 0)
520 sent += new[] { 0, 1, 1, 2, 2, 3, 3, 4, 4, 4, 5 }[Math.Min(combo, 10)];
521 bool perfect = n > 0 && State.Board.All(row => row == 0);
522 int perfectPoints =
523 perfect ? (n == 4 && back ? 3200 : new[] { 800, 1200, 1800, 2000 }[n - 1]) * level : 0;
524 if (perfect) {
525 sent += 10;
526 State.Stats.Perfect++;
527 }
528 int gained = points + comboPoints + perfectPoints;
529 State.Score += gained;
530 State.Lines += n;
531 State.Stats.Sent += sent;
532 State.Stats.Combo = Math.Max(State.Stats.Combo, combo);
533 State.Stats.Chain = Math.Max(State.Stats.Chain, State.Chain);
534 Console.WriteLine(
535 $"lock {State.Pieces} {Tetris.Kinds[kind]} {Tetris.States[p.State]} {p.X} {p.Y} keys {State.Keys} {(shortest==null?"-":shortest.Count.ToString())} drop {State.Drop} lines {n} sent {sent} +{gained} total {State.Score}");
536 if (shortest != null && State.Keys > shortest.Count) {
537 State.Stats.Faults++;
538 Console.WriteLine("finesse " +
539 (shortest.Count == 0 ? "none" : Tetris.OperationText(shortest)));
540 }
541 if (info.Name >= 0) {
542 State.Stats.Clears[info.Name]++;
543 var line = new StringBuilder($"clear {Tetris.ClearNames[info.Name]} {points}");
544 if (back)
545 line.Append(" b2b");
546 if (combo >= 1)
547 line.Append($" combo {combo} {comboPoints}");
548 if (perfect)
549 line.Append($" perfect {perfectPoints}");
550 Console.WriteLine(line.ToString());
551 }
552 int nextLevel = State.InitialLevel + State.Lines / 10;
553 if (nextLevel > level) {
554 State.Level = nextLevel;
555 Console.WriteLine($"level {State.Level}");
556 }
557 if (oldCells.All(cell => cell.Y >= 21))
558 GameOver("lock out");
559 int cancelled = Math.Min(sent, State.Pending.Sum());
560 int remaining = cancelled;
561 while (remaining > 0) {
562 int used = Math.Min(remaining, State.Pending[0]);
563 remaining -= used;
564 State.Pending[0] -= used;
565 if (State.Pending[0] == 0)
566 State.Pending.RemoveAt(0);
567 }
568 if (cancelled > 0) {
569 State.Stats.Cancelled += cancelled;
570 Console.WriteLine($"cancel {cancelled} pending {State.Pending.Sum()}");
571 }
572 if (n == 0) {
573 int risen = 0;
574 while (State.Pending.Count > 0 && risen < 8) {
575 int count = Math.Min(State.Pending[0], 8 - risen);
576 State.Random = Tetris.SeedStep(State.Random);
577 int hole = (int)(State.Random % 10) + 1;
578 InsertGarbage(count, hole);
579 State.Stats.Received += count;
580 risen += count;
581 State.Pending[0] -= count;
582 if (State.Pending[0] == 0)
583 State.Pending.RemoveAt(0);
584 Console.WriteLine($"rise {count} hole {hole} pending {State.Pending.Sum()}");
585 }
586 }
587 State.UsedHold = false;
588 SpawnPiece(TakeNext());
589 start = State.Copy();
590 }
591 private List<Candidate> Candidates(int kind, Position start, List<int> prefix) {
592 var result = new List<Candidate>();
593 var seen = new HashSet<string>();
594 foreach (var turns in new[] { Array.Empty<int>(), new[] { 6 }, new[] { 6, 6 },
595 new[] { 7 } }) {
596 var p = start;
597 bool valid = true;
598 foreach (int turn in turns) {
599 var next = Tetris.Move(State.Board, kind, p, turn);
600 if (!next.HasValue) {
601 valid = false;
602 break;
603 }
604 p = next.Value.Position;
605 }
606 if (!valid)
607 continue;
608 void Add(Position q, List<int> shifts) {
609 var landed = Tetris.Landing(State.Board, kind, q);
610 var occupied = Tetris.Cells(kind, landed)
611 .Select(cell => (cell.Y - 1) * 10 + cell.X - 1)
612 .OrderBy(x => x)
613 .ToArray();
614 string id = string.Join(",", occupied);
615 if (!seen.Add(id))
616 return;
617 var placed = Tetris.Placed(State.Board, kind, landed);
618 int value = Tetris.Evaluate(placed.Item1, placed.Item2).Value;
619 var ops = prefix.Concat(turns).Concat(shifts).Append(8).ToList();
620 result.Add(new Candidate(kind, value, ops, id));
621 }
622 Add(p, new List<int>());
623 foreach (int direction in new[] { 0, 1 }) {
624 var q = p;
625 var shifts = new List<int>();
626 while (true) {
627 var next = Tetris.Move(State.Board, kind, q, direction);
628 if (!next.HasValue)
629 break;
630 q = next.Value.Position;
631 shifts.Add(direction);
632 Add(q, shifts);
633 }
634 }
635 }
636 return result;
637 }
638 private List<Candidate> Ranked() {
639 var all = Candidates(State.Kind, State.Position, new List<int>());
640 int held = State.Hold ?? State.Queue[0];
641 if (!State.UsedHold && Tetris.Fits(State.Board, held, Tetris.Spawn(held)))
642 all.AddRange(Candidates(held, Tetris.Spawn(held), new List<int> { 10 }));
643 for (int i = 0; i < all.Count; i++)
644 all[i] = all[i] with { Order = i };
645 return all.OrderByDescending(c => c.Value).ThenBy(c => c.Order).ToList();
646 }
647 private void Spins() {
648 if (State.Kind != 2) {
649 Console.WriteLine("spins 0");
650 return;
651 }
652 var found = new Dictionary<Position, int>();
653 foreach (var route in Tetris.Routes(State.Board, 2, State.Position).Item1) {
654 foreach (int turn in new[] { 6, 7 }) {
655 var next = Tetris.Move(State.Board, 2, route.Position, turn);
656 if (!next.HasValue || Tetris.Move(State.Board, 2, next.Value.Position, 4).HasValue)
657 continue;
658 int spin = Tetris.TSpin(State.Board, next.Value.Position, next.Value.Test);
659 if (spin > 0) {
660 found.TryGetValue(next.Value.Position, out int previous);
661 found[next.Value.Position] = Math.Max(previous, spin);
662 }
663 }
664 }
665 var rows = found.Select(item => new SpinCandidate(item.Key, item.Value))
666 .OrderBy(item => item.Position.Y)
667 .ThenBy(item => item.Position.X)
668 .ThenBy(item => item.Position.State)
669 .ToList();
670 foreach (var item in rows) {
671 int lines = Tetris.Placed(State.Board, 2, item.Position).Item2;
672 int kind = Tetris.Clearance(item.Spin, lines).Name;
673 Console.WriteLine(
674 $"spin {Tetris.States[item.Position.State]} {item.Position.X} {item.Position.Y} {Tetris.ClearNames[kind]}");
675 }
676 Console.WriteLine($"spins {rows.Count}");
677 }

701 public void Command(string name, string[] args) {
702 switch (name) {
703 case "PRINT": {
704 var current = State.Ended ? new HashSet<Cell>()
705 : Tetris.Cells(State.Kind, State.Position).ToHashSet();
706 var ghost = State.Ended ? new HashSet<Cell>()
707 : Tetris
708 .Cells(State.Kind, Tetris.Landing(State.Board, State.Kind,
709 State.Position))
710 .ToHashSet();
711 for (int row = 22; row >= 1; row--) {
712 var text = new StringBuilder($"{row,2} ");
713 for (int col = 1; col <= 10; col++) {
714 char character = (State.Board[row - 1] & (1 << (col - 1))) != 0 ? '#'
715 : current.Contains(new Cell(col, row)) ? '@'
716 : ghost.Contains(new Cell(col, row)) ? '+'
717 : '.';
718 text.Append(character);
719 }
720 Console.WriteLine(text.ToString());
721 }
722 break;
723 }
724 case "NEXT":
725 Console.WriteLine(
726 $"hold {(State.Hold.HasValue?Tetris.Kinds[State.Hold.Value]:'-')} next {new string(State.Queue.Take(5).Select(i=>Tetris.Kinds[i]).ToArray())}");
727 break;
728 case "PREVIEW": {
729 int count = int.Parse(args[0]);
730 var queue = new List<int>(State.Queue);
731 long random = State.Random;
732 while (queue.Count < count) {
733 var result = Tetris.Bag(random);
734 random = result.Item1;
735 queue.AddRange(result.Item2);
736 }
737 Console.WriteLine("preview " +
738 new string(queue.Take(count).Select(i => Tetris.Kinds[i]).ToArray()));
739 break;
740 }
741 case "STATUS": {
742 Console.WriteLine(
743 $"score {State.Score} level {State.Level} lines {State.Lines} pieces {State.Pieces}");
744 string piece =
745 State.Ended
746 ? "-"
747 : $"{Tetris.Kinds[State.Kind]} {Tetris.States[State.Position.State]} {State.Position.X} {State.Position.Y}";
748 Console.WriteLine(
749 $"piece {piece} streak {State.Streak} chain {State.Chain} pending {State.Pending.Sum()}");
750 break;
751 }
752 case "ANALYZE": {

756 $"holes {metrics.Holes} bumpiness {metrics.Bumpiness} value {metrics.Value}");
757 break;

759 case "ROW": {

787 break;
