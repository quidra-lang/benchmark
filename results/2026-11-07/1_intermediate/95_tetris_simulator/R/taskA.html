74move_piece <- function(board, kind, position, operation) {
75 if (operation %in% c("CW", "CCW")) {
76 next_state <- (position[1] + if (operation == "CW") 1L else 3L) %% 4L
77 key <- paste0(position[1], next_state)
78 offsets <- if (kind == "O") matrix(c(0L, 0L), ncol = 2L) else if (kind == "I") i_kicks[[key]] else kicks[[key]]
79 for (i in seq_len(nrow(offsets))) {
80 next_position <- c(next_state, position[2:3] + offsets[i, ])
81 if (fits(board, kind, next_position)) return(list(position = next_position, test = i))
82 }
83 return(NULL)
84 }
85 step <- switch(operation, L = c(0,-1,0), LL = c(0,-1,0),
86 R = c(0,1,0), RR = c(0,1,0), SD = c(0,0,-1), DD = c(0,0,-1))
87 next_position <- position + step
88 if (!fits(board, kind, next_position)) return(NULL)
89 if (operation %in% c("LL", "RR", "DD")) {
90 while (fits(board, kind, next_position + step)) next_position <- next_position + step
91 }
92 list(position = next_position, test = 0L)
93}

101well_metrics <- function(board, lines = 0) {
102 heights <- vapply(seq_len(10), function(col) {
103 filled <- which(board[, col])
104 if (length(filled)) max(filled) else 0L
105 }, integer(1))
106 holes <- sum(vapply(seq_len(10), function(col) {
107 h <- heights[col]
108 if (h) sum(!board[seq_len(h), col]) else 0L
109 }, integer(1)))
110 bumpiness <- sum(abs(diff(heights)))
111 list(heights = heights, holes = holes, bumpiness = bumpiness,
112 value = 76 * lines - 51 * sum(heights) - 36 * holes - 18 * bumpiness)
113}

183reachable <- function(board, kind, start) {
184 seen <- new.env(hash = TRUE, parent = emptyenv())
185 nodes <- list(list(position = start, previous = 0L, operation = ""))
186 assign(position_key(start), TRUE, envir = seen)
187 next_index <- 1L
188 while (next_index <= length(nodes)) {
189 position <- nodes[[next_index]]$position
190 for (operation in movements) {
191 result <- move_piece(board, kind, position, operation)
192 if (is.null(result)) next
193 key <- position_key(result$position)
194 if (exists(key, envir = seen, inherits = FALSE)) next
195 assign(key, TRUE, envir = seen)
196 nodes[[length(nodes) + 1L]] <- list(position = result$position, previous = next_index, operation = operation)
197 }
198 next_index <- next_index + 1L
199 }
200 nodes
201}
202shortest <- function(board, kind, target) {
203 start <- spawn_position(kind)
204 if (!fits(board, kind, start)) return(NULL)
205 nodes <- reachable(board, kind, start)
206 for (i in seq_along(nodes)) {
207 if (!all(landing(board, kind, nodes[[i]]$position) == target)) next
208 sequence <- character()
209 current <- i
210 while (nodes[[current]]$previous != 0L) {
211 sequence <- c(nodes[[current]]$operation, sequence)
212 current <- nodes[[current]]$previous
213 }
214 return(sequence)
215 }
216 NULL
217}
218spin_check <- function(board, kind, position, motion, test) {
219 if (kind != "T" || motion != "rotation") return(0L)
220 x <- position[2]; y <- position[3]
221 corners <- c(blocked(board,x,y), blocked(board,x+2,y),
222 blocked(board,x+2,y-2), blocked(board,x,y-2))
223 if (sum(corners) < 3L) return(0L)
224 front <- switch(as.character(position[1]), `0` = c(1L,2L), `1` = c(2L,3L),
225 `2` = c(3L,4L), `3` = c(4L,1L))
226 if (all(corners[front]) || test == 5L) 2L else 1L
227}

239lock_piece <- function() {
240 kind <- game$kind
241 position <- game$position
242 cells <- cells_at(kind, position)
243 sequence <- shortest(game$board, kind, position)
244 spin <- spin_check(game$board, kind, position, game$motion, game$test)
245 placed <- placed_board(game$board, kind, position)
246 game$board <<- placed$board
247 lines <- placed$lines
248 game$history[[length(game$history) + 1L]] <<- game$start_state
249 game$pieces <<- game$pieces + 1
250 game$kind_counts[[kind]] <<- game$kind_counts[[kind]] + 1
251 game$operation_total <<- game$operation_total + game$keys
252 type <- clear_type(spin, lines)
253 level <- game$level
254 points <- if (type == 0L) 0 else base_points[type] * level
255 sent <- if (type == 0L) 0 else send_lines[type]
256 b2b <- FALSE
257 if (lines > 0L) {
258 game$streak <<- game$streak + 1L
259 if (lines == 4L || spin != 0L) {
260 game$chain <<- game$chain + 1L
261 b2b <- game$chain >= 2L
262 } else game$chain <<- 0L
263 } else game$streak <<- 0L
264 if (b2b) {
265 points <- (points * 3) %/% 2
266 sent <- sent + 1
267 }
268 combo <- game$streak - 1L
269 combo_points <- if (combo >= 1L) 50 * combo * level else 0
270 if (combo >= 0L) sent <- sent + c(0,1,1,2,2,3,3,4,4,4,5)[min(combo, 10L) + 1L]
271 perfect <- lines > 0L && !any(game$board)
272 perfect_points <- 0
273 if (perfect) {
274 perfect_points <- (if (lines == 4L && b2b) 3200 else c(800,1200,1800,2000)[lines]) * level
275 sent <- sent + 10
276 game$perfect <<- game$perfect + 1
277 }
278 gained <- points + combo_points + perfect_points
279 game$score <<- game$score + gained
280 game$lines <<- game$lines + lines
281 game$sent <<- game$sent + sent
282 game$max_combo <<- max(game$max_combo, combo)
283 game$max_chain <<- max(game$max_chain, game$chain)
284 emit("lock", game$pieces, kind, state_names[position[1] + 1L], position[2], position[3],
285 "keys", game$keys, if (is.null(sequence)) "-" else length(sequence),
286 "drop", game$drop, "lines", lines, "sent", sent,
287 paste0("+", format(gained, scientific = FALSE, trim = TRUE)), "total", game$score)
288 if (!is.null(sequence) && game$keys > length(sequence)) {
289 game$faults <<- game$faults + 1
290 emit("finesse", if (length(sequence)) paste(sequence, collapse = " ") else "none")
291 }
292 if (type != 0L) {
293 game$clears[type] <<- game$clears[type] + 1
294 suffix <- character()
295 if (b2b) suffix <- c(suffix, "b2b")
296 if (combo >= 1L) suffix <- c(suffix, "combo", sprintf("%.0f", combo), sprintf("%.0f", combo_points))
297 if (perfect) suffix <- c(suffix, "perfect", sprintf("%.0f", perfect_points))
298 emit("clear", clear_names[type], points, suffix)
299 }
300 next_level <- game$starting_level + game$lines %/% 10
301 if (next_level > level) {
302 game$level <<- next_level
303 emit("level", next_level)
304 }
305 if (all(cells[, 2] >= 21L)) game_over("lock out")
306 cancelled <- min(sent, sum(game$pending))
307 if (cancelled > 0) {
308 remaining <- cancelled
309 while (remaining > 0) {
310 used <- min(remaining, game$pending[1])
311 game$pending[1] <<- game$pending[1] - used
312 remaining <- remaining - used
313 if (game$pending[1] == 0) game$pending <<- game$pending[-1]
314 }
315 game$cancelled <<- game$cancelled + cancelled
316 emit("cancel", cancelled, "pending", sum(game$pending))
317 }
318 if (lines == 0L) {
319 risen <- 0L
320 while (length(game$pending) && risen < 8L) {
321 count <- min(game$pending[1], 8L - risen)
322 game$random <<- random_next(game$random)
323 hole <- game$random %% 10 + 1L
324 insert_garbage(count, hole)
325 game$received <<- game$received + count
326 risen <- risen + count
327 game$pending[1] <<- game$pending[1] - count
328 if (game$pending[1] == 0) game$pending <<- game$pending[-1]
329 emit("rise", count, "hole", hole, "pending", sum(game$pending))
330 }
331 }
332 game$hold_used <<- FALSE
333 spawn(take_piece())
334 game$start_state <<- core_state()
335}

341operate <- function(operation) {
342 kind <- game$kind
343 position <- game$position
344 if (operation %in% movements) {
345 game$keys <<- game$keys + 1
346 result <- move_piece(game$board, kind, position, operation)
347 if (is.null(result)) { emit("fail", operation); return(invisible(NULL)) }
348 if (operation %in% c("SD", "DD")) {
349 add_drop(position[3] - result$position[3])
350 update_position(result$position, "move")
351 game$fall_counter <<- 0L
352 game$ground_counter <<- 0L
353 } else {
354 update_position(result$position, if (operation %in% c("CW", "CCW")) "rotation" else "move", result$test)
355 if (game$ground_counter > 0L && game$resets < 15L) {
356 game$ground_counter <<- 0L
357 game$resets <<- game$resets + 1L
358 }
359 }
360 } else if (operation == "HD") {
361 target <- landing(game$board, kind, position)
362 distance <- position[3] - target[3]
363 add_drop(2 * distance)
364 if (distance > 0) update_position(target, "move")
365 lock_piece()
366 } else if (operation == "LOCK") {
367 if (fits(game$board, kind, lower(position))) emit("fail LOCK") else lock_piece()
368 } else {
369 if (game$hold_used) { emit("fail HOLD"); return(invisible(NULL)) }
370 game$hold_used <<- TRUE
371 game$holds <<- game$holds + 1
372 held <- game$hold
373 game$hold <<- kind
374 spawn(if (held == "") take_piece() else held)
375 }
376 invisible(NULL)
377}

393candidate_list <- function(kind, start, prefix) {
394 result <- list()
395 seen <- new.env(hash = TRUE, parent = emptyenv())
396 rotations <- list(character(), "CW", c("CW", "CW"), "CCW")
397 for (rotation in rotations) {
398 position <- start
399 valid <- TRUE
400 for (operation in rotation) {
401 moved <- move_piece(game$board, kind, position, operation)
402 if (is.null(moved)) { valid <- FALSE; break }
403 position <- moved$position
404 }
405 if (!valid) next
406 starts <- list(list(position = position, sequence = character()))
407 for (direction in c("L", "R")) {
408 moved_position <- position
409 sequence <- character()
410 repeat {
411 moved <- move_piece(game$board, kind, moved_position, direction)
412 if (is.null(moved)) break
413 moved_position <- moved$position
414 sequence <- c(sequence, direction)
415 starts[[length(starts) + 1L]] <- list(position = moved_position, sequence = sequence)
416 }
417 }
418 for (entry in starts) {
419 target <- landing(game$board, kind, entry$position)
420 cells <- cells_at(kind, target)
421 key <- paste(sort((cells[, 2] - 1L) * 10L + cells[, 1]), collapse = ",")
422 if (exists(key, envir = seen, inherits = FALSE)) next
423 assign(key, TRUE, envir = seen)
424 placed <- placed_board(game$board, kind, target)
425 result[[length(result) + 1L]] <- list(
426 kind = kind, value = well_metrics(placed$board, placed$lines)$value,
427 sequence = c(prefix, rotation, entry$sequence, "HD"))
428 }
429 }
430 result
431}

511 n <- length(args)
512 if (command == "NEW") return(n == 2L && valid_integer(args[1],1,2147483646) && valid_integer(args[2],1,15))
513 if (command %in% c("PRINT", "NEXT", "STATUS", "ANALYZE", "UNDO", "SPINS")) return(n == 0L)
514 if (command %in% c("SAVE", "LOAD")) return(n == 1L && grepl("^[a-z]{1,8}$", args[1]))
515 if (command == "ROW") return(n == 2L && valid_integer(args[1],1,22) && grepl("^[#.]{10}$", args[2]) && grepl(".",args[2],fixed=TRUE))
516 if (command == "GARBAGE") return(n == 2L && valid_integer(args[1],1,10) && valid_integer(args[2],1,10))
517 maximum <- switch(command, PREVIEW=14, ATTACK=20, TICK=1000, HINT=10, AUTO=50)
518 n == 1L && valid_integer(args[1],1,maximum)
519}
520
521process_line <- function(text) {
522 words <- strsplit(text, " ", fixed = TRUE)[[1]]
523 command <- words[1]
524 args <- words[-1]
525 if (!command %in% commands) {
526 unknown <- words[!words %in% operations]
527 if (length(unknown)) { emit("ERROR unknown", unknown[1]); return(invisible(NULL)) }
528 if (is.null(game)) { emit("ERROR no game"); return(invisible(NULL)) }
529 if (game$ended) { emit("ERROR game over"); return(invisible(NULL)) }
530 for (operation in words) operate(operation)
531 return(invisible(NULL))
532 }
533 if (!validate_arguments(command, args)) { emit("ERROR bad arguments"); return(invisible(NULL)) }
534 if (command == "NEW") { new_game(as.numeric(args[1]), as.numeric(args[2])); return(invisible(NULL)) }
535 if (is.null(game)) { emit("ERROR no game"); return(invisible(NULL)) }
536 if (game$ended && !command %in% allowed_after_end) { emit("ERROR game over"); return(invisible(NULL)) }
537 if (command == "PRINT") print_board()
538 else if (command == "NEXT") emit("hold", if (game$hold == "") "-" else game$hold,
539 "next", paste(game$queue[1:5], collapse = ""))
540 else if (command == "PREVIEW") {
541 count <- as.numeric(args[1])
542 queue <- game$queue
543 random <- game$random
544 while (length(queue) < count) {
545 generated <- bag(random)
546 random <- generated$random
547 queue <- c(queue, generated$pieces)
548 }
549 emit("preview", paste(queue[seq_len(count)], collapse = ""))
550 } else if (command == "STATUS") {
551 emit("score", game$score, "level", game$level, "lines", game$lines, "pieces", game$pieces)
552 piece <- if (game$ended) "-" else paste(game$kind, state_names[game$position[1] + 1L],
553 game$position[2], game$position[3])
554 emit("piece", piece, "streak", game$streak, "chain", game$chain, "pending", sum(game$pending))
555 } else if (command == "ANALYZE") {
556 metrics <- well_metrics(game$board)
557 emit("heights", metrics$heights)
558 emit("holes", metrics$holes, "bumpiness", metrics$bumpiness, "value", metrics$value)
559 } else if (command == "ROW") {
560 row <- as.numeric(args[1])
561 marks <- strsplit(args[2], "", fixed = TRUE)[[1]] == "#"
562 cells <- cells_at(game$kind, game$position)
563 occupied <- cells[cells[, 2] == row, 1]
564 if (any(marks[occupied])) emit("ERROR overlap")
565 else {
566 game$board[row, ] <<- marks
567 emit("OK")
568 }
569 } else if (command == "GARBAGE") {
570 count <- as.numeric(args[1])
571 insert_garbage(count, as.numeric(args[2]))
572 original <- game$position
573 found <- FALSE
574 for (j in 0:count) {
575 position <- original + c(0,0,j)
576 if (fits(game$board, game$kind, position)) {
577 game$position <<- position
578 found <- TRUE
579 break
580 }
581 }
582 if (!found) game_over("top out")
583 emit("OK")
584 } else if (command == "ATTACK") {
585 game$pending <<- c(game$pending, as.numeric(args[1]))
586 emit("pending", sum(game$pending))
587 } else if (command == "TICK") {
588 for (i in seq_len(as.numeric(args[1]))) tick()
589 } else if (command == "UNDO") {
590 if (!length(game$history)) emit("ERROR nothing to undo")
591 else {
592 last <- length(game$history)
593 saved <- game$history[[last]]
594 previous <- game$history[-last]
595 restore_core(saved, previous)
596 emit("undo", game$pieces, game$kind)
597 }
598 } else if (command == "SAVE") {
599 saved <- game[setdiff(names(game), "saves")]
600 game$saves[[args[1]]] <<- saved
601 emit("OK")
602 } else if (command == "LOAD") {
603 saved <- game$saves[[args[1]]]
604 if (is.null(saved)) emit("ERROR unknown save")
605 else {
606 saves <- game$saves
607 game <<- c(saved, list(saves = saves))
608 emit("load", game$pieces, game$kind)
609 }
610 } else if (command == "SPINS") show_spins()
611 else if (command == "HINT") {
612 candidates <- ranked_candidates()
613 count <- min(as.numeric(args[1]), length(candidates))
614 for (i in seq_len(count)) {
615 candidate <- candidates[[i]]
616 emit("hint", i, candidate$kind, paste(candidate$sequence, collapse = " "), "value", candidate$value)
617 }
618 } else if (command == "AUTO") {
619 for (i in seq_len(as.numeric(args[1]))) {
620 candidate <- ranked_candidates()[[1]]
621 emit("auto", candidate$kind, paste(candidate$sequence, collapse = " "))
622 for (operation in candidate$sequence) operate(operation)
623 }
624 }
625 invisible(NULL)
626}
627
628input <- file("stdin", open = "r")
629header <- readLines(input, n = 1L, warn = FALSE)
630if (length(header)) {
631 lines <- readLines(input, n = as.integer(header), warn = FALSE)
632 for (line in lines) tryCatch(process_line(line), game_over = function(condition) invisible(NULL))
633 if (!is.null(game) && !game$ended) { emit("game stopped"); summary() }
634}
635close(input)
