140local function bag(x)
141 local pieces = clone(kinds)
142 for i = 7, 2, -1 do
143 x = seed_step(x)

149local function fits(board, kind, p)
150 for _, b in ipairs(cells(kind, p)) do
151 local c, r = b[1], b[2]
152 if c < 1 or c > 10 or r < 1 or r > 22 or board[r] & (1 << (c - 1)) ~= 0 then
153 return false
154 end
155 end
156 return true
157end
158local function move(board, kind, p, op)
159 local s, x, y = p[1], p[2], p[3]
160 if op == "CW" or op == "CCW" then
161 local b = (s + (op == "CW" and 1 or 3)) % 4
162 local offsets =
163 kind == "O" and {{0, 0}} or
164 (kind == "I" and i_kicks or ordinary_kicks)[tostring(s) .. tostring(b)]
165 for i, k in ipairs(offsets) do
166 local q = {b, x + k[1], y + k[2]}
167 if fits(board, kind, q) then
168 return q, i
169 end
170 end
171 return nil
172 end
173 local dx = (op == "L" or op == "LL") and -1 or (op == "R" or op == "RR") and 1 or 0
174 local dy = dx == 0 and -1 or 0
175 local repeat_move = op == "LL" or op == "RR" or op == "DD"
176 local position = p
177 local changed = false
178 repeat
179 local q = {s, position[2] + dx, position[3] + dy}
180 if not fits(board, kind, q) then
181 break
182 end
183 position = q
184 changed = true
185 until not repeat_move
186 if changed then
187 return position, 0
188 end
189 return nil
190end
191local function landing(board, kind, p)
192 return move(board, kind, p, "DD") or p
193end
194local function placed(board, kind, p)
195 local result = clone(board)
196 for _, b in ipairs(cells(kind, p)) do
197 result[b[2]] = result[b[2]] | (1 << (b[1] - 1))
198 end
199 local remaining = {}
200 for _, row in ipairs(result) do
201 if row ~= 1023 then
202 remaining[#remaining + 1] = row
203 end
204 end
205 local lines = 22 - #remaining
206 for i = 1, lines do
207 remaining[#remaining + 1] = 0
208 end
209 return remaining, lines
210end
211local function evaluate(board, cleared)
212 local heights = {}
213 local holes, bumpiness = 0, 0
214 for c = 1, 10 do
215 local height = 0
216 for r = 22, 1, -1 do
217 if board[r] & (1 << (c - 1)) ~= 0 then
218 height = r
219 break
220 end
221 end
222 heights[c] = height
223 for r = 1, height do
224 if board[r] & (1 << (c - 1)) == 0 then
225 holes = holes + 1
226 end
227 end
228 end
229 for c = 1, 9 do
230 bumpiness = bumpiness + math.abs(heights[c] - heights[c + 1])
231 end
232 return {
233 heights = heights,
234 holes = holes,
235 bumpiness = bumpiness,
236 value = 76 * cleared - 51 * sum(heights) - 36 * holes - 18 * bumpiness
237 }
238end
239local function routes(board, kind, start, target)
240 if not fits(board, kind, start) then
241 return {}
242 end
243 local queue = {{position = start, keys = {}}}
244 local seen = {[pkey(start)] = true}
245 local i = 1
246 while i <= #queue do
247 local entry = queue[i]
248 i = i + 1
249 if target and pkey(landing(board, kind, entry.position)) == pkey(target) then
250 return {entry}
251 end
252 for _, op in ipairs(movements) do
253 local p = move(board, kind, entry.position, op)
254 if p and not seen[pkey(p)] then
255 seen[pkey(p)] = true
256 local keys = clone(entry.keys)
257 keys[#keys + 1] = op
258 queue[#queue + 1] = {position = p, keys = keys}
259 end
260 end
261 end
262 if target then
263 return {}
264 end
265 return queue
266end
267local function tspin(board, p, test)
268 local s, x, y = p[1], p[2], p[3]
269 local corners = {{x, y}, {x + 2, y}, {x + 2, y - 2}, {x, y - 2}}
270 local blocked = {}
271 local total = 0
272 for i, b in ipairs(corners) do
273 local c, r = b[1], b[2]
274 blocked[i] = c < 1 or c > 10 or r < 1 or r > 22 or board[r] & (1 << (c - 1)) ~= 0
275 if blocked[i] then
276 total = total + 1
277 end
278 end
279 if total < 3 then
280 return ""
281 end
282 if blocked[s + 1] and blocked[(s + 1) % 4 + 1] or test == 5 then
283 return "full"
284 end
285 return "mini"
286end

304 lines + 1
305 ], ({0, 0, 1, 2, 4})[lines + 1]
306end

422function Game:insert_garbage(rows, hole)
423 local s = self.s
424 for r = 23 - rows, 22 do
425 if s.board[r] ~= 0 then
426 self:over("top out")
427 end
428 end
429 local result = {}
430 for i = 1, rows do
431 result[i] = 1023 ~ (1 << (hole - 1))
432 end
433 for i = 1, 22 - rows do
434 result[rows + i] = s.board[i]
435 end
436 s.board = result

438function Game:operation(op)
439 local s = self.s
440 if member(movements, op) then
441 s.keys = s.keys + 1
442 local p, test = move(s.board, s.kind, s.position, op)
443 if not p then
444 emit("fail " .. op)
445 return
446 end
447 if op == "SD" or op == "DD" then
448 self:drops(s.position[3] - p[3])
449 self:update(p, "move")
450 s.fall = 0
451 s.ground = 0
452 else
453 self:update(p, (op == "CW" or op == "CCW") and "rotation" or "move", test)
454 if s.ground > 0 and s.resets < 15 then
455 s.ground = 0
456 s.resets = s.resets + 1
457 end
458 end
459 elseif op == "HD" then
460 local p = landing(s.board, s.kind, s.position)
461 local rows = s.position[3] - p[3]
462 self:drops(2 * rows)
463 if rows > 0 then
464 self:update(p, "move")
465 end
466 self:lock()
467 elseif op == "LOCK" then
468 if self:grounded() then
469 self:lock()
470 else
471 emit("fail LOCK")
472 end
473 else
474 if s.used_hold then
475 emit("fail HOLD")
476 return
477 end
478 s.used_hold = true
479 s.stats.holds = s.stats.holds + 1
480 local prev = s.hold
481 s.hold = s.kind
482 self:spawn_piece(prev or self:next_piece())
483 end
484end
485function Game:tick()
486 local s = self.s
487 if not self:grounded() then
488 s.ground = 0
489 s.fall = s.fall + 1
490 local interval =
491 s.level >= 13 and 1 or ({60, 48, 37, 28, 21, 16, 11, 8, 6, 4, 3, 2})[s.level]
492 if s.fall >= interval then
493 self:update({s.position[1], s.position[2], s.position[3] - 1}, "move")
494 s.fall = 0
495 end
496 else
497 s.ground = s.ground + 1
498 if s.ground >= 30 then
499 self:lock()
500 end
501 end
502end
503function Game:lock()
504 local s = self.s
505 local kind, p = s.kind, s.position
506 local old_cells = cells(kind, p)
507 local shortlist = routes(s.board, kind, spawn(kind), p)
508 local shortest = shortlist[1]
509 local spin = kind == "T" and s.last == "rotation" and tspin(s.board, p, s.test) or ""
510 local n
511 s.board, n = placed(s.board, kind, p)
512 self.history[#self.history + 1] = clone(self.start)
513 s.pieces = s.pieces + 1
514 s.stats.kinds[kind] = s.stats.kinds[kind] + 1
515 s.stats.operations = s.stats.operations + s.keys
516 local name, points, sent = clear_info(spin, n)
517 local level = s.level
518 points = points * level
519 local back = false
520 if n > 0 then
521 s.streak = s.streak + 1
522 if n == 4 or spin ~= "" then
523 s.chain = s.chain + 1
524 back = s.chain >= 2
525 else
526 s.chain = 0
527 end
528 else
529 s.streak = 0
530 end
531 if back then
532 points = math.floor(points * 3 / 2)
533 sent = sent + 1
534 end
535 local combo = s.streak - 1
536 local combo_points = combo >= 1 and 50 * combo * level or 0
537 if combo >= 0 then
538 sent = sent + ({0, 1, 1, 2, 2, 3, 3, 4, 4, 4, 5})[math.min(combo, 10) + 1]
539 end
540 local perfect = n > 0
541 if perfect then
542 for _, row in ipairs(s.board) do
543 if row ~= 0 then
544 perfect = false
545 break
546 end
547 end
548 end
549 local perfect_points =
550 perfect and (n == 4 and back and 3200 or ({800, 1200, 1800, 2000})[n]) * level or 0
551 if perfect then
552 sent = sent + 10
553 s.stats.perfect = s.stats.perfect + 1
554 end
555 local gained = points + combo_points + perfect_points
556 s.score = s.score + gained
557 s.lines = s.lines + n
558 s.stats.sent = s.stats.sent + sent
559 s.stats.combo = math.max(s.stats.combo, combo)
560 s.stats.chain = math.max(s.stats.chain, s.chain)
561 emit(
562 string.format(
563 "lock %d %s %s %d %d keys %d %s drop %d lines %d sent %d +%d total %d",
564 s.pieces,
565 kind,
566 states[p[1] + 1],
567 p[2],
568 p[3],
569 s.keys,
570 shortest and tostring(#shortest.keys) or "-",
571 s.drop,
572 n,
573 sent,
574 gained,
575 s.score
576 )
577 )
578 if shortest and s.keys > #shortest.keys then
579 s.stats.faults = s.stats.faults + 1
580 emit("finesse " .. (#shortest.keys > 0 and table.concat(shortest.keys, " ") or "none"))
581 end
582 if name ~= "" then
583 s.stats.clears[name] = s.stats.clears[name] + 1
584 local line = "clear " .. name .. " " .. points
585 if back then
586 line = line .. " b2b"
587 end
588 if combo >= 1 then
589 line = line .. " combo " .. combo .. " " .. combo_points
590 end
591 if perfect then
592 line = line .. " perfect " .. perfect_points
593 end
594 emit(line)
595 end
596 local next_level = s.initial_level + math.floor(s.lines / 10)
597 if next_level > level then
598 s.level = next_level
599 emit("level " .. s.level)
600 end
601 local lock_out = true
602 for _, b in ipairs(old_cells) do
603 if b[2] < 21 then
604 lock_out = false
605 break
606 end
607 end
608 if lock_out then
609 self:over("lock out")
610 end
611 local cancelled = math.min(sent, sum(s.pending))
612 local remaining = cancelled
613 while remaining > 0 do
614 local n1 = math.min(remaining, s.pending[1])
615 remaining = remaining - n1
616 s.pending[1] = s.pending[1] - n1
617 if s.pending[1] == 0 then
618 table.remove(s.pending, 1)
619 end
620 end
621 if cancelled > 0 then
622 s.stats.cancelled = s.stats.cancelled + cancelled
623 emit("cancel " .. cancelled .. " pending " .. sum(s.pending))
624 end
625 if n == 0 then
626 local risen = 0
627 while #s.pending > 0 and risen < 8 do
628 local count = math.min(s.pending[1], 8 - risen)
629 s.random = seed_step(s.random)
630 local hole = s.random % 10 + 1
631 self:insert_garbage(count, hole)
632 s.stats.received = s.stats.received + count
633 risen = risen + count
634 s.pending[1] = s.pending[1] - count
635 if s.pending[1] == 0 then
636 table.remove(s.pending, 1)
637 end
638 emit("rise " .. count .. " hole " .. hole .. " pending " .. sum(s.pending))
639 end
640 end
641 s.used_hold = false
642 self:spawn_piece(self:next_piece())
643 self.start = clone(s)
644end
645function Game:candidates(kind, start, prefix)
646 local s = self.s
647 local result = {}
648 local seen = {}
649 for _, turns in ipairs({{}, {"CW"}, {"CW", "CW"}, {"CCW"}}) do
650 local p = start
651 local valid = true
652 for _, turn in ipairs(turns) do
653 local q = move(s.board, kind, p, turn)
654 if not q then
655 valid = false
656 break
657 end
658 p = q
659 end
660 if valid then
661 local function add(q, shifts)
662 local landed = landing(s.board, kind, q)
663 local indexes = {}
664 for _, b in ipairs(cells(kind, landed)) do
665 indexes[#indexes + 1] = (b[2] - 1) * 10 + b[1] - 1
666 end
667 table.sort(indexes)
668 local key = table.concat(indexes, ",")
669 if seen[key] then
670 return
671 end
672 seen[key] = true
673 local board, n = placed(s.board, kind, landed)
674 local keys = {}
675 for _, part in ipairs({prefix, turns, shifts, {"HD"}}) do
676 for _, name in ipairs(part) do
677 keys[#keys + 1] = name
678 end
679 end
680 result[#result + 1] = {value = evaluate(board, n).value, kind = kind, keys = keys}
681 end
682 add(p, {})
683 for _, direction in ipairs({"L", "R"}) do
684 local q = p
685 local shifts = {}
686 while true do
687 local nextp = move(s.board, kind, q, direction)
688 if not nextp then
689 break
690 end
691 q = nextp
692 shifts[#shifts + 1] = direction
693 add(q, shifts)
694 end
695 end
696 end
697 end
698 return result
699end
700function Game:ranked()
701 local s = self.s
702 local result = self:candidates(s.kind, s.position, {})
703 local held = s.hold or s.queue[1]
704 if not s.used_hold and fits(s.board, held, spawn(held)) then
705 for _, c in ipairs(self:candidates(held, spawn(held), {"HOLD"})) do
706 result[#result + 1] = c
707 end
708 end
709 for i, c in ipairs(result) do
710 c.order = i
711 end
712 table.sort(
713 result,
714 function(a, b)
715 if a.value == b.value then
716 return a.order < b.order
717 else
718 return a.value > b.value
719 end
720 end
721 )
722 return result
723end
724function Game:spins()
725 local s = self.s
726 if s.kind ~= "T" then
727 emit("spins 0")
728 return
729 end
730 local found = {}
731 for _, route in ipairs(routes(s.board, s.kind, s.position)) do
732 for _, turn in ipairs({"CW", "CCW"}) do
733 local p, test = move(s.board, "T", route.position, turn)
734 if p and not move(s.board, "T", p, "SD") then
735 local spin = tspin(s.board, p, test)
736 local key = pkey(p)
737 if spin ~= "" and (not found[key] or spin == "full") then
738 found[key] = {position = p, spin = spin}
739 end
740 end
741 end
742 end
743 local candidates = {}
744 for _, candidate in pairs(found) do
745 candidates[#candidates + 1] = candidate
746 end
747 table.sort(
748 candidates,
749 function(a, b)
750 local x, y = a.position, b.position
751 if x[3] ~= y[3] then
752 return x[3] < y[3]
753 end
754 if x[2] ~= y[2] then
755 return x[2] < y[2]
756 end
757 return x[1] < y[1]

830function Game:command(command, args)
831 local s = self.s
832 if command == "PRINT" then
833 local current, ghost = {}, {}
834 if not s.ended then
835 for _, b in ipairs(cells(s.kind, s.position)) do
836 current[b[2] * 10 + b[1]] = true
837 end
838 for _, b in ipairs(cells(s.kind, landing(s.board, s.kind, s.position))) do
839 ghost[b[2] * 10 + b[1]] = true
840 end
841 end
842 for r = 22, 1, -1 do
843 local line = string.format("%2d ", r)
844 for c = 1, 10 do
845 if s.board[r] & (1 << (c - 1)) ~= 0 then
846 line = line .. "#"
847 elseif current[r * 10 + c] then
848 line = line .. "@"
849 elseif ghost[r * 10 + c] then
850 line = line .. "+"
851 else
852 line = line .. "."
853 end
854 end
855 emit(line)
856 end
857 elseif command == "NEXT" then
858 local queue = {}
859 for i = 1, 5 do
860 queue[i] = s.queue[i]
861 end
862 emit("hold " .. (s.hold or "-") .. " next " .. table.concat(queue))
863 elseif command == "PREVIEW" then
864 local n = tonumber(args[1])
865 local queue = clone(s.queue)
866 local random = s.random
867 while #queue < n do
868 local pieces
869 random, pieces = bag(random)
870 for _, k in ipairs(pieces) do
871 queue[#queue + 1] = k
872 end
873 end
874 local list = {}
875 for i = 1, n do
876 list[i] = queue[i]
877 end
878 emit("preview " .. table.concat(list))
879 elseif command == "STATUS" then
880 emit(
881 "score " ..
882 s.score .. " level " .. s.level .. " lines " .. s.lines .. " pieces " .. s.pieces
883 )
884 local piece =
885 s.ended and "-" or
886 s.kind ..
887 " " .. states[s.position[1] + 1] .. " " .. s.position[2] .. " " .. s.position[3]
888 emit(
889 "piece " ..
890 piece ..
891 " streak " .. s.streak .. " chain " .. s.chain .. " pending " .. sum(s.pending)
892 )
893 elseif command == "ANALYZE" then
894 local e = evaluate(s.board, 0)
895 emit("heights " .. table.concat(e.heights, " "))
896 emit("holes " .. e.holes .. " bumpiness " .. e.bumpiness .. " value " .. e.value)
897 elseif command == "ROW" then
898 local r = tonumber(args[1])
899 local pattern = args[2]
900 for _, b in ipairs(cells(s.kind, s.position)) do
901 if b[2] == r and pattern:sub(b[1], b[1]) == "#" then
902 emit("ERROR overlap")
903 return
904 end
905 end
906 local row = 0
907 for i = 1, 10 do
908 if pattern:sub(i, i) == "#" then
909 row = row | (1 << (i - 1))
910 end
911 end
912 s.board[r] = row
913 emit("OK")
914 elseif command == "GARBAGE" then
915 local n, h = tonumber(args[1]), tonumber(args[2])
916 self:insert_garbage(n, h)
917 for j = 0, n do
918 local p = {s.position[1], s.position[2], s.position[3] + j}
919 if fits(s.board, s.kind, p) then
920 s.position = p
921 emit("OK")
922 return
923 end
924 end
925 self:over("top out")
926 elseif command == "ATTACK" then
927 s.pending[#s.pending + 1] = tonumber(args[1])
928 emit("pending " .. sum(s.pending))
929 elseif command == "TICK" then
930 for i = 1, tonumber(args[1]) do
931 self:tick()
932 end
933 elseif command == "UNDO" then
934 if #self.history == 0 then
935 emit("ERROR nothing to undo")
936 else
937 self.s = table.remove(self.history)
938 self.start = clone(self.s)
939 emit("undo " .. self.s.pieces .. " " .. self.s.kind)
940 end
941 elseif command == "SAVE" then
942 self.saves[args[1]] = {
943 state = clone(s),
944 history = clone(self.history),
945 start = clone(self.start)
946 }
947 emit("OK")
948 elseif command == "LOAD" then
949 local saved = self.saves[args[1]]
950 if not saved then
951 emit("ERROR unknown save")
952 else
953 self.s = clone(saved.state)
954 self.history = clone(saved.history)
955 self.start = clone(saved.start)
956 emit("load " .. self.s.pieces .. " " .. self.s.kind)
957 end
958 elseif command == "SPINS" then
959 self:spins()
960 elseif command == "HINT" then
961 local ranked = self:ranked()
962 for i = 1, math.min(tonumber(args[1]), #ranked) do
963 local c = ranked[i]
964 emit(
965 "hint " ..
966 i .. " " .. c.kind .. " " .. table.concat(c.keys, " ") .. " value " .. c.value
967 )
968 end
969 elseif command == "AUTO" then
970 for i = 1, tonumber(args[1]) do
971 local c = self:ranked()[1]
972 emit("auto " .. c.kind .. " " .. table.concat(c.keys, " "))
973 for _, op in ipairs(c.keys) do
974 self:operation(op)
975 end
976 end
977 end
978end
979local function valid_arguments(command, args)
980 local function valid_number(text, min, max)
981 if not text or not text:match("^%d+$") or (#text > 1 and text:sub(1, 1) == "0") then
982 return false
983 end
984 local n = tonumber(text)
985 return n >= min and n <= max
986 end
987 if command == "NEW" then
988 return #args == 2 and valid_number(args[1], 1, 2147483646) and valid_number(args[2], 1, 15)
989 end
990 if command == "ROW" then
991 return #args == 2 and valid_number(args[1], 1, 22) and #args[2] == 10 and
992 args[2]:match("^[#%.]+$") ~= nil and
993 args[2]:find("%.") ~= nil
994 end
995 if command == "GARBAGE" then
996 return #args == 2 and valid_number(args[1], 1, 10) and valid_number(args[2], 1, 10)
997 end
998 if command == "SAVE" or command == "LOAD" then
999 return #args == 1 and #args[1] >= 1 and #args[1] <= 8 and args[1]:match("^[a-z]+$") ~= nil
1000 end
1001 local limits = {PREVIEW = 14, ATTACK = 20, TICK = 1000, HINT = 10, AUTO = 50}
1002 if limits[command] then
1003 return #args == 1 and valid_number(args[1], 1, limits[command])
1004 end
1005 return #args == 0
1006end
