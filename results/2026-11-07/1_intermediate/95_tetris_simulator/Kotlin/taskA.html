183private fun bag(seed: Long): Pair<Long, List<Int>> {
184 var random = seed
185 val pieces = (0..6).toMutableList()
186 for (i in 6 downTo 1) {
187 random = seedStep(random)
188 val j = (random % (i + 1)).toInt()
189 val temp = pieces[i]
190 pieces[i] = pieces[j]
191 pieces[j] = temp
192 }
193 return random to pieces
194}

201private fun move(board: IntArray, kind: Int, p: Pos, op: Int): Movement? {
202 if (op == 6 || op == 7) {
203 val b = (p.s + if (op == 6) 1 else 3) % 4
204 val row = kickStates.indexOfFirst { it.first == p.s && it.second == b }
205 val offsets =
206 if (kind == 1) arrayOf(0 to 0) else (if (kind == 0) iKicks else ordinaryKicks)[row]
207 for ((index, d) in offsets.withIndex()) {
208 val q = Pos(b, p.x + d.first, p.y + d.second)
209 if (fits(board, kind, q)) return Movement(q, index + 1)
210 }
211 return null
212 }
213 val dx =
214 when (op) {
215 0,
216 2 -> -1
217 1,
218 3 -> 1
219 else -> 0
220 }
221 val dy = if (dx == 0) -1 else 0
222 val repeated = op == 2 || op == 3 || op == 5
223 var q = p
224 var changed = false
225 do {
226 val next = Pos(p.s, q.x + dx, q.y + dy)
227 if (!fits(board, kind, next)) break
228 q = next
229 changed = true
230 } while (repeated)
231 return if (changed) Movement(q, 0) else null
232}

236private fun placed(board: IntArray, kind: Int, p: Pos): Pair<IntArray, Int> {
237 val result = board.clone()
238 for ((x, y) in cells(kind, p)) result[y - 1] = result[y - 1] or (1 shl (x - 1))
239 val rest = result.filter { it != 1023 }
240 val cleared = 22 - rest.size
241 return (rest + List(cleared) { 0 }).toIntArray() to cleared
242}

244private fun evaluate(board: IntArray, n: Int): Metrics {
245 val heights = IntArray(10)
246 var holes = 0
247 for (c in 0..9) {
248 for (r in 21 downTo 0) if ((board[r] and (1 shl c)) != 0) {
249 heights[c] = r + 1
250 break
251 }
252 for (r in 0 until heights[c]) if ((board[r] and (1 shl c)) == 0) holes++
253 }
254 var bump = 0
255 for (c in 0..8) bump += abs(heights[c] - heights[c + 1])
256 return Metrics(heights, holes, bump, 76 * n - 51 * heights.sum() - 36 * holes - 18 * bump)
257}

259private fun routes(
260 board: IntArray,
261 kind: Int,
262 start: Pos,
263 target: Pos? = null,
264): Pair<List<Route>, Int?> {
265 if (!fits(board, kind, start)) return emptyList<Route>() to null
266 val entries = mutableListOf(Route(start, -1, -1))
267 val seen = hashSetOf(start)
268 var i = 0
269 while (i < entries.size) {
270 val entry = entries[i]
271 if (target != null && landing(board, kind, entry.pos) == target) return entries to i
272 for (op in 0..7) {
273 val result = move(board, kind, entry.pos, op) ?: continue
274 if (seen.add(result.pos)) entries.add(Route(result.pos, i, op))
275 }
276 i++
277 }
278 return (if (target == null) entries else emptyList<Route>()) to null
279}

281private fun shortest(nodes: List<Route>, index: Int): List<Int> {
282 val ops = mutableListOf<Int>()
283 var i = index
284 while (nodes[i].parent >= 0) {
285 ops.add(nodes[i].op)
286 i = nodes[i].parent
287 }
288 ops.reverse()
289 return ops

292private fun tSpin(board: IntArray, p: Pos, test: Int): Int {
293 val corners = listOf(p.x to p.y, (p.x + 2) to p.y, (p.x + 2) to (p.y - 2), p.x to (p.y - 2))
294 val blocked = corners.map { (x, y) ->
295 x < 1 || x > 10 || y < 1 || y > 22 || (board[y - 1] and (1 shl (x - 1))) != 0
296 }
297 if (blocked.count { it } < 3) return 0
298 return if ((blocked[p.s] && blocked[(p.s + 1) % 4]) || test == 5) 2 else 1
299}

390 fun operation(op: Int) {
391 if (op in 0..7) {
392 s.keys++
393 val next = move(s.board, s.kind, s.pos, op)
394 if (next == null) {
395 emit("fail ${operationNames[op]}")
396 return
397 }
398 if (op == 4 || op == 5) {
399 drops(s.pos.y - next.pos.y)
400 update(next.pos, 1)
401 s.fall = 0
402 s.ground = 0
403 } else {
404 update(next.pos, if (op >= 6) 2 else 1, next.test)
405 if (s.ground > 0 && s.resets < 15) {
406 s.ground = 0
407 s.resets++
408 }
409 }
410 } else if (op == 8) {
411 val landed = landing(s.board, s.kind, s.pos)
412 val rows = s.pos.y - landed.y
413 drops(2 * rows)
414 if (rows > 0) update(landed, 1)
415 lock()
416 } else if (op == 9) {
417 if (grounded()) lock() else emit("fail LOCK")
418 } else {
419 if (s.usedHold) {
420 emit("fail HOLD")
421 return
422 }
423 s.usedHold = true
424 s.stats.holds++
425 val previous = s.hold
426 s.hold = s.kind
427 spawnPiece(previous ?: takeNext())
428 }

431 private fun tick() {
432 if (!grounded()) {
433 s.ground = 0
434 s.fall++
435 val interval =
436 if (s.level >= 13) 1
437 else intArrayOf(60, 48, 37, 28, 21, 16, 11, 8, 6, 4, 3, 2)[s.level - 1]
438 if (s.fall >= interval) {
439 update(s.pos.copy(y = s.pos.y - 1), 1)
440 s.fall = 0
441 }
442 } else {
443 s.ground++
444 if (s.ground >= 30) lock()
445 }

448 private fun lock() {
449 val kind = s.kind
450 val position = s.pos
451 val oldCells = cells(kind, position)
452 val route = routes(s.board, kind, spawn(kind), position)
453 val shortestOps = route.second?.let { shortest(route.first, it) }
454 val spin = if (kind == 2 && s.last == 2) tSpin(s.board, position, s.test) else 0
455 val (board, n) = placed(s.board, kind, position)
456 s.board = board
457 history.add(start.snapshot())
458 s.pieces++
459 s.stats.kinds[kind]++
460 s.stats.operations += s.keys
461 val info = clearInfo(spin, n)
462 val level = s.level
463 var points = info.points * level
464 var sent = info.sent
465 var back = false
466 if (n > 0) {
467 s.streak++
468 if (n == 4 || spin != 0) {
469 s.chain++
470 back = s.chain >= 2
471 } else s.chain = 0
472 } else s.streak = 0
473 if (back) {
474 points = points * 3 / 2
475 sent++
476 }
477 val combo = s.streak - 1
478 val comboPoints = if (combo >= 1) 50 * combo * level else 0
479 if (combo >= 0) sent += intArrayOf(0, 1, 1, 2, 2, 3, 3, 4, 4, 4, 5)[min(combo, 10)]
480 val perfect = n > 0 && s.board.all { it == 0 }
481 val perfectPoints =
482 if (perfect)
483 (if (n == 4 && back) 3200 else intArrayOf(800, 1200, 1800, 2000)[n - 1]) * level
484 else 0
485 if (perfect) {
486 sent += 10
487 s.stats.perfect++
488 }
489 val gained = points + comboPoints + perfectPoints
490 s.score += gained
491 s.lines += n
492 s.stats.sent += sent
493 s.stats.combo = max(s.stats.combo, combo)
494 s.stats.chain = max(s.stats.chain, s.chain)
495 emit(
496 "lock ${s.pieces} ${kinds[kind]} ${states[position.s]} ${position.x} ${position.y} keys ${s.keys} ${shortestOps?.size?:"-"} drop ${s.drop} lines $n sent $sent +$gained total ${s.score}",
497 )
498 if (shortestOps != null && s.keys > shortestOps.size) {
499 s.stats.faults++
500 emit("finesse ${if (shortestOps.isEmpty()) "none" else operationsText(shortestOps)}")
501 }
502 if (info.name >= 0) {
503 s.stats.clears[info.name]++
504 val line = buildString {
505 append("clear ${clearNames[info.name]} $points")
506 if (back) append(" b2b")
507 if (combo >= 1) append(" combo $combo $comboPoints")
508 if (perfect) append(" perfect $perfectPoints")
509 }
510 emit(line)
511 }
512 val nextLevel = s.initialLevel + s.lines / 10
513 if (nextLevel > level) {
514 s.level = nextLevel
515 emit("level ${s.level}")
516 }
517 if (oldCells.all { it.second >= 21 }) gameOver("lock out")
518 val cancelled = min(sent, s.pending.sum())
519 var remaining = cancelled
520 while (remaining > 0) {
521 val used = min(remaining, s.pending[0])
522 remaining -= used
523 s.pending[0] -= used
524 if (s.pending[0] == 0) s.pending.removeAt(0)
525 }
526 if (cancelled > 0) {
527 s.stats.cancelled += cancelled
528 emit("cancel $cancelled pending ${s.pending.sum()}")
529 }
530 if (n == 0) {
531 var risen = 0
532 while (s.pending.isNotEmpty() && risen < 8) {
533 val count = min(s.pending[0], 8 - risen)
534 s.random = seedStep(s.random)
535 val hole = (s.random % 10).toInt() + 1
536 insertGarbage(count, hole)
537 s.stats.received += count
538 risen += count
539 s.pending[0] -= count
540 if (s.pending[0] == 0) s.pending.removeAt(0)
541 emit("rise $count hole $hole pending ${s.pending.sum()}")
542 }
543 }
544 s.usedHold = false
545 spawnPiece(takeNext())
546 start = s.snapshot()
547 }

549 private fun candidates(kind: Int, start: Pos, prefix: List<Int>): List<Candidate> {
550 val result = mutableListOf<Candidate>()
551 val seen = mutableSetOf<List<Int>>()
552 for (turns in listOf(emptyList(), listOf(6), listOf(6, 6), listOf(7))) {
553 var p = start
554 var valid = true
555 for (op in turns) {
556 val next = move(s.board, kind, p, op)
557 if (next == null) {
558 valid = false
559 break
560 }
561 p = next.pos
562 }
563 if (!valid) continue
564 fun add(q: Pos, shifts: List<Int>) {
565 val landed = landing(s.board, kind, q)
566 val occupied =
567 cells(kind, landed).map { (it.second - 1) * 10 + it.first - 1 }.sorted()
568 if (!seen.add(occupied)) return
569 val (board, lines) = placed(s.board, kind, landed)
570 result.add(
571 Candidate(
572 kind,
573 evaluate(board, lines).value,
574 prefix + turns + shifts + 8,
575 occupied,
576 ),
577 )
578 }
579 add(p, emptyList())
580 for (direction in 0..1) {
581 var q = p
582 val shifts = mutableListOf<Int>()
583 while (true) {
584 val next = move(s.board, kind, q, direction) ?: break
585 q = next.pos
586 shifts.add(direction)
587 add(q, shifts)
588 }
589 }
590 }
591 return result
592 }

594 private fun ranked(): List<Candidate> {
595 val result = candidates(s.kind, s.pos, emptyList()).toMutableList()
596 val held = s.hold ?: s.queue[0]
597 if (!s.usedHold && fits(s.board, held, spawn(held)))
598 result.addAll(candidates(held, spawn(held), listOf(10)))
599 for ((i, c) in result.withIndex()) c.order = i
600 return result.sortedWith(compareByDescending<Candidate> { it.value }.thenBy { it.order })
601 }

603 private fun spins() {
604 if (s.kind != 2) {
605 emit("spins 0")
606 return
607 }
608 val found = mutableMapOf<Pos, Int>()
609 for (entry in routes(s.board, 2, s.pos).first) for (op in 6..7) {
610 val next = move(s.board, 2, entry.pos, op) ?: continue
611 if (move(s.board, 2, next.pos, 4) != null) continue
612 val spin = tSpin(s.board, next.pos, next.test)
613 if (spin != 0) found[next.pos] = max(found[next.pos] ?: 0, spin)
614 }
615 val rows =
616 found
617 .map { SpinCandidate(it.key, it.value) }
618 .sortedWith(
619 compareBy<SpinCandidate> { it.pos.y }.thenBy { it.pos.x }.thenBy { it.pos.s },
620 )
621 for (c in rows) {
622 val lines = placed(s.board, 2, c.pos).second
623 val info = clearInfo(c.spin, lines)
624 emit("spin ${states[c.pos.s]} ${c.pos.x} ${c.pos.y} ${clearNames[info.name]}")
625 }
626 emit("spins ${rows.size}")
627 }

655 fun command(name: String, args: List<String>) {
656 when (name) {
657 "PRINT" -> {
658 val current = if (s.ended) emptySet() else cells(s.kind, s.pos).toSet()
659 val ghost =
660 if (s.ended) emptySet()
661 else cells(s.kind, landing(s.board, s.kind, s.pos)).toSet()
662 for (r in 22 downTo 1) {
663 val line = buildString {
664 append("%2d ".format(Locale.US, r))
665 for (c in 1..10) {
666 val p = c to r
667 append(
668 when {
669 (s.board[r - 1] and (1 shl (c - 1))) != 0 -> '#'
670 p in current -> '@'
671 p in ghost -> '+'
672 else -> '.'
673 },
674 )
675 }
676 }
677 emit(line)
678 }
679 }
680 "NEXT" ->
681 emit(
682 "hold ${s.hold?.let{kinds[it]}?:'-'} next " +
683 s.queue.take(5).joinToString("") { kinds[it].toString() },
684 )
685 "PREVIEW" -> {
686 val n = args[0].toInt()
687 val queue = s.queue.toMutableList()
688 var random = s.random
689 while (queue.size < n) {
690 val next = bag(random)
691 random = next.first
692 queue.addAll(next.second)
693 }
694 emit("preview " + queue.take(n).joinToString("") { kinds[it].toString() })
695 }
696 "STATUS" -> {
697 emit("score ${s.score} level ${s.level} lines ${s.lines} pieces ${s.pieces}")
698 val piece =
699 if (s.ended) "-"
700 else "${kinds[s.kind]} ${states[s.pos.s]} ${s.pos.x} ${s.pos.y}"
701 emit("piece $piece streak ${s.streak} chain ${s.chain} pending ${s.pending.sum()}")
702 }
703 "ANALYZE" -> {
704 val e = evaluate(s.board, 0)
705 emit("heights " + e.heights.joinToString(" "))
706 emit("holes ${e.holes} bumpiness ${e.bump} value ${e.value}")
707 }
708 "ROW" -> {
709 val row = args[0].toInt()
710 val pattern = args[1]
711 if (cells(s.kind, s.pos).any { it.second == row && pattern[it.first - 1] == '#' }) {
712 emit("ERROR overlap")
713 return
714 }
715 var mask = 0
716 for (i in 0..9) if (pattern[i] == '#') mask = mask or (1 shl i)
717 s.board[row - 1] = mask
718 emit("OK")
719 }
720 "GARBAGE" -> {
721 val count = args[0].toInt()
722 val hole = args[1].toInt()
723 insertGarbage(count, hole)
724 for (j in 0..count) {
725 val p = s.pos.copy(y = s.pos.y + j)
726 if (fits(s.board, s.kind, p)) {
727 s.pos = p
728 emit("OK")
729 return
730 }
731 }
732 gameOver("top out")
733 }
734 "ATTACK" -> {
735 s.pending.add(args[0].toInt())
736 emit("pending ${s.pending.sum()}")
737 }
738 "TICK" -> repeat(args[0].toInt()) { tick() }
739 "UNDO" -> {
740 if (history.isEmpty()) emit("ERROR nothing to undo")
741 else {
742 s = history.removeAt(history.lastIndex)
743 start = s.snapshot()
744 emit("undo ${s.pieces} ${kinds[s.kind]}")
745 }
746 }
747 "SAVE" -> {
748 saves[args[0]] = Save(s.snapshot(), history.map { it.snapshot() }, start.snapshot())
749 emit("OK")
750 }
751 "LOAD" -> {
752 val saved = saves[args[0]]
753 if (saved == null) emit("ERROR unknown save")
754 else {
755 s = saved.state.snapshot()
756 history = saved.history.map { it.snapshot() }.toMutableList()
757 start = saved.start.snapshot()
758 emit("load ${s.pieces} ${kinds[s.kind]}")
759 }
760 }
761 "SPINS" -> spins()
762 "HINT" -> {
763 for ((i, c) in ranked().take(args[0].toInt()).withIndex()) emit(
764 "hint ${i+1} ${kinds[c.kind]} ${operationsText(c.ops)} value ${c.value}",
765 )
766 }
767 "AUTO" -> {
768 repeat(args[0].toInt()) {
769 val c = ranked()[0]
770 emit("auto ${kinds[c.kind]} ${operationsText(c.ops)}")
771 for (op in c.ops) operation(op)
772 }
773 }
774 }
775 }
776}

778private fun validNumber(text: String, minValue: Int, maxValue: Int): Boolean {
779 if (text.isEmpty() || (text.length > 1 && text[0] == '0') || text.any { it !in '0'..'9' })
780 return false
781 val n = text.toLongOrNull() ?: return false
782 return n in minValue.toLong()..maxValue.toLong()
783}
784
785private fun validArgs(name: String, args: List<String>): Boolean {
786 if (name == "NEW")
787 return args.size == 2 && validNumber(args[0], 1, 2147483646) && validNumber(args[1], 1, 15)
788 if (name == "ROW")
789 return args.size == 2 &&
790 validNumber(args[0], 1, 22) &&
791 args[1].length == 10 &&
792 args[1].all { it == '#' || it == '.' } &&
793 '.' in args[1]
794 if (name == "GARBAGE")
795 return args.size == 2 && validNumber(args[0], 1, 10) && validNumber(args[1], 1, 10)
796 if (name == "SAVE" || name == "LOAD")
797 return args.size == 1 && args[0].length in 1..8 && args[0].all { it in 'a'..'z' }
798 val limits = mapOf("PREVIEW" to 14, "ATTACK" to 20, "TICK" to 1000, "HINT" to 10, "AUTO" to 50)
799 if (name in limits) return args.size == 1 && validNumber(args[0], 1, limits.getValue(name))
800 return args.isEmpty()
801}

803fun main() {
804 val reader = BufferedReader(InputStreamReader(System.`in`))
805 val count = reader.readLine().toInt()
806 var game: Game? = null
807 var serial = 0
808 repeat(count) {
809 val words = reader.readLine().split(' ')
810 val name = words[0]
811 val args = words.drop(1)
812 try {
813 if (name in commandNames) {
814 if (!validArgs(name, args)) emit("ERROR bad arguments")
815 else if (name == "NEW") {
816 if (game != null && !game!!.s.ended) {
817 emit("game stopped")
818 game!!.summary()
819 }
820 serial++
821 emit("game $serial seed ${args[0].toInt()} level ${args[1].toInt()}")
822 game = Game(serial, args[0].toInt(), args[1].toInt())
823 } else if (game == null) emit("ERROR no game")
824 else if (game!!.s.ended && name !in afterEnd) emit("ERROR game over")
825 else game!!.command(name, args)
826 } else {
827 val bad = words.firstOrNull { it !in operationNames }
828 if (bad != null) emit("ERROR unknown $bad")
829 else if (game == null) emit("ERROR no game")
830 else if (game!!.s.ended) emit("ERROR game over")
831 else for (word in words) game!!.operation(operationNames.indexOf(word))
832 }
833 } catch (_: GameOver) {}
834 }
835 if (game != null && !game!!.s.ended) {
836 emit("game stopped")
837 game!!.summary()
838 }
839 print(output.joinToString("\n", postfix = "\n"))
840}
