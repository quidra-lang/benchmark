147vector<pair<int, int>> frame(char kind) {
148 vector<string> lines;
149 switch (kind) {
150 case 'I': lines = {"....", "####", "....", "...."}; break;
151 case 'O': lines = {"##", "##"}; break;
152 case 'T': lines = {".#.", "###", "..."}; break;
153 case 'S': lines = {".##", "##.", "..."}; break;
154 case 'Z': lines = {"##.", ".##", "..."}; break;
155 case 'J': lines = {"#..", "###", "..."}; break;
156 case 'L': lines = {"..#", "###", "..."}; break;
157 }
158 vector<pair<int, int>> out;
159 for (int j = 0; j < static_cast<int>(lines.size()); ++j)
160 for (int i = 0; i < static_cast<int>(lines[j].size()); ++i)
161 if (lines[j][i] == '#') out.emplace_back(i, j);
162 return out;
163}

165vector<pair<int, int>> cells(char kind, Pos position) {
166 int side = kind == 'I' ? 4 : kind == 'O' ? 2 : 3;
167 auto blocks = frame(kind);
168 for (int state = 0; state < position.s; ++state)
169 for (auto& block : blocks) {
170 int i = block.first;
171 block.first = side - 1 - block.second;
172 block.second = i;
173 }
174 for (auto& block : blocks) {
175 block.first += position.x;
176 block.second = position.y - block.second;
177 }
178 return blocks;
179}

189bool fits(const array<int, 22>& board, char kind, Pos pos) {
190 for (auto [c, r] : cells(kind, pos)) {
191 if (c < 1 || c > 10 || r < 1 || r > 22) return false;
192 if ((board[r - 1] & (1 << (c - 1))) != 0) return false;
193 }
194 return true;
195}

221optional<MoveResult> moved(const array<int, 22>& board, char kind, Pos position, const string& action) {
222 if (action == "CW" || action == "CCW") {
223 int target = (position.s + (action == "CW" ? 1 : 3)) % 4;
224 const vector<pair<int, int>> own = {{0,0}};
225 const auto& offsets = kind == 'O' ? own : kicks(kind == 'I').at(to_string(position.s) + to_string(target));
226 for (size_t i = 0; i < offsets.size(); ++i) {
227 Pos candidate = {target, position.x + offsets[i].first, position.y + offsets[i].second};
228 if (fits(board, kind, candidate)) return MoveResult{candidate, static_cast<int>(i) + 1};
229 }
230 return nullopt;
231 }
232 int dx = action == "L" || action == "LL" ? -1 : action == "R" || action == "RR" ? 1 : 0;
233 int dy = dx == 0 ? -1 : 0;
234 bool repeat = action == "LL" || action == "RR" || action == "DD";
235 Pos current = position;
236 bool changed = false;
237 do {
238 Pos next = {current.s, current.x + dx, current.y + dy};
239 if (!fits(board, kind, next)) break;
240 current = next;
241 changed = true;
242 } while (repeat);
243 if (!changed) return nullopt;
244 return MoveResult{current, 0};
245}

252Placement placed(const array<int, 22>& board, char kind, Pos pos) {
253 Placement out;
254 out.board = board;
255 for (auto [c, r] : cells(kind, pos)) out.board[r - 1] |= 1 << (c - 1);
256 array<int, 22> filtered{};
257 int next = 0;
258 for (int row : out.board) if (row != 1023) filtered[next++] = row;
259 out.lines = 22 - next;
260 out.board = filtered;
261 return out;
262}

264Evaluation evaluate(const array<int, 22>& board, int cleared) {
265 Evaluation result;
266 for (int col = 0; col < 10; ++col) {
267 for (int r = 21; r >= 0; --r) if (board[r] & (1 << col)) {
268 result.heights[col] = r + 1;
269 break;
270 }
271 for (int r = 0; r < result.heights[col]; ++r)
272 if (!(board[r] & (1 << col))) result.holes++;
273 }
274 int sum = 0;
275 for (int col = 0; col < 10; ++col) {
276 sum += result.heights[col];
277 if (col < 9) result.bumpiness += abs(result.heights[col] - result.heights[col + 1]);
278 }
279 result.value = 76 * cleared - 51 * sum - 36 * result.holes - 18 * result.bumpiness;
280 return result;
281}

283vector<Route> routes(const array<int, 22>& board, char kind, Pos start, optional<Pos> target = nullopt) {
284 vector<Route> queue;
285 if (!fits(board, kind, start)) return queue;
286 set<Pos> visited = {start};
287 queue.push_back({start, {}});
288 for (size_t index = 0; index < queue.size(); ++index) {
289 const Route& entry = queue[index];
290 Pos position = entry.position;
291 if (target && landing(board, kind, position) == *target) return {entry};
292 vector<string> keys = entry.keys;
293 for (const string& action : MOVEMENTS) {
294 auto next = moved(board, kind, position, action);
295 if (next && visited.insert(next->position).second) {
296 auto path = keys;
297 path.push_back(action);
298 queue.push_back({next->position, move(path)});
299 }
300 }
301 }
302 return target ? vector<Route>() : queue;
303}

305string spinType(const array<int, 22>& board, Pos pos, int test) {
306 const int x = pos.x, y = pos.y;
307 const array<pair<int, int>, 4> corners = {{{x,y},{x+2,y},{x+2,y-2},{x,y-2}}};
308 array<bool, 4> occupied{};
309 int count = 0;
310 for (int i = 0; i < 4; ++i) {
311 auto [col, row] = corners[i];
312 occupied[i] = col < 1 || col > 10 || row < 1 || row > 22 || (board[row - 1] & (1 << (col - 1)));
313 if (occupied[i]) count++;
314 }
315 if (count < 3) return "";
316 return (occupied[pos.s] && occupied[(pos.s + 1) % 4]) || test == 5 ? "full" : "mini";
317}

319ClearInfo clearInfo(const string& spin, int lines) {
320 if (spin == "full") {
321 const array<int, 4> points = {400, 800, 1200, 1600};
322 const array<int, 4> sent = {0, 2, 4, 6};
323 return {CLEAR_NAMES[7 + lines], points[lines], sent[lines], 7 + lines};
324 }
325 if (spin == "mini") {
326 const array<int, 3> points = {100, 200, 400};
327 const array<int, 3> sent = {0, 0, 1};
328 return {CLEAR_NAMES[4 + lines], points[lines], sent[lines], 4 + lines};
329 }
330 if (lines == 0) return {};
331 const array<int, 5> points = {0, 100, 300, 500, 800};
332 const array<int, 5> sent = {0, 0, 1, 2, 4};
333 return {CLEAR_NAMES[lines - 1], points[lines], sent[lines], lines - 1};
334}

418
419 void operation(const string& action) {
420 auto& s = state;
421 if (find(MOVEMENTS.begin(), MOVEMENTS.end(), action) != MOVEMENTS.end()) {
422 s.keys++;
423 auto result = moved(s.board, s.kind, s.position, action);
424 if (!result) { emit("fail " + action); return; }
425 if (action == "SD" || action == "DD") {
426 drops(s.position.y - result->position.y);
427 updatePosition(result->position, 1);
428 s.fall = s.ground = 0;
429 } else {
430 bool rotation = action == "CW" || action == "CCW";
431 updatePosition(result->position, rotation ? 2 : 1, result->test);
432 if (s.ground > 0 && s.resets < 15) {
433 s.ground = 0;
434 s.resets++;
435 }
436 }
437 return;
438 }
439 if (action == "HD") {
440 Pos to = landing(s.board, s.kind, s.position);
441 int rows = s.position.y - to.y;
442 drops(2 * rows);
443 if (rows) updatePosition(to, 1);
444 lock();
445 } else if (action == "LOCK") {
446 if (grounded()) lock();
447 else emit("fail LOCK");
448 } else if (action == "HOLD") {
449 if (s.usedHold) { emit("fail HOLD"); return; }
450 s.usedHold = true;
451 s.stats.holds++;
452 char current = s.kind;
453 char next = s.hold == '-' ? takeNext() : s.hold;
454 s.hold = current;
455 spawnPiece(next);
456 }
457 }

459 void tick() {
460 auto& s = state;
461 if (!grounded()) {
462 s.ground = 0;
463 s.fall++;
464 int g = s.level >= 13 ? 1 : INTERVALS[s.level - 1];
465 if (s.fall >= g) {
466 updatePosition({s.position.s, s.position.x, s.position.y - 1}, 1);
467 s.fall = 0;
468 }
469 } else {
470 s.ground++;
471 if (s.ground >= 30) lock();
472 }
473 }
474
475 void lock() {
476 auto& s = state;
477 char kind = s.kind;
478 Pos pos = s.position;
479 auto oldCells = cells(kind, pos);
480 auto shortest = routes(s.board, kind, spawn(kind), pos);
481 string spin = kind == 'T' && s.last == 2 ? spinType(s.board, pos, s.test) : "";
482 auto place = placed(s.board, kind, pos);
483 s.board = place.board;
484 int lines = place.lines;
485 history.push_back(start);
486 s.pieces++;
487 s.stats.kinds[kindIndex(kind)]++;
488 s.stats.operations += s.keys;
489 auto info = clearInfo(spin, lines);
490 int level = s.level;
491 int points = info.points * level, sent = info.sent;
492 bool b2b = false;
493 if (lines) {
494 s.streak++;
495 if (lines == 4 || !spin.empty()) {
496 s.chain++;
497 b2b = s.chain >= 2;
498 } else s.chain = 0;
499 } else s.streak = 0;
500 if (b2b) { points = points * 3 / 2; sent++; }
501 int combo = s.streak - 1;
502 int comboPoints = combo >= 1 ? 50 * combo * level : 0;
503 const array<int, 11> comboSent = {0, 1, 1, 2, 2, 3, 3, 4, 4, 4, 5};
504 if (combo >= 0) sent += comboSent[min(combo, 10)];
505 bool perfect = lines > 0 && all_of(s.board.begin(), s.board.end(), [](int row) { return row == 0; });
506 int perfectPoints = 0;
507 if (perfect) {
508 const array<int, 4> full = {800, 1200, 1800, 2000};
509 perfectPoints = (lines == 4 && b2b ? 3200 : full[lines - 1]) * level;
510 sent += 10;
511 s.stats.perfect++;
512 }
513 int gained = points + comboPoints + perfectPoints;
514 s.score += gained;
515 s.lines += lines;
516 s.stats.sent += sent;
517 s.stats.combo = max(s.stats.combo, combo);
518 s.stats.chain = max(s.stats.chain, s.chain);
519 string length = shortest.empty() ? "-" : to_string(shortest[0].keys.size());
520 emit("lock " + to_string(s.pieces) + " " + kind + " " + STATES[pos.s] + " " + to_string(pos.x) + " " + to_string(pos.y) +
521 " keys " + to_string(s.keys) + " " + length + " drop " + to_string(s.drop) + " lines " + to_string(lines) +
522 " sent " + to_string(sent) + " +" + to_string(gained) + " total " + to_string(s.score));
523 if (!shortest.empty() && s.keys > static_cast<int>(shortest[0].keys.size())) {
524 s.stats.faults++;
525 string sequence = shortest[0].keys.empty() ? "none" : join(shortest[0].keys, " ");
526 emit("finesse " + sequence);
527 }
528 if (info.index >= 0) {
529 s.stats.clears[info.index]++;
530 string report = "clear " + info.name + " " + to_string(points);
531 if (b2b) report += " b2b";
532 if (combo >= 1) report += " combo " + to_string(combo) + " " + to_string(comboPoints);
533 if (perfect) report += " perfect " + to_string(perfectPoints);
534 emit(report);
535 }
536 int nextLevel = s.initialLevel + s.lines / 10;
537 if (nextLevel > level) {
538 s.level = nextLevel;
539 emit("level " + to_string(s.level));
540 }
541 if (all_of(oldCells.begin(), oldCells.end(), [](auto cell) { return cell.second >= 21; })) gameOver("lock out");
542 int cancelled = min(sent, sumPending(s.pending));
543 int remaining = cancelled;
544 while (remaining > 0) {
545 int count = min(remaining, s.pending.front());
546 s.pending.front() -= count;
547 remaining -= count;
548 if (!s.pending.front()) s.pending.pop_front();
549 }
550 if (cancelled) {
551 s.stats.cancelled += cancelled;
552 emit("cancel " + to_string(cancelled) + " pending " + to_string(sumPending(s.pending)));
553 }
554 if (lines == 0) {
555 int risen = 0;
556 while (!s.pending.empty() && risen < 8) {
557 int count = min(s.pending.front(), 8 - risen);
558 int hole = static_cast<int>(nextRandom(s.random) % 10) + 1;
559 garbage(count, hole);
560 s.stats.received += count;
561 risen += count;
562 s.pending.front() -= count;
563 if (!s.pending.front()) s.pending.pop_front();
564 emit("rise " + to_string(count) + " hole " + to_string(hole) + " pending " + to_string(sumPending(s.pending)));
565 }
566 }
567 s.usedHold = false;
568 spawnPiece(takeNext());
569 start = s;
570 }

572 vector<Candidate> candidates(char kind, Pos first, vector<string> prefix) const {
573 vector<Candidate> out;
574 set<array<int,4>> seen;
575 auto consider = [&](Pos at, const vector<string>& turns, const vector<string>& shifts) {
576 Pos down = landing(state.board, kind, at);
577 array<int,4> id{};
578 int i = 0;
579 for (auto [c,r] : cells(kind, down)) id[i++] = (r - 1) * 10 + c - 1;
580 sort(id.begin(), id.end());
581 if (!seen.insert(id).second) return;
582 auto placement = placed(state.board, kind, down);
583 vector<string> sequence = prefix;
584 sequence.insert(sequence.end(), turns.begin(), turns.end());
585 sequence.insert(sequence.end(), shifts.begin(), shifts.end());
586 sequence.push_back("HD");
587 out.push_back({evaluate(placement.board, placement.lines).value, kind, move(sequence)});
588 };
589 const vector<vector<string>> orientations = {{}, {"CW"}, {"CW", "CW"}, {"CCW"}};
590 for (const auto& turns : orientations) {
591 Pos pos = first;
592 bool valid = true;
593 for (const string& turn : turns) {
594 auto next = moved(state.board, kind, pos, turn);
595 if (!next) { valid = false; break; }
596 pos = next->position;
597 }
598 if (!valid) continue;
599 consider(pos, turns, {});
600 for (const string& direction : {string("L"), string("R")}) {
601 Pos shifted = pos;
602 vector<string> shifts;
603 while (true) {
604 auto next = moved(state.board, kind, shifted, direction);
605 if (!next) break;
606 shifted = next->position;
607 shifts.push_back(direction);
608 consider(shifted, turns, shifts);
609 }
610 }
611 }
612 return out;
613 }

615 vector<Candidate> ranked() const {
616 vector<Candidate> out = candidates(state.kind, state.position, {});
617 char held = state.hold == '-' ? state.queue.front() : state.hold;
618 if (!state.usedHold && fits(state.board, held, spawn(held))) {
619 auto more = candidates(held, spawn(held), {"HOLD"});
620 out.insert(out.end(), more.begin(), more.end());
621 }
622 stable_sort(out.begin(), out.end(), [](const Candidate& a, const Candidate& b) { return a.value > b.value; });
623 return out;
624 }

626 void spins() {
627 if (state.kind != 'T') { emit("spins 0"); return; }
628 map<Pos, string> found;
629 for (const Route& route : routes(state.board, 'T', state.position)) {
630 for (const string& turn : {string("CW"), string("CCW")}) {
631 auto next = moved(state.board, 'T', route.position, turn);
632 if (!next || moved(state.board, 'T', next->position, "SD")) continue;
633 string spin = spinType(state.board, next->position, next->test);
634 if (!spin.empty() && (found.find(next->position) == found.end() || spin == "full"))
635 found[next->position] = spin;
636 }
637 }
638 vector<pair<Pos,string>> result(found.begin(), found.end());
639 sort(result.begin(), result.end(), [](const auto& a, const auto& b) {
640 return tie(a.first.y, a.first.x, a.first.s) < tie(b.first.y, b.first.x, b.first.s);
641 });
642 for (const auto& [pos,spin] : result) {
643 int cleared = placed(state.board, 'T', pos).lines;
644 emit("spin " + STATES[pos.s] + " " + to_string(pos.x) + " " + to_string(pos.y) + " " + clearInfo(spin, cleared).name);
645 }
646 emit("spins " + to_string(result.size()));
647 }

664 void command(const string& name, const vector<string>& args) {
665 auto& s = state;
666 if (name == "PRINT") {
667 set<int> current, ghost;
668 if (!s.ended) {
669 for (auto [c,r] : cells(s.kind, s.position)) current.insert(r * 10 + c);
670 for (auto [c,r] : cells(s.kind, landing(s.board, s.kind, s.position))) ghost.insert(r * 10 + c);
671 }
672 for (int r = 22; r >= 1; --r) {
673 string text = (r < 10 ? " " : "") + to_string(r) + " ";
674 for (int c = 1; c <= 10; ++c) {
675 int id = r * 10 + c;
676 text += (s.board[r-1] & (1 << (c-1))) ? '#' : current.count(id) ? '@' : ghost.count(id) ? '+' : '.';
677 }
678 emit(text);
679 }
680 } else if (name == "NEXT") {
681 string queue;
682 for (int i = 0; i < 5; ++i) queue += s.queue[i];
683 emit("hold " + string(1,s.hold) + " next " + queue);
684 } else if (name == "PREVIEW") {
685 int count = stoi(args[0]);
686 auto copy = s.queue;
687 long long random = s.random;
688 while (static_cast<int>(copy.size()) < count) {
689 vector<char> pieces(KINDS.begin(), KINDS.end());
690 for (int i = 6; i >= 1; --i) {
691 int j = static_cast<int>(nextRandom(random) % (i+1));
692 swap(pieces[i], pieces[j]);
693 }
694 for (char kind : pieces) copy.push_back(kind);
695 }
696 string value;
697 for (int i = 0; i < count; ++i) value += copy[i];
698 emit("preview " + value);
699 } else if (name == "STATUS") {
700 emit("score " + to_string(s.score) + " level " + to_string(s.level) + " lines " + to_string(s.lines) + " pieces " + to_string(s.pieces));
701 string description = s.ended ? "-" : string(1,s.kind) + " " + STATES[s.position.s] + " " + to_string(s.position.x) + " " + to_string(s.position.y);
702 emit("piece " + description + " streak " + to_string(s.streak) + " chain " + to_string(s.chain) + " pending " + to_string(sumPending(s.pending)));
703 } else if (name == "ANALYZE") {
704 auto value = evaluate(s.board, 0);
705 string heights = "heights";
706 for (int n : value.heights) heights += " " + to_string(n);
707 emit(heights);
708 emit("holes " + to_string(value.holes) + " bumpiness " + to_string(value.bumpiness) + " value " + to_string(value.value));
709 } else if (name == "ROW") {
710 int r = stoi(args[0]);
711 const string& pattern = args[1];
712 for (auto [c,row] : cells(s.kind, s.position)) if (row == r && pattern[c - 1] == '#') {
713 emit("ERROR overlap");
714 return;
715 }
716 int mask = 0;
717 for (int i = 0; i < 10; ++i) if (pattern[i] == '#') mask |= 1 << i;
718 s.board[r-1] = mask;
719 emit("OK");
720 } else if (name == "GARBAGE") {
721 int n = stoi(args[0]), hole = stoi(args[1]);
722 garbage(n, hole);
723 for (int j = 0; j <= n; ++j) {
724 Pos shifted = {s.position.s,s.position.x,s.position.y+j};
725 if (fits(s.board,s.kind,shifted)) {
726 s.position = shifted;
727 emit("OK");
728 return;
729 }
730 }
731 gameOver("top out");
732 } else if (name == "ATTACK") {
733 s.pending.push_back(stoi(args[0]));
734 emit("pending " + to_string(sumPending(s.pending)));
735 } else if (name == "TICK") {
736 int n = stoi(args[0]);
737 for (int i = 0; i < n; ++i) tick();
738 } else if (name == "UNDO") {
739 if (history.empty()) emit("ERROR nothing to undo");
740 else {
741 s = history.back();
742 history.pop_back();
743 start = s;
744 emit("undo " + to_string(s.pieces) + " " + s.kind);
745 }
746 } else if (name == "SAVE") {
747 saves[args[0]] = {s, history, start};
748 emit("OK");
749 } else if (name == "LOAD") {
750 auto it = saves.find(args[0]);
751 if (it == saves.end()) emit("ERROR unknown save");
752 else {
753 Snapshot snap = it->second;
754 s = snap.state;
755 history = move(snap.history);
756 start = snap.start;
757 emit("load " + to_string(s.pieces) + " " + s.kind);
758 }
759 } else if (name == "SPINS") {
760 spins();
761 } else if (name == "HINT") {
762 auto choice = ranked();
763 int count = min(static_cast<int>(choice.size()), stoi(args[0]));
764 for (int i = 0; i < count; ++i) {
765 const auto& entry = choice[i];
766 emit("hint " + to_string(i+1) + " " + entry.kind + " " + join(entry.keys, " ") + " value " + to_string(entry.value));
767 }
768 } else if (name == "AUTO") {
769 int count = stoi(args[0]);
770 for (int i = 0; i < count; ++i) {
771 auto choice = ranked();
772 if (choice.empty()) break;
773 const auto best = choice[0];
774 emit("auto " + string(1,best.kind) + " " + join(best.keys," "));
775 for (const string& action : best.keys) operation(action);
776 }
777 }
778 }
779};

781bool number(const string& word, long long low, long long high) {
782 if (word.empty() || (word.size() > 1 && word[0] == '0')) return false;
783 long long value = 0;
784 for (char c : word) {
785 if (c < '0' || c > '9') return false;
786 value = value * 10 + c - '0';
787 if (value > high) return false;
788 }
789 return value >= low;
790}

792bool argumentsValid(const string& name, const vector<string>& args) {
793 if (name == "NEW") return args.size() == 2 && number(args[0],1,2147483646) && number(args[1],1,15);
794 if (name == "ROW") {
795 if (args.size() != 2 || !number(args[0],1,22) || args[1].size() != 10 || args[1].find('.') == string::npos) return false;
796 return all_of(args[1].begin(),args[1].end(),[](char c) { return c == '#' || c == '.'; });
797 }
798 if (name == "GARBAGE") return args.size() == 2 && number(args[0],1,10) && number(args[1],1,10);
799 if (name == "SAVE" || name == "LOAD") return args.size() == 1 && args[0].size() >= 1 && args[0].size() <= 8 && all_of(args[0].begin(),args[0].end(),[](char c) { return c >= 'a' && c <= 'z'; });
800 const map<string, int> limits = {{"PREVIEW",14},{"ATTACK",20},{"TICK",1000},{"HINT",10},{"AUTO",50}};
801 auto limit = limits.find(name);
802 if (limit != limits.end()) return args.size() == 1 && number(args[0],1,limit->second);
803 return args.empty();
804}

806int main() {
807 ios::sync_with_stdio(false);
808 cin.tie(nullptr);
809 const set<string> commands = {"NEW","PRINT","NEXT","PREVIEW","STATUS","ANALYZE","ROW","GARBAGE","ATTACK","TICK","UNDO","SAVE","LOAD","SPINS","HINT","AUTO"};
810 const set<string> afterEnd = {"PRINT","NEXT","PREVIEW","STATUS","ANALYZE","LOAD"};
811 set<string> operations(MOVEMENTS.begin(), MOVEMENTS.end());
812 operations.insert({"HD","LOCK","HOLD"});
813 int n = 0;
814 if (!(cin >> n)) return 0;
815 string line;
816 getline(cin,line);
817 unique_ptr<Game> game;
818 int serial = 0;
819 for (int i = 0; i < n && getline(cin,line); ++i) {
820 istringstream input(line);
821 vector<string> words;
822 string word;
823 while (input >> word) words.push_back(word);
824 if (words.empty()) continue;
825 string name = words.front();
826 vector<string> args(words.begin()+1,words.end());
827 try {
828 if (commands.count(name)) {
829 if (!argumentsValid(name,args)) cout << "ERROR bad arguments\n";
830 else if (name == "NEW") {
831 if (game && !game->state.ended) {
832 game->emit("game stopped");
833 game->summary();
834 }
835 int seed = stoi(args[0]), level = stoi(args[1]);
836 cout << "game " << ++serial << " seed " << seed << " level " << level << '\n';
837 game = make_unique<Game>(serial,seed,level);
838 } else if (!game) cout << "ERROR no game\n";
839 else if (game->state.ended && !afterEnd.count(name)) cout << "ERROR game over\n";
840 else game->command(name,args);
841 } else {
842 auto bad = find_if(words.begin(),words.end(),[&](const string& op) { return !operations.count(op); });
843 if (bad != words.end()) cout << "ERROR unknown " << *bad << '\n';
844 else if (!game) cout << "ERROR no game\n";
845 else if (game->state.ended) cout << "ERROR game over\n";
846 else for (const string& op : words) game->operation(op);
847 }
848 } catch (const EndGame&) {}
849 }
850 if (game && !game->state.ended) {
851 game->emit("game stopped");
852 game->summary();
853 }
