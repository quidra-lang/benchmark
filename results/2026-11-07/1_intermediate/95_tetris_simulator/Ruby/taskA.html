116def spawn(kind) = [0, kind == "O" ? 5 : 4, 22]
117def cells(kind, position)
118 state, x, y = position
119 SHAPES[kind][state].map { |i, j| [x + i, y - j] }

122def lowest(kind, position) = cells(kind, position).map(&:last).min
123def bag(random)
124 pieces = KINDS.dup
125 6.downto(1) do |i|
126 random = 48271 * random % 2147483647
127 j = random % (i + 1)
128 pieces[i], pieces[j] = pieces[j], pieces[i]
129 end
130
131 [random, pieces]
132end

134def fits?(board, kind, position)
135 cells(kind, position).all? { |c, r| c.between?(1, 10) && r.between?(1, 22) && (board[r - 1] & (1 << (c - 1))).zero? }
136end

138def move(board, kind, position, operation)
139 state, x, y = position
140 if %w[CW CCW].include?(operation)
141 target = (state + (operation == "CW" ? 1 : 3)) % 4
142 offsets = kind == "O" ? [[0, 0]] : (kind == "I" ? I_KICKS : KICKS)[[state, target]]
143 offsets.each_with_index do |(dx, dy), i|
144 result = [target, x + dx, y + dy]
145 return [result, i + 1] if fits?(board, kind, result)
146 end
147
148 return nil
149 end
150
151 dx = %w[L LL].include?(operation) ? -1 : %w[R RR].include?(operation) ? 1 : 0
152 dy = dx.zero? ? -1 : 0
153 repeated = %w[LL RR DD].include?(operation)
154 result = position
155 loop do
156 candidate = [state, result[1] + dx, result[2] + dy]
157 break unless fits?(board, kind, candidate)
158 result = candidate
159 break unless repeated
160 end
161
162 result == position ? nil : [result, 0]
163end

165def landing(board, kind, position) = move(board, kind, position, "DD")&.first || position
166def placed(board, kind, position)
167 result = board.dup
168 cells(kind, position).each { |c, r| result[r - 1] |= 1 << (c - 1) }
169 remaining = result.reject { |row| row == 1023 }
170 count = 22 - remaining.length
171 [remaining + Array.new(count, 0), count]
172end

174def evaluation(board, cleared)
175 heights = Array.new(10, 0)
176 holes = 0
177 10.times do |c|
178 row = 21.downto(0).find { |r| (board[r] & (1 << c)) != 0 }
179 heights[c] = row ? row + 1 : 0
180 heights[c].times { |r| holes += 1 if (board[r] & (1 << c)).zero? }
181 end
182
183 bumpiness = heights.each_cons(2).sum { |a, b| (a - b).abs }
184 [heights, holes, bumpiness, 76 * cleared - 51 * heights.sum - 36 * holes - 18 * bumpiness]
185end

187def routes(board, kind, start, target = nil)
188 return [] unless fits?(board, kind, start)
189 queue = [[start, []]]
190 seen = Set[start]
191 cursor = 0
192 while cursor < queue.length
193 position, keys = queue[cursor]
194 return [queue[cursor]] if target && landing(board, kind, position) == target
195 MOVEMENTS.each do |operation|
196 result = move(board, kind, position, operation)
197 next unless result && seen.add?(result[0])
198 queue << [result[0], keys + [operation]]
199 end
200
201 cursor += 1
202 end
203
204 target ? [] : queue
205end

207def t_spin(board, position, test)
208 state, x, y = position
209 blocked = [[x, y], [x + 2, y], [x + 2, y - 2], [x, y - 2]].map do |c, r|
210 !c.between?(1, 10) || !r.between?(1, 22) || (board[r - 1] & (1 << (c - 1))) != 0
211 end
212
213 return "" if blocked.count(true) < 3
214 (blocked[state] && blocked[(state + 1) % 4]) || test == 5 ? "full" : "mini"
215end

217def clear_info(spin, lines)
218 case spin
219 when "full"
220 [%w[TSPIN TSPIN_SINGLE TSPIN_DOUBLE TSPIN_TRIPLE][lines], [400, 800, 1200, 1600][lines], [0, 2, 4, 6][lines]]
221 when "mini"
222 [%w[TSPIN_MINI TSPIN_MINI_SINGLE TSPIN_MINI_DOUBLE][lines], [100, 200, 400][lines], [0, 0, 1][lines]]
223 else
224 [["", "SINGLE", "DOUBLE", "TRIPLE", "TETRIS"][lines], [0, 100, 300, 500, 800][lines], [0, 0, 1, 2, 4][lines]]
225 end

328 def insert_garbage(rows, hole)
329 s = @state
330 over("top out") if s.board.last(rows).any? { |row| row != 0 }
331 s.board = Array.new(rows, 1023 ^ (1 << (hole - 1))) + s.board.first(22 - rows)

334 def operation(operation)
335 s = @state
336 if MOVEMENTS.include?(operation)
337 s.keys += 1
338 result = move(s.board, s.kind, s.position, operation)
339 unless result
340 emit("fail " + operation)
341 return
342 end
343
344 position, test = result
345 if %w[SD DD].include?(operation)
346 drops(s.position[2] - position[2])
347 update(position, "move")
348 s.fall = s.ground = 0
349 else
350 update(position, %w[CW CCW].include?(operation) ? "rotation" : "move", test)
351 if s.ground > 0 && s.resets < 15
352 s.ground = 0
353 s.resets += 1
354 end
355 end
356 elsif operation == "HD"
357 position = landing(s.board, s.kind, s.position)
358 rows = s.position[2] - position[2]
359 drops(2 * rows)
360 update(position, "move") if rows > 0
361 lock
362 elsif operation == "LOCK"
363 grounded? ? lock : emit("fail LOCK")
364 else
365 if s.used_hold
366 emit("fail HOLD")
367 return
368 end
369
370 s.used_hold = true
371 s.stats.holds += 1
372 previous = s.hold
373 s.hold = s.kind
374 spawn_piece(previous || next_piece)
375 end
376 end

378 def tick
379 s = @state
380 if grounded?
381 s.ground += 1
382 lock if s.ground >= 30
383 else
384 s.ground = 0
385 s.fall += 1
386 interval = s.level >= 13 ? 1 : [60, 48, 37, 28, 21, 16, 11, 8, 6, 4, 3, 2][s.level - 1]
387 if s.fall >= interval
388 update([s.position[0], s.position[1], s.position[2] - 1], "move")
389 s.fall = 0
390 end
391 end
392 end

394 def lock
395 s = @state
396 kind, position = s.kind, s.position
397 old_cells = cells(kind, position)
398 shortest = routes(s.board, kind, spawn(kind), position).first
399 spin = kind == "T" && s.last == "rotation" ? t_spin(s.board, position, s.test) : ""
400 s.board, lines = placed(s.board, kind, position)
401 @history << copy(@start)
402 s.pieces += 1
403 s.stats.kinds[kind] += 1
404 s.stats.operations += s.keys
405 name, base, sent = clear_info(spin, lines)
406 level = s.level
407 points = base * level
408 back_to_back = false
409 if lines > 0
410 s.streak += 1
411 if lines == 4 || spin != ""
412 s.chain += 1
413 back_to_back = s.chain >= 2
414 else
415 s.chain = 0
416 end
417 else
418 s.streak = 0
419 end
420
421 if back_to_back
422 points = points * 3 / 2
423 sent += 1
424 end
425
426 combo = s.streak - 1
427 combo_points = combo >= 1 ? 50 * combo * level : 0
428 sent += [0, 1, 1, 2, 2, 3, 3, 4, 4, 4, 5][[combo, 10].min] if combo >= 0
429 perfect = lines > 0 && s.board.all?(&:zero?)
430 perfect_points = perfect ? (lines == 4 && back_to_back ? 3200 : [800, 1200, 1800, 2000][lines - 1]) * level : 0
431 if perfect
432 sent += 10
433 s.stats.perfect += 1
434 end
435
436 gained = points + combo_points + perfect_points
437 s.score += gained
438 s.lines += lines
439 s.stats.sent += sent
440 s.stats.combo = [s.stats.combo, combo].max
441 s.stats.chain = [s.stats.chain, s.chain].max
442 emit(
443 "lock #{s.pieces} #{kind} #{STATES[position[0]]} #{position[1]} #{position[2]} keys #{s.keys} #{shortest ? shortest[1].length : "-"} drop #{s.drop} lines #{lines} sent #{sent} +#{gained} total #{s.score}"
444 )
445 if shortest && s.keys > shortest[1].length
446 s.stats.faults += 1
447 emit("finesse " + (shortest[1].empty? ? "none" : shortest[1].join(" ")))
448 end
449
450 unless name.empty?
451 s.stats.clears[name] += 1
452 emit(
453 "clear #{name} #{points}" +
454 (back_to_back ? " b2b" : "") +
455 (combo >= 1 ? " combo #{combo} #{combo_points}" : "") +
456 (perfect ? " perfect #{perfect_points}" : "")
457 )
458 end
459
460 next_level = s.initial_level + s.lines / 10
461 if next_level > level
462 s.level = next_level
463 emit("level #{s.level}")
464 end
465
466 over("lock out") if old_cells.all? { |_, row| row >= 21 }
467 cancelled = [sent, s.pending.sum].min
468 remaining = cancelled
469 while remaining > 0
470 count = [remaining, s.pending[0]].min
471 remaining -= count
472 s.pending[0] -= count
473 s.pending.shift if s.pending[0].zero?
474 end
475
476 if cancelled > 0
477 s.stats.cancelled += cancelled
478 emit("cancel #{cancelled} pending #{s.pending.sum}")
479 end
480
481 if lines.zero?
482 risen = 0
483 while !s.pending.empty? && risen < 8
484 count = [s.pending[0], 8 - risen].min
485 s.random = 48271 * s.random % 2147483647
486 hole = s.random % 10 + 1
487 insert_garbage(count, hole)
488 s.stats.received += count
489 risen += count
490 s.pending[0] -= count
491 s.pending.shift if s.pending[0].zero?
492 emit("rise #{count} hole #{hole} pending #{s.pending.sum}")
493 end
494 end
495
496 s.used_hold = false
497 spawn_piece(next_piece)
498 @start = copy(s)
499 end

501 def candidates(kind, start, prefix)
502 result = []
503 seen = Set.new
504 board = @state.board
505 [[], %w[CW], %w[CW CW], %w[CCW]].each do |turns|
506 position = start
507 valid = true
508 turns.each do |turn|
509 step = move(board, kind, position, turn)
510 unless step
511 valid = false
512 break
513 end
514
515 position = step[0]
516 end
517
518 next unless valid
519 add = lambda do |p, shifts|
520 landed = landing(board, kind, p)
521 id = cells(kind, landed).map { |c, r| (r - 1) * 10 + c - 1 }.sort
522 next unless seen.add?(id)
523 placed_board, lines = placed(board, kind, landed)
524 result << Candidate.new(evaluation(placed_board, lines)[3], kind, prefix + turns + shifts + ["HD"])
525 end
526
527 add.call(position, [])
528 %w[L R].each do |direction|
529 p = position
530 shifts = []
531 while (step = move(board, kind, p, direction))
532 p = step[0]
533 shifts << direction
534 add.call(p, shifts)
535 end
536 end
537 end
538
539 result
540 end

550 def spins
551 s = @state
552 if s.kind != "T"
553 emit("spins 0")
554 return
555 end
556
557 found = {}
558 routes(s.board, "T", s.position).each do |position, _|
559 %w[CW CCW].each do |turn|
560 step = move(s.board, "T", position, turn)
561 next unless step && move(s.board, "T", step[0], "SD").nil?
562 spin = t_spin(s.board, step[0], step[1])
563 found[step[0]] = spin if spin != "" && (!found.key?(step[0]) || spin == "full")
564 end
565 end

567 found.sort_by { |p, _| [p[2], p[1], p[0]] }.each do |position, spin|
568 _, lines = placed(s.board, "T", position)
569 emit("spin #{STATES[position[0]]} #{position[1]} #{position[2]} #{clear_info(spin, lines)[0]}")
570 end
571
572 emit("spins #{found.length}")
573 end

575 def summary
576 s = @state
577 a = s.stats
578 c = a.clears
579 ratio = lambda { |x, y|
580 q = y.zero? ? 0 : 100 * x / y
581 "#{q / 100}.#{(q % 100).to_s.rjust(2, "0")}"
582 }
583 emit("summary game #{@number} score #{s.score} level #{s.level} lines #{s.lines} pieces #{s.pieces}")
584 emit(
585 "clears single #{c["SINGLE"]} double #{c["DOUBLE"]} triple #{c["TRIPLE"]} tetris #{c["TETRIS"]} perfect #{a.perfect}"
586 )
587 emit(
588 "tspins mini #{c["TSPIN_MINI"]} mini_single #{c["TSPIN_MINI_SINGLE"]} mini_double #{c["TSPIN_MINI_DOUBLE"]} zero #{c["TSPIN"]} single #{c["TSPIN_SINGLE"]} double #{c["TSPIN_DOUBLE"]} triple #{c["TSPIN_TRIPLE"]}"
589 )
590 emit("pieces " + KINDS.map { |k| "#{k} #{a.kinds[k]}" }.join(" "))
591 emit("best combo #{a.combo} chain #{a.chain} holds #{a.holds} drops #{a.drops} faults #{a.faults}")
592 emit("garbage sent #{a.sent} cancelled #{a.cancelled} received #{a.received}")
593 emit(
594 "rates app #{ratio.call(a.sent, s.pieces)} apl #{ratio.call(a.sent, s.lines)} kpp #{ratio.call(a.operations, s.pieces)}"
595 )
596 end

598 def command(command, args)
599 s = @state
600 case command
601 when "PRINT"
602 current = s.ended ? Set.new : cells(s.kind, s.position).to_set
603 ghost = s.ended ? Set.new : cells(s.kind, landing(s.board, s.kind, s.position)).to_set
604 22.downto(1) do |r|
605 row = r.to_s.rjust(2) + " "
606 1.upto(10) { |c|
607 row += (s.board[r - 1] & (1 << (c - 1))) != 0 ? "#" : current.include?([c, r]) ? "@" : ghost.include?([c, r]) ? "+" : "."
608 }
609 emit(row)
610 end
611
612 when "NEXT"
613 emit("hold #{s.hold || "-"} next #{s.queue.first(5).join}")
614 when "PREVIEW"
615 n = args[0].to_i
616 queue = s.queue.dup
617 random = s.random
618 while queue.length < n
619 random, pieces = bag(random)
620 queue.concat(pieces)
621 end
622
623 emit("preview " + queue.first(n).join)
624 when "STATUS"
625 emit("score #{s.score} level #{s.level} lines #{s.lines} pieces #{s.pieces}")
626 piece = s.ended ? "-" : "#{s.kind} #{STATES[s.position[0]]} #{s.position[1]} #{s.position[2]}"
627 emit("piece #{piece} streak #{s.streak} chain #{s.chain} pending #{s.pending.sum}")
628 when "ANALYZE"
629 heights, holes, bumpiness, value = evaluation(s.board, 0)
630 emit("heights " + heights.join(" "))
631 emit("holes #{holes} bumpiness #{bumpiness} value #{value}")
632 when "ROW"
633 r = args[0].to_i
634 pattern = args[1]
635 if cells(s.kind, s.position).any? { |c, row| row == r && pattern[c - 1] == "#" }
636 emit("ERROR overlap")
637 else
638 s.board[r - 1] = pattern.chars.each_with_index.sum { |ch, i| ch == "#" ? 1 << i : 0 }
639 emit("OK")
640 end
641
642 when "GARBAGE"
643 n, hole = args.map(&:to_i)
644 insert_garbage(n, hole)
645 0.upto(n) do |j|
646 position = [s.position[0], s.position[1], s.position[2] + j]
647 if fits?(s.board, s.kind, position)
648 s.position = position
649 emit("OK")
650 return
651 end
652 end
653
654 over("top out")
655 when "ATTACK"
656 s.pending << args[0].to_i
657 emit("pending #{s.pending.sum}")
658 when "TICK"
659 args[0].to_i.times { tick }
660 when "UNDO"
661 if @history.empty?
662 emit("ERROR nothing to undo")
663 else
664 @state = copy(@history.pop)
665 @start = copy(@state)
666 emit("undo #{@state.pieces} #{@state.kind}")
667 end
668
669 when "SAVE"
670 @saves[args[0]] = copy([s, @history, @start])
671 emit("OK")
672 when "LOAD"
673 if @saves.key?(args[0])
674 @state, @history, @start = copy(@saves[args[0]])
675 emit("load #{@state.pieces} #{@state.kind}")
676 else
677 emit("ERROR unknown save")
678 end
679
680 when "SPINS"
681 spins
682 when "HINT"
683 ranked.first(args[0].to_i).each_with_index { |c, i|
684 emit("hint #{i + 1} #{c.kind} #{c.keys.join(" ")} value #{c.value}")
685 }
686 when "AUTO"
687 args[0].to_i.times do
688 best = ranked.first
689 emit("auto #{best.kind} #{best.keys.join(" ")}")
690 best.keys.each { |op| operation(op) }
691 end
692 end
693 end
694end

696def valid_arguments?(command, args)
697 number = -> (text, low, high) { /\A(?:0|[1-9][0-9]*)\z/.match?(text) && text.to_i.between?(low, high) }
698 case command
699 when "NEW"
700 args.length == 2 && number.call(args[0], 1, 2147483646) && number.call(args[1], 1, 15)
701 when "ROW"
702 args.length == 2 && number.call(args[0], 1, 22) && /\A[#.]{10}\z/.match?(args[1]) && args[1].include?(".")
703 when "GARBAGE"
704 args.length == 2 && number.call(args[0], 1, 10) && number.call(args[1], 1, 10)
705 when "SAVE", "LOAD"
706 args.length == 1 && /\A[a-z]{1,8}\z/.match?(args[0])
707 when "PREVIEW", "ATTACK", "TICK", "HINT", "AUTO"
708 args.length == 1 &&
709 number.call(args[0], 1, {"PREVIEW" => 14, "ATTACK" => 20, "TICK" => 1000, "HINT" => 10, "AUTO" => 50}[command])
710 else
711 args.empty?
712 end
713end
714
715lines = $stdin.each_line.map(&:chomp)
716game = nil
717serial = 0
718lines.drop(1).first(lines[0].to_i).each do |line|
719 command, *args = line.split(" ")
720 begin
721 if COMMANDS.include?(command)
722 if !valid_arguments?(command, args)
723 emit("ERROR bad arguments")
724 elsif command == "NEW"
725 if game && !game.state.ended
726 emit("game stopped")
727 game.summary
728 end
729
730 serial += 1
731 emit("game #{serial} seed #{args[0].to_i} level #{args[1].to_i}")
732 game = Game.new(serial, args[0].to_i, args[1].to_i)
733 elsif !game
734 emit("ERROR no game")
735 elsif game.state.ended && !AFTER_END.include?(command)
736 emit("ERROR game over")
737 else
738 game.command(command, args)
739 end
740 else
741 invalid = [command, *args].find { |word| !OPERATIONS.include?(word) }
742 if invalid
743 emit("ERROR unknown " + invalid)
744 elsif !game
745 emit("ERROR no game")
746 elsif game.state.ended
747 emit("ERROR game over")
748 else
749 [command, *args].each { |op| game.operation(op) }
750 end
751 end
752
753 rescue GameOver
754 end
755end
756
757if game && !game.state.ended
758 emit("game stopped")
759 game.summary
760end
