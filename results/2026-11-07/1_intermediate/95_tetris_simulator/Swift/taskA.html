185func bag(_ initial: Int) -> (Int, [Int]) {
186 var random = initial
187 var pieces = Array(0..<7)
188 for i in stride(from: 6, through: 1, by: -1) {
189 random = seedStep(random)
190 pieces.swapAt(i, random % (i + 1))
191 }
192 return (random, pieces)

194func fits(_ board: [Int], _ kind: Int, _ p: Position) -> Bool {
195 for cell in cells(kind, p) {
196 if cell.x < 1 || cell.x > 10 || cell.y < 1 || cell.y > 22 {
197 return false
198 }
199 if (board[cell.y - 1] & (1 << (cell.x - 1))) != 0 {
200 return false
201 }
202 }
203 return true
204}
205func move(_ board: [Int], _ kind: Int, _ p: Position, _ op: Int) -> Movement? {
206 if op == 6 || op == 7 {
207 let b = (p.s + (op == 6 ? 1 : 3)) % 4
208 let row = kickStates.firstIndex { $0[0] == p.s && $0[1] == b }!
209 let kicks = kind == 1 ? [[0, 0]] : (kind == 0 ? iKicks : ordinaryKicks)[row]
210 for (index, offset) in kicks.enumerated() {
211 let target = Position(s: b, x: p.x + offset[0], y: p.y + offset[1])
212 if fits(board, kind, target) {
213 return Movement(position: target, test: index + 1)
214 }
215 }
216 return nil
217 }
218 let dx = (op == 0 || op == 2) ? -1 : (op == 1 || op == 3) ? 1 : 0
219 let dy = dx == 0 ? -1 : 0
220 var q = p
221 var changed = false
222 repeat {
223 let next = Position(s: p.s, x: q.x + dx, y: q.y + dy)
224 if !fits(board, kind, next) {
225 break
226 }
227 q = next
228 changed = true
229 } while op == 2 || op == 3 || op == 5
230 return changed ? Movement(position: q, test: 0) : nil
231}
232func landing(_ board: [Int], _ kind: Int, _ p: Position) -> Position {
233 move(board, kind, p, 5)?.position ?? p
234}
235func placed(_ board: [Int], _ kind: Int, _ p: Position) -> ([Int], Int) {
236 var next = board
237 for cell in cells(kind, p) {
238 next[cell.y - 1] |= 1 << (cell.x - 1)
239 }
240 let remaining = next.filter { $0 != 1023 }
241 let n = 22 - remaining.count
242 return (remaining + Array(repeating: 0, count: n), n)
243}
244func evaluate(_ board: [Int], _ cleared: Int) -> Metrics {
245 var heights = Array(repeating: 0, count: 10)
246 var holes = 0
247 for c in 0..<10 {
248 for r in stride(from: 21, through: 0, by: -1) where (board[r] & (1 << c)) != 0 {
249 heights[c] = r + 1
250 break
251 }
252 if heights[c] > 0 {
253 for r in 0..<heights[c] where (board[r] & (1 << c)) == 0 {
254 holes += 1
255 }
256 }
257 }
258 var bumpiness = 0
259 for c in 0..<9 {
260 bumpiness += abs(heights[c] - heights[c + 1])
261 }
262 return Metrics(
263 heights: heights,
264 holes: holes,
265 bumpiness: bumpiness,
266 value: 76 * cleared - 51 * heights.reduce(0,+) - 36 * holes - 18 * bumpiness
267 )
268}
269func routes(_ board: [Int], _ kind: Int, _ start: Position, _ target: Position? = nil) -> ([RouteNode], Int?) {
270 guard fits(board, kind, start) else {
271 return ([], nil)
272 }
273 var queue = [RouteNode(position: start, parent: -1, operation: -1)]
274 var seen: Set<Position> = [start]
275 var i = 0
276 while i < queue.count {
277 let entry = queue[i]
278 if let target, landing(board, kind, entry.position) == target {
279 return (queue, i)
280 }
281 for op in 0..<8 {
282 guard let result = move(board, kind, entry.position, op) else {
283 continue
284 }
285 if seen.insert(result.position).inserted {
286 queue.append(RouteNode(position: result.position, parent: i, operation: op))
287 }
288 }
289 i += 1
290 }
291 return (target == nil ? queue : [], nil)
292}
293func shortest(_ nodes: [RouteNode], _ index: Int) -> [Int] {
294 var i = index
295 var operations: [Int] = []
296 while nodes[i].parent >= 0 {
297 operations.append(nodes[i].operation)
298 i = nodes[i].parent
299 }
300 return operations.reversed()
301}
302func tSpin(_ board: [Int], _ p: Position, _ test: Int) -> Int {
303 let corners = [
304 Cell(x: p.x, y: p.y),
305 Cell(x: p.x + 2, y: p.y),
306 Cell(x: p.x + 2, y: p.y - 2),
307 Cell(x: p.x, y: p.y - 2)
308 ]
309 let blocked = corners
310 .map { c in c.x < 1 || c.x > 10 || c.y < 1 || c.y > 22 || (board[c.y - 1] & (1 << (c.x - 1))) != 0 }
311 if blocked.filter({ $0 }).count < 3 {
312 return 0
313 }
314 return (blocked[p.s] && blocked[(p.s + 1) % 4]) || test == 5 ? 2 : 1
315}

404 func operation(_ op: Int) throws {
405 if op < 8 {
406 state.keys += 1
407 guard let next = move(state.board, state.kind, state.position, op) else {
408 emit("fail \(operations[op])")
409 return
410 }
411 if op == 4 || op == 5 {
412 drops(state.position.y - next.position.y)
413 update(next.position, 1)
414 state.fall = 0
415 state.ground = 0
416 } else {
417 update(next.position, op >= 6 ? 2 : 1, next.test)
418 if state.ground > 0 && state.resets < 15 {
419 state.ground = 0
420 state.resets += 1
421 }
422 }
423 } else if op == 8 {
424 let p = landing(state.board, state.kind, state.position)
425 let rows = state.position.y - p.y
426 drops(2 * rows)
427 if rows > 0 {
428 update(p, 1)
429 }
430 try lock()
431 } else if op == 9 {
432 if grounded() {
433 try lock()
434 } else {
435 emit("fail LOCK")
436 }
437 } else {

439 emit("fail HOLD")

443 state.stats.holds += 1

445 state.hold = state.kind

447 }

449 func tick() throws {
450 if !grounded() {
451 state.ground = 0
452 state.fall += 1
453 let interval = state.level >= 13 ? 1 : [60, 48, 37, 28, 21, 16, 11, 8, 6, 4, 3, 2][state.level - 1]
454 if state.fall >= interval {
455 update(Position(s: state.position.s, x: state.position.x, y: state.position.y - 1), 1)
456 state.fall = 0
457 }
458 } else {
459 state.ground += 1
460 if state.ground >= 30 {

462 }

465 func lock() throws {
466 let kind = state.kind
467 let position = state.position
468 let oldCells = cells(kind, position)
469 let route = routes(state.board, kind, spawn(kind), position)
470 let shortestOperations = route.1.map { shortest(route.0, $0) }
471 let spin = kind == 2 && state.last == 2 ? tSpin(state.board, position, state.test) : 0
472 let placement = placed(state.board, kind, position)
473 let n = placement.1
474 state.board = placement.0
475 history.append(start)
476 state.pieces += 1
477 state.stats.kinds[kind] += 1
478 state.stats.operations += state.keys
479 let info = clearInfo(spin, n)
480 let level = state.level
481 var points = info.points * level
482 var sent = info.sent
483 var back = false
484 if n > 0 {
485 state.streak += 1
486 if n == 4 || spin != 0 {
487 state.chain += 1
488 back = state.chain >= 2
489 } else {
490 state.chain = 0
491 }
492 } else {
493 state.streak = 0
494 }
495 if back {
496 points = points * 3 / 2
497 sent += 1
498 }
499 let combo = state.streak - 1
500 let comboPoints = combo >= 1 ? 50 * combo * level : 0
501 if combo >= 0 {
502 sent += [0, 1, 1, 2, 2, 3, 3, 4, 4, 4, 5][min(combo, 10)]
503 }
504 let perfect = n > 0 && state.board.allSatisfy { $0 == 0 }
505 let perfectPoints = perfect ? (n == 4 && back ? 3200 : [800, 1200, 1800, 2000][n - 1]) * level : 0
506 if perfect {
507 sent += 10
508 state.stats.perfect += 1
509 }
510 let gained = points + comboPoints + perfectPoints
511 state.score += gained
512 state.lines += n
513 state.stats.sent += sent
514 state.stats.combo = max(state.stats.combo, combo)
515 state.stats.chain = max(state.stats.chain, state.chain)
516 emit(
517 "lock \(state.pieces) \(kinds[kind]) \(stateNames[position.s]) \(position.x) \(position.y) keys \(state.keys) \(shortestOperations.map { String($0.count) } ?? "-") drop \(state.drop) lines \(n) sent \(sent) +\(gained) total \(state.score)"
518 )
519 if let shortestOperations, state.keys > shortestOperations.count {
520 state.stats.faults += 1
521 emit("finesse \(shortestOperations.isEmpty ? "none" : operationText(shortestOperations))")
522 }
523 if info.name >= 0 {
524 state.stats.clears[info.name] += 1
525 var line = "clear \(clearNames[info.name]) \(points)"
526 if back {
527 line += " b2b"
528 }
529 if combo >= 1 {
530 line += " combo \(combo) \(comboPoints)"
531 }
532 if perfect {
533 line += " perfect \(perfectPoints)"
534 }
535 emit(line)
536 }
537 let nextLevel = state.initialLevel + state.lines / 10
538 if nextLevel > level {
539 state.level = nextLevel
540 emit("level \(nextLevel)")
541 }
542 if oldCells.allSatisfy({ $0.y >= 21 }) {
543 try gameOver("lock out")
544 }
545 let cancelled = min(sent, state.pending.reduce(0,+))
546 var remaining = cancelled
547 while remaining > 0 {
548 let used = min(remaining, state.pending[0])
549 remaining -= used
550 state.pending[0] -= used
551 if state.pending[0] == 0 {
552 state.pending.removeFirst()
553 }
554 }
555 if cancelled > 0 {
556 state.stats.cancelled += cancelled
557 emit("cancel \(cancelled) pending \(state.pending.reduce(0,+))")
558 }
559 if n == 0 {
560 var risen = 0
561 while !state.pending.isEmpty && risen < 8 {
562 let count = min(state.pending[0], 8 - risen)
563 state.random = seedStep(state.random)
564 let hole = state.random % 10 + 1
565 try insertGarbage(count, hole)
566 state.stats.received += count
567 risen += count
568 state.pending[0] -= count
569 if state.pending[0] == 0 {
570 state.pending.removeFirst()
571 }
572 emit("rise \(count) hole \(hole) pending \(state.pending.reduce(0,+))")
573 }
574 }
575 state.usedHold = false
576 try spawnPiece(takeNext())
577 start = state
578 }

580 var result: [Candidate] = []
581 var seen = Set<[Int]>()
582 for turns in [[], [6], [6, 6], [7]] {
583 var p = start
584 var valid = true
585 for turn in turns {
586 guard let next = move(state.board, kind, p, turn) else {

589 }
590 p = next.position
591 }

593 continue

595 func add(_ p: Position, _ shifts: [Int]) {
596 let landed = landing(state.board, kind, p)
597 let occupied = cells(kind, landed).map { ($0.y - 1) * 10 + $0.x - 1 }.sorted()
598 if !seen.insert(occupied).inserted {
599 return
600 }
601 let placedResult = placed(state.board, kind, landed)
602 let value = evaluate(placedResult.0, placedResult.1).value
603 result.append(Candidate(
604 kind: kind,
605 value: value,
606 operations: prefix + turns + shifts + [8],
607 occupied: occupied
608 ))
609 }
610 add(p, [])
611 for direction in [0, 1] {
612 var q = p
613 var shifts: [Int] = []
614 while let next = move(state.board, kind, q, direction) {
615 q = next.position
616 shifts.append(direction)
617 add(q, shifts)
618 }
619 }
620 }
621 return result
622 }

632 result.sort { a, b in a.value == b.value ? a.order < b.order : a.value > b.value }

687 func command(_ name: String, _ args: [String]) throws {
688 switch name {
689 case "PRINT":
690 var current = Set<Cell>(), ghost = Set<Cell>()
691 if !state.ended {
692 current = Set(cells(state.kind, state.position))
693 ghost = Set(cells(state.kind, landing(state.board, state.kind, state.position)))
694 }
695 for r in stride(from: 22, through: 1, by: -1) {
696 var line = String(format: "%2d ", r)
697 for c in 1...10 {
698 let cell = Cell(x: c, y: r)
699 if (state.board[r - 1] & (1 << (c - 1))) != 0 {
700 line += "#"
701 } else if current.contains(cell) {
702 line += "@"
703 } else if ghost.contains(cell) {
704 line += "+"
705 } else {
706 line += "."
707 }
708 }
709 emit(line)
710 }
711 case "NEXT":
712 emit("hold \(state.hold.map { String(kinds[$0]) } ?? "-") next " + state.queue.prefix(5)
713 .map { String(kinds[$0]) }.joined())
714 case "PREVIEW":
715 var queue = state.queue, random = state.random
716 let n = Int(args[0])!
717 while queue.count < n {
718 let result = bag(random)
719 random = result.0
720 queue += result.1
721 }
722 emit("preview " + queue.prefix(n).map { String(kinds[$0]) }.joined())
723 case "STATUS":
724 emit("score \(state.score) level \(state.level) lines \(state.lines) pieces \(state.pieces)")
725 let piece = state.ended ? "-" : "\(kinds[state.kind]) \(stateNames[state.position.s]) \(state.position.x) \(state.position.y)"
726 emit("piece \(piece) streak \(state.streak) chain \(state.chain) pending \(state.pending.reduce(0,+))")
727 case "ANALYZE":
728 let e = evaluate(state.board, 0)
729 emit("heights " + e.heights.map(String.init).joined(separator: " "))
730 emit("holes \(e.holes) bumpiness \(e.bumpiness) value \(e.value)")
731 case "ROW":
732 let row = Int(args[0])!, pattern = Array(args[1])
733 if cells(state.kind, state.position)
734 .contains(where: { $0.y == row && pattern[$0.x - 1] == "#" })
735 {
736 emit("ERROR overlap")
737 return
738 }
739 var mask = 0
740 for i in 0..<10 where pattern[i] == "#" {
741 mask |= 1 << i
742 }
743 state.board[row - 1] = mask
744 emit("OK")
745 case "GARBAGE":
746 let n = Int(args[0])!, hole = Int(args[1])!
747 try insertGarbage(n, hole)
748 for j in 0...n {
749 let p = Position(s: state.position.s, x: state.position.x, y: state.position.y + j)
750 if fits(state.board, state.kind, p) {
751 state.position = p
752 emit("OK")
753 return
754 }
755 }
756 try gameOver("top out")
757 case "ATTACK":
758 state.pending.append(Int(args[0])!)
759 emit("pending \(state.pending.reduce(0,+))")
760 case "TICK":
761 for _ in 0..<Int(args[0])! {
762 try tick()
763 }
764 case "UNDO":
765 guard !history.isEmpty else {
766 emit("ERROR nothing to undo")
767 return
768 }
769 state = history.removeLast()
770 start = state
771 emit("undo \(state.pieces) \(kinds[state.kind])")
772 case "SAVE":
773 saves[args[0]] = SavedGame(state: state, history: history, start: start)
774 emit("OK")
775 case "LOAD":
776 guard let saved = saves[args[0]] else {
777 emit("ERROR unknown save")
778 return
779 }
780 state = saved.state
781 history = saved.history
782 start = saved.start
783 emit("load \(state.pieces) \(kinds[state.kind])")
784 case "SPINS":
785 spins()
786 case "HINT":
787 for (i, c) in ranked().prefix(Int(args[0])!).enumerated() {
788 emit("hint \(i + 1) \(kinds[c.kind]) \(operationText(c.operations)) value \(c.value)")
789 }
790 case "AUTO":
791 for _ in 0..<Int(args[0])! {
792 let c = ranked()[0]
793 emit("auto \(kinds[c.kind]) \(operationText(c.operations))")
794 for op in c.operations {
795 try operation(op)
796 }
797 }
798 default:
799 break
800 }
801 }
802}
803func validNumber(_ string: String, _ minValue: Int, _ maxValue: Int) -> Bool {
804 if string.isEmpty || (string.count > 1 && string.first == "0") || !string.utf8
805 .allSatisfy({ $0 >= 48 && $0 <= 57 })
806 {
807 return false
808 }
809 guard let n = Int(string) else {
810 return false
811 }
812 return n >= minValue && n <= maxValue
813}
814func validArgs(_ command: String, _ args: [String]) -> Bool {
815 if command == "NEW" {
816 return args.count == 2 && validNumber(args[0], 1, 2147483646) && validNumber(args[1], 1, 15)
817 }
818 if command == "ROW" {
819 return args.count == 2 && validNumber(args[0], 1, 22) && args[1].count == 10 && args[1]
820 .allSatisfy({ $0 == "#" || $0 == "." }) && args[1].contains(".")
821 }
822 if command == "GARBAGE" {
823 return args.count == 2 && validNumber(args[0], 1, 10) && validNumber(args[1], 1, 10)
824 }
825 if command == "SAVE" || command ==
826 "LOAD"
827 {
828 return args.count == 1 && (1...8).contains(args[0].count) && args[0].utf8.allSatisfy({ $0 >= 97 && $0 <= 122 })
829 }
830 let limits = ["PREVIEW": 14, "ATTACK": 20, "TICK": 1000, "HINT": 10, "AUTO": 50]
831 if let limit = limits[command] {
832 return args.count == 1 && validNumber(args[0], 1, limit)
833 }
834 return args.isEmpty
835}

838var game: Game?
839var serial = 0
840for _ in 0..<inputCount {
841 guard let line = readLine() else {

844 let words = line.split(separator: " ").map(String.init)

847 }
848 let args = Array(words.dropFirst())
849 do {
850 if commands.contains(name) {
851 if !validArgs(name, args) {
852 emit("ERROR bad arguments")
853 } else if name == "NEW" {

857 }
858 serial += 1

860 game = try Game(number: serial, seed: Int(args[0])!, level: Int(args[1])!)
861 } else if let game {
862 if game.state.ended && !allowedAfterEnd.contains(name) {
863 emit("ERROR game over")
864 } else {
865 try game.command(name, args)
866 }
867 } else {
868 emit("ERROR no game")
869 }
870 } else {
871 let unknown = words.first(where: { !operations.contains($0) })
872 if let unknown {
873 emit("ERROR unknown \(unknown)")
874 } else if let game {
875 if game.state.ended {
876 emit("ERROR game over")
877 } else {
878 for word in words {
879 try game.operation(operations.firstIndex(of: word)!)
880 }
881 }
882 } else {
883 emit("ERROR no game")
884 }
885 }
886 } catch GameOver.ended {}
887}
