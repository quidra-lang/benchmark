141func cells(kind byte, pos position) [4]point {
142 frame := frames[kind]
143 size := len(frame)
144 var result [4]point
145 count := 0
146 for j, row := range frame {
147 for i := range row {
148 if row[i] != '#' {
149 continue
150 }
151 x, y := i, j
152 for r := 0; r < pos.rotation; r++ {
153 x, y = size-1-y, x
154 }
155 result[count] = point{pos.x + x, pos.y - y}
156 count++
157 }
158 }
159 return result
160}

162func lowest(kind byte, pos position) int {
163 row := 23
164 for _, p := range cells(kind, pos) {
165 if p.y < row {
166 row = p.y
167 }
168 }
169 return row
170}

172func fits(board [22]int, kind byte, pos position) bool {
173 for _, p := range cells(kind, pos) {
174 if p.x < 1 || p.x > 10 || p.y < 1 || p.y > 22 {
175 return false
176 }
177 if board[p.y-1]&(1<<uint(p.x-1)) != 0 {
178 return false
179 }
180 }
181 return true
182}

206func moved(board [22]int, kind byte, pos position, action string) (movement, bool) {
207 if action == "CW" || action == "CCW" {
208 increment := 1
209 if action == "CCW" {
210 increment = 3
211 }
212 target := (pos.rotation + increment) % 4
213 offsets := []point{{0, 0}}
214 if kind != 'O' {
215 key := strconv.Itoa(pos.rotation) + strconv.Itoa(target)
216 if kind == 'I' {
217 offsets = iKicks[key]
218 } else {
219 offsets = ordinaryKicks[key]
220 }
221 }
222 for i, delta := range offsets {
223 to := position{target, pos.x + delta.x, pos.y + delta.y}
224 if fits(board, kind, to) {
225 return movement{to, i + 1}, true
226 }
227 }
228 return movement{}, false
229 }
230 dx := 0
231 if action == "L" || action == "LL" {
232 dx = -1
233 }
234 if action == "R" || action == "RR" {
235 dx = 1
236 }
237 dy := 0
238 if dx == 0 {
239 dy = -1
240 }
241 repeat := action == "LL" || action == "RR" || action == "DD"
242 changed := false
243 current := pos
244 for {
245 next := position{current.rotation, current.x + dx, current.y + dy}
246 if !fits(board, kind, next) {
247 break
248 }
249 current = next
250 changed = true
251 if !repeat {
252 break
253 }
254 }
255 return movement{current, 0}, changed
256}

258func landing(board [22]int, kind byte, pos position) position {
259 result, ok := moved(board, kind, pos, "DD")
260 if ok {
261 return result.position
262 }
263 return pos

266func placed(board [22]int, kind byte, pos position) placement {
267 result := board
268 for _, p := range cells(kind, pos) {
269 result[p.y-1] |= 1 << uint(p.x-1)
270 }
271 var filtered [22]int
272 length := 0
273 for _, row := range result {
274 if row != 1023 {
275 filtered[length] = row
276 length++
277 }
278 }
279 return placement{filtered, 22 - length}
280}

282func evaluate(board [22]int, cleared int) evaluation {
283 result := evaluation{}
284 sum := 0
285 for col := 0; col < 10; col++ {
286 for row := 21; row >= 0; row-- {
287 if board[row]&(1<<uint(col)) != 0 {
288 result.heights[col] = row + 1
289 break
290 }
291 }
292 sum += result.heights[col]
293 for row := 0; row < result.heights[col]; row++ {
294 if board[row]&(1<<uint(col)) == 0 {
295 result.holes++
296 }
297 }
298 if col > 0 {
299 delta := result.heights[col] - result.heights[col-1]
300 if delta < 0 {
301 delta = -delta
302 }
303 result.bumpiness += delta
304 }
305 }
306 result.value = 76*cleared - 51*sum - 36*result.holes - 18*result.bumpiness
307 return result
308}

310func routes(board [22]int, kind byte, first position, target *position) []route {
311 if !fits(board, kind, first) {
312 return nil
313 }
314 queue := []route{{first, nil}}
315 seen := map[position]bool{first: true}
316 for i := 0; i < len(queue); i++ {
317 entry := queue[i]
318 if target != nil && landing(board, kind, entry.position) == *target {
319 return []route{entry}
320 }
321 for _, action := range movements {
322 next, ok := moved(board, kind, entry.position, action)
323 if ok && !seen[next.position] {
324 seen[next.position] = true
325 keys := append(append([]string(nil), entry.keys...), action)
326 queue = append(queue, route{next.position, keys})
327 }
328 }
329 }
330 if target != nil {
331 return nil
332 }
333 return queue
334}

336func spinType(board [22]int, pos position, test int) string {
337 corners := [4]point{{pos.x, pos.y}, {pos.x + 2, pos.y}, {pos.x + 2, pos.y - 2}, {pos.x, pos.y - 2}}
338 var occupied [4]bool
339 count := 0
340 for i, p := range corners {
341 occupied[i] = p.x < 1 || p.x > 10 || p.y < 1 || p.y > 22 || board[p.y-1]&(1<<uint(p.x-1)) != 0
342 if occupied[i] {
343 count++
344 }
345 }
346 if count < 3 {
347 return ""
348 }
349 if (occupied[pos.rotation] && occupied[(pos.rotation+1)%4]) || test == 5 {
350 return "full"
351 }
352 return "mini"
353}

355func cleared(spin string, lines int) clearInfo {
356 if spin == "full" {
357 points, sent := [4]int{400, 800, 1200, 1600}, [4]int{0, 2, 4, 6}
358 return clearInfo{clearNames[7+lines], points[lines], sent[lines], 7 + lines}
359 }
360 if spin == "mini" {
361 points, sent := [3]int{100, 200, 400}, [3]int{0, 0, 1}
362 return clearInfo{clearNames[4+lines], points[lines], sent[lines], 4 + lines}
363 }
364 if lines == 0 {
365 return clearInfo{"", 0, 0, -1}
366 }
367 points, sent := [5]int{0, 100, 300, 500, 800}, [5]int{0, 0, 1, 2, 4}
368 return clearInfo{clearNames[lines-1], points[lines], sent[lines], lines - 1}
369}

397func (g *game) addBag() {
398 pieces := []byte(kinds)
399 for i := 6; i >= 1; i-- {
400 g.state.random = nextRandom(g.state.random)
401 j := int(g.state.random % int64(i+1))
402 pieces[i], pieces[j] = pieces[j], pieces[i]
403 }
404 g.state.queue = append(g.state.queue, pieces...)
405}

410 if len(g.state.queue) < 5 {

412 }

455func (g *game) garbage(count, hole int) {
456 s := &g.state
457 for r := 22 - count; r < 22; r++ {
458 if s.board[r] != 0 {
459 g.gameOver("top out")
460 }
461 }
462 for r := 21; r >= count; r-- {
463 s.board[r] = s.board[r-count]
464 }
465 for r := 0; r < count; r++ {
466 s.board[r] = 1023 ^ (1 << uint(hole-1))
467 }
468}

470func (g *game) operation(action string) {
471 s := &g.state
472 for _, candidate := range movements {
473 if candidate != action {
474 continue
475 }
476 s.keys++
477 result, ok := moved(s.board, s.kind, s.position, action)
478 if !ok {
479 fmt.Println("fail", action)
480 return
481 }
482 if action == "SD" || action == "DD" {
483 g.drops(s.position.y - result.position.y)
484 g.updatePosition(result.position, 1, 0)
485 s.fall, s.ground = 0, 0
486 } else {
487 rotate := action == "CW" || action == "CCW"
488 last := 1
489 if rotate {
490 last = 2
491 }
492 g.updatePosition(result.position, last, result.test)
493 if s.ground > 0 && s.resets < 15 {
494 s.ground = 0
495 s.resets++
496 }
497 }
498 return
499 }
500 switch action {
501 case "HD":
502 to := landing(s.board, s.kind, s.position)
503 rows := s.position.y - to.y
504 g.drops(2 * rows)
505 if rows > 0 {
506 g.updatePosition(to, 1, 0)
507 }
508 g.lock()
509 case "LOCK":
510 if g.grounded() {
511 g.lock()
512 } else {
513 fmt.Println("fail LOCK")
514 }
515 case "HOLD":
516 if s.usedHold {
517 fmt.Println("fail HOLD")
518 return
519 }
520 s.usedHold = true
521 s.stats.holds++
522 current := s.kind
523 next := s.hold
524 if next == '-' {
525 next = g.takeNext()
526 }
527 s.hold = current
528 g.spawnPiece(next)
529 }
530}

532func (g *game) tick() {
533 s := &g.state
534 if !g.grounded() {
535 s.ground = 0
536 s.fall++
537 interval := 1
538 if s.level < 13 {
539 interval = intervals[s.level-1]
540 }
541 if s.fall >= interval {
542 g.updatePosition(position{s.position.rotation, s.position.x, s.position.y - 1}, 1, 0)
543 s.fall = 0
544 }
545 } else {
546 s.ground++
547 if s.ground >= 30 {
548 g.lock()
549 }
550 }
551}

553func (g *game) lock() {
554 s := &g.state
555 kind, pos := s.kind, s.position
556 oldCells := cells(kind, pos)
557 shortest := routes(s.board, kind, spawn(kind), &pos)
558 spin := ""
559 if kind == 'T' && s.last == 2 {
560 spin = spinType(s.board, pos, s.test)
561 }
562 p := placed(s.board, kind, pos)
563 s.board = p.board
564 n := p.lines
565 g.history = append(g.history, g.start.clone())
566 s.pieces++
567 s.stats.kinds[strings.IndexByte(kinds, kind)]++
568 s.stats.operations += s.keys
569 info := cleared(spin, n)
570 level := s.level
571 points, sent := info.points*level, info.sent
572 backToBack := false
573 if n > 0 {
574 s.streak++
575 if n == 4 || spin != "" {
576 s.chain++
577 backToBack = s.chain >= 2
578 } else {
579 s.chain = 0
580 }
581 } else {
582 s.streak = 0
583 }
584 if backToBack {
585 points = points * 3 / 2
586 sent++
587 }
588 combo := s.streak - 1
589 comboPoints := 0
590 if combo >= 1 {
591 comboPoints = 50 * combo * level
592 }
593 if combo >= 0 {
594 i := combo
595 if i > 10 {
596 i = 10
597 }
598 sent += comboSent[i]
599 }
600 perfect := n > 0
601 for _, row := range s.board {
602 if row != 0 {
603 perfect = false
604 break
605 }
606 }
607 perfectPoints := 0
608 if perfect {
609 table := [4]int{800, 1200, 1800, 2000}
610 value := table[n-1]
611 if n == 4 && backToBack {
612 value = 3200
613 }
614 perfectPoints = value * level
615 sent += 10
616 s.stats.perfect++
617 }
618 gained := points + comboPoints + perfectPoints
619 s.score += int64(gained)
620 s.lines += n
621 s.stats.sent += sent
622 if combo > s.stats.combo {
623 s.stats.combo = combo
624 }
625 if s.chain > s.stats.chain {
626 s.stats.chain = s.chain
627 }
628 distance := "-"
629 if len(shortest) > 0 {
630 distance = strconv.Itoa(len(shortest[0].keys))
631 }
632 fmt.Printf("lock %d %c %s %d %d keys %d %s drop %d lines %d sent %d +%d total %d\n", s.pieces, kind, stateNames[pos.rotation], pos.x, pos.y, s.keys, distance, s.drop, n, sent, gained, s.score)
633 if len(shortest) > 0 && s.keys > len(shortest[0].keys) {
634 s.stats.faults++
635 sequence := strings.Join(shortest[0].keys, " ")
636 if sequence == "" {
637 sequence = "none"
638 }
639 fmt.Println("finesse", sequence)
640 }
641 if info.index >= 0 {
642 s.stats.clears[info.index]++
643 output := fmt.Sprintf("clear %s %d", info.name, points)
644 if backToBack {
645 output += " b2b"
646 }
647 if combo >= 1 {
648 output += fmt.Sprintf(" combo %d %d", combo, comboPoints)
649 }
650 if perfect {
651 output += fmt.Sprintf(" perfect %d", perfectPoints)
652 }
653 fmt.Println(output)
654 }
655 nextLevel := s.initialLevel + s.lines/10
656 if nextLevel > level {
657 s.level = nextLevel
658 fmt.Println("level", s.level)
659 }
660 lockOut := true
661 for _, c := range oldCells {
662 if c.y < 21 {
663 lockOut = false
664 }
665 }
666 if lockOut {
667 g.gameOver("lock out")
668 }
669 cancelled := sent
670 if pending := sumPending(s.pending); pending < cancelled {
671 cancelled = pending
672 }
673 remaining := cancelled
674 for remaining > 0 {
675 count := s.pending[0]
676 if count > remaining {
677 count = remaining
678 }
679 s.pending[0] -= count
680 remaining -= count
681 if s.pending[0] == 0 {
682 s.pending = s.pending[1:]
683 }
684 }
685 if cancelled > 0 {
686 s.stats.cancelled += cancelled
687 fmt.Printf("cancel %d pending %d\n", cancelled, sumPending(s.pending))
688 }
689 if n == 0 {
690 risen := 0
691 for len(s.pending) > 0 && risen < 8 {
692 count := s.pending[0]
693 if count > 8-risen {
694 count = 8 - risen
695 }
696 s.random = nextRandom(s.random)
697 hole := int(s.random%10) + 1
698 g.garbage(count, hole)
699 s.stats.received += count
700 risen += count
701 s.pending[0] -= count
702 if s.pending[0] == 0 {
703 s.pending = s.pending[1:]
704 }
705 fmt.Printf("rise %d hole %d pending %d\n", count, hole, sumPending(s.pending))
706 }
707 }
708 s.usedHold = false
709 g.spawnPiece(g.takeNext())
710 g.start = s.clone()
711}

713func (g *game) appendCandidate(list *[]candidate, seen map[[4]int]bool, kind byte, pos position, prefix, turns, shifts []string) {
714 down := landing(g.state.board, kind, pos)
715 var id [4]int
716 for i, p := range cells(kind, down) {
717 id[i] = (p.y-1)*10 + p.x - 1
718 }
719 sort.Ints(id[:])
720 if seen[id] {
721 return
722 }
723 seen[id] = true
724 p := placed(g.state.board, kind, down)
725 keys := make([]string, 0, len(prefix)+len(turns)+len(shifts)+1)
726 keys = append(keys, prefix...)
727 keys = append(keys, turns...)
728 keys = append(keys, shifts...)
729 keys = append(keys, "HD")
730 *list = append(*list, candidate{evaluate(p.board, p.lines).value, kind, keys})
731}

733func (g *game) candidates(kind byte, first position, prefix []string) []candidate {
734 result := []candidate{}
735 seen := make(map[[4]int]bool)
736 rotations := [][]string{{}, {"CW"}, {"CW", "CW"}, {"CCW"}}
737 for _, turns := range rotations {
738 pos := first
739 valid := true
740 for _, turn := range turns {
741 next, ok := moved(g.state.board, kind, pos, turn)
742 if !ok {
743 valid = false
744 break
745 }
746 pos = next.position
747 }
748 if !valid {
749 continue
750 }
751 g.appendCandidate(&result, seen, kind, pos, prefix, turns, nil)
752 for _, direction := range []string{"L", "R"} {
753 shifted := pos
754 shifts := []string{}
755 for {
756 next, ok := moved(g.state.board, kind, shifted, direction)
757 if !ok {
758 break
759 }
760 shifted = next.position
761 shifts = append(shifts, direction)
762 g.appendCandidate(&result, seen, kind, shifted, prefix, turns, shifts)
763 }
764 }
765 }
766 return result
767}

769func (g *game) ranked() []candidate {
770 s := &g.state
771 result := g.candidates(s.kind, s.position, nil)
772 held := s.hold
773 if held == '-' {
774 held = s.queue[0]
775 }
776 if !s.usedHold && fits(s.board, held, spawn(held)) {
777 result = append(result, g.candidates(held, spawn(held), []string{"HOLD"})...)
778 }
779 sort.SliceStable(result, func(i, j int) bool { return result[i].value > result[j].value })
780 return result

783func (g *game) spins() {
784 s := &g.state
785 if s.kind != 'T' {
786 fmt.Println("spins 0")
787 return
788 }
789 found := map[position]string{}
790 for _, r := range routes(s.board, 'T', s.position, nil) {
791 for _, action := range []string{"CW", "CCW"} {
792 next, ok := moved(s.board, 'T', r.position, action)
793 if !ok {
794 continue
795 }
796 if _, ok = moved(s.board, 'T', next.position, "SD"); ok {
797 continue
798 }
799 spin := spinType(s.board, next.position, next.test)
800 if spin != "" && (found[next.position] == "" || spin == "full") {
801 found[next.position] = spin
802 }
803 }
804 }
805 positions := make([]position, 0, len(found))
806 for pos := range found {
807 positions = append(positions, pos)
808 }
809 sort.Slice(positions, func(i, j int) bool {
810 a, b := positions[i], positions[j]
811 if a.y != b.y {
812 return a.y < b.y
813 }
814 if a.x != b.x {
815 return a.x < b.x
816 }
817 return a.rotation < b.rotation
818 })
819 for _, pos := range positions {
820 n := placed(s.board, 'T', pos).lines
821 fmt.Printf("spin %s %d %d %s\n", stateNames[pos.rotation], pos.x, pos.y, cleared(found[pos], n).name)
822 }
823 fmt.Println("spins", len(positions))
824}

839func (g *game) command(name string, args []string) {
840 s := &g.state
841 switch name {
842 case "PRINT":
843 current, ghost := map[int]bool{}, map[int]bool{}
844 if !s.ended {
845 for _, p := range cells(s.kind, s.position) {
846 current[p.y*10+p.x] = true
847 }
848 for _, p := range cells(s.kind, landing(s.board, s.kind, s.position)) {
849 ghost[p.y*10+p.x] = true
850 }
851 }
852 for row := 22; row >= 1; row-- {
853 text := fmt.Sprintf("%2d ", row)
854 for col := 1; col <= 10; col++ {
855 id := row*10 + col
856 if s.board[row-1]&(1<<uint(col-1)) != 0 {
857 text += "#"
858 } else if current[id] {
859 text += "@"
860 } else if ghost[id] {
861 text += "+"
862 } else {
863 text += "."
864 }
865 }
866 fmt.Println(text)
867 }
868 case "NEXT":
869 fmt.Printf("hold %c next %s\n", s.hold, string(s.queue[:5]))
870 case "PREVIEW":
871 n, _ := strconv.Atoi(args[0])
872 queue := append([]byte(nil), s.queue...)
873 random := s.random
874 for len(queue) < n {
875 pieces := []byte(kinds)
876 for i := 6; i >= 1; i-- {
877 random = nextRandom(random)
878 j := int(random % int64(i+1))
879 pieces[i], pieces[j] = pieces[j], pieces[i]
880 }
881 queue = append(queue, pieces...)
882 }
883 fmt.Println("preview", string(queue[:n]))
884 case "STATUS":
885 fmt.Printf("score %d level %d lines %d pieces %d\n", s.score, s.level, s.lines, s.pieces)
886 piece := "-"
887 if !s.ended {
888 piece = fmt.Sprintf("%c %s %d %d", s.kind, stateNames[s.position.rotation], s.position.x, s.position.y)
889 }
890 fmt.Printf("piece %s streak %d chain %d pending %d\n", piece, s.streak, s.chain, sumPending(s.pending))
891 case "ANALYZE":
892 e := evaluate(s.board, 0)
893 text := "heights"
894 for _, value := range e.heights {
895 text += " " + strconv.Itoa(value)
896 }
897 fmt.Println(text)
898 fmt.Printf("holes %d bumpiness %d value %d\n", e.holes, e.bumpiness, e.value)
899 case "ROW":
900 row, _ := strconv.Atoi(args[0])
901 pattern := args[1]
902 for _, p := range cells(s.kind, s.position) {
903 if p.y == row && pattern[p.x-1] == '#' {
904 fmt.Println("ERROR overlap")
905 return
906 }
907 }
908 mask := 0
909 for c := range pattern {
910 if pattern[c] == '#' {
911 mask |= 1 << uint(c)
912 }
913 }
914 s.board[row-1] = mask
915 fmt.Println("OK")
916 case "GARBAGE":
917 n, _ := strconv.Atoi(args[0])
918 hole, _ := strconv.Atoi(args[1])
919 g.garbage(n, hole)
920 for j := 0; j <= n; j++ {
921 next := position{s.position.rotation, s.position.x, s.position.y + j}
922 if fits(s.board, s.kind, next) {
923 s.position = next
924 fmt.Println("OK")
925 return
926 }
927 }
928 g.gameOver("top out")
929 case "ATTACK":
930 n, _ := strconv.Atoi(args[0])
931 s.pending = append(s.pending, n)
932 fmt.Println("pending", sumPending(s.pending))
933 case "TICK":
934 n, _ := strconv.Atoi(args[0])
935 for i := 0; i < n; i++ {
936 g.tick()
937 }
938 case "UNDO":
939 if len(g.history) == 0 {
940 fmt.Println("ERROR nothing to undo")
941 } else {
942 s = &g.state
943 g.state = g.history[len(g.history)-1].clone()
944 g.history = g.history[:len(g.history)-1]
945 g.start = g.state.clone()
946 fmt.Printf("undo %d %c\n", g.state.pieces, g.state.kind)
947 }
948 case "SAVE":
949 g.saves[args[0]] = snapshot{s.clone(), copyStates(g.history), g.start.clone()}
950 fmt.Println("OK")
951 case "LOAD":
952 snap, ok := g.saves[args[0]]
953 if !ok {
954 fmt.Println("ERROR unknown save")
955 } else {
956 g.state = snap.state.clone()
957 g.history = copyStates(snap.history)
958 g.start = snap.start.clone()
959 fmt.Printf("load %d %c\n", g.state.pieces, g.state.kind)
960 }
961 case "SPINS":
962 g.spins()
963 case "HINT":
964 n, _ := strconv.Atoi(args[0])
965 ranked := g.ranked()
966 if n > len(ranked) {
967 n = len(ranked)
968 }
969 for i := 0; i < n; i++ {
970 entry := ranked[i]
971 fmt.Printf("hint %d %c %s value %d\n", i+1, entry.kind, strings.Join(entry.keys, " "), entry.value)
972 }
973 case "AUTO":
974 n, _ := strconv.Atoi(args[0])
975 for i := 0; i < n; i++ {
976 ranked := g.ranked()
977 if len(ranked) == 0 {
978 break
979 }
980 best := ranked[0]
981 fmt.Printf("auto %c %s\n", best.kind, strings.Join(best.keys, " "))
982 for _, action := range best.keys {
983 g.operation(action)
984 }
985 }
986 }
987}

989func number(word string, low, high int64) bool {
990 if len(word) == 0 || (len(word) > 1 && word[0] == '0') {
991 return false
992 }
993 var value int64
994 for _, c := range word {
995 if c < '0' || c > '9' {
996 return false
997 }
998 value = value*10 + int64(c-'0')
999 if value > high {
1000 return false
1001 }
1002 }
1003 return value >= low
1004}

1006func argumentsValid(name string, args []string) bool {
1007 switch name {
1008 case "NEW":
1009 return len(args) == 2 && number(args[0], 1, 2147483646) && number(args[1], 1, 15)
1010 case "ROW":
1011 if len(args) != 2 || !number(args[0], 1, 22) || len(args[1]) != 10 || !strings.Contains(args[1], ".") {
1012 return false
1013 }
1014 for _, c := range args[1] {
1015 if c != '#' && c != '.' {
1016 return false
1017 }
1018 }
1019 return true
1020 case "GARBAGE":
1021 return len(args) == 2 && number(args[0], 1, 10) && number(args[1], 1, 10)
1022 case "SAVE", "LOAD":
1023 if len(args) != 1 || len(args[0]) == 0 || len(args[0]) > 8 {
1024 return false
1025 }
1026 for _, c := range args[0] {
1027 if c < 'a' || c > 'z' {
1028 return false
1029 }
1030 }
1031 return true
1032 }
1033 limits := map[string]int{"PREVIEW": 14, "ATTACK": 20, "TICK": 1000, "HINT": 10, "AUTO": 50}
1034 if high, ok := limits[name]; ok {
1035 return len(args) == 1 && number(args[0], 1, int64(high))
1036 }
1037 return len(args) == 0
1038}

1040func process(line string, current **game, serial *int) {
1041 defer func() {
1042 if value := recover(); value != nil {
1043 if _, ok := value.(gameEnd); !ok {
1044 panic(value)
1045 }
1046 }
1047 }()
1048 words := strings.Fields(line)
1049 if len(words) == 0 {
1050 return
1051 }
1052 name, args := words[0], words[1:]
1053 if commands[name] {
1054 if !argumentsValid(name, args) {
1055 fmt.Println("ERROR bad arguments")
1056 return
1057 }
1058 if name == "NEW" {
1059 if *current != nil && !(*current).state.ended {
1060 fmt.Println("game stopped")
1061 (*current).summary()
1062 }
1063 seed, _ := strconv.ParseInt(args[0], 10, 64)
1064 level, _ := strconv.Atoi(args[1])
1065 *serial = *serial + 1
1066 fmt.Printf("game %d seed %d level %d\n", *serial, seed, level)
1067 *current = newGame(*serial, seed, level)
1068 } else if *current == nil {
1069 fmt.Println("ERROR no game")
1070 } else if (*current).state.ended && !afterEnd[name] {
1071 fmt.Println("ERROR game over")
1072 } else {
1073 (*current).command(name, args)
1074 }
1075 } else {
1076 for _, word := range words {
1077 if !operations[word] {
1078 fmt.Println("ERROR unknown", word)
1079 return
1080 }
1081 }
1082 if *current == nil {
1083 fmt.Println("ERROR no game")
1084 return
1085 }
1086 if (*current).state.ended {
1087 fmt.Println("ERROR game over")
1088 return
1089 }
1090 for _, action := range words {
1091 (*current).operation(action)
1092 }
1093 }
1094}

1096func main() {
1097 input := bufio.NewScanner(os.Stdin)
1098 input.Buffer(make([]byte, 1024), 1024*1024)
1099 if !input.Scan() {
1100 return
1101 }
1102 count, _ := strconv.Atoi(strings.TrimSpace(input.Text()))
1103 var current *game
1104 serial := 0

1107 }

1110 current.summary()
1111 }
