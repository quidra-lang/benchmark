156static bool fits(const int board[22], int kind, Position p) {
157 Cell c[4];
158 piece_cells(kind, p, c);
159 for (int i = 0; i < 4; ++i) {
160 int x = c[i].x, y = c[i].y;
161 if (x < 1 || x > 10 || y < 1 || y > 22 || (board[y - 1] & (1 << (x - 1))) != 0)
162 return false;
163 }
164 return true;

166static bool try_move(const int board[22], int kind, Position p, int op, Position *result,
167 int *test) {
168 if (op >= 6 && op <= 7) {
169 int b = (p.s + (op == 6 ? 1 : 3)) % 4, row = 0;
170 for (int i = 0; i < 8; ++i)
171 if (kick_states[i][0] == p.s && kick_states[i][1] == b) {
172 row = i;
173 break;
174 }
175 for (int i = 0; i < (kind == 1 ? 1 : 5); ++i) {
176 int dx = kind == 1 ? 0 : (kind == 0 ? kick_i : kick_other)[row][i][0];
177 int dy = kind == 1 ? 0 : (kind == 0 ? kick_i : kick_other)[row][i][1];
178 Position q = position(b, p.x + dx, p.y + dy);
179 if (fits(board, kind, q)) {
180 *result = q;
181 *test = i + 1;
182 return true;
183 }
184 }
185 return false;
186 }
187 int dx = (op == 0 || op == 2) ? -1 : (op == 1 || op == 3) ? 1 : 0;
188 int dy = dx == 0 ? -1 : 0;
189 bool repeat = op == 2 || op == 3 || op == 5, changed = false;
190 Position q = p;
191 do {
192 Position n = position(q.s, q.x + dx, q.y + dy);
193 if (!fits(board, kind, n))
194 break;
195 q = n;
196 changed = true;
197 } while (repeat);
198 if (changed) {
199 *result = q;
200 *test = 0;
201 return true;
202 }
203 return false;
204}
205static Position landing(const int board[22], int kind, Position p) {

210static int place_piece(const int board[22], int kind, Position p, int result[22]) {
211 memcpy(result, board, sizeof(int) * 22);
212 Cell c[4];
213 piece_cells(kind, p, c);
214 for (int i = 0; i < 4; ++i)
215 result[c[i].y - 1] |= 1 << (c[i].x - 1);
216 int count = 0;
217 for (int i = 0; i < 22; ++i)
218 if (result[i] != 1023)
219 result[count++] = result[i];
220 int lines = 22 - count;
221 for (int i = count; i < 22; ++i)
222 result[i] = 0;
223 return lines;
224}
225static int evaluate_well(const int board[22], int cleared, int heights[10], int *holes,
226 int *bumpiness) {
227 *holes = 0;
228 *bumpiness = 0;
229 int total = 0;
230 for (int c = 0; c < 10; ++c) {
231 int h = 0;
232 for (int r = 21; r >= 0; --r)
233 if ((board[r] & (1 << c)) != 0) {
234 h = r + 1;
235 break;
236 }
237 heights[c] = h;
238 total += h;
239 for (int r = 0; r < h; ++r)
240 if ((board[r] & (1 << c)) == 0)
241 ++*holes;
242 }
243 for (int c = 0; c < 9; ++c)
244 *bumpiness += abs(heights[c] - heights[c + 1]);
245 return 76 * cleared - 51 * total - 36 * (*holes) - 18 * (*bumpiness);
246}
247static int search_routes(const int board[22], int kind, Position start, const Position *target,
248 int *last_index) {
249 if (!fits(board, kind, start))
250 return 0;
251 static bool seen[4][32][32];
252 memset(seen, 0, sizeof(seen));
253 bfs[0] = (BFSNode){start, -1, -1};
254 int count = 1;
255 if (start.s >= 0 && start.s < 4 && start.x >= -8 && start.x < 24 && start.y >= 0 &&
256 start.y < 32)
257 seen[start.s][start.x + 8][start.y] = true;
258 for (int i = 0; i < count; ++i) {
259 if (target && same_position(landing(board, kind, bfs[i].p), *target)) {
260 *last_index = i;
261 return count;
262 }
263 for (int op = 0; op < 8; ++op) {
264 Position q;
265 int test;
266 if (!try_move(board, kind, bfs[i].p, op, &q, &test))
267 continue;
268 if (q.s < 0 || q.s >= 4 || q.x < -8 || q.x >= 24 || q.y < 0 || q.y >= 32)
269 continue;
270 if (seen[q.s][q.x + 8][q.y])
271 continue;
272 seen[q.s][q.x + 8][q.y] = true;
273 if (count >= MAX_BFS)
274 break;
275 bfs[count++] = (BFSNode){q, i, op};
276 }
277 }
278 return target ? 0 : count;
279}
280static int reconstruct_ops(int last, int result[MAX_OPERATIONS]) {
281 int reversed[MAX_OPERATIONS], count = 0;
282 while (last >= 0 && bfs[last].parent >= 0 && count < MAX_OPERATIONS) {
283 reversed[count++] = bfs[last].operation;
284 last = bfs[last].parent;
285 }
286 for (int i = 0; i < count; ++i)
287 result[i] = reversed[count - 1 - i];
288 return count;
289}
290static int t_spin(const int board[22], Position p, int test) {
291 Cell corners[4] = {{p.x, p.y}, {p.x + 2, p.y}, {p.x + 2, p.y - 2}, {p.x, p.y - 2}};
292 bool blocked[4];
293 int count = 0;
294 for (int i = 0; i < 4; ++i) {
295 int x = corners[i].x, y = corners[i].y;
296 blocked[i] = x < 1 || x > 10 || y < 1 || y > 22 || (board[y - 1] & (1 << (x - 1))) != 0;
297 if (blocked[i])
298 ++count;
299 }
300 if (count < 3)
301 return 0;
302 return (blocked[p.s] && blocked[(p.s + 1) % 4]) || test == 5 ? 2 : 1;
303}

450static void lock_piece(Game *g) {
451 State *s = &g->state;
452 int kind = s->kind;
453 Position p = s->position;
454 Cell old_cells[4];
455 piece_cells(kind, p, old_cells);
456 int last = 0, route_count = search_routes(s->board, kind, spawn_position(kind), &p, &last);
457 int shortest[MAX_OPERATIONS], shortest_len = route_count ? reconstruct_ops(last, shortest) : -1;
458 int spin = kind == 2 && s->last == 2 ? t_spin(s->board, p, s->test) : 0;
459 int board[22];
460 int lines = place_piece(s->board, kind, p, board);
461 memcpy(s->board, board, sizeof(board));
462 if (g->history_len < MAX_HISTORY)
463 g->history[g->history_len++] = g->start;
464 ++s->pieces;
465 ++s->stats.kinds[kind];
466 s->stats.operations += s->keys;
467 int name, points, sent;
468 clear_info(spin, lines, &name, &points, &sent);
469 int level = s->level;
470 points *= level;
471 bool back = false;
472 if (lines > 0) {
473 ++s->streak;
474 if (lines == 4 || spin) {
475 ++s->chain;
476 back = s->chain >= 2;
477 } else
478 s->chain = 0;
479 } else
480 s->streak = 0;
481 if (back) {
482 points = points * 3 / 2;
483 ++sent;
484 }
485 int combo = s->streak - 1, combo_points = combo >= 1 ? 50 * combo * level : 0;
486 static const int combo_sent[] = {0, 1, 1, 2, 2, 3, 3, 4, 4, 4, 5};
487 if (combo >= 0)
488 sent += combo_sent[combo > 10 ? 10 : combo];
489 bool perfect = lines > 0;
490 if (perfect)
491 for (int i = 0; i < 22; ++i)
492 if (s->board[i]) {
493 perfect = false;
494 break;
495 }
496 int perfect_points = 0;
497 if (perfect) {
498 static const int p_points[] = {800, 1200, 1800, 2000};
499 perfect_points = (lines == 4 && back ? 3200 : p_points[lines - 1]) * level;
500 sent += 10;
501 ++s->stats.perfect;
502 }
503 int gained = points + combo_points + perfect_points;
504 s->score += gained;
505 s->lines += lines;
506 s->stats.sent += sent;
507 if (combo > s->stats.combo)
508 s->stats.combo = combo;
509 if (s->chain > s->stats.chain)
510 s->stats.chain = s->chain;
511 printf("lock %d %c %c %d %d keys %d ", s->pieces, kind_names[kind], "0R2L"[p.s], p.x, p.y,
512 s->keys);
513 if (shortest_len < 0)
514 printf("-");
515 else
516 printf("%d", shortest_len);
517 printf(" drop %d lines %d sent %d +%d total %d\n", s->drop, lines, sent, gained, s->score);
518 if (shortest_len >= 0 && s->keys > shortest_len) {
519 ++s->stats.faults;
520 printf("finesse ");
521 if (shortest_len)
522 print_ops(shortest, shortest_len);
523 else
524 printf("none");
525 puts("");
526 }
527 if (name >= 0) {
528 ++s->stats.clears[name];
529 printf("clear %s %d", clear_names[name], points);
530 if (back)
531 printf(" b2b");
532 if (combo >= 1)
533 printf(" combo %d %d", combo, combo_points);
534 if (perfect)
535 printf(" perfect %d", perfect_points);
536 puts("");
537 }
538 int next_level = s->initial_level + s->lines / 10;
539 if (next_level > level) {
540 s->level = next_level;
541 printf("level %d\n", s->level);
542 }
543 bool lockout = true;
544 for (int i = 0; i < 4; ++i)
545 if (old_cells[i].y < 21)
546 lockout = false;
547 if (lockout) {
548 game_over(g, "lock out");
549 return;
550 }
551 int pending = sum_pending(s), cancelled = sent < pending ? sent : pending;
552 cancel_garbage(s, cancelled);
553 if (cancelled) {
554 s->stats.cancelled += cancelled;
555 printf("cancel %d pending %d\n", cancelled, sum_pending(s));
556 }
557 if (lines == 0) {
558 int risen = 0;
559 while (s->pending_len > 0 && risen < 8) {
560 int count = s->pending[0] < 8 - risen ? s->pending[0] : 8 - risen;
561 s->random = seed_step(s->random);
562 int hole = (int)(s->random % 10) + 1;
563 if (!insert_garbage(g, count, hole))
564 return;
565 s->stats.received += count;
566 risen += count;
567 cancel_garbage(s, count);
568 printf("rise %d hole %d pending %d\n", count, hole, sum_pending(s));
569 }
570 }
571 s->used_hold = false;
572 int next = next_piece(s);
573 spawn_piece(g, next);
574 if (!s->ended)
575 g->start = *s;
576}
577static void operation(Game *g, int op) {
578 State *s = &g->state;
579 if (op >= 0 && op < 8) {
580 ++s->keys;
581 Position p;
582 int test;
583 if (!try_move(s->board, s->kind, s->position, op, &p, &test)) {
584 printf("fail %s\n", operation_names[op]);
585 return;
586 }
587 if (op == 4 || op == 5) {
588 give_drops(s, s->position.y - p.y);
589 update_position(s, p, 1, 0);
590 s->fall = 0;
591 s->ground = 0;
592 } else {
593 update_position(s, p, op >= 6 ? 2 : 1, test);
594 if (s->ground > 0 && s->resets < 15) {
595 s->ground = 0;
596 ++s->resets;
597 }
598 }
599 } else if (op == 8) {
600 Position p = landing(s->board, s->kind, s->position);
601 int rows = s->position.y - p.y;
602 give_drops(s, 2 * rows);
603 if (rows > 0)
604 update_position(s, p, 1, 0);
605 lock_piece(g);
606 } else if (op == 9) {
607 if (grounded(s))
608 lock_piece(g);
609 else
610 puts("fail LOCK");
611 } else {
612 if (s->used_hold) {
613 puts("fail HOLD");
614 return;
615 }
616 s->used_hold = true;
617 ++s->stats.holds;
618 int previous = s->hold;
619 s->hold = s->kind;
620 spawn_piece(g, previous >= 0 ? previous : next_piece(s));
621 }
622}
623static void tick(Game *g) {
624 State *s = &g->state;
625 if (!grounded(s)) {
626 s->ground = 0;
627 ++s->fall;
628 static const int intervals[] = {60, 48, 37, 28, 21, 16, 11, 8, 6, 4, 3, 2};
629 int interval = s->level >= 13 ? 1 : intervals[s->level - 1];
630 if (s->fall >= interval) {
631 Position p = s->position;
632 --p.y;
633 update_position(s, p, 1, 0);
634 s->fall = 0;
635 }
636 } else {
637 ++s->ground;
638 if (s->ground >= 30)
639 lock_piece(g);
640 }
641}
642static void add_candidate(const int board[22], int kind, Position p, const int *prefix,
643 int prefix_len, const int *rot, int rot_len, const int *shift,
644 int shift_len, int start_index, int *count) {
645 Position end = landing(board, kind, p);
646 Cell cells[4];
647 piece_cells(kind, end, cells);
648 int indexes[4];
649 for (int i = 0; i < 4; ++i)
650 indexes[i] = (cells[i].y - 1) * 10 + cells[i].x - 1;
651 qsort(indexes, 4, sizeof(int), compare_int);
652 for (int i = start_index; i < *count; ++i) {
653 bool same = true;
654 for (int j = 0; j < 4; ++j)
655 if (candidates[i].occupied[j] != indexes[j]) {
656 same = false;
657 break;
658 }
659 if (same)
660 return;
661 }
662 if (*count >= MAX_CANDIDATES)
663 return;
664 Candidate *c = &candidates[*count];
665 c->kind = kind;
666 c->count = 0;
667 c->index = *count;
668 memcpy(c->occupied, indexes, sizeof(indexes));
669 for (int i = 0; i < prefix_len && c->count < MAX_OPERATIONS; ++i)
670 c->ops[c->count++] = prefix[i];
671 for (int i = 0; i < rot_len && c->count < MAX_OPERATIONS; ++i)
672 c->ops[c->count++] = rot[i];
673 for (int i = 0; i < shift_len && c->count < MAX_OPERATIONS; ++i)
674 c->ops[c->count++] = shift[i];
675 if (c->count < MAX_OPERATIONS)
676 c->ops[c->count++] = 8;
677 int after[22];
678 int lines = place_piece(board, kind, end, after);
679 int heights[10], holes, bump;
680 c->value = evaluate_well(after, lines, heights, &holes, &bump);
681 ++*count;
682}
683static int append_candidates(const State *s, int kind, Position start, const int *prefix,
684 int prefix_len, int count) {
685 static const int rotations[4][2] = {{-1, -1}, {6, -1}, {6, 6}, {7, -1}};
686 int start_index = count;
687 for (int r = 0; r < 4; ++r) {
688 Position p = start;
689 bool valid = true;
690 int rot[2], rot_len = 0;
691 for (int j = 0; j < 2; ++j)
692 if (rotations[r][j] >= 0) {
693 Position q;
694 int test;
695 if (!try_move(s->board, kind, p, rotations[r][j], &q, &test)) {
696 valid = false;
697 break;
698 }
699 p = q;
700 rot[rot_len++] = rotations[r][j];
701 }
702 if (!valid)
703 continue;
704 add_candidate(s->board, kind, p, prefix, prefix_len, rot, rot_len, NULL, 0, start_index,
705 &count);
706 for (int direction = 0; direction <= 1; ++direction) {
707 Position q = p;
708 int shifts[MAX_OPERATIONS], shift_len = 0;
709 while (shift_len < MAX_OPERATIONS - 5) {
710 Position next;
711 int test;
712 if (!try_move(s->board, kind, q, direction, &next, &test))
713 break;
714 q = next;
715 shifts[shift_len++] = direction;
716 add_candidate(s->board, kind, q, prefix, prefix_len, rot, rot_len, shifts,
717 shift_len, start_index, &count);
718 }
719 }
720 }
721 return count;
722}
723static int compare_candidates(const void *left, const void *right) {
724 const Candidate *a = left, *b = right;
725 if (a->value != b->value)
726 return a->value > b->value ? -1 : 1;
727 return a->index < b->index ? -1 : a->index > b->index ? 1 : 0;
728}
729static int ranked(const State *s) {
730 int count = append_candidates(s, s->kind, s->position, NULL, 0, 0);
731 int held = s->hold >= 0 ? s->hold : s->queue[0];
732 int prefix[] = {10};
733 if (!s->used_hold && fits(s->board, held, spawn_position(held)))
734 count = append_candidates(s, held, spawn_position(held), prefix, 1, count);
735 qsort(candidates, (size_t)count, sizeof(Candidate), compare_candidates);
736 return count;
737}
738static int compare_spin(const void *left, const void *right) {
739 const SpinCandidate *a = left, *b = right;
740 if (a->p.y != b->p.y)
741 return a->p.y - b->p.y;
742 if (a->p.x != b->p.x)
743 return a->p.x - b->p.x;
744 return a->p.s - b->p.s;
745}
746static void command_spins(const State *s) {
747 if (s->kind != 2) {
748 puts("spins 0");
749 return;
750 }
751 int count = search_routes(s->board, 2, s->position, NULL, NULL), found = 0;
752 for (int i = 0; i < count; ++i) {
753 Position start = bfs[i].p;
754 for (int op = 6; op <= 7; ++op) {
755 Position p, down;
756 int test, unused;
757 if (!try_move(s->board, 2, start, op, &p, &test))
758 continue;
759 if (try_move(s->board, 2, p, 4, &down, &unused))
760 continue;
761 int spin = t_spin(s->board, p, test);
762 if (!spin)
763 continue;
764 int old = -1;
765 for (int j = 0; j < found; ++j)
766 if (same_position(spins[j].p, p)) {
767 old = j;
768 break;
769 }
770 if (old >= 0) {
771 if (spin > spins[old].spin)
772 spins[old].spin = spin;
773 } else if (found < MAX_BFS)
774 spins[found++] = (SpinCandidate){p, spin};
775 }
776 }
777 qsort(spins, (size_t)found, sizeof(SpinCandidate), compare_spin);
778 for (int i = 0; i < found; ++i) {
779 int after[22], name, points, sent;
780 int lines = place_piece(s->board, 2, spins[i].p, after);
781 clear_info(spins[i].spin, lines, &name, &points, &sent);
782 printf("spin %c %d %d %s\n", "0R2L"[spins[i].p.s], spins[i].p.x, spins[i].p.y,
783 clear_names[name]);
784 }
785 printf("spins %d\n", found);
786}
787static void print_board(const State *s) {
788 bool current[22][10] = {0}, ghost[22][10] = {0};
789 if (!s->ended) {
790 Cell c[4], g[4];
791 piece_cells(s->kind, s->position, c);
792 piece_cells(s->kind, landing(s->board, s->kind, s->position), g);
793 for (int i = 0; i < 4; ++i) {
794 current[c[i].y - 1][c[i].x - 1] = true;
795 ghost[g[i].y - 1][g[i].x - 1] = true;
796 }
797 }
798 for (int r = 22; r >= 1; --r) {
799 printf("%2d ", r);
800 for (int c = 1; c <= 10; ++c) {
801 char ch = s->board[r - 1] & (1 << (c - 1)) ? '#'
802 : current[r - 1][c - 1] ? '@'
803 : ghost[r - 1][c - 1] ? '+'
804 : '.';
805 putchar(ch);
806 }
807 puts("");
808 }

1010static bool valid_number(const char *s, int min, int max) {
1011 int64_t n = 0;
1012 if (!s || !*s || (s[0] == '0' && s[1]))
1013 return false;
1014 for (const char *p = s; *p; ++p) {
1015 if (!isdigit((unsigned char)*p))
1016 return false;
1017 n = n * 10 + (*p - '0');
1018 if (n > max)
1019 return false;
1020 }
1021 return n >= min;
1022}
1023static bool valid_arguments(int command_id, int argc, char *argv[]) {
1024 if (command_id == 0)
1025 return argc == 2 && valid_number(argv[0], 1, 2147483646) && valid_number(argv[1], 1, 15);
1026 if (command_id == 6) {
1027 if (argc != 2 || !valid_number(argv[0], 1, 22) || strlen(argv[1]) != 10)
1028 return false;
1029 bool empty = false;
1030 for (int i = 0; i < 10; ++i) {
1031 if (argv[1][i] == '.')
1032 empty = true;
1033 else if (argv[1][i] != '#')
1034 return false;
1035 }
1036 return empty;
1037 }
1038 if (command_id == 7)
1039 return argc == 2 && valid_number(argv[0], 1, 10) && valid_number(argv[1], 1, 10);
1040 if (command_id == 11 || command_id == 12) {
1041 if (argc != 1) {
1042 return false;
1043 }
1044 size_t len = strlen(argv[0]);
1045 if (len < 1 || len > 8)
1046 return false;
1047 for (size_t i = 0; i < len; ++i)
1048 if (argv[0][i] < 'a' || argv[0][i] > 'z')
1049 return false;
1050 return true;
1051 }
1052 int limit = command_id == 3 ? 14
1053 : command_id == 8 ? 20
1054 : command_id == 9 ? 1000
1055 : command_id == 14 ? 10
1056 : command_id == 15 ? 50
1057 : 0;
1058 if (limit)
1059 return argc == 1 && valid_number(argv[0], 1, limit);
1060 return argc == 0;
1061}
