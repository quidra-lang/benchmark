1ALGORITHMS = %w[FCFS SJF SRTF RR PRIORITY HRRN EDF LOTTERY MLFQ].freeze
2KEYS = %w[cs q k s seed].freeze
3RANGES = {'cs'=>0..5,'q'=>1..20,'k'=>1..50,'s'=>0..500,'seed'=>1..2147483646}.freeze
4DEFAULTS = {'cs'=>0,'q'=>4,'k'=>5,'s'=>0,'seed'=>1}.freeze
5METRICS = %w[turnaround waiting response].freeze
6EVENTS = %w[PREEMPT EXPIRE DEMOTE BOOST AGE].freeze
7SYMBOLS = {'FUTURE'=>'-','DONE'=>'-','RUN'=>'#','SWITCH'=>'s','READY'=>'.','IO'=>'i'}.freeze
8class CommandError < StandardError
9end
10def allowed(a) = ['cs'] + ({'RR'=>['q'],'LOTTERY'=>%w[q seed],'PRIORITY'=>['k'],'MLFQ'=>['s']}[a] || [])
11def notation(p, a = nil) = (a ? allowed(a) : KEYS).map { |k| "#{k}=#{p[k]}" }.join(' ')
12def truncated(u, v)
13 n = u * 100 / v
14 "#{n / 100}.#{(n % 100).to_s.rjust(2, '0')}"
15end
16def fail_if(bad, reason) = (raise CommandError, reason if bad)
17def params(words, a = nil, preset = [])
18 p = DEFAULTS.dup
19 seen = preset.dup
20 words.each do |word|
21 k, text = word.split('=')
22 v = text.to_i
23 fail_if(!RANGES.key?(k), "unknown key #{k}")
24 fail_if(a && !allowed(a).include?(k), "key #{k} not allowed for #{a}")
25 fail_if(seen.include?(k), "duplicate key #{k}")
26 fail_if(!RANGES[k].include?(v), "invalid #{k}")
27 seen << k
28 p[k] = v
29 end
30 p
31end
32ProcessDef = Struct.new(:id, :arrival, :priority, :bursts, :deadline) do
33 def describe = "#{id} arrival=#{arrival} priority=#{priority} deadline=#{deadline || 'none'} bursts=#{bursts.each_with_index.map { |n, i| "#{i.even? ? 'C' : 'I'}#{n}" }.join(',')}"
34end
35Burst = Struct.new(:length, :ready, :start, :finish, :slices)
36Job = Struct.new(:process, :state, :burst, :left, :effective, :wait, :level, :used, :enqueued, :start, :finish, :records, :timeline) do
37 def id = process.id
38 def deadline = process.deadline && process.arrival + process.deadline
39 def metrics
40 turn = finish - process.arrival
41 [turn, turn - process.bursts.sum, start - process.arrival]
42 end
43 def slices = records.sum(&:slices)
44 def missed? = deadline && finish > deadline
45end
46class Simulation
47 attr_reader :jobs, :events, :counts, :labels, :states, :lengths, :snapshot, :time, :switches
48 def initialize(processes, algo, p, snapshot_time = nil)
49 @a, @p, @snapshot_time = algo, p, snapshot_time
50 @jobs = processes.sort_by(&:id).map { |d| Job.new(d, 'FUTURE', 0, d.bursts[0], d.priority, 0, 0, 0, 0, nil, nil, [], '') }
51 @queues = [[], [], []]
52 @cpu = 'IDLE'
53 @current = @previous = nil
54 @switch_left = @run = @switches = 0
55 @random = p['seed']
56 @events = []
57 @counts = Hash.new(0)
58 @labels, @states, @lengths = [], [], []
59 @time = 0
60 simulate
61 end
62 def record(kind, *details)
63 @events << [@time, kind, *details].join(' ')
64 @counts[kind] += 1
65 end
66 def enqueue(j)
67 j.state = 'READY'
68 j.enqueued = @time
69 @queues[j.level] << j
70 end
71 def advance(j) = (j.burst += 1; j.left = j.process.bursts[j.burst])
72 def release = (@cpu = 'IDLE'; @current = nil)
73 def start(j)
74 @cpu, @current, @run, j.state = 'RUN', j, 0, 'RUN'
75 j.start ||= @time
76 j.records.last.start ||= @time
77 end
78 def preempts?(j)
79 case @a
80 when 'SRTF'
81 @queues[0].any? { |x| x.left < j.left }
82 when 'PRIORITY'
83 @queues[0].any? { |x| x.effective < j.effective }
84 when 'EDF'
85 @queues[0].any? { |x| x.deadline && (!j.deadline || x.deadline < j.deadline) }
86 when 'MLFQ'
87 @queues.take(j.level).any? { |q| !q.empty? }
88 else
89 false
90 end
91 end
92 def choose(q)
93 case @a
94 when 'SJF', 'SRTF'
95 q.min_by(&:left)
96 when 'PRIORITY'
97 q.min_by(&:effective)
98 when 'EDF'
99 q.select(&:deadline).min_by(&:deadline) || q.first
100 when 'HRRN'
101 q.reduce { |b, j| (@time - j.enqueued + j.left) * b.left > (@time - b.enqueued + b.left) * j.left ? j : b }
102 when 'LOTTERY'
103 @random = 48271 * @random % 2147483647
104 target = @random % q.sum { |j| 10 - j.process.priority }
105 q.find { |j| target -= 10 - j.process.priority
106 target < 0 }
107 else
108 q.first
109 end
110 end
111 def simulate
112 loop do
113 if @a == 'PRIORITY'
114 @queues[0].each do |j|
115 j.wait += 1
116 if j.wait == @p['k']
117 j.wait = 0
118 if j.effective > 0
119 j.effective -= 1
120 record('AGE', j.id, j.effective)
121 end
122 end
123 end
124 end
125 @jobs.each do |j|
126 if j.state == 'IO' && j.left.zero?
127 advance(j)
128 j.records << Burst.new(j.left, @time, nil, nil, 0)
129 enqueue(j)
130 record('IODONE', j.id)
131 end
132 end
133 @jobs.each do |j|
134 if j.state == 'FUTURE' && j.process.arrival == @time
135 j.records << Burst.new(j.left, @time, nil, nil, 0)
136 enqueue(j)
137 record('ARRIVE', j.id)
138 end
139 end
140 j = @current
141 if @cpu == 'SWITCH' && @switch_left.zero?
142 start(j)
143 record('START', j.id)
144 elsif @cpu == 'RUN'
145 if j.left.zero? && j.burst == j.process.bursts.length - 1
146 j.state, j.finish = 'DONE', @time
147 record('DONE', j.id)
148 release
149 else
150 demoted = @a == 'MLFQ' && j.level < 2 && j.used == [4, 8][j.level]
151 if demoted
152 j.level += 1
153 j.used = 0
154 record('DEMOTE', j.id, j.level)
155 end
156 if j.left.zero?
157 advance(j)
158 j.state = 'IO'
159 record('IO', j.id, j.left)
160 release
161 elsif demoted
162 enqueue(j)
163 release
164 elsif %w[RR LOTTERY].include?(@a) && @run == @p['q']
165 enqueue(j)
166 record('EXPIRE', j.id)
167 release
168 elsif preempts?(j)
169 enqueue(j)
170 record('PREEMPT', j.id)
171 release
172 end
173 end
174 end
175 break if @jobs.all? { |x| x.state == 'DONE' }
176 if @a == 'MLFQ' && @p['s'] > 0 && @time > 0 && @time % @p['s'] == 0
177 @queues = [@queues.flatten, [], []]
178 @jobs.each { |x| x.level = x.used = 0 }
179 record('BOOST')
180 end
181 if @cpu == 'IDLE' && @queues.any? { |q| !q.empty? }
182 q = @queues.find { |queue| !queue.empty? }
183 j = choose(q)
184 q.delete(j)
185 j.effective, j.wait = j.process.priority, 0
186 j.records.last.slices += 1
187 switched = j != @previous
188 @switches += 1 if switched
189 if switched && @p['cs'] > 0
190 @cpu, @current, @switch_left, j.state = 'SWITCH', j, @p['cs'], 'SWITCH'
191 record('SWITCH', j.id)
192 else
193 start(j)
194 record('DISPATCH', j.id)
195 end
196 end
197 capture if @time == @snapshot_time
198 @jobs.each { |x| x.timeline << SYMBOLS[x.state] }
199 @lengths << @queues.sum(&:length)
200 @states << @cpu
201 if @cpu == 'RUN'
202 j = @current
203 j.left -= 1
204 j.used += 1
205 @run += 1
206 @previous = j
207 @labels << j.id
208 j.records.last.finish = @time + 1 if j.left.zero?
209 elsif @cpu == 'SWITCH'
210 @switch_left -= 1
211 @labels << 'CS'
212 else
213 @labels << 'IDLE'
214 end
215 @jobs.each { |x| x.left -= 1 if x.state == 'IO' }
216 @time += 1
217 end
218 capture unless @snapshot
219 end
220 def capture
221 @snapshot = [@cpu == 'IDLE' ? 'CPU IDLE' : "CPU #{@cpu} #{@current.id}"]
222 (@a == 'MLFQ' ? (0..2) : [0]).each do |i|
223 @snapshot << "QUEUE#{@a == 'MLFQ' ? i : ''} #{@queues[i].empty? ? 'none' : @queues[i].map(&:id).join(' ')}"
224 end
225 @jobs.each do |j|
226 text = if j.state == 'FUTURE'
227 "#{j.id} FUTURE at=#{j.process.arrival}"
228 elsif j.state == 'DONE'
229 "#{j.id} DONE at=#{j.finish}"
230 else
231 s = "#{j.id} #{j.state} burst=#{j.burst + 1} left=#{j.left}"
232 s += " prio=#{j.effective}" if @a == 'PRIORITY'
233 s += " level=#{j.level} used=#{j.used}" if @a == 'MLFQ'
234 s
235 end
236 @snapshot << text
237 end
238 end
239 def totals = (0..2).map { |i| @jobs.sum { |j| j.metrics[i] } }
240 def averages = totals.map { |n| truncated(n, @jobs.length) }
241 def utilization = truncated(@states.count('RUN') * 100, @time)
242 def missed = @jobs.count(&:missed?)
243 def met = @jobs.count { |j| j.deadline && !j.missed? }
244 def ticks(a, b) = %w[RUN SWITCH IDLE].map { |s| @states[a...b].count(s) }
245 def segments
246 result = []
247 start = 0
248 (1..@time).each do |t|
249 if t == @time || @labels[t] != @labels[start]
250 result << "[#{start}-#{t} #{@labels[start]}]"
251 start = t
252 end
253 end
254 result
255 end
256end
257class Scheduler
258 def initialize = (@registry = {})
259 def emit(*parts) = puts(parts.join(' '))
260 def algorithm(a) = fail_if(!ALGORITHMS.include?(a), "unknown algorithm #{a}")
261 def process(id) = @registry.fetch(id) { raise CommandError, "unknown process #{id}" }
262 def prepare(words, a = nil, preset = [])
263 p = params(words, a, preset)
264 fail_if(@registry.empty?, 'no processes')
265 p
266 end
267 def simulate(a, p, t = nil) = Simulation.new(@registry.values, a, p, t)
268 def table(cells) = cells[0].ljust(8) + cells.drop(1).map { |v| ' ' + v.to_s.rjust(8) }.join
269 def execute(line)
270 c, *a = line.split
271 case c
272 when 'PROC'
273 id, arrival, priority, *words = a
274 at, pr = arrival.to_i, priority.to_i
275 fail_if(@registry.key?(id), "duplicate process #{id}")
276 fail_if(@registry.length >= 10, 'too many processes')
277 fail_if(at > 200, "invalid arrival #{id}")
278 fail_if(pr > 9, "invalid priority #{id}")
279 fail_if(words.length.even? || words.each_with_index.any? { |w, i| w[0] != (i.even? ? 'C' : 'I') || !(1..30).include?(w[1..].to_i) }, "invalid bursts #{id}")
280 bs = words.map { |w| w[1..].to_i }
281 @registry[id] = ProcessDef.new(id, at, pr, bs, nil)
282 emit(c, id, "cpu=#{bs.each_with_index.sum { |v, i| i.even? ? v : 0 }}", "io=#{bs.each_with_index.sum { |v, i| i.odd? ? v : 0 }}")
283 when 'GEN'
284 prefix, count, seed = a
285 ids = (1..count.to_i).map { |i| prefix + i.to_s }
286 ids.each { |id| fail_if(@registry.key?(id), "duplicate process #{id}") }
287 fail_if(@registry.length + ids.length > 10, 'too many processes')
288 emit(c, "count=#{ids.length}")
289 x = seed.to_i
290 draw = ->(m) { x = 48271 * x % 2147483647
291 x % m }
292 ids.each do |id|
293 at, pr, m = draw.call(21), draw.call(10), 1 + draw.call(3)
294 bs = []
295 m.times { |i| bs << 1 + draw.call(8) if i > 0
296 bs << 1 + draw.call(9) }
297 @registry[id] = ProcessDef.new(id, at, pr, bs, nil)
298 emit(@registry[id].describe)
299 end
300 when 'SETPRIO', 'DEADLINE'
301 d = process(a[0])
302 n = a[1].to_i
303 fail_if(n > (c == 'SETPRIO' ? 9 : 1000), "invalid #{c == 'SETPRIO' ? 'priority' : 'deadline'} #{d.id}")
304 if c == 'SETPRIO'
305 d.priority = n
306 emit(c, d.id, "priority=#{n}")
307 else
308 d.deadline = n.zero? ? nil : n
309 emit(c, d.id, d.deadline || 'none')
310 end
311 when 'DROP'
312 process(a[0])
313 @registry.delete(a[0])
314 emit(c, a[0])
315 when 'CLEAR'
316 emit(c, "count=#{@registry.length}")
317 @registry.clear
318 when 'LIST'
319 emit(c, "count=#{@registry.length}")
320 @registry.keys.sort.each { |id| emit(@registry[id].describe) }
321 when 'COMPARE', 'DIFF', 'RANK'
322 first = second = metric = nil
323 if c == 'DIFF'
324 first, second = a.shift(2)
325 algorithm(first)
326 algorithm(second)
327 elsif c == 'RANK'
328 metric = a.shift
329 fail_if(!(METRICS + %w[end switches missed]).include?(metric), "unknown metric #{metric}")
330 end
331 p = prepare(a)
332 emit(c, *[first, second, metric].compact, notation(p))
333 algos = c == 'DIFF' ? [first, second] : ALGORITHMS
334 sims = algos.map { |algo| simulate(algo, p) }
335 if c == 'COMPARE'
336 emit(table(%w[ALGO TURN WAIT RESP UTIL END SWITCH MISS]))
337 sims.each_with_index { |s, i| emit(table([algos[i], *s.averages, s.utilization, s.time, s.switches, s.missed])) }
338 emit('BEST', *METRICS.each_with_index.map { |m, i| "#{m}=#{algos.each_index.min_by { |j| sims[j].totals[i] }.then { |j| algos[j] }}" })
339 elsif c == 'DIFF'
340 wins = [0, 0, 0]
341 sims[0].jobs.zip(sims[1].jobs).each do |j, k|
342 x, y = j.metrics, k.metrics
343 emit(j.id, *METRICS.each_with_index.map { |m, i| "#{m}=#{x[i]}/#{y[i]}" })
344 wins[x[0] < y[0] ? 0 : x[0] > y[0] ? 1 : 2] += 1
345 end
346 emit('WINS', "#{first}=#{wins[0]}", "#{second}=#{wins[1]}", "ties=#{wins[2]}")
347 else
348 idx = METRICS.index(metric)
349 values = sims.map { |s| idx ? s.totals[idx] : metric == 'end' ? s.time : metric == 'switches' ? s.switches : s.missed }
350 algos.each_index.sort_by { |i| [values[i], i] }.each { |i| emit(1 + values.count { |v| v < values[i] }, algos[i], idx ? truncated(values[i], @registry.length) : values[i]) }
351 end
352 else
353 algo = a.shift
354 algorithm(algo)
355 extra = nil
356 preset = []
357 if c == 'WINDOW'
358 extra = a.shift.to_i
359 fail_if(!(1..1000).include?(extra), 'invalid window')
360 elsif c == 'SNAPSHOT'
361 extra = a.shift.to_i
362 elsif c == 'SWEEP'
363 key, lo, hi = a.shift(3)
364 lo, hi = lo.to_i, hi.to_i
365 fail_if(!RANGES.key?(key), "unknown key #{key}")
366 fail_if(!allowed(algo).include?(key), "key #{key} not allowed for #{algo}")
367 fail_if(lo > hi || !RANGES[key].include?(lo) || !RANGES[key].include?(hi) || hi - lo > 20, 'invalid range')
368 preset = [key]
369 end
370 p = prepare(a, algo, preset)
371 if c == 'SWEEP'
372 emit(c, algo, key, lo, hi)
373 best = nil
374 best_value = lo
375 (lo..hi).each do |v|
376 p[key] = v
377 s = simulate(algo, p)
378 av = s.averages
379 emit(notation(p, algo), *METRICS.each_with_index.map { |m, i| "#{m}=#{av[i]}" }, "util=#{s.utilization}", "end=#{s.time}", "switches=#{s.switches}")
380 if best.nil? || s.totals[1] < best
381 best, best_value = s.totals[1], v
382 end
383 end
384 emit('BEST', "#{key}=#{best_value}")
385 return
386 end
387 s = simulate(algo, p, c == 'SNAPSHOT' ? extra : nil)
388 fail_if(c == 'TIMELINE' && s.time > 100, "too long #{s.time}")
389 suffix = c == 'TIMELINE' ? ["end=#{s.time}"] : c == 'SNAPSHOT' ? ["t=#{extra}"] : c == 'WINDOW' ? ["width=#{extra}"] : []
390 emit(c, algo, notation(p, algo), *suffix)
391 case c
392 when 'TRACE'
393 s.events.each { |v| emit(v) }
394 emit('END', s.time)
395 when 'SNAPSHOT'
396 s.snapshot.each { |v| emit(v) }
397 when 'TIMELINE'
398 s.jobs.each { |j| emit(j.id.ljust(8), j.timeline) }
399 when 'BURSTS'
400 s.jobs.each { |j| j.records.each_with_index { |r, i| emit(j.id, "##{i + 1}", "len=#{r.length}", "ready=#{r.ready}", "start=#{r.start}", "end=#{r.finish}", "wait=#{r.finish - r.ready - r.length}", "slices=#{r.slices}") } }
401 when 'WINDOW'
402 (0...s.time).step(extra) do |start|
403 stop = [start + extra, s.time].min
404 b, cs, idle = s.ticks(start, stop)
405 counts = Hash.new(0)
406 (start...stop).each { |t| counts[s.labels[t]] += 1 if s.states[t] == 'RUN' }
407 top = counts.keys.min_by { |id| [-counts[id], id] } || 'none'
408 emit("#{start}-#{stop}", "busy=#{b}", "switch=#{cs}", "idle=#{idle}", "util=#{truncated(b * 100, stop - start)}", "top=#{top}")
409 end
410 when 'STATS'
411 METRICS.each_with_index do |m, i|
412 vs = s.jobs.map { |j| j.metrics[i] }
413 sorted = vs.sort
414 n = vs.length
415 doubled = sorted[(n - 1) / 2] + sorted[n / 2]
416 emit(m, "min=#{vs.min}", "(#{s.jobs[vs.index(vs.min)].id})", "max=#{vs.max}", "(#{s.jobs[vs.index(vs.max)].id})", "median=#{doubled / 2}.#{doubled.odd? ? 5 : 0}", "total=#{vs.sum}")
417 end
418 when 'RUN'
419 s.segments.each_slice(8) { |g| emit('GANTT', g.join) }
420 s.jobs.each do |j|
421 t, w, r = j.metrics
422 text = "#{j.id} arrival=#{j.process.arrival} done=#{j.finish} turnaround=#{t} waiting=#{w} response=#{r} slices=#{j.slices}"
423 text += " deadline=#{j.deadline} late=#{[0, j.finish - j.deadline].max}" if j.deadline
424 emit(text)
425 end
426 emit('AVERAGE', *METRICS.zip(s.averages).map { |m, v| "#{m}=#{v}" })
427 b, cs, idle = s.ticks(0, s.time)
428 emit('CPU', "end=#{s.time}", "busy=#{b}", "switch=#{cs}", "idle=#{idle}", "switches=#{s.switches}", "util=#{s.utilization}")
429 emit('QUEUE', "max=#{s.lengths.max}", "avg=#{truncated(s.lengths.sum, s.time)}")
430 emit('EVENTS', *EVENTS.map { |k| "#{k.downcase}=#{s.counts[k]}" })
431 emit('DEADLINE', "met=#{s.met}", "missed=#{s.missed}")
432 end
433 end
434 rescue CommandError => e
435 emit('ERROR', e.message)
436 end
437end
438scheduler = Scheduler.new
439STDIN.gets.to_i.times { scheduler.execute(STDIN.gets) }
