1private val ALGOS = listOf("FCFS", "SJF", "SRTF", "RR", "PRIORITY", "HRRN", "EDF", "LOTTERY", "MLFQ")
2private val KEYS = listOf("cs", "q", "k", "s", "seed")
3private val METRICS = listOf("turnaround", "waiting", "response")
4private val EVENTS = listOf("PREEMPT", "EXPIRE", "DEMOTE", "BOOST", "AGE")
5private val RANGES = mapOf("cs" to 0L..5L, "q" to 1L..20L, "k" to 1L..50L, "s" to 0L..500L, "seed" to 1L..2147483646L)
6private class Failure(message: String) : RuntimeException(message)
7private fun checkBad(bad: Boolean, message: String) {
8 if (bad) throw Failure(message)
9}
10private fun ratio(u: Long, v: Int): String {
11 val n = u * 100 / v
12 return "${n / 100}.${(n % 100).toString().padStart(2, '0')}"
13}
14private fun allowed(a: String) = when (a) {
15 "RR" -> listOf("cs", "q")
16 "PRIORITY" -> listOf("cs", "k")
17 "MLFQ" -> listOf("cs", "s")
18 "LOTTERY" -> listOf("cs", "q", "seed")
19 else -> listOf("cs")
20}
21private fun parameters(words: List<String>, algo: String?, preset: String? = null): MutableMap<String, Long> {
22 val p = linkedMapOf("cs" to 0L, "q" to 4L, "k" to 5L, "s" to 0L, "seed" to 1L)
23 val seen = mutableSetOf<String>()
24 if (preset != null) seen.add(preset)
25 for (word in words) {
26 val (k, text) = word.split('=')
27 val v = text.toLong()
28 checkBad(k !in KEYS, "unknown key $k")
29 checkBad(algo != null && k !in allowed(algo), "key $k not allowed for $algo")
30 checkBad(k in seen, "duplicate key $k")
31 checkBad(v !in RANGES.getValue(k), "invalid $k")
32 seen.add(k)
33 p[k] = v
34 }
35 return p
36}
37private fun notation(p: Map<String, Long>, algo: String? = null) = (if (algo == null) KEYS else allowed(algo)).joinToString(" ") {
38 "$it=${p.getValue(it)}"
39}
40private class RandomState(var x: Long) {
41 fun draw(n: Int): Int {
42 x = 48271 * x % 2147483647
43 return (x % n).toInt()
44 }
45}
46private class Process(val id: String, val arrival: Int, var priority: Int, val bursts: List<Int>) {
47 var deadline = 0
48 fun absolute() = if (deadline == 0) Int.MAX_VALUE else arrival + deadline
49 fun describe() = "$id arrival=$arrival priority=$priority deadline=${if (deadline == 0) "none" else deadline} bursts=${bursts.mapIndexed { i, b -> "${if (i % 2 == 0) "C" else "I"}$b" }.joinToString(",")}"
50}
51private class Burst(val length: Int, val ready: Int) {
52 var start = -1
53 var end = 0
54 var slices = 0
55}
56private class Job(val p: Process) {
57 val id get() = p.id
58 var state = "FUTURE"
59 var burst = 0
60 var left = p.bursts[0]
61 var effective = p.priority
62 var wait = 0
63 var level = 0
64 var used = 0
65 var enqueued = 0
66 var start = -1
67 var finish = 0
68 val records = mutableListOf<Burst>()
69 val timeline = StringBuilder()
70 fun advance() {
71 left = p.bursts[++burst]
72 }
73 fun slices() = records.sumOf {
74 it.slices
75 }
76 fun metrics(): List<Long> {
77 val turn = finish - p.arrival
78 return listOf(turn.toLong(), (turn - p.bursts.sum()).toLong(), (start - p.arrival).toLong())
79 }
80 fun missed() = p.deadline > 0 && finish > p.absolute()
81}
82private class Simulation(processes: Collection<Process>, val algo: String, val p: Map<String, Long>, val snapshotTime: Int = -1) {
83 val jobs = processes.sortedBy {
84 it.id
85 }.map {
86 Job(it)
87 }
88 val queues = List(3) {
89 mutableListOf<Job>()
90 }
91 var cpu = "IDLE"
92 var current: Job? = null
93 var previous: Job? = null
94 var time = 0
95 var switchLeft = 0
96 var runLength = 0
97 var switches = 0
98 val random = RandomState(p.getValue("seed"))
99 val events = mutableListOf<String>()
100 val labels = mutableListOf<String>()
101 val cpuStates = mutableListOf<String>()
102 val lengths = mutableListOf<Int>()
103 val counts = mutableMapOf<String, Int>()
104 var snapshot: List<String>? = null
105 init {
106 run()
107 }
108 fun event(kind: String, vararg details: Any) {
109 events.add((listOf(time, kind) + details).joinToString(" "))
110 counts[kind] = counts.getOrDefault(kind, 0) + 1
111 }
112 fun enqueue(j: Job) {
113 j.state = "READY"
114 j.enqueued = time
115 queues[if (algo == "MLFQ") j.level else 0].add(j)
116 }
117 fun release() {
118 cpu = "IDLE"
119 current = null
120 }
121 fun start(j: Job) {
122 current = j
123 cpu = "RUN"
124 j.state = "RUN"
125 runLength = 0
126 if (j.start < 0) j.start = time
127 if (j.records.last().start < 0) j.records.last().start = time
128 }
129 fun preempts(j: Job): Boolean = when (algo) {
130 "SRTF" -> queues[0].any {
131 it.left < j.left
132 }
133 "PRIORITY" -> queues[0].any {
134 it.effective < j.effective
135 }
136 "EDF" -> queues[0].any {
137 it.p.absolute() < j.p.absolute()
138 }
139 "MLFQ" -> queues.take(j.level).any {
140 it.isNotEmpty()
141 }
142 else -> false
143 }
144 fun choose(q: List<Job>): Job {
145 var best = q.first()
146 if (algo == "LOTTERY") {
147 var r = random.draw(q.sumOf {
148 10 - it.p.priority
149 })
150 for (j in q) {
151 r -= 10 - j.p.priority
152 if (r < 0) return j
153 }
154 }
155 for (j in q) {
156 val better = when (algo) {
157 "SJF", "SRTF" -> j.left < best.left
158 "PRIORITY" -> j.effective < best.effective
159 "EDF" -> j.p.absolute() < best.p.absolute()
160 "HRRN" -> (time - j.enqueued + j.left).toLong() * best.left > (time - best.enqueued + best.left).toLong() * j.left
161 else -> false
162 }
163 if (better) best = j
164 }
165 return best
166 }
167 fun run() {
168 while (true) {
169 if (algo == "PRIORITY") for (j in queues[0]) {
170 j.wait++
171 if (j.wait.toLong() == p.getValue("k")) {
172 j.wait = 0
173 if (j.effective > 0) {
174 j.effective--
175 event("AGE", j.id, j.effective)
176 }
177 }
178 }
179 for (j in jobs) if (j.state == "IO" && j.left == 0) {
180 j.advance()
181 j.records.add(Burst(j.left, time))
182 enqueue(j)
183 event("IODONE", j.id)
184 }
185 for (j in jobs) if (j.state == "FUTURE" && j.p.arrival == time) {
186 j.records.add(Burst(j.left, time))
187 enqueue(j)
188 event("ARRIVE", j.id)
189 }
190 if (cpu == "SWITCH" && switchLeft == 0) {
191 start(current!!)
192 event("START", current!!.id)
193 }
194 else if (cpu == "RUN") {
195 val j = current!!
196 if (j.left == 0 && j.burst == j.p.bursts.lastIndex) {
197 j.state = "DONE"
198 j.finish = time
199 event("DONE", j.id)
200 release()
201 }
202 else {
203 val demoted = algo == "MLFQ" && j.level < 2 && j.used == (if (j.level == 0) 4 else 8)
204 if (demoted) {
205 j.level++
206 j.used = 0
207 event("DEMOTE", j.id, j.level)
208 }
209 when {
210 j.left == 0 -> {
211 j.advance()
212 j.state = "IO"
213 event("IO", j.id, j.left)
214 release()
215 }
216 demoted -> {
217 enqueue(j)
218 release()
219 }
220 algo in listOf("RR", "LOTTERY") && runLength.toLong() == p.getValue("q") -> {
221 enqueue(j)
222 event("EXPIRE", j.id)
223 release()
224 }
225 preempts(j) -> {
226 enqueue(j)
227 event("PREEMPT", j.id)
228 release()
229 }
230 }
231 }
232 }
233 if (jobs.all {
234 it.state == "DONE"
235 }) break
236 val interval = p.getValue("s")
237 if (algo == "MLFQ" && interval > 0 && time > 0 && time % interval == 0L) {
238 queues[0].addAll(queues[1])
239 queues[0].addAll(queues[2])
240 queues[1].clear()
241 queues[2].clear()
242 jobs.forEach {
243 it.level = 0
244 it.used = 0
245 }
246 event("BOOST")
247 }
248 if (cpu == "IDLE") {
249 val q = queues.firstOrNull {
250 it.isNotEmpty()
251 }
252 if (q != null) {
253 val j = if (algo == "MLFQ") q.first() else choose(q)
254 q.remove(j)
255 j.effective = j.p.priority
256 j.wait = 0
257 j.records.last().slices++
258 val switched = j !== previous
259 if (switched) switches++
260 if (switched && p.getValue("cs") > 0) {
261 current = j
262 cpu = "SWITCH"
263 j.state = "SWITCH"
264 switchLeft = p.getValue("cs").toInt()
265 event("SWITCH", j.id)
266 }
267 else {
268 start(j)
269 event("DISPATCH", j.id)
270 }
271 }
272 }
273 if (time == snapshotTime) capture()
274 for (j in jobs) j.timeline.append(when (j.state) {
275 "READY" -> '.'
276 "SWITCH" -> 's'
277 "RUN" -> '#'
278 "IO" -> 'i'
279 else -> '-'
280 })
281 lengths.add(queues.sumOf {
282 it.size
283 })
284 cpuStates.add(cpu)
285 when (cpu) {
286 "RUN" -> {
287 val j = current!!
288 j.left--
289 j.used++
290 runLength++
291 previous = j
292 labels.add(j.id)
293 if (j.left == 0) j.records.last().end = time + 1
294 }
295 "SWITCH" -> {
296 switchLeft--
297 labels.add("CS")
298 }
299 else -> labels.add("IDLE")
300 }
301 for (j in jobs) if (j.state == "IO") j.left--
302 time++
303 }
304 if (snapshot == null) capture()
305 }
306 fun capture() {
307 val lines = mutableListOf("CPU $cpu" + (current?.let {
308 " ${it.id}"
309 } ?: ""))
310 for (i in 0 until (if (algo == "MLFQ") 3 else 1)) lines.add("QUEUE${if (algo == "MLFQ") i else ""} ${queues[i].joinToString(" ") { it.id }.ifEmpty { "none" }}")
311 for (j in jobs) {
312 var s = "${j.id} ${j.state}"
313 s += when (j.state) {
314 "FUTURE" -> " at=${j.p.arrival}"
315 "DONE" -> " at=${j.finish}"
316 else -> " burst=${j.burst + 1} left=${j.left}" + if (algo == "PRIORITY") " prio=${j.effective}" else if (algo == "MLFQ") " level=${j.level} used=${j.used}" else ""
317 }
318 lines.add(s)
319 }
320 snapshot = lines
321 }
322 fun totals() = List(3) {
323 i -> jobs.sumOf {
324 it.metrics()[i]
325 }
326 }
327 fun averages(): String {
328 val t = totals()
329 return "turnaround=${ratio(t[0], jobs.size)} waiting=${ratio(t[1], jobs.size)} response=${ratio(t[2], jobs.size)}"
330 }
331 fun ticks(a: Int, b: Int) = listOf("RUN", "SWITCH", "IDLE").map {
332 state -> cpuStates.subList(a, b).count {
333 it == state
334 }
335 }
336 fun util() = ratio(ticks(0, time)[0] * 100L, time)
337 fun missed() = jobs.count {
338 it.missed()
339 }
340 fun met() = jobs.count {
341 it.p.deadline > 0 && !it.missed()
342 }
343}
344private class Scheduler {
345 val registry = sortedMapOf<String, Process>()
346 val output = StringBuilder()
347 fun emit(s: String) {
348 output.append(s).append('\n')
349 }
350 fun process(id: String): Process {
351 checkBad(id !in registry, "unknown process $id")
352 return registry.getValue(id)
353 }
354 fun algorithm(algo: String) {
355 checkBad(algo !in ALGOS, "unknown algorithm $algo")
356 }
357 fun prepare(words: List<String>, algo: String? = null, preset: String? = null): MutableMap<String, Long> {
358 val p = parameters(words, algo, preset)
359 checkBad(registry.isEmpty(), "no processes")
360 return p
361 }
362 fun sim(algo: String, p: Map<String, Long>) = Simulation(registry.values, algo, p)
363 fun table(cells: List<String>) = cells.first().padEnd(8) + cells.drop(1).joinToString("") {
364 " ${it.padStart(8)}"
365 }
366 fun execute(line: String) {
367 val a = line.split(' ')
368 val c = a[0]
369 try {
370 when (c) {
371 "PROC" -> {
372 val id = a[1]
373 checkBad(id in registry, "duplicate process $id")
374 checkBad(registry.size == 10, "too many processes")
375 val arrival = a[2].toInt()
376 val priority = a[3].toInt()
377 checkBad(arrival > 200, "invalid arrival $id")
378 checkBad(priority > 9, "invalid priority $id")
379 val words = a.drop(4)
380 checkBad(words.size % 2 == 0, "invalid bursts $id")
381 val bursts = words.mapIndexed {
382 i, word -> val n = word.drop(1).toInt()
383 checkBad(word[0] != (if (i % 2 == 0) 'C' else 'I') || n !in 1..30, "invalid bursts $id")
384 n
385 }
386 registry[id] = Process(id, arrival, priority, bursts)
387 emit("PROC $id cpu=${bursts.filterIndexed { i, _ -> i % 2 == 0 }.sum()} io=${bursts.filterIndexed { i, _ -> i % 2 == 1 }.sum()}")
388 }
389 "GEN" -> {
390 val n = a[2].toInt()
391 for (i in 1..n) checkBad("${a[1]}$i" in registry, "duplicate process ${a[1]}$i")
392 checkBad(registry.size + n > 10, "too many processes")
393 emit("GEN count=$n")
394 val random = RandomState(a[3].toLong())
395 for (i in 1..n) {
396 val arrival = random.draw(21)
397 val priority = random.draw(10)
398 val m = 1 + random.draw(3)
399 val bursts = mutableListOf<Int>()
400 repeat(m) {
401 j -> if (j > 0) bursts.add(1 + random.draw(8))
402 bursts.add(1 + random.draw(9))
403 }
404 val p = Process("${a[1]}$i", arrival, priority, bursts)
405 registry[p.id] = p
406 emit(p.describe())
407 }
408 }
409 "SETPRIO" -> {
410 val p = process(a[1])
411 val v = a[2].toInt()
412 checkBad(v > 9, "invalid priority ${p.id}")
413 p.priority = v
414 emit("SETPRIO ${p.id} priority=$v")
415 }
416 "DEADLINE" -> {
417 val p = process(a[1])
418 val v = a[2].toInt()
419 checkBad(v > 1000, "invalid deadline ${p.id}")
420 p.deadline = v
421 emit("DEADLINE ${p.id} ${if (v == 0) "none" else v}")
422 }
423 "DROP" -> {
424 val p = process(a[1])
425 registry.remove(p.id)
426 emit("DROP ${p.id}")
427 }
428 "CLEAR" -> {
429 val n = registry.size
430 registry.clear()
431 emit("CLEAR count=$n")
432 }
433 "LIST" -> {
434 emit("LIST count=${registry.size}")
435 registry.values.forEach {
436 emit(it.describe())
437 }
438 }
439 else -> simulationCommand(c, a)
440 }
441 } catch (e: Failure) {
442 emit("ERROR ${e.message}")
443 }
444 }
445 fun simulationCommand(c: String, a: List<String>) {
446 if (c == "COMPARE" || c == "RANK") {
447 val metric = if (c == "RANK") a[1] else null
448 if (metric != null) checkBad(metric !in METRICS + listOf("end", "switches", "missed"), "unknown metric $metric")
449 val p = prepare(a.drop(if (metric == null) 1 else 2))
450 val sims = ALGOS.map {
451 sim(it, p)
452 }
453 if (metric == null) {
454 emit("COMPARE ${notation(p)}")
455 emit(table(listOf("ALGO", "TURN", "WAIT", "RESP", "UTIL", "END", "SWITCH", "MISS")))
456 for (s in sims) emit(table(listOf(s.algo) + s.totals().map {
457 ratio(it, s.jobs.size)
458 } + listOf(s.util(), s.time.toString(), s.switches.toString(), s.missed().toString())))
459 emit("BEST " + METRICS.mapIndexed {
460 i, name -> "$name=${sims.minBy { it.totals()[i] }.algo}"
461 }.joinToString(" "))
462 } else {
463 emit("RANK $metric ${notation(p)}")
464 val index = METRICS.indexOf(metric)
465 val values = sims.map {
466 s -> if (index >= 0) s.totals()[index] else when (metric) {
467 "end" -> s.time.toLong()
468 "switches" -> s.switches.toLong()
469 else -> s.missed().toLong()
470 }
471 }
472 for (i in values.indices.sortedBy {
473 values[it]
474 }) {
475 val value = values[i]
476 emit("${1 + values.count { it < value }} ${ALGOS[i]} ${if (index >= 0) ratio(value, registry.size) else value}")
477 }
478 }
479 return
480 }
481 val algo = a[1]
482 algorithm(algo)
483 if (c == "DIFF") {
484 val second = a[2]
485 algorithm(second)
486 val p = prepare(a.drop(3))
487 val one = sim(algo, p)
488 val two = sim(second, p)
489 emit("DIFF $algo $second ${notation(p)}")
490 var win = 0
491 var loss = 0
492 var ties = 0
493 one.jobs.zip(two.jobs).forEach {
494 (j, other) -> val x = j.metrics()
495 val y = other.metrics()
496 emit("${j.id} turnaround=${x[0]}/${y[0]} waiting=${x[1]}/${y[1]} response=${x[2]}/${y[2]}")
497 when {
498 x[0] < y[0] -> win++
499 x[0] > y[0] -> loss++
500 else -> ties++
501 }
502 }
503 emit("WINS $algo=$win $second=$loss ties=$ties")
504 return
505 }
506 if (c == "SWEEP") {
507 val key = a[2]
508 checkBad(key !in KEYS, "unknown key $key")
509 checkBad(key !in allowed(algo), "key $key not allowed for $algo")
510 val lo = a[3].toLong()
511 val hi = a[4].toLong()
512 checkBad(lo > hi || lo !in RANGES.getValue(key) || hi !in RANGES.getValue(key) || hi - lo > 20, "invalid range")
513 val p = prepare(a.drop(5), algo, key)
514 emit("SWEEP $algo $key $lo $hi")
515 var best = lo
516 var total = Long.MAX_VALUE
517 for (v in lo..hi) {
518 p[key] = v
519 val s = sim(algo, p)
520 emit("${notation(p, algo)} ${s.averages()} util=${s.util()} end=${s.time} switches=${s.switches}")
521 if (s.totals()[1] < total) {
522 total = s.totals()[1]
523 best = v
524 }
525 }
526 emit("BEST $key=$best")
527 return
528 }
529 val extra = if (c in listOf("WINDOW", "SNAPSHOT")) a[2].toInt() else 0
530 if (c == "WINDOW") checkBad(extra !in 1..1000, "invalid window")
531 val p = prepare(a.drop(if (c in listOf("WINDOW", "SNAPSHOT")) 3 else 2), algo)
532 val s = Simulation(registry.values, algo, p, if (c == "SNAPSHOT") extra else -1)
533 val header = "$c $algo ${notation(p, algo)}"
534 when (c) {
535 "TRACE" -> {
536 emit(header)
537 s.events.forEach {
538 emit(it)
539 }
540 emit("END ${s.time}")
541 }
542 "TIMELINE" -> {
543 checkBad(s.time > 100, "too long ${s.time}")
544 emit("$header end=${s.time}")
545 for (j in s.jobs) emit("${j.id.padEnd(8)} ${j.timeline}")
546 }
547 "SNAPSHOT" -> {
548 emit("$header t=$extra")
549 s.snapshot!!.forEach {
550 emit(it)
551 }
552 }
553 "BURSTS" -> {
554 emit(header)
555 for (j in s.jobs) j.records.forEachIndexed {
556 i, b -> emit("${j.id} #${i + 1} len=${b.length} ready=${b.ready} start=${b.start} end=${b.end} wait=${b.end - b.ready - b.length} slices=${b.slices}")
557 }
558 }
559 "WINDOW" -> {
560 emit("$header width=$extra")
561 for (start in 0 until s.time step extra) {
562 val end = minOf(start + extra, s.time)
563 val t = s.ticks(start, end)
564 val counts = sortedMapOf<String, Int>()
565 for (i in start until end) if (s.cpuStates[i] == "RUN") counts[s.labels[i]] = counts.getOrDefault(s.labels[i], 0) + 1
566 val top = counts.entries.maxByOrNull {
567 it.value
568 }?.key ?: "none"
569 emit("$start-$end busy=${t[0]} switch=${t[1]} idle=${t[2]} util=${ratio(t[0] * 100L, end - start)} top=$top")
570 }
571 }
572 "STATS" -> {
573 emit(header)
574 for (i in 0..2) {
575 val values = s.jobs.map {
576 it.metrics()[i]
577 }
578 val low = values.min()
579 val high = values.max()
580 val sorted = values.sorted()
581 val n = sorted.size
582 val doubled = sorted[n / 2] + sorted[(n - 1) / 2]
583 emit("${METRICS[i]} min=$low (${s.jobs[values.indexOf(low)].id}) max=$high (${s.jobs[values.indexOf(high)].id}) median=${doubled / 2}.${if (doubled % 2 == 0L) 0 else 5} total=${s.totals()[i]}")
584 }
585 }
586 "RUN" -> {
587 emit(header)
588 val segments = mutableListOf<String>()
589 var start = 0
590 for (t in 1..s.time) if (t == s.time || s.labels[t] != s.labels[start]) {
591 segments.add("[$start-$t ${s.labels[start]}]")
592 start = t
593 }
594 segments.chunked(8).forEach {
595 emit("GANTT " + it.joinToString(""))
596 }
597 for (j in s.jobs) {
598 val m = j.metrics()
599 emit("${j.id} arrival=${j.p.arrival} done=${j.finish} turnaround=${m[0]} waiting=${m[1]} response=${m[2]} slices=${j.slices()}" + if (j.p.deadline == 0) "" else " deadline=${j.p.absolute()} late=${maxOf(0, j.finish - j.p.absolute())}")
600 }
601 emit("AVERAGE ${s.averages()}")
602 val t = s.ticks(0, s.time)
603 emit("CPU end=${s.time} busy=${t[0]} switch=${t[1]} idle=${t[2]} switches=${s.switches} util=${s.util()}")
604 emit("QUEUE max=${s.lengths.max()} avg=${ratio(s.lengths.sumOf { it.toLong() }, s.time)}")
605 emit("EVENTS " + EVENTS.joinToString(" ") {
606 "${it.lowercase()}=${s.counts.getOrDefault(it, 0)}"
607 })
608 emit("DEADLINE met=${s.met()} missed=${s.missed()}")
609 }
610 }
611 }
612}
613fun main() {
614 val reader = System.`in`.bufferedReader()
615 val s = Scheduler()
616 repeat(reader.readLine().toInt()) {
617 s.execute(reader.readLine())
618 }
619 print(s.output)
620}
