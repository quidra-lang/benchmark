24function truncated(int $u,int $v): string {
25 $n=intdiv($u*100,$v);
26 return intdiv($n,100).'.'.str_pad((string)($n%100),2,'0',STR_PAD_LEFT);

28function parseParams(array $words,?string $a=null,array $seen=[]): array {
29 $p=DEFAULTS;
30 foreach($words as $word) {
31 [$k,$text]=explode('=',$word);
32 $v=(int)$text;
33 check(!isset(RANGES[$k]),"unknown key $k");
34 check($a!==null&&!in_array($k,allowed($a),true),"key $k not allowed for $a");
35 check(in_array($k,$seen,true),"duplicate key $k");
36 [$lo,$hi]=RANGES[$k];
37 check($v<$lo||$v>$hi,"invalid $k");
38 $seen[]=$k;
39 $p[$k]=$v;
40 }
41 return $p;

59final class Job {
60 public string $state='FUTURE',$timeline='';
61 public int $burst=0,$left,$effective,$wait=0,$level=0,$used=0,$enqueued=0;
62 public ?int $start=null,$finish=null;
63 public array $records=[];
64 public function __construct(public ProcessDef $process) {
65 $this->left=$process->bursts[0];
66 $this->effective=$process->priority;
67 }
68 public function id(): string {
69 return $this->process->id;
70 }
71 public function deadline(): ?int {
72 return $this->process->deadline===null?null:$this->process->arrival+$this->process->deadline;
73 }
74 public function metrics(): array {
75 $t=$this->finish-$this->process->arrival;
76 return [$t,$t-array_sum($this->process->bursts),$this->start-$this->process->arrival];
77 }

79 return array_sum(array_map(fn(Burst $r)=>$r->slices,$this->records));
80 }

83 }

88final class Simulation {
89 public array $jobs=[],$events=[],$counts=[],$labels=[],$states=[],$lengths=[],$snapshot=[];
90 public int $time=0,$switches=0;
91 private array $queues=[[],[],[]];
92 private string $cpu='IDLE';
93 private ?Job $current=null,$previous=null;
94 private int $switchLeft=0,$run=0,$random;
95 public function __construct(array $processes,private string $a,private array $p,private ?int $snapshotTime=null) {
96 usort($processes,fn(ProcessDef $x,ProcessDef $y)=>strcmp($x->id,$y->id));
97 foreach($processes as $d) $this->jobs[]=new Job($d);
98 $this->random=$p['seed'];
99 $this->simulate();
100 }
101 private function event(string $kind,mixed ...$parts): void {
102 $this->events[]=implode(' ',[$this->time,$kind,...$parts]);
103 $this->counts[$kind]=($this->counts[$kind]??0)+1;
104 }
105 private function enqueue(Job $j): void {
106 $j->state='READY';
107 $j->enqueued=$this->time;
108 $this->queues[$j->level][]=$j;
109 }
110 private function advance(Job $j): void {
111 $j->burst++;
112 $j->left=$j->process->bursts[$j->burst];
113 }
114 private function release(): void {
115 $this->cpu='IDLE';
116 $this->current=null;
117 }
118 private function start(Job $j): void {
119 $this->cpu='RUN';
120 $this->current=$j;
121 $this->run=0;
122 $j->state='RUN';
123 $j->start??=$this->time;
124 $j->record()->start??=$this->time;
125 }
126 private function preempts(Job $j): bool {
127 if($this->a==='MLFQ') {
128 for($i=0;$i<$j->level;$i++) if($this->queues[$i]!==[]) return true;
129 return false;
130 }
131 foreach($this->queues[0] as $x) {
132 if($this->a==='SRTF'&&$x->left<$j->left) return true;
133 if($this->a==='PRIORITY'&&$x->effective<$j->effective) return true;
134 if($this->a==='EDF'&&$x->deadline()!==null&&($j->deadline()===null||$x->deadline()<$j->deadline())) return true;
135 }
136 return false;
137 }
138 private function choose(array $q): Job {
139 if($this->a==='LOTTERY') {
140 $this->random=48271*$this->random%2147483647;
141 $z=array_sum(array_map(fn(Job $j)=>10-$j->process->priority,$q));
142 $target=$this->random%$z;
143 foreach($q as $j) {
144 $target-=10-$j->process->priority;
145 if($target<0) return $j;
146 }
147 }
148 $best=$q[0];
149 foreach(array_slice($q,1) as $j) {
150 $better=match($this->a) {
151 'SJF','SRTF'=>$j->left<$best->left, 'PRIORITY'=>$j->effective<$best->effective,
152 'EDF'=>$j->deadline()!==null&&($best->deadline()===null||$j->deadline()<$best->deadline()),
153 'HRRN'=>($this->time-$j->enqueued+$j->left)*$best->left>($this->time-$best->enqueued+$best->left)*$j->left,
154 default=>false
155 };
156 if($better) $best=$j;
157 }
158 return $best;
159 }
160 private function simulate(): void {
161 while(true) {
162 if($this->a==='PRIORITY') foreach($this->queues[0] as $j) {
163 $j->wait++;
164 if($j->wait===$this->p['k']) {
165 $j->wait=0;
166 if($j->effective>0) {
167 $j->effective--;
168 $this->event('AGE',$j->id(),$j->effective);
169 }
170 }
171 }
172 foreach($this->jobs as $j) if($j->state==='IO'&&$j->left===0) {
173 $this->advance($j);
174 $j->records[]=new Burst($j->left,$this->time);
175 $this->enqueue($j);
176 $this->event('IODONE',$j->id());
177 }
178 foreach($this->jobs as $j) if($j->state==='FUTURE'&&$j->process->arrival===$this->time) {
179 $j->records[]=new Burst($j->left,$this->time);
180 $this->enqueue($j);
181 $this->event('ARRIVE',$j->id());
182 }
183 $j=$this->current;
184 if($this->cpu==='SWITCH'&&$this->switchLeft===0) {
185 $this->start($j);
186 $this->event('START',$j->id());
187 } elseif($this->cpu==='RUN') {
188 if($j->left===0&&$j->burst===count($j->process->bursts)-1) {
189 $j->state='DONE';
190 $j->finish=$this->time;
191 $this->event('DONE',$j->id());
192 $this->release();
193 } else {
194 $demoted=$this->a==='MLFQ'&&$j->level<2&&$j->used===[4,8][$j->level];
195 if($demoted) {
196 $j->level++;
197 $j->used=0;
198 $this->event('DEMOTE',$j->id(),$j->level);
199 }
200 if($j->left===0) {
201 $this->advance($j);
202 $j->state='IO';
203 $this->event('IO',$j->id(),$j->left);
204 $this->release();
205 } elseif($demoted) {
206 $this->enqueue($j);
207 $this->release();
208 } elseif(in_array($this->a,['RR','LOTTERY'],true)&&$this->run===$this->p['q']) {
209 $this->enqueue($j);
210 $this->event('EXPIRE',$j->id());
211 $this->release();
212 } elseif($this->preempts($j)) {
213 $this->enqueue($j);
214 $this->event('PREEMPT',$j->id());
215 $this->release();
216 }
217 }
218 }
219 if(count(array_filter($this->jobs,fn(Job $x)=>$x->state!=='DONE'))===0) break;
220 if($this->a==='MLFQ'&&$this->p['s']>0&&$this->time>0&&$this->time%$this->p['s']===0) {
221 $this->queues=[array_merge(...$this->queues),[],[]];
222 foreach($this->jobs as $x) $x->level=$x->used=0;
223 $this->event('BOOST');
224 }
225 if($this->cpu==='IDLE') for($i=0;$i<3;$i++) if($this->queues[$i]!==[]) {
226 $j=$this->choose($this->queues[$i]);
227 $index=array_search($j,$this->queues[$i],true);
228 array_splice($this->queues[$i],$index,1);
229 $j->effective=$j->process->priority;
230 $j->wait=0;
231 $j->record()->slices++;
232 $switched=$j!==$this->previous;
233 if($switched) $this->switches++;
234 if($switched&&$this->p['cs']>0) {
235 $this->cpu='SWITCH';
236 $this->current=$j;
237 $this->switchLeft=$this->p['cs'];
238 $j->state='SWITCH';
239 $this->event('SWITCH',$j->id());
240 } else {
241 $this->start($j);
242 $this->event('DISPATCH',$j->id());
243 }
244 break;
245 }
246 if($this->time===$this->snapshotTime) $this->capture();
247 foreach($this->jobs as $x) $x->timeline.=SYMBOLS[$x->state];
248 $this->lengths[]=array_sum(array_map('count',$this->queues));
249 $this->states[]=$this->cpu;
250 if($this->cpu==='RUN') {
251 $j=$this->current;
252 $j->left--;
253 $j->used++;
254 $this->run++;
255 $this->previous=$j;
256 $this->labels[]=$j->id();
257 if($j->left===0) $j->record()->end=$this->time+1;
258 } elseif($this->cpu==='SWITCH') {
259 $this->switchLeft--;
260 $this->labels[]='CS';
261 } else $this->labels[]='IDLE';
262 foreach($this->jobs as $x) if($x->state==='IO') $x->left--;
263 $this->time++;
264 }
265 if($this->snapshot===[]) $this->capture();
266 }
267 private function capture(): void {
268 $this->snapshot=[$this->cpu==='IDLE'?'CPU IDLE':"CPU {$this->cpu} ".$this->current->id()];
269 for($i=0;$i<($this->a==='MLFQ'?3:1);$i++) {
270 $ids=array_map(fn(Job $j)=>$j->id(),$this->queues[$i]);
271 $this->snapshot[]='QUEUE'.($this->a==='MLFQ'?$i:'').' '.($ids===[]?'none':implode(' ',$ids));
272 }
273 foreach($this->jobs as $j) {
274 if($j->state==='FUTURE') $text=$j->id()." FUTURE at={$j->process->arrival}";
275 elseif($j->state==='DONE') $text=$j->id()." DONE at={$j->finish}";
276 else {
277 $text=$j->id()." {$j->state} burst=".($j->burst+1)." left={$j->left}";
278 if($this->a==='PRIORITY') $text.=" prio={$j->effective}";
279 if($this->a==='MLFQ') $text.=" level={$j->level} used={$j->used}";
280 }
281 $this->snapshot[]=$text;
282 }
283 }
284 public function totals(): array {
285 $v=[0,0,0];
286 foreach($this->jobs as $j) foreach($j->metrics() as $i=>$n) $v[$i]+=$n;
287 return $v;
288 }
289 public function averages(): array {
290 return array_map(fn($n)=>truncated($n,count($this->jobs)),$this->totals());
291 }
292 public function ticks(int $a,int $b): array {
293 $ss=array_count_values(array_slice($this->states,$a,$b-$a));
294 return [$ss['RUN']??0,$ss['SWITCH']??0,$ss['IDLE']??0];
295 }
296 public function utilization(): string {
297 return truncated($this->ticks(0,$this->time)[0]*100,$this->time);
298 }
299 public function missed(): int {
300 return count(array_filter($this->jobs,fn(Job $j)=>$j->missed()));
301 }
302 public function met(): int {
303 return count(array_filter($this->jobs,fn(Job $j)=>$j->deadline()!==null&&!$j->missed()));
304 }
305 public function segments(): array {
306 $result=[];
307 $start=0;
308 for($t=1;$t<=$this->time;$t++) if($t===$this->time||$this->labels[$t]!==$this->labels[$start]) {
309 $result[]="[$start-$t {$this->labels[$start]}]";
310 $start=$t;
311 }
312 return $result;
313 }

315final class Scheduler {
316 private array $registry=[];
317 private function algorithm(string $a): void {
318 check(!in_array($a,ALGOS,true),"unknown algorithm $a");
319 }
320 private function process(string $id): ProcessDef {
321 return $this->registry[$id]??throw new CommandError("unknown process $id");
322 }
323 private function prepare(array $words,?string $a=null,array $preset=[]): array {
324 $p=parseParams($words,$a,$preset);
325 check($this->registry===[],'no processes');
326 return $p;
327 }
328 private function sim(string $a,array $p,?int $t=null): Simulation {
329 return new Simulation(array_values($this->registry),$a,$p,$t);
330 }
331 private function table(array $cells): string {
332 $s=str_pad(array_shift($cells),8);
333 foreach($cells as $v) $s.=' '.str_pad((string)$v,8,' ',STR_PAD_LEFT);
334 return $s;
335 }
336 private function metricParts(array $v): array {
337 $p=[];
338 foreach(METRICS as $i=>$m) $p[]="$m={$v[$i]}";
339 return $p;
340 }
341 public function execute(string $line): void {
342 $a=explode(' ',trim($line));
343 $c=array_shift($a);
344 try {
345 switch($c) {
346 case 'PROC':
347 [$id,$at,$pr]=[$a[0],(int)$a[1],(int)$a[2]];
348 $words=array_slice($a,3);
349 check(isset($this->registry[$id]),"duplicate process $id");
350 check(count($this->registry)>=10,'too many processes');
351 check($at>200,"invalid arrival $id");
352 check($pr>9,"invalid priority $id");
353 $valid=count($words)%2===1;
354 $bs=[];
355 $cpu=$io=0;
356 foreach($words as $i=>$w) {
357 $n=(int)substr($w,1);
358 $valid=$valid&&$w[0]===($i%2===0?'C':'I')&&$n>=1&&$n<=30;
359 $bs[]=$n;
360 if($i%2===0) $cpu+=$n;
361 else $io+=$n;
362 }
363 check(!$valid,"invalid bursts $id");
364 $this->registry[$id]=new ProcessDef($id,$at,$pr,$bs);
365 emit($c,$id,"cpu=$cpu","io=$io");
366 return;
367 case 'GEN':
368 [$prefix,$count,$seed]=[$a[0],(int)$a[1],(int)$a[2]];
369 for($i=1;$i<=$count;$i++) check(isset($this->registry[$prefix.$i]),'duplicate process '.$prefix.$i);
370 check(count($this->registry)+$count>10,'too many processes');
371 emit($c,"count=$count");
372 $draw=function(int $m) use (&$seed): int {
373 $seed=48271*$seed%2147483647;
374 return $seed%$m;
375 };
376 for($i=1;$i<=$count;$i++) {
377 $id=$prefix.$i;
378 $at=$draw(21);
379 $pr=$draw(10);
380 $m=1+$draw(3);
381 $bs=[];
382 for($j=0;$j<$m;$j++) {
383 if($j>0) $bs[]=1+$draw(8);
384 $bs[]=1+$draw(9);
385 }
386 $d=new ProcessDef($id,$at,$pr,$bs);
387 $this->registry[$id]=$d;
388 emit($d->describe());
389 }
390 return;
391 case 'SETPRIO':
392 case 'DEADLINE':
393 $d=$this->process($a[0]);
394 $n=(int)$a[1];
395 check($n>($c==='SETPRIO'?9:1000),'invalid '.($c==='SETPRIO'?'priority':'deadline')." {$d->id}");
396 if($c==='SETPRIO') {
397 $d->priority=$n;
398 emit($c,$d->id,"priority=$n");
399 } else {
400 $d->deadline=$n===0?null:$n;
401 emit($c,$d->id,$d->deadline??'none');
402 }
403 return;
404 case 'DROP':
405 $this->process($a[0]);
406 unset($this->registry[$a[0]]);
407 emit($c,$a[0]);
408 return;
409 case 'CLEAR':
410 emit($c,'count='.count($this->registry));
411 $this->registry=[];
412 return;
413 case 'LIST':
414 emit($c,'count='.count($this->registry));
415 $ids=array_keys($this->registry);
416 sort($ids,SORT_STRING);
417 foreach($ids as $id) emit($this->registry[$id]->describe());
418 return;
419 }
420 if(in_array($c,['COMPARE','DIFF','RANK'],true)) {
421 $first=$second=$metric=null;
422 $header=[];
423 if($c==='DIFF') {
424 $first=array_shift($a);
425 $second=array_shift($a);
426 $this->algorithm($first);
427 $this->algorithm($second);
428 $header=[$first,$second];
429 } elseif($c==='RANK') {
430 $metric=array_shift($a);
431 check(!in_array($metric,[...METRICS,'end','switches','missed'],true),"unknown metric $metric");
432 $header=[$metric];
433 }
434 $p=$this->prepare($a);
435 emit($c,...[...$header,notation($p)]);
436 $algos=$c==='DIFF'?[$first,$second]:ALGOS;
437 $sims=array_map(fn($algo)=>$this->sim($algo,$p),$algos);
438 if($c==='COMPARE') {
439 emit($this->table(['ALGO','TURN','WAIT','RESP','UTIL','END','SWITCH','MISS']));
440 foreach($sims as $i=>$s) emit($this->table([$algos[$i],...$s->averages(),$s->utilization(),$s->time,$s->switches,$s->missed()]));
441 $best=[];
442 foreach(METRICS as $i=>$m) {
443 $b=0;
444 foreach($sims as $j=>$s) if($s->totals()[$i]<$sims[$b]->totals()[$i]) $b=$j;
445 $best[]="$m={$algos[$b]}";
446 }
447 emit('BEST',...$best);
448 } elseif($c==='DIFF') {
449 $wins=[0,0,0];
450 foreach($sims[0]->jobs as $i=>$j) {
451 $x=$j->metrics();
452 $y=$sims[1]->jobs[$i]->metrics();
453 $parts=[];
454 foreach(METRICS as $k=>$m) $parts[]="$m={$x[$k]}/{$y[$k]}";
455 emit($j->id(),...$parts);
456 $wins[$x[0]<$y[0]?0:($x[0]>$y[0]?1:2)]++;
457 }
458 emit('WINS',"$first={$wins[0]}","$second={$wins[1]}","ties={$wins[2]}");
459 } else {
460 $idx=array_search($metric,METRICS,true);
461 $values=[];
462 foreach($sims as $s) $values[]=$idx!==false?$s->totals()[$idx]:match($metric) {
463 'end'=>$s->time,'switches'=>$s->switches,default=>$s->missed()
464 };
465 $indices=array_keys($values);
466 usort($indices,fn($i,$j)=>($values[$i]<=>$values[$j])?:($i<=>$j));
467 foreach($indices as $i) {
468 $rank=1;
469 foreach($values as $v) if($v<$values[$i]) $rank++;
470 emit($rank,$algos[$i],$idx!==false?truncated($values[$i],count($this->registry)):$values[$i]);
471 }
472 }
473 return;
474 }
475 $algo=array_shift($a);
476 $this->algorithm($algo);
477 $extra=null;
478 $preset=[];
479 if($c==='WINDOW') {
480 $extra=(int)array_shift($a);
481 check($extra<1||$extra>1000,'invalid window');
482 } elseif($c==='SNAPSHOT') $extra=(int)array_shift($a);
483 elseif($c==='SWEEP') {
484 $key=array_shift($a);
485 $lo=(int)array_shift($a);
486 $hi=(int)array_shift($a);
487 check(!isset(RANGES[$key]),"unknown key $key");
488 check(!in_array($key,allowed($algo),true),"key $key not allowed for $algo");
489 [$min,$max]=RANGES[$key];
490 check($lo>$hi||$lo<$min||$hi>$max||$hi-$lo>20,'invalid range');
491 $preset=[$key];
492 }
493 $p=$this->prepare($a,$algo,$preset);
494 if($c==='SWEEP') {
495 emit($c,$algo,$key,$lo,$hi);
496 $best=null;
497 $bestValue=$lo;
498 for($v=$lo;$v<=$hi;$v++) {
499 $p[$key]=$v;
500 $s=$this->sim($algo,$p);
501 emit(notation($p,$algo),...array_merge($this->metricParts($s->averages()),['util='.$s->utilization(),"end={$s->time}","switches={$s->switches}"]));
502 $total=$s->totals()[1];
503 if($best===null||$total<$best) {
504 $best=$total;
505 $bestValue=$v;
506 }
507 }
508 emit('BEST',"$key=$bestValue");
509 return;
510 }
511 $s=$this->sim($algo,$p,$c==='SNAPSHOT'?$extra:null);
512 check($c==='TIMELINE'&&$s->time>100,"too long {$s->time}");
513 $suffix=match($c) {
514 'TIMELINE'=>["end={$s->time}"],'SNAPSHOT'=>["t=$extra"],'WINDOW'=>["width=$extra"],default=>[]
515 };
516 emit($c,$algo,notation($p,$algo),...$suffix);
517 switch($c) {
518 case 'TRACE':
519 foreach($s->events as $v) emit($v);
520 emit('END',$s->time);
521 break;
522 case 'SNAPSHOT':
523 foreach($s->snapshot as $v) emit($v);
524 break;
525 case 'TIMELINE':
526 foreach($s->jobs as $j) emit(str_pad($j->id(),8),$j->timeline);
527 break;
528 case 'BURSTS':
529 foreach($s->jobs as $j) foreach($j->records as $i=>$r) emit($j->id(),'#'.($i+1),"len={$r->length}","ready={$r->ready}","start={$r->start}","end={$r->end}",'wait='.($r->end-$r->ready-$r->length),"slices={$r->slices}");
530 break;
531 case 'WINDOW':
532 for($start=0;$start<$s->time;$start+=$extra) {
533 $stop=min($start+$extra,$s->time);
534 [$b,$cs,$idle]=$s->ticks($start,$stop);
535 $counts=[];
536 for($t=$start;$t<$stop;$t++) if($s->states[$t]==='RUN') $counts[$s->labels[$t]]=($counts[$s->labels[$t]]??0)+1;
537 $ids=array_keys($counts);
538 usort($ids,fn($x,$y)=>($counts[$y]<=>$counts[$x])?:strcmp($x,$y));
539 emit("$start-$stop","busy=$b","switch=$cs","idle=$idle",'util='.truncated($b*100,$stop-$start),'top='.($ids[0]??'none'));
540 }
541 break;
542 case 'STATS':
543 foreach(METRICS as $i=>$m) {
544 $vs=array_map(fn(Job $j)=>$j->metrics()[$i],$s->jobs);
545 $sorted=$vs;
546 sort($sorted,SORT_NUMERIC);
