51func has(a []string, s string) bool {

60func ratio(u, v int) string {
61 n := 100 * u / v
62 return fmt.Sprintf("%d.%02d", n/100, n%100)

65func allowed(a, k string) bool {
66 return k == "cs" || k == "q" && (a == "RR" || a == "LOTTERY") || k == "seed" && a == "LOTTERY" || k == "k" && a == "PRIORITY" || k == "s" && a == "MLFQ"
67}

69func valid(k string, v int) bool {
70 switch k {
71 case "cs":
72 return v <= 5
73 case "q":
74 return v >= 1 && v <= 20
75 case "k":
76 return v >= 1 && v <= 50
77 case "s":
78 return v <= 500
79 case "seed":
80 return v >= 1 && v <= 2147483646
81 }
82 return false
83}

85func notation(a string, p map[string]int, full bool) string {
86 r := []string{}
87 for _, k := range keys {
88 if full || allowed(a, k) {
89 r = append(r, fmt.Sprintf("%s=%d", k, p[k]))
90 }
91 }
92 return strings.Join(r, " ")
93}

95func simulate(p []Process, a string, params map[string]int) *Result {
96 n := len(p)
97 r := &Result{p: p, states: make([]State, n), bursts: make([][]Burst, n)}
98 for i, x := range p {
99 r.states[i] = State{state: "FUTURE", left: x.bursts[0], effective: x.priority, start: -1, finish: -1}
100 r.bursts[i] = make([]Burst, (len(x.bursts)+1)/2)
101 for j := range r.bursts[i] {
102 r.bursts[i][j].start = -1
103 }
104 }
105 var queues [3][]int
106 cpu, previous, mode, sw, run := -1, -1, 0, 0, 0
107 x := int64(params["seed"])
108 for t := 0; ; t++ {
109 log := func(event string) {
110 r.events = append(r.events, fmt.Sprintf("%d %s", t, event))
111 }
112 enqueue := func(i int) {
113 s := &r.states[i]
114 s.state = "READY"
115 s.enqueued = t
116 l := 0
117 if a == "MLFQ" {
118 l = s.level
119 }
120 queues[l] = append(queues[l], i)
121 }
122 start := func(i int) {
123 cpu = i
124 mode = 2
125 run = 0
126 s := &r.states[i]
127 s.state = "RUN"
128 if s.start < 0 {
129 s.start = t
130 }
131 b := &r.bursts[i][s.burst/2]
132 if b.start < 0 {
133 b.start = t
134 }
135 }
136 deadline := func(i int) int {
137 if p[i].deadline == 0 {
138 return 1 << 60
139 }
140 return p[i].arrival + p[i].deadline
141 }
142 if a == "PRIORITY" {
143 for _, i := range queues[0] {
144 s := &r.states[i]
145 s.wait++
146 if s.wait == params["k"] {
147 s.wait = 0
148 if s.effective > 0 {
149 s.effective--
150 log(fmt.Sprintf("AGE %s %d", p[i].id, s.effective))
151 }
152 }
153 }
154 }
155 for i := range p {
156 s := &r.states[i]
157 if s.state == "IO" && s.left == 0 {
158 s.burst++
159 s.left = p[i].bursts[s.burst]
160 r.bursts[i][s.burst/2].ready = t
161 enqueue(i)
162 log("IODONE " + p[i].id)
163 }
164 }
165 for i := range p {
166 if r.states[i].state == "FUTURE" && p[i].arrival == t {
167 r.bursts[i][0].ready = t
168 enqueue(i)
169 log("ARRIVE " + p[i].id)
170 }
171 }
172 if mode == 1 && sw == 0 {
173 start(cpu)
174 log("START " + p[cpu].id)
175 } else if mode == 2 {
176 i := cpu
177 s := &r.states[i]
178 if s.left == 0 && s.burst == len(p[i].bursts)-1 {
179 s.state = "DONE"
180 s.finish = t
181 log("DONE " + p[i].id)
182 mode = 0
183 cpu = -1
184 } else {
185 demoted := false
186 if a == "MLFQ" && s.level < 2 && s.used == []int{4, 8}[s.level] {
187 s.level++
188 s.used = 0
189 demoted = true
190 log(fmt.Sprintf("DEMOTE %s %d", p[i].id, s.level))
191 }
192 preempt := false
193 for l, q := range queues {
194 for _, j := range q {
195 z := r.states[j]
196 if a == "SRTF" && z.left < s.left || a == "PRIORITY" && z.effective < s.effective || a == "EDF" && deadline(j) < deadline(i) || a == "MLFQ" && l < s.level {
197 preempt = true
198 }
199 }
200 }
201 if s.left == 0 {
202 s.burst++
203 s.left = p[i].bursts[s.burst]
204 s.state = "IO"
205 log(fmt.Sprintf("IO %s %d", p[i].id, s.left))
206 mode = 0
207 cpu = -1
208 } else if demoted {
209 enqueue(i)
210 mode = 0
211 cpu = -1
212 } else if (a == "RR" || a == "LOTTERY") && run == params["q"] {
213 enqueue(i)
214 log("EXPIRE " + p[i].id)
215 mode = 0
216 cpu = -1
217 } else if preempt {
218 enqueue(i)
219 log("PREEMPT " + p[i].id)
220 mode = 0
221 cpu = -1
222 }
223 }
224 }
225 finished := true
226 for _, s := range r.states {
227 if s.state != "DONE" {
228 finished = false
229 }
230 }
231 if finished {
232 r.end = t
233 shot := Snapshot{states: append([]State(nil), r.states...), cpu: -1}
234 r.snapshots = append(r.snapshots, shot)
235 break
236 }
237 if a == "MLFQ" && params["s"] > 0 && t > 0 && t%params["s"] == 0 {
238 queues[0] = append(queues[0], queues[1]...)
239 queues[0] = append(queues[0], queues[2]...)
240 queues[1] = nil
241 queues[2] = nil
242 for i := range r.states {
243 r.states[i].level = 0
244 r.states[i].used = 0
245 }
246 log("BOOST")
247 }
248 if mode == 0 {
249 l := 0
250 for l < 3 && len(queues[l]) == 0 {
251 l++
252 }
253 if l < 3 {
254 q := queues[l]
255 selected := 0
256 if a == "LOTTERY" {
257 total := 0
258 for _, i := range q {
259 total += 10 - p[i].priority
260 }
261 x = 48271 * x % 2147483647
262 draw := int(x % int64(total))
263 sum := 0
264 for j, i := range q {
265 sum += 10 - p[i].priority
266 if sum > draw {
267 selected = j
268 break
269 }
270 }
271 } else {
272 for j := 1; j < len(q); j++ {
273 i, b := q[j], q[selected]
274 u, v := r.states[i], r.states[b]
275 better := false
276 switch a {
277 case "SJF", "SRTF":
278 better = u.left < v.left
279 case "PRIORITY":
280 better = u.effective < v.effective
281 case "EDF":
282 better = deadline(i) < deadline(b)
283 case "HRRN":
284 better = (t-u.enqueued+u.left)*v.left > (t-v.enqueued+v.left)*u.left
285 }
286 if better {
287 selected = j
288 }
289 }
290 }
291 i := q[selected]
292 queues[l] = append(q[:selected], q[selected+1:]...)
293 s := &r.states[i]
294 s.effective = p[i].priority
295 s.wait = 0
296 s.slices++
297 r.bursts[i][s.burst/2].slices++
298 change := i != previous
299 if change {
300 r.switches++
301 }
302 if change && params["cs"] > 0 {
303 cpu = i
304 mode = 1
305 sw = params["cs"]
306 s.state = "SWITCH"
307 log("SWITCH " + p[i].id)
308 } else {
309 start(i)
310 log("DISPATCH " + p[i].id)
311 }
312 }
313 }
314 shot := Snapshot{states: append([]State(nil), r.states...), cpu: cpu, mode: mode}
315 qlen := 0
316 for l, q := range queues {
317 shot.queues[l] = append([]int(nil), q...)
318 qlen += len(q)
319 }
320 r.snapshots = append(r.snapshots, shot)
321 r.qsum += qlen
322 r.qmax = max(r.qmax, qlen)
323 label := "IDLE"
324 if mode == 2 {
325 s := &r.states[cpu]
326 s.left--
327 s.used++
328 run++
329 previous = cpu
330 label = p[cpu].id
331 r.busy++
332 if s.left == 0 {
333 r.bursts[cpu][s.burst/2].end = t + 1
334 }
335 } else if mode == 1 {
336 sw--
337 label = "CS"
338 r.swtime++
339 } else {
340 r.idle++
341 }
342 r.labels = append(r.labels, label)
343 for i := range r.states {
344 if r.states[i].state == "IO" {
345 r.states[i].left--
346 }
347 }
348 }
349 for i, z := range p {
350 s := r.states[i]
351 turn := s.finish - z.arrival
352 total := 0
353 for _, b := range z.bursts {
354 total += b
355 }
356 r.totals[0] += turn
357 r.totals[1] += turn - total
358 r.totals[2] += s.start - z.arrival
359 if z.deadline > 0 {
360 if s.finish <= z.arrival+z.deadline {
361 r.met++
362 } else {
363 r.missed++
364 }
365 }
366 }
367 return r
368}

370func metrics(r *Result, i int) [3]int {
371 p, s := r.p[i], r.states[i]
372 total := 0
373 for _, b := range p.bursts {
374 total += b
375 }
376 t := s.finish - p.arrival
377 return [3]int{t, t - total, s.start - p.arrival}
378}
379
380func summary(r *Result) string {
381 n := len(r.p)
382 return fmt.Sprintf("turnaround=%s waiting=%s response=%s util=%s end=%d switches=%d", ratio(r.totals[0], n), ratio(r.totals[1], n), ratio(r.totals[2], n), ratio(r.busy*100, r.end), r.end, r.switches)
383}

385func listline(p Process) string {
386 b := []string{}
387 for i, n := range p.bursts {
388 c := "C"
389 if i%2 == 1 {
390 c = "I"
391 }
392 b = append(b, c+strconv.Itoa(n))
393 }
394 d := "none"
395 if p.deadline > 0 {
396 d = strconv.Itoa(p.deadline)
397 }
398 return fmt.Sprintf("%s arrival=%d priority=%d deadline=%s bursts=%s", p.id, p.arrival, p.priority, d, strings.Join(b, ","))
399}

410func (app *App) command(w []string) {
411 c := w[0]
412 err := func(s string) {
413 panic(s)
414 }
415 n := func(i int) int {
416 return integer(w[i])
417 }
418 p, found := Process{}, false
419 if len(w) > 1 {
420 p, found = app.registry[w[1]]
421 }
422 switch c {
423 case "PROC":
424 if found {
425 err("duplicate process " + w[1])
426 }
427 if len(app.registry) == 10 {
428 err("too many processes")
429 }
430 if n(2) > 200 {
431 err("invalid arrival " + w[1])
432 }
433 if n(3) > 9 {
434 err("invalid priority " + w[1])
435 }
436 b := []int{}
437 ok := (len(w)-4)%2 == 1
438 for i, s := range w[4:] {
439 v := integer(s[1:])
440 expected := byte('C')
441 if i%2 == 1 {
442 expected = 'I'
443 }
444 if s[0] != expected || v < 1 || v > 30 {
445 ok = false
446 }
447 b = append(b, v)
448 }
449 if !ok {
450 err("invalid bursts " + w[1])
451 }
452 p = Process{w[1], n(2), n(3), 0, b}
453 app.registry[p.id] = p
454 cpu, io := 0, 0
455 for i, v := range b {
456 if i%2 == 0 {
457 cpu += v
458 } else {
459 io += v
460 }
461 }
462 app.emit("PROC", p.id, fmt.Sprintf("cpu=%d io=%d", cpu, io))
463 return
464 case "GEN":
465 count := n(2)
466 for i := 1; i <= count; i++ {
467 id := w[1] + strconv.Itoa(i)
468 if _, ok := app.registry[id]; ok {
469 err("duplicate process " + id)
470 }
471 }
472 if len(app.registry)+count > 10 {
473 err("too many processes")
474 }
475 app.emit("GEN", fmt.Sprintf("count=%d", count))
476 x := int64(n(3))
477 next := func() int {
478 x = x * 48271 % 2147483647
479 return int(x)
480 }
481 for i := 1; i <= count; i++ {
482 id := w[1] + strconv.Itoa(i)
483 arrival, priority, m := next()%21, next()%10, 1+next()%3
484 b := []int{}
485 for j := 0; j < m; j++ {
486 if j > 0 {
487 b = append(b, 1+next()%8)
488 }
489 b = append(b, 1+next()%9)
490 }
491 p := Process{id, arrival, priority, 0, b}
492 app.registry[id] = p
493 app.emit(listline(p))
494 }
495 return
496 case "SETPRIO", "DEADLINE", "DROP":
497 if !found {
498 err("unknown process " + w[1])
499 }
500 if c == "DROP" {
501 delete(app.registry, w[1])
502 app.emit(c, w[1])
503 } else if c == "SETPRIO" {
504 if n(2) > 9 {
505 err("invalid priority " + w[1])
506 }
507 p.priority = n(2)
508 app.registry[p.id] = p
509 app.emit(c, p.id, fmt.Sprintf("priority=%d", p.priority))
510 } else {
511 if n(2) > 1000 {
512 err("invalid deadline " + w[1])
513 }
514 p.deadline = n(2)
515 app.registry[p.id] = p
516 d := "none"
517 if p.deadline > 0 {
518 d = strconv.Itoa(p.deadline)
519 }
520 app.emit(c, p.id, d)
521 }
522 return
523 case "CLEAR":
524 app.emit("CLEAR", fmt.Sprintf("count=%d", len(app.registry)))
525 app.registry = map[string]Process{}
526 return
527 case "LIST":
528 ids := []string{}
529 for id := range app.registry {
530 ids = append(ids, id)
531 }
532 sort.Strings(ids)
533 app.emit("LIST", fmt.Sprintf("count=%d", len(ids)))
534 for _, id := range ids {
535 app.emit(listline(app.registry[id]))
536 }
537 return
538 }
539 a, metric, start := "", "", 2
540 multi := c == "COMPARE" || c == "RANK" || c == "DIFF"
541 if c == "COMPARE" {
542 start = 1
543 } else if c == "RANK" {
544 metric = w[1]
545 if !has([]string{"turnaround", "waiting", "response", "end", "switches", "missed"}, metric) {
546 err("unknown metric " + metric)
547 }
548 } else {
549 a = w[1]
550 if !has(algorithms, a) {
551 err("unknown algorithm " + a)
552 }
553 if c == "DIFF" {
554 if !has(algorithms, w[2]) {
555 err("unknown algorithm " + w[2])
556 }
557 start = 3
558 }
559 }
560 if c == "SNAPSHOT" || c == "WINDOW" {
561 start = 3
562 }
563 if c == "WINDOW" && (n(2) < 1 || n(2) > 1000) {
564 err("invalid window")
565 }
566 seen := map[string]bool{}
567 if c == "SWEEP" {
568 start = 5
569 k := w[2]
570 if !has(keys, k) {
571 err("unknown key " + k)
572 }
573 if !allowed(a, k) {
574 err("key " + k + " not allowed for " + a)
575 }
576 if n(3) > n(4) || !valid(k, n(3)) || !valid(k, n(4)) || n(4)-n(3) > 20 {
577 err("invalid range")
578 }
579 seen[k] = true
580 }
581 params := map[string]int{"cs": 0, "q": 4, "k": 5, "s": 0, "seed": 1}
582 for _, word := range w[start:] {
583 kv := strings.Split(word, "=")
584 k, v := kv[0], integer(kv[1])
585 if !has(keys, k) {
586 err("unknown key " + k)
587 }
588 if !multi && !allowed(a, k) {
589 err("key " + k + " not allowed for " + a)
590 }
591 if seen[k] {
592 err("duplicate key " + k)
593 }
594 if !valid(k, v) {
595 err("invalid " + k)
596 }
597 seen[k] = true
598 params[k] = v
599 }
600 if len(app.registry) == 0 {
601 err("no processes")
602 }
603 ps := []Process{}
604 for _, p := range app.registry {
605 ps = append(ps, p)
606 }
607 sort.Slice(ps, func(i, j int) bool {
608 return ps[i].id < ps[j].id
609 })
610 if c == "SWEEP" {
611 app.emit(c, a, w[2], n(3), n(4))
612 best, bestv := -1, 0
613 for v := n(3); v <= n(4); v++ {
614 params[w[2]] = v
615 r := simulate(ps, a, params)
616 app.emit(notation(a, params, false), summary(r))
617 if best < 0 || r.totals[1] < best {
618 best = r.totals[1]
619 bestv = v
620 }
621 }
622 app.emit("BEST", fmt.Sprintf("%s=%d", w[2], bestv))
623 return
624 }
625 if multi {
626 if c == "DIFF" {
627 r, s := simulate(ps, a, params), simulate(ps, w[2], params)
628 app.emit(c, a, w[2], notation("", params, true))
629 wins, loss, ties := 0, 0, 0
630 for i, p := range ps {
631 u, v := metrics(r, i), metrics(s, i)
632 app.emit(fmt.Sprintf("%s turnaround=%d/%d waiting=%d/%d response=%d/%d", p.id, u[0], v[0], u[1], v[1], u[2], v[2]))
633 if u[0] < v[0] {
634 wins++
635 } else if u[0] > v[0] {
636 loss++
637 } else {
638 ties++
639 }
640 }
641 app.emit("WINS", fmt.Sprintf("%s=%d %s=%d ties=%d", a, wins, w[2], loss, ties))
642 return
643 }
644 results := make([]*Result, 9)
645 for i, a := range algorithms {
646 results[i] = simulate(ps, a, params)
647 }
648 if c == "COMPARE" {
649 app.emit(c, notation("", params, true))
650 row := func(v []string) {
651 fmt.Fprintf(app.out, "%-8s", v[0])
652 for _, x := range v[1:] {
653 fmt.Fprintf(app.out, " %8s", x)
654 }
655 fmt.Fprintln(app.out)
656 }
657 row([]string{"ALGO", "TURN", "WAIT", "RESP", "UTIL", "END", "SWITCH", "MISS"})
658 for i, r := range results {
659 row([]string{algorithms[i], ratio(r.totals[0], len(ps)), ratio(r.totals[1], len(ps)), ratio(r.totals[2], len(ps)), ratio(r.busy*100, r.end), strconv.Itoa(r.end), strconv.Itoa(r.switches), strconv.Itoa(r.missed)})
660 }
661 best := []string{}
662 for j, k := range []string{"turnaround", "waiting", "response"} {
663 b := 0
664 for i := 1; i < 9; i++ {
665 if results[i].totals[j] < results[b].totals[j] {
666 b = i
667 }
668 }
669 best = append(best, k+"="+algorithms[b])
670 }
671 app.emit("BEST", strings.Join(best, " "))
672 } else {
673 app.emit(c, metric, notation("", params, true))
674 value := func(i int) int {
675 r := results[i]
676 switch metric {
677 case "turnaround":
678 return r.totals[0]
679 case "waiting":
680 return r.totals[1]
681 case "response":
682 return r.totals[2]
683 case "end":
684 return r.end
685 case "switches":
686 return r.switches
687 }
688 return r.missed
689 }
690 ids := []int{0, 1, 2, 3, 4, 5, 6, 7, 8}
691 sort.SliceStable(ids, func(i, j int) bool {
692 return value(ids[i]) < value(ids[j])
693 })
694 rank := 1
695 for j, i := range ids {
696 if j > 0 && value(i) != value(ids[j-1]) {
697 rank = j + 1
698 }
699 v := strconv.Itoa(value(i))
700 if metric == "turnaround" || metric == "waiting" || metric == "response" {
701 v = ratio(value(i), len(ps))
702 }
703 app.emit(rank, algorithms[i], v)
704 }
705 }
706 return
707 }
708 r := simulate(ps, a, params)
709 note := notation(a, params, false)
710 switch c {
711 case "TRACE":
712 app.emit(c, a, note)
713 for _, s := range r.events {
714 app.emit(s)
715 }
716 app.emit("END", r.end)
717 case "TIMELINE":
718 if r.end > 100 {
719 err(fmt.Sprintf("too long %d", r.end))
720 }
721 app.emit(c, a, note, fmt.Sprintf("end=%d", r.end))
722 for i, p := range ps {
723 var line strings.Builder
724 for t := 0; t < r.end; t++ {
725 ch := byte('-')
726 switch r.snapshots[t].states[i].state {
727 case "READY":
728 ch = '.'
729 case "SWITCH":
730 ch = 's'
731 case "RUN":
732 ch = '#'
733 case "IO":
734 ch = 'i'
735 }
736 line.WriteByte(ch)
737 }
738 app.emit(fmt.Sprintf("%-8s %s", p.id, line.String()))
739 }
740 case "SNAPSHOT":
741 t := n(2)
742 s := r.snapshots[min(t, r.end)]
743 app.emit(c, a, note, fmt.Sprintf("t=%d", t))
744 if s.mode == 0 {
745 app.emit("CPU IDLE")
746 } else {
747 mode := "RUN"
748 if s.mode == 1 {
749 mode = "SWITCH"
750 }
751 app.emit("CPU", mode, ps[s.cpu].id)
752 }
753 count := 1
754 if a == "MLFQ" {
755 count = 3
756 }
757 for l := 0; l < count; l++ {
758 name := "QUEUE"
759 if a == "MLFQ" {
760 name += strconv.Itoa(l)
761 }
762 ids := []string{}
763 for _, i := range s.queues[l] {
764 ids = append(ids, ps[i].id)
765 }
766 text := "none"
767 if len(ids) > 0 {
768 text = strings.Join(ids, " ")
769 }
770 app.emit(name, text)
771 }
772 for i, p := range ps {
773 z := s.states[i]
774 switch z.state {
775 case "FUTURE":
776 app.emit(p.id, "FUTURE", fmt.Sprintf("at=%d", p.arrival))
777 case "DONE":
778 app.emit(p.id, "DONE", fmt.Sprintf("at=%d", z.finish))
779 default:
780 text := fmt.Sprintf("%s %s burst=%d left=%d", p.id, z.state, z.burst+1, z.left)
781 if a == "PRIORITY" {
782 text += fmt.Sprintf(" prio=%d", z.effective)
783 }
784 if a == "MLFQ" {
785 text += fmt.Sprintf(" level=%d used=%d", z.level, z.used)
786 }
787 app.emit(text)
788 }
789 }
790 case "BURSTS":
791 app.emit(c, a, note)
792 for i, p := range ps {
793 for j, b := range r.bursts[i] {
794 length := p.bursts[2*j]
795 app.emit(fmt.Sprintf("%s #%d len=%d ready=%d start=%d end=%d wait=%d slices=%d", p.id, j+1, length, b.ready, b.start, b.end, b.end-b.ready-length, b.slices))
796 }
797 }
798 case "WINDOW":
799 width := n(2)
800 app.emit(c, a, note, fmt.Sprintf("width=%d", width))
801 for lo := 0; lo < r.end; lo += width {
802 hi := min(lo+width, r.end)
803 busy, cs, idle := 0, 0, 0
804 counts := map[string]int{}
805 for tick := lo; tick < hi; tick++ {
806 shot := r.snapshots[tick]
807 if shot.mode == 1 {
808 cs++
809 } else if shot.mode == 0 {
810 idle++
811 } else {
812 busy++
813 counts[ps[shot.cpu].id]++
814 }
815 }
816 top, best := "none", 0
817 for _, p := range ps {
818 if counts[p.id] > best {
819 best = counts[p.id]
820 top = p.id
821 }
822 }
823 app.emit(fmt.Sprintf("%d-%d busy=%d switch=%d idle=%d util=%s top=%s", lo, hi, busy, cs, idle, ratio(busy*100, hi-lo), top))
824 }
825 case "STATS":
826 app.emit(c, a, note)
827 for j, k := range []string{"turnaround", "waiting", "response"} {
828 v := []int{}
829 imin, imax := 0, 0
830 for i := range ps {
831 m := metrics(r, i)
832 v = append(v, m[j])
833 if v[i] < v[imin] {
834 imin = i
835 }
836 if v[i] > v[imax] {
837 imax = i
838 }
839 }
840 lo, hi := v[imin], v[imax]
841 sort.Ints(v)
842 med := v[len(v)/2] * 2
843 if len(v)%2 == 0 {
844 med = v[len(v)/2-1] + v[len(v)/2]
845 }
846 app.emit(fmt.Sprintf("%s min=%d (%s) max=%d (%s) median=%d.%d total=%d", k, lo, ps[imin].id, hi, ps[imax].id, med/2, med%2*5, r.totals[j]))
847 }
848 case "RUN":
849 app.emit(c, a, note)
850 segments := []string{}
851 for lo := 0; lo < r.end; {
852 hi := lo + 1
853 for hi < r.end && r.labels[hi] == r.labels[lo] {
854 hi++
855 }
856 segments = append(segments, fmt.Sprintf("[%d-%d %s]", lo, hi, r.labels[lo]))
857 lo = hi
858 }
859 for i := 0; i < len(segments); i += 8 {
860 app.emit("GANTT", strings.Join(segments[i:min(i+8, len(segments))], ""))
861 }
862 for i, p := range ps {
863 s := r.states[i]
864 m := metrics(r, i)
865 line := fmt.Sprintf("%s arrival=%d done=%d turnaround=%d waiting=%d response=%d slices=%d", p.id, p.arrival, s.finish, m[0], m[1], m[2], s.slices)
866 if p.deadline > 0 {
867 d := p.arrival + p.deadline
868 line += fmt.Sprintf(" deadline=%d late=%d", d, max(0, s.finish-d))
869 }
870 app.emit(line)
871 }
872 app.emit(fmt.Sprintf("AVERAGE turnaround=%s waiting=%s response=%s", ratio(r.totals[0], len(ps)), ratio(r.totals[1], len(ps)), ratio(r.totals[2], len(ps))))
873 app.emit(fmt.Sprintf("CPU end=%d busy=%d switch=%d idle=%d switches=%d util=%s", r.end, r.busy, r.swtime, r.idle, r.switches, ratio(r.busy*100, r.end)))
874 app.emit(fmt.Sprintf("QUEUE max=%d avg=%s", r.qmax, ratio(r.qsum, r.end)))
875 counts := map[string]int{}
876 for _, s := range r.events {
877 counts[strings.Fields(s)[1]]++
878 }
879 app.emit(fmt.Sprintf("EVENTS preempt=%d expire=%d demote=%d boost=%d age=%d", counts["PREEMPT"], counts["EXPIRE"], counts["DEMOTE"], counts["BOOST"], counts["AGE"]))
880 app.emit(fmt.Sprintf("DEADLINE met=%d missed=%d", r.met, r.missed))
881 }
882}
