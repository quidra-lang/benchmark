1local ALGOS={'FCFS','SJF','SRTF','RR','PRIORITY','HRRN','EDF','LOTTERY','MLFQ'}
2local KEYS={'cs','q','k','s','seed'}
3local RANGES={cs={0,5},q={1,20},k={1,50},s={0,500},seed={1,2147483646}}
4local DEFAULTS={cs=0,q=4,k=5,s=0,seed=1}
5local METRICS={'turnaround','waiting','response'}
6local EVENT_KINDS={'PREEMPT','EXPIRE','DEMOTE','BOOST','AGE'}
7local SYMBOLS={FUTURE='-',DONE='-',RUN='#',SWITCH='s',READY='.',IO='i'}
8local function emit(...)
9 local a={...}
10 for i,v in ipairs(a) do
11 a[i]=tostring(v)
12 end
13 io.write(table.concat(a,' '),'\n')
14end
15local function check(bad,message)
16 if bad then
17 error(message,0)
18 end
19end
20local function contains(a,x)
21 for _,v in ipairs(a) do
22 if v==x then
23 return true
24 end
25 end
26 return false
27end
28local function allowed(a)
29 local k={'cs'}
30 for _,v in ipairs(({RR={'q'},LOTTERY={'q','seed'},PRIORITY={'k'},MLFQ={'s'}})[a] or {}) do
31 k[#k+1]=v
32 end
33 return k
34end
35local function notation(p,a)
36 local words={}
37 for _,k in ipairs(a and allowed(a) or KEYS) do
38 words[#words+1]=k..'='..p[k]
39 end
40 return table.concat(words,' ')
41end
42local function truncated(u,v)
43 local n=u*100//v
44 return string.format('%d.%02d',n//100,n%100)
45end
46local function parse_params(words,a,preset)
47 local p,seen={},{}
48 for k,v in pairs(DEFAULTS) do
49 p[k]=v
50 end
51 for _,k in ipairs(preset or {}) do
52 seen[k]=true
53 end
54 for _,word in ipairs(words) do
55 local k,text=word:match('^(%a+)=(%d+)$')
56 local v=tonumber(text)
57 check(not RANGES[k],'unknown key '..k)
58 check(a and not contains(allowed(a),k),'key '..k..' not allowed for '..(a or ''))
59 check(seen[k],'duplicate key '..k)
60 check(v<RANGES[k][1] or v>RANGES[k][2],'invalid '..k)
61 seen[k]=true
62 p[k]=v
63 end
64 return p
65end
66local function describe(d)
67 local bs={}
68 for i,n in ipairs(d.bursts) do
69 bs[#bs+1]=(i%2==1 and 'C' or 'I')..n
70 end
71 return d.id..' arrival='..d.arrival..' priority='..d.priority..' deadline='..(d.deadline or 'none')..' bursts='..table.concat(bs,',')
72end
73local function deadline(j)
74 return j.process.deadline and j.process.arrival+j.process.deadline or nil
75end
76local function metrics(j)
77 local t,total=j.finish-j.process.arrival,0
78 for _,n in ipairs(j.process.bursts) do
79 total=total+n
80 end
81 return {t,t-total,j.start-j.process.arrival}
82end
83local function slices(j)
84 local n=0
85 for _,r in ipairs(j.records) do
86 n=n+r.slices
87 end
88 return n
89end
90local function missed(j)
91 return deadline(j) and j.finish>deadline(j)
92end

103function Sim:enqueue(j)
104 j.state='READY'
105 j.enqueued=self.time
106 local q=self.queues[j.level+1]
107 q[#q+1]=j
108end
109function Sim:advance(j)
110 j.burst=j.burst+1
111 j.left=j.process.bursts[j.burst]
112end
113function Sim:release()
114 self.cpu='IDLE'
115 self.current=nil
116end
117function Sim:start(j)
118 self.cpu='RUN'
119 self.current=j
120 self.run=0
121 j.state='RUN'
122 j.start=j.start or self.time
123 local r=j.records[#j.records]
124 r.start=r.start or self.time
125end
126function Sim:preempts(j)
127 if self.a=='MLFQ' then
128 for i=1,j.level do
129 if #self.queues[i]>0 then
130 return true
131 end
132 end
133 end
134 for _,x in ipairs(self.queues[1]) do
135 if self.a=='SRTF' and x.left<j.left then
136 return true
137 end
138 if self.a=='PRIORITY' and x.effective<j.effective then
139 return true
140 end
141 if self.a=='EDF' and deadline(x) and (not deadline(j) or deadline(x)<deadline(j)) then
142 return true
143 end
144 end
145 return false
146end
147function Sim:choose(q)
148 if self.a=='LOTTERY' then
149 self.random=48271*self.random%2147483647
150 local total=0
151 for _,j in ipairs(q) do
152 total=total+10-j.process.priority
153 end
154 local target=self.random%total
155 for _,j in ipairs(q) do
156 target=target-10+j.process.priority
157 if target<0 then
158 return j
159 end
160 end
161 end
162 local best=q[1]
163 for i=2,#q do
164 local j=q[i]
165 local better=false
166 if self.a=='SJF' or self.a=='SRTF' then
167 better=j.left<best.left
168 elseif self.a=='PRIORITY' then
169 better=j.effective<best.effective
170 elseif self.a=='EDF' then
171 better=deadline(j) and (not deadline(best) or deadline(j)<deadline(best))
172 elseif self.a=='HRRN' then
173 better=(self.time-j.enqueued+j.left)*best.left>(self.time-best.enqueued+best.left)*j.left
174 end
175 if better then
176 best=j
177 end
178 end
179 return best
180end
181function Sim:capture()
182 self.snapshot={self.cpu=='IDLE' and 'CPU IDLE' or 'CPU '..self.cpu..' '..self.current.id}
183 for i=1,(self.a=='MLFQ' and 3 or 1) do
184 local ids={}
185 for _,j in ipairs(self.queues[i]) do
186 ids[#ids+1]=j.id
187 end
188 self.snapshot[#self.snapshot+1]='QUEUE'..(self.a=='MLFQ' and i-1 or '')..' '..(#ids==0 and 'none' or table.concat(ids,' '))
189 end
190 for _,j in ipairs(self.jobs) do
191 local text
192 if j.state=='FUTURE' then
193 text=j.id..' FUTURE at='..j.process.arrival
194 elseif j.state=='DONE' then
195 text=j.id..' DONE at='..j.finish
196 else
197 text=j.id..' '..j.state..' burst='..j.burst..' left='..j.left
198 if self.a=='PRIORITY' then
199 text=text..' prio='..j.effective
200 elseif self.a=='MLFQ' then
201 text=text..' level='..j.level..' used='..j.used
202 end
203 end
204 self.snapshot[#self.snapshot+1]=text
205 end
206end
207function Sim:simulate()
208 while true do
209 if self.a=='PRIORITY' then
210 for _,j in ipairs(self.queues[1]) do
211 j.wait=j.wait+1
212 if j.wait==self.p.k then
213 j.wait=0
214 if j.effective>0 then
215 j.effective=j.effective-1
216 self:event('AGE',j.id,j.effective)
217 end
218 end
219 end
220 end
221 for _,j in ipairs(self.jobs) do
222 if j.state=='IO' and j.left==0 then
223 self:advance(j)
224 j.records[#j.records+1]={length=j.left,ready=self.time,slices=0}
225 self:enqueue(j)
226 self:event('IODONE',j.id)
227 end
228 end
229 for _,j in ipairs(self.jobs) do
230 if j.state=='FUTURE' and j.process.arrival==self.time then
231 j.records[#j.records+1]={length=j.left,ready=self.time,slices=0}
232 self:enqueue(j)
233 self:event('ARRIVE',j.id)
234 end
235 end
236 local j=self.current
237 if self.cpu=='SWITCH' and self.switch_left==0 then
238 self:start(j)
239 self:event('START',j.id)
240 elseif self.cpu=='RUN' then
241 if j.left==0 and j.burst==#j.process.bursts then
242 j.state='DONE'
243 j.finish=self.time
244 self:event('DONE',j.id)
245 self:release()
246 else
247 local demoted=self.a=='MLFQ' and j.level<2 and j.used==({4,8})[j.level+1]
248 if demoted then
249 j.level=j.level+1
250 j.used=0
251 self:event('DEMOTE',j.id,j.level)
252 end
253 if j.left==0 then
254 self:advance(j)
255 j.state='IO'
256 self:event('IO',j.id,j.left)
257 self:release()
258 elseif demoted then
259 self:enqueue(j)
260 self:release()
261 elseif (self.a=='RR' or self.a=='LOTTERY') and self.run==self.p.q then
262 self:enqueue(j)
263 self:event('EXPIRE',j.id)
264 self:release()
265 elseif self:preempts(j) then
266 self:enqueue(j)
267 self:event('PREEMPT',j.id)
268 self:release()
269 end
270 end
271 end
272 local done=true
273 for _,x in ipairs(self.jobs) do
274 if x.state~='DONE' then
275 done=false
276 break
277 end
278 end
279 if done then
280 break
281 end
282 if self.a=='MLFQ' and self.p.s>0 and self.time>0 and self.time%self.p.s==0 then
283 local merged={}
284 for _,q in ipairs(self.queues) do
285 for _,x in ipairs(q) do
286 merged[#merged+1]=x
287 end
288 end
289 self.queues={merged,{},{}}
290 for _,x in ipairs(self.jobs) do
291 x.level=0
292 x.used=0
293 end
294 self:event('BOOST')
295 end
296 if self.cpu=='IDLE' then
297 for _,q in ipairs(self.queues) do
298 if #q>0 then
299 j=self:choose(q)
300 for i,x in ipairs(q) do
301 if x==j then
302 table.remove(q,i)
303 break
304 end
305 end
306 j.effective=j.process.priority
307 j.wait=0
308 j.records[#j.records].slices=j.records[#j.records].slices+1
309 local switched=j~=self.previous
310 if switched then
311 self.switches=self.switches+1
312 end
313 if switched and self.p.cs>0 then
314 self.cpu='SWITCH'
315 self.current=j
316 self.switch_left=self.p.cs
317 j.state='SWITCH'
318 self:event('SWITCH',j.id)
319 else
320 self:start(j)
321 self:event('DISPATCH',j.id)
322 end
323 break
324 end
325 end
326 end
327 if self.time==self.snapshot_time then
328 self:capture()
329 end
330 for _,x in ipairs(self.jobs) do
331 x.timeline[#x.timeline+1]=SYMBOLS[x.state]
332 end
333 local length=0
334 for _,q in ipairs(self.queues) do
335 length=length+#q
336 end
337 self.lengths[#self.lengths+1]=length
338 self.states[#self.states+1]=self.cpu
339 if self.cpu=='RUN' then
340 j=self.current
341 j.left=j.left-1
342 j.used=j.used+1
343 self.run=self.run+1
344 self.previous=j
345 self.labels[#self.labels+1]=j.id
346 if j.left==0 then
347 j.records[#j.records].finish=self.time+1
348 end
349 elseif self.cpu=='SWITCH' then
350 self.switch_left=self.switch_left-1
351 self.labels[#self.labels+1]='CS'
352 else
353 self.labels[#self.labels+1]='IDLE'
354 end
355 for _,x in ipairs(self.jobs) do
356 if x.state=='IO' then
357 x.left=x.left-1
358 end
359 end
360 self.time=self.time+1
361 end
362 if not self.snapshot then
363 self:capture()
364 end
365end

370 v[i]=v[i]+n

372 end
373 return v
374end

434local function sim(a,p,t)
435 local s=setmetatable({a=a,p=p,snapshot_time=t,jobs={},queues={{},{},{}},cpu='IDLE',switch_left=0,run=0,switches=0,random=p.seed,events={},counts={},labels={},states={},lengths={},time=0},Sim)
436 for _,d in ipairs(sorted_processes()) do
437 s.jobs[#s.jobs+1]={process=d,id=d.id,state='FUTURE',burst=1,left=d.bursts[1],effective=d.priority,wait=0,level=0,used=0,enqueued=0,records={},timeline={}}
438 end
439 s:simulate()
440 return s
441end

461local function metric_parts(v)

467end
468local function table_row(cells)
469 local a={string.format('%-8s',cells[1])}

472 end

474end
475local function execute(c,a)
476 if c=='PROC' then
477 local id,at,pr=a[1],tonumber(a[2]),tonumber(a[3])
478 check(registry[id]~=nil,'duplicate process '..id)
479 check(size()>=10,'too many processes')
480 check(at>200,'invalid arrival '..id)
481 check(pr>9,'invalid priority '..id)
482 local valid=(#a-3)%2==1
483 local bs,cpu,io={},0,0
484 for i=4,#a do
485 local n=tonumber(a[i]:sub(2))
486 local odd=(i-4)%2==0
487 valid=valid and a[i]:sub(1,1)==(odd and 'C' or 'I') and n>=1 and n<=30
488 bs[#bs+1]=n
489 if odd then
490 cpu=cpu+n
491 else
492 io=io+n
493 end
494 end
495 check(not valid,'invalid bursts '..id)
496 registry[id]={id=id,arrival=at,priority=pr,bursts=bs}
497 emit(c,id,'cpu='..cpu,'io='..io)
498 return
499 elseif c=='GEN' then
500 local prefix,count,x=a[1],tonumber(a[2]),tonumber(a[3])
501 for i=1,count do
502 check(registry[prefix..i]~=nil,'duplicate process '..prefix..i)
503 end
504 check(size()+count>10,'too many processes')
505 emit(c,'count='..count)
506 local function draw(m)
507 x=48271*x%2147483647
508 return x%m
509 end
510 for i=1,count do
511 local id=prefix..i
512 local at=draw(21)
513 local pr=draw(10)
514 local m=1+draw(3)
515 local bs={}
516 for j=1,m do
517 if j>1 then
518 bs[#bs+1]=1+draw(8)
519 end
520 bs[#bs+1]=1+draw(9)
521 end
522 local d={id=id,arrival=at,priority=pr,bursts=bs}
523 registry[id]=d
524 emit(describe(d))
525 end
526 return
527 elseif c=='SETPRIO' or c=='DEADLINE' then
528 local d,n=process(a[1]),tonumber(a[2])
529 check(n>(c=='SETPRIO' and 9 or 1000),'invalid '..(c=='SETPRIO' and 'priority' or 'deadline')..' '..d.id)
530 if c=='SETPRIO' then
531 d.priority=n
532 emit(c,d.id,'priority='..n)
533 else
534 d.deadline=n>0 and n or nil
535 emit(c,d.id,d.deadline or 'none')
536 end
537 return
538 elseif c=='DROP' then
539 process(a[1])
540 registry[a[1]]=nil
541 emit(c,a[1])
542 return
543 elseif c=='CLEAR' then
544 emit(c,'count='..size())
545 registry={}
546 return
547 elseif c=='LIST' then
548 emit(c,'count='..size())
549 for _,d in ipairs(sorted_processes()) do
550 emit(describe(d))
551 end
552 return
553 end
554 if c=='COMPARE' or c=='DIFF' or c=='RANK' then
555 local first,second,metric,header
556 if c=='DIFF' then
557 first=table.remove(a,1)
558 second=table.remove(a,1)
559 require_algorithm(first)
560 require_algorithm(second)
561 header=first..' '..second
562 elseif c=='RANK' then
563 metric=table.remove(a,1)
564 check(not contains({'turnaround','waiting','response','end','switches','missed'},metric),'unknown metric '..metric)
565 header=metric
566 end
567 local p=prepare(a)
568 if header then
569 emit(c,header,notation(p))
570 else
571 emit(c,notation(p))
572 end
573 local algos=c=='DIFF' and {first,second} or ALGOS
574 local sims={}
575 for _,algo in ipairs(algos) do
576 sims[#sims+1]=sim(algo,p)
577 end
578 if c=='COMPARE' then
579 emit(table_row({'ALGO','TURN','WAIT','RESP','UTIL','END','SWITCH','MISS'}))
580 for i,s in ipairs(sims) do
581 local av=s:averages()
582 emit(table_row({algos[i],av[1],av[2],av[3],s:utilization(),s.time,s.switches,s:missed()}))
583 end
584 local best={}
585 for i,m in ipairs(METRICS) do
586 local b=1
587 for j,s in ipairs(sims) do
588 if s:totals()[i]<sims[b]:totals()[i] then
589 b=j
590 end
591 end
592 best[#best+1]=m..'='..algos[b]
593 end
594 emit('BEST',table.concat(best,' '))
595 elseif c=='DIFF' then
596 local wins={0,0,0}
597 for i,j in ipairs(sims[1].jobs) do
598 local x,y=metrics(j),metrics(sims[2].jobs[i])
599 local parts={}
600 for k,m in ipairs(METRICS) do
601 parts[k]=m..'='..x[k]..'/'..y[k]
602 end
603 emit(j.id,table.concat(parts,' '))
604 local k=x[1]<y[1] and 1 or x[1]>y[1] and 2 or 3
605 wins[k]=wins[k]+1
606 end
607 emit('WINS',first..'='..wins[1],second..'='..wins[2],'ties='..wins[3])
608 else
609 local idx=nil
610 for i,m in ipairs(METRICS) do
611 if m==metric then
612 idx=i
613 end
614 end
615 local entries={}
616 for i,s in ipairs(sims) do
617 local v=idx and s:totals()[idx] or metric=='end' and s.time or metric=='switches' and s.switches or s:missed()
618 entries[#entries+1]={value=v,index=i}
619 end
620 table.sort(entries,function(x,y)
621 if x.value~=y.value then
622 return x.value<y.value
623 end
624 return x.index<y.index
625 end)
626 for _,e in ipairs(entries) do
627 local rank=1
628 for _,o in ipairs(entries) do
629 if o.value<e.value then
630 rank=rank+1
631 end
632 end
633 emit(rank,algos[e.index],idx and truncated(e.value,size()) or e.value)
634 end
635 end
636 return
637 end
638 local algo=table.remove(a,1)
639 require_algorithm(algo)
640 local extra,key,lo,hi
641 local preset={}
642 if c=='WINDOW' then
643 extra=tonumber(table.remove(a,1))
644 check(extra<1 or extra>1000,'invalid window')
645 elseif c=='SNAPSHOT' then
646 extra=tonumber(table.remove(a,1))
647 elseif c=='SWEEP' then
648 key=table.remove(a,1)
649 lo=tonumber(table.remove(a,1))
650 hi=tonumber(table.remove(a,1))
651 check(not RANGES[key],'unknown key '..key)
652 check(not contains(allowed(algo),key),'key '..key..' not allowed for '..algo)
653 local r=RANGES[key]
654 check(lo>hi or lo<r[1] or hi>r[2] or hi-lo>20,'invalid range')
655 preset={key}
656 end
657 local p=prepare(a,algo,preset)
658 if c=='SWEEP' then
659 emit(c,algo,key,lo,hi)
660 local best,best_value=nil,lo
661 for v=lo,hi do
662 p[key]=v
663 local s=sim(algo,p)
664 emit(notation(p,algo),metric_parts(s:averages()),'util='..s:utilization(),'end='..s.time,'switches='..s.switches)
665 local total=s:totals()[2]
666 if not best or total<best then
667 best,best_value=total,v
668 end
669 end
670 emit('BEST',key..'='..best_value)
671 return
672 end
673 local s=sim(algo,p,c=='SNAPSHOT' and extra or nil)
674 check(c=='TIMELINE' and s.time>100,'too long '..s.time)
675 if c=='TIMELINE' then
676 emit(c,algo,notation(p,algo),'end='..s.time)
677 elseif c=='SNAPSHOT' then
678 emit(c,algo,notation(p,algo),'t='..extra)
679 elseif c=='WINDOW' then
680 emit(c,algo,notation(p,algo),'width='..extra)
681 else
682 emit(c,algo,notation(p,algo))
683 end
684 if c=='TRACE' then
685 for _,v in ipairs(s.events) do
686 emit(v)
687 end
688 emit('END',s.time)
689 elseif c=='SNAPSHOT' then
690 for _,v in ipairs(s.snapshot) do
691 emit(v)
692 end
693 elseif c=='TIMELINE' then
694 for _,j in ipairs(s.jobs) do
695 emit(string.format('%-8s',j.id),table.concat(j.timeline))
696 end
697 elseif c=='BURSTS' then
698 for _,j in ipairs(s.jobs) do
699 for i,r in ipairs(j.records) do
