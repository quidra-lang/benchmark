67allowed(a, k) = k == "cs" || (k == "q" && a in ("RR", "LOTTERY")) || (k == "seed" && a == "LOTTERY") || (k == "k" && a == "PRIORITY") || (k == "s" && a == "MLFQ")
68valid(k, v) = k == "cs" ? 0 <= v <= 5 : k == "q" ? 1 <= v <= 20 : k == "k" ? 1 <= v <= 50 : k == "s" ? 0 <= v <= 500 : k == "seed" && 1 <= v <= 2147483646

71function simulate(p, a, param)
72 n = length(p)
73 states = [ProcessState("FUTURE", 1, x.bursts[1], x.priority, 0, 0, 0, 0, -1, -1, 0) for x in p]
74 bursts = [[BurstRecord(0, -1, 0, 0) for _ in 1:cld(length(x.bursts), 2)] for x in p]
75 queues = [Int[], Int[], Int[]]
76 cpu = previous = 0
77 mode = sw = run = switches = busy = switching = idle = qmax = qsum = 0
78 x = param["seed"]
79 labels = String[]
80 events = String[]
81 shots = CPUSnapshot[]
82 t = 0
83 deadline(i) = p[i].deadline == 0 ? typemax(Int) : p[i].arrival+p[i].deadline
84 while true
85 log(s) = push!(events, "$t $s")
86 function enqueue(i)
87 z = states[i]
88 z.state = "READY"
89 z.enqueued = t
90 push!(queues[a == "MLFQ" ? z.level+1 : 1], i)
91 end
92 function start(i)
93 cpu = i
94 mode = 2
95 run = 0
96 z = states[i]
97 z.state = "RUN"
98 z.start<0 && (z.start = t)
99 b = bursts[i][cld(z.burst, 2)]
100 b.start<0 && (b.start = t)
101 end
102 if a == "PRIORITY"
103 for i in queues[1]
104 z = states[i]
105 z.wait += 1
106 if z.wait == param["k"]
107 z.wait = 0
108 if z.effective>0
109 z.effective -= 1
110 log("AGE $(p[i].id) $(z.effective)")
111 end
112 end
113 end
114 end
115 for i in eachindex(p)
116 z = states[i]
117 if z.state == "IO" && z.left == 0
118 z.burst += 1
119 z.left = p[i].bursts[z.burst]
120 bursts[i][cld(z.burst, 2)].ready = t
121 enqueue(i)
122 log("IODONE $(p[i].id)")
123 end
124 end
125 for i in eachindex(p)
126 if states[i].state == "FUTURE" && p[i].arrival == t
127 bursts[i][1].ready = t
128 enqueue(i)
129 log("ARRIVE $(p[i].id)")
130 end
131 end
132 if mode == 1 && sw == 0
133 start(cpu)
134 log("START $(p[cpu].id)")
135 elseif mode == 2
136 i = cpu
137 z = states[i]
138 if z.left == 0 && z.burst == length(p[i].bursts)
139 z.state = "DONE"
140 z.finish = t
141 log("DONE $(p[i].id)")
142 mode = 0
143 cpu = 0
144 else
145 demoted = false
146 if a == "MLFQ" && z.level<2 && z.used == (4, 8)[z.level+1]
147 z.level += 1
148 z.used = 0
149 demoted = true
150 log("DEMOTE $(p[i].id) $(z.level)")
151 end
152 preempt = any((a == "SRTF" && states[j].left<z.left) || (a == "PRIORITY" && states[j].effective<z.effective) || (a == "EDF" && deadline(j)<deadline(i)) || (a == "MLFQ" && l-1<z.level) for (l, q) in enumerate(queues) for j in q)
153 if z.left == 0
154 z.burst += 1
155 z.left = p[i].bursts[z.burst]
156 z.state = "IO"
157 log("IO $(p[i].id) $(z.left)")
158 mode = 0
159 cpu = 0
160 elseif demoted
161 enqueue(i)
162 mode = 0
163 cpu = 0
164 elseif a in ("RR", "LOTTERY") && run == param["q"]
165 enqueue(i)
166 log("EXPIRE $(p[i].id)")
167 mode = 0
168 cpu = 0
169 elseif preempt
170 enqueue(i)
171 log("PREEMPT $(p[i].id)")
172 mode = 0
173 cpu = 0
174 end
175 end
176 end
177 if all(z->z.state == "DONE", states)
178 push!(shots, CPUSnapshot(deepcopy(states), [Int[], Int[], Int[]], 0, 0))
179 break
180 end
181 if a == "MLFQ" && param["s"]>0 && t>0 && t%param["s"] == 0
182 append!(queues[1], queues[2])
183 append!(queues[1], queues[3])
184 empty!(queues[2])
185 empty!(queues[3])
186 for z in states
187 z.level = 0
188 z.used = 0
189 end
190 log("BOOST")
191 end
192 if mode == 0
193 l = findfirst(q->!isempty(q), queues)
194 if !isnothing(l)
195 q = queues[l]
196 selected = 1
197 if a == "LOTTERY"
198 total = sum(10-p[i].priority for i in q)
199 x = 48271x%2147483647
200 draw = x%total
201 count = 0
202 for (j, i) in enumerate(q)
203 count += 10-p[i].priority
204 if count>draw
205 selected = j
206 break
207 end
208 end
209 else
210 for j in 2:length(q)
211 i = q[j]
212 b = q[selected]
213 u = states[i]
214 v = states[b]
215 better = a in ("SJF", "SRTF") ? u.left<v.left : a == "PRIORITY" ? u.effective<v.effective : a == "EDF" ? deadline(i)<deadline(b) : a == "HRRN" ? (t-u.enqueued+u.left)*v.left>(t-v.enqueued+v.left)*u.left : false
216 better && (selected = j)
217 end
218 end
219 i = q[selected]
220 deleteat!(q, selected)
221 z = states[i]
222 z.effective = p[i].priority
223 z.wait = 0
224 z.slices += 1
225 bursts[i][cld(z.burst, 2)].slices += 1
226 change = i != previous
227 change && (switches += 1)
228 if change && param["cs"]>0
229 cpu = i
230 mode = 1
231 sw = param["cs"]
232 z.state = "SWITCH"
233 log("SWITCH $(p[i].id)")
234 else
235 start(i)
236 log("DISPATCH $(p[i].id)")
237 end
238 end
239 end
240 push!(shots, CPUSnapshot(deepcopy(states), deepcopy(queues), cpu, mode))
241 qlen = sum(length, queues)
242 qmax = max(qmax, qlen)
243 qsum += qlen
244 label = "IDLE"
245 if mode == 2
246 z = states[cpu]
247 z.left -= 1
248 z.used += 1
249 run += 1
250 previous = cpu
251 label = p[cpu].id
252 busy += 1
253 z.left == 0 && (bursts[cpu][cld(z.burst, 2)].finish = t+1)
254 elseif mode == 1
255 sw -= 1
256 label = "CS"
257 switching += 1
258 else
259 idle += 1
260 end
261 push!(labels, label)
262 for z in states
263 z.state == "IO" && (z.left -= 1)
264 end
265 t += 1
266 end
267 totals = zeros(Int, 3)
268 met = missed = 0
269 for i in eachindex(p)
270 z = states[i]
271 u = p[i]
272 turn = z.finish-u.arrival
273 totals.+= [turn, turn-sum(u.bursts), z.start-u.arrival]
274 if u.deadline>0
275 z.finish <= u.arrival+u.deadline ? (met += 1) : (missed += 1)
276 end
277 end
278 ScheduleResult(p, states, bursts, labels, events, shots, t, switches, busy, switching, idle, qmax, qsum, met, missed, totals)
279end
280metrics(r, i) = let p = r.p[i], z = r.states[i]
281 t = z.finish-p.arrival
282 [t, t-sum(p.bursts), z.start-p.arrival]
283end
284summaryline(r) = let n = length(r.p)
285 "turnaround=$(ratio(r.totals[1],n)) waiting=$(ratio(r.totals[2],n)) response=$(ratio(r.totals[3],n)) util=$(ratio(r.busy*100,r.finish)) end=$(r.finish) switches=$(r.switches)"
286end

288function listline(p)
289 b = join([string(isodd(i) ? "C" : "I", v) for (i, v) in enumerate(p.bursts)], ",")
290 d = p.deadline == 0 ? "none" : string(p.deadline)
291 "$(p.id) arrival=$(p.arrival) priority=$(p.priority) deadline=$d bursts=$b"
292end

294function command!(registry, w)
295 c = w[1]
296 n(i) = parse(Int, w[i])
297 p = length(w)>1 ? get(registry, w[2], nothing) : nothing
298 if c == "PROC"
299 !isnothing(p) && fail("duplicate process $(w[2])")
300 length(registry) == 10 && fail("too many processes")
301 n(3)>200 && fail("invalid arrival $(w[2])")
302 n(4)>9 && fail("invalid priority $(w[2])")
303 b = [parse(Int, s[2:end]) for s in w[5:end]]
304 (iseven(length(b)) || any(!(1 <= v <= 30) || w[i+4][1] != (isodd(i) ? 'C' : 'I') for (i, v) in enumerate(b))) && fail("invalid bursts $(w[2])")
305 p = CPUProcess(w[2], n(3), n(4), 0, b)
306 registry[p.id] = p
307 emit(c, p.id, "cpu=$(sum(b[1:2:end])) io=$(sum(b[2:2:end]))")
308 return
309 elseif c == "GEN"
310 generated_count = n(3)
311 ids = [w[2]*string(i) for i in 1:generated_count]
312 for id in ids
313 haskey(registry, id) && fail("duplicate process $id")
314 end
315 length(registry)+generated_count>10 && fail("too many processes")
316 emit(c, "count=$generated_count")
317 x = n(4)
318 next() = (x = 48271x%2147483647)
319 for id in ids
320 arrival = next()%21
321 priority = next()%10
322 m = 1+next()%3
323 b = Int[]
324 for j in 1:m
325 j>1 && push!(b, 1+next()%8)
326 push!(b, 1+next()%9)
327 end
328 p = CPUProcess(id, arrival, priority, 0, b)
329 registry[id] = p
330 emit(listline(p))
331 end
332 return
333 elseif c in ("SETPRIO", "DEADLINE", "DROP")
334 isnothing(p) && fail("unknown process $(w[2])")
335 if c == "DROP"
336 delete!(registry, p.id)
337 emit(c, p.id)
338 elseif c == "SETPRIO"
339 n(3)>9 && fail("invalid priority $(p.id)")
340 p.priority = n(3)
341 emit(c, p.id, "priority=$(p.priority)")
342 else
343 n(3)>1000 && fail("invalid deadline $(p.id)")
344 p.deadline = n(3)
345 emit(c, p.id, p.deadline == 0 ? "none" : p.deadline)
346 end
347 return
348 elseif c == "CLEAR"
349 emit(c, "count=$(length(registry))")
350 empty!(registry)
351 return
352 elseif c == "LIST"
353 emit(c, "count=$(length(registry))")
354 for id in sort(collect(keys(registry)))
355 emit(listline(registry[id]))
356 end
357 return
358 end
359 a = ""
360 metric = ""
361 start = 3
362 multi = c in ("COMPARE", "RANK", "DIFF")
363 if c == "COMPARE"
364 start = 2
365 elseif c == "RANK"
366 metric = w[2]
367 metric in ("turnaround", "waiting", "response", "end", "switches", "missed") || fail("unknown metric $metric")
368 else
369 a = w[2]
370 a in ALGORITHMS || fail("unknown algorithm $a")
371 if c == "DIFF"
372 w[3] in ALGORITHMS || fail("unknown algorithm $(w[3])")
373 start = 4
374 end
375 end
376 c in ("SNAPSHOT", "WINDOW") && (start = 4)
377 c == "WINDOW" && !(1 <= n(3) <= 1000) && fail("invalid window")
378 seen = Set{String}()
379 if c == "SWEEP"
380 start = 6
381 k = w[3]
382 k in KEYS || fail("unknown key $k")
383 allowed(a, k) || fail("key $k not allowed for $a")
384 (n(4)>n(5) || !valid(k, n(4)) || !valid(k, n(5)) || n(5)-n(4)>20) && fail("invalid range")
385 push!(seen, k)
386 end
387 param = Dict("cs"=>0, "q"=>4, "k"=>5, "s"=>0, "seed"=>1)
388 for word in w[start:end]
389 kv = split(word, '=')
390 k = String(kv[1])
391 v = parse(Int, kv[2])
392 k in KEYS || fail("unknown key $k")
393 !multi && !allowed(a, k) && fail("key $k not allowed for $a")
394 k in seen && fail("duplicate key $k")
395 valid(k, v) || fail("invalid $k")
396 push!(seen, k)
397 param[k] = v
398 end
399 isempty(registry) && fail("no processes")
400 ps = [registry[id] for id in sort(collect(keys(registry)))]
401 N = length(ps)
402 if c == "SWEEP"
403 emit(c, a, w[3], n(4), n(5))
404 best = typemax(Int)
405 bestv = 0
406 for v in n(4):n(5)
407 param[w[3]] = v
408 r = simulate(ps, a, param)
409 emit(notation(a, param), summaryline(r))
410 if r.totals[2]<best
411 best = r.totals[2]
412 bestv = v
413 end
414 end
415 emit("BEST", "$(w[3])=$bestv")
416 return
417 end
418 if multi
419 if c == "DIFF"
420 r = simulate(ps, a, param)
421 s = simulate(ps, w[3], param)
422 emit(c, a, w[3], notation("", param, true))
423 wins = loss = ties = 0
424 for i in eachindex(ps)
425 u = metrics(r, i)
426 v = metrics(s, i)
427 emit(ps[i].id, "turnaround=$(u[1])/$(v[1]) waiting=$(u[2])/$(v[2]) response=$(u[3])/$(v[3])")
428 u[1]<v[1] ? (wins += 1) : u[1]>v[1] ? (loss += 1) : (ties += 1)
429 end
430 emit("WINS", "$a=$wins $(w[3])=$loss ties=$ties")
431 return
432 end
433 results = [simulate(ps, a, param) for a in ALGORITHMS]
434 if c == "COMPARE"
435 emit(c, notation("", param, true))
436 row(v) = println(rpad(v[1], 8)*join([" "*lpad(x, 8) for x in v[2:end]], ""))
437 row(["ALGO", "TURN", "WAIT", "RESP", "UTIL", "END", "SWITCH", "MISS"])
438 for (i, r) in enumerate(results)
439 row([ALGORITHMS[i], ratio(r.totals[1], N), ratio(r.totals[2], N), ratio(r.totals[3], N), ratio(r.busy*100, r.finish), string(r.finish), string(r.switches), string(r.missed)])
440 end
441 emit("BEST", ("$k=$(ALGORITHMS[argmin([r.totals[j] for r in results])])" for (j, k) in enumerate(("turnaround", "waiting", "response")))...)
442 else
443 emit(c, metric, notation("", param, true))
444 j = findfirst(==(metric), ["turnaround", "waiting", "response"])
445 values = [!isnothing(j) ? r.totals[j] : metric == "end" ? r.finish : metric == "switches" ? r.switches : r.missed for r in results]
446 for i in sortperm(values)
447 rank = count(<(values[i]), values)+1
448 emit(rank, ALGORITHMS[i], isnothing(j) ? string(values[i]) : ratio(values[i], N))
449 end
450 end
451 return
452 end
453 r = simulate(ps, a, param)
454 note = notation(a, param)
455 T = r.finish
456 if c == "TRACE"
457 emit(c, a, note)
458 foreach(emit, r.events)
459 emit("END", T)
460 elseif c == "TIMELINE"
461 T>100 && fail("too long $T")
462 emit(c, a, note, "end=$T")
463 chars = Dict("READY"=>'.', "RUN"=>'#', "SWITCH"=>'s', "IO"=>'i', "FUTURE"=>'-', "DONE"=>'-')
464 for i in eachindex(ps)
465 emit(rpad(ps[i].id, 8), join([chars[r.shots[t].states[i].state] for t in 1:T]))
466 end
467 elseif c == "SNAPSHOT"
468 t = n(3)
469 s = r.shots[min(t, T)+1]
470 emit(c, a, note, "t=$t")
471 s.mode == 0 ? emit("CPU IDLE") : emit("CPU", s.mode == 1 ? "SWITCH" : "RUN", ps[s.cpu].id)
472 for l in 1:(a == "MLFQ" ? 3 : 1)
473 ids = [ps[i].id for i in s.queues[l]]
474 emit(a == "MLFQ" ? "QUEUE$(l-1)" : "QUEUE", isempty(ids) ? "none" : join(ids, " "))
475 end
476 for (i, p) in enumerate(ps)
477 z = s.states[i]
478 if z.state == "FUTURE"
479 emit(p.id, "FUTURE at=$(p.arrival)")
480 elseif z.state == "DONE"
481 emit(p.id, "DONE at=$(z.finish)")
482 else
483 line = "$(p.id) $(z.state) burst=$(z.burst) left=$(z.left)"
484 a == "PRIORITY" && (line*= " prio=$(z.effective)")
485 a == "MLFQ" && (line*= " level=$(z.level) used=$(z.used)")
486 emit(line)
487 end
488 end
489 elseif c == "BURSTS"
490 emit(c, a, note)
491 for (i, p) in enumerate(ps), (j, b) in enumerate(r.bursts[i])
492 len = p.bursts[2j-1]
493 emit(p.id, "#$j len=$len ready=$(b.ready) start=$(b.start) end=$(b.finish) wait=$(b.finish-b.ready-len) slices=$(b.slices)")
494 end
495 elseif c == "WINDOW"
496 width = n(3)
497 emit(c, a, note, "width=$width")
498 for lo in 0:width:T-1
499 hi = min(lo+width, T)
500 shots = r.shots[lo+1:hi]
501 labels = [ps[s.cpu].id for s in shots if s.mode == 2]
502 cs = count(s->s.mode == 1, shots)
503 idle = count(s->s.mode == 0, shots)
504 busy = hi-lo-cs-idle
505 counts = [count(==(p.id), labels) for p in ps]
506 top = maximum(counts) == 0 ? "none" : ps[argmax(counts)].id
507 emit("$lo-$hi busy=$busy switch=$cs idle=$idle util=$(ratio(busy*100,hi-lo)) top=$top")
508 end
509 elseif c == "STATS"
510 emit(c, a, note)
511 for (j, k) in enumerate(("turnaround", "waiting", "response"))
512 values = [metrics(r, i)[j] for i in eachindex(ps)]
513 lo = argmin(values)
514 hi = argmax(values)
515 v = sort(values)
516 med = isodd(N) ? 2v[cld(N, 2)] : v[N÷2]+v[N÷2+1]
517 emit("$k min=$(values[lo]) ($(ps[lo].id)) max=$(values[hi]) ($(ps[hi].id)) median=$(med÷2).$(med%2*5) total=$(r.totals[j])")
518 end
519 elseif c == "RUN"
520 emit(c, a, note)
521 segments = String[]
522 lo = 0
523 while lo<T
524 hi = lo+1
525 while hi<T && r.labels[hi+1] == r.labels[lo+1]
526 hi += 1
527 end
528 push!(segments, "[$lo-$hi $(r.labels[lo+1])]")
529 lo = hi
530 end
531 for i in 1:8:length(segments)
532 emit("GANTT", join(segments[i:min(i+7, length(segments))], ""))
533 end
534 for (i, p) in enumerate(ps)
535 z = r.states[i]
536 m = metrics(r, i)
537 line = "$(p.id) arrival=$(p.arrival) done=$(z.finish) turnaround=$(m[1]) waiting=$(m[2]) response=$(m[3]) slices=$(z.slices)"
538 if p.deadline>0
539 d = p.arrival+p.deadline
540 line*= " deadline=$d late=$(max(0,z.finish-d))"
541 end
542 emit(line)
543 end
544 emit("AVERAGE turnaround=$(ratio(r.totals[1],N)) waiting=$(ratio(r.totals[2],N)) response=$(ratio(r.totals[3],N))")
545 emit("CPU end=$T busy=$(r.busy) switch=$(r.switching) idle=$(r.idle) switches=$(r.switches) util=$(ratio(r.busy*100,T))")
546 emit("QUEUE max=$(r.qmax) avg=$(ratio(r.qsum,T))")
547 counts = Dict(k=>count(s->split(s)[2] == k, r.events) for k in ("PREEMPT", "EXPIRE", "DEMOTE", "BOOST", "AGE"))
548 emit("EVENTS preempt=$(counts["PREEMPT"]) expire=$(counts["EXPIRE"]) demote=$(counts["DEMOTE"]) boost=$(counts["BOOST"]) age=$(counts["AGE"])")
549 emit("DEADLINE met=$(r.met) missed=$(r.missed)")
550 end
551end

553function main()
554 registry = Dict{String, CPUProcess}()
555 for _ in 1:parse(Int, readline())
556 w = String.(split(readline()))
557 try
558 command!(registry, w)
559 catch err
560 err isa CommandError || rethrow()
561 emit("ERROR", err.reason)
562 end
563 end
564end
