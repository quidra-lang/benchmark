28simulate <- function(p, a, param) {
29 n <- length(p)
30 states <- lapply(p, function(x) mutable(state = "FUTURE", burst = 1L, left = x$bursts[1], effective = x$priority, wait = 0, level = 0L, used = 0, enqueued = 0, start = -1, finish = -1, slices = 0))
31 bursts <- lapply(p, function(x) lapply(seq_len((length(x$bursts) + 1L) %/% 2L), function(j) mutable(ready = 0, start = -1, finish = 0, slices = 0)))
32 queues <- list(integer(), integer(), integer())
33 cpu <- previous <- 0L
34 mode <- sw <- run <- switches <- busy <- switching <- idle <- qmax <- qsum <- 0
35 x <- param$seed
36 labels <- events <- character()
37 shots <- list()
38 t <- 0
39 deadline <- function(i) if (p[[i]]$deadline == 0) 2^52 else p[[i]]$arrival + p[[i]]$deadline
40 log <- function(s) events <<- c(events, paste(t, s))
41 enqueue <- function(i) {
42 z <- states[[i]]
43 z$state <- "READY"
44 z$enqueued <- t
45 l <- if (a == "MLFQ") z$level + 1L else 1L
46 queues[[l]] <<- c(queues[[l]], i)
47 }
48 start <- function(i) {
49 cpu <<- i
50 mode <<- 2
51 run <<- 0
52 z <- states[[i]]
53 z$state <- "RUN"
54 if (z$start<0) z$start <- t
55 b <- bursts[[i]][[(z$burst + 1L) %/% 2L]]
56 if (b$start<0) b$start <- t
57 }
58 snapshot <- function() list(states = lapply(states, as.list), queues = lapply(queues, identity), cpu = cpu, mode = mode)
59 repeat {
60 if (a == "PRIORITY") for (i in queues[[1]]) {
61 z <- states[[i]]
62 z$wait <- z$wait + 1
63 if (z$wait == param$k) {
64 z$wait <- 0
65 if (z$effective>0) {
66 z$effective <- z$effective - 1
67 log(paste("AGE", p[[i]]$id, z$effective))
68 }
69 }
70 }
71 for (i in seq_len(n)) {
72 z <- states[[i]]
73 if (z$state == "IO" && z$left == 0) {
74 z$burst <- z$burst + 1L
75 z$left <- p[[i]]$bursts[z$burst]
76 bursts[[i]][[(z$burst + 1L) %/% 2L]]$ready <- t
77 enqueue(i)
78 log(paste("IODONE", p[[i]]$id))
79 }
80 }
81 for (i in seq_len(n)) if (states[[i]]$state == "FUTURE" && p[[i]]$arrival == t) {
82 bursts[[i]][[1]]$ready <- t
83 enqueue(i)
84 log(paste("ARRIVE", p[[i]]$id))
85 }
86 if (mode == 1 && sw == 0) {
87 start(cpu)
88 log(paste("START", p[[cpu]]$id))
89 } else if (mode == 2) {
90 i <- cpu
91 z <- states[[i]]
92 if (z$left == 0 && z$burst == length(p[[i]]$bursts)) {
93 z$state <- "DONE"
94 z$finish <- t
95 log(paste("DONE", p[[i]]$id))
96 mode <- 0
97 cpu <- 0L
98 } else {
99 demoted <- FALSE
100 if (a == "MLFQ" && z$level<2 && z$used == c(4, 8)[z$level + 1L]) {
101 z$level <- z$level + 1L
102 z$used <- 0
103 demoted <- TRUE
104 log(paste("DEMOTE", p[[i]]$id, z$level))
105 }
106 preempt <- FALSE
107 for (l in seq_len(3L)) for (j in queues[[l]]) {
108 u <- states[[j]]
109 if ((a == "SRTF" && u$left<z$left) || (a == "PRIORITY" && u$effective<z$effective) || (a == "EDF" && deadline(j)<deadline(i)) || (a == "MLFQ" && l - 1L<z$level)) preempt <- TRUE
110 }
111 if (z$left == 0) {
112 z$burst <- z$burst + 1L
113 z$left <- p[[i]]$bursts[z$burst]
114 z$state <- "IO"
115 log(paste("IO", p[[i]]$id, z$left))
116 mode <- 0
117 cpu <- 0L
118 } else if (demoted) {
119 enqueue(i)
120 mode <- 0
121 cpu <- 0L
122 } else if (a %in% c("RR", "LOTTERY") && run == param$q) {
123 enqueue(i)
124 log(paste("EXPIRE", p[[i]]$id))
125 mode <- 0
126 cpu <- 0L
127 } else if (preempt) {
128 enqueue(i)
129 log(paste("PREEMPT", p[[i]]$id))
130 mode <- 0
131 cpu <- 0L
132 }
133 }
134 }
135 if (all(vapply(states, function(z) z$state == "DONE", logical(1)))) {
136 shots[[length(shots) + 1L]] <- snapshot()
137 break
138 }
139 if (a == "MLFQ" && param$s>0 && t>0 && t %% param$s == 0) {
140 queues[[1]] <- unlist(queues, use.names = FALSE)
141 queues[[2]] <- queues[[3]] <- integer()
142 for (z in states) {
143 z$level <- 0L
144 z$used <- 0
145 }
146 log("BOOST")
147 }
148 if (mode == 0) {
149 nonempty <- which(lengths(queues)>0)
150 if (length(nonempty)) {
151 l <- nonempty[1]
152 q <- queues[[l]]
153 selected <- 1L
154 if (a == "LOTTERY") {
155 total <- sum(vapply(q, function(i) 10 - p[[i]]$priority, numeric(1)))
156 x <- (48271 * x) %% 2147483647
157 draw <- x %% total
158 count <- 0
159 for (j in seq_along(q)) {
160 count <- count + 10 - p[[q[j]]]$priority
161 if (count>draw) {
162 selected <- j
163 break
164 }
165 }
166 } else if (length(q)>1L) for (j in 2:length(q)) {
167 i <- q[j]
168 b <- q[selected]
169 u <- states[[i]]
170 v <- states[[b]]
171 better <- switch(a, SJF = u$left<v$left, SRTF = u$left<v$left, PRIORITY = u$effective<v$effective, EDF = deadline(i)<deadline(b), HRRN = (t - u$enqueued + u$left) * v$left>(t - v$enqueued + v$left) * u$left, FALSE)
172 if (better) selected <- j
173 }
174 i <- q[selected]
175 queues[[l]] <- q[-selected]
176 z <- states[[i]]
177 z$effective <- p[[i]]$priority
178 z$wait <- 0
179 z$slices <- z$slices + 1
180 b <- bursts[[i]][[(z$burst + 1L) %/% 2L]]
181 b$slices <- b$slices + 1
182 change <- i != previous
183 if (change) switches <- switches + 1
184 if (change && param$cs>0) {
185 cpu <- i
186 mode <- 1
187 sw <- param$cs
188 z$state <- "SWITCH"
189 log(paste("SWITCH", p[[i]]$id))
190 } else {
191 start(i)
192 log(paste("DISPATCH", p[[i]]$id))
193 }
194 }
195 }
196 shots[[length(shots) + 1L]] <- snapshot()
197 qlen <- sum(lengths(queues))
198 qmax <- max(qmax, qlen)
199 qsum <- qsum + qlen
200 label <- "IDLE"
201 if (mode == 2) {
202 z <- states[[cpu]]
203 z$left <- z$left - 1
204 z$used <- z$used + 1
205 run <- run + 1
206 previous <- cpu
207 label <- p[[cpu]]$id
208 busy <- busy + 1
209 if (z$left == 0) bursts[[cpu]][[(z$burst + 1L) %/% 2L]]$finish <- t + 1
210 } else if (mode == 1) {
211 sw <- sw - 1
212 label <- "CS"
213 switching <- switching + 1
214 } else idle <- idle + 1
215 labels <- c(labels, label)
216 for (z in states) if (z$state == "IO") z$left <- z$left - 1
217 t <- t + 1
218 }
219 totals <- numeric(3)
220 met <- missed <- 0
221 for (i in seq_len(n)) {
222 z <- states[[i]]
223 u <- p[[i]]
224 turn <- z$finish - u$arrival
225 totals <- totals + c(turn, turn - sum(u$bursts), z$start - u$arrival)
226 if (u$deadline>0) {
227 if (z$finish <= u$arrival + u$deadline) met <- met + 1 else missed <- missed + 1
228 }
229 }
230 list(p = p, states = states, bursts = bursts, labels = labels, events = events, shots = shots, finish = t, switches = switches, busy = busy, switching = switching, idle = idle, qmax = qmax, qsum = qsum, met = met, missed = missed, totals = totals)
231}

252 if (n(3)>200) fail(paste("invalid arrival", w[2]))
253 if (n(4)>9) fail(paste("invalid priority", w[2]))
254 b <- as.numeric(substring(w[-seq_len(4L)], 2))
255 letters <- substring(w[-seq_len(4L)], 1, 1)

257 p <- list(id = w[2], arrival = n(3), priority = n(4), deadline = 0, bursts = b)

270 x <<- (48271 * x) %% 2147483647

293 } else if (c == "SETPRIO") {
294 if (n(3)>9) fail(paste("invalid priority", p$id))

311 if (c == "LIST") {
312 emit(c, paste0("count=", length(registry)))
313 for (id in sort(names(registry), method = "radix")) emit(listline(registry[[id]]))
314 return(invisible(NULL))
315 }
316 a <- metric <- ""
317 start <- 3L
318 multi <- c %in% c("COMPARE", "RANK", "DIFF")
319 if (c == "COMPARE") start <- 2L else if (c == "RANK") {
320 metric <- w[2]
321 if (!(metric %in% c("turnaround", "waiting", "response", "end", "switches", "missed"))) fail(paste("unknown metric", metric))
322 } else {
323 a <- w[2]
324 if (!(a %in% algorithms)) fail(paste("unknown algorithm", a))
325 if (c == "DIFF") {
326 if (!(w[3] %in% algorithms)) fail(paste("unknown algorithm", w[3]))
327 start <- 4L
328 }
329 }
330 if (c %in% c("SNAPSHOT", "WINDOW")) start <- 4L
331 if (c == "WINDOW" && (n(3)<1 || n(3)>1000)) fail("invalid window")
332 seen <- character()
333 if (c == "SWEEP") {
334 start <- 6L
335 k <- w[3]
336 if (!(k %in% keys)) fail(paste("unknown key", k))
337 if (!allowed(a, k)) fail(paste("key", k, "not allowed for", a))
338 if (n(4)>n(5) || !valid(k, n(4)) || !valid(k, n(5)) || n(5) - n(4)>20) fail("invalid range")
339 seen <- k
340 }
341 param <- list(cs = 0, q = 4, k = 5, s = 0, seed = 1)
342 if (start <= length(w)) for (word in w[start:length(w)]) {
343 kv <- strsplit(word, "=", fixed = TRUE)[[1]]
344 k <- kv[1]
345 v <- as.numeric(kv[2])
346 if (!(k %in% keys)) fail(paste("unknown key", k))
347 if (!multi && !allowed(a, k)) fail(paste("key", k, "not allowed for", a))
348 if (k %in% seen) fail(paste("duplicate key", k))
349 if (!valid(k, v)) fail(paste("invalid", k))
350 seen <- c(seen, k)
351 param[[k]] <- v
352 }
353 if (!length(registry)) fail("no processes")
354 ps <- registry[sort(names(registry), method = "radix")]
355 N <- length(ps)
356 if (c == "SWEEP") {
357 emit(c, a, w[3], n(4), n(5))
358 best <- Inf
359 bestv <- 0
360 for (v in seq(n(4), n(5))) {
361 param[[w[3]]] <- v

363 emit(notation(a, param), summary_line(r))

433 if (a == "MLFQ") line <- paste(line, paste0("level=", z$level), paste0("used=", z$used))

441 len <- ps[[i]]$bursts[2 * j - 1]
