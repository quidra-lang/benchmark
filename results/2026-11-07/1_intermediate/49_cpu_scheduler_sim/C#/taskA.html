5static class Rules
6{
7 public static readonly string[] Algorithms = {
8 "FCFS", "SJF", "SRTF", "RR", "PRIORITY", "HRRN", "EDF", "LOTTERY", "MLFQ"
9 };
10 public static readonly string[] Keys = {
11 "cs", "q", "k", "s", "seed"
12 }, Metrics = {
13 "turnaround", "waiting", "response"
14 }, EventKinds = {
15 "PREEMPT", "EXPIRE", "DEMOTE", "BOOST", "AGE"
16 };
17 public static bool Allowed(string a, string k) => k == "cs" || k == "q" && (a is "RR" or "LOTTERY") || k == "k" && a == "PRIORITY" || k == "s" && a == "MLFQ" || k == "seed" && a == "LOTTERY";
18 public static bool Valid(string k, long v) => k switch {
19 "cs" => v <= 5,
20 "q" => v >= 1 && v <= 20,
21 "k" => v >= 1 && v <= 50,
22 "s" => v <= 500,
23 "seed" => v >= 1 && v <= 2147483646, _ => false
24 };
25 public static string Decimal(long n, long d) {
26 long x = n * 100 / d;
27 return $"{x / 100}.{x % 100:D2}";
28 }
29 public static void Require(bool good, string reason) {
30 if (!good) throw new ArgumentException(reason);
31 }
32 public static string Notation(Dictionary<string, long> p, string? a = null) => string.Join(" ", Keys.Where(k => a == null || Allowed(a, k)).Select(k => $"{k}={p[k]}"));
33 public static string Row(IEnumerable<string> cells) => string.Join(" ", cells.Select((s, i) => i == 0 ? s.PadRight(8) : s.PadLeft(8)));
34}

58sealed class Job
59{
60 public Process Process;
61 public string State = "FUTURE";
62 public int Burst, Left, Effective, Wait, Level, Usage, Enqueued, Start = -1, Finish;
63 public List<BurstRecord> Records = new();
64 public List<char> Timeline = new();
65 public Job(Process p) {
66 Process = p;
67 Left = p.Bursts[0];
68 Effective = p.Priority;
69 }

71 public int[] Metrics => new[] {
72 Finish - Process.Arrival, Finish - Process.Arrival - Process.Bursts.Sum(), Start - Process.Arrival

80sealed class Simulation
81{
82 public List<Job> Jobs;
83 readonly string algo;
84 readonly Dictionary<string, long> p;
85 readonly int? requested;
86 List<Job>[] queues = {
87 new(), new(), new()
88 };
89 string cpu = "IDLE";
90 Job? current, previous;
91 int switchLeft, runLength, time;
92 long random;
93 public int End, Switches;
94 public List<string> Events = new(), Labels = new(), CpuStates = new(), Snapshot = new();
95 public List<int> QueueLengths = new();
96 public Dictionary<string, int> Counts = new();
97 public Simulation(IEnumerable<Process> processes, string a, Dictionary<string, long> parameters, int? snapshot = null) {
98 Jobs = processes.OrderBy(x => x.Id, StringComparer.Ordinal).Select(x => new Job(x)).ToList();
99 algo = a;
100 p = parameters;
101 requested = snapshot;
102 random = p["seed"];
103 Run();
104 }
105 void Record(string kind, string details = "") {
106 Events.Add($"{time} {kind}" + (details.Length == 0 ? "" : " " + details));
107 Counts[kind] = Counts.GetValueOrDefault(kind) + 1;
108 }
109 void Enqueue(Job j) {
110 j.State = "READY";
111 j.Enqueued = time;
112 queues[j.Level].Add(j);
113 }
114 void Release() {
115 cpu = "IDLE";
116 current = null;
117 }
118 void Start(Job j) {
119 cpu = "RUN";
120 current = j;
121 runLength = 0;
122 j.State = "RUN";
123 if (j.Start < 0) j.Start = time;
124 if (j.Records[^1].Start < 0) j.Records[^1].Start = time;
125 }
126 bool Preempt(Job j) => algo switch
127 {
128 "SRTF" => queues[0].Any(x => x.Left < j.Left),
129 "PRIORITY" => queues[0].Any(x => x.Effective < j.Effective),
130 "EDF" => queues[0].Any(x => x.Process.Deadline != null && (j.Process.Deadline == null || x.Process.Absolute < j.Process.Absolute)),
131 "MLFQ" => queues.Take(j.Level).Any(q => q.Count > 0), _ => false
132 };
133 Job Choose()
134 {
135 List<Job> q = queues[0];
136 Job best = q[0];
137 if (algo == "LOTTERY") {
138 random = random * 48271 % 2147483647;
139 long r = random % q.Sum(j => 10 - j.Process.Priority), total = 0;
140 foreach (Job j in q) {
141 total += 10 - j.Process.Priority;
142 if (total > r) return j;
143 }
144 }
145 foreach (Job j in q.Skip(1))
146 {
147 bool better = algo switch {
148 "SJF" or "SRTF" => j.Left < best.Left,
149 "PRIORITY" => j.Effective < best.Effective,
150 "HRRN" => (long)(time - j.Enqueued + j.Left) * best.Left > (long)(time - best.Enqueued + best.Left) * j.Left,
151 "EDF" => j.Process.Deadline != null && (best.Process.Deadline == null || j.Process.Absolute < best.Process.Absolute), _ => false
152 };
153 if (better) best = j;
154 }
155 return best;
156 }
157 void Capture()
158 {
159 Snapshot = new() {
160 cpu == "IDLE" ? "CPU IDLE" : $"CPU {cpu} {current!.Id}"
161 };
162 string Queue(List<Job> q) => q.Count == 0 ? "none" : string.Join(" ", q.Select(j => j.Id));
163 if (algo == "MLFQ") for (int i = 0;
164 i < 3;
165 i++) Snapshot.Add($"QUEUE{i} {Queue(queues[i])}");
166 else Snapshot.Add("QUEUE " + Queue(queues[0]));
167 foreach (Job j in Jobs)
168 {
169 string line = j.State == "FUTURE" ? $"{j.Id} FUTURE at={j.Process.Arrival}" : j.State == "DONE" ? $"{j.Id} DONE at={j.Finish}" : $"{j.Id} {j.State} burst={j.Burst + 1} left={j.Left}";
170 if (j.State is not ("FUTURE" or "DONE")) {
171 if (algo == "PRIORITY") line += $" prio={j.Effective}";
172 if (algo == "MLFQ") line += $" level={j.Level} used={j.Usage}";
173 }
174 Snapshot.Add(line);
175 }
176 }
177 void Run()
178 {
179 for (time = 0;
180 ;
181 time++)
182 {
183 if (algo == "PRIORITY") foreach (Job j in queues[0]) {
184 j.Wait++;
185 if (j.Wait == p["k"]) {
186 j.Wait = 0;
187 if (j.Effective > 0) {
188 j.Effective--;
189 Record("AGE", $"{j.Id} {j.Effective}");
190 }
191 }
192 }
193 foreach (Job j in Jobs.Where(j => j.State == "IO" && j.Left == 0)) {
194 j.Advance();
195 j.Records.Add(new(j.Left, time));
196 Enqueue(j);
197 Record("IODONE", j.Id);
198 }
199 foreach (Job j in Jobs.Where(j => j.State == "FUTURE" && j.Process.Arrival == time)) {
200 j.Records.Add(new(j.Left, time));
201 Enqueue(j);
202 Record("ARRIVE", j.Id);
203 }
204 if (cpu == "SWITCH" && switchLeft == 0) {
205 Start(current!);
206 Record("START", current!.Id);
207 } else if (cpu == "RUN")
208 {
209 Job j = current!;
210 if (j.Left == 0 && j.Burst == j.Process.Bursts.Length - 1) {
211 j.State = "DONE";
212 j.Finish = time;
213 Record("DONE", j.Id);
214 Release();
215 } else
216 {
217 bool demoted = algo == "MLFQ" && j.Level < 2 && j.Usage == (j.Level == 0 ? 4 : 8);
218 if (demoted) {
219 j.Level++;
220 j.Usage = 0;
221 Record("DEMOTE", $"{j.Id} {j.Level}");
222 }
223 if (j.Left == 0) {
224 j.Advance();
225 j.State = "IO";
226 Record("IO", $"{j.Id} {j.Left}");
227 Release();
228 } else if (demoted) {
229 Enqueue(j);
230 Release();
231 } else if ((algo is "RR" or "LOTTERY") && runLength == p["q"]) {
232 Enqueue(j);
233 Record("EXPIRE", j.Id);
234 Release();
235 } else if (Preempt(j)) {
236 Enqueue(j);
237 Record("PREEMPT", j.Id);
238 Release();
239 }
240 }
241 }
242 if (Jobs.All(j => j.State == "DONE")) break;
243 if (algo == "MLFQ" && p["s"] > 0 && time > 0 && time % p["s"] == 0) {
244 queues = new[] {
245 queues.SelectMany(q => q).ToList(), new List<Job>(), new List<Job>()
246 };
247 foreach (Job j in Jobs) {
248 j.Level = 0;
249 j.Usage = 0;
250 }
251 Record("BOOST");
252 }
253 if (cpu == "IDLE" && queues.Any(q => q.Count > 0))
254 {
255 List<Job> q = algo == "MLFQ" ? queues.First(q => q.Count > 0) : queues[0];
256 Job j = algo == "MLFQ" ? q[0] : Choose();
257 q.Remove(j);
258 j.Effective = j.Process.Priority;
259 j.Wait = 0;
260 j.Records[^1].Slices++;
261 bool switched = j != previous;
262 if (switched) Switches++;
263 if (switched && p["cs"] > 0) {
264 cpu = "SWITCH";
265 current = j;
266 switchLeft = (int)p["cs"];
267 j.State = "SWITCH";
268 Record("SWITCH", j.Id);
269 } else {
270 Start(j);
271 Record("DISPATCH", j.Id);
272 }
273 }
274 if (time == requested) Capture();
275 foreach (Job j in Jobs) j.Timeline.Add(j.State switch {
276 "READY" => '.',
277 "SWITCH" => 's',
278 "RUN" => '#',
279 "IO" => 'i',
280 _ => '-'
281 });
282 QueueLengths.Add(queues.Sum(q => q.Count));
283 CpuStates.Add(cpu);
284 if (cpu == "RUN") {
285 Job j = current!;
286 j.Left--;
287 j.Usage++;
288 runLength++;
289 previous = j;
290 Labels.Add(j.Id);
291 if (j.Left == 0) j.Records[^1].End = time + 1;
292 } else if (cpu == "SWITCH") {
293 switchLeft--;
294 Labels.Add("CS");
295 } else Labels.Add("IDLE");
296 foreach (Job j in Jobs.Where(j => j.State == "IO")) j.Left--;
297 }
298 End = time;
299 if (requested != null && Snapshot.Count == 0) Capture();
300 }
301 public long[] Totals => Enumerable.Range(0, 3).Select(i => Jobs.Sum(j => (long)j.Metrics[i])).ToArray();
302 public string[] Averages => Totals.Select(n => Rules.Decimal(n, Jobs.Count)).ToArray();
303 public string Util => Rules.Decimal(CpuStates.Count(s => s == "RUN") * 100L, End);
304 public int Missed => Jobs.Count(j => j.Missed);
305 public (int Busy, int Switch, int Idle) TickCounts(int a, int b) {
306 var s = CpuStates.Skip(a).Take(b - a).ToList();
307 return (s.Count(x => x == "RUN"), s.Count(x => x == "SWITCH"), s.Count(x => x == "IDLE"));
308 }
309}

314 Process RequireProcess(string id) {
315 Rules.Require(registry.ContainsKey(id), $"unknown process {id}");

318 static void Algorithm(string a) => Rules.Require(Rules.Algorithms.Contains(a), $"unknown algorithm {a}");
319 Dictionary<string, long> Prepare(IEnumerable<string> words, string? a = null, string? preset = null)
320 {
321 var p = new Dictionary<string, long> {
322 ["cs"] = 0, ["q"] = 4, ["k"] = 5, ["s"] = 0, ["seed"] = 1
323 };
324 var seen = new HashSet<string>();
325 if (preset != null) seen.Add(preset);
326 foreach (string w in words) {
327 string[] kv = w.Split('=');
328 string k = kv[0];
329 long v = long.Parse(kv[1]);
330 Rules.Require(Rules.Keys.Contains(k), $"unknown key {k}");
331 Rules.Require(a == null || Rules.Allowed(a, k), $"key {k} not allowed for {a}");
332 Rules.Require(!seen.Contains(k), $"duplicate key {k}");
333 Rules.Require(Rules.Valid(k, v), $"invalid {k}");
334 seen.Add(k);
335 p[k] = v;
336 }
337 Rules.Require(registry.Count > 0, "no processes");
338 return p;
339 }
340 Simulation Sim(string a, Dictionary<string, long> p, int? t = null) => new(registry.Values, a, p, t);
341 void Command(string[] w)
342 {
343 string c = w[0];
344 int Num(int i) => int.Parse(w[i]);
345 if (c == "PROC")
346 {
347 string id = w[1];
348 Rules.Require(!registry.ContainsKey(id), $"duplicate process {id}");
349 Rules.Require(registry.Count < 10, "too many processes");
350 Rules.Require(Num(2) <= 200, $"invalid arrival {id}");
351 Rules.Require(Num(3) <= 9, $"invalid priority {id}");
352 var bs = w.Skip(4).ToArray();
353 int[] bursts = bs.Select(s => int.Parse(s[1..])).ToArray();
354 Rules.Require(bs.Length % 2 == 1 && bs.Select((s, i) => s[0] == (i % 2 == 0 ? 'C' : 'I') && bursts[i] >= 1 && bursts[i] <= 30).All(x => x), $"invalid bursts {id}");
355 registry[id] = new(id, Num(2), Num(3), bursts);
356 Output.Add($"PROC {id} cpu={bursts.Where((_, i) => i % 2 == 0).Sum()} io={bursts.Where((_, i) => i % 2 == 1).Sum()}");
357 return;
358 }
359 if (c == "GEN")
360 {
361 var ids = Enumerable.Range(1, Num(2)).Select(i => w[1] + i).ToArray();
362 foreach (string id in ids) Rules.Require(!registry.ContainsKey(id), $"duplicate process {id}");
363 Rules.Require(registry.Count + ids.Length <= 10, "too many processes");
364 Output.Add($"GEN count={ids.Length}");
365 long x = Num(3);
366 int Draw(int m) {
367 x = x * 48271 % 2147483647;
368 return (int)(x % m);
369 }
370 foreach (string id in ids) {
371 int arrival = Draw(21), priority = Draw(10), count = 1 + Draw(3);
372 var bursts = new List<int>();
373 for (int j = 0;
374 j < count;
375 j++) {
376 if (j > 0) bursts.Add(1 + Draw(8));
377 bursts.Add(1 + Draw(9));
378 }
379 var proc = new Process(id, arrival, priority, bursts.ToArray());
380 registry[id] = proc;
381 Output.Add(proc.Describe);
382 }
383 return;
384 }
385 if (c is "SETPRIO" or "DEADLINE" or "DROP")
386 {
387 Process proc = RequireProcess(w[1]);
388 if (c == "DROP") {
389 registry.Remove(w[1]);
390 Output.Add($"DROP {w[1]}");
391 } else if (c == "SETPRIO") {
392 Rules.Require(Num(2) <= 9, $"invalid priority {w[1]}");
393 proc.Priority = Num(2);
394 Output.Add($"SETPRIO {w[1]} priority={w[2]}");
395 } else {
396 Rules.Require(Num(2) <= 1000, $"invalid deadline {w[1]}");
397 proc.Deadline = Num(2) == 0 ? null : Num(2);
398 Output.Add($"DEADLINE {w[1]} {proc.Deadline?.ToString() ?? "none"}");
399 }
400 return;
401 }
402 if (c == "CLEAR") {
403 Output.Add($"CLEAR count={registry.Count}");
404 registry.Clear();
405 return;
406 }
407 if (c == "LIST") {
408 Output.Add($"LIST count={registry.Count}");
409 Output.AddRange(registry.Values.OrderBy(x => x.Id, StringComparer.Ordinal).Select(x => x.Describe));
410 return;
411 }
412 if (c is "COMPARE" or "RANK" or "DIFF")
413 {
414 int start = c == "COMPARE" ? 1 : c == "RANK" ? 2 : 3;
415 if (c == "DIFF") {
416 Algorithm(w[1]);
417 Algorithm(w[2]);
418 }
419 if (c == "RANK") Rules.Require(Rules.Metrics.Concat(new[] {
420 "end", "switches", "missed"
421 }
422 ).Contains(w[1]), $"unknown metric {w[1]}");
423 var p = Prepare(w.Skip(start));
424 if (c == "DIFF")
425 {
426 var a = Sim(w[1], p);
427 var b = Sim(w[2], p);
428 Output.Add($"DIFF {w[1]} {w[2]} {Rules.Notation(p)}");
429 int wins = 0, losses = 0;
430 for (int i = 0;
431 i < a.Jobs.Count;
432 i++) {
433 int[] x = a.Jobs[i].Metrics, y = b.Jobs[i].Metrics;
434 Output.Add(a.Jobs[i].Id + " " + string.Join(" ", Rules.Metrics.Select((m, j) => $"{m}={x[j]}/{y[j]}")));
435 if (x[0] < y[0]) wins++;
436 if (x[0] > y[0]) losses++;
437 }
438 Output.Add($"WINS {w[1]}={wins} {w[2]}={losses} ties={a.Jobs.Count - wins - losses}");
439 return;
440 }
441 var sims = Rules.Algorithms.Select(a => Sim(a, p)).ToArray();
442 Output.Add(c == "COMPARE" ? $"COMPARE {Rules.Notation(p)}" : $"RANK {w[1]} {Rules.Notation(p)}");
443 if (c == "COMPARE")
444 {
445 Output.Add(Rules.Row(new[] {
446 "ALGO", "TURN", "WAIT", "RESP", "UTIL", "END", "SWITCH", "MISS"
447 }
448 ));
449 for (int i = 0;
450 i < sims.Length;
451 i++) {
452 var s = sims[i];
453 Output.Add(Rules.Row(new[] {
454 Rules.Algorithms[i]
455 }
456 .Concat(s.Averages).Concat(new[] {
457 s.Util, s.End.ToString(), s.Switches.ToString(), s.Missed.ToString()
458 }
459 )));
460 }
461 Output.Add("BEST " + string.Join(" ", Rules.Metrics.Select((m, j) => $"{m}={Rules.Algorithms[Enumerable.Range(0, 9).OrderBy(i => sims[i].Totals[j]).First()]}")));
462 } else
463 {
464 int metric = Array.IndexOf(Rules.Metrics, w[1]);
465 long Value(int i) => metric >= 0 ? sims[i].Totals[metric] : w[1] == "end" ? sims[i].End : w[1] == "switches" ? sims[i].Switches : sims[i].Missed;
466 foreach (int i in Enumerable.Range(0, 9).OrderBy(Value)) Output.Add($"{1 + Enumerable.Range(0, 9).Count(j => Value(j) < Value(i))} {Rules.Algorithms[i]} {(metric >= 0 ? Rules.Decimal(Value(i), registry.Count) : Value(i).ToString())}");
467 }
468 return;
469 }
470 string algo = w[1];
471 Algorithm(algo);
472 int offset = c is "SNAPSHOT" or "WINDOW" ? 3 : c == "SWEEP" ? 5 : 2;
473 if (c == "WINDOW") Rules.Require(Num(2) >= 1 && Num(2) <= 1000, "invalid window");
474 if (c == "SWEEP") {
475 string k = w[2];
476 Rules.Require(Rules.Keys.Contains(k), $"unknown key {k}");
477 Rules.Require(Rules.Allowed(algo, k), $"key {k} not allowed for {algo}");
478 Rules.Require(Num(3) <= Num(4) && Rules.Valid(k, Num(3)) && Rules.Valid(k, Num(4)) && (long)Num(4) - Num(3) <= 20, "invalid range");
479 }
480 var parameters = Prepare(w.Skip(offset), algo, c == "SWEEP" ? w[2] : null);
481 if (c == "SWEEP")
482 {
483 Output.Add($"SWEEP {algo} {w[2]} {w[3]} {w[4]}");

485 int value = Num(3);
486 for (int v = Num(3);
487 v <= Num(4);
488 v++) {
489 parameters[w[2]] = v;
490 var s = Sim(algo, parameters);

494 best = s.Totals[1];

496 }
497 }
498 Output.Add($"BEST {w[2]}={value}");
499 return;
500 }

503 if (c == "TIMELINE") Rules.Require(sim.End <= 100, $"too long {sim.End}");
504 Output.Add($"{c} {algo} {notation}" + (c == "TIMELINE" ? $" end={sim.End}" : c == "SNAPSHOT" ? $" t={w[2]}" : c == "WINDOW" ? $" width={w[2]}" : ""));
505 switch (c)
506 {
507 case "TRACE": Output.AddRange(sim.Events);
508 Output.Add($"END {sim.End}");
509 break;
510 case "SNAPSHOT": Output.AddRange(sim.Snapshot);
511 break;
512 case "TIMELINE": foreach (Job j in sim.Jobs) Output.Add(j.Id.PadRight(8) + " " + new string(j.Timeline.ToArray()));
513 break;
514 case "BURSTS": foreach (Job j in sim.Jobs) for (int i = 0;
515 i < j.Records.Count;
516 i++) {
517 var r = j.Records[i];
518 Output.Add($"{j.Id} #{i + 1} len={r.Length} ready={r.Ready} start={r.Start} end={r.End} wait={r.End - r.Ready - r.Length} slices={r.Slices}");
519 }
520 break;
521 case "WINDOW":
522 for (int a = 0;
523 a < sim.End;
524 a += Num(2)) {
525 int b = Math.Min(sim.End, a + Num(2));
526 var t = sim.TickCounts(a, b);
527 string top = Enumerable.Range(a, b - a).Where(i => sim.CpuStates[i] == "RUN").GroupBy(i => sim.Labels[i]).OrderByDescending(g => g.Count()).ThenBy(g => g.Key, StringComparer.Ordinal).Select(g => g.Key).FirstOrDefault() ?? "none";
528 Output.Add($"{a}-{b} busy={t.Busy} switch={t.Switch} idle={t.Idle} util={Rules.Decimal(t.Busy * 100L, b - a)} top={top}");
529 }
530 break;
531 case "STATS":
532 for (int i = 0;
533 i < 3;
534 i++) {
535 int[] v = sim.Jobs.Select(j => j.Metrics[i]).ToArray(), sorted = v.OrderBy(x => x).ToArray();
536 int doubled = sorted.Length % 2 == 1 ? 2 * sorted[sorted.Length / 2] : sorted[sorted.Length / 2 - 1] + sorted[sorted.Length / 2];
537 Output.Add($"{Rules.Metrics[i]} min={v.Min()} ({sim.Jobs[Array.IndexOf(v, v.Min())].Id}) max={v.Max()} ({sim.Jobs[Array.IndexOf(v, v.Max())].Id}) median={doubled / 2}.{doubled % 2 * 5} total={v.Sum()}");
538 }
539 break;
540 case "RUN":
541 var segments = new List<string>();
542 int start = 0;
543 for (int t = 1;
544 t <= sim.End;
545 t++) if (t == sim.End || sim.Labels[t] != sim.Labels[start]) {
546 segments.Add($"[{start}-{t} {sim.Labels[start]}]");
547 start = t;
548 }
549 for (int i = 0;
550 i < segments.Count;
551 i += 8) Output.Add("GANTT " + string.Concat(segments.Skip(i).Take(8)));
552 foreach (Job j in sim.Jobs) {
553 int[] m = j.Metrics;
554 Output.Add($"{j.Id} arrival={j.Process.Arrival} done={j.Finish} turnaround={m[0]} waiting={m[1]} response={m[2]} slices={j.Records.Sum(r => r.Slices)}" + (j.Process.Deadline == null ? "" : $" deadline={j.Process.Absolute} late={Math.Max(0, j.Finish - j.Process.Absolute)}"));
555 }
556 string[] av = sim.Averages;
557 Output.Add($"AVERAGE turnaround={av[0]} waiting={av[1]} response={av[2]}");
558 var counts = sim.TickCounts(0, sim.End);
559 Output.Add($"CPU end={sim.End} busy={counts.Busy} switch={counts.Switch} idle={counts.Idle} switches={sim.Switches} util={sim.Util}");
560 Output.Add($"QUEUE max={sim.QueueLengths.Max()} avg={Rules.Decimal(sim.QueueLengths.Sum(), sim.End)}");
561 Output.Add("EVENTS " + string.Join(" ", Rules.EventKinds.Select(k => $"{k.ToLowerInvariant()}={sim.Counts.GetValueOrDefault(k)}")));
562 Output.Add($"DEADLINE met={sim.Jobs.Count(j => j.Process.Deadline != null && !j.Missed)} missed={sim.Missed}");
563 break;
564 }
565 }
