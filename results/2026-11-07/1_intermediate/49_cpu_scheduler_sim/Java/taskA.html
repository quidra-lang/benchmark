142 this.snapshotTime=snapshotTime;
143 for(Process process:processes) jobs.add(new Job(process));

155 void enqueue(Job j) {
156 j.state="READY";
157 j.enqueued=time;
158 queues.get(algo.equals("MLFQ")?j.level:0).add(j);
159 }
160 void release() {
161 cpu="IDLE";

164 void start(Job j) {
165 current=j;
166 cpu=j.state="RUN";
167 runLength=0;
168 if(j.start<0) j.start=time;
169 if(j.records.getLast().start<0) j.records.getLast().start=time;
170 }
171 boolean preempts(Job j) {
172 return switch(algo) {
173 case "SRTF" -> queues.getFirst().stream().anyMatch(o -> o.left<j.left);
174 case "PRIORITY" -> queues.getFirst().stream().anyMatch(o -> o.effective<j.effective);
175 case "EDF" -> queues.getFirst().stream().anyMatch(o -> o.p.absolute()<j.p.absolute());
176 case "MLFQ" -> queues.subList(0,j.level).stream().anyMatch(q -> !q.isEmpty());
177 default -> false;
178 };
179 }
180 Job choose(List<Job> q) {
181 Job best=q.getFirst();
182 if(algo.equals("LOTTERY")) {
183 int r=random.draw(q.stream().mapToInt(j -> 10-j.p.priority).sum());
184 for(Job j:q) {
185 r-=10-j.p.priority;
186 if(r<0) return j;
187 }
188 }
189 for(Job j:q) {
190 boolean better=switch(algo) {
191 case "SJF","SRTF" -> j.left<best.left;
192 case "PRIORITY" -> j.effective<best.effective;
193 case "EDF" -> j.p.absolute()<best.p.absolute();
194 case "HRRN" -> (long)(time-j.enqueued+j.left)*best.left>(long)(time-best.enqueued+best.left)*j.left;
195 default -> false;
196 };
197 if(better) best=j;
198 }
199 return best;
200 }
201 void run() {
202 while(true) {
203 if(algo.equals("PRIORITY")) for(Job j:queues.getFirst()) {
204 if(++j.wait==p.get("k")) {
205 j.wait=0;
206 if(j.effective>0) {
207 j.effective--;
208 event("AGE",j.id(),j.effective);
209 }
210 }
211 }
212 for(Job j:jobs) if(j.state.equals("IO")&&j.left==0) {
213 j.advance();
214 j.records.add(new Burst(j.left,time));
215 enqueue(j);
216 event("IODONE",j.id());
217 }
218 for(Job j:jobs) if(j.state.equals("FUTURE")&&j.p.arrival==time) {
219 j.records.add(new Burst(j.left,time));
220 enqueue(j);
221 event("ARRIVE",j.id());
222 }
223 if(cpu.equals("SWITCH")&&switchLeft==0) {
224 start(current);
225 event("START",current.id());
226 }
227 else if(cpu.equals("RUN")) {
228 Job j=current;
229 if(j.left==0&&j.burst==j.p.bursts.size()-1) {
230 j.state="DONE";
231 j.finish=time;
232 event("DONE",j.id());
233 release();
234 }
235 else {
236 boolean demoted=algo.equals("MLFQ")&&j.level<2&&j.used==(j.level==0?4:8);
237 if(demoted) {
238 j.level++;
239 j.used=0;
240 event("DEMOTE",j.id(),j.level);
241 }
242 if(j.left==0) {
243 j.advance();
244 j.state="IO";
245 event("IO",j.id(),j.left);
246 release();
247 }
248 else if(demoted) {
249 enqueue(j);
250 release();
251 }
252 else if((algo.equals("RR")||algo.equals("LOTTERY"))&&runLength==p.get("q")) {
253 enqueue(j);
254 event("EXPIRE",j.id());
255 release();
256 }
257 else if(preempts(j)) {
258 enqueue(j);
259 event("PREEMPT",j.id());
260 release();
261 }
262 }
263 }
264 if(jobs.stream().allMatch(j -> j.state.equals("DONE"))) break;
265 if(algo.equals("MLFQ")&&p.get("s")>0&&time>0&&time%p.get("s")==0) {
266 queues.getFirst().addAll(queues.get(1));
267 queues.getFirst().addAll(queues.get(2));
268 queues.get(1).clear();
269 queues.get(2).clear();
270 for(Job j:jobs) {
271 j.level=0;
272 j.used=0;
273 }
274 event("BOOST");
275 }
276 if(cpu.equals("IDLE")) {
277 List<Job> q=queues.stream().filter(xs -> !xs.isEmpty()).findFirst().orElse(null);
278 if(q!=null) {
279 Job j=algo.equals("MLFQ")?q.getFirst():choose(q);
280 q.remove(j);
281 j.effective=j.p.priority;
282 j.wait=0;
283 j.records.getLast().slices++;
284 boolean switched=j!=previous;
285 if(switched) switches++;
286 if(switched&&p.get("cs")>0) {
287 current=j;
288 cpu=j.state="SWITCH";
289 switchLeft=p.get("cs").intValue();
290 event("SWITCH",j.id());
291 }
292 else {
293 start(j);
294 event("DISPATCH",j.id());
295 }
296 }
297 }
298 if(time==snapshotTime) capture();
299 for(Job j:jobs) j.timeline.append(switch(j.state) {
300 case "READY" -> '.';
301 case "SWITCH" -> 's';
302 case "RUN" -> '#';
303 case "IO" -> 'i';
304 default -> '-';
305 });
306 lengths.add(queues.stream().mapToInt(List::size).sum());
307 cpuStates.add(cpu);
308 if(cpu.equals("RUN")) {
309 current.left--;
310 current.used++;
311 runLength++;
312 previous=current;
313 labels.add(current.id());
314 if(current.left==0) current.records.getLast().end=time+1;
315 }
316 else if(cpu.equals("SWITCH")) {
317 switchLeft--;
318 labels.add("CS");
319 } else labels.add("IDLE");
320 for(Job j:jobs) if(j.state.equals("IO")) j.left--;
321 time++;
322 }
323 if(snapshot==null) capture();
324 }
325 void capture() {
326 snapshot=new ArrayList<>();
327 snapshot.add("CPU "+cpu+(current==null?"":" "+current.id()));
328 for(int i=0;i<(algo.equals("MLFQ")?3:1);i++) {
329 List<Job> q=queues.get(i);
330 snapshot.add("QUEUE"+(algo.equals("MLFQ")?i:"")+" "+(q.isEmpty()?"none":String.join(" ",q.stream().map(Job::id).toList())));
331 }
332 for(Job j:jobs) {
333 String s=j.id()+" "+j.state;
334 if(j.state.equals("FUTURE")) s+=" at="+j.p.arrival;
335 else if(j.state.equals("DONE")) s+=" at="+j.finish;
336 else {
337 s+=" burst="+(j.burst+1)+" left="+j.left;
338 if(algo.equals("PRIORITY")) s+=" prio="+j.effective;
339 if(algo.equals("MLFQ")) s+=" level="+j.level+" used="+j.used;
340 }
341 snapshot.add(s);
342 }

397 void execute(String line) {
398 List<String> a=Arrays.asList(line.split(" "));
399 String c=a.getFirst();
400 try {
401 switch(c) {
402 case "PROC" -> {
403 String id=a.get(1);
404 check(registry.containsKey(id),"duplicate process "+id);
405 check(registry.size()==10,"too many processes");
406 int arrival=Integer.parseInt(a.get(2)),priority=Integer.parseInt(a.get(3));
407 check(arrival>200,"invalid arrival "+id);
408 check(priority>9,"invalid priority "+id);
409 List<Integer> bursts=new ArrayList<>();
410 check((a.size()-4)%2==0,"invalid bursts "+id);
411 int cpu=0,io=0;
412 for(int i=4;i<a.size();i++) {
413 String b=a.get(i);
414 int n=Integer.parseInt(b.substring(1));
415 check(b.charAt(0)!=((i-4)%2==0?'C':'I')||n<1||n>30,"invalid bursts "+id);
416 bursts.add(n);
417 if(i%2==0) cpu+=n;
418 else io+=n;
419 }
420 registry.put(id,new Process(id,arrival,priority,bursts));
421 emit("PROC "+id+" cpu="+cpu+" io="+io);
422 }
423 case "GEN" -> {
424 int n=Integer.parseInt(a.get(2));
425 for(int i=1;i<=n;i++) check(registry.containsKey(a.get(1)+i),"duplicate process "+a.get(1)+i);
426 check(registry.size()+n>10,"too many processes");
427 emit("GEN count="+n);
428 RandomState random=new RandomState(Long.parseLong(a.get(3)));
429 for(int i=1;i<=n;i++) {
430 int arrival=random.draw(21),priority=random.draw(10),m=1+random.draw(3);
431 List<Integer> bursts=new ArrayList<>();
432 for(int j=0;j<m;j++) {
433 if(j>0) bursts.add(1+random.draw(8));
434 bursts.add(1+random.draw(9));
435 }
436 Process p=new Process(a.get(1)+i,arrival,priority,bursts);
437 registry.put(p.id,p);
438 emit(p.describe());
439 }
440 }
441 case "SETPRIO" -> {
442 Process p=process(a.get(1));
443 int v=Integer.parseInt(a.get(2));
444 check(v>9,"invalid priority "+p.id);
445 p.priority=v;
446 emit("SETPRIO "+p.id+" priority="+v);
447 }
448 case "DEADLINE" -> {
449 Process p=process(a.get(1));
450 int v=Integer.parseInt(a.get(2));
451 check(v>1000,"invalid deadline "+p.id);
452 p.deadline=v;
453 emit("DEADLINE "+p.id+" "+(v==0?"none":v));
454 }
455 case "DROP" -> {
456 Process p=process(a.get(1));
457 registry.remove(p.id);
458 emit("DROP "+p.id);
459 }
460 case "CLEAR" -> {
461 int n=registry.size();
462 registry.clear();
463 emit("CLEAR count="+n);
464 }
465 case "LIST" -> {
466 emit("LIST count="+registry.size());
467 for(Process p:registry.values()) emit(p.describe());
468 }
469 default -> simulationCommand(c,a);
470 }
471 } catch(Failure e) {
472 emit("ERROR "+e.getMessage());
473 }
474 }
475 void simulationCommand(String c,List<String> a) {
476 if(c.equals("COMPARE")||c.equals("RANK")) {
477 String metric=c.equals("RANK")?a.get(1):null;
478 if(metric!=null) check(!List.of("turnaround","waiting","response","end","switches","missed").contains(metric),"unknown metric "+metric);
479 Map<String,Long> p=prepare(a.subList(metric==null?1:2,a.size()),null,null);
480 List<Simulation> sims=new ArrayList<>();
481 for(String algo:ALGOS) sims.add(sim(algo,p));
482 if(metric==null) {
483 emit("COMPARE "+notation(p,null));
484 emit(table(List.of("ALGO","TURN","WAIT","RESP","UTIL","END","SWITCH","MISS")));
485 for(Simulation s:sims) {
486 long[] t=s.totals();
487 emit(table(List.of(s.algo,ratio(t[0],s.jobs.size()),ratio(t[1],s.jobs.size()),ratio(t[2],s.jobs.size()),s.util(),""+s.time,""+s.switches,""+s.missed())));
488 }
489 String best="BEST";
490 for(int i=0;i<3;i++) {
491 int winner=0;
492 for(int j=1;j<9;j++) if(sims.get(j).totals()[i]<sims.get(winner).totals()[i]) winner=j;
493 best+=" "+METRICS.get(i)+"="+ALGOS.get(winner);
494 }
495 emit(best);
496 } else {
497 emit("RANK "+metric+" "+notation(p,null));
498 int index=METRICS.indexOf(metric);
499 List<Long> values=new ArrayList<>();
500 for(Simulation s:sims) values.add(index>=0?s.totals()[index]:metric.equals("end")?(long)s.time:metric.equals("switches")?(long)s.switches:(long)s.missed());
501 List<Integer> indices=new ArrayList<>();
502 for(int i=0;i<9;i++) indices.add(i);
503 indices.sort(Comparator.comparingLong(values::get));
504 for(int i:indices) {
505 long value=values.get(i);
506 long rank=1+values.stream().filter(v -> v<value).count();
507 emit(rank+" "+ALGOS.get(i)+" "+(index>=0?ratio(value,registry.size()):value));
508 }
509 }
510 return;
511 }
512 String algo=a.get(1);
513 algorithm(algo);
514 if(c.equals("DIFF")) {
515 String second=a.get(2);
516 algorithm(second);
517 Map<String,Long> p=prepare(a.subList(3,a.size()),null,null);
518 Simulation one=sim(algo,p),two=sim(second,p);
519 emit("DIFF "+algo+" "+second+" "+notation(p,null));
520 int win=0,loss=0,ties=0;
521 for(int i=0;i<one.jobs.size();i++) {
522 Job j=one.jobs.get(i);
523 long[] x=j.metrics(),y=two.jobs.get(i).metrics();
524 emit(j.id()+" turnaround="+x[0]+"/"+y[0]+" waiting="+x[1]+"/"+y[1]+" response="+x[2]+"/"+y[2]);
525 if(x[0]<y[0]) win++;
526 else if(x[0]>y[0]) loss++;
527 else ties++;
528 }
529 emit("WINS "+algo+"="+win+" "+second+"="+loss+" ties="+ties);
530 return;
531 }
532 if(c.equals("SWEEP")) {
533 String key=a.get(2);
534 check(!KEYS.contains(key),"unknown key "+key);
535 check(!allowed(algo).contains(key),"key "+key+" not allowed for "+algo);
536 long lo=Long.parseLong(a.get(3)),hi=Long.parseLong(a.get(4));
537 check(lo>hi||!valid(key,lo)||!valid(key,hi)||hi-lo>20,"invalid range");
538 Map<String,Long> p=prepare(a.subList(5,a.size()),algo,key);
539 emit("SWEEP "+algo+" "+key+" "+lo+" "+hi);
540 long best=lo,total=Long.MAX_VALUE;
541 for(long v=lo;v<=hi;v++) {
542 p.put(key,v);
543 Simulation s=sim(algo,p);
544 emit(notation(p,algo)+" "+s.averages()+" util="+s.util()+" end="+s.time+" switches="+s.switches);
545 if(s.totals()[1]<total) {
546 total=s.totals()[1];
547 best=v;
548 }
549 }
550 emit("BEST "+key+"="+best);
551 return;
552 }
553 int extra=c.equals("WINDOW")||c.equals("SNAPSHOT")?Integer.parseInt(a.get(2)):0;
554 if(c.equals("WINDOW")) check(extra<1||extra>1000,"invalid window");
555 Map<String,Long> p=prepare(a.subList(c.equals("WINDOW")||c.equals("SNAPSHOT")?3:2,a.size()),algo,null);
556 Simulation s=new Simulation(registry.values(),algo,p,c.equals("SNAPSHOT")?extra:-1);
557 String header=c+" "+algo+" "+notation(p,algo);
558 switch(c) {
559 case "TRACE" -> {
560 emit(header);
561 for(String event:s.events) emit(event);
562 emit("END "+s.time);
563 }
564 case "TIMELINE" -> {
565 check(s.time>100,"too long "+s.time);
566 emit(header+" end="+s.time);
567 for(Job j:s.jobs) emit(String.format("%-8s",j.id())+" "+j.timeline);
568 }
569 case "SNAPSHOT" -> {
570 emit(header+" t="+extra);
571 for(String row:s.snapshot) emit(row);
572 }
573 case "BURSTS" -> {
574 emit(header);
575 for(Job j:s.jobs) for(int i=0;i<j.records.size();i++) {
576 Burst b=j.records.get(i);
577 emit(j.id()+" #"+(i+1)+" len="+b.length+" ready="+b.ready+" start="+b.start+" end="+b.end+" wait="+(b.end-b.ready-b.length)+" slices="+b.slices);
578 }
579 }
580 case "WINDOW" -> {
581 emit(header+" width="+extra);
582 for(int start=0;start<s.time;start+=extra) {
583 int end=Math.min(start+extra,s.time);
584 int[] t=s.ticks(start,end);
585 Map<String,Integer> count=new TreeMap<>();
586 for(int i=start;i<end;i++) if(s.cpuStates.get(i).equals("RUN")) count.merge(s.labels.get(i),1,Integer::sum);
587 String top="none";
588 int best=0;
589 for(var entry:count.entrySet()) if(entry.getValue()>best) {
590 top=entry.getKey();
591 best=entry.getValue();
592 }
593 emit(start+"-"+end+" busy="+t[0]+" switch="+t[1]+" idle="+t[2]+" util="+ratio(t[0]*100L,end-start)+" top="+top);
594 }
595 }
596 case "STATS" -> {
597 emit(header);
598 for(int i=0;i<3;i++) {
599 List<Long> values=new ArrayList<>();
600 for(Job j:s.jobs) values.add(j.metrics()[i]);
601 long low=Collections.min(values),high=Collections.max(values);
602 List<Long> sorted=new ArrayList<>(values);
603 Collections.sort(sorted);
604 int n=sorted.size();
605 long doubled=sorted.get(n/2)+sorted.get((n-1)/2);
606 emit(METRICS.get(i)+" min="+low+" ("+s.jobs.get(values.indexOf(low)).id()+") max="+high+" ("+s.jobs.get(values.indexOf(high)).id()+") median="+doubled/2+"."+(doubled%2==0?0:5)+" total="+s.totals()[i]);
607 }
608 }
609 case "RUN" -> {
610 emit(header);
611 List<String> segments=new ArrayList<>();
612 int start=0;
613 for(int t=1;t<=s.time;t++) if(t==s.time||!s.labels.get(t).equals(s.labels.get(start))) {
614 segments.add("["+start+"-"+t+" "+s.labels.get(start)+"]");
615 start=t;
616 }
617 for(int i=0;i<segments.size();i+=8) emit("GANTT "+String.join("",segments.subList(i,Math.min(i+8,segments.size()))));
618 for(Job j:s.jobs) {
619 long[] m=j.metrics();
620 emit(j.id()+" arrival="+j.p.arrival+" done="+j.finish+" turnaround="+m[0]+" waiting="+m[1]+" response="+m[2]+" slices="+j.slices()+(j.p.deadline==0?"":" deadline="+j.p.absolute()+" late="+Math.max(0,j.finish-j.p.absolute())));
621 }
622 emit("AVERAGE "+s.averages());
623 int[] t=s.ticks(0,s.time);
624 emit("CPU end="+s.time+" busy="+t[0]+" switch="+t[1]+" idle="+t[2]+" switches="+s.switches+" util="+s.util());
625 emit("QUEUE max="+Collections.max(s.lengths)+" avg="+ratio(s.lengths.stream().mapToLong(Integer::longValue).sum(),s.time));
626 String events="EVENTS";
627 for(String e:EVENTS) events+=" "+e.toLowerCase(Locale.ROOT)+"="+s.counts.getOrDefault(e,0);
628 emit(events);
629 emit("DEADLINE met="+s.met()+" missed="+s.missed());
630 }
631 }
632 }
633 }
