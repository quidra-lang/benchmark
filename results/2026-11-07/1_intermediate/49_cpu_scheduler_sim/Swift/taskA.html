4func allowed(_ a: String, _ k: String) -> Bool {
5 k == "cs" || k == "q" && ["RR", "LOTTERY"].contains(a) || k == "k" && a == "PRIORITY" || k == "s" && a == "MLFQ" || k == "seed" && a == "LOTTERY"
6}
7func valid(_ k: String, _ v: Int) -> Bool {
8 switch k {
9 case "cs": return v <= 5
10 case "q": return (1...20).contains(v)
11 case "k": return (1...50).contains(v)
12 case "s": return v <= 500
13 case "seed": return (1...2147483646).contains(v)
14 default: return false
15 }
16}

68final class Job {
69 let process: Process
70 var state = "FUTURE", burst = 0, left: Int, effective: Int, wait = 0, level = 0, usage = 0, enqueued = 0, start = -1, finish = 0
71 var records: [BurstRecord] = [], timeline = ""
72 init(_ p: Process) {
73 process = p
74 left = p.bursts[0]
75 effective = p.priority
76 }
77 var id: String {
78 process.id
79 }
80 var metrics: [Int] {
81 [finish - process.arrival, finish - process.arrival - process.bursts.reduce(0, +), start - process.arrival]

84 process.deadline != nil && finish > process.absolute
85 }

87 burst += 1

91final class Simulation {
92 let jobs: [Job], algo: String, p: [String: Int], requested: Int?
93 var queues: [[Job]] = [[], [], []], cpu = "IDLE", current: Job?, previous: Job?, switchLeft = 0, runLength = 0, time = 0, random: Int, end = 0, switches = 0
94 var events: [String] = [], labels: [String] = [], cpuStates: [String] = [], snapshot: [String] = [], queueLengths: [Int] = [], counts: [String: Int] = [:]
95 init(_ processes: [Process], _ algo: String, _ p: [String: Int], _ snapshot: Int? = nil) {
96 jobs = processes.sorted {
97 $0.id < $1.id
98 }
99 .map(Job.init)
100 self.algo = algo
101 self.p = p
102 requested = snapshot
103 random = p["seed"]!
104 run()
105 }
106 func record(_ kind: String, _ details: String = "") {
107 events.append("\(time) \(kind)" + (details.isEmpty ? "" : " " + details))
108 counts[kind, default: 0] += 1
109 }
110 func enqueue(_ j: Job) {
111 j.state = "READY"
112 j.enqueued = time
113 queues[j.level].append(j)
114 }
115 func release() {
116 cpu = "IDLE"
117 current = nil
118 }
119 func start(_ j: Job) {
120 cpu = "RUN"
121 current = j
122 runLength = 0
123 j.state = "RUN"
124 if j.start < 0 {
125 j.start = time
126 }
127 if j.records.last!.start < 0 {
128 j.records.last!.start = time
129 }
130 }
131 func preempt(_ j: Job) -> Bool {
132 switch algo {
133 case "SRTF": return queues[0].contains {
134 $0.left < j.left
135 }
136 case "PRIORITY": return queues[0].contains {
137 $0.effective < j.effective
138 }
139 case "EDF": return queues[0].contains {
140 $0.process.deadline != nil && (j.process.deadline == nil || $0.process.absolute < j.process.absolute)
141 }
142 case "MLFQ": return queues.prefix(j.level).contains {
143 !$0.isEmpty
144 }
145 default: return false
146 }
147 }
148 func choose() -> Job {
149 let q = queues[0]
150 var best = q[0]
151 if algo == "LOTTERY" {
152 random = random * 48271 % 2147483647
153 let r = random % q.reduce(0) {
154 $0 + 10 - $1.process.priority
155 }
156 var total = 0
157 for j in q {
158 total += 10 - j.process.priority
159 if total > r {
160 return j
161 }
162 }
163 }
164 for j in q.dropFirst() {
165 let better: Bool
166 switch algo {
167 case "SJF", "SRTF": better = j.left < best.left
168 case "PRIORITY": better = j.effective < best.effective
169 case "HRRN": better = (time - j.enqueued + j.left) * best.left > (time - best.enqueued + best.left) * j.left
170 case "EDF": better = j.process.deadline != nil && (best.process.deadline == nil || j.process.absolute < best.process.absolute)
171 default: better = false
172 }
173 if better {
174 best = j
175 }
176 }
177 return best
178 }
179 func capture() {
180 snapshot = [cpu == "IDLE" ? "CPU IDLE" : "CPU \(cpu) \(current!.id)"]
181 func text(_ q: [Job]) -> String {
182 q.isEmpty ? "none" : q.map(\.id).joined(separator: " ")
183 }
184 if algo == "MLFQ" {
185 for i in 0..<3 {
186 snapshot.append("QUEUE\(i) \(text(queues[i]))")
187 }
188 } else {
189 snapshot.append("QUEUE \(text(queues[0]))")
190 }
191 for j in jobs {
192 var line = j.state == "FUTURE" ? "\(j.id) FUTURE at=\(j.process.arrival)" : j.state == "DONE" ? "\(j.id) DONE at=\(j.finish)" : "\(j.id) \(j.state) burst=\(j.burst + 1) left=\(j.left)"
193 if j.state != "FUTURE" && j.state != "DONE" {
194 if algo == "PRIORITY" {
195 line += " prio=\(j.effective)"
196 }
197 if algo == "MLFQ" {
198 line += " level=\(j.level) used=\(j.usage)"
199 }
200 }
201 snapshot.append(line)
202 }
203 }
204 func run() {
205 while true {
206 if algo == "PRIORITY" {
207 for j in queues[0] {
208 j.wait += 1
209 if j.wait == p["k"]! {
210 j.wait = 0
211 if j.effective > 0 {
212 j.effective -= 1
213 record("AGE", "\(j.id) \(j.effective)")
214 }
215 }
216 }
217 }
218 for j in jobs where j.state == "IO" && j.left == 0 {
219 j.advance()
220 j.records.append(BurstRecord(j.left, time))
221 enqueue(j)
222 record("IODONE", j.id)
223 }
224 for j in jobs where j.state == "FUTURE" && j.process.arrival == time {
225 j.records.append(BurstRecord(j.left, time))
226 enqueue(j)
227 record("ARRIVE", j.id)
228 }
229 if cpu == "SWITCH" && switchLeft == 0 {
230 start(current!)
231 record("START", current!.id)
232 } else if cpu == "RUN" {
233 let j = current!
234 if j.left == 0 && j.burst == j.process.bursts.count - 1 {
235 j.state = "DONE"
236 j.finish = time
237 record("DONE", j.id)
238 release()
239 } else {
240 let demoted = algo == "MLFQ" && j.level < 2 && j.usage == (j.level == 0 ? 4 : 8)
241 if demoted {
242 j.level += 1
243 j.usage = 0
244 record("DEMOTE", "\(j.id) \(j.level)")
245 }
246 if j.left == 0 {
247 j.advance()
248 j.state = "IO"
249 record("IO", "\(j.id) \(j.left)")
250 release()
251 } else if demoted {
252 enqueue(j)
253 release()
254 } else if ["RR", "LOTTERY"].contains(algo) && runLength == p["q"]! {
255 enqueue(j)
256 record("EXPIRE", j.id)
257 release()
258 } else if preempt(j) {
259 enqueue(j)
260 record("PREEMPT", j.id)
261 release()
262 }
263 }
264 }
265 if jobs.allSatisfy({
266 $0.state == "DONE"
267 }
268 ) {
269 break
270 }
271 if algo == "MLFQ" && p["s"]! > 0 && time > 0 && time % p["s"]! == 0 {
272 queues = [queues.flatMap {
273 $0
274 }, [], []]
275 for j in jobs {
276 j.level = 0
277 j.usage = 0
278 }
279 record("BOOST")
280 }
281 if cpu == "IDLE" && queues.contains(where: {
282 !$0.isEmpty
283 }
284 ) {
285 let level = algo == "MLFQ" ? queues.firstIndex {
286 !$0.isEmpty
287 }! : 0, j = algo == "MLFQ" ? queues[level][0] : choose()
288 queues[level].removeAll {
289 $0 === j
290 }
291 j.effective = j.process.priority
292 j.wait = 0
293 j.records.last!.slices += 1
294 let switched = j !== previous
295 if switched {
296 switches += 1
297 }
298 if switched && p["cs"]! > 0 {
299 cpu = "SWITCH"
300 current = j
301 switchLeft = p["cs"]!
302 j.state = "SWITCH"
303 record("SWITCH", j.id)
304 } else {
305 start(j)
306 record("DISPATCH", j.id)
307 }
308 }
309 if time == requested {
310 capture()
311 }
312 for j in jobs {
313 j.timeline += ["READY": ".", "SWITCH": "s", "RUN": "#", "IO": "i"][j.state] ?? "-"
314 }
315 queueLengths.append(queues.reduce(0) {
316 $0 + $1.count
317 })
318 cpuStates.append(cpu)
319 if cpu == "RUN" {
320 let j = current!
321 j.left -= 1
322 j.usage += 1
323 runLength += 1
324 previous = j
325 labels.append(j.id)
326 if j.left == 0 {
327 j.records.last!.end = time + 1
328 }
329 } else if cpu == "SWITCH" {
330 switchLeft -= 1
331 labels.append("CS")
332 } else {
333 labels.append("IDLE")
334 }
335 for j in jobs where j.state == "IO" {
336 j.left -= 1
337 }
338 time += 1
339 }
340 end = time
341 if requested != nil && snapshot.isEmpty {
342 capture()
343 }
344 }
345 var totals: [Int] {
346 (0..<3).map {
347 i in jobs.reduce(0) {
348 $0 + $1.metrics[i]
349 }
350 }
351 }
352 var averages: [String] {
353 totals.map {
354 decimal($0, jobs.count)
355 }
356 }
357 var util: String {
358 decimal(cpuStates.filter {
359 $0 == "RUN"
360 }
361 .count * 100, end)
362 }
363 var missed: Int {
364 jobs.filter(\.missed).count
365 }
366 func ticks(_ a: Int, _ b: Int) -> (Int, Int, Int) {
367 let s = cpuStates[a..<b]
368 return (s.filter {
369 $0 == "RUN"
370 }
371 .count, s.filter {
372 $0 == "SWITCH"
373 }
374 .count, s.filter {
375 $0 == "IDLE"
376 }
377 .count)
378 }
379}
380final class Scheduler {
381 var registry: [String: Process] = [:], output: [String] = []
382 func process(_ id: String) throws -> Process {
383 try require(registry[id] != nil, "unknown process \(id)")
384 return registry[id]!
385 }
386 func algorithm(_ a: String) throws {
387 try require(algorithms.contains(a), "unknown algorithm \(a)")

389 func prepare(_ words: ArraySlice<String>, _ algo: String? = nil, _ preset: String? = nil) throws -> [String: Int] {
390 var p = ["cs": 0, "q": 4, "k": 5, "s": 0, "seed": 1], seen: Set<String> = []
391 if let preset {
392 seen.insert(preset)
393 }
394 for word in words {
395 let kv = word.split(separator: "=").map(String.init), k = kv[0], v = Int(kv[1])!
396 try require(keys.contains(k), "unknown key \(k)")
397 try require(algo == nil || allowed(algo!, k), "key \(k) not allowed for \(algo ?? "")")
398 try require(!seen.contains(k), "duplicate key \(k)")
399 try require(valid(k, v), "invalid \(k)")
400 seen.insert(k)
401 p[k] = v
402 }
403 try require(!registry.isEmpty, "no processes")
404 return p
405 }
406 func sim(_ a: String, _ p: [String: Int], _ t: Int? = nil) -> Simulation {
407 Simulation(Array(registry.values), a, p, t)

409 func command(_ w: [String]) throws {

411 func num(_ i: Int) -> Int {

414 if c == "PROC" {

417 try require(registry.count < 10, "too many processes")

419 try require(num(3) <= 9, "invalid priority \(id)")

458 }
459 if ["SETPRIO", "DEADLINE", "DROP"].contains(c) {

462 registry.removeValue(forKey: w[1])

474 }
475 if c == "CLEAR" {
476 output.append("CLEAR count=\(registry.count)")

479 }
480 if c == "LIST" {
481 output.append("LIST count=\(registry.count)")

483 registry[$0]!.describe
484 }
485 return
486 }
487 if ["COMPARE", "RANK", "DIFF"].contains(c) {
488 let offset = c == "COMPARE" ? 1 : c == "RANK" ? 2 : 3
489 if c == "DIFF" {
490 try algorithm(w[1])
491 try algorithm(w[2])
492 }
493 if c == "RANK" {

495 }
496 let p = try prepare(w.dropFirst(offset))
497 if c == "DIFF" {

509 }

511 losses += 1
512 }
513 }
514 output.append("WINS \(w[1])=\(wins) \(w[2])=\(losses) ties=\(a.jobs.count - wins - losses)")

629 output.append("GANTT " + segments[i..<min(i + 8, segments.count)].joined())
630 }

633 output.append("\(j.id) arrival=\(j.process.arrival) done=\(j.finish) turnaround=\(m[0]) waiting=\(m[1]) response=\(m[2]) slices=\(j.records.reduce(0) { $0 + $1.slices })" + (j.process.deadline == nil ? "" : " deadline=\(j.process.absolute) late=\(max(0, j.finish - j.process.absolute))"))
