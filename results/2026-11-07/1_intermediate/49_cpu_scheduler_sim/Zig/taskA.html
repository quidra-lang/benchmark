70 const algorithm = a orelse return true;

97 fn notation(s: *Scheduler, p: [5]I, a: ?usize) !Text {
98 var buffer: std.Io.Writer.Allocating = .init(s.gpa);
99 var first = true;
100 for (keys, 0..) |key, k| if (uses(a, k)) {
101 if (!first) try buffer.writer.writeByte(' ');
102 first = false;
103 try buffer.writer.print("{s}={d}", .{ key, p[k] });

105 return buffer.toOwnedSlice();

113 fn describe(s: *Scheduler, p: Process) !void {
114 try s.out.print("{s} arrival={d} priority={d} deadline=", .{ p.id, p.arrival, p.priority });
115 if (p.deadline == 0) try s.out.writeAll("none") else try s.out.print("{d}", .{p.deadline});
116 try s.out.writeAll(" bursts=");
117 for (p.bursts[0..p.count], 0..) |length, i| try s.out.print("{s}{s}{d}", .{ if (i == 0) "" else ",", if (i % 2 == 0) "C" else "I", length });

121 fn sorted(s: *Scheduler) [10]Process {
122 var result = s.registry;
123 for (0..s.count) |i| {
124 const value = result[i];
125 var j = i;
126 while (j > 0 and std.mem.order(u8, result[j - 1].id, value.id) == .gt) : (j -= 1) result[j] = result[j - 1];
127 result[j] = value;
128 }
129 return result;
130 }

132 fn parameters(s: *Scheduler, words: []const Text, algorithm: ?usize, preset: ?usize) !?[5]I {
133 var p: [5]I = .{ 0, 4, 5, 0, 1 };
134 var seen: [5]bool = @splat(false);
135 if (preset) |k| seen[k] = true;
136 for (words) |word| {
137 const split = std.mem.findScalar(u8, word, '=').?;
138 const name = word[0..split];
139 const value = number(word[split + 1 ..]);
140 const key = lookup(name, &keys) orelse {
141 try s.line("ERROR unknown key {s}", .{name});
142 return null;
143 };
144 if (!uses(algorithm, key)) {
145 try s.line("ERROR key {s} not allowed for {s}", .{ name, algorithms[algorithm.?] });
146 return null;
147 }
148 if (seen[key]) {
149 try s.line("ERROR duplicate key {s}", .{name});
150 return null;
151 }
152 if (value < lower[key] or value > upper[key]) {
153 try s.line("ERROR invalid {s}", .{name});
154 return null;
155 }
156 seen[key] = true;
157 p[key] = value;
158 }
159 if (s.count == 0) {
160 try s.line("ERROR no processes", .{});
161 return null;
162 }
163 return p;
164 }

166 fn simulate(s: *Scheduler, algorithm: usize, p: [5]I, snapshot: ?I, trace: bool) !*Simulation {
167 const result = try std.heap.page_allocator.create(Simulation);
168 result.* = .{ .scheduler = s, .algorithm = algorithm, .p = s.sorted(), .count = s.count, .param = p, .random = p[4], .trace = trace };
169 for (0..s.count) |i| {
170 result.j[i].left = result.p[i].bursts[0];
171 result.j[i].effective = result.p[i].priority;
172 }
173 try result.runSimulation(snapshot);
174 return result;
175 }

177 fn command(s: *Scheduler, w: []const Text) !void {
178 const c = w[0];
179 if (eq(c, "PROC")) {
180 if (s.find(w[1]) != null) return s.line("ERROR duplicate process {s}", .{w[1]});
181 if (s.count == 10) return s.line("ERROR too many processes", .{});
182 const arrival = number(w[2]);
183 const priority = number(w[3]);
184 if (arrival > 200) return s.line("ERROR invalid arrival {s}", .{w[1]});
185 if (priority > 9) return s.line("ERROR invalid priority {s}", .{w[1]});
186 var p: Process = .{ .id = w[1], .arrival = arrival, .priority = priority, .count = w.len - 4 };
187 var valid = p.count % 2 == 1;
188 var cpu: I = 0;
189 var io: I = 0;
190 for (w[4..], 0..) |word, i| {
191 const length = number(word[1..]);
192 if (word[0] != (if (i % 2 == 0) @as(u8, 'C') else @as(u8, 'I')) or length < 1 or length > 30) valid = false;
193 p.bursts[i] = length;
194 if (i % 2 == 0) cpu += length else io += length;
195 }
196 if (!valid) return s.line("ERROR invalid bursts {s}", .{w[1]});
197 s.registry[s.count] = p;
198 s.count += 1;
199 return s.line("PROC {s} cpu={d} io={d}", .{ p.id, cpu, io });
200 }
201 if (eq(c, "GEN")) {
202 const amount = index(number(w[2]));
203 for (1..amount + 1) |i| {
204 const id = try s.format("{s}{d}", .{ w[1], i });
205 if (s.find(id) != null) return s.line("ERROR duplicate process {s}", .{id});
206 }
207 if (s.count + amount > 10) return s.line("ERROR too many processes", .{});
208 var random = number(w[3]);
209 try s.line("GEN count={d}", .{amount});
210 for (1..amount + 1) |i| {
211 const arrival = draw(&random, 21);
212 const priority = draw(&random, 10);
213 var p: Process = .{ .id = try s.format("{s}{d}", .{ w[1], i }), .arrival = arrival, .priority = priority };
214 const bursts = index(1 + draw(&random, 3));
215 for (0..bursts) |b| {
216 if (b > 0) {
217 p.bursts[p.count] = 1 + draw(&random, 8);
218 p.count += 1;
219 }
220 p.bursts[p.count] = 1 + draw(&random, 9);
221 p.count += 1;
222 }
223 s.registry[s.count] = p;
224 s.count += 1;
225 try s.describe(p);
226 }
227 return;
228 }
229 if (eq(c, "SETPRIO") or eq(c, "DEADLINE") or eq(c, "DROP")) {
230 const i = s.find(w[1]) orelse return s.line("ERROR unknown process {s}", .{w[1]});
231 if (eq(c, "DROP")) {
232 for (i + 1..s.count) |j| s.registry[j - 1] = s.registry[j];
233 s.count -= 1;
234 return s.line("DROP {s}", .{w[1]});
235 }
236 const value = number(w[2]);
237 const priority = eq(c, "SETPRIO");
238 if (value > (if (priority) @as(I, 9) else @as(I, 1000))) return s.line("ERROR invalid {s} {s}", .{ if (priority) "priority" else "deadline", w[1] });
239 if (priority) {
240 s.registry[i].priority = value;
241 return s.line("SETPRIO {s} priority={d}", .{ w[1], value });
242 }
243 s.registry[i].deadline = value;
244 if (value == 0) return s.line("DEADLINE {s} none", .{w[1]});
245 return s.line("DEADLINE {s} {d}", .{ w[1], value });
246 }
247 if (eq(c, "CLEAR")) {
248 try s.line("CLEAR count={d}", .{s.count});
249 s.count = 0;
250 return;
251 }
252 if (eq(c, "LIST")) {
253 try s.line("LIST count={d}", .{s.count});
254 const list = s.sorted();
255 for (list[0..s.count]) |p| try s.describe(p);
256 return;
257 }
258 const all = eq(c, "COMPARE") or eq(c, "RANK");
259 const diff = eq(c, "DIFF");
260 var algorithm: ?usize = null;
261 var second: usize = 0;
262 var metric_index: ?usize = null;
263 var offset: usize = 2;
264 if (!all) {
265 algorithm = lookup(w[1], &algorithms) orelse return s.line("ERROR unknown algorithm {s}", .{w[1]});
266 if (diff) {
267 second = lookup(w[2], &algorithms) orelse return s.line("ERROR unknown algorithm {s}", .{w[2]});
268 offset = 3;
269 }
270 } else if (eq(c, "RANK")) metric_index = lookup(w[1], &metrics) orelse return s.line("ERROR unknown metric {s}", .{w[1]}) else offset = 1;
271 var argument: I = -1;
272 var preset: ?usize = null;
273 var lo: I = 0;
274 var hi: I = 0;
275 if (eq(c, "WINDOW") or eq(c, "SNAPSHOT")) {
276 argument = number(w[2]);
277 offset = 3;
278 if (eq(c, "WINDOW") and (argument < 1 or argument > 1000)) return s.line("ERROR invalid window", .{});
279 }
280 if (eq(c, "SWEEP")) {
281 const key = lookup(w[2], &keys) orelse return s.line("ERROR unknown key {s}", .{w[2]});
282 preset = key;
283 if (!uses(algorithm, key)) return s.line("ERROR key {s} not allowed for {s}", .{ w[2], algorithms[algorithm.?] });
284 lo = number(w[3]);
285 hi = number(w[4]);
286 if (lo > hi or lo < lower[key] or hi > upper[key] or hi - lo > 20) return s.line("ERROR invalid range", .{});
287 offset = 5;
288 }
289 var p = (try s.parameters(w[offset..], if (all or diff) null else algorithm, preset)) orelse return;
290 const note = try s.notation(p, if (all or diff) null else algorithm);
291 if (preset) |key| {
292 try s.line("SWEEP {s} {s} {d} {d}", .{ algorithms[algorithm.?], keys[key], lo, hi });
293 var best_total: ?I = null;
294 var best_value = lo;
295 var value = lo;
296 while (value <= hi) : (value += 1) {
297 p[key] = value;
298 const simulation = try s.simulate(algorithm.?, p, null, false);
299 try s.line("{s} turnaround={s} waiting={s} response={s} util={s} end={d} switches={d}", .{ try s.notation(p, algorithm), try s.decimal(simulation.totals[0], @intCast(s.count)), try s.decimal(simulation.totals[1], @intCast(s.count)), try s.decimal(simulation.totals[2], @intCast(s.count)), try simulation.util(), simulation.end, simulation.switches });
300 if (best_total == null or simulation.totals[1] < best_total.?) {
301 best_total = simulation.totals[1];
302 best_value = value;
303 }
304 std.heap.page_allocator.destroy(simulation);
305 }
306 return s.line("BEST {s}={d}", .{ keys[key], best_value });
307 }
308 if (diff) {
309 const one = try s.simulate(algorithm.?, p, null, false);
310 defer std.heap.page_allocator.destroy(one);
311 const two = try s.simulate(second, p, null, false);
312 defer std.heap.page_allocator.destroy(two);
313 try s.line("DIFF {s} {s} {s}", .{ w[1], w[2], note });
314 var wins: I = 0;
315 var losses: I = 0;
316 var ties: I = 0;
317 for (0..s.count) |i| {
318 try s.line("{s} turnaround={d}/{d} waiting={d}/{d} response={d}/{d}", .{ one.p[i].id, one.metric(i, 0), two.metric(i, 0), one.metric(i, 1), two.metric(i, 1), one.metric(i, 2), two.metric(i, 2) });
319 if (one.metric(i, 0) < two.metric(i, 0)) wins += 1 else if (one.metric(i, 0) > two.metric(i, 0)) losses += 1 else ties += 1;
320 }
321 return s.line("WINS {s}={d} {s}={d} ties={d}", .{ w[1], wins, w[2], losses, ties });
322 }
323 if (all) {
324 var totals: [9][3]I = undefined;
325 var values: [9]I = undefined;
326 var texts: [9]Text = undefined;
327 if (metric_index) |_| try s.line("RANK {s} {s}", .{ w[1], note }) else {
328 try s.line("COMPARE {s}", .{note});
329 try s.line("{s: <8} {s: >8} {s: >8} {s: >8} {s: >8} {s: >8} {s: >8} {s: >8}", .{ "ALGO", "TURN", "WAIT", "RESP", "UTIL", "END", "SWITCH", "MISS" });
330 }
331 for (0..9) |a| {
332 const simulation = try s.simulate(a, p, null, false);
333 totals[a] = simulation.totals;
334 if (metric_index) |m| {
335 values[a] = simulation.rankValue(m);
336 texts[a] = if (m < 3) try s.decimal(values[a], @intCast(s.count)) else try s.format("{d}", .{values[a]});
337 } else try s.line("{s: <8} {s: >8} {s: >8} {s: >8} {s: >8} {s: >8} {s: >8} {s: >8}", .{ algorithms[a], try s.decimal(totals[a][0], @intCast(s.count)), try s.decimal(totals[a][1], @intCast(s.count)), try s.decimal(totals[a][2], @intCast(s.count)), try simulation.util(), try s.format("{d}", .{simulation.end}), try s.format("{d}", .{simulation.switches}), try s.format("{d}", .{simulation.missed}) });
338 std.heap.page_allocator.destroy(simulation);
339 }
340 if (metric_index == null) {
341 var best: [3]usize = @splat(0);
342 for (0..3) |m| for (1..9) |a| {
343 if (totals[a][m] < totals[best[m]][m]) best[m] = a;
344 };
345 return s.line("BEST turnaround={s} waiting={s} response={s}", .{ algorithms[best[0]], algorithms[best[1]], algorithms[best[2]] });
346 }
347 var ids: [9]usize = undefined;
348 for (0..9) |i| {
349 var j = i;
350 while (j > 0 and values[ids[j - 1]] > values[i]) : (j -= 1) ids[j] = ids[j - 1];
351 ids[j] = i;
352 }
353 for (ids) |i| {
354 var rank: usize = 1;
355 for (values) |v| if (v < values[i]) {
356 rank += 1;
357 };
358 try s.line("{d} {s} {s}", .{ rank, algorithms[i], texts[i] });
359 }
360 return;
361 }
362 const trace = eq(c, "TRACE");
363 if (trace) try s.line("TRACE {s} {s}", .{ algorithms[algorithm.?], note });
364 const simulation = try s.simulate(algorithm.?, p, if (eq(c, "SNAPSHOT")) argument else null, trace);
365 defer std.heap.page_allocator.destroy(simulation);
366 try simulation.output(c, argument);
367 }

399 fn event(s: *Simulation, t: I, kind: usize, id: ?usize, value: ?I) !void {
400 const names = [_]Text{ "PREEMPT", "EXPIRE", "DEMOTE", "BOOST", "AGE", "ARRIVE", "IODONE", "START", "DONE", "IO", "SWITCH", "DISPATCH" };
401 if (kind < 5) s.events[kind] += 1;
402 if (s.trace) {
403 try s.scheduler.out.print("{d} {s}", .{ t, names[kind] });
404 if (id) |i| try s.scheduler.out.print(" {s}", .{s.p[i].id});
405 if (value) |v| try s.scheduler.out.print(" {d}", .{v});
406 try s.scheduler.out.writeByte('\n');
407 }
408 }

410 fn enqueue(s: *Simulation, i: usize, t: I) void {
411 s.j[i].state = 1;
412 s.j[i].enqueued = t;
413 s.q[if (s.algorithm == 8) s.j[i].level else 0].append(i);

416 fn start(s: *Simulation, i: usize, t: I) void {
417 s.cpu = 3;
418 s.current = i;
419 s.run = 0;
420 s.j[i].state = 3;
421 if (s.j[i].start < 0) s.j[i].start = t;
422 if (s.j[i].began[s.j[i].burst / 2] < 0) s.j[i].began[s.j[i].burst / 2] = t;
423 }

425 fn deadline(s: *Simulation, i: usize) I {
426 return if (s.p[i].deadline == 0) 1000000000 else s.p[i].arrival + s.p[i].deadline;
427 }

429 fn preempts(s: *Simulation, i: usize) bool {
430 for (s.q, 0..) |queue, level| for (queue.ids[0..queue.count]) |other| {
431 if ((s.algorithm == 2 and s.j[other].left < s.j[i].left) or (s.algorithm == 4 and s.j[other].effective < s.j[i].effective) or (s.algorithm == 6 and s.deadline(other) < s.deadline(i)) or (s.algorithm == 8 and level < s.j[i].level)) return true;
432 };
433 return false;
434 }

436 fn choose(s: *Simulation, queue: *Queue, t: I) usize {
437 if (s.algorithm == 7) {
438 var total: I = 0;
439 for (queue.ids[0..queue.count]) |i| total += 10 - s.p[i].priority;
440 var target = draw(&s.random, total);
441 for (queue.ids[0..queue.count], 0..) |i, position| {
442 target -= 10 - s.p[i].priority;
443 if (target < 0) return position;
444 }
445 }
446 var selected: usize = 0;
447 for (queue.ids[0..queue.count], 0..) |i, position| {
448 const other = queue.ids[selected];
449 const a = s.j[i];
450 const b = s.j[other];
451 const better = switch (s.algorithm) {
452 1, 2 => a.left < b.left,
453 4 => a.effective < b.effective,
454 5 => (t - a.enqueued + a.left) * b.left > (t - b.enqueued + b.left) * a.left,
455 6 => s.deadline(i) < s.deadline(other),
456 else => false,
457 };
458 if (better) selected = position;
459 }
460 return selected;
461 }

463 fn capture(s: *Simulation) !void {
464 var buffer: std.Io.Writer.Allocating = .init(s.scheduler.gpa);
465 const out = &buffer.writer;
466 if (s.cpu == 0) try out.writeAll("CPU IDLE\n") else try out.print("CPU {s} {s}\n", .{ states[s.cpu], s.p[s.current].id });
467 for (0..(if (s.algorithm == 8) @as(usize, 3) else @as(usize, 1))) |level| {
468 if (s.algorithm == 8) try out.print("QUEUE{d}", .{level}) else try out.writeAll("QUEUE");
469 const queue = s.q[level];
470 if (queue.count == 0) try out.writeAll(" none");
471 for (queue.ids[0..queue.count]) |i| try out.print(" {s}", .{s.p[i].id});
472 try out.writeByte('\n');
473 }
474 for (0..s.count) |i| {
475 const j = s.j[i];
476 try out.print("{s} {s}", .{ s.p[i].id, states[j.state] });
477 if (j.state == 0) try out.print(" at={d}", .{s.p[i].arrival}) else if (j.state == 5) try out.print(" at={d}", .{j.finish}) else {
478 try out.print(" burst={d} left={d}", .{ j.burst + 1, j.left });
479 if (s.algorithm == 4) try out.print(" prio={d}", .{j.effective});
480 if (s.algorithm == 8) try out.print(" level={d} used={d}", .{ j.level, j.used });
481 }
482 try out.writeByte('\n');
483 }
484 s.snapshot = try buffer.toOwnedSlice();
485 }

487 fn runSimulation(s: *Simulation, snapshot_time: ?I) !void {
488 var t: I = 0;
489 while (true) : (t += 1) {
490 if (s.algorithm == 4) for (s.q[0].ids[0..s.q[0].count]) |i| {
491 s.j[i].wait += 1;
492 if (s.j[i].wait == s.param[2]) {
493 s.j[i].wait = 0;
494 if (s.j[i].effective > 0) {
495 s.j[i].effective -= 1;
496 try s.event(t, 4, i, s.j[i].effective);
497 }
498 }
499 };
500 for (0..s.count) |i| if (s.j[i].state == 4 and s.j[i].left == 0) {
501 s.j[i].burst += 1;
502 s.j[i].left = s.p[i].bursts[s.j[i].burst];
503 s.j[i].ready[s.j[i].burst / 2] = t;
504 s.enqueue(i, t);
505 try s.event(t, 6, i, null);
506 };
507 for (0..s.count) |i| if (s.j[i].state == 0 and s.p[i].arrival == t) {
508 s.j[i].ready[0] = t;
509 s.enqueue(i, t);
510 try s.event(t, 5, i, null);
511 };
512 if (s.cpu == 2 and s.switch_left == 0) {
513 s.start(s.current, t);
514 try s.event(t, 7, s.current, null);
515 } else if (s.cpu == 3) {
516 const i = s.current;
517 const j = &s.j[i];
518 if (j.left == 0 and j.burst + 1 == s.p[i].count) {
519 j.state = 5;
520 j.finish = t;
521 try s.event(t, 8, i, null);
522 s.cpu = 0;
523 } else {
524 const demoted = s.algorithm == 8 and j.level < 2 and j.used == (if (j.level == 0) @as(I, 4) else @as(I, 8));
525 if (demoted) {
526 j.level += 1;
527 j.used = 0;
528 try s.event(t, 2, i, @intCast(j.level));
529 }
530 if (j.left == 0) {
531 j.burst += 1;
532 j.left = s.p[i].bursts[j.burst];
533 j.state = 4;
534 try s.event(t, 9, i, j.left);
535 s.cpu = 0;
536 } else if (demoted) {
537 s.enqueue(i, t);
538 s.cpu = 0;
539 } else if ((s.algorithm == 3 or s.algorithm == 7) and s.run == s.param[1]) {
540 s.enqueue(i, t);
541 try s.event(t, 1, i, null);
542 s.cpu = 0;
543 } else if (s.preempts(i)) {
544 s.enqueue(i, t);
545 try s.event(t, 0, i, null);
546 s.cpu = 0;
547 }
548 }
549 }
550 var done: usize = 0;
551 for (s.j[0..s.count]) |j| if (j.state == 5) {
552 done += 1;
553 };
554 if (done == s.count) {
555 s.end = t;
556 if (snapshot_time != null and s.snapshot.len == 0) try s.capture();
557 break;
558 }
559 if (s.algorithm == 8 and s.param[3] > 0 and t > 0 and @mod(t, s.param[3]) == 0) {
560 for (1..3) |level| {
561 for (s.q[level].ids[0..s.q[level].count]) |i| s.q[0].append(i);
562 s.q[level].count = 0;
563 }
564 for (s.j[0..s.count]) |*j| {
565 j.level = 0;
566 j.used = 0;
567 }
568 try s.event(t, 3, null, null);
569 }
570 if (s.cpu == 0) {
571 var selected_level: ?usize = null;
572 for (0..3) |level| if (s.q[level].count > 0) {
573 selected_level = level;
574 break;
575 };
576 if (selected_level) |level| {
577 const queue = &s.q[level];
578 const position = s.choose(queue, t);
579 const i = queue.remove(position);
580 const j = &s.j[i];
581 j.effective = s.p[i].priority;
582 j.wait = 0;
583 j.slices += 1;
584 j.selections[j.burst / 2] += 1;
585 const changed = s.previous == null or i != s.previous.?;
586 if (changed) s.switches += 1;
587 if (changed and s.param[0] > 0) {
588 s.cpu = 2;
589 s.current = i;
590 s.switch_left = s.param[0];
591 j.state = 2;
592 try s.event(t, 10, i, null);
593 } else {
594 s.start(i, t);
595 try s.event(t, 11, i, null);
596 }
597 }
598 }
599 if (snapshot_time != null and t == snapshot_time.?) try s.capture();
600 const qlen = s.q[0].count + s.q[1].count + s.q[2].count;
601 s.sumq += @intCast(qlen);
602 s.maxq = @max(s.maxq, qlen);
603 for (0..s.count) |i| s.timeline[i][index(t)] = "-.s#i-"[s.j[i].state];
604 if (s.cpu == 3) {
605 const j = &s.j[s.current];
606 j.left -= 1;
607 j.used += 1;
608 s.run += 1;
609 s.previous = s.current;
610 s.labels[index(t)] = @intCast(s.current);
611 s.busy += 1;
612 if (j.left == 0) j.ended[j.burst / 2] = t + 1;
613 } else if (s.cpu == 2) {
614 s.switch_left -= 1;
615 s.labels[index(t)] = -2;
616 s.switching += 1;
617 } else {
618 s.labels[index(t)] = -1;
619 s.idle += 1;
620 }
621 for (s.j[0..s.count]) |*j| if (j.state == 4) {
622 j.left -= 1;
623 };
624 }
625 for (0..s.count) |i| {
626 for (0..3) |m| s.totals[m] += s.metric(i, m);
627 if (s.p[i].deadline > 0) {
628 if (s.j[i].finish > s.deadline(i)) s.missed += 1 else s.met += 1;
629 }
630 }
631 }

633 fn metric(s: *Simulation, i: usize, m: usize) I {
634 const turnaround = s.j[i].finish - s.p[i].arrival;
635 if (m == 0) return turnaround;
636 if (m == 2) return s.j[i].start - s.p[i].arrival;
637 var total: I = 0;
638 for (s.p[i].bursts[0..s.p[i].count]) |length| total += length;
639 return turnaround - total;
640 }
641
642 fn rankValue(s: *Simulation, m: usize) I {
643 return if (m < 3) s.totals[m] else if (m == 3) s.end else if (m == 4) s.switches else s.missed;
644 }
645
646 fn label(s: *Simulation, time: usize) Text {
647 const i = s.labels[time];
648 return if (i == -1) "IDLE" else if (i == -2) "CS" else s.p[index(i)].id;
649 }
650
651 fn util(s: *Simulation) !Text {
652 return s.scheduler.decimal(s.busy * 100, s.end);
653 }
654
655 fn output(s: *Simulation, command: Text, argument: I) !void {
656 const host = s.scheduler;
657 const out = host.out;
658 const note = try host.notation(s.param, s.algorithm);
659 if (eq(command, "TRACE")) return host.line("END {d}", .{s.end});
660 if (eq(command, "TIMELINE")) {
661 if (s.end > 100) return host.line("ERROR too long {d}", .{s.end});
662 try host.line("TIMELINE {s} {s} end={d}", .{ algorithms[s.algorithm], note, s.end });
663 for (0..s.count) |i| try host.line("{s: <8} {s}", .{ s.p[i].id, s.timeline[i][0..index(s.end)] });
664 return;
665 }
666 if (eq(command, "SNAPSHOT")) {
667 try host.line("SNAPSHOT {s} {s} t={d}", .{ algorithms[s.algorithm], note, argument });
668 return out.writeAll(s.snapshot);
669 }
670 if (eq(command, "WINDOW")) {
671 try host.line("WINDOW {s} {s} width={d}", .{ algorithms[s.algorithm], note, argument });
672 var begin: I = 0;
673 while (begin < s.end) : (begin += argument) {
674 const end = @min(begin + argument, s.end);
675 var busy: I = 0;
676 var cs: I = 0;
677 var idle: I = 0;
678 var hits: [10]I = @splat(0);
679 for (s.labels[index(begin)..index(end)]) |i| {
680 if (i == -1) idle += 1 else if (i == -2) cs += 1 else {
681 busy += 1;
682 hits[index(i)] += 1;
683 }
684 }
685 var top: ?usize = null;
686 for (0..s.count) |i| if (hits[i] > 0 and (top == null or hits[i] > hits[top.?])) {
687 top = i;
688 };
689 try host.line("{d}-{d} busy={d} switch={d} idle={d} util={s} top={s}", .{ begin, end, busy, cs, idle, try host.decimal(busy * 100, end - begin), if (top) |i| s.p[i].id else "none" });
690 }
691 return;
692 }
693 try host.line("{s} {s} {s}", .{ command, algorithms[s.algorithm], note });
694 if (eq(command, "BURSTS")) {
695 for (0..s.count) |i| {
696 var b: usize = 0;
697 while (b < s.p[i].count) : (b += 2) {
698 const k = b / 2;
699 const j = s.j[i];
700 const length = s.p[i].bursts[b];
701 try host.line("{s} #{d} len={d} ready={d} start={d} end={d} wait={d} slices={d}", .{ s.p[i].id, k + 1, length, j.ready[k], j.began[k], j.ended[k], j.ended[k] - j.ready[k] - length, j.selections[k] });
702 }
703 }
704 return;
705 }
706 if (eq(command, "STATS")) {
707 for (0..3) |m| {
708 var values: [10]I = undefined;
709 var lo: usize = 0;
710 var hi: usize = 0;
711 for (0..s.count) |i| {
712 values[i] = s.metric(i, m);
713 if (values[i] < s.metric(lo, m)) lo = i;
714 if (values[i] > s.metric(hi, m)) hi = i;
715 }
716 for (0..s.count) |i| {
717 const value = values[i];
718 var j = i;
719 while (j > 0 and values[j - 1] > value) : (j -= 1) values[j] = values[j - 1];
720 values[j] = value;
721 }
722 const middle = s.count / 2;
723 const doubled = if (s.count % 2 == 1) 2 * values[middle] else values[middle - 1] + values[middle];
724 try host.line("{s} min={d} ({s}) max={d} ({s}) median={d}.{d} total={d}", .{ metrics[m], s.metric(lo, m), s.p[lo].id, s.metric(hi, m), s.p[hi].id, @divTrunc(doubled, 2), 5 * @mod(doubled, 2), s.totals[m] });
725 }
726 return;
727 }
728 var begin: usize = 0;
729 var segments: usize = 0;
730 var t: usize = 1;
731 while (t <= index(s.end)) : (t += 1) if (t == index(s.end) or !eq(s.label(t), s.label(begin))) {
732 if (segments % 8 == 0) try out.writeAll("GANTT ");
733 try out.print("[{d}-{d} {s}]", .{ begin, t, s.label(begin) });
734 segments += 1;
735 if (segments % 8 == 0 or t == index(s.end)) try out.writeByte('\n');
736 begin = t;
737 };
738 for (0..s.count) |i| {
739 try out.print("{s} arrival={d} done={d} turnaround={d} waiting={d} response={d} slices={d}", .{ s.p[i].id, s.p[i].arrival, s.j[i].finish, s.metric(i, 0), s.metric(i, 1), s.metric(i, 2), s.j[i].slices });
740 if (s.p[i].deadline > 0) try out.print(" deadline={d} late={d}", .{ s.deadline(i), @max(0, s.j[i].finish - s.deadline(i)) });
741 try out.writeByte('\n');
742 }
743 try host.line("AVERAGE turnaround={s} waiting={s} response={s}", .{ try host.decimal(s.totals[0], @intCast(s.count)), try host.decimal(s.totals[1], @intCast(s.count)), try host.decimal(s.totals[2], @intCast(s.count)) });
744 try host.line("CPU end={d} busy={d} switch={d} idle={d} switches={d} util={s}", .{ s.end, s.busy, s.switching, s.idle, s.switches, try s.util() });
745 try host.line("QUEUE max={d} avg={s}", .{ s.maxq, try host.decimal(s.sumq, s.end) });
746 try host.line("EVENTS preempt={d} expire={d} demote={d} boost={d} age={d}", .{ s.events[0], s.events[1], s.events[2], s.events[3], s.events[4] });
747 try host.line("DEADLINE met={d} missed={d}", .{ s.met, s.missed });
748 }
749};

751fn readLine(reader: *std.Io.Reader, gpa: std.mem.Allocator) !?Text {
752 var buffer: std.Io.Writer.Allocating = .init(gpa);
753 defer buffer.deinit();
754 _ = try reader.streamDelimiterEnding(&buffer.writer, '\n');
755 if (reader.bufferedLen() == 0 and buffer.written().len == 0) return null;
756 if (reader.bufferedLen() > 0) reader.toss(1);
757 return buffer.toOwnedSlice();
758}

760pub fn main(init: std.process.Init) !void {
761 var arena: std.heap.ArenaAllocator = .init(std.heap.page_allocator);
762 defer arena.deinit();
763 const gpa = arena.allocator();
764 var ib: [4096]u8 = undefined;
765 var ob: [4096]u8 = undefined;
766 var input = std.Io.File.stdin().readerStreaming(init.io, &ib);
767 var output = std.Io.File.stdout().writerStreaming(init.io, &ob);
768 var scheduler: Scheduler = .{ .gpa = gpa, .out = &output.interface };
769 const first = std.mem.trim(u8, (try readLine(&input.interface, gpa)).?, "\r");
770 const count = index(number(first));
771 for (0..count) |_| {
772 const text = std.mem.trim(u8, (try readLine(&input.interface, gpa)).?, "\r");
773 var words: std.ArrayList(Text) = .empty;
774 defer words.deinit(gpa);
775 var it = std.mem.splitScalar(u8, text, ' ');
776 while (it.next()) |word| try words.append(gpa, word);
777 try scheduler.command(words.items);
778 }
779 try output.interface.flush();
780}
