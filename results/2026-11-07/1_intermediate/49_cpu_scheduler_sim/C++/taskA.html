178 string algo, cpu = "IDLE";

189 void event(const string& name, const string& detail = "") {
190 r.events.push_back(num(time)+" "+name+(detail.empty()?"":" "+detail));

192 }
193 void enqueue(int i) {
194 Job& j = r.jobs[i];
195 j.state = "READY";
196 j.enqueue = time;
197 queues[algo == "MLFQ"?j.level:0].push_back(i);
198 }

200 Job& j = r.jobs[i];
201 j.records.push_back({
202 j.left, time
203 });
204 enqueue(i);

206 void start() {
207 Job& j = r.jobs[current];
208 cpu = "RUN";
209 j.state = "RUN";
210 runLength = 0;
211 if (j.start<0)j.start = time;
212 if (j.records.back().start<0)j.records.back().start = time;

214 int absolute(int i) const {
215 const Process& process = r.jobs[i].process;
216 return process.deadline<0?2147483647:process.arrival+process.deadline;
217 }
218 vector<string> snapshot() const {
219 vector<string> output {
220 cpu == "IDLE"?"CPU IDLE":"CPU "+cpu+" "+r.jobs[current].process.id
221 };
222 for (int q = 0;q<(algo == "MLFQ"?3:1);++q) {
223 string s = "QUEUE"+(algo == "MLFQ"?num(q):"")+" ";
224 if (queues[q].empty())s += "none";
225 else for (int i:queues[q]) {
226 if (s.back() != ' ')s += ' ';
227 s += r.jobs[i].process.id;
228 }
229 output.push_back(s);
230 }
231 for (const Job& j:r.jobs) {
232 string s = j.process.id+" "+j.state;
233 if (j.state == "FUTURE")s += " at="+num(j.process.arrival);
234 else if (j.state == "DONE")s += " at="+num(j.finish);
235 else {
236 s += " burst="+num(j.burst+1)+" left="+num(j.left);
237 if (algo == "PRIORITY")s += " prio="+num(j.effective);
238 if (algo == "MLFQ")s += " level="+num(j.level)+" used="+num(j.usage);
239 }
240 output.push_back(s);
241 }
242 return output;
243 }
244 bool preempts() const {
245 const Job& j = r.jobs[current];
246 for (int i:ready()) {
247 const Job& q = r.jobs[i];
248 if (algo == "SRTF" && q.left<j.left)return true;
249 if (algo == "PRIORITY" && q.effective<j.effective)return true;
250 if (algo == "EDF" && absolute(i)<absolute(current))return true;
251 if (algo == "MLFQ" && q.level<j.level)return true;
252 }
253 return false;
254 }
255 int select() {
256 vector<int> list = ready();
257 int chosen = list[0];
258 if (algo == "LOTTERY") {
259 random = random*48271%2147483647;
260 int tickets = 0;
261 for (int i:list)tickets += 10-r.jobs[i].process.priority;
262 Integer draw = random%tickets;
263 for (int i:list) {
264 draw -= 10-r.jobs[i].process.priority;
265 if (draw<0)return i;
266 }
267 }
268 for (int i:list) {
269 const Job& a = r.jobs[i];
270 const Job& b = r.jobs[chosen];
271 bool better = false;
272 if (algo == "SJF" || algo == "SRTF")better = a.left<b.left;
273 else if (algo == "PRIORITY")better = a.effective<b.effective;
274 else if (algo == "HRRN")better = (Integer)(time-a.enqueue+a.left)*b.left>(Integer)(time-b.enqueue+b.left)*a.left;
275 else if (algo == "EDF")better = absolute(i)<absolute(chosen);
276 if (better)chosen = i;
277 }
278 return chosen;
279 }
280 public:
281 Simulation(const vector<Process>& processes, string algorithm, Parameters params, int snapshotAt = -1):algo(std::move(algorithm)),
282 p(std::move(params)), snapshotTime(snapshotAt), random(p.at("seed")) {
283 for (const Process& process:processes) {
284 Job j;
285 j.process = process;
286 j.left = process.bursts[0];
287 j.effective = process.priority;
288 r.jobs.push_back(j);
289 }
290 }
291 Result run() {
292 for (time = 0;;++time) {
293 if (algo == "PRIORITY")for (int i:queues[0]) {
294 Job& j = r.jobs[i];
295 if (++j.wait == p["k"]) {
296 j.wait = 0;
297 if (j.effective)event("AGE", j.process.id+" "+num(--j.effective));
298 }
299 }
300 for (int i = 0;i<(int)r.jobs.size();++i) {
301 Job& j = r.jobs[i];
302 if (j.state == "IO" && !j.left) {
303 ++j.burst;
304 j.left = j.process.bursts[j.burst];
305 fresh(i);
306 event("IODONE", j.process.id);
307 }
308 }
309 for (int i = 0;i<(int)r.jobs.size();++i) {
310 Job& j = r.jobs[i];
311 if (j.state == "FUTURE" && j.process.arrival == time) {
312 fresh(i);
313 event("ARRIVE", j.process.id);
314 }
315 }
316 if (cpu == "SWITCH" && !switchLeft) {
317 start();
318 event("START", r.jobs[current].process.id);
319 }
320 else if (cpu == "RUN") {
321 Job& j = r.jobs[current];
322 if (!j.left && j.burst == (int)j.process.bursts.size()-1) {
323 j.state = "DONE";
324 j.finish = time;
325 event("DONE", j.process.id);
326 cpu = "IDLE";
327 }
328 else {
329 bool demoted = false;
330 if (algo == "MLFQ" && j.level<2 && j.usage == (j.level == 0?4:8)) {
331 ++j.level;
332 j.usage = 0;
333 demoted = true;
334 event("DEMOTE", j.process.id+" "+num(j.level));
335 }
336 if (!j.left) {
337 ++j.burst;
338 j.left = j.process.bursts[j.burst];
339 j.state = "IO";
340 cpu = "IDLE";
341 event("IO", j.process.id+" "+num(j.left));
342 }
343 else if (demoted) {
344 enqueue(current);
345 cpu = "IDLE";
346 }
347 else if ((algo == "RR" || algo == "LOTTERY") && runLength == p["q"]) {
348 enqueue(current);
349 cpu = "IDLE";
350 event("EXPIRE", j.process.id);
351 }
352 else if (preempts()) {
353 enqueue(current);
354 cpu = "IDLE";
355 event("PREEMPT", j.process.id);
356 }
357 }
358 }
359 if (all_of(r.jobs.begin(), r.jobs.end(), [](const Job& j) {
360 return j.state == "DONE";
361 })) {
362 r.end = time;
363 if (r.snapshot.empty() && snapshotTime >= 0)r.snapshot = snapshot();
364 break;
365 }
366 if (algo == "MLFQ" && p["s"] && time>0 && time%p["s"] == 0) {
367 vector<int> all = ready();
368 queues[0] = all;
369 queues[1].clear();
370 queues[2].clear();
371 for (Job& j:r.jobs) {
372 j.level = 0;
373 j.usage = 0;
374 }
375 event("BOOST");
376 }
377 if (cpu == "IDLE" && !ready().empty()) {
378 current = select();
379 Job& j = r.jobs[current];
380 vector<int>& q = queues[algo == "MLFQ"?j.level:0];
381 q.erase(find(q.begin(), q.end(), current));
382 j.effective = j.process.priority;
383 j.wait = 0;
384 ++j.records.back().slices;
385 bool change = current != previous;
386 if (change)++r.switches;
387 if (change && p["cs"]) {
388 cpu = "SWITCH";
389 j.state = "SWITCH";
390 switchLeft = (int)p["cs"];
391 event("SWITCH", j.process.id);
392 } else {
393 start();
394 event("DISPATCH", j.process.id);
395 }
396 }
397 if (time == snapshotTime)r.snapshot = snapshot();
398 r.queueLengths.push_back((int)ready().size());
399 r.modes.push_back(cpu);
400 for (Job& j:r.jobs)j.timeline += j.state == "RUN"?'#':j.state == "SWITCH"?'s':j.state == "READY"?'.':j.state == "IO"?'i':'-';
401 if (cpu == "RUN") {
402 Job& j = r.jobs[current];
403 --j.left;
404 ++runLength;
405 ++j.usage;
406 previous = current;
407 r.labels.push_back(j.process.id);
408 ++r.busy;
409 if (!j.left)j.records.back().end = time+1;
410 }
411 else if (cpu == "SWITCH") {
412 --switchLeft;
413 r.labels.push_back("CS");
414 ++r.switching;
415 } else {
416 r.labels.push_back("IDLE");
417 ++r.idle;
418 }
419 for (Job& j:r.jobs)if (j.state == "IO")--j.left;
420 }
421 for (const Job& j:r.jobs) {
422 array<Integer, 3> m = metrics(j);
423 for (int k = 0;k<3;++k)r.totals[k] += m[k];
424 if (j.process.deadline >= 0) {
425 if (j.finish <= j.process.arrival+j.process.deadline)++r.met;
426 else ++r.missed;
427 }
428 }
429 return r;
430 }

571 void command(const string& c, const vector<string>& a) {
572 string id = a.empty()?"":a[0];
573 if (c == "PROC") {
574 require(registry.count(id),"duplicate process "+id);
575 require(registry.size() == 10,"too many processes");
576 int arrival = stoi(a[1]), priority = stoi(a[2]);
577 require(arrival>200,"invalid arrival "+id);
578 require(priority>9,"invalid priority "+id);
579 bool bad = (a.size()-3)%2 == 0;
580 vector<int> bursts;
581 Integer cpu = 0, io = 0;
582 for (int i = 3;i<(int)a.size();++i) {
583 int n = stoi(a[i].substr(1));
584 bad|=a[i][0] != ((i-3)%2?'I':'C') || n<1 || n>30;
585 bursts.push_back(n);
586 ((i-3)%2?io:cpu) += n;
587 }
588 require(bad,"invalid bursts "+id);
589 registry[id] = {
590 id, arrival, priority, -1, bursts
591 };
592 cout<<"PROC "<<id<<" cpu="<<cpu<<" io="<<io<<'\n';
593 return;
594 }
595 if (c == "GEN") {
596 int n = stoi(a[1]);
597 for (int i = 1;i <= n;++i)require(registry.count(id+num(i)),"duplicate process "+id+num(i));
598 require(registry.size()+n>10,"too many processes");
599 Integer random = stoll(a[2]);
600 auto draw = [&](int modulus) {
601 random = random*48271%2147483647;
602 return (int)(random%modulus);
603 };
604 cout<<"GEN count="<<n<<'\n';
605 for (int i = 1;i <= n;++i) {
606 Process p;
607 p.id = id+num(i);
608 p.arrival = draw(21);
609 p.priority = draw(10);
610 int m = 1+draw(3);
611 for (int j = 0;j<m;++j) {
612 if (j)p.bursts.push_back(1+draw(8));
613 p.bursts.push_back(1+draw(9));
614 }
615 registry[p.id] = p;
616 cout<<describe(p)<<'\n';
617 }
618 return;
619 }
620 if (c == "SETPRIO" || c == "DEADLINE" || c == "DROP") {
621 Process& p = process(id);
622 if (c == "DROP") {
623 registry.erase(id);
624 cout<<"DROP "<<id<<'\n';
625 } else {
626 int v = stoi(a[1]);
627 if (c == "SETPRIO") {
628 require(v>9,"invalid priority "+id);
629 p.priority = v;
630 cout<<"SETPRIO "<<id<<" priority="<<v<<'\n';
631 } else {
632 require(v>1000,"invalid deadline "+id);
633 p.deadline = v?v:-1;
634 cout<<"DEADLINE "<<id<<' '<<(v?num(v):"none")<<'\n';
635 }
636 }
637 return;
638 }
639 if (c == "CLEAR") {
640 cout<<"CLEAR count="<<registry.size()<<'\n';
641 registry.clear();
642 return;
643 }
644 if (c == "LIST") {
645 cout<<"LIST count="<<registry.size()<<'\n';
646 for (const Process& p:processes())cout<<describe(p)<<'\n';
647 return;
648 }
649 bool multiple = c == "DIFF" || c == "COMPARE" || c == "RANK";
650 string algo = c == "COMPARE" || c == "RANK"?"":id;
651 if (!algo.empty())require(!contains(algorithms, algo),"unknown algorithm "+algo);
652 if (c == "DIFF")require(!contains(algorithms, a[1]),"unknown algorithm "+a[1]);
653 if (c == "RANK")require(!contains(metricNames, id) && id != "end" && id != "switches" && id != "missed","unknown metric "+id);
654 int start = c == "COMPARE"?0:1;
655 string preset;
656 if (c == "SNAPSHOT" || c == "WINDOW" || c == "DIFF")start = 2;
657 if (c == "WINDOW")require(stoll(a[1])<1 || stoll(a[1])>1000,"invalid window");
658 if (c == "SWEEP") {
659 string key = a[1];
660 Integer lo = stoll(a[2]), hi = stoll(a[3]);
661 require(!ranges.count(key),"unknown key "+key);
662 require(!contains(used.at(algo), key),"key "+key+" not allowed for "+algo);
663 require(lo>hi || lo<ranges.at(key).first || hi>ranges.at(key).second || hi-lo>20,"invalid range");
664 start = 4;
665 preset = key;
666 }
667 Parameters p = parameters(vector<string>(a.begin()+start, a.end()), multiple?"":algo, preset);
668 require(registry.empty(),"no processes");
669 vector<Process> ps = processes();
670 if (c == "SWEEP") {
671 string key = a[1];
672 Integer lo = stoll(a[2]), hi = stoll(a[3]), best = lo, value = -1;
673 cout<<"SWEEP "<<algo<<' '<<key<<' '<<lo<<' '<<hi<<'\n';
674 for (Integer v = lo;v <= hi;++v) {
675 p[key] = v;
676 Result r = Simulation(ps, algo, p).run();
677 cout<<notation(p, algo)<<" turnaround="<<average(r, 0)<<" waiting="<<average(r, 1)<<" response="<<average(r, 2)
678 <<" util="<<util(r)<<" end="<<r.end<<" switches="<<r.switches<<'\n';
679 if (value<0 || r.totals[1]<value) {
680 best = v;
681 value = r.totals[1];
682 }
683 }
684 cout<<"BEST "<<key<<'='<<best<<'\n';
685 return;
686 }
687 if (c == "DIFF") {
688 Result x = Simulation(ps, algo, p).run(), y = Simulation(ps, a[1], p).run();
689 cout<<"DIFF "<<algo<<' '<<a[1]<<' '<<notation(p)<<'\n';
690 int wins = 0, loses = 0, ties = 0;
691 for (int i = 0;i<(int)ps.size();++i) {
692 array<Integer, 3> m = metrics(x.jobs[i]), n = metrics(y.jobs[i]);
693 cout<<ps[i].id<<" turnaround="<<m[0]<<'/'<<n[0]<<" waiting="<<m[1]<<'/'<<n[1]<<" response="<<m[2]<<'/'<<n[2]
694 <<'\n';
695 if (m[0]<n[0])++wins;
696 else if (m[0]>n[0])++loses;
697 else ++ties;
698 }
699 cout<<"WINS "<<algo<<'='<<wins<<' '<<a[1]<<'='<<loses<<" ties="<<ties<<'\n';
700 return;
701 }
702 if (c == "COMPARE" || c == "RANK") {
703 vector<Result> rs;
704 for (const string& m:algorithms)rs.push_back(Simulation(ps, m, p).run());
705 if (c == "COMPARE") {
706 cout<<"COMPARE "<<notation(p)<<'\n';
707 tableRow({
708 "ALGO","TURN","WAIT","RESP","UTIL","END","SWITCH","MISS"
709 });
710 for (int i = 0;i<9;++i) {
711 const Result& r = rs[i];
712 tableRow({
713 algorithms[i], average(r, 0), average(r, 1), average(r, 2), util(r), num(r.end), num(r.switches), num(r.missed)
714 });
715 }
716 cout<<"BEST";
717 for (int k = 0;k<3;++k) {
718 int best = 0;
719 for (int i = 1;i<9;++i)if (rs[i].totals[k]<rs[best].totals[k])best = i;
720 cout<<' '<<metricNames[k]<<'='<<algorithms[best];
721 }
722 cout<<'\n';
723 }
724 else {
725 cout<<"RANK "<<id<<' '<<notation(p)<<'\n';
726 int k = (int)(find(metricNames.begin(), metricNames.end(), id)-metricNames.begin());
727 auto value = [&](int i)->Integer {
728 return k<3?rs[i].totals[k]:id == "end"?rs[i].end:id == "switches"?rs[i].switches:rs[i].missed;
729 };
730 vector<int> order {
731 0, 1, 2, 3, 4, 5, 6, 7, 8
732 };
733 stable_sort(order.begin(), order.end(), [&](int a, int b) {
734 return value(a)<value(b);
735 });
736 for (int i:order) {
737 int rank = 1;
738 for (int j = 0;j<9;++j)if (value(j)<value(i))++rank;
739 cout<<rank<<' '<<algorithms[i]<<' '<<(k<3?truncated(value(i), ps.size()):num(value(i)))<<'\n';
740 }
741 }
742 return;
743 }
744 single(c, algo, a, p, ps);
745 }
