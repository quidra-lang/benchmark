53 return a < 0 || k == 0 || (k == 1 && (a == 3 || a == 7)) || (k == 2 && a == 4) ||
54 (k == 3 && a == 8) || (k == 4 && a == 7);

62static void notation(I *p, int a, char *s) {
63 s[0] = 0;
64 for (int k = 0; k < 5; k++)
65 if (uses(a, k)) {
66 char part[80];
67 sprintf(part, "%s%s=%lld", s[0] ? " " : "", keys[k], p[k]);
68 strcat(s, part);
69 }

72static int params(char **w, int n, int a, int preset, I *p) {
73 I defaults[] = {0, 4, 5, 0, 1};
74 int seen[5] = {0};
75 memcpy(p, defaults, sizeof defaults);
76 if (preset >= 0)
77 seen[preset] = 1;
78 for (int i = 0; i < n; i++) {
79 char *eq = strchr(w[i], '=');
80 *eq = 0;
81 int k = lookup(w[i], keys, 5);
82 I v = atoll(eq + 1);
83 if (k < 0) {
84 printf("ERROR unknown key %s\n", w[i]);
85 return 0;
86 }
87 if (!uses(a, k)) {
88 printf("ERROR key %s not allowed for %s\n", w[i], algos[a]);
89 return 0;
90 }
91 if (seen[k]) {
92 printf("ERROR duplicate key %s\n", w[i]);
93 return 0;
94 }
95 if (v < low[k] || v > high[k]) {
96 printf("ERROR invalid %s\n", w[i]);
97 return 0;
98 }
99 seen[k] = 1;
100 p[k] = v;
101 }
102 if (!count) {
103 puts("ERROR no processes");
104 return 0;
105 }
106 return 1;
107}

130static void event(Simulation *s, int t, int kind, int id, int value, int trace) {
131 static const char *names[] = {"PREEMPT", "EXPIRE", "DEMOTE", "BOOST", "AGE", "ARRIVE",
132 "IODONE", "START", "DONE", "IO", "SWITCH", "DISPATCH"};

145static void enqueue(Simulation *s, int i, int t) {
146 Job *j = &s->j[i];
147 j->state = 1;
148 j->enqueued = t;
149 Queue *q = &s->q[s->algo == 8 ? j->level : 0];
150 q->ids[q->count++] = i;

153static void start(Simulation *s, int i, int t) {
154 Job *j = &s->j[i];
155 s->cpu = 3;
156 s->current = i;
157 s->run = 0;
158 j->state = 3;
159 if (j->start < 0)
160 j->start = t;
161 if (j->began[j->burst / 2] < 0)
162 j->began[j->burst / 2] = t;

166 return s->p[i].deadline ? s->p[i].arrival + s->p[i].deadline : 1000000000;

169static int preempts(Simulation *s, int i) {
170 for (int k = 0; k < 3; k++)
171 for (int a = 0; a < s->q[k].count; a++) {
172 int b = s->q[k].ids[a];
173 if ((s->algo == 2 && s->j[b].left < s->j[i].left) ||
174 (s->algo == 4 && s->j[b].effective < s->j[i].effective) ||
175 (s->algo == 6 && deadline(s, b) < deadline(s, i)) ||
176 (s->algo == 8 && k < s->j[i].level))
177 return 1;
178 }
179 return 0;
180}

182static int choose(Simulation *s, Queue *q, int t) {
183 int selected = 0;
184 if (s->algo == 7) {
185 I total = 0;
186 for (int k = 0; k < q->count; k++)
187 total += 10 - s->p[q->ids[k]].priority;
188 I r = draw(&s->random, total);
189 for (int k = 0; k < q->count; k++) {
190 r -= 10 - s->p[q->ids[k]].priority;
191 if (r < 0)
192 return k;
193 }
194 }
195 for (int k = 1; k < q->count; k++) {
196 int x = q->ids[k], y = q->ids[selected];
197 Job *a = &s->j[x], *b = &s->j[y];
198 int better = 0;
199 if (s->algo == 1 || s->algo == 2)
200 better = a->left < b->left;
201 else if (s->algo == 4)
202 better = a->effective < b->effective;
203 else if (s->algo == 5)
204 better =
205 (I)(t - a->enqueued + a->left) * b->left > (I)(t - b->enqueued + b->left) * a->left;
206 else if (s->algo == 6)
207 better = deadline(s, x) < deadline(s, y);
208 if (better)
209 selected = k;
210 }
211 return selected;
212}

214static void capture(Simulation *s) {
215 char *out = s->snapshot;
216 int used = 0;
217 if (s->cpu == 0)
218 used += sprintf(out + used, "CPU IDLE\n");
219 else
220 used += sprintf(out + used, "CPU %s %s\n", states[s->cpu], s->p[s->current].id);
221 for (int k = 0; k < (s->algo == 8 ? 3 : 1); k++) {
222 used += sprintf(out + used, s->algo == 8 ? "QUEUE%d" : "QUEUE", k);
223 Queue *q = &s->q[k];
224 if (!q->count)
225 used += sprintf(out + used, " none");
226 for (int a = 0; a < q->count; a++)
227 used += sprintf(out + used, " %s", s->p[q->ids[a]].id);
228 used += sprintf(out + used, "\n");
229 }
230 for (int i = 0; i < s->count; i++) {
231 Job *j = &s->j[i];
232 used += sprintf(out + used, "%s %s", s->p[i].id, states[j->state]);
233 if (j->state == 0)
234 used += sprintf(out + used, " at=%d", s->p[i].arrival);
235 else if (j->state == 5)
236 used += sprintf(out + used, " at=%d", j->finish);
237 else {
238 used += sprintf(out + used, " burst=%d left=%d", j->burst + 1, j->left);
239 if (s->algo == 4)
240 used += sprintf(out + used, " prio=%d", j->effective);
241 if (s->algo == 8)
242 used += sprintf(out + used, " level=%d used=%d", j->level, j->used);
243 }
244 used += sprintf(out + used, "\n");
245 }
246}

248static Simulation *simulate(int algo, I *param, int snapshot, int trace) {
249 Simulation *s = calloc(1, sizeof *s);
250 s->count = count;
251 s->algo = algo;
252 s->current = s->previous = -1;
253 s->random = param[4];
254 memcpy(s->param, param, sizeof s->param);
255 memcpy(s->p, registry, (size_t)count * sizeof *registry);
256 qsort(s->p, (size_t)count, sizeof *s->p, process_compare);
257 for (int i = 0; i < count; i++) {
258 s->j[i].left = s->p[i].bursts[0];
259 s->j[i].effective = s->p[i].priority;
260 s->j[i].start = -1;
261 for (int b = 0; b < 8; b++)
262 s->j[i].began[b] = -1;
263 }
264 for (int t = 0;; t++) {
265 if (algo == 4)
266 for (int k = 0; k < s->q[0].count; k++) {
267 int i = s->q[0].ids[k];
268 Job *j = &s->j[i];
269 if (++j->wait == param[2]) {
270 j->wait = 0;
271 if (j->effective) {
272 j->effective--;
273 event(s, t, 4, i, j->effective, trace);
274 }
275 }
276 }
277 for (int i = 0; i < count; i++)
278 if (s->j[i].state == 4 && !s->j[i].left) {
279 Job *j = &s->j[i];
280 j->burst++;
281 j->left = s->p[i].bursts[j->burst];
282 j->ready[j->burst / 2] = t;
283 enqueue(s, i, t);
284 event(s, t, 6, i, -1, trace);
285 }
286 for (int i = 0; i < count; i++)
287 if (!s->j[i].state && s->p[i].arrival == t) {
288 s->j[i].ready[0] = t;
289 enqueue(s, i, t);
290 event(s, t, 5, i, -1, trace);
291 }
292 if (s->cpu == 2 && !s->switch_left) {
293 start(s, s->current, t);
294 event(s, t, 7, s->current, -1, trace);
295 } else if (s->cpu == 3) {
296 int i = s->current;
297 Job *j = &s->j[i];
298 if (!j->left && j->burst == s->p[i].count - 1) {
299 j->state = 5;
300 j->finish = t;
301 event(s, t, 8, i, -1, trace);
302 s->cpu = 0;
303 } else {
304 int demoted = algo == 8 && j->level < 2 && j->used == (j->level ? 8 : 4);
305 if (demoted) {
306 j->level++;
307 j->used = 0;
308 event(s, t, 2, i, j->level, trace);
309 }
310 if (!j->left) {
311 j->burst++;
312 j->left = s->p[i].bursts[j->burst];
313 j->state = 4;
314 event(s, t, 9, i, j->left, trace);
315 s->cpu = 0;
316 } else if (demoted) {
317 enqueue(s, i, t);
318 s->cpu = 0;
319 } else if ((algo == 3 || algo == 7) && s->run == param[1]) {
320 enqueue(s, i, t);
321 event(s, t, 1, i, -1, trace);
322 s->cpu = 0;
323 } else if (preempts(s, i)) {
324 enqueue(s, i, t);
325 event(s, t, 0, i, -1, trace);
326 s->cpu = 0;
327 }
328 }
329 }
330 int done = 0;
331 for (int i = 0; i < count; i++)
332 done += s->j[i].state == 5;
333 if (done == count) {
334 s->end = t;
335 if (!s->snapshot[0] && snapshot >= 0)
336 capture(s);
337 break;
338 }
339 if (algo == 8 && param[3] && t && t % param[3] == 0) {
340 for (int k = 1; k < 3; k++) {
341 for (int a = 0; a < s->q[k].count; a++)
342 s->q[0].ids[s->q[0].count++] = s->q[k].ids[a];
343 s->q[k].count = 0;
344 }
345 for (int i = 0; i < count; i++)
346 s->j[i].level = s->j[i].used = 0;
347 event(s, t, 3, -1, -1, trace);
348 }
349 if (!s->cpu) {
350 Queue *q = NULL;
351 for (int k = 0; k < 3; k++)
352 if (s->q[k].count) {
353 q = &s->q[k];
354 break;
355 }
356 if (q) {
357 int a = choose(s, q, t), i = q->ids[a];
358 for (int k = a + 1; k < q->count; k++)
359 q->ids[k - 1] = q->ids[k];
360 q->count--;
361 Job *j = &s->j[i];
362 j->effective = s->p[i].priority;
363 j->wait = 0;
364 j->slices++;
365 j->selections[j->burst / 2]++;
366 int changed = i != s->previous;
367 if (changed)
368 s->switches++;
369 if (changed && param[0]) {
370 s->cpu = 2;
371 s->current = i;
372 s->switch_left = (int)param[0];
373 j->state = 2;
374 event(s, t, 10, i, -1, trace);
375 } else {
376 start(s, i, t);
377 event(s, t, 11, i, -1, trace);
378 }
379 }
380 }
381 if (t == snapshot)
382 capture(s);
383 int qlen = s->q[0].count + s->q[1].count + s->q[2].count;
384 s->lengths[t] = qlen;
385 s->sumq += qlen;
386 if (qlen > s->maxq)
387 s->maxq = qlen;
388 for (int i = 0; i < count; i++)
389 s->timeline[i][t] = "-.s#i-"[s->j[i].state];
390 if (s->cpu == 3) {
391 Job *j = &s->j[s->current];
392 j->left--;
393 j->used++;
394 s->run++;
395 s->previous = s->current;
396 s->labels[t] = s->current;
397 s->busy++;
398 if (!j->left)
399 j->ended[j->burst / 2] = t + 1;
400 } else if (s->cpu == 2) {
401 s->switch_left--;
402 s->labels[t] = -2;
403 s->switching++;
404 } else {
405 s->labels[t] = -1;
406 s->idle++;
407 }
408 for (int i = 0; i < count; i++)
409 if (s->j[i].state == 4)
410 s->j[i].left--;
411 }
412 for (int i = 0; i < count; i++) {
413 Job *j = &s->j[i];
414 int turnaround = j->finish - s->p[i].arrival, total = 0;
415 for (int b = 0; b < s->p[i].count; b++)
416 total += s->p[i].bursts[b];
417 s->totals[0] += turnaround;
418 s->totals[1] += turnaround - total;
419 s->totals[2] += j->start - s->p[i].arrival;
420 if (s->p[i].deadline) {
421 if (j->finish > deadline(s, i))
422 s->missed++;
423 else
424 s->met++;
425 }
426 }
427 return s;
428}

430static I metric(Simulation *s, int i, int m) {
431 if (m == 0)
432 return s->j[i].finish - s->p[i].arrival;
433 if (m == 2)
434 return s->j[i].start - s->p[i].arrival;
435 I x = metric(s, i, 0);
436 for (int b = 0; b < s->p[i].count; b++)
437 x -= s->p[i].bursts[b];
438 return x;
439}

455static void simulation_output(const char *command, int algo, I *p, Simulation *s, int argument) {
456 char note[200], a[3][40], util[40];
457 notation(p, algo, note);
458 averages(s, a, util);
459 if (!strcmp(command, "TRACE")) {
460 printf("END %d\n", s->end);
461 return;
462 }
463 if (!strcmp(command, "TIMELINE")) {
464 if (s->end > 100) {
465 printf("ERROR too long %d\n", s->end);
466 return;
467 }
468 printf("TIMELINE %s %s end=%d\n", algos[algo], note, s->end);
469 for (int i = 0; i < count; i++)
470 printf("%-8s %s\n", s->p[i].id, s->timeline[i]);
471 return;
472 }
473 if (!strcmp(command, "SNAPSHOT")) {
474 printf("SNAPSHOT %s %s t=%d\n%s", algos[algo], note, argument, s->snapshot);
475 return;
476 }
477 if (!strcmp(command, "WINDOW")) {
478 printf("WINDOW %s %s width=%d\n", algos[algo], note, argument);
479 for (int start = 0; start < s->end; start += argument) {
480 int end = start + argument;
481 if (end > s->end)
482 end = s->end;
483 int busy = 0, cs = 0, idle = 0, hits[10] = {0};
484 for (int t = start; t < end; t++)
485 if (s->labels[t] == -1)
486 idle++;
487 else if (s->labels[t] == -2)
488 cs++;
489 else {
490 busy++;
491 hits[s->labels[t]]++;
492 }
493 int top = -1;
494 for (int i = 0; i < count; i++)
495 if (hits[i] && (top < 0 || hits[i] > hits[top]))
496 top = i;
497 char u[40];
498 decimal(busy * 100, end - start, u);
499 printf("%d-%d busy=%d switch=%d idle=%d util=%s top=%s\n", start, end, busy, cs, idle,
500 u, top < 0 ? "none" : s->p[top].id);
501 }
502 return;
503 }
504 printf("%s %s %s\n", command, algos[algo], note);
505 if (!strcmp(command, "BURSTS")) {
506 for (int i = 0; i < count; i++)
507 for (int b = 0; b < s->p[i].count; b += 2) {
508 Job *j = &s->j[i];
509 int k = b / 2, len = s->p[i].bursts[b];
510 printf("%s #%d len=%d ready=%d start=%d end=%d wait=%d slices=%d\n", s->p[i].id,
511 k + 1, len, j->ready[k], j->began[k], j->ended[k],
512 j->ended[k] - j->ready[k] - len, j->selections[k]);
513 }
514 return;
515 }
516 if (!strcmp(command, "STATS")) {
517 for (int m = 0; m < 3; m++) {
518 I values[10];
519 int lo = 0, hi = 0;
520 for (int i = 0; i < count; i++) {
521 values[i] = metric(s, i, m);
522 if (values[i] < metric(s, lo, m))
523 lo = i;
524 if (values[i] > metric(s, hi, m))
525 hi = i;
526 }
527 for (int i = 1; i < count; i++) {
528 I v = values[i];
529 int j = i;
530 while (j && values[j - 1] > v) {
531 values[j] = values[j - 1];
532 j--;
533 }
534 values[j] = v;
535 }
536 I twice = count % 2 ? 2 * values[count / 2] : values[count / 2 - 1] + values[count / 2];
537 printf("%s min=%lld (%s) max=%lld (%s) median=%lld.%d total=%lld\n", metrics[m],
538 metric(s, lo, m), s->p[lo].id, metric(s, hi, m), s->p[hi].id, twice / 2,
539 twice % 2 ? 5 : 0, s->totals[m]);
540 }
541 return;
542 }
543 int start = 0, segments = 0;
544 for (int t = 1; t <= s->end; t++)
545 if (t == s->end || strcmp(label(s, t), label(s, start))) {
546 if (segments % 8 == 0)
547 printf("GANTT ");
548 printf("[%d-%d %s]", start, t, label(s, start));
549 segments++;
550 if (segments % 8 == 0 || t == s->end)
551 puts("");
552 start = t;
553 }
554 for (int i = 0; i < count; i++) {
555 printf("%s arrival=%d done=%d turnaround=%lld waiting=%lld response=%lld slices=%d",
556 s->p[i].id, s->p[i].arrival, s->j[i].finish, metric(s, i, 0), metric(s, i, 1),
557 metric(s, i, 2), s->j[i].slices);
558 if (s->p[i].deadline) {
559 int d = deadline(s, i), late = s->j[i].finish - d;
560 printf(" deadline=%d late=%d", d, late > 0 ? late : 0);
561 }
562 puts("");
563 }
564 printf("AVERAGE turnaround=%s waiting=%s response=%s\n", a[0], a[1], a[2]);
565 printf("CPU end=%d busy=%d switch=%d idle=%d switches=%d util=%s\n", s->end, s->busy,
566 s->switching, s->idle, s->switches, util);
567 char avgq[40];
568 decimal(s->sumq, s->end, avgq);
569 printf("QUEUE max=%d avg=%s\n", s->maxq, avgq);
570 printf("EVENTS preempt=%d expire=%d demote=%d boost=%d age=%d\n", s->events[0], s->events[1],
571 s->events[2], s->events[3], s->events[4]);
572 printf("DEADLINE met=%d missed=%d\n", s->met, s->missed);
573}

575static void command(char **w, int n) {
576 char *c = w[0];
577 if (!strcmp(c, "PROC")) {
578 int at = find(w[1]);
579 if (at >= 0) {
580 printf("ERROR duplicate process %s\n", w[1]);
581 return;
582 }
583 if (count == 10) {
584 puts("ERROR too many processes");
585 return;
586 }
587 if (atoll(w[2]) > 200) {
588 printf("ERROR invalid arrival %s\n", w[1]);
589 return;
590 }
591 if (atoll(w[3]) > 9) {
592 printf("ERROR invalid priority %s\n", w[1]);
593 return;
594 }
595 int bursts = n - 4, valid = bursts % 2;
596 for (int i = 0; i < bursts; i++)
597 if (w[i + 4][0] != (i % 2 ? 'I' : 'C') || atoll(w[i + 4] + 1) < 1 ||
598 atoll(w[i + 4] + 1) > 30)
599 valid = 0;
600 if (!valid) {
601 printf("ERROR invalid bursts %s\n", w[1]);
602 return;
603 }
604 Process p = {0};
605 strcpy(p.id, w[1]);
606 p.arrival = atoi(w[2]);
607 p.priority = atoi(w[3]);
608 p.count = bursts;
609 int cpu = 0, io = 0;
610 for (int i = 0; i < bursts; i++) {
611 p.bursts[i] = atoi(w[i + 4] + 1);
612 if (i % 2)
613 io += p.bursts[i];
614 else
615 cpu += p.bursts[i];
616 }
617 registry[count++] = p;
618 printf("PROC %s cpu=%d io=%d\n", p.id, cpu, io);
619 return;
620 }
621 if (!strcmp(c, "GEN")) {
622 int amount = atoi(w[2]);
623 for (int i = 1; i <= amount; i++) {
624 char id[9];
625 sprintf(id, "%s%d", w[1], i);
626 if (find(id) >= 0) {
627 printf("ERROR duplicate process %s\n", id);
628 return;
629 }
630 }
631 if (count + amount > 10) {
632 puts("ERROR too many processes");
633 return;
634 }
635 I random = atoll(w[3]);
636 printf("GEN count=%d\n", amount);
637 for (int i = 1; i <= amount; i++) {
638 Process p = {0};
639 sprintf(p.id, "%s%d", w[1], i);
640 p.arrival = (int)draw(&random, 21);
641 p.priority = (int)draw(&random, 10);
642 int m = 1 + (int)draw(&random, 3);
643 for (int b = 0; b < m; b++) {
644 if (b)
645 p.bursts[p.count++] = 1 + (int)draw(&random, 8);
646 p.bursts[p.count++] = 1 + (int)draw(&random, 9);
647 }
648 registry[count++] = p;
649 describe(&p);
650 }
651 return;
652 }
653 if (!strcmp(c, "SETPRIO") || !strcmp(c, "DEADLINE") || !strcmp(c, "DROP")) {
654 int i = find(w[1]);
655 if (i < 0) {
656 printf("ERROR unknown process %s\n", w[1]);
657 return;
658 }
659 if (!strcmp(c, "DROP")) {
660 for (int j = i + 1; j < count; j++)
661 registry[j - 1] = registry[j];
662 count--;
663 printf("DROP %s\n", w[1]);
664 return;
665 }
666 int v = atoi(w[2]), isprio = !strcmp(c, "SETPRIO");
667 if (v > (isprio ? 9 : 1000)) {
668 printf("ERROR invalid %s %s\n", isprio ? "priority" : "deadline", w[1]);
669 return;
670 }
671 if (isprio) {
672 registry[i].priority = v;
673 printf("SETPRIO %s priority=%d\n", w[1], v);
674 } else {
675 registry[i].deadline = v;
676 if (v)
677 printf("DEADLINE %s %d\n", w[1], v);
678 else
679 printf("DEADLINE %s none\n", w[1]);
680 }
681 return;
682 }
683 if (!strcmp(c, "CLEAR")) {
684 printf("CLEAR count=%d\n", count);
685 count = 0;
686 return;
687 }
688 if (!strcmp(c, "LIST")) {
689 printf("LIST count=%d\n", count);
690 Process sorted[10];
691 memcpy(sorted, registry, (size_t)count * sizeof *registry);
692 qsort(sorted, (size_t)count, sizeof *sorted, process_compare);
693 for (int i = 0; i < count; i++)
694 describe(&sorted[i]);
695 return;
696 }
697 int all = !strcmp(c, "COMPARE") || !strcmp(c, "RANK"), diff = !strcmp(c, "DIFF"), a = -1,
698 b = -1, m = -1, offset = 2, key = -1, arg = -1;
699 I lo = 0, hi = 0, p[5];
700 if (!all) {
701 a = lookup(w[1], algos, 9);
702 if (a < 0) {
703 printf("ERROR unknown algorithm %s\n", w[1]);
704 return;
705 }
706 if (diff) {
707 b = lookup(w[2], algos, 9);
708 offset = 3;
709 if (b < 0) {
710 printf("ERROR unknown algorithm %s\n", w[2]);
711 return;
712 }
713 }
714 } else if (!strcmp(c, "RANK")) {
715 m = lookup(w[1], metrics, 6);
716 if (m < 0) {
717 printf("ERROR unknown metric %s\n", w[1]);
718 return;
719 }
720 } else
721 offset = 1;
722 if (!strcmp(c, "WINDOW")) {
723 I width = atoll(w[2]);
724 if (width < 1 || width > 1000) {
725 puts("ERROR invalid window");
726 return;
727 }
728 arg = (int)width;
729 offset = 3;
730 }
731 if (!strcmp(c, "SNAPSHOT")) {
732 arg = atoi(w[2]);
733 offset = 3;
734 }
735 if (!strcmp(c, "SWEEP")) {
736 key = lookup(w[2], keys, 5);
737 if (key < 0) {
738 printf("ERROR unknown key %s\n", w[2]);
739 return;
740 }
741 if (!uses(a, key)) {
742 printf("ERROR key %s not allowed for %s\n", w[2], algos[a]);
743 return;
744 }
745 lo = atoll(w[3]);
746 hi = atoll(w[4]);
747 if (lo > hi || lo < low[key] || hi > high[key] || hi - lo > 20) {
748 puts("ERROR invalid range");
749 return;
750 }
751 offset = 5;
752 }
753 if (!params(w + offset, n - offset, all || diff ? -1 : a, key, p))
754 return;
755 char note[200];
756 notation(p, all || diff ? -1 : a, note);
757 if (key >= 0) {
758 printf("SWEEP %s %s %lld %lld\n", algos[a], keys[key], lo, hi);
759 I best = -1, bestv = lo;
760 for (I v = lo; v <= hi; v++) {
761 p[key] = v;
762 Simulation *s = simulate(a, p, -1, 0);
763 char av[3][40], u[40], pn[200];
764 averages(s, av, u);
765 notation(p, a, pn);
766 printf("%s turnaround=%s waiting=%s response=%s util=%s end=%d switches=%d\n", pn,
767 av[0], av[1], av[2], u, s->end, s->switches);
768 if (best < 0 || s->totals[1] < best) {
769 best = s->totals[1];
770 bestv = v;
771 }
772 free(s);
773 }
774 printf("BEST %s=%lld\n", keys[key], bestv);
775 return;
776 }
777 if (diff) {
778 Simulation *one = simulate(a, p, -1, 0), *two = simulate(b, p, -1, 0);
779 printf("DIFF %s %s %s\n", algos[a], algos[b], note);
780 int wins = 0, losses = 0, ties = 0;
781 for (int i = 0; i < count; i++) {
782 printf("%s turnaround=%lld/%lld waiting=%lld/%lld response=%lld/%lld\n", one->p[i].id,
783 metric(one, i, 0), metric(two, i, 0), metric(one, i, 1), metric(two, i, 1),
784 metric(one, i, 2), metric(two, i, 2));
785 I x = metric(one, i, 0), y = metric(two, i, 0);
786 wins += x < y;
787 losses += x > y;
788 ties += x == y;
789 }
790 printf("WINS %s=%d %s=%d ties=%d\n", algos[a], wins, algos[b], losses, ties);
791 free(one);
792 free(two);
793 return;
794 }
795 if (all) {
796 I totals[9][3], values[9];
797 char texts[9][40];
798 if (m < 0) {
799 printf("COMPARE %s\n", note);
800 printf("%-8s %8s %8s %8s %8s %8s %8s %8s\n", "ALGO", "TURN", "WAIT", "RESP", "UTIL",
801 "END", "SWITCH", "MISS");
802 } else
803 printf("RANK %s %s\n", metrics[m], note);
804 for (int algo = 0; algo < 9; algo++) {
805 Simulation *s = simulate(algo, p, -1, 0);
806 memcpy(totals[algo], s->totals, sizeof s->totals);
807 if (m < 0) {
808 char av[3][40], u[40];
809 averages(s, av, u);
810 printf("%-8s %8s %8s %8s %8s %8d %8d %8d\n", algos[algo], av[0], av[1], av[2], u,
811 s->end, s->switches, s->missed);
812 } else {
813 values[algo] = rank_value(s, m);
814 if (m < 3)
815 decimal(values[algo], count, texts[algo]);
816 else
817 sprintf(texts[algo], "%lld", values[algo]);
818 }
819 free(s);
820 }
821 if (m < 0) {
822 int best[3] = {0};
823 for (int k = 0; k < 3; k++)
824 for (int algo = 1; algo < 9; algo++)
825 if (totals[algo][k] < totals[best[k]][k])
826 best[k] = algo;
827 printf("BEST turnaround=%s waiting=%s response=%s\n", algos[best[0]], algos[best[1]],
828 algos[best[2]]);
829 } else {
830 int ids[9];
831 for (int i = 0; i < 9; i++) {
832 int j = i;
833 while (j && values[ids[j - 1]] > values[i]) {
834 ids[j] = ids[j - 1];
835 j--;
836 }
837 ids[j] = i;
838 }
839 for (int k = 0; k < 9; k++) {
840 int i = ids[k], rank = 1;
841 for (int j = 0; j < 9; j++)
842 rank += values[j] < values[i];
843 printf("%d %s %s\n", rank, algos[i], texts[i]);
844 }
845 }
846 return;
847 }
848 int trace = !strcmp(c, "TRACE");
849 if (trace)
850 printf("TRACE %s %s\n", algos[a], note);
851 Simulation *s = simulate(a, p, !strcmp(c, "SNAPSHOT") ? arg : -1, trace);
852 simulation_output(c, a, p, s, arg);
853 free(s);
854}
855static char *read_line(FILE *input) {
856 size_t length = 0, capacity = 128;
857 char *line = malloc(capacity);
858 int character;
859 while ((character = fgetc(input)) != EOF && character != '\n') {
860 if (length + 1 == capacity) {
861 capacity *= 2;
862 line = realloc(line, capacity);
863 }
864 line[length++] = (char)character;
865 }
866 if (!length && character == EOF) {
867 free(line);
868 return NULL;
869 }
870 line[length] = 0;
871 return line;
872}

874int main(void) {
875 char *line = read_line(stdin);
876 if (!line)
877 return 0;
878 int n = atoi(line);
879 free(line);
880 size_t capacity = 32;
881 char **words = malloc(capacity * sizeof *words);
882 for (int i = 0; i < n; i++) {
883 line = read_line(stdin);
884 if (!line)
885 break;
886 int nw = 0;
887 for (char *s = strtok(line, " \r\n"); s; s = strtok(NULL, " \r\n")) {
888 if ((size_t)nw == capacity) {
889 capacity *= 2;
890 words = realloc(words, capacity * sizeof *words);
891 }
892 words[nw++] = s;
893 }
894 command(words, nw);
895 free(line);
896 }
897 free(words);
898 return 0;
