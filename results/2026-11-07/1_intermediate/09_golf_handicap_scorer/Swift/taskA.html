8func away(_ n: Int, _ d: Int) -> Int { n < 0 ? -((2*(-n)+d)/(2*d)) : (2*n+d)/(2*d) }
9func up(_ n: Int, _ d: Int) -> Int {
10 let a = 2*n+d
11 return a/(2*d)-(a%(2*d) < 0 ? 1 : 0)
12}
13func signed(_ n: Int) -> String { (n < 0 ? "-" : "")+"\(abs(n)/10).\(abs(n)%10)" }
14func index(_ n: Int?) -> String {
15 guard let n else { return "none" }
16 return n < 0 ? "+"+signed(-n) : signed(n)
17}
18func handicap(_ n: Int?) -> String {

120func allocate(_ c: Course, _ h: Int) -> [Int] {
121 (0..<18).map { i in
122 h >= 0 ? h/18+(c.si[i] <= h%18 ? 1 : 0) : -((-h)/18)-(c.si[i] >= 19-(-h)%18 ? 1 : 0)

125func courseHandicap(_ c: Course, _ t: Tee, _ i: Int) -> Int { up(i*t.slope+113*(t.rating-10*c.total),1130) }

145 p.records.filter { $0.day >= day-365 && $0.day < day }.min {

149 func eligible(_ p: Player) -> Eligible {
150 let start = max(0,p.rounds.count-20)
151 let n = p.rounds.count-start
152 var reduction = 0
153 var adjusted = Array(repeating:0,count:n)
154 if n > 0 {
155 for i in stride(from:n-1,through:0,by:-1) {
156 let r = p.rounds[start+i]
157 reduction += r.exceptional
158 adjusted[i] = r.diff-reduction
159 }
160 }
161 let k: Int
162 switch n {
163 case 0...2: k = 0
164 case 3...5: k = 1
165 case 6...8: k = 2
166 case 9...11: k = 3
167 case 12...14: k = 4
168 case 15...16: k = 5
169 case 17...18: k = 6
170 case 19: k = 7
171 default: k = 8
172 }
173 let adjustment = n == 3 ? -20 : n == 4 || n == 6 ? -10 : 0
174 let counting = Array((0..<n).sorted { adjusted[$0] == adjusted[$1] ? $0 > $1 : adjusted[$0] < adjusted[$1] }.prefix(k))
175 return Eligible(start:start,adjusted:adjusted,counting:counting,k:k,adjustment:adjustment)
176 }
177 func update(_ p: Player, _ day: Int) {
178 let e = eligible(p)
179 if e.k == 0 {
180 p.calculation = nil
181 return
182 }
183 let a = away(e.counting.reduce(0) { $0+e.adjusted[$1] },e.k)
184 var c = a+e.adjustment
185 let l = low(p,day)?.value
186 var soft = false
187 var hard = false
188 var maximum = false
189 if let l, c-l > 30 {
190 let next = l+30+(c-l-30)/2
191 soft = next != c
192 c = next
193 }
194 if let l, c-l > 50 {
195 c = l+50
196 hard = true
197 }
198 if c > 540 {
199 c = 540
200 maximum = true
201 }
202 p.index = c
203 p.calculation = Calculation(k:e.k,average:a,adjustment:e.adjustment,low:l,hard:hard,soft:soft,maximum:maximum)
204 }
205 func data(_ e: Event) throws -> EventData {
206 var rounds: [Round] = []
207 let n = e.players.count
208 var ph = Array(repeating:0,count:n)
209 for i in 0..<n {
210 guard let r = e.players[i].rounds.first(where:{ $0.course === e.course && $0.tee === e.tee && $0.day == e.day }) else {
211 throw Fault.reason("missing round")
212 }
213 guard let h = r.handicap else { throw Fault.reason("no index") }
214 rounds.append(r)
215 ph[i] = up(h*e.allowance,100)
216 }
217 let minimum = ph.min()!
218 let relative = ["MATCH","NASSAU","FOURBALL"].contains(e.format)
219 let strokes = ph.map { $0-(relative ? minimum : 0) }
220 var nets = Array(repeating:Array(repeating:0,count:18),count:n)
221 var points = nets
222 var par = nets
223 for i in 0..<n {
224 let allocation = allocate(e.course,strokes[i])
225 for h in 0..<18 {
226 nets[i][h] = rounds[i].scores[h] == noNet ? noNet : rounds[i].scores[h]-allocation[h]
227 points[i][h] = nets[i][h] == noNet ? 0 : max(0,e.course.par[h]+2-nets[i][h])
228 par[i][h] = nets[i][h] == noNet || nets[i][h] > e.course.par[h] ? -1 : nets[i][h] < e.course.par[h] ? 1 : 0
229 }
230 }
231 return EventData(e:e,rounds:rounds,ph:ph,strokes:strokes,nets:nets,points:points,par:par)
232 }
233 func rank(_ values: [[Int]], _ candidates: [Int], _ high: Bool, _ ids: [String]) -> [Int] {
234 candidates.map { i in
235 (index:i,key:[0,9,12,15,17].map { (high ? -1 : 1)*sum(values[i],$0) })
236 }.sorted { a,b in
237 a.key == b.key ? ids[a.index] < ids[b.index] : a.key.lexicographicallyPrecedes(b.key)
238 }.map { $0.index }
239 }
240 func ranked(_ d: EventData) -> [Int] {
241 let candidates = d.ph.indices.filter { d.e.format != "STROKE" || !d.rounds[$0].scores.contains(noNet) }
242 let values = d.e.format == "STROKE" ? d.nets : d.e.format == "PAR" ? d.par : d.points
243 return rank(values,candidates,d.e.format != "STROKE",d.e.players.map { $0.id })
244 }
245 func status(_ lead: Int) -> String { lead == 0 ? "AS" : lead > 0 ? "\(lead) UP" : "\(-lead) DN" }
246 func match(_ wins: [Int], _ first: Int, _ last: Int, _ a: String, _ b: String) -> Match {
247 var lead = 0
248 var remaining = 0
249 var statuses: [String] = []
250 for h in first...last {
251 lead += wins[h]
252 remaining = last-h
253 statuses.append(status(lead))
254 if abs(lead) > remaining { break }
255 }
256 let result = lead == 0 ? "AS" : (lead > 0 ? a : b)+" \(abs(lead))"+(remaining > 0 ? "&\(remaining)" : " UP")
257 return Match(statuses:statuses,result:result)
258 }
259 func results(_ e: Event, _ d: EventData) {
260 output.append("results \(e.id): \(e.format) \(e.course.id) \(e.tee.id) day \(e.day) allowance \(e.allowance)")
261 let n = e.players.count
262 let ids = e.players.map { $0.id }
263 switch e.format {
264 case "STROKE", "STABLEFORD", "PAR":
265 let order = ranked(d)
266 for (r,i) in order.enumerated() {
267 let value: String
268 if e.format == "STROKE" { value = "net \(sum(d.nets[i])) gross \(sum(d.rounds[i].scores))" }
269 else if e.format == "STABLEFORD" { value = "points \(sum(d.points[i]))" }
270 else { value = status(sum(d.par[i])) }
271 output.append("\(r+1) \(ids[i]) \(value) hcp \(handicap(d.ph[i]))")
272 }
273 if e.format == "STROKE" {
274 for i in ids.indices.filter({ !order.contains($0) }).sorted(by:{ ids[$0] < ids[$1] }) {
275 output.append("NR \(ids[i]) hcp \(handicap(d.ph[i]))")
276 }
277 if !order.isEmpty {
278 let gross = d.rounds.map { $0.scores }
279 let winner = rank(gross,order,false,ids)[0]
280 output.append("gross \(ids[winner]) \(sum(gross[winner]))")
281 }
282 }
283 case "MATCH", "NASSAU", "FOURBALL":
284 output.append("strokes"+ids.indices.map { " \(ids[$0]) \(d.strokes[$0])" }.joined())
285 let size = n/2
286 let a = ids.prefix(size).joined(separator:"+")
287 let b = ids.dropFirst(size).joined(separator:"+")
288 let wins = (0..<18).map { h -> Int in
289 let left = (0..<size).map { d.nets[$0][h] }.min()!
290 let right = (size..<n).map { d.nets[$0][h] }.min()!
291 return right < left ? -1 : right > left ? 1 : 0
292 }
293 if e.format == "NASSAU" {
294 output.append("front "+match(wins,0,8,a,b).result)
295 output.append("back "+match(wins,9,17,a,b).result)
296 output.append("total "+match(wins,0,17,a,b).result)
297 } else {
298 let m = match(wins,0,17,a,b)
299 output.append("status "+m.statuses.joined(separator:", "))
300 output.append("result "+m.result)
301 }
302 case "BETTERBALL":
303 let values = (0..<n/2).map { i in (0..<18).map { h in max(d.points[2*i][h],d.points[2*i+1][h]) } }
304 let order = rank(values,Array(0..<n/2),true,(0..<n/2).map { ids[2*$0] })
305 for (r,i) in order.enumerated() {
306 output.append("\(r+1) \(ids[2*i])+\(ids[2*i+1]) points \(sum(values[i])) hcp \(handicap(d.ph[2*i])) \(handicap(d.ph[2*i+1]))")
307 }
308 case "SKINS":
309 var carry = 0
310 var totals = Array(repeating:0,count:n)
311 var tied = Array(ids.indices)
312 for h in 0..<18 {
313 carry += 1
314 let minimum = ids.indices.map { d.nets[$0][h] }.min()!
315 tied = ids.indices.filter { d.nets[$0][h] == minimum }
316 if minimum != noNet && tied.count == 1 {
317 let i = tied[0]
318 output.append("hole \(h+1) \(ids[i]) \(carry)")
319 totals[i] += carry*100
320 carry = 0
321 }
322 }
323 if carry > 0 {
324 tied.sort { ids[$0] < ids[$1] }
325 for i in tied { totals[i] += carry*100/tied.count }
326 output.append("carry \(carry)"+tied.map { " \(ids[$0])" }.joined())
327 }
328 let order = ids.indices.sorted { totals[$0] == totals[$1] ? ids[$0] < ids[$1] : totals[$0] > totals[$1] }
329 for i in order { output.append("\(ids[i]) \(cents(totals[i]))") }
330 default: break
331 }
332 }
333 func cardTotal(_ label: String, _ r: Round, _ strokes: [Int], _ nets: [Int], _ points: [Int], _ from: Int, _ to: Int) {
334 let complete = !r.scores[from..<to].contains(noNet)
335 let gross = complete ? String(sum(r.scores,from,to)) : "-"
336 let net = complete ? String(sum(nets,from,to)) : "-"
337 output.append("\(label) \(sum(r.course.par,from,to)) \(gross) \(sum(r.adjusted,from,to)) \(sum(strokes,from,to)) \(net) \(sum(points,from,to))")
338 }
339 func card(_ p: Player, _ number: Int, _ r: Round) {
340 output.append("card \(p.id) \(number): \(r.course.id) \(r.tee.id) day \(r.day) handicap \(handicap(r.handicap))")
341 let strokes = allocate(r.course,r.handicap ?? 0)
342 let nets = (0..<18).map { r.scores[$0] == noNet ? noNet : r.scores[$0]-strokes[$0] }
343 let points = (0..<18).map { nets[$0] == noNet ? 0 : max(0,r.course.par[$0]+2-nets[$0]) }
344 for h in 0..<18 {
345 let net = nets[h] == noNet ? "-" : String(nets[h])
346 output.append("\(h+1) \(r.course.par[h]) \(r.course.si[h]) \(r.raw[h]) \(r.adjusted[h]) \(strokes[h]) \(net) \(points[h])")
347 if h == 8 { cardTotal("OUT",r,strokes,nets,points,0,9) }
348 if h == 17 {
349 cardTotal("IN",r,strokes,nets,points,9,18)
350 cardTotal("TOTAL",r,strokes,nets,points,0,18)
351 }
352 }
353 output.append("ags \(r.ags) diff \(signed(r.diff))")
354 }
355 func statistics(_ p: Player) {
356 var category = Array(repeating:0,count:7)
357 var totals = Array(repeating:0,count:3)
358 var counts = Array(repeating:0,count:3)
359 var complete: [Int] = []
360 for r in p.rounds {
361 var full = true
362 for h in 0..<18 {
363 if r.raw[h] == "X" { category[6] += 1 }
364 if r.scores[h] == noNet {
365 full = false
366 continue
367 }
368 let relative = r.scores[h]-r.course.par[h]
369 let k = relative <= -2 ? 0 : relative <= 2 ? relative+2 : 5
370 category[k] += 1
371 let i = r.course.par[h]-3
372 totals[i] += r.scores[h]
373 counts[i] += 1
374 }
375 if full { complete.append(sum(r.scores)) }
376 }
377 let best = complete.min().map(String.init) ?? "-"
378 var line = "stats \(p.id): rounds \(p.rounds.count) best \(best) average \(average(complete.reduce(0,+),complete.count))"
379 let labels = ["eagles","birdies","pars","bogeys","doubles","others","pickups"]
380 for i in 0..<7 { line += " \(labels[i]) \(category[i])" }
381 for i in 0..<3 { line += " par\(i+3) \(average(totals[i],counts[i]))" }
382 output.append(line)
383 }
384 func execute(_ t: [String]) throws {
385 switch t[0] {
386 case "COURSE":
387 try error(courses[t[1]] != nil,"duplicate course")
388 let par = (0..<18).map { Int(t[$0+2])! }
389 let si = (0..<18).map { Int(t[$0+20])! }
390 try error(par.contains { $0 < 3 || $0 > 5 } || Set(si).count != 18 || si.contains { $0 < 1 || $0 > 18 },"invalid course")
391 courses[t[1]] = Course(t[1],par,si)
392 case "TEE":
393 let c = try course(t[1])
394 try error(c.tees[t[2]] != nil,"duplicate tee")
395 let rating = Int(t[3])!
396 let slope = Int(t[4])!
397 try error(rating < 500 || rating > 900 || slope < 55 || slope > 155,"invalid tee")
398 c.tees[t[2]] = Tee(t[2],rating,slope)
399 case "PLAYER":
400 try error(players[t[1]] != nil,"duplicate player")
401 let i: Int? = t.count == 4 ? Int(t[3])! : nil
402 try error(i != nil && (i! < -100 || i! > 540),"invalid index")
403 players[t[1]] = Player(t[1],t[2],i)
404 case "ROUND":
405 let p = try player(t[1])
406 let c = try course(t[2])
407 let tee = try tee(c,t[3])
408 let day = Int(t[4])!
409 try error(!p.rounds.isEmpty && day < p.rounds.last!.day,"invalid day")
410 let raw = Array(t.dropFirst(5))
411 try error(raw.filter { $0 == "-" }.count >= 5,"too few holes")
412 let previous = p.index
413 let hcp = previous.map { courseHandicap(c,tee,$0) }
414 let strokes = allocate(c,hcp ?? 0)
415 let scores = raw.map { Int($0) ?? noNet }
416 let adjusted = (0..<18).map { h -> Int in
417 let limit = c.par[h]+(previous == nil ? 5 : 2+strokes[h])
418 if raw[h] == "-" { return c.par[h]+strokes[h] }
419 if raw[h] == "X" { return limit }
420 return min(scores[h],limit)
421 }
422 let ags = adjusted.reduce(0,+)
423 let diff = away(113*(10*ags-tee.rating),tee.slope)
424 let exceptional = previous != nil && previous!-diff >= 70 ? (previous!-diff >= 100 ? 20 : 10) : 0
425 let r = Round(course:c,tee:tee,day:day,raw:raw,handicap:hcp,scores:scores,adjusted:adjusted,ags:ags,diff:diff,
426 exceptional:exceptional,oldIndex:previous,oldCalculation:p.calculation,oldRecords:p.records.count)
427 p.rounds.append(r)
428 update(p,day)
429 if let i = p.index { p.records.append(IndexRecord(day:day,value:i)) }
430 output.append("round \(p.id) \(p.rounds.count): ags \(ags) diff \(signed(diff))"+(exceptional == 0 ? "" : " exceptional \(signed(-exceptional))")+" index \(index(p.index))")
431 case "REMOVE":
432 let p = try player(t[1])
433 try error(p.rounds.isEmpty,"no rounds")
434 let number = p.rounds.count
435 let r = p.rounds.removeLast()
436 p.index = r.oldIndex
437 p.calculation = r.oldCalculation
438 p.records.removeSubrange(r.oldRecords..<p.records.count)
439 output.append("removed \(p.id) \(number): index \(index(p.index))")
440 case "INDEX":
441 let p = try player(t[1])
442 var line = "index \(p.id): \(index(p.index)) rounds \(p.rounds.count)"
443 if let c = p.calculation {
444 line += " lowest \(c.k) average \(signed(c.average)) adjustment \(signed(c.adjustment))"+
445 (c.low == nil ? "" : " low \(index(c.low))")+(c.hard ? " hard cap" : c.soft ? " soft cap" : "")+(c.maximum ? " maximum" : "")
446 }
447 output.append(line)
448 case "HISTORY":
449 let p = try player(t[1])
450 let e = eligible(p)
451 output.append("history \(p.id): rounds \(p.rounds.count)")
452 for i in e.adjusted.indices {
453 let r = p.rounds[e.start+i]
454 output.append("\(e.start+i+1) day \(r.day) \(r.course.id) \(r.tee.id) ags \(r.ags) diff \(signed(r.diff))"+
455 (r.diff == e.adjusted[i] ? "" : " adjusted \(signed(e.adjusted[i]))")+(e.counting.contains(i) ? " *" : ""))
456 }
457 case "LOWINDEX":
458 let p = try player(t[1])
459 let day = Int(t[2])!
460 let r = low(p,day)
461 output.append("lowindex \(p.id) \(day): "+(r == nil ? "none" : "\(index(r!.value)) day \(r!.day)"))
462 case "COURSEHCP":
463 let p = try player(t[1])
464 let c = try course(t[2])
465 let tee = try tee(c,t[3])
466 guard let i = p.index else { throw Fault.reason("no index") }
467 let h = courseHandicap(c,tee,i)
468 output.append("coursehcp \(p.id) \(c.id) \(tee.id): \(handicap(h)) strokes "+allocate(c,h).map(String.init).joined(separator:" "))
469 case "HCPTABLE":
470 let c = try course(t[1])
471 let tee = try tee(c,t[2])
472 let first = Int(t[3])!
473 let last = Int(t[4])!
474 try error(first < -100 || first > last || last > 540,"invalid index")
475 output.append("hcptable \(c.id) \(tee.id): \(index(first)) to \(index(last))")
476 var start = first
477 var h = courseHandicap(c,tee,first)
478 for i in (first+1)...(last+1) {
479 let next = i <= last ? courseHandicap(c,tee,i) : h+1
480 if next != h {
481 output.append("\(index(start)) to \(index(i-1)): \(handicap(h))")
482 start = i
483 h = next
484 }
485 }
486 case "CARD":
487 let p = try player(t[1])
488 let n = Int(t[2])!
489 try error(n > p.rounds.count,"unknown round")
490 card(p,n,p.rounds[n-1])
491 case "STATS": try statistics(player(t[1]))
492 case "HOLESTATS":
493 let c = try course(t[1])
494 let rounds = players.values.flatMap { $0.rounds }.filter { $0.course === c }
495 var totals = Array(repeating:0,count:18)
496 var counts = Array(repeating:0,count:18)
497 var ranks = Array(repeating:0,count:18)
498 for r in rounds {
499 for h in 0..<18 where r.scores[h] != noNet {
500 totals[h] += r.scores[h]
501 counts[h] += 1
502 }
503 }
504 let order = (0..<18).sorted { a,b in
505 if counts[a] == 0 || counts[b] == 0 {
506 return counts[a] == counts[b] ? a < b : counts[a] != 0
507 }
508 let left = (totals[a]-c.par[a]*counts[a])*counts[b]
509 let right = (totals[b]-c.par[b]*counts[b])*counts[a]
510 return left == right ? a < b : left > right
511 }
512 for (i,h) in order.enumerated() { ranks[h] = i+1 }
513 output.append("holestats \(c.id): rounds \(rounds.count)")
514 for h in 0..<18 { output.append("\(h+1) par \(c.par[h]) si \(c.si[h]) scores \(counts[h]) avg \(average(totals[h],counts[h])) rank \(ranks[h])") }
515 case "PLAYERS":
516 let order = players.values.sorted { a,b in
517 if a.index == nil || b.index == nil { return a.index == b.index ? a.id < b.id : a.index != nil }
518 return a.index == b.index ? a.id < b.id : a.index! < b.index!
519 }
520 output.append("players \(order.count)")
521 for p in order { output.append("\(p.id) \(p.name) \(index(p.index)) rounds \(p.rounds.count)") }
522 case "EVENT":
523 try error(events[t[1]] != nil,"duplicate event")
524 let c = try course(t[3])
525 let tee = try tee(c,t[4])
526 let allowance = Int(t[6])!
527 try error(allowance < 1 || allowance > 100,"invalid allowance")
528 let list = try t.dropFirst(7).map { try player($0) }
529 let n = list.count
530 let valid: Bool
531 switch t[2] {
532 case "MATCH", "NASSAU": valid = n == 2
533 case "FOURBALL": valid = n == 4
534 case "BETTERBALL": valid = [4,6,8].contains(n)
535 default: valid = n >= 2 && n <= 8
536 }
537 try error(Set(list.map { $0.id }).count != n || !valid,"invalid players")
538 let e = Event(id:t[1],format:t[2],course:c,tee:tee,day:Int(t[5])!,allowance:allowance,players:list)
539 events[t[1]] = e
540 eventOrder.append(e)
541 case "RESULTS":
542 guard let e = events[t[1]] else { throw Fault.reason("unknown event") }
543 try results(e,data(e))
544 case "MERIT":
545 var merit: [String:[Int]] = [:]
546 let awards = [10,8,6,5,4,3,2,1]
547 var count = 0
548 for e in eventOrder where ["STROKE","STABLEFORD","PAR"].contains(e.format) {
549 guard let d = try? data(e) else { continue }
550 count += 1
551 for p in e.players { merit[p.id,default:[0,0,0]][2] += 1 }
552 for (i,p) in ranked(d).enumerated() {
553 let id = e.players[p].id
554 merit[id]![0] += awards[i]
555 if i == 0 { merit[id]![1] += 1 }
556 }
557 }
558 let order = merit.keys.sorted { a,b in
559 if merit[a]![0] != merit[b]![0] { return merit[a]![0] > merit[b]![0] }
560 if merit[a]![1] != merit[b]![1] { return merit[a]![1] > merit[b]![1] }
561 return a < b
562 }
563 output.append("merit: events \(count)")
564 for (i,id) in order.enumerated() {
565 let a = merit[id]!
566 output.append("\(i+1) \(id) \(a[0]) wins \(a[1]) played \(a[2])")
567 }
568 default: break
569 }
570 }

574 catch Fault.reason(let reason) { output.append("ERROR "+reason) }
575 catch { fatalError("Unexpected error") }
576 return output
577 }
