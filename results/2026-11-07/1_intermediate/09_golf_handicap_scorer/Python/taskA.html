9def round_half_up(numerator, denominator):
10 return (2 * numerator + denominator) // (2 * denominator)

13def round_half_away(numerator, denominator):
14 if numerator < 0:
15 return -round_half_up(-numerator, denominator)
16 return round_half_up(numerator, denominator)

40def allocation(handicap, indexes):
41 quotient, remainder = divmod(abs(handicap), 18)
42 if handicap >= 0:
43 return [quotient + int(index <= remainder) for index in indexes]
44 return [-quotient - int(index >= 19 - remainder) for index in indexes]

48 return sum(values), sum(values[9:]), sum(values[12:]), sum(values[15:]), values[17]

51def ranked(entries, values, larger=False):
52 def key(identifier):
53 counts = countback(values[identifier])

55 counts = tuple(-value for value in counts)

95 def low_index(self, day):
96 records = [(index, date) for date, index in self.records if day - 365 <= date < day]
97 return min(records) if records else None

99 def eligible(self):
100 start = max(0, len(self.rounds) - 20)
101 reduction = 0
102 eligible = []
103 for position in range(len(self.rounds) - 1, start - 1, -1):
104 record = self.rounds[position]
105 reduction += record.exceptional
106 eligible.append((position + 1, record, record.differential - reduction))
107 return eligible[::-1]
108
109 def counting(self, eligible):
110 count = len(eligible)
111 if count < 3:
112 return [], 0
113 lowest = next(
114 k for k, limit in enumerate((5, 8, 11, 14, 16, 18, 19, 20), 1)
115 if count <= limit
116 )
117 adjustment = -20 if count == 3 else -10 if count in (4, 6) else 0
118 return sorted(eligible, key=lambda row: (row[2], -row[0]))[:lowest], adjustment
119
120 def update_index(self, day):
121 selected, adjustment = self.counting(self.eligible())
122 if not selected:
123 self.calculation = None
124 return
125 average = round_half_away(sum(row[2] for row in selected), len(selected))
126 candidate = average + adjustment
127 low_record = self.low_index(day)
128 low = low_record[0] if low_record is not None else None
129 soft = False
130 hard = False
131 maximum = False
132 if low is not None:
133 if candidate - low > 30:
134 candidate = low + 30 + (candidate - low - 30) // 2
135 soft = True
136 if candidate - low > 50:
137 candidate = low + 50
138 hard = True
139 if candidate > 540:
140 candidate = 540
141 maximum = True
142 self.index = candidate
143 self.calculation = (len(selected), average, adjustment, low, hard, soft, maximum)

146class Round:
147 def __init__(self, player, course, tee, day, scores):
148 self.course = course
149 self.tee = tee
150 self.day = day
151 self.scores = scores
152 self.previous_index = player.index
153 self.previous_records = len(player.records)
154 self.previous_calculation = player.calculation
155 self.handicap = tee.handicap(player.index, course) if player.index is not None else None
156 strokes = allocation(self.handicap or 0, course.indexes)
157 self.adjusted = []
158 for par, score, stroke in zip(course.pars, scores, strokes):
159 limit = par + 5 if self.handicap is None else par + 2 + stroke
160 if score == "-":
161 adjusted = par + stroke
162 elif score == "X":
163 adjusted = limit
164 else:
165 adjusted = min(score, limit)
166 self.adjusted.append(adjusted)
167 self.ags = sum(self.adjusted)
168 self.differential = round_half_away(113 * (10 * self.ags - tee.rating), tee.slope)
169 self.exceptional = 0
170 if player.index is not None and player.index - self.differential >= 70:
171 self.exceptional = 20 if player.index - self.differential >= 100 else 10
172
173 def nets(self, strokes):
174 return [score - stroke if isinstance(score, int) else None for score, stroke in zip(self.scores, strokes)]
175
176 def points(self, nets):
177 return [max(0, par + 2 - net) if net is not None else 0 for par, net in zip(self.course.pars, nets)]
178
179 def complete(self):
180 return all(isinstance(score, int) for score in self.scores)

407 def stats(self, arguments):
408 player = self.player(arguments[0])
409 labels = ("eagles", "birdies", "pars", "bogeys", "doubles", "others")
410 categories = dict.fromkeys(labels + ("pickups",), 0)
411 par_scores = {par: [] for par in (3, 4, 5)}
412 complete_totals = []
413 for record in player.rounds:
414 if record.complete():
415 complete_totals.append(sum(record.scores))
416 for par, score in zip(record.course.pars, record.scores):
417 if score == "X":
418 categories["pickups"] += 1
419 elif isinstance(score, int):
420 relative = score - par

422 categories[category] += 1
423 par_scores[par].append(score)
424 parts = [
425 "stats", player.identifier + ":", "rounds", len(player.rounds),
426 "best", min(complete_totals) if complete_totals else "-",
427 "average", average_display(sum(complete_totals), len(complete_totals)),
428 ]
429 for category, count in categories.items():
430 parts.extend((category, count))
431 for par, scores in par_scores.items():
432 parts.extend(("par" + str(par), average_display(sum(scores), len(scores))))

485 def event_data(self, event):
486 rounds = event.rounds()
487 handicaps = {
488 identifier: round_half_up(record.handicap * event.allowance, 100)
489 for identifier, record in rounds.items()
490 }
491 minimum = min(handicaps.values()) if event.form in ("MATCH", "NASSAU", "FOURBALL") else 0
492 nets = {
493 identifier: record.nets(allocation(handicaps[identifier] - minimum, event.course.indexes))
494 for identifier, record in rounds.items()
495 }
496 points = {identifier: record.points(nets[identifier]) for identifier, record in rounds.items()}
497 return rounds, handicaps, nets, points

499 def event_ranking(self, event, rounds, nets, points):
500 identifiers = [player.identifier for player in event.players]
501 if event.form == "STROKE":
502 complete = [identifier for identifier in identifiers if rounds[identifier].complete()]
503 return ranked(complete, nets), nets
504 if event.form == "STABLEFORD":
505 return ranked(identifiers, points, True), points
506 values = {
507 identifier: [
508 -1 if net is None or net > par else 1 if net < par else 0
509 for net, par in zip(nets[identifier], event.course.pars)
510 ]
511 for identifier in identifiers
512 }
513 return ranked(identifiers, values, True), values

515 def individual_results(self, event, rounds, handicaps, nets, points):
516 order, values = self.event_ranking(event, rounds, nets, points)

521 elif event.form == "STABLEFORD":
522 self.emit(rank, identifier, "points", sum(points[identifier]), "hcp", handicap)
523 else:
524 self.emit(rank, identifier, match_status(sum(values[identifier])), "hcp", handicap)
525 if event.form == "STROKE":
526 nonreturners = sorted(player.identifier for player in event.players if not rounds[player.identifier].complete())
527 for identifier in nonreturners:
528 self.emit("NR", identifier, "hcp", handicap_display(handicaps[identifier]))
529 if order:
530 gross = {identifier: rounds[identifier].scores for identifier in order}
531 winner = ranked(order, gross)[0]
532 self.emit("gross", winner, sum(gross[winner]))

534 def match_holes(self, event, nets):
535 identifiers = [player.identifier for player in event.players]
536 if event.form == "FOURBALL":
537 sides = (identifiers[:2], identifiers[2:])
538 else:
539 sides = (identifiers[:1], identifiers[1:])
540 names = tuple("+".join(side) for side in sides)
541 holes = []
542 for hole in range(18):
543 side_scores = []
544 for side in sides:
545 scores = [nets[player][hole] for player in side if nets[player][hole] is not None]
546 side_scores.append(min(scores) if scores else None)
547 first, second = side_scores
548 if first is None:
549 winner = 0 if second is None else -1
550 elif second is None or first < second:
551 winner = 1
552 else:
553 winner = -1 if first > second else 0
554 holes.append(winner)
555 return names, holes

557 def play_match(self, names, holes, first, last):
558 lead = 0
559 statuses = []
560 for hole in range(first, last):
561 lead += holes[hole]
562 statuses.append(match_status(lead))
563 remaining = last - hole - 1
564 if abs(lead) > remaining or remaining == 0:
565 break
566 if lead == 0:
567 result = "AS"
568 else:
569 name = names[0] if lead > 0 else names[1]
570 margin = f"{abs(lead)}&{remaining}" if remaining else str(abs(lead)) + " UP"
571 result = name + " " + margin
572 return statuses, result

574 def match_results(self, event, handicaps, nets):
575 minimum = min(handicaps.values())
576 parts = ["strokes"]
577 for player in event.players:
578 parts.extend((player.identifier, handicaps[player.identifier] - minimum))
579 self.emit(*parts)
580 names, holes = self.match_holes(event, nets)
581 if event.form == "NASSAU":
582 for label, first, last in (("front", 0, 9), ("back", 9, 18), ("total", 0, 18)):
583 _, result = self.play_match(names, holes, first, last)
584 self.emit(label, result)
585 else:
586 statuses, result = self.play_match(names, holes, 0, 18)
587 self.emit("status", ", ".join(statuses))
588 self.emit("result", result)

604 def skins_results(self, event, nets):
605 identifiers = [player.identifier for player in event.players]
606 totals = dict.fromkeys(identifiers, 0)
607 carry = 0
608 last_minimum = []
609 for hole in range(18):
610 carry += 1
611 available = [player for player in identifiers if nets[player][hole] is not None]
612 if available:
613 best = min(nets[player][hole] for player in available)
614 winners = [player for player in available if nets[player][hole] == best]
615 else:
616 winners = []
617 last_minimum = winners
618 if len(winners) == 1:
619 self.emit("hole", hole + 1, winners[0], carry)
620 totals[winners[0]] += 100 * carry
621 carry = 0
622 if carry:
623 sharing = sorted(last_minimum or identifiers)
624 share = carry * 100 // len(sharing)
625 for identifier in sharing:
626 totals[identifier] += share
627 self.emit("carry", carry, *sharing)
628 for identifier in sorted(identifiers, key=lambda player: (-totals[player], player)):
629 value = totals[identifier]
630 self.emit(identifier, f"{value // 100}.{value % 100:02d}")

651 def merit(self, arguments):
652 players = {}
653 counted = 0
654 awards = (10, 8, 6, 5, 4, 3, 2, 1)
655 for event in self.events.values():
656 if event.form not in ("STROKE", "STABLEFORD", "PAR"):
657 continue
658 try:
659 rounds, _, nets, points = self.event_data(event)
660 except CommandError:
661 continue
662 counted += 1
663 order, _ = self.event_ranking(event, rounds, nets, points)
664 for player in event.players:
665 players.setdefault(player.identifier, [0, 0, 0])[2] += 1
666 for rank, identifier in enumerate(order):
667 players[identifier][0] += awards[rank]
668 players[identifier][1] += int(rank == 0)
669 self.emit("merit:", "events", counted)
670 order = sorted(players, key=lambda player: (-players[player][0], -players[player][1], player))
671 for rank, identifier in enumerate(order, 1):
672 points, wins, appearances = players[identifier]
673 self.emit(rank, identifier, points, "wins", wins, "played", appearances)

677 handlers = {
