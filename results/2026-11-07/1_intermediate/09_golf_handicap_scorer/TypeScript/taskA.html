3type Score = number | string;
4type Course = {id: string; pars: number[]; si: number[]; par: number; tees: Map<string, Tee>};
5type Tee = {id: string; rating: number; slope: number};
6type Calculation = {k: number; average: number; adjustment: number; low: number | null; hard: boolean; soft: boolean; maximum: boolean};
7type Eligible = {number: number; round: GolfRound; adjusted: number};
8type Event = {id: string; form: string; course: Course; tee: Tee; day: number; allowance: number; players: Player[]};
9type EventData = {rounds: Record<string, GolfRound>; hcp: Record<string, number>; nets: Record<string, (number | null)[]>; points: Record<string, number[]>};
10class CommandError extends Error {}
11const sum = (v: number[]): number => v.reduce((a, b) => a + b, 0);
12const lex = (a: string, b: string): number => a < b ? -1 : a > b ? 1 : 0;
13const halfUp = (n: number, d: number): number => Math.floor((2 * n + d) / (2 * d));
14const halfAway = (n: number, d: number): number => n < 0 ? -halfUp(-n, d) : halfUp(n, d);
15const decimal = (n: number, sign = "-"): string => (n < 0 ? sign : "") + Math.floor(Math.abs(n) / 10) + "." + Math.abs(n) % 10;
16const indexText = (n: number | null): string => n === null ? "none" : decimal(n, "+");
17const hcpText = (n: number | null): string => n === null ? "none" : n < 0 ? "+" + -n : String(n);
18const hundredths = (n: number): string => Math.floor(n / 100) + "." + String(n % 100).padStart(2, "0");
19const averageText = (total: number, count: number): string => count ? hundredths(Math.floor(total * 100 / count)) : "-";
20const status = (n: number): string => n === 0 ? "AS" : Math.abs(n) + (n > 0 ? " UP" : " DN");

22function allocation(h: number, si: number[]): number[] {
23 const q = Math.floor(Math.abs(h) / 18);
24 const r = Math.abs(h) % 18;
25 return si.map(i => h >= 0 ? q + Number(i <= r) : -q - Number(i >= 19 - r));
26}
27
28function handicap(index: number, c: Course, t: Tee): number {
29 return halfUp(index * t.slope + 113 * (t.rating - 10 * c.par), 1130);
30}
31
32function ranked(ids: string[], values: Record<string, number[]>, larger = false): string[] {
33 const keys: Record<string, number[]> = {};
34 for (const id of ids) {
35 const v = values[id];
36 keys[id] = [sum(v), sum(v.slice(9)), sum(v.slice(12)), sum(v.slice(15)), v[17]];
37 }
38 return [...ids].sort((a, b) => {
39 for (let i = 0; i < 5; i++) {
40 const difference = keys[a][i] - keys[b][i];
41 if (difference) return larger ? -difference : difference;
42 }
43 return lex(a, b);
44 });
45}

47class Player {
48 id: string;
49 name: string;
50 index: number | null;
51 rounds: GolfRound[] = [];
52 records: number[][] = [];
53 calculation: Calculation | null = null;
54
55 constructor(id: string, name: string, index: number | null) {
56 this.id = id;
57 this.name = name;
58 this.index = index;
59 }
60
61 low(day: number): number[] | null {
62 const records = this.records.filter(([d]) => d >= day - 365 && d < day);
63 records.sort((a, b) => a[1] - b[1] || a[0] - b[0]);
64 return records.length ? records[0] : null;
65 }
66
67 eligible(): Eligible[] {
68 const rows: Eligible[] = [];
69 let reduction = 0;
70 for (let i = this.rounds.length - 1; i >= Math.max(0, this.rounds.length - 20); i--) {
71 const round = this.rounds[i];
72 reduction += round.exceptional;
73 rows.push({number: i + 1, round, adjusted: round.diff - reduction});
74 }
75 return rows.reverse();
76 }
77
78 counting(rows: Eligible[]): {selected: Eligible[]; adjustment: number} {
79 const n = rows.length;
80 if (n < 3) return {selected: [], adjustment: 0};
81 const k = [5, 8, 11, 14, 16, 18, 19, 20].findIndex(limit => n <= limit) + 1;
82 const adjustment = n === 3 ? -20 : n === 4 || n === 6 ? -10 : 0;
83 const selected = [...rows].sort((a, b) => a.adjusted - b.adjusted || b.number - a.number).slice(0, k);
84 return {selected, adjustment};
85 }
86
87 update(day: number): void {
88 const {selected, adjustment} = this.counting(this.eligible());
89 if (!selected.length) {
90 this.calculation = null;
91 return;
92 }
93 const average = halfAway(sum(selected.map(r => r.adjusted)), selected.length);
94 let candidate = average + adjustment;
95 const record = this.low(day);
96 const low = record === null ? null : record[1];
97 let soft = false;
98 let hard = false;
99 let maximum = false;
100 if (low !== null) {
101 if (candidate - low > 30) {
102 candidate = low + 30 + Math.floor((candidate - low - 30) / 2);
103 soft = true;
104 }
105 if (candidate - low > 50) {
106 candidate = low + 50;
107 hard = true;
108 }
109 }
110 if (candidate > 540) {
111 candidate = 540;
112 maximum = true;
113 }
114 this.index = candidate;
115 this.calculation = {k: selected.length, average, adjustment, low, hard, soft, maximum};
116 }
117}

119class GolfRound {
120 course: Course;
121 tee: Tee;
122 day: number;
123 scores: Score[];
124 previousIndex: number | null;
125 previousRecords: number;
126 previousCalculation: Calculation | null;
127 hcp: number | null;
128 adjusted: number[];
129 ags: number;
130 diff: number;
131 exceptional: number;
132
133 constructor(p: Player, c: Course, t: Tee, day: number, scores: Score[]) {
134 this.course = c;
135 this.tee = t;
136 this.day = day;
137 this.scores = scores;
138 this.previousIndex = p.index;
139 this.previousRecords = p.records.length;
140 this.previousCalculation = p.calculation;
141 this.hcp = p.index === null ? null : handicap(p.index, c, t);
142 const strokes = allocation(this.hcp ?? 0, c.si);
143 this.adjusted = scores.map((score, h) => {
144 const limit = this.hcp === null ? c.pars[h] + 5 : c.pars[h] + 2 + strokes[h];
145 return score === "-" ? c.pars[h] + strokes[h] : score === "X" ? limit : Math.min(score as number, limit);
146 });
147 this.ags = sum(this.adjusted);
148 this.diff = halfAway(113 * (10 * this.ags - t.rating), t.slope);
149 this.exceptional = p.index !== null && p.index - this.diff >= 70 ? p.index - this.diff >= 100 ? 20 : 10 : 0;
150 }
151
152 nets(strokes: number[]): (number | null)[] {
153 return this.scores.map((s, h) => typeof s === "number" ? s - strokes[h] : null);
154 }
155
156 points(nets: (number | null)[]): number[] {
157 return nets.map((n, h) => n === null ? 0 : Math.max(0, this.course.pars[h] + 2 - n));
158 }
159
160 complete(): boolean {
161 return this.scores.every(s => typeof s === "number");
162 }
163}

194 eventData(e: Event): EventData {
195 const rounds: Record<string, GolfRound> = {};
196 const hcp: Record<string, number> = {};
197 const nets: Record<string, (number | null)[]> = {};
198 const points: Record<string, number[]> = {};
199 for (const p of e.players) {
200 const r = p.rounds.find(r => r.course === e.course && r.tee === e.tee && r.day === e.day);
201 if (!r) throw new CommandError("missing round");
202 if (r.hcp === null) throw new CommandError("no index");
203 rounds[p.id] = r;
204 hcp[p.id] = halfUp(r.hcp * e.allowance, 100);
205 }
206 const minimum = ["MATCH", "NASSAU", "FOURBALL"].includes(e.form) ? Math.min(...Object.values(hcp)) : 0;
207 for (const p of e.players) {
208 nets[p.id] = rounds[p.id].nets(allocation(hcp[p.id] - minimum, e.course.si));
209 points[p.id] = rounds[p.id].points(nets[p.id]);
210 }
211 return {rounds, hcp, nets, points};
212 }

214 eventRanking(e: Event, d: EventData): {order: string[]; values: Record<string, number[]>} {
215 const ids = e.players.map(p => p.id);
216 if (e.form === "STROKE") {
217 const complete = ids.filter(id => d.rounds[id].complete());
218 const values: Record<string, number[]> = {};
219 for (const id of complete) values[id] = d.nets[id] as number[];
220 return {order: ranked(complete, values), values};
221 }
222 if (e.form === "STABLEFORD") return {order: ranked(ids, d.points, true), values: d.points};
223 const values: Record<string, number[]> = {};
224 for (const id of ids) values[id] = d.nets[id].map((n, h) => n === null || n > e.course.pars[h] ? -1 : n < e.course.pars[h] ? 1 : 0);
225 return {order: ranked(ids, values, true), values};
226 }

228 individual(e: Event, d: EventData): void {
229 const {order, values} = this.eventRanking(e, d);
230 for (let rank = 0; rank < order.length; rank++) {
231 const id = order[rank];
232 if (e.form === "STROKE") this.emit(rank + 1, id, "net", sum(values[id]), "gross", sum(d.rounds[id].scores as number[]), "hcp", hcpText(d.hcp[id]));
233 else if (e.form === "STABLEFORD") this.emit(rank + 1, id, "points", sum(values[id]), "hcp", hcpText(d.hcp[id]));
234 else this.emit(rank + 1, id, status(sum(values[id])), "hcp", hcpText(d.hcp[id]));
235 }
236 if (e.form === "STROKE") {
237 for (const id of e.players.map(p => p.id).filter(id => !d.rounds[id].complete()).sort()) this.emit("NR", id, "hcp", hcpText(d.hcp[id]));
238 if (order.length) {
239 const gross: Record<string, number[]> = {};
240 for (const id of order) gross[id] = d.rounds[id].scores as number[];
241 const winner = ranked(order, gross)[0];
242 this.emit("gross", winner, sum(gross[winner]));
243 }
244 }

247 playMatch(names: string[], holes: number[], first: number, last: number): {statuses: string[]; result: string} {
248 let lead = 0;
249 let remaining = last - first;
250 const statuses: string[] = [];
251 for (let h = first; h < last; h++) {
252 lead += holes[h];
253 statuses.push(status(lead));
254 remaining = last - h - 1;
255 if (Math.abs(lead) > remaining || remaining === 0) break;
256 }
257 const result = lead === 0 ? "AS" : names[lead > 0 ? 0 : 1] + " " + Math.abs(lead) + (remaining ? "&" + remaining : " UP");
258 return {statuses, result};
259 }

261 matches(e: Event, d: EventData): void {
262 const ids = e.players.map(p => p.id);
263 const sides = e.form === "FOURBALL" ? [ids.slice(0, 2), ids.slice(2)] : [[ids[0]], [ids[1]]];
264 const names = sides.map(side => side.join("+"));
265 const holes: number[] = [];
266 for (let h = 0; h < 18; h++) {
267 const scores = sides.map(side => {
268 const values = side.map(id => d.nets[id][h]).filter((n): n is number => n !== null);
269 return values.length ? Math.min(...values) : null;
270 });
271 const [a, b] = scores;
272 holes.push(a === null ? b === null ? 0 : -1 : b === null || a < b ? 1 : a > b ? -1 : 0);
273 }
274 const minimum = Math.min(...Object.values(d.hcp));
275 const parts: (string | number)[] = ["strokes"];
276 for (const id of ids) parts.push(id, d.hcp[id] - minimum);
277 this.emit(...parts);
278 if (e.form === "NASSAU") {
279 for (const [label, first, last] of [["front", 0, 9], ["back", 9, 18], ["total", 0, 18]] as [string, number, number][]) {
280 this.emit(label, this.playMatch(names, holes, first, last).result);
281 }
282 } else {
283 const match = this.playMatch(names, holes, 0, 18);
284 this.emit("status", match.statuses.join(", "));
285 this.emit("result", match.result);
286 }
287 }

289 betterball(e: Event, d: EventData): void {
290 const teams: Record<string, string> = {};
291 const values: Record<string, number[]> = {};
292 for (let i = 0; i < e.players.length; i += 2) {
293 const a = e.players[i].id;
294 const b = e.players[i + 1].id;
295 teams[a] = b;
296 values[a] = d.points[a].map((v, h) => Math.max(v, d.points[b][h]));
297 }
298 const order = ranked(Object.keys(teams), values, true);
299 for (let i = 0; i < order.length; i++) {
300 const a = order[i];
301 const b = teams[a];
302 this.emit(i + 1, a + "+" + b, "points", sum(values[a]), "hcp", hcpText(d.hcp[a]), hcpText(d.hcp[b]));
303 }
304 }

306 skins(e: Event, d: EventData): void {
307 const ids = e.players.map(p => p.id);
308 const totals: Record<string, number> = Object.fromEntries(ids.map(id => [id, 0]));
309 let carry = 0;
310 let winners: string[] = [];
311 for (let h = 0; h < 18; h++) {
312 carry++;
313 const available = ids.filter(id => d.nets[id][h] !== null);
314 const best = available.length ? Math.min(...available.map(id => d.nets[id][h]!)) : null;
315 winners = available.filter(id => d.nets[id][h] === best);
316 if (winners.length === 1) {
317 this.emit("hole", h + 1, winners[0], carry);
318 totals[winners[0]] += 100 * carry;
319 carry = 0;
320 }
321 }
322 if (carry) {
323 const sharing = (winners.length ? winners : ids).toSorted();
324 const share = Math.floor(carry * 100 / sharing.length);
325 for (const id of sharing) totals[id] += share;
326 this.emit("carry", carry, ...sharing);
327 }
328 for (const id of [...ids].sort((a, b) => totals[b] - totals[a] || lex(a, b))) this.emit(id, hundredths(totals[id]));
329 }

331 cardTotal(label: string, r: GolfRound, strokes: number[], nets: (number | null)[], points: number[], first: number, last: number): void {
332 const scores = r.scores.slice(first, last);
333 const complete = scores.every(s => typeof s === "number");
334 this.emit(label, sum(r.course.pars.slice(first, last)), complete ? sum(scores as number[]) : "-",
335 sum(r.adjusted.slice(first, last)), sum(strokes.slice(first, last)),
336 complete ? sum(nets.slice(first, last) as number[]) : "-", sum(points.slice(first, last)));
337 }
338
339 command(cmd: string, a: string[]): void {
340 switch (cmd) {
341 case "COURSE": {
342 if (this.courses.has(a[0])) throw new CommandError("duplicate course");
343 const pars = a.slice(1, 19).map(Number);
344 const si = a.slice(19).map(Number);
345 if (pars.some(p => ![3, 4, 5].includes(p)) || [...si].sort((a, b) => a - b).some((v, i) => v !== i + 1)) throw new CommandError("invalid course");
346 this.courses.set(a[0], {id: a[0], pars, si, par: sum(pars), tees: new Map()});
347 break;
348 }
349 case "TEE": {
350 const c = this.course(a[0]);
351 if (c.tees.has(a[1])) throw new CommandError("duplicate tee");
352 const rating = Number(a[2]);
353 const slope = Number(a[3]);
354 if (rating < 500 || rating > 900 || slope < 55 || slope > 155) throw new CommandError("invalid tee");
355 c.tees.set(a[1], {id: a[1], rating, slope});
356 break;
357 }
358 case "PLAYER": {
359 if (this.players.has(a[0])) throw new CommandError("duplicate player");
360 const index = a.length === 3 ? Number(a[2]) : null;
361 if (index !== null && (index < -100 || index > 540)) throw new CommandError("invalid index");
362 this.players.set(a[0], new Player(a[0], a[1], index));
363 break;
364 }
365 case "ROUND": {
366 const p = this.player(a[0]);
367 const [c, t] = this.tee(a[1], a[2]);
368 const day = Number(a[3]);
369 if (p.rounds.length && day < p.rounds[p.rounds.length - 1].day) throw new CommandError("invalid day");
370 const scores = a.slice(4).map(s => s === "-" || s === "X" ? s : Number(s));
371 if (scores.filter(s => s === "-").length >= 5) throw new CommandError("too few holes");
372 const r = new GolfRound(p, c, t, day, scores);
373 p.rounds.push(r);
374 p.update(day);
375 if (p.index !== null) p.records.push([day, p.index]);
376 const parts: (string | number)[] = ["round", p.id, p.rounds.length + ":", "ags", r.ags, "diff", decimal(r.diff)];
377 if (r.exceptional) parts.push("exceptional", decimal(-r.exceptional));
378 this.emit(...parts, "index", indexText(p.index));
379 break;
380 }
381 case "REMOVE": {
382 const p = this.player(a[0]);
383 if (!p.rounds.length) throw new CommandError("no rounds");
384 const number = p.rounds.length;
385 const r = p.rounds.pop()!;
386 p.index = r.previousIndex;
387 p.records.length = r.previousRecords;
388 p.calculation = r.previousCalculation;
389 this.emit("removed", p.id, number + ":", "index", indexText(p.index));
390 break;
391 }
392 case "INDEX": {
393 const p = this.player(a[0]);
394 const parts: (string | number)[] = ["index", p.id + ":", indexText(p.index), "rounds", p.rounds.length];
395 const c = p.calculation;
396 if (c) {
397 parts.push("lowest", c.k, "average", decimal(c.average), "adjustment", decimal(c.adjustment));
398 if (c.low !== null) parts.push("low", indexText(c.low));
399 if (c.hard) parts.push("hard", "cap");
400 else if (c.soft) parts.push("soft", "cap");
401 if (c.maximum) parts.push("maximum");
402 }
403 this.emit(...parts);
404 break;
405 }
406 case "HISTORY": {
407 const p = this.player(a[0]);
408 const rows = p.eligible();
409 const counted = new Set(p.counting(rows).selected.map(r => r.number));
410 this.emit("history", p.id + ":", "rounds", p.rounds.length);
411 for (const row of rows) {
412 const r = row.round;
413 const parts: (string | number)[] = [row.number, "day", r.day, r.course.id, r.tee.id, "ags", r.ags, "diff", decimal(r.diff)];
414 if (row.adjusted !== r.diff) parts.push("adjusted", decimal(row.adjusted));
415 if (counted.has(row.number)) parts.push("*");
416 this.emit(...parts);
417 }
418 break;
419 }
420 case "LOWINDEX": {
421 const p = this.player(a[0]);
422 const day = Number(a[1]);
423 const low = p.low(day);
424 this.emit("lowindex", p.id, day + ":", ...(low === null ? ["none"] : [indexText(low[1]), "day", low[0]]));
425 break;
426 }
427 case "COURSEHCP": {
428 const p = this.player(a[0]);
429 const [c, t] = this.tee(a[1], a[2]);
430 if (p.index === null) throw new CommandError("no index");
431 const h = handicap(p.index, c, t);
432 this.emit("coursehcp", p.id, c.id, t.id + ":", hcpText(h), "strokes", ...allocation(h, c.si));
433 break;
434 }
435 case "HCPTABLE": {
436 const [c, t] = this.tee(a[0], a[1]);
437 const first = Number(a[2]);
438 const last = Number(a[3]);
439 if (first < -100 || first > last || last > 540) throw new CommandError("invalid index");
440 this.emit("hcptable", c.id, t.id + ":", indexText(first), "to", indexText(last));
441 let start = first;
442 let h = handicap(first, c, t);
443 for (let i = first + 1; i <= last; i++) {
444 const next = handicap(i, c, t);
445 if (next !== h) {
446 this.emit(indexText(start), "to", indexText(i - 1) + ":", hcpText(h));
447 start = i;
448 h = next;
449 }
450 }
451 this.emit(indexText(start), "to", indexText(last) + ":", hcpText(h));
452 break;
453 }
454 case "CARD": {
455 const p = this.player(a[0]);
456 const number = Number(a[1]);
457 if (number > p.rounds.length) throw new CommandError("unknown round");
458 const r = p.rounds[number - 1];
459 const c = r.course;
460 const strokes = allocation(r.hcp ?? 0, c.si);
461 const nets = r.nets(strokes);
462 const points = r.points(nets);
463 this.emit("card", p.id, number + ":", c.id, r.tee.id, "day", r.day, "handicap", hcpText(r.hcp));
464 for (let h = 0; h < 18; h++) {
465 this.emit(h + 1, c.pars[h], c.si[h], r.scores[h], r.adjusted[h], strokes[h], nets[h] ?? "-", points[h]);
466 if (h === 8) this.cardTotal("OUT", r, strokes, nets, points, 0, 9);
467 if (h === 17) {
468 this.cardTotal("IN", r, strokes, nets, points, 9, 18);
469 this.cardTotal("TOTAL", r, strokes, nets, points, 0, 18);
470 }
471 }
472 this.emit("ags", r.ags, "diff", decimal(r.diff));
473 break;
474 }
475 case "STATS": {
476 const p = this.player(a[0]);
477 const labels = ["eagles", "birdies", "pars", "bogeys", "doubles", "others", "pickups"];
478 const counts = Array(7).fill(0) as number[];
479 const parScores: Record<number, number[]> = {3: [], 4: [], 5: []};
480 const totals: number[] = [];
481 for (const r of p.rounds) {
482 if (r.complete()) totals.push(sum(r.scores as number[]));
483 for (let h = 0; h < 18; h++) {
484 const score = r.scores[h];
485 if (score === "X") counts[6]++;
486 else if (typeof score === "number") {
487 const relative = score - r.course.pars[h];
488 counts[Math.max(0, Math.min(relative + 2, 5))]++;
489 parScores[r.course.pars[h]].push(score);
490 }
491 }
492 }
493 const parts: (string | number)[] = ["stats", p.id + ":", "rounds", p.rounds.length, "best", totals.length ? Math.min(...totals) : "-", "average", averageText(sum(totals), totals.length)];
494 for (let i = 0; i < labels.length; i++) parts.push(labels[i], counts[i]);
495 for (const par of [3, 4, 5]) parts.push("par" + par, averageText(sum(parScores[par]), parScores[par].length));
496 this.emit(...parts);
497 break;
498 }
499 case "HOLESTATS": {
500 const c = this.course(a[0]);
501 const rounds = [...this.players.values()].flatMap(p => p.rounds).filter(r => r.course === c);
502 const scores: number[][] = Array.from({length: 18}, (_, h) => rounds.map(r => r.scores[h]).filter((s): s is number => typeof s === "number"));
503 const totals = scores.map(sum);
504 const order = Array.from({length: 18}, (_, h) => h).sort((a, b) => {
505 if (!scores[a].length || !scores[b].length) return Number(!scores[a].length) - Number(!scores[b].length) || a - b;
506 return (totals[b] - c.pars[b] * scores[b].length) * scores[a].length - (totals[a] - c.pars[a] * scores[a].length) * scores[b].length || a - b;
507 });
508 this.emit("holestats", c.id + ":", "rounds", rounds.length);
509 for (let h = 0; h < 18; h++) this.emit(h + 1, "par", c.pars[h], "si", c.si[h], "scores", scores[h].length, "avg", averageText(totals[h], scores[h].length), "rank", order.indexOf(h) + 1);
510 break;
511 }
512 case "PLAYERS": {
513 const players = [...this.players.values()].sort((a, b) => Number(a.index === null) - Number(b.index === null) || (a.index ?? 0) - (b.index ?? 0) || lex(a.id, b.id));
514 this.emit("players", players.length);
515 for (const p of players) this.emit(p.id, p.name, indexText(p.index), "rounds", p.rounds.length);
516 break;
517 }
518 case "EVENT": {
519 if (this.events.has(a[0])) throw new CommandError("duplicate event");
520 const [c, t] = this.tee(a[2], a[3]);
521 const allowance = Number(a[5]);
522 if (allowance < 1 || allowance > 100) throw new CommandError("invalid allowance");
523 const ids = a.slice(6);
524 if (ids.some(id => !this.players.has(id))) throw new CommandError("unknown player");
525 const n = ids.length;
526 const valid = ["MATCH", "NASSAU"].includes(a[1]) ? n === 2 : a[1] === "FOURBALL" ? n === 4 : a[1] === "BETTERBALL" ? [4, 6, 8].includes(n) : n >= 2 && n <= 8;
527 if (!valid || new Set(ids).size !== n) throw new CommandError("invalid players");
528 this.events.set(a[0], {id: a[0], form: a[1], course: c, tee: t, day: Number(a[4]), allowance, players: ids.map(id => this.players.get(id)!)});
529 break;
530 }
531 case "RESULTS": {
532 const e = this.events.get(a[0]);
533 if (!e) throw new CommandError("unknown event");
534 const d = this.eventData(e);
535 this.emit("results", e.id + ":", e.form, e.course.id, e.tee.id, "day", e.day, "allowance", e.allowance);
536 if (["STROKE", "STABLEFORD", "PAR"].includes(e.form)) this.individual(e, d);
537 else if (["MATCH", "NASSAU", "FOURBALL"].includes(e.form)) this.matches(e, d);
538 else if (e.form === "BETTERBALL") this.betterball(e, d);
539 else this.skins(e, d);
540 break;
541 }
542 case "MERIT": {
543 const totals = new Map<string, number[]>();
544 const awards = [10, 8, 6, 5, 4, 3, 2, 1];
545 let counted = 0;
546 for (const e of this.events.values()) {
547 if (!["STROKE", "STABLEFORD", "PAR"].includes(e.form)) continue;
548 let d: EventData;
549 try {
550 d = this.eventData(e);
551 } catch (error) {
552 if (!(error instanceof CommandError)) throw error;
553 continue;
554 }
555 counted++;
556 for (const p of e.players) {
557 if (!totals.has(p.id)) totals.set(p.id, [0, 0, 0]);
558 totals.get(p.id)![2]++;
559 }
560 const order = this.eventRanking(e, d).order;
561 for (let rank = 0; rank < order.length; rank++) {
562 const row = totals.get(order[rank])!;
563 row[0] += awards[rank];
564 if (rank === 0) row[1]++;
565 }
566 }
567 this.emit("merit:", "events", counted);
568 const ids = [...totals.keys()].sort((a, b) => totals.get(b)![0] - totals.get(a)![0] || totals.get(b)![1] - totals.get(a)![1] || lex(a, b));
569 for (let i = 0; i < ids.length; i++) {
570 const [points, wins, played] = totals.get(ids[i])!;
571 this.emit(i + 1, ids[i], points, "wins", wins, "played", played);
572 }
573 break;
574 }
575 }
576 }

580 try {
581 this.command(cmd, args);
582 } catch (error) {
583 if (!(error instanceof CommandError)) throw error;
584 this.emit("ERROR", error.message);
585 }
586 }
