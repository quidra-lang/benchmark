78List<int> allocation(int h, Course c) {
79 if (h >= 0) return c.si.map((si) => h ~/ 18 + (si <= h % 18 ? 1 : 0)).toList();
80 return c.si.map((si) => -((-h) ~/ 18) - (si >= 19 - (-h) % 18 ? 1 : 0)).toList();

82int handicap(int index, Course c, Tee t) => halfUp(index * t.slope + 113 * (t.rating - 10 * c.par), 1130);
83int points(int? net, int par) => net == null ? 0 : math.max(0, par + 2 - net);
84(int, int) countRule(int n) {
85 if (n <= 5) return (1, n == 3 ? -20 : n == 4 ? -10 : 0);
86 if (n == 6) return (2, -10);
87 return (n <= 8 ? 2 : n <= 11 ? 3 : n <= 14 ? 4 : n <= 16 ? 5 : n <= 18 ? 6 : n == 19 ? 7 : 8, 0);
88}
89List<Pick> eligible(Player p) {
90 final result = <Pick>[];
91 var reduction = 0;
92 for (var i = p.rounds.length - 1; i >= math.max(0, p.rounds.length - 20); i--) {
93 reduction += p.rounds[i].exceptional;
94 result.add(Pick(i + 1, p.rounds[i].diff - reduction));
95 }
96 if (result.length >= 3) {
97 final ranked = result.toList()..sort((a, b) {
98 final c = a.value.compareTo(b.value);
99 return c != 0 ? c : b.number.compareTo(a.number);
100 });
101 for (final pick in ranked.take(countRule(result.length).$1)) pick.counting = true;
102 }
103 return result.reversed.toList();
104}

125 void update(Player p, int day) {
126 final picks = eligible(p);
127 if (picks.length < 3) {
128 p.calculation = null;
129 return;
130 }
131 final rule = countRule(picks.length);
132 final avg = away(sum(picks.where((x) => x.counting).map((x) => x.value)), rule.$1);
133 var value = avg + rule.$2;
134 final low = lowIndex(p, day)?.$2;
135 var soft = false, hard = false, maximum = false;
136 if (low != null && value - low > 30) {
137 final capped = low + 30 + (value - low - 30) ~/ 2;
138 soft = capped != value;
139 value = capped;
140 }
141 if (low != null && value - low > 50) {
142 value = low + 50;
143 hard = true;
144 }
145 if (value > 540) {
146 value = 540;
147 maximum = true;
148 }
149 p.index = value;
150 p.calculation = Calculation(rule.$1, avg, rule.$2, low, hard, soft, maximum);
151 }
152 Result evaluate(Event e) {
153 final rs = <Round>[];
154 for (final p in e.players) {
155 final r = p.rounds.where((r) => r.course == e.course && r.tee == e.tee && r.day == e.day).firstOrNull;
156 if (r == null) fail('missing round');
157 if (r.handicap == null) fail('no index');
158 rs.add(r);
159 }
160 final ph = rs.map((r) => halfUp(r.handicap! * e.allowance, 100)).toList();
161 final minimum = ph.reduce(math.min);
162 final relative = ['MATCH', 'NASSAU', 'FOURBALL'].contains(e.format);
163 final nets = <List<int?>>[], pts = <List<int>>[];
164 for (var i = 0; i < rs.length; i++) {
165 final strokes = allocation(ph[i] - (relative ? minimum : 0), e.course);
166 final net = List<int?>.generate(18, (h) => rs[i].scores[h] == null || rs[i].scores[h]! <= 0 ? null : rs[i].scores[h]! - strokes[h]);
167 nets.add(net);
168 pts.add(List.generate(18, (h) => points(net[h], e.course.pars[h])));
169 }
170 return Result(e, rs, ph, nets, pts);
171 }
172 List<int> rank(Result r, {bool gross = false}) {
173 final e = r.event, rows = <int>[];
174 final values = <int, List<int>>{};
175 for (var i = 0; i < e.players.length; i++) {
176 if (e.format == 'STROKE' && r.nets[i].contains(null)) continue;
177 rows.add(i);
178 values[i] = countback(gross ? r.rounds[i].scores.cast<int>() : e.format == 'STABLEFORD' ? r.points[i] : e.format == 'PAR' ? List.generate(18, (h) => r.nets[i][h] == null || r.nets[i][h]! > e.course.pars[h] ? -1 : r.nets[i][h] == e.course.pars[h] ? 0 : 1) : r.nets[i].cast<int>());
179 }
180 rows.sort((a, b) {
181 final c = compareKeys(values[a]!, values[b]!, !gross && e.format != 'STROKE');
182 return c != 0 ? c : e.players[a].id.compareTo(e.players[b].id);
183 });
184 return rows;
185 }
186 (List<String>, String) match(List<int> wins, int first, int end, String side1, String side2) {
187 var lead = 0, remaining = end - first;
188 final statuses = <String>[];
189 for (var h = first; h < end; h++) {
190 lead += wins[h];
191 remaining = end - h - 1;
192 statuses.add(status(lead));
193 if (lead.abs() > remaining) break;
194 }
195 final result = lead == 0 ? 'AS' : '${lead > 0 ? side1 : side2} ${lead.abs()}${remaining > 0 ? '&$remaining' : ' UP'}';
196 return (statuses, result);
197 }

222 } else if (['MATCH', 'NASSAU', 'FOURBALL'].contains(e.format)) {

225 final four = e.format == 'FOURBALL';

228 final wins = List<int>.generate(18, (h) {
229 final a = (four ? [r.nets[0][h], r.nets[1][h]] : [r.nets[0][h]]).whereType<int>().toList();
230 final b = (four ? [r.nets[2][h], r.nets[3][h]] : [r.nets[1][h]]).whereType<int>().toList();

233 return -a.reduce(math.min).compareTo(b.reduce(math.min));

235 if (e.format == 'NASSAU') {

242 out.writeln('result ${m.$2}');
243 }
244 } else if (e.format == 'BETTERBALL') {

246 final values = List.generate(teams.length, (i) => List.generate(18, (h) => math.max(r.points[2 * i][h], r.points[2 * i + 1][h])));
247 final keys = values.map(countback).toList();
248 teams.sort((a, b) {
249 final c = compareKeys(keys[a], keys[b], true);
250 return c != 0 ? c : e.players[2 * a].id.compareTo(e.players[2 * b].id);
251 });
252 for (var rank = 0; rank < teams.length; rank++) {
253 final i = 2 * teams[rank];
254 out.writeln('${rank + 1} ${e.players[i].id}+${e.players[i + 1].id} points ${sum(values[i ~/ 2])} hcp ${hcpText(r.ph[i])} ${hcpText(r.ph[i + 1])}');

258 final totals = List.filled(e.players.length, 0);

281 }
282 final ids = List.generate(e.players.length, (i) => i)..sort((a, b) {
283 final c = totals[b].compareTo(totals[a]);
284 return c != 0 ? c : e.players[a].id.compareTo(e.players[b].id);
285 });
286 for (final i in ids) out.writeln('${e.players[i].id} ${two(totals[i], 100)}');
287 }

327 void execute(List<String> a, StringBuffer out) {
328 final op = a[0], id = a.length > 1 ? a[1] : '';
329 if (op == 'COURSE') {
330 if (courses.containsKey(id)) fail('duplicate course');
331 final pars = a.sublist(2, 20).map(int.parse).toList(), si = a.sublist(20).map(int.parse).toList();
332 if (pars.any((p) => p < 3 || p > 5) || si.toSet().length != 18 || si.any((s) => s < 1 || s > 18)) fail('invalid course');
333 courses[id] = Course(id, pars, si);
334 } else if (op == 'TEE') {
335 final c = course(id);
336 if (c.tees.containsKey(a[2])) fail('duplicate tee');
337 final r = int.parse(a[3]), s = int.parse(a[4]);
338 if (r < 500 || r > 900 || s < 55 || s > 155) fail('invalid tee');
339 c.tees[a[2]] = Tee(a[2], r, s);
340 } else if (op == 'PLAYER') {
341 if (players.containsKey(id)) fail('duplicate player');
342 final i = a.length > 3 ? int.parse(a[3]) : null;
343 if (i != null && (i < -100 || i > 540)) fail('invalid index');
344 players[id] = Player(id, a[2], i);
345 } else if (op == 'ROUND') {
346 final p = player(id), c = course(a[2]), t = tee(c, a[3]), day = int.parse(a[4]);
347 if (p.rounds.isNotEmpty && day < p.rounds.last.day) fail('invalid day');
348 final scores = a.sublist(5).map((s) => s == '-' ? null : s == 'X' ? 0 : int.parse(s)).toList();
349 if (scores.where((s) => s == null).length >= 5) fail('too few holes');
350 final hcp = p.index == null ? null : handicap(p.index!, c, t), strokes = allocation(hcp ?? 0, c);
351 final adjusted = List.generate(18, (h) {
352 final limit = c.pars[h] + (p.index == null ? 5 : 2 + strokes[h]);
353 return scores[h] == null ? c.pars[h] + strokes[h] : scores[h] == 0 ? limit : math.min(scores[h]!, limit);
354 });
355 final ags = sum(adjusted), diff = away(113 * (10 * ags - t.rating), t.slope);
356 final exceptional = p.index != null && p.index! - diff >= 70 ? (p.index! - diff < 100 ? 10 : 20) : 0;
357 p.rounds.add(Round(c, t, day, scores, hcp, adjusted, ags, diff, exceptional, p.index, p.calculation, p.records.length));
358 update(p, day);
359 if (p.index != null) p.records.add((day, p.index!));
360 out.writeln('round $id ${p.rounds.length}: ags $ags diff ${dec(diff)}${exceptional == 0 ? '' : ' exceptional ${dec(-exceptional)}'} index ${indexText(p.index)}');
361 } else if (op == 'REMOVE') {
362 final p = player(id);
363 if (p.rounds.isEmpty) fail('no rounds');
364 final no = p.rounds.length, r = p.rounds.removeLast();
365 p.index = r.oldIndex;
366 p.calculation = r.oldCalculation;
367 p.records.length = r.oldRecords;
368 out.writeln('removed $id $no: index ${indexText(p.index)}');
369 } else if (['INDEX', 'HISTORY', 'LOWINDEX', 'STATS', 'CARD'].contains(op)) {
370 final p = player(id);
371 if (op == 'INDEX') {
372 final c = p.calculation;
373 out.writeln('index $id: ${indexText(p.index)} rounds ${p.rounds.length}${c == null ? '' : ' lowest ${c.k} average ${dec(c.average)} adjustment ${dec(c.adjustment)}${c.low == null ? '' : ' low ${indexText(c.low)}'}${c.hard ? ' hard cap' : c.soft ? ' soft cap' : ''}${c.maximum ? ' maximum' : ''}'}');
374 } else if (op == 'HISTORY') {
375 out.writeln('history $id: rounds ${p.rounds.length}');
376 for (final pick in eligible(p)) {
377 final r = p.rounds[pick.number - 1];
378 out.writeln('${pick.number} day ${r.day} ${r.course.id} ${r.tee.id} ags ${r.ags} diff ${dec(r.diff)}${pick.value == r.diff ? '' : ' adjusted ${dec(pick.value)}'}${pick.counting ? ' *' : ''}');
379 }
380 } else if (op == 'LOWINDEX') {
381 final d = int.parse(a[2]), r = lowIndex(p, d);
382 out.writeln('lowindex $id $d: ${r == null ? 'none' : '${indexText(r.$2)} day ${r.$1}'}');
383 } else if (op == 'STATS') {
384 stats(p, out);
385 } else {
386 final n = int.parse(a[2]);
387 if (n > p.rounds.length) fail('unknown round');
388 card(p, n, p.rounds[n - 1], out);
389 }
390 } else if (op == 'COURSEHCP') {
391 final p = player(id), c = course(a[2]), t = tee(c, a[3]);
392 if (p.index == null) fail('no index');
393 final h = handicap(p.index!, c, t);
394 out.writeln('coursehcp $id ${c.id} ${t.id}: ${hcpText(h)} strokes ${allocation(h, c).join(' ')}');
395 } else if (op == 'HCPTABLE') {
396 final c = course(id), t = tee(c, a[2]), first = int.parse(a[3]), last = int.parse(a[4]);
397 if (first < -100 || first > last || last > 540) fail('invalid index');
398 out.writeln('hcptable $id ${t.id}: ${indexText(first)} to ${indexText(last)}');
399 var start = first, current = handicap(first, c, t);
400 for (var i = first + 1; i <= last + 1; i++) {
401 final h = i > last ? null : handicap(i, c, t);
402 if (h != current) {
403 out.writeln('${indexText(start)} to ${indexText(i - 1)}: ${hcpText(current)}');
404 start = i;
405 if (h != null) current = h;
406 }
407 }
408 } else if (op == 'HOLESTATS') {
409 final c = course(id), sums = List.filled(18, 0), counts = List.filled(18, 0);
410 var rounds = 0;
411 for (final p in players.values) {
412 for (final r in p.rounds.where((r) => r.course == c)) {
413 rounds++;
414 for (var h = 0; h < 18; h++) {
415 if (r.scores[h] != null && r.scores[h]! > 0) {
416 counts[h]++;
417 sums[h] += r.scores[h]!;
418 }
419 }
420 }
421 }
422 final holes = List.generate(18, (h) => h)..sort((a, b) {
423 if (counts[a] == 0 || counts[b] == 0) {
424 if (counts[a] == counts[b]) return a.compareTo(b);
425 return counts[a] == 0 ? 1 : -1;
426 }
427 final c1 = ((sums[b] - c.pars[b] * counts[b]) * counts[a]).compareTo((sums[a] - c.pars[a] * counts[a]) * counts[b]);
428 return c1 != 0 ? c1 : a.compareTo(b);
429 });
430 final ranks = List.filled(18, 0);
431 for (var i = 0; i < 18; i++) ranks[holes[i]] = i + 1;
432 out.writeln('holestats $id: rounds $rounds');
433 for (var h = 0; h < 18; h++) out.writeln('${h + 1} par ${c.pars[h]} si ${c.si[h]} scores ${counts[h]} avg ${counts[h] == 0 ? '-' : two(sums[h], counts[h])} rank ${ranks[h]}');
434 } else if (op == 'PLAYERS') {
435 final ps = players.values.toList()..sort((a, b) {
436 final c = a.index == null ? (b.index == null ? 0 : 1) : b.index == null ? -1 : a.index!.compareTo(b.index!);
437 return c != 0 ? c : a.id.compareTo(b.id);
438 });
439 out.writeln('players ${ps.length}');
440 for (final p in ps) out.writeln('${p.id} ${p.name} ${indexText(p.index)} rounds ${p.rounds.length}');
441 } else if (op == 'EVENT') {
442 if (events.containsKey(id)) fail('duplicate event');
443 final c = course(a[3]), t = tee(c, a[4]), allowance = int.parse(a[6]), ids = a.sublist(7), f = a[2];
444 if (allowance < 1 || allowance > 100) fail('invalid allowance');
445 final ps = ids.map(player).toList(), n = ids.length;
446 final valid = f == 'MATCH' || f == 'NASSAU' ? n == 2 : f == 'FOURBALL' ? n == 4 : f == 'BETTERBALL' ? [4, 6, 8].contains(n) : n >= 2 && n <= 8;
447 if (ids.toSet().length != n || !valid) fail('invalid players');
448 events[id] = Event(id, f, c, t, int.parse(a[5]), allowance, ps);
449 } else if (op == 'RESULTS') {
450 final e = events[id] ?? fail('unknown event');
451 results(evaluate(e), out);
452 } else if (op == 'MERIT') {
453 final values = <String, List<int>>{};
454 var counted = 0;
455 for (final e in events.values) {
456 if (!['STROKE', 'STABLEFORD', 'PAR'].contains(e.format)) continue;
457 Result r;
458 try {
459 r = evaluate(e);
460 } on GolfError {
461 continue;
462 }
463 counted++;
464 for (final p in e.players) values.putIfAbsent(p.id, () => [0, 0, 0])[2]++;
465 final ranked = rank(r);
466 for (var i = 0; i < ranked.length; i++) {
467 final v = values[e.players[ranked[i]].id]!;
468 v[0] += [10, 8, 6, 5, 4, 3, 2, 1][i];
469 if (i == 0) v[1]++;
470 }
471 }
472 final ids = values.keys.toList()..sort((a, b) {
473 var c = values[b]![0].compareTo(values[a]![0]);
474 if (c == 0) c = values[b]![1].compareTo(values[a]![1]);
475 return c != 0 ? c : a.compareTo(b);
476 });
477 out.writeln('merit: events $counted');
478 for (var i = 0; i < ids.length; i++) {
479 final id = ids[i], v = values[ids[i]]!;
480 out.writeln('${i + 1} $id ${v[0]} wins ${v[1]} played ${v[2]}');
481 }
482 }
483 }
484}
