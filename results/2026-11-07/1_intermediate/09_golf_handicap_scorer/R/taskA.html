8fd <- function(a,b) {
9 q <- floor(a/b)
10 while (q*b>a) q <- q-1
11 while ((q+1)*b<=a) q <- q+1
12 q
13}

57counting_rule <- function(n) {
58 if (n<3L) return(c(0L,0L))
59 if (n==3L) return(c(1L,-20L))
60 if (n==4L) return(c(1L,-10L))
61 if (n==5L) return(c(1L,0L))
62 if (n==6L) return(c(2L,-10L))
63 if (n<=8L) return(c(2L,0L))
64 if (n<=11L) return(c(3L,0L))
65 if (n<=14L) return(c(4L,0L))
66 if (n<=16L) return(c(5L,0L))
67 if (n<=18L) return(c(6L,0L))
68 c(if (n==19L) 7L else 8L,0L)
69}
70eligible <- function(p) {
71 if (!length(p$rounds)) return(list())
72 result <- list()
73 reduction <- 0
74 indices <- seq.int(max(1L,length(p$rounds)-19L),length(p$rounds))
75 for (i in rev(indices)) {
76 r <- p$rounds[[i]]
77 reduction <- reduction+r$exceptional
78 result[[length(result)+1L]] <- list(number=i,round=r,adjusted=r$diff-reduction)
79 }
80 rev(result)
81}
82counting <- function(items) {
83 k <- counting_rule(length(items))[1L]
84 values <- vapply(items,function(item) item$adjusted,numeric(1L))
85 numbers <- vapply(items,function(item) item$number,integer(1L))
86 items[order(values,-numbers,method="radix")][seq_len(k)]
87}
88update_index <- function(p,day) {
89 items <- eligible(p)
90 rule <- counting_rule(length(items))
91 k <- rule[1L]
92 adjustment <- rule[2L]
93 if (k==0L) {
94 p$calculation <- NULL
95 return(invisible(NULL))
96 }
97 total <- sum(vapply(counting(items),function(item) item$adjusted,numeric(1L)))
98 average <- half_away(total,k)
99 value <- average+adjustment
100 low <- low_index(p,day)
101 soft <- hard <- maximum <- FALSE
102 if (!is.null(low)) {
103 l <- low[2L]
104 if (value-l>30) {
105 new <- l+30+fd(value-l-30,2)
106 soft <- new!=value
107 value <- new
108 }
109 if (value-l>50) {
110 value <- l+50
111 hard <- TRUE
112 }
113 }
114 if (value>540) {
115 value <- 540
116 maximum <- TRUE
117 }
118 p$index <- value
119 p$calculation <- list(k=k,average=average,adjustment=adjustment,low=if (is.null(low)) NA_real_ else low[2L],hard=hard,soft=soft,maximum=maximum)
120}

133event_context <- function(e) {
134 checked <- tee_check(e$course,e$tee)
135 c <- checked$course
136 rounds <- ph <- list()
137 for (id in e$players) {
138 found <- NULL
139 for (r in players[[id]]$rounds) {
140 if (r$course==e$course && r$tee==e$tee && r$day==e$day) {
141 found <- r
142 break
143 }
144 }
145 if (is.null(found)) fail("missing round")
146 if (is.na(found$handicap)) fail("no index")
147 rounds[[id]] <- found
148 ph[[id]] <- half_up(found$handicap*e$allowance,100)
149 }
150 minimum <- if (e$format %in% c("MATCH","NASSAU","FOURBALL")) min(unlist(ph,use.names=FALSE)) else 0
151 nets <- points <- list()
152 for (id in e$players) {
153 values <- hole_values(rounds[[id]],allocate(ph[[id]]-minimum,c$si),c)
154 nets[[id]] <- values$nets
155 points[[id]] <- values$points
156 }
157 list(course=c,rounds=rounds,ph=ph,minimum=minimum,nets=nets,points=points)

159par_values <- function(nets,pars) ifelse(is.na(nets),-1,sign(pars-nets))
160ranking <- function(e,ctx) {

177play_match <- function(wins,names,first,last) {
178 lead <- remaining <- 0
179 statuses <- character()
180 for (h in first:last) {
181 lead <- lead+wins[h]
182 statuses <- c(statuses,status_text(lead))
183 remaining <- last-h
184 if (abs(lead)>remaining) break
185 }
186 result <- "AS"
187 if (lead!=0) {
188 name <- names[if (lead>0) 1L else 2L]
189 margin <- if (remaining>0) paste0(num(abs(lead)),"&",num(remaining)) else text(abs(lead),"UP")
190 result <- text(name,margin)
191 }
192 list(statuses=statuses,result=result)
193}

252 }
253 } else if (e$format=="BETTERBALL") {
