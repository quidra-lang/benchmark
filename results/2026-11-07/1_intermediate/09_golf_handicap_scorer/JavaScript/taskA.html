4class CommandError extends Error {}
5const sum = (v) => v.reduce((a, b) => a + b, 0);
6const lex = (a, b) => a < b ? -1 : a > b ? 1 : 0;
7const halfUp = (n, d) => Math.floor((2 * n + d) / (2 * d));
8const halfAway = (n, d) => n < 0 ? -halfUp(-n, d) : halfUp(n, d);
9const decimal = (n, sign = "-") => (n < 0 ? sign : "") + Math.floor(Math.abs(n) / 10) + "." + Math.abs(n) % 10;
10const indexText = (n) => n === null ? "none" : decimal(n, "+");
11const hcpText = (n) => n === null ? "none" : n < 0 ? "+" + -n : String(n);
12const hundredths = (n) => Math.floor(n / 100) + "." + String(n % 100).padStart(2, "0");
13const averageText = (total, count) => count ? hundredths(Math.floor(total * 100 / count)) : "-";
14const status = (n) => n === 0 ? "AS" : Math.abs(n) + (n > 0 ? " UP" : " DN");
15
16function allocation(h, si) {
17 const q = Math.floor(Math.abs(h) / 18);
18 const r = Math.abs(h) % 18;
19 return si.map(i => h >= 0 ? q + Number(i <= r) : -q - Number(i >= 19 - r));
20}
21
22function handicap(index, c, t) {
23 return halfUp(index * t.slope + 113 * (t.rating - 10 * c.par), 1130);
24}
25
26function ranked(ids, values, larger = false) {
27 const keys = {};
28 for (const id of ids) {
29 const v = values[id];
30 keys[id] = [sum(v), sum(v.slice(9)), sum(v.slice(12)), sum(v.slice(15)), v[17]];
31 }
32 return [...ids].sort((a, b) => {
33 for (let i = 0; i < 5; i++) {
34 const difference = keys[a][i] - keys[b][i];
35 if (difference) return larger ? -difference : difference;
36 }
37 return lex(a, b);
38 });
39}

41class Player {
42 id;
43 name;
44 index;
45 rounds = [];
46 records = [];
47 calculation = null;
48
49 constructor(id, name, index) {
50 this.id = id;
51 this.name = name;
52 this.index = index;
53 }
54
55 low(day) {
56 const records = this.records.filter(([d]) => d >= day - 365 && d < day);
57 records.sort((a, b) => a[1] - b[1] || a[0] - b[0]);
58 return records.length ? records[0] : null;
59 }
60
61 eligible() {
62 const rows = [];
63 let reduction = 0;
64 for (let i = this.rounds.length - 1; i >= Math.max(0, this.rounds.length - 20); i--) {
65 const round = this.rounds[i];
66 reduction += round.exceptional;
67 rows.push({number: i + 1, round, adjusted: round.diff - reduction});
68 }
69 return rows.reverse();
70 }
71
72 counting(rows) {
73 const n = rows.length;
74 if (n < 3) return {selected: [], adjustment: 0};
75 const k = [5, 8, 11, 14, 16, 18, 19, 20].findIndex(limit => n <= limit) + 1;
76 const adjustment = n === 3 ? -20 : n === 4 || n === 6 ? -10 : 0;
77 const selected = [...rows].sort((a, b) => a.adjusted - b.adjusted || b.number - a.number).slice(0, k);
78 return {selected, adjustment};
79 }
80
81 update(day) {
82 const {selected, adjustment} = this.counting(this.eligible());
83 if (!selected.length) {
84 this.calculation = null;
85 return;
86 }
87 const average = halfAway(sum(selected.map(r => r.adjusted)), selected.length);
88 let candidate = average + adjustment;
89 const record = this.low(day);
90 const low = record === null ? null : record[1];
91 let soft = false;
92 let hard = false;
93 let maximum = false;
94 if (low !== null) {
95 if (candidate - low > 30) {
96 candidate = low + 30 + Math.floor((candidate - low - 30) / 2);
97 soft = true;
98 }
99 if (candidate - low > 50) {
100 candidate = low + 50;
101 hard = true;
102 }
103 }
104 if (candidate > 540) {
105 candidate = 540;
106 maximum = true;
107 }
108 this.index = candidate;
109 this.calculation = {k: selected.length, average, adjustment, low, hard, soft, maximum};
110 }
111}

113class GolfRound {
114 course;
115 tee;
116 day;
117 scores;
118 previousIndex;
119 previousRecords;
120 previousCalculation;
121 hcp;
122 adjusted;
123 ags;
124 diff;
125 exceptional;
126
127 constructor(p, c, t, day, scores) {
128 this.course = c;
129 this.tee = t;
130 this.day = day;
131 this.scores = scores;
132 this.previousIndex = p.index;
133 this.previousRecords = p.records.length;
134 this.previousCalculation = p.calculation;
135 this.hcp = p.index === null ? null : handicap(p.index, c, t);
136 const strokes = allocation(this.hcp ?? 0, c.si);
137 this.adjusted = scores.map((score, h) => {
138 const limit = this.hcp === null ? c.pars[h] + 5 : c.pars[h] + 2 + strokes[h];
139 return score === "-" ? c.pars[h] + strokes[h] : score === "X" ? limit : Math.min(score, limit);
140 });
141 this.ags = sum(this.adjusted);
142 this.diff = halfAway(113 * (10 * this.ags - t.rating), t.slope);
143 this.exceptional = p.index !== null && p.index - this.diff >= 70 ? p.index - this.diff >= 100 ? 20 : 10 : 0;
144 }
145
146 nets(strokes) {
147 return this.scores.map((s, h) => typeof s === "number" ? s - strokes[h] : null);
148 }
149
150 points(nets) {
151 return nets.map((n, h) => n === null ? 0 : Math.max(0, this.course.pars[h] + 2 - n));
152 }
153
154 complete() {
155 return this.scores.every(s => typeof s === "number");
156 }
157}

188 eventData(e) {
189 const rounds = {};
190 const hcp = {};
191 const nets = {};
192 const points = {};
193 for (const p of e.players) {
194 const r = p.rounds.find(r => r.course === e.course && r.tee === e.tee && r.day === e.day);
195 if (!r) throw new CommandError("missing round");
196 if (r.hcp === null) throw new CommandError("no index");
197 rounds[p.id] = r;
198 hcp[p.id] = halfUp(r.hcp * e.allowance, 100);
199 }
200 const minimum = ["MATCH", "NASSAU", "FOURBALL"].includes(e.form) ? Math.min(...Object.values(hcp)) : 0;
201 for (const p of e.players) {
202 nets[p.id] = rounds[p.id].nets(allocation(hcp[p.id] - minimum, e.course.si));
203 points[p.id] = rounds[p.id].points(nets[p.id]);
204 }
205 return {rounds, hcp, nets, points};
206 }
207
208 eventRanking(e, d) {
209 const ids = e.players.map(p => p.id);
210 if (e.form === "STROKE") {
211 const complete = ids.filter(id => d.rounds[id].complete());
212 return {order: ranked(complete, d.nets), values: d.nets};
213 }
214 if (e.form === "STABLEFORD") return {order: ranked(ids, d.points, true), values: d.points};
215 const values = {};
216 for (const id of ids) values[id] = d.nets[id].map((n, h) => n === null || n > e.course.pars[h] ? -1 : n < e.course.pars[h] ? 1 : 0);
217 return {order: ranked(ids, values, true), values};
218 }
219
220 individual(e, d) {
221 const {order, values} = this.eventRanking(e, d);
222 for (let rank = 0; rank < order.length; rank++) {
223 const id = order[rank];
224 if (e.form === "STROKE") this.emit(rank + 1, id, "net", sum(values[id]), "gross", sum(d.rounds[id].scores), "hcp", hcpText(d.hcp[id]));
225 else if (e.form === "STABLEFORD") this.emit(rank + 1, id, "points", sum(values[id]), "hcp", hcpText(d.hcp[id]));
226 else this.emit(rank + 1, id, status(sum(values[id])), "hcp", hcpText(d.hcp[id]));
227 }
228 if (e.form === "STROKE") {
229 for (const id of e.players.map(p => p.id).filter(id => !d.rounds[id].complete()).sort()) this.emit("NR", id, "hcp", hcpText(d.hcp[id]));
230 if (order.length) {
231 const gross = {};
232 for (const id of order) gross[id] = d.rounds[id].scores;
233 const winner = ranked(order, gross)[0];
234 this.emit("gross", winner, sum(gross[winner]));
235 }
236 }
237 }

239 playMatch(names, holes, first, last) {
240 let lead = 0;
241 let remaining = last - first;
242 const statuses = [];
243 for (let h = first; h < last; h++) {
244 lead += holes[h];
245 statuses.push(status(lead));
246 remaining = last - h - 1;
247 if (Math.abs(lead) > remaining || remaining === 0) break;
248 }
249 const result = lead === 0 ? "AS" : names[lead > 0 ? 0 : 1] + " " + Math.abs(lead) + (remaining ? "&" + remaining : " UP");
250 return {statuses, result};
251 }
252
253 matches(e, d) {
254 const ids = e.players.map(p => p.id);
255 const sides = e.form === "FOURBALL" ? [ids.slice(0, 2), ids.slice(2)] : [[ids[0]], [ids[1]]];
256 const names = sides.map(side => side.join("+"));
257 const holes = [];
258 for (let h = 0; h < 18; h++) {
259 const scores = sides.map(side => {
260 const values = side.map(id => d.nets[id][h]).filter((n) => n !== null);
261 return values.length ? Math.min(...values) : null;
262 });
263 const [a, b] = scores;
264 holes.push(a === null ? b === null ? 0 : -1 : b === null || a < b ? 1 : a > b ? -1 : 0);
265 }
266 const minimum = Math.min(...Object.values(d.hcp));
267 const parts = ["strokes"];
268 for (const id of ids) parts.push(id, d.hcp[id] - minimum);
269 this.emit(...parts);
270 if (e.form === "NASSAU") {
271 for (const [label, first, last] of [["front", 0, 9], ["back", 9, 18], ["total", 0, 18]]) {
272 this.emit(label, this.playMatch(names, holes, first, last).result);
273 }
274 } else {
275 const match = this.playMatch(names, holes, 0, 18);
276 this.emit("status", match.statuses.join(", "));
277 this.emit("result", match.result);
278 }
279 }

281 betterball(e, d) {
282 const teams = {};
283 const values = {};
284 for (let i = 0; i < e.players.length; i += 2) {
285 const a = e.players[i].id;
286 const b = e.players[i + 1].id;
287 teams[a] = b;
288 values[a] = d.points[a].map((v, h) => Math.max(v, d.points[b][h]));
289 }
290 const order = ranked(Object.keys(teams), values, true);
291 for (let i = 0; i < order.length; i++) {
292 const a = order[i];
293 const b = teams[a];
294 this.emit(i + 1, a + "+" + b, "points", sum(values[a]), "hcp", hcpText(d.hcp[a]), hcpText(d.hcp[b]));
295 }
296 }

298 skins(e, d) {
299 const ids = e.players.map(p => p.id);
300 const totals = Object.fromEntries(ids.map(id => [id, 0]));
301 let carry = 0;
302 let winners = [];
303 for (let h = 0; h < 18; h++) {
304 carry++;
305 const available = ids.filter(id => d.nets[id][h] !== null);
306 const best = available.length ? Math.min(...available.map(id => d.nets[id][h])) : null;
307 winners = available.filter(id => d.nets[id][h] === best);
308 if (winners.length === 1) {
309 this.emit("hole", h + 1, winners[0], carry);
310 totals[winners[0]] += 100 * carry;
311 carry = 0;
312 }
313 }
314 if (carry) {
315 const sharing = (winners.length ? winners : ids).toSorted();
316 const share = Math.floor(carry * 100 / sharing.length);
317 for (const id of sharing) totals[id] += share;
318 this.emit("carry", carry, ...sharing);
319 }
320 for (const id of [...ids].sort((a, b) => totals[b] - totals[a] || lex(a, b))) this.emit(id, hundredths(totals[id]));
321 }

323 cardTotal(label, r, strokes, nets, points, first, last) {
324 const scores = r.scores.slice(first, last);
325 const complete = scores.every(s => typeof s === "number");
326 this.emit(label, sum(r.course.pars.slice(first, last)), complete ? sum(scores) : "-",
327 sum(r.adjusted.slice(first, last)), sum(strokes.slice(first, last)),
328 complete ? sum(nets.slice(first, last)) : "-", sum(points.slice(first, last)));
329 }

331 command(cmd, a) {
332 switch (cmd) {
333 case "COURSE": {
334 if (this.courses.has(a[0])) throw new CommandError("duplicate course");
335 const pars = a.slice(1, 19).map(Number);
336 const si = a.slice(19).map(Number);
337 if (pars.some(p => ![3, 4, 5].includes(p)) || [...si].sort((a, b) => a - b).some((v, i) => v !== i + 1)) throw new CommandError("invalid course");
338 this.courses.set(a[0], {id: a[0], pars, si, par: sum(pars), tees: new Map()});
339 break;
340 }
341 case "TEE": {
342 const c = this.course(a[0]);
343 if (c.tees.has(a[1])) throw new CommandError("duplicate tee");
344 const rating = Number(a[2]);
345 const slope = Number(a[3]);
346 if (rating < 500 || rating > 900 || slope < 55 || slope > 155) throw new CommandError("invalid tee");
347 c.tees.set(a[1], {id: a[1], rating, slope});
348 break;
349 }
350 case "PLAYER": {
351 if (this.players.has(a[0])) throw new CommandError("duplicate player");
352 const index = a.length === 3 ? Number(a[2]) : null;
353 if (index !== null && (index < -100 || index > 540)) throw new CommandError("invalid index");
354 this.players.set(a[0], new Player(a[0], a[1], index));
355 break;
356 }
357 case "ROUND": {
358 const p = this.player(a[0]);
359 const [c, t] = this.tee(a[1], a[2]);
360 const day = Number(a[3]);
361 if (p.rounds.length && day < p.rounds[p.rounds.length - 1].day) throw new CommandError("invalid day");
362 const scores = a.slice(4).map(s => s === "-" || s === "X" ? s : Number(s));
363 if (scores.filter(s => s === "-").length >= 5) throw new CommandError("too few holes");
364 const r = new GolfRound(p, c, t, day, scores);
365 p.rounds.push(r);
366 p.update(day);
367 if (p.index !== null) p.records.push([day, p.index]);
368 const parts = ["round", p.id, p.rounds.length + ":", "ags", r.ags, "diff", decimal(r.diff)];
369 if (r.exceptional) parts.push("exceptional", decimal(-r.exceptional));
370 this.emit(...parts, "index", indexText(p.index));
371 break;
372 }
373 case "REMOVE": {
374 const p = this.player(a[0]);
375 if (!p.rounds.length) throw new CommandError("no rounds");
376 const number = p.rounds.length;
377 const r = p.rounds.pop();
378 p.index = r.previousIndex;
379 p.records.length = r.previousRecords;
380 p.calculation = r.previousCalculation;
381 this.emit("removed", p.id, number + ":", "index", indexText(p.index));
382 break;
383 }
384 case "INDEX": {
385 const p = this.player(a[0]);
386 const parts = ["index", p.id + ":", indexText(p.index), "rounds", p.rounds.length];
387 const c = p.calculation;
388 if (c) {
389 parts.push("lowest", c.k, "average", decimal(c.average), "adjustment", decimal(c.adjustment));
390 if (c.low !== null) parts.push("low", indexText(c.low));
391 if (c.hard) parts.push("hard", "cap");
392 else if (c.soft) parts.push("soft", "cap");
393 if (c.maximum) parts.push("maximum");
394 }
395 this.emit(...parts);
396 break;
397 }
398 case "HISTORY": {
399 const p = this.player(a[0]);
400 const rows = p.eligible();
401 const counted = new Set(p.counting(rows).selected.map(r => r.number));
402 this.emit("history", p.id + ":", "rounds", p.rounds.length);
403 for (const row of rows) {
404 const r = row.round;
405 const parts = [row.number, "day", r.day, r.course.id, r.tee.id, "ags", r.ags, "diff", decimal(r.diff)];
406 if (row.adjusted !== r.diff) parts.push("adjusted", decimal(row.adjusted));
407 if (counted.has(row.number)) parts.push("*");
408 this.emit(...parts);
409 }
410 break;
411 }
412 case "LOWINDEX": {
413 const p = this.player(a[0]);
414 const day = Number(a[1]);
415 const low = p.low(day);
416 this.emit("lowindex", p.id, day + ":", ...(low === null ? ["none"] : [indexText(low[1]), "day", low[0]]));
417 break;
418 }
419 case "COURSEHCP": {
420 const p = this.player(a[0]);
421 const [c, t] = this.tee(a[1], a[2]);
422 if (p.index === null) throw new CommandError("no index");
423 const h = handicap(p.index, c, t);
424 this.emit("coursehcp", p.id, c.id, t.id + ":", hcpText(h), "strokes", ...allocation(h, c.si));
425 break;
426 }
427 case "HCPTABLE": {
428 const [c, t] = this.tee(a[0], a[1]);
429 const first = Number(a[2]);
430 const last = Number(a[3]);
431 if (first < -100 || first > last || last > 540) throw new CommandError("invalid index");
432 this.emit("hcptable", c.id, t.id + ":", indexText(first), "to", indexText(last));
433 let start = first;
434 let h = handicap(first, c, t);
435 for (let i = first + 1; i <= last; i++) {
436 const next = handicap(i, c, t);
437 if (next !== h) {
438 this.emit(indexText(start), "to", indexText(i - 1) + ":", hcpText(h));
439 start = i;
440 h = next;
441 }
442 }
443 this.emit(indexText(start), "to", indexText(last) + ":", hcpText(h));
444 break;
445 }
446 case "CARD": {
447 const p = this.player(a[0]);
448 const number = Number(a[1]);
449 if (number > p.rounds.length) throw new CommandError("unknown round");
450 const r = p.rounds[number - 1];
451 const c = r.course;
452 const strokes = allocation(r.hcp ?? 0, c.si);
453 const nets = r.nets(strokes);
454 const points = r.points(nets);
455 this.emit("card", p.id, number + ":", c.id, r.tee.id, "day", r.day, "handicap", hcpText(r.hcp));
456 for (let h = 0; h < 18; h++) {
457 this.emit(h + 1, c.pars[h], c.si[h], r.scores[h], r.adjusted[h], strokes[h], nets[h] ?? "-", points[h]);
458 if (h === 8) this.cardTotal("OUT", r, strokes, nets, points, 0, 9);
459 if (h === 17) {
460 this.cardTotal("IN", r, strokes, nets, points, 9, 18);
461 this.cardTotal("TOTAL", r, strokes, nets, points, 0, 18);
462 }
463 }
464 this.emit("ags", r.ags, "diff", decimal(r.diff));
465 break;
466 }
467 case "STATS": {
468 const p = this.player(a[0]);
469 const labels = ["eagles", "birdies", "pars", "bogeys", "doubles", "others", "pickups"];
470 const counts = Array(7).fill(0);
471 const parScores = {3: [], 4: [], 5: []};
472 const totals = [];
473 for (const r of p.rounds) {
474 if (r.complete()) totals.push(sum(r.scores));
475 for (let h = 0; h < 18; h++) {
476 const score = r.scores[h];
477 if (score === "X") counts[6]++;
478 else if (typeof score === "number") {
479 const relative = score - r.course.pars[h];
480 counts[Math.max(0, Math.min(relative + 2, 5))]++;
481 parScores[r.course.pars[h]].push(score);
482 }
483 }
484 }
485 const parts = ["stats", p.id + ":", "rounds", p.rounds.length, "best", totals.length ? Math.min(...totals) : "-", "average", averageText(sum(totals), totals.length)];
486 for (let i = 0; i < labels.length; i++) parts.push(labels[i], counts[i]);
487 for (const par of [3, 4, 5]) parts.push("par" + par, averageText(sum(parScores[par]), parScores[par].length));
488 this.emit(...parts);
489 break;
490 }
491 case "HOLESTATS": {
492 const c = this.course(a[0]);
493 const rounds = [...this.players.values()].flatMap(p => p.rounds).filter(r => r.course === c);
494 const scores = Array.from({length: 18}, (_, h) => rounds.map(r => r.scores[h]).filter((s) => typeof s === "number"));
495 const totals = scores.map(sum);
496 const order = Array.from({length: 18}, (_, h) => h).sort((a, b) => {
497 if (!scores[a].length || !scores[b].length) return Number(!scores[a].length) - Number(!scores[b].length) || a - b;
498 return (totals[b] - c.pars[b] * scores[b].length) * scores[a].length - (totals[a] - c.pars[a] * scores[a].length) * scores[b].length || a - b;
499 });
500 this.emit("holestats", c.id + ":", "rounds", rounds.length);
501 for (let h = 0; h < 18; h++) this.emit(h + 1, "par", c.pars[h], "si", c.si[h], "scores", scores[h].length, "avg", averageText(totals[h], scores[h].length), "rank", order.indexOf(h) + 1);
502 break;
503 }
504 case "PLAYERS": {
505 const players = [...this.players.values()].sort((a, b) => Number(a.index === null) - Number(b.index === null) || (a.index ?? 0) - (b.index ?? 0) || lex(a.id, b.id));
506 this.emit("players", players.length);
507 for (const p of players) this.emit(p.id, p.name, indexText(p.index), "rounds", p.rounds.length);
508 break;
509 }
510 case "EVENT": {
511 if (this.events.has(a[0])) throw new CommandError("duplicate event");
512 const [c, t] = this.tee(a[2], a[3]);
513 const allowance = Number(a[5]);
514 if (allowance < 1 || allowance > 100) throw new CommandError("invalid allowance");
515 const ids = a.slice(6);
516 if (ids.some(id => !this.players.has(id))) throw new CommandError("unknown player");
517 const n = ids.length;
518 const valid = ["MATCH", "NASSAU"].includes(a[1]) ? n === 2 : a[1] === "FOURBALL" ? n === 4 : a[1] === "BETTERBALL" ? [4, 6, 8].includes(n) : n >= 2 && n <= 8;
519 if (!valid || new Set(ids).size !== n) throw new CommandError("invalid players");
520 this.events.set(a[0], {id: a[0], form: a[1], course: c, tee: t, day: Number(a[4]), allowance, players: ids.map(id => this.players.get(id))});
521 break;
522 }
523 case "RESULTS": {
524 const e = this.events.get(a[0]);
525 if (!e) throw new CommandError("unknown event");
526 const d = this.eventData(e);
527 this.emit("results", e.id + ":", e.form, e.course.id, e.tee.id, "day", e.day, "allowance", e.allowance);
528 if (["STROKE", "STABLEFORD", "PAR"].includes(e.form)) this.individual(e, d);
529 else if (["MATCH", "NASSAU", "FOURBALL"].includes(e.form)) this.matches(e, d);
530 else if (e.form === "BETTERBALL") this.betterball(e, d);
531 else this.skins(e, d);
532 break;
533 }
534 case "MERIT": {
535 const totals = new Map();
536 const awards = [10, 8, 6, 5, 4, 3, 2, 1];
537 let counted = 0;

541 try {

551 }

554 const row = totals.get(order[rank]);
555 row[0] += awards[rank];
556 if (rank === 0) row[1]++;
557 }
558 }
559 this.emit("merit:", "events", counted);
560 const ids = [...totals.keys()].sort((a, b) => totals.get(b)[0] - totals.get(a)[0] || totals.get(b)[1] - totals.get(a)[1] || lex(a, b));
561 for (let i = 0; i < ids.length; i++) {

563 this.emit(i + 1, ids[i], points, "wins", wins, "played", played);
