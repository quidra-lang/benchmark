7function floor_div(int $a,int $b): int
8{
9 $q = intdiv($a,$b);
10 return $a<0 && $a%$b!==0 ? $q-1 : $q;
11}
12function half_up(int $a,int $b): int { return floor_div(2*$a+$b,2*$b); }
13function half_away(int $a,int $b): int { return ($a<0 ? -1 : 1)*intdiv(2*abs($a)+$b,2*$b); }
14function decimal(int $v): string { return intdiv(abs($v),10) . '.' . abs($v)%10; }
15function index_text(?int $v): string { return $v===null ? 'none' : ($v<0 ? '+' : '') . decimal($v); }
16function signed_text(int $v): string { return ($v<0 ? '-' : '') . decimal($v); }
17function handicap_text(?int $v): string { return $v===null ? 'none' : ($v<0 ? '+' . -$v : (string)$v); }
18function hundredths(int $v): string { return intdiv($v,100) . '.' . sprintf('%02d',$v%100); }
19function mean_text(int $sum,int $count): string { return $count===0 ? '-' : hundredths(intdiv($sum*100,$count)); }

47function allocate(int $handicap,array $si): array
48{
49 $q = intdiv(abs($handicap),18);
50 $r = abs($handicap)%18;
51 return array_map(fn($rank)=>$handicap>=0 ? $q+(int)($rank<=$r) : -$q-(int)($rank>=19-$r),$si);
52}
53function low_index(object $p,int $day): ?array
54{
55 $best = null;
56 foreach ($p->records as $record) {
57 if ($record[0]>=$day-365 && $record[0]<$day && ($best===null || [$record[1],$record[0]]<[$best[1],$best[0]])) $best = $record;
58 }
59 return $best;

61function counting_rule(int $n): array
62{
63 if ($n<3) return [0,0];
64 if ($n===3) return [1,-20];
65 if ($n===4) return [1,-10];
66 if ($n===5) return [1,0];
67 if ($n===6) return [2,-10];
68 if ($n<=8) return [2,0];
69 if ($n<=11) return [3,0];
70 if ($n<=14) return [4,0];
71 if ($n<=16) return [5,0];
72 if ($n<=18) return [6,0];
73 return [$n===19 ? 7 : 8,0];
74}
75function eligible(object $p): array
76{
77 $result = [];
78 $reduction = 0;
79 for ($i=count($p->rounds)-1; $i>=max(0,count($p->rounds)-20); --$i) {
80 $r = $p->rounds[$i];
81 $reduction += $r->exceptional;
82 $result[] = ['number'=>$i+1,'round'=>$r,'adjusted'=>$r->diff-$reduction];
83 }
84 return array_reverse($result);
85}
86function counting(array $eligible): array
87{
88 [$k] = counting_rule(count($eligible));
89 usort($eligible,fn($a,$b)=>[$a['adjusted'],-$a['number']]<=>[$b['adjusted'],-$b['number']]);
90 return array_slice($eligible,0,$k);
91}
92function update_index(object $p,int $day): void
93{
94 $eligible = eligible($p);
95 [$k,$adjustment] = counting_rule(count($eligible));
96 if ($k===0) {
97 $p->calculation = null;
98 return;
99 }
100 $sum = array_sum(array_column(counting($eligible),'adjusted'));
101 $average = half_away($sum,$k);
102 $value = $average+$adjustment;
103 $low = low_index($p,$day);
104 $soft = false;
105 $hard = false;
106 $maximum = false;
107 if ($low!==null) {
108 $l = $low[1];
109 if ($value-$l>30) {
110 $new = $l+30+intdiv($value-$l-30,2);
111 $soft = $new!==$value;
112 $value = $new;
113 }
114 if ($value-$l>50) {
115 $value = $l+50;
116 $hard = true;
117 }
118 }
119 if ($value>540) {
120 $value = 540;
121 $maximum = true;
122 }
123 $p->index = $value;
124 $p->calculation = ['k'=>$k,'average'=>$average,'adjustment'=>$adjustment,'low'=>$low===null ? null : $low[1],'hard'=>$hard,'soft'=>$soft,'maximum'=>$maximum];
125}

139 $key = [];

143function event_context(object $e): array
144{
145 global $players;
146 [$c] = tee_check($e->course,$e->tee);
147 $rounds = [];
148 $ph = [];
149 foreach ($e->players as $id) {
150 $found = array_find($players[$id]->rounds,fn($r)=>
151 $r->course===$e->course && $r->tee===$e->tee && $r->day===$e->day);
152 if ($found===null) fail('missing round');
153 if ($found->handicap===null) fail('no index');
154 $rounds[$id] = $found;
155 $ph[$id] = half_up($found->handicap*$e->allowance,100);
156 }
157 $minimum = in_array($e->format,['MATCH','NASSAU','FOURBALL'],true) ? min($ph) : 0;
158 $nets = [];
159 $points = [];
160 foreach ($e->players as $id) [$nets[$id],$points[$id]] = hole_values($rounds[$id],allocate($ph[$id]-$minimum,$c->si),$c);
161 return ['course'=>$c,'rounds'=>$rounds,'ph'=>$ph,'minimum'=>$minimum,'nets'=>$nets,'points'=>$points];
162}

167function ranking(object $e,array $ctx): array
168{
169 $ranked = [];
170 $keys = [];
171 foreach ($e->players as $id) {
172 if ($e->format==='STROKE') {
173 if (in_array(null,$ctx['nets'][$id],true)) continue;
174 $values = $ctx['nets'][$id];
175 } elseif ($e->format==='STABLEFORD') {
176 $values = $ctx['points'][$id];
177 } else {
178 $values = par_values($ctx['nets'][$id],$ctx['course']->pars);
179 }
180 $ranked[] = $id;
181 $keys[$id] = countback($values,$e->format!=='STROKE');
182 }
183 usort($ranked,fn($a,$b)=>($keys[$a]<=>$keys[$b]) ?: strcmp($a,$b));
184 return $ranked;

186function play_match(array $wins,array $names,int $first,int $last): array
187{
188 $lead = 0;
189 $statuses = [];
190 $remaining = 0;
191 for ($h=$first; $h<=$last; ++$h) {
192 $lead += $wins[$h];
193 $statuses[] = status_text($lead);
194 $remaining = $last-$h;
195 if (abs($lead)>$remaining) break;
196 }
197 $result = 'AS';
198 if ($lead!==0) $result = $names[$lead>0 ? 0 : 1] . ' ' . ($remaining>0 ? abs($lead) . "&$remaining" : abs($lead) . ' UP');
199 return [$statuses,$result];
200}
