85 Eligible eligible(Player p) {
86 int start=Math.max(0,p.rounds.size()-20), n=p.rounds.size()-start, reduction=0;
87 int[] adjusted=new int[n];
88 for (int i=n-1;i>=0;i--) {

90 reduction+=r.exceptional;
91 adjusted[i]=r.diff-reduction;

94 int adjustment=n==3 ? -20 : n==4 || n==6 ? -10 : 0;
95 List<Integer> counting=IntStream.range(0,n).boxed().sorted(Comparator.comparingInt((Integer i) -> adjusted[i])
96 .thenComparing(Comparator.reverseOrder())).limit(k).toList();
97 return new Eligible(start,adjusted,counting,k,adjustment);

99 void update(Player p, int day) {
100 Eligible e=eligible(p);
101 if (e.k()==0) {
102 p.calculation=null;
103 return;
104 }
105 int total=e.counting().stream().mapToInt(i -> e.adjusted()[i]).sum();
106 int a=away(total,e.k()), c=a+e.adjustment();
107 IndexRecord record=low(p,day);
108 Integer l=record==null ? null : record.value();
109 boolean soft=false,hard=false,maximum=false;
110 if (l!=null && c-l>30) {
111 int next=l+30+(c-l-30)/2;
112 soft=next!=c;
113 c=next;
114 }
115 if (l!=null && c-l>50) {
116 c=l+50;
117 hard=true;
118 }
119 if (c>540) {
120 c=540;
121 maximum=true;
122 }
123 p.index=c;
124 p.calculation=new Calculation(e.k(),a,e.adjustment(),l,hard,soft,maximum);

126 EventData data(Event e) {
127 List<Round> rounds=new ArrayList<>();
128 int n=e.players().size();
129 int[] ph=new int[n], strokes=new int[n];
130 for (int i=0;i<n;i++) {
131 Round r=e.players().get(i).rounds.stream().filter(x -> x.course==e.course() && x.tee==e.tee() && x.day==e.day()).findFirst().orElse(null);
132 error(r==null,"missing round");
133 error(r.handicap==null,"no index");
134 rounds.add(r);
135 ph[i]=up((long)r.handicap*e.allowance(),100);
136 }
137 int minimum=Arrays.stream(ph).min().orElse(0);
138 boolean relative=List.of("MATCH","NASSAU","FOURBALL").contains(e.format());
139 int[][] nets=new int[n][18], points=new int[n][18], par=new int[n][18];
140 for (int i=0;i<n;i++) {
141 strokes[i]=ph[i]-(relative ? minimum : 0);
142 int[] allocation=allocate(e.course(),strokes[i]);
143 for (int h=0;h<18;h++) {
144 nets[i][h]=rounds.get(i).scores[h]==NO_NET ? NO_NET : rounds.get(i).scores[h]-allocation[h];
145 points[i][h]=nets[i][h]==NO_NET ? 0 : Math.max(0,e.course().par[h]+2-nets[i][h]);
146 par[i][h]=nets[i][h]==NO_NET || nets[i][h]>e.course().par[h] ? -1 : nets[i][h]<e.course().par[h] ? 1 : 0;
147 }
148 }
149 return new EventData(e,rounds,ph,strokes,nets,points,par);

156 .map(Keyed::index).toList();

164 int[][] values=d.event().format().equals("STROKE") ? d.nets() : d.event().format().equals("PAR") ? d.par() : d.points();

167 String status(int lead) { return lead==0 ? "AS" : lead>0 ? lead+" UP" : (-lead)+" DN"; }
168 Match match(int[] wins, int first, int last, String a, String b) {
169 int lead=0,remaining=0;
170 List<String> statuses=new ArrayList<>();
171 for (int h=first;h<=last;h++) {
172 lead+=wins[h];
173 remaining=last-h;
174 statuses.add(status(lead));
175 if (Math.abs(lead)>remaining) break;
176 }
177 String result=lead==0 ? "AS" : (lead>0 ? a : b)+" "+Math.abs(lead)+(remaining>0 ? "&"+remaining : " UP");
178 return new Match(statuses,result);

180 void results(Event e, EventData d) {
181 output.add("results "+e.id()+": "+e.format()+" "+e.course().id+" "+e.tee().id()+" day "+e.day()+" allowance "+e.allowance());
182 int n=e.players().size();
183 List<String> ids=e.players().stream().map(p -> p.id).toList();
184 if (List.of("STROKE","STABLEFORD","PAR").contains(e.format())) {
185 List<Integer> order=ranked(d);
186 for (int r=0;r<order.size();r++) {
187 int i=order.get(r);
188 String value=e.format().equals("STROKE") ? "net "+sum(d.nets()[i],0,18)+" gross "+sum(d.rounds().get(i).scores,0,18) :
189 e.format().equals("STABLEFORD") ? "points "+sum(d.points()[i],0,18) : status(sum(d.par()[i],0,18));
190 output.add((r+1)+" "+ids.get(i)+" "+value+" hcp "+handicap(d.ph()[i]));
191 }
192 if (e.format().equals("STROKE")) {
193 List<Integer> nr=all(n);
194 nr.removeAll(order);
195 nr.sort(Comparator.comparing(ids::get));
196 for (int i:nr) output.add("NR "+ids.get(i)+" hcp "+handicap(d.ph()[i]));
197 if (!order.isEmpty()) {
198 int[][] gross=d.rounds().stream().map(r -> r.scores).toArray(int[][]::new);
199 int winner=rank(gross,order,false,ids).get(0);
200 output.add("gross "+ids.get(winner)+" "+sum(gross[winner],0,18));
201 }
202 }
203 } else if (List.of("MATCH","NASSAU","FOURBALL").contains(e.format())) {
204 String line="strokes";
205 for (int i=0;i<n;i++) line+=" "+ids.get(i)+" "+d.strokes()[i];
206 output.add(line);
207 int size=n/2;
208 String a=String.join("+",ids.subList(0,size)), b=String.join("+",ids.subList(size,n));
209 int[] wins=new int[18];
210 for (int h=0;h<18;h++) {
211 int left=NO_NET,right=NO_NET;
212 for (int i=0;i<size;i++) left=Math.min(left,d.nets()[i][h]);
213 for (int i=size;i<n;i++) right=Math.min(right,d.nets()[i][h]);
214 wins[h]=Integer.compare(right,left);
215 }
216 if (e.format().equals("NASSAU")) {
217 output.add("front "+match(wins,0,8,a,b).result());
218 output.add("back "+match(wins,9,17,a,b).result());
219 output.add("total "+match(wins,0,17,a,b).result());
220 } else {
221 Match m=match(wins,0,17,a,b);
222 output.add("status "+String.join(", ",m.statuses()));
223 output.add("result "+m.result());
224 }
225 } else if (e.format().equals("BETTERBALL")) {
226 int[][] values=new int[n/2][18];
227 List<String> keys=new ArrayList<>();
228 for (int i=0;i<n/2;i++) {
229 keys.add(ids.get(2*i));
230 for (int h=0;h<18;h++) values[i][h]=Math.max(d.points()[2*i][h],d.points()[2*i+1][h]);
231 }
232 List<Integer> order=rank(values,all(n/2),true,keys);
233 for (int r=0;r<order.size();r++) {
234 int i=order.get(r);
235 output.add((r+1)+" "+ids.get(2*i)+"+"+ids.get(2*i+1)+" points "+sum(values[i],0,18)+" hcp "+handicap(d.ph()[2*i])+" "+handicap(d.ph()[2*i+1]));
236 }
237 } else {
238 int carry=0;
239 int[] totals=new int[n];
240 List<Integer> tied=all(n);
241 for (int h=0;h<18;h++) {
242 carry++;
243 int minimum=NO_NET;
244 for (int i=0;i<n;i++) minimum=Math.min(minimum,d.nets()[i][h]);
245 tied=new ArrayList<>();
246 for (int i=0;i<n;i++) if (d.nets()[i][h]==minimum) tied.add(i);
247 if (minimum!=NO_NET && tied.size()==1) {
248 int i=tied.get(0);
249 output.add("hole "+(h+1)+" "+ids.get(i)+" "+carry);
250 totals[i]+=carry*100;
251 carry=0;
252 }
253 }
254 if (carry>0) {
255 tied.sort(Comparator.comparing(ids::get));
256 String line="carry "+carry;
257 for (int i:tied) {
258 line+=" "+ids.get(i);
259 totals[i]+=carry*100/tied.size();
260 }
261 output.add(line);
262 }
263 List<Integer> order=all(n);
264 order.sort(Comparator.comparingInt((Integer i) -> -totals[i]).thenComparing(ids::get));
265 for (int i:order) output.add(ids.get(i)+" "+cents(totals[i]));
266 }

288 void statistics(Player p) {
289 int[] category=new int[7], totals=new int[3], counts=new int[3];
290 List<Integer> complete=new ArrayList<>();
291 for (Round r:p.rounds) {
292 boolean full=true;
293 for (int h=0;h<18;h++) {
294 if (r.raw[h].equals("X")) category[6]++;
295 if (r.scores[h]==NO_NET) {
296 full=false;
297 continue;
298 }
299 int relative=r.scores[h]-r.course.par[h];
300 category[relative<=-2 ? 0 : relative==-1 ? 1 : relative==0 ? 2 : relative==1 ? 3 : relative==2 ? 4 : 5]++;
301 int i=r.course.par[h]-3;
302 totals[i]+=r.scores[h];
303 counts[i]++;
304 }
305 if (full) complete.add(sum(r.scores,0,18));
306 }
307 String line="stats "+p.id+": rounds "+p.rounds.size()+" best "+(complete.isEmpty() ? "-" : Collections.min(complete))+" average "+
308 average(complete.stream().mapToInt(Integer::intValue).sum(),complete.size());
309 String[] labels={"eagles","birdies","pars","bogeys","doubles","others","pickups"};
310 for (int i=0;i<7;i++) line+=" "+labels[i]+" "+category[i];
311 for (int i=0;i<3;i++) line+=" par"+(i+3)+" "+average(totals[i],counts[i]);
312 output.add(line);
313 }
314 void execute(String[] t) {
315 switch (t[0]) {
316 case "COURSE": {
317 error(courses.containsKey(t[1]),"duplicate course");
318 int[] par=new int[18],si=new int[18];
319 for (int i=0;i<18;i++) {
320 par[i]=Integer.parseInt(t[i+2]);
321 si[i]=Integer.parseInt(t[i+20]);
322 }
323 error(Arrays.stream(par).anyMatch(x -> x<3 || x>5) || Arrays.stream(si).distinct().count()!=18 || Arrays.stream(si).anyMatch(x -> x<1 || x>18),"invalid course");
324 courses.put(t[1],new Course(t[1],par,si));
325 break;
326 }
327 case "TEE": {
328 Course c=course(t[1]);
329 error(c.tees.containsKey(t[2]),"duplicate tee");
330 int rating=Integer.parseInt(t[3]),slope=Integer.parseInt(t[4]);
331 error(rating<500 || rating>900 || slope<55 || slope>155,"invalid tee");
332 c.tees.put(t[2],new Tee(t[2],rating,slope));
333 break;
334 }
335 case "PLAYER": {
336 error(players.containsKey(t[1]),"duplicate player");
337 Integer i=t.length==4 ? Integer.parseInt(t[3]) : null;
338 error(i!=null && (i< -100 || i>540),"invalid index");
339 players.put(t[1],new Player(t[1],t[2],i));
340 break;
341 }
342 case "ROUND": {
343 Player p=player(t[1]);
344 Course c=course(t[2]);
345 Tee tee=tee(c,t[3]);
346 int day=Integer.parseInt(t[4]);
347 error(!p.rounds.isEmpty() && day<p.rounds.get(p.rounds.size()-1).day,"invalid day");
348 String[] raw=Arrays.copyOfRange(t,5,23);
349 error(Arrays.stream(raw).filter(s -> s.equals("-")).count()>=5,"too few holes");
350 Round r=new Round();
351 r.course=c;
352 r.tee=tee;
353 r.day=day;
354 r.raw=raw;
355 r.oldIndex=p.index;
356 r.oldCalculation=p.calculation;
357 r.oldRecords=p.records.size();
358 r.handicap=p.index==null ? null : courseHandicap(c,tee,p.index);
359 int[] strokes=allocate(c,r.handicap==null ? 0 : r.handicap);
360 r.scores=new int[18];
361 r.adjusted=new int[18];
362 for (int h=0;h<18;h++) {
363 int limit=c.par[h]+(p.index==null ? 5 : 2+strokes[h]);
364 r.scores[h]=raw[h].equals("X") || raw[h].equals("-") ? NO_NET : Integer.parseInt(raw[h]);
365 r.adjusted[h]=raw[h].equals("-") ? c.par[h]+strokes[h] : raw[h].equals("X") ? limit : Math.min(r.scores[h],limit);
366 }
367 r.ags=sum(r.adjusted,0,18);
368 r.diff=away(113L*(10*r.ags-tee.rating()),tee.slope());
369 r.exceptional=p.index!=null && p.index-r.diff>=70 ? (p.index-r.diff>=100 ? 20 : 10) : 0;
370 p.rounds.add(r);
371 update(p,day);
372 if (p.index!=null) p.records.add(new IndexRecord(day,p.index));
373 output.add("round "+p.id+" "+p.rounds.size()+": ags "+r.ags+" diff "+signed(r.diff)+(r.exceptional==0 ? "" : " exceptional "+signed(-r.exceptional))+" index "+index(p.index));
374 break;
375 }
376 case "REMOVE": {
377 Player p=player(t[1]);
378 error(p.rounds.isEmpty(),"no rounds");
379 int number=p.rounds.size();
380 Round r=p.rounds.remove(number-1);
381 p.index=r.oldIndex;
382 p.calculation=r.oldCalculation;
383 p.records.subList(r.oldRecords,p.records.size()).clear();
384 output.add("removed "+p.id+" "+number+": index "+index(p.index));
385 break;
386 }
387 case "INDEX": {
388 Player p=player(t[1]);
389 String line="index "+p.id+": "+index(p.index)+" rounds "+p.rounds.size();
390 Calculation c=p.calculation;
391 if (c!=null) line+=" lowest "+c.k()+" average "+signed(c.average())+" adjustment "+signed(c.adjustment())+
392 (c.low()==null ? "" : " low "+index(c.low()))+(c.hard() ? " hard cap" : c.soft() ? " soft cap" : "")+(c.maximum() ? " maximum" : "");
393 output.add(line);
394 break;
395 }
396 case "HISTORY": {
397 Player p=player(t[1]);
398 Eligible e=eligible(p);
399 output.add("history "+p.id+": rounds "+p.rounds.size());
400 for (int i=0;i<e.adjusted().length;i++) {
401 Round r=p.rounds.get(e.start()+i);
402 output.add((e.start()+i+1)+" day "+r.day+" "+r.course.id+" "+r.tee.id()+" ags "+r.ags+" diff "+signed(r.diff)+
403 (r.diff==e.adjusted()[i] ? "" : " adjusted "+signed(e.adjusted()[i]))+(e.counting().contains(i) ? " *" : ""));
404 }
405 break;
406 }
407 case "LOWINDEX": {
408 Player p=player(t[1]);
409 int day=Integer.parseInt(t[2]);
410 IndexRecord r=low(p,day);
411 output.add("lowindex "+p.id+" "+day+": "+(r==null ? "none" : index(r.value())+" day "+r.day()));
412 break;
413 }
414 case "COURSEHCP": {
415 Player p=player(t[1]);
416 Course c=course(t[2]);
417 Tee tee=tee(c,t[3]);
418 error(p.index==null,"no index");
419 int h=courseHandicap(c,tee,p.index);
420 output.add("coursehcp "+p.id+" "+c.id+" "+tee.id()+": "+handicap(h)+" strokes "+numbers(allocate(c,h)));
421 break;
422 }
423 case "HCPTABLE": {
424 Course c=course(t[1]);
425 Tee tee=tee(c,t[2]);
426 int first=Integer.parseInt(t[3]),last=Integer.parseInt(t[4]);
427 error(first< -100 || first>last || last>540,"invalid index");
428 output.add("hcptable "+c.id+" "+tee.id()+": "+index(first)+" to "+index(last));
429 int start=first,h=courseHandicap(c,tee,first);
430 for (int i=first+1;i<=last+1;i++) {
431 int next=i<=last ? courseHandicap(c,tee,i) : h+1;
432 if (next!=h) {
433 output.add(index(start)+" to "+index(i-1)+": "+handicap(h));
434 start=i;
435 h=next;
436 }
437 }
438 break;
439 }
440 case "CARD": {
441 Player p=player(t[1]);
442 int n=Integer.parseInt(t[2]);
443 error(n>p.rounds.size(),"unknown round");
444 card(p,n,p.rounds.get(n-1));
445 break;
446 }
447 case "STATS":
448 statistics(player(t[1]));
449 break;
450 case "HOLESTATS": {
451 Course c=course(t[1]);
452 List<Round> rounds=players.values().stream().flatMap(p -> p.rounds.stream()).filter(r -> r.course==c).toList();
453 int[] totals=new int[18],counts=new int[18],ranks=new int[18];
454 for (Round r:rounds) {
455 for (int h=0;h<18;h++) {
456 if (r.scores[h]!=NO_NET) {
457 totals[h]+=r.scores[h];
458 counts[h]++;
459 }
460 }
461 }
462 List<Integer> order=all(18);
463 order.sort((a,b) -> {
464 if (counts[a]==0 || counts[b]==0) return counts[a]==counts[b] ? Integer.compare(a,b) : counts[a]==0 ? 1 : -1;
465 int cmp=Long.compare((long)(totals[b]-c.par[b]*counts[b])*counts[a],(long)(totals[a]-c.par[a]*counts[a])*counts[b]);
466 return cmp==0 ? Integer.compare(a,b) : cmp;
467 });
468 for (int i=0;i<18;i++) ranks[order.get(i)]=i+1;
469 output.add("holestats "+c.id+": rounds "+rounds.size());
470 for (int h=0;h<18;h++) output.add((h+1)+" par "+c.par[h]+" si "+c.si[h]+" scores "+counts[h]+" avg "+average(totals[h],counts[h])+" rank "+ranks[h]);
471 break;
472 }
473 case "PLAYERS": {
474 List<Player> order=new ArrayList<>(players.values());
475 order.sort(Comparator.comparing((Player p) -> p.index,Comparator.nullsLast(Comparator.naturalOrder())).thenComparing(p -> p.id));
476 output.add("players "+order.size());
477 for (Player p:order) output.add(p.id+" "+p.name+" "+index(p.index)+" rounds "+p.rounds.size());
478 break;
479 }
480 case "EVENT": {
481 error(events.containsKey(t[1]),"duplicate event");
482 Course c=course(t[3]);
483 Tee tee=tee(c,t[4]);
484 int allowance=Integer.parseInt(t[6]);
485 error(allowance<1 || allowance>100,"invalid allowance");
486 List<Player> list=Arrays.stream(t,7,t.length).map(this::player).toList();
487 int n=list.size();
488 boolean valid=switch(t[2]) { case "MATCH","NASSAU" -> n==2; case "FOURBALL" -> n==4; case "BETTERBALL" -> n==4 || n==6 || n==8; default -> n>=2 && n<=8; };
489 error(new HashSet<>(list).size()!=n || !valid,"invalid players");
490 events.put(t[1],new Event(t[1],t[2],c,tee,Integer.parseInt(t[5]),allowance,list));
491 break;
492 }
493 case "RESULTS": {
494 Event e=events.get(t[1]);
495 error(e==null,"unknown event");
496 EventData d=data(e);
497 results(e,d);
498 break;
499 }
500 case "MERIT": {
501 Map<String,int[]> merit=new HashMap<>();
502 int count=0;
503 int[] awards={10,8,6,5,4,3,2,1};
504 for (Event e:events.values()) {
505 if (!List.of("STROKE","STABLEFORD","PAR").contains(e.format())) continue;
506 EventData d;
507 try { d=data(e); }
508 catch (Fault f) { continue; }
509 count++;
510 for (Player p:e.players()) merit.computeIfAbsent(p.id,id -> new int[3])[2]++;
511 List<Integer> order=ranked(d);
512 for (int i=0;i<order.size();i++) {
513 String id=e.players().get(order.get(i)).id;
514 merit.get(id)[0]+=awards[i];
515 if (i==0) merit.get(id)[1]++;
516 }
517 }
518 List<String> order=new ArrayList<>(merit.keySet());
519 order.sort(Comparator.comparingInt((String id) -> -merit.get(id)[0]).thenComparingInt(id -> -merit.get(id)[1]).thenComparing(Comparator.naturalOrder()));
520 output.add("merit: events "+count);
521 for (int i=0;i<order.size();i++) {
522 String id=order.get(i);
523 int[] a=merit.get(id);
524 output.add((i+1)+" "+id+" "+a[0]+" wins "+a[1]+" played "+a[2]);
525 }
526 break;
527 }
528 }
529 }
530 List<String> process(String line) {
531 output=new ArrayList<>();
532 try { execute(line.split(" ")); }
533 catch (Fault f) { output.add("ERROR "+f.getMessage()); }
534 return output;
535 }
