29 return n / d

81func strokes(handicap int, si []int) []int {
82 absolute := handicap
83 if absolute < 0 {
84 absolute = -absolute
85 }
86 q, r := absolute/18, absolute%18
87 result := make([]int, len(si))
88 for hole, rank := range si {
89 if handicap >= 0 {
90 result[hole] = q
91 if rank <= r {
92 result[hole]++
93 }
94 } else {
95 result[hole] = -q
96 if rank >= 19-r {
97 result[hole]--
98 }
99 }
100 }
101 return result
102}

104func countback(values []int) [5]int {
105 return [5]int{
106 total(values),
107 total(values[9:]),
108 total(values[12:]),
109 total(values[15:]),
110 values[17],
111 }
112}

114func ranked(indices []int, values [][]int, ids []string, larger bool) []int {
115 result := append([]int(nil), indices...)
116 sort.Slice(result, func(i, j int) bool {
117 left, right := countback(values[result[i]]), countback(values[result[j]])
118 for k := range left {
119 if left[k] != right[k] {
120 if larger {
121 return left[k] > right[k]
122 }
123 return left[k] < right[k]
124 }
125 }
126 return ids[result[i]] < ids[result[j]]
127 })
128 return result
129}

175func (p *Player) lowIndex(day int) (IndexRecord, bool) {
176 var result IndexRecord
177 found := false
178 for _, record := range p.Records {
179 if record.Day < day-365 || record.Day >= day {
180 continue
181 }
182 if !found || record.Value < result.Value || (record.Value == result.Value && record.Day < result.Day) {
183 result = record
184 found = true
185 }
186 }
187 return result, found
188}

190type Eligible struct {
191 Number, Adjusted int
192 Round *Round
193}

195func (p *Player) eligible() []Eligible {
196 first := len(p.Rounds) - 20
197 if first < 0 {
198 first = 0
199 }
200 rows := make([]Eligible, 0, len(p.Rounds)-first)
201 reduction := 0
202 for i := len(p.Rounds) - 1; i >= first; i-- {
203 record := p.Rounds[i]
204 reduction += record.Exceptional
205 rows = append(rows, Eligible{i + 1, record.Differential - reduction, record})
206 }
207 for i, j := 0, len(rows)-1; i < j; i, j = i+1, j-1 {
208 rows[i], rows[j] = rows[j], rows[i]
209 }
210 return rows
211}

213func counting(rows []Eligible) ([]Eligible, int) {
214 n := len(rows)
215 if n < 3 {
216 return nil, 0
217 }
218 k := 1
219 switch {
220 case n <= 5:
221 k = 1
222 case n <= 8:
223 k = 2
224 case n <= 11:
225 k = 3
226 case n <= 14:
227 k = 4
228 case n <= 16:
229 k = 5
230 case n <= 18:
231 k = 6
232 case n == 19:
233 k = 7
234 default:
235 k = 8
236 }
237 adjustment := 0
238 if n == 3 {
239 adjustment = -20
240 } else if n == 4 || n == 6 {
241 adjustment = -10
242 }
243 ordered := append([]Eligible(nil), rows...)
244 sort.Slice(ordered, func(i, j int) bool {
245 if ordered[i].Adjusted != ordered[j].Adjusted {
246 return ordered[i].Adjusted < ordered[j].Adjusted
247 }
248 return ordered[i].Number > ordered[j].Number
249 })
250 return ordered[:k], adjustment
251}

253func (p *Player) updateIndex(day int) {
254 selected, adjustment := counting(p.eligible())
255 if len(selected) == 0 {
256 p.Calculation = nil
257 return
258 }
259 sum := 0
260 for _, row := range selected {
261 sum += row.Adjusted
262 }
263 avg := halfAway(sum, len(selected))
264 candidate := avg + adjustment
265 lowRecord, hasLow := p.lowIndex(day)
266 var low *int
267 soft, hard, maximum := false, false, false
268 if hasLow {
269 low = pointer(lowRecord.Value)
270 if candidate-*low > 30 {
271 candidate = *low + 30 + (candidate-*low-30)/2
272 soft = true
273 }
274 if candidate-*low > 50 {
275 candidate = *low + 50
276 hard = true
277 }
278 }
279 if candidate > 540 {
280 candidate = 540
281 maximum = true
282 }
283 p.Index = pointer(candidate)
284 p.Calculation = &Calculation{len(selected), avg, adjustment, low, hard, soft, maximum}
285}

299func newRound(p *Player, course *Course, tee *Tee, day int, scores []int) *Round {
300 r := &Round{
301 Course: course, Tee: tee, Day: day, Scores: scores,
302 Previous: p.Index, PreviousRecords: len(p.Records),
303 PreviousCalculation: p.Calculation,
304 }
305 if p.Index != nil {
306 r.Handicap = pointer(tee.handicap(*p.Index, course))
307 }
308 hcp := 0
309 if r.Handicap != nil {
310 hcp = *r.Handicap
311 }
312 allocation := strokes(hcp, course.SI)
313 r.Adjusted = make([]int, 18)
314 for hole, score := range scores {
315 par, stroke := course.Pars[hole], allocation[hole]
316 limit := par + 5
317 if r.Handicap != nil {
318 limit = par + 2 + stroke
319 }
320 switch {
321 case score == 0:
322 r.Adjusted[hole] = par + stroke
323 case score < 0:
324 r.Adjusted[hole] = limit
325 case score > limit:
326 r.Adjusted[hole] = limit
327 default:
328 r.Adjusted[hole] = score
329 }
330 }
331 r.AGS = total(r.Adjusted)
332 r.Differential = halfAway(113*(10*r.AGS-tee.Rating), tee.Slope)
333 if p.Index != nil && *p.Index-r.Differential >= 70 {
334 r.Exceptional = 10
335 if *p.Index-r.Differential >= 100 {
336 r.Exceptional = 20
337 }
338 }
339 return r
340}

351func (r *Round) nets(allocation []int) []int {
352 result := make([]int, 18)
353 for hole, score := range r.Scores {
354 if score <= 0 {
355 result[hole] = noNet
356 } else {
357 result[hole] = score - allocation[hole]
358 }
359 }
360 return result
361}

363func (r *Round) points(nets []int) []int {
364 result := make([]int, 18)
365 for hole, net := range nets {
366 if net == noNet {
367 continue
368 }
369 value := r.Course.Pars[hole] + 2 - net
370 if value > 0 {
371 result[hole] = value
372 }
373 }
374 return result

444func (g *Golf) eventData(event *Event) (*EventData, string) {
445 d := &EventData{
446 Rounds: make([]*Round, len(event.Players)),
447 Handicaps: make([]int, len(event.Players)),
448 Nets: make([][]int, len(event.Players)),
449 Points: make([][]int, len(event.Players)),
450 }
451 for i, p := range event.Players {
452 var found *Round
453 for _, r := range p.Rounds {
454 if r.Course == event.Course && r.Tee == event.Tee && r.Day == event.Day {
455 found = r
456 break
457 }
458 }
459 if found == nil {
460 return nil, "missing round"
461 }
462 if found.Handicap == nil {
463 return nil, "no index"
464 }
465 d.Rounds[i] = found
466 d.Handicaps[i] = halfUp(*found.Handicap*event.Allowance, 100)
467 }
468 minimum := 0
469 if event.Format == "MATCH" || event.Format == "NASSAU" || event.Format == "FOURBALL" {
470 minimum = d.Handicaps[0]
471 for _, hcp := range d.Handicaps[1:] {
472 if hcp < minimum {
473 minimum = hcp
474 }
475 }
476 }
477 for i, r := range d.Rounds {
478 d.Nets[i] = r.nets(strokes(d.Handicaps[i]-minimum, event.Course.SI))
479 d.Points[i] = r.points(d.Nets[i])
480 }
481 return d, ""
482}

484func (g *Golf) individualRanking(event *Event, data *EventData) ([]int, [][]int) {
485 n := len(event.Players)
486 ids := make([]string, n)
487 positions := make([]int, 0, n)
488 values := make([][]int, n)
489 for i, p := range event.Players {
490 ids[i] = p.ID
491 if event.Format == "STROKE" {
492 if data.Rounds[i].complete() {
493 positions = append(positions, i)
494 }
495 values[i] = data.Nets[i]
496 } else if event.Format == "STABLEFORD" {
497 positions = append(positions, i)
498 values[i] = data.Points[i]
499 } else {
500 positions = append(positions, i)
501 values[i] = make([]int, 18)
502 for hole, net := range data.Nets[i] {
503 if net == noNet || net > event.Course.Pars[hole] {
504 values[i][hole] = -1
505 } else if net < event.Course.Pars[hole] {
506 values[i][hole] = 1
507 }
508 }
509 }
510 }
511 return ranked(positions, values, ids, event.Format != "STROKE"), values
512}

514func (g *Golf) individualResults(event *Event, data *EventData) {
515 order, values := g.individualRanking(event, data)
516 for rank, i := range order {
517 p := event.Players[i]
518 hcp := handicapText(pointer(data.Handicaps[i]))
519 switch event.Format {
520 case "STROKE":
521 g.emit(rank+1, p.ID, "net", total(data.Nets[i]), "gross", total(data.Rounds[i].Scores), "hcp", hcp)
522 case "STABLEFORD":
523 g.emit(rank+1, p.ID, "points", total(data.Points[i]), "hcp", hcp)
524 case "PAR":
525 g.emit(rank+1, p.ID, matchStatus(total(values[i])), "hcp", hcp)
526 }
527 }
528 if event.Format != "STROKE" {
529 return
530 }
531 nonreturns := make([]int, 0)
532 for i, r := range data.Rounds {
533 if !r.complete() {
534 nonreturns = append(nonreturns, i)
535 }
536 }
537 sort.Slice(nonreturns, func(i, j int) bool {
538 return event.Players[nonreturns[i]].ID < event.Players[nonreturns[j]].ID
539 })
540 for _, i := range nonreturns {
541 g.emit("NR", event.Players[i].ID, "hcp", handicapText(pointer(data.Handicaps[i])))
542 }
543 if len(order) > 0 {
544 gross := make([][]int, len(event.Players))
545 ids := make([]string, len(event.Players))
546 for i, p := range event.Players {
547 ids[i] = p.ID
548 if data.Rounds[i].complete() {
549 gross[i] = data.Rounds[i].Scores
550 }
551 }
552 winner := ranked(order, gross, ids, false)[0]
553 g.emit("gross", event.Players[winner].ID, total(gross[winner]))
554 }
555}

557func (g *Golf) matchHoles(event *Event, data *EventData) ([2]string, []int) {
558 var sides [2][]int
559 if event.Format == "FOURBALL" {
560 sides[0], sides[1] = []int{0, 1}, []int{2, 3}
561 } else {
562 sides[0], sides[1] = []int{0}, []int{1}
563 }
564 var names [2]string
565 for side := range sides {
566 ids := make([]string, len(sides[side]))
567 for i, index := range sides[side] {
568 ids[i] = event.Players[index].ID
569 }
570 names[side] = strings.Join(ids, "+")
571 }
572 holes := make([]int, 18)
573 for h := range holes {
574 var scores [2]int
575 for side := range sides {
576 scores[side] = noNet
577 for _, index := range sides[side] {
578 net := data.Nets[index][h]
579 if net < scores[side] {
580 scores[side] = net
581 }
582 }
583 }
584 if scores[0] == scores[1] {
585 holes[h] = 0
586 } else if scores[0] < scores[1] {
587 holes[h] = 1
588 } else {
589 holes[h] = -1
590 }
591 }
592 return names, holes
593}

595func playMatch(names [2]string, holes []int, first, last int) ([]string, string) {
596 lead := 0
597 statuses := []string{}
598 remaining := 0
599 for hole := first; hole < last; hole++ {
600 lead += holes[hole]
601 statuses = append(statuses, matchStatus(lead))
602 remaining = last - hole - 1
603 if abs(lead) > remaining || remaining == 0 {
604 break
605 }
606 }
607 if lead == 0 {
608 return statuses, "AS"
609 }
610 winner := names[0]
611 if lead < 0 {
612 winner = names[1]
613 }
614 if remaining > 0 {
615 return statuses, fmt.Sprintf("%s %d&%d", winner, abs(lead), remaining)
616 }
617 return statuses, fmt.Sprintf("%s %d UP", winner, abs(lead))
618}

627func (g *Golf) matchResults(event *Event, data *EventData) {
628 minimum := data.Handicaps[0]
629 for _, hcp := range data.Handicaps {
630 if hcp < minimum {
631 minimum = hcp
632 }
633 }
634 parts := []interface{}{"strokes"}
635 for i, p := range event.Players {
636 parts = append(parts, p.ID, data.Handicaps[i]-minimum)
637 }
638 g.emit(parts...)
639 names, holes := g.matchHoles(event, data)
640 if event.Format == "NASSAU" {
641 for _, section := range []struct {
642 Name string
643 First, Last int
644 }{{"front", 0, 9}, {"back", 9, 18}, {"total", 0, 18}} {
645 _, result := playMatch(names, holes, section.First, section.Last)
646 g.emit(section.Name, result)
647 }
648 } else {
649 statuses, result := playMatch(names, holes, 0, 18)
650 g.emit("status", strings.Join(statuses, ", "))
651 g.emit("result", result)
652 }
653}

655func (g *Golf) betterballResults(event *Event, data *EventData) {
656 type Team struct {
657 First, Second int
658 Values []int
659 }
660 teams := []Team{}
661 for i := 0; i < len(event.Players); i += 2 {
662 values := make([]int, 18)
663 for hole := range values {
664 values[hole] = data.Points[i][hole]
665 if data.Points[i+1][hole] > values[hole] {
666 values[hole] = data.Points[i+1][hole]
667 }
668 }
669 teams = append(teams, Team{i, i + 1, values})
670 }
671 ids := make([]string, len(teams))
672 values := make([][]int, len(teams))
673 positions := make([]int, len(teams))
674 for i, team := range teams {
675 ids[i], values[i], positions[i] = event.Players[team.First].ID, team.Values, i
676 }
677 for rank, index := range ranked(positions, values, ids, true) {
678 team := teams[index]
679 g.emit(rank+1, event.Players[team.First].ID+"+"+event.Players[team.Second].ID,
680 "points", total(team.Values), "hcp",
681 handicapText(pointer(data.Handicaps[team.First])),
682 handicapText(pointer(data.Handicaps[team.Second])))
683 }
684}

686func (g *Golf) skinsResults(event *Event, data *EventData) {
687 totals := make([]int, len(event.Players))
688 carry := 0
689 var lastWinners []int
690 for hole := 0; hole < 18; hole++ {
691 carry++
692 best := noNet
693 winners := []int{}
694 for i, nets := range data.Nets {
695 if nets[hole] == noNet {
696 continue
697 }
698 if nets[hole] < best {
699 best, winners = nets[hole], []int{i}
700 } else if nets[hole] == best {
701 winners = append(winners, i)
702 }
703 }
704 lastWinners = winners
705 if len(winners) == 1 {
706 g.emit("hole", hole+1, event.Players[winners[0]].ID, carry)
707 totals[winners[0]] += 100 * carry
708 carry = 0
709 }
710 }
711 if carry > 0 {
712 sharing := lastWinners
713 if len(sharing) == 0 {
714 sharing = make([]int, len(event.Players))
715 for i := range sharing {
716 sharing[i] = i
717 }
718 }
719 sort.Slice(sharing, func(i, j int) bool {
720 return event.Players[sharing[i]].ID < event.Players[sharing[j]].ID
721 })
722 parts := []interface{}{"carry", carry}
723 for _, i := range sharing {
724 totals[i] += carry * 100 / len(sharing)
725 parts = append(parts, event.Players[i].ID)
726 }
727 g.emit(parts...)
728 }
729 positions := make([]int, len(event.Players))
730 for i := range positions {
731 positions[i] = i
732 }
733 sort.Slice(positions, func(i, j int) bool {
734 a, b := positions[i], positions[j]
735 if totals[a] != totals[b] {
736 return totals[a] > totals[b]
737 }
738 return event.Players[a].ID < event.Players[b].ID
739 })
740 for _, i := range positions {
741 g.emit(event.Players[i].ID, fmt.Sprintf("%d.%02d", totals[i]/100, totals[i]%100))
742 }
743}

760func (g *Golf) merit() {
761 type Standing struct {
762 ID string
763 Points, Wins, Appearances int
764 }
765 standings := make(map[string]*Standing)
766 counted := 0
767 awards := []int{10, 8, 6, 5, 4, 3, 2, 1}
768 for _, event := range g.EventOrder {
769 if event.Format != "STROKE" && event.Format != "STABLEFORD" && event.Format != "PAR" {
770 continue
771 }
772 data, err := g.eventData(event)
773 if err != "" {
774 continue
775 }
776 counted++
777 order, _ := g.individualRanking(event, data)
778 for _, p := range event.Players {
779 if standings[p.ID] == nil {
780 standings[p.ID] = &Standing{ID: p.ID}
781 }
782 standings[p.ID].Appearances++
783 }
784 for rank, i := range order {
785 standing := standings[event.Players[i].ID]
786 standing.Points += awards[rank]
787 if rank == 0 {
788 standing.Wins++
789 }
790 }
791 }
792 g.emit("merit:", "events", counted)
793 sorted := make([]*Standing, 0, len(standings))
794 for _, entry := range standings {
795 sorted = append(sorted, entry)
796 }
797 sort.Slice(sorted, func(i, j int) bool {
798 a, b := sorted[i], sorted[j]
799 if a.Points != b.Points {
800 return a.Points > b.Points
801 }
802 if a.Wins != b.Wins {
803 return a.Wins > b.Wins
804 }
805 return a.ID < b.ID
806 })
807 for rank, entry := range sorted {
808 g.emit(rank+1, entry.ID, entry.Points, "wins", entry.Wins, "played", entry.Appearances)
809 }
810}

812func (g *Golf) cardTotal(label string, r *Round, allocation, nets, points []int, first, last int) {
813 complete, scores, netTotal := true, 0, 0
814 for hole := first; hole < last; hole++ {
815 if r.Scores[hole] <= 0 {
816 complete = false
817 } else {
818 scores += r.Scores[hole]
819 netTotal += nets[hole]
820 }
821 }
822 scoreText, netText := "-", "-"
823 if complete {
824 scoreText, netText = strconv.Itoa(scores), strconv.Itoa(netTotal)
825 }
826 g.emit(label, total(r.Course.Pars[first:last]), scoreText,
827 total(r.Adjusted[first:last]), total(allocation[first:last]),
828 netText, total(points[first:last]))
829}

831func (g *Golf) card(p *Player, number int) {
832 r := p.Rounds[number-1]
833 hcp := 0
834 if r.Handicap != nil {
835 hcp = *r.Handicap
836 }
837 allocation := strokes(hcp, r.Course.SI)
838 nets, points := r.nets(allocation), r.points(r.nets(allocation))
839 g.emit("card", p.ID, fmt.Sprint(number, ":"), r.Course.ID, r.Tee.ID,
840 "day", r.Day, "handicap", handicapText(r.Handicap))
841 for hole, score := range r.Scores {
842 scoreText, netText := strconv.Itoa(score), "-"
843 if score == 0 {
844 scoreText = "-"
845 } else if score < 0 {
846 scoreText = "X"
847 } else {
848 netText = strconv.Itoa(nets[hole])
849 }
850 g.emit(hole+1, r.Course.Pars[hole], r.Course.SI[hole], scoreText,
851 r.Adjusted[hole], allocation[hole], netText, points[hole])
852 if hole == 8 {
853 g.cardTotal("OUT", r, allocation, nets, points, 0, 9)
854 } else if hole == 17 {
855 g.cardTotal("IN", r, allocation, nets, points, 9, 18)
856 g.cardTotal("TOTAL", r, allocation, nets, points, 0, 18)
857 }
858 }
859 g.emit("ags", r.AGS, "diff", decimal(r.Differential, "-"))
860}

862func (g *Golf) stats(p *Player) {
863 labels := []string{"eagles", "birdies", "pars", "bogeys", "doubles", "others", "pickups"}
864 categories := make([]int, len(labels))
865 parTotal := [3]int{}
866 parCount := [3]int{}
867 complete := []int{}
868 for _, r := range p.Rounds {
869 if r.complete() {
870 complete = append(complete, total(r.Scores))
871 }
872 for hole, score := range r.Scores {
873 par := r.Course.Pars[hole]
874 if score == -1 {
875 categories[6]++
876 } else if score > 0 {
877 bucket := score - par + 2
878 if bucket < 0 {
879 bucket = 0
880 }
881 if bucket > 5 {
882 bucket = 5
883 }
884 categories[bucket]++
885 parTotal[par-3] += score
886 parCount[par-3]++
887 }
888 }
889 }
890 best := "-"
891 if len(complete) != 0 {
892 sort.Ints(complete)
893 best = strconv.Itoa(complete[0])
894 }
895 parts := []interface{}{"stats", p.ID+":", "rounds", len(p.Rounds),
896 "best", best, "average", average(total(complete), len(complete))}
897 for i, label := range labels {
898 parts = append(parts, label, categories[i])
899 }
900 for i := 0; i < 3; i++ {
901 parts = append(parts, fmt.Sprint("par", i+3), average(parTotal[i], parCount[i]))
902 }
903 g.emit(parts...)
904}

906func (g *Golf) holeStats(course *Course) {
907 sums, counts := [18]int{}, [18]int{}
908 rounds := 0
909 for _, p := range g.Players {
910 for _, r := range p.Rounds {
911 if r.Course != course {
912 continue
913 }
914 rounds++
915 for h, score := range r.Scores {
916 if score > 0 {
917 sums[h] += score
918 counts[h]++
919 }
920 }
921 }
922 }
923 order := make([]int, 18)
924 ranks := make([]int, 18)
925 for h := range order {
926 order[h] = h
927 }
928 sort.Slice(order, func(i, j int) bool {
929 a, b := order[i], order[j]
930 if counts[a] == 0 || counts[b] == 0 {
931 if counts[a] == 0 && counts[b] == 0 {
932 return a < b
933 }
934 return counts[a] != 0
935 }
936 ar := (sums[a] - course.Pars[a]*counts[a]) * counts[b]
937 br := (sums[b] - course.Pars[b]*counts[b]) * counts[a]
938 if ar != br {
939 return ar > br
940 }
941 return a < b
942 })
943 for rank, hole := range order {
944 ranks[hole] = rank + 1
945 }
946 g.emit("holestats", course.ID+":", "rounds", rounds)
947 for hole := 0; hole < 18; hole++ {
948 g.emit(hole+1, "par", course.Pars[hole], "si", course.SI[hole],
949 "scores", counts[hole], "avg", average(sums[hole], counts[hole]), "rank", ranks[hole])
950 }
951}

953func (g *Golf) playersList() {
954 players := make([]*Player, 0, len(g.Players))
955 for _, p := range g.Players {
956 players = append(players, p)
957 }
958 sort.Slice(players, func(i, j int) bool {
959 a, b := players[i], players[j]
960 if a.Index == nil || b.Index == nil {
961 if a.Index == nil && b.Index == nil {
962 return a.ID < b.ID
963 }
964 return a.Index != nil
965 }
966 if *a.Index != *b.Index {
967 return *a.Index < *b.Index
968 }
969 return a.ID < b.ID
970 })
971 g.emit("players", len(players))
972 for _, p := range players {
973 g.emit(p.ID, p.Name, indexText(p.Index), "rounds", len(p.Rounds))
974 }
975}

977func (g *Golf) dispatch(command string, a []string) string {
978 switch command {
979 case "COURSE":
980 id := a[0]
981 if g.Courses[id] != nil {
982 return "duplicate course"
983 }
984 pars, si := make([]int, 18), make([]int, 18)
985 seen := make(map[int]bool)
986 for i := 0; i < 18; i++ {
987 pars[i], si[i] = number(a[i+1]), number(a[i+19])
988 if pars[i] < 3 || pars[i] > 5 || si[i] < 1 || si[i] > 18 || seen[si[i]] {
989 return "invalid course"
990 }
991 seen[si[i]] = true
992 }
993 g.Courses[id] = &Course{id, pars, si, total(pars), make(map[string]*Tee)}
994 case "TEE":
995 c, err := g.course(a[0])
996 if err != "" {
997 return err
998 }
999 if c.Tees[a[1]] != nil {
1000 return "duplicate tee"
1001 }
1002 rating, slope := number(a[2]), number(a[3])
1003 if rating < 500 || rating > 900 || slope < 55 || slope > 155 {
1004 return "invalid tee"
1005 }
1006 c.Tees[a[1]] = &Tee{a[1], rating, slope}
1007 case "PLAYER":
1008 if g.Players[a[0]] != nil {
1009 return "duplicate player"
1010 }
1011 var index *int
1012 if len(a) == 3 {
1013 n := number(a[2])
1014 if n < -100 || n > 540 {
1015 return "invalid index"
1016 }
1017 index = pointer(n)
1018 }
1019 g.Players[a[0]] = &Player{ID: a[0], Name: a[1], Index: index}
1020 case "EVENT":
1021 id, form := a[0], a[1]
1022 if g.Events[id] != nil {
1023 return "duplicate event"
1024 }
1025 c, t, err := g.tee(a[2], a[3])
1026 if err != "" {
1027 return err
1028 }
1029 allowance := number(a[5])
1030 if allowance < 1 || allowance > 100 {
1031 return "invalid allowance"
1032 }
1033 players := make([]*Player, 0, len(a)-6)
1034 seen := make(map[string]bool)
1035 duplicate := false
1036 for _, pid := range a[6:] {
1037 if g.Players[pid] == nil {
1038 return "unknown player"
1039 }
1040 if seen[pid] {
1041 duplicate = true
1042 }
1043 seen[pid] = true
1044 players = append(players, g.Players[pid])
1045 }
1046 n := len(players)
1047 valid := n >= 2 && n <= 8
1048 switch form {
1049 case "MATCH", "NASSAU":
1050 valid = n == 2
1051 case "FOURBALL":
1052 valid = n == 4
1053 case "BETTERBALL":
1054 valid = n == 4 || n == 6 || n == 8
1055 }
1056 if duplicate || !valid {
1057 return "invalid players"
1058 }
1059 event := &Event{id, form, c, t, number(a[4]), allowance, players}
1060 g.Events[id] = event
1061 g.EventOrder = append(g.EventOrder, event)
1062 case "RESULTS":
1063 event := g.Events[a[0]]
1064 if event == nil {
1065 return "unknown event"
1066 }
1067 data, err := g.eventData(event)
1068 if err != "" {
1069 return err
1070 }
1071 g.results(event, data)
1072 case "MERIT":
1073 g.merit()
1074 case "PLAYERS":
1075 g.playersList()
1076 case "HOLESTATS":
1077 c, err := g.course(a[0])
1078 if err != "" {
1079 return err
1080 }
1081 g.holeStats(c)
1082 case "HCPTABLE":
1083 c, t, err := g.tee(a[0], a[1])
1084 if err != "" {
1085 return err
1086 }
1087 first, last := number(a[2]), number(a[3])
1088 if first < -100 || first > last || last > 540 {
1089 return "invalid index"
1090 }
1091 g.emit("hcptable", c.ID, t.ID+":", indexText(pointer(first)), "to", indexText(pointer(last)))
1092 begin := first
1093 hcp := t.handicap(first, c)
1094 for i := first+1; i <= last; i++ {
1095 next := t.handicap(i, c)
1096 if next != hcp {
1097 g.emit(indexText(pointer(begin)), "to", indexText(pointer(i-1))+":", handicapText(pointer(hcp)))
1098 begin, hcp = i, next
1099 }
1100 }
1101 g.emit(indexText(pointer(begin)), "to", indexText(pointer(last))+":", handicapText(pointer(hcp)))
1102 case "REMOVE", "INDEX", "HISTORY", "LOWINDEX", "COURSEHCP", "ROUND", "CARD", "STATS":
1103 p, err := g.player(a[0])
1104 if err != "" {
1105 return err
1106 }
1107 switch command {
1108 case "COURSEHCP", "ROUND":
1109 c, t, err := g.tee(a[1], a[2])
1110 if err != "" {
1111 return err
1112 }
1113 if command == "COURSEHCP" {
1114 if p.Index == nil {
1115 return "no index"
1116 }
1117 hcp := t.handicap(*p.Index, c)
1118 parts := []interface{}{"coursehcp", p.ID, c.ID, t.ID+":", handicapText(pointer(hcp)), "strokes"}
1119 for _, stroke := range strokes(hcp, c.SI) {
1120 parts = append(parts, stroke)
1121 }
1122 g.emit(parts...)
1123 } else {
1124 day := number(a[3])
1125 if len(p.Rounds) != 0 && day < p.Rounds[len(p.Rounds)-1].Day {
1126 return "invalid day"
1127 }
1128 scores, missing := make([]int, 18), 0
1129 for i, text := range a[4:] {
1130 if text == "-" {
1131 missing++
1132 } else if text == "X" {
1133 scores[i] = -1
1134 } else {
1135 scores[i] = number(text)
1136 }
1137 }
1138 if missing >= 5 {
1139 return "too few holes"
1140 }
1141 r := newRound(p, c, t, day, scores)
1142 p.Rounds = append(p.Rounds, r)
1143 p.updateIndex(day)
1144 if p.Index != nil {
1145 p.Records = append(p.Records, IndexRecord{day, *p.Index})
1146 }
1147 parts := []interface{}{"round", p.ID, fmt.Sprint(len(p.Rounds), ":"),
1148 "ags", r.AGS, "diff", decimal(r.Differential, "-")}
1149 if r.Exceptional != 0 {
1150 parts = append(parts, "exceptional", decimal(-r.Exceptional, "-"))
1151 }
1152 parts = append(parts, "index", indexText(p.Index))
1153 g.emit(parts...)
1154 }
1155 case "REMOVE":
1156 if len(p.Rounds) == 0 {
1157 return "no rounds"
1158 }
1159 n := len(p.Rounds)
1160 r := p.Rounds[n-1]
1161 p.Rounds = p.Rounds[:n-1]
1162 p.Index = r.Previous
1163 p.Records = p.Records[:r.PreviousRecords]
1164 p.Calculation = r.PreviousCalculation
1165 g.emit("removed", p.ID, fmt.Sprint(n, ":"), "index", indexText(p.Index))
1166 case "INDEX":
1167 parts := []interface{}{"index", p.ID+":", indexText(p.Index), "rounds", len(p.Rounds)}
1168 if c := p.Calculation; c != nil {
1169 parts = append(parts, "lowest", c.K, "average", decimal(c.Average, "-"),
1170 "adjustment", decimal(c.Adjustment, "-"))
1171 if c.Low != nil {
1172 parts = append(parts, "low", indexText(c.Low))
1173 }
1174 if c.Hard {
1175 parts = append(parts, "hard", "cap")
1176 } else if c.Soft {
1177 parts = append(parts, "soft", "cap")
1178 }
1179 if c.Maximum {
1180 parts = append(parts, "maximum")
1181 }
1182 }
1183 g.emit(parts...)
1184 case "HISTORY":
1185 rows := p.eligible()
1186 selected, _ := counting(rows)
1187 marked := make(map[int]bool)
1188 for _, row := range selected {
1189 marked[row.Number] = true
1190 }
1191 g.emit("history", p.ID+":", "rounds", len(p.Rounds))
1192 for _, row := range rows {
1193 r := row.Round
1194 parts := []interface{}{row.Number, "day", r.Day, r.Course.ID, r.Tee.ID,
1195 "ags", r.AGS, "diff", decimal(r.Differential, "-")}
1196 if row.Adjusted != r.Differential {
1197 parts = append(parts, "adjusted", decimal(row.Adjusted, "-"))
1198 }
1199 if marked[row.Number] {
1200 parts = append(parts, "*")
1201 }
1202 g.emit(parts...)
1203 }
1204 case "LOWINDEX":
1205 day := number(a[1])
1206 record, ok := p.lowIndex(day)
1207 if !ok {
1208 g.emit("lowindex", p.ID, fmt.Sprint(day, ":"), "none")
1209 } else {
1210 g.emit("lowindex", p.ID, fmt.Sprint(day, ":"), indexText(pointer(record.Value)), "day", record.Day)
1211 }
1212 case "CARD":
1213 n := number(a[1])
1214 if n > len(p.Rounds) {
1215 return "unknown round"
1216 }
1217 g.card(p, n)
1218 case "STATS":
1219 g.stats(p)
1220 }
1221 }
1222 return ""
1223}

1225func (g *Golf) execute(line string) {

1228 return
1229 }
1230 if err := g.dispatch(words[0], words[1:]); err != "" {
1231 g.emit("ERROR", err)
1232 }
1233}
