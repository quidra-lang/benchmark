94fn allocation(h: i64, c: *Course) [18]i64 {
95 var strokes: [18]i64 = undefined;
96 for (c.si, 0..) |si, i| {
97 strokes[i] = if (h >= 0) @divTrunc(h, 18) + @as(i64, if (si <= @mod(h, 18)) 1 else 0) else -@divTrunc(-h, 18) - @as(i64, if (si >= 19 - @mod(-h, 18)) 1 else 0);
98 }
99 return strokes;
100}
101fn handicap(index: i64, c: *Course, t: *Tee) i64 {
102 return halfUp(index * t.slope + 113 * (t.rating - 10 * c.par), 1130);
103}
104fn points(net: ?i64, par: i64) i64 {
105 return if (net) |n| @max(0, par + 2 - n) else 0;
106}
107fn rule(n: usize) struct { k: usize, adjustment: i64 } {
108 if (n <= 5) return .{ .k = 1, .adjustment = if (n == 3) -20 else if (n == 4) -10 else 0 };
109 if (n == 6) return .{ .k = 2, .adjustment = -10 };
110 return .{ .k = if (n <= 8) 2 else if (n <= 11) 3 else if (n <= 14) 4 else if (n <= 16) 5 else if (n <= 18) 6 else if (n == 19) 7 else 8, .adjustment = 0 };
111}
112fn eligible(p: *Player) Picks {
113 var result: Picks = .{};
114 result.len = @min(20, p.rounds.items.len);
115 var reduction: i64 = 0;
116 var i = p.rounds.items.len;
117 while (i > p.rounds.items.len - result.len) {
118 i -= 1;
119 const r = p.rounds.items[i];
120 reduction += r.exceptional;
121 result.items[i - (p.rounds.items.len - result.len)] = .{ .number = i + 1, .value = r.diff - reduction };
122 }
123 if (result.len >= 3) {
124 var order: [20]usize = undefined;
125 for (0..result.len) |j| order[j] = j;
126 std.mem.sort(usize, order[0..result.len], @as([]const Pick, result.items[0..result.len]), pickLess);
127 for (order[0..rule(result.len).k]) |j| result.items[j].counting = true;
128 }
129 return result;
130}
131fn lowIndex(p: *Player, day: i64) ?IndexRecord {
132 var best: ?IndexRecord = null;
133 for (p.records.items) |r| {
134 if (r.day < day - 365 or r.day >= day) continue;
135 if (best == null or r.value < best.?.value or (r.value == best.?.value and r.day < best.?.day)) best = r;
136 }
137 return best;
138}
139fn pickLess(picks: []const Pick, a: usize, b: usize) bool {

141}

144 for ([_]usize{ 0, 9, 12, 15, 17 }, 0..) |start, i| keys[i] = sum(values[start..]);

154 const order = std.mem.order(i64, &self.keys[a], &self.keys[b]);

201 fn update(self: *Golf, p: *Player, day: i64) void {
202 _ = self;
203 const picks = eligible(p);
204 if (picks.len < 3) {
205 p.calculation = null;
206 return;
207 }
208 const r = rule(picks.len);
209 var total: i64 = 0;
210 for (picks.items[0..picks.len]) |pick| {
211 if (pick.counting) total += pick.value;
212 }
213 const average = away(total, @intCast(r.k));
214 var value = average + r.adjustment;
215 const low: ?i64 = if (lowIndex(p, day)) |record| record.value else null;
216 var soft = false;
217 var hard = false;
218 var maximum = false;
219 if (low) |l| {
220 if (value - l > 30) {
221 const capped = l + 30 + @divTrunc(value - l - 30, 2);
222 soft = capped != value;
223 value = capped;
224 }
225 if (value - l > 50) {
226 value = l + 50;
227 hard = true;
228 }
229 }
230 if (value > 540) {
231 value = 540;
232 maximum = true;
233 }
234 p.index = value;
235 p.calculation = .{ .k = r.k, .average = average, .adjustment = r.adjustment, .low = low, .hard = hard, .soft = soft, .maximum = maximum };
236 }
237 fn evaluate(self: *Golf, e: *Event) Error!Result {
238 var r = Result{ .event = e };
239 var minimum: i64 = std.math.maxInt(i64);
240 for (e.players, 0..) |p, i| {
241 var selected: ?*Round = null;
242 for (p.rounds.items) |round| {
243 if (round.course == e.course and round.tee == e.tee and round.day == e.day) {
244 selected = round;
245 break;
246 }
247 }
248 const round = selected orelse return self.fail("missing round");
249 const hcp = round.handicap orelse return self.fail("no index");
250 r.rounds[i] = round;
251 r.ph[i] = halfUp(hcp * e.allowance, 100);
252 minimum = @min(minimum, r.ph[i]);
253 }
254 const relative = std.mem.eql(u8, e.format, "MATCH") or std.mem.eql(u8, e.format, "NASSAU") or std.mem.eql(u8, e.format, "FOURBALL");
255 for (e.players, 0..) |_, i| {
256 const strokes = allocation(r.ph[i] - (if (relative) minimum else @as(i64, 0)), e.course);
257 for (0..18) |h| {
258 r.nets[i][h] = if (r.rounds[i].scores[h] > 0) r.rounds[i].scores[h] - strokes[h] else null;
259 r.points[i][h] = points(r.nets[i][h], e.course.pars[h]);
260 }
261 }
262 return r;
263 }
264 fn rank(r: *const Result, gross: bool) Rows {
265 const e = r.event;
266 var rows: Rows = .{};
267 var values: [10][18]i64 = undefined;
268 var keys: [10][5]i64 = undefined;
269 for (e.players, 0..) |_, i| {
270 var complete = true;
271 for (r.nets[i]) |n| {
272 if (n == null) complete = false;
273 }
274 if (std.mem.eql(u8, e.format, "STROKE") and !complete) continue;
275 rows.items[rows.len] = i;
276 rows.len += 1;
277 for (0..18) |h| {
278 values[i][h] = if (gross) r.rounds[i].scores[h] else if (std.mem.eql(u8, e.format, "STABLEFORD")) r.points[i][h] else if (std.mem.eql(u8, e.format, "PAR")) (if (r.nets[i][h] == null or r.nets[i][h].? > e.course.pars[h]) @as(i64, -1) else if (r.nets[i][h].? == e.course.pars[h]) @as(i64, 0) else @as(i64, 1)) else r.nets[i][h].?;
279 }
280 }
281 for (rows.items[0..rows.len]) |i| keys[i] = countback(values[i]);
282 std.mem.sort(usize, rows.items[0..rows.len], CountbackOrder{
283 .keys = keys[0..e.players.len],
284 .players = e.players,
285 .higher = !gross and !std.mem.eql(u8, e.format, "STROKE"),
286 }, CountbackOrder.less);
287 return rows;
288 }
289 fn match(self: *Golf, wins: [18]i64, first: usize, end: usize, side1: []const u8, side2: []const u8) Error!struct { statuses: []const u8, result: []const u8 } {
290 var lead: i64 = 0;
291 var remaining: usize = end - first;
292 var statuses: std.Io.Writer.Allocating = .init(self.gpa);
293 for (first..end) |h| {
294 lead += wins[h];
295 remaining = end - h - 1;
296 if (h > first) try statuses.writer.writeAll(", ");
297 try statuses.writer.writeAll(try self.status(lead));
298 if (@abs(lead) > remaining) break;
299 }
300 var result: []const u8 = "AS";
301 if (lead != 0) {
302 const side = if (lead > 0) side1 else side2;
303 result = if (remaining > 0) try std.fmt.allocPrint(self.gpa, "{s} {d}&{d}", .{ side, @abs(lead), remaining }) else try std.fmt.allocPrint(self.gpa, "{s} {d} UP", .{ side, @abs(lead) });
304 }
305 return .{ .statuses = statuses.writer.buffered(), .result = result };
306 }
307 fn results(self: *Golf, r: *const Result) Error!void {
308 const e = r.event;
309 const f = e.format;
310 try self.out.print("results {s}: {s} {s} {s} day {d} allowance {d}\n", .{ e.id, f, e.course.id, e.tee.id, e.day, e.allowance });
311 if (std.mem.eql(u8, f, "STROKE") or std.mem.eql(u8, f, "STABLEFORD") or std.mem.eql(u8, f, "PAR")) {
312 const ranked = rank(r, false);
313 for (ranked.items[0..ranked.len], 0..) |i, place| {
314 try self.out.print("{d} {s}", .{ place + 1, e.players[i].id });
315 if (std.mem.eql(u8, f, "STROKE")) {
316 var net: i64 = 0;
317 for (r.nets[i]) |n| net += n.?;
318 try self.out.print(" net {d} gross {d}", .{ net, sum(&r.rounds[i].scores) });
319 } else if (std.mem.eql(u8, f, "STABLEFORD")) {
320 try self.out.print(" points {d}", .{sum(&r.points[i])});
321 } else {
322 var record: i64 = 0;
323 for (0..18) |h| record += if (r.nets[i][h] == null or r.nets[i][h].? > e.course.pars[h]) @as(i64, -1) else if (r.nets[i][h].? == e.course.pars[h]) @as(i64, 0) else @as(i64, 1);
324 try self.out.print(" {s}", .{try self.status(record)});
325 }
326 try self.out.print(" hcp {s}\n", .{try self.hcpText(r.ph[i])});
327 }
328 if (std.mem.eql(u8, f, "STROKE")) {
329 var nr: Rows = .{};
330 for (e.players, 0..) |_, i| {
331 var complete = true;
332 for (r.nets[i]) |n| {
333 if (n == null) complete = false;
334 }
335 if (!complete) {
336 nr.items[nr.len] = i;
337 nr.len += 1;
338 }
339 }
340 std.mem.sort(usize, nr.items[0..nr.len], e, playerIDLess);
341 for (nr.items[0..nr.len]) |i| try self.out.print("NR {s} hcp {s}\n", .{ e.players[i].id, try self.hcpText(r.ph[i]) });
342 if (ranked.len > 0) {
343 const i = rank(r, true).items[0];
344 try self.out.print("gross {s} {d}\n", .{ e.players[i].id, sum(&r.rounds[i].scores) });
345 }
346 }
347 } else if (std.mem.eql(u8, f, "MATCH") or std.mem.eql(u8, f, "NASSAU") or std.mem.eql(u8, f, "FOURBALL")) {
348 const minimum = std.mem.min(i64, r.ph[0..e.players.len]);
349 try self.out.writeAll("strokes");
350 for (e.players, 0..) |p, i| try self.out.print(" {s} {d}", .{ p.id, r.ph[i] - minimum });
351 try self.out.writeByte('\n');
352 const four = std.mem.eql(u8, f, "FOURBALL");
353 const side1 = if (four) try std.fmt.allocPrint(self.gpa, "{s}+{s}", .{ e.players[0].id, e.players[1].id }) else e.players[0].id;
354 const side2 = if (four) try std.fmt.allocPrint(self.gpa, "{s}+{s}", .{ e.players[2].id, e.players[3].id }) else e.players[1].id;
355 var wins: [18]i64 = undefined;
356 for (0..18) |h| {
357 var a = r.nets[0][h];
358 var b = r.nets[if (four) 2 else 1][h];
359 if (four) {
360 if (r.nets[1][h]) |n| {
361 if (a == null or n < a.?) a = n;
362 }
363 if (r.nets[3][h]) |n| {
364 if (b == null or n < b.?) b = n;
365 }
366 }
367 wins[h] = if (a == null) (if (b == null) @as(i64, 0) else @as(i64, -1)) else if (b == null) 1 else if (a.? < b.?) 1 else if (a.? > b.?) -1 else 0;
368 }
369 if (std.mem.eql(u8, f, "NASSAU")) {
370 try self.out.print("front {s}\nback {s}\ntotal {s}\n", .{ (try self.match(wins, 0, 9, side1, side2)).result, (try self.match(wins, 9, 18, side1, side2)).result, (try self.match(wins, 0, 18, side1, side2)).result });
371 } else {
372 const m = try self.match(wins, 0, 18, side1, side2);
373 try self.out.print("status {s}\nresult {s}\n", .{ m.statuses, m.result });
374 }
375 } else if (std.mem.eql(u8, f, "BETTERBALL")) {
376 var values: [5][18]i64 = undefined;
377 var teams: [5]usize = undefined;
378 const count = e.players.len / 2;
379 for (0..count) |i| {
380 teams[i] = i;
381 for (0..18) |h| values[i][h] = @max(r.points[2 * i][h], r.points[2 * i + 1][h]);
382 }
383 var keys: [5][5]i64 = undefined;
384 for (0..count) |i| keys[i] = countback(values[i]);
385 std.mem.sort(usize, teams[0..count], CountbackOrder{
386 .keys = keys[0..count],
387 .players = e.players,
388 .higher = true,
389 .stride = 2,
390 }, CountbackOrder.less);
391 for (teams[0..count], 0..) |team, place| {
392 const i = team * 2;
393 try self.out.print("{d} {s}+{s} points {d} hcp {s} {s}\n", .{ place + 1, e.players[i].id, e.players[i + 1].id, sum(&values[team]), try self.hcpText(r.ph[i]), try self.hcpText(r.ph[i + 1]) });
394 }
395 } else {
396 var carry: i64 = 0;
397 var totals: [10]i64 = @splat(0);
398 for (0..18) |h| {
399 carry += 1;
400 const ws = winners(r, h);
401 if (ws.len == 1) {
402 const i = ws.items[0];
403 totals[i] += carry * 100;
404 try self.out.print("hole {d} {s} {d}\n", .{ h + 1, e.players[i].id, carry });
405 carry = 0;
406 }
407 }
408 if (carry > 0) {
409 var ws = winners(r, 17);
410 if (ws.len == 0) {
411 ws.len = e.players.len;
412 for (0..ws.len) |i| ws.items[i] = i;
413 }
414 std.mem.sort(usize, ws.items[0..ws.len], e, playerIDLess);
415 try self.out.print("carry {d}", .{carry});
416 for (ws.items[0..ws.len]) |i| {
417 totals[i] += @divTrunc(carry * 100, @as(i64, @intCast(ws.len)));
418 try self.out.print(" {s}", .{e.players[i].id});
419 }
420 try self.out.writeByte('\n');
421 }
422 var ids: [10]usize = undefined;
423 for (0..e.players.len) |i| ids[i] = i;

431 }
432 };
433 std.mem.sort(usize, ids[0..e.players.len], Order{ .totals = &totals, .event = e }, Order.less);
434 for (ids[0..e.players.len]) |i| try self.out.print("{s} {s}\n", .{ e.players[i].id, try self.two(totals[i], 100) });
435 }
436 }
437 fn cardTotal(self: *Golf, label: []const u8, r: *Round, strokes: [18]i64, first: usize, end: usize) Error!void {
438 var complete = true;
439 var score: i64 = 0;
440 var net: i64 = 0;
441 var pts: i64 = 0;
442 for (first..end) |h| {
443 if (r.scores[h] <= 0)
444 complete = false
445 else {
446 score += r.scores[h];
447 net += r.scores[h] - strokes[h];
448 pts += points(r.scores[h] - strokes[h], r.course.pars[h]);
449 }
450 }
451 try self.out.print("{s} {d} {s} {d} {d} {s} {d}\n", .{ label, sum(r.course.pars[first..end]), if (complete) try std.fmt.allocPrint(self.gpa, "{d}", .{score}) else "-", sum(r.adjusted[first..end]), sum(strokes[first..end]), if (complete) try std.fmt.allocPrint(self.gpa, "{d}", .{net}) else "-", pts });
452 }

457 for (0..18) |h| {

469 fn stats(self: *Golf, p: *Player) Error!void {
470 var categories: [7]i64 = @splat(0);
471 var par_sum: [6]i64 = @splat(0);
472 var par_count: [6]i64 = @splat(0);
473 var best: ?i64 = null;
474 var complete_sum: i64 = 0;
475 var complete_count: i64 = 0;
476 for (p.rounds.items) |r| {
477 var complete = true;
478 for (0..18) |h| {
479 const s = r.scores[h];
480 const par: usize = @intCast(r.course.pars[h]);
481 if (s <= 0) {
482 complete = false;
483 if (s == 0) categories[6] += 1;
484 continue;
485 }
486 const d = s - r.course.pars[h];
487 const cat: usize = if (d <= -2) 0 else if (d == -1) 1 else if (d == 0) 2 else if (d == 1) 3 else if (d == 2) 4 else 5;
488 categories[cat] += 1;
489 par_sum[par] += s;
490 par_count[par] += 1;
491 }
492 if (complete) {
493 const s = sum(&r.scores);
494 best = if (best) |b| @min(b, s) else s;
495 complete_sum += s;
496 complete_count += 1;
497 }
498 }
499 try self.out.print("stats {s}: rounds {d} best {s} average {s} eagles {d} birdies {d} pars {d} bogeys {d} doubles {d} others {d} pickups {d}", .{ p.id, p.rounds.items.len, if (best) |b| try std.fmt.allocPrint(self.gpa, "{d}", .{b}) else "-", if (complete_count == 0) "-" else try self.two(complete_sum, complete_count), categories[0], categories[1], categories[2], categories[3], categories[4], categories[5], categories[6] });
500 for (3..6) |k| try self.out.print(" par{d} {s}", .{ k, if (par_count[k] == 0) "-" else try self.two(par_sum[k], par_count[k]) });
501 try self.out.writeByte('\n');
502 }
503 fn execute(self: *Golf, a: []const []const u8) Error!void {
504 const op = a[0];
505 const id: []const u8 = if (a.len > 1) a[1] else "";
506 if (std.mem.eql(u8, op, "COURSE")) {
507 if (self.courses.get(id) != null) return self.fail("duplicate course");
508 var pars: [18]i64 = undefined;
509 var si: [18]i64 = undefined;
510 var seen: [19]bool = @splat(false);
511 for (0..18) |h| {
512 pars[h] = integer(a[h + 2]);
513 si[h] = integer(a[h + 20]);
514 if (pars[h] < 3 or pars[h] > 5 or si[h] < 1 or si[h] > 18) return self.fail("invalid course");
515 const s: usize = @intCast(si[h]);
516 if (seen[s]) return self.fail("invalid course");
517 seen[s] = true;
518 }
519 const c = try self.gpa.create(Course);
520 c.* = .{ .id = id, .pars = pars, .si = si, .par = sum(&pars) };
521 try self.courses.put(self.gpa, id, c);
522 } else if (std.mem.eql(u8, op, "TEE")) {
523 const c = try self.course(id);
524 if (c.tees.get(a[2]) != null) return self.fail("duplicate tee");
525 const r = integer(a[3]);
526 const s = integer(a[4]);
527 if (r < 500 or r > 900 or s < 55 or s > 155) return self.fail("invalid tee");
528 const t = try self.gpa.create(Tee);
529 t.* = .{ .id = a[2], .rating = r, .slope = s };
530 try c.tees.put(self.gpa, t.id, t);
531 } else if (std.mem.eql(u8, op, "PLAYER")) {
532 if (self.players.get(id) != null) return self.fail("duplicate player");
533 const value: ?i64 = if (a.len > 3) integer(a[3]) else null;
534 if (value) |v| {
535 if (v < -100 or v > 540) return self.fail("invalid index");
536 }
537 const p = try self.gpa.create(Player);
538 p.* = .{ .id = id, .name = a[2], .index = value };
539 try self.players.put(self.gpa, id, p);
540 } else if (std.mem.eql(u8, op, "ROUND")) {
541 const p = try self.player(id);
542 const c = try self.course(a[2]);
543 const t = try self.tee(c, a[3]);
544 const day = integer(a[4]);
545 if (p.rounds.items.len > 0 and day < p.rounds.items[p.rounds.items.len - 1].day) return self.fail("invalid day");
546 var scores: [18]i64 = undefined;
547 var missing: usize = 0;
548 for (0..18) |h| {
549 scores[h] = if (std.mem.eql(u8, a[h + 5], "-")) -1 else if (std.mem.eql(u8, a[h + 5], "X")) 0 else integer(a[h + 5]);
550 if (scores[h] < 0) missing += 1;
551 }
552 if (missing >= 5) return self.fail("too few holes");
553 const hcp: ?i64 = if (p.index) |value| handicap(value, c, t) else null;
554 const strokes = allocation(hcp orelse 0, c);
555 var adjusted: [18]i64 = undefined;
556 for (0..18) |h| {
557 const limit = c.pars[h] + (if (p.index == null) @as(i64, 5) else 2 + strokes[h]);
558 adjusted[h] = if (scores[h] < 0) c.pars[h] + strokes[h] else if (scores[h] == 0) limit else @min(scores[h], limit);
559 }
560 const ags = sum(&adjusted);
561 const diff = away(113 * (10 * ags - t.rating), t.slope);
562 var exceptional: i64 = 0;
563 if (p.index) |value| {
564 if (value - diff >= 70) exceptional = if (value - diff < 100) 10 else 20;
565 }
566 const r = try self.gpa.create(Round);
567 r.* = .{ .course = c, .tee = t, .day = day, .scores = scores, .handicap = hcp, .adjusted = adjusted, .ags = ags, .diff = diff, .exceptional = exceptional, .old_index = p.index, .old_calculation = p.calculation, .old_records = p.records.items.len };
568 try p.rounds.append(self.gpa, r);
569 self.update(p, day);
570 if (p.index) |value| try p.records.append(self.gpa, .{ .day = day, .value = value });
571 try self.out.print("round {s} {d}: ags {d} diff {s}", .{ id, p.rounds.items.len, ags, try self.decimal(diff) });
572 if (exceptional != 0) try self.out.print(" exceptional {s}", .{try self.decimal(-exceptional)});
573 try self.out.print(" index {s}\n", .{try self.indexText(p.index)});
574 } else if (std.mem.eql(u8, op, "REMOVE")) {
575 const p = try self.player(id);
576 if (p.rounds.items.len == 0) return self.fail("no rounds");
577 const no = p.rounds.items.len;
578 const r = p.rounds.pop().?;
579 p.index = r.old_index;
580 p.calculation = r.old_calculation;
581 p.records.shrinkRetainingCapacity(r.old_records);
582 try self.out.print("removed {s} {d}: index {s}\n", .{ id, no, try self.indexText(p.index) });
583 } else if (std.mem.eql(u8, op, "INDEX") or std.mem.eql(u8, op, "HISTORY") or std.mem.eql(u8, op, "LOWINDEX") or std.mem.eql(u8, op, "STATS") or std.mem.eql(u8, op, "CARD")) {
584 const p = try self.player(id);
585 if (std.mem.eql(u8, op, "INDEX")) {
586 try self.out.print("index {s}: {s} rounds {d}", .{ id, try self.indexText(p.index), p.rounds.items.len });
587 if (p.calculation) |c| {
588 try self.out.print(" lowest {d} average {s} adjustment {s}", .{ c.k, try self.decimal(c.average), try self.decimal(c.adjustment) });
589 if (c.low) |l| try self.out.print(" low {s}", .{try self.indexText(l)});
590 if (c.hard) try self.out.writeAll(" hard cap") else if (c.soft) try self.out.writeAll(" soft cap");
591 if (c.maximum) try self.out.writeAll(" maximum");
592 }
593 try self.out.writeByte('\n');
594 } else if (std.mem.eql(u8, op, "HISTORY")) {
595 try self.out.print("history {s}: rounds {d}\n", .{ id, p.rounds.items.len });
596 const picks = eligible(p);
597 for (picks.items[0..picks.len]) |pick| {
598 const r = p.rounds.items[pick.number - 1];
599 try self.out.print("{d} day {d} {s} {s} ags {d} diff {s}", .{ pick.number, r.day, r.course.id, r.tee.id, r.ags, try self.decimal(r.diff) });
600 if (pick.value != r.diff) try self.out.print(" adjusted {s}", .{try self.decimal(pick.value)});
601 if (pick.counting) try self.out.writeAll(" *");
602 try self.out.writeByte('\n');
603 }
604 } else if (std.mem.eql(u8, op, "LOWINDEX")) {
605 const day = integer(a[2]);
606 try self.out.print("lowindex {s} {d}: ", .{ id, day });
607 if (lowIndex(p, day)) |r| try self.out.print("{s} day {d}\n", .{ try self.indexText(r.value), r.day }) else try self.out.writeAll("none\n");
608 } else if (std.mem.eql(u8, op, "STATS")) {
609 try self.stats(p);
610 } else {
611 const no: usize = @intCast(integer(a[2]));
612 if (no > p.rounds.items.len) return self.fail("unknown round");
613 try self.card(p, no, p.rounds.items[no - 1]);
614 }
615 } else if (std.mem.eql(u8, op, "COURSEHCP")) {
616 const p = try self.player(id);
617 const c = try self.course(a[2]);
618 const t = try self.tee(c, a[3]);
619 const value = p.index orelse return self.fail("no index");
620 const hcp = handicap(value, c, t);
621 try self.out.print("coursehcp {s} {s} {s}: {s} strokes", .{ id, c.id, t.id, try self.hcpText(hcp) });
622 for (allocation(hcp, c)) |s| try self.out.print(" {d}", .{s});
623 try self.out.writeByte('\n');
624 } else if (std.mem.eql(u8, op, "HCPTABLE")) {
625 const c = try self.course(id);
626 const t = try self.tee(c, a[2]);
627 const first = integer(a[3]);
628 const last = integer(a[4]);
629 if (first < -100 or first > last or last > 540) return self.fail("invalid index");
630 try self.out.print("hcptable {s} {s}: {s} to {s}\n", .{ id, t.id, try self.indexText(first), try self.indexText(last) });
631 var start = first;
632 var current = handicap(first, c, t);
633 var i = first + 1;
634 while (i <= last + 1) : (i += 1) {
635 const hcp: ?i64 = if (i > last) null else handicap(i, c, t);
636 if (hcp == null or hcp.? != current) {
637 try self.out.print("{s} to {s}: {s}\n", .{ try self.indexText(start), try self.indexText(i - 1), try self.hcpText(current) });
638 start = i;
639 if (hcp) |h| current = h;
640 }
641 }
642 } else if (std.mem.eql(u8, op, "HOLESTATS")) {
643 const c = try self.course(id);
644 var sums: [18]i64 = @splat(0);
645 var counts: [18]i64 = @splat(0);
646 var rounds: usize = 0;
647 for (self.players.values()) |p| {
648 for (p.rounds.items) |r| {
649 if (r.course != c) continue;
650 rounds += 1;
651 for (0..18) |h| {
652 if (r.scores[h] > 0) {
653 counts[h] += 1;
654 sums[h] += r.scores[h];
655 }
656 }
657 }
658 }
659 var holes: [18]usize = undefined;
660 for (0..18) |h| holes[h] = h;
661 const Order = struct {
662 course: *Course,
663 sums: *const [18]i64,
664 counts: *const [18]i64,
665
666 fn less(self: @This(), a: usize, b: usize) bool {
667 if (self.counts[a] == 0 or self.counts[b] == 0)
668 return if (self.counts[a] == self.counts[b]) a < b else self.counts[a] > 0;
669 const av = (self.sums[a] - self.course.pars[a] * self.counts[a]) * self.counts[b];
670 const bv = (self.sums[b] - self.course.pars[b] * self.counts[b]) * self.counts[a];
671 return av > bv or (av == bv and a < b);
672 }
673 };
674 std.mem.sort(usize, &holes, Order{ .course = c, .sums = &sums, .counts = &counts }, Order.less);
675 var ranks: [18]usize = undefined;
676 for (holes, 0..) |h, i| ranks[h] = i + 1;
677 try self.out.print("holestats {s}: rounds {d}\n", .{ id, rounds });
678 for (0..18) |h| try self.out.print("{d} par {d} si {d} scores {d} avg {s} rank {d}\n", .{ h + 1, c.pars[h], c.si[h], counts[h], if (counts[h] == 0) "-" else try self.two(sums[h], counts[h]), ranks[h] });
679 } else if (std.mem.eql(u8, op, "PLAYERS")) {
680 const ps = try self.gpa.dupe(*Player, self.players.values());
681 std.mem.sort(*Player, ps, {}, playerLess);
682 try self.out.print("players {d}\n", .{ps.len});
683 for (ps) |p| try self.out.print("{s} {s} {s} rounds {d}\n", .{ p.id, p.name, try self.indexText(p.index), p.rounds.items.len });
684 } else if (std.mem.eql(u8, op, "EVENT")) {
685 if (self.events.get(id) != null) return self.fail("duplicate event");
686 const c = try self.course(a[3]);
687 const t = try self.tee(c, a[4]);
688 const allowance = integer(a[6]);
689 if (allowance < 1 or allowance > 100) return self.fail("invalid allowance");
690 const ps = try self.gpa.alloc(*Player, a.len - 7);
691 for (a[7..], 0..) |name, i| ps[i] = try self.player(name);
692 const f = a[2];
693 const n = ps.len;
694 var valid = if (std.mem.eql(u8, f, "MATCH") or std.mem.eql(u8, f, "NASSAU")) n == 2 else if (std.mem.eql(u8, f, "FOURBALL")) n == 4 else if (std.mem.eql(u8, f, "BETTERBALL")) n == 4 or n == 6 or n == 8 else n >= 2 and n <= 8;
695 for (ps, 0..) |p, i| {
696 if (std.mem.findScalar(*Player, ps[0..i], p) != null) valid = false;
697 }
698 if (!valid) return self.fail("invalid players");
699 const e = try self.gpa.create(Event);
700 e.* = .{ .id = id, .format = f, .course = c, .tee = t, .day = integer(a[5]), .allowance = allowance, .players = ps };
701 try self.events.put(self.gpa, id, e);
702 } else if (std.mem.eql(u8, op, "RESULTS")) {
703 const e = self.events.get(id) orelse return self.fail("unknown event");
704 const r = try self.evaluate(e);
705 try self.results(&r);
706 } else if (std.mem.eql(u8, op, "MERIT")) {
707 const Merit = struct {
708 player: *Player,
709 points: i64 = 0,
710 wins: i64 = 0,
711 appearances: i64 = 0,
712
713 fn less(_: void, a: @This(), b: @This()) bool {
714 return a.points > b.points or (a.points == b.points and (a.wins > b.wins or
715 (a.wins == b.wins and std.mem.order(u8, a.player.id, b.player.id) == .lt)));
716 }
717 };
718 const values = try self.gpa.alloc(Merit, self.players.values().len);
719 for (self.players.values(), 0..) |p, i| values[i] = .{ .player = p };
720 var counted: usize = 0;
721 for (self.events.values()) |e| {
722 if (!std.mem.eql(u8, e.format, "STROKE") and !std.mem.eql(u8, e.format, "STABLEFORD") and !std.mem.eql(u8, e.format, "PAR")) continue;
723 const r = self.evaluate(e) catch |err| {
724 if (err == error.BadCommand) continue;
725 return err;
726 };
727 counted += 1;
728 for (e.players) |p| values[self.players.getIndex(p.id).?].appearances += 1;
729 const ranked = rank(&r, false);
730 const awards = [_]i64{ 10, 8, 6, 5, 4, 3, 2, 1 };
731 for (ranked.items[0..ranked.len], 0..) |i, place| {
732 const v = &values[self.players.getIndex(e.players[i].id).?];
733 v.points += awards[place];
734 if (place == 0) v.wins += 1;
735 }
736 }
737 std.mem.sort(Merit, values, {}, Merit.less);
738 try self.out.print("merit: events {d}\n", .{counted});
739 var place: usize = 0;
740 for (values) |v| {
741 if (v.appearances == 0) continue;
742 place += 1;
743 try self.out.print("{d} {s} {d} wins {d} played {d}\n", .{ place, v.player.id, v.points, v.wins, v.appearances });
744 }
745 }
746 }
747};
748fn playerIDLess(e: *Event, a: usize, b: usize) bool {
749 return std.mem.order(u8, e.players[a].id, e.players[b].id) == .lt;
750}
751fn playerLess(_: void, a: *Player, b: *Player) bool {
752 const ai = a.index orelse std.math.maxInt(i64);
753 const bi = b.index orelse std.math.maxInt(i64);
754 return ai < bi or (ai == bi and std.mem.order(u8, a.id, b.id) == .lt);
755}
756fn winners(r: *const Result, h: usize) Rows {
757 var rows: Rows = .{};
758 var minimum: ?i64 = null;
759 for (r.event.players, 0..) |_, i| {
760 if (r.nets[i][h]) |net| {
761 if (minimum == null or net < minimum.?) {
762 minimum = net;
763 rows.len = 0;
764 }
765 if (net == minimum.?) {
766 rows.items[rows.len] = i;
767 rows.len += 1;
768 }
769 }
770 }
771 return rows;
772}
773pub fn main(init: std.process.Init) !void {
774 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
775 defer arena.deinit();
776 const gpa = arena.allocator();
777 var ib: [4096]u8 = undefined;
778 var input = std.Io.File.stdin().readerStreaming(init.io, &ib);
779 var ob: [8192]u8 = undefined;
780 var output = std.Io.File.stdout().writerStreaming(init.io, &ob);
781 var golf = Golf{ .gpa = gpa, .out = &output.interface };
782 const n: usize = @intCast(integer((try input.interface.takeDelimiter('\n')).?));
783 for (0..n) |_| {
784 const line = try gpa.dupe(u8, (try input.interface.takeDelimiter('\n')).?);
785 var tokens: std.ArrayList([]const u8) = .empty;
786 var it = std.mem.splitScalar(u8, line, ' ');
787 while (it.next()) |part| try tokens.append(gpa, part);
788 var draft: std.Io.Writer.Allocating = .init(gpa);
789 golf.out = &draft.writer;
790 golf.execute(tokens.items) catch |err| {
791 if (err == error.BadCommand) {
792 try output.interface.print("ERROR {s}\n", .{golf.error_message});
793 continue;
794 }
795 return err;
796 };
797 try output.interface.writeAll(draft.writer.buffered());
798 }
799 try output.interface.flush();
800}
