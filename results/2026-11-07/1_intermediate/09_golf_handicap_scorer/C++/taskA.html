18int round_half_up(long long numerator, long long denominator) {
19 return static_cast<int>((2 * numerator + denominator >= 0)
20 ? (2 * numerator + denominator) / (2 * denominator)
21 : -((-2 * numerator - denominator + 2 * denominator - 1) / (2 * denominator)));
22}
23
24int round_half_away(long long numerator, long long denominator) {
25 return numerator < 0 ? -round_half_up(-numerator, denominator)
26 : round_half_up(numerator, denominator);
27}

65array<int, 18> allocate(int handicap, const array<int, 18>& stroke_indexes) {
66 array<int, 18> strokes{};
67 int quotient = abs(handicap) / 18;
68 int remainder = abs(handicap) % 18;
69 for (int hole = 0; hole < 18; ++hole) {
70 strokes[hole] = handicap >= 0
71 ? quotient + (stroke_indexes[hole] <= remainder)
72 : -quotient - (stroke_indexes[hole] >= 19 - remainder);
73 }
74 return strokes;
75}

77array<long long, 5> countback(const vector<int>& values) {
78 array<long long, 5> result{};
79 for (int hole = 0; hole < 18; ++hole) {
80 result[0] += values[hole];
81 if (hole >= 9) result[1] += values[hole];
82 if (hole >= 12) result[2] += values[hole];
83 if (hole >= 15) result[3] += values[hole];
84 }
85 result[4] = values[17];
86 return result;
87}

89vector<string> ranked(vector<string> identifiers, const map<string, vector<int>>& values,
90 bool larger = false) {
91 sort(identifiers.begin(), identifiers.end(), [&](const string& a, const string& b) {
92 auto left = countback(values.at(a));
93 auto right = countback(values.at(b));
94 return left != right ? (larger ? left > right : left < right) : a < b;
95 });
96 return identifiers;
97}

169 optional<pair<int, int>> low_index(int day) const {
170 optional<pair<int, int>> lowest;
171 for (auto [date, value] : records) {
172 if (day - 365 <= date && date < day) {
173 pair<int, int> candidate{value, date};
174 if (!lowest || candidate < *lowest) lowest = candidate;
175 }
176 }
177 return lowest;
178 }
179
180 vector<Eligible> eligible() const {
181 vector<Eligible> selected;
182 int reduction = 0;
183 int start = max(0, static_cast<int>(rounds.size()) - 20);
184 for (int i = static_cast<int>(rounds.size()) - 1; i >= start; --i) {
185 reduction += rounds[i].exceptional;
186 selected.push_back({i + 1, &rounds[i], rounds[i].differential - reduction});
187 }
188 reverse(selected.begin(), selected.end());
189 return selected;
190 }
191
192 pair<vector<Eligible>, int> counting(vector<Eligible> selected) const {
193 int n = static_cast<int>(selected.size());
194 if (n < 3) return {{}, 0};
195 const array<int, 8> limits{5, 8, 11, 14, 16, 18, 19, 20};
196 int k = 1;
197 while (n > limits[k - 1]) ++k;
198 sort(selected.begin(), selected.end(), [](const Eligible& a, const Eligible& b) {
199 return a.adjusted != b.adjusted ? a.adjusted < b.adjusted
200 : a.number > b.number;
201 });
202 selected.resize(k);
203 int adjustment = n == 3 ? -20 : (n == 4 || n == 6 ? -10 : 0);
204 return {selected, adjustment};
205 }
206
207 void update_index(int day) {
208 auto [selected, adjustment] = counting(eligible());
209 if (selected.empty()) {
210 calculation.reset();
211 return;
212 }
213 long long total = 0;
214 for (const Eligible& entry : selected) total += entry.adjusted;
215 int average = round_half_away(total, selected.size());
216 int candidate = average + adjustment;
217 auto low_record = low_index(day);
218 optional<int> low = low_record ? optional<int>(low_record->first) : nullopt;
219 bool soft = false, hard = false, maximum = false;
220 if (low) {
221 if (candidate - *low > 30) {
222 candidate = *low + 30 + (candidate - *low - 30) / 2;
223 soft = true;
224 }
225 if (candidate - *low > 50) {
226 candidate = *low + 50;
227 hard = true;
228 }
229 }
230 if (candidate > 540) {
231 candidate = 540;
232 maximum = true;
233 }
234 index = candidate;
235 calculation = Calculation{static_cast<int>(selected.size()), average, adjustment,
236 low, hard, soft, maximum};
237 }

240Round::Round(const Player& player, Course* c, Tee* t, int d,
241 const array<int, 18>& raw)
242 : course(c), tee(t), day(d), scores(raw), previous_index(player.index),
243 previous_records(player.records.size()),
244 previous_calculation(player.calculation),
245 handicap(player.index ? optional<int>(t->handicap(*player.index, *c)) : nullopt) {
246 auto strokes = allocate(handicap.value_or(0), c->stroke_indexes);
247 for (int hole = 0; hole < 18; ++hole) {
248 int limit = handicap ? c->pars[hole] + 2 + strokes[hole] : c->pars[hole] + 5;
249 adjusted[hole] = raw[hole] == -1 ? c->pars[hole] + strokes[hole]
250 : raw[hole] == 0 ? limit : min(raw[hole], limit);
251 ags += adjusted[hole];
252 }
253 differential = round_half_away(113LL * (10 * ags - t->rating), t->slope);
254 if (player.index && *player.index - differential >= 70) {
255 exceptional = *player.index - differential >= 100 ? 20 : 10;
256 }
257}

259vector<optional<int>> Round::nets(const array<int, 18>& strokes) const {
260 vector<optional<int>> result;
261 for (int i = 0; i < 18; ++i)
262 result.push_back(scores[i] > 0 ? optional<int>(scores[i] - strokes[i]) : nullopt);
263 return result;
264}
265
266vector<int> Round::points(const vector<optional<int>>& nets) const {
267 vector<int> result;
268 for (int i = 0; i < 18; ++i)
269 result.push_back(nets[i] ? max(0, course->pars[i] + 2 - *nets[i]) : 0);
270 return result;
271}

328 const string& id = a[0];
329 if (courses.count(id)) throw runtime_error("duplicate course");
330 array<int, 18> pars{}, indexes{};
331 for (int i = 0; i < 18; ++i) {
332 pars[i] = stoi(a[i + 1]);
333 indexes[i] = stoi(a[i + 19]);
334 }
335 bool valid = all_of(pars.begin(), pars.end(),
336 [](int par) { return par >= 3 && par <= 5; });
337 auto ordered = indexes;
338 sort(ordered.begin(), ordered.end());
339 for (int i = 0; i < 18; ++i) valid &= ordered[i] == i + 1;
340 if (!valid) throw runtime_error("invalid course");
341 courses.emplace(id, Course{id, pars, indexes, static_cast<int>(sum(pars)), {}});

483 const vector<int>& points, int first, int last) {
484 int pars = 0, scores = 0, adjusted = 0, allocated = 0, net = 0, pts = 0;

524 void stats(const vector<string>& a) {
525 Player& p = player(a[0]);
526 const array<string, 7> labels{"eagles", "birdies", "pars", "bogeys",
527 "doubles", "others", "pickups"};
528 array<int, 7> counts{};
529 array<int, 6> par_total{}, par_count{};
530 int best = 0;
531 long long complete_total = 0;
532 int complete_count = 0;
533 for (const Round& r : p.rounds) {
534 if (r.complete()) {
535 int total = static_cast<int>(sum(r.scores));
536 if (complete_count == 0 || total < best) best = total;
537 complete_total += total;
538 ++complete_count;
539 }
540 for (int i = 0; i < 18; ++i) {
541 int score = r.scores[i], par = r.course->pars[i];
542 if (score == 0) ++counts[6];
543 else if (score > 0) {
544 int category = max(0, min(score - par + 2, 5));
545 ++counts[category];
546 par_total[par] += score;
547 ++par_count[par];
548 }
549 }
550 }
551 vector<string> parts{"stats", p.id + ":", "rounds", to_string(p.rounds.size()),
552 "best", complete_count ? to_string(best) : "-",
553 "average", average_display(complete_total, complete_count)};
554 for (int i = 0; i < 7; ++i) {
555 parts.push_back(labels[i]);
556 parts.push_back(to_string(counts[i]));
557 }
558 for (int par = 3; par <= 5; ++par) {
559 parts.push_back("par" + to_string(par));
560 parts.push_back(average_display(par_total[par], par_count[par]));
561 }
562 emit(join(parts));
563 }

565 void holestats(const vector<string>& a) {
566 Course& c = course(a[0]);
567 array<int, 18> scores{}, totals{}, rank{};
568 int rounds = 0;
569 for (const auto& [id, p] : players)
570 for (const Round& r : p.rounds)
571 if (r.course == &c) {
572 ++rounds;
573 for (int i = 0; i < 18; ++i)
574 if (r.scores[i] > 0) {
575 ++scores[i];
576 totals[i] += r.scores[i];
577 }
578 }
579 vector<int> holes(18);
580 iota(holes.begin(), holes.end(), 0);
581 sort(holes.begin(), holes.end(), [&](int left, int right) {
582 if (!scores[left]) return !scores[right] && left < right;
583 if (!scores[right]) return true;
584 long long a = 1LL * (totals[left] - c.pars[left] * scores[left]) * scores[right];
585 long long b = 1LL * (totals[right] - c.pars[right] * scores[right]) * scores[left];
586 return a != b ? a > b : left < right;
587 });
588 for (int i = 0; i < 18; ++i) rank[holes[i]] = i + 1;
589 emit("holestats", c.id + ":", "rounds", rounds);
590 for (int i = 0; i < 18; ++i)
591 emit(i + 1, "par", c.pars[i], "si", c.stroke_indexes[i], "scores", scores[i],
592 "avg", average_display(totals[i], scores[i]), "rank", rank[i]);
593 }

608 void register_event(const vector<string>& a) {
609 if (events.count(a[0])) throw runtime_error("duplicate event");
610 auto [c, t] = tee(a[2], a[3]);
611 int allowance = stoi(a[5]);
612 if (allowance < 1 || allowance > 100) throw runtime_error("invalid allowance");
613 vector<Player*> participants;
614 set<string> unique;
615 for (size_t i = 6; i < a.size(); ++i) {
616 participants.push_back(&player(a[i]));
617 unique.insert(a[i]);
618 }
619 int count = static_cast<int>(participants.size());
620 bool valid = a[1] == "MATCH" || a[1] == "NASSAU" ? count == 2
621 : a[1] == "FOURBALL" ? count == 4
622 : a[1] == "BETTERBALL" ? (count == 4 || count == 6 || count == 8)
623 : count >= 2 && count <= 8;
624 if (!valid || unique.size() != participants.size())
625 throw runtime_error("invalid players");
626 events.emplace(a[0], Event{a[0], a[1], c, t, stoi(a[4]), allowance,
627 move(participants)});
628 }

630 EventData event_data(const Event& event) const {
631 EventData data;
632 for (const Player* p : event.players) {
633 const Round* found = nullptr;
634 for (const Round& r : p->rounds) {
635 if (r.course == event.course && r.tee == event.tee && r.day == event.day) {
636 found = &r;
637 break;
638 }
639 }
640 if (!found) throw runtime_error("missing round");
641 if (!found->handicap) throw runtime_error("no index");
642 data.rounds[p->id] = found;
643 data.handicaps[p->id] = round_half_up(1LL * *found->handicap * event.allowance, 100);
644 }
645 int minimum = 0;
646 if (event.format == "MATCH" || event.format == "NASSAU" || event.format == "FOURBALL") {
647 minimum = min_element(data.handicaps.begin(), data.handicaps.end(),
648 [](const auto& a, const auto& b) { return a.second < b.second; })->second;
649 }
650 for (const Player* p : event.players) {
651 auto strokes = allocate(data.handicaps.at(p->id) - minimum,
652 event.course->stroke_indexes);
653 data.nets[p->id] = data.rounds.at(p->id)->nets(strokes);
654 data.points[p->id] = data.rounds.at(p->id)->points(data.nets.at(p->id));
655 }
656 return data;
657 }
658
659 pair<vector<string>, map<string, vector<int>>> event_ranking(
660 const Event& event, const EventData& data) const {
661 vector<string> ids;
662 map<string, vector<int>> values;
663 for (const Player* p : event.players) {
664 if (event.format == "STROKE" && !data.rounds.at(p->id)->complete()) continue;
665 ids.push_back(p->id);
666 if (event.format == "STABLEFORD") values[p->id] = data.points.at(p->id);
667 else if (event.format == "STROKE") {
668 vector<int> v;
669 for (const auto& net : data.nets.at(p->id)) v.push_back(*net);
670 values[p->id] = move(v);
671 } else {
672 vector<int> v;
673 for (int h = 0; h < 18; ++h) {
674 const auto& net = data.nets.at(p->id)[h];
675 v.push_back(!net || *net > event.course->pars[h] ? -1
676 : *net < event.course->pars[h] ? 1 : 0);
677 }
678 values[p->id] = move(v);
679 }
680 }
681 return {ranked(ids, values, event.format != "STROKE"), values};
682 }

684 void individual_results(const Event& event, const EventData& data) {
685 auto [order, values] = event_ranking(event, data);
686 int position = 1;
687 for (const string& id : order) {
688 string handicap = handicap_display(data.handicaps.at(id));
689 if (event.format == "STROKE")
690 emit(position, id, "net", sum(values.at(id)), "gross",
691 sum(data.rounds.at(id)->scores), "hcp", handicap);
692 else if (event.format == "STABLEFORD")
693 emit(position, id, "points", sum(data.points.at(id)), "hcp", handicap);
694 else
695 emit(position, id, match_status(static_cast<int>(sum(values.at(id)))),
696 "hcp", handicap);
697 ++position;
698 }
699 if (event.format == "STROKE") {
700 vector<string> nr;
701 for (const Player* p : event.players)
702 if (!data.rounds.at(p->id)->complete()) nr.push_back(p->id);
703 sort(nr.begin(), nr.end());
704 for (const string& id : nr)
705 emit("NR", id, "hcp", handicap_display(data.handicaps.at(id)));
706 if (!order.empty()) {
707 map<string, vector<int>> gross;
708 for (const string& id : order) {
709 auto& scores = data.rounds.at(id)->scores;
710 gross[id] = vector<int>(scores.begin(), scores.end());
711 }
712 string winner = ranked(order, gross).front();
713 emit("gross", winner, sum(gross.at(winner)));
714 }
715 }
716 }

718 pair<array<string, 2>, vector<int>> match_holes(const Event& event,
719 const EventData& data) const {
720 vector<string> ids;
721 for (const Player* p : event.players) ids.push_back(p->id);
722 vector<string> first{ids[0]}, second{ids[event.format == "FOURBALL" ? 2 : 1]};
723 if (event.format == "FOURBALL") {
724 first.push_back(ids[1]);
725 second.push_back(ids[3]);
726 }
727 array<string, 2> names{join(first, "+"), join(second, "+")};
728 vector<int> holes;
729 for (int h = 0; h < 18; ++h) {
730 auto best = [&](const vector<string>& side) -> optional<int> {
731 optional<int> minimum;
732 for (const string& id : side) {
733 auto net = data.nets.at(id)[h];
734 if (net && (!minimum || *net < *minimum)) minimum = *net;
735 }
736 return minimum;
737 };
738 auto left = best(first), right = best(second);
739 int winner = !left ? (!right ? 0 : -1)
740 : !right || *left < *right ? 1
741 : *left > *right ? -1 : 0;
742 holes.push_back(winner);
743 }
744 return {names, holes};
745 }
746
747 pair<vector<string>, string> play_match(const array<string, 2>& names,
748 const vector<int>& holes,
749 int first, int last) const {
750 int lead = 0, remaining = 0;
751 vector<string> statuses;
752 for (int h = first; h < last; ++h) {
753 lead += holes[h];
754 statuses.push_back(match_status(lead));
755 remaining = last - h - 1;
756 if (abs(lead) > remaining || remaining == 0) break;
757 }
758 if (lead == 0) return {statuses, "AS"};
759 string margin = remaining ? to_string(abs(lead)) + "&" + to_string(remaining)
760 : to_string(abs(lead)) + " UP";
761 return {statuses, names[lead > 0 ? 0 : 1] + " " + margin};
762 }
763
764 void match_results(const Event& event, const EventData& data) {
765 int minimum = min_element(data.handicaps.begin(), data.handicaps.end(),
766 [](const auto& a, const auto& b) { return a.second < b.second; })->second;
767 vector<string> parts{"strokes"};
768 for (const Player* p : event.players) {
769 parts.push_back(p->id);
770 parts.push_back(to_string(data.handicaps.at(p->id) - minimum));
771 }
772 emit(join(parts));
773 auto [names, holes] = match_holes(event, data);
774 if (event.format == "NASSAU") {
775 for (auto [label, first, last] :
776 vector<tuple<string, int, int>>{{"front", 0, 9},
777 {"back", 9, 18}, {"total", 0, 18}}) {
778 emit(label, play_match(names, holes, first, last).second);
779 }
780 } else {
781 auto [statuses, result] = play_match(names, holes, 0, 18);
782 emit("status", join(statuses, ", "));
783 emit("result", result);
784 }
785 }

787 void betterball_results(const Event& event, const EventData& data) {
788 map<string, pair<string, string>> teams;
789 map<string, vector<int>> values;
790 vector<string> ids;
791 for (size_t i = 0; i < event.players.size(); i += 2) {
792 string first = event.players[i]->id, second = event.players[i + 1]->id;
793 ids.push_back(first);
794 teams[first] = {first, second};
795 vector<int> points;
796 for (int h = 0; h < 18; ++h)
797 points.push_back(max(data.points.at(first)[h], data.points.at(second)[h]));
798 values[first] = move(points);
799 }
800 auto order = ranked(ids, values, true);
801 for (size_t i = 0; i < order.size(); ++i) {
802 auto [first, second] = teams.at(order[i]);
803 emit(i + 1, first + "+" + second, "points", sum(values.at(first)), "hcp",
804 handicap_display(data.handicaps.at(first)),
805 handicap_display(data.handicaps.at(second)));
806 }
807 }

809 void skins_results(const Event& event, const EventData& data) {
810 vector<string> ids;
811 map<string, int> totals;
812 for (const Player* p : event.players) {
813 ids.push_back(p->id);
814 totals[p->id] = 0;
815 }
816 int carry = 0;
817 vector<string> last_minimum;
818 for (int hole = 0; hole < 18; ++hole) {
819 ++carry;
820 optional<int> best;
821 vector<string> winners;
822 for (const string& id : ids) {
823 auto net = data.nets.at(id)[hole];
824 if (!net) continue;
825 if (!best || *net < *best) {
826 best = net;
827 winners = {id};
828 } else if (*net == *best) {
829 winners.push_back(id);
830 }
831 }
832 last_minimum = winners;
833 if (winners.size() == 1) {
834 emit("hole", hole + 1, winners.front(), carry);
835 totals[winners.front()] += 100 * carry;
836 carry = 0;
837 }
838 }
839 if (carry) {
840 auto sharing = last_minimum.empty() ? ids : last_minimum;
841 sort(sharing.begin(), sharing.end());
842 int share = carry * 100 / sharing.size();
843 for (const string& id : sharing) totals[id] += share;
844 vector<string> parts{"carry", to_string(carry)};
845 parts.insert(parts.end(), sharing.begin(), sharing.end());
846 emit(join(parts));
847 }
848 sort(ids.begin(), ids.end(), [&](const string& a, const string& b) {
849 return totals[a] == totals[b] ? a < b : totals[a] > totals[b];
850 });
851 for (const string& id : ids) {
852 int value = totals[id];
853 emit(id, to_string(value / 100) + "." +
854 (value % 100 < 10 ? "0" : "") + to_string(value % 100));
855 }
856 }

858 void results(const vector<string>& a) {
859 auto found = events.find(a[0]);
860 if (found == events.end()) throw runtime_error("unknown event");
861 const Event& event = found->second;
862 auto data = event_data(event);
863 emit("results", event.id + ":", event.format, event.course->id, event.tee->id,
864 "day", event.day, "allowance", event.allowance);
865 if (event.format == "STROKE" || event.format == "STABLEFORD" ||
866 event.format == "PAR") {
867 individual_results(event, data);
868 } else if (event.format == "BETTERBALL") {
869 betterball_results(event, data);
870 } else if (event.format == "SKINS") {
871 skins_results(event, data);
872 } else {
873 match_results(event, data);
874 }
875 }

877 void merit(const vector<string>&) {
878 map<string, array<int, 3>> tally;
879 const array<int, 8> awards{10, 8, 6, 5, 4, 3, 2, 1};
880 int counted = 0;
881 for (const auto& [id, event] : events) {
882 if (event.format != "STROKE" && event.format != "STABLEFORD" &&
883 event.format != "PAR") continue;
884 EventData data;
885 try { data = event_data(event); }
886 catch (const runtime_error&) { continue; }
887 ++counted;
888 auto order = event_ranking(event, data).first;
889 for (const Player* p : event.players) ++tally[p->id][2];
890 for (size_t i = 0; i < order.size(); ++i) {
891 tally[order[i]][0] += awards[i];
892 if (i == 0) ++tally[order[i]][1];
893 }
894 }
895 emit("merit:", "events", counted);
896 vector<string> ids;
897 for (const auto& [id, values] : tally) ids.push_back(id);
898 sort(ids.begin(), ids.end(), [&](const string& a, const string& b) {
899 if (tally[a][0] != tally[b][0]) return tally[a][0] > tally[b][0];
900 if (tally[a][1] != tally[b][1]) return tally[a][1] > tally[b][1];
901 return a < b;
902 });
903 for (size_t i = 0; i < ids.size(); ++i) {
904 const auto& row = tally.at(ids[i]);
905 emit(i + 1, ids[i], row[0], "wins", row[1], "played", row[2]);
906 }
907 }

910 void execute(const string& line) {
911 istringstream in(line);
912 string command, part;
913 in >> command;
914 vector<string> arguments;
915 while (in >> part) arguments.push_back(part);
916 try {
917 if (command == "COURSE") register_course(arguments);
918 else if (command == "TEE") register_tee(arguments);
919 else if (command == "PLAYER") register_player(arguments);
920 else if (command == "ROUND") add_round(arguments);
921 else if (command == "REMOVE") remove(arguments);
922 else if (command == "INDEX") show_index(arguments);
923 else if (command == "HISTORY") history(arguments);
924 else if (command == "LOWINDEX") lowindex(arguments);
925 else if (command == "COURSEHCP") coursehcp(arguments);
926 else if (command == "HCPTABLE") hcptable(arguments);
927 else if (command == "CARD") card(arguments);
928 else if (command == "STATS") stats(arguments);
929 else if (command == "HOLESTATS") holestats(arguments);
930 else if (command == "PLAYERS") show_players(arguments);
931 else if (command == "EVENT") register_event(arguments);
932 else if (command == "RESULTS") results(arguments);
933 else if (command == "MERIT") merit(arguments);
934 } catch (const runtime_error& error) {
935 emit("ERROR", error.what());
936 }
