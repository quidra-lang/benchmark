5private const val NO_NET = 1000000
6private class Fault(reason: String) : RuntimeException(reason)
7private fun error(condition: Boolean, reason: String) { if (condition) throw Fault(reason) }
8private fun away(n: Long, d: Long) = (if (n < 0) -((2*(-n)+d)/(2*d)) else (2*n+d)/(2*d)).toInt()
9private fun up(n: Long, d: Long) = Math.floorDiv(2*n+d,2*d).toInt()
10private fun signed(n: Int) = (if (n < 0) "-" else "")+"${abs(n)/10}.${abs(n)%10}"
11private fun index(n: Int?) = if (n == null) "none" else if (n < 0) "+"+signed(-n) else signed(n)
12private fun handicap(n: Int?) = if (n == null) "none" else if (n < 0) "+${-n}" else n.toString()
13private fun cents(n: Int) = "${n/100}.${(n%100).toString().padStart(2,'0')}"
14private fun average(total: Int, count: Int) = if (count == 0) "-" else cents(total*100/count)
15private fun sum(a: IntArray, from: Int = 0, to: Int = 18) = (from until to).sumOf { a[it] }

36private fun allocate(c: Course, h: Int) = IntArray(18) { i ->
37 if (h >= 0) h/18+(if (c.si[i] <= h%18) 1 else 0)
38 else -((-h)/18)-(if (c.si[i] >= 19-(-h)%18) 1 else 0)
39}
40private fun courseHandicap(c: Course, t: Tee, i: Int) = up(i.toLong()*t.slope+113L*(t.rating-10*c.total),1130)

46 fun player(id: String) = players[id] ?: throw Fault("unknown player")
47 fun course(id: String) = courses[id] ?: throw Fault("unknown course")
48 fun tee(c: Course, id: String) = c.tees[id] ?: throw Fault("unknown tee")
49 fun low(p: Player, day: Int) = p.records.filter { it.day >= day-365 && it.day < day }
50 .minWithOrNull(compareBy<IndexRecord> { it.value }.thenBy { it.day })
51 fun eligible(p: Player): Eligible {
52 val start = max(0,p.rounds.size-20)
53 val n = p.rounds.size-start
54 var reduction = 0
55 val adjusted = IntArray(n)
56 for (i in n-1 downTo 0) {
57 val r = p.rounds[start+i]
58 reduction += r.exceptional
59 adjusted[i] = r.diff-reduction
60 }
61 val k = when {
62 n < 3 -> 0
63 n <= 5 -> 1
64 n <= 8 -> 2
65 n <= 11 -> 3
66 n <= 14 -> 4
67 n <= 16 -> 5
68 n <= 18 -> 6
69 n == 19 -> 7
70 else -> 8
71 }
72 val adjustment = if (n == 3) -20 else if (n == 4 || n == 6) -10 else 0
73 val counting = (0 until n).sortedWith(compareBy<Int> { adjusted[it] }.thenByDescending { it }).take(k)
74 return Eligible(start,adjusted,counting,k,adjustment)
75 }
76 fun update(p: Player, day: Int) {
77 val e = eligible(p)
78 if (e.k == 0) {
79 p.calculation = null
80 return
81 }
82 val a = away(e.counting.sumOf { e.adjusted[it] }.toLong(),e.k.toLong())
83 var c = a+e.adjustment
84 val l = low(p,day)?.value
85 var soft = false
86 var hard = false
87 var maximum = false
88 if (l != null && c-l > 30) {
89 val next = l+30+(c-l-30)/2
90 soft = next != c
91 c = next
92 }
93 if (l != null && c-l > 50) {
94 c = l+50
95 hard = true
96 }
97 if (c > 540) {
98 c = 540
99 maximum = true
100 }
101 p.index = c
102 p.calculation = Calculation(e.k,a,e.adjustment,l,hard,soft,maximum)
103 }
104 fun data(e: Event): EventData {
105 val rounds = mutableListOf<Round>()
106 val n = e.players.size
107 val ph = IntArray(n)
108 for (i in 0 until n) {
109 val r = e.players[i].rounds.firstOrNull { it.course === e.course && it.tee === e.tee && it.day == e.day }
110 ?: throw Fault("missing round")
111 val h = r.handicap ?: throw Fault("no index")
112 rounds.add(r)
113 ph[i] = up(h.toLong()*e.allowance,100)
114 }
115 val minimum = ph.minOrNull()!!
116 val relative = e.format in listOf("MATCH","NASSAU","FOURBALL")
117 val strokes = IntArray(n) { ph[it]-(if (relative) minimum else 0) }
118 val nets = Array(n) { IntArray(18) }
119 val points = Array(n) { IntArray(18) }
120 val par = Array(n) { IntArray(18) }
121 for (i in 0 until n) {
122 val allocation = allocate(e.course,strokes[i])
123 for (h in 0 until 18) {
124 nets[i][h] = if (rounds[i].scores[h] == NO_NET) NO_NET else rounds[i].scores[h]-allocation[h]
125 points[i][h] = if (nets[i][h] == NO_NET) 0 else max(0,e.course.par[h]+2-nets[i][h])
126 par[i][h] = if (nets[i][h] == NO_NET || nets[i][h] > e.course.par[h]) -1 else if (nets[i][h] < e.course.par[h]) 1 else 0
127 }
128 }
129 return EventData(e,rounds,ph,strokes,nets,points,par)
130 }
131 fun rank(values: Array<IntArray>, candidates: List<Int>, high: Boolean, ids: List<String>) = candidates.sortedWith { a,b ->
132 var cmp = 0
133 for (start in intArrayOf(0,9,12,15,17)) {
134 cmp = sum(values[a],start).compareTo(sum(values[b],start))
135 if (cmp != 0) break
136 }
137 if (cmp == 0) ids[a].compareTo(ids[b]) else if (high) -cmp else cmp
138 }
139 fun ranked(d: EventData): List<Int> {
140 val candidates = d.ph.indices.filter { d.e.format != "STROKE" || d.rounds[it].scores.none { s -> s == NO_NET } }
141 val values = if (d.e.format == "STROKE") d.nets else if (d.e.format == "PAR") d.par else d.points
142 return rank(values,candidates,d.e.format != "STROKE",d.e.players.map { it.id })
143 }
144 fun status(lead: Int) = if (lead == 0) "AS" else if (lead > 0) "$lead UP" else "${-lead} DN"
145 fun match(wins: IntArray, first: Int, last: Int, a: String, b: String): Match {
146 var lead = 0
147 var remaining = 0
148 val statuses = mutableListOf<String>()
149 for (h in first..last) {
150 lead += wins[h]
151 remaining = last-h
152 statuses.add(status(lead))
153 if (abs(lead) > remaining) break
154 }
155 val result = if (lead == 0) "AS" else (if (lead > 0) a else b)+" ${abs(lead)}"+(if (remaining > 0) "&$remaining" else " UP")
156 return Match(statuses,result)
157 }
158 fun results(e: Event, d: EventData) {
159 output.add("results ${e.id}: ${e.format} ${e.course.id} ${e.tee.id} day ${e.day} allowance ${e.allowance}")
160 val n = e.players.size
161 val ids = e.players.map { it.id }
162 when (e.format) {
163 "STROKE", "STABLEFORD", "PAR" -> {
164 val order = ranked(d)
165 for ((r,i) in order.withIndex()) {
166 val value = when (e.format) {
167 "STROKE" -> "net ${sum(d.nets[i])} gross ${sum(d.rounds[i].scores)}"
168 "STABLEFORD" -> "points ${sum(d.points[i])}"
169 else -> status(sum(d.par[i]))
170 }
171 output.add("${r+1} ${ids[i]} $value hcp ${handicap(d.ph[i])}")
172 }
173 if (e.format == "STROKE") {
174 for (i in ids.indices.filter { it !in order }.sortedBy { ids[it] }) output.add("NR ${ids[i]} hcp ${handicap(d.ph[i])}")
175 if (order.isNotEmpty()) {
176 val gross = Array(n) { d.rounds[it].scores }
177 val winner = rank(gross,order,false,ids)[0]
178 output.add("gross ${ids[winner]} ${sum(gross[winner])}")
179 }
180 }
181 }
182 "MATCH", "NASSAU", "FOURBALL" -> {
183 output.add("strokes"+ids.indices.joinToString("") { " ${ids[it]} ${d.strokes[it]}" })
184 val size = n/2
185 val a = ids.take(size).joinToString("+")
186 val b = ids.drop(size).joinToString("+")
187 val wins = IntArray(18) { h ->
188 val left = (0 until size).minOf { d.nets[it][h] }
189 val right = (size until n).minOf { d.nets[it][h] }
190 right.compareTo(left)
191 }
192 if (e.format == "NASSAU") {
193 output.add("front ${match(wins,0,8,a,b).result}")
194 output.add("back ${match(wins,9,17,a,b).result}")
195 output.add("total ${match(wins,0,17,a,b).result}")
196 } else {
197 val m = match(wins,0,17,a,b)
198 output.add("status "+m.statuses.joinToString(", "))
199 output.add("result ${m.result}")
200 }
201 }
202 "BETTERBALL" -> {
203 val values = Array(n/2) { i -> IntArray(18) { h -> max(d.points[2*i][h],d.points[2*i+1][h]) } }
204 val order = rank(values,(0 until n/2).toList(),true,(0 until n/2).map { ids[2*it] })
205 for ((r,i) in order.withIndex()) output.add("${r+1} ${ids[2*i]}+${ids[2*i+1]} points ${sum(values[i])} hcp ${handicap(d.ph[2*i])} ${handicap(d.ph[2*i+1])}")
206 }
207 "SKINS" -> {
208 var carry = 0
209 val totals = IntArray(n)
210 var tied = ids.indices.toList()
211 for (h in 0 until 18) {
212 carry++
213 val minimum = ids.indices.minOf { d.nets[it][h] }
214 tied = ids.indices.filter { d.nets[it][h] == minimum }
215 if (minimum != NO_NET && tied.size == 1) {
216 val i = tied[0]
217 output.add("hole ${h+1} ${ids[i]} $carry")
218 totals[i] += carry*100
219 carry = 0
220 }
221 }
222 if (carry > 0) {
223 tied = tied.sortedBy { ids[it] }
224 for (i in tied) totals[i] += carry*100/tied.size
225 output.add("carry $carry"+tied.joinToString("") { " ${ids[it]}" })
226 }
227 val order = ids.indices.sortedWith(compareByDescending<Int> { totals[it] }.thenBy { ids[it] })
228 for (i in order) output.add("${ids[i]} ${cents(totals[i])}")
229 }
230 }
231 }
232 fun cardTotal(label: String, r: Round, strokes: IntArray, nets: IntArray, points: IntArray, from: Int, to: Int) {
233 val complete = (from until to).none { r.scores[it] == NO_NET }
234 output.add("$label ${sum(r.course.par,from,to)} "+(if (complete) sum(r.scores,from,to).toString() else "-")+
235 " ${sum(r.adjusted,from,to)} ${sum(strokes,from,to)} "+(if (complete) sum(nets,from,to).toString() else "-")+" ${sum(points,from,to)}")
236 }
237 fun card(p: Player, number: Int, r: Round) {
238 output.add("card ${p.id} $number: ${r.course.id} ${r.tee.id} day ${r.day} handicap ${handicap(r.handicap)}")
239 val strokes = allocate(r.course,r.handicap ?: 0)
240 val nets = IntArray(18) { if (r.scores[it] == NO_NET) NO_NET else r.scores[it]-strokes[it] }
241 val points = IntArray(18) { if (nets[it] == NO_NET) 0 else max(0,r.course.par[it]+2-nets[it]) }
242 for (h in 0 until 18) {
243 val net = if (nets[h] == NO_NET) "-" else nets[h].toString()
244 output.add("${h+1} ${r.course.par[h]} ${r.course.si[h]} ${r.raw[h]} ${r.adjusted[h]} ${strokes[h]} $net ${points[h]}")
245 if (h == 8) cardTotal("OUT",r,strokes,nets,points,0,9)
246 if (h == 17) {
247 cardTotal("IN",r,strokes,nets,points,9,18)
248 cardTotal("TOTAL",r,strokes,nets,points,0,18)
249 }
250 }
251 output.add("ags ${r.ags} diff ${signed(r.diff)}")
252 }
253 fun statistics(p: Player) {
254 val category = IntArray(7)
255 val totals = IntArray(3)
256 val counts = IntArray(3)
257 val complete = mutableListOf<Int>()
258 for (r in p.rounds) {
259 var full = true
260 for (h in 0 until 18) {
261 if (r.raw[h] == "X") category[6]++
262 if (r.scores[h] == NO_NET) {
263 full = false
264 continue
265 }
266 val relative = r.scores[h]-r.course.par[h]
267 val k = if (relative <= -2) 0 else if (relative <= 2) relative+2 else 5
268 category[k]++
269 val i = r.course.par[h]-3
270 totals[i] += r.scores[h]
271 counts[i]++
272 }
273 if (full) complete.add(sum(r.scores))
274 }
275 var line = "stats ${p.id}: rounds ${p.rounds.size} best ${complete.minOrNull() ?: "-"} average ${average(complete.sum(),complete.size)}"
276 val labels = listOf("eagles","birdies","pars","bogeys","doubles","others","pickups")
277 for (i in 0..6) line += " ${labels[i]} ${category[i]}"
278 for (i in 0..2) line += " par${i+3} ${average(totals[i],counts[i])}"
279 output.add(line)
280 }
281 fun execute(t: List<String>) {
282 when (t[0]) {
283 "COURSE" -> {
284 error(t[1] in courses,"duplicate course")
285 val par = IntArray(18) { t[it+2].toInt() }
286 val si = IntArray(18) { t[it+20].toInt() }
287 error(par.any { it !in 3..5 } || si.toSet().size != 18 || si.any { it !in 1..18 },"invalid course")
288 courses[t[1]] = Course(t[1],par,si)
289 }
290 "TEE" -> {
291 val c = course(t[1])
292 error(t[2] in c.tees,"duplicate tee")
293 val rating = t[3].toInt()
294 val slope = t[4].toInt()
295 error(rating !in 500..900 || slope !in 55..155,"invalid tee")
296 c.tees[t[2]] = Tee(t[2],rating,slope)
297 }
298 "PLAYER" -> {
299 error(t[1] in players,"duplicate player")
300 val i = if (t.size == 4) t[3].toInt() else null
301 error(i != null && i !in -100..540,"invalid index")
302 players[t[1]] = Player(t[1],t[2],i)
303 }
304 "ROUND" -> {
305 val p = player(t[1])
306 val c = course(t[2])
307 val tee = tee(c,t[3])
308 val day = t[4].toInt()
309 error(p.rounds.isNotEmpty() && day < p.rounds.last().day,"invalid day")
310 val raw = t.drop(5)
311 error(raw.count { it == "-" } >= 5,"too few holes")
312 val previous = p.index
313 val hcp = previous?.let { courseHandicap(c,tee,it) }
314 val strokes = allocate(c,hcp ?: 0)
315 val scores = IntArray(18) { raw[it].toIntOrNull() ?: NO_NET }
316 val adjusted = IntArray(18) { h ->
317 val limit = c.par[h]+(if (previous == null) 5 else 2+strokes[h])
318 when (raw[h]) {
319 "-" -> c.par[h]+strokes[h]
320 "X" -> limit
321 else -> min(scores[h],limit)
322 }
323 }
324 val ags = adjusted.sum()
325 val diff = away(113L*(10*ags-tee.rating),tee.slope.toLong())
326 val exceptional = if (previous != null && previous-diff >= 70) (if (previous-diff >= 100) 20 else 10) else 0
327 val r = Round(c,tee,day,raw,hcp,scores,adjusted,ags,diff,exceptional,previous,p.calculation,p.records.size)
328 p.rounds.add(r)
329 update(p,day)
330 p.index?.let { p.records.add(IndexRecord(day,it)) }
331 output.add("round ${p.id} ${p.rounds.size}: ags $ags diff ${signed(diff)}"+(if (exceptional == 0) "" else " exceptional ${signed(-exceptional)}")+" index ${index(p.index)}")
332 }
333 "REMOVE" -> {
334 val p = player(t[1])
335 error(p.rounds.isEmpty(),"no rounds")
336 val number = p.rounds.size
337 val r = p.rounds.removeAt(number-1)
338 p.index = r.oldIndex
339 p.calculation = r.oldCalculation
340 p.records.subList(r.oldRecords,p.records.size).clear()
341 output.add("removed ${p.id} $number: index ${index(p.index)}")
342 }
343 "INDEX" -> {
344 val p = player(t[1])
345 var line = "index ${p.id}: ${index(p.index)} rounds ${p.rounds.size}"
346 val c = p.calculation
347 if (c != null) line += " lowest ${c.k} average ${signed(c.average)} adjustment ${signed(c.adjustment)}"+
348 (if (c.low == null) "" else " low ${index(c.low)}")+(if (c.hard) " hard cap" else if (c.soft) " soft cap" else "")+(if (c.maximum) " maximum" else "")
349 output.add(line)
350 }
351 "HISTORY" -> {
352 val p = player(t[1])
353 val e = eligible(p)
354 output.add("history ${p.id}: rounds ${p.rounds.size}")
355 for (i in e.adjusted.indices) {
356 val r = p.rounds[e.start+i]
357 output.add("${e.start+i+1} day ${r.day} ${r.course.id} ${r.tee.id} ags ${r.ags} diff ${signed(r.diff)}"+
358 (if (r.diff == e.adjusted[i]) "" else " adjusted ${signed(e.adjusted[i])}")+(if (i in e.counting) " *" else ""))
359 }
360 }
361 "LOWINDEX" -> {
362 val p = player(t[1])
363 val day = t[2].toInt()
364 val r = low(p,day)
365 output.add("lowindex ${p.id} $day: "+(if (r == null) "none" else "${index(r.value)} day ${r.day}"))
366 }
367 "COURSEHCP" -> {
368 val p = player(t[1])
369 val c = course(t[2])
370 val tee = tee(c,t[3])
371 val h = courseHandicap(c,tee,p.index ?: throw Fault("no index"))
372 output.add("coursehcp ${p.id} ${c.id} ${tee.id}: ${handicap(h)} strokes "+allocate(c,h).joinToString(" "))
373 }
374 "HCPTABLE" -> {
375 val c = course(t[1])
376 val tee = tee(c,t[2])
377 val first = t[3].toInt()
378 val last = t[4].toInt()
379 error(first < -100 || first > last || last > 540,"invalid index")
380 output.add("hcptable ${c.id} ${tee.id}: ${index(first)} to ${index(last)}")
381 var start = first
382 var h = courseHandicap(c,tee,first)
383 for (i in first+1..last+1) {
384 val next = if (i <= last) courseHandicap(c,tee,i) else h+1
385 if (next != h) {
386 output.add("${index(start)} to ${index(i-1)}: ${handicap(h)}")
387 start = i
388 h = next
389 }
390 }
391 }
392 "CARD" -> {
393 val p = player(t[1])
394 val n = t[2].toInt()
395 error(n > p.rounds.size,"unknown round")
396 card(p,n,p.rounds[n-1])
397 }
398 "STATS" -> statistics(player(t[1]))
399 "HOLESTATS" -> {
400 val c = course(t[1])
401 val rounds = players.values.flatMap { it.rounds }.filter { it.course === c }
402 val totals = IntArray(18)
403 val counts = IntArray(18)
404 val ranks = IntArray(18)
405 for (r in rounds) {
406 for (h in 0 until 18) {
407 if (r.scores[h] != NO_NET) {
408 totals[h] += r.scores[h]
409 counts[h]++
410 }
411 }
412 }
413 val order = (0 until 18).sortedWith { a,b ->
414 if (counts[a] == 0 || counts[b] == 0) {
415 if (counts[a] == counts[b]) a.compareTo(b) else if (counts[a] == 0) 1 else -1
416 } else {
417 val cmp = ((totals[b]-c.par[b]*counts[b]).toLong()*counts[a]).compareTo((totals[a]-c.par[a]*counts[a]).toLong()*counts[b])
418 if (cmp == 0) a.compareTo(b) else cmp
419 }
420 }
421 for ((i,h) in order.withIndex()) ranks[h] = i+1
422 output.add("holestats ${c.id}: rounds ${rounds.size}")
423 for (h in 0 until 18) output.add("${h+1} par ${c.par[h]} si ${c.si[h]} scores ${counts[h]} avg ${average(totals[h],counts[h])} rank ${ranks[h]}")
424 }
425 "PLAYERS" -> {
426 val order = players.values.sortedWith(compareBy<Player> { it.index == null }.thenBy { it.index }.thenBy { it.id })
427 output.add("players ${order.size}")
428 for (p in order) output.add("${p.id} ${p.name} ${index(p.index)} rounds ${p.rounds.size}")
429 }
430 "EVENT" -> {
431 error(t[1] in events,"duplicate event")
432 val c = course(t[3])
433 val tee = tee(c,t[4])
434 val allowance = t[6].toInt()
435 error(allowance !in 1..100,"invalid allowance")
436 val list = t.drop(7).map { player(it) }
437 val n = list.size
438 val valid = when (t[2]) {
439 "MATCH", "NASSAU" -> n == 2
440 "FOURBALL" -> n == 4
441 "BETTERBALL" -> n in listOf(4,6,8)
442 else -> n in 2..8
443 }
444 error(list.toSet().size != n || !valid,"invalid players")
445 events[t[1]] = Event(t[1],t[2],c,tee,t[5].toInt(),allowance,list)
446 }
447 "RESULTS" -> {
448 val e = events[t[1]] ?: throw Fault("unknown event")
449 results(e,data(e))
450 }
451 "MERIT" -> {
452 val merit = mutableMapOf<String,IntArray>()
453 val awards = intArrayOf(10,8,6,5,4,3,2,1)
454 var count = 0
455 for (e in events.values) {
456 if (e.format !in listOf("STROKE","STABLEFORD","PAR")) continue
457 val d = try { data(e) } catch (_: Fault) { continue }
458 count++
459 for (p in e.players) merit.getOrPut(p.id) { IntArray(3) }[2]++
460 for ((i,p) in ranked(d).withIndex()) {
461 val a = merit.getValue(e.players[p].id)
462 a[0] += awards[i]
463 if (i == 0) a[1]++
464 }
465 }
466 val order = merit.keys.sortedWith(compareByDescending<String> { merit.getValue(it)[0] }.thenByDescending { merit.getValue(it)[1] }.thenBy { it })
467 output.add("merit: events $count")
468 for ((i,id) in order.withIndex()) {
469 val a = merit.getValue(id)
470 output.add("${i+1} $id ${a[0]} wins ${a[1]} played ${a[2]}")
471 }
472 }
473 }
474 }
475 fun process(line: String): List<String> {
476 output = mutableListOf()
