1class CommandError < StandardError; end

3def half_up(n, d)
4 (2 * n + d).div(2 * d)
5end
6
7def half_away(n, d)
8 n.negative? ? -half_up(-n, d) : half_up(n, d)
9end
10
11def decimal(n, sign = '-')
12 (n.negative? ? sign : '') + "#{n.abs / 10}.#{n.abs % 10}"
13end
14
15def index_text(n)
16 n.nil? ? 'none' : decimal(n, '+')
17end
18
19def hcp_text(n)
20 n.nil? ? 'none' : n.negative? ? "+#{-n}" : n.to_s
21end
22
23def hundredths(n)
24 format('%d.%02d', n / 100, n % 100)
25end
26
27def average_text(total, count)
28 count.zero? ? '-' : hundredths(total * 100 / count)
29end
30
31def status(n)
32 n.zero? ? 'AS' : "#{n.abs}#{n.positive? ? ' UP' : ' DN'}"
33end
34
35def allocation(h, si)
36 q, r = h.abs.divmod(18)
37 si.map { |i| h >= 0 ? q + (i <= r ? 1 : 0) : -q - (i >= 19 - r ? 1 : 0) }
38end
39
40def handicap(index, course, tee)
41 half_up(index * tee.slope + 113 * (tee.rating - 10 * course.par), 1130)
42end
43
44def ranked(ids, values, larger = false)
45 ids.sort_by do |id|
46 v = values[id]
47 counts = [v.sum, v[9..].sum, v[12..].sum, v[15..].sum, v[17]]
48 [larger ? counts.map { |n| -n } : counts, id]
49 end
50end

56class Player
57 attr_reader :id, :name
58 attr_accessor :index, :rounds, :records, :calculation
59
60 def initialize(id, name, index)
61 @id = id
62 @name = name
63 @index = index
64 @rounds = []
65 @records = []
66 @calculation = nil
67 end
68
69 def low(day)
70 @records.select { |d, _| d >= day - 365 && d < day }.min_by { |d, i| [i, d] }
71 end
72
73 def eligible
74 reduction = 0
75 rows = []
76 (@rounds.length - 1).downto([0, @rounds.length - 20].max) do |i|
77 r = @rounds[i]
78 reduction += r.exceptional
79 rows << [i + 1, r, r.diff - reduction]
80 end
81 rows.reverse
82 end
83
84 def counting(rows)
85 n = rows.length
86 return [[], 0] if n < 3
87 k = [5, 8, 11, 14, 16, 18, 19, 20].index { |limit| n <= limit } + 1
88 adjustment = n == 3 ? -20 : [4, 6].include?(n) ? -10 : 0
89 [rows.sort_by { |number, _, adjusted| [adjusted, -number] }.first(k), adjustment]
90 end
91
92 def update(day)
93 selected, adjustment = counting(eligible)
94 if selected.empty?
95 @calculation = nil
96 return
97 end
98 average = half_away(selected.sum { |row| row[2] }, selected.length)
99 candidate = average + adjustment
100 record = low(day)
101 low_index = record&.last
102 soft = false
103 hard = false
104 maximum = false
105 unless low_index.nil?
106 if candidate - low_index > 30
107 candidate = low_index + 30 + (candidate - low_index - 30) / 2
108 soft = true
109 end
110 if candidate - low_index > 50
111 candidate = low_index + 50
112 hard = true
113 end
114 end
115 if candidate > 540
116 candidate = 540
117 maximum = true
118 end
119 @index = candidate
120 @calculation = {k: selected.length, average: average, adjustment: adjustment,
121 low: low_index, hard: hard, soft: soft, maximum: maximum}
122 end
123end

125class GolfRound
126 attr_reader :course, :tee, :day, :scores, :previous_index, :previous_records,
127 :previous_calculation, :hcp, :adjusted, :ags, :diff, :exceptional
128
129 def initialize(player, course, tee, day, scores)
130 @course = course
131 @tee = tee
132 @day = day
133 @scores = scores
134 @previous_index = player.index
135 @previous_records = player.records.length
136 @previous_calculation = player.calculation
137 @hcp = player.index.nil? ? nil : handicap(player.index, course, tee)
138 strokes = allocation(@hcp || 0, course.si)
139 @adjusted = scores.each_with_index.map do |score, h|
140 limit = @hcp.nil? ? course.pars[h] + 5 : course.pars[h] + 2 + strokes[h]
141 score == '-' ? course.pars[h] + strokes[h] : score == 'X' ? limit : [score, limit].min
142 end
143 @ags = @adjusted.sum
144 @diff = half_away(113 * (10 * @ags - tee.rating), tee.slope)
145 @exceptional = if !player.index.nil? && player.index - @diff >= 70
146 player.index - @diff >= 100 ? 20 : 10
147 else
148 0
149 end
150 end
151
152 def nets(strokes)
153 @scores.each_with_index.map { |s, h| s.is_a?(Integer) ? s - strokes[h] : nil }
154 end
155
156 def points(nets)
157 nets.each_with_index.map { |n, h| n.nil? ? 0 : [0, @course.pars[h] + 2 - n].max }
158 end
159
160 def complete?
161 @scores.all? { |s| s.is_a?(Integer) }
162 end
163end

165class GolfScorer
166 attr_reader :output
167
168 def initialize
169 @courses = {}
170 @players = {}
171 @events = {}
172 @output = []
173 end
174
175 def emit(*parts)
176 @output << parts.join(' ')

179 def player(id)
180 raise CommandError, 'unknown player' unless @players.key?(id)
181 @players[id]
182 end
183
184 def course(id)
185 raise CommandError, 'unknown course' unless @courses.key?(id)
186 @courses[id]
187 end
188
189 def tee(cid, tid)
190 c = course(cid)
191 raise CommandError, 'unknown tee' unless c.tees.key?(tid)
192 [c, c.tees[tid]]
193 end

195 def event_data(e)
196 rounds = {}
197 hcp = {}
198 nets = {}
199 points = {}
200 e.players.each do |p|
201 r = p.rounds.find { |r| r.course.equal?(e.course) && r.tee.equal?(e.tee) && r.day == e.day }
202 raise CommandError, 'missing round' unless r
203 raise CommandError, 'no index' if r.hcp.nil?
204 rounds[p.id] = r
205 hcp[p.id] = half_up(r.hcp * e.allowance, 100)
206 end
207 minimum = %w[MATCH NASSAU FOURBALL].include?(e.form) ? hcp.values.min : 0
208 e.players.each do |p|
209 nets[p.id] = rounds[p.id].nets(allocation(hcp[p.id] - minimum, e.course.si))
210 points[p.id] = rounds[p.id].points(nets[p.id])
211 end
212 {rounds: rounds, hcp: hcp, nets: nets, points: points}
213 end
214
215 def event_ranking(e, d)
216 ids = e.players.map(&:id)
217 if e.form == 'STROKE'
218 complete = ids.select { |id| d[:rounds][id].complete? }
219 [ranked(complete, d[:nets]), d[:nets]]
220 elsif e.form == 'STABLEFORD'
221 [ranked(ids, d[:points], true), d[:points]]
222 else
223 values = ids.to_h do |id|
224 [id, d[:nets][id].each_with_index.map { |n, h| n.nil? || n > e.course.pars[h] ? -1 : n < e.course.pars[h] ? 1 : 0 }]
225 end
226 [ranked(ids, values, true), values]
227 end
228 end
229
230 def individual(e, d)
231 order, values = event_ranking(e, d)
232 order.each_with_index do |id, i|
233 if e.form == 'STROKE'
234 emit(i + 1, id, 'net', values[id].sum, 'gross', d[:rounds][id].scores.sum, 'hcp', hcp_text(d[:hcp][id]))
235 elsif e.form == 'STABLEFORD'
236 emit(i + 1, id, 'points', values[id].sum, 'hcp', hcp_text(d[:hcp][id]))
237 else
238 emit(i + 1, id, status(values[id].sum), 'hcp', hcp_text(d[:hcp][id]))
239 end
240 end
241 return unless e.form == 'STROKE'
242 e.players.map(&:id).reject { |id| d[:rounds][id].complete? }.sort.each do |id|
243 emit('NR', id, 'hcp', hcp_text(d[:hcp][id]))
244 end
245 unless order.empty?
246 gross = order.to_h { |id| [id, d[:rounds][id].scores] }
247 winner = ranked(order, gross).first
248 emit('gross', winner, gross[winner].sum)
249 end
250 end

252 def play_match(names, holes, first, last)
253 lead = 0
254 remaining = last - first
255 statuses = []
256 (first...last).each do |h|
257 lead += holes[h]
258 statuses << status(lead)
259 remaining = last - h - 1
260 break if lead.abs > remaining || remaining.zero?
261 end
262 result = if lead.zero?
263 'AS'
264 else
265 names[lead.positive? ? 0 : 1] + ' ' + lead.abs.to_s + (remaining.positive? ? "&#{remaining}" : ' UP')
266 end
267 [statuses, result]
268 end
269
270 def matches(e, d)
271 ids = e.players.map(&:id)
272 sides = e.form == 'FOURBALL' ? [ids.first(2), ids.drop(2)] : [[ids[0]], [ids[1]]]
273 names = sides.map { |s| s.join('+') }
274 holes = (0...18).map do |h|
275 a, b = sides.map { |side| side.filter_map { |id| d[:nets][id][h] }.min }
276 a.nil? ? b.nil? ? 0 : -1 : b.nil? || a < b ? 1 : a > b ? -1 : 0
277 end
278 minimum = d[:hcp].values.min
279 parts = ['strokes']
280 ids.each { |id| parts.push(id, d[:hcp][id] - minimum) }
281 emit(*parts)
282 if e.form == 'NASSAU'
283 [['front', 0, 9], ['back', 9, 18], ['total', 0, 18]].each do |label, first, last|
284 emit(label, play_match(names, holes, first, last)[1])
285 end
286 else
287 statuses, result = play_match(names, holes, 0, 18)
288 emit('status', statuses.join(', '))
289 emit('result', result)
290 end
291 end

293 def betterball(e, d)
294 teams = {}
295 values = {}
296 e.players.each_slice(2) do |a, b|
297 teams[a.id] = b.id
298 values[a.id] = (0...18).map { |h| [d[:points][a.id][h], d[:points][b.id][h]].max }
299 end
300 ranked(teams.keys, values, true).each_with_index do |a, i|
301 b = teams[a]
302 emit(i + 1, "#{a}+#{b}", 'points', values[a].sum, 'hcp', hcp_text(d[:hcp][a]), hcp_text(d[:hcp][b]))
303 end
304 end

306 def skins(e, d)
307 ids = e.players.map(&:id)
308 totals = ids.to_h { |id| [id, 0] }
309 carry = 0
310 winners = []
311 18.times do |h|
312 carry += 1
313 available = ids.reject { |id| d[:nets][id][h].nil? }
314 best = available.map { |id| d[:nets][id][h] }.min
315 winners = available.select { |id| d[:nets][id][h] == best }
316 if winners.length == 1
317 emit('hole', h + 1, winners[0], carry)
318 totals[winners[0]] += 100 * carry
319 carry = 0
320 end
321 end
322 if carry.positive?
323 sharing = (winners.empty? ? ids : winners).sort
324 share = carry * 100 / sharing.length
325 sharing.each { |id| totals[id] += share }
326 emit('carry', carry, *sharing)
327 end
328 ids.sort_by { |id| [-totals[id], id] }.each { |id| emit(id, hundredths(totals[id])) }
329 end

331 def card_total(label, r, strokes, nets, points, first, last)
332 scores = r.scores[first...last]
333 complete = scores.all? { |s| s.is_a?(Integer) }
334 emit(label, r.course.pars[first...last].sum, complete ? scores.sum : '-',
335 r.adjusted[first...last].sum, strokes[first...last].sum,
336 complete ? nets[first...last].sum : '-', points[first...last].sum)
337 end

339 def command(cmd, a)
340 case cmd
341 when 'COURSE'
342 raise CommandError, 'duplicate course' if @courses.key?(a[0])
343 pars = a[1, 18].map(&:to_i)
344 si = a[19, 18].map(&:to_i)
345 raise CommandError, 'invalid course' unless pars.all? { |p| [3, 4, 5].include?(p) } && si.sort == (1..18).to_a
346 @courses[a[0]] = Course.new(a[0], pars, si, pars.sum, {})
347 when 'TEE'
348 c = course(a[0])
349 raise CommandError, 'duplicate tee' if c.tees.key?(a[1])
350 rating, slope = a[2, 2].map(&:to_i)
351 raise CommandError, 'invalid tee' unless (500..900).cover?(rating) && (55..155).cover?(slope)
352 c.tees[a[1]] = Tee.new(a[1], rating, slope)
353 when 'PLAYER'
354 raise CommandError, 'duplicate player' if @players.key?(a[0])
355 index = a.length == 3 ? a[2].to_i : nil
356 raise CommandError, 'invalid index' if !index.nil? && !(-100..540).cover?(index)
357 @players[a[0]] = Player.new(a[0], a[1], index)
358 when 'ROUND'
359 p = player(a[0])
360 c, t = tee(a[1], a[2])
361 day = a[3].to_i
362 raise CommandError, 'invalid day' if !p.rounds.empty? && day < p.rounds.last.day
363 scores = a.drop(4).map { |s| %w[X -].include?(s) ? s : s.to_i }
364 raise CommandError, 'too few holes' if scores.count('-') >= 5
365 r = GolfRound.new(p, c, t, day, scores)
366 p.rounds << r
367 p.update(day)
368 p.records << [day, p.index] unless p.index.nil?
369 parts = ['round', p.id, "#{p.rounds.length}:", 'ags', r.ags, 'diff', decimal(r.diff)]
370 parts.push('exceptional', decimal(-r.exceptional)) unless r.exceptional.zero?
371 emit(*parts, 'index', index_text(p.index))
372 when 'REMOVE'
373 p = player(a[0])
374 raise CommandError, 'no rounds' if p.rounds.empty?
375 number = p.rounds.length
376 r = p.rounds.pop
377 p.index = r.previous_index
378 p.records = p.records.first(r.previous_records)
379 p.calculation = r.previous_calculation
380 emit('removed', p.id, "#{number}:", 'index', index_text(p.index))
381 when 'INDEX'
382 p = player(a[0])
383 parts = ['index', "#{p.id}:", index_text(p.index), 'rounds', p.rounds.length]
384 c = p.calculation
385 if c
386 parts.push('lowest', c[:k], 'average', decimal(c[:average]), 'adjustment', decimal(c[:adjustment]))
387 parts.push('low', index_text(c[:low])) unless c[:low].nil?
388 if c[:hard]
389 parts.push('hard', 'cap')
390 elsif c[:soft]
391 parts.push('soft', 'cap')
392 end
393 parts << 'maximum' if c[:maximum]
394 end
395 emit(*parts)
396 when 'HISTORY'
397 p = player(a[0])
398 rows = p.eligible
399 counted = p.counting(rows)[0].map(&:first)
400 emit('history', "#{p.id}:", 'rounds', p.rounds.length)
401 rows.each do |number, r, adjusted|
402 parts = [number, 'day', r.day, r.course.id, r.tee.id, 'ags', r.ags, 'diff', decimal(r.diff)]
403 parts.push('adjusted', decimal(adjusted)) if adjusted != r.diff
404 parts << '*' if counted.include?(number)
405 emit(*parts)
406 end
407 when 'LOWINDEX'
408 p = player(a[0])
409 day = a[1].to_i
410 low = p.low(day)
411 parts = ['lowindex', p.id, "#{day}:"]
412 low ? parts.push(index_text(low[1]), 'day', low[0]) : parts.push('none')
413 emit(*parts)
414 when 'COURSEHCP'
415 p = player(a[0])
416 c, t = tee(a[1], a[2])
417 raise CommandError, 'no index' if p.index.nil?
418 h = handicap(p.index, c, t)
419 emit('coursehcp', p.id, c.id, "#{t.id}:", hcp_text(h), 'strokes', *allocation(h, c.si))
420 when 'HCPTABLE'
421 c, t = tee(a[0], a[1])
422 first, last = a[2, 2].map(&:to_i)
423 raise CommandError, 'invalid index' unless -100 <= first && first <= last && last <= 540
424 emit('hcptable', c.id, "#{t.id}:", index_text(first), 'to', index_text(last))
425 start = first
426 h = handicap(first, c, t)
427 ((first + 1)..last).each do |i|
428 following = handicap(i, c, t)
429 next if following == h
430 emit(index_text(start), 'to', index_text(i - 1) + ':', hcp_text(h))
431 start = i
432 h = following
433 end
434 emit(index_text(start), 'to', index_text(last) + ':', hcp_text(h))
435 when 'CARD'
436 p = player(a[0])
437 number = a[1].to_i
438 raise CommandError, 'unknown round' if number > p.rounds.length
439 r = p.rounds[number - 1]
440 c = r.course
441 strokes = allocation(r.hcp || 0, c.si)
442 nets = r.nets(strokes)
443 points = r.points(nets)
444 emit('card', p.id, "#{number}:", c.id, r.tee.id, 'day', r.day, 'handicap', hcp_text(r.hcp))
445 18.times do |h|
446 emit(h + 1, c.pars[h], c.si[h], r.scores[h], r.adjusted[h], strokes[h], nets[h].nil? ? '-' : nets[h], points[h])
447 card_total('OUT', r, strokes, nets, points, 0, 9) if h == 8
448 if h == 17
449 card_total('IN', r, strokes, nets, points, 9, 18)
450 card_total('TOTAL', r, strokes, nets, points, 0, 18)
451 end
452 end
453 emit('ags', r.ags, 'diff', decimal(r.diff))
454 when 'STATS'
455 p = player(a[0])
456 labels = %w[eagles birdies pars bogeys doubles others pickups]
457 counts = Array.new(7, 0)
458 par_scores = {3 => [], 4 => [], 5 => []}
459 totals = []
460 p.rounds.each do |r|
461 totals << r.scores.sum if r.complete?
462 r.scores.each_with_index do |score, h|
463 if score == 'X'
464 counts[6] += 1
465 elsif score.is_a?(Integer)
466 relative = score - r.course.pars[h]
467 category = (relative + 2).clamp(0, 5)
468 counts[category] += 1
469 par_scores[r.course.pars[h]] << score
470 end
471 end
472 end
473 parts = ['stats', "#{p.id}:", 'rounds', p.rounds.length, 'best', totals.empty? ? '-' : totals.min,
474 'average', average_text(totals.sum, totals.length)]
475 labels.each_with_index { |label, i| parts.push(label, counts[i]) }
476 [3, 4, 5].each { |par| parts.push("par#{par}", average_text(par_scores[par].sum, par_scores[par].length)) }
477 emit(*parts)
478 when 'HOLESTATS'
479 c = course(a[0])
480 rounds = @players.values.flat_map(&:rounds).select { |r| r.course.equal?(c) }
481 scores = (0...18).map { |h| rounds.map { |r| r.scores[h] }.select { |s| s.is_a?(Integer) } }
482 order = (0...18).sort_by do |h|
483 values = scores[h]
484 [values.empty? ? 1 : 0, values.empty? ? 0 : -Rational(values.sum - c.pars[h] * values.length, values.length), h]
485 end
486 emit('holestats', "#{c.id}:", 'rounds', rounds.length)
487 18.times do |h|
488 emit(h + 1, 'par', c.pars[h], 'si', c.si[h], 'scores', scores[h].length,
489 'avg', average_text(scores[h].sum, scores[h].length), 'rank', order.index(h) + 1)
490 end
491 when 'PLAYERS'
492 players = @players.values.sort_by { |p| [p.index.nil? ? 1 : 0, p.index || 0, p.id] }
493 emit('players', players.length)
494 players.each { |p| emit(p.id, p.name, index_text(p.index), 'rounds', p.rounds.length) }
495 when 'EVENT'
496 raise CommandError, 'duplicate event' if @events.key?(a[0])
497 c, t = tee(a[2], a[3])
498 allowance = a[5].to_i
499 raise CommandError, 'invalid allowance' unless (1..100).cover?(allowance)
500 ids = a.drop(6)
501 raise CommandError, 'unknown player' unless ids.all? { |id| @players.key?(id) }
502 n = ids.length
503 valid = if %w[MATCH NASSAU].include?(a[1])
504 n == 2
505 elsif a[1] == 'FOURBALL'
506 n == 4
507 elsif a[1] == 'BETTERBALL'
508 [4, 6, 8].include?(n)
509 else
510 (2..8).cover?(n)
511 end
512 raise CommandError, 'invalid players' unless valid && ids.uniq.length == n
513 @events[a[0]] = Event.new(a[0], a[1], c, t, a[4].to_i, allowance, ids.map { |id| @players[id] })
514 when 'RESULTS'
515 e = @events[a[0]]
516 raise CommandError, 'unknown event' unless e
517 d = event_data(e)
518 emit('results', "#{e.id}:", e.form, e.course.id, e.tee.id, 'day', e.day, 'allowance', e.allowance)
519 if %w[STROKE STABLEFORD PAR].include?(e.form)
520 individual(e, d)
521 elsif %w[MATCH NASSAU FOURBALL].include?(e.form)
522 matches(e, d)
523 elsif e.form == 'BETTERBALL'
524 betterball(e, d)
525 else
526 skins(e, d)
527 end
528 when 'MERIT'
529 totals = {}
530 awards = [10, 8, 6, 5, 4, 3, 2, 1]
531 counted = 0
532 @events.each_value do |e|
533 next unless %w[STROKE STABLEFORD PAR].include?(e.form)
534 begin
535 d = event_data(e)
536 rescue CommandError
537 next
538 end
539 counted += 1
540 e.players.each do |p|
541 totals[p.id] ||= [0, 0, 0]
542 totals[p.id][2] += 1
543 end
544 event_ranking(e, d)[0].each_with_index do |id, rank|
545 totals[id][0] += awards[rank]
546 totals[id][1] += 1 if rank.zero?
547 end
548 end
549 emit('merit:', 'events', counted)
550 totals.keys.sort_by { |id| [-totals[id][0], -totals[id][1], id] }.each_with_index do |id, i|
551 points, wins, played = totals[id]
552 emit(i + 1, id, points, 'wins', wins, 'played', played)
553 end
554 end
555 end
556
557 def execute(line)
558 cmd, *args = line.split
559 begin
560 command(cmd, args)
561 rescue CommandError => error
562 emit('ERROR', error.message)
563 end
564 end
565end
