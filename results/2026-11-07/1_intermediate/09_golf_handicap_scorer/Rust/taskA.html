4fn number(text: &str) -> i32 {
5 text.parse().unwrap()
6}
7
8fn half_up(numerator: i64, denominator: i64) -> i32 {
9 (2 * numerator + denominator).div_euclid(2 * denominator) as i32
10}
11
12fn half_away(numerator: i64, denominator: i64) -> i32 {
13 if numerator < 0 {
14 -half_up(-numerator, denominator)
15 } else {
16 half_up(numerator, denominator)
17 }
18}
19
20fn decimal(value: i32, negative_sign: &str) -> String {
21 let sign = if value < 0 { negative_sign } else { "" };
22 format!("{sign}{}.{:01}", value.abs() / 10, value.abs() % 10)
23}
24
25fn index_display(value: Option<i32>) -> String {
26 value.map_or_else(|| "none".to_owned(), |v| decimal(v, "+"))
27}
28
29fn handicap_display(value: Option<i32>) -> String {
30 value.map_or_else(|| "none".to_owned(), |v| {
31 if v < 0 { format!("+{}", -v) } else { v.to_string() }
32 })
33}
34
35fn average(total: i64, count: usize) -> String {
36 if count == 0 {
37 "-".to_owned()
38 } else {
39 let hundredths = total * 100 / count as i64;
40 format!("{}.{:02}", hundredths / 100, hundredths % 100)
41 }
42}
43
44fn allocation(handicap: i32, indexes: &[i32; 18]) -> [i32; 18] {
45 let magnitude = handicap.abs();
46 let quotient = magnitude / 18;
47 let remainder = magnitude % 18;
48 std::array::from_fn(|i| {
49 if handicap >= 0 {
50 quotient + i32::from(indexes[i] <= remainder)
51 } else {
52 -quotient - i32::from(indexes[i] >= 19 - remainder)
53 }
54 })
55}
56
57fn countback(values: &[i32; 18]) -> [i64; 5] {
58 let sum = |start: usize| values[start..].iter().map(|&v| i64::from(v)).sum();
59 [sum(0), sum(9), sum(12), sum(15), i64::from(values[17])]
60}

62fn match_status(lead: i32) -> String {
63 if lead == 0 {
64 "AS".to_owned()
65 } else {
66 format!("{} {}", lead.abs(), if lead > 0 { "UP" } else { "DN" })
67 }
68}

135#[derive(Clone)]
136struct Eligible {
137 number: usize,
138 adjusted: i32,
139}
140
141impl Player {
142 fn low_index(&self, day: i32) -> Option<(i32, i32)> {
143 self.records.iter()
144 .filter(|(date, _)| *date >= day - 365 && *date < day)
145 .map(|&(date, index)| (index, date))
146 .min()
147 }
148
149 fn eligible(&self) -> Vec<Eligible> {
150 let start = self.rounds.len().saturating_sub(20);
151 let mut reduction = 0;
152 let mut rows = Vec::new();
153 for i in (start..self.rounds.len()).rev() {
154 let record = &self.rounds[i];
155 reduction += record.exceptional;
156 rows.push(Eligible {
157 number: i + 1,
158 adjusted: record.differential - reduction,
159 });
160 }
161 rows.reverse();
162 rows
163 }
164
165 fn counting(&self, eligible: &[Eligible]) -> (Vec<Eligible>, i32) {
166 let n = eligible.len();
167 if n < 3 {
168 return (Vec::new(), 0);
169 }
170 let limits = [5, 8, 11, 14, 16, 18, 19, 20];
171 let k = limits.iter().position(|&limit| n <= limit).unwrap() + 1;
172 let mut selected = eligible.to_vec();
173 selected.sort_by_key(|row| (row.adjusted, -(row.number as i32)));
174 selected.truncate(k);
175 let adjustment = if n == 3 { -20 } else if n == 4 || n == 6 { -10 } else { 0 };
176 (selected, adjustment)
177 }
178
179 fn update_index(&mut self, day: i32) {
180 let (selected, adjustment) = self.counting(&self.eligible());
181 if selected.is_empty() {
182 self.calculation = None;
183 return;
184 }
185 let total: i64 = selected.iter().map(|row| i64::from(row.adjusted)).sum();
186 let average = half_away(total, selected.len() as i64);
187 let mut candidate = average + adjustment;
188 let low = self.low_index(day).map(|(index, _)| index);
189 let (mut hard, mut soft, mut maximum) = (false, false, false);
190 if let Some(lowest) = low {
191 if candidate - lowest > 30 {
192 candidate = lowest + 30 + (candidate - lowest - 30) / 2;
193 soft = true;
194 }
195 if candidate - lowest > 50 {
196 candidate = lowest + 50;
197 hard = true;
198 }
199 }
200 if candidate > 540 {
201 candidate = 540;
202 maximum = true;
203 }
204 self.index = Some(candidate);
205 self.calculation = Some(Calculation {
206 lowest: selected.len(), average, adjustment, low, hard, soft, maximum,
207 });
208 }
209}

211impl Round {
212 fn new(player: &Player, course: &Course, tee: &Tee,
213 day: i32, scores: [i32; 18]) -> Self {
214 let handicap = player.index.map(|index| tee.handicap(index, course));
215 let strokes = allocation(handicap.unwrap_or(0), &course.indexes);
216 let adjusted = std::array::from_fn(|hole| {
217 let limit = if handicap.is_some() {
218 course.pars[hole] + 2 + strokes[hole]
219 } else {
220 course.pars[hole] + 5
221 };
222 match scores[hole] {
223 -1 => course.pars[hole] + strokes[hole],
224 0 => limit,
225 value => value.min(limit),
226 }
227 });
228 let ags: i32 = adjusted.iter().sum();
229 let differential = half_away(
230 113 * i64::from(10 * ags - tee.rating), i64::from(tee.slope));
231 let exceptional = match player.index {
232 Some(index) if index - differential >= 100 => 20,
233 Some(index) if index - differential >= 70 => 10,
234 _ => 0,
235 };
236 Self {
237 course_id: course.id.clone(), tee_id: tee.id.clone(),
238 pars: course.pars, indexes: course.indexes,
239 day, scores, previous_index: player.index,
240 previous_records: player.records.len(),
241 previous_calculation: player.calculation.clone(),
242 handicap, adjusted, ags, differential, exceptional,
243 }
244 }
245
246 fn nets(&self, strokes: &[i32; 18]) -> [Option<i32>; 18] {
247 std::array::from_fn(|i| (self.scores[i] > 0).then_some(self.scores[i] - strokes[i]))
248 }
249
250 fn points(&self, nets: &[Option<i32>; 18]) -> [i32; 18] {
251 std::array::from_fn(|i| {
252 nets[i].map_or(0, |net| (self.pars[i] + 2 - net).max(0))
253 })
254 }
255
256 fn complete(&self) -> bool {
257 self.scores.iter().all(|&score| score > 0)
258 }
259}

315 fn register_course(&mut self, a: &[&str]) -> Result<(), String> {
316 if self.courses.contains_key(a[0]) {
317 return Err("duplicate course".into());
318 }
319 let pars = std::array::from_fn(|i| number(a[i + 1]));
320 let indexes = std::array::from_fn(|i| number(a[i + 19]));
321 let mut sorted = indexes;
322 sorted.sort();
323 if !pars.iter().all(|par| (3..=5).contains(par)) ||
324 !sorted.iter().enumerate().all(|(i, &si)| si == i as i32 + 1) {
325 return Err("invalid course".into());
326 }
327 let course = Course {
328 id: a[0].into(), pars, indexes, par: pars.iter().sum(),
329 tees: BTreeMap::new(),
330 };
331 self.courses.insert(a[0].into(), course);
332 Ok(())
333 }

335 fn register_tee(&mut self, a: &[&str]) -> Result<(), String> {
336 let mut course = self.course(a[0])?;
337 if course.tees.contains_key(a[1]) {
338 return Err("duplicate tee".into());
339 }
340 let rating = number(a[2]);
341 let slope = number(a[3]);
342 if !(500..=900).contains(&rating) || !(55..=155).contains(&slope) {
343 return Err("invalid tee".into());
344 }
345 course.tees.insert(a[1].into(), Tee {
346 id: a[1].into(), rating, slope,
347 });
348 self.courses.insert(a[0].into(), course);
349 Ok(())
350 }

367 fn add_round(&mut self, a: &[&str]) -> Result<(), String> {
368 let player = self.player(a[0])?;
369 let (course, tee) = self.tee(a[1], a[2])?;
370 let day = number(a[3]);
371 if player.rounds.last().is_some_and(|r| day < r.day) {
372 return Err("invalid day".into());
373 }
374 let scores = std::array::from_fn(|i| match a[i + 4] {
375 "-" => -1,
376 "X" => 0,
377 value => number(value),
378 });
379 if scores.iter().filter(|&&score| score == -1).count() >= 5 {
380 return Err("too few holes".into());
381 }
382 let record = Round::new(&player, &course, &tee, day, scores);
383 let p = self.players.get_mut(a[0]).unwrap();
384 p.rounds.push(record.clone());
385 p.update_index(day);
386 if let Some(index) = p.index {
387 p.records.push((day, index));
388 }
389 let mut parts = vec![
390 "round".to_owned(), p.id.clone(), format!("{}:", p.rounds.len()),
391 "ags".into(), record.ags.to_string(),
392 "diff".into(), decimal(record.differential, "-"),
393 ];
394 if record.exceptional != 0 {
395 parts.push("exceptional".into());
396 parts.push(decimal(-record.exceptional, "-"));
397 }
398 parts.push("index".into());
399 parts.push(index_display(p.index));
400 self.emit(parts.join(" "));
401 Ok(())
402 }

497 fn hcptable(&mut self, a: &[&str]) -> Result<(), String> {
498 let (c, t) = self.tee(a[0], a[1])?;
499 let first = number(a[2]);
500 let last = number(a[3]);
501 if first < -100 || last > 540 || first > last {

504 self.emit(format!(
505 "hcptable {} {}: {} to {}", c.id, t.id,
506 index_display(Some(first)), index_display(Some(last))));
507 let mut start = first;
508 let mut handicap = t.handicap(first, &c);
509 for index in first + 1..=last {
510 let next = t.handicap(index, &c);
511 if next != handicap {
512 self.emit(format!(
513 "{} to {}: {}", index_display(Some(start)),
514 index_display(Some(index - 1)), handicap_display(Some(handicap))));
515 start = index;
516 handicap = next;
517 }
518 }
519 self.emit(format!(
520 "{} to {}: {}", index_display(Some(start)),
521 index_display(Some(last)), handicap_display(Some(handicap))));
522 Ok(())

525 fn card_total(&mut self, label: &str, record: &Round,
526 strokes: &[i32; 18], nets: &[Option<i32>; 18],
527 points: &[i32; 18], first: usize, last: usize) {
528 let complete = record.scores[first..last].iter().all(|&score| score > 0);
529 let gross: i32 = record.scores[first..last].iter().filter(|&&v| v > 0).sum();
530 let net: i32 = nets[first..last].iter().flatten().copied().sum();
531 let pars: i32 = record.pars[first..last].iter().sum();
532 let ags: i32 = record.adjusted[first..last].iter().sum();
533 let allocated: i32 = strokes[first..last].iter().sum();
534 let pts: i32 = points[first..last].iter().sum();
535 self.emit(format!(
536 "{label} {pars} {} {ags} {allocated} {} {pts}",
537 if complete { gross.to_string() } else { "-".into() },
538 if complete { net.to_string() } else { "-".into() },
539 ));
540 }

542 fn card(&mut self, a: &[&str]) -> Result<(), String> {
543 let p = self.player(a[0])?;
544 let number = number(a[1]) as usize;
545 let record = p.rounds.get(number - 1).ok_or("unknown round")?;
546 let strokes = allocation(record.handicap.unwrap_or(0), &record.indexes);
547 let nets = record.nets(&strokes);
548 let points = record.points(&nets);
549 self.emit(format!(
550 "card {} {}: {} {} day {} handicap {}", p.id, number,
551 record.course_id, record.tee_id, record.day,
552 handicap_display(record.handicap)));
553 for h in 0..18 {
554 let score = match record.scores[h] {
555 -1 => "-".into(),
556 0 => "X".into(),
557 value => value.to_string(),
558 };
559 self.emit(format!(
560 "{} {} {} {} {} {} {} {}", h + 1, record.pars[h],
561 record.indexes[h], score, record.adjusted[h], strokes[h],
562 nets[h].map_or_else(|| "-".into(), |v| v.to_string()), points[h]));
563 if h == 8 {
564 self.card_total("OUT", record, &strokes, &nets, &points, 0, 9);
565 }
566 if h == 17 {
567 self.card_total("IN", record, &strokes, &nets, &points, 9, 18);
568 self.card_total("TOTAL", record, &strokes, &nets, &points, 0, 18);
569 }
570 }
571 self.emit(format!(
572 "ags {} diff {}", record.ags, decimal(record.differential, "-")));
573 Ok(())
574 }

576 fn stats(&mut self, a: &[&str]) -> Result<(), String> {
577 let p = self.player(a[0])?;
578 let labels = [
579 "eagles", "birdies", "pars", "bogeys", "doubles", "others", "pickups"
580 ];
581 let mut categories = [0usize; 7];
582 let mut par_total = [0i64; 6];
583 let mut par_count = [0usize; 6];
584 let mut complete_totals = Vec::new();
585 for record in &p.rounds {
586 if record.complete() {
587 complete_totals.push(record.scores.iter().sum::<i32>());
588 }
589 for (&score, &par) in record.scores.iter().zip(record.pars.iter()) {
590 if score == 0 {
591 categories[6] += 1;
592 } else if score > 0 {
593 let category = (score - par + 2).clamp(0, 5) as usize;
594 categories[category] += 1;
595 par_total[par as usize] += i64::from(score);
596 par_count[par as usize] += 1;
597 }
598 }
599 }
600 let best = complete_totals.iter().min().map_or_else(
601 || "-".to_owned(), i32::to_string);
602 let mut parts = vec![
603 format!("stats {}:", p.id), "rounds".into(), p.rounds.len().to_string(),
604 "best".into(), best, "average".into(),
605 average(complete_totals.iter().map(|&v| i64::from(v)).sum(),
606 complete_totals.len()),
607 ];
608 for (label, count) in labels.iter().zip(categories) {
609 parts.push((*label).into());
610 parts.push(count.to_string());
611 }
612 for par in 3..=5 {
613 parts.push(format!("par{par}"));
614 parts.push(average(par_total[par], par_count[par]));
615 }
616 self.emit(parts.join(" "));
617 Ok(())
618 }

620 fn holestats(&mut self, a: &[&str]) -> Result<(), String> {
621 let c = self.course(a[0])?;
622 let mut totals = [0i64; 18];
623 let mut counts = [0i64; 18];
624 let mut considered = 0usize;
625 for p in self.players.values() {
626 for r in &p.rounds {
627 if r.course_id != c.id {
628 continue;
629 }
630 considered += 1;
631 for h in 0..18 {
632 if r.scores[h] > 0 {
633 counts[h] += 1;
634 totals[h] += i64::from(r.scores[h]);
635 }
636 }
637 }
638 }
639 let mut holes: Vec<usize> = (0..18).collect();
640 holes.sort_by(|&a, &b| match (counts[a], counts[b]) {
641 (0, 0) => a.cmp(&b),
642 (0, _) => std::cmp::Ordering::Greater,
643 (_, 0) => std::cmp::Ordering::Less,
644 _ => {
645 let left = (totals[a] - i64::from(c.pars[a]) * counts[a]) * counts[b];
646 let right = (totals[b] - i64::from(c.pars[b]) * counts[b]) * counts[a];
647 right.cmp(&left).then(a.cmp(&b))
648 }
649 });
650 let mut ranks = [0usize; 18];
651 for (rank, hole) in holes.into_iter().enumerate() {
652 ranks[hole] = rank + 1;
653 }
654 self.emit(format!("holestats {}: rounds {}", c.id, considered));
655 for h in 0..18 {
656 self.emit(format!(
657 "{} par {} si {} scores {} avg {} rank {}",
658 h + 1, c.pars[h], c.indexes[h], counts[h],
659 average(totals[h], counts[h] as usize), ranks[h]));
660 }
661 Ok(())
662 }

664 fn show_players(&mut self) {
665 let mut players: Vec<Player> = self.players.values().cloned().collect();
666 players.sort_by(|a, b| {
667 (a.index.is_none(), a.index.unwrap_or(0), &a.id)
668 .cmp(&(b.index.is_none(), b.index.unwrap_or(0), &b.id))
669 });
670 self.emit(format!("players {}", players.len()));
671 for p in players {
672 self.emit(format!(
673 "{} {} {} rounds {}", p.id, p.name,
674 index_display(p.index), p.rounds.len()));
675 }
676 }

678 fn register_event(&mut self, a: &[&str]) -> Result<(), String> {
679 if self.events.contains_key(a[0]) {
680 return Err("duplicate event".into());
681 }
682 let _ = self.tee(a[2], a[3])?;
683 let allowance = number(a[5]);
684 if !(1..=100).contains(&allowance) {
685 return Err("invalid allowance".into());
686 }
687 for id in &a[6..] {
688 let _ = self.player(id)?;
689 }
690 let players: Vec<String> = a[6..].iter().map(|s| (*s).to_owned()).collect();
691 let count = players.len();
692 let valid = match a[1] {
693 "MATCH" | "NASSAU" => count == 2,
694 "FOURBALL" => count == 4,
695 "BETTERBALL" => [4, 6, 8].contains(&count),
696 _ => (2..=8).contains(&count),
697 };
698 if !valid || players.iter().collect::<BTreeSet<_>>().len() != count {
699 return Err("invalid players".into());
700 }
701 self.events.insert(a[0].into(), Event {
702 id: a[0].into(), format: a[1].into(),
703 course_id: a[2].into(), tee_id: a[3].into(),
704 day: number(a[4]), allowance, players,
705 });
706 Ok(())
707 }

709 fn event_data(&self, event: &Event) -> Result<EventData, String> {
710 let course = self.course(&event.course_id)?;
711 let mut rounds = Vec::new();
712 let mut handicaps = Vec::new();
713 for id in &event.players {
714 let p = self.player(id)?;
715 let record = p.rounds.iter().find(|r| {
716 r.course_id == event.course_id && r.tee_id == event.tee_id &&
717 r.day == event.day
718 }).ok_or("missing round")?;
719 let handicap = record.handicap.ok_or("no index")?;
720 rounds.push(record.clone());
721 handicaps.push(half_up(
722 i64::from(handicap) * i64::from(event.allowance), 100));
723 }
724 let minimum = if ["MATCH", "NASSAU", "FOURBALL"].contains(&event.format.as_str()) {
725 *handicaps.iter().min().unwrap()
726 } else {
727 0
728 };
729 let mut nets = Vec::new();
730 let mut points = Vec::new();
731 for (record, &handicap) in rounds.iter().zip(handicaps.iter()) {
732 let strokes = allocation(handicap - minimum, &course.indexes);
733 let net = record.nets(&strokes);
734 points.push(record.points(&net));
735 nets.push(net);
736 }
737 Ok(EventData { rounds, handicaps, nets, points })
738 }

740 fn event_ranking(&self, event: &Event, data: &EventData)
741 -> (Vec<usize>, Vec<[i32; 18]>) {
742 let mut order = Vec::new();
743 let mut values = Vec::new();
744 for i in 0..event.players.len() {
745 let record = &data.rounds[i];
746 let row = std::array::from_fn(|h| match event.format.as_str() {
747 "STABLEFORD" => data.points[i][h],
748 "STROKE" => data.nets[i][h].unwrap_or(0),
749 _ => match data.nets[i][h] {
750 None => -1,
751 Some(net) if net > record.pars[h] => -1,
752 Some(net) if net < record.pars[h] => 1,
753 _ => 0,
754 },
755 });
756 values.push(row);
757 if event.format != "STROKE" || record.complete() {
758 order.push(i);
759 }
760 }
761 let larger = event.format != "STROKE";
762 order.sort_by(|&a, &b| {
763 let left = countback(&values[a]);
764 let right = countback(&values[b]);
765 let relation = if larger { right.cmp(&left) } else { left.cmp(&right) };
766 relation.then(event.players[a].cmp(&event.players[b]))
767 });
768 (order, values)
769 }

771 fn individual_results(&mut self, event: &Event, data: &EventData) {
772 let (order, values) = self.event_ranking(event, data);
773 for (rank, &i) in order.iter().enumerate() {
774 let handicap = handicap_display(Some(data.handicaps[i]));
775 let details = match event.format.as_str() {
776 "STROKE" => format!(
777 "net {} gross {}", values[i].iter().sum::<i32>(),
778 data.rounds[i].scores.iter().sum::<i32>()),
779 "STABLEFORD" => format!("points {}", data.points[i].iter().sum::<i32>()),
780 _ => match_status(values[i].iter().sum::<i32>()),
781 };
782 self.emit(format!(
783 "{} {} {} hcp {}", rank + 1, event.players[i], details, handicap));
784 }
785 if event.format == "STROKE" {
786 let mut no_returns: Vec<usize> = (0..event.players.len())
787 .filter(|&i| !data.rounds[i].complete()).collect();
788 no_returns.sort_by_key(|&i| &event.players[i]);
789 for i in no_returns {
790 self.emit(format!(
791 "NR {} hcp {}", event.players[i],
792 handicap_display(Some(data.handicaps[i]))));
793 }
794 if !order.is_empty() {
795 let winner = *order.iter().min_by(|&&a, &&b| {
796 countback(&data.rounds[a].scores)
797 .cmp(&countback(&data.rounds[b].scores))
798 .then(event.players[a].cmp(&event.players[b]))
799 }).unwrap();
800 self.emit(format!(
801 "gross {} {}", event.players[winner],
802 data.rounds[winner].scores.iter().sum::<i32>()));
803 }
804 }
805 }

807 fn match_holes(&self, event: &Event, data: &EventData)
808 -> ([String; 2], [i32; 18]) {
809 let (sides, names) = if event.format == "FOURBALL" {
810 (
811 [vec![0, 1], vec![2, 3]],
812 [
813 format!("{}+{}", event.players[0], event.players[1]),
814 format!("{}+{}", event.players[2], event.players[3]),
815 ],
816 )
817 } else {
818 (
819 [vec![0], vec![1]],
820 [event.players[0].clone(), event.players[1].clone()],
821 )
822 };
823 let holes = std::array::from_fn(|h| {
824 let best = |side: &Vec<usize>| {
825 side.iter().filter_map(|&i| data.nets[i][h]).min()
826 };
827 match (best(&sides[0]), best(&sides[1])) {
828 (None, None) => 0,
829 (None, Some(_)) => -1,
830 (Some(_), None) => 1,
831 (Some(a), Some(b)) => i32::from(a < b) - i32::from(a > b),
832 }
833 });
834 (names, holes)
835 }
836
837 fn play_match(names: &[String; 2], holes: &[i32; 18],
838 first: usize, last: usize) -> (Vec<String>, String) {
839 let mut lead: i32 = 0;
840 let mut remaining = 0usize;
841 let mut statuses = Vec::new();
842 for hole in first..last {
843 lead += holes[hole];
844 statuses.push(match_status(lead));
845 remaining = last - hole - 1;
846 if lead.abs() as usize > remaining || remaining == 0 {
847 break;
848 }
849 }
850 let result = if lead == 0 {
851 "AS".to_owned()
852 } else {
853 let winner = &names[usize::from(lead < 0)];
854 let margin = if remaining == 0 {
855 format!("{} UP", lead.abs())
856 } else {
857 format!("{}&{remaining}", lead.abs())
858 };
859 format!("{winner} {margin}")
860 };
861 (statuses, result)
862 }
863
864 fn match_results(&mut self, event: &Event, data: &EventData) {
865 let minimum = *data.handicaps.iter().min().unwrap();
866 let mut parts = vec!["strokes".to_owned()];
867 for (id, &handicap) in event.players.iter().zip(data.handicaps.iter()) {
868 parts.push(id.clone());
869 parts.push((handicap - minimum).to_string());
870 }
871 self.emit(parts.join(" "));
872 let (names, holes) = self.match_holes(event, data);
873 if event.format == "NASSAU" {
874 for (label, first, last) in [("front", 0, 9), ("back", 9, 18), ("total", 0, 18)] {
875 let (_, result) = Self::play_match(&names, &holes, first, last);
876 self.emit(format!("{label} {result}"));
877 }
878 } else {
879 let (statuses, result) = Self::play_match(&names, &holes, 0, 18);
880 self.emit(format!("status {}", statuses.join(", ")));
881 self.emit(format!("result {result}"));
882 }
883 }

885 fn betterball_results(&mut self, event: &Event, data: &EventData) {
886 let mut values = Vec::new();
887 for pair in (0..event.players.len()).step_by(2) {
888 values.push(std::array::from_fn(|h| data.points[pair][h]
889 .max(data.points[pair + 1][h])));
890 }
891 let mut order: Vec<usize> = (0..values.len()).collect();
892 order.sort_by(|&a, &b| {
893 countback(&values[b]).cmp(&countback(&values[a]))
894 .then(event.players[2 * a].cmp(&event.players[2 * b]))
895 });
896 for (rank, team) in order.into_iter().enumerate() {
897 self.emit(format!(
898 "{} {}+{} points {} hcp {} {}", rank + 1,
899 event.players[2 * team], event.players[2 * team + 1],
900 values[team].iter().sum::<i32>(),
901 handicap_display(Some(data.handicaps[2 * team])),
902 handicap_display(Some(data.handicaps[2 * team + 1])),
903 ));
904 }
905 }
906
907 fn skins_results(&mut self, event: &Event, data: &EventData) {
908 let mut totals = vec![0i32; event.players.len()];
909 let mut carry = 0i32;
910 let mut last_minimum: Vec<usize> = Vec::new();
911 for hole in 0..18 {
912 carry += 1;
913 let best = (0..event.players.len())
914 .filter_map(|i| data.nets[i][hole]).min();
915 let winners: Vec<usize> = (0..event.players.len())
916 .filter(|&i| best.is_some() && data.nets[i][hole] == best).collect();
917 last_minimum = winners.clone();
918 if winners.len() == 1 {
919 self.emit(format!(
920 "hole {} {} {}", hole + 1, event.players[winners[0]], carry));
921 totals[winners[0]] += 100 * carry;
922 carry = 0;
923 }
924 }
925 if carry != 0 {
926 let mut sharing = if last_minimum.is_empty() {
927 (0..event.players.len()).collect::<Vec<_>>()
928 } else {
929 last_minimum
930 };
931 sharing.sort_by_key(|&i| &event.players[i]);
932 let portion = 100 * carry / sharing.len() as i32;
933 for &i in &sharing {
934 totals[i] += portion;
935 }
936 let mut parts = vec!["carry".into(), carry.to_string()];
937 parts.extend(sharing.iter().map(|&i| event.players[i].clone()));
938 self.emit(parts.join(" "));
939 }
940 let mut order: Vec<usize> = (0..event.players.len()).collect();
941 order.sort_by(|&a, &b| {
942 totals[b].cmp(&totals[a]).then(event.players[a].cmp(&event.players[b]))
943 });
944 for i in order {
945 self.emit(format!(
946 "{} {}.{:02}", event.players[i], totals[i] / 100, totals[i] % 100));
947 }
948 }

950 fn results(&mut self, a: &[&str]) -> Result<(), String> {
951 let event = self.events.get(a[0]).cloned().ok_or("unknown event")?;
952 let data = self.event_data(&event)?;
953 self.emit(format!(
954 "results {}: {} {} {} day {} allowance {}", event.id, event.format,
955 event.course_id, event.tee_id, event.day, event.allowance));
956 match event.format.as_str() {
957 "STROKE" | "STABLEFORD" | "PAR" => self.individual_results(&event, &data),
958 "MATCH" | "NASSAU" | "FOURBALL" => self.match_results(&event, &data),
959 "BETTERBALL" => self.betterball_results(&event, &data),
960 _ => self.skins_results(&event, &data),
961 }
962 Ok(())
963 }

965 fn merit(&mut self) {
966 let awards = [10, 8, 6, 5, 4, 3, 2, 1];
967 let mut tally: BTreeMap<String, [i32; 3]> = BTreeMap::new();
968 let mut counted = 0;
969 for event in self.events.values() {
970 if !["STROKE", "STABLEFORD", "PAR"].contains(&event.format.as_str()) {
971 continue;
972 }
973 let Ok(data) = self.event_data(event) else { continue };
974 counted += 1;
975 let (order, _) = self.event_ranking(event, &data);
976 for id in &event.players {
977 tally.entry(id.clone()).or_default()[2] += 1;
978 }
979 for (rank, i) in order.into_iter().enumerate() {
980 let row = tally.entry(event.players[i].clone()).or_default();
981 row[0] += awards[rank];
982 if rank == 0 {
983 row[1] += 1;
984 }
985 }
986 }
987 self.emit(format!("merit: events {counted}"));
988 let mut rows: Vec<_> = tally.into_iter().collect();
989 rows.sort_by(|(id_a, a), (id_b, b)| {
990 b[0].cmp(&a[0]).then(b[1].cmp(&a[1])).then(id_a.cmp(id_b))
991 });
992 for (rank, (id, row)) in rows.into_iter().enumerate() {
993 self.emit(format!(
994 "{} {} {} wins {} played {}", rank + 1, id, row[0], row[1], row[2]));
995 }
996 }

998 fn dispatch(&mut self, command: &str, a: &[&str]) -> Result<(), String> {
999 match command {
1000 "COURSE" => self.register_course(a),
1001 "TEE" => self.register_tee(a),
1002 "PLAYER" => self.register_player(a),
1003 "ROUND" => self.add_round(a),
1004 "REMOVE" => self.remove(a),
1005 "INDEX" => self.show_index(a),
1006 "HISTORY" => self.history(a),
1007 "LOWINDEX" => self.lowindex(a),
1008 "COURSEHCP" => self.coursehcp(a),
1009 "HCPTABLE" => self.hcptable(a),
1010 "CARD" => self.card(a),
1011 "STATS" => self.stats(a),
1012 "HOLESTATS" => self.holestats(a),
1013 "PLAYERS" => { self.show_players(); Ok(()) }
1014 "EVENT" => self.register_event(a),
1015 "RESULTS" => self.results(a),
1016 "MERIT" => { self.merit(); Ok(()) }
1017 _ => unreachable!(),
1018 }
1019 }

1021 fn execute(&mut self, line: &str) {
1022 let mut words = line.split_whitespace();
1023 let command = words.next().unwrap();
1024 let arguments: Vec<_> = words.collect();
1025 if let Err(reason) = self.dispatch(command, &arguments) {
1026 self.emit(format!("ERROR {reason}"));
1027 }
1028 }
1029}
