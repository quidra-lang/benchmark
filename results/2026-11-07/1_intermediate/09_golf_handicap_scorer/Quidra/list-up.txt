int integer_value(string text)
    return int(text)
int floor_ratio(int numerator, int denominator)
    return if numerator >= 0 then numerator / denominator else -((-numerator + denominator - 1) / denominator)
int half_up(int numerator, int denominator)
    return floor_ratio(2 * numerator + denominator, 2 * denominator)
int half_away(int numerator, int denominator)
    result := (2 * math.abs(numerator) + denominator) / (2 * denominator)
    return if numerator < 0 then -result else result
string decimal_text(int value, string negative)
    absolute := math.abs(value)
    return (if value < 0 then negative else "") + string(absolute / 10) + "." + string(absolute % 10)
string index_text(int value)
    return if value == 1000000000000 then "none" else decimal_text(value, "+")
bool alphabetic_before(string a, string b)
    int[] aa := a.codepoints()
    int[] bb := b.codepoints()
    for: 0 <= k < math.min(int(#aa), int(#bb))
        if aa[k] != bb[k]
            return aa[k] < bb[k]
    return #aa < #bb
class Rank(int index, int[] keys, string id)
bool rank_before(const Rank a, const Rank b)
    for: 0 <= k < int(#a.keys)
        if a.keys[k] != b.keys[k]
            return a.keys[k] < b.keys[k]
    return alphabetic_before(a.id, b.id)
int[] ranked(Rank[] entries)
    for: 1 <= k < int(#entries)
        item := entries[k]
        position := k
        while position > 0 and rank_before(item, entries[position - 1])
            entries[position] = entries[position - 1]
            position -= 1
        entries[position] = item
    return [for item := entries[] then item.index]
class Course(string id, int[] pars, int[] si, int par)
class Tee(int course, string id, int rating, int slope)
class Calculation(int k, int average, int adjustment, int low, bool hard, bool soft, bool capped)
class Player(string id, string name, int index, int[] rounds, int[] record_days, int[] record_indexes, Calculation last)
class Round(int course, int tee, int day, int[] scores, int handicap, int[] adjusted, int ags, int differential, int exceptional, int prior_index, int prior_records, Calculation prior_last)
class State()
    Course[] courses := []
    Tee[] tees := []
    Player[] players := []
    rounds := array(1000, fill = Round(0, 0, 0, [], 1000000000000, [], 0, 0, 0, 1000000000000, 0, Calculation(0, 0, 0, 1000000000000, false, false, false)))
    round_used := 0
void emit(string text)
    println(text)
void fail(string reason)
    emit("ERROR " + reason)
int course_id(const State state, string text)
    for: 0 <= k < int(#state.courses)
        if state.courses[k].id == text
            return k
    return -1
int tee_id(const State state, int course, string text)
    for: 0 <= k < int(#state.tees)
        if state.tees[k].course == course and state.tees[k].id == text
            return k
    return -1
int player_id(const State state, string text)
    for: 0 <= k < int(#state.players)
        if state.players[k].id == text
            return k
    return -1
int course_handicap(const State state, int course, int tee, int index)
    return half_up(index * state.tees[tee].slope + 113 * (state.tees[tee].rating - 10 * state.courses[course].par), 1130)
int[] allocation(const Course course, int handicap)
    absolute := math.abs(handicap)
    quotient := absolute / 18
    remainder := absolute % 18
    return [for k := (0:<18)[] then
        if handicap >= 0 then quotient + (if course.si[k] <= remainder then 1 else 0)
        else -quotient - (if course.si[k] >= 19 - remainder then 1 else 0)]
int[] low_record(const Player player, int day)
    value := 1000000000000
    date := 0
    for: 0 <= k < int(#player.record_days)
        candidate_day := player.record_days[k]
        candidate := player.record_indexes[k]
        if day - 365 <= candidate_day < day and (candidate < value or (candidate == value and candidate_day < date))
            value = candidate
            date = candidate_day
    return [value, date]
int[] count_rule(int count)
    if count == 3
        return [1, -20]
    if count == 4
        return [1, -10]
    if count == 5
        return [1, 0]
    if count == 6
        return [2, -10]
    if count <= 8
        return [2, 0]
    if count <= 11
        return [3, 0]
    if count <= 14
        return [4, 0]
    if count <= 16
        return [5, 0]
    if count <= 18
        return [6, 0]
    return [if count == 19 then 7 else 8, 0]
int[] adjusted_differentials(const State state, int person)
    ids := state.players[person].rounds
    first := math.max(0, int(#ids) - 20)
    result := array(nat(int(#ids) - first), fill = 0)
    reduction := 0
    position := int(#ids) - 1
    while position >= first
        item := state.rounds[ids[position]]
        reduction += item.exceptional
        result[position - first] = item.differential - reduction
        position -= 1
    return result
int[] counting_order(int[] adjusted)
    return ranked([for k := (0:<int(#adjusted))[] then Rank(k, [adjusted[k], -k], "")])
void update_index(State &state, int person, int day)
    adjusted := adjusted_differentials(state, person)
    count := int(#adjusted)
    if count < 3
        state.players[person].last = Calculation(0, 0, 0, 1000000000000, false, false, false)
        return
    rules := count_rule(count)
    order := counting_order(adjusted)
    int total := adjusted[order[:<rules[0]]].sum()
    average := half_away(total, rules[0])
    candidate := average + rules[1]
    low := low_record(state.players[person], day)
    soft := false
    hard := false
    if low[0] != 1000000000000
        if candidate - low[0] > 30
            candidate = low[0] + 30 + (candidate - low[0] - 30) / 2
            soft = true
        if candidate - low[0] > 50
            candidate = low[0] + 50
            hard = true
    capped := candidate > 540
    candidate = math.min(candidate, 540)
    state.players[person].index = candidate
    state.players[person].last = Calculation(rules[0], average, rules[1], low[0], hard, soft, capped)
void command(State &state, string line)
    words := line.split(" ")
    action := words[0]
    if action == "COURSE"
        if course_id(state, words[1]) >= 0
            fail("duplicate course")
            return
        int[] pars := []
        int[] si := []
        valid := true
        par := 0
        for: 0 <= k < 18
            value := integer_value(words[k + 2])
            if value < 3 or value > 5
                valid = false
            pars = pars.append(value)
            par += value
            si = si.append(integer_value(words[k + 20]))
        for: 1 <= target <= 18
            found := 0
            for value := si[]
                if value == target
                    found += 1
            if found != 1
                valid = false
        if not valid
            fail("invalid course")
        else
            state.courses = state.courses.append(Course(words[1], pars, si, par))
        return
    if action == "TEE" or action == "HCPTABLE" or action == "HOLESTATS"
        course := course_id(state, words[1])
        if course < 0
            fail("unknown course")
            return
        tee := tee_id(state, course, words[2])
        if action == "TEE"
            if tee >= 0
                fail("duplicate tee")
                return
            rating := integer_value(words[3])
            slope := integer_value(words[4])
            if rating < 500 or rating > 900 or slope < 55 or slope > 155
                fail("invalid tee")
            else
                state.tees = state.tees.append(Tee(course, words[2], rating, slope))
        return
    if action == "PLAYER"
        if player_id(state, words[1]) >= 0
            fail("duplicate player")
            return
        index := if #words == 4 then integer_value(words[3]) else 1000000000000
        if index != 1000000000000 and (index < -100 or index > 540)
            fail("invalid index")
        else
            state.players = state.players.append(Player(words[1], words[2], index, [], [], [], Calculation(0, 0, 0, 1000000000000, false, false, false)))
        return
    person := player_id(state, words[1])
    if person < 0
        fail("unknown player")
        return
    id := state.players[person].id
    count := int(#state.players[person].rounds)
    if action == "ROUND" or action == "COURSEHCP"
        course := course_id(state, words[2])
        tee := tee_id(state, course, words[3])
        if course < 0
            fail("unknown course")
            return
        if tee < 0
            fail("unknown tee")
            return
        index := state.players[person].index
        day := integer_value(words[4])
        if count > 0 and day < state.rounds[state.players[person].rounds[count - 1]].day
            fail("invalid day")
            return
        int[] scores := []
        missing := 0
        for: 0 <= hole < 18
            text := words[hole + 5]
            score := if text == "X" then 0 elif text == "-" then -1 else integer_value(text)
            if score == -1
                missing += 1
            scores = scores.append(score)
        if missing >= 5
            fail("too few holes")
            return
        handicap := if index == 1000000000000 then 1000000000000 else course_handicap(state, course, tee, index)
        strokes := allocation(state.courses[course], if handicap == 1000000000000 then 0 else handicap)
        adjusted := array(18, fill = 0)
        for: 0 <= hole < 18
            par := state.courses[course].pars[hole]
            limit := if index == 1000000000000 then par + 5 else par + 2 + strokes[hole]
            adjusted[hole] = if scores[hole] > 0 then math.min(scores[hole], limit) elif scores[hole] == 0 then limit else par + strokes[hole]
        ags := adjusted.sum()
        differential := half_away(113 * (10 * ags - state.tees[tee].rating), state.tees[tee].slope)
        exceptional := if index == 1000000000000 or index - differential < 70 then 0 elif index - differential < 100 then 10 else 20
        round_index := state.round_used
        state.round_used += 1
        state.rounds[round_index] = Round(course, tee, day, scores, handicap, adjusted, ags, differential, exceptional, index, int(#state.players[person].record_days), state.players[person].last)
        state.players[person].rounds = state.players[person].rounds.append(round_index)
        update_index(&state, person, day)
        updated := state.players[person].index
        if updated != 1000000000000
            state.players[person].record_days = state.players[person].record_days.append(day)
            state.players[person].record_indexes = state.players[person].record_indexes.append(updated)
        emit("round " + id + " " + string(count + 1) + ": ags " + string(ags) + " diff " + decimal_text(differential, "-") + (if exceptional > 0 then " exceptional " + decimal_text(-exceptional, "-") else "") + " index " + index_text(updated))
    elif action == "REMOVE"
        if count == 0
            fail("no rounds")
        else
            item := state.rounds[state.players[person].rounds[count - 1]]
            state.players[person].rounds = state.players[person].rounds[:<count - 1]
            state.players[person].index = item.prior_index
            state.players[person].record_days = state.players[person].record_days[:<item.prior_records]
            state.players[person].record_indexes = state.players[person].record_indexes[:<item.prior_records]
            state.players[person].last = item.prior_last
            emit("removed " + id + " " + string(count) + ": index " + index_text(item.prior_index))
    elif action == "INDEX"
        item := state.players[person]
        text := "index " + id + ": " + index_text(item.index) + " rounds " + string(count)
        calculation := item.last
        if calculation.k > 0
            text += " lowest " + string(calculation.k) + " average " + decimal_text(calculation.average, "-") + " adjustment " + decimal_text(calculation.adjustment, "-")
            if calculation.low != 1000000000000
                text += " low " + index_text(calculation.low)
            if calculation.hard
                text += " hard cap"
            elif calculation.soft
                text += " soft cap"
            if calculation.capped
                text += " maximum"
        emit(text)
    elif action == "LOWINDEX"
        day := integer_value(words[2])
        low := low_record(state.players[person], day)
        emit("lowindex " + id + " " + string(day) + ": " + index_text(low[0]) + (if low[0] == 1000000000000 then "" else " day " + string(low[1])))
    elif action == "HISTORY"
        emit("history " + id + ": rounds " + string(count))
        adjusted := adjusted_differentials(state, person)
        order := counting_order(adjusted)
        used := if #adjusted < 3 then 0 else count_rule(int(#adjusted))[0]
        first := math.max(0, count - 20)
        for: 0 <= k < int(#adjusted)
            item := state.rounds[state.players[person].rounds[first + k]]
            counting := false
            for: 0 <= j < used
                if order[j] == k
                    counting = true
            emit(string(first + k + 1) + " day " + string(item.day) + " " + state.courses[item.course].id + " " + state.tees[item.tee].id + " ags " + string(item.ags) + " diff " + decimal_text(item.differential, "-") + (if adjusted[k] != item.differential then " adjusted " + decimal_text(adjusted[k], "-") else "") + (if counting then " *" else ""))
command_count := 0
scan(&command_count)
golf := State()
for: 0 <= line_number < command_count
    input_line := ""
    scan(&input_line)
    command(&golf, input_line)
