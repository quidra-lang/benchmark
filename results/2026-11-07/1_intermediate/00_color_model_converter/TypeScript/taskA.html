13function floorBig(n: bigint, d: bigint): bigint {
14 let quotient = n / d;
15 if (n < 0n && n % d !== 0n) quotient--;
16 return quotient;
17}

19function floorRatio(n: number, d: number): number {
20 return Number(floorBig(BigInt(n), BigInt(d)));

23function roundRatio(n: number, d: number): number {
24 const denominator = BigInt(d);
25 return Number(floorBig(2n * BigInt(n) + denominator, 2n * denominator));

28function awayRatio(n: number, d: number): number {

81function colorArgument(token: string): Color {
82 if (/^#[0-9a-fA-F]{6}$/.test(token)) return parseHex(token);
83 if (/^#[0-9a-fA-F]{3}$/.test(token)) {
84 return [...token.slice(1)].map(digit => parseInt(digit + digit, 16));
85 }
86 const match = token.match(/^rgb\((\d{1,3}),(\d{1,3}),(\d{1,3})\)$/);
87 if (match) {
88 const color = match.slice(1).map(Number);
89 requireValid(color.every(component => component <= 255));
90 return color;
91 }
92 const color = builtinMap.get(token) ?? registered.get(token);
93 requireValid(color !== undefined);
94 return [...color!];
95}

106 return a.reduce((sum, value, i) => sum + (value - b[i]) * (value - b[i]), 0);

109function hue(color: Color): number {
110 const [r, g, b] = color;
111 const maximum = Math.max(...color), difference = maximum - Math.min(...color);
112 if (difference === 0) return 0;
113 let numerator;
114 if (maximum === r) {
115 numerator = 60 * (g - b);
116 if (numerator < 0) numerator += 360 * difference;
117 } else if (maximum === g) numerator = 60 * (b - r) + 120 * difference;
118 else numerator = 60 * (r - g) + 240 * difference;
119 return roundRatio(numerator, difference) % 360;
120}

122function hsl(color: Color): number[] {
123 const maximum = Math.max(...color), minimum = Math.min(...color);
124 const difference = maximum - minimum;
125 const saturation = difference === 0 ? 0 : roundRatio(100 * difference, 255 - Math.abs(maximum + minimum - 255));
126 return [hue(color), saturation, roundRatio(100 * (maximum + minimum), 510)];
127}

129function base60(h: number): Color {
130 const sector = floorRatio(h, 60), fraction = h % 60;
131 const x = sector % 2 === 0 ? fraction : 60 - fraction;
132 const bases = [[60, x, 0], [x, 60, 0], [0, 60, x], [0, x, 60], [x, 0, 60], [60, 0, x]];
133 return bases[sector];
134}

136function fromHsl(h: number, s: number, l: number): Color {
137 const chroma = (100 - Math.abs(2 * l - 100)) * s;
138 return base60(h).map(component => roundRatio(255 * (chroma * component + 30 * (200 * l - chroma)), 600000));
139}
140
141function fromHsv(h: number, s: number, v: number): Color {
142 const chroma = v * s;
143 return base60(h).map(component => roundRatio(255 * (chroma * component + 60 * (100 * v - chroma)), 600000));
144}

146function fromHwb(h: number, w: number, k: number): Color {
147 if (w + k >= 100) return Array(3).fill(roundRatio(255 * w, w + k));
148 return base60(h).map(component => roundRatio(255 * (component * (100 - w - k) + 60 * w), 6000));
149}

151function modelValues(command: string, color: Color): number[] {
152 const [r, g, b] = color;
153 const maximum = Math.max(...color), minimum = Math.min(...color);
154 switch (command) {
155 case "HSL": return hsl(color);
156 case "HSV":
157 return [hue(color), maximum === 0 ? 0 : roundRatio(100 * (maximum - minimum), maximum), roundRatio(100 * maximum, 255)];
158 case "HWB":
159 return [hue(color), roundRatio(100 * minimum, 255), roundRatio(100 * (255 - maximum), 255)];
160 case "CMYK":
161 if (maximum === 0) return [0, 0, 0, 100];
162 return [...color.map(component => roundRatio(100 * (maximum - component), maximum)), roundRatio(100 * (255 - maximum), 255)];
163 case "YCBCR":
164 return [
165 clamp(roundRatio(299 * r + 587 * g + 114 * b, 1000)),
166 clamp(roundRatio(128000 - 169 * r - 331 * g + 500 * b, 1000)),
167 clamp(roundRatio(128000 + 500 * r - 419 * g - 81 * b, 1000))
168 ];
169 case "YIQ":
170 return [
171 roundRatio(299 * r + 587 * g + 114 * b, 1000),
172 roundRatio(596 * r - 274 * g - 322 * b, 1000),
173 roundRatio(211 * r - 523 * g + 312 * b, 1000)
174 ];
175 default: throw new Invalid();
176 }
177}

179function modelText(command: string, color: Color): string {
180 const values = modelValues(command, color);
181 let body;
182 if (["HSL", "HSV", "HWB"].includes(command)) body = values[0] + "," + values[1] + "%," + values[2] + "%";
183 else if (command === "CMYK") body = values.map(value => value + "%").join(",");
184 else body = values.join(",");
185 return command.toLowerCase() + "(" + body + ")";
186}

188function ansiColor(code: number): Color {
189 if (code <= 15) return [...builtins[code].color];
190 if (code >= 232) return Array(3).fill(8 + 10 * (code - 232));
191 const levels = [0, 95, 135, 175, 215, 255], value = code - 16;
192 return [levels[floorRatio(value, 36)], levels[floorRatio(value, 6) % 6], levels[value % 6]];
193}

195function rootScaled(value: number, scale: number): number {
196 const target = BigInt(value), factor = BigInt(scale);
197 let low = 0n, high = 1n;
198 while (factor * high * high <= target) high *= 2n;
199 while (low + 1n < high) {
200 const middle = (low + high) / 2n;
201 if (factor * middle * middle <= target) low = middle;
202 else high = middle;
203 }
204 return Number(low);
205}

222function blendChannel(mode: string, x: number, y: number): number {
223 switch (mode) {
224 case "MULTIPLY": return floorRatio(x * y, 255);
225 case "SCREEN": return 255 - floorRatio((255 - x) * (255 - y), 255);
226 case "OVERLAY":
227 case "HARDLIGHT":
228 if ((mode === "OVERLAY" ? x : y) < 128) return floorRatio(2 * x * y, 255);
229 return 255 - floorRatio(2 * (255 - x) * (255 - y), 255);
230 case "SOFTLIGHT": {
231 const multiply = floorRatio(x * y, 255);
232 const screen = 255 - floorRatio((255 - x) * (255 - y), 255);
233 return floorRatio((255 - x) * multiply + x * screen, 255);
234 }
235 case "DARKEN": return Math.min(x, y);
236 case "LIGHTEN": return Math.max(x, y);
237 case "DIFFERENCE": return Math.abs(x - y);
238 case "EXCLUSION": return x + y - floorRatio(2 * x * y, 255);
239 case "ADD": return Math.min(255, x + y);
240 case "SUBTRACT": return Math.max(0, x - y);
241 case "COLORDODGE": return y === 255 ? 255 : Math.min(255, floorRatio(255 * x, 255 - y));
242 case "COLORBURN": return y === 0 ? 0 : Math.max(0, 255 - floorRatio(255 * (255 - x), y));
243 default: throw new Invalid();
244 }
245}

247function blend(mode: string, base: Color, top: Color): Color {
248 const [h1, s1, l1] = hsl(base), [h2, s2, l2] = hsl(top);
249 switch (mode) {
250 case "HUE": return fromHsl(h2, s1, l1);
251 case "SATURATION": return fromHsl(h1, s2, l1);
252 case "COLOR": return fromHsl(h2, s2, l1);
253 case "LUMINOSITY": return fromHsl(h1, s1, l2);
254 default: return base.map((value, i) => blendChannel(mode, value, top[i]));
255 }
256}

265function contrast(a: Color, b: Color): ContrastInfo {
266 const luminance = (c: Color): number => 2126 * c[0] * c[0] + 7152 * c[1] * c[1] + 722 * c[2] * c[2];
267 const first = luminance(a), second = luminance(b);
268 const high = Math.max(first, second) + 32512500;
269 const low = Math.min(first, second) + 32512500;
270 const verdict = high >= 7 * low ? "AAA" : 2 * high >= 9 * low ? "AA" : "FAIL";
271 return {ratio: roundRatio(100 * high, low), verdict};
272}

274function contrastText(info: ContrastInfo): string {
275 return floorRatio(info.ratio, 100) + "." + String(info.ratio % 100).padStart(2, "0") + " " + info.verdict;
276}

278function accessible(foreground: Color, background: Color): Color {
279 if (contrast(foreground, background).verdict !== "FAIL") return foreground;
280 const [h, s, l] = hsl(foreground);
281 for (let step = 1; step <= 100; step++) {
282 for (const lightness of [l - step, l + step]) {
283 if (lightness < 0 || lightness > 100) continue;
284 const candidate = fromHsl(h, s, lightness);
285 if (contrast(candidate, background).verdict !== "FAIL") return candidate;
286 }
287 }
288 throw new Error("Accessible color must exist");
289}

291function describe(color: Color): string {
292 const [h, s, l] = hsl(color);
293 if (l <= 5) return "black";
294 if (l >= 95) return "white";
295 const lightness = l <= 20 ? "very dark" : l <= 40 ? "dark" : l <= 60 ? "medium" : l <= 80 ? "light" : "very light";
296 if (s < 10) return lightness + " gray";
297 const saturation = s < 40 ? "muted" : s < 80 ? "moderate" : "vivid";
298 const names = ["red", "orange", "yellow", "chartreuse", "green", "spring-green", "cyan", "azure", "blue", "violet", "magenta", "rose"];
299 return lightness + " " + saturation + " " + names[floorRatio((h + 15) % 360, 30)];
300}

302function mean(colors: Color[]): Color {
303 return [0, 1, 2].map(component => roundRatio(colors.reduce((sum, c) => sum + c[component], 0), colors.length));
304}

306function palette(colors: Color[], count: number): Color[][] {
307 const boxes = [colors.slice()];
308 while (boxes.length < count) {
309 let chosen = -1, chosenWidth = -1, component = 0;
310 boxes.forEach((box, index) => {
311 if (box.length < 2) return;
312 const widths = [0, 1, 2].map(c => Math.max(...box.map(color => color[c])) - Math.min(...box.map(color => color[c])));
313 const width = Math.max(...widths);
314 if (width > chosenWidth) {
315 chosen = index;
316 chosenWidth = width;
317 component = widths.indexOf(width);
318 }
319 });
320 const colorsInOrder = boxes[chosen].toSorted((a, b) => a[component] - b[component]);
321 const middle = floorRatio(colorsInOrder.length, 2);
322 boxes.splice(chosen, 1, colorsInOrder.slice(0, middle), colorsInOrder.slice(middle));
323 }
324 return boxes;
325}

327function kmeans(colors: Color[], count: number): string {
328 let representatives = palette(colors, count).map(mean);
329 let counts: number[] = [], iterations = 0;
330 for (let iteration = 0; iteration < 20; iteration++) {
331 const groups: Color[][] = Array.from({length: count}, () => []);
332 for (const color of colors) {
333 let selected = 0, best = distance(color, representatives[0]);
334 for (let j = 1; j < count; j++) {
335 const value = distance(color, representatives[j]);
336 if (value < best) {
337 selected = j;
338 best = value;
339 }
340 }
341 groups[selected].push(color);
342 }
343 const next = groups.map((group, j) => group.length > 0 ? mean(group) : representatives[j]);
344 counts = groups.map(group => group.length);
345 iterations++;
346 const unchanged = next.every((color, j) => same(color, representatives[j]));
347 representatives = next;
348 if (unchanged) break;
349 }
350 return iterations + " " + representatives.map((color, j) => hex(color) + ":" + counts[j]).join(" ");
351}

353function dither(colors: Color[]): string {
354 let error = [0, 0, 0];
355 const names: string[] = [];
356 for (const color of colors) {
357 const adjusted = color.map((value, c) => clamp(value + error[c]));
358 let selected = 0, best = distance(adjusted, builtins[0].color);
359 for (let j = 1; j < 16; j++) {
360 const value = distance(adjusted, builtins[j].color);
361 if (value < best) {
362 selected = j;
363 best = value;
364 }
365 }
366 names.push(builtins[selected].name);
367 error = adjusted.map((value, c) => value - builtins[selected].color[c]);
368 }
369 return names.join(" ");
370}

386function execute(line: string): string {
387 const [command, ...args] = line.split(" ");
388 const arity = arities.get(command);
389 requireValid(arity !== undefined);
390 requireValid(args.length >= arity![0] && args.length <= arity![1]);
391 if (command === "DEFINE") {
392 const name = nameArgument(args[0]);
393 requireValid(!builtinMap.has(name));
394 const color = colorArgument(args[1]);
395 registered.set(name, color);
396 return name + " " + hex(color);
397 }
398 if (command === "UNDEFINE") {
399 const name = nameArgument(args[0]);
400 requireValid(registered.has(name));
401 registered.delete(name);
402 return name + " removed";
403 }
404 if (command === "LIST") {
405 const names = [...registered.keys()].sort();
406 return names.length === 0 ? "NONE" : names.map(name => name + "=" + hex(registered.get(name)!)).join(" ");
407 }
408 if (command === "PARSE") {
409 const color = colorArgument(args[0]);
410 const first = table().find(entry => same(entry.color, color));
411 const models = ["HSL", "HSV", "HWB", "CMYK", "YCBCR", "YIQ"].map(model => modelText(model, color));
412 return [hex(color), "rgb(" + color.join(",") + ")", ...models, first ? first.name : "-"].join(" ");
413 }
414 if (["HSL", "HSV", "HWB", "CMYK", "YCBCR", "YIQ"].includes(command)) {
415 return modelText(command, colorArgument(args[0]));
416 }
417 if (["FROMHSL", "FROMHSV", "FROMHWB"].includes(command)) {
418 const [h, second, third] = integers(args, [[0, 359], [0, 100], [0, 100]]);
419 if (command === "FROMHSL") return hex(fromHsl(h, second, third));
420 if (command === "FROMHSV") return hex(fromHsv(h, second, third));
421 return hex(fromHwb(h, second, third));
422 }
423 if (command === "FROMCMYK") {
424 const [c, m, y, k] = integers(args, Array.from({length: 4}, () => [0, 100] as [number, number]));
425 return hex([c, m, y].map(value => roundRatio(255 * (100 - value) * (100 - k), 10000)));
426 }
427 if (command === "FROMYCBCR") {
428 const [y, cb, cr] = integers(args, [[0, 255], [0, 255], [0, 255]]);
429 return hex([
430 clamp(roundRatio(1000 * y + 1402 * (cr - 128), 1000)),
431 clamp(roundRatio(1000 * y - 344 * (cb - 128) - 714 * (cr - 128), 1000)),
432 clamp(roundRatio(1000 * y + 1772 * (cb - 128), 1000))
433 ]);
434 }
435 if (command === "FROMYIQ") {
436 const [y, i, q] = integers(args, [[0, 255], [-255, 255], [-255, 255]]);
437 return hex([
438 clamp(roundRatio(1000 * y + 956 * i + 619 * q, 1000)),
439 clamp(roundRatio(1000 * y - 272 * i - 647 * q, 1000)),
440 clamp(roundRatio(1000 * y - 1106 * i + 1703 * q, 1000))
441 ]);
442 }
443 if (command === "FROM565") {
444 requireValid(/^0x[0-9a-fA-F]{4}$/.test(args[0]));
445 const value = parseInt(args[0].slice(2), 16);
446 return hex([
447 roundRatio(255 * floorRatio(value, 2048), 31),
448 roundRatio(255 * (floorRatio(value, 32) % 64), 63),
449 roundRatio(255 * (value % 32), 31)
450 ]);
451 }
452 if (command === "FROMANSI") return hex(ansiColor(integer(args[0], 0, 255)));
453 if (command === "RGB565") {
454 const [r, g, b] = colorArgument(args[0]);
455 const value = 2048 * roundRatio(31 * r, 255) + 32 * roundRatio(63 * g, 255) + roundRatio(31 * b, 255);
456 return "0x" + value.toString(16).toUpperCase().padStart(4, "0");
457 }
458 if (command === "ANSI") {
459 const color = colorArgument(args[0]);
460 let selected = 16, best = distance(color, ansiColor(16));
461 for (let code = 17; code <= 255; code++) {
462 const value = distance(color, ansiColor(code));
463 if (value < best) {
464 selected = code;
465 best = value;
466 }
467 }
468 return selected + " " + hex(ansiColor(selected)) + " " + best;
469 }
470 if (command === "WEBSAFE") return hex(colorArgument(args[0]).map(value => 51 * roundRatio(value, 51)));
471 if (command === "INVERT") return hex(colorArgument(args[0]).map(value => 255 - value));
472 if (command === "GRAY") {
473 const color = colorArgument(args[0]), [r, g, b] = color;
474 const method = keyword(args[1], ["LUMA", "AVERAGE", "LIGHTNESS", "HSP"]);
475 let value;
476 if (method === "LUMA") value = roundRatio(299 * r + 587 * g + 114 * b, 1000);
477 else if (method === "AVERAGE") value = roundRatio(r + g + b, 3);
478 else if (method === "LIGHTNESS") value = roundRatio(Math.max(...color) + Math.min(...color), 2);
479 else value = rootScaled(299 * r * r + 587 * g * g + 114 * b * b, 1000);
480 return hex([value, value, value]);
481 }
482 if (command === "MIX" || command === "MIXHSL") {
483 const a = colorArgument(args[0]), b = colorArgument(args[1]), p = integer(args[2], 0, 100);
484 if (command === "MIX") return hex(a.map((value, i) => roundRatio(value * (100 - p) + b[i] * p, 100)));
485 const [h1, s1, l1] = hsl(a), [h2, s2, l2] = hsl(b);
486 let delta = mod(h2 - h1, 360);
487 if (delta > 180) delta -= 360;
488 return hex(fromHsl(
489 mod(h1 + awayRatio(delta * p, 100), 360),
490 s1 + awayRatio((s2 - s1) * p, 100),
491 l1 + awayRatio((l2 - l1) * p, 100)
492 ));
493 }
494 if (command === "GRADIENT") {
495 const a = colorArgument(args[0]), b = colorArgument(args[1]), count = integer(args[2], 2, 16);
496 return gradient(a, b, count).map(hex).join(" ");
497 }
498 if (["TINTS", "SHADES", "TONES"].includes(command)) {
499 const color = colorArgument(args[0]), count = integer(args[1], 1, 10);
500 const target = command === "TINTS" ? [255, 255, 255] : command === "SHADES" ? [0, 0, 0] : [128, 128, 128];
501 return gradient(color, target, count + 2).slice(1, -1).map(hex).join(" ");
502 }
503 if (command === "BLEND") {
504 const mode = keyword(args[0], blendModes), base = colorArgument(args[1]), top = colorArgument(args[2]);
505 return hex(blend(mode, base, top));
506 }
507 if (command === "ALPHA") {
508 const foreground = colorArgument(args[0]), p = integer(args[1], 0, 100);
509 const background = colorArgument(args[2]), q = integer(args[3], 0, 100);
510 const opacity = 100 * p + q * (100 - p);
511 const color = opacity === 0 ? [0, 0, 0] : foreground.map((value, i) => {
512 const numerator = 100 * p * value + q * (100 - p) * background[i];
513 return roundRatio(numerator, opacity);
514 });
515 return hex(color) + " " + roundRatio(opacity, 100) + "%";
516 }
517 if (command === "SIMULATE") {
518 const color = colorArgument(args[0]), type = keyword(args[1], [...simulations.keys()]);
519 const result = simulations.get(type)!.map(row => roundRatio(row.reduce((sum, value, i) => sum + value * color[i], 0), 1000));
520 return hex(result);
521 }
522 if (command === "ROTATE") return hex(rotate(colorArgument(args[0]), integer(args[1], -720, 720)));
