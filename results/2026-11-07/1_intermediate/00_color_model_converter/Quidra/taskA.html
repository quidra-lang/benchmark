20int root_ratio(int numerator, int denominator)
21 low := 0
22 high := 1

135int hue(int[] color)
136 r := color[0]
137 g := color[1]
138 b := color[2]
139 largest := color.max()
140 delta := largest - color.min()
141 if delta == 0
142 return 0
143 numerator := 0
144 if largest == r
145 numerator = 60 * (g - b)
146 if numerator < 0
147 numerator += 360 * delta
148 elif largest == g
149 numerator = 60 * (b - r) + 120 * delta
150 else
151 numerator = 60 * (r - g) + 240 * delta
152 return modulo(rounded(numerator, delta), 360)

270int[][][] palette(int[][] colors, int count)
271 boxes := [colors]
272 while int(#boxes) < count
273 selected := -1
274 widest := -1
275 channel := 0
276 for: 0 <= i < int(#boxes)
277 if #boxes[i] < 2
278 continue
279 spans := widths(boxes[i])
280 component := 0
281 for: 1 <= j < 3
282 if spans[j] > spans[component]
283 component = j
284 if spans[component] > widest
285 selected = i
286 widest = spans[component]
287 channel = component
288 box := boxes[selected]
289 for: 1 <= i < int(#box)
290 value := box[i]
291 j := i - 1
292 while j >= 0 and box[j][channel] > value[channel]
293 box[j + 1] = box[j]
294 j -= 1
295 box[j + 1] = value
296 middle := int(#box) / 2
297 int[][][] next := []
298 for: 0 <= i < int(#boxes)
299 if i == selected
300 next = next.append(box[:<middle])
301 next = next.append(box[middle:])
302 else
303 next = next.append(boxes[i])
304 boxes = next
305 return boxes

307int blend_component(string mode, int x, int y)
308 multiplied := x * y / 255
309 screened := 255 - (255 - x) * (255 - y) / 255
310 if mode == "MULTIPLY"
311 return multiplied
312 if mode == "SCREEN"
313 return screened
314 if mode == "OVERLAY" or mode == "HARDLIGHT"
315 low := if mode == "OVERLAY" then x < 128 else y < 128
316 return if low then 2 * x * y / 255 else 255 - 2 * (255 - x) * (255 - y) / 255
317 if mode == "SOFTLIGHT"
318 return ((255 - x) * multiplied + x * screened) / 255
319 if mode == "DARKEN"
320 return math.min(x, y)
321 if mode == "LIGHTEN"
322 return math.max(x, y)
323 if mode == "DIFFERENCE"
324 return math.abs(x - y)
325 if mode == "EXCLUSION"
326 return x + y - 2 * x * y / 255
327 if mode == "ADD"
328 return math.min(255, x + y)
329 if mode == "SUBTRACT"
330 return math.max(0, x - y)
331 if mode == "COLORDODGE"
332 return if y == 255 then 255 else math.min(255, 255 * x / (255 - y))
333 return if y == 0 then 0 else math.max(0, 255 - 255 * (255 - x) / y)
