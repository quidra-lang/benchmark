4class ColorConverter
5 BUILTIN_HEX = {
6 'black' => '#000000', 'maroon' => '#800000', 'green' => '#008000', 'olive' => '#808000',
7 'navy' => '#000080', 'purple' => '#800080', 'teal' => '#008080', 'silver' => '#C0C0C0',
8 'gray' => '#808080', 'red' => '#FF0000', 'lime' => '#00FF00', 'yellow' => '#FFFF00',
9 'blue' => '#0000FF', 'magenta' => '#FF00FF', 'cyan' => '#00FFFF', 'white' => '#FFFFFF',
10 'orange' => '#FFA500', 'pink' => '#FFC0CB', 'brown' => '#A52A2A', 'gold' => '#FFD700',
11 'coral' => '#FF7F50', 'salmon' => '#FA8072', 'violet' => '#EE82EE', 'indigo' => '#4B0082'
12 }.freeze

18 MULTIPLY SCREEN OVERLAY HARDLIGHT SOFTLIGHT DARKEN LIGHTEN DIFFERENCE EXCLUSION

20 ].freeze
21 SIMULATIONS = {
22 'PROTANOPIA' => [[567, 433, 0], [558, 442, 0], [0, 242, 758]],
23 'DEUTERANOPIA' => [[625, 375, 0], [700, 300, 0], [0, 300, 700]],
24 'TRITANOPIA' => [[950, 50, 0], [0, 433, 567], [0, 475, 525]],
25 'ACHROMATOPSIA' => [[299, 587, 114], [299, 587, 114], [299, 587, 114]]

28 'COMPLEMENT' => [180], 'TRIADIC' => [120, 240],

53 def integer(token, low, high)
54 validate(token.match?(/\A-?\d{1,9}\z/))
55 value = token.to_i
56 validate(value.between?(low, high))
57 value

60 def integers(tokens, ranges)
61 tokens.each_with_index.map { |token, i| integer(token, *ranges[i]) }
62 end

64 def name_argument(token)
65 validate(token.match?(/\A[a-z]{1,12}\z/))
66 token

69 def keyword(token, choices)
70 validate(choices.include?(token))
71 token

74 def color_argument(token)
75 if token.match?(/\A#[0-9a-fA-F]{6}\z/)
76 return [1, 3, 5].map { |start| token[start, 2].to_i(16) }
77 end
78 if token.match?(/\A#[0-9a-fA-F]{3}\z/)
79 return token[1..].chars.map { |digit| (digit * 2).to_i(16) }
80 end
81 match = token.match(/\Argb\((\d{1,3}),(\d{1,3}),(\d{1,3})\)\z/)
82 if match
83 color = match.captures.map(&:to_i)
84 validate(color.all? { |component| component <= 255 })
85 return color
86 end
87 color = BUILTINS[token] || @registered[token]
88 validate(!color.nil?)
89 color.dup
90 end

96 def hex(color)
97 '#' + color.map { |component| format('%02X', component) }.join

100 def round_exact(value)
101 (value + Rational(1, 2)).floor
102 end
103
104 def round_ratio(numerator, denominator)
105 (2 * numerator + denominator) / (2 * denominator)
106 end
107
108 def away_ratio(numerator, denominator)
109 numerator.negative? ? -round_ratio(-numerator, denominator) : round_ratio(numerator, denominator)
110 end
111
112 def clamp(value)
113 value.clamp(0, 255)
114 end
115
116 def distance(a, b)
117 a.each_with_index.sum { |value, i| (value - b[i])**2 }
118 end
119
120 def hue(color)
121 r, g, b = color
122 maximum, minimum = color.max, color.min
123 difference = maximum - minimum
124 return 0 if difference.zero?
125 if maximum == r
126 value = Rational(60 * (g - b), difference)
127 value += 360 if value.negative?
128 elsif maximum == g
129 value = Rational(60 * (b - r), difference) + 120
130 else
131 value = Rational(60 * (r - g), difference) + 240
132 end
133 round_exact(value) % 360
134 end
135
136 def hsl(color)
137 maximum, minimum = color.max, color.min
138 difference = maximum - minimum
139 saturation = difference.zero? ? 0 : round_ratio(100 * difference, 255 - (maximum + minimum - 255).abs)
140 [hue(color), saturation, round_ratio(100 * (maximum + minimum), 510)]
141 end
142
143 def base(h, chroma)
144 sector, fraction = h.divmod(60)
145 x = chroma * Rational(sector.even? ? fraction : 60 - fraction, 60)
146 [
147 [chroma, x, 0], [x, chroma, 0], [0, chroma, x],
148 [0, x, chroma], [x, 0, chroma], [chroma, 0, x]
149 ][sector]
150 end
151
152 def from_hsl(h, s, l)
153 chroma = Rational((100 - (2 * l - 100).abs) * s, 10000)
154 offset = Rational(l, 100) - chroma / 2
155 base(h, chroma).map { |component| round_exact(255 * (component + offset)) }
156 end
157
158 def from_hsv(h, s, v)
159 chroma = Rational(v * s, 10000)
160 offset = Rational(v, 100) - chroma
161 base(h, chroma).map { |component| round_exact(255 * (component + offset)) }
162 end
163
164 def from_hwb(h, w, k)
165 return Array.new(3, round_ratio(255 * w, w + k)) if w + k >= 100
166 base(h, Rational(1, 1)).map do |component|
167 round_exact(255 * (component * Rational(100 - w - k, 100) + Rational(w, 100)))
168 end
169 end
170
171 def model_values(command, color)
172 r, g, b = color
173 maximum, minimum = color.max, color.min
174 case command
175 when 'HSL'
176 hsl(color)
177 when 'HSV'
178 saturation = maximum.zero? ? 0 : round_ratio(100 * (maximum - minimum), maximum)
179 [hue(color), saturation, round_ratio(100 * maximum, 255)]
180 when 'HWB'
181 [hue(color), round_ratio(100 * minimum, 255), round_ratio(100 * (255 - maximum), 255)]
182 when 'CMYK'
183 return [0, 0, 0, 100] if maximum.zero?
184 color.map { |component| round_ratio(100 * (maximum - component), maximum) } +
185 [round_ratio(100 * (255 - maximum), 255)]
186 when 'YCBCR'
187 [
188 clamp(round_ratio(299 * r + 587 * g + 114 * b, 1000)),
189 clamp(round_ratio(128000 - 169 * r - 331 * g + 500 * b, 1000)),
190 clamp(round_ratio(128000 + 500 * r - 419 * g - 81 * b, 1000))
191 ]
192 when 'YIQ'
193 [
194 round_ratio(299 * r + 587 * g + 114 * b, 1000),
195 round_ratio(596 * r - 274 * g - 322 * b, 1000),
196 round_ratio(211 * r - 523 * g + 312 * b, 1000)
197 ]
198 end
199 end

201 def model_text(command, color)
202 values = model_values(command, color)
203 body = if %w[HSL HSV HWB].include?(command)
204 "#{values[0]},#{values[1]}%,#{values[2]}%"
205 elsif command == 'CMYK'
206 values.map { |value| "#{value}%" }.join(',')
207 else
208 values.join(',')
209 end
210 "#{command.downcase}(#{body})"
211 end

213 def ansi_color(code)
214 return BUILTINS.values[code].dup if code <= 15
215 return Array.new(3, 8 + 10 * (code - 232)) if code >= 232
216 levels = [0, 95, 135, 175, 215, 255]
217 value = code - 16
218 [levels[value / 36], levels[value / 6 % 6], levels[value % 6]]
219 end

221 def gradient(a, b, count)
222 (0...count).map do |index|
223 a.each_with_index.map do |value, component|
224 value + away_ratio((b[component] - value) * index, count - 1)
225 end
226 end
227 end

229 def rotate(color, degrees)
230 h, s, l = hsl(color)
231 from_hsl((h + degrees) % 360, s, l)
232 end

234 def blend_channel(mode, x, y)
235 case mode
236 when 'MULTIPLY'
237 x * y / 255
238 when 'SCREEN'
239 255 - (255 - x) * (255 - y) / 255
240 when 'OVERLAY', 'HARDLIGHT'
241 if (mode == 'OVERLAY' ? x : y) < 128
242 2 * x * y / 255
243 else
244 255 - 2 * (255 - x) * (255 - y) / 255
245 end
246 when 'SOFTLIGHT'
247 multiply = x * y / 255
248 screen = 255 - (255 - x) * (255 - y) / 255
249 ((255 - x) * multiply + x * screen) / 255
250 when 'DARKEN'
251 [x, y].min
252 when 'LIGHTEN'
253 [x, y].max
254 when 'DIFFERENCE'
255 (x - y).abs
256 when 'EXCLUSION'
257 x + y - 2 * x * y / 255
258 when 'ADD'
259 [255, x + y].min
260 when 'SUBTRACT'
261 [0, x - y].max
262 when 'COLORDODGE'
263 y == 255 ? 255 : [255, 255 * x / (255 - y)].min
264 when 'COLORBURN'
265 y.zero? ? 0 : [0, 255 - 255 * (255 - x) / y].max
266 end
267 end

269 def blend(mode, base_color, top)
270 h1, s1, l1 = hsl(base_color)
271 h2, s2, l2 = hsl(top)
272 case mode
273 when 'HUE' then from_hsl(h2, s1, l1)
274 when 'SATURATION' then from_hsl(h1, s2, l1)
275 when 'COLOR' then from_hsl(h2, s2, l1)
276 when 'LUMINOSITY' then from_hsl(h1, s1, l2)
277 else
278 base_color.each_with_index.map { |value, i| blend_channel(mode, value, top[i]) }
279 end
280 end

282 def contrast(a, b)
283 luminance = ->(color) { 2126 * color[0]**2 + 7152 * color[1]**2 + 722 * color[2]**2 }
284 first, second = luminance.call(a), luminance.call(b)
285 high, low = [first, second].max + 32512500, [first, second].min + 32512500
286 verdict = high >= 7 * low ? 'AAA' : 2 * high >= 9 * low ? 'AA' : 'FAIL'
287 [round_ratio(100 * high, low), verdict]
288 end

290 def contrast_text(info)
291 format('%d.%02d %s', info[0] / 100, info[0] % 100, info[1])
292 end

294 def accessible(foreground, background)
295 return foreground if contrast(foreground, background)[1] != 'FAIL'
296 h, s, l = hsl(foreground)
297 (1..100).each do |step|
298 [l - step, l + step].each do |lightness|
299 next unless lightness.between?(0, 100)
300 candidate = from_hsl(h, s, lightness)
301 return candidate if contrast(candidate, background)[1] != 'FAIL'
302 end
303 end
304 raise 'Accessible color must exist'
305 end

307 def describe(color)
308 h, s, l = hsl(color)
309 return 'black' if l <= 5
310 return 'white' if l >= 95
311 lightness = if l <= 20 then 'very dark'
312 elsif l <= 40 then 'dark'
313 elsif l <= 60 then 'medium'
314 elsif l <= 80 then 'light'
315 else 'very light'
316 end
317 return "#{lightness} gray" if s < 10
318 saturation = s < 40 ? 'muted' : s < 80 ? 'moderate' : 'vivid'
319 names = %w[red orange yellow chartreuse green spring-green cyan azure blue violet magenta rose]
320 "#{lightness} #{saturation} #{names[((h + 15) % 360) / 30]}"
321 end

323 def mean(colors)
324 (0...3).map { |component| round_ratio(colors.sum { |color| color[component] }, colors.length) }
325 end

327 def palette(colors, count)
328 boxes = [colors.dup]
329 while boxes.length < count
330 chosen = -1
331 chosen_width = -1
332 component = 0
333 boxes.each_with_index do |box, index|
334 next if box.length < 2
335 widths = (0...3).map do |c|
336 minimum, maximum = box.map { |color| color[c] }.minmax
337 maximum - minimum
338 end
339 width = widths.max
340 if width > chosen_width
341 chosen = index
342 chosen_width = width
343 component = widths.index(width)
344 end
345 end
346 sorted = boxes[chosen].each_with_index.sort_by { |color, index| [color[component], index] }.map(&:first)
347 middle = sorted.length / 2
348 boxes[chosen, 1] = [sorted.take(middle), sorted.drop(middle)]
349 end
350 boxes
351 end

353 def kmeans(colors, count)
354 representatives = palette(colors, count).map { |box| mean(box) }
355 counts = []
356 iterations = 0
357 20.times do
358 groups = Array.new(count) { [] }
359 colors.each do |color|
360 selected = (0...count).min_by { |j| distance(color, representatives[j]) }
361 groups[selected] << color
362 end
363 following = groups.each_with_index.map do |group, j|
364 group.empty? ? representatives[j] : mean(group)
365 end
366 counts = groups.map(&:length)
367 iterations += 1
368 unchanged = following == representatives
369 representatives = following
370 break if unchanged
371 end
372 items = representatives.each_with_index.map { |color, j| "#{hex(color)}:#{counts[j]}" }
373 "#{iterations} #{items.join(' ')}"
374 end

376 def dither(colors)
377 error = [0, 0, 0]
378 names = []
379 entries = BUILTINS.first(16)
380 colors.each do |color|
381 adjusted = color.each_with_index.map { |value, component| clamp(value + error[component]) }
382 selected = (0...16).min_by { |j| distance(adjusted, entries[j][1]) }
383 names << entries[selected][0]
384 error = adjusted.each_with_index.map { |value, component| value - entries[selected][1][component] }
385 end
386 names.join(' ')
387 end

389 def execute(line)
390 command, *args = line.split(' ')
391 limits = ARITIES[command]
392 validate(!limits.nil? && args.length.between?(*limits))
393 case command
394 when 'DEFINE'
395 name = name_argument(args[0])
396 validate(!BUILTINS.key?(name))
397 color = color_argument(args[1])
398 @registered[name] = color
399 "#{name} #{hex(color)}"
400 when 'UNDEFINE'
401 name = name_argument(args[0])
402 validate(@registered.key?(name))
403 @registered.delete(name)
404 "#{name} removed"
405 when 'LIST'
406 return 'NONE' if @registered.empty?
407 @registered.keys.sort.map { |name| "#{name}=#{hex(@registered[name])}" }.join(' ')
408 when 'PARSE'
409 color = color_argument(args[0])
410 first = table.find { |_, value| value == color }
411 models = %w[HSL HSV HWB CMYK YCBCR YIQ].map { |model| model_text(model, color) }
412 [hex(color), "rgb(#{color.join(',')})", *models, first ? first[0] : '-'].join(' ')
413 when 'HSL', 'HSV', 'HWB', 'CMYK', 'YCBCR', 'YIQ'
414 model_text(command, color_argument(args[0]))
415 when 'FROMHSL', 'FROMHSV', 'FROMHWB'
416 h, second, third = integers(args, [[0, 359], [0, 100], [0, 100]])
417 color = if command == 'FROMHSL' then from_hsl(h, second, third)
418 elsif command == 'FROMHSV' then from_hsv(h, second, third)
419 else from_hwb(h, second, third)
420 end
421 hex(color)
422 when 'FROMCMYK'
423 c, m, y, k = integers(args, Array.new(4) { [0, 100] })
424 hex([c, m, y].map { |value| round_ratio(255 * (100 - value) * (100 - k), 10000) })
425 when 'FROMYCBCR'
426 y, cb, cr = integers(args, [[0, 255], [0, 255], [0, 255]])
427 hex([
428 clamp(round_ratio(1000 * y + 1402 * (cr - 128), 1000)),
429 clamp(round_ratio(1000 * y - 344 * (cb - 128) - 714 * (cr - 128), 1000)),
430 clamp(round_ratio(1000 * y + 1772 * (cb - 128), 1000))
431 ])
432 when 'FROMYIQ'
433 y, i, q = integers(args, [[0, 255], [-255, 255], [-255, 255]])
434 hex([
435 clamp(round_ratio(1000 * y + 956 * i + 619 * q, 1000)),
436 clamp(round_ratio(1000 * y - 272 * i - 647 * q, 1000)),
437 clamp(round_ratio(1000 * y - 1106 * i + 1703 * q, 1000))
438 ])
439 when 'FROM565'
440 validate(args[0].match?(/\A0x[0-9a-fA-F]{4}\z/))
441 value = args[0][2..].to_i(16)
442 hex([
443 round_ratio(255 * (value / 2048), 31),
444 round_ratio(255 * (value / 32 % 64), 63),
445 round_ratio(255 * (value % 32), 31)
446 ])
447 when 'FROMANSI'
448 hex(ansi_color(integer(args[0], 0, 255)))
449 when 'RGB565'
450 r, g, b = color_argument(args[0])
451 value = 2048 * round_ratio(31 * r, 255) + 32 * round_ratio(63 * g, 255) + round_ratio(31 * b, 255)
452 format('0x%04X', value)
453 when 'ANSI'
454 color = color_argument(args[0])
455 selected = (16..255).min_by { |code| distance(color, ansi_color(code)) }
456 best = distance(color, ansi_color(selected))
457 "#{selected} #{hex(ansi_color(selected))} #{best}"
458 when 'WEBSAFE'
459 hex(color_argument(args[0]).map { |value| 51 * round_ratio(value, 51) })
460 when 'INVERT'
461 hex(color_argument(args[0]).map { |value| 255 - value })
462 when 'GRAY'
463 color = color_argument(args[0])
464 r, g, b = color
465 method = keyword(args[1], %w[LUMA AVERAGE LIGHTNESS HSP])
466 value = case method
467 when 'LUMA' then round_ratio(299 * r + 587 * g + 114 * b, 1000)
468 when 'AVERAGE' then round_ratio(r + g + b, 3)
469 when 'LIGHTNESS' then round_ratio(color.max + color.min, 2)
470 else Integer.sqrt((299 * r * r + 587 * g * g + 114 * b * b) / 1000)
471 end
472 hex([value, value, value])
473 when 'MIX', 'MIXHSL'
474 a, b = color_argument(args[0]), color_argument(args[1])
475 p = integer(args[2], 0, 100)
476 if command == 'MIX'
477 return hex(a.each_with_index.map { |value, i| round_ratio(value * (100 - p) + b[i] * p, 100) })
478 end
479 h1, s1, l1 = hsl(a)
480 h2, s2, l2 = hsl(b)
481 delta = (h2 - h1) % 360
482 delta -= 360 if delta > 180
483 hex(from_hsl(
484 (h1 + away_ratio(delta * p, 100)) % 360,
485 s1 + away_ratio((s2 - s1) * p, 100),
486 l1 + away_ratio((l2 - l1) * p, 100)
487 ))
488 when 'GRADIENT'
489 a, b = color_argument(args[0]), color_argument(args[1])
490 count = integer(args[2], 2, 16)
491 gradient(a, b, count).map { |color| hex(color) }.join(' ')
492 when 'TINTS', 'SHADES', 'TONES'
493 color = color_argument(args[0])
494 count = integer(args[1], 1, 10)
495 target = command == 'TINTS' ? [255, 255, 255] : command == 'SHADES' ? [0, 0, 0] : [128, 128, 128]
496 gradient(color, target, count + 2)[1...-1].map { |value| hex(value) }.join(' ')
497 when 'BLEND'
498 mode = keyword(args[0], BLEND_MODES)
499 base_color, top = color_argument(args[1]), color_argument(args[2])
500 hex(blend(mode, base_color, top))
501 when 'ALPHA'
502 foreground = color_argument(args[0])
503 p = integer(args[1], 0, 100)
504 background = color_argument(args[2])
505 q = integer(args[3], 0, 100)
506 opacity = 100 * p + q * (100 - p)
507 color = if opacity.zero?
508 [0, 0, 0]
509 else
510 foreground.each_with_index.map do |value, i|
511 round_ratio(100 * p * value + q * (100 - p) * background[i], opacity)
512 end
513 end
514 "#{hex(color)} #{round_ratio(opacity, 100)}%"
515 when 'SIMULATE'
516 color = color_argument(args[0])
517 type = keyword(args[1], SIMULATIONS.keys)
518 result = SIMULATIONS[type].map do |row|
519 round_ratio(row.each_with_index.sum { |value, i| value * color[i] }, 1000)
520 end
521 hex(result)
522 when 'ROTATE'
523 hex(rotate(color_argument(args[0]), integer(args[1], -720, 720)))
524 when 'COMPLEMENT', 'TRIADIC', 'TETRADIC', 'SPLITCOMP'
525 color = color_argument(args[0])
526 ROTATIONS[command].map { |degrees| hex(rotate(color, degrees)) }.join(' ')
527 when 'ANALOGOUS'
528 color = color_argument(args[0])
529 k, step = integer(args[1], 1, 6), integer(args[2], 1, 60)
530 (-k..k).reject(&:zero?).map { |i| hex(rotate(color, i * step)) }.join(' ')
531 when 'LIGHTEN', 'DARKEN', 'SATURATE', 'DESATURATE'
532 color = color_argument(args[0])
533 p = integer(args[1], 0, 100)
534 h, s, l = hsl(color)
535 l = [100, l + p].min if command == 'LIGHTEN'
536 l = [0, l - p].max if command == 'DARKEN'
537 s = [100, s + p].min if command == 'SATURATE'
538 s = [0, s - p].max if command == 'DESATURATE'
539 hex(from_hsl(h, s, l))
540 when 'NEAREST'
541 color = color_argument(args[0])
542 selected = table.min_by { |entry| distance(color, entry[1]) }
543 best = distance(color, selected[1])
544 "#{selected[0]} #{hex(selected[1])} #{best}"
545 when 'DISTANCE'
546 a, b = color_argument(args[0]), color_argument(args[1])
547 dr, dg, db = a.each_with_index.map { |value, i| value - b[i] }
548 weighted = (1024 + a[0] + b[0]) * dr * dr + 2048 * dg * dg + (1534 - a[0] - b[0]) * db * db
549 "#{distance(a, b)} #{Integer.sqrt(weighted / 512)}"
550 when 'CONTRAST'
551 contrast_text(contrast(color_argument(args[0]), color_argument(args[1])))
552 when 'ACCESSIBLE'
553 foreground, background = color_argument(args[0]), color_argument(args[1])
554 result = accessible(foreground, background)
555 "#{hex(result)} #{contrast_text(contrast(result, background))}"
556 when 'BEST'
557 background, *colors = args.map { |token| color_argument(token) }
558 chosen = colors.max_by { |color| contrast(color, background)[0] }
559 info = contrast(chosen, background)
560 "#{hex(chosen)} #{contrast_text(info)}"
561 when 'DESCRIBE'
562 describe(color_argument(args[0]))
563 when 'PALETTE', 'KMEANS'
564 count = integer(args[0], 1, args.length - 1)
565 colors = args.drop(1).map { |token| color_argument(token) }
566 return kmeans(colors, count) if command == 'KMEANS'
567 palette(colors, count).map { |box| "#{hex(mean(box))}:#{box.length}" }.join(' ')
568 when 'SORT'
569 records = args.each_with_index.map { |token, index| [color_argument(token), index] }
570 records.sort_by { |color, index| [*hsl(color), index] }.map { |color, _| hex(color) }.join(' ')
571 when 'DITHER'
