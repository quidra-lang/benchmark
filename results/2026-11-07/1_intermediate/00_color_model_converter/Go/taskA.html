39func abs(x int64) int64 {
40 if x < 0 { return -x }

44func clamp(x int64) int64 { return min(255, max(0, x)) }

46func rounded(numerator, denominator int64) int64 {
47 numerator = 2*numerator + denominator
48 denominator *= 2
49 quotient := numerator / denominator
50 if numerator < 0 && numerator%denominator != 0 { quotient-- }
51 return quotient
52}

54func away(numerator, denominator int64) int64 {
55 if numerator < 0 { return -rounded(-numerator, denominator) }
56 return rounded(numerator, denominator)
57}

59func modulo(value, modulus int64) int64 {
60 result := value % modulus
61 if result < 0 { result += modulus }
62 return result
63}

65func distance(a, b Color) int64 {
66 result := int64(0)
67 for j := 0; j < 3; j++ {
68 difference := a[j] - b[j]
69 result += difference * difference
70 }
71 return result
72}

74func extrema(c Color) (int64, int64) {
75 return max(c[0], c[1], c[2]), min(c[0], c[1], c[2])
76}

78func hue(c Color) int64 {
79 maximum, minimum := extrema(c)
80 delta := maximum - minimum
81 if delta == 0 { return 0 }
82 numerator := int64(0)
83 if maximum == c[0] {
84 numerator = 60 * (c[1] - c[2])
85 if numerator < 0 { numerator += 360 * delta }
86 } else if maximum == c[1] {
87 numerator = 60*(c[2]-c[0]) + 120*delta
88 } else {
89 numerator = 60*(c[0]-c[1]) + 240*delta
90 }
91 return rounded(numerator, delta) % 360
92}

94func hsl(c Color) [3]int64 {
95 maximum, minimum := extrema(c)
96 delta := maximum - minimum
97 saturation := int64(0)
98 if delta != 0 {
99 saturation = rounded(100*delta, 255-abs(maximum+minimum-255))
100 }
101 return [3]int64{hue(c), saturation, rounded(100*(maximum+minimum), 510)}
102}

104func sector(h int64) Color {
105 q, f := h/60, h%60
106 x := f
107 if q%2 != 0 { x = 60-f }
108 switch q {
109 case 0: return Color{60, x, 0}
110 case 1: return Color{x, 60, 0}
111 case 2: return Color{0, 60, x}
112 case 3: return Color{0, x, 60}
113 case 4: return Color{x, 0, 60}
114 default: return Color{60, 0, x}
115 }
116}

118func fromHSL(h, s, l int64) Color {
119 chroma := (100-abs(2*l-100)) * s
120 offset := 30 * (200*l-chroma)
121 result := sector(h)
122 for j := range result {
123 result[j] = rounded(255*(result[j]*chroma+offset), 600000)
124 }
125 return result
126}

128func fromHSV(h, s, v int64) Color {
129 chroma := v*s
130 offset := 60 * (100*v-chroma)
131 result := sector(h)
132 for j := range result {
133 result[j] = rounded(255*(result[j]*chroma+offset), 600000)
134 }
135 return result
136}

138func fromHWB(h, w, k int64) Color {
139 if w+k >= 100 {
140 value := rounded(255*w, w+k)
141 return Color{value, value, value}
142 }
143 result := sector(h)
144 for j := range result {
145 result[j] = rounded(255*(result[j]*(100-w-k)+60*w), 6000)
146 }
147 return result
148}

150func integerRoot(value int64) int64 {
151 low, high := int64(0), int64(1)
152 for high*high <= value { high *= 2 }
153 for low+1 < high {
154 middle := (low+high)/2
155 if middle*middle <= value { low = middle } else { high = middle }
156 }
157 return low
158}

161 levels := [6]int64{0, 95, 135, 175, 215, 255}
162 if index < 16 { return builtins[index].color }
163 if index >= 232 {
164 value := int64(8 + 10*(index-232))

167 code := index-16
168 return Color{levels[code/36], levels[code/6%6], levels[code%6]}

171func contrast(a, b Color) (int64, string) {
172 luminance := func(c Color) int64 {
173 return 2126*c[0]*c[0] + 7152*c[1]*c[1] + 722*c[2]*c[2]
174 }
175 y1, y2 := luminance(a), luminance(b)
176 high, low := max(y1,y2)+32512500, min(y1,y2)+32512500
177 verdict := "FAIL"
178 if high >= 7*low { verdict = "AAA" } else if 2*high >= 9*low { verdict = "AA" }
179 return rounded(100*high, low), verdict
180}

202func blend(mode int, base, top Color) Color {
203 if mode >= 13 {
204 a, b := hsl(base), hsl(top)
205 if mode == 13 || mode == 15 { a[0] = b[0] }
206 if mode == 14 || mode == 15 { a[1] = b[1] }
207 if mode == 16 { a[2] = b[2] }
208 return fromHSL(a[0], a[1], a[2])
209 }
210 result := base
211 for j := range result {
212 x, y := base[j], top[j]
213 switch mode {
214 case 0: result[j] = x*y/255
215 case 1: result[j] = 255-(255-x)*(255-y)/255
216 case 2, 3:
217 condition := x
218 if mode == 3 { condition = y }
219 if condition < 128 { result[j] = 2*x*y/255
220 } else { result[j] = 255-2*(255-x)*(255-y)/255 }
221 case 4:
222 multiply := x*y/255
223 screen := 255-(255-x)*(255-y)/255
224 result[j] = ((255-x)*multiply+x*screen)/255
225 case 5: result[j] = min(x,y)
226 case 6: result[j] = max(x,y)
227 case 7: result[j] = abs(x-y)
228 case 8: result[j] = x+y-2*x*y/255
229 case 9: result[j] = min(255,x+y)
230 case 10: result[j] = max(0,x-y)
231 case 11:
232 result[j] = 255
233 if y != 255 { result[j] = min(255,255*x/(255-y)) }
234 case 12:
235 result[j] = 0
236 if y != 0 { result[j] = max(0,255-255*(255-x)/y) }
237 }
238 }
239 return result
240}

242func mean(colors []Color) Color {
243 result := Color{}
244 for _, color := range colors {
245 for j := range result { result[j] += color[j] }
246 }
247 for j := range result { result[j] = rounded(result[j],int64(len(colors))) }
248 return result
249}

251func palette(colors []Color, wanted int) []Group {
252 boxes := [][]Color{slices.Clone(colors)}
253 for len(boxes) < wanted {
254 chosen, channel, bestWidth := 0, 0, int64(-1)
255 for i, colors := range boxes {
256 if len(colors) < 2 { continue }
257 width, component := int64(-1), 0
258 for j := 0; j < 3; j++ {
259 minimum, maximum := int64(255), int64(0)
260 for _, color := range colors {
261 minimum = min(minimum,color[j])
262 maximum = max(maximum,color[j])
263 }
264 if maximum-minimum > width { width, component = maximum-minimum, j }
265 }
266 if width > bestWidth { chosen, channel, bestWidth = i, component, width }
267 }
268 selected := boxes[chosen]
269 sort.SliceStable(selected,func(i,j int) bool { return selected[i][channel] < selected[j][channel] })
270 middle := len(selected)/2
271 boxes = slices.Replace(boxes, chosen, chosen+1, selected[:middle], selected[middle:])
272 }
273 result := make([]Group,len(boxes))
274 for i, colors := range boxes { result[i] = Group{mean(colors),len(colors)} }
275 return result
276}

278func kmeans(colors []Color, wanted int) (int, []Group) {
279 representatives := palette(colors,wanted)
280 for iteration := 1; iteration <= 20; iteration++ {
281 sums, counts := make([]Color,wanted), make([]int,wanted)
282 for _, color := range colors {
283 closest := 0
284 for j := 1; j < wanted; j++ {
285 if distance(color,representatives[j].color) < distance(color,representatives[closest].color) {
286 closest = j
287 }
288 }
289 counts[closest]++
290 for j := 0; j < 3; j++ { sums[closest][j] += color[j] }
291 }
292 unchanged := true
293 for i := range representatives {
294 representatives[i].count = counts[i]
295 if counts[i] == 0 { continue }
296 for j := 0; j < 3; j++ { sums[i][j] = rounded(sums[i][j],int64(counts[i])) }
297 if sums[i] != representatives[i].color { unchanged = false }
298 representatives[i].color = sums[i]
299 }
300 if unchanged || iteration == 20 { return iteration, representatives }
301 }
302 panic("unreachable")
303}

305func parseInteger(text string, low, high int64) (int64, bool) {
306 digits := text
307 if strings.HasPrefix(text,"-") { digits = text[1:] }
308 if len(digits) < 1 || len(digits) > 9 { return 0,false }
309 for i := range digits { if digits[i] < '0' || digits[i] > '9' { return 0,false } }
310 value, err := strconv.ParseInt(text,10,64)
311 return value, err == nil && value >= low && value <= high
312}

314func validName(text string) bool {
315 if len(text) < 1 || len(text) > 12 { return false }
316 for i := range text { if text[i] < 'a' || text[i] > 'z' { return false } }
317 return true

320func hexDigit(c byte) (int64, bool) {
321 if c >= '0' && c <= '9' { return int64(c-'0'),true }
322 if c >= 'a' && c <= 'f' { return int64(c-'a'+10),true }
323 if c >= 'A' && c <= 'F' { return int64(c-'A'+10),true }
324 return 0,false
325}

327func parseColor(text string, registry []Entry) (Color, bool) {
328 result := Color{}
329 if strings.HasPrefix(text,"#") {
330 if len(text) != 4 && len(text) != 7 { return result,false }
331 for j := range result {
332 if len(text) == 4 {
333 digit, ok := hexDigit(text[j+1])
334 if !ok { return result,false }
335 result[j] = 17*digit
336 } else {
337 hi, ok1 := hexDigit(text[2*j+1])
338 lo, ok2 := hexDigit(text[2*j+2])
339 if !ok1 || !ok2 { return result,false }
340 result[j] = 16*hi+lo
341 }
342 }
343 return result,true
344 }
345 if strings.HasPrefix(text,"rgb(") {
346 if !strings.HasSuffix(text,")") { return result,false }
347 parts := strings.Split(text[4:len(text)-1],",")
348 if len(parts) != 3 { return result,false }
349 for j, part := range parts {
350 if len(part) < 1 || len(part) > 3 { return result,false }
351 for i := range part { if part[i] < '0' || part[i] > '9' { return result,false } }
352 value, ok := parseInteger(part,0,255)
353 if !ok { return result,false }
354 result[j] = value
355 }
356 return result,true
357 }
358 for _, entry := range builtins { if entry.name == text { return entry.color,true } }
359 for _, entry := range registry { if entry.name == text { return entry.color,true } }
360 return result,false
361}

368func prepare(spec Spec, words []string, registry []Entry) (Arguments, bool) {
369 args := Arguments{}
370 if len(words) < spec.minimum || len(words) > spec.maximum { return args,false }
371 for i, word := range words {
372 kind := byte(0)
373 switch spec.signature[0] {
374 case '*': kind = 'C'
375 case 'P':
376 kind = 'C'
377 if i == 0 { kind = 'I' }
378 default: kind = spec.signature[i]
379 }
380 switch kind {
381 case 'C':
382 color, ok := parseColor(word,registry)
383 if !ok { return args,false }
384 args.colors[i] = color
385 case 'I':
386 low, high := int64(1), int64(32)
387 if spec.signature != "P" { low, high = spec.low[i], spec.high[i] }
388 value, ok := parseInteger(word,low,high)
389 if !ok { return args,false }
390 args.numbers[i] = value
391 case 'N':
392 if !validName(word) { return args,false }
393 case 'K':
394 args.keyword = slices.Index(strings.Split(spec.keywords, "|"), word)
395 if args.keyword < 0 { return args,false }
396 case 'W':
397 if len(word) != 6 || !strings.HasPrefix(word,"0x") { return args,false }
398 value := int64(0)
399 for j := 2; j < 6; j++ {
400 digit, ok := hexDigit(word[j])
401 if !ok { return args,false }
402 value = 16*value+digit
403 }
404 args.numbers[i] = value
405 }
406 }
407 if spec.signature == "P" && args.numbers[0] > int64(len(words)-1) { return args,false }
408 if spec.name == "DEFINE" {
409 for _, entry := range builtins { if entry.name == words[0] { return args,false } }
410 }
411 if spec.name == "UNDEFINE" {
412 found := false
413 for _, entry := range registry { if entry.name == words[0] { found = true } }
414 if !found { return args,false }
415 }
416 return args,true
417}

419func model(name string, c Color) string {
420 maximum, minimum := extrema(c)
421 r, g, b := c[0], c[1], c[2]
422 switch name {
423 case "HSL":
424 value := hsl(c)
425 return fmt.Sprintf("hsl(%d,%d%%,%d%%)",value[0],value[1],value[2])
426 case "HSV":
427 saturation := int64(0)
428 if maximum != 0 { saturation = rounded(100*(maximum-minimum),maximum) }
429 return fmt.Sprintf("hsv(%d,%d%%,%d%%)",hue(c),saturation,rounded(100*maximum,255))
430 case "HWB":
431 return fmt.Sprintf("hwb(%d,%d%%,%d%%)",hue(c),rounded(100*minimum,255),rounded(100*(255-maximum),255))
432 case "CMYK":
433 cyan, magenta, yellow := int64(0), int64(0), int64(0)
434 if maximum != 0 {
435 cyan = rounded(100*(maximum-r),maximum)
436 magenta = rounded(100*(maximum-g),maximum)
437 yellow = rounded(100*(maximum-b),maximum)
438 }
439 return fmt.Sprintf("cmyk(%d%%,%d%%,%d%%,%d%%)",cyan,magenta,yellow,rounded(100*(255-maximum),255))
440 case "YCBCR":
441 return fmt.Sprintf("ycbcr(%d,%d,%d)",clamp(rounded(299*r+587*g+114*b,1000)),
442 clamp(rounded(128000-169*r-331*g+500*b,1000)),clamp(rounded(128000+500*r-419*g-81*b,1000)))
443 default:
444 return fmt.Sprintf("yiq(%d,%d,%d)",rounded(299*r+587*g+114*b,1000),
445 rounded(596*r-274*g-322*b,1000),rounded(211*r-523*g+312*b,1000))
446 }
447}

532func run(line string, registry *[]Entry) (string, bool) {
533 tokens := strings.Split(line," ")
534 var spec *Spec
535 for i := range specs {
536 if specs[i].name == tokens[0] { spec = &specs[i]; break }
537 }
538 if spec == nil { return "",false }
539 words := tokens[1:]
540 args, valid := prepare(*spec,words,*registry)
541 if !valid { return "",false }
542 c := args.colors[0]
543 a, b, d := args.numbers[0], args.numbers[1], args.numbers[2]
544 output := []string{}
545 switch spec.name {
546 case "DEFINE":
547 output = append(output,fmt.Sprintf("%s %s",words[0],hex(args.colors[1])))
548 index := -1
549 for i, entry := range *registry { if entry.name == words[0] { index = i; break } }
550 if index >= 0 { (*registry)[index].color = args.colors[1]
551 } else { *registry = append(*registry,Entry{words[0],args.colors[1]}) }
552 case "UNDEFINE":
553 output = append(output,words[0]+" removed")
554 for i, entry := range *registry {
555 if entry.name == words[0] {
556 *registry = slices.Delete(*registry, i, i+1)
557 break
558 }
559 }
560 case "LIST":
561 sorted := slices.Clone(*registry)
562 sort.Slice(sorted,func(i,j int) bool { return sorted[i].name < sorted[j].name })
563 for _, entry := range sorted { output = append(output,entry.name+"="+hex(entry.color)) }
564 if len(output) == 0 { output = append(output,"NONE") }
565 case "PARSE":
566 output = append(output,hex(c),fmt.Sprintf("rgb(%d,%d,%d)",c[0],c[1],c[2]))
567 for _, name := range []string{"HSL","HSV","HWB","CMYK","YCBCR","YIQ"} {
568 output = append(output,model(name,c))
569 }
570 name := "-"
571 for _, entry := range colorTable(*registry) {
572 if entry.color == c { name = entry.name; break }
573 }
574 output = append(output,name)
575 case "HSL", "HSV", "HWB", "CMYK", "YCBCR", "YIQ":
576 output = append(output,model(spec.name,c))
577 case "FROMHSL": output = append(output,hex(fromHSL(a,b,d)))
578 case "FROMHSV": output = append(output,hex(fromHSV(a,b,d)))
579 case "FROMHWB": output = append(output,hex(fromHWB(a,b,d)))
580 case "FROMCMYK":
581 result := Color{}
582 for j := range result { result[j] = rounded(255*(100-args.numbers[j])*(100-args.numbers[3]),10000) }
583 output = append(output,hex(result))
584 case "FROMYCBCR":
585 output = append(output,hex(Color{
586 clamp(rounded(1000*a+1402*(d-128),1000)),
587 clamp(rounded(1000*a-344*(b-128)-714*(d-128),1000)),
588 clamp(rounded(1000*a+1772*(b-128),1000)),
589 }))
590 case "FROMYIQ":
591 output = append(output,hex(Color{
592 clamp(rounded(1000*a+956*b+619*d,1000)),
593 clamp(rounded(1000*a-272*b-647*d,1000)),
594 clamp(rounded(1000*a-1106*b+1703*d,1000)),
595 }))
596 case "RGB565":
597 code := 2048*rounded(31*c[0],255)+32*rounded(63*c[1],255)+rounded(31*c[2],255)
598 output = append(output,fmt.Sprintf("0x%04X",code))
599 case "FROM565":
600 output = append(output,hex(Color{
601 rounded(255*(a/2048),31),rounded(255*(a/32%64),63),rounded(255*(a%32),31),
602 }))
603 case "FROMANSI": output = append(output,hex(ansiColor(int(a))))
604 case "ANSI":
605 best := 16
606 for i := 17; i <= 255; i++ {
607 if distance(c,ansiColor(i)) < distance(c,ansiColor(best)) { best = i }
608 }
609 output = append(output,fmt.Sprintf("%d %s %d",best,hex(ansiColor(best)),distance(c,ansiColor(best))))
610 case "WEBSAFE":
611 result := c
612 for j := range result { result[j] = 51*rounded(c[j],51) }
613 output = append(output,hex(result))
614 case "GRAY":
615 maximum, minimum := extrema(c)
616 value := int64(0)
617 switch args.keyword {
618 case 0: value = rounded(299*c[0]+587*c[1]+114*c[2],1000)
619 case 1: value = rounded(c[0]+c[1]+c[2],3)
620 case 2: value = rounded(maximum+minimum,2)
621 case 3: value = integerRoot((299*c[0]*c[0]+587*c[1]*c[1]+114*c[2]*c[2])/1000)
622 }
623 output = append(output,hex(Color{value,value,value}))
624 case "INVERT": output = append(output,hex(Color{255-c[0],255-c[1],255-c[2]}))
625 case "MIX":
626 result := c
627 for j := range result { result[j] = rounded(c[j]*(100-d)+args.colors[1][j]*d,100) }
628 output = append(output,hex(result))
629 case "MIXHSL":
630 first, last := hsl(c), hsl(args.colors[1])
631 delta := modulo(last[0]-first[0],360)
632 if delta > 180 { delta -= 360 }
633 output = append(output,hex(fromHSL(modulo(first[0]+away(delta*d,100),360),
634 first[1]+away((last[1]-first[1])*d,100),first[2]+away((last[2]-first[2])*d,100))))
635 case "GRADIENT":
636 for i := int64(0); i < d; i++ { output = append(output,hex(interpolate(c,args.colors[1],i,d-1))) }
637 case "TINTS", "SHADES", "TONES":
638 target := int64(128)
639 if spec.name == "TINTS" { target = 255 } else if spec.name == "SHADES" { target = 0 }
640 for i := int64(1); i <= b; i++ { output = append(output,hex(interpolate(c,Color{target,target,target},i,b+1))) }
641 case "BLEND": output = append(output,hex(blend(args.keyword,args.colors[1],args.colors[2])))
642 case "ALPHA":
643 q := args.numbers[3]
644 alpha := 100*b+q*(100-b)
645 result := Color{}
646 if alpha > 0 {
647 for j := range result { result[j] = rounded(100*b*c[j]+q*(100-b)*args.colors[2][j],alpha) }
648 }
649 output = append(output,fmt.Sprintf("%s %d%%",hex(result),rounded(alpha,100)))
650 case "SIMULATE":
651 coefficients := [4][3][3]int64{
652 {{567,433,0},{558,442,0},{0,242,758}},
653 {{625,375,0},{700,300,0},{0,300,700}},
654 {{950,50,0},{0,433,567},{0,475,525}},
655 {{299,587,114},{299,587,114},{299,587,114}},
656 }
657 result := Color{}
658 for j := range result {
659 sum := int64(0)
660 for k := 0; k < 3; k++ { sum += coefficients[args.keyword][j][k]*c[k] }
661 result[j] = rounded(sum,1000)
662 }
663 output = append(output,hex(result))
664 case "ROTATE": output = append(output,hex(rotate(c,b)))
665 case "COMPLEMENT": output = append(output,hex(rotate(c,180)))
666 case "TRIADIC":
667 for _, shift := range []int64{120,240} { output = append(output,hex(rotate(c,shift))) }
668 case "TETRADIC":
669 for _, shift := range []int64{90,180,270} { output = append(output,hex(rotate(c,shift))) }
670 case "SPLITCOMP":
671 for _, shift := range []int64{150,210} { output = append(output,hex(rotate(c,shift))) }
672 case "ANALOGOUS":
673 for i := -b; i <= b; i++ { if i != 0 { output = append(output,hex(rotate(c,i*d))) } }
674 case "LIGHTEN", "DARKEN", "SATURATE", "DESATURATE":
675 value := hsl(c)
676 switch spec.name {
677 case "LIGHTEN": value[2] = min(100,value[2]+b)
678 case "DARKEN": value[2] = max(0,value[2]-b)
679 case "SATURATE": value[1] = min(100,value[1]+b)
680 case "DESATURATE": value[1] = max(0,value[1]-b)
681 }
682 output = append(output,hex(fromHSL(value[0],value[1],value[2])))
683 case "NEAREST":
684 entries := colorTable(*registry)
685 best := 0
686 for i := 1; i < len(entries); i++ {
687 if distance(c,entries[i].color) < distance(c,entries[best].color) { best = i }
688 }
689 entry := entries[best]
690 output = append(output,fmt.Sprintf("%s %s %d",entry.name,hex(entry.color),distance(c,entry.color)))
691 case "DISTANCE":
692 other := args.colors[1]
693 r, g, blue := c[0]-other[0], c[1]-other[1], c[2]-other[2]
694 weight := (1024+c[0]+other[0])*r*r + 2048*g*g + (1534-c[0]-other[0])*blue*blue
695 output = append(output,fmt.Sprintf("%d %d",distance(c,other),integerRoot(weight/512)))
696 case "CONTRAST": output = append(output,contrastText(c,args.colors[1]))
697 case "ACCESSIBLE":
698 background, result := args.colors[1], c
699 _, verdict := contrast(c,background)
700 if verdict == "FAIL" {
701 value, found := hsl(c), false
702 for step := int64(1); step <= 100 && !found; step++ {
703 for _, sign := range []int64{-1,1} {
704 lightness := value[2]+sign*step
705 if lightness < 0 || lightness > 100 { continue }
706 candidate := fromHSL(value[0],value[1],lightness)
707 _, verdict := contrast(candidate,background)
708 if verdict != "FAIL" { result, found = candidate,true; break }
709 }
710 }
711 }
712 output = append(output,hex(result)+" "+contrastText(result,background))
713 case "BEST":
714 best := 1
715 bestRatio, _ := contrast(args.colors[best],c)
716 for i := 2; i < len(words); i++ {
717 ratio, _ := contrast(args.colors[i],c)
718 if ratio > bestRatio { best, bestRatio = i,ratio }
719 }
720 output = append(output,hex(args.colors[best])+" "+contrastText(args.colors[best],c))
721 case "DESCRIBE":
722 value := hsl(c)
723 h, s, l := value[0], value[1], value[2]
724 if l <= 5 { output = append(output,"black")
725 } else if l >= 95 { output = append(output,"white")
726 } else {
727 lightness := "very light"
728 switch {
729 case l <= 20: lightness = "very dark"
730 case l <= 40: lightness = "dark"
731 case l <= 60: lightness = "medium"
732 case l <= 80: lightness = "light"
733 }
734 if s < 10 { output = append(output,lightness+" gray")
735 } else {
736 names := []string{"red","orange","yellow","chartreuse","green","spring-green",
737 "cyan","azure","blue","violet","magenta","rose"}
738 saturation := "vivid"
739 if s < 40 { saturation = "muted" } else if s < 80 { saturation = "moderate" }
740 output = append(output,lightness+" "+saturation+" "+names[(h+15)%360/30])
741 }
742 }
743 case "PALETTE", "KMEANS":
744 colors := args.colors[1:len(words)]
745 var groups []Group
746 if spec.name == "PALETTE" { groups = palette(colors,int(a))
747 } else {
748 iterations, result := kmeans(colors,int(a))
749 groups = result
750 output = append(output,strconv.Itoa(iterations))
751 }
752 for _, group := range groups { output = append(output,fmt.Sprintf("%s:%d",hex(group.color),group.count)) }
753 case "SORT":
754 colors := slices.Clone(args.colors[:len(words)])
755 sort.SliceStable(colors,func(i,j int) bool {
756 a, b := hsl(colors[i]), hsl(colors[j])
757 return slices.Compare(a[:], b[:]) < 0
758 })
759 for _, color := range colors { output = append(output,hex(color)) }
760 case "DITHER":
761 error := Color{}
762 for _, color := range args.colors[:len(words)] {
763 target := color
764 for j := range target { target[j] = clamp(target[j]+error[j]) }
765 best := 0
766 for i := 1; i < 16; i++ {
767 if distance(target,builtins[i].color) < distance(target,builtins[best].color) { best = i }
768 }
769 output = append(output,builtins[best].name)
770 for j := range error { error[j] = target[j]-builtins[best].color[j] }
771 }
772 case "RANDOM":
773 state := a
774 for i := int64(0); i < b; i++ {
775 state = state*48271%2147483647
776 h := state%360
777 state = state*48271%2147483647
778 s := 40+state%61
779 state = state*48271%2147483647
780 l := 25+state%51
781 output = append(output,hex(fromHSL(h,s,l)))
782 }
783 }
784 return strings.Join(output," "),true
785}
