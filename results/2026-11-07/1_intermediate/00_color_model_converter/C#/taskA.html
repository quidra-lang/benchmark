33 static int Number(string text, int low, int high) {
34 Require(INTEGER.IsMatch(text));
35 int value = int.Parse(text,CultureInfo.InvariantCulture);
36 Require(value >= low && value <= high);
37 return value;
38 }
39 static int Round(long numerator, long denominator) {
40 return (int)Floor(2 * numerator + denominator, 2 * denominator);
41 }
42 static int Away(long numerator, long denominator) {
43 return numerator < 0 ? -Round(-numerator, denominator) : Round(numerator, denominator);
44 }
45 static int Percent(long a, long b) {
46 return Round(100 * a, b);
47 }
48 static int Mod(int value, int modulus) {
49 return (value % modulus + modulus) % modulus;
50 }
51 static int Clamp(int value) {
52 return Math.Clamp(value, 0, 255);
53 }
54 static int Root(long numerator, long scale) {
55 long value = numerator / scale;
56 long low = 0, high = value + 1;
57 while (low + 1 < high) {
58 long middle = (low + high) / 2;
59 if (middle * middle <= value) low = middle;
60 else high = middle;
61 }
62 return (int)low;
63 }
64 static int[] Unpack(int value) {
65 return new int[]{value >> 16, value >> 8 & 255, value & 255};
66 }
67 static string Hex(int[] color) {
68 return string.Format(CultureInfo.InvariantCulture,"#{0:X2}{1:X2}{2:X2}",color[0],color[1],color[2]);
69 }
70 List<Entry> Table() {
71 return NAMES.Select((name, i) => new Entry(name, Unpack(RGB[i]))).Concat(registered).ToList();

73 int[] Color(string text) {
74 if (HEX.IsMatch(text)) {
75 string digits = text.Substring(1);
76 if (digits.Length == 3) {
77 digits = "" + digits[0] + digits[0] + digits[1] + digits[1]
78 + digits[2] + digits[2];
79 }
80 return Unpack(int.Parse(digits,NumberStyles.HexNumber,CultureInfo.InvariantCulture));
81 }
82 Match matcher = RGB_TEXT.Match(text);
83 if (matcher.Success) {
84 int[] result = Enumerable.Range(1, 3).Select(i => int.Parse(matcher.Groups[i].Value, CultureInfo.InvariantCulture)).ToArray();
85 Require(result.All(value => value <= 255));
86 return result;
87 }
88 return Table().FirstOrDefault(entry => entry.Name == text)?.Color ?? throw new Invalid();
89 }
90 static int Hue(int[] c) {
91 int maximum = c.Max(), difference = maximum - c.Min();
92 if (difference == 0) return 0;
93 long numerator;
94 if (maximum == c[0]) {
95 numerator = 60L * (c[1] - c[2]);
96 if (numerator < 0) numerator += 360L * difference;
97 } else if (maximum == c[1]) numerator = 60L * (c[2] - c[0]) + 120L * difference;
98 else numerator = 60L * (c[0] - c[1]) + 240L * difference;
99 return Round(numerator, difference) % 360;
100 }
101 static int[] Hsl(int[] c) {
102 int maximum = c.Max(), minimum = c.Min(), difference = maximum - minimum;
103 int saturation = difference == 0 ? 0 : Percent(difference, 255 - Math.Abs(maximum + minimum - 255));
104 return new int[]{Hue(c), saturation, Percent(maximum + minimum, 510)};
105 }
106 static int[] FromHue(int h, long chroma, long offset, long denominator) {
107 int sector = h / 60, f = h % 60;
108 int x = sector % 2 == 0 ? f : 60 - f;
109 int[][] factors = {new[]{60,x,0},new[]{x,60,0},new[]{0,60,x},new[]{0,x,60},new[]{x,0,60},new[]{60,0,x}};
110 int[] result = new int[3];
111 for (int i = 0; i < 3; i++) result[i] = Round(255 * (chroma * factors[sector][i] + 60 * offset), 60 * denominator);
112 return result;
113 }
114 static int[] FromHsl(int h, int s, int l) {
115 long c = (100 - Math.Abs(2 * l - 100)) * s;
116 return FromHue(h, 2 * c, 200L * l - c, 20000);
117 }
118 static long Distance(int[] a, int[] b) {
119 long result = 0;
120 for (int i = 0; i < 3; i++) result += (long)(a[i] - b[i]) * (a[i] - b[i]);
121 return result;
122 }
123 static string Model(string command, int[] c) {
124 int r = c[0], g = c[1], b = c[2], m = c.Max(), n = c.Min(), d = m - n;
125 int[] v = Hsl(c);
126 return command switch {
127 "HSL" => "hsl(" + v[0] + "," + v[1] + "%," + v[2] + "%)",
128 "HSV" => "hsv(" + Hue(c) + "," + (m == 0 ? 0 : Percent(d,m)) + "%," + Percent(m,255) + "%)",
129 "HWB" => "hwb(" + Hue(c) + "," + Percent(n,255) + "%," + Percent(255 - m,255) + "%)",
130 "CMYK" => m == 0 ? "cmyk(0%,0%,0%,100%)" :
131 "cmyk(" + Percent(m-r,m) + "%," + Percent(m-g,m) + "%," + Percent(m-b,m) + "%," + Percent(255-m,255) + "%)",
132 "YCBCR" => "ycbcr(" + Clamp(Round(299L*r+587L*g+114L*b,1000)) + "," +
133 Clamp(Round(128000L-169L*r-331L*g+500L*b,1000)) + "," + Clamp(Round(128000L+500L*r-419L*g-81L*b,1000)) + ")",
134 "YIQ" => "yiq(" + Round(299L*r+587L*g+114L*b,1000) + "," +
135 Round(596L*r-274L*g-322L*b,1000) + "," + Round(211L*r-523L*g+312L*b,1000) + ")",
136 _ => throw new Invalid(),
137 };
138 }
139 static int[] Ansi(int number) {
140 if (number < 16) return Unpack(RGB[number]);
141 if (number >= 232) {
142 int value = 8 + 10 * (number - 232);
143 return new int[]{value, value, value};
144 }
145 int[] values = {0,95,135,175,215,255};
146 int v = number - 16;
147 return new int[]{values[v / 36], values[v / 6 % 6], values[v % 6]};
148 }
149 static int[] Gradient(int[] a, int[] b, int index, int denominator) {
150 int[] result = new int[3];
151 for (int i = 0; i < 3; i++) result[i] = a[i] + Away((long)(b[i] - a[i]) * index, denominator);
152 return result;
153 }
154 static long Luminance(int[] c) {
155 return 2126L*c[0]*c[0] + 7152L*c[1]*c[1] + 722L*c[2]*c[2];
156 }
157 static long[] Contrast(int[] a, int[] b) {
158 long first = Luminance(a), second = Luminance(b);
159 return new long[]{Math.Max(first,second) + 32512500, Math.Min(first,second) + 32512500};
160 }
161 static int Ratio(int[] a, int[] b) {
162 long[] values = Contrast(a,b);
163 return Round(100 * values[0], values[1]);
164 }
165 static bool Readable(int[] a, int[] b) {
166 long[] values = Contrast(a,b);
167 return 2 * values[0] >= 9 * values[1];
168 }
169 static string ContrastText(int[] a, int[] b) {
170 long[] values = Contrast(a,b);
171 int ratio = Round(100 * values[0], values[1]);
172 string verdict = values[0] >= 7 * values[1] ? "AAA" : 2 * values[0] >= 9 * values[1] ? "AA" : "FAIL";
173 return $"{ratio / 100}.{ratio % 100:D2} {verdict}";
174 }
175 static int BlendComponent(string mode, int x, int y) {
176 int multiply = x * y / 255, screen = 255 - (255-x)*(255-y)/255;
177 return mode switch {
178 "MULTIPLY" => multiply,
179 "SCREEN" => screen,
180 "OVERLAY" => x < 128 ? 2*x*y/255 : 255-2*(255-x)*(255-y)/255,
181 "HARDLIGHT" => y < 128 ? 2*x*y/255 : 255-2*(255-x)*(255-y)/255,
182 "SOFTLIGHT" => ((255-x)*multiply+x*screen)/255,
183 "DARKEN" => Math.Min(x,y),
184 "LIGHTEN" => Math.Max(x,y),
185 "DIFFERENCE" => Math.Abs(x-y),
186 "EXCLUSION" => x+y-2*x*y/255,
187 "ADD" => Math.Min(255,x+y),
188 "SUBTRACT" => Math.Max(0,x-y),
189 "COLORDODGE" => y == 255 ? 255 : Math.Min(255,255*x/(255-y)),
190 "COLORBURN" => y == 0 ? 0 : Math.Max(0,255-255*(255-x)/y),
191 _ => throw new Invalid(),
192 };
193 }
194 static int[] Blend(string mode, int[] baseColor, int[] top) {
195 if (new[]{"HUE","SATURATION","COLOR","LUMINOSITY"}.Contains(mode)) {
196 int[] a = Hsl(baseColor), b = Hsl(top);
197 return FromHsl(mode.Equals("HUE") || mode.Equals("COLOR") ? b[0] : a[0],
198 mode.Equals("SATURATION") || mode.Equals("COLOR") ? b[1] : a[1], mode.Equals("LUMINOSITY") ? b[2] : a[2]);
199 }
200 int[] result = new int[3];
201 for (int i = 0; i < 3; i++) result[i] = BlendComponent(mode,baseColor[i],top[i]);
202 return result;
203 }
204 static int[] Mean(List<int[]> colors) {
205 int[] result = new int[3];
206 for (int component = 0; component < 3; component++) {
207 result[component] = Round(colors.Sum(color => color[component]), colors.Count);
208 }
209 return result;
210 }
211 static int[] Ranges(List<int[]> colors) {
212 return Enumerable.Range(0, 3).Select(i => colors.Max(color => color[i]) - colors.Min(color => color[i])).ToArray();
213 }
214 static List<Cluster> Palette(int k, List<int[]> colors) {
215 List<List<int[]>> boxes = new();
216 boxes.Add(new(colors));
217 while (boxes.Count < k) {
218 int selected = Enumerable.Range(0, boxes.Count).Where(i => boxes[i].Count > 1)
219 .MaxBy(i => Ranges(boxes[i]).Max());
220 List<int[]> box = boxes[selected];
221 int[] ranges = Ranges(box);
222 int component = Array.IndexOf(ranges, ranges.Max());
223 box = box.OrderBy(c => c[component]).ToList();
224 int middle = box.Count / 2;
225 boxes[selected] = box.GetRange(0,middle);
226 boxes.Insert(selected + 1,box.GetRange(middle,box.Count-middle));
227 }
228 return boxes.Select(box => new Cluster(Mean(box), box.Count)).ToList();
229 }
230 static string ClusterText(List<Cluster> clusters) {
231 return string.Join(" ", clusters.Select(cluster => $"{Hex(cluster.Color)}:{cluster.Count}"));
232 }
233 static string Kmeans(int k, List<int[]> colors) {
234 List<int[]> representatives = Palette(k, colors).Select(cluster => cluster.Color).ToList();
235 List<Cluster> result = new();
236 int iteration = 0;
237 while (iteration < 20) {
238 iteration++;
239 var groups = Enumerable.Range(0, k).Select(_ => new List<int[]>()).ToList();
240 foreach (int[] color in colors) {
241 int best = Enumerable.Range(0, k).MinBy(j => Distance(color, representatives[j]));
242 groups[best].Add(color);
243 }
244 result = new();
245 bool unchanged = true;
246 for (int j = 0; j < k; j++) {
247 int[] next = groups[j].Count == 0 ? representatives[j] : Mean(groups[j]);
248 if (!next.SequenceEqual(representatives[j])) unchanged = false;
249 result.Add(new Cluster(next,groups[j].Count));
250 }
251 for (int j = 0; j < k; j++) representatives[j] = result[j].Color;
252 if (unchanged) break;
253 }
254 return iteration + " " + ClusterText(result);
255 }
256 List<int[]> Colors(string[] tokens, int start) {
257 return tokens.Skip(start).Select(Color).ToList();

259 static string ColorsText(List<int[]> colors) {

262 string Execute(string line) {
263 string[] t = line.Split(' ');
264 string command = t[0];
265 switch (command) {
266 case "DEFINE": {
267 Arity(t,2);
268 Require(NAME.IsMatch(t[1]) && !NAMES.Contains(t[1],StringComparer.Ordinal));
269 int[] c = Color(t[2]);
270 Entry entry = new Entry(t[1],c);
271 int index = registered.FindIndex(existing => existing.Name == t[1]);
272 if (index < 0) registered.Add(entry);
273 else registered[index] = entry;
274 return t[1] + " " + Hex(c);
275 }
276 case "UNDEFINE": {
277 Arity(t,1);
278 Require(NAME.IsMatch(t[1]));
279 int index = registered.FindIndex(entry => entry.Name == t[1]);
280 Require(index >= 0);
281 registered.RemoveAt(index);
282 return t[1] + " removed";
283 }
284 case "LIST": {
285 Arity(t,0);
286 if (registered.Count == 0) return "NONE";
287 return string.Join(" ",registered.OrderBy(e => e.Name,StringComparer.Ordinal)
288 .Select(e => e.Name + "=" + Hex(e.Color)));
289 }
290 case "PARSE": {
291 Arity(t,1);
292 int[] c = Color(t[1]);
293 string name = Table().FirstOrDefault(entry => c.SequenceEqual(entry.Color))?.Name ?? "-";
294 return Hex(c) + " rgb(" + c[0] + "," + c[1] + "," + c[2] + ") " +
295 Model("HSL",c) + " " + Model("HSV",c) + " " + Model("HWB",c) + " " +
296 Model("CMYK",c) + " " + Model("YCBCR",c) + " " + Model("YIQ",c) + " " + name;
297 }
298 case "HSL":
299 case "HSV":
300 case "HWB":
301 case "CMYK":
302 case "YCBCR":
303 case "YIQ": {
304 Arity(t,1);
305 return Model(command,Color(t[1]));
306 }
307 case "FROMHSL":
308 case "FROMHSV":
309 case "FROMHWB": {
310 Arity(t,3);
311 int h = Number(t[1],0,359), s = Number(t[2],0,100), third = Number(t[3],0,100);
312 int[] c;
313 if (command.Equals("FROMHSL")) c = FromHsl(h,s,third);
314 else if (command.Equals("FROMHSV")) c = FromHue(h,(long)third*s,100L*third-third*s,10000);
315 else if (s + third >= 100) {
316 int value = Round(255L*s,s+third);
317 c = new int[]{value,value,value};
318 } else c = FromHue(h,100-s-third,s,100);
319 return Hex(c);
320 }
321 case "FROMCMYK": {
322 Arity(t,4);
323 int k = Number(t[4],0,100);
324 int[] c = new int[3];
325 for (int i = 0; i < 3; i++) c[i] = Round(255L*(100-Number(t[i+1],0,100))*(100-k),10000);
326 return Hex(c);
327 }
328 case "FROMYCBCR":
329 case "FROMYIQ": {
330 Arity(t,3);
331 int y = Number(t[1],0,255);
332 bool iq = command.Equals("FROMYIQ");
333 int a = Number(t[2],iq ? -255 : 0,255), b = Number(t[3],iq ? -255 : 0,255);
334 long r, g, blue;
335 if (iq) {
336 r = 1000L*y+956L*a+619L*b;
337 g = 1000L*y-272L*a-647L*b;
338 blue = 1000L*y-1106L*a+1703L*b;
339 } else {
340 r = 1000L*y+1402L*(b-128);
341 g = 1000L*y-344L*(a-128)-714L*(b-128);
342 blue = 1000L*y+1772L*(a-128);
343 }
344 return Hex(new int[]{Clamp(Round(r,1000)),Clamp(Round(g,1000)),Clamp(Round(blue,1000))});
345 }
346 case "RGB565": {
347 Arity(t,1);
348 int[] c = Color(t[1]);
349 int value = 2048*Round(31L*c[0],255)+32*Round(63L*c[1],255)+Round(31L*c[2],255);
350 return $"0x{value:X4}";
351 }
352 case "FROM565": {
353 Arity(t,1);
354 Require(Regex.IsMatch(t[1],@"\A0x[0-9a-fA-F]{4}\z"));
355 int v = int.Parse(t[1].Substring(2),NumberStyles.HexNumber,CultureInfo.InvariantCulture);
356 return Hex(new int[]{Round(255L*(v/2048),31),Round(255L*(v/32%64),63),Round(255L*(v%32),31)});
357 }
358 case "FROMANSI": {
359 Arity(t,1);
360 return Hex(Ansi(Number(t[1],0,255)));
361 }
362 case "ANSI": {
363 Arity(t,1);
364 int[] c = Color(t[1]);
365 int best = Enumerable.Range(16, 240).MinBy(i => Distance(c, Ansi(i)));
366 return best + " " + Hex(Ansi(best)) + " " + Distance(c,Ansi(best));
367 }
368 case "WEBSAFE":
369 case "INVERT": {
370 Arity(t,1);
371 int[] c = Color(t[1]), result = new int[3];
372 for (int i = 0; i < 3; i++) result[i] = command.Equals("INVERT") ? 255-c[i] : 51*Round(c[i],51);
373 return Hex(result);
374 }
375 case "GRAY": {
376 Arity(t,2);
377 int[] c = Color(t[1]);
378 int value = t[2] switch {
379 "LUMA" => Round(299L*c[0]+587L*c[1]+114L*c[2],1000),
380 "AVERAGE" => Round(c.Sum(),3),
381 "LIGHTNESS" => Round(c.Max()+c.Min(),2),
382 "HSP" => Root(299L*c[0]*c[0]+587L*c[1]*c[1]+114L*c[2]*c[2],1000),
383 _ => throw new Invalid(),
384 };
385 return Hex(new int[]{value,value,value});
386 }
387 case "MIX":
388 case "MIXHSL": {
389 Arity(t,3);
390 int[] a = Color(t[1]), b = Color(t[2]);
391 int p = Number(t[3],0,100);
392 int[] result = new int[3];
393 if (command.Equals("MIX")) {
394 for (int i = 0; i < 3; i++) result[i] = Round((long)a[i]*(100-p)+(long)b[i]*p,100);
395 } else {
396 int[] first = Hsl(a), second = Hsl(b);
397 int e = Mod(second[0]-first[0],360);
398 if (e > 180) e -= 360;
399 result = FromHsl(Mod(first[0]+Away((long)e*p,100),360),
400 first[1]+Away((long)(second[1]-first[1])*p,100),first[2]+Away((long)(second[2]-first[2])*p,100));
401 }
402 return Hex(result);
403 }
404 case "GRADIENT":
405 case "TINTS":
406 case "SHADES":
407 case "TONES": {
408 Arity(t,command.Equals("GRADIENT") ? 3 : 2);
409 int[] a = Color(t[1]);
410 List<int[]> result = new();
411 if (command.Equals("GRADIENT")) {
412 int[] b = Color(t[2]);
413 int n = Number(t[3],2,16);
414 for (int i = 0; i < n; i++) result.Add(Gradient(a,b,i,n-1));
415 } else {
416 int n = Number(t[2],1,10);
417 int target = command.Equals("TINTS") ? 255 : command.Equals("SHADES") ? 0 : 128;
418 int[] b = {target,target,target};
419 for (int i = 1; i <= n; i++) result.Add(Gradient(a,b,i,n+1));
420 }
421 return ColorsText(result);
422 }
423 case "BLEND": {
424 Arity(t,3);
425 return Hex(Blend(t[1],Color(t[2]),Color(t[3])));
426 }
427 case "ALPHA": {
428 Arity(t,4);
429 int[] fg = Color(t[1]), bg = Color(t[3]);
430 int p = Number(t[2],0,100), q = Number(t[4],0,100), a = 100*p+q*(100-p);
431 int[] result = new int[3];
432 if (a > 0) {
433 for (int i = 0; i < 3; i++) result[i] = Round(100L*p*fg[i]+(long)q*(100-p)*bg[i],a);
434 }
435 return Hex(result) + " " + Round(a,100) + "%";
436 }
437 case "SIMULATE": {
438 Arity(t,2);
439 int[] c = Color(t[1]);
440 int[][] coefficients = t[2] switch {
441 "PROTANOPIA" => new int[][]{new[]{567,433,0},new[]{558,442,0},new[]{0,242,758}},
442 "DEUTERANOPIA" => new int[][]{new[]{625,375,0},new[]{700,300,0},new[]{0,300,700}},
443 "TRITANOPIA" => new int[][]{new[]{950,50,0},new[]{0,433,567},new[]{0,475,525}},
444 "ACHROMATOPSIA" => new int[][]{new[]{299,587,114},new[]{299,587,114},new[]{299,587,114}},
445 _ => throw new Invalid(),
446 };
447 int[] result = new int[3];
448 for (int i = 0; i < 3; i++) result[i] = Round((long)coefficients[i][0]*c[0]+
449 (long)coefficients[i][1]*c[1]+(long)coefficients[i][2]*c[2],1000);
450 return Hex(result);
451 }
452 case "ROTATE":
453 case "COMPLEMENT":
454 case "TRIADIC":
455 case "TETRADIC":
456 case "SPLITCOMP":
457 case "ANALOGOUS": {
458 Arity(t,command.Equals("ROTATE") ? 2 : command.Equals("ANALOGOUS") ? 3 : 1);
459 int[] values = Hsl(Color(t[1]));
460 List<int> rotations = new();
461 switch (command) {
462 case "ROTATE":
463 rotations.Add(Number(t[2],-720,720));
464 break;
465 case "COMPLEMENT":
466 rotations.Add(180);
467 break;
468 case "TRIADIC":
469 rotations.AddRange(new[]{120,240});
470 break;
471 case "TETRADIC":
472 rotations.AddRange(new[]{90,180,270});
473 break;
474 case "SPLITCOMP":
475 rotations.AddRange(new[]{150,210});
476 break;
477 default: {
478 int k = Number(t[2],1,6), step = Number(t[3],1,60);
479 for (int i = -k; i <= k; i++) {
480 if (i != 0) rotations.Add(i*step);
481 }
482 break;
483 }
484 }
485 List<int[]> result = new();
486 foreach (int d in rotations) result.Add(FromHsl(Mod(values[0]+d,360),values[1],values[2]));
487 return ColorsText(result);
488 }
489 case "LIGHTEN":
490 case "DARKEN":
491 case "SATURATE":
492 case "DESATURATE": {
493 Arity(t,2);
494 int[] values = Hsl(Color(t[1]));
495 int p = Number(t[2],0,100);
496 if (command.Equals("LIGHTEN")) values[2] = Math.Min(100,values[2]+p);
497 if (command.Equals("DARKEN")) values[2] = Math.Max(0,values[2]-p);
498 if (command.Equals("SATURATE")) values[1] = Math.Min(100,values[1]+p);
499 if (command.Equals("DESATURATE")) values[1] = Math.Max(0,values[1]-p);
500 return Hex(FromHsl(values[0],values[1],values[2]));
501 }
502 case "NEAREST": {
503 Arity(t,1);
504 int[] c = Color(t[1]);
505 Entry best = Table().MinBy(entry => Distance(c, entry.Color))!;
506 return best.Name + " " + Hex(best.Color) + " " + Distance(c,best.Color);
507 }
508 case "DISTANCE": {
509 Arity(t,2);
510 int[] a = Color(t[1]), b = Color(t[2]);
