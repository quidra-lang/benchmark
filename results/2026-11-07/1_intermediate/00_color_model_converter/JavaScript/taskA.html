10function floorBig(n, d) {
11 let quotient = n / d;
12 if (n < 0n && n % d !== 0n) quotient--;
13 return quotient;

20function roundRatio(n, d) {

78function colorArgument(token) {
79 if (/^#[0-9a-fA-F]{6}$/.test(token)) return parseHex(token);
80 if (/^#[0-9a-fA-F]{3}$/.test(token)) {
81 return [...token.slice(1)].map(digit => parseInt(digit + digit, 16));
82 }
83 const match = token.match(/^rgb\((\d{1,3}),(\d{1,3}),(\d{1,3})\)$/);
84 if (match) {
85 const color = match.slice(1).map(Number);
86 requireValid(color.every(component => component <= 255));
87 return color;
88 }
89 const color = builtinMap.get(token) ?? registered.get(token);
90 requireValid(color !== undefined);
91 return [...color];
92}

106function hue(color) {
107 const [r, g, b] = color;
108 const maximum = Math.max(...color), difference = maximum - Math.min(...color);
109 if (difference === 0) return 0;
110 let numerator;
111 if (maximum === r) {
112 numerator = 60 * (g - b);
113 if (numerator < 0) numerator += 360 * difference;
114 } else if (maximum === g) numerator = 60 * (b - r) + 120 * difference;
115 else numerator = 60 * (r - g) + 240 * difference;
116 return roundRatio(numerator, difference) % 360;
117}

119function hsl(color) {
120 const maximum = Math.max(...color), minimum = Math.min(...color);
121 const difference = maximum - minimum;
122 const saturation = difference === 0 ? 0 : roundRatio(100 * difference, 255 - Math.abs(maximum + minimum - 255));
123 return [hue(color), saturation, roundRatio(100 * (maximum + minimum), 510)];
124}

126function base60(h) {
127 const sector = floorRatio(h, 60), fraction = h % 60;
128 const x = sector % 2 === 0 ? fraction : 60 - fraction;
129 const bases = [[60, x, 0], [x, 60, 0], [0, 60, x], [0, x, 60], [x, 0, 60], [60, 0, x]];
130 return bases[sector];
131}

133function fromHsl(h, s, l) {
134 const chroma = (100 - Math.abs(2 * l - 100)) * s;
135 return base60(h).map(component => roundRatio(255 * (chroma * component + 30 * (200 * l - chroma)), 600000));
136}

138function fromHsv(h, s, v) {
139 const chroma = v * s;
140 return base60(h).map(component => roundRatio(255 * (chroma * component + 60 * (100 * v - chroma)), 600000));
141}

143function fromHwb(h, w, k) {
144 if (w + k >= 100) return Array(3).fill(roundRatio(255 * w, w + k));
145 return base60(h).map(component => roundRatio(255 * (component * (100 - w - k) + 60 * w), 6000));
146}

148function modelValues(command, color) {
149 const [r, g, b] = color;
150 const maximum = Math.max(...color), minimum = Math.min(...color);
151 switch (command) {
152 case "HSL": return hsl(color);
153 case "HSV":
154 return [hue(color), maximum === 0 ? 0 : roundRatio(100 * (maximum - minimum), maximum), roundRatio(100 * maximum, 255)];
155 case "HWB":
156 return [hue(color), roundRatio(100 * minimum, 255), roundRatio(100 * (255 - maximum), 255)];
157 case "CMYK":
158 if (maximum === 0) return [0, 0, 0, 100];
159 return [...color.map(component => roundRatio(100 * (maximum - component), maximum)), roundRatio(100 * (255 - maximum), 255)];
160 case "YCBCR":
161 return [
162 clamp(roundRatio(299 * r + 587 * g + 114 * b, 1000)),
163 clamp(roundRatio(128000 - 169 * r - 331 * g + 500 * b, 1000)),
164 clamp(roundRatio(128000 + 500 * r - 419 * g - 81 * b, 1000))
165 ];
166 case "YIQ":
167 return [
168 roundRatio(299 * r + 587 * g + 114 * b, 1000),
169 roundRatio(596 * r - 274 * g - 322 * b, 1000),
170 roundRatio(211 * r - 523 * g + 312 * b, 1000)
171 ];
172 default: throw new Invalid();
173 }
174}

185function ansiColor(code) {
186 if (code <= 15) return [...builtins[code].color];
187 if (code >= 232) return Array(3).fill(8 + 10 * (code - 232));
188 const levels = [0, 95, 135, 175, 215, 255], value = code - 16;
189 return [levels[floorRatio(value, 36)], levels[floorRatio(value, 6) % 6], levels[value % 6]];

192function rootScaled(value, scale) {
193 const target = BigInt(value), factor = BigInt(scale);
194 let low = 0n, high = 1n;
195 while (factor * high * high <= target) high *= 2n;
196 while (low + 1n < high) {
197 const middle = (low + high) / 2n;
198 if (factor * middle * middle <= target) low = middle;
199 else high = middle;
200 }
201 return Number(low);
202}

219function blendChannel(mode, x, y) {
220 switch (mode) {
221 case "MULTIPLY": return floorRatio(x * y, 255);
222 case "SCREEN": return 255 - floorRatio((255 - x) * (255 - y), 255);
223 case "OVERLAY":
224 case "HARDLIGHT":
225 if ((mode === "OVERLAY" ? x : y) < 128) return floorRatio(2 * x * y, 255);
226 return 255 - floorRatio(2 * (255 - x) * (255 - y), 255);
227 case "SOFTLIGHT": {
228 const multiply = floorRatio(x * y, 255);
229 const screen = 255 - floorRatio((255 - x) * (255 - y), 255);
230 return floorRatio((255 - x) * multiply + x * screen, 255);
231 }
232 case "DARKEN": return Math.min(x, y);
233 case "LIGHTEN": return Math.max(x, y);
234 case "DIFFERENCE": return Math.abs(x - y);
235 case "EXCLUSION": return x + y - floorRatio(2 * x * y, 255);
236 case "ADD": return Math.min(255, x + y);
237 case "SUBTRACT": return Math.max(0, x - y);
238 case "COLORDODGE": return y === 255 ? 255 : Math.min(255, floorRatio(255 * x, 255 - y));
239 case "COLORBURN": return y === 0 ? 0 : Math.max(0, 255 - floorRatio(255 * (255 - x), y));
240 default: throw new Invalid();
241 }
242}

244function blend(mode, base, top) {
245 const [h1, s1, l1] = hsl(base), [h2, s2, l2] = hsl(top);
246 switch (mode) {
247 case "HUE": return fromHsl(h2, s1, l1);
248 case "SATURATION": return fromHsl(h1, s2, l1);
249 case "COLOR": return fromHsl(h2, s2, l1);
250 case "LUMINOSITY": return fromHsl(h1, s1, l2);
251 default: return base.map((value, i) => blendChannel(mode, value, top[i]));
252 }
253}

262function contrast(a, b) {
263 const luminance = (c) => 2126 * c[0] * c[0] + 7152 * c[1] * c[1] + 722 * c[2] * c[2];
264 const first = luminance(a), second = luminance(b);
265 const high = Math.max(first, second) + 32512500;
266 const low = Math.min(first, second) + 32512500;
267 const verdict = high >= 7 * low ? "AAA" : 2 * high >= 9 * low ? "AA" : "FAIL";
268 return {ratio: roundRatio(100 * high, low), verdict};
269}
270
271function contrastText(info) {
272 return floorRatio(info.ratio, 100) + "." + String(info.ratio % 100).padStart(2, "0") + " " + info.verdict;
273}

275function accessible(foreground, background) {
276 if (contrast(foreground, background).verdict !== "FAIL") return foreground;
277 const [h, s, l] = hsl(foreground);
278 for (let step = 1; step <= 100; step++) {
279 for (const lightness of [l - step, l + step]) {
280 if (lightness < 0 || lightness > 100) continue;
281 const candidate = fromHsl(h, s, lightness);
282 if (contrast(candidate, background).verdict !== "FAIL") return candidate;
283 }
284 }
285 throw new Error("Accessible color must exist");
286}

288function describe(color) {
289 const [h, s, l] = hsl(color);
290 if (l <= 5) return "black";
291 if (l >= 95) return "white";
292 const lightness = l <= 20 ? "very dark" : l <= 40 ? "dark" : l <= 60 ? "medium" : l <= 80 ? "light" : "very light";
293 if (s < 10) return lightness + " gray";
294 const saturation = s < 40 ? "muted" : s < 80 ? "moderate" : "vivid";
295 const names = ["red", "orange", "yellow", "chartreuse", "green", "spring-green", "cyan", "azure", "blue", "violet", "magenta", "rose"];
296 return lightness + " " + saturation + " " + names[floorRatio((h + 15) % 360, 30)];
297}

303function palette(colors, count) {
304 const boxes = [colors.slice()];
305 while (boxes.length < count) {
306 let chosen = -1, chosenWidth = -1, component = 0;
307 boxes.forEach((box, index) => {
308 if (box.length < 2) return;
309 const widths = [0, 1, 2].map(c => Math.max(...box.map(color => color[c])) - Math.min(...box.map(color => color[c])));
310 const width = Math.max(...widths);
311 if (width > chosenWidth) {
312 chosen = index;
313 chosenWidth = width;
314 component = widths.indexOf(width);
315 }
316 });
317 const colorsInOrder = boxes[chosen].toSorted((a, b) => a[component] - b[component]);
318 const middle = floorRatio(colorsInOrder.length, 2);
319 boxes.splice(chosen, 1, colorsInOrder.slice(0, middle), colorsInOrder.slice(middle));
320 }
321 return boxes;
322}

324function kmeans(colors, count) {
325 let representatives = palette(colors, count).map(mean);
326 let counts = [], iterations = 0;
327 for (let iteration = 0; iteration < 20; iteration++) {
328 const groups = Array.from({length: count}, () => []);
329 for (const color of colors) {
330 let selected = 0, best = distance(color, representatives[0]);
331 for (let j = 1; j < count; j++) {
332 const value = distance(color, representatives[j]);
333 if (value < best) {
334 selected = j;
335 best = value;
336 }
337 }
338 groups[selected].push(color);
339 }
340 const next = groups.map((group, j) => group.length > 0 ? mean(group) : representatives[j]);
341 counts = groups.map(group => group.length);
342 iterations++;
343 const unchanged = next.every((color, j) => same(color, representatives[j]));
344 representatives = next;
345 if (unchanged) break;
346 }
347 return iterations + " " + representatives.map((color, j) => hex(color) + ":" + counts[j]).join(" ");
348}

350function dither(colors) {
351 let error = [0, 0, 0];
352 const names = [];
353 for (const color of colors) {
354 const adjusted = color.map((value, c) => clamp(value + error[c]));
355 let selected = 0, best = distance(adjusted, builtins[0].color);
356 for (let j = 1; j < 16; j++) {
357 const value = distance(adjusted, builtins[j].color);
358 if (value < best) {
359 selected = j;
360 best = value;
361 }
362 }
363 names.push(builtins[selected].name);
364 error = adjusted.map((value, c) => value - builtins[selected].color[c]);
365 }
366 return names.join(" ");
367}

383function execute(line) {
384 const [command, ...args] = line.split(" ");
385 const arity = arities.get(command);
386 requireValid(arity !== undefined);
387 requireValid(args.length >= arity[0] && args.length <= arity[1]);
388 if (command === "DEFINE") {
389 const name = nameArgument(args[0]);
390 requireValid(!builtinMap.has(name));
391 const color = colorArgument(args[1]);
392 registered.set(name, color);
393 return name + " " + hex(color);
394 }
395 if (command === "UNDEFINE") {
396 const name = nameArgument(args[0]);
397 requireValid(registered.has(name));
398 registered.delete(name);
399 return name + " removed";
400 }
401 if (command === "LIST") {
402 const names = [...registered.keys()].sort();
403 return names.length === 0 ? "NONE" : names.map(name => name + "=" + hex(registered.get(name))).join(" ");
404 }
405 if (command === "PARSE") {
406 const color = colorArgument(args[0]);
407 const first = table().find(entry => same(entry.color, color));
408 const models = ["HSL", "HSV", "HWB", "CMYK", "YCBCR", "YIQ"].map(model => modelText(model, color));
409 return [hex(color), "rgb(" + color.join(",") + ")", ...models, first ? first.name : "-"].join(" ");
410 }
411 if (["HSL", "HSV", "HWB", "CMYK", "YCBCR", "YIQ"].includes(command)) {
412 return modelText(command, colorArgument(args[0]));
413 }
414 if (["FROMHSL", "FROMHSV", "FROMHWB"].includes(command)) {
415 const [h, second, third] = integers(args, [[0, 359], [0, 100], [0, 100]]);
416 if (command === "FROMHSL") return hex(fromHsl(h, second, third));
417 if (command === "FROMHSV") return hex(fromHsv(h, second, third));
418 return hex(fromHwb(h, second, third));
419 }
420 if (command === "FROMCMYK") {
421 const [c, m, y, k] = integers(args, Array.from({length: 4}, () => [0, 100]));
422 return hex([c, m, y].map(value => roundRatio(255 * (100 - value) * (100 - k), 10000)));
423 }
424 if (command === "FROMYCBCR") {
425 const [y, cb, cr] = integers(args, [[0, 255], [0, 255], [0, 255]]);
426 return hex([
427 clamp(roundRatio(1000 * y + 1402 * (cr - 128), 1000)),
428 clamp(roundRatio(1000 * y - 344 * (cb - 128) - 714 * (cr - 128), 1000)),
429 clamp(roundRatio(1000 * y + 1772 * (cb - 128), 1000))
430 ]);
431 }
432 if (command === "FROMYIQ") {
433 const [y, i, q] = integers(args, [[0, 255], [-255, 255], [-255, 255]]);
434 return hex([
435 clamp(roundRatio(1000 * y + 956 * i + 619 * q, 1000)),
436 clamp(roundRatio(1000 * y - 272 * i - 647 * q, 1000)),
437 clamp(roundRatio(1000 * y - 1106 * i + 1703 * q, 1000))
438 ]);
439 }
440 if (command === "FROM565") {
441 requireValid(/^0x[0-9a-fA-F]{4}$/.test(args[0]));
442 const value = parseInt(args[0].slice(2), 16);
443 return hex([
444 roundRatio(255 * floorRatio(value, 2048), 31),
445 roundRatio(255 * (floorRatio(value, 32) % 64), 63),
446 roundRatio(255 * (value % 32), 31)
447 ]);
448 }
449 if (command === "FROMANSI") return hex(ansiColor(integer(args[0], 0, 255)));
450 if (command === "RGB565") {
451 const [r, g, b] = colorArgument(args[0]);
452 const value = 2048 * roundRatio(31 * r, 255) + 32 * roundRatio(63 * g, 255) + roundRatio(31 * b, 255);
453 return "0x" + value.toString(16).toUpperCase().padStart(4, "0");
454 }
455 if (command === "ANSI") {
456 const color = colorArgument(args[0]);
457 let selected = 16, best = distance(color, ansiColor(16));
458 for (let code = 17; code <= 255; code++) {
459 const value = distance(color, ansiColor(code));
460 if (value < best) {
461 selected = code;
462 best = value;
463 }
464 }
465 return selected + " " + hex(ansiColor(selected)) + " " + best;
466 }
467 if (command === "WEBSAFE") return hex(colorArgument(args[0]).map(value => 51 * roundRatio(value, 51)));
468 if (command === "INVERT") return hex(colorArgument(args[0]).map(value => 255 - value));
469 if (command === "GRAY") {
470 const color = colorArgument(args[0]), [r, g, b] = color;
471 const method = keyword(args[1], ["LUMA", "AVERAGE", "LIGHTNESS", "HSP"]);
472 let value;
473 if (method === "LUMA") value = roundRatio(299 * r + 587 * g + 114 * b, 1000);
474 else if (method === "AVERAGE") value = roundRatio(r + g + b, 3);
475 else if (method === "LIGHTNESS") value = roundRatio(Math.max(...color) + Math.min(...color), 2);
476 else value = rootScaled(299 * r * r + 587 * g * g + 114 * b * b, 1000);
477 return hex([value, value, value]);
478 }
479 if (command === "MIX" || command === "MIXHSL") {
480 const a = colorArgument(args[0]), b = colorArgument(args[1]), p = integer(args[2], 0, 100);
481 if (command === "MIX") return hex(a.map((value, i) => roundRatio(value * (100 - p) + b[i] * p, 100)));
482 const [h1, s1, l1] = hsl(a), [h2, s2, l2] = hsl(b);
483 let delta = mod(h2 - h1, 360);
484 if (delta > 180) delta -= 360;
485 return hex(fromHsl(
486 mod(h1 + awayRatio(delta * p, 100), 360),
487 s1 + awayRatio((s2 - s1) * p, 100),
488 l1 + awayRatio((l2 - l1) * p, 100)
489 ));
490 }
491 if (command === "GRADIENT") {
492 const a = colorArgument(args[0]), b = colorArgument(args[1]), count = integer(args[2], 2, 16);
493 return gradient(a, b, count).map(hex).join(" ");
494 }
495 if (["TINTS", "SHADES", "TONES"].includes(command)) {
496 const color = colorArgument(args[0]), count = integer(args[1], 1, 10);
497 const target = command === "TINTS" ? [255, 255, 255] : command === "SHADES" ? [0, 0, 0] : [128, 128, 128];
498 return gradient(color, target, count + 2).slice(1, -1).map(hex).join(" ");
499 }
500 if (command === "BLEND") {
501 const mode = keyword(args[0], blendModes), base = colorArgument(args[1]), top = colorArgument(args[2]);
502 return hex(blend(mode, base, top));
503 }
504 if (command === "ALPHA") {
505 const foreground = colorArgument(args[0]), p = integer(args[1], 0, 100);
506 const background = colorArgument(args[2]), q = integer(args[3], 0, 100);
507 const opacity = 100 * p + q * (100 - p);
508 const color = opacity === 0 ? [0, 0, 0] : foreground.map((value, i) => {
509 const numerator = 100 * p * value + q * (100 - p) * background[i];
510 return roundRatio(numerator, opacity);
511 });
512 return hex(color) + " " + roundRatio(opacity, 100) + "%";
513 }
514 if (command === "SIMULATE") {
515 const color = colorArgument(args[0]), type = keyword(args[1], [...simulations.keys()]);
516 const result = simulations.get(type).map(row => roundRatio(row.reduce((sum, value, i) => sum + value * color[i], 0), 1000));
517 return hex(result);
518 }
519 if (command === "ROTATE") return hex(rotate(colorArgument(args[0]), integer(args[1], -720, 720)));
520 const rotations = new Map([
521 ["COMPLEMENT", [180]], ["TRIADIC", [120, 240]], ["TETRADIC", [90, 180, 270]], ["SPLITCOMP", [150, 210]]
522 ]);
523 if (rotations.has(command)) {
524 const color = colorArgument(args[0]);
525 return rotations.get(command).map(degrees => hex(rotate(color, degrees))).join(" ");
526 }
527 if (command === "ANALOGOUS") {
528 const color = colorArgument(args[0]), k = integer(args[1], 1, 6), step = integer(args[2], 1, 60);
529 const result = [];
530 for (let i = -k; i <= k; i++) {
531 if (i !== 0) result.push(rotate(color, i * step));
532 }
533 return result.map(hex).join(" ");
534 }
535 if (["LIGHTEN", "DARKEN", "SATURATE", "DESATURATE"].includes(command)) {
536 const color = colorArgument(args[0]), p = integer(args[1], 0, 100);
537 let [h, s, l] = hsl(color);
