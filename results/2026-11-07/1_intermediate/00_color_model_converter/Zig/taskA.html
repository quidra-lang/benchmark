92fn squareRoot(n: i64) i64 {
93 var lo: i64 = 0;
94 var hi = n + 1;
95 while (lo + 1 < hi) {
96 const mid = @divTrunc(lo + hi, 2);
97 if (mid <= @divTrunc(n, mid)) {
98 lo = mid;
99 } else {
100 hi = mid;
101 }
102 }
103 return lo;
104}

118fn number(s: []const u8, lo: i64, hi: i64) Invalid!i64 {
119 const start: usize = if (s.len > 0 and s[0] == '-') 1 else 0;
120 if (s.len <= start or s.len - start > 9)
121 return error.Invalid;
122 for (s[start..]) |c| {
123 if (c < '0' or c > '9')
124 return error.Invalid;
125 }
126 const value = std.fmt.parseInt(i64, s, 10) catch return error.Invalid;
127 if (value < lo or value > hi)
128 return error.Invalid;
129 return value;
130}

146fn hue(c: Color) i64 {
147 const r = c[0];
148 const g = c[1];
149 const b = c[2];
150 const m = std.mem.max(i64, &c);
151 const d = m - std.mem.min(i64, &c);
152 if (d == 0)
153 return 0;
154 var numerator = if (m == r) 60 * (g - b) else if (m == g) 60 * (b - r) + 120 * d else 60 * (r - g) + 240 * d;
155 if (numerator < 0)
156 numerator += 360 * d;
157 return @mod(rounded(numerator, d), 360);
158}

160fn hsl(c: Color) [3]i64 {
161 const m = std.mem.max(i64, &c);
162 const n = std.mem.min(i64, &c);
163 const d = m - n;
164 const delta = if (m + n >= 255) m + n - 255 else 255 - m - n;
165 return .{ hue(c), if (d == 0) 0 else percent(d, 255 - delta), percent(m + n, 510) };
166}
167
168fn hsv(c: Color) [3]i64 {
169 const m = std.mem.max(i64, &c);
170 return .{ hue(c), if (m == 0) 0 else percent(m - std.mem.min(i64, &c), m), percent(m, 255) };
171}
172
173fn hwb(c: Color) [3]i64 {
174 return .{ hue(c), percent(std.mem.min(i64, &c), 255), percent(255 - std.mem.max(i64, &c), 255) };
175}

177fn cmyk(c: Color) [4]i64 {
178 const m = std.mem.max(i64, &c);
179 if (m == 0)
180 return .{ 0, 0, 0, 100 };
181 return .{ percent(m - c[0], m), percent(m - c[1], m), percent(m - c[2], m), percent(255 - m, 255) };
182}

184fn ycbcr(c: Color) [3]i64 {
185 const r = c[0];
186 const g = c[1];
187 const b = c[2];
188 return .{ clamp(rounded(299 * r + 587 * g + 114 * b, 1000)), clamp(rounded(128000 - 169 * r - 331 * g + 500 * b, 1000)), clamp(rounded(128000 + 500 * r - 419 * g - 81 * b, 1000)) };

191fn yiq(c: Color) [3]i64 {
192 const r = c[0];
193 const g = c[1];
194 const b = c[2];
195 return .{ rounded(299 * r + 587 * g + 114 * b, 1000), rounded(596 * r - 274 * g - 322 * b, 1000), rounded(211 * r - 523 * g + 312 * b, 1000) };
196}
197fn baseFactors(h: i64) [3]i64 {
198 const q = @divTrunc(h, 60);
199 const f = @mod(h, 60);
200 const x = if (@mod(q, 2) == 0) f else 60 - f;
201 return switch (q) {
202 0 => .{ 60, x, 0 },
203 1 => .{ x, 60, 0 },
204 2 => .{ 0, 60, x },
205 3 => .{ 0, x, 60 },
206 4 => .{ x, 0, 60 },
207 else => .{ 60, 0, x },
208 };
209}

211fn fromHsl(h: i64, s: i64, l: i64) Color {
212 const delta = if (2 * l >= 100) 2 * l - 100 else 100 - 2 * l;
213 const c = (100 - delta) * s;
214 var result: Color = undefined;
215 for (baseFactors(h), 0..) |f, j|
216 result[j] = rounded(255 * (2 * c * f + 60 * (200 * l - c)), 1200000);
217 return result;
218}
219
220fn fromHsv(h: i64, s: i64, v: i64) Color {
221 const c = v * s;
222 var result: Color = undefined;
223 for (baseFactors(h), 0..) |f, j|
224 result[j] = rounded(255 * (c * f + 60 * (100 * v - c)), 600000);
225 return result;
226}
227
228fn fromHwb(h: i64, w: i64, k: i64) Color {
229 if (w + k >= 100) {
230 const value = rounded(255 * w, w + k);
231 return .{ value, value, value };
232 }
233 var result: Color = undefined;
234 for (baseFactors(h), 0..) |f, j|
235 result[j] = rounded(255 * (f * (100 - w - k) + 60 * w), 6000);
236 return result;
237}

239fn writeModel(out: *std.Io.Writer, op: []const u8, c: Color) std.Io.Writer.Error!void {
240 if (std.mem.eql(u8, op, "CMYK")) {
241 const v = cmyk(c);
242 try out.print("cmyk({d}%,{d}%,{d}%,{d}%)", .{ v[0], v[1], v[2], v[3] });
243 return;
244 }
245 const v = if (std.mem.eql(u8, op, "HSL")) hsl(c) else if (std.mem.eql(u8, op, "HSV")) hsv(c) else if (std.mem.eql(u8, op, "HWB")) hwb(c) else if (std.mem.eql(u8, op, "YCBCR")) ycbcr(c) else yiq(c);
246 const label: []const u8 = if (std.mem.eql(u8, op, "HSL")) "hsl" else if (std.mem.eql(u8, op, "HSV")) "hsv" else if (std.mem.eql(u8, op, "HWB")) "hwb" else if (std.mem.eql(u8, op, "YCBCR")) "ycbcr" else "yiq";
247 const suffix: []const u8 = if (oneOf(op, &.{ "HSL", "HSV", "HWB" })) "%" else "";
248 try out.print("{s}({d},{d}{s},{d}{s})", .{ label, v[0], v[1], suffix, v[2], suffix });
249}

251fn ansi(n: i64) Color {
252 if (n < 16)
253 return builtins[@intCast(n)];
254 if (n >= 232) {
255 const v = 8 + 10 * (n - 232);
256 return .{ v, v, v };
257 }
258 const levels = [_]i64{ 0, 95, 135, 175, 215, 255 };
259 const i = n - 16;
260 return .{ levels[@intCast(@divTrunc(i, 36))], levels[@intCast(@mod(@divTrunc(i, 6), 6))], levels[@intCast(@mod(i, 6))] };
261}

269fn contrast(a: Color, b: Color) Contrast {
270 const y1 = luminance(a);
271 const y2 = luminance(b);
272 const high = @max(y1, y2) + 32512500;
273 const low = @min(y1, y2) + 32512500;
274 return .{ .ratio = rounded(100 * high, low), .verdict = if (high >= 7 * low) "AAA" else if (2 * high >= 9 * low) "AA" else "FAIL" };
275}

294const Box = struct {
295 colors: [32]Color = undefined,
296 len: usize = 0,
297
298 fn ranges(self: *const Box) [3]i64 {
299 var lo = self.colors[0];
300 var hi = lo;
301 for (self.colors[1..self.len]) |c| {
302 for (0..3) |j| {
303 lo[j] = @min(lo[j], c[j]);
304 hi[j] = @max(hi[j], c[j]);
305 }
306 }
307 return .{ hi[0] - lo[0], hi[1] - lo[1], hi[2] - lo[2] };
308 }
309
310 fn mean(self: *const Box) Color {
311 var sums = Color{ 0, 0, 0 };
312 for (self.colors[0..self.len]) |c| {
313 for (0..3) |j|
314 sums[j] += c[j];
315 }
316 for (0..3) |j|
317 sums[j] = rounded(sums[j], @intCast(self.len));
318 return sums;
319 }
320};

326fn palette(k: usize, colors: []const Color) [32]Box {
327 var boxes: [32]Box = undefined;
328 boxes[0] = Box{};
329 boxes[0].len = colors.len;
330 @memcpy(boxes[0].colors[0..colors.len], colors);
331 var len: usize = 1;
332 while (len < k) {
333 var selected: usize = 0;
334 var width: i64 = -1;
335 for (boxes[0..len], 0..) |box, i| {
336 if (box.len < 2)
337 continue;
338 const r = box.ranges();
339 const w = @max(r[0], r[1], r[2]);
340 if (w > width) {
341 width = w;
342 selected = i;
343 }
344 }
345 const ranges = boxes[selected].ranges();
346 var component: usize = 0;
347 for (1..3) |j| {
348 if (ranges[j] > ranges[component])
349 component = j;
350 }
351 var sorted = boxes[selected];
352 std.mem.sort(Color, sorted.colors[0..sorted.len], component, channelLess);
353 const middle = sorted.len / 2;
354 var left = Box{};
355 var right = Box{};
356 left.len = middle;
357 right.len = sorted.len - middle;
358 @memcpy(left.colors[0..middle], sorted.colors[0..middle]);
359 @memcpy(right.colors[0..right.len], sorted.colors[middle..sorted.len]);
360 std.mem.copyBackwards(Box, boxes[selected + 2 .. len + 1], boxes[selected + 1 .. len]);
361 boxes[selected] = left;
362 boxes[selected + 1] = right;
363 len += 1;
364 }
365 return boxes;
366}

368fn blendComponent(mode: []const u8, x: i64, y: i64) Invalid!i64 {
369 const multiply = @divFloor(x * y, 255);
370 const screen = 255 - @divFloor((255 - x) * (255 - y), 255);
371 if (std.mem.eql(u8, mode, "MULTIPLY"))
372 return multiply;
373 if (std.mem.eql(u8, mode, "SCREEN"))
374 return screen;
375 if (std.mem.eql(u8, mode, "OVERLAY") or std.mem.eql(u8, mode, "HARDLIGHT")) {
376 const low = if (std.mem.eql(u8, mode, "OVERLAY")) x < 128 else y < 128;
377 return if (low) @divFloor(2 * x * y, 255) else 255 - @divFloor(2 * (255 - x) * (255 - y), 255);
378 }
379 if (std.mem.eql(u8, mode, "SOFTLIGHT"))
380 return @divFloor((255 - x) * multiply + x * screen, 255);
381 if (std.mem.eql(u8, mode, "DARKEN"))
382 return @min(x, y);
383 if (std.mem.eql(u8, mode, "LIGHTEN"))
384 return @max(x, y);
385 if (std.mem.eql(u8, mode, "DIFFERENCE"))
386 return if (x >= y) x - y else y - x;
387 if (std.mem.eql(u8, mode, "EXCLUSION"))
388 return x + y - @divFloor(2 * x * y, 255);
389 if (std.mem.eql(u8, mode, "ADD"))
390 return @min(255, x + y);
391 if (std.mem.eql(u8, mode, "SUBTRACT"))
392 return @max(0, x - y);
393 if (std.mem.eql(u8, mode, "COLORDODGE"))
394 return if (y == 255) 255 else @min(255, @divFloor(255 * x, 255 - y));
395 if (std.mem.eql(u8, mode, "COLORBURN"))
396 return if (y == 0) 0 else @max(0, 255 - @divFloor(255 * (255 - x), y));
397 return error.Invalid;
398}

410const Engine = struct {
411 entries: [500]Entry = undefined,
412 len: usize = 0,
413
414 fn nameLess(self: *const Engine, a: usize, b: usize) bool {
415 return std.mem.order(u8, self.entries[a].getName(), self.entries[b].getName()) == .lt;
416 }

418 fn registeredIndex(self: *const Engine, name: []const u8) ?usize {
419 for (self.entries[0..self.len], 0..) |*entry, i| {
420 if (std.mem.eql(u8, entry.getName(), name))
421 return i;
422 }
423 return null;
424 }

426 fn color(self: *const Engine, s: []const u8) Invalid!Color {
427 if ((s.len == 4 or s.len == 7) and s[0] == '#') {
428 var result: Color = undefined;
429 if (s.len == 4) {
430 for (0..3) |j|
431 result[j] = 17 * (try hexDigit(s[j + 1]));
432 } else {
433 for (0..3) |j|
434 result[j] = 16 * (try hexDigit(s[1 + 2 * j])) + (try hexDigit(s[2 + 2 * j]));
435 }
436 return result;
437 }
438 if (s.len >= 5 and std.mem.eql(u8, s[0..4], "rgb(") and s[s.len - 1] == ')') {
439 var parts = std.mem.splitScalar(u8, s[4 .. s.len - 1], ',');
440 var result: Color = undefined;
441 var count: usize = 0;
442 while (parts.next()) |part| {
443 if (count == 3 or part.len < 1 or part.len > 3)
444 return error.Invalid;
445 for (part) |c| {
446 if (c < '0' or c > '9')
447 return error.Invalid;
448 }
449 result[count] = try number(part, 0, 255);
450 count += 1;
451 }
452 if (count != 3)
453 return error.Invalid;
454 return result;
455 }
456 for (names, builtins) |name, c| {
457 if (std.mem.eql(u8, s, name))
458 return c;
459 }
460 if (self.registeredIndex(s)) |index|
461 return self.entries[index].color;
462 return error.Invalid;
463 }

465 fn resolveColors(self: *const Engine, args: []const []const u8, buffer: *[33]Color) Invalid![]Color {
466 for (args, 0..) |arg, i|
467 buffer[i] = try self.color(arg);
468 return buffer[0..args.len];
469 }

471 fn execute(self: *Engine, words: []const []const u8, out: *std.Io.Writer) CommandError!void {
472 const op = words[0];
473 const a = words[1..];
474 if (std.mem.eql(u8, op, "DEFINE")) {
475 try requireCount(a, 2);
476 if (!validName(a[0]))
477 return error.Invalid;
478 for (names) |name| {
479 if (std.mem.eql(u8, name, a[0]))
480 return error.Invalid;
481 }
482 const value = try self.color(a[1]);
483 if (self.registeredIndex(a[0])) |index| {
484 self.entries[index].color = value;
485 } else {
486 var entry = Entry{ .color = value, .len = a[0].len };
487 @memcpy(entry.name[0..a[0].len], a[0]);
488 self.entries[self.len] = entry;
489 self.len += 1;
490 }
491 try out.print("{s} ", .{a[0]});
492 try writeColor(out, value);
493 return;
494 }
495 if (std.mem.eql(u8, op, "UNDEFINE")) {
496 try requireCount(a, 1);
497 if (!validName(a[0]))
498 return error.Invalid;
499 const index = self.registeredIndex(a[0]) orelse return error.Invalid;
500 std.mem.copyForwards(Entry, self.entries[index .. self.len - 1], self.entries[index + 1 .. self.len]);
501 self.len -= 1;
502 try out.print("{s} removed", .{a[0]});
503 return;
504 }
505 if (std.mem.eql(u8, op, "LIST")) {
506 try requireCount(a, 0);
507 if (self.len == 0) {
508 try out.writeAll("NONE");
509 return;
510 }
511 var order: [500]usize = undefined;
512 for (0..self.len) |i|
513 order[i] = i;
514 std.mem.sort(usize, order[0..self.len], @as(*const Engine, self), nameLess);
515 for (order[0..self.len], 0..) |index, i| {
516 if (i > 0)
517 try out.writeByte(' ');
518 try out.print("{s}=", .{self.entries[index].getName()});
519 try writeColor(out, self.entries[index].color);
520 }
521 return;
522 }
523 if (oneOf(op, &.{ "HSL", "HSV", "HWB", "CMYK", "YCBCR", "YIQ", "PARSE" })) {
524 try requireCount(a, 1);
525 const c = try self.color(a[0]);
526 if (!std.mem.eql(u8, op, "PARSE")) {
527 try writeModel(out, op, c);
528 return;
529 }
530 var name: []const u8 = "-";
531 for (names, builtins) |n, value| {
532 if (std.mem.eql(i64, &c, &value)) {
533 name = n;
534 break;
535 }
536 }
537 if (std.mem.eql(u8, name, "-")) {
538 for (self.entries[0..self.len]) |*entry| {
539 if (std.mem.eql(i64, &c, &entry.color)) {
540 name = entry.getName();
541 break;
542 }
543 }
544 }
545 try writeColor(out, c);
546 try out.print(" rgb({d},{d},{d})", .{ c[0], c[1], c[2] });
547 for ([_][]const u8{ "HSL", "HSV", "HWB", "CMYK", "YCBCR", "YIQ" }) |m| {
548 try out.writeByte(' ');
549 try writeModel(out, m, c);
550 }
551 try out.print(" {s}", .{name});
552 return;
553 }
554 if (oneOf(op, &.{ "FROMHSL", "FROMHSV", "FROMHWB" })) {
555 try requireCount(a, 3);
556 const h = try number(a[0], 0, 359);
557 const s = try number(a[1], 0, 100);
558 const v = try number(a[2], 0, 100);
559 try writeColor(out, if (std.mem.eql(u8, op, "FROMHSL")) fromHsl(h, s, v) else if (std.mem.eql(u8, op, "FROMHSV")) fromHsv(h, s, v) else fromHwb(h, s, v));
560 return;
561 }
562 if (std.mem.eql(u8, op, "FROMCMYK")) {
563 try requireCount(a, 4);
564 const c = try number(a[0], 0, 100);
565 const m = try number(a[1], 0, 100);
566 const y = try number(a[2], 0, 100);
567 const k = try number(a[3], 0, 100);
568 try writeColor(out, .{ rounded(255 * (100 - c) * (100 - k), 10000), rounded(255 * (100 - m) * (100 - k), 10000), rounded(255 * (100 - y) * (100 - k), 10000) });
569 return;
570 }
571 if (std.mem.eql(u8, op, "FROMYCBCR") or std.mem.eql(u8, op, "FROMYIQ")) {
572 try requireCount(a, 3);
573 const is_yiq = std.mem.eql(u8, op, "FROMYIQ");
574 const y = try number(a[0], 0, 255);
575 const u = try number(a[1], if (is_yiq) -255 else 0, 255);
576 const v = try number(a[2], if (is_yiq) -255 else 0, 255);
577 var result: Color = if (is_yiq) .{ 1000 * y + 956 * u + 619 * v, 1000 * y - 272 * u - 647 * v, 1000 * y - 1106 * u + 1703 * v } else .{ 1000 * y + 1402 * (v - 128), 1000 * y - 344 * (u - 128) - 714 * (v - 128), 1000 * y + 1772 * (u - 128) };
578 for (0..3) |j|
579 result[j] = clamp(rounded(result[j], 1000));
580 try writeColor(out, result);
581 return;
582 }
583 if (std.mem.eql(u8, op, "FROM565")) {
584 try requireCount(a, 1);
585 if (a[0].len != 6 or !std.mem.eql(u8, a[0][0..2], "0x"))
586 return error.Invalid;
587 var v: i64 = 0;
588 for (a[0][2..]) |c|
589 v = 16 * v + (try hexDigit(c));
590 try writeColor(out, .{ rounded(255 * @divTrunc(v, 2048), 31), rounded(255 * @mod(@divTrunc(v, 32), 64), 63), rounded(255 * @mod(v, 32), 31) });
591 return;
592 }
593 if (std.mem.eql(u8, op, "FROMANSI")) {

595 try writeColor(out, ansi(try number(a[0], 0, 255)));

603 return;

605 if (std.mem.eql(u8, op, "ANSI")) {
606 try requireCount(a, 1);
607 const c = try self.color(a[0]);
608 var best: i64 = 16;

611 if (distance(c, ansi(i)) < distance(c, ansi(best)))
612 best = i;
613 }
614 try out.print("{d} ", .{best});
615 try writeColor(out, ansi(best));
616 try out.print(" {d}", .{distance(c, ansi(best))});
617 return;
618 }
619 if (std.mem.eql(u8, op, "WEBSAFE") or std.mem.eql(u8, op, "INVERT")) {
620 try requireCount(a, 1);
621 const c = try self.color(a[0]);
622 var result: Color = undefined;
623 for (c, 0..) |x, j|
624 result[j] = if (std.mem.eql(u8, op, "INVERT")) 255 - x else 51 * rounded(x, 51);
625 try writeColor(out, result);

628 if (std.mem.eql(u8, op, "GRAY")) {

632 const g = c[1];
633 const b = c[2];

635 try writeColor(out, .{ value, value, value });
636 return;
637 }
638 if (std.mem.eql(u8, op, "MIX") or std.mem.eql(u8, op, "MIXHSL")) {
639 try requireCount(a, 3);
640 const c1 = try self.color(a[0]);
641 const c2 = try self.color(a[1]);
642 const p = try number(a[2], 0, 100);
643 if (std.mem.eql(u8, op, "MIX")) {
644 var result: Color = undefined;
645 for (0..3) |j|
646 result[j] = rounded(c1[j] * (100 - p) + c2[j] * p, 100);
647 try writeColor(out, result);
648 return;
649 }
650 const v1 = hsl(c1);
651 const v2 = hsl(c2);
652 var e = @mod(v2[0] - v1[0], 360);
653 if (e > 180)
654 e -= 360;
655 try writeColor(out, fromHsl(@mod(v1[0] + away(e * p, 100), 360), v1[1] + away((v2[1] - v1[1]) * p, 100), v1[2] + away((v2[2] - v1[2]) * p, 100)));
656 return;
657 }
658 if (std.mem.eql(u8, op, "GRADIENT")) {
659 try requireCount(a, 3);
660 const c1 = try self.color(a[0]);
661 const c2 = try self.color(a[1]);
662 const n = try number(a[2], 2, 16);
663 var result: [16]Color = undefined;
664 for (0..@as(usize, @intCast(n))) |i|
665 result[i] = gradientColor(c1, c2, @intCast(i), n - 1);
666 try writeColors(out, result[0..@as(usize, @intCast(n))]);

669 if (oneOf(op, &.{ "TINTS", "SHADES", "TONES" })) {
670 try requireCount(a, 2);
671 const c = try self.color(a[0]);
672 const n = try number(a[1], 1, 10);
673 const value: i64 = if (std.mem.eql(u8, op, "TINTS")) 255 else if (std.mem.eql(u8, op, "SHADES")) 0 else 128;
674 const target = Color{ value, value, value };
675 var result: [10]Color = undefined;
676 for (0..@as(usize, @intCast(n))) |i|
677 result[i] = gradientColor(c, target, @as(i64, @intCast(i)) + 1, n + 1);
678 try writeColors(out, result[0..@as(usize, @intCast(n))]);
679 return;

681 if (std.mem.eql(u8, op, "BLEND")) {
682 try requireCount(a, 3);
683 const mode = a[0];
684 const base = try self.color(a[1]);
685 const top = try self.color(a[2]);
686 if (oneOf(mode, &.{ "HUE", "SATURATION", "COLOR", "LUMINOSITY" })) {

688 const v2 = hsl(top);
689 try writeColor(out, fromHsl(if (std.mem.eql(u8, mode, "HUE") or std.mem.eql(u8, mode, "COLOR")) v2[0] else v1[0], if (std.mem.eql(u8, mode, "SATURATION") or std.mem.eql(u8, mode, "COLOR")) v2[1] else v1[1], if (std.mem.eql(u8, mode, "LUMINOSITY")) v2[2] else v1[2]));
690 return;
691 }
692 var result: Color = undefined;
693 for (0..3) |j|
694 result[j] = try blendComponent(mode, base[j], top[j]);
695 try writeColor(out, result);
696 return;
697 }
698 if (std.mem.eql(u8, op, "ALPHA")) {
699 try requireCount(a, 4);
700 const fg = try self.color(a[0]);
701 const p = try number(a[1], 0, 100);
702 const bg = try self.color(a[2]);
703 const q = try number(a[3], 0, 100);
704 const opacity = 100 * p + q * (100 - p);
705 var result = Color{ 0, 0, 0 };
706 if (opacity > 0) {
707 for (0..3) |j|
708 result[j] = rounded(100 * p * fg[j] + q * (100 - p) * bg[j], opacity);
709 }
710 try writeColor(out, result);
711 try out.print(" {d}%", .{rounded(opacity, 100)});
712 return;
713 }
714 if (std.mem.eql(u8, op, "SIMULATE")) {
715 try requireCount(a, 2);
716 const c = try self.color(a[0]);
717 const coefficients: [3][3]i64 = if (std.mem.eql(u8, a[1], "PROTANOPIA"))
718 .{ .{ 567, 433, 0 }, .{ 558, 442, 0 }, .{ 0, 242, 758 } }
719 else if (std.mem.eql(u8, a[1], "DEUTERANOPIA"))
720 .{ .{ 625, 375, 0 }, .{ 700, 300, 0 }, .{ 0, 300, 700 } }
721 else if (std.mem.eql(u8, a[1], "TRITANOPIA"))
722 .{ .{ 950, 50, 0 }, .{ 0, 433, 567 }, .{ 0, 475, 525 } }
723 else if (std.mem.eql(u8, a[1], "ACHROMATOPSIA"))
724 .{ .{ 299, 587, 114 }, .{ 299, 587, 114 }, .{ 299, 587, 114 } }
725 else
726 return error.Invalid;
727 var result: Color = undefined;
728 for (coefficients, 0..) |row, j|
729 result[j] = rounded(row[0] * c[0] + row[1] * c[1] + row[2] * c[2], 1000);
730 try writeColor(out, result);
731 return;
732 }
733 if (std.mem.eql(u8, op, "ROTATE")) {
734 try requireCount(a, 2);
735 const c = try self.color(a[0]);
736 const d = try number(a[1], -720, 720);
737 try writeColor(out, rotated(c, d));
738 return;
739 }
740 if (oneOf(op, &.{ "COMPLEMENT", "TRIADIC", "TETRADIC", "SPLITCOMP" })) {
741 try requireCount(a, 1);
742 const c = try self.color(a[0]);
743 const offsets: []const i64 = if (std.mem.eql(u8, op, "COMPLEMENT")) &.{180} else if (std.mem.eql(u8, op, "TRIADIC")) &.{ 120, 240 } else if (std.mem.eql(u8, op, "TETRADIC")) &.{ 90, 180, 270 } else &.{ 150, 210 };
744 var result: [3]Color = undefined;
745 for (offsets, 0..) |d, i|
746 result[i] = rotated(c, d);
747 try writeColors(out, result[0..offsets.len]);
748 return;
749 }
750 if (std.mem.eql(u8, op, "ANALOGOUS")) {
751 try requireCount(a, 3);
752 const c = try self.color(a[0]);
753 const k = try number(a[1], 1, 6);
754 const step = try number(a[2], 1, 60);
755 var result: [12]Color = undefined;
756 var len: usize = 0;
757 var i = -k;
758 while (i <= k) : (i += 1) {
759 if (i == 0)
760 continue;
761 result[len] = rotated(c, i * step);
762 len += 1;
763 }
764 try writeColors(out, result[0..len]);
765 return;
766 }
767 if (oneOf(op, &.{ "LIGHTEN", "DARKEN", "SATURATE", "DESATURATE" })) {
768 try requireCount(a, 2);
769 var v = hsl(try self.color(a[0]));
770 const p = try number(a[1], 0, 100);
771 const j: usize = if (std.mem.eql(u8, op, "LIGHTEN") or std.mem.eql(u8, op, "DARKEN")) 2 else 1;
772 v[j] = if (std.mem.eql(u8, op, "LIGHTEN") or std.mem.eql(u8, op, "SATURATE")) @min(100, v[j] + p) else @max(0, v[j] - p);
773 try writeColor(out, fromHsl(v[0], v[1], v[2]));
774 return;
775 }
776 if (std.mem.eql(u8, op, "NEAREST")) {
777 try requireCount(a, 1);
778 const c = try self.color(a[0]);
779 var best_name = names[0];
780 var best_color = builtins[0];
781 var d = distance(c, best_color);
782 for (names[1..], builtins[1..]) |name, value| {
783 const next = distance(c, value);
784 if (next < d) {
785 d = next;
786 best_name = name;
787 best_color = value;
788 }
789 }
790 for (self.entries[0..self.len]) |*entry| {
791 const next = distance(c, entry.color);
792 if (next < d) {
793 d = next;
794 best_name = entry.getName();
795 best_color = entry.color;
796 }
797 }
798 try out.print("{s} ", .{best_name});
799 try writeColor(out, best_color);
800 try out.print(" {d}", .{d});
801 return;
802 }
803 if (std.mem.eql(u8, op, "DISTANCE")) {
804 try requireCount(a, 2);
805 const c1 = try self.color(a[0]);
806 const c2 = try self.color(a[1]);
807 const r = c1[0] - c2[0];
808 const g = c1[1] - c2[1];
809 const b = c1[2] - c2[2];
810 const w = (1024 + c1[0] + c2[0]) * r * r + 2048 * g * g + (1534 - c1[0] - c2[0]) * b * b;
811 try out.print("{d} {d}", .{ distance(c1, c2), squareRoot(@divTrunc(w, 512)) });
812 return;
813 }
814 if (std.mem.eql(u8, op, "CONTRAST")) {
815 try requireCount(a, 2);
816 const c1 = try self.color(a[0]);
817 const c2 = try self.color(a[1]);
818 try writeContrast(out, c1, c2);

821 if (std.mem.eql(u8, op, "ACCESSIBLE")) {
822 try requireCount(a, 2);
823 const fg = try self.color(a[0]);
824 const bg = try self.color(a[1]);
825 var result = fg;
826 if (std.mem.eql(u8, contrast(fg, bg).verdict, "FAIL")) {
827 const v = hsl(fg);
828 var t: i64 = 1;
829 search: while (t <= 100) : (t += 1) {
830 for ([_]i64{ v[2] - t, v[2] + t }) |l| {
831 if (l < 0 or l > 100)
832 continue;
833 const candidate = fromHsl(v[0], v[1], l);
834 if (!std.mem.eql(u8, contrast(candidate, bg).verdict, "FAIL")) {
835 result = candidate;
836 break :search;
837 }
838 }
839 }
840 }
841 try writeColor(out, result);
842 try out.writeByte(' ');
843 try writeContrast(out, result, bg);
844 return;
845 }
846 if (std.mem.eql(u8, op, "BEST")) {
847 if (a.len < 2 or a.len > 33)
848 return error.Invalid;
849 var buffer: [33]Color = undefined;
850 const colors = try self.resolveColors(a, &buffer);
851 const bg = colors[0];
852 var best: usize = 1;
853 for (2..colors.len) |i| {
854 if (contrast(colors[i], bg).ratio > contrast(colors[best], bg).ratio)
855 best = i;
856 }
857 try writeColor(out, colors[best]);
858 try out.writeByte(' ');
859 try writeContrast(out, colors[best], bg);
860 return;
861 }
862 if (std.mem.eql(u8, op, "DESCRIBE")) {
863 try requireCount(a, 1);
864 const v = hsl(try self.color(a[0]));
865 const h = v[0];
866 const s = v[1];
867 const l = v[2];
868 if (l <= 5) {
869 try out.writeAll("black");
870 return;
871 }
872 if (l >= 95) {
873 try out.writeAll("white");
874 return;
875 }
876 const lightness: []const u8 = if (l <= 20) "very dark" else if (l <= 40) "dark" else if (l <= 60) "medium" else if (l <= 80) "light" else "very light";
877 if (s < 10) {
878 try out.print("{s} gray", .{lightness});
879 return;
880 }
881 const hues = [_][]const u8{ "red", "orange", "yellow", "chartreuse", "green", "spring-green", "cyan", "azure", "blue", "violet", "magenta", "rose" };
882 const saturation: []const u8 = if (s < 40) "muted" else if (s < 80) "moderate" else "vivid";
883 try out.print("{s} {s} {s}", .{ lightness, saturation, hues[@intCast(@divTrunc(@mod(h + 15, 360), 30))] });
884 return;
885 }
886 if (std.mem.eql(u8, op, "PALETTE") or std.mem.eql(u8, op, "KMEANS")) {
887 if (a.len < 2 or a.len > 33)
888 return error.Invalid;
889 const k: usize = @intCast(try number(a[0], 1, @intCast(a.len - 1)));
890 var buffer: [33]Color = undefined;
891 const colors = try self.resolveColors(a[1..], &buffer);
892 const boxes = palette(k, colors);
893 if (std.mem.eql(u8, op, "PALETTE")) {
894 for (boxes[0..k], 0..) |box, j| {
895 if (j > 0)
896 try out.writeByte(' ');
897 try writeColor(out, box.mean());
898 try out.print(":{d}", .{box.len});
899 }
900 return;
901 }
902 var representatives: [32]Color = undefined;
903 for (0..k) |j|
904 representatives[j] = boxes[j].mean();
905 var counts: [32]i64 = undefined;
906 var iterations: usize = 0;
907 for (0..20) |_| {
908 iterations += 1;
909 counts = @splat(0);
910 var sums: [32]Color = @splat(.{ 0, 0, 0 });
911 for (colors) |c| {
912 var best: usize = 0;
913 for (1..k) |j| {
914 if (distance(c, representatives[j]) < distance(c, representatives[best]))
915 best = j;
916 }
917 counts[best] += 1;
918 for (0..3) |j|
919 sums[best][j] += c[j];
920 }
921 var next = representatives;
922 var unchanged = true;
923 for (0..k) |j| {
924 if (counts[j] > 0) {
925 for (0..3) |channel|
926 next[j][channel] = rounded(sums[j][channel], counts[j]);
927 }
928 if (!std.mem.eql(i64, &next[j], &representatives[j]))
929 unchanged = false;
930 }
931 representatives = next;
932 if (unchanged)
933 break;
934 }
935 try out.print("{d}", .{iterations});
936 for (0..k) |j| {
937 try out.writeByte(' ');
938 try writeColor(out, representatives[j]);
939 try out.print(":{d}", .{counts[j]});
940 }
941 return;
942 }
943 if (std.mem.eql(u8, op, "SORT")) {
944 if (a.len < 1 or a.len > 32)
945 return error.Invalid;
946 var buffer: [33]Color = undefined;
947 const colors = try self.resolveColors(a, &buffer);
948 var sortable: [32]SortColor = undefined;
949 for (colors, 0..) |c, i|
950 sortable[i] = .{ .color = c, .key = hsl(c) };
951 std.mem.sort(SortColor, sortable[0..colors.len], {}, hslLess);
952 for (sortable[0..colors.len], 0..) |entry, i|
953 colors[i] = entry.color;
954 try writeColors(out, colors);
955 return;
956 }
957 if (std.mem.eql(u8, op, "DITHER")) {
958 if (a.len < 1 or a.len > 32)
959 return error.Invalid;
960 var buffer: [33]Color = undefined;
961 const colors = try self.resolveColors(a, &buffer);
962 var err = Color{ 0, 0, 0 };
963 for (colors, 0..) |c, i| {
964 var corrected: Color = undefined;
965 for (0..3) |j|
966 corrected[j] = clamp(c[j] + err[j]);
967 var best: usize = 0;
968 for (1..16) |j| {
969 if (distance(corrected, builtins[j]) < distance(corrected, builtins[best]))
970 best = j;
971 }
972 if (i > 0)
973 try out.writeByte(' ');
974 try out.writeAll(names[best]);
975 for (0..3) |j|
976 err[j] = corrected[j] - builtins[best][j];
977 }
978 return;
979 }
980 if (std.mem.eql(u8, op, "RANDOM")) {
981 try requireCount(a, 2);
982 var state = try number(a[0], 1, 999999999);
983 const n: usize = @intCast(try number(a[1], 1, 16));
984 var result: [16]Color = undefined;
985 for (0..n) |i| {
986 state = @mod(state * 48271, 2147483647);
987 const h = @mod(state, 360);
988 state = @mod(state * 48271, 2147483647);
989 const s = 40 + @mod(state, 61);
990 state = @mod(state * 48271, 2147483647);
991 const l = 25 + @mod(state, 51);
992 result[i] = fromHsl(h, s, l);
993 }
994 try writeColors(out, result[0..n]);
995 return;
996 }
997 return error.Invalid;
998 }
999};
