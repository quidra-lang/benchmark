84def hue(color):
85 red, green, blue = color
86 maximum = max(color)
87 delta = maximum - min(color)
88 if delta == 0:
89 return 0
90 if maximum == red:
91 angle = Fraction(60 * (green - blue), delta)
92 if angle < 0:
93 angle += 360
94 elif maximum == green:
95 angle = Fraction(60 * (blue - red), delta) + 120
96 else:
97 angle = Fraction(60 * (red - green), delta) + 240
98 return round_exact(angle) % 360

101def hsl(color):
102 maximum, minimum = max(color), min(color)
103 delta = maximum - minimum
104 total = maximum + minimum
105 saturation = 0 if delta == 0 else round_ratio(100 * delta, 255 - abs(total - 255))
106 return hue(color), saturation, round_ratio(100 * total, 510)

153def hue_base(angle, chroma):
154 sector, fraction = divmod(angle, 60)
155 cross = chroma * Fraction(fraction if sector % 2 == 0 else 60 - fraction, 60)
156 return (
157 (chroma, cross, 0), (cross, chroma, 0), (0, chroma, cross),
158 (0, cross, chroma), (cross, 0, chroma), (chroma, 0, cross),
159 )[sector]

186def contrast(first, second):
187 def luminance(color):
188 return sum(weight * value * value for weight, value in zip((2126, 7152, 722), color))
189
190 first_y, second_y = luminance(first), luminance(second)
191 upper = max(first_y, second_y) + 32512500
192 lower = min(first_y, second_y) + 32512500
193 verdict = "AAA" if upper >= 7 * lower else "AA" if 2 * upper >= 9 * lower else "FAIL"
194 return round_ratio(100 * upper, lower), verdict

210def palette(colors, count):
211 boxes = [colors[:]]
212 while len(boxes) < count:
213 candidates = [i for i, box in enumerate(boxes) if len(box) >= 2]
214 chosen = max(candidates, key=lambda i: max(box_widths(boxes[i])))
215 widths = box_widths(boxes[chosen])
216 channel = widths.index(max(widths))
217 ordered = sorted(boxes[chosen], key=lambda color: color[channel])
218 middle = len(ordered) // 2
219 boxes[chosen:chosen + 1] = [ordered[:middle], ordered[middle:]]
220 return boxes

223def blend_component(mode, base, top):
224 if mode == "MULTIPLY":
225 return base * top // 255
226 if mode == "SCREEN":
227 return 255 - (255 - base) * (255 - top) // 255
228 if mode in ("OVERLAY", "HARDLIGHT"):
229 condition = base if mode == "OVERLAY" else top
230 if condition < 128:
231 return 2 * base * top // 255
232 return 255 - 2 * (255 - base) * (255 - top) // 255
233 if mode == "SOFTLIGHT":
234 multiply = base * top // 255
235 screen = 255 - (255 - base) * (255 - top) // 255
236 return ((255 - base) * multiply + base * screen) // 255
237 if mode == "DARKEN":
238 return min(base, top)
239 if mode == "LIGHTEN":
240 return max(base, top)
241 if mode == "DIFFERENCE":
242 return abs(base - top)
243 if mode == "EXCLUSION":
244 return base + top - 2 * base * top // 255
245 if mode == "ADD":
246 return min(255, base + top)
247 if mode == "SUBTRACT":
248 return max(0, base - top)
249 if mode == "COLORDODGE":
250 return 255 if top == 255 else min(255, 255 * base // (255 - top))
251 return 0 if top == 0 else max(0, 255 - 255 * (255 - base) // top)

287 def color(self, text):
288 if re.fullmatch(r"#[0-9A-Fa-f]{6}", text):
289 return tuple(int(text[i:i + 2], 16) for i in (1, 3, 5))
290 if re.fullmatch(r"#[0-9A-Fa-f]{3}", text):
291 return tuple(17 * int(digit, 16) for digit in text[1:])
292 match = re.fullmatch(r"rgb\(([0-9]{1,3}),([0-9]{1,3}),([0-9]{1,3})\)", text)
293 if match:
294 color = tuple(map(int, match.groups()))
295 require(all(component <= 255 for component in color))
296 return color
297 if text in BUILTIN_NAMES:
298 return BUILTIN_NAMES[text]
299 require(text in self.registered)
300 return self.registered[text]

302 def table(self):
303 return BUILTINS + list(self.registered.items())

305 def execute(self, line):
306 words = line.split(" ")
307 try:
308 require(words[0] in self.handlers)
309 arity, handler = self.handlers[words[0]]
310 count = len(words) - 1
311 if isinstance(arity, tuple):
312 require(arity[0] <= count <= arity[1])
313 else:
314 require(count == arity)
315 return handler(words[1:])
316 except InvalidCommand:
317 return "INVALID"

319 def parse_notations(self, args):
320 color = self.color(args[0])
321 name = next((name for name, value in self.table() if value == color), "-")
322 rgb = "rgb(" + ",".join(map(str, color)) + ")"
323 models = (("hsl", hsl), ("hsv", hsv), ("hwb", hwb), ("cmyk", cmyk),
324 ("ycbcr", ycbcr), ("yiq", yiq))
325 return " ".join([hex_color(color), rgb] + [
326 model_notation(model, convert(color)) for model, convert in models
327 ] + [name])

466 def mix_hsl(self, args):
467 first, second = hsl(self.color(args[0])), hsl(self.color(args[1]))
468 percent = integer(args[2], 0, 100)
469 difference = (second[0] - first[0]) % 360
470 if difference > 180:
471 difference -= 360
472 angle = (first[0] + away(Fraction(difference * percent, 100))) % 360
473 saturation = first[1] + away(Fraction((second[1] - first[1]) * percent, 100))
474 lightness = first[2] + away(Fraction((second[2] - first[2]) * percent, 100))
475 return hex_color(from_hsl(angle, saturation, lightness))

496 def blend(self, args):
497 mode = args[0]
498 channel_modes = ("MULTIPLY", "SCREEN", "OVERLAY", "HARDLIGHT", "SOFTLIGHT", "DARKEN",
499 "LIGHTEN", "DIFFERENCE", "EXCLUSION", "ADD", "SUBTRACT", "COLORDODGE", "COLORBURN")
500 hsl_modes = ("HUE", "SATURATION", "COLOR", "LUMINOSITY")
501 require(mode in channel_modes + hsl_modes)
502 base, top = self.color(args[1]), self.color(args[2])
503 if mode in channel_modes:
504 color = tuple(blend_component(mode, x, y) for x, y in zip(base, top))
505 else:
506 h1, s1, l1 = hsl(base)
507 h2, s2, l2 = hsl(top)
508 values = {"HUE": (h2, s1, l1), "SATURATION": (h1, s2, l1),
509 "COLOR": (h2, s2, l1), "LUMINOSITY": (h1, s1, l2)}
510 color = from_hsl(*values[mode])
511 return hex_color(color)

513 def alpha(self, args):

584 def distance(self, args):
585 first, second = self.color(args[0]), self.color(args[1])
586 red_difference, green_difference, blue_difference = [a - b for a, b in zip(first, second)]
587 red_sum = first[0] + second[0]
588 weighted = ((1024 + red_sum) * red_difference ** 2 + 2048 * green_difference ** 2
589 + (1534 - red_sum) * blue_difference ** 2)
590 return f"{distance(first, second)} {isqrt(weighted // 512)}"

592 def contrast(self, args):
593 return contrast_notation(self.color(args[0]), self.color(args[1]))

595 def accessible(self, args):
596 foreground, background = self.color(args[0]), self.color(args[1])
597 if contrast(foreground, background)[1] != "FAIL":
598 result = foreground
599 else:
600 angle, saturation, lightness = hsl(foreground)
601 result = None
602 for step in range(1, 101):
603 for candidate_l in (lightness - step, lightness + step):
604 if 0 <= candidate_l <= 100:
605 candidate = from_hsl(angle, saturation, candidate_l)
606 if contrast(candidate, background)[1] != "FAIL":
607 result = candidate
608 break
609 if result is not None:
610 break
611 return f"{hex_color(result)} {contrast_notation(result, background)}"

613 def best(self, args):
614 background = self.color(args[0])
615 candidates = [self.color(text) for text in args[1:]]
616 chosen = max(candidates, key=lambda color: contrast(color, background)[0])
617 return f"{hex_color(chosen)} {contrast_notation(chosen, background)}"

619 def describe(self, args):
620 angle, saturation, lightness = hsl(self.color(args[0]))
621 if lightness <= 5:
622 return "black"
623 if lightness >= 95:
624 return "white"
625 if lightness <= 20:
626 light_word = "very dark"
627 elif lightness <= 40:
628 light_word = "dark"
629 elif lightness <= 60:
630 light_word = "medium"
631 elif lightness <= 80:
632 light_word = "light"
633 else:
634 light_word = "very light"
635 if saturation < 10:
636 return f"{light_word} gray"
637 saturation_word = "muted" if saturation < 40 else "moderate" if saturation < 80 else "vivid"
638 return f"{light_word} {saturation_word} {HUE_NAMES[((angle + 15) % 360) // 30]}"

640 def cluster_arguments(self, args):
641 colors = [self.color(text) for text in args[1:]]
642 count = integer(args[0], 1, len(colors))

645 def palette(self, args):
646 count, colors = self.cluster_arguments(args)
647 boxes = palette(colors, count)
648 return " ".join(f"{hex_color(color_mean(box))}:{len(box)}" for box in boxes)

650 def kmeans(self, args):
651 count, colors = self.cluster_arguments(args)
652 representatives = [color_mean(box) for box in palette(colors, count)]
653 for iteration in range(1, 21):
654 groups = [[] for _ in range(count)]
655 for color in colors:
656 chosen = min(range(count), key=lambda i: distance(color, representatives[i]))
657 groups[chosen].append(color)
658 updated = [color_mean(group) if group else representatives[i]
659 for i, group in enumerate(groups)]
660 unchanged = updated == representatives
661 representatives = updated
662 if unchanged:
663 break
664 entries = [f"{hex_color(color)}:{len(group)}" for color, group in zip(representatives, groups)]
665 return " ".join([str(iteration)] + entries)

671 def dither(self, args):
672 colors = [self.color(text) for text in args]
673 error = (0, 0, 0)
674 names = []
675 for color in colors:
676 target = tuple(clamp(x + e) for x, e in zip(color, error))
677 number = min(range(16), key=lambda i: distance(target, BUILTINS[i][1]))
678 name, quantized = BUILTINS[number]
679 names.append(name)
680 error = tuple(x - y for x, y in zip(target, quantized))
681 return " ".join(names)
