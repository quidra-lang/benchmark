36int squareRoot(int n) {
37 var lo = 0, hi = n + 1;
38 while (lo + 1 < hi) {
39 final mid = (lo + hi) ~/ 2;
40 if (mid <= n ~/ mid) {
41 lo = mid;
42 } else {
43 hi = mid;
44 }
45 }
46 return lo;
47}

64int hue(Color c) {
65 final r = c[0], g = c[1], b = c[2];
66 final m = c.reduce(math.max), n = c.reduce(math.min), d = m - n;
67 if (d == 0)
68 return 0;
69 var numerator = m == r ? 60 * (g - b) : m == g ? 60 * (b - r) + 120 * d : 60 * (r - g) + 240 * d;
70 if (numerator < 0)
71 numerator += 360 * d;
72 return rounded(numerator, d) % 360;
73}
74List<int> baseFactors(int h) {
75 final q = h ~/ 60, f = h % 60, x = q.isEven ? f : 60 - f;
76 return switch (q) {
77 0 => [60, x, 0],
78 1 => [x, 60, 0],
79 2 => [0, 60, x],
80 3 => [0, x, 60],
81 4 => [x, 0, 60],
82 _ => [60, 0, x],
83 };
84}
85Color fromHsl(int h, int s, int l) {
86 final c = (100 - (2 * l - 100).abs()) * s;
87 return [for (final f in baseFactors(h)) rounded(255 * (2 * c * f + 60 * (200 * l - c)), 1200000)];
88}
89Color fromHsv(int h, int s, int v) {
90 final c = v * s;
91 return [for (final f in baseFactors(h)) rounded(255 * (c * f + 60 * (100 * v - c)), 600000)];
92}
93Color fromHwb(int h, int w, int k) {
94 if (w + k >= 100)
95 return List.filled(3, rounded(255 * w, w + k));
96 return [for (final f in baseFactors(h)) rounded(255 * (f * (100 - w - k) + 60 * w), 6000)];
97}
98List<int> hsv(Color c) {

142 return (rounded(100 * high, low), high >= 7 * low ? 'AAA' : 2 * high >= 9 * low ? 'AA' : 'FAIL');

162List<Box> palette(int k, List<Color> colors) {
163 final boxes = [Box(colors.toList())];
164 while (boxes.length < k) {
165 var selected = -1, width = -1;
166 for (var i = 0; i < boxes.length; i++) {
167 if (boxes[i].colors.length < 2)
168 continue;
169 final w = boxes[i].ranges().reduce(math.max);
170 if (w > width) {
171 selected = i;
172 width = w;
173 }
174 }
175 final box = boxes[selected], ranges = box.ranges();
176 final component = ranges.indexOf(ranges.reduce(math.max));
177 final indexed = box.colors.indexed.toList()..sort((a, b) {
178 final order = a.$2[component].compareTo(b.$2[component]);
179 return order != 0 ? order : a.$1.compareTo(b.$1);
180 });
181 final sorted = indexed.map((entry) => entry.$2).toList();
182 final middle = sorted.length ~/ 2;
183 boxes.replaceRange(selected, selected + 1, [Box(sorted.sublist(0, middle)), Box(sorted.sublist(middle))]);
184 }
185 return boxes;
186}

188int blendComponent(String mode, int x, int y) {
189 final multiply = x * y ~/ 255, screen = 255 - (255 - x) * (255 - y) ~/ 255;
190 return switch (mode) {
191 'MULTIPLY' => multiply,
192 'SCREEN' => screen,
193 'OVERLAY' => x < 128 ? 2 * x * y ~/ 255 : 255 - 2 * (255 - x) * (255 - y) ~/ 255,
194 'HARDLIGHT' => y < 128 ? 2 * x * y ~/ 255 : 255 - 2 * (255 - x) * (255 - y) ~/ 255,
195 'SOFTLIGHT' => ((255 - x) * multiply + x * screen) ~/ 255,
196 'DARKEN' => math.min(x, y),
197 'LIGHTEN' => math.max(x, y),
198 'DIFFERENCE' => (x - y).abs(),
199 'EXCLUSION' => x + y - 2 * x * y ~/ 255,
200 'ADD' => math.min(255, x + y),
201 'SUBTRACT' => math.max(0, x - y),
202 'COLORDODGE' => y == 255 ? 255 : math.min(255, 255 * x ~/ (255 - y)),
203 'COLORBURN' => y == 0 ? 0 : math.max(0, 255 - 255 * (255 - x) ~/ y),
204 _ => throw Invalid(),
205 };
206}
