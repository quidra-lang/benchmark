60require_valid(condition) = condition || throw(ArgumentError("INVALID"))
61round_ratio(n,d) = fld(2*n+d,2*d)
62away_ratio(n,d) = n < 0 ? -round_ratio(-n,d) : round_ratio(n,d)
63byte_clamp(x) = clamp(x,0,255)
64hex_color(c) = @sprintf("#%02X%02X%02X",c[1],c[2],c[3])
65color_sequence(colors) = join(hex_color.(colors)," ")
66color_distance(a,b) = sum((a[i]-b[i])^2 for i in 1:3)

68function integer_arg(text,low,high)
69 require_valid(occursin(r"^-?[0-9]{1,9}$",text))
70 value = parse(Int,text)
71 require_valid(low <= value <= high)
72 return value
73end
74
75function parse_color(text,registered)
76 if occursin(r"^#[0-9A-Fa-f]{6}$",text)
77 return [parse(Int,text[i:i+1];base=16) for i in (2,4,6)]
78 elseif occursin(r"^#[0-9A-Fa-f]{3}$",text)
79 return [17*parse(Int,string(text[i]);base=16) for i in 2:4]
80 end
81 matched = match(r"^rgb\(([0-9]{1,3}),([0-9]{1,3}),([0-9]{1,3})\)$",text)
82 if matched !== nothing
83 color = parse.(Int,matched.captures)
84 require_valid(all(x <= 255 for x in color))
85 return color
86 end
87 index = findfirst(==(text),BUILTIN_NAMES)
88 index !== nothing && return copy(BUILTIN_COLORS[index])
89 require_valid(haskey(registered,text))
90 return copy(registered[text])
91end

93function typed_arg(text,kind,registered)
94 kind == "c" && return parse_color(text,registered)
95 if kind == "name"
96 require_valid(occursin(r"^[a-z]{1,12}$",text))
97 return String(text)
98 elseif kind == "code"
99 require_valid(occursin(r"^0x[0-9a-fA-F]{4}$",text))
100 return parse(Int,text[3:end];base=16)
101 elseif haskey(KEYWORDS,kind)
102 require_valid(text in KEYWORDS[kind])
103 return String(text)
104 end
105 low,high = parse.(Int,split(kind,':'))
106 return integer_arg(text,low,high)
107end

109function hue(c)
110 r,g,b = c
111 high,low = maximum(c),minimum(c)
112 delta = high-low
113 delta == 0 && return 0
114 if high == r
115 numerator = 60*(g-b)
116 numerator < 0 && (numerator += 360*delta)
117 elseif high == g
118 numerator = 60*(b-r)+120*delta
119 else
120 numerator = 60*(r-g)+240*delta
121 end
122 return mod(round_ratio(numerator,delta),360)
123end

125function hsl(c)
126 high,low = maximum(c),minimum(c)
127 delta = high-low
128 saturation = delta == 0 ? 0 : round_ratio(100*delta,255-abs(high+low-255))
129 return [hue(c),saturation,round_ratio(100*(high+low),510)]
130end

132function base60(h)
133 sector,fraction = divrem(h,60)
134 x = iseven(sector) ? fraction : 60-fraction
135 return [[60,x,0],[x,60,0],[0,60,x],[0,x,60],[x,0,60],[60,0,x]][sector+1]
136end

138function from_hsl(h,s,l)
139 chroma = (100-abs(2*l-100))*s
140 offset = 6000*l-30*chroma
141 return [round_ratio(255*(u*chroma+offset),600000) for u in base60(h)]
142end

144function model_values(model,c)
145 r,g,b = c
146 high,low = maximum(c),minimum(c)
147 if model == "HSL"
148 return hsl(c)
149 elseif model == "HSV"
150 return [hue(c),high == 0 ? 0 : round_ratio(100*(high-low),high),round_ratio(100*high,255)]
151 elseif model == "HWB"
152 return [hue(c),round_ratio(100*low,255),round_ratio(100*(255-high),255)]
153 elseif model == "CMYK"
154 high == 0 && return [0,0,0,100]
155 return vcat([round_ratio(100*(high-x),high) for x in c],round_ratio(100*(255-high),255))
156 elseif model == "YCBCR"
157 return byte_clamp.([
158 round_ratio(299*r+587*g+114*b,1000),
159 round_ratio(128000-169*r-331*g+500*b,1000),
160 round_ratio(128000+500*r-419*g-81*b,1000)])
161 end
162 return [
163 round_ratio(299*r+587*g+114*b,1000),
164 round_ratio(596*r-274*g-322*b,1000),
165 round_ratio(211*r-523*g+312*b,1000)]
166end

168function model_text(model,c)
169 values = model_values(model,c)
170 if model in ("HSL","HSV","HWB")
171 return string(lowercase(model),'(',values[1],',',values[2],"%,",values[3],"%)")
172 elseif model == "CMYK"
173 return string("cmyk(",join([string(value,'%') for value in values],","),')')
174 end
175 return string(lowercase(model),'(',join(values,","),')')
176end

178function ansi_color(number)
179 number < 16 && return copy(BUILTIN_COLORS[number+1])
180 if number >= 232
181 return fill(8+10*(number-232),3)
182 end
183 value = number-16
184 levels = [0,95,135,175,215,255]
185 return levels[[div(value,36)+1,div(value,6)%6+1,value%6+1]]
186end

188root_ratio(n,d) = isqrt(fld(n,d))
189nearest_index(color,colors) = argmin(i -> color_distance(color,colors[i]),eachindex(colors))

191function contrast(a,b)
192 luminance(c) = 2126*c[1]^2+7152*c[2]^2+722*c[3]^2
193 y1,y2 = luminance(a),luminance(b)
194 high,low = max(y1,y2)+32512500,min(y1,y2)+32512500
195 if high >= 7*low
196 verdict = "AAA"
197 elseif 2*high >= 9*low
198 verdict = "AA"
199 else
200 verdict = "FAIL"
201 end
202 return round_ratio(100*high,low),verdict
203end

205function contrast_text(a,b)
206 ratio,verdict = contrast(a,b)
207 return @sprintf("%d.%02d %s",div(ratio,100),ratio%100,verdict)
208end

210gradient(a,b,n) = [[a[j]+away_ratio((b[j]-a[j])*i,n-1) for j in 1:3] for i in 0:n-1]
211mean_color(colors) = [round_ratio(sum(c[j] for c in colors),length(colors)) for j in 1:3]
212widths(colors) = [maximum(c[j] for c in colors)-minimum(c[j] for c in colors) for j in 1:3]

214function palette(colors,k)
215 boxes = [copy(colors)]
216 while length(boxes) < k
217 candidates = findall(box -> length(box) >= 2,boxes)
218 selected = argmax(i -> maximum(widths(boxes[i])),candidates)
219 box = boxes[selected]
220 axis = argmax(widths(box))
221 indices = sortperm(1:length(box);by=i -> (box[i][axis],i))
222 ordered = box[indices]
223 middle = div(length(box),2)
224 splice!(boxes,selected:selected,[ordered[1:middle],ordered[middle+1:end]])
225 end
226 return boxes
227end

229function blend_component(mode,x,y)
230 multiply = div(x*y,255)
231 screen = 255-div((255-x)*(255-y),255)
232 if mode == "MULTIPLY"
233 return multiply
234 elseif mode == "SCREEN"
235 return screen
236 elseif mode == "OVERLAY" || mode == "HARDLIGHT"
237 lower = mode == "OVERLAY" ? x < 128 : y < 128
238 return lower ? div(2*x*y,255) : 255-div(2*(255-x)*(255-y),255)
239 elseif mode == "SOFTLIGHT"
240 return div((255-x)*multiply+x*screen,255)
241 elseif mode == "DARKEN"
242 return min(x,y)
243 elseif mode == "LIGHTEN"
244 return max(x,y)
245 elseif mode == "DIFFERENCE"
246 return abs(x-y)
247 elseif mode == "EXCLUSION"
248 return x+y-div(2*x*y,255)
249 elseif mode == "ADD"
250 return min(255,x+y)
251 elseif mode == "SUBTRACT"
252 return max(0,x-y)
253 elseif mode == "COLORDODGE"
254 return y == 255 ? 255 : min(255,div(255*x,255-y))
255 end
256 return y == 0 ? 0 : max(0,255-div(255*(255-x),y))
257end

259function execute(command,raw,registered,registration_order)
260 if command in ("PALETTE","KMEANS","BEST","SORT","DITHER")
261 if command in ("PALETTE","KMEANS")
262 require_valid(2 <= length(raw) <= 33)
263 k = integer_arg(raw[1],1,length(raw)-1)
264 args = Any[k,[parse_color(token,registered) for token in raw[2:end]]]
265 else
266 require_valid((command == "BEST" ? 2 : 1) <= length(raw) <= (command == "BEST" ? 33 : 32))
267 args = Any[[parse_color(token,registered) for token in raw]]
268 end
269 else
270 require_valid(haskey(SIGNATURES,command))
271 kinds = split(SIGNATURES[command])
272 require_valid(length(raw) == length(kinds))
273 args = Any[typed_arg(raw[i],kinds[i],registered) for i in eachindex(raw)]
274 end
275 if command == "DEFINE"
276 name,color = args
277 require_valid(!(name in BUILTIN_NAMES))
278 if !haskey(registered,name)
279 push!(registration_order,name)
280 end
281 registered[name] = copy(color)
282 return name*" "*hex_color(color)
283 elseif command == "UNDEFINE"
284 name = args[1]
285 require_valid(haskey(registered,name))
286 delete!(registered,name)
287 filter!(!=(name),registration_order)
288 return name*" removed"
289 elseif command == "LIST"
290 return isempty(registered) ? "NONE" : join([name*"="*hex_color(registered[name]) for name in sort(collect(keys(registered)))]," ")
291 elseif command == "PARSE" || command == "NEAREST"
292 color = args[1]
293 names = vcat(String.(BUILTIN_NAMES),registration_order)
294 colors = vcat(BUILTIN_COLORS,[registered[name] for name in registration_order])
295 if command == "NEAREST"
296 i = nearest_index(color,colors)
297 return string(names[i],' ',hex_color(colors[i]),' ',color_distance(color,colors[i]))
298 end
299 i = findfirst(==(color),colors)
300 parts = [hex_color(color),"rgb("*join(color,",")*")"]
301 append!(parts,[model_text(model,color) for model in ("HSL","HSV","HWB","CMYK","YCBCR","YIQ")])
302 push!(parts,i === nothing ? "-" : names[i])
303 return join(parts," ")
304 elseif command in ("HSL","HSV","HWB","CMYK","YCBCR","YIQ")
305 return model_text(command,args[1])
306 elseif command == "FROMHSL"
307 result = from_hsl(args...)
308 elseif command == "FROMHSV"
309 h,s,v = args
310 chroma = v*s
311 result = [round_ratio(255*(u*chroma+6000*v-60*chroma),600000) for u in base60(h)]
312 elseif command == "FROMHWB"
313 h,w,k = args
314 if w+k >= 100
315 result = fill(round_ratio(255*w,w+k),3)
316 else
317 result = [round_ratio(255*(u*(100-w-k)+60*w),6000) for u in base60(h)]
318 end
319 elseif command == "FROMCMYK"
320 result = [round_ratio(255*(100-args[i])*(100-args[4]),10000) for i in 1:3]
321 elseif command == "FROMYCBCR"
322 y,cb,cr = args
323 result = byte_clamp.([
324 round_ratio(1000*y+1402*(cr-128),1000),
325 round_ratio(1000*y-344*(cb-128)-714*(cr-128),1000),
326 round_ratio(1000*y+1772*(cb-128),1000)])
327 elseif command == "FROMYIQ"
328 y,i,q = args
329 result = byte_clamp.([
330 round_ratio(1000*y+956*i+619*q,1000),
331 round_ratio(1000*y-272*i-647*q,1000),
332 round_ratio(1000*y-1106*i+1703*q,1000)])
333 elseif command == "RGB565"
334 r,g,b = args[1]
335 value = 2048*round_ratio(31*r,255)+32*round_ratio(63*g,255)+round_ratio(31*b,255)
336 return @sprintf("0x%04X",value)
337 elseif command == "FROM565"
338 value = args[1]
339 result = [round_ratio(255*div(value,2048),31),round_ratio(255*(div(value,32)%64),63),round_ratio(255*(value%32),31)]
340 elseif command == "ANSI"
341 colors = ansi_color.(16:255)
342 i = nearest_index(args[1],colors)
343 return string(i+15,' ',hex_color(colors[i]),' ',color_distance(args[1],colors[i]))
344 elseif command == "FROMANSI"
345 result = ansi_color(args[1])
346 elseif command == "WEBSAFE"
347 result = [51*round_ratio(x,51) for x in args[1]]
348 elseif command == "GRAY"
349 color,method = args
350 r,g,b = color
351 if method == "LUMA"
352 value = round_ratio(299*r+587*g+114*b,1000)
353 elseif method == "AVERAGE"
354 value = round_ratio(r+g+b,3)
355 elseif method == "LIGHTNESS"
356 value = round_ratio(maximum(color)+minimum(color),2)
357 else
358 value = root_ratio(299*r*r+587*g*g+114*b*b,1000)
359 end
360 result = fill(value,3)
361 elseif command == "INVERT"
362 result = 255 .- args[1]
363 elseif command == "MIX"
364 a,b,p = args
365 result = [round_ratio(a[i]*(100-p)+b[i]*p,100) for i in 1:3]
366 elseif command == "MIXHSL"
367 a,b,p = args
368 h1,s1,l1 = hsl(a)
369 h2,s2,l2 = hsl(b)
370 delta = mod(h2-h1,360)
371 delta > 180 && (delta -= 360)
372 result = from_hsl(
373 mod(h1+away_ratio(delta*p,100),360),
374 s1+away_ratio((s2-s1)*p,100),
375 l1+away_ratio((l2-l1)*p,100))
376 elseif command == "GRADIENT"
377 return color_sequence(gradient(args...))
378 elseif command in ("TINTS","SHADES","TONES")
379 color,n = args
380 target_value = Dict("TINTS"=>255,"SHADES"=>0,"TONES"=>128)[command]
381 target = fill(target_value,3)
382 return color_sequence(gradient(color,target,n+2)[2:end-1])
383 elseif command == "BLEND"
384 mode,a,b = args
385 if mode in ("HUE","SATURATION","COLOR","LUMINOSITY")
386 h1,s1,l1 = hsl(a)
387 h2,s2,l2 = hsl(b)
388 if mode == "HUE"
389 result = from_hsl(h2,s1,l1)
390 elseif mode == "SATURATION"
391 result = from_hsl(h1,s2,l1)
392 elseif mode == "COLOR"
393 result = from_hsl(h2,s2,l1)
394 else
395 result = from_hsl(h1,s1,l2)
396 end
397 else
398 result = [blend_component(mode,a[i],b[i]) for i in 1:3]
399 end
400 elseif command == "ALPHA"
401 a,p,b,q = args
402 opacity = 100*p+q*(100-p)
403 result = opacity == 0 ? [0,0,0] : [round_ratio(100*p*a[i]+q*(100-p)*b[i],opacity) for i in 1:3]
404 return string(hex_color(result),' ',round_ratio(opacity,100),'%')
405 elseif command == "SIMULATE"
406 color,kind = args
407 matrices = Dict(
408 "PROTANOPIA"=>[[567,433,0],[558,442,0],[0,242,758]],
409 "DEUTERANOPIA"=>[[625,375,0],[700,300,0],[0,300,700]],
410 "TRITANOPIA"=>[[950,50,0],[0,433,567],[0,475,525]],
411 "ACHROMATOPSIA"=>fill([299,587,114],3))
412 result = [round_ratio(sum(row[i]*color[i] for i in 1:3),1000) for row in matrices[kind]]
413 elseif command in ("ROTATE","COMPLEMENT","TRIADIC","TETRADIC","SPLITCOMP","ANALOGOUS")
414 h,s,l = hsl(args[1])
415 if command == "ROTATE"
416 offsets = [args[2]]
417 elseif command == "ANALOGOUS"
418 offsets = [i*args[3] for i in -args[2]:args[2] if i != 0]
419 else
420 offsets = Dict("COMPLEMENT"=>[180],"TRIADIC"=>[120,240],"TETRADIC"=>[90,180,270],"SPLITCOMP"=>[150,210])[command]
421 end
422 return color_sequence([from_hsl(mod(h+d,360),s,l) for d in offsets])
423 elseif command in ("LIGHTEN","DARKEN","SATURATE","DESATURATE")
424 h,s,l = hsl(args[1])
425 p = args[2]
426 if command == "LIGHTEN"
427 l = min(100,l+p)
428 elseif command == "DARKEN"
429 l = max(0,l-p)
430 elseif command == "SATURATE"
431 s = min(100,s+p)
432 else
433 s = max(0,s-p)
434 end
435 result = from_hsl(h,s,l)
436 elseif command == "DISTANCE"
437 a,b = args
438 r1,g1,b1 = a
439 r2,g2,b2 = b
440 weighted = (1024+r1+r2)*(r1-r2)^2+2048*(g1-g2)^2+(1534-r1-r2)*(b1-b2)^2
441 return string(color_distance(a,b),' ',root_ratio(weighted,512))
442 elseif command == "CONTRAST"
443 return contrast_text(args...)
444 elseif command == "ACCESSIBLE"
445 result,bg = args
446 if contrast(result,bg)[2] == "FAIL"
447 h,s,l = hsl(result)
448 found = false
449 for step in 1:100
450 for lightness in (l-step,l+step)
451 if 0 <= lightness <= 100
452 candidate = from_hsl(h,s,lightness)
453 if contrast(candidate,bg)[2] != "FAIL"
454 result = candidate
455 found = true
456 break
457 end
458 end
459 end
460 found && break
461 end
462 end
463 return hex_color(result)*" "*contrast_text(result,bg)
464 elseif command == "BEST"
465 colors = args[1]
466 bg = colors[1]
467 selected = argmax(i -> contrast(colors[i],bg)[1],2:length(colors))
468 return hex_color(colors[selected])*" "*contrast_text(colors[selected],bg)
469 elseif command == "DESCRIBE"
470 h,s,l = hsl(args[1])
471 l <= 5 && return "black"
472 l >= 95 && return "white"
473 if l <= 20
474 lightness = "very dark"
475 elseif l <= 40
476 lightness = "dark"
477 elseif l <= 60
478 lightness = "medium"
479 elseif l <= 80
480 lightness = "light"
481 else
482 lightness = "very light"
483 end
484 s < 10 && return lightness*" gray"
485 if s < 40
486 saturation = "muted"
487 elseif s < 80
488 saturation = "moderate"
489 else
490 saturation = "vivid"
491 end
492 names = split("red orange yellow chartreuse green spring-green cyan azure blue violet magenta rose")
493 return join([lightness,saturation,names[div(mod(h+15,360),30)+1]]," ")
494 elseif command in ("PALETTE","KMEANS")
495 k,colors = args
496 boxes = palette(colors,k)
497 representatives = mean_color.(boxes)
498 counts = length.(boxes)
499 iterations = 0
500 if command == "KMEANS"
501 for iteration in 1:20
502 assigned = [Vector{Int}[] for j in 1:k]
503 for color in colors
504 push!(assigned[nearest_index(color,representatives)],color)
505 end
506 updated = [isempty(assigned[j]) ? representatives[j] : mean_color(assigned[j]) for j in 1:k]
507 counts = length.(assigned)
508 iterations = iteration
509 unchanged = updated == representatives
510 representatives = updated
511 unchanged && break
512 end
513 end
514 output = join([string(hex_color(representatives[j]),':',counts[j]) for j in 1:k]," ")
515 return command == "PALETTE" ? output : string(iterations,' ',output)
516 elseif command == "SORT"
517 colors = args[1]
518 order = sortperm(1:length(colors);by=i -> (Tuple(hsl(colors[i])),i))
519 return color_sequence(colors[order])
520 elseif command == "DITHER"
521 error = [0,0,0]
522 output = String[]
523 for color in args[1]
524 adjusted = byte_clamp.(color.+error)
525 selected = nearest_index(adjusted,BUILTIN_COLORS[1:16])
526 push!(output,BUILTIN_NAMES[selected])
527 error = adjusted.-BUILTIN_COLORS[selected]
528 end
529 return join(output," ")
530 elseif command == "RANDOM"
531 state,n = args
532 output = Vector{Int}[]
533 for i in 1:n
534 state = (48271*state)%2147483647
535 u = state
536 state = (48271*state)%2147483647
537 v = state
538 state = (48271*state)%2147483647
539 push!(output,from_hsl(u%360,40+v%61,25+state%51))
540 end
541 return color_sequence(output)
542 end
543 return hex_color(result)
544end

547 count = parse(Int,readline())
548 registered = Dict{String,Vector{Int}}()
549 registration_order = String[]
550 for i in 1:count
551 parts = split(readline(),' ')
552 try
553 println(execute(String(parts[1]),parts[2:end],registered,registration_order))
554 catch exception
555 exception isa ArgumentError || rethrow()
556 println("INVALID")
557 end
558 end
