51 static int root(long numerator, long scale) {
52 long value = numerator / scale;
53 long low = 0, high = value + 1;
54 while (low + 1 < high) {
55 long middle = (low + high) / 2;
56 if (middle * middle <= value) low = middle;
57 else high = middle;
58 }
59 return (int)low;
60 }

94 static int hue(int[] c) {
95 var statistics = Arrays.stream(c).summaryStatistics();
96 int maximum = statistics.getMax(), difference = maximum - statistics.getMin();
97 if (difference == 0) return 0;
98 long numerator;
99 if (maximum == c[0]) {
100 numerator = 60L * (c[1] - c[2]);
101 if (numerator < 0) numerator += 360L * difference;
102 } else if (maximum == c[1]) numerator = 60L * (c[2] - c[0]) + 120L * difference;
103 else numerator = 60L * (c[0] - c[1]) + 240L * difference;
104 return round(numerator, difference) % 360;
105 }
106 static int[] hsl(int[] c) {
107 var statistics = Arrays.stream(c).summaryStatistics();
108 int maximum = statistics.getMax(), minimum = statistics.getMin(), difference = maximum - minimum;
109 int saturation = difference == 0 ? 0 : percent(difference, 255 - Math.abs(maximum + minimum - 255));
110 return new int[]{hue(c), saturation, percent(maximum + minimum, 510)};
111 }
112 static int[] fromHue(int h, long chroma, long offset, long denominator) {
113 int sector = h / 60, f = h % 60;
114 int x = sector % 2 == 0 ? f : 60 - f;
115 int[][] factors = {{60,x,0},{x,60,0},{0,60,x},{0,x,60},{x,0,60},{60,0,x}};
116 int[] result = new int[3];
117 for (int i = 0; i < 3; i++) result[i] = round(255 * (chroma * factors[sector][i] + 60 * offset), 60 * denominator);
118 return result;
119 }
120 static int[] fromHsl(int h, int s, int l) {
121 long c = (100 - Math.abs(2 * l - 100)) * s;
122 return fromHue(h, 2 * c, 200L * l - c, 20000);
123 }
124 static long distance(int[] a, int[] b) {
125 long result = 0;
126 for (int i = 0; i < 3; i++) result += (long)(a[i] - b[i]) * (a[i] - b[i]);
127 return result;
128 }
129 static String model(String command, int[] c) {
130 var statistics = Arrays.stream(c).summaryStatistics();
131 int r = c[0], g = c[1], b = c[2], m = statistics.getMax(), n = statistics.getMin(), d = m - n;
132 return switch (command) {
133 case "HSL" -> {
134 int[] v = hsl(c);
135 yield "hsl(" + v[0] + "," + v[1] + "%," + v[2] + "%)";
136 }
137 case "HSV" -> "hsv(" + hue(c) + "," + (m == 0 ? 0 : percent(d,m)) + "%," + percent(m,255) + "%)";
138 case "HWB" -> "hwb(" + hue(c) + "," + percent(n,255) + "%," + percent(255 - m,255) + "%)";
139 case "CMYK" -> m == 0 ? "cmyk(0%,0%,0%,100%)" :
140 "cmyk(" + percent(m-r,m) + "%," + percent(m-g,m) + "%," + percent(m-b,m) + "%," + percent(255-m,255) + "%)";
141 case "YCBCR" -> "ycbcr(" + clamp(round(299L*r+587L*g+114L*b,1000)) + "," +
142 clamp(round(128000L-169L*r-331L*g+500L*b,1000)) + "," + clamp(round(128000L+500L*r-419L*g-81L*b,1000)) + ")";
143 case "YIQ" -> "yiq(" + round(299L*r+587L*g+114L*b,1000) + "," +
144 round(596L*r-274L*g-322L*b,1000) + "," + round(211L*r-523L*g+312L*b,1000) + ")";
145 default -> throw new Invalid();
146 };
147 }
148 static int[] ansi(int number) {
149 if (number < 16) return unpack(RGB[number]);
150 if (number >= 232) {
151 int value = 8 + 10 * (number - 232);
152 return new int[]{value, value, value};
153 }
154 int[] values = {0,95,135,175,215,255};
155 int v = number - 16;
156 return new int[]{values[v / 36], values[v / 6 % 6], values[v % 6]};
157 }
158 static int[] gradient(int[] a, int[] b, int index, int denominator) {
159 int[] result = new int[3];
160 for (int i = 0; i < 3; i++) result[i] = a[i] + away((long)(b[i] - a[i]) * index, denominator);
161 return result;
162 }
163 static long luminance(int[] c) {
164 return 2126L*c[0]*c[0] + 7152L*c[1]*c[1] + 722L*c[2]*c[2];
165 }
166 static long[] contrast(int[] a, int[] b) {
167 long first = luminance(a), second = luminance(b);
168 return new long[]{Math.max(first,second) + 32512500, Math.min(first,second) + 32512500};
169 }
170 static int ratio(int[] a, int[] b) {
171 long[] values = contrast(a,b);
172 return round(100 * values[0], values[1]);
173 }
174 static boolean readable(int[] a, int[] b) {
175 long[] values = contrast(a,b);

177 }
178 static String contrastText(int[] a, int[] b) {
179 long[] values = contrast(a,b);
180 int ratio = round(100 * values[0], values[1]);
181 String verdict = values[0] >= 7 * values[1] ? "AAA" : 2 * values[0] >= 9 * values[1] ? "AA" : "FAIL";
182 return String.format(Locale.ROOT, "%d.%02d %s", ratio / 100, ratio % 100, verdict);
183 }
184 static int blendComponent(String mode, int x, int y) {
185 int multiply = x * y / 255, screen = 255 - (255-x)*(255-y)/255;
186 return switch (mode) {
187 case "MULTIPLY" -> multiply;
188 case "SCREEN" -> screen;
189 case "OVERLAY" -> x < 128 ? 2*x*y/255 : 255-2*(255-x)*(255-y)/255;
190 case "HARDLIGHT" -> y < 128 ? 2*x*y/255 : 255-2*(255-x)*(255-y)/255;
191 case "SOFTLIGHT" -> ((255-x)*multiply+x*screen)/255;
192 case "DARKEN" -> Math.min(x,y);
193 case "LIGHTEN" -> Math.max(x,y);
194 case "DIFFERENCE" -> Math.abs(x-y);
195 case "EXCLUSION" -> x+y-2*x*y/255;
196 case "ADD" -> Math.min(255,x+y);
197 case "SUBTRACT" -> Math.max(0,x-y);
198 case "COLORDODGE" -> y == 255 ? 255 : Math.min(255,255*x/(255-y));
199 case "COLORBURN" -> y == 0 ? 0 : Math.max(0,255-255*(255-x)/y);
200 default -> throw new Invalid();
201 };
202 }
203 static int[] blend(String mode, int[] base, int[] top) {
204 if (Set.of("HUE","SATURATION","COLOR","LUMINOSITY").contains(mode)) {
205 int[] a = hsl(base), b = hsl(top);
206 return fromHsl(mode.equals("HUE") || mode.equals("COLOR") ? b[0] : a[0],
207 mode.equals("SATURATION") || mode.equals("COLOR") ? b[1] : a[1], mode.equals("LUMINOSITY") ? b[2] : a[2]);
208 }
209 int[] result = new int[3];
210 for (int i = 0; i < 3; i++) result[i] = blendComponent(mode,base[i],top[i]);
211 return result;
212 }

217 static int[] ranges(List<int[]> colors) {
218 return IntStream.range(0, 3).map(component -> {
219 var statistics = colors.stream().mapToInt(color -> color[component]).summaryStatistics();
220 return statistics.getMax() - statistics.getMin();
221 }).toArray();

223 static List<Cluster> palette(int k, List<int[]> colors) {
224 List<List<int[]>> boxes = new ArrayList<>();
225 boxes.add(new ArrayList<>(colors));
226 while (boxes.size() < k) {
227 int selected = -1, width = -1;
228 for (int i = 0; i < boxes.size(); i++) {
229 if (boxes.get(i).size() < 2) continue;
230 int candidate = Arrays.stream(ranges(boxes.get(i))).max().orElseThrow();
231 if (candidate > width) {
232 width = candidate;
233 selected = i;
234 }
235 }
236 List<int[]> box = boxes.get(selected);
237 int[] ranges = ranges(box);
238 int component = 0;
239 for (int i = 1; i < 3; i++) {
240 if (ranges[i] > ranges[component]) component = i;
241 }
242 final int axis = component;
243 box.sort(Comparator.comparingInt(c -> c[axis]));
244 int middle = box.size() / 2;
245 boxes.set(selected,new ArrayList<>(box.subList(0,middle)));
246 boxes.add(selected + 1,new ArrayList<>(box.subList(middle,box.size())));
247 }
248 return boxes.stream().map(box -> new Cluster(mean(box), box.size())).toList();
249 }
250 static String clusterText(List<Cluster> clusters) {
251 return String.join(" ", clusters.stream().map(cluster -> hex(cluster.color()) + ":" + cluster.count()).toList());
252 }
253 static String kmeans(int k, List<int[]> colors) {
254 List<Cluster> initial = palette(k,colors);
255 List<int[]> representatives = new ArrayList<>();
256 for (Cluster cluster : initial) representatives.add(cluster.color());
257 List<Cluster> result = new ArrayList<>();
258 int iteration = 0;
259 while (iteration < 20) {
260 iteration++;
261 List<List<int[]>> groups = new ArrayList<>();
262 for (int i = 0; i < k; i++) groups.add(new ArrayList<>());
263 for (int[] color : colors) {
264 int best = 0;
265 for (int j = 1; j < k; j++) {
266 if (distance(color,representatives.get(j)) < distance(color,representatives.get(best))) best = j;
267 }
268 groups.get(best).add(color);
269 }
270 result = new ArrayList<>();
271 boolean unchanged = true;
272 for (int j = 0; j < k; j++) {
273 int[] next = groups.get(j).isEmpty() ? representatives.get(j) : mean(groups.get(j));
274 if (!Arrays.equals(next,representatives.get(j))) unchanged = false;
275 result.add(new Cluster(next,groups.get(j).size()));
276 }
277 for (int j = 0; j < k; j++) representatives.set(j,result.get(j).color());
278 if (unchanged) break;
279 }
280 return iteration + " " + clusterText(result);
281 }
