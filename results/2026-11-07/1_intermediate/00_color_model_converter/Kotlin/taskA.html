3class Invalid : RuntimeException()
4data class Entry(val name: String, val color: IntArray)
5data class Cluster(val color: IntArray, val count: Int)

7class Converter {
8 private val names = listOf(
9 "black", "maroon", "green", "olive", "navy", "purple", "teal", "silver",
10 "gray", "red", "lime", "yellow", "blue", "magenta", "cyan", "white",
11 "orange", "pink", "brown", "gold", "coral", "salmon", "violet", "indigo"
12 )
13 private val rgb = intArrayOf(
14 0x000000, 0x800000, 0x008000, 0x808000, 0x000080, 0x800080, 0x008080, 0xC0C0C0,
15 0x808080, 0xFF0000, 0x00FF00, 0xFFFF00, 0x0000FF, 0xFF00FF, 0x00FFFF, 0xFFFFFF,
16 0xFFA500, 0xFFC0CB, 0xA52A2A, 0xFFD700, 0xFF7F50, 0xFA8072, 0xEE82EE, 0x4B0082
17 )
18 private val integerPattern = Regex("-?[0-9]{1,9}")
19 private val namePattern = Regex("[a-z]{1,12}")
20 private val hexPattern = Regex("#[0-9a-fA-F]{3}(?:[0-9a-fA-F]{3})?")
21 private val rgbPattern = Regex("rgb\\(([0-9]{1,3}),([0-9]{1,3}),([0-9]{1,3})\\)")
22 private val registered = mutableListOf<Entry>()

24 private fun check(condition: Boolean) {
25 if (!condition) throw Invalid()
26 }
27 private fun arity(tokens: List<String>, count: Int) {
28 check(tokens.size == count + 1)
29 }
30 private fun number(text: String, low: Int, high: Int): Int {
31 check(integerPattern.matches(text))
32 val value = text.toInt()
33 check(value in low..high)
34 return value
35 }
36 private fun round(n: Long, d: Long): Int = Math.floorDiv(2 * n + d, 2 * d).toInt()
37 private fun round(n: Long, d: Int): Int = round(n, d.toLong())
38 private fun round(n: Int, d: Int): Int = round(n.toLong(), d.toLong())
39 private fun away(n: Long, d: Int): Int = if (n < 0) -round(-n,d) else round(n,d)
40 private fun percent(a: Int, b: Int): Int = round(100L*a,b)
41 private fun clamp(value: Int): Int = value.coerceIn(0,255)
42 private fun root(numerator: Long, scale: Long): Int {
43 val value = numerator / scale
44 var low = 0L
45 var high = value + 1
46 while (low + 1 < high) {
47 val middle = (low + high) / 2
48 if (middle * middle <= value) low = middle else high = middle
49 }
50 return low.toInt()
51 }
52 private fun unpack(value: Int): IntArray = intArrayOf(value shr 16, (value shr 8) and 255, value and 255)
53 private fun hex(color: IntArray): String = String.format(Locale.ROOT,"#%02X%02X%02X",color[0],color[1],color[2])
54 private fun table(): List<Entry> = names.indices.map { Entry(names[it],unpack(rgb[it])) } + registered
55 private fun color(text: String): IntArray {
56 if (hexPattern.matches(text)) {
57 var digits = text.substring(1)
58 if (digits.length == 3) digits = digits.map { "$it$it" }.joinToString("")
59 return unpack(digits.toInt(16))
60 }
61 val match = rgbPattern.matchEntire(text)
62 if (match != null) {
63 val result = IntArray(3) { match.groupValues[it + 1].toInt() }
64 check(result.all { it <= 255 })
65 return result
66 }
67 return table().firstOrNull { it.name == text }?.color ?: throw Invalid()
68 }
69 private fun hue(c: IntArray): Int {
70 val largest = c.max()
71 val difference = largest - c.min()
72 if (difference == 0) return 0
73 var numerator = when (largest) {
74 c[0] -> 60L * (c[1] - c[2])
75 c[1] -> 60L * (c[2] - c[0]) + 120L * difference
76 else -> 60L * (c[0] - c[1]) + 240L * difference
77 }
78 if (numerator < 0) numerator += 360L * difference
79 return round(numerator,difference) % 360
80 }
81 private fun hsl(c: IntArray): IntArray {
82 val largest = c.max()
83 val smallest = c.min()
84 val difference = largest - smallest
85 val saturation = if (difference == 0) 0 else percent(difference,255 - kotlin.math.abs(largest + smallest - 255))
86 return intArrayOf(hue(c),saturation,percent(largest + smallest,510))
87 }
88 private fun fromHue(h: Int, chroma: Long, offset: Long, denominator: Long): IntArray {
89 val sector = h / 60
90 val f = h % 60
91 val x = if (sector % 2 == 0) f else 60 - f
92 val factors = arrayOf(intArrayOf(60,x,0),intArrayOf(x,60,0),intArrayOf(0,60,x),
93 intArrayOf(0,x,60),intArrayOf(x,0,60),intArrayOf(60,0,x))
94 return IntArray(3) { round(255 * (chroma * factors[sector][it] + 60 * offset),60 * denominator) }
95 }
96 private fun fromHsl(h: Int, s: Int, l: Int): IntArray {
97 val c = (100 - kotlin.math.abs(2 * l - 100)).toLong() * s
98 return fromHue(h,2*c,200L*l-c,20000)
99 }
100 private fun distance(a: IntArray, b: IntArray): Long = (0..2).sumOf {
101 (a[it] - b[it]).toLong() * (a[it] - b[it])
102 }
103 private fun model(command: String, c: IntArray): String {
104 val r = c[0]
105 val g = c[1]
106 val b = c[2]
107 val m = c.max()
108 val n = c.min()
109 val d = m - n
110 return when (command) {
111 "HSL" -> {
112 val v = hsl(c)
113 "hsl(${v[0]},${v[1]}%,${v[2]}%)"
114 }
115 "HSV" -> "hsv(${hue(c)},${if (m == 0) 0 else percent(d,m)}%,${percent(m,255)}%)"
116 "HWB" -> "hwb(${hue(c)},${percent(n,255)}%,${percent(255-m,255)}%)"
117 "CMYK" -> if (m == 0) "cmyk(0%,0%,0%,100%)" else
118 "cmyk(${percent(m-r,m)}%,${percent(m-g,m)}%,${percent(m-b,m)}%,${percent(255-m,255)}%)"
119 "YCBCR" -> "ycbcr(${clamp(round(299L*r+587L*g+114L*b,1000))}," +
120 "${clamp(round(128000L-169L*r-331L*g+500L*b,1000))},${clamp(round(128000L+500L*r-419L*g-81L*b,1000))})"
121 "YIQ" -> "yiq(${round(299L*r+587L*g+114L*b,1000)}," +
122 "${round(596L*r-274L*g-322L*b,1000)},${round(211L*r-523L*g+312L*b,1000)})"
123 else -> throw Invalid()
124 }
125 }
126 private fun ansi(number: Int): IntArray {
127 if (number < 16) return unpack(rgb[number])
128 if (number >= 232) return IntArray(3) { 8 + 10 * (number - 232) }
129 val values = intArrayOf(0,95,135,175,215,255)
130 val v = number - 16
131 return intArrayOf(values[v / 36],values[v / 6 % 6],values[v % 6])
132 }
133 private fun gradient(a: IntArray, b: IntArray, index: Int, denominator: Int): IntArray =
134 IntArray(3) { a[it] + away((b[it]-a[it]).toLong()*index,denominator) }
135 private fun luminance(c: IntArray): Long = 2126L*c[0]*c[0] + 7152L*c[1]*c[1] + 722L*c[2]*c[2]
136 private fun contrast(a: IntArray, b: IntArray): LongArray {
137 val first = luminance(a)
138 val second = luminance(b)
139 return longArrayOf(maxOf(first,second)+32512500,minOf(first,second)+32512500)
140 }
141 private fun ratio(a: IntArray, b: IntArray): Int {
142 val values = contrast(a,b)
143 return round(100*values[0],values[1])
144 }
145 private fun readable(a: IntArray, b: IntArray): Boolean {
146 val values = contrast(a,b)
147 return 2*values[0] >= 9*values[1]
148 }
149 private fun contrastText(a: IntArray, b: IntArray): String {
150 val values = contrast(a,b)
151 val ratio = round(100*values[0],values[1])
152 val verdict = if (values[0] >= 7*values[1]) "AAA" else if (2*values[0] >= 9*values[1]) "AA" else "FAIL"
153 return "${ratio / 100}.${(ratio % 100).toString().padStart(2,'0')} $verdict"
154 }
155 private fun blendComponent(mode: String, x: Int, y: Int): Int {
156 val multiply = x*y/255
157 val screen = 255-(255-x)*(255-y)/255
158 return when (mode) {
159 "MULTIPLY" -> multiply
160 "SCREEN" -> screen
161 "OVERLAY" -> if (x < 128) 2*x*y/255 else 255-2*(255-x)*(255-y)/255
162 "HARDLIGHT" -> if (y < 128) 2*x*y/255 else 255-2*(255-x)*(255-y)/255
163 "SOFTLIGHT" -> ((255-x)*multiply+x*screen)/255
164 "DARKEN" -> minOf(x,y)
165 "LIGHTEN" -> maxOf(x,y)
166 "DIFFERENCE" -> kotlin.math.abs(x-y)
167 "EXCLUSION" -> x+y-2*x*y/255
168 "ADD" -> minOf(255,x+y)
169 "SUBTRACT" -> maxOf(0,x-y)
170 "COLORDODGE" -> if (y == 255) 255 else minOf(255,255*x/(255-y))
171 "COLORBURN" -> if (y == 0) 0 else maxOf(0,255-255*(255-x)/y)
172 else -> throw Invalid()
173 }
174 }
175 private fun blend(mode: String, base: IntArray, top: IntArray): IntArray {
176 if (mode in setOf("HUE","SATURATION","COLOR","LUMINOSITY")) {
177 val a = hsl(base)
178 val b = hsl(top)
179 return fromHsl(if (mode == "HUE" || mode == "COLOR") b[0] else a[0],
180 if (mode == "SATURATION" || mode == "COLOR") b[1] else a[1],if (mode == "LUMINOSITY") b[2] else a[2])
181 }
182 return IntArray(3) { blendComponent(mode,base[it],top[it]) }
183 }
184 private fun mean(colors: List<IntArray>): IntArray = IntArray(3) { component ->
185 round(colors.sumOf { it[component] },colors.size)
186 }
187 private fun ranges(colors: List<IntArray>): IntArray = IntArray(3) { component ->
188 colors.maxOf { it[component] } - colors.minOf { it[component] }
189 }
190 private fun palette(k: Int, colors: List<IntArray>): List<Cluster> {
191 val boxes = mutableListOf(colors)
192 while (boxes.size < k) {
193 val selected = boxes.indices.filter { boxes[it].size > 1 }.maxBy { ranges(boxes[it]).max() }
194 val range = ranges(boxes[selected])
195 val component = range.indices.maxBy { range[it] }
196 val sorted = boxes[selected].sortedBy { it[component] }
197 val middle = sorted.size / 2
198 boxes[selected] = sorted.take(middle)
199 boxes.add(selected + 1,sorted.drop(middle))
200 }
201 return boxes.map { Cluster(mean(it),it.size) }
202 }
203 private fun clusterText(clusters: List<Cluster>): String =
204 clusters.joinToString(" ") { "${hex(it.color)}:${it.count}" }
205 private fun kmeans(k: Int, colors: List<IntArray>): String {
206 var representatives = palette(k,colors).map { it.color }
207 var result = emptyList<Cluster>()
208 var iteration = 0
209 while (iteration < 20) {
210 iteration++
211 val groups = List(k) { mutableListOf<IntArray>() }
212 for (color in colors) {
213 val best = (0 until k).minBy { distance(color, representatives[it]) }
214 groups[best].add(color)
215 }
216 result = groups.indices.map { j ->
217 Cluster(if (groups[j].isEmpty()) representatives[j] else mean(groups[j]),groups[j].size)
218 }
219 val unchanged = result.indices.all { result[it].color.contentEquals(representatives[it]) }
220 representatives = result.map { it.color }
221 if (unchanged) break
222 }
223 return "$iteration ${clusterText(result)}"
224 }
225 private fun colors(tokens: List<String>, start: Int): List<IntArray> = tokens.drop(start).map { color(it) }
226 private fun colorsText(colors: List<IntArray>): String = colors.joinToString(" ") { hex(it) }
227 fun execute(line: String): String {
228 val t = line.split(' ')
229 val command = t[0]
230 when (command) {
231 "DEFINE" -> {
232 arity(t,2)
233 check(namePattern.matches(t[1]) && t[1] !in names)
234 val c = color(t[2])
235 val entry = Entry(t[1],c)
236 val index = registered.indexOfFirst { it.name == t[1] }
237 if (index < 0) registered.add(entry) else registered[index] = entry
238 return "${t[1]} ${hex(c)}"
239 }
240 "UNDEFINE" -> {
241 arity(t,1)
242 check(namePattern.matches(t[1]))
243 val index = registered.indexOfFirst { it.name == t[1] }
244 check(index >= 0)
245 registered.removeAt(index)
246 return "${t[1]} removed"
247 }
248 "LIST" -> {
249 arity(t,0)
250 return if (registered.isEmpty()) "NONE" else
251 registered.sortedBy { it.name }.joinToString(" ") { "${it.name}=${hex(it.color)}" }
252 }
253 "PARSE" -> {
254 arity(t,1)
255 val c = color(t[1])
256 val name = table().firstOrNull { it.color.contentEquals(c) }?.name ?: "-"
257 return "${hex(c)} rgb(${c[0]},${c[1]},${c[2]}) " +
258 "${model("HSL",c)} ${model("HSV",c)} ${model("HWB",c)} ${model("CMYK",c)} " +
259 "${model("YCBCR",c)} ${model("YIQ",c)} $name"
260 }
261 "HSL", "HSV", "HWB", "CMYK", "YCBCR", "YIQ" -> {
262 arity(t,1)
263 return model(command,color(t[1]))
264 }
265 "FROMHSL", "FROMHSV", "FROMHWB" -> {
266 arity(t,3)
267 val h = number(t[1],0,359)
268 val s = number(t[2],0,100)
269 val third = number(t[3],0,100)
270 val c = when {
271 command == "FROMHSL" -> fromHsl(h,s,third)
272 command == "FROMHSV" -> fromHue(h,third.toLong()*s,100L*third-third*s,10000)
273 s + third >= 100 -> IntArray(3) { round(255L*s,s+third) }
274 else -> fromHue(h,(100-s-third).toLong(),s.toLong(),100)
275 }
276 return hex(c)
277 }
278 "FROMCMYK" -> {
279 arity(t,4)
280 val k = number(t[4],0,100)
281 return hex(IntArray(3) { round(255L*(100-number(t[it+1],0,100))*(100-k),10000) })
282 }
283 "FROMYCBCR", "FROMYIQ" -> {
284 arity(t,3)
285 val y = number(t[1],0,255)
286 val iq = command == "FROMYIQ"
287 val a = number(t[2],if (iq) -255 else 0,255)
288 val b = number(t[3],if (iq) -255 else 0,255)
289 val numerators = if (iq) longArrayOf(1000L*y+956L*a+619L*b,
290 1000L*y-272L*a-647L*b,1000L*y-1106L*a+1703L*b)
291 else longArrayOf(1000L*y+1402L*(b-128),1000L*y-344L*(a-128)-714L*(b-128),1000L*y+1772L*(a-128))
292 return hex(IntArray(3) { clamp(round(numerators[it],1000)) })
293 }
294 "RGB565" -> {
295 arity(t,1)
296 val c = color(t[1])
297 val value = 2048*round(31L*c[0],255)+32*round(63L*c[1],255)+round(31L*c[2],255)
298 return "0x" + value.toString(16).uppercase().padStart(4,'0')
299 }
300 "FROM565" -> {
301 arity(t,1)
302 check(Regex("0x[0-9a-fA-F]{4}").matches(t[1]))
303 val v = t[1].substring(2).toInt(16)
304 return hex(intArrayOf(round(255L*(v/2048),31),round(255L*(v/32%64),63),round(255L*(v%32),31)))
305 }
306 "FROMANSI" -> {
307 arity(t,1)
308 return hex(ansi(number(t[1],0,255)))
309 }
310 "ANSI" -> {
311 arity(t,1)
312 val c = color(t[1])
313 val best = (16..255).minBy { distance(c, ansi(it)) }
314 return "$best ${hex(ansi(best))} ${distance(c,ansi(best))}"
315 }
316 "WEBSAFE", "INVERT" -> {
317 arity(t,1)
318 val c = color(t[1])
319 return hex(IntArray(3) { if (command == "INVERT") 255-c[it] else 51*round(c[it],51) })
320 }
321 "GRAY" -> {
322 arity(t,2)
323 val c = color(t[1])
324 val value = when (t[2]) {
325 "LUMA" -> round(299L*c[0]+587L*c[1]+114L*c[2],1000)
326 "AVERAGE" -> round(c.sum(),3)
327 "LIGHTNESS" -> round(c.max()+c.min(),2)
328 "HSP" -> root(299L*c[0]*c[0]+587L*c[1]*c[1]+114L*c[2]*c[2],1000)
329 else -> throw Invalid()
330 }
331 return hex(IntArray(3) { value })
332 }
333 "MIX", "MIXHSL" -> {
334 arity(t,3)
335 val a = color(t[1])
336 val b = color(t[2])
337 val p = number(t[3],0,100)
338 if (command == "MIX") return hex(IntArray(3) { round(a[it].toLong()*(100-p)+b[it].toLong()*p,100) })
339 val first = hsl(a)
340 val second = hsl(b)
341 var e = Math.floorMod(second[0]-first[0],360)
342 if (e > 180) e -= 360
343 return hex(fromHsl(Math.floorMod(first[0]+away(e.toLong()*p,100),360),
344 first[1]+away((second[1]-first[1]).toLong()*p,100),first[2]+away((second[2]-first[2]).toLong()*p,100)))
345 }
346 "GRADIENT", "TINTS", "SHADES", "TONES" -> {
347 arity(t,if (command == "GRADIENT") 3 else 2)
348 val a = color(t[1])
349 val result: List<IntArray>
350 if (command == "GRADIENT") {
351 val b = color(t[2])
352 val n = number(t[3],2,16)
353 result = (0 until n).map { gradient(a,b,it,n-1) }
354 } else {
355 val n = number(t[2],1,10)
356 val target = if (command == "TINTS") 255 else if (command == "SHADES") 0 else 128
357 val b = IntArray(3) { target }
358 result = (1..n).map { gradient(a,b,it,n+1) }
359 }
360 return colorsText(result)
361 }
362 "BLEND" -> {
363 arity(t,3)
364 return hex(blend(t[1],color(t[2]),color(t[3])))
365 }
366 "ALPHA" -> {
367 arity(t,4)
368 val fg = color(t[1])
369 val p = number(t[2],0,100)
370 val bg = color(t[3])
371 val q = number(t[4],0,100)
372 val a = 100*p+q*(100-p)
373 val result = IntArray(3) { if (a == 0) 0 else round(100L*p*fg[it]+q.toLong()*(100-p)*bg[it],a) }
374 return "${hex(result)} ${round(a,100)}%"
375 }
376 "SIMULATE" -> {
377 arity(t,2)
378 val c = color(t[1])
379 val coefficients = when (t[2]) {
380 "PROTANOPIA" -> listOf(listOf(567,433,0),listOf(558,442,0),listOf(0,242,758))
381 "DEUTERANOPIA" -> listOf(listOf(625,375,0),listOf(700,300,0),listOf(0,300,700))
382 "TRITANOPIA" -> listOf(listOf(950,50,0),listOf(0,433,567),listOf(0,475,525))
383 "ACHROMATOPSIA" -> List(3) { listOf(299,587,114) }
384 else -> throw Invalid()
385 }
386 return hex(IntArray(3) { i -> round((0..2).sumOf { coefficients[i][it].toLong()*c[it] },1000) })
387 }
388 "ROTATE", "COMPLEMENT", "TRIADIC", "TETRADIC", "SPLITCOMP", "ANALOGOUS" -> {
389 arity(t,if (command == "ROTATE") 2 else if (command == "ANALOGOUS") 3 else 1)
390 val values = hsl(color(t[1]))
391 val rotations = when (command) {
392 "ROTATE" -> listOf(number(t[2],-720,720))
393 "COMPLEMENT" -> listOf(180)
394 "TRIADIC" -> listOf(120,240)
395 "TETRADIC" -> listOf(90,180,270)
396 "SPLITCOMP" -> listOf(150,210)
397 else -> {
398 val k = number(t[2],1,6)
399 val step = number(t[3],1,60)
400 (-k..k).filter { it != 0 }.map { it*step }
401 }
402 }
403 return colorsText(rotations.map { fromHsl(Math.floorMod(values[0]+it,360),values[1],values[2]) })
404 }
405 "LIGHTEN", "DARKEN", "SATURATE", "DESATURATE" -> {
406 arity(t,2)
407 val values = hsl(color(t[1]))
408 val p = number(t[2],0,100)
409 when (command) {
410 "LIGHTEN" -> values[2] = minOf(100,values[2]+p)
411 "DARKEN" -> values[2] = maxOf(0,values[2]-p)
412 "SATURATE" -> values[1] = minOf(100,values[1]+p)
413 "DESATURATE" -> values[1] = maxOf(0,values[1]-p)
414 }
415 return hex(fromHsl(values[0],values[1],values[2]))
416 }
417 "NEAREST" -> {
418 arity(t,1)
419 val c = color(t[1])
420 val best = table().minBy { distance(c, it.color) }
421 return "${best.name} ${hex(best.color)} ${distance(c,best.color)}"
422 }
423 "DISTANCE" -> {
424 arity(t,2)
425 val a = color(t[1])
426 val b = color(t[2])
427 val dr = (a[0]-b[0]).toLong()
428 val dg = (a[1]-b[1]).toLong()
429 val db = (a[2]-b[2]).toLong()
430 val w = (1024L+a[0]+b[0])*dr*dr+2048*dg*dg+(1534L-a[0]-b[0])*db*db
431 return "${distance(a,b)} ${root(w,512)}"
432 }
433 "CONTRAST" -> {
434 arity(t,2)
435 return contrastText(color(t[1]),color(t[2]))
436 }
437 "ACCESSIBLE" -> {
438 arity(t,2)
439 val fg = color(t[1])
440 val bg = color(t[2])
441 if (readable(fg,bg)) return "${hex(fg)} ${contrastText(fg,bg)}"
442 val values = hsl(fg)
443 for (d in 1..100) {
444 for (l in listOf(values[2]-d,values[2]+d)) {
445 if (l !in 0..100) continue
446 val candidate = fromHsl(values[0],values[1],l)
447 if (readable(candidate,bg)) return "${hex(candidate)} ${contrastText(candidate,bg)}"
448 }
449 }
450 throw Invalid()
451 }
452 "BEST" -> {
453 check(t.size in 3..34)
454 val bg = color(t[1])
455 val options = colors(t,2)
456 val best = options.maxBy { ratio(it, bg) }
457 return "${hex(best)} ${contrastText(best,bg)}"
458 }
459 "DESCRIBE" -> {
460 arity(t,1)
461 val (h,s,l) = hsl(color(t[1])).toList()
462 if (l <= 5) return "black"
463 if (l >= 95) return "white"
464 val lightness = when {
465 l <= 20 -> "very dark"
466 l <= 40 -> "dark"
467 l <= 60 -> "medium"
468 l <= 80 -> "light"
469 else -> "very light"
470 }
471 if (s < 10) return "$lightness gray"
472 val saturation = if (s < 40) "muted" else if (s < 80) "moderate" else "vivid"
473 val hues = listOf("red","orange","yellow","chartreuse","green","spring-green",
474 "cyan","azure","blue","violet","magenta","rose")
475 return "$lightness $saturation ${hues[(h+15)%360/30]}"
476 }
477 "PALETTE", "KMEANS" -> {
478 check(t.size in 3..34)
479 val k = number(t[1],1,t.size-2)
480 val options = colors(t,2)
481 return if (command == "PALETTE") clusterText(palette(k,options)) else kmeans(k,options)
482 }
483 "SORT" -> {
484 check(t.size in 2..33)
485 val options = colors(t,1).sortedWith(compareBy<IntArray> { hsl(it)[0] }
486 .thenBy { hsl(it)[1] }.thenBy { hsl(it)[2] })
487 return colorsText(options)
488 }
489 "DITHER" -> {
490 check(t.size in 2..33)
491 val options = colors(t,1)
492 val error = IntArray(3)
493 val result = mutableListOf<String>()
494 for (c in options) {
495 val adjusted = IntArray(3) { clamp(c[it]+error[it]) }
496 val best = (0 until 16).minBy { distance(adjusted, unpack(rgb[it])) }
497 result.add(names[best])
498 val chosen = unpack(rgb[best])
499 for (i in 0..2) error[i] = adjusted[i]-chosen[i]
500 }
501 return result.joinToString(" ")
502 }
503 "RANDOM" -> {
504 arity(t,2)
505 var state = number(t[1],1,999999999).toLong()
506 val n = number(t[2],1,16)

511 state = state*48271%2147483647

513 state = state*48271%2147483647

518 }
519 else -> throw Invalid()
520 }
