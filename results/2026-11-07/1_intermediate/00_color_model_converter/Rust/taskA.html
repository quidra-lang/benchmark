26fn rounded(numerator: i64, denominator: i64) -> i64 {
27 (2 * numerator + denominator).div_euclid(2 * denominator)
28}
29
30fn away(numerator: i64, denominator: i64) -> i64 {
31 if numerator < 0 { -rounded(-numerator, denominator) }
32 else { rounded(numerator, denominator) }
33}
34
35fn distance(a: Color, b: Color) -> i64 {
36 (0..3).map(|j| (a[j] - b[j]) * (a[j] - b[j])).sum()
37}

39fn hue(c: Color) -> i64 {
40 let maximum = *c.iter().max().unwrap();
41 let minimum = *c.iter().min().unwrap();
42 let delta = maximum - minimum;
43 if delta == 0 { return 0; }
44 let mut numerator;
45 if maximum == c[0] {
46 numerator = 60 * (c[1] - c[2]);
47 if numerator < 0 { numerator += 360 * delta; }
48 } else if maximum == c[1] {
49 numerator = 60 * (c[2] - c[0]) + 120 * delta;
50 } else {
51 numerator = 60 * (c[0] - c[1]) + 240 * delta;
52 }
53 rounded(numerator, delta) % 360
54}

56fn hsl(c: Color) -> [i64; 3] {
57 let maximum = *c.iter().max().unwrap();
58 let minimum = *c.iter().min().unwrap();
59 let delta = maximum - minimum;
60 let saturation = if delta == 0 { 0 }
61 else { rounded(100 * delta, 255 - (maximum + minimum - 255).abs()) };
62 [hue(c), saturation, rounded(100 * (maximum + minimum), 510)]
63}

65fn sector(h: i64) -> Color {
66 let q = h / 60;
67 let f = h % 60;
68 let x = if q % 2 == 0 { f } else { 60 - f };
69 match q {
70 0 => [60, x, 0],
71 1 => [x, 60, 0],
72 2 => [0, 60, x],
73 3 => [0, x, 60],
74 4 => [x, 0, 60],
75 _ => [60, 0, x],
76 }
77}

79fn from_hsl(h: i64, s: i64, l: i64) -> Color {
80 let chroma = (100 - (2 * l - 100).abs()) * s;
81 let offset = 30 * (200 * l - chroma);
82 sector(h).map(|x| rounded(255 * (x * chroma + offset), 600000))
83}
84
85fn from_hsv(h: i64, s: i64, v: i64) -> Color {
86 let chroma = v * s;
87 let offset = 60 * (100 * v - chroma);
88 sector(h).map(|x| rounded(255 * (x * chroma + offset), 600000))
89}
90
91fn from_hwb(h: i64, w: i64, k: i64) -> Color {
92 if w + k >= 100 { return [rounded(255 * w, w + k); 3]; }
93 sector(h).map(|x| rounded(255 * (x * (100 - w - k) + 60 * w), 6000))
94}

96fn ansi_color(index: usize) -> Color {
97 const LEVELS: [i64; 6] = [0, 95, 135, 175, 215, 255];
98 if index < 16 { return BUILTINS[index].1; }
99 if index >= 232 { return [8 + 10 * (index as i64 - 232); 3]; }
100 let code = index - 16;
101 [LEVELS[code / 36], LEVELS[code / 6 % 6], LEVELS[code % 6]]
102}

104fn contrast(a: Color, b: Color) -> (i64, &'static str) {
105 let luminance = |c: Color| 2126 * c[0] * c[0] + 7152 * c[1] * c[1] + 722 * c[2] * c[2];
106 let (y1, y2) = (luminance(a), luminance(b));
107 let high = y1.max(y2) + 32512500;
108 let low = y1.min(y2) + 32512500;
109 let verdict = if high >= 7 * low { "AAA" }
110 else if 2 * high >= 9 * low { "AA" } else { "FAIL" };
111 (rounded(100 * high, low), verdict)
112}

114fn contrast_text(c: Color, background: Color) -> String {
115 let (ratio, verdict) = contrast(c, background);
116 format!("{}.{:02} {}", ratio / 100, ratio % 100, verdict)
117}

123fn rotate(c: Color, degrees: i64) -> Color {
124 let [h, s, l] = hsl(c);

128fn interpolate(first: Color, last: Color, i: i64, denominator: i64) -> Color {
129 std::array::from_fn(|j| first[j] + away((last[j] - first[j]) * i, denominator))

132fn blend(mode: usize, base: Color, top: Color) -> Color {
133 if mode >= 13 {
134 let mut a = hsl(base);
135 let b = hsl(top);
136 if mode == 13 || mode == 15 { a[0] = b[0]; }
137 if mode == 14 || mode == 15 { a[1] = b[1]; }
138 if mode == 16 { a[2] = b[2]; }
139 return from_hsl(a[0], a[1], a[2]);
140 }
141 std::array::from_fn(|j| {
142 let (x, y) = (base[j], top[j]);
143 match mode {
144 0 => x * y / 255,
145 1 => 255 - (255 - x) * (255 - y) / 255,
146 2 | 3 => {
147 if (if mode == 2 { x } else { y }) < 128 { 2 * x * y / 255 }
148 else { 255 - 2 * (255 - x) * (255 - y) / 255 }
149 }
150 4 => {
151 let multiply = x * y / 255;
152 let screen = 255 - (255 - x) * (255 - y) / 255;
153 ((255 - x) * multiply + x * screen) / 255
154 }
155 5 => x.min(y),
156 6 => x.max(y),
157 7 => (x - y).abs(),
158 8 => x + y - 2 * x * y / 255,
159 9 => 255.min(x + y),
160 10 => 0.max(x - y),
161 11 => if y == 255 { 255 } else { 255.min(255 * x / (255 - y)) },
162 12 => if y == 0 { 0 } else { 0.max(255 - 255 * (255 - x) / y) },
163 _ => unreachable!(),
164 }
165 })
166}

168fn mean(colors: &[Color]) -> Color {
169 std::array::from_fn(|j| rounded(colors.iter().map(|c| c[j]).sum(), colors.len() as i64))
170}

172fn palette(colors: &[Color], wanted: usize) -> Vec<(Color, usize)> {
173 let mut boxes = vec![colors.to_vec()];
174 while boxes.len() < wanted {
175 let (mut chosen, mut channel, mut best_width) = (0, 0, -1);
176 for (i, colors) in boxes.iter().enumerate() {
177 if colors.len() < 2 { continue; }
178 let (mut width, mut component) = (-1, 0);
179 for j in 0..3 {
180 let minimum = colors.iter().map(|c| c[j]).min().unwrap();
181 let maximum = colors.iter().map(|c| c[j]).max().unwrap();
182 if maximum - minimum > width {
183 width = maximum - minimum;
184 component = j;
185 }
186 }
187 if width > best_width {
188 best_width = width;
189 chosen = i;
190 channel = component;
191 }
192 }
193 let mut selected = boxes.remove(chosen);
194 selected.sort_by_key(|c| c[channel]);
195 let right = selected.split_off(selected.len() / 2);
196 boxes.insert(chosen, selected);
197 boxes.insert(chosen + 1, right);
198 }
199 boxes.iter().map(|colors| (mean(colors), colors.len())).collect()
200}

202fn kmeans(colors: &[Color], wanted: usize) -> (usize, Vec<(Color, usize)>) {
203 let mut representatives: Vec<Color> = palette(colors, wanted).iter().map(|item| item.0).collect();
204 for iteration in 1..=20 {
205 let mut sums = vec![[0; 3]; wanted];
206 let mut counts = vec![0; wanted];
207 for &color in colors {
208 let mut closest = 0;
209 for j in 1..wanted {
210 if distance(color, representatives[j]) < distance(color, representatives[closest]) {
211 closest = j;
212 }
213 }
214 counts[closest] += 1;
215 for j in 0..3 { sums[closest][j] += color[j]; }
216 }
217 let mut unchanged = true;
218 for i in 0..wanted {
219 if counts[i] == 0 { continue; }
220 let updated = sums[i].map(|x| rounded(x, counts[i] as i64));
221 if updated != representatives[i] { unchanged = false; }
222 representatives[i] = updated;
223 }
224 if unchanged || iteration == 20 {
225 return (iteration, representatives.into_iter().zip(counts).collect());
226 }
227 }
228 unreachable!()
229}

231fn parse_integer(text: &str, low: i64, high: i64) -> Option<i64> {
232 let digits = text.strip_prefix('-').unwrap_or(text);
233 if digits.is_empty() || digits.len() > 9 || !digits.bytes().all(|c| c.is_ascii_digit()) {
234 return None;
235 }
236 let value = text.parse::<i64>().ok()?;
237 if value < low || value > high { None } else { Some(value) }
238}

240fn valid_name(text: &str) -> bool {
241 (1..=12).contains(&text.len()) && text.bytes().all(|c| c.is_ascii_lowercase())
242}

244fn hex_digit(c: u8) -> Option<i64> {
245 match c {
246 b'0'..=b'9' => Some(i64::from(c - b'0')),
247 b'a'..=b'f' => Some(i64::from(c - b'a' + 10)),
248 b'A'..=b'F' => Some(i64::from(c - b'A' + 10)),
249 _ => None,
250 }
251}

253fn parse_color(text: &str, registry: &[Entry]) -> Option<Color> {
254 let bytes = text.as_bytes();
255 if text.starts_with('#') {
256 if bytes.len() != 4 && bytes.len() != 7 { return None; }
257 let mut result = [0; 3];
258 for j in 0..3 {
259 result[j] = if bytes.len() == 4 { 17 * hex_digit(bytes[j + 1])? }
260 else { 16 * hex_digit(bytes[2 * j + 1])? + hex_digit(bytes[2 * j + 2])? };
261 }
262 return Some(result);
263 }
264 if text.starts_with("rgb(") {
265 let inner = text.strip_prefix("rgb(")?.strip_suffix(')')?;
266 let parts: Vec<&str> = inner.split(',').collect();
267 if parts.len() != 3 { return None; }
268 let mut result = [0; 3];
269 for j in 0..3 {
270 if parts[j].is_empty() || parts[j].len() > 3 || !parts[j].bytes().all(|c| c.is_ascii_digit()) {
271 return None;
272 }
273 result[j] = parts[j].parse().ok()?;
274 if result[j] > 255 { return None; }
275 }
276 return Some(result);
277 }
278 for &(name, color) in &BUILTINS {
279 if text == name { return Some(color); }
280 }
281 registry.iter().find(|entry| entry.name == text).map(|entry| entry.color)
282}

284fn table(registry: &[Entry]) -> impl Iterator<Item = (&str, Color)> + '_ {
285 BUILTINS.iter().copied()
286 .chain(registry.iter().map(|entry| (entry.name.as_str(), entry.color)))
287}

289fn prepare(spec: &Spec, words: &[&str], registry: &[Entry]) -> Option<Arguments> {
290 if words.len() < spec.minimum || words.len() > spec.maximum { return None; }
291 let mut args = Arguments { colors: [[0; 3]; 33], numbers: [0; 33], keyword: 0 };
292 let signature = spec.signature.as_bytes();
293 for (i, &word) in words.iter().enumerate() {
294 let kind = match signature.first() {
295 Some(b'*') => b'C',
296 Some(b'P') => if i == 0 { b'I' } else { b'C' },
297 _ => signature[i],
298 };
299 match kind {
300 b'C' => args.colors[i] = parse_color(word, registry)?,
301 b'I' => {
302 let (low, high) = if signature[0] == b'P' { (1, 32) } else { (spec.low[i], spec.high[i]) };
303 args.numbers[i] = parse_integer(word, low, high)?;
304 }
305 b'N' => if !valid_name(word) { return None; },
306 b'K' => args.keyword = spec.keywords.split('|').position(|candidate| candidate == word)?,
307 b'W' => {
308 if word.len() != 6 || !word.starts_with("0x") { return None; }
309 let mut value = 0;
310 for digit in word.bytes().skip(2) { value = 16 * value + hex_digit(digit)?; }
311 args.numbers[i] = value;
312 }
313 _ => unreachable!(),
314 }
315 }
316 if spec.signature == "P" && args.numbers[0] as usize > words.len() - 1 { return None; }
317 if spec.name == "DEFINE" && BUILTINS.iter().any(|&(name, _)| name == words[0]) { return None; }
318 if spec.name == "UNDEFINE" && !registry.iter().any(|entry| entry.name == words[0]) { return None; }
319 Some(args)
320}

322fn model(name: &str, c: Color) -> String {
323 let maximum = *c.iter().max().unwrap();
324 let minimum = *c.iter().min().unwrap();
325 let [r, g, b] = c;
326 match name {
327 "HSL" => {
328 let [h, s, l] = hsl(c);
329 format!("hsl({h},{s}%,{l}%)")
330 }
331 "HSV" => {
332 let s = if maximum == 0 { 0 } else { rounded(100 * (maximum - minimum), maximum) };
333 format!("hsv({},{}%,{}%)", hue(c), s, rounded(100 * maximum, 255))
334 }
335 "HWB" => format!("hwb({},{}%,{}%)", hue(c), rounded(100 * minimum, 255), rounded(100 * (255 - maximum), 255)),
336 "CMYK" => {
337 let cyan = if maximum == 0 { 0 } else { rounded(100 * (maximum - r), maximum) };
338 let magenta = if maximum == 0 { 0 } else { rounded(100 * (maximum - g), maximum) };
339 let yellow = if maximum == 0 { 0 } else { rounded(100 * (maximum - b), maximum) };
340 format!("cmyk({}%,{}%,{}%,{}%)", cyan, magenta, yellow, rounded(100 * (255 - maximum), 255))
341 }
342 "YCBCR" => format!("ycbcr({},{},{})", rounded(299 * r + 587 * g + 114 * b, 1000).clamp(0, 255),
343 rounded(128000 - 169 * r - 331 * g + 500 * b, 1000).clamp(0, 255),
344 rounded(128000 + 500 * r - 419 * g - 81 * b, 1000).clamp(0, 255)),
345 "YIQ" => format!("yiq({},{},{})", rounded(299 * r + 587 * g + 114 * b, 1000),
346 rounded(596 * r - 274 * g - 322 * b, 1000), rounded(211 * r - 523 * g + 312 * b, 1000)),
347 _ => unreachable!(),
348 }
349}

487fn run(line: &str, registry: &mut Vec<Entry>) -> Option<String> {
488 let tokens: Vec<&str> = line.split(' ').collect();
489 let spec = SPECS.iter().find(|spec| spec.name == tokens[0])?;
490 let words = &tokens[1..];
491 let args = prepare(spec, words, registry)?;
492 let c = args.colors[0];
493 let [a, b, d] = [args.numbers[0], args.numbers[1], args.numbers[2]];
494 let mut output = Vec::new();
495 match spec.name {
496 "DEFINE" => {
497 output.push(format!("{} {}", words[0], hex(args.colors[1])));
498 if let Some(entry) = registry.iter_mut().find(|entry| entry.name == words[0]) {
499 entry.color = args.colors[1];
500 } else {
501 registry.push(Entry { name: words[0].to_string(), color: args.colors[1] });
502 }
503 }
504 "UNDEFINE" => {
505 output.push(format!("{} removed", words[0]));
506 let index = registry.iter().position(|entry| entry.name == words[0]).unwrap();
507 registry.remove(index);
508 }
509 "LIST" => {
510 let mut sorted: Vec<&Entry> = registry.iter().collect();
511 sorted.sort_by(|a, b| a.name.cmp(&b.name));
512 for entry in sorted { output.push(format!("{}={}", entry.name, hex(entry.color))); }
513 if output.is_empty() { output.push("NONE".to_string()); }
514 }
515 "PARSE" => {
516 output.push(hex(c));
517 output.push(format!("rgb({},{},{})", c[0], c[1], c[2]));
518 for name in ["HSL", "HSV", "HWB", "CMYK", "YCBCR", "YIQ"] { output.push(model(name, c)); }
519 let name = table(registry).find(|entry| entry.1 == c).map_or("-", |entry| entry.0);
520 output.push(name.to_string());
521 }
522 "HSL" | "HSV" | "HWB" | "CMYK" | "YCBCR" | "YIQ" => output.push(model(spec.name, c)),
523 "FROMHSL" => output.push(hex(from_hsl(a, b, d))),
524 "FROMHSV" => output.push(hex(from_hsv(a, b, d))),
525 "FROMHWB" => output.push(hex(from_hwb(a, b, d))),
526 "FROMCMYK" => output.push(hex(std::array::from_fn(|j|
527 rounded(255 * (100 - args.numbers[j]) * (100 - args.numbers[3]), 10000)))),
528 "FROMYCBCR" => output.push(hex([
529 rounded(1000 * a + 1402 * (d - 128), 1000).clamp(0, 255),
530 rounded(1000 * a - 344 * (b - 128) - 714 * (d - 128), 1000).clamp(0, 255),
531 rounded(1000 * a + 1772 * (b - 128), 1000).clamp(0, 255),
532 ])),
533 "FROMYIQ" => output.push(hex([
534 rounded(1000 * a + 956 * b + 619 * d, 1000).clamp(0, 255),
535 rounded(1000 * a - 272 * b - 647 * d, 1000).clamp(0, 255),
536 rounded(1000 * a - 1106 * b + 1703 * d, 1000).clamp(0, 255),
537 ])),
538 "RGB565" => {
539 let value = 2048 * rounded(31 * c[0], 255) + 32 * rounded(63 * c[1], 255)
540 + rounded(31 * c[2], 255);
541 output.push(format!("0x{value:04X}"));
542 }
543 "FROM565" => output.push(hex([
544 rounded(255 * (a / 2048), 31), rounded(255 * (a / 32 % 64), 63), rounded(255 * (a % 32), 31),
545 ])),
546 "FROMANSI" => output.push(hex(ansi_color(a as usize))),
547 "ANSI" => {
548 let mut best = 16;
549 for i in 17..=255 {
550 if distance(c, ansi_color(i)) < distance(c, ansi_color(best)) { best = i; }
551 }
552 output.push(format!("{} {} {}", best, hex(ansi_color(best)), distance(c, ansi_color(best))));
553 }
554 "WEBSAFE" => output.push(hex(c.map(|x| 51 * rounded(x, 51)))),
555 "GRAY" => {
556 let maximum = *c.iter().max().unwrap();
557 let minimum = *c.iter().min().unwrap();
558 let value = match args.keyword {
559 0 => rounded(299 * c[0] + 587 * c[1] + 114 * c[2], 1000),
560 1 => rounded(c[0] + c[1] + c[2], 3),
561 2 => rounded(maximum + minimum, 2),
562 _ => ((299 * c[0] * c[0] + 587 * c[1] * c[1] + 114 * c[2] * c[2]) / 1000).isqrt(),
563 };
564 output.push(hex([value; 3]));
565 }
566 "INVERT" => output.push(hex(c.map(|x| 255 - x))),
567 "MIX" => output.push(hex(std::array::from_fn(|j|
568 rounded(c[j] * (100 - d) + args.colors[1][j] * d, 100)))),
569 "MIXHSL" => {
570 let first = hsl(c);
571 let last = hsl(args.colors[1]);
572 let mut delta = (last[0] - first[0]).rem_euclid(360);
573 if delta > 180 { delta -= 360; }
574 output.push(hex(from_hsl((first[0] + away(delta * d, 100)).rem_euclid(360),
575 first[1] + away((last[1] - first[1]) * d, 100),
576 first[2] + away((last[2] - first[2]) * d, 100))));
577 }
578 "GRADIENT" => {
579 for i in 0..d { output.push(hex(interpolate(c, args.colors[1], i, d - 1))); }
580 }
581 "TINTS" | "SHADES" | "TONES" => {
582 let target = match spec.name { "TINTS" => 255, "SHADES" => 0, _ => 128 };
583 for i in 1..=b { output.push(hex(interpolate(c, [target; 3], i, b + 1))); }
584 }
585 "BLEND" => output.push(hex(blend(args.keyword, args.colors[1], args.colors[2]))),
586 "ALPHA" => {
587 let q = args.numbers[3];
588 let alpha = 100 * b + q * (100 - b);
589 let color = if alpha == 0 { [0; 3] } else {
590 std::array::from_fn(|j| rounded(100 * b * c[j] + q * (100 - b) * args.colors[2][j], alpha))
591 };
592 output.push(format!("{} {}%", hex(color), rounded(alpha, 100)));
593 }
594 "SIMULATE" => {
595 const COEFFICIENTS: [[[i64; 3]; 3]; 4] = [
596 [[567,433,0],[558,442,0],[0,242,758]],
597 [[625,375,0],[700,300,0],[0,300,700]],
598 [[950,50,0],[0,433,567],[0,475,525]],
599 [[299,587,114],[299,587,114],[299,587,114]],
600 ];
601 let color = std::array::from_fn(|j|
602 rounded((0..3).map(|k| COEFFICIENTS[args.keyword][j][k] * c[k]).sum(), 1000));
603 output.push(hex(color));
604 }
605 "ROTATE" => output.push(hex(rotate(c, b))),
606 "COMPLEMENT" => output.push(hex(rotate(c, 180))),
607 "TRIADIC" => for shift in [120, 240] { output.push(hex(rotate(c, shift))); },
608 "TETRADIC" => for shift in [90, 180, 270] { output.push(hex(rotate(c, shift))); },
609 "SPLITCOMP" => for shift in [150, 210] { output.push(hex(rotate(c, shift))); },
610 "ANALOGOUS" => {
611 for i in -b..=b {
612 if i != 0 { output.push(hex(rotate(c, i * d))); }
613 }
614 }
615 "LIGHTEN" | "DARKEN" | "SATURATE" | "DESATURATE" => {
616 let mut value = hsl(c);
617 match spec.name {
618 "LIGHTEN" => value[2] = 100.min(value[2] + b),
619 "DARKEN" => value[2] = 0.max(value[2] - b),
620 "SATURATE" => value[1] = 100.min(value[1] + b),
621 _ => value[1] = 0.max(value[1] - b),
622 }
623 output.push(hex(from_hsl(value[0], value[1], value[2])));
624 }
625 "NEAREST" => {
626 let (name, color) = table(registry).min_by_key(|entry| distance(c, entry.1)).unwrap();
627 output.push(format!("{} {} {}", name, hex(color), distance(c, color)));
628 }
629 "DISTANCE" => {
630 let other = args.colors[1];
631 let differences: Color = std::array::from_fn(|j| c[j] - other[j]);
632 let [r, g, b] = differences;
633 let weight = (1024 + c[0] + other[0]) * r * r + 2048 * g * g
634 + (1534 - c[0] - other[0]) * b * b;
635 output.push(format!("{} {}", distance(c, other), (weight / 512).isqrt()));
636 }
637 "CONTRAST" => output.push(contrast_text(c, args.colors[1])),
638 "ACCESSIBLE" => {
639 let background = args.colors[1];
640 let mut result = c;
641 if contrast(c, background).1 == "FAIL" {
642 let [h, s, l] = hsl(c);
643 let mut found = false;
644 for step in 1..=100 {
645 for sign in [-1, 1] {
646 let lightness = l + sign * step;
647 if !(0..=100).contains(&lightness) { continue; }
648 let candidate = from_hsl(h, s, lightness);
649 if contrast(candidate, background).1 != "FAIL" {
650 result = candidate;
651 found = true;
652 break;
653 }
654 }
655 if found { break; }
656 }
657 }
658 output.push(format!("{} {}", hex(result), contrast_text(result, background)));
659 }
660 "BEST" => {
661 let mut best = 1;
662 for i in 2..words.len() {
663 if contrast(args.colors[i], c).0 > contrast(args.colors[best], c).0 { best = i; }
664 }
665 output.push(format!("{} {}", hex(args.colors[best]), contrast_text(args.colors[best], c)));
666 }
667 "DESCRIBE" => {
668 let [h, s, l] = hsl(c);
669 if l <= 5 { output.push("black".to_string()); }
670 else if l >= 95 { output.push("white".to_string()); }
671 else {
672 let lightness = if l <= 20 { "very dark" } else if l <= 40 { "dark" }
673 else if l <= 60 { "medium" } else if l <= 80 { "light" } else { "very light" };
674 if s < 10 { output.push(format!("{lightness} gray")); }
675 else {
676 const NAMES: [&str; 12] = ["red","orange","yellow","chartreuse","green","spring-green",
677 "cyan","azure","blue","violet","magenta","rose"];
678 let saturation = if s < 40 { "muted" } else if s < 80 { "moderate" } else { "vivid" };
679 output.push(format!("{} {} {}", lightness, saturation, NAMES[((h + 15) % 360 / 30) as usize]));
680 }
681 }
682 }
683 "PALETTE" | "KMEANS" => {
684 let colors = &args.colors[1..words.len()];
685 let means = if spec.name == "PALETTE" { palette(colors, a as usize) }
686 else {
687 let (iterations, result) = kmeans(colors, a as usize);
688 output.push(iterations.to_string());
689 result
690 };
691 for (color, size) in means { output.push(format!("{}:{}", hex(color), size)); }
692 }
693 "SORT" => {
694 let mut colors = args.colors[..words.len()].to_vec();
695 colors.sort_by_key(|&color| hsl(color));
696 for color in colors { output.push(hex(color)); }
697 }
698 "DITHER" => {
699 let mut error = [0; 3];
700 for color in &args.colors[..words.len()] {
701 let target: Color = std::array::from_fn(|j| (color[j] + error[j]).clamp(0, 255));
702 let mut best = 0;
703 for i in 1..16 {
704 if distance(target, BUILTINS[i].1) < distance(target, BUILTINS[best].1) { best = i; }
705 }
706 output.push(BUILTINS[best].0.to_string());
707 error = std::array::from_fn(|j| target[j] - BUILTINS[best].1[j]);
708 }
709 }
710 "RANDOM" => {
711 let mut state = a;
712 for _ in 0..b {
713 state = state * 48271 % 2147483647;
714 let h = state % 360;
715 state = state * 48271 % 2147483647;
716 let s = 40 + state % 61;
717 state = state * 48271 % 2147483647;
718 let l = 25 + state % 51;
719 output.push(hex(from_hsl(h, s, l)));
720 }
721 }
722 _ => unreachable!(),
723 }
724 Some(output.join(" "))
725}

731 let stdout = io::stdout();

738 }
