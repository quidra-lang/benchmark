26 func check(_ condition: Bool) throws {
27 if !condition { throw Invalid.command }
28 }
29 func arity(_ tokens: [String], _ count: Int) throws {
30 try check(tokens.count == count + 1)
31 }
32 func matches(_ text: String, _ pattern: String) -> Bool {
33 text.range(of: "\\A(?:" + pattern + ")\\z", options: .regularExpression) != nil
34 }
35 func number(_ text: String, _ low: Int, _ high: Int) throws -> Int {
36 try check(matches(text, "-?[0-9]{1,9}"))
37 let value = Int(text)!
38 try check(value >= low && value <= high)
39 return value
40 }
41 func floor(_ n: Int64, _ d: Int64) -> Int64 {
42 let quotient = n / d
43 return n < 0 && n % d != 0 ? quotient - 1 : quotient
44 }
45 func round(_ n: Int64, _ d: Int64) -> Int {
46 Int(floor(2*n+d,2*d))
47 }
48 func round(_ n: Int, _ d: Int) -> Int {
49 round(Int64(n),Int64(d))
50 }
51 func away(_ n: Int, _ d: Int) -> Int {
52 n < 0 ? -round(-n,d) : round(n,d)
53 }
54 func percent(_ a: Int, _ b: Int) -> Int { round(100*a,b) }
55 func mod(_ value: Int, _ modulus: Int) -> Int { (value % modulus + modulus) % modulus }
56 func clamp(_ value: Int) -> Int { min(255,max(0,value)) }
57 func root(_ numerator: Int64, _ scale: Int64) -> Int {
58 let value = numerator / scale
59 var low: Int64 = 0
60 var high = value + 1
61 while low + 1 < high {
62 let middle = (low + high) / 2
63 if middle * middle <= value { low = middle }
64 else { high = middle }
65 }
66 return Int(low)
67 }
68 func unpack(_ value: Int) -> [Int] { [value >> 16,(value >> 8) & 255,value & 255] }
69 func hex(_ color: [Int]) -> String {
70 String(format: "#%02X%02X%02X", locale: Locale(identifier: "en_US_POSIX"), color[0], color[1], color[2])
71 }
72 func table() -> [Entry] {
73 names.indices.map { Entry(name: names[$0],color: unpack(rgb[$0])) } + registered
74 }
75 func color(_ text: String) throws -> [Int] {
76 if matches(text,"#[0-9a-fA-F]{3}(?:[0-9a-fA-F]{3})?") {
77 var digits = String(text.dropFirst())
78 if digits.count == 3 { digits = digits.map { "\($0)\($0)" }.joined() }
79 return unpack(Int(digits,radix: 16)!)
80 }
81 if matches(text,#"rgb\(([0-9]{1,3}),([0-9]{1,3}),([0-9]{1,3})\)"#) {
82 let components = text.dropFirst(4).dropLast().split(separator: ",").map { Int($0)! }
83 try check(components.allSatisfy { $0 <= 255 })
84 return components
85 }
86 if let entry = table().first(where: { $0.name == text }) { return entry.color }
87 throw Invalid.command
88 }
89 func hue(_ c: [Int]) -> Int {
90 let largest = c.max()!, difference = largest - c.min()!
91 if difference == 0 { return 0 }
92 var numerator: Int
93 if largest == c[0] { numerator = 60*(c[1]-c[2]) }
94 else if largest == c[1] { numerator = 60*(c[2]-c[0])+120*difference }
95 else { numerator = 60*(c[0]-c[1])+240*difference }
96 if numerator < 0 { numerator += 360*difference }
97 return round(numerator,difference) % 360
98 }
99 func hsl(_ c: [Int]) -> [Int] {
100 let largest = c.max()!, smallest = c.min()!, difference = largest - smallest
101 let saturation = difference == 0 ? 0 : percent(difference,255-abs(largest+smallest-255))
102 return [hue(c),saturation,percent(largest+smallest,510)]
103 }
104 func fromHue(_ h: Int, _ chroma: Int, _ offset: Int, _ denominator: Int) -> [Int] {
105 let sector = h / 60, f = h % 60, x = sector % 2 == 0 ? f : 60-f
106 let factors = [[60,x,0],[x,60,0],[0,60,x],[0,x,60],[x,0,60],[60,0,x]]
107 return (0..<3).map { i in
108 round(255*(Int64(chroma)*Int64(factors[sector][i])+60*Int64(offset)),60*Int64(denominator))
109 }
110 }
111 func fromHsl(_ h: Int, _ s: Int, _ l: Int) -> [Int] {
112 let c = (100-abs(2*l-100))*s
113 return fromHue(h,2*c,200*l-c,20000)
114 }
115 func distance(_ a: [Int], _ b: [Int]) -> Int64 {
116 return (0..<3).reduce(Int64(0)) { sum,i in
117 let difference = Int64(a[i]-b[i])
118 return sum + difference*difference
119 }
120 }
121 func model(_ command: String, _ c: [Int]) -> String {
122 let r = c[0], g = c[1], b = c[2], m = c.max()!, n = c.min()!, d = m-n
123 switch command {
124 case "HSL":
125 let v = hsl(c)
126 return "hsl(\(v[0]),\(v[1])%,\(v[2])%)"
127 case "HSV": return "hsv(\(hue(c)),\(m == 0 ? 0 : percent(d,m))%,\(percent(m,255))%)"
128 case "HWB": return "hwb(\(hue(c)),\(percent(n,255))%,\(percent(255-m,255))%)"
129 case "CMYK":
130 if m == 0 { return "cmyk(0%,0%,0%,100%)" }
131 return "cmyk(\(percent(m-r,m))%,\(percent(m-g,m))%,\(percent(m-b,m))%,\(percent(255-m,255))%)"
132 case "YCBCR":
133 return "ycbcr(\(clamp(round(299*r+587*g+114*b,1000)))," +
134 "\(clamp(round(128000-169*r-331*g+500*b,1000))),\(clamp(round(128000+500*r-419*g-81*b,1000))))"
135 case "YIQ":
136 return "yiq(\(round(299*r+587*g+114*b,1000))," +
137 "\(round(596*r-274*g-322*b,1000)),\(round(211*r-523*g+312*b,1000)))"
138 default: preconditionFailure("Unknown internal color model")
139 }
140 }
141 func ansi(_ number: Int) -> [Int] {
142 if number < 16 { return unpack(rgb[number]) }
143 if number >= 232 { return [Int](repeating: 8+10*(number-232),count: 3) }
144 let values = [0,95,135,175,215,255], v = number-16
145 return [values[v/36],values[v/6%6],values[v%6]]
146 }
147 func gradient(_ a: [Int], _ b: [Int], _ index: Int, _ denominator: Int) -> [Int] {
148 (0..<3).map { a[$0]+away((b[$0]-a[$0])*index,denominator) }
149 }
150 func luminance(_ c: [Int]) -> Int64 {
151 2126*Int64(c[0])*Int64(c[0])+7152*Int64(c[1])*Int64(c[1])+722*Int64(c[2])*Int64(c[2])
152 }
153 func contrast(_ a: [Int], _ b: [Int]) -> [Int64] {
154 let first = luminance(a), second = luminance(b)
155 return [max(first,second)+32512500,min(first,second)+32512500]
156 }
157 func ratio(_ a: [Int], _ b: [Int]) -> Int {
158 let values = contrast(a,b)
159 return round(100*values[0],values[1])
160 }
161 func readable(_ a: [Int], _ b: [Int]) -> Bool {
162 let values = contrast(a,b)
163 return 2*values[0] >= 9*values[1]
164 }
165 func contrastText(_ a: [Int], _ b: [Int]) -> String {
166 let values = contrast(a,b), ratio = round(100*values[0],values[1])
167 let verdict = values[0] >= 7*values[1] ? "AAA" : 2*values[0] >= 9*values[1] ? "AA" : "FAIL"
168 return String(format: "%d.%02d", locale: Locale(identifier: "en_US_POSIX"), ratio / 100, ratio % 100) + " " + verdict
169 }
170 func blendComponent(_ mode: String, _ x: Int, _ y: Int) throws -> Int {
171 let multiply = x*y/255, screen = 255-(255-x)*(255-y)/255
172 switch mode {
173 case "MULTIPLY": return multiply
174 case "SCREEN": return screen
175 case "OVERLAY": return x < 128 ? 2*x*y/255 : 255-2*(255-x)*(255-y)/255
176 case "HARDLIGHT": return y < 128 ? 2*x*y/255 : 255-2*(255-x)*(255-y)/255
177 case "SOFTLIGHT": return ((255-x)*multiply+x*screen)/255
178 case "DARKEN": return min(x,y)
179 case "LIGHTEN": return max(x,y)
180 case "DIFFERENCE": return abs(x-y)
181 case "EXCLUSION": return x+y-2*x*y/255
182 case "ADD": return min(255,x+y)
183 case "SUBTRACT": return max(0,x-y)
184 case "COLORDODGE": return y == 255 ? 255 : min(255,255*x/(255-y))
185 case "COLORBURN": return y == 0 ? 0 : max(0,255-255*(255-x)/y)
186 default: throw Invalid.command
187 }
188 }
189 func blend(_ mode: String, _ base: [Int], _ top: [Int]) throws -> [Int] {
190 if ["HUE","SATURATION","COLOR","LUMINOSITY"].contains(mode) {
191 let a = hsl(base), b = hsl(top)
192 return fromHsl(mode == "HUE" || mode == "COLOR" ? b[0] : a[0],
193 mode == "SATURATION" || mode == "COLOR" ? b[1] : a[1],mode == "LUMINOSITY" ? b[2] : a[2])
194 }
195 return try (0..<3).map { try blendComponent(mode,base[$0],top[$0]) }
196 }
197 func mean(_ colors: [[Int]]) -> [Int] {
198 (0..<3).map { i in round(colors.reduce(0) { $0+$1[i] },colors.count) }
199 }
200 func ranges(_ colors: [[Int]]) -> [Int] {
201 return (0..<3).map { i in colors.map { $0[i] }.max()! - colors.map { $0[i] }.min()! }
202 }
203 func palette(_ k: Int, _ colors: [[Int]]) -> [Cluster] {
204 var boxes = [colors]
205 while boxes.count < k {
206 let selected = boxes.indices.filter { boxes[$0].count > 1 }.max {
207 ranges(boxes[$0]).max()! < ranges(boxes[$1]).max()!
208 }!
209 let range = ranges(boxes[selected])
210 let component = range.indices.max { range[$0] < range[$1] }!
211 let sorted = boxes[selected].sorted { $0[component] < $1[component] }
212 let middle = sorted.count / 2
213 boxes[selected] = Array(sorted.prefix(middle))
214 boxes.insert(Array(sorted.dropFirst(middle)),at: selected+1)
215 }
216 return boxes.map { Cluster(color: mean($0),count: $0.count) }
217 }
218 func clusterText(_ clusters: [Cluster]) -> String {
219 clusters.map { "\(hex($0.color)):\($0.count)" }.joined(separator: " ")
220 }
221 func kmeans(_ k: Int, _ colors: [[Int]]) -> String {
222 var representatives = palette(k,colors).map { $0.color }
223 var result = [Cluster](), iteration = 0
224 while iteration < 20 {
225 iteration += 1
226 var groups = [[[Int]]](repeating: [],count: k)
227 for color in colors {
228 let best = (0..<k).min { distance(color, representatives[$0]) < distance(color, representatives[$1]) }!
229 groups[best].append(color)
230 }
231 result = (0..<k).map { j in
232 Cluster(color: groups[j].isEmpty ? representatives[j] : mean(groups[j]),count: groups[j].count)
233 }
234 let unchanged = result.indices.allSatisfy { result[$0].color == representatives[$0] }
235 representatives = result.map { $0.color }
236 if unchanged { break }
237 }
238 return "\(iteration) \(clusterText(result))"
239 }
240 func colors(_ tokens: [String], _ start: Int) throws -> [[Int]] {
241 try tokens.dropFirst(start).map { try color($0) }
242 }
243 func colorsText(_ colors: [[Int]]) -> String { colors.map { hex($0) }.joined(separator: " ") }
244 func execute(_ line: String) throws -> String {
245 let t = line.split(separator: " ").map(String.init), command = t[0]
246 switch command {
247 case "DEFINE":
248 try arity(t,2)
249 try check(matches(t[1],"[a-z]{1,12}") && !names.contains(t[1]))
250 let c = try color(t[2])
251 let entry = Entry(name: t[1],color: c)
252 if let index = registered.firstIndex(where: { $0.name == t[1] }) { registered[index] = entry }
253 else { registered.append(entry) }
254 return "\(t[1]) \(hex(c))"
255 case "UNDEFINE":
256 try arity(t,1)
257 try check(matches(t[1],"[a-z]{1,12}"))
258 guard let index = registered.firstIndex(where: { $0.name == t[1] }) else { throw Invalid.command }
259 registered.remove(at: index)
260 return "\(t[1]) removed"
261 case "LIST":
262 try arity(t,0)
263 if registered.isEmpty { return "NONE" }
264 return registered.sorted { $0.name < $1.name }.map { "\($0.name)=\(hex($0.color))" }.joined(separator: " ")
265 case "PARSE":
266 try arity(t,1)
267 let c = try color(t[1]), name = table().first(where: { $0.color == c })?.name ?? "-"
268 return "\(hex(c)) rgb(\(c[0]),\(c[1]),\(c[2])) " +
269 "\(model("HSL",c)) \(model("HSV",c)) \(model("HWB",c)) \(model("CMYK",c)) " +
270 "\(model("YCBCR",c)) \(model("YIQ",c)) \(name)"
271 case "HSL", "HSV", "HWB", "CMYK", "YCBCR", "YIQ":
272 try arity(t,1)
273 return model(command,try color(t[1]))
274 case "FROMHSL", "FROMHSV", "FROMHWB":
275 try arity(t,3)
276 let h = try number(t[1],0,359), s = try number(t[2],0,100), third = try number(t[3],0,100)
277 if command == "FROMHSL" { return hex(fromHsl(h,s,third)) }
278 if command == "FROMHSV" { return hex(fromHue(h,third*s,100*third-third*s,10000)) }
279 if s+third >= 100 { return hex([Int](repeating: round(255*s,s+third),count: 3)) }
280 return hex(fromHue(h,100-s-third,s,100))
281 case "FROMCMYK":
282 try arity(t,4)
283 let k = try number(t[4],0,100)
284 let c = try (0..<3).map { round(255*(100-(try number(t[$0+1],0,100)))*(100-k),10000) }
285 return hex(c)
286 case "FROMYCBCR", "FROMYIQ":
287 try arity(t,3)
288 let y = try number(t[1],0,255), iq = command == "FROMYIQ"
289 let a = try number(t[2],iq ? -255 : 0,255), b = try number(t[3],iq ? -255 : 0,255)
290 let numerators: [Int]
291 if iq { numerators = [1000*y+956*a+619*b,1000*y-272*a-647*b,1000*y-1106*a+1703*b] }
292 else { numerators = [1000*y+1402*(b-128),1000*y-344*(a-128)-714*(b-128),1000*y+1772*(a-128)] }
293 return hex(numerators.map { clamp(round($0,1000)) })
294 case "RGB565":
295 try arity(t,1)
296 let c = try color(t[1])
297 let value = 2048*round(31*c[0],255)+32*round(63*c[1],255)+round(31*c[2],255)
298 return String(format: "0x%04X", locale: Locale(identifier: "en_US_POSIX"), value)
299 case "FROM565":
300 try arity(t,1)
301 try check(matches(t[1],"0x[0-9a-fA-F]{4}"))
302 let v = Int(t[1].dropFirst(2),radix: 16)!
303 return hex([round(255*(v/2048),31),round(255*(v/32%64),63),round(255*(v%32),31)])
304 case "FROMANSI":
305 try arity(t,1)
306 return hex(ansi(try number(t[1],0,255)))
307 case "ANSI":
308 try arity(t,1)
309 let c = try color(t[1])
310 let best = (16...255).min { distance(c, ansi($0)) < distance(c, ansi($1)) }!
311 return "\(best) \(hex(ansi(best))) \(distance(c,ansi(best)))"
312 case "WEBSAFE", "INVERT":
313 try arity(t,1)
314 let c = try color(t[1])
315 return hex(c.map { command == "INVERT" ? 255-$0 : 51*round($0,51) })
316 case "GRAY":
317 try arity(t,2)
318 let c = try color(t[1])
319 let value: Int
320 switch t[2] {
321 case "LUMA": value = round(299*c[0]+587*c[1]+114*c[2],1000)
322 case "AVERAGE": value = round(c.reduce(0,+),3)
323 case "LIGHTNESS": value = round(c.max()!+c.min()!,2)
324 case "HSP": value = root(Int64(299*c[0]*c[0]+587*c[1]*c[1]+114*c[2]*c[2]),1000)
325 default: throw Invalid.command
326 }
327 return hex([Int](repeating: value,count: 3))
328 case "MIX", "MIXHSL":
329 try arity(t,3)
330 let a = try color(t[1]), b = try color(t[2]), p = try number(t[3],0,100)
331 if command == "MIX" { return hex((0..<3).map { round(a[$0]*(100-p)+b[$0]*p,100) }) }
332 let first = hsl(a), second = hsl(b)
333 var e = mod(second[0]-first[0],360)
334 if e > 180 { e -= 360 }
335 return hex(fromHsl(mod(first[0]+away(e*p,100),360),
336 first[1]+away((second[1]-first[1])*p,100),first[2]+away((second[2]-first[2])*p,100)))
337 case "GRADIENT", "TINTS", "SHADES", "TONES":
338 try arity(t,command == "GRADIENT" ? 3 : 2)
339 let a = try color(t[1])
340 if command == "GRADIENT" {
341 let b = try color(t[2]), n = try number(t[3],2,16)
342 return colorsText((0..<n).map { gradient(a,b,$0,n-1) })
343 }
344 let n = try number(t[2],1,10), target = command == "TINTS" ? 255 : command == "SHADES" ? 0 : 128
345 let b = [Int](repeating: target,count: 3)
346 return colorsText((1...n).map { gradient(a,b,$0,n+1) })
347 case "BLEND":
348 try arity(t,3)
349 let base = try color(t[2]), top = try color(t[3])
350 return hex(try blend(t[1],base,top))
351 case "ALPHA":
352 try arity(t,4)
353 let fg = try color(t[1]), p = try number(t[2],0,100)
354 let bg = try color(t[3]), q = try number(t[4],0,100), a = 100*p+q*(100-p)
355 let result = (0..<3).map { a == 0 ? 0 : round(100*p*fg[$0]+q*(100-p)*bg[$0],a) }
356 return "\(hex(result)) \(round(a,100))%"
357 case "SIMULATE":
358 try arity(t,2)
359 let c = try color(t[1])
360 let coefficients: [[Int]]
361 switch t[2] {
362 case "PROTANOPIA": coefficients = [[567,433,0],[558,442,0],[0,242,758]]
363 case "DEUTERANOPIA": coefficients = [[625,375,0],[700,300,0],[0,300,700]]
364 case "TRITANOPIA": coefficients = [[950,50,0],[0,433,567],[0,475,525]]
365 case "ACHROMATOPSIA": coefficients = [[299,587,114],[299,587,114],[299,587,114]]
366 default: throw Invalid.command
367 }
368 return hex((0..<3).map { i in round((0..<3).reduce(0) { $0+coefficients[i][$1]*c[$1] },1000) })
369 case "ROTATE", "COMPLEMENT", "TRIADIC", "TETRADIC", "SPLITCOMP", "ANALOGOUS":
370 try arity(t,command == "ROTATE" ? 2 : command == "ANALOGOUS" ? 3 : 1)
371 let values = hsl(try color(t[1]))
372 let rotations: [Int]
373 switch command {
374 case "ROTATE": rotations = [try number(t[2],-720,720)]
375 case "COMPLEMENT": rotations = [180]
376 case "TRIADIC": rotations = [120,240]
377 case "TETRADIC": rotations = [90,180,270]
378 case "SPLITCOMP": rotations = [150,210]
379 default:
380 let k = try number(t[2],1,6), step = try number(t[3],1,60)
381 rotations = (-k...k).filter { $0 != 0 }.map { $0*step }
382 }
383 return colorsText(rotations.map { fromHsl(mod(values[0]+$0,360),values[1],values[2]) })
384 case "LIGHTEN", "DARKEN", "SATURATE", "DESATURATE":
385 try arity(t,2)
386 var values = hsl(try color(t[1]))
387 let p = try number(t[2],0,100)
388 switch command {
389 case "LIGHTEN": values[2] = min(100,values[2]+p)
390 case "DARKEN": values[2] = max(0,values[2]-p)
391 case "SATURATE": values[1] = min(100,values[1]+p)
392 default: values[1] = max(0,values[1]-p)
393 }
394 return hex(fromHsl(values[0],values[1],values[2]))
395 case "NEAREST":
396 try arity(t,1)
397 let c = try color(t[1])
398 let best = table().min { distance(c, $0.color) < distance(c, $1.color) }!
399 return "\(best.name) \(hex(best.color)) \(distance(c,best.color))"
400 case "DISTANCE":
401 try arity(t,2)
402 let a = try color(t[1]), b = try color(t[2])
403 let dr = Int64(a[0]-b[0]), dg = Int64(a[1]-b[1]), db = Int64(a[2]-b[2])
404 let w = (1024+Int64(a[0]+b[0]))*dr*dr+2048*dg*dg+(1534-Int64(a[0]+b[0]))*db*db
405 return "\(distance(a,b)) \(root(w,512))"
406 case "CONTRAST":
407 try arity(t,2)
408 return contrastText(try color(t[1]),try color(t[2]))
409 case "ACCESSIBLE":
410 try arity(t,2)
411 let fg = try color(t[1]), bg = try color(t[2])
412 if readable(fg,bg) { return "\(hex(fg)) \(contrastText(fg,bg))" }
413 let values = hsl(fg)
414 for d in 1...100 {
415 for l in [values[2]-d,values[2]+d] {
416 if l < 0 || l > 100 { continue }
417 let candidate = fromHsl(values[0],values[1],l)
418 if readable(candidate,bg) { return "\(hex(candidate)) \(contrastText(candidate,bg))" }
419 }
420 }
421 throw Invalid.command
422 case "BEST":
423 try check(t.count >= 3 && t.count <= 34)
424 let bg = try color(t[1]), options = try colors(t,2)
425 let best = options.max { ratio($0, bg) < ratio($1, bg) }!
426 return "\(hex(best)) \(contrastText(best,bg))"
427 case "DESCRIBE":
428 try arity(t,1)
429 let values = hsl(try color(t[1])), h = values[0], s = values[1], l = values[2]
430 if l <= 5 { return "black" }
431 if l >= 95 { return "white" }
432 let lightness = l <= 20 ? "very dark" : l <= 40 ? "dark" : l <= 60 ? "medium" : l <= 80 ? "light" : "very light"
433 if s < 10 { return lightness + " gray" }
434 let saturation = s < 40 ? "muted" : s < 80 ? "moderate" : "vivid"
435 let hues = ["red","orange","yellow","chartreuse","green","spring-green",
436 "cyan","azure","blue","violet","magenta","rose"]
437 return lightness + " " + saturation + " " + hues[(h+15)%360/30]
438 case "PALETTE", "KMEANS":
439 try check(t.count >= 3 && t.count <= 34)
440 let k = try number(t[1],1,t.count-2), options = try colors(t,2)
441 return command == "PALETTE" ? clusterText(palette(k,options)) : kmeans(k,options)
442 case "SORT":
