240static void extrema(Color c, Int *maximum, Int *minimum) {
241 *maximum = std::max({c.x[0], c.x[1], c.x[2]});
242 *minimum = std::min({c.x[0], c.x[1], c.x[2]});
243}

245static Int hue(Color c) {
246 Int maximum, minimum;
247 extrema(c, &maximum, &minimum);
248 Int delta = maximum - minimum;
249 if (delta == 0) return 0;
250 Int numerator;
251 if (maximum == c.x[0]) {
252 numerator = 60 * (c.x[1] - c.x[2]);
253 if (numerator < 0) numerator += 360 * delta;
254 } else if (maximum == c.x[1]) {
255 numerator = 60 * (c.x[2] - c.x[0]) + 120 * delta;
256 } else {
257 numerator = 60 * (c.x[0] - c.x[1]) + 240 * delta;
258 }
259 return rounded(numerator, delta) % 360;
260}

262static Hsl hsl(Color c) {
263 Int maximum, minimum;
264 extrema(c, &maximum, &minimum);
265 Int delta = maximum - minimum;
266 Hsl result = {hue(c), 0, rounded(100 * (maximum + minimum), 510)};
267 if (delta != 0)
268 result.s = rounded(100 * delta, 255 - std::abs(maximum + minimum - 255));
269 return result;
270}

272static Color sector(Int h) {
273 Int q = h / 60;
274 Int f = h % 60;
275 Int x = q % 2 == 0 ? f : 60 - f;
276 switch (q) {
277 case 0: return rgb(60, x, 0);
278 case 1: return rgb(x, 60, 0);
279 case 2: return rgb(0, 60, x);
280 case 3: return rgb(0, x, 60);
281 case 4: return rgb(x, 0, 60);
282 default: return rgb(60, 0, x);
283 }
284}

286static Color from_hsl(Int h, Int s, Int l) {
287 Int chroma = (100 - std::abs(2 * l - 100)) * s;
288 Int offset = 30 * (200 * l - chroma);
289 Color result = sector(h);
290 for (int j = 0; j < 3; ++j)
291 result.x[j] = rounded(255 * (result.x[j] * chroma + offset), 600000);
292 return result;
293}

295static Color from_hsv(Int h, Int s, Int v) {
296 Int chroma = v * s;
297 Int offset = 60 * (100 * v - chroma);
298 Color result = sector(h);
299 for (int j = 0; j < 3; ++j)
300 result.x[j] = rounded(255 * (result.x[j] * chroma + offset), 600000);
301 return result;
302}

304static Color from_hwb(Int h, Int w, Int k) {
305 if (w + k >= 100) {
306 Int value = rounded(255 * w, w + k);
307 return rgb(value, value, value);
308 }
309 Color result = sector(h);
310 for (int j = 0; j < 3; ++j)
311 result.x[j] = rounded(255 * (result.x[j] * (100 - w - k) + 60 * w), 6000);
312 return result;
313}

315static Int integer_root(Int value) {
316 Int low = 0, high = 1;
317 while (high * high <= value) high *= 2;
318 while (low + 1 < high) {
319 Int middle = (low + high) / 2;
320 if (middle * middle <= value) low = middle;
321 else high = middle;
322 }
323 return low;
324}

337static Int luminance(Color c) {
338 return 2126 * c.x[0] * c.x[0] + 7152 * c.x[1] * c.x[1]
339 + 722 * c.x[2] * c.x[2];

342static Contrast contrast(Color a, Color b) {
343 Int y1 = luminance(a), y2 = luminance(b);
344 Int high = std::max(y1, y2) + 32512500;
345 Int low = std::min(y1, y2) + 32512500;
346 Contrast result = {rounded(100 * high, low), "FAIL"};
347 if (high >= 7 * low) result.verdict = "AAA";
348 else if (2 * high >= 9 * low) result.verdict = "AA";
349 return result;
350}

353 Color result = first;

364static Color blend(int mode, Color base, Color top) {
365 if (mode >= 13) {
366 Hsl a = hsl(base), b = hsl(top);
367 if (mode == 13 || mode == 15) a.h = b.h;
368 if (mode == 14 || mode == 15) a.s = b.s;
369 if (mode == 16) a.l = b.l;
370 return from_hsl(a.h, a.s, a.l);
371 }
372 Color result = base;
373 for (int j = 0; j < 3; ++j) {
374 Int x = base.x[j], y = top.x[j], value = 0;
375 switch (mode) {
376 case 0: value = x * y / 255; break;
377 case 1: value = 255 - (255 - x) * (255 - y) / 255; break;
378 case 2:
379 case 3:
380 if ((mode == 2 ? x : y) < 128) value = 2 * x * y / 255;
381 else value = 255 - 2 * (255 - x) * (255 - y) / 255;
382 break;
383 case 4: {
384 Int multiply = x * y / 255;
385 Int screen = 255 - (255 - x) * (255 - y) / 255;
386 value = ((255 - x) * multiply + x * screen) / 255;
387 break;
388 }
389 case 5: value = std::min(x, y); break;
390 case 6: value = std::max(x, y); break;
391 case 7: value = std::abs(x - y); break;
392 case 8: value = x + y - 2 * x * y / 255; break;
393 case 9: value = std::min<Int>(255, x + y); break;
394 case 10: value = std::max<Int>(0, x - y); break;
395 case 11: value = y == 255 ? 255 : std::min<Int>(255, 255 * x / (255 - y)); break;
396 case 12: value = y == 0 ? 0 : std::max<Int>(0, 255 - 255 * (255 - x) / y); break;
397 }
398 result.x[j] = value;
399 }
400 return result;
401}

403static void palette(const Color *colors, int count, int wanted, Color *means, int *sizes) {
404 Box boxes[32]{};
405 boxes[0].count = count;
406 std::copy_n(colors, count, boxes[0].colors.begin());
407 int box_count = 1;
408 while (box_count < wanted) {
409 int chosen = -1, channel = 0;
410 Int best_width = -1;
411 for (int i = 0; i < box_count; ++i) {
412 if (boxes[i].count < 2) continue;
413 Int width = -1;
414 int component = 0;
415 for (int j = 0; j < 3; ++j) {
416 Int minimum = 255, maximum = 0;
417 for (int p = 0; p < boxes[i].count; ++p) {
418 minimum = std::min(minimum, boxes[i].colors[p].x[j]);
419 maximum = std::max(maximum, boxes[i].colors[p].x[j]);
420 }
421 if (maximum - minimum > width) {
422 width = maximum - minimum;
423 component = j;
424 }
425 }
426 if (width > best_width) {
427 chosen = i;
428 channel = component;
429 best_width = width;
430 }
431 }
432 Box sorted = boxes[chosen];
433 std::stable_sort(sorted.colors.begin(), sorted.colors.begin() + sorted.count,
434 [channel](Color a, Color b) { return a.x[channel] < b.x[channel]; });
435 for (int i = box_count; i > chosen + 1; --i) boxes[i] = boxes[i - 1];
436 int middle = sorted.count / 2;
437 boxes[chosen].count = middle;
438 boxes[chosen + 1].count = sorted.count - middle;
439 for (int i = 0; i < middle; ++i) boxes[chosen].colors[i] = sorted.colors[i];
440 for (int i = middle; i < sorted.count; ++i)
441 boxes[chosen + 1].colors[i - middle] = sorted.colors[i];
442 ++box_count;
443 }
444 for (int i = 0; i < wanted; ++i) {
445 sizes[i] = boxes[i].count;
446 means[i] = rgb(0, 0, 0);
447 for (int p = 0; p < boxes[i].count; ++p)
448 for (int j = 0; j < 3; ++j) means[i].x[j] += boxes[i].colors[p].x[j];
449 for (int j = 0; j < 3; ++j) means[i].x[j] = rounded(means[i].x[j], sizes[i]);
450 }
451}

453static int kmeans(const Color *colors, int count, int wanted, Color *means, int *sizes) {
454 palette(colors, count, wanted, means, sizes);
455 for (int iteration = 1; iteration <= 20; ++iteration) {
456 Color sums[32];
457 for (int j = 0; j < wanted; ++j) {
458 sums[j] = rgb(0, 0, 0);
459 sizes[j] = 0;
460 }
461 for (int i = 0; i < count; ++i) {
462 int closest = 0;
463 for (int j = 1; j < wanted; ++j)
464 if (distance(colors[i], means[j]) < distance(colors[i], means[closest])) closest = j;
465 ++sizes[closest];
466 for (int j = 0; j < 3; ++j) sums[closest].x[j] += colors[i].x[j];
467 }
468 int unchanged = 1;
469 for (int i = 0; i < wanted; ++i) {
470 if (sizes[i] == 0) continue;
471 for (int j = 0; j < 3; ++j) sums[i].x[j] = rounded(sums[i].x[j], sizes[i]);
472 if (!equal_color(means[i], sums[i])) unchanged = 0;
473 means[i] = sums[i];
474 }
475 if (unchanged || iteration == 20) return iteration;
476 }
477 return 20;
478}

480static int hex_digit(char c) {
481 if (c >= '0' && c <= '9') return c - '0';
482 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
483 if (c >= 'A' && c <= 'F') return c - 'A' + 10;
484 return -1;
485}

487static int parse_integer(const char *text, Int low, Int high, Int *value) {
488 int negative = text[0] == '-';
489 int digits = 0;
490 Int result = 0;
491 for (int i = negative; text[i]; ++i) {
492 if (text[i] < '0' || text[i] > '9' || ++digits > 9) return 0;
493 result = result * 10 + text[i] - '0';
494 }
495 if (digits == 0) return 0;
496 if (negative) result = -result;
497 if (result < low || result > high) return 0;
498 *value = result;
499 return 1;
500}

520static int parse_color(const State *state, const char *text, Color *result) {
521 size_t length = strlen(text);
522 if (text[0] == '#') {
523 if (length != 4 && length != 7) return 0;
524 for (size_t i = 1; i < length; ++i) if (hex_digit(text[i]) < 0) return 0;
525 for (int j = 0; j < 3; ++j) {
526 if (length == 4) result->x[j] = 17 * hex_digit(text[j + 1]);
527 else result->x[j] = 16 * hex_digit(text[2 * j + 1]) + hex_digit(text[2 * j + 2]);
528 }
529 return 1;
530 }
531 if (length >= 4 && strncmp(text, "rgb(", 4) == 0) {
532 size_t position = 4;
533 for (int j = 0; j < 3; ++j) {
534 Int value = 0;
535 int digits = 0;
536 while (text[position] >= '0' && text[position] <= '9') {
537 if (++digits > 3) return 0;
538 value = value * 10 + text[position++] - '0';
539 }
540 if (digits == 0 || value > 255) return 0;
541 char expected = j == 2 ? ')' : ',';
542 if (text[position] != expected) return 0;
543 ++position;
544 result->x[j] = value;
545 }
546 return text[position] == '\0';
547 }
548 for (int i = 0; i < 24 + registered_count(state); ++i) {
549 const Entry *entry = table_entry(state, i);
550 if (entry->name == text) {
551 *result = entry->color;
552 return 1;
553 }
554 }
555 return 0;
556}

558static int keyword_index(const char *word, const char *choices) {
559 const char *start = choices;
560 int index = 0;
561 while (*start) {
562 const char *end = strchr(start, '|');
563 size_t length = end ? (size_t)(end - start) : strlen(start);
564 if (strlen(word) == length && strncmp(word, start, length) == 0) return index;
565 if (!end) break;
566 start = end + 1;
567 ++index;
568 }
569 return -1;
570}

572static int prepare(const State *state, const Spec *spec, int operation,
573 const char **words, int count, Arguments *arguments) {
574 if (count < spec->minimum || count > spec->maximum) return 0;
575 for (int i = 0; i < count; ++i) {
576 char kind;
577 if (spec->signature[0] == '*') kind = 'C';
578 else if (spec->signature[0] == 'P') kind = i == 0 ? 'I' : 'C';
579 else kind = spec->signature[i];
580 if (kind == 'C') {
581 if (!parse_color(state, words[i], &arguments->colors[i])) return 0;
582 } else if (kind == 'I') {
583 Int low = spec->signature[0] == 'P' ? 1 : spec->low[i];
584 Int high = spec->signature[0] == 'P' ? 32 : spec->high[i];
585 if (!parse_integer(words[i], low, high, &arguments->numbers[i])) return 0;
586 } else if (kind == 'N') {
587 if (!valid_name(words[i])) return 0;
588 } else if (kind == 'K') {
589 arguments->keyword = keyword_index(words[i], spec->keywords);
590 if (arguments->keyword < 0) return 0;
591 } else if (kind == 'W') {
592 if (strlen(words[i]) != 6 || strncmp(words[i], "0x", 2) != 0) return 0;
593 Int value = 0;
594 for (int j = 2; j < 6; ++j) {
595 int digit = hex_digit(words[i][j]);
596 if (digit < 0) return 0;
597 value = 16 * value + digit;
598 }
599 arguments->numbers[i] = value;
600 }
601 }
602 if (spec->signature[0] == 'P' && arguments->numbers[0] > count - 1) return 0;
603 if (operation == OP_DEFINE) {
604 for (int i = 0; i < 24; ++i)
605 if (builtins[i].name == words[0]) return 0;
606 }
607 if (operation == OP_UNDEFINE && registry_find(state, words[0]) < 0) return 0;
608 return 1;
609}

611static void color_text(Color c, char *text) {
612 snprintf(text, 8, "#%02X%02X%02X", (unsigned)c.x[0], (unsigned)c.x[1], (unsigned)c.x[2]);
613}

615static void output_color(Text *output, Color c) {
616 char text[8];
617 color_text(c, text);
618 wordf(output, "%s", text);
619}

621static void output_contrast(Text *output, Color c, Color background) {
622 Contrast value = contrast(c, background);
623 wordf(output, "%lld.%02lld %s", value.ratio / 100, value.ratio % 100, value.verdict);
624}

626static void output_model(Text *output, int operation, Color c) {
627 Int maximum, minimum;
628 extrema(c, &maximum, &minimum);
629 Int r = c.x[0], g = c.x[1], b = c.x[2];
630 if (operation == OP_HSL) {
631 Hsl value = hsl(c);
632 wordf(output, "hsl(%lld,%lld%%,%lld%%)", value.h, value.s, value.l);
633 } else if (operation == OP_HSV) {
634 wordf(output, "hsv(%lld,%lld%%,%lld%%)", hue(c),
635 maximum == 0 ? 0 : rounded(100 * (maximum - minimum), maximum), rounded(100 * maximum, 255));
636 } else if (operation == OP_HWB) {
637 wordf(output, "hwb(%lld,%lld%%,%lld%%)", hue(c), rounded(100 * minimum, 255),
638 rounded(100 * (255 - maximum), 255));
639 } else if (operation == OP_CMYK) {
640 wordf(output, "cmyk(%lld%%,%lld%%,%lld%%,%lld%%)",
641 maximum == 0 ? 0 : rounded(100 * (maximum - r), maximum),
642 maximum == 0 ? 0 : rounded(100 * (maximum - g), maximum),
643 maximum == 0 ? 0 : rounded(100 * (maximum - b), maximum),
644 rounded(100 * (255 - maximum), 255));
645 } else if (operation == OP_YCBCR) {
646 wordf(output, "ycbcr(%lld,%lld,%lld)", clamp(rounded(299 * r + 587 * g + 114 * b, 1000)),
647 clamp(rounded(128000 - 169 * r - 331 * g + 500 * b, 1000)),
648 clamp(rounded(128000 + 500 * r - 419 * g - 81 * b, 1000)));
649 } else {
650 wordf(output, "yiq(%lld,%lld,%lld)", rounded(299 * r + 587 * g + 114 * b, 1000),
651 rounded(596 * r - 274 * g - 322 * b, 1000), rounded(211 * r - 523 * g + 312 * b, 1000));
652 }
653}

655static void execute(State *state, int operation, const char **words, int count,
656 const Arguments *args, Text *output) {
657 Color c = args->colors[0];
658 Int a = args->numbers[0], b = args->numbers[1], d = args->numbers[2];
659 switch (operation) {
660 case OP_DEFINE:
661 wordf(output, "%s", words[0]);
662 output_color(output, args->colors[1]);
663 registry_set(state, words[0], args->colors[1]);
664 break;
665 case OP_UNDEFINE:
666 wordf(output, "%s removed", words[0]);
667 registry_delete(state, registry_find(state, words[0]));
668 break;
669 case OP_LIST: {
670 std::vector<const Entry *> sorted;
671 for (const Entry &entry : state->entries) sorted.push_back(&entry);
672 std::sort(sorted.begin(), sorted.end(), [](const Entry *a, const Entry *b) {
673 return a->name < b->name;
674 });
675 if (sorted.empty()) wordf(output, "NONE");
676 for (const Entry *entry : sorted) {
677 char text[8];
678 color_text(entry->color, text);
679 wordf(output, "%s=%s", entry->name.c_str(), text);
680 }
681 break;
682 }
683 case OP_PARSE: {
684 output_color(output, c);
685 wordf(output, "rgb(%lld,%lld,%lld)", c.x[0], c.x[1], c.x[2]);
686 const int models[] = {OP_HSL, OP_HSV, OP_HWB, OP_CMYK, OP_YCBCR, OP_YIQ};
687 for (int i = 0; i < 6; ++i) output_model(output, models[i], c);
688 const char *name = "-";
689 for (int i = 0; i < 24 + registered_count(state); ++i) {
690 const Entry *entry = table_entry(state, i);
691 if (equal_color(c, entry->color)) {
692 name = entry->name.c_str();
693 break;
694 }
695 }
696 wordf(output, "%s", name);
697 break;
698 }
699 case OP_HSL: case OP_HSV: case OP_HWB: case OP_CMYK: case OP_YCBCR: case OP_YIQ:
700 output_model(output, operation, c);
701 break;
702 case OP_FROMHSL: output_color(output, from_hsl(a, b, d)); break;
703 case OP_FROMHSV: output_color(output, from_hsv(a, b, d)); break;
704 case OP_FROMHWB: output_color(output, from_hwb(a, b, d)); break;
705 case OP_FROMCMYK: {
706 Color result = rgb(0, 0, 0);
707 for (int j = 0; j < 3; ++j)
708 result.x[j] = rounded(255 * (100 - args->numbers[j]) * (100 - args->numbers[3]), 10000);
709 output_color(output, result);
710 break;
711 }
712 case OP_FROMYCBCR:
713 output_color(output, rgb(clamp(rounded(1000 * a + 1402 * (d - 128), 1000)),
714 clamp(rounded(1000 * a - 344 * (b - 128) - 714 * (d - 128), 1000)),
715 clamp(rounded(1000 * a + 1772 * (b - 128), 1000))));
716 break;
717 case OP_FROMYIQ:
718 output_color(output, rgb(clamp(rounded(1000 * a + 956 * b + 619 * d, 1000)),
719 clamp(rounded(1000 * a - 272 * b - 647 * d, 1000)),
720 clamp(rounded(1000 * a - 1106 * b + 1703 * d, 1000))));
721 break;
722 case OP_RGB565: {
723 Int code = 2048 * rounded(31 * c.x[0], 255) + 32 * rounded(63 * c.x[1], 255)
724 + rounded(31 * c.x[2], 255);
725 wordf(output, "0x%04X", (unsigned)code);
726 break;
727 }
728 case OP_FROM565:
729 output_color(output, rgb(rounded(255 * (a / 2048), 31),
730 rounded(255 * ((a / 32) % 64), 63), rounded(255 * (a % 32), 31)));
731 break;
732 case OP_FROMANSI: output_color(output, ansi_color((int)a)); break;
733 case OP_ANSI: {
734 int best = 16;
735 for (int i = 17; i <= 255; ++i)
736 if (distance(c, ansi_color(i)) < distance(c, ansi_color(best))) best = i;
737 wordf(output, "%d", best);
738 output_color(output, ansi_color(best));
739 wordf(output, "%lld", distance(c, ansi_color(best)));
740 break;
741 }
742 case OP_WEBSAFE: {
743 Color result = c;
744 for (int j = 0; j < 3; ++j) result.x[j] = 51 * rounded(c.x[j], 51);
745 output_color(output, result);
746 break;
747 }
748 case OP_GRAY: {
749 Int maximum, minimum, value;
750 extrema(c, &maximum, &minimum);
751 if (args->keyword == 0) value = rounded(299 * c.x[0] + 587 * c.x[1] + 114 * c.x[2], 1000);
752 else if (args->keyword == 1) value = rounded(c.x[0] + c.x[1] + c.x[2], 3);
753 else if (args->keyword == 2) value = rounded(maximum + minimum, 2);
754 else value = integer_root((299 * c.x[0] * c.x[0] + 587 * c.x[1] * c.x[1]
755 + 114 * c.x[2] * c.x[2]) / 1000);
756 output_color(output, rgb(value, value, value));
757 break;
758 }
759 case OP_INVERT:
760 output_color(output, rgb(255 - c.x[0], 255 - c.x[1], 255 - c.x[2]));
761 break;
762 case OP_MIX: {
763 Color result = c;
764 for (int j = 0; j < 3; ++j)
765 result.x[j] = rounded(c.x[j] * (100 - d) + args->colors[1].x[j] * d, 100);
766 output_color(output, result);
767 break;
768 }
769 case OP_MIXHSL: {
770 Hsl first = hsl(c), last = hsl(args->colors[1]);
771 Int delta = modulo(last.h - first.h, 360);
772 if (delta > 180) delta -= 360;
773 output_color(output, from_hsl(modulo(first.h + away(delta * d, 100), 360),
774 first.s + away((last.s - first.s) * d, 100), first.l + away((last.l - first.l) * d, 100)));
775 break;
776 }
777 case OP_GRADIENT:
778 for (Int i = 0; i < d; ++i) output_color(output, interpolate(c, args->colors[1], i, d - 1));
779 break;
780 case OP_TINTS: case OP_SHADES: case OP_TONES: {
781 Int target = operation == OP_TINTS ? 255 : operation == OP_SHADES ? 0 : 128;
782 for (Int i = 1; i <= b; ++i) output_color(output, interpolate(c, rgb(target, target, target), i, b + 1));
783 break;
784 }
785 case OP_BLEND: output_color(output, blend(args->keyword, args->colors[1], args->colors[2])); break;
786 case OP_ALPHA: {
787 Int q = args->numbers[3];
788 Int alpha = 100 * b + q * (100 - b);
789 Color result = rgb(0, 0, 0);
790 if (alpha > 0)
791 for (int j = 0; j < 3; ++j)
792 result.x[j] = rounded(100 * b * c.x[j] + q * (100 - b) * args->colors[2].x[j], alpha);
793 output_color(output, result);
794 wordf(output, "%lld%%", rounded(alpha, 100));
795 break;
796 }
797 case OP_SIMULATE: {
798 static const Int coefficients[4][3][3] = {
799 {{567,433,0},{558,442,0},{0,242,758}},
800 {{625,375,0},{700,300,0},{0,300,700}},
801 {{950,50,0},{0,433,567},{0,475,525}},
802 {{299,587,114},{299,587,114},{299,587,114}}
803 };
804 Color result = rgb(0, 0, 0);
805 for (int j = 0; j < 3; ++j) {
806 Int sum = 0;
807 for (int k = 0; k < 3; ++k) sum += coefficients[args->keyword][j][k] * c.x[k];
808 result.x[j] = rounded(sum, 1000);
809 }
810 output_color(output, result);
811 break;
812 }
813 case OP_ROTATE: output_color(output, rotate(c, b)); break;
814 case OP_COMPLEMENT: output_color(output, rotate(c, 180)); break;
815 case OP_TRIADIC: output_color(output, rotate(c, 120)); output_color(output, rotate(c, 240)); break;
816 case OP_TETRADIC:
817 for (Int shift = 90; shift <= 270; shift += 90) output_color(output, rotate(c, shift));
818 break;
819 case OP_SPLITCOMP: output_color(output, rotate(c, 150)); output_color(output, rotate(c, 210)); break;
820 case OP_ANALOGOUS:
821 for (Int i = -b; i <= b; ++i) if (i != 0) output_color(output, rotate(c, i * d));
822 break;
823 case OP_LIGHTEN: case OP_DARKEN: case OP_SATURATE: case OP_DESATURATE: {
824 Hsl value = hsl(c);
825 if (operation == OP_LIGHTEN) value.l = std::min<Int>(100, value.l + b);
826 else if (operation == OP_DARKEN) value.l = std::max<Int>(0, value.l - b);
827 else if (operation == OP_SATURATE) value.s = std::min<Int>(100, value.s + b);
828 else value.s = std::max<Int>(0, value.s - b);
829 output_color(output, from_hsl(value.h, value.s, value.l));
830 break;
831 }
832 case OP_NEAREST: {
833 int best = 0;
834 for (int i = 1; i < 24 + registered_count(state); ++i)
835 if (distance(c, table_entry(state, i)->color) < distance(c, table_entry(state, best)->color)) best = i;
836 const Entry *entry = table_entry(state, best);
837 wordf(output, "%s", entry->name.c_str());
838 output_color(output, entry->color);
839 wordf(output, "%lld", distance(c, entry->color));
840 break;
841 }
842 case OP_DISTANCE: {
843 Color other = args->colors[1];
844 Int r = c.x[0] - other.x[0], g = c.x[1] - other.x[1], blue = c.x[2] - other.x[2];
845 Int weight = (1024 + c.x[0] + other.x[0]) * r * r + 2048 * g * g
846 + (1534 - c.x[0] - other.x[0]) * blue * blue;
847 wordf(output, "%lld %lld", distance(c, other), integer_root(weight / 512));
848 break;
849 }
850 case OP_CONTRAST: output_contrast(output, c, args->colors[1]); break;
851 case OP_ACCESSIBLE: {
852 Color result = c, background = args->colors[1];
853 if (strcmp(contrast(c, background).verdict, "FAIL") == 0) {
854 Hsl value = hsl(c);
855 int found = 0;
856 for (Int step = 1; step <= 100 && !found; ++step)
857 for (int sign = -1; sign <= 1; sign += 2) {
858 Int lightness = value.l + sign * step;
859 if (lightness < 0 || lightness > 100) continue;
860 Color candidate = from_hsl(value.h, value.s, lightness);
861 if (strcmp(contrast(candidate, background).verdict, "FAIL") != 0) {
862 result = candidate;
863 found = 1;
864 break;
865 }
866 }
867 }
868 output_color(output, result);
869 output_contrast(output, result, background);
870 break;
871 }
872 case OP_BEST: {
873 int best = 1;
874 for (int i = 2; i < count; ++i)
875 if (contrast(args->colors[i], c).ratio > contrast(args->colors[best], c).ratio) best = i;
876 output_color(output, args->colors[best]);
877 output_contrast(output, args->colors[best], c);
878 break;
879 }
880 case OP_DESCRIBE: {
881 Hsl value = hsl(c);
882 if (value.l <= 5) wordf(output, "black");
883 else if (value.l >= 95) wordf(output, "white");
884 else {
885 const char *lightness = value.l <= 20 ? "very dark" : value.l <= 40 ? "dark"
886 : value.l <= 60 ? "medium" : value.l <= 80 ? "light" : "very light";
887 wordf(output, "%s", lightness);
888 if (value.s < 10) wordf(output, "gray");
889 else {
890 static const char *names[] = {"red","orange","yellow","chartreuse","green","spring-green",
891 "cyan","azure","blue","violet","magenta","rose"};
892 wordf(output, "%s %s", value.s < 40 ? "muted" : value.s < 80 ? "moderate" : "vivid",
893 names[((value.h + 15) % 360) / 30]);
894 }
895 }
896 break;
897 }
898 case OP_PALETTE: case OP_KMEANS: {
899 Color means[32];
900 int sizes[32], wanted = (int)a;
901 if (operation == OP_PALETTE) palette(args->colors.data() + 1, count - 1, wanted, means, sizes);
902 else wordf(output, "%d", kmeans(args->colors.data() + 1, count - 1, wanted, means, sizes));
903 for (int i = 0; i < wanted; ++i) {
904 char text[8];
905 color_text(means[i], text);
906 wordf(output, "%s:%d", text, sizes[i]);
907 }
908 break;
909 }
910 case OP_SORT: {
911 Color colors[32];
912 std::copy_n(args->colors.begin(), count, colors);
913 std::stable_sort(colors, colors + count, [](Color a, Color b) {
914 Hsl x = hsl(a), y = hsl(b);
915 return std::tie(x.h, x.s, x.l) < std::tie(y.h, y.s, y.l);
916 });
917 for (int i = 0; i < count; ++i) output_color(output, colors[i]);
918 break;
919 }
920 case OP_DITHER: {
921 Color error = rgb(0, 0, 0);
922 for (int i = 0; i < count; ++i) {
923 Color target = args->colors[i];
924 for (int j = 0; j < 3; ++j) target.x[j] = clamp(target.x[j] + error.x[j]);
925 int best = 0;
926 for (int j = 1; j < 16; ++j)
927 if (distance(target, builtins[j].color) < distance(target, builtins[best].color)) best = j;
928 wordf(output, "%s", builtins[best].name.c_str());
929 for (int j = 0; j < 3; ++j) error.x[j] = target.x[j] - builtins[best].color.x[j];
930 }
931 break;
932 }
933 case OP_RANDOM: {
934 Int state_value = a;
935 for (Int i = 0; i < b; ++i) {
936 state_value = state_value * 48271 % 2147483647;
937 Int h = state_value % 360;
938 state_value = state_value * 48271 % 2147483647;
939 Int s = 40 + state_value % 61;
940 state_value = state_value * 48271 % 2147483647;
941 Int l = 25 + state_value % 51;
942 output_color(output, from_hsl(h, s, l));
943 }
944 break;
945 }
946 }
947}

949static int process(State *state, const char **tokens, int count, Text *output) {
950 int operation = -1;
951 for (int i = 0; i < 53; ++i)
952 if (strcmp(tokens[0], specifications[i].name) == 0) {
953 operation = i;
954 break;
955 }
956 if (operation < 0) return 0;
957 Arguments arguments{};
958 if (!prepare(state, &specifications[operation], operation, tokens + 1, count - 1, &arguments)) return 0;
959 execute(state, operation, tokens + 1, count - 1, &arguments, output);
960 return 1;
961}

963int main() {
964 std::string line;
965 if (!std::getline(std::cin, line)) return 0;
966 int commands = std::stoi(line);
967 State state;
968 for (int i = 0; i < commands; ++i) {
969 std::getline(std::cin, line);
970 std::istringstream input(line);
971 std::vector<std::string> words;
972 for (std::string word; input >> word;) words.push_back(word);
973 std::vector<const char *> tokens;
974 for (const std::string &word : words) tokens.push_back(word.c_str());
975 Text output;
976 if (process(&state, tokens.data(), static_cast<int>(tokens.size()), &output))
977 std::cout << output << '\n';
978 else std::cout << "INVALID\n";
979 }
