83function floor_div(int $n, int $d): int
84{
85 $quotient = intdiv($n, $d);
86 if ($n < 0 && $n % $d !== 0) {
87 --$quotient;
88 }
89 return $quotient;

135function parse_color(string $text, array $registry): array
136{
137 global $builtinColors;
138 if (preg_match('/^#[0-9a-fA-F]{6}$/D', $text) === 1) {
139 return [hexdec(substr($text, 1, 2)), hexdec(substr($text, 3, 2)), hexdec(substr($text, 5, 2))];
140 }
141 if (preg_match('/^#[0-9a-fA-F]{3}$/D', $text) === 1) {
142 return [17 * hexdec($text[1]), 17 * hexdec($text[2]), 17 * hexdec($text[3])];
143 }
144 if (preg_match('/^rgb\(([0-9]{1,3}),([0-9]{1,3}),([0-9]{1,3})\)$/D', $text, $matches) === 1) {
145 $color = [(int) $matches[1], (int) $matches[2], (int) $matches[3]];
146 require_valid(max($color) <= 255);
147 return $color;
148 }
149 $index = array_search($text, BUILTIN_NAMES, true);
150 if ($index !== false) {
151 return $builtinColors[$index];
152 }
153 require_valid(array_key_exists($text, $registry['colors']));
154 return $registry['colors'][$text];
155}

157function typed_arg(string $text, string $kind, array $registry): mixed
158{
159 if ($kind === 'c') {
160 return parse_color($text, $registry);
161 }
162 if ($kind === 'name') {
163 require_valid(preg_match('/^[a-z]{1,12}$/D', $text) === 1);
164 return $text;
165 }
166 if ($kind === 'code') {
167 require_valid(preg_match('/^0x[0-9a-fA-F]{4}$/D', $text) === 1);
168 return hexdec(substr($text, 2));
169 }
170 if (array_key_exists($kind, KEYWORDS)) {
171 require_valid(in_array($text, KEYWORDS[$kind], true));
172 return $text;
173 }
174 [$low, $high] = array_map('intval', explode(':', $kind));
175 return integer_arg($text, $low, $high);
176}

178function hue(array $color): int
179{
180 [$r, $g, $b] = $color;
181 $high = max($color);
182 $delta = $high - min($color);
183 if ($delta === 0) {
184 return 0;
185 }
186 if ($high === $r) {
187 $numerator = 60 * ($g - $b);
188 if ($numerator < 0) {
189 $numerator += 360 * $delta;
190 }
191 } elseif ($high === $g) {
192 $numerator = 60 * ($b - $r) + 120 * $delta;
193 } else {
194 $numerator = 60 * ($r - $g) + 240 * $delta;
195 }
196 return round_ratio($numerator, $delta) % 360;
197}

199function hsl(array $color): array
200{
201 $high = max($color);
202 $low = min($color);
203 $delta = $high - $low;
204 $saturation = $delta === 0 ? 0 : round_ratio(100 * $delta, 255 - abs($high + $low - 255));
205 return [hue($color), $saturation, round_ratio(100 * ($high + $low), 510)];
206}

208function base60(int $h): array
209{
210 $sector = intdiv($h, 60);
211 $fraction = $h % 60;
212 $x = $sector % 2 === 0 ? $fraction : 60 - $fraction;
213 return [[60, $x, 0], [$x, 60, 0], [0, 60, $x], [0, $x, 60], [$x, 0, 60], [60, 0, $x]][$sector];
214}

216function from_hsl(int $h, int $s, int $l): array
217{
218 $chroma = (100 - abs(2 * $l - 100)) * $s;
219 $offset = 6000 * $l - 30 * $chroma;
220 return array_map(fn($u) => round_ratio(255 * ($u * $chroma + $offset), 600000), base60($h));
221}

223function model_values(string $model, array $color): array
224{
225 [$r, $g, $b] = $color;
226 $high = max($color);
227 $low = min($color);
228 if ($model === 'HSL') {
229 return hsl($color);
230 }
231 if ($model === 'HSV') {
232 $saturation = $high === 0 ? 0 : round_ratio(100 * ($high - $low), $high);
233 return [hue($color), $saturation, round_ratio(100 * $high, 255)];
234 }
235 if ($model === 'HWB') {
236 return [hue($color), round_ratio(100 * $low, 255), round_ratio(100 * (255 - $high), 255)];
237 }
238 if ($model === 'CMYK') {
239 if ($high === 0) {
240 return [0, 0, 0, 100];
241 }
242 return [
243 round_ratio(100 * ($high - $r), $high),
244 round_ratio(100 * ($high - $g), $high),
245 round_ratio(100 * ($high - $b), $high),
246 round_ratio(100 * (255 - $high), 255)
247 ];
248 }
249 if ($model === 'YCBCR') {
250 return [
251 byte_clamp(round_ratio(299 * $r + 587 * $g + 114 * $b, 1000)),
252 byte_clamp(round_ratio(128000 - 169 * $r - 331 * $g + 500 * $b, 1000)),
253 byte_clamp(round_ratio(128000 + 500 * $r - 419 * $g - 81 * $b, 1000))
254 ];
255 }
256 return [
257 round_ratio(299 * $r + 587 * $g + 114 * $b, 1000),
258 round_ratio(596 * $r - 274 * $g - 322 * $b, 1000),
259 round_ratio(211 * $r - 523 * $g + 312 * $b, 1000)
260 ];
261}

275function ansi_color(int $number): array
276{
277 global $builtinColors;
278 if ($number < 16) {
279 return $builtinColors[$number];
280 }
281 if ($number >= 232) {
282 return array_fill(0, 3, 8 + 10 * ($number - 232));
283 }
284 $value = $number - 16;
285 $levels = [0, 95, 135, 175, 215, 255];
286 return [$levels[intdiv($value, 36)], $levels[intdiv($value, 6) % 6], $levels[$value % 6]];
287}

289function root_ratio(int $n, int $d): int
290{
291 $low = 0;
292 $high = 1;
293 while ($d * $high * $high <= $n) {
294 $high *= 2;
295 }
296 while ($high - $low > 1) {
297 $middle = intdiv($low + $high, 2);
298 if ($d * $middle * $middle <= $n) {
299 $low = $middle;
300 } else {
301 $high = $middle;
302 }
303 }
304 return $low;
305}

313function contrast(array $a, array $b): array
314{
315 $luminance = fn($c) => 2126 * $c[0] * $c[0] + 7152 * $c[1] * $c[1] + 722 * $c[2] * $c[2];
316 $y1 = $luminance($a);
317 $y2 = $luminance($b);
318 $high = max($y1, $y2) + 32512500;
319 $low = min($y1, $y2) + 32512500;
320 if ($high >= 7 * $low) {
321 $verdict = 'AAA';
322 } elseif (2 * $high >= 9 * $low) {
323 $verdict = 'AA';
324 } else {
325 $verdict = 'FAIL';
326 }
327 return [round_ratio(100 * $high, $low), $verdict];
328}

336function gradient(array $a, array $b, int $n): array
337{
338 $output = [];
339 for ($i = 0; $i < $n; ++$i) {
340 $color = [];
341 for ($j = 0; $j < 3; ++$j) {
342 $color[] = $a[$j] + away_ratio(($b[$j] - $a[$j]) * $i, $n - 1);
343 }
344 $output[] = $color;
345 }
346 return $output;
347}

362function palette(array $colors, int $k): array
363{
364 $boxes = [$colors];
365 while (count($boxes) < $k) {
366 $boxWidths = array_map(fn($box) => count($box) >= 2 ? max(widths($box)) : -1,$boxes);
367 $selected = array_search(max($boxWidths),$boxWidths,true);
368 $box = $boxes[$selected];
369 $ranges = widths($box);
370 $axis = array_search(max($ranges), $ranges, true);
371 usort($box,fn($a,$b) => $a[$axis]<=>$b[$axis]);
372 $middle = intdiv(count($box), 2);
373 array_splice($boxes, $selected, 1, [array_slice($box, 0, $middle), array_slice($box, $middle)]);
374 }
375 return $boxes;
376}

378function blend_component(string $mode, int $x, int $y): int
379{
380 $multiply = intdiv($x * $y, 255);
381 $screen = 255 - intdiv((255 - $x) * (255 - $y), 255);
382 if ($mode === 'MULTIPLY') {
383 return $multiply;
384 }
385 if ($mode === 'SCREEN') {
386 return $screen;
387 }
388 if ($mode === 'OVERLAY' || $mode === 'HARDLIGHT') {
389 $lower = $mode === 'OVERLAY' ? $x < 128 : $y < 128;
390 if ($lower) {
391 return intdiv(2 * $x * $y, 255);
392 }
393 return 255 - intdiv(2 * (255 - $x) * (255 - $y), 255);
394 }
395 if ($mode === 'SOFTLIGHT') {
396 return intdiv((255 - $x) * $multiply + $x * $screen, 255);
397 }
398 if ($mode === 'DARKEN') {
399 return min($x, $y);
400 }
401 if ($mode === 'LIGHTEN') {
402 return max($x, $y);
403 }
404 if ($mode === 'DIFFERENCE') {
405 return abs($x - $y);
406 }
407 if ($mode === 'EXCLUSION') {
408 return $x + $y - intdiv(2 * $x * $y, 255);
409 }
410 if ($mode === 'ADD') {
411 return min(255, $x + $y);
412 }
413 if ($mode === 'SUBTRACT') {
414 return max(0, $x - $y);
415 }
416 if ($mode === 'COLORDODGE') {
417 return $y === 255 ? 255 : min(255, intdiv(255 * $x, 255 - $y));
418 }
419 return $y === 0 ? 0 : max(0, 255 - intdiv(255 * (255 - $x), $y));
420}

422function execute(string $command, array $raw, array &$registry): string
423{
424 global $builtinColors;
425 if ($command === 'PALETTE' || $command === 'KMEANS') {
426 require_valid(count($raw) >= 2 && count($raw) <= 33);
427 $k = integer_arg($raw[0], 1, count($raw) - 1);
428 $colors = array_map(fn($token) => parse_color($token, $registry), array_slice($raw, 1));
429 $args = [$k, $colors];
430 } elseif (in_array($command, ['BEST', 'SORT', 'DITHER'], true)) {
431 $minimum = $command === 'BEST' ? 2 : 1;
432 $maximum = $command === 'BEST' ? 33 : 32;
433 require_valid(count($raw) >= $minimum && count($raw) <= $maximum);
434 $args = [array_map(fn($token) => parse_color($token, $registry), $raw)];
435 } else {
436 require_valid(array_key_exists($command, SIGNATURES));
437 $kinds = SIGNATURES[$command] === '' ? [] : explode(' ', SIGNATURES[$command]);
438 require_valid(count($raw) === count($kinds));
439 $args = [];
440 foreach ($raw as $i => $token) {
441 $args[] = typed_arg($token, $kinds[$i], $registry);
442 }
443 }
444 if ($command === 'DEFINE') {
445 [$name, $color] = $args;
446 require_valid(!in_array($name, BUILTIN_NAMES, true));
447 if (!array_key_exists($name, $registry['colors'])) {
448 $registry['order'][] = $name;
449 }
450 $registry['colors'][$name] = $color;
451 return $name . ' ' . hex_color($color);
452 }
453 if ($command === 'UNDEFINE') {
454 $name = $args[0];
455 require_valid(array_key_exists($name, $registry['colors']));
456 unset($registry['colors'][$name]);
457 $registry['order'] = array_values(array_filter($registry['order'], fn($other) => $other !== $name));
458 return $name . ' removed';
459 }
460 if ($command === 'LIST') {
461 if ($registry['colors'] === []) {
462 return 'NONE';
463 }
464 $names = array_keys($registry['colors']);
465 sort($names, SORT_STRING);
466 return implode(' ', array_map(fn($name) => $name . '=' . hex_color($registry['colors'][$name]), $names));
467 }
468 if ($command === 'PARSE' || $command === 'NEAREST') {
469 $color = $args[0];
470 $names = BUILTIN_NAMES;
471 $colors = $builtinColors;
472 foreach ($registry['order'] as $name) {
473 $names[] = $name;
474 $colors[] = $registry['colors'][$name];
475 }
476 if ($command === 'NEAREST') {
477 $i = nearest_index($color, $colors);
478 return sprintf('%s %s %d', $names[$i], hex_color($colors[$i]), color_distance($color, $colors[$i]));
479 }
480 $index = array_search($color,$colors,true);
481 $exact = $index === false ? '-' : $names[$index];
482 $parts = [hex_color($color), sprintf('rgb(%d,%d,%d)', ...$color)];
483 foreach (['HSL', 'HSV', 'HWB', 'CMYK', 'YCBCR', 'YIQ'] as $model) {
484 $parts[] = model_text($model, $color);
485 }
486 $parts[] = $exact;
487 return implode(' ', $parts);
488 }
489 if (in_array($command, ['HSL', 'HSV', 'HWB', 'CMYK', 'YCBCR', 'YIQ'], true)) {
490 return model_text($command, $args[0]);
491 }
492 if ($command === 'FROMHSL') {
493 $result = from_hsl(...$args);
494 } elseif ($command === 'FROMHSV') {
495 [$h, $s, $v] = $args;
496 $chroma = $v * $s;
497 $result = array_map(fn($u) => round_ratio(255 * ($u * $chroma + 6000 * $v - 60 * $chroma), 600000), base60($h));
498 } elseif ($command === 'FROMHWB') {
499 [$h, $w, $k] = $args;
500 if ($w + $k >= 100) {
501 $result = array_fill(0, 3, round_ratio(255 * $w, $w + $k));
502 } else {
503 $result = array_map(fn($u) => round_ratio(255 * ($u * (100 - $w - $k) + 60 * $w), 6000), base60($h));
504 }
505 } elseif ($command === 'FROMCMYK') {
506 $result = [];
507 for ($i = 0; $i < 3; ++$i) {
508 $result[] = round_ratio(255 * (100 - $args[$i]) * (100 - $args[3]), 10000);
509 }
510 } elseif ($command === 'FROMYCBCR') {
511 [$y, $cb, $cr] = $args;
512 $result = [
513 byte_clamp(round_ratio(1000 * $y + 1402 * ($cr - 128), 1000)),
514 byte_clamp(round_ratio(1000 * $y - 344 * ($cb - 128) - 714 * ($cr - 128), 1000)),
515 byte_clamp(round_ratio(1000 * $y + 1772 * ($cb - 128), 1000))
516 ];
517 } elseif ($command === 'FROMYIQ') {
518 [$y, $i, $q] = $args;
519 $result = [
520 byte_clamp(round_ratio(1000 * $y + 956 * $i + 619 * $q, 1000)),
521 byte_clamp(round_ratio(1000 * $y - 272 * $i - 647 * $q, 1000)),
522 byte_clamp(round_ratio(1000 * $y - 1106 * $i + 1703 * $q, 1000))
523 ];
524 } elseif ($command === 'RGB565') {
525 [$r, $g, $b] = $args[0];
526 $value = 2048 * round_ratio(31 * $r, 255) + 32 * round_ratio(63 * $g, 255) + round_ratio(31 * $b, 255);
527 return sprintf('0x%04X', $value);
528 } elseif ($command === 'FROM565') {
529 $value = $args[0];
530 $result = [round_ratio(255 * intdiv($value, 2048), 31), round_ratio(255 * (intdiv($value, 32) % 64), 63), round_ratio(255 * ($value % 32), 31)];
531 } elseif ($command === 'ANSI') {
532 $colors = array_map('ansi_color', range(16, 255));
533 $i = nearest_index($args[0], $colors);
534 return sprintf('%d %s %d', $i + 16, hex_color($colors[$i]), color_distance($args[0], $colors[$i]));
535 } elseif ($command === 'FROMANSI') {
536 $result = ansi_color($args[0]);
537 } elseif ($command === 'WEBSAFE') {
538 $result = array_map(fn($x) => 51 * round_ratio($x, 51), $args[0]);
539 } elseif ($command === 'GRAY') {
540 [$color, $method] = $args;
541 [$r, $g, $b] = $color;
542 if ($method === 'LUMA') {
543 $value = round_ratio(299 * $r + 587 * $g + 114 * $b, 1000);
544 } elseif ($method === 'AVERAGE') {
545 $value = round_ratio($r + $g + $b, 3);
546 } elseif ($method === 'LIGHTNESS') {
547 $value = round_ratio(max($color) + min($color), 2);
548 } else {
549 $value = root_ratio(299 * $r * $r + 587 * $g * $g + 114 * $b * $b, 1000);
550 }
551 $result = array_fill(0, 3, $value);
552 } elseif ($command === 'INVERT') {
553 $result = array_map(fn($x) => 255 - $x, $args[0]);
554 } elseif ($command === 'MIX') {
555 [$a, $b, $p] = $args;
556 $result = [];
557 for ($i = 0; $i < 3; ++$i) {
558 $result[] = round_ratio($a[$i] * (100 - $p) + $b[$i] * $p, 100);
559 }
560 } elseif ($command === 'MIXHSL') {
561 [$a, $b, $p] = $args;
562 [$h1, $s1, $l1] = hsl($a);
563 [$h2, $s2, $l2] = hsl($b);
564 $delta = positive_mod($h2 - $h1, 360);
565 if ($delta > 180) {
566 $delta -= 360;
567 }
568 $result = from_hsl(positive_mod($h1 + away_ratio($delta * $p, 100), 360),
569 $s1 + away_ratio(($s2 - $s1) * $p, 100), $l1 + away_ratio(($l2 - $l1) * $p, 100));
570 } elseif ($command === 'GRADIENT') {
571 return color_sequence(gradient(...$args));
572 } elseif (in_array($command, ['TINTS', 'SHADES', 'TONES'], true)) {
573 $value = ['TINTS' => 255, 'SHADES' => 0, 'TONES' => 128][$command];
574 return color_sequence(array_slice(gradient($args[0], array_fill(0, 3, $value), $args[1] + 2), 1, $args[1]));
575 } elseif ($command === 'BLEND') {
576 [$mode, $a, $b] = $args;
577 if (in_array($mode, ['HUE', 'SATURATION', 'COLOR', 'LUMINOSITY'], true)) {
578 [$h1, $s1, $l1] = hsl($a);
579 [$h2, $s2, $l2] = hsl($b);
580 if ($mode === 'HUE') {
581 $result = from_hsl($h2, $s1, $l1);
582 } elseif ($mode === 'SATURATION') {
583 $result = from_hsl($h1, $s2, $l1);
584 } elseif ($mode === 'COLOR') {
585 $result = from_hsl($h2, $s2, $l1);
586 } else {
587 $result = from_hsl($h1, $s1, $l2);
588 }
589 } else {
590 $result = [];
591 for ($i = 0; $i < 3; ++$i) {
592 $result[] = blend_component($mode, $a[$i], $b[$i]);
593 }
594 }
595 } elseif ($command === 'ALPHA') {
596 [$a, $p, $b, $q] = $args;
597 $opacity = 100 * $p + $q * (100 - $p);
598 $result = [0, 0, 0];
599 if ($opacity > 0) {
600 for ($i = 0; $i < 3; ++$i) {
601 $result[$i] = round_ratio(100 * $p * $a[$i] + $q * (100 - $p) * $b[$i], $opacity);
602 }
603 }
604 return sprintf('%s %d%%', hex_color($result), round_ratio($opacity, 100));
605 } elseif ($command === 'SIMULATE') {
606 [$color, $kind] = $args;
607 $matrices = [
608 'PROTANOPIA' => [[567, 433, 0], [558, 442, 0], [0, 242, 758]],
609 'DEUTERANOPIA' => [[625, 375, 0], [700, 300, 0], [0, 300, 700]],
610 'TRITANOPIA' => [[950, 50, 0], [0, 433, 567], [0, 475, 525]],
611 'ACHROMATOPSIA' => [[299, 587, 114], [299, 587, 114], [299, 587, 114]]
612 ];
613 $result = [];
614 foreach ($matrices[$kind] as $row) {
615 $result[] = round_ratio($row[0] * $color[0] + $row[1] * $color[1] + $row[2] * $color[2], 1000);
616 }
617 } elseif (in_array($command, ['ROTATE', 'COMPLEMENT', 'TRIADIC', 'TETRADIC', 'SPLITCOMP', 'ANALOGOUS'], true)) {
618 [$h, $s, $l] = hsl($args[0]);
619 if ($command === 'ROTATE') {
620 $offsets = [$args[1]];
621 } elseif ($command === 'ANALOGOUS') {
622 $offsets = [];
623 for ($i = -$args[1]; $i <= $args[1]; ++$i) {
624 if ($i !== 0) {
625 $offsets[] = $i * $args[2];
626 }
627 }
628 } else {
629 $offsets = ['COMPLEMENT' => [180], 'TRIADIC' => [120, 240], 'TETRADIC' => [90, 180, 270], 'SPLITCOMP' => [150, 210]][$command];
630 }
631 return color_sequence(array_map(fn($d) => from_hsl(positive_mod($h + $d, 360), $s, $l), $offsets));
632 } elseif (in_array($command, ['LIGHTEN', 'DARKEN', 'SATURATE', 'DESATURATE'], true)) {
633 [$h, $s, $l] = hsl($args[0]);
634 $p = $args[1];
635 if ($command === 'LIGHTEN') {
636 $l = min(100, $l + $p);
637 } elseif ($command === 'DARKEN') {
638 $l = max(0, $l - $p);
639 } elseif ($command === 'SATURATE') {
640 $s = min(100, $s + $p);
641 } else {
642 $s = max(0, $s - $p);
643 }
644 $result = from_hsl($h, $s, $l);
645 } elseif ($command === 'DISTANCE') {
646 [$a, $b] = $args;
647 $dr = $a[0] - $b[0];
648 $dg = $a[1] - $b[1];
649 $db = $a[2] - $b[2];
650 $weighted = (1024 + $a[0] + $b[0]) * $dr * $dr + 2048 * $dg * $dg + (1534 - $a[0] - $b[0]) * $db * $db;
651 return sprintf('%d %d', color_distance($a, $b), root_ratio($weighted, 512));
652 } elseif ($command === 'CONTRAST') {
653 return contrast_text(...$args);
654 } elseif ($command === 'ACCESSIBLE') {
655 [$result, $bg] = $args;
656 if (contrast($result, $bg)[1] === 'FAIL') {
657 [$h, $s, $l] = hsl($result);
658 $found = false;
659 for ($step = 1; $step <= 100; ++$step) {
660 foreach ([$l - $step, $l + $step] as $lightness) {
661 if ($lightness >= 0 && $lightness <= 100) {
662 $candidate = from_hsl($h, $s, $lightness);
663 if (contrast($candidate, $bg)[1] !== 'FAIL') {
664 $result = $candidate;
665 $found = true;
666 break;
667 }
668 }
669 }
670 if ($found) {
671 break;
672 }
673 }
674 }
675 return hex_color($result) . ' ' . contrast_text($result, $bg);
676 } elseif ($command === 'BEST') {
677 $colors = $args[0];
678 $bg = $colors[0];
679 $selected = 1;
680 for ($i = 2; $i < count($colors); ++$i) {
681 if (contrast($colors[$i], $bg)[0] > contrast($colors[$selected], $bg)[0]) {
682 $selected = $i;
683 }
684 }
685 return hex_color($colors[$selected]) . ' ' . contrast_text($colors[$selected], $bg);
686 } elseif ($command === 'DESCRIBE') {
687 [$h, $s, $l] = hsl($args[0]);
688 if ($l <= 5) {
689 return 'black';
690 }
691 if ($l >= 95) {
692 return 'white';
693 }
694 if ($l <= 20) {
695 $lightness = 'very dark';
696 } elseif ($l <= 40) {
697 $lightness = 'dark';
698 } elseif ($l <= 60) {
699 $lightness = 'medium';
700 } elseif ($l <= 80) {
701 $lightness = 'light';
702 } else {
703 $lightness = 'very light';
704 }
705 if ($s < 10) {
706 return $lightness . ' gray';
707 }
708 if ($s < 40) {
709 $saturation = 'muted';
710 } elseif ($s < 80) {
711 $saturation = 'moderate';
712 } else {
713 $saturation = 'vivid';
714 }
715 $names = explode(' ', 'red orange yellow chartreuse green spring-green cyan azure blue violet magenta rose');
716 return $lightness . ' ' . $saturation . ' ' . $names[intdiv(($h + 15) % 360, 30)];
717 } elseif ($command === 'PALETTE' || $command === 'KMEANS') {
718 [$k, $colors] = $args;
719 $boxes = palette($colors, $k);
720 $representatives = array_map('mean_color', $boxes);
721 $counts = array_map('count', $boxes);
722 $iterations = 0;
723 if ($command === 'KMEANS') {
724 for ($iteration = 1; $iteration <= 20; ++$iteration) {
725 $assigned = array_fill(0, $k, []);
726 foreach ($colors as $color) {
727 $j = nearest_index($color, $representatives);
728 $assigned[$j][] = $color;
729 }
730 $updated = [];
731 $unchanged = true;
732 for ($j = 0; $j < $k; ++$j) {
733 $counts[$j] = count($assigned[$j]);
734 $updated[$j] = $counts[$j] === 0 ? $representatives[$j] : mean_color($assigned[$j]);
735 if ($updated[$j] !== $representatives[$j]) {
736 $unchanged = false;
737 }
738 }
739 $representatives = $updated;
740 $iterations = $iteration;
741 if ($unchanged) {
742 break;
743 }
744 }
745 }
746 $output = [];
747 for ($j = 0; $j < $k; ++$j) {
748 $output[] = hex_color($representatives[$j]) . ':' . $counts[$j];
749 }
750 return ($command === 'KMEANS' ? $iterations . ' ' : '') . implode(' ', $output);
751 } elseif ($command === 'SORT') {
752 $indexed = [];
753 foreach ($args[0] as $i => $color) {
754 $indexed[] = ['color' => $color, 'value' => hsl($color), 'index' => $i];
755 }
756 usort($indexed, function($a, $b) {
757 $comparison = $a['value'] <=> $b['value'];
758 return $comparison !== 0 ? $comparison : $a['index'] <=> $b['index'];
759 });
760 return color_sequence(array_column($indexed, 'color'));
761 } elseif ($command === 'DITHER') {
762 $error = [0, 0, 0];
763 $colors = array_slice($builtinColors, 0, 16);
764 $output = [];
765 foreach ($args[0] as $color) {
766 $adjusted = [];
767 for ($j = 0; $j < 3; ++$j) {
768 $adjusted[$j] = byte_clamp($color[$j] + $error[$j]);
769 }
770 $selected = nearest_index($adjusted, $colors);
771 $output[] = BUILTIN_NAMES[$selected];
772 for ($j = 0; $j < 3; ++$j) {
773 $error[$j] = $adjusted[$j] - $colors[$selected][$j];
774 }
775 }
776 return implode(' ', $output);
777 } elseif ($command === 'RANDOM') {
778 [$state, $n] = $args;
779 $output = [];
780 for ($i = 0; $i < $n; ++$i) {
781 $state = (48271 * $state) % 2147483647;
782 $u = $state;
783 $state = (48271 * $state) % 2147483647;
784 $v = $state;
785 $state = (48271 * $state) % 2147483647;
786 $output[] = from_hsl($u % 360, 40 + $v % 61, 25 + $state % 51);
787 }
788 return color_sequence($output);
789 }
790 return hex_color($result);
