185static Int smaller(Int a, Int b) { return a < b ? a : b; }
186static Int larger(Int a, Int b) { return a > b ? a : b; }
187static Int clamp(Int x) { return smaller(255, larger(0, x)); }

189static Int floor_div(Int a, Int b) {
190 Int q = a / b;
191 return q - (a < 0 && a % b != 0);
192}

194static Int rounded(Int numerator, Int denominator) {
195 return floor_div(2 * numerator + denominator, 2 * denominator);

198static Int away(Int numerator, Int denominator) {
199 return numerator < 0 ? -rounded(-numerator, denominator)
200 : rounded(numerator, denominator);

216static Int distance(Color a, Color b) {
217 Int result = 0;
218 for (int j = 0; j < 3; ++j) {
219 Int difference = a.x[j] - b.x[j];
220 result += difference * difference;
221 }
222 return result;

225static void extrema(Color c, Int *maximum, Int *minimum) {
226 *maximum = larger(c.x[0], larger(c.x[1], c.x[2]));
227 *minimum = smaller(c.x[0], smaller(c.x[1], c.x[2]));
228}

230static Int hue(Color c) {
231 Int maximum, minimum;
232 extrema(c, &maximum, &minimum);
233 Int delta = maximum - minimum;
234 if (delta == 0) return 0;
235 Int numerator;
236 if (maximum == c.x[0]) {
237 numerator = 60 * (c.x[1] - c.x[2]);
238 if (numerator < 0) numerator += 360 * delta;
239 } else if (maximum == c.x[1]) {
240 numerator = 60 * (c.x[2] - c.x[0]) + 120 * delta;
241 } else {
242 numerator = 60 * (c.x[0] - c.x[1]) + 240 * delta;
243 }
244 return rounded(numerator, delta) % 360;
245}

247static Hsl hsl(Color c) {
248 Int maximum, minimum;
249 extrema(c, &maximum, &minimum);
250 Int delta = maximum - minimum;
251 Hsl result = {hue(c), 0, rounded(100 * (maximum + minimum), 510)};
252 if (delta != 0)
253 result.s = rounded(100 * delta, 255 - llabs(maximum + minimum - 255));
254 return result;
255}

257static Color sector(Int h) {
258 Int q = h / 60;
259 Int f = h % 60;
260 Int x = q % 2 == 0 ? f : 60 - f;
261 switch (q) {
262 case 0: return rgb(60, x, 0);
263 case 1: return rgb(x, 60, 0);
264 case 2: return rgb(0, 60, x);
265 case 3: return rgb(0, x, 60);
266 case 4: return rgb(x, 0, 60);
267 default: return rgb(60, 0, x);
268 }
269}

271static Color from_hsl(Int h, Int s, Int l) {
272 Int chroma = (100 - llabs(2 * l - 100)) * s;
273 Int offset = 30 * (200 * l - chroma);
274 Color result = sector(h);
275 for (int j = 0; j < 3; ++j)
276 result.x[j] = rounded(255 * (result.x[j] * chroma + offset), 600000);
277 return result;
278}

280static Color from_hsv(Int h, Int s, Int v) {
281 Int chroma = v * s;
282 Int offset = 60 * (100 * v - chroma);
283 Color result = sector(h);
284 for (int j = 0; j < 3; ++j)
285 result.x[j] = rounded(255 * (result.x[j] * chroma + offset), 600000);
286 return result;
287}

289static Color from_hwb(Int h, Int w, Int k) {
290 if (w + k >= 100) {
291 Int value = rounded(255 * w, w + k);
292 return rgb(value, value, value);
293 }
294 Color result = sector(h);
295 for (int j = 0; j < 3; ++j)
296 result.x[j] = rounded(255 * (result.x[j] * (100 - w - k) + 60 * w), 6000);
297 return result;
298}

300static Int integer_root(Int value) {
301 Int low = 0, high = 1;
302 while (high * high <= value) high *= 2;
303 while (low + 1 < high) {
304 Int middle = (low + high) / 2;
305 if (middle * middle <= value) low = middle;
306 else high = middle;
307 }
308 return low;
309}

322static Int luminance(Color c) {
323 return 2126 * c.x[0] * c.x[0] + 7152 * c.x[1] * c.x[1]
324 + 722 * c.x[2] * c.x[2];

327static Contrast contrast(Color a, Color b) {
328 Int y1 = luminance(a), y2 = luminance(b);
329 Int high = larger(y1, y2) + 32512500;
330 Int low = smaller(y1, y2) + 32512500;
331 Contrast result = {rounded(100 * high, low), "FAIL"};
332 if (high >= 7 * low) result.verdict = "AAA";
333 else if (2 * high >= 9 * low) result.verdict = "AA";
334 return result;
335}

338 Color result = first;
339 for (int j = 0; j < 3; ++j)
340 result.x[j] += away((last.x[j] - first.x[j]) * position, denominator);
341 return result;
342}

349static Color blend(int mode, Color base, Color top) {
350 if (mode >= 13) {
351 Hsl a = hsl(base), b = hsl(top);
352 if (mode == 13 || mode == 15) a.h = b.h;
353 if (mode == 14 || mode == 15) a.s = b.s;
354 if (mode == 16) a.l = b.l;
355 return from_hsl(a.h, a.s, a.l);
356 }
357 Color result = base;
358 for (int j = 0; j < 3; ++j) {
359 Int x = base.x[j], y = top.x[j], value = 0;
360 switch (mode) {
361 case 0: value = x * y / 255; break;
362 case 1: value = 255 - (255 - x) * (255 - y) / 255; break;
363 case 2:
364 case 3:
365 if ((mode == 2 ? x : y) < 128) value = 2 * x * y / 255;
366 else value = 255 - 2 * (255 - x) * (255 - y) / 255;
367 break;
368 case 4: {
369 Int multiply = x * y / 255;
370 Int screen = 255 - (255 - x) * (255 - y) / 255;
371 value = ((255 - x) * multiply + x * screen) / 255;
372 break;
373 }
374 case 5: value = smaller(x, y); break;
375 case 6: value = larger(x, y); break;
376 case 7: value = llabs(x - y); break;
377 case 8: value = x + y - 2 * x * y / 255; break;
378 case 9: value = smaller(255, x + y); break;
379 case 10: value = larger(0, x - y); break;
380 case 11: value = y == 255 ? 255 : smaller(255, 255 * x / (255 - y)); break;
381 case 12: value = y == 0 ? 0 : larger(0, 255 - 255 * (255 - x) / y); break;
382 }
383 result.x[j] = value;
384 }
385 return result;
386}

388static void palette(const Color *colors, int count, int wanted, Color *means, int *sizes) {
389 Box boxes[32] = {0};
390 boxes[0].count = count;
391 memcpy(boxes[0].colors, colors, count * sizeof *colors);
392 int box_count = 1;
393 while (box_count < wanted) {
394 int chosen = -1, channel = 0;
395 Int best_width = -1;
396 for (int i = 0; i < box_count; ++i) {
397 if (boxes[i].count < 2) continue;
398 Int width = -1;
399 int component = 0;
400 for (int j = 0; j < 3; ++j) {
401 Int minimum = 255, maximum = 0;
402 for (int p = 0; p < boxes[i].count; ++p) {
403 minimum = smaller(minimum, boxes[i].colors[p].x[j]);
404 maximum = larger(maximum, boxes[i].colors[p].x[j]);
405 }
406 if (maximum - minimum > width) {
407 width = maximum - minimum;
408 component = j;
409 }
410 }
411 if (width > best_width) {
412 chosen = i;
413 channel = component;
414 best_width = width;
415 }
416 }
417 Box sorted = boxes[chosen];
418 for (int i = 1; i < sorted.count; ++i) {
419 Color value = sorted.colors[i];
420 int position = i;
421 while (position > 0 && sorted.colors[position - 1].x[channel] > value.x[channel]) {
422 sorted.colors[position] = sorted.colors[position - 1];
423 --position;
424 }
425 sorted.colors[position] = value;
426 }
427 for (int i = box_count; i > chosen + 1; --i) boxes[i] = boxes[i - 1];
428 int middle = sorted.count / 2;
429 boxes[chosen].count = middle;
430 boxes[chosen + 1].count = sorted.count - middle;
431 for (int i = 0; i < middle; ++i) boxes[chosen].colors[i] = sorted.colors[i];
432 for (int i = middle; i < sorted.count; ++i)
433 boxes[chosen + 1].colors[i - middle] = sorted.colors[i];
434 ++box_count;
435 }
436 for (int i = 0; i < wanted; ++i) {
437 sizes[i] = boxes[i].count;
438 means[i] = rgb(0, 0, 0);
439 for (int p = 0; p < boxes[i].count; ++p)
440 for (int j = 0; j < 3; ++j) means[i].x[j] += boxes[i].colors[p].x[j];
441 for (int j = 0; j < 3; ++j) means[i].x[j] = rounded(means[i].x[j], sizes[i]);
442 }
443}

445static int kmeans(const Color *colors, int count, int wanted, Color *means, int *sizes) {
446 palette(colors, count, wanted, means, sizes);
447 for (int iteration = 1; iteration <= 20; ++iteration) {
448 Color sums[32];
449 for (int j = 0; j < wanted; ++j) {
450 sums[j] = rgb(0, 0, 0);
451 sizes[j] = 0;
452 }
453 for (int i = 0; i < count; ++i) {
454 int closest = 0;
455 for (int j = 1; j < wanted; ++j)
456 if (distance(colors[i], means[j]) < distance(colors[i], means[closest])) closest = j;
457 ++sizes[closest];
458 for (int j = 0; j < 3; ++j) sums[closest].x[j] += colors[i].x[j];
459 }
460 int unchanged = 1;
461 for (int i = 0; i < wanted; ++i) {
462 if (sizes[i] == 0) continue;
463 for (int j = 0; j < 3; ++j) sums[i].x[j] = rounded(sums[i].x[j], sizes[i]);
464 if (!equal_color(means[i], sums[i])) unchanged = 0;
465 means[i] = sums[i];
466 }
467 if (unchanged || iteration == 20) return iteration;
468 }
469 return 20;
470}

472static int hex_digit(char c) {
473 if (c >= '0' && c <= '9') return c - '0';
474 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
475 if (c >= 'A' && c <= 'F') return c - 'A' + 10;
476 return -1;
477}

479static int parse_integer(const char *text, Int low, Int high, Int *value) {
480 int negative = text[0] == '-';
481 int digits = 0;
482 Int result = 0;
483 for (int i = negative; text[i]; ++i) {
484 if (text[i] < '0' || text[i] > '9' || ++digits > 9) return 0;
485 result = result * 10 + text[i] - '0';
486 }
487 if (digits == 0) return 0;
488 if (negative) result = -result;
489 if (result < low || result > high) return 0;
490 *value = result;
491 return 1;
492}

512static int parse_color(const State *state, const char *text, Color *result) {
513 size_t length = strlen(text);
514 if (text[0] == '#') {
515 if (length != 4 && length != 7) return 0;
516 for (size_t i = 1; i < length; ++i) if (hex_digit(text[i]) < 0) return 0;
517 for (int j = 0; j < 3; ++j) {
518 if (length == 4) result->x[j] = 17 * hex_digit(text[j + 1]);
519 else result->x[j] = 16 * hex_digit(text[2 * j + 1]) + hex_digit(text[2 * j + 2]);
520 }
521 return 1;
522 }
523 if (length >= 4 && strncmp(text, "rgb(", 4) == 0) {
524 size_t position = 4;
525 for (int j = 0; j < 3; ++j) {
526 Int value = 0;
527 int digits = 0;
528 while (text[position] >= '0' && text[position] <= '9') {
529 if (++digits > 3) return 0;
530 value = value * 10 + text[position++] - '0';
531 }
532 if (digits == 0 || value > 255) return 0;
533 char expected = j == 2 ? ')' : ',';
534 if (text[position] != expected) return 0;
535 ++position;
536 result->x[j] = value;
537 }
538 return text[position] == '\0';
539 }
540 for (int i = 0; i < 24 + state->count; ++i) {
541 const Entry *entry = table_entry(state, i);
542 if (strcmp(entry->name, text) == 0) {
543 *result = entry->color;
544 return 1;
545 }
546 }
547 return 0;
548}

550static int keyword_index(const char *word, const char *choices) {
551 const char *start = choices;
552 int index = 0;
553 while (*start) {
554 const char *end = strchr(start, '|');
555 size_t length = end ? (size_t)(end - start) : strlen(start);
556 if (strlen(word) == length && strncmp(word, start, length) == 0) return index;
557 if (!end) break;
558 start = end + 1;
559 ++index;
560 }
561 return -1;
562}

564static int prepare(const State *state, const Spec *spec, int operation,
565 const char **words, int count, Arguments *arguments) {
566 if (count < spec->minimum || count > spec->maximum) return 0;
567 for (int i = 0; i < count; ++i) {
568 char kind;
569 if (spec->signature[0] == '*') kind = 'C';
570 else if (spec->signature[0] == 'P') kind = i == 0 ? 'I' : 'C';
571 else kind = spec->signature[i];
572 if (kind == 'C') {
573 if (!parse_color(state, words[i], &arguments->colors[i])) return 0;
574 } else if (kind == 'I') {
575 Int low = spec->signature[0] == 'P' ? 1 : spec->low[i];
576 Int high = spec->signature[0] == 'P' ? 32 : spec->high[i];
577 if (!parse_integer(words[i], low, high, &arguments->numbers[i])) return 0;
578 } else if (kind == 'N') {
579 if (!valid_name(words[i])) return 0;
580 } else if (kind == 'K') {
581 arguments->keyword = keyword_index(words[i], spec->keywords);
582 if (arguments->keyword < 0) return 0;
583 } else if (kind == 'W') {
584 if (strlen(words[i]) != 6 || strncmp(words[i], "0x", 2) != 0) return 0;
585 Int value = 0;
586 for (int j = 2; j < 6; ++j) {
587 int digit = hex_digit(words[i][j]);
588 if (digit < 0) return 0;
589 value = 16 * value + digit;
590 }
591 arguments->numbers[i] = value;
592 }
593 }
594 if (spec->signature[0] == 'P' && arguments->numbers[0] > count - 1) return 0;
595 if (operation == OP_DEFINE) {
596 for (int i = 0; i < 24; ++i)
597 if (strcmp(words[0], builtins[i].name) == 0) return 0;
598 }
599 if (operation == OP_UNDEFINE && registry_find(state, words[0]) < 0) return 0;
600 return 1;
601}

603static void color_text(Color c, char *text) {
604 snprintf(text, 8, "#%02X%02X%02X", (unsigned)c.x[0], (unsigned)c.x[1], (unsigned)c.x[2]);
605}

607static void output_color(Text *output, Color c) {
608 char text[8];
609 color_text(c, text);
610 wordf(output, "%s", text);
611}

613static void output_contrast(Text *output, Color c, Color background) {
614 Contrast value = contrast(c, background);
615 wordf(output, "%lld.%02lld %s", value.ratio / 100, value.ratio % 100, value.verdict);
616}

618static void output_model(Text *output, int operation, Color c) {
619 Int maximum, minimum;
620 extrema(c, &maximum, &minimum);
621 Int r = c.x[0], g = c.x[1], b = c.x[2];
622 if (operation == OP_HSL) {
623 Hsl value = hsl(c);
624 wordf(output, "hsl(%lld,%lld%%,%lld%%)", value.h, value.s, value.l);
625 } else if (operation == OP_HSV) {
626 wordf(output, "hsv(%lld,%lld%%,%lld%%)", hue(c),
627 maximum == 0 ? 0 : rounded(100 * (maximum - minimum), maximum), rounded(100 * maximum, 255));
628 } else if (operation == OP_HWB) {
629 wordf(output, "hwb(%lld,%lld%%,%lld%%)", hue(c), rounded(100 * minimum, 255),
630 rounded(100 * (255 - maximum), 255));
631 } else if (operation == OP_CMYK) {
632 wordf(output, "cmyk(%lld%%,%lld%%,%lld%%,%lld%%)",
633 maximum == 0 ? 0 : rounded(100 * (maximum - r), maximum),
634 maximum == 0 ? 0 : rounded(100 * (maximum - g), maximum),
635 maximum == 0 ? 0 : rounded(100 * (maximum - b), maximum),
636 rounded(100 * (255 - maximum), 255));
637 } else if (operation == OP_YCBCR) {
638 wordf(output, "ycbcr(%lld,%lld,%lld)", clamp(rounded(299 * r + 587 * g + 114 * b, 1000)),
639 clamp(rounded(128000 - 169 * r - 331 * g + 500 * b, 1000)),
640 clamp(rounded(128000 + 500 * r - 419 * g - 81 * b, 1000)));
641 } else {
642 wordf(output, "yiq(%lld,%lld,%lld)", rounded(299 * r + 587 * g + 114 * b, 1000),
643 rounded(596 * r - 274 * g - 322 * b, 1000), rounded(211 * r - 523 * g + 312 * b, 1000));
644 }
645}

647static void execute(State *state, int operation, const char **words, int count,
648 const Arguments *args, Text *output) {
649 Color c = args->colors[0];
650 Int a = args->numbers[0], b = args->numbers[1], d = args->numbers[2];
651 switch (operation) {
652 case OP_DEFINE:
653 wordf(output, "%s", words[0]);
654 output_color(output, args->colors[1]);
655 registry_set(state, words[0], args->colors[1]);
656 break;
657 case OP_UNDEFINE:
658 wordf(output, "%s removed", words[0]);
659 registry_delete(state, registry_find(state, words[0]));
660 break;
661 case OP_LIST: {
662 int indices[500], length = state->count;
663 for (int i = 0; i < length; ++i) indices[i] = i;
664 for (int i = 1; i < length; ++i) {
665 int value = indices[i], j = i;
666 while (j > 0 && strcmp(state->entries[indices[j - 1]].name,
667 state->entries[value].name) > 0) {
668 indices[j] = indices[j - 1];
669 --j;
670 }
671 indices[j] = value;
672 }
673 if (length == 0) wordf(output, "NONE");
674 for (int i = 0; i < length; ++i) {
675 const Entry *entry = &state->entries[indices[i]];
676 char text[8];
677 color_text(entry->color, text);
678 wordf(output, "%s=%s", entry->name, text);
679 }
680 break;
681 }
682 case OP_PARSE: {
683 output_color(output, c);
684 wordf(output, "rgb(%lld,%lld,%lld)", c.x[0], c.x[1], c.x[2]);
685 const int models[] = {OP_HSL, OP_HSV, OP_HWB, OP_CMYK, OP_YCBCR, OP_YIQ};
686 for (int i = 0; i < 6; ++i) output_model(output, models[i], c);
687 const char *name = "-";
688 for (int i = 0; i < 24 + state->count; ++i) {
689 const Entry *entry = table_entry(state, i);
690 if (equal_color(c, entry->color)) {
691 name = entry->name;
692 break;
693 }
694 }
695 wordf(output, "%s", name);
696 break;
697 }
698 case OP_HSL: case OP_HSV: case OP_HWB: case OP_CMYK: case OP_YCBCR: case OP_YIQ:
699 output_model(output, operation, c);
700 break;
701 case OP_FROMHSL: output_color(output, from_hsl(a, b, d)); break;
702 case OP_FROMHSV: output_color(output, from_hsv(a, b, d)); break;
703 case OP_FROMHWB: output_color(output, from_hwb(a, b, d)); break;
704 case OP_FROMCMYK: {
705 Color result = rgb(0, 0, 0);
706 for (int j = 0; j < 3; ++j)
707 result.x[j] = rounded(255 * (100 - args->numbers[j]) * (100 - args->numbers[3]), 10000);
708 output_color(output, result);
709 break;
710 }
711 case OP_FROMYCBCR:
712 output_color(output, rgb(clamp(rounded(1000 * a + 1402 * (d - 128), 1000)),
713 clamp(rounded(1000 * a - 344 * (b - 128) - 714 * (d - 128), 1000)),
714 clamp(rounded(1000 * a + 1772 * (b - 128), 1000))));
715 break;
716 case OP_FROMYIQ:
717 output_color(output, rgb(clamp(rounded(1000 * a + 956 * b + 619 * d, 1000)),
718 clamp(rounded(1000 * a - 272 * b - 647 * d, 1000)),
719 clamp(rounded(1000 * a - 1106 * b + 1703 * d, 1000))));
720 break;
721 case OP_RGB565: {
722 Int code = 2048 * rounded(31 * c.x[0], 255) + 32 * rounded(63 * c.x[1], 255)
723 + rounded(31 * c.x[2], 255);
724 wordf(output, "0x%04X", (unsigned)code);
725 break;
726 }
727 case OP_FROM565:
728 output_color(output, rgb(rounded(255 * (a / 2048), 31),
729 rounded(255 * ((a / 32) % 64), 63), rounded(255 * (a % 32), 31)));
730 break;
731 case OP_FROMANSI: output_color(output, ansi_color((int)a)); break;
732 case OP_ANSI: {
733 int best = 16;
734 for (int i = 17; i <= 255; ++i)
735 if (distance(c, ansi_color(i)) < distance(c, ansi_color(best))) best = i;
736 wordf(output, "%d", best);
737 output_color(output, ansi_color(best));
738 wordf(output, "%lld", distance(c, ansi_color(best)));
739 break;
740 }
741 case OP_WEBSAFE: {
742 Color result = c;
743 for (int j = 0; j < 3; ++j) result.x[j] = 51 * rounded(c.x[j], 51);
744 output_color(output, result);
745 break;
746 }
747 case OP_GRAY: {
748 Int maximum, minimum, value;
749 extrema(c, &maximum, &minimum);
750 if (args->keyword == 0) value = rounded(299 * c.x[0] + 587 * c.x[1] + 114 * c.x[2], 1000);
751 else if (args->keyword == 1) value = rounded(c.x[0] + c.x[1] + c.x[2], 3);
752 else if (args->keyword == 2) value = rounded(maximum + minimum, 2);
753 else value = integer_root((299 * c.x[0] * c.x[0] + 587 * c.x[1] * c.x[1]
754 + 114 * c.x[2] * c.x[2]) / 1000);
755 output_color(output, rgb(value, value, value));
756 break;
757 }
758 case OP_INVERT:
759 output_color(output, rgb(255 - c.x[0], 255 - c.x[1], 255 - c.x[2]));
760 break;
761 case OP_MIX: {
762 Color result = c;
763 for (int j = 0; j < 3; ++j)
764 result.x[j] = rounded(c.x[j] * (100 - d) + args->colors[1].x[j] * d, 100);
765 output_color(output, result);
766 break;
767 }
768 case OP_MIXHSL: {
769 Hsl first = hsl(c), last = hsl(args->colors[1]);
770 Int delta = modulo(last.h - first.h, 360);
771 if (delta > 180) delta -= 360;
772 output_color(output, from_hsl(modulo(first.h + away(delta * d, 100), 360),
773 first.s + away((last.s - first.s) * d, 100), first.l + away((last.l - first.l) * d, 100)));
774 break;
775 }
776 case OP_GRADIENT:
777 for (Int i = 0; i < d; ++i) output_color(output, interpolate(c, args->colors[1], i, d - 1));
778 break;
779 case OP_TINTS: case OP_SHADES: case OP_TONES: {
780 Int target = operation == OP_TINTS ? 255 : operation == OP_SHADES ? 0 : 128;
781 for (Int i = 1; i <= b; ++i) output_color(output, interpolate(c, rgb(target, target, target), i, b + 1));
782 break;
783 }
784 case OP_BLEND: output_color(output, blend(args->keyword, args->colors[1], args->colors[2])); break;
785 case OP_ALPHA: {
786 Int q = args->numbers[3];
787 Int alpha = 100 * b + q * (100 - b);
788 Color result = rgb(0, 0, 0);
789 if (alpha > 0)
790 for (int j = 0; j < 3; ++j)
791 result.x[j] = rounded(100 * b * c.x[j] + q * (100 - b) * args->colors[2].x[j], alpha);
792 output_color(output, result);
793 wordf(output, "%lld%%", rounded(alpha, 100));
794 break;
795 }
796 case OP_SIMULATE: {
797 static const Int coefficients[4][3][3] = {
798 {{567,433,0},{558,442,0},{0,242,758}},
799 {{625,375,0},{700,300,0},{0,300,700}},
800 {{950,50,0},{0,433,567},{0,475,525}},
801 {{299,587,114},{299,587,114},{299,587,114}}
802 };
803 Color result = rgb(0, 0, 0);
804 for (int j = 0; j < 3; ++j) {
805 Int sum = 0;
806 for (int k = 0; k < 3; ++k) sum += coefficients[args->keyword][j][k] * c.x[k];
807 result.x[j] = rounded(sum, 1000);
808 }
809 output_color(output, result);
810 break;
811 }
812 case OP_ROTATE: output_color(output, rotate(c, b)); break;
813 case OP_COMPLEMENT: output_color(output, rotate(c, 180)); break;
814 case OP_TRIADIC: output_color(output, rotate(c, 120)); output_color(output, rotate(c, 240)); break;
815 case OP_TETRADIC:
816 for (Int shift = 90; shift <= 270; shift += 90) output_color(output, rotate(c, shift));
817 break;
818 case OP_SPLITCOMP: output_color(output, rotate(c, 150)); output_color(output, rotate(c, 210)); break;
819 case OP_ANALOGOUS:
820 for (Int i = -b; i <= b; ++i) if (i != 0) output_color(output, rotate(c, i * d));
821 break;
822 case OP_LIGHTEN: case OP_DARKEN: case OP_SATURATE: case OP_DESATURATE: {
823 Hsl value = hsl(c);
824 if (operation == OP_LIGHTEN) value.l = smaller(100, value.l + b);
825 else if (operation == OP_DARKEN) value.l = larger(0, value.l - b);
826 else if (operation == OP_SATURATE) value.s = smaller(100, value.s + b);
827 else value.s = larger(0, value.s - b);
828 output_color(output, from_hsl(value.h, value.s, value.l));
829 break;
830 }
831 case OP_NEAREST: {
832 int best = 0;
833 for (int i = 1; i < 24 + state->count; ++i)
834 if (distance(c, table_entry(state, i)->color) < distance(c, table_entry(state, best)->color)) best = i;
835 const Entry *entry = table_entry(state, best);
836 wordf(output, "%s", entry->name);
837 output_color(output, entry->color);
838 wordf(output, "%lld", distance(c, entry->color));
839 break;
840 }
841 case OP_DISTANCE: {
842 Color other = args->colors[1];
843 Int r = c.x[0] - other.x[0], g = c.x[1] - other.x[1], blue = c.x[2] - other.x[2];
844 Int weight = (1024 + c.x[0] + other.x[0]) * r * r + 2048 * g * g
845 + (1534 - c.x[0] - other.x[0]) * blue * blue;
846 wordf(output, "%lld %lld", distance(c, other), integer_root(weight / 512));
847 break;
848 }
849 case OP_CONTRAST: output_contrast(output, c, args->colors[1]); break;
850 case OP_ACCESSIBLE: {
851 Color result = c, background = args->colors[1];
852 if (strcmp(contrast(c, background).verdict, "FAIL") == 0) {
853 Hsl value = hsl(c);
854 int found = 0;
855 for (Int step = 1; step <= 100 && !found; ++step)
856 for (int sign = -1; sign <= 1; sign += 2) {
857 Int lightness = value.l + sign * step;
858 if (lightness < 0 || lightness > 100) continue;
859 Color candidate = from_hsl(value.h, value.s, lightness);
860 if (strcmp(contrast(candidate, background).verdict, "FAIL") != 0) {
861 result = candidate;
862 found = 1;
863 break;
864 }
865 }
866 }
867 output_color(output, result);
868 output_contrast(output, result, background);
869 break;
870 }
871 case OP_BEST: {
872 int best = 1;
873 for (int i = 2; i < count; ++i)
874 if (contrast(args->colors[i], c).ratio > contrast(args->colors[best], c).ratio) best = i;
875 output_color(output, args->colors[best]);
876 output_contrast(output, args->colors[best], c);
877 break;
878 }
879 case OP_DESCRIBE: {
880 Hsl value = hsl(c);
881 if (value.l <= 5) wordf(output, "black");
882 else if (value.l >= 95) wordf(output, "white");
883 else {
884 const char *lightness = value.l <= 20 ? "very dark" : value.l <= 40 ? "dark"
885 : value.l <= 60 ? "medium" : value.l <= 80 ? "light" : "very light";
886 wordf(output, "%s", lightness);
887 if (value.s < 10) wordf(output, "gray");
888 else {
889 static const char *names[] = {"red","orange","yellow","chartreuse","green","spring-green",
890 "cyan","azure","blue","violet","magenta","rose"};
891 wordf(output, "%s %s", value.s < 40 ? "muted" : value.s < 80 ? "moderate" : "vivid",
892 names[((value.h + 15) % 360) / 30]);
893 }
894 }
895 break;
896 }
897 case OP_PALETTE: case OP_KMEANS: {
898 Color means[32];
899 int sizes[32], wanted = (int)a;
900 if (operation == OP_PALETTE) palette(args->colors + 1, count - 1, wanted, means, sizes);
901 else wordf(output, "%d", kmeans(args->colors + 1, count - 1, wanted, means, sizes));
902 for (int i = 0; i < wanted; ++i) {
903 char text[8];
904 color_text(means[i], text);
905 wordf(output, "%s:%d", text, sizes[i]);
906 }
907 break;
908 }
909 case OP_SORT: {
910 Color colors[32];
911 memcpy(colors, args->colors, count * sizeof *colors);
912 for (int i = 1; i < count; ++i) {
913 Color value = colors[i];
914 Hsl key = hsl(value);
915 int j = i;
916 while (j > 0) {
917 Hsl previous = hsl(colors[j - 1]);
918 int greater = previous.h > key.h || (previous.h == key.h &&
919 (previous.s > key.s || (previous.s == key.s && previous.l > key.l)));
920 if (!greater) break;
921 colors[j] = colors[j - 1];
922 --j;
923 }
924 colors[j] = value;
925 }
926 for (int i = 0; i < count; ++i) output_color(output, colors[i]);
927 break;
928 }
929 case OP_DITHER: {
930 Color error = rgb(0, 0, 0);
931 for (int i = 0; i < count; ++i) {
932 Color target = args->colors[i];
933 for (int j = 0; j < 3; ++j) target.x[j] = clamp(target.x[j] + error.x[j]);
934 int best = 0;
935 for (int j = 1; j < 16; ++j)
936 if (distance(target, builtins[j].color) < distance(target, builtins[best].color)) best = j;
937 wordf(output, "%s", builtins[best].name);
938 for (int j = 0; j < 3; ++j) error.x[j] = target.x[j] - builtins[best].color.x[j];
939 }
940 break;
941 }
942 case OP_RANDOM: {
943 Int state_value = a;
944 for (Int i = 0; i < b; ++i) {
945 state_value = state_value * 48271 % 2147483647;
946 Int h = state_value % 360;
947 state_value = state_value * 48271 % 2147483647;
948 Int s = 40 + state_value % 61;
949 state_value = state_value * 48271 % 2147483647;
950 Int l = 25 + state_value % 51;
951 output_color(output, from_hsl(h, s, l));
952 }
953 break;
954 }
955 }
956}

958static int process(State *state, const char **tokens, int count, Text *output) {
959 int operation = -1;
960 for (int i = 0; i < 53; ++i)
961 if (strcmp(tokens[0], specifications[i].name) == 0) {
962 operation = i;
963 break;
964 }
965 if (operation < 0) return 0;
966 Arguments arguments = {0};
967 if (!prepare(state, &specifications[operation], operation, tokens + 1, count - 1, &arguments)) return 0;
968 execute(state, operation, tokens + 1, count - 1, &arguments, output);
969 return 1;
970}

972int main(void) {
973 char line[1004];
974 if (!fgets(line, sizeof line, stdin)) return 0;
975 int commands = atoi(line);
976 State state = {0};
977 for (int i = 0; i < commands; ++i) {
978 if (!fgets(line, sizeof line, stdin)) break;
979 line[strcspn(line, "\n")] = '\0';
980 const char *tokens[501];
981 int count = 0;
982 for (char *word = strtok(line, " "); word; word = strtok(NULL, " ")) tokens[count++] = word;
983 Text output = {{0}, 0};
984 if (process(&state, tokens, count, &output)) puts(output.data);
985 else puts("INVALID");
986 }
987 return 0;
