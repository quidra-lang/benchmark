1private class Problem(val reason: String) : RuntimeException(reason)

3private fun fail(reason: String): Nothing = throw Problem(reason)

5private data class Term(val kind: String, val name: String = "", val left: Term? = null, val right: Term? = null, val number: Int = 0)
6
7private fun variable(name: String) = Term("var", name)
8
9private fun lambda(name: String, body: Term) = Term("lam", name, body)
10
11private fun application(left: Term, right: Term) = Term("app", left = left, right = right)
12
13private fun free(t: Term): Set<String> = when (t.kind) {
14 "var" -> setOf(t.name)
15 "lam" -> free(t.left!!) - t.name
16 "app" -> free(t.left!!) + free(t.right!!)
17 else -> emptySet()
18}
19
20private fun references(t: Term): List<String> = when (t.kind) {
21 "macro" -> listOf(t.name)
22 "lam" -> references(t.left!!)
23 "app" -> references(t.left!!) + references(t.right!!)
24 else -> emptyList()
25}

27private fun printed(t: Term, end: Boolean = true, db: Boolean = false): String = when (t.kind) {
28 "var", "macro" -> t.name
29 "index" -> t.number.toString()
30 "num" -> "#${t.number}"
31 "lam" -> {
32 var body = t
33 val names = mutableListOf<String>()
34 while (body.kind == "lam") {
35 names.add(body.name)
36 body = body.left!!
37 }
38 val s = if (db) "\\".repeat(names.size) + printed(body, true, true) else "\\" + names.joinToString(" ") + ". " + printed(body)
39 if (end) s else "($s)"
40 } else -> {
41 val a = t.left!!
42 val b = t.right!!
43 val left = if (a.kind == "lam") "(${printed(a, true, db)})" else printed(a, false, db)
44 val right = if (b.kind == "app") "(${printed(b, true, db)})" else printed(b, end, db)
45 "$left $right"
46 }
47}

49private fun sequenceName(n: Int) = ('a'.code + n % 26).toChar().toString() + if (n < 26) "" else (n / 26).toString()
50
51private fun canonical(t: Term): Term {
52 val f = free(t)
53 var next = 0
54
55 fun visit(s: Term, env: List<Pair<String, String>>): Term = when (s.kind) {
56 "var" -> variable(env.lastOrNull {
57 it.first == s.name
58 }?.second ?: s.name)
59 "lam" -> {
60 while (sequenceName(next) in f) next++
61 val name = sequenceName(next++)
62 lambda(name, visit(s.left!!, env + (s.name to name)))
63 }
64 "app" -> application(visit(s.left!!, env), visit(s.right!!, env))
65 else -> s
66 }
67 return visit(t, emptyList())
68}

70private fun debruijn(t: Term, env: List<String> = emptyList()): Term = when (t.kind) {
71 "var" -> {
72 val i = env.indexOfLast {
73 it == t.name
74 }
75 if (i < 0) t else Term("index", number = env.lastIndex - i)
76 }
77 "lam" -> lambda("", debruijn(t.left!!, env + t.name))
78 "app" -> application(debruijn(t.left!!, env), debruijn(t.right!!, env))
79 else -> t
80}

82private fun substitute(t: Term, x: String, n: Term): Term = when (t.kind) {
83 "var" -> if (t.name == x) n else t
84 "app" -> application(substitute(t.left!!, x, n), substitute(t.right!!, x, n))
85 "lam" -> {
86 val body = t.left!!
87 if (t.name == x || x !in free(body)) t
88 else if (t.name !in free(n)) lambda(t.name, substitute(body, x, n))
89 else {
90 val forbidden = free(n) + free(body)
91 val base = t.name.trimEnd {
92 it in '0'..'9'
93 }
94 var i = 1
95 while (base + i in forbidden) i++
96 val z = base + i
97 lambda(z, substitute(substitute(body, t.name, variable(z)), x, n))
98 }
99 } else -> t
100}

102private fun eta(t: Term): Term? {
103 if (t.kind != "lam") return null
104 val b = t.left!!
105 return if (b.kind == "app" && b.right!!.kind == "var" && b.right.name == t.name && t.name !in free(b.left!!)) b.left else null
106}

108private fun step(t: Term, strategy: String, useEta: Boolean): Pair<Term, String>? {
109 if (t.kind == "app") {
110 val a = t.left!!
111 val b = t.right!!
112 if (strategy in setOf("NORMAL", "CBN") && a.kind == "lam") return substitute(a.left!!, a.name, b) to "beta"
113 step(a, strategy, useEta)?.let {
114 return application(it.first, b) to it.second
115 }
116 if (strategy != "CBN") step(b, strategy, useEta)?.let {
117 return application(a, it.first) to it.second
118 }
119 if (strategy in setOf("APPLICATIVE", "CBV") && a.kind == "lam" && (strategy != "CBV" || b.kind in setOf("var", "lam"))) return substitute(a.left!!, a.name, b) to "beta"
120 } else if (t.kind == "lam" && strategy in setOf("NORMAL", "APPLICATIVE")) {
121 if (strategy == "NORMAL" && useEta) eta(t)?.let {
122 return it to "eta"
123 }
124 step(t.left!!, strategy, useEta)?.let {
125 return lambda(t.name, it.first) to it.second
126 }
127 if (strategy == "APPLICATIVE" && useEta) eta(t)?.let {
128 return it to "eta"
129 }
130 }
131 return null
132}

134private class Ty(val id: Int = -1, val left: Ty? = null, val right: Ty? = null)
135
136private fun infer(t: Term): String {
137 var next = 0
138 val bindings = mutableMapOf<Int, Ty>()
139 val globals = mutableMapOf<String, Ty>()
140
141 fun fresh() = Ty(next++)
142
143 fun resolve(t: Ty): Ty = if (t.id >= 0 && t.id in bindings) resolve(bindings.getValue(t.id)) else t
144
145 fun occurs(id: Int, t: Ty): Boolean {
146 val a = resolve(t)
147 return if (a.id >= 0) a.id == id else occurs(id, a.left!!) || occurs(id, a.right!!)
148 }
149
150 fun unify(a0: Ty, b0: Ty) {
151 val a = resolve(a0)
152 val b = resolve(b0)
153 if (a === b || a.id >= 0 && a.id == b.id) return
154 if (a.id >= 0) {
155 if (occurs(a.id, b)) throw IllegalStateException()
156 bindings[a.id] = b
157 } else if (b.id >= 0) {
158 if (occurs(b.id, a)) throw IllegalStateException()
159 bindings[b.id] = a
160 } else {
161 unify(a.left!!, b.left!!)
162 unify(a.right!!, b.right!!)
163 }
164 }
165
166 fun visit(t: Term, env: List<Pair<String, Ty>>): Ty = when (t.kind) {
167 "var" -> env.lastOrNull {
168 it.first == t.name
169 }?.second ?: globals.getOrPut(t.name) {
170 fresh()
171 }
172 "lam" -> {
173 val a = fresh()
174 Ty(left = a, right = visit(t.left!!, env + (t.name to a)))
175 } else -> {
176 val a = visit(t.left!!, env)
177 val b = visit(t.right!!, env)
178 val result = fresh()
179 unify(a, Ty(left = b, right = result))
180 result
181 }
182 }
183 return try {
184 val result = visit(t, emptyList())
185 val names = mutableMapOf<Int, String>()
186
187 fun show(t: Ty): String {
188 val a = resolve(t)
189 if (a.id >= 0) return names.getOrPut(a.id) {
190 sequenceName(names.size)
191 }
192 val l = resolve(a.left!!)
193 return (if (l.id < 0) "(${show(l)})" else show(l)) + " -> " + show(a.right!!)
194 }
195 show(result)
196 } catch (_: IllegalStateException) {
197 "not typable"
198 }
199}

201private fun abstract(x: String, e: Term): Term {
202 if (x !in free(e)) return application(Term("macro", "K"), e)
203 if (e.kind == "var" && e.name == x) return Term("macro", "I")
204 if (e.right!!.kind == "var" && e.right.name == x && x !in free(e.left!!)) return e.left
205 return application(application(Term("macro", "S"), abstract(x, e.left!!)), abstract(x, e.right))
206}
207
208private fun ski(t: Term): Term = when (t.kind) {
209 "var" -> t
210 "lam" -> abstract(t.name, ski(t.left!!))
211 else -> application(ski(t.left!!), ski(t.right!!))
212}
213
214private fun skiStep(t: Term): Pair<Term, String>? {
215 var h = t
216 val args = mutableListOf<Term>()
217 while (h.kind == "app") {
218 args.add(0, h.right!!)
219 h = h.left!!
220 }
221
222 fun build(head: Term, rest: List<Term>) = rest.fold(head, ::application)
223 if (h.kind == "macro") {
224 when (h.name) {
225 "I" -> if (args.size >= 1) return build(args[0], args.drop(1)) to "I"
226 "K" -> if (args.size >= 2) return build(args[0], args.drop(2)) to "K"
227 "S" -> if (args.size >= 3) return build(application(application(args[0], args[2]), application(args[1], args[2])), args.drop(3)) to "S"
228 }
229 }
230 for (i in args.indices) skiStep(args[i])?.let {
231 args[i] = it.first
232 return build(h, args) to it.second
233 }
234 return null
235}

237private fun church(t: Term): String {
238 if (t.kind == "lam" && t.left!!.kind == "lam") {
239 val f = t.name
240 val x = t.left.name
241 var b = t.left.left!!
242 var count = 0
243 while (b.kind == "app" && b.left!!.kind == "var" && b.left.name == f) {
244 count++
245 b = b.right!!
246 }
247 if (b.kind == "var" && b.name == x) return count.toString()
248 if (t.left.left!!.kind == "var" && t.left.left.name == f) return "true"
249 }
250 if (t.kind == "lam") {
251 val b = t.left!!
252 if (b.kind == "app" && b.left!!.kind == "app" && b.left.left!!.kind == "var" && b.left.left.name == t.name) {
253 val p = b.left.right!!
254 val q = b.right!!
255 if (t.name !in free(p) && t.name !in free(q)) return "(${church(p)}, ${church(q)})"
256 }
257 }
258 return printed(canonical(t))
259}

261private data class Token(val kind: String, val text: String, val column: Int)
262
263private class Parser(line: String, argument: String, offset: Int) {
264 val tokens = mutableListOf<Token>()
265 var at = 0
266 init {
267 var i = 0
268 while (i < argument.length) {
269 val c = argument[i]
270 if (c == ' ') {
271 i++
272 continue
273 }
274 val start = i
275 val kind = when {
276 c in "\\.()=;" -> {
277 i++
278 c.toString()
279 }
280 c in 'a'..'z' -> {
281 i++
282 while (i < argument.length && (argument[i] in 'a'..'z' || argument[i] in '0'..'9')) i++
283 "var"
284 }
285 c in 'A'..'Z' -> {
286 i++
287 while (i < argument.length && (argument[i] in 'A'..'Z' || argument[i] in '0'..'9' || argument[i] == '_')) i++
288 "macro"
289 }
290 c == '#' && i + 1 < argument.length && argument[i + 1] in '0'..'9' -> {
291 i += 2
292 while (i < argument.length && argument[i] in '0'..'9') i++
293 "num"
294 } else -> fail("column ${offset + i}: unexpected character '$c'")
295 }
296 tokens.add(Token(kind, argument.substring(start, i), offset + start))
297 }
298 tokens.add(Token("end", "", line.length + 1))
299 }
300
301 fun expect(kind: String): Token {
302 val t = tokens[at]
303 if (t.kind != kind) fail("column ${t.column}: expected " + when (kind) {
304 "var" -> "variable"
305 "macro" -> "macro name"
306 else -> "'$kind'"
307 })
308 at++
309 return t
310 }
311
312 fun term(): Term {
313 val elements = mutableListOf<Term>()
314 while (tokens[at].kind in setOf("var", "macro", "num", "(")) {
315 val t = tokens[at++]
316 elements.add(when (t.kind) {
317 "(" -> {
318 val e = term()
319 expect(")")
320 e
321 }
322 "num" -> Term("num", number = t.text.drop(1).toInt())
323 else -> Term(t.kind, t.text)
324 })
325 }
326 if (tokens[at].kind == "\\") {
327 at++
328 val names = mutableListOf(expect("var").text)
329 while (tokens[at].kind == "var") names.add(tokens[at++].text)
330 expect(".")
331 var body = term()
332 for (n in names.asReversed()) body = lambda(n, body)
333 elements.add(body)
334 }
335 if (elements.isEmpty()) fail("column ${tokens[at].column}: expected term")
336 return elements.drop(1).fold(elements[0], ::application)
337 }
338
339 fun end() {
340 val t = tokens[at]
341 if (t.kind != "end") fail("column ${t.column}: unexpected '${t.text}'")
342 }
343}

347private class Workbench {
348 val macros = sortedMapOf<String, Term>()
349 val reserved = setOf("S", "K", "I", "NORMAL", "APPLICATIVE", "CBN", "CBV")
350 var display = false
351 var etaMode = false
352 var limit = 10000
353 init {
354 val x = variable("x")
355 val y = variable("y")
356 val z = variable("z")
357 macros["I"] = lambda("x", x)
358 macros["K"] = lambda("x", lambda("y", x))
359 macros["S"] = lambda("x", lambda("y", lambda("z", application(application(x, z), application(y, z)))))
360 }

362 fun expand(t: Term): Term = when (t.kind) {
363 "macro" -> expand(macros[t.name] ?: fail("unbound macro ${t.name}"))
364 "num" -> {
365 var b = variable("x")
366 repeat(t.number) {
367 b = application(variable("f"), b)
368 }
369 lambda("f", lambda("x", b))
370 }
371 "lam" -> lambda(t.name, expand(t.left!!))
372 "app" -> application(expand(t.left!!), expand(t.right!!))
373 else -> t
374 }

376 fun depends(name: String, target: String): Boolean {
377 val visited = mutableSetOf<String>()
378
379 fun visit(n: String): Boolean {
380 if (!visited.add(n)) return false
381 return macros[n]?.let {
382 references(it).any {
383 r -> r == target || visit(r)
384 }
385 } ?: false
386 }
387 return visit(name)
388 }

390 fun reduce(t: Term, strategy: String, useEta: Boolean, output: MutableList<String>?, combinators: Boolean = false): Pair<Term, Int>? {
391 var current = t
392 var count = 0
393 while (true) {
394 val next = if (combinators) skiStep(current) else step(current, strategy, useEta)
395 if (next == null) return current to count
396 if (count == limit) return null
397 current = next.first
398 count++
399 if (display && output != null) output.add("step $count (${next.second}): ${printed(current)}")
400 }
401 }

403 fun kam(t: Term): List<String> {
404 var code = debruijn(t)
405 var env = emptyList<Closure>()
406 val stack = mutableListOf<Closure>()
407 var count = 0
408 val output = mutableListOf<String>()
409 while (true) {
410 val kind = when {
411 code.kind == "app" -> "push"
412 code.kind == "lam" && stack.isNotEmpty() -> "grab"
413 code.kind == "index" -> "access"
414 else -> break
415 }
416 if (count == limit) {
417 output.add("no result within $limit transitions")
418 return output
419 }
420 when (kind) {
421 "push" -> {
422 stack.add(Closure(code.right!!, env))
423 code = code.left!!
424 }
425 "grab" -> {
426 env = listOf(stack.removeAt(stack.lastIndex)) + env
427 code = code.left!!
428 }
429 "access" -> {
430 val c = env[code.number]
431 code = c.code
432 env = c.env
433 }
434 }
435 count++
436 if (display) output.add("step $count ($kind): ${printed(code, db = true)} [env ${env.size}, stack ${stack.size}]")
437 }

439 fun reify(initialTerm: Term, initialEnvironment: List<Closure>, initialDepth: Int): Term {
440 var term = initialTerm
441 var environment = initialEnvironment
442 var depth = initialDepth
443 while (term.kind == "index" && term.number >= depth) {
444 val closure = environment[term.number - depth]
445 term = closure.code
446 environment = closure.env
447 depth = 0
448 }
449 return when (term.kind) {
450 "lam" -> lambda("", reify(term.left!!, environment, depth + 1))
451 "app" -> application(reify(term.left!!, environment, depth), reify(term.right!!, environment, depth))
452 else -> term
453 }
454 }
455 var result = reify(code, env, 0)
456 for (c in stack.asReversed()) result = application(result, reify(c.code, c.env, 0))
457 val f = free(result)
458 var nextName = 0

460 fun named(t: Term, bound: List<String>): Term = when (t.kind) {
461 "index" -> variable(bound[bound.lastIndex - t.number])
462 "lam" -> {
463 while (sequenceName(nextName) in f) nextName++
464 val n = sequenceName(nextName++)
465 lambda(n, named(t.left!!, bound + n))
466 }
467 "app" -> application(named(t.left!!, bound), named(t.right!!, bound))
468 else -> t
469 }
470 output.add("result: ${printed(named(result, emptyList()))} (transitions: $count)")
471 return output
472 }

474 fun command(line: String): List<String> {
475 val command = line.substringBefore(' ')
476 val arg = if (' ' in line) line.substringAfter(' ') else ""
477 val known = setOf("let", "unlet", "macros", "steps", "eta", "limit", "expand", "subst", "reduce", "equal", "alpha", "debruijn", "fv", "size", "depth", "type", "church", "ski", "skireduce", "kam")
478 if (command !in known) fail("unknown command $command")
479 val words = arg.split(' ').filter {
480 it.isNotEmpty()
481 }
482 if (command in setOf("steps", "eta")) {
483 if (words.size != 1 || words[0] !in setOf("on", "off")) fail("bad argument")
484 if (command == "steps") display = words[0] == "on" else etaMode = words[0] == "on"
485 return emptyList()
486 }
487 if (command == "limit") {
488 if (words.size != 1 || words[0].length !in 1..5 || words[0].any {
489 it !in '0'..'9'
490 } || words[0].toInt() !in 1..10000) fail("bad argument")
491 limit = words[0].toInt()
492 return emptyList()
493 }
494 if (command == "macros") {
495 if (words.isNotEmpty()) fail("bad argument")
496 return macros.filterKeys {
497 it !in reserved
498 }.map {
499 (n, t) -> "$n = ${printed(t)}"
500 }.ifEmpty {
501 listOf("(none)")
502 }
503 }
504 if (command == "unlet") {
505 if (words.size != 1 || !words[0].matches(Regex("[A-Z][A-Z0-9_]*"))) fail("bad argument")
506 val name = words[0]
507 if (name in reserved) fail("reserved name $name")
508 if (name !in macros) fail("unbound macro $name")
509 macros.keys.firstOrNull {
510 it !in reserved && name in references(macros.getValue(it))
511 }?.let {
512 fail("$name is used by $it")
513 }
514 macros.remove(name)
515 return listOf("removed $name")
516 }
517 val p = Parser(line, arg, if (' ' in line) command.length + 2 else line.length + 1)
518 var name = ""
519 var strategy = "NORMAL"
520 val terms = mutableListOf<Term>()
521 when (command) {
522 "let" -> {
523 name = p.expect("macro").text
524 p.expect("=")
525 terms.add(p.term())
526 }
527 "subst" -> {
528 name = p.expect("var").text
529 p.expect("=")
530 terms.add(p.term())
531 p.expect(";")
532 terms.add(p.term())
533 }
534 "equal", "alpha" -> {
535 terms.add(p.term())
536 p.expect(";")
537 terms.add(p.term())
538 }
539 "reduce" -> {
540 if (p.tokens[p.at].kind == "macro" && p.tokens[p.at].text in setOf("NORMAL", "APPLICATIVE", "CBN", "CBV")) strategy = p.tokens[p.at++].text
541 terms.add(p.term())
542 } else -> terms.add(p.term())
543 }
544 p.end()
545 if (command == "let") {
546 val t = terms[0]
547 if (name in reserved) fail("reserved name $name")
548 for (r in references(t)) {
549 if (r == name || depends(r, name)) fail("recursive definition of $name")
550 if (r !in macros) fail("unbound macro $r")
551 }
552 free(t).minOrNull()?.let {
553 fail("free variable $it in definition of $name")
554 }
555 val existed = name in macros
556 macros[name] = t
557 return listOf("${if (existed) "redefined" else "defined"} $name")
558 }
559 val expanded = terms.map(::expand)
560 val t = expanded[0]
561 val failure = "no normal form within $limit steps"
562 return when (command) {
563 "expand" -> listOf(printed(t))
564 "subst" -> listOf(printed(substitute(expanded[1], name, t)))
565 "alpha" -> listOf(if (debruijn(t) == debruijn(expanded[1])) "yes" else "no")
566 "debruijn" -> listOf(printed(debruijn(t), db = true))
567 "fv" -> listOf(free(t).sorted().joinToString(" ").ifEmpty {
568 "(none)"
569 })
570 "size", "depth" -> {
571
572 fun measure(t: Term): Int = if (t.kind == "var") 1 else if (t.kind == "lam") 1 + measure(t.left!!) else 1 + if (command == "size") measure(t.left!!) + measure(t.right!!) else maxOf(measure(t.left!!), measure(t.right!!))
573 listOf(measure(t).toString())
574 }
575 "type" -> listOf(infer(t))
576 "ski" -> listOf(printed(ski(t)))
577 "kam" -> kam(t)
578 "equal" -> {
579 val a = reduce(t, "NORMAL", etaMode, null)
580 val b = reduce(expanded[1], "NORMAL", etaMode, null)
581 listOf(if (a == null || b == null) "unknown" else if (debruijn(a.first) == debruijn(b.first)) "yes" else "no")
582 }
583 "church" -> {
584 val result = reduce(t, "NORMAL", false, null)
585 listOf(if (result == null) failure else church(canonical(result.first)))
586 } else -> {
587 val output = mutableListOf<String>()
588 val combinators = command == "skireduce"
589 val result = reduce(if (combinators) ski(t) else t, strategy, etaMode, output, combinators)
590 output.add(if (result == null) failure else "result: ${printed(if (combinators) result.first else canonical(result.first))} (steps: ${result.second})")
591 output
592 }
593 }
594 }
595}
