15func free(_ t: Term) -> Set<String> {
16 switch t {
17 case let .variable(n):
18 return [n]
19 case let .lambda(n, b):
20 return free(b).subtracting([n])
21 case let .application(a, b):
22 return free(a).union(free(b))
23 default:
24 return []
25 }
26}

28func references(_ t: Term) -> [String] {
29 switch t {
30 case let .macro(n):
31 return [n]
32 case let .lambda(_, b):
33 return references(b)
34 case let .application(a, b):
35 return references(a) + references(b)
36 default:
37 return []

41func printed(_ t: Term, _ end: Bool = true, _ db: Bool = false) -> String {
42 switch t {
43 case let .variable(n), let .macro(n):
44 return n
45 case let .numeral(n):
46 return "#\(n)"
47 case let .index(i):
48 return String(i)
49 case .lambda:
50 var body = t
51 var names = [String]()
52 while case let .lambda(n, b) = body {
53 names.append(n)
54 body = b
55 }
56 let s = db ? String(repeating: "\\", count: names.count) + printed(body, true, true) : "\\" + names.joined(separator: " ") + ". " + printed(body)
57 return end ? s : "(\(s))"
58 case let .application(a, b):
59 let left: String
60 if case .lambda = a {
61 left = "(\(printed(a, true, db)))"
62 } else {
63 left = printed(a, false, db)
64 }
65 let right: String
66 if case .application = b {
67 right = "(\(printed(b, true, db)))"
68 } else {
69 right = printed(b, end, db)
70 }
71 return left + " " + right
72 }

121func substitute(_ t: Term, _ x: String, _ n: Term) -> Term {
122 switch t {
123 case let .variable(v):
124 return v == x ? n : t
125 case let .application(a, b):
126 return .application(substitute(a, x, n), substitute(b, x, n))
127 case let .lambda(y, b):
128 let fb = free(b)
129 let fnn = free(n)
130 if y == x || !fb.contains(x) {
131 return t
132 }
133 if !fnn.contains(y) {
134 return .lambda(y, substitute(b, x, n))
135 }
136 var base = y
137 while let c = base.utf8.last, c >= 48 && c <= 57 {
138 base.removeLast()
139 }
140 var i = 1
141 while fb.contains(base + String(i)) || fnn.contains(base + String(i)) {
142 i += 1
143 }
144 let z = base + String(i)
145 return .lambda(z, substitute(substitute(b, y, .variable(z)), x, n))
146 default:
147 return t
148 }
149}

158func step(_ t: Term, _ strategy: String, _ useEta: Bool) -> (Term, String)? {
159 switch t {
160 case let .application(a, b):
161 if ["NORMAL", "CBN"].contains(strategy), case let .lambda(x, body) = a {
162 return (substitute(body, x, b), "beta")
163 }
164 if let (n, kind) = step(a, strategy, useEta) {
165 return (.application(n, b), kind)
166 }
167 if strategy != "CBN", let (n, kind) = step(b, strategy, useEta) {
168 return (.application(a, n), kind)
169 }
170 if ["APPLICATIVE", "CBV"].contains(strategy), case let .lambda(x, body) = a {
171 let value: Bool
172 switch b {
173 case .variable, .lambda:
174 value = true
175 default:
176 value = false
177 }
178 if strategy != "CBV" || value {
179 return (substitute(body, x, b), "beta")
180 }
181 }
182 case let .lambda(x, b) where ["NORMAL", "APPLICATIVE"].contains(strategy):
183 if strategy == "NORMAL" && useEta, let n = eta(t) {
184 return (n, "eta")
185 }
186 if let (n, kind) = step(b, strategy, useEta) {
187 return (.lambda(x, n), kind)
188 }
189 if strategy == "APPLICATIVE" && useEta, let n = eta(t) {
190 return (n, "eta")
191 }
192 default:
193 break
194 }
195 return nil
196}

225 func resolve(_ t: Ty) -> Ty {

232 func occurs(_ id: Int, _ t: Ty) -> Bool {
233 let a = resolve(t)
234 return a.id >= 0 ? a.id == id : occurs(id, a.left!) || occurs(id, a.right!)
235 }

237 func unify(_ a0: Ty, _ b0: Ty) throws {
238 let a = resolve(a0)
239 let b = resolve(b0)
240 if a === b || a.id >= 0 && a.id == b.id {
241 return
242 }
243 if a.id >= 0 {
244 if occurs(a.id, b) {
245 throw Untypable()
246 }
247 bindings[a.id] = b
248 } else if b.id >= 0 {
249 if occurs(b.id, a) {
250 throw Untypable()
251 }
252 bindings[b.id] = a
253 } else {
254 try unify(a.left!, b.left!)
255 try unify(a.right!, b.right!)
256 }
257 }

259 func visit(_ t: Term, _ env: [(String, Ty)]) throws -> Ty {
260 switch t {
261 case let .variable(n):
262 if let a = env.last(where: {
263 $0.0 == n
264 })?.1 {
265 return a
266 }
267 if let a = globals[n] {
268 return a
269 }
270 let a = fresh()
271 globals[n] = a
272 return a
273 case let .lambda(n, b):
274 let a = fresh()
275 return Ty(-1, a, try visit(b, env + [(n, a)]))
276 case let .application(a, b):
277 let f = try visit(a, env)
278 let x = try visit(b, env)
279 let result = fresh()
280 try unify(f, Ty(-1, x, result))
281 return result
282 default:
283 fatalError("pure term required")
284 }
285 }

287 let result = try visit(t, [])
288 var names = [Int: String]()

290 func show(_ t: Ty) -> String {
291 let a = resolve(t)
292 if a.id >= 0 {
293 if names[a.id] == nil {
294 names[a.id] = sequenceName(names.count)

296 return names[a.id]!
297 }
298 let l = resolve(a.left!)
299 let text = show(l)
300 return (l.id < 0 ? "(\(text))" : text) + " -> " + show(a.right!)
301 }

303 } catch {
304 return "not typable"

308func abstract(_ x: String, _ e: Term) -> Term {
309 if !free(e).contains(x) {
310 return .application(.macro("K"), e)
311 }
312 if case let .variable(v) = e, v == x {
313 return .macro("I")
314 }
315 guard case let .application(a, b) = e else {
316 fatalError("abstraction invariant")
317 }
318 if case let .variable(v) = b, v == x, !free(a).contains(x) {
319 return a
320 }
321 return .application(.application(.macro("S"), abstract(x, a)), abstract(x, b))
322}

324func ski(_ t: Term) -> Term {
325 switch t {
326 case .variable:
327 return t
328 case let .lambda(n, b):
329 return abstract(n, ski(b))
330 case let .application(a, b):
331 return .application(ski(a), ski(b))
332 default:
333 fatalError("pure term required")
334 }

343func skiStep(_ t: Term) -> (Term, String)? {
344 var h = t
345 var args = [Term]()
346 while case let .application(a, b) = h {
347 args.insert(b, at: 0)
348 h = a
349 }
350 if case let .macro(n) = h {
351 if n == "I" && args.count >= 1 {
352 return (build(args[0], Array(args.dropFirst())), "I")
353 }
354 if n == "K" && args.count >= 2 {
355 return (build(args[0], Array(args.dropFirst(2))), "K")
356 }
357 if n == "S" && args.count >= 3 {
358 return (build(.application(.application(args[0], args[2]), .application(args[1], args[2])), Array(args.dropFirst(3))), "S")
359 }
360 }
361 for i in args.indices {
362 if let (n, kind) = skiStep(args[i]) {
363 args[i] = n
364 return (build(h, args), kind)
365 }
366 }
367 return nil
368}

370func church(_ t: Term) -> String {
371 if case let .lambda(f, .lambda(x, body)) = t {
372 var b = body
373 var count = 0
374 while case let .application(.variable(v), n) = b, v == f {
375 count += 1
376 b = n
377 }
378 if case let .variable(v) = b, v == x {
379 return String(count)
380 }
381 if case let .variable(v) = body, v == f {
382 return "true"
383 }
384 }
385 if case let .lambda(f, .application(.application(.variable(v), p), q)) = t, f == v, !free(p).contains(f), !free(q).contains(f) {
386 return "(\(church(p)), \(church(q)))"
387 }
388 return printed(canonical(t))
389}

404
405 func digit(_ c: UInt8) -> Bool {
406 c >= 48 && c <= 57
407 }
408
409 func lower(_ c: UInt8) -> Bool {
410 c >= 97 && c <= 122
411 }
412
413 func upper(_ c: UInt8) -> Bool {
414 c >= 65 && c <= 90
415 }
416 while i < b.count {
417 let c = b[i]
418 if c == 32 {
419 i += 1
420 continue
421 }
422 let start = i
423 let kind: String
424 if Array("\\.()=;".utf8).contains(c) {
425 i += 1
426 kind = String(UnicodeScalar(c))
427 } else if lower(c) {
428 i += 1
429 while i < b.count && (lower(b[i]) || digit(b[i])) {
430 i += 1
431 }
432 kind = "var"
433 } else if upper(c) {
434 i += 1
435 while i < b.count && (upper(b[i]) || digit(b[i]) || b[i] == 95) {
436 i += 1
437 }
438 kind = "macro"
439 } else if c == 35 && i + 1 < b.count && digit(b[i + 1]) {
440 i += 2
441 while i < b.count && digit(b[i]) {
442 i += 1
443 }
444 kind = "num"
445 } else {
446 try fail("column \(offset + i): unexpected character '\(Character(UnicodeScalar(c)))'")
447 }
448 tokens.append(Token(kind: kind, text: String(decoding: b[start..<i], as: UTF8.self), column: offset + start))
449 }
450 tokens.append(Token(kind: "end", text: "", column: line.utf8.count + 1))
451 }

453 func expect(_ kind: String) throws -> Token {
454 let t = tokens[at]
455 if t.kind != kind {
456 let expected = kind == "var" ? "variable" : kind == "macro" ? "macro name" : "'\(kind)'"
457 try fail("column \(t.column): expected \(expected)")
458 }
459 at += 1
460 return t
461 }
462
463 func term() throws -> Term {
464 var elements = [Term]()
465 while ["var", "macro", "num", "("].contains(tokens[at].kind) {
466 let t = tokens[at]
467 at += 1
468 switch t.kind {
469 case "(":
470 let e = try term()
471 _ = try expect(")")
472 elements.append(e)
473 case "num":
474 elements.append(.numeral(Int(t.text.dropFirst())!))
475 case "var":
476 elements.append(.variable(t.text))
477 default:
478 elements.append(.macro(t.text))
479 }
480 }
481 if tokens[at].kind == "\\" {
482 at += 1
483 var names = [try expect("var").text]
484 while tokens[at].kind == "var" {
485 names.append(tokens[at].text)
486 at += 1
487 }
488 _ = try expect(".")
489 var body = try term()
490 for n in names.reversed() {
491 body = .lambda(n, body)
492 }
493 elements.append(body)
494 }
495 if elements.isEmpty {
496 try fail("column \(tokens[at].column): expected term")
497 }
498 return build(elements[0], Array(elements.dropFirst()))
499 }
500
501 func end() throws {
502 let t = tokens[at]
503 if t.kind != "end" {
504 try fail("column \(t.column): unexpected '\(t.text)'")
505 }
506 }
507}

519func reify(_ initialTerm: Term, _ initialEnvironment: [Closure], _ initialDepth: Int = 0) -> Term {
520 var term = initialTerm
521 var environment = initialEnvironment
522 var depth = initialDepth
523 while case let .index(i) = term {
524 if i < depth {
525 break
526 }
527 let closure = environment[i - depth]
528 term = closure.code
529 environment = closure.env
530 depth = 0
531 }
532 switch term {
533 case let .lambda(_, body):
534 return .lambda("", reify(body, environment, depth + 1))
535 case let .application(left, right):
536 return .application(reify(left, environment, depth), reify(right, environment, depth))
537 default:
538 return term
539 }
540}

614 func reduce(_ term: Term, _ strategy: String, _ useEta: Bool, _ show: Bool, _ combinators: Bool, _ output: inout [String]) -> (Term, Int)? {
615 var t = term
616 var count = 0
617 while true {
618 let next = combinators ? skiStep(t) : step(t, strategy, useEta)
619 guard let (n, kind) = next else {
620 return (t, count)
621 }
622 if count == limit {
623 return nil
624 }
625 t = n
626 count += 1
627 if show && display {
628 output.append("step \(count) (\(kind)): \(printed(t))")
629 }
630 }

633 func kam(_ term: Term) -> [String] {
634 var code = debruijn(term)
635 var env = [Closure]()
636 var stack = [Closure]()
637 var count = 0
638 var out = [String]()
639 while true {
640 let kind: String
641 switch code {
642 case .application:
643 kind = "push"
644 case .lambda where !stack.isEmpty:
645 kind = "grab"
646 case .index:
647 kind = "access"
648 default:
649 kind = ""
650 }
651 if kind.isEmpty {
652 break
653 }
654 if count == limit {
655 out.append("no result within \(limit) transitions")
656 return out
657 }
658 switch code {
659 case let .application(a, b):
660 stack.append(Closure(b, env))
661 code = a
662 case let .lambda(_, b):
663 env.insert(stack.removeLast(), at: 0)
664 code = b
665 case let .index(i):
666 let c = env[i]
667 code = c.code
668 env = c.env
669 default:
670 break
671 }
672 count += 1
673 if display {
674 out.append("step \(count) (\(kind)): \(printed(code, true, true)) [env \(env.count), stack \(stack.count)]")
675 }
676 }
677 var result = reify(code, env)
678 for c in stack.reversed() {
679 result = .application(result, reify(c.code, c.env))
680 }
681 out.append("result: \(printed(namedDB(result))) (transitions: \(count))")
682 return out
683 }

685 func command(_ line: String) throws -> [String] {
686 let parts = line.split(separator: " ", maxSplits: 1, omittingEmptySubsequences: false)
687 let command = String(parts[0])
688 let arg = parts.count == 2 ? String(parts[1]) : ""
689 let known: Set<String> = ["let", "unlet", "macros", "steps", "eta", "limit", "expand", "subst", "reduce", "equal", "alpha", "debruijn", "fv", "size", "depth", "type", "church", "ski", "skireduce", "kam"]
690 if !known.contains(command) {
691 try fail("unknown command \(command)")
692 }
693 let words = arg.split(separator: " ").map(String.init)
694 if command == "steps" || command == "eta" {
695 if words.count != 1 || !["on", "off"].contains(words[0]) {
696 try fail("bad argument")
697 }
698 if command == "steps" {
699 display = words[0] == "on"
700 } else {
701 etaMode = words[0] == "on"
702 }
703 return []
704 }
705 if command == "limit" {
706 if words.count != 1 || words[0].count < 1 || words[0].count > 5 || !words[0].utf8.allSatisfy( {
707 $0 >= 48 && $0 <= 57
708 }) || Int(words[0])! < 1 || Int(words[0])! > 10000 {
709 try fail("bad argument")
710 }
711 limit = Int(words[0])!
712 return []
713 }
714 if command == "macros" {
715 if !words.isEmpty {
716 try fail("bad argument")
717 }
718 let names = macros.keys.filter {
719 !reserved.contains($0)
720 }.sorted()
721 return names.isEmpty ? ["(none)"] : names.map {
722 "\($0) = \(printed(macros[$0]!))"
723 }
724 }
725 if command == "unlet" {
726 if words.count != 1 {
727 try fail("bad argument")
728 }
729 let n = words[0]
730 let bytes = Array(n.utf8)
731 if bytes.isEmpty || bytes[0] < 65 || bytes[0] > 90 || !bytes.allSatisfy( {
732 $0 >= 65 && $0 <= 90 || $0 >= 48 && $0 <= 57 || $0 == 95
733 }) {
734 try fail("bad argument")
735 }
736 if reserved.contains(n) {
737 try fail("reserved name \(n)")
738 }
739 if macros[n] == nil {
740 try fail("unbound macro \(n)")
741 }
742 for used in macros.keys.sorted() where !reserved.contains(used) {
743 if references(macros[used]!).contains(n) {
744 try fail("\(n) is used by \(used)")
745 }
746 }
747 macros.removeValue(forKey: n)
748 return ["removed \(n)"]
749 }
750 let p = try Parser(line, arg, parts.count == 2 ? command.count + 2 : line.count + 1)
751 var name = ""
752 var strategy = "NORMAL"
753 var terms = [Term]()
754 switch command {
755 case "let":
756 name = try p.expect("macro").text
757 _ = try p.expect("=")
758 terms.append(try p.term())
759 case "subst":
760 name = try p.expect("var").text
761 _ = try p.expect("=")
762 terms.append(try p.term())
763 _ = try p.expect(";")
764 terms.append(try p.term())
765 case "equal", "alpha":
766 terms.append(try p.term())
767 _ = try p.expect(";")
768 terms.append(try p.term())
769 case "reduce":
770 if p.tokens[p.at].kind == "macro" && ["NORMAL", "APPLICATIVE", "CBN", "CBV"].contains(p.tokens[p.at].text) {
771 strategy = p.tokens[p.at].text
772 p.at += 1
773 }
774 terms.append(try p.term())
775 default:
776 terms.append(try p.term())
777 }
778 try p.end()
779 if command == "let" {
780 let t = terms[0]
781 if reserved.contains(name) {
782 try fail("reserved name \(name)")
783 }
784 for r in references(t) {
785 if r == name || depends(r, name) {
786 try fail("recursive definition of \(name)")
787 }
788 if macros[r] == nil {
789 try fail("unbound macro \(r)")
790 }
791 }
792 if let v = free(t).sorted().first {
793 try fail("free variable \(v) in definition of \(name)")
794 }
795 let exists = macros[name] != nil
796 macros[name] = t
797 return ["\(exists ? "redefined" : "defined") \(name)"]
798 }
799 let expanded = try terms.map {
