113def tokenize(text, first_column, end_column):
114 tokens = []
115 position = 0
116 while position < len(text):
117 if text[position] == " ":
118 position += 1
119 continue
120 column = first_column + position
121 matched = TOKEN_PATTERN.match(text, position)
122 if matched is None:
123 raise CommandError(f"column {column}: unexpected character '{text[position]}'")
124 kind = matched[0] if matched.lastgroup == SYMBOL else matched.lastgroup
125 tokens.append(Token(kind, matched[0], column))
126 position = matched.end()
127 tokens.append(Token(END, "", end_column))
128 return tokens

131class Parser:
132 def __init__(self, tokens):
133 self.tokens = tokens
134 self.position = 0
135
136 def peek(self):
137 return self.tokens[self.position]
138
139 def advance(self):
140 token = self.tokens[self.position]
141 self.position += 1
142 return token
143
144 def expect(self, kind):
145 token = self.peek()
146 if token.kind != kind:
147 raise CommandError(f"column {token.column}: expected {EXPECTED[kind]}")
148 return self.advance()
149
150 def finish(self):
151 token = self.peek()
152 if token.kind != END:
153 raise CommandError(f"column {token.column}: unexpected '{token.text}'")
154
155 def term(self):
156 elements = []
157 while self.peek().kind in (VARIABLE, NAME, NUMERAL, "("):
158 elements.append(self.atom())
159 if self.peek().kind == "\\":
160 elements.append(self.abstraction())
161 if not elements:
162 raise CommandError(f"column {self.peek().column}: expected term")
163 return apply_all(elements[0], elements[1:])
164
165 def atom(self):
166 token = self.advance()
167 if token.kind == VARIABLE:
168 return Var(token.text)
169 if token.kind == NAME:
170 return Macro(token.text)
171 if token.kind == NUMERAL:
172 return Numeral(int(token.text[1:]))
173 inner = self.term()
174 self.expect(")")
175 return inner
176
177 def abstraction(self):
178 self.advance()
179 names = [self.expect(VARIABLE).text]
180 while self.peek().kind == VARIABLE:
181 names.append(self.advance().text)
182 self.expect(".")
183 body = self.term()
184 for name in reversed(names):
185 body = Lam(name, body)
186 return body

197def show(term, at_end=True):
198 match term:
199 case Var(name) | Macro(name) | Const(name):
200 return name
201 case Numeral(value):
202 return f"#{value}"
203 case Index(value):
204 return str(value)
205 case Lam():
206 names = []
207 while isinstance(term, Lam):
208 names.append(term.var)
209 term = term.body
210 text = f"\\{' '.join(names)}. {show(term)}"
211 return text if at_end else f"({text})"
212 case DeBruijnLam(body):
213 text = "\\" + show(body)
214 return text if at_end else f"({text})"
215 case App(func, arg):
216 left = f"({show(func)})" if isinstance(func, (Lam, DeBruijnLam)) else show(func, False)
217 right = f"({show(arg)})" if isinstance(arg, App) else show(arg, at_end)
218 return f"{left} {right}"

221def free_vars(term):
222 match term:
223 case Var(name):
224 return {name}
225 case Lam(var, body):
226 return free_vars(body) - {var}
227 case DeBruijnLam(body):
228 return free_vars(body)
229 case App(func, arg):
230 return free_vars(func) | free_vars(arg)
231 return set()

234def macro_references(term):
235 match term:
236 case Macro(name):
237 return [name]
238 case Lam(_, body):
239 return macro_references(body)
240 case App(func, arg):
241 return macro_references(func) + macro_references(arg)
242 return []

245def numeral_term(value):
246 body = Var("x")
247 for _ in range(value):
248 body = App(Var("f"), body)
249 return Lam("f", Lam("x", body))

252def substitute(term, name, value):
253 match term:
254 case Var(var):
255 return value if var == name else term
256 case App(func, arg):
257 return App(substitute(func, name, value), substitute(arg, name, value))
258 case Lam(var, body):
259 body_free = free_vars(body)
260 if var == name or name not in body_free:
261 return term
262 value_free = free_vars(value)
263 if var not in value_free:
264 return Lam(var, substitute(body, name, value))
265 base = var.rstrip("0123456789")
266 suffix = 1
267 while f"{base}{suffix}" in value_free or f"{base}{suffix}" in body_free:
268 suffix += 1
269 fresh = f"{base}{suffix}"
270 return Lam(fresh, substitute(substitute(body, var, Var(fresh)), name, value))
271 return term

283def to_de_bruijn(term, bound=()):
284 match term:
285 case Var(name):
286 return Index(bound.index(name)) if name in bound else term
287 case Lam(var, body):
288 return DeBruijnLam(to_de_bruijn(body, (var, *bound)))
289 case App(func, arg):
290 return App(to_de_bruijn(func, bound), to_de_bruijn(arg, bound))
291 return term

294def from_de_bruijn(term):
295 names = fresh_names(free_vars(term))

297 def rename(node, bound):
298 match node:
299 case Index(value):
300 return Var(bound[value])
301 case DeBruijnLam(body):
302 name = next(names)
303 return Lam(name, rename(body, (name, *bound)))
304 case App(func, arg):
305 return App(rename(func, bound), rename(arg, bound))
306 return node

344def one_step(term, strategy, eta):
345 match term:
346 case App(Lam(var, body), arg) if strategy in (NORMAL, CBN):
347 return Step(substitute(body, var, arg), "beta")
348 case App(func, arg):
349 if found := one_step(func, strategy, eta):
350 return Step(App(found.term, arg), found.kind)
351 if strategy != CBN and (found := one_step(arg, strategy, eta)):
352 return Step(App(func, found.term), found.kind)
353 if isinstance(func, Lam) and (strategy == APPLICATIVE or isinstance(arg, (Var, Lam))):
354 return Step(substitute(func.body, func.var, arg), "beta")
355 case Lam(var, body) if strategy in (NORMAL, APPLICATIVE):
356 contractible = eta and is_eta_redex(term)
357 if contractible and strategy == NORMAL:
358 return Step(body.func, "eta")
359 if found := one_step(body, strategy, eta):
360 return Step(Lam(var, found.term), found.kind)
361 if contractible:
362 return Step(body.func, "eta")
363 return None

366def numeral_value(term):
367 if not (isinstance(term, Lam) and isinstance(term.body, Lam)):
368 return None
369 body = term.body.body
370 applications = 0
371 while isinstance(body, App) and body.func == Var(term.var):
372 body = body.arg
373 applications += 1
374 return applications if body == Var(term.body.var) else None

377def read_church(term):
378 value = numeral_value(term)
379 if value is not None:
380 return str(value)
381 match term:
382 case Lam(outer, Lam(_, Var(name))) if name == outer:
383 return "true"
384 case Lam(var, App(App(Var(name), first), second)) if (
385 name == var and var not in free_vars(first) | free_vars(second)
386 ):
387 return f"({read_church(first)}, {read_church(second)})"
388 return show(canonical(term))

392 match term:

400def abstract(var, term):
401 if var not in free_vars(term):
402 return App(Const("K"), term)
403 if term == Var(var):
404 return Const("I")
405 if term.arg == Var(var) and var not in free_vars(term.func):
406 return term.func
407 return App(App(Const("S"), abstract(var, term.func)), abstract(var, term.arg))

410def unwind(term):
411 args = []
412 while isinstance(term, App):
413 args.append(term.arg)
414 term = term.func
415 return term, args[::-1]

418def ski_step(term):
419 head, args = unwind(term)
420 match head:
421 case Const("I") if len(args) >= 1:
422 return Step(apply_all(args[0], args[1:]), "I")
423 case Const("K") if len(args) >= 2:
424 return Step(apply_all(args[0], args[2:]), "K")
425 case Const("S") if len(args) >= 3:
426 first, second, third = args[:3]
427 return Step(apply_all(App(App(first, third), App(second, third)), args[3:]), "S")
428 for position, arg in enumerate(args):
429 if found := ski_step(arg):
430 args[position] = found.term
431 return Step(apply_all(head, args), found.kind)
432 return None

435def read_back(code, env, binders=0):
436 match code:
437 case Index(value) if value >= binders:
438 return read_back(*env[value - binders])
439 case DeBruijnLam(body):
440 return DeBruijnLam(read_back(body, env, binders + 1))
441 case App(func, arg):
442 return App(read_back(func, env, binders), read_back(arg, env, binders))
443 return code

446class TypeInference:
447 def __init__(self):
448 self.bindings = {}
449 self.variables = 0
450
451 def fresh(self):
452 self.variables += 1
453 return self.variables
454
455 def resolve(self, type_):
456 while isinstance(type_, int) and type_ in self.bindings:
457 type_ = self.bindings[type_]
458 return type_
459
460 def occurs(self, variable, type_):
461 type_ = self.resolve(type_)
462 if isinstance(type_, Arrow):
463 return self.occurs(variable, type_.domain) or self.occurs(variable, type_.codomain)
464 return type_ == variable
465
466 def unify(self, left, right):
467 left = self.resolve(left)
468 right = self.resolve(right)
469 if isinstance(left, int):
470 if left != right:
471 if self.occurs(left, right):
472 raise NotTypable
473 self.bindings[left] = right
474 elif isinstance(right, int):
475 self.unify(right, left)
476 else:
477 self.unify(left.domain, right.domain)
478 self.unify(left.codomain, right.codomain)
479
480 def infer(self, term, scope):
481 match term:
482 case Var(name):
483 return scope[name]
484 case Lam(var, body):
485 parameter = self.fresh()
486 return Arrow(parameter, self.infer(body, {**scope, var: parameter}))
487 case App(func, arg):
488 func_type = self.infer(func, scope)
489 result = self.fresh()
490 self.unify(func_type, Arrow(self.infer(arg, scope), result))
491 return result
492
493 def render(self, type_, names):
494 type_ = self.resolve(type_)
495 if isinstance(type_, int):
496 return names.setdefault(type_, sequence_name(len(names)))
497 domain = self.render(type_.domain, names)
498 if isinstance(self.resolve(type_.domain), Arrow):
499 domain = f"({domain})"
500 return f"{domain} -> {self.render(type_.codomain, names)}"
501
502
503def principal_type(term):
504 inference = TypeInference()
505 scope = {name: inference.fresh() for name in free_vars(term)}
506 try:
507 type_ = inference.infer(term, scope)
508 except NotTypable:
509 return "not typable"
510 return inference.render(type_, {})

513class Workbench:
514 def __init__(self):
515 self.macros = {name: parse_term(text) for name, text in PREDEFINED.items()}
516 self.show_steps = False
517 self.eta = False
518 self.limit = MAX_LIMIT
519
520 def execute(self, line):
521 name, _, argument = line.partition(" ")
522 if name in WORD_COMMANDS:
523 getattr(self, f"command_{name}")(argument.split())
524 elif name in TERM_COMMANDS:
525 parser = Parser(tokenize(argument, len(name) + 2, len(line) + 1))
526 getattr(self, f"command_{name}")(parser)
527 else:
528 raise CommandError(f"unknown command {name}")
529
530 def expand(self, term):
531 match term:
532 case Macro(name):
533 if name not in self.macros:
534 raise CommandError(f"unbound macro {name}")
535 return self.expand(self.macros[name])
536 case Numeral(value):
537 return numeral_term(value)
538 case Lam(var, body):
539 return Lam(var, self.expand(body))
540 case App(func, arg):
541 return App(self.expand(func), self.expand(arg))
542 return term
543
544 def read_term(self, parser):
545 term = parser.term()
546 parser.finish()
547 return self.expand(term)
548
549 def read_pair(self, parser):
550 first = parser.term()
551 parser.expect(";")
552 second = parser.term()
553 parser.finish()
554 return self.expand(first), self.expand(second)
555
556 def dependencies(self, name):
557 found = set()
558 pending = [name]
559 while pending:
560 body = self.macros.get(pending.pop())
561 if body is None:
562 continue
563 for reference in macro_references(body):
564 if reference not in found:
565 found.add(reference)
566 pending.append(reference)
567 return found
568
569 def run(self, term, step, show_steps):
570 for number in count(1):
571 found = step(term)
572 if found is None:
573 return term, number - 1
574 if number > self.limit:
575 return None
576 term = found.term
577 if show_steps:
578 print(f"step {number} ({found.kind}): {show(term)}")
579
580 def normalize(self, term, strategy, eta, show_steps=False):
581 return self.run(term, lambda current: one_step(current, strategy, eta), show_steps)
582
583 def report(self, outcome, render):
584 if outcome is None:
585 print(f"no normal form within {self.limit} steps")
586 else:
587 result, steps = outcome
588 print(f"result: {render(result)} (steps: {steps})")

590 def command_let(self, parser):
591 name = parser.expect(NAME).text
592 parser.expect("=")
593 body = parser.term()
594 parser.finish()
595 if name in RESERVED:
596 raise CommandError(f"reserved name {name}")
597 for reference in macro_references(body):
598 if reference == name or name in self.dependencies(reference):
599 raise CommandError(f"recursive definition of {name}")
600 if reference not in self.macros:
601 raise CommandError(f"unbound macro {reference}")
602 free = free_vars(body)
603 if free:
604 raise CommandError(f"free variable {min(free)} in definition of {name}")
605 print(f"{'redefined' if name in self.macros else 'defined'} {name}")
606 self.macros[name] = body
607
608 def command_unlet(self, words):
609 if len(words) != 1 or not NAME_PATTERN.fullmatch(words[0]):
610 raise CommandError("bad argument")
611 name = words[0]
612 if name in RESERVED:
613 raise CommandError(f"reserved name {name}")
614 if name not in self.macros:
615 raise CommandError(f"unbound macro {name}")
616 user = min(
617 (other for other, body in self.macros.items() if name in macro_references(body)),
618 default=None,
619 )
620 if user is not None:
621 raise CommandError(f"{name} is used by {user}")
622 del self.macros[name]
623 print(f"removed {name}")

625 def command_macros(self, words):
626 if words:
627 raise CommandError("bad argument")
628 names = sorted(name for name in self.macros if name not in PREDEFINED)
629 for name in names:
630 print(f"{name} = {show(self.macros[name])}")
631 if not names:
632 print("(none)")

642 raise CommandError("bad argument")

661 def command_reduce(self, parser):
662 strategy = NORMAL
663 if parser.peek().kind == NAME and parser.peek().text in STRATEGIES:
664 strategy = parser.advance().text
665 term = self.read_term(parser)
666 outcome = self.normalize(term, strategy, self.eta, self.show_steps)
667 self.report(outcome, lambda result: show(canonical(result)))

669 def command_equal(self, parser):
670 first, second = (self.normalize(term, NORMAL, self.eta) for term in self.read_pair(parser))
671 if first is None or second is None:

673 else:
674 print("yes" if alpha_equivalent(first[0], second[0]) else "no")

682 def command_fv(self, parser):

686 def command_size(self, parser):
687 print(size(self.read_term(parser)))
688
689 def command_depth(self, parser):
690 print(depth(self.read_term(parser)))
691
692 def command_type(self, parser):

694
695 def command_church(self, parser):
696 outcome = self.normalize(self.read_term(parser), NORMAL, False)
697 if outcome is None:
698 print(f"no normal form within {self.limit} steps")
699 else:
700 print(read_church(canonical(outcome[0])))
701
702 def command_ski(self, parser):
703 print(show(to_ski(self.read_term(parser))))
704
705 def command_skireduce(self, parser):
706 combinator = to_ski(self.read_term(parser))
707 self.report(self.run(combinator, ski_step, self.show_steps), show)
708
709 def command_kam(self, parser):
710 code = to_de_bruijn(self.read_term(parser))
711 env = ()
712 stack = []
713 for number in count(1):
714 if isinstance(code, Var) or isinstance(code, DeBruijnLam) and not stack:
715 break
716 if number > self.limit:
717 print(f"no result within {self.limit} transitions")
718 return
719 match code:
720 case App(func, arg):
721 stack.append((arg, env))
722 code = func
723 kind = "push"
724 case DeBruijnLam(body):
725 env = (stack.pop(), *env)
726 code = body
727 kind = "grab"
728 case Index(value):
729 code, env = env[value]
730 kind = "access"
731 if self.show_steps:
732 state = f"[env {len(env)}, stack {len(stack)}]"
733 print(f"step {number} ({kind}): {show(code)} {state}")
734 arguments = [read_back(*closure) for closure in reversed(stack)]
735 result = apply_all(read_back(code, env), arguments)
736 print(f"result: {show(from_de_bruijn(result))} (transitions: {number - 1})")
