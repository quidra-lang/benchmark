9class Term
10 attr_reader :kind, :name, :left, :right, :free, :size, :depth
11
12 def initialize(kind, name = nil, left = nil, right = nil)
13 @kind, @name, @left, @right = kind, name, left, right
14 case kind
15 when :var
16 @free = Set[name]
17 when :abs
18 @free = left.free - Set[name]
19 when :app
20 @free = left.free | right.free
21 else
22 @free = Set.new
23 end
24 @size = 1 + (left&.size || 0) + (right&.size || 0)
25 @depth = 1 + [left&.depth || 0, right&.depth || 0].max
26 end
27end
28
29def variable(name) = Term.new(:var, name)
30def abstraction(name, body) = Term.new(:abs, name, body)
31def application(left, right) = Term.new(:app, nil, left, right)
32
33def printed(term, end_position = true, debruijn = false)
34 case term.kind
35 when :var, :macro, :constant
36 term.name
37 when :number
38 "##{term.name}"
39 when :index
40 term.name.to_s
41 when :abs
42 names = []
43 body = term
44 while body.kind == :abs
45 names << body.name
46 body = body.left
47 end
48 text = if debruijn
49 '\\' * names.length + printed(body, true, true)
50 else
51 '\\' + names.join(' ') + '. ' + printed(body)
52 end
53 end_position ? text : "(#{text})"
54 when :app
55 left = term.left.kind == :abs ? "(#{printed(term.left, true, debruijn)})" : printed(term.left, false, debruijn)
56 right = term.right.kind == :app ? "(#{printed(term.right, true, debruijn)})" : printed(term.right, end_position, debruijn)
57 left + ' ' + right
58 end
59end

61class Parser
62 def initialize(argument, offset, end_column)
63 @tokens = []
64 i = 0
65 while i < argument.length
66 c = argument[i]
67 if c == ' '
68 i += 1
69 next
70 end
71 start = i
72 if '\\.()=;'.include?(c)
73 kind = c
74 i += 1
75 elsif c.match?(/[a-z]/)
76 kind = :var
77 i += 1
78 i += 1 while i < argument.length && argument[i].match?(/[a-z0-9]/)
79 elsif c.match?(/[A-Z]/)
80 kind = :macro
81 i += 1
82 i += 1 while i < argument.length && argument[i].match?(/[A-Z0-9_]/)
83 elsif c == '#' && i + 1 < argument.length && argument[i + 1].match?(/[0-9]/)
84 kind = :number
85 i += 2
86 i += 1 while i < argument.length && argument[i].match?(/[0-9]/)
87 else
88 raise LambdaError, "column #{offset + i}: unexpected character '#{c}'"
89 end
90 @tokens << [kind, argument[start...i], offset + start]
91 end
92 @tokens << [:end, '', end_column]
93 @position = 0
94 end
95
96 def peek = @tokens[@position]
97 def advance
98 result = peek
99 @position += 1
100 result
101 end
102
103 def expect(kind)
104 notation = { var: 'variable', macro: 'macro name' }[kind] || "'#{kind}'"
105 raise LambdaError, "column #{peek[2]}: expected #{notation}" unless peek[0] == kind
106 advance[1]
107 end
108
109 def term
110 elements = []
111 while [:var, :macro, :number, '('].include?(peek[0])
112 kind, text = advance
113 if kind == '('
114 elements << term
115 expect(')')
116 else
117 elements << Term.new(kind, kind == :number ? text[1..].to_i : text)
118 end
119 end
120 if peek[0] == '\\'
121 advance
122 names = [expect(:var)]
123 names << advance[1] while peek[0] == :var
124 expect('.')
125 body = term
126 names.reverse_each { |name| body = abstraction(name, body) }
127 elements << body
128 end
129 raise LambdaError, "column #{peek[2]}: expected term" if elements.empty?
130 elements.reduce { |left, right| application(left, right) }
131 end
132
133 def finish
134 raise LambdaError, "column #{peek[2]}: unexpected '#{peek[1]}'" unless peek[0] == :end
135 end
136end
137
138def references(term)
139 case term.kind
140 when :macro then [term.name]
141 when :abs then references(term.left)
142 when :app then references(term.left) + references(term.right)
143 else []
144 end
145end

147def name_at(index)
148 (97 + index % 26).chr + (index >= 26 ? (index / 26).to_s : '')
149end
150
151def canonical(term)
152 free = term.free
153 counter = 0
154 visit = lambda do |node, bindings|
155 case node.kind
156 when :var
157 variable(bindings[node.name] || node.name)
158 when :abs
159 counter += 1 while free.include?(name_at(counter))
160 name = name_at(counter)
161 counter += 1
162 abstraction(name, visit.call(node.left, bindings.merge(node.name => name)))
163 when :app
164 application(visit.call(node.left, bindings), visit.call(node.right, bindings))
165 else
166 node
167 end
168 end
169 visit.call(term, {})
170end

172def to_debruijn(term, bindings = [])
173 case term.kind
174 when :var
175 index = bindings.index(term.name)
176 index ? Term.new(:index, index) : term
177 when :abs
178 abstraction(nil, to_debruijn(term.left, [term.name] + bindings))
179 when :app
180 application(to_debruijn(term.left, bindings), to_debruijn(term.right, bindings))
181 end
182end
183
184def from_debruijn(term)
185 counter = 0
186 free = term.free
187 visit = lambda do |node, bindings|
188 case node.kind
189 when :index
190 variable(bindings[node.name])
191 when :var
192 node
193 when :abs
194 counter += 1 while free.include?(name_at(counter))
195 name = name_at(counter)
196 counter += 1
197 abstraction(name, visit.call(node.left, [name] + bindings))
198 when :app
199 application(visit.call(node.left, bindings), visit.call(node.right, bindings))
200 end
201 end
202 visit.call(term, [])
203end

205def substitute(term, name, replacement)
206 case term.kind
207 when :var
208 term.name == name ? replacement : term
209 when :app
210 application(substitute(term.left, name, replacement), substitute(term.right, name, replacement))
211 when :abs
212 bound = term.name
213 body = term.left
214 return term if bound == name || !body.free.include?(name)
215 if replacement.free.include?(bound)
216 stem = bound.sub(/[0-9]+\z/, '')
217 suffix = 1
218 suffix += 1 while replacement.free.include?(stem + suffix.to_s) || body.free.include?(stem + suffix.to_s)
219 fresh = stem + suffix.to_s
220 body = substitute(body, bound, variable(fresh))
221 bound = fresh
222 end
223 abstraction(bound, substitute(body, name, replacement))
224 end
225end

227def eta_target(term)
228 body = term.left
229 if body.kind == :app && body.right.kind == :var && body.right.name == term.name && !body.left.free.include?(term.name)
230 body.left
231 end
232end

234def one_step(term, strategy, eta)
235 if term.kind == :app
236 left, right = term.left, term.right
237 if %w[NORMAL CBN].include?(strategy) && left.kind == :abs
238 return [substitute(left.left, left.name, right), 'beta']
239 end
240 step = one_step(left, strategy, eta)
241 return [application(step[0], right), step[1]] if step
242 unless strategy == 'CBN'
243 step = one_step(right, strategy, eta)
244 return [application(left, step[0]), step[1]] if step
245 end
246 if %w[APPLICATIVE CBV].include?(strategy) && left.kind == :abs && (strategy != 'CBV' || [:var, :abs].include?(right.kind))
247 return [substitute(left.left, left.name, right), 'beta']
248 end
249 elsif term.kind == :abs && %w[NORMAL APPLICATIVE].include?(strategy)
250 target = eta ? eta_target(term) : nil
251 return [target, 'eta'] if target && strategy == 'NORMAL'
252 step = one_step(term.left, strategy, eta)
253 return [abstraction(term.name, step[0]), step[1]] if step
254 return [target, 'eta'] if target
255 end
256 nil
257end

259def ski_abstract(name, term)
260 return application(Term.new(:constant, 'K'), term) unless term.free.include?(name)
261 return Term.new(:constant, 'I') if term.kind == :var && term.name == name
262 if term.right.kind == :var && term.right.name == name && !term.left.free.include?(name)
263 return term.left
264 end
265 application(application(Term.new(:constant, 'S'), ski_abstract(name, term.left)), ski_abstract(name, term.right))
266end

268def to_ski(term)
269 case term.kind
270 when :var then term
271 when :abs then ski_abstract(term.name, to_ski(term.left))
272 when :app then application(to_ski(term.left), to_ski(term.right))
273 end
274end

276def ski_step(term)
277 head = term
278 args = []
279 while head.kind == :app
280 args.unshift(head.right)
281 head = head.left
282 end
283 arity = head.kind == :constant ? { 'I' => 1, 'K' => 2, 'S' => 3 }[head.name] : nil
284 if arity && args.length >= arity
285 result = head.name == 'S' ? application(application(args[0], args[2]), application(args[1], args[2])) : args[0]
286 args.drop(arity).each { |arg| result = application(result, arg) }
287 return [result, head.name]
288 end
289 args.each_with_index do |arg, i|
290 step = ski_step(arg)
291 next unless step
292 changed = args.dup
293 changed[i] = step[0]
294 return [changed.reduce(head) { |left, right| application(left, right) }, step[1]]
295 end
296 nil
297end

299class TypeInference
300 def initialize
301 @next = 0
302 @links = {}
303 @free = {}
304 end
305
306 def fresh
307 result = [:variable, @next]
308 @next += 1
309 result
310 end
311
312 def prune(type)
313 path = []
314 while type[0] == :variable && @links.key?(type[1])
315 path << type[1]
316 type = @links[type[1]]
317 end
318 path.each { |id| @links[id] = type }
319 type
320 end
321
322 def occurs(id, type)
323 type = prune(type)
324 type[0] == :variable ? type[1] == id : occurs(id, type[1]) || occurs(id, type[2])
325 end
326
327 def unify(left, right)
328 left, right = prune(left), prune(right)
329 return if left == right
330 if left[0] == :variable
331 raise Untypable if occurs(left[1], right)
332 @links[left[1]] = right
333 elsif right[0] == :variable
334 unify(right, left)
335 else
336 unify(left[1], right[1])
337 unify(left[2], right[2])
338 end
339 end
340
341 def infer(term, environment = {})
342 case term.kind
343 when :var
344 environment[term.name] || (@free[term.name] ||= fresh)
345 when :abs
346 parameter = fresh
347 [:arrow, parameter, infer(term.left, environment.merge(term.name => parameter))]
348 when :app
349 function = infer(term.left, environment)
350 argument = infer(term.right, environment)
351 result = fresh
352 unify(function, [:arrow, argument, result])
353 result
354 end
355 end
356
357 def show(term)
358 type = infer(term)
359 names = {}
360 visit = lambda do |node|
361 node = prune(node)
362 if node[0] == :variable
363 names[node[1]] ||= name_at(names.length)
364 else
365 left = visit.call(node[1])
366 left = "(#{left})" if prune(node[1])[0] == :arrow
367 left + ' -> ' + visit.call(node[2])
368 end
369 end
370 visit.call(type)
371 rescue Untypable
372 'not typable'
373 end
374end
375
376def church_value(term)
377 if term.kind == :abs && term.left.kind == :abs
378 f, x = term.name, term.left.name
379 body = term.left.left
380 count = 0
381 while body.kind == :app && body.left.kind == :var && body.left.name == f
382 count += 1
383 body = body.right
384 end
385 return count.to_s if body.kind == :var && body.name == x
386 return 'true' if term.left.left.kind == :var && term.left.left.name == f
387 end
388 if term.kind == :abs
389 body = term.left
390 if body.kind == :app && body.left.kind == :app && body.left.left.kind == :var && body.left.left.name == term.name
391 p, q = body.left.right, body.right
392 return "(#{church_value(p)}, #{church_value(q)})" unless p.free.include?(term.name) || q.free.include?(term.name)
393 end
394 end
395 printed(canonical(term))
396end
397
398def materialize(code, environment, level = 0)
399 while code.kind == :index && code.name >= level
400 code, environment = environment[code.name - level]
401 level = 0
402 end
403 case code.kind
404 when :app
405 application(materialize(code.left, environment, level), materialize(code.right, environment, level))
406 when :abs
407 abstraction(nil, materialize(code.left, environment, level + 1))
408 else
409 code
410 end
411end

413class Workbench
414 RESERVED = %w[S K I NORMAL APPLICATIVE CBN CBV].freeze
415 COMMANDS = %w[let unlet macros steps eta limit expand subst reduce equal alpha debruijn fv size depth type church ski skireduce kam].freeze
416
417 def initialize
418 @macros = {
419 'S' => abstraction('x', abstraction('y', abstraction('z', application(application(variable('x'), variable('z')), application(variable('y'), variable('z')))))),
420 'K' => abstraction('x', abstraction('y', variable('x'))),
421 'I' => abstraction('x', variable('x'))
422 }
423 @steps = false
424 @eta = false
425 @limit = 10_000
426 end
427
428 def depends?(name, target, visited = {})
429 return false if visited.key?(name)
430 visited[name] = true
431 return false unless @macros.key?(name) && !%w[S K I].include?(name)
432 references(@macros[name]).any? { |ref| ref == target || depends?(ref, target, visited) }
433 end
434
435 def expand(term)
436 case term.kind
437 when :macro
438 raise LambdaError, "unbound macro #{term.name}" unless @macros.key?(term.name)
439 expand(@macros[term.name])
440 when :number
441 body = variable('x')
442 term.name.times { body = application(variable('f'), body) }
443 abstraction('f', abstraction('x', body))
444 when :abs
445 abstraction(term.name, expand(term.left))
446 when :app
447 application(expand(term.left), expand(term.right))
448 else
449 term
450 end
451 end
452
453 def reduce(term, strategy = 'NORMAL', eta = @eta, display = false, ski = false)
454 count = 0
455 loop do
456 step = ski ? ski_step(term) : one_step(term, strategy, eta)
457 return [term, count] unless step
458 return nil if count == @limit
459 term = step[0]
460 count += 1
461 puts "step #{count} (#{step[1]}): #{printed(term)}" if display && @steps
462 end
463 end
464
465 def kam(term)
466 code = to_debruijn(term)
467 environment = []
468 stack = []
469 count = 0
470 loop do
471 kind = case code.kind
472 when :app then 'push'
473 when :abs then 'grab' unless stack.empty?
474 when :index then 'access'
475 end
476 break unless kind
477 if count == @limit
478 puts "no result within #{@limit} transitions"
479 return
480 end
481 case kind
482 when 'push'
483 stack << [code.right, environment]
484 code = code.left
485 when 'grab'
486 environment = [stack.pop] + environment
487 code = code.left
488 when 'access'
489 code, environment = environment[code.name]
490 end
491 count += 1
492 puts "step #{count} (#{kind}): #{printed(code, true, true)} [env #{environment.length}, stack #{stack.length}]" if @steps
493 end
494 result = materialize(code, environment)
495 stack.reverse_each { |closure| result = application(result, materialize(*closure)) }
496 puts "result: #{printed(from_debruijn(result))} (transitions: #{count})"
497 end

499 def command(line)
500 separator = line.index(' ')
501 command = separator ? line[0...separator] : line
502 argument = separator ? line[(separator + 1)..] : ''
503 raise LambdaError, "unknown command #{command}" unless COMMANDS.include?(command)
504 if %w[steps eta limit macros unlet].include?(command)
505 words = argument.split(' ')
506 case command
507 when 'steps', 'eta'
508 raise LambdaError, 'bad argument' unless words.length == 1 && %w[on off].include?(words[0])
509 command == 'steps' ? @steps = words[0] == 'on' : @eta = words[0] == 'on'
510 when 'limit'
511 raise LambdaError, 'bad argument' unless words.length == 1 && words[0].match?(/\A[0-9]{1,5}\z/) && (1..10_000).cover?(words[0].to_i)
512 @limit = words[0].to_i
513 when 'macros'
514 raise LambdaError, 'bad argument' unless words.empty?
515 names = @macros.keys.reject { |name| %w[S K I].include?(name) }.sort
516 puts '(none)' if names.empty?
517 names.each { |name| puts "#{name} = #{printed(@macros[name])}" }
518 when 'unlet'
519 raise LambdaError, 'bad argument' unless words.length == 1 && words[0].match?(/\A[A-Z][A-Z0-9_]*\z/)
520 name = words[0]
521 raise LambdaError, "reserved name #{name}" if RESERVED.include?(name)
522 raise LambdaError, "unbound macro #{name}" unless @macros.key?(name)
523 user = @macros.keys.sort.find { |key| !%w[S K I].include?(key) && references(@macros[key]).include?(name) }
524 raise LambdaError, "#{name} is used by #{user}" if user
525 @macros.delete(name)
526 puts "removed #{name}"
527 end
528 return
529 end
530 parser = Parser.new(argument, command.length + 2, line.length + 1)
531 strategy = 'NORMAL'
532 name = nil
533 terms = []
534 case command
535 when 'let', 'subst'
536 name = parser.expect(command == 'let' ? :macro : :var)
537 parser.expect('=')
538 terms << parser.term
539 if command == 'subst'
540 parser.expect(';')
541 terms << parser.term
542 end
543 when 'equal', 'alpha'
544 terms << parser.term
545 parser.expect(';')
546 terms << parser.term
547 when 'reduce'
548 strategy = parser.advance[1] if parser.peek[0] == :macro && %w[NORMAL APPLICATIVE CBN CBV].include?(parser.peek[1])
549 terms << parser.term
550 else
551 terms << parser.term
552 end
553 parser.finish
554 if command == 'let'
555 raise LambdaError, "reserved name #{name}" if RESERVED.include?(name)
556 references(terms[0]).each do |ref|
557 raise LambdaError, "recursive definition of #{name}" if ref == name || depends?(ref, name)
558 raise LambdaError, "unbound macro #{ref}" unless @macros.key?(ref)
559 end
560 unless terms[0].free.empty?
561 raise LambdaError, "free variable #{terms[0].free.min} in definition of #{name}"
562 end
563 puts "#{@macros.key?(name) ? 'redefined' : 'defined'} #{name}"
564 @macros[name] = terms[0]
565 return
566 end
567 terms = terms.map { |term| expand(term) }
568 term = terms[0]
569 case command
570 when 'expand' then puts printed(term)
571 when 'subst' then puts printed(substitute(terms[1], name, term))
572 when 'alpha' then puts printed(to_debruijn(term), true, true) == printed(to_debruijn(terms[1]), true, true) ? 'yes' : 'no'
573 when 'debruijn' then puts printed(to_debruijn(term), true, true)
574 when 'fv' then puts term.free.empty? ? '(none)' : term.free.to_a.sort.join(' ')
575 when 'size' then puts term.size
576 when 'depth' then puts term.depth
577 when 'type' then puts TypeInference.new.show(term)
578 when 'ski' then puts printed(to_ski(term))
579 when 'kam' then kam(term)
580 when 'equal'
581 left = reduce(term)
582 right = reduce(terms[1])
583 if !left || !right
584 puts 'unknown'
585 elsif printed(canonical(left[0])) == printed(canonical(right[0]))
586 puts 'yes'
587 else
588 puts 'no'
589 end
590 when 'church', 'reduce', 'skireduce'
591 result = if command == 'church'
592 reduce(term, 'NORMAL', false)
593 elsif command == 'skireduce'
594 reduce(to_ski(term), 'NORMAL', false, true, true)
595 else
596 reduce(term, strategy, @eta, true)
597 end
598 if result.nil?
599 puts "no normal form within #{@limit} steps"
600 elsif command == 'church'
601 puts church_value(canonical(result[0]))
602 else
603 text = command == 'skireduce' ? printed(result[0]) : printed(canonical(result[0]))
604 puts "result: #{text} (steps: #{result[1]})"
605 end
606 end
607 end
608
609 def run
610 STDIN.each_line do |input|
611 line = input.delete_suffix("\n")
612 break if line == 'quit'
613 next if line.empty? || line.start_with?('--')
614 begin
615 command(line)
616 rescue LambdaError => error
617 puts "error: #{error.message}"
618 end
619 end
620 end
621end
