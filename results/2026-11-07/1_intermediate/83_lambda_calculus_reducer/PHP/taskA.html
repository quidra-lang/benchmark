10class Term
11{
12 public array $free;
13 public int $size;
14 public int $depth;
15
16 public function __construct(
17 public string $kind,
18 public string|int|null $name = null,
19 public ?Term $left = null,
20 public ?Term $right = null
21 ) {
22 $this->free = match ($kind) {
23 'var' => [$name => true],
24 'abs' => $left->free,
25 'app' => $left->free + $right->free,
26 default => []
27 };
28 if ($kind === 'abs') {
29 unset($this->free[$name]);
30 }
31 $this->size = 1 + ($left?->size ?? 0) + ($right?->size ?? 0);
32 $this->depth = 1 + max($left?->depth ?? 0, $right?->depth ?? 0);
33 }
34}

51function printed(Term $term, bool $end = true, bool $db = false): string
52{
53 if (in_array($term->kind, ['var', 'macro', 'constant'], true)) {
54 return $term->name;
55 }
56 if ($term->kind === 'number') {
57 return '#' . $term->name;
58 }
59 if ($term->kind === 'index') {
60 return (string)$term->name;
61 }
62 if ($term->kind === 'abs') {
63 $names = [];
64 $body = $term;
65 while ($body->kind === 'abs') {
66 $names[] = $body->name;
67 $body = $body->left;
68 }
69 $text = $db
70 ? str_repeat('\\', count($names)) . printed($body, true, true)
71 : '\\' . implode(' ', $names) . '. ' . printed($body);
72 return $end ? $text : '(' . $text . ')';
73 }
74 $left = $term->left->kind === 'abs' ? '(' . printed($term->left, true, $db) . ')' : printed($term->left, false, $db);
75 $right = $term->right->kind === 'app' ? '(' . printed($term->right, true, $db) . ')' : printed($term->right, $end, $db);
76 return $left . ' ' . $right;
77}

79class LambdaParser
80{
81 private array $tokens = [];
82 private int $position = 0;
83
84 public function __construct(string $argument, int $offset, int $endColumn)
85 {
86 $i = 0;
87 $length = strlen($argument);
88 while ($i < $length) {
89 $c = $argument[$i];
90 if ($c === ' ') {
91 $i++;
92 continue;
93 }
94 $start = $i;
95 if (str_contains('\\.()=;', $c)) {
96 $kind = $c;
97 $i++;
98 } elseif ($c >= 'a' && $c <= 'z') {
99 $kind = 'var';
100 $i++;
101 while ($i < $length && preg_match('/[a-z0-9]/', $argument[$i])) {
102 $i++;
103 }
104 } elseif ($c >= 'A' && $c <= 'Z') {
105 $kind = 'macro';
106 $i++;
107 while ($i < $length && preg_match('/[A-Z0-9_]/', $argument[$i])) {
108 $i++;
109 }
110 } elseif ($c === '#' && $i + 1 < $length && ctype_digit($argument[$i + 1])) {
111 $kind = 'number';
112 $i += 2;
113 while ($i < $length && ctype_digit($argument[$i])) {
114 $i++;
115 }
116 } else {
117 throw new LambdaError('column ' . ($offset + $i) . ": unexpected character '" . $c . "'");
118 }
119 $this->tokens[] = [$kind, substr($argument, $start, $i - $start), $offset + $start];
120 }
121 $this->tokens[] = ['end', '', $endColumn];
122 }
123
124 public function peek(): array
125 {
126 return $this->tokens[$this->position];
127 }
128
129 public function advance(): array
130 {
131 return $this->tokens[$this->position++];
132 }
133
134 public function expect(string $kind): string
135 {
136 $notation = ['var' => 'variable', 'macro' => 'macro name'][$kind] ?? "'" . $kind . "'";
137 if ($this->peek()[0] !== $kind) {
138 throw new LambdaError('column ' . $this->peek()[2] . ': expected ' . $notation);
139 }
140 return $this->advance()[1];
141 }
142
143 public function term(): Term
144 {
145 $elements = [];
146 while (in_array($this->peek()[0], ['var', 'macro', 'number', '('], true)) {
147 [$kind, $text] = $this->advance();
148 if ($kind === '(') {
149 $elements[] = $this->term();
150 $this->expect(')');
151 } else {
152 $elements[] = new Term($kind, $kind === 'number' ? (int)substr($text, 1) : $text);
153 }
154 }
155 if ($this->peek()[0] === '\\') {
156 $this->advance();
157 $names = [$this->expect('var')];
158 while ($this->peek()[0] === 'var') {
159 $names[] = $this->advance()[1];
160 }
161 $this->expect('.');
162 $body = $this->term();
163 foreach (array_reverse($names) as $name) {
164 $body = abstraction($name, $body);
165 }
166 $elements[] = $body;
167 }
168 if ($elements === []) {
169 throw new LambdaError('column ' . $this->peek()[2] . ': expected term');
170 }
171 $result = array_shift($elements);
172 foreach ($elements as $element) {
173 $result = application($result, $element);
174 }
175 return $result;
176 }
177
178 public function finish(): void
179 {
180 if ($this->peek()[0] !== 'end') {
181 throw new LambdaError('column ' . $this->peek()[2] . ": unexpected '" . $this->peek()[1] . "'");
182 }
183 }
184}

186function references(Term $term): array
187{
188 return match ($term->kind) {
189 'macro' => [$term->name],
190 'abs' => references($term->left),
191 'app' => array_merge(references($term->left), references($term->right)),
192 default => []
193 };

196function nameAt(int $index): string
197{
198 return chr(97 + $index % 26) . ($index >= 26 ? intdiv($index, 26) : '');
199}

201function canonical(Term $term): Term
202{
203 $free = $term->free;
204 $counter = 0;
205 $visit = function (Term $node, array $bindings) use (&$visit, &$counter, $free): Term {
206 if ($node->kind === 'var') {
207 return variable($bindings[$node->name] ?? $node->name);
208 }
209 if ($node->kind === 'abs') {
210 while (isset($free[nameAt($counter)])) {
211 $counter++;
212 }
213 $name = nameAt($counter++);
214 $bindings[$node->name] = $name;
215 return abstraction($name, $visit($node->left, $bindings));
216 }
217 if ($node->kind === 'app') {
218 $left = $visit($node->left, $bindings);
219 $right = $visit($node->right, $bindings);
220 return application($left, $right);
221 }
222 return $node;
223 };
224 return $visit($term, []);
225}

227function toDebruijn(Term $term, array $bindings = []): Term
228{
229 if ($term->kind === 'var') {
230 $index = array_search($term->name, $bindings, true);
231 return $index === false ? $term : new Term('index', $index);
232 }
233 if ($term->kind === 'abs') {
234 return abstraction(null, toDebruijn($term->left, array_merge([$term->name], $bindings)));
235 }
236 return application(toDebruijn($term->left, $bindings), toDebruijn($term->right, $bindings));
237}
238
239function fromDebruijn(Term $term): Term
240{
241 $free = $term->free;
242 $counter = 0;
243 $visit = function (Term $node, array $bindings) use (&$visit, &$counter, $free): Term {
244 if ($node->kind === 'index') {
245 return variable($bindings[$node->name]);
246 }
247 if ($node->kind === 'var') {
248 return $node;
249 }
250 if ($node->kind === 'abs') {
251 while (isset($free[nameAt($counter)])) {
252 $counter++;
253 }
254 $name = nameAt($counter++);
255 return abstraction($name, $visit($node->left, array_merge([$name], $bindings)));
256 }
257 $left = $visit($node->left, $bindings);
258 $right = $visit($node->right, $bindings);
259 return application($left, $right);
260 };
261 return $visit($term, []);
262}

264function substitute(Term $term, string $name, Term $replacement): Term
265{
266 if ($term->kind === 'var') {
267 return $term->name === $name ? $replacement : $term;
268 }
269 if ($term->kind === 'app') {
270 return application(substitute($term->left, $name, $replacement), substitute($term->right, $name, $replacement));
271 }
272 $bound = $term->name;
273 $body = $term->left;
274 if ($bound === $name || !isset($body->free[$name])) {
275 return $term;
276 }
277 if (isset($replacement->free[$bound])) {
278 $stem = preg_replace('/[0-9]+\z/', '', $bound);
279 $suffix = 1;
280 while (isset($replacement->free[$stem . $suffix]) || isset($body->free[$stem . $suffix])) {
281 $suffix++;
282 }
283 $fresh = $stem . $suffix;
284 $body = substitute($body, $bound, variable($fresh));
285 $bound = $fresh;
286 }
287 return abstraction($bound, substitute($body, $name, $replacement));
288}

290function etaTarget(Term $term): ?Term
291{
292 $body = $term->left;
293 return $body->kind === 'app' && $body->right->kind === 'var' && $body->right->name === $term->name && !isset($body->left->free[$term->name]) ? $body->left : null;
294}

296function oneStep(Term $term, string $strategy, bool $eta): ?array
297{
298 if ($term->kind === 'app') {
299 $left = $term->left;
300 $right = $term->right;
301 if (in_array($strategy, ['NORMAL', 'CBN'], true) && $left->kind === 'abs') {
302 return [substitute($left->left, $left->name, $right), 'beta'];
303 }
304 $step = oneStep($left, $strategy, $eta);
305 if ($step !== null) {
306 return [application($step[0], $right), $step[1]];
307 }
308 if ($strategy !== 'CBN') {
309 $step = oneStep($right, $strategy, $eta);
310 if ($step !== null) {
311 return [application($left, $step[0]), $step[1]];
312 }
313 }
314 if (in_array($strategy, ['APPLICATIVE', 'CBV'], true) && $left->kind === 'abs' && ($strategy !== 'CBV' || in_array($right->kind, ['var', 'abs'], true))) {
315 return [substitute($left->left, $left->name, $right), 'beta'];
316 }
317 } elseif ($term->kind === 'abs' && in_array($strategy, ['NORMAL', 'APPLICATIVE'], true)) {
318 $target = $eta ? etaTarget($term) : null;
319 if ($target !== null && $strategy === 'NORMAL') {
320 return [$target, 'eta'];
321 }
322 $step = oneStep($term->left, $strategy, $eta);
323 if ($step !== null) {
324 return [abstraction($term->name, $step[0]), $step[1]];
325 }
326 if ($target !== null) {
327 return [$target, 'eta'];
328 }
329 }
330 return null;
331}

333function skiAbstract(string $name, Term $term): Term
334{
335 if (!isset($term->free[$name])) {
336 return application(new Term('constant', 'K'), $term);
337 }
338 if ($term->kind === 'var' && $term->name === $name) {
339 return new Term('constant', 'I');
340 }
341 if ($term->right->kind === 'var' && $term->right->name === $name && !isset($term->left->free[$name])) {
342 return $term->left;
343 }
344 return application(application(new Term('constant', 'S'), skiAbstract($name, $term->left)), skiAbstract($name, $term->right));
345}

347function toSki(Term $term): Term
348{
349 return match ($term->kind) {
350 'var' => $term,
351 'abs' => skiAbstract($term->name, toSki($term->left)),
352 'app' => application(toSki($term->left), toSki($term->right))
353 };
354}

356function skiStep(Term $term): ?array
357{
358 $head = $term;
359 $args = [];
360 while ($head->kind === 'app') {
361 array_unshift($args, $head->right);
362 $head = $head->left;
363 }
364 $arity = $head->kind === 'constant' ? ['I' => 1, 'K' => 2, 'S' => 3][$head->name] : null;
365 if ($arity !== null && count($args) >= $arity) {
366 $result = $head->name === 'S' ? application(application($args[0], $args[2]), application($args[1], $args[2])) : $args[0];
367 foreach (array_slice($args, $arity) as $arg) {
368 $result = application($result, $arg);
369 }
370 return [$result, $head->name];
371 }
372 foreach ($args as $i => $arg) {
373 $step = skiStep($arg);
374 if ($step !== null) {
375 $args[$i] = $step[0];
376 $result = $head;
377 foreach ($args as $argument) {
378 $result = application($result, $argument);
379 }
380 return [$result, $step[1]];
381 }
382 }
383 return null;
384}

386class TypeInference
387{
388 private int $next = 0;
389 private array $links = [];
390 private array $free = [];
391
392 private function fresh(): array
393 {
394 return ['variable', $this->next++];
395 }
396
397 private function prune(array $type): array
398 {
399 $path = [];
400 while ($type[0] === 'variable' && isset($this->links[$type[1]])) {
401 $path[] = $type[1];
402 $type = $this->links[$type[1]];
403 }
404 foreach ($path as $id) {
405 $this->links[$id] = $type;
406 }
407 return $type;
408 }
409
410 private function occurs(int $id, array $type): bool
411 {
412 $type = $this->prune($type);
413 return $type[0] === 'variable' ? $type[1] === $id : $this->occurs($id, $type[1]) || $this->occurs($id, $type[2]);
414 }
415
416 private function unify(array $left, array $right): void
417 {
418 $left = $this->prune($left);
419 $right = $this->prune($right);
420 if ($left === $right) {
421 return;
422 }
423 if ($left[0] === 'variable') {
424 if ($this->occurs($left[1], $right)) {
425 throw new Untypable();
426 }
427 $this->links[$left[1]] = $right;
428 } elseif ($right[0] === 'variable') {
429 $this->unify($right, $left);
430 } else {
431 $this->unify($left[1], $right[1]);
432 $this->unify($left[2], $right[2]);
433 }
434 }
435
436 private function infer(Term $term, array $environment = []): array
437 {
438 if ($term->kind === 'var') {
439 if (isset($environment[$term->name])) {
440 return $environment[$term->name];
441 }
442 return $this->free[$term->name] ??= $this->fresh();
443 }
444 if ($term->kind === 'abs') {
445 $parameter = $this->fresh();
446 $environment[$term->name] = $parameter;
447 return ['arrow', $parameter, $this->infer($term->left, $environment)];
448 }
449 $function = $this->infer($term->left, $environment);
450 $argument = $this->infer($term->right, $environment);
451 $result = $this->fresh();
452 $this->unify($function, ['arrow', $argument, $result]);
453 return $result;
454 }
455
456 public function show(Term $term): string
457 {
458 try {
459 $type = $this->infer($term);
460 $names = [];
461 $visit = function (array $node) use (&$visit, &$names): string {
462 $node = $this->prune($node);
463 if ($node[0] === 'variable') {
464 return $names[$node[1]] ??= nameAt(count($names));
465 }
466 $left = $visit($node[1]);
467 if ($this->prune($node[1])[0] === 'arrow') {
468 $left = '(' . $left . ')';
469 }
470 return $left . ' -> ' . $visit($node[2]);
471 };
472 return $visit($type);
473 } catch (Untypable) {
474 return 'not typable';
475 }
476 }
477}

479function churchValue(Term $term): string
480{
481 if ($term->kind === 'abs' && $term->left->kind === 'abs') {
482 $f = $term->name;
483 $x = $term->left->name;
484 $body = $term->left->left;
485 $count = 0;
486 while ($body->kind === 'app' && $body->left->kind === 'var' && $body->left->name === $f) {
487 $count++;
488 $body = $body->right;
489 }
490 if ($body->kind === 'var' && $body->name === $x) {
491 return (string)$count;
492 }
493 if ($term->left->left->kind === 'var' && $term->left->left->name === $f) {
494 return 'true';
495 }
496 }
497 if ($term->kind === 'abs') {
498 $body = $term->left;
499 if ($body->kind === 'app' && $body->left->kind === 'app' && $body->left->left->kind === 'var' && $body->left->left->name === $term->name) {
500 $p = $body->left->right;
501 $q = $body->right;
502 if (!isset($p->free[$term->name]) && !isset($q->free[$term->name])) {
503 return '(' . churchValue($p) . ', ' . churchValue($q) . ')';
504 }
505 }
506 }
507 return printed(canonical($term));
508}

510function materialize(Term $code, array $environment, int $level = 0): Term
511{
512 while ($code->kind === 'index' && $code->name >= $level) {
513 [$code, $environment] = $environment[$code->name - $level];
514 $level = 0;
515 }
516 if ($code->kind === 'app') {
517 return application(materialize($code->left, $environment, $level), materialize($code->right, $environment, $level));
518 }
519 if ($code->kind === 'abs') {
520 return abstraction(null, materialize($code->left, $environment, $level + 1));
521 }
522 return $code;
523}

525class Workbench
526{
527 private const RESERVED = ['S', 'K', 'I', 'NORMAL', 'APPLICATIVE', 'CBN', 'CBV'];
528 private const COMMANDS = ['let', 'unlet', 'macros', 'steps', 'eta', 'limit', 'expand', 'subst',
529 'reduce', 'equal', 'alpha', 'debruijn', 'fv', 'size', 'depth', 'type', 'church', 'ski', 'skireduce', 'kam'];
530 private array $macros;
531 private bool $steps = false;
532 private bool $eta = false;
533 private int $limit = 10000;
534
535 public function __construct()
536 {
537 $this->macros = [
538 'S' => abstraction('x', abstraction('y', abstraction('z', application(application(variable('x'), variable('z')), application(variable('y'), variable('z')))))),
539 'K' => abstraction('x', abstraction('y', variable('x'))),
540 'I' => abstraction('x', variable('x'))
541 ];
542 }
543
544 private function depends(string $name, string $target, ?array &$visited = null): bool
545 {
546 if ($visited === null) {
547 $visited = [];
548 }
549 if (isset($visited[$name])) {
550 return false;
551 }
552 $visited[$name] = true;
553 if (!isset($this->macros[$name]) || in_array($name, ['S', 'K', 'I'], true)) {
554 return false;
555 }
556 foreach (references($this->macros[$name]) as $ref) {
557 if ($ref === $target || $this->depends($ref, $target, $visited)) {
558 return true;
559 }
560 }
561 return false;
562 }
563
564 private function expand(Term $term): Term
565 {
566 if ($term->kind === 'macro') {
567 if (!isset($this->macros[$term->name])) {
568 throw new LambdaError('unbound macro ' . $term->name);
569 }
570 return $this->expand($this->macros[$term->name]);
571 }
572 if ($term->kind === 'number') {
573 $body = variable('x');
574 for ($i = 0; $i < $term->name; $i++) {
575 $body = application(variable('f'), $body);
576 }
577 return abstraction('f', abstraction('x', $body));
578 }
579 if ($term->kind === 'abs') {
580 return abstraction($term->name, $this->expand($term->left));
581 }
582 if ($term->kind === 'app') {
583 $left = $this->expand($term->left);
584 $right = $this->expand($term->right);
585 return application($left, $right);
586 }
587 return $term;
588 }
589
590 private function reduce(Term $term, string $strategy, bool $eta, bool $display = false, bool $ski = false): ?array
591 {
592 $count = 0;
593 while (true) {
594 $step = $ski ? skiStep($term) : oneStep($term, $strategy, $eta);
595 if ($step === null) {
596 return [$term, $count];
597 }
598 if ($count === $this->limit) {
599 return null;
600 }
601 $term = $step[0];
602 $count++;
603 if ($display && $this->steps) {
604 echo 'step ', $count, ' (', $step[1], '): ', printed($term), "\n";
605 }
606 }
607 }
608
609 private function kam(Term $term): void
610 {
611 $code = toDebruijn($term);
612 $environment = [];
613 $stack = [];
614 $count = 0;
615 while (true) {
616 $kind = match ($code->kind) {
617 'app' => 'push', 'abs' => $stack === [] ? null : 'grab', 'index' => 'access', default => null
618 };
619 if ($kind === null) {
620 break;
621 }
622 if ($count === $this->limit) {
623 echo 'no result within ', $this->limit, " transitions\n";
624 return;
625 }
626 if ($kind === 'push') {
627 $stack[] = [$code->right, $environment];
628 $code = $code->left;
629 } elseif ($kind === 'grab') {
630 array_unshift($environment, array_pop($stack));
631 $code = $code->left;
632 } else {
633 [$code, $environment] = $environment[$code->name];
634 }
635 $count++;
636 if ($this->steps) {
637 echo 'step ', $count, ' (', $kind, '): ', printed($code, true, true), ' [env ', count($environment), ', stack ', count($stack), "]\n";
638 }
639 }
640 $result = materialize($code, $environment);
641 foreach (array_reverse($stack) as $closure) {
642 $result = application($result, materialize($closure[0], $closure[1]));
643 }
644 echo 'result: ', printed(fromDebruijn($result)), ' (transitions: ', $count, ")\n";
645 }

647 private function simpleCommand(string $command, string $argument): void
648 {
649 $words = preg_split('/ +/', $argument, -1, PREG_SPLIT_NO_EMPTY);
650 if ($command === 'steps' || $command === 'eta') {
651 if (count($words) !== 1 || !in_array($words[0], ['on', 'off'], true)) {
652 throw new LambdaError('bad argument');
653 }
654 if ($command === 'steps') {
655 $this->steps = $words[0] === 'on';
656 } else {
657 $this->eta = $words[0] === 'on';
658 }
659 } elseif ($command === 'limit') {
660 if (count($words) !== 1 || !preg_match('/\A[0-9]{1,5}\z/', $words[0]) || (int)$words[0] < 1 || (int)$words[0] > 10000) {
661 throw new LambdaError('bad argument');
662 }
663 $this->limit = (int)$words[0];
664 } elseif ($command === 'macros') {
665 if ($words !== []) {
666 throw new LambdaError('bad argument');
667 }
668 $names = array_values(array_diff(array_keys($this->macros), ['S', 'K', 'I']));
669 sort($names, SORT_STRING);
670 if ($names === []) {
671 echo "(none)\n";
672 }
673 foreach ($names as $name) {
674 echo $name, ' = ', printed($this->macros[$name]), "\n";
675 }
676 } else {
677 if (count($words) !== 1 || !preg_match('/\A[A-Z][A-Z0-9_]*\z/', $words[0])) {
678 throw new LambdaError('bad argument');
679 }
680 $name = $words[0];
681 if (in_array($name, self::RESERVED, true)) {
682 throw new LambdaError('reserved name ' . $name);
683 }
684 if (!isset($this->macros[$name])) {
685 throw new LambdaError('unbound macro ' . $name);
686 }
687 $names = array_keys($this->macros);
688 sort($names, SORT_STRING);
689 foreach ($names as $key) {
690 if (!in_array($key, ['S', 'K', 'I'], true) && in_array($name, references($this->macros[$key]), true)) {
691 throw new LambdaError($name . ' is used by ' . $key);
692 }
693 }
694 unset($this->macros[$name]);
695 echo 'removed ', $name, "\n";
696 }
697 }

699 public function command(string $line): void
700 {
701 $parts = explode(' ', $line, 2);
702 $command = $parts[0];
703 $argument = $parts[1] ?? '';
704 if (!in_array($command, self::COMMANDS, true)) {
705 throw new LambdaError('unknown command ' . $command);
706 }
707 if (in_array($command, ['steps', 'eta', 'limit', 'macros', 'unlet'], true)) {
708 $this->simpleCommand($command, $argument);
709 return;
710 }
711 $parser = new LambdaParser($argument, strlen($command) + 2, strlen($line) + 1);
712 $strategy = 'NORMAL';
713 $name = null;
714 $terms = [];
715 if ($command === 'let' || $command === 'subst') {
716 $name = $parser->expect($command === 'let' ? 'macro' : 'var');
717 $parser->expect('=');
718 $terms[] = $parser->term();
719 if ($command === 'subst') {
720 $parser->expect(';');
721 $terms[] = $parser->term();
722 }
723 } elseif ($command === 'equal' || $command === 'alpha') {
724 $terms[] = $parser->term();
725 $parser->expect(';');
726 $terms[] = $parser->term();
727 } else {
728 if ($command === 'reduce' && $parser->peek()[0] === 'macro' && in_array($parser->peek()[1], ['NORMAL', 'APPLICATIVE', 'CBN', 'CBV'], true)) {
729 $strategy = $parser->advance()[1];
730 }
731 $terms[] = $parser->term();
732 }
733 $parser->finish();
734 if ($command === 'let') {
735 if (in_array($name, self::RESERVED, true)) {
736 throw new LambdaError('reserved name ' . $name);
737 }
738 foreach (references($terms[0]) as $ref) {
739 if ($ref === $name || $this->depends($ref, $name)) {
740 throw new LambdaError('recursive definition of ' . $name);
741 }
742 if (!isset($this->macros[$ref])) {
743 throw new LambdaError('unbound macro ' . $ref);
744 }
745 }
746 if ($terms[0]->free !== []) {
747 $free = array_keys($terms[0]->free);
748 sort($free, SORT_STRING);
749 throw new LambdaError('free variable ' . $free[0] . ' in definition of ' . $name);
750 }
751 echo isset($this->macros[$name]) ? 'redefined ' : 'defined ', $name, "\n";
752 $this->macros[$name] = $terms[0];
753 return;
754 }
755 foreach ($terms as $i => $term) {
756 $terms[$i] = $this->expand($term);
757 }
758 $term = $terms[0];
759 switch ($command) {
760 case 'expand':
761 echo printed($term), "\n";
762 break;
763 case 'subst':
764 echo printed(substitute($terms[1], $name, $term)), "\n";
765 break;
766 case 'alpha':
767 echo printed(toDebruijn($term), true, true) === printed(toDebruijn($terms[1]), true, true) ? "yes\n" : "no\n";
768 break;
769 case 'debruijn':
770 echo printed(toDebruijn($term), true, true), "\n";
771 break;
772 case 'fv':
773 $free = array_keys($term->free);
774 sort($free, SORT_STRING);
775 echo $free === [] ? '(none)' : implode(' ', $free), "\n";
776 break;
777 case 'size':
778 echo $term->size, "\n";
779 break;
780 case 'depth':
781 echo $term->depth, "\n";
782 break;
783 case 'type':
784 echo (new TypeInference())->show($term), "\n";
785 break;
786 case 'ski':
787 echo printed(toSki($term)), "\n";
788 break;
789 case 'kam':
790 $this->kam($term);
791 break;
792 case 'equal':
793 $left = $this->reduce($term, 'NORMAL', $this->eta);
794 $right = $this->reduce($terms[1], 'NORMAL', $this->eta);
795 if ($left === null || $right === null) {
796 echo "unknown\n";
797 } else {
798 echo printed(canonical($left[0])) === printed(canonical($right[0])) ? "yes\n" : "no\n";
799 }
800 break;
801 default:
802 if ($command === 'church') {
803 $result = $this->reduce($term, 'NORMAL', false);
804 } elseif ($command === 'skireduce') {
805 $result = $this->reduce(toSki($term), 'NORMAL', false, true, true);
806 } else {
807 $result = $this->reduce($term, $strategy, $this->eta, true);
808 }
809 if ($result === null) {
810 echo 'no normal form within ', $this->limit, " steps\n";
811 } elseif ($command === 'church') {
812 echo churchValue(canonical($result[0])), "\n";
813 } else {
814 $text = $command === 'skireduce' ? printed($result[0]) : printed(canonical($result[0]));
815 echo 'result: ', $text, ' (steps: ', $result[1], ")\n";
816 }
817 }
818 }
819
820 public function run(): void
821 {
822 while (($input = fgets(STDIN)) !== false) {
823 $line = str_ends_with($input, "\n") ? substr($input, 0, -1) : $input;
824 if ($line === 'quit') {
825 break;
826 }
827 if ($line === '' || str_starts_with($line, '--')) {
828 continue;
829 }
830 try {
831 $this->command($line);
832 } catch (LambdaError $error) {
833 echo 'error: ', $error->getMessage(), "\n";
834 }
835 }
836 }
837}
