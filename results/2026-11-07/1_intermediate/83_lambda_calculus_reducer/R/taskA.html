6term <- function(kind, name = NULL, left = NULL, right = NULL) {
7 free <- switch(kind, var = name, abs = setdiff(left$free, name), app = union(left$free, right$free), character())
8 list(kind = kind, name = name, left = left, right = right, free = free,
9 size = 1L + (if (is.null(left)) 0L else left$size) + (if (is.null(right)) 0L else right$size),
10 depth = 1L + max(if (is.null(left)) 0L else left$depth, if (is.null(right)) 0L else right$depth))

39new_parser <- function(argument, offset, end_column) {
40 tokens <- list()
41 chars <- strsplit(argument, "", fixed = TRUE)[[1]]
42 i <- 1L
43 while (i <= length(chars)) {
44 c <- chars[[i]]
45 if (c == " ") {
46 i <- i + 1L
47 next
48 }
49 start <- i
50 if (c %in% c("\\", ".", "(", ")", "=", ";")) {
51 kind <- c
52 i <- i + 1L
53 } else if (grepl("^[a-z]$", c)) {
54 kind <- "var"
55 i <- i + 1L
56 while (i <= length(chars) && grepl("^[a-z0-9]$", chars[[i]])) i <- i + 1L
57 } else if (grepl("^[A-Z]$", c)) {
58 kind <- "macro"
59 i <- i + 1L
60 while (i <= length(chars) && grepl("^[A-Z0-9_]$", chars[[i]])) i <- i + 1L
61 } else if (c == "#" && i < length(chars) && grepl("^[0-9]$", chars[[i + 1L]])) {
62 kind <- "number"
63 i <- i + 2L
64 while (i <= length(chars) && grepl("^[0-9]$", chars[[i]])) i <- i + 1L
65 } else {
66 lambda_error(paste0("column ", offset + i - 1L, ": unexpected character '", c, "'"))
67 }
68 tokens[[length(tokens) + 1L]] <- list(kind = kind, text = substr(argument, start, i - 1L), column = offset + start - 1L)
69 }
70 tokens[[length(tokens) + 1L]] <- list(kind = "end", text = "", column = end_column)
71 position <- 1L
72 peek <- function() tokens[[position]]
73 advance <- function() {
74 result <- peek()
75 position <<- position + 1L
76 result
77 }
78 expect <- function(kind) {
79 notation <- if (kind == "var") "variable" else if (kind == "macro") "macro name" else paste0("'", kind, "'")
80 if (peek()$kind != kind) lambda_error(paste0("column ", peek()$column, ": expected ", notation))
81 advance()$text
82 }
83 read_term <- function() {
84 elements <- list()
85 while (peek()$kind %in% c("var", "macro", "number", "(")) {
86 token <- advance()
87 if (token$kind == "(") {
88 elements[[length(elements) + 1L]] <- read_term()
89 expect(")")
90 } else {
91 elements[[length(elements) + 1L]] <- term(token$kind, if (token$kind == "number") as.integer(substring(token$text, 2L)) else token$text)
92 }
93 }
94 if (peek()$kind == "\\") {
95 advance()
96 names <- expect("var")
97 while (peek()$kind == "var") names <- c(names, advance()$text)
98 expect(".")
99 body <- read_term()
100 for (name in rev(names)) body <- abstraction(name, body)
101 elements[[length(elements) + 1L]] <- body
102 }
103 if (length(elements) == 0L) lambda_error(paste0("column ", peek()$column, ": expected term"))
104 result <- elements[[1]]
105 for (element in elements[-1L]) result <- application(result, element)
106 result
107 }
108 finish <- function() {
109 if (peek()$kind != "end") lambda_error(paste0("column ", peek()$column, ": unexpected '", peek()$text, "'"))
110 }
111 list(peek = peek, advance = advance, expect = expect, term = read_term, finish = finish)
112}

142from_debruijn <- function(node) {
143 free <- node$free
144 counter <- 0L
145 visit <- function(node, bindings) {

158substitute <- function(node, name, replacement) {
159 if (node$kind == "var") return(if (node$name == name) replacement else node)
160 if (node$kind == "app") return(application(substitute(node$left, name, replacement), substitute(node$right, name, replacement)))
161 bound <- node$name
162 body <- node$left
163 if (bound == name || !(name %in% body$free)) return(node)
164 if (bound %in% replacement$free) {
165 stem <- sub("[0-9]+$", "", bound)
166 suffix <- 1L
167 while (paste0(stem, suffix) %in% c(replacement$free, body$free)) suffix <- suffix + 1L
168 fresh <- paste0(stem, suffix)
169 body <- substitute(body, bound, variable(fresh))
170 bound <- fresh
171 }
172 abstraction(bound, substitute(body, name, replacement))

178one_step <- function(node, strategy, eta) {
179 if (node$kind == "app") {
180 left <- node$left
181 right <- node$right
182 if (strategy %in% c("NORMAL", "CBN") && left$kind == "abs") return(list(substitute(left$left, left$name, right), "beta"))
183 step <- one_step(left, strategy, eta)
184 if (!is.null(step)) return(list(application(step[[1]], right), step[[2]]))
185 if (strategy != "CBN") {
186 step <- one_step(right, strategy, eta)
187 if (!is.null(step)) return(list(application(left, step[[1]]), step[[2]]))
188 }
189 if (strategy %in% c("APPLICATIVE", "CBV") && left$kind == "abs" && (strategy != "CBV" || right$kind %in% c("var", "abs"))) return(list(substitute(left$left, left$name, right), "beta"))
190 } else if (node$kind == "abs" && strategy %in% c("NORMAL", "APPLICATIVE")) {
191 target <- if (eta) eta_target(node) else NULL
192 if (!is.null(target) && strategy == "NORMAL") return(list(target, "eta"))
193 step <- one_step(node$left, strategy, eta)
194 if (!is.null(step)) return(list(abstraction(node$name, step[[1]]), step[[2]]))
195 if (!is.null(target)) return(list(target, "eta"))
196 }
197 NULL
198}

208ski_step <- function(node) {
209 head <- node
210 args <- list()
211 while (head$kind == "app") {
212 args <- c(list(head$right), args)
213 head <- head$left
214 }
215 arity <- if (head$kind == "constant") c(I = 1L, K = 2L, S = 3L)[[head$name]] else NULL
216 if (!is.null(arity) && length(args) >= arity) {
217 result <- if (head$name == "S") application(application(args[[1]], args[[3]]), application(args[[2]], args[[3]])) else args[[1]]
218 for (arg in tail(args, length(args) - arity)) result <- application(result, arg)
219 return(list(result, head$name))
220 }
221 for (i in seq_along(args)) {
222 step <- ski_step(args[[i]])
223 if (!is.null(step)) {
224 args[[i]] <- step[[1]]
225 result <- head
226 for (arg in args) result <- application(result, arg)
227 return(list(result, step[[2]]))
228 }
229 }
230 NULL

241 prune <- function(type) {

244 key <- as.character(type$id)

251 occurs <- function(id, type) {

255 unify <- function(left, right) {
256 left <- prune(left)
257 right <- prune(right)
258 if (identical(left, right)) return(invisible(NULL))
259 if (left$kind == "variable") {

261 assign(as.character(left$id), right, envir = links)
262 } else if (right$kind == "variable") {
263 unify(right, left)
264 } else {
265 unify(left$left, right$left)

269 infer <- function(node, environment = list()) {

271 bound <- environment[[node$name]]

394 kam <- function(node) {
395 code <- to_debruijn(node)
396 environment <- list()
397 stack <- list()
398 count <- 0L
399 repeat {
400 kind <- switch(code$kind, app = "push", abs = if (length(stack) == 0L) NULL else "grab", index = "access", NULL)
401 if (is.null(kind)) break
402 if (count == state$limit) {
403 emit("no result within ", state$limit, " transitions")
404 return(invisible(NULL))
405 }
406 if (kind == "push") {

408 code <- code$left
409 } else if (kind == "grab") {

411 stack <- head(stack, -1L)

416 environment <- closure$environment

420 }
421 result <- materialize(code, environment)
422 for (closure in rev(stack)) result <- application(result, materialize(closure$code, closure$environment))
423 emit("result: ", printed(from_debruijn(result)), " (transitions: ", count, ")")
424 }

433 if (name %in% c("steps", "eta")) {

444 } else {
445 if (length(words) != 1L || !grepl("^[A-Z][A-Z0-9_]*$", words[[1]])) lambda_error("bad argument")
446 macro <- words[[1]]

449 for (other in sort(names(state$macros), method = "radix")) {
450 if (!(other %in% c("S", "K", "I")) && macro %in% references(state$macros[[other]])) lambda_error(paste0(macro, " is used by ", other))

453 emit("removed ", macro)
454 }
455 return(invisible(NULL))
