6struct LambdaTerm
7 kind::Symbol
8 name::String
9 number::Int
10 children::Vector{LambdaTerm}
11 free::Set{String}
12 size::Int
13 depth::Int
14end

16function make_term(kind, name="", number=0, children=LambdaTerm[])
17 free = if kind == :variable
18 Set([name])
19 elseif kind == :abstraction
20 setdiff(children[1].free, [name])
21 elseif kind == :application
22 union(children[1].free, children[2].free)
23 else
24 Set{String}()
25 end
26 size = 1 + sum((child.size for child in children); init=0)
27 depth = 1 + maximum((child.depth for child in children); init=0)
28 LambdaTerm(kind, name, number, children, free, size, depth)
29end
30varterm(name) = make_term(:variable, name)
31absterm(name, body) = make_term(:abstraction, name, 0, [body])
32appterm(left, right) = make_term(:application, "", 0, [left, right])

34function print_term(term, at_end=true; db=false)
35 kind = term.kind
36 kind in (:variable, :macro, :constant) && return term.name
37 kind == :number && return "#$(term.number)"
38 kind == :index && return string(term.number)
39 if kind == :abstraction
40 names = String[]
41 body = term
42 while body.kind == :abstraction
43 push!(names, body.name)
44 body = body.children[1]
45 end
46 text = db ? repeat("\\", length(names)) * print_term(body; db=true) :
47 "\\" * join(names, " ") * ". " * print_term(body)
48 return at_end ? text : "(" * text * ")"
49 end
50 left, right = term.children
51 a = left.kind == :abstraction ? "(" * print_term(left; db=db) * ")" :
52 print_term(left, false; db=db)
53 b = right.kind == :application ? "(" * print_term(right; db=db) * ")" :
54 print_term(right, at_end; db=db)
55 a * " " * b
56end

68function lambda_parser(argument, offset, end_column)
69 tokens = LambdaToken[]
70 i = 1
71 while i <= length(argument)
72 c = argument[i]
73 if c == ' '
74 i += 1
75 continue
76 end
77 start = i
78 if c in "\\.()=;"
79 kind = :symbol
80 i += 1
81 elseif 'a' <= c <= 'z'
82 kind = :variable
83 i += 1
84 while i <= length(argument) && ('a' <= argument[i] <= 'z' || isdigit(argument[i]))
85 i += 1
86 end
87 elseif 'A' <= c <= 'Z'
88 kind = :macro
89 i += 1
90 while i <= length(argument) &&
91 ('A' <= argument[i] <= 'Z' || isdigit(argument[i]) || argument[i] == '_')
92 i += 1
93 end
94 elseif c == '#' && i < length(argument) && isdigit(argument[i + 1])
95 kind = :number
96 i += 2
97 while i <= length(argument) && isdigit(argument[i])
98 i += 1
99 end
100 else
101 lambda_error("column $(offset + i - 1): unexpected character '$c'")
102 end
103 push!(tokens, LambdaToken(kind, argument[start:i - 1], offset + start - 1))
104 end
105 push!(tokens, LambdaToken(:eof, "", end_column))
106 LambdaParser(tokens, 1)
107end
108peek_token(p) = p.tokens[p.position]
109function read_token!(p::LambdaParser)
110 token = peek_token(p)
111 p.position += 1
112 token
113end
114function expect_token!(p, expected)
115 token = peek_token(p)
116 accepted = expected == "variable" ? token.kind == :variable :
117 expected == "macro name" ? token.kind == :macro : token.text == expected
118 if !accepted
119 notation = expected in ("variable", "macro name") ? expected : "'$expected'"
120 lambda_error("column $(token.column): expected $notation")
121 end
122 read_token!(p).text
123end
124function read_term!(p)
125 elements = LambdaTerm[]
126 while peek_token(p).kind in (:variable, :macro, :number) || peek_token(p).text == "("
127 token = read_token!(p)
128 if token.text == "("
129 push!(elements, read_term!(p))
130 expect_token!(p, ")")
131 elseif token.kind == :number
132 push!(elements, make_term(:number, "", parse(Int, token.text[2:end])))
133 else
134 push!(elements, make_term(token.kind, token.text))
135 end
136 end
137 if peek_token(p).text == "\\"
138 read_token!(p)
139 names = [expect_token!(p, "variable")]
140 while peek_token(p).kind == :variable
141 push!(names, read_token!(p).text)
142 end
143 expect_token!(p, ".")
144 body = read_term!(p)
145 for name in Iterators.reverse(names)
146 body = absterm(name, body)
147 end
148 push!(elements, body)
149 end
150 isempty(elements) && lambda_error("column $(peek_token(p).column): expected term")
151 foldl(appterm, elements)
152end
153function finish_parse(p)
154 token = peek_token(p)
155 token.kind == :eof || lambda_error("column $(token.column): unexpected '$(token.text)'")
156end

159 result = String[]
160 function visit(node)
161 node.kind == :macro && push!(result, node.name)
162 for child in node.children
163 visit(child)
164 end

166 visit(term)

169name_at(index) = string(Char(97 + index % 26)) * (index >= 26 ? string(index ÷ 26) : "")

171function canonical_term(term)
172 counter = 0
173 function visit(node, bindings)
174 if node.kind == :variable
175 return varterm(get(bindings, node.name, node.name))
176 elseif node.kind == :abstraction
177 while name_at(counter) in term.free
178 counter += 1
179 end
180 name = name_at(counter)
181 counter += 1
182 inner = copy(bindings)
183 inner[node.name] = name
184 return absterm(name, visit(node.children[1], inner))
185 elseif node.kind == :application
186 left = visit(node.children[1], bindings)
187 right = visit(node.children[2], bindings)
188 return appterm(left, right)
189 end
190 node
191 end
192 visit(term, Dict{String,String}())
193end
194function to_debruijn(term, bindings=String[])
195 if term.kind == :variable
196 index = findfirst(==(term.name), bindings)
197 return index === nothing ? term : make_term(:index, "", index - 1)
198 elseif term.kind == :abstraction
199 return absterm("", to_debruijn(term.children[1], [term.name; bindings]))
200 end
201 appterm(to_debruijn(term.children[1], bindings), to_debruijn(term.children[2], bindings))
202end
203function from_debruijn(term)
204 counter = 0
205 function visit(node, bindings)
206 if node.kind == :index
207 return varterm(bindings[node.number + 1])
208 elseif node.kind == :variable
209 return node
210 elseif node.kind == :abstraction
211 while name_at(counter) in term.free
212 counter += 1
213 end
214 name = name_at(counter)
215 counter += 1
216 return absterm(name, visit(node.children[1], [name; bindings]))
217 end
218 left = visit(node.children[1], bindings)
219 right = visit(node.children[2], bindings)
220 appterm(left, right)
221 end
222 visit(term, String[])
223end

225function substitute(term, name, replacement)
226 if term.kind == :variable
227 return term.name == name ? replacement : term
228 elseif term.kind == :application
229 return appterm(substitute(term.children[1], name, replacement),
230 substitute(term.children[2], name, replacement))
231 end
232 bound = term.name
233 body = term.children[1]
234 (bound == name || !(name in body.free)) && return term
235 if bound in replacement.free
236 stem = replace(bound, r"[0-9]+$" => "")
237 suffix = 1
238 while stem * string(suffix) in replacement.free || stem * string(suffix) in body.free
239 suffix += 1
240 end
241 fresh = stem * string(suffix)
242 body = substitute(body, bound, varterm(fresh))
243 bound = fresh
244 end
245 absterm(bound, substitute(body, name, replacement))
246end
247function eta_target(term)
248 body = term.children[1]
249 if body.kind == :application
250 left, right = body.children
251 if right.kind == :variable && right.name == term.name && !(term.name in left.free)
252 return left
253 end
254 end
255 nothing
256end
257function lambda_step(term, strategy, eta)
258 if term.kind == :application
259 left, right = term.children
260 if strategy in ("NORMAL", "CBN") && left.kind == :abstraction
261 return substitute(left.children[1], left.name, right), "beta"
262 end
263 step = lambda_step(left, strategy, eta)
264 step !== nothing && return (appterm(step[1], right), step[2])
265 if strategy != "CBN"
266 step = lambda_step(right, strategy, eta)
267 step !== nothing && return (appterm(left, step[1]), step[2])
268 end
269 if strategy in ("APPLICATIVE", "CBV") && left.kind == :abstraction &&
270 (strategy != "CBV" || right.kind in (:variable, :abstraction))
271 return substitute(left.children[1], left.name, right), "beta"
272 end
273 elseif term.kind == :abstraction && strategy in ("NORMAL", "APPLICATIVE")
274 target = eta ? eta_target(term) : nothing
275 target !== nothing && strategy == "NORMAL" && return (target, "eta")
276 step = lambda_step(term.children[1], strategy, eta)
277 step !== nothing && return (absterm(term.name, step[1]), step[2])
278 target !== nothing && return (target, "eta")
279 end
280 nothing
281end
282function ski_abstract(name, term)
283 !(name in term.free) && return appterm(make_term(:constant, "K"), term)
284 term.kind == :variable && term.name == name && return make_term(:constant, "I")
285 left, right = term.children
286 if right.kind == :variable && right.name == name && !(name in left.free)
287 return left
288 end
289 appterm(appterm(make_term(:constant, "S"), ski_abstract(name, left)), ski_abstract(name, right))
290end
291function to_ski(term)
292 term.kind == :variable && return term
293 term.kind == :abstraction && return ski_abstract(term.name, to_ski(term.children[1]))
294 appterm(to_ski(term.children[1]), to_ski(term.children[2]))
295end
296function ski_step(term)
297 head = term
298 args = LambdaTerm[]
299 while head.kind == :application
300 pushfirst!(args, head.children[2])
301 head = head.children[1]
302 end
303 arity = head.kind == :constant ? (head.name == "I" ? 1 : head.name == "K" ? 2 : 3) : 0
304 if arity > 0 && length(args) >= arity
305 result = head.name == "S" ? appterm(appterm(args[1], args[3]), appterm(args[2], args[3])) : args[1]
306 for arg in args[arity + 1:end]
307 result = appterm(result, arg)
308 end
309 return result, head.name
310 end
311 for i in eachindex(args)
312 step = ski_step(args[i])
313 if step !== nothing
314 args[i] = step[1]
315 return foldl(appterm, args; init=head), step[2]
316 end
317 end
318 nothing
319end

321mutable struct SimpleType
322 id::Int
323 left::Union{Nothing,SimpleType}
324 right::Union{Nothing,SimpleType}
325 link::Union{Nothing,SimpleType}
326end
327struct NotTypable <: Exception end
328function prune_type(type)
329 path = SimpleType[]
330 while type.link !== nothing
331 push!(path, type)
332 type = type.link
333 end
334 for node in path
335 node.link = type
336 end
337 type
338end
339function occurs_in(id, type)
340 type = prune_type(type)
341 type.id >= 0 ? type.id == id : occurs_in(id, type.left) || occurs_in(id, type.right)
342end
343function unify_types(left, right)
344 left, right = prune_type(left), prune_type(right)
345 left === right && return
346 if left.id >= 0
347 occurs_in(left.id, right) && throw(NotTypable())
348 left.link = right
349 elseif right.id >= 0
350 unify_types(right, left)
351 else
352 unify_types(left.left, right.left)
353 unify_types(left.right, right.right)
354 end
355end
356function principal_type(term)
357 next_id = 0
358 free = Dict{String,SimpleType}()
359 function fresh()
360 type = SimpleType(next_id, nothing, nothing, nothing)
361 next_id += 1
362 type
363 end
364 function infer(node, environment)
365 if node.kind == :variable
366 haskey(environment, node.name) && return environment[node.name]
367 if !haskey(free, node.name)
368 free[node.name] = fresh()
369 end
370 return free[node.name]
371 elseif node.kind == :abstraction
372 parameter = fresh()
373 inner = copy(environment)
374 inner[node.name] = parameter
375 return SimpleType(-1, parameter, infer(node.children[1], inner), nothing)
376 end
377 function_type = infer(node.children[1], environment)
378 argument_type = infer(node.children[2], environment)
379 result = fresh()
380 unify_types(function_type, SimpleType(-1, argument_type, result, nothing))
381 result
382 end
383 try
384 type = infer(term, Dict{String,SimpleType}())
385 names = Dict{Int,String}()
386 function visit(node)
387 node = prune_type(node)
388 if node.id >= 0
389 if !haskey(names, node.id)
390 names[node.id] = name_at(length(names))
391 end
392 return names[node.id]
393 end
394 left = visit(node.left)
395 prune_type(node.left).id < 0 && (left = "(" * left * ")")
396 left * " -> " * visit(node.right)
397 end
398 visit(type)
399 catch error
400 error isa NotTypable || rethrow()
401 "not typable"
402 end
403end

405function church_value(term)
406 if term.kind == :abstraction && term.children[1].kind == :abstraction
407 f = term.name
408 x = term.children[1].name
409 body = term.children[1].children[1]
410 count = 0
411 while body.kind == :application && body.children[1].kind == :variable &&
412 body.children[1].name == f
413 count += 1
414 body = body.children[2]
415 end
416 body.kind == :variable && body.name == x && return string(count)
417 original = term.children[1].children[1]
418 original.kind == :variable && original.name == f && return "true"
419 end
420 if term.kind == :abstraction
421 body = term.children[1]
422 if body.kind == :application && body.children[1].kind == :application
423 head, p = body.children[1].children
424 q = body.children[2]
425 if head.kind == :variable && head.name == term.name &&
426 !(term.name in p.free) && !(term.name in q.free)
427 return "(" * church_value(p) * ", " * church_value(q) * ")"
428 end
429 end
430 end

438function materialize_closure(code, environment, level=0)
439 while code.kind == :index && code.number >= level
440 closure = environment[code.number - level + 1]
441 code, environment = closure.code, closure.environment
442 level = 0
443 end
444 code.kind == :abstraction && return absterm("", materialize_closure(code.children[1], environment, level + 1))
445 if code.kind == :application
446 return appterm(materialize_closure(code.children[1], environment, level),
447 materialize_closure(code.children[2], environment, level))
448 end
449 code
450end

452const RESERVED_MACROS = Set(["S", "K", "I", "NORMAL", "APPLICATIVE", "CBN", "CBV"])
453const STRATEGIES = ("NORMAL", "APPLICATIVE", "CBN", "CBV")
454const COMMANDS = Set(["let", "unlet", "macros", "steps", "eta", "limit", "expand", "subst",
455 "reduce", "equal", "alpha", "debruijn", "fv", "size", "depth", "type", "church", "ski", "skireduce", "kam"])
456mutable struct LambdaWorkbench
457 macros::Dict{String,LambdaTerm}
458 steps::Bool
459 eta::Bool
460 limit::Int
461end
462function LambdaWorkbench()
463 macros = Dict(
464 "S" => absterm("x", absterm("y", absterm("z",
465 appterm(appterm(varterm("x"), varterm("z")), appterm(varterm("y"), varterm("z")))))),
466 "K" => absterm("x", absterm("y", varterm("x"))),
467 "I" => absterm("x", varterm("x")))
468 LambdaWorkbench(macros, false, false, 10000)

470function macro_depends(w, name, target, visited=Set{String}())
471 name in visited && return false
472 push!(visited, name)
473 (!haskey(w.macros, name) || name in ("S", "K", "I")) && return false
474 any(ref -> ref == target || macro_depends(w, ref, target, visited), macro_references(w.macros[name]))
475end
476function expand_term(w, term)
477 if term.kind == :macro
478 haskey(w.macros, term.name) || lambda_error("unbound macro $(term.name)")
479 return expand_term(w, w.macros[term.name])
480 elseif term.kind == :number
481 body = varterm("x")
482 for _ in 1:term.number
483 body = appterm(varterm("f"), body)
484 end
485 return absterm("f", absterm("x", body))
486 elseif term.kind == :abstraction
487 return absterm(term.name, expand_term(w, term.children[1]))
488 elseif term.kind == :application
489 left = expand_term(w, term.children[1])
490 right = expand_term(w, term.children[2])
491 return appterm(left, right)
492 end
493 term
494end
495function reduce_term(w, term; strategy="NORMAL", eta=w.eta, display=false, combinators=false)
496 count = 0
497 while true
498 step = combinators ? ski_step(term) : lambda_step(term, strategy, eta)
499 step === nothing && return (term, count)
500 count == w.limit && return nothing
501 term = step[1]
502 count += 1
503 display && w.steps && println("step $count ($(step[2])): ", print_term(term))
504 end
505end
506function run_krivine(w, term)
507 code = to_debruijn(term)
508 environment = LambdaClosure[]
509 stack = LambdaClosure[]
510 count = 0
511 while true
512 kind = code.kind == :application ? "push" :
513 code.kind == :abstraction && !isempty(stack) ? "grab" :
514 code.kind == :index ? "access" : ""
515 isempty(kind) && break
516 if count == w.limit
517 println("no result within $(w.limit) transitions")
518 return
519 end
520 if kind == "push"
521 push!(stack, LambdaClosure(code.children[2], environment))
522 code = code.children[1]
523 elseif kind == "grab"
524 environment = [pop!(stack); environment]
525 code = code.children[1]
526 else
527 closure = environment[code.number + 1]
528 code, environment = closure.code, closure.environment
529 end
530 count += 1
531 if w.steps
532 println("step $count ($kind): ", print_term(code; db=true),
533 " [env $(length(environment)), stack $(length(stack))]")
534 end
535 end
536 result = materialize_closure(code, environment)
537 for closure in Iterators.reverse(stack)
538 result = appterm(result, materialize_closure(closure.code, closure.environment))
539 end
540 println("result: ", print_term(from_debruijn(result)), " (transitions: $count)")
541end

543function simple_command!(w, command, argument)
544 words = split(argument)
545 if command in ("steps", "eta")
546 length(words) == 1 && words[1] in ("on", "off") || lambda_error("bad argument")
547 if command == "steps"
548 w.steps = words[1] == "on"
549 else
550 w.eta = words[1] == "on"
551 end
552 elseif command == "limit"
553 length(words) == 1 && occursin(r"^[0-9]{1,5}$", words[1]) &&
554 1 <= parse(Int, words[1]) <= 10000 || lambda_error("bad argument")
555 w.limit = parse(Int, words[1])
556 elseif command == "macros"
557 isempty(words) || lambda_error("bad argument")
558 names = sort([name for name in keys(w.macros) if !(name in ("S", "K", "I"))])
559 isempty(names) && println("(none)")
560 for name in names
561 println(name, " = ", print_term(w.macros[name]))
562 end
563 else
564 length(words) == 1 && occursin(r"^[A-Z][A-Z0-9_]*$", words[1]) || lambda_error("bad argument")
565 name = words[1]
566 name in RESERVED_MACROS && lambda_error("reserved name $name")
567 haskey(w.macros, name) || lambda_error("unbound macro $name")
568 for user in sort(collect(keys(w.macros)))
569 if !(user in ("S", "K", "I")) && name in macro_references(w.macros[user])
570 lambda_error("$name is used by $user")
571 end
572 end
573 delete!(w.macros, name)
574 println("removed $name")
575 end
576end
577function lambda_command!(w, line)
578 separator = findfirst(==(' '), line)
579 command = separator === nothing ? line : line[1:separator - 1]
580 argument = separator === nothing ? "" : line[separator + 1:end]
581 command in COMMANDS || lambda_error("unknown command $command")
582 if command in ("steps", "eta", "limit", "macros", "unlet")
583 simple_command!(w, command, argument)
584 return
585 end
586 parser = lambda_parser(argument, length(command) + 2, length(line) + 1)
587 strategy = "NORMAL"
588 name = ""
589 terms = LambdaTerm[]
590 if command in ("let", "subst")
591 name = expect_token!(parser, command == "let" ? "macro name" : "variable")
592 expect_token!(parser, "=")
593 push!(terms, read_term!(parser))
594 if command == "subst"
595 expect_token!(parser, ";")
596 push!(terms, read_term!(parser))
597 end
598 elseif command in ("equal", "alpha")
599 push!(terms, read_term!(parser))
600 expect_token!(parser, ";")
601 push!(terms, read_term!(parser))
602 else
603 if command == "reduce" && peek_token(parser).kind == :macro && peek_token(parser).text in STRATEGIES
604 strategy = read_token!(parser).text
605 end
606 push!(terms, read_term!(parser))
607 end
608 finish_parse(parser)
609 if command == "let"
610 name in RESERVED_MACROS && lambda_error("reserved name $name")
611 for ref in macro_references(terms[1])
612 (ref == name || macro_depends(w, ref, name)) && lambda_error("recursive definition of $name")
613 haskey(w.macros, ref) || lambda_error("unbound macro $ref")
614 end
615 isempty(terms[1].free) || lambda_error("free variable $(minimum(terms[1].free)) in definition of $name")
616 println(haskey(w.macros, name) ? "redefined $name" : "defined $name")
617 w.macros[name] = terms[1]
618 return
619 end
620 for i in eachindex(terms)
621 terms[i] = expand_term(w, terms[i])
622 end
623 term = terms[1]
624 if command == "expand"
625 println(print_term(term))
626 elseif command == "subst"
627 println(print_term(substitute(terms[2], name, term)))
628 elseif command == "alpha"
629 println(print_term(to_debruijn(term); db=true) == print_term(to_debruijn(terms[2]); db=true) ? "yes" : "no")
630 elseif command == "debruijn"
631 println(print_term(to_debruijn(term); db=true))
632 elseif command == "fv"
633 println(isempty(term.free) ? "(none)" : join(sort(collect(term.free)), " "))
634 elseif command == "size"
635 println(term.size)
636 elseif command == "depth"
637 println(term.depth)
638 elseif command == "type"
639 println(principal_type(term))
640 elseif command == "ski"
641 println(print_term(to_ski(term)))
642 elseif command == "kam"
643 run_krivine(w, term)
644 elseif command == "equal"
645 left = reduce_term(w, term)
646 right = reduce_term(w, terms[2])
647 println(left === nothing || right === nothing ? "unknown" :
648 print_term(canonical_term(left[1])) == print_term(canonical_term(right[1])) ? "yes" : "no")
649 else
650 result = if command == "church"
651 reduce_term(w, term; eta=false)
652 elseif command == "skireduce"
653 reduce_term(w, to_ski(term); eta=false, display=true, combinators=true)
654 else
655 reduce_term(w, term; strategy=strategy, display=true)
656 end
657 if result === nothing
658 println("no normal form within $(w.limit) steps")
659 elseif command == "church"
660 println(church_value(canonical_term(result[1])))
661 else
662 text = print_term(command == "skireduce" ? result[1] : canonical_term(result[1]))
663 println("result: $text (steps: $(result[2]))")
664 end
665 end
666end
667function main()
668 workbench = LambdaWorkbench()
669 for line in eachline()
670 line == "quit" && break
671 (isempty(line) || startswith(line, "--")) && continue
