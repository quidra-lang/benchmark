20function lex(argument, start, end) {
21 const result = [];
22 for (let i = 0; i < argument.length;) {
23 const c = argument[i],
24 column = start + i;
25 if (c === " ") {
26 i++;
27 continue;
28 }
29 if ("\\.()=;".includes(c)) {
30 result.push({ kind: c, text: c, column });
31 i++;
32 continue;
33 }
34 const begin = i;
35 let kind;
36 if (c >= "a" && c <= "z") {
37 kind = "variable";
38 while (i < argument.length && /[a-z0-9]/.test(argument[i])) i++;
39 } else if (c >= "A" && c <= "Z") {
40 kind = "macro";
41 while (i < argument.length && /[A-Z0-9_]/.test(argument[i])) i++;
42 } else if (c === "#" && i + 1 < argument.length && /[0-9]/.test(argument[i + 1])) {
43 kind = "numeral";
44 i += 2;
45 while (i < argument.length && /[0-9]/.test(argument[i])) i++;
46 } else fail(`column ${column}: unexpected character '${c}'`);
47 result.push({ kind, text: argument.slice(begin, i), column });
48 }
49 result.push({ kind: "end", text: "", column: end });
50 return result;
51}
52class Parser {
53 constructor(tokens) {
54 this.tokens = tokens;
55 this.at = 0;
56 }
57 peek() {
58 return this.tokens[this.at];
59 }
60 expect(kind) {
61 const token = this.peek();
62 if (token.kind !== kind)
63 fail(
64 `column ${token.column}: expected ${kind === "variable" ? "variable" : kind === "macro" ? "macro name" : "'" + kind + "'"}`,
65 );
66 this.at++;
67 return token;
68 }
69 term() {
70 const elements = [];
71 while (["variable", "macro", "numeral", "("].includes(this.peek().kind))
72 elements.push(this.atom());
73 if (this.peek().kind === "\\") {
74 this.at++;
75 const names = [this.expect("variable").text];
76 while (this.peek().kind === "variable") names.push(this.tokens[this.at++].text);
77 this.expect(".");
78 let body = this.term();
79 for (let i = names.length - 1; i >= 0; i--) body = abstraction(names[i], body);
80 elements.push(body);
81 }
82 if (!elements.length) fail(`column ${this.peek().column}: expected term`);
83 return elements.slice(1).reduce(application, elements[0]);
84 }
85 atom() {
86 const token = this.tokens[this.at++];
87 if (token.kind === "variable") return variable(token.text);
88 if (token.kind === "macro") return { kind: "macro", name: token.text };
89 if (token.kind === "numeral")
90 return { kind: "numeral", value: Number(token.text.slice(1)) };
91 const term = this.term();
92 this.expect(")");
93 return term;
94 }
95 end() {
96 const token = this.peek();
97 if (token.kind !== "end") fail(`column ${token.column}: unexpected '${token.text}'`);
98 }

241function oneStep(term, strategy, eta) {
242 if (term.kind === "variable") return null;
243 if (term.kind === "abstraction") {
244 if (strategy === "CBN" || strategy === "CBV") return null;
245 const candidate = eta ? etaBody(term) : null;
246 if (strategy === "NORMAL" && candidate !== null) return { term: candidate, kind: "eta" };
247 const step = oneStep(term.body, strategy, eta);
248 if (step !== null) return { term: abstraction(term.name, step.term), kind: step.kind };
249 return candidate === null ? null : { term: candidate, kind: "eta" };
250 }
251 const early = strategy === "NORMAL" || strategy === "CBN";
252 if (early && term.left.kind === "abstraction")
253 return { term: substitute(term.left.body, term.left.name, term.right), kind: "beta" };
254 const left = oneStep(term.left, strategy, eta);
255 if (left !== null) return { term: application(left.term, term.right), kind: left.kind };
256 if (strategy !== "CBN") {
257 const right = oneStep(term.right, strategy, eta);
258 if (right !== null) return { term: application(term.left, right.term), kind: right.kind };
259 }
260 if (
261 !early &&
262 term.left.kind === "abstraction" &&
263 (strategy !== "CBV" || term.right.kind === "variable" || term.right.kind === "abstraction")
264 )
265 return { term: substitute(term.left.body, term.left.name, term.right), kind: "beta" };
266 return null;

285function skiStep(term) {
286 let head = term;
287 const args = [];
288 while (head.kind === "application") {
289 args.unshift(head.right);
290 head = head.left;
291 }
292 let result = null,
293 used = 0;
294 if (head.kind === "constant") {
295 if (head.name === "I" && args.length >= 1) {
296 result = args[0];
297 used = 1;
298 } else if (head.name === "K" && args.length >= 2) {
299 result = args[0];
300 used = 2;
301 } else if (head.name === "S" && args.length >= 3) {
302 result = application(application(args[0], args[2]), application(args[1], args[2]));
303 used = 3;
304 }
305 }
306 if (result !== null)
307 return { term: args.slice(used).reduce(application, result), kind: head.name };
308 for (let i = 0; i < args.length; i++) {
309 const step = skiStep(args[i]);
310 if (step !== null) {
311 args[i] = step.term;
312 return { term: args.reduce(application, head), kind: step.kind };
313 }
314 }
315 return null;
