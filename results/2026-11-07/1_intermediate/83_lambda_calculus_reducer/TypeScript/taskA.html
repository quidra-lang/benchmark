43function lex(argument: string, start: number, end: number): Token[] {
44 const result: Token[] = [];
45 for (let i = 0; i < argument.length;) {
46 const c = argument[i],
47 column = start + i;
48 if (c === " ") {
49 i++;
50 continue;
51 }
52 if ("\\.()=;".includes(c)) {
53 result.push({ kind: c, text: c, column });
54 i++;
55 continue;
56 }
57 const begin = i;
58 let kind: string;
59 if (c >= "a" && c <= "z") {
60 kind = "variable";
61 while (i < argument.length && /[a-z0-9]/.test(argument[i])) i++;
62 } else if (c >= "A" && c <= "Z") {
63 kind = "macro";
64 while (i < argument.length && /[A-Z0-9_]/.test(argument[i])) i++;
65 } else if (c === "#" && i + 1 < argument.length && /[0-9]/.test(argument[i + 1])) {
66 kind = "numeral";
67 i += 2;
68 while (i < argument.length && /[0-9]/.test(argument[i])) i++;
69 } else fail(`column ${column}: unexpected character '${c}'`);
70 result.push({ kind, text: argument.slice(begin, i), column });
71 }
72 result.push({ kind: "end", text: "", column: end });
73 return result;
74}

76class Parser {
77 at = 0;
78 constructor(readonly tokens: Token[]) {}
79 peek(): Token {
80 return this.tokens[this.at];
81 }
82 expect(kind: string): Token {
83 const token = this.peek();
84 if (token.kind !== kind) {
85 const notation =
86 kind === "variable"
87 ? "variable"
88 : kind === "macro"
89 ? "macro name"
90 : "'" + kind + "'";
91 fail(`column ${token.column}: expected ${notation}`);
92 }
93 this.at++;
94 return token;
95 }
96 term(): Term {
97 const elements: Term[] = [];
98 while (["variable", "macro", "numeral", "("].includes(this.peek().kind))
99 elements.push(this.atom());
100 if (this.peek().kind === "\\") {
101 this.at++;
102 const names = [this.expect("variable").text];
103 while (this.peek().kind === "variable") names.push(this.tokens[this.at++].text);
104 this.expect(".");
105 let body = this.term();
106 for (let i = names.length - 1; i >= 0; i--) body = abstraction(names[i], body);
107 elements.push(body);
108 }
109 if (elements.length === 0) fail(`column ${this.peek().column}: expected term`);
110 let result = elements[0];
111 for (const element of elements.slice(1)) result = application(result, element);
112 return result;
113 }
114 atom(): Term {
115 const token = this.tokens[this.at++];
116 if (token.kind === "variable") return variable(token.text);
117 if (token.kind === "macro") return { kind: "macro", name: token.text };
118 if (token.kind === "numeral")
119 return { kind: "numeral", value: Number(token.text.slice(1)) };
120 const term = this.term();
121 this.expect(")");
122 return term;
123 }
124 end(): void {
125 const token = this.peek();
126 if (token.kind !== "end") fail(`column ${token.column}: unexpected '${token.text}'`);
127 }
128}

131function free(term: Term): Set<string> {
132 const cached = freeCache.get(term);
133 if (cached !== undefined) return cached;
134 let result: Set<string>;
135 if (term.kind === "variable") result = new Set([term.name]);

137 result = new Set(free(term.body));
138 result.delete(term.name);
139 } else if (term.kind === "application")

141 else result = new Set();

146function print(term: Term, end = true): string {
147 if (term.kind === "variable" || term.kind === "macro" || term.kind === "constant")
148 return term.name;
149 if (term.kind === "numeral") return "#" + term.value;
150 if (term.kind === "abstraction") {
151 const names: string[] = [];
152 let body: Term = term;
153 while (body.kind === "abstraction") {
154 names.push(body.name);
155 body = body.body;
156 }
157 const text = "\\" + names.join(" ") + ". " + print(body);
158 return end ? text : "(" + text + ")";
159 }
160 const left =
161 term.left.kind === "abstraction" ? "(" + print(term.left) + ")" : print(term.left, false);
162 const right =
163 term.right.kind === "application" ? "(" + print(term.right) + ")" : print(term.right, end);
164 return left + " " + right;

174function substitute(term: Term, name: string, replacement: Term): Term {
175 if (term.kind === "variable") return term.name === name ? replacement : term;
176 if (term.kind === "application")
177 return application(
178 substitute(term.left, name, replacement),
179 substitute(term.right, name, replacement),
180 );
181 if (term.kind !== "abstraction") throw new Error("Substitution requires a pure term");
182 if (term.name === name || !free(term.body).has(name)) return term;
183 if (!free(replacement).has(term.name))
184 return abstraction(term.name, substitute(term.body, name, replacement));
185 const base = term.name.replace(/[0-9]+$/, "");
186 let next: string,
187 i = 1;
188 do {
189 next = base + i++;
190 } while (free(replacement).has(next) || free(term.body).has(next));
191 return abstraction(
192 next,
193 substitute(substitute(term.body, term.name, variable(next)), name, replacement),
194 );
195}

198 if (term.kind === "variable") {
199 const index = environment.indexOf(term.name);
200 return index < 0 ? { kind: "free", name: term.name } : { kind: "index", value: index };
201 }
202 if (term.kind === "abstraction")
203 return { kind: "abstraction", body: toDB(term.body, [term.name, ...environment]) };
204 if (term.kind !== "application") throw new Error("de Bruijn conversion requires a pure term");
205 return {
206 kind: "application",
207 left: toDB(term.left, environment),
208 right: toDB(term.right, environment),

212function printDB(term: DBTerm, end = true): string {
213 if (term.kind === "free") return term.name;
214 if (term.kind === "index") return String(term.value);
215 if (term.kind === "abstraction") {
216 let body: DBTerm = term,
217 prefix = "";
218 while (body.kind === "abstraction") {
219 prefix += "\\";
220 body = body.body;
221 }
222 const text = prefix + printDB(body);
223 return end ? text : "(" + text + ")";
224 }
225 const left =
226 term.left.kind === "abstraction"
227 ? "(" + printDB(term.left) + ")"
228 : printDB(term.left, false);
229 const right =
230 term.right.kind === "application"
231 ? "(" + printDB(term.right) + ")"
232 : printDB(term.right, end);
233 return left + " " + right;

239function fromDB(term: DBTerm): Term {
240 const names = new Set<string>();
241 function collect(node: DBTerm): void {
242 if (node.kind === "free") names.add(node.name);
243 else if (node.kind === "abstraction") collect(node.body);
244 else if (node.kind === "application") {
245 collect(node.left);
246 collect(node.right);
247 }
248 }
249 collect(term);
250 let counter = 0;
251 function walk(node: DBTerm, environment: string[]): Term {
252 if (node.kind === "free") return variable(node.name);
253 if (node.kind === "index") return variable(environment[node.value]);
254 if (node.kind === "abstraction") {
255 let name: string;
256 do {
257 name = generatedName(counter++);
258 } while (names.has(name));
259 return abstraction(name, walk(node.body, [name, ...environment]));
260 }
261 return application(walk(node.left, environment), walk(node.right, environment));
262 }
263 return walk(term, []);
264}

278function oneStep(term: Term, strategy: string, eta: boolean): Step | null {
279 if (term.kind === "variable") return null;
280 if (term.kind === "abstraction") {
281 if (strategy === "CBN" || strategy === "CBV") return null;
282 const candidate = eta ? etaBody(term) : null;
283 if (strategy === "NORMAL" && candidate !== null) return { term: candidate, kind: "eta" };
284 const step = oneStep(term.body, strategy, eta);
285 if (step !== null) return { term: abstraction(term.name, step.term), kind: step.kind };
286 return candidate === null ? null : { term: candidate, kind: "eta" };
287 }
288 if (term.kind !== "application") throw new Error("Reduction requires a pure term");
289 const early = strategy === "NORMAL" || strategy === "CBN";
290 if (early && term.left.kind === "abstraction")
291 return { term: substitute(term.left.body, term.left.name, term.right), kind: "beta" };
292 const left = oneStep(term.left, strategy, eta);
293 if (left !== null) return { term: application(left.term, term.right), kind: left.kind };
294 if (strategy !== "CBN") {
295 const right = oneStep(term.right, strategy, eta);
296 if (right !== null) return { term: application(term.left, right.term), kind: right.kind };
297 }
298 if (
299 !early &&
300 term.left.kind === "abstraction" &&
301 (strategy !== "CBV" || term.right.kind === "variable" || term.right.kind === "abstraction")
302 ) {
303 return { term: substitute(term.left.body, term.left.name, term.right), kind: "beta" };
304 }
305 return null;
306}

308function abstractSki(name: string, term: Term): Term {
309 if (!free(term).has(name)) return application(constant("K"), term);
310 if (term.kind === "variable") return constant("I");
311 if (term.kind !== "application") throw new Error("Invalid SKI abstraction");
312 if (term.right.kind === "variable" && term.right.name === name && !free(term.left).has(name))
313 return term.left;
314 return application(
315 application(constant("S"), abstractSki(name, term.left)),
316 abstractSki(name, term.right),
317 );
318}
319function toSki(term: Term): Term {
320 if (term.kind === "variable") return term;
321 if (term.kind === "abstraction") return abstractSki(term.name, toSki(term.body));
322 if (term.kind !== "application") throw new Error("SKI conversion requires a pure term");
323 return application(toSki(term.left), toSki(term.right));
324}
325function skiStep(term: Term): Step | null {
326 let head = term;
327 const args: Term[] = [];
328 while (head.kind === "application") {
329 args.unshift(head.right);
330 head = head.left;
331 }
332 let result: Term | null = null,
333 used = 0;
334 if (head.kind === "constant") {
335 if (head.name === "I" && args.length >= 1) {
336 result = args[0];
337 used = 1;
338 } else if (head.name === "K" && args.length >= 2) {
339 result = args[0];
340 used = 2;
341 } else if (head.name === "S" && args.length >= 3) {
342 result = application(application(args[0], args[2]), application(args[1], args[2]));
343 used = 3;
344 }
345 }
346 if (result !== null) {
347 for (const argument of args.slice(used)) result = application(result, argument);
348 return { term: result, kind: head.kind === "constant" ? head.name : "" };
349 }
350 for (let i = 0; i < args.length; i++) {
351 const step = skiStep(args[i]);
352 if (step !== null) {
353 args[i] = step.term;
354 let rebuilt: Term = head;
355 for (const argument of args) rebuilt = application(rebuilt, argument);
356 return { term: rebuilt, kind: step.kind };
357 }
358 }
359 return null;
360}

363 const freeTypes = new Map<string, Type>();
364 let next = 0;
365 function fresh(): TypeVariable {

368 function arrow(left: Type, right: Type): Type {
369 return { kind: "arrow", left, right };
370 }
371 function prune(type: Type): Type {
372 if (type.kind === "variable" && type.link !== null) {
373 type.link = prune(type.link);
374 return type.link;
375 }
376 return type;
377 }
378 function occurs(variable: TypeVariable, type: Type): boolean {
379 type = prune(type);
380 return (
381 type === variable ||
382 (type.kind === "arrow" && (occurs(variable, type.left) || occurs(variable, type.right)))
383 );
384 }
385 function unify(a: Type, b: Type): void {
386 a = prune(a);
387 b = prune(b);
388 if (a === b) return;
389 if (a.kind === "variable") {
390 if (occurs(a, b)) throw new TypeFailure();
391 a.link = b;
392 } else if (b.kind === "variable") unify(b, a);
393 else {
394 unify(a.left, b.left);
395 unify(a.right, b.right);
396 }
397 }
398 function infer(node: Term, bound: Map<string, Type>): Type {
399 if (node.kind === "variable") {
400 const known = bound.get(node.name);
401 if (known !== undefined) return known;
402 if (!freeTypes.has(node.name)) freeTypes.set(node.name, fresh());
403 return freeTypes.get(node.name)!;
404 }
405 if (node.kind === "abstraction") {
406 const type = fresh(),
407 environment = new Map(bound);
408 environment.set(node.name, type);
409 return arrow(type, infer(node.body, environment));
410 }
411 if (node.kind !== "application") throw new Error("Type inference requires a pure term");
412 const left = infer(node.left, bound),
413 right = infer(node.right, bound),
414 result = fresh();
415 unify(left, arrow(right, result));
416 return result;
417 }
418 try {
419 const type = infer(term, new Map()),
420 names = new Map<TypeVariable, string>();
421 function show(type: Type): string {
422 type = prune(type);
423 if (type.kind === "variable") {
424 if (!names.has(type)) names.set(type, generatedName(names.size));
425 return names.get(type)!;
426 }
427 const left = show(type.left);
428 return (
429 (prune(type.left).kind === "arrow" ? "(" + left + ")" : left) +
430 " -> " +
431 show(type.right)
432 );
433 }
434 return show(type);
435 } catch (error) {
436 if (!(error instanceof TypeFailure)) throw error;
437 return "not typable";
438 }

441function readChurch(term: Term): string {
442 if (term.kind === "abstraction" && term.body.kind === "abstraction") {
443 const outer = term.name,
444 inner = term.body.name;
445 let body = term.body.body,
446 count = 0;
447 while (
448 body.kind === "application" &&
449 body.left.kind === "variable" &&
450 body.left.name === outer
451 ) {
452 count++;
453 body = body.right;
454 }
455 if (body.kind === "variable" && body.name === inner) return String(count);
456 if (term.body.body.kind === "variable" && term.body.body.name === outer) return "true";
457 }
458 if (
459 term.kind === "abstraction" &&
460 term.body.kind === "application" &&
461 term.body.left.kind === "application" &&
462 term.body.left.left.kind === "variable" &&
463 term.body.left.left.name === term.name
464 ) {
465 const left = term.body.left.right,
466 right = term.body.right;
467 if (!free(left).has(term.name) && !free(right).has(term.name))
468 return "(" + readChurch(left) + ", " + readChurch(right) + ")";
469 }
470 return canonical(term);
471}

475 readonly output: string[] = [];

477 eta = false;

481 ["S", "\\x y z. x z (y z)"],

491 expand(term: Term): Term {

515 reduction(term: Term, strategy: string, eta: boolean, show = false, ski = false): Reduction {
516 let count = 0;
517 while (true) {
518 const step = ski ? skiStep(term) : oneStep(term, strategy, eta);
519 if (step === null) return { term, count, success: true };
520 if (count === this.limit) return { term, count, success: false };

522 count++;

527 kam(term: Term): void {
528 let code = toDB(term),
529 environment: Closure[] = [],
530 count = 0;
531 const stack: Closure[] = [];
532 while (true) {
533 if (
534 code.kind !== "application" &&
535 code.kind !== "index" &&
536 !(code.kind === "abstraction" && stack.length > 0)
537 )
538 break;
539 if (count === this.limit) {
540 this.output.push(`no result within ${this.limit} transitions`);
541 return;
542 }
543 let kind: string;
544 if (code.kind === "application") {
545 kind = "push";
546 stack.push({ code: code.right, environment });
547 code = code.left;
548 } else if (code.kind === "abstraction") {
549 kind = "grab";
550 environment = [stack.pop()!, ...environment];
551 code = code.body;
552 } else if (code.kind === "index") {
553 kind = "access";
554 const closure = environment[code.value];
555 code = closure.code;
556 environment = closure.environment;
557 } else throw new Error("Invalid machine transition");
558 count++;
559 if (this.display)
560 this.output.push(
561 `step ${count} (${kind}): ${printDB(code)} [env ${environment.length}, stack ${stack.length}]`,
562 );
563 }
564 function restore(code: DBTerm, environment: Closure[], depth = 0): DBTerm {
565 while (code.kind === "index" && code.value >= depth) {
566 const closure = environment[code.value - depth];
567 code = closure.code;
568 environment = closure.environment;
569 depth = 0;
570 }
571 if (code.kind === "abstraction")
572 return { kind: "abstraction", body: restore(code.body, environment, depth + 1) };
573 if (code.kind === "application")
574 return {
575 kind: "application",
576 left: restore(code.left, environment, depth),
577 right: restore(code.right, environment, depth),
578 };
579 return code;
580 }
581 let result = restore(code, environment);
582 for (const closure of stack.reverse())
583 result = {
584 kind: "application",
585 left: result,
586 right: restore(closure.code, closure.environment),
587 };
588 this.output.push(`result: ${print(fromDB(result))} (transitions: ${count})`);
589 }
590 command(line: string): void {

592 const command = position < 0 ? line : line.slice(0, position);
593 const argument = position < 0 ? "" : line.slice(position + 1);
594 const commands = [
595 "let",
596 "unlet",
597 "macros",
598 "steps",
599 "eta",
600 "limit",
601 "expand",
602 "subst",
603 "reduce",
604 "equal",
605 "alpha",
606 "debruijn",
607 "fv",
608 "size",
609 "depth",
610 "type",
611 "church",
612 "ski",
613 "skireduce",
614 "kam",
615 ];

617 const reserved = ["S", "K", "I", "NORMAL", "APPLICATIVE", "CBN", "CBV"];

641 const name = words[0];

645 references(this.macros.get(other)!).includes(name),
646 );

654 lex(argument, position < 0 ? line.length + 1 : position + 2, line.length + 1),

658 terms: Term[];

660 name = parser.expect(command === "let" ? "macro" : "variable").text;

669 parser.expect(";");

680 parser.end();
681 if (command === "let") {
682 const term = terms[0];
683 if (reserved.includes(name)) fail("reserved name " + name);
684 for (const reference of references(term)) {
685 if (reference === name || this.depends(reference, name))
686 fail("recursive definition of " + name);
687 if (!this.macros.has(reference)) fail("unbound macro " + reference);

690 if (variables.length > 0)
691 fail("free variable " + variables[0] + " in definition of " + name);
692 this.output.push((this.macros.has(name) ? "redefined " : "defined ") + name);
693 this.macros.set(name, term);
694 return;
695 }
696 terms = terms.map((term) => this.expand(term));
697 const term = terms[0];
698 if (command === "expand") this.output.push(print(term));
699 else if (command === "subst") this.output.push(print(substitute(terms[1], name, term)));
700 else if (command === "fv") this.output.push([...free(term)].sort().join(" ") || "(none)");
701 else if (command === "debruijn") this.output.push(printDB(toDB(term)));
702 else if (command === "alpha")
703 this.output.push(printDB(toDB(term)) === printDB(toDB(terms[1])) ? "yes" : "no");
704 else if (command === "size" || command === "depth") {
705 function measure(node: Term): number {
706 if (node.kind === "variable") return 1;
707 if (node.kind === "abstraction") return 1 + measure(node.body);
708 if (node.kind !== "application") throw new Error("Size requires a pure term");
709 return (
710 1 +
711 (command === "size"
712 ? measure(node.left) + measure(node.right)
713 : Math.max(measure(node.left), measure(node.right)))
714 );
715 }
716 this.output.push(String(measure(term)));
717 } else if (command === "type") this.output.push(principalType(term));
718 else if (command === "ski") this.output.push(print(toSki(term)));
719 else if (command === "kam") this.kam(term);
720 else if (command === "equal") {
721 const a = this.reduction(term, "NORMAL", this.eta),
722 b = this.reduction(terms[1], "NORMAL", this.eta);
723 this.output.push(
724 !a.success || !b.success
725 ? "unknown"
726 : printDB(toDB(a.term)) === printDB(toDB(b.term))
727 ? "yes"
728 : "no",
729 );
730 } else {
731 const ski = command === "skireduce";
732 const result = this.reduction(
733 ski ? toSki(term) : term,
734 strategy,
735 command === "church" ? false : this.eta,
736 command === "reduce" || ski,
737 ski,
738 );
739 this.output.push(
740 !result.success
741 ? `no normal form within ${this.limit} steps`
742 : command === "church"
743 ? readChurch(fromDB(toDB(result.term)))
744 : `result: ${ski ? print(result.term) : canonical(result.term)} (steps: ${result.count})`,
745 );
746 }
