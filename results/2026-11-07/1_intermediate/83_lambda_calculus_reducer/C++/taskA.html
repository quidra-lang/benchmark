31set<string> free_variables(T term) {
32 if (term->kind == "var") return {term->name};
33 if (term->kind == "abs") {
34 auto result = free_variables(term->left);
35 result.erase(term->name);
36 return result;
37 }
38 if (term->kind == "app") {
39 auto result = free_variables(term->left), argument = free_variables(term->right);
40 result.insert(argument.begin(), argument.end());
41 return result;
42 }
43 return {};
44}
45string printed(T term, bool end = true, bool debruijn = false) {
46 if (term->kind == "var" || term->kind == "macro" || term->kind == "constant") return term->name;
47 if (term->kind == "numeral") return "#" + to_string(term->number);
48 if (term->kind == "index") return to_string(term->number);
49 if (term->kind == "abs") {
50 string prefix = "\\";
51 if (debruijn) {
52 T body = term;
53 prefix.clear();
54 while (body->kind == "abs") {
55 prefix += '\\';
56 body = body->left;
57 }
58 prefix += printed(body, true, true);
59 } else {
60 T body = term;
61 bool first = true;
62 while (body->kind == "abs") {
63 if (!first) prefix += ' ';
64 prefix += body->name;
65 body = body->left;
66 first = false;
67 }
68 prefix += ". " + printed(body, true, false);
69 }
70 return end ? prefix : "(" + prefix + ")";
71 }
72 string left = term->left->kind == "abs" ? "(" + printed(term->left, true, debruijn) + ")" : printed(term->left, false, debruijn);
73 string right = term->right->kind == "app" ? "(" + printed(term->right, true, debruijn) + ")" : printed(term->right, end, debruijn);
74 return left + " " + right;
75}
76string sequence_name(size_t index) {
77 return string(1, char('a' + index % 26)) + (index >= 26 ? to_string(index / 26) : "");
78}
79T canonical_term(T term) {
80 auto free = free_variables(term);
81 size_t next = 0;
82 vector<pair<string, string>> bindings;
83 auto rename = [&](auto&& self, T value) -> T {
84 if (value->kind == "var") {
85 for (auto it = bindings.rbegin(); it != bindings.rend(); ++it) if (it->first == value->name) return leaf("var", it->second);
86 return value;
87 }
88 if (value->kind == "abs") {
89 string name;
90 do {
91 name = sequence_name(next++);
92 }
93 while (free.count(name));
94 bindings.push_back({value->name, name});
95 T body = self(self, value->left);
96 bindings.pop_back();
97 return abstraction(name, body);
98 }
99 if (value->kind == "app") {
100 T left = self(self, value->left);
101 T right = self(self, value->right);
102 return application(left, right);
103 }
104 return value;
105 };
106 return rename(rename, term);
107}
108string canonical(T term) {
109 return printed(canonical_term(term));
110}
111T debruijn_term(T term, vector<string>& bindings) {
112 if (term->kind == "var") {
113 for (size_t i = bindings.size(); i > 0; --i) if (bindings[i - 1] == term->name) return leaf("index", "", bindings.size() - i);
114 return term;
115 }
116 if (term->kind == "abs") {
117 bindings.push_back(term->name);
118 T body = debruijn_term(term->left, bindings);
119 bindings.pop_back();
120 return abstraction("", body);
121 }
122 T left = debruijn_term(term->left, bindings), right = debruijn_term(term->right, bindings);
123 return application(left, right);
124}
125T debruijn_term(T term) {
126 vector<string> bindings;
127 return debruijn_term(term, bindings);
128}
129bool alpha_equal(T a, T b) {
130 return printed(debruijn_term(a), true, true) == printed(debruijn_term(b), true, true);
131}
132T substitute(T term, const string& variable, T replacement) {
133 if (term->kind == "var") return term->name == variable ? replacement : term;
134 if (term->kind == "app") {
135 T left = substitute(term->left, variable, replacement), right = substitute(term->right, variable, replacement);
136 return application(left, right);
137 }
138 auto body_free = free_variables(term->left);
139 if (term->name == variable || !body_free.count(variable)) return term;
140 auto replacement_free = free_variables(replacement);
141 if (!replacement_free.count(term->name)) return abstraction(term->name, substitute(term->left, variable, replacement));
142 string base = term->name;
143 while (!base.empty() && base.back() >= '0' && base.back() <= '9') base.pop_back();
144 string fresh;
145 for (size_t i = 1;; ++i) {
146 fresh = base + to_string(i);
147 if (!replacement_free.count(fresh) && !body_free.count(fresh)) break;
148 }
149 T renamed = substitute(term->left, term->name, leaf("var", fresh));
150 return abstraction(fresh, substitute(renamed, variable, replacement));
151}
152struct Token {
153 string kind, text;
154 int column;
155};
156[[noreturn]] void failure(const string& message) {
157 throw runtime_error(message);
158}
159vector<Token> lexical(const string& line, size_t start) {
160 vector<Token> tokens;
161 for (size_t i = start; i < line.size();) {
162 char ch = line[i];
163 if (ch == ' ') {
164 ++i;
165 continue;
166 }
167 size_t first = i++;
168 string kind = "symbol";
169 if (ch >= 'a' && ch <= 'z') {
170 kind = "var";
171 while (i < line.size() && ((line[i] >= 'a' && line[i] <= 'z') || (line[i] >= '0' && line[i] <= '9'))) ++i;
172 } else if (ch >= 'A' && ch <= 'Z') {
173 kind = "macro";
174 while (i < line.size() && ((line[i] >= 'A' && line[i] <= 'Z') || (line[i] >= '0' && line[i] <= '9') || line[i] == '_')) ++i;
175 } else if (ch == '#' && i < line.size() && line[i] >= '0' && line[i] <= '9') {
176 kind = "numeral";
177 while (i < line.size() && line[i] >= '0' && line[i] <= '9') ++i;
178 } else if (string("\\.()=;").find(ch) == string::npos)
179 failure("column " + to_string(first + 1) + ": unexpected character '" + string(1, ch) + "'");
180 tokens.push_back({kind, line.substr(first, i - first), int(first + 1)});
181 }
182 tokens.push_back({"end", "", int(line.size() + 1)});
183 return tokens;
184}
185class Parser {
186 vector<Token> tokens;
187 size_t position = 0;
188public:
189 explicit Parser(vector<Token> tokens) : tokens(move(tokens)) {
190 }
191 const Token& next() const {
192 return tokens[position];
193 }
194 Token consume() {
195 return tokens[position++];
196 }
197 string expect(const string& kind, const string& notation) {
198 const auto& token = next();
199 if ((kind == "var" || kind == "macro") ? token.kind != kind : token.text != kind)
200 failure("column " + to_string(token.column) + ": expected " + notation);
201 return consume().text;
202 }
203 T term() {
204 vector<T> elements;
205 while (next().kind == "var" || next().kind == "macro" || next().kind == "numeral" || next().text == "(") {
206 Token token = consume();
207 if (token.text == "(") {
208 elements.push_back(term());
209 expect(")", "')'");
210 }
211 else elements.push_back(leaf(token.kind, token.kind == "numeral" ? "" : token.text,
212 token.kind == "numeral" ? stoi(token.text.substr(1)) : 0));
213 }
214 if (next().text == "\\") {
215 consume();
216 vector<string> bindings{expect("var", "variable")};
217 while (next().kind == "var") bindings.push_back(consume().text);
218 expect(".", "'.'");
219 T body = term();
220 for (auto it = bindings.rbegin(); it != bindings.rend(); ++it) body = abstraction(*it, body);
221 elements.push_back(body);
222 }
223 if (elements.empty()) failure("column " + to_string(next().column) + ": expected term");
224 T result = elements[0];
225 for (size_t i = 1; i < elements.size(); ++i) result = application(result, elements[i]);
226 return result;
227 }
228 void finish() const {
229 if (next().kind != "end") failure("column " + to_string(next().column) + ": unexpected '" + next().text + "'");
230 }

232struct Step {
233 T term;
234 string kind;
235};
236optional<Step> one_step(T term, const string& strategy, bool eta) {
237 bool normal = strategy == "NORMAL", applicative = strategy == "APPLICATIVE";
238 if (term->kind == "var") return nullopt;
239 if (term->kind == "abs") {
240 if (!normal && !applicative) return nullopt;
241 bool eta_redex = eta && term->left->kind == "app" && term->left->right->kind == "var" &&
242 term->left->right->name == term->name && !free_variables(term->left->left).count(term->name);
243 if (normal && eta_redex) return Step{term->left->left, "eta"};
244 if (auto step = one_step(term->left, strategy, eta)) return Step{abstraction(term->name, step->term), step->kind};
245 if (applicative && eta_redex) return Step{term->left->left, "eta"};
246 return nullopt;
247 }
248 T function = term->left, argument = term->right;
249 if ((normal || strategy == "CBN") && function->kind == "abs")
250 return Step{substitute(function->left, function->name, argument), "beta"};
251 if (auto step = one_step(function, strategy, eta)) return Step{application(step->term, argument), step->kind};
252 if (strategy != "CBN") if (auto step = one_step(argument, strategy, eta)) return Step{application(function, step->term), step->kind};
253 if ((applicative || strategy == "CBV") && function->kind == "abs" &&
254 (applicative || argument->kind == "var" || argument->kind == "abs"))
255 return Step{substitute(function->left, function->name, argument), "beta"};
256 return nullopt;
257}
258T abstract_ski(const string& variable, T term) {
259 if (!free_variables(term).count(variable)) return application(leaf("constant", "K"), term);
260 if (term->kind == "var") return leaf("constant", "I");
261 if (term->right->kind == "var" && term->right->name == variable && !free_variables(term->left).count(variable)) return term->left;
262 T left = abstract_ski(variable, term->left), right = abstract_ski(variable, term->right);
263 return application(application(leaf("constant", "S"), left), right);
264}
265T to_ski(T term) {
266 if (term->kind == "var") return term;
267 if (term->kind == "abs") return abstract_ski(term->name, to_ski(term->left));
268 T left = to_ski(term->left), right = to_ski(term->right);
269 return application(left, right);
270}
271optional<Step> ski_step(T term) {
272 vector<T> arguments;
273 T head = term;
274 while (head->kind == "app") {
275 arguments.push_back(head->right);
276 head = head->left;
277 }
278 reverse(arguments.begin(), arguments.end());
279 T result;
280 size_t used = 0;
281 if (head->kind == "constant") {
282 if (head->name == "I" && arguments.size() >= 1) {
283 result = arguments[0];
284 used = 1;
285 }
286 else if (head->name == "K" && arguments.size() >= 2) {
287 result = arguments[0];
288 used = 2;
289 }
290 else if (head->name == "S" && arguments.size() >= 3) {
291 result = application(application(arguments[0], arguments[2]), application(arguments[1], arguments[2]));
292 used = 3;
293 }
294 }
295 if (result) {
296 for (size_t i = used; i < arguments.size(); ++i) result = application(result, arguments[i]);
297 return Step{result, head->name};
298 }
299 for (size_t i = 0; i < arguments.size(); ++i) if (auto step = ski_step(arguments[i])) {
300 result = head;
301 for (size_t j = 0; j < arguments.size(); ++j) result = application(result, j == i ? step->term : arguments[j]);
302 return Step{result, step->kind};
303 }
304 return nullopt;
305}

314class Inference {
315 int counter = 0;
316 map<string, TypePtr> free;
317 vector<pair<string, TypePtr>> bindings;
318 TypePtr variable() {
319 return make_shared<Type>(Type{counter++, nullptr, nullptr, nullptr});
320 }
321 static TypePtr arrow(TypePtr a, TypePtr b) {
322 return make_shared<Type>(Type{-1, a, b, nullptr});
323 }
324 static TypePtr prune(TypePtr type) {
325 if (type->binding) type->binding = prune(type->binding);
326 return type->binding ? type->binding : type;
327 }
328 static bool occurs(int id, TypePtr type) {
329 type = prune(type);
330 if (type->id >= 0) return type->id == id;
331 return occurs(id, type->left) || occurs(id, type->right);
332 }
333 static void unify(TypePtr a, TypePtr b) {
334 a = prune(a);
335 b = prune(b);
336 if (a == b) return;
337 if (a->id >= 0) {
338 if (occurs(a->id, b)) throw NotTypable{};
339 a->binding = b;
340 } else if (b->id >= 0) unify(b, a);
341 else {
342 unify(a->left, b->left);
343 unify(a->right, b->right);
344 }
345 }
346 TypePtr infer(T term) {
347 if (term->kind == "var") {
348 for (auto it = bindings.rbegin(); it != bindings.rend(); ++it) if (it->first == term->name) return it->second;
349 if (!free.count(term->name)) free[term->name] = variable();
350 return free.at(term->name);
351 }
352 if (term->kind == "abs") {
353 TypePtr input = variable();
354 bindings.push_back({term->name, input});
355 TypePtr output = infer(term->left);
356 bindings.pop_back();
357 return arrow(input, output);
358 }
359 TypePtr function = infer(term->left), argument = infer(term->right), result = variable();
360 unify(function, arrow(argument, result));
361 return result;
362 }
363 static string show(TypePtr type, map<int, string>& names) {
364 type = prune(type);
365 if (type->id >= 0) {
366 if (!names.count(type->id)) names[type->id] = sequence_name(names.size());
367 return names.at(type->id);
368 }
369 string left = show(type->left, names);
370 if (prune(type->left)->id < 0) left = "(" + left + ")";
371 string right = show(type->right, names);
372 return left + " -> " + right;
373 }
374public:
375 string principal(T term) {
376 try {
377 TypePtr result = infer(term);
378 map<int, string> names;
379 return show(result, names);
380 }
381 catch (const NotTypable&) {
382 return "not typable";
383 }
384 }

386struct Closure;
387using Environment = shared_ptr<const vector<Closure>>;
388struct Closure {
389 T code;
390 Environment environment;
391};
392T reify(T code, Environment environment, int depth = 0) {
393 while (code->kind == "index" && code->number >= depth) {
394 Closure closure = (*environment)[code->number - depth];
395 code = closure.code;
396 environment = closure.environment;
397 depth = 0;
398 }
399 if (code->kind == "index") return code;
400 if (code->kind == "var") return code;
401 if (code->kind == "abs") return abstraction("", reify(code->left, environment, depth + 1));
402 T left = reify(code->left, environment, depth), right = reify(code->right, environment, depth);
403 return application(left, right);
404}
405T named_from_debruijn(T term) {
406 auto free = free_variables(term);
407 size_t next = 0;
408 vector<string> bindings;
409 auto convert = [&](auto&& self, T code) -> T {
410 if (code->kind == "var") return code;
411 if (code->kind == "index") return leaf("var", bindings[bindings.size() - 1 - code->number]);
412 if (code->kind == "abs") {
413 string name;
414 do {
415 name = sequence_name(next++);
416 }
417 while (free.count(name));
418 bindings.push_back(name);
419 T body = self(self, code->left);
420 bindings.pop_back();
421 return abstraction(name, body);
422 }
423 T left = self(self, code->left), right = self(self, code->right);
424 return application(left, right);
425 };
426 return convert(convert, term);
427}

429class Workbench {
430 map<string, T> macros;
431 bool display = false, eta = false;
432 int limit = 10000;
433 const set<string> reserved = {"S", "K", "I", "NORMAL", "APPLICATIVE", "CBN", "CBV"};
434 static vector<string> references(T term) {
435 if (term->kind == "macro") return {term->name};
436 if (term->kind == "abs") return references(term->left);
437 if (term->kind == "app") {
438 auto result = references(term->left), right = references(term->right);
439 result.insert(result.end(), right.begin(), right.end());
440 return result;
441 }
442 return {};
443 }
444 bool depends(const string& macro, const string& target, set<string>& visited) const {
445 if (!macros.count(macro) || reserved.count(macro) || !visited.insert(macro).second) return false;
446 for (const auto& name : references(macros.at(macro))) if (name == target || depends(name, target, visited)) return true;
447 return false;
448 }
449 T expand(T term) const {
450 if (term->kind == "macro") {
451 auto found = macros.find(term->name);
452 if (found == macros.end()) failure("unbound macro " + term->name);
453 return expand(found->second);
454 }
455 if (term->kind == "numeral") {
456 T body = leaf("var", "x");
457 for (int i = 0; i < term->number; ++i) body = application(leaf("var", "f"), body);
458 return abstraction("f", abstraction("x", body));
459 }
460 if (term->kind == "var") return term;
461 if (term->kind == "abs") return abstraction(term->name, expand(term->left));
462 T left = expand(term->left), right = expand(term->right);
463 return application(left, right);
464 }
465 struct Reduction {
466 T term;
467 int count;
468 bool success;
469 };
470 Reduction reduce(T term, const string& strategy, bool use_eta, bool show, bool ski = false) const {
471 for (int count = 0;; ++count) {
472 auto next = ski ? ski_step(term) : one_step(term, strategy, use_eta);
473 if (!next) return {term, count, true};
474 if (count == limit) return {term, count, false};
475 term = next->term;
476 if (show) cout << "step " << count + 1 << " (" << next->kind << "): " << printed(term) << '\n';
477 }

479 string church_value(T term) const {
480 if (term->kind == "abs" && term->left->kind == "abs") {
481 T body = term->left->left;
482 int number = 0;
483 while (body->kind == "app" && body->left->kind == "var" && body->left->name == term->name) {
484 ++number;
485 body = body->right;
486 }
487 if (body->kind == "var" && body->name == term->left->name) return to_string(number);
488 if (term->left->left->kind == "var" && term->left->left->name == term->name) return "true";
489 }
490 if (term->kind == "abs" && term->left->kind == "app" && term->left->left->kind == "app") {
491 T head = term->left->left->left, first = term->left->left->right, second = term->left->right;
492 if (head->kind == "var" && head->name == term->name && !free_variables(first).count(term->name) && !free_variables(second).count(term->name)) {
493 string left = church_value(first), right = church_value(second);
494 return "(" + left + ", " + right + ")";
495 }
496 }
497 return canonical(term);

499 void kam(T term) const {
500 T code = debruijn_term(term);
501 Environment environment = make_shared<vector<Closure>>();
502 vector<Closure> stack;
503 int count = 0;
504 while (code->kind == "app" || code->kind == "index" || (code->kind == "abs" && !stack.empty())) {
505 if (count == limit) {
506 cout << "no result within " << limit << " transitions\n";
507 return;
508 }
509 string kind;
510 if (code->kind == "app") {
511 kind = "push";
512 stack.push_back({code->right, environment});
513 code = code->left;
514 }
515 else if (code->kind == "abs") {
516 kind = "grab";
517 auto extended = make_shared<vector<Closure>>(*environment);
518 extended->insert(extended->begin(), stack.back());
519 stack.pop_back();
520 environment = extended;
521 code = code->left;
522 } else {
523 kind = "access";
524 Closure closure = (*environment)[code->number];
525 code = closure.code;
526 environment = closure.environment;
527 }
528 ++count;
529 if (display) cout << "step " << count << " (" << kind << "): " << printed(code, true, true)
530 << " [env " << environment->size() << ", stack " << stack.size() << "]\n";
531 }
532 T result = reify(code, environment);
533 for (auto it = stack.rbegin(); it != stack.rend(); ++it) result = application(result, reify(it->code, it->environment));
534 cout << "result: " << canonical(named_from_debruijn(result)) << " (transitions: " << count << ")\n";
535 }

537 Workbench() {
538 for (const auto& [name, body] : vector<pair<string, string>>{{"S", "\\x y z. x z (y z)"}, {"K", "\\x y. x"}, {"I", "\\x. x"}})

541 void command(const string& line) {
542 size_t space = line.find(' ');
543 string command = line.substr(0, space);
544 const set<string> commands = {"let", "unlet", "macros", "steps", "eta", "limit", "expand", "subst", "reduce", "equal", "alpha", "debruijn", "fv", "size", "depth", "type", "church", "ski", "skireduce", "kam"};
545 if (!commands.count(command)) failure("unknown command " + command);
546 string argument = space == string::npos ? "" : line.substr(space + 1);
547 if (command == "unlet" || command == "macros" || command == "steps" || command == "eta" || command == "limit") {
548 istringstream input(argument);
549 vector<string> args;
550 string word;
551 while (input >> word) args.push_back(word);
552 if (command == "macros") {
553 if (!args.empty()) failure("bad argument");
554 bool any = false;
555 for (const auto& [name, body] : macros) if (!reserved.count(name)) {
556 cout << name << " = " << printed(body) << '\n';
557 any = true;
558 }
559 if (!any) cout << "(none)\n";
560 } else if (command == "steps" || command == "eta") {
561 if (args.size() != 1 || (args[0] != "on" && args[0] != "off")) failure("bad argument");
562 (command == "steps" ? display : eta) = args[0] == "on";
563 } else if (command == "limit") {
564 if (args.size() != 1 || args[0].empty() || args[0].size() > 5 || !all_of(args[0].begin(), args[0].end(), [](char c) {
565 return c >= '0' && c <= '9';
566 })) failure("bad argument");
567 int value = stoi(args[0]);
568 if (value < 1 || value > 10000) failure("bad argument");
569 limit = value;
570 } else {
571 if (args.size() != 1 || args[0].empty() || args[0][0] < 'A' || args[0][0] > 'Z' ||
572 !all_of(args[0].begin(), args[0].end(), [](char c) {
573 return (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9') || c == '_';
574 })) failure("bad argument");
575 string name = args[0];
576 if (reserved.count(name)) failure("reserved name " + name);
577 if (!macros.count(name)) failure("unbound macro " + name);
578 for (const auto& [user, body] : macros) if (!reserved.count(user)) {
579 auto refs = references(body);
580 if (find(refs.begin(), refs.end(), name) != refs.end()) failure(name + " is used by " + user);
581 }
582 macros.erase(name);
583 cout << "removed " << name << '\n';
584 }
585 return;
586 }
587 Parser parser(lexical(line, space == string::npos ? line.size() : space + 1));
588 string name, strategy = "NORMAL";
589 vector<T> terms;
590 if (command == "let") {
591 name = parser.expect("macro", "macro name");
592 parser.expect("=", "'='");
593 terms.push_back(parser.term());
594 }
595 else if (command == "subst") {
596 name = parser.expect("var", "variable");
597 parser.expect("=", "'='");
598 terms.push_back(parser.term());
599 parser.expect(";", "';'");
600 terms.push_back(parser.term());
601 } else if (command == "equal" || command == "alpha") {
602 terms.push_back(parser.term());
603 parser.expect(";", "';'");
604 terms.push_back(parser.term());
605 } else {
606 if (command == "reduce" && parser.next().kind == "macro" &&
607 (parser.next().text == "NORMAL" || parser.next().text == "APPLICATIVE" || parser.next().text == "CBN" || parser.next().text == "CBV")) strategy = parser.consume().text;
608 terms.push_back(parser.term());
609 }
610 parser.finish();
611 if (command == "let") {
612 if (reserved.count(name)) failure("reserved name " + name);
613 for (const auto& reference : references(terms[0])) {
614 set<string> visited;
615 if (reference == name || depends(reference, name, visited)) failure("recursive definition of " + name);
616 if (!macros.count(reference)) failure("unbound macro " + reference);
617 }
618 auto free = free_variables(terms[0]);
619 if (!free.empty()) failure("free variable " + *free.begin() + " in definition of " + name);
620 bool existed = macros.count(name);
621 macros[name] = terms[0];
622 cout << (existed ? "redefined " : "defined ") << name << '\n';
623 return;
624 }
625 for (T& term : terms) term = expand(term);
626 T term = terms[0];
627 if (command == "expand") cout << printed(term) << '\n';
628 else if (command == "subst") cout << printed(substitute(terms[1], name, term)) << '\n';
629 else if (command == "alpha") cout << (alpha_equal(term, terms[1]) ? "yes" : "no") << '\n';
630 else if (command == "equal") {
631 auto first = reduce(term, "NORMAL", eta, false), second = reduce(terms[1], "NORMAL", eta, false);
632 cout << (!first.success || !second.success ? "unknown" : alpha_equal(first.term, second.term) ? "yes" : "no") << '\n';
633 } else if (command == "debruijn") cout << printed(debruijn_term(term), true, true) << '\n';
634 else if (command == "fv") {
635 auto free = free_variables(term);
636 bool first = true;
637 for (const auto& variable : free) {
638 cout << (first ? "" : " ") << variable;
639 first = false;
640 }
641 if (free.empty()) cout << "(none)";
642 cout << '\n';
643 } else if (command == "size" || command == "depth") {
644 auto measure = [&](auto&& self, T value) -> int {
645 if (value->kind == "var") return 1;
646 int left = self(self, value->left);
647 if (value->kind == "abs") return 1 + left;
648 int right = self(self, value->right);
649 return 1 + (command == "size" ? left + right : max(left, right));
650 };
651 cout << measure(measure, term) << '\n';
652 } else if (command == "type") cout << Inference().principal(term) << '\n';
653 else if (command == "ski") cout << printed(to_ski(term)) << '\n';
654 else if (command == "kam") kam(term);
655 else {
656 bool ski = command == "skireduce";
657 if (ski) term = to_ski(term);
658 auto result = reduce(term, strategy, command == "church" ? false : eta, display && command != "church", ski);
659 if (!result.success) cout << "no normal form within " << limit << " steps\n";
660 else if (command == "church") cout << church_value(canonical_term(result.term)) << '\n';
661 else cout << "result: " << (ski ? printed(result.term) : canonical(result.term)) << " (steps: " << result.count << ")\n";
662 }
663 }
664};
