32 class Parser
33 {
34 public List<Token> Tokens = new();
35 public int At;
36 public Parser(string line, int start)
37 {
38 for (int i = start; i < line.Length;)
39 {
40 char c = line[i];
41 if (c == ' ')
42 {
43 i++;
44 continue;
45 }
46 int first = i;
47 string kind;
48 if ("\\.()=;".Contains(c))
49 {
50 kind = c.ToString();
51 i++;
52 }
53 else if (Lower(c))
54 {
55 kind = "variable";
56 while (i < line.Length && (Lower(line[i]) || Digit(line[i]))) i++;
57 }
58 else if (Upper(c))
59 {
60 kind = "macro";
61 while (i < line.Length && (Upper(line[i]) || Digit(line[i]) || line[i] == '_')) i++;
62 }
63 else if (c == '#' && i + 1 < line.Length && Digit(line[i + 1]))
64 {
65 kind = "numeral";
66 i++;
67 while (i < line.Length && Digit(line[i])) i++;
68 }
69 else
70 {
71 Fail($"column {i + 1}: unexpected character '{c}'");
72 return;
73 }
74 Tokens.Add(new Token(kind, line[first..i], first + 1));
75 }
76 Tokens.Add(new Token("end", "", line.Length + 1));
77 }
78 public Token Peek() => Tokens[At];
79 public Token Expect(string kind)
80 {
81 Token token = Peek();
82 if (token.Kind != kind) Fail($"column {token.Column}: expected " + (kind == "macro" ? "macro name" : kind == "variable" ? "variable" : $"'{kind}'"));
83 At++;
84 return token;
85 }
86 public Term ReadTerm()
87 {
88 List<Term> elements = new();
89 while (Peek().Kind is "variable" or "macro" or "numeral" or "(") elements.Add(Atom());
90 if (Peek().Kind == "\\")
91 {
92 At++;
93 List<string> names = new() { Expect("variable").Text };
94 while (Peek().Kind == "variable") names.Add(Tokens[At++].Text);
95 Expect(".");
96 Term body = ReadTerm();
97 for (int i = names.Count - 1; i >= 0; i--) body = Abstraction(names[i], body);
98 elements.Add(body);
99 }
100 if (elements.Count == 0) Fail($"column {Peek().Column}: expected term");
101 Term result = elements[0];
102 for (int i = 1; i < elements.Count; i++) result = Application(result, elements[i]);
103 return result;
104 }
105 Term Atom()
106 {
107 Token token = Tokens[At++];
108 if (token.Kind == "variable") return Variable(token.Text);
109 if (token.Kind == "macro") return new('m', token.Text);
110 if (token.Kind == "numeral") return new('n', Number: int.Parse(token.Text[1..]));
111 Term body = ReadTerm();
112 Expect(")");
113 return body;
114 }
115 public void End()
116 {
117 if (Peek().Kind != "end") Fail($"column {Peek().Column}: unexpected '{Peek().Text}'");
118 }
119 }

121 static string Print(Term term, bool end = true)
122 {
123 if (term.Kind == 'n') return "#" + term.Number;
124 if (term.Kind == 'd') return term.Number.ToString(CultureInfo.InvariantCulture);
125 if (term.Kind != 'a' && term.Kind != 'p') return term.Name;
126 if (term.Kind == 'a')
127 {
128 List<string> names = new();
129 Term body = term;
130 while (body.Kind == 'a')
131 {
132 names.Add(body.Name);
133 body = body.Left!;
134 }
135 string text = term.Name.Length == 0 ? new string('\\', names.Count) + Print(body) : "\\" + string.Join(" ", names) + ". " + Print(body);
136 return end ? text : "(" + text + ")";
137 }
138 string left = term.Left!.Kind == 'a' ? "(" + Print(term.Left) + ")" : Print(term.Left, false);
139 string right = term.Right!.Kind == 'p' ? "(" + Print(term.Right) + ")" : Print(term.Right, end);
140 return left + " " + right;
141 }

143 static SortedSet<string> Free(Term term)
144 {
145 SortedSet<string> result = new(StringComparer.Ordinal);
146 if (term.Kind == 'v') result.Add(term.Name);
147 else if (term.Kind == 'a')
148 {
149 result.UnionWith(Free(term.Left!));
150 result.Remove(term.Name);
151 }
152 else if (term.Kind == 'p')
153 {
154 result.UnionWith(Free(term.Left!));
155 result.UnionWith(Free(term.Right!));
156 }
157 return result;
158 }

160 static void References(Term term, List<string> result)
161 {
162 if (term.Kind == 'm') result.Add(term.Name);
163 else
164 {
165 if (term.Left != null) References(term.Left, result);
166 if (term.Right != null) References(term.Right, result);
167 }

170 static Term Substitute(Term term, string name, Term value)
171 {
172 if (term.Kind == 'v') return term.Name == name ? value : term;
173 if (term.Kind == 'p') return Application(Substitute(term.Left!, name, value), Substitute(term.Right!, name, value));
174 string bound = term.Name;
175 var bodyFree = Free(term.Left!);
176 if (bound == name || !bodyFree.Contains(name)) return term;
177 var valueFree = Free(value);
178 if (!valueFree.Contains(bound)) return Abstraction(bound, Substitute(term.Left!, name, value));
179 int end = bound.Length;
180 while (end > 0 && Digit(bound[end - 1])) end--;
181 string prefix = bound[..end], fresh;
182 int index = 1;
183 do
184 {
185 fresh = prefix + index++;
186 }
187 while (bodyFree.Contains(fresh) || valueFree.Contains(fresh));
188 return Abstraction(fresh, Substitute(Substitute(term.Left!, bound, Variable(fresh)), name, value));
189 }

201 static Term Rename(Term term, SortedSet<string> free, ref int next, List<Binding> bindings)
202 {
203 if (term.Kind == 'v')
204 {
205 for (int i = bindings.Count - 1; i >= 0; i--) if (bindings[i].OldName == term.Name) return Variable(bindings[i].NewName);
206 return term;
207 }
208 if (term.Kind == 'p')
209 {
210 Term left = Rename(term.Left!, free, ref next, bindings);
211 Term right = Rename(term.Right!, free, ref next, bindings);
212 return Application(left, right);
213 }
214 string name;
215 do
216 {
217 name = Sequence(next++);
218 }
219 while (free.Contains(name));
220 bindings.Add(new Binding(term.Name, name));
221 Term body = Rename(term.Left!, free, ref next, bindings);
222 bindings.RemoveAt(bindings.Count - 1);
223 return Abstraction(name, body);
224 }

228 static Term Debruijn(Term term, List<string> bindings)
229 {
230 if (term.Kind == 'v')
231 {

242 static bool Same(Term a, Term b)
243 {
244 if (a.Kind != b.Kind) return false;
245 return a.Kind switch
246 {
247 'v' or 'm' or 'c' => a.Name == b.Name,
248 'd' or 'n' => a.Number == b.Number,
249 'a' => Same(a.Left!, b.Left!),
250 _ => Same(a.Left!, b.Left!) && Same(a.Right!, b.Right!)
251 };
252 }

256 static bool EtaRedex(Term term) => term.Kind == 'a' && term.Left!.Kind == 'p' && term.Left.Right!.Kind == 'v' && term.Left.Right.Name == term.Name && !Free(term.Left.Left!).Contains(term.Name);

258 static Step? OneStep(Term term, string strategy, bool eta)
259 {
260 bool normal = strategy == "NORMAL", applicative = strategy == "APPLICATIVE", cbn = strategy == "CBN";
261 if (term.Kind == 'v') return null;
262 if (term.Kind == 'a')
263 {
264 if (!normal && !applicative) return null;
265 if (normal && eta && EtaRedex(term)) return new(term.Left!.Left!, "eta");
266 Step? body = OneStep(term.Left!, strategy, eta);
267 if (body != null) return new(Abstraction(term.Name, body.Term), body.Kind);
268 return eta && EtaRedex(term) ? new(term.Left!.Left!, "eta") : null;
269 }
270 if ((normal || cbn) && term.Left!.Kind == 'a') return new(Substitute(term.Left.Left!, term.Left.Name, term.Right!), "beta");
271 Step? left = OneStep(term.Left!, strategy, eta);
272 if (left != null) return new(Application(left.Term, term.Right!), left.Kind);
273 if (cbn) return null;
274 Step? right = OneStep(term.Right!, strategy, eta);
275 if (right != null) return new(Application(term.Left!, right.Term), right.Kind);
276 return term.Left!.Kind == 'a' && (normal || applicative || term.Right!.Kind is 'v' or 'a') ? new(Substitute(term.Left.Left!, term.Left.Name, term.Right!), "beta") : null;
277 }

291 Term Expand(Term term)
292 {
293 if (term.Kind == 'v') return term;
294 if (term.Kind == 'm')
295 {
296 if (!macros.TryGetValue(term.Name, out Term? value)) Fail("unbound macro " + term.Name);
297 return Expand(value!);
298 }
299 if (term.Kind == 'n')
300 {
301 Term body = Variable("x");
302 for (int i = 0; i < term.Number; i++) body = Application(Variable("f"), body);
303 return Abstraction("f", Abstraction("x", body));
304 }
305 if (term.Kind == 'a') return Abstraction(term.Name, Expand(term.Left!));
306 return Application(Expand(term.Left!), Expand(term.Right!));
307 }

309 bool Depends(string from, string on, HashSet<string>? seen = null)
310 {
311 seen ??= new(StringComparer.Ordinal);
312 if (!macros.TryGetValue(from, out Term? term) || !seen.Add(from)) return false;
313 List<string> refs = new();
314 References(term, refs);
315 return refs.Any(name => name == on || Depends(name, on, seen));
316 }

318 Reduced Reduce(Term term, string strategy, bool useEta, bool show = false, bool ski = false)
319 {
320 int count = 0;
321 while (true)
322 {
323 Step? next = ski ? SkiStep(term) : OneStep(term, strategy, useEta);
324 if (next == null) return new(term, count, true);
325 if (count == limit) return new(term, count, false);
326 term = next.Term;
327 count++;
328 if (show) Console.WriteLine($"step {count} ({next.Kind}): {Print(term)}");
329 }
330 }

356 static Type Resolved(Type type)
357 {
358 Type end = type;
359 while (end.Link != null) end = end.Link;
360 while (type.Link != null)
361 {
362 Type next = type.Link;
363 type.Link = end;
364 type = next;
365 }
366 return end;
367 }

369 static bool Occurs(Type variable, Type within)
370 {
371 within = Resolved(within);
372 return within == variable || within.Variable < 0 && (Occurs(variable, within.Left!) || Occurs(variable, within.Right!));
373 }
374
375 static void Unify(Type a, Type b)
376 {
377 a = Resolved(a);
378 b = Resolved(b);
379 if (a == b) return;
380 if (a.Variable >= 0)
381 {
382 if (Occurs(a, b)) throw new Untypable();
383 a.Link = b;
384 }
385 else if (b.Variable >= 0) Unify(b, a);
386 else
387 {
388 Unify(a.Left!, b.Left!);
389 Unify(a.Right!, b.Right!);
390 }
391 }

393 record TypedBinding(string Name, Type Type);

395 static Type Infer(Term term, List<TypedBinding> bindings, Dictionary<string, Type> free, ref int next)
396 {
397 if (term.Kind == 'v')
398 {
399 for (int i = bindings.Count - 1; i >= 0; i--) if (bindings[i].Name == term.Name) return bindings[i].Type;
400 if (!free.TryGetValue(term.Name, out Type? type)) free[term.Name] = type = new Type(next++);
401 return type;
402 }
403 if (term.Kind == 'a')
404 {
405 Type parameter = new(next++);
406 bindings.Add(new(term.Name, parameter));
407 Type body = Infer(term.Left!, bindings, free, ref next);
408 bindings.RemoveAt(bindings.Count - 1);
409 return new(parameter, body);
410 }
411 Type function = Infer(term.Left!, bindings, free, ref next), argument = Infer(term.Right!, bindings, free, ref next), result = new(next++);
412 Unify(function, new Type(argument, result));
413 return result;
414 }

417 {
418 type = Resolved(type);

429 static string Church(Term term)
430 {
431 if (term.Kind == 'a' && term.Left!.Kind == 'a')
432 {
433 string f = term.Name, x = term.Left.Name;
434 Term body = term.Left.Left!;
435 int count = 0;
436 while (body.Kind == 'p' && body.Left!.Kind == 'v' && body.Left.Name == f)
437 {
438 count++;
439 body = body.Right!;
440 }
441 if (body.Kind == 'v' && body.Name == x) return count.ToString(CultureInfo.InvariantCulture);
442 if (term.Left.Left!.Kind == 'v' && term.Left.Left.Name == f) return "true";
443 }
444 if (term.Kind == 'a' && term.Left!.Kind == 'p' && term.Left.Left!.Kind == 'p' && term.Left.Left.Left!.Kind == 'v' && term.Left.Left.Left.Name == term.Name)
445 {
446 Term p = term.Left.Left.Right!, q = term.Left.Right!;
447 if (!Free(p).Contains(term.Name) && !Free(q).Contains(term.Name)) return "(" + Church(p) + ", " + Church(q) + ")";
448 }
449 return Print(Canonical(term));
450 }

468 static Step? SkiStep(Term term)
469 {
470 List<Term> args = new();
471 Term head = term;
472 while (head.Kind == 'p')
473 {
474 args.Insert(0, head.Right!);
475 head = head.Left!;
476 }
477 if (head.Kind == 'c')
478 {
479 if (head.Name == "I" && args.Count > 0) return new(Apply(args[0], args, 1), "I");
480 if (head.Name == "K" && args.Count >= 2) return new(Apply(args[0], args, 2), "K");
481 if (head.Name == "S" && args.Count >= 3) return new(Apply(Application(Application(args[0], args[2]), Application(args[1], args[2])), args, 3), "S");
482 }
483 for (int i = 0; i < args.Count; i++)
484 {
485 Step? next = SkiStep(args[i]);
486 if (next != null)
487 {
488 args[i] = next.Term;
489 return new(Apply(head, args, 0), next.Kind);
490 }
491 }
492 return null;
493 }

497 static Term Realize(Term code, List<Closure> environment, int depth)
498 {
499 while (code.Kind == 'd' && code.Number >= depth)
500 {
501 Closure closure = environment[code.Number - depth];
502 code = closure.Term;
503 environment = closure.Environment;
504 depth = 0;
505 }
506 if (code.Kind == 'v' || code.Kind == 'd') return code;
507 if (code.Kind == 'a') return Abstraction("", Realize(code.Left!, environment, depth + 1));
508 return Application(Realize(code.Left!, environment, depth), Realize(code.Right!, environment, depth));
509 }

511 static Term FromDebruijn(Term term, SortedSet<string> free, List<string> bindings, ref int next)
512 {
513 if (term.Kind == 'v') return term;
514 if (term.Kind == 'd') return Variable(bindings[bindings.Count - 1 - term.Number]);
515 if (term.Kind == 'p')
516 {
517 Term left = FromDebruijn(term.Left!, free, bindings, ref next), right = FromDebruijn(term.Right!, free, bindings, ref next);
518 return Application(left, right);
519 }
520 string name;
521 do
522 {
523 name = Sequence(next++);
524 }
525 while (free.Contains(name));
526 bindings.Add(name);
527 Term body = FromDebruijn(term.Left!, free, bindings, ref next);
528 bindings.RemoveAt(bindings.Count - 1);
529 return Abstraction(name, body);

532 void Kam(Term term)
533 {
534 Term code = Debruijn(term);
535 List<Closure> environment = new(), stack = new();
536 int count = 0;
537 while (true)
538 {
539 string kind = code.Kind == 'p' ? "push" : code.Kind == 'a' && stack.Count > 0 ? "grab" : code.Kind == 'd' ? "access" : "";
540 if (kind.Length == 0) break;
541 if (count == limit)
542 {
543 Console.WriteLine($"no result within {limit} transitions");
544 return;
545 }
546 if (kind == "push")
547 {
548 stack.Add(new(code.Right!, environment));
549 code = code.Left!;
550 }
551 else if (kind == "grab")
552 {
553 List<Closure> next = new() { stack[^1] };
554 stack.RemoveAt(stack.Count - 1);
555 next.AddRange(environment);
556 environment = next;
557 code = code.Left!;
558 }
559 else
560 {
561 Closure closure = environment[code.Number];
562 code = closure.Term;
563 environment = closure.Environment;
564 }
565 count++;
566 if (display) Console.WriteLine($"step {count} ({kind}): {Print(code)} [env {environment.Count}, stack {stack.Count}]");
567 }
568 Term result = Realize(code, environment, 0);
569 for (int i = stack.Count - 1; i >= 0; i--) result = Application(result, Realize(stack[i].Term, stack[i].Environment, 0));
570 int index = 0;
571 result = FromDebruijn(result, Free(result), new(), ref index);
572 Console.WriteLine($"result: {Print(result)} (transitions: {count})");
573 }
574 static readonly HashSet<string> Reserved = new(StringComparer.Ordinal) { "S", "K", "I", "NORMAL", "APPLICATIVE", "CBN", "CBV" };

576 void Command(string line)
577 {
578 int split = line.IndexOf(' ');
579 string command = split < 0 ? line : line[..split], argument = split < 0 ? "" : line[(split + 1)..];
580 if (!new[] { "let", "unlet", "macros", "steps", "eta", "limit", "expand", "subst", "reduce", "equal", "alpha", "debruijn", "fv", "size", "depth", "type", "church", "ski", "skireduce", "kam" }.Contains(command)) Fail("unknown command " + command);
581 if (new[] { "unlet", "macros", "steps", "eta", "limit" }.Contains(command))
582 {
583 string[] words = argument.Split(' ', StringSplitOptions.RemoveEmptyEntries);
584 switch (command)
585 {
586 case "steps":
587 case "eta":
588 if (words.Length != 1 || (words[0] != "on" && words[0] != "off")) Fail("bad argument");
589 if (command == "steps") display = words[0] == "on";
590 else eta = words[0] == "on";
591 break;
592 case "limit":
593 if (words.Length != 1 || words[0].Length is < 1 or > 5 || !words[0].All(Digit)) Fail("bad argument");
594 int value = int.Parse(words[0]);
595 if (value < 1 || value > 10000) Fail("bad argument");
596 limit = value;
597 break;
598 case "macros":
599 if (words.Length != 0) Fail("bad argument");
600 bool any = false;
601 foreach (var entry in macros) if (entry.Key != "S" && entry.Key != "K" && entry.Key != "I")
602 {
603 Console.WriteLine(entry.Key + " = " + Print(entry.Value));
604 any = true;
605 }
606 if (!any) Console.WriteLine("(none)");
607 break;
608 case "unlet":
609 if (words.Length != 1 || !MacroName(words[0])) Fail("bad argument");
610 string name = words[0];
611 if (Reserved.Contains(name)) Fail("reserved name " + name);
612 if (!macros.ContainsKey(name)) Fail("unbound macro " + name);
613 foreach (var entry in macros)
614 {
615 List<string> refs = new();
616 References(entry.Value, refs);
617 if (refs.Contains(name)) Fail(name + " is used by " + entry.Key);
618 }
619 macros.Remove(name);
620 Console.WriteLine("removed " + name);
621 break;
622 }
623 return;
624 }
625 Parser parser = new(line, split < 0 ? line.Length : split + 1);
626 string? identifier = null;
627 string strategy = "NORMAL";
628 Term first;
629 Term? second = null;
630 if (command == "let")
631 {
632 identifier = parser.Expect("macro").Text;
633 parser.Expect("=");
634 first = parser.ReadTerm();
635 }
636 else if (command == "subst")
637 {
638 identifier = parser.Expect("variable").Text;
639 parser.Expect("=");
640 first = parser.ReadTerm();
641 parser.Expect(";");
642 second = parser.ReadTerm();
643 }
644 else
645 {
646 if (command == "reduce" && parser.Peek().Kind == "macro" && new[] { "NORMAL", "APPLICATIVE", "CBN", "CBV" }.Contains(parser.Peek().Text)) strategy = parser.Tokens[parser.At++].Text;
647 first = parser.ReadTerm();
648 if (command is "equal" or "alpha")
649 {
650 parser.Expect(";");
651 second = parser.ReadTerm();
652 }
653 }
654 parser.End();
655 if (command == "let")
656 {
657 if (Reserved.Contains(identifier!)) Fail("reserved name " + identifier);
658 List<string> refs = new();
659 References(first, refs);
660 foreach (string name in refs)
661 {
662 if (name == identifier || Depends(name, identifier!)) Fail("recursive definition of " + identifier);
663 if (!macros.ContainsKey(name)) Fail("unbound macro " + name);
664 }
665 var variables = Free(first);
666 if (variables.Count > 0) Fail("free variable " + variables.Min + " in definition of " + identifier);
667 Console.WriteLine((macros.ContainsKey(identifier!) ? "redefined " : "defined ") + identifier);
668 macros[identifier!] = first;
669 return;
670 }
671 first = Expand(first);
672 if (second != null) second = Expand(second);
673 switch (command)
674 {
675 case "expand":
676 Console.WriteLine(Print(first));
677 break;
678 case "subst":
679 Console.WriteLine(Print(Substitute(second!, identifier!, first)));
680 break;
681 case "debruijn":
682 Console.WriteLine(Print(Debruijn(first)));
683 break;
684 case "fv":
685 var variables = Free(first);
686 Console.WriteLine(variables.Count == 0 ? "(none)" : string.Join(" ", variables));
687 break;
688 case "size":
689 Console.WriteLine(Size(first));
690 break;
691 case "depth":
692 Console.WriteLine(Depth(first));
693 break;
694 case "alpha":
695 Console.WriteLine(Same(Debruijn(first), Debruijn(second!)) ? "yes" : "no");
696 break;
697 case "equal":
698 Reduced left = Reduce(first, "NORMAL", eta), right = Reduce(second!, "NORMAL", eta);
699 Console.WriteLine(!left.Success || !right.Success ? "unknown" : Same(Debruijn(left.Term), Debruijn(right.Term)) ? "yes" : "no");
700 break;
701 case "type":
702 try
703 {
704 int index = 0;
705 Console.WriteLine(PrintType(Infer(first, new(), new(StringComparer.Ordinal), ref index), new()));
706 }
707 catch (Untypable)
708 {
709 Console.WriteLine("not typable");
710 }
711 break;
712 case "church":
713 Reduced church = Reduce(first, "NORMAL", false);
714 Console.WriteLine(church.Success ? Church(Canonical(church.Term)) : Failure());
715 break;
716 case "ski":
717 Console.WriteLine(Print(Ski(first)));
718 break;
719 case "kam":
720 Kam(first);
721 break;
722 case "reduce":
723 case "skireduce":
724 bool ski = command == "skireduce";
725 Reduced result = Reduce(ski ? Ski(first) : first, strategy, eta, display, ski);
726 Console.WriteLine(result.Success ? $"result: {Print(ski ? result.Term : Canonical(result.Term))} (steps: {result.Count})" : Failure());
727 break;
728 }
729 }
