40 kind: Kind,
41 name: Text = "",
42 value: usize = 0,
43 left: ?*Term = null,

50const Pool = struct {
51 gpa: Allocator,
52 nodes: std.ArrayList(*Term) = .empty,
53 available: std.ArrayList(*Term) = .empty,
54 fn node(self: *Pool, term: Term) Error!*Term {
55 if (self.available.pop()) |result| {
56 result.* = term;
57 return result;
58 }
59 const result = try self.gpa.create(Term);
60 result.* = term;
61 try self.nodes.append(self.gpa, result);
62 return result;
63 }
64 fn variable(self: *Pool, name: Text) Error!*Term {
65 return self.node(.{ .kind = .variable, .name = name });
66 }
67 fn abstraction(self: *Pool, name: Text, body: *Term) Error!*Term {
68 return self.node(.{ .kind = .abstraction, .name = name, .left = body });
69 }
70 fn application(self: *Pool, function: *Term, argument: *Term) Error!*Term {
71 return self.node(.{ .kind = .application, .left = function, .right = argument });
72 }
73 fn constant(self: *Pool, name: Text) Error!*Term {
74 return self.node(.{ .kind = .constant, .name = name });
75 }
76 fn mark(term: *Term) void {
77 if (term.marked) return;
78 term.marked = true;
79 if (term.left) |child| mark(child);
80 if (term.right) |child| mark(child);
81 }
82 fn collect(self: *Pool, macros: std.StringHashMapUnmanaged(*Term), roots: []const *Term, current: ?*Term) Error!void {
83 for (self.nodes.items) |term| term.marked = false;
84 var iterator = macros.valueIterator();
85 while (iterator.next()) |term| mark(term.*);
86 for (roots) |term| mark(term);
87 if (current) |term| mark(term);
88 for (self.nodes.items) |term| if (term.alive and !term.marked) {
89 term.free.deinit(std.heap.page_allocator);
90 term.free = .empty;
91 term.alive = false;
92 try self.available.append(self.gpa, term);
93 };
94 }
95 fn deinit(self: *Pool) void {
96 for (self.nodes.items) |term| if (term.alive) term.free.deinit(std.heap.page_allocator);
97 }

99fn freeVariables(term: *Term) Error!*std.StringHashMapUnmanaged(void) {
100 if (term.freeReady) return &term.free;
101 if (term.kind == .variable) {
102 try term.free.put(std.heap.page_allocator, term.name, {});
103 } else if (term.kind == .abstraction or term.kind == .application) {
104 var iterator = (try freeVariables(term.left.?)).keyIterator();
105 while (iterator.next()) |name| if (term.kind != .abstraction or !equal(name.*, term.name)) try term.free.put(std.heap.page_allocator, name.*, {});
106 if (term.kind == .application) {
107 var right = (try freeVariables(term.right.?)).keyIterator();
108 while (right.next()) |name| try term.free.put(std.heap.page_allocator, name.*, {});
109 }
110 }
111 term.freeReady = true;
112 return &term.free;
113}
114fn hasFree(term: *Term, name: Text) Error!bool {
115 return (try freeVariables(term)).contains(name);
116}
117fn sortedKeys(gpa: Allocator, set: anytype) Error![]const Text {
118 var result: std.ArrayList(Text) = .empty;
119 var iterator = set.keyIterator();
120 while (iterator.next()) |key| try result.append(gpa, key.*);
121 std.mem.sort(Text, result.items, {}, struct {
122 fn less(_: void, a: Text, b: Text) bool {
123 return std.mem.order(u8, a, b) == .lt;
124 }
125 }.less);
126 return result.items;
127}
128fn printInto(term: *Term, end: bool, debruijn: bool, out: *Builder) Error!void {
129 switch (term.kind) {
130 .variable, .macro, .constant => try out.add(term.name),
131 .index => try out.print("{d}", .{term.value}),
132 .numeral => try out.print("#{d}", .{term.value}),
133 .abstraction => {
134 if (!end) try out.add("(");
135 var body = term;
136 if (!debruijn) try out.add("\\");
137 var count: usize = 0;
138 while (body.kind == .abstraction) {
139 if (debruijn) {
140 try out.add("\\");
141 } else {
142 if (count > 0) try out.add(" ");
143 try out.add(body.name);
144 }
145 count += 1;
146 body = body.left.?;
147 }
148 if (!debruijn) try out.add(". ");
149 try printInto(body, true, debruijn, out);
150 if (!end) try out.add(")");
151 },
152 .application => {
153 try printInto(term.left.?, false, debruijn, out);
154 try out.add(" ");
155 if (term.right.?.kind == .application) {
156 try out.add("(");
157 try printInto(term.right.?, true, debruijn, out);
158 try out.add(")");
159 } else try printInto(term.right.?, end, debruijn, out);
160 },
161 }
162}
163fn printed(term: *Term, debruijn: bool) Error![]u8 {
164 var out: Builder = .{ .gpa = std.heap.page_allocator };
165 errdefer out.bytes.deinit(out.gpa);
166 try printInto(term, true, debruijn, &out);
167 return out.bytes.toOwnedSlice(out.gpa);
168}
169fn nameSequence(gpa: Allocator, index: usize) Error!Text {
170 const letter: u8 = 'a' + @as(u8, @intCast(index % 26));
171 return if (index < 26) try std.fmt.allocPrint(gpa, "{c}", .{letter}) else try std.fmt.allocPrint(gpa, "{c}{d}", .{ letter, index / 26 });
172}
173fn substitute(pool: *Pool, term: *Term, name: Text, replacement: *Term) Error!*Term {
174 if (term.kind == .variable) return if (equal(term.name, name)) replacement else term;
175 if (term.kind == .application) return pool.application(try substitute(pool, term.left.?, name, replacement), try substitute(pool, term.right.?, name, replacement));
176 if (term.kind != .abstraction or equal(term.name, name) or !try hasFree(term.left.?, name)) return term;
177 var binder = term.name;
178 var body = term.left.?;
179 if (try hasFree(replacement, binder)) {
180 var end = binder.len;
181 while (end > 0 and digit(binder[end - 1])) end -= 1;
182 var index: usize = 1;
183 while (true) : (index += 1) {
184 const candidate = try std.fmt.allocPrint(pool.gpa, "{s}{d}", .{ binder[0..end], index });
185 if (!try hasFree(replacement, candidate) and !try hasFree(body, candidate)) {
186 binder = candidate;
187 break;
188 }
189 }
190 body = try substitute(pool, body, term.name, try pool.variable(binder));
191 }
192 return pool.abstraction(binder, try substitute(pool, body, name, replacement));
193}
194const Binding = struct { original: Text, renamed: Text };
195const Canonicalizer = struct {
196 pool: *Pool,
197 free: *std.StringHashMapUnmanaged(void),
198 bound: std.ArrayList(Binding) = .empty,
199 counter: usize = 0,
200 fromDB: bool = false,
201 fn visit(self: *Canonicalizer, term: *Term) Error!*Term {
202 if (term.kind == .variable) {
203 if (!self.fromDB) {
204 var i = self.bound.items.len;
205 while (i > 0) {
206 i -= 1;
207 if (equal(self.bound.items[i].original, term.name)) return self.pool.variable(self.bound.items[i].renamed);
208 }
209 }
210 return term;
211 }
212 if (term.kind == .index) return self.pool.variable(self.bound.items[self.bound.items.len - 1 - term.value].renamed);
213 if (term.kind == .application) return self.pool.application(try self.visit(term.left.?), try self.visit(term.right.?));
214 var name: Text = undefined;
215 while (true) {
216 name = try nameSequence(self.pool.gpa, self.counter);
217 self.counter += 1;
218 if (!self.free.contains(name)) break;
219 }
220 try self.bound.append(std.heap.page_allocator, .{ .original = term.name, .renamed = name });
221 const body = try self.visit(term.left.?);
222 _ = self.bound.pop();
223 return self.pool.abstraction(name, body);
224 }
225};
226fn canonical(pool: *Pool, term: *Term, fromDB: bool) Error!*Term {
227 var converter: Canonicalizer = .{ .pool = pool, .free = try freeVariables(term), .fromDB = fromDB };
228 defer converter.bound.deinit(std.heap.page_allocator);
229 return converter.visit(term);
230}
231fn deBruijn(pool: *Pool, term: *Term, bound: *std.ArrayList(Text)) Error!*Term {
232 if (term.kind == .variable) {
233 var i = bound.items.len;
234 while (i > 0) {
235 i -= 1;
236 if (equal(bound.items[i], term.name)) return pool.node(.{ .kind = .index, .value = bound.items.len - 1 - i });
237 }
238 return term;
239 }
240 if (term.kind == .application) return pool.application(try deBruijn(pool, term.left.?, bound), try deBruijn(pool, term.right.?, bound));
241 try bound.append(std.heap.page_allocator, term.name);
242 const body = try deBruijn(pool, term.left.?, bound);
243 _ = bound.pop();
244 return pool.abstraction("", body);
245}
246const TokenKind = enum { variable, macro, numeral, symbol, end };
247const Token = struct { kind: TokenKind, text: Text = "", column: usize, value: usize = 0 };
248const Parser = struct {
249 bench: *Workbench,
250 tokens: []const Token,
251 position: usize = 0,
252 fn peek(self: *Parser) Token {
253 return self.tokens[self.position];
254 }
255 fn expect(self: *Parser, kind: Text) Error!Token {
256 const token = self.peek();
257 const matches = if (equal(kind, "variable")) token.kind == .variable else if (equal(kind, "macro")) token.kind == .macro else token.kind == .symbol and equal(token.text, kind);
258 if (!matches) {
259 if (equal(kind, "variable")) return self.bench.bad("column {d}: expected variable", .{token.column});
260 if (equal(kind, "macro")) return self.bench.bad("column {d}: expected macro name", .{token.column});
261 return self.bench.bad("column {d}: expected '{s}'", .{ token.column, kind });
262 }
263 self.position += 1;
264 return token;
265 }
266 fn readTerm(self: *Parser) Error!*Term {
267 var elements: std.ArrayList(*Term) = .empty;
268 while (true) {
269 const token = self.peek();
270 if (!(token.kind == .variable or token.kind == .macro or token.kind == .numeral or equal(token.text, "("))) break;
271 self.position += 1;
272 var value: *Term = undefined;
273 if (equal(token.text, "(")) {
274 value = try self.readTerm();
275 _ = try self.expect(")");
276 } else value = try self.bench.pool.node(.{ .kind = switch (token.kind) {
277 .variable => .variable,
278 .macro => .macro,
279 .numeral => .numeral,
280 else => unreachable,
281 }, .name = token.text, .value = token.value });
282 try elements.append(self.bench.scratch, value);
283 }
284 if (equal(self.peek().text, "\\")) {
285 self.position += 1;
286 var names: std.ArrayList(Text) = .empty;
287 try names.append(self.bench.scratch, (try self.expect("variable")).text);
288 while (self.peek().kind == .variable) try names.append(self.bench.scratch, (try self.expect("variable")).text);
289 _ = try self.expect(".");
290 var value = try self.readTerm();
291 var i = names.items.len;
292 while (i > 0) {
293 i -= 1;
294 value = try self.bench.pool.abstraction(names.items[i], value);
295 }
296 try elements.append(self.bench.scratch, value);
297 }
298 if (elements.items.len == 0) return self.bench.bad("column {d}: expected term", .{self.peek().column});
299 var result = elements.items[0];
300 for (elements.items[1..]) |next| result = try self.bench.pool.application(result, next);
301 return result;
302 }
303 fn finish(self: *Parser) Error!void {
304 const token = self.peek();
305 if (token.kind != .end) return self.bench.bad("column {d}: unexpected '{s}'", .{ token.column, token.text });
306 }

308fn references(term: *Term, result: *std.ArrayList(Text), gpa: Allocator) Error!void {
309 if (term.kind == .macro) {
310 try result.append(gpa, term.name);
311 } else if (term.kind == .abstraction) {
312 try references(term.left.?, result, gpa);
313 } else if (term.kind == .application) {
314 try references(term.left.?, result, gpa);
315 try references(term.right.?, result, gpa);
316 }

318const Step = struct { term: *Term, kind: Text };
319fn etaRedex(term: *Term) Error!bool {
320 return term.kind == .abstraction and term.left.?.kind == .application and term.left.?.right.?.kind == .variable and equal(term.left.?.right.?.name, term.name) and !try hasFree(term.left.?.left.?, term.name);
321}
322fn oneStep(pool: *Pool, term: *Term, strategy: Text, eta: bool) Error!?Step {
323 if (term.kind == .application) {
324 if (member(&.{ "NORMAL", "CBN" }, strategy) and term.left.?.kind == .abstraction) return Step{ .term = try substitute(pool, term.left.?.left.?, term.left.?.name, term.right.?), .kind = "beta" };
325 if (try oneStep(pool, term.left.?, strategy, eta)) |step| return Step{ .term = try pool.application(step.term, term.right.?), .kind = step.kind };
326 if (equal(strategy, "CBN")) return null;
327 if (try oneStep(pool, term.right.?, strategy, eta)) |step| return Step{ .term = try pool.application(term.left.?, step.term), .kind = step.kind };
328 if (term.left.?.kind == .abstraction and (!equal(strategy, "CBV") or term.right.?.kind == .variable or term.right.?.kind == .abstraction)) return Step{ .term = try substitute(pool, term.left.?.left.?, term.left.?.name, term.right.?), .kind = "beta" };
329 } else if (term.kind == .abstraction and !member(&.{ "CBN", "CBV" }, strategy)) {
330 if (equal(strategy, "NORMAL") and eta and try etaRedex(term)) return Step{ .term = term.left.?.left.?, .kind = "eta" };
331 if (try oneStep(pool, term.left.?, strategy, eta)) |step| return Step{ .term = try pool.abstraction(term.name, step.term), .kind = step.kind };
332 if (eta and try etaRedex(term)) return Step{ .term = term.left.?.left.?, .kind = "eta" };
333 }
334 return null;
335}
336fn termSize(term: *Term) usize {
337 return switch (term.kind) {
338 .variable => 1,
339 .abstraction => 1 + termSize(term.left.?),
340 else => 1 + termSize(term.left.?) + termSize(term.right.?),
341 };
342}
343fn termDepth(term: *Term) usize {
344 return switch (term.kind) {
345 .variable => 1,
346 .abstraction => 1 + termDepth(term.left.?),
347 else => 1 + @max(termDepth(term.left.?), termDepth(term.right.?)),
348 };
349}

351const SimpleType = struct { left: ?*SimpleType = null, right: ?*SimpleType = null, link: ?*SimpleType = null };
352fn resolve(value: *SimpleType) *SimpleType {
353 if (value.link) |next| {
354 value.link = resolve(next);
355 return value.link.?;
356 }
357 return value;
358}
359fn occurs(variableType: *SimpleType, raw: *SimpleType) bool {
360 const value = resolve(raw);
361 return variableType == value or value.left != null and (occurs(variableType, value.left.?) or occurs(variableType, value.right.?));
362}
363fn unify(rawA: *SimpleType, rawB: *SimpleType) Error!void {
364 const a = resolve(rawA);
365 const b = resolve(rawB);
366 if (a == b) return;
367 if (a.left == null) {
368 if (occurs(a, b)) return error.NotTypable;
369 a.link = b;
370 return;
371 }
372 if (b.left == null) return unify(b, a);
373 try unify(a.left.?, b.left.?);
374 try unify(a.right.?, b.right.?);
375}
376const TypeBinding = struct { name: Text, value: *SimpleType };
377const TypeInference = struct {
378 gpa: Allocator,
379 free: std.StringHashMapUnmanaged(*SimpleType) = .empty,
380 bound: std.ArrayList(TypeBinding) = .empty,
381 names: std.AutoHashMapUnmanaged(*SimpleType, usize) = .empty,
382 fn fresh(self: *TypeInference) Error!*SimpleType {
383 const value = try self.gpa.create(SimpleType);
384 value.* = .{};
385 return value;
386 }
387 fn arrow(self: *TypeInference, left: *SimpleType, right: *SimpleType) Error!*SimpleType {
388 const value = try self.fresh();
389 value.left = left;
390 value.right = right;
391 return value;
392 }
393 fn infer(self: *TypeInference, term: *Term) Error!*SimpleType {
394 if (term.kind == .variable) {
395 var i = self.bound.items.len;
396 while (i > 0) {
397 i -= 1;
398 if (equal(self.bound.items[i].name, term.name)) return self.bound.items[i].value;
399 }
400 if (self.free.get(term.name)) |value| return value;
401 const value = try self.fresh();
402 try self.free.put(self.gpa, term.name, value);
403 return value;
404 }
405 if (term.kind == .abstraction) {
406 const argument = try self.fresh();
407 try self.bound.append(self.gpa, .{ .name = term.name, .value = argument });
408 const body = try self.infer(term.left.?);
409 _ = self.bound.pop();
410 return self.arrow(argument, body);
411 }
412 const function = try self.infer(term.left.?);
413 const argument = try self.infer(term.right.?);
414 const result = try self.fresh();
415 try unify(function, try self.arrow(argument, result));
416 return result;
417 }
418 fn print(self: *TypeInference, raw: *SimpleType, out: *Builder) Error!void {
419 const value = resolve(raw);
420 if (value.left == null) {
421 const entry = try self.names.getOrPut(self.gpa, value);
422 if (!entry.found_existing) entry.value_ptr.* = self.names.count() - 1;
423 try out.add(try nameSequence(self.gpa, entry.value_ptr.*));
424 return;
425 }
426 const parentheses = resolve(value.left.?).left != null;
427 if (parentheses) try out.add("(");
428 try self.print(value.left.?, out);
429 if (parentheses) try out.add(")");
430 try out.add(" -> ");
431 try self.print(value.right.?, out);
432 }
433};
434fn churchInto(pool: *Pool, term: *Term, out: *Builder) Error!void {
435 if (term.kind == .abstraction and term.left.?.kind == .abstraction) {
436 const outer = term.name;
437 const inner = term.left.?.name;
438 var body = term.left.?.left.?;
439 var count: usize = 0;
440 while (body.kind == .application and body.left.?.kind == .variable and equal(body.left.?.name, outer)) {
441 count += 1;
442 body = body.right.?;
443 }
444 if (body.kind == .variable and equal(body.name, inner)) {
445 try out.print("{d}", .{count});
446 return;
447 }
448 if (term.left.?.left.?.kind == .variable and equal(term.left.?.left.?.name, outer)) {
449 try out.add("true");
450 return;
451 }
452 }
453 if (term.kind == .abstraction and term.left.?.kind == .application and term.left.?.left.?.kind == .application) {
454 const body = term.left.?;
455 const function = body.left.?.left.?;
456 if (function.kind == .variable and equal(function.name, term.name) and !try hasFree(body.left.?.right.?, term.name) and !try hasFree(body.right.?, term.name)) {
457 try out.add("(");
458 try churchInto(pool, body.left.?.right.?, out);
459 try out.add(", ");
460 try churchInto(pool, body.right.?, out);
461 try out.add(")");
462 return;
463 }
464 }
465 try printInto(try canonical(pool, term, false), true, false, out);
466}
467fn abstractSKI(pool: *Pool, name: Text, term: *Term) Error!*Term {
468 if (!try hasFree(term, name)) return pool.application(try pool.constant("K"), term);
469 if (term.kind == .variable) return pool.constant("I");
470 if (term.right.?.kind == .variable and equal(term.right.?.name, name) and !try hasFree(term.left.?, name)) return term.left.?;
471 return pool.application(try pool.application(try pool.constant("S"), try abstractSKI(pool, name, term.left.?)), try abstractSKI(pool, name, term.right.?));
472}
473fn ski(pool: *Pool, term: *Term) Error!*Term {
474 if (term.kind == .variable) return term;
475 if (term.kind == .application) return pool.application(try ski(pool, term.left.?), try ski(pool, term.right.?));
476 return abstractSKI(pool, term.name, try ski(pool, term.left.?));
477}
478fn skiStep(pool: *Pool, term: *Term) Error!?Step {
479 var head = term;
480 var arguments: std.ArrayList(*Term) = .empty;
481 defer arguments.deinit(std.heap.page_allocator);
482 while (head.kind == .application) {
483 try arguments.append(std.heap.page_allocator, head.right.?);
484 head = head.left.?;
485 }
486 std.mem.reverse(*Term, arguments.items);
487 var result: ?*Term = null;
488 var used: usize = 0;
489 var kind: Text = "";
490 if (head.kind == .constant) {
491 if (equal(head.name, "I") and arguments.items.len >= 1) {
492 result = arguments.items[0];
493 used = 1;
494 kind = "I";
495 } else if (equal(head.name, "K") and arguments.items.len >= 2) {
496 result = arguments.items[0];
497 used = 2;
498 kind = "K";
499 } else if (equal(head.name, "S") and arguments.items.len >= 3) {
500 result = try pool.application(try pool.application(arguments.items[0], arguments.items[2]), try pool.application(arguments.items[1], arguments.items[2]));
501 used = 3;
502 kind = "S";
503 }
504 }
505 if (result != null) {
506 for (arguments.items[used..]) |argument| result = try pool.application(result.?, argument);
507 return Step{ .term = result.?, .kind = kind };
508 }
509 for (arguments.items, 0..) |argument, i| if (try skiStep(pool, argument)) |step| {
510 var value = head;
511 for (arguments.items, 0..) |item, j| value = try pool.application(value, if (i == j) step.term else item);
512 return Step{ .term = value, .kind = step.kind };
513 };
514 return null;
515}
516const Closure = struct { code: *Term, environment: []const *Closure };
517fn reify(pool: *Pool, initialCode: *Term, initialEnvironment: []const *Closure, initialInside: usize) Error!*Term {
518 var code = initialCode;
519 var environment = initialEnvironment;
520 var inside = initialInside;
521 while (code.kind == .index and code.value >= inside) {
522 const closure = environment[code.value - inside];
523 code = closure.code;
524 environment = closure.environment;
525 inside = 0;
526 }
527 if (code.kind == .variable or code.kind == .index) return code;
528 if (code.kind == .abstraction) return pool.abstraction("", try reify(pool, code.left.?, environment, inside + 1));
529 return pool.application(try reify(pool, code.left.?, environment, inside), try reify(pool, code.right.?, environment, inside));
530}

535 out: *Builder,
536 pool: Pool,
537 macros: std.StringHashMapUnmanaged(*Term) = .empty,
538 roots: []const *Term = &.{},
539 steps: bool = false,
540 eta: bool = false,
541 limit: usize = 10000,
542 fn bad(self: *Workbench, comptime format: Text, args: anytype) Error {
543 failure = std.fmt.allocPrint(self.gpa, format, args) catch return error.OutOfMemory;
544 return error.Invalid;
545 }
546 fn initialize(self: *Workbench) Error!void {
547 try self.macros.put(self.gpa, "I", try self.pool.abstraction("x", try self.pool.variable("x")));
548 try self.macros.put(self.gpa, "K", try self.pool.abstraction("x", try self.pool.abstraction("y", try self.pool.variable("x"))));
549 const body = try self.pool.application(try self.pool.application(try self.pool.variable("x"), try self.pool.variable("z")), try self.pool.application(try self.pool.variable("y"), try self.pool.variable("z")));
550 try self.macros.put(self.gpa, "S", try self.pool.abstraction("x", try self.pool.abstraction("y", try self.pool.abstraction("z", body))));
551 }
552 fn lex(self: *Workbench, line: Text, text: Text, offset: usize) Error![]const Token {
553 var tokens: std.ArrayList(Token) = .empty;
554 var i: usize = 0;
555 while (i < text.len) {
556 if (text[i] == ' ') {
557 i += 1;
558 continue;
559 }
560 const start = i;
561 var kind: TokenKind = undefined;
562 const c = text[i];
563 if (std.mem.findScalar(u8, "\\.()=;", c) != null) {
564 kind = .symbol;
565 i += 1;
566 } else if (lower(c)) {
567 kind = .variable;
568 while (i < text.len and (lower(text[i]) or digit(text[i]))) i += 1;
569 } else if (upper(c)) {
570 kind = .macro;
571 while (i < text.len and (upper(text[i]) or digit(text[i]) or text[i] == '_')) i += 1;
572 } else if (c == '#' and i + 1 < text.len and digit(text[i + 1])) {
573 kind = .numeral;
574 i += 1;
575 while (i < text.len and digit(text[i])) i += 1;
576 } else return self.bad("column {d}: unexpected character '{c}'", .{ offset + i + 1, c });
577 try tokens.append(self.scratch, .{ .kind = kind, .text = text[start..i], .column = offset + start + 1, .value = if (kind == .numeral) std.fmt.parseInt(usize, text[start + 1 .. i], 10) catch unreachable else 0 });
578 }
579 try tokens.append(self.scratch, .{ .kind = .end, .column = line.len + 1 });
580 return tokens.items;
581 }
582 fn expand(self: *Workbench, term: *Term) Error!*Term {
583 switch (term.kind) {
584 .variable => return term,
585 .macro => {
586 const value = self.macros.get(term.name) orelse return self.bad("unbound macro {s}", .{term.name});
587 return self.expand(value);
588 },
589 .numeral => {
590 var body = try self.pool.variable("x");
591 for (0..term.value) |_| body = try self.pool.application(try self.pool.variable("f"), body);
592 return self.pool.abstraction("f", try self.pool.abstraction("x", body));
593 },
594 .abstraction => return self.pool.abstraction(term.name, try self.expand(term.left.?)),
595 else => return self.pool.application(try self.expand(term.left.?), try self.expand(term.right.?)),
596 }
597 }
598 fn refs(self: *Workbench, term: *Term) Error![]const Text {
599 var result: std.ArrayList(Text) = .empty;
600 try references(term, &result, self.scratch);
601 return result.items;
602 }
603 fn depends(self: *Workbench, name: Text, target: Text, seen: *std.StringHashMapUnmanaged(void)) Error!bool {
604 if (equal(name, target)) return true;
605 if (seen.contains(name) or member(reserved[0..3], name)) return false;
606 const term = self.macros.get(name) orelse return false;
607 try seen.put(self.scratch, name, {});
608 for (try self.refs(term)) |reference| if (try self.depends(reference, target, seen)) return true;
609 return false;
610 }
611 fn show(self: *Workbench, term: *Term, db: bool) Error!void {
612 const text = try printed(term, db);
613 defer std.heap.page_allocator.free(text);
614 try self.out.line("{s}", .{text});
615 }
616 fn reduce(self: *Workbench, initialTerm: *Term, strategy: Text, eta: bool, display: bool) Error!Reduction {
617 var term = initialTerm;
618 var count: usize = 0;
619 while (true) {
620 const step = (try oneStep(&self.pool, term, strategy, eta)) orelse return .{ .term = term, .count = count, .success = true };
621 if (count == self.limit) return .{ .term = term, .count = count, .success = false };
622 term = step.term;
623 count += 1;
624 if (display) {
625 const text = try printed(term, false);
626 defer std.heap.page_allocator.free(text);
627 try self.out.line("step {d} ({s}): {s}", .{ count, step.kind, text });
628 }
629 try self.pool.collect(self.macros, self.roots, term);
630 }
631 }
632 fn equivalent(self: *Workbench, a: *Term, b: *Term) Error!bool {
633 var bound: std.ArrayList(Text) = .empty;
634 defer bound.deinit(std.heap.page_allocator);
635 const left = try deBruijn(&self.pool, a, &bound);
636 const right = try deBruijn(&self.pool, b, &bound);
637 const first = try printed(left, true);
638 defer std.heap.page_allocator.free(first);
639 const second = try printed(right, true);
640 defer std.heap.page_allocator.free(second);
641 return equal(first, second);
642 }
643 fn kam(self: *Workbench, term: *Term) Error!void {
644 var bound: std.ArrayList(Text) = .empty;
645 defer bound.deinit(std.heap.page_allocator);
646 var code = try deBruijn(&self.pool, term, &bound);
647 var environment: []const *Closure = &.{};
648 var stack: std.ArrayList(*Closure) = .empty;
649 var count: usize = 0;
650 while (true) {
651 const kind: Text = if (code.kind == .application) "push" else if (code.kind == .index) "access" else if (code.kind == .abstraction and stack.items.len > 0) "grab" else "";
652 if (kind.len == 0) break;
653 if (count == self.limit) {
654 try self.out.line("no result within {d} transitions", .{self.limit});
655 return;
656 }
657 count += 1;
658 if (equal(kind, "push")) {
659 const closure = try self.scratch.create(Closure);
660 closure.* = .{ .code = code.right.?, .environment = environment };
661 try stack.append(self.scratch, closure);
662 code = code.left.?;
663 } else if (equal(kind, "grab")) {
664 const next = try self.scratch.alloc(*Closure, environment.len + 1);
665 next[0] = stack.pop().?;
666 @memcpy(next[1..], environment);
667 environment = next;
668 code = code.left.?;
669 } else {
670 const closure = environment[code.value];
671 code = closure.code;
672 environment = closure.environment;
673 }
674 if (self.steps) {
675 const text = try printed(code, true);
676 defer std.heap.page_allocator.free(text);
677 try self.out.line("step {d} ({s}): {s} [env {d}, stack {d}]", .{ count, kind, text, environment.len, stack.items.len });
678 }
679 }
680 var result = try reify(&self.pool, code, environment, 0);
681 var i = stack.items.len;
682 while (i > 0) {
683 i -= 1;
684 result = try self.pool.application(result, try reify(&self.pool, stack.items[i].code, stack.items[i].environment, 0));
685 }
686 const text = try printed(try canonical(&self.pool, result, true), false);
687 defer std.heap.page_allocator.free(text);
688 try self.out.line("result: {s} (transitions: {d})", .{ text, count });
689 }
690 fn command(self: *Workbench, line: Text) Error!void {
691 if (line.len == 0 or std.mem.startsWith(u8, line, "--")) return;
692 const separator = std.mem.findScalar(u8, line, ' ') orelse line.len;
693 const name = line[0..separator];
694 const arguments = if (separator < line.len) line[separator + 1 ..] else "";
695 const offset = if (separator < line.len) separator + 1 else separator;
696 if (!member(&.{ "let", "unlet", "macros", "steps", "eta", "limit", "expand", "subst", "reduce", "equal", "alpha", "debruijn", "fv", "size", "depth", "type", "church", "ski", "skireduce", "kam" }, name)) return self.bad("unknown command {s}", .{name});
697 if (member(&.{ "unlet", "macros", "steps", "eta", "limit" }, name)) {
698 var parts: std.ArrayList(Text) = .empty;
699 var iterator = std.mem.tokenizeScalar(u8, arguments, ' ');
700 while (iterator.next()) |part| try parts.append(self.scratch, part);
701 if (equal(name, "steps") or equal(name, "eta")) {
702 if (parts.items.len != 1 or !member(&.{ "on", "off" }, parts.items[0])) return self.bad("bad argument", .{});
703 if (equal(name, "steps")) {
704 self.steps = equal(parts.items[0], "on");
705 } else self.eta = equal(parts.items[0], "on");
706 } else if (equal(name, "limit")) {
707 if (parts.items.len != 1 or parts.items[0].len == 0 or parts.items[0].len > 5) return self.bad("bad argument", .{});
708 for (parts.items[0]) |c| if (!digit(c)) return self.bad("bad argument", .{});
709 const value = std.fmt.parseInt(usize, parts.items[0], 10) catch unreachable;
710 if (value < 1 or value > 10000) return self.bad("bad argument", .{});
711 self.limit = value;
712 } else if (equal(name, "macros")) {
713 if (parts.items.len != 0) return self.bad("bad argument", .{});
714 var count: usize = 0;
715 for (try sortedKeys(self.scratch, self.macros)) |macroName| {
716 if (member(reserved[0..3], macroName)) continue;
717 const text = try printed(self.macros.get(macroName).?, false);
718 defer std.heap.page_allocator.free(text);
719 try self.out.line("{s} = {s}", .{ macroName, text });
720 count += 1;
721 }
722 if (count == 0) try self.out.line("(none)", .{});
723 } else {
724 if (parts.items.len != 1 or !validMacro(parts.items[0])) return self.bad("bad argument", .{});
725 const macroName = parts.items[0];
726 if (member(&reserved, macroName)) return self.bad("reserved name {s}", .{macroName});
727 if (!self.macros.contains(macroName)) return self.bad("unbound macro {s}", .{macroName});
728 for (try sortedKeys(self.scratch, self.macros)) |other| if (member(try self.refs(self.macros.get(other).?), macroName)) return self.bad("{s} is used by {s}", .{ macroName, other });
729 _ = self.macros.remove(macroName);
730 try self.out.line("removed {s}", .{macroName});
731 }
732 return;
733 }
734 var parser: Parser = .{ .bench = self, .tokens = try self.lex(line, arguments, offset) };
735 var targetName: Text = "";
736 var strategy: Text = "NORMAL";
737 if (equal(name, "let") or equal(name, "subst")) {
738 targetName = (try parser.expect(if (equal(name, "let")) "macro" else "variable")).text;
739 _ = try parser.expect("=");
740 }
741 if (equal(name, "reduce") and parser.peek().kind == .macro and member(reserved[3..], parser.peek().text)) {
742 strategy = parser.peek().text;
743 parser.position += 1;
744 }
745 var first = try parser.readTerm();
746 var second: ?*Term = null;
747 if (member(&.{ "subst", "equal", "alpha" }, name)) {
748 _ = try parser.expect(";");
749 second = try parser.readTerm();
750 }
751 try parser.finish();
752 if (equal(name, "let")) {
753 if (member(&reserved, targetName)) return self.bad("reserved name {s}", .{targetName});
754 var seen: std.StringHashMapUnmanaged(void) = .empty;
755 for (try self.refs(first)) |reference| {
756 if (try self.depends(reference, targetName, &seen)) return self.bad("recursive definition of {s}", .{targetName});
757 if (!self.macros.contains(reference)) return self.bad("unbound macro {s}", .{reference});
758 }
759 const free = try sortedKeys(self.scratch, (try freeVariables(first)).*);
760 if (free.len > 0) return self.bad("free variable {s} in definition of {s}", .{ free[0], targetName });
761 const existed = self.macros.contains(targetName);
762 try self.macros.put(self.gpa, targetName, first);
763 try self.out.line("{s} {s}", .{ if (existed) "redefined" else "defined", targetName });
764 return;
765 }
766 first = try self.expand(first);
767 if (second) |term| second = try self.expand(term);
768 var roots: std.ArrayList(*Term) = .empty;
769 try roots.append(self.scratch, first);
770 if (second) |term| try roots.append(self.scratch, term);
771 self.roots = roots.items;
772 if (equal(name, "expand")) {
773 try self.show(first, false);
774 } else if (equal(name, "subst")) {
775 try self.show(try substitute(&self.pool, second.?, targetName, first), false);
776 } else if (equal(name, "alpha")) {
777 try self.out.line("{s}", .{if (try self.equivalent(first, second.?)) "yes" else "no"});
778 } else if (equal(name, "equal")) {
779 const a = try self.reduce(first, "NORMAL", self.eta, false);
780 const preserved = try self.scratch.dupe(*Term, &.{ first, second.?, a.term });
781 self.roots = preserved;
782 const b = try self.reduce(second.?, "NORMAL", self.eta, false);
783 try self.out.line("{s}", .{if (!a.success or !b.success) "unknown" else if (try self.equivalent(a.term, b.term)) "yes" else "no"});
784 } else if (equal(name, "debruijn")) {
785 var bound: std.ArrayList(Text) = .empty;
786 defer bound.deinit(std.heap.page_allocator);
787 try self.show(try deBruijn(&self.pool, first, &bound), true);
788 } else if (equal(name, "fv")) {
789 const names = try sortedKeys(self.scratch, (try freeVariables(first)).*);
790 if (names.len == 0) {
791 try self.out.line("(none)", .{});
792 } else {
793 for (names, 0..) |item, i| {
794 if (i > 0) try self.out.add(" ");
795 try self.out.add(item);
796 }
797 try self.out.add("\n");
798 }
799 } else if (equal(name, "size")) {
800 try self.out.line("{d}", .{termSize(first)});
801 } else if (equal(name, "depth")) {
802 try self.out.line("{d}", .{termDepth(first)});
803 } else if (equal(name, "type")) {
804 var inference: TypeInference = .{ .gpa = self.scratch };
805 const principal = inference.infer(first) catch |err| {
806 if (err != error.NotTypable) return err;
807 try self.out.line("not typable", .{});
808 return;
809 };
810 var text: Builder = .{ .gpa = std.heap.page_allocator };
811 defer text.bytes.deinit(text.gpa);
812 try inference.print(principal, &text);
813 try self.out.line("{s}", .{text.bytes.items});
814 } else if (equal(name, "church")) {
815 const result = try self.reduce(first, "NORMAL", false, false);
816 if (!result.success) {
817 try self.out.line("no normal form within {d} steps", .{self.limit});
818 } else {
819 var text: Builder = .{ .gpa = std.heap.page_allocator };
820 defer text.bytes.deinit(text.gpa);
821 try churchInto(&self.pool, try canonical(&self.pool, result.term, false), &text);
822 try self.out.line("{s}", .{text.bytes.items});
823 }
824 } else if (equal(name, "ski")) {
825 try self.show(try ski(&self.pool, first), false);
826 } else if (equal(name, "skireduce")) {
827 var value = try ski(&self.pool, first);
828 var count: usize = 0;
829 while (true) {
830 const step = (try skiStep(&self.pool, value)) orelse {
831 const text = try printed(value, false);
832 defer std.heap.page_allocator.free(text);
833 try self.out.line("result: {s} (steps: {d})", .{ text, count });
834 break;
835 };
836 if (count == self.limit) {
837 try self.out.line("no normal form within {d} steps", .{self.limit});
838 break;
839 }
840 value = step.term;
841 count += 1;
842 if (self.steps) {
843 const text = try printed(value, false);
844 defer std.heap.page_allocator.free(text);
845 try self.out.line("step {d} ({s}): {s}", .{ count, step.kind, text });
846 }
847 try self.pool.collect(self.macros, self.roots, value);
848 }
849 } else if (equal(name, "kam")) {
850 try self.kam(first);
851 } else if (equal(name, "reduce")) {
852 const result = try self.reduce(first, strategy, self.eta, self.steps);
853 if (!result.success) {
854 try self.out.line("no normal form within {d} steps", .{self.limit});
855 } else {
856 const text = try printed(try canonical(&self.pool, result.term, false), false);
857 defer std.heap.page_allocator.free(text);
858 try self.out.line("result: {s} (steps: {d})", .{ text, result.count });
859 }
860 }
861 }
862};
863fn validMacro(name: Text) bool {
864 if (name.len == 0 or !upper(name[0])) return false;
865 for (name) |c| if (!upper(c) and !digit(c) and c != '_') return false;
866 return true;
867}

869pub fn main(init: std.process.Init) !void {
870 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
871 defer arena.deinit();
872 const gpa = arena.allocator();
873 var inputBuffer: [8192]u8 = undefined;
874 var input = std.Io.File.stdin().readerStreaming(init.io, &inputBuffer);
875 const text = try input.interface.allocRemaining(gpa, .unlimited);
876 var lines = std.mem.splitScalar(u8, text, '\n');
877 var out: Builder = .{ .gpa = std.heap.page_allocator };
878 defer out.bytes.deinit(out.gpa);
879 var bench: Workbench = .{ .gpa = gpa, .scratch = gpa, .out = &out, .pool = .{ .gpa = gpa } };
880 defer bench.pool.deinit();
881 try bench.initialize();
882 var outputBuffer: [8192]u8 = undefined;
883 var output = std.Io.File.stdout().writerStreaming(init.io, &outputBuffer);
884 while (lines.next()) |raw| {
885 const line = std.mem.trimEnd(u8, raw, "\r");
886 if (equal(line, "quit")) break;
887 var commandArena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
888 defer commandArena.deinit();
889 bench.scratch = commandArena.allocator();
890 bench.roots = &.{};
891 bench.command(line) catch |err| {
892 if (err != error.Invalid) return err;
893 try out.line("error: {s}", .{failure});
894 };
895 bench.roots = &.{};
896 try bench.pool.collect(bench.macros, &.{}, null);
897 try output.interface.writeAll(out.bytes.items);
898 out.bytes.clearRetainingCapacity();
899 }
900 try output.interface.flush();
901}
