21def poly_hash(text, base, modulus):
22 value = 0
23 for char in text:
24 value = (value * base + ord(char)) % modulus
25 return value

28def is_prime(number):
29 if number < 2:
30 return False
31 divisor = 2
32 while divisor * divisor <= number:
33 if number % divisor == 0:
34 return False
35 divisor += 1
36 return True

39def next_prime(number):
40 while not is_prime(number):
41 number += 1
42 return number

84def circular_runs(slots):
85 if None not in slots:
86 return [len(slots)]
87 start = slots.index(None)
88 runs = []
89 length = 0
90 for offset in range(1, len(slots) + 1):
91 if slots[(start + offset) % len(slots)] is None:
92 if length > 0:
93 runs.append(length)
94 length = 0
95 else:
96 length += 1
97 return runs

194class Table:
195 def __init__(self, mode, capacity):
196 self.mode = mode
197 self.capacity = capacity
198 self.probes = 0
199 self.rehashes = 0
200 self.inserts = 0
201 self.updates = 0
202 self.hits = 0
203 self.misses = 0
204 self.deletes = 0
205 self.absent = 0
206 self.clear()

208 def clear(self):
209 self.size = 0
210 self.tomb = 0
211 if self.mode == "CHAIN":
212 self.buckets = [[] for _ in range(self.capacity)]
213 self.slots = []
214 else:
215 self.buckets = []
216 self.slots = [None] * self.slot_count()

218 def slot_count(self):
219 return slots_for(self.mode, self.capacity)

221 def h1(self, key):
222 return poly_hash(key, 31, self.capacity)
223
224 def h2(self, key):
225 return poly_hash(key, 37, self.capacity)
226
227 def step(self, key):
228 return 1 + poly_hash(key, 37, self.capacity - 1)
229
230 def distance(self, item, place):
231 return (place - self.h1(item.key) + self.capacity) % self.capacity
232
233 def cuckoo_place(self, key, part):
234 if part == 1:
235 return self.h1(key)
236 return self.capacity + self.h2(key)

238 def position(self, place):
239 if self.mode != "CUCKOO":
240 return str(place)
241 if place < self.capacity:
242 return f"T1.{place}"
243 return f"T2.{place - self.capacity}"

245 def items(self):
246 if self.mode == "CHAIN":
247 return [item for chain in self.buckets for item in chain]
248 return [slot for slot in self.slots if holds_item(slot)]
249
250 def keys_at(self, cursor):
251 if self.mode == "CHAIN":
252 return [item.key for item in self.buckets[cursor]]
253 slot = self.slots[cursor]
254 return [slot.key] if holds_item(slot) else []

256 def costs(self):
257 return [self.search(item.key).probes for item in self.items()]
258
259 def runs(self):
260 if self.mode == "CHAIN":
261 return [len(chain) for chain in self.buckets if chain]
262 if self.mode == "CUCKOO":
263 first = circular_runs(self.slots[:self.capacity])
264 return first + circular_runs(self.slots[self.capacity:])
265 return circular_runs(self.slots)
266
267 def map_symbols(self):
268 if self.mode == "CHAIN":
269 return [gauge(len(chain)) if chain else "." for chain in self.buckets]
270 symbols = []
271 for place, slot in enumerate(self.slots):
272 if slot is None:
273 symbols.append(".")
274 elif slot is TOMBSTONE:
275 symbols.append("D")
276 elif self.mode == "CUCKOO":
277 symbols.append("#")
278 elif self.mode == "ROBINHOOD":
279 symbols.append(gauge(self.distance(slot, place)))
280 else:
281 symbols.append(gauge(self.search(slot.key).probes - 1))
282 return symbols

284 def search(self, key):
285 search = Search()
286 if self.mode == "CHAIN":
287 self.search_chain(key, search)
288 elif self.mode == "ROBINHOOD":
289 self.search_robinhood(key, search)
290 elif self.mode == "CUCKOO":
291 self.search_cuckoo(key, search)
292 else:
293 self.search_probing(key, search)
294 return search

296 def examine(self, search, place):
297 slot = self.slots[place]
298 search.probes += 1
299 search.path.append(f"{self.position(place)}:{content(slot)}")
300 return slot
301
302 def search_chain(self, key, search):
303 bucket = self.h1(key)
304 search.place = bucket
305 for offset, item in enumerate(self.buckets[bucket]):
306 search.probes += 1
307 search.path.append(f"{bucket}:{item.key}")
308 if item.key == key:
309 search.item = item
310 search.offset = offset
311 return
312
313 def search_probing(self, key, search):
314 start = self.h1(key)
315 stride = self.step(key)
316 for attempt in range(self.capacity):
317 if self.mode == "LINEAR":
318 place = (start + attempt) % self.capacity
319 elif self.mode == "QUADRATIC":
320 place = (start + attempt * attempt) % self.capacity
321 else:
322 place = (start + attempt * stride) % self.capacity
323 slot = self.examine(search, place)
324 if slot is None:
325 search.vacant = place
326 return
327 if slot is TOMBSTONE:
328 if search.tombstone is None:
329 search.tombstone = place
330 elif slot.key == key:
331 search.item = slot
332 search.place = place
333 return
334
335 def search_robinhood(self, key, search):
336 start = self.h1(key)
337 for attempt in range(self.capacity):
338 place = (start + attempt) % self.capacity
339 slot = self.examine(search, place)
340 if slot is None:
341 return
342 if slot.key == key:
343 search.item = slot
344 search.place = place
345 return
346 if self.distance(slot, place) < attempt:
347 return
348
349 def search_cuckoo(self, key, search):
350 for part in (1, 2):
351 place = self.cuckoo_place(key, part)
352 slot = self.examine(search, place)
353 if slot is not None and slot.key == key:
354 search.item = slot
355 search.place = place
356 return

358 def place(self, item):
359 placement = Placement()
360 if self.mode == "CHAIN":
361 placement.place = self.h1(item.key)
362 self.buckets[placement.place].append(item)
363 elif self.mode == "ROBINHOOD":
364 self.place_robinhood(item, placement)
365 elif self.mode == "CUCKOO":
366 self.place_cuckoo(item, placement)
367 else:
368 self.place_probing(item, placement)
369 if placement.success:
370 self.size += 1
371 return placement
372
373 def place_probing(self, item, placement):
374 search = self.search(item.key)
375 if search.tombstone is not None:
376 placement.place = search.tombstone
377 placement.reused = True
378 self.tomb -= 1
379 elif search.vacant is not None:
380 placement.place = search.vacant
381 else:
382 placement.success = False
383 return
384 self.slots[placement.place] = item
385
386 def place_robinhood(self, item, placement):
387 current = item
388 place = self.h1(item.key)
389 distance = 0
390 origin = None
391 while True:
392 occupant = self.slots[place]
393 if occupant is None or self.distance(occupant, place) < distance:
394 self.slots[place] = current
395 if origin is None:
396 placement.place = place
397 else:
398 placement.moves.append(f"SHIFT {current.key} {origin} -> {place}")
399 if occupant is None:
400 return
401 current = occupant
402 origin = place
403 distance = self.distance(occupant, place)
404 place = (place + 1) % self.capacity
405 distance += 1
406
407 def place_cuckoo(self, item, placement):
408 current = item
409 part = 1
410 while True:
411 place = self.cuckoo_place(current.key, part)
412 occupant = self.slots[place]
413 if occupant is None:
414 self.slots[place] = current
415 break
416 if len(placement.moves) == self.capacity:
417 placement.success = False
418 placement.homeless = current
419 return
420 self.slots[place] = current
421 placement.moves.append(f"KICK {occupant.key} {self.position(place)}")
422 current = occupant
423 part = 3 - part
424 first = self.cuckoo_place(item.key, 1)
425 if self.slots[first] is item:
426 placement.place = first
427 else:
428 placement.place = self.cuckoo_place(item.key, 2)

430 def placement_note(self, placement):
431 if self.mode == "ROBINHOOD":
432 return f" shifted={len(placement.moves)}"
433 if self.mode == "CUCKOO":
434 return f" kicks={len(placement.moves)}"
435 if placement.reused:
436 return " reused"
437 return ""

439 def remove(self, search):
440 self.size -= 1
441 place = search.place
442 if self.mode == "CHAIN":
443 del self.buckets[place][search.offset]
444 return []
445 if self.mode in PROBING_MODES:
446 self.slots[place] = TOMBSTONE
447 self.tomb += 1
448 return []
449 self.slots[place] = None
450 if self.mode == "CUCKOO":
451 return []
452 return self.backward_shift(place)
453
454 def backward_shift(self, place):
455 moves = []
456 current = (place + 1) % self.capacity
457 while True:
458 item = self.slots[current]
459 if item is None or self.distance(item, current) < 1:
460 return moves
461 target = (current - 1 + self.capacity) % self.capacity
462 self.slots[target] = item
463 self.slots[current] = None
464 moves.append(f"SHIFT {item.key} {current} -> {target}")
465 current = (current + 1) % self.capacity

467 def rebuild(self, items, mode, capacity):
468 table = Table(mode, capacity)
469 for item in items:
470 if not table.place(item).success:
471 return False
472 self.mode = table.mode
473 self.capacity = table.capacity
474 self.size = table.size
475 self.tomb = 0
476 self.buckets = table.buckets
477 self.slots = table.slots
478 return True

480 def auto_rebuild(self, reason, capacity, pending):
481 items = self.items()
482 if pending is not None:
483 items.append(pending)
484 old_capacity = self.capacity
485 while not self.rebuild(items, self.mode, capacity):
486 capacity = next_prime(2 * capacity)
487 self.rehashes += 1
488 return f"REHASH {reason} {old_capacity} -> {capacity}"

491 costs = self.costs()

494 f"probes={self.probes} rehashes={self.rehashes}"
495 )

542 def load_check(self, table, notices):
543 limit = self.max_load * table.slot_count()
544 if (table.size + table.tomb) * 100 <= limit:
545 return
546 if table.size * 200 > limit:
547 notices.append(table.auto_rebuild("load", next_prime(2 * table.capacity), None))
548 else:
549 notices.append(table.auto_rebuild("tomb", table.capacity, None))

551 def shrink_check(self, table, notices):
552 capacity = next_prime(table.capacity // 2)
553 if table.size * 100 >= self.min_load * table.slot_count():
554 return
555 if capacity >= table.capacity:
556 return
557 if table.size * 100 > self.max_load * slots_for(table.mode, capacity):
558 return
559 old_capacity = table.capacity
560 if table.rebuild(table.items(), table.mode, capacity):
561 table.rehashes += 1
562 notices.append(f"REHASH shrink {old_capacity} -> {capacity}")

564 def put(self, table, key, value, adding):
565 search = table.search(key)
566 table.probes += search.probes
567 outcome = Outcome(search)
568 name = "ADD" if adding else "PUT"
569 item = search.item
570 if item is not None:
571 old_value = item.value
572 item.value = old_value + value if adding else value
573 table.updates += 1
574 outcome.line = (
575 f"{name} {key}={item.value} UPDATED old={old_value} "
576 f"at={table.position(search.place)} probes={search.probes}"
577 )
578 return outcome
579 table.inserts += 1
580 item = Item(key, value)
581 placement = table.place(item)
582 outcome.moves = placement.moves
583 head = f"{name} {key}={value} NEW"
584 if placement.success:
585 outcome.line = (
586 f"{head} at={table.position(placement.place)} probes={search.probes}"
587 + table.placement_note(placement)
588 )
589 elif table.mode == "QUADRATIC":
590 outcome.line = f"{head} probes={search.probes} full"
591 capacity = next_prime(2 * table.capacity)
592 outcome.notices.append(table.auto_rebuild("full", capacity, item))
593 else:
594 outcome.line = f"{head} probes={search.probes} kicks={len(placement.moves)} failed"
595 capacity = next_prime(2 * table.capacity)
596 outcome.notices.append(table.auto_rebuild("kicks", capacity, placement.homeless))
597 self.load_check(table, outcome.notices)
598 return outcome

600 def delete(self, table, key):

603 outcome = Outcome(search)

605 table.absent += 1

607 return outcome

612 if table.mode == "ROBINHOOD":
613 outcome.line += f" shifted={len(outcome.moves)}"
614 self.shrink_check(table, outcome.notices)
615 return outcome

775 def top(self, arguments):
776 limit = int(arguments[0])
777 require(1 <= limit <= MAX_LISTED, "BAD ARGUMENT")
778 table = self.table

788 def scan(self, arguments):
789 cursor = int(arguments[0])
790 limit = int(arguments[1])
791 require(1 <= limit <= MAX_LISTED, "BAD ARGUMENT")
792 total = self.table.slot_count()
793 require(cursor < total, "BAD CURSOR")
794 keys = []
795 while cursor < total and len(keys) < limit:
796 keys.extend(self.table.keys_at(cursor))
797 cursor += 1
798 if cursor == total:
799 cursor = 0
800 self.emit(f"SCAN next={cursor}", *keys)

822 def bench(self, arguments):
823 capacity = next_prime(capacity_argument(arguments[0]))
824 count, length, seed = generation_arguments(arguments[1:])
825 keys = generate_keys(2 * count, length, seed)
826 inserted = keys[:count]
827 queries = keys[count:]
828 self.emit(f"BENCH cap={capacity} count={count} len={length} seed={seed}")
829 for mode in MODES:
830 trial = Table(mode, capacity)
831 for number, key in enumerate(inserted, 1):
832 self.put(trial, key, number, False)
833 misses = 0
834 miss_probes = 0
835 for key in queries:
836 search = trial.search(key)
837 if search.item is None:
838 misses += 1
839 miss_probes += search.probes
840 costs = trial.costs()
841 self.emit(
842 f"{mode} cap={trial.capacity} keys={trial.size} "
843 f"hit={two_decimal(sum(costs), trial.size)} misses={misses} "
844 f"miss={two_decimal(miss_probes, misses)} rehashes={trial.rehashes}"
845 )

847 def execute(self, line):
848 command, *arguments = line.split()
849 if command not in self.handlers:
850 self.emit("ERROR UNKNOWN COMMAND", command)
851 return
852 if self.table is None and command not in TABLELESS_COMMANDS:
853 self.emit("ERROR NO TABLE")
854 return
855 try:
856 self.handlers[command](arguments)
857 except CommandError as error:
858 self.emit("ERROR", error)
