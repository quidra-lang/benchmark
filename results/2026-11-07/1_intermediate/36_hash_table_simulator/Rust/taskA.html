5#[derive(Clone, Copy, PartialEq, Eq)]
6enum Mode { Chain, Linear, Quadratic, Double, RobinHood, Cuckoo }
7
8impl Mode {
9 const ALL: [Self; 6] = [Self::Chain, Self::Linear, Self::Quadratic, Self::Double, Self::RobinHood, Self::Cuckoo];
10 fn name(self) -> &'static str {
11 match self {
12 Self::Chain => "CHAIN", Self::Linear => "LINEAR", Self::Quadratic => "QUADRATIC",
13 Self::Double => "DOUBLE", Self::RobinHood => "ROBINHOOD", Self::Cuckoo => "CUCKOO",
14 }
15 }
16 fn parse(text: &str) -> Result<Self, String> {
17 Self::ALL.iter().copied().find(|mode| mode.name() == text).ok_or_else(|| format!("BAD MODE {text}"))
18 }
19}

21#[derive(Clone)]
22struct Item { key: String, value: i64 }
23impl fmt::Display for Item {
24 fn fmt(&self, output: &mut fmt::Formatter<'_>) -> fmt::Result { write!(output, "{}={}", self.key, self.value) }
25}
26#[derive(Clone, Default)]
27enum Slot { #[default] Empty, Tomb, Live(Item) }
28impl Slot {
29 fn content(&self) -> &str {
30 match self { Self::Empty => "-", Self::Tomb => "D", Self::Live(item) => &item.key }
31 }
32}

34struct Counters { probes: i64, rehashes: i64, inserts: i64, updates: i64, hits: i64, misses: i64, deletes: i64, absent: i64 }

49fn hash(key: &str, base: usize, modulus: usize) -> usize {
50 key.bytes().fold(0, |value, ch| (value * base + ch as usize) % modulus)
51}
52fn prime(value: usize) -> bool {
53 if value < 2 { return false; }
54 let mut divisor = 2;
55 while divisor <= value / divisor {
56 if value % divisor == 0 { return false; }
57 divisor += 1;
58 }
59 true
60}
61fn next_prime(mut value: usize) -> usize {
62 value = value.max(2);
63 while !prime(value) { value += 1; }
64 value
65}
66fn decimal(numerator: i64, denominator: i64) -> String {
67 let value = if denominator == 0 { 0 } else { 100 * numerator / denominator };
68 format!("{}.{:02}", value / 100, value % 100)
69}
70fn gauge(value: usize) -> char { if value < 10 { (b'0' + value as u8) as char } else { '+' } }
71fn histogram(values: impl IntoIterator<Item = usize>) -> Vec<String> {
72 let mut frequencies = BTreeMap::<usize, usize>::new();
73 for value in values { *frequencies.entry(value).or_default() += 1; }
74 frequencies.into_iter().map(|(value, count)| {
75 format!("{value} {count} {}{}", "#".repeat(count.min(50)), if count > 50 { "+" } else { "" })
76 }).collect()
77}
78fn generate(count: usize, length: usize, mut seed: i64) -> Vec<String> {
79 (0..count).map(|_| (0..length).map(|_| {
80 seed = 48271 * seed % 2147483647;
81 (b'a' + (seed % 26) as u8) as char
82 }).collect()).collect()
83}
84fn capacity(value: i64) -> Result<usize, String> {
85 if !(2..=10000).contains(&value) { return Err("BAD CAPACITY".into()); }
86 Ok(value as usize)
87}
88fn generation(count: i64, length: i64, seed: i64) -> Result<(usize, usize, i64), String> {
89 if !(1..=500).contains(&count) || !(1..=10).contains(&length) || !(1..=2147483646).contains(&seed) {
90 return Err("BAD ARGUMENT".into());
91 }
92 Ok((count as usize, length as usize, seed))
93}

95struct Table {
96 mode: Mode,
97 cap: usize,
98 size: usize,
99 tomb: usize,
100 buckets: Vec<Vec<Item>>,
101 slots: Vec<Slot>,
102 counters: Counters,
103}
104
105impl Table {
106 fn new(mode: Mode, cap: usize) -> Self {
107 Self {
108 mode, cap, size: 0, tomb: 0, counters: Counters::default(),
109 buckets: if mode == Mode::Chain { vec![Vec::new(); cap] } else { Vec::new() },
110 slots: if mode == Mode::Chain { Vec::new() } else { vec![Slot::Empty; cap * if mode == Mode::Cuckoo { 2 } else { 1 }] },
111 }
112 }
113 fn slot_count(&self) -> usize { self.cap * if self.mode == Mode::Cuckoo { 2 } else { 1 } }
114 fn position(&self, position: usize) -> String {
115 if self.mode == Mode::Cuckoo { format!("T{}.{}", position / self.cap + 1, position % self.cap) }
116 else { position.to_string() }
117 }
118 fn distance(&self, item: &Item, position: usize) -> usize { (position + self.cap - hash(&item.key, 31, self.cap)) % self.cap }
119 fn items(&self) -> Vec<Item> {
120 if self.mode == Mode::Chain { self.buckets.iter().flatten().cloned().collect() }
121 else { self.slots.iter().filter_map(|slot| if let Slot::Live(item) = slot { Some(item.clone()) } else { None }).collect() }
122 }
123 fn observe(&self, search: &mut Search, position: usize, content: &str, trace: bool) {
124 search.probes += 1;
125 if trace { search.path.push(format!("{}:{content}", self.position(position))); }
126 }
127 fn search(&self, key: &str, trace: bool) -> Search {
128 let mut result = Search::default();
129 let home = hash(key, 31, self.cap);
130 if self.mode == Mode::Chain {
131 for (index, item) in self.buckets[home].iter().enumerate() {
132 self.observe(&mut result, home, &item.key, trace);
133 if item.key == key { result.found = Some(Location { position: home, chain_index: index }); break; }
134 }
135 return result;
136 }
137 let step = if self.mode == Mode::Double { 1 + hash(key, 37, self.cap - 1) } else { 1 };
138 let limit = if self.mode == Mode::Cuckoo { 2 } else { self.cap };
139 for i in 0..limit {
140 let position = if self.mode == Mode::Cuckoo {
141 if i == 0 { home } else { self.cap + hash(key, 37, self.cap) }
142 } else { (home + if self.mode == Mode::Quadratic { i * i } else { i * step }) % self.cap };
143 let slot = &self.slots[position];
144 self.observe(&mut result, position, slot.content(), trace);
145 if let Slot::Live(item) = slot {
146 if item.key == key { result.found = Some(Location { position, chain_index: 0 }); break; }
147 }
148 if self.mode == Mode::Cuckoo { continue; }
149 match slot {
150 Slot::Empty => { result.available.get_or_insert(position); break; }
151 Slot::Tomb => { result.available.get_or_insert(position); }
152 Slot::Live(item) => {
153 if self.mode == Mode::RobinHood && self.distance(item, position) < i { break; }
154 }
155 }
156 }
157 result
158 }
159 fn item(&self, location: Location) -> &Item {
160 if self.mode == Mode::Chain { &self.buckets[location.position][location.chain_index] }
161 else if let Slot::Live(item) = &self.slots[location.position] { item }
162 else { unreachable!() }
163 }
164 fn place(&mut self, item: Item, initial: Option<&Search>, trace: bool) -> Placement {
165 let mut result = Placement::default();
166 let home = hash(&item.key, 31, self.cap);
167 if self.mode == Mode::Chain {
168 self.buckets[home].push(item);
169 result.position = Some(home);
170 return result;
171 }
172 if matches!(self.mode, Mode::Linear | Mode::Quadratic | Mode::Double) {
173 let available = match initial { Some(search) => search.available, None => self.search(&item.key, false).available };
174 if let Some(position) = available {
175 result.reused = matches!(self.slots[position], Slot::Tomb);
176 if result.reused { self.tomb -= 1; }
177 self.slots[position] = Slot::Live(item);
178 result.position = Some(position);
179 } else { result.pending = Some(item); }
180 return result;
181 }
182 let original_key = item.key.clone();
183 let mut carried = item;
184 if self.mode == Mode::RobinHood {
185 let (mut position, mut traveled) = (home, 0);
186 let mut from: Option<usize> = None;
187 loop {
188 if matches!(self.slots[position], Slot::Empty) {
189 if trace {
190 if let Some(source) = from { result.moves.push(format!("SHIFT {} {source} -> {position}", carried.key)); }
191 }
192 self.slots[position] = Slot::Live(carried);
193 result.position.get_or_insert(position);
194 return result;
195 }
196 let resident_distance = match &self.slots[position] {
197 Slot::Live(resident) => self.distance(resident, position), _ => unreachable!(),
198 };
199 if resident_distance < traveled {
200 if trace {
201 if let Some(source) = from { result.moves.push(format!("SHIFT {} {source} -> {position}", carried.key)); }
202 }
203 let Slot::Live(displaced) = std::mem::replace(&mut self.slots[position], Slot::Live(carried)) else { unreachable!() };
204 carried = displaced;
205 result.position.get_or_insert(position);
206 from = Some(position);
207 traveled = resident_distance;
208 result.movements += 1;
209 }
210 position = (position + 1) % self.cap;
211 traveled += 1;
212 }
213 }
214 let mut part = 0;
215 loop {
216 let position = part * self.cap + hash(&carried.key, if part == 0 { 31 } else { 37 }, self.cap);
217 if matches!(self.slots[position], Slot::Empty) {
218 self.slots[position] = Slot::Live(carried);
219 result.position = Some(self.search(&original_key, false).found.unwrap().position);
220 return result;
221 }
222 if result.movements == self.cap { result.pending = Some(carried); return result; }
223 let Slot::Live(displaced) = std::mem::replace(&mut self.slots[position], Slot::Live(carried)) else { unreachable!() };
224 if trace { result.moves.push(format!("KICK {} {}", displaced.key, self.position(position))); }
225 carried = displaced;
226 result.movements += 1;
227 part = 1 - part;
228 }
229 }
230 fn remove(&mut self, location: Location, trace: bool) -> (usize, Vec<String>) {
231 let mut shifted = 0;
232 let mut moves = Vec::new();
233 let position = location.position;
234 if self.mode == Mode::Chain { self.buckets[position].remove(location.chain_index); }
235 else if matches!(self.mode, Mode::Linear | Mode::Quadratic | Mode::Double) {
236 self.slots[position] = Slot::Tomb;
237 self.tomb += 1;
238 } else {
239 self.slots[position] = Slot::Empty;
240 if self.mode == Mode::RobinHood {
241 let mut source = (position + 1) % self.cap;
242 while let Slot::Live(item) = &self.slots[source] {
243 if self.distance(item, source) == 0 { break; }
244 let destination = (source + self.cap - 1) % self.cap;
245 if trace { moves.push(format!("SHIFT {} {source} -> {destination}", item.key)); }
246 self.slots[destination] = std::mem::replace(&mut self.slots[source], Slot::Empty);
247 shifted += 1;
248 source = (source + 1) % self.cap;
249 }
250 }
251 }
252 self.size -= 1;
253 (shifted, moves)
254 }
255 fn rebuilt(mode: Mode, cap: usize, items: &[Item]) -> Option<Self> {
256 let mut table = Self::new(mode, cap);
257 for item in items {
258 table.place(item.clone(), None, false).position?;
259 table.size += 1;
260 }
261 Some(table)
262 }
263 fn adopt(&mut self, mut replacement: Self) {
264 replacement.counters = std::mem::take(&mut self.counters);
265 *self = replacement;
266 }
267 fn automatic(&mut self, mut cap: usize, reason: &str, pending: Option<Item>) -> String {
268 let old = self.cap;
269 let mut items = self.items();
270 if let Some(item) = pending { items.push(item); }
271 loop {
272 if let Some(replacement) = Self::rebuilt(self.mode, cap, &items) {
273 self.adopt(replacement);
274 self.counters.rehashes += 1;
275 return format!("REHASH {reason} {old} -> {cap}");
276 }
277 cap = next_prime(2 * cap);
278 }
279 }
280 fn load_check(&mut self, maximum: usize) -> Option<String> {
281 let threshold = maximum * self.slot_count();
282 if (self.size + self.tomb) * 100 <= threshold { return None; }
283 let grow = self.size * 200 > threshold;
284 Some(self.automatic(if grow { next_prime(2 * self.cap) } else { self.cap }, if grow { "load" } else { "tomb" }, None))
285 }
286 fn shrink_check(&mut self, minimum: usize, maximum: usize) -> Option<String> {
287 let cap = next_prime(self.cap / 2);
288 let slots = cap * if self.mode == Mode::Cuckoo { 2 } else { 1 };
289 if self.size * 100 >= minimum * self.slot_count() || cap >= self.cap || self.size * 100 > maximum * slots { return None; }
290 let replacement = Self::rebuilt(self.mode, cap, &self.items())?;
291 let old = self.cap;
292 self.adopt(replacement);
293 self.counters.rehashes += 1;
294 Some(format!("REHASH shrink {old} -> {cap}"))
295 }
296 fn path_line(search: &Search) -> String {
297 format!("PATH {}", if search.path.is_empty() { "none".to_owned() } else { search.path.join(" ") })
298 }
299 fn put(&mut self, item: Item, add: bool, maximum: usize, reporting: Reporting) -> (bool, Vec<String>) {
300 let search = self.search(&item.key, reporting.trace());
301 self.counters.probes += search.probes as i64;
302 let command = if add { "ADD" } else { "PUT" };
303 let mut lines = Vec::new();
304 if let Some(location) = search.found {
305 let old = self.item(location).value;
306 let stored = Item { key: item.key, value: if add { old + item.value } else { item.value } };
307 if reporting.details() {
308 lines.push(format!("{command} {stored} UPDATED old={old} at={} probes={}", self.position(location.position), search.probes));
309 }
310 if self.mode == Mode::Chain { self.buckets[location.position][location.chain_index] = stored; }
311 else { self.slots[location.position] = Slot::Live(stored); }
312 self.counters.updates += 1;
313 if reporting.trace() { lines.push(Self::path_line(&search)); }
314 return (false, lines);
315 }
316 let mut row = if reporting.details() { format!("{command} {item} NEW") } else { String::new() };
317 let placed = self.place(item, Some(&search), reporting.trace());
318 self.counters.inserts += 1;
319 if reporting.details() {
320 if let Some(position) = placed.position {
321 row.push_str(&format!(" at={} probes={}", self.position(position), search.probes));
322 if placed.reused { row.push_str(" reused"); }
323 if self.mode == Mode::RobinHood { row.push_str(&format!(" shifted={}", placed.movements)); }
324 if self.mode == Mode::Cuckoo { row.push_str(&format!(" kicks={}", placed.movements)); }
325 } else {
326 row.push_str(&format!(" probes={}", search.probes));
327 if self.mode == Mode::Quadratic { row.push_str(" full"); }
328 else { row.push_str(&format!(" kicks={} failed", placed.movements)); }
329 }
330 lines.push(row);
331 }
332 if reporting.trace() { lines.push(Self::path_line(&search)); lines.extend(placed.moves); }
333 if placed.position.is_some() { self.size += 1; }
334 else {
335 let rehash = self.automatic(next_prime(2 * self.cap), if self.mode == Mode::Quadratic { "full" } else { "kicks" }, placed.pending);
336 if reporting != Reporting::Silent { lines.push(rehash); }
337 }
338 if let Some(rehash) = self.load_check(maximum) {
339 if reporting != Reporting::Silent { lines.push(rehash); }
340 }
341 (true, lines)
342 }
343 fn delete(&mut self, key: &str, minimum: usize, maximum: usize, reporting: Reporting) -> (bool, Vec<String>) {
344 let search = self.search(key, reporting.trace());
345 self.counters.probes += search.probes as i64;
346 let mut lines = Vec::new();
347 if let Some(location) = search.found {
348 let (shifted, moves) = self.remove(location, reporting.trace());
349 self.counters.deletes += 1;
350 if reporting.details() {
351 let mut row = format!("DEL {key} OK at={} probes={}", self.position(location.position), search.probes);
352 if self.mode == Mode::RobinHood { row.push_str(&format!(" shifted={shifted}")); }
353 lines.push(row);
354 }
355 if reporting.trace() { lines.push(Self::path_line(&search)); lines.extend(moves); }
356 if let Some(rehash) = self.shrink_check(minimum, maximum) {
357 if reporting != Reporting::Silent { lines.push(rehash); }
358 }
359 (true, lines)
360 } else {
361 self.counters.absent += 1;
362 if reporting.details() { lines.push(format!("DEL {key} NOT FOUND probes={}", search.probes)); }
363 if reporting.trace() { lines.push(Self::path_line(&search)); }
364 (false, lines)
365 }
366 }
367 fn get(&mut self, key: &str, trace: bool) -> Vec<String> {
368 let search = self.search(key, trace);
369 self.counters.probes += search.probes as i64;
370 let row = if let Some(location) = search.found {
371 self.counters.hits += 1;
372 format!("GET {} at={} probes={}", self.item(location), self.position(location.position), search.probes)
373 } else {
374 self.counters.misses += 1;
375 format!("GET {key} NOT FOUND probes={}", search.probes)
376 };
377 let mut lines = vec![row];
378 if trace { lines.push(Self::path_line(&search)); }
379 lines
380 }
381 fn records(&self) -> Vec<Record> {
382 let mut records = Vec::with_capacity(self.size);
383 if self.mode == Mode::Chain {
384 for (position, bucket) in self.buckets.iter().enumerate() {
385 for (index, item) in bucket.iter().enumerate() { records.push(Record { item: item.clone(), position, cost: index + 1 }); }
386 }
387 } else {
388 for (position, slot) in self.slots.iter().enumerate() {
389 if let Slot::Live(item) = slot {
390 let cost = match self.mode {
391 Mode::Cuckoo => position / self.cap + 1,
392 Mode::Linear | Mode::RobinHood => self.distance(item, position) + 1,
393 _ => self.search(&item.key, false).probes,
394 };
395 records.push(Record { item: item.clone(), position, cost });
396 }
397 }
398 }
399 records
400 }
401 fn metrics(&self) -> (usize, String) {
402 let records = self.records();
403 (records.iter().map(|record| record.cost).max().unwrap_or(0),
404 decimal(records.iter().map(|record| record.cost as i64).sum(), self.size as i64))
405 }
406 fn dump(&self) -> Vec<String> {
407 let mut lines = vec![format!("DUMP {} cap={} size={} tomb={}", self.mode.name(), self.cap, self.size, self.tomb)];
408 if self.mode == Mode::Chain {
409 for (position, bucket) in self.buckets.iter().enumerate() {
410 let content = if bucket.is_empty() { "-".into() } else { bucket.iter().map(ToString::to_string).collect::<Vec<_>>().join(" ") };
411 lines.push(format!("{position} {content}"));
412 }
413 } else {
414 for (position, slot) in self.slots.iter().enumerate() {
415 let prefix = if self.mode == Mode::Cuckoo { format!("T{} {}", position / self.cap + 1, position % self.cap) } else { position.to_string() };
416 let content = match slot {
417 Slot::Live(item) if self.mode == Mode::RobinHood => format!("{item} d={}", self.distance(item, position)),
418 Slot::Live(item) => item.to_string(), _ => slot.content().to_owned(),
419 };
420 lines.push(format!("{prefix} {content}"));
421 }
422 }
423 lines
424 }
425 fn map(&self) -> Vec<String> {
426 let mut lines = vec![format!("MAP {} cap={}", self.mode.name(), self.cap)];
427 let mut costs = vec![0; self.slot_count()];
428 if self.mode != Mode::Chain { for record in self.records() { costs[record.position] = record.cost; } }
429 for part in 0..if self.mode == Mode::Cuckoo { 2 } else { 1 } {
430 for start in (0..self.cap).step_by(40) {
431 let mut text = String::new();
432 for index in start..(start + 40).min(self.cap) {
433 let ch = if self.mode == Mode::Chain {
434 if self.buckets[index].is_empty() { '.' } else { gauge(self.buckets[index].len()) }
435 } else {
436 let position = part * self.cap + index;
437 match &self.slots[position] {
438 Slot::Empty => '.', Slot::Tomb => 'D',
439 Slot::Live(_) if self.mode == Mode::Cuckoo => '#', Slot::Live(_) => gauge(costs[position] - 1),
440 }
441 };
442 text.push(ch);
443 }
444 let prefix = if self.mode == Mode::Cuckoo { format!("T{} ", part + 1) } else { String::new() };
445 lines.push(format!("{prefix}{start:05} {text}"));
446 }
447 }
448 lines
449 }
450 fn statistics(&self) -> Vec<String> {
451 let (longest, average) = self.metrics();
452 let c = &self.counters;
453 vec![format!("STATS {} size={} cap={} slots={} tomb={} load={}% longest={longest} avg={average} probes={} rehashes={}",
454 self.mode.name(), self.size, self.cap, self.slot_count(), self.tomb, decimal(100 * self.size as i64, self.slot_count() as i64), c.probes, c.rehashes),
455 format!("OPS inserts={} updates={} hits={} misses={} deletes={} absent={}", c.inserts, c.updates, c.hits, c.misses, c.deletes, c.absent)]
456 }
457 fn scan(&self, mut cursor: usize, limit: usize) -> String {
458 let mut keys = Vec::new();
459 while cursor < self.slot_count() && keys.len() < limit {
460 if self.mode == Mode::Chain { keys.extend(self.buckets[cursor].iter().map(|item| item.key.as_str())); }
461 else if let Slot::Live(item) = &self.slots[cursor] { keys.push(item.key.as_str()); }
462 cursor += 1;
463 }
464 if cursor == self.slot_count() { cursor = 0; }
465 let mut line = format!("SCAN next={cursor}");
466 for key in keys { line.push(' '); line.push_str(key); }
467 line
468 }
469 fn clusters(&self) -> Vec<String> {
470 let mut runs = Vec::new();
471 if self.mode == Mode::Chain { runs.extend(self.buckets.iter().filter(|bucket| !bucket.is_empty()).map(Vec::len)); }
472 else {
473 for part in self.slots.chunks(self.cap) {
474 if let Some(empty) = part.iter().position(|slot| matches!(slot, Slot::Empty)) {
475 let mut length = 0;
476 for offset in 1..=self.cap {
477 if !matches!(part[(empty + offset) % self.cap], Slot::Empty) { length += 1; }
478 else if length > 0 { runs.push(length); length = 0; }
479 }
480 } else { runs.push(self.cap); }
481 }
482 }
483 let mut lines = vec![format!("CLUSTERS runs={} longest={}", runs.len(), runs.iter().max().unwrap_or(&0))];
484 lines.extend(histogram(runs));
485 lines
486 }
487 fn distribution(&self) -> Vec<String> {
488 let values: Vec<_> = self.records().into_iter().map(|record| record.cost).collect();
489 let mut lines = vec![format!("DIST keys={} longest={}", self.size, values.iter().max().unwrap_or(&0))];
490 lines.extend(histogram(values));
491 lines
492 }
493}

495fn compare(table: &Table, cap: usize, maximum: usize) -> Vec<String> {
496 let mut items = table.items();
497 items.sort_by(|a, b| a.key.cmp(&b.key));
498 let mut lines = vec![format!("COMPARE keys={} cap={cap}", items.len())];
499 for mode in Mode::ALL {
500 let mut trial = Table::new(mode, cap);
501 for item in &items { trial.put(item.clone(), false, maximum, Reporting::Silent); }
502 let (longest, average) = trial.metrics();
503 lines.push(format!("{} cap={} longest={longest} avg={average} probes={} rehashes={}",
504 mode.name(), trial.cap, trial.counters.probes, trial.counters.rehashes));
505 }
506 lines
507}
508fn benchmark(cap: usize, count: usize, length: usize, seed: i64, maximum: usize) -> Vec<String> {
509 let keys = generate(2 * count, length, seed);
510 let mut lines = vec![format!("BENCH cap={cap} count={count} len={length} seed={seed}")];
511 for mode in Mode::ALL {
512 let mut trial = Table::new(mode, cap);
513 for (index, key) in keys[..count].iter().enumerate() {
514 trial.put(Item { key: key.clone(), value: index as i64 + 1 }, false, maximum, Reporting::Silent);
515 }
516 let (mut misses, mut probes) = (0, 0);
517 for key in &keys[count..] {
518 let search = trial.search(key, false);
519 if search.found.is_none() { misses += 1; probes += search.probes as i64; }
520 }
521 lines.push(format!("{} cap={} keys={} hit={} misses={misses} miss={} rehashes={}",
522 mode.name(), trial.cap, trial.size, trial.metrics().1, decimal(probes, misses), trial.counters.rehashes));
523 }
524 lines
525}
526
527struct Simulator { table: Option<Table>, maximum: usize, minimum: usize, trace: bool }
528impl Simulator {
529 fn execute(&mut self, line: &str) -> Result<Vec<String>, String> {
530 let words: Vec<_> = line.split(' ').collect();
531 let command = words[0];
532 let number = |index: usize| words[index].parse::<i64>().unwrap();
533 if !matches!(command, "CREATE" | "MAXLOAD" | "MINLOAD" | "TRACE" | "HASH" | "PUT" | "ADD" | "GET" | "DEL" |
534 "PROBE" | "GEN" | "GENDEL" | "RESIZE" | "CONVERT" | "CLEAR" | "DUMP" | "MAP" | "STATS" | "KEYS" |
535 "TOP" | "SCAN" | "CLUSTERS" | "DIST" | "COMPARE" | "BENCH") {
536 return Err(format!("UNKNOWN COMMAND {command}"));
537 }
538 match command {
539 "CREATE" => {
540 let mode = Mode::parse(words[1])?;
541 let cap = next_prime(capacity(number(2))?);
542 self.table = Some(Table::new(mode, cap));
543 return Ok(vec![format!("CREATE {} cap={cap}", mode.name())]);
544 }
545 "MAXLOAD" | "MINLOAD" => {
546 let value = number(1);
547 let range = if command == "MAXLOAD" { 10..=95 } else { 0..=40 };
548 if !range.contains(&value) { return Err("BAD LOAD".into()); }
549 if command == "MAXLOAD" { self.maximum = value as usize; } else { self.minimum = value as usize; }
550 return Ok(vec![format!("{command} {value}")]);
551 }
552 "TRACE" => {
553 if !matches!(words[1], "ON" | "OFF") { return Err("BAD ARGUMENT".into()); }
554 self.trace = words[1] == "ON";
555 return Ok(vec![format!("TRACE {}", words[1])]);
556 }
557 "HASH" => {
558 let cap = capacity(number(2))?;
559 let key = words[1];
560 return Ok(vec![format!("HASH {key} m={cap} h31={} h37={} step={}", hash(key, 31, cap), hash(key, 37, cap), 1 + hash(key, 37, cap - 1))]);
561 }
562 "BENCH" => {
563 let cap = next_prime(capacity(number(1))?);
564 let (count, length, seed) = generation(number(2), number(3), number(4))?;
565 return Ok(benchmark(cap, count, length, seed, self.maximum));
566 }
567 _ => {}
568 }
569 let table = self.table.as_mut().ok_or_else(|| "NO TABLE".to_owned())?;
570 let lines = match command {
571 "PUT" | "ADD" => table.put(Item { key: words[1].into(), value: number(2) }, command == "ADD", self.maximum, Reporting::Full(self.trace)).1,
572 "GET" => table.get(words[1], self.trace),
573 "DEL" => table.delete(words[1], self.minimum, self.maximum, Reporting::Full(self.trace)).1,
574 "PROBE" => {
575 let search = table.search(words[1], true);
576 let mut row = format!("PROBE {}", words[1]);
577 for element in search.path { row.push(' '); row.push_str(&element); }
578 row.push_str(&format!(" {} probes={}", if search.found.is_some() { "FOUND" } else { "MISS" }, search.probes));
579 vec![row]
580 }
581 "GEN" | "GENDEL" => {
582 let (count, length, seed) = generation(number(1), number(2), number(3))?;
583 let mut lines = Vec::new();
584 let mut changed = 0;
585 for (index, key) in generate(count, length, seed).into_iter().enumerate() {
586 let (did_change, output) = if command == "GEN" {
587 table.put(Item { key, value: index as i64 + 1 }, false, self.maximum, Reporting::Rehashes)
588 } else { table.delete(&key, self.minimum, self.maximum, Reporting::Rehashes) };
589 if did_change { changed += 1; }
590 lines.extend(output);
591 }
592 lines.push(if command == "GEN" { format!("GEN {count} new={changed} updated={}", count - changed) }
593 else { format!("GENDEL {count} deleted={changed} missing={}", count - changed) });
594 lines
595 }
596 "RESIZE" | "CONVERT" => {
597 let cap = if command == "RESIZE" { next_prime(capacity(number(1))?) } else { table.cap };
598 let mode = if command == "CONVERT" { Mode::parse(words[1])? } else { table.mode };
599 let slots = cap * if mode == Mode::Cuckoo { 2 } else { 1 };
600 if table.size * 100 > self.maximum * slots { return Err("TOO SMALL".into()); }
601 let replacement = Table::rebuilt(mode, cap, &table.items()).ok_or_else(|| "REBUILD FAILED".to_owned())?;
602 let row = if command == "RESIZE" { format!("RESIZE {} -> {cap}", table.cap) }
603 else { format!("CONVERT {} -> {} cap={cap}", table.mode.name(), mode.name()) };
604 table.adopt(replacement);
605 vec![row]
606 }
607 "CLEAR" => {
608 let row = format!("CLEAR {}", table.size);
609 table.adopt(Table::new(table.mode, table.cap));
610 vec![row]
611 }
612 "DUMP" => table.dump(), "MAP" => table.map(), "STATS" => table.statistics(),
613 "KEYS" => {
614 let mut keys: Vec<_> = table.items().into_iter().map(|item| item.key).collect();
615 keys.sort();
616 let mut row = format!("KEYS {}", keys.len());
617 for key in keys { row.push(' '); row.push_str(&key); }
618 vec![row]
619 }
620 "TOP" => {
621 let limit = number(1);
622 if !(1..=100).contains(&limit) { return Err("BAD ARGUMENT".into()); }
623 let mut records = table.records();
624 records.sort_by(|a, b| b.cost.cmp(&a.cost).then_with(|| a.item.key.cmp(&b.item.key)));
625 records.truncate(limit as usize);
626 let mut lines = vec![format!("TOP {}", records.len())];
627 lines.extend(records.into_iter().map(|record| format!("{} cost={} at={}", record.item.key, record.cost, table.position(record.position))));
628 lines
629 }
630 "SCAN" => {
631 let (cursor, limit) = (number(1), number(2));
632 if !(1..=100).contains(&limit) { return Err("BAD ARGUMENT".into()); }
633 if cursor >= table.slot_count() as i64 { return Err("BAD CURSOR".into()); }
634 vec![table.scan(cursor as usize, limit as usize)]
635 }
636 "CLUSTERS" => table.clusters(), "DIST" => table.distribution(),
637 "COMPARE" => compare(table, next_prime(capacity(number(1))?), self.maximum),
638 _ => unreachable!(),
639 };
640 Ok(lines)
641 }
642}
643
644fn main() {
645 let stdin = io::stdin();
646 let mut input = stdin.lock().lines();
647 let Some(Ok(first)) = input.next() else { return; };
648 let count: usize = first.parse().unwrap();
649 let mut simulator = Simulator { table: None, maximum: 75, minimum: 0, trace: false };
650 let stdout = io::stdout();
651 let mut output = io::BufWriter::new(stdout.lock());
652 for line in input.take(count) {
653 match simulator.execute(&line.unwrap()) {
654 Ok(lines) => for line in lines { writeln!(output, "{line}").unwrap(); },
655 Err(reason) => writeln!(output, "ERROR {reason}").unwrap(),
656 }
657 }
658}
