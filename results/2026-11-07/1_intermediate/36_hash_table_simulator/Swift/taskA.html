46struct Item {

53enum Slot {
54 case empty, item(Item), tombstone
55 var entry: Item? {
56 if case let .item(item) = self {
57 return item
58 }
59 return nil
60 }
61 var isEmpty: Bool {
62 if case .empty = self {
63 return true
64 }
65 return false
66 }
67 var isTombstone: Bool {
68 if case .tombstone = self {
69 return true
70 }
71 return false

77func hash(_ key: String, _ base: Int, _ modulus: Int) -> Int {
78 key.utf8.reduce(0) { ($0 * base + Int($1)) % modulus }
79}
80func prime(_ value: Int) -> Int {
81 var candidate = max(2, value)
82 while true {
83 var valid = true, divisor = 2
84 while divisor * divisor <= candidate {
85 if candidate % divisor == 0 {
86 valid = false
87 break
88 }
89 divisor += 1
90 }
91 if valid {
92 return candidate
93 }
94 candidate += 1
95 }
96}
97func padded(_ value: Int, _ width: Int) -> String {
98 let text = String(value)
99 return String(repeating: "0", count: max(0, width - text.count)) + text
100}
101func decimal(_ numerator: Int, _ denominator: Int) -> String {
102 let value = denominator == 0 ? 0 : 100 * numerator / denominator
103 return "\(value / 100)." + padded(value % 100, 2)
104}
105func gauge(_ value: Int) -> Character {
106 value < 10 ? Character(String(value)) : "+"
107}
108func generatedKeys(_ count: Int, _ length: Int, _ seed: Int) -> [String] {
109 var state = seed, keys: [String] = []
110 for _ in 0..<count {
111 var bytes: [UInt8] = []
112 for _ in 0..<length {
113 state = 48271 * state % 2147483647
114 bytes.append(UInt8(97 + state % 26))
115 }
116 keys.append(String(decoding: bytes, as: UTF8.self))
117 }
118 return keys
119}
120func histogram(_ values: [Int]) -> [String] {
121 var counts: [Int: Int] = [:]
122 for value in values {
123 counts[value, default: 0] += 1
124 }
125 return counts.keys.sorted().map { value in
126 let count = counts[value]!
127 return "\(value) \(count) " + String(repeating: "#", count: min(count, 50)) + (count > 50 ? "+" : "")
128 }
129}
130func runs(_ slots: [Slot], start: Int, count: Int) -> [Int] {
131 guard let empty = (0..<count).first(where: { slots[start + $0].isEmpty }) else {
132 return [count]
133 }
134 var lengths: [Int] = [], length = 0
135 for offset in 1...count {
136 if !slots[start + (empty + offset) % count].isEmpty {
137 length += 1
138 } else if length > 0 {
139 lengths.append(length)
140 length = 0
141 }
142 }
143 return lengths
144}

177final class Table {
178 private(set) var mode: Mode
179 private(set) var cap: Int
180 private(set) var size = 0, tomb = 0
181 private(set) var counters = Counters()
182 private var buckets: [[Item]]
183 private var slots: [Slot]
184 init(_ mode: Mode, _ capacity: Int) {
185 self.mode = mode
186 cap = capacity
187 buckets = mode == .chain ? Array(repeating: [], count: capacity) : []
188 slots = mode == .chain ? [] : Array(repeating: .empty, count: capacity * mode.parts)
189 }
190 var slotCount: Int {
191 cap * mode.parts
192 }
193 func position(_ index: Int) -> String {
194 mode == .cuckoo ? "T\(index / cap + 1).\(index % cap)" : String(index)
195 }
196 func distance(_ key: String, _ position: Int) -> Int {
197 (position - hash(key, 31, cap) + cap) % cap
198 }
199 func items() -> [Item] {
200 mode == .chain ? buckets.flatMap { $0 } : slots.compactMap { $0.entry }
201 }
202 func search(_ key: String, tracing: Bool = false) -> Search {
203 var result = Search()
204 let first = hash(key, 31, cap)
205 func examine(_ index: Int, _ slot: Slot) {
206 result.probes += 1
207 if tracing {
208 result.path.append("\(position(index)):\(slot.content)")
209 }
210 }
211 func found(_ item: Item, _ position: Int, _ chainIndex: Int = -1) {
212 result.item = item
213 result.position = position
214 result.chainIndex = chainIndex
215 }
216 if mode == .chain {
217 for (index, item) in buckets[first].enumerated() {
218 examine(first, .item(item))
219 if item.key == key {
220 found(item, first, index)
221 break
222 }
223 }
224 return result
225 }
226 if mode == .cuckoo {
227 for index in [first, cap + hash(key, 37, cap)] {
228 examine(index, slots[index])
229 if let item = slots[index].entry, item.key == key {
230 found(item, index)
231 break
232 }
233 }
234 return result
235 }
236 let step = mode == .double ? 1 + hash(key, 37, cap - 1) : 1
237 for i in 0..<cap {
238 let offset = mode == .quadratic ? i * i : i * step
239 let index = (first + offset) % cap, slot = slots[index]
240 examine(index, slot)
241 if slot.isEmpty {
242 if result.available == nil {
243 result.available = index
244 }
245 break
246 }
247 if slot.isTombstone {
248 if result.available == nil {
249 result.available = index
250 }
251 continue
252 }
253 let item = slot.entry!
254 if item.key == key {
255 found(item, index)
256 break
257 }
258 if mode == .robinHood && distance(item.key, index) < i {
259 break
260 }
261 }
262 return result
263 }
264 func place(_ item: Item, initial: Search? = nil, tracing: Bool = false) -> Placement {
265 var result = Placement()
266 if mode == .chain {
267 result.position = hash(item.key, 31, cap)
268 buckets[result.position].append(item)
269 result.success = true
270 return result
271 }
272 if mode.usesTombstones {
273 let found = initial ?? search(item.key)
274 guard let index = found.available else {
275 result.homeless = item
276 return result
277 }
278 result.position = index
279 result.reused = slots[index].isTombstone
280 slots[index] = .item(item)
281 if result.reused {
282 tomb -= 1
283 }
284 result.success = true
285 return result
286 }
287 var current = item
288 if mode == .robinHood {
289 var index = hash(item.key, 31, cap), dist = 0
290 var from: Int?
291 while true {
292 guard let occupant = slots[index].entry else {
293 slots[index] = .item(current)
294 if tracing, let from {
295 result.moves.append("SHIFT \(current.key) \(from) -> \(index)")
296 }
297 if result.position < 0 {
298 result.position = index
299 }
300 result.success = true
301 return result
302 }
303 let otherDistance = distance(occupant.key, index)
304 if otherDistance < dist {
305 slots[index] = .item(current)
306 if tracing, let from {
307 result.moves.append("SHIFT \(current.key) \(from) -> \(index)")
308 }
309 if result.position < 0 {
310 result.position = index
311 }
312 current = occupant
313 from = index
314 dist = otherDistance
315 result.shifted += 1
316 }
317 index = (index + 1) % cap
318 dist += 1
319 }
320 }
321 var part = 0
322 while true {
323 let index = part * cap + hash(current.key, part == 0 ? 31 : 37, cap)
324 guard let occupant = slots[index].entry else {
325 slots[index] = .item(current)
326 result.success = true
327 result.position = search(item.key).position
328 return result
329 }
330 if result.kicks == cap {
331 result.homeless = current
332 return result
333 }
334 slots[index] = .item(current)
335 result.kicks += 1
336 if tracing {
337 result.moves.append("KICK \(occupant.key) \(position(index))")
338 }
339 current = occupant
340 part = 1 - part
341 }
342 }
343 func remove(_ found: Search, tracing: Bool) -> Removal {
344 var result = Removal()
345 if mode == .chain {
346 buckets[found.position].remove(at: found.chainIndex)
347 } else if mode.usesTombstones {
348 slots[found.position] = .tombstone
349 tomb += 1
350 } else {
351 slots[found.position] = .empty
352 if mode == .robinHood {
353 var next = (found.position + 1) % cap
354 while let item = slots[next].entry, distance(item.key, next) >= 1 {
355 let destination = (next - 1 + cap) % cap
356 slots[destination] = .item(item)
357 slots[next] = .empty
358 if tracing {
359 result.moves.append("SHIFT \(item.key) \(next) -> \(destination)")
360 }
361 result.shifted += 1
362 next = (next + 1) % cap
363 }
364 }
365 }
366 size -= 1
367 return result
368 }
369 static func rebuild(_ items: [Item], mode: Mode, capacity: Int) -> Table? {
370 let table = Table(mode, capacity)
371 for item in items {
372 if !table.place(item).success {
373 return nil
374 }
375 table.size += 1
376 }
377 return table
378 }
379 func adopt(_ replacement: Table) {
380 mode = replacement.mode
381 cap = replacement.cap
382 size = replacement.size
383 tomb = replacement.tomb
384 buckets = replacement.buckets
385 slots = replacement.slots
386 }
387 func automatic(_ reason: String, capacity: Int, pending: Item? = nil) -> String {
388 var listed = items(), target = capacity
389 if let pending {
390 listed.append(pending)
391 }
392 let old = cap
393 while true {
394 if let replacement = Table.rebuild(listed, mode: mode, capacity: target) {
395 adopt(replacement)
396 counters.rehashes += 1
397 return "REHASH \(reason) \(old) -> \(cap)"
398 }
399 target = prime(2 * target)
400 }
401 }
402 func loadCheck(_ maxLoad: Int) -> [String] {
403 let threshold = maxLoad * slotCount
404 if 100 * (size + tomb) <= threshold {
405 return []
406 }
407 let grow = 200 * size > threshold
408 return [automatic(grow ? "load" : "tomb", capacity: grow ? prime(2 * cap) : cap)]
409 }
410 func shrinkCheck(_ minLoad: Int, _ maxLoad: Int) -> [String] {
411 let capacity = prime(cap / 2), newSlots = capacity * mode.parts
412 if 100 * size >= minLoad * slotCount || capacity >= cap || 100 * size > maxLoad * newSlots {
413 return []
414 }
415 guard let replacement = Table.rebuild(items(), mode: mode, capacity: capacity) else {
416 return []
417 }
418 let old = cap
419 adopt(replacement)
420 counters.rehashes += 1
421 return ["REHASH shrink \(old) -> \(capacity)"]
422 }
423 func put(_ key: String, _ value: Int, command: String, maxLoad: Int, tracing: Bool = false) -> PutResult {
424 let found = search(key, tracing: tracing)
425 let trace = tracing ? ["PATH " + (found.path.isEmpty ? "none" : found.path.joined(separator: " "))] : []
426 counters.probes += found.probes
427 if let old = found.item?.value {
428 let item = Item(key: key, value: command == "ADD" ? old + value : value)
429 if mode == .chain {
430 buckets[found.position][found.chainIndex] = item
431 } else {
432 slots[found.position] = .item(item)
433 }
434 counters.updates += 1
435 let row = "\(command) \(item.text) UPDATED old=\(old) at=\(position(found.position)) probes=\(found.probes)"
436 return PutResult(fresh: false, rehashes: [], rows: [row] + trace)
437 }
438 let item = Item(key: key, value: value), placed = place(item, initial: found, tracing: tracing)
439 size += 1
440 counters.inserts += 1
441 var row = "\(command) \(item.text) NEW"
442 if placed.success {
443 row += " at=\(position(placed.position)) probes=\(found.probes)"
444 if placed.reused {
445 row += " reused"
446 }
447 if mode == .robinHood {
448 row += " shifted=\(placed.shifted)"
449 }
450 if mode == .cuckoo {
451 row += " kicks=\(placed.kicks)"
452 }
453 } else {
454 row += " probes=\(found.probes)" + (mode == .cuckoo ? " kicks=\(placed.kicks) failed" : " full")
455 }
456 var rehashes: [String] = []
457 if !placed.success {
458 rehashes.append(automatic(
459 mode == .cuckoo ? "kicks" : "full",
460 capacity: prime(2 * cap),
461 pending: placed.homeless
462 ))
463 }
464 rehashes += loadCheck(maxLoad)
465 return PutResult(fresh: true, rehashes: rehashes, rows: [row] + trace + placed.moves + rehashes)
466 }
467 func get(_ key: String, tracing: Bool) -> [String] {
468 let found = search(key, tracing: tracing)
469 counters.probes += found.probes
470 var row: String
471 if let item = found.item {
472 counters.hits += 1
473 row = "GET \(item.text) at=\(position(found.position))"
474 } else {
475 counters.misses += 1
476 row = "GET \(key) NOT FOUND"
477 }
478 row += " probes=\(found.probes)"
479 return tracing ? [row, "PATH " + (found.path.isEmpty ? "none" : found.path.joined(separator: " "))] : [row]
480 }
481 func delete(_ key: String, minLoad: Int, maxLoad: Int, tracing: Bool = false) -> DeleteResult {
482 let found = search(key, tracing: tracing)
483 let trace = tracing ? ["PATH " + (found.path.isEmpty ? "none" : found.path.joined(separator: " "))] : []
484 counters.probes += found.probes
485 if found.item == nil {
486 counters.absent += 1
487 return DeleteResult(
488 deleted: false,
489 rehashes: [],
490 rows: ["DEL \(key) NOT FOUND probes=\(found.probes)"] + trace
491 )
492 }
493 let removed = remove(found, tracing: tracing)
494 counters.deletes += 1
495 var row = "DEL \(key) OK at=\(position(found.position)) probes=\(found.probes)"
496 if mode == .robinHood {
497 row += " shifted=\(removed.shifted)"
498 }
499 let rehashes = shrinkCheck(minLoad, maxLoad)
500 return DeleteResult(deleted: true, rehashes: rehashes, rows: [row] + trace + removed.moves + rehashes)
501 }
502 func costs() -> [Cost] {
503 var records: [Cost] = []
504 if mode == .chain {
505 for (position, bucket) in buckets.enumerated() {
506 for (index, item) in bucket.enumerated() {
507 records.append(Cost(
508 key: item.key,
509 cost: index + 1,
510 position: position
511 ))
512 }
513 }
514 } else {
515 for (position, slot) in slots.enumerated() {
516 guard let item = slot.entry else {
517 continue
518 }
519 let cost = mode == .cuckoo ? (position < cap ? 1 : 2)
520 : mode == .linear || mode == .robinHood ? distance(item.key, position) + 1 : search(item.key).probes
521 records.append(Cost(key: item.key, cost: cost, position: position))
522 }
523 }
524 return records
525 }
526 func metrics() -> Metrics {
527 let values = costs().map { $0.cost }
528 return Metrics(longest: values.max() ?? 0, average: decimal(values.reduce(0, +), size))
529 }
530 func dump() -> [String] {
531 var rows = ["DUMP \(mode.rawValue) cap=\(cap) size=\(size) tomb=\(tomb)"]
532 if mode == .chain {
533 for (index, bucket) in buckets.enumerated() {
534 rows.append("\(index) " + (bucket.isEmpty ? "-" : bucket.map { $0.text }.joined(separator: " ")))
535 }
536 } else {
537 for (index, slot) in slots.enumerated() {
538 let position = mode == .cuckoo ? "T\(index / cap + 1) \(index % cap)" : String(index)
539 var value = slot.entry?.text ?? slot.content
540 if mode == .robinHood, let item = slot.entry {
541 value += " d=\(distance(item.key, index))"
542 }
543 rows.append(position + " " + value)
544 }
545 }
546 return rows
547 }
548 func map() -> [String] {
549 var rows = ["MAP \(mode.rawValue) cap=\(cap)"], costBySlot = Array(repeating: 0, count: slotCount)
550 if mode != .chain {
551 for record in costs() {
552 costBySlot[record.position] = record.cost
553 }
554 }
555 for part in 0..<mode.parts {
556 var characters: [Character] = []
557 for index in 0..<cap {
558 if mode == .chain {
559 characters.append(buckets[index].isEmpty ? "." : gauge(buckets[index].count))
560 } else {
561 let flat = part * cap + index, slot = slots[flat]
562 characters
563 .append(slot.isEmpty ? "." : slot
564 .isTombstone ? "D" : mode == .cuckoo ? "#" : gauge(costBySlot[flat] - 1))
565 }
566 }
567 for index in stride(from: 0, to: cap, by: 40) {
568 let prefix = mode == .cuckoo ? "T\(part + 1) " : ""
569 rows.append(prefix + padded(index, 5) + " " + String(characters[index..<min(index + 40, cap)]))
570 }
571 }
572 return rows
573 }
574 func statistics() -> [String] {
575 let values = metrics(), count = counters
576 return [
577 "STATS \(mode.rawValue) size=\(size) cap=\(cap) slots=\(slotCount) tomb=\(tomb) load=\(decimal(100 * size, slotCount))% longest=\(values.longest) avg=\(values.average) probes=\(count.probes) rehashes=\(count.rehashes)",
578 "OPS inserts=\(count.inserts) updates=\(count.updates) hits=\(count.hits) misses=\(count.misses) deletes=\(count.deletes) absent=\(count.absent)"
579 ]
580 }
581 func scan(_ cursor: Int, _ limit: Int) -> [String] {
582 var keys: [String] = [], next = cursor
583 while next < slotCount && keys.count < limit {
584 if mode == .chain {
585 keys += buckets[next].map { $0.key }
586 } else if let item = slots[next].entry {
587 keys.append(item.key)
588 }
589 next += 1
590 }
591 if next == slotCount {
592 next = 0
593 }
594 return [(["SCAN next=\(next)"] + keys).joined(separator: " ")]
595 }
596 func clusters() -> [String] {
597 var lengths: [Int]
598 if mode == .chain {
599 lengths = buckets.filter { !$0.isEmpty }.map { $0.count }
600 } else {
601 lengths = runs(slots, start: 0, count: cap)
602 if mode == .cuckoo {
603 lengths += runs(slots, start: cap, count: cap)
604 }
605 }
606 return ["CLUSTERS runs=\(lengths.count) longest=\(lengths.max() ?? 0)"] + histogram(lengths)
607 }
608 func distribution() -> [String] {
609 let values = costs().map { $0.cost }
610 return ["DIST keys=\(size) longest=\(values.max() ?? 0)"] + histogram(values)
611 }

632final class Simulator {
633 private var table: Table?
634 private var maxLoad = 75, minLoad = 0
635 private var tracing = false
636 func compare(_ capacity: Int) -> [String] {
637 let items = table!.items().sorted { $0.key < $1.key }
638 var rows = ["COMPARE keys=\(items.count) cap=\(capacity)"]
639 for mode in Mode.allCases {
640 let experiment = Table(mode, capacity)
641 for item in items {
642 _ = experiment.put(item.key, item.value, command: "PUT", maxLoad: maxLoad)
643 }
644 let values = experiment.metrics()
645 rows
646 .append(
647 "\(mode.rawValue) cap=\(experiment.cap) longest=\(values.longest) avg=\(values.average) probes=\(experiment.counters.probes) rehashes=\(experiment.counters.rehashes)"
648 )
649 }
650 return rows
651 }
652 func bench(_ capacity: Int, _ count: Int, _ length: Int, _ seed: Int) -> [String] {
653 let keys = generatedKeys(2 * count, length, seed)
654 var rows = ["BENCH cap=\(capacity) count=\(count) len=\(length) seed=\(seed)"]
655 for mode in Mode.allCases {
656 let experiment = Table(mode, capacity)
657 for index in 0..<count {
658 _ = experiment.put(keys[index], index + 1, command: "PUT", maxLoad: maxLoad)
659 }
660 var misses = 0, probes = 0
661 for index in count..<(2 * count) {
662 let found = experiment.search(keys[index])
663 if found.item == nil {
664 misses += 1
665 probes += found.probes
666 }
667 }
668 rows
669 .append(
670 "\(mode.rawValue) cap=\(experiment.cap) keys=\(experiment.size) hit=\(experiment.metrics().average) misses=\(misses) miss=\(decimal(probes, misses)) rehashes=\(experiment.counters.rehashes)"
671 )
672 }
673 return rows
674 }
675 func execute(_ line: String) throws -> [String] {
676 let words = line.split(separator: " ").map(String.init), command = words[0]
677 let args = Array(words.dropFirst())
678 if !commands.contains(command) {
679 throw CommandError("UNKNOWN COMMAND " + command)
680 }
681 if command == "CREATE" {
682 let mode = try checkedMode(args[0]), capacity = Int(args[1])!
683 try checkCapacity(capacity)
684 let replacement = Table(mode, prime(capacity))
685 self.table = replacement
686 return ["CREATE \(mode.rawValue) cap=\(replacement.cap)"]
687 }
688 if command == "MAXLOAD" || command == "MINLOAD" {
689 let value = Int(args[0])!, maximum = command == "MAXLOAD"
690 if maximum ? !(10...95).contains(value) : !(0...40).contains(value) {
691 throw CommandError("BAD LOAD")
692 }
693 if maximum {
694 maxLoad = value
695 } else {
696 minLoad = value
697 }
698 return ["\(command) \(value)"]
699 }
700 if command == "TRACE" {
701 if args[0] != "ON" && args[0] != "OFF" {
702 throw CommandError("BAD ARGUMENT")
703 }
704 tracing = args[0] == "ON"
705 return ["TRACE " + args[0]]
706 }
707 if command == "HASH" {
708 let key = args[0], capacity = Int(args[1])!
709 try checkCapacity(capacity)
710 return [
711 "HASH \(key) m=\(capacity) h31=\(hash(key, 31, capacity)) h37=\(hash(key, 37, capacity)) step=\(1 + hash(key, 37, capacity - 1))"
712 ]
713 }
714 if command == "BENCH" {
715 let capacity = Int(args[0])!, count = Int(args[1])!, length = Int(args[2])!, seed = Int(args[3])!
716 try checkCapacity(capacity)
717 try checkGeneration(count, length, seed)
718 return bench(prime(capacity), count, length, seed)
719 }
720 guard let table = self.table else {
721 throw CommandError("NO TABLE")
722 }
723 switch command {
724 case "PUT",
725 "ADD":
726 return table.put(args[0], Int(args[1])!, command: command, maxLoad: maxLoad, tracing: tracing).rows
727 case "GET":
728 return table.get(args[0], tracing: tracing)
729 case "DEL":
730 return table.delete(args[0], minLoad: minLoad, maxLoad: maxLoad, tracing: tracing).rows
731 case "PROBE":
732 let found = table.search(args[0], tracing: true)
733 return [(["PROBE " + args[0]] + found
734 .path + ["\(found.item == nil ? "MISS" : "FOUND") probes=\(found.probes)"]).joined(separator: " ")]
735 case "GEN", "GENDEL":
736 let count = Int(args[0])!, length = Int(args[1])!, seed = Int(args[2])!
737 try checkGeneration(count, length, seed)
738 let keys = generatedKeys(count, length, seed)
739 var rows: [String] = [], successes = 0
740 for (index, key) in keys.enumerated() {
741 if command == "GEN" {
742 let result = table.put(key, index + 1, command: "PUT", maxLoad: maxLoad)
743 if result.fresh {
744 successes += 1
745 }
746 rows += result.rehashes
747 } else {
748 let result = table.delete(key, minLoad: minLoad, maxLoad: maxLoad)
749 if result.deleted {
750 successes += 1
751 }
752 rows += result.rehashes
753 }
754 }
755 rows
756 .append(command == "GEN" ? "GEN \(count) new=\(successes) updated=\(count - successes)" :
757 "GENDEL \(count) deleted=\(successes) missing=\(count - successes)")
758 return rows
759 case "RESIZE", "CONVERT":
760 var mode = table.mode, capacity = table.cap
761 if command == "RESIZE" {
762 capacity = Int(args[0])!
763 try checkCapacity(capacity)
764 capacity = prime(capacity)
765 } else {
766 mode = try checkedMode(args[0])
767 }
768 if 100 * table.size > maxLoad * capacity * mode.parts {
769 throw CommandError("TOO SMALL")
770 }
771 guard let replacement = Table.rebuild(table.items(), mode: mode, capacity: capacity)
772 else {
773 throw CommandError("REBUILD FAILED")
774 }
775 let row = command == "RESIZE" ? "RESIZE \(table.cap) -> \(capacity)" : "CONVERT \(table.mode.rawValue) -> \(mode.rawValue) cap=\(capacity)"
776 table.adopt(replacement)
777 return [row]
778 case "CLEAR":
779 let size = table.size
780 table.adopt(Table(table.mode, table.cap))
781 return ["CLEAR \(size)"]
782 case "DUMP":
783 return table.dump()
784 case "MAP":
785 return table.map()
786 case "STATS":
787 return table.statistics()
788 case "KEYS":
789 return [(["KEYS \(table.size)"] + table.items().map { $0.key }.sorted()).joined(separator: " ")]
790 case "TOP":
791 let limit = Int(args[0])!
792 if !(1...100).contains(limit) {
793 throw CommandError("BAD ARGUMENT")
794 }
795 let records = table.costs().sorted { $0.cost != $1.cost ? $0.cost > $1.cost : $0.key < $1.key }
796 .prefix(limit)
797 return ["TOP \(records.count)"] + records
798 .map { "\($0.key) cost=\($0.cost) at=\(table.position($0.position))" }
799 case "SCAN":
800 let cursor = Int(args[0])!, limit = Int(args[1])!
801 if !(1...100).contains(limit) {
802 throw CommandError("BAD ARGUMENT")
803 }
804 if cursor >= table.slotCount {
805 throw CommandError("BAD CURSOR")
806 }
807 return table.scan(cursor, limit)
808 case "CLUSTERS":
809 return table.clusters()
810 case "DIST":
811 return table.distribution()
812 case "COMPARE":
813 let capacity = Int(args[0])!
814 try checkCapacity(capacity)
815 return compare(prime(capacity))
816 default:
817 preconditionFailure("Unhandled command " + command)
818 }
819 }
820}
