9next_prime <- function(value) {
10 value <- max(2L, value)
11 repeat {
12 divisor <- 2L
13 while (divisor * divisor <= value && value %% divisor != 0L) divisor <- divisor + 1L
14 if (divisor * divisor > value) return(value)
15 value <- value + 1L
16 }
17}
18hash <- function(key, base, modulus) {
19 value <- 0L
20 for (code in utf8ToInt(key)) value <- (value * base + code) %% modulus
21 value

25 scaled <- as.integer((gmp::as.bigz(numerator) * 100L) %/% denominator)

37new_table <- function(mode, cap) {
38 table <- new.env(parent = emptyenv())
39 table$mode <- mode
40 table$cap <- cap

42 table$buckets <- vector("list", if (mode == "CHAIN") cap else 0L)
43 count <- if (mode == "CHAIN") 0L else slot_count(table)
44 table$state <- integer(count)
45 table$keys <- rep("", count)
46 table$values <- integer(count)
47 table$counters <- setNames(integer(8L), c("probes", "rehashes", "inserts", "updates",

73search <- function(table, key, tracing = FALSE) {
74 result <- list(found = FALSE, position = 0L, chain = 0L, probes = 0L, available = 0L,
75 item = NULL, path = character())
76 home <- hash(key, 31L, table$cap) + 1L
77 if (table$mode == "CHAIN") {
78 bucket <- table$buckets[[home]]
79 for (index in seq_along(bucket)) {
80 item <- bucket[[index]]
81 result$probes <- result$probes + 1L
82 if (tracing) result$path <- c(result$path, paste0(home - 1L, ":", item$key))
83 if (item$key == key) {
84 result$found <- TRUE
85 result$position <- home
86 result$chain <- index
87 result$item <- item
88 return(result)
89 }
90 }
91 } else if (table$mode == "CUCKOO") {
92 for (position in c(home, table$cap + hash(key, 37L, table$cap) + 1L)) {
93 result$probes <- result$probes + 1L
94 if (tracing) result$path <- c(result$path, paste0(position_text(table, position), ":", content(table, position)))
95 if (table$state[position] == 1L && table$keys[position] == key) {
96 result$found <- TRUE
97 result$position <- position
98 result$item <- slot_item(table, position)
99 return(result)
100 }
101 }
102 } else {
103 step <- 1L + hash(key, 37L, table$cap - 1L)
104 for (index in seq_len(table$cap) - 1L) {
105 offset <- switch(table$mode, QUADRATIC = index * index, DOUBLE = index * step, index)
106 position <- (home - 1L + offset) %% table$cap + 1L
107 result$probes <- result$probes + 1L
108 if (tracing) result$path <- c(result$path, paste0(position - 1L, ":", content(table, position)))
109 if (table$state[position] == 0L) {
110 if (result$available == 0L) result$available <- position
111 return(result)
112 }
113 if (table$state[position] == 2L) {
114 if (result$available == 0L) result$available <- position
115 } else {
116 if (table$keys[position] == key) {
117 result$found <- TRUE
118 result$position <- position
119 result$item <- slot_item(table, position)
120 return(result)
121 }
122 if (table$mode == "ROBINHOOD" && distance(table, table$keys[position], position) < index) return(result)
123 }
124 }
125 }
126 result
127}
128place <- function(table, item, initial = NULL, tracing = FALSE) {
129 result <- list(success = TRUE, position = 0L, reused = FALSE, shifted = 0L,
130 kicks = 0L, homeless = NULL, moves = character())
131 original <- item$key
132 if (table$mode == "CHAIN") {
133 result$position <- hash(item$key, 31L, table$cap) + 1L
134 table$buckets[[result$position]] <- c(table$buckets[[result$position]], list(item))
135 } else if (table$mode %in% c("LINEAR", "QUADRATIC", "DOUBLE")) {
136 if (is.null(initial)) initial <- search(table, item$key)
137 position <- initial$available
138 result$position <- position
139 if (position == 0L) {
140 result$success <- FALSE
141 result$homeless <- item
142 } else {
143 result$reused <- table$state[position] == 2L
144 if (result$reused) table$tomb <- table$tomb - 1L
145 put_slot(table, position, item)
146 }
147 } else if (table$mode == "ROBINHOOD") {
148 position <- hash(item$key, 31L, table$cap) + 1L
149 travelled <- 0L
150 origin <- 0L
151 repeat {
152 if (table$state[position] == 0L) {
153 put_slot(table, position, item)
154 if (result$position == 0L) result$position <- position
155 if (tracing && origin > 0L) result$moves <- c(result$moves, sprintf("SHIFT %s %d -> %d", item$key, origin - 1L, position - 1L))
156 break
157 }
158 other <- slot_item(table, position)
159 old_distance <- distance(table, other$key, position)
160 if (old_distance < travelled) {
161 put_slot(table, position, item)
162 if (result$position == 0L) result$position <- position
163 if (tracing && origin > 0L) result$moves <- c(result$moves, sprintf("SHIFT %s %d -> %d", item$key, origin - 1L, position - 1L))
164 item <- other
165 origin <- position
166 travelled <- old_distance
167 result$shifted <- result$shifted + 1L
168 }
169 position <- position %% table$cap + 1L
170 travelled <- travelled + 1L
171 }
172 } else {
173 part <- 0L
174 repeat {
175 position <- part * table$cap + hash(item$key, 31L + 6L * part, table$cap) + 1L
176 if (table$state[position] == 0L) {
177 put_slot(table, position, item)
178 result$position <- search(table, original)$position
179 break
180 }
181 if (result$kicks == table$cap) {
182 result$success <- FALSE
183 result$homeless <- item
184 break
185 }
186 other <- slot_item(table, position)
187 put_slot(table, position, item)
188 result$kicks <- result$kicks + 1L
189 if (tracing) result$moves <- c(result$moves, paste("KICK", other$key, position_text(table, position)))
190 item <- other
191 part <- 1L - part
192 }
193 }
194 result
195}
196remove_item <- function(table, found, tracing) {
197 shifted <- 0L
198 moves <- character()
199 position <- found$position
200 if (table$mode == "CHAIN") {
201 table$buckets[[position]] <- table$buckets[[position]][-found$chain]
202 } else if (table$mode %in% c("LINEAR", "QUADRATIC", "DOUBLE")) {
203 table$state[position] <- 2L
204 table$tomb <- table$tomb + 1L
205 } else {
206 table$state[position] <- 0L
207 if (table$mode == "ROBINHOOD") {
208 following <- position %% table$cap + 1L
209 while (table$state[following] == 1L && distance(table, table$keys[following], following) > 0L) {
210 destination <- (following - 2L + table$cap) %% table$cap + 1L
211 item <- slot_item(table, following)
212 put_slot(table, destination, item)
213 table$state[following] <- 0L
214 shifted <- shifted + 1L
215 if (tracing) moves <- c(moves, sprintf("SHIFT %s %d -> %d", item$key, following - 1L, destination - 1L))
216 following <- following %% table$cap + 1L
217 }
218 }
219 }
220 table$size <- table$size - 1L
221 list(shifted = shifted, moves = moves)
222}

224rebuild <- function(table, mode, capacity, items) {
225 trial <- new_table(mode, capacity)
226 for (item in items) {
227 if (!place(trial, item)$success) return(FALSE)
228 }
229 trial$size <- length(items)
230 adopt(table, trial)
231 TRUE

233automatic <- function(table, capacity, reason, pending = NULL) {
234 old <- table$cap
235 items <- table_items(table)
236 if (!is.null(pending)) items <- c(items, list(pending))
237 while (!rebuild(table, table$mode, capacity, items)) capacity <- next_prime(2L * capacity)
238 increment(table, "rehashes")
239 sprintf("REHASH %s %d -> %d", reason, old, table$cap)
240}
241load_check <- function(table, maximum) {
242 if ((table$size + table$tomb) * 100L <= maximum * slot_count(table)) return(character())
243 grow <- table$size * 200L > maximum * slot_count(table)
244 automatic(table, if (grow) next_prime(2L * table$cap) else table$cap, if (grow) "load" else "tomb")
245}
246shrink_check <- function(table, minimum, maximum) {
247 capacity <- next_prime(table$cap %/% 2L)
248 slots <- capacity * if (table$mode == "CUCKOO") 2L else 1L
249 if (table$size * 100L >= minimum * slot_count(table) || capacity >= table$cap || table$size * 100L > maximum * slots) return(character())
250 old <- table$cap
251 if (!rebuild(table, table$mode, capacity, table_items(table))) return(character())
252 increment(table, "rehashes")
253 sprintf("REHASH shrink %d -> %d", old, capacity)
254}
255trace_path <- function(found, tracing) {
256 if (tracing) paste("PATH", if (length(found$path)) paste(found$path, collapse = " ") else "none") else character()
257}
258put <- function(table, key, value, command, maximum, tracing = FALSE) {
259 found <- search(table, key, tracing)
260 increment(table, "probes", found$probes)
261 trace <- trace_path(found, tracing)
262 if (found$found) {
263 old <- found$item$value
264 if (command == "ADD") value <- old + value
265 item <- list(key = key, value = value)
266 if (table$mode == "CHAIN") table$buckets[[found$position]][[found$chain]] <- item else put_slot(table, found$position, item)
267 increment(table, "updates")
268 row <- sprintf("%s %s=%d UPDATED old=%d at=%s probes=%d", command, key, value, old, position_text(table, found$position), found$probes)
269 return(list(changed = FALSE, rehashes = character(), rows = c(row, trace)))
270 }
271 placed <- place(table, list(key = key, value = value), found, tracing)
272 table$size <- table$size + 1L
273 increment(table, "inserts")
274 row <- sprintf("%s %s=%d NEW", command, key, value)
275 rehashes <- character()
276 if (placed$success) {
277 row <- paste0(row, sprintf(" at=%s probes=%d", position_text(table, placed$position), found$probes))
278 if (placed$reused) row <- paste0(row, " reused")
279 if (table$mode == "ROBINHOOD") row <- paste0(row, " shifted=", placed$shifted)
280 if (table$mode == "CUCKOO") row <- paste0(row, " kicks=", placed$kicks)
281 } else {
282 row <- paste0(row, " probes=", found$probes)
283 cuckoo <- table$mode == "CUCKOO"
284 row <- paste0(row, if (cuckoo) paste0(" kicks=", placed$kicks, " failed") else " full")
285 rehashes <- automatic(table, next_prime(2L * table$cap), if (cuckoo) "kicks" else "full", placed$homeless)
286 }
287 rehashes <- c(rehashes, load_check(table, maximum))
288 list(changed = TRUE, rehashes = rehashes, rows = c(row, trace, placed$moves, rehashes))
289}
290get_item <- function(table, key, tracing) {
291 found <- search(table, key, tracing)
292 increment(table, "probes", found$probes)
293 if (found$found) {
294 increment(table, "hits")
295 row <- sprintf("GET %s=%d at=%s probes=%d", key, found$item$value, position_text(table, found$position), found$probes)
296 } else {
297 increment(table, "misses")
298 row <- sprintf("GET %s NOT FOUND probes=%d", key, found$probes)
299 }
300 c(row, trace_path(found, tracing))
301}
302delete_item <- function(table, key, minimum, maximum, tracing = FALSE) {
303 found <- search(table, key, tracing)
304 increment(table, "probes", found$probes)
305 trace <- trace_path(found, tracing)
306 if (!found$found) {
307 increment(table, "absent")
308 return(list(changed = FALSE, rehashes = character(), rows = c(sprintf("DEL %s NOT FOUND probes=%d", key, found$probes), trace)))
309 }
310 removed <- remove_item(table, found, tracing)
311 increment(table, "deletes")
312 row <- sprintf("DEL %s OK at=%s probes=%d", key, position_text(table, found$position), found$probes)
313 if (table$mode == "ROBINHOOD") row <- paste0(row, " shifted=", removed$shifted)
314 rehashes <- shrink_check(table, minimum, maximum)
315 list(changed = TRUE, rehashes = rehashes, rows = c(row, trace, removed$moves, rehashes))
316}

318costs <- function(table) {
319 keys <- character()
320 values <- positions <- integer()
321 if (table$mode == "CHAIN") {
322 for (position in seq_len(table$cap)) {
323 bucket <- table$buckets[[position]]
324 keys <- c(keys, item_keys(bucket))
325 values <- c(values, seq_along(bucket))
326 positions <- c(positions, rep.int(position, length(bucket)))
327 }
328 } else {
329 positions <- which(table$state == 1L)
330 keys <- table$keys[positions]
331 values <- vapply(positions, function(position) {
332 if (table$mode == "CUCKOO") return((position - 1L) %/% table$cap + 1L)
333 if (table$mode %in% c("LINEAR", "ROBINHOOD")) return(distance(table, table$keys[position], position) + 1L)
334 search(table, table$keys[position])$probes
335 }, integer(1), USE.NAMES = FALSE)
336 }
337 list(keys = keys, values = values, positions = positions)
338}

381 }

486execute <- function(simulator, line) {
487 words <- strsplit(line, " ", fixed = TRUE)[[1L]]
488 command <- words[1L]
489 if (!command %in% commands) fail(paste("UNKNOWN COMMAND", command))
490 if (command == "CREATE") {
491 mode <- check_mode(words[2L])
492 capacity <- next_prime(check_capacity(as.integer(words[3L])))
493 simulator$current <- new_table(mode, capacity)
494 return(sprintf("CREATE %s cap=%d", mode, capacity))
495 }
496 if (command == "MAXLOAD" || command == "MINLOAD") {
497 value <- as.integer(words[2L])
498 limits <- if (command == "MAXLOAD") c(10L, 95L) else c(0L, 40L)
499 if (value < limits[1L] || value > limits[2L]) fail("BAD LOAD")
500 if (command == "MAXLOAD") simulator$max_load <- value else simulator$min_load <- value
501 return(line)
502 }
503 if (command == "TRACE") {
504 if (!words[2L] %in% c("ON", "OFF")) fail("BAD ARGUMENT")
505 simulator$tracing <- words[2L] == "ON"
506 return(line)
507 }
508 if (command == "HASH") {
509 capacity <- check_capacity(as.integer(words[3L]))
510 key <- words[2L]
511 return(sprintf("HASH %s m=%d h31=%d h37=%d step=%d", key, capacity, hash(key, 31L, capacity),
512 hash(key, 37L, capacity), 1L + hash(key, 37L, capacity - 1L)))
513 }
514 if (command == "BENCH") {
515 capacity <- next_prime(check_capacity(as.integer(words[2L])))
516 args <- check_generation(as.integer(words[3:5]))
517 return(bench(capacity, args[1L], args[2L], args[3L], simulator$max_load))
518 }
519 table <- simulator$current
520 if (is.null(table)) fail("NO TABLE")
521 switch(command,
522 PUT =, ADD = put(table, words[2L], as.integer(words[3L]), command, simulator$max_load, simulator$tracing)$rows,
523 GET = get_item(table, words[2L], simulator$tracing),
524 DEL = delete_item(table, words[2L], simulator$min_load, simulator$max_load, simulator$tracing)$rows,
525 PROBE = {
526 found <- search(table, words[2L], TRUE)
527 ending <- paste0(if (found$found) "FOUND" else "MISS", " probes=", found$probes)
528 paste(c(paste("PROBE", words[2L]), found$path, ending), collapse = " ")
529 },
530 GEN =, GENDEL = {
531 args <- check_generation(as.integer(words[2:4]))
532 keys <- generated_keys(args[1L], args[2L], args[3L])
533 successes <- 0L
534 rows <- character()
535 for (index in seq_along(keys)) {
536 result <- if (command == "GEN") put(table, keys[index], index, "PUT", simulator$max_load) else
537 delete_item(table, keys[index], simulator$min_load, simulator$max_load)
538 if (result$changed) successes <- successes + 1L
539 rows <- c(rows, result$rehashes)
540 }
541 summary <- if (command == "GEN") sprintf("GEN %d new=%d updated=%d", args[1L], successes, args[1L] - successes) else
542 sprintf("GENDEL %d deleted=%d missing=%d", args[1L], successes, args[1L] - successes)
543 c(rows, summary)
544 },
545 RESIZE =, CONVERT = {
546 mode <- table$mode
547 capacity <- table$cap
548 if (command == "RESIZE") capacity <- next_prime(check_capacity(as.integer(words[2L]))) else mode <- check_mode(words[2L])
549 slots <- capacity * if (mode == "CUCKOO") 2L else 1L
550 if (table$size * 100L > simulator$max_load * slots) fail("TOO SMALL")
551 row <- if (command == "RESIZE") sprintf("RESIZE %d -> %d", table$cap, capacity) else
552 sprintf("CONVERT %s -> %s cap=%d", table$mode, mode, capacity)
553 if (!rebuild(table, mode, capacity, table_items(table))) fail("REBUILD FAILED")
554 row
555 },
556 CLEAR = {
557 row <- paste("CLEAR", table$size)
558 adopt(table, new_table(table$mode, table$cap))
559 row
560 },
561 DUMP = dump(table),
562 MAP = map_report(table),
563 STATS = statistics(table),
564 KEYS = paste(c(paste("KEYS", table$size), sort(item_keys(table_items(table)), method = "radix")), collapse = " "),
565 TOP = {
566 limit <- as.integer(words[2L])
567 if (limit < 1L || limit > 100L) fail("BAD ARGUMENT")
568 records <- costs(table)
569 selected <- head(order(-records$values, records$keys, method = "radix"), limit)
570 c(paste("TOP", length(selected)), sprintf("%s cost=%d at=%s", records$keys[selected],
571 records$values[selected], position_text(table, records$positions[selected])))
572 },
573 SCAN = {
574 cursor <- as.integer(words[2L])
575 limit <- as.integer(words[3L])
576 if (limit < 1L || limit > 100L) fail("BAD ARGUMENT")
577 if (cursor >= slot_count(table)) fail("BAD CURSOR")
578 scan_table(table, cursor, limit)
579 },
580 CLUSTERS = clusters(table),
581 DIST = {
582 values <- costs(table)$values
583 c(sprintf("DIST keys=%d longest=%d", table$size, max(0L, values)), histogram(values))
584 },
585 COMPARE = compare(table, next_prime(check_capacity(as.integer(words[2L]))), simulator$max_load)
586 )
587}
588main <- function() {
589 input <- readLines(file("stdin"), warn = FALSE)
590 simulator <- new.env(parent = emptyenv())
591 simulator$current <- NULL
592 simulator$max_load <- 75L
593 simulator$min_load <- 0L
594 simulator$tracing <- FALSE
595 for (index in seq_len(as.integer(input[1L]))) {
596 rows <- tryCatch(execute(simulator, input[index + 1L]),
597 command_error = function(problem) paste("ERROR", conditionMessage(problem)))
598 cat(paste(rows, collapse = "\n"), "\n", sep = "")
599 }
600}
601main()
