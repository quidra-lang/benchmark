44int hash(const std::string& key, int base, int modulus) {
45 int value = 0;
46 for (unsigned char character : key) value = (value * base + character) % modulus;
47 return value;

49int prime(int value) {
50 for (int candidate = std::max(2, value); ; ++candidate) {
51 bool valid = true;
52 for (int divisor = 2; divisor * divisor <= candidate; ++divisor) {
53 if (candidate % divisor == 0) { valid = false; break; }
54 }
55 if (valid) return candidate;
56 }
57}
58std::string decimal(Integer numerator, Integer denominator) {
59 const Integer value = denominator == 0 ? 0 : 100 * numerator / denominator;
60 return std::to_string(value / 100) + "." + (value % 100 < 10 ? "0" : "") + std::to_string(value % 100);
61}

74Lines histogram(const std::vector<int>& values) {
75 std::map<int, int> counts;
76 for (int value : values) ++counts[value];
77 Lines rows;
78 for (const auto& [value, count] : counts) {
79 rows.push_back(std::to_string(value) + " " + std::to_string(count) + " " + std::string(std::min(count, 50), '#') + (count > 50 ? "+" : ""));
80 }
81 return rows;

83std::vector<int> runs(const std::vector<Slot>& slots, int begin, int count) {
84 int firstEmpty = -1;
85 for (int i = 0; i < count; ++i) if (empty(slots[begin + i])) { firstEmpty = i; break; }
86 if (firstEmpty < 0) return {count};
87 std::vector<int> lengths;
88 int length = 0;
89 for (int offset = 1; offset <= count; ++offset) {
90 if (!empty(slots[begin + (firstEmpty + offset) % count])) ++length;
91 else if (length > 0) { lengths.push_back(length); length = 0; }
92 }
93 return lengths;

117class Table {
118 std::vector<std::vector<Item>> buckets;
119 std::vector<Slot> slots;
120public:
121 Mode mode;
122 int cap, size = 0, tomb = 0;
123 Counters counters;
124 Table(Mode selected, int capacity)
125 : buckets(selected == Mode::Chain ? capacity : 0),
126 slots(selected == Mode::Chain ? 0 : capacity * (selected == Mode::Cuckoo ? 2 : 1)),
127 mode(selected), cap(capacity) {}
128 int slotCount() const { return cap * (mode == Mode::Cuckoo ? 2 : 1); }
129 bool usesTombstones() const { return mode == Mode::Linear || mode == Mode::Quadratic || mode == Mode::Double; }
130 std::string position(int index) const {
131 return mode == Mode::Cuckoo ? "T" + std::to_string(index / cap + 1) + "." + std::to_string(index % cap) : std::to_string(index);
132 }
133 int distance(const std::string& key, int index) const { return (index - hash(key, 31, cap) + cap) % cap; }
134 std::vector<Item> items() const {
135 std::vector<Item> result;
136 if (mode == Mode::Chain) {
137 for (const std::vector<Item>& bucket : buckets) result.insert(result.end(), bucket.begin(), bucket.end());
138 } else {
139 for (const Slot& slot : slots) if (const Item* item = std::get_if<Item>(&slot)) result.push_back(*item);
140 }
141 return result;
142 }
143 Placement place(const Item& item, const Search* initial = nullptr, bool tracing = false) {
144 Placement result;
145 if (mode == Mode::Chain) {
146 result.position = hash(item.key, 31, cap);
147 buckets[result.position].push_back(item);
148 result.success = true;
149 return result;
150 }
151 if (usesTombstones()) {
152 const Search fresh = initial ? Search{} : search(item.key);
153 const Search& found = initial ? *initial : fresh;
154 if (!found.available) { result.homeless = item; return result; }
155 result.position = *found.available;
156 result.reused = std::holds_alternative<Tombstone>(slots[result.position]);
157 slots[result.position] = item;
158 if (result.reused) --tomb;
159 result.success = true;
160 return result;
161 }
162 Item current = item;
163 if (mode == Mode::RobinHood) {
164 int index = hash(item.key, 31, cap), dist = 0;
165 std::optional<int> from;
166 for (;;) {
167 Slot& slot = slots[index];
168 if (empty(slot)) {
169 slot = current;
170 if (tracing && from) result.moves.push_back("SHIFT " + current.key + " " + std::to_string(*from) + " -> " + std::to_string(index));
171 if (result.position < 0) result.position = index;
172 result.success = true;
173 return result;
174 }
175 const Item occupant = std::get<Item>(slot);
176 const int otherDistance = distance(occupant.key, index);
177 if (otherDistance < dist) {
178 slot = current;
179 if (tracing && from) result.moves.push_back("SHIFT " + current.key + " " + std::to_string(*from) + " -> " + std::to_string(index));
180 if (result.position < 0) result.position = index;
181 current = occupant;
182 from = index;
183 dist = otherDistance;
184 ++result.shifted;
185 }
186 index = (index + 1) % cap;
187 ++dist;
188 }
189 }
190 int part = 0;
191 for (;;) {
192 const int index = part * cap + hash(current.key, part == 0 ? 31 : 37, cap);
193 Slot& slot = slots[index];
194 if (empty(slot)) {
195 slot = current;
196 result.success = true;
197 result.position = search(item.key).position;
198 return result;
199 }
200 if (result.kicks == cap) { result.homeless = current; return result; }
201 const Item occupant = std::get<Item>(slot);
202 slot = current;
203 ++result.kicks;
204 if (tracing) result.moves.push_back("KICK " + occupant.key + " " + position(index));
205 current = occupant;
206 part = 1 - part;
207 }
208 }
209 Removal remove(const Search& found, bool tracing) {
210 Removal result;
211 if (mode == Mode::Chain) {
212 std::vector<Item>& bucket = buckets[found.position];
213 bucket.erase(bucket.begin() + found.chainIndex);
214 } else if (usesTombstones()) {
215 slots[found.position] = Tombstone{};
216 ++tomb;
217 } else {
218 slots[found.position] = std::monostate{};
219 if (mode == Mode::RobinHood) {
220 int next = (found.position + 1) % cap;
221 while (!empty(slots[next])) {
222 const Item item = std::get<Item>(slots[next]);
223 if (distance(item.key, next) == 0) break;
224 const int destination = (next - 1 + cap) % cap;
225 slots[destination] = item;
226 slots[next] = std::monostate{};
227 if (tracing) result.moves.push_back("SHIFT " + item.key + " " + std::to_string(next) + " -> " + std::to_string(destination));
228 ++result.shifted;
229 next = (next + 1) % cap;
230 }
231 }
232 }
233 --size;
234 return result;
235 }
236 static std::unique_ptr<Table> rebuild(const std::vector<Item>& items, Mode mode, int capacity) {
237 std::unique_ptr<Table> table = std::make_unique<Table>(mode, capacity);
238 for (const Item& item : items) {
239 if (!table->place(item).success) return nullptr;
240 ++table->size;
241 }
242 return table;
243 }
244 void adopt(Table&& replacement) {
245 const Counters saved = counters;
246 *this = std::move(replacement);
247 counters = saved;
248 }
249 std::string automatic(const std::string& reason, int capacity, const std::optional<Item>& pending = std::nullopt) {
250 std::vector<Item> listed = items();
251 if (pending) listed.push_back(*pending);
252 const int old = cap;
253 for (;;) {
254 std::unique_ptr<Table> replacement = rebuild(listed, mode, capacity);
255 if (replacement) {
256 adopt(std::move(*replacement));
257 ++counters.rehashes;
258 return "REHASH " + reason + " " + std::to_string(old) + " -> " + std::to_string(cap);
259 }
260 capacity = prime(2 * capacity);
261 }
262 }
263 Lines loadCheck(int maxLoad) {
264 const Integer threshold = maxLoad * slotCount();
265 if (100LL * (size + tomb) <= threshold) return {};
266 const bool grow = 200LL * size > threshold;
267 return {automatic(grow ? "load" : "tomb", grow ? prime(2 * cap) : cap)};
268 }
269 Lines shrinkCheck(int minLoad, int maxLoad) {
270 const int capacity = prime(cap / 2), newSlots = capacity * (mode == Mode::Cuckoo ? 2 : 1);
271 if (100LL * size >= minLoad * slotCount() || capacity >= cap || 100LL * size > maxLoad * newSlots) return {};
272 std::unique_ptr<Table> replacement = rebuild(items(), mode, capacity);
273 if (!replacement) return {};
274 const int old = cap;
275 adopt(std::move(*replacement));
276 ++counters.rehashes;
277 return {"REHASH shrink " + std::to_string(old) + " -> " + std::to_string(capacity)};
278 }
279 PutResult put(const std::string& key, Integer value, const std::string& command, int maxLoad, bool tracing = false) {
280 const Search found = search(key, tracing);
281 const Lines trace = tracing ? Lines{"PATH " + (found.path.empty() ? "none" : join(found.path))} : Lines{};
282 counters.probes += found.probes;
283 if (found.item) {
284 const Integer old = found.item->value, next = command == "ADD" ? old + value : value;
285 const Item item{key, next};
286 if (mode == Mode::Chain) buckets[found.position][found.chainIndex] = item;
287 else slots[found.position] = item;
288 ++counters.updates;
289 Lines rows{command + " " + pairText(item) + " UPDATED old=" + std::to_string(old) + " at=" + position(found.position) + " probes=" + std::to_string(found.probes)};
290 append(rows, trace);
291 return {false, {}, rows};
292 }
293 const Item item{key, value};
294 const Placement placed = place(item, &found, tracing);
295 ++size;
296 ++counters.inserts;
297 std::string row = command + " " + pairText(item) + " NEW";
298 if (placed.success) {
299 row += " at=" + position(placed.position) + " probes=" + std::to_string(found.probes);
300 if (placed.reused) row += " reused";
301 if (mode == Mode::RobinHood) row += " shifted=" + std::to_string(placed.shifted);
302 if (mode == Mode::Cuckoo) row += " kicks=" + std::to_string(placed.kicks);
303 } else {
304 row += " probes=" + std::to_string(found.probes);
305 row += mode == Mode::Cuckoo ? " kicks=" + std::to_string(placed.kicks) + " failed" : " full";
306 }
307 Lines rehashes;
308 if (!placed.success) rehashes.push_back(automatic(mode == Mode::Cuckoo ? "kicks" : "full", prime(2 * cap), placed.homeless));
309 append(rehashes, loadCheck(maxLoad));
310 Lines rows{row};
311 append(rows, trace); append(rows, placed.moves); append(rows, rehashes);
312 return {true, rehashes, rows};
313 }
314 Lines get(const std::string& key, bool tracing) {
315 const Search found = search(key, tracing);
316 counters.probes += found.probes;
317 std::string row;
318 if (found.item) { ++counters.hits; row = "GET " + pairText(*found.item) + " at=" + position(found.position); }
319 else { ++counters.misses; row = "GET " + key + " NOT FOUND"; }
320 Lines rows{row + " probes=" + std::to_string(found.probes)};
321 if (tracing) rows.push_back("PATH " + (found.path.empty() ? "none" : join(found.path)));
322 return rows;
323 }
324 DeleteResult erase(const std::string& key, int minLoad, int maxLoad, bool tracing = false) {
325 const Search found = search(key, tracing);
326 const Lines trace = tracing ? Lines{"PATH " + (found.path.empty() ? "none" : join(found.path))} : Lines{};
327 counters.probes += found.probes;
328 if (!found.item) {
329 ++counters.absent;
330 Lines rows{"DEL " + key + " NOT FOUND probes=" + std::to_string(found.probes)};
331 append(rows, trace);
332 return {false, {}, rows};
333 }
334 const Removal removed = remove(found, tracing);
335 ++counters.deletes;
336 std::string row = "DEL " + key + " OK at=" + position(found.position) + " probes=" + std::to_string(found.probes);
337 if (mode == Mode::RobinHood) row += " shifted=" + std::to_string(removed.shifted);
338 const Lines rehashes = shrinkCheck(minLoad, maxLoad);
339 Lines rows{row};
340 append(rows, trace); append(rows, removed.moves); append(rows, rehashes);
341 return {true, rehashes, rows};
342 }
343 Search search(const std::string& key, bool tracing = false) const {
344 Search result;
345 const int first = hash(key, 31, cap);
346 const auto examine = [&](int index, const Slot& slot) {
347 ++result.probes;
348 if (tracing) result.path.push_back(position(index) + ":" + content(slot));
349 };
350 const auto found = [&](const Item& item, int index, int chainIndex = -1) {
351 result.item = item; result.position = index; result.chainIndex = chainIndex;
352 };
353 if (mode == Mode::Chain) {
354 const std::vector<Item>& bucket = buckets[first];
355 for (int i = 0; i < static_cast<int>(bucket.size()); ++i) {
356 examine(first, bucket[i]);
357 if (bucket[i].key == key) { found(bucket[i], first, i); break; }
358 }
359 return result;
360 }
361 if (mode == Mode::Cuckoo) {
362 for (int index : {first, cap + hash(key, 37, cap)}) {
363 examine(index, slots[index]);
364 const Item* item = std::get_if<Item>(&slots[index]);
365 if (item && item->key == key) { found(*item, index); break; }
366 }
367 return result;
368 }
369 const int step = mode == Mode::Double ? 1 + hash(key, 37, cap - 1) : 1;
370 for (int i = 0; i < cap; ++i) {
371 const int offset = mode == Mode::Quadratic ? i * i : i * step;
372 const int index = (first + offset) % cap;
373 const Slot& slot = slots[index];
374 examine(index, slot);
375 if (empty(slot)) { if (!result.available) result.available = index; break; }
376 if (std::holds_alternative<Tombstone>(slot)) { if (!result.available) result.available = index; continue; }
377 const Item& item = std::get<Item>(slot);
378 if (item.key == key) { found(item, index); break; }
379 if (mode == Mode::RobinHood && distance(item.key, index) < i) break;
380 }
381 return result;
382 }
383 std::vector<Cost> costs() const {
384 std::vector<Cost> result;
385 if (mode == Mode::Chain) {
386 for (int index = 0; index < cap; ++index) {
387 const std::vector<Item>& bucket = buckets[index];
388 for (int i = 0; i < static_cast<int>(bucket.size()); ++i) result.push_back({bucket[i].key, i + 1, index});
389 }
390 } else {
391 for (int index = 0; index < slotCount(); ++index) if (const Item* item = std::get_if<Item>(&slots[index])) {
392 const int cost = mode == Mode::Cuckoo ? (index < cap ? 1 : 2)
393 : mode == Mode::RobinHood || mode == Mode::Linear ? distance(item->key, index) + 1 : search(item->key).probes;
394 result.push_back({item->key, cost, index});
395 }
396 }
397 return result;
398 }
399 Metrics metrics() const {
400 int longest = 0;
401 Integer total = 0;
402 for (const Cost& record : costs()) { longest = std::max(longest, record.cost); total += record.cost; }
403 return {longest, decimal(total, size)};
404 }
405 Lines dump() const {
406 Lines rows{"DUMP " + name(mode) + " cap=" + std::to_string(cap) + " size=" + std::to_string(size) + " tomb=" + std::to_string(tomb)};
407 if (mode == Mode::Chain) {
408 for (int index = 0; index < cap; ++index) {
409 Lines words{std::to_string(index)};
410 for (const Item& item : buckets[index]) words.push_back(pairText(item));
411 if (buckets[index].empty()) words.push_back("-");
412 rows.push_back(join(words));
413 }
414 } else {
415 for (int index = 0; index < slotCount(); ++index) {
416 const std::string prefix = mode == Mode::Cuckoo ? "T" + std::to_string(index / cap + 1) + " " + std::to_string(index % cap) : std::to_string(index);
417 const Item* item = std::get_if<Item>(&slots[index]);
418 std::string value = item ? pairText(*item) : content(slots[index]);
419 if (mode == Mode::RobinHood && item) value += " d=" + std::to_string(distance(item->key, index));
420 rows.push_back(prefix + " " + value);
421 }
422 }
423 return rows;
424 }
425 Lines map() const {
426 Lines rows{"MAP " + name(mode) + " cap=" + std::to_string(cap)};
427 std::vector<int> costBySlot(slotCount());
428 if (mode != Mode::Chain) for (const Cost& record : costs()) costBySlot[record.position] = record.cost;
429 for (int part = 0; part < (mode == Mode::Cuckoo ? 2 : 1); ++part) {
430 std::string characters;
431 for (int index = 0; index < cap; ++index) {
432 if (mode == Mode::Chain) characters += buckets[index].empty() ? '.' : gauge(static_cast<int>(buckets[index].size()));
433 else {
434 const int flat = part * cap + index;
435 const Slot& slot = slots[flat];
436 characters += empty(slot) ? '.' : std::holds_alternative<Tombstone>(slot) ? 'D'
437 : mode == Mode::Cuckoo ? '#' : gauge(costBySlot[flat] - 1);
438 }
439 }
440 for (int index = 0; index < cap; index += 40) {
441 std::ostringstream line;
442 if (mode == Mode::Cuckoo) line << "T" << part + 1 << ' ';
443 line << std::setw(5) << std::setfill('0') << index << ' ' << characters.substr(index, 40);
444 rows.push_back(line.str());
445 }
446 }
447 return rows;
448 }
449 Lines statistics() const {
450 const Metrics values = metrics();
451 return {"STATS " + name(mode) + " size=" + std::to_string(size) + " cap=" + std::to_string(cap)
452 + " slots=" + std::to_string(slotCount()) + " tomb=" + std::to_string(tomb) + " load=" + decimal(100LL * size, slotCount())
453 + "% longest=" + std::to_string(values.longest) + " avg=" + values.average + " probes=" + std::to_string(counters.probes) + " rehashes=" + std::to_string(counters.rehashes),
454 "OPS inserts=" + std::to_string(counters.inserts) + " updates=" + std::to_string(counters.updates)
455 + " hits=" + std::to_string(counters.hits) + " misses=" + std::to_string(counters.misses)
456 + " deletes=" + std::to_string(counters.deletes) + " absent=" + std::to_string(counters.absent)};
457 }
458 Lines scan(int cursor, int limit) const {
459 Lines keys;
460 int next = cursor;
461 while (next < slotCount() && static_cast<int>(keys.size()) < limit) {
462 if (mode == Mode::Chain) {
463 for (const Item& item : buckets[next]) keys.push_back(item.key);
464 } else if (const Item* item = std::get_if<Item>(&slots[next])) keys.push_back(item->key);
465 ++next;
466 }
467 if (next == slotCount()) next = 0;
468 Lines words{"SCAN next=" + std::to_string(next)};
469 append(words, keys);
470 return {join(words)};
471 }
472 Lines clusters() const {
473 std::vector<int> lengths;
474 if (mode == Mode::Chain) {
475 for (const std::vector<Item>& bucket : buckets) if (!bucket.empty()) lengths.push_back(static_cast<int>(bucket.size()));
476 } else {
477 lengths = runs(slots, 0, cap);
478 if (mode == Mode::Cuckoo) {
479 const std::vector<int> second = runs(slots, cap, cap);
480 lengths.insert(lengths.end(), second.begin(), second.end());
481 }
482 }
483 const int longest = lengths.empty() ? 0 : *std::max_element(lengths.begin(), lengths.end());
484 Lines rows{"CLUSTERS runs=" + std::to_string(lengths.size()) + " longest=" + std::to_string(longest)};
485 append(rows, histogram(lengths));
486 return rows;
487 }
488 Lines distribution() const {
489 std::vector<int> values;
490 for (const Cost& record : costs()) values.push_back(record.cost);
491 const int longest = values.empty() ? 0 : *std::max_element(values.begin(), values.end());
492 Lines rows{"DIST keys=" + std::to_string(size) + " longest=" + std::to_string(longest)};
493 append(rows, histogram(values));
494 return rows;
495 }

520 Lines bench(int capacity, int count, int length, Integer seed) const {
521 const Lines keys = generatedKeys(2 * count, length, seed);
522 Lines rows{"BENCH cap=" + std::to_string(capacity) + " count=" + std::to_string(count) + " len=" + std::to_string(length) + " seed=" + std::to_string(seed)};
523 for (int index = 0; index < 6; ++index) {
524 const Mode mode = static_cast<Mode>(index);
525 Table experiment(mode, capacity);
526 for (int i = 0; i < count; ++i) experiment.put(keys[i], i + 1, "PUT", maxLoad);
527 int misses = 0;
528 Integer probes = 0;
529 for (int i = count; i < 2 * count; ++i) {
530 const Search found = experiment.search(keys[i]);
531 if (!found.item) { ++misses; probes += found.probes; }
532 }
533 rows.push_back(name(mode) + " cap=" + std::to_string(experiment.cap) + " keys=" + std::to_string(experiment.size)
534 + " hit=" + experiment.metrics().average + " misses=" + std::to_string(misses) + " miss=" + decimal(probes, misses)
535 + " rehashes=" + std::to_string(experiment.counters.rehashes));
536 }
537 return rows;
538 }
539public:
540 Lines execute(const std::string& line) {
541 std::istringstream input(line);
542 std::string command, word;
543 input >> command;
544 Lines args;
545 while (input >> word) args.push_back(word);
546 if (std::find(commands.begin(), commands.end(), command) == commands.end()) throw CommandError("UNKNOWN COMMAND " + command);
547 if (command == "CREATE") {
548 const Mode mode = parseMode(args[0]);
549 const int capacity = std::stoi(args[1]);
550 checkCapacity(capacity);
551 table.emplace(mode, prime(capacity));
552 return {"CREATE " + name(mode) + " cap=" + std::to_string(table->cap)};
553 }
554 if (command == "MAXLOAD" || command == "MINLOAD") {
555 const int value = std::stoi(args[0]);
556 const bool maximum = command == "MAXLOAD";
557 if (maximum ? value < 10 || value > 95 : value > 40) throw CommandError("BAD LOAD");
558 if (maximum) maxLoad = value;
559 else minLoad = value;
560 return {command + " " + std::to_string(value)};
561 }
562 if (command == "TRACE") {
563 if (args[0] != "ON" && args[0] != "OFF") throw CommandError("BAD ARGUMENT");
564 tracing = args[0] == "ON";
565 return {"TRACE " + args[0]};
566 }
567 if (command == "HASH") {
568 const std::string& key = args[0];
569 const int capacity = std::stoi(args[1]);
570 checkCapacity(capacity);
571 return {"HASH " + key + " m=" + std::to_string(capacity) + " h31=" + std::to_string(hash(key, 31, capacity))
572 + " h37=" + std::to_string(hash(key, 37, capacity)) + " step=" + std::to_string(1 + hash(key, 37, capacity - 1))};
573 }
574 if (command == "BENCH") {
575 const int capacity = std::stoi(args[0]), count = std::stoi(args[1]), length = std::stoi(args[2]);
576 const Integer seed = std::stoll(args[3]);
577 checkCapacity(capacity);
578 checkGeneration(count, length, seed);
579 return bench(prime(capacity), count, length, seed);
580 }
581 if (!table) throw CommandError("NO TABLE");
582 Table& current = *table;
583 if (command == "PUT" || command == "ADD") return current.put(args[0], std::stoll(args[1]), command, maxLoad, tracing).rows;
584 if (command == "GET") return current.get(args[0], tracing);
585 if (command == "DEL") return current.erase(args[0], minLoad, maxLoad, tracing).rows;
586 if (command == "PROBE") {
587 const Search found = current.search(args[0], true);
588 Lines words{"PROBE " + args[0]};
589 append(words, found.path);
590 words.push_back(std::string(found.item ? "FOUND" : "MISS") + " probes=" + std::to_string(found.probes));
591 return {join(words)};
592 }
593 if (command == "GEN" || command == "GENDEL") {
594 const int count = std::stoi(args[0]), length = std::stoi(args[1]);
595 const Integer seed = std::stoll(args[2]);
596 checkGeneration(count, length, seed);
597 const Lines keys = generatedKeys(count, length, seed);
598 Lines rows;
599 int successes = 0;
600 for (int i = 0; i < count; ++i) {
601 if (command == "GEN") {
602 const PutResult result = current.put(keys[i], i + 1, "PUT", maxLoad);
603 successes += result.fresh;
604 append(rows, result.rehashes);
605 } else {
606 const DeleteResult result = current.erase(keys[i], minLoad, maxLoad);
607 successes += result.deleted;
608 append(rows, result.rehashes);
609 }
610 }
611 rows.push_back(command + " " + std::to_string(count) + (command == "GEN" ? " new=" : " deleted=")
612 + std::to_string(successes) + (command == "GEN" ? " updated=" : " missing=") + std::to_string(count - successes));
613 return rows;
614 }
615 if (command == "RESIZE" || command == "CONVERT") {
616 Mode mode = current.mode;
617 int capacity = current.cap;
618 if (command == "RESIZE") { capacity = std::stoi(args[0]); checkCapacity(capacity); capacity = prime(capacity); }
619 else mode = parseMode(args[0]);
620 const int slots = capacity * (mode == Mode::Cuckoo ? 2 : 1);
621 if (100LL * current.size > maxLoad * slots) throw CommandError("TOO SMALL");
622 std::unique_ptr<Table> replacement = Table::rebuild(current.items(), mode, capacity);
623 if (!replacement) throw CommandError("REBUILD FAILED");
624 const std::string row = command == "RESIZE" ? "RESIZE " + std::to_string(current.cap) + " -> " + std::to_string(capacity)
625 : "CONVERT " + name(current.mode) + " -> " + name(mode) + " cap=" + std::to_string(capacity);
626 current.adopt(std::move(*replacement));
627 return {row};
628 }
629 if (command == "CLEAR") {
630 const int size = current.size;
631 current.adopt(Table(current.mode, current.cap));
632 return {"CLEAR " + std::to_string(size)};
633 }
634 if (command == "DUMP") return current.dump();
635 if (command == "MAP") return current.map();
636 if (command == "STATS") return current.statistics();
637 if (command == "KEYS") {
638 Lines keys;
639 for (const Item& item : current.items()) keys.push_back(item.key);
640 std::sort(keys.begin(), keys.end());
641 Lines words{"KEYS " + std::to_string(current.size)};
642 append(words, keys);
643 return {join(words)};
644 }
645 if (command == "TOP") {
646 const int limit = std::stoi(args[0]);
647 if (limit < 1 || limit > 100) throw CommandError("BAD ARGUMENT");
648 std::vector<Cost> records = current.costs();
649 std::sort(records.begin(), records.end(), [](const Cost& a, const Cost& b) { return a.cost != b.cost ? a.cost > b.cost : a.key < b.key; });
650 const int count = std::min(limit, static_cast<int>(records.size()));
651 Lines rows{"TOP " + std::to_string(count)};
652 for (int i = 0; i < count; ++i) rows.push_back(records[i].key + " cost=" + std::to_string(records[i].cost) + " at=" + current.position(records[i].position));
653 return rows;
654 }
655 if (command == "SCAN") {
656 const int cursor = std::stoi(args[0]), limit = std::stoi(args[1]);
657 if (limit < 1 || limit > 100) throw CommandError("BAD ARGUMENT");
658 if (cursor >= current.slotCount()) throw CommandError("BAD CURSOR");
659 return current.scan(cursor, limit);
660 }
661 if (command == "CLUSTERS") return current.clusters();
662 if (command == "DIST") return current.distribution();
663 if (command == "COMPARE") {
664 const int capacity = std::stoi(args[0]);
665 checkCapacity(capacity);
666 return compare(prime(capacity));
667 }
668 throw std::logic_error("Unhandled command " + command);
669 }
670};

673 std::ios::sync_with_stdio(false);
674 std::cin.tie(nullptr);
675 int count;
676 if (!(std::cin >> count)) return 0;
677 std::string line;

679 Simulator simulator;

681 std::getline(std::cin, line);
682 try { for (const std::string& row : simulator.execute(line)) std::cout << row << '\n'; }
683 catch (const CommandError& error) { std::cout << "ERROR " << error.what() << '\n'; }
684 std::cout.flush();
685 }
