51fun parseMode(word: String) = Mode.entries.firstOrNull { it.name == word } ?: fail("BAD MODE $word")
52fun hash(key: String, base: Int, modulus: Int) = key.fold(0) { value, ch -> (value * base + ch.code) % modulus }
53fun prime(value: Int): Int {
54 var candidate = maxOf(value, 2)
55 while (true) {
56 var divisor = 2
57 while (divisor <= candidate / divisor && candidate % divisor != 0) divisor++
58 if (divisor > candidate / divisor) return candidate
59 candidate++
60 }
61}
62fun decimal(numerator: Long, denominator: Long): String {
63 val value = if (denominator == 0L) 0L else 100 * numerator / denominator
64 return "${value / 100}.${(value % 100).toString().padStart(2, '0')}"
65}
66fun gauge(value: Int) = if (value < 10) ('0'.code + value).toChar() else '+'
67fun histogram(values: List<Int>) = values.groupingBy { it }.eachCount().toSortedMap().map { (value, count) ->
68 "$value $count ${"#".repeat(minOf(count, 50))}${if (count > 50) "+" else ""}"
69}
70fun generate(count: Int, length: Int, seed: Long): List<String> {
71 var state = seed
72 return List(count) {
73 buildString {
74 repeat(length) {
75 state = 48271 * state % 2147483647
76 append(('a'.code + state % 26).toInt().toChar())
77 }
78 }

89class HashTable(var mode: Mode, var cap: Int) {
90 var size = 0
91 var tomb = 0
92 val counters = Counters()
93 var buckets: List<MutableList<Item>> = if (mode == Mode.CHAIN) List(cap) { mutableListOf() } else emptyList()
94 var slots = Array(if (mode == Mode.CHAIN) 0 else slotCount) { Slot.EMPTY }
95 val slotCount get() = cap * if (mode == Mode.CUCKOO) 2 else 1
96
97 fun position(index: Int) = if (mode == Mode.CUCKOO) "T${index / cap + 1}.${index % cap}" else index.toString()
98 fun distance(item: Item, index: Int) = (index + cap - hash(item.key, 31, cap)) % cap
99 fun items(): List<Item> = if (mode == Mode.CHAIN) buckets.flatten() else slots.mapNotNull { it.item }
100 fun search(key: String, trace: Boolean = false): Search {
101 val result = Search()
102 fun observe(index: Int, content: String) {
103 result.probes++
104 if (trace) result.path += "${position(index)}:$content"
105 }
106 val home = hash(key, 31, cap)
107 if (mode == Mode.CHAIN) {
108 for ((index, item) in buckets[home].withIndex()) {
109 observe(home, item.key)
110 if (item.key == key) {
111 result.item = item
112 result.position = home
113 result.chainIndex = index
114 break
115 }
116 }
117 return result
118 }
119 val step = if (mode == Mode.DOUBLE) 1 + hash(key, 37, cap - 1) else 1
120 for (i in 0 until if (mode == Mode.CUCKOO) 2 else cap) {
121 val index = if (mode == Mode.CUCKOO) {
122 if (i == 0) home else cap + hash(key, 37, cap)
123 } else ((home + if (mode == Mode.QUADRATIC) i.toLong() * i else i.toLong() * step) % cap).toInt()
124 val slot = slots[index]
125 observe(index, slot.content)
126 if (slot.item?.key == key) {
127 result.item = slot.item
128 result.position = index
129 break
130 }
131 if (mode == Mode.CUCKOO) continue
132 if (slot.empty) { result.available = result.available ?: index; break }
133 if (slot.deleted) { result.available = result.available ?: index; continue }
134 if (mode == Mode.ROBINHOOD && distance(slot.item!!, index) < i) break
135 }
136 return result
137 }
138 fun place(item: Item, initial: Search? = null, trace: Boolean = false): Placement {
139 val result = Placement()
140 val home = hash(item.key, 31, cap)
141 if (mode == Mode.CHAIN) {
142 buckets[home].add(item)
143 result.position = home
144 return result
145 }
146 if (mode == Mode.LINEAR || mode == Mode.QUADRATIC || mode == Mode.DOUBLE) {
147 val index = (initial ?: search(item.key)).available
148 if (index == null) { result.pending = item; return result }
149 result.reused = slots[index].deleted
150 if (result.reused) tomb--
151 slots[index] = Slot(item)
152 result.position = index
153 return result
154 }
155 var carried = item
156 if (mode == Mode.ROBINHOOD) {
157 var index = home
158 var traveled = 0
159 var from: Int? = null
160 while (true) {
161 val resident = slots[index].item
162 if (resident == null) {
163 if (trace && from != null) result.moves += "SHIFT ${carried.key} $from -> $index"
164 slots[index] = Slot(carried)
165 result.position = result.position ?: index
166 return result
167 }
168 val residentDistance = distance(resident, index)
169 if (residentDistance < traveled) {
170 if (trace && from != null) result.moves += "SHIFT ${carried.key} $from -> $index"
171 slots[index] = Slot(carried)
172 carried = resident
173 result.position = result.position ?: index
174 from = index
175 traveled = residentDistance
176 result.movements++
177 }
178 index = (index + 1) % cap
179 traveled++
180 }
181 }
182 var part = 0
183 while (true) {
184 val index = part * cap + hash(carried.key, if (part == 0) 31 else 37, cap)
185 val resident = slots[index].item
186 if (resident == null) {
187 slots[index] = Slot(carried)
188 result.position = search(item.key).position
189 return result
190 }
191 if (result.movements == cap) { result.pending = carried; return result }
192 if (trace) result.moves += "KICK ${resident.key} ${position(index)}"
193 slots[index] = Slot(carried)
194 carried = resident
195 result.movements++
196 part = 1 - part
197 }
198 }
199 fun remove(found: Search, trace: Boolean): Removal {
200 var shifted = 0
201 val moves = mutableListOf<String>()
202 if (mode == Mode.CHAIN) buckets[found.position].removeAt(found.chainIndex)
203 else if (mode == Mode.LINEAR || mode == Mode.QUADRATIC || mode == Mode.DOUBLE) {
204 slots[found.position] = Slot.TOMB
205 tomb++
206 } else {
207 slots[found.position] = Slot.EMPTY
208 if (mode == Mode.ROBINHOOD) {
209 var source = (found.position + 1) % cap
210 while (true) {
211 val item = slots[source].item ?: break
212 if (distance(item, source) == 0) break
213 val destination = (source + cap - 1) % cap
214 if (trace) moves += "SHIFT ${item.key} $source -> $destination"
215 slots[destination] = slots[source]
216 slots[source] = Slot.EMPTY
217 shifted++
218 source = (source + 1) % cap
219 }
220 }
221 }
222 size--
223 return Removal(shifted, moves)
224 }
225 companion object {
226 fun rebuild(mode: Mode, cap: Int, items: List<Item>): HashTable? {
227 val trial = HashTable(mode, cap)
228 for (item in items) {
229 if (trial.place(item).position == null) return null
230 trial.size++
231 }
232 return trial
233 }
234 }
235 fun adopt(replacement: HashTable) {
236 mode = replacement.mode
237 cap = replacement.cap
238 size = replacement.size
239 tomb = replacement.tomb
240 buckets = replacement.buckets
241 slots = replacement.slots
242 }
243 fun automatic(capacity: Int, reason: String, pending: Item? = null): String {
244 var target = capacity
245 val old = cap
246 val listed = items().toMutableList()
247 if (pending != null) listed += pending
248 while (true) {
249 val trial = rebuild(mode, target, listed)
250 if (trial != null) {
251 adopt(trial)
252 counters.rehashes++
253 return "REHASH $reason $old -> $cap"
254 }
255 target = prime(2 * target)
256 }
257 }
258 fun loadCheck(maximum: Int): String? {
259 val threshold = maximum.toLong() * slotCount
260 if ((size + tomb) * 100L <= threshold) return null
261 val grow = size * 200L > threshold
262 return automatic(if (grow) prime(2 * cap) else cap, if (grow) "load" else "tomb")
263 }
264 fun shrinkCheck(minimum: Int, maximum: Int): String? {
265 val target = prime(cap / 2)
266 val count = target * if (mode == Mode.CUCKOO) 2 else 1
267 if (size * 100L >= minimum.toLong() * slotCount || target >= cap || size * 100L > maximum.toLong() * count) return null
268 val trial = rebuild(mode, target, items()) ?: return null
269 val old = cap
270 adopt(trial)
271 counters.rehashes++
272 return "REHASH shrink $old -> $cap"
273 }
274 fun pathLine(found: Search) = "PATH " + if (found.path.isEmpty()) "none" else found.path.joinToString(" ")
275 fun put(item: Item, add: Boolean, maximum: Int, reporting: Reporting): Mutation {
276 val found = search(item.key, reporting.trace)
277 counters.probes += found.probes
278 val command = if (add) "ADD" else "PUT"
279 val lines = mutableListOf<String>()
280 val old = found.item
281 if (old != null) {
282 val stored = Item(item.key, if (add) old.value + item.value else item.value)
283 if (mode == Mode.CHAIN) buckets[found.position][found.chainIndex] = stored else slots[found.position] = Slot(stored)
284 counters.updates++
285 if (reporting.details) lines += "$command $stored UPDATED old=${old.value} at=${position(found.position)} probes=${found.probes}"
286 if (reporting.trace) lines += pathLine(found)
287 return Mutation(false, lines)
288 }
289 val placed = place(item, found, reporting.trace)
290 counters.inserts++
291 val index = placed.position
292 if (reporting.details) {
293 var row = "$command $item NEW"
294 if (index != null) {
295 row += " at=${position(index)} probes=${found.probes}"
296 if (placed.reused) row += " reused"
297 if (mode == Mode.ROBINHOOD) row += " shifted=${placed.movements}"
298 if (mode == Mode.CUCKOO) row += " kicks=${placed.movements}"
299 } else {
300 row += " probes=${found.probes}"
301 row += if (mode == Mode.QUADRATIC) " full" else " kicks=${placed.movements} failed"
302 }
303 lines += row
304 }
305 if (reporting.trace) { lines += pathLine(found); lines += placed.moves }
306 if (index != null) size++ else {
307 val rehash = automatic(prime(2 * cap), if (mode == Mode.QUADRATIC) "full" else "kicks", placed.pending)
308 if (reporting != Reporting.SILENT) lines += rehash
309 }
310 loadCheck(maximum)?.let { if (reporting != Reporting.SILENT) lines += it }
311 return Mutation(true, lines)
312 }
313 fun delete(key: String, minimum: Int, maximum: Int, reporting: Reporting): Mutation {
314 val found = search(key, reporting.trace)
315 counters.probes += found.probes
316 val lines = mutableListOf<String>()
317 if (found.item == null) {
318 counters.absent++
319 if (reporting.details) lines += "DEL $key NOT FOUND probes=${found.probes}"
320 if (reporting.trace) lines += pathLine(found)
321 return Mutation(false, lines)
322 }
323 val removed = remove(found, reporting.trace)
324 counters.deletes++
325 if (reporting.details) {
326 var row = "DEL $key OK at=${position(found.position)} probes=${found.probes}"
327 if (mode == Mode.ROBINHOOD) row += " shifted=${removed.shifted}"
328 lines += row
329 }
330 if (reporting.trace) { lines += pathLine(found); lines += removed.moves }
331 shrinkCheck(minimum, maximum)?.let { if (reporting != Reporting.SILENT) lines += it }
332 return Mutation(true, lines)
333 }
334 fun get(key: String, trace: Boolean): List<String> {
335 val found = search(key, trace)
336 counters.probes += found.probes
337 val item = found.item
338 val row = if (item != null) {
339 counters.hits++
340 "GET $item at=${position(found.position)} probes=${found.probes}"
341 } else {
342 counters.misses++
343 "GET $key NOT FOUND probes=${found.probes}"
344 }
345 return if (trace) listOf(row, pathLine(found)) else listOf(row)
346 }
347 fun costs(): List<Cost> = buildList {
348 if (mode == Mode.CHAIN) {
349 for ((index, bucket) in buckets.withIndex()) {
350 for ((place, item) in bucket.withIndex()) add(Cost(item.key, index, place + 1))
351 }
352 } else {
353 for ((index, slot) in slots.withIndex()) {
354 val item = slot.item ?: continue
355 val cost = when (mode) {
356 Mode.CUCKOO -> index / cap + 1
357 Mode.LINEAR, Mode.ROBINHOOD -> distance(item, index) + 1
358 else -> search(item.key).probes
359 }
360 add(Cost(item.key, index, cost))
361 }
362 }
363 }
364 fun metrics(): Metrics {
365 val values = costs()
366 return Metrics(values.maxOfOrNull { it.probes } ?: 0, decimal(values.sumOf { it.probes.toLong() }, size.toLong()))
367 }
368 fun dump(): List<String> = buildList {
369 add("DUMP $mode cap=$cap size=${this@HashTable.size} tomb=$tomb")
370 if (mode == Mode.CHAIN) {
371 for ((index, bucket) in buckets.withIndex()) add("$index " + if (bucket.isEmpty()) "-" else bucket.joinToString(" "))
372 } else {
373 for ((index, slot) in slots.withIndex()) {
374 val prefix = if (mode == Mode.CUCKOO) "T${index / cap + 1} ${index % cap}" else index.toString()
375 var content = slot.item?.toString() ?: slot.content
376 if (mode == Mode.ROBINHOOD && slot.item != null) content += " d=${distance(slot.item, index)}"
377 add("$prefix $content")
378 }
379 }
380 }
381 fun map(): List<String> = buildList {
382 add("MAP $mode cap=$cap")
383 val costBySlot = IntArray(slotCount)
384 if (mode != Mode.CHAIN) for (cost in costs()) costBySlot[cost.position] = cost.probes
385 for (part in 0 until if (mode == Mode.CUCKOO) 2 else 1) {
386 for (start in 0 until cap step 40) {
387 val text = buildString {
388 for (index in start until minOf(start + 40, cap)) {
389 append(if (mode == Mode.CHAIN) {
390 if (buckets[index].isEmpty()) '.' else gauge(buckets[index].size)
391 } else {
392 val position = part * cap + index
393 val slot = slots[position]
394 when {
395 slot.empty -> '.'
396 slot.deleted -> 'D'
397 mode == Mode.CUCKOO -> '#'
398 else -> gauge(costBySlot[position] - 1)
399 }
400 })
401 }
402 }
403 val prefix = if (mode == Mode.CUCKOO) "T${part + 1} " else ""
404 add("$prefix${start.toString().padStart(5, '0')} $text")
405 }
406 }
407 }
408 fun statistics(): List<String> {
409 val values = metrics()
410 val c = counters
411 return listOf("STATS $mode size=$size cap=$cap slots=$slotCount tomb=$tomb load=${decimal(100L * size, slotCount.toLong())}% " +
412 "longest=${values.longest} avg=${values.average} probes=${c.probes} rehashes=${c.rehashes}",
413 "OPS inserts=${c.inserts} updates=${c.updates} hits=${c.hits} misses=${c.misses} deletes=${c.deletes} absent=${c.absent}")
414 }
415 fun scan(cursor: Int, limit: Int): String {
416 var next = cursor
417 val keys = mutableListOf<String>()
418 while (next < slotCount && keys.size < limit) {
419 if (mode == Mode.CHAIN) keys += buckets[next].map { it.key }
420 else slots[next].item?.let { keys += it.key }
421 next++
422 }
423 if (next == slotCount) next = 0
424 return "SCAN next=$next" + if (keys.isEmpty()) "" else " " + keys.joinToString(" ")
425 }
426 fun clusters(): List<String> {
427 val runs = mutableListOf<Int>()
428 if (mode == Mode.CHAIN) runs += buckets.filter { it.isNotEmpty() }.map { it.size }
429 else {
430 for (base in slots.indices step cap) {
431 val empty = (0 until cap).firstOrNull { slots[base + it].empty }
432 if (empty == null) { runs += cap; continue }
433 var length = 0
434 for (offset in 1..cap) {
435 if (!slots[base + (empty + offset) % cap].empty) length++
436 else if (length > 0) { runs += length; length = 0 }
437 }
438 }
439 }
440 return listOf("CLUSTERS runs=${runs.size} longest=${runs.maxOrNull() ?: 0}") + histogram(runs)
441 }
442 fun distribution(): List<String> {
443 val values = costs().map { it.probes }
444 return listOf("DIST keys=$size longest=${values.maxOrNull() ?: 0}") + histogram(values)
445 }
446}

448fun compare(table: HashTable, cap: Int, maximum: Int): List<String> = buildList {
449 val items = table.items().sortedBy { it.key }
450 add("COMPARE keys=${items.size} cap=$cap")
451 for (mode in Mode.entries) {
452 val trial = HashTable(mode, cap)
453 for (item in items) trial.put(item, false, maximum, Reporting.SILENT)
454 val values = trial.metrics()
455 add("$mode cap=${trial.cap} longest=${values.longest} avg=${values.average} probes=${trial.counters.probes} rehashes=${trial.counters.rehashes}")
456 }
457}
458fun benchmark(cap: Int, count: Int, length: Int, seed: Long, maximum: Int): List<String> = buildList {
459 val keys = generate(2 * count, length, seed)
460 add("BENCH cap=$cap count=$count len=$length seed=$seed")
461 for (mode in Mode.entries) {
462 val trial = HashTable(mode, cap)
463 for (index in 0 until count) trial.put(Item(keys[index], index + 1L), false, maximum, Reporting.SILENT)
464 var misses = 0L
465 var probes = 0L
466 for (key in keys.subList(count, 2 * count)) {
467 val found = trial.search(key)
468 if (found.item == null) { misses++; probes += found.probes }
469 }
470 add("$mode cap=${trial.cap} keys=${trial.size} hit=${trial.metrics().average} misses=$misses miss=${decimal(probes, misses)} rehashes=${trial.counters.rehashes}")
471 }

474class Simulator {
475 var table: HashTable? = null
476 var maximum = 75
477 var minimum = 0
478 var trace = false
479 companion object {
480 val COMMANDS = setOf("CREATE", "MAXLOAD", "MINLOAD", "TRACE", "HASH", "PUT", "ADD", "GET", "DEL", "PROBE", "GEN", "GENDEL", "RESIZE", "CONVERT",
481 "CLEAR", "DUMP", "MAP", "STATS", "KEYS", "TOP", "SCAN", "CLUSTERS", "DIST", "COMPARE", "BENCH")
482 }
483 fun execute(line: String): List<String> {
484 val words = line.split(' ')
485 val command = words[0]
486 fun number(index: Int) = words[index].toLong()
487 if (command !in COMMANDS) fail("UNKNOWN COMMAND $command")
488 when (command) {
489 "CREATE" -> {
490 val mode = parseMode(words[1])
491 val cap = prime(capacity(number(2)))
492 table = HashTable(mode, cap)
493 return listOf("CREATE $mode cap=$cap")
494 }
495 "MAXLOAD", "MINLOAD" -> {
496 val value = number(1)
497 if (value !in if (command == "MAXLOAD") 10L..95L else 0L..40L) fail("BAD LOAD")
498 if (command == "MAXLOAD") maximum = value.toInt() else minimum = value.toInt()
499 return listOf("$command $value")
500 }
501 "TRACE" -> {
502 if (words[1] != "ON" && words[1] != "OFF") fail("BAD ARGUMENT")
503 trace = words[1] == "ON"
504 return listOf("TRACE ${words[1]}")
505 }
506 "HASH" -> {
507 val cap = capacity(number(2))
508 val key = words[1]
509 return listOf("HASH $key m=$cap h31=${hash(key, 31, cap)} h37=${hash(key, 37, cap)} step=${1 + hash(key, 37, cap - 1)}")
510 }
511 "BENCH" -> {
512 val cap = prime(capacity(number(1)))
513 val count = number(2); val length = number(3); val seed = number(4)
514 checkGeneration(count, length, seed)
515 return benchmark(cap, count.toInt(), length.toInt(), seed, maximum)
516 }
517 }
518 val current = table ?: fail("NO TABLE")
519 val reporting = if (trace) Reporting.TRACE else Reporting.FULL
520 return when (command) {
521 "PUT", "ADD" -> current.put(Item(words[1], number(2)), command == "ADD", maximum, reporting).lines
522 "GET" -> current.get(words[1], trace)
523 "DEL" -> current.delete(words[1], minimum, maximum, reporting).lines
524 "PROBE" -> {
525 val found = current.search(words[1], true)
526 val path = if (found.path.isEmpty()) "" else " " + found.path.joinToString(" ")
527 listOf("PROBE ${words[1]}$path ${if (found.item == null) "MISS" else "FOUND"} probes=${found.probes}")
528 }
529 "GEN", "GENDEL" -> {
530 val count = number(1); val length = number(2); val seed = number(3)
531 checkGeneration(count, length, seed)
532 val lines = mutableListOf<String>()
533 var changed = 0
534 for ((index, key) in generate(count.toInt(), length.toInt(), seed).withIndex()) {
535 val result = if (command == "GEN") current.put(Item(key, index + 1L), false, maximum, Reporting.REHASHES)
536 else current.delete(key, minimum, maximum, Reporting.REHASHES)
537 if (result.changed) changed++
538 lines += result.lines
539 }
540 lines += if (command == "GEN") "GEN $count new=$changed updated=${count - changed}"
541 else "GENDEL $count deleted=$changed missing=${count - changed}"
542 lines
543 }
544 "RESIZE", "CONVERT" -> {
545 val cap = if (command == "RESIZE") prime(capacity(number(1))) else current.cap
546 val mode = if (command == "CONVERT") parseMode(words[1]) else current.mode
547 val slots = cap * if (mode == Mode.CUCKOO) 2 else 1
548 if (current.size * 100L > maximum.toLong() * slots) fail("TOO SMALL")
549 val replacement = HashTable.rebuild(mode, cap, current.items()) ?: fail("REBUILD FAILED")
550 val row = if (command == "RESIZE") "RESIZE ${current.cap} -> $cap" else "CONVERT ${current.mode} -> $mode cap=$cap"
551 current.adopt(replacement)
552 listOf(row)
553 }
554 "CLEAR" -> {
555 val size = current.size
556 current.adopt(HashTable(current.mode, current.cap))
557 listOf("CLEAR $size")
558 }
559 "DUMP" -> current.dump()
560 "MAP" -> current.map()
561 "STATS" -> current.statistics()
562 "KEYS" -> {
563 val keys = current.items().map { it.key }.sorted()
564 listOf("KEYS ${keys.size}" + if (keys.isEmpty()) "" else " " + keys.joinToString(" "))
565 }
566 "TOP" -> {
567 val limit = number(1)
568 if (limit !in 1L..100L) fail("BAD ARGUMENT")
569 val selected = current.costs().sortedWith(compareByDescending<Cost> { it.probes }.thenBy { it.key }).take(limit.toInt())
570 listOf("TOP ${selected.size}") + selected.map { "${it.key} cost=${it.probes} at=${current.position(it.position)}" }
571 }
572 "SCAN" -> {
573 val cursor = number(1); val limit = number(2)
574 if (limit !in 1L..100L) fail("BAD ARGUMENT")
575 if (cursor >= current.slotCount) fail("BAD CURSOR")
576 listOf(current.scan(cursor.toInt(), limit.toInt()))
577 }
578 "CLUSTERS" -> current.clusters()
579 "DIST" -> current.distribution()
580 "COMPARE" -> compare(current, prime(capacity(number(1))), maximum)
581 else -> error("Unreachable command")
582 }
583 }
584}

588 val output = System.out.bufferedWriter()
589 val count = input.readLine()?.toInt() ?: return
590 val simulator = Simulator()
