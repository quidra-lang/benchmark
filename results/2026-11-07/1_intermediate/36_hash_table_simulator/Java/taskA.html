42 static Mode parseMode(String word) {
43 for (Mode mode : Mode.values()) if (mode.name().equals(word)) return mode;
44 throw new CommandError("BAD MODE " + word);

46 static int hash(String key, int base, int modulus) {
47 int value = 0;
48 for (int i = 0; i < key.length(); i++) value = (value * base + key.charAt(i)) % modulus;
49 return value;
50 }
51 static int prime(int value) {
52 for (int candidate = Math.max(value, 2); ; candidate++) {
53 boolean valid = true;
54 for (int divisor = 2; divisor <= candidate / divisor; divisor++) {
55 if (candidate % divisor == 0) { valid = false; break; }
56 }
57 if (valid) return candidate;
58 }
59 }
60 static String decimal(long numerator, long denominator) {
61 long value = denominator == 0 ? 0 : 100 * numerator / denominator;
62 return "%d.%02d".formatted(value / 100, value % 100);
63 }
64 static char gauge(int value) { return value < 10 ? (char) ('0' + value) : '+'; }
65 static List<String> histogram(List<Integer> values) {
66 var frequencies = new TreeMap<Integer, Integer>();
67 for (int value : values) frequencies.merge(value, 1, Integer::sum);
68 var lines = new ArrayList<String>();
69 frequencies.forEach((value, count) -> lines.add(value + " " + count + " " + "#".repeat(Math.min(50, count)) + (count > 50 ? "+" : "")));
70 return lines;
71 }

92 static final class HashTable {
93 Mode mode;
94 int cap, size, tomb;
95 List<List<Item>> buckets;
96 Slot[] slots;
97 final Counters counters = new Counters();

99 HashTable(Mode mode, int cap) {
100 this.mode = mode;
101 this.cap = cap;
102 buckets = new ArrayList<>();
103 if (mode == Mode.CHAIN) {
104 for (int i = 0; i < cap; i++) buckets.add(new ArrayList<>());
105 slots = new Slot[0];
106 } else {
107 slots = new Slot[slotCount()];
108 Arrays.fill(slots, Slot.EMPTY);
109 }
110 }
111 int slotCount() { return cap * (mode == Mode.CUCKOO ? 2 : 1); }
112 String position(int index) { return mode == Mode.CUCKOO ? "T" + (index / cap + 1) + "." + index % cap : Integer.toString(index); }
113 int distance(Item item, int index) { return (index + cap - hash(item.key, 31, cap)) % cap; }
114 List<Item> items() {
115 var items = new ArrayList<Item>(size);
116 if (mode == Mode.CHAIN) for (var bucket : buckets) items.addAll(bucket);
117 else for (Slot slot : slots) if (slot.item != null) items.add(slot.item);
118 return items;
119 }
120 void observe(Search search, int position, String content, boolean trace) {
121 search.probes++;
122 if (trace) search.path.add(position(position) + ":" + content);
123 }
124 Search search(String key, boolean trace) {
125 var result = new Search();
126 int home = hash(key, 31, cap);
127 if (mode == Mode.CHAIN) {
128 var bucket = buckets.get(home);
129 for (int index = 0; index < bucket.size(); index++) {
130 Item item = bucket.get(index);
131 observe(result, home, item.key, trace);
132 if (item.key.equals(key)) {
133 result.item = item;
134 result.position = home;
135 result.chainIndex = index;
136 break;
137 }
138 }
139 return result;
140 }
141 int step = mode == Mode.DOUBLE ? 1 + hash(key, 37, cap - 1) : 1;
142 int limit = mode == Mode.CUCKOO ? 2 : cap;
143 for (int i = 0; i < limit; i++) {
144 int index = mode == Mode.CUCKOO ? (i == 0 ? home : cap + hash(key, 37, cap))
145 : (int) ((home + (mode == Mode.QUADRATIC ? (long) i * i : (long) i * step)) % cap);
146 Slot slot = slots[index];
147 observe(result, index, slot.content(), trace);
148 if (slot.item != null && slot.item.key.equals(key)) {
149 result.item = slot.item;
150 result.position = index;
151 break;
152 }
153 if (mode == Mode.CUCKOO) continue;
154 if (slot.empty()) { if (result.available < 0) result.available = index; break; }
155 if (slot.deleted) { if (result.available < 0) result.available = index; continue; }
156 if (mode == Mode.ROBINHOOD && distance(slot.item, index) < i) break;
157 }
158 return result;
159 }
160 Placement place(Item item, Search initial, boolean trace) {
161 var result = new Placement();
162 int home = hash(item.key, 31, cap);
163 if (mode == Mode.CHAIN) {
164 buckets.get(home).add(item);
165 result.position = home;
166 return result;
167 }
168 if (mode == Mode.LINEAR || mode == Mode.QUADRATIC || mode == Mode.DOUBLE) {
169 int index = (initial == null ? search(item.key, false) : initial).available;
170 if (index < 0) { result.pending = item; return result; }
171 result.reused = slots[index].deleted;
172 if (result.reused) tomb--;
173 slots[index] = new Slot(item);
174 result.position = index;
175 return result;
176 }
177 Item carried = item;
178 if (mode == Mode.ROBINHOOD) {
179 int index = home, traveled = 0, from = -1;
180 while (true) {
181 Item resident = slots[index].item;
182 if (resident == null) {
183 if (trace && from >= 0) result.moves.add("SHIFT " + carried.key + " " + from + " -> " + index);
184 slots[index] = new Slot(carried);
185 if (result.position < 0) result.position = index;
186 return result;
187 }
188 int residentDistance = distance(resident, index);
189 if (residentDistance < traveled) {
190 if (trace && from >= 0) result.moves.add("SHIFT " + carried.key + " " + from + " -> " + index);
191 slots[index] = new Slot(carried);
192 carried = resident;
193 if (result.position < 0) result.position = index;
194 from = index;
195 traveled = residentDistance;
196 result.movements++;
197 }
198 index = (index + 1) % cap;
199 traveled++;
200 }
201 }
202 int part = 0;
203 while (true) {
204 int index = part * cap + hash(carried.key, part == 0 ? 31 : 37, cap);
205 Item resident = slots[index].item;
206 if (resident == null) {
207 slots[index] = new Slot(carried);
208 result.position = search(item.key, false).position;
209 return result;
210 }
211 if (result.movements == cap) { result.pending = carried; return result; }
212 if (trace) result.moves.add("KICK " + resident.key + " " + position(index));
213 slots[index] = new Slot(carried);
214 carried = resident;
215 result.movements++;
216 part = 1 - part;
217 }
218 }
219 Removal remove(Search found, boolean trace) {
220 int shifted = 0;
221 var moves = new ArrayList<String>();
222 if (mode == Mode.CHAIN) buckets.get(found.position).remove(found.chainIndex);
223 else if (mode == Mode.LINEAR || mode == Mode.QUADRATIC || mode == Mode.DOUBLE) {
224 slots[found.position] = Slot.TOMB;
225 tomb++;
226 } else {
227 slots[found.position] = Slot.EMPTY;
228 if (mode == Mode.ROBINHOOD) {
229 int source = (found.position + 1) % cap;
230 while (slots[source].item != null && distance(slots[source].item, source) > 0) {
231 int destination = (source + cap - 1) % cap;
232 if (trace) moves.add("SHIFT " + slots[source].item.key + " " + source + " -> " + destination);
233 slots[destination] = slots[source];
234 slots[source] = Slot.EMPTY;
235 shifted++;
236 source = (source + 1) % cap;
237 }
238 }
239 }
240 size--;
241 return new Removal(shifted, moves);
242 }
243 static HashTable rebuild(Mode mode, int cap, List<Item> items) {
244 var trial = new HashTable(mode, cap);
245 for (Item item : items) {
246 if (trial.place(item, null, false).position < 0) return null;
247 trial.size++;
248 }
249 return trial;
250 }
251 void adopt(HashTable replacement) {
252 mode = replacement.mode;
253 cap = replacement.cap;
254 size = replacement.size;
255 tomb = replacement.tomb;
256 buckets = replacement.buckets;
257 slots = replacement.slots;
258 }
259 String automatic(int capacity, String reason, Item pending) {
260 int old = cap;
261 var items = items();
262 if (pending != null) items.add(pending);
263 while (true) {
264 HashTable trial = rebuild(mode, capacity, items);
265 if (trial != null) {
266 adopt(trial);
267 counters.rehashes++;
268 return "REHASH " + reason + " " + old + " -> " + cap;
269 }
270 capacity = prime(2 * capacity);
271 }
272 }
273 String loadCheck(int maximum) {
274 long threshold = (long) maximum * slotCount();
275 if ((size + tomb) * 100L <= threshold) return null;
276 boolean grow = size * 200L > threshold;
277 return automatic(grow ? prime(2 * cap) : cap, grow ? "load" : "tomb", null);
278 }
279 String shrinkCheck(int minimum, int maximum) {
280 int capacity = prime(cap / 2), slots = capacity * (mode == Mode.CUCKOO ? 2 : 1);
281 if (size * 100L >= (long) minimum * slotCount() || capacity >= cap || size * 100L > (long) maximum * slots) return null;
282 HashTable trial = rebuild(mode, capacity, items());
283 if (trial == null) return null;
284 int old = cap;
285 adopt(trial);
286 counters.rehashes++;
287 return "REHASH shrink " + old + " -> " + cap;
288 }
289 static String pathLine(Search found) { return "PATH " + (found.path.isEmpty() ? "none" : String.join(" ", found.path)); }
290 Mutation put(Item item, boolean add, int maximum, Reporting reporting) {
291 Search found = search(item.key, reporting.trace());
292 counters.probes += found.probes;
293 String command = add ? "ADD" : "PUT";
294 var lines = new ArrayList<String>();
295 if (found.item != null) {
296 long old = found.item.value;
297 Item stored = new Item(item.key, add ? old + item.value : item.value);
298 if (mode == Mode.CHAIN) buckets.get(found.position).set(found.chainIndex, stored);
299 else slots[found.position] = new Slot(stored);
300 counters.updates++;
301 if (reporting.details()) lines.add(command + " " + stored + " UPDATED old=" + old + " at=" + position(found.position) + " probes=" + found.probes);
302 if (reporting.trace()) lines.add(pathLine(found));
303 return new Mutation(false, lines);
304 }
305 Placement placed = place(item, found, reporting.trace());
306 counters.inserts++;
307 if (reporting.details()) {
308 String row = command + " " + item + " NEW";
309 if (placed.position >= 0) {
310 row += " at=" + position(placed.position) + " probes=" + found.probes;
311 if (placed.reused) row += " reused";
312 if (mode == Mode.ROBINHOOD) row += " shifted=" + placed.movements;
313 if (mode == Mode.CUCKOO) row += " kicks=" + placed.movements;
314 } else {
315 row += " probes=" + found.probes;
316 row += mode == Mode.QUADRATIC ? " full" : " kicks=" + placed.movements + " failed";
317 }
318 lines.add(row);
319 }
320 if (reporting.trace()) { lines.add(pathLine(found)); lines.addAll(placed.moves); }
321 if (placed.position >= 0) size++;
322 else {
323 String rehash = automatic(prime(2 * cap), mode == Mode.QUADRATIC ? "full" : "kicks", placed.pending);
324 if (reporting != Reporting.SILENT) lines.add(rehash);
325 }
326 String rehash = loadCheck(maximum);
327 if (rehash != null && reporting != Reporting.SILENT) lines.add(rehash);
328 return new Mutation(true, lines);
329 }
330 Mutation delete(String key, int minimum, int maximum, Reporting reporting) {
331 Search found = search(key, reporting.trace());
332 counters.probes += found.probes;
333 var lines = new ArrayList<String>();
334 if (found.item == null) {
335 counters.absent++;
336 if (reporting.details()) lines.add("DEL " + key + " NOT FOUND probes=" + found.probes);
337 if (reporting.trace()) lines.add(pathLine(found));
338 return new Mutation(false, lines);
339 }
340 Removal removed = remove(found, reporting.trace());
341 counters.deletes++;
342 if (reporting.details()) {
343 String row = "DEL " + key + " OK at=" + position(found.position) + " probes=" + found.probes;
344 if (mode == Mode.ROBINHOOD) row += " shifted=" + removed.shifted;
345 lines.add(row);
346 }
347 if (reporting.trace()) { lines.add(pathLine(found)); lines.addAll(removed.moves); }
348 String rehash = shrinkCheck(minimum, maximum);
349 if (rehash != null && reporting != Reporting.SILENT) lines.add(rehash);
350 return new Mutation(true, lines);
351 }
352 List<String> get(String key, boolean trace) {
353 Search found = search(key, trace);
354 counters.probes += found.probes;
355 String row;
356 if (found.item != null) {
357 counters.hits++;
358 row = "GET " + found.item + " at=" + position(found.position) + " probes=" + found.probes;
359 } else {
360 counters.misses++;
361 row = "GET " + key + " NOT FOUND probes=" + found.probes;
362 }
363 var lines = new ArrayList<>(List.of(row));
364 if (trace) lines.add(pathLine(found));
365 return lines;
366 }
367 List<Cost> costs() {
368 var costs = new ArrayList<Cost>(size);
369 if (mode == Mode.CHAIN) {
370 for (int position = 0; position < cap; position++) {
371 var bucket = buckets.get(position);
372 for (int index = 0; index < bucket.size(); index++) costs.add(new Cost(bucket.get(index).key, position, index + 1));
373 }
374 } else {
375 for (int position = 0; position < slots.length; position++) {
376 Item item = slots[position].item;
377 if (item == null) continue;
378 int probes = switch (mode) {
379 case CUCKOO -> position / cap + 1;
380 case LINEAR, ROBINHOOD -> distance(item, position) + 1;
381 default -> search(item.key, false).probes;
382 };
383 costs.add(new Cost(item.key, position, probes));
384 }
385 }
386 return costs;
387 }
388 Metrics metrics() {
389 int longest = 0;
390 long total = 0;
391 for (Cost cost : costs()) { longest = Math.max(longest, cost.probes); total += cost.probes; }
392 return new Metrics(longest, decimal(total, size));
393 }
394 List<String> dump() {
395 var lines = new ArrayList<String>();
396 lines.add("DUMP " + mode + " cap=" + cap + " size=" + size + " tomb=" + tomb);
397 if (mode == Mode.CHAIN) {
398 for (int position = 0; position < cap; position++) {
399 var row = new StringBuilder().append(position);
400 var bucket = buckets.get(position);
401 if (bucket.isEmpty()) row.append(" -");
402 for (Item item : bucket) row.append(' ').append(item);
403 lines.add(row.toString());
404 }
405 } else {
406 for (int position = 0; position < slots.length; position++) {
407 Slot slot = slots[position];
408 String prefix = mode == Mode.CUCKOO ? "T" + (position / cap + 1) + " " + position % cap : Integer.toString(position);
409 String value = slot.item == null ? slot.content() : slot.item.toString();
410 if (mode == Mode.ROBINHOOD && slot.item != null) value += " d=" + distance(slot.item, position);
411 lines.add(prefix + " " + value);
412 }
413 }
414 return lines;
415 }
416 List<String> map() {
417 var lines = new ArrayList<String>();
418 lines.add("MAP " + mode + " cap=" + cap);
419 int[] costs = new int[slotCount()];
420 if (mode != Mode.CHAIN) for (Cost cost : costs()) costs[cost.position] = cost.probes;
421 for (int part = 0; part < (mode == Mode.CUCKOO ? 2 : 1); part++) {
422 for (int start = 0; start < cap; start += 40) {
423 var text = new StringBuilder();
424 for (int index = start; index < Math.min(start + 40, cap); index++) {
425 char ch;
426 if (mode == Mode.CHAIN) ch = buckets.get(index).isEmpty() ? '.' : gauge(buckets.get(index).size());
427 else {
428 int position = part * cap + index;
429 Slot slot = slots[position];
430 ch = slot.empty() ? '.' : slot.deleted ? 'D' : mode == Mode.CUCKOO ? '#' : gauge(costs[position] - 1);
431 }
432 text.append(ch);
433 }
434 String prefix = mode == Mode.CUCKOO ? "T" + (part + 1) + " " : "";
435 lines.add(prefix + "%05d %s".formatted(start, text));
436 }
437 }
438 return lines;
439 }

448 String scan(int cursor, int limit) {
449 var keys = new ArrayList<String>();
450 while (cursor < slotCount() && keys.size() < limit) {
451 if (mode == Mode.CHAIN) for (Item item : buckets.get(cursor)) keys.add(item.key);
452 else if (slots[cursor].item != null) keys.add(slots[cursor].item.key);
453 cursor++;
454 }
455 if (cursor == slotCount()) cursor = 0;
456 return "SCAN next=" + cursor + (keys.isEmpty() ? "" : " " + String.join(" ", keys));
457 }
458 List<String> clusters() {
459 var runs = new ArrayList<Integer>();
460 if (mode == Mode.CHAIN) {
461 for (var bucket : buckets) if (!bucket.isEmpty()) runs.add(bucket.size());
462 } else {
463 for (int base = 0; base < slots.length; base += cap) {
464 int firstEmpty = -1;
465 for (int i = 0; i < cap; i++) if (slots[base + i].empty()) { firstEmpty = i; break; }
466 if (firstEmpty < 0) { runs.add(cap); continue; }
467 int length = 0;
468 for (int offset = 1; offset <= cap; offset++) {
469 if (!slots[base + (firstEmpty + offset) % cap].empty()) length++;
470 else if (length > 0) { runs.add(length); length = 0; }
471 }
472 }
473 }
474 var lines = new ArrayList<String>();
475 lines.add("CLUSTERS runs=" + runs.size() + " longest=" + runs.stream().mapToInt(Integer::intValue).max().orElse(0));
476 lines.addAll(histogram(runs));
477 return lines;
478 }

480 var values = costs().stream().map(Cost::probes).toList();
481 var lines = new ArrayList<String>();
482 lines.add("DIST keys=" + size + " longest=" + values.stream().mapToInt(Integer::intValue).max().orElse(0));
483 lines.addAll(histogram(values));
484 return lines;
485 }

488 static List<String> compare(HashTable table, int cap, int maximum) {
489 var items = table.items();
490 items.sort(Comparator.comparing(Item::key));
491 var lines = new ArrayList<String>();
492 lines.add("COMPARE keys=" + items.size() + " cap=" + cap);
493 for (Mode mode : Mode.values()) {
494 var trial = new HashTable(mode, cap);
495 for (Item item : items) trial.put(item, false, maximum, Reporting.SILENT);
496 Metrics values = trial.metrics();
497 lines.add(mode + " cap=" + trial.cap + " longest=" + values.longest + " avg=" + values.average
498 + " probes=" + trial.counters.probes + " rehashes=" + trial.counters.rehashes);
499 }
500 return lines;
501 }
502 static List<String> benchmark(int cap, int count, int length, long seed, int maximum) {
503 var keys = generate(2 * count, length, seed);
504 var lines = new ArrayList<String>();
505 lines.add("BENCH cap=" + cap + " count=" + count + " len=" + length + " seed=" + seed);
506 for (Mode mode : Mode.values()) {
507 var trial = new HashTable(mode, cap);
508 for (int i = 0; i < count; i++) trial.put(new Item(keys.get(i), i + 1), false, maximum, Reporting.SILENT);
509 int misses = 0;
510 long probes = 0;
511 for (String key : keys.subList(count, 2 * count)) {
512 Search found = trial.search(key, false);
513 if (found.item == null) { misses++; probes += found.probes; }
514 }
515 lines.add(mode + " cap=" + trial.cap + " keys=" + trial.size + " hit=" + trial.metrics().average
516 + " misses=" + misses + " miss=" + decimal(probes, misses) + " rehashes=" + trial.counters.rehashes);
517 }
518 return lines;
519 }

521 static final class Simulator {
522 static final Set<String> COMMANDS = Set.of("CREATE", "MAXLOAD", "MINLOAD", "TRACE", "HASH", "PUT", "ADD", "GET", "DEL", "PROBE", "GEN", "GENDEL",
523 "RESIZE", "CONVERT", "CLEAR", "DUMP", "MAP", "STATS", "KEYS", "TOP", "SCAN", "CLUSTERS", "DIST", "COMPARE", "BENCH");
524 HashTable table;
525 int maximum = 75, minimum;
526 boolean trace;
527
528 List<String> execute(String line) {
529 String[] words = line.split(" ");
530 String command = words[0];
531 if (!COMMANDS.contains(command)) throw new CommandError("UNKNOWN COMMAND " + command);
532 switch (command) {
533 case "CREATE" -> {
534 Mode mode = parseMode(words[1]);
535 int cap = prime(capacity(Long.parseLong(words[2])));
536 table = new HashTable(mode, cap);
537 return List.of("CREATE " + mode + " cap=" + cap);
538 }
539 case "MAXLOAD", "MINLOAD" -> {
540 long value = Long.parseLong(words[1]);
541 if (command.equals("MAXLOAD") ? value < 10 || value > 95 : value < 0 || value > 40) throw new CommandError("BAD LOAD");
542 if (command.equals("MAXLOAD")) maximum = (int) value; else minimum = (int) value;
543 return List.of(command + " " + value);
544 }
545 case "TRACE" -> {
546 if (!words[1].equals("ON") && !words[1].equals("OFF")) throw new CommandError("BAD ARGUMENT");
547 trace = words[1].equals("ON");
548 return List.of("TRACE " + words[1]);
549 }
550 case "HASH" -> {
551 int cap = capacity(Long.parseLong(words[2]));
552 String key = words[1];
553 return List.of("HASH " + key + " m=" + cap + " h31=" + hash(key, 31, cap) + " h37=" + hash(key, 37, cap) + " step=" + (1 + hash(key, 37, cap - 1)));
554 }
555 case "BENCH" -> {
556 int cap = prime(capacity(Long.parseLong(words[1])));
557 long count = Long.parseLong(words[2]), length = Long.parseLong(words[3]), seed = Long.parseLong(words[4]);
558 checkGeneration(count, length, seed);
559 return benchmark(cap, (int) count, (int) length, seed, maximum);
560 }
561 }
562 if (table == null) throw new CommandError("NO TABLE");
563 Reporting reporting = trace ? Reporting.TRACE : Reporting.FULL;
564 return switch (command) {
565 case "PUT", "ADD" -> table.put(new Item(words[1], Long.parseLong(words[2])), command.equals("ADD"), maximum, reporting).lines;
566 case "GET" -> table.get(words[1], trace);
567 case "DEL" -> table.delete(words[1], minimum, maximum, reporting).lines;
568 case "PROBE" -> {
569 Search found = table.search(words[1], true);
570 String path = found.path.isEmpty() ? "" : " " + String.join(" ", found.path);
571 yield List.of("PROBE " + words[1] + path + (found.item == null ? " MISS" : " FOUND") + " probes=" + found.probes);
572 }
573 case "GEN", "GENDEL" -> {
574 long count = Long.parseLong(words[1]), length = Long.parseLong(words[2]), seed = Long.parseLong(words[3]);
575 checkGeneration(count, length, seed);
576 var lines = new ArrayList<String>();
577 int index = 0, changed = 0;
578 for (String key : generate((int) count, (int) length, seed)) {
579 index++;
580 Mutation result = command.equals("GEN") ? table.put(new Item(key, index), false, maximum, Reporting.REHASHES)
581 : table.delete(key, minimum, maximum, Reporting.REHASHES);
582 if (result.changed) changed++;
583 lines.addAll(result.lines);
584 }
585 lines.add(command.equals("GEN") ? "GEN " + count + " new=" + changed + " updated=" + (count - changed)
586 : "GENDEL " + count + " deleted=" + changed + " missing=" + (count - changed));
587 yield lines;
588 }
589 case "RESIZE", "CONVERT" -> {
590 int cap = command.equals("RESIZE") ? prime(capacity(Long.parseLong(words[1]))) : table.cap;
591 Mode mode = command.equals("CONVERT") ? parseMode(words[1]) : table.mode;
592 int slots = cap * (mode == Mode.CUCKOO ? 2 : 1);
593 if (table.size * 100L > (long) maximum * slots) throw new CommandError("TOO SMALL");
594 HashTable replacement = HashTable.rebuild(mode, cap, table.items());
595 if (replacement == null) throw new CommandError("REBUILD FAILED");
596 String row = command.equals("RESIZE") ? "RESIZE " + table.cap + " -> " + cap : "CONVERT " + table.mode + " -> " + mode + " cap=" + cap;
597 table.adopt(replacement);
598 yield List.of(row);
599 }
600 case "CLEAR" -> {
601 int size = table.size;
602 table.adopt(new HashTable(table.mode, table.cap));
603 yield List.of("CLEAR " + size);
604 }
605 case "DUMP" -> table.dump();
606 case "MAP" -> table.map();
607 case "STATS" -> table.statistics();
608 case "KEYS" -> {
609 var keys = table.items().stream().map(Item::key).sorted().toList();
610 yield List.of("KEYS " + keys.size() + (keys.isEmpty() ? "" : " " + String.join(" ", keys)));
611 }
612 case "TOP" -> {
613 long limit = Long.parseLong(words[1]);
614 if (limit < 1 || limit > 100) throw new CommandError("BAD ARGUMENT");
615 var costs = table.costs();
616 costs.sort(Comparator.comparingInt(Cost::probes).reversed().thenComparing(Cost::key));
617 var selected = costs.subList(0, Math.min(costs.size(), (int) limit));
618 var lines = new ArrayList<String>();
619 lines.add("TOP " + selected.size());
620 for (Cost cost : selected) lines.add(cost.key + " cost=" + cost.probes + " at=" + table.position(cost.position));
621 yield lines;
622 }
623 case "SCAN" -> {
624 long cursor = Long.parseLong(words[1]), limit = Long.parseLong(words[2]);
625 if (limit < 1 || limit > 100) throw new CommandError("BAD ARGUMENT");
626 if (cursor >= table.slotCount()) throw new CommandError("BAD CURSOR");
627 yield List.of(table.scan((int) cursor, (int) limit));
628 }
629 case "CLUSTERS" -> table.clusters();
630 case "DIST" -> table.distribution();
631 case "COMPARE" -> compare(table, prime(capacity(Long.parseLong(words[1]))), maximum);
632 default -> throw new IllegalStateException(command);
633 };
634 }
635 }

637 public static void main(String[] args) throws IOException {
638 Locale.setDefault(Locale.ROOT);
639 var input = new BufferedReader(new InputStreamReader(System.in));
640 var output = new PrintWriter(new BufferedWriter(new OutputStreamWriter(System.out)));
641 String first = input.readLine();
642 if (first == null) return;
643 int count = Integer.parseInt(first);
644 var simulator = new Simulator();
645 for (int i = 0; i < count; i++) {
646 try { for (String line : simulator.execute(input.readLine())) output.print(line + '\n'); }
647 catch (CommandError error) { output.print("ERROR " + error.getMessage() + '\n'); }
648 }
649 output.flush();
650 }
