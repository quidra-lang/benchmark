40static class Utilities
41{
42 public static string Name(Mode mode) => mode.ToString().ToUpperInvariant();
43 public static Mode ParseMode(string text) => text switch
44 {
45 "CHAIN" => Mode.Chain, "LINEAR" => Mode.Linear, "QUADRATIC" => Mode.Quadratic,
46 "DOUBLE" => Mode.Double, "ROBINHOOD" => Mode.RobinHood, "CUCKOO" => Mode.Cuckoo,
47 _ => throw new CommandError("BAD MODE " + text)
48 };
49 public static int Hash(string key, int basis, int modulus)
50 {
51 int value = 0;
52 foreach (char character in key) value = (value * basis + character) % modulus;
53 return value;
54 }
55 public static int Prime(int value)
56 {
57 for (int candidate = Math.Max(value, 2); ; candidate++)
58 {
59 bool prime = true;
60 for (int divisor = 2; divisor * divisor <= candidate; divisor++)
61 if (candidate % divisor == 0) { prime = false; break; }
62 if (prime) return candidate;
63 }
64 }
65 public static string Ratio(long numerator, long denominator)
66 {
67 long value = denominator == 0 ? 0 : 100 * numerator / denominator;
68 return $"{value / 100}.{value % 100:D2}";
69 }
70 public static char Gauge(int value) => value >= 10 ? '+' : (char)('0' + value);
71 public static IEnumerable<string> Generate(int count, int length, long seed)
72 {
73 for (int i = 0; i < count; i++)
74 {
75 char[] key = new char[length];
76 for (int j = 0; j < length; j++)
77 {
78 seed = 48271 * seed % 2147483647;
79 key[j] = (char)('a' + seed % 26);
80 }
81 yield return new string(key);
82 }
83 }
84 public static List<string> Histogram(IEnumerable<int> values) => values.GroupBy(x => x).OrderBy(g => g.Key)
85 .Select(g => $"{g.Key} {g.Count()} {new string('#', Math.Min(g.Count(), 50))}{(g.Count() > 50 ? "+" : "")}").ToList();
86 public static void CheckCapacity(int capacity)
87 {
88 if (capacity is < 2 or > 10000) throw new CommandError("BAD CAPACITY");
89 }
90 public static void CheckGeneration(int count, int length, long seed)
91 {
92 if (count is < 1 or > 500 || length is < 1 or > 10 || seed is < 1 or > 2147483646)
93 throw new CommandError("BAD ARGUMENT");
94 }
95}

97sealed class HashTable
98{
99 List<Item>[] buckets;
100 Slot[] slots;
101 public Mode Mode { get; private set; }
102 public int Capacity { get; private set; }
103 public int Size { get; private set; }
104 public int Tombstones { get; private set; }
105 public Counters Counters { get; } = new();
106 public int SlotCount => Capacity * (Mode == Mode.Cuckoo ? 2 : 1);
107 bool UsesTombstones => Mode is Mode.Linear or Mode.Quadratic or Mode.Double;
108 public HashTable(Mode mode, int capacity)
109 {
110 Mode = mode;
111 Capacity = capacity;
112 buckets = mode == Mode.Chain ? Enumerable.Range(0, capacity).Select(_ => new List<Item>()).ToArray() : [];
113 slots = mode == Mode.Chain ? [] : new Slot[SlotCount];
114 }
115 public string Position(int index) => Mode == Mode.Cuckoo ? $"T{index / Capacity + 1}.{index % Capacity}" : index.ToString();
116 int Distance(string key, int index) => (index - Utilities.Hash(key, 31, Capacity) + Capacity) % Capacity;
117 public List<Item> Items()
118 {
119 if (Mode == Mode.Chain) return buckets.SelectMany(bucket => bucket).ToList();
120 List<Item> items = [];
121 foreach (Slot slot in slots) if (slot.Item is { } item) items.Add(item);
122 return items;
123 }
124 public SearchResult Search(string key, bool trace = false)
125 {
126 SearchResult result = new();
127 int first = Utilities.Hash(key, 31, Capacity);
128 void Examine(int position, string content)
129 {
130 result.Probes++;
131 if (trace) result.Path.Add($"{Position(position)}:{content}");
132 }
133 void Found(Item item, int position, int chainIndex = -1)
134 {
135 result.Item = item;
136 result.Position = position;
137 result.ChainIndex = chainIndex;
138 }
139 if (Mode == Mode.Chain)
140 {
141 for (int i = 0; i < buckets[first].Count; i++)
142 {
143 Item item = buckets[first][i];
144 Examine(first, item.Key);
145 if (item.Key == key) { Found(item, first, i); break; }
146 }
147 return result;
148 }
149 if (Mode == Mode.Cuckoo)
150 {
151 foreach (int index in new[] { first, Capacity + Utilities.Hash(key, 37, Capacity) })
152 {
153 Examine(index, slots[index].Content);
154 if (slots[index].Item is { } item && item.Key == key) { Found(item, index); break; }
155 }
156 return result;
157 }
158 int step = Mode == Mode.Double ? 1 + Utilities.Hash(key, 37, Capacity - 1) : 1;
159 for (int i = 0; i < Capacity; i++)
160 {
161 long offset = Mode == Mode.Quadratic ? (long)i * i : (long)i * step;
162 int index = (int)((first + offset) % Capacity);
163 Slot slot = slots[index];
164 Examine(index, slot.Content);
165 if (slot.Empty) { result.Available ??= index; break; }
166 if (slot.Deleted) { result.Available ??= index; continue; }
167 Item item = slot.Item!.Value;
168 if (item.Key == key) { Found(item, index); break; }
169 if (Mode == Mode.RobinHood && Distance(item.Key, index) < i) break;
170 }
171 return result;
172 }
173 Placement Place(Item item, SearchResult? initial = null, bool trace = false)
174 {
175 Placement result = new();
176 if (Mode == Mode.Chain)
177 {
178 result.Position = Utilities.Hash(item.Key, 31, Capacity);
179 buckets[result.Position].Add(item);
180 result.Success = true;
181 return result;
182 }
183 if (UsesTombstones)
184 {
185 SearchResult found = initial ?? Search(item.Key);
186 if (found.Available is null) { result.Homeless = item; return result; }
187 int index = found.Available.Value;
188 result.Position = index;
189 result.Reused = slots[index].Deleted;
190 slots[index] = new Slot(item);
191 if (result.Reused) Tombstones--;
192 result.Success = true;
193 return result;
194 }
195 Item current = item;
196 if (Mode == Mode.RobinHood)
197 {
198 int index = Utilities.Hash(item.Key, 31, Capacity), distance = 0;
199 int? from = null;
200 while (true)
201 {
202 if (slots[index].Empty)
203 {
204 slots[index] = new Slot(current);
205 if (trace && from.HasValue) result.Moves.Add($"SHIFT {current.Key} {from.Value} -> {index}");
206 if (result.Position < 0) result.Position = index;
207 result.Success = true;
208 return result;
209 }
210 Item resident = slots[index].Item!.Value;
211 int residentDistance = Distance(resident.Key, index);
212 if (residentDistance < distance)
213 {
214 slots[index] = new Slot(current);
215 if (trace && from.HasValue) result.Moves.Add($"SHIFT {current.Key} {from.Value} -> {index}");
216 if (result.Position < 0) result.Position = index;
217 current = resident;
218 from = index;
219 distance = residentDistance;
220 result.Shifted++;
221 }
222 index = (index + 1) % Capacity;
223 distance++;
224 }
225 }
226 int part = 0;
227 while (true)
228 {
229 int index = part * Capacity + Utilities.Hash(current.Key, part == 0 ? 31 : 37, Capacity);
230 if (slots[index].Empty)
231 {
232 slots[index] = new Slot(current);
233 result.Success = true;
234 result.Position = Search(item.Key).Position;
235 return result;
236 }
237 if (result.Kicks == Capacity) { result.Homeless = current; return result; }
238 Item resident = slots[index].Item!.Value;
239 slots[index] = new Slot(current);
240 result.Kicks++;
241 if (trace) result.Moves.Add($"KICK {resident.Key} {Position(index)}");
242 current = resident;
243 part = 1 - part;
244 }
245 }
246 Removal Remove(SearchResult found, bool trace)
247 {
248 int shifted = 0;
249 List<string> moves = [];
250 if (Mode == Mode.Chain) buckets[found.Position].RemoveAt(found.ChainIndex);
251 else if (UsesTombstones) { slots[found.Position] = new Slot(null, true); Tombstones++; }
252 else
253 {
254 slots[found.Position] = default;
255 if (Mode == Mode.RobinHood)
256 {
257 int next = (found.Position + 1) % Capacity;
258 while (slots[next].Item is { } item && Distance(item.Key, next) > 0)
259 {
260 int destination = (next - 1 + Capacity) % Capacity;
261 slots[destination] = slots[next];
262 slots[next] = default;
263 if (trace) moves.Add($"SHIFT {item.Key} {next} -> {destination}");
264 shifted++;
265 next = (next + 1) % Capacity;
266 }
267 }
268 }
269 Size--;
270 return new Removal(shifted, moves);
271 }
272 public static HashTable? Rebuild(List<Item> items, Mode mode, int capacity)
273 {
274 HashTable replacement = new(mode, capacity);
275 foreach (Item item in items)
276 {
277 if (!replacement.Place(item).Success) return null;
278 replacement.Size++;
279 }
280 return replacement;
281 }
282 public void Adopt(HashTable replacement)
283 {
284 Mode = replacement.Mode;
285 Capacity = replacement.Capacity;
286 Size = replacement.Size;
287 Tombstones = replacement.Tombstones;
288 buckets = replacement.buckets;
289 slots = replacement.slots;
290 }
291 string Automatic(string reason, int capacity, Item? pending = null)
292 {
293 List<Item> items = Items();
294 if (pending is { } item) items.Add(item);
295 int old = Capacity;
296 while (true)
297 {
298 HashTable? replacement = Rebuild(items, Mode, capacity);
299 if (replacement is not null)
300 {
301 Adopt(replacement);
302 Counters.Rehashes++;
303 return $"REHASH {reason} {old} -> {Capacity}";
304 }
305 capacity = Utilities.Prime(2 * capacity);
306 }
307 }
308 string? LoadCheck(int maximum)
309 {
310 long threshold = (long)maximum * SlotCount;
311 if ((Size + Tombstones) * 100L <= threshold) return null;
312 bool grow = Size * 200L > threshold;
313 return Automatic(grow ? "load" : "tomb", grow ? Utilities.Prime(2 * Capacity) : Capacity);
314 }
315 string? ShrinkCheck(int minimum, int maximum)
316 {
317 int capacity = Utilities.Prime(Capacity / 2), count = capacity * (Mode == Mode.Cuckoo ? 2 : 1);
318 if (Size * 100L >= (long)minimum * SlotCount || capacity >= Capacity || Size * 100L > (long)maximum * count) return null;
319 HashTable? replacement = Rebuild(Items(), Mode, capacity);
320 if (replacement is null) return null;
321 int old = Capacity;
322 Adopt(replacement);
323 Counters.Rehashes++;
324 return $"REHASH shrink {old} -> {capacity}";
325 }
326 static string PathLine(SearchResult found) => "PATH " + (found.Path.Count == 0 ? "none" : string.Join(' ', found.Path));
327 public Mutation Put(string key, long value, bool add, int maximum, bool trace = false)
328 {
329 string command = add ? "ADD" : "PUT";
330 SearchResult found = Search(key, trace);
331 Counters.Probes += found.Probes;
332 List<string> lines = [], rehashes = [];
333 if (found.Item is { } old)
334 {
335 Item item = new(key, add ? old.Value + value : value);
336 if (Mode == Mode.Chain) buckets[found.Position][found.ChainIndex] = item;
337 else slots[found.Position] = new Slot(item);
338 Counters.Updates++;
339 lines.Add($"{command} {item} UPDATED old={old.Value} at={Position(found.Position)} probes={found.Probes}");
340 if (trace) lines.Add(PathLine(found));
341 return new Mutation(false, lines, rehashes);
342 }
343 Placement placed = Place(new Item(key, value), found, trace);
344 Size++;
345 Counters.Inserts++;
346 string row = $"{command} {key}={value} NEW";
347 if (placed.Success)
348 {
349 row += $" at={Position(placed.Position)} probes={found.Probes}";
350 if (placed.Reused) row += " reused";
351 if (Mode == Mode.RobinHood) row += $" shifted={placed.Shifted}";
352 if (Mode == Mode.Cuckoo) row += $" kicks={placed.Kicks}";
353 }
354 else
355 {
356 row += $" probes={found.Probes}" + (Mode == Mode.Cuckoo ? $" kicks={placed.Kicks} failed" : " full");
357 rehashes.Add(Automatic(Mode == Mode.Cuckoo ? "kicks" : "full", Utilities.Prime(2 * Capacity), placed.Homeless));
358 }
359 if (LoadCheck(maximum) is { } rehash) rehashes.Add(rehash);
360 lines.Add(row);
361 if (trace) lines.Add(PathLine(found));
362 lines.AddRange(placed.Moves);
363 lines.AddRange(rehashes);
364 return new Mutation(true, lines, rehashes);
365 }
366 public List<string> Get(string key, bool trace)
367 {
368 SearchResult found = Search(key, trace);
369 Counters.Probes += found.Probes;
370 string row;
371 if (found.Item is { } item) { Counters.Hits++; row = $"GET {item} at={Position(found.Position)}"; }
372 else { Counters.Misses++; row = $"GET {key} NOT FOUND"; }
373 List<string> lines = [$"{row} probes={found.Probes}"];
374 if (trace) lines.Add(PathLine(found));
375 return lines;
376 }
377 public Mutation Delete(string key, int minimum, int maximum, bool trace = false)
378 {
379 SearchResult found = Search(key, trace);
380 Counters.Probes += found.Probes;
381 List<string> lines = [], rehashes = [];
382 if (found.Item is null)
383 {
384 Counters.Absent++;
385 lines.Add($"DEL {key} NOT FOUND probes={found.Probes}");
386 if (trace) lines.Add(PathLine(found));
387 return new Mutation(false, lines, rehashes);
388 }
389 Removal removed = Remove(found, trace);
390 Counters.Deletes++;
391 string row = $"DEL {key} OK at={Position(found.Position)} probes={found.Probes}";
392 if (Mode == Mode.RobinHood) row += $" shifted={removed.Shifted}";
393 if (ShrinkCheck(minimum, maximum) is { } rehash) rehashes.Add(rehash);
394 lines.Add(row);
395 if (trace) lines.Add(PathLine(found));
396 lines.AddRange(removed.Moves);
397 lines.AddRange(rehashes);
398 return new Mutation(true, lines, rehashes);
399 }
400 public List<Cost> Costs()
401 {
402 List<Cost> result = [];
403 if (Mode == Mode.Chain)
404 {
405 for (int index = 0; index < Capacity; index++)
406 for (int i = 0; i < buckets[index].Count; i++) result.Add(new Cost(buckets[index][i].Key, i + 1, index));
407 }
408 else
409 {
410 for (int index = 0; index < SlotCount; index++) if (slots[index].Item is { } item)
411 {
412 int cost = Mode == Mode.Cuckoo ? index / Capacity + 1
413 : Mode is Mode.Linear or Mode.RobinHood ? Distance(item.Key, index) + 1 : Search(item.Key).Probes;
414 result.Add(new Cost(item.Key, cost, index));
415 }
416 }
417 return result;
418 }
419 public Metrics Metrics()
420 {
421 List<Cost> costs = Costs();
422 return new Metrics(costs.Count == 0 ? 0 : costs.Max(c => c.Probes), Utilities.Ratio(costs.Sum(c => (long)c.Probes), Size));
423 }
424 public List<string> Dump()
425 {
426 List<string> lines = [$"DUMP {Utilities.Name(Mode)} cap={Capacity} size={Size} tomb={Tombstones}"];
427 if (Mode == Mode.Chain)
428 {
429 for (int i = 0; i < Capacity; i++)
430 lines.Add($"{i} " + (buckets[i].Count == 0 ? "-" : string.Join(' ', buckets[i])));
431 }
432 else
433 {
434 for (int i = 0; i < SlotCount; i++)
435 {
436 string prefix = Mode == Mode.Cuckoo ? $"T{i / Capacity + 1} {i % Capacity}" : i.ToString();
437 string value = slots[i].Item is { } item ? item.ToString() : slots[i].Content;
438 if (Mode == Mode.RobinHood && slots[i].Item is { } resident) value += $" d={Distance(resident.Key, i)}";
439 lines.Add(prefix + " " + value);
440 }
441 }
442 return lines;
443 }
444 public List<string> Map()
445 {
446 List<string> lines = [$"MAP {Utilities.Name(Mode)} cap={Capacity}"];
447 int[] costs = new int[SlotCount];
448 if (Mode != Mode.Chain) foreach (Cost cost in Costs()) costs[cost.Position] = cost.Probes;
449 for (int part = 0; part < (Mode == Mode.Cuckoo ? 2 : 1); part++)
450 {
451 char[] characters = new char[Capacity];
452 for (int i = 0; i < Capacity; i++)
453 {
454 if (Mode == Mode.Chain) characters[i] = buckets[i].Count == 0 ? '.' : Utilities.Gauge(buckets[i].Count);
455 else
456 {
457 int index = part * Capacity + i;
458 Slot slot = slots[index];
459 characters[i] = slot.Empty ? '.' : slot.Deleted ? 'D' : Mode == Mode.Cuckoo ? '#' : Utilities.Gauge(costs[index] - 1);
460 }
461 }
462 for (int i = 0; i < Capacity; i += 40)
463 lines.Add((Mode == Mode.Cuckoo ? $"T{part + 1} " : "") + $"{i:D5} " + new string(characters, i, Math.Min(40, Capacity - i)));
464 }
465 return lines;
466 }
467 public List<string> Statistics()

469 Metrics metrics = Metrics();
470 return [$"STATS {Utilities.Name(Mode)} size={Size} cap={Capacity} slots={SlotCount} tomb={Tombstones} load={Utilities.Ratio(100L * Size, SlotCount)}% longest={metrics.Longest} avg={metrics.Average} probes={Counters.Probes} rehashes={Counters.Rehashes}",
471 $"OPS inserts={Counters.Inserts} updates={Counters.Updates} hits={Counters.Hits} misses={Counters.Misses} deletes={Counters.Deletes} absent={Counters.Absent}"];
472 }
473 public List<string> Scan(int cursor, int limit)
474 {
475 List<string> keys = [];
476 while (cursor < SlotCount && keys.Count < limit)
477 {
478 if (Mode == Mode.Chain) keys.AddRange(buckets[cursor].Select(item => item.Key));
479 else if (slots[cursor].Item is { } item) keys.Add(item.Key);
480 cursor++;
481 }
482 if (cursor == SlotCount) cursor = 0;
483 return [$"SCAN next={cursor}" + (keys.Count == 0 ? "" : " " + string.Join(' ', keys))];
484 }
485 List<int> Runs(int start)
486 {
487 int firstEmpty = -1;
488 for (int i = 0; i < Capacity; i++) if (slots[start + i].Empty) { firstEmpty = i; break; }
489 if (firstEmpty < 0) return [Capacity];
490 List<int> lengths = [];
491 int length = 0;
492 for (int offset = 1; offset <= Capacity; offset++)
493 {
494 if (!slots[start + (firstEmpty + offset) % Capacity].Empty) length++;
495 else if (length > 0) { lengths.Add(length); length = 0; }
496 }
497 return lengths;
498 }
499 public List<string> Clusters()
500 {
501 List<int> lengths;
502 if (Mode == Mode.Chain) lengths = buckets.Where(b => b.Count > 0).Select(b => b.Count).ToList();
503 else
504 {
505 lengths = Runs(0);
506 if (Mode == Mode.Cuckoo) lengths.AddRange(Runs(Capacity));
507 }
508 int longest = lengths.Count == 0 ? 0 : lengths.Max();
509 return [$"CLUSTERS runs={lengths.Count} longest={longest}", .. Utilities.Histogram(lengths)];
510 }
511 public List<string> Distribution()
512 {
513 List<int> values = Costs().Select(c => c.Probes).ToList();
514 int longest = values.Count == 0 ? 0 : values.Max();
515 return [$"DIST keys={Size} longest={longest}", .. Utilities.Histogram(values)];
516 }

519sealed class Simulator
520{
521 static readonly HashSet<string> Commands = ["CREATE", "MAXLOAD", "MINLOAD", "TRACE", "HASH", "PUT", "ADD", "GET", "DEL", "PROBE", "GEN", "GENDEL", "RESIZE", "CONVERT", "CLEAR", "DUMP", "MAP", "STATS", "KEYS", "TOP", "SCAN", "CLUSTERS", "DIST", "COMPARE", "BENCH"];
522 HashTable? table;
523 int maximum = 75, minimum;
524 bool tracing;
525 List<string> Compare(int capacity)
526 {
527 List<Item> items = table!.Items().OrderBy(item => item.Key, StringComparer.Ordinal).ToList();
528 List<string> lines = [$"COMPARE keys={items.Count} cap={capacity}"];
529 foreach (Mode mode in Enum.GetValues<Mode>())
530 {
531 HashTable experiment = new(mode, capacity);
532 foreach (Item item in items) experiment.Put(item.Key, item.Value, false, maximum);
533 Metrics metrics = experiment.Metrics();
534 lines.Add($"{Utilities.Name(mode)} cap={experiment.Capacity} longest={metrics.Longest} avg={metrics.Average} probes={experiment.Counters.Probes} rehashes={experiment.Counters.Rehashes}");
535 }
536 return lines;
537 }
538 List<string> Bench(int capacity, int count, int length, long seed)
539 {
540 List<string> keys = Utilities.Generate(2 * count, length, seed).ToList();
541 List<string> lines = [$"BENCH cap={capacity} count={count} len={length} seed={seed}"];
542 foreach (Mode mode in Enum.GetValues<Mode>())
543 {
544 HashTable experiment = new(mode, capacity);
545 for (int i = 0; i < count; i++) experiment.Put(keys[i], i + 1, false, maximum);
546 int misses = 0;
547 long probes = 0;
548 for (int i = count; i < 2 * count; i++)
549 {
550 SearchResult found = experiment.Search(keys[i]);
551 if (found.Item is null) { misses++; probes += found.Probes; }
552 }
553 lines.Add($"{Utilities.Name(mode)} cap={experiment.Capacity} keys={experiment.Size} hit={experiment.Metrics().Average} misses={misses} miss={Utilities.Ratio(probes, misses)} rehashes={experiment.Counters.Rehashes}");
554 }
555 return lines;
556 }
557 public List<string> Execute(string line)
558 {
559 string[] words = line.Split(' ');
560 string command = words[0];
561 int Number(int index) => int.Parse(words[index], CultureInfo.InvariantCulture);
562 if (!Commands.Contains(command)) throw new CommandError("UNKNOWN COMMAND " + command);
563 switch (command)
564 {
565 case "CREATE":
566 {
567 Mode mode = Utilities.ParseMode(words[1]);
568 int capacity = Number(2);
569 Utilities.CheckCapacity(capacity);
570 table = new HashTable(mode, Utilities.Prime(capacity));
571 return [$"CREATE {Utilities.Name(mode)} cap={table.Capacity}"];
572 }
573 case "MAXLOAD":
574 case "MINLOAD":
575 {
576 int value = Number(1);
577 if (command == "MAXLOAD" ? value is < 10 or > 95 : value > 40) throw new CommandError("BAD LOAD");
578 if (command == "MAXLOAD") maximum = value; else minimum = value;
579 return [$"{command} {value}"];
580 }
581 case "TRACE":
582 if (words[1] is not ("ON" or "OFF")) throw new CommandError("BAD ARGUMENT");
583 tracing = words[1] == "ON";
584 return ["TRACE " + words[1]];
585 case "HASH":
586 {
587 int capacity = Number(2);
588 Utilities.CheckCapacity(capacity);
589 string key = words[1];
590 return [$"HASH {key} m={capacity} h31={Utilities.Hash(key, 31, capacity)} h37={Utilities.Hash(key, 37, capacity)} step={1 + Utilities.Hash(key, 37, capacity - 1)}"];
591 }
592 case "BENCH":
593 {
594 int capacity = Number(1), count = Number(2), length = Number(3);
595 long seed = Number(4);
596 Utilities.CheckCapacity(capacity);
597 Utilities.CheckGeneration(count, length, seed);
598 return Bench(Utilities.Prime(capacity), count, length, seed);
599 }
600 }
601 HashTable current = table ?? throw new CommandError("NO TABLE");
602 switch (command)
603 {
604 case "PUT":
605 case "ADD": return current.Put(words[1], Number(2), command == "ADD", maximum, tracing).Lines;
606 case "GET": return current.Get(words[1], tracing);
607 case "DEL": return current.Delete(words[1], minimum, maximum, tracing).Lines;
608 case "PROBE":
609 {
610 SearchResult found = current.Search(words[1], true);
611 string path = found.Path.Count == 0 ? "" : " " + string.Join(' ', found.Path);
612 return [$"PROBE {words[1]}{path} {(found.Item is null ? "MISS" : "FOUND")} probes={found.Probes}"];
613 }
614 case "GEN":
615 case "GENDEL":
616 {
617 int count = Number(1), length = Number(2);
618 long seed = Number(3);
619 Utilities.CheckGeneration(count, length, seed);
620 List<string> lines = [];
621 int changed = 0, index = 0;
622 foreach (string key in Utilities.Generate(count, length, seed))
623 {
624 index++;
625 Mutation mutation = command == "GEN" ? current.Put(key, index, false, maximum) : current.Delete(key, minimum, maximum);
626 if (mutation.Changed) changed++;
627 lines.AddRange(mutation.Rehashes);
628 }
629 lines.Add(command == "GEN" ? $"GEN {count} new={changed} updated={count - changed}" : $"GENDEL {count} deleted={changed} missing={count - changed}");
630 return lines;
631 }
632 case "RESIZE":
633 case "CONVERT":
634 {
635 int capacity = current.Capacity;
636 Mode mode = current.Mode;
637 if (command == "RESIZE") { capacity = Number(1); Utilities.CheckCapacity(capacity); capacity = Utilities.Prime(capacity); }
638 else mode = Utilities.ParseMode(words[1]);
639 int slots = capacity * (mode == Mode.Cuckoo ? 2 : 1);
640 if (current.Size * 100L > (long)maximum * slots) throw new CommandError("TOO SMALL");
641 HashTable replacement = HashTable.Rebuild(current.Items(), mode, capacity) ?? throw new CommandError("REBUILD FAILED");
642 string row = command == "RESIZE" ? $"RESIZE {current.Capacity} -> {capacity}"
643 : $"CONVERT {Utilities.Name(current.Mode)} -> {Utilities.Name(mode)} cap={capacity}";
644 current.Adopt(replacement);
645 return [row];
646 }
647 case "CLEAR":
648 {
649 int size = current.Size;
650 current.Adopt(new HashTable(current.Mode, current.Capacity));
651 return [$"CLEAR {size}"];
652 }
653 case "DUMP": return current.Dump();
654 case "MAP": return current.Map();
655 case "STATS": return current.Statistics();
656 case "KEYS":
657 {
658 string keys = string.Join(' ', current.Items().Select(item => item.Key).Order(StringComparer.Ordinal));
659 return [$"KEYS {current.Size}" + (keys.Length == 0 ? "" : " " + keys)];
660 }
661 case "TOP":
662 {
663 int limit = Number(1);
664 if (limit is < 1 or > 100) throw new CommandError("BAD ARGUMENT");
665 List<Cost> costs = current.Costs().OrderByDescending(c => c.Probes).ThenBy(c => c.Key, StringComparer.Ordinal).Take(limit).ToList();
666 return [$"TOP {costs.Count}", .. costs.Select(c => $"{c.Key} cost={c.Probes} at={current.Position(c.Position)}")];
667 }
668 case "SCAN":
669 {
670 int cursor = Number(1), limit = Number(2);
671 if (limit is < 1 or > 100) throw new CommandError("BAD ARGUMENT");
672 if (cursor >= current.SlotCount) throw new CommandError("BAD CURSOR");
673 return current.Scan(cursor, limit);
674 }
675 case "CLUSTERS": return current.Clusters();
676 case "DIST": return current.Distribution();
677 case "COMPARE":
678 {
679 int capacity = Number(1);
680 Utilities.CheckCapacity(capacity);
681 return Compare(Utilities.Prime(capacity));
682 }
683 default: throw new InvalidOperationException();
684 }
685 }
686}
