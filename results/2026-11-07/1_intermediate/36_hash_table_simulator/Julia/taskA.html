76function hash_key(key::String, base::Int, modulus::Int)
77 value = 0
78 for code in codeunits(key)
79 value = mod(value * base + Int(code), modulus)
80 end
81 value
82end
83function is_prime(value::Int)
84 value < 2 && return false
85 for divisor in 2:isqrt(value)
86 mod(value, divisor) == 0 && return false
87 end
88 true
89end
90function next_prime(value::Int)
91 value = max(value, 2)
92 while !is_prime(value)
93 value += 1
94 end
95 value
96end
97function two_decimal(numerator::Int, denominator::Int)
98 value = denominator == 0 ? 0 : div(100numerator, denominator)
99 @sprintf("%d.%02d", div(value, 100), mod(value, 100))
100end

113 result = String[]

124function position_text(table::Table, position::Int)
125 index = position - 1
126 table.mode == CUCKOO ? "T$(div(index, table.capacity) + 1).$(mod(index, table.capacity))" : string(index)
127end
128distance(table::Table, item::Item, position::Int) =
129 mod(position - 1 - hash_key(item.key, 31, table.capacity), table.capacity)
130with_words(prefix::String, words::Vector{String}) = isempty(words) ? prefix : prefix * " " * join(words, " ")
131path_line(search::Search) = with_words("PATH", isempty(search.path) ? ["none"] : search.path)
132found(search::Search) = search.position != 0

134function item_list(table::Table)
135 if table.mode == CHAIN
136 return Item[item for bucket in table.buckets for item in bucket]
137 end
138 Item[slot for slot in table.slots if slot isa Item]
139end
140function lookup(table::Table, key::String; trace::Bool = false)
141 result = Search()
142 function observe(position::Int, content::String)
143 result.probes += 1
144 trace && push!(result.path, "$(position_text(table, position)):$content")
145 end
146 home = hash_key(key, 31, table.capacity)
147 if table.mode == CHAIN
148 for (index, item) in enumerate(table.buckets[home + 1])
149 observe(home + 1, item.key)
150 if item.key == key
151 result.position, result.chain_index = home + 1, index
152 break
153 end
154 end
155 return result
156 end
157 step = table.mode == DOUBLE ? 1 + hash_key(key, 37, table.capacity - 1) : 1
158 limit = table.mode == CUCKOO ? 2 : table.capacity
159 for offset in 0:limit-1
160 index = mod(home + offset * step, table.capacity)
161 if table.mode == QUADRATIC
162 index = mod(home + offset * offset, table.capacity)
163 elseif table.mode == CUCKOO && offset == 1
164 index = table.capacity + hash_key(key, 37, table.capacity)
165 end
166 position = index + 1
167 slot = table.slots[position]
168 observe(position, slot_text(slot))
169 if slot isa Item && slot.key == key
170 result.position = position
171 break
172 end
173 table.mode == CUCKOO && continue
174 if isnothing(slot)
175 result.available == 0 && (result.available = position)
176 break
177 end
178 slot isa Tombstone && result.available == 0 && (result.available = position)
179 table.mode == ROBINHOOD && distance(table, slot, position) < offset && break
180 end
181 result
182end
183function found_item(table::Table, search::Search)
184 table.mode == CHAIN && return table.buckets[search.position][search.chain_index]
185 table.slots[search.position]::Item
186end

188function place!(table::Table, item::Item; initial::Union{Nothing,Search} = nothing, trace::Bool = false)
189 home = hash_key(item.key, 31, table.capacity) + 1
190 if table.mode == CHAIN
191 push!(table.buckets[home], item)
192 return Placement(success = true, position = home)
193 elseif table.mode in (LINEAR, QUADRATIC, DOUBLE)
194 search = isnothing(initial) ? lookup(table, item.key) : initial
195 position = search.available
196 position == 0 && return Placement(pending = item)
197 reused = table.slots[position] isa Tombstone
198 reused && (table.tomb -= 1)
199 table.slots[position] = item
200 return Placement(success = true, position = position, reused = reused)
201 end
202 carried, movements, original = item, 0, 0
203 moves = String[]
204 if table.mode == ROBINHOOD
205 position, traveled, source = home, 0, 0
206 while true
207 if isnothing(table.slots[position])
208 trace && source != 0 && push!(moves, "SHIFT $(carried.key) $(source - 1) -> $(position - 1)")
209 table.slots[position] = carried
210 original == 0 && (original = position)
211 return Placement(success = true, position = original, movements = movements, moves = moves)
212 end
213 resident = table.slots[position]::Item
214 old_distance = distance(table, resident, position)
215 if old_distance < traveled
216 trace && source != 0 && push!(moves, "SHIFT $(carried.key) $(source - 1) -> $(position - 1)")
217 table.slots[position], carried = carried, resident
218 original == 0 && (original = position)
219 source, traveled = position, old_distance
220 movements += 1
221 end
222 position = mod(position, table.capacity) + 1
223 traveled += 1
224 end
225 end
226 part = 0
227 while true
228 position = part * table.capacity + hash_key(carried.key, part == 0 ? 31 : 37, table.capacity) + 1
229 if isnothing(table.slots[position])
230 table.slots[position] = carried
231 original = lookup(table, item.key).position
232 return Placement(success = true, position = original, movements = movements, moves = moves)
233 end
234 movements == table.capacity && return Placement(movements = movements, pending = carried, moves = moves)
235 resident = table.slots[position]::Item
236 trace && push!(moves, "KICK $(resident.key) $(position_text(table, position))")
237 table.slots[position], carried = carried, resident
238 movements += 1
239 part = 1 - part
240 end
241end
242function remove_item!(table::Table, search::Search; trace::Bool = false)
243 position, shifted = search.position, 0
244 moves = String[]
245 if table.mode == CHAIN
246 deleteat!(table.buckets[position], search.chain_index)
247 elseif table.mode in (LINEAR, QUADRATIC, DOUBLE)
248 table.slots[position] = TOMB
249 table.tomb += 1
250 else
251 table.slots[position] = nothing
252 if table.mode == ROBINHOOD
253 source = mod(position, table.capacity) + 1
254 while table.slots[source] isa Item && distance(table, table.slots[source], source) > 0
255 destination = mod(source - 2, table.capacity) + 1
256 trace && push!(moves, "SHIFT $(table.slots[source].key) $(source - 1) -> $(destination - 1)")
257 table.slots[destination], table.slots[source] = table.slots[source], nothing
258 shifted += 1
259 source = mod(source, table.capacity) + 1
260 end
261 end
262 end
263 table.size -= 1
264 shifted, moves
265end

267function rebuilt(mode::Mode, capacity::Int, items::Vector{Item})
268 trial = Table(mode, capacity)
269 for item in items
270 place!(trial, item).success || return nothing
271 trial.size += 1
272 end
273 trial
274end
275function adopt!(table::Table, trial::Table)
276 table.mode, table.capacity = trial.mode, trial.capacity
277 table.size, table.tomb = trial.size, trial.tomb
278 table.buckets, table.slots = trial.buckets, trial.slots
279 table
280end
281function automatic!(table::Table, capacity::Int, reason::String; pending::Union{Nothing,Item} = nothing)
282 old = table.capacity
283 items = item_list(table)
284 !isnothing(pending) && push!(items, pending)
285 while true
286 trial = rebuilt(table.mode, capacity, items)
287 if !isnothing(trial)
288 adopt!(table, trial)
289 table.counters.rehashes += 1
290 return "REHASH $reason $old -> $capacity"
291 end
292 capacity = next_prime(2capacity)
293 end
294end
295function load_check!(table::Table, maxload::Int)
296 threshold = maxload * slot_count(table)
297 100 * (table.size + table.tomb) <= threshold && return nothing
298 200 * table.size > threshold && return automatic!(table, next_prime(2 * table.capacity), "load")
299 automatic!(table, table.capacity, "tomb")
300end
301function shrink_check!(table::Table, minload::Int, maxload::Int)
302 capacity = next_prime(div(table.capacity, 2))
303 if 100 * table.size >= minload * slot_count(table) || capacity >= table.capacity ||
304 100 * table.size > maxload * slot_count(table.mode, capacity)
305 return nothing
306 end
307 trial = rebuilt(table.mode, capacity, item_list(table))
308 isnothing(trial) && return nothing
309 old = table.capacity
310 adopt!(table, trial)
311 table.counters.rehashes += 1
312 "REHASH shrink $old -> $capacity"
313end
314function put_item!(table::Table, item::Item, add::Bool, maxload::Int, reporting::Reporting)
315 trace = reporting == TRACED
316 detailed = reporting in (FULL, TRACED)
317 search = lookup(table, item.key; trace)
318 table.counters.probes += search.probes
319 command = add ? "ADD" : "PUT"
320 lines = String[]
321 if found(search)
322 old = found_item(table, search).value
323 add && (item = Item(item.key, old + item.value))
324 if table.mode == CHAIN
325 table.buckets[search.position][search.chain_index] = item
326 else
327 table.slots[search.position] = item
328 end
329 table.counters.updates += 1
330 detailed && push!(lines, "$command $(item_text(item)) UPDATED old=$old at=$(position_text(table, search.position)) probes=$(search.probes)")
331 trace && push!(lines, path_line(search))
332 return false, lines
333 end
334 placed = place!(table, item; initial = search, trace)
335 table.counters.inserts += 1
336 if detailed
337 row = "$command $(item_text(item)) NEW"
338 if placed.success
339 row *= " at=$(position_text(table, placed.position)) probes=$(search.probes)"
340 placed.reused && (row *= " reused")
341 table.mode == ROBINHOOD && (row *= " shifted=$(placed.movements)")
342 table.mode == CUCKOO && (row *= " kicks=$(placed.movements)")
343 else
344 row *= " probes=$(search.probes)"
345 row *= table.mode == QUADRATIC ? " full" : " kicks=$(placed.movements) failed"
346 end
347 push!(lines, row)
348 end
349 if trace
350 push!(lines, path_line(search))
351 append!(lines, placed.moves)
352 end
353 if placed.success
354 table.size += 1
355 else
356 reason = table.mode == QUADRATIC ? "full" : "kicks"
357 event = automatic!(table, next_prime(2 * table.capacity), reason; pending = placed.pending)
358 reporting != SILENT && push!(lines, event)
359 end
360 event = load_check!(table, maxload)
361 reporting != SILENT && !isnothing(event) && push!(lines, event)
362 true, lines
363end
364function delete_item!(table::Table, key::String, minload::Int, maxload::Int, reporting::Reporting)
365 trace = reporting == TRACED
366 search = lookup(table, key; trace)
367 table.counters.probes += search.probes
368 lines, moves = String[], String[]
369 if found(search)
370 shifted, moves = remove_item!(table, search; trace)
371 table.counters.deletes += 1
372 row = "DEL $key OK at=$(position_text(table, search.position)) probes=$(search.probes)"
373 table.mode == ROBINHOOD && (row *= " shifted=$shifted")
374 else
375 table.counters.absent += 1
376 row = "DEL $key NOT FOUND probes=$(search.probes)"
377 end
378 reporting in (FULL, TRACED) && push!(lines, row)
379 if trace
380 push!(lines, path_line(search))
381 append!(lines, moves)
382 end
383 if found(search)
384 event = shrink_check!(table, minload, maxload)
385 reporting != SILENT && !isnothing(event) && push!(lines, event)
386 end
387 found(search), lines
388end
389function get_item!(table::Table, key::String, trace::Bool)
390 search = lookup(table, key; trace)
391 table.counters.probes += search.probes
392 if found(search)
393 table.counters.hits += 1
394 row = "GET $(item_text(found_item(table, search))) at=$(position_text(table, search.position)) probes=$(search.probes)"
395 else
396 table.counters.misses += 1
397 row = "GET $key NOT FOUND probes=$(search.probes)"
398 end
399 trace ? [row, path_line(search)] : [row]
400end

402function record_items(table::Table)
403 records = ItemRecord[]
404 if table.mode == CHAIN
405 for (position, bucket) in enumerate(table.buckets), (cost, item) in enumerate(bucket)
406 push!(records, ItemRecord(item, position, cost))
407 end
408 else
409 for (position, slot) in enumerate(table.slots)
410 slot isa Item || continue
411 cost = if table.mode == CUCKOO
412 div(position - 1, table.capacity) + 1
413 elseif table.mode in (LINEAR, ROBINHOOD)
414 distance(table, slot, position) + 1
415 else
416 lookup(table, slot.key).probes
417 end
418 push!(records, ItemRecord(slot, position, cost))
419 end
420 end
421 records
422end
423function table_metrics(table::Table)
424 costs = [record.cost for record in record_items(table)]
425 maximum(costs; init = 0), two_decimal(sum(costs), table.size)
426end
427function dump_table(table::Table)
428 lines = ["DUMP $(table.mode) cap=$(table.capacity) size=$(table.size) tomb=$(table.tomb)"]
429 if table.mode == CHAIN
430 for (position, bucket) in enumerate(table.buckets)
431 content = isempty(bucket) ? "-" : join(item_text.(bucket), " ")
432 push!(lines, "$(position - 1) $content")
433 end
434 else
435 for (position, slot) in enumerate(table.slots)
436 index = position - 1
437 prefix = table.mode == CUCKOO ? "T$(div(index, table.capacity) + 1) $(mod(index, table.capacity))" : string(index)
438 content = slot isa Item ? item_text(slot) : slot_text(slot)
439 slot isa Item && table.mode == ROBINHOOD && (content *= " d=$(distance(table, slot, position))")
440 push!(lines, "$prefix $content")
441 end
442 end
443 lines
444end
445gauge(value::Int) = value >= 10 ? '+' : Char(Int('0') + value)
446function map_table(table::Table)
447 lines = ["MAP $(table.mode) cap=$(table.capacity)"]
448 costs = zeros(Int, slot_count(table))
449 for record in record_items(table)
450 costs[record.position] = record.cost
451 end
452 for part in 0:(table.mode == CUCKOO ? 1 : 0), start in 0:40:table.capacity-1
453 chars = Char[]
454 for index in start:min(start + 39, table.capacity - 1)
455 if table.mode == CHAIN
456 count = length(table.buckets[index + 1])
457 push!(chars, count == 0 ? '.' : gauge(count))
458 else
459 position = part * table.capacity + index + 1
460 slot = table.slots[position]
461 char = isnothing(slot) ? '.' : slot isa Tombstone ? 'D' :
462 table.mode == CUCKOO ? '#' : gauge(costs[position] - 1)
463 push!(chars, char)
464 end
465 end
466 prefix = table.mode == CUCKOO ? "T$(part + 1) " : ""
467 push!(lines, prefix * @sprintf("%05d", start) * " " * join(chars))
468 end
469 lines
470end
471function table_stats(table::Table)
472 longest, average = table_metrics(table)
473 load = two_decimal(100 * table.size, slot_count(table))
474 c = table.counters
475 ["STATS $(table.mode) size=$(table.size) cap=$(table.capacity) slots=$(slot_count(table)) tomb=$(table.tomb) load=$load% longest=$longest avg=$average probes=$(c.probes) rehashes=$(c.rehashes)",
476 "OPS inserts=$(c.inserts) updates=$(c.updates) hits=$(c.hits) misses=$(c.misses) deletes=$(c.deletes) absent=$(c.absent)"]
477end
478function table_scan(table::Table, cursor::Int, limit::Int)
479 collected = String[]
480 while cursor < slot_count(table) && length(collected) < limit
481 if table.mode == CHAIN
482 append!(collected, [item.key for item in table.buckets[cursor + 1]])
483 elseif table.slots[cursor + 1] isa Item
484 push!(collected, table.slots[cursor + 1].key)
485 end
486 cursor += 1
487 end
488 cursor == slot_count(table) && (cursor = 0)
489 with_words("SCAN next=$cursor", collected)
490end
491function histogram(values::Vector{Int})
492 frequencies = Dict{Int,Int}()
493 for value in values
494 frequencies[value] = get(frequencies, value, 0) + 1
495 end
496 lines = String[]
497 for value in sort!(collect(keys(frequencies)))
498 count = frequencies[value]
499 bar = repeat("#", min(count, 50)) * (count > 50 ? "+" : "")
500 push!(lines, "$value $count $bar")
501 end
502 lines
503end
504function cluster_report(table::Table)
505 runs = Int[]
506 if table.mode == CHAIN
507 runs = [length(bucket) for bucket in table.buckets if !isempty(bucket)]
508 else
509 for base in 0:table.capacity:length(table.slots)-1
510 first_empty = findfirst(isnothing, view(table.slots, base+1:base+table.capacity))
511 if isnothing(first_empty)
512 push!(runs, table.capacity)
513 continue
514 end
515 count = 0
516 for offset in 1:table.capacity
517 position = base + mod(first_empty - 1 + offset, table.capacity) + 1
518 if !isnothing(table.slots[position])
519 count += 1
520 elseif count > 0
521 push!(runs, count)
522 count = 0
523 end
524 end
525 end
526 end
527 vcat(["CLUSTERS runs=$(length(runs)) longest=$(maximum(runs; init = 0))"], histogram(runs))
528end

531 vcat(["DIST keys=$(table.size) longest=$(maximum(costs; init = 0))"], histogram(costs))

535 lines = ["COMPARE keys=$(length(items)) cap=$capacity"]
536 for mode in instances(Mode)

546function benchmark(capacity::Int, count::Int, width::Int, seed::Int, maxload::Int)
547 generated = generate_keys(2count, width, seed)
548 lines = ["BENCH cap=$capacity count=$count len=$width seed=$seed"]
549 for mode in instances(Mode)
550 trial = Table(mode, capacity)
551 for index in 1:count
552 put_item!(trial, Item(generated[index], index), false, maxload, SILENT)
553 end
554 misses, probes = 0, 0
555 for key in view(generated, count+1:2count)
556 search = lookup(trial, key)
557 if !found(search)
558 misses += 1
559 probes += search.probes
560 end
561 end
562 _, average = table_metrics(trial)
563 push!(lines, "$mode cap=$(trial.capacity) keys=$(trial.size) hit=$average misses=$misses miss=$(two_decimal(probes, misses)) rehashes=$(trial.counters.rehashes)")
564 end
565 lines
566end

568function execute!(sim::Simulator, line::String)
569 words = String.(split(line, ' '))
570 command = words[1]
571 command in COMMANDS || fail("UNKNOWN COMMAND $command")
572 number(index::Int) = parse(Int, words[index])
573 if command == "CREATE"
574 mode = checked_mode(words[2])
575 capacity = next_prime(checked_capacity(number(3)))
576 sim.table = Table(mode, capacity)
577 return ["CREATE $mode cap=$capacity"]
578 elseif command in ("MAXLOAD", "MINLOAD")
579 value = number(2)
580 if command == "MAXLOAD"
581 10 <= value <= 95 || fail("BAD LOAD")
582 sim.maxload = value
583 else
584 0 <= value <= 40 || fail("BAD LOAD")
585 sim.minload = value
586 end
587 return ["$command $value"]
588 elseif command == "TRACE"
589 words[2] in ("ON", "OFF") || fail("BAD ARGUMENT")
590 sim.trace = words[2] == "ON"
591 return ["TRACE $(words[2])"]
592 elseif command == "HASH"
593 capacity = checked_capacity(number(3))
594 key = words[2]
595 return ["HASH $key m=$capacity h31=$(hash_key(key, 31, capacity)) h37=$(hash_key(key, 37, capacity)) step=$(1 + hash_key(key, 37, capacity - 1))"]
596 elseif command == "BENCH"
597 capacity = checked_capacity(number(2))
598 count, width, seed = number(3), number(4), number(5)
599 check_generation(count, width, seed)
600 return benchmark(next_prime(capacity), count, width, seed, sim.maxload)
601 end
602 table = sim.table
603 isnothing(table) && fail("NO TABLE")
604 reporting = sim.trace ? TRACED : FULL
605 if command in ("PUT", "ADD")
606 _, lines = put_item!(table, Item(words[2], number(3)), command == "ADD", sim.maxload, reporting)
607 return lines
608 elseif command == "GET"
609 return get_item!(table, words[2], sim.trace)
610 elseif command == "DEL"
611 _, lines = delete_item!(table, words[2], sim.minload, sim.maxload, reporting)
612 return lines
613 elseif command == "PROBE"
614 search = lookup(table, words[2]; trace = true)
615 row = with_words("PROBE $(words[2])", search.path)
616 return [row * " $(found(search) ? "FOUND" : "MISS") probes=$(search.probes)"]
617 elseif command in ("GEN", "GENDEL")
618 count, width, seed = number(2), number(3), number(4)
619 check_generation(count, width, seed)
620 changed, lines = 0, String[]
621 for (index, key) in enumerate(generate_keys(count, width, seed))
622 success, output = command == "GEN" ? put_item!(table, Item(key, index), false, sim.maxload, REHASHES) :
623 delete_item!(table, key, sim.minload, sim.maxload, REHASHES)
624 changed += success
625 append!(lines, output)
626 end
627 summary = command == "GEN" ? "new=$changed updated=$(count - changed)" : "deleted=$changed missing=$(count - changed)"
628 push!(lines, "$command $count $summary")
629 return lines
630 elseif command in ("RESIZE", "CONVERT")
631 mode, capacity = table.mode, table.capacity
632 if command == "RESIZE"
633 capacity = next_prime(checked_capacity(number(2)))
634 else
635 mode = checked_mode(words[2])
636 end
637 100 * table.size <= sim.maxload * slot_count(mode, capacity) || fail("TOO SMALL")
638 trial = rebuilt(mode, capacity, item_list(table))
639 isnothing(trial) && fail("REBUILD FAILED")
640 row = command == "RESIZE" ? "RESIZE $(table.capacity) -> $capacity" : "CONVERT $(table.mode) -> $mode cap=$capacity"
641 adopt!(table, trial)
642 return [row]
643 elseif command == "CLEAR"
644 old = table.size
645 adopt!(table, Table(table.mode, table.capacity))
646 return ["CLEAR $old"]
647 elseif command == "DUMP"
648 return dump_table(table)
649 elseif command == "MAP"
650 return map_table(table)
651 elseif command == "STATS"
652 return table_stats(table)
653 elseif command == "KEYS"
654 sorted_keys = sort!([item.key for item in item_list(table)])
655 return [with_words("KEYS $(table.size)", sorted_keys)]
656 elseif command == "TOP"
657 limit = number(2)
658 1 <= limit <= 100 || fail("BAD ARGUMENT")
659 records = sort(record_items(table); by = record -> (-record.cost, record.item.key))
660 selected = records[1:min(limit, length(records))]
661 return vcat(["TOP $(length(selected))"], ["$(record.item.key) cost=$(record.cost) at=$(position_text(table, record.position))" for record in selected])
662 elseif command == "SCAN"
663 cursor, limit = number(2), number(3)
664 1 <= limit <= 100 || fail("BAD ARGUMENT")
665 cursor < slot_count(table) || fail("BAD CURSOR")
666 return [table_scan(table, cursor, limit)]
667 elseif command == "CLUSTERS"
668 return cluster_report(table)
669 elseif command == "DIST"
670 return cost_report(table)
671 elseif command == "COMPARE"
672 return compare_modes(table, next_prime(checked_capacity(number(2))), sim.maxload)
673 end
674 error("unreachable command")
675end
676function main()
677 count = parse(Int, readline())
678 simulator = Simulator()
679 for _ in 1:count
680 line = readline()
681 try
682 foreach(println, execute!(simulator, line))
683 catch error
684 error isa CommandError || rethrow()
685 println("ERROR ", error.message)
686 end
687 end
688end
689main()
