50module Helpers
51 def parse_mode(word)
52 raise CommandError, "BAD MODE #{word}" unless MODES.include?(word)
53 word
54 end
55
56 def hash_key(key, base, modulus)
57 key.each_byte.reduce(0) { |value, byte| (value * base + byte) % modulus }
58 end
59
60 def prime(value)
61 candidate = [value, 2].max
62 loop do
63 divisor = 2
64 divisor += 1 while divisor <= candidate / divisor && candidate % divisor != 0
65 return candidate if divisor > candidate / divisor
66 candidate += 1
67 end
68 end
69
70 def decimal(numerator, denominator)
71 value = denominator.zero? ? 0 : 100 * numerator / denominator
72 format('%d.%02d', value / 100, value % 100)
73 end
74
75 def gauge(value) = value < 10 ? (48 + value).chr : '+'
76
77 def histogram(values)
78 values.tally.sort.map do |value, count|
79 "#{value} #{count} #{'#' * [count, 50].min}#{count > 50 ? '+' : ''}"
80 end
81 end

83 def generate(count, length, seed)
84 state = seed
85 Array.new(count) do
86 Array.new(length) do
87 state = 48_271 * state % 2_147_483_647
88 (97 + state % 26).chr
89 end.join.freeze
90 end
91 end

94 raise CommandError, 'BAD CAPACITY' unless (2..10_000).cover?(value)

99 return if (1..500).cover?(count) && (1..10).cover?(length) && (1..2_147_483_646).cover?(seed)

104class HashTable
105 include Helpers
106 attr_reader :mode, :cap, :tomb, :counters, :buckets, :slots
107 attr_accessor :size
108
109 def initialize(mode, cap)
110 @mode = mode
111 @cap = cap
112 @size = @tomb = 0
113 @counters = Counters.new
114 @buckets = mode == 'CHAIN' ? Array.new(cap) { [] } : []
115 @slots = Array.new(mode == 'CHAIN' ? 0 : slot_count, EMPTY)
116 end
117
118 def slot_count = @cap * (@mode == 'CUCKOO' ? 2 : 1)
119 def position(index) = @mode == 'CUCKOO' ? "T#{index / @cap + 1}.#{index % @cap}" : index.to_s
120 def distance(item, index) = (index - hash_key(item.key, 31, @cap)) % @cap
121 def items = @mode == 'CHAIN' ? @buckets.flatten : @slots.filter_map(&:item)
122
123 def search(key, trace = false)
124 result = Search.new
125 observe = lambda do |index, content|
126 result.probes += 1
127 result.path << "#{position(index)}:#{content}" if trace
128 end
129 home = hash_key(key, 31, @cap)
130 if @mode == 'CHAIN'
131 @buckets[home].each_with_index do |item, index|
132 observe.call(home, item.key)
133 next unless item.key == key
134 result.item = item
135 result.position = home
136 result.chain_index = index
137 break
138 end
139 return result
140 end
141 step = @mode == 'DOUBLE' ? 1 + hash_key(key, 37, @cap - 1) : 1
142 (@mode == 'CUCKOO' ? 2 : @cap).times do |i|
143 index = if @mode == 'CUCKOO'
144 i.zero? ? home : @cap + hash_key(key, 37, @cap)
145 else
146 (home + (@mode == 'QUADRATIC' ? i * i : i * step)) % @cap
147 end
148 slot = @slots[index]
149 observe.call(index, slot.content)
150 if slot.item&.key == key
151 result.item = slot.item
152 result.position = index
153 break
154 end
155 next if @mode == 'CUCKOO'
156 if slot.empty?
157 result.available ||= index
158 break
159 end
160 if slot.deleted
161 result.available ||= index
162 next
163 end
164 break if @mode == 'ROBINHOOD' && distance(slot.item, index) < i
165 end
166 result
167 end
168
169 def place(item, initial = nil, trace = false)
170 result = Placement.new
171 home = hash_key(item.key, 31, @cap)
172 if @mode == 'CHAIN'
173 @buckets[home] << item
174 result.position = home
175 return result
176 end
177 if %w[LINEAR QUADRATIC DOUBLE].include?(@mode)
178 index = (initial || search(item.key)).available
179 unless index
180 result.pending = item
181 return result
182 end
183 result.reused = @slots[index].deleted
184 @tomb -= 1 if result.reused
185 @slots[index] = Slot.new(item, false)
186 result.position = index
187 return result
188 end
189 carried = item
190 if @mode == 'ROBINHOOD'
191 index = home
192 traveled = 0
193 from = nil
194 loop do
195 resident = @slots[index].item
196 unless resident
197 result.moves << "SHIFT #{carried.key} #{from} -> #{index}" if trace && from
198 @slots[index] = Slot.new(carried, false)
199 result.position ||= index
200 return result
201 end
202 resident_distance = distance(resident, index)
203 if resident_distance < traveled
204 result.moves << "SHIFT #{carried.key} #{from} -> #{index}" if trace && from
205 @slots[index] = Slot.new(carried, false)
206 carried = resident
207 result.position ||= index
208 from = index
209 traveled = resident_distance
210 result.movements += 1
211 end
212 index = (index + 1) % @cap
213 traveled += 1
214 end
215 end
216 part = 0
217 loop do
218 index = part * @cap + hash_key(carried.key, part.zero? ? 31 : 37, @cap)
219 resident = @slots[index].item
220 unless resident
221 @slots[index] = Slot.new(carried, false)
222 result.position = search(item.key).position
223 return result
224 end
225 if result.movements == @cap
226 result.pending = carried
227 return result
228 end
229 result.moves << "KICK #{resident.key} #{position(index)}" if trace
230 @slots[index] = Slot.new(carried, false)
231 carried = resident
232 result.movements += 1
233 part = 1 - part
234 end
235 end

237 def remove(found, trace)
238 shifted = 0
239 moves = []
240 if @mode == 'CHAIN'
241 @buckets[found.position].delete_at(found.chain_index)
242 elsif %w[LINEAR QUADRATIC DOUBLE].include?(@mode)
243 @slots[found.position] = TOMB
244 @tomb += 1
245 else
246 @slots[found.position] = EMPTY
247 if @mode == 'ROBINHOOD'
248 source = (found.position + 1) % @cap
249 loop do
250 item = @slots[source].item
251 break unless item && distance(item, source).positive?
252 destination = (source - 1) % @cap
253 moves << "SHIFT #{item.key} #{source} -> #{destination}" if trace
254 @slots[destination] = @slots[source]
255 @slots[source] = EMPTY
256 shifted += 1
257 source = (source + 1) % @cap
258 end
259 end
260 end
261 @size -= 1
262 Removal.new(shifted, moves)
263 end
264
265 def self.rebuild(mode, cap, items)
266 trial = new(mode, cap)
267 items.each do |item|
268 return nil unless trial.place(item).position
269 end
270 trial.size = items.length
271 trial
272 end

274 def adopt(replacement)
275 @mode = replacement.mode
276 @cap = replacement.cap
277 @size = replacement.size
278 @tomb = replacement.tomb
279 @buckets = replacement.buckets
280 @slots = replacement.slots
281 end

283 def automatic(target, reason, pending = nil)
284 old = @cap
285 listed = items
286 listed << pending if pending
287 loop do
288 trial = HashTable.rebuild(@mode, target, listed)
289 if trial
290 adopt(trial)
291 @counters.rehashes += 1
292 return "REHASH #{reason} #{old} -> #{@cap}"
293 end
294 target = prime(2 * target)
295 end
296 end

298 def load_check(maximum)
299 threshold = maximum * slot_count
300 return nil if (@size + @tomb) * 100 <= threshold
301 grow = @size * 200 > threshold
302 automatic(grow ? prime(2 * @cap) : @cap, grow ? 'load' : 'tomb')
303 end
304
305 def shrink_check(minimum, maximum)
306 target = prime(@cap / 2)
307 count = target * (@mode == 'CUCKOO' ? 2 : 1)
308 return nil if @size * 100 >= minimum * slot_count || target >= @cap || @size * 100 > maximum * count
309 trial = HashTable.rebuild(@mode, target, items)
310 return nil unless trial
311 old = @cap
312 adopt(trial)
313 @counters.rehashes += 1
314 "REHASH shrink #{old} -> #{@cap}"
315 end

317 def path_line(found) = "PATH #{found.path.empty? ? 'none' : found.path.join(' ')}"
318
319 def put(item, add, maximum, reporting)
320 trace = reporting == :trace
321 details = trace || reporting == :full
322 found = search(item.key, trace)
323 @counters.probes += found.probes
324 command = add ? 'ADD' : 'PUT'
325 lines = []
326 if (old = found.item)
327 stored = Item.new(item.key, add ? old.value + item.value : item.value)
328 if @mode == 'CHAIN'
329 @buckets[found.position][found.chain_index] = stored
330 else
331 @slots[found.position] = Slot.new(stored, false)
332 end
333 @counters.updates += 1
334 lines << "#{command} #{stored} UPDATED old=#{old.value} at=#{position(found.position)} probes=#{found.probes}" if details
335 lines << path_line(found) if trace
336 return Mutation.new(false, lines)
337 end
338 placed = place(item, found, trace)
339 @counters.inserts += 1
340 if details
341 row = "#{command} #{item} NEW"
342 if placed.position
343 row << " at=#{position(placed.position)} probes=#{found.probes}"
344 row << ' reused' if placed.reused
345 row << " shifted=#{placed.movements}" if @mode == 'ROBINHOOD'
346 row << " kicks=#{placed.movements}" if @mode == 'CUCKOO'
347 else
348 row << " probes=#{found.probes}"
349 row << (@mode == 'QUADRATIC' ? ' full' : " kicks=#{placed.movements} failed")
350 end
351 lines << row
352 end
353 if trace
354 lines << path_line(found)
355 lines.concat(placed.moves)
356 end
357 if placed.position
358 @size += 1
359 else
360 rehash = automatic(prime(2 * @cap), @mode == 'QUADRATIC' ? 'full' : 'kicks', placed.pending)
361 lines << rehash unless reporting == :silent
362 end
363 rehash = load_check(maximum)
364 lines << rehash if rehash && reporting != :silent
365 Mutation.new(true, lines)
366 end
367
368 def delete(key, minimum, maximum, reporting)
369 trace = reporting == :trace
370 details = trace || reporting == :full
371 found = search(key, trace)
372 @counters.probes += found.probes
373 lines = []
374 unless found.item
375 @counters.absent += 1
376 lines << "DEL #{key} NOT FOUND probes=#{found.probes}" if details
377 lines << path_line(found) if trace
378 return Mutation.new(false, lines)
379 end
380 removed = remove(found, trace)
381 @counters.deletes += 1
382 if details
383 row = "DEL #{key} OK at=#{position(found.position)} probes=#{found.probes}"
384 row << " shifted=#{removed.shifted}" if @mode == 'ROBINHOOD'
385 lines << row
386 end
387 if trace
388 lines << path_line(found)
389 lines.concat(removed.moves)
390 end
391 rehash = shrink_check(minimum, maximum)
392 lines << rehash if rehash && reporting != :silent
393 Mutation.new(true, lines)
394 end

396 def get(key, trace)
397 found = search(key, trace)
398 @counters.probes += found.probes
399 row = if found.item
400 @counters.hits += 1
401 "GET #{found.item} at=#{position(found.position)} probes=#{found.probes}"
402 else
403 @counters.misses += 1
404 "GET #{key} NOT FOUND probes=#{found.probes}"
405 end
406 trace ? [row, path_line(found)] : [row]
407 end

409 def costs
410 result = []
411 if @mode == 'CHAIN'
412 @buckets.each_with_index do |bucket, index|
413 bucket.each_with_index { |item, place| result << Cost.new(item.key, index, place + 1) }
414 end
415 else
416 @slots.each_with_index do |slot, index|
417 next unless slot.item
418 cost = case @mode
419 when 'CUCKOO' then index / @cap + 1
420 when 'LINEAR', 'ROBINHOOD' then distance(slot.item, index) + 1
421 else search(slot.item.key).probes
422 end
423 result << Cost.new(slot.item.key, index, cost)
424 end
425 end
426 result
427 end

430 values = costs
431 Metrics.new(values.map(&:probes).max || 0, decimal(values.sum(&:probes), @size))
432 end

434 def dump
435 lines = ["DUMP #{@mode} cap=#{@cap} size=#{@size} tomb=#{@tomb}"]
436 if @mode == 'CHAIN'
437 @buckets.each_with_index { |bucket, index| lines << "#{index} #{bucket.empty? ? '-' : bucket.join(' ')}" }
438 else
439 @slots.each_with_index do |slot, index|
440 prefix = @mode == 'CUCKOO' ? "T#{index / @cap + 1} #{index % @cap}" : index.to_s
441 content = slot.item ? slot.item.to_s : slot.content
442 content += " d=#{distance(slot.item, index)}" if @mode == 'ROBINHOOD' && slot.item
443 lines << "#{prefix} #{content}"
444 end
445 end
446 lines
447 end

449 def map
450 lines = ["MAP #{@mode} cap=#{@cap}"]
451 cost_by_slot = Array.new(slot_count, 0)
452 costs.each { |cost| cost_by_slot[cost.position] = cost.probes } unless @mode == 'CHAIN'
453 (@mode == 'CUCKOO' ? 2 : 1).times do |part|
454 (0...@cap).step(40) do |start|
455 text = (start...[start + 40, @cap].min).map do |index|
456 if @mode == 'CHAIN'
457 @buckets[index].empty? ? '.' : gauge(@buckets[index].length)
458 else
459 position = part * @cap + index
460 slot = @slots[position]
461 if slot.empty? then '.'
462 elsif slot.deleted then 'D'
463 elsif @mode == 'CUCKOO' then '#'
464 else gauge(cost_by_slot[position] - 1)
465 end
466 end
467 end.join
468 prefix = @mode == 'CUCKOO' ? "T#{part + 1} " : ''
469 lines << "#{prefix}#{format('%05d', start)} #{text}"
470 end
471 end
472 lines
473 end

475 def statistics
476 values = metrics
477 c = @counters
478 ["STATS #{@mode} size=#{@size} cap=#{@cap} slots=#{slot_count} tomb=#{@tomb} load=#{decimal(100 * @size, slot_count)}% " \
479 "longest=#{values.longest} avg=#{values.average} probes=#{c.probes} rehashes=#{c.rehashes}",
480 "OPS inserts=#{c.inserts} updates=#{c.updates} hits=#{c.hits} misses=#{c.misses} deletes=#{c.deletes} absent=#{c.absent}"]
481 end

483 def scan(cursor, limit)
484 keys = []
485 while cursor < slot_count && keys.length < limit
486 if @mode == 'CHAIN'
487 keys.concat(@buckets[cursor].map(&:key))
488 elsif (item = @slots[cursor].item)
489 keys << item.key
490 end
491 cursor += 1
492 end
493 cursor = 0 if cursor == slot_count
494 "SCAN next=#{cursor}#{keys.empty? ? '' : ' ' + keys.join(' ')}"
495 end

497 def clusters
498 runs = []
499 if @mode == 'CHAIN'
500 runs = @buckets.reject(&:empty?).map(&:length)
501 else
502 (0...slot_count).step(@cap) do |base|
503 empty = (0...@cap).find { |i| @slots[base + i].empty? }
504 unless empty
505 runs << @cap
506 next
507 end
508 length = 0
509 (1..@cap).each do |offset|
510 if !@slots[base + (empty + offset) % @cap].empty?
511 length += 1
512 elsif length.positive?
513 runs << length
514 length = 0
515 end
516 end
517 end
518 end
519 ["CLUSTERS runs=#{runs.length} longest=#{runs.max || 0}"] + histogram(runs)
520 end

522 def distribution
523 values = costs.map(&:probes)
524 ["DIST keys=#{@size} longest=#{values.max || 0}"] + histogram(values)
525 end

528class Simulator
529 include Helpers
530 COMMANDS = %w[CREATE MAXLOAD MINLOAD TRACE HASH PUT ADD GET DEL PROBE GEN GENDEL RESIZE CONVERT
531 CLEAR DUMP MAP STATS KEYS TOP SCAN CLUSTERS DIST COMPARE BENCH].freeze
532
533 def initialize
534 @table = nil
535 @maximum = 75
536 @minimum = 0
537 @trace = false
538 end

540 def compare(cap)
541 items = @table.items.sort_by(&:key)
542 lines = ["COMPARE keys=#{items.length} cap=#{cap}"]
543 MODES.each do |mode|
544 trial = HashTable.new(mode, cap)
545 items.each { |item| trial.put(item, false, @maximum, :silent) }
546 values = trial.metrics
547 lines << "#{mode} cap=#{trial.cap} longest=#{values.longest} avg=#{values.average} probes=#{trial.counters.probes} rehashes=#{trial.counters.rehashes}"
548 end
549 lines
550 end

552 def benchmark(cap, count, length, seed)
553 keys = generate(2 * count, length, seed)
554 lines = ["BENCH cap=#{cap} count=#{count} len=#{length} seed=#{seed}"]
555 MODES.each do |mode|
556 trial = HashTable.new(mode, cap)
557 count.times { |i| trial.put(Item.new(keys[i], i + 1), false, @maximum, :silent) }
558 misses = probes = 0
559 keys.drop(count).each do |key|
560 found = trial.search(key)
561 next if found.item
562 misses += 1
563 probes += found.probes
564 end
565 lines << "#{mode} cap=#{trial.cap} keys=#{trial.size} hit=#{trial.metrics.average} misses=#{misses} miss=#{decimal(probes, misses)} rehashes=#{trial.counters.rehashes}"
566 end
567 lines
568 end

570 def execute(line)
571 words = line.split(' ')
572 command = words[0]
573 number = ->(index) { Integer(words[index], 10) }
574 raise CommandError, "UNKNOWN COMMAND #{command}" unless COMMANDS.include?(command)
575 case command
576 when 'CREATE'
577 mode = parse_mode(words[1])
578 cap = prime(capacity(number.call(2)))
579 @table = HashTable.new(mode, cap)
580 return ["CREATE #{mode} cap=#{cap}"]
581 when 'MAXLOAD', 'MINLOAD'
582 value = number.call(1)
583 raise CommandError, 'BAD LOAD' unless (command == 'MAXLOAD' ? 10..95 : 0..40).cover?(value)
584 if command == 'MAXLOAD'
585 @maximum = value
586 else
587 @minimum = value
588 end
589 return ["#{command} #{value}"]
590 when 'TRACE'
591 raise CommandError, 'BAD ARGUMENT' unless %w[ON OFF].include?(words[1])
592 @trace = words[1] == 'ON'
593 return ["TRACE #{words[1]}"]
594 when 'HASH'
595 cap = capacity(number.call(2))
596 key = words[1]
597 return ["HASH #{key} m=#{cap} h31=#{hash_key(key, 31, cap)} h37=#{hash_key(key, 37, cap)} step=#{1 + hash_key(key, 37, cap - 1)}"]
598 when 'BENCH'
599 cap = prime(capacity(number.call(1)))
600 count, length, seed = (2..4).map(&number)
601 check_generation(count, length, seed)
602 return benchmark(cap, count, length, seed)
603 end
604 raise CommandError, 'NO TABLE' unless @table
605 reporting = @trace ? :trace : :full
606 case command
607 when 'PUT', 'ADD'
608 @table.put(Item.new(words[1].freeze, number.call(2)), command == 'ADD', @maximum, reporting).lines
609 when 'GET' then @table.get(words[1], @trace)
610 when 'DEL' then @table.delete(words[1], @minimum, @maximum, reporting).lines
611 when 'PROBE'
612 found = @table.search(words[1], true)
613 path = found.path.empty? ? '' : ' ' + found.path.join(' ')
614 ["PROBE #{words[1]}#{path} #{found.item ? 'FOUND' : 'MISS'} probes=#{found.probes}"]
615 when 'GEN', 'GENDEL'
616 count, length, seed = (1..3).map(&number)
617 check_generation(count, length, seed)
618 lines = []
619 changed = 0
620 generate(count, length, seed).each_with_index do |key, index|
621 result = if command == 'GEN'
622 @table.put(Item.new(key, index + 1), false, @maximum, :rehashes)
623 else
624 @table.delete(key, @minimum, @maximum, :rehashes)
625 end
626 changed += 1 if result.changed
627 lines.concat(result.lines)
628 end
629 lines << if command == 'GEN'
630 "GEN #{count} new=#{changed} updated=#{count - changed}"
631 else
632 "GENDEL #{count} deleted=#{changed} missing=#{count - changed}"
633 end
634 lines
635 when 'RESIZE', 'CONVERT'
636 cap = command == 'RESIZE' ? prime(capacity(number.call(1))) : @table.cap
637 mode = command == 'CONVERT' ? parse_mode(words[1]) : @table.mode
638 slots = cap * (mode == 'CUCKOO' ? 2 : 1)
639 raise CommandError, 'TOO SMALL' if @table.size * 100 > @maximum * slots
640 replacement = HashTable.rebuild(mode, cap, @table.items)
641 raise CommandError, 'REBUILD FAILED' unless replacement
642 row = command == 'RESIZE' ? "RESIZE #{@table.cap} -> #{cap}" : "CONVERT #{@table.mode} -> #{mode} cap=#{cap}"
643 @table.adopt(replacement)
644 [row]
645 when 'CLEAR'
646 size = @table.size
647 @table.adopt(HashTable.new(@table.mode, @table.cap))
648 ["CLEAR #{size}"]
649 when 'DUMP' then @table.dump
650 when 'MAP' then @table.map
651 when 'STATS' then @table.statistics
652 when 'KEYS'
653 keys = @table.items.map(&:key).sort
654 ["KEYS #{keys.length}#{keys.empty? ? '' : ' ' + keys.join(' ')}"]
655 when 'TOP'
656 limit = number.call(1)
657 raise CommandError, 'BAD ARGUMENT' unless (1..100).cover?(limit)
658 selected = @table.costs.sort_by { |cost| [-cost.probes, cost.key] }.take(limit)
659 ["TOP #{selected.length}"] + selected.map { |cost| "#{cost.key} cost=#{cost.probes} at=#{@table.position(cost.position)}" }
660 when 'SCAN'
661 cursor = number.call(1)
662 limit = number.call(2)
663 raise CommandError, 'BAD ARGUMENT' unless (1..100).cover?(limit)
664 raise CommandError, 'BAD CURSOR' if cursor >= @table.slot_count
665 [@table.scan(cursor, limit)]
666 when 'CLUSTERS' then @table.clusters
667 when 'DIST' then @table.distribution
668 when 'COMPARE' then compare(prime(capacity(number.call(1))))
669 end
670 end
671end

674 simulator = Simulator.new
675 Integer(first, 10).times do
676 begin
677 simulator.execute($stdin.gets).each { |line| puts line }
678 rescue CommandError => error
679 puts "ERROR #{error.message}"
680 end
681 end
