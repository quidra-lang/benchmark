1local modes = {"CHAIN", "LINEAR", "QUADRATIC", "DOUBLE", "ROBINHOOD", "CUCKOO"}
2local known_modes, commands = {}, {}
3for _, mode in ipairs(modes) do known_modes[mode] = true end
4for command in ("CREATE MAXLOAD MINLOAD TRACE HASH PUT ADD GET DEL PROBE GEN GENDEL " ..
5 "RESIZE CONVERT CLEAR DUMP MAP STATS KEYS TOP SCAN CLUSTERS DIST COMPARE BENCH"):gmatch("%S+") do
6 commands[command] = true
7end
8local TOMBSTONE = {}

10local function fail(message) error({command_error = true, message = message}, 0) end
11local function append(target, source)
12 for _, value in ipairs(source) do target[#target + 1] = value end
13end
14local function ascii_less(a, b)
15 for i = 1, math.min(#a, #b) do
16 local x, y = a:byte(i), b:byte(i)
17 if x ~= y then return x < y end
18 end
19 return #a < #b
20end
21local function checked_mode(mode)
22 if not known_modes[mode] then fail("BAD MODE " .. mode) end
23 return mode
24end
25local function hash(key, base, modulus)
26 local value = 0
27 for i = 1, #key do value = (value * base + key:byte(i)) % modulus end
28 return value
29end
30local function prime(value)
31 local candidate = math.max(value, 2)
32 while true do
33 local valid, divisor = true, 2
34 while divisor * divisor <= candidate do
35 if candidate % divisor == 0 then valid = false; break end
36 divisor = divisor + 1
37 end
38 if valid then return candidate end
39 candidate = candidate + 1
40 end
41end
42local function ratio(numerator, denominator)
43 local value = denominator == 0 and 0 or 100 * numerator // denominator
44 return string.format("%d.%02d", value // 100, value % 100)
45end
46local function gauge(value) return value >= 10 and "+" or tostring(value) end
47local function item_text(item) return item.key .. "=" .. item.value end
48local function content(slot)
49 if slot == TOMBSTONE then return "D" end
50 return slot and slot.key or "-"
51end
52local function parts(mode) return mode == "CUCKOO" and 2 or 1 end
53local function uses_tombstones(mode) return mode == "LINEAR" or mode == "QUADRATIC" or mode == "DOUBLE" end
54local function generator(length, seed)
55 return function()
56 local bytes = {}
57 for i = 1, length do
58 seed = 48271 * seed % 2147483647
59 bytes[i] = string.char(97 + seed % 26)
60 end
61 return table.concat(bytes)
62 end
63end
64local function histogram(values)
65 local counts, distinct, rows = {}, {}, {}
66 for _, value in ipairs(values) do
67 if not counts[value] then distinct[#distinct + 1] = value; counts[value] = 0 end
68 counts[value] = counts[value] + 1
69 end
70 table.sort(distinct)
71 for _, value in ipairs(distinct) do
72 local count = counts[value]
73 rows[#rows + 1] = string.format("%d %d %s%s", value, count, string.rep("#", math.min(count, 50)), count > 50 and "+" or "")
74 end
75 return rows
76end
77local function check_capacity(capacity)
78 if capacity < 2 or capacity > 10000 then fail("BAD CAPACITY") end
79end
80local function check_generation(count, length, seed)
81 if count < 1 or count > 500 or length < 1 or length > 10 or seed < 1 or seed > 2147483646 then fail("BAD ARGUMENT") end
82end
83local function path_lines(found, trace)
84 return trace and {"PATH " .. (#found.path == 0 and "none" or table.concat(found.path, " "))} or {}
85end
86local function mutation(changed, row, path, moves, rehashes)
87 local lines = {row}
88 append(lines, path)
89 append(lines, moves)
90 append(lines, rehashes)
91 return {changed = changed, lines = lines, rehashes = rehashes}
92end

96function Table.new(mode, capacity)
97 local self = setmetatable({mode = mode, capacity = capacity, size = 0, tombstones = 0,
98 buckets = {}, slots = {}, counters = {probes = 0, rehashes = 0, inserts = 0, updates = 0,
99 hits = 0, misses = 0, deletes = 0, absent = 0}}, Table)
100 if mode == "CHAIN" then
101 for i = 1, capacity do self.buckets[i] = {} end
102 else
103 for i = 1, capacity * parts(mode) do self.slots[i] = false end
104 end
105 return self
106end
107function Table:slot_count() return self.capacity * parts(self.mode) end
108function Table:position(index)
109 index = index - 1
110 return self.mode == "CUCKOO" and string.format("T%d.%d", index // self.capacity + 1, index % self.capacity) or tostring(index)
111end
112function Table:distance(key, index) return (index - 1 - hash(key, 31, self.capacity)) % self.capacity end
113function Table:items()
114 local result = {}
115 if self.mode == "CHAIN" then
116 for _, bucket in ipairs(self.buckets) do append(result, bucket) end
117 else
118 for _, slot in ipairs(self.slots) do
119 if slot and slot ~= TOMBSTONE then result[#result + 1] = slot end
120 end
121 end
122 return result
123end
124function Table:search(key, trace)
125 local result = {probes = 0, path = {}}
126 local first = hash(key, 31, self.capacity) + 1
127 local function examine(index, slot)
128 result.probes = result.probes + 1
129 if trace then result.path[#result.path + 1] = self:position(index) .. ":" .. content(slot) end
130 end
131 local function found(item, index, chain_index)
132 result.item, result.position, result.chain_index = item, index, chain_index
133 end
134 if self.mode == "CHAIN" then
135 for i, item in ipairs(self.buckets[first]) do
136 examine(first, item)
137 if item.key == key then found(item, first, i); break end
138 end
139 return result
140 end
141 if self.mode == "CUCKOO" then
142 for _, index in ipairs({first, self.capacity + hash(key, 37, self.capacity) + 1}) do
143 local slot = self.slots[index]
144 examine(index, slot)
145 if slot and slot.key == key then found(slot, index); break end
146 end
147 return result
148 end
149 local step = self.mode == "DOUBLE" and 1 + hash(key, 37, self.capacity - 1) or 1
150 for i = 0, self.capacity - 1 do
151 local offset = self.mode == "QUADRATIC" and i * i or i * step
152 local index = (first - 1 + offset) % self.capacity + 1
153 local slot = self.slots[index]
154 examine(index, slot)
155 if not slot then result.available = result.available or index; break end
156 if slot == TOMBSTONE then
157 result.available = result.available or index
158 else
159 if slot.key == key then found(slot, index); break end
160 if self.mode == "ROBINHOOD" and self:distance(slot.key, index) < i then break end
161 end
162 end
163 return result
164end
165function Table:place(item, initial, trace)
166 local result = {success = false, reused = false, shifted = 0, kicks = 0, moves = {}}
167 if self.mode == "CHAIN" then
168 local index = hash(item.key, 31, self.capacity) + 1
169 self.buckets[index][#self.buckets[index] + 1] = item
170 result.position, result.success = index, true
171 return result
172 end
173 if uses_tombstones(self.mode) then
174 local found = initial or self:search(item.key)
175 local index = found.available
176 if not index then result.homeless = item; return result end
177 result.position, result.reused = index, self.slots[index] == TOMBSTONE
178 self.slots[index] = item
179 if result.reused then self.tombstones = self.tombstones - 1 end
180 result.success = true
181 return result
182 end
183 local current = item
184 if self.mode == "ROBINHOOD" then
185 local index, distance, origin = hash(item.key, 31, self.capacity) + 1, 0, nil
186 while true do
187 local resident = self.slots[index]
188 if not resident then
189 self.slots[index] = current
190 if trace and origin then result.moves[#result.moves + 1] = string.format("SHIFT %s %d -> %d", current.key, origin - 1, index - 1) end
191 result.position = result.position or index
192 result.success = true
193 return result
194 end
195 local resident_distance = self:distance(resident.key, index)
196 if resident_distance < distance then
197 self.slots[index] = current
198 if trace and origin then result.moves[#result.moves + 1] = string.format("SHIFT %s %d -> %d", current.key, origin - 1, index - 1) end
199 result.position = result.position or index
200 current, origin, distance = resident, index, resident_distance
201 result.shifted = result.shifted + 1
202 end
203 index, distance = index % self.capacity + 1, distance + 1
204 end
205 end
206 local part = 0
207 while true do
208 local index = part * self.capacity + hash(current.key, part == 0 and 31 or 37, self.capacity) + 1
209 local resident = self.slots[index]
210 if not resident then
211 self.slots[index] = current
212 result.success = true
213 result.position = self:search(item.key).position
214 return result
215 end
216 if result.kicks == self.capacity then result.homeless = current; return result end
217 self.slots[index] = current
218 result.kicks = result.kicks + 1
219 if trace then result.moves[#result.moves + 1] = "KICK " .. resident.key .. " " .. self:position(index) end
220 current, part = resident, 1 - part
221 end
222end
223function Table:remove(found, trace)
224 local result = {shifted = 0, moves = {}}
225 if self.mode == "CHAIN" then
226 table.remove(self.buckets[found.position], found.chain_index)
227 elseif uses_tombstones(self.mode) then
228 self.slots[found.position] = TOMBSTONE
229 self.tombstones = self.tombstones + 1
230 else
231 self.slots[found.position] = false
232 if self.mode == "ROBINHOOD" then
233 local following = found.position % self.capacity + 1
234 while self.slots[following] and self:distance(self.slots[following].key, following) > 0 do
235 local item = self.slots[following]
236 local destination = (following - 2) % self.capacity + 1
237 self.slots[destination], self.slots[following] = item, false
238 if trace then result.moves[#result.moves + 1] = string.format("SHIFT %s %d -> %d", item.key, following - 1, destination - 1) end
239 result.shifted = result.shifted + 1
240 following = following % self.capacity + 1
241 end
242 end
243 end
244 self.size = self.size - 1
245 return result
246end
247function Table.rebuild(items, mode, capacity)
248 local trial = Table.new(mode, capacity)
249 for _, item in ipairs(items) do
250 if not trial:place(item).success then return nil end
251 trial.size = trial.size + 1
252 end
253 return trial
254end
255function Table:adopt(replacement)
256 self.mode, self.capacity = replacement.mode, replacement.capacity
257 self.size, self.tombstones = replacement.size, replacement.tombstones
258 self.buckets, self.slots = replacement.buckets, replacement.slots
259end
260function Table:automatic(reason, capacity, pending)
261 local items, old = self:items(), self.capacity
262 if pending then items[#items + 1] = pending end
263 while true do
264 local trial = Table.rebuild(items, self.mode, capacity)
265 if trial then
266 self:adopt(trial)
267 self.counters.rehashes = self.counters.rehashes + 1
268 return string.format("REHASH %s %d -> %d", reason, old, capacity)
269 end
270 capacity = prime(2 * capacity)
271 end
272end
273function Table:load_check(max_load)
274 local threshold = max_load * self:slot_count()
275 if 100 * (self.size + self.tombstones) <= threshold then return {} end
276 local grow = 200 * self.size > threshold
277 return {self:automatic(grow and "load" or "tomb", grow and prime(2 * self.capacity) or self.capacity)}
278end
279function Table:shrink_check(min_load, max_load)
280 local capacity = prime(self.capacity // 2)
281 if 100 * self.size >= min_load * self:slot_count() or capacity >= self.capacity or
282 100 * self.size > max_load * capacity * parts(self.mode) then return {} end
283 local trial = Table.rebuild(self:items(), self.mode, capacity)
284 if not trial then return {} end
285 local old = self.capacity
286 self:adopt(trial)
287 self.counters.rehashes = self.counters.rehashes + 1
288 return {string.format("REHASH shrink %d -> %d", old, capacity)}
289end
290function Table:put(key, value, max_load, command, trace)
291 command = command or "PUT"
292 local found = self:search(key, trace)
293 local path = path_lines(found, trace)
294 self.counters.probes = self.counters.probes + found.probes
295 if found.item then
296 local old = found.item.value
297 local item = {key = key, value = command == "ADD" and old + value or value}
298 if self.mode == "CHAIN" then self.buckets[found.position][found.chain_index] = item else self.slots[found.position] = item end
299 self.counters.updates = self.counters.updates + 1
300 local row = string.format("%s %s UPDATED old=%d at=%s probes=%d", command, item_text(item), old, self:position(found.position), found.probes)
301 return mutation(false, row, path, {}, {})
302 end
303 local item = {key = key, value = value}
304 local placed = self:place(item, found, trace)
305 self.size = self.size + 1
306 self.counters.inserts = self.counters.inserts + 1
307 local row = command .. " " .. item_text(item) .. " NEW"
308 if placed.success then
309 row = row .. string.format(" at=%s probes=%d", self:position(placed.position), found.probes)
310 if placed.reused then row = row .. " reused" end
311 if self.mode == "ROBINHOOD" then row = row .. " shifted=" .. placed.shifted end
312 if self.mode == "CUCKOO" then row = row .. " kicks=" .. placed.kicks end
313 else
314 row = row .. " probes=" .. found.probes .. (self.mode == "CUCKOO" and " kicks=" .. placed.kicks .. " failed" or " full")
315 end
316 local rehashes = {}
317 if not placed.success then
318 rehashes[#rehashes + 1] = self:automatic(self.mode == "CUCKOO" and "kicks" or "full", prime(2 * self.capacity), placed.homeless)
319 end
320 append(rehashes, self:load_check(max_load))
321 return mutation(true, row, path, placed.moves, rehashes)
322end
323function Table:get(key, trace)
324 local found = self:search(key, trace)
325 self.counters.probes = self.counters.probes + found.probes
326 local row
327 if found.item then
328 self.counters.hits = self.counters.hits + 1
329 row = string.format("GET %s at=%s probes=%d", item_text(found.item), self:position(found.position), found.probes)
330 else
331 self.counters.misses = self.counters.misses + 1
332 row = string.format("GET %s NOT FOUND probes=%d", key, found.probes)
333 end
334 local rows = {row}
335 append(rows, path_lines(found, trace))
336 return rows
337end
338function Table:delete(key, min_load, max_load, trace)
339 local found = self:search(key, trace)
340 local path = path_lines(found, trace)
341 self.counters.probes = self.counters.probes + found.probes
342 if not found.item then
343 self.counters.absent = self.counters.absent + 1
344 return mutation(false, string.format("DEL %s NOT FOUND probes=%d", key, found.probes), path, {}, {})
345 end
346 local removed = self:remove(found, trace)
347 self.counters.deletes = self.counters.deletes + 1
348 local row = string.format("DEL %s OK at=%s probes=%d", key, self:position(found.position), found.probes)
349 if self.mode == "ROBINHOOD" then row = row .. " shifted=" .. removed.shifted end
350 return mutation(true, row, path, removed.moves, self:shrink_check(min_load, max_load))
351end
352function Table:costs()
353 local records = {}
354 if self.mode == "CHAIN" then
355 for index, bucket in ipairs(self.buckets) do
356 for i, item in ipairs(bucket) do records[#records + 1] = {key = item.key, probes = i, position = index} end
357 end
358 else
359 for index, slot in ipairs(self.slots) do
360 if slot and slot ~= TOMBSTONE then
361 local cost
362 if self.mode == "CUCKOO" then cost = (index - 1) // self.capacity + 1
363 elseif self.mode == "LINEAR" or self.mode == "ROBINHOOD" then cost = self:distance(slot.key, index) + 1
364 else cost = self:search(slot.key).probes end
365 records[#records + 1] = {key = slot.key, probes = cost, position = index}
366 end
367 end
368 end
369 return records
370end
371function Table:metrics()
372 local total, longest = 0, 0
373 for _, cost in ipairs(self:costs()) do total, longest = total + cost.probes, math.max(longest, cost.probes) end
374 return {longest = longest, average = ratio(total, self.size)}
375end
376function Table:dump()
377 local rows = {string.format("DUMP %s cap=%d size=%d tomb=%d", self.mode, self.capacity, self.size, self.tombstones)}
378 if self.mode == "CHAIN" then
379 for index, bucket in ipairs(self.buckets) do
380 local words = {tostring(index - 1)}
381 if #bucket == 0 then words[#words + 1] = "-" end
382 for _, item in ipairs(bucket) do words[#words + 1] = item_text(item) end
383 rows[#rows + 1] = table.concat(words, " ")
384 end
385 else
386 for index, slot in ipairs(self.slots) do
387 local label = self.mode == "CUCKOO" and string.format("T%d %d", (index - 1) // self.capacity + 1, (index - 1) % self.capacity) or tostring(index - 1)
388 local value = content(slot)
389 if slot and slot ~= TOMBSTONE then
390 value = item_text(slot)
391 if self.mode == "ROBINHOOD" then value = value .. " d=" .. self:distance(slot.key, index) end
392 end
393 rows[#rows + 1] = label .. " " .. value
394 end
395 end
396 return rows
397end
398function Table:map()
399 local rows = {string.format("MAP %s cap=%d", self.mode, self.capacity)}
400 local cost_by_slot = {}
401 if self.mode ~= "CHAIN" then
402 for _, cost in ipairs(self:costs()) do cost_by_slot[cost.position] = cost.probes end
403 end
404 for part = 0, parts(self.mode) - 1 do
405 local characters = {}
406 for index = 1, self.capacity do
407 if self.mode == "CHAIN" then
408 local count = #self.buckets[index]
409 characters[index] = count == 0 and "." or gauge(count)
410 else
411 local flat = part * self.capacity + index
412 local slot = self.slots[flat]
413 if not slot then characters[index] = "."
414 elseif slot == TOMBSTONE then characters[index] = "D"
415 elseif self.mode == "CUCKOO" then characters[index] = "#"
416 else characters[index] = gauge(cost_by_slot[flat] - 1) end
417 end
418 end
419 for index = 1, self.capacity, 40 do
420 local prefix = self.mode == "CUCKOO" and "T" .. (part + 1) .. " " or ""
421 rows[#rows + 1] = prefix .. string.format("%05d ", index - 1) .. table.concat(characters, "", index, math.min(index + 39, self.capacity))
422 end
423 end
424 return rows
425end
426function Table:statistics()
427 local values, count = self:metrics(), self.counters
428 return {string.format("STATS %s size=%d cap=%d slots=%d tomb=%d load=%s%% longest=%d avg=%s probes=%d rehashes=%d",
429 self.mode, self.size, self.capacity, self:slot_count(), self.tombstones, ratio(100 * self.size, self:slot_count()),
430 values.longest, values.average, count.probes, count.rehashes),
431 string.format("OPS inserts=%d updates=%d hits=%d misses=%d deletes=%d absent=%d", count.inserts, count.updates, count.hits, count.misses, count.deletes, count.absent)}
432end
433function Table:scan(cursor, limit)
434 local keys = {}
435 while cursor < self:slot_count() and #keys < limit do
436 local index = cursor + 1
437 if self.mode == "CHAIN" then
438 for _, item in ipairs(self.buckets[index]) do keys[#keys + 1] = item.key end
439 else
440 local slot = self.slots[index]
441 if slot and slot ~= TOMBSTONE then keys[#keys + 1] = slot.key end
442 end
443 cursor = cursor + 1
444 end
445 if cursor == self:slot_count() then cursor = 0 end
446 local words = {"SCAN next=" .. cursor}
447 append(words, keys)
448 return {table.concat(words, " ")}
449end
450function Table:runs(start, count)
451 local empty
452 for i = 0, count - 1 do if not self.slots[start + i] then empty = i; break end end
453 if empty == nil then return {count} end
454 local lengths, length = {}, 0
455 for offset = 1, count do
456 if self.slots[start + (empty + offset) % count] then length = length + 1
457 elseif length > 0 then lengths[#lengths + 1] = length; length = 0 end
458 end
459 return lengths
460end
461function Table:clusters()
462 local lengths = {}
463 if self.mode == "CHAIN" then
464 for _, bucket in ipairs(self.buckets) do if #bucket > 0 then lengths[#lengths + 1] = #bucket end end
465 else
466 for part = 0, parts(self.mode) - 1 do append(lengths, self:runs(part * self.capacity + 1, self.capacity)) end
467 end
468 local longest = 0
469 for _, length in ipairs(lengths) do longest = math.max(longest, length) end
470 local rows = {string.format("CLUSTERS runs=%d longest=%d", #lengths, longest)}
471 append(rows, histogram(lengths))
472 return rows
473end
474function Table:distribution()
475 local values, longest = {}, 0
476 for _, cost in ipairs(self:costs()) do values[#values + 1] = cost.probes; longest = math.max(longest, cost.probes) end
477 local rows = {string.format("DIST keys=%d longest=%d", self.size, longest)}
478 append(rows, histogram(values))
479 return rows
480end

484function Simulator.new() return setmetatable({max_load = 75, min_load = 0, trace = false}, Simulator) end
485function Simulator:compare(capacity)
486 local items = self.current:items()
487 table.sort(items, function(a, b) return ascii_less(a.key, b.key) end)
488 local rows = {string.format("COMPARE keys=%d cap=%d", #items, capacity)}
489 for _, mode in ipairs(modes) do
490 local trial = Table.new(mode, capacity)
491 for _, item in ipairs(items) do trial:put(item.key, item.value, self.max_load) end
492 local values = trial:metrics()
493 rows[#rows + 1] = string.format("%s cap=%d longest=%d avg=%s probes=%d rehashes=%d", mode, trial.capacity, values.longest, values.average, trial.counters.probes, trial.counters.rehashes)
494 end
495 return rows
496end
497function Simulator:bench(capacity, count, length, seed)
498 local next_key, keys = generator(length, seed), {}
499 for i = 1, 2 * count do keys[i] = next_key() end
500 local rows = {string.format("BENCH cap=%d count=%d len=%d seed=%d", capacity, count, length, seed)}
501 for _, mode in ipairs(modes) do
502 local trial = Table.new(mode, capacity)
503 for i = 1, count do trial:put(keys[i], i, self.max_load) end
504 local misses, probes = 0, 0
505 for i = count + 1, 2 * count do
506 local found = trial:search(keys[i])
507 if not found.item then misses, probes = misses + 1, probes + found.probes end
508 end
509 rows[#rows + 1] = string.format("%s cap=%d keys=%d hit=%s misses=%d miss=%s rehashes=%d", mode, trial.capacity, trial.size,
510 trial:metrics().average, misses, ratio(probes, misses), trial.counters.rehashes)
511 end
512 return rows
513end
514function Simulator:execute(line)
515 local words = {}
516 for word in line:gmatch("%S+") do words[#words + 1] = word end
517 local command = words[1]
518 if not commands[command] then fail("UNKNOWN COMMAND " .. command) end
519 local function number(index) return tonumber(words[index], 10) end
520 if command == "CREATE" then
521 local mode = checked_mode(words[2])
522 local capacity = number(3)
523 check_capacity(capacity)
524 self.current = Table.new(mode, prime(capacity))
525 return {string.format("CREATE %s cap=%d", mode, self.current.capacity)}
526 elseif command == "MAXLOAD" or command == "MINLOAD" then
527 local value, maximum = number(2), command == "MAXLOAD"
528 if maximum and (value < 10 or value > 95) or not maximum and (value < 0 or value > 40) then fail("BAD LOAD") end
529 if maximum then self.max_load = value else self.min_load = value end
530 return {command .. " " .. value}
531 elseif command == "TRACE" then
532 if words[2] ~= "ON" and words[2] ~= "OFF" then fail("BAD ARGUMENT") end
533 self.trace = words[2] == "ON"
534 return {"TRACE " .. words[2]}
535 elseif command == "HASH" then
536 local key, capacity = words[2], number(3)
537 check_capacity(capacity)
538 return {string.format("HASH %s m=%d h31=%d h37=%d step=%d", key, capacity, hash(key, 31, capacity), hash(key, 37, capacity), 1 + hash(key, 37, capacity - 1))}
539 elseif command == "BENCH" then
540 local capacity = number(2)
541 check_capacity(capacity)
542 local count, length, seed = number(3), number(4), number(5)
543 check_generation(count, length, seed)
544 return self:bench(prime(capacity), count, length, seed)
545 end
546 local current = self.current
547 if not current then fail("NO TABLE") end
548 if command == "PUT" or command == "ADD" then
549 return current:put(words[2], number(3), self.max_load, command, self.trace).lines
550 elseif command == "GET" then return current:get(words[2], self.trace)
551 elseif command == "DEL" then return current:delete(words[2], self.min_load, self.max_load, self.trace).lines
552 elseif command == "PROBE" then
553 local found = current:search(words[2], true)
554 local output = {"PROBE " .. words[2]}
555 append(output, found.path)
556 output[#output + 1] = (found.item and "FOUND" or "MISS") .. " probes=" .. found.probes
557 return {table.concat(output, " ")}
558 elseif command == "GEN" or command == "GENDEL" then
559 local count, length, seed = number(2), number(3), number(4)
560 check_generation(count, length, seed)
561 local next_key, rows, changed = generator(length, seed), {}, 0
562 for i = 1, count do
563 local key = next_key()
564 local result = command == "GEN" and current:put(key, i, self.max_load) or current:delete(key, self.min_load, self.max_load)
565 if result.changed then changed = changed + 1 end
566 append(rows, result.rehashes)
567 end
568 rows[#rows + 1] = command == "GEN" and string.format("GEN %d new=%d updated=%d", count, changed, count - changed)
569 or string.format("GENDEL %d deleted=%d missing=%d", count, changed, count - changed)
570 return rows
571 elseif command == "RESIZE" or command == "CONVERT" then
572 local mode, capacity = current.mode, current.capacity
573 if command == "RESIZE" then capacity = number(2); check_capacity(capacity); capacity = prime(capacity)
574 else mode = checked_mode(words[2]) end
575 if 100 * current.size > self.max_load * capacity * parts(mode) then fail("TOO SMALL") end
576 local trial = Table.rebuild(current:items(), mode, capacity)
577 if not trial then fail("REBUILD FAILED") end
578 local row = command == "RESIZE" and string.format("RESIZE %d -> %d", current.capacity, capacity)
579 or string.format("CONVERT %s -> %s cap=%d", current.mode, mode, capacity)
580 current:adopt(trial)
581 return {row}
582 elseif command == "CLEAR" then
583 local size = current.size
584 current:adopt(Table.new(current.mode, current.capacity))
585 return {"CLEAR " .. size}
586 elseif command == "DUMP" then return current:dump()
587 elseif command == "MAP" then return current:map()
588 elseif command == "STATS" then return current:statistics()
589 elseif command == "KEYS" then
590 local keys = {}
591 for _, item in ipairs(current:items()) do keys[#keys + 1] = item.key end
592 table.sort(keys, ascii_less)
593 local output = {"KEYS " .. current.size}
594 append(output, keys)
595 return {table.concat(output, " ")}
596 elseif command == "TOP" then
597 local limit = number(2)
598 if limit < 1 or limit > 100 then fail("BAD ARGUMENT") end
599 local records = current:costs()
600 table.sort(records, function(a, b)
601 if a.probes ~= b.probes then return a.probes > b.probes end
602 return ascii_less(a.key, b.key)
603 end)
604 local count = math.min(limit, #records)
605 local rows = {"TOP " .. count}
606 for i = 1, count do
607 local cost = records[i]
608 rows[#rows + 1] = string.format("%s cost=%d at=%s", cost.key, cost.probes, current:position(cost.position))
609 end
610 return rows
611 elseif command == "SCAN" then
612 local cursor, limit = number(2), number(3)
613 if limit < 1 or limit > 100 then fail("BAD ARGUMENT") end
614 if cursor >= current:slot_count() then fail("BAD CURSOR") end
615 return current:scan(cursor, limit)
616 elseif command == "CLUSTERS" then return current:clusters()
617 elseif command == "DIST" then return current:distribution()
618 elseif command == "COMPARE" then
619 local capacity = number(2)
620 check_capacity(capacity)
621 return self:compare(prime(capacity))
622 end
623 error("Unhandled command " .. command, 0)
624end

631 if ok then io.write(table.concat(rows, "\n"), "\n")
632 elseif type(rows) == "table" and rows.command_error then io.write("ERROR ", rows.message, "\n")
