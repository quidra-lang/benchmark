4enum Mode {

6 doubleHash('DOUBLE'), robinHood('ROBINHOOD'), cuckoo('CUCKOO');

8 const Mode(this.label);
9 final String label;
10 int get parts => this == cuckoo ? 2 : 1;
11 bool get usesTombstones => this == linear || this == quadratic || this == doubleHash;

73int hash(String key, int base, int modulus) {
74 var value = 0;
75 for (final character in key.codeUnits) {
76 value = (value * base + character) % modulus;
77 }
78 return value;
79}

81int prime(int value) {
82 for (var candidate = math.max(value, 2); ; candidate++) {
83 var valid = true;
84 for (var divisor = 2; divisor * divisor <= candidate; divisor++) {
85 if (candidate % divisor == 0) { valid = false; break; }
86 }
87 if (valid) return candidate;
88 }
89}

92 final value = denominator == 0 ? 0 : 100 * numerator ~/ denominator;
93 return '${value ~/ 100}.${(value % 100).toString().padLeft(2, '0')}';
94}

127final class HashTable {
128 HashTable(this.mode, this.capacity) {
129 buckets = mode == Mode.chain ? List.generate(capacity, (_) => <Item>[]) : [];
130 slots = mode == Mode.chain ? [] : List.filled(slotCount, const Slot());
131 }
132
133 Mode mode;
134 int capacity;
135 int size = 0, tombstones = 0;
136 final counters = Counters();
137 late List<List<Item>> buckets;
138 late List<Slot> slots;
139 int get slotCount => capacity * mode.parts;
140
141 String position(int index) => mode == Mode.cuckoo ? 'T${index ~/ capacity + 1}.${index % capacity}' : '$index';
142 int distance(String key, int index) => (index - hash(key, 31, capacity)) % capacity;
143
144 List<Item> items() => mode == Mode.chain
145 ? buckets.expand((bucket) => bucket).toList()
146 : [for (final slot in slots) if (slot.item case final item?) item];
147
148 SearchResult search(String key, {bool trace = false}) {
149 final result = SearchResult();
150 final first = hash(key, 31, capacity);
151 void examine(int index, String content) {
152 result.probes++;
153 if (trace) result.path.add('${position(index)}:$content');
154 }
155 void found(Item item, int index, [int chainIndex = -1]) {
156 result.item = item;
157 result.position = index;
158 result.chainIndex = chainIndex;
159 }
160 if (mode == Mode.chain) {
161 for (var i = 0; i < buckets[first].length; i++) {
162 final item = buckets[first][i];
163 examine(first, item.key);
164 if (item.key == key) { found(item, first, i); break; }
165 }
166 return result;
167 }
168 if (mode == Mode.cuckoo) {
169 for (final index in [first, capacity + hash(key, 37, capacity)]) {
170 final slot = slots[index];
171 examine(index, slot.content);
172 final item = slot.item;
173 if (item != null && item.key == key) { found(item, index); break; }
174 }
175 return result;
176 }
177 final step = mode == Mode.doubleHash ? 1 + hash(key, 37, capacity - 1) : 1;
178 for (var i = 0; i < capacity; i++) {
179 final offset = mode == Mode.quadratic ? i * i : i * step;
180 final index = (first + offset) % capacity;
181 final slot = slots[index];
182 examine(index, slot.content);
183 if (slot.empty) { result.available ??= index; break; }
184 if (slot.deleted) { result.available ??= index; continue; }
185 final item = slot.item!;
186 if (item.key == key) { found(item, index); break; }
187 if (mode == Mode.robinHood && distance(item.key, index) < i) break;
188 }
189 return result;
190 }
191
192 Placement place(Item item, {SearchResult? initial, bool trace = false}) {
193 final result = Placement();
194 if (mode == Mode.chain) {
195 result.position = hash(item.key, 31, capacity);
196 buckets[result.position].add(item);
197 result.success = true;
198 return result;
199 }
200 if (mode.usesTombstones) {
201 final found = initial ?? search(item.key);
202 final index = found.available;
203 if (index == null) { result.homeless = item; return result; }
204 result.position = index;
205 result.reused = slots[index].deleted;
206 slots[index] = Slot(item);
207 if (result.reused) tombstones--;
208 result.success = true;
209 return result;
210 }
211 var current = item;
212 if (mode == Mode.robinHood) {
213 var index = hash(item.key, 31, capacity), dist = 0;
214 int? from;
215 while (true) {
216 final resident = slots[index].item;
217 if (resident == null) {
218 slots[index] = Slot(current);
219 if (trace && from != null) result.moves.add('SHIFT ${current.key} $from -> $index');
220 if (result.position < 0) result.position = index;
221 result.success = true;
222 return result;
223 }
224 final residentDistance = distance(resident.key, index);
225 if (residentDistance < dist) {
226 slots[index] = Slot(current);
227 if (trace && from != null) result.moves.add('SHIFT ${current.key} $from -> $index');
228 if (result.position < 0) result.position = index;
229 current = resident;
230 from = index;
231 dist = residentDistance;
232 result.shifted++;
233 }
234 index = (index + 1) % capacity;
235 dist++;
236 }
237 }
238 var part = 0;
239 while (true) {
240 final index = part * capacity + hash(current.key, part == 0 ? 31 : 37, capacity);
241 final resident = slots[index].item;
242 if (resident == null) {
243 slots[index] = Slot(current);
244 result.success = true;
245 result.position = search(item.key).position;
246 return result;
247 }
248 if (result.kicks == capacity) { result.homeless = current; return result; }
249 slots[index] = Slot(current);
250 result.kicks++;
251 if (trace) result.moves.add('KICK ${resident.key} ${position(index)}');
252 current = resident;
253 part = 1 - part;
254 }
255 }
256
257 Removal remove(SearchResult found, {bool trace = false}) {
258 var shifted = 0;
259 final moves = <String>[];
260 if (mode == Mode.chain) { buckets[found.position].removeAt(found.chainIndex); }
261 else if (mode.usesTombstones) { slots[found.position] = const Slot(null, true); tombstones++; }
262 else {
263 slots[found.position] = const Slot();
264 if (mode == Mode.robinHood) {
265 var next = (found.position + 1) % capacity;
266 while (true) {
267 final item = slots[next].item;
268 if (item == null || distance(item.key, next) == 0) break;
269 final destination = (next - 1) % capacity;
270 slots[destination] = Slot(item);
271 slots[next] = const Slot();
272 if (trace) moves.add('SHIFT ${item.key} $next -> $destination');
273 shifted++;
274 next = (next + 1) % capacity;
275 }
276 }
277 }
278 size--;
279 return (shifted: shifted, moves: moves);
280 }

282 static HashTable? rebuild(List<Item> items, Mode mode, int capacity) {
283 final table = HashTable(mode, capacity);
284 for (final item in items) {
285 if (!table.place(item).success) return null;
286 table.size++;
287 }
288 return table;
289 }

291 void adopt(HashTable replacement) {
292 mode = replacement.mode;
293 capacity = replacement.capacity;
294 size = replacement.size;
295 tombstones = replacement.tombstones;
296 buckets = replacement.buckets;
297 slots = replacement.slots;

300 String automatic(String reason, int target, [Item? pending]) {
301 final listed = items();
302 if (pending != null) listed.add(pending);
303 final old = capacity;
304 while (true) {
305 final replacement = rebuild(listed, mode, target);
306 if (replacement != null) {
307 adopt(replacement);
308 counters.rehashes++;
309 return 'REHASH $reason $old -> $capacity';
310 }
311 target = prime(2 * target);
312 }
313 }

315 List<String> loadCheck(int maxLoad) {
316 final threshold = maxLoad * slotCount;
317 if (100 * (size + tombstones) <= threshold) return [];
318 final grow = 200 * size > threshold;
319 return [automatic(grow ? 'load' : 'tomb', grow ? prime(2 * capacity) : capacity)];
320 }
321
322 List<String> shrinkCheck(int minLoad, int maxLoad) {
323 final target = prime(capacity ~/ 2);
324 if (100 * size >= minLoad * slotCount || target >= capacity || 100 * size > maxLoad * target * mode.parts) return [];
325 final replacement = rebuild(items(), mode, target);
326 if (replacement == null) return [];
327 final old = capacity;
328 adopt(replacement);
329 counters.rehashes++;
330 return ['REHASH shrink $old -> $target'];
331 }

333 Mutation put(String key, int value, int maxLoad, {String command = 'PUT', bool trace = false}) {
334 final found = search(key, trace: trace);
335 final path = trace ? ['PATH ${found.path.isEmpty ? 'none' : found.path.join(' ')}'] : <String>[];
336 counters.probes += found.probes;
337 final old = found.item;
338 if (old != null) {
339 final item = Item(key, command == 'ADD' ? old.value + value : value);
340 if (mode == Mode.chain) { buckets[found.position][found.chainIndex] = item; }
341 else { slots[found.position] = Slot(item); }
342 counters.updates++;
343 final row = '$command $item UPDATED old=${old.value} at=${position(found.position)} probes=${found.probes}';
344 return Mutation(false, [row, ...path], []);
345 }
346 final item = Item(key, value);
347 final placed = place(item, initial: found, trace: trace);
348 size++;
349 counters.inserts++;
350 var row = '$command $item NEW';
351 if (placed.success) {
352 row += ' at=${position(placed.position)} probes=${found.probes}';
353 if (placed.reused) row += ' reused';
354 if (mode == Mode.robinHood) row += ' shifted=${placed.shifted}';
355 if (mode == Mode.cuckoo) row += ' kicks=${placed.kicks}';
356 } else {
357 row += ' probes=${found.probes}' + (mode == Mode.cuckoo ? ' kicks=${placed.kicks} failed' : ' full');
358 }
359 final rehashes = <String>[];
360 if (!placed.success) {
361 rehashes.add(automatic(mode == Mode.cuckoo ? 'kicks' : 'full', prime(2 * capacity), placed.homeless));
362 }
363 rehashes.addAll(loadCheck(maxLoad));
364 return Mutation(true, [row, ...path, ...placed.moves, ...rehashes], rehashes);
365 }
366
367 List<String> get(String key, {bool trace = false}) {
368 final found = search(key, trace: trace);
369 counters.probes += found.probes;
370 final item = found.item;
371 String row;
372 if (item != null) {
373 counters.hits++;
374 row = 'GET $item at=${position(found.position)} probes=${found.probes}';
375 } else {
376 counters.misses++;
377 row = 'GET $key NOT FOUND probes=${found.probes}';
378 }
379 return [row, if (trace) 'PATH ${found.path.isEmpty ? 'none' : found.path.join(' ')}'];
380 }
381
382 Mutation delete(String key, int minLoad, int maxLoad, {bool trace = false}) {
383 final found = search(key, trace: trace);
384 final path = trace ? ['PATH ${found.path.isEmpty ? 'none' : found.path.join(' ')}'] : <String>[];
385 counters.probes += found.probes;
386 if (found.item == null) {
387 counters.absent++;
388 return Mutation(false, ['DEL $key NOT FOUND probes=${found.probes}', ...path], []);
389 }
390 final removed = remove(found, trace: trace);
391 counters.deletes++;
392 var row = 'DEL $key OK at=${position(found.position)} probes=${found.probes}';
393 if (mode == Mode.robinHood) row += ' shifted=${removed.shifted}';
394 final rehashes = shrinkCheck(minLoad, maxLoad);
395 return Mutation(true, [row, ...path, ...removed.moves, ...rehashes], rehashes);
396 }

398 List<Cost> costs() {
399 final records = <Cost>[];
400 if (mode == Mode.chain) {
401 for (var index = 0; index < capacity; index++) {
402 for (var i = 0; i < buckets[index].length; i++) {
403 records.add((key: buckets[index][i].key, probes: i + 1, position: index));
404 }
405 }
406 } else {
407 for (var index = 0; index < slots.length; index++) {
408 final item = slots[index].item;
409 if (item == null) continue;
410 final cost = mode == Mode.cuckoo ? index ~/ capacity + 1
411 : mode == Mode.linear || mode == Mode.robinHood ? distance(item.key, index) + 1
412 : search(item.key).probes;
413 records.add((key: item.key, probes: cost, position: index));
414 }
415 }
416 return records;
417 }
418
419 Metrics metrics() {
420 var total = 0, longest = 0;
421 for (final cost in costs()) {
422 total += cost.probes;
423 longest = math.max(longest, cost.probes);
424 }
425 return (longest: longest, average: ratio(total, size));
426 }

428 List<String> dump() {
429 final rows = ['DUMP ${mode.label} cap=$capacity size=$size tomb=$tombstones'];
430 if (mode == Mode.chain) {
431 for (var index = 0; index < capacity; index++) {
432 rows.add('$index ${buckets[index].isEmpty ? '-' : buckets[index].join(' ')}');
433 }
434 } else {
435 for (var index = 0; index < slots.length; index++) {
436 final slot = slots[index];
437 final item = slot.item;
438 final label = mode == Mode.cuckoo ? 'T${index ~/ capacity + 1} ${index % capacity}' : '$index';
439 var value = item?.toString() ?? slot.content;
440 if (mode == Mode.robinHood && item != null) value += ' d=${distance(item.key, index)}';
441 rows.add('$label $value');
442 }
443 }
444 return rows;
445 }

447 List<String> map() {
448 final rows = ['MAP ${mode.label} cap=$capacity'];
449 final costBySlot = List.filled(slotCount, 0);
450 if (mode != Mode.chain) {
451 for (final cost in costs()) { costBySlot[cost.position] = cost.probes; }
452 }
453 for (var part = 0; part < mode.parts; part++) {
454 final characters = <String>[];
455 for (var index = 0; index < capacity; index++) {
456 if (mode == Mode.chain) {
457 characters.add(buckets[index].isEmpty ? '.' : gauge(buckets[index].length));
458 } else {
459 final flat = part * capacity + index;
460 final slot = slots[flat];
461 characters.add(slot.empty ? '.' : slot.deleted ? 'D'
462 : mode == Mode.cuckoo ? '#' : gauge(costBySlot[flat] - 1));
463 }
464 }
465 for (var index = 0; index < capacity; index += 40) {
466 final prefix = mode == Mode.cuckoo ? 'T${part + 1} ' : '';
467 rows.add('$prefix${index.toString().padLeft(5, '0')} ${characters.sublist(index, math.min(index + 40, capacity)).join()}');
468 }
469 }
470 return rows;
471 }

474 final values = metrics();
475 final count = counters;
476 return ['STATS ${mode.label} size=$size cap=$capacity slots=$slotCount tomb=$tombstones '
477 'load=${ratio(100 * size, slotCount)}% longest=${values.longest} avg=${values.average} '
478 'probes=${count.probes} rehashes=${count.rehashes}',
479 'OPS inserts=${count.inserts} updates=${count.updates} hits=${count.hits} misses=${count.misses} '

483 List<String> scan(int cursor, int limit) {
484 final keys = <String>[];
485 var next = cursor;
486 while (next < slotCount && keys.length < limit) {
487 if (mode == Mode.chain) { keys.addAll(buckets[next].map((item) => item.key)); }
488 else if (slots[next].item case final item?) { keys.add(item.key); }
489 next++;
490 }
491 if (next == slotCount) next = 0;
492 return [['SCAN next=$next', ...keys].join(' ')];
493 }
494
495 List<int> runs(int start, int count) {
496 var empty = -1;
497 for (var i = 0; i < count; i++) {
498 if (slots[start + i].empty) { empty = i; break; }
499 }
500 if (empty < 0) return [count];
501 final lengths = <int>[];
502 var length = 0;
503 for (var offset = 1; offset <= count; offset++) {
504 if (!slots[start + (empty + offset) % count].empty) { length++; }
505 else if (length > 0) { lengths.add(length); length = 0; }
506 }
507 return lengths;
508 }

510 List<String> clusters() {
511 final lengths = mode == Mode.chain
512 ? [for (final bucket in buckets) if (bucket.isNotEmpty) bucket.length]
513 : [for (var part = 0; part < mode.parts; part++) ...runs(part * capacity, capacity)];
514 final longest = lengths.fold(0, (value, length) => math.max(value, length));
515 return ['CLUSTERS runs=${lengths.length} longest=$longest', ...histogram(lengths)];
516 }
517
518 List<String> distribution() {
519 final values = costs().map((cost) => cost.probes).toList();
520 final longest = values.fold(0, (value, cost) => math.max(value, cost));
521 return ['DIST keys=$size longest=$longest', ...histogram(values)];
522 }

526 HashTable? table;
527 int maxLoad = 75, minLoad = 0;
528 bool tracing = false;
529 static const commands = {'CREATE', 'MAXLOAD', 'MINLOAD', 'TRACE', 'HASH', 'PUT', 'ADD',
530 'GET', 'DEL', 'PROBE', 'GEN', 'GENDEL', 'RESIZE', 'CONVERT', 'CLEAR', 'DUMP', 'MAP',
531 'STATS', 'KEYS', 'TOP', 'SCAN', 'CLUSTERS', 'DIST', 'COMPARE', 'BENCH'};

533 List<String> compare(int capacity) {
534 final listed = table!.items()..sort((a, b) => a.key.compareTo(b.key));
535 final rows = ['COMPARE keys=${listed.length} cap=$capacity'];
536 for (final mode in Mode.values) {
537 final trial = HashTable(mode, capacity);
538 for (final item in listed) { trial.put(item.key, item.value, maxLoad); }
539 final values = trial.metrics();
540 rows.add('${mode.label} cap=${trial.capacity} longest=${values.longest} avg=${values.average} '
541 'probes=${trial.counters.probes} rehashes=${trial.counters.rehashes}');
542 }
543 return rows;
544 }

546 List<String> bench(int capacity, int count, int length, int seed) {
547 final keys = generate(2 * count, length, seed).toList();
548 final rows = ['BENCH cap=$capacity count=$count len=$length seed=$seed'];
549 for (final mode in Mode.values) {
550 final trial = HashTable(mode, capacity);
551 for (var i = 0; i < count; i++) { trial.put(keys[i], i + 1, maxLoad); }
552 var misses = 0, probes = 0;
553 for (var i = count; i < 2 * count; i++) {
554 final found = trial.search(keys[i]);
555 if (found.item == null) { misses++; probes += found.probes; }
556 }
557 rows.add('${mode.label} cap=${trial.capacity} keys=${trial.size} hit=${trial.metrics().average} '
558 'misses=$misses miss=${ratio(probes, misses)} rehashes=${trial.counters.rehashes}');
559 }
560 return rows;
561 }

563 List<String> execute(String line) {
564 final words = line.split(' ');
565 final command = words[0];
566 if (!commands.contains(command)) throw CommandError('UNKNOWN COMMAND $command');
567 switch (command) {
568 case 'CREATE':
569 final mode = parseMode(words[1]);
570 final capacity = int.parse(words[2]);
571 checkCapacity(capacity);
572 final replacement = HashTable(mode, prime(capacity));
573 table = replacement;
574 return ['CREATE ${mode.label} cap=${replacement.capacity}'];
575 case 'MAXLOAD':
576 case 'MINLOAD':
577 final value = int.parse(words[1]);
578 final maximum = command == 'MAXLOAD';
579 if (maximum ? value < 10 || value > 95 : value < 0 || value > 40) throw const CommandError('BAD LOAD');
580 if (maximum) { maxLoad = value; } else { minLoad = value; }
581 return ['$command $value'];
582 case 'TRACE':
583 if (words[1] != 'ON' && words[1] != 'OFF') throw const CommandError('BAD ARGUMENT');
584 tracing = words[1] == 'ON';
585 return ['TRACE ${words[1]}'];
586 case 'HASH':
587 final key = words[1];
588 final capacity = int.parse(words[2]);
589 checkCapacity(capacity);
590 return ['HASH $key m=$capacity h31=${hash(key, 31, capacity)} h37=${hash(key, 37, capacity)} '
591 'step=${1 + hash(key, 37, capacity - 1)}'];
592 case 'BENCH':
593 final capacity = int.parse(words[1]);
594 checkCapacity(capacity);
595 final count = int.parse(words[2]), length = int.parse(words[3]), seed = int.parse(words[4]);
596 checkGeneration(count, length, seed);
597 return bench(prime(capacity), count, length, seed);
598 }
599 final current = table;
600 if (current == null) throw const CommandError('NO TABLE');
601 switch (command) {
602 case 'PUT':
603 case 'ADD': return current.put(words[1], int.parse(words[2]), maxLoad, command: command, trace: tracing).lines;
604 case 'GET': return current.get(words[1], trace: tracing);
605 case 'DEL': return current.delete(words[1], minLoad, maxLoad, trace: tracing).lines;
606 case 'PROBE':
607 final found = current.search(words[1], trace: true);
608 return [['PROBE ${words[1]}', ...found.path, '${found.item == null ? 'MISS' : 'FOUND'} probes=${found.probes}'].join(' ')];
609 case 'GEN':
610 case 'GENDEL':
611 final count = int.parse(words[1]), length = int.parse(words[2]), seed = int.parse(words[3]);
612 checkGeneration(count, length, seed);
613 final rows = <String>[];
614 var changed = 0, index = 0;
615 for (final key in generate(count, length, seed)) {
616 index++;
617 final result = command == 'GEN' ? current.put(key, index, maxLoad) : current.delete(key, minLoad, maxLoad);
618 if (result.changed) changed++;
619 rows.addAll(result.rehashes);
620 }
621 rows.add(command == 'GEN' ? 'GEN $count new=$changed updated=${count - changed}'
622 : 'GENDEL $count deleted=$changed missing=${count - changed}');
623 return rows;
624 case 'RESIZE':
625 case 'CONVERT':
626 var mode = current.mode;
627 var capacity = current.capacity;
628 if (command == 'RESIZE') { capacity = int.parse(words[1]); checkCapacity(capacity); capacity = prime(capacity); }
629 else { mode = parseMode(words[1]); }
630 if (100 * current.size > maxLoad * capacity * mode.parts) throw const CommandError('TOO SMALL');
631 final replacement = HashTable.rebuild(current.items(), mode, capacity);
632 if (replacement == null) throw const CommandError('REBUILD FAILED');
633 final row = command == 'RESIZE' ? 'RESIZE ${current.capacity} -> $capacity'
634 : 'CONVERT ${current.mode.label} -> ${mode.label} cap=$capacity';
635 current.adopt(replacement);
636 return [row];
637 case 'CLEAR':
638 final size = current.size;
639 current.adopt(HashTable(current.mode, current.capacity));
640 return ['CLEAR $size'];
641 case 'DUMP': return current.dump();
642 case 'MAP': return current.map();
643 case 'STATS': return current.statistics();
644 case 'KEYS':
645 final keys = current.items().map((item) => item.key).toList()..sort();
646 return [['KEYS ${current.size}', ...keys].join(' ')];
647 case 'TOP':
648 final limit = int.parse(words[1]);
649 if (limit < 1 || limit > 100) throw const CommandError('BAD ARGUMENT');
650 final records = current.costs()..sort((a, b) {
651 final order = b.probes.compareTo(a.probes);
652 return order != 0 ? order : a.key.compareTo(b.key);
653 });
654 final selected = records.take(limit).toList();
655 return ['TOP ${selected.length}', for (final cost in selected)
656 '${cost.key} cost=${cost.probes} at=${current.position(cost.position)}'];
657 case 'SCAN':
658 final cursor = int.parse(words[1]), limit = int.parse(words[2]);
659 if (limit < 1 || limit > 100) throw const CommandError('BAD ARGUMENT');
660 if (cursor >= current.slotCount) throw const CommandError('BAD CURSOR');
661 return current.scan(cursor, limit);
662 case 'CLUSTERS': return current.clusters();
663 case 'DIST': return current.distribution();
664 case 'COMPARE':
665 final capacity = int.parse(words[1]);
666 checkCapacity(capacity);
667 return compare(prime(capacity));
668 default: throw StateError('Unhandled command $command');
669 }
670 }
671}
