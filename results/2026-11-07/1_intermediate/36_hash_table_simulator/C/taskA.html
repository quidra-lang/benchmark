49static void append(String *text, const char *format, ...) {
50 va_list args, copy;
51 va_start(args, format);
52 va_copy(copy, args);
53 int added = vsnprintf(NULL, 0, format, copy);
54 va_end(copy);
55 size_t needed = text->length + (size_t)added + 1;
56 if (needed > text->capacity) {
57 size_t capacity = text->capacity ? text->capacity : 64;
58 while (capacity < needed) capacity *= 2;
59 char *memory = realloc(text->data, capacity);
60 if (!memory) exit(EXIT_FAILURE);
61 text->data = memory;
62 text->capacity = capacity;
63 }
64 vsnprintf(text->data + text->length, text->capacity - text->length, format, args);
65 text->length += (size_t)added;
66 va_end(args);
67}

75static int hash(const char *key, int base, int modulus) {
76 int value = 0;
77 for (const unsigned char *p = (const unsigned char *)key; *p; p++) value = (value * base + *p) % modulus;
78 return value;
79}
80static bool prime(int value) {
81 if (value < 2) return false;
82 for (int divisor = 2; divisor <= value / divisor; divisor++) if (value % divisor == 0) return false;
83 return true;
84}
85static int next_prime(int value) {
86 if (value < 2) value = 2;
87 while (!prime(value)) value++;
88 return value;
89}
90static int slot_count(const Table *table) { return table->cap * (table->mode == CUCKOO ? 2 : 1); }
91static int distance(const Table *table, const Item *item, int position) {
92 return (position - hash(item->key, 31, table->cap) + table->cap) % table->cap;
93}
94static Text position_text(const Table *table, int position) {
95 Text text;
96 if (table->mode == CUCKOO) snprintf(text.data, sizeof(text.data), "T%d.%d", position / table->cap + 1, position % table->cap);
97 else snprintf(text.data, sizeof(text.data), "%d", position);
98 return text;
99}
100static Text decimal(int64_t numerator, int64_t denominator) {
101 int64_t value = denominator ? 100 * numerator / denominator : 0;
102 Text text;
103 snprintf(text.data, sizeof(text.data), "%" PRId64 ".%02" PRId64, value / 100, value % 100);
104 return text;
105}

107static Table table_new(Mode mode, int cap) {
108 Table table = {.mode = mode, .cap = cap};
109 if (mode == CHAIN) {
110 table.heads = allocate((size_t)cap, sizeof(Node *));
111 table.tails = allocate((size_t)cap, sizeof(Node *));
112 } else table.slots = allocate((size_t)slot_count(&table), sizeof(Slot));
113 return table;
114}
115static void table_destroy(Table *table) {
116 if (table->mode == CHAIN) {
117 for (int i = 0; i < table->cap; i++) {
118 Node *node = table->heads[i];
119 while (node) { Node *next = node->next; free(node); node = next; }
120 }
121 free(table->heads);
122 free(table->tails);
123 } else free(table->slots);
124}
125static void replace_table(Table *table, Table replacement) {
126 Counters counters = table->counters;
127 table_destroy(table);
128 *table = replacement;
129 table->counters = counters;
130}

132static void observe(Search *search, const Table *table, int position, const char *content, bool record_path) {
133 search->probes++;
134 if (record_path) append(&search->path, "%s%s:%s", search->path.length ? " " : "", position_text(table, position).data, content);
135}
136static Search search_key(const Table *table, const char *key, bool record_path) {
137 Search search = {.position = -1, .first_tomb = -1, .empty = -1};
138 int home = hash(key, 31, table->cap);
139 if (table->mode == CHAIN) {
140 search.position = home;
141 for (Node *node = table->heads[home]; node; node = node->next) {
142 observe(&search, table, home, node->item.key, record_path);
143 if (strcmp(node->item.key, key) == 0) { search.found = true; search.node = node; break; }
144 search.previous = node;
145 }
146 return search;
147 }
148 int step = table->mode == DOUBLE ? 1 + hash(key, 37, table->cap - 1) : 1;
149 int limit = table->mode == CUCKOO ? 2 : table->cap;
150 for (int i = 0; i < limit; i++) {
151 int position;
152 if (table->mode == CUCKOO) position = i == 0 ? home : table->cap + hash(key, 37, table->cap);
153 else position = (home + (table->mode == QUADRATIC ? i * i : i * step)) % table->cap;
154 const Slot *slot = &table->slots[position];
155 const char *content = slot->state == LIVE ? slot->item.key : slot->state == TOMB ? "D" : "-";
156 observe(&search, table, position, content, record_path);
157 if (slot->state == LIVE && strcmp(slot->item.key, key) == 0) {
158 search.found = true;
159 search.position = position;
160 break;
161 }
162 if (table->mode == CUCKOO) continue;
163 if (slot->state == EMPTY) { search.empty = position; break; }
164 if (slot->state == TOMB && search.first_tomb < 0) search.first_tomb = position;
165 if (table->mode == ROBINHOOD && distance(table, &slot->item, position) < i) break;
166 }
167 return search;
168}
169static Item *found_item(Table *table, const Search *search) {
170 return table->mode == CHAIN ? &search->node->item : &table->slots[search->position].item;
171}

173static Placement place(Table *table, Item item, const Search *initial, String *events) {
174 Placement result = {.position = -1};
175 int home = hash(item.key, 31, table->cap);
176 if (table->mode == CHAIN) {
177 Node *node = allocate(1, sizeof(Node));
178 node->item = item;
179 if (table->tails[home]) table->tails[home]->next = node;
180 else table->heads[home] = node;
181 table->tails[home] = node;
182 result.success = true;
183 result.position = home;
184 } else if (table->mode == ROBINHOOD) {
185 Item carried = item;
186 int position = home, traveled = 0, from = -1;
187 for (;;) {
188 Slot *slot = &table->slots[position];
189 if (slot->state == EMPTY) {
190 if (from >= 0 && events) append(events, "SHIFT %s %d -> %d\n", carried.key, from, position);
191 slot->state = LIVE;
192 slot->item = carried;
193 if (result.position < 0) result.position = position;
194 result.success = true;
195 break;
196 }
197 int resident_distance = distance(table, &slot->item, position);
198 if (resident_distance < traveled) {
199 if (from >= 0 && events) append(events, "SHIFT %s %d -> %d\n", carried.key, from, position);
200 Item displaced = slot->item;
201 slot->item = carried;
202 carried = displaced;
203 if (result.position < 0) result.position = position;
204 from = position;
205 traveled = resident_distance;
206 result.movements++;
207 }
208 position = (position + 1) % table->cap;
209 traveled++;
210 }
211 } else if (table->mode == CUCKOO) {
212 Item carried = item;
213 int part = 0;
214 for (;;) {
215 int position = part * table->cap + hash(carried.key, part == 0 ? 31 : 37, table->cap);
216 Slot *slot = &table->slots[position];
217 if (slot->state == EMPTY) {
218 slot->state = LIVE;
219 slot->item = carried;
220 result.success = true;
221 result.position = search_key(table, item.key, false).position;
222 break;
223 }
224 if (result.movements == table->cap) { result.homeless = carried; break; }
225 Item displaced = slot->item;
226 if (events) append(events, "KICK %s %s\n", displaced.key, position_text(table, position).data);
227 slot->item = carried;
228 carried = displaced;
229 result.movements++;
230 part = 1 - part;
231 }
232 } else {
233 Search own = {0};
234 if (!initial) { own = search_key(table, item.key, false); initial = &own; }
235 int position = initial->first_tomb >= 0 ? initial->first_tomb : initial->empty;
236 if (position >= 0) {
237 result.reused = table->slots[position].state == TOMB;
238 if (result.reused) table->tomb--;
239 table->slots[position] = (Slot){.state = LIVE, .item = item};
240 result.position = position;
241 result.success = true;
242 } else result.homeless = item;
243 }
244 return result;
245}

247static int delete_item(Table *table, const Search *search, String *events) {
248 int shifted = 0, position = search->position;
249 if (table->mode == CHAIN) {
250 Node *node = search->node;
251 if (search->previous) search->previous->next = node->next;
252 else table->heads[position] = node->next;
253 if (table->tails[position] == node) table->tails[position] = search->previous;
254 free(node);
255 } else if (table->mode == LINEAR || table->mode == QUADRATIC || table->mode == DOUBLE) {
256 table->slots[position].state = TOMB;
257 table->tomb++;
258 } else {
259 table->slots[position].state = EMPTY;
260 if (table->mode == ROBINHOOD) {
261 int source = (position + 1) % table->cap;
262 while (table->slots[source].state == LIVE && distance(table, &table->slots[source].item, source) > 0) {
263 int destination = (source + table->cap - 1) % table->cap;
264 if (events) append(events, "SHIFT %s %d -> %d\n", table->slots[source].item.key, source, destination);
265 table->slots[destination] = table->slots[source];
266 table->slots[source].state = EMPTY;
267 shifted++;
268 source = (source + 1) % table->cap;
269 }
270 }
271 }
272 table->size--;
273 return shifted;
274}

276static Item *old_order(const Table *table, const Item *pending, int *count) {
277 Item *items = allocate((size_t)table->size + (pending != NULL), sizeof(Item));
278 int length = 0;
279 if (table->mode == CHAIN) {
280 for (int i = 0; i < table->cap; i++) for (Node *node = table->heads[i]; node; node = node->next) items[length++] = node->item;
281 } else {
282 for (int i = 0; i < slot_count(table); i++) if (table->slots[i].state == LIVE) items[length++] = table->slots[i].item;
283 }
284 if (pending) items[length++] = *pending;
285 *count = length;
286 return items;
287}
288static bool rebuild(Table *table, Mode mode, int cap, const Item *items, int count) {
289 Table trial = table_new(mode, cap);
290 for (int i = 0; i < count; i++) {
291 Placement result = place(&trial, items[i], NULL, NULL);
292 if (!result.success) { table_destroy(&trial); return false; }
293 trial.size++;
294 }
295 replace_table(table, trial);
296 return true;
297}
298static void automatic_rebuild(Table *table, int cap, const char *reason, const Item *pending, FILE *output) {
299 int count, old_cap = table->cap;
300 Item *items = old_order(table, pending, &count);
301 while (!rebuild(table, table->mode, cap, items, count)) cap = next_prime(2 * cap);
302 table->counters.rehashes++;
303 if (output) fprintf(output, "REHASH %s %d -> %d\n", reason, old_cap, cap);
304 free(items);
305}
306static void load_check(Table *table, const Settings *settings, FILE *output) {
307 int64_t limit = (int64_t)settings->maxload * slot_count(table);
308 if ((int64_t)(table->size + table->tomb) * 100 <= limit) return;
309 bool grow = (int64_t)table->size * 200 > limit;
310 automatic_rebuild(table, grow ? next_prime(2 * table->cap) : table->cap, grow ? "load" : "tomb", NULL, output);
311}
312static void shrink_check(Table *table, const Settings *settings, FILE *output) {
313 int cap = next_prime(table->cap / 2);
314 int slots = cap * (table->mode == CUCKOO ? 2 : 1);
315 if ((int64_t)table->size * 100 >= (int64_t)settings->minload * slot_count(table) || cap >= table->cap ||
316 (int64_t)table->size * 100 > (int64_t)settings->maxload * slots) return;
317 int count, old_cap = table->cap;
318 Item *items = old_order(table, NULL, &count);
319 if (rebuild(table, table->mode, cap, items, count)) {
320 table->counters.rehashes++;
321 if (output) fprintf(output, "REHASH shrink %d -> %d\n", old_cap, cap);
322 }
323 free(items);
324}

330static bool put_item(Table *table, Item item, bool add_value, const Settings *settings, FILE *output, bool details) {
331 bool trace = details && settings->trace;
332 Search search = search_key(table, item.key, trace);
333 table->counters.probes += search.probes;
334 const char *command = add_value ? "ADD" : "PUT";
335 String events = {0};
336 if (search.found) {
337 Item *stored = found_item(table, &search);
338 int64_t old = stored->value;
339 stored->value = add_value ? old + item.value : item.value;
340 table->counters.updates++;
341 if (details) fprintf(output, "%s %s=%" PRId64 " UPDATED old=%" PRId64 " at=%s probes=%d\n",
342 command, item.key, stored->value, old, position_text(table, search.position).data, search.probes);
343 if (trace) trace_lines(output, &search, &events);
344 free(search.path.data);
345 return false;
346 }
347 Placement result = place(table, item, &search, trace ? &events : NULL);
348 table->counters.inserts++;
349 if (details) {
350 fprintf(output, "%s %s=%" PRId64 " NEW", command, item.key, item.value);
351 if (result.success) {
352 fprintf(output, " at=%s probes=%d", position_text(table, result.position).data, search.probes);
353 if (result.reused) fputs(" reused", output);
354 if (table->mode == ROBINHOOD) fprintf(output, " shifted=%d", result.movements);
355 if (table->mode == CUCKOO) fprintf(output, " kicks=%d", result.movements);
356 } else {
357 fprintf(output, " probes=%d", search.probes);
358 if (table->mode == QUADRATIC) fputs(" full", output);
359 else fprintf(output, " kicks=%d failed", result.movements);
360 }
361 fputc('\n', output);
362 }
363 if (trace) trace_lines(output, &search, &events);
364 free(search.path.data);
365 free(events.data);
366 if (result.success) table->size++;
367 else automatic_rebuild(table, next_prime(2 * table->cap), table->mode == QUADRATIC ? "full" : "kicks", &result.homeless, output);
368 load_check(table, settings, output);
369 return true;
370}
371static bool remove_key(Table *table, const char *key, const Settings *settings, FILE *output, bool details) {
372 bool trace = details && settings->trace;
373 Search search = search_key(table, key, trace);
374 String events = {0};
375 table->counters.probes += search.probes;
376 if (search.found) {
377 int shifted = delete_item(table, &search, trace ? &events : NULL);
378 table->counters.deletes++;
379 if (details) {
380 fprintf(output, "DEL %s OK at=%s probes=%d", key, position_text(table, search.position).data, search.probes);
381 if (table->mode == ROBINHOOD) fprintf(output, " shifted=%d", shifted);
382 fputc('\n', output);
383 }
384 } else {
385 table->counters.absent++;
386 if (details) fprintf(output, "DEL %s NOT FOUND probes=%d\n", key, search.probes);
387 }
388 if (trace) trace_lines(output, &search, &events);
389 free(search.path.data);
390 free(events.data);
391 if (search.found) shrink_check(table, settings, output);
392 return search.found;
393}
394static void get_key(Table *table, const char *key, const Settings *settings) {
395 Search search = search_key(table, key, settings->trace);
396 table->counters.probes += search.probes;
397 if (search.found) {
398 table->counters.hits++;
399 printf("GET %s=%" PRId64 " at=%s probes=%d\n", key, found_item(table, &search)->value,
400 position_text(table, search.position).data, search.probes);
401 } else {
402 table->counters.misses++;
403 printf("GET %s NOT FOUND probes=%d\n", key, search.probes);
404 }
405 if (settings->trace) trace_lines(stdout, &search, &(String){0});
406 free(search.path.data);
407}
408static int item_cost(const Table *table, const Item *item, int position) {
409 if (table->mode == CUCKOO) return position / table->cap + 1;
410 if (table->mode == LINEAR || table->mode == ROBINHOOD) return distance(table, item, position) + 1;
411 return search_key(table, item->key, false).probes;
412}
413static Record *records(const Table *table) {
414 Record *result = allocate((size_t)table->size, sizeof(Record));
415 int count = 0;
416 if (table->mode == CHAIN) {
417 for (int i = 0; i < table->cap; i++) {
418 int cost = 0;
419 for (Node *node = table->heads[i]; node; node = node->next) {
420 result[count++] = (Record){.item = node->item, .position = i, .cost = ++cost};
421 }
422 }
423 } else {
424 for (int i = 0; i < slot_count(table); i++) if (table->slots[i].state == LIVE) {
425 const Item *item = &table->slots[i].item;
426 result[count++] = (Record){.item = *item, .position = i, .cost = item_cost(table, item, i)};
427 }
428 }
429 return result;
430}
431static Costs costs(const Table *table) {
432 Record *items = records(table);
433 Costs result = {0};
434 for (int i = 0; i < table->size; i++) {
435 result.sum += items[i].cost;
436 if (items[i].cost > result.longest) result.longest = items[i].cost;
437 }
438 free(items);
439 return result;
440}
441static int compare_items(const void *left, const void *right) {
442 const Item *a = left, *b = right;
443 return strcmp(a->key, b->key);
444}
445static int compare_costs(const void *left, const void *right) {
446 const Record *a = left, *b = right;
447 if (a->cost != b->cost) return a->cost > b->cost ? -1 : 1;
448 return strcmp(a->item.key, b->item.key);
449}

451static void dump(const Table *table) {
452 printf("DUMP %s cap=%d size=%d tomb=%d\n", MODE_NAMES[table->mode], table->cap, table->size, table->tomb);
453 if (table->mode == CHAIN) {
454 for (int i = 0; i < table->cap; i++) {
455 printf("%d", i);
456 if (!table->heads[i]) fputs(" -", stdout);
457 for (Node *node = table->heads[i]; node; node = node->next) printf(" %s=%" PRId64, node->item.key, node->item.value);
458 putchar('\n');
459 }
460 } else {
461 for (int i = 0; i < slot_count(table); i++) {
462 if (table->mode == CUCKOO) printf("T%d %d ", i / table->cap + 1, i % table->cap);
463 else printf("%d ", i);
464 const Slot *slot = &table->slots[i];
465 if (slot->state == EMPTY) putchar('-');
466 else if (slot->state == TOMB) putchar('D');
467 else {
468 printf("%s=%" PRId64, slot->item.key, slot->item.value);
469 if (table->mode == ROBINHOOD) printf(" d=%d", distance(table, &slot->item, i));
470 }
471 putchar('\n');
472 }
473 }

475static char gauge(int value) { return value < 10 ? (char)('0' + value) : '+'; }
476static void map_table(const Table *table) {
477 printf("MAP %s cap=%d\n", MODE_NAMES[table->mode], table->cap);
478 int parts = table->mode == CUCKOO ? 2 : 1;
479 for (int part = 0; part < parts; part++) {
480 for (int start = 0; start < table->cap; start += 40) {
481 char text[41];
482 int length = table->cap - start < 40 ? table->cap - start : 40;
483 for (int j = 0; j < length; j++) {
484 int i = start + j;
485 if (table->mode == CHAIN) {
486 int count = 0;
487 for (Node *node = table->heads[i]; node; node = node->next) count++;
488 text[j] = count ? gauge(count) : '.';
489 } else {
490 int position = part * table->cap + i;
491 const Slot *slot = &table->slots[position];
492 text[j] = slot->state == EMPTY ? '.' : slot->state == TOMB ? 'D' : table->mode == CUCKOO ? '#'
493 : gauge(item_cost(table, &slot->item, position) - 1);
494 }
495 }
496 text[length] = '\0';
497 if (table->mode == CUCKOO) printf("T%d ", part + 1);
498 printf("%05d %s\n", start, text);
499 }
500 }
501}
502static void statistics(const Table *table) {
503 Costs result = costs(table);
504 const Counters *c = &table->counters;
505 printf("STATS %s size=%d cap=%d slots=%d tomb=%d load=%s%% longest=%d avg=%s probes=%" PRId64 " rehashes=%" PRId64 "\n",
506 MODE_NAMES[table->mode], table->size, table->cap, slot_count(table), table->tomb,
507 decimal((int64_t)table->size * 100, slot_count(table)).data, result.longest, decimal(result.sum, table->size).data, c->probes, c->rehashes);
508 printf("OPS inserts=%" PRId64 " updates=%" PRId64 " hits=%" PRId64 " misses=%" PRId64 " deletes=%" PRId64 " absent=%" PRId64 "\n",
509 c->inserts, c->updates, c->hits, c->misses, c->deletes, c->absent);
510}
511static void histogram(const int *values, int count) {
512 int maximum = 0;
513 for (int i = 0; i < count; i++) if (values[i] > maximum) maximum = values[i];
514 int *frequencies = allocate((size_t)maximum + 1, sizeof(int));
515 for (int i = 0; i < count; i++) frequencies[values[i]]++;
516 for (int value = 0; value <= maximum; value++) if (frequencies[value]) {
517 int frequency = frequencies[value];
518 printf("%d %d ", value, frequency);
519 for (int j = 0; j < frequency && j < 50; j++) putchar('#');
520 if (frequency > 50) putchar('+');
521 putchar('\n');
522 }
523 free(frequencies);
524}
525static void clusters(const Table *table) {
526 int *runs = allocate((size_t)slot_count(table), sizeof(int));
527 int count = 0, longest = 0;
528 if (table->mode == CHAIN) {
529 for (int i = 0; i < table->cap; i++) {
530 int length = 0;
531 for (Node *node = table->heads[i]; node; node = node->next) length++;
532 if (length) runs[count++] = length;
533 }
534 } else {
535 int parts = table->mode == CUCKOO ? 2 : 1;
536 for (int part = 0; part < parts; part++) {
537 int base = part * table->cap, empty = -1;
538 for (int i = 0; i < table->cap; i++) if (table->slots[base + i].state == EMPTY) { empty = i; break; }
539 if (empty < 0) runs[count++] = table->cap;
540 else {
541 int length = 0;
542 for (int offset = 1; offset <= table->cap; offset++) {
543 int i = (empty + offset) % table->cap;
544 if (table->slots[base + i].state != EMPTY) length++;
545 else if (length) { runs[count++] = length; length = 0; }
546 }
547 }
548 }
549 }
550 for (int i = 0; i < count; i++) if (runs[i] > longest) longest = runs[i];
551 printf("CLUSTERS runs=%d longest=%d\n", count, longest);
552 histogram(runs, count);
553 free(runs);
554}
555static void distribution(const Table *table) {
556 Record *items = records(table);
557 int *values = allocate((size_t)table->size, sizeof(int));
558 int longest = 0;
559 for (int i = 0; i < table->size; i++) {
560 values[i] = items[i].cost;
561 if (values[i] > longest) longest = values[i];
562 }
563 printf("DIST keys=%d longest=%d\n", table->size, longest);
564 histogram(values, table->size);
565 free(values);
566 free(items);
567}
568static void scan(const Table *table, int cursor, int limit) {
569 String keys = {0};
570 int count = 0, end = cursor, slots = slot_count(table);
571 while (end < slots && count < limit) {
572 if (table->mode == CHAIN) {
573 for (Node *node = table->heads[end]; node; node = node->next) { append(&keys, " %s", node->item.key); count++; }
574 } else if (table->slots[end].state == LIVE) { append(&keys, " %s", table->slots[end].item.key); count++; }
575 end++;
576 }
577 printf("SCAN next=%d%s\n", end == slots ? 0 : end, string_data(&keys));
578 free(keys.data);
579}

581typedef char Key[11];
582static Key *generate_keys(int count, int length, int64_t seed) {
583 Key *keys = allocate((size_t)count, sizeof(Key));
584 int64_t state = seed;
585 for (int i = 0; i < count; i++) {
586 for (int j = 0; j < length; j++) {
587 state = 48271 * state % 2147483647;
588 keys[i][j] = (char)('a' + state % 26);
589 }
590 keys[i][length] = '\0';
591 }
592 return keys;
593}
594static void compare_modes(const Table *table, int cap, const Settings *settings) {
595 int count;
596 Item *items = old_order(table, NULL, &count);
597 qsort(items, (size_t)count, sizeof(Item), compare_items);
598 printf("COMPARE keys=%d cap=%d\n", count, cap);
599 for (Mode mode = CHAIN; mode <= CUCKOO; mode++) {
600 Table trial = table_new(mode, cap);
601 for (int i = 0; i < count; i++) put_item(&trial, items[i], false, settings, NULL, false);
602 Costs result = costs(&trial);
603 printf("%s cap=%d longest=%d avg=%s probes=%" PRId64 " rehashes=%" PRId64 "\n", MODE_NAMES[mode], trial.cap,
604 result.longest, decimal(result.sum, trial.size).data, trial.counters.probes, trial.counters.rehashes);
605 table_destroy(&trial);
606 }
607 free(items);
608}
609static void benchmark(int cap, int count, int length, int64_t seed, const Settings *settings) {
610 Key *keys = generate_keys(2 * count, length, seed);
611 printf("BENCH cap=%d count=%d len=%d seed=%" PRId64 "\n", cap, count, length, seed);
612 for (Mode mode = CHAIN; mode <= CUCKOO; mode++) {
613 Table trial = table_new(mode, cap);
614 for (int i = 0; i < count; i++) put_item(&trial, item_make(keys[i], i + 1), false, settings, NULL, false);
615 int misses = 0;
616 int64_t miss_probes = 0;
617 for (int i = count; i < 2 * count; i++) {
618 Search search = search_key(&trial, keys[i], false);
619 if (!search.found) { misses++; miss_probes += search.probes; }
620 }
621 Costs result = costs(&trial);
622 printf("%s cap=%d keys=%d hit=%s misses=%d miss=%s rehashes=%" PRId64 "\n", MODE_NAMES[mode], trial.cap, trial.size,
623 decimal(result.sum, trial.size).data, misses, decimal(miss_probes, misses).data, trial.counters.rehashes);
624 table_destroy(&trial);
625 }
626 free(keys);
627}

629typedef enum {
630 CREATE, MAXLOAD, MINLOAD, TRACE, HASH, PUT, ADD, GET, DEL, PROBE, GEN, GENDEL,
631 RESIZE, CONVERT, CLEAR, DUMP, MAP, STATS, KEYS, TOP, SCAN, CLUSTERS, DIST, COMPARE, BENCH, UNKNOWN
632} Command;
633static const char *const COMMAND_NAMES[] = {
634 "CREATE", "MAXLOAD", "MINLOAD", "TRACE", "HASH", "PUT", "ADD", "GET", "DEL", "PROBE", "GEN", "GENDEL",
635 "RESIZE", "CONVERT", "CLEAR", "DUMP", "MAP", "STATS", "KEYS", "TOP", "SCAN", "CLUSTERS", "DIST", "COMPARE", "BENCH"
636};
637static Command command_parse(const char *word) {
638 for (int i = CREATE; i < UNKNOWN; i++) if (strcmp(word, COMMAND_NAMES[i]) == 0) return (Command)i;
639 return UNKNOWN;
640}
641static bool mode_parse(const char *word, Mode *mode) {
642 for (int i = CHAIN; i <= CUCKOO; i++) if (strcmp(word, MODE_NAMES[i]) == 0) { *mode = (Mode)i; return true; }
643 return false;
644}
645static int64_t number(const char *word) { return strtoll(word, NULL, 10); }
646static bool valid_capacity(int64_t cap) { return cap >= 2 && cap <= 10000; }
647static bool valid_generation(int64_t count, int64_t length, int64_t seed) {
648 return count >= 1 && count <= 500 && length >= 1 && length <= 10 && seed >= 1 && seed <= 2147483646;
649}
650
651static void execute(Table **current, Settings *settings, char **words) {
652 Command command = command_parse(words[0]);
653 if (command == UNKNOWN) { printf("ERROR UNKNOWN COMMAND %s\n", words[0]); return; }
654 if (command == CREATE) {
655 Mode mode;
656 if (!mode_parse(words[1], &mode)) { printf("ERROR BAD MODE %s\n", words[1]); return; }
657 int64_t capacity = number(words[2]);
658 if (!valid_capacity(capacity)) { puts("ERROR BAD CAPACITY"); return; }
659 if (*current) table_destroy(*current);
660 else *current = allocate(1, sizeof(Table));
661 **current = table_new(mode, next_prime((int)capacity));
662 printf("CREATE %s cap=%d\n", MODE_NAMES[mode], (*current)->cap);
663 return;
664 }
665 if (command == MAXLOAD || command == MINLOAD) {
666 int64_t value = number(words[1]);
667 if (value < (command == MAXLOAD ? 10 : 0) || value > (command == MAXLOAD ? 95 : 40)) { puts("ERROR BAD LOAD"); return; }
668 if (command == MAXLOAD) settings->maxload = (int)value;
669 else settings->minload = (int)value;
670 printf("%s %" PRId64 "\n", words[0], value);
671 return;
672 }
673 if (command == TRACE) {
674 if (strcmp(words[1], "ON") != 0 && strcmp(words[1], "OFF") != 0) { puts("ERROR BAD ARGUMENT"); return; }
675 settings->trace = strcmp(words[1], "ON") == 0;
676 printf("TRACE %s\n", words[1]);
677 return;
678 }
679 if (command == HASH) {
680 int64_t capacity = number(words[2]);
681 if (!valid_capacity(capacity)) { puts("ERROR BAD CAPACITY"); return; }
682 int m = (int)capacity;
683 printf("HASH %s m=%d h31=%d h37=%d step=%d\n", words[1], m, hash(words[1], 31, m), hash(words[1], 37, m), 1 + hash(words[1], 37, m - 1));
684 return;
685 }
686 if (command == BENCH) {
687 int64_t capacity = number(words[1]);
688 if (!valid_capacity(capacity)) { puts("ERROR BAD CAPACITY"); return; }
689 int64_t count = number(words[2]), length = number(words[3]), seed = number(words[4]);
690 if (!valid_generation(count, length, seed)) { puts("ERROR BAD ARGUMENT"); return; }
691 benchmark(next_prime((int)capacity), (int)count, (int)length, seed, settings);
692 return;
693 }
694 Table *table = *current;
695 if (!table) { puts("ERROR NO TABLE"); return; }
696 switch (command) {
697 case PUT: case ADD:
698 put_item(table, item_make(words[1], number(words[2])), command == ADD, settings, stdout, true);
699 break;
700 case GET:
701 get_key(table, words[1], settings);
702 break;
703 case DEL:
704 remove_key(table, words[1], settings, stdout, true);
705 break;
706 case PROBE: {
707 Search search = search_key(table, words[1], true);
708 printf("PROBE %s", words[1]);
709 if (search.path.length) printf(" %s", search.path.data);
710 printf(" %s probes=%d\n", search.found ? "FOUND" : "MISS", search.probes);
711 free(search.path.data);
712 break;
713 }
714 case GEN: case GENDEL: {
715 int64_t count = number(words[1]), length = number(words[2]), seed = number(words[3]);
716 if (!valid_generation(count, length, seed)) { puts("ERROR BAD ARGUMENT"); return; }
717 Key *keys = generate_keys((int)count, (int)length, seed);
718 int changed = 0;
719 for (int i = 0; i < count; i++) {
720 if (command == GEN) changed += put_item(table, item_make(keys[i], i + 1), false, settings, stdout, false);
721 else changed += remove_key(table, keys[i], settings, stdout, false);
722 }
723 if (command == GEN) printf("GEN %" PRId64 " new=%d updated=%" PRId64 "\n", count, changed, count - changed);
724 else printf("GENDEL %" PRId64 " deleted=%d missing=%" PRId64 "\n", count, changed, count - changed);
725 free(keys);
726 break;
727 }
728 case RESIZE: case CONVERT: {
729 int cap = table->cap, old_cap = table->cap;
730 Mode mode = table->mode, old_mode = table->mode;
731 if (command == RESIZE) {
732 int64_t capacity = number(words[1]);
733 if (!valid_capacity(capacity)) { puts("ERROR BAD CAPACITY"); return; }
734 cap = next_prime((int)capacity);
735 } else if (!mode_parse(words[1], &mode)) { printf("ERROR BAD MODE %s\n", words[1]); return; }
736 int slots = cap * (mode == CUCKOO ? 2 : 1);
737 if ((int64_t)table->size * 100 > (int64_t)settings->maxload * slots) { puts("ERROR TOO SMALL"); return; }
738 int count;
739 Item *items = old_order(table, NULL, &count);
740 bool success = rebuild(table, mode, cap, items, count);
741 free(items);
742 if (!success) { puts("ERROR REBUILD FAILED"); return; }
743 if (command == RESIZE) printf("RESIZE %d -> %d\n", old_cap, cap);
744 else printf("CONVERT %s -> %s cap=%d\n", MODE_NAMES[old_mode], MODE_NAMES[mode], cap);
745 break;
746 }
747 case CLEAR:
748 printf("CLEAR %d\n", table->size);
749 replace_table(table, table_new(table->mode, table->cap));
750 break;
751 case DUMP:
752 dump(table);
753 break;
754 case MAP:
755 map_table(table);
756 break;
757 case STATS:
758 statistics(table);
759 break;
760 case KEYS: {
761 int count;
762 Item *items = old_order(table, NULL, &count);
763 qsort(items, (size_t)count, sizeof(Item), compare_items);
764 printf("KEYS %d", count);
765 for (int i = 0; i < count; i++) printf(" %s", items[i].key);
766 putchar('\n');
767 free(items);
768 break;
769 }
770 case TOP: {
771 int64_t limit = number(words[1]);
772 if (limit < 1 || limit > 100) { puts("ERROR BAD ARGUMENT"); return; }
773 Record *items = records(table);
774 qsort(items, (size_t)table->size, sizeof(Record), compare_costs);
775 int count = table->size < limit ? table->size : (int)limit;
776 printf("TOP %d\n", count);
777 for (int i = 0; i < count; i++) printf("%s cost=%d at=%s\n", items[i].item.key, items[i].cost, position_text(table, items[i].position).data);
778 free(items);
779 break;
780 }
781 case SCAN: {
782 int64_t cursor = number(words[1]), limit = number(words[2]);
783 if (limit < 1 || limit > 100) { puts("ERROR BAD ARGUMENT"); return; }
784 if (cursor >= slot_count(table)) { puts("ERROR BAD CURSOR"); return; }
785 scan(table, (int)cursor, (int)limit);
786 break;
787 }
788 case CLUSTERS:
789 clusters(table);
790 break;
791 case DIST:
792 distribution(table);
793 break;
794 case COMPARE: {
795 int64_t capacity = number(words[1]);
796 if (!valid_capacity(capacity)) { puts("ERROR BAD CAPACITY"); return; }
797 compare_modes(table, next_prime((int)capacity), settings);
798 break;
799 }
800 default: break;
801 }
802}

804static char *read_line(void) {
805 size_t length = 0, capacity = 128;
806 char *line = allocate(capacity, 1);
807 int ch;
808 while ((ch = getchar()) != EOF && ch != '\n') {
809 if (length + 1 == capacity) {
810 capacity *= 2;
811 char *memory = realloc(line, capacity);
812 if (!memory) exit(EXIT_FAILURE);
813 line = memory;
814 }
815 line[length++] = (char)ch;
816 }
817 if (ch == EOF && length == 0) { free(line); return NULL; }
818 if (length && line[length - 1] == '\r') length--;
819 line[length] = '\0';
820 return line;
821}

823int main(void) {
824 char *line = read_line();
825 if (!line) return 0;
826 int count = (int)number(line);
827 free(line);
828 Table *table = NULL;
829 Settings settings = {.maxload = 75};
830 for (int i = 0; i < count; i++) {
831 line = read_line();
832 if (!line) break;
833 char *words[5];
834 int size = 0;
835 char *word = strtok(line, " ");
836 while (word && size < 5) { words[size++] = word; word = strtok(NULL, " "); }
837 execute(&table, &settings, words);
838 free(line);
839 }
840 if (table) { table_destroy(table); free(table); }
841 return 0;
842}
