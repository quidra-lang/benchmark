75function hashKey(string $key, int $base, int $modulus): int
76{
77 $value = 0;
78 for ($i = 0; $i < strlen($key); $i++) $value = ($value * $base + ord($key[$i])) % $modulus;

81function prime(int $value): int
82{
83 for ($candidate = max($value, 2); ; $candidate++) {
84 $valid = true;
85 for ($divisor = 2; $divisor * $divisor <= $candidate; $divisor++) {
86 if ($candidate % $divisor === 0) { $valid = false; break; }
87 }
88 if ($valid) return $candidate;
89 }

129final class HashTable
130{
131 private array $buckets, $slots;
132 public int $size = 0, $tombstones = 0;
133 public readonly Counters $counters;
134
135 public function __construct(public Mode $mode, public int $capacity)
136 {
137 $this->buckets = $mode === Mode::CHAIN ? array_fill(0, $capacity, []) : [];
138 $this->slots = $mode === Mode::CHAIN ? [] : array_fill(0, $this->slotCount(), new Slot());
139 $this->counters = new Counters();
140 }
141 public function slotCount(): int { return $this->capacity * $this->mode->parts(); }
142 public function position(int $index): string
143 {
144 return $this->mode === Mode::CUCKOO ? 'T' . (intdiv($index, $this->capacity) + 1) . '.' . ($index % $this->capacity) : (string)$index;
145 }
146 private function distance(string $key, int $index): int
147 {
148 return ($index + $this->capacity - hashKey($key, 31, $this->capacity)) % $this->capacity;
149 }
150 public function items(): array
151 {
152 $result = [];
153 if ($this->mode === Mode::CHAIN) {
154 foreach ($this->buckets as $bucket) foreach ($bucket as $item) $result[] = $item;
155 } else {
156 foreach ($this->slots as $slot) if ($slot->item !== null) $result[] = $slot->item;
157 }
158 return $result;
159 }
160 public function search(string $key, bool $trace = false): SearchResult
161 {
162 $result = new SearchResult();
163 $first = hashKey($key, 31, $this->capacity);
164 $examine = function (int $index, string $content) use ($result, $trace): void {
165 $result->probes++;
166 if ($trace) $result->path[] = $this->position($index) . ':' . $content;
167 };
168 $found = static function (Item $item, int $index, int $chainIndex = -1) use ($result): void {
169 $result->item = $item;
170 $result->position = $index;
171 $result->chainIndex = $chainIndex;
172 };
173 if ($this->mode === Mode::CHAIN) {
174 foreach ($this->buckets[$first] as $index => $item) {
175 $examine($first, $item->key);
176 if ($item->key === $key) { $found($item, $first, $index); break; }
177 }
178 return $result;
179 }
180 if ($this->mode === Mode::CUCKOO) {
181 foreach ([$first, $this->capacity + hashKey($key, 37, $this->capacity)] as $index) {
182 $slot = $this->slots[$index];
183 $examine($index, $slot->content());
184 if ($slot->item !== null && $slot->item->key === $key) { $found($slot->item, $index); break; }
185 }
186 return $result;
187 }
188 $step = $this->mode === Mode::DOUBLE ? 1 + hashKey($key, 37, $this->capacity - 1) : 1;
189 for ($i = 0; $i < $this->capacity; $i++) {
190 $offset = $this->mode === Mode::QUADRATIC ? $i * $i : $i * $step;
191 $index = ($first + $offset) % $this->capacity;
192 $slot = $this->slots[$index];
193 $examine($index, $slot->content());
194 if ($slot->empty()) { $result->available ??= $index; break; }
195 if ($slot->deleted) { $result->available ??= $index; continue; }
196 $item = $slot->item;
197 if ($item->key === $key) { $found($item, $index); break; }
198 if ($this->mode === Mode::ROBINHOOD && $this->distance($item->key, $index) < $i) break;
199 }
200 return $result;
201 }
202 private function place(Item $item, ?SearchResult $initial = null, bool $trace = false): Placement
203 {
204 $result = new Placement();
205 if ($this->mode === Mode::CHAIN) {
206 $result->position = hashKey($item->key, 31, $this->capacity);
207 $this->buckets[$result->position][] = $item;
208 $result->success = true;
209 return $result;
210 }
211 if ($this->mode->usesTombstones()) {
212 $found = $initial ?? $this->search($item->key);
213 $index = $found->available;
214 if ($index === null) { $result->homeless = $item; return $result; }
215 $result->position = $index;
216 $result->reused = $this->slots[$index]->deleted;
217 $this->slots[$index] = new Slot($item);
218 if ($result->reused) $this->tombstones--;
219 $result->success = true;
220 return $result;
221 }
222 $current = $item;
223 if ($this->mode === Mode::ROBINHOOD) {
224 $index = hashKey($item->key, 31, $this->capacity);
225 $distance = 0;
226 $from = null;
227 while (true) {
228 $resident = $this->slots[$index]->item;
229 if ($resident === null) {
230 $this->slots[$index] = new Slot($current);
231 if ($trace && $from !== null) $result->moves[] = "SHIFT $current->key $from -> $index";
232 if ($result->position < 0) $result->position = $index;
233 $result->success = true;
234 return $result;
235 }
236 $residentDistance = $this->distance($resident->key, $index);
237 if ($residentDistance < $distance) {
238 $this->slots[$index] = new Slot($current);
239 if ($trace && $from !== null) $result->moves[] = "SHIFT $current->key $from -> $index";
240 if ($result->position < 0) $result->position = $index;
241 $current = $resident;
242 $from = $index;
243 $distance = $residentDistance;
244 $result->shifted++;
245 }
246 $index = ($index + 1) % $this->capacity;
247 $distance++;
248 }
249 }
250 $part = 0;
251 while (true) {
252 $index = $part * $this->capacity + hashKey($current->key, $part === 0 ? 31 : 37, $this->capacity);
253 $resident = $this->slots[$index]->item;
254 if ($resident === null) {
255 $this->slots[$index] = new Slot($current);
256 $result->success = true;
257 $result->position = $this->search($item->key)->position;
258 return $result;
259 }
260 if ($result->kicks === $this->capacity) { $result->homeless = $current; return $result; }
261 $this->slots[$index] = new Slot($current);
262 $result->kicks++;
263 if ($trace) $result->moves[] = "KICK $resident->key " . $this->position($index);
264 $current = $resident;
265 $part = 1 - $part;
266 }
267 }
268 private function remove(SearchResult $found, bool $trace): Removal
269 {
270 $shifted = 0;
271 $moves = [];
272 if ($this->mode === Mode::CHAIN) {
273 array_splice($this->buckets[$found->position], $found->chainIndex, 1);
274 } elseif ($this->mode->usesTombstones()) {
275 $this->slots[$found->position] = new Slot(null, true);
276 $this->tombstones++;
277 } else {
278 $this->slots[$found->position] = new Slot();
279 if ($this->mode === Mode::ROBINHOOD) {
280 $next = ($found->position + 1) % $this->capacity;
281 while (($item = $this->slots[$next]->item) !== null && $this->distance($item->key, $next) > 0) {
282 $destination = ($next + $this->capacity - 1) % $this->capacity;
283 $this->slots[$destination] = new Slot($item);
284 $this->slots[$next] = new Slot();
285 if ($trace) $moves[] = "SHIFT $item->key $next -> $destination";
286 $shifted++;
287 $next = ($next + 1) % $this->capacity;
288 }
289 }
290 }
291 $this->size--;
292 return new Removal($shifted, $moves);
293 }
294 public static function rebuild(array $items, Mode $mode, int $capacity): ?self
295 {
296 $trial = new self($mode, $capacity);
297 foreach ($items as $item) {
298 if (!$trial->place($item)->success) return null;
299 $trial->size++;
300 }
301 return $trial;
302 }
303 public function adopt(self $replacement): void
304 {
305 $this->mode = $replacement->mode;
306 $this->capacity = $replacement->capacity;
307 $this->size = $replacement->size;

312 private function automatic(string $reason, int $capacity, ?Item $pending = null): string
313 {
314 $items = $this->items();
315 if ($pending !== null) $items[] = $pending;
316 $old = $this->capacity;
317 while (true) {
318 $trial = self::rebuild($items, $this->mode, $capacity);
319 if ($trial !== null) {
320 $this->adopt($trial);
321 $this->counters->rehashes++;
322 return "REHASH $reason $old -> $capacity";
323 }
324 $capacity = prime(2 * $capacity);
325 }
326 }
327 private function loadCheck(int $maxLoad): array
328 {
329 $threshold = $maxLoad * $this->slotCount();
330 if (100 * ($this->size + $this->tombstones) <= $threshold) return [];
331 $grow = 200 * $this->size > $threshold;
332 return [$this->automatic($grow ? 'load' : 'tomb', $grow ? prime(2 * $this->capacity) : $this->capacity)];
333 }
334 private function shrinkCheck(int $minLoad, int $maxLoad): array
335 {
336 $capacity = prime(intdiv($this->capacity, 2));
337 if (100 * $this->size >= $minLoad * $this->slotCount() || $capacity >= $this->capacity ||
338 100 * $this->size > $maxLoad * $capacity * $this->mode->parts()) return [];
339 $trial = self::rebuild($this->items(), $this->mode, $capacity);
340 if ($trial === null) return [];
341 $old = $this->capacity;
342 $this->adopt($trial);
343 $this->counters->rehashes++;
344 return ["REHASH shrink $old -> $capacity"];
345 }
346 public function put(string $key, int $value, int $maxLoad, string $command = 'PUT', bool $trace = false): Mutation
347 {
348 $found = $this->search($key, $trace);
349 $path = $trace ? ['PATH ' . ($found->path === [] ? 'none' : implode(' ', $found->path))] : [];
350 $this->counters->probes += $found->probes;
351 if ($found->item !== null) {
352 $old = $found->item->value;
353 $item = new Item($key, $command === 'ADD' ? $old + $value : $value);
354 if ($this->mode === Mode::CHAIN) $this->buckets[$found->position][$found->chainIndex] = $item;
355 else $this->slots[$found->position] = new Slot($item);
356 $this->counters->updates++;
357 $row = "$command $item UPDATED old=$old at=" . $this->position($found->position) . " probes=$found->probes";
358 return new Mutation(false, [$row, ...$path], []);
359 }
360 $item = new Item($key, $value);
361 $placed = $this->place($item, $found, $trace);
362 $this->size++;
363 $this->counters->inserts++;
364 $row = "$command $item NEW";
365 if ($placed->success) {
366 $row .= ' at=' . $this->position($placed->position) . " probes=$found->probes";
367 if ($placed->reused) $row .= ' reused';
368 if ($this->mode === Mode::ROBINHOOD) $row .= " shifted=$placed->shifted";
369 if ($this->mode === Mode::CUCKOO) $row .= " kicks=$placed->kicks";
370 } else {
371 $row .= " probes=$found->probes" . ($this->mode === Mode::CUCKOO ? " kicks=$placed->kicks failed" : ' full');
372 }
373 $rehashes = [];
374 if (!$placed->success) {
375 $rehashes[] = $this->automatic($this->mode === Mode::CUCKOO ? 'kicks' : 'full', prime(2 * $this->capacity), $placed->homeless);
376 }
377 array_push($rehashes, ...$this->loadCheck($maxLoad));
378 return new Mutation(true, [$row, ...$path, ...$placed->moves, ...$rehashes], $rehashes);
379 }
380 public function get(string $key, bool $trace): array
381 {
382 $found = $this->search($key, $trace);
383 $this->counters->probes += $found->probes;
384 if ($found->item !== null) {
385 $this->counters->hits++;
386 $row = 'GET ' . $found->item . ' at=' . $this->position($found->position) . " probes=$found->probes";
387 } else {
388 $this->counters->misses++;
389 $row = "GET $key NOT FOUND probes=$found->probes";
390 }
391 return [$row, ...($trace ? ['PATH ' . ($found->path === [] ? 'none' : implode(' ', $found->path))] : [])];
392 }
393 public function delete(string $key, int $minLoad, int $maxLoad, bool $trace = false): Mutation
394 {
395 $found = $this->search($key, $trace);
396 $path = $trace ? ['PATH ' . ($found->path === [] ? 'none' : implode(' ', $found->path))] : [];
397 $this->counters->probes += $found->probes;
398 if ($found->item === null) {
399 $this->counters->absent++;
400 return new Mutation(false, ["DEL $key NOT FOUND probes=$found->probes", ...$path], []);
401 }
402 $removed = $this->remove($found, $trace);
403 $this->counters->deletes++;
404 $row = "DEL $key OK at=" . $this->position($found->position) . " probes=$found->probes";
405 if ($this->mode === Mode::ROBINHOOD) $row .= " shifted=$removed->shifted";
406 $rehashes = $this->shrinkCheck($minLoad, $maxLoad);
407 return new Mutation(true, [$row, ...$path, ...$removed->moves, ...$rehashes], $rehashes);
408 }
409 public function costs(): array
410 {
411 $records = [];
412 if ($this->mode === Mode::CHAIN) {
413 foreach ($this->buckets as $index => $bucket) {
414 foreach ($bucket as $i => $item) $records[] = new Cost($item->key, $i + 1, $index);
415 }
416 } else {
417 foreach ($this->slots as $index => $slot) {
418 if ($slot->item === null) continue;
419 $key = $slot->item->key;
420 $cost = match ($this->mode) {
421 Mode::CUCKOO => intdiv($index, $this->capacity) + 1,
422 Mode::LINEAR, Mode::ROBINHOOD => $this->distance($key, $index) + 1,
423 default => $this->search($key)->probes,
424 };
425 $records[] = new Cost($key, $cost, $index);
426 }
427 }
428 return $records;
429 }

433 foreach ($this->costs() as $cost) { $total += $cost->probes; $longest = max($longest, $cost->probes); }
434 return new Metrics($longest, ratio($total, $this->size));

451 public function map(): array
452 {
453 $rows = ["MAP {$this->mode->value} cap=$this->capacity"];
454 $costBySlot = array_fill(0, $this->slotCount(), 0);
455 if ($this->mode !== Mode::CHAIN) foreach ($this->costs() as $cost) $costBySlot[$cost->position] = $cost->probes;
456 for ($part = 0; $part < $this->mode->parts(); $part++) {
457 $characters = '';
458 for ($index = 0; $index < $this->capacity; $index++) {
459 if ($this->mode === Mode::CHAIN) {
460 $count = count($this->buckets[$index]);
461 $characters .= $count === 0 ? '.' : gauge($count);
462 } else {
463 $flat = $part * $this->capacity + $index;
464 $slot = $this->slots[$flat];
465 $characters .= $slot->empty() ? '.' : ($slot->deleted ? 'D' : ($this->mode === Mode::CUCKOO ? '#' : gauge($costBySlot[$flat] - 1)));
466 }
467 }
468 for ($index = 0; $index < $this->capacity; $index += 40) {
469 $prefix = $this->mode === Mode::CUCKOO ? 'T' . ($part + 1) . ' ' : '';
470 $rows[] = $prefix . sprintf('%05d ', $index) . substr($characters, $index, 40);
471 }
472 }
473 return $rows;

484 public function scan(int $cursor, int $limit): array
485 {
486 $keys = [];
487 while ($cursor < $this->slotCount() && count($keys) < $limit) {
488 if ($this->mode === Mode::CHAIN) {
489 foreach ($this->buckets[$cursor] as $item) $keys[] = $item->key;
490 } elseif ($this->slots[$cursor]->item !== null) $keys[] = $this->slots[$cursor]->item->key;
491 $cursor++;
492 }
493 if ($cursor === $this->slotCount()) $cursor = 0;
494 return [implode(' ', ["SCAN next=$cursor", ...$keys])];
495 }
496 private function runs(int $start, int $count): array
497 {
498 $empty = null;
499 for ($i = 0; $i < $count; $i++) if ($this->slots[$start + $i]->empty()) { $empty = $i; break; }
500 if ($empty === null) return [$count];
501 $lengths = [];
502 $length = 0;
503 for ($offset = 1; $offset <= $count; $offset++) {
504 if (!$this->slots[$start + ($empty + $offset) % $count]->empty()) $length++;
505 elseif ($length > 0) { $lengths[] = $length; $length = 0; }
506 }
507 return $lengths;
508 }
509 public function clusters(): array
510 {
511 $lengths = [];
512 if ($this->mode === Mode::CHAIN) {
513 foreach ($this->buckets as $bucket) if ($bucket !== []) $lengths[] = count($bucket);
514 } else {
515 for ($part = 0; $part < $this->mode->parts(); $part++) array_push($lengths, ...$this->runs($part * $this->capacity, $this->capacity));
516 }
517 return ['CLUSTERS runs=' . count($lengths) . ' longest=' . ($lengths === [] ? 0 : max($lengths)), ...histogram($lengths)];
518 }
519 public function distribution(): array
520 {
521 $values = array_map(static fn(Cost $cost): int => $cost->probes, $this->costs());
522 return ["DIST keys=$this->size longest=" . ($values === [] ? 0 : max($values)), ...histogram($values)];
523 }

535 private function compare(int $capacity): array
536 {
537 $items = $this->table->items();
538 usort($items, static fn(Item $a, Item $b): int => strcmp($a->key, $b->key));
539 $rows = ['COMPARE keys=' . count($items) . " cap=$capacity"];
540 foreach (Mode::cases() as $mode) {
541 $trial = new HashTable($mode, $capacity);
542 foreach ($items as $item) $trial->put($item->key, $item->value, $this->maxLoad);
543 $values = $trial->metrics();
544 $rows[] = "$mode->value cap=$trial->capacity longest=$values->longest avg=$values->average probes={$trial->counters->probes} rehashes={$trial->counters->rehashes}";
545 }
546 return $rows;
547 }
548 private function bench(int $capacity, int $count, int $length, int $seed): array
549 {
550 $keys = iterator_to_array(generatedKeys(2 * $count, $length, $seed), false);
551 $rows = ["BENCH cap=$capacity count=$count len=$length seed=$seed"];
552 foreach (Mode::cases() as $mode) {
553 $trial = new HashTable($mode, $capacity);
554 for ($i = 0; $i < $count; $i++) $trial->put($keys[$i], $i + 1, $this->maxLoad);
555 $misses = $probes = 0;
556 for ($i = $count; $i < 2 * $count; $i++) {
557 $found = $trial->search($keys[$i]);
558 if ($found->item === null) { $misses++; $probes += $found->probes; }
559 }
560 $values = $trial->metrics();
561 $rows[] = "$mode->value cap=$trial->capacity keys=$trial->size hit=$values->average misses=$misses miss=" .
562 ratio($probes, $misses) . " rehashes={$trial->counters->rehashes}";
563 }
564 return $rows;
565 }
566 public function execute(string $line): array
567 {
568 $words = explode(' ', $line);
569 $command = $words[0];
570 if (!in_array($command, self::COMMANDS, true)) throw new CommandError('UNKNOWN COMMAND ' . $command);
571 switch ($command) {
572 case 'CREATE':
573 $mode = checkedMode($words[1]);
574 $capacity = (int)$words[2];
575 checkCapacity($capacity);
576 $this->table = new HashTable($mode, prime($capacity));
577 return ["CREATE $mode->value cap={$this->table->capacity}"];
578 case 'MAXLOAD':
579 case 'MINLOAD':
580 $value = (int)$words[1];
581 $maximum = $command === 'MAXLOAD';
582 if ($maximum ? $value < 10 || $value > 95 : $value < 0 || $value > 40) throw new CommandError('BAD LOAD');
583 if ($maximum) $this->maxLoad = $value; else $this->minLoad = $value;
584 return ["$command $value"];
585 case 'TRACE':
586 if ($words[1] !== 'ON' && $words[1] !== 'OFF') throw new CommandError('BAD ARGUMENT');
587 $this->tracing = $words[1] === 'ON';
588 return ['TRACE ' . $words[1]];
589 case 'HASH':
590 $key = $words[1];
591 $capacity = (int)$words[2];
592 checkCapacity($capacity);
593 return ["HASH $key m=$capacity h31=" . hashKey($key, 31, $capacity) . ' h37=' . hashKey($key, 37, $capacity) . ' step=' . (1 + hashKey($key, 37, $capacity - 1))];
594 case 'BENCH':
595 $capacity = (int)$words[1];
596 checkCapacity($capacity);
597 [$count, $length, $seed] = array_map('intval', array_slice($words, 2));
598 checkGeneration($count, $length, $seed);
599 return $this->bench(prime($capacity), $count, $length, $seed);
600 }
601 $table = $this->table;
602 if ($table === null) throw new CommandError('NO TABLE');
603 switch ($command) {
604 case 'PUT':
605 case 'ADD': return $table->put($words[1], (int)$words[2], $this->maxLoad, $command, $this->tracing)->lines;
606 case 'GET': return $table->get($words[1], $this->tracing);
607 case 'DEL': return $table->delete($words[1], $this->minLoad, $this->maxLoad, $this->tracing)->lines;
608 case 'PROBE':
609 $found = $table->search($words[1], true);
610 return [implode(' ', ['PROBE ' . $words[1], ...$found->path, ($found->item === null ? 'MISS' : 'FOUND') . " probes=$found->probes"])];
611 case 'GEN':
612 case 'GENDEL':
613 [$count, $length, $seed] = array_map('intval', array_slice($words, 1));
614 checkGeneration($count, $length, $seed);
615 $rows = [];
616 $changed = $index = 0;
617 foreach (generatedKeys($count, $length, $seed) as $key) {
618 $index++;
619 $result = $command === 'GEN' ? $table->put($key, $index, $this->maxLoad) : $table->delete($key, $this->minLoad, $this->maxLoad);
620 if ($result->changed) $changed++;
621 array_push($rows, ...$result->rehashes);
622 }
623 $rows[] = $command === 'GEN' ? "GEN $count new=$changed updated=" . ($count - $changed)
624 : "GENDEL $count deleted=$changed missing=" . ($count - $changed);
625 return $rows;
626 case 'RESIZE':
627 case 'CONVERT':
628 $mode = $table->mode;
629 $capacity = $table->capacity;
630 if ($command === 'RESIZE') { $capacity = (int)$words[1]; checkCapacity($capacity); $capacity = prime($capacity); }
631 else $mode = checkedMode($words[1]);
632 if (100 * $table->size > $this->maxLoad * $capacity * $mode->parts()) throw new CommandError('TOO SMALL');
633 $trial = HashTable::rebuild($table->items(), $mode, $capacity);
634 if ($trial === null) throw new CommandError('REBUILD FAILED');
635 $row = $command === 'RESIZE' ? "RESIZE $table->capacity -> $capacity" : "CONVERT {$table->mode->value} -> $mode->value cap=$capacity";
636 $table->adopt($trial);
637 return [$row];
638 case 'CLEAR':
639 $size = $table->size;
640 $table->adopt(new HashTable($table->mode, $table->capacity));
641 return ["CLEAR $size"];
642 case 'DUMP': return $table->dump();
643 case 'MAP': return $table->map();
644 case 'STATS': return $table->statistics();
645 case 'KEYS':
646 $keys = array_map(static fn(Item $item): string => $item->key, $table->items());
647 sort($keys, SORT_STRING);
648 return [implode(' ', ["KEYS $table->size", ...$keys])];
649 case 'TOP':
650 $limit = (int)$words[1];
651 if ($limit < 1 || $limit > 100) throw new CommandError('BAD ARGUMENT');
652 $records = $table->costs();
653 usort($records, static fn(Cost $a, Cost $b): int => ($b->probes <=> $a->probes) ?: strcmp($a->key, $b->key));
654 $selected = array_slice($records, 0, $limit);
655 return ['TOP ' . count($selected), ...array_map(fn(Cost $cost): string => "$cost->key cost=$cost->probes at=" . $table->position($cost->position), $selected)];
656 case 'SCAN':
657 $cursor = (int)$words[1];
658 $limit = (int)$words[2];
659 if ($limit < 1 || $limit > 100) throw new CommandError('BAD ARGUMENT');
660 if ($cursor >= $table->slotCount()) throw new CommandError('BAD CURSOR');
661 return $table->scan($cursor, $limit);
662 case 'CLUSTERS': return $table->clusters();
663 case 'DIST': return $table->distribution();
664 case 'COMPARE':
665 $capacity = (int)$words[1];
666 checkCapacity($capacity);
667 return $this->compare(prime($capacity));
668 default: throw new LogicException('Unhandled command ' . $command);
669 }
670 }
671}

673$count = (int)fgets(STDIN);
674$simulator = new Simulator();
675for ($i = 0; $i < $count; $i++) {
676 $line = rtrim(fgets(STDIN), "\r\n");
677 try { echo implode("\n", $simulator->execute($line)), "\n"; }
678 catch (CommandError $error) { echo $error->getMessage(), "\n"; }
679}
