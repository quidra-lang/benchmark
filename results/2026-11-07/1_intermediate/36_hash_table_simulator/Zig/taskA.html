33fn prime(value: usize) usize {
34 var candidate = @max(value, 2);
35 while (true) : (candidate += 1) {
36 var valid = true;
37 var divisor: usize = 2;
38 while (divisor * divisor <= candidate) : (divisor += 1) {
39 if (candidate % divisor == 0) {
40 valid = false;
41 break;
42 }
43 }
44 if (valid) return candidate;
45 }
46}

59const Key = struct {

61 len: usize = 0,
62 fn init(text_value: []const u8) Key {
63 var key: Key = .{ .len = text_value.len };
64 @memcpy(key.bytes[0..text_value.len], text_value);

67 fn text(key: *const Key) []const u8 {

73 fn hash(key: Key, base: usize, modulus: usize) usize {
74 var value: usize = 0;
75 for (key.text()) |character| value = (value * base + character) % modulus;
76 return value;
77 }

79 return std.mem.order(u8, a.text(), b.text()) == .lt;

176const Table = struct {

178 mode: Mode,
179 capacity: usize,

186 fn init(allocator: Allocator, mode: Mode, capacity: usize) Allocator.Error!Table {
187 var table: Table = .{ .allocator = allocator, .mode = mode, .capacity = capacity };
188 if (mode == .CHAIN) {
189 table.buckets = try allocator.alloc(std.ArrayList(Item), capacity);
190 @memset(table.buckets, .empty);
191 } else {
192 table.slots = try allocator.alloc(Slot, table.slotCount());
193 @memset(table.slots, .{});
194 }
195 return table;
196 }
197 fn deinit(table: *Table) void {
198 for (table.buckets) |*bucket| bucket.deinit(table.allocator);
199 table.allocator.free(table.buckets);
200 table.allocator.free(table.slots);
201 }
202 fn slotCount(table: *const Table) usize {
203 return table.capacity * table.mode.parts();
204 }
205 fn position(table: *const Table, index: usize) Position {
206 return .{ .mode = table.mode, .capacity = table.capacity, .index = index };
207 }
208 fn distance(table: *const Table, key: Key, index: usize) usize {
209 return (index + table.capacity - key.hash(31, table.capacity)) % table.capacity;
210 }
211 fn items(table: *const Table, gpa: Allocator) Allocator.Error![]Item {
212 var result: std.ArrayList(Item) = .empty;
213 if (table.mode == .CHAIN) {
214 for (table.buckets) |bucket| try result.appendSlice(gpa, bucket.items);
215 } else {
216 for (table.slots) |slot| if (slot.item) |item| try result.append(gpa, item);
217 }
218 return result.items;
219 }
220 fn search(table: *const Table, gpa: Allocator, key: Key, trace: bool) Allocator.Error!Search {
221 var result: Search = .{};
222 const first = key.hash(31, table.capacity);
223 if (table.mode == .CHAIN) {
224 for (table.buckets[first].items, 0..) |item, i| {
225 try result.examine(gpa, first, .{ .item = item }, trace);
226 if (item.key.eql(key)) {
227 result.found(item, first, i);
228 break;
229 }
230 }
231 return result;
232 }
233 if (table.mode == .CUCKOO) {
234 for ([_]usize{ first, table.capacity + key.hash(37, table.capacity) }) |index| {
235 const slot = table.slots[index];
236 try result.examine(gpa, index, slot, trace);
237 if (slot.item) |item| if (item.key.eql(key)) {
238 result.found(item, index, 0);
239 break;
240 };
241 }
242 return result;
243 }
244 const step: usize = if (table.mode == .DOUBLE) 1 + key.hash(37, table.capacity - 1) else 1;
245 for (0..table.capacity) |i| {
246 const offset = if (table.mode == .QUADRATIC) i * i else i * step;
247 const index = (first + offset) % table.capacity;
248 const slot = table.slots[index];
249 try result.examine(gpa, index, slot, trace);
250 if (slot.empty()) {
251 if (result.available == null) result.available = index;
252 break;
253 }
254 if (slot.deleted) {
255 if (result.available == null) result.available = index;
256 continue;
257 }
258 const item = slot.item.?;
259 if (item.key.eql(key)) {
260 result.found(item, index, 0);
261 break;
262 }
263 if (table.mode == .ROBINHOOD and table.distance(item.key, index) < i) break;
264 }
265 return result;
266 }
267 fn place(table: *Table, gpa: Allocator, item: Item, initial: ?Search, trace: bool) Allocator.Error!Placement {
268 var result: Placement = .{};
269 if (table.mode == .CHAIN) {
270 const index = item.key.hash(31, table.capacity);
271 try table.buckets[index].append(table.allocator, item);
272 result.position = index;
273 result.success = true;
274 return result;
275 }
276 if (table.mode.usesTombstones()) {
277 const found = initial orelse try table.search(gpa, item.key, false);
278 const index = found.available orelse {
279 result.homeless = item;
280 return result;
281 };
282 result.position = index;
283 result.reused = table.slots[index].deleted;
284 table.slots[index] = .{ .item = item };
285 if (result.reused) table.tombstones -= 1;
286 result.success = true;
287 return result;
288 }
289 var current = item;
290 if (table.mode == .ROBINHOOD) {
291 var index = item.key.hash(31, table.capacity);
292 var dist: usize = 0;
293 var from: ?usize = null;
294 while (true) {
295 const resident = table.slots[index].item orelse {
296 table.slots[index] = .{ .item = current };
297 if (trace) if (from) |source| try result.moves.append(gpa, .{ .key = current.key, .from = source, .to = index });
298 if (result.position == null) result.position = index;
299 result.success = true;
300 return result;
301 };
302 const resident_distance = table.distance(resident.key, index);
303 if (resident_distance < dist) {
304 table.slots[index] = .{ .item = current };
305 if (trace) if (from) |source| try result.moves.append(gpa, .{ .key = current.key, .from = source, .to = index });
306 if (result.position == null) result.position = index;
307 current = resident;
308 from = index;
309 dist = resident_distance;
310 result.shifted += 1;
311 }
312 index = (index + 1) % table.capacity;
313 dist += 1;
314 }
315 }
316 var part: usize = 0;
317 while (true) {
318 const index = part * table.capacity + current.key.hash(if (part == 0) 31 else 37, table.capacity);
319 const resident = table.slots[index].item orelse {
320 table.slots[index] = .{ .item = current };
321 result.success = true;
322 result.position = (try table.search(gpa, item.key, false)).position;
323 return result;
324 };
325 if (result.kicks == table.capacity) {
326 result.homeless = current;
327 return result;
328 }
329 table.slots[index] = .{ .item = current };
330 result.kicks += 1;
331 if (trace) try result.moves.append(gpa, .{ .key = resident.key, .from = index });
332 current = resident;
333 part = 1 - part;
334 }
335 }
336 fn remove(table: *Table, gpa: Allocator, found: Search, trace: bool) Allocator.Error!Removal {
337 var result: Removal = .{};
338 if (table.mode == .CHAIN) {
339 _ = table.buckets[found.position].orderedRemove(found.chain_index);
340 } else if (table.mode.usesTombstones()) {
341 table.slots[found.position] = .{ .deleted = true };
342 table.tombstones += 1;
343 } else {
344 table.slots[found.position] = .{};
345 if (table.mode == .ROBINHOOD) {
346 var next = (found.position + 1) % table.capacity;
347 while (table.slots[next].item) |item| {
348 if (table.distance(item.key, next) == 0) break;
349 const destination = (next + table.capacity - 1) % table.capacity;
350 table.slots[destination] = .{ .item = item };
351 table.slots[next] = .{};
352 if (trace) try result.moves.append(gpa, .{ .key = item.key, .from = next, .to = destination });
353 result.shifted += 1;
354 next = (next + 1) % table.capacity;
355 }
356 }
357 }
358 table.size -= 1;
359 return result;
360 }
361 fn rebuild(allocator: Allocator, gpa: Allocator, listed: []const Item, mode: Mode, capacity: usize) Allocator.Error!?Table {
362 var replacement = try Table.init(allocator, mode, capacity);
363 errdefer replacement.deinit();
364 for (listed) |item| {
365 if (!(try replacement.place(gpa, item, null, false)).success) {
366 replacement.deinit();
367 return null;
368 }
369 replacement.size += 1;
370 }
371 return replacement;
372 }
373 fn adopt(table: *Table, replacement: Table) void {
374 const counters = table.counters;
375 table.deinit();
376 table.* = replacement;
377 table.counters = counters;
378 }
379 fn automatic(table: *Table, gpa: Allocator, out: ?*Writer, reason: []const u8, initial_capacity: usize, pending: ?Item) Error!void {
380 var listed: std.ArrayList(Item) = .empty;
381 try listed.appendSlice(gpa, try table.items(gpa));
382 if (pending) |item| try listed.append(gpa, item);
383 const old = table.capacity;
384 var capacity = initial_capacity;
385 while (true) {
386 if (try Table.rebuild(table.allocator, gpa, listed.items, table.mode, capacity)) |replacement| {
387 table.adopt(replacement);
388 table.counters.rehashes += 1;
389 if (out) |writer| try line(writer, "REHASH {s} {d} -> {d}", .{ reason, old, table.capacity });
390 return;
391 }
392 capacity = prime(2 * capacity);
393 }
394 }
395 fn loadCheck(table: *Table, gpa: Allocator, out: ?*Writer, max_load: usize) Error!void {
396 const threshold = max_load * table.slotCount();
397 if (100 * (table.size + table.tombstones) <= threshold) return;
398 const grow = 200 * table.size > threshold;
399 try table.automatic(gpa, out, if (grow) "load" else "tomb", if (grow) prime(2 * table.capacity) else table.capacity, null);
400 }
401 fn shrinkCheck(table: *Table, gpa: Allocator, out: ?*Writer, min_load: usize, max_load: usize) Error!void {
402 const capacity = prime(table.capacity / 2);
403 if (100 * table.size >= min_load * table.slotCount() or capacity >= table.capacity or
404 100 * table.size > max_load * capacity * table.mode.parts()) return;
405 const listed = try table.items(gpa);
406 if (try Table.rebuild(table.allocator, gpa, listed, table.mode, capacity)) |replacement| {
407 const old = table.capacity;
408 table.adopt(replacement);
409 table.counters.rehashes += 1;
410 if (out) |writer| try line(writer, "REHASH shrink {d} -> {d}", .{ old, capacity });
411 }
412 }

424 fn writeMoves(table: *const Table, out: *Writer, moves: []const Move) Writer.Error!void {
425 for (moves) |move| {
426 if (move.to) |destination| try line(out, "SHIFT {f} {d} -> {d}", .{ move.key, move.from, destination }) else try line(out, "KICK {f} {f}", .{ move.key, table.position(move.from) });
427 }

429 fn put(table: *Table, gpa: Allocator, out: ?*Writer, key: Key, value: i64, add: bool, max_load: usize, detailed: bool, trace: bool) Error!bool {
430 const command: []const u8 = if (add) "ADD" else "PUT";
431 const found = try table.search(gpa, key, trace);
432 table.counters.probes += found.probes;
433 if (found.item) |old| {
434 const item: Item = .{ .key = key, .value = if (add) old.value + value else value };
435 if (table.mode == .CHAIN) table.buckets[found.position].items[found.chain_index] = item else table.slots[found.position] = .{ .item = item };
436 table.counters.updates += 1;
437 if (detailed) {
438 const writer = out.?;
439 try line(writer, "{s} {f} UPDATED old={d} at={f} probes={d}", .{ command, item, old.value, table.position(found.position), found.probes });
440 if (trace) try table.writePath(writer, found);
441 }
442 return false;
443 }
444 const item: Item = .{ .key = key, .value = value };
445 const placed = try table.place(gpa, item, found, trace);
446 table.size += 1;
447 table.counters.inserts += 1;
448 if (detailed) {
449 const writer = out.?;
450 try writer.print("{s} {f} NEW", .{ command, item });
451 if (placed.success) {
452 try writer.print(" at={f} probes={d}", .{ table.position(placed.position.?), found.probes });
453 if (placed.reused) try writer.writeAll(" reused");
454 if (table.mode == .ROBINHOOD) try writer.print(" shifted={d}", .{placed.shifted});
455 if (table.mode == .CUCKOO) try writer.print(" kicks={d}", .{placed.kicks});
456 } else {
457 try writer.print(" probes={d}", .{found.probes});
458 if (table.mode == .CUCKOO) try writer.print(" kicks={d} failed", .{placed.kicks}) else try writer.writeAll(" full");
459 }
460 try writer.writeByte('\n');
461 if (trace) {
462 try table.writePath(writer, found);
463 try table.writeMoves(writer, placed.moves.items);
464 }
465 }
466 if (!placed.success) try table.automatic(gpa, out, if (table.mode == .CUCKOO) "kicks" else "full", prime(2 * table.capacity), placed.homeless);
467 try table.loadCheck(gpa, out, max_load);
468 return true;
469 }
470 fn get(table: *Table, gpa: Allocator, out: *Writer, key: Key, trace: bool) Error!void {
471 const found = try table.search(gpa, key, trace);
472 table.counters.probes += found.probes;
473 if (found.item) |item| {
474 table.counters.hits += 1;
475 try line(out, "GET {f} at={f} probes={d}", .{ item, table.position(found.position), found.probes });
476 } else {
477 table.counters.misses += 1;
478 try line(out, "GET {f} NOT FOUND probes={d}", .{ key, found.probes });
479 }
480 if (trace) try table.writePath(out, found);
481 }
482 fn delete(table: *Table, gpa: Allocator, out: ?*Writer, key: Key, min_load: usize, max_load: usize, detailed: bool, trace: bool) Error!bool {
483 const found = try table.search(gpa, key, trace);
484 table.counters.probes += found.probes;
485 if (found.item == null) {
486 table.counters.absent += 1;
487 if (detailed) {
488 try line(out.?, "DEL {f} NOT FOUND probes={d}", .{ key, found.probes });
489 if (trace) try table.writePath(out.?, found);
490 }
491 return false;
492 }
493 const removed = try table.remove(gpa, found, trace);
494 table.counters.deletes += 1;
495 if (detailed) {
496 const writer = out.?;
497 try writer.print("DEL {f} OK at={f} probes={d}", .{ key, table.position(found.position), found.probes });
498 if (table.mode == .ROBINHOOD) try writer.print(" shifted={d}", .{removed.shifted});
499 try writer.writeByte('\n');
500 if (trace) {
501 try table.writePath(writer, found);
502 try table.writeMoves(writer, removed.moves.items);
503 }
504 }
505 try table.shrinkCheck(gpa, out, min_load, max_load);
506 return true;
507 }
508 fn costs(table: *const Table, gpa: Allocator) Allocator.Error![]Cost {
509 var result: std.ArrayList(Cost) = .empty;
510 if (table.mode == .CHAIN) {
511 for (table.buckets, 0..) |bucket, index| for (bucket.items, 0..) |item, i| {
512 try result.append(gpa, .{ .key = item.key, .probes = i + 1, .position = index });
513 };
514 } else {
515 for (table.slots, 0..) |slot, index| if (slot.item) |item| {
516 const probes = switch (table.mode) {
517 .CUCKOO => index / table.capacity + 1,
518 .LINEAR, .ROBINHOOD => table.distance(item.key, index) + 1,
519 else => (try table.search(gpa, item.key, false)).probes,
520 };
521 try result.append(gpa, .{ .key = item.key, .probes = probes, .position = index });
522 };
523 }
524 return result.items;
525 }
526 fn metrics(table: *const Table, gpa: Allocator) Allocator.Error!Metrics {
527 var total: u64 = 0;
528 var longest: usize = 0;
529 for (try table.costs(gpa)) |cost| {
530 total += cost.probes;
531 longest = @max(longest, cost.probes);
532 }
533 return .{ .longest = longest, .average = Ratio.init(total, table.size) };
534 }
535 fn dump(table: *const Table, out: *Writer) Writer.Error!void {
536 try line(out, "DUMP {s} cap={d} size={d} tomb={d}", .{ @tagName(table.mode), table.capacity, table.size, table.tombstones });
537 if (table.mode == .CHAIN) {
538 for (table.buckets, 0..) |bucket, index| {
539 try out.print("{d}", .{index});
540 if (bucket.items.len == 0) try out.writeAll(" -");
541 for (bucket.items) |item| try out.print(" {f}", .{item});
542 try out.writeByte('\n');
543 }
544 } else {
545 for (table.slots, 0..) |slot, index| {
546 if (table.mode == .CUCKOO) try out.print("T{d} {d} ", .{ index / table.capacity + 1, index % table.capacity }) else try out.print("{d} ", .{index});
547 if (slot.item) |item| {
548 try out.print("{f}", .{item});
549 if (table.mode == .ROBINHOOD) try out.print(" d={d}", .{table.distance(item.key, index)});
550 } else try slot.content(out);
551 try out.writeByte('\n');
552 }
553 }
554 }
555 fn map(table: *const Table, gpa: Allocator, out: *Writer) Error!void {
556 try line(out, "MAP {s} cap={d}", .{ @tagName(table.mode), table.capacity });
557 const cost_by_slot = try gpa.alloc(usize, table.slotCount());
558 @memset(cost_by_slot, 0);
559 if (table.mode != .CHAIN) for (try table.costs(gpa)) |cost| {
560 cost_by_slot[cost.position] = cost.probes;
561 };
562 const characters = try gpa.alloc(u8, table.capacity);
563 for (0..table.mode.parts()) |part| {
564 for (characters, 0..) |*character, index| {
565 if (table.mode == .CHAIN) {
566 const length = table.buckets[index].items.len;
567 character.* = if (length == 0) '.' else gauge(length);
568 } else {
569 const flat = part * table.capacity + index;
570 const slot = table.slots[flat];
571 character.* = if (slot.empty()) '.' else if (slot.deleted) 'D' else if (table.mode == .CUCKOO) '#' else gauge(cost_by_slot[flat] - 1);
572 }
573 }
574 var index: usize = 0;
575 while (index < table.capacity) : (index += 40) {
576 if (table.mode == .CUCKOO) try out.print("T{d} ", .{part + 1});
577 try line(out, "{d:0>5} {s}", .{ index, characters[index..@min(index + 40, table.capacity)] });
578 }
579 }
580 }
581 fn statistics(table: *const Table, gpa: Allocator, out: *Writer) Error!void {
582 const values = try table.metrics(gpa);
583 const count = table.counters;
584 try line(out, "STATS {s} size={d} cap={d} slots={d} tomb={d} load={f}% longest={d} avg={f} probes={d} rehashes={d}", .{
585 @tagName(table.mode), table.size, table.capacity, table.slotCount(), table.tombstones,
586 Ratio.init(100 * table.size, table.slotCount()), values.longest, values.average, count.probes, count.rehashes,
587 });
588 try line(out, "OPS inserts={d} updates={d} hits={d} misses={d} deletes={d} absent={d}", .{ count.inserts, count.updates, count.hits, count.misses, count.deletes, count.absent });
589 }
590 fn scan(table: *const Table, gpa: Allocator, out: *Writer, cursor: usize, limit: usize) Error!void {
591 var keys: std.ArrayList(Key) = .empty;
592 var next = cursor;
593 while (next < table.slotCount() and keys.items.len < limit) : (next += 1) {
594 if (table.mode == .CHAIN) {
595 for (table.buckets[next].items) |item| try keys.append(gpa, item.key);
596 } else if (table.slots[next].item) |item| try keys.append(gpa, item.key);
597 }
598 if (next == table.slotCount()) next = 0;
599 try out.print("SCAN next={d}", .{next});
600 for (keys.items) |key| try out.print(" {f}", .{key});
601 try out.writeByte('\n');
602 }
603 fn appendRuns(table: *const Table, gpa: Allocator, lengths: *std.ArrayList(usize), part: usize) Allocator.Error!void {
604 const offset = part * table.capacity;
605 var empty: ?usize = null;
606 for (0..table.capacity) |i| if (table.slots[offset + i].empty()) {
607 empty = i;
608 break;
609 };
610 const start = empty orelse {
611 try lengths.append(gpa, table.capacity);
612 return;
613 };
614 var length: usize = 0;
615 for (1..table.capacity + 1) |i| {
616 if (!table.slots[offset + (start + i) % table.capacity].empty()) {
617 length += 1;
618 } else if (length > 0) {
619 try lengths.append(gpa, length);
620 length = 0;
621 }
622 }
623 }
624 fn clusters(table: *const Table, gpa: Allocator, out: *Writer) Error!void {
625 var lengths: std.ArrayList(usize) = .empty;
626 if (table.mode == .CHAIN) {
627 for (table.buckets) |bucket| if (bucket.items.len > 0) try lengths.append(gpa, bucket.items.len);
628 } else for (0..table.mode.parts()) |part| try table.appendRuns(gpa, &lengths, part);
629 var longest: usize = 0;
630 for (lengths.items) |length| longest = @max(longest, length);
631 try line(out, "CLUSTERS runs={d} longest={d}", .{ lengths.items.len, longest });
632 try histogram(out, lengths.items);
633 }
634 fn distribution(table: *const Table, gpa: Allocator, out: *Writer) Error!void {
635 var values: std.ArrayList(usize) = .empty;
636 var longest: usize = 0;
637 for (try table.costs(gpa)) |cost| {
638 try values.append(gpa, cost.probes);
639 longest = @max(longest, cost.probes);
640 }
641 try line(out, "DIST keys={d} longest={d}", .{ table.size, longest });
642 try histogram(out, values.items);
643 }
644};

646fn smaller(_: void, a: usize, b: usize) bool {
647 return a < b;
648}
649fn histogram(out: *Writer, values: []usize) Writer.Error!void {
650 std.sort.heap(usize, values, {}, smaller);
651 var index: usize = 0;
652 while (index < values.len) {
653 var end = index + 1;
654 while (end < values.len and values[end] == values[index]) : (end += 1) {}
655 const count = end - index;
656 try out.print("{d} {d} ", .{ values[index], count });
657 try out.splatByteAll('#', @min(count, 50));
658 if (count > 50) try out.writeByte('+');
659 try out.writeByte('\n');
660 index = end;
661 }

663const State = struct {
664 allocator: Allocator,
665 table: ?Table = null,
666 max_load: usize = 75,
667 min_load: usize = 0,
668 tracing: bool = false,
669 fn deinit(state: *State) void {
670 if (state.table) |*table| table.deinit();
671 }
672};
673const modes = [_]Mode{ .CHAIN, .LINEAR, .QUADRATIC, .DOUBLE, .ROBINHOOD, .CUCKOO };
674const Session = struct {
675 gpa: Allocator,
676 state: *State,
677 out: *Writer,
678 problem: []const u8 = "",
679 fn fail(session: *Session, comptime fmt: []const u8, args: anytype) Error {
680 session.problem = std.fmt.allocPrint(session.gpa, fmt, args) catch return error.OutOfMemory;
681 return error.Invalid;
682 }
683 fn checkedMode(session: *Session, text: []const u8) Error!Mode {
684 return std.meta.stringToEnum(Mode, text) orelse return session.fail("BAD MODE {s}", .{text});
685 }
686 fn checkCapacity(session: *Session, value: usize) Error!void {
687 if (value < 2 or value > 10000) return session.fail("BAD CAPACITY", .{});
688 }
689 fn checkGeneration(session: *Session, count: usize, length: usize, seed: usize) Error!void {
690 if (count < 1 or count > 500 or length < 1 or length > 10 or seed < 1 or seed > 2147483646) return session.fail("BAD ARGUMENT", .{});
691 }
692 fn compare(session: *Session, capacity: usize) Error!void {
693 const listed = try session.state.table.?.items(session.gpa);
694 std.sort.heap(Item, listed, {}, Item.less);
695 try line(session.out, "COMPARE keys={d} cap={d}", .{ listed.len, capacity });
696 for (modes) |mode| {
697 var trial = try Table.init(session.state.allocator, mode, capacity);
698 defer trial.deinit();
699 for (listed) |item| _ = try trial.put(session.gpa, null, item.key, item.value, false, session.state.max_load, false, false);
700 const values = try trial.metrics(session.gpa);
701 try line(session.out, "{s} cap={d} longest={d} avg={f} probes={d} rehashes={d}", .{ @tagName(mode), trial.capacity, values.longest, values.average, trial.counters.probes, trial.counters.rehashes });
702 }
703 }
704 fn bench(session: *Session, capacity: usize, count: usize, length: usize, seed: usize) Error!void {
705 const keys = try session.gpa.alloc(Key, 2 * count);
706 var generator: Generator = .{ .state = seed, .length = length };
707 for (keys) |*key| key.* = generator.next();
708 try line(session.out, "BENCH cap={d} count={d} len={d} seed={d}", .{ capacity, count, length, seed });
709 for (modes) |mode| {
710 var trial = try Table.init(session.state.allocator, mode, capacity);
711 defer trial.deinit();
712 for (keys[0..count], 1..) |key, i| _ = try trial.put(session.gpa, null, key, @intCast(i), false, session.state.max_load, false, false);
713 var misses: u64 = 0;
714 var probes: u64 = 0;
715 for (keys[count..]) |key| {
716 const found = try trial.search(session.gpa, key, false);
717 if (found.item == null) {
718 misses += 1;
719 probes += found.probes;
720 }
721 }
722 const values = try trial.metrics(session.gpa);
723 try line(session.out, "{s} cap={d} keys={d} hit={f} misses={d} miss={f} rehashes={d}", .{ @tagName(mode), trial.capacity, trial.size, values.average, misses, Ratio.init(probes, misses), trial.counters.rehashes });
724 }
725 }
726 fn execute(session: *Session, text: []const u8) Error!void {
727 var words: [5][]const u8 = undefined;
728 var tokens = std.mem.tokenizeScalar(u8, text, ' ');
729 for (&words) |*word| word.* = tokens.next() orelse break;
730 const command = std.meta.stringToEnum(Command, words[0]) orelse return session.fail("UNKNOWN COMMAND {s}", .{words[0]});
731 const state = session.state;
732 const out = session.out;
733 const gpa = session.gpa;
734 switch (command) {
735 .CREATE => {
736 const mode = try session.checkedMode(words[1]);
737 const capacity = integer(words[2]);
738 try session.checkCapacity(capacity);
739 const replacement = try Table.init(state.allocator, mode, prime(capacity));
740 if (state.table) |*old| old.deinit();
741 state.table = replacement;
742 return line(out, "CREATE {s} cap={d}", .{ @tagName(mode), replacement.capacity });
743 },
744 .MAXLOAD, .MINLOAD => {
745 const value = integer(words[1]);
746 const maximum = command == .MAXLOAD;
747 if (if (maximum) value < 10 or value > 95 else value > 40) return session.fail("BAD LOAD", .{});
748 if (maximum) state.max_load = value else state.min_load = value;
749 return line(out, "{s} {d}", .{ @tagName(command), value });
750 },
751 .TRACE => {
752 if (!equal(words[1], "ON") and !equal(words[1], "OFF")) return session.fail("BAD ARGUMENT", .{});
753 state.tracing = equal(words[1], "ON");
754 return line(out, "TRACE {s}", .{words[1]});
755 },
756 .HASH => {
757 const key = Key.init(words[1]);
758 const capacity = integer(words[2]);
759 try session.checkCapacity(capacity);
760 return line(out, "HASH {f} m={d} h31={d} h37={d} step={d}", .{ key, capacity, key.hash(31, capacity), key.hash(37, capacity), 1 + key.hash(37, capacity - 1) });
761 },
762 .BENCH => {
763 const capacity = integer(words[1]);
764 try session.checkCapacity(capacity);
765 const count = integer(words[2]);
766 const length = integer(words[3]);
767 const seed = integer(words[4]);
768 try session.checkGeneration(count, length, seed);
769 return session.bench(prime(capacity), count, length, seed);
770 },
771 else => {},
772 }
773 if (state.table == null) return session.fail("NO TABLE", .{});
774 const table = &state.table.?;
775 switch (command) {
776 .PUT, .ADD => {
777 _ = try table.put(gpa, out, Key.init(words[1]), signedInteger(words[2]), command == .ADD, state.max_load, true, state.tracing);
778 },
779 .GET => try table.get(gpa, out, Key.init(words[1]), state.tracing),
780 .DEL => {
781 _ = try table.delete(gpa, out, Key.init(words[1]), state.min_load, state.max_load, true, state.tracing);
782 },
783 .PROBE => {
784 const key = Key.init(words[1]);
785 const found = try table.search(gpa, key, true);
786 try out.print("PROBE {f}", .{key});
787 try table.pathElements(out, found.path.items);
788 try line(out, " {s} probes={d}", .{ if (found.item == null) "MISS" else "FOUND", found.probes });
789 },
790 .GEN, .GENDEL => {
791 const count = integer(words[1]);
792 const length = integer(words[2]);
793 const seed = integer(words[3]);
794 try session.checkGeneration(count, length, seed);
795 var generator: Generator = .{ .state = seed, .length = length };
796 var changed: usize = 0;
797 for (0..count) |i| {
798 const key = generator.next();
799 const success = if (command == .GEN) try table.put(gpa, out, key, @intCast(i + 1), false, state.max_load, false, false) else try table.delete(gpa, out, key, state.min_load, state.max_load, false, false);
800 if (success) changed += 1;
801 }
802 if (command == .GEN) try line(out, "GEN {d} new={d} updated={d}", .{ count, changed, count - changed }) else try line(out, "GENDEL {d} deleted={d} missing={d}", .{ count, changed, count - changed });
803 },
804 .RESIZE, .CONVERT => {
805 var mode = table.mode;
806 var capacity = table.capacity;
807 if (command == .RESIZE) {
808 capacity = integer(words[1]);
809 try session.checkCapacity(capacity);
810 capacity = prime(capacity);
811 } else mode = try session.checkedMode(words[1]);
812 if (100 * table.size > state.max_load * capacity * mode.parts()) return session.fail("TOO SMALL", .{});
813 const listed = try table.items(gpa);
814 const replacement = (try Table.rebuild(state.allocator, gpa, listed, mode, capacity)) orelse return session.fail("REBUILD FAILED", .{});
815 if (command == .RESIZE) try line(out, "RESIZE {d} -> {d}", .{ table.capacity, capacity }) else try line(out, "CONVERT {s} -> {s} cap={d}", .{ @tagName(table.mode), @tagName(mode), capacity });
816 table.adopt(replacement);
817 },
818 .CLEAR => {
819 const size = table.size;
820 table.adopt(try Table.init(state.allocator, table.mode, table.capacity));
821 try line(out, "CLEAR {d}", .{size});
822 },
823 .DUMP => try table.dump(out),
824 .MAP => try table.map(gpa, out),
825 .STATS => try table.statistics(gpa, out),
826 .KEYS => {
827 const listed = try table.items(gpa);
828 std.sort.heap(Item, listed, {}, Item.less);
829 try out.print("KEYS {d}", .{table.size});
830 for (listed) |item| try out.print(" {f}", .{item.key});
831 try out.writeByte('\n');
832 },
833 .TOP => {
834 const limit = integer(words[1]);
835 if (limit < 1 or limit > 100) return session.fail("BAD ARGUMENT", .{});
836 const records = try table.costs(gpa);
837 std.sort.heap(Cost, records, {}, Cost.less);
838 const count = @min(limit, records.len);
839 try line(out, "TOP {d}", .{count});
840 for (records[0..count]) |cost| try line(out, "{f} cost={d} at={f}", .{ cost.key, cost.probes, table.position(cost.position) });
841 },
842 .SCAN => {
843 const cursor = integer(words[1]);
844 const limit = integer(words[2]);
845 if (limit < 1 or limit > 100) return session.fail("BAD ARGUMENT", .{});
846 if (cursor >= table.slotCount()) return session.fail("BAD CURSOR", .{});
847 try table.scan(gpa, out, cursor, limit);
848 },
849 .CLUSTERS => try table.clusters(gpa, out),
850 .DIST => try table.distribution(gpa, out),
851 .COMPARE => {
852 const capacity = integer(words[1]);
853 try session.checkCapacity(capacity);
854 try session.compare(prime(capacity));
855 },
856 else => unreachable,
857 }
858 }
859};

861fn readLine(gpa: Allocator, input: *std.Io.Reader) ![]const u8 {
862 var buffer: Writer.Allocating = .init(gpa);
863 defer buffer.deinit();
864 _ = try input.streamDelimiterEnding(&buffer.writer, '\n');
865 if (input.seek < input.end) input.toss(1) else if (buffer.written().len == 0) return error.EndOfStream;
866 const text = try buffer.toOwnedSlice();
867 return if (text.len > 0 and text[text.len - 1] == '\r') text[0 .. text.len - 1] else text;
868}
869pub fn main(init: std.process.Init) !void {
870 var header = std.heap.ArenaAllocator.init(std.heap.page_allocator);
871 defer header.deinit();
872 var state: State = .{ .allocator = std.heap.page_allocator };
873 defer state.deinit();
874 var input_buffer: [1024]u8 = undefined;
875 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
876 var output_buffer: [4096]u8 = undefined;
877 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
878 const count = integer(try readLine(header.allocator(), &input.interface));
879 for (0..count) |_| {
880 var scratch = std.heap.ArenaAllocator.init(std.heap.page_allocator);
881 defer scratch.deinit();
882 const gpa = scratch.allocator();
883 const text = try readLine(gpa, &input.interface);
884 var response: Writer.Allocating = .init(gpa);
885 defer response.deinit();
886 var session: Session = .{ .gpa = gpa, .state = &state, .out = &response.writer };
887 session.execute(text) catch |err| {
888 if (err != error.Invalid) return err;
889 try line(&output.interface, "ERROR {s}", .{session.problem});
890 try output.interface.flush();
891 continue;
892 };
893 try output.interface.writeAll(response.written());
894 try output.interface.flush();
895 }
