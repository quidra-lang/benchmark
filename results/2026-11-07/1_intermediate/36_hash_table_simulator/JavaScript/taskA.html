14function hash(key, base, modulus) {
15 let value = 0;
16 for (const character of key) value = (value * base + character.charCodeAt(0)) % modulus;
17 return value;
18}
19function prime(value) {
20 for (let candidate = Math.max(2, value); ; ++candidate) {
21 let valid = true;
22 for (let divisor = 2; divisor * divisor <= candidate; ++divisor) if (candidate % divisor === 0) { valid = false; break; }
23 if (valid) return candidate;
24 }
25}
26function decimal(numerator, denominator) {
27 const value = denominator === 0 ? 0 : Math.floor(100 * numerator / denominator);
28 return `${Math.floor(value / 100)}.${String(value % 100).padStart(2, '0')}`;
29}
30function generatedKeys(count, length, seed) {
31 let state = seed;
32 const keys = [];
33 for (let index = 0; index < count; ++index) {
34 let key = '';
35 for (let place = 0; place < length; ++place) {
36 state = 48271 * state % 2147483647;
37 key += String.fromCharCode(97 + state % 26);
38 }
39 keys.push(key);
40 }
41 return keys;
42}
43function histogram(values) {
44 const counts = new Map();
45 for (const value of values) counts.set(value, (counts.get(value) ?? 0) + 1);
46 return [...counts].sort((a, b) => a[0] - b[0]).map(([value, count]) => `${value} ${count} ${'#'.repeat(Math.min(count, 50))}${count > 50 ? '+' : ''}`);
47}
48function runs(slots) {
49 const empty = slots.indexOf(null);
50 if (empty < 0) return [slots.length];
51 const lengths = [];
52 let length = 0;
53 for (let offset = 1; offset <= slots.length; ++offset) {
54 if (slots[(empty + offset) % slots.length] !== null) ++length;
55 else if (length > 0) { lengths.push(length); length = 0; }
56 }
57 return lengths;

59class Table {
60 constructor(mode, capacity) {
61 this.mode = mode;
62 this.cap = capacity;
63 this.size = 0;
64 this.tomb = 0;
65 this.buckets = mode === 'CHAIN' ? Array.from({ length: capacity }, () => []) : [];
66 this.slots = mode === 'CHAIN' ? [] : Array(capacity * (mode === 'CUCKOO' ? 2 : 1)).fill(null);
67 this.counters = { probes: 0, rehashes: 0, inserts: 0, updates: 0, hits: 0, misses: 0, deletes: 0, absent: 0 };
68 }
69 slotCount() { return this.cap * (this.mode === 'CUCKOO' ? 2 : 1); }
70 position(index) { return this.mode === 'CUCKOO' ? `T${Math.floor(index / this.cap) + 1}.${index % this.cap}` : String(index); }
71 distance(key, position) { return (position - hash(key, 31, this.cap) + this.cap) % this.cap; }
72 items() { return this.mode === 'CHAIN' ? this.buckets.flat() : this.slots.filter(isItem); }
73 search(key, tracing = false) {
74 let probes = 0, available = null;
75 const path = [], first = hash(key, 31, this.cap);
76 const result = (item = null, position = -1, chainIndex = -1) => ({ item, position, chainIndex, probes, path, available });
77 const examine = (index, slot) => {
78 ++probes;
79 if (tracing) path.push(`${this.position(index)}:${content(slot)}`);
80 };
81 if (this.mode === 'CHAIN') {
82 const bucket = this.buckets[first];
83 for (let index = 0; index < bucket.length; ++index) {
84 examine(first, bucket[index]);
85 if (bucket[index].key === key) return result(bucket[index], first, index);
86 }
87 return result();
88 }
89 if (this.mode === 'CUCKOO') {
90 for (const index of [first, this.cap + hash(key, 37, this.cap)]) {
91 const item = this.slots[index];
92 examine(index, item);
93 if (item !== null && item.key === key) return result(item, index);
94 }
95 return result();
96 }
97 const step = this.mode === 'DOUBLE' ? 1 + hash(key, 37, this.cap - 1) : 1;
98 for (let i = 0; i < this.cap; ++i) {
99 const offset = this.mode === 'QUADRATIC' ? i * i : i * step;
100 const index = (first + offset) % this.cap, slot = this.slots[index];
101 examine(index, slot);
102 if (slot === null) { if (available === null) available = index; return result(); }
103 if (slot === tombstone) { if (available === null) available = index; continue; }
104 if (slot.key === key) return result(slot, index);
105 if (this.mode === 'ROBINHOOD' && this.distance(slot.key, index) < i) return result();
106 }
107 return result();
108 }
109 place(item, initial = null, tracing = false) {
110 const moves = [];
111 const result = (success, position = -1, extra = {}) => ({ success, position, reused: false, shifted: 0, kicks: 0, homeless: null, moves, ...extra });
112 if (this.mode === 'CHAIN') {
113 const index = hash(item.key, 31, this.cap);
114 this.buckets[index].push(item);
115 return result(true, index);
116 }
117 if (['LINEAR', 'QUADRATIC', 'DOUBLE'].includes(this.mode)) {
118 const search = initial ?? this.search(item.key), index = search.available;
119 if (index === null) return result(false, -1, { homeless: item });
120 const reused = this.slots[index] === tombstone;
121 this.slots[index] = item;
122 if (reused) --this.tomb;
123 return result(true, index, { reused });
124 }
125 if (this.mode === 'ROBINHOOD') {
126 let current = item, position = hash(item.key, 31, this.cap), distance = 0, from = null, original = null, shifted = 0;
127 for (;;) {
128 const occupant = this.slots[position];
129 if (occupant === null) {
130 this.slots[position] = current;
131 if (tracing && from !== null) moves.push(`SHIFT ${current.key} ${from} -> ${position}`);
132 if (original === null) original = position;
133 return result(true, original, { shifted });
134 }
135 const otherDistance = this.distance(occupant.key, position);
136 if (otherDistance < distance) {
137 this.slots[position] = current;
138 if (tracing && from !== null) moves.push(`SHIFT ${current.key} ${from} -> ${position}`);
139 if (original === null) original = position;
140 current = occupant;
141 from = position;
142 distance = otherDistance;
143 ++shifted;
144 }
145 position = (position + 1) % this.cap;
146 ++distance;
147 }
148 }
149 let current = item, part = 0, kicks = 0;
150 for (;;) {
151 const index = part * this.cap + hash(current.key, part === 0 ? 31 : 37, this.cap);
152 const occupant = this.slots[index];
153 if (occupant === null) {
154 this.slots[index] = current;
155 return result(true, this.search(item.key).position, { kicks });
156 }
157 if (kicks === this.cap) return result(false, -1, { kicks, homeless: current });
158 this.slots[index] = current;
159 ++kicks;
160 if (tracing) moves.push(`KICK ${occupant.key} ${this.position(index)}`);
161 current = occupant;
162 part = 1 - part;
163 }
164 }
165 remove(found, tracing) {
166 const moves = [];
167 let shifted = 0;
168 if (this.mode === 'CHAIN') this.buckets[found.position].splice(found.chainIndex, 1);
169 else if (['LINEAR', 'QUADRATIC', 'DOUBLE'].includes(this.mode)) {
170 this.slots[found.position] = tombstone;
171 ++this.tomb;
172 } else {
173 this.slots[found.position] = null;
174 if (this.mode === 'ROBINHOOD') {
175 let next = (found.position + 1) % this.cap;
176 while (this.slots[next] !== null && this.distance(this.slots[next].key, next) >= 1) {
177 const item = this.slots[next], destination = (next - 1 + this.cap) % this.cap;
178 this.slots[destination] = item;
179 this.slots[next] = null;
180 if (tracing) moves.push(`SHIFT ${item.key} ${next} -> ${destination}`);
181 ++shifted;
182 next = (next + 1) % this.cap;
183 }
184 }
185 }
186 --this.size;
187 return { shifted, moves };
188 }
189 static rebuild(items, mode, capacity) {
190 const table = new Table(mode, capacity);
191 for (const item of items) {
192 if (!table.place(item).success) return null;
193 ++table.size;
194 }
195 return table;
196 }
197 adopt(table) {
198 this.mode = table.mode;
199 this.cap = table.cap;
200 this.size = table.size;
201 this.tomb = table.tomb;
202 this.buckets = table.buckets;
203 this.slots = table.slots;
204 }
205 automatic(reason, capacity, pending = null) {
206 const items = this.items(), old = this.cap;
207 if (pending !== null) items.push(pending);
208 for (;;) {
209 const replacement = Table.rebuild(items, this.mode, capacity);
210 if (replacement !== null) {
211 this.adopt(replacement);
212 ++this.counters.rehashes;
213 return `REHASH ${reason} ${old} -> ${this.cap}`;
214 }
215 capacity = prime(2 * capacity);
216 }
217 }
218 loadCheck(maxLoad) {
219 const threshold = maxLoad * this.slotCount();
220 if (100 * (this.size + this.tomb) <= threshold) return [];
221 const grow = 200 * this.size > threshold;
222 return [this.automatic(grow ? 'load' : 'tomb', grow ? prime(2 * this.cap) : this.cap)];
223 }
224 shrinkCheck(minLoad, maxLoad) {
225 const capacity = prime(Math.floor(this.cap / 2)), slots = capacity * (this.mode === 'CUCKOO' ? 2 : 1);
226 if (100 * this.size >= minLoad * this.slotCount() || capacity >= this.cap || 100 * this.size > maxLoad * slots) return [];
227 const replacement = Table.rebuild(this.items(), this.mode, capacity);
228 if (replacement === null) return [];
229 const old = this.cap;
230 this.adopt(replacement);
231 ++this.counters.rehashes;
232 return [`REHASH shrink ${old} -> ${capacity}`];
233 }
234 put(key, value, command, maxLoad, tracing = false) {
235 const found = this.search(key, tracing), trace = tracing ? [`PATH ${found.path.length ? found.path.join(' ') : 'none'}`] : [];
236 this.counters.probes += found.probes;
237 if (found.item !== null) {
238 const old = found.item.value, next = command === 'ADD' ? old + value : value;
239 const item = { key, value: next };
240 if (this.mode === 'CHAIN') this.buckets[found.position][found.chainIndex] = item;
241 else this.slots[found.position] = item;
242 ++this.counters.updates;
243 return { fresh: false, rehashes: [], rows: [`${command} ${key}=${next} UPDATED old=${old} at=${this.position(found.position)} probes=${found.probes}`, ...trace] };
244 }
245 const item = { key, value }, placed = this.place(item, found, tracing);
246 ++this.size;
247 ++this.counters.inserts;
248 let row = `${command} ${key}=${value} NEW`;
249 if (placed.success) {
250 row += ` at=${this.position(placed.position)} probes=${found.probes}`;
251 if (placed.reused) row += ' reused';
252 if (this.mode === 'ROBINHOOD') row += ` shifted=${placed.shifted}`;
253 if (this.mode === 'CUCKOO') row += ` kicks=${placed.kicks}`;
254 } else row += ` probes=${found.probes}` + (this.mode === 'CUCKOO' ? ` kicks=${placed.kicks} failed` : ' full');
255 const rehashes = [];
256 if (!placed.success) rehashes.push(this.automatic(this.mode === 'CUCKOO' ? 'kicks' : 'full', prime(2 * this.cap), placed.homeless));
257 rehashes.push(...this.loadCheck(maxLoad));
258 return { fresh: true, rehashes, rows: [row, ...trace, ...placed.moves, ...rehashes] };
259 }
260 get(key, tracing) {
261 const found = this.search(key, tracing);
262 this.counters.probes += found.probes;
263 let row;
264 if (found.item === null) { ++this.counters.misses; row = `GET ${key} NOT FOUND probes=${found.probes}`; }
265 else { ++this.counters.hits; row = `GET ${key}=${found.item.value} at=${this.position(found.position)} probes=${found.probes}`; }
266 return tracing ? [row, `PATH ${found.path.length ? found.path.join(' ') : 'none'}`] : [row];
267 }
268 delete(key, minLoad, maxLoad, tracing = false) {
269 const found = this.search(key, tracing), trace = tracing ? [`PATH ${found.path.length ? found.path.join(' ') : 'none'}`] : [];
270 this.counters.probes += found.probes;
271 if (found.item === null) {
272 ++this.counters.absent;
273 return { deleted: false, rehashes: [], rows: [`DEL ${key} NOT FOUND probes=${found.probes}`, ...trace] };
274 }
275 const removed = this.remove(found, tracing);
276 ++this.counters.deletes;
277 let row = `DEL ${key} OK at=${this.position(found.position)} probes=${found.probes}`;
278 if (this.mode === 'ROBINHOOD') row += ` shifted=${removed.shifted}`;
279 const rehashes = this.shrinkCheck(minLoad, maxLoad);
280 return { deleted: true, rehashes, rows: [row, ...trace, ...removed.moves, ...rehashes] };
281 }
282 costs() {
283 const records = [];
284 if (this.mode === 'CHAIN') {
285 this.buckets.forEach((bucket, position) => bucket.forEach((item, index) => records.push({ key: item.key, cost: index + 1, position: String(position) })));
286 } else this.slots.forEach((item, position) => {
287 if (!isItem(item)) return;
288 const cost = this.mode === 'CUCKOO' ? (position < this.cap ? 1 : 2) : this.mode === 'ROBINHOOD' || this.mode === 'LINEAR' ? this.distance(item.key, position) + 1 : this.search(item.key).probes;
289 records.push({ key: item.key, cost, position: this.position(position) });
290 });
291 return records;
292 }
293 metrics() {
294 const costs = this.costs();
295 const longest = costs.reduce((value, item) => Math.max(value, item.cost), 0), total = costs.reduce((value, item) => value + item.cost, 0);
296 return { longest, average: decimal(total, this.size) };
297 }
298 dump() {
299 const rows = [`DUMP ${this.mode} cap=${this.cap} size=${this.size} tomb=${this.tomb}`];
300 if (this.mode === 'CHAIN') this.buckets.forEach((bucket, index) => rows.push(`${index} ${bucket.length ? bucket.map(pairText).join(' ') : '-'}`));
301 else this.slots.forEach((slot, index) => {
302 const position = this.mode === 'CUCKOO' ? `T${Math.floor(index / this.cap) + 1} ${index % this.cap}` : String(index);
303 let value = isItem(slot) ? pairText(slot) : content(slot);
304 if (this.mode === 'ROBINHOOD' && slot !== null) value += ` d=${this.distance(slot.key, index)}`;
305 rows.push(`${position} ${value}`);
306 });
307 return rows;
308 }
309 map() {
310 const rows = [`MAP ${this.mode} cap=${this.cap}`];
311 const costs = new Map(this.costs().map(item => [item.key, item.cost]));
312 const parts = this.mode === 'CUCKOO' ? [this.slots.slice(0, this.cap), this.slots.slice(this.cap)] : [this.mode === 'CHAIN' ? this.buckets : this.slots];
313 parts.forEach((part, number) => {
314 const characters = part.map(slot => {
315 if (this.mode === 'CHAIN') return slot.length === 0 ? '.' : gauge(slot.length);
316 if (slot === null) return '.';
317 if (slot === tombstone) return 'D';
318 return this.mode === 'CUCKOO' ? '#' : gauge(costs.get(slot.key) - 1);
319 });
320 for (let index = 0; index < characters.length; index += 40) {
321 const prefix = this.mode === 'CUCKOO' ? `T${number + 1} ` : '';
322 rows.push(`${prefix}${String(index).padStart(5, '0')} ${characters.slice(index, index + 40).join('')}`);
323 }
324 });
325 return rows;
326 }
327 statistics() {
328 const { longest, average } = this.metrics(), count = this.counters;
329 return [`STATS ${this.mode} size=${this.size} cap=${this.cap} slots=${this.slotCount()} tomb=${this.tomb} load=${decimal(100 * this.size, this.slotCount())}% longest=${longest} avg=${average} probes=${count.probes} rehashes=${count.rehashes}`,
330 `OPS inserts=${count.inserts} updates=${count.updates} hits=${count.hits} misses=${count.misses} deletes=${count.deletes} absent=${count.absent}`];
331 }
332 scan(cursor, limit) {
333 const keys = [];
334 let next = cursor;
335 while (next < this.slotCount() && keys.length < limit) {
336 if (this.mode === 'CHAIN') {
337 for (const item of this.buckets[next]) keys.push(item.key);
338 } else if (isItem(this.slots[next])) keys.push(this.slots[next].key);
339 ++next;
340 }
341 if (next === this.slotCount()) next = 0;
342 return [[`SCAN next=${next}`, ...keys].join(' ')];
343 }
344 clusters() {
345 const lengths = this.mode === 'CHAIN' ? this.buckets.filter(bucket => bucket.length > 0).map(bucket => bucket.length)
346 : this.mode === 'CUCKOO' ? [...runs(this.slots.slice(0, this.cap)), ...runs(this.slots.slice(this.cap))] : runs(this.slots);
347 const longest = lengths.reduce((maximum, length) => Math.max(maximum, length), 0);
348 return [`CLUSTERS runs=${lengths.length} longest=${longest}`, ...histogram(lengths)];
349 }
350 distribution() {
351 const costs = this.costs().map(item => item.cost), longest = costs.reduce((maximum, cost) => Math.max(maximum, cost), 0);
352 return [`DIST keys=${this.size} longest=${longest}`, ...histogram(costs)];
353 }
354}

361class Simulator {
362 constructor() { this.table = null; this.maxLoad = 75; this.minLoad = 0; this.tracing = false; }
363 compare(capacity) {
364 const items = this.table.items().slice().sort((a, b) => ascii(a.key, b.key)), rows = [`COMPARE keys=${items.length} cap=${capacity}`];
365 for (const mode of modes) {
366 const table = new Table(mode, capacity);
367 for (const item of items) table.put(item.key, item.value, 'PUT', this.maxLoad);
368 const { longest, average } = table.metrics();
369 rows.push(`${mode} cap=${table.cap} longest=${longest} avg=${average} probes=${table.counters.probes} rehashes=${table.counters.rehashes}`);
370 }
371 return rows;
372 }
373 bench(capacity, count, length, seed) {
374 const keys = generatedKeys(2 * count, length, seed), rows = [`BENCH cap=${capacity} count=${count} len=${length} seed=${seed}`];
375 for (const mode of modes) {
376 const table = new Table(mode, capacity);
377 for (let index = 0; index < count; ++index) table.put(keys[index], index + 1, 'PUT', this.maxLoad);
378 let misses = 0, probes = 0;
379 for (let index = count; index < keys.length; ++index) {
380 const found = table.search(keys[index]);
381 if (found.item === null) { ++misses; probes += found.probes; }
382 }
383 rows.push(`${mode} cap=${table.cap} keys=${table.size} hit=${table.metrics().average} misses=${misses} miss=${decimal(probes, misses)} rehashes=${table.counters.rehashes}`);
384 }
385 return rows;
386 }
387 execute(line) {
388 const [command, ...args] = line.split(' ');
389 if (!commands.has(command)) fail(`UNKNOWN COMMAND ${command}`);
390 if (command === 'CREATE') {
391 checkMode(args[0]);
392 const capacity = Number(args[1]);
393 checkCapacity(capacity);
394 this.table = new Table(args[0], prime(capacity));
395 return [`CREATE ${args[0]} cap=${this.table.cap}`];
396 }
397 if (command === 'MAXLOAD' || command === 'MINLOAD') {
398 const value = Number(args[0]), maximum = command === 'MAXLOAD';
399 if (maximum ? value < 10 || value > 95 : value > 40) fail('BAD LOAD');
400 if (maximum) this.maxLoad = value;
401 else this.minLoad = value;
402 return [`${command} ${value}`];
403 }
404 if (command === 'TRACE') {
405 if (args[0] !== 'ON' && args[0] !== 'OFF') fail('BAD ARGUMENT');
406 this.tracing = args[0] === 'ON';
407 return [`TRACE ${args[0]}`];
408 }
409 if (command === 'HASH') {
410 const key = args[0], capacity = Number(args[1]);
411 checkCapacity(capacity);
412 return [`HASH ${key} m=${capacity} h31=${hash(key, 31, capacity)} h37=${hash(key, 37, capacity)} step=${1 + hash(key, 37, capacity - 1)}`];
413 }
414 if (command === 'BENCH') {
415 const [capacity, count, length, seed] = args.map(Number);
416 checkCapacity(capacity);
417 checkGeneration(count, length, seed);
418 return this.bench(prime(capacity), count, length, seed);
419 }
420 const table = this.table;
421 if (table === null) fail('NO TABLE');
422 switch (command) {
423 case 'PUT':
424 case 'ADD': return table.put(args[0], Number(args[1]), command, this.maxLoad, this.tracing).rows;
425 case 'GET': return table.get(args[0], this.tracing);
426 case 'DEL': return table.delete(args[0], this.minLoad, this.maxLoad, this.tracing).rows;
427 case 'PROBE': {
428 const found = table.search(args[0], true);
429 return [[`PROBE ${args[0]}`, ...found.path, `${found.item === null ? 'MISS' : 'FOUND'} probes=${found.probes}`].join(' ')];
430 }
431 case 'GEN':
432 case 'GENDEL': {
433 const [count, length, seed] = args.map(Number);
434 checkGeneration(count, length, seed);
435 const keys = generatedKeys(count, length, seed), rows = [];
436 let successes = 0;
437 keys.forEach((key, index) => {
438 if (command === 'GEN') {
439 const result = table.put(key, index + 1, 'PUT', this.maxLoad);
440 successes += Number(result.fresh);
441 rows.push(...result.rehashes);
442 } else {
443 const result = table.delete(key, this.minLoad, this.maxLoad);
444 successes += Number(result.deleted);
445 rows.push(...result.rehashes);
446 }
447 });
448 rows.push(command === 'GEN' ? `GEN ${count} new=${successes} updated=${count - successes}` : `GENDEL ${count} deleted=${successes} missing=${count - successes}`);
449 return rows;
450 }
451 case 'RESIZE':
452 case 'CONVERT': {
453 let mode = table.mode, capacity = table.cap;
454 if (command === 'RESIZE') { checkCapacity(Number(args[0])); capacity = prime(Number(args[0])); }
455 else { checkMode(args[0]); mode = args[0]; }
456 const slots = capacity * (mode === 'CUCKOO' ? 2 : 1);
457 if (100 * table.size > this.maxLoad * slots) fail('TOO SMALL');
458 const replacement = Table.rebuild(table.items(), mode, capacity);
459 if (replacement === null) fail('REBUILD FAILED');
460 const row = command === 'RESIZE' ? `RESIZE ${table.cap} -> ${capacity}` : `CONVERT ${table.mode} -> ${mode} cap=${capacity}`;
461 table.adopt(replacement);
462 return [row];
463 }
464 case 'CLEAR': {
465 const size = table.size;
466 table.adopt(new Table(table.mode, table.cap));
467 return [`CLEAR ${size}`];
468 }
469 case 'DUMP': return table.dump();
470 case 'MAP': return table.map();
471 case 'STATS': return table.statistics();
472 case 'KEYS': return [[`KEYS ${table.size}`, ...table.items().map(item => item.key).sort(ascii)].join(' ')];
473 case 'TOP': {
474 const limit = Number(args[0]);
475 if (limit < 1 || limit > 100) fail('BAD ARGUMENT');
476 const items = table.costs().sort((a, b) => b.cost - a.cost || ascii(a.key, b.key)).slice(0, limit);
477 return [`TOP ${items.length}`, ...items.map(item => `${item.key} cost=${item.cost} at=${item.position}`)];
478 }
479 case 'SCAN': {
480 const [cursor, limit] = args.map(Number);
481 if (limit < 1 || limit > 100) fail('BAD ARGUMENT');
482 if (cursor >= table.slotCount()) fail('BAD CURSOR');
483 return table.scan(cursor, limit);
484 }
485 case 'CLUSTERS': return table.clusters();
486 case 'DIST': return table.distribution();
487 case 'COMPARE': {
488 const capacity = Number(args[0]);
489 checkCapacity(capacity);
