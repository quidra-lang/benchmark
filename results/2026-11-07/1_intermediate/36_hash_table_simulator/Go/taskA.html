100func hash(key string, base, modulus int) int {
101 value := 0
102 for i := range len(key) {
103 value = (value*base + int(key[i])) % modulus
104 }
105 return value
106}

108func prime(value int) bool {
109 if value < 2 {
110 return false
111 }
112 for divisor := 2; divisor <= value/divisor; divisor++ {
113 if value%divisor == 0 {
114 return false
115 }
116 }
117 return true
118}

120func nextPrime(value int) int {
121 value = max(value, 2)
122 for !prime(value) {
123 value++
124 }
125 return value
126}

143func histogram(values []int) []string {
144 frequencies := make(map[int]int)
145 for _, value := range values {
146 frequencies[value]++
147 }
148 keys := make([]int, 0, len(frequencies))
149 for value := range frequencies {
150 keys = append(keys, value)
151 }
152 slices.Sort(keys)
153 var lines []string
154 for _, value := range keys {
155 count := frequencies[value]
156 bar := strings.Repeat("#", min(count, 50))
157 if count > 50 {
158 bar += "+"
159 }
160 lines = append(lines, fmt.Sprintf("%d %d %s", value, count, bar))
161 }
162 return lines
163}

200func newTable(mode Mode, capacity int) *Table {
201 table := &Table{mode: mode, capacity: capacity}
202 if mode == chain {

204 } else {
205 table.slots = make([]Slot, table.slotCount())
206 }
207 return table

224func (table *Table) distance(item Item, index int) int {
225 return (index + table.capacity - hash(item.key, 31, table.capacity)) % table.capacity

244func (table *Table) search(key string, trace bool) Search {
245 result := Search{position: -1, available: -1}
246 observe := func(position int, content string) {
247 result.probes++
248 if trace {
249 result.path = append(result.path, table.position(position)+":"+content)
250 }
251 }
252 home := hash(key, 31, table.capacity)
253 if table.mode == chain {
254 for index, item := range table.buckets[home] {
255 observe(home, item.key)
256 if item.key == key {
257 result.found, result.position, result.chainIndex = true, home, index
258 break
259 }
260 }
261 return result
262 }
263 step, limit := 1, table.capacity
264 if table.mode == double {
265 step += hash(key, 37, table.capacity-1)
266 }
267 if table.mode == cuckoo {
268 limit = 2
269 }
270 for i := range limit {
271 position := (home + i*step) % table.capacity
272 if table.mode == quadratic {
273 position = (home + i*i) % table.capacity
274 } else if table.mode == cuckoo && i == 1 {
275 position = table.capacity + hash(key, 37, table.capacity)
276 }
277 slot := table.slots[position]
278 observe(position, slot.content())
279 if slot.state == live && slot.item.key == key {
280 result.found, result.position = true, position
281 break
282 }
283 if table.mode == cuckoo {
284 continue
285 }
286 if slot.state == empty {
287 if result.available < 0 {
288 result.available = position
289 }
290 break
291 }
292 if slot.state == tombstone && result.available < 0 {
293 result.available = position
294 }
295 if table.mode == robinHood && table.distance(slot.item, position) < i {
296 break
297 }
298 }
299 return result
300}

302func (table *Table) foundItem(search Search) Item {
303 if table.mode == chain {
304 return table.buckets[search.position][search.chainIndex]
305 }
306 return table.slots[search.position].item

309func (table *Table) place(item Item, initial *Search, trace bool) Placement {
310 result := Placement{position: -1}
311 home := hash(item.key, 31, table.capacity)
312 if table.mode == chain {
313 table.buckets[home] = append(table.buckets[home], item)
314 result.success, result.position = true, home
315 return result
316 }
317 if table.mode == linear || table.mode == quadratic || table.mode == double {
318 if initial == nil {
319 search := table.search(item.key, false)
320 initial = &search
321 }
322 position := initial.available
323 if position < 0 {
324 result.pending = item
325 return result
326 }
327 result.reused = table.slots[position].state == tombstone
328 if result.reused {
329 table.tomb--
330 }
331 table.slots[position] = Slot{state: live, item: item}
332 result.success, result.position = true, position
333 return result
334 }
335 carried := item
336 if table.mode == robinHood {
337 position, traveled, from := home, 0, -1
338 for {
339 if table.slots[position].state == empty {
340 if trace && from >= 0 {
341 result.moves = append(result.moves, fmt.Sprintf("SHIFT %s %d -> %d", carried.key, from, position))
342 }
343 table.slots[position] = Slot{state: live, item: carried}
344 if result.position < 0 {
345 result.position = position
346 }
347 result.success = true
348 return result
349 }
350 resident := table.slots[position].item
351 distance := table.distance(resident, position)
352 if distance < traveled {
353 if trace && from >= 0 {
354 result.moves = append(result.moves, fmt.Sprintf("SHIFT %s %d -> %d", carried.key, from, position))
355 }
356 table.slots[position].item, carried = carried, resident
357 if result.position < 0 {
358 result.position = position
359 }
360 from, traveled = position, distance
361 result.movements++
362 }
363 position = (position + 1) % table.capacity
364 traveled++
365 }
366 }
367 part := 0
368 for {
369 base := 31
370 if part == 1 {
371 base = 37
372 }
373 position := part*table.capacity + hash(carried.key, base, table.capacity)
374 if table.slots[position].state == empty {
375 table.slots[position] = Slot{state: live, item: carried}
376 result.success = true
377 result.position = table.search(item.key, false).position
378 return result
379 }
380 if result.movements == table.capacity {
381 result.pending = carried
382 return result
383 }
384 resident := table.slots[position].item
385 if trace {
386 result.moves = append(result.moves, fmt.Sprintf("KICK %s %s", resident.key, table.position(position)))
387 }
388 table.slots[position].item, carried = carried, resident
389 result.movements++
390 part = 1 - part
391 }
392}

394func (table *Table) remove(search Search, trace bool) (int, []string) {
395 shifted := 0
396 var moves []string
397 position := search.position
398 if table.mode == chain {
399 table.buckets[position] = slices.Delete(table.buckets[position], search.chainIndex, search.chainIndex+1)
400 } else if table.mode == linear || table.mode == quadratic || table.mode == double {
401 table.slots[position] = Slot{state: tombstone}
402 table.tomb++
403 } else {
404 table.slots[position] = Slot{}
405 if table.mode == robinHood {
406 source := (position + 1) % table.capacity
407 for table.slots[source].state == live && table.distance(table.slots[source].item, source) > 0 {
408 destination := (source + table.capacity - 1) % table.capacity
409 if trace {
410 moves = append(moves, fmt.Sprintf("SHIFT %s %d -> %d", table.slots[source].item.key, source, destination))
411 }
412 table.slots[destination], table.slots[source] = table.slots[source], Slot{}
413 shifted++
414 source = (source + 1) % table.capacity
415 }
416 }
417 }
418 table.size--
419 return shifted, moves
420}

422func rebuild(mode Mode, capacity int, items []Item) *Table {
423 trial := newTable(mode, capacity)
424 for _, item := range items {
425 if !trial.place(item, nil, false).success {
426 return nil
427 }
428 trial.size++
429 }
430 return trial
431}

433func (table *Table) adopt(replacement *Table) {
434 counters := table.counters
435 *table = *replacement
436 table.counters = counters
437}

439func (table *Table) automatic(capacity int, reason string, pending *Item) string {
440 old := table.capacity
441 items := table.items()
442 if pending != nil {
443 items = append(items, *pending)
444 }
445 for {
446 if trial := rebuild(table.mode, capacity, items); trial != nil {
447 table.adopt(trial)
448 table.counters.rehashes++
449 return fmt.Sprintf("REHASH %s %d -> %d", reason, old, capacity)
450 }
451 capacity = nextPrime(2 * capacity)
452 }
453}

455func (table *Table) loadCheck(maximum int) string {
456 threshold := int64(maximum) * int64(table.slotCount())
457 if int64(table.size+table.tomb)*100 <= threshold {
458 return ""
459 }
460 if int64(table.size)*200 > threshold {
461 return table.automatic(nextPrime(2*table.capacity), "load", nil)
462 }
463 return table.automatic(table.capacity, "tomb", nil)
464}

466func (table *Table) shrinkCheck(minimum, maximum int) string {
467 capacity := nextPrime(table.capacity / 2)
468 slots := capacity
469 if table.mode == cuckoo {
470 slots *= 2
471 }
472 if int64(table.size)*100 >= int64(minimum)*int64(table.slotCount()) || capacity >= table.capacity ||
473 int64(table.size)*100 > int64(maximum)*int64(slots) {
474 return ""
475 }
476 trial := rebuild(table.mode, capacity, table.items())
477 if trial == nil {
478 return ""
479 }
480 old := table.capacity
481 table.adopt(trial)
482 table.counters.rehashes++
483 return fmt.Sprintf("REHASH shrink %d -> %d", old, capacity)
484}

493func (table *Table) put(item Item, add bool, maximum int, reporting Reporting) (bool, []string) {
494 search := table.search(item.key, reporting.trace())
495 table.counters.probes += int64(search.probes)
496 command := "PUT"
497 if add {
498 command = "ADD"
499 }
500 var lines []string
501 if search.found {
502 old := table.foundItem(search).value
503 if add {
504 item.value += old
505 }
506 if table.mode == chain {
507 table.buckets[search.position][search.chainIndex] = item
508 } else {
509 table.slots[search.position].item = item
510 }
511 table.counters.updates++
512 if reporting.details() {
513 lines = append(lines, fmt.Sprintf("%s %s UPDATED old=%d at=%s probes=%d", command, item, old, table.position(search.position), search.probes))
514 }
515 if reporting.trace() {
516 lines = append(lines, pathLine(search))
517 }
518 return false, lines
519 }
520 placed := table.place(item, &search, reporting.trace())
521 table.counters.inserts++
522 if reporting.details() {
523 row := fmt.Sprintf("%s %s NEW", command, item)
524 if placed.success {
525 row += fmt.Sprintf(" at=%s probes=%d", table.position(placed.position), search.probes)
526 if placed.reused {
527 row += " reused"
528 }
529 if table.mode == robinHood {
530 row += fmt.Sprintf(" shifted=%d", placed.movements)
531 }
532 if table.mode == cuckoo {
533 row += fmt.Sprintf(" kicks=%d", placed.movements)
534 }
535 } else {
536 row += fmt.Sprintf(" probes=%d", search.probes)
537 if table.mode == quadratic {
538 row += " full"
539 } else {
540 row += fmt.Sprintf(" kicks=%d failed", placed.movements)
541 }
542 }
543 lines = append(lines, row)
544 }
545 if reporting.trace() {
546 lines = append(lines, pathLine(search))
547 lines = append(lines, placed.moves...)
548 }
549 if placed.success {
550 table.size++
551 } else {
552 reason := "full"
553 if table.mode == cuckoo {
554 reason = "kicks"
555 }
556 rehash := table.automatic(nextPrime(2*table.capacity), reason, &placed.pending)
557 if reporting != reportSilent {
558 lines = append(lines, rehash)
559 }
560 }
561 if rehash := table.loadCheck(maximum); rehash != "" && reporting != reportSilent {
562 lines = append(lines, rehash)
563 }
564 return true, lines
565}

567func (table *Table) delete(key string, minimum, maximum int, reporting Reporting) (bool, []string) {
568 search := table.search(key, reporting.trace())
569 table.counters.probes += int64(search.probes)
570 var lines, moves []string
571 if search.found {
572 shifted := 0
573 shifted, moves = table.remove(search, reporting.trace())
574 table.counters.deletes++
575 if reporting.details() {
576 row := fmt.Sprintf("DEL %s OK at=%s probes=%d", key, table.position(search.position), search.probes)
577 if table.mode == robinHood {
578 row += fmt.Sprintf(" shifted=%d", shifted)
579 }
580 lines = append(lines, row)
581 }
582 } else {
583 table.counters.absent++
584 if reporting.details() {
585 lines = append(lines, fmt.Sprintf("DEL %s NOT FOUND probes=%d", key, search.probes))
586 }
587 }
588 if reporting.trace() {
589 lines = append(lines, pathLine(search))
590 lines = append(lines, moves...)
591 }
592 if search.found {
593 if rehash := table.shrinkCheck(minimum, maximum); rehash != "" && reporting != reportSilent {
594 lines = append(lines, rehash)
595 }
596 }
597 return search.found, lines
598}

600func (table *Table) get(key string, trace bool) []string {
601 search := table.search(key, trace)
602 table.counters.probes += int64(search.probes)
603 var row string
604 if search.found {
605 table.counters.hits++
606 row = fmt.Sprintf("GET %s at=%s probes=%d", table.foundItem(search), table.position(search.position), search.probes)
607 } else {
608 table.counters.misses++
609 row = fmt.Sprintf("GET %s NOT FOUND probes=%d", key, search.probes)
610 }
611 lines := []string{row}
612 if trace {
613 lines = append(lines, pathLine(search))
614 }
615 return lines
616}

618func (table *Table) records() []Record {
619 records := make([]Record, 0, table.size)
620 if table.mode == chain {
621 for position, bucket := range table.buckets {
622 for index, item := range bucket {
623 records = append(records, Record{item, position, index + 1})
624 }
625 }
626 } else {
627 for position, slot := range table.slots {
628 if slot.state != live {
629 continue
630 }
631 cost := 0
632 switch table.mode {
633 case cuckoo:
634 cost = position/table.capacity + 1
635 case linear, robinHood:
636 cost = table.distance(slot.item, position) + 1
637 default:
638 cost = table.search(slot.item.key, false).probes
639 }
640 records = append(records, Record{slot.item, position, cost})
641 }
642 }
643 return records
644}

646func (table *Table) metrics() (int, string) {
647 longest, total := 0, int64(0)
648 for _, record := range table.records() {
649 longest = max(longest, record.cost)
650 total += int64(record.cost)
651 }
652 return longest, decimal(total, int64(table.size))
653}

655func (table *Table) dump() []string {
656 lines := []string{fmt.Sprintf("DUMP %s cap=%d size=%d tomb=%d", table.mode, table.capacity, table.size, table.tomb)}
657 if table.mode == chain {
658 for position, bucket := range table.buckets {
659 row := strconv.Itoa(position)
660 if len(bucket) == 0 {
661 row += " -"
662 }
663 for _, item := range bucket {
664 row += " " + item.String()
665 }
666 lines = append(lines, row)
667 }
668 } else {
669 for position, slot := range table.slots {
670 prefix := strconv.Itoa(position)
671 if table.mode == cuckoo {
672 prefix = fmt.Sprintf("T%d %d", position/table.capacity+1, position%table.capacity)
673 }
674 content := slot.content()
675 if slot.state == live {
676 content = slot.item.String()
677 if table.mode == robinHood {
678 content += fmt.Sprintf(" d=%d", table.distance(slot.item, position))
679 }
680 }
681 lines = append(lines, prefix+" "+content)
682 }
683 }
684 return lines
685}

687func (table *Table) mapReport() []string {
688 lines := []string{fmt.Sprintf("MAP %s cap=%d", table.mode, table.capacity)}
689 costs := make([]int, table.slotCount())
690 if table.mode != chain {
691 for _, record := range table.records() {
692 costs[record.position] = record.cost
693 }
694 }
695 parts := 1
696 if table.mode == cuckoo {
697 parts = 2
698 }
699 for part := range parts {
700 for start := 0; start < table.capacity; start += 40 {
701 text := make([]byte, min(40, table.capacity-start))
702 for j := range text {
703 index := start + j
704 if table.mode == chain {
705 text[j] = '.'
706 if len(table.buckets[index]) > 0 {
707 text[j] = gauge(len(table.buckets[index]))
708 }
709 } else {
710 position := part*table.capacity + index
711 switch table.slots[position].state {
712 case empty:
713 text[j] = '.'
714 case tombstone:
715 text[j] = 'D'
716 case live:
717 text[j] = '#'
718 if table.mode != cuckoo {
719 text[j] = gauge(costs[position] - 1)
720 }
721 }
722 }
723 }
724 prefix := ""
725 if table.mode == cuckoo {
726 prefix = fmt.Sprintf("T%d ", part+1)
727 }
728 lines = append(lines, fmt.Sprintf("%s%05d %s", prefix, start, text))
729 }
730 }
731 return lines
732}

744func (table *Table) scan(cursor, limit int) string {
745 var keys []string
746 for cursor < table.slotCount() && len(keys) < limit {
747 if table.mode == chain {
748 for _, item := range table.buckets[cursor] {
749 keys = append(keys, item.key)
750 }
751 } else if table.slots[cursor].state == live {
752 keys = append(keys, table.slots[cursor].item.key)
753 }
754 cursor++
755 }
756 if cursor == table.slotCount() {
757 cursor = 0
758 }
759 row := fmt.Sprintf("SCAN next=%d", cursor)
760 if len(keys) > 0 {
761 row += " " + strings.Join(keys, " ")
762 }
763 return row
764}

766func (table *Table) clusters() []string {
767 var runs []int
768 if table.mode == chain {
769 for _, bucket := range table.buckets {
770 if len(bucket) > 0 {
771 runs = append(runs, len(bucket))
772 }
773 }
774 } else {
775 for base := 0; base < len(table.slots); base += table.capacity {
776 firstEmpty := -1
777 for i := range table.capacity {
778 if table.slots[base+i].state == empty {
779 firstEmpty = i
780 break
781 }
782 }
783 if firstEmpty < 0 {
784 runs = append(runs, table.capacity)
785 continue
786 }
787 length := 0
788 for offset := 1; offset <= table.capacity; offset++ {
789 if table.slots[base+(firstEmpty+offset)%table.capacity].state != empty {
790 length++
791 } else if length > 0 {
792 runs = append(runs, length)
793 length = 0
794 }
795 }
796 }
797 }
798 longest := 0
799 for _, length := range runs {
800 longest = max(longest, length)
801 }
802 return append([]string{fmt.Sprintf("CLUSTERS runs=%d longest=%d", len(runs), longest)}, histogram(runs)...)
803}

811 }

815func compareModes(table *Table, capacity, maximum int) []string {
816 items := table.items()
817 slices.SortFunc(items, func(a, b Item) int { return strings.Compare(a.key, b.key) })
818 lines := []string{fmt.Sprintf("COMPARE keys=%d cap=%d", len(items), capacity)}
819 for mode := chain; mode <= cuckoo; mode++ {
820 trial := newTable(mode, capacity)
821 for _, item := range items {
822 trial.put(item, false, maximum, reportSilent)
823 }
824 longest, average := trial.metrics()
825 lines = append(lines, fmt.Sprintf("%s cap=%d longest=%d avg=%s probes=%d rehashes=%d", mode, trial.capacity, longest, average, trial.counters.probes, trial.counters.rehashes))
826 }
827 return lines
828}

830func benchmark(capacity, count, length int, seed int64, maximum int) []string {
831 keys := generate(2*count, length, seed)
832 lines := []string{fmt.Sprintf("BENCH cap=%d count=%d len=%d seed=%d", capacity, count, length, seed)}
833 for mode := chain; mode <= cuckoo; mode++ {
834 trial := newTable(mode, capacity)
835 for i, key := range keys[:count] {
836 trial.put(Item{key, int64(i + 1)}, false, maximum, reportSilent)
837 }
838 misses, probes := int64(0), int64(0)
839 for _, key := range keys[count:] {
840 search := trial.search(key, false)
841 if !search.found {
842 misses++
843 probes += int64(search.probes)
844 }
845 }
846 _, average := trial.metrics()
847 lines = append(lines, fmt.Sprintf("%s cap=%d keys=%d hit=%s misses=%d miss=%s rehashes=%d", mode, trial.capacity, trial.size, average, misses, decimal(probes, misses), trial.counters.rehashes))
848 }
849 return lines
850}

852type Simulator struct {
853 table *Table
854 maximum, minimum int
855 trace bool
856}
857
858func (sim *Simulator) execute(line string) ([]string, error) {
859 words := strings.Split(line, " ")
860 command := words[0]
861 number := func(index int) int64 {
862 value, _ := strconv.ParseInt(words[index], 10, 64)
863 return value
864 }
865 switch command {
866 case "CREATE", "MAXLOAD", "MINLOAD", "TRACE", "HASH", "PUT", "ADD", "GET", "DEL", "PROBE", "GEN", "GENDEL",
867 "RESIZE", "CONVERT", "CLEAR", "DUMP", "MAP", "STATS", "KEYS", "TOP", "SCAN", "CLUSTERS", "DIST", "COMPARE", "BENCH":
868 default:
869 return nil, fmt.Errorf("UNKNOWN COMMAND %s", command)
870 }
871 switch command {
872 case "CREATE":
873 mode, err := parseMode(words[1])
874 if err != nil {
875 return nil, err
876 }
877 capacity, err := checkCapacity(number(2))
878 if err != nil {
879 return nil, err
880 }
881 sim.table = newTable(mode, nextPrime(capacity))
882 return []string{fmt.Sprintf("CREATE %s cap=%d", mode, sim.table.capacity)}, nil
883 case "MAXLOAD", "MINLOAD":
884 value := number(1)
885 if command == "MAXLOAD" && (value < 10 || value > 95) || command == "MINLOAD" && value > 40 {
886 return nil, fmt.Errorf("BAD LOAD")
887 }
888 if command == "MAXLOAD" {
889 sim.maximum = int(value)
890 } else {
891 sim.minimum = int(value)
892 }
893 return []string{fmt.Sprintf("%s %d", command, value)}, nil
894 case "TRACE":
895 if words[1] != "ON" && words[1] != "OFF" {
896 return nil, fmt.Errorf("BAD ARGUMENT")
897 }
898 sim.trace = words[1] == "ON"
899 return []string{"TRACE " + words[1]}, nil
900 case "HASH":
901 capacity, err := checkCapacity(number(2))
902 if err != nil {
903 return nil, err
904 }
905 key := words[1]
906 return []string{fmt.Sprintf("HASH %s m=%d h31=%d h37=%d step=%d", key, capacity, hash(key, 31, capacity), hash(key, 37, capacity), 1+hash(key, 37, capacity-1))}, nil
907 case "BENCH":
908 capacity, err := checkCapacity(number(1))
909 if err != nil {
910 return nil, err
911 }
912 count, length, seed := number(2), number(3), number(4)
913 if err := checkGeneration(count, length, seed); err != nil {
914 return nil, err
915 }
916 return benchmark(nextPrime(capacity), int(count), int(length), seed, sim.maximum), nil
917 }
918 table := sim.table
919 if table == nil {
920 return nil, fmt.Errorf("NO TABLE")
921 }
922 reporting := reportFull
923 if sim.trace {
924 reporting = reportTrace
925 }
926 switch command {
927 case "PUT", "ADD":
928 _, lines := table.put(Item{words[1], number(2)}, command == "ADD", sim.maximum, reporting)
929 return lines, nil
930 case "GET":
931 return table.get(words[1], sim.trace), nil
932 case "DEL":
933 _, lines := table.delete(words[1], sim.minimum, sim.maximum, reporting)
934 return lines, nil
935 case "PROBE":
936 search := table.search(words[1], true)
937 row := "PROBE " + words[1]
938 if len(search.path) > 0 {
939 row += " " + strings.Join(search.path, " ")
940 }
941 outcome := "MISS"
942 if search.found {
943 outcome = "FOUND"
944 }
945 return []string{fmt.Sprintf("%s %s probes=%d", row, outcome, search.probes)}, nil
946 case "GEN", "GENDEL":
947 count, length, seed := number(1), number(2), number(3)
948 if err := checkGeneration(count, length, seed); err != nil {
949 return nil, err
950 }
951 var lines []string
952 changed := int64(0)
953 for i, key := range generate(int(count), int(length), seed) {
954 var didChange bool
955 var output []string
956 if command == "GEN" {
957 didChange, output = table.put(Item{key, int64(i + 1)}, false, sim.maximum, reportRehashes)
958 } else {
959 didChange, output = table.delete(key, sim.minimum, sim.maximum, reportRehashes)
960 }
961 if didChange {
962 changed++
963 }
964 lines = append(lines, output...)
965 }
966 if command == "GEN" {
967 lines = append(lines, fmt.Sprintf("GEN %d new=%d updated=%d", count, changed, count-changed))
968 } else {
969 lines = append(lines, fmt.Sprintf("GENDEL %d deleted=%d missing=%d", count, changed, count-changed))
970 }
971 return lines, nil
972 case "RESIZE", "CONVERT":
973 capacity, mode := table.capacity, table.mode
974 var err error
975 if command == "RESIZE" {
976 capacity, err = checkCapacity(number(1))
977 if err == nil {
978 capacity = nextPrime(capacity)
979 }
980 } else {
981 mode, err = parseMode(words[1])
982 }
983 if err != nil {
984 return nil, err
985 }
986 slots := capacity
987 if mode == cuckoo {
988 slots *= 2
989 }
990 if int64(table.size)*100 > int64(sim.maximum)*int64(slots) {
991 return nil, fmt.Errorf("TOO SMALL")
992 }
993 trial := rebuild(mode, capacity, table.items())
994 if trial == nil {
995 return nil, fmt.Errorf("REBUILD FAILED")
996 }
997 row := fmt.Sprintf("RESIZE %d -> %d", table.capacity, capacity)
998 if command == "CONVERT" {
999 row = fmt.Sprintf("CONVERT %s -> %s cap=%d", table.mode, mode, capacity)
1000 }
1001 table.adopt(trial)
1002 return []string{row}, nil
1003 case "CLEAR":
1004 size := table.size
1005 table.adopt(newTable(table.mode, table.capacity))
1006 return []string{fmt.Sprintf("CLEAR %d", size)}, nil
1007 case "DUMP":
1008 return table.dump(), nil
1009 case "MAP":
1010 return table.mapReport(), nil
1011 case "STATS":
1012 return table.statistics(), nil
1013 case "KEYS":
1014 keys := make([]string, 0, table.size)
1015 for _, item := range table.items() {
1016 keys = append(keys, item.key)
1017 }
1018 slices.Sort(keys)
1019 row := fmt.Sprintf("KEYS %d", len(keys))
1020 if len(keys) > 0 {
1021 row += " " + strings.Join(keys, " ")
1022 }
1023 return []string{row}, nil
1024 case "TOP":
1025 limit := number(1)
1026 if limit < 1 || limit > 100 {
1027 return nil, fmt.Errorf("BAD ARGUMENT")
1028 }
1029 records := table.records()
1030 slices.SortFunc(records, func(a, b Record) int {
1031 if a.cost != b.cost {
1032 return b.cost - a.cost
1033 }
1034 return strings.Compare(a.item.key, b.item.key)
1035 })
1036 records = records[:min(len(records), int(limit))]
1037 lines := []string{fmt.Sprintf("TOP %d", len(records))}
1038 for _, record := range records {
1039 lines = append(lines, fmt.Sprintf("%s cost=%d at=%s", record.item.key, record.cost, table.position(record.position)))
1040 }
1041 return lines, nil
1042 case "SCAN":
1043 cursor, limit := number(1), number(2)
1044 if limit < 1 || limit > 100 {
1045 return nil, fmt.Errorf("BAD ARGUMENT")
1046 }
1047 if cursor >= int64(table.slotCount()) {
1048 return nil, fmt.Errorf("BAD CURSOR")
1049 }
1050 return []string{table.scan(int(cursor), int(limit))}, nil
1051 case "CLUSTERS":
1052 return table.clusters(), nil
1053 case "DIST":
1054 return table.distribution(), nil
1055 case "COMPARE":
1056 capacity, err := checkCapacity(number(1))
1057 if err != nil {
1058 return nil, err
1059 }
1060 return compareModes(table, nextPrime(capacity), sim.maximum), nil
1061 }
1062 panic("unreachable command")
1063}

1065func readLine(input *bufio.Reader) string {
1066 line, _ := input.ReadString('\n')
1067 return strings.TrimSuffix(strings.TrimSuffix(line, "\n"), "\r")
1068}

1070func main() {
1071 input := bufio.NewReader(os.Stdin)
1072 output := bufio.NewWriter(os.Stdout)
1073 defer output.Flush()
1074 count, _ := strconv.Atoi(readLine(input))
1075 sim := Simulator{maximum: 75}
1076 for range count {
1077 lines, err := sim.execute(readLine(input))
1078 if err != nil {
1079 fmt.Fprintln(output, "ERROR", err)
1080 } else {
1081 for _, line := range lines {
1082 fmt.Fprintln(output, line)
1083 }
1084 }
1085 }
