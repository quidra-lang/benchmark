1using System;

8class Program {
9 record Cue(long Index, long Start, long End, List<string> Text) {
10 public long Duration() => End-Start;
11 public long Chars() => Text.Sum(t => (long)t.Length);
12 public bool Dialogue() => Text.Any(t => t.StartsWith("- ",StringComparison.Ordinal));
13 }
14 record Undo(string Command, List<Cue> Cues);
15 record Item(string Text, int Kind, int Digit);
16 class Problem : Exception {
17 public Problem(string message) : base(message) {}
18 }
19 List<Cue> cues = new();
20 readonly Stack<Undo> history = new();
21 static readonly Dictionary<string,string> Schemas = new(StringComparer.Ordinal) {
22 ["LIST"] = "",
23 ["VALIDATE"] = "",
24 ["SEARCH"] = "X",
25 ["AT"] = "T",
26 ["GAPS"] = "g",
27 ["STATS"] = "",
28 ["PRINT"] = "",
29 ["TOVTT"] = "",
30 ["TOASS"] = "",
31 ["FRAMES"] = "F",
32 ["SHIFT"] = "I",
33 ["SHIFTFROM"] = "PI",
34 ["SCALE"] = "rr",
35 ["SYNC"] = "PTPT",
36 ["SNAP"] = "F",
37 ["FIXOVERLAP"] = "g",
38 ["EXTEND"] = "e",
39 ["TIME"] = "PTT",
40 ["SORT"] = "",
41 ["RENUMBER"] = "",
42 ["MERGE"] = "g",
43 ["SPLIT"] = "w",
44 ["INSERT"] = "PTTX",
45 ["DELETE"] = "P",
46 ["TEXT"] = "PX",
47 ["BALANCE"] = "",
48 ["STRIP"] = "",
49 ["CLEANHI"] = "",
50 ["REPLACE"] = "X",
51 ["UNDO"] = "",
52 };
53 static readonly HashSet<string> Modify = new(StringComparer.Ordinal) {
54 "SHIFT","SHIFTFROM","SCALE","SYNC","SNAP","FIXOVERLAP","EXTEND","TIME","SORT",
55 "RENUMBER","MERGE","SPLIT","INSERT","DELETE","TEXT","BALANCE","STRIP","CLEANHI","REPLACE"
56 };
57 static long Floor(long a, long b) {
58 long q = a/b;
59 return a%b < 0 ? q-1 : q;
60 }
61 static void Bad() { throw new Problem("bad arguments"); }
62 static long Number(string s) {
63 if (!Regex.IsMatch(s,@"\A[0-9]{1,9}\z")) Bad();
64 return long.Parse(s);
65 }
66 static long Time(string s) {
67 if (!Regex.IsMatch(s,@"\A[0-9]{2}:[0-5][0-9]:[0-5][0-9],[0-9]{3}\z")) Bad();
68 return long.Parse(s.Substring(0,2))*3600000+long.Parse(s.Substring(3,2))*60000+
69 long.Parse(s.Substring(6,2))*1000+long.Parse(s.Substring(9));
70 }
71 static long[] Fps(string s) {
72 string[] pieces = s.Split('/');
73 if (pieces.Length > 2) Bad();
74 long a = Number(pieces[0]), b = pieces.Length == 2 ? Number(pieces[1]) : 1;
75 if (a < 1 || a > 100000 || b < 1 || b > 100000) Bad();
76 return new long[]{a,b};
77 }
78 static List<string> Text(string s) {
79 List<string> lines = s.Split('|').ToList();
80 if (lines.Any(x => x.Length == 0)) Bad();
81 return lines;
82 }
83 static string Srt(long t) => $"{t/3600000:D2}:{t/60000%60:D2}:{t/1000%60:D2},{t%1000:D3}";
84 static string Ass(long t) {
85 long c = (t+5)/10;
86 return $"{c/360000}:{c/6000%60:D2}:{c/100%60:D2}.{c%100:D2}";
87 }
88 static string Decimal(long v) => $"{v/100}.{v%100:D2}";
89 static long Round(long a, long b) {
90 if (b < 0) {
91 a = -a;
92 b = -b;
93 }
94 return Floor(2*a+b,2*b);
95 }
96 static List<string> Words(List<string> lines) {
97 return string.Join(" ",lines).Split(' ',StringSplitOptions.RemoveEmptyEntries).ToList();
98 }
99 static List<string> Wrap(List<string> lines, long width) {
100 List<string> result = new();
101 foreach (string word in Words(lines)) {
102 if (result.Count == 0 || result.Last().Length+1+word.Length > width) {
103 result.Add(word);
104 } else {
105 result[result.Count-1] = result.Last()+" "+word;
106 }
107 }
108 return result;
109 }
110 static string Normalize(string s) { return string.Join(" ",Words(new List<string> {s})); }
111 static readonly Regex MarkupTokens = new(@"</?[ibu]>|</font>|<font(?=[ >])[^>]*>|\{\\an[1-9]\}|.");
112 static List<Item> Markup(string line) => MarkupTokens.Matches(line).Select(match => {
113 string text = match.Value;
114 int kind = text.StartsWith("{\\an",StringComparison.Ordinal) ? 3 :
115 text.StartsWith("<font",StringComparison.Ordinal) || text == "</font>" ? 2 : text.Length > 1 ? 1 : 0;
116 return new Item(text,kind,kind == 3 ? text[4]-'0' : 0);
117 }).ToList();
118 static string Listing(int p, Cue c) {
119 return p+" "+c.Index+" "+Srt(c.Start)+" "+Srt(c.End)+" "+string.Join("|",c.Text);
120 }
121 void Load(List<List<string>> blocks, List<string> output) {
122 int bad = 0;
123 for (int i = 0; i < blocks.Count; i++) {
124 List<string> block = blocks[i];
125 string reason = "";
126 long start = 0, end = 0;
127 if (!Regex.IsMatch(block.First(),@"\A[0-9]{1,6}\z")) reason = "index";
128 else {
129 try {
130 if (block.Count < 2 || block[1].Length != 29 || !block[1].Substring(12,5).Equals(" --> ")) Bad();
131 start = Time(block[1].Substring(0,12));
132 end = Time(block[1].Substring(17));
133 } catch (Problem) {
134 reason = "time";
135 }
136 if (reason.Length == 0 && block.Count < 3) reason = "no text";
137 }
138 if (reason.Length != 0) {
139 output.Add("BAD CUE "+(i+1)+": "+reason);
140 bad++;
141 } else {
142 cues.Add(new Cue(long.Parse(block.First()),start,end,block.Skip(2).ToList()));
143 }
144 }
145 output.Add("LOAD cues="+cues.Count+" bad="+bad);
146 }
147 List<string> Execute(string line) {
148 string command = line.Split(new[]{' '},2,StringSplitOptions.None)[0];
149 if (!Schemas.ContainsKey(command)) throw new Problem("unknown command");
150 string schema = Schemas[command];
151 if (command.Equals("DELETE") && line.Split(' ').Length == 3) schema = "PP";
152 string[] tokens = schema.EndsWith("X",StringComparison.Ordinal) ?
153 line.Split(new[]{' '},schema.Length+1,StringSplitOptions.None) : line.Split(' ');
154 if (tokens.Length != schema.Length+1) Bad();
155 string[] args = tokens.Skip(1).ToArray();
156 long[] values = new long[args.Length];
157 for (int i = 0; i < args.Length; i++) {
158 char type = schema[i];
159 if (type == 'T') values[i] = Time(args[i]);
160 else if (type == 'F') Fps(args[i]);
161 else if (type == 'X') {
162 if (command.Equals("INSERT") || command.Equals("TEXT")) Text(args[i]);
163 if (command.Equals("REPLACE")) {
164 string[] parts = args[i].Split('|');
165 if (parts.Length != 2 || parts[0].Length == 0 || parts[1].Length == 0) Bad();
166 }
167 } else {
168 string raw = args[i];
169 bool negative = type == 'I' && raw.StartsWith("-");
170 values[i] = Number(negative ? raw.Substring(1) : raw)*(negative ? -1 : 1);
171 if (type == 'P' && values[i] < 1 || type == 'r' && (values[i] < 1 || values[i] > 100000) ||
172 type == 'g' && values[i] > 60000 || type == 'e' && (values[i] < 1 || values[i] > 60000) ||
173 type == 'w' && (values[i] < 10 || values[i] > 80)) Bad();
174 }
175 }
176 if (command.Equals("DELETE") && values.Length == 2 && values[1] < values[0]) Bad();
177 for (int i = 0; i < args.Length; i++) {
178 if (schema[i] == 'P' && values[i] > cues.Count+(command.Equals("INSERT") ? 1 : 0)) {
179 throw new Problem("no cue "+values[i]);
180 }
181 }
182 if (command.Equals("SYNC") && cues[(int)values[0]-1].Start == cues[(int)values[2]-1].Start) {
183 throw new Problem("sync points coincide");
184 }
185 if (command.Equals("UNDO")) {
186 if (history.Count == 0) throw new Problem("nothing to undo");
187 Undo undo = history.Pop();
188 cues = undo.Cues;
189 return new List<string> {"UNDO command="+undo.Command};
190 }
191 if (Modify.Contains(command)) history.Push(new Undo(command,new List<Cue>(cues)));
192 List<string> output = new();
193 int n = cues.Count;
194 long v = values.Length == 0 ? 0 : values[0];
195 switch (command) {
196 case "LIST":
197 case "AT":
198 case "SEARCH":
199 case "GAPS": {
200 int count = 0;
201 for (int i = 0; i < n; i++) {
202 Cue cue = cues[i];
203 if (command.Equals("LIST") || command.Equals("AT") && cue.Start <= v && v < cue.End) {
204 output.Add(Listing(i+1,cue));
205 count++;
206 }
207 if (command.Equals("SEARCH")) {
208 foreach (string t in cue.Text) {
209 if (t.Contains(args[0])) {
210 output.Add((i+1)+" "+Srt(cue.Start)+" "+t);
211 count++;
212 break;
213 }
214 }
215 }
216 if (command.Equals("GAPS") && i+1 < n && cues[i+1].Start-cue.End < v) {
217 output.Add((i+1)+" "+(i+2)+" "+(cues[i+1].Start-cue.End));
218 count++;
219 }
220 }
221 if (!command.Equals("LIST")) output.Add(command+(command == "AT" ? " active=" : command == "SEARCH" ? " found=" : " count=")+count);
222 break;
223 }
224 case "VALIDATE": {
225 for (int i = 0; i < n; i++) {
226 Cue c = cues[i];
227 long d = c.Duration();
228 string p = "cue "+(i+1)+": ";
229 if (c.Index != i+1) output.Add(p+"index "+c.Index+" (expected "+(i+1)+")");
230 if (d <= 0) output.Add(p+"end not after start");
231 if (i > 0 && c.Start < cues[i-1].End) output.Add(p+"overlaps previous by "+(cues[i-1].End-c.Start)+" ms");
232 if (d >= 1 && d < 700) output.Add(p+"too short ("+d+" ms)");
233 if (d > 7000) output.Add(p+"too long ("+d+" ms)");
234 for (int j = 0; j < c.Text.Count; j++) {
235 int size = c.Text[j].Length;
236 if (size > 42) output.Add(p+"line "+(j+1)+" too long ("+size+" chars)");
237 }
238 if (c.Text.Count >= 3) output.Add(p+"too many lines ("+c.Text.Count+")");
239 if (d >= 1 && c.Chars()*1000 > 21*d) output.Add(p+"too fast ("+Decimal(c.Chars()*100000/d)+" cps)");
240 }
241 output.Add("VALIDATE problems="+output.Count);
242 break;
243 }
244 case "STATS": {
245 output.Add("STATS cues="+n);
246 if (n == 0) break;
247 var durations = cues.Select(c => Math.Max(0,c.Duration())).ToArray();
248 long total = durations.Sum(), longest = durations.Max();
249 long chars = cues.Where(c => c.Duration() > 0).Sum(c => c.Chars());
250 output.Add("duration total="+Srt(total)+" mean="+Srt(total/n));
251 output.Add("speed "+(total == 0 ? "-" : Decimal(chars*100000/total)));
252 output.Add("longest cue="+(Array.IndexOf(durations,longest)+1)+" duration="+Srt(longest));
253 if (n == 1) output.Add("gap none");
254 else {
255 var gaps = Enumerable.Range(0,n-1).Select(i => cues[i+1].Start-cues[i].End).ToArray();
256 long gap = gaps.Max();
257 output.Add("gap cue="+(Array.IndexOf(gaps,gap)+1)+" ms="+gap);
258 }
259 break;
260 }
261 case "PRINT":
262 case "TOVTT":
263 case "TOASS": {
264 if (command.Equals("TOVTT")) output.AddRange(new List<string> {"WEBVTT",""});
265 if (command.Equals("TOASS")) output.AddRange(new List<string> {"[Events]","Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text"});
266 foreach (Cue c in cues) {
267 List<string> lines = new();
268 int code = 0;
269 foreach (string t in c.Text) {
270 StringBuilder converted = new StringBuilder();
271 foreach (Item item in Markup(t)) {
272 if (code == 0 && item.Kind == 3) code = item.Digit;
273 if (command.Equals("PRINT")) converted.Append(item.Text);
274 else if (command.Equals("TOASS")) {
275 if (item.Kind == 1) {
276 char tag = item.Text[item.Text.StartsWith("</") ? 2 : 1];
277 converted.Append("{\\").Append(tag).Append(item.Text.StartsWith("</") ? "0}" : "1}");
278 } else if (item.Kind != 2) converted.Append(item.Text);
279 } else if (item.Kind == 1) converted.Append(item.Text);
280 else if (item.Kind == 0) converted.Append(item.Text switch {
281 "&" => "&amp;",
282 "<" => "&lt;",
283 ">" => "&gt;",
284 _ => item.Text
285 });
286 }
287 string convertedText = converted.ToString();
288 if (command.Equals("TOVTT")) convertedText = convertedText.Trim(' ');
289 if (!command.Equals("TOVTT") || convertedText.Length != 0) lines.Add(convertedText);
290 }
291 if (command.Equals("TOASS")) {
292 output.Add("Dialogue: 0,"+Ass(c.Start)+","+Ass(c.End)+",Default,,0,0,0,,"+string.Join("\\N",lines));
293 } else if (lines.Count != 0) {
294 string settings = "";
295 if (command.Equals("TOVTT") && code != 0) {
296 if (code >= 7) settings += " line:0";
297 else if (code >= 4) settings += " line:50%";
298 if (code%3 == 1) settings += " align:start";
299 if (code%3 == 0) settings += " align:end";
300 }
301 output.Add(Convert.ToString(c.Index));
302 string times = Srt(c.Start)+" --> "+Srt(c.End);
303 output.Add((command.Equals("TOVTT") ? times.Replace(',','.') : times)+settings);
304 output.AddRange(lines);
305 output.Add("");
306 }
307 }
308 break;
309 }
310 case "FRAMES": {
311 long[] rate = Fps(args[0]);
312 for (int i = 0; i < n; i++) {
313 Cue c = cues[i];
314 output.Add((i+1)+" "+c.Start*rate[0]/(1000*rate[1])+" "+c.End*rate[0]/(1000*rate[1]));
315 }
316 break;
317 }
318 case "SHIFT":
319 case "SHIFTFROM":
320 case "SCALE":
321 case "SYNC":
322 case "SNAP": {
323 int first = command.Equals("SHIFTFROM") ? (int)v-1 : 0;
324 long delta = command.Equals("SHIFTFROM") ? values[1] : v;
325 long s1 = command.Equals("SYNC") ? cues[(int)v-1].Start : 0;
326 long s2 = command.Equals("SYNC") ? cues[(int)values[2]-1].Start : 0;
327 long[] rate = command.Equals("SNAP") ? Fps(args[0]) : new long[]{1,1};
328 int clamped = 0, changed = 0;
329 for (int i = first; i < n; i++) {
330 Cue c = cues[i];
331 long[] times = {c.Start,c.End};
332 bool clamp = false;
333 for (int j = 0; j < 2; j++) {
334 long old = times[j], next;
335 if (command.Equals("SCALE")) next = Round(old*v,values[1]);
336 else if (command.Equals("SYNC")) next = Round(values[1]*(s2-s1)+(old-s1)*(values[3]-values[1]),s2-s1);
337 else if (command.Equals("SNAP")) next = Round(Round(old*rate[0],1000*rate[1])*1000*rate[1],rate[0]);
338 else next = old+delta;
339 if (next < 0) clamp = true;
340 times[j] = Math.Max(0,next);
341 if (times[j] != old) changed++;
342 }
343 if (clamp) clamped++;
344 cues[i] = new Cue(c.Index,times[0],times[1],c.Text);
345 }
346 if (command.Equals("SNAP")) output.Add("SNAP changed="+changed);
347 else output.Add(command+" cues="+(n-first)+(command.Equals("SCALE") ? "" : " clamped="+clamped));
348 break;
349 }
350 case "FIXOVERLAP":
351 case "EXTEND": {
352 int changed = 0, failed = 0;
353 for (int i = 0; i < n; i++) {
354 Cue c = cues[i];
355 long end = c.End;
356 if (command.Equals("FIXOVERLAP") && i+1 < n && end > cues[i+1].Start-v) {
357 long candidate = cues[i+1].Start-v;
358 if (candidate > c.Start) {
359 end = candidate;
360 changed++;
361 } else {
362 output.Add("cue "+(i+1)+": cannot fix");
363 failed++;
364 }
365 }
366 if (command.Equals("EXTEND") && c.Duration() < v) {
367 long candidate = c.Start+v;
368 if (i+1 < n) candidate = Math.Min(candidate,cues[i+1].Start);
369 if (candidate > end) {
370 end = candidate;
371 changed++;
372 }
373 }
374 cues[i] = new Cue(c.Index,c.Start,end,c.Text);
375 }
376 if (command.Equals("FIXOVERLAP")) output.Add("FIXOVERLAP fixed="+changed+" failed="+failed);
377 else output.Add("EXTEND extended="+changed+" short="+cues.Count(c => c.Duration() < v));
378 break;
379 }
380 case "TIME": {
381 Cue c = cues[(int)v-1];
382 cues[(int)v-1] = new Cue(c.Index,values[1],values[2],c.Text);
383 output.Add("TIME cue="+v);
384 break;
385 }
386 case "SORT": {
387 List<Cue> old = new List<Cue>(cues);
388 cues = cues.OrderBy(c => c.Start).ThenBy(c => c.End).ToList();
389 int moved = 0;
390 for (int i = 0; i < n; i++) {
391 if (!ReferenceEquals(old[i],cues[i])) moved++;
392 }
393 output.Add("SORT moved="+moved);
394 break;
395 }
396 case "RENUMBER": {
397 int changed = 0;
398 for (int i = 0; i < n; i++) {
399 Cue c = cues[i];
400 if (c.Index != i+1) changed++;
401 cues[i] = new Cue(i+1,c.Start,c.End,c.Text);
402 }
403 output.Add("RENUMBER changed="+changed);
404 break;
405 }
406 case "MERGE": {
407 List<Cue> result = new();
408 int merged = 0;
409 foreach (Cue b in cues) {
410 if (result.Count != 0) {
411 Cue a = result.Last();
412 List<string> wrapped = Wrap(a.Text.Concat(b.Text).ToList(),42);
413 if (b.Start-a.End < v && !a.Dialogue() && !b.Dialogue() && wrapped.Count <= 2) {
414 result[result.Count-1] = new Cue(a.Index,a.Start,Math.Max(a.End,b.End),wrapped);
415 merged++;
416 continue;
417 }
418 }
419 result.Add(b);
420 }
421 cues = result;
422 output.Add("MERGE merged="+merged+" cues="+cues.Count);
423 break;
424 }
425 case "SPLIT": {
426 List<Cue> result = new();
427 int split = 0;
428 foreach (Cue c in cues) {
429 if (c.Duration() <= 0 || c.Dialogue()) {
430 result.Add(c);
431 continue;
432 }
433 List<string> wrapped = Wrap(c.Text,v);
434 if (wrapped.Count <= 2) {
435 result.Add(new Cue(c.Index,c.Start,c.End,wrapped));
436 continue;
437 }
438 split++;
439 var parts = wrapped.Chunk(2).Select(part => part.ToList()).ToList();
440 long total = parts.Sum(part => (long)part.Sum(t => t.Length));
441 long[] durations = new long[parts.Count], remainders = new long[parts.Count];
442 List<int> order = new();
443 long allocated = 0;
444 for (int i = 0; i < parts.Count; i++) {
445 long count = parts[i].Sum(t => (long)t.Length);
446 durations[i] = c.Duration()*count/total;
447 remainders[i] = c.Duration()*count%total;
448 allocated += durations[i];
449 order.Add(i);
450 }
451 order = order.OrderByDescending(i => remainders[i]).ThenBy(i => i).ToList();
452 for (int i = 0; i < c.Duration()-allocated; i++) durations[order[i]]++;
453 long start = c.Start;
454 for (int i = 0; i < parts.Count; i++) {
455 result.Add(new Cue(c.Index,start,start+durations[i],parts[i]));
456 start += durations[i];
457 }
458 }
459 cues = result;
460 output.Add("SPLIT split="+split+" cues="+cues.Count);
461 break;
462 }
463 case "INSERT": {
464 cues.Insert((int)v-1,new Cue(v,values[1],values[2],Text(args[3])));
465 output.Add("INSERT cue="+v+" cues="+cues.Count);
466 break;
467 }
468 case "DELETE": {
469 long last = values.Length == 2 ? values[1] : v;
470 cues.RemoveRange((int)v-1,(int)(last-v+1));
471 output.Add("DELETE removed="+(last-v+1)+" cues="+cues.Count);
472 break;
473 }
474 case "TEXT": {
475 Cue c = cues[(int)v-1];
476 List<string> lines = Text(args[1]);
477 cues[(int)v-1] = new Cue(c.Index,c.Start,c.End,lines);
478 output.Add("TEXT cue="+v+" lines="+lines.Count);
479 break;
480 }
481 case "BALANCE": {
482 int changed = 0, skipped = 0, dialogue = 0;
483 for (int i = 0; i < n; i++) {
484 Cue c = cues[i];
485 if (c.Dialogue()) {
486 dialogue++;
487 continue;
488 }
489 List<string> tokens2 = Words(c.Text);
490 string joined = string.Join(" ",tokens2);
491 List<string> best = joined.Length <= 42 ? new List<string> {joined} : null;
492 int difference = int.MaxValue, firstLength = int.MaxValue;
493 for (int j = 1; joined.Length > 42 && j < tokens2.Count; j++) {
494 string a = string.Join(" ",tokens2.Take(j)), b = string.Join(" ",tokens2.Skip(j));
495 int d = Math.Abs(a.Length-b.Length);
496 if (a.Length <= 42 && b.Length <= 42 && (d < difference || d == difference && a.Length < firstLength)) {
497 best = new List<string> {a,b};
498 difference = d;
499 firstLength = a.Length;
500 }
501 }
502 if (best == null) skipped++;
503 else if (!best.SequenceEqual(c.Text)) {
504 changed++;
505 cues[i] = new Cue(c.Index,c.Start,c.End,best);
506 }
507 }
508 output.Add("BALANCE changed="+changed+" skipped="+skipped+" dialogue="+dialogue);
509 break;
510 }
511 case "STRIP":
512 case "CLEANHI": {
513 List<Cue> result = new();
514 int tags = 0, segments = 0, labels = 0, removed = 0;
515 foreach (Cue c in cues) {
516 List<string> lines = new();
517 foreach (string t in c.Text) {
518 StringBuilder builder = new StringBuilder();
519 if (command.Equals("STRIP")) {
520 foreach (Item item in Markup(t)) {
521 if (item.Kind == 0) builder.Append(item.Text);
522 else tags++;
523 }
524 } else {
525 int p = 0;
526 while (p < t.Length) {
527 char ch = t[p];
528 int end = ch == '[' || ch == '(' ? t.IndexOf(ch == '[' ? ']' : ')',p+1) : -1;
529 if (end >= 0) {
530 segments++;
531 p = end+1;
532 } else {
533 builder.Append(ch);
534 p++;
535 }
536 }
537 }
538 string cleaned = command.Equals("STRIP") ? builder.ToString().Trim(' ') : Normalize(builder.ToString());
539 if (command.Equals("CLEANHI")) {
540 string prefix = cleaned.StartsWith("- ") ? "- " : "";
541 string body = cleaned.Substring(prefix.Length);
542 int colon = body.IndexOf(':');
543 if (colon > 0 && Regex.IsMatch(body.Substring(0,colon),@"\A[A-Z][A-Z0-9 ]*\z")) {
544 labels++;
545 cleaned = Normalize(prefix+body.Substring(colon+1));
546 }
547 }
548 if (cleaned.Length == 0 || command.Equals("CLEANHI") && cleaned.Equals("-")) removed++;
549 else lines.Add(cleaned);
550 }
551 if (lines.Count != 0) result.Add(new Cue(c.Index,c.Start,c.End,lines));
552 }
553 int removedCues = n-result.Count;
554 cues = result;
555 output.Add(command.Equals("STRIP") ? "STRIP tags="+tags+" removed_lines="+removed+" removed_cues="+removedCues :
556 "CLEANHI segments="+segments+" labels="+labels+" removed_lines="+removed+" removed_cues="+removedCues);
557 break;
558 }
559 case "REPLACE": {
560 string[] parts = args[0].Split('|');
561 string old = parts[0], replacement = parts[1];
562 int count = 0, changed = 0;
563 for (int i = 0; i < n; i++) {
564 Cue c = cues[i];
565 List<string> lines = new();
566 int before = count;
567 foreach (string t in c.Text) {
568 string[] pieces = t.Split(new[] { old },StringSplitOptions.None);
569 count += pieces.Length-1;
570 lines.Add(string.Join(replacement,pieces));
571 }
572 if (count > before) changed++;
573 cues[i] = new Cue(c.Index,c.Start,c.End,lines);
574 }
575 output.Add("REPLACE count="+count+" cues="+changed);
576 break;
577 }
578 }
579 return output;
580 }
