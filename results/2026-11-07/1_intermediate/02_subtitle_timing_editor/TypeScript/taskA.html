9function bad(): never {
10 throw new Invalid("bad arguments");
11}

13function floorRatio(a: bigint, b: bigint): bigint {
14 if (b < 0n) {
15 a = -a;
16 b = -b;
17 }
18 const q = a / b;
19 return a < 0n && a % b !== 0n ? q - 1n : q;
20}
21
22function rounded(a: bigint, b: bigint): number {
23 if (b < 0n) {
24 a = -a;
25 b = -b;
26 }
27 return Number(floorRatio(2n * a + b, 2n * b));
28}
29
30function quotient(a: number, b: number): number {
31 return Number(floorRatio(BigInt(a), BigInt(b)));
32}
33
34function pad(value: number, width: number): string {
35 return String(value).padStart(width, "0");
36}

38function srt(time: number): string {
39 return pad(quotient(time, 3600000), 2) + ":" + pad(quotient(time, 60000) % 60, 2) +
40 ":" + pad(quotient(time, 1000) % 60, 2) + "," + pad(time % 1000, 3);
41}
42
43function ass(time: number): string {
44 const c = quotient(time + 5, 10);
45 return quotient(c, 360000) + ":" + pad(quotient(c, 6000) % 60, 2) +
46 ":" + pad(quotient(c, 100) % 60, 2) + "." + pad(c % 100, 2);
47}
48
49function decimal(value: number): string {
50 return quotient(value, 100) + "." + pad(value % 100, 2);
51}
52
53function parseTime(value: string): number {
54 if (!/^\d{2}:[0-5]\d:[0-5]\d,\d{3}$/.test(value)) {
55 bad();
56 }
57 return Number(value.slice(0, 2)) * 3600000 + Number(value.slice(3, 5)) * 60000 +
58 Number(value.slice(6, 8)) * 1000 + Number(value.slice(9));
59}
60
61function integer(value: string, minimum: number = 0, maximum: number = 999999999): number {
62 if (!/^\d{1,9}$/.test(value)) {
63 bad();
64 }
65 const result = Number(value);
66 if (result < minimum || result > maximum) {
67 bad();
68 }
69 return result;
70}
71
72function signed(value: string): number {
73 if (!/^-?\d{1,9}$/.test(value)) {
74 bad();
75 }
76 return Number(value);
77}
78
79function fps(value: string): number[] {
80 const parts = value.split("/");
81 if (parts.length > 2) {
82 bad();
83 }
84 return [integer(parts[0], 1, 100000), parts.length === 2 ? integer(parts[1], 1, 100000) : 1];
85}
86
87function textLines(value: string): string[] {
88 const parts = value.split("|");
89 if (parts.some(part => part.length === 0)) {
90 bad();
91 }
92 return parts;
93}

95function chars(cue: Cue): number {
96 return cue.text.reduce((sum, line) => sum + line.length, 0);
97}
98
99function dialogue(cue: Cue): boolean {
100 return cue.text.some(line => line.startsWith("- "));
101}

103function words(lines: string[]): string[] {
104 return lines.join(" ").split(" ").filter(Boolean);
105}
106
107function wrap(lines: string[], width: number): string[] {
108 const result: string[] = [];
109 for (const word of words(lines)) {
110 if (result.length && result[result.length - 1].length + 1 + word.length <= width) {
111 result[result.length - 1] += " " + word;
112 } else {
113 result.push(word);
114 }
115 }
116 return result;
117}

119function markup(line: string): Token[] {
120 const result: Token[] = [];
121 const formats = ["<i>", "</i>", "<b>", "</b>", "<u>", "</u>"];
122 let p = 0;
123 while (p < line.length) {
124 const format = formats.find(tag => line.startsWith(tag, p));
125 if (format) {
126 result.push({ kind: "format", text: format, digit: 0 });
127 p += format.length;
128 continue;
129 }
130 let font = "";
131 if (line.startsWith("</font>", p)) {
132 font = "</font>";
133 } else if (line.startsWith("<font", p) && [" ", ">"].includes(line[p + 5])) {
134 const end = line.indexOf(">", p + 5);
135 if (end >= 0) {
136 font = line.slice(p, end + 1);
137 }
138 }
139 if (font) {
140 result.push({ kind: "font", text: font, digit: 0 });
141 p += font.length;
142 continue;
143 }
144 const position = /^\{\\an([1-9])\}/.exec(line.slice(p));
145 if (position) {
146 result.push({ kind: "position", text: position[0], digit: Number(position[1]) });
147 p += position[0].length;
148 } else {
149 result.push({ kind: "char", text: line[p], digit: 0 });
150 p++;
151 }
152 }
153 return result;
154}

156function listing(cue: Cue, position: number): string {
157 return position + " " + cue.index + " " + srt(cue.start) + " " + srt(cue.end) + " " + cue.text.join("|");

177 load(lines: string[]): string[] {
178 const blocks: string[][] = [];
179 let current: string[] = [];
180 for (const line of [...lines, ""]) {
181 if (line) {
182 current.push(line);
183 } else if (current.length) {
184 blocks.push(current);
185 current = [];
186 }
187 }
188 const output: string[] = [];
189 for (let i = 0; i < blocks.length; i++) {
190 const block = blocks[i];
191 let reason = "";
192 let start = 0;
193 let end = 0;
194 if (!/^\d{1,6}$/.test(block[0])) {
195 reason = "index";
196 } else if (block.length < 2 || !/^\d{2}:[0-5]\d:[0-5]\d,\d{3} --> \d{2}:[0-5]\d:[0-5]\d,\d{3}$/.test(block[1])) {
197 reason = "time";
198 } else {
199 start = parseTime(block[1].slice(0, 12));
200 end = parseTime(block[1].slice(17));
201 if (block.length < 3) {
202 reason = "no text";
203 }
204 }
205 if (reason) {
206 output.push("BAD CUE " + (i + 1) + ": " + reason);
207 } else {
208 this.cues.push({ index: Number(block[0]), start, end, text: block.slice(2) });
209 }
210 }
211 output.push("LOAD cues=" + this.cues.length + " bad=" + (blocks.length - this.cues.length));
212 return output;
213 }

215 execute(line: string): string[] {
216 const name = line.split(" ", 1)[0];
217 if (!Object.hasOwn(arities, name)) {
218 throw new Invalid("unknown command");
219 }
220 const fixed: Record<string, number> = { SEARCH: 0, INSERT: 3, TEXT: 1, REPLACE: 0 };
221 let args: string[];
222 if (Object.hasOwn(fixed, name)) {
223 args = [];
224 let p = name.length + 1;
225 if (line.length <= name.length) {
226 bad();
227 }
228 for (let i = 0; i < fixed[name]; i++) {
229 const next = line.indexOf(" ", p);
230 if (next < 0) {
231 bad();
232 }
233 args.push(line.slice(p, next));
234 p = next + 1;
235 }
236 args.push(line.slice(p));
237 } else {
238 args = line.split(" ").slice(1);
239 }
240 if (!arities[name].includes(args.length)) {
241 bad();
242 }
243 const values: number[] = [];
244 let texts: string[] = [];
245 let rate = [1, 1];
246 let old = "";
247 let replacement = "";
248 const positions: number[] = [];
249 const position = (value: string): number => {
250 const p = integer(value, 1);
251 positions.push(p);
252 return p;
253 };
254 switch (name) {
255 case "AT":
256 values.push(parseTime(args[0]));
257 break;
258 case "GAPS": case "FIXOVERLAP": case "MERGE":
259 values.push(integer(args[0], 0, 60000));
260 break;
261 case "EXTEND":
262 values.push(integer(args[0], 1, 60000));
263 break;
264 case "SPLIT":
265 values.push(integer(args[0], 10, 80));
266 break;
267 case "SHIFT":
268 values.push(signed(args[0]));
269 break;
270 case "SHIFTFROM":
271 values.push(position(args[0]), signed(args[1]));
272 break;
273 case "SCALE":
274 values.push(integer(args[0], 1, 100000), integer(args[1], 1, 100000));
275 break;
276 case "SYNC":
277 values.push(position(args[0]), parseTime(args[1]), position(args[2]), parseTime(args[3]));
278 break;
279 case "TIME": case "INSERT":
280 values.push(position(args[0]), parseTime(args[1]), parseTime(args[2]));
281 if (name === "INSERT") {
282 texts = textLines(args[3]);
283 }
284 break;
285 case "DELETE":
286 values.push(...args.map(position));
287 if (values.length === 2 && values[1] < values[0]) {
288 bad();
289 }
290 break;
291 case "TEXT":
292 values.push(position(args[0]));
293 texts = textLines(args[1]);
294 break;
295 case "FRAMES": case "SNAP":
296 rate = fps(args[0]);
297 break;
298 case "REPLACE": {
299 const parts = args[0].split("|");
300 if (parts.length !== 2 || parts.some(part => !part)) {
301 bad();
302 }
303 [old, replacement] = parts;
304 break;
305 }
306 }
307 for (const p of positions) {
308 if (p > this.cues.length + (name === "INSERT" ? 1 : 0)) {
309 throw new Invalid("no cue " + p);
310 }
311 }
312 if (name === "SYNC" && this.cues[values[0] - 1].start === this.cues[values[2] - 1].start) {
313 throw new Invalid("sync points coincide");
314 }
315 if (name === "UNDO") {
316 const record = this.history.pop();
317 if (!record) {
318 throw new Invalid("nothing to undo");
319 }
320 this.cues = record.cues;
321 return ["UNDO command=" + record.name];
322 }
323 if (modifying.has(name)) {
324 this.history.push({ name, cues: structuredClone(this.cues) });
325 }
326 return this.process(name, args, values, texts, rate, old, replacement);
327 }
328
329 process(name: string, args: string[], v: number[], texts: string[], rate: number[],
330 old: string, replacement: string): string[] {
331 const output: string[] = [];
332 const n = this.cues.length;
333 if (name === "LIST" || name === "AT") {
334 this.cues.forEach((cue, i) => {
335 if (name === "LIST" || (cue.start <= v[0] && v[0] < cue.end)) {
336 output.push(listing(cue, i + 1));
337 }
338 });
339 if (name === "AT") {
340 output.push("AT active=" + output.length);
341 }
342 } else if (name === "VALIDATE") {
343 this.cues.forEach((cue, i) => {
344 const p = i + 1;
345 const prefix = "cue " + p + ": ";
346 const d = cue.end - cue.start;
347 if (cue.index !== p) {
348 output.push(prefix + "index " + cue.index + " (expected " + p + ")");
349 }
350 if (d <= 0) {
351 output.push(prefix + "end not after start");
352 }
353 if (i > 0 && cue.start < this.cues[i - 1].end) {
354 output.push(prefix + "overlaps previous by " + (this.cues[i - 1].end - cue.start) + " ms");
355 }
356 if (d >= 1 && d < 700) {
357 output.push(prefix + "too short (" + d + " ms)");
358 } else if (d > 7000) {
359 output.push(prefix + "too long (" + d + " ms)");
360 }
361 cue.text.forEach((text, j) => {
362 if (text.length > 42) {
363 output.push(prefix + "line " + (j + 1) + " too long (" + text.length + " chars)");
364 }
365 });
366 if (cue.text.length >= 3) {
367 output.push(prefix + "too many lines (" + cue.text.length + ")");
368 }
369 const c = chars(cue);
370 if (d >= 1 && c * 1000 > 21 * d) {
371 output.push(prefix + "too fast (" + decimal(quotient(c * 100000, d)) + " cps)");
372 }
373 });
374 output.push("VALIDATE problems=" + output.length);
375 } else if (name === "SEARCH") {
376 this.cues.forEach((cue, i) => {
377 const match = cue.text.find(text => text.includes(args[0]));
378 if (match !== undefined) {
379 output.push((i + 1) + " " + srt(cue.start) + " " + match);
380 }
381 });
382 output.push("SEARCH found=" + output.length);
383 } else if (name === "GAPS") {
384 for (let i = 0; i + 1 < n; i++) {
385 const gap = this.cues[i + 1].start - this.cues[i].end;
386 if (gap < v[0]) {
387 output.push((i + 1) + " " + (i + 2) + " " + gap);
388 }
389 }
390 output.push("GAPS count=" + output.length);
391 } else if (name === "STATS") {
392 output.push("STATS cues=" + n);
393 if (n) {
394 const durations = this.cues.map(cue => Math.max(0, cue.end - cue.start));
395 const total = durations.reduce((a, b) => a + b, 0);
396 const c = this.cues.reduce((sum, cue, i) => sum + (durations[i] > 0 ? chars(cue) : 0), 0);
397 output.push("duration total=" + srt(total) + " mean=" + srt(quotient(total, n)));
398 output.push("speed " + (total ? decimal(quotient(c * 100000, total)) : "-"));
399 const maximum = Math.max(...durations);
400 output.push("longest cue=" + (durations.indexOf(maximum) + 1) + " duration=" + srt(maximum));
401 if (n === 1) {
402 output.push("gap none");
403 } else {
404 const gaps = this.cues.slice(1).map((cue, i) => cue.start - this.cues[i].end);
405 const maximumGap = Math.max(...gaps);
406 output.push("gap cue=" + (gaps.indexOf(maximumGap) + 1) + " ms=" + maximumGap);
407 }
408 }
409 } else if (name === "PRINT") {
410 for (const cue of this.cues) {
411 output.push(String(cue.index), srt(cue.start) + " --> " + srt(cue.end), ...cue.text, "");
412 }
413 } else if (name === "TOVTT") {
414 output.push("WEBVTT", "");
415 const escapes: Record<string, string> = { "&": "&amp;", "<": "&lt;", ">": "&gt;" };
416 for (const cue of this.cues) {
417 let digit = 0;
418 const converted = cue.text.map(line => {
419 let result = "";
420 for (const token of markup(line)) {
421 if (token.kind === "position" && !digit) {
422 digit = token.digit;
423 }
424 if (token.kind === "format") {
425 result += token.text;
426 } else if (token.kind === "char") {
427 result += escapes[token.text] ?? token.text;
428 }
429 }
430 return result.trim();
431 }).filter(Boolean);
432 if (!converted.length) {
433 continue;
434 }
435 let settings = digit >= 7 ? " line:0" : digit >= 4 ? " line:50%" : "";
436 if ([1, 4, 7].includes(digit)) {
437 settings += " align:start";
438 } else if ([3, 6, 9].includes(digit)) {
439 settings += " align:end";
440 }
441 output.push(String(cue.index), srt(cue.start).replace(",", ".") + " --> " +
442 srt(cue.end).replace(",", ".") + settings, ...converted, "");
443 }
444 } else if (name === "TOASS") {
445 output.push("[Events]", "Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text");
446 const tags: Record<string, string> = {
447 "<i>": "{\\i1}", "</i>": "{\\i0}", "<b>": "{\\b1}", "</b>": "{\\b0}",
448 "<u>": "{\\u1}", "</u>": "{\\u0}"
449 };
450 for (const cue of this.cues) {
451 const converted = cue.text.map(line => markup(line).map(token =>
452 token.kind === "format" ? tags[token.text] : token.kind === "font" ? "" : token.text).join(""));
453 output.push("Dialogue: 0," + ass(cue.start) + "," + ass(cue.end) + ",Default,,0,0,0,," + converted.join("\\N"));
454 }
455 } else if (name === "FRAMES") {
456 this.cues.forEach((cue, i) => {
457 const frames = [cue.start, cue.end].map(t =>
458 floorRatio(BigInt(t) * BigInt(rate[0]), 1000n * BigInt(rate[1])));
459 output.push((i + 1) + " " + frames[0] + " " + frames[1]);
460 });
461 } else if (["SHIFT", "SHIFTFROM", "SCALE", "SYNC", "SNAP"].includes(name)) {
462 let clamped = 0;
463 let changed = 0;
464 const first = name === "SHIFTFROM" ? v[0] - 1 : 0;
465 const shift = name === "SHIFTFROM" ? v[1] : v[0];
466 const s1 = name === "SYNC" ? this.cues[v[0] - 1].start : 0;
467 const s2 = name === "SYNC" ? this.cues[v[2] - 1].start : 0;
468 for (let i = first; i < n; i++) {
469 const cue = this.cues[i];
470 let wasClamped = false;
471 for (const key of ["start", "end"] as const) {
472 const t = cue[key];
473 let result = t;
474 if (name === "SHIFT" || name === "SHIFTFROM") {
475 result = t + shift;
476 } else if (name === "SCALE") {
477 result = rounded(BigInt(t) * BigInt(v[0]), BigInt(v[1]));
478 } else if (name === "SYNC") {
479 const den = BigInt(s2 - s1);
480 const num = BigInt(v[1]) * den + BigInt(t - s1) * BigInt(v[3] - v[1]);
481 result = rounded(num, den);
482 } else {
483 const frame = rounded(BigInt(t) * BigInt(rate[0]), 1000n * BigInt(rate[1]));
484 result = rounded(BigInt(frame) * 1000n * BigInt(rate[1]), BigInt(rate[0]));
485 }
486 if (result < 0) {
487 wasClamped = true;
488 result = 0;
489 }
490 if (result !== t) {
491 changed++;
492 }
493 cue[key] = result;
494 }
495 if (wasClamped) {
496 clamped++;
497 }
498 }
499 if (name === "SNAP") {
500 output.push("SNAP changed=" + changed);
501 } else if (name === "SCALE") {
502 output.push("SCALE cues=" + n);
503 } else {
504 output.push(name + " cues=" + (n - first) + " clamped=" + clamped);
505 }
506 } else if (name === "FIXOVERLAP") {
507 let fixed = 0;
508 let failed = 0;
509 for (let i = 0; i + 1 < n; i++) {
510 const cue = this.cues[i];
511 const limit = this.cues[i + 1].start - v[0];
512 if (cue.end > limit) {
513 if (limit > cue.start) {
514 cue.end = limit;
515 fixed++;
516 } else {
517 failed++;
518 output.push("cue " + (i + 1) + ": cannot fix");
519 }
520 }
521 }
522 output.push("FIXOVERLAP fixed=" + fixed + " failed=" + failed);
523 } else if (name === "EXTEND") {
524 let extended = 0;
525 for (let i = 0; i < n; i++) {
526 const cue = this.cues[i];
527 if (cue.end - cue.start < v[0]) {
528 const candidate = Math.min(cue.start + v[0], this.cues[i + 1]?.start ?? Infinity);
529 if (candidate > cue.end) {
530 cue.end = candidate;
531 extended++;
532 }
533 }
534 }
535 output.push("EXTEND extended=" + extended + " short=" + this.cues.filter(cue => cue.end - cue.start < v[0]).length);
536 } else if (name === "TIME") {
537 this.cues[v[0] - 1].start = v[1];
538 this.cues[v[0] - 1].end = v[2];
539 output.push("TIME cue=" + v[0]);
540 } else if (name === "SORT") {
541 const sorted = this.cues.map((cue, i) => ({ cue, i }));
542 sorted.sort((a, b) => a.cue.start - b.cue.start || a.cue.end - b.cue.end || a.i - b.i);
543 output.push("SORT moved=" + sorted.filter((entry, i) => entry.i !== i).length);
544 this.cues = sorted.map(entry => entry.cue);
545 } else if (name === "RENUMBER") {
546 let changed = 0;
547 this.cues.forEach((cue, i) => {
548 if (cue.index !== i + 1) {
549 cue.index = i + 1;
550 changed++;
551 }
552 });
553 output.push("RENUMBER changed=" + changed);
554 } else if (name === "MERGE") {
555 const merged: Cue[] = [];
556 let joins = 0;
557 for (const cue of this.cues) {
558 const previous = merged[merged.length - 1];
559 const lines = previous ? wrap([...previous.text, ...cue.text], 42) : [];
560 if (previous && cue.start - previous.end < v[0] && !dialogue(previous) && !dialogue(cue) && lines.length <= 2) {
561 previous.end = Math.max(previous.end, cue.end);
562 previous.text = lines;
563 joins++;
564 } else {
565 merged.push(cue);
566 }
567 }
568 this.cues = merged;
569 output.push("MERGE merged=" + joins + " cues=" + merged.length);
570 } else if (name === "SPLIT") {
571 const result: Cue[] = [];
572 let split = 0;
573 for (const cue of this.cues) {
574 const duration = cue.end - cue.start;
575 if (duration <= 0 || dialogue(cue)) {
576 result.push(cue);
577 continue;
578 }
579 const lines = wrap(cue.text, v[0]);
580 if (lines.length <= 2) {
581 cue.text = lines;
582 result.push(cue);
583 continue;
584 }
585 split++;
586 const parts: string[][] = [];
587 for (let i = 0; i < lines.length; i += 2) {
588 parts.push(lines.slice(i, i + 2));
589 }
590 const counts = parts.map(part => part.reduce((sum, line) => sum + line.length, 0));
591 const total = counts.reduce((a, b) => a + b, 0);
592 const times = counts.map(c => quotient(duration * c, total));
593 const order = counts.map((c, i) => ({ i, remainder: BigInt(duration) * BigInt(c) % BigInt(total) }));
594 order.sort((a, b) => a.remainder === b.remainder ? a.i - b.i : a.remainder > b.remainder ? -1 : 1);
595 const remaining = duration - times.reduce((a, b) => a + b, 0);
596 for (let i = 0; i < remaining; i++) {
597 times[order[i].i]++;
598 }
599 let start = cue.start;
600 parts.forEach((text, i) => {
601 result.push({ index: cue.index, start, end: start + times[i], text });
602 start += times[i];
603 });
604 }
605 this.cues = result;
606 output.push("SPLIT split=" + split + " cues=" + result.length);
607 } else if (name === "INSERT") {
608 this.cues.splice(v[0] - 1, 0, { index: v[0], start: v[1], end: v[2], text: texts });
609 output.push("INSERT cue=" + v[0] + " cues=" + this.cues.length);
610 } else if (name === "DELETE") {
611 const count = (v[1] ?? v[0]) - v[0] + 1;
612 this.cues.splice(v[0] - 1, count);
613 output.push("DELETE removed=" + count + " cues=" + this.cues.length);
614 } else if (name === "TEXT") {
615 this.cues[v[0] - 1].text = texts;
616 output.push("TEXT cue=" + v[0] + " lines=" + texts.length);
617 } else if (name === "BALANCE") {
618 let changed = 0;
619 let skipped = 0;
620 let dialogues = 0;
621 for (const cue of this.cues) {
622 if (dialogue(cue)) {
623 dialogues++;
624 continue;
625 }

637 candidate[0].length < best[0].length)) {

643 skipped++;
644 } else if (JSON.stringify(best) !== JSON.stringify(cue.text)) {

646 changed++;
647 }
648 }
649 output.push("BALANCE changed=" + changed + " skipped=" + skipped + " dialogue=" + dialogues);
650 } else if (name === "STRIP" || name === "CLEANHI") {
651 let tags = 0;

679 labels++;

681 }

684 deletedLines++;
685 } else {
686 lines.push(line);
687 }

690 cue.text = lines;
691 result.push(cue);

696 output.push(name === "STRIP" ? "STRIP tags=" + tags + suffix :
697 "CLEANHI segments=" + segments + " labels=" + labels + suffix);
