21function floor_div(int $n, int $d): int
22{
23 $q = intdiv($n, $d);
24 return $n < 0 && $n % $d !== 0 ? $q - 1 : $q;
25}

27function round_ratio(int $n, int $d): int
28{
29 return floor_div(2 * $n + $d, 2 * $d);
30}

52function time_value(string $text): int
53{
54 if (preg_match('/^([0-9]{2}):([0-9]{2}):([0-9]{2}),([0-9]{3})$/D', $text, $m) !== 1) {
55 fail('bad arguments');
56 }
57 if ((int) $m[2] > 59 || (int) $m[3] > 59) {
58 fail('bad arguments');
59 }
60 return (int) $m[1] * 3600000 + (int) $m[2] * 60000 + (int) $m[3] * 1000 + (int) $m[4];
61}

63function time_text(int $t, string $kind = 'SRT'): string
64{
65 if ($kind === 'ASS') {
66 $c = intdiv($t + 5, 10);
67 return sprintf('%d:%02d:%02d.%02d', intdiv($c, 360000), intdiv($c, 6000) % 60, intdiv($c, 100) % 60, $c % 100);
68 }
69 return sprintf('%02d:%02d:%02d%s%03d', intdiv($t, 3600000), intdiv($t, 60000) % 60, intdiv($t, 1000) % 60,
70 $kind === 'VTT' ? '.' : ',', $t % 1000);
71}

73function unsigned_arg(string $text, int $low = 0, int $high = 999999999): int
74{
75 if (preg_match('/^[0-9]{1,9}$/D', $text) !== 1) {
76 fail('bad arguments');
77 }
78 $value = (int) $text;
79 if ($value < $low || $value > $high) {
80 fail('bad arguments');
81 }
82 return $value;
83}

85function typed_arg(string $text, string $kind): mixed
86{
87 if ($kind === 'string') {
88 return $text;
89 }
90 if ($kind === 'time') {
91 return time_value($text);
92 }
93 if ($kind === 'pos') {
94 return unsigned_arg($text, 1);
95 }
96 if ($kind === 'signed') {
97 if (preg_match('/^-?[0-9]{1,9}$/D', $text) !== 1) {
98 fail('bad arguments');
99 }
100 return (int) $text;
101 }
102 if ($kind === 'fps') {
103 $parts = explode('/', $text);
104 if (count($parts) > 2) {
105 fail('bad arguments');
106 }
107 return [unsigned_arg($parts[0], 1, 100000), count($parts) === 1 ? 1 : unsigned_arg($parts[1], 1, 100000)];
108 }
109 if ($kind === 'text' || $kind === 'replace') {
110 $parts = explode('|', $text);
111 if (in_array('', $parts, true) || ($kind === 'replace' && count($parts) !== 2)) {
112 fail('bad arguments');
113 }
114 return $parts;
115 }
116 [$low, $high] = array_map('intval', explode(':', $kind));
117 return unsigned_arg($text, $low, $high);
118}

120function arguments(string $line, array $cues): array
121{
122 $command = explode(' ', $line, 2)[0];
123 if (!array_key_exists($command, SCHEMAS)) {
124 fail('unknown command');
125 }
126 if ($command === 'DELETE') {
127 $raw = array_slice(explode(' ', $line), 1);
128 if (count($raw) < 1 || count($raw) > 2) {
129 fail('bad arguments');
130 }
131 $args = array_map(fn($text) => unsigned_arg($text, 1), $raw);
132 if (count($args) === 2 && $args[1] < $args[0]) {
133 fail('bad arguments');
134 }
135 $kinds = array_fill(0, count($args), 'pos');
136 } else {
137 $kinds = SCHEMAS[$command] === '' ? [] : explode(' ', SCHEMAS[$command]);
138 $last = $kinds[count($kinds) - 1] ?? '';
139 $parts = in_array($last, ['string', 'text', 'replace'], true)
140 ? explode(' ', $line, count($kinds) + 1) : explode(' ', $line);
141 if (count($parts) !== count($kinds) + 1) {
142 fail('bad arguments');
143 }
144 $args = [];
145 foreach ($kinds as $i => $kind) {
146 $args[] = typed_arg($parts[$i + 1], $kind);
147 }
148 }
149 foreach ($kinds as $i => $kind) {
150 if ($kind === 'pos' && $args[$i] > count($cues) + ($command === 'INSERT' ? 1 : 0)) {
151 fail('no cue ' . $args[$i]);
152 }
153 }
154 if ($command === 'SYNC' && $cues[$args[0] - 1]['start'] === $cues[$args[2] - 1]['start']) {
155 fail('sync points coincide');
156 }
157 return [$command, $args];
158}

160function wrap_lines(array $lines, int $width): array
161{
162 $output = [];
163 foreach (cue_words($lines) as $word) {
164 $last = count($output) - 1;
165 if ($last >= 0 && strlen($output[$last]) + 1 + strlen($word) <= $width) {
166 $output[$last] .= ' ' . $word;
167 } else {
168 $output[] = $word;
169 }
170 }
171 return $output;

174function markup(string $line): array
175{
176 $items = [];
177 $p = 0;
178 while ($p < strlen($line)) {
179 $tag = null;
180 foreach (FORMAT_TAGS as $text => $replacement) {
181 if (substr($line, $p, strlen($text)) === $text) {
182 $tag = $text;
183 break;
184 }
185 }
186 $rest = substr($line, $p);
187 if ($tag !== null) {
188 $items[] = ['format', $tag, 0];
189 $p += strlen($tag);
190 } elseif (str_starts_with($rest, '</font>')) {
191 $items[] = ['font', '</font>', 0];
192 $p += 7;
193 } elseif (str_starts_with($rest, '<font') && in_array($line[$p + 5] ?? '', [' ', '>'], true)
194 && ($ending = strpos($line, '>', $p + 5)) !== false) {
195 $items[] = ['font', substr($line, $p, $ending - $p + 1), 0];
196 $p = $ending + 1;
197 } elseif (substr($line, $p, 4) === '{\an' && ($line[$p + 4] ?? '') >= '1'
198 && ($line[$p + 4] ?? '') <= '9' && ($line[$p + 5] ?? '') === '}') {
199 $items[] = ['position', substr($line, $p, 6), (int) $line[$p + 4]];
200 $p += 6;
201 } else {
202 $items[] = ['ordinary', $line[$p], 0];
203 ++$p;
204 }
205 }
206 return $items;
207}

209function converted_line(string $line, string $mode): array
210{
211 $output = '';
212 $tags = 0;
213 $position = 0;
214 foreach (markup($line) as [$kind, $text, $digit]) {
215 if ($kind !== 'ordinary') {
216 ++$tags;
217 }
218 if ($kind === 'position' && $position === 0) {
219 $position = $digit;
220 }
221 if ($kind === 'ordinary') {
222 $output .= $mode === 'VTT' ? strtr($text, ['&' => '&amp;', '<' => '&lt;', '>' => '&gt;']) : $text;
223 } elseif ($mode === 'VTT' && $kind === 'format') {
224 $output .= $text;
225 } elseif ($mode === 'ASS' && $kind === 'format') {
226 $output .= FORMAT_TAGS[$text];
227 } elseif ($mode === 'ASS' && $kind === 'position') {
228 $output .= $text;
229 }
230 }
231 return [$mode === 'ASS' ? $output : trim($output, ' '), $tags, $position];
232}

234function clean_hi(string $line): array
235{
236 $cleaned = preg_replace('/\[[^\]]*\]|\([^)]*\)/','',$line,-1,$segments);
237 $normalize = fn($text) => trim(preg_replace('/ +/', ' ', $text), ' ');
238 $result = $normalize($cleaned);
239 $prefix = str_starts_with($result, '- ') ? '- ' : '';
240 $body = $prefix === '' ? $result : substr($result, 2);
241 $labels = preg_match('/^[A-Z][A-Z0-9 ]*:/', $body, $matches);
242 if ($labels === 1) {
243 $result = $normalize($prefix . substr($body, strlen($matches[0])));
244 }
245 return [$result, $segments, $labels];
246}

253function inspect(string $command, array $args, array $cues): array
254{
255 $output = [];
256 $n = count($cues);
257 if ($command === 'LIST' || $command === 'AT') {
258 foreach ($cues as $i => $cue) {
259 if ($command === 'LIST' || ($cue['start'] <= $args[0] && $args[0] < $cue['stop'])) {
260 $output[] = list_line($cue, $i + 1);
261 }
262 }
263 if ($command === 'AT') {
264 $output[] = 'AT active=' . count($output);
265 }
266 } elseif ($command === 'VALIDATE') {
267 foreach ($cues as $i => $cue) {
268 $p = $i + 1;
269 $d = $cue['stop'] - $cue['start'];
270 if ($cue['index'] !== $p) {
271 $output[] = "cue $p: index $cue[index] (expected $p)";
272 }
273 if ($d <= 0) {
274 $output[] = "cue $p: end not after start";
275 }
276 if ($i > 0 && $cue['start'] < $cues[$i - 1]['stop']) {
277 $output[] = "cue $p: overlaps previous by " . ($cues[$i - 1]['stop'] - $cue['start']) . ' ms';
278 }
279 if ($d >= 1 && $d < 700) {
280 $output[] = "cue $p: too short ($d ms)";
281 } elseif ($d > 7000) {
282 $output[] = "cue $p: too long ($d ms)";
283 }
284 foreach ($cue['lines'] as $j => $line) {
285 if (strlen($line) > 42) {
286 $output[] = "cue $p: line " . ($j + 1) . ' too long (' . strlen($line) . ' chars)';
287 }
288 }
289 if (count($cue['lines']) >= 3) {
290 $output[] = "cue $p: too many lines (" . count($cue['lines']) . ')';
291 }
292 if ($d >= 1 && char_count($cue) * 1000 > 21 * $d) {
293 $output[] = "cue $p: too fast (" . two_decimal(intdiv(char_count($cue) * 100000, $d)) . ' cps)';
294 }
295 }
296 $output[] = 'VALIDATE problems=' . count($output);
297 } elseif ($command === 'SEARCH') {
298 foreach ($cues as $i => $cue) {
299 $line = array_find($cue['lines'],fn($line) => str_contains($line,$args[0]));
300 if ($line !== null) {
301 $output[] = ($i + 1) . ' ' . time_text($cue['start']) . ' ' . $line;
302 }
303 }
304 $output[] = 'SEARCH found=' . count($output);
305 } elseif ($command === 'GAPS') {
306 for ($i = 0; $i + 1 < $n; ++$i) {
307 $gap = $cues[$i + 1]['start'] - $cues[$i]['stop'];
308 if ($gap < $args[0]) {
309 $output[] = ($i + 1) . ' ' . ($i + 2) . " $gap";
310 }
311 }
312 $output[] = 'GAPS count=' . count($output);
313 } elseif ($command === 'STATS') {
314 $output[] = "STATS cues=$n";
315 if ($n === 0) {
316 return $output;
317 }
318 $durations = array_map(fn($cue) => max(0, $cue['stop'] - $cue['start']), $cues);
319 $total = array_sum($durations);
320 $characters = array_sum(array_map(fn($cue,$duration) => $duration > 0 ? char_count($cue) : 0,$cues,$durations));
321 $output[] = 'duration total=' . time_text($total) . ' mean=' . time_text(intdiv($total, $n));
322 $output[] = $total === 0 ? 'speed -' : 'speed ' . two_decimal(intdiv($characters * 100000, $total));
323 $i = array_search(max($durations), $durations, true);
324 $output[] = 'longest cue=' . ($i + 1) . ' duration=' . time_text($durations[$i]);
325 if ($n === 1) {
326 $output[] = 'gap none';
327 } else {
328 $gaps = [];
329 for ($i = 0; $i + 1 < $n; ++$i) {
330 $gaps[] = $cues[$i + 1]['start'] - $cues[$i]['stop'];
331 }
332 $i = array_search(max($gaps), $gaps, true);
333 $output[] = 'gap cue=' . ($i + 1) . ' ms=' . $gaps[$i];
334 }
335 } elseif ($command === 'PRINT') {
336 foreach ($cues as $cue) {
337 $output = [...$output, (string) $cue['index'], time_text($cue['start']) . ' --> ' . time_text($cue['stop']), ...$cue['lines'], ''];
338 }
339 } elseif ($command === 'TOVTT') {
340 $output = ['WEBVTT', ''];
341 foreach ($cues as $cue) {
342 $lines = [];
343 $position = 0;
344 foreach ($cue['lines'] as $line) {
345 [$text, $tags, $digit] = converted_line($line, 'VTT');
346 if ($position === 0 && $digit > 0) {
347 $position = $digit;
348 }
349 if ($text !== '') {
350 $lines[] = $text;
351 }
352 }
353 if ($lines === []) {
354 continue;
355 }
356 $settings = '';
357 if ($position >= 7) {
358 $settings .= ' line:0';
359 } elseif ($position >= 4) {
360 $settings .= ' line:50%';
361 }
362 if (in_array($position, [1, 4, 7], true)) {
363 $settings .= ' align:start';
364 } elseif (in_array($position, [3, 6, 9], true)) {
365 $settings .= ' align:end';
366 }
367 $output = [...$output, (string) $cue['index'], time_text($cue['start'], 'VTT') . ' --> ' . time_text($cue['stop'], 'VTT') . $settings, ...$lines, ''];
368 }
369 } elseif ($command === 'TOASS') {
370 $output = ['[Events]', 'Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text'];
371 foreach ($cues as $cue) {
372 $text = implode('\N', array_map(fn($line) => converted_line($line, 'ASS')[0], $cue['lines']));
373 $output[] = 'Dialogue: 0,' . time_text($cue['start'], 'ASS') . ',' . time_text($cue['stop'], 'ASS') . ',Default,,0,0,0,,' . $text;
374 }
375 } elseif ($command === 'FRAMES') {
376 [$a, $b] = $args[0];
377 foreach ($cues as $i => $cue) {
378 $output[] = ($i + 1) . ' ' . intdiv($cue['start'] * $a, 1000 * $b) . ' ' . intdiv($cue['stop'] * $a, 1000 * $b);
379 }
380 }
381 return $output;
382}

384function modify(string $command, array $args, array $cues): array
385{
386 $output = [];
387 $n = count($cues);
388 if (in_array($command, ['SHIFT', 'SHIFTFROM', 'SCALE', 'SYNC', 'SNAP'], true)) {
389 $first = $command === 'SHIFTFROM' ? $args[0] - 1 : 0;
390 $clamped = 0;
391 $changed = 0;
392 if ($command === 'SYNC') {
393 [$p1, $t1, $p2, $t2] = $args;
394 $s1 = $cues[$p1 - 1]['start'];
395 $den = $cues[$p2 - 1]['start'] - $s1;
396 $transform = function($t) use ($t1, $t2, $s1, $den) {
397 $num = $t1 * $den + ($t - $s1) * ($t2 - $t1);
398 return $den > 0 ? round_ratio($num, $den) : round_ratio(-$num, -$den);
399 };
400 } elseif ($command === 'SCALE') {
401 $transform = fn($t) => round_ratio($t * $args[0], $args[1]);
402 } elseif ($command === 'SNAP') {
403 [$a, $b] = $args[0];
404 $transform = fn($t) => round_ratio(round_ratio($t * $a, 1000 * $b) * 1000 * $b, $a);
405 } else {
406 $delta = $command === 'SHIFT' ? $args[0] : $args[1];
407 $transform = fn($t) => $t + $delta;
408 }
409 for ($i = $first; $i < $n; ++$i) {
410 $start = $transform($cues[$i]['start']);
411 $stop = $transform($cues[$i]['stop']);
412 $clamped += $start < 0 || $stop < 0 ? 1 : 0;
413 $start = max(0, $start);
414 $stop = max(0, $stop);
415 $changed += ($start !== $cues[$i]['start'] ? 1 : 0) + ($stop !== $cues[$i]['stop'] ? 1 : 0);
416 $cues[$i]['start'] = $start;
417 $cues[$i]['stop'] = $stop;
418 }
419 if ($command === 'SNAP') {
420 $output[] = "SNAP changed=$changed";
421 } elseif ($command === 'SCALE') {
422 $output[] = "SCALE cues=$n";
423 } else {
424 $output[] = "$command cues=" . ($n - $first) . " clamped=$clamped";
425 }
426 } elseif ($command === 'FIXOVERLAP') {
427 $fixed = 0;
428 $failed = 0;
429 for ($i = 0; $i + 1 < $n; ++$i) {
430 $candidate = $cues[$i + 1]['start'] - $args[0];
431 if ($cues[$i]['stop'] > $candidate) {
432 if ($candidate > $cues[$i]['start']) {
433 $cues[$i]['stop'] = $candidate;
434 ++$fixed;
435 } else {
436 $output[] = 'cue ' . ($i + 1) . ': cannot fix';
437 ++$failed;
438 }
439 }
440 }
441 $output[] = "FIXOVERLAP fixed=$fixed failed=$failed";
442 } elseif ($command === 'EXTEND') {
443 $extended = 0;
444 foreach ($cues as $i => $cue) {
445 if ($cue['stop'] - $cue['start'] < $args[0]) {
446 $candidate = $cue['start'] + $args[0];
447 if ($i + 1 < $n) {
448 $candidate = min($candidate, $cues[$i + 1]['start']);
449 }
450 if ($candidate > $cue['stop']) {
451 $cues[$i]['stop'] = $candidate;
452 ++$extended;
453 }
454 }
455 }
456 $short = count(array_filter($cues, fn($cue) => $cue['stop'] - $cue['start'] < $args[0]));
457 $output[] = "EXTEND extended=$extended short=$short";
458 } elseif ($command === 'TIME') {
459 [$p, $start, $stop] = $args;
460 $cues[$p - 1]['start'] = $start;
461 $cues[$p - 1]['stop'] = $stop;
462 $output[] = "TIME cue=$p";
463 } elseif ($command === 'SORT') {
464 foreach ($cues as $i => $cue) {
465 $cues[$i]['original'] = $i;
466 }
467 usort($cues, fn($a, $b) => [$a['start'], $a['stop'], $a['original']] <=> [$b['start'], $b['stop'], $b['original']]);
468 $moved = 0;
469 foreach ($cues as $i => $cue) {
470 $moved += $cue['original'] !== $i ? 1 : 0;
471 unset($cues[$i]['original']);
472 }
473 $output[] = "SORT moved=$moved";
474 } elseif ($command === 'RENUMBER') {
475 $changed = 0;
476 foreach ($cues as $i => $cue) {
477 $changed += $cue['index'] !== $i + 1 ? 1 : 0;
478 $cues[$i]['index'] = $i + 1;
479 }
480 $output[] = "RENUMBER changed=$changed";
481 } elseif ($command === 'MERGE') {
482 $result = [];
483 $merged = 0;
484 foreach ($cues as $cue) {
485 $last = count($result) - 1;
486 if ($last >= 0 && $cue['start'] - $result[$last]['stop'] < $args[0] && !dialogue($result[$last]) && !dialogue($cue)) {
487 $lines = wrap_lines([...$result[$last]['lines'], ...$cue['lines']], 42);
488 if (count($lines) <= 2) {
489 $result[$last]['stop'] = max($result[$last]['stop'], $cue['stop']);
490 $result[$last]['lines'] = $lines;
491 ++$merged;
492 continue;
493 }
494 }
495 $result[] = $cue;
496 }
497 $cues = $result;
498 $output[] = "MERGE merged=$merged cues=" . count($cues);
499 } elseif ($command === 'SPLIT') {
500 $result = [];
501 $split = 0;
502 foreach ($cues as $cue) {
503 $duration = $cue['stop'] - $cue['start'];
504 if ($duration <= 0 || dialogue($cue)) {
505 $result[] = $cue;
506 continue;
507 }
508 $lines = wrap_lines($cue['lines'], $args[0]);
509 if (count($lines) <= 2) {
510 $cue['lines'] = $lines;
511 $result[] = $cue;
512 continue;
513 }
514 $parts = array_chunk($lines, 2);
515 $weights = array_map(fn($part) => array_sum(array_map('strlen', $part)), $parts);
516 $total = array_sum($weights);
517 $allotted = array_map(fn($weight) => intdiv($duration * $weight, $total), $weights);
518 $order = array_keys($parts);
519 usort($order, function($a, $b) use ($duration, $weights, $total) {
520 $ra = $duration * $weights[$a] % $total;
521 $rb = $duration * $weights[$b] % $total;
522 return $ra === $rb ? $a <=> $b : $rb <=> $ra;
523 });
524 $remaining = $duration - array_sum($allotted);
525 for ($i = 0; $i < $remaining; ++$i) {
526 ++$allotted[$order[$i]];
527 }
528 $start = $cue['start'];
529 foreach ($parts as $i => $part) {
530 $result[] = ['index' => $cue['index'], 'start' => $start, 'stop' => $start + $allotted[$i], 'lines' => $part];
531 $start += $allotted[$i];
532 }
533 ++$split;
534 }
535 $cues = $result;
536 $output[] = "SPLIT split=$split cues=" . count($cues);
537 } elseif ($command === 'INSERT') {
538 [$p, $start, $stop, $lines] = $args;
539 array_splice($cues, $p - 1, 0, [['index' => $p, 'start' => $start, 'stop' => $stop, 'lines' => $lines]]);
540 $output[] = "INSERT cue=$p cues=" . count($cues);
541 } elseif ($command === 'DELETE') {
542 $p = $args[0];
543 $q = $args[1] ?? $p;
544 array_splice($cues, $p - 1, $q - $p + 1);
545 $output[] = 'DELETE removed=' . ($q - $p + 1) . ' cues=' . count($cues);
546 } elseif ($command === 'TEXT') {
547 [$p, $lines] = $args;
548 $cues[$p - 1]['lines'] = $lines;
549 $output[] = "TEXT cue=$p lines=" . count($lines);
550 } elseif ($command === 'BALANCE') {
551 $changed = 0;
552 $skipped = 0;
553 $dialogues = 0;
554 foreach ($cues as $i => $cue) {
555 if (dialogue($cue)) {
556 ++$dialogues;
557 continue;
558 }
559 $words = cue_words($cue['lines']);
560 $text = implode(' ', $words);
561 $candidate = null;
562 $best = [PHP_INT_MAX, PHP_INT_MAX];
563 if (strlen($text) <= 42) {
564 $candidate = [$text];
565 } else {
566 for ($j = 1; $j < count($words); ++$j) {
567 $left = implode(' ', array_slice($words, 0, $j));
568 $right = implode(' ', array_slice($words, $j));
569 $key = [abs(strlen($left) - strlen($right)), strlen($left)];
570 if (strlen($left) <= 42 && strlen($right) <= 42 && $key < $best) {
571 $candidate = [$left, $right];
572 $best = $key;
573 }
574 }
575 }
576 if ($candidate === null) {
577 ++$skipped;
578 } else {
579 $changed += $cue['lines'] !== $candidate ? 1 : 0;
580 $cues[$i]['lines'] = $candidate;
581 }
582 }
583 $output[] = "BALANCE changed=$changed skipped=$skipped dialogue=$dialogues";
584 } elseif ($command === 'STRIP' || $command === 'CLEANHI') {
585 $result = [];
586 $tags = 0;
587 $labels = 0;
588 $removedLines = 0;
589 $removedCues = 0;
590 foreach ($cues as $cue) {
591 $lines = [];
592 foreach ($cue['lines'] as $line) {
593 if ($command === 'STRIP') {
594 [$text, $count] = converted_line($line, 'STRIP');
595 $tags += $count;
596 } else {
597 [$text, $count, $speakers] = clean_hi($line);
598 $tags += $count;
599 $labels += $speakers;
600 }
601 if ($text === '' || ($command === 'CLEANHI' && $text === '-')) {
602 ++$removedLines;
603 } else {
604 $lines[] = $text;
605 }
606 }
607 if ($lines === []) {
608 ++$removedCues;
609 } else {
610 $cue['lines'] = $lines;
611 $result[] = $cue;
612 }
613 }
614 $cues = $result;
615 $prefix = $command === 'STRIP' ? "STRIP tags=$tags" : "CLEANHI segments=$tags labels=$labels";
616 $output[] = "$prefix removed_lines=$removedLines removed_cues=$removedCues";
617 } elseif ($command === 'REPLACE') {
618 [$old, $new] = $args[0];
619 $replacements = 0;
620 $changed = 0;
621 foreach ($cues as $i => $cue) {
622 $cues[$i]['lines'] = str_replace($old,$new,$cue['lines'],$local);
623 $replacements += $local;
624 $changed += $local > 0 ? 1 : 0;
625 }
626 $output[] = "REPLACE count=$replacements cues=$changed";
627 }
628 return [$cues, $output];
629}

631function load_block(array $block, int $k, array &$cues): ?string
632{
633 if (preg_match('/^[0-9]{1,6}$/D', $block[0]) !== 1) {
634 return "BAD CUE $k: index";
635 }
636 try {
637 if (!isset($block[1]) || strlen($block[1]) !== 29 || substr($block[1], 12, 5) !== ' --> ') {
638 fail('bad arguments');
639 }
640 $start = time_value(substr($block[1], 0, 12));
641 $stop = time_value(substr($block[1], 17, 12));
642 } catch (InvalidArgumentException $exception) {
643 return "BAD CUE $k: time";
644 }
645 if (count($block) < 3) {
646 return "BAD CUE $k: no text";
647 }
648 $cues[] = ['index' => (int) $block[0], 'start' => $start, 'stop' => $stop, 'lines' => array_slice($block, 2)];
649 return null;
650}
651
652$cues = [];
653$block = [];
654$blocks = 0;
655$bad = 0;
656while (true) {
657 $line = rtrim(fgets(STDIN), "\r\n");
658 if ($line === '' || $line === '###') {
659 if ($block !== []) {
660 $problem = load_block($block, ++$blocks, $cues);
661 if ($problem !== null) {
662 echo $problem, "\n";
663 ++$bad;
664 }
665 $block = [];
666 }
667 if ($line === '###') {
668 break;
669 }
670 } else {
671 $block[] = $line;
672 }
673}
674echo 'LOAD cues=', count($cues), " bad=$bad\n";
675$history = [];
676$count = (int) fgets(STDIN);
677for ($i = 0; $i < $count; ++$i) {
678 $line = rtrim(fgets(STDIN), "\r\n");
679 try {
680 [$command, $args] = arguments($line, $cues);
681 if ($command === 'UNDO') {
682 if ($history === []) {
683 fail('nothing to undo');
684 }
685 [$previous, $cues] = array_pop($history);
686 $output = ["UNDO command=$previous"];
687 } elseif (in_array($command, MODIFYING, true)) {
688 $history[] = [$command, $cues];
689 [$cues, $output] = modify($command, $args, $cues);
690 } else {
691 $output = inspect($command, $args, $cues);
692 }
693 foreach ($output as $line) {
694 echo $line, "\n";
695 }
696 } catch (InvalidArgumentException $exception) {
697 echo 'ERROR ', $exception->getMessage(), "\n";
698 }
699}
