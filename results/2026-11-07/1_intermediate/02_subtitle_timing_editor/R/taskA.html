38floor_div <- function(n,d) {
39 quotient <- trunc(n/d)
40 remainder <- n-quotient*d
41 if (remainder < 0) {
42 quotient <- quotient-1
43 } else if (remainder >= d) {
44 quotient <- quotient+1
45 }
46 quotient
47}

66split_literal <- function(text,separator) {
67 positions <- gregexpr(separator,text,fixed=TRUE)[[1]]
68 if (positions[1] == -1) return(text)
69 starts <- c(1,positions+nchar(separator))
70 ends <- c(positions-1,nchar(text))
71 vapply(seq_along(starts),function(i) substr(text,starts[i],ends[i]),character(1))
72}

74time_value <- function(text) {
75 if (!grepl("^[0-9]{2}:[0-9]{2}:[0-9]{2},[0-9]{3}$",text)) fail("bad arguments")
76 values <- as.numeric(strsplit(text,"[:,]")[[1]])
77 if (values[2] > 59 || values[3] > 59) fail("bad arguments")
78 sum(values*c(3600000,60000,1000,1))
79}

81time_text <- function(t,kind="SRT") {
82 if (kind == "ASS") {
83 c <- floor_div(t+5,10)
84 return(sprintf("%.0f:%02d:%02d.%02d",floor_div(c,360000),floor_div(c,6000) %% 60,floor_div(c,100) %% 60,c %% 100))
85 }
86 sprintf("%02d:%02d:%02d%s%03d",floor_div(t,3600000),floor_div(t,60000) %% 60,floor_div(t,1000) %% 60,
87 if (kind == "VTT") "." else ",",t %% 1000)
88}

90unsigned_arg <- function(text,low=0,high=999999999) {
91 if (!grepl("^[0-9]{1,9}$",text)) fail("bad arguments")
92 value <- as.numeric(text)
93 if (value < low || value > high) fail("bad arguments")
94 value
95}

97typed_arg <- function(text,kind) {
98 if (kind == "string") return(text)
99 if (kind == "time") return(time_value(text))
100 if (kind == "pos") return(unsigned_arg(text,1))
101 if (kind == "signed") {
102 if (!grepl("^-?[0-9]{1,9}$",text)) fail("bad arguments")
103 return(as.numeric(text))
104 }
105 if (kind == "fps") {
106 parts <- split_literal(text,"/")
107 if (length(parts) > 2) fail("bad arguments")
108 return(c(unsigned_arg(parts[1],1,100000),if (length(parts) == 1) 1 else unsigned_arg(parts[2],1,100000)))
109 }
110 if (kind %in% c("text","replace")) {
111 parts <- split_literal(text,"|")
112 if (any(parts == "") || (kind == "replace" && length(parts) != 2)) fail("bad arguments")
113 return(parts)
114 }
115 bounds <- as.numeric(strsplit(kind,":",fixed=TRUE)[[1]])
116 unsigned_arg(text,bounds[1],bounds[2])
117}

119arguments <- function(line,cues) {
120 parts <- strsplit(line," ",fixed=TRUE)[[1]]
121 command <- parts[1]
122 if (!command %in% names(schemas)) fail("unknown command")
123 if (command == "DELETE") {
124 raw <- parts[-1]
125 if (!length(raw) %in% c(1,2)) fail("bad arguments")
126 args <- lapply(raw,unsigned_arg,1)
127 if (length(args) == 2 && args[[2]] < args[[1]]) fail("bad arguments")
128 kinds <- rep("pos",length(args))
129 } else {
130 kinds <- if (schemas[[command]] == "") character(0) else strsplit(schemas[[command]]," ",fixed=TRUE)[[1]]
131 if (length(kinds) > 0 && tail(kinds,1) %in% c("string","text","replace")) {
132 if (length(parts) <= length(kinds)) fail("bad arguments")
133 raw <- character(0)
134 position <- nchar(command)+2
135 for (i in seq_along(kinds)) {
136 rest <- substring(line,position)
137 if (i == length(kinds)) {
138 raw <- c(raw,rest)
139 } else {
140 ending <- regexpr(" ",rest,fixed=TRUE)[1]
141 if (ending == -1) fail("bad arguments")
142 raw <- c(raw,substr(rest,1,ending-1))
143 position <- position+ending
144 }
145 }
146 } else {
147 raw <- parts[-1]
148 if (length(raw) != length(kinds)) fail("bad arguments")
149 }
150 args <- lapply(seq_along(kinds),function(i) typed_arg(raw[i],kinds[i]))
151 }
152 for (i in seq_along(kinds)) {
153 if (kinds[i] == "pos" && args[[i]] > length(cues)+(if (command == "INSERT") 1 else 0)) {
154 fail(paste("no cue",number(args[[i]])))
155 }
156 }
157 if (command == "SYNC" && cues[[args[[1]]]]$start == cues[[args[[3]]]]$start) fail("sync points coincide")
158 list(command=command,args=args)
159}

174markup <- function(line) {
175 items <- list()
176 p <- 1
177 while (p <= nchar(line)) {
178 rest <- substring(line,p)
179 tags <- names(format_tags)[vapply(names(format_tags),function(tag) startsWith(rest,tag),logical(1))]
180 ending <- regexpr(">",substring(line,p+5),fixed=TRUE)[1]
181 if (length(tags) > 0) {
182 text <- tags[1]
183 item <- list(kind="format",text=text,digit=0)
184 p <- p+nchar(text)
185 } else if (startsWith(rest,"</font>")) {
186 item <- list(kind="font",text="</font>",digit=0)
187 p <- p+7
188 } else if (startsWith(rest,"<font") && substr(line,p+5,p+5) %in% c(" ",">") && ending != -1) {
189 ending <- p+5+ending-1
190 item <- list(kind="font",text=substr(line,p,ending),digit=0)
191 p <- ending+1
192 } else if (substr(line,p,p+3) == "{\\an" && grepl("^[1-9]$",substr(line,p+4,p+4)) && substr(line,p+5,p+5) == "}") {
193 item <- list(kind="position",text=substr(line,p,p+5),digit=as.numeric(substr(line,p+4,p+4)))
194 p <- p+6
195 } else {
196 item <- list(kind="ordinary",text=substr(line,p,p),digit=0)
197 p <- p+1
198 }
199 items[[length(items)+1]] <- item
200 }
201 items
202}

204converted_line <- function(line,mode) {
205 output <- character(0)
206 tags <- 0
207 position <- 0
208 escapes <- c("&"="&amp;","<"="&lt;",">"="&gt;")
209 for (item in markup(line)) {
210 kind <- item$kind
211 text <- item$text
212 if (kind != "ordinary") tags <- tags+1
213 if (kind == "position" && position == 0) position <- item$digit
214 if (kind == "ordinary") {
215 output <- c(output,if (mode == "VTT" && text %in% names(escapes)) escapes[[text]] else text)
216 } else if (mode == "VTT" && kind == "format") {
217 output <- c(output,text)
218 } else if (mode == "ASS" && kind == "format") {
219 output <- c(output,format_tags[[text]])
220 } else if (mode == "ASS" && kind == "position") {
221 output <- c(output,text)
222 }
223 }
224 result <- paste(output,collapse="")
225 list(text=if (mode == "ASS") result else trim_spaces(result),tags=tags,position=position)
226}

228clean_hi <- function(line) {
229 pattern <- "\\[[^\\]]*\\]|\\([^)]*\\)"
230 segments <- sum(gregexpr(pattern,line,perl=TRUE)[[1]] > 0)
231 normalize <- function(text) trim_spaces(gsub(" +"," ",text))
232 result <- normalize(gsub(pattern,"",line,perl=TRUE))
233 prefix <- if (startsWith(result,"- ")) "- " else ""
234 body <- if (prefix == "") result else substring(result,3)
235 label <- regexpr("^[A-Z][A-Z0-9 ]*:",body)
236 labels <- as.numeric(label[1] != -1)
237 if (labels == 1) result <- normalize(paste0(prefix,substring(body,attr(label,"match.length")+1)))
238 list(text=result,segments=segments,labels=labels)
239}

245inspect <- function(command,args,cues) {
246 output <- character(0)
247 n <- length(cues)
248 if (command %in% c("LIST","AT")) {
249 for (p in seq_along(cues)) {
250 cue <- cues[[p]]
251 if (command == "LIST" || (cue$start <= args[[1]] && args[[1]] < cue$stop)) output <- c(output,list_line(cue,p))
252 }
253 if (command == "AT") output <- c(output,paste0("AT active=",length(output)))
254 } else if (command == "VALIDATE") {
255 for (p in seq_along(cues)) {
256 cue <- cues[[p]]
257 d <- cue$stop-cue$start
258 if (cue$index != p) output <- c(output,sprintf("cue %d: index %.0f (expected %d)",p,cue$index,p))
259 if (d <= 0) output <- c(output,sprintf("cue %d: end not after start",p))
260 if (p > 1 && cue$start < cues[[p-1]]$stop) output <- c(output,sprintf("cue %d: overlaps previous by %.0f ms",p,cues[[p-1]]$stop-cue$start))
261 if (d >= 1 && d < 700) {
262 output <- c(output,sprintf("cue %d: too short (%.0f ms)",p,d))
263 } else if (d > 7000) {
264 output <- c(output,sprintf("cue %d: too long (%.0f ms)",p,d))
265 }
266 for (j in seq_along(cue$lines)) {
267 if (nchar(cue$lines[j]) > 42) output <- c(output,sprintf("cue %d: line %d too long (%d chars)",p,j,nchar(cue$lines[j])))
268 }
269 if (length(cue$lines) >= 3) output <- c(output,sprintf("cue %d: too many lines (%d)",p,length(cue$lines)))
270 if (d >= 1 && char_count(cue)*1000 > 21*d) output <- c(output,sprintf("cue %d: too fast (%s cps)",p,two_decimal(floor_div(char_count(cue)*100000,d))))
271 }
272 output <- c(output,paste0("VALIDATE problems=",length(output)))
273 } else if (command == "SEARCH") {
274 for (p in seq_along(cues)) {
275 cue <- cues[[p]]
276 matches <- which(grepl(args[[1]],cue$lines,fixed=TRUE))
277 if (length(matches) > 0) output <- c(output,paste(p,time_text(cue$start),cue$lines[matches[1]]))
278 }
279 output <- c(output,paste0("SEARCH found=",length(output)))
280 } else if (command == "GAPS") {
281 for (p in seq_len(max(0,n-1))) {
282 gap <- cues[[p+1]]$start-cues[[p]]$stop
283 if (gap < args[[1]]) output <- c(output,paste(p,p+1,number(gap)))
284 }
285 output <- c(output,paste0("GAPS count=",length(output)))
286 } else if (command == "STATS") {
287 output <- paste0("STATS cues=",n)
288 if (n == 0) return(output)
289 durations <- vapply(cues,function(cue) max(0,cue$stop-cue$start),numeric(1))
290 total <- sum(durations)
291 characters <- sum(vapply(cues[durations > 0],char_count,numeric(1)))
292 output <- c(output,paste0("duration total=",time_text(total)," mean=",time_text(floor_div(total,n))))
293 output <- c(output,if (total == 0) "speed -" else paste("speed",two_decimal(floor_div(characters*100000,total))))
294 p <- which.max(durations)
295 output <- c(output,paste0("longest cue=",p," duration=",time_text(durations[p])))
296 if (n == 1) {
297 output <- c(output,"gap none")
298 } else {
299 gaps <- vapply(seq_len(n-1),function(i) cues[[i+1]]$start-cues[[i]]$stop,numeric(1))
300 p <- which.max(gaps)
301 output <- c(output,paste0("gap cue=",p," ms=",number(gaps[p])))
302 }
303 } else if (command == "PRINT") {
304 for (cue in cues) output <- c(output,number(cue$index),paste(time_text(cue$start),"-->",time_text(cue$stop)),cue$lines,"")
305 } else if (command == "TOVTT") {
306 output <- c("WEBVTT","")
307 for (cue in cues) {
308 lines <- character(0)
309 position <- 0
310 for (line in cue$lines) {
311 converted <- converted_line(line,"VTT")
312 if (position == 0 && converted$position > 0) position <- converted$position
313 if (converted$text != "") lines <- c(lines,converted$text)
314 }
315 if (length(lines) == 0) next
316 settings <- ""
317 if (position >= 7) {
318 settings <- " line:0"
319 } else if (position >= 4) {
320 settings <- " line:50%"
321 }
322 if (position %in% c(1,4,7)) settings <- paste0(settings," align:start")
323 if (position %in% c(3,6,9)) settings <- paste0(settings," align:end")
324 output <- c(output,number(cue$index),paste0(time_text(cue$start,"VTT")," --> ",time_text(cue$stop,"VTT"),settings),lines,"")
325 }
326 } else if (command == "TOASS") {
327 output <- c("[Events]","Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text")
328 for (cue in cues) {
329 text <- paste(vapply(cue$lines,function(line) converted_line(line,"ASS")$text,character(1)),collapse="\\N")
330 output <- c(output,paste0("Dialogue: 0,",time_text(cue$start,"ASS"),",",time_text(cue$stop,"ASS"),",Default,,0,0,0,,",text))
331 }
332 } else if (command == "FRAMES") {
333 a <- args[[1]][1]
334 b <- args[[1]][2]
335 for (p in seq_along(cues)) {
336 cue <- cues[[p]]
337 output <- c(output,paste(p,number(floor_div(cue$start*a,1000*b)),number(floor_div(cue$stop*a,1000*b))))
338 }
339 }
340 output
341}

343modify <- function(command,args,cues) {
344 output <- character(0)
345 n <- length(cues)
346 if (command %in% c("SHIFT","SHIFTFROM","SCALE","SYNC","SNAP")) {
347 first <- if (command == "SHIFTFROM") args[[1]] else 1
348 clamped <- 0
349 changed <- 0
350 if (command == "SYNC") {
351 s1 <- cues[[args[[1]]]]$start
352 den <- cues[[args[[3]]]]$start-s1
353 t1 <- args[[2]]
354 t2 <- args[[4]]
355 transform <- function(t) {
356 num <- t1*den+(t-s1)*(t2-t1)
357 if (den < 0) round_ratio(-num,-den) else round_ratio(num,den)
358 }
359 } else if (command == "SCALE") {
360 transform <- function(t) round_ratio(t*args[[1]],args[[2]])
361 } else if (command == "SNAP") {
362 a <- args[[1]][1]
363 b <- args[[1]][2]
364 transform <- function(t) round_ratio(round_ratio(t*a,1000*b)*1000*b,a)
365 } else {
366 delta <- if (command == "SHIFT") args[[1]] else args[[2]]
367 transform <- function(t) t+delta
368 }
369 for (p in seq_len(max(0,n-first+1))+first-1) {
370 start <- transform(cues[[p]]$start)
371 stop <- transform(cues[[p]]$stop)
372 clamped <- clamped+as.numeric(start < 0 || stop < 0)
373 start <- max(0,start)
374 stop <- max(0,stop)
375 changed <- changed+(start != cues[[p]]$start)+(stop != cues[[p]]$stop)
376 cues[[p]]$start <- start
377 cues[[p]]$stop <- stop
378 }
379 if (command == "SNAP") {
380 output <- paste0("SNAP changed=",changed)
381 } else if (command == "SCALE") {
382 output <- paste0("SCALE cues=",n)
383 } else {
384 output <- paste0(command," cues=",max(0,n-first+1)," clamped=",clamped)
385 }
386 } else if (command == "FIXOVERLAP") {
387 fixed <- 0
388 failed <- 0
389 for (p in seq_len(max(0,n-1))) {
390 candidate <- cues[[p+1]]$start-args[[1]]
391 if (cues[[p]]$stop > candidate) {
392 if (candidate > cues[[p]]$start) {
393 cues[[p]]$stop <- candidate
394 fixed <- fixed+1
395 } else {
396 output <- c(output,paste0("cue ",p,": cannot fix"))
397 failed <- failed+1
398 }
399 }
400 }
401 output <- c(output,paste0("FIXOVERLAP fixed=",fixed," failed=",failed))
402 } else if (command == "EXTEND") {
403 extended <- 0
404 for (p in seq_along(cues)) {
405 cue <- cues[[p]]
406 if (cue$stop-cue$start < args[[1]]) {
407 candidate <- cue$start+args[[1]]
408 if (p < n) candidate <- min(candidate,cues[[p+1]]$start)
409 if (candidate > cue$stop) {
410 cues[[p]]$stop <- candidate
411 extended <- extended+1
412 }
413 }
414 }
415 short <- sum(vapply(cues,function(cue) cue$stop-cue$start < args[[1]],logical(1)))
416 output <- paste0("EXTEND extended=",extended," short=",short)
417 } else if (command == "TIME") {
418 p <- args[[1]]
419 cues[[p]]$start <- args[[2]]
420 cues[[p]]$stop <- args[[3]]
421 output <- paste0("TIME cue=",p)
422 } else if (command == "SORT") {
423 indices <- order(vapply(cues,function(cue) cue$start,numeric(1)),vapply(cues,function(cue) cue$stop,numeric(1)),seq_along(cues))
424 moved <- sum(indices != seq_along(cues))
425 cues <- cues[indices]
426 output <- paste0("SORT moved=",moved)
427 } else if (command == "RENUMBER") {
428 changed <- 0
429 for (p in seq_along(cues)) {
430 changed <- changed+(cues[[p]]$index != p)
431 cues[[p]]$index <- p
432 }
433 output <- paste0("RENUMBER changed=",changed)
434 } else if (command == "MERGE") {
435 result <- list()
436 merged <- 0
437 for (cue in cues) {
438 last <- length(result)
439 joined <- FALSE
440 if (last > 0 && cue$start-result[[last]]$stop < args[[1]] && !dialogue(result[[last]]) && !dialogue(cue)) {
441 lines <- wrap_lines(c(result[[last]]$lines,cue$lines),42)
442 if (length(lines) <= 2) {
443 result[[last]]$stop <- max(result[[last]]$stop,cue$stop)
444 result[[last]]$lines <- lines
445 merged <- merged+1
446 joined <- TRUE
447 }
448 }
449 if (!joined) result[[length(result)+1]] <- cue
450 }
451 cues <- result
452 output <- paste0("MERGE merged=",merged," cues=",length(cues))
453 } else if (command == "SPLIT") {
454 result <- list()
455 split_count <- 0
456 for (cue in cues) {
457 duration <- cue$stop-cue$start
458 if (duration <= 0 || dialogue(cue)) {
459 result[[length(result)+1]] <- cue
460 next
461 }
462 lines <- wrap_lines(cue$lines,args[[1]])
463 if (length(lines) <= 2) {
464 cue$lines <- lines
465 result[[length(result)+1]] <- cue
466 next
467 }
468 starts <- seq(1,length(lines),by=2)
469 parts <- lapply(starts,function(i) lines[i:min(i+1,length(lines))])
470 weights <- vapply(parts,function(part) sum(nchar(part)),numeric(1))
471 total <- sum(weights)
472 allotted <- vapply(weights,function(weight) floor_div(duration*weight,total),numeric(1))
473 remainders <- duration*weights-allotted*total
474 ordering <- order(-remainders,seq_along(parts))
475 for (i in seq_len(duration-sum(allotted))) allotted[ordering[i]] <- allotted[ordering[i]]+1
476 start <- cue$start
477 for (i in seq_along(parts)) {
478 result[[length(result)+1]] <- list(index=cue$index,start=start,stop=start+allotted[i],lines=parts[[i]])
479 start <- start+allotted[i]
480 }
481 split_count <- split_count+1
482 }
483 cues <- result
484 output <- paste0("SPLIT split=",split_count," cues=",length(cues))
485 } else if (command == "INSERT") {
486 p <- args[[1]]
487 cue <- list(index=p,start=args[[2]],stop=args[[3]],lines=args[[4]])
488 cues <- append(cues,list(cue),after=p-1)
489 output <- paste0("INSERT cue=",p," cues=",length(cues))
490 } else if (command == "DELETE") {
491 p <- args[[1]]
492 q <- if (length(args) == 1) p else args[[2]]
493 cues <- cues[-(p:q)]
494 output <- paste0("DELETE removed=",q-p+1," cues=",length(cues))
495 } else if (command == "TEXT") {
496 p <- args[[1]]
497 cues[[p]]$lines <- args[[2]]
498 output <- paste0("TEXT cue=",p," lines=",length(args[[2]]))
499 } else if (command == "BALANCE") {
500 changed <- 0
501 skipped <- 0
502 dialogues <- 0
503 for (p in seq_along(cues)) {
504 cue <- cues[[p]]
505 if (dialogue(cue)) {
506 dialogues <- dialogues+1
507 next
508 }
509 words <- cue_words(cue$lines)
510 text <- paste(words,collapse=" ")
511 candidate <- NULL
512 best_difference <- 1000000
513 best_first <- 1000000
514 if (nchar(text) <= 42) {
515 candidate <- text
516 } else {
517 for (j in seq_len(max(0,length(words)-1))) {
518 left <- paste(words[seq_len(j)],collapse=" ")
519 right <- paste(words[(j+1):length(words)],collapse=" ")
520 difference <- abs(nchar(left)-nchar(right))
521 if (nchar(left) <= 42 && nchar(right) <= 42 && (difference < best_difference || (difference == best_difference && nchar(left) < best_first))) {
522 candidate <- c(left,right)
523 best_difference <- difference
524 best_first <- nchar(left)
525 }
526 }
527 }
528 if (is.null(candidate)) {
529 skipped <- skipped+1
530 } else {
531 changed <- changed+!identical(cue$lines,candidate)
532 cues[[p]]$lines <- candidate
533 }
534 }
535 output <- paste0("BALANCE changed=",changed," skipped=",skipped," dialogue=",dialogues)
536 } else if (command %in% c("STRIP","CLEANHI")) {
537 result <- list()
538 tags <- 0
539 labels <- 0
540 removed_lines <- 0
541 removed_cues <- 0
542 for (cue in cues) {
543 lines <- character(0)
544 for (line in cue$lines) {
545 if (command == "STRIP") {
546 converted <- converted_line(line,"STRIP")
547 tags <- tags+converted$tags
548 } else {
549 converted <- clean_hi(line)
550 tags <- tags+converted$segments
551 labels <- labels+converted$labels
552 }
553 text <- converted$text
554 if (text == "" || (command == "CLEANHI" && text == "-")) {
555 removed_lines <- removed_lines+1
556 } else {
557 lines <- c(lines,text)
558 }
559 }
560 if (length(lines) == 0) {
561 removed_cues <- removed_cues+1
562 } else {
563 cue$lines <- lines
564 result[[length(result)+1]] <- cue
565 }
566 }
567 cues <- result
568 prefix <- if (command == "STRIP") paste0("STRIP tags=",tags) else paste0("CLEANHI segments=",tags," labels=",labels)
569 output <- paste0(prefix," removed_lines=",removed_lines," removed_cues=",removed_cues)
570 } else if (command == "REPLACE") {
571 old <- args[[1]][1]
572 new <- args[[1]][2]
573 replacements <- 0
574 changed <- 0
575 for (p in seq_along(cues)) {
576 lines <- cues[[p]]$lines
577 matches <- gregexpr(old,lines,fixed=TRUE)
578 counts <- vapply(matches,function(positions) sum(positions > 0),numeric(1))
579 regmatches(lines,matches) <- lapply(counts,function(count) rep(new,count))
580 cues[[p]]$lines <- lines
581 local_count <- sum(counts)
582 replacements <- replacements+local_count
583 changed <- changed+(local_count > 0)
584 }
585 output <- paste0("REPLACE count=",number(replacements)," cues=",changed)
586 }
587 list(cues=cues,output=output)
588}

590load_block <- function(block,k) {
591 if (!grepl("^[0-9]{1,6}$",block[1])) return(list(problem=paste0("BAD CUE ",k,": index")))
592 times <- tryCatch({
593 if (length(block) < 2 || nchar(block[2]) != 29 || substr(block[2],13,17) != " --> ") fail("bad arguments")
594 c(time_value(substr(block[2],1,12)),time_value(substr(block[2],18,29)))
595 },error=function(e) NULL)
596 if (is.null(times)) return(list(problem=paste0("BAD CUE ",k,": time")))
597 if (length(block) < 3) return(list(problem=paste0("BAD CUE ",k,": no text")))
598 list(cue=list(index=as.numeric(block[1]),start=times[1],stop=times[2],lines=block[-c(1,2)]))
599}

601main <- function() {
602 input <- readLines(file("stdin"),warn=FALSE)
603 cues <- list()
604 block <- character(0)
605 blocks <- 0
606 bad <- 0
607 position <- 1
608 repeat {
609 line <- input[position]
610 position <- position+1
611 if (line == "" || line == "###") {
612 if (length(block) > 0) {
613 blocks <- blocks+1
614 loaded <- load_block(block,blocks)
615 if (!is.null(loaded$problem)) {
616 cat(loaded$problem,"\n",sep="")
617 bad <- bad+1
618 } else {
619 cues[[length(cues)+1]] <- loaded$cue
620 }
621 block <- character(0)
622 }
623 if (line == "###") break
624 } else {
625 block <- c(block,line)
626 }
627 }
628 cat("LOAD cues=",length(cues)," bad=",bad,"\n",sep="")
629 history <- list()
630 count <- as.integer(input[position])
631 for (i in seq_len(count)) {
632 line <- input[position+i]
633 output <- tryCatch({
634 parsed <- arguments(line,cues)
635 command <- parsed$command
636 args <- parsed$args
637 if (command == "UNDO") {
638 if (length(history) == 0) fail("nothing to undo")
639 previous <- history[[length(history)]]
640 history <- history[-length(history)]
641 cues <- previous$cues
642 paste0("UNDO command=",previous$command)
643 } else if (command %in% modifying) {
644 result <- modify(command,args,cues)
645 history[[length(history)+1]] <- list(command=command,cues=cues)
646 cues <- result$cues
647 result$output
648 } else {
649 inspect(command,args,cues)
650 }
651 },error=function(e) paste("ERROR",conditionMessage(e)))
652 if (length(output) > 0) cat(paste0(output,"\n"),sep="")
653 }
654}
655main()
