1using Printf

3mutable struct SubtitleCue
4 index::Int
5 start::Int
6 stop::Int
7 lines::Vector{String}
8end

10const SCHEMAS = Dict(
11 "LIST"=>"", "VALIDATE"=>"", "SEARCH"=>"string", "AT"=>"time", "GAPS"=>"0:60000",
12 "STATS"=>"", "PRINT"=>"", "TOVTT"=>"", "TOASS"=>"", "FRAMES"=>"fps",
13 "SHIFT"=>"signed", "SHIFTFROM"=>"pos signed", "SCALE"=>"1:100000 1:100000",
14 "SYNC"=>"pos time pos time", "SNAP"=>"fps", "FIXOVERLAP"=>"0:60000", "EXTEND"=>"1:60000",
15 "TIME"=>"pos time time", "SORT"=>"", "RENUMBER"=>"", "MERGE"=>"0:60000", "SPLIT"=>"10:80",
16 "INSERT"=>"pos time time text", "DELETE"=>"delete", "TEXT"=>"pos text", "BALANCE"=>"",
17 "STRIP"=>"", "CLEANHI"=>"", "REPLACE"=>"replace", "UNDO"=>"")
18const MODIFYING = Set(split("SHIFT SHIFTFROM SCALE SYNC SNAP FIXOVERLAP EXTEND TIME SORT RENUMBER MERGE SPLIT INSERT DELETE TEXT BALANCE STRIP CLEANHI REPLACE"))
19const FORMAT_TAGS = Dict("<i>"=>raw"{\i1}", "</i>"=>raw"{\i0}", "<b>"=>raw"{\b1}",
20 "</b>"=>raw"{\b0}", "<u>"=>raw"{\u1}", "</u>"=>raw"{\u0}")
21
22bad_arguments() = throw(ArgumentError("bad arguments"))
23round_ratio(n,d) = fld(2*n+d,2*d)
24char_count(cue) = sum(length,cue.lines)
25dialogue(cue) = any(startswith(line,"- ") for line in cue.lines)
26cue_words(lines) = String.(split(join(lines," ")))
27trim_spaces(text) = String(strip(text,[' ']))
28two_decimal(value) = @sprintf("%d.%02d",div(value,100),value%100)

30function time_value(text)
31 occursin(r"^[0-9]{2}:[0-9]{2}:[0-9]{2},[0-9]{3}$",text) || bad_arguments()
32 h,m,s,ms = parse.(Int,split(text,r"[:,]"))
33 m <= 59 && s <= 59 || bad_arguments()
34 return h*3600000+m*60000+s*1000+ms
35end
36
37function time_text(t,kind="SRT")
38 if kind == "ASS"
39 c = div(t+5,10)
40 return @sprintf("%d:%02d:%02d.%02d",div(c,360000),div(c,6000)%60,div(c,100)%60,c%100)
41 end
42 separator = kind == "VTT" ? '.' : ','
43 return @sprintf("%02d:%02d:%02d%c%03d",div(t,3600000),div(t,60000)%60,div(t,1000)%60,separator,t%1000)
44end

46function unsigned_arg(text,low=0,high=999999999)
47 occursin(r"^[0-9]{1,9}$",text) || bad_arguments()
48 value = parse(Int,text)
49 low <= value <= high || bad_arguments()
50 return value
51end
52
53function typed_arg(text,kind)
54 kind == "string" && return String(text)
55 kind == "time" && return time_value(text)
56 kind == "pos" && return unsigned_arg(text,1)
57 if kind == "signed"
58 occursin(r"^-?[0-9]{1,9}$",text) || bad_arguments()
59 return parse(Int,text)
60 elseif kind == "fps"
61 parts = split(text,'/';keepempty=true)
62 1 <= length(parts) <= 2 || bad_arguments()
63 a = unsigned_arg(parts[1],1,100000)
64 b = length(parts) == 1 ? 1 : unsigned_arg(parts[2],1,100000)
65 return (a,b)
66 elseif kind in ("text","replace")
67 parts = String.(split(text,'|';keepempty=true))
68 all(!isempty(part) for part in parts) || bad_arguments()
69 kind == "replace" && length(parts) != 2 && bad_arguments()
70 return parts
71 end
72 low,high = parse.(Int,split(kind,':'))
73 return unsigned_arg(text,low,high)
74end

76function arguments(line,cues)
77 command = String(split(line,' ';limit=2)[1])
78 haskey(SCHEMAS,command) || throw(ArgumentError("unknown command"))
79 if command == "DELETE"
80 raw = split(line,' ')[2:end]
81 1 <= length(raw) <= 2 || bad_arguments()
82 args = unsigned_arg.(raw,1)
83 length(args) == 2 && args[2] < args[1] && bad_arguments()
84 kinds = fill("pos",length(args))
85 else
86 kinds = split(SCHEMAS[command])
87 if !isempty(kinds) && kinds[end] in ("string","text","replace")
88 parts = split(line,' ';limit=length(kinds)+1)
89 else
90 parts = split(line,' ')
91 end
92 length(parts) == length(kinds)+1 || bad_arguments()
93 args = Any[typed_arg(parts[i+1],kinds[i]) for i in eachindex(kinds)]
94 end
95 for (kind,arg) in zip(kinds,args)
96 if kind == "pos" && arg > length(cues)+(command == "INSERT" ? 1 : 0)
97 throw(ArgumentError("no cue $arg"))
98 end
99 end
100 command == "SYNC" && cues[args[1]].start == cues[args[3]].start && throw(ArgumentError("sync points coincide"))
101 return command,args
102end

104function wrap_lines(lines,width)
105 output = String[]
106 for word in cue_words(lines)
107 if !isempty(output) && length(output[end])+1+length(word) <= width
108 output[end] *= " "*word
109 else
110 push!(output,word)
111 end
112 end
113 return output
114end

116function markup(line)
117 items = Tuple{String,String,Int}[]
118 p = 1
119 tags = collect(keys(FORMAT_TAGS))
120 while p <= length(line)
121 rest = SubString(line,p)
122 tag = findfirst(tag -> startswith(rest,tag),tags)
123 if tag !== nothing
124 text = tags[tag]
125 push!(items,("format",text,0))
126 p += length(text)
127 elseif startswith(rest,"</font>")
128 push!(items,("font","</font>",0))
129 p += 7
130 elseif startswith(rest,"<font") && p+5 <= length(line) && line[p+5] in (' ','>') && findnext('>',line,p+5) !== nothing
131 ending = findnext('>',line,p+5)
132 push!(items,("font",line[p:ending],0))
133 p = ending+1
134 elseif p+5 <= length(line) && line[p:p+3] == raw"{\an" && '1' <= line[p+4] <= '9' && line[p+5] == '}'
135 push!(items,("position",line[p:p+5],Int(line[p+4]-'0')))
136 p += 6
137 else
138 push!(items,("ordinary",string(line[p]),0))
139 p += 1
140 end
141 end
142 return items
143end

145function converted_line(line,mode)
146 output = IOBuffer()
147 tags = 0
148 position = 0
149 for (kind,text,digit) in markup(line)
150 if kind != "ordinary"
151 tags += 1
152 end
153 kind == "position" && position == 0 && (position = digit)
154 if kind == "ordinary"
155 value = mode == "VTT" ? get(Dict("&"=>"&amp;","<"=>"&lt;",">"=>"&gt;"),text,text) : text
156 print(output,value)
157 elseif mode == "VTT" && kind == "format"
158 print(output,text)
159 elseif mode == "ASS" && kind == "format"
160 print(output,FORMAT_TAGS[text])
161 elseif mode == "ASS" && kind == "position"
162 print(output,text)
163 end
164 end
165 result = String(take!(output))
166 return mode == "ASS" ? result : trim_spaces(result),tags,position
167end

169function clean_hi(line)
170 pattern = r"\[[^\]]*\]|\([^)]*\)"
171 segments = length(findall(pattern,line))
172 normalize(text) = trim_spaces(replace(text,r" +"=>" "))
173 result = normalize(replace(line,pattern=>""))
174 prefix = startswith(result,"- ") ? "- " : ""
175 body = isempty(prefix) ? result : result[3:end]
176 matched = match(r"^[A-Z][A-Z0-9 ]*:",body)
177 labels = matched === nothing ? 0 : 1
178 if matched !== nothing
179 result = normalize(prefix*body[length(matched.match)+1:end])
180 end
181 return result,segments,labels
182end

184function list_line(cue,p)
185 return string(p,' ',cue.index,' ',time_text(cue.start),' ',time_text(cue.stop),' ',join(cue.lines,"|"))
186end

188function inspect(command,args,cues)
189 output = String[]
190 n = length(cues)
191 if command in ("LIST","AT")
192 for (p,cue) in enumerate(cues)
193 if command == "LIST" || cue.start <= args[1] < cue.stop
194 push!(output,list_line(cue,p))
195 end
196 end
197 command == "AT" && push!(output,"AT active=$(length(output))")
198 elseif command == "VALIDATE"
199 for (p,cue) in enumerate(cues)
200 d = cue.stop-cue.start
201 cue.index != p && push!(output,"cue $p: index $(cue.index) (expected $p)")
202 d <= 0 && push!(output,"cue $p: end not after start")
203 p > 1 && cue.start < cues[p-1].stop && push!(output,"cue $p: overlaps previous by $(cues[p-1].stop-cue.start) ms")
204 if 1 <= d < 700
205 push!(output,"cue $p: too short ($d ms)")
206 elseif d > 7000
207 push!(output,"cue $p: too long ($d ms)")
208 end
209 for (j,line) in enumerate(cue.lines)
210 length(line) > 42 && push!(output,"cue $p: line $j too long ($(length(line)) chars)")
211 end
212 length(cue.lines) >= 3 && push!(output,"cue $p: too many lines ($(length(cue.lines)))")
213 d >= 1 && char_count(cue)*1000 > 21*d && push!(output,"cue $p: too fast ($(two_decimal(div(char_count(cue)*100000,d))) cps)")
214 end
215 push!(output,"VALIDATE problems=$(length(output))")
216 elseif command == "SEARCH"
217 for (p,cue) in enumerate(cues)
218 j = findfirst(line -> occursin(args[1],line),cue.lines)
219 j !== nothing && push!(output,"$p $(time_text(cue.start)) $(cue.lines[j])")
220 end
221 push!(output,"SEARCH found=$(length(output))")
222 elseif command == "GAPS"
223 for p in 1:n-1
224 gap = cues[p+1].start-cues[p].stop
225 gap < args[1] && push!(output,"$p $(p+1) $gap")
226 end
227 push!(output,"GAPS count=$(length(output))")
228 elseif command == "STATS"
229 push!(output,"STATS cues=$n")
230 n == 0 && return output
231 durations = [max(0,cue.stop-cue.start) for cue in cues]
232 total = sum(durations)
233 characters = sum((char_count(cues[i]) for i in eachindex(cues) if durations[i] > 0);init=0)
234 push!(output,"duration total=$(time_text(total)) mean=$(time_text(div(total,n)))")
235 push!(output,total == 0 ? "speed -" : "speed $(two_decimal(div(characters*100000,total)))")
236 p = argmax(durations)
237 push!(output,"longest cue=$p duration=$(time_text(durations[p]))")
238 if n == 1
239 push!(output,"gap none")
240 else
241 gaps = [cues[i+1].start-cues[i].stop for i in 1:n-1]
242 p = argmax(gaps)
243 push!(output,"gap cue=$p ms=$(gaps[p])")
244 end
245 elseif command == "PRINT"
246 for cue in cues
247 append!(output,[string(cue.index),time_text(cue.start)*" --> "*time_text(cue.stop)])
248 append!(output,cue.lines)
249 push!(output,"")
250 end
251 elseif command == "TOVTT"
252 append!(output,["WEBVTT",""])
253 for cue in cues
254 lines = String[]
255 position = 0
256 for line in cue.lines
257 text,tags,digit = converted_line(line,"VTT")
258 position == 0 && digit > 0 && (position = digit)
259 !isempty(text) && push!(lines,text)
260 end
261 isempty(lines) && continue
262 settings = ""
263 if position >= 7
264 settings *= " line:0"
265 elseif position >= 4
266 settings *= " line:50%"
267 end
268 if position in (1,4,7)
269 settings *= " align:start"
270 elseif position in (3,6,9)
271 settings *= " align:end"
272 end
273 append!(output,[string(cue.index),time_text(cue.start,"VTT")*" --> "*time_text(cue.stop,"VTT")*settings])
274 append!(output,lines)
275 push!(output,"")
276 end
277 elseif command == "TOASS"
278 append!(output,["[Events]","Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text"])
279 for cue in cues
280 text = join((converted_line(line,"ASS")[1] for line in cue.lines),raw"\N")
281 push!(output,"Dialogue: 0,$(time_text(cue.start,"ASS")),$(time_text(cue.stop,"ASS")),Default,,0,0,0,,$text")
282 end
283 elseif command == "FRAMES"
284 a,b = args[1]
285 for (p,cue) in enumerate(cues)
286 push!(output,"$p $(div(cue.start*a,1000*b)) $(div(cue.stop*a,1000*b))")
287 end
288 end
289 return output
290end

292function modify(command,args,cues)
293 output = String[]
294 n = length(cues)
295 if command in ("SHIFT","SHIFTFROM","SCALE","SYNC","SNAP")
296 first = command == "SHIFTFROM" ? args[1] : 1
297 clamped = 0
298 changed = 0
299 if command == "SYNC"
300 p1,t1,p2,t2 = args
301 s1,s2 = cues[p1].start,cues[p2].start
302 den = s2-s1
303 transform = t -> den > 0 ? round_ratio(t1*den+(t-s1)*(t2-t1),den) : round_ratio(-t1*den-(t-s1)*(t2-t1),-den)
304 elseif command == "SCALE"
305 transform = t -> round_ratio(t*args[1],args[2])
306 elseif command == "SNAP"
307 a,b = args[1]
308 transform = t -> round_ratio(round_ratio(t*a,1000*b)*1000*b,a)
309 else
310 delta = command == "SHIFT" ? args[1] : args[2]
311 transform = t -> t+delta
312 end
313 for p in first:n
314 cue = cues[p]
315 start,stop = transform(cue.start),transform(cue.stop)
316 clamped += start < 0 || stop < 0
317 start,stop = max(0,start),max(0,stop)
318 changed += (start != cue.start)+(stop != cue.stop)
319 cue.start,cue.stop = start,stop
320 end
321 if command == "SNAP"
322 push!(output,"SNAP changed=$changed")
323 elseif command == "SCALE"
324 push!(output,"SCALE cues=$n")
325 else
326 push!(output,"$command cues=$(max(0,n-first+1)) clamped=$clamped")
327 end
328 elseif command == "FIXOVERLAP"
329 fixed,failed = 0,0
330 for p in 1:n-1
331 cue = cues[p]
332 candidate = cues[p+1].start-args[1]
333 if cue.stop > candidate
334 if candidate > cue.start
335 cue.stop = candidate
336 fixed += 1
337 else
338 push!(output,"cue $p: cannot fix")
339 failed += 1
340 end
341 end
342 end
343 push!(output,"FIXOVERLAP fixed=$fixed failed=$failed")
344 elseif command == "EXTEND"
345 extended = 0
346 for (p,cue) in enumerate(cues)
347 if cue.stop-cue.start < args[1]
348 candidate = cue.start+args[1]
349 p < n && (candidate = min(candidate,cues[p+1].start))
350 if candidate > cue.stop
351 cue.stop = candidate
352 extended += 1
353 end
354 end
355 end
356 short = count(cue -> cue.stop-cue.start < args[1],cues)
357 push!(output,"EXTEND extended=$extended short=$short")
358 elseif command == "TIME"
359 p,start,stop = args
360 cues[p].start,cues[p].stop = start,stop
361 push!(output,"TIME cue=$p")
362 elseif command == "SORT"
363 indices = sortperm(1:n;by=i -> (cues[i].start,cues[i].stop,i))
364 moved = count(i -> indices[i] != i,1:n)
365 cues = cues[indices]
366 push!(output,"SORT moved=$moved")
367 elseif command == "RENUMBER"
368 changed = 0
369 for (p,cue) in enumerate(cues)
370 changed += cue.index != p
371 cue.index = p
372 end
373 push!(output,"RENUMBER changed=$changed")
374 elseif command == "MERGE"
375 result = SubtitleCue[]
376 merged = 0
377 for cue in cues
378 if !isempty(result) && cue.start-result[end].stop < args[1] && !dialogue(result[end]) && !dialogue(cue)
379 lines = wrap_lines(vcat(result[end].lines,cue.lines),42)
380 if length(lines) <= 2
381 result[end].stop = max(result[end].stop,cue.stop)
382 result[end].lines = lines
383 merged += 1
384 continue
385 end
386 end
387 push!(result,cue)
388 end
389 cues = result
390 push!(output,"MERGE merged=$merged cues=$(length(cues))")
391 elseif command == "SPLIT"
392 result = SubtitleCue[]
393 split_count = 0
394 for cue in cues
395 duration = cue.stop-cue.start
396 if duration <= 0 || dialogue(cue)
397 push!(result,cue)
398 continue
399 end
400 lines = wrap_lines(cue.lines,args[1])
401 if length(lines) <= 2
402 cue.lines = lines
403 push!(result,cue)
404 continue
405 end
406 parts = [lines[i:min(i+1,length(lines))] for i in 1:2:length(lines)]
407 weights = [sum(length(line) for line in part) for part in parts]
408 total = sum(weights)
409 allotted = [div(duration*weight,total) for weight in weights]
410 order = sortperm(1:length(parts);by=i -> (-mod(duration*weights[i],total),i))
411 for i in 1:duration-sum(allotted)
412 allotted[order[i]] += 1
413 end
414 start = cue.start
415 for (i,part) in enumerate(parts)
416 push!(result,SubtitleCue(cue.index,start,start+allotted[i],part))
417 start += allotted[i]
418 end
419 split_count += 1
420 end
421 cues = result
422 push!(output,"SPLIT split=$split_count cues=$(length(cues))")
423 elseif command == "INSERT"
424 p,start,stop,lines = args
425 insert!(cues,p,SubtitleCue(p,start,stop,lines))
426 push!(output,"INSERT cue=$p cues=$(length(cues))")
427 elseif command == "DELETE"
428 p = args[1]
429 q = length(args) == 1 ? p : args[2]
430 deleteat!(cues,p:q)
431 push!(output,"DELETE removed=$(q-p+1) cues=$(length(cues))")
432 elseif command == "TEXT"
433 p,lines = args
434 cues[p].lines = lines
435 push!(output,"TEXT cue=$p lines=$(length(lines))")
436 elseif command == "BALANCE"
437 changed,skipped,dialogues = 0,0,0
438 for cue in cues
439 if dialogue(cue)
440 dialogues += 1
441 continue
442 end
443 words = cue_words(cue.lines)
444 text = join(words," ")
445 candidate = nothing
446 best = (typemax(Int),typemax(Int))
447 if length(text) <= 42
448 candidate = [text]
449 else
450 for j in 1:length(words)-1
451 left,right = join(words[1:j]," "),join(words[j+1:end]," ")
452 key = (abs(length(left)-length(right)),length(left))
453 if length(left) <= 42 && length(right) <= 42 && isless(key,best)
454 candidate,best = [left,right],key
455 end
456 end
457 end
458 if candidate === nothing
459 skipped += 1
460 else
461 changed += cue.lines != candidate
462 cue.lines = candidate
463 end
464 end
465 push!(output,"BALANCE changed=$changed skipped=$skipped dialogue=$dialogues")
466 elseif command in ("STRIP","CLEANHI")
467 result = SubtitleCue[]
468 tags,labels,removed_lines,removed_cues = 0,0,0,0
469 for cue in cues
470 lines = String[]
471 for line in cue.lines
472 if command == "STRIP"
473 text,count,digit = converted_line(line,"STRIP")
474 tags += count
475 else
476 text,count,speakers = clean_hi(line)
477 tags += count
478 labels += speakers
479 end
480 if isempty(text) || (command == "CLEANHI" && text == "-")
481 removed_lines += 1
482 else
483 push!(lines,text)
484 end
485 end
486 if isempty(lines)
487 removed_cues += 1
488 else
489 cue.lines = lines
490 push!(result,cue)
491 end
492 end
493 cues = result
494 if command == "STRIP"
495 push!(output,"STRIP tags=$tags removed_lines=$removed_lines removed_cues=$removed_cues")
496 else
497 push!(output,"CLEANHI segments=$tags labels=$labels removed_lines=$removed_lines removed_cues=$removed_cues")
498 end
499 elseif command == "REPLACE"
500 old,new = args[1]
501 replacements,changed = 0,0
502 for cue in cues
503 local_count = sum(length(findall(old,line)) for line in cue.lines)
504 replacements += local_count
505 changed += local_count > 0
506 cue.lines = [replace(line,old=>new) for line in cue.lines]
507 end
508 push!(output,"REPLACE count=$replacements cues=$changed")
509 end
510 return cues,output
511end

513function load_block(block,k,cues)
514 if !occursin(r"^[0-9]{1,6}$",block[1])
515 return "BAD CUE $k: index"
516 end
517 try
518 length(block) >= 2 && length(block[2]) == 29 && block[2][13:17] == " --> " || bad_arguments()
519 start,stop = time_value(block[2][1:12]),time_value(block[2][18:29])
520 length(block) >= 3 || return "BAD CUE $k: no text"
521 push!(cues,SubtitleCue(parse(Int,block[1]),start,stop,block[3:end]))
522 catch exception
523 exception isa ArgumentError || rethrow()
524 return "BAD CUE $k: time"
525 end
526 return ""
527end

529function main()
530 cues = SubtitleCue[]
531 block = String[]
532 blocks,bad = 0,0
533 while true
534 line = readline()
535 if isempty(line) || line == "###"
536 if !isempty(block)
537 blocks += 1
538 problem = load_block(block,blocks,cues)
539 if !isempty(problem)
540 println(problem)
541 bad += 1
542 end
543 empty!(block)
544 end
545 line == "###" && break
546 else
547 push!(block,line)
548 end
549 end
550 println("LOAD cues=$(length(cues)) bad=$bad")
551 history = Tuple{String,Vector{SubtitleCue}}[]
552 count = parse(Int,readline())
553 for i in 1:count
554 line = readline()
555 try
556 command,args = arguments(line,cues)
557 if command == "UNDO"
558 isempty(history) && throw(ArgumentError("nothing to undo"))
559 previous,cues = pop!(history)
560 output = ["UNDO command=$previous"]
561 elseif command in MODIFYING
562 push!(history,(command,deepcopy(cues)))
563 cues,output = modify(command,args,cues)
564 else
565 output = inspect(command,args,cues)
566 end
567 foreach(println,output)
568 catch exception
569 exception isa ArgumentError || rethrow()
570 println("ERROR ",exception.msg)
571 end
572 end
573end
