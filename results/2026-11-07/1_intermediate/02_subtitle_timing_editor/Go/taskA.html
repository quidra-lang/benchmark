38func clone(cues []cue) []cue {
39 result := slices.Clone(cues)
40 for i := range result {
41 result[i].text = slices.Clone(result[i].text)
42 }
43 return result
44}
45func fail(message string) { panic(message) }
46func unsigned(s string) (int64, bool) {
47 if len(s) < 1 || len(s) > 9 { return 0, false }
48 for i := range s {
49 if s[i] < '0' || s[i] > '9' { return 0, false }
50 }
51 n, _ := strconv.ParseInt(s, 10, 64)
52 return n, true
53}
54func timeValue(s string) (int64, bool) {
55 if len(s) != 12 || s[2] != ':' || s[5] != ':' || s[8] != ',' { return 0, false }
56 for i := range s {
57 if i != 2 && i != 5 && i != 8 && (s[i] < '0' || s[i] > '9') { return 0, false }
58 }
59 h, _ := strconv.ParseInt(s[:2], 10, 64)
60 m, _ := strconv.ParseInt(s[3:5], 10, 64)
61 sec, _ := strconv.ParseInt(s[6:8], 10, 64)
62 ms, _ := strconv.ParseInt(s[9:], 10, 64)
63 return h*3600000+m*60000+sec*1000+ms, m <= 59 && sec <= 59
64}
65func timeText(t int64, kind int) string {
66 if kind == 2 {
67 c := (t+5)/10
68 return fmt.Sprintf("%d:%02d:%02d.%02d", c/360000, c/6000%60, c/100%60, c%100)
69 }
70 mark := ","
71 if kind == 1 { mark = "." }
72 return fmt.Sprintf("%02d:%02d:%02d%s%03d", t/3600000, t/60000%60, t/1000%60, mark, t%1000)
73}
74func decimal(v int64) string { return fmt.Sprintf("%d.%02d", v/100, v%100) }
75func floorRatio(a, b int64) int64 {
76 if b < 0 { a, b = -a, -b }
77 q, r := a/b, a%b
78 if r < 0 { q-- }
79 return q
80}
81func halfUp(a, b int64) int64 {
82 if b < 0 { a, b = -a, -b }
83 return floorRatio(2*a+b, 2*b)
84}
85func countChars(c cue) int64 {
86 var count int64
87 for _, line := range c.text { count += int64(len(line)) }
88 return count
89}
90func dialogue(c cue) bool {
91 for _, line := range c.text {
92 if strings.HasPrefix(line, "- ") { return true }
93 }
94 return false
95}
96func words(lines []string) []string { return strings.Fields(strings.Join(lines, " ")) }
97func wrap(lines []string, width int) []string {
98 var result []string
99 for _, w := range words(lines) {
100 if len(result) == 0 || len(result[len(result)-1])+1+len(w) > width {
101 result = append(result, w)
102 } else { result[len(result)-1] += " " + w }
103 }
104 return result
105}
106func scanMarkup(line string) []markup {
107 tags := []string{"<i>", "</i>", "<b>", "</b>", "<u>", "</u>"}
108 var result []markup
109 for p := 0; p < len(line); {
110 size, kind, digit := 1, 0, 0
111 for _, tag := range tags {
112 if strings.HasPrefix(line[p:], tag) {
113 size, kind = len(tag), 1
114 break
115 }
116 }
117 if kind == 0 && strings.HasPrefix(line[p:], "</font>") { size, kind = 7, 2 }
118 if kind == 0 && strings.HasPrefix(line[p:], "<font") && p+5 < len(line) && (line[p+5] == ' ' || line[p+5] == '>') {
119 if end := strings.IndexByte(line[p+5:], '>'); end >= 0 { size, kind = 6+end, 2 }
120 }
121 if kind == 0 && p+6 <= len(line) && line[p:p+4] == "{\\an" && line[p+4] >= '1' && line[p+4] <= '9' && line[p+5] == '}' {
122 size, kind, digit = 6, 3, int(line[p+4]-'0')
123 }
124 result = append(result, markup{kind, line[p:p+size], digit})
125 p += size
126 }
127 return result
128}
129func converted(lines []string, mode int) ([]string, int, int) {
130 var result []string
131 first, removed := 0, 0
132 ass := map[string]string{"<i>": "{\\i1}", "</i>": "{\\i0}", "<b>": "{\\b1}", "</b>": "{\\b0}", "<u>": "{\\u1}", "</u>": "{\\u0}"}
133 for _, line := range lines {
134 var output strings.Builder
135 for _, m := range scanMarkup(line) {
136 if m.kind == 3 && first == 0 { first = m.digit }
137 if mode == 0 {
138 if m.kind == 0 { output.WriteString(m.text) } else { removed++ }
139 } else if mode == 1 {
140 if m.kind == 1 { output.WriteString(m.text) }
141 if m.kind == 0 {
142 switch m.text {
143 case "&": output.WriteString("&amp;")
144 case "<": output.WriteString("&lt;")
145 case ">": output.WriteString("&gt;")
146 default: output.WriteString(m.text)
147 }
148 }
149 } else {
150 if m.kind == 1 { output.WriteString(ass[m.text]) }
151 if m.kind == 0 || m.kind == 3 { output.WriteString(m.text) }
152 }
153 }
154 text := output.String()
155 if mode != 2 { text = strings.Trim(text, " ") }
156 if mode == 2 || text != "" { result = append(result, text) }
157 }
158 return result, first, removed
159}
160func cleanHI(line string) (string, int, int) {
161 var output strings.Builder
162 segments := 0
163 for p := 0; p < len(line); {
164 end := -1
165 if line[p] == '[' { end = strings.IndexByte(line[p+1:], ']') }
166 if line[p] == '(' { end = strings.IndexByte(line[p+1:], ')') }
167 if end >= 0 {
168 segments++
169 p += end+2
170 } else {
171 output.WriteByte(line[p])
172 p++
173 }
174 }
175 text := strings.Join(strings.Fields(output.String()), " ")
176 prefix, body := "", text
177 if strings.HasPrefix(body, "- ") { prefix, body = "- ", body[2:] }
178 labels := 0
179 if colon := strings.IndexByte(body, ':'); colon > 0 {
180 label := body[:colon]
181 valid := label[0] >= 'A' && label[0] <= 'Z'
182 for i := range label {
183 c := label[i]
184 if !(c >= 'A' && c <= 'Z' || c >= '0' && c <= '9' || c == ' ') { valid = false }
185 }
186 if valid {
187 labels = 1
188 text = strings.Join(strings.Fields(prefix+body[colon+1:]), " ")
189 }
190 }
191 return text, segments, labels
192}
193func parseArguments(line string, count int) arguments {
194 schemas := map[string]string{
195 "LIST":"", "VALIDATE":"", "SEARCH":"S", "AT":"T", "GAPS":"U", "STATS":"",
196 "PRINT":"", "TOVTT":"", "TOASS":"", "FRAMES":"F", "SHIFT":"I", "SHIFTFROM":"PI",
197 "SCALE":"UU", "SYNC":"PTPT", "SNAP":"F", "FIXOVERLAP":"U", "EXTEND":"U", "TIME":"PTT",
198 "SORT":"", "RENUMBER":"", "MERGE":"U", "SPLIT":"U", "INSERT":"PTTX", "DELETE":"P",
199 "TEXT":"PX", "BALANCE":"", "STRIP":"", "CLEANHI":"", "REPLACE":"R", "UNDO":"",
200 }
201 name, rest, hasSpace := strings.Cut(line, " ")
202 schema, known := schemas[name]
203 if !known { fail("unknown command") }
204 var raw []string
205 if hasSpace {
206 if strings.ContainsAny(schema, "SXR") { raw = strings.SplitN(rest, " ", len(schema)) } else { raw = strings.Split(rest, " ") }
207 }
208 if name == "DELETE" && len(raw) == 2 { schema = "PP" }
209 if len(raw) != len(schema) { fail("bad arguments") }
210 a := arguments{name: name, raw: raw, values: make([]int64, len(raw)), denominator: 1}
211 for i, kind := range schema {
212 s := raw[i]
213 valid := true
214 var value int64
215 switch kind {
216 case 'P', 'U':
217 value, valid = unsigned(s)
218 if kind == 'P' && value == 0 { valid = false }
219 case 'I':
220 negative := strings.HasPrefix(s, "-")
221 if negative { s = s[1:] }
222 value, valid = unsigned(s)
223 if negative { value = -value }
224 case 'T': value, valid = timeValue(s)
225 case 'F':
226 parts := strings.Split(s, "/")
227 if len(parts) > 2 { valid = false } else {
228 value, valid = unsigned(parts[0])
229 if len(parts) == 2 {
230 denominator, ok := unsigned(parts[1])
231 a.denominator = denominator
232 valid = valid && ok
233 }
234 valid = valid && value >= 1 && value <= 100000 && a.denominator >= 1 && a.denominator <= 100000
235 }
236 case 'X':
237 for _, part := range strings.Split(s, "|") {
238 if part == "" { valid = false }
239 }
240 case 'R':
241 old, replacement, found := strings.Cut(s, "|")
242 valid = found && old != "" && replacement != "" && !strings.Contains(replacement, "|")
243 case 'S': valid = s != ""
244 }
245 if !valid { fail("bad arguments") }
246 a.values[i] = value
247 }
248 if name == "SCALE" && (a.values[0] < 1 || a.values[0] > 100000 || a.values[1] < 1 || a.values[1] > 100000) { fail("bad arguments") }
249 if name == "GAPS" || name == "FIXOVERLAP" || name == "MERGE" || name == "EXTEND" {
250 minimum := int64(0)
251 if name == "EXTEND" { minimum = 1 }
252 if a.values[0] < minimum || a.values[0] > 60000 { fail("bad arguments") }
253 }
254 if name == "SPLIT" && (a.values[0] < 10 || a.values[0] > 80) { fail("bad arguments") }
255 if name == "DELETE" && len(raw) == 2 && a.values[1] < a.values[0] { fail("bad arguments") }
256 limit := int64(count)
257 if name == "INSERT" { limit++ }
258 for i, kind := range schema {
259 if kind == 'P' && a.values[i] > limit { fail(fmt.Sprintf("no cue %d", a.values[i])) }
260 }
261 return a
262}
263func cueLine(position int, c cue) string {
264 return fmt.Sprintf("%d %d %s %s %s", position, c.index, timeText(c.start, 0), timeText(c.end, 0), strings.Join(c.text, "|"))
265}
266func (e *editor) execute(line string) []string {
267 a := parseArguments(line, len(e.cues))
268 name, v := a.name, a.values
269 n := len(e.cues)
270 if name == "UNDO" {
271 if len(e.history) == 0 { fail("nothing to undo") }
272 previous := e.history[len(e.history)-1]
273 e.history = e.history[:len(e.history)-1]
274 e.cues = previous.cues
275 return []string{"UNDO command="+previous.command}
276 }
277 if name == "SYNC" && e.cues[v[0]-1].start == e.cues[v[2]-1].start { fail("sync points coincide") }
278 modifying := strings.Contains(" SHIFT SHIFTFROM SCALE SYNC SNAP FIXOVERLAP EXTEND TIME SORT RENUMBER MERGE SPLIT INSERT DELETE TEXT BALANCE STRIP CLEANHI REPLACE ", " "+name+" ")
279 if modifying { e.history = append(e.history, record{name, clone(e.cues)}) }
280 var output []string
281 switch name {
282 case "LIST", "AT":
283 matches := 0
284 for i, c := range e.cues {
285 if name == "LIST" || c.start <= v[0] && v[0] < c.end {
286 output = append(output, cueLine(i+1, c))
287 matches++
288 }
289 }
290 if name == "AT" { output = append(output, fmt.Sprintf("AT active=%d", matches)) }
291 case "VALIDATE":
292 for i, c := range e.cues {
293 prefix := fmt.Sprintf("cue %d: ", i+1)
294 d := c.end-c.start
295 if c.index != int64(i+1) { output = append(output, prefix+fmt.Sprintf("index %d (expected %d)", c.index, i+1)) }
296 if d <= 0 { output = append(output, prefix+"end not after start") }
297 if i > 0 && c.start < e.cues[i-1].end { output = append(output, prefix+fmt.Sprintf("overlaps previous by %d ms", e.cues[i-1].end-c.start)) }
298 if d >= 1 && d < 700 { output = append(output, prefix+fmt.Sprintf("too short (%d ms)", d)) }
299 if d > 7000 { output = append(output, prefix+fmt.Sprintf("too long (%d ms)", d)) }
300 for j, text := range c.text {
301 if len(text) > 42 { output = append(output, prefix+fmt.Sprintf("line %d too long (%d chars)", j+1, len(text))) }
302 }
303 if len(c.text) >= 3 { output = append(output, prefix+fmt.Sprintf("too many lines (%d)", len(c.text))) }
304 chars := countChars(c)
305 if d >= 1 && chars*1000 > 21*d { output = append(output, prefix+"too fast ("+decimal(chars*100000/d)+" cps)") }
306 }
307 output = append(output, fmt.Sprintf("VALIDATE problems=%d", len(output)))
308 case "SEARCH":
309 for i, c := range e.cues {
310 for _, text := range c.text {
311 if strings.Contains(text, a.raw[0]) {
312 output = append(output, fmt.Sprintf("%d %s %s", i+1, timeText(c.start, 0), text))
313 break
314 }
315 }
316 }
317 output = append(output, fmt.Sprintf("SEARCH found=%d", len(output)))
318 case "GAPS":
319 for i := 0; i+1 < n; i++ {
320 gap := e.cues[i+1].start-e.cues[i].end
321 if gap < v[0] { output = append(output, fmt.Sprintf("%d %d %d", i+1, i+2, gap)) }
322 }
323 output = append(output, fmt.Sprintf("GAPS count=%d", len(output)))
324 case "STATS":
325 output = append(output, fmt.Sprintf("STATS cues=%d", n))
326 if n == 0 { return output }
327 var total, chars, longest int64
328 longestPosition := 1
329 for i, c := range e.cues {
330 d := max(int64(0), c.end-c.start)
331 total += d
332 if d > 0 { chars += countChars(c) }
333 if d > longest { longest, longestPosition = d, i+1 }
334 }
335 output = append(output, "duration total="+timeText(total, 0)+" mean="+timeText(total/int64(n), 0))
336 speed := "-"
337 if total > 0 { speed = decimal(chars*100000/total) }
338 output = append(output, "speed "+speed, fmt.Sprintf("longest cue=%d duration=%s", longestPosition, timeText(longest, 0)))
339 if n == 1 { output = append(output, "gap none") } else {
340 position, gap := 1, e.cues[1].start-e.cues[0].end
341 for i := 1; i+1 < n; i++ {
342 candidate := e.cues[i+1].start-e.cues[i].end
343 if candidate > gap { position, gap = i+1, candidate }
344 }
345 output = append(output, fmt.Sprintf("gap cue=%d ms=%d", position, gap))
346 }
347 case "PRINT", "TOVTT":
348 if name == "TOVTT" { output = append(output, "WEBVTT", "") }
349 for _, c := range e.cues {
350 text, code := c.text, 0
351 kind := 0
352 if name == "TOVTT" {
353 text, code, _ = converted(c.text, 1)
354 kind = 1
355 }
356 if len(text) == 0 { continue }
357 setting := ""
358 if code >= 7 { setting = " line:0" } else if code >= 4 { setting = " line:50%" }
359 if code > 0 && code%3 == 1 { setting += " align:start" }
360 if code > 0 && code%3 == 0 { setting += " align:end" }
361 output = append(output, strconv.FormatInt(c.index, 10), timeText(c.start, kind)+" --> "+timeText(c.end, kind)+setting)
362 output = append(output, text...)
363 output = append(output, "")
364 }
365 case "TOASS":
366 output = append(output, "[Events]", "Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text")
367 for _, c := range e.cues {
368 text, _, _ := converted(c.text, 2)
369 output = append(output, "Dialogue: 0,"+timeText(c.start, 2)+","+timeText(c.end, 2)+",Default,,0,0,0,,"+strings.Join(text, "\\N"))
370 }
371 case "FRAMES":
372 for i, c := range e.cues { output = append(output, fmt.Sprintf("%d %d %d", i+1, c.start*v[0]/(1000*a.denominator), c.end*v[0]/(1000*a.denominator))) }
373 case "SHIFT", "SHIFTFROM", "SYNC", "SCALE", "SNAP":
374 start, shift := 0, int64(0)
375 if name == "SHIFT" { shift = v[0] }
376 if name == "SHIFTFROM" { start, shift = int(v[0]-1), v[1] }
377 var s1, s2 int64
378 if name == "SYNC" { s1, s2 = e.cues[v[0]-1].start, e.cues[v[2]-1].start }
379 clamped, changed := 0, 0
380 for i := start; i < n; i++ {
381 cueClamped := false
382 times := []*int64{&e.cues[i].start, &e.cues[i].end}
383 for _, t := range times {
384 old, value := *t, *t
385 switch name {
386 case "SHIFT", "SHIFTFROM": value += shift
387 case "SCALE": value = halfUp(value*v[0], v[1])
388 case "SYNC": value = halfUp(v[1]*(s2-s1)+(value-s1)*(v[3]-v[1]), s2-s1)
389 case "SNAP":
390 frame := halfUp(value*v[0], 1000*a.denominator)
391 value = halfUp(frame*1000*a.denominator, v[0])
392 }
393 if value < 0 {
394 value = 0
395 cueClamped = true
396 }
397 if old != value { changed++ }
398 *t = value
399 }
400 if cueClamped { clamped++ }
401 }
402 if name == "SNAP" { output = append(output, fmt.Sprintf("SNAP changed=%d", changed)) } else if name == "SCALE" {
403 output = append(output, fmt.Sprintf("SCALE cues=%d", n))
404 } else { output = append(output, fmt.Sprintf("%s cues=%d clamped=%d", name, n-start, clamped)) }
405 case "FIXOVERLAP":
406 fixed, failed := 0, 0
407 for i := 0; i+1 < n; i++ {
408 candidate := e.cues[i+1].start-v[0]
409 if e.cues[i].end > candidate {
410 if candidate > e.cues[i].start {
411 e.cues[i].end = candidate
412 fixed++
413 } else {
414 failed++
415 output = append(output, fmt.Sprintf("cue %d: cannot fix", i+1))
416 }
417 }
418 }
419 output = append(output, fmt.Sprintf("FIXOVERLAP fixed=%d failed=%d", fixed, failed))
420 case "EXTEND":
421 extended, short := 0, 0
422 for i := range e.cues {
423 if e.cues[i].end-e.cues[i].start < v[0] {
424 candidate := e.cues[i].start+v[0]
425 if i+1 < n { candidate = min(candidate, e.cues[i+1].start) }
426 if candidate > e.cues[i].end {
427 e.cues[i].end = candidate
428 extended++
429 }
430 }
431 if e.cues[i].end-e.cues[i].start < v[0] { short++ }
432 }
433 output = append(output, fmt.Sprintf("EXTEND extended=%d short=%d", extended, short))
434 case "TIME":
435 e.cues[v[0]-1].start, e.cues[v[0]-1].end = v[1], v[2]
436 output = append(output, fmt.Sprintf("TIME cue=%d", v[0]))
437 case "SORT":
438 order := make([]int, n)
439 for i := range order { order[i] = i }
440 sort.SliceStable(order, func(i, j int) bool {
441 x, y := e.cues[order[i]], e.cues[order[j]]
442 if x.start != y.start { return x.start < y.start }
443 return x.end < y.end
444 })
445 sorted := make([]cue, n)
446 moved := 0
447 for i, original := range order {
448 sorted[i] = e.cues[original]
449 if i != original { moved++ }
450 }
451 e.cues = sorted
452 output = append(output, fmt.Sprintf("SORT moved=%d", moved))
453 case "RENUMBER":
454 changed := 0
455 for i := range e.cues {
456 if e.cues[i].index != int64(i+1) { changed++ }
457 e.cues[i].index = int64(i+1)
458 }
459 output = append(output, fmt.Sprintf("RENUMBER changed=%d", changed))
460 case "MERGE":
461 var result []cue
462 merged := 0
463 for _, c := range e.cues {
464 if len(result) > 0 {
465 previous := &result[len(result)-1]
466 if c.start-previous.end < v[0] && !dialogue(*previous) && !dialogue(c) {
467 lines := slices.Concat(previous.text, c.text)
468 wrapped := wrap(lines, 42)
469 if len(wrapped) <= 2 {
470 previous.end = max(previous.end, c.end)
471 previous.text = wrapped
472 merged++
473 continue
474 }
475 }
476 }
477 result = append(result, c)
478 }
479 e.cues = result
480 output = append(output, fmt.Sprintf("MERGE merged=%d cues=%d", merged, len(result)))
481 case "SPLIT":
482 var result []cue
483 split := 0
484 for _, c := range e.cues {
485 d := c.end-c.start
486 if d < 1 || dialogue(c) {
487 result = append(result, c)
488 continue
489 }
490 wrapped := wrap(c.text, int(v[0]))
491 if len(wrapped) <= 2 {
492 c.text = wrapped
493 result = append(result, c)
494 continue
495 }
496 split++
497 parts := make([]cue, (len(wrapped)+1)/2)
498 var total int64
499 for i := range parts {
500 parts[i] = cue{index: c.index, text: wrapped[2*i:min(2*i+2, len(wrapped))]}
501 total += countChars(parts[i])
502 }
503 lengths, remainders := make([]int64, len(parts)), make([]int64, len(parts))
504 order := make([]int, len(parts))
505 remaining := d
506 for i := range parts {
507 product := d*countChars(parts[i])
508 lengths[i], remainders[i] = product/total, product%total
509 remaining -= lengths[i]
510 order[i] = i
511 }
512 sort.SliceStable(order, func(i, j int) bool { return remainders[order[i]] > remainders[order[j]] })
513 for i := int64(0); i < remaining; i++ { lengths[order[i]]++ }
514 start := c.start
515 for i := range parts {
516 parts[i].start, parts[i].end = start, start+lengths[i]
517 start += lengths[i]
518 result = append(result, parts[i])
519 }
520 }
521 e.cues = result
522 output = append(output, fmt.Sprintf("SPLIT split=%d cues=%d", split, len(result)))
523 case "INSERT":
524 p := int(v[0]-1)
525 c := cue{index: v[0], start: v[1], end: v[2], text: strings.Split(a.raw[3], "|")}
526 e.cues = slices.Insert(e.cues, p, c)
527 output = append(output, fmt.Sprintf("INSERT cue=%d cues=%d", v[0], len(e.cues)))
528 case "DELETE":
529 p, q := int(v[0]-1), int(v[0])
530 if len(v) == 2 { q = int(v[1]) }
531 e.cues = slices.Delete(e.cues, p, q)
532 output = append(output, fmt.Sprintf("DELETE removed=%d cues=%d", q-p, len(e.cues)))
533 case "TEXT":
534 e.cues[v[0]-1].text = strings.Split(a.raw[1], "|")
535 output = append(output, fmt.Sprintf("TEXT cue=%d lines=%d", v[0], len(e.cues[v[0]-1].text)))
536 case "BALANCE":
537 changed, skipped, dialogues := 0, 0, 0
538 for i, c := range e.cues {
539 if dialogue(c) {
540 dialogues++
541 continue
542 }
543 w := words(c.text)
544 text := strings.Join(w, " ")
545 var candidate []string
546 if len(text) <= 42 { candidate = []string{text} } else if len(text) <= 85 {
547 bestDifference, bestFirst := int(^uint(0)>>1), int(^uint(0)>>1)
548 for j := 1; j < len(w); j++ {
549 first, second := strings.Join(w[:j], " "), strings.Join(w[j:], " ")
550 difference := len(first)-len(second)
551 if difference < 0 { difference = -difference }
552 if len(first) <= 42 && len(second) <= 42 && (difference < bestDifference || difference == bestDifference && len(first) < bestFirst) {
553 candidate = []string{first, second}
554 bestDifference, bestFirst = difference, len(first)
555 }
556 }
557 }
558 if candidate == nil { skipped++ } else {
559 if !slices.Equal(c.text, candidate) { changed++ }
560 e.cues[i].text = candidate
561 }
562 }
563 output = append(output, fmt.Sprintf("BALANCE changed=%d skipped=%d dialogue=%d", changed, skipped, dialogues))
564 case "STRIP", "CLEANHI":
565 removedItems, labels, removedLines, removedCues := 0, 0, 0, 0
566 var result []cue
567 for _, c := range e.cues {
568 var lines []string
569 if name == "STRIP" {
570 var tags int
571 lines, _, tags = converted(c.text, 0)
572 removedItems += tags
573 removedLines += len(c.text)-len(lines)
574 } else {
575 for _, line := range c.text {
576 text, segments, names := cleanHI(line)
577 removedItems += segments
578 labels += names
579 if text == "" || text == "-" { removedLines++ } else { lines = append(lines, text) }
580 }
581 }
582 if len(lines) == 0 {
583 removedCues++
584 } else {
585 c.text = lines
586 result = append(result, c)
587 }
588 }
589 e.cues = result
590 if name == "STRIP" { output = append(output, fmt.Sprintf("STRIP tags=%d removed_lines=%d removed_cues=%d", removedItems, removedLines, removedCues)) } else {
591 output = append(output, fmt.Sprintf("CLEANHI segments=%d labels=%d removed_lines=%d removed_cues=%d", removedItems, labels, removedLines, removedCues))
592 }
593 case "REPLACE":
594 old, replacement, _ := strings.Cut(a.raw[0], "|")
595 count, cues := 0, 0
596 for i := range e.cues {
597 matches := 0
598 for j, text := range e.cues[i].text {
599 matches += strings.Count(text, old)
600 e.cues[i].text[j] = strings.ReplaceAll(text, old, replacement)
601 }
602 count += matches
603 if matches > 0 { cues++ }
604 }
605 output = append(output, fmt.Sprintf("REPLACE count=%d cues=%d", count, cues))
606 }
607 return output
608}
609func commandResult(e *editor, line string) (output []string) {
610 defer func() {
611 if message := recover(); message != nil { output = []string{"ERROR "+message.(string)} }
612 }()
613 return e.execute(line)
614}
615func loadBlock(block []string, position int, e *editor) string {
616 index, valid := unsigned(block[0])
617 if !valid || len(block[0]) > 6 { return fmt.Sprintf("BAD CUE %d: index", position) }
618 var start, end int64
619 valid = false
620 if len(block) >= 2 && len(block[1]) == 29 && block[1][12:17] == " --> " {
621 var ok bool
622 start, valid = timeValue(block[1][:12])
623 end, ok = timeValue(block[1][17:])
624 valid = valid && ok
625 }
626 if !valid { return fmt.Sprintf("BAD CUE %d: time", position) }
627 if len(block) < 3 { return fmt.Sprintf("BAD CUE %d: no text", position) }
628 e.cues = append(e.cues, cue{index, start, end, slices.Clone(block[2:])})
629 return ""
630}
631func main() {
632 scanner := bufio.NewScanner(os.Stdin)
633 output := bufio.NewWriter(os.Stdout)
634 defer output.Flush()
635 e := editor{}
636 var block []string
637 blocks, bad := 0, 0
638 finish := func() {
639 if len(block) == 0 { return }
640 blocks++
641 if message := loadBlock(block, blocks, &e); message != "" {
642 fmt.Fprintln(output, message)
643 bad++
644 }
645 block = nil
646 }
647 for scanner.Scan() {
648 line := scanner.Text()
649 if line == "###" {
650 finish()
651 break
652 }
653 if line == "" { finish() } else { block = append(block, line) }
654 }
655 fmt.Fprintf(output, "LOAD cues=%d bad=%d\n", len(e.cues), bad)
656 scanner.Scan()
657 count, _ := strconv.Atoi(scanner.Text())
658 for i := 0; i < count && scanner.Scan(); i++ {
659 for _, line := range commandResult(&e, scanner.Text()) { fmt.Fprintln(output, line) }
660 }
661}
