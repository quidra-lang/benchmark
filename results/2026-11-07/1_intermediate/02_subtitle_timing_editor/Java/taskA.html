7 record Cue(long index, long start, long end, List<String> text) {
8 long duration() { return end-start; }
9 long chars() { return text.stream().mapToLong(String::length).sum(); }
10 boolean dialogue() { return text.stream().anyMatch(s -> s.startsWith("- ")); }
11 }
12 record Undo(String command, List<Cue> cues) {}
13 record Item(String text, int kind, int digit) {}

30 static void bad() { throw new Problem("bad arguments"); }
31 static long number(String s) {
32 if (!s.matches("[0-9]{1,9}")) bad();
33 return Long.parseLong(s);
34 }
35 static long time(String s) {
36 if (!s.matches("[0-9]{2}:[0-5][0-9]:[0-5][0-9],[0-9]{3}")) bad();
37 return Long.parseLong(s.substring(0,2))*3600000+Long.parseLong(s.substring(3,5))*60000+
38 Long.parseLong(s.substring(6,8))*1000+Long.parseLong(s.substring(9));
39 }
40 static long[] fps(String s) {
41 String[] pieces = s.split("/",-1);
42 if (pieces.length > 2) bad();
43 long a = number(pieces[0]), b = pieces.length == 2 ? number(pieces[1]) : 1;
44 if (a < 1 || a > 100000 || b < 1 || b > 100000) bad();
45 return new long[]{a,b};
46 }
47 static List<String> text(String s) {
48 List<String> lines = Arrays.asList(s.split("\\|",-1));
49 if (lines.stream().anyMatch(String::isEmpty)) bad();
50 return lines;
51 }
52 static String srt(long t) {
53 return String.format(Locale.ROOT,"%02d:%02d:%02d,%03d",t/3600000,t/60000%60,t/1000%60,t%1000);
54 }
55 static String ass(long t) {
56 long c = (t+5)/10;
57 return String.format(Locale.ROOT,"%d:%02d:%02d.%02d",c/360000,c/6000%60,c/100%60,c%100);
58 }
59 static String decimal(long v) { return String.format(Locale.ROOT,"%d.%02d",v/100,v%100); }
60 static long round(long a, long b) {
61 if (b < 0) {
62 a = -a;
63 b = -b;
64 }
65 return Math.floorDiv(2*a+b,2*b);
66 }
67 static List<String> words(List<String> lines) {
68 return Arrays.stream(String.join(" ",lines).split(" ")).filter(s -> !s.isEmpty()).toList();
69 }
70 static List<String> wrap(List<String> lines, long width) {
71 List<String> result = new ArrayList<>();
72 for (String word : words(lines)) {
73 if (result.isEmpty() || result.getLast().length()+1+word.length() > width) {
74 result.add(word);
75 } else {
76 result.set(result.size()-1,result.getLast()+" "+word);
77 }
78 }
79 return result;
80 }
81 static String normalize(String s) { return String.join(" ",words(List.of(s))); }
82 static final Pattern MARKUP = Pattern.compile("</?[ibu]>|</font>|<font(?=[ >])[^>]*>|\\{\\\\an[1-9]\\}|.");
83 static List<Item> markup(String line) {
84 return MARKUP.matcher(line).results().map(match -> {
85 String text = match.group();
86 int kind = text.startsWith("{\\an") ? 3 :
87 text.startsWith("<font") || text.equals("</font>") ? 2 : text.length() > 1 ? 1 : 0;
88 return new Item(text,kind,kind == 3 ? text.charAt(4)-'0' : 0);
89 }).toList();
90 }
91 static String listing(int p, Cue c) {
92 return p+" "+c.index()+" "+srt(c.start())+" "+srt(c.end())+" "+String.join("|",c.text());
93 }
94 void load(List<List<String>> blocks, List<String> out) {
95 int bad = 0;
96 for (int i = 0; i < blocks.size(); i++) {
97 List<String> block = blocks.get(i);
98 String reason = "";
99 long start = 0, end = 0;
100 if (!block.getFirst().matches("[0-9]{1,6}")) reason = "index";
101 else {
102 try {
103 if (block.size() < 2 || block.get(1).length() != 29 || !block.get(1).substring(12,17).equals(" --> ")) bad();
104 start = time(block.get(1).substring(0,12));
105 end = time(block.get(1).substring(17));
106 } catch (Problem problem) {
107 reason = "time";
108 }
109 if (reason.isEmpty() && block.size() < 3) reason = "no text";
110 }
111 if (!reason.isEmpty()) {
112 out.add("BAD CUE "+(i+1)+": "+reason);
113 bad++;
114 } else {
115 cues.add(new Cue(Long.parseLong(block.getFirst()),start,end,List.copyOf(block.subList(2,block.size()))));
116 }
117 }
118 out.add("LOAD cues="+cues.size()+" bad="+bad);
119 }
120 List<String> execute(String line) {
121 String command = line.split(" ",2)[0];
122 if (!SCHEMAS.containsKey(command)) throw new Problem("unknown command");
123 String schema = SCHEMAS.get(command);
124 if (command.equals("DELETE") && line.split(" ").length == 3) schema = "PP";
125 String[] tokens = schema.endsWith("X") ? line.split(" ",schema.length()+1) : line.split(" ");
126 if (tokens.length != schema.length()+1) bad();
127 String[] args = Arrays.copyOfRange(tokens,1,tokens.length);
128 long[] values = new long[args.length];
129 for (int i = 0; i < args.length; i++) {
130 char type = schema.charAt(i);
131 if (type == 'T') values[i] = time(args[i]);
132 else if (type == 'F') fps(args[i]);
133 else if (type == 'X') {
134 if (command.equals("INSERT") || command.equals("TEXT")) text(args[i]);
135 if (command.equals("REPLACE")) {
136 String[] parts = args[i].split("\\|",-1);
137 if (parts.length != 2 || parts[0].isEmpty() || parts[1].isEmpty()) bad();
138 }
139 } else {
140 String raw = args[i];
141 boolean negative = type == 'I' && raw.startsWith("-");
142 values[i] = number(negative ? raw.substring(1) : raw)*(negative ? -1 : 1);
143 if (type == 'P' && values[i] < 1 || type == 'r' && (values[i] < 1 || values[i] > 100000) ||
144 type == 'g' && values[i] > 60000 || type == 'e' && (values[i] < 1 || values[i] > 60000) ||
145 type == 'w' && (values[i] < 10 || values[i] > 80)) bad();
146 }
147 }
148 if (command.equals("DELETE") && values.length == 2 && values[1] < values[0]) bad();
149 for (int i = 0; i < args.length; i++) {
150 if (schema.charAt(i) == 'P' && values[i] > cues.size()+(command.equals("INSERT") ? 1 : 0)) {
151 throw new Problem("no cue "+values[i]);
152 }
153 }
154 if (command.equals("SYNC") && cues.get((int)values[0]-1).start() == cues.get((int)values[2]-1).start()) {
155 throw new Problem("sync points coincide");
156 }
157 if (command.equals("UNDO")) {
158 if (history.isEmpty()) throw new Problem("nothing to undo");
159 Undo undo = history.pop();
160 cues = undo.cues();
161 return List.of("UNDO command="+undo.command());
162 }
163 if (MODIFY.contains(command)) history.push(new Undo(command,new ArrayList<>(cues)));
164 List<String> out = new ArrayList<>();
165 int n = cues.size();
166 long v = values.length == 0 ? 0 : values[0];
167 switch (command) {
168 case "LIST", "AT", "SEARCH", "GAPS": {
169 int count = 0;
170 for (int i = 0; i < n; i++) {
171 Cue cue = cues.get(i);
172 if (command.equals("LIST") || command.equals("AT") && cue.start() <= v && v < cue.end()) {
173 out.add(listing(i+1,cue));
174 count++;
175 }
176 if (command.equals("SEARCH")) {
177 for (String t : cue.text()) {
178 if (t.contains(args[0])) {
179 out.add((i+1)+" "+srt(cue.start())+" "+t);
180 count++;
181 break;
182 }
183 }
184 }
185 if (command.equals("GAPS") && i+1 < n && cues.get(i+1).start()-cue.end() < v) {
186 out.add((i+1)+" "+(i+2)+" "+(cues.get(i+1).start()-cue.end()));
187 count++;
188 }
189 }
190 if (!command.equals("LIST")) out.add(command+switch (command) {
191 case "AT" -> " active=";
192 case "SEARCH" -> " found=";
193 default -> " count=";
194 }+count);
195 break;
196 }
197 case "VALIDATE": {
198 for (int i = 0; i < n; i++) {
199 Cue c = cues.get(i);
200 long d = c.duration();
201 String p = "cue "+(i+1)+": ";
202 if (c.index() != i+1) out.add(p+"index "+c.index()+" (expected "+(i+1)+")");
203 if (d <= 0) out.add(p+"end not after start");
204 if (i > 0 && c.start() < cues.get(i-1).end()) out.add(p+"overlaps previous by "+(cues.get(i-1).end()-c.start())+" ms");
205 if (d >= 1 && d < 700) out.add(p+"too short ("+d+" ms)");
206 if (d > 7000) out.add(p+"too long ("+d+" ms)");
207 for (int j = 0; j < c.text().size(); j++) {
208 int size = c.text().get(j).length();
209 if (size > 42) out.add(p+"line "+(j+1)+" too long ("+size+" chars)");
210 }
211 if (c.text().size() >= 3) out.add(p+"too many lines ("+c.text().size()+")");
212 if (d >= 1 && c.chars()*1000 > 21*d) out.add(p+"too fast ("+decimal(c.chars()*100000/d)+" cps)");
213 }
214 out.add("VALIDATE problems="+out.size());
215 break;
216 }
217 case "STATS": {
218 out.add("STATS cues="+n);
219 if (n == 0) break;
220 List<Long> durations = cues.stream().map(c -> Math.max(0,c.duration())).toList();
221 long total = durations.stream().mapToLong(Long::longValue).sum(), longest = Collections.max(durations);
222 long chars = cues.stream().filter(c -> c.duration() > 0).mapToLong(Cue::chars).sum();
223 out.add("duration total="+srt(total)+" mean="+srt(total/n));
224 out.add("speed "+(total == 0 ? "-" : decimal(chars*100000/total)));
225 out.add("longest cue="+(durations.indexOf(longest)+1)+" duration="+srt(longest));
226 if (n == 1) out.add("gap none");
227 else {
228 List<Long> gaps = IntStream.range(0,n-1).mapToObj(i -> cues.get(i+1).start()-cues.get(i).end()).toList();
229 long gap = Collections.max(gaps);
230 out.add("gap cue="+(gaps.indexOf(gap)+1)+" ms="+gap);
231 }
232 break;
233 }
234 case "PRINT", "TOVTT", "TOASS": {
235 if (command.equals("TOVTT")) out.addAll(List.of("WEBVTT",""));
236 if (command.equals("TOASS")) out.addAll(List.of("[Events]","Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text"));
237 for (Cue c : cues) {
238 List<String> lines = new ArrayList<>();
239 int code = 0;
240 for (String t : c.text()) {
241 StringBuilder converted = new StringBuilder();
242 for (Item item : markup(t)) {
243 if (code == 0 && item.kind() == 3) code = item.digit();
244 if (command.equals("PRINT")) converted.append(item.text());
245 else if (command.equals("TOASS")) {
246 if (item.kind() == 1) {
247 char tag = item.text().charAt(item.text().startsWith("</") ? 2 : 1);
248 converted.append("{\\").append(tag).append(item.text().startsWith("</") ? "0}" : "1}");
249 } else if (item.kind() != 2) converted.append(item.text());
250 } else if (item.kind() == 1) converted.append(item.text());
251 else if (item.kind() == 0) converted.append(switch (item.text()) {
252 case "&" -> "&amp;";
253 case "<" -> "&lt;";
254 case ">" -> "&gt;";
255 default -> item.text();
256 });
257 }
258 String convertedText = converted.toString();
259 if (command.equals("TOVTT")) convertedText = convertedText.trim();
260 if (!command.equals("TOVTT") || !convertedText.isEmpty()) lines.add(convertedText);
261 }
262 if (command.equals("TOASS")) {
263 out.add("Dialogue: 0,"+ass(c.start())+","+ass(c.end())+",Default,,0,0,0,,"+String.join("\\N",lines));
264 } else if (!lines.isEmpty()) {
265 String settings = "";
266 if (command.equals("TOVTT") && code != 0) {
267 if (code >= 7) settings += " line:0";
268 else if (code >= 4) settings += " line:50%";
269 if (code%3 == 1) settings += " align:start";
270 if (code%3 == 0) settings += " align:end";
271 }
272 out.add(Long.toString(c.index()));
273 String times = srt(c.start())+" --> "+srt(c.end());
274 out.add((command.equals("TOVTT") ? times.replace(',','.') : times)+settings);
275 out.addAll(lines);
276 out.add("");
277 }
278 }
279 break;
280 }
281 case "FRAMES": {
282 long[] rate = fps(args[0]);
283 for (int i = 0; i < n; i++) {
284 Cue c = cues.get(i);
285 out.add((i+1)+" "+c.start()*rate[0]/(1000*rate[1])+" "+c.end()*rate[0]/(1000*rate[1]));
286 }
287 break;
288 }
289 case "SHIFT", "SHIFTFROM", "SCALE", "SYNC", "SNAP": {
290 int first = command.equals("SHIFTFROM") ? (int)v-1 : 0;
291 long delta = command.equals("SHIFTFROM") ? values[1] : v;
292 long s1 = command.equals("SYNC") ? cues.get((int)v-1).start() : 0;
293 long s2 = command.equals("SYNC") ? cues.get((int)values[2]-1).start() : 0;
294 long[] rate = command.equals("SNAP") ? fps(args[0]) : new long[]{1,1};
295 int clamped = 0, changed = 0;
296 for (int i = first; i < n; i++) {
297 Cue c = cues.get(i);
298 long[] times = {c.start(),c.end()};
299 boolean clamp = false;
300 for (int j = 0; j < 2; j++) {
301 long old = times[j], next;
302 if (command.equals("SCALE")) next = round(old*v,values[1]);
303 else if (command.equals("SYNC")) next = round(values[1]*(s2-s1)+(old-s1)*(values[3]-values[1]),s2-s1);
304 else if (command.equals("SNAP")) next = round(round(old*rate[0],1000*rate[1])*1000*rate[1],rate[0]);
305 else next = old+delta;
306 if (next < 0) clamp = true;
307 times[j] = Math.max(0,next);
308 if (times[j] != old) changed++;
309 }
310 if (clamp) clamped++;
311 cues.set(i,new Cue(c.index(),times[0],times[1],c.text()));
312 }
313 if (command.equals("SNAP")) out.add("SNAP changed="+changed);
314 else out.add(command+" cues="+(n-first)+(command.equals("SCALE") ? "" : " clamped="+clamped));
315 break;
316 }
317 case "FIXOVERLAP", "EXTEND": {
318 int changed = 0, failed = 0;
319 for (int i = 0; i < n; i++) {
320 Cue c = cues.get(i);
321 long end = c.end();
322 if (command.equals("FIXOVERLAP") && i+1 < n && end > cues.get(i+1).start()-v) {
323 long candidate = cues.get(i+1).start()-v;
324 if (candidate > c.start()) {
325 end = candidate;
326 changed++;
327 } else {
328 out.add("cue "+(i+1)+": cannot fix");
329 failed++;
330 }
331 }
332 if (command.equals("EXTEND") && c.duration() < v) {
333 long candidate = c.start()+v;
334 if (i+1 < n) candidate = Math.min(candidate,cues.get(i+1).start());
335 if (candidate > end) {
336 end = candidate;
337 changed++;
338 }
339 }
340 cues.set(i,new Cue(c.index(),c.start(),end,c.text()));
341 }
342 if (command.equals("FIXOVERLAP")) out.add("FIXOVERLAP fixed="+changed+" failed="+failed);
343 else out.add("EXTEND extended="+changed+" short="+cues.stream().filter(c -> c.duration() < v).count());
344 break;
345 }
346 case "TIME": {
347 Cue c = cues.get((int)v-1);
348 cues.set((int)v-1,new Cue(c.index(),values[1],values[2],c.text()));
349 out.add("TIME cue="+v);
350 break;
351 }
352 case "SORT": {
353 List<Cue> old = new ArrayList<>(cues);
354 cues.sort(Comparator.comparingLong(Cue::start).thenComparingLong(Cue::end));
355 int moved = 0;
356 for (int i = 0; i < n; i++) {
357 if (old.get(i) != cues.get(i)) moved++;
358 }
359 out.add("SORT moved="+moved);
360 break;
361 }
362 case "RENUMBER": {
363 int changed = 0;
364 for (int i = 0; i < n; i++) {
365 Cue c = cues.get(i);
366 if (c.index() != i+1) changed++;
367 cues.set(i,new Cue(i+1,c.start(),c.end(),c.text()));
368 }
369 out.add("RENUMBER changed="+changed);
370 break;
371 }
372 case "MERGE": {
373 List<Cue> result = new ArrayList<>();
374 int merged = 0;
375 for (Cue b : cues) {
376 if (!result.isEmpty()) {
377 Cue a = result.getLast();
378 List<String> both = new ArrayList<>(a.text());
379 both.addAll(b.text());
380 List<String> wrapped = wrap(both,42);
381 if (b.start()-a.end() < v && !a.dialogue() && !b.dialogue() && wrapped.size() <= 2) {
382 result.set(result.size()-1,new Cue(a.index(),a.start(),Math.max(a.end(),b.end()),wrapped));
383 merged++;
384 continue;
385 }
386 }
387 result.add(b);
388 }
389 cues = result;
390 out.add("MERGE merged="+merged+" cues="+cues.size());
391 break;
392 }
393 case "SPLIT": {
394 List<Cue> result = new ArrayList<>();
395 int split = 0;
396 for (Cue c : cues) {
397 if (c.duration() <= 0 || c.dialogue()) {
398 result.add(c);
399 continue;
400 }
401 List<String> wrapped = wrap(c.text(),v);
402 if (wrapped.size() <= 2) {
403 result.add(new Cue(c.index(),c.start(),c.end(),wrapped));
404 continue;
405 }
406 split++;
407 List<List<String>> parts = new ArrayList<>();
408 for (int i = 0; i < wrapped.size(); i += 2) parts.add(List.copyOf(wrapped.subList(i,Math.min(i+2,wrapped.size()))));
409 long total = parts.stream().flatMap(List::stream).mapToLong(String::length).sum();
410 long[] durations = new long[parts.size()], remainders = new long[parts.size()];
411 List<Integer> order = new ArrayList<>();
412 long allocated = 0;
413 for (int i = 0; i < parts.size(); i++) {
414 long count = parts.get(i).stream().mapToLong(String::length).sum();
415 durations[i] = c.duration()*count/total;
416 remainders[i] = c.duration()*count%total;
417 allocated += durations[i];
418 order.add(i);
419 }
420 order.sort(Comparator.<Integer>comparingLong(i -> remainders[i]).reversed().thenComparingInt(i -> i));
421 for (int i = 0; i < c.duration()-allocated; i++) durations[order.get(i)]++;
422 long start = c.start();
423 for (int i = 0; i < parts.size(); i++) {
424 result.add(new Cue(c.index(),start,start+durations[i],parts.get(i)));
425 start += durations[i];
426 }
427 }
428 cues = result;
429 out.add("SPLIT split="+split+" cues="+cues.size());
430 break;
431 }
432 case "INSERT": {
433 cues.add((int)v-1,new Cue(v,values[1],values[2],text(args[3])));
434 out.add("INSERT cue="+v+" cues="+cues.size());
435 break;
436 }
437 case "DELETE": {
438 long last = values.length == 2 ? values[1] : v;
439 cues.subList((int)v-1,(int)last).clear();
440 out.add("DELETE removed="+(last-v+1)+" cues="+cues.size());
441 break;
442 }
443 case "TEXT": {
444 Cue c = cues.get((int)v-1);
445 List<String> lines = text(args[1]);
446 cues.set((int)v-1,new Cue(c.index(),c.start(),c.end(),lines));
447 out.add("TEXT cue="+v+" lines="+lines.size());
448 break;
449 }
450 case "BALANCE": {
451 int changed = 0, skipped = 0, dialogue = 0;
452 for (int i = 0; i < n; i++) {
453 Cue c = cues.get(i);
454 if (c.dialogue()) {
455 dialogue++;
456 continue;
457 }
458 List<String> tokens2 = words(c.text());
459 String joined = String.join(" ",tokens2);
460 List<String> best = joined.length() <= 42 ? List.of(joined) : null;
461 int difference = Integer.MAX_VALUE, firstLength = Integer.MAX_VALUE;
462 for (int j = 1; joined.length() > 42 && j < tokens2.size(); j++) {
463 String a = String.join(" ",tokens2.subList(0,j)), b = String.join(" ",tokens2.subList(j,tokens2.size()));
464 int d = Math.abs(a.length()-b.length());
465 if (a.length() <= 42 && b.length() <= 42 && (d < difference || d == difference && a.length() < firstLength)) {
466 best = List.of(a,b);
467 difference = d;
468 firstLength = a.length();
469 }
470 }
471 if (best == null) skipped++;
472 else if (!best.equals(c.text())) {
473 changed++;
474 cues.set(i,new Cue(c.index(),c.start(),c.end(),best));
475 }
476 }
477 out.add("BALANCE changed="+changed+" skipped="+skipped+" dialogue="+dialogue);
478 break;
479 }
480 case "STRIP", "CLEANHI": {
481 List<Cue> result = new ArrayList<>();
482 int tags = 0, segments = 0, labels = 0, removed = 0;
483 for (Cue c : cues) {
484 List<String> lines = new ArrayList<>();
485 for (String t : c.text()) {
486 StringBuilder builder = new StringBuilder();
487 if (command.equals("STRIP")) {
488 for (Item item : markup(t)) {
489 if (item.kind() == 0) builder.append(item.text());
490 else tags++;
491 }
492 } else {
493 int p = 0;
494 while (p < t.length()) {
495 char ch = t.charAt(p);
496 int end = ch == '[' || ch == '(' ? t.indexOf(ch == '[' ? ']' : ')',p+1) : -1;
497 if (end >= 0) {
498 segments++;
499 p = end+1;
500 } else {
501 builder.append(ch);
502 p++;
503 }
504 }
505 }
506 String cleaned = command.equals("STRIP") ? builder.toString().trim() : normalize(builder.toString());
507 if (command.equals("CLEANHI")) {
508 String prefix = cleaned.startsWith("- ") ? "- " : "";
509 String body = cleaned.substring(prefix.length());
510 int colon = body.indexOf(':');
511 if (colon > 0 && body.substring(0,colon).matches("[A-Z][A-Z0-9 ]*")) {
512 labels++;
513 cleaned = normalize(prefix+body.substring(colon+1));
514 }
515 }
516 if (cleaned.isEmpty() || command.equals("CLEANHI") && cleaned.equals("-")) removed++;
517 else lines.add(cleaned);
518 }
519 if (!lines.isEmpty()) result.add(new Cue(c.index(),c.start(),c.end(),lines));
520 }
521 int removedCues = n-result.size();
522 cues = result;
523 out.add(command.equals("STRIP") ? "STRIP tags="+tags+" removed_lines="+removed+" removed_cues="+removedCues :
524 "CLEANHI segments="+segments+" labels="+labels+" removed_lines="+removed+" removed_cues="+removedCues);
525 break;
526 }
527 case "REPLACE": {
528 String[] parts = args[0].split("\\|",-1);
529 String old = parts[0], replacement = parts[1];
530 int count = 0, changed = 0;
531 for (int i = 0; i < n; i++) {
532 Cue c = cues.get(i);
533 List<String> lines = new ArrayList<>();
534 int before = count;
535 for (String t : c.text()) {
536 String[] pieces = t.split(Pattern.quote(old),-1);
537 count += pieces.length-1;
538 lines.add(String.join(replacement,pieces));
539 }
540 if (count > before) changed++;
541 cues.set(i,new Cue(c.index(),c.start(),c.end(),lines));
542 }
543 out.add("REPLACE count="+count+" cues="+changed);
544 break;
545 }
546 }
547 return out;
548 }
