6fn unsigned(text: []const u8, limit: usize) ?i64 {
7 if (text.len == 0 or text.len > limit) return null;
8 for (text) |c| {
9 if (!std.ascii.isDigit(c)) return null;
10 }
11 return std.fmt.parseInt(i64, text, 10) catch null;
12}
13fn timeValue(text: []const u8) ?i64 {
14 if (text.len != 12 or text[2] != ':' or text[5] != ':' or text[8] != ',') return null;
15 const h = unsigned(text[0..2], 2) orelse return null;
16 const m = unsigned(text[3..5], 2) orelse return null;
17 const s = unsigned(text[6..8], 2) orelse return null;
18 const ms = unsigned(text[9..12], 3) orelse return null;
19 if (m > 59 or s > 59) return null;
20 return h * 3600000 + m * 60000 + s * 1000 + ms;
21}
22fn roundRatio(numerator: i64, denominator: i64) i64 {
23 const a = if (denominator < 0) -numerator else numerator;
24 const b = if (denominator < 0) -denominator else denominator;
25 return @divFloor(2 * a + b, 2 * b);
26}
27fn timeText(gpa: Allocator, value: i64, vtt: bool) Allocator.Error![]const u8 {
28 const t: u64 = @intCast(value);
29 return std.fmt.allocPrint(gpa, "{d:0>2}:{d:0>2}:{d:0>2}{c}{d:0>3}", .{
30 t / 3600000, t / 60000 % 60, t / 1000 % 60,
31 @as(u8, if (vtt) '.' else ','), t % 1000,
32 });
33}
34fn assTime(gpa: Allocator, value: i64) Allocator.Error![]const u8 {
35 const c: u64 = @intCast(@divFloor(value + 5, 10));
36 return std.fmt.allocPrint(gpa, "{d}:{d:0>2}:{d:0>2}.{d:0>2}", .{
37 c / 360000, c / 6000 % 60, c / 100 % 60, c % 100,
38 });
39}
40fn decimal(gpa: Allocator, value: i64) Allocator.Error![]const u8 {
41 const v: u64 = @intCast(value);
42 return std.fmt.allocPrint(gpa, "{d}.{d:0>2}", .{ v / 100, v % 100 });
43}
44fn split(gpa: Allocator, text: []const u8, delimiter: u8, skip_empty: bool) Allocator.Error!Strings {
45 var result: std.ArrayList([]const u8) = .empty;
46 var parts = std.mem.splitScalar(u8, text, delimiter);
47 while (parts.next()) |part| {
48 if (!skip_empty or part.len > 0) try result.append(gpa, part);
49 }
50 return result.items;
51}
52fn words(gpa: Allocator, lines: []const []const u8) Allocator.Error!Strings {
53 var result: std.ArrayList([]const u8) = .empty;
54 for (lines) |line| try result.appendSlice(gpa, try split(gpa, line, ' ', true));
55 return result.items;
56}
57fn wrap(gpa: Allocator, lines: []const []const u8, width: usize) Allocator.Error!Strings {
58 var result: std.ArrayList([]const u8) = .empty;
59 for (try words(gpa, lines)) |word| {
60 if (result.items.len > 0 and result.items[result.items.len - 1].len + 1 + word.len <= width) {
61 const last = result.items.len - 1;
62 result.items[last] = try std.fmt.allocPrint(gpa, "{s} {s}", .{ result.items[last], word });
63 } else try result.append(gpa, word);
64 }
65 return result.items;
66}
67fn sameLines(a: []const []const u8, b: []const []const u8) bool {
68 if (a.len != b.len) return false;
69 for (a, b) |x, y| {
70 if (!std.mem.eql(u8, x, y)) return false;
71 }
72 return true;
73}
74fn normalized(gpa: Allocator, text: []const u8) Allocator.Error![]const u8 {
75 return std.mem.join(gpa, " ", try split(gpa, text, ' ', true));
76}

78fn remainderLess(remainders: []const i64, a: usize, b: usize) bool {
79 return remainders[a] > remainders[b];
80}

82const Cue = struct {
83 index: i64,
84 start: i64,
85 end: i64,
86 text: Strings,
87 fn duration(self: Cue) i64 {
88 return self.end - self.start;
89 }
90 fn characters(self: Cue) i64 {
91 var count: i64 = 0;
92 for (self.text) |line| count += @intCast(line.len);
93 return count;
94 }
95 fn dialogue(self: Cue) bool {
96 for (self.text) |line| {
97 if (std.mem.startsWith(u8, line, "- ")) return true;
98 }
99 return false;
100 }
101};
102const format_tags = [_][]const u8{ "<i>", "</i>", "<b>", "</b>", "<u>", "</u>" };
103const ass_tags = [_][]const u8{ "{\\i1}", "{\\i0}", "{\\b1}", "{\\b0}", "{\\u1}", "{\\u0}" };
104const Markup = struct { len: usize, kind: enum { format, font, position, ordinary }, detail: usize = 0 };
105fn markup(line: []const u8, p: usize) Markup {
106 const tail = line[p..];
107 for (format_tags, 0..) |tag, i| {
108 if (std.mem.startsWith(u8, tail, tag)) return .{ .len = tag.len, .kind = .format, .detail = i };
109 }
110 if (std.mem.startsWith(u8, tail, "</font>")) return .{ .len = 7, .kind = .font };
111 if (std.mem.startsWith(u8, tail, "<font") and tail.len > 5 and (tail[5] == ' ' or tail[5] == '>')) {
112 if (std.mem.findScalar(u8, tail[5..], '>')) |end| return .{ .len = end + 6, .kind = .font };
113 }
114 if (tail.len >= 6 and std.mem.startsWith(u8, tail, "{\\an") and
115 tail[4] >= '1' and tail[4] <= '9' and tail[5] == '}')
116 return .{ .len = 6, .kind = .position, .detail = tail[4] - '0' };
117 return .{ .len = 1, .kind = .ordinary };
118}

173const Editor = struct {
174 gpa: Allocator,
175 cues: []Cue = &.{},
176 history: std.ArrayList(Record) = .empty,
177 blocks: usize = 0,
178 bad: usize = 0,
179 error_buffer: [1024]u8 = undefined,

182 fn fail(self: *Editor, message: []const u8) error{BadCommand} {
183 @memcpy(self.error_buffer[0..message.len], message);
184 self.error_len = message.len;
185 return error.BadCommand;
186 }
187 fn failFmt(self: *Editor, comptime fmt: []const u8, args: anytype) error{BadCommand} {
188 const text = std.fmt.bufPrint(&self.error_buffer, fmt, args) catch unreachable;
189 self.error_len = text.len;
190 return error.BadCommand;
191 }
192 fn parseCommand(self: *Editor, line: []const u8) Error!Command {
193 const fields = try split(self.gpa, line, ' ', false);
194 const op = std.meta.stringToEnum(Op, fields[0]) orelse return self.fail("unknown command");
195 const schema: []const Kind = switch (op) {
196 .SEARCH => &.{.string},
197 .AT => &.{.time},
198 .GAPS, .FIXOVERLAP, .MERGE => &.{.gap},
199 .FRAMES, .SNAP => &.{.fps},
200 .SHIFT => &.{.signed},
201 .SHIFTFROM => &.{ .position, .signed },
202 .SCALE => &.{ .scale, .scale },
203 .SYNC => &.{ .position, .time, .position, .time },
204 .EXTEND => &.{.minimum},
205 .TIME => &.{ .position, .time, .time },
206 .SPLIT => &.{.width},
207 .INSERT => &.{ .position, .time, .time, .text },
208 .DELETE => if (fields.len == 3) &.{ .position, .position } else &.{.position},
209 .TEXT => &.{ .position, .text },
210 .REPLACE => &.{.replacement},
211 else => &.{},
212 };
213 const string_last = schema.len > 0 and switch (schema[schema.len - 1]) {
214 .string, .text, .replacement => true,
215 else => false,
216 };
217 if (fields.len < schema.len + 1 or (!string_last and fields.len != schema.len + 1))
218 return self.fail("bad arguments");
219 var command = Command{ .op = op, .count = schema.len };
220 for (schema, 0..) |kind, i| {
221 const raw = if (string_last and i == schema.len - 1)
222 try std.mem.join(self.gpa, " ", fields[i + 1 ..])
223 else
224 fields[i + 1];
225 var arg = Arg{};
226 switch (kind) {
227 .string => arg.string = raw,
228 .text, .replacement => {
229 arg.lines = try split(self.gpa, raw, '|', false);
230 for (arg.lines) |part| {
231 if (part.len == 0) return self.fail("bad arguments");
232 }
233 if (kind == .replacement and arg.lines.len != 2) return self.fail("bad arguments");
234 },
235 .fps => {
236 const parts = try split(self.gpa, raw, '/', false);
237 if (parts.len > 2) return self.fail("bad arguments");
238 arg.a = unsigned(parts[0], 9) orelse return self.fail("bad arguments");
239 arg.b = if (parts.len == 1) 1 else unsigned(parts[1], 9) orelse return self.fail("bad arguments");
240 if (arg.a < 1 or arg.a > 100000 or arg.b < 1 or arg.b > 100000) return self.fail("bad arguments");
241 },
242 else => {
243 arg.number = if (kind == .time) timeValue(raw) orelse return self.fail("bad arguments") else if (kind == .signed and raw.len > 0 and raw[0] == '-')
244 -(unsigned(raw[1..], 9) orelse return self.fail("bad arguments"))
245 else
246 unsigned(raw, 9) orelse return self.fail("bad arguments");
247 const value = arg.number;
248 if ((kind == .position and value < 1) or (kind == .gap and value > 60000) or
249 (kind == .minimum and (value < 1 or value > 60000)) or
250 (kind == .width and (value < 10 or value > 80)) or
251 (kind == .scale and (value < 1 or value > 100000))) return self.fail("bad arguments");
252 },
253 }
254 command.args[i] = arg;
255 }
256 if (op == .DELETE and command.count == 2 and command.n(1) < command.n(0)) return self.fail("bad arguments");
257 for (schema, 0..) |kind, i| {
258 const limit = self.cues.len + @as(usize, if (op == .INSERT) 1 else 0);
259 if (kind == .position and command.n(i) > @as(i64, @intCast(limit)))
260 return self.failFmt("no cue {d}", .{command.n(i)});
261 }
262 return command;
263 }
264 fn load(self: *Editor, block: []const []const u8, out: *std.Io.Writer) Error!void {
265 if (block.len == 0) return;
266 self.blocks += 1;
267 const index = unsigned(block[0], 6);
268 var start: ?i64 = null;
269 var end: ?i64 = null;
270 var load_problem: ?[]const u8 = null;
271 if (index == null) load_problem = "index" else if (block.len < 2 or block[1].len != 29 or
272 !std.mem.eql(u8, block[1][12..17], " --> ")) load_problem = "time" else {
273 start = timeValue(block[1][0..12]);
274 end = timeValue(block[1][17..]);
275 if (start == null or end == null) load_problem = "time" else if (block.len < 3) load_problem = "no text";
276 }
277 if (load_problem) |message| {
278 self.bad += 1;
279 try out.print("BAD CUE {d}: {s}\n", .{ self.blocks, message });
280 } else {
281 const cues = try self.gpa.alloc(Cue, self.cues.len + 1);
282 @memcpy(cues[0..self.cues.len], self.cues);
283 cues[self.cues.len] = .{ .index = index.?, .start = start.?, .end = end.?, .text = try self.gpa.dupe([]const u8, block[2..]) };
284 self.cues = cues;
285 }
286 }
287 fn snapshot(self: *Editor) Allocator.Error![]Cue {
288 const result = try self.gpa.dupe(Cue, self.cues);
289 for (result) |*c| c.text = try self.gpa.dupe([]const u8, c.text);
290 return result;
291 }
292 fn listCue(self: *Editor, out: *std.Io.Writer, i: usize) Error!void {
293 const c = self.cues[i];
294 try out.print("{d} {d} {s} {s} {s}\n", .{ i + 1, c.index, try timeText(self.gpa, c.start, false), try timeText(self.gpa, c.end, false), try std.mem.join(self.gpa, "|", c.text) });
295 }
296 fn problem(out: *std.Io.Writer, count: *usize, comptime fmt: []const u8, args: anytype) Error!void {
297 try out.print(fmt, args);
298 try out.writeByte('\n');
299 count.* += 1;
300 }
301 fn execute(self: *Editor, line: []const u8, out: *std.Io.Writer) Error!void {
302 const command = try self.parseCommand(line);
303 const op = command.op;
304 const args = command.args;
305 if (op == .SYNC and self.cues[command.p(0)].start == self.cues[command.p(2)].start)
306 return self.fail("sync points coincide");
307 if (op == .UNDO) {
308 if (self.history.items.len == 0) return self.fail("nothing to undo");
309 const entry = self.history.pop().?;
310 self.cues = entry.cues;
311 try out.print("UNDO command={s}\n", .{@tagName(entry.op)});
312 return;
313 }
314 switch (op) {
315 .SHIFT, .SHIFTFROM, .SCALE, .SYNC, .SNAP, .FIXOVERLAP, .EXTEND, .TIME, .SORT, .RENUMBER, .MERGE, .SPLIT, .INSERT, .DELETE, .TEXT, .BALANCE, .STRIP, .CLEANHI, .REPLACE => try self.history.append(self.gpa, .{ .op = op, .cues = try self.snapshot() }),
316 else => {},
317 }
318 switch (op) {
319 .LIST => for (0..self.cues.len) |i| try self.listCue(out, i),
320 .VALIDATE => {
321 var problems: usize = 0;
322 for (self.cues, 0..) |c, i| {
323 const p = i + 1;
324 const d = c.duration();
325 if (c.index != @as(i64, @intCast(p))) try problem(out, &problems, "cue {d}: index {d} (expected {d})", .{ p, c.index, p });
326 if (d <= 0) try problem(out, &problems, "cue {d}: end not after start", .{p});
327 if (i > 0 and c.start < self.cues[i - 1].end) try problem(out, &problems, "cue {d}: overlaps previous by {d} ms", .{ p, self.cues[i - 1].end - c.start });
328 if (d >= 1 and d < 700) try problem(out, &problems, "cue {d}: too short ({d} ms)", .{ p, d });
329 if (d > 7000) try problem(out, &problems, "cue {d}: too long ({d} ms)", .{ p, d });
330 for (c.text, 0..) |text, j| {
331 if (text.len > 42) try problem(out, &problems, "cue {d}: line {d} too long ({d} chars)", .{ p, j + 1, text.len });
332 }
333 if (c.text.len >= 3) try problem(out, &problems, "cue {d}: too many lines ({d})", .{ p, c.text.len });
334 if (d > 0 and c.characters() * 1000 > 21 * d) try problem(out, &problems, "cue {d}: too fast ({s} cps)", .{ p, try decimal(self.gpa, @divFloor(c.characters() * 100000, d)) });
335 }
336 try out.print("VALIDATE problems={d}\n", .{problems});
337 },
338 .SEARCH => {
339 var found: usize = 0;
340 for (self.cues, 0..) |c, i| {
341 for (c.text) |text| {
342 if (std.mem.find(u8, text, args[0].string) != null) {
343 try out.print("{d} {s} {s}\n", .{ i + 1, try timeText(self.gpa, c.start, false), text });
344 found += 1;
345 break;
346 }
347 }
348 }
349 try out.print("SEARCH found={d}\n", .{found});
350 },
351 .AT => {
352 var active: usize = 0;
353 for (self.cues, 0..) |c, i| {
354 if (c.start <= command.n(0) and command.n(0) < c.end) {
355 try self.listCue(out, i);
356 active += 1;
357 }
358 }
359 try out.print("AT active={d}\n", .{active});
360 },
361 .GAPS => {
362 var count: usize = 0;
363 var i: usize = 0;
364 while (i + 1 < self.cues.len) : (i += 1) {
365 const gap = self.cues[i + 1].start - self.cues[i].end;
366 if (gap < command.n(0)) {
367 try out.print("{d} {d} {d}\n", .{ i + 1, i + 2, gap });
368 count += 1;
369 }
370 }
371 try out.print("GAPS count={d}\n", .{count});
372 },
373 .STATS => {
374 try out.print("STATS cues={d}\n", .{self.cues.len});
375 if (self.cues.len == 0) return;
376 var total: i64 = 0;
377 var chars: i64 = 0;
378 var longest: usize = 0;
379 var maximum: i64 = -1;
380 for (self.cues, 0..) |c, i| {
381 const d = @max(0, c.duration());
382 total += d;
383 if (d > 0) chars += c.characters();
384 if (d > maximum) {
385 maximum = d;
386 longest = i;
387 }
388 }
389 try out.print("duration total={s} mean={s}\n", .{ try timeText(self.gpa, total, false), try timeText(self.gpa, @divFloor(total, @as(i64, @intCast(self.cues.len))), false) });
390 try out.print("speed {s}\n", .{if (total == 0) "-" else try decimal(self.gpa, @divFloor(chars * 100000, total))});
391 try out.print("longest cue={d} duration={s}\n", .{ longest + 1, try timeText(self.gpa, maximum, false) });
392 if (self.cues.len == 1) try out.writeAll("gap none\n") else {
393 var position: usize = 0;
394 var gap = self.cues[1].start - self.cues[0].end;
395 var i: usize = 1;
396 while (i + 1 < self.cues.len) : (i += 1) {
397 const candidate = self.cues[i + 1].start - self.cues[i].end;
398 if (candidate > gap) {
399 gap = candidate;
400 position = i;
401 }
402 }
403 try out.print("gap cue={d} ms={d}\n", .{ position + 1, gap });
404 }
405 },
406 .PRINT => for (self.cues) |c| {
407 try out.print("{d}\n{s} --> {s}\n", .{ c.index, try timeText(self.gpa, c.start, false), try timeText(self.gpa, c.end, false) });
408 for (c.text) |text| try out.print("{s}\n", .{text});
409 try out.writeByte('\n');
410 },
411 .TOVTT => {
412 try out.writeAll("WEBVTT\n\n");
413 for (self.cues) |c| {
414 var lines: std.ArrayList([]const u8) = .empty;
415 var code: usize = 0;
416 for (c.text) |text| {
417 var bytes: std.ArrayList(u8) = .empty;
418 var p: usize = 0;
419 while (p < text.len) {
420 const item = markup(text, p);
421 if (item.kind == .position and code == 0) code = item.detail;
422 if (item.kind == .format) try bytes.appendSlice(self.gpa, text[p .. p + item.len]);
423 if (item.kind == .ordinary) try bytes.appendSlice(self.gpa, switch (text[p]) {
424 '&' => "&amp;",
425 '<' => "&lt;",
426 '>' => "&gt;",
427 else => text[p .. p + 1],
428 });
429 p += item.len;
430 }
431 const converted = std.mem.trim(u8, bytes.items, " ");
432 if (converted.len > 0) try lines.append(self.gpa, converted);
433 }
434 if (lines.items.len == 0) continue;
435 try out.print("{d}\n{s} --> {s}", .{ c.index, try timeText(self.gpa, c.start, true), try timeText(self.gpa, c.end, true) });
436 if (code >= 7) try out.writeAll(" line:0") else if (code >= 4) try out.writeAll(" line:50%");
437 if (code == 1 or code == 4 or code == 7) try out.writeAll(" align:start");
438 if (code == 3 or code == 6 or code == 9) try out.writeAll(" align:end");
439 try out.writeByte('\n');
440 for (lines.items) |text| try out.print("{s}\n", .{text});
441 try out.writeByte('\n');
442 }
443 },
444 .TOASS => {
445 try out.writeAll("[Events]\nFormat: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text\n");
446 for (self.cues) |c| {
447 try out.print("Dialogue: 0,{s},{s},Default,,0,0,0,,", .{ try assTime(self.gpa, c.start), try assTime(self.gpa, c.end) });
448 for (c.text, 0..) |text, i| {
449 if (i > 0) try out.writeAll("\\N");
450 var p: usize = 0;
451 while (p < text.len) {
452 const item = markup(text, p);
453 if (item.kind == .format) try out.writeAll(ass_tags[item.detail]) else if (item.kind != .font)
454 try out.writeAll(text[p .. p + item.len]);
455 p += item.len;
456 }
457 }
458 try out.writeByte('\n');
459 }
460 },
461 .FRAMES => for (self.cues, 0..) |c, i| {
462 try out.print("{d} {d} {d}\n", .{ i + 1, @divFloor(c.start * args[0].a, 1000 * args[0].b), @divFloor(c.end * args[0].a, 1000 * args[0].b) });
463 },
464 .SHIFT, .SHIFTFROM, .SCALE, .SYNC, .SNAP => {
465 const first = if (op == .SHIFTFROM) command.p(0) else 0;
466 var clamped: usize = 0;
467 var changed: usize = 0;
468 const s1 = if (op == .SYNC) self.cues[command.p(0)].start else 0;
469 const s2 = if (op == .SYNC) self.cues[command.p(2)].start else 0;
470 for (self.cues[first..]) |*c| {
471 var clamp = false;
472 const before = [_]i64{ c.start, c.end };
473 var after: [2]i64 = undefined;
474 for (before, 0..) |t, i| {
475 var value: i64 = switch (op) {
476 .SCALE => roundRatio(t * command.n(0), command.n(1)),
477 .SYNC => roundRatio(command.n(1) * (s2 - s1) + (t - s1) * (command.n(3) - command.n(1)), s2 - s1),
478 .SNAP => roundRatio(roundRatio(t * args[0].a, 1000 * args[0].b) * 1000 * args[0].b, args[0].a),
479 else => t + command.n(if (op == .SHIFT) 0 else 1),
480 };
481 if (value < 0) {
482 value = 0;
483 clamp = true;
484 }
485 if (value != t) changed += 1;
486 after[i] = value;
487 }
488 c.start = after[0];
489 c.end = after[1];
490 if (clamp) clamped += 1;
491 }
492 if (op == .SNAP) try out.print("SNAP changed={d}\n", .{changed}) else if (op == .SCALE)
493 try out.print("SCALE cues={d}\n", .{self.cues.len})
494 else
495 try out.print("{s} cues={d} clamped={d}\n", .{ @tagName(op), self.cues.len - first, clamped });
496 },
497 .FIXOVERLAP => {
498 var fixed: usize = 0;
499 var failed: usize = 0;
500 var i: usize = 0;
501 while (i + 1 < self.cues.len) : (i += 1) {
502 const candidate = self.cues[i + 1].start - command.n(0);
503 if (self.cues[i].end > candidate) {
504 if (candidate > self.cues[i].start) {
505 self.cues[i].end = candidate;
506 fixed += 1;
507 } else {
508 try out.print("cue {d}: cannot fix\n", .{i + 1});
509 failed += 1;
510 }
511 }
512 }
513 try out.print("FIXOVERLAP fixed={d} failed={d}\n", .{ fixed, failed });
514 },
515 .EXTEND => {
516 var extended: usize = 0;
517 var short: usize = 0;
518 for (self.cues, 0..) |*c, i| {
519 if (c.duration() < command.n(0)) {
520 var candidate = c.start + command.n(0);
521 if (i + 1 < self.cues.len) candidate = @min(candidate, self.cues[i + 1].start);
522 if (candidate > c.end) {
523 c.end = candidate;
524 extended += 1;
525 }
526 }
527 if (c.duration() < command.n(0)) short += 1;
528 }
529 try out.print("EXTEND extended={d} short={d}\n", .{ extended, short });
530 },
531 .TIME => {
532 self.cues[command.p(0)].start = command.n(1);
533 self.cues[command.p(0)].end = command.n(2);
534 try out.print("TIME cue={d}\n", .{command.n(0)});
535 },
536 .SORT => {
537 const Entry = struct {
538 cue: Cue,
539 position: usize,
540
541 fn less(_: void, a: @This(), b: @This()) bool {
542 return if (a.cue.start != b.cue.start) a.cue.start < b.cue.start else a.cue.end < b.cue.end;
543 }
544 };
545 const ordered = try self.gpa.alloc(Entry, self.cues.len);
546 for (self.cues, 0..) |c, i|
547 ordered[i] = .{ .cue = c, .position = i };
548 std.mem.sort(Entry, ordered, {}, Entry.less);
549 var moved: usize = 0;
550 for (ordered, 0..) |entry, i| {
551 if (entry.position != i) moved += 1;
552 self.cues[i] = entry.cue;
553 }
554 try out.print("SORT moved={d}\n", .{moved});
555 },
556 .RENUMBER => {
557 var changed: usize = 0;
558 for (self.cues, 0..) |*c, i| {
559 if (c.index != @as(i64, @intCast(i + 1))) {
560 c.index = @intCast(i + 1);
561 changed += 1;
562 }
563 }
564 try out.print("RENUMBER changed={d}\n", .{changed});
565 },
566 .MERGE => {
567 var result: std.ArrayList(Cue) = .empty;
568 var merged: usize = 0;
569 for (self.cues) |b| {
570 if (result.items.len > 0) {
571 const a = &result.items[result.items.len - 1];
572 const combined = try std.mem.concat(self.gpa, []const u8, &.{ a.text, b.text });
573 const wrapped = try wrap(self.gpa, combined, 42);
574 if (b.start - a.end < command.n(0) and !a.dialogue() and !b.dialogue() and wrapped.len <= 2) {
575 a.end = @max(a.end, b.end);
576 a.text = wrapped;
577 merged += 1;
578 continue;
579 }
580 }
581 try result.append(self.gpa, b);
582 }
583 self.cues = result.items;
584 try out.print("MERGE merged={d} cues={d}\n", .{ merged, self.cues.len });
585 },
586 .SPLIT => {
587 var result: std.ArrayList(Cue) = .empty;
588 var count: usize = 0;
589 for (self.cues) |original| {
590 var c = original;
591 if (c.duration() < 1 or c.dialogue()) {
592 try result.append(self.gpa, c);
593 continue;
594 }
595 const lines = try wrap(self.gpa, c.text, @intCast(command.n(0)));
596 if (lines.len <= 2) {
597 c.text = lines;
598 try result.append(self.gpa, c);
599 continue;
600 }
601 count += 1;
602 const len = (lines.len + 1) / 2;
603 const counts = try self.gpa.alloc(i64, len);
604 const times = try self.gpa.alloc(i64, len);
605 const order = try self.gpa.alloc(usize, len);
606 const remainders = try self.gpa.alloc(i64, len);
607 var total: i64 = 0;
608 for (0..len) |i| {
609 counts[i] = 0;
610 for (lines[i * 2 .. @min(i * 2 + 2, lines.len)]) |text| counts[i] += @intCast(text.len);
611 total += counts[i];
612 order[i] = i;
613 }
614 var allotted: i64 = 0;
615 for (0..len) |i| {
616 const share = c.duration() * counts[i];
617 times[i] = @divFloor(share, total);
618 remainders[i] = @mod(share, total);
619 allotted += times[i];
620 }
621 std.mem.sort(usize, order, @as([]const i64, remainders), remainderLess);
622 for (0..@as(usize, @intCast(c.duration() - allotted))) |i| times[order[i]] += 1;
623 var start = c.start;
624 for (0..len) |i| {
625 try result.append(self.gpa, .{ .index = c.index, .start = start, .end = start + times[i], .text = lines[i * 2 .. @min(i * 2 + 2, lines.len)] });
626 start += times[i];
627 }
628 }
629 self.cues = result.items;
630 try out.print("SPLIT split={d} cues={d}\n", .{ count, self.cues.len });
631 },
632 .INSERT => {
633 const p = command.p(0);
634 const result = try self.gpa.alloc(Cue, self.cues.len + 1);
635 @memcpy(result[0..p], self.cues[0..p]);
636 result[p] = .{ .index = command.n(0), .start = command.n(1), .end = command.n(2), .text = args[3].lines };
637 @memcpy(result[p + 1 ..], self.cues[p..]);
638 self.cues = result;
639 try out.print("INSERT cue={d} cues={d}\n", .{ command.n(0), self.cues.len });
640 },
641 .DELETE => {
642 const first = command.p(0);
643 const end = if (command.count == 2) command.p(1) + 1 else first + 1;
644 const result = try self.gpa.alloc(Cue, self.cues.len - (end - first));
645 @memcpy(result[0..first], self.cues[0..first]);
646 @memcpy(result[first..], self.cues[end..]);
647 self.cues = result;
648 try out.print("DELETE removed={d} cues={d}\n", .{ end - first, self.cues.len });
649 },
650 .TEXT => {
651 self.cues[command.p(0)].text = args[1].lines;
652 try out.print("TEXT cue={d} lines={d}\n", .{ command.n(0), args[1].lines.len });
653 },
654 .BALANCE => {
655 var changed: usize = 0;
656 var skipped: usize = 0;
657 var dialogue: usize = 0;
658 for (self.cues) |*c| {
659 if (c.dialogue()) {
660 dialogue += 1;
661 continue;
662 }
663 const ws = try words(self.gpa, c.text);
664 const text = try std.mem.join(self.gpa, " ", ws);
665 var candidate: ?Strings = null;
666 if (text.len <= 42) candidate = try self.gpa.dupe([]const u8, &.{text}) else {
667 var best_difference: usize = std.math.maxInt(usize);
668 var best_first: usize = std.math.maxInt(usize);
669 for (1..@max(1, ws.len)) |j| {
670 const left = try std.mem.join(self.gpa, " ", ws[0..j]);
671 const right = try std.mem.join(self.gpa, " ", ws[j..]);
672 const difference = @max(left.len, right.len) - @min(left.len, right.len);
673 if (left.len <= 42 and right.len <= 42 and (difference < best_difference or
674 (difference == best_difference and left.len < best_first)))
675 {
676 best_difference = difference;
677 best_first = left.len;
678 candidate = try self.gpa.dupe([]const u8, &.{ left, right });
679 }
680 }
681 }
682 if (candidate) |lines| {
683 if (!sameLines(c.text, lines)) {
684 c.text = lines;
685 changed += 1;
686 }
687 } else skipped += 1;
688 }
689 try out.print("BALANCE changed={d} skipped={d} dialogue={d}\n", .{ changed, skipped, dialogue });
690 },
691 .STRIP, .CLEANHI => {
692 var tags: usize = 0;
693 var segments: usize = 0;
694 var labels: usize = 0;
695 var removed_lines: usize = 0;
696 var removed_cues: usize = 0;
697 var result: std.ArrayList(Cue) = .empty;
698 for (self.cues) |original| {
699 var c = original;
700 var lines: std.ArrayList([]const u8) = .empty;
701 for (c.text) |text| {
702 var bytes: std.ArrayList(u8) = .empty;
703 var p: usize = 0;
704 while (p < text.len) {
705 if (op == .STRIP) {
706 const item = markup(text, p);
707 if (item.kind == .ordinary) try bytes.append(self.gpa, text[p]) else tags += 1;
708 p += item.len;
709 } else {
710 const close: u8 = if (text[p] == '[') ']' else if (text[p] == '(') ')' else 0;
711 const offset = if (close == 0) null else std.mem.findScalar(u8, text[p + 1 ..], close);
712 if (offset) |end| {
713 segments += 1;
714 p += end + 2;
715 } else {
716 try bytes.append(self.gpa, text[p]);
717 p += 1;
718 }
719 }
720 }
721 var line_text = if (op == .STRIP) std.mem.trim(u8, bytes.items, " ") else try normalized(self.gpa, bytes.items);
722 if (op == .CLEANHI) {
723 const dialogue_prefix = std.mem.startsWith(u8, line_text, "- ");
724 const body = if (dialogue_prefix) line_text[2..] else line_text;
725 if (std.mem.findScalar(u8, body, ':')) |colon| {
726 var is_label = colon > 0 and body[0] >= 'A' and body[0] <= 'Z';
727 for (body[0..colon]) |ch| {
728 if (!((ch >= 'A' and ch <= 'Z') or (ch >= '0' and ch <= '9') or ch == ' ')) is_label = false;
729 }
730 if (is_label) {
731 line_text = try normalized(self.gpa, try std.fmt.allocPrint(self.gpa, "{s}{s}", .{
732 @as([]const u8, if (dialogue_prefix) "- " else ""), body[colon + 1 ..],
733 }));
734 labels += 1;
735 }
736 }
737 }
738 if (line_text.len == 0 or (op == .CLEANHI and std.mem.eql(u8, line_text, "-"))) removed_lines += 1 else try lines.append(self.gpa, line_text);
739 }
740 if (lines.items.len == 0) removed_cues += 1 else {
741 c.text = lines.items;
742 try result.append(self.gpa, c);
743 }
744 }
745 self.cues = result.items;
746 if (op == .STRIP) try out.print("STRIP tags={d} removed_lines={d} removed_cues={d}\n", .{ tags, removed_lines, removed_cues }) else try out.print("CLEANHI segments={d} labels={d} removed_lines={d} removed_cues={d}\n", .{ segments, labels, removed_lines, removed_cues });
747 },
748 .REPLACE => {
749 const old = args[0].lines[0];
750 const replacement = args[0].lines[1];
751 var count: usize = 0;
752 var changed_cues: usize = 0;
753 for (self.cues) |*c| {
754 var changed = false;
755 for (c.text) |*text| {
756 const replacements = std.mem.count(u8, text.*, old);
757 if (replacements > 0) {
758 count += replacements;
759 changed = true;
760 text.* = try std.mem.replaceOwned(u8, self.gpa, text.*, old, replacement);
761 }
762 }
763 if (changed) changed_cues += 1;
764 }
765 try out.print("REPLACE count={d} cues={d}\n", .{ count, changed_cues });
766 },
767 .UNDO => unreachable,
768 }
769 }
770};

772pub fn main(init: std.process.Init) !void {
773 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
774 defer arena.deinit();
775 const gpa = arena.allocator();
776 var input_buffer: [4096]u8 = undefined;
777 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
778 var output_buffer: [8192]u8 = undefined;
779 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
780 const out = &output.interface;
781 var editor = Editor{ .gpa = gpa };
782 var block: std.ArrayList([]const u8) = .empty;
783 while (true) {
784 const line = try gpa.dupe(u8, (try input.interface.takeDelimiter('\n')).?);
785 if (std.mem.eql(u8, line, "###")) {
786 try editor.load(block.items, out);
787 break;
788 }
789 if (line.len == 0) {
790 try editor.load(block.items, out);
791 block = .empty;
792 } else try block.append(gpa, line);
793 }
794 try out.print("LOAD cues={d} bad={d}\n", .{ editor.cues.len, editor.bad });
795 const count: usize = @intCast(unsigned((try input.interface.takeDelimiter('\n')).?, 9).?);
796 for (0..count) |_| {
797 const line = try gpa.dupe(u8, (try input.interface.takeDelimiter('\n')).?);
798 var draft: std.Io.Writer.Allocating = .init(gpa);
799 editor.execute(line, &draft.writer) catch |err| {
800 if (err == error.BadCommand) {
801 try out.print("ERROR {s}\n", .{editor.error_buffer[0..editor.error_len]});
802 continue;
803 }
804 return err;
805 };
806 try out.writeAll(draft.writer.buffered());
807 }
808 try out.flush();
809}
