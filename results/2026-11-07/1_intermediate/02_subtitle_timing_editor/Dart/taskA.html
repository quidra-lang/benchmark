10int? unsigned(String text, [int limit = 9]) {
11 if (text.isEmpty || text.length > limit ||
12 text.codeUnits.any((c) => c < 48 || c > 57)) return null;
13 return int.parse(text);
14}

16int? timeValue(String text) {
17 if (text.length != 12 || text[2] != ':' || text[5] != ':' ||
18 text[8] != ',') return null;
19 final h = unsigned(text.substring(0, 2));
20 final m = unsigned(text.substring(3, 5));
21 final s = unsigned(text.substring(6, 8));
22 final ms = unsigned(text.substring(9));
23 if (h == null || m == null || s == null || ms == null || m > 59 || s > 59) return null;
24 return h * 3600000 + m * 60000 + s * 1000 + ms;
25}

27int floorDiv(int a, int b) {
28 final q = a ~/ b;
29 return a % b == 0 || a >= 0 ? q : q - 1;
30}
31
32int roundRatio(int a, int b) {
33 if (b < 0) {
34 a = -a;
35 b = -b;
36 }
37 return floorDiv(2 * a + b, 2 * b);
38}

40String padded(int n, int width) => n.toString().padLeft(width, '0');
41String timeText(int t, [bool vtt = false]) =>
42 '${padded(t ~/ 3600000, 2)}:${padded(t ~/ 60000 % 60, 2)}:'
43 '${padded(t ~/ 1000 % 60, 2)}${vtt ? '.' : ','}${padded(t % 1000, 3)}';
44String assTime(int t) {
45 final c = (t + 5) ~/ 10;
46 return '${c ~/ 360000}:${padded(c ~/ 6000 % 60, 2)}:'
47 '${padded(c ~/ 100 % 60, 2)}.${padded(c % 100, 2)}';
48}
49String decimal(int v) => '${v ~/ 100}.${padded(v % 100, 2)}';

51class Cue {
52 int index, start, end;
53 List<String> text;
54 Cue(this.index, this.start, this.end, this.text);
55 Cue copy() => Cue(index, start, end, List.of(text));
56 int get duration => end - start;
57 int get characters => text.fold(0, (sum, s) => sum + s.length);
58 bool get dialogue => text.any((s) => s.startsWith('- '));

61List<String> words(List<String> lines) =>
62 lines.expand((s) => s.split(' ')).where((s) => s.isNotEmpty).toList();
63List<String> wrap(List<String> lines, int width) {
64 final result = <String>[];
65 for (final word in words(lines)) {
66 if (result.isNotEmpty && result.last.length + 1 + word.length <= width) {
67 result[result.length - 1] += ' $word';
68 } else {
69 result.add(word);
70 }
71 }
72 return result;
73}
74bool sameLines(List<String> a, List<String> b) =>
75 a.length == b.length && a.indexed.every((entry) => entry.$2 == b[entry.$1]);
76String normalized(String s) => s.split(' ').where((w) => w.isNotEmpty).join(' ');

78const formatTags = ['<i>', '</i>', '<b>', '</b>', '<u>', '</u>'];
79const assTags = [r'{\i1}', r'{\i0}', r'{\b1}', r'{\b0}', r'{\u1}', r'{\u0}'];
80(int, int, int) markup(String line, int p) {
81 for (var i = 0; i < formatTags.length; i++) {
82 if (line.startsWith(formatTags[i], p)) return (formatTags[i].length, 0, i);
83 }
84 if (line.startsWith('</font>', p)) return (7, 1, 0);
85 if (line.startsWith('<font', p) && p + 5 < line.length &&
86 (line[p + 5] == ' ' || line[p + 5] == '>')) {
87 final end = line.indexOf('>', p + 5);
88 if (end >= 0) return (end - p + 1, 1, 0);
89 }
90 if (p + 6 <= line.length && line.startsWith(r'{\an', p) &&
91 line.codeUnitAt(p + 4) >= 49 && line.codeUnitAt(p + 4) <= 57 && line[p + 5] == '}') {
92 return (6, 2, line.codeUnitAt(p + 4) - 48);
93 }
94 return (1, 3, 0);
95}

107const schemas = <String, List<String>>{
108 'LIST': [], 'VALIDATE': [], 'SEARCH': ['string'], 'AT': ['time'],
109 'GAPS': ['gap'], 'STATS': [], 'PRINT': [], 'TOVTT': [], 'TOASS': [],
110 'FRAMES': ['fps'], 'SHIFT': ['signed'], 'SHIFTFROM': ['position', 'signed'],
111 'SCALE': ['scale', 'scale'], 'SYNC': ['position', 'time', 'position', 'time'],
112 'SNAP': ['fps'], 'FIXOVERLAP': ['gap'], 'EXTEND': ['minimum'],
113 'TIME': ['position', 'time', 'time'], 'SORT': [], 'RENUMBER': [],
114 'MERGE': ['gap'], 'SPLIT': ['width'], 'INSERT': ['position', 'time', 'time', 'text'],
115 'DELETE': ['position'], 'TEXT': ['position', 'text'], 'BALANCE': [],
116 'STRIP': [], 'CLEANHI': [], 'REPLACE': ['replacement'], 'UNDO': [],
117};
118const modifying = ['SHIFT', 'SHIFTFROM', 'SCALE', 'SYNC', 'SNAP', 'FIXOVERLAP',
119 'EXTEND', 'TIME', 'SORT', 'RENUMBER', 'MERGE', 'SPLIT', 'INSERT', 'DELETE',
120 'TEXT', 'BALANCE', 'STRIP', 'CLEANHI', 'REPLACE'];

122Command parseCommand(String line, int cueCount) {
123 final parts = line.split(' '), name = parts.first;
124 final declared = schemas[name];
125 if (declared == null) throw CommandError('unknown command');
126 final schema = List<String>.of(declared);
127 if (name == 'DELETE' && parts.length == 3) schema.add('position');
128 final stringLast = schema.isNotEmpty &&
129 ['string', 'text', 'replacement'].contains(schema.last);
130 if (parts.length < schema.length + 1 ||
131 (!stringLast && parts.length != schema.length + 1)) badArguments();
132 final raw = parts.skip(1).toList();
133 if (stringLast) {
134 final joined = raw.skip(schema.length - 1).join(' ');
135 raw.removeRange(schema.length - 1, raw.length);
136 raw.add(joined);
137 }
138 final values = <Object>[];
139 for (var i = 0; i < schema.length; i++) {
140 final type = schema[i], s = raw[i];
141 if (type == 'string') {
142 values.add(s);
143 continue;
144 }
145 if (type == 'text' || type == 'replacement') {
146 final lines = s.split('|');
147 if (lines.any((l) => l.isEmpty) || type == 'replacement' && lines.length != 2) badArguments();
148 values.add(lines);
149 continue;
150 }
151 if (type == 'fps') {
152 final rate = s.split('/');
153 if (rate.length > 2) badArguments();
154 final a = unsigned(rate[0]), b = rate.length == 1 ? 1 : unsigned(rate[1]);
155 if (a == null || b == null || a < 1 || a > 100000 || b < 1 || b > 100000) badArguments();
156 values.add((a, b));
157 continue;
158 }
159 int? value;
160 if (type == 'time') {
161 value = timeValue(s);
162 } else if (type == 'signed' && s.startsWith('-')) {
163 final magnitude = unsigned(s.substring(1));
164 value = magnitude == null ? null : -magnitude;
165 } else {
166 value = unsigned(s);
167 }
168 if (value == null) badArguments();
169 if (type == 'position' && value < 1 ||
170 type == 'gap' && value > 60000 ||
171 type == 'minimum' && (value < 1 || value > 60000) ||
172 type == 'scale' && (value < 1 || value > 100000) ||
173 type == 'width' && (value < 10 || value > 80)) badArguments();
174 values.add(value);
175 }
176 final command = Command(name, values);
177 if (name == 'DELETE' && values.length == 2 && command.number(1) < command.number(0)) badArguments();
178 for (var i = 0; i < schema.length; i++) {
179 if (schema[i] == 'position' && command.number(i) > cueCount + (name == 'INSERT' ? 1 : 0)) {
180 throw CommandError('no cue ${command.number(i)}');
181 }
182 }
183 return command;
184}

186class Editor {
187 List<Cue> cues = [];
188 final history = <(String, List<Cue>)>[];
189 int blocks = 0, bad = 0;
190 void load(List<String> block, StringBuffer out) {
191 if (block.isEmpty) return;
192 blocks++;
193 String? problem;
194 final index = unsigned(block[0], 6);
195 int? start, end;
196 if (index == null) {
197 problem = 'index';
198 } else if (block.length < 2 || block[1].length != 29 || block[1].substring(12, 17) != ' --> ') {
199 problem = 'time';
200 } else {
201 start = timeValue(block[1].substring(0, 12));
202 end = timeValue(block[1].substring(17));
203 if (start == null || end == null) problem = 'time';
204 else if (block.length < 3) problem = 'no text';
205 }
206 if (problem != null) {
207 bad++;
208 out.writeln('BAD CUE $blocks: $problem');
209 } else {
210 cues.add(Cue(index!, start!, end!, block.skip(2).toList()));
211 }
212 }
213 void listCue(StringBuffer out, int i) {
214 final c = cues[i];
215 out.writeln('${i + 1} ${c.index} ${timeText(c.start)} ${timeText(c.end)} ${c.text.join('|')}');
216 }
217 String execute(String line) {
218 final command = parseCommand(line, cues.length), op = command.name;
219 int n(int i) => command.number(i);
220 if (op == 'SYNC' && cues[n(0) - 1].start == cues[n(2) - 1].start) {
221 throw CommandError('sync points coincide');
222 }
223 if (op == 'UNDO') {
224 if (history.isEmpty) throw CommandError('nothing to undo');
225 final entry = history.removeLast();
226 cues = entry.$2;
227 return 'UNDO command=${entry.$1}\n';
228 }
229 if (modifying.contains(op)) history.add((op, cues.map((c) => c.copy()).toList()));
230 final out = StringBuffer();
231 switch (op) {
232 case 'LIST':
233 for (var i = 0; i < cues.length; i++) listCue(out, i);
234 case 'VALIDATE':
235 var problems = 0;
236 void problem(String s) {
237 out.writeln(s);
238 problems++;
239 }
240 for (var i = 0; i < cues.length; i++) {
241 final c = cues[i], p = i + 1, d = c.duration;
242 if (c.index != p) problem('cue $p: index ${c.index} (expected $p)');
243 if (d <= 0) problem('cue $p: end not after start');
244 if (i > 0 && c.start < cues[i - 1].end) problem('cue $p: overlaps previous by ${cues[i - 1].end - c.start} ms');
245 if (d >= 1 && d < 700) problem('cue $p: too short ($d ms)');
246 if (d > 7000) problem('cue $p: too long ($d ms)');
247 for (var j = 0; j < c.text.length; j++) {
248 if (c.text[j].length > 42) problem('cue $p: line ${j + 1} too long (${c.text[j].length} chars)');
249 }
250 if (c.text.length >= 3) problem('cue $p: too many lines (${c.text.length})');
251 if (d > 0 && c.characters * 1000 > 21 * d) problem('cue $p: too fast (${decimal(c.characters * 100000 ~/ d)} cps)');
252 }
253 out.writeln('VALIDATE problems=$problems');
254 case 'SEARCH':
255 var found = 0;
256 for (var i = 0; i < cues.length; i++) {
257 final matches = cues[i].text.where((s) => s.contains(command.string(0)));
258 if (matches.isNotEmpty) {
259 out.writeln('${i + 1} ${timeText(cues[i].start)} ${matches.first}');
260 found++;
261 }
262 }
263 out.writeln('SEARCH found=$found');
264 case 'AT':
265 var active = 0;
266 for (var i = 0; i < cues.length; i++) {
267 if (cues[i].start <= n(0) && n(0) < cues[i].end) {
268 listCue(out, i);
269 active++;
270 }
271 }
272 out.writeln('AT active=$active');
273 case 'GAPS':
274 var count = 0;
275 for (var i = 0; i + 1 < cues.length; i++) {
276 final gap = cues[i + 1].start - cues[i].end;
277 if (gap < n(0)) {
278 out.writeln('${i + 1} ${i + 2} $gap');
279 count++;
280 }
281 }
282 out.writeln('GAPS count=$count');
283 case 'STATS':
284 out.writeln('STATS cues=${cues.length}');
285 if (cues.isNotEmpty) {
286 var total = 0, chars = 0, longest = 0, maximum = -1;
287 for (var i = 0; i < cues.length; i++) {
288 final d = cues[i].duration < 0 ? 0 : cues[i].duration;
289 total += d;
290 if (d > 0) chars += cues[i].characters;
291 if (d > maximum) {
292 maximum = d;
293 longest = i;
294 }
295 }
296 out.writeln('duration total=${timeText(total)} mean=${timeText(total ~/ cues.length)}');
297 out.writeln('speed ${total == 0 ? '-' : decimal(chars * 100000 ~/ total)}');
298 out.writeln('longest cue=${longest + 1} duration=${timeText(maximum)}');
299 if (cues.length == 1) out.writeln('gap none');
300 else {
301 var position = 0, gap = cues[1].start - cues[0].end;
302 for (var i = 1; i + 1 < cues.length; i++) {
303 final candidate = cues[i + 1].start - cues[i].end;
304 if (candidate > gap) {
305 gap = candidate;
306 position = i;
307 }
308 }
309 out.writeln('gap cue=${position + 1} ms=$gap');
310 }
311 }
312 case 'PRINT':
313 for (final c in cues) {
314 out.writeln(c.index);
315 out.writeln('${timeText(c.start)} --> ${timeText(c.end)}');
316 for (final s in c.text) out.writeln(s);
317 out.writeln();
318 }
319 case 'TOVTT':
320 out.write('WEBVTT\n\n');
321 for (final c in cues) {
322 final lines = <String>[];
323 var code = 0;
324 for (final s in c.text) {
325 final converted = StringBuffer();
326 for (var p = 0; p < s.length;) {
327 final item = markup(s, p);
328 if (item.$2 == 2 && code == 0) code = item.$3;
329 if (item.$2 == 0) converted.write(s.substring(p, p + item.$1));
330 if (item.$2 == 3) converted.write(s[p] == '&' ? '&amp;' : s[p] == '<' ? '&lt;' : s[p] == '>' ? '&gt;' : s[p]);
331 p += item.$1;
332 }
333 final text = converted.toString().trim();
334 if (text.isNotEmpty) lines.add(text);
335 }
336 if (lines.isEmpty) continue;
337 var settings = code >= 7 ? ' line:0' : code >= 4 ? ' line:50%' : '';
338 if ([1, 4, 7].contains(code)) settings += ' align:start';
339 if ([3, 6, 9].contains(code)) settings += ' align:end';
340 out.writeln(c.index);
341 out.writeln('${timeText(c.start, true)} --> ${timeText(c.end, true)}$settings');
342 for (final text in lines) out.writeln(text);
343 out.writeln();
344 }
345 case 'TOASS':
346 out.writeln('[Events]');
347 out.writeln('Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text');
348 for (final c in cues) {
349 final lines = <String>[];
350 for (final s in c.text) {
351 final converted = StringBuffer();
352 for (var p = 0; p < s.length;) {
353 final item = markup(s, p);
354 if (item.$2 == 0) converted.write(assTags[item.$3]);
355 else if (item.$2 != 1) converted.write(s.substring(p, p + item.$1));
356 p += item.$1;
357 }
358 lines.add(converted.toString());
359 }
360 out.writeln('Dialogue: 0,${assTime(c.start)},${assTime(c.end)},Default,,0,0,0,,${lines.join(r'\N')}');
361 }
362 case 'FRAMES':
363 final rate = command.rate(0);
364 for (var i = 0; i < cues.length; i++) out.writeln('${i + 1} ${cues[i].start * rate.$1 ~/ (1000 * rate.$2)} ${cues[i].end * rate.$1 ~/ (1000 * rate.$2)}');
365 case 'SHIFT' || 'SHIFTFROM' || 'SCALE' || 'SYNC' || 'SNAP':
366 final first = op == 'SHIFTFROM' ? n(0) - 1 : 0;
367 var clamped = 0, changed = 0;
368 final s1 = op == 'SYNC' ? cues[n(0) - 1].start : 0;
369 final s2 = op == 'SYNC' ? cues[n(2) - 1].start : 0;
370 for (var i = first; i < cues.length; i++) {
371 final c = cues[i], before = [c.start, c.end], after = <int>[];
372 var clamp = false;
373 for (final t in before) {
374 int value;
375 if (op == 'SCALE') value = roundRatio(t * n(0), n(1));
376 else if (op == 'SYNC') value = roundRatio(n(1) * (s2 - s1) + (t - s1) * (n(3) - n(1)), s2 - s1);
377 else if (op == 'SNAP') {
378 final rate = command.rate(0);
379 final f = roundRatio(t * rate.$1, 1000 * rate.$2);
380 value = roundRatio(f * 1000 * rate.$2, rate.$1);
381 } else value = t + n(op == 'SHIFT' ? 0 : 1);
382 if (value < 0) {
383 value = 0;
384 clamp = true;
385 }
386 if (value != t) changed++;
387 after.add(value);
388 }
389 c.start = after[0];
390 c.end = after[1];
391 if (clamp) clamped++;
392 }
393 if (op == 'SNAP') out.writeln('SNAP changed=$changed');
394 else if (op == 'SCALE') out.writeln('SCALE cues=${cues.length}');
395 else out.writeln('$op cues=${cues.length - first} clamped=$clamped');
396 case 'FIXOVERLAP':
397 var fixed = 0, failed = 0;
398 for (var i = 0; i + 1 < cues.length; i++) {
399 final candidate = cues[i + 1].start - n(0);
400 if (cues[i].end > candidate) {
401 if (candidate > cues[i].start) {
402 cues[i].end = candidate;
403 fixed++;
404 } else {
405 out.writeln('cue ${i + 1}: cannot fix');
406 failed++;
407 }
408 }
409 }
410 out.writeln('FIXOVERLAP fixed=$fixed failed=$failed');
411 case 'EXTEND':
412 var extended = 0, short = 0;
413 for (var i = 0; i < cues.length; i++) {
414 final c = cues[i];
415 if (c.duration < n(0)) {
416 var candidate = c.start + n(0);
417 if (i + 1 < cues.length && cues[i + 1].start < candidate) candidate = cues[i + 1].start;
418 if (candidate > c.end) {
419 c.end = candidate;
420 extended++;
421 }
422 }
423 if (c.duration < n(0)) short++;
424 }
425 out.writeln('EXTEND extended=$extended short=$short');
426 case 'TIME':
427 cues[n(0) - 1].start = n(1);
428 cues[n(0) - 1].end = n(2);
429 out.writeln('TIME cue=${n(0)}');
430 case 'SORT':
431 final ordered = [for (var i = 0; i < cues.length; i++) (cue: cues[i], original: i)];
432 ordered.sort((a, b) {
433 final start = a.cue.start.compareTo(b.cue.start), end = a.cue.end.compareTo(b.cue.end);
434 return start != 0 ? start : end != 0 ? end : a.original.compareTo(b.original);
435 });
436 final moved = ordered.indexed.where((entry) => entry.$2.original != entry.$1).length;
437 cues = ordered.map((entry) => entry.cue).toList();
438 out.writeln('SORT moved=$moved');
439 case 'RENUMBER':
440 var changed = 0;
441 for (var i = 0; i < cues.length; i++) {
442 if (cues[i].index != i + 1) {
443 cues[i].index = i + 1;
444 changed++;
445 }
446 }
447 out.writeln('RENUMBER changed=$changed');
448 case 'MERGE':
449 final result = <Cue>[];
450 var merged = 0;
451 for (final b in cues) {
452 if (result.isNotEmpty) {
453 final a = result.last, combined = wrap([...a.text, ...b.text], 42);
454 if (b.start - a.end < n(0) && !a.dialogue && !b.dialogue && combined.length <= 2) {
455 if (b.end > a.end) a.end = b.end;
456 a.text = combined;
457 merged++;
458 continue;
459 }
460 }
461 result.add(b);
462 }
463 cues = result;
464 out.writeln('MERGE merged=$merged cues=${cues.length}');
465 case 'SPLIT':
466 final result = <Cue>[];
467 var split = 0;
468 for (final c in cues) {
469 if (c.duration < 1 || c.dialogue) {
470 result.add(c);
471 continue;
472 }
473 final lines = wrap(c.text, n(0));
474 if (lines.length <= 2) {
475 c.text = lines;
476 result.add(c);
477 continue;
478 }
479 split++;
480 final parts = [for (var i = 0; i < lines.length; i += 2) lines.sublist(i, i + 2 < lines.length ? i + 2 : lines.length)];
481 final counts = parts.map((p) => p.fold(0, (s, line) => s + line.length)).toList();
482 final total = counts.fold(0, (s, count) => s + count), duration = c.duration;
483 final times = counts.map((count) => duration * count ~/ total).toList();
484 final order = [for (var i = 0; i < counts.length; i++) i];
485 order.sort((a, b) {
486 final cmp = (duration * counts[b] % total).compareTo(duration * counts[a] % total);
487 return cmp != 0 ? cmp : a.compareTo(b);
488 });
489 final remainder = duration - times.fold(0, (s, t) => s + t);
490 for (var i = 0; i < remainder; i++) times[order[i]]++;
491 var start = c.start;
492 for (var i = 0; i < parts.length; i++) {
493 result.add(Cue(c.index, start, start + times[i], parts[i]));
494 start += times[i];
495 }
496 }
497 cues = result;
498 out.writeln('SPLIT split=$split cues=${cues.length}');
499 case 'INSERT':
500 cues.insert(n(0) - 1, Cue(n(0), n(1), n(2), command.lines(3)));
501 out.writeln('INSERT cue=${n(0)} cues=${cues.length}');
502 case 'DELETE':
503 final end = command.args.length == 2 ? n(1) : n(0);
504 cues.removeRange(n(0) - 1, end);
505 out.writeln('DELETE removed=${end - n(0) + 1} cues=${cues.length}');
506 case 'TEXT':
507 cues[n(0) - 1].text = command.lines(1);
508 out.writeln('TEXT cue=${n(0)} lines=${command.lines(1).length}');
509 case 'BALANCE':
510 var changed = 0, skipped = 0, dialogue = 0;
511 for (final c in cues) {
512 if (c.dialogue) {
513 dialogue++;
514 continue;
515 }
516 final ws = words(c.text), text = ws.join(' ');
517 List<String>? candidate;
518 if (text.length <= 42) candidate = [text];
519 else {
520 var bestDifference = 1000000, bestFirst = 1000000;
521 for (var j = 1; j < ws.length; j++) {
522 final left = ws.take(j).join(' '), right = ws.skip(j).join(' ');
523 final difference = (left.length - right.length).abs();
524 if (left.length <= 42 && right.length <= 42 &&
525 (difference < bestDifference || difference == bestDifference && left.length < bestFirst)) {
526 bestDifference = difference;
527 bestFirst = left.length;
528 candidate = [left, right];
529 }
530 }
531 }
532 if (candidate == null) skipped++;
533 else if (!sameLines(c.text, candidate)) {
534 c.text = candidate;
535 changed++;
536 }
537 }
538 out.writeln('BALANCE changed=$changed skipped=$skipped dialogue=$dialogue');
539 case 'STRIP' || 'CLEANHI':
540 var tags = 0, segments = 0, labels = 0, removedLines = 0, removedCues = 0;
541 final result = <Cue>[];
542 for (final c in cues) {
543 final lines = <String>[];
544 for (final text in c.text) {
545 final converted = StringBuffer();
546 for (var p = 0; p < text.length;) {
547 if (op == 'STRIP') {
548 final item = markup(text, p);
549 if (item.$2 == 3) converted.write(text[p]);
550 else tags++;
551 p += item.$1;
552 } else {
553 final close = text[p] == '[' ? ']' : text[p] == '(' ? ')' : '';
554 final end = close.isEmpty ? -1 : text.indexOf(close, p + 1);
555 if (end >= 0) {
556 segments++;
557 p = end + 1;
558 }
559 else {
560 converted.write(text[p]);
561 p++;
562 }
563 }
564 }
565 var line = op == 'STRIP' ? converted.toString().trim() : normalized(converted.toString());
566 if (op == 'CLEANHI') {
567 final prefix = line.startsWith('- ') ? '- ' : '', body = line.startsWith('- ') ? line.substring(2) : line;
568 final colon = body.indexOf(':');
569 if (colon > 0 && body.codeUnitAt(0) >= 65 && body.codeUnitAt(0) <= 90 &&
570 body.substring(0, colon).codeUnits.every((c) => c >= 65 && c <= 90 || c >= 48 && c <= 57 || c == 32)) {
571 line = normalized(prefix + body.substring(colon + 1));
572 labels++;
573 }
574 }
575 if (line.isEmpty || op == 'CLEANHI' && line == '-') removedLines++;
576 else lines.add(line);
577 }
578 if (lines.isEmpty) removedCues++;
579 else {
580 c.text = lines;
581 result.add(c);
582 }
583 }
584 cues = result;
585 out.writeln(op == 'STRIP' ? 'STRIP tags=$tags removed_lines=$removedLines removed_cues=$removedCues' :
586 'CLEANHI segments=$segments labels=$labels removed_lines=$removedLines removed_cues=$removedCues');
587 case 'REPLACE':
588 final pair = command.lines(0), old = pair[0], replacement = pair[1];
589 var count = 0, changedCues = 0;
590 for (final c in cues) {
591 var changed = false;
592 for (var i = 0; i < c.text.length; i++) {
593 c.text[i] = c.text[i].replaceAllMapped(old, (_) {
594 count++;
595 changed = true;
596 return replacement;
597 });
598 }
599 if (changed) changedCues++;
600 }
601 out.writeln('REPLACE count=$count cues=$changedCues');
602 }
603 return out.toString();
604 }
605}
