7TIME_PATTERN = re.compile(r"([0-9]{2}):([0-9]{2}):([0-9]{2}),([0-9]{3})")
8POSITION_CODE = re.compile(r"\{\\an[1-9]\}")
9ASS_TAGS = {
10 "<i>": r"{\i1}", "</i>": r"{\i0}",
11 "<b>": r"{\b1}", "</b>": r"{\b0}",
12 "<u>": r"{\u1}", "</u>": r"{\u0}",
13}
14SCHEMAS = {
15 "LIST": (), "VALIDATE": (), "SEARCH": ("string",), "AT": ("time",),
16 "GAPS": ("gap",), "STATS": (), "PRINT": (), "TOVTT": (), "TOASS": (),
17 "FRAMES": ("fps",), "SHIFT": ("signed",),
18 "SHIFTFROM": ("position", "signed"), "SCALE": ("factor", "factor"),
19 "SYNC": ("position", "time", "position", "time"), "SNAP": ("fps",),
20 "FIXOVERLAP": ("gap",), "EXTEND": ("minimum",),
21 "TIME": ("position", "time", "time"), "SORT": (), "RENUMBER": (),
22 "MERGE": ("gap",), "SPLIT": ("width",),
23 "INSERT": ("position", "time", "time", "text"), "DELETE": ("position",),
24 "TEXT": ("position", "text"), "BALANCE": (), "STRIP": (), "CLEANHI": (),
25 "REPLACE": ("replacement",), "UNDO": (),
26}
27STRING_COMMANDS = {"SEARCH", "INSERT", "TEXT", "REPLACE"}
28MODIFYING = set((
29 "SHIFT SHIFTFROM SCALE SYNC SNAP FIXOVERLAP EXTEND TIME SORT RENUMBER "
30 "MERGE SPLIT INSERT DELETE TEXT BALANCE STRIP CLEANHI REPLACE"
31).split())

42def integer(text, minimum=0, maximum=999999999, signed=False):
43 pattern = r"-?[0-9]{1,9}" if signed else r"[0-9]{1,9}"
44 if re.fullmatch(pattern, text) is None:
45 bad_arguments()
46 value = int(text)
47 if not minimum <= value <= maximum:
48 bad_arguments()
49 return value

52def parse_time(text):
53 match = TIME_PATTERN.fullmatch(text)
54 if match is None:
55 bad_arguments()
56 hours, minutes, seconds, milliseconds = map(int, match.groups())
57 if minutes > 59 or seconds > 59:
58 bad_arguments()
59 return hours * 3600000 + minutes * 60000 + seconds * 1000 + milliseconds

62def frame_rate(text):
63 parts = text.split("/")
64 if len(parts) not in (1, 2):
65 bad_arguments()
66 numerator = integer(parts[0], 1, 100000)
67 denominator = integer(parts[1], 1, 100000) if len(parts) == 2 else 1
68 return numerator, denominator

71def argument(text, kind):
72 if kind == "time":
73 return parse_time(text)
74 if kind == "fps":
75 return frame_rate(text)
76 if kind == "signed":
77 return integer(text, -999999999, 999999999, True)
78 if kind in ("position", "factor", "gap", "minimum", "width"):
79 minimum, maximum = {
80 "position": (1, 999999999), "factor": (1, 100000),
81 "gap": (0, 60000), "minimum": (1, 60000), "width": (10, 80),
82 }[kind]
83 return integer(text, minimum, maximum)
84 if kind == "text":
85 parts = text.split("|")
86 if not all(parts):
87 bad_arguments()
88 return parts
89 if kind == "replacement":
90 old, separator, new = text.partition("|")
91 if not separator or not old or not new or "|" in new:
92 bad_arguments()
93 return old, new
94 if not text:
95 bad_arguments()
96 return text

99def round_ratio(numerator, denominator):
100 if denominator < 0:
101 numerator, denominator = -numerator, -denominator
102 return (2 * numerator + denominator) // (2 * denominator)

105def srt_time(value, separator=","):
106 return (f"{value // 3600000:02d}:{value // 60000 % 60:02d}:"
107 f"{value // 1000 % 60:02d}{separator}{value % 1000:03d}")

110def ass_time(value):
111 centiseconds = (value + 5) // 10
112 return (f"{centiseconds // 360000}:{centiseconds // 6000 % 60:02d}:"
113 f"{centiseconds // 100 % 60:02d}.{centiseconds % 100:02d}")

120def markup(line):
121 position = 0
122 while position < len(line):
123 tag = next((tag for tag in ASS_TAGS if line.startswith(tag, position)), None)
124 if tag is not None:
125 yield "format", tag
126 position += len(tag)
127 continue
128 if line.startswith("</font>", position):
129 yield "font", "</font>"
130 position += 7
131 continue
132 if (line.startswith("<font", position) and position + 5 < len(line)
133 and line[position + 5] in (" ", ">")):
134 end = line.find(">", position + 5)
135 if end != -1:
136 yield "font", line[position:end + 1]
137 position = end + 1
138 continue
139 match = POSITION_CODE.match(line, position)
140 if match is not None:
141 yield "position", match.group()
142 position = match.end()
143 continue
144 yield "character", line[position]
145 position += 1

156def clean_hi_line(line):
157 cleaned, segments = re.subn(r"\[.*?\]|\(.*?\)", "", line)
158 normalized = " ".join(cleaned.split())
159 prefix = "- " if normalized.startswith("- ") else ""
160 body = normalized[len(prefix):]
161 label, separator, rest = body.partition(":")
162 labels = 0
163 if separator and re.fullmatch(r"[A-Z][A-Z0-9 ]*", label):
164 normalized = " ".join(words([prefix + rest]))
165 labels = 1
166 return normalized, segments, labels

169class Cue:
170 def __init__(self, index, start, end, lines):
171 self.index = index
172 self.start = start
173 self.end = end
174 self.lines = lines
175
176 @property
177 def duration(self):
178 return self.end - self.start
179
180 @property
181 def characters(self):
182 return sum(map(len, self.lines))
183
184 @property
185 def dialogue(self):
186 return any(line.startswith("- ") for line in self.lines)

189def load_cues(blocks):
190 cues = []
191 output = []
192 for number, block in enumerate(blocks, 1):
193 if re.fullmatch(r"[0-9]{1,6}", block[0]) is None:
194 output.append(f"BAD CUE {number}: index")
195 continue
196 try:
197 if len(block) < 2 or len(block[1]) != 29 or block[1][12:17] != " --> ":
198 bad_arguments()
199 start, end = parse_time(block[1][:12]), parse_time(block[1][17:])
200 except CommandError:
201 output.append(f"BAD CUE {number}: time")
202 continue
203 if len(block) < 3:
204 output.append(f"BAD CUE {number}: no text")
205 continue
206 cues.append(Cue(int(block[0]), start, end, block[2:]))
207 output.append(f"LOAD cues={len(cues)} bad={len(output)}")
208 return cues, output

211class SubtitleEditor:
212 def __init__(self, cues):
213 self.cues = cues
214 self.history = []
215 self.handlers = {
216 "LIST": self.list_cues, "VALIDATE": self.validate, "SEARCH": self.search,
217 "AT": self.at, "GAPS": self.gaps, "STATS": self.stats,
218 "PRINT": self.print_srt, "TOVTT": self.to_vtt, "TOASS": self.to_ass,
219 "FRAMES": self.frames, "SHIFT": self.shift, "SHIFTFROM": self.shift_from,
220 "SCALE": self.scale, "SYNC": self.sync, "SNAP": self.snap,
221 "FIXOVERLAP": self.fix_overlap, "EXTEND": self.extend, "TIME": self.set_time,
222 "SORT": self.sort, "RENUMBER": self.renumber, "MERGE": self.merge,
223 "SPLIT": self.split, "INSERT": self.insert, "DELETE": self.delete,
224 "TEXT": self.set_text, "BALANCE": self.balance, "STRIP": self.strip,
225 "CLEANHI": self.clean_hi, "REPLACE": self.replace, "UNDO": self.undo,
226 }

228 def parse(self, line):
229 command = line.split(" ", 1)[0]
230 if command not in SCHEMAS:
231 raise CommandError("unknown command")
232 schema = SCHEMAS[command]
233 if command in STRING_COMMANDS:
234 parts = line.split(" ", len(schema))
235 else:
236 parts = line.split(" ")
237 if command == "DELETE" and len(parts) == 3:
238 schema = ("position", "position")
239 if len(parts) != len(schema) + 1:
240 bad_arguments()
241 values = [argument(text, kind) for text, kind in zip(parts[1:], schema)]
242 if command == "DELETE" and len(values) == 2 and values[1] < values[0]:
243 bad_arguments()
244 limit = len(self.cues) + (1 if command == "INSERT" else 0)
245 for value, kind in zip(values, schema):
246 if kind == "position" and value > limit:
247 raise CommandError(f"no cue {value}")
248 return command, values
249
250 def execute(self, line):
251 try:
252 command, values = self.parse(line)
253 before = copy.deepcopy(self.cues) if command in MODIFYING else None
254 output = self.handlers[command](*values)
255 if command in MODIFYING:
256 self.history.append((command, before))
257 return output
258 except CommandError as error:
259 return ["ERROR " + str(error)]

268 def validate(self):
269 output = []
270 for position, cue in enumerate(self.cues, 1):
271 prefix = f"cue {position}: "
272 duration = cue.duration
273 if cue.index != position:
274 output.append(prefix + f"index {cue.index} (expected {position})")
275 if duration <= 0:
276 output.append(prefix + "end not after start")
277 if position >= 2 and cue.start < self.cues[position - 2].end:
278 overlap = self.cues[position - 2].end - cue.start
279 output.append(prefix + f"overlaps previous by {overlap} ms")
280 if 1 <= duration < 700:
281 output.append(prefix + f"too short ({duration} ms)")
282 elif duration > 7000:
283 output.append(prefix + f"too long ({duration} ms)")
284 for line_number, line in enumerate(cue.lines, 1):
285 if len(line) > 42:
286 output.append(prefix + f"line {line_number} too long ({len(line)} chars)")
287 if len(cue.lines) >= 3:
288 output.append(prefix + f"too many lines ({len(cue.lines)})")
289 if duration >= 1 and cue.characters * 1000 > 21 * duration:
290 speed = two_decimals(cue.characters * 100000 // duration)
291 output.append(prefix + f"too fast ({speed} cps)")
292 output.append(f"VALIDATE problems={len(output)}")
293 return output

295 def search(self, text):
296 output = []
297 for position, cue in enumerate(self.cues, 1):
298 line = next((line for line in cue.lines if text in line), None)
299 if line is not None:
300 output.append(f"{position} {srt_time(cue.start)} {line}")
301 output.append(f"SEARCH found={len(output)}")
302 return output

304 def at(self, time):
305 output = [self.cue_line(position, cue) for position, cue in enumerate(self.cues, 1)
306 if cue.start <= time < cue.end]
307 output.append(f"AT active={len(output)}")

310 def gaps(self, minimum):
311 output = []
312 for index in range(len(self.cues) - 1):
313 gap = self.cues[index + 1].start - self.cues[index].end
314 if gap < minimum:
315 output.append(f"{index + 1} {index + 2} {gap}")
316 output.append(f"GAPS count={len(output)}")

319 def stats(self):
320 count = len(self.cues)
321 output = [f"STATS cues={count}"]
322 if count == 0:
323 return output
324 durations = [max(0, cue.duration) for cue in self.cues]
325 total = sum(durations)
326 output.append(f"duration total={srt_time(total)} mean={srt_time(total // count)}")
327 characters = sum(cue.characters for cue, duration in zip(self.cues, durations) if duration >= 1)
328 output.append("speed " + (two_decimals(characters * 100000 // total) if total else "-"))
329 longest = durations.index(max(durations))
330 output.append(f"longest cue={longest + 1} duration={srt_time(durations[longest])}")
331 if count == 1:
332 output.append("gap none")
333 else:
334 gaps = [self.cues[i + 1].start - self.cues[i].end for i in range(count - 1)]
335 largest = gaps.index(max(gaps))
336 output.append(f"gap cue={largest + 1} ms={gaps[largest]}")
337 return output

347 def to_vtt(self):
348 output = ["WEBVTT", ""]
349 escapes = {"&": "&amp;", "<": "&lt;", ">": "&gt;"}
350 for cue in self.cues:
351 lines = []
352 first_code = None
353 for line in cue.lines:
354 parts = []
355 for kind, item in markup(line):
356 if kind == "position" and first_code is None:
357 first_code = int(item[4])
358 if kind == "format":
359 parts.append(item)
360 elif kind == "character":
361 parts.append(escapes.get(item, item))
362 converted = "".join(parts).strip(" ")
363 if converted:
364 lines.append(converted)
365 if not lines:
366 continue
367 settings = ""
368 if first_code in (7, 8, 9):
369 settings += " line:0"
370 elif first_code in (4, 5, 6):
371 settings += " line:50%"
372 if first_code in (1, 4, 7):
373 settings += " align:start"
374 elif first_code in (3, 6, 9):
375 settings += " align:end"
376 output.extend([str(cue.index), f"{srt_time(cue.start, '.')} --> {srt_time(cue.end, '.')}{settings}"])
377 output.extend(lines)
378 output.append("")
379 return output

381 def to_ass(self):
382 output = ["[Events]", "Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text"]
383 for cue in self.cues:
384 lines = []
385 for line in cue.lines:
386 parts = []
387 for kind, item in markup(line):
388 if kind == "format":
389 parts.append(ASS_TAGS[item])
390 elif kind != "font":
391 parts.append(item)
392 lines.append("".join(parts))
393 text = r"\N".join(lines)
394 output.append(f"Dialogue: 0,{ass_time(cue.start)},{ass_time(cue.end)},Default,,0,0,0,,{text}")
395 return output

397 def frames(self, rate):
398 numerator, denominator = rate
399 return [f"{position} {cue.start * numerator // (1000 * denominator)} "
400 f"{cue.end * numerator // (1000 * denominator)}"
401 for position, cue in enumerate(self.cues, 1)]

403 def transform_times(self, transform, first=0):
404 clamped = 0
405 for cue in self.cues[first:]:
406 start, end = transform(cue.start), transform(cue.end)
407 clamped += start < 0 or end < 0
408 cue.start, cue.end = max(0, start), max(0, end)
409 return clamped

411 def shift(self, milliseconds):
412 clamped = self.transform_times(lambda time: time + milliseconds)
413 return [f"SHIFT cues={len(self.cues)} clamped={clamped}"]
414
415 def shift_from(self, position, milliseconds):
416 clamped = self.transform_times(lambda time: time + milliseconds, position - 1)
417 return [f"SHIFTFROM cues={len(self.cues) - position + 1} clamped={clamped}"]
418
419 def scale(self, numerator, denominator):
420 self.transform_times(lambda time: round_ratio(time * numerator, denominator))
421 return [f"SCALE cues={len(self.cues)}"]
422
423 def sync(self, first, target_first, second, target_second):
424 source_first, source_second = self.cues[first - 1].start, self.cues[second - 1].start
425 if source_first == source_second:
426 raise CommandError("sync points coincide")
427 denominator = source_second - source_first
428 clamped = self.transform_times(lambda time: round_ratio(
429 target_first * denominator + (time - source_first) * (target_second - target_first), denominator
430 ))
431 return [f"SYNC cues={len(self.cues)} clamped={clamped}"]
432
433 def snap(self, rate):
434 numerator, denominator = rate
435 changed = 0
436 for cue in self.cues:
437 old = cue.start, cue.end
438 updated = []
439 for time in old:
440 frame = round_ratio(time * numerator, 1000 * denominator)
441 updated.append(round_ratio(frame * 1000 * denominator, numerator))
442 changed += sum(first != second for first, second in zip(old, updated))
443 cue.start, cue.end = updated
444 return [f"SNAP changed={changed}"]

446 def fix_overlap(self, gap):
447 output = []
448 fixed = 0
449 for index in range(len(self.cues) - 1):
450 cue = self.cues[index]
451 candidate = self.cues[index + 1].start - gap
452 if cue.end > candidate:
453 if candidate > cue.start:
454 cue.end = candidate
455 fixed += 1
456 else:
457 output.append(f"cue {index + 1}: cannot fix")
458 output.append(f"FIXOVERLAP fixed={fixed} failed={len(output)}")
459 return output

461 def extend(self, minimum):
462 extended = 0
463 for index, cue in enumerate(self.cues):
464 if cue.duration < minimum:
465 candidate = cue.start + minimum
466 if index + 1 < len(self.cues):
467 candidate = min(candidate, self.cues[index + 1].start)
468 if candidate > cue.end:
469 cue.end = candidate
470 extended += 1
471 short = sum(cue.duration < minimum for cue in self.cues)
472 return [f"EXTEND extended={extended} short={short}"]

479 def sort(self):
480 ordered = sorted(enumerate(self.cues, 1), key=lambda item: (item[1].start, item[1].end))
481 moved = sum(old != new for new, (old, cue) in enumerate(ordered, 1))
482 self.cues = [cue for old, cue in ordered]
483 return [f"SORT moved={moved}"]

492 def merge(self, gap):
493 merged = []
494 joins = 0
495 for cue in self.cues:
496 if merged:
497 previous = merged[-1]
498 if cue.start - previous.end < gap and not previous.dialogue and not cue.dialogue:
499 lines = wrap(words(previous.lines + cue.lines), 42)
500 if len(lines) <= 2:
501 previous.end = max(previous.end, cue.end)
502 previous.lines = lines
503 joins += 1
504 continue
505 merged.append(cue)
506 self.cues = merged
507 return [f"MERGE merged={joins} cues={len(self.cues)}"]

509 def split(self, width):
510 rebuilt = []
511 split_count = 0
512 for cue in self.cues:
513 if cue.duration <= 0 or cue.dialogue:
514 rebuilt.append(cue)
515 continue
516 lines = wrap(words(cue.lines), width)
517 if len(lines) <= 2:
518 cue.lines = lines
519 rebuilt.append(cue)
520 continue
521 split_count += 1
522 parts = [lines[index:index + 2] for index in range(0, len(lines), 2)]
523 counts = [sum(map(len, part)) for part in parts]
524 total = sum(counts)
525 durations = [cue.duration * count // total for count in counts]
526 remainders = [cue.duration * count % total for count in counts]
527 missing = cue.duration - sum(durations)
528 order = sorted(range(len(parts)), key=lambda index: (-remainders[index], index))
529 for index in order[:missing]:
530 durations[index] += 1
531 start = cue.start
532 for part, duration in zip(parts, durations):
533 rebuilt.append(Cue(cue.index, start, start + duration, part))
534 start += duration
535 self.cues = rebuilt
536 return [f"SPLIT split={split_count} cues={len(self.cues)}"]

551 def balance(self):
552 changed = skipped = dialogue = 0
553 for cue in self.cues:
554 if cue.dialogue:
555 dialogue += 1
556 continue
557 sequence = words(cue.lines)
558 joined = " ".join(sequence)
559 updated = [joined] if len(joined) <= 42 else None
560 if updated is None:
561 best_key = None
562 for position in range(1, len(sequence)):
563 first = " ".join(sequence[:position])
564 second = " ".join(sequence[position:])
565 if len(first) <= 42 and len(second) <= 42:
566 key = abs(len(first) - len(second)), len(first)
567 if best_key is None or key < best_key:
568 updated, best_key = [first, second], key
569 if updated is None:
570 skipped += 1
571 else:
572 changed += updated != cue.lines
573 cue.lines = updated
574 return [f"BALANCE changed={changed} skipped={skipped} dialogue={dialogue}"]

576 def strip(self):
577 tags = removed_lines = removed_cues = 0
578 remaining = []
579 for cue in self.cues:
580 lines = []
581 for line in cue.lines:
582 parts = []
583 for kind, item in markup(line):
584 if kind == "character":
585 parts.append(item)
586 else:
587 tags += 1
588 converted = "".join(parts).strip(" ")
589 if converted:
590 lines.append(converted)
591 else:
592 removed_lines += 1
593 if lines:
594 cue.lines = lines
595 remaining.append(cue)
596 else:
597 removed_cues += 1
598 self.cues = remaining
599 return [f"STRIP tags={tags} removed_lines={removed_lines} removed_cues={removed_cues}"]

601 def clean_hi(self):
602 segments = labels = removed_lines = removed_cues = 0
603 remaining = []
604 for cue in self.cues:
605 lines = []
606 for line in cue.lines:
607 converted, bracket_count, label_count = clean_hi_line(line)
608 segments += bracket_count
609 labels += label_count
610 if converted and converted != "-":
611 lines.append(converted)
612 else:
613 removed_lines += 1
614 if lines:
615 cue.lines = lines
616 remaining.append(cue)
617 else:
618 removed_cues += 1
619 self.cues = remaining
620 return [f"CLEANHI segments={segments} labels={labels} removed_lines={removed_lines} removed_cues={removed_cues}"]

622 def replace(self, replacement):
623 old, new = replacement
624 replacements = affected_cues = 0
625 for cue in self.cues:
626 count = sum(line.count(old) for line in cue.lines)
627 replacements += count
628 affected_cues += count > 0
629 cue.lines = [line.replace(old, new) for line in cue.lines]
630 return [f"REPLACE count={replacements} cues={affected_cues}"]

632 def undo(self):
633 if not self.history:
634 raise CommandError("nothing to undo")
635 command, self.cues = self.history.pop()
636 return [f"UNDO command={command}"]

639def read_blocks(stream):
640 blocks = []
641 current = []
642 for raw in stream:
643 line = raw.rstrip("\n")
644 if line == "###":
645 break
646 if line:
647 current.append(line)
648 elif current:
649 blocks.append(current)
650 current = []
651 if current:
652 blocks.append(current)
653 return blocks
