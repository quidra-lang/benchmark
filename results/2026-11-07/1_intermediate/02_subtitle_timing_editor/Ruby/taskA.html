1class Invalid < StandardError
2end
3
4class Editor
5 ARITIES = {
6 "LIST" => [0], "VALIDATE" => [0], "SEARCH" => [1], "AT" => [1],
7 "GAPS" => [1], "STATS" => [0], "PRINT" => [0], "TOVTT" => [0],
8 "TOASS" => [0], "FRAMES" => [1], "SHIFT" => [1], "SHIFTFROM" => [2],
9 "SCALE" => [2], "SYNC" => [4], "SNAP" => [1], "FIXOVERLAP" => [1],
10 "EXTEND" => [1], "TIME" => [3], "SORT" => [0], "RENUMBER" => [0],
11 "MERGE" => [1], "SPLIT" => [1], "INSERT" => [4], "DELETE" => [1, 2],
12 "TEXT" => [2], "BALANCE" => [0], "STRIP" => [0], "CLEANHI" => [0],
13 "REPLACE" => [1], "UNDO" => [0]
14 }.freeze
15 MODIFYING = %w[SHIFT SHIFTFROM SCALE SYNC SNAP FIXOVERLAP EXTEND TIME SORT
16 RENUMBER MERGE SPLIT INSERT DELETE TEXT BALANCE STRIP CLEANHI REPLACE].freeze
17 FORMATS = %w[<i> </i> <b> </b> <u> </u>].freeze

19 def initialize
20 @cues = []
21 @history = []
22 end

24 def bad
25 raise Invalid, "bad arguments"
26 end

28 def integer(value, minimum = 0, maximum = 999_999_999)
29 bad unless value.match?(/\A\d{1,9}\z/)
30 result = value.to_i
31 bad unless result.between?(minimum, maximum)
32 result
33 end
34
35 def signed(value)
36 bad unless value.match?(/\A-?\d{1,9}\z/)
37 value.to_i
38 end
39
40 def parse_time(value)
41 bad unless value.match?(/\A\d{2}:[0-5]\d:[0-5]\d,\d{3}\z/)
42 value[0, 2].to_i * 3_600_000 + value[3, 2].to_i * 60_000 +
43 value[6, 2].to_i * 1000 + value[9, 3].to_i
44 end
45
46 def fps(value)
47 parts = value.split("/", -1)
48 bad if parts.length > 2
49 [integer(parts[0], 1, 100_000), parts.length == 2 ? integer(parts[1], 1, 100_000) : 1]
50 end
51
52 def text_lines(value)
53 parts = value.split("|", -1)
54 bad if parts.empty? || parts.any?(&:empty?)
55 parts
56 end
57
58 def rounded(a, b)
59 if b.negative?
60 a = -a
61 b = -b
62 end
63 (2 * a + b).div(2 * b)
64 end
65
66 def srt(time)
67 format("%02d:%02d:%02d,%03d", time.div(3_600_000), time.div(60_000) % 60,
68 time.div(1000) % 60, time % 1000)
69 end
70
71 def ass(time)
72 c = (time + 5).div(10)
73 format("%d:%02d:%02d.%02d", c.div(360_000), c.div(6000) % 60, c.div(100) % 60, c % 100)
74 end
75
76 def decimal(value)
77 format("%d.%02d", value.div(100), value % 100)
78 end

80 def chars(cue)
81 cue[:text].sum(&:length)
82 end

84 def dialogue?(cue)
85 cue[:text].any? { |line| line.start_with?("- ") }
86 end

88 def words(lines)
89 lines.join(" ").split(" ")
90 end

92 def wrap(lines, width)
93 result = []
94 words(lines).each do |word|
95 if !result.empty? && result[-1].length + 1 + word.length <= width
96 result[-1] += " " + word
97 else
98 result << word
99 end
100 end
101 result
102 end

104 def markup(line)
105 result = []
106 p = 0
107 while p < line.length
108 tag = FORMATS.find { |item| line[p, item.length] == item }
109 if tag
110 result << { kind: :format, text: tag, digit: 0 }
111 p += tag.length
112 next
113 end
114 font = nil
115 if line[p, 7] == "</font>"
116 font = "</font>"
117 elsif line[p, 5] == "<font" && [" ", ">"].include?(line[p + 5])
118 ending = line.index(">", p + 5)
119 font = line[p..ending] if ending
120 end
121 if font
122 result << { kind: :font, text: font, digit: 0 }
123 p += font.length
124 next
125 end
126 position = /\A\{\\an([1-9])\}/.match(line[p..])
127 if position
128 result << { kind: :position, text: position[0], digit: position[1].to_i }
129 p += position[0].length
130 else
131 result << { kind: :char, text: line[p], digit: 0 }
132 p += 1
133 end
134 end
135 result
136 end

138 def listing(cue, position)
139 "#{position} #{cue[:index]} #{srt(cue[:start])} #{srt(cue[:end])} " + cue[:text].join("|")
140 end

142 def load(lines)
143 blocks = []
144 current = []
145 (lines + [""]).each do |line|
146 if !line.empty?
147 current << line
148 elsif !current.empty?
149 blocks << current
150 current = []
151 end
152 end
153 output = []
154 blocks.each_with_index do |block, i|
155 reason = nil
156 if !block[0].match?(/\A\d{1,6}\z/)
157 reason = "index"
158 elsif block.length < 2 ||
159 !block[1].match?(/\A\d{2}:[0-5]\d:[0-5]\d,\d{3} --> \d{2}:[0-5]\d:[0-5]\d,\d{3}\z/)
160 reason = "time"
161 elsif block.length < 3
162 reason = "no text"
163 end
164 if reason
165 output << "BAD CUE #{i + 1}: #{reason}"
166 else
167 @cues << {
168 index: block[0].to_i, start: parse_time(block[1][0, 12]),
169 end: parse_time(block[1][17, 12]), text: block[2..]
170 }
171 end
172 end
173 output << "LOAD cues=#{@cues.length} bad=#{blocks.length - @cues.length}"
174 output
175 end

177 def execute(line)
178 name = line.split(" ", 2)[0]
179 raise Invalid, "unknown command" unless ARITIES.key?(name)
180
181 fixed = { "SEARCH" => 0, "INSERT" => 3, "TEXT" => 1, "REPLACE" => 0 }
182 if fixed.key?(name)
183 bad if line.length <= name.length
184 args = []
185 p = name.length + 1
186 fixed[name].times do
187 following = line.index(" ", p)
188 bad unless following
189 args << line[p...following]
190 p = following + 1
191 end
192 args << line[p..]
193 else
194 args = line.split(" ")[1..]
195 end
196 bad unless ARITIES[name].include?(args.length)
197
198 values = []
199 texts = []
200 rate = [1, 1]
201 old = ""
202 replacement = ""
203 positions = []
204 position = lambda do |value|
205 p = integer(value, 1)
206 positions << p
207 p
208 end
209 case name
210 when "AT"
211 values << parse_time(args[0])
212 when "GAPS", "FIXOVERLAP", "MERGE"
213 values << integer(args[0], 0, 60_000)
214 when "EXTEND"
215 values << integer(args[0], 1, 60_000)
216 when "SPLIT"
217 values << integer(args[0], 10, 80)
218 when "SHIFT"
219 values << signed(args[0])
220 when "SHIFTFROM"
221 values = [position.call(args[0]), signed(args[1])]
222 when "SCALE"
223 values = args.map { |arg| integer(arg, 1, 100_000) }
224 when "SYNC"
225 values = [position.call(args[0]), parse_time(args[1]), position.call(args[2]), parse_time(args[3])]
226 when "TIME", "INSERT"
227 values = [position.call(args[0]), parse_time(args[1]), parse_time(args[2])]
228 texts = text_lines(args[3]) if name == "INSERT"
229 when "DELETE"
230 values = args.map { |arg| position.call(arg) }
231 bad if values.length == 2 && values[1] < values[0]
232 when "TEXT"
233 values = [position.call(args[0])]
234 texts = text_lines(args[1])
235 when "FRAMES", "SNAP"
236 rate = fps(args[0])
237 when "REPLACE"
238 parts = args[0].split("|", -1)
239 bad unless parts.length == 2 && parts.none?(&:empty?)
240 old, replacement = parts
241 end
242 positions.each do |p|
243 limit = @cues.length + (name == "INSERT" ? 1 : 0)
244 raise Invalid, "no cue #{p}" if p > limit
245 end
246 if name == "SYNC" && @cues[values[0] - 1][:start] == @cues[values[2] - 1][:start]
247 raise Invalid, "sync points coincide"
248 end
249 if name == "UNDO"
250 record = @history.pop
251 raise Invalid, "nothing to undo" unless record
252 @cues = record[:cues]
253 return ["UNDO command=#{record[:name]}"]
254 end
255 if MODIFYING.include?(name)
256 snapshot = @cues.map { |cue| cue.merge(text: cue[:text].dup) }
257 @history << { name: name, cues: snapshot }
258 end
259 process(name, args, values, texts, rate, old, replacement)
260 end
261
262 def process(name, args, v, texts, rate, old, replacement)
263 output = []
264 n = @cues.length
265 case name
266 when "LIST", "AT"
267 @cues.each_with_index do |cue, i|
268 if name == "LIST" || (cue[:start] <= v[0] && v[0] < cue[:end])
269 output << listing(cue, i + 1)
270 end
271 end
272 output << "AT active=#{output.length}" if name == "AT"
273 when "VALIDATE"
274 @cues.each_with_index do |cue, i|
275 p = i + 1
276 prefix = "cue #{p}: "
277 d = cue[:end] - cue[:start]
278 output << prefix + "index #{cue[:index]} (expected #{p})" if cue[:index] != p
279 output << prefix + "end not after start" if d <= 0
280 if i.positive? && cue[:start] < @cues[i - 1][:end]
281 output << prefix + "overlaps previous by #{@cues[i - 1][:end] - cue[:start]} ms"
282 end
283 if d.between?(1, 699)
284 output << prefix + "too short (#{d} ms)"
285 elsif d > 7000
286 output << prefix + "too long (#{d} ms)"
287 end
288 cue[:text].each_with_index do |text, j|
289 output << prefix + "line #{j + 1} too long (#{text.length} chars)" if text.length > 42
290 end
291 output << prefix + "too many lines (#{cue[:text].length})" if cue[:text].length >= 3
292 c = chars(cue)
293 if d.positive? && c * 1000 > 21 * d
294 output << prefix + "too fast (#{decimal((c * 100_000).div(d))} cps)"
295 end
296 end
297 output << "VALIDATE problems=#{output.length}"
298 when "SEARCH"
299 @cues.each_with_index do |cue, i|
300 match = cue[:text].find { |text| text.include?(args[0]) }
301 output << "#{i + 1} #{srt(cue[:start])} #{match}" if match
302 end
303 output << "SEARCH found=#{output.length}"
304 when "GAPS"
305 (0...n - 1).each do |i|
306 gap = @cues[i + 1][:start] - @cues[i][:end]
307 output << "#{i + 1} #{i + 2} #{gap}" if gap < v[0]
308 end
309 output << "GAPS count=#{output.length}"
310 when "STATS"
311 output << "STATS cues=#{n}"
312 if n.positive?
313 durations = @cues.map { |cue| [0, cue[:end] - cue[:start]].max }
314 total = durations.sum
315 c = @cues.each_with_index.sum { |cue, i| durations[i].positive? ? chars(cue) : 0 }
316 output << "duration total=#{srt(total)} mean=#{srt(total.div(n))}"
317 output << "speed " + (total.positive? ? decimal((c * 100_000).div(total)) : "-")
318 maximum = durations.max
319 output << "longest cue=#{durations.index(maximum) + 1} duration=#{srt(maximum)}"
320 if n == 1
321 output << "gap none"
322 else
323 gaps = @cues.drop(1).each_with_index.map { |cue, i| cue[:start] - @cues[i][:end] }
324 maximum_gap = gaps.max
325 output << "gap cue=#{gaps.index(maximum_gap) + 1} ms=#{maximum_gap}"
326 end
327 end
328 when "PRINT"
329 @cues.each do |cue|
330 output.concat([cue[:index].to_s, "#{srt(cue[:start])} --> #{srt(cue[:end])}", *cue[:text], ""])
331 end
332 when "TOVTT"
333 output.concat(["WEBVTT", ""])
334 escapes = { "&" => "&amp;", "<" => "&lt;", ">" => "&gt;" }
335 @cues.each do |cue|
336 digit = 0
337 converted = cue[:text].map do |line|
338 result = +""
339 markup(line).each do |token|
340 digit = token[:digit] if token[:kind] == :position && digit.zero?
341 if token[:kind] == :format
342 result << token[:text]
343 elsif token[:kind] == :char
344 result << escapes.fetch(token[:text], token[:text])
345 end
346 end
347 result.strip
348 end.reject(&:empty?)
349 next if converted.empty?
350
351 settings = digit >= 7 ? " line:0" : digit >= 4 ? " line:50%" : ""
352 settings += " align:start" if [1, 4, 7].include?(digit)
353 settings += " align:end" if [3, 6, 9].include?(digit)
354 timing = "#{srt(cue[:start]).tr(',', '.')} --> #{srt(cue[:end]).tr(',', '.')}#{settings}"
355 output.concat([cue[:index].to_s, timing, *converted, ""])
356 end
357 when "TOASS"
358 output.concat(["[Events]", "Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text"])
359 tags = {
360 "<i>" => "{\\i1}", "</i>" => "{\\i0}", "<b>" => "{\\b1}",
361 "</b>" => "{\\b0}", "<u>" => "{\\u1}", "</u>" => "{\\u0}"
362 }
363 @cues.each do |cue|
364 converted = cue[:text].map do |line|
365 markup(line).map do |token|
366 if token[:kind] == :format
367 tags[token[:text]]
368 elsif token[:kind] == :font
369 ""
370 else
371 token[:text]
372 end
373 end.join
374 end
375 output << "Dialogue: 0,#{ass(cue[:start])},#{ass(cue[:end])},Default,,0,0,0,," + converted.join("\\N")
376 end
377 when "FRAMES"
378 @cues.each_with_index do |cue, i|
379 frames = [cue[:start], cue[:end]].map { |t| (t * rate[0]).div(1000 * rate[1]) }
380 output << "#{i + 1} #{frames[0]} #{frames[1]}"
381 end
382 when "SHIFT", "SHIFTFROM", "SCALE", "SYNC", "SNAP"
383 clamped = 0
384 changed = 0
385 first = name == "SHIFTFROM" ? v[0] - 1 : 0
386 shift = name == "SHIFTFROM" ? v[1] : v[0]
387 s1 = name == "SYNC" ? @cues[v[0] - 1][:start] : 0
388 s2 = name == "SYNC" ? @cues[v[2] - 1][:start] : 0
389 (first...n).each do |i|
390 cue = @cues[i]
391 was_clamped = false
392 %i[start end].each do |key|
393 t = cue[key]
394 result = case name
395 when "SHIFT", "SHIFTFROM"
396 t + shift
397 when "SCALE"
398 rounded(t * v[0], v[1])
399 when "SYNC"
400 den = s2 - s1
401 rounded(v[1] * den + (t - s1) * (v[3] - v[1]), den)
402 else
403 frame = rounded(t * rate[0], 1000 * rate[1])
404 rounded(frame * 1000 * rate[1], rate[0])
405 end
406 if result.negative?
407 was_clamped = true
408 result = 0
409 end
410 changed += 1 if result != t
411 cue[key] = result
412 end
413 clamped += 1 if was_clamped
414 end
415 output << if name == "SNAP"
416 "SNAP changed=#{changed}"
417 elsif name == "SCALE"
418 "SCALE cues=#{n}"
419 else
420 "#{name} cues=#{n - first} clamped=#{clamped}"
421 end
422 when "FIXOVERLAP"
423 fixed = 0
424 failed = 0
425 (0...n - 1).each do |i|
426 cue = @cues[i]
427 limit = @cues[i + 1][:start] - v[0]
428 next unless cue[:end] > limit
429 if limit > cue[:start]
430 cue[:end] = limit
431 fixed += 1
432 else
433 failed += 1
434 output << "cue #{i + 1}: cannot fix"
435 end
436 end
437 output << "FIXOVERLAP fixed=#{fixed} failed=#{failed}"
438 when "EXTEND"
439 extended = 0
440 @cues.each_with_index do |cue, i|
441 next unless cue[:end] - cue[:start] < v[0]
442 candidate = cue[:start] + v[0]
443 candidate = [candidate, @cues[i + 1][:start]].min if i + 1 < n
444 if candidate > cue[:end]
445 cue[:end] = candidate
446 extended += 1
447 end
448 end
449 short = @cues.count { |cue| cue[:end] - cue[:start] < v[0] }
450 output << "EXTEND extended=#{extended} short=#{short}"
451 when "TIME"
452 @cues[v[0] - 1][:start] = v[1]
453 @cues[v[0] - 1][:end] = v[2]
454 output << "TIME cue=#{v[0]}"
455 when "SORT"
456 sorted = @cues.each_with_index.sort_by { |cue, i| [cue[:start], cue[:end], i] }
457 moved = sorted.each_with_index.count { |(_, old_position), i| old_position != i }
458 @cues = sorted.map(&:first)
459 output << "SORT moved=#{moved}"
460 when "RENUMBER"
461 changed = 0
462 @cues.each_with_index do |cue, i|
463 if cue[:index] != i + 1
464 cue[:index] = i + 1
465 changed += 1
466 end
467 end
468 output << "RENUMBER changed=#{changed}"
469 when "MERGE"
470 merged = []
471 joins = 0
472 @cues.each do |cue|
473 previous = merged[-1]
474 lines = previous ? wrap(previous[:text] + cue[:text], 42) : []
475 if previous && cue[:start] - previous[:end] < v[0] &&
476 !dialogue?(previous) && !dialogue?(cue) && lines.length <= 2
477 previous[:end] = [previous[:end], cue[:end]].max
478 previous[:text] = lines
479 joins += 1
480 else
481 merged << cue
482 end
483 end
484 @cues = merged
485 output << "MERGE merged=#{joins} cues=#{merged.length}"
486 when "SPLIT"
487 result = []
488 split = 0
489 @cues.each do |cue|
490 duration = cue[:end] - cue[:start]
491 if duration <= 0 || dialogue?(cue)
492 result << cue
493 next
494 end
495 lines = wrap(cue[:text], v[0])
496 if lines.length <= 2
497 cue[:text] = lines
498 result << cue
499 next
500 end
501 split += 1
502 parts = lines.each_slice(2).to_a
503 counts = parts.map { |part| part.sum(&:length) }
504 total = counts.sum
505 times = counts.map { |c| (duration * c).div(total) }
506 order = counts.each_index.sort_by { |i| [-(duration * counts[i] % total), i] }
507 (duration - times.sum).times { |i| times[order[i]] += 1 }
508 start = cue[:start]
509 parts.each_with_index do |text, i|
510 result << { index: cue[:index], start: start, end: start + times[i], text: text }
511 start += times[i]
512 end
513 end
514 @cues = result
515 output << "SPLIT split=#{split} cues=#{result.length}"
516 when "INSERT"
517 @cues.insert(v[0] - 1, { index: v[0], start: v[1], end: v[2], text: texts })
518 output << "INSERT cue=#{v[0]} cues=#{@cues.length}"
519 when "DELETE"
520 count = (v[1] || v[0]) - v[0] + 1
521 @cues.slice!(v[0] - 1, count)
522 output << "DELETE removed=#{count} cues=#{@cues.length}"
523 when "TEXT"
524 @cues[v[0] - 1][:text] = texts
525 output << "TEXT cue=#{v[0]} lines=#{texts.length}"
526 when "BALANCE"
527 changed = 0
528 skipped = 0
529 dialogues = 0
530 @cues.each do |cue|
531 if dialogue?(cue)
532 dialogues += 1
533 next
534 end
535 all = words(cue[:text])
536 joined = all.join(" ")
537 best = joined.length <= 42 ? [joined] : nil
538 unless best
539 (1...all.length).each do |j|
540 candidate = [all.take(j).join(" "), all.drop(j).join(" ")]
541 next if candidate.any? { |text| text.length > 42 }
542 key = [(candidate[0].length - candidate[1].length).abs, candidate[0].length]
543 best_key = best && [(best[0].length - best[1].length).abs, best[0].length]
544 best = candidate if best.nil? || (key <=> best_key).negative?
545 end
546 end
547 if best.nil?
548 skipped += 1
549 elsif best != cue[:text]
550 cue[:text] = best
551 changed += 1
552 end
553 end
554 output << "BALANCE changed=#{changed} skipped=#{skipped} dialogue=#{dialogues}"
555 when "STRIP", "CLEANHI"
556 tags = 0
557 segments = 0
558 labels = 0
559 deleted_lines = 0
560 result = []
561 @cues.each do |cue|
562 lines = []
563 cue[:text].each do |original|
564 line = +""
565 if name == "STRIP"
566 markup(original).each do |token|
567 if token[:kind] == :char
568 line << token[:text]
569 else
570 tags += 1
571 end
572 end
573 line = line.strip
574 else
575 line = original.gsub(/\[.*?\]|\(.*?\)/) do
576 segments += 1
577 ""
578 end
579 line = line.gsub(/ +/, " ").strip
580 prefix = line.start_with?("- ") ? "- " : ""
581 body = line[prefix.length..]
582 colon = body.index(":")
583 if colon && body[0...colon].match?(/\A[A-Z][A-Z0-9 ]*\z/)
584 labels += 1
585 line = (prefix + body[colon + 1..]).gsub(/ +/, " ").strip
586 end
587 end
588 if line.empty? || (name == "CLEANHI" && line == "-")
589 deleted_lines += 1
590 else
591 lines << line
592 end
593 end
594 unless lines.empty?
595 cue[:text] = lines
596 result << cue
597 end
598 end
599 @cues = result
600 suffix = " removed_lines=#{deleted_lines} removed_cues=#{n - result.length}"
601 output << if name == "STRIP"
602 "STRIP tags=#{tags}" + suffix
603 else
604 "CLEANHI segments=#{segments} labels=#{labels}" + suffix
605 end
606 when "REPLACE"
607 count = 0
608 changed = 0
609 @cues.each do |cue|
610 matches = 0
611 cue[:text] = cue[:text].map do |line|
612 parts = line.split(old, -1)
613 matches += parts.length - 1
614 parts.join(replacement)
615 end
616 count += matches
617 changed += 1 if matches.positive?
618 end
619 output << "REPLACE count=#{count} cues=#{changed}"
620 end
621 output
622 end
623end
624
625lines = STDIN.read.split("\n", -1)
626terminator = lines.index("###")
627editor = Editor.new
628output = editor.load(lines.take(terminator))
629count = lines[terminator + 1].to_i
630count.times do |i|
631 begin
632 output.concat(editor.execute(lines[terminator + 2 + i]))
633 rescue Invalid => error
634 output << "ERROR " + error.message
635 end
636end
637STDOUT.write(output.join("\n") + "\n")
638
