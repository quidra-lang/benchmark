1data class Cue(val index: Long, val start: Long, val end: Long, val text: List<String>) {
2 val duration get() = end-start
3 val chars get() = text.sumOf { it.length.toLong() }
4 val dialogue get() = text.any { it.startsWith("- ") }
5}
6data class Undo(val command: String, val cues: List<Cue>)
7data class Item(val text: String, val kind: Int, val digit: Int = 0)
8class Problem(message: String) : RuntimeException(message)

10class Editor {
11 private var cues = mutableListOf<Cue>()
12 private val history = ArrayDeque<Undo>()
13 private val schemas = mapOf("LIST" to "", "VALIDATE" to "", "SEARCH" to "X", "AT" to "T",
14 "GAPS" to "g", "STATS" to "", "PRINT" to "", "TOVTT" to "", "TOASS" to "", "FRAMES" to "F",
15 "SHIFT" to "I", "SHIFTFROM" to "PI", "SCALE" to "rr", "SYNC" to "PTPT", "SNAP" to "F",
16 "FIXOVERLAP" to "g", "EXTEND" to "e", "TIME" to "PTT", "SORT" to "", "RENUMBER" to "",
17 "MERGE" to "g", "SPLIT" to "w", "INSERT" to "PTTX", "DELETE" to "P", "TEXT" to "PX",
18 "BALANCE" to "", "STRIP" to "", "CLEANHI" to "", "REPLACE" to "X", "UNDO" to "")
19 private val modifying = setOf("SHIFT","SHIFTFROM","SCALE","SYNC","SNAP","FIXOVERLAP","EXTEND",
20 "TIME","SORT","RENUMBER","MERGE","SPLIT","INSERT","DELETE","TEXT","BALANCE","STRIP","CLEANHI","REPLACE")
21 private fun bad(): Nothing = throw Problem("bad arguments")
22 private fun number(s: String): Long {
23 if (!Regex("[0-9]{1,9}").matches(s)) bad()
24 return s.toLong()
25 }
26 private fun time(s: String): Long {
27 if (!Regex("[0-9]{2}:[0-5][0-9]:[0-5][0-9],[0-9]{3}").matches(s)) bad()
28 return s.substring(0,2).toLong()*3600000+s.substring(3,5).toLong()*60000+
29 s.substring(6,8).toLong()*1000+s.substring(9).toLong()
30 }
31 private fun fps(s: String): Pair<Long,Long> {
32 val parts = s.split('/')
33 if (parts.size !in 1..2) bad()
34 val a = number(parts[0])
35 val b = if (parts.size == 2) number(parts[1]) else 1L
36 if (a !in 1..100000 || b !in 1..100000) bad()
37 return a to b
38 }
39 private fun text(s: String): List<String> {
40 val parts = s.split('|')
41 if (parts.any { it.isEmpty() }) bad()
42 return parts
43 }
44 private fun pad(n: Long, width: Int) = n.toString().padStart(width,'0')
45 private fun srt(t: Long) = "${pad(t/3600000,2)}:${pad(t/60000%60,2)}:${pad(t/1000%60,2)},${pad(t%1000,3)}"
46 private fun ass(t: Long): String {
47 val c = (t+5)/10
48 return "${c/360000}:${pad(c/6000%60,2)}:${pad(c/100%60,2)}.${pad(c%100,2)}"
49 }
50 private fun decimal(v: Long) = "${v/100}.${pad(v%100,2)}"
51 private fun round(numerator: Long, denominator: Long): Long {
52 val a = if (denominator < 0) -numerator else numerator
53 val b = kotlin.math.abs(denominator)
54 return Math.floorDiv(2*a+b,2*b)
55 }
56 private fun words(lines: List<String>) = lines.joinToString(" ").split(' ').filter { it.isNotEmpty() }
57 private fun normalize(s: String) = words(listOf(s)).joinToString(" ")
58 private fun wrap(lines: List<String>, width: Long): List<String> {
59 val result = mutableListOf<String>()
60 for (word in words(lines)) {
61 if (result.isEmpty() || result.last().length+1+word.length > width) {
62 result.add(word)
63 } else {
64 result[result.lastIndex] += " $word"
65 }
66 }
67 return result
68 }
69 private val markupTokens = Regex("""</?[ibu]>|</font>|<font(?=[ >])[^>]*>|\{\\an[1-9]\}|.""")
70 private fun markup(line: String): List<Item> = markupTokens.findAll(line).map { match ->
71 val text = match.value
72 val kind = when {
73 text.startsWith("{\\an") -> 3
74 text.startsWith("<font") || text == "</font>" -> 2
75 text.length > 1 -> 1
76 else -> 0
77 }
78 Item(text,kind,if (kind == 3) text[4]-'0' else 0)
79 }.toList()
80 private fun listing(position: Int, cue: Cue) = "$position ${cue.index} ${srt(cue.start)} ${srt(cue.end)} ${cue.text.joinToString("|")}"
81 fun load(blocks: List<List<String>>): List<String> {
82 val output = mutableListOf<String>()
83 var badCount = 0
84 for ((i,block) in blocks.withIndex()) {
85 var reason = ""
86 var start = 0L
87 var end = 0L
88 if (!Regex("[0-9]{1,6}").matches(block[0])) {
89 reason = "index"
90 } else {
91 try {
92 if (block.size < 2 || block[1].length != 29 || block[1].substring(12,17) != " --> ") bad()
93 start = time(block[1].substring(0,12))
94 end = time(block[1].substring(17))
95 } catch (problem: Problem) {
96 reason = "time"
97 }
98 if (reason.isEmpty() && block.size < 3) reason = "no text"
99 }
100 if (reason.isNotEmpty()) {
101 output.add("BAD CUE ${i+1}: $reason")
102 badCount++
103 } else {
104 cues.add(Cue(block[0].toLong(),start,end,block.drop(2)))
105 }
106 }
107 output.add("LOAD cues=${cues.size} bad=$badCount")
108 return output
109 }
110 fun execute(line: String): List<String> {
111 val command = line.substringBefore(' ')
112 var schema = schemas[command] ?: throw Problem("unknown command")
113 if (command == "DELETE" && line.split(' ').size == 3) schema = "PP"
114 val tokens = if (schema.endsWith('X')) line.split(' ',limit = schema.length+1) else line.split(' ')
115 if (tokens.size != schema.length+1) bad()
116 val args = tokens.drop(1)
117 val values = LongArray(args.size)
118 for ((i,arg) in args.withIndex()) {
119 when (val type = schema[i]) {
120 'T' -> values[i] = time(arg)
121 'F' -> fps(arg)
122 'X' -> {
123 if (command in setOf("INSERT","TEXT")) text(arg)
124 if (command == "REPLACE") {
125 val parts = arg.split('|')
126 if (parts.size != 2 || parts.any { it.isEmpty() }) bad()
127 }
128 }
129 else -> {
130 val negative = type == 'I' && arg.startsWith('-')
131 values[i] = number(if (negative) arg.drop(1) else arg)*(if (negative) -1 else 1)
132 val v = values[i]
133 if (type == 'P' && v < 1 || type == 'r' && v !in 1..100000 || type == 'g' && v > 60000 ||
134 type == 'e' && v !in 1..60000 || type == 'w' && v !in 10..80) bad()
135 }
136 }
137 }
138 if (command == "DELETE" && values.size == 2 && values[1] < values[0]) bad()
139 for (i in args.indices) {
140 if (schema[i] == 'P' && values[i] > cues.size+(if (command == "INSERT") 1 else 0)) {
141 throw Problem("no cue ${values[i]}")
142 }
143 }
144 if (command == "SYNC" && cues[values[0].toInt()-1].start == cues[values[2].toInt()-1].start) {
145 throw Problem("sync points coincide")
146 }
147 if (command == "UNDO") {
148 if (history.isEmpty()) throw Problem("nothing to undo")
149 val undo = history.removeLast()
150 cues = undo.cues.toMutableList()
151 return listOf("UNDO command=${undo.command}")
152 }
153 if (command in modifying) history.addLast(Undo(command,cues.toList()))
154 val output = mutableListOf<String>()
155 val n = cues.size
156 val v = values.firstOrNull() ?: 0L
157 when (command) {
158 "LIST", "AT", "SEARCH", "GAPS" -> {
159 var count = 0
160 for ((i,cue) in cues.withIndex()) {
161 if (command == "LIST" || command == "AT" && cue.start <= v && v < cue.end) {
162 output.add(listing(i+1,cue))
163 count++
164 }
165 if (command == "SEARCH") {
166 val found = cue.text.firstOrNull { args[0] in it }
167 if (found != null) {
168 output.add("${i+1} ${srt(cue.start)} $found")
169 count++
170 }
171 }
172 if (command == "GAPS" && i+1 < n && cues[i+1].start-cue.end < v) {
173 output.add("${i+1} ${i+2} ${cues[i+1].start-cue.end}")
174 count++
175 }
176 }
177 if (command != "LIST") {
178 val label = when (command) {
179 "AT" -> "active"
180 "SEARCH" -> "found"
181 else -> "count"
182 }
183 output.add("$command $label=$count")
184 }
185 }
186 "VALIDATE" -> {
187 for ((i,c) in cues.withIndex()) {
188 val d = c.duration
189 val p = "cue ${i+1}: "
190 if (c.index != (i+1).toLong()) output.add("${p}index ${c.index} (expected ${i+1})")
191 if (d <= 0) output.add("${p}end not after start")
192 if (i > 0 && c.start < cues[i-1].end) output.add("${p}overlaps previous by ${cues[i-1].end-c.start} ms")
193 if (d in 1..699) output.add("${p}too short ($d ms)")
194 if (d > 7000) output.add("${p}too long ($d ms)")
195 for ((j,t) in c.text.withIndex()) {
196 if (t.length > 42) output.add("${p}line ${j+1} too long (${t.length} chars)")
197 }
198 if (c.text.size >= 3) output.add("${p}too many lines (${c.text.size})")
199 if (d > 0 && c.chars*1000 > 21*d) output.add("${p}too fast (${decimal(c.chars*100000/d)} cps)")
200 }
201 output.add("VALIDATE problems=${output.size}")
202 }
203 "STATS" -> {
204 output.add("STATS cues=$n")
205 if (n > 0) {
206 val durations = cues.map { maxOf(0,it.duration) }
207 val total = durations.sum()
208 val chars = cues.filter { it.duration > 0 }.sumOf { it.chars }
209 val longest = durations.maxOrNull()!!
210 output.add("duration total=${srt(total)} mean=${srt(total/n)}")
211 output.add("speed "+if (total == 0L) "-" else decimal(chars*100000/total))
212 output.add("longest cue=${durations.indexOf(longest)+1} duration=${srt(longest)}")
213 if (n == 1) output.add("gap none")
214 else {
215 val gaps = (0 until n-1).map { cues[it+1].start-cues[it].end }
216 val gap = gaps.maxOrNull()!!
217 output.add("gap cue=${gaps.indexOf(gap)+1} ms=$gap")
218 }
219 }
220 }
221 "PRINT", "TOVTT", "TOASS" -> {
222 if (command == "TOVTT") output.addAll(listOf("WEBVTT",""))
223 if (command == "TOASS") output.addAll(listOf("[Events]","Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text"))
224 for (c in cues) {
225 var code = 0
226 val lines = c.text.map { t ->
227 val converted = buildString {
228 for (item in markup(t)) {
229 if (code == 0 && item.kind == 3) code = item.digit
230 when {
231 command == "PRINT" -> append(item.text)
232 command == "TOASS" -> {
233 if (item.kind == 1) {
234 val closing = item.text.startsWith("</")
235 val tag = item.text[if (closing) 2 else 1]
236 append("{\\$tag${if (closing) 0 else 1}}")
237 } else if (item.kind != 2) append(item.text)
238 }
239 item.kind == 1 -> append(item.text)
240 item.kind == 0 -> append(when (item.text) {
241 "&" -> "&amp;"
242 "<" -> "&lt;"
243 ">" -> "&gt;"
244 else -> item.text
245 })
246 }
247 }
248 }
249 if (command == "TOVTT") converted.trim(' ') else converted
250 }.filter { command != "TOVTT" || it.isNotEmpty() }
251 if (command == "TOASS") output.add("Dialogue: 0,${ass(c.start)},${ass(c.end)},Default,,0,0,0,,${lines.joinToString("\\N")}")
252 else if (lines.isNotEmpty()) {
253 var settings = ""
254 if (command == "TOVTT" && code != 0) {
255 if (code >= 7) settings += " line:0"
256 else if (code >= 4) settings += " line:50%"
257 if (code%3 == 1) settings += " align:start"
258 if (code%3 == 0) settings += " align:end"
259 }
260 output.add(c.index.toString())
261 val times = "${srt(c.start)} --> ${srt(c.end)}"
262 output.add((if (command == "TOVTT") times.replace(',','.') else times)+settings)
263 output.addAll(lines)
264 output.add("")
265 }
266 }
267 }
268 "FRAMES" -> {
269 val (a,b) = fps(args[0])
270 for ((i,c) in cues.withIndex()) output.add("${i+1} ${c.start*a/(1000*b)} ${c.end*a/(1000*b)}")
271 }
272 "SHIFT", "SHIFTFROM", "SCALE", "SYNC", "SNAP" -> {
273 val first = if (command == "SHIFTFROM") v.toInt()-1 else 0
274 val delta = if (command == "SHIFTFROM") values[1] else v
275 val s1 = if (command == "SYNC") cues[v.toInt()-1].start else 0L
276 val s2 = if (command == "SYNC") cues[values[2].toInt()-1].start else 0L
277 val (a,b) = if (command == "SNAP") fps(args[0]) else (1L to 1L)
278 var clamped = 0
279 var changed = 0
280 for (i in first until n) {
281 val c = cues[i]
282 var clamp = false
283 val times = listOf(c.start,c.end).map { old ->
284 val next = when (command) {
285 "SCALE" -> round(old*v,values[1])
286 "SYNC" -> round(values[1]*(s2-s1)+(old-s1)*(values[3]-values[1]),s2-s1)
287 "SNAP" -> round(round(old*a,1000*b)*1000*b,a)
288 else -> old+delta
289 }
290 if (next < 0) clamp = true
291 val result = maxOf(0,next)
292 if (result != old) changed++
293 result
294 }
295 if (clamp) clamped++
296 cues[i] = c.copy(start = times[0],end = times[1])
297 }
298 output.add(if (command == "SNAP") "SNAP changed=$changed" else
299 "$command cues=${n-first}"+(if (command == "SCALE") "" else " clamped=$clamped"))
300 }
301 "FIXOVERLAP", "EXTEND" -> {
302 var changed = 0
303 var failed = 0
304 for (i in cues.indices) {
305 val c = cues[i]
306 var end = c.end
307 if (command == "FIXOVERLAP" && i+1 < n && end > cues[i+1].start-v) {
308 val candidate = cues[i+1].start-v
309 if (candidate > c.start) {
310 end = candidate
311 changed++
312 } else {
313 output.add("cue ${i+1}: cannot fix")
314 failed++
315 }
316 }
317 if (command == "EXTEND" && c.duration < v) {
318 val candidate = if (i+1 < n) minOf(c.start+v,cues[i+1].start) else c.start+v
319 if (candidate > end) {
320 end = candidate
321 changed++
322 }
323 }
324 cues[i] = c.copy(end = end)
325 }
326 output.add(if (command == "FIXOVERLAP") "FIXOVERLAP fixed=$changed failed=$failed" else
327 "EXTEND extended=$changed short=${cues.count { it.duration < v }}")
328 }
329 "TIME" -> {
330 cues[v.toInt()-1] = cues[v.toInt()-1].copy(start = values[1],end = values[2])
331 output.add("TIME cue=$v")
332 }
333 "SORT" -> {
334 val old = cues.toList()
335 cues.sortWith(compareBy<Cue> { it.start }.thenBy { it.end })
336 output.add("SORT moved=${cues.indices.count { old[it] !== cues[it] }}")
337 }
338 "RENUMBER" -> {
339 val changed = cues.indices.count { cues[it].index != (it+1).toLong() }
340 cues = cues.mapIndexed { i,c -> c.copy(index = (i+1).toLong()) }.toMutableList()
341 output.add("RENUMBER changed=$changed")
342 }
343 "MERGE" -> {
344 val result = mutableListOf<Cue>()
345 var merged = 0
346 for (b in cues) {
347 val a = result.lastOrNull()
348 if (a != null && b.start-a.end < v && !a.dialogue && !b.dialogue) {
349 val wrapped = wrap(a.text+b.text,42)
350 if (wrapped.size <= 2) {
351 result[result.lastIndex] = a.copy(end = maxOf(a.end,b.end),text = wrapped)
352 merged++
353 continue
354 }
355 }
356 result.add(b)
357 }
358 cues = result
359 output.add("MERGE merged=$merged cues=${cues.size}")
360 }
361 "SPLIT" -> {
362 val result = mutableListOf<Cue>()
363 var split = 0
364 for (c in cues) {
365 if (c.duration <= 0 || c.dialogue) {
366 result.add(c)
367 continue
368 }
369 val wrapped = wrap(c.text,v)
370 if (wrapped.size <= 2) {
371 result.add(c.copy(text = wrapped))
372 continue
373 }
374 split++
375 val parts = wrapped.chunked(2)
376 val counts = parts.map { it.sumOf { line2 -> line2.length.toLong() } }
377 val total = counts.sum()
378 val durations = counts.map { c.duration*it/total }.toMutableList()
379 val order = parts.indices.sortedWith(compareByDescending<Int> { c.duration*counts[it]%total }.thenBy { it })
380 for (i in 0 until (c.duration-durations.sum()).toInt()) durations[order[i]]++
381 var start = c.start
382 for (i in parts.indices) {
383 result.add(Cue(c.index,start,start+durations[i],parts[i]))
384 start += durations[i]
385 }
386 }
387 cues = result
388 output.add("SPLIT split=$split cues=${cues.size}")
389 }
390 "INSERT" -> {
391 cues.add(v.toInt()-1,Cue(v,values[1],values[2],text(args[3])))
392 output.add("INSERT cue=$v cues=${cues.size}")
393 }
394 "DELETE" -> {
395 val last = values.getOrElse(1) { v }
396 cues.subList(v.toInt()-1,last.toInt()).clear()
397 output.add("DELETE removed=${last-v+1} cues=${cues.size}")
398 }
399 "TEXT" -> {
400 val lines = text(args[1])
401 cues[v.toInt()-1] = cues[v.toInt()-1].copy(text = lines)
402 output.add("TEXT cue=$v lines=${lines.size}")
403 }
404 "BALANCE" -> {
405 var changed = 0
406 var skipped = 0
407 var dialogue = 0
408 for ((i,c) in cues.withIndex()) {
409 if (c.dialogue) {
410 dialogue++
411 continue
412 }
413 val parts = words(c.text)
414 val joined = parts.joinToString(" ")
415 var best: List<String>? = if (joined.length <= 42) listOf(joined) else null
416 var difference = Int.MAX_VALUE
417 var firstLength = Int.MAX_VALUE
418 if (joined.length > 42) {
419 for (j in 1 until parts.size) {
420 val a = parts.take(j).joinToString(" ")
421 val b = parts.drop(j).joinToString(" ")
422 val d = kotlin.math.abs(a.length-b.length)
423 if (a.length <= 42 && b.length <= 42 && (d < difference || d == difference && a.length < firstLength)) {
424 best = listOf(a,b)
425 difference = d
426 firstLength = a.length
427 }
428 }
429 }
430 if (best == null) skipped++
431 else if (best != c.text) {
432 changed++
433 cues[i] = c.copy(text = best)
434 }
435 }
436 output.add("BALANCE changed=$changed skipped=$skipped dialogue=$dialogue")
437 }
438 "STRIP", "CLEANHI" -> {
439 val result = mutableListOf<Cue>()
440 var tags = 0
441 var segments = 0
442 var labels = 0
443 var removed = 0
444 for (c in cues) {
445 val lines = mutableListOf<String>()
446 for (t in c.text) {
447 val builder = StringBuilder()
448 if (command == "STRIP") {
449 for (item in markup(t)) {
450 if (item.kind == 0) {
451 builder.append(item.text)
452 } else {
453 tags++
454 }
455 }
456 } else {
457 var p = 0
458 while (p < t.length) {
459 val end = when (t[p]) {
460 '[' -> t.indexOf(']',p+1)
461 '(' -> t.indexOf(')',p+1)
462 else -> -1
463 }
464 if (end >= 0) {
465 segments++
466 p = end+1
467 } else {
468 builder.append(t[p])
469 p++
470 }
471 }
472 }
473 var cleaned = if (command == "STRIP") builder.toString().trim(' ') else normalize(builder.toString())
474 if (command == "CLEANHI") {
475 val prefix = if (cleaned.startsWith("- ")) "- " else ""
476 val body = cleaned.drop(prefix.length)
477 val colon = body.indexOf(':')
478 if (colon > 0 && Regex("[A-Z][A-Z0-9 ]*").matches(body.take(colon))) {
479 labels++
480 cleaned = normalize(prefix+body.drop(colon+1))
481 }
482 }
483 if (cleaned.isEmpty() || command == "CLEANHI" && cleaned == "-") {
484 removed++
485 } else {
486 lines.add(cleaned)
487 }
488 }
489 if (lines.isNotEmpty()) result.add(c.copy(text = lines))
490 }
491 val removedCues = n-result.size
492 cues = result
493 output.add(if (command == "STRIP") "STRIP tags=$tags removed_lines=$removed removed_cues=$removedCues" else
494 "CLEANHI segments=$segments labels=$labels removed_lines=$removed removed_cues=$removedCues")
495 }
496 "REPLACE" -> {
497 val (old,replacement) = args[0].split('|')
498 var count = 0
499 var changed = 0
500 for ((i,c) in cues.withIndex()) {
501 val before = count
502 val lines = c.text.map { t ->
503 val pieces = t.split(old)
504 count += pieces.size-1
505 pieces.joinToString(replacement)
506 }
507 if (count > before) changed++
508 cues[i] = c.copy(text = lines)
509 }
510 output.add("REPLACE count=$count cues=$changed")
511 }
512 }
513 return output
514 }
515}

517fun main() {
518 val input = System.`in`.bufferedReader()
519 val output = System.out.bufferedWriter()
520 val blocks = mutableListOf<List<String>>()
521 var block = mutableListOf<String>()
522 while (true) {
523 val line = input.readLine()
524 if (line == "###") break
525 if (line.isEmpty()) {
526 if (block.isNotEmpty()) blocks.add(block)
527 block = mutableListOf()
528 } else block.add(line)
529 }
530 if (block.isNotEmpty()) blocks.add(block)
531 val editor = Editor()
532 fun emit(lines: List<String>) {
533 for (line in lines) {
534 output.write(line)
535 output.newLine()
536 }
537 }
538 emit(editor.load(blocks))
539 repeat(input.readLine().toInt()) {
540 try {
541 emit(editor.execute(input.readLine()))
542 } catch (problem: Problem) {
543 emit(listOf("ERROR ${problem.message}"))
544 }
545 }
546 output.flush()
547}
