3struct Cue {
4 var index: Int
5 var start: Int
6 var end: Int
7 var text: [String]
8 var duration: Int { end-start }
9 var chars: Int { text.reduce(0) { $0+$1.count } }

11}

26 var cues: [Cue] = []
27 var history: [Undo] = []
28 let schemas = [
29 "LIST":"", "VALIDATE":"", "SEARCH":"X", "AT":"T", "GAPS":"g", "STATS":"",
30 "PRINT":"", "TOVTT":"", "TOASS":"", "FRAMES":"F", "SHIFT":"I", "SHIFTFROM":"PI",
31 "SCALE":"rr", "SYNC":"PTPT", "SNAP":"F", "FIXOVERLAP":"g", "EXTEND":"e", "TIME":"PTT",
32 "SORT":"", "RENUMBER":"", "MERGE":"g", "SPLIT":"w", "INSERT":"PTTX", "DELETE":"P",
33 "TEXT":"PX", "BALANCE":"", "STRIP":"", "CLEANHI":"", "REPLACE":"X", "UNDO":""
34 ]
35 let modifying: Set<String> = ["SHIFT","SHIFTFROM","SCALE","SYNC","SNAP","FIXOVERLAP","EXTEND",
36 "TIME","SORT","RENUMBER","MERGE","SPLIT","INSERT","DELETE","TEXT","BALANCE","STRIP","CLEANHI","REPLACE"]

38 func bad() -> Problem { Problem(message: "bad arguments") }
39 func sub(_ s: String, _ start: Int, _ end: Int? = nil) -> String {
40 String(s.dropFirst(start).prefix((end ?? s.count)-start))
41 }
42 func number(_ s: String) throws -> Int {
43 guard (1...9).contains(s.count) && s.allSatisfy({ $0 >= "0" && $0 <= "9" }) else {
44 throw bad()
45 }
46 return Int(s)!
47 }
48 func time(_ s: String) throws -> Int {
49 let c = Array(s)
50 guard c.count == 12 && c[2] == ":" && c[5] == ":" && c[8] == "," else {
51 throw bad()
52 }
53 for i in [0,1,3,4,6,7,9,10,11] {
54 guard c[i] >= "0" && c[i] <= "9" else { throw bad() }
55 }
56 let hours = Int(sub(s,0,2))!
57 let minutes = Int(sub(s,3,5))!
58 let seconds = Int(sub(s,6,8))!
59 guard minutes <= 59 && seconds <= 59 else { throw bad() }
60 return hours*3600000+minutes*60000+seconds*1000+Int(sub(s,9))!
61 }
62 func fps(_ s: String) throws -> (Int,Int) {
63 let pieces = s.components(separatedBy: "/")
64 guard pieces.count <= 2 else { throw bad() }
65 let a = try number(pieces[0])
66 let b: Int
67 if pieces.count == 2 {
68 b = try number(pieces[1])
69 } else {
70 b = 1
71 }
72 guard (1...100000).contains(a) && (1...100000).contains(b) else { throw bad() }
73 return (a,b)
74 }
75 func text(_ s: String) throws -> [String] {
76 let lines = s.components(separatedBy: "|")
77 guard lines.allSatisfy({ !$0.isEmpty }) else { throw bad() }
78 return lines
79 }
80 func srt(_ t: Int) -> String {
81 String(format: "%02ld:%02ld:%02ld,%03ld",locale: Locale(identifier: "en_US_POSIX"),t/3600000,t/60000%60,t/1000%60,t%1000)
82 }
83 func ass(_ t: Int) -> String {
84 let c = (t+5)/10
85 return String(format: "%ld:%02ld:%02ld.%02ld",locale: Locale(identifier: "en_US_POSIX"),c/360000,c/6000%60,c/100%60,c%100)
86 }
87 func decimal(_ v: Int) -> String {
88 String(format: "%ld.%02ld",locale: Locale(identifier: "en_US_POSIX"),v/100,v%100)
89 }
90 func round(_ numerator: Int, _ denominator: Int) -> Int {
91 let a = denominator < 0 ? -numerator : numerator
92 let b = abs(denominator)
93 let n = 2*a+b
94 let d = 2*b
95 return n/d-(n%d < 0 ? 1 : 0)
96 }
97 func words(_ lines: [String]) -> [String] {
98 lines.joined(separator: " ").split(separator: " ").map(String.init)
99 }
100 func normalize(_ s: String) -> String { words([s]).joined(separator: " ") }
101 func wrap(_ lines: [String], _ width: Int) -> [String] {
102 var result: [String] = []
103 for word in words(lines) {
104 if result.isEmpty || result.last!.count+1+word.count > width {
105 result.append(word)
106 } else {
107 result[result.count-1] += " "+word
108 }
109 }
110 return result
111 }
112 func markup(_ line: String) -> [Item] {
113 line.matches(of: #/</?[ibu]>|</font>|<font(?=[ >])[^>]*>|\{\\an[1-9]\}|./#).map {
114 let text = String($0.output)
115 let kind = text.hasPrefix("{\\an") ? 3 :
116 (text.hasPrefix("<font") || text == "</font>" ? 2 : (text.count > 1 ? 1 : 0))
117 return Item(text: text,kind: kind,digit: kind == 3 ? Int(String(text.dropFirst(4).first!))! : 0)
118 }
119 }
120 func listing(_ p: Int, _ c: Cue) -> String {
121 "\(p) \(c.index) \(srt(c.start)) \(srt(c.end)) "+c.text.joined(separator: "|")
122 }
123 func load(_ blocks: [[String]]) -> [String] {
124 var output: [String] = []
125 var badCount = 0
126 for (i,block) in blocks.enumerated() {
127 var reason = ""
128 var start = 0
129 var end = 0
130 if !(1...6).contains(block[0].count) || !block[0].allSatisfy({ $0 >= "0" && $0 <= "9" }) {
131 reason = "index"
132 } else {
133 do {
134 guard block.count >= 2 && block[1].count == 29 && sub(block[1],12,17) == " --> " else {
135 throw bad()
136 }
137 start = try time(sub(block[1],0,12))
138 end = try time(sub(block[1],17))
139 } catch {
140 reason = "time"
141 }
142 if reason.isEmpty && block.count < 3 { reason = "no text" }
143 }
144 if !reason.isEmpty {
145 output.append("BAD CUE \(i+1): "+reason)
146 badCount += 1
147 } else {
148 cues.append(Cue(index: Int(block[0])!,start: start,end: end,text: Array(block.dropFirst(2))))
149 }
150 }
151 output.append("LOAD cues=\(cues.count) bad=\(badCount)")
152 return output
153 }
154 func execute(_ line: String) throws -> [String] {
155 let command = String(line.split(separator: " ",maxSplits: 1)[0])
156 guard var schema = schemas[command] else { throw Problem(message: "unknown command") }
157 if command == "DELETE" && line.split(separator: " ").count == 3 { schema = "PP" }
158 let tokens = schema.hasSuffix("X") ?
159 line.split(separator: " ",maxSplits: schema.count,omittingEmptySubsequences: false).map(String.init) :
160 line.components(separatedBy: " ")
161 guard tokens.count == schema.count+1 else { throw bad() }
162 let args = Array(tokens.dropFirst())
163 let types = Array(schema)
164 var values = Array(repeating: 0,count: args.count)
165 for i in args.indices {
166 let type = types[i]
167 let arg = args[i]
168 if type == "T" { values[i] = try time(arg) }
169 else if type == "F" { _ = try fps(arg) }
170 else if type == "X" {
171 if command == "INSERT" || command == "TEXT" { _ = try text(arg) }
172 if command == "REPLACE" {
173 let pieces = arg.components(separatedBy: "|")
174 guard pieces.count == 2 && pieces.allSatisfy({ !$0.isEmpty }) else { throw bad() }
175 }
176 } else {
177 let negative = type == "I" && arg.hasPrefix("-")
178 values[i] = try number(negative ? sub(arg,1) : arg)*(negative ? -1 : 1)
179 let v = values[i]
180 if (type == "P" && v < 1) || (type == "r" && !(1...100000).contains(v)) ||
181 (type == "g" && v > 60000) || (type == "e" && !(1...60000).contains(v)) ||
182 (type == "w" && !(10...80).contains(v)) { throw bad() }
183 }
184 }
185 if command == "DELETE" && values.count == 2 && values[1] < values[0] { throw bad() }
186 for i in args.indices {
187 if types[i] == "P" && values[i] > cues.count+(command == "INSERT" ? 1 : 0) {
188 throw Problem(message: "no cue \(values[i])")
189 }
190 }
191 if command == "SYNC" && cues[values[0]-1].start == cues[values[2]-1].start {
192 throw Problem(message: "sync points coincide")
193 }
194 if command == "UNDO" {
195 guard let undo = history.popLast() else { throw Problem(message: "nothing to undo") }
196 cues = undo.cues
197 return ["UNDO command="+undo.command]
198 }
199 if modifying.contains(command) { history.append(Undo(command: command,cues: cues)) }
200 var output: [String] = []
201 let n = cues.count
202 let v = values.first ?? 0
203 switch command {
204 case "LIST", "AT", "SEARCH", "GAPS":
205 var count = 0
206 for (i,c) in cues.enumerated() {
207 if command == "LIST" || (command == "AT" && c.start <= v && v < c.end) {
208 output.append(listing(i+1,c))
209 count += 1
210 }
211 if command == "SEARCH", let found = c.text.first(where: { $0.contains(args[0]) }) {
212 output.append("\(i+1) \(srt(c.start)) "+found)
213 count += 1
214 }
215 if command == "GAPS" && i+1 < n && cues[i+1].start-c.end < v {
216 output.append("\(i+1) \(i+2) \(cues[i+1].start-c.end)")
217 count += 1
218 }
219 }
220 if command != "LIST" {
221 let label = command == "AT" ? "active" : (command == "SEARCH" ? "found" : "count")
222 output.append(command+" "+label+"=\(count)")
223 }
224 case "VALIDATE":
225 for (i,c) in cues.enumerated() {
226 let d = c.duration
227 let p = "cue \(i+1): "
228 if c.index != i+1 { output.append(p+"index \(c.index) (expected \(i+1))") }
229 if d <= 0 { output.append(p+"end not after start") }
230 if i > 0 && c.start < cues[i-1].end { output.append(p+"overlaps previous by \(cues[i-1].end-c.start) ms") }
231 if d > 0 && d < 700 { output.append(p+"too short (\(d) ms)") }
232 if d > 7000 { output.append(p+"too long (\(d) ms)") }
233 for (j,t) in c.text.enumerated() {
234 if t.count > 42 { output.append(p+"line \(j+1) too long (\(t.count) chars)") }
235 }
236 if c.text.count >= 3 { output.append(p+"too many lines (\(c.text.count))") }
237 if d > 0 && c.chars*1000 > 21*d {
238 output.append(p+"too fast (\(decimal(c.chars*100000/d)) cps)")
239 }
240 }
241 output.append("VALIDATE problems=\(output.count)")
242 case "STATS":
243 output.append("STATS cues=\(n)")
244 if n > 0 {
245 let durations = cues.map { max(0,$0.duration) }
246 let total = durations.reduce(0,+)
247 let chars = cues.filter { $0.duration > 0 }.reduce(0) { $0+$1.chars }
248 let longest = durations.max()!
249 output.append("duration total=\(srt(total)) mean=\(srt(total/n))")
250 output.append("speed "+(total == 0 ? "-" : decimal(chars*100000/total)))
251 output.append("longest cue=\(durations.firstIndex(of: longest)!+1) duration=\(srt(longest))")
252 if n == 1 { output.append("gap none") }
253 else {
254 let gaps = (0..<(n-1)).map { cues[$0+1].start-cues[$0].end }
255 let gap = gaps.max()!
256 output.append("gap cue=\(gaps.firstIndex(of: gap)!+1) ms=\(gap)")
257 }
258 }
259 case "PRINT", "TOVTT", "TOASS":
260 if command == "TOVTT" { output += ["WEBVTT",""] }
261 if command == "TOASS" { output += ["[Events]","Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text"] }
262 for c in cues {
263 var code = 0
264 var lines: [String] = []
265 for t in c.text {
266 var converted = ""
267 for item in markup(t) {
268 if code == 0 && item.kind == 3 { code = item.digit }
269 if command == "PRINT" { converted += item.text }
270 else if command == "TOASS" {
271 if item.kind == 1 {
272 let closing = item.text.hasPrefix("</")
273 let tag = Array(item.text)[closing ? 2 : 1]
274 converted += "{\\\(tag)\(closing ? 0 : 1)}"
275 } else if item.kind != 2 { converted += item.text }
276 } else if item.kind == 1 { converted += item.text }
277 else if item.kind == 0 {
278 switch item.text {
279 case "&": converted += "&amp;"
280 case "<": converted += "&lt;"
281 case ">": converted += "&gt;"
282 default: converted += item.text
283 }
284 }
285 }
286 if command == "TOVTT" { converted = converted.trimmingCharacters(in: CharacterSet(charactersIn: " ")) }
287 if command != "TOVTT" || !converted.isEmpty { lines.append(converted) }
288 }
289 if command == "TOASS" {
290 output.append("Dialogue: 0,\(ass(c.start)),\(ass(c.end)),Default,,0,0,0,,"+lines.joined(separator: "\\N"))
291 } else if !lines.isEmpty {
292 var settings = ""
293 if command == "TOVTT" && code != 0 {
294 if code >= 7 { settings += " line:0" }
295 else if code >= 4 { settings += " line:50%" }
296 if code%3 == 1 { settings += " align:start" }
297 if code%3 == 0 { settings += " align:end" }
298 }
299 output.append(String(c.index))
300 let times = "\(srt(c.start)) --> \(srt(c.end))"
301 output.append((command == "TOVTT" ? times.replacingOccurrences(of: ",",with: ".") : times)+settings)
302 output += lines
303 output.append("")
304 }
305 }
306 case "FRAMES":
307 let (a,b) = try fps(args[0])
308 for (i,c) in cues.enumerated() { output.append("\(i+1) \(c.start*a/(1000*b)) \(c.end*a/(1000*b))") }
309 case "SHIFT", "SHIFTFROM", "SCALE", "SYNC", "SNAP":
310 let first = command == "SHIFTFROM" ? v-1 : 0
311 let delta = command == "SHIFTFROM" ? values[1] : v
312 let s1 = command == "SYNC" ? cues[v-1].start : 0
313 let s2 = command == "SYNC" ? cues[values[2]-1].start : 0
314 let rate: (Int,Int)
315 if command == "SNAP" {
316 rate = try fps(args[0])
317 } else {
318 rate = (1,1)
319 }
320 let (a,b) = rate
321 var clamped = 0
322 var changed = 0
323 for i in first..<n {
324 var c = cues[i]
325 var times = [c.start,c.end]
326 var clamp = false
327 for j in times.indices {
328 let old = times[j]
329 let next: Int
330 switch command {
331 case "SCALE": next = round(old*v,values[1])
332 case "SYNC": next = round(values[1]*(s2-s1)+(old-s1)*(values[3]-values[1]),s2-s1)
333 case "SNAP": next = round(round(old*a,1000*b)*1000*b,a)
334 default: next = old+delta
335 }
336 if next < 0 { clamp = true }
337 times[j] = max(0,next)
338 if times[j] != old { changed += 1 }
339 }
340 if clamp { clamped += 1 }
341 c.start = times[0]
342 c.end = times[1]
343 cues[i] = c
344 }
345 output.append(command == "SNAP" ? "SNAP changed=\(changed)" :
346 command+" cues=\(n-first)"+(command == "SCALE" ? "" : " clamped=\(clamped)"))
347 case "FIXOVERLAP", "EXTEND":
348 var changed = 0
349 var failed = 0
350 for i in cues.indices {
351 var c = cues[i]
352 if command == "FIXOVERLAP" && i+1 < n && c.end > cues[i+1].start-v {
353 let candidate = cues[i+1].start-v
354 if candidate > c.start {
355 c.end = candidate
356 changed += 1
357 } else {
358 output.append("cue \(i+1): cannot fix")
359 failed += 1
360 }
361 }
362 if command == "EXTEND" && c.duration < v {
363 let candidate = i+1 < n ? min(c.start+v,cues[i+1].start) : c.start+v
364 if candidate > c.end {
365 c.end = candidate
366 changed += 1
367 }
368 }
369 cues[i] = c
370 }
371 output.append(command == "FIXOVERLAP" ? "FIXOVERLAP fixed=\(changed) failed=\(failed)" :
372 "EXTEND extended=\(changed) short=\(cues.filter { $0.duration < v }.count)")
373 case "TIME":
374 cues[v-1].start = values[1]
375 cues[v-1].end = values[2]
376 output.append("TIME cue=\(v)")
377 case "SORT":
378 let ordered = cues.enumerated().sorted {
379 if $0.element.start != $1.element.start { return $0.element.start < $1.element.start }
380 if $0.element.end != $1.element.end { return $0.element.end < $1.element.end }
381 return $0.offset < $1.offset
382 }
383 let moved = ordered.enumerated().filter { $0.offset != $0.element.offset }.count
384 cues = ordered.map { $0.element }
385 output.append("SORT moved=\(moved)")
386 case "RENUMBER":
387 var changed = 0
388 for i in cues.indices {
389 if cues[i].index != i+1 { changed += 1 }
390 cues[i].index = i+1
391 }
392 output.append("RENUMBER changed=\(changed)")
393 case "MERGE":
394 var result: [Cue] = []
395 var merged = 0
396 for b in cues {
397 if var a = result.last, b.start-a.end < v && !a.dialogue && !b.dialogue {
398 let wrapped = wrap(a.text+b.text,42)
399 if wrapped.count <= 2 {
400 a.end = max(a.end,b.end)
401 a.text = wrapped
402 result[result.count-1] = a
403 merged += 1
404 continue
405 }
406 }
407 result.append(b)
408 }
409 cues = result
410 output.append("MERGE merged=\(merged) cues=\(cues.count)")
411 case "SPLIT":
412 var result: [Cue] = []
413 var split = 0
414 for var c in cues {
415 if c.duration <= 0 || c.dialogue {
416 result.append(c)
417 continue
418 }
419 let wrapped = wrap(c.text,v)
420 if wrapped.count <= 2 {
421 c.text = wrapped
422 result.append(c)
423 continue
424 }
425 split += 1
426 var parts: [[String]] = []
427 for i in stride(from: 0,to: wrapped.count,by: 2) {
428 parts.append(Array(wrapped[i..<min(i+2,wrapped.count)]))
429 }
430 let counts = parts.map { $0.reduce(0) { $0+$1.count } }
431 let total = counts.reduce(0,+)
432 var durations = counts.map { c.duration*$0/total }
433 let order = parts.indices.sorted {
434 let a = c.duration*counts[$0]%total
435 let b = c.duration*counts[$1]%total
436 return a == b ? $0 < $1 : a > b
437 }
438 let remainder = c.duration-durations.reduce(0,+)
439 for i in 0..<remainder { durations[order[i]] += 1 }
440 var start = c.start
441 for i in parts.indices {
442 result.append(Cue(index: c.index,start: start,end: start+durations[i],text: parts[i]))
443 start += durations[i]
444 }
445 }
446 cues = result
447 output.append("SPLIT split=\(split) cues=\(cues.count)")
448 case "INSERT":
449 cues.insert(Cue(index: v,start: values[1],end: values[2],text: try text(args[3])),at: v-1)
450 output.append("INSERT cue=\(v) cues=\(cues.count)")
451 case "DELETE":
452 let last = values.count == 2 ? values[1] : v
453 cues.removeSubrange((v-1)..<last)
454 output.append("DELETE removed=\(last-v+1) cues=\(cues.count)")
455 case "TEXT":
456 let lines = try text(args[1])
457 cues[v-1].text = lines
458 output.append("TEXT cue=\(v) lines=\(lines.count)")
459 case "BALANCE":
460 var changed = 0
461 var skipped = 0
462 var dialogue = 0
463 for i in cues.indices {
464 var c = cues[i]
465 if c.dialogue {
466 dialogue += 1
467 continue
468 }
469 let tokens = words(c.text)
470 let joined = tokens.joined(separator: " ")
471 var best: [String]? = joined.count <= 42 ? [joined] : nil
472 var difference = Int.max
473 var firstLength = Int.max
474 if joined.count > 42 && tokens.count > 1 {
475 for j in 1..<tokens.count {
476 let a = tokens.prefix(j).joined(separator: " ")
477 let b = tokens.dropFirst(j).joined(separator: " ")
478 let d = abs(a.count-b.count)
479 if a.count <= 42 && b.count <= 42 && (d < difference || (d == difference && a.count < firstLength)) {
480 best = [a,b]
481 difference = d
482 firstLength = a.count
483 }
484 }
485 }
486 if let best {
487 if best != c.text {
488 changed += 1
489 c.text = best
490 cues[i] = c
491 }
492 } else { skipped += 1 }
493 }
494 output.append("BALANCE changed=\(changed) skipped=\(skipped) dialogue=\(dialogue)")
495 case "STRIP", "CLEANHI":
496 var result: [Cue] = []
497 var tags = 0
498 var segments = 0
499 var labels = 0
500 var removed = 0
501 for var c in cues {
502 var lines: [String] = []
503 for t in c.text {
504 var cleaned = ""
505 if command == "STRIP" {
506 for item in markup(t) {
507 if item.kind == 0 { cleaned += item.text }
508 else { tags += 1 }
509 }
510 cleaned = cleaned.trimmingCharacters(in: CharacterSet(charactersIn: " "))
511 } else {
512 let chars = Array(t)
513 var p = 0
514 while p < chars.count {
515 let closer: Character? = chars[p] == "[" ? "]" : (chars[p] == "(" ? ")" : nil)
516 if let closer, let end = chars[(p+1)...].firstIndex(of: closer) {
517 segments += 1
518 p = end+1
519 } else {
520 cleaned.append(chars[p])
521 p += 1
522 }
523 }
524 cleaned = normalize(cleaned)
525 let prefix = cleaned.hasPrefix("- ") ? "- " : ""
526 let body = sub(cleaned,prefix.count)
527 let bodyChars = Array(body)
528 if let colon = bodyChars.firstIndex(of: ":"), colon > 0 {
529 let name = Array(bodyChars[..<colon])
530 if name[0] >= "A" && name[0] <= "Z" && name.allSatisfy({
531 ($0 >= "A" && $0 <= "Z") || ($0 >= "0" && $0 <= "9") || $0 == " "
532 }) {
533 labels += 1
534 cleaned = normalize(prefix+String(bodyChars[(colon+1)...]))
535 }
536 }
537 }
538 if cleaned.isEmpty || (command == "CLEANHI" && cleaned == "-") { removed += 1 }
539 else { lines.append(cleaned) }
540 }
541 if !lines.isEmpty {
542 c.text = lines
543 result.append(c)
544 }
545 }
546 let removedCues = n-result.count
547 cues = result
548 output.append(command == "STRIP" ? "STRIP tags=\(tags) removed_lines=\(removed) removed_cues=\(removedCues)" :
549 "CLEANHI segments=\(segments) labels=\(labels) removed_lines=\(removed) removed_cues=\(removedCues)")
550 case "REPLACE":
551 let parts = args[0].components(separatedBy: "|")
552 let old = parts[0]
553 let replacement = parts[1]
554 var count = 0
555 var changed = 0
556 for i in cues.indices {
557 var c = cues[i]
558 let before = count
559 c.text = c.text.map { t in
560 let pieces = t.components(separatedBy: old)
561 count += pieces.count-1
562 return pieces.joined(separator: replacement)
563 }
564 if count > before { changed += 1 }
565 cues[i] = c
566 }
567 output.append("REPLACE count=\(count) cues=\(changed)")
568 default:
569 preconditionFailure("Unreachable command")
570 }
571 return output
572 }
573}
