22fn unsigned(s: &str) -> Option<Integer> {
23 if s.is_empty() || s.len() > 9 || !s.bytes().all(|c| c.is_ascii_digit()) { None }
24 else { s.parse().ok() }
25}
26fn time_value(s: &str) -> Option<Integer> {
27 let b = s.as_bytes();
28 if b.len() != 12 || b[2] != b':' || b[5] != b':' || b[8] != b',' { return None; }
29 if b.iter().enumerate().any(|(i, c)| ![2, 5, 8].contains(&i) && !c.is_ascii_digit()) { return None; }
30 let h: Integer = s[..2].parse().ok()?;
31 let m: Integer = s[3..5].parse().ok()?;
32 let sec: Integer = s[6..8].parse().ok()?;
33 let ms: Integer = s[9..].parse().ok()?;
34 if m > 59 || sec > 59 { None } else { Some(h*3600000+m*60000+sec*1000+ms) }
35}
36fn time_text(t: Integer, kind: usize) -> String {
37 if kind == 2 {
38 let c = (t+5)/10;
39 format!("{}:{:02}:{:02}.{:02}", c/360000, c/6000%60, c/100%60, c%100)
40 } else {
41 format!("{:02}:{:02}:{:02}{}{:03}", t/3600000, t/60000%60, t/1000%60, if kind == 1 { "." } else { "," }, t%1000)
42 }
43}
44fn decimal(v: Integer) -> String { format!("{}.{:02}", v/100, v%100) }
45fn floor_ratio(mut a: Integer, mut b: Integer) -> Integer {
46 if b < 0 {
47 a = -a;
48 b = -b;
49 }
50 a.div_euclid(b)
51}
52fn half_up(mut a: Integer, mut b: Integer) -> Integer {
53 if b < 0 {
54 a = -a;
55 b = -b;
56 }
57 floor_ratio(2*a+b, 2*b)
58}
59fn count_chars(c: &Cue) -> Integer { c.text.iter().map(|s| s.len() as Integer).sum() }
60fn dialogue(c: &Cue) -> bool { c.text.iter().any(|line| line.starts_with("- ")) }
61fn words(lines: &[String]) -> Vec<&str> {
62 lines.iter().flat_map(|line| line.split(' ').filter(|s| !s.is_empty())).collect()
63}
64fn wrap(lines: &[String], width: usize) -> Vec<String> {
65 let mut result: Vec<String> = Vec::new();
66 for word in words(lines) {
67 if result.last().is_none_or(|s| s.len()+1+word.len() > width) { result.push(word.to_owned()); }
68 else {
69 result.last_mut().unwrap().push(' ');
70 result.last_mut().unwrap().push_str(&word);
71 }
72 }
73 result
74}
75fn scan_markup(line: &str) -> Vec<Markup<'_>> {
76 let tags = ["<i>", "</i>", "<b>", "</b>", "<u>", "</u>"];
77 let mut result = Vec::new();
78 let mut p = 0;
79 let b = line.as_bytes();
80 while p < line.len() {
81 let (mut size, mut kind, mut digit) = (1, 0, 0);
82 for tag in tags {
83 if line[p..].starts_with(tag) {
84 size = tag.len();
85 kind = 1;
86 break;
87 }
88 }
89 if kind == 0 && line[p..].starts_with("</font>") {
90 size = 7;
91 kind = 2;
92 }
93 if kind == 0 && line[p..].starts_with("<font") && p+5 < line.len() && [b' ', b'>'].contains(&b[p+5]) {
94 if let Some(end) = line[p+5..].find('>') {
95 size = 6+end;
96 kind = 2;
97 }
98 }
99 if kind == 0 && p+6 <= line.len() && line[p..].starts_with("{\\an") && (b'1'..=b'9').contains(&b[p+4]) && b[p+5] == b'}' {
100 size = 6;
101 kind = 3;
102 digit = (b[p+4]-b'0') as usize;
103 }
104 result.push(Markup { kind, text: &line[p..p+size], digit });
105 p += size;
106 }
107 result
108}
109fn converted(lines: &[String], mode: usize) -> (Vec<String>, usize, usize) {
110 let tags = ["<i>", "</i>", "<b>", "</b>", "<u>", "</u>"];
111 let ass = ["{\\i1}", "{\\i0}", "{\\b1}", "{\\b0}", "{\\u1}", "{\\u0}"];
112 let (mut first, mut removed) = (0, 0);
113 let mut result = Vec::new();
114 for line in lines {
115 let mut output = String::new();
116 for m in scan_markup(line) {
117 if m.kind == 3 && first == 0 { first = m.digit; }
118 match mode {
119 0 => {
120 if m.kind == 0 { output.push_str(m.text); }
121 else { removed += 1; }
122 }
123 1 => {
124 if m.kind == 1 { output.push_str(m.text); }
125 if m.kind == 0 {
126 output.push_str(match m.text {
127 "&" => "&amp;", "<" => "&lt;", ">" => "&gt;", _ => m.text,
128 });
129 }
130 }
131 _ => {
132 if m.kind == 1 { output.push_str(ass[tags.iter().position(|&s| s == m.text).unwrap()]); }
133 if m.kind == 0 || m.kind == 3 { output.push_str(m.text); }
134 }
135 }
136 }
137 if mode != 2 { output = output.trim_matches(' ').to_owned(); }
138 if mode == 2 || !output.is_empty() { result.push(output); }
139 }
140 (result, first, removed)
141}
142fn normalize(line: &str) -> String {
143 line.split(' ').filter(|s| !s.is_empty()).collect::<Vec<_>>().join(" ")
144}
145fn clean_hi(line: &str) -> (String, usize, usize) {
146 let mut output = String::new();
147 let (mut p, mut segments) = (0, 0);
148 let b = line.as_bytes();
149 while p < line.len() {
150 let end = match b[p] {
151 b'[' => line[p+1..].find(']'),
152 b'(' => line[p+1..].find(')'),
153 _ => None,
154 };
155 if let Some(end) = end {
156 segments += 1;
157 p += end+2;
158 } else {
159 output.push(b[p] as char);
160 p += 1;
161 }
162 }
163 let mut text = normalize(&output);
164 let (prefix, body) = if text.starts_with("- ") { ("- ", &text[2..]) } else { ("", text.as_str()) };
165 let mut labels = 0;
166 if let Some(colon) = body.find(':') {
167 let label = &body[..colon];
168 if label.as_bytes().first().is_some_and(u8::is_ascii_uppercase)
169 && label.bytes().all(|c| c.is_ascii_uppercase() || c.is_ascii_digit() || c == b' ') {
170 labels = 1;
171 text = normalize(&format!("{prefix}{}", &body[colon+1..]));
172 }
173 }
174 (text, segments, labels)
175}
176fn parse_arguments(line: &str, count: usize) -> Result<Arguments, String> {
177 let names = ["LIST", "VALIDATE", "SEARCH", "AT", "GAPS", "STATS", "PRINT", "TOVTT", "TOASS", "FRAMES", "SHIFT", "SHIFTFROM", "SCALE", "SYNC", "SNAP", "FIXOVERLAP", "EXTEND", "TIME", "SORT", "RENUMBER", "MERGE", "SPLIT", "INSERT", "DELETE", "TEXT", "BALANCE", "STRIP", "CLEANHI", "REPLACE", "UNDO"];
178 let schemas = ["", "", "S", "T", "U", "", "", "", "", "F", "I", "PI", "UU", "PTPT", "F", "U", "U", "PTT", "", "", "U", "U", "PTTX", "P", "PX", "", "", "", "R", ""];
179 let (name, rest) = line.split_once(' ').map_or((line, None), |(a, b)| (a, Some(b)));
180 let position = names.iter().position(|&s| s == name).ok_or("unknown command")?;
181 let mut schema = schemas[position];
182 let raw: Vec<String> = match rest {
183 None => Vec::new(),
184 Some(s) if schema.bytes().any(|c| b"SXR".contains(&c)) => s.splitn(schema.len(), ' ').map(str::to_owned).collect(),
185 Some(s) => s.split(' ').map(str::to_owned).collect(),
186 };
187 if name == "DELETE" && raw.len() == 2 { schema = "PP"; }
188 if raw.len() != schema.len() { return Err("bad arguments".to_owned()); }
189 let mut a = Arguments { name: name.to_owned(), values: vec![0; raw.len()], raw, denominator: 1 };
190 for (i, kind) in schema.bytes().enumerate() {
191 let s = a.raw[i].as_str();
192 let value = match kind {
193 b'P' => unsigned(s).filter(|&v| v > 0),
194 b'U' => unsigned(s),
195 b'I' => if let Some(rest) = s.strip_prefix('-') { unsigned(rest).map(|v| -v) } else { unsigned(s) },
196 b'T' => time_value(s),
197 b'F' => {
198 let parts: Vec<&str> = s.split('/').collect();
199 if parts.len() > 2 { None } else {
200 a.denominator = if parts.len() == 2 { unsigned(parts[1]).unwrap_or(0) } else { 1 };
201 unsigned(parts[0]).filter(|&v| (1..=100000).contains(&v) && (1..=100000).contains(&a.denominator))
202 }
203 }
204 b'X' => if s.split('|').all(|part| !part.is_empty()) { Some(0) } else { None },
205 b'R' => s.split_once('|').filter(|(old, new)| !old.is_empty() && !new.is_empty() && !new.contains('|')).map(|_| 0),
206 _ => if !s.is_empty() { Some(0) } else { None },
207 };
208 a.values[i] = value.ok_or("bad arguments")?;
209 }
210 let v = &a.values;
211 if name == "SCALE" && (!(1..=100000).contains(&v[0]) || !(1..=100000).contains(&v[1])) { return Err("bad arguments".to_owned()); }
212 if ["GAPS", "FIXOVERLAP", "MERGE", "EXTEND"].contains(&name) {
213 let minimum = if name == "EXTEND" { 1 } else { 0 };
214 if !(minimum..=60000).contains(&v[0]) { return Err("bad arguments".to_owned()); }
215 }
216 if name == "SPLIT" && !(10..=80).contains(&v[0]) { return Err("bad arguments".to_owned()); }
217 if name == "DELETE" && v.len() == 2 && v[1] < v[0] { return Err("bad arguments".to_owned()); }
218 let limit = count as Integer + Integer::from(name == "INSERT");
219 for (i, kind) in schema.bytes().enumerate() {
220 if kind == b'P' && v[i] > limit { return Err(format!("no cue {}", v[i])); }
221 }
222 Ok(a)
223}
224fn cue_line(position: usize, c: &Cue) -> String {
225 format!("{position} {} {} {} {}", c.index, time_text(c.start, 0), time_text(c.end, 0), c.text.join("|"))
226}
227impl Editor {
228 fn execute(&mut self, line: &str) -> Result<Vec<String>, String> {
229 let a = parse_arguments(line, self.cues.len())?;
230 let name = a.name.as_str();
231 let v = &a.values;
232 let n = self.cues.len();
233 if name == "UNDO" {
234 let previous = self.history.pop().ok_or("nothing to undo")?;
235 self.cues = previous.cues;
236 return Ok(vec![format!("UNDO command={}", previous.command)]);
237 }
238 if name == "SYNC" && self.cues[v[0] as usize-1].start == self.cues[v[2] as usize-1].start { return Err("sync points coincide".to_owned()); }
239 let modifying = ["SHIFT", "SHIFTFROM", "SCALE", "SYNC", "SNAP", "FIXOVERLAP", "EXTEND", "TIME", "SORT", "RENUMBER", "MERGE", "SPLIT", "INSERT", "DELETE", "TEXT", "BALANCE", "STRIP", "CLEANHI", "REPLACE"];
240 if modifying.contains(&name) { self.history.push(Record { command: name.to_owned(), cues: self.cues.clone() }); }
241 let mut out = Vec::new();
242 match name {
243 "LIST" | "AT" => {
244 let mut matches = 0;
245 for (i, c) in self.cues.iter().enumerate() {
246 if name == "LIST" || c.start <= v[0] && v[0] < c.end {
247 out.push(cue_line(i+1, c));
248 matches += 1;
249 }
250 }
251 if name == "AT" { out.push(format!("AT active={matches}")); }
252 }
253 "VALIDATE" => {
254 for (i, c) in self.cues.iter().enumerate() {
255 let prefix = format!("cue {}: ", i+1);
256 let d = c.end-c.start;
257 if c.index != (i+1) as Integer { out.push(format!("{prefix}index {} (expected {})", c.index, i+1)); }
258 if d <= 0 { out.push(format!("{prefix}end not after start")); }
259 if i > 0 && c.start < self.cues[i-1].end { out.push(format!("{prefix}overlaps previous by {} ms", self.cues[i-1].end-c.start)); }
260 if (1..700).contains(&d) { out.push(format!("{prefix}too short ({d} ms)")); }
261 if d > 7000 { out.push(format!("{prefix}too long ({d} ms)")); }
262 for (j, text) in c.text.iter().enumerate() {
263 if text.len() > 42 { out.push(format!("{prefix}line {} too long ({} chars)", j+1, text.len())); }
264 }
265 if c.text.len() >= 3 { out.push(format!("{prefix}too many lines ({})", c.text.len())); }
266 let chars = count_chars(c);
267 if d >= 1 && chars*1000 > 21*d { out.push(format!("{prefix}too fast ({} cps)", decimal(chars*100000/d))); }
268 }
269 out.push(format!("VALIDATE problems={}", out.len()));
270 }
271 "SEARCH" => {
272 for (i, c) in self.cues.iter().enumerate() {
273 if let Some(line) = c.text.iter().find(|s| s.contains(a.raw[0].as_str())) {
274 out.push(format!("{} {} {line}", i+1, time_text(c.start, 0)));
275 }
276 }
277 out.push(format!("SEARCH found={}", out.len()));
278 }
279 "GAPS" => {
280 for i in 0..n.saturating_sub(1) {
281 let gap = self.cues[i+1].start-self.cues[i].end;
282 if gap < v[0] { out.push(format!("{} {} {gap}", i+1, i+2)); }
283 }
284 out.push(format!("GAPS count={}", out.len()));
285 }
286 "STATS" => {
287 out.push(format!("STATS cues={n}"));
288 if n == 0 { return Ok(out); }
289 let (mut total, mut chars, mut longest) = (0, 0, 0);
290 let mut longest_position = 1;
291 for (i, c) in self.cues.iter().enumerate() {
292 let d = (c.end-c.start).max(0);
293 total += d;
294 if d > 0 { chars += count_chars(c); }
295 if d > longest {
296 longest = d;
297 longest_position = i+1;
298 }
299 }
300 out.push(format!("duration total={} mean={}", time_text(total, 0), time_text(total/n as Integer, 0)));
301 out.push(format!("speed {}", if total > 0 { decimal(chars*100000/total) } else { "-".to_owned() }));
302 out.push(format!("longest cue={longest_position} duration={}", time_text(longest, 0)));
303 if n == 1 { out.push("gap none".to_owned()); }
304 else {
305 let (mut position, mut gap) = (1, self.cues[1].start-self.cues[0].end);
306 for i in 1..n-1 {
307 let candidate = self.cues[i+1].start-self.cues[i].end;
308 if candidate > gap {
309 position = i+1;
310 gap = candidate;
311 }
312 }
313 out.push(format!("gap cue={position} ms={gap}"));
314 }
315 }
316 "PRINT" | "TOVTT" => {
317 if name == "TOVTT" { out.extend(["WEBVTT".to_owned(), String::new()]); }
318 for c in &self.cues {
319 let kind = usize::from(name == "TOVTT");
320 let (lines, code, _) = if kind == 1 { converted(&c.text, 1) } else { (c.text.clone(), 0, 0) };
321 if lines.is_empty() { continue; }
322 let mut setting = if code >= 7 { " line:0" } else if code >= 4 { " line:50%" } else { "" }.to_owned();
323 if code > 0 && code%3 == 1 { setting.push_str(" align:start"); }
324 if code > 0 && code%3 == 0 { setting.push_str(" align:end"); }
325 out.push(c.index.to_string());
326 out.push(format!("{} --> {}{setting}", time_text(c.start, kind), time_text(c.end, kind)));
327 out.extend(lines);
328 out.push(String::new());
329 }
330 }
331 "TOASS" => {
332 out.extend(["[Events]".to_owned(), "Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text".to_owned()]);
333 for c in &self.cues {
334 let (lines, _, _) = converted(&c.text, 2);
335 out.push(format!("Dialogue: 0,{},{},Default,,0,0,0,,{}", time_text(c.start, 2), time_text(c.end, 2), lines.join("\\N")));
336 }
337 }
338 "FRAMES" => {
339 for (i, c) in self.cues.iter().enumerate() {
340 out.push(format!("{} {} {}", i+1, c.start*v[0]/(1000*a.denominator), c.end*v[0]/(1000*a.denominator)));
341 }
342 }
343 "SHIFT" | "SHIFTFROM" | "SYNC" | "SCALE" | "SNAP" => {
344 let start = if name == "SHIFTFROM" { v[0] as usize-1 } else { 0 };
345 let shift = if name == "SHIFT" { v[0] } else if name == "SHIFTFROM" { v[1] } else { 0 };
346 let (s1, s2) = if name == "SYNC" { (self.cues[v[0] as usize-1].start, self.cues[v[2] as usize-1].start) } else { (0, 0) };
347 let (mut clamped, mut changed) = (0, 0);
348 for c in &mut self.cues[start..] {
349 let mut cue_clamped = false;
350 for t in [&mut c.start, &mut c.end] {
351 let old = *t;
352 let mut value = match name {
353 "SHIFT" | "SHIFTFROM" => old+shift,
354 "SCALE" => half_up(old*v[0], v[1]),
355 "SYNC" => half_up(v[1]*(s2-s1)+(old-s1)*(v[3]-v[1]), s2-s1),
356 _ => {
357 let frame = half_up(old*v[0], 1000*a.denominator);
358 half_up(frame*1000*a.denominator, v[0])
359 }
360 };
361 if value < 0 {
362 value = 0;
363 cue_clamped = true;
364 }
365 if old != value { changed += 1; }
366 *t = value;
367 }
368 if cue_clamped { clamped += 1; }
369 }
370 out.push(match name {
371 "SNAP" => format!("SNAP changed={changed}"),
372 "SCALE" => format!("SCALE cues={n}"),
373 _ => format!("{name} cues={} clamped={clamped}", n-start),
374 });
375 }
376 "FIXOVERLAP" => {
377 let (mut fixed, mut failed) = (0, 0);
378 for i in 0..n.saturating_sub(1) {
379 let candidate = self.cues[i+1].start-v[0];
380 if self.cues[i].end > candidate {
381 if candidate > self.cues[i].start {
382 self.cues[i].end = candidate;
383 fixed += 1;
384 }
385 else {
386 failed += 1;
387 out.push(format!("cue {}: cannot fix", i+1));
388 }
389 }
390 }
391 out.push(format!("FIXOVERLAP fixed={fixed} failed={failed}"));
392 }
393 "EXTEND" => {
394 let (mut extended, mut short) = (0, 0);
395 for i in 0..n {
396 if self.cues[i].end-self.cues[i].start < v[0] {
397 let mut candidate = self.cues[i].start+v[0];
398 if i+1 < n { candidate = candidate.min(self.cues[i+1].start); }
399 if candidate > self.cues[i].end {
400 self.cues[i].end = candidate;
401 extended += 1;
402 }
403 }
404 if self.cues[i].end-self.cues[i].start < v[0] { short += 1; }
405 }
406 out.push(format!("EXTEND extended={extended} short={short}"));
407 }
408 "TIME" => {
409 let c = &mut self.cues[v[0] as usize-1];
410 c.start = v[1];
411 c.end = v[2];
412 out.push(format!("TIME cue={}", v[0]));
413 }
414 "SORT" => {
415 let mut sorted: Vec<_> = std::mem::take(&mut self.cues).into_iter().enumerate().collect();
416 sorted.sort_by_key(|(_, c)| (c.start, c.end));
417 let moved = sorted.iter().enumerate().filter(|(i, (original, _))| i != original).count();
418 self.cues = sorted.into_iter().map(|(_, c)| c).collect();
419 out.push(format!("SORT moved={moved}"));
420 }
421 "RENUMBER" => {
422 let mut changed = 0;
423 for (i, c) in self.cues.iter_mut().enumerate() {
424 if c.index != (i+1) as Integer { changed += 1; }
425 c.index = (i+1) as Integer;
426 }
427 out.push(format!("RENUMBER changed={changed}"));
428 }
429 "MERGE" => {
430 let mut result: Vec<Cue> = Vec::new();
431 let mut merged = 0;
432 for c in std::mem::take(&mut self.cues) {
433 if let Some(previous) = result.last_mut() {
434 if c.start-previous.end < v[0] && !dialogue(previous) && !dialogue(&c) {
435 let mut lines = previous.text.clone();
436 lines.extend(c.text.iter().cloned());
437 let wrapped = wrap(&lines, 42);
438 if wrapped.len() <= 2 {
439 previous.end = previous.end.max(c.end);
440 previous.text = wrapped;
441 merged += 1;
442 continue;
443 }
444 }
445 }
446 result.push(c);
447 }
448 self.cues = result;
449 out.push(format!("MERGE merged={merged} cues={}", self.cues.len()));
450 }
451 "SPLIT" => {
452 let mut result = Vec::new();
453 let mut split = 0;
454 for mut c in std::mem::take(&mut self.cues) {
455 let d = c.end-c.start;
456 if d < 1 || dialogue(&c) {
457 result.push(c);
458 continue;
459 }
460 let wrapped = wrap(&c.text, v[0] as usize);
461 if wrapped.len() <= 2 {
462 c.text = wrapped;
463 result.push(c);
464 continue;
465 }
466 split += 1;
467 let mut parts: Vec<Cue> = wrapped.chunks(2).map(|lines| Cue { index: c.index, start: 0, end: 0, text: lines.to_vec() }).collect();
468 let total: Integer = parts.iter().map(count_chars).sum();
469 let mut lengths: Vec<Integer> = parts.iter().map(|p| d*count_chars(p)/total).collect();
470 let remainders: Vec<Integer> = parts.iter().map(|p| d*count_chars(p)%total).collect();
471 let remaining = d-lengths.iter().sum::<Integer>();
472 let mut order: Vec<usize> = (0..parts.len()).collect();
473 order.sort_by(|&i, &j| remainders[j].cmp(&remainders[i]));
474 for &i in &order[..remaining as usize] { lengths[i] += 1; }
475 let mut start = c.start;
476 for (i, part) in parts.iter_mut().enumerate() {
477 part.start = start;
478 part.end = start+lengths[i];
479 start += lengths[i];
480 }
481 result.extend(parts);
482 }
483 self.cues = result;
484 out.push(format!("SPLIT split={split} cues={}", self.cues.len()));
485 }
486 "INSERT" => {
487 self.cues.insert(v[0] as usize-1, Cue { index: v[0], start: v[1], end: v[2], text: a.raw[3].split('|').map(str::to_owned).collect() });
488 out.push(format!("INSERT cue={} cues={}", v[0], self.cues.len()));
489 }
490 "DELETE" => {
491 let p = v[0] as usize-1;
492 let q = if v.len() == 2 { v[1] } else { v[0] } as usize;
493 self.cues.drain(p..q);
494 out.push(format!("DELETE removed={} cues={}", q-p, self.cues.len()));
495 }
496 "TEXT" => {
497 let c = &mut self.cues[v[0] as usize-1];
498 c.text = a.raw[1].split('|').map(str::to_owned).collect();
499 out.push(format!("TEXT cue={} lines={}", v[0], c.text.len()));
500 }
501 "BALANCE" => {
502 let (mut changed, mut skipped, mut dialogues) = (0, 0, 0);
503 for c in &mut self.cues {
504 if dialogue(c) {
505 dialogues += 1;
506 continue;
507 }
508 let w = words(&c.text);
509 let all = w.join(" ");
510 let mut candidate = None;
511 if all.len() <= 42 { candidate = Some(vec![all]); }
512 else if all.len() <= 85 {
513 let mut best = (usize::MAX, usize::MAX);
514 for j in 1..w.len() {
515 let first = w[..j].join(" ");
516 let second = w[j..].join(" ");
517 let score = (first.len().abs_diff(second.len()), first.len());
518 if first.len() <= 42 && second.len() <= 42 && score < best {
519 best = score;
520 candidate = Some(vec![first, second]);
521 }
522 }
523 }
524 if let Some(lines) = candidate {
525 if lines != c.text { changed += 1; }
526 c.text = lines;
527 } else { skipped += 1; }
528 }
529 out.push(format!("BALANCE changed={changed} skipped={skipped} dialogue={dialogues}"));
530 }
531 "STRIP" | "CLEANHI" => {
532 let (mut removed_items, mut labels, mut removed_lines, mut removed_cues) = (0, 0, 0, 0);
533 let mut result = Vec::new();
534 for mut c in std::mem::take(&mut self.cues) {
535 let mut lines = Vec::new();
536 if name == "STRIP" {
537 let (converted, _, tags) = converted(&c.text, 0);
538 removed_items += tags;
539 removed_lines += c.text.len()-converted.len();
540 lines = converted;
541 } else {
542 for line in &c.text {
543 let (text, segments, names) = clean_hi(line);
544 removed_items += segments;
545 labels += names;
546 if text.is_empty() || text == "-" { removed_lines += 1; }
547 else { lines.push(text); }
548 }
549 }
550 if lines.is_empty() { removed_cues += 1; }
551 else {
552 c.text = lines;
553 result.push(c);
554 }
555 }
556 self.cues = result;
557 if name == "STRIP" { out.push(format!("STRIP tags={removed_items} removed_lines={removed_lines} removed_cues={removed_cues}")); }
558 else { out.push(format!("CLEANHI segments={removed_items} labels={labels} removed_lines={removed_lines} removed_cues={removed_cues}")); }
559 }
560 "REPLACE" => {
561 let (old, replacement) = a.raw[0].split_once('|').unwrap();
562 let (mut count, mut cues) = (0, 0);
563 for c in &mut self.cues {
564 let mut matches = 0;
565 for line in &mut c.text {
566 matches += line.matches(old).count();
567 *line = line.replace(old, replacement);
568 }
569 count += matches;
570 if matches > 0 { cues += 1; }
571 }
572 out.push(format!("REPLACE count={count} cues={cues}"));
573 }
574 _ => unreachable!(),
575 }
576 Ok(out)
577 }
578}
579fn load_block(block: &[String], position: usize, editor: &mut Editor) -> Option<String> {
580 let index = unsigned(&block[0]).filter(|_| block[0].len() <= 6);
581 if index.is_none() { return Some(format!("BAD CUE {position}: index")); }
582 let times = block.get(1).filter(|s| s.len() == 29 && &s[12..17] == " --> ")
583 .and_then(|s| Some((time_value(&s[..12])?, time_value(&s[17..])?)));
584 let Some((start, end)) = times else { return Some(format!("BAD CUE {position}: time")); };
585 if block.len() < 3 { return Some(format!("BAD CUE {position}: no text")); }
586 editor.cues.push(Cue { index: index.unwrap(), start, end, text: block[2..].to_vec() });
587 None
588}
589fn main() {
590 let stdin = io::stdin();
591 let mut lines = stdin.lock().lines();
592 let stdout = io::stdout();
593 let mut output = io::BufWriter::new(stdout.lock());
594 let mut editor = Editor::default();
595 let mut block = Vec::new();
596 let (mut blocks, mut bad) = (0, 0);
597 for line in lines.by_ref() {
598 let line = line.unwrap();
599 if line.is_empty() || line == "###" {
600 if !block.is_empty() {
601 blocks += 1;
602 if let Some(message) = load_block(&block, blocks, &mut editor) {
603 writeln!(output, "{message}").unwrap();
604 bad += 1;
605 }
606 block.clear();
607 }
608 if line == "###" { break; }
609 } else { block.push(line); }
610 }
611 writeln!(output, "LOAD cues={} bad={bad}", editor.cues.len()).unwrap();
612 let count: usize = lines.next().unwrap().unwrap().parse().unwrap();
613 for line in lines.take(count) {
614 let line = line.unwrap();
615 let result = editor.execute(&line).unwrap_or_else(|message| vec![format!("ERROR {message}")]);
616 for line in result { writeln!(output, "{line}").unwrap(); }
617 }
618}
