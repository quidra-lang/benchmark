40static void *allocate(size_t size) {

42 if (!p) exit(EXIT_FAILURE);

44}
45static char *slice(const char *s, size_t n) {
46 char *copy = allocate(n + 1);
47 memcpy(copy, s, n);
48 copy[n] = '\0';
49 return copy;
50}
51static char *copy_string(const char *s) { return slice(s, strlen(s)); }
52static char *format(const char *pattern, ...) {
53 va_list args, copy;
54 va_start(args, pattern);
55 va_copy(copy, args);
56 int length = vsnprintf(NULL, 0, pattern, copy);
57 va_end(copy);
58 char *s = allocate((size_t)length + 1);
59 vsnprintf(s, (size_t)length + 1, pattern, args);
60 va_end(args);
61 return s;
62}
63static void push_owned(Strings *s, char *item) {
64 if (s->count == s->capacity) {
65 s->capacity = s->capacity ? s->capacity * 2 : 8;
66 s->items = realloc(s->items, (size_t)s->capacity * sizeof *s->items);
67 if (!s->items) exit(EXIT_FAILURE);
68 }
69 s->items[s->count++] = item;
70}
71static void push(Strings *s, const char *item) { push_owned(s, copy_string(item)); }
72static void free_strings(Strings *s) {
73 for (int i = 0; i < s->count; ++i) free(s->items[i]);
74 free(s->items);
75 *s = (Strings){0};
76}
77static Strings clone_strings(const Strings *s) {
78 Strings result = {0};
79 for (int i = 0; i < s->count; ++i) push(&result, s->items[i]);
80 return result;
81}
82static Strings split(const char *s, char delimiter, int limit) {
83 Strings result = {0};
84 const char *start = s;
85 for (const char *p = s;; ++p) {
86 if (!*p || (*p == delimiter && (!limit || result.count + 1 < limit))) {
87 push_owned(&result, slice(start, (size_t)(p - start)));
88 start = p + 1;
89 if (!*p) break;
90 }
91 }
92 return result;
93}
94static char *join(const Strings *s, const char *separator) {
95 size_t size = 1, sep = strlen(separator);
96 for (int i = 0; i < s->count; ++i) size += strlen(s->items[i]) + (i ? sep : 0);
97 char *result = allocate(size), *p = result;
98 for (int i = 0; i < s->count; ++i) {
99 if (i) {
100 memcpy(p, separator, sep);
101 p += sep;
102 }
103 size_t n = strlen(s->items[i]);
104 memcpy(p, s->items[i], n);
105 p += n;
106 }
107 *p = '\0';
108 return result;
109}
110static void append_char(Strings *parts, char c) {
111 char s[] = {c, '\0'};

114static char *trim(const char *s) {
115 while (*s == ' ') ++s;
116 size_t length = strlen(s);
117 while (length && s[length - 1] == ' ') --length;
118 return slice(s, length);
119}

147static bool unsigned_value(const char *s, Integer *value) {
148 size_t n = strlen(s);
149 if (!n || n > 9) return false;
150 *value = 0;
151 for (size_t i = 0; i < n; ++i) {
152 if (s[i] < '0' || s[i] > '9') return false;
153 *value = *value * 10 + s[i] - '0';
154 }
155 return true;
156}
157static bool time_value(const char *s, Integer *value) {
158 if (strlen(s) != 12 || s[2] != ':' || s[5] != ':' || s[8] != ',') return false;
159 for (int i = 0; i < 12; ++i) {
160 if (i != 2 && i != 5 && i != 8 && (s[i] < '0' || s[i] > '9')) return false;
161 }
162 int h = (s[0] - '0') * 10 + s[1] - '0';
163 int m = (s[3] - '0') * 10 + s[4] - '0';
164 int sec = (s[6] - '0') * 10 + s[7] - '0';
165 int ms = (s[9] - '0') * 100 + (s[10] - '0') * 10 + s[11] - '0';
166 *value = (Integer)h * 3600000 + m * 60000 + sec * 1000 + ms;
167 return m <= 59 && sec <= 59;
168}
169static char *time_text(Integer t, int kind) {
170 if (kind == 2) {
171 Integer c = (t + 5) / 10;
172 return format("%" PRId64 ":%02" PRId64 ":%02" PRId64 ".%02" PRId64, c / 360000, c / 6000 % 60, c / 100 % 60, c % 100);
173 }
174 return format("%02" PRId64 ":%02" PRId64 ":%02" PRId64 "%c%03" PRId64, t / 3600000, t / 60000 % 60, t / 1000 % 60, kind == 1 ? '.' : ',', t % 1000);
175}
176static char *decimal(Integer v) { return format("%" PRId64 ".%02" PRId64, v / 100, v % 100); }
177static Integer floor_ratio(Integer a, Integer b) {
178 if (b < 0) {
179 a = -a;
180 b = -b;
181 }
182 Integer q = a / b;
183 if (a % b < 0) --q;
184 return q;
185}
186static Integer half_up(Integer a, Integer b) {
187 if (b < 0) {
188 a = -a;
189 b = -b;
190 }
191 return floor_ratio(2 * a + b, 2 * b);
192}

202static Strings words(const Strings *lines) {
203 Strings result = {0};
204 for (int i = 0; i < lines->count; ++i) {
205 Strings parts = split(lines->items[i], ' ', 0);
206 for (int j = 0; j < parts.count; ++j) if (*parts.items[j]) push(&result, parts.items[j]);
207 free_strings(&parts);
208 }
209 return result;
210}
211static Strings wrap(const Strings *lines, int width) {
212 Strings w = words(lines), result = {0};
213 for (int i = 0; i < w.count; ++i) {
214 if (!result.count || strlen(result.items[result.count - 1]) + 1 + strlen(w.items[i]) > (size_t)width) push(&result, w.items[i]);
215 else {
216 char **last = &result.items[result.count - 1];
217 char *replacement = format("%s %s", *last, w.items[i]);
218 free(*last);
219 *last = replacement;
220 }
221 }
222 free_strings(&w);
223 return result;
224}
225static bool same_text(const Strings *a, const Strings *b) {
226 if (a->count != b->count) return false;
227 for (int i = 0; i < a->count; ++i) if (strcmp(a->items[i], b->items[i])) return false;
228 return true;
229}
230static Markup markup_at(const char *line) {
231 const char *tags[] = {"<i>", "</i>", "<b>", "</b>", "<u>", "</u>"};
232 for (int i = 0; i < 6; ++i) {
233 if (!strncmp(line, tags[i], strlen(tags[i]))) return (Markup){1, (int)strlen(tags[i]), 0, i};
234 }
235 if (!strncmp(line, "</font>", 7)) return (Markup){2, 7, 0, 0};
236 if (!strncmp(line, "<font", 5) && (line[5] == ' ' || line[5] == '>')) {
237 const char *end = strchr(line + 5, '>');
238 if (end) return (Markup){2, (int)(end - line + 1), 0, 0};
239 }
240 if (strlen(line) >= 6 && !strncmp(line, "{\\an", 4) && line[4] >= '1' && line[4] <= '9' && line[5] == '}') return (Markup){3, 6, line[4] - '0', 0};
241 return (Markup){0, 1, 0, 0};
242}
243static Strings converted(const Strings *lines, int mode, int *first, int *removed) {
244 const char *ass[] = {"{\\i1}", "{\\i0}", "{\\b1}", "{\\b0}", "{\\u1}", "{\\u0}"};
245 Strings result = {0};
246 *first = *removed = 0;
247 for (int i = 0; i < lines->count; ++i) {
248 Strings pieces = {0};
249 for (const char *p = lines->items[i]; *p;) {
250 Markup m = markup_at(p);
251 if (m.kind == 3 && !*first) *first = m.digit;
252 if (mode == 0) {
253 if (!m.kind) append_char(&pieces, *p);
254 else ++*removed;
255 } else if (mode == 1) {
256 if (m.kind == 1) push_owned(&pieces, slice(p, m.size));
257 if (!m.kind) {
258 if (*p == '&') push(&pieces, "&amp;");
259 else if (*p == '<') push(&pieces, "&lt;");
260 else if (*p == '>') push(&pieces, "&gt;");
261 else append_char(&pieces, *p);
262 }
263 } else {
264 if (m.kind == 1) push(&pieces, ass[m.tag]);
265 if (m.kind == 0 || m.kind == 3) push_owned(&pieces, slice(p, m.size));
266 }
267 p += m.size;
268 }
269 char *raw = join(&pieces, "");
270 char *text = mode == 2 ? raw : trim(raw);
271 if (mode != 2) free(raw);
272 if (mode == 2 || *text) push_owned(&result, text);
273 else free(text);
274 free_strings(&pieces);
275 }
276 return result;
277}
278static char *normalize(const char *s) {
279 Strings lines = {0};
280 push(&lines, s);
281 Strings w = words(&lines);
282 char *result = join(&w, " ");
283 free_strings(&lines);
284 free_strings(&w);
285 return result;
286}
287static char *clean_hi(const char *line, int *segments, int *labels) {
288 Strings parts = {0};
289 *segments = *labels = 0;
290 for (const char *p = line; *p;) {
291 const char *end = *p == '[' ? strchr(p + 1, ']') : *p == '(' ? strchr(p + 1, ')') : NULL;
292 if (end) {
293 ++*segments;
294 p = end + 1;
295 }
296 else append_char(&parts, *p++);
297 }
298 char *joined = join(&parts, ""), *text = normalize(joined);
299 free(joined);
300 free_strings(&parts);
301 const char *prefix = "", *body = text;
302 if (!strncmp(body, "- ", 2)) {
303 prefix = "- ";
304 body += 2;
305 }
306 const char *colon = strchr(body, ':');
307 if (colon && colon > body) {
308 bool valid = *body >= 'A' && *body <= 'Z';
309 for (const char *p = body; p < colon; ++p) {
310 if (!((*p >= 'A' && *p <= 'Z') || (*p >= '0' && *p <= '9') || *p == ' ')) valid = false;
311 }
312 if (valid) {
313 *labels = 1;
314 char *raw = format("%s%s", prefix, colon + 1), *replacement = normalize(raw);
315 free(raw);
316 free(text);
317 text = replacement;
318 }
319 }
320 return text;
321}
322static int parse_arguments(const char *line) {
323 const char *names[] = {"LIST", "VALIDATE", "SEARCH", "AT", "GAPS", "STATS", "PRINT", "TOVTT", "TOASS", "FRAMES", "SHIFT", "SHIFTFROM", "SCALE", "SYNC", "SNAP", "FIXOVERLAP", "EXTEND", "TIME", "SORT", "RENUMBER", "MERGE", "SPLIT", "INSERT", "DELETE", "TEXT", "BALANCE", "STRIP", "CLEANHI", "REPLACE", "UNDO"};
324 const char *schemas[] = {"", "", "S", "T", "U", "", "", "", "", "F", "I", "PI", "UU", "PTPT", "F", "U", "U", "PTT", "", "", "U", "U", "PTTX", "P", "PX", "", "", "", "R", ""};
325 const char *space = strchr(line, ' ');
326 size_t length = space ? (size_t)(space - line) : strlen(line);
327 int cmd = -1;
328 for (int i = 0; i < 30; ++i) if (strlen(names[i]) == length && !strncmp(line, names[i], length)) cmd = i;
329 if (cmd < 0) fail("unknown command");
330 Arguments *a = &arguments;
331 strcpy(a->name, names[cmd]);
332 strcpy(a->schema, schemas[cmd]);
333 a->denominator = 1;
334 if (space) a->raw = split(space + 1, ' ', strpbrk(a->schema, "SXR") ? (int)strlen(a->schema) : 0);
335 if (cmd == 23 && a->raw.count == 2) strcpy(a->schema, "PP");
336 if (a->raw.count != (int)strlen(a->schema)) fail("bad arguments");
337 for (int i = 0; i < a->raw.count; ++i) {
338 const char *s = a->raw.items[i];
339 char kind = a->schema[i];
340 Integer value = 0;
341 bool valid = true;
342 if (kind == 'P' || kind == 'U') {
343 valid = unsigned_value(s, &value);
344 if (kind == 'P' && value == 0) valid = false;
345 } else if (kind == 'I') {
346 bool negative = *s == '-';
347 valid = unsigned_value(s + (negative ? 1 : 0), &value);
348 if (negative) value = -value;
349 } else if (kind == 'T') valid = time_value(s, &value);
350 else if (kind == 'F') {
351 Strings parts = split(s, '/', 0);
352 if (parts.count > 2) valid = false;
353 else {
354 valid = unsigned_value(parts.items[0], &value);
355 if (parts.count == 2) valid = unsigned_value(parts.items[1], &a->denominator) && valid;
356 valid = valid && value >= 1 && value <= 100000 && a->denominator >= 1 && a->denominator <= 100000;
357 }
358 free_strings(&parts);
359 } else if (kind == 'X') {
360 Strings parts = split(s, '|', 0);
361 for (int j = 0; j < parts.count; ++j) if (!*parts.items[j]) valid = false;
362 free_strings(&parts);
363 } else if (kind == 'R') {
364 Strings parts = split(s, '|', 0);
365 valid = parts.count == 2 && *parts.items[0] && *parts.items[1];
366 free_strings(&parts);
367 } else if (kind == 'S') valid = *s != '\0';
368 if (!valid) fail("bad arguments");
369 a->values[i] = value;
370 }
371 Integer *v = a->values;
372 if (cmd == 12 && (v[0] < 1 || v[0] > 100000 || v[1] < 1 || v[1] > 100000)) fail("bad arguments");
373 if (cmd == 4 || cmd == 15 || cmd == 16 || cmd == 20) {
374 if (v[0] < (cmd == 16 ? 1 : 0) || v[0] > 60000) fail("bad arguments");
375 }
376 if (cmd == 21 && (v[0] < 10 || v[0] > 80)) fail("bad arguments");
377 if (cmd == 23 && a->raw.count == 2 && v[1] < v[0]) fail("bad arguments");
378 Integer limit = cues.count + (cmd == 22 ? 1 : 0);
379 for (int i = 0; i < a->raw.count; ++i) {
380 if (a->schema[i] == 'P' && v[i] > limit) {
381 char message[64];
382 snprintf(message, sizeof message, "no cue %" PRId64, v[i]);
383 fail(message);
384 }
385 }
386 return cmd;
387}
388static char *cue_line(int position, const Cue *c) {
389 char *start = time_text(c->start, 0), *end = time_text(c->end, 0), *joined = join(&c->text, "|");
390 char *result = format("%d %" PRId64 " %s %s %s", position, c->index, start, end, joined);
391 free(start);
392 free(end);
393 free(joined);
394 return result;
395}
396static int compare_order(const void *left, const void *right) {
397 int i = *(const int *)left, j = *(const int *)right;
398 const Cue *a = &cues.items[i], *b = &cues.items[j];
399 if (a->start != b->start) return a->start < b->start ? -1 : 1;
400 if (a->end != b->end) return a->end < b->end ? -1 : 1;
401 return i < j ? -1 : i > j;
402}
403static void execute(const char *line, Strings *out) {
404 int cmd = parse_arguments(line), n = cues.count;
405 Arguments *a = &arguments;
406 Integer *v = a->values;
407 if (cmd == 29) {
408 if (!history_count) fail("nothing to undo");
409 Record *previous = &history[--history_count];
410 free_cues(&cues);
411 cues = previous->cues;
412 previous->cues = (Cues){0};
413 push_owned(out, format("UNDO command=%s", previous->name));
414 return;
415 }
416 if (cmd == 13 && cues.items[v[0] - 1].start == cues.items[v[2] - 1].start) fail("sync points coincide");
417 if (cmd >= 10 && cmd <= 28) {
418 Record *r = &history[history_count++];
419 strcpy(r->name, a->name);
420 r->cues = clone_cues(&cues);
421 }
422 if (cmd == 0 || cmd == 3) {
423 int matches = 0;
424 for (int i = 0; i < n; ++i) {
425 const Cue *c = &cues.items[i];
426 if (cmd == 0 || (c->start <= v[0] && v[0] < c->end)) {
427 push_owned(out, cue_line(i + 1, c));
428 ++matches;
429 }
430 }
431 if (cmd == 3) push_owned(out, format("AT active=%d", matches));
432 } else if (cmd == 1) {
433 for (int i = 0; i < n; ++i) {
434 const Cue *c = &cues.items[i];
435 Integer d = c->end - c->start;
436 if (c->index != i + 1) push_owned(out, format("cue %d: index %" PRId64 " (expected %d)", i + 1, c->index, i + 1));
437 if (d <= 0) push_owned(out, format("cue %d: end not after start", i + 1));
438 if (i && c->start < cues.items[i - 1].end) push_owned(out, format("cue %d: overlaps previous by %" PRId64 " ms", i + 1, cues.items[i - 1].end - c->start));
439 if (d >= 1 && d < 700) push_owned(out, format("cue %d: too short (%" PRId64 " ms)", i + 1, d));
440 if (d > 7000) push_owned(out, format("cue %d: too long (%" PRId64 " ms)", i + 1, d));
441 for (int j = 0; j < c->text.count; ++j) {
442 size_t length = strlen(c->text.items[j]);
443 if (length > 42) push_owned(out, format("cue %d: line %d too long (%zu chars)", i + 1, j + 1, length));
444 }
445 if (c->text.count >= 3) push_owned(out, format("cue %d: too many lines (%d)", i + 1, c->text.count));
446 Integer chars = count_chars(c);
447 if (d >= 1 && chars * 1000 > 21 * d) {
448 char *speed = decimal(chars * 100000 / d);
449 push_owned(out, format("cue %d: too fast (%s cps)", i + 1, speed));
450 free(speed);
451 }
452 }
453 push_owned(out, format("VALIDATE problems=%d", out->count));
454 } else if (cmd == 2) {
455 for (int i = 0; i < n; ++i) {
456 const Cue *c = &cues.items[i];
457 for (int j = 0; j < c->text.count; ++j) {
458 if (strstr(c->text.items[j], a->raw.items[0])) {
459 char *start = time_text(c->start, 0);
460 push_owned(out, format("%d %s %s", i + 1, start, c->text.items[j]));
461 free(start);
462 break;
463 }
464 }
465 }
466 push_owned(out, format("SEARCH found=%d", out->count));
467 } else if (cmd == 4) {
468 for (int i = 0; i + 1 < n; ++i) {
469 Integer gap = cues.items[i + 1].start - cues.items[i].end;
470 if (gap < v[0]) push_owned(out, format("%d %d %" PRId64, i + 1, i + 2, gap));
471 }
472 push_owned(out, format("GAPS count=%d", out->count));
473 } else if (cmd == 5) {
474 push_owned(out, format("STATS cues=%d", n));
475 if (!n) return;
476 Integer total = 0, chars = 0, longest = 0;
477 int position = 1;
478 for (int i = 0; i < n; ++i) {
479 Integer d = cues.items[i].end - cues.items[i].start;
480 if (d < 0) d = 0;
481 total += d;
482 if (d) chars += count_chars(&cues.items[i]);
483 if (d > longest) {
484 longest = d;
485 position = i + 1;
486 }
487 }
488 char *total_text = time_text(total, 0), *mean = time_text(total / n, 0);
489 push_owned(out, format("duration total=%s mean=%s", total_text, mean));
490 free(total_text);
491 free(mean);
492 char *speed = total ? decimal(chars * 100000 / total) : copy_string("-");
493 push_owned(out, format("speed %s", speed));
494 free(speed);
495 char *long_text = time_text(longest, 0);
496 push_owned(out, format("longest cue=%d duration=%s", position, long_text));
497 free(long_text);
498 if (n == 1) push(out, "gap none");
499 else {
500 position = 1;
501 Integer gap = cues.items[1].start - cues.items[0].end;
502 for (int i = 1; i + 1 < n; ++i) {
503 Integer candidate = cues.items[i + 1].start - cues.items[i].end;
504 if (candidate > gap) {
505 gap = candidate;
506 position = i + 1;
507 }
508 }
509 push_owned(out, format("gap cue=%d ms=%" PRId64, position, gap));
510 }
511 } else if (cmd == 6 || cmd == 7) {
512 if (cmd == 7) {
513 push(out, "WEBVTT");
514 push(out, "");
515 }
516 for (int i = 0; i < n; ++i) {
517 const Cue *c = &cues.items[i];
518 int code = 0, removed = 0;
519 Strings lines = cmd == 7 ? converted(&c->text, 1, &code, &removed) : c->text;
520 if (!lines.count) {
521 if (cmd == 7) free_strings(&lines);
522 continue;
523 }
524 char setting[64] = "";
525 if (code >= 7) strcat(setting, " line:0");
526 else if (code >= 4) strcat(setting, " line:50%");
527 if (code && code % 3 == 1) strcat(setting, " align:start");
528 if (code && code % 3 == 0) strcat(setting, " align:end");
529 char *start = time_text(c->start, cmd == 7), *end = time_text(c->end, cmd == 7);
530 push_owned(out, format("%" PRId64, c->index));
531 push_owned(out, format("%s --> %s%s", start, end, setting));
532 free(start);
533 free(end);
534 for (int j = 0; j < lines.count; ++j) push(out, lines.items[j]);
535 push(out, "");
536 if (cmd == 7) free_strings(&lines);
537 }
538 } else if (cmd == 8) {
539 push(out, "[Events]");
540 push(out, "Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text");
541 for (int i = 0; i < n; ++i) {
542 Cue *c = &cues.items[i];
543 int first, removed;
544 Strings lines = converted(&c->text, 2, &first, &removed);
545 char *joined = join(&lines, "\\N"), *start = time_text(c->start, 2), *end = time_text(c->end, 2);
546 push_owned(out, format("Dialogue: 0,%s,%s,Default,,0,0,0,,%s", start, end, joined));
547 free(joined);
548 free(start);
549 free(end);
550 free_strings(&lines);
551 }
552 } else if (cmd == 9) {
553 for (int i = 0; i < n; ++i) push_owned(out, format("%d %" PRId64 " %" PRId64, i + 1, cues.items[i].start * v[0] / (1000 * a->denominator), cues.items[i].end * v[0] / (1000 * a->denominator)));
554 } else if (cmd >= 10 && cmd <= 14) {
555 int start = cmd == 11 ? (int)v[0] - 1 : 0, clamped = 0, changed = 0;
556 Integer shift = cmd == 10 ? v[0] : cmd == 11 ? v[1] : 0;
557 Integer s1 = 0, s2 = 0;
558 if (cmd == 13) {
559 s1 = cues.items[v[0] - 1].start;
560 s2 = cues.items[v[2] - 1].start;
561 }
562 for (int i = start; i < n; ++i) {
563 bool cue_clamped = false;
564 Integer *times[] = {&cues.items[i].start, &cues.items[i].end};
565 for (int j = 0; j < 2; ++j) {
566 Integer old = *times[j], value = old;
567 if (cmd == 10 || cmd == 11) value += shift;
568 if (cmd == 12) value = half_up(value * v[0], v[1]);
569 if (cmd == 13) value = half_up(v[1] * (s2 - s1) + (value - s1) * (v[3] - v[1]), s2 - s1);
570 if (cmd == 14) {
571 Integer frame = half_up(value * v[0], 1000 * a->denominator);
572 value = half_up(frame * 1000 * a->denominator, v[0]);
573 }
574 if (value < 0) {
575 value = 0;
576 cue_clamped = true;
577 }
578 if (value != old) ++changed;
579 *times[j] = value;
580 }
581 if (cue_clamped) ++clamped;
582 }
583 if (cmd == 14) push_owned(out, format("SNAP changed=%d", changed));
584 else if (cmd == 12) push_owned(out, format("SCALE cues=%d", n));
585 else push_owned(out, format("%s cues=%d clamped=%d", a->name, n - start, clamped));
586 } else if (cmd == 15) {
587 int fixed = 0, failed = 0;
588 for (int i = 0; i + 1 < n; ++i) {
589 Integer candidate = cues.items[i + 1].start - v[0];
590 if (cues.items[i].end > candidate) {
591 if (candidate > cues.items[i].start) {
592 cues.items[i].end = candidate;
593 ++fixed;
594 } else {
595 ++failed;
596 push_owned(out, format("cue %d: cannot fix", i + 1));
597 }
598 }
599 }
600 push_owned(out, format("FIXOVERLAP fixed=%d failed=%d", fixed, failed));
601 } else if (cmd == 16) {
602 int extended = 0, short_count = 0;
603 for (int i = 0; i < n; ++i) {
604 Cue *c = &cues.items[i];
605 if (c->end - c->start < v[0]) {
606 Integer candidate = c->start + v[0];
607 if (i + 1 < n && cues.items[i + 1].start < candidate) candidate = cues.items[i + 1].start;
608 if (candidate > c->end) {
609 c->end = candidate;
610 ++extended;
611 }
612 }
613 if (c->end - c->start < v[0]) ++short_count;
614 }
615 push_owned(out, format("EXTEND extended=%d short=%d", extended, short_count));
616 } else if (cmd == 17) {
617 cues.items[v[0] - 1].start = v[1];
618 cues.items[v[0] - 1].end = v[2];
619 push_owned(out, format("TIME cue=%" PRId64, v[0]));
620 } else if (cmd == 18) {
621 int *order = allocate((size_t)n * sizeof *order), moved = 0;
622 for (int i = 0; i < n; ++i) order[i] = i;
623 qsort(order, n, sizeof *order, compare_order);
624 Cues sorted = {0};
625 for (int i = 0; i < n; ++i) {
626 push_cue(&sorted, cues.items[order[i]]);
627 if (i != order[i]) ++moved;
628 }
629 free(order);
630 free(cues.items);
631 cues = sorted;
632 push_owned(out, format("SORT moved=%d", moved));
633 } else if (cmd == 19) {
634 int changed = 0;
635 for (int i = 0; i < n; ++i) {
636 if (cues.items[i].index != i + 1) ++changed;
637 cues.items[i].index = i + 1;
638 }
639 push_owned(out, format("RENUMBER changed=%d", changed));
640 } else if (cmd == 20) {
641 Cues result = {0};
642 int merged = 0;
643 for (int i = 0; i < n; ++i) {
644 Cue c = cues.items[i];
645 if (result.count) {
646 Cue *previous = &result.items[result.count - 1];
647 if (c.start - previous->end < v[0] && !dialogue(previous) && !dialogue(&c)) {
648 Strings lines = clone_strings(&previous->text);
649 for (int j = 0; j < c.text.count; ++j) push(&lines, c.text.items[j]);
650 Strings wrapped = wrap(&lines, 42);
651 free_strings(&lines);
652 if (wrapped.count <= 2) {
653 if (c.end > previous->end) previous->end = c.end;
654 free_strings(&previous->text);
655 previous->text = wrapped;
656 free_strings(&c.text);
657 ++merged;
658 continue;
659 }
660 free_strings(&wrapped);
661 }
662 }
663 push_cue(&result, c);
664 }
665 free(cues.items);
666 cues = result;
667 push_owned(out, format("MERGE merged=%d cues=%d", merged, cues.count));
668 } else if (cmd == 21) {
669 Cues result = {0};
670 int split_count = 0;
671 for (int i = 0; i < n; ++i) {
672 Cue c = cues.items[i];
673 Integer d = c.end - c.start;
674 if (d < 1 || dialogue(&c)) {
675 push_cue(&result, c);
676 continue;
677 }
678 Strings wrapped = wrap(&c.text, (int)v[0]);
679 free_strings(&c.text);
680 if (wrapped.count <= 2) {
681 c.text = wrapped;
682 push_cue(&result, c);
683 continue;
684 }
685 ++split_count;
686 int count = (wrapped.count + 1) / 2;
687 Integer *lengths = allocate((size_t)count * sizeof *lengths);
688 Integer *remainders = allocate((size_t)count * sizeof *remainders);
689 int *order = allocate((size_t)count * sizeof *order);
690 Cues parts = {0};
691 Integer total = 0;
692 for (int j = 0; j < wrapped.count; j += 2) {
693 Cue part = {c.index, 0, 0, {0}};
694 push(&part.text, wrapped.items[j]);
695 if (j + 1 < wrapped.count) push(&part.text, wrapped.items[j + 1]);
696 total += count_chars(&part);
697 push_cue(&parts, part);
698 }
699 free_strings(&wrapped);
700 Integer remaining = d;
701 for (int j = 0; j < count; ++j) {
702 Integer product = d * count_chars(&parts.items[j]);
703 lengths[j] = product / total;
704 remainders[j] = product % total;
705 remaining -= lengths[j];
706 order[j] = j;
707 }
708 for (int j = 1; j < count; ++j) {
709 int part = order[j], k = j;
710 while (k > 0 && remainders[order[k - 1]] < remainders[part]) {
711 order[k] = order[k - 1];
712 --k;
713 }
714 order[k] = part;
715 }
716 for (int j = 0; j < remaining; ++j) ++lengths[order[j]];
717 Integer start = c.start;
718 for (int j = 0; j < count; ++j) {
719 parts.items[j].start = start;
720 parts.items[j].end = start + lengths[j];
721 start += lengths[j];
722 push_cue(&result, parts.items[j]);
723 }
724 free(parts.items);
725 free(lengths);
726 free(remainders);
727 free(order);
728 }
729 free(cues.items);
730 cues = result;
731 push_owned(out, format("SPLIT split=%d cues=%d", split_count, cues.count));
732 } else if (cmd == 22) {
733 Cue c = {v[0], v[1], v[2], split(a->raw.items[3], '|', 0)};
734 int p = (int)v[0] - 1;
735 push_cue(&cues, c);
736 memmove(cues.items + p + 1, cues.items + p, (size_t)(n - p) * sizeof c);
737 cues.items[p] = c;
738 push_owned(out, format("INSERT cue=%" PRId64 " cues=%d", v[0], cues.count));
739 } else if (cmd == 23) {
740 int p = (int)v[0] - 1, q = (int)(a->raw.count == 2 ? v[1] : v[0]);
741 for (int i = p; i < q; ++i) free_strings(&cues.items[i].text);
742 memmove(cues.items + p, cues.items + q, (size_t)(n - q) * sizeof *cues.items);
743 cues.count -= q - p;
744 push_owned(out, format("DELETE removed=%d cues=%d", q - p, cues.count));
745 } else if (cmd == 24) {
746 Cue *c = &cues.items[v[0] - 1];
747 free_strings(&c->text);
748 c->text = split(a->raw.items[1], '|', 0);
749 push_owned(out, format("TEXT cue=%" PRId64 " lines=%d", v[0], c->text.count));
750 } else if (cmd == 25) {
751 int changed = 0, skipped = 0, dialogues = 0;
752 for (int i = 0; i < n; ++i) {
753 Cue *c = &cues.items[i];
754 if (dialogue(c)) {
755 ++dialogues;
756 continue;
757 }
758 Strings w = words(&c->text), candidate = {0};
759 char *all = join(&w, " ");
760 if (strlen(all) <= 42) push(&candidate, all);
761 else if (strlen(all) <= 85) {
762 size_t best_difference = SIZE_MAX, best_first = SIZE_MAX;
763 for (int j = 1; j < w.count; ++j) {
764 Strings left = {w.items, j, j}, right = {w.items + j, w.count - j, w.count - j};
765 char *first = join(&left, " "), *second = join(&right, " ");
766 size_t a_length = strlen(first), b_length = strlen(second);
767 size_t difference = a_length > b_length ? a_length - b_length : b_length - a_length;
768 if (a_length <= 42 && b_length <= 42 && (difference < best_difference || (difference == best_difference && a_length < best_first))) {
769 free_strings(&candidate);
770 push(&candidate, first);
771 push(&candidate, second);
772 best_difference = difference;
773 best_first = a_length;
774 }
775 free(first);
776 free(second);
777 }
778 }
779 if (!candidate.count) ++skipped;
780 else {
781 if (!same_text(&candidate, &c->text)) ++changed;
782 free_strings(&c->text);
783 c->text = candidate;
784 }
785 free(all);
786 free_strings(&w);
787 }
788 push_owned(out, format("BALANCE changed=%d skipped=%d dialogue=%d", changed, skipped, dialogues));
789 } else if (cmd == 26 || cmd == 27) {
790 int removed_items = 0, labels = 0, removed_lines = 0, removed_cues = 0;
791 Cues result = {0};
792 for (int i = 0; i < n; ++i) {
793 Cue c = cues.items[i];
794 Strings lines = {0};
795 if (cmd == 26) {
796 int first, tags;
797 lines = converted(&c.text, 0, &first, &tags);
798 removed_items += tags;
799 removed_lines += c.text.count - lines.count;
800 } else {
801 for (int j = 0; j < c.text.count; ++j) {
802 int segments, names;
803 char *cleaned = clean_hi(c.text.items[j], &segments, &names);
804 removed_items += segments;
805 labels += names;
806 if (!*cleaned || !strcmp(cleaned, "-")) {
807 ++removed_lines;
808 free(cleaned);
809 }
810 else push_owned(&lines, cleaned);
811 }
812 }
813 free_strings(&c.text);
814 if (!lines.count) {
815 ++removed_cues;
816 free_strings(&lines);
817 } else {
818 c.text = lines;
819 push_cue(&result, c);
820 }
821 }
822 free(cues.items);
823 cues = result;
824 if (cmd == 26) push_owned(out, format("STRIP tags=%d removed_lines=%d removed_cues=%d", removed_items, removed_lines, removed_cues));
825 else push_owned(out, format("CLEANHI segments=%d labels=%d removed_lines=%d removed_cues=%d", removed_items, labels, removed_lines, removed_cues));
826 } else if (cmd == 28) {
827 Strings pair = split(a->raw.items[0], '|', 0);
828 const char *old = pair.items[0], *replacement = pair.items[1];
829 int count = 0, matching_cues = 0;
830 for (int i = 0; i < n; ++i) {
831 int matches = 0;
832 for (int j = 0; j < cues.items[i].text.count; ++j) {
833 char **line = &cues.items[i].text.items[j];
834 const char *p = *line, *found;
835 Strings pieces = {0};
836 while ((found = strstr(p, old)) != NULL) {
837 push_owned(&pieces, slice(p, (size_t)(found - p)));
838 push(&pieces, replacement);
839 p = found + strlen(old);
840 ++matches;
841 }
842 push(&pieces, p);
843 char *result = join(&pieces, "");
844 free_strings(&pieces);
845 free(*line);
846 *line = result;
847 }
848 count += matches;
849 if (matches) ++matching_cues;
850 }
851 free_strings(&pair);
852 push_owned(out, format("REPLACE count=%d cues=%d", count, matching_cues));
853 }

855static bool load_block(const Strings *block, int position) {
856 Integer index = 0, start = 0, end = 0;
857 if (!unsigned_value(block->items[0], &index) || strlen(block->items[0]) > 6) {
858 printf("BAD CUE %d: index\n", position);
859 return false;
860 }
861 bool valid = block->count >= 2 && strlen(block->items[1]) == 29 && !strncmp(block->items[1] + 12, " --> ", 5);
862 if (valid) {
863 char time[13];
864 memcpy(time, block->items[1], 12);
865 time[12] = '\0';
866 valid = time_value(time, &start) && time_value(block->items[1] + 17, &end);
867 }
868 if (!valid) {
869 printf("BAD CUE %d: time\n", position);
870 return false;
871 }
872 if (block->count < 3) {
873 printf("BAD CUE %d: no text\n", position);
874 return false;
875 }
876 Cue c = {index, start, end, {0}};
877 for (int i = 2; i < block->count; ++i) push(&c.text, block->items[i]);
878 push_cue(&cues, c);
879 return true;
880}
881int main(void) {
882 char line[256];
883 Strings block = {0};
884 int blocks = 0, bad = 0;
885 while (fgets(line, sizeof line, stdin)) {
886 line[strcspn(line, "\n")] = '\0';
887 bool terminator = !strcmp(line, "###");
888 if (!*line || terminator) {
889 if (block.count) {
890 if (!load_block(&block, ++blocks)) ++bad;

899 for (int i = 0; i < count; ++i) {

907 free_strings(&output);
908 } else printf("ERROR %s\n", error_message);

910 }
911 free_cues(&cues);
