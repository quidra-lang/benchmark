38[[noreturn]] void fail(const string &message) { throw std::runtime_error(message); }
39vector<string> split(const string &s, char delimiter, int limit = 0) {
40 vector<string> result;
41 std::size_t start = 0;
42 for (std::size_t i = 0; i <= s.size(); ++i) {
43 if (i == s.size() || (s[i] == delimiter && (!limit || int(result.size()) + 1 < limit))) {
44 result.push_back(s.substr(start, i - start));
45 start = i + 1;
46 }
47 }
48 return result;
49}
50string join(const vector<string> &items, const string &separator) {
51 string result;
52 for (std::size_t i = 0; i < items.size(); ++i) {
53 if (i) result += separator;
54 result += items[i];
55 }
56 return result;
57}
58bool begins(const string &s, const string &prefix) { return s.compare(0, prefix.size(), prefix) == 0; }
59string trim(const string &s) {
60 std::size_t first = s.find_first_not_of(' ');
61 return first == string::npos ? "" : s.substr(first, s.find_last_not_of(' ') - first + 1);
62}
63bool unsignedValue(const string &s, Integer &value) {
64 if (s.empty() || s.size() > 9) return false;
65 value = 0;
66 for (char c : s) {
67 if (c < '0' || c > '9') return false;
68 value = value * 10 + c - '0';
69 }
70 return true;
71}
72bool timeValue(const string &s, Integer &value) {
73 if (s.size() != 12 || s[2] != ':' || s[5] != ':' || s[8] != ',') return false;
74 for (int i = 0; i < 12; ++i) {
75 if (i != 2 && i != 5 && i != 8 && (s[i] < '0' || s[i] > '9')) return false;
76 }
77 Integer h = std::stoll(s.substr(0, 2)), m = std::stoll(s.substr(3, 2));
78 Integer sec = std::stoll(s.substr(6, 2)), ms = std::stoll(s.substr(9));
79 value = h * 3600000 + m * 60000 + sec * 1000 + ms;
80 return m <= 59 && sec <= 59;
81}
82string timeText(Integer t, int kind = 0) {
83 std::ostringstream out;
84 out << std::setfill('0');
85 if (kind == 2) {
86 Integer c = (t + 5) / 10;
87 out << c / 360000 << ':' << std::setw(2) << c / 6000 % 60 << ':'
88 << std::setw(2) << c / 100 % 60 << '.' << std::setw(2) << c % 100;
89 } else {
90 out << std::setw(2) << t / 3600000 << ':' << std::setw(2) << t / 60000 % 60 << ':'
91 << std::setw(2) << t / 1000 % 60 << (kind == 1 ? '.' : ',') << std::setw(3) << t % 1000;
92 }
93 return out.str();
94}
95string decimal(Integer value) {
96 return std::to_string(value / 100) + "." + (value % 100 < 10 ? "0" : "") + std::to_string(value % 100);
97}
98Integer floorRatio(Integer a, Integer b) {
99 if (b < 0) {
100 a = -a;
101 b = -b;
102 }
103 Integer q = a / b;
104 if (a % b < 0) --q;
105 return q;
106}
107Integer halfUp(Integer a, Integer b) {
108 if (b < 0) {
109 a = -a;
110 b = -b;
111 }
112 return floorRatio(2 * a + b, 2 * b);
113}

121vector<string> words(const vector<string> &lines) {
122 vector<string> result;
123 for (const string &line : lines) {
124 for (const string &word : split(line, ' ')) if (!word.empty()) result.push_back(word);
125 }
126 return result;

128vector<string> wrap(const vector<string> &lines, int width) {
129 vector<string> result;
130 for (const string &word : words(lines)) {
131 if (result.empty() || int(result.back().size() + 1 + word.size()) > width) result.push_back(word);
132 else result.back() += " " + word;
133 }
134 return result;

136vector<Markup> scanMarkup(const string &line) {
137 const vector<string> tags = {"<i>", "</i>", "<b>", "</b>", "<u>", "</u>"};
138 vector<Markup> result;
139 for (std::size_t p = 0; p < line.size();) {
140 int size = 1, kind = 0, digit = 0;
141 for (const string &tag : tags) {
142 if (line.compare(p, tag.size(), tag) == 0) {
143 size = int(tag.size());
144 kind = 1;
145 break;
146 }
147 }
148 if (!kind && line.compare(p, 7, "</font>") == 0) {
149 size = 7;
150 kind = 2;
151 }
152 if (!kind && line.compare(p, 5, "<font") == 0 && p + 5 < line.size() && (line[p + 5] == ' ' || line[p + 5] == '>')) {
153 std::size_t end = line.find('>', p + 5);
154 if (end != string::npos) {
155 size = int(end - p + 1);
156 kind = 2;
157 }
158 }
159 if (!kind && p + 6 <= line.size() && line.compare(p, 4, "{\\an") == 0 && line[p + 4] >= '1' && line[p + 4] <= '9' && line[p + 5] == '}') {
160 size = 6;
161 kind = 3;
162 digit = line[p + 4] - '0';
163 }
164 result.push_back({kind, line.substr(p, size), digit});
165 p += size;
166 }
167 return result;
168}
169vector<string> converted(const vector<string> &lines, int mode, int &first, int &removed) {
170 vector<string> result;
171 first = removed = 0;
172 const vector<string> tags = {"<i>", "</i>", "<b>", "</b>", "<u>", "</u>"};
173 const vector<string> ass = {"{\\i1}", "{\\i0}", "{\\b1}", "{\\b0}", "{\\u1}", "{\\u0}"};
174 for (const string &line : lines) {
175 string output;
176 for (const Markup &m : scanMarkup(line)) {
177 if (m.kind == 3 && !first) first = m.digit;
178 if (mode == 0) {
179 if (!m.kind) output += m.text;
180 else ++removed;
181 } else if (mode == 1) {
182 if (m.kind == 1) output += m.text;
183 if (!m.kind) {
184 if (m.text == "&") output += "&amp;";
185 else if (m.text == "<") output += "&lt;";
186 else if (m.text == ">") output += "&gt;";
187 else output += m.text;
188 }
189 } else {
190 if (m.kind == 1) output += ass[std::find(tags.begin(), tags.end(), m.text) - tags.begin()];
191 if (m.kind == 0 || m.kind == 3) output += m.text;
192 }
193 }
194 if (mode != 2) output = trim(output);
195 if (mode == 2 || !output.empty()) result.push_back(output);
196 }
197 return result;
198}
199string normalize(const string &line) { return join(words(vector<string>{line}), " "); }
200string cleanHI(const string &line, int &segments, int &labels) {
201 string output;
202 segments = labels = 0;
203 for (std::size_t p = 0; p < line.size();) {
204 std::size_t end = string::npos;
205 if (line[p] == '[') end = line.find(']', p + 1);
206 if (line[p] == '(') end = line.find(')', p + 1);
207 if (end != string::npos) {
208 ++segments;
209 p = end + 1;
210 }
211 else output += line[p++];
212 }
213 string text = normalize(output), prefix, body = text;
214 if (begins(body, "- ")) {
215 prefix = "- ";
216 body = body.substr(2);
217 }
218 std::size_t colon = body.find(':');
219 if (colon != string::npos && colon > 0) {
220 string label = body.substr(0, colon);
221 bool valid = label[0] >= 'A' && label[0] <= 'Z';
222 for (char c : label) {
223 if (!((c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9') || c == ' ')) valid = false;
224 }
225 if (valid) {
226 labels = 1;
227 text = normalize(prefix + body.substr(colon + 1));
228 }
229 }
230 return text;

232Arguments parseArguments(const string &line) {
233 const vector<string> names = {"LIST", "VALIDATE", "SEARCH", "AT", "GAPS", "STATS", "PRINT", "TOVTT", "TOASS", "FRAMES", "SHIFT", "SHIFTFROM", "SCALE", "SYNC", "SNAP", "FIXOVERLAP", "EXTEND", "TIME", "SORT", "RENUMBER", "MERGE", "SPLIT", "INSERT", "DELETE", "TEXT", "BALANCE", "STRIP", "CLEANHI", "REPLACE", "UNDO"};
234 const vector<string> schemas = {"", "", "S", "T", "U", "", "", "", "", "F", "I", "PI", "UU", "PTPT", "F", "U", "U", "PTT", "", "", "U", "U", "PTTX", "P", "PX", "", "", "", "R", ""};
235 std::size_t space = line.find(' ');
236 string name = line.substr(0, space);
237 vector<string>::const_iterator found = std::find(names.begin(), names.end(), name);
238 if (found == names.end()) fail("unknown command");
239 string schema = schemas[found - names.begin()];
240 Arguments a;
241 a.name = name;
242 if (space != string::npos) a.raw = split(line.substr(space + 1), ' ', schema.find_first_of("SXR") != string::npos ? int(schema.size()) : 0);
243 if (name == "DELETE" && a.raw.size() == 2) schema = "PP";
244 if (a.raw.size() != schema.size()) fail("bad arguments");
245 a.values.resize(a.raw.size());
246 for (int i = 0; i < int(schema.size()); ++i) {
247 string s = a.raw[i];
248 bool valid = true;
249 Integer value = 0;
250 char kind = schema[i];
251 if (kind == 'P' || kind == 'U') {
252 valid = unsignedValue(s, value);
253 if (kind == 'P' && value == 0) valid = false;
254 } else if (kind == 'I') {
255 bool negative = begins(s, "-");
256 valid = unsignedValue(negative ? s.substr(1) : s, value);
257 if (negative) value = -value;
258 } else if (kind == 'T') valid = timeValue(s, value);
259 else if (kind == 'F') {
260 vector<string> parts = split(s, '/');
261 if (parts.size() > 2) valid = false;
262 else {
263 valid = unsignedValue(parts[0], value);
264 if (parts.size() == 2) valid = unsignedValue(parts[1], a.denominator) && valid;
265 valid = valid && value >= 1 && value <= 100000 && a.denominator >= 1 && a.denominator <= 100000;
266 }
267 } else if (kind == 'X') {
268 for (const string &part : split(s, '|')) if (part.empty()) valid = false;
269 } else if (kind == 'R') {
270 vector<string> parts = split(s, '|');
271 valid = parts.size() == 2 && !parts[0].empty() && !parts[1].empty();
272 } else if (kind == 'S') valid = !s.empty();
273 if (!valid) fail("bad arguments");
274 a.values[i] = value;
275 }
276 const vector<Integer> &v = a.values;
277 if (name == "SCALE" && (v[0] < 1 || v[0] > 100000 || v[1] < 1 || v[1] > 100000)) fail("bad arguments");
278 if (name == "GAPS" || name == "FIXOVERLAP" || name == "MERGE" || name == "EXTEND") {
279 if (v[0] < (name == "EXTEND" ? 1 : 0) || v[0] > 60000) fail("bad arguments");
280 }
281 if (name == "SPLIT" && (v[0] < 10 || v[0] > 80)) fail("bad arguments");
282 if (name == "DELETE" && v.size() == 2 && v[1] < v[0]) fail("bad arguments");
283 Integer limit = Integer(cues.size()) + (name == "INSERT" ? 1 : 0);
284 for (int i = 0; i < int(schema.size()); ++i) {
285 if (schema[i] == 'P' && v[i] > limit) fail("no cue " + std::to_string(v[i]));
286 }
287 return a;
288}

291}

297vector<string> execute(const string &line) {
298 Arguments a = parseArguments(line);
299 const string &name = a.name;
300 const vector<Integer> &v = a.values;
301 int n = int(cues.size());
302 vector<string> out;
303 if (name == "UNDO") {
304 if (history.empty()) fail("nothing to undo");
305 Record previous = std::move(history.back());
306 history.pop_back();
307 cues = std::move(previous.cues);
308 return {"UNDO command=" + previous.command};
309 }
310 if (name == "SYNC" && cues[v[0] - 1].start == cues[v[2] - 1].start) fail("sync points coincide");
311 const string modifying = " SHIFT SHIFTFROM SCALE SYNC SNAP FIXOVERLAP EXTEND TIME SORT RENUMBER MERGE SPLIT INSERT DELETE TEXT BALANCE STRIP CLEANHI REPLACE ";
312 if (modifying.find(" " + name + " ") != string::npos) history.push_back({name, cues});
313 if (name == "LIST" || name == "AT") {
314 int matches = 0;
315 for (int i = 0; i < n; ++i) {
316 if (name == "LIST" || (cues[i].start <= v[0] && v[0] < cues[i].end)) {
317 out.push_back(cueLine(i + 1, cues[i]));
318 ++matches;
319 }
320 }
321 if (name == "AT") out.push_back(text("AT active=", matches));
322 } else if (name == "VALIDATE") {
323 for (int i = 0; i < n; ++i) {
324 const Cue &c = cues[i];
325 string prefix = text("cue ", i + 1, ": ");
326 Integer d = c.end - c.start;
327 if (c.index != i + 1) out.push_back(text(prefix, "index ", c.index, " (expected ", i + 1, ")"));
328 if (d <= 0) out.push_back(prefix + "end not after start");
329 if (i && c.start < cues[i - 1].end) out.push_back(text(prefix, "overlaps previous by ", cues[i - 1].end - c.start, " ms"));
330 if (d >= 1 && d < 700) out.push_back(text(prefix, "too short (", d, " ms)"));
331 if (d > 7000) out.push_back(text(prefix, "too long (", d, " ms)"));
332 for (int j = 0; j < int(c.text.size()); ++j) {
333 if (c.text[j].size() > 42) out.push_back(text(prefix, "line ", j + 1, " too long (", c.text[j].size(), " chars)"));
334 }
335 if (c.text.size() >= 3) out.push_back(text(prefix, "too many lines (", c.text.size(), ")"));
336 Integer chars = countChars(c);
337 if (d >= 1 && chars * 1000 > 21 * d) out.push_back(prefix + "too fast (" + decimal(chars * 100000 / d) + " cps)");
338 }
339 out.push_back(text("VALIDATE problems=", out.size()));
340 } else if (name == "SEARCH") {
341 for (int i = 0; i < n; ++i) {
342 for (const string &line : cues[i].text) {
343 if (line.find(a.raw[0]) != string::npos) {
344 out.push_back(text(i + 1, " ", timeText(cues[i].start), " ", line));
345 break;
346 }
347 }
348 }
349 out.push_back(text("SEARCH found=", out.size()));
350 } else if (name == "GAPS") {
351 for (int i = 0; i + 1 < n; ++i) {
352 Integer gap = cues[i + 1].start - cues[i].end;
353 if (gap < v[0]) out.push_back(text(i + 1, " ", i + 2, " ", gap));
354 }
355 out.push_back(text("GAPS count=", out.size()));
356 } else if (name == "STATS") {
357 out.push_back(text("STATS cues=", n));
358 if (!n) return out;
359 Integer total = 0, chars = 0, longest = 0;
360 int longestPosition = 1;
361 for (int i = 0; i < n; ++i) {
362 Integer d = std::max(Integer(0), cues[i].end - cues[i].start);
363 total += d;
364 if (d) chars += countChars(cues[i]);
365 if (d > longest) {
366 longest = d;
367 longestPosition = i + 1;
368 }
369 }
370 out.push_back("duration total=" + timeText(total) + " mean=" + timeText(total / n));
371 out.push_back("speed " + (total ? decimal(chars * 100000 / total) : "-"));
372 out.push_back(text("longest cue=", longestPosition, " duration=", timeText(longest)));
373 if (n == 1) out.push_back("gap none");
374 else {
375 int position = 1;
376 Integer gap = cues[1].start - cues[0].end;
377 for (int i = 1; i + 1 < n; ++i) {
378 Integer candidate = cues[i + 1].start - cues[i].end;
379 if (candidate > gap) {
380 gap = candidate;
381 position = i + 1;
382 }
383 }
384 out.push_back(text("gap cue=", position, " ms=", gap));
385 }
386 } else if (name == "PRINT" || name == "TOVTT") {
387 if (name == "TOVTT") out.insert(out.end(), {"WEBVTT", ""});
388 for (const Cue &c : cues) {
389 int code = 0, removed = 0, kind = name == "TOVTT" ? 1 : 0;
390 vector<string> convertedLines;
391 if (kind) convertedLines = converted(c.text, 1, code, removed);
392 const vector<string> &lines = kind ? convertedLines : c.text;
393 if (lines.empty()) continue;
394 string setting;
395 if (code >= 7) setting = " line:0";
396 else if (code >= 4) setting = " line:50%";
397 if (code && code % 3 == 1) setting += " align:start";
398 if (code && code % 3 == 0) setting += " align:end";
399 out.push_back(std::to_string(c.index));
400 out.push_back(timeText(c.start, kind) + " --> " + timeText(c.end, kind) + setting);
401 out.insert(out.end(), lines.begin(), lines.end());
402 out.push_back("");
403 }
404 } else if (name == "TOASS") {
405 out = {"[Events]", "Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text"};
406 for (const Cue &c : cues) {
407 int code, removed;
408 vector<string> lines = converted(c.text, 2, code, removed);
409 out.push_back("Dialogue: 0," + timeText(c.start, 2) + "," + timeText(c.end, 2) + ",Default,,0,0,0,," + join(lines, "\\N"));
410 }
411 } else if (name == "FRAMES") {
412 for (int i = 0; i < n; ++i) out.push_back(text(i + 1, " ", cues[i].start * v[0] / (1000 * a.denominator), " ", cues[i].end * v[0] / (1000 * a.denominator)));
413 } else if (name == "SHIFT" || name == "SHIFTFROM" || name == "SYNC" || name == "SCALE" || name == "SNAP") {
414 int start = name == "SHIFTFROM" ? int(v[0] - 1) : 0;
415 Integer shift = name == "SHIFT" ? v[0] : name == "SHIFTFROM" ? v[1] : 0;
416 Integer s1 = 0, s2 = 0;
417 if (name == "SYNC") {
418 s1 = cues[v[0] - 1].start;
419 s2 = cues[v[2] - 1].start;
420 }
421 int clamped = 0, changed = 0;
422 for (int i = start; i < n; ++i) {
423 bool cueClamped = false;
424 for (Integer *t : {&cues[i].start, &cues[i].end}) {
425 Integer old = *t, value = old;
426 if (name == "SHIFT" || name == "SHIFTFROM") value += shift;
427 if (name == "SCALE") value = halfUp(value * v[0], v[1]);
428 if (name == "SYNC") value = halfUp(v[1] * (s2 - s1) + (value - s1) * (v[3] - v[1]), s2 - s1);
429 if (name == "SNAP") {
430 Integer frame = halfUp(value * v[0], 1000 * a.denominator);
431 value = halfUp(frame * 1000 * a.denominator, v[0]);
432 }
433 if (value < 0) {
434 value = 0;
435 cueClamped = true;
436 }
437 if (old != value) ++changed;
438 *t = value;
439 }
440 if (cueClamped) ++clamped;
441 }
442 if (name == "SNAP") out.push_back(text("SNAP changed=", changed));
443 else if (name == "SCALE") out.push_back(text("SCALE cues=", n));
444 else out.push_back(text(name, " cues=", n - start, " clamped=", clamped));
445 } else if (name == "FIXOVERLAP") {
446 int fixed = 0, failed = 0;
447 for (int i = 0; i + 1 < n; ++i) {
448 Integer candidate = cues[i + 1].start - v[0];
449 if (cues[i].end > candidate) {
450 if (candidate > cues[i].start) {
451 cues[i].end = candidate;
452 ++fixed;
453 } else {
454 ++failed;
455 out.push_back(text("cue ", i + 1, ": cannot fix"));
456 }
457 }
458 }
459 out.push_back(text("FIXOVERLAP fixed=", fixed, " failed=", failed));
460 } else if (name == "EXTEND") {
461 int extended = 0, shortCount = 0;
462 for (int i = 0; i < n; ++i) {
463 if (cues[i].end - cues[i].start < v[0]) {
464 Integer candidate = cues[i].start + v[0];
465 if (i + 1 < n) candidate = std::min(candidate, cues[i + 1].start);
466 if (candidate > cues[i].end) {
467 cues[i].end = candidate;
468 ++extended;
469 }
470 }
471 if (cues[i].end - cues[i].start < v[0]) ++shortCount;
472 }
473 out.push_back(text("EXTEND extended=", extended, " short=", shortCount));
474 } else if (name == "TIME") {
475 cues[v[0] - 1].start = v[1];
476 cues[v[0] - 1].end = v[2];
477 out.push_back(text("TIME cue=", v[0]));
478 } else if (name == "SORT") {
479 vector<int> order(n);
480 std::iota(order.begin(), order.end(), 0);
481 std::stable_sort(order.begin(), order.end(), [](int i, int j) {
482 if (cues[i].start != cues[j].start) return cues[i].start < cues[j].start;
483 return cues[i].end < cues[j].end;
484 });
485 vector<Cue> sorted;
486 int moved = 0;
487 for (int i = 0; i < n; ++i) {
488 sorted.push_back(std::move(cues[order[i]]));
489 if (i != order[i]) ++moved;
490 }
491 cues = std::move(sorted);
492 out.push_back(text("SORT moved=", moved));
493 } else if (name == "RENUMBER") {
494 int changed = 0;
495 for (int i = 0; i < n; ++i) {
496 if (cues[i].index != i + 1) ++changed;
497 cues[i].index = i + 1;
498 }
499 out.push_back(text("RENUMBER changed=", changed));
500 } else if (name == "MERGE") {
501 vector<Cue> result;
502 int merged = 0;
503 for (Cue &c : cues) {
504 if (!result.empty()) {
505 Cue &previous = result.back();
506 if (c.start - previous.end < v[0] && !dialogue(previous) && !dialogue(c)) {
507 vector<string> lines = previous.text;
508 lines.insert(lines.end(), c.text.begin(), c.text.end());
509 vector<string> wrapped = wrap(lines, 42);
510 if (wrapped.size() <= 2) {
511 previous.end = std::max(previous.end, c.end);
512 previous.text = std::move(wrapped);
513 ++merged;
514 continue;
515 }
516 }
517 }
518 result.push_back(std::move(c));
519 }
520 cues = std::move(result);
521 out.push_back(text("MERGE merged=", merged, " cues=", cues.size()));
522 } else if (name == "SPLIT") {
523 vector<Cue> result;
524 int splitCount = 0;
525 for (Cue &c : cues) {
526 Integer d = c.end - c.start;
527 if (d < 1 || dialogue(c)) {
528 result.push_back(std::move(c));
529 continue;
530 }
531 vector<string> wrapped = wrap(c.text, int(v[0]));
532 if (wrapped.size() <= 2) {
533 c.text = std::move(wrapped);
534 result.push_back(std::move(c));
535 continue;
536 }
537 ++splitCount;
538 vector<Cue> parts;
539 Integer total = 0;
540 for (int i = 0; i < int(wrapped.size()); i += 2) {
541 Cue part{c.index, 0, 0, {wrapped[i]}};
542 if (i + 1 < int(wrapped.size())) part.text.push_back(wrapped[i + 1]);
543 total += countChars(part);
544 parts.push_back(std::move(part));
545 }
546 vector<Integer> lengths(parts.size()), remainders(parts.size());
547 vector<int> order(parts.size());
548 Integer remaining = d;
549 for (int i = 0; i < int(parts.size()); ++i) {
550 Integer product = d * countChars(parts[i]);
551 lengths[i] = product / total;
552 remainders[i] = product % total;
553 remaining -= lengths[i];
554 order[i] = i;
555 }
556 std::stable_sort(order.begin(), order.end(), [&](int i, int j) { return remainders[i] > remainders[j]; });
557 for (int i = 0; i < remaining; ++i) ++lengths[order[i]];
558 Integer start = c.start;
559 for (int i = 0; i < int(parts.size()); ++i) {
560 parts[i].start = start;
561 parts[i].end = start + lengths[i];
562 start += lengths[i];
563 result.push_back(std::move(parts[i]));
564 }
565 }
566 cues = std::move(result);
567 out.push_back(text("SPLIT split=", splitCount, " cues=", cues.size()));
568 } else if (name == "INSERT") {
569 cues.insert(cues.begin() + v[0] - 1, Cue{v[0], v[1], v[2], split(a.raw[3], '|')});
570 out.push_back(text("INSERT cue=", v[0], " cues=", cues.size()));
571 } else if (name == "DELETE") {
572 int p = int(v[0] - 1), q = int(v.size() == 2 ? v[1] : v[0]);
573 cues.erase(cues.begin() + p, cues.begin() + q);
574 out.push_back(text("DELETE removed=", q - p, " cues=", cues.size()));
575 } else if (name == "TEXT") {
576 cues[v[0] - 1].text = split(a.raw[1], '|');
577 out.push_back(text("TEXT cue=", v[0], " lines=", cues[v[0] - 1].text.size()));
578 } else if (name == "BALANCE") {
579 int changed = 0, skipped = 0, dialogues = 0;
580 for (Cue &c : cues) {
581 if (dialogue(c)) {
582 ++dialogues;
583 continue;
584 }
585 vector<string> w = words(c.text), candidate;
586 string all = join(w, " ");
587 if (all.size() <= 42) candidate.push_back(all);
588 else if (all.size() <= 85) {
589 int bestDifference = 1000000, bestFirst = 1000000;
590 for (int j = 1; j < int(w.size()); ++j) {
591 string first = join(vector<string>(w.begin(), w.begin() + j), " ");
592 string second = join(vector<string>(w.begin() + j, w.end()), " ");
593 int difference = std::abs(int(first.size()) - int(second.size()));
594 if (first.size() <= 42 && second.size() <= 42 && (difference < bestDifference || (difference == bestDifference && int(first.size()) < bestFirst))) {
595 candidate = {first, second};
596 bestDifference = difference;
597 bestFirst = int(first.size());
598 }
599 }
600 }
601 if (candidate.empty()) ++skipped;
602 else {
603 if (candidate != c.text) ++changed;
604 c.text = std::move(candidate);
605 }
606 }
607 out.push_back(text("BALANCE changed=", changed, " skipped=", skipped, " dialogue=", dialogues));
608 } else if (name == "STRIP" || name == "CLEANHI") {
609 int removedItems = 0, labels = 0, removedLines = 0, removedCues = 0;
610 vector<Cue> result;
611 for (Cue &c : cues) {
612 vector<string> lines;
613 if (name == "STRIP") {
614 int first, tags;
615 lines = converted(c.text, 0, first, tags);
616 removedItems += tags;
617 removedLines += int(c.text.size() - lines.size());
618 } else {
619 for (const string &line : c.text) {
620 int segments, names;
621 string cleaned = cleanHI(line, segments, names);
622 removedItems += segments;
623 labels += names;
624 if (cleaned.empty() || cleaned == "-") ++removedLines;
625 else lines.push_back(cleaned);
626 }
627 }
628 if (lines.empty()) ++removedCues;
629 else {
630 c.text = std::move(lines);
631 result.push_back(std::move(c));
632 }
633 }
634 cues = std::move(result);
635 if (name == "STRIP") out.push_back(text("STRIP tags=", removedItems, " removed_lines=", removedLines, " removed_cues=", removedCues));
636 else out.push_back(text("CLEANHI segments=", removedItems, " labels=", labels, " removed_lines=", removedLines, " removed_cues=", removedCues));
637 } else if (name == "REPLACE") {
638 vector<string> parts = split(a.raw[0], '|');
639 int count = 0, matchingCues = 0;
640 for (Cue &c : cues) {
641 int matches = 0;
642 for (string &line : c.text) {
643 string result;
644 std::size_t p = 0, found;
645 while ((found = line.find(parts[0], p)) != string::npos) {
646 result += line.substr(p, found - p) + parts[1];
647 p = found + parts[0].size();
648 ++matches;
649 }
650 line = result + line.substr(p);
651 }
652 count += matches;
653 if (matches) ++matchingCues;
654 }
655 out.push_back(text("REPLACE count=", count, " cues=", matchingCues));
656 }
657 return out;
658}
659string loadBlock(const vector<string> &block, int position) {
660 Integer index, start = 0, end = 0;
661 if (!unsignedValue(block[0], index) || block[0].size() > 6) return text("BAD CUE ", position, ": index");
662 bool valid = block.size() >= 2 && block[1].size() == 29 && block[1].substr(12, 5) == " --> ";
663 if (valid) valid = timeValue(block[1].substr(0, 12), start) && timeValue(block[1].substr(17), end);
664 if (!valid) return text("BAD CUE ", position, ": time");
665 if (block.size() < 3) return text("BAD CUE ", position, ": no text");
666 cues.push_back({index, start, end, vector<string>(block.begin() + 2, block.end())});
667 return "";
668}
