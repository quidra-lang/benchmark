1Edge = Struct.new(:u, :v, :capacity, :cost, :lower, :flow)

3class FlowNetwork
4 attr_reader :n, :edges
5
6 def initialize(n, edges = [])
7 @n = n
8 @edges = edges.map(&:dup)
9 end
10
11 def arcs(vertex = nil)
12 forward = @edges.each_index.select { |i| vertex.nil? || @edges[i].u == vertex }.map { |i| i + 1 }
13 backward = @edges.each_index.select { |i| vertex.nil? || @edges[i].v == vertex }.map { |i| -i - 1 }
14 forward + backward
15 end
16
17 def edge(arc)
18 @edges[arc.abs - 1]
19 end
20
21 def from(arc)
22 arc.positive? ? edge(arc).u : edge(arc).v
23 end
24
25 def to(arc)
26 arc.positive? ? edge(arc).v : edge(arc).u
27 end
28
29 def residual(arc)
30 arc.positive? ? edge(arc).capacity - edge(arc).flow : edge(arc).flow
31 end
32
33 def arc_cost(arc)
34 arc.positive? ? edge(arc).cost : -edge(arc).cost
35 end
36
37 def push(path, amount)
38 path.each { |arc| edge(arc).flow += arc.positive? ? amount : -amount }
39 end
40
41 def bottleneck(path)
42 path.map { |arc| residual(arc) }.min
43 end
44
45 def bfs(source, threshold = 1)
46 levels = Array.new(@n)
47 parents = Array.new(@n)
48 levels[source] = 0
49 queue = [source]
50 queue.each do |u|
51 arcs(u).each do |arc|
52 v = to(arc)
53 next unless residual(arc) >= threshold && levels[v].nil?
54 levels[v] = levels[u] + 1
55 parents[v] = arc
56 queue << v
57 end
58 end
59 [levels, parents]
60 end
61
62 def path_to(source, target, parents)
63 path = []
64 until target == source
65 arc = parents[target]
66 path << arc
67 target = from(arc)
68 end
69 path.reverse
70 end
71
72 def augment(source, target, threshold = 1)
73 levels, parents = bfs(source, threshold)
74 return nil if levels[target].nil?
75 path = path_to(source, target, parents)
76 amount = bottleneck(path)
77 push(path, amount)
78 { path: path, amount: amount, delta: threshold }
79 end
80
81 def maxflow(source, target, method = 'MAXFLOW')
82 records = []
83 phases = 0
84 if method == 'DINIC'
85 loop do
86 levels, = bfs(source)
87 break if levels[target].nil?
88 phases += 1
89 lists = (0...@n).map { |v| arcs(v) }
90 pointers = Array.new(@n, 0)
91 path = []
92 u = source
93 loop do
94 if u == target
95 amount = bottleneck(path)
96 push(path, amount)
97 records << { path: path.dup, amount: amount, phase: phases }
98 path = []
99 u = source
100 elsif pointers[u] >= lists[u].length
101 break if u == source
102 u = from(path.pop)
103 pointers[u] += 1
104 else
105 arc = lists[u][pointers[u]]
106 v = to(arc)
107 if residual(arc).positive? && levels[v] && levels[v] == levels[u] + 1
108 path << arc
109 u = v
110 else
111 pointers[u] += 1
112 end
113 end
114 end
115 end
116 else
117 threshold = 1
118 maximum = @edges.map(&:capacity).max || 0
119 threshold *= 2 while method == 'SCALING' && threshold * 2 <= maximum
120 while threshold >= 1
121 loop do
122 record = augment(source, target, threshold)
123 break unless record
124 records << record
125 end
126 threshold = method == 'SCALING' ? threshold / 2 : 0
127 end
128 end
129 [records, phases]
130 end

132 def negative_cycle?
133 distance = Array.new(@n, 0)
134 @n.times do
135 changed = false
136 @edges.each do |e|
137 next unless e.capacity.positive? && distance[e.u] + e.cost < distance[e.v]
138 distance[e.v] = distance[e.u] + e.cost
139 changed = true
140 end
141 return false unless changed
142 end
143 @n.positive?
144 end
145
146 def mincost(source, target, limit = nil)
147 amount = 0
148 cost = 0
149 records = []
150 while limit.nil? || amount < limit
151 usable = arcs.select { |arc| residual(arc).positive? }
152 distance = Array.new(@n)
153 parents = Array.new(@n)
154 distance[source] = 0
155 loop do
156 changed = false
157 usable.each do |arc|
158 u = from(arc)
159 v = to(arc)
160 next if distance[u].nil?
161 value = distance[u] + arc_cost(arc)
162 next unless distance[v].nil? || value < distance[v]
163 distance[v] = value
164 parents[v] = arc
165 changed = true
166 end
167 break unless changed
168 end
169 break if distance[target].nil?
170 path = path_to(source, target, parents)
171 delta = bottleneck(path)
172 delta = [delta, limit - amount].min if limit
173 push(path, delta)
174 amount += delta
175 cost += delta * distance[target]
176 records << { path: path, amount: delta, cost: distance[target] }
177 end
178 [amount, cost, records]
179 end
180
181 def balance
182 values = Array.new(@n, 0)
183 @edges.each do |e|
184 values[e.u] += e.flow
185 values[e.v] -= e.flow
186 end
187 values
188 end
189
190 def total_cost
191 @edges.sum { |e| e.flow * e.cost }
192 end
193
194 def decompose
195 remaining = @edges.map(&:flow)
196 balances = balance
197 result = []
198 while (first = remaining.index(&:positive?))
199 start = balances.index(&:positive?) || @edges[first].u
200 vertices = [start]
201 path = []
202 loop do
203 u = vertices.last
204 i = @edges.each_index.find { |j| @edges[j].u == u && remaining[j].positive? }
205 v = @edges[i].v
206 path << i + 1
207 if (q = vertices.index(v))
208 cycle = path[q..]
209 delta = cycle.map { |arc| remaining[arc - 1] }.min
210 cycle.each { |arc| remaining[arc - 1] -= delta }
211 result << { kind: 'CYCLE', start: v, path: cycle, amount: delta }
212 break
213 end
214 vertices << v
215 next unless balances[v].negative?
216 delta = [path.map { |arc| remaining[arc - 1] }.min, balances[start], -balances[v]].min
217 path.each { |arc| remaining[arc - 1] -= delta }
218 balances[start] -= delta
219 balances[v] += delta
220 result << { kind: 'PATH', start: start, path: path, amount: delta }
221 break
222 end
223 end
224 result
225 end
226end

242 def vertex(name)
243 index = @names.index(name)
244 raise FlowError, "UNKNOWN NODE #{name}" unless index
245 index

248 def endpoints(s, t)
249 source = vertex(s)
250 target = vertex(t)
251 raise FlowError, 'SAME NODE' if source == target
252 [source, target]
253 end

255 def checked_edge(text)
256 number = text.to_i
257 raise FlowError, "UNKNOWN EDGE #{text}" unless number.between?(1, @network.edges.length)
258 @network.edges[number - 1]
259 end

261 def start_flow(source, target)
262 @network.edges.each { |e| e.flow = 0 } unless @owner == [:flow, source, target]
263 @owner = [:flow, source, target]
264 @network.balance[source]
265 end

267 def notation(network, start, path, original_only = nil)
268 words = [@names[start]]
269 path.each do |arc|
270 next if original_only && arc.abs > original_only
271 words << "#{arc.positive? ? '+' : '-'}#{arc.abs}"
272 words << (original_only ? @names[@network.edges[arc.abs - 1].v] : @names[network.to(arc)])
273 end
274 words.join(' ')
275 end

277 def flow_list
278 @network.edges.each_with_index { |e, i| puts "FLOW #{i + 1} #{@names[e.u]} #{@names[e.v]} #{e.flow}/#{e.capacity}" }
279 end

281 def fresh
282 network = FlowNetwork.new(@names.length, @network.edges)
283 network.edges.each { |e| e.flow = 0 }
284 network
285 end

287 def circulation(mode)
288 work = fresh
289 work.edges.each { |e| e.capacity -= e.lower }
290 raise FlowError, 'NEGATIVE CYCLE' if mode == 'MINCOST' && work.negative_cycle?
291 total = @demands.sum
292 unless total.zero?
293 puts "CIRCULATION infeasible unbalanced=#{total}"
294 return
295 end
296 residual = @demands.dup
297 @network.edges.each do |e|
298 residual[e.v] -= e.lower
299 residual[e.u] += e.lower
300 end
301 source = @names.length
302 target = source + 1
303 edges = work.edges.map(&:dup)
304 residual.each_with_index { |r, v| edges << Edge.new(source, v, -r, 0, 0, 0) if r.negative? }
305 sinks = []
306 residual.each_with_index do |r, v|
307 next unless r.positive?
308 sinks << edges.length
309 edges << Edge.new(v, target, r, 0, 0, 0)
310 end
311 work = FlowNetwork.new(target + 1, edges)
312 mode == 'ANY' ? work.maxflow(source, target) : work.mincost(source, target)
313 required = residual.select(&:positive?).sum
314 achieved = sinks.sum { |i| work.edges[i].flow }
315 if achieved < required
316 puts "CIRCULATION infeasible required=#{required} achieved=#{achieved}"
317 reachable, = work.bfs(source)
318 puts 'CUT ' + @names.each_index.select { |v| reachable[v] }.map { |v| @names[v] }.join(' ')
319 else
320 @network.edges.each_with_index { |e, i| e.flow = e.lower + work.edges[i].flow }
321 @owner = [:circulation]
322 puts "CIRCULATION feasible required=#{required} cost=#{@network.total_cost}"
323 flow_list
324 end
325 end

327 def matching(left_text, right_text, pair_text)
328 left = left_text.split(',')
329 right = right_text.split(',')
330 seen = []
331 (left + right).each do |name|
332 raise FlowError, "DUPLICATE VERTEX #{name}" if seen.include?(name)
333 seen << name
334 end
335 pairs = pair_text == '-' ? [] : pair_text.split(',').map { |word| word.split('-') }
336 seen_pairs = []
337 pairs.each do |a, b|
338 raise FlowError, "UNKNOWN VERTEX #{a}" unless left.include?(a)
339 raise FlowError, "UNKNOWN VERTEX #{b}" unless right.include?(b)
340 raise FlowError, "DUPLICATE PAIR #{a}-#{b}" if seen_pairs.include?([a, b])
341 seen_pairs << [a, b]
342 end
343 lp = {}
344 rp = {}
345 examined = []
346 attempt = lambda do |a|
347 pairs.each do |x, b|
348 next unless x == a && !examined.include?(b)
349 examined << b
350 if !rp.key?(b) || attempt.call(rp[b])
351 rp[b] = a
352 lp[a] = b
353 return true
354 end
355 end
356 false
357 end
358 left.each do |a|
359 examined = []
360 attempt.call(a)
361 end
362 free_left = left.reject { |a| lp.key?(a) }
363 free_right = right.reject { |b| rp.key?(b) }
364 zl = free_left.dup
365 zr = []
366 zl.each do |a|
367 pairs.each do |x, b|
368 next unless x == a && lp[a] != b && !zr.include?(b)
369 zr << b
370 zl << rp[b] if rp.key?(b) && !zl.include?(rp[b])
371 end
372 end
373 cover = left.reject { |a| zl.include?(a) } + right.select { |b| zr.include?(b) }
374 puts "MATCH size=#{lp.length} left=#{left.length} right=#{right.length}"
375 left.each { |a| puts "PAIR #{a} #{lp[a]}" if lp.key?(a) }
376 puts "FREE #{free_left.empty? ? '-' : free_left.join(',')} #{free_right.empty? ? '-' : free_right.join(',')}"
377 puts "COVER size=#{cover.length}#{cover.empty? ? '' : ' ' + cover.join(' ')}"
378 puts "HALL #{left.select { |a| zl.include?(a) }.join(',')} #{zr.empty? ? '-' : right.select { |b| zr.include?(b) }.join(',')}" unless free_left.empty?
379 end

381 def assignment(mode, n, values)
382 raise FlowError, 'BAD SIZE' unless n.between?(1, 8)
383 raise FlowError, 'BAD LENGTH' unless values.length == n * n
384 best = nil
385 count = 0
386 chosen = nil
387 (0...n).to_a.permutation do |permutation|
388 cells = permutation.each_with_index.map { |column, row| values[row * n + column] }
389 next if cells.include?('x')
390 total = cells.sum(&:to_i)
391 if best.nil? || (mode == 'MIN' ? total < best : total > best)
392 best = total
393 count = 1
394 chosen = permutation.dup
395 elsif total == best
396 count += 1
397 chosen = permutation.dup if (permutation <=> chosen).negative?
398 end
399 end
400 if best.nil?
401 puts "ASSIGN #{mode} n=#{n} infeasible"
402 else
403 puts "ASSIGN #{mode} n=#{n} total=#{best} optimal=#{count}"
404 chosen.each_with_index { |column, row| puts "MATE #{row + 1} #{column + 1} #{values[row * n + column]}" }
405 end
406 end

408 def execute(line)
409 command, *args = line.split(' ')
410 case command
411 when 'NODE'
412 seen = @names.dup
413 args.each do |name|
414 raise FlowError, "DUPLICATE NODE #{name}" if seen.include?(name)
415 seen << name
416 end
417 raise FlowError, 'TOO MANY NODES' if seen.length > 20
418 @names = seen
419 @demands.concat(Array.new(args.length, 0))
420 @network = FlowNetwork.new(@names.length, @network.edges)
421 puts "NODE added=#{args.length} total=#{@names.length}"
422 when 'EDGE'
423 u = vertex(args[0])
424 v = vertex(args[1])
425 raise FlowError, 'SELF LOOP' if u == v
426 raise FlowError, 'TOO MANY EDGES' if @network.edges.length == 60
427 e = Edge.new(u, v, args[2].to_i, (args[3] || '0').to_i, 0, 0)
428 @network.edges << e
429 puts "EDGE #{@network.edges.length} #{args[0]} #{args[1]} cap=#{e.capacity} cost=#{e.cost}"
430 when 'LOWER', 'CAP', 'COST'
431 e = checked_edge(args[0])
432 value = args[1].to_i
433 if command == 'LOWER'
434 raise FlowError, 'BAD BOUND' if value > e.capacity
435 e.lower = value
436 puts "LOWER #{args[0]} low=#{value} cap=#{e.capacity}"
437 elsif command == 'CAP'
438 raise FlowError, 'BAD BOUND' if value < e.lower
439 raise FlowError, 'BELOW FLOW' if value < e.flow
440 e.capacity = value
441 puts "CAP #{args[0]} low=#{e.lower} cap=#{value}"
442 else
443 e.cost = value
444 puts "COST #{args[0]} cost=#{value}"
445 end
446 when 'DEMAND'
447 v = vertex(args[0])
448 @demands[v] = args[1].to_i
449 puts "DEMAND #{args[0]} #{args[1]} total=#{@demands.sum}"
450 when 'RESET'
451 puts "RESET edges=#{@network.edges.count { |e| !e.flow.zero? }}"
452 @network.edges.each { |e| e.flow = 0 }
453 @owner = nil
454 when 'CLEAR'
455 puts "CLEAR nodes=#{@names.length} edges=#{@network.edges.length}"
456 clear
457 when 'SHOW'
458 balances = @network.balance
459 puts "NETWORK nodes=#{@names.length} edges=#{@network.edges.length}"
460 if @owner.nil?
461 puts 'STATE none'
462 elsif @owner[0] == :circulation
463 puts "STATE circulation cost=#{@network.total_cost}"
464 else
465 puts "STATE flow #{@names[@owner[1]]} #{@names[@owner[2]]} value=#{balances[@owner[1]]} cost=#{@network.total_cost}"
466 end
467 @names.each_with_index do |name, v|
468 out = @network.edges.select { |e| e.u == v }.sum(&:flow)
469 incoming = out - balances[v]
470 puts "NODE #{name} demand=#{@demands[v]} out=#{out} in=#{incoming}"
471 end
472 @network.edges.each_with_index { |e, i| puts "EDGE #{i + 1} #{@names[e.u]} #{@names[e.v]} cap=#{e.capacity} cost=#{e.cost} low=#{e.lower} flow=#{e.flow}" }
473 when 'AUGMENT', 'MAXFLOW', 'DINIC', 'SCALING', 'MINCUT', 'SENSITIVITY', 'MINCOST', 'DISJOINT'
474 s, t = endpoints(args[0], args[1])
475 flow_command(command, args, s, t)
476 when 'DECOMPOSE'
477 records = @network.decompose
478 puts "DECOMPOSE paths=#{records.count { |r| r[:kind] == 'PATH' }} cycles=#{records.count { |r| r[:kind] == 'CYCLE' }}"
479 counters = Hash.new(0)
480 records.each do |r|
481 counters[r[:kind]] += 1
482 puts "#{r[:kind]} #{counters[r[:kind]]} amount=#{r[:amount]} #{notation(@network, r[:start], r[:path])}"
483 end
484 when 'VERIFY'
485 violations = []
486 @network.edges.each_with_index { |e, i| violations << "LOW #{i + 1} flow=#{e.flow} low=#{e.lower}" if e.flow < e.lower }
487 balance = @network.balance
488 @names.each_with_index do |name, v|
489 next if @owner && @owner[0] == :flow && @owner.drop(1).include?(v)
490 required = @owner && @owner[0] == :flow ? 0 : @demands[v]
491 violations << "BALANCE #{name} net=#{-balance[v]} need=#{required}" if -balance[v] != required
492 end
493 puts(violations.empty? ? 'VERIFY ok' : ["VERIFY violations=#{violations.length}", *violations])
494 when 'CIRCULATION'
495 circulation(args[0])
496 when 'MATCH'
497 matching(*args)
498 when 'ASSIGN'
499 assignment(args[0], args[1].to_i, args.drop(2))
500 end
501 rescue FlowError => error
502 puts "ERROR #{error.message}"
503 end

505 def flow_command(command, args, source, target)
506 prefix = "#{command} #{args[0]} #{args[1]}"
507 if command == 'SENSITIVITY'
508 maximum = lambda do |index, delta|
509 work = fresh
510 work.edges[index].capacity += delta if index
511 work.maxflow(source, target)
512 work.balance[source]
513 end
514 value = maximum.call(nil, 0)
515 critical = @network.edges.each_index.map do |i|
516 [i, maximum.call(i, 1) > value, @network.edges[i].capacity.positive? && maximum.call(i, -1) < value]
517 end
518 puts "#{prefix} value=#{value} up=#{critical.count { |r| r[1] }} down=#{critical.count { |r| r[2] }}"
519 critical.each { |i, up, down| puts "CRITICAL #{i + 1} #{@names[@network.edges[i].u]} #{@names[@network.edges[i].v]} up=#{up ? 'yes' : 'no'} down=#{down ? 'yes' : 'no'}" if up || down }
520 elsif command == 'MINCOST'
521 work = fresh
522 raise FlowError, 'NEGATIVE CYCLE' if work.negative_cycle?
523 limit = args[2] == 'MAX' ? nil : args[2].to_i
524 amount, cost, records = work.mincost(source, target, limit)
525 if limit && amount < limit
526 puts "#{prefix} k=#{args[2]} infeasible max=#{amount}"
527 return
528 end
529 @network.edges.each_with_index { |e, i| e.flow = work.edges[i].flow }
530 @owner = [:flow, source, target]
531 puts "#{prefix} k=#{args[2]} flow=#{amount} cost=#{cost}"
532 records.each_with_index { |r, i| puts "AUG #{i + 1} amount=#{r[:amount]} cost=#{r[:cost]} #{notation(work, source, r[:path])}" }
533 flow_list
534 elsif command == 'DISJOINT'
535 edges = @network.edges.map { |e| Edge.new(e.u, e.v, e.capacity.positive? ? 1 : 0, 0, 0, 0) }
536 n = @names.length
537 if args[2] == 'VERTEX'
538 exits = (0...n).to_a
539 @names.each_index do |v|
540 next if v == source || v == target
541 exits[v] = n
542 n += 1
543 end
544 edges.each { |e| e.u = exits[e.u] }
545 @names.each_index { |v| edges << Edge.new(v, exits[v], 1, 0, 0, 0) unless v == source || v == target }
546 end
547 work = FlowNetwork.new(n, edges)
548 work.maxflow(source, target)
549 records = work.decompose.select { |r| r[:kind] == 'PATH' }
550 puts "#{prefix} #{args[2]} count=#{records.length}"
551 records.each_with_index { |r, i| puts "PATH #{i + 1} #{notation(work, source, r[:path], @network.edges.length)}" }
552 else
553 start = start_flow(source, target)
554 if command == 'AUGMENT'
555 record = @network.augment(source, target)
556 ending = record ? "amount=#{record[:amount]} #{notation(@network, source, record[:path])}" : 'none'
557 puts "#{prefix} value=#{@network.balance[source]} #{ending}"
558 return
559 end
560 records, phases = @network.maxflow(source, target, command)
561 value = @network.balance[source]
562 if command == 'MINCUT'
563 reachable, = @network.bfs(source)
564 cut = @network.edges.each_index.select { |i| reachable[@network.edges[i].u] && reachable[@network.edges[i].v].nil? }
565 puts "#{prefix} value=#{value} edges=#{cut.length}"
566 puts 'SOURCE ' + @names.each_index.select { |v| reachable[v] }.map { |v| @names[v] }.join(' ')
567 puts 'SINK ' + @names.each_index.select { |v| reachable[v].nil? }.map { |v| @names[v] }.join(' ')
568 cut.each do |i|
569 e = @network.edges[i]
570 puts "CUT #{i + 1} #{@names[e.u]} #{@names[e.v]} #{e.capacity}"
571 end
572 else
573 puts "#{prefix} value=#{value} start=#{start}#{command == 'DINIC' ? " phases=#{phases}" : ''} paths=#{records.length}"
574 records.each_with_index do |r, i|
575 extra = command == 'DINIC' ? "phase=#{r[:phase]} " : command == 'SCALING' ? "delta=#{r[:delta]} " : ''
576 puts "AUG #{i + 1} #{extra}amount=#{r[:amount]} #{notation(@network, source, r[:path])}"
577 end
578 flow_list
579 end
580 end
581 end
