27 val n = Network()

36 fun node(name: String) = names.indexOf(name).takeIf {
37 it >= 0
38 }?: fail("UNKNOWN NODE $name")
39 fun endpoints(a: String, b: String): Pair<Int,Int> {
40 val s = node(a)
41 val t = node(b)
42 if (s == t) fail("SAME NODE")
43 return s to t

45 fun edgeNumber(text: String) = text.toInt().takeIf {
46 it in 1..edges.size
47 }?.minus(1) ?: fail("UNKNOWN EDGE $text")
48 fun globalArcs() = listOf(true,false).flatMap {
49 forward -> edges.indices.map {
50 Arc(it,forward)
51 }
52 }
53 fun arcs(node: Int) = globalArcs().filter {
54 start(it) == node
55 }
56 fun start(a: Arc) = if (a.forward) edges[a.edge].from else edges[a.edge].to
57 fun end(a: Arc) = if (a.forward) edges[a.edge].to else edges[a.edge].from
58 fun residual(a: Arc) = edges[a.edge].let {
59 if (a.forward) it.capacity-it.flow else it.flow
60 }
61 fun arcCost(a: Arc) = if (a.forward) edges[a.edge].cost else -edges[a.edge].cost
62 fun push(path: List<Arc>, amount: Long) {
63 for (a in path) edges[a.edge].flow += if (a.forward) amount else -amount
64 }
65 fun bottleneck(path: List<Arc>) = path.minOf {
66 residual(it)
67 }
68 fun balances(): LongArray {
69 val b = LongArray(names.size)
70 for (e in edges) {
71 b[e.from] += e.flow
72 b[e.to] -= e.flow
73 }
74 return b
75 }
76 fun cost() = edges.sumOf {
77 it.flow*it.cost
78 }
79 fun zero() {
80 for (e in edges) e.flow = 0
81 }
82 fun begin(s: Int, t: Int): Long {
83 if (owner != FlowOwner(s,t)) zero()
84 owner = FlowOwner(s,t)
85 return balances()[s]

87 fun bfs(s: Int, delta: Long = 1): Pair<IntArray,List<Arc?>> {
88 val level = IntArray(names.size) {
89 -1
90 }
91 val parent = MutableList<Arc?>(names.size) {
92 null
93 }
94 val queue = ArrayDeque<Int>()
95 queue.addLast(s)
96 level[s] = 0
97 while (queue.isNotEmpty()) {
98 val u = queue.removeFirst()
99 for (a in arcs(u)) {
100 val v = end(a)
101 if (residual(a) >= delta && level[v] < 0) {
102 level[v] = level[u]+1
103 parent[v] = a
104 queue.addLast(v)
105 }
106 }
107 }
108 return level to parent
109 }
110 fun path(s: Int, t: Int, parent: List<Arc?>): List<Arc> {
111 val path = mutableListOf<Arc>()
112 var at = t
113 while (at != s) {
114 val a = parent[at]!!
115 path.add(a)
116 at = start(a)
117 }
118 return path.asReversed()
119 }
120 fun augment(s: Int, t: Int, delta: Long = 1): Pushed? {
121 val (level,parent) = bfs(s,delta)
122 if (level[t] < 0) return null
123 val arcs = path(s,t,parent)
124 val amount = bottleneck(arcs)
125 push(arcs,amount)
126 return Pushed(arcs,amount,delta=delta)
127 }
128 fun maxflow(s: Int, t: Int): List<Pushed> {
129 val result = mutableListOf<Pushed>()
130 while (true) result.add(augment(s,t) ?: break)
131 return result
132 }
133 fun dinic(s: Int, t: Int): Pair<Int,List<Pushed>> {
134 var phases = 0
135 val pushed = mutableListOf<Pushed>()
136 while (true) {
137 val (level,_) = bfs(s)
138 if (level[t] < 0) return phases to pushed
139 phases++
140 val lists = names.indices.map {
141 arcs(it)
142 }
143 val pointers = IntArray(names.size)
144 var path = mutableListOf<Arc>()
145 var u = s
146 while (true) {
147 if (u == t) {
148 val amount = bottleneck(path)
149 push(path,amount)
150 pushed.add(Pushed(path.toList(),amount,phase=phases))
151 path = mutableListOf()
152 u = s
153 } else if (pointers[u] == lists[u].size) {
154 if (u == s) break
155 val a = path.removeAt(path.lastIndex)
156 u = start(a)
157 pointers[u]++
158 } else {
159 val a = lists[u][pointers[u]]
160 val v = end(a)
161 if (residual(a) > 0 && level[v] >= 0 && level[v] == level[u]+1) {
162 path.add(a)
163 u = v
164 } else pointers[u]++
165 }
166 }
167 }
168 }
169 fun scaling(s: Int, t: Int): List<Pushed> {
170 val capacity = edges.maxOfOrNull {
171 it.capacity
172 }?: 0
173 var delta = 1L
174 while (delta*2 <= capacity) delta *= 2
175 val paths = mutableListOf<Pushed>()
176 while (delta >= 1) {
177 while (true) paths.add(augment(s,t,delta) ?: break)
178 delta /= 2
179 }
180 return paths
181 }
182 fun negativeCycle(residualLower: Boolean): Boolean {
183 val distance = LongArray(names.size)
184 repeat(names.size) {
185 pass -> var changed = false
186 for (e in edges) if (e.capacity-(if (residualLower) e.lower else 0) > 0 && distance[e.to] > distance[e.from]+e.cost) {
187 distance[e.to] = distance[e.from]+e.cost
188 changed = true
189 }
190 if (!changed) return false
191 if (pass == names.lastIndex) return true
192 }
193 return false
194 }
195 fun shortest(s: Int): Pair<List<Long?>,List<Arc?>> {
196 val arcs = globalArcs().filter {
197 residual(it) > 0
198 }
199 val distance = MutableList<Long?>(names.size) {
200 null
201 }
202 val parent = MutableList<Arc?>(names.size) {
203 null
204 }
205 distance[s] = 0
206 while (true) {
207 var changed = false
208 for (a in arcs) {
209 val u = start(a)
210 val v = end(a)
211 val d = distance[u] ?: continue
212 val next = d+arcCost(a)
213 if (distance[v] == null || next < distance[v]!!) {
214 distance[v] = next
215 parent[v] = a
216 changed = true
217 }
218 }
219 if (!changed) return distance to parent
220 }
221 }
222 fun mincost(s: Int, t: Int, target: Long? = null): Triple<Long,Long,List<Pushed>> {
223 var amount = 0L
224 var cost = 0L
225 val paths = mutableListOf<Pushed>()
226 while (target == null || amount < target) {
227 val (distance,parent) = shortest(s)
228 val unit = distance[t] ?: break
229 val arcs = path(s,t,parent)
230 var delta = bottleneck(arcs)
231 if (target != null) delta = minOf(delta,target-amount)
232 push(arcs,delta)
233 amount += delta
234 cost += delta*unit
235 paths.add(Pushed(arcs,delta,cost=unit))
236 }
237 return Triple(amount,cost,paths)
238 }
239 fun notation(start: Int, path: List<Arc>) = names[start]+path.joinToString("") {
240 " ${if (it.forward) "+" else "-"}${it.edge+1} ${names[end(it)]}"
241 }
242 fun flows() = edges.mapIndexed {
243 i,e -> "FLOW ${i+1} ${names[e.from]} ${names[e.to]} ${e.flow}/${e.capacity}"
244 }
245 fun decompose(): List<Component> {
246 val g = edges.map {
247 it.flow
248 }.toLongArray()
249 val balance = balances()
250 val result = mutableListOf<Component>()
251 while (g.any {
252 it > 0
253 }) {
254 val first = g.indexOfFirst {
255 it > 0
256 }
257 val start = balance.indexOfFirst {
258 it > 0
259 }.takeIf {
260 it >= 0
261 }?: edges[first].from
262 val vertices = mutableListOf(start)
263 val path = mutableListOf<Int>()
264 while (true) {
265 val u = vertices.last()
266 val i = edges.indices.first {
267 edges[it].from == u && g[it] > 0
268 }
269 val v = edges[i].to
270 path.add(i)
271 val q = vertices.indexOf(v)
272 if (q >= 0) {
273 val cycle = path.drop(q)
274 val delta = cycle.minOf {
275 g[it]
276 }
277 for (e in cycle) g[e] -= delta
278 result.add(Component(v,cycle,delta,true))
279 break
280 }
281 vertices.add(v)
282 if (balance[v] < 0) {
283 val delta = minOf(path.minOf {
284 g[it]
285 },balance[start],-balance[v])
286 for (e in path) g[e] -= delta
287 balance[start] -= delta
288 balance[v] += delta
289 result.add(Component(start,path.toList(),delta,false))
290 break
291 }
292 }
293 }
294 return result
295 }
296 fun command(w: List<String>): List<String> {
297 val c = w[0]
298 val a = w.drop(1)
299 fun num(i: Int) = a[i].toLong()
300 val out = mutableListOf<String>()
301 when(c) {
302 "NODE" -> {
303 val seen = names.toMutableSet()
304 for (name in a) if (!seen.add(name)) fail("DUPLICATE NODE $name")
305 if (names.size+a.size > 20) fail("TOO MANY NODES")
306 names.addAll(a)
307 repeat(a.size) {
308 demands.add(0)
309 }
310 out.add("NODE added=${a.size} total=${names.size}")
311 }
312 "EDGE" -> {
313 val u = node(a[0])
314 val v = node(a[1])
315 if (u == v) fail("SELF LOOP")
316 if (edges.size == 60) fail("TOO MANY EDGES")
317 val cost = if (a.size == 4) num(3) else 0L
318 edges.add(Edge(u,v,num(2),cost))
319 out.add("EDGE ${edges.size} ${a[0]} ${a[1]} cap=${num(2)} cost=$cost")
320 }
321 "LOWER","CAP","COST" -> {
322 val e = edges[edgeNumber(a[0])]
323 val n = num(1)
324 when(c) {
325 "LOWER" -> {
326 if (n > e.capacity) fail("BAD BOUND")
327 e.lower = n
328 out.add("LOWER ${a[0]} low=$n cap=${e.capacity}")
329 }
330 "CAP" -> {
331 if (n < e.lower) fail("BAD BOUND")
332 if (n < e.flow) fail("BELOW FLOW")
333 e.capacity = n
334 out.add("CAP ${a[0]} low=${e.lower} cap=$n")
335 } else -> {
336 e.cost = n
337 out.add("COST ${a[0]} cost=$n")
338 }
339 }
340 }
341 "DEMAND" -> {
342 demands[node(a[0])] = num(1)
343 out.add("DEMAND ${a[0]} ${num(1)} total=${demands.sum()}")
344 }
345 "RESET" -> {
346 out.add("RESET edges=${edges.count { it.flow != 0L }}")
347 zero()
348 owner = NoOwner
349 }
350 "CLEAR" -> {
351 out.add("CLEAR nodes=${names.size} edges=${edges.size}")
352 names.clear()
353 demands.clear()
354 edges.clear()
355 owner = NoOwner
356 }
357 "SHOW" -> {
358 val balance = balances()
359 out.add("NETWORK nodes=${names.size} edges=${edges.size}")
360 out.add(when(val owner = owner) {
361 NoOwner -> "STATE none"
362 CirculationOwner -> "STATE circulation cost=${cost()}"
363 is FlowOwner -> "STATE flow ${names[owner.source]} ${names[owner.sink]} value=${balance[owner.source]} cost=${cost()}"
364 })
365 for ((i,name) in names.withIndex()) out.add("NODE $name demand=${demands[i]} out=${edges.filter { it.from == i }.sumOf { it.flow }} in=${edges.filter { it.to == i }.sumOf { it.flow }}")
366 for ((i,e) in edges.withIndex()) out.add("EDGE ${i+1} ${names[e.from]} ${names[e.to]} cap=${e.capacity} cost=${e.cost} low=${e.lower} flow=${e.flow}")
367 }
368 "AUGMENT","MAXFLOW","DINIC","SCALING","MINCUT" -> {
369 val (s,t) = endpoints(a[0],a[1])
370 val start = begin(s,t)
371 if (c == "AUGMENT") {
372 val p = augment(s,t)
373 out.add(if (p == null) "AUGMENT ${a[0]} ${a[1]} value=${balances()[s]} none" else "AUGMENT ${a[0]} ${a[1]} value=${balances()[s]} amount=${p.amount} ${notation(s,p.arcs)}")
374 } else if (c == "MINCUT") {
375 maxflow(s,t)
376 val (level,_) = bfs(s)
377 val cut = edges.indices.filter {
378 level[edges[it].from] >= 0 && level[edges[it].to] < 0
379 }
380 out.add("MINCUT ${a[0]} ${a[1]} value=${balances()[s]} edges=${cut.size}")
381 out.add("SOURCE "+names.indices.filter {
382 level[it] >= 0
383 }.joinToString(" ") {
384 names[it]
385 })
386 out.add("SINK "+names.indices.filter {
387 level[it] < 0
388 }.joinToString(" ") {
389 names[it]
390 })
391 for (i in cut) {
392 val e = edges[i]
393 out.add("CUT ${i+1} ${names[e.from]} ${names[e.to]} ${e.capacity}")
394 }
395 } else {
396 val (phases,paths) = when(c) {
397 "DINIC" -> dinic(s,t)
398 "SCALING" -> 0 to scaling(s,t)
399 else -> 0 to maxflow(s,t)
400 }
401 out.add("$c ${a[0]} ${a[1]} value=${balances()[s]} start=$start${if (c == "DINIC") " phases=$phases" else ""} paths=${paths.size}")
402 for ((i,p) in paths.withIndex()) out.add("AUG ${i+1}${if (c == "DINIC") " phase=${p.phase}" else if (c == "SCALING") " delta=${p.delta}" else ""} amount=${p.amount} ${notation(s,p.arcs)}")
403 out.addAll(flows())
404 }
405 }
406 "SENSITIVITY" -> {
407 val (s,t) = endpoints(a[0],a[1])
408 val base = copy()
409 base.zero()
410 base.maxflow(s,t)
411 val value = base.balances()[s]
412 var up = 0
413 var down = 0
414 val details = mutableListOf<String>()
415 for (i in edges.indices) {
416 val work = copy()
417 work.zero()
418 work.edges[i].capacity++
419 work.maxflow(s,t)
420 val effectiveUp = work.balances()[s] > value
421 var effectiveDown = false
422 if (edges[i].capacity > 0) {
423 val reduced = copy()
424 reduced.zero()
425 reduced.edges[i].capacity--
426 reduced.maxflow(s,t)
427 effectiveDown = reduced.balances()[s] < value
428 }
429 if (effectiveUp) up++
430 if (effectiveDown) down++
431 if (effectiveUp || effectiveDown) {
432 val e = edges[i]
433 details.add("CRITICAL ${i+1} ${names[e.from]} ${names[e.to]} up=${if (effectiveUp) "yes" else "no"} down=${if (effectiveDown) "yes" else "no"}")
434 }
435 }
436 out.add("SENSITIVITY ${a[0]} ${a[1]} value=$value up=$up down=$down")
437 out.addAll(details)
438 }
439 "MINCOST" -> {
440 val (s,t) = endpoints(a[0],a[1])
441 if (negativeCycle(false)) fail("NEGATIVE CYCLE")
442 val target = if (a[2] == "MAX") null else num(2)
443 val work = copy()
444 work.zero()
445 val (amount,cost,paths) = work.mincost(s,t,target)
446 if (target != null && amount < target) out.add("MINCOST ${a[0]} ${a[1]} k=${a[2]} infeasible max=$amount") else {
447 for (i in edges.indices) edges[i].flow = work.edges[i].flow
448 owner = FlowOwner(s,t)
449 out.add("MINCOST ${a[0]} ${a[1]} k=${a[2]} flow=$amount cost=$cost")
450 for ((i,p) in paths.withIndex()) out.add("AUG ${i+1} amount=${p.amount} cost=${p.cost} ${notation(s,p.arcs)}")
451 out.addAll(flows())
452 }
453 }
454 "DISJOINT" -> {
455 val (s,t) = endpoints(a[0],a[1])
456 val original = edges.size
457 val work = copy()
458 work.zero()
459 val exits = IntArray(names.size) {
460 it
461 }
462 if (a[2] == "VERTEX") for (v in names.indices) if (v != s && v != t) {
463 exits[v] = work.names.size
464 work.names.add(names[v])
465 work.demands.add(0)
466 }
467 for (i in work.edges.indices) {
468 val e = work.edges[i]
469 work.edges[i] = e.copy(from=exits[e.from],capacity=if (e.capacity > 0) 1 else 0,cost=0)
470 }
471 if (a[2] == "VERTEX") for (v in names.indices) if (v != s && v != t) work.edges.add(Edge(v,exits[v],1))
472 work.maxflow(s,t)
473 val paths = work.decompose().filter {
474 !it.cycle
475 }
476 out.add("DISJOINT ${a[0]} ${a[1]} ${a[2]} count=${paths.size}")
477 for ((i,p) in paths.withIndex()) out.add("PATH ${i+1} ${notation(s,p.edges.filter { it < original }.map { Arc(it,true) })}")
478 }
479 "DECOMPOSE" -> {
480 val parts = decompose()
481 out.add("DECOMPOSE paths=${parts.count { !it.cycle }} cycles=${parts.count { it.cycle }}")
482 val counts = IntArray(2)
483 for (p in parts) {
484 val kind = if (p.cycle) 1 else 0
485 counts[kind]++
486 out.add("${if (p.cycle) "CYCLE" else "PATH"} ${counts[kind]} amount=${p.amount} ${notation(p.start,p.edges.map { Arc(it,true) })}")
487 }
488 }
489 "VERIFY" -> {
490 val violations = mutableListOf<String>()
491 for ((i,e) in edges.withIndex()) if (e.flow < e.lower) violations.add("LOW ${i+1} flow=${e.flow} low=${e.lower}")
492 val balance = balances()
493 for (i in names.indices) {
494 val owner = owner
495 val need = if (owner is FlowOwner) {
496 if (i == owner.source || i == owner.sink) null else 0L
497 } else demands[i]
498 if (need != null && -balance[i] != need) violations.add("BALANCE ${names[i]} net=${-balance[i]} need=$need")
499 }
500 if (violations.isEmpty()) out.add("VERIFY ok") else {
501 out.add("VERIFY violations=${violations.size}")
502 out.addAll(violations)
503 }
504 }
505 "CIRCULATION" -> {
506 if (a[0] == "MINCOST" && negativeCycle(true)) fail("NEGATIVE CYCLE")
507 val sum = demands.sum()
508 if (sum != 0L) return listOf("CIRCULATION infeasible unbalanced=$sum")
509 val n = names.size
510 val work = copy()
511 work.names.addAll(listOf("super source","super sink"))
512 work.demands.addAll(listOf(0L,0L))
513 val required = demands.toLongArray()
514 for (e in work.edges) {
515 required[e.to] -= e.lower
516 required[e.from] += e.lower
517 e.capacity -= e.lower
518 e.flow = 0
519 }
520 for (v in required.indices) if (required[v] < 0) work.edges.add(Edge(n,v,-required[v]))
521 val sinkStart = work.edges.size
522 for (v in required.indices) if (required[v] > 0) work.edges.add(Edge(v,n+1,required[v]))
523 if (a[0] == "ANY") work.maxflow(n,n+1) else work.mincost(n,n+1)
524 val total = required.filter {
525 it > 0
526 }.sum()
527 val achieved = work.edges.drop(sinkStart).sumOf {
528 it.flow
529 }
530 if (achieved < total) {
531 val (level,_) = work.bfs(n)
532 out.add("CIRCULATION infeasible required=$total achieved=$achieved")
533 out.add("CUT "+names.indices.filter {
534 level[it] >= 0
535 }.joinToString(" ") {
536 names[it]
537 })
538 } else {
539 for (i in edges.indices) edges[i].flow = edges[i].lower+work.edges[i].flow
540 owner = CirculationOwner
541 out.add("CIRCULATION feasible required=$total cost=${cost()}")
542 out.addAll(flows())
543 }
544 }
545 "MATCH" -> out.addAll(matching(a))
546 "ASSIGN" -> out.addAll(assignment(a))
547 }
548 return out
549 }

552private fun matching(a: List<String>): List<String> {
553 val left = a[0].split(',')
554 val right = a[1].split(',')
555 val seen = mutableSetOf<String>()
556 for (n in left+right) if (!seen.add(n)) fail("DUPLICATE VERTEX $n")
557 val pairs = mutableListOf<Pair<Int,Int>>()
558 if (a[2] != "-") for (p in a[2].split(',')) {
559 val words = p.split('-')
560 val u = left.indexOf(words[0]).takeIf {
561 it >= 0
562 }?: fail("UNKNOWN VERTEX ${words[0]}")
563 val v = right.indexOf(words[1]).takeIf {
564 it >= 0
565 }?: fail("UNKNOWN VERTEX ${words[1]}")
566 if ((u to v) in pairs) fail("DUPLICATE PAIR $p")
567 pairs.add(u to v)
568 }
569 val lp = IntArray(left.size) {
570 -1
571 }
572 val rp = IntArray(right.size) {
573 -1
574 }
575 fun attempt(u: Int, examined: BooleanArray): Boolean {
576 for ((x,b) in pairs) if (x == u && !examined[b]) {
577 examined[b] = true
578 if (rp[b] < 0 || attempt(rp[b],examined)) {
579 lp[u] = b
580 rp[b] = u
581 return true
582 }
583 }
584 return false
585 }
586 for (u in left.indices) attempt(u,BooleanArray(right.size))
587 val zl = BooleanArray(left.size) {
588 lp[it] < 0
589 }
590 val zr = BooleanArray(right.size)
591 while (true) {
592 var changed = false
593 for ((u,v) in pairs) if (zl[u] && lp[u] != v && !zr[v]) {
594 zr[v] = true
595 changed = true
596 }
597 for (v in right.indices) if (zr[v] && rp[v] >= 0 && !zl[rp[v]]) {
598 zl[rp[v]] = true
599 changed = true
600 }
601 if (!changed) break
602 }
603 val size = lp.count {
604 it >= 0
605 }
606 val out = mutableListOf("MATCH size=$size left=${left.size} right=${right.size}")
607 for (u in left.indices) if (lp[u] >= 0) out.add("PAIR ${left[u]} ${right[lp[u]]}")
608 fun list(values: List<String>) = values.joinToString(",").ifEmpty {
609 "-"
610 }
611 out.add("FREE ${list(left.indices.filter { lp[it] < 0 }.map { left[it] })} ${list(right.indices.filter { rp[it] < 0 }.map { right[it] })}")
612 val cover = left.indices.filter {
613 !zl[it]
614 }.map {
615 left[it]
616 }+right.indices.filter {
617 zr[it]
618 }.map {
619 right[it]
620 }
621 out.add("COVER size=${cover.size}"+cover.joinToString("") {
622 " $it"
623 })
624 if (lp.any {
625 it < 0
626 }) out.add("HALL ${list(left.indices.filter { zl[it] }.map { left[it] })} ${list(right.indices.filter { zr[it] }.map { right[it] })}")
627 return out
628}

630private fun assignment(a: List<String>): List<String> {
631 val n = a[1].toInt()
632 if (n !in 1..8) fail("BAD SIZE")
633 if (a.size-2 != n*n) fail("BAD LENGTH")
634 val values = a.drop(2).map {
635 if (it == "x") null else it.toLong()
636 }
637 var best: Long? = null
638 var count = 0
639 var first = emptyList<Int>()
640 val chosen = mutableListOf<Int>()
641 fun visit(used: Int, sum: Long) {
642 val row = chosen.size
643 if (row == n) {
644 if (best == null || (if (a[0] == "MAX") sum > best!! else sum < best!!)) {
645 best = sum
646 count = 1
647 first = chosen.toList()
648 } else if (sum == best) count++
649 return
650 }
651 for (col in 0 until n) if (used and (1 shl col) == 0) {
652 val v = values[row*n+col] ?: continue
653 chosen.add(col)
654 visit(used or (1 shl col),sum+v)
655 chosen.removeAt(chosen.lastIndex)
656 }
657 }
658 visit(0,0)
659 if (best == null) return listOf("ASSIGN ${a[0]} n=$n infeasible")
660 return listOf("ASSIGN ${a[0]} n=$n total=$best optimal=$count")+first.mapIndexed {
661 row,col -> "MATE ${row+1} ${col+1} ${values[row*n+col]}"
662 }
663}
