23class Graph
24{
25 public List<string> Names = [];
26 public List<Edge> Edges = [];
27
28 public Graph Copy(bool keepFlow)
29 {
30 Graph g = new();
31 g.Names.AddRange(Names);
32 foreach (Edge e in Edges)
33 {
34 Edge f = e.Copy();
35 if (!keepFlow) f.Flow = 0;
36 g.Edges.Add(f);
37 }
38 return g;
39 }
40
41 public int Add(int u, int v, int capacity, int cost)
42 {
43 Edges.Add(new(u, v, capacity, cost));
44 return Edges.Count;
45 }
46 public Edge EdgeOf(int arc) => Edges[Math.Abs(arc) - 1];
47 public int From(int arc) => arc > 0 ? EdgeOf(arc).U : EdgeOf(arc).V;
48 public int To(int arc) => arc > 0 ? EdgeOf(arc).V : EdgeOf(arc).U;
49 public int Residual(int arc) => arc > 0 ? EdgeOf(arc).Capacity - EdgeOf(arc).Flow : EdgeOf(arc).Flow;
50 public int Cost(int arc) => arc > 0 ? EdgeOf(arc).Cost : -EdgeOf(arc).Cost;
51
52 public List<int> Arcs(int u)
53 {
54 List<int> result = [];
55 for (int i = 0; i < Edges.Count; i++) if (Edges[i].U == u) result.Add(i + 1);
56 for (int i = 0; i < Edges.Count; i++) if (Edges[i].V == u) result.Add(-i - 1);
57 return result;
58 }
59
60 public int[] Balances()
61 {
62 int[] result = new int[Names.Count];
63 foreach (Edge e in Edges)
64 {
65 result[e.U] += e.Flow;
66 result[e.V] -= e.Flow;
67 }
68 return result;
69 }
70 public int TotalCost() => Edges.Sum(e => e.Flow * e.Cost);
71 public int Bottleneck(List<int> path) => path.Min(Residual);
72
73 public void Push(List<int> path, int amount)
74 {
75 foreach (int a in path) EdgeOf(a).Flow += a > 0 ? amount : -amount;
76 }
77 public string Notation(int start, List<int> path) => Names[start] + string.Concat(path.Select(a => $" {(a > 0 ? "+" : "-")}{Math.Abs(a)} {Names[To(a)]}"));
78
79 public Search Bfs(int source, int threshold)
80 {
81 Search search = new(Names.Count);
82 Queue<int> queue = new();
83 search.Level[source] = 0;
84 queue.Enqueue(source);
85 while (queue.Count != 0)
86 {
87 int u = queue.Dequeue();
88 foreach (int a in Arcs(u))
89 {
90 int v = To(a);
91 if (Residual(a) >= threshold && search.Level[v] < 0)
92 {
93 search.Level[v] = search.Level[u] + 1;
94 search.Parent[v] = a;
95 queue.Enqueue(v);
96 }
97 }
98 }
99 return search;
100 }
101
102 public List<int> Path(int source, int target, int[] parent)
103 {
104 List<int> result = [];
105 for (int v = target; v != source; v = From(parent[v])) result.Add(parent[v]);
106 result.Reverse();
107 return result;
108 }
109
110 public List<Augmentation> Maximum(int source, int target, string method)
111 {
112 List<Augmentation> result = [];
113 if (method == "DINIC")
114 {
115 int phase = 0;
116 while (true)
117 {
118 Search search = Bfs(source, 1);
119 if (search.Level[target] < 0) break;
120 phase++;
121 List<int>[] lists = Enumerable.Range(0, Names.Count).Select(Arcs).ToArray();
122 int[] pointer = new int[Names.Count];
123 List<int> path = [];
124 int u = source;
125 while (true)
126 {
127 if (u == target)
128 {
129 int amount = Bottleneck(path);
130 result.Add(new([.. path], amount, phase, 0));
131 Push(path, amount);
132 path.Clear();
133 u = source;
134 }
135 else if (pointer[u] == lists[u].Count)
136 {
137 if (u == source) break;
138 u = From(path[^1]);
139 path.RemoveAt(path.Count - 1);
140 pointer[u]++;
141 }
142 else
143 {
144 int a = lists[u][pointer[u]], v = To(a);
145 if (Residual(a) > 0 && search.Level[v] == search.Level[u] + 1)
146 {
147 path.Add(a);
148 u = v;
149 }
150 else pointer[u]++;
151 }
152 }
153 }
154 return result;
155 }
156 int threshold = 1;
157 if (method == "SCALING")
158 {
159 int largest = Edges.Select(e => e.Capacity).DefaultIfEmpty().Max();
160 while (threshold <= largest / 2) threshold *= 2;
161 }
162 do
163 {
164 while (true)
165 {
166 Search search = Bfs(source, threshold);
167 if (search.Level[target] < 0) break;
168 List<int> path = Path(source, target, search.Parent);
169 int amount = Bottleneck(path);
170 Push(path, amount);
171 result.Add(new(path, amount, threshold, 0));
172 if (method == "AUGMENT") return result;
173 }
174 threshold /= 2;
175 }
176 while (threshold > 0);
177 return result;
178 }

180 public bool NegativeCycle(bool subtractLower)
181 {
182 int[] distance = new int[Names.Count];
183 for (int pass = 0; pass < Names.Count; pass++)
184 {
185 bool changed = false;
186 foreach (Edge e in Edges) if (e.Capacity - (subtractLower ? e.Lower : 0) > 0 && distance[e.V] > distance[e.U] + e.Cost)
187 {
188 distance[e.V] = distance[e.U] + e.Cost;
189 changed = true;
190 }
191 if (!changed) return false;
192 }
193 return Names.Count != 0;
194 }
195
196 public List<Augmentation> Cheapest(int source, int target, int wanted)
197 {
198 List<Augmentation> result = [];
199 int pushed = 0;
200 while (wanted < 0 || pushed < wanted)
201 {
202 List<int> usable = [];
203 for (int i = 1; i <= Edges.Count; i++) if (Residual(i) > 0) usable.Add(i);
204 for (int i = 1; i <= Edges.Count; i++) if (Residual(-i) > 0) usable.Add(-i);
205 int?[] distance = new int?[Names.Count];
206 int[] parent = new int[Names.Count];
207 distance[source] = 0;
208 bool changed;
209 do
210 {
211 changed = false;
212 foreach (int a in usable)
213 {
214 int u = From(a), v = To(a);
215 if (distance[u].HasValue && (!distance[v].HasValue || distance[u]!.Value + Cost(a) < distance[v]!.Value))
216 {
217 distance[v] = distance[u]!.Value + Cost(a);
218 parent[v] = a;
219 changed = true;
220 }
221 }
222 }
223 while (changed);
224 if (!distance[target].HasValue) break;
225 List<int> path = Path(source, target, parent);
226 int amount = Bottleneck(path);
227 if (wanted >= 0) amount = Math.Min(amount, wanted - pushed);
228 Push(path, amount);
229 pushed += amount;
230 result.Add(new(path, amount, 0, distance[target]!.Value));
231 }
232 return result;
233 }
234
235 public List<Piece> Decompose()
236 {
237 int[] remaining = Edges.Select(e => e.Flow).ToArray(), balance = Balances();
238 List<Piece> result = [];
239 while (remaining.Any(x => x > 0))
240 {
241 int start = Array.FindIndex(balance, x => x > 0);
242 if (start < 0) start = Edges[Array.FindIndex(remaining, x => x > 0)].U;
243 List<int> vertices = [start], path = [];
244 while (true)
245 {
246 int u = vertices[^1], i = 0;
247 while (Edges[i].U != u || remaining[i] == 0) i++;
248 int v = Edges[i].V;
249 path.Add(i + 1);
250 int repeated = vertices.IndexOf(v);
251 if (repeated >= 0)
252 {
253 List<int> cycle = path.GetRange(repeated, path.Count - repeated);
254 int amount = cycle.Min(a => remaining[a - 1]);
255 foreach (int a in cycle) remaining[a - 1] -= amount;
256 result.Add(new(true, v, cycle, amount));
257 break;
258 }
259 vertices.Add(v);
260 if (balance[v] < 0)
261 {
262 int amount = Math.Min(Math.Min(balance[start], -balance[v]), path.Min(a => remaining[a - 1]));
263 foreach (int a in path) remaining[a - 1] -= amount;
264 balance[start] -= amount;
265 balance[v] += amount;
266 result.Add(new(false, start, path, amount));
267 break;
268 }
269 }
270 }
271 return result;
272 }

320 List<string> Command(string line)
321 {
322 string[] w = line.Split(' ');
323 string c = w[0];
324 List<string> output = [];
325 switch (c)
326 {
327 case "NODE":
328 {
329 HashSet<string> seen = new(network.Names, StringComparer.Ordinal);
330 foreach (string name in w.Skip(1)) if (!seen.Add(name)) Fail("DUPLICATE NODE " + name);
331 if (network.Names.Count + w.Length - 1 > 20) Fail("TOO MANY NODES");
332 foreach (string name in w.Skip(1))
333 {
334 network.Names.Add(name);
335 demand.Add(0);
336 }
337 output.Add($"NODE added={w.Length - 1} total={network.Names.Count}");
338 break;
339 }
340 case "EDGE":
341 {
342 int u = Node(w[1]), v = Node(w[2]);
343 if (u == v) Fail("SELF LOOP");
344 if (network.Edges.Count == 60) Fail("TOO MANY EDGES");
345 int capacity = int.Parse(w[3]), cost = w.Length == 5 ? int.Parse(w[4]) : 0;
346 int number = network.Add(u, v, capacity, cost);
347 output.Add($"EDGE {number} {w[1]} {w[2]} cap={capacity} cost={cost}");
348 break;
349 }
350 case "LOWER":
351 case "CAP":
352 case "COST":
353 {
354 Edge e = CheckedEdge(w[1]);
355 int value = int.Parse(w[2]);
356 if (c == "LOWER")
357 {
358 if (value > e.Capacity) Fail("BAD BOUND");
359 e.Lower = value;
360 output.Add($"LOWER {w[1]} low={value} cap={e.Capacity}");
361 }
362 else if (c == "CAP")
363 {
364 if (value < e.Lower) Fail("BAD BOUND");
365 if (value < e.Flow) Fail("BELOW FLOW");
366 e.Capacity = value;
367 output.Add($"CAP {w[1]} low={e.Lower} cap={value}");
368 }
369 else
370 {
371 e.Cost = value;
372 output.Add($"COST {w[1]} cost={value}");
373 }
374 break;
375 }
376 case "DEMAND":
377 {
378 int u = Node(w[1]), value = int.Parse(w[2]);
379 demand[u] = value;
380 output.Add($"DEMAND {w[1]} {value} total={demand.Sum()}");
381 break;
382 }
383 case "RESET":
384 {
385 int count = network.Edges.Count(e => e.Flow != 0);
386 foreach (Edge e in network.Edges) e.Flow = 0;
387 owner = 0;
388 output.Add($"RESET edges={count}");
389 break;
390 }
391 case "CLEAR":
392 output.Add($"CLEAR nodes={network.Names.Count} edges={network.Edges.Count}");
393 network = new();
394 demand.Clear();
395 owner = 0;
396 break;
397 case "SHOW":
398 {
399 output.Add($"NETWORK nodes={network.Names.Count} edges={network.Edges.Count}");
400 output.Add(owner == 0 ? "STATE none" : owner == 2 ? $"STATE circulation cost={network.TotalCost()}" : $"STATE flow {network.Names[source]} {network.Names[target]} value={network.Balances()[source]} cost={network.TotalCost()}");
401 for (int u = 0; u < network.Names.Count; u++) output.Add($"NODE {network.Names[u]} demand={demand[u]} out={network.Edges.Where(e => e.U == u).Sum(e => e.Flow)} in={network.Edges.Where(e => e.V == u).Sum(e => e.Flow)}");
402 output.AddRange(network.Edges.Select((e, i) => $"EDGE {i + 1} {network.Names[e.U]} {network.Names[e.V]} cap={e.Capacity} cost={e.Cost} low={e.Lower} flow={e.Flow}"));
403 break;
404 }
405 case "AUGMENT":
406 case "MAXFLOW":
407 case "DINIC":
408 case "SCALING":
409 case "MINCUT":
410 {
411 (int s, int t) = Endpoints(w[1], w[2]);
412 int initial = Start(s, t);
413 List<Augmentation> paths = network.Maximum(s, t, c);
414 string heading = $"{c} {w[1]} {w[2]} value={network.Balances()[s]}";
415 if (c == "AUGMENT") output.Add(heading + (paths.Count == 0 ? " none" : $" amount={paths[0].Amount} {network.Notation(s, paths[0].Path)}"));
416 else if (c == "MINCUT")
417 {
418 Search search = network.Bfs(s, 1);
419 List<int> cut = Enumerable.Range(0, network.Edges.Count).Where(i => search.Level[network.Edges[i].U] >= 0 && search.Level[network.Edges[i].V] < 0).ToList();
420 output.Add($"{heading} edges={cut.Count}");
421 foreach (bool reachable in new[] { true, false }) output.Add((reachable ? "SOURCE" : "SINK") + string.Concat(Enumerable.Range(0, network.Names.Count).Where(u => (search.Level[u] >= 0) == reachable).Select(u => " " + network.Names[u])));
422 foreach (int i in cut)
423 {
424 Edge e = network.Edges[i];
425 output.Add($"CUT {i + 1} {network.Names[e.U]} {network.Names[e.V]} {e.Capacity}");
426 }
427 }
428 else
429 {
430 output.Add($"{heading} start={initial}" + (c == "DINIC" ? $" phases={(paths.Count == 0 ? 0 : paths[^1].Stage)}" : "") + $" paths={paths.Count}");
431 for (int i = 0; i < paths.Count; i++)
432 {
433 Augmentation a = paths[i];
434 output.Add($"AUG {i + 1}" + (c == "DINIC" ? $" phase={a.Stage}" : c == "SCALING" ? $" delta={a.Stage}" : "") + $" amount={a.Amount} {network.Notation(s, a.Path)}");
435 }
436 output.AddRange(Flows());
437 }
438 break;
439 }
440 case "SENSITIVITY":
441 {
442 (int s, int t) = Endpoints(w[1], w[2]);
443 Graph g = network.Copy(false);
444 g.Maximum(s, t, "MAXFLOW");
445 int value = g.Balances()[s], upCount = 0, downCount = 0;
446 List<string> critical = [];
447 for (int i = 0; i < network.Edges.Count; i++)
448 {
449 g = network.Copy(false);
450 g.Edges[i].Capacity++;
451 g.Maximum(s, t, "MAXFLOW");
452 bool up = g.Balances()[s] > value, down = false;
453 if (network.Edges[i].Capacity > 0)
454 {
455 g = network.Copy(false);
456 g.Edges[i].Capacity--;
457 g.Maximum(s, t, "MAXFLOW");
458 down = g.Balances()[s] < value;
459 }
460 if (up) upCount++;
461 if (down) downCount++;
462 Edge e = network.Edges[i];
463 if (up || down) critical.Add($"CRITICAL {i + 1} {network.Names[e.U]} {network.Names[e.V]} up={(up ? "yes" : "no")} down={(down ? "yes" : "no")}");
464 }
465 output.Add($"SENSITIVITY {w[1]} {w[2]} value={value} up={upCount} down={downCount}");
466 output.AddRange(critical);
467 break;
468 }
469 case "MINCOST":
470 {
471 (int s, int t) = Endpoints(w[1], w[2]);
472 if (network.NegativeCycle(false)) Fail("NEGATIVE CYCLE");
473 int wanted = w[3] == "MAX" ? -1 : int.Parse(w[3]);
474 Graph g = network.Copy(false);
475 List<Augmentation> paths = g.Cheapest(s, t, wanted);
476 int amount = g.Balances()[s];
477 string heading = $"MINCOST {w[1]} {w[2]} k={w[3]}";
478 if (wanted >= 0 && amount < wanted) output.Add($"{heading} infeasible max={amount}");
479 else
480 {
481 CopyFlows(g, false);
482 owner = 1;
483 source = s;
484 target = t;
485 output.Add($"{heading} flow={amount} cost={g.TotalCost()}");
486 for (int i = 0; i < paths.Count; i++)
487 {
488 Augmentation a = paths[i];
489 output.Add($"AUG {i + 1} amount={a.Amount} cost={a.UnitCost} {g.Notation(s, a.Path)}");
490 }
491 output.AddRange(Flows());
492 }
493 break;
494 }
495 case "DISJOINT":
496 {
497 (int s, int t) = Endpoints(w[1], w[2]);
498 int n = network.Names.Count;
499 Graph g = new();
500 g.Names.AddRange(network.Names);
501 int[] exit = new int[n];
502 for (int u = 0; u < n; u++)
503 {
504 exit[u] = u;
505 if (w[3] == "VERTEX" && u != s && u != t)
506 {
507 exit[u] = g.Names.Count;
508 g.Names.Add(network.Names[u]);
509 }
510 }
511 foreach (Edge e in network.Edges) g.Add(exit[e.U], e.V, e.Capacity == 0 ? 0 : 1, 0);
512 for (int u = 0; u < n; u++) if (exit[u] != u) g.Add(u, exit[u], 1, 0);
513 g.Maximum(s, t, "MAXFLOW");
514 List<Piece> pieces = g.Decompose().Where(p => !p.Cycle).ToList();
515 output.Add($"DISJOINT {w[1]} {w[2]} {w[3]} count={pieces.Count}");
516 for (int i = 0; i < pieces.Count; i++) output.Add($"PATH {i + 1} {w[1]}" + string.Concat(pieces[i].Path.Where(a => a <= network.Edges.Count).Select(a => $" +{a} {network.Names[network.Edges[a - 1].V]}")));
517 break;
518 }
519 case "DECOMPOSE":
520 {
521 List<Piece> pieces = network.Decompose();
522 int cycles = pieces.Count(p => p.Cycle), pathNumber = 0, cycleNumber = 0;
523 output.Add($"DECOMPOSE paths={pieces.Count - cycles} cycles={cycles}");
524 foreach (Piece p in pieces) output.Add((p.Cycle ? $"CYCLE {++cycleNumber}" : $"PATH {++pathNumber}") + $" amount={p.Amount} {network.Notation(p.Start, p.Path)}");
525 break;
526 }
527 case "VERIFY":
528 {
529 List<string> violations = [];
530 for (int i = 0; i < network.Edges.Count; i++)
531 {
532 Edge e = network.Edges[i];
533 if (e.Flow < e.Lower) violations.Add($"LOW {i + 1} flow={e.Flow} low={e.Lower}");
534 }
535 int[] balance = network.Balances();
536 for (int u = 0; u < network.Names.Count; u++) if (owner != 1 || (u != source && u != target))
537 {
538 int need = owner == 1 ? 0 : demand[u];
539 if (-balance[u] != need) violations.Add($"BALANCE {network.Names[u]} net={-balance[u]} need={need}");
540 }
541 output.Add(violations.Count == 0 ? "VERIFY ok" : $"VERIFY violations={violations.Count}");
542 output.AddRange(violations);
543 break;
544 }
545 case "CIRCULATION":
546 {
547 bool cheapest = w[1] == "MINCOST";
548 if (cheapest && network.NegativeCycle(true)) Fail("NEGATIVE CYCLE");
549 int sum = demand.Sum();
550 if (sum != 0)
551 {
552 output.Add($"CIRCULATION infeasible unbalanced={sum}");
553 break;
554 }
555 Graph g = network.Copy(false);
556 int n = g.Names.Count, required = 0;
557 g.Names.AddRange(["source", "sink"]);
558 int[] remaining = demand.ToArray();
559 foreach (Edge e in g.Edges)
560 {
561 remaining[e.U] += e.Lower;
562 remaining[e.V] -= e.Lower;
563 e.Capacity -= e.Lower;
564 }
565 for (int u = 0; u < n; u++) if (remaining[u] < 0) g.Add(n, u, -remaining[u], 0);
566 for (int u = 0; u < n; u++) if (remaining[u] > 0)
567 {
568 g.Add(u, n + 1, remaining[u], 0);
569 required += remaining[u];
570 }
571 if (cheapest) g.Cheapest(n, n + 1, -1);
572 else g.Maximum(n, n + 1, "MAXFLOW");
573 int achieved = -g.Balances()[n + 1];
574 if (achieved < required)
575 {
576 output.Add($"CIRCULATION infeasible required={required} achieved={achieved}");
577 Search search = g.Bfs(n, 1);
578 output.Add("CUT" + string.Concat(Enumerable.Range(0, n).Where(u => search.Level[u] >= 0).Select(u => " " + network.Names[u])));
579 }
580 else
581 {
582 CopyFlows(g, true);
583 owner = 2;
584 output.Add($"CIRCULATION feasible required={required} cost={network.TotalCost()}");
585 output.AddRange(Flows());
586 }
587 break;
588 }
589 case "MATCH":
590 output.AddRange(Matching(w));
591 break;
592 case "ASSIGN":
593 output.AddRange(Assignment(w));
594 break;
595 }
596 return output;
597 }

599 static List<string> Matching(string[] w)
600 {
601 List<string> left = [.. w[1].Split(',')], right = [.. w[2].Split(',')];
602 HashSet<string> seen = new(StringComparer.Ordinal);
603 foreach (string name in left) if (!seen.Add(name)) Fail("DUPLICATE VERTEX " + name);
604 foreach (string name in right) if (!seen.Add(name)) Fail("DUPLICATE VERTEX " + name);
605 List<(int U, int V)> pairs = [];
606 HashSet<string> pairSeen = new(StringComparer.Ordinal);
607 if (w[3] != "-") foreach (string pair in w[3].Split(','))
608 {
609 string[] p = pair.Split('-');
610 int u = left.IndexOf(p[0]), v = right.IndexOf(p[1]);
611 if (u < 0) Fail("UNKNOWN VERTEX " + p[0]);
612 if (v < 0) Fail("UNKNOWN VERTEX " + p[1]);
613 if (!pairSeen.Add(pair)) Fail("DUPLICATE PAIR " + pair);
614 pairs.Add((u, v));
615 }
616 int[] a = Enumerable.Repeat(-1, left.Count).ToArray(), b = Enumerable.Repeat(-1, right.Count).ToArray();
617 bool Visit(int u, bool[] examined)
618 {
619 foreach ((int x, int v) in pairs) if (x == u && !examined[v])
620 {
621 examined[v] = true;
622 if (b[v] < 0 || Visit(b[v], examined))
623 {
624 b[v] = u;
625 a[u] = v;
626 return true;
627 }
628 }
629 return false;
630 }
631 for (int u = 0; u < a.Length; u++) Visit(u, new bool[b.Length]);
632 bool[] za = new bool[a.Length], zb = new bool[b.Length];
633 Queue<int> queue = new();
634 List<string> freeA = [], freeB = [], cover = [], hallA = [], hallB = [];
635 for (int u = 0; u < a.Length; u++) if (a[u] < 0)
636 {
637 za[u] = true;
638 queue.Enqueue(u);
639 freeA.Add(left[u]);
640 }
641 while (queue.Count > 0)
642 {
643 int u = queue.Dequeue();
644 foreach ((int x, int v) in pairs) if (x == u && a[u] != v && !zb[v])
645 {
646 zb[v] = true;
647 if (b[v] >= 0 && !za[b[v]])
648 {
649 za[b[v]] = true;
650 queue.Enqueue(b[v]);
651 }
652 }
653 }
654 List<string> output = [$"MATCH size={a.Length - freeA.Count} left={a.Length} right={b.Length}"];
655 for (int u = 0; u < a.Length; u++)
656 {
657 if (a[u] >= 0) output.Add($"PAIR {left[u]} {right[a[u]]}");
658 if (za[u]) hallA.Add(left[u]);
659 else cover.Add(left[u]);
660 }
661 for (int v = 0; v < b.Length; v++)
662 {
663 if (b[v] < 0) freeB.Add(right[v]);
664 if (zb[v])
665 {
666 cover.Add(right[v]);
667 hallB.Add(right[v]);
668 }
669 }
670 output.Add($"FREE {Join(freeA)} {Join(freeB)}");
671 output.Add($"COVER size={cover.Count}" + (cover.Count == 0 ? "" : " " + string.Join(" ", cover)));
672 if (freeA.Count != 0) output.Add($"HALL {Join(hallA)} {Join(hallB)}");
673 return output;
674 }

676 static List<string> Assignment(string[] w)
677 {
678 int n = int.Parse(w[2]);
679 if (n < 1 || n > 8) Fail("BAD SIZE");
680 if (w.Length - 3 != n * n) Fail("BAD LENGTH");
681 int?[,] values = new int?[n, n];
682 for (int i = 0; i < n * n; i++) if (w[i + 3] != "x") values[i / n, i % n] = int.Parse(w[i + 3]);
683 int best = 0, count = 0;
684 int[] chosen = [], current = new int[n];
685 void Visit(int row, int used, int sum)
686 {
687 if (row == n)
688 {
689 if (count == 0 || (w[1] == "MIN" ? sum < best : sum > best))
690 {
691 best = sum;
692 count = 1;
693 chosen = (int[])current.Clone();
694 }
695 else if (sum == best) count++;
696 return;
697 }
698 for (int col = 0; col < n; col++) if ((used & (1 << col)) == 0 && values[row, col].HasValue)
699 {
700 current[row] = col;
701 Visit(row + 1, used | (1 << col), sum + values[row, col]!.Value);
702 }
703 }
704 Visit(0, 0, 0);
705 string heading = $"ASSIGN {w[1]} n={n}";
706 List<string> output = [];
707 if (count == 0) output.Add(heading + " infeasible");
708 else
709 {
710 output.Add($"{heading} total={best} optimal={count}");
711 for (int row = 0; row < n; row++) output.Add($"MATE {row + 1} {chosen[row] + 1} {values[row, chosen[row]]}");
712 }
713 return output;
714 }
