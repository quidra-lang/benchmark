2class Edge
3{
4 public function __construct(public int $u, public int $v, public int $capacity,
5 public int $cost = 0, public int $lower = 0, public int $flow = 0) {}

8class FlowNetwork
9{
10 public array $edges;
11 public function __construct(public int $n, array $edges = [])
12 {
13 $this->edges = array_map(fn($e) => clone $e, $edges);
14 }
15
16 public function arcs(?int $vertex = null): array
17 {
18 $result = [];
19 foreach ($this->edges as $i => $edge) {
20 if ($vertex === null || $edge->u === $vertex) {
21 $result[] = $i + 1;
22 }
23 }
24 foreach ($this->edges as $i => $edge) {
25 if ($vertex === null || $edge->v === $vertex) {
26 $result[] = -$i - 1;
27 }
28 }
29 return $result;
30 }
31
32 public function edge(int $arc): Edge
33 {
34 return $this->edges[abs($arc) - 1];
35 }
36
37 public function from(int $arc): int
38 {
39 return $arc > 0 ? $this->edge($arc)->u : $this->edge($arc)->v;
40 }
41
42 public function to(int $arc): int
43 {
44 return $arc > 0 ? $this->edge($arc)->v : $this->edge($arc)->u;
45 }
46
47 public function residual(int $arc): int
48 {
49 $e = $this->edge($arc);
50 return $arc > 0 ? $e->capacity - $e->flow : $e->flow;
51 }
52
53 public function arcCost(int $arc): int
54 {
55 return $arc > 0 ? $this->edge($arc)->cost : -$this->edge($arc)->cost;
56 }
57
58 public function push(array $path, int $amount): void
59 {
60 foreach ($path as $arc) {
61 $this->edge($arc)->flow += $arc > 0 ? $amount : -$amount;
62 }
63 }
64
65 public function bottleneck(array $path): int
66 {
67 return min(array_map(fn($arc) => $this->residual($arc), $path));
68 }
69
70 public function bfs(int $source, int $threshold = 1): array
71 {
72 $levels = array_fill(0, $this->n, null);
73 $parents = array_fill(0, $this->n, null);
74 $levels[$source] = 0;
75 $queue = [$source];
76 for ($i = 0; $i < count($queue); $i++) {
77 $u = $queue[$i];
78 foreach ($this->arcs($u) as $arc) {
79 $v = $this->to($arc);
80 if ($this->residual($arc) >= $threshold && $levels[$v] === null) {
81 $levels[$v] = $levels[$u] + 1;
82 $parents[$v] = $arc;
83 $queue[] = $v;
84 }
85 }
86 }
87 return [$levels, $parents];
88 }
89
90 private function pathTo(int $source, int $target, array $parents): array
91 {
92 $path = [];
93 while ($target !== $source) {
94 $arc = $parents[$target];
95 $path[] = $arc;
96 $target = $this->from($arc);
97 }
98 return array_reverse($path);
99 }
100
101 public function augment(int $source, int $target, int $threshold = 1): ?array
102 {
103 [$levels, $parents] = $this->bfs($source, $threshold);
104 if ($levels[$target] === null) {
105 return null;
106 }
107 $path = $this->pathTo($source, $target, $parents);
108 $amount = $this->bottleneck($path);
109 $this->push($path, $amount);
110 return ['path' => $path, 'amount' => $amount, 'delta' => $threshold];
111 }
112
113 public function maxflow(int $source, int $target, string $method = 'MAXFLOW'): array
114 {
115 $records = [];
116 $phases = 0;
117 if ($method === 'DINIC') {
118 while (true) {
119 [$levels] = $this->bfs($source);
120 if ($levels[$target] === null) {
121 break;
122 }
123 $phases++;
124 $lists = [];
125 for ($v = 0; $v < $this->n; $v++) {
126 $lists[] = $this->arcs($v);
127 }
128 $pointers = array_fill(0, $this->n, 0);
129 $path = [];
130 $u = $source;
131 while (true) {
132 if ($u === $target) {
133 $amount = $this->bottleneck($path);
134 $this->push($path, $amount);
135 $records[] = ['path' => $path, 'amount' => $amount, 'phase' => $phases];
136 $path = [];
137 $u = $source;
138 } elseif ($pointers[$u] >= count($lists[$u])) {
139 if ($u === $source) {
140 break;
141 }
142 $u = $this->from(array_pop($path));
143 $pointers[$u]++;
144 } else {
145 $arc = $lists[$u][$pointers[$u]];
146 $v = $this->to($arc);
147 if ($this->residual($arc) > 0 && $levels[$v] !== null && $levels[$v] === $levels[$u] + 1) {
148 $path[] = $arc;
149 $u = $v;
150 } else {
151 $pointers[$u]++;
152 }
153 }
154 }
155 }
156 } else {
157 $threshold = 1;
158 $maximum = $this->edges ? max(array_map(fn($e) => $e->capacity, $this->edges)) : 0;
159 while ($method === 'SCALING' && $threshold * 2 <= $maximum) {
160 $threshold *= 2;
161 }
162 while ($threshold >= 1) {
163 while (($record = $this->augment($source, $target, $threshold)) !== null) {
164 $records[] = $record;
165 }
166 $threshold = $method === 'SCALING' ? intdiv($threshold, 2) : 0;
167 }
168 }
169 return [$records, $phases];
170 }
171
172 public function negativeCycle(): bool
173 {
174 $distance = array_fill(0, $this->n, 0);
175 for ($pass = 0; $pass < $this->n; $pass++) {
176 $changed = false;
177 foreach ($this->edges as $e) {
178 if ($e->capacity > 0 && $distance[$e->u] + $e->cost < $distance[$e->v]) {
179 $distance[$e->v] = $distance[$e->u] + $e->cost;
180 $changed = true;
181 }
182 }
183 if (!$changed) {
184 return false;
185 }
186 }
187 return $this->n > 0;
188 }
189
190 public function mincost(int $source, int $target, ?int $limit = null): array
191 {
192 $amount = 0;
193 $cost = 0;
194 $records = [];
195 while ($limit === null || $amount < $limit) {
196 $usable = array_filter($this->arcs(), fn($arc) => $this->residual($arc) > 0);
197 $distance = array_fill(0, $this->n, null);
198 $parents = array_fill(0, $this->n, null);
199 $distance[$source] = 0;
200 do {
201 $changed = false;
202 foreach ($usable as $arc) {
203 $u = $this->from($arc);
204 $v = $this->to($arc);
205 if ($distance[$u] === null) {
206 continue;
207 }
208 $value = $distance[$u] + $this->arcCost($arc);
209 if ($distance[$v] === null || $value < $distance[$v]) {
210 $distance[$v] = $value;
211 $parents[$v] = $arc;
212 $changed = true;
213 }
214 }
215 } while ($changed);
216 if ($distance[$target] === null) {
217 break;
218 }
219 $path = $this->pathTo($source, $target, $parents);
220 $delta = $this->bottleneck($path);
221 if ($limit !== null) {
222 $delta = min($delta, $limit - $amount);
223 }
224 $this->push($path, $delta);
225 $amount += $delta;
226 $cost += $delta * $distance[$target];
227 $records[] = ['path' => $path, 'amount' => $delta, 'cost' => $distance[$target]];
228 }
229 return [$amount, $cost, $records];
230 }
231
232 public function balance(): array
233 {
234 $result = array_fill(0, $this->n, 0);
235 foreach ($this->edges as $e) {
236 $result[$e->u] += $e->flow;
237 $result[$e->v] -= $e->flow;
238 }
239 return $result;
240 }
241
242 public function totalCost(): int
243 {
244 return array_sum(array_map(fn($e) => $e->flow * $e->cost, $this->edges));
245 }
246
247 public function decompose(): array
248 {
249 $remaining = array_map(fn($e) => $e->flow, $this->edges);
250 $balances = $this->balance();
251 $result = [];
252 while (true) {
253 $positive = array_keys(array_filter($remaining, fn($g) => $g > 0));
254 if (!$positive) {
255 break;
256 }
257 $supplies = array_keys(array_filter($balances, fn($b) => $b > 0));
258 $start = $supplies ? $supplies[0] : $this->edges[$positive[0]]->u;
259 $vertices = [$start];
260 $path = [];
261 while (true) {
262 $u = $vertices[count($vertices) - 1];
263 foreach ($this->edges as $i => $edge) {
264 if ($edge->u === $u && $remaining[$i] > 0) {
265 break;
266 }
267 }
268 $v = $this->edges[$i]->v;
269 $path[] = $i + 1;
270 $q = array_search($v, $vertices, true);
271 if ($q !== false) {
272 $cycle = array_slice($path, $q);
273 $delta = min(array_map(fn($arc) => $remaining[$arc - 1], $cycle));
274 foreach ($cycle as $arc) {
275 $remaining[$arc - 1] -= $delta;
276 }
277 $result[] = ['kind' => 'CYCLE', 'start' => $v, 'path' => $cycle, 'amount' => $delta];
278 break;
279 }
280 $vertices[] = $v;
281 if ($balances[$v] < 0) {
282 $delta = min(min(array_map(fn($arc) => $remaining[$arc - 1], $path)), $balances[$start], -$balances[$v]);
283 foreach ($path as $arc) {
284 $remaining[$arc - 1] -= $delta;
285 }
286 $balances[$start] -= $delta;
287 $balances[$v] += $delta;
288 $result[] = ['kind' => 'PATH', 'start' => $start, 'path' => $path, 'amount' => $delta];
289 break;
290 }
291 }
292 }
293 return $result;
294 }

299class FlowSolver
300{
301 private array $names = [];
302 private array $demands = [];
303 private FlowNetwork $network;
304 private ?array $owner = null;

306 public function __construct()
307 {
308 $this->network = new FlowNetwork(0);

311 private function vertex(string $name): int
312 {
313 $v = array_search($name, $this->names, true);
314 if ($v === false) {
315 throw new FlowError("UNKNOWN NODE $name");
316 }
317 return $v;
318 }
319
320 private function endpoints(string $s, string $t): array
321 {
322 $source = $this->vertex($s);
323 $target = $this->vertex($t);
324 if ($source === $target) {
325 throw new FlowError('SAME NODE');
326 }
327 return [$source, $target];
328 }

330 private function checkedEdge(string $text): Edge
331 {
332 $number = (int)$text;
333 if ($number < 1 || $number > count($this->network->edges)) {
334 throw new FlowError("UNKNOWN EDGE $text");
335 }
336 return $this->network->edges[$number - 1];
337 }
338
339 private function fresh(): FlowNetwork
340 {
341 $copy = new FlowNetwork(count($this->names), $this->network->edges);
342 foreach ($copy->edges as $e) {
343 $e->flow = 0;
344 }
345 return $copy;
346 }
347
348 private function startFlow(int $source, int $target): int
349 {
350 if ($this->owner !== ['flow', $source, $target]) {
351 foreach ($this->network->edges as $e) {
352 $e->flow = 0;
353 }
354 }
355 $this->owner = ['flow', $source, $target];
356 return $this->network->balance()[$source];
357 }

359 private function notation(FlowNetwork $network, int $start, array $path, ?int $originalOnly = null): string
360 {
361 $words = [$this->names[$start]];
362 foreach ($path as $arc) {
363 if ($originalOnly !== null && abs($arc) > $originalOnly) {
364 continue;
365 }
366 $words[] = ($arc > 0 ? '+' : '-') . abs($arc);
367 $v = $originalOnly !== null ? $this->network->edges[abs($arc) - 1]->v : $network->to($arc);
368 $words[] = $this->names[$v];
369 }
370 return implode(' ', $words);
371 }

373 private function flowList(): void
374 {
375 foreach ($this->network->edges as $i => $e) {
376 echo 'FLOW ', $i + 1, " {$this->names[$e->u]} {$this->names[$e->v]} {$e->flow}/{$e->capacity}\n";
377 }

380 private function circulation(string $mode): void
381 {
382 $work = $this->fresh();
383 foreach ($work->edges as $e) {
384 $e->capacity -= $e->lower;
385 }
386 if ($mode === 'MINCOST' && $work->negativeCycle()) {
387 throw new FlowError('NEGATIVE CYCLE');
388 }
389 $total = array_sum($this->demands);
390 if ($total !== 0) {
391 echo "CIRCULATION infeasible unbalanced=$total\n";
392 return;
393 }
394 $residual = $this->demands;
395 foreach ($this->network->edges as $e) {
396 $residual[$e->v] -= $e->lower;
397 $residual[$e->u] += $e->lower;
398 }
399 $source = count($this->names);
400 $target = $source + 1;
401 foreach ($residual as $v => $r) {
402 if ($r < 0) {
403 $work->edges[] = new Edge($source, $v, -$r);
404 }
405 }
406 $sinks = [];
407 foreach ($residual as $v => $r) {
408 if ($r > 0) {
409 $sinks[] = count($work->edges);
410 $work->edges[] = new Edge($v, $target, $r);
411 }
412 }
413 $work->n = $target + 1;
414 if ($mode === 'ANY') {
415 $work->maxflow($source, $target);
416 } else {
417 $work->mincost($source, $target);
418 }
419 $required = array_sum(array_filter($residual, fn($r) => $r > 0));
420 $achieved = array_sum(array_map(fn($i) => $work->edges[$i]->flow, $sinks));
421 if ($achieved < $required) {
422 echo "CIRCULATION infeasible required=$required achieved=$achieved\n";
423 [$reachable] = $work->bfs($source);
424 $cut = [];
425 foreach ($this->names as $v => $name) {
426 if ($reachable[$v] !== null) {
427 $cut[] = $name;
428 }
429 }
430 echo 'CUT ', implode(' ', $cut), "\n";
431 } else {
432 foreach ($this->network->edges as $i => $e) {
433 $e->flow = $e->lower + $work->edges[$i]->flow;
434 }
435 $this->owner = ['circulation'];
436 echo "CIRCULATION feasible required=$required cost=", $this->network->totalCost(), "\n";
437 $this->flowList();
438 }
439 }

441 private function matching(string $leftText, string $rightText, string $pairText): void
442 {
443 $left = explode(',', $leftText);
444 $right = explode(',', $rightText);
445 $seen = [];
446 foreach (array_merge($left, $right) as $name) {
447 if (isset($seen[$name])) {
448 throw new FlowError("DUPLICATE VERTEX $name");
449 }
450 $seen[$name] = true;
451 }
452 $pairs = $pairText === '-' ? [] : array_map(fn($p) => explode('-', $p), explode(',', $pairText));
453 $seenPairs = [];
454 foreach ($pairs as [$a, $b]) {
455 if (!in_array($a, $left, true)) {
456 throw new FlowError("UNKNOWN VERTEX $a");
457 }
458 if (!in_array($b, $right, true)) {
459 throw new FlowError("UNKNOWN VERTEX $b");
460 }
461 if (isset($seenPairs["$a-$b"])) {
462 throw new FlowError("DUPLICATE PAIR $a-$b");
463 }
464 $seenPairs["$a-$b"] = true;
465 }
466 $lp = [];
467 $rp = [];
468 $examined = [];
469 $attempt = function (string $a) use (&$attempt, &$lp, &$rp, &$examined, $pairs): bool {
470 foreach ($pairs as [$x, $b]) {
471 if ($x !== $a || isset($examined[$b])) {
472 continue;
473 }
474 $examined[$b] = true;
475 if (!isset($rp[$b]) || $attempt($rp[$b])) {
476 $rp[$b] = $a;
477 $lp[$a] = $b;
478 return true;
479 }
480 }
481 return false;
482 };
483 foreach ($left as $a) {
484 $examined = [];
485 $attempt($a);
486 }
487 $freeLeft = array_values(array_filter($left, fn($a) => !isset($lp[$a])));
488 $freeRight = array_values(array_filter($right, fn($b) => !isset($rp[$b])));
489 $zl = array_fill_keys($freeLeft, true);
490 $zr = [];
491 $queue = $freeLeft;
492 for ($i = 0; $i < count($queue); $i++) {
493 $a = $queue[$i];
494 foreach ($pairs as [$x, $b]) {
495 if ($x === $a && ($lp[$a] ?? null) !== $b && !isset($zr[$b])) {
496 $zr[$b] = true;
497 if (isset($rp[$b]) && !isset($zl[$rp[$b]])) {
498 $zl[$rp[$b]] = true;
499 $queue[] = $rp[$b];
500 }
501 }
502 }
503 }
504 $cover = array_merge(array_filter($left, fn($a) => !isset($zl[$a])), array_filter($right, fn($b) => isset($zr[$b])));
505 echo 'MATCH size=', count($lp), ' left=', count($left), ' right=', count($right), "\n";
506 foreach ($left as $a) {
507 if (isset($lp[$a])) {
508 echo "PAIR $a {$lp[$a]}\n";
509 }
510 }
511 echo 'FREE ', $freeLeft ? implode(',', $freeLeft) : '-', ' ', $freeRight ? implode(',', $freeRight) : '-', "\n";
512 echo 'COVER size=', count($cover), $cover ? ' ' . implode(' ', $cover) : '', "\n";
513 if ($freeLeft) {
514 echo 'HALL ', implode(',', array_filter($left, fn($a) => isset($zl[$a]))), ' ',
515 $zr ? implode(',', array_filter($right, fn($b) => isset($zr[$b]))) : '-', "\n";
516 }
517 }

519 private function assignment(string $mode, int $n, array $values): void
520 {
521 if ($n < 1 || $n > 8) {
522 throw new FlowError('BAD SIZE');
523 }
524 if (count($values) !== $n * $n) {
525 throw new FlowError('BAD LENGTH');
526 }
527 $best = null;
528 $count = 0;
529 $chosen = [];
530 $search = function (array $columns, int $total) use (&$search, &$best, &$count, &$chosen, $mode, $n, $values): void {
531 $row = count($columns);
532 if ($row === $n) {
533 if ($best === null || ($mode === 'MIN' ? $total < $best : $total > $best)) {
534 $best = $total;
535 $count = 1;
536 $chosen = $columns;
537 } elseif ($total === $best) {
538 $count++;
539 }
540 return;
541 }
542 for ($column = 0; $column < $n; $column++) {
543 $value = $values[$row * $n + $column];
544 if (!in_array($column, $columns, true) && $value !== 'x') {
545 $search([...$columns, $column], $total + (int)$value);
546 }
547 }
548 };
549 $search([], 0);
550 if ($best === null) {
551 echo "ASSIGN $mode n=$n infeasible\n";
552 } else {
553 echo "ASSIGN $mode n=$n total=$best optimal=$count\n";
554 foreach ($chosen as $row => $column) {
555 echo 'MATE ', $row + 1, ' ', $column + 1, ' ', $values[$row * $n + $column], "\n";
556 }
557 }
558 }

560 public function execute(string $line): void
561 {
562 $args = explode(' ', $line);
563 $command = array_shift($args);
564 try {
565 switch ($command) {
566 case 'NODE':
567 $seen = $this->names;
568 foreach ($args as $name) {
569 if (in_array($name, $seen, true)) {
570 throw new FlowError("DUPLICATE NODE $name");
571 }
572 $seen[] = $name;
573 }
574 if (count($seen) > 20) {
575 throw new FlowError('TOO MANY NODES');
576 }
577 $this->names = $seen;
578 $this->demands = array_merge($this->demands, array_fill(0, count($args), 0));
579 $this->network->n = count($seen);
580 echo 'NODE added=', count($args), ' total=', count($seen), "\n";
581 break;
582 case 'EDGE':
583 $u = $this->vertex($args[0]);
584 $v = $this->vertex($args[1]);
585 if ($u === $v) {
586 throw new FlowError('SELF LOOP');
587 }
588 if (count($this->network->edges) === 60) {
589 throw new FlowError('TOO MANY EDGES');
590 }
591 $e = new Edge($u, $v, (int)$args[2], (int)($args[3] ?? 0));
592 $this->network->edges[] = $e;
593 echo 'EDGE ', count($this->network->edges), " {$args[0]} {$args[1]} cap={$e->capacity} cost={$e->cost}\n";
594 break;
595 case 'LOWER':
596 case 'CAP':
597 case 'COST':
598 $e = $this->checkedEdge($args[0]);
599 $value = (int)$args[1];
600 if ($command === 'LOWER') {
601 if ($value > $e->capacity) {
602 throw new FlowError('BAD BOUND');
603 }
604 $e->lower = $value;
605 echo "LOWER {$args[0]} low=$value cap={$e->capacity}\n";
606 } elseif ($command === 'CAP') {
607 if ($value < $e->lower) {
608 throw new FlowError('BAD BOUND');
609 }
610 if ($value < $e->flow) {
611 throw new FlowError('BELOW FLOW');
612 }
613 $e->capacity = $value;
614 echo "CAP {$args[0]} low={$e->lower} cap=$value\n";
615 } else {
616 $e->cost = $value;
617 echo "COST {$args[0]} cost=$value\n";
618 }
619 break;
620 case 'DEMAND':
621 $v = $this->vertex($args[0]);
622 $this->demands[$v] = (int)$args[1];
623 echo "DEMAND {$args[0]} {$args[1]} total=", array_sum($this->demands), "\n";
624 break;
625 case 'RESET':
626 echo 'RESET edges=', count(array_filter($this->network->edges, fn($e) => $e->flow !== 0)), "\n";
627 foreach ($this->network->edges as $e) {
628 $e->flow = 0;
629 }
630 $this->owner = null;
631 break;
632 case 'CLEAR':
633 echo 'CLEAR nodes=', count($this->names), ' edges=', count($this->network->edges), "\n";
634 $this->names = [];
635 $this->demands = [];
636 $this->network = new FlowNetwork(0);
637 $this->owner = null;
638 break;
639 case 'SHOW':
640 $balance = $this->network->balance();
641 echo 'NETWORK nodes=', count($this->names), ' edges=', count($this->network->edges), "\n";
642 if ($this->owner === null) {
643 echo "STATE none\n";
644 } elseif ($this->owner[0] === 'circulation') {
645 echo 'STATE circulation cost=', $this->network->totalCost(), "\n";
646 } else {
647 echo "STATE flow {$this->names[$this->owner[1]]} {$this->names[$this->owner[2]]} value={$balance[$this->owner[1]]} cost=",
648 $this->network->totalCost(), "\n";
649 }
650 foreach ($this->names as $v => $name) {
651 $out = 0;
652 foreach ($this->network->edges as $e) {
653 if ($e->u === $v) {
654 $out += $e->flow;
655 }
656 }
657 echo "NODE $name demand={$this->demands[$v]} out=$out in=", $out - $balance[$v], "\n";
658 }
659 foreach ($this->network->edges as $i => $e) {
660 echo 'EDGE ', $i + 1, " {$this->names[$e->u]} {$this->names[$e->v]} cap={$e->capacity} cost={$e->cost} low={$e->lower} flow={$e->flow}\n";
661 }
662 break;
663 case 'DECOMPOSE':
664 $records = $this->network->decompose();
665 $paths = count(array_filter($records, fn($r) => $r['kind'] === 'PATH'));
666 echo "DECOMPOSE paths=$paths cycles=", count($records) - $paths, "\n";
667 $counters = ['PATH' => 0, 'CYCLE' => 0];
668 foreach ($records as $r) {
669 $number = ++$counters[$r['kind']];
670 echo "{$r['kind']} $number amount={$r['amount']} ", $this->notation($this->network, $r['start'], $r['path']), "\n";
671 }
672 break;
673 case 'VERIFY':
674 $violations = [];
675 foreach ($this->network->edges as $i => $e) {
676 if ($e->flow < $e->lower) {
677 $violations[] = 'LOW ' . ($i + 1) . " flow={$e->flow} low={$e->lower}";
678 }
679 }
680 $balance = $this->network->balance();
681 foreach ($this->names as $v => $name) {
682 $isFlow = $this->owner !== null && $this->owner[0] === 'flow';
683 if ($isFlow && ($v === $this->owner[1] || $v === $this->owner[2])) {
684 continue;
685 }
686 $required = $isFlow ? 0 : $this->demands[$v];
687 if (-$balance[$v] !== $required) {
688 $violations[] = "BALANCE $name net=" . (-$balance[$v]) . " need=$required";
689 }
690 }
691 echo $violations ? 'VERIFY violations=' . count($violations) . "\n" . implode("\n", $violations) : 'VERIFY ok', "\n";
692 break;
693 case 'CIRCULATION':
694 $this->circulation($args[0]);
695 break;
696 case 'MATCH':
697 $this->matching(...$args);
698 break;
699 case 'ASSIGN':
700 $this->assignment($args[0], (int)$args[1], array_slice($args, 2));
701 break;
702 default:
703 [$source, $target] = $this->endpoints($args[0], $args[1]);
704 $this->flowCommand($command, $args, $source, $target);
705 }
706 } catch (FlowError $error) {
707 echo 'ERROR ', $error->getMessage(), "\n";
708 }
709 }

711 private function flowCommand(string $command, array $args, int $source, int $target): void
712 {
713 $prefix = "$command {$args[0]} {$args[1]}";
714 if ($command === 'SENSITIVITY') {
715 $maximum = function (?int $index, int $delta) use ($source, $target): int {
716 $work = $this->fresh();
717 if ($index !== null) {
718 $work->edges[$index]->capacity += $delta;
719 }
720 $work->maxflow($source, $target);
721 return $work->balance()[$source];
722 };
723 $value = $maximum(null, 0);
724 $critical = [];
725 foreach ($this->network->edges as $i => $e) {
726 $critical[] = [$i, $maximum($i, 1) > $value, $e->capacity > 0 && $maximum($i, -1) < $value];
727 }
728 echo "$prefix value=$value up=", count(array_filter($critical, fn($r) => $r[1])),
729 ' down=', count(array_filter($critical, fn($r) => $r[2])), "\n";
730 foreach ($critical as [$i, $up, $down]) {
731 if ($up || $down) {
732 $e = $this->network->edges[$i];
733 echo 'CRITICAL ', $i + 1, " {$this->names[$e->u]} {$this->names[$e->v]} up=", $up ? 'yes' : 'no',
734 ' down=', $down ? 'yes' : 'no', "\n";
735 }
736 }
737 } elseif ($command === 'MINCOST') {
738 $work = $this->fresh();
739 if ($work->negativeCycle()) {
740 throw new FlowError('NEGATIVE CYCLE');
741 }
742 $limit = $args[2] === 'MAX' ? null : (int)$args[2];
743 [$amount, $cost, $records] = $work->mincost($source, $target, $limit);
744 if ($limit !== null && $amount < $limit) {
745 echo "$prefix k={$args[2]} infeasible max=$amount\n";
746 return;
747 }
748 foreach ($this->network->edges as $i => $e) {
749 $e->flow = $work->edges[$i]->flow;
750 }
751 $this->owner = ['flow', $source, $target];
752 echo "$prefix k={$args[2]} flow=$amount cost=$cost\n";
753 foreach ($records as $i => $r) {
754 echo 'AUG ', $i + 1, " amount={$r['amount']} cost={$r['cost']} ", $this->notation($work, $source, $r['path']), "\n";
755 }
756 $this->flowList();
757 } elseif ($command === 'DISJOINT') {
758 $edges = array_map(fn($e) => new Edge($e->u, $e->v, $e->capacity > 0 ? 1 : 0), $this->network->edges);
759 $n = count($this->names);
760 if ($args[2] === 'VERTEX') {
761 $exits = range(0, $n - 1);
762 for ($v = 0; $v < count($this->names); $v++) {
763 if ($v !== $source && $v !== $target) {
764 $exits[$v] = $n++;
765 }
766 }
767 foreach ($edges as $e) {
768 $e->u = $exits[$e->u];
769 }
770 for ($v = 0; $v < count($this->names); $v++) {
771 if ($v !== $source && $v !== $target) {
772 $edges[] = new Edge($v, $exits[$v], 1);
773 }
774 }
775 }
776 $work = new FlowNetwork($n, $edges);
777 $work->maxflow($source, $target);
778 $records = array_values(array_filter($work->decompose(), fn($r) => $r['kind'] === 'PATH'));
779 echo "$prefix {$args[2]} count=", count($records), "\n";
780 foreach ($records as $i => $r) {
781 echo 'PATH ', $i + 1, ' ', $this->notation($work, $source, $r['path'], count($this->network->edges)), "\n";
782 }
783 } else {
784 $start = $this->startFlow($source, $target);
785 if ($command === 'AUGMENT') {
786 $r = $this->network->augment($source, $target);
787 echo "$prefix value=", $this->network->balance()[$source], ' ',
788 $r === null ? 'none' : "amount={$r['amount']} " . $this->notation($this->network, $source, $r['path']), "\n";
789 return;
790 }
791 [$records, $phases] = $this->network->maxflow($source, $target, $command);
792 $value = $this->network->balance()[$source];
793 if ($command === 'MINCUT') {
794 [$reachable] = $this->network->bfs($source);
795 $cut = [];
796 foreach ($this->network->edges as $i => $e) {
797 if ($reachable[$e->u] !== null && $reachable[$e->v] === null) {
798 $cut[] = $i;
799 }
800 }
801 echo "$prefix value=$value edges=", count($cut), "\n";
802 $s = [];
803 $t = [];
804 foreach ($this->names as $v => $name) {
805 if ($reachable[$v] === null) {
806 $t[] = $name;
807 } else {
808 $s[] = $name;
809 }
810 }
811 echo 'SOURCE ', implode(' ', $s), "\nSINK ", implode(' ', $t), "\n";
812 foreach ($cut as $i) {
813 $e = $this->network->edges[$i];
814 echo 'CUT ', $i + 1, " {$this->names[$e->u]} {$this->names[$e->v]} {$e->capacity}\n";
815 }
816 } else {
817 echo "$prefix value=$value start=$start", $command === 'DINIC' ? " phases=$phases" : '',
818 ' paths=', count($records), "\n";
819 foreach ($records as $i => $r) {
820 $extra = match ($command) {
821 'DINIC' => "phase={$r['phase']} ",
822 'SCALING' => "delta={$r['delta']} ",
823 default => ''
824 };
825 echo 'AUG ', $i + 1, " {$extra}amount={$r['amount']} ", $this->notation($this->network, $source, $r['path']), "\n";
826 }
827 $this->flowList();
828 }
829 }
830 }
831}
