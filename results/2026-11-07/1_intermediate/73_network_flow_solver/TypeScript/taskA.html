3interface Edge { u: number; v: number; capacity: number; cost: number; lower: number; flow: number }
4interface Owner { kind: 'none' | 'flow' | 'circulation'; source: number; sink: number }
5interface Arc { edge: number; forward: boolean; from: number; to: number }
6interface Augmentation { path: Arc[]; amount: number; phase: number; delta: number; cost: number }
7interface Decomposition { cycle: boolean; start: number; edges: number[]; amount: number }
8interface BreadthFirst { level: number[]; parent: (Arc | null)[] }
9interface Cheapest { path: Arc[]; cost: number }

11class Network {
12 names: string[] = [];
13 edges: Edge[] = [];
14 demands: number[] = [];
15 owner: Owner = { kind: 'none', source: -1, sink: -1 };

17 const result = new Network();

26function fail(message: string): never { throw new Error('ERROR ' + message); }
27function vertex(name: string): number {
28 const index = network.names.indexOf(name);
29 if (index < 0) fail('UNKNOWN NODE ' + name);
30 return index;

32function endpoints(a: string, b: string): [number, number] {
33 const source = vertex(a), sink = vertex(b);
34 if (source === sink) fail('SAME NODE');
35 return [source, sink];

37function edgeAt(word: string): Edge {
38 const index = Number(word) - 1;
39 if (index < 0 || index >= network.edges.length) fail('UNKNOWN EDGE ' + word);
40 return network.edges[index];
41}
42function balances(net: Network): number[] {
43 const result = Array<number>(net.names.length).fill(0);
44 for (const edge of net.edges) { result[edge.u] += edge.flow; result[edge.v] -= edge.flow; }
45 return result;
46}
47function totalCost(net: Network): number { return net.edges.reduce((sum, edge) => sum + edge.flow * edge.cost, 0); }
48function globalArcs(net: Network): Arc[] {
49 const forward = net.edges.map((edge, index) => ({ edge: index, forward: true, from: edge.u, to: edge.v }));
50 const backward = net.edges.map((edge, index) => ({ edge: index, forward: false, from: edge.v, to: edge.u }));
51 return [...forward, ...backward];
52}
53function adjacency(net: Network): Arc[][] {
54 const result: Arc[][] = Array.from({ length: net.names.length }, () => []);
55 for (const arc of globalArcs(net)) result[arc.from].push(arc);
56 return result;
57}
58function residual(net: Network, arc: Arc): number {
59 const edge = net.edges[arc.edge];
60 return arc.forward ? edge.capacity - edge.flow : edge.flow;
61}
62function push(net: Network, path: Arc[], amount: number): void {
63 for (const arc of path) net.edges[arc.edge].flow += arc.forward ? amount : -amount;
64}
65function bottleneck(net: Network, path: Arc[]): number { return Math.min(...path.map(arc => residual(net, arc))); }
66function bfs(net: Network, source: number, threshold = 1): BreadthFirst {
67 const level = Array<number>(net.names.length).fill(-1), parent = Array<Arc | null>(net.names.length).fill(null), queue = [source];
68 const lists = adjacency(net);
69 level[source] = 0;
70 for (let i = 0; i < queue.length; i++) {
71 const u = queue[i];
72 for (const arc of lists[u]) if (residual(net, arc) >= threshold && level[arc.to] < 0) {
73 level[arc.to] = level[u] + 1; parent[arc.to] = arc; queue.push(arc.to);
74 }
75 }
76 return { level, parent };
77}
78function restorePath(parent: (Arc | null)[], source: number, sink: number): Arc[] {
79 const result: Arc[] = [];
80 for (let at = sink; at !== source;) { const arc = parent[at]!; result.push(arc); at = arc.from; }
81 return result.reverse();
82}
83function single(net: Network, source: number, sink: number, threshold = 1): Augmentation | null {
84 const search = bfs(net, source, threshold);
85 if (search.level[sink] < 0) return null;
86 const path = restorePath(search.parent, source, sink), amount = bottleneck(net, path);
87 push(net, path, amount);
88 return { path, amount, phase: 0, delta: threshold, cost: 0 };
89}
90function edmondsKarp(net: Network, source: number, sink: number): Augmentation[] {
91 const paths: Augmentation[] = [];
92 while (true) { const augmentation = single(net, source, sink); if (augmentation === null) return paths; paths.push(augmentation); }
93}
94function dinic(net: Network, source: number, sink: number): { paths: Augmentation[]; phases: number } {
95 const paths: Augmentation[] = [];
96 let phases = 0;
97 const lists = adjacency(net);
98 while (true) {
99 const { level } = bfs(net, source);
100 if (level[sink] < 0) return { paths, phases };
101 phases++;
102 const pointers = Array<number>(net.names.length).fill(0);
103 let path: Arc[] = [], current = source;
104 while (true) {
105 if (current === sink) {
106 const amount = bottleneck(net, path);
107 push(net, path, amount);
108 paths.push({ path: [...path], amount, phase: phases, delta: 1, cost: 0 });
109 path = []; current = source;
110 } else if (pointers[current] >= lists[current].length) {
111 if (current === source) break;
112 const arc = path.pop()!; current = arc.from; pointers[current]++;
113 } else {
114 const arc = lists[current][pointers[current]];
115 if (residual(net, arc) > 0 && level[arc.to] === level[current] + 1) { path.push(arc); current = arc.to; }
116 else pointers[current]++;
117 }
118 }
119 }
120}
121function scaling(net: Network, source: number, sink: number): Augmentation[] {
122 const paths: Augmentation[] = [], maximum = Math.max(1, ...net.edges.map(edge => edge.capacity));
123 let delta = 1;
124 while (delta * 2 <= maximum) delta *= 2;
125 while (delta >= 1) {
126 while (true) { const augmentation = single(net, source, sink, delta); if (augmentation === null) break; paths.push(augmentation); }
127 delta = Math.floor(delta / 2);
128 }
129 return paths;
130}
131function negativeCycle(net: Network, lowerBounds = false): boolean {
132 const distance = Array<number>(net.names.length).fill(0);
133 for (let pass = 0; pass < net.names.length; pass++) {
134 let changed = false;
135 for (const edge of net.edges) if (edge.capacity - (lowerBounds ? edge.lower : 0) > 0 && distance[edge.v] > distance[edge.u] + edge.cost) {
136 distance[edge.v] = distance[edge.u] + edge.cost; changed = true;
137 }
138 if (!changed) return false;
139 if (pass + 1 === net.names.length) return true;
140 }
141 return false;
142}
143function shortest(net: Network, source: number, sink: number): Cheapest | null {
144 const arcs = globalArcs(net).filter(arc => residual(net, arc) > 0);
145 const distance = Array<number | null>(net.names.length).fill(null), parent = Array<Arc | null>(net.names.length).fill(null);
146 distance[source] = 0;
147 while (true) {
148 let changed = false;
149 for (const arc of arcs) {
150 if (distance[arc.from] === null) continue;
151 const cost = net.edges[arc.edge].cost * (arc.forward ? 1 : -1), next = distance[arc.from]! + cost;
152 if (distance[arc.to] === null || next < distance[arc.to]!) { distance[arc.to] = next; parent[arc.to] = arc; changed = true; }
153 }
154 if (!changed) break;
155 }
156 return distance[sink] === null ? null : { path: restorePath(parent, source, sink), cost: distance[sink]! };
157}
158function minCost(net: Network, source: number, sink: number, target: number | null = null): { paths: Augmentation[]; flow: number; cost: number } {
159 const paths: Augmentation[] = [];
160 let flow = 0, cost = 0;
161 while (target === null || flow < target) {
162 const cheapest = shortest(net, source, sink);
163 if (cheapest === null) break;
164 const amount = Math.min(bottleneck(net, cheapest.path), target === null ? Infinity : target - flow);
165 push(net, cheapest.path, amount); flow += amount; cost += amount * cheapest.cost;
166 paths.push({ path: cheapest.path, amount, phase: 0, delta: 1, cost: cheapest.cost });
167 }
168 return { paths, flow, cost };
169}
170function decompose(net: Network): Decomposition[] {
171 const remaining = net.edges.map(edge => edge.flow), balance = balances(net), result: Decomposition[] = [];
172 while (remaining.some(value => value > 0)) {
173 let start = balance.findIndex(value => value > 0);
174 if (start < 0) start = net.edges[remaining.findIndex(value => value > 0)].u;
175 const vertices = [start], edges: number[] = [];
176 while (true) {
177 const current = vertices[vertices.length - 1];
178 const index = net.edges.findIndex((edge, i) => edge.u === current && remaining[i] > 0);
179 const next = net.edges[index].v;
180 edges.push(index);
181 const repeat = vertices.indexOf(next);
182 if (repeat >= 0) {
183 const cycle = edges.slice(repeat), amount = Math.min(...cycle.map(i => remaining[i]));
184 for (const i of cycle) remaining[i] -= amount;
185 result.push({ cycle: true, start: next, edges: cycle, amount });
186 break;
187 }
188 vertices.push(next);
189 if (balance[next] < 0) {
190 const amount = Math.min(...edges.map(i => remaining[i]), balance[start], -balance[next]);
191 for (const i of edges) remaining[i] -= amount;
192 balance[start] -= amount; balance[next] += amount;
193 result.push({ cycle: false, start, edges: [...edges], amount });
194 break;
195 }
196 }
197 }
198 return result;
199}
200function pathText(net: Network, path: Arc[], source: number): string {
201 return [net.names[source], ...path.flatMap(arc => [(arc.forward ? '+' : '-') + (arc.edge + 1), net.names[arc.to]])].join(' ');
202}
203function decompositionText(net: Network, item: Decomposition, originalEdges = Infinity): string {
204 const result = [net.names[item.start]];
205 for (const index of item.edges) if (index < originalEdges) result.push('+' + (index + 1), net.names[net.edges[index].v]);
206 return result.join(' ');
207}
208function flowList(): void {
209 network.edges.forEach((edge, i) => output.push(`FLOW ${i + 1} ${network.names[edge.u]} ${network.names[edge.v]} ${edge.flow}/${edge.capacity}`));
210}
211function startFlow(source: number, sink: number): number {
212 if (network.owner.kind !== 'flow' || network.owner.source !== source || network.owner.sink !== sink) for (const edge of network.edges) edge.flow = 0;
213 network.owner = { kind: 'flow', source, sink };
214 return balances(network)[source];
215}
216function matching(a: string, b: string, pairs: string): void {
217 const left = a.split(','), right = b.split(','), links = pairs === '-' ? [] : pairs.split(',').map(pair => pair.split('-'));
218 const seen = new Set<string>();
219 for (const name of [...left, ...right]) { if (seen.has(name)) fail('DUPLICATE VERTEX ' + name); seen.add(name); }
220 const used = new Set<string>();
221 for (const [u, v] of links) {
222 if (!left.includes(u)) fail('UNKNOWN VERTEX ' + u);
223 if (!right.includes(v)) fail('UNKNOWN VERTEX ' + v);
224 if (used.has(u + '-' + v)) fail('DUPLICATE PAIR ' + u + '-' + v);
225 used.add(u + '-' + v);
226 }
227 const partnerLeft = new Map<string, string>(), partnerRight = new Map<string, string>();
228 function attempt(name: string, examined: Set<string>): boolean {
229 for (const [u, v] of links) if (u === name && !examined.has(v)) {
230 examined.add(v);
231 if (!partnerRight.has(v) || attempt(partnerRight.get(v)!, examined)) { partnerLeft.set(u, v); partnerRight.set(v, u); return true; }
232 }
233 return false;
234 }
235 for (const name of left) attempt(name, new Set<string>());
236 const free = left.filter(name => !partnerLeft.has(name)), zl = new Set(free), zr = new Set<string>(), queue = [...free];
237 for (let i = 0; i < queue.length; i++) for (const [u, v] of links) if (u === queue[i] && partnerLeft.get(u) !== v && !zr.has(v)) {
238 zr.add(v);
239 const partner = partnerRight.get(v);
240 if (partner !== undefined && !zl.has(partner)) { zl.add(partner); queue.push(partner); }
241 }
242 const cover = [...left.filter(name => !zl.has(name)), ...right.filter(name => zr.has(name))];
243 output.push(`MATCH size=${partnerLeft.size} left=${left.length} right=${right.length}`);
244 for (const name of left) if (partnerLeft.has(name)) output.push(`PAIR ${name} ${partnerLeft.get(name)}`);
245 output.push(`FREE ${free.join(',') || '-'} ${right.filter(name => !partnerRight.has(name)).join(',') || '-'}`,
246 `COVER size=${cover.length}${cover.length ? ' ' + cover.join(' ') : ''}`);
247 if (free.length) output.push(`HALL ${left.filter(name => zl.has(name)).join(',')} ${right.filter(name => zr.has(name)).join(',') || '-'}`);
248}
249function assignment(mode: string, size: number, words: string[]): void {
250 if (size < 1 || size > 8) fail('BAD SIZE');
251 if (words.length !== size * size) fail('BAD LENGTH');
252 let best: number | null = null, count = 0, chosen: number[] = [];
253 const current: number[] = [];
254 function visit(row: number, used: number, sum: number): void {
255 if (row === size) {
256 if (best === null || (mode === 'MIN' ? sum < best : sum > best)) { best = sum; count = 1; chosen = [...current]; }
257 else if (sum === best) count++;
258 return;
259 }
260 for (let col = 0; col < size; col++) if (!(used & (1 << col)) && words[row * size + col] !== 'x') {
261 current.push(col); visit(row + 1, used | (1 << col), sum + Number(words[row * size + col])); current.pop();
262 }
263 }
264 visit(0, 0, 0);
265 if (best === null) output.push(`ASSIGN ${mode} n=${size} infeasible`);
266 else {
267 output.push(`ASSIGN ${mode} n=${size} total=${best} optimal=${count}`);
268 chosen.forEach((col, row) => output.push(`MATE ${row + 1} ${col + 1} ${words[row * size + col]}`));
269 }
270}
271function execute(commandLine: string): void {
272 const [command, ...args] = commandLine.split(' '), a = args[0], b = args[1];
273 switch (command) {
274 case 'NODE': {
275 const seen = new Set(network.names);
276 for (const name of args) { if (seen.has(name)) fail('DUPLICATE NODE ' + name); seen.add(name); }
277 if (network.names.length + args.length > 20) fail('TOO MANY NODES');
278 network.names.push(...args); network.demands.push(...args.map(() => 0));
279 output.push(`NODE added=${args.length} total=${network.names.length}`);
280 break;
281 }
282 case 'EDGE': {
283 const u = vertex(a), v = vertex(b);
284 if (u === v) fail('SELF LOOP');
285 if (network.edges.length === 60) fail('TOO MANY EDGES');
286 const capacity = Number(args[2]), cost = Number(args[3] ?? 0);
287 network.edges.push({ u, v, capacity, cost, lower: 0, flow: 0 });
288 output.push(`EDGE ${network.edges.length} ${a} ${b} cap=${capacity} cost=${cost}`);
289 break;
290 }
291 case 'LOWER': {
292 const edge = edgeAt(a), lower = Number(b);
293 if (lower > edge.capacity) fail('BAD BOUND');
294 edge.lower = lower; output.push(`LOWER ${a} low=${lower} cap=${edge.capacity}`);
295 break;
296 }
297 case 'CAP': {
298 const edge = edgeAt(a), capacity = Number(b);
299 if (capacity < edge.lower) fail('BAD BOUND');
300 if (capacity < edge.flow) fail('BELOW FLOW');
301 edge.capacity = capacity; output.push(`CAP ${a} low=${edge.lower} cap=${capacity}`);
302 break;
303 }
304 case 'COST': { const edge = edgeAt(a); edge.cost = Number(b); output.push(`COST ${a} cost=${b}`); break; }
305 case 'DEMAND': {
306 const index = vertex(a); network.demands[index] = Number(b);
307 output.push(`DEMAND ${a} ${b} total=${network.demands.reduce((sum, demand) => sum + demand, 0)}`);
308 break;
309 }
310 case 'RESET': {
311 const count = network.edges.filter(edge => edge.flow !== 0).length;
312 for (const edge of network.edges) edge.flow = 0;
313 network.owner = { kind: 'none', source: -1, sink: -1 };
314 output.push(`RESET edges=${count}`); break;
315 }
316 case 'CLEAR':
317 output.push(`CLEAR nodes=${network.names.length} edges=${network.edges.length}`);
318 network.names = []; network.demands = []; network.edges = []; network.owner = { kind: 'none', source: -1, sink: -1 };
319 break;
320 case 'SHOW': {
321 const out = Array<number>(network.names.length).fill(0), incoming = Array<number>(network.names.length).fill(0);
322 for (const edge of network.edges) { out[edge.u] += edge.flow; incoming[edge.v] += edge.flow; }
323 const owner = network.owner;
324 output.push(`NETWORK nodes=${network.names.length} edges=${network.edges.length}`);
325 output.push(owner.kind === 'none' ? 'STATE none' : owner.kind === 'circulation' ? `STATE circulation cost=${totalCost(network)}` : `STATE flow ${network.names[owner.source]} ${network.names[owner.sink]} value=${balances(network)[owner.source]} cost=${totalCost(network)}`);
326 network.names.forEach((name, i) => output.push(`NODE ${name} demand=${network.demands[i]} out=${out[i]} in=${incoming[i]}`));
327 network.edges.forEach((edge, i) => output.push(`EDGE ${i + 1} ${network.names[edge.u]} ${network.names[edge.v]} cap=${edge.capacity} cost=${edge.cost} low=${edge.lower} flow=${edge.flow}`));
328 break;
329 }
330 case 'AUGMENT': case 'MAXFLOW': case 'DINIC': case 'SCALING': case 'MINCUT': {
331 const [source, sink] = endpoints(a, b), start = startFlow(source, sink);
332 if (command === 'AUGMENT') {
333 const augmentation = single(network, source, sink);
334 output.push(`AUGMENT ${a} ${b} value=${balances(network)[source]}${augmentation === null ? ' none' : ` amount=${augmentation.amount} ${pathText(network, augmentation.path, source)}`}`);
335 break;
336 }
337 const result = command === 'DINIC' ? dinic(network, source, sink) : { paths: command === 'SCALING' ? scaling(network, source, sink) : edmondsKarp(network, source, sink), phases: 0 };
338 const value = balances(network)[source];
339 if (command === 'MINCUT') {
340 const { level } = bfs(network, source), cut = network.edges.map((edge, i) => i).filter(i => level[network.edges[i].u] >= 0 && level[network.edges[i].v] < 0);
341 output.push(`MINCUT ${a} ${b} value=${value} edges=${cut.length}`,
342 'SOURCE ' + network.names.filter((_, i) => level[i] >= 0).join(' '), 'SINK ' + network.names.filter((_, i) => level[i] < 0).join(' '));
343 for (const i of cut) { const edge = network.edges[i]; output.push(`CUT ${i + 1} ${network.names[edge.u]} ${network.names[edge.v]} ${edge.capacity}`); }
344 } else {
345 output.push(`${command} ${a} ${b} value=${value} start=${start}${command === 'DINIC' ? ` phases=${result.phases}` : ''} paths=${result.paths.length}`);
346 result.paths.forEach((item, i) => output.push(`AUG ${i + 1}${command === 'DINIC' ? ` phase=${item.phase}` : command === 'SCALING' ? ` delta=${item.delta}` : ''} amount=${item.amount} ${pathText(network, item.path, source)}`));
347 flowList();
348 }
349 break;
350 }
351 case 'SENSITIVITY': {
352 const [source, sink] = endpoints(a, b), base = network.copy();
353 edmondsKarp(base, source, sink); const value = balances(base)[source], effective: { up: boolean; down: boolean }[] = [];
354 for (let i = 0; i < network.edges.length; i++) {
355 const up = network.copy(); up.edges[i].capacity++; edmondsKarp(up, source, sink);
356 let downEffective = false;
357 if (network.edges[i].capacity > 0) { const down = network.copy(); down.edges[i].capacity--; edmondsKarp(down, source, sink); downEffective = balances(down)[source] < value; }
358 effective.push({ up: balances(up)[source] > value, down: downEffective });
359 }
360 output.push(`SENSITIVITY ${a} ${b} value=${value} up=${effective.filter(item => item.up).length} down=${effective.filter(item => item.down).length}`);
361 effective.forEach((item, i) => { if (item.up || item.down) { const edge = network.edges[i]; output.push(`CRITICAL ${i + 1} ${network.names[edge.u]} ${network.names[edge.v]} up=${item.up ? 'yes' : 'no'} down=${item.down ? 'yes' : 'no'}`); } });
362 break;
363 }
364 case 'MINCOST': {
365 const [source, sink] = endpoints(a, b);
366 if (negativeCycle(network)) fail('NEGATIVE CYCLE');
367 const working = network.copy(), target = args[2] === 'MAX' ? null : Number(args[2]), result = minCost(working, source, sink, target);
368 if (target !== null && result.flow < target) { output.push(`MINCOST ${a} ${b} k=${args[2]} infeasible max=${result.flow}`); break; }
369 network.edges.forEach((edge, i) => edge.flow = working.edges[i].flow); network.owner = { kind: 'flow', source, sink };
370 output.push(`MINCOST ${a} ${b} k=${args[2]} flow=${result.flow} cost=${result.cost}`);
371 result.paths.forEach((item, i) => output.push(`AUG ${i + 1} amount=${item.amount} cost=${item.cost} ${pathText(working, item.path, source)}`));
372 flowList(); break;
373 }
374 case 'DISJOINT': {
375 const [source, sink] = endpoints(a, b), working = network.copy(), original = network.edges.length;
376 for (const edge of working.edges) edge.capacity = edge.capacity > 0 ? 1 : 0;
377 if (args[2] === 'VERTEX') {
378 const exits = working.names.map((_, i) => i);
379 for (let i = 0; i < network.names.length; i++) if (i !== source && i !== sink) { exits[i] = working.names.length; working.names.push(network.names[i]); working.demands.push(0); }
380 for (const edge of working.edges) edge.u = exits[edge.u];
381 for (let i = 0; i < network.names.length; i++) if (i !== source && i !== sink) working.edges.push({ u: i, v: exits[i], capacity: 1, cost: 0, lower: 0, flow: 0 });
382 }
383 edmondsKarp(working, source, sink);
384 const paths = decompose(working).filter(item => !item.cycle);
385 output.push(`DISJOINT ${a} ${b} ${args[2]} count=${paths.length}`);
386 paths.forEach((item, i) => output.push(`PATH ${i + 1} ${decompositionText(working, item, original)}`));
387 break;
388 }
389 case 'DECOMPOSE': {
390 const pieces = decompose(network);
391 output.push(`DECOMPOSE paths=${pieces.filter(item => !item.cycle).length} cycles=${pieces.filter(item => item.cycle).length}`);
392 let paths = 0, cycles = 0;
393 for (const item of pieces) output.push(`${item.cycle ? 'CYCLE' : 'PATH'} ${item.cycle ? ++cycles : ++paths} amount=${item.amount} ${decompositionText(network, item)}`);
394 break;
395 }
396 case 'VERIFY': {
397 const violations: string[] = [], balance = balances(network);
398 network.edges.forEach((edge, i) => { if (edge.flow < edge.lower) violations.push(`LOW ${i + 1} flow=${edge.flow} low=${edge.lower}`); });
399 network.names.forEach((name, i) => {
400 if (network.owner.kind === 'flow' && (network.owner.source === i || network.owner.sink === i)) return;
401 const need = network.owner.kind === 'flow' ? 0 : network.demands[i];
402 if (-balance[i] !== need) violations.push(`BALANCE ${name} net=${-balance[i]} need=${need}`);
403 });
404 output.push(...(violations.length ? [`VERIFY violations=${violations.length}`, ...violations] : ['VERIFY ok']));
405 break;
406 }
407 case 'CIRCULATION': {
408 if (a === 'MINCOST' && negativeCycle(network, true)) fail('NEGATIVE CYCLE');
409 const sum = network.demands.reduce((total, demand) => total + demand, 0);
410 if (sum !== 0) { output.push(`CIRCULATION infeasible unbalanced=${sum}`); break; }
411 const working = network.copy(), balance = [...network.demands], source = working.names.length, sink = source + 1;
412 working.names.push('superSource', 'superSink'); working.demands.push(0, 0);
413 working.edges.forEach((edge, i) => { const lower = network.edges[i].lower; edge.capacity -= lower; balance[edge.u] += lower; balance[edge.v] -= lower; });
414 for (let i = 0; i < balance.length; i++) if (balance[i] < 0) working.edges.push({ u: source, v: i, capacity: -balance[i], cost: 0, lower: 0, flow: 0 });
