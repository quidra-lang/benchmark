106def has_negative_cycle(node_count, edges):
107 distance = [0] * node_count
108 for _ in range(node_count + 1):
109 updated = False
110 for edge in edges:
111 if distance[edge.start] + edge.cost < distance[edge.end]:
112 distance[edge.end] = distance[edge.start] + edge.cost
113 updated = True
114 if not updated:
115 return False
116 return True

126def trace_piece(edges, remaining, balance, origin):
127 walk = [origin]
128 used = []
129 while True:
130 vertex = walk[-1]
131 index = next(
132 i for i, edge in enumerate(edges)
133 if edge.start == vertex and remaining[i] > 0
134 )
135 target = edges[index].end
136 used.append(index)
137 if target in walk:
138 cycle = used[walk.index(target):]
139 amount = min(remaining[i] for i in cycle)
140 for i in cycle:
141 remaining[i] -= amount
142 return FlowPiece(True, target, [edges[i] for i in cycle], amount)
143 walk.append(target)
144 if balance[target] < 0:
145 amount = min(min(remaining[i] for i in used), balance[origin], -balance[target])
146 for i in used:
147 remaining[i] -= amount
148 balance[origin] -= amount
149 balance[target] += amount
150 return FlowPiece(False, origin, [edges[i] for i in used], amount)

153def flow_decomposition(node_count, edges):
154 remaining = [edge.flow for edge in edges]
155 balance = [0] * node_count
156 for edge in edges:
157 balance[edge.start] += edge.flow
158 balance[edge.end] -= edge.flow
159 pieces = []
160 while first_positive(remaining) is not None:
161 origin = first_positive(balance)
162 if origin is None:
163 origin = edges[first_positive(remaining)].start
164 pieces.append(trace_piece(edges, remaining, balance, origin))
165 return pieces

168class FlowGraph:
169 def __init__(self, node_count, edges):
170 self.node_count = node_count
171 self.edges = edges
172 forward = [Arc(edge, True) for edge in edges]
173 backward = [Arc(edge, False) for edge in edges]
174 self.global_arcs = forward + backward
175 self.arcs = [[] for _ in range(node_count)]
176 for arc in self.global_arcs:
177 self.arcs[arc.start()].append(arc)
178
179 def breadth_first(self, source, threshold):
180 level = [None] * self.node_count
181 parent = [None] * self.node_count
182 level[source] = 0
183 queue = deque([source])
184 while queue:
185 vertex = queue.popleft()
186 for arc in self.arcs[vertex]:
187 target = arc.end()
188 if arc.residual() >= threshold and level[target] is None:
189 level[target] = level[vertex] + 1
190 parent[target] = arc
191 queue.append(target)
192 return level, parent
193
194 def single_augmentation(self, source, sink, threshold):
195 level, parent = self.breadth_first(source, threshold)
196 if level[sink] is None:
197 return None
198 path = traced_path(parent, source, sink)
199 amount = bottleneck(path)
200 push_path(path, amount)
201 return amount, path
202
203 def edmonds_karp(self, source, sink):
204 pushes = []
205 result = self.single_augmentation(source, sink, 1)
206 while result is not None:
207 pushes.append(result)
208 result = self.single_augmentation(source, sink, 1)
209 return pushes
210
211 def blocking_flow(self, source, sink, level):
212 pushes = []
213 pointer = [0] * self.node_count
214 path = []
215 vertex = source
216 while True:
217 if vertex == sink:
218 amount = bottleneck(path)
219 push_path(path, amount)
220 pushes.append((amount, path))
221 path = []
222 vertex = source
223 elif pointer[vertex] >= len(self.arcs[vertex]):
224 if vertex == source:
225 return pushes
226 vertex = path.pop().start()
227 pointer[vertex] += 1
228 else:
229 arc = self.arcs[vertex][pointer[vertex]]
230 target = arc.end()
231 if (
232 arc.residual() > 0
233 and level[target] is not None
234 and level[target] == level[vertex] + 1
235 ):
236 path.append(arc)
237 vertex = target
238 else:
239 pointer[vertex] += 1
240
241 def dinic(self, source, sink):
242 pushes = []
243 phases = 0
244 level, _ = self.breadth_first(source, 1)
245 while level[sink] is not None:
246 phases += 1
247 for amount, path in self.blocking_flow(source, sink, level):
248 pushes.append((phases, amount, path))
249 level, _ = self.breadth_first(source, 1)
250 return phases, pushes
251
252 def capacity_scaling(self, source, sink):
253 largest = max((edge.capacity for edge in self.edges), default=0)
254 delta = 1
255 while delta * 2 <= largest:
256 delta *= 2
257 pushes = []
258 while delta >= 1:
259 result = self.single_augmentation(source, sink, delta)
260 while result is not None:
261 amount, path = result
262 pushes.append((delta, amount, path))
263 result = self.single_augmentation(source, sink, delta)
264 delta //= 2
265 return pushes

267 def cheapest_path(self, source, sink):
268 usable = [arc for arc in self.global_arcs if arc.residual() > 0]
269 distance = [None] * self.node_count
270 parent = [None] * self.node_count
271 distance[source] = 0
272 updated = True
273 while updated:
274 updated = False
275 for arc in usable:
276 start, end = arc.start(), arc.end()
277 if distance[start] is None:
278 continue
279 candidate = distance[start] + arc.cost()
280 if distance[end] is None or candidate < distance[end]:
281 distance[end] = candidate
282 parent[end] = arc
283 updated = True
284 if distance[sink] is None:
285 return None
286 return distance[sink], traced_path(parent, source, sink)

288 def successive_shortest_paths(self, source, sink, target):
289 pushes = []
290 flow = 0
291 cost = 0
292 while target is None or flow < target:
293 result = self.cheapest_path(source, sink)
294 if result is None:
295 break
296 unit_cost, path = result
297 amount = bottleneck(path)
298 if target is not None:
299 amount = min(amount, target - flow)
300 push_path(path, amount)
301 flow += amount
302 cost += amount * unit_cost
303 pushes.append((amount, unit_cost, path))
304 return flow, cost, pushes

312class BipartiteMatching:
313 def __init__(self, left_count, right_count, links):
314 self.links = links
315 self.left_partner = [None] * left_count
316 self.right_partner = [None] * right_count
317 self.examined = []
318 for left in range(left_count):
319 self.examined = [False] * right_count
320 self.try_left(left)
321
322 def try_left(self, left):
323 for a, b in self.links:
324 if a != left or self.examined[b]:
325 continue
326 self.examined[b] = True
327 partner = self.right_partner[b]
328 if partner is None or self.try_left(partner):
329 self.right_partner[b] = left
330 self.left_partner[left] = b
331 return True
332 return False
333
334 def alternating_set(self):
335 left_in = [partner is None for partner in self.left_partner]
336 right_in = [False] * len(self.right_partner)
337 changed = True
338 while changed:
339 changed = False
340 for a, b in self.links:
341 if left_in[a] and not right_in[b] and self.left_partner[a] != b:
342 right_in[b] = True
343 changed = True
344 for b, partner in enumerate(self.right_partner):
345 if right_in[b] and partner is not None and not left_in[partner]:
346 left_in[partner] = True
347 changed = True
348 return left_in, right_in

351class AssignmentSolver:
352 def __init__(self, table, maximize):
353 self.table = table
354 self.maximize = maximize
355 self.size = len(table)
356 full = 1 << self.size
357 self.best = [[0] * full for _ in range(self.size + 1)]
358 self.ways = [[0] * full for _ in range(self.size)] + [[1] * full]
359 for row in range(self.size - 1, -1, -1):
360 for mask in range(full):
361 for _, total, count in self.options(row, mask):
362 if self.ways[row][mask] == 0 or self.improves(total, self.best[row][mask]):
363 self.best[row][mask] = total
364 self.ways[row][mask] = count
365 elif total == self.best[row][mask]:
366 self.ways[row][mask] += count
367
368 def improves(self, total, current):
369 return total > current if self.maximize else total < current
370
371 def options(self, row, mask):
372 result = []
373 for column in range(self.size):
374 bit = 1 << column
375 value = self.table[row][column]
376 following = mask | bit
377 if mask & bit or value is None or self.ways[row + 1][following] == 0:
378 continue
379 total = value + self.best[row + 1][following]
380 result.append((column, total, self.ways[row + 1][following]))
381 return result
382
383 def feasible(self):
384 return self.ways[0][0] > 0
385
386 def optimum(self):
387 return self.best[0][0], self.ways[0][0]
388
389 def chosen_columns(self):
390 columns = []
391 mask = 0
392 for row in range(self.size):
393 for column, total, _ in self.options(row, mask):
394 if total == self.best[row][mask]:
395 columns.append(column)
396 mask |= 1 << column
397 break
398 return columns

692 def sensitivity(self, arguments):
693 source, sink = self.require_endpoints(arguments)
694 network = self.network
695 count = network.node_count()
696 value = maximum_flow_amount(count, network.working_edges(), source, sink)
697 critical = []
698 for index, edge in enumerate(network.edges):
699 raised = network.working_edges()
700 raised[index].capacity += 1
701 up = maximum_flow_amount(count, raised, source, sink) > value
702 down = False
703 if edge.capacity >= 1:
704 lowered = network.working_edges()
705 lowered[index].capacity -= 1
706 down = maximum_flow_amount(count, lowered, source, sink) < value
707 if up or down:
708 critical.append((edge, up, down))
709 ups = sum(up for _, up, _ in critical)
710 downs = sum(down for _, _, down in critical)
711 self.emit(
712 "SENSITIVITY", arguments[0], arguments[1],
713 f"value={value}", f"up={ups}", f"down={downs}",
714 )
715 for edge, up, down in critical:
716 self.emit(
717 "CRITICAL", edge.number, self.name(edge.start), self.name(edge.end),
718 "up=" + ("yes" if up else "no"), "down=" + ("yes" if down else "no"),
719 )

721 def mincost(self, arguments):
722 source, sink = self.require_endpoints(arguments)
723 network = self.network
724 usable = [edge for edge in network.edges if edge.capacity >= 1]
725 if has_negative_cycle(network.node_count(), usable):
726 raise CommandError("NEGATIVE CYCLE")
727 target = None if arguments[2] == "MAX" else int(arguments[2])
728 work = network.working_edges()
729 graph = FlowGraph(network.node_count(), work)
730 flow, cost, pushes = graph.successive_shortest_paths(source, sink, target)
731 prefix = ("MINCOST", arguments[0], arguments[1], f"k={arguments[2]}")
732 if target is not None and flow < target:
733 self.emit(*prefix, "infeasible", f"max={flow}")
734 return
735 for edge, worked in zip(network.edges, work):
736 edge.flow = worked.flow
737 network.set_owner("flow", source, sink)
738 self.emit(*prefix, f"flow={flow}", f"cost={cost}")
739 for number, (amount, unit_cost, path) in enumerate(pushes, 1):
740 self.emit(
741 "AUG", number, f"amount={amount}", f"cost={unit_cost}",
742 self.notation(source, path),
743 )
744 self.flow_list()

746 def disjoint(self, arguments):
747 source, sink = self.require_endpoints(arguments)
748 mode = arguments[2]
749 network = self.network
750 count = network.node_count()
751 exits = list(range(count))
752 node_total = count
753 if mode == "VERTEX":
754 node_total = 2 * count
755 for vertex in range(count):
756 if vertex != source and vertex != sink:
757 exits[vertex] = count + vertex
758 work = [
759 Edge(edge.number, exits[edge.start], edge.end, min(edge.capacity, 1), 0)
760 for edge in network.edges
761 ]
762 if mode == "VERTEX":
763 for vertex in range(count):
764 if vertex != source and vertex != sink:
765 work.append(Edge(len(work) + 1, vertex, exits[vertex], 1, 0))
766 FlowGraph(node_total, work).edmonds_karp(source, sink)
767 paths = [piece for piece in flow_decomposition(node_total, work) if not piece.cycle]
768 self.emit("DISJOINT", arguments[0], arguments[1], mode, f"count={len(paths)}")
769 limit = len(network.edges)
770 for number, piece in enumerate(paths, 1):
771 original = [edge for edge in piece.edges if edge.number <= limit]
772 self.emit("PATH", number, self.notation(piece.start, forward_arcs(original)))

774 def decompose(self, arguments):
775 network = self.network
776 pieces = flow_decomposition(network.node_count(), network.edges)
777 cycles = sum(piece.cycle for piece in pieces)
778 self.emit("DECOMPOSE", f"paths={len(pieces) - cycles}", f"cycles={cycles}")
779 path_number = 0
780 cycle_number = 0
781 for piece in pieces:
782 notation = self.notation(piece.start, forward_arcs(piece.edges))
783 if piece.cycle:
784 cycle_number += 1
785 self.emit("CYCLE", cycle_number, f"amount={piece.amount}", notation)
786 else:
787 path_number += 1
788 self.emit("PATH", path_number, f"amount={piece.amount}", notation)

790 def verify(self, arguments):
791 network = self.network
792 violations = []
793 for edge in network.edges:
794 if edge.flow < edge.lower:
795 violations.append(f"LOW {edge.number} flow={edge.flow} low={edge.lower}")
796 for vertex, name in enumerate(network.names):
797 if network.owner == "flow":
798 if vertex == network.source or vertex == network.sink:
799 continue
800 need = 0
801 else:
802 need = network.demands[vertex]
803 net = network.inflow(vertex) - network.outflow(vertex)
804 if net != need:
805 violations.append(f"BALANCE {name} net={net} need={need}")
806 if not violations:
807 self.emit("VERIFY", "ok")
808 return
809 self.emit("VERIFY", f"violations={len(violations)}")
810 for violation in violations:
811 self.emit(violation)

813 def circulation(self, arguments):
814 mode = arguments[0]
815 network = self.network
816 count = network.node_count()
817 if mode == "MINCOST":
818 usable = [edge for edge in network.edges if edge.capacity - edge.lower >= 1]
819 if has_negative_cycle(count, usable):
820 raise CommandError("NEGATIVE CYCLE")
821 imbalance = sum(network.demands)
822 if imbalance != 0:
823 self.emit("CIRCULATION", "infeasible", f"unbalanced={imbalance}")
824 return
825 required = list(network.demands)
826 for edge in network.edges:
827 required[edge.end] -= edge.lower
828 required[edge.start] += edge.lower
829 super_source = count
830 super_sink = count + 1
831 work = [
832 Edge(edge.number, edge.start, edge.end, edge.capacity - edge.lower, edge.cost)
833 for edge in network.edges
834 ]
835 for vertex in range(count):
836 if required[vertex] < 0:
837 work.append(Edge(len(work) + 1, super_source, vertex, -required[vertex], 0))
838 sink_edges = []
839 for vertex in range(count):
840 if required[vertex] > 0:
841 edge = Edge(len(work) + 1, vertex, super_sink, required[vertex], 0)
842 work.append(edge)
843 sink_edges.append(edge)
844 graph = FlowGraph(count + 2, work)
845 if mode == "ANY":
846 graph.edmonds_karp(super_source, super_sink)
847 else:
848 graph.successive_shortest_paths(super_source, super_sink, None)
849 total_required = sum(edge.capacity for edge in sink_edges)
850 achieved = sum(edge.flow for edge in sink_edges)
851 if achieved < total_required:
852 self.emit(
853 "CIRCULATION", "infeasible",
854 f"required={total_required}", f"achieved={achieved}",
855 )
856 level, _ = graph.breadth_first(super_source, 1)
857 self.emit("CUT", *(network.names[v] for v in range(count) if level[v] is not None))
858 return
859 for edge, worked in zip(network.edges, work):
860 edge.flow = edge.lower + worked.flow
861 network.set_owner("circulation", None, None)
862 self.emit(
863 "CIRCULATION", "feasible",
864 f"required={total_required}", f"cost={network.total_cost()}",
865 )
866 self.flow_list()

868 def match(self, arguments):
869 left_names = arguments[0].split(",")
870 right_names = arguments[1].split(",")
871 pairs = [] if arguments[2] == "-" else arguments[2].split(",")
872 left_index = {}
873 for name in left_names:
874 if name in left_index:
875 raise CommandError(f"DUPLICATE VERTEX {name}")
876 left_index[name] = len(left_index)
877 right_index = {}
878 for name in right_names:
879 if name in right_index or name in left_index:
880 raise CommandError(f"DUPLICATE VERTEX {name}")
881 right_index[name] = len(right_index)
882 links = []
883 for pair in pairs:
884 left, right = pair.split("-")
885 if left not in left_index:
886 raise CommandError(f"UNKNOWN VERTEX {left}")
887 if right not in right_index:
888 raise CommandError(f"UNKNOWN VERTEX {right}")
889 link = (left_index[left], right_index[right])
890 if link in links:
891 raise CommandError(f"DUPLICATE PAIR {left}-{right}")
892 links.append(link)
893 matching = BipartiteMatching(len(left_names), len(right_names), links)
894 left_partner = matching.left_partner
895 right_partner = matching.right_partner
896 size = sum(partner is not None for partner in left_partner)
897 self.emit("MATCH", f"size={size}", f"left={len(left_names)}", f"right={len(right_names)}")
898 for a, partner in enumerate(left_partner):
899 if partner is not None:
900 self.emit("PAIR", left_names[a], right_names[partner])
901 free_left = [left_names[a] for a, p in enumerate(left_partner) if p is None]
902 free_right = [right_names[b] for b, p in enumerate(right_partner) if p is None]
903 self.emit("FREE", ",".join(free_left) or "-", ",".join(free_right) or "-")
904 left_in, right_in = matching.alternating_set()
905 cover = [name for a, name in enumerate(left_names) if not left_in[a]]
906 cover += [name for b, name in enumerate(right_names) if right_in[b]]
907 self.emit("COVER", f"size={len(cover)}", *cover)
908 if free_left:
909 hall_left = [name for a, name in enumerate(left_names) if left_in[a]]
910 hall_right = [name for b, name in enumerate(right_names) if right_in[b]]
911 self.emit("HALL", ",".join(hall_left), ",".join(hall_right) or "-")

913 def assign(self, arguments):
914 mode = arguments[0]
915 size = int(arguments[1])
916 values = arguments[2:]
917 if size < MIN_ASSIGN_SIZE or size > MAX_ASSIGN_SIZE:
918 raise CommandError("BAD SIZE")
919 if len(values) != size * size:
920 raise CommandError("BAD LENGTH")
921 table = [
922 [None if text == "x" else int(text) for text in values[row * size:(row + 1) * size]]
923 for row in range(size)
924 ]
925 solver = AssignmentSolver(table, mode == "MAX")
926 if not solver.feasible():
927 self.emit("ASSIGN", mode, f"n={size}", "infeasible")
928 return
929 total, optimal = solver.optimum()
930 self.emit("ASSIGN", mode, f"n={size}", f"total={total}", f"optimal={optimal}")
931 for row, column in enumerate(solver.chosen_columns()):
932 self.emit("MATE", row + 1, column + 1, table[row][column])
