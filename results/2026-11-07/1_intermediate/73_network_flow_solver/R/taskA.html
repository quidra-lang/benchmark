18arc_from <- function(network, arc) network$edges[abs(arc), if (arc > 0L) "u" else "v"]
19arc_to <- function(network, arc) network$edges[abs(arc), if (arc > 0L) "v" else "u"]
20residual <- function(network, arc) {
21 e <- network$edges[abs(arc), ]
22 if (arc > 0L) e[["capacity"]] - e[["flow"]] else e[["flow"]]
23}
24arc_cost <- function(network, arc) network$edges[abs(arc), "cost"] * sign(arc)
25push <- function(network, path, amount) {
26 for (arc in path) network$edges[abs(arc), "flow"] <- network$edges[abs(arc), "flow"] + sign(arc) * amount
27}
28bottleneck <- function(network, path) min(vapply(path, function(arc) residual(network, arc), numeric(1)))
29bfs <- function(network, source, threshold = 1L) {
30 levels <- rep(NA_integer_, network$n)
31 parents <- integer(network$n)
32 levels[[source]] <- 0L
33 queue <- source
34 at <- 1L
35 while (at <= length(queue)) {
36 u <- queue[[at]]
37 at <- at + 1L
38 for (arc in arcs(network, u)) {
39 v <- arc_to(network, arc)
40 if (residual(network, arc) >= threshold && is.na(levels[[v]])) {
41 levels[[v]] <- levels[[u]] + 1L
42 parents[[v]] <- arc
43 queue <- c(queue, v)
44 }
45 }
46 }
47 list(levels = levels, parents = parents)
48}
49path_to <- function(network, source, target, parents) {
50 path <- integer()
51 while (target != source) {
52 arc <- parents[[target]]
53 path <- c(path, arc)
54 target <- arc_from(network, arc)
55 }
56 rev(path)
57}
58augment <- function(network, source, target, threshold = 1L) {
59 search <- bfs(network, source, threshold)
60 if (is.na(search$levels[[target]])) return(NULL)
61 path <- path_to(network, source, target, search$parents)
62 amount <- bottleneck(network, path)
63 push(network, path, amount)
64 list(path = path, amount = amount, delta = threshold)

66maxflow <- function(network, source, target, method = "MAXFLOW") {
67 records <- list()
68 phases <- 0L
69 if (method == "DINIC") {
70 repeat {
71 levels <- bfs(network, source)$levels
72 if (is.na(levels[[target]])) break
73 phases <- phases + 1L
74 lists <- lapply(seq_len(network$n), function(v) arcs(network, v))
75 pointers <- rep(1L, network$n)
76 path <- integer()
77 u <- source
78 repeat {
79 if (u == target) {
80 amount <- bottleneck(network, path)
81 push(network, path, amount)
82 records[[length(records) + 1L]] <- list(path = path, amount = amount, phase = phases)
83 path <- integer()
84 u <- source
85 } else if (pointers[[u]] > length(lists[[u]])) {
86 if (u == source) break
87 u <- arc_from(network, tail(path, 1L))
88 path <- head(path, -1L)
89 pointers[[u]] <- pointers[[u]] + 1L
90 } else {
91 arc <- lists[[u]][[pointers[[u]]]]
92 v <- arc_to(network, arc)
93 if (residual(network, arc) > 0L && !is.na(levels[[v]]) && levels[[v]] == levels[[u]] + 1L) {
94 path <- c(path, arc)
95 u <- v
96 } else {
97 pointers[[u]] <- pointers[[u]] + 1L
98 }
99 }
100 }
101 }
102 } else {
103 threshold <- 1L
104 maximum <- max(c(0L, network$edges[, "capacity"]))
105 while (method == "SCALING" && threshold * 2L <= maximum) threshold <- threshold * 2L
106 while (threshold >= 1L) {
107 repeat {
108 record <- augment(network, source, target, threshold)
109 if (is.null(record)) break
110 records[[length(records) + 1L]] <- record
111 }
112 threshold <- if (method == "SCALING") threshold %/% 2L else 0L
113 }
114 }
115 list(records = records, phases = phases)
116}
117negative_cycle <- function(network) {
118 distance <- rep(0, network$n)
119 for (pass in seq_len(network$n)) {
120 changed <- FALSE
121 for (i in seq_len(nrow(network$edges))) {
122 e <- network$edges[i, ]
123 if (e[["capacity"]] > 0L && distance[[e[["u"]]]] + e[["cost"]] < distance[[e[["v"]]]]) {
124 distance[[e[["v"]]]] <- distance[[e[["u"]]]] + e[["cost"]]
125 changed <- TRUE
126 }
127 }
128 if (!changed) return(FALSE)
129 }
130 network$n > 0L
131}
132mincost <- function(network, source, target, limit = NULL) {
133 amount <- 0
134 cost <- 0
135 records <- list()
136 while (is.null(limit) || amount < limit) {
137 usable <- Filter(function(arc) residual(network, arc) > 0L, arcs(network))
138 distance <- rep(Inf, network$n)
139 parents <- integer(network$n)
140 distance[[source]] <- 0
141 repeat {
142 changed <- FALSE
143 for (arc in usable) {
144 u <- arc_from(network, arc)
145 v <- arc_to(network, arc)
146 if (!is.finite(distance[[u]])) next
147 value <- distance[[u]] + arc_cost(network, arc)
148 if (value < distance[[v]]) {
149 distance[[v]] <- value
150 parents[[v]] <- arc
151 changed <- TRUE
152 }
153 }
154 if (!changed) break
155 }
156 if (!is.finite(distance[[target]])) break
157 path <- path_to(network, source, target, parents)
158 delta <- bottleneck(network, path)
159 if (!is.null(limit)) delta <- min(delta, limit - amount)
160 push(network, path, delta)
161 amount <- amount + delta
162 cost <- cost + delta * distance[[target]]
163 records[[length(records) + 1L]] <- list(path = path, amount = delta, cost = distance[[target]])
164 }
165 list(amount = amount, cost = cost, records = records)
166}
167balance <- function(network) {
168 result <- rep(0, network$n)
169 for (i in seq_len(nrow(network$edges))) {
170 e <- network$edges[i, ]
171 result[[e[["u"]]]] <- result[[e[["u"]]]] + e[["flow"]]
172 result[[e[["v"]]]] <- result[[e[["v"]]]] - e[["flow"]]
173 }
174 result
175}
176total_cost <- function(network) sum(network$edges[, "flow"] * network$edges[, "cost"])
177decompose <- function(network) {
178 remaining <- network$edges[, "flow"]
179 balances <- balance(network)
180 result <- list()
181 while (any(remaining > 0L)) {
182 positive <- which(remaining > 0L)
183 supplies <- which(balances > 0L)
184 start <- if (length(supplies)) supplies[[1]] else network$edges[positive[[1]], "u"]
185 vertices <- start
186 path <- integer()
187 repeat {
188 u <- tail(vertices, 1L)
189 i <- which(network$edges[, "u"] == u & remaining > 0L)[[1]]
190 v <- network$edges[i, "v"]
191 path <- c(path, i)
192 q <- match(v, vertices)
193 if (!is.na(q)) {
194 cycle <- path[q:length(path)]
195 delta <- min(remaining[cycle])
196 remaining[cycle] <- remaining[cycle] - delta
197 result[[length(result) + 1L]] <- list(kind = "CYCLE", start = v, path = cycle, amount = delta)
198 break
199 }
200 vertices <- c(vertices, v)
201 if (balances[[v]] < 0L) {
202 delta <- min(remaining[path], balances[[start]], -balances[[v]])
203 remaining[path] <- remaining[path] - delta
204 balances[[start]] <- balances[[start]] - delta
205 balances[[v]] <- balances[[v]] + delta
206 result[[length(result) + 1L]] <- list(kind = "PATH", start = start, path = path, amount = delta)
207 break
208 }
209 }
210 }
211 result
212}

263circulation <- function(mode) {
264 work <- fresh()
265 work$edges[, "capacity"] <- work$edges[, "capacity"] - work$edges[, "lower"]
266 if (mode == "MINCOST" && negative_cycle(work)) fail("NEGATIVE CYCLE")
267 total <- sum(state$demands)
268 if (total != 0L) {
269 say("CIRCULATION infeasible unbalanced=", total)
270 return(invisible(NULL))
271 }
272 residuals <- state$demands
273 for (i in seq_len(nrow(state$network$edges))) {
274 e <- state$network$edges[i, ]
275 residuals[[e[["v"]]]] <- residuals[[e[["v"]]]] - e[["lower"]]
276 residuals[[e[["u"]]]] <- residuals[[e[["u"]]]] + e[["lower"]]
277 }
278 source <- length(state$names) + 1L
279 target <- source + 1L
280 for (v in which(residuals < 0L)) work$edges <- rbind(work$edges, new_edge(source, v, -residuals[[v]]))
281 sinks <- integer()
282 for (v in which(residuals > 0L)) {
283 sinks <- c(sinks, nrow(work$edges) + 1L)
284 work$edges <- rbind(work$edges, new_edge(v, target, residuals[[v]]))
285 }
286 work$n <- target
287 if (mode == "ANY") maxflow(work, source, target) else mincost(work, source, target)
288 required <- sum(residuals[residuals > 0L])
289 achieved <- sum(work$edges[sinks, "flow"])
290 if (achieved < required) {
291 say("CIRCULATION infeasible required=", required, " achieved=", achieved)
292 reachable <- bfs(work, source)$levels[seq_along(state$names)]
293 say("CUT ", paste(state$names[!is.na(reachable)], collapse = " "))
294 } else {
295 ids <- seq_len(nrow(state$network$edges))
296 state$network$edges[, "flow"] <- state$network$edges[, "lower"] + work$edges[ids, "flow"]
297 state$owner <- list(kind = "circulation")
298 say("CIRCULATION feasible required=", required, " cost=", total_cost(state$network))
299 flow_list()
300 }

302matching <- function(left_text, right_text, pair_text) {
303 left <- strsplit(left_text, ",", fixed = TRUE)[[1]]
304 right <- strsplit(right_text, ",", fixed = TRUE)[[1]]
305 seen <- character()
306 for (name in c(left, right)) {
307 if (name %in% seen) fail(paste("DUPLICATE VERTEX", name))
308 seen <- c(seen, name)
309 }
310 pairs <- if (pair_text == "-") list() else lapply(strsplit(pair_text, ",", fixed = TRUE)[[1]], function(p) strsplit(p, "-", fixed = TRUE)[[1]])
311 seen_pairs <- character()
312 for (pair in pairs) {
313 a <- pair[[1]]
314 b <- pair[[2]]
315 if (!(a %in% left)) fail(paste("UNKNOWN VERTEX", a))
316 if (!(b %in% right)) fail(paste("UNKNOWN VERTEX", b))
317 key <- paste0(a, "-", b)
318 if (key %in% seen_pairs) fail(paste("DUPLICATE PAIR", key))
319 seen_pairs <- c(seen_pairs, key)
320 }
321 lp <- list()
322 rp <- list()
323 examined <- character()
324 attempt <- function(a) {
325 for (pair in pairs) {
326 b <- pair[[2]]
327 if (pair[[1]] != a || b %in% examined) next
328 examined <<- c(examined, b)
329 if (is.null(rp[[b]]) || attempt(rp[[b]])) {
330 rp[[b]] <<- a
331 lp[[a]] <<- b
332 return(TRUE)
333 }
334 }
335 FALSE
336 }
337 for (a in left) {
338 examined <- character()
339 attempt(a)
340 }
341 free_left <- setdiff(left, names(lp))
342 free_right <- setdiff(right, names(rp))
343 zl <- free_left
344 zr <- character()
345 at <- 1L
346 while (at <= length(zl)) {
347 a <- zl[[at]]
348 at <- at + 1L
349 for (pair in pairs) {
350 b <- pair[[2]]
351 if (pair[[1]] == a && !identical(lp[[a]], b) && !(b %in% zr)) {
352 zr <- c(zr, b)
353 if (!is.null(rp[[b]]) && !(rp[[b]] %in% zl)) zl <- c(zl, rp[[b]])
354 }
355 }
356 }
357 cover <- c(left[!(left %in% zl)], right[right %in% zr])
358 say("MATCH size=", length(lp), " left=", length(left), " right=", length(right))
359 for (a in left) if (!is.null(lp[[a]])) say("PAIR ", a, " ", lp[[a]])
360 list_text <- function(x) if (length(x)) paste(x, collapse = ",") else "-"
361 say("FREE ", list_text(free_left), " ", list_text(free_right))
362 say("COVER size=", length(cover), if (length(cover)) paste0(" ", paste(cover, collapse = " ")) else "")
363 if (length(free_left)) say("HALL ", list_text(left[left %in% zl]), " ", list_text(right[right %in% zr]))
364}
365assignment <- function(mode, n, values) {
366 if (n < 1L || n > 8L) fail("BAD SIZE")
367 if (length(values) != n * n) fail("BAD LENGTH")
368 best <- NULL
369 count <- 0L
370 chosen <- integer()
371 search <- function(columns, total) {
372 row <- length(columns)
373 if (row == n) {
374 if (is.null(best) || (if (mode == "MIN") total < best else total > best)) {
375 best <<- total
376 count <<- 1L
377 chosen <<- columns
378 } else if (total == best) {
379 count <<- count + 1L
380 }
381 return(invisible(NULL))
382 }
383 for (column in seq_len(n)) {
384 value <- values[[row * n + column]]
385 if (!(column %in% columns) && value != "x") search(c(columns, column), total + as.integer(value))
386 }
387 }
388 search(integer(), 0L)
389 if (is.null(best)) {
390 say("ASSIGN ", mode, " n=", n, " infeasible")
391 } else {
392 say("ASSIGN ", mode, " n=", n, " total=", best, " optimal=", count)
393 for (row in seq_len(n)) say("MATE ", row, " ", chosen[[row]], " ", values[[(row - 1L) * n + chosen[[row]]]])
394 }

396flow_command <- function(command, args, source, target) {
397 prefix <- paste(command, args[[1]], args[[2]])
398 if (command == "SENSITIVITY") {
399 maximum <- function(index = NULL, delta = 0L) {
400 work <- fresh()
401 if (!is.null(index)) work$edges[index, "capacity"] <- work$edges[index, "capacity"] + delta
402 maxflow(work, source, target)
403 balance(work)[[source]]
404 }
405 value <- maximum()
406 critical <- lapply(seq_len(nrow(state$network$edges)), function(i) {
407 list(i = i, up = maximum(i, 1L) > value, down = state$network$edges[i, "capacity"] > 0L && maximum(i, -1L) < value)
408 })
409 say(prefix, " value=", value, " up=", sum(vapply(critical, function(r) r$up, logical(1))),
410 " down=", sum(vapply(critical, function(r) r$down, logical(1))))
411 for (r in critical) {
412 if (!(r$up || r$down)) next
413 e <- state$network$edges[r$i, ]
414 say("CRITICAL ", r$i, " ", state$names[[e[["u"]]]], " ", state$names[[e[["v"]]]],
415 " up=", if (r$up) "yes" else "no", " down=", if (r$down) "yes" else "no")
416 }
417 } else if (command == "MINCOST") {
418 work <- fresh()
419 if (negative_cycle(work)) fail("NEGATIVE CYCLE")
420 limit <- if (args[[3]] == "MAX") NULL else as.integer(args[[3]])
421 result <- mincost(work, source, target, limit)
422 if (!is.null(limit) && result$amount < limit) {
423 say(prefix, " k=", args[[3]], " infeasible max=", result$amount)
424 return(invisible(NULL))
425 }
426 state$network$edges[, "flow"] <- work$edges[, "flow"]
427 state$owner <- list(kind = "flow", source = source, target = target)
428 say(prefix, " k=", args[[3]], " flow=", result$amount, " cost=", result$cost)
429 for (i in seq_along(result$records)) {
430 r <- result$records[[i]]
431 say("AUG ", i, " amount=", r$amount, " cost=", r$cost, " ", notation(work, source, r$path))
432 }
433 flow_list()
434 } else if (command == "DISJOINT") {
435 work <- fresh()
436 work$edges[, "capacity"] <- as.integer(work$edges[, "capacity"] > 0L)
437 work$edges[, "cost"] <- 0L
438 if (args[[3]] == "VERTEX") {
439 exits <- seq_along(state$names)
440 for (v in seq_along(state$names)) {
441 if (v == source || v == target) next
442 work$n <- work$n + 1L
443 exits[[v]] <- work$n
444 }
445 work$edges[, "u"] <- exits[work$edges[, "u"]]
446 for (v in seq_along(state$names)) {
447 if (v != source && v != target) work$edges <- rbind(work$edges, new_edge(v, exits[[v]], 1L))
448 }
449 }
450 maxflow(work, source, target)
451 records <- Filter(function(r) r$kind == "PATH", decompose(work))
452 say(prefix, " ", args[[3]], " count=", length(records))
453 for (i in seq_along(records)) say("PATH ", i, " ", notation(work, source, records[[i]]$path, nrow(state$network$edges)))
454 } else {
455 start <- start_flow(source, target)
456 if (command == "AUGMENT") {
457 r <- augment(state$network, source, target)
458 ending <- if (is.null(r)) "none" else paste0("amount=", r$amount, " ", notation(state$network, source, r$path))
459 say(prefix, " value=", balance(state$network)[[source]], " ", ending)
460 return(invisible(NULL))
461 }
462 result <- maxflow(state$network, source, target, command)
463 value <- balance(state$network)[[source]]
464 if (command == "MINCUT") {
465 reachable <- !is.na(bfs(state$network, source)$levels)
466 cut <- which(reachable[state$network$edges[, "u"]] & !reachable[state$network$edges[, "v"]])
467 say(prefix, " value=", value, " edges=", length(cut))
468 say("SOURCE ", paste(state$names[reachable], collapse = " "))
469 say("SINK ", paste(state$names[!reachable], collapse = " "))
470 for (i in cut) {
471 e <- state$network$edges[i, ]
472 say("CUT ", i, " ", state$names[[e[["u"]]]], " ", state$names[[e[["v"]]]], " ", e[["capacity"]])
473 }
474 } else {
475 say(prefix, " value=", value, " start=", start,
476 if (command == "DINIC") paste0(" phases=", result$phases) else "", " paths=", length(result$records))
477 for (i in seq_along(result$records)) {
478 r <- result$records[[i]]
479 extra <- if (command == "DINIC") paste0("phase=", r$phase, " ") else if (command == "SCALING") paste0("delta=", r$delta, " ") else ""
480 say("AUG ", i, " ", extra, "amount=", r$amount, " ", notation(state$network, source, r$path))
481 }
482 flow_list()
483 }
484 }

487 words <- strsplit(line, " ", fixed = TRUE)[[1]]
488 command <- words[[1]]
489 args <- words[-1]
490 switch(command,
491 NODE = {
492 seen <- state$names

513 CAP = edit_edge(command, args),

528 },

539 }

543 }

545 e <- state$network$edges[i, ]
546 say("EDGE ", i, " ", state$names[[e[["u"]]]], " ", state$names[[e[["v"]]]], " cap=", e[["capacity"]],
547 " cost=", e[["cost"]], " low=", e[["lower"]], " flow=", e[["flow"]])
548 }
549 },

551 records <- decompose(state$network)
552 count_paths <- sum(vapply(records, function(r) r$kind == "PATH", logical(1)))
553 say("DECOMPOSE paths=", count_paths, " cycles=", length(records) - count_paths)
554 counters <- c(PATH = 0L, CYCLE = 0L)
555 for (r in records) {
556 counters[[r$kind]] <- counters[[r$kind]] + 1L
557 say(r$kind, " ", counters[[r$kind]], " amount=", r$amount, " ", notation(state$network, r$start, r$path))
558 }
559 },
560 VERIFY = {
561 violations <- character()

598 state$network$edges[i, "capacity"] <- value

600 } else {
