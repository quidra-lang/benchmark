3class Network {
4 names = [];
5 edges = [];
6 demands = [];
7 owner = { kind: 'none', source: -1, sink: -1 };

18function fail(message) { throw new Error('ERROR ' + message); }
19function vertex(name) {
20 const index = network.names.indexOf(name);
21 if (index < 0) fail('UNKNOWN NODE ' + name);
22 return index;
23}
24function endpoints(a, b) {
25 const source = vertex(a), sink = vertex(b);
26 if (source === sink) fail('SAME NODE');
27 return [source, sink];
28}
29function edgeAt(word) {
30 const index = Number(word) - 1;
31 if (index < 0 || index >= network.edges.length) fail('UNKNOWN EDGE ' + word);
32 return network.edges[index];
33}
34function balances(net) {
35 const result = Array(net.names.length).fill(0);
36 for (const edge of net.edges) { result[edge.u] += edge.flow; result[edge.v] -= edge.flow; }
37 return result;
38}
39function totalCost(net) { return net.edges.reduce((sum, edge) => sum + edge.flow * edge.cost, 0); }
40function globalArcs(net) {
41 const forward = net.edges.map((edge, index) => ({ edge: index, forward: true, from: edge.u, to: edge.v }));
42 const backward = net.edges.map((edge, index) => ({ edge: index, forward: false, from: edge.v, to: edge.u }));
43 return [...forward, ...backward];
44}
45function adjacency(net) {
46 const result = Array.from({ length: net.names.length }, () => []);
47 for (const arc of globalArcs(net)) result[arc.from].push(arc);
48 return result;
49}
50function residual(net, arc) {
51 const edge = net.edges[arc.edge];
52 return arc.forward ? edge.capacity - edge.flow : edge.flow;
53}
54function push(net, path, amount) {
55 for (const arc of path) net.edges[arc.edge].flow += arc.forward ? amount : -amount;
56}
57function bottleneck(net, path) { return Math.min(...path.map(arc => residual(net, arc))); }
58function bfs(net, source, threshold = 1) {
59 const level = Array(net.names.length).fill(-1), parent = Array(net.names.length).fill(null), queue = [source];
60 const lists = adjacency(net);
61 level[source] = 0;
62 for (let i = 0; i < queue.length; i++) {
63 const u = queue[i];
64 for (const arc of lists[u]) if (residual(net, arc) >= threshold && level[arc.to] < 0) {
65 level[arc.to] = level[u] + 1; parent[arc.to] = arc; queue.push(arc.to);
66 }
67 }
68 return { level, parent };
69}
70function restorePath(parent, source, sink) {
71 const result = [];
72 for (let at = sink; at !== source;) { const arc = parent[at]; result.push(arc); at = arc.from; }
73 return result.reverse();
74}
75function single(net, source, sink, threshold = 1) {
76 const search = bfs(net, source, threshold);
77 if (search.level[sink] < 0) return null;
78 const path = restorePath(search.parent, source, sink), amount = bottleneck(net, path);
79 push(net, path, amount);
80 return { path, amount, phase: 0, delta: threshold, cost: 0 };
81}
82function edmondsKarp(net, source, sink) {
83 const paths = [];
84 while (true) { const augmentation = single(net, source, sink); if (augmentation === null) return paths; paths.push(augmentation); }
85}
86function dinic(net, source, sink) {
87 const paths = [];
88 let phases = 0;
89 const lists = adjacency(net);
90 while (true) {
91 const { level } = bfs(net, source);
92 if (level[sink] < 0) return { paths, phases };
93 phases++;
94 const pointers = Array(net.names.length).fill(0);
95 let path = [], current = source;
96 while (true) {
97 if (current === sink) {
98 const amount = bottleneck(net, path);
99 push(net, path, amount);
100 paths.push({ path: [...path], amount, phase: phases, delta: 1, cost: 0 });
101 path = []; current = source;
102 } else if (pointers[current] >= lists[current].length) {
103 if (current === source) break;
104 const arc = path.pop(); current = arc.from; pointers[current]++;
105 } else {
106 const arc = lists[current][pointers[current]];
107 if (residual(net, arc) > 0 && level[arc.to] === level[current] + 1) { path.push(arc); current = arc.to; }
108 else pointers[current]++;
109 }
110 }
111 }
112}
113function scaling(net, source, sink) {
114 const paths = [], maximum = Math.max(1, ...net.edges.map(edge => edge.capacity));
115 let delta = 1;
116 while (delta * 2 <= maximum) delta *= 2;
117 while (delta >= 1) {
118 while (true) { const augmentation = single(net, source, sink, delta); if (augmentation === null) break; paths.push(augmentation); }
119 delta = Math.floor(delta / 2);
120 }
121 return paths;
122}
123function negativeCycle(net, lowerBounds = false) {
124 const distance = Array(net.names.length).fill(0);
125 for (let pass = 0; pass < net.names.length; pass++) {
126 let changed = false;
127 for (const edge of net.edges) if (edge.capacity - (lowerBounds ? edge.lower : 0) > 0 && distance[edge.v] > distance[edge.u] + edge.cost) {
128 distance[edge.v] = distance[edge.u] + edge.cost; changed = true;
129 }
130 if (!changed) return false;
131 if (pass + 1 === net.names.length) return true;
132 }
133 return false;
134}
135function shortest(net, source, sink) {
136 const arcs = globalArcs(net).filter(arc => residual(net, arc) > 0);
137 const distance = Array(net.names.length).fill(null), parent = Array(net.names.length).fill(null);
138 distance[source] = 0;
139 while (true) {
140 let changed = false;
141 for (const arc of arcs) {
142 if (distance[arc.from] === null) continue;
143 const cost = net.edges[arc.edge].cost * (arc.forward ? 1 : -1), next = distance[arc.from] + cost;
144 if (distance[arc.to] === null || next < distance[arc.to]) { distance[arc.to] = next; parent[arc.to] = arc; changed = true; }
145 }
146 if (!changed) break;
147 }
148 return distance[sink] === null ? null : { path: restorePath(parent, source, sink), cost: distance[sink] };
149}
150function minCost(net, source, sink, target = null) {
151 const paths = [];
152 let flow = 0, cost = 0;
153 while (target === null || flow < target) {
154 const cheapest = shortest(net, source, sink);
155 if (cheapest === null) break;
156 const amount = Math.min(bottleneck(net, cheapest.path), target === null ? Infinity : target - flow);
157 push(net, cheapest.path, amount); flow += amount; cost += amount * cheapest.cost;
158 paths.push({ path: cheapest.path, amount, phase: 0, delta: 1, cost: cheapest.cost });
159 }
160 return { paths, flow, cost };
161}
162function decompose(net) {
163 const remaining = net.edges.map(edge => edge.flow), balance = balances(net), result = [];
164 while (remaining.some(value => value > 0)) {
165 let start = balance.findIndex(value => value > 0);
166 if (start < 0) start = net.edges[remaining.findIndex(value => value > 0)].u;
167 const vertices = [start], edges = [];
168 while (true) {
169 const current = vertices[vertices.length - 1];
170 const index = net.edges.findIndex((edge, i) => edge.u === current && remaining[i] > 0);
171 const next = net.edges[index].v;
172 edges.push(index);
173 const repeat = vertices.indexOf(next);
174 if (repeat >= 0) {
175 const cycle = edges.slice(repeat), amount = Math.min(...cycle.map(i => remaining[i]));
176 for (const i of cycle) remaining[i] -= amount;
177 result.push({ cycle: true, start: next, edges: cycle, amount });
178 break;
179 }
180 vertices.push(next);
181 if (balance[next] < 0) {
182 const amount = Math.min(...edges.map(i => remaining[i]), balance[start], -balance[next]);
183 for (const i of edges) remaining[i] -= amount;
184 balance[start] -= amount; balance[next] += amount;
185 result.push({ cycle: false, start, edges: [...edges], amount });
186 break;
187 }
188 }
189 }
190 return result;
191}
192function pathText(net, path, source) {
193 return [net.names[source], ...path.flatMap(arc => [(arc.forward ? '+' : '-') + (arc.edge + 1), net.names[arc.to]])].join(' ');
194}
195function decompositionText(net, item, originalEdges = Infinity) {
196 const result = [net.names[item.start]];
197 for (const index of item.edges) if (index < originalEdges) result.push('+' + (index + 1), net.names[net.edges[index].v]);
198 return result.join(' ');
199}
200function flowList() {
201 network.edges.forEach((edge, i) => output.push(`FLOW ${i + 1} ${network.names[edge.u]} ${network.names[edge.v]} ${edge.flow}/${edge.capacity}`));
202}
203function startFlow(source, sink) {
204 if (network.owner.kind !== 'flow' || network.owner.source !== source || network.owner.sink !== sink) for (const edge of network.edges) edge.flow = 0;
205 network.owner = { kind: 'flow', source, sink };
206 return balances(network)[source];
207}
208function matching(a, b, pairs) {
209 const left = a.split(','), right = b.split(','), links = pairs === '-' ? [] : pairs.split(',').map(pair => pair.split('-'));
210 const seen = new Set();
211 for (const name of [...left, ...right]) { if (seen.has(name)) fail('DUPLICATE VERTEX ' + name); seen.add(name); }
212 const used = new Set();
213 for (const [u, v] of links) {
214 if (!left.includes(u)) fail('UNKNOWN VERTEX ' + u);
215 if (!right.includes(v)) fail('UNKNOWN VERTEX ' + v);
216 if (used.has(u + '-' + v)) fail('DUPLICATE PAIR ' + u + '-' + v);
217 used.add(u + '-' + v);
218 }
219 const partnerLeft = new Map(), partnerRight = new Map();
220 function attempt(name, examined) {
221 for (const [u, v] of links) if (u === name && !examined.has(v)) {
222 examined.add(v);
223 if (!partnerRight.has(v) || attempt(partnerRight.get(v), examined)) { partnerLeft.set(u, v); partnerRight.set(v, u); return true; }
224 }
225 return false;
226 }
227 for (const name of left) attempt(name, new Set());
228 const free = left.filter(name => !partnerLeft.has(name)), zl = new Set(free), zr = new Set(), queue = [...free];
229 for (let i = 0; i < queue.length; i++) for (const [u, v] of links) if (u === queue[i] && partnerLeft.get(u) !== v && !zr.has(v)) {
230 zr.add(v);
231 const partner = partnerRight.get(v);
232 if (partner !== undefined && !zl.has(partner)) { zl.add(partner); queue.push(partner); }
233 }
234 const cover = [...left.filter(name => !zl.has(name)), ...right.filter(name => zr.has(name))];
235 output.push(`MATCH size=${partnerLeft.size} left=${left.length} right=${right.length}`);
236 for (const name of left) if (partnerLeft.has(name)) output.push(`PAIR ${name} ${partnerLeft.get(name)}`);
237 output.push(`FREE ${free.join(',') || '-'} ${right.filter(name => !partnerRight.has(name)).join(',') || '-'}`,
238 `COVER size=${cover.length}${cover.length ? ' ' + cover.join(' ') : ''}`);
239 if (free.length) output.push(`HALL ${left.filter(name => zl.has(name)).join(',')} ${right.filter(name => zr.has(name)).join(',') || '-'}`);
240}
241function assignment(mode, size, words) {
242 if (size < 1 || size > 8) fail('BAD SIZE');
243 if (words.length !== size * size) fail('BAD LENGTH');
244 let best = null, count = 0, chosen = [];
245 const current = [];
246 function visit(row, used, sum) {
247 if (row === size) {
248 if (best === null || (mode === 'MIN' ? sum < best : sum > best)) { best = sum; count = 1; chosen = [...current]; }
249 else if (sum === best) count++;
250 return;
251 }
252 for (let col = 0; col < size; col++) if (!(used & (1 << col)) && words[row * size + col] !== 'x') {
253 current.push(col); visit(row + 1, used | (1 << col), sum + Number(words[row * size + col])); current.pop();
254 }
255 }
256 visit(0, 0, 0);
257 if (best === null) output.push(`ASSIGN ${mode} n=${size} infeasible`);
258 else {
259 output.push(`ASSIGN ${mode} n=${size} total=${best} optimal=${count}`);
260 chosen.forEach((col, row) => output.push(`MATE ${row + 1} ${col + 1} ${words[row * size + col]}`));
261 }
262}
263function execute(commandLine) {
264 const [command, ...args] = commandLine.split(' '), a = args[0], b = args[1];
265 switch (command) {
266 case 'NODE': {
267 const seen = new Set(network.names);
268 for (const name of args) { if (seen.has(name)) fail('DUPLICATE NODE ' + name); seen.add(name); }
269 if (network.names.length + args.length > 20) fail('TOO MANY NODES');
270 network.names.push(...args); network.demands.push(...args.map(() => 0));
271 output.push(`NODE added=${args.length} total=${network.names.length}`);
272 break;
273 }
274 case 'EDGE': {
275 const u = vertex(a), v = vertex(b);
276 if (u === v) fail('SELF LOOP');
277 if (network.edges.length === 60) fail('TOO MANY EDGES');
278 const capacity = Number(args[2]), cost = Number(args[3] ?? 0);
279 network.edges.push({ u, v, capacity, cost, lower: 0, flow: 0 });
280 output.push(`EDGE ${network.edges.length} ${a} ${b} cap=${capacity} cost=${cost}`);
281 break;
282 }
283 case 'LOWER': {
284 const edge = edgeAt(a), lower = Number(b);
285 if (lower > edge.capacity) fail('BAD BOUND');
286 edge.lower = lower; output.push(`LOWER ${a} low=${lower} cap=${edge.capacity}`);
287 break;
288 }
289 case 'CAP': {
290 const edge = edgeAt(a), capacity = Number(b);
291 if (capacity < edge.lower) fail('BAD BOUND');
292 if (capacity < edge.flow) fail('BELOW FLOW');
293 edge.capacity = capacity; output.push(`CAP ${a} low=${edge.lower} cap=${capacity}`);
294 break;
295 }
296 case 'COST': { const edge = edgeAt(a); edge.cost = Number(b); output.push(`COST ${a} cost=${b}`); break; }
297 case 'DEMAND': {
298 const index = vertex(a); network.demands[index] = Number(b);
299 output.push(`DEMAND ${a} ${b} total=${network.demands.reduce((sum, demand) => sum + demand, 0)}`);
300 break;
301 }
302 case 'RESET': {
303 const count = network.edges.filter(edge => edge.flow !== 0).length;
304 for (const edge of network.edges) edge.flow = 0;
305 network.owner = { kind: 'none', source: -1, sink: -1 };
306 output.push(`RESET edges=${count}`); break;
307 }
308 case 'CLEAR':
309 output.push(`CLEAR nodes=${network.names.length} edges=${network.edges.length}`);
310 network.names = []; network.demands = []; network.edges = []; network.owner = { kind: 'none', source: -1, sink: -1 };
311 break;
312 case 'SHOW': {
313 const out = Array(network.names.length).fill(0), incoming = Array(network.names.length).fill(0);
314 for (const edge of network.edges) { out[edge.u] += edge.flow; incoming[edge.v] += edge.flow; }
315 const owner = network.owner;
316 output.push(`NETWORK nodes=${network.names.length} edges=${network.edges.length}`);
317 output.push(owner.kind === 'none' ? 'STATE none' : owner.kind === 'circulation' ? `STATE circulation cost=${totalCost(network)}` : `STATE flow ${network.names[owner.source]} ${network.names[owner.sink]} value=${balances(network)[owner.source]} cost=${totalCost(network)}`);
318 network.names.forEach((name, i) => output.push(`NODE ${name} demand=${network.demands[i]} out=${out[i]} in=${incoming[i]}`));
319 network.edges.forEach((edge, i) => output.push(`EDGE ${i + 1} ${network.names[edge.u]} ${network.names[edge.v]} cap=${edge.capacity} cost=${edge.cost} low=${edge.lower} flow=${edge.flow}`));
320 break;
321 }
322 case 'AUGMENT': case 'MAXFLOW': case 'DINIC': case 'SCALING': case 'MINCUT': {
323 const [source, sink] = endpoints(a, b), start = startFlow(source, sink);
324 if (command === 'AUGMENT') {
325 const augmentation = single(network, source, sink);
326 output.push(`AUGMENT ${a} ${b} value=${balances(network)[source]}${augmentation === null ? ' none' : ` amount=${augmentation.amount} ${pathText(network, augmentation.path, source)}`}`);
327 break;
328 }
329 const result = command === 'DINIC' ? dinic(network, source, sink) : { paths: command === 'SCALING' ? scaling(network, source, sink) : edmondsKarp(network, source, sink), phases: 0 };
330 const value = balances(network)[source];
331 if (command === 'MINCUT') {
332 const { level } = bfs(network, source), cut = network.edges.map((edge, i) => i).filter(i => level[network.edges[i].u] >= 0 && level[network.edges[i].v] < 0);
333 output.push(`MINCUT ${a} ${b} value=${value} edges=${cut.length}`,
334 'SOURCE ' + network.names.filter((_, i) => level[i] >= 0).join(' '), 'SINK ' + network.names.filter((_, i) => level[i] < 0).join(' '));
335 for (const i of cut) { const edge = network.edges[i]; output.push(`CUT ${i + 1} ${network.names[edge.u]} ${network.names[edge.v]} ${edge.capacity}`); }
336 } else {
337 output.push(`${command} ${a} ${b} value=${value} start=${start}${command === 'DINIC' ? ` phases=${result.phases}` : ''} paths=${result.paths.length}`);
338 result.paths.forEach((item, i) => output.push(`AUG ${i + 1}${command === 'DINIC' ? ` phase=${item.phase}` : command === 'SCALING' ? ` delta=${item.delta}` : ''} amount=${item.amount} ${pathText(network, item.path, source)}`));
339 flowList();
340 }
341 break;
342 }
343 case 'SENSITIVITY': {
344 const [source, sink] = endpoints(a, b), base = network.copy();
345 edmondsKarp(base, source, sink); const value = balances(base)[source], effective = [];
346 for (let i = 0; i < network.edges.length; i++) {
347 const up = network.copy(); up.edges[i].capacity++; edmondsKarp(up, source, sink);
348 let downEffective = false;
349 if (network.edges[i].capacity > 0) { const down = network.copy(); down.edges[i].capacity--; edmondsKarp(down, source, sink); downEffective = balances(down)[source] < value; }
350 effective.push({ up: balances(up)[source] > value, down: downEffective });
351 }
352 output.push(`SENSITIVITY ${a} ${b} value=${value} up=${effective.filter(item => item.up).length} down=${effective.filter(item => item.down).length}`);
353 effective.forEach((item, i) => { if (item.up || item.down) { const edge = network.edges[i]; output.push(`CRITICAL ${i + 1} ${network.names[edge.u]} ${network.names[edge.v]} up=${item.up ? 'yes' : 'no'} down=${item.down ? 'yes' : 'no'}`); } });
354 break;
355 }
356 case 'MINCOST': {
357 const [source, sink] = endpoints(a, b);
358 if (negativeCycle(network)) fail('NEGATIVE CYCLE');
359 const working = network.copy(), target = args[2] === 'MAX' ? null : Number(args[2]), result = minCost(working, source, sink, target);
360 if (target !== null && result.flow < target) { output.push(`MINCOST ${a} ${b} k=${args[2]} infeasible max=${result.flow}`); break; }
361 network.edges.forEach((edge, i) => edge.flow = working.edges[i].flow); network.owner = { kind: 'flow', source, sink };
362 output.push(`MINCOST ${a} ${b} k=${args[2]} flow=${result.flow} cost=${result.cost}`);
363 result.paths.forEach((item, i) => output.push(`AUG ${i + 1} amount=${item.amount} cost=${item.cost} ${pathText(working, item.path, source)}`));
364 flowList(); break;
365 }
366 case 'DISJOINT': {
367 const [source, sink] = endpoints(a, b), working = network.copy(), original = network.edges.length;
368 for (const edge of working.edges) edge.capacity = edge.capacity > 0 ? 1 : 0;
369 if (args[2] === 'VERTEX') {
370 const exits = working.names.map((_, i) => i);
371 for (let i = 0; i < network.names.length; i++) if (i !== source && i !== sink) { exits[i] = working.names.length; working.names.push(network.names[i]); working.demands.push(0); }
372 for (const edge of working.edges) edge.u = exits[edge.u];
373 for (let i = 0; i < network.names.length; i++) if (i !== source && i !== sink) working.edges.push({ u: i, v: exits[i], capacity: 1, cost: 0, lower: 0, flow: 0 });
374 }
375 edmondsKarp(working, source, sink);
376 const paths = decompose(working).filter(item => !item.cycle);
377 output.push(`DISJOINT ${a} ${b} ${args[2]} count=${paths.length}`);
378 paths.forEach((item, i) => output.push(`PATH ${i + 1} ${decompositionText(working, item, original)}`));
379 break;
380 }
381 case 'DECOMPOSE': {
382 const pieces = decompose(network);
383 output.push(`DECOMPOSE paths=${pieces.filter(item => !item.cycle).length} cycles=${pieces.filter(item => item.cycle).length}`);
384 let paths = 0, cycles = 0;
385 for (const item of pieces) output.push(`${item.cycle ? 'CYCLE' : 'PATH'} ${item.cycle ? ++cycles : ++paths} amount=${item.amount} ${decompositionText(network, item)}`);
386 break;
387 }
388 case 'VERIFY': {
389 const violations = [], balance = balances(network);
390 network.edges.forEach((edge, i) => { if (edge.flow < edge.lower) violations.push(`LOW ${i + 1} flow=${edge.flow} low=${edge.lower}`); });
391 network.names.forEach((name, i) => {
392 if (network.owner.kind === 'flow' && (network.owner.source === i || network.owner.sink === i)) return;
393 const need = network.owner.kind === 'flow' ? 0 : network.demands[i];
394 if (-balance[i] !== need) violations.push(`BALANCE ${name} net=${-balance[i]} need=${need}`);
395 });
396 output.push(...(violations.length ? [`VERIFY violations=${violations.length}`, ...violations] : ['VERIFY ok']));
397 break;
398 }
399 case 'CIRCULATION': {
400 if (a === 'MINCOST' && negativeCycle(network, true)) fail('NEGATIVE CYCLE');
401 const sum = network.demands.reduce((total, demand) => total + demand, 0);
402 if (sum !== 0) { output.push(`CIRCULATION infeasible unbalanced=${sum}`); break; }
403 const working = network.copy(), balance = [...network.demands], source = working.names.length, sink = source + 1;
404 working.names.push('superSource', 'superSink'); working.demands.push(0, 0);
405 working.edges.forEach((edge, i) => { const lower = network.edges[i].lower; edge.capacity -= lower; balance[edge.u] += lower; balance[edge.v] -= lower; });
406 for (let i = 0; i < balance.length; i++) if (balance[i] < 0) working.edges.push({ u: source, v: i, capacity: -balance[i], cost: 0, lower: 0, flow: 0 });
407 for (let i = 0; i < balance.length; i++) if (balance[i] > 0) working.edges.push({ u: i, v: sink, capacity: balance[i], cost: 0, lower: 0, flow: 0 });
408 if (a === 'ANY') edmondsKarp(working, source, sink); else minCost(working, source, sink);
409 const required = balance.filter(value => value > 0).reduce((total, value) => total + value, 0), achieved = working.edges.filter(edge => edge.v === sink).reduce((total, edge) => total + edge.flow, 0);
410 if (achieved < required) {
411 const { level } = bfs(working, source);
412 output.push(`CIRCULATION infeasible required=${required} achieved=${achieved}`, 'CUT ' + network.names.filter((_, i) => level[i] >= 0).join(' '));
413 } else {
414 network.edges.forEach((edge, i) => edge.flow = edge.lower + working.edges[i].flow); network.owner = { kind: 'circulation', source: -1, sink: -1 };
415 output.push(`CIRCULATION feasible required=${required} cost=${totalCost(network)}`); flowList();
416 }
417 break;
418 }
419 case 'MATCH': matching(a, b, args[2]); break;
420 case 'ASSIGN': assignment(a, Number(b), args.slice(2)); break;
421 }
422}
423const input = fs.readFileSync(0, 'utf8').split(/\r?\n/), count = Number(input[0]);
424for (const command of input.slice(1, count + 1)) {
425 try { execute(command); } catch (error) { output.push(error.message); }
426}
427if (output.length) process.stdout.write(output.join('\n') + '\n');
