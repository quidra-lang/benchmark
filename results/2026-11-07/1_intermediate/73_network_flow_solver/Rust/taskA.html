4#[derive(Clone)]
5struct Edge {
6 from: usize,
7 to: usize,
8 capacity: i64,
9 cost: i64,
10 lower: i64,
11 flow: i64,
12}

14#[derive(Clone,Copy,PartialEq,Eq)]
15struct Arc {
16 edge: usize,
17 forward: bool,
18}

20#[derive(Clone,Copy,PartialEq,Eq)]
21enum Owner {
22 None,
23 Flow(usize,usize),

54impl Network {
55 fn new() -> Self {
56 Self {
57 names:Vec::new(),demands:Vec::new(),edges:Vec::new(),owner:Owner::None

60 fn node(&self, name: &str) -> Result<usize,String> {
61 self.names.iter().position(|n|n == name).ok_or_else(||error(format!("UNKNOWN NODE {name}")))
62 }
63 fn endpoints(&self, a: &str, b: &str) -> Result<(usize,usize),String> {
64 let s = self.node(a)?;
65 let t = self.node(b)?;
66 if s == t {
67 Err(error("SAME NODE"))
68 } else {
69 Ok((s,t))
70 }
71 }
72 fn edge_number(&self, text: &str) -> Result<usize,String> {
73 let i: usize = text.parse().unwrap();
74 if i == 0 || i > self.edges.len() {
75 Err(error(format!("UNKNOWN EDGE {text}")))
76 } else {
77 Ok(i-1)
78 }
79 }
80 fn global_arcs(&self) -> Vec<Arc> {
81 [true,false].iter().copied().flat_map(|forward|(0..self.edges.len()).map(move |edge|Arc {
82 edge,forward
83 })).collect()
84 }
85 fn arcs(&self, node: usize) -> Vec<Arc> {
86 self.global_arcs().into_iter().filter(|&a|self.start(a) == node).collect()
87 }
88 fn start(&self, a: Arc) -> usize {
89 if a.forward {
90 self.edges[a.edge].from
91 } else {
92 self.edges[a.edge].to
93 }
94 }
95 fn end(&self, a: Arc) -> usize {
96 if a.forward {
97 self.edges[a.edge].to
98 } else {
99 self.edges[a.edge].from
100 }
101 }
102 fn residual(&self, a: Arc) -> i64 {
103 let e = &self.edges[a.edge];
104 if a.forward {
105 e.capacity-e.flow
106 } else {
107 e.flow
108 }
109 }
110 fn arc_cost(&self, a: Arc) -> i64 {
111 if a.forward {
112 self.edges[a.edge].cost
113 } else {
114 -self.edges[a.edge].cost
115 }
116 }
117 fn push(&mut self, path: &[Arc], amount: i64) {
118 for &a in path {
119 self.edges[a.edge].flow += if a.forward {
120 amount
121 } else {
122 -amount
123 };
124 }
125 }
126 fn bottleneck(&self, path: &[Arc]) -> i64 {
127 path.iter().map(|&a|self.residual(a)).min().unwrap()
128 }
129 fn balances(&self) -> Vec<i64> {
130 let mut b = vec![0;self.names.len()];
131 for e in &self.edges {
132 b[e.from] += e.flow;
133 b[e.to] -= e.flow;
134 }
135 b
136 }
137 fn cost(&self) -> i64 {
138 self.edges.iter().map(|e|e.flow*e.cost).sum()
139 }
140 fn zero(&mut self) {
141 for e in &mut self.edges {
142 e.flow = 0;
143 }
144 }
145 fn begin(&mut self, s: usize, t: usize) -> i64 {
146 if self.owner != Owner::Flow(s,t) {
147 self.zero();
148 }
149 self.owner = Owner::Flow(s,t);
150 self.balances()[s]
151 }
152 fn bfs(&self, s: usize, delta: i64) -> (Vec<Option<usize>>,Vec<Option<Arc>>) {
153 let mut level = vec![None;self.names.len()];
154 let mut parent = vec![None;self.names.len()];
155 let mut queue = VecDeque::from([s]);
156 level[s] = Some(0);
157 while let Some(u) = queue.pop_front() {
158 for a in self.arcs(u) {
159 let v = self.end(a);
160 if self.residual(a) >= delta && level[v].is_none() {
161 level[v] = Some(level[u].unwrap()+1);
162 parent[v] = Some(a);
163 queue.push_back(v);
164 }
165 }
166 }
167 (level,parent)
168 }
169 fn path(&self, s: usize, t: usize, parent: &[Option<Arc>]) -> Vec<Arc> {
170 let mut path = Vec::new();
171 let mut at = t;
172 while at != s {
173 let a = parent[at].unwrap();
174 path.push(a);
175 at = self.start(a);
176 }
177 path.reverse();
178 path
179 }
180 fn augment(&mut self, s: usize, t: usize, delta: i64) -> Option<Pushed> {
181 let (level,parent) = self.bfs(s,delta);
182 level[t]?;
183 let arcs = self.path(s,t,&parent);
184 let amount = self.bottleneck(&arcs);
185 self.push(&arcs,amount);
186 Some(Pushed {
187 arcs,amount,phase:0,delta,cost:0
188 })
189 }
190 fn maxflow(&mut self, s: usize, t: usize) -> Vec<Pushed> {
191 let mut paths = Vec::new();
192 while let Some(p) = self.augment(s,t,1) {
193 paths.push(p);
194 }
195 paths
196 }
197 fn dinic(&mut self, s: usize, t: usize) -> (usize,Vec<Pushed>) {
198 let mut phases = 0;
199 let mut pushed = Vec::new();
200 loop {
201 let (level,_) = self.bfs(s,1);
202 if level[t].is_none() {
203 return (phases,pushed);
204 }
205 phases += 1;
206 let lists: Vec<_> = (0..self.names.len()).map(|v|self.arcs(v)).collect();
207 let mut pointers = vec![0;self.names.len()];
208 let mut path = Vec::new();
209 let mut u = s;
210 loop {
211 if u == t {
212 let amount = self.bottleneck(&path);
213 self.push(&path,amount);
214 pushed.push(Pushed {
215 arcs:path,amount,phase:phases,delta:1,cost:0
216 });
217 path = Vec::new();
218 u = s;
219 } else if pointers[u] == lists[u].len() {
220 if u == s {
221 break;
222 }
223 let a = path.pop().unwrap();
224 u = self.start(a);
225 pointers[u] += 1;
226 } else {
227 let a = lists[u][pointers[u]];
228 let v = self.end(a);
229 if self.residual(a) > 0 && level[v].is_some() && level[v].unwrap() == level[u].unwrap()+1 {
230 path.push(a);
231 u = v;
232 } else {
233 pointers[u] += 1;
234 }
235 }
236 }
237 }
238 }
239 fn scaling(&mut self, s: usize, t: usize) -> Vec<Pushed> {
240 let capacity = self.edges.iter().map(|e|e.capacity).max().unwrap_or(0);
241 let mut delta = 1;
242 while delta*2 <= capacity {
243 delta *= 2;
244 }
245 let mut paths = Vec::new();
246 while delta >= 1 {
247 while let Some(p) = self.augment(s,t,delta) {
248 paths.push(p);
249 }
250 delta /= 2;
251 }
252 paths
253 }
254 fn negative_cycle(&self, residual_lower: bool) -> bool {
255 let mut distance = vec![0i64;self.names.len()];
256 for pass in 0..self.names.len() {
257 let mut changed = false;
258 for e in &self.edges {
259 if e.capacity-(if residual_lower {
260 e.lower
261 } else {
262 0
263 }) > 0 && distance[e.to] > distance[e.from]+e.cost {
264 distance[e.to] = distance[e.from]+e.cost;
265 changed = true;
266 }
267 }
268 if !changed {
269 return false;
270 }
271 if pass+1 == self.names.len() {
272 return true;
273 }
274 }
275 false
276 }
277 fn shortest(&self, s: usize) -> (Vec<Option<i64>>,Vec<Option<Arc>>) {
278 let arcs: Vec<_> = self.global_arcs().into_iter().filter(|&a|self.residual(a) > 0).collect();
279 let mut distance = vec![None;self.names.len()];
280 let mut parent = vec![None;self.names.len()];
281 distance[s] = Some(0);
282 loop {
283 let mut changed = false;
284 for &a in &arcs {
285 let u = self.start(a);
286 let v = self.end(a);
287 if let Some(d) = distance[u] {
288 let next = d+self.arc_cost(a);
289 if distance[v].is_none_or(|old|next < old) {
290 distance[v] = Some(next);
291 parent[v] = Some(a);
292 changed = true;
293 }
294 }
295 }
296 if !changed {
297 return (distance,parent);
298 }
299 }
300 }
301 fn mincost(&mut self, s: usize, t: usize, target: Option<i64>) -> (i64,i64,Vec<Pushed>) {
302 let mut amount = 0;
303 let mut cost = 0;
304 let mut paths = Vec::new();
305 while target.is_none_or(|k|amount < k) {
306 let (distance,parent) = self.shortest(s);
307 let Some(unit) = distance[t] else {
308 break;
309 };
310 let arcs = self.path(s,t,&parent);
311 let mut delta = self.bottleneck(&arcs);
312 if let Some(k) = target {
313 delta = delta.min(k-amount);
314 }
315 self.push(&arcs,delta);
316 amount += delta;
317 cost += delta*unit;
318 paths.push(Pushed {
319 arcs,amount:delta,phase:0,delta:1,cost:unit
320 });
321 }
322 (amount,cost,paths)
323 }
324 fn notation(&self, start: usize, path: &[Arc]) -> String {
325 let mut text = self.names[start].clone();
326 for &a in path {
327 text += &format!(" {}{} {}",if a.forward {
328 "+"
329 } else {
330 "-"
331 },a.edge+1,self.names[self.end(a)]);
332 }
333 text
334 }
335 fn flows(&self) -> Vec<String> {
336 self.edges.iter().enumerate().map(|(i,e)|format!("FLOW {} {} {} {}/{}",i+1,self.names[e.from],self.names[e.to],e.flow,e.capacity)).collect()
337 }
338 fn decompose(&self) -> Vec<Component> {
339 let mut g: Vec<_> = self.edges.iter().map(|e|e.flow).collect();
340 let mut balance = self.balances();
341 let mut result = Vec::new();
342 while let Some(first) = g.iter().position(|&f|f > 0) {
343 let start = balance.iter().position(|&b|b > 0).unwrap_or(self.edges[first].from);
344 let mut vertices = vec![start];
345 let mut path = Vec::new();
346 loop {
347 let u = *vertices.last().unwrap();
348 let i = self.edges.iter().enumerate().position(|(i,e)|e.from == u && g[i] > 0).unwrap();
349 let v = self.edges[i].to;
350 path.push(i);
351 if let Some(q) = vertices.iter().position(|&x|x == v) {
352 let cycle = path[q..].to_vec();
353 let delta = cycle.iter().map(|&i|g[i]).min().unwrap();
354 for &i in &cycle {
355 g[i] -= delta;
356 }
357 result.push(Component {
358 start:v,edges:cycle,amount:delta,cycle:true
359 });
360 break;
361 }
362 vertices.push(v);
363 if balance[v] < 0 {
364 let delta = path.iter().map(|&i|g[i]).min().unwrap().min(balance[start]).min(-balance[v]);
365 for &i in &path {
366 g[i] -= delta;
367 }
368 balance[start] -= delta;
369 balance[v] += delta;
370 result.push(Component {
371 start,edges:path,amount:delta,cycle:false
372 });
373 break;
374 }
375 }
376 }
377 result
378 }
379 fn command(&mut self, w: &[&str]) -> Result<Vec<String>,String> {
380 let c = w[0];
381 let a = &w[1..];
382 let num = |i:usize|a[i].parse::<i64>().unwrap();
383 let mut out = Vec::new();
384 match c {
385 "NODE" => {
386 let mut seen: BTreeSet<_> = self.names.iter().map(String::as_str).collect();
387 for &name in a {
388 if !seen.insert(name) {
389 return Err(error(format!("DUPLICATE NODE {name}")));
390 }
391 }
392 if self.names.len()+a.len() > 20 {
393 return Err(error("TOO MANY NODES"));
394 }
395 self.names.extend(a.iter().map(|n|n.to_string()));
396 self.demands.resize(self.names.len(),0);
397 out.push(format!("NODE added={} total={}",a.len(),self.names.len()));
398 }
399 "EDGE" => {
400 let u = self.node(a[0])?;
401 let v = self.node(a[1])?;
402 if u == v {
403 return Err(error("SELF LOOP"));
404 }
405 if self.edges.len() == 60 {
406 return Err(error("TOO MANY EDGES"));
407 }
408 let cost = if a.len() == 4 {
409 num(3)
410 } else {
411 0
412 };
413 self.edges.push(Edge {
414 from:u,to:v,capacity:num(2),cost,lower:0,flow:0
415 });
416 out.push(format!("EDGE {} {} {} cap={} cost={cost}",self.edges.len(),a[0],a[1],num(2)));
417 }
418 "LOWER" | "CAP" | "COST" => {
419 let i = self.edge_number(a[0])?;
420 let e = &mut self.edges[i];
421 let n = num(1);
422 match c {
423 "LOWER" => {
424 if n > e.capacity {
425 return Err(error("BAD BOUND"));
426 }
427 e.lower = n;
428 out.push(format!("LOWER {} low={n} cap={}",a[0],e.capacity));
429 }, "CAP" => {
430 if n < e.lower {
431 return Err(error("BAD BOUND"));
432 }
433 if n < e.flow {
434 return Err(error("BELOW FLOW"));
435 }
436 e.capacity = n;
437 out.push(format!("CAP {} low={} cap={n}",a[0],e.lower));
438 }, _ => {
439 e.cost = n;
440 out.push(format!("COST {} cost={n}",a[0]));
441 }
442 }
443 }
444 "DEMAND" => {
445 let i = self.node(a[0])?;
446 self.demands[i] = num(1);
447 out.push(format!("DEMAND {} {} total={}",a[0],num(1),self.demands.iter().sum::<i64>()));
448 }
449 "RESET" => {
450 out.push(format!("RESET edges={}",self.edges.iter().filter(|e|e.flow != 0).count()));
451 self.zero();
452 self.owner = Owner::None;
453 }
454 "CLEAR" => {
455 out.push(format!("CLEAR nodes={} edges={}",self.names.len(),self.edges.len()));
456 *self = Self::new();
457 }
458 "SHOW" => {
459 let balance = self.balances();
460 out.push(format!("NETWORK nodes={} edges={}",self.names.len(),self.edges.len()));
461 out.push(match self.owner {
462 Owner::None => "STATE none".into(), Owner::Circulation => format!("STATE circulation cost={}",self.cost()), Owner::Flow(s,t) => format!("STATE flow {} {} value={} cost={}",self.names[s],self.names[t],balance[s],self.cost())
463 });
464 for (i,name) in self.names.iter().enumerate() {
465 let sent: i64 = self.edges.iter().filter(|e|e.from == i).map(|e|e.flow).sum();
466 let received: i64 = self.edges.iter().filter(|e|e.to == i).map(|e|e.flow).sum();
467 out.push(format!("NODE {name} demand={} out={sent} in={received}",self.demands[i]));
468 }
469 for (i,e) in self.edges.iter().enumerate() {
470 out.push(format!("EDGE {} {} {} cap={} cost={} low={} flow={}",i+1,self.names[e.from],self.names[e.to],e.capacity,e.cost,e.lower,e.flow));
471 }
472 }
473 "AUGMENT" | "MAXFLOW" | "DINIC" | "SCALING" | "MINCUT" => {
474 let (s,t) = self.endpoints(a[0],a[1])?;
475 let start = self.begin(s,t);
476 if c == "AUGMENT" {
477 if let Some(p) = self.augment(s,t,1) {
478 out.push(format!("AUGMENT {} {} value={} amount={} {}",a[0],a[1],self.balances()[s],p.amount,self.notation(s,&p.arcs)));
479 } else {
480 out.push(format!("AUGMENT {} {} value={} none",a[0],a[1],self.balances()[s]));
481 }
482 } else if c == "MINCUT" {
483 self.maxflow(s,t);
484 let (level,_) = self.bfs(s,1);
485 let cut: Vec<_> = self.edges.iter().enumerate().filter(|(_,e)|level[e.from].is_some() && level[e.to].is_none()).collect();
486 out.push(format!("MINCUT {} {} value={} edges={}",a[0],a[1],self.balances()[s],cut.len()));
487 out.push(format!("SOURCE {}",self.names.iter().enumerate().filter(|(i,_)|level[*i].is_some()).map(|(_,n)|n.as_str()).collect::<Vec<_>>().join(" ")));
488 out.push(format!("SINK {}",self.names.iter().enumerate().filter(|(i,_)|level[*i].is_none()).map(|(_,n)|n.as_str()).collect::<Vec<_>>().join(" ")));
489 for (i,e) in cut {
490 out.push(format!("CUT {} {} {} {}",i+1,self.names[e.from],self.names[e.to],e.capacity));
491 }
492 } else {
493 let (phases,paths) = if c == "DINIC" {
494 self.dinic(s,t)
495 } else if c == "SCALING" {
496 (0,self.scaling(s,t))
497 } else {
498 (0,self.maxflow(s,t))
499 };
500 out.push(format!("{c} {} {} value={} start={start}{} paths={}",a[0],a[1],self.balances()[s],if c == "DINIC" {
501 format!(" phases={phases}")
502 } else {
503 String::new()
504 },paths.len()));
505 for (i,p) in paths.iter().enumerate() {
506 let extra = if c == "DINIC" {
507 format!(" phase={}",p.phase)
508 } else if c == "SCALING" {
509 format!(" delta={}",p.delta)
510 } else {
511 String::new()
512 };
513 out.push(format!("AUG {}{extra} amount={} {}",i+1,p.amount,self.notation(s,&p.arcs)));
514 }
515 out.extend(self.flows());
516 }
517 }
518 "SENSITIVITY" => {
519 let (s,t) = self.endpoints(a[0],a[1])?;
520 let mut base = self.clone();
521 base.zero();
522 base.maxflow(s,t);
523 let value = base.balances()[s];
524 let mut up = 0;
525 let mut down = 0;
526 let mut lines = Vec::new();
527 for i in 0..self.edges.len() {
528 let mut work = self.clone();
529 work.zero();
530 work.edges[i].capacity += 1;
531 work.maxflow(s,t);
532 let effective_up = work.balances()[s] > value;
533 let mut effective_down = false;
534 if self.edges[i].capacity > 0 {
535 work = self.clone();
536 work.zero();
537 work.edges[i].capacity -= 1;
538 work.maxflow(s,t);
539 effective_down = work.balances()[s] < value;
540 }
541 up += usize::from(effective_up);
542 down += usize::from(effective_down);
543 if effective_up || effective_down {
544 let e = &self.edges[i];
545 lines.push(format!("CRITICAL {} {} {} up={} down={}",i+1,self.names[e.from],self.names[e.to],if effective_up {
546 "yes"
547 } else {
548 "no"
549 },if effective_down {
550 "yes"
551 } else {
552 "no"
553 }));
554 }
555 }
556 out.push(format!("SENSITIVITY {} {} value={value} up={up} down={down}",a[0],a[1]));
557 out.extend(lines);
558 }
559 "MINCOST" => {
560 let (s,t) = self.endpoints(a[0],a[1])?;
561 if self.negative_cycle(false) {
562 return Err(error("NEGATIVE CYCLE"));
563 }
564 let target = if a[2] == "MAX" {
565 None
566 } else {
567 Some(num(2))
568 };
569 let mut work = self.clone();
570 work.zero();
571 let (amount,cost,paths) = work.mincost(s,t,target);
572 if target.is_some_and(|k|amount < k) {
573 out.push(format!("MINCOST {} {} k={} infeasible max={amount}",a[0],a[1],a[2]));
574 } else {
575 for (e,new) in self.edges.iter_mut().zip(work.edges.iter()) {
576 e.flow = new.flow;
577 }
578 self.owner = Owner::Flow(s,t);
579 out.push(format!("MINCOST {} {} k={} flow={amount} cost={cost}",a[0],a[1],a[2]));
580 for (i,p) in paths.iter().enumerate() {
581 out.push(format!("AUG {} amount={} cost={} {}",i+1,p.amount,p.cost,self.notation(s,&p.arcs)));
582 }
583 out.extend(self.flows());
584 }
585 }
586 "DISJOINT" => {
587 let (s,t) = self.endpoints(a[0],a[1])?;
588 let original = self.edges.len();
589 let mut work = self.clone();
590 work.zero();
591 let mut exits: Vec<_> = (0..self.names.len()).collect();
592 if a[2] == "VERTEX" {
593 for (v,exit) in exits.iter_mut().enumerate() {
594 if v != s && v != t {
595 *exit = work.names.len();
596 work.names.push(self.names[v].clone());
597 work.demands.push(0);
598 }
599 }
600 }
601 for e in &mut work.edges {
602 e.from = exits[e.from];
603 e.capacity = i64::from(e.capacity > 0);
604 e.cost = 0;
605 }
606 if a[2] == "VERTEX" {
607 for (v,&exit) in exits.iter().enumerate() {
608 if v != s && v != t {
609 work.edges.push(Edge {
610 from:v,to:exit,capacity:1,cost:0,lower:0,flow:0
611 });
612 }
613 }
614 }
615 work.maxflow(s,t);
616 let paths: Vec<_> = work.decompose().into_iter().filter(|p|!p.cycle).collect();
617 out.push(format!("DISJOINT {} {} {} count={}",a[0],a[1],a[2],paths.len()));
618 for (i,p) in paths.iter().enumerate() {
619 let arcs: Vec<_> = p.edges.iter().filter(|&&e|e < original).map(|&edge|Arc {
620 edge,forward:true
621 }).collect();
622 out.push(format!("PATH {} {}",i+1,self.notation(s,&arcs)));
623 }
624 }
625 "DECOMPOSE" => {
626 let parts = self.decompose();
627 out.push(format!("DECOMPOSE paths={} cycles={}",parts.iter().filter(|p|!p.cycle).count(),parts.iter().filter(|p|p.cycle).count()));
628 let mut counts = [0,0];
629 for p in parts {
630 let kind = usize::from(p.cycle);
631 counts[kind] += 1;
632 let arcs: Vec<_> = p.edges.iter().map(|&edge|Arc {
633 edge,forward:true
634 }).collect();
635 out.push(format!("{} {} amount={} {}",if p.cycle {
636 "CYCLE"
637 } else {
638 "PATH"
639 },counts[kind],p.amount,self.notation(p.start,&arcs)));
640 }
641 }
642 "VERIFY" => {
643 let mut violations = Vec::new();
644 for (i,e) in self.edges.iter().enumerate() {
645 if e.flow < e.lower {
646 violations.push(format!("LOW {} flow={} low={}",i+1,e.flow,e.lower));
647 }
648 }
649 let balance = self.balances();
650 for (i,&b) in balance.iter().enumerate() {
651 let need = match self.owner {
652 Owner::Flow(s,t) if i == s || i == t => None, Owner::Flow(_,_) => Some(0), _ => Some(self.demands[i])
653 };
654 if let Some(need) = need {
655 if -b != need {
656 violations.push(format!("BALANCE {} net={} need={need}",self.names[i],-b));
657 }
658 }
659 }
660 if violations.is_empty() {
661 out.push("VERIFY ok".into());
662 } else {
663 out.push(format!("VERIFY violations={}",violations.len()));
664 out.extend(violations);
665 }
666 }
667 "CIRCULATION" => {
668 if a[0] == "MINCOST" && self.negative_cycle(true) {
669 return Err(error("NEGATIVE CYCLE"));
670 }
671 let sum: i64 = self.demands.iter().sum();
672 if sum != 0 {
673 return Ok(vec![format!("CIRCULATION infeasible unbalanced={sum}")]);
674 }
675 let n = self.names.len();
676 let mut work = self.clone();
677 work.names.extend(["super source".into(),"super sink".into()]);
678 work.demands.extend([0,0]);
679 let mut required = self.demands.clone();
680 for e in &mut work.edges {
681 required[e.to] -= e.lower;
682 required[e.from] += e.lower;
683 e.capacity -= e.lower;
684 e.flow = 0;
685 }
686 for (v,&r) in required.iter().enumerate() {
687 if r < 0 {
688 work.edges.push(Edge {
689 from:n,to:v,capacity:-r,cost:0,lower:0,flow:0
690 });
691 }
692 }
693 let sink_start = work.edges.len();
694 for (v,&r) in required.iter().enumerate() {
695 if r > 0 {
696 work.edges.push(Edge {
697 from:v,to:n+1,capacity:r,cost:0,lower:0,flow:0
698 });
699 }
700 }
701 if a[0] == "ANY" {
702 work.maxflow(n,n+1);
703 } else {
704 work.mincost(n,n+1,None);
705 }
706 let total: i64 = required.iter().filter(|&&r|r > 0).sum();
707 let achieved: i64 = work.edges[sink_start..].iter().map(|e|e.flow).sum();
708 if achieved < total {
709 let (level,_) = work.bfs(n,1);
710 out.push(format!("CIRCULATION infeasible required={total} achieved={achieved}"));
711 out.push(format!("CUT {}",self.names.iter().enumerate().filter(|(v,_)|level[*v].is_some()).map(|(_,s)|s.as_str()).collect::<Vec<_>>().join(" ")));
712 } else {
713 for (e,new) in self.edges.iter_mut().zip(work.edges.iter()) {
714 e.flow = e.lower+new.flow;
715 }
716 self.owner = Owner::Circulation;
717 out.push(format!("CIRCULATION feasible required={total} cost={}",self.cost()));
718 out.extend(self.flows());
719 }
720 }
721 "MATCH" => out = matching(a)?,
722 "ASSIGN" => out = assignment(a)?,
723 _ => unreachable!(),
724 }
725 Ok(out)
726 }
727}

729fn matching(a: &[&str]) -> Result<Vec<String>,String> {
730 let left: Vec<_> = a[0].split(',').collect();
731 let right: Vec<_> = a[1].split(',').collect();
732 let mut seen = BTreeSet::new();
733 for &n in left.iter().chain(right.iter()) {
734 if !seen.insert(n) {
735 return Err(error(format!("DUPLICATE VERTEX {n}")));
736 }
737 }
738 let mut pairs = Vec::new();
739 if a[2] != "-" {
740 for p in a[2].split(',') {
741 let (x,y) = p.split_once('-').unwrap();
742 let u = left.iter().position(|&n|n == x).ok_or_else(||error(format!("UNKNOWN VERTEX {x}")))?;
743 let v = right.iter().position(|&n|n == y).ok_or_else(||error(format!("UNKNOWN VERTEX {y}")))?;
744 if pairs.contains(&(u,v)) {
745 return Err(error(format!("DUPLICATE PAIR {p}")));
746 }
747 pairs.push((u,v));
748 }
749 }
750 fn attempt(u: usize, pairs: &[(usize,usize)], examined: &mut [bool], lp: &mut [Option<usize>], rp: &mut [Option<usize>]) -> bool {
751 for &(a,b) in pairs {
752 if a != u || examined[b] {
753 continue;
754 }
755 examined[b] = true;
756 if rp[b].is_none() || attempt(rp[b].unwrap(),pairs,examined,lp,rp) {
757 lp[u] = Some(b);
758 rp[b] = Some(u);
759 return true;
760 }
761 }
762 false
763 }
764 let mut lp = vec![None;left.len()];
765 let mut rp = vec![None;right.len()];
766 for u in 0..left.len() {
767 attempt(u,&pairs,&mut vec![false;right.len()],&mut lp,&mut rp);
768 }
769 let mut zl: Vec<_> = lp.iter().map(Option::is_none).collect();
770 let mut zr = vec![false;right.len()];
771 loop {
772 let mut changed = false;
773 for &(u,v) in &pairs {
774 if zl[u] && lp[u] != Some(v) && !zr[v] {
775 zr[v] = true;
776 changed = true;
777 }
778 }
779 for v in 0..right.len() {
780 if zr[v] {
781 if let Some(u) = rp[v] {
782 if !zl[u] {
783 zl[u] = true;
784 changed = true;
785 }
786 }
787 }
788 }
789 if !changed {
790 break;
791 }
792 }
793 let size = lp.iter().filter(|p|p.is_some()).count();
794 let mut out = vec![format!("MATCH size={size} left={} right={}",left.len(),right.len())];
795 for (u,p) in lp.iter().enumerate() {
796 if let Some(v) = p {
797 out.push(format!("PAIR {} {}",left[u],right[*v]));
798 }
799 }
800 let list = |values:Vec<&str>|if values.is_empty() {
801 "-".into()
802 } else {
803 values.join(",")
804 };
805 out.push(format!("FREE {} {}",list(left.iter().enumerate().filter(|(i,_)|lp[*i].is_none()).map(|(_,n)|*n).collect()),list(right.iter().enumerate().filter(|(i,_)|rp[*i].is_none()).map(|(_,n)|*n).collect())));
806 let cover: Vec<_> = left.iter().enumerate().filter(|(i,_)|!zl[*i]).map(|(_,n)|*n).chain(right.iter().enumerate().filter(|(i,_)|zr[*i]).map(|(_,n)|*n)).collect();
807 out.push(format!("COVER size={}{}",cover.len(),if cover.is_empty() {
808 String::new()
809 } else {
810 " ".to_string()+&cover.join(" ")
811 }));
812 if lp.iter().any(Option::is_none) {
813 out.push(format!("HALL {} {}",list(left.iter().enumerate().filter(|(i,_)|zl[*i]).map(|(_,n)|*n).collect()),list(right.iter().enumerate().filter(|(i,_)|zr[*i]).map(|(_,n)|*n).collect())));
814 }
815 Ok(out)
816}

818fn assignment(a: &[&str]) -> Result<Vec<String>,String> {
819 let n: usize = a[1].parse().unwrap();
820 if !(1..=8).contains(&n) {
821 return Err(error("BAD SIZE"));
822 }
823 if a.len()-2 != n*n {
824 return Err(error("BAD LENGTH"));
825 }
826 let values: Vec<Option<i64>> = a[2..].iter().map(|s|if *s == "x" {
827 None
828 } else {
829 Some(s.parse().unwrap())
830 }).collect();
831 fn visit(n: usize, values: &[Option<i64>], maximize: bool, chosen: &mut Vec<usize>, used: u32, sum: i64, best: &mut Option<(i64,usize,Vec<usize>)>) {
832 let row = chosen.len();
833 if row == n {
834 if best.as_ref().is_none_or(|b|if maximize {
835 sum > b.0
836 } else {
837 sum < b.0
838 }) {
839 *best = Some((sum,1,chosen.clone()));
840 } else if best.as_ref().unwrap().0 == sum {
841 best.as_mut().unwrap().1 += 1;
842 }
843 return;
844 }
845 for col in 0..n {
846 if used & (1 << col) == 0 {
847 if let Some(v) = values[row*n+col] {
848 chosen.push(col);
849 visit(n,values,maximize,chosen,used | (1 << col),sum+v,best);
850 chosen.pop();
851 }
852 }
853 }
854 }
855 let mut best = None;
856 visit(n,&values,a[0] == "MAX",&mut Vec::new(),0,0,&mut best);
857 if let Some((sum,count,chosen)) = best {
858 let mut out = vec![format!("ASSIGN {} n={n} total={sum} optimal={count}",a[0])];
859 for (row,&col) in chosen.iter().enumerate() {
860 out.push(format!("MATE {} {} {}",row+1,col+1,values[row*n+col].unwrap()));
861 }
862 Ok(out)
863 } else {
864 Ok(vec![format!("ASSIGN {} n={n} infeasible",a[0])])
865 }
866}
