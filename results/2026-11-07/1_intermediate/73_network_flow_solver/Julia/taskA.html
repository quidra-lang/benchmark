1struct CommandError <: Exception
2 message::String
3end
4fail(s) = throw(CommandError(s))
5check(b, s) = b ? nothing : fail(s)
6yesno(b) = b ? "yes" : "no"
7mutable struct FlowEdge
8 u::Int
9 v::Int
10 capacity::Int
11 cost::Int
12 lower::Int
13 flow::Int
14end
15FlowEdge(u, v, c, w = 0) = FlowEdge(u, v, c, w, 0, 0)
16clone_edges(es) = [FlowEdge(e.u, e.v, e.capacity, e.cost) for e in es]
17arcstart(es, a) = a>0 ? es[a].u : es[-a].v
18arcend(es, a) = a>0 ? es[a].v : es[-a].u
19residual(es, a) = a>0 ? es[a].capacity-es[a].flow : es[-a].flow
20arccost(es, a) = a>0 ? es[a].cost : -es[-a].cost
21struct FlowGraph
22 n::Int
23 edges::Vector{FlowEdge}
24 globalarcs::Vector{Int}
25 arcs::Vector{Vector{Int}}
26end
27function FlowGraph(n, es)
28 all=[collect(1:length(es)); -collect(1:length(es))]
29 FlowGraph(n, es, all, [[a for a in all if arcstart(es, a)==v] for v in 1:n])
30end
31struct Push
32 amount::Int
33 path::Vector{Int}
34 marker::Int
35 cost::Int
36end
37function pushpath!(g, path, amount)
38 for a in path
39 g.edges[abs(a)].flow+=sign(a)*amount
40 end
41end
42bottleneck(g, path) = minimum(residual(g.edges, a) for a in path)
43function bfs(g, s, threshold = 1)
44 levels=fill(-1, g.n)
45 parents=zeros(Int, g.n)
46 levels[s]=0
47 queue=[s]
48 at=1
49 while at<=length(queue)
50 u=queue[at]
51 at+=1
52 for a in g.arcs[u]
53 v=arcend(g.edges, a)
54 if residual(g.edges, a)>=threshold && levels[v]<0
55 levels[v]=levels[u]+1
56 parents[v]=a
57 push!(queue, v)
58 end
59 end
60 end
61 levels, parents
62end
63function tracepath(g, parents, s, t)
64 path=Int[]
65 v=t
66 while v!=s
67 a=parents[v]
68 push!(path, a)
69 v=arcstart(g.edges, a)
70 end
71 reverse!(path)
72end
73function augment!(g, s, t, threshold = 1)
74 levels, parents=bfs(g, s, threshold)
75 levels[t]<0 && return nothing
76 path=tracepath(g, parents, s, t)
77 amount=bottleneck(g, path)
78 pushpath!(g, path, amount)
79 Push(amount, path, threshold, 0)
80end
81function edmonds_karp!(g, s, t)
82 pushes=Push[]
83 while true
84 item=augment!(g, s, t)
85 item===nothing && return pushes
86 push!(pushes, item)
87 end
88end
89function dinic!(g, s, t)
90 pushes=Push[]
91 phases=0
92 while true
93 levels, _=bfs(g, s)
94 levels[t]<0 && return phases, pushes
95 phases+=1
96 pointers=ones(Int, g.n)
97 path=Int[]
98 u=s
99 while true
100 if u==t
101 amount=bottleneck(g, path)
102 pushpath!(g, path, amount)
103 push!(pushes, Push(amount, path, phases, 0))
104 path=Int[]
105 u=s
106 elseif pointers[u]>length(g.arcs[u])
107 u==s && break
108 a=pop!(path)
109 u=arcstart(g.edges, a)
110 pointers[u]+=1
111 else
112 a=g.arcs[u][pointers[u]]
113 v=arcend(g.edges, a)
114 if residual(g.edges, a)>0 && levels[v]==levels[u]+1
115 push!(path, a)
116 u=v
117 else
118 pointers[u]+=1
119 end
120 end
121 end
122 end
123end
124function scaling!(g, s, t)
125 largest=maximum((e.capacity for e in g.edges); init = 0)
126 threshold=1
127 while threshold*2<=largest
128 threshold*=2
129 end
130 pushes=Push[]
131 while threshold>=1
132 while true
133 item=augment!(g, s, t, threshold)
134 item===nothing && break
135 push!(pushes, item)
136 end
137 threshold=div(threshold, 2)
138 end
139 pushes
140end
141function negativecycle(n, es)
142 d=zeros(Int, n)
143 for pass in 1:n
144 changed=false
145 for e in es
146 if d[e.u]+e.cost<d[e.v]
147 d[e.v]=d[e.u]+e.cost
148 changed=true
149 end
150 end
151 !changed && return false
152 pass==n && return true
153 end
154 false
155end
156function cheapestpath(g, s, t)
157 usable=[a for a in g.globalarcs if residual(g.edges, a)>0]
158 d=Union{Nothing,Int}[nothing for _ in 1:g.n]
159 parents=zeros(Int, g.n)
160 d[s]=0
161 while true
162 changed=false
163 for a in usable
164 u=arcstart(g.edges, a)
165 v=arcend(g.edges, a)
166 d[u]===nothing && continue
167 candidate=d[u]+arccost(g.edges, a)
168 if d[v]===nothing || candidate<d[v]
169 d[v]=candidate
170 parents[v]=a
171 changed=true
172 end
173 end
174 !changed && break
175 end
176 d[t]===nothing ? nothing : (d[t], tracepath(g, parents, s, t))
177end
178function mincost!(g, s, t, target = nothing)
179 flow=0
180 cost=0
181 pushes=Push[]
182 while target===nothing || flow<target
183 result=cheapestpath(g, s, t)
184 result===nothing && break
185 unit, path=result
186 amount=bottleneck(g, path)
187 target!==nothing && (amount=min(amount, target-flow))
188 pushpath!(g, path, amount)
189 flow+=amount
190 cost+=amount*unit
191 push!(pushes, Push(amount, path, 0, unit))
192 end
193 flow, cost, pushes
194end
195outflow(es, v) = sum((e.flow for e in es if e.u==v); init = 0)
196inflow(es, v) = sum((e.flow for e in es if e.v==v); init = 0)
197flowvalue(es, s) = outflow(es, s)-inflow(es, s)
198totalcost(es) = sum((e.flow*e.cost for e in es); init = 0)
199function maximum_amount(n, es, s, t)
200 g=FlowGraph(n, es)
201 edmonds_karp!(g, s, t)
202 flowvalue(es, s)
203end
204struct FlowPiece
205 cycle::Bool
206 origin::Int
207 edges::Vector{Int}
208 amount::Int
209end
210function decompose(n, es)
211 remaining=[e.flow for e in es]
212 balance=[flowvalue(es, v) for v in 1:n]
213 pieces=FlowPiece[]
214 while any(>(0), remaining)
215 origin=findfirst(>(0), balance)
216 origin===nothing && (origin=es[findfirst(>(0), remaining)].u)
217 walk=[origin]
218 used=Int[]
219 while true
220 u=last(walk)
221 i=findfirst(j->es[j].u==u && remaining[j]>0, eachindex(es))
222 v=es[i].v
223 push!(used, i)
224 repeated=findfirst(==(v), walk)
225 if repeated!==nothing
226 selected=used[repeated:end]
227 amount=minimum(remaining[selected])
228 remaining[selected].-=amount
229 push!(pieces, FlowPiece(true, v, selected, amount))
230 break
231 end
232 push!(walk, v)
233 if balance[v]<0
234 amount=min(minimum(remaining[used]), balance[origin], -balance[v])
235 remaining[used].-=amount
236 balance[origin]-=amount
237 balance[v]+=amount
238 push!(pieces, FlowPiece(false, origin, used, amount))
239 break
240 end
241 end
242 end
243 pieces
244end
245function notation(names, es, origin, path)
246 words=[names[origin]]
247 for a in path
248 push!(words, (a>0 ? "+" : "-")*string(abs(a)), names[arcend(es, a)])
249 end
250 join(words, ' ')
251end

261function nodeindex(n, name)

263 check(at!==nothing, "UNKNOWN NODE $name")
264 at
265end
266function edgeindex(n, text)
267 at=parse(Int, text)

269 at

271function endpoints(n, args)
272 s=nodeindex(n, args[1])

275 s, t
276end
277function startflow!(n, s, t)
278 if n.owner != :flow || n.source!=s || n.sink!=t
279 for e in n.edges
280 e.flow=0
281 end
282 end
283 n.owner=:flow
284 n.source=s
285 n.sink=t
286 flowvalue(n.edges, s)
287end

293function matchreport(args)
294 left=String.(split(args[1], ','))
295 right=String.(split(args[2], ','))
296 seen=Set{String}()
297 for name in [left; right]
298 check(!(name in seen), "DUPLICATE VERTEX $name")
299 push!(seen, name)
300 end
301 links=Tuple{Int,Int}[]
302 for text in (args[3]=="-" ? String[] : split(args[3], ','))
303 a, b=split(text, '-')
304 u=findfirst(==(a), left)
305 v=findfirst(==(b), right)
306 check(u!==nothing, "UNKNOWN VERTEX $a")
307 check(v!==nothing, "UNKNOWN VERTEX $b")
308 check(!((u, v) in links), "DUPLICATE PAIR $a-$b")
309 push!(links, (u, v))
310 end
311 lp=zeros(Int, length(left))
312 rp=zeros(Int, length(right))
313 examined=falses(length(right))
314 function try_left(u)
315 for (a, b) in links
316 a==u && !examined[b] || continue
317 examined[b]=true
318 if rp[b]==0 || try_left(rp[b])
319 rp[b]=u
320 lp[u]=b
321 return true
322 end
323 end
324 false
325 end
326 for u in eachindex(left)
327 fill!(examined, false)
328 try_left(u)
329 end
330 lin=lp .== 0
331 rin=falses(length(right))
332 while true
333 changed=false
334 for (a, b) in links
335 if lin[a] && !rin[b] && lp[a]!=b
336 rin[b]=true
337 changed=true
338 end
339 end
340 for b in eachindex(right)
341 if rin[b] && rp[b]!=0 && !lin[rp[b]]
342 lin[rp[b]]=true
343 changed=true
344 end
345 end
346 !changed && break
347 end
348 freeleft=left[lp .== 0]
349 freeright=right[rp .== 0]
350 cover=[left[.!lin]; right[rin]]
351 println("MATCH size=$(count(!=(0),lp)) left=$(length(left)) right=$(length(right))")
352 for a in eachindex(left)
353 lp[a]!=0 && println("PAIR $(left[a]) $(right[lp[a]])")
354 end
355 println(
356 "FREE ",
357 isempty(freeleft) ? "-" : join(freeleft, ','),
358 " ",
359 isempty(freeright) ? "-" : join(freeright, ','),
360 )
361 println("COVER size=$(length(cover))", isempty(cover) ? "" : " "*join(cover, ' '))
362 !isempty(freeleft) &&
363 println("HALL ", join(left[lin], ','), " ", any(rin) ? join(right[rin], ',') : "-")
364end
365function assignmentreport(args)
366 mode=args[1]
367 n=parse(Int, args[2])
368 check(1<=n<=8, "BAD SIZE")
369 values=args[3:end]
370 check(length(values)==n*n, "BAD LENGTH")
371 chosen=Int[]
372 best=nothing
373 ways=0
374 bestchosen=Int[]
375 function search(total)
376 row=length(chosen)+1
377 if row>n
378 if best===nothing || (mode=="MAX" ? total>best : total<best)
379 best=total
380 ways=1
381 bestchosen=copy(chosen)
382 elseif total==best
383 ways+=1
384 end
385 return
386 end
387 for col in 1:n
388 col in chosen && continue
389 text=values[(row-1)*n+col]
390 text=="x" && continue
391 push!(chosen, col)
392 search(total+parse(Int, text))
393 pop!(chosen)
394 end
395 end
396 search(0)
397 if best===nothing
398 println("ASSIGN $mode n=$n infeasible")
399 return
400 end
401 println("ASSIGN $mode n=$n total=$best optimal=$ways")
402 for (row, col) in enumerate(bestchosen)
403 println("MATE $row $col $(values[(row-1)*n+col])")
404 end
405end
406function execute!(n, line)
407 words=String.(split(line, ' '))
408 command=words[1]
409 args=words[2:end]
410 if command=="NODE"
411 seen=Set(n.names)
412 for name in args
413 check(!(name in seen), "DUPLICATE NODE $name")
414 push!(seen, name)
415 end
416 check(length(n.names)+length(args)<=20, "TOO MANY NODES")
417 append!(n.names, args)
418 append!(n.demands, zeros(Int, length(args)))
419 println("NODE added=$(length(args)) total=$(length(n.names))")
420 elseif command=="EDGE"
421 s=nodeindex(n, args[1])
422 t=nodeindex(n, args[2])
423 check(s!=t, "SELF LOOP")
424 check(length(n.edges)<60, "TOO MANY EDGES")
425 e=FlowEdge(s, t, parse(Int, args[3]), length(args)==4 ? parse(Int, args[4]) : 0)
426 push!(n.edges, e)
427 println("EDGE $(length(n.edges)) $(args[1]) $(args[2]) cap=$(e.capacity) cost=$(e.cost)")
428 elseif command in ("LOWER", "CAP", "COST")
429 at=edgeindex(n, args[1])
430 e=n.edges[at]
431 value=parse(Int, args[2])
432 if command=="LOWER"
433 check(value<=e.capacity, "BAD BOUND")
434 e.lower=value
435 println("LOWER $at low=$(e.lower) cap=$(e.capacity)")
436 elseif command=="CAP"
437 check(value>=e.lower, "BAD BOUND")
438 check(value>=e.flow, "BELOW FLOW")
439 e.capacity=value
440 println("CAP $at low=$(e.lower) cap=$(e.capacity)")
441 else
442 e.cost=value
443 println("COST $at cost=$(e.cost)")
444 end
445 elseif command=="DEMAND"
446 at=nodeindex(n, args[1])
447 n.demands[at]=parse(Int, args[2])
448 println("DEMAND $(args[1]) $(args[2]) total=$(sum(n.demands))")
449 elseif command=="RESET"
450 println("RESET edges=$(count(e->e.flow!=0,n.edges))")
451 for e in n.edges
452 e.flow=0
453 end
454 n.owner=:none
455 elseif command=="CLEAR"
456 println("CLEAR nodes=$(length(n.names)) edges=$(length(n.edges))")
457 empty!(n.names)
458 empty!(n.demands)
459 empty!(n.edges)
460 n.owner=:none
461 n.source=0
462 n.sink=0
463 elseif command=="SHOW"
464 println("NETWORK nodes=$(length(n.names)) edges=$(length(n.edges))")
465 println(
466 n.owner == :none ? "STATE none" :
467 n.owner == :circulation ? "STATE circulation cost=$(totalcost(n.edges))" :
468 "STATE flow $(n.names[n.source]) $(n.names[n.sink]) value=$(flowvalue(n.edges,n.source)) cost=$(totalcost(n.edges))",
469 )
470 for (v, name) in enumerate(n.names)
471 println(
472 "NODE $name demand=$(n.demands[v]) out=$(outflow(n.edges,v)) in=$(inflow(n.edges,v))",
473 )
474 end
475 for (i, e) in enumerate(n.edges)
476 println(
477 "EDGE $i $(n.names[e.u]) $(n.names[e.v]) cap=$(e.capacity) cost=$(e.cost) low=$(e.lower) flow=$(e.flow)",
478 )
479 end
480 elseif command in ("AUGMENT", "MAXFLOW", "DINIC", "SCALING", "MINCUT")
481 s, t=endpoints(n, args)
482 start=startflow!(n, s, t)
483 g=FlowGraph(length(n.names), n.edges)
484 heading="$command $(args[1]) $(args[2])"
485 if command=="AUGMENT"
486 item=augment!(g, s, t)
487 println(
488 "$heading value=$(flowvalue(n.edges,s))",
489 item===nothing ? " none" :
490 " amount=$(item.amount) "*notation(n.names, n.edges, s, item.path),
491 )
492 return
493 end
494 phases=0
495 if command=="DINIC"
496 phases, pushes=dinic!(g, s, t)
497 elseif command=="SCALING"
498 pushes=scaling!(g, s, t)
499 else
500 pushes=edmonds_karp!(g, s, t)
501 end
502 if command=="MINCUT"
503 levels, _=bfs(g, s)
504 cut=[i for (i, e) in enumerate(n.edges) if levels[e.u]>=0 && levels[e.v]<0]
505 println("$heading value=$(flowvalue(n.edges,s)) edges=$(length(cut))")
506 println("SOURCE ", join(n.names[levels .>= 0], ' '))
507 println("SINK ", join(n.names[levels .< 0], ' '))
508 for i in cut
509 e=n.edges[i]
510 println("CUT $i $(n.names[e.u]) $(n.names[e.v]) $(e.capacity)")
511 end
512 else
513 println(
514 "$heading value=$(flowvalue(n.edges,s)) start=$start",
515 command=="DINIC" ? " phases=$phases" : "",
516 " paths=$(length(pushes))",
517 )
518 for (i, item) in enumerate(pushes)
519 detail=command=="DINIC" ? " phase=$(item.marker)" :
520 command=="SCALING" ? " delta=$(item.marker)" : ""
521 println(
522 "AUG $i$detail amount=$(item.amount) ",
523 notation(n.names, n.edges, s, item.path),
524 )
525 end
526 flowlist(n)
527 end
528 elseif command=="SENSITIVITY"
529 s, t=endpoints(n, args)
530 value=maximum_amount(length(n.names), clone_edges(n.edges), s, t)
531 effective=Tuple{Int,Bool,Bool}[]
532 for (i, e) in enumerate(n.edges)
533 work=clone_edges(n.edges)
534 work[i].capacity+=1
535 up=maximum_amount(length(n.names), work, s, t)>value
536 down=false
537 if e.capacity>0
538 work=clone_edges(n.edges)
539 work[i].capacity-=1
540 down=maximum_amount(length(n.names), work, s, t)<value
541 end
542 (up || down) && push!(effective, (i, up, down))
543 end
544 println(
545 "SENSITIVITY $(args[1]) $(args[2]) value=$value up=$(count(x->x[2],effective)) down=$(count(x->x[3],effective))",
546 )
547 for (i, up, down) in effective
548 e=n.edges[i]
549 println(
550 "CRITICAL $i $(n.names[e.u]) $(n.names[e.v]) up=$(yesno(up)) down=$(yesno(down))",
551 )
552 end
553 elseif command=="MINCOST"
554 s, t=endpoints(n, args)
555 check(
556 !negativecycle(length(n.names), [e for e in n.edges if e.capacity>0]),
557 "NEGATIVE CYCLE",
558 )
559 target=args[3]=="MAX" ? nothing : parse(Int, args[3])
560 g=FlowGraph(length(n.names), clone_edges(n.edges))
561 flow, cost, pushes=mincost!(g, s, t, target)
562 heading="MINCOST $(args[1]) $(args[2]) k=$(args[3])"
563 if target!==nothing && flow<target
564 println("$heading infeasible max=$flow")
565 return
566 end
567 for (original, worked) in zip(n.edges, g.edges)
568 original.flow=worked.flow
569 end
570 n.owner=:flow
571 n.source=s
572 n.sink=t
573 println("$heading flow=$flow cost=$cost")
574 for (i, item) in enumerate(pushes)
575 println(
576 "AUG $i amount=$(item.amount) cost=$(item.cost) ",
577 notation(n.names, n.edges, s, item.path),
578 )
579 end
580 flowlist(n)
581 elseif command=="DISJOINT"
582 s, t=endpoints(n, args)
583 nodecount=length(n.names)
584 exits=collect(1:nodecount)
585 total=nodecount
586 if args[3]=="VERTEX"
587 total=2nodecount
588 for v in 1:nodecount
589 v!=s && v!=t && (exits[v]=nodecount+v)
590 end
591 end
592 es=[FlowEdge(exits[e.u], e.v, min(1, e.capacity)) for e in n.edges]
593 if args[3]=="VERTEX"
594 for v in 1:nodecount
595 v!=s && v!=t && push!(es, FlowEdge(v, exits[v], 1))
596 end
597 end
598 edmonds_karp!(FlowGraph(total, es), s, t)
599 pieces=[p for p in decompose(total, es) if !p.cycle]
600 println("DISJOINT $(args[1]) $(args[2]) $(args[3]) count=$(length(pieces))")
601 for (i, piece) in enumerate(pieces)
602 selected=[j for j in piece.edges if j<=length(n.edges)]
603 println("PATH $i ", notation(n.names, n.edges, piece.origin, selected))
604 end
605 elseif command=="DECOMPOSE"
606 pieces=decompose(length(n.names), n.edges)
607 cycles=count(p->p.cycle, pieces)
608 println("DECOMPOSE paths=$(length(pieces)-cycles) cycles=$cycles")
609 p=0
610 c=0
611 for piece in pieces
612 if piece.cycle
613 c+=1
614 title="CYCLE $c"
615 else
616 p+=1
617 title="PATH $p"
618 end
619 println(
620 "$title amount=$(piece.amount) ",
621 notation(n.names, n.edges, piece.origin, piece.edges),
622 )
623 end
624 elseif command=="VERIFY"
625 violations=[
626 "LOW $i flow=$(e.flow) low=$(e.lower)" for
627 (i, e) in enumerate(n.edges) if e.flow<e.lower
628 ]
629 for (v, name) in enumerate(n.names)
630 n.owner == :flow && v in (n.source, n.sink) && continue
631 need=n.owner == :flow ? 0 : n.demands[v]
632 net=inflow(n.edges, v)-outflow(n.edges, v)
633 net!=need && push!(violations, "BALANCE $name net=$net need=$need")
634 end
635 if isempty(violations)
636 println("VERIFY ok")
637 else
638 println("VERIFY violations=$(length(violations))")
639 foreach(println, violations)
640 end
641 elseif command=="CIRCULATION"
642 mode=args[1]
643 nodecount=length(n.names)
644 mode=="MINCOST" && check(
645 !negativecycle(nodecount, [e for e in n.edges if e.capacity>e.lower]),
646 "NEGATIVE CYCLE",
647 )
648 imbalance=sum(n.demands)
649 if imbalance!=0
650 println("CIRCULATION infeasible unbalanced=$imbalance")
651 return
652 end
653 required=copy(n.demands)
654 es=[FlowEdge(e.u, e.v, e.capacity-e.lower, e.cost) for e in n.edges]
655 for e in n.edges
656 required[e.v]-=e.lower
657 required[e.u]+=e.lower
658 end
659 for v in 1:nodecount
660 required[v]<0 && push!(es, FlowEdge(nodecount+1, v, -required[v]))
661 end
662 sinkedges=Int[]
663 for v in 1:nodecount
664 if required[v]>0
665 push!(es, FlowEdge(v, nodecount+2, required[v]))
666 push!(sinkedges, length(es))
667 end
668 end
669 g=FlowGraph(nodecount+2, es)
670 mode=="ANY" ? edmonds_karp!(g, nodecount+1, nodecount+2) :
671 mincost!(g, nodecount+1, nodecount+2)
672 total=sum((es[i].capacity for i in sinkedges); init = 0)
673 achieved=sum((es[i].flow for i in sinkedges); init = 0)
674 if achieved<total
675 println("CIRCULATION infeasible required=$total achieved=$achieved")
676 levels, _=bfs(g, nodecount+1)
677 println("CUT ", join([n.names[v] for v in 1:nodecount if levels[v]>=0], ' '))
678 return
679 end
680 for (i, e) in enumerate(n.edges)
681 e.flow=e.lower+es[i].flow
682 end
683 n.owner=:circulation
684 println("CIRCULATION feasible required=$total cost=$(totalcost(n.edges))")
685 flowlist(n)
686 elseif command=="MATCH"
687 matchreport(args)
688 elseif command=="ASSIGN"
689 assignmentreport(args)
690 end
691end
