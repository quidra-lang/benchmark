72List<int> balances(Network net) {
73 final result = List<int>.filled(net.names.length, 0);
74 for (final edge in net.edges) {
75 result[edge.u] += edge.flow;
76 result[edge.v] -= edge.flow;

80int totalCost(Network net) => net.edges.fold<int>(0, (sum, edge) => sum + edge.flow * edge.cost);

82 for (var i = 0; i < net.edges.length; i++) Arc(i, true, net.edges[i].u, net.edges[i].v),

85List<List<Arc>> adjacency(Network net) {

90int residual(Network net, Arc arc) {
91 final edge = net.edges[arc.edge];
92 return arc.forward ? edge.capacity - edge.flow : edge.flow;
93}
94void push(Network net, List<Arc> path, int amount) {
95 for (final arc in path) { net.edges[arc.edge].flow += arc.forward ? amount : -amount; }
96}
97int bottleneck(Network net, List<Arc> path) => path.map((arc) => residual(net, arc)).reduce(math.min);
98BreadthFirst bfs(Network net, int source, [int threshold = 1]) {
99 final level = List<int>.filled(net.names.length, -1);
100 final parent = List<Arc?>.filled(net.names.length, null), queue = <int>[source];
101 final lists = adjacency(net);
102 level[source] = 0;
103 for (var i = 0; i < queue.length; i++) {
104 final u = queue[i];
105 for (final arc in lists[u]) {
106 if (residual(net, arc) >= threshold && level[arc.to] < 0) {
107 level[arc.to] = level[u] + 1;
108 parent[arc.to] = arc;
109 queue.add(arc.to);
110 }
111 }
112 }
113 return BreadthFirst(level, parent);
114}
115List<Arc> restorePath(List<Arc?> parent, int source, int sink) {
116 final result = <Arc>[];
117 for (var at = sink; at != source;) {
118 final arc = parent[at]!;
119 result.add(arc);
120 at = arc.from;
121 }
122 return result.reversed.toList();
123}
124Augmentation? single(Network net, int source, int sink, [int threshold = 1]) {
125 final search = bfs(net, source, threshold);
126 if (search.level[sink] < 0) return null;
127 final path = restorePath(search.parent, source, sink), amount = bottleneck(net, path);
128 push(net, path, amount);
129 return Augmentation(path, amount, delta: threshold);
130}
131List<Augmentation> edmondsKarp(Network net, int source, int sink) {
132 final paths = <Augmentation>[];
133 while (true) {
134 final augmentation = single(net, source, sink);
135 if (augmentation == null) return paths;
136 paths.add(augmentation);
137 }

139(List<Augmentation>, int) dinic(Network net, int source, int sink) {
140 final paths = <Augmentation>[], lists = adjacency(net);
141 var phases = 0;
142 while (true) {
143 final level = bfs(net, source).level;
144 if (level[sink] < 0) return (paths, phases);
145 phases++;
146 final pointers = List<int>.filled(net.names.length, 0);
147 var path = <Arc>[], current = source;
148 while (true) {
149 if (current == sink) {
150 final amount = bottleneck(net, path);
151 push(net, path, amount);
152 paths.add(Augmentation(List.of(path), amount, phase: phases));
153 path = [];
154 current = source;
155 } else if (pointers[current] >= lists[current].length) {
156 if (current == source) break;
157 final arc = path.removeLast();
158 current = arc.from;
159 pointers[current]++;
160 } else {
161 final arc = lists[current][pointers[current]];
162 if (residual(net, arc) > 0 && level[arc.to] == level[current] + 1) {
163 path.add(arc);
164 current = arc.to;
165 } else {
166 pointers[current]++;
167 }
168 }
169 }
170 }

172List<Augmentation> scaling(Network net, int source, int sink) {
173 final paths = <Augmentation>[];
174 final maximum = net.edges.fold<int>(1, (value, edge) => math.max(value, edge.capacity));
175 var delta = 1;
176 while (delta * 2 <= maximum) { delta *= 2; }
177 while (delta >= 1) {
178 while (true) {
179 final augmentation = single(net, source, sink, delta);
180 if (augmentation == null) break;
181 paths.add(augmentation);
182 }
183 delta ~/= 2;
184 }
185 return paths;
186}
187bool negativeCycle(Network net, [bool lowerBounds = false]) {
188 final distance = List<int>.filled(net.names.length, 0);
189 for (var pass = 0; pass < net.names.length; pass++) {
190 var changed = false;
191 for (final edge in net.edges) {
192 if (edge.capacity - (lowerBounds ? edge.lower : 0) > 0 && distance[edge.v] > distance[edge.u] + edge.cost) {
193 distance[edge.v] = distance[edge.u] + edge.cost;
194 changed = true;
195 }
196 }
197 if (!changed) return false;
198 if (pass + 1 == net.names.length) return true;
199 }
200 return false;
201}
202Cheapest? shortest(Network net, int source, int sink) {
203 final arcs = globalArcs(net).where((arc) => residual(net, arc) > 0).toList();
204 final distance = List<int?>.filled(net.names.length, null), parent = List<Arc?>.filled(net.names.length, null);
205 distance[source] = 0;
206 while (true) {
207 var changed = false;
208 for (final arc in arcs) {
209 if (distance[arc.from] == null) continue;
210 final cost = net.edges[arc.edge].cost * (arc.forward ? 1 : -1);
211 final next = distance[arc.from]! + cost;
212 if (distance[arc.to] == null || next < distance[arc.to]!) {
213 distance[arc.to] = next;
214 parent[arc.to] = arc;
215 changed = true;
216 }
217 }
218 if (!changed) break;
219 }
220 return distance[sink] == null ? null : Cheapest(restorePath(parent, source, sink), distance[sink]!);
221}
222(List<Augmentation>, int, int) minCost(Network net, int source, int sink, [int? target]) {
223 final paths = <Augmentation>[];
224 var flow = 0, cost = 0;
225 while (target == null || flow < target) {
226 final cheapest = shortest(net, source, sink);
227 if (cheapest == null) break;
228 final amount = target == null ? bottleneck(net, cheapest.path) : math.min(bottleneck(net, cheapest.path), target - flow);
229 push(net, cheapest.path, amount);
230 flow += amount;
231 cost += amount * cheapest.cost;
232 paths.add(Augmentation(cheapest.path, amount, cost: cheapest.cost));
233 }
234 return (paths, flow, cost);
235}
236List<Decomposition> decompose(Network net) {
237 final remaining = net.edges.map((edge) => edge.flow).toList(), balance = balances(net);
238 final result = <Decomposition>[];
239 while (remaining.any((value) => value > 0)) {
240 var start = balance.indexWhere((value) => value > 0);
241 if (start < 0) start = net.edges[remaining.indexWhere((value) => value > 0)].u;
242 final vertices = <int>[start], edges = <int>[];
243 while (true) {
244 final current = vertices.last;
245 var index = 0;
246 while (net.edges[index].u != current || remaining[index] <= 0) { index++; }
247 final next = net.edges[index].v;
248 edges.add(index);
249 final repeat = vertices.indexOf(next);
250 if (repeat >= 0) {
251 final cycle = edges.sublist(repeat), amount = edges.sublist(repeat).map((i) => remaining[i]).reduce(math.min);
252 for (final i in cycle) { remaining[i] -= amount; }
253 result.add(Decomposition(true, next, cycle, amount));
254 break;
255 }
256 vertices.add(next);
257 if (balance[next] < 0) {
258 final amount = math.min(edges.map((i) => remaining[i]).reduce(math.min), math.min(balance[start], -balance[next]));
259 for (final i in edges) { remaining[i] -= amount; }
260 balance[start] -= amount;
261 balance[next] += amount;
262 result.add(Decomposition(false, start, List.of(edges), amount));
263 break;
264 }
265 }
266 }
267 return result;
268}

286void matching(String a, String b, String pairs) {
287 final left = a.split(','), right = b.split(',');
288 final links = pairs == '-' ? <(String, String)>[] : pairs.split(',').map((pair) {
289 final parts = pair.split('-');
290 return (parts[0], parts[1]);
291 }).toList();
292 final seen = <String>{};
293 for (final name in [...left, ...right]) {
294 if (!seen.add(name)) fail('DUPLICATE VERTEX $name');
295 }
296 final used = <String>{};
297 for (final (u, v) in links) {
298 if (!left.contains(u)) fail('UNKNOWN VERTEX $u');
299 if (!right.contains(v)) fail('UNKNOWN VERTEX $v');
300 if (!used.add('$u-$v')) fail('DUPLICATE PAIR $u-$v');
301 }
302 final partnerLeft = <String, String>{}, partnerRight = <String, String>{};
303 bool attempt(String name, Set<String> examined) {
304 for (final (u, v) in links) {
305 if (u == name && examined.add(v)) {
306 if (!partnerRight.containsKey(v) || attempt(partnerRight[v]!, examined)) {
307 partnerLeft[u] = v;
308 partnerRight[v] = u;
309 return true;
310 }
311 }
312 }
313 return false;
314 }
315 for (final name in left) { attempt(name, {}); }
316 final free = left.where((name) => !partnerLeft.containsKey(name)).toList();
317 final zl = free.toSet(), zr = <String>{}, queue = List.of(free);
318 for (var i = 0; i < queue.length; i++) {
319 for (final (u, v) in links) {
320 if (u == queue[i] && partnerLeft[u] != v && zr.add(v)) {
321 final partner = partnerRight[v];
322 if (partner != null && zl.add(partner)) queue.add(partner);
323 }
324 }
325 }
326 final cover = [...left.where((name) => !zl.contains(name)), ...right.where(zr.contains)];
327 output.add('MATCH size=${partnerLeft.length} left=${left.length} right=${right.length}');
328 for (final name in left) {
329 if (partnerLeft.containsKey(name)) output.add('PAIR $name ${partnerLeft[name]}');
330 }
331 final freeRight = right.where((name) => !partnerRight.containsKey(name)).toList();
332 output.add('FREE ${free.isEmpty ? '-' : free.join(',')} ${freeRight.isEmpty ? '-' : freeRight.join(',')}');
333 output.add('COVER size=${cover.length}${cover.isEmpty ? '' : ' ${cover.join(' ')}'}');
334 if (free.isNotEmpty) {
335 final rightHall = right.where(zr.contains).toList();
336 output.add('HALL ${left.where(zl.contains).join(',')} ${rightHall.isEmpty ? '-' : rightHall.join(',')}');
337 }

339void assignment(String mode, int size, List<String> words) {
340 if (size < 1 || size > 8) fail('BAD SIZE');
341 if (words.length != size * size) fail('BAD LENGTH');
342 int? best;
343 var count = 0, chosen = <int>[];
344 final current = <int>[];
345 void visit(int row, int used, int sum) {
346 if (row == size) {
347 if (best == null || (mode == 'MIN' ? sum < best! : sum > best!)) {
348 best = sum;
349 count = 1;
350 chosen = List.of(current);
351 } else if (sum == best) {
352 count++;
353 }
354 return;
355 }
356 for (var col = 0; col < size; col++) {
357 if ((used & (1 << col)) == 0 && words[row * size + col] != 'x') {
358 current.add(col);
359 visit(row + 1, used | (1 << col), sum + int.parse(words[row * size + col]));
360 current.removeLast();
361 }
362 }
363 }
364 visit(0, 0, 0);
365 if (best == null) {
366 output.add('ASSIGN $mode n=$size infeasible');
367 } else {
368 output.add('ASSIGN $mode n=$size total=$best optimal=$count');
369 for (var row = 0; row < size; row++) {
370 final col = chosen[row];
371 output.add('MATE ${row + 1} ${col + 1} ${words[row * size + col]}');
372 }
373 }

549 case 'VERIFY':
550 final violations = <String>[], balance = balances(network);
551 for (var i = 0; i < network.edges.length; i++) {
552 final edge = network.edges[i];
553 if (edge.flow < edge.lower) violations.add('LOW ${i + 1} flow=${edge.flow} low=${edge.lower}');
554 }
555 for (var i = 0; i < network.names.length; i++) {
556 if (network.owner.kind == 'flow' && (network.owner.source == i || network.owner.sink == i)) continue;
557 final need = network.owner.kind == 'flow' ? 0 : network.demands[i];
558 if (-balance[i] != need) violations.add('BALANCE ${network.names[i]} net=${-balance[i]} need=$need');
559 }
560 output.addAll(violations.isEmpty ? ['VERIFY ok'] : ['VERIFY violations=${violations.length}', ...violations]);
561 case 'CIRCULATION':
562 if (a == 'MINCOST' && negativeCycle(network, true)) fail('NEGATIVE CYCLE');
563 final sum = network.demands.fold<int>(0, (total, demand) => total + demand);
564 if (sum != 0) {
565 output.add('CIRCULATION infeasible unbalanced=$sum');
566 } else {
567 final working = network.copy(), balance = List<int>.of(network.demands);
568 final source = working.names.length, sink = source + 1;
569 working.names.addAll(['superSource', 'superSink']);
570 working.demands.addAll([0, 0]);
571 for (var i = 0; i < network.edges.length; i++) {
572 final edge = working.edges[i], lower = network.edges[i].lower;
573 edge.capacity -= lower;
574 balance[edge.u] += lower;
575 balance[edge.v] -= lower;
576 }
577 for (var i = 0; i < balance.length; i++) {
578 if (balance[i] < 0) working.edges.add(Edge(source, i, -balance[i], 0));
579 }
580 for (var i = 0; i < balance.length; i++) {
581 if (balance[i] > 0) working.edges.add(Edge(i, sink, balance[i], 0));
582 }
583 if (a == 'ANY') { edmondsKarp(working, source, sink); } else { minCost(working, source, sink); }
584 final required = balance.where((value) => value > 0).fold<int>(0, (total, value) => total + value);
585 final achieved = working.edges.where((edge) => edge.v == sink).fold<int>(0, (total, edge) => total + edge.flow);
586 if (achieved < required) {
587 final level = bfs(working, source).level;
588 output.add('CIRCULATION infeasible required=$required achieved=$achieved');
589 output.add('CUT ${[for (var i = 0; i < network.names.length; i++) if (level[i] >= 0) network.names[i]].join(' ')}');
590 } else {
591 for (var i = 0; i < network.edges.length; i++) { network.edges[i].flow = network.edges[i].lower + working.edges[i].flow; }
592 network.owner = Owner('circulation');
593 output.add('CIRCULATION feasible required=$required cost=${totalCost(network)}');
594 flowList();
595 }
596 }
597 case 'MATCH':
598 matching(a, b, args[2]);
599 case 'ASSIGN':
600 assignment(a, int.parse(b), args.skip(2).toList());
601 }
602}
603void main() {
604 final count = int.parse(stdin.readLineSync()!);
605 for (var i = 0; i < count; i++) {
606 try { execute(stdin.readLineSync()!); }
607 on FlowError catch (error) { output.add(error.message); }
608 }
609 if (output.isNotEmpty) stdout.write('${output.join('\n')}\n');
610}
