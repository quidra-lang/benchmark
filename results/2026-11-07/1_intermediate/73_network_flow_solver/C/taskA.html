106static int arcs(Graph *g, int u, int *result) {
107 int count = 0;
108 for (int i = 0; i < g->m; i++) if (g->edges[i].u == u) result[count++] = i + 1;
109 for (int i = 0; i < g->m; i++) if (g->edges[i].v == u) result[count++] = -i - 1;
110 return count;
111}

113static void balances(Graph *g, int *b) {
114 memset(b, 0, (size_t)g->n * sizeof *b);
115 for (int i = 0; i < g->m; i++) {
116 Edge *e = &g->edges[i];
117 b[e->u] += e->flow;
118 b[e->v] -= e->flow;
119 }
120}

134static int bottleneck(Graph *g, Path *path) {
135 int result = INT_MAX;
136 for (int i = 0; i < path->count; i++) result = minimum(result, residual(g, path->arcs[i]));
137 return result;

155static Search bfs(Graph *g, int s, int threshold) {
156 Search search = {0};
157 for (int u = 0; u < g->n; u++) search.level[u] = -1;
158 int queue[WORK_NODES], head = 0, tail = 0;
159 search.level[s] = 0;
160 queue[tail++] = s;
161 while (head < tail) {
162 int u = queue[head++], list[2 * WORK_EDGES], count = arcs(g, u, list);
163 for (int i = 0; i < count; i++) {
164 int a = list[i], v = to(g, a);
165 if (residual(g, a) >= threshold && search.level[v] < 0) {
166 search.level[v] = search.level[u] + 1;
167 search.parent[v] = a;
168 queue[tail++] = v;
169 }
170 }
171 }
172 return search;
173}

175static Path parent_path(Graph *g, int s, int t, int *parent) {
176 Path result = {0};
177 for (int v = t; v != s; v = from(g, parent[v])) result.arcs[result.count++] = parent[v];
178 for (int i = 0; i < result.count / 2; i++) {
179 int a = result.arcs[i];
180 result.arcs[i] = result.arcs[result.count - 1 - i];
181 result.arcs[result.count - 1 - i] = a;
182 }
183 return result;
184}

196static Augmentations maximum(Graph *g, int s, int t, const char *method) {
197 Augmentations result = {0};
198 if (!strcmp(method, "DINIC")) {
199 int phase = 0;
200 for (;;) {
201 Search search = bfs(g, s, 1);
202 if (search.level[t] < 0) break;
203 phase++;
204 int lists[WORK_NODES][2 * WORK_EDGES], counts[WORK_NODES], pointer[WORK_NODES] = {0};
205 for (int u = 0; u < g->n; u++) counts[u] = arcs(g, u, lists[u]);
206 Path path = {0};
207 int u = s;
208 for (;;) {
209 if (u == t) {
210 int amount = bottleneck(g, &path);
211 record_augmentation(&result, path, amount, phase, 0);
212 push(g, &path, amount);
213 path.count = 0;
214 u = s;
215 } else if (pointer[u] == counts[u]) {
216 if (u == s) break;
217 u = from(g, path.arcs[--path.count]);
218 pointer[u]++;
219 } else {
220 int a = lists[u][pointer[u]], v = to(g, a);
221 if (residual(g, a) > 0 && search.level[v] == search.level[u] + 1) {
222 path.arcs[path.count++] = a;
223 u = v;
224 } else pointer[u]++;
225 }
226 }
227 }
228 return result;
229 }
230 int threshold = 1;
231 if (!strcmp(method, "SCALING")) {
232 int largest = 0;
233 for (int i = 0; i < g->m; i++) if (g->edges[i].capacity > largest) largest = g->edges[i].capacity;
234 while (threshold <= largest / 2) threshold *= 2;
235 }
236 do {
237 for (;;) {
238 Search search = bfs(g, s, threshold);
239 if (search.level[t] < 0) break;
240 Path path = parent_path(g, s, t, search.parent);
241 int amount = bottleneck(g, &path);
242 push(g, &path, amount);
243 record_augmentation(&result, path, amount, threshold, 0);
244 if (!strcmp(method, "AUGMENT")) return result;
245 }
246 threshold /= 2;
247 }
248 while (threshold > 0);
249 return result;
250}

257static bool negative_cycle(Graph *g, bool subtract_lower) {
258 int distance[WORK_NODES] = {0};
259 for (int pass = 0; pass < g->n; pass++) {
260 bool changed = false;
261 for (int i = 0; i < g->m; i++) {
262 Edge *e = &g->edges[i];
263 if (e->capacity - (subtract_lower ? e->lower : 0) > 0 && distance[e->v] > distance[e->u] + e->cost) {
264 distance[e->v] = distance[e->u] + e->cost;
265 changed = true;
266 }
267 }
268 if (!changed) return false;
269 }
270 return g->n != 0;
271}

273static Augmentations cheapest(Graph *g, int s, int t, int wanted) {
274 Augmentations result = {0};
275 int pushed = 0;
276 while (wanted < 0 || pushed < wanted) {
277 int usable[2 * WORK_EDGES], count = 0;
278 for (int i = 1; i <= g->m; i++) if (residual(g, i) > 0) usable[count++] = i;
279 for (int i = 1; i <= g->m; i++) if (residual(g, -i) > 0) usable[count++] = -i;
280 int distance[WORK_NODES] = {0}, parent[WORK_NODES] = {0};
281 bool known[WORK_NODES] = {false}, changed;
282 known[s] = true;
283 do {
284 changed = false;
285 for (int i = 0; i < count; i++) {
286 int a = usable[i], u = from(g, a), v = to(g, a);
287 if (known[u] && (!known[v] || distance[u] + arc_cost(g, a) < distance[v])) {
288 distance[v] = distance[u] + arc_cost(g, a);
289 known[v] = true;
290 parent[v] = a;
291 changed = true;
292 }
293 }
294 }
295 while (changed);
296 if (!known[t]) break;
297 Path path = parent_path(g, s, t, parent);
298 int amount = bottleneck(g, &path);
299 if (wanted >= 0) amount = minimum(amount, wanted - pushed);
300 push(g, &path, amount);
301 pushed += amount;
302 record_augmentation(&result, path, amount, 0, distance[t]);
303 }
304 return result;
305}

317static Pieces decompose(Graph *g) {
318 int remaining[WORK_EDGES], balance[WORK_NODES];
319 balances(g, balance);
320 for (int i = 0; i < g->m; i++) remaining[i] = g->edges[i].flow;
321 Pieces result = {0};
322 for (;;) {
323 int first = -1;
324 for (int i = 0; i < g->m; i++) if (remaining[i] > 0) {
325 first = i;
326 break;
327 }
328 if (first < 0) break;
329 int start = -1;
330 for (int u = 0; u < g->n; u++) if (balance[u] > 0) {
331 start = u;
332 break;
333 }
334 if (start < 0) start = g->edges[first].u;
335 int vertices[WORK_NODES], length = 1;
336 vertices[0] = start;
337 Path path = {0};
338 for (;;) {
339 int u = vertices[length - 1], i = 0;
340 while (g->edges[i].u != u || remaining[i] == 0) i++;
341 int v = g->edges[i].v;
342 path.arcs[path.count++] = i + 1;
343 int repeated = -1;
344 for (int j = 0; j < length; j++) if (vertices[j] == v) {
345 repeated = j;
346 break;
347 }
348 if (repeated >= 0) {
349 Path cycle = {0};
350 int amount = INT_MAX;
351 for (int j = repeated; j < path.count; j++) {
352 int a = path.arcs[j];
353 cycle.arcs[cycle.count++] = a;
354 amount = minimum(amount, remaining[a - 1]);
355 }
356 for (int j = 0; j < cycle.count; j++) remaining[cycle.arcs[j] - 1] -= amount;
357 record_piece(&result, true, v, cycle, amount);
358 break;
359 }
360 vertices[length++] = v;
361 if (balance[v] < 0) {
362 int amount = minimum(balance[start], -balance[v]);
363 for (int j = 0; j < path.count; j++) amount = minimum(amount, remaining[path.arcs[j] - 1]);
364 for (int j = 0; j < path.count; j++) remaining[path.arcs[j] - 1] -= amount;
365 balance[start] -= amount;
366 balance[v] += amount;
367 record_piece(&result, false, start, path, amount);
368 break;
369 }
370 }
371 }
372 return result;
373}

426static bool try_match(Matching *m, int u, bool *examined) {
427 for (int i = 0; i < m->np; i++) if (m->pairs[i][0] == u) {
428 int v = m->pairs[i][1];
429 if (examined[v]) continue;
430 examined[v] = true;
431 if (m->b[v] < 0 || try_match(m, m->b[v], examined)) {
432 m->a[u] = v;
433 m->b[v] = u;
434 return true;
435 }
436 }
437 return false;
438}

450static void matching(char **w) {
451 Matching m = {0};
452 m.na = split(w[1], ',', m.left);
453 m.nb = split(w[2], ',', m.right);
454 for (int i = 0; i < m.na; i++) for (int j = 0; j < i; j++) if (!strcmp(m.left[i], m.left[j])) fail("DUPLICATE VERTEX %s", m.left[i]);
455 for (int i = 0; i < m.nb; i++) {
456 for (int j = 0; j < i; j++) if (!strcmp(m.right[i], m.right[j])) fail("DUPLICATE VERTEX %s", m.right[i]);
457 for (int j = 0; j < m.na; j++) if (!strcmp(m.right[i], m.left[j])) fail("DUPLICATE VERTEX %s", m.right[i]);
458 }
459 if (strcmp(w[3], "-")) {
460 char *pairs[60];
461 m.np = split(w[3], ',', pairs);
462 for (int i = 0; i < m.np; i++) {
463 char *p[2];
464 split(pairs[i], '-', p);
465 int u = -1, v = -1;
466 for (int j = 0; j < m.na; j++) if (!strcmp(p[0], m.left[j])) u = j;
467 for (int j = 0; j < m.nb; j++) if (!strcmp(p[1], m.right[j])) v = j;
468 if (u < 0) fail("UNKNOWN VERTEX %s", p[0]);
469 if (v < 0) fail("UNKNOWN VERTEX %s", p[1]);
470 for (int j = 0; j < i; j++) if (m.pairs[j][0] == u && m.pairs[j][1] == v) fail("DUPLICATE PAIR %s-%s", p[0], p[1]);
471 m.pairs[i][0] = u;
472 m.pairs[i][1] = v;
473 }
474 }
475 for (int i = 0; i < m.na; i++) m.a[i] = -1;
476 for (int i = 0; i < m.nb; i++) m.b[i] = -1;
477 for (int i = 0; i < m.na; i++) {
478 bool examined[12] = {false};
479 try_match(&m, i, examined);
480 }
481 bool za[12] = {false}, zb[12] = {false}, free_a[12] = {false}, free_b[12] = {false};
482 int queue[12], head = 0, tail = 0, size = 0;
483 for (int u = 0; u < m.na; u++) {
484 if (m.a[u] < 0) {
485 free_a[u] = za[u] = true;
486 queue[tail++] = u;
487 } else size++;
488 }
489 while (head < tail) {
490 int u = queue[head++];
491 for (int i = 0; i < m.np; i++) if (m.pairs[i][0] == u) {
492 int v = m.pairs[i][1];
493 if (m.a[u] == v || zb[v]) continue;
494 zb[v] = true;
495 if (m.b[v] >= 0 && !za[m.b[v]]) {
496 za[m.b[v]] = true;
497 queue[tail++] = m.b[v];
498 }
499 }
500 }
501 printf("MATCH size=%d left=%d right=%d\n", size, m.na, m.nb);
502 for (int u = 0; u < m.na; u++) if (m.a[u] >= 0) printf("PAIR %s %s\n", m.left[u], m.right[m.a[u]]);
503 for (int v = 0; v < m.nb; v++) free_b[v] = m.b[v] < 0;
504 fputs("FREE ", stdout);
505 comma_names(m.left, m.na, free_a);
506 putchar(' ');
507 comma_names(m.right, m.nb, free_b);
508 putchar('\n');
509 printf("COVER size=%d", size);
510 for (int u = 0; u < m.na; u++) if (!za[u]) printf(" %s", m.left[u]);
511 for (int v = 0; v < m.nb; v++) if (zb[v]) printf(" %s", m.right[v]);
512 putchar('\n');
513 if (size < m.na) {
514 fputs("HALL ", stdout);
515 comma_names(m.left, m.na, za);
516 putchar(' ');
517 comma_names(m.right, m.nb, zb);
518 putchar('\n');
519 }

521typedef struct {
522 int n, values[8][8], best, count, chosen[8], current[8];
523 bool forbidden[8][8], minimize;
524} Assignment;
525
526static void assign_visit(Assignment *a, int row, unsigned used, int sum) {
527 if (row == a->n) {
528 if (a->count == 0 || (a->minimize ? sum < a->best : sum > a->best)) {
529 a->best = sum;
530 a->count = 1;
531 memcpy(a->chosen, a->current, sizeof a->chosen);
532 } else if (sum == a->best) a->count++;
533 return;
534 }
535 for (int col = 0; col < a->n; col++) if (!(used & (1u << col)) && !a->forbidden[row][col]) {
536 a->current[row] = col;
537 assign_visit(a, row + 1, used | (1u << col), sum + a->values[row][col]);
538 }
539}

541static void assignment(char **w, int count) {
542 int n = atoi(w[2]);
543 if (n < 1 || n > 8) fail("BAD SIZE");
544 if (count - 3 != n * n) fail("BAD LENGTH");
545 Assignment a = {0};
546 a.n = n;
547 a.minimize = !strcmp(w[1], "MIN");
548 for (int i = 0; i < n * n; i++) {
549 a.forbidden[i / n][i % n] = !strcmp(w[i + 3], "x");
550 if (!a.forbidden[i / n][i % n]) a.values[i / n][i % n] = atoi(w[i + 3]);
551 }
552 assign_visit(&a, 0, 0, 0);
553 if (a.count == 0) printf("ASSIGN %s n=%d infeasible\n", w[1], n);
554 else {
555 printf("ASSIGN %s n=%d total=%d optimal=%d\n", w[1], n, a.best, a.count);
556 for (int i = 0; i < n; i++) printf("MATE %d %d %d\n", i + 1, a.chosen[i] + 1, a.values[i][a.chosen[i]]);
557 }
558}

560static void command(char *line) {
561 char *w[1001];
562 int count = split(line, ' ', w);
563 char *c = w[0];
564 if (!strcmp(c, "NODE")) {
565 for (int i = 1; i < count; i++) {
566 for (int j = 0; j < network.n; j++) if (!strcmp(w[i], network.names[j])) fail("DUPLICATE NODE %s", w[i]);
567 for (int j = 1; j < i; j++) if (!strcmp(w[i], w[j])) fail("DUPLICATE NODE %s", w[i]);
568 }
569 if (network.n + count - 1 > MAX_NODES) fail("TOO MANY NODES");
570 for (int i = 1; i < count; i++) {
571 demand[network.n] = 0;
572 network.names[network.n++] = copy_string(w[i]);
573 }
574 printf("NODE added=%d total=%d\n", count - 1, network.n);
575 } else if (!strcmp(c, "EDGE")) {
576 int u = node(w[1]), v = node(w[2]);
577 if (u == v) fail("SELF LOOP");
578 if (network.m == MAX_EDGES) fail("TOO MANY EDGES");
579 int capacity = atoi(w[3]), cost = count == 5 ? atoi(w[4]) : 0;
580 add_edge(&network, u, v, capacity, cost);
581 printf("EDGE %d %s %s cap=%d cost=%d\n", network.m, w[1], w[2], capacity, cost);
582 } else if (!strcmp(c, "LOWER") || !strcmp(c, "CAP") || !strcmp(c, "COST")) {
583 Edge *e = checked_edge(w[1]);
584 int number = atoi(w[1]), amount = atoi(w[2]);
585 if (!strcmp(c, "LOWER")) {
586 if (amount > e->capacity) fail("BAD BOUND");
587 e->lower = amount;
588 printf("LOWER %d low=%d cap=%d\n", number, amount, e->capacity);
589 } else if (!strcmp(c, "CAP")) {
590 if (amount < e->lower) fail("BAD BOUND");
591 if (amount < e->flow) fail("BELOW FLOW");
592 e->capacity = amount;
593 printf("CAP %d low=%d cap=%d\n", number, e->lower, amount);
594 } else {
595 e->cost = amount;
596 printf("COST %d cost=%d\n", number, amount);
597 }
598 } else if (!strcmp(c, "DEMAND")) {
599 int u = node(w[1]), amount = atoi(w[2]), sum = 0;
600 demand[u] = amount;
601 for (int i = 0; i < network.n; i++) sum += demand[i];
602 printf("DEMAND %s %d total=%d\n", w[1], amount, sum);
603 } else if (!strcmp(c, "RESET")) {
604 int changed = 0;
605 for (int i = 0; i < network.m; i++) {
606 if (network.edges[i].flow) changed++;
607 network.edges[i].flow = 0;
608 }
609 owner = 0;
610 printf("RESET edges=%d\n", changed);
611 } else if (!strcmp(c, "CLEAR")) {
612 printf("CLEAR nodes=%d edges=%d\n", network.n, network.m);
613 for (int i = 0; i < network.n; i++) free(network.names[i]);
614 memset(&network, 0, sizeof network);
615 owner = 0;
616 } else if (!strcmp(c, "SHOW")) {
617 printf("NETWORK nodes=%d edges=%d\n", network.n, network.m);
618 if (owner == 0) puts("STATE none");
619 else if (owner == 2) printf("STATE circulation cost=%d\n", total_cost(&network));
620 else printf("STATE flow %s %s value=%d cost=%d\n", network.names[source], network.names[target], value(&network, source), total_cost(&network));
621 for (int u = 0; u < network.n; u++) {
622 int sent = 0, received = 0;
623 for (int i = 0; i < network.m; i++) {
624 if (network.edges[i].u == u) sent += network.edges[i].flow;
625 if (network.edges[i].v == u) received += network.edges[i].flow;
626 }
627 printf("NODE %s demand=%d out=%d in=%d\n", network.names[u], demand[u], sent, received);
628 }
629 for (int i = 0; i < network.m; i++) {
630 Edge *e = &network.edges[i];
631 printf("EDGE %d %s %s cap=%d cost=%d low=%d flow=%d\n", i + 1, network.names[e->u], network.names[e->v], e->capacity, e->cost, e->lower, e->flow);
632 }
633 } else if (!strcmp(c, "AUGMENT") || !strcmp(c, "MAXFLOW") || !strcmp(c, "DINIC") || !strcmp(c, "SCALING") || !strcmp(c, "MINCUT")) {
634 int s, t;
635 endpoints(w[1], w[2], &s, &t);
636 int initial = start_flow(s, t);
637 Augmentations paths = maximum(&network, s, t, c);
638 printf("%s %s %s value=%d", c, w[1], w[2], value(&network, s));
639 if (!strcmp(c, "AUGMENT")) {
640 if (!paths.count) puts(" none");
641 else {
642 printf(" amount=%d ", paths.items[0].amount);
643 notation(&network, s, &paths.items[0].path);
644 putchar('\n');
645 }
646 } else if (!strcmp(c, "MINCUT")) {
647 Search search = bfs(&network, s, 1);
648 int cut = 0;
649 for (int i = 0; i < network.m; i++) {
650 Edge *e = &network.edges[i];
651 if (search.level[e->u] >= 0 && search.level[e->v] < 0) cut++;
652 }
653 printf(" edges=%d\nSOURCE", cut);
654 for (int u = 0; u < network.n; u++) if (search.level[u] >= 0) printf(" %s", network.names[u]);
655 fputs("\nSINK", stdout);
656 for (int u = 0; u < network.n; u++) if (search.level[u] < 0) printf(" %s", network.names[u]);
657 putchar('\n');
658 for (int i = 0; i < network.m; i++) {
659 Edge *e = &network.edges[i];
660 if (search.level[e->u] >= 0 && search.level[e->v] < 0) printf("CUT %d %s %s %d\n", i + 1, network.names[e->u], network.names[e->v], e->capacity);
661 }
662 } else {
663 printf(" start=%d", initial);
664 if (!strcmp(c, "DINIC")) printf(" phases=%d", paths.count ? paths.items[paths.count - 1].stage : 0);
665 printf(" paths=%d\n", paths.count);
666 for (int i = 0; i < paths.count; i++) {
667 Augmentation *a = &paths.items[i];
668 printf("AUG %d", i + 1);
669 if (!strcmp(c, "DINIC")) printf(" phase=%d", a->stage);
670 if (!strcmp(c, "SCALING")) printf(" delta=%d", a->stage);
671 printf(" amount=%d ", a->amount);
672 notation(&network, s, &a->path);
673 putchar('\n');
674 }
675 flow_list();
676 }
677 free(paths.items);
678 } else if (!strcmp(c, "SENSITIVITY")) {
679 int s, t;
680 endpoints(w[1], w[2], &s, &t);
681 Graph g = zero_copy(&network);
682 finish_maximum(&g, s, t);
683 int amount = value(&g, s), up_count = 0, down_count = 0;
684 bool up[MAX_EDGES], down[MAX_EDGES];
685 for (int i = 0; i < network.m; i++) {
686 g = zero_copy(&network);
687 g.edges[i].capacity++;
688 finish_maximum(&g, s, t);
689 up[i] = value(&g, s) > amount;
690 down[i] = false;
691 if (network.edges[i].capacity > 0) {
692 g = zero_copy(&network);
693 g.edges[i].capacity--;
694 finish_maximum(&g, s, t);
695 down[i] = value(&g, s) < amount;
696 }
697 up_count += up[i];
698 down_count += down[i];
699 }
700 printf("SENSITIVITY %s %s value=%d up=%d down=%d\n", w[1], w[2], amount, up_count, down_count);
701 for (int i = 0; i < network.m; i++) if (up[i] || down[i]) {
702 Edge *e = &network.edges[i];
703 printf("CRITICAL %d %s %s up=%s down=%s\n", i + 1, network.names[e->u], network.names[e->v], up[i] ? "yes" : "no", down[i] ? "yes" : "no");
704 }
705 } else if (!strcmp(c, "MINCOST")) {
706 int s, t;
707 endpoints(w[1], w[2], &s, &t);
708 if (negative_cycle(&network, false)) fail("NEGATIVE CYCLE");
709 int wanted = !strcmp(w[3], "MAX") ? -1 : atoi(w[3]);
710 Graph g = zero_copy(&network);
711 Augmentations paths = cheapest(&g, s, t, wanted);
712 int amount = value(&g, s);
713 printf("MINCOST %s %s k=%s", w[1], w[2], w[3]);
714 if (wanted >= 0 && amount < wanted) printf(" infeasible max=%d\n", amount);
715 else {
716 copy_flows(&g, false);
717 owner = 1;
718 source = s;
719 target = t;
720 printf(" flow=%d cost=%d\n", amount, total_cost(&g));
721 for (int i = 0; i < paths.count; i++) {
722 Augmentation *a = &paths.items[i];
723 printf("AUG %d amount=%d cost=%d ", i + 1, a->amount, a->unit_cost);
724 notation(&g, s, &a->path);
725 putchar('\n');
726 }
727 flow_list();
728 }
729 free(paths.items);
730 } else if (!strcmp(c, "DISJOINT")) {
731 int s, t;
732 endpoints(w[1], w[2], &s, &t);
733 Graph g = {0};
734 g.n = network.n;
735 memcpy(g.names, network.names, (size_t)g.n * sizeof *g.names);
736 int exits[MAX_NODES];
737 for (int u = 0; u < network.n; u++) {
738 exits[u] = u;
739 if (!strcmp(w[3], "VERTEX") && u != s && u != t) {
740 exits[u] = g.n;
741 g.names[g.n++] = network.names[u];
742 }
743 }
744 for (int i = 0; i < network.m; i++) {
745 Edge *e = &network.edges[i];
746 add_edge(&g, exits[e->u], e->v, e->capacity > 0, 0);
747 }
748 for (int u = 0; u < network.n; u++) if (exits[u] != u) add_edge(&g, u, exits[u], 1, 0);
749 finish_maximum(&g, s, t);
750 Pieces pieces = decompose(&g);
751 int paths = 0;
752 for (int i = 0; i < pieces.count; i++) if (!pieces.items[i].cycle) paths++;
753 printf("DISJOINT %s %s %s count=%d\n", w[1], w[2], w[3], paths);
754 int number = 0;
755 for (int i = 0; i < pieces.count; i++) if (!pieces.items[i].cycle) {
756 printf("PATH %d %s", ++number, w[1]);
757 Path *path = &pieces.items[i].path;
758 for (int j = 0; j < path->count; j++) {
759 int a = path->arcs[j];
760 if (a <= network.m) printf(" +%d %s", a, network.names[network.edges[a - 1].v]);
761 }
762 putchar('\n');
763 }
764 free(pieces.items);
765 } else if (!strcmp(c, "DECOMPOSE")) {
766 Pieces pieces = decompose(&network);
767 int cycles = 0, path_number = 0, cycle_number = 0;
768 for (int i = 0; i < pieces.count; i++) cycles += pieces.items[i].cycle;
769 printf("DECOMPOSE paths=%d cycles=%d\n", pieces.count - cycles, cycles);
770 for (int i = 0; i < pieces.count; i++) {
771 Piece *p = &pieces.items[i];
772 printf("%s %d amount=%d ", p->cycle ? "CYCLE" : "PATH", p->cycle ? ++cycle_number : ++path_number, p->amount);
773 notation(&network, p->start, &p->path);
774 putchar('\n');
775 }
776 free(pieces.items);
777 } else if (!strcmp(c, "VERIFY")) {
778 int violations = 0, balance[WORK_NODES];
779 balances(&network, balance);
780 for (int i = 0; i < network.m; i++) if (network.edges[i].flow < network.edges[i].lower) violations++;
781 for (int u = 0; u < network.n; u++) if ((owner != 1 || (u != source && u != target)) && -balance[u] != (owner == 1 ? 0 : demand[u])) violations++;
782 if (!violations) puts("VERIFY ok");
783 else {
784 printf("VERIFY violations=%d\n", violations);
785 for (int i = 0; i < network.m; i++) {
786 Edge *e = &network.edges[i];
787 if (e->flow < e->lower) printf("LOW %d flow=%d low=%d\n", i + 1, e->flow, e->lower);
788 }
789 for (int u = 0; u < network.n; u++) if ((owner != 1 || (u != source && u != target)) && -balance[u] != (owner == 1 ? 0 : demand[u])) printf("BALANCE %s net=%d need=%d\n", network.names[u], -balance[u], owner == 1 ? 0 : demand[u]);
790 }
791 } else if (!strcmp(c, "CIRCULATION")) {
792 bool mincost = !strcmp(w[1], "MINCOST");
793 if (mincost && negative_cycle(&network, true)) fail("NEGATIVE CYCLE");
794 int sum = 0;
795 for (int u = 0; u < network.n; u++) sum += demand[u];
796 if (sum) {
797 printf("CIRCULATION infeasible unbalanced=%d\n", sum);
798 return;
799 }
800 Graph g = zero_copy(&network);
801 int n = g.n, remaining[MAX_NODES], required = 0;
802 g.names[g.n++] = "source";
803 g.names[g.n++] = "sink";
804 memcpy(remaining, demand, (size_t)n * sizeof *remaining);
805 for (int i = 0; i < g.m; i++) {
806 Edge *e = &g.edges[i];
807 remaining[e->u] += e->lower;
808 remaining[e->v] -= e->lower;
809 e->capacity -= e->lower;
810 }
811 for (int u = 0; u < n; u++) if (remaining[u] < 0) add_edge(&g, n, u, -remaining[u], 0);
812 for (int u = 0; u < n; u++) if (remaining[u] > 0) {
813 add_edge(&g, u, n + 1, remaining[u], 0);
814 required += remaining[u];
815 }
816 if (mincost) {
817 Augmentations paths = cheapest(&g, n, n + 1, -1);
818 free(paths.items);
819 } else finish_maximum(&g, n, n + 1);
820 int achieved = -value(&g, n + 1);
821 if (achieved < required) {
822 printf("CIRCULATION infeasible required=%d achieved=%d\nCUT", required, achieved);
823 Search search = bfs(&g, n, 1);
824 for (int u = 0; u < n; u++) if (search.level[u] >= 0) printf(" %s", network.names[u]);
825 putchar('\n');
826 } else {
827 copy_flows(&g, true);
828 owner = 2;
829 printf("CIRCULATION feasible required=%d cost=%d\n", required, total_cost(&network));
830 flow_list();
831 }
832 } else if (!strcmp(c, "MATCH")) matching(w);
833 else if (!strcmp(c, "ASSIGN")) assignment(w, count);
834}

836static char *read_line(void) {
837 size_t length = 0, capacity = 128;
838 char *line = allocate(capacity);
839 int ch;
840 while ((ch = getchar()) != EOF && ch != '\n') {
841 if (length + 1 == capacity) {
842 capacity *= 2;
843 char *p = realloc(line, capacity);
844 if (!p) exit(EXIT_FAILURE);
845 line = p;
846 }
847 line[length++] = (char)ch;
848 }
849 line[length] = '\0';
850 return line;
851}

858 if (setjmp(command_error) == 0) command(line);
