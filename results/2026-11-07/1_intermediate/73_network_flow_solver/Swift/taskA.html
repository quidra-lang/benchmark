44final class Network {
45 var names = [String]()
46 var demands = [Int]()
47 var edges = [Edge]()
48 var owner = Owner.none
49 func copy() -> Network {
50 let n = Network()
51 n.names = names
52 n.demands = demands
53 n.edges = edges
54 n.owner = owner
55 return n
56 }
57 func node(_ name: String) throws -> Int {
58 guard let i = names.firstIndex(of: name) else {
59 try fail("UNKNOWN NODE \(name)")
60 }
61 return i
62 }
63 func endpoints(_ a: String, _ b: String) throws -> (Int, Int) {
64 let s = try node(a)
65 let t = try node(b)
66 if s == t {
67 try fail("SAME NODE")
68 }
69 return (s,t)
70 }
71 func edgeNumber(_ text: String) throws -> Int {
72 let i = Int(text)!
73 if i < 1 || i > edges.count {
74 try fail("UNKNOWN EDGE \(text)")
75 }
76 return i-1
77 }
78 func globalArcs() -> [Arc] {
79 [true,false].flatMap {
80 forward in edges.indices.map {
81 Arc(edge: $0,forward: forward)
82 }
83 }
84 }
85 func arcs(_ node: Int) -> [Arc] {
86 globalArcs().filter {
87 start($0) == node
88 }
89 }
90 func start(_ a: Arc) -> Int {
91 a.forward ? edges[a.edge].from : edges[a.edge].to
92 }
93 func end(_ a: Arc) -> Int {
94 a.forward ? edges[a.edge].to : edges[a.edge].from
95 }
96 func residual(_ a: Arc) -> Int {
97 a.forward ? edges[a.edge].capacity-edges[a.edge].flow : edges[a.edge].flow
98 }
99 func arcCost(_ a: Arc) -> Int {
100 a.forward ? edges[a.edge].cost : -edges[a.edge].cost
101 }
102 func push(_ path: [Arc], _ amount: Int) {
103 for a in path {
104 edges[a.edge].flow += a.forward ? amount : -amount
105 }
106 }
107 func bottleneck(_ path: [Arc]) -> Int {
108 path.map {
109 residual($0)
110 }.min()!
111 }
112 func balances() -> [Int] {
113 var b = Array(repeating: 0,count: names.count)
114 for e in edges {
115 b[e.from] += e.flow
116 b[e.to] -= e.flow
117 }
118 return b
119 }
120 func cost() -> Int {
121 edges.reduce(0) {
122 $0+$1.flow*$1.cost
123 }
124 }
125 func zero() {
126 for i in edges.indices {
127 edges[i].flow = 0
128 }
129 }
130 func begin(_ s: Int, _ t: Int) -> Int {
131 if owner != .flow(s,t) {
132 zero()
133 }
134 owner = .flow(s,t)
135 return balances()[s]
136 }
137 func bfs(_ s: Int, _ delta: Int = 1) -> ([Int],[Arc?]) {
138 var levels = Array(repeating: -1,count: names.count)
139 var parent = [Arc?](repeating: nil,count: names.count)
140 var queue = [s]
141 var head = 0
142 levels[s] = 0
143 while head < queue.count {
144 let u = queue[head]
145 head += 1
146 for a in arcs(u) {
147 let v = end(a)
148 if residual(a) >= delta && levels[v] < 0 {
149 levels[v] = levels[u]+1
150 parent[v] = a
151 queue.append(v)
152 }
153 }
154 }
155 return (levels,parent)
156 }
157 func path(_ s: Int, _ t: Int, _ parent: [Arc?]) -> [Arc] {
158 var result = [Arc]()
159 var u = t
160 while u != s {
161 let a = parent[u]!
162 result.append(a)
163 u = start(a)
164 }
165 return Array(result.reversed())
166 }
167 func augment(_ s: Int, _ t: Int, _ delta: Int = 1) -> Pushed? {
168 let (level,parent) = bfs(s,delta)
169 if level[t] < 0 {
170 return nil
171 }
172 let p = path(s,t,parent)
173 let amount = bottleneck(p)
174 push(p,amount)
175 return Pushed(arcs: p,amount: amount,delta: delta)
176 }
177 @discardableResult func maxflow(_ s: Int, _ t: Int) -> [Pushed] {
178 var result = [Pushed]()
179 while let p = augment(s,t) {
180 result.append(p)
181 }
182 return result
183 }
184 func dinic(_ s: Int, _ t: Int) -> (Int,[Pushed]) {
185 var phases = 0
186 var result = [Pushed]()
187 while true {
188 let (level,_) = bfs(s)
189 if level[t] < 0 {
190 return (phases,result)
191 }
192 phases += 1
193 let lists = names.indices.map {
194 arcs($0)
195 }
196 var pointers = Array(repeating: 0,count: names.count)
197 var p = [Arc]()
198 var u = s
199 while true {
200 if u == t {
201 let amount = bottleneck(p)
202 push(p,amount)
203 result.append(Pushed(arcs: p,amount: amount,phase: phases))
204 p = []
205 u = s
206 } else if pointers[u] == lists[u].count {
207 if u == s {
208 break
209 }
210 let a = p.removeLast()
211 u = start(a)
212 pointers[u] += 1
213 } else {
214 let a = lists[u][pointers[u]]
215 let v = end(a)
216 if residual(a) > 0 && level[v] >= 0 && level[v] == level[u]+1 {
217 p.append(a)
218 u = v
219 } else {
220 pointers[u] += 1
221 }
222 }
223 }
224 }
225 }
226 func scaling(_ s: Int, _ t: Int) -> [Pushed] {
227 let maximum = edges.map(\.capacity).max() ?? 0
228 var delta = 1
229 while delta*2 <= maximum {
230 delta *= 2
231 }
232 var result = [Pushed]()
233 while delta >= 1 {
234 while let p = augment(s,t,delta) {
235 result.append(p)
236 }
237 delta /= 2
238 }
239 return result
240 }
241 func negativeCycle(_ lower: Bool) -> Bool {
242 var distance = Array(repeating: 0,count: names.count)
243 for pass in names.indices {
244 var changed = false
245 for e in edges where e.capacity-(lower ? e.lower : 0) > 0 {
246 if distance[e.to] > distance[e.from]+e.cost {
247 distance[e.to] = distance[e.from]+e.cost
248 changed = true
249 }
250 }
251 if !changed {
252 return false
253 }
254 if pass == names.count-1 {
255 return true
256 }
257 }
258 return false
259 }
260 func shortest(_ s: Int) -> ([Int?],[Arc?]) {
261 let q = globalArcs().filter {
262 residual($0) > 0
263 }
264 var distance = [Int?](repeating: nil,count: names.count)
265 var parent = [Arc?](repeating: nil,count: names.count)
266 distance[s] = 0
267 while true {
268 var changed = false
269 for a in q {
270 let u = start(a)
271 let v = end(a)
272 if let d = distance[u] {
273 let next = d+arcCost(a)
274 if distance[v] == nil || next < distance[v]! {
275 distance[v] = next
276 parent[v] = a
277 changed = true
278 }
279 }
280 }
281 if !changed {
282 return (distance,parent)
283 }
284 }
285 }
286 func mincost(_ s: Int, _ t: Int, _ target: Int? = nil) -> (Int,Int,[Pushed]) {
287 var amount = 0
288 var cost = 0
289 var result = [Pushed]()
290 while target == nil || amount < target! {
291 let (distance,parent) = shortest(s)
292 guard let unit = distance[t] else {
293 break
294 }
295 let p = path(s,t,parent)
296 let delta = min(bottleneck(p),target.map {
297 $0-amount
298 } ?? Int.max)
299 push(p,delta)
300 amount += delta
301 cost += delta*unit
302 result.append(Pushed(arcs: p,amount: delta,cost: unit))
303 }
304 return (amount,cost,result)
305 }
306 func notation(_ s: Int, _ path: [Arc]) -> String {
307 names[s]+path.map {
308 " \($0.forward ? "+" : "-")\($0.edge+1) \(names[end($0)])"
309 }.joined()
310 }
311 func flows() -> [String] {
312 edges.enumerated().map {
313 i,e in "FLOW \(i+1) \(names[e.from]) \(names[e.to]) \(e.flow)/\(e.capacity)"
314 }
315 }
316 func decompose() -> [Component] {
317 var g = edges.map(\.flow)
318 var balance = balances()
319 var result = [Component]()
320 while let first = g.firstIndex(where: {
321 $0 > 0
322 }) {
323 let initial = balance.firstIndex(where: {
324 $0 > 0
325 }) ?? edges[first].from
326 var vertices = [initial]
327 var p = [Int]()
328 while true {
329 let u = vertices.last!
330 let i = edges.indices.first {
331 edges[$0].from == u && g[$0] > 0
332 }!
333 let v = edges[i].to
334 p.append(i)
335 if let q = vertices.firstIndex(of: v) {
336 let cycle = Array(p.dropFirst(q))
337 let delta = cycle.map {
338 g[$0]
339 }.min()!
340 for e in cycle {
341 g[e] -= delta
342 }
343 result.append(Component(start: v,edges: cycle,amount: delta,cycle: true))
344 break
345 }
346 vertices.append(v)
347 if balance[v] < 0 {
348 let delta = min(p.map {
349 g[$0]
350 }.min()!,balance[initial],-balance[v])
351 for e in p {
352 g[e] -= delta
353 }
354 balance[initial] -= delta
355 balance[v] += delta
356 result.append(Component(start: initial,edges: p,amount: delta,cycle: false))
357 break
358 }
359 }
360 }
361 return result
362 }
363 func command(_ w: [String]) throws -> [String] {
364 let c = w[0]
365 let a = Array(w.dropFirst())
366 func num(_ i: Int) -> Int {
367 Int(a[i])!
368 }
369 var out = [String]()
370 switch c {
371 case "NODE":
372 var seen = Set(names)
373 for name in a {
374 if !seen.insert(name).inserted {
375 try fail("DUPLICATE NODE \(name)")
376 }
377 }
378 if names.count+a.count > 20 {
379 try fail("TOO MANY NODES")
380 }
381 names += a
382 demands += Array(repeating: 0,count: a.count)
383 out.append("NODE added=\(a.count) total=\(names.count)")
384 case "EDGE":
385 let u = try node(a[0])
386 let v = try node(a[1])
387 if u == v {
388 try fail("SELF LOOP")
389 }
390 if edges.count == 60 {
391 try fail("TOO MANY EDGES")
392 }
393 let cost = a.count == 4 ? num(3) : 0
394 edges.append(Edge(from: u,to: v,capacity: num(2),cost: cost))
395 out.append("EDGE \(edges.count) \(a[0]) \(a[1]) cap=\(num(2)) cost=\(cost)")
396 case "LOWER","CAP","COST":
397 let i = try edgeNumber(a[0])
398 let n = num(1)
399 if c == "LOWER" {
400 if n > edges[i].capacity {
401 try fail("BAD BOUND")
402 }
403 edges[i].lower = n
404 out.append("LOWER \(a[0]) low=\(n) cap=\(edges[i].capacity)")
405 } else if c == "CAP" {
406 if n < edges[i].lower {
407 try fail("BAD BOUND")
408 }
409 if n < edges[i].flow {
410 try fail("BELOW FLOW")
411 }
412 edges[i].capacity = n
413 out.append("CAP \(a[0]) low=\(edges[i].lower) cap=\(n)")
414 } else {
415 edges[i].cost = n
416 out.append("COST \(a[0]) cost=\(n)")
417 }
418 case "DEMAND":
419 let i = try node(a[0])
420 demands[i] = num(1)
421 out.append("DEMAND \(a[0]) \(num(1)) total=\(demands.reduce(0,+))")
422 case "RESET":
423 out.append("RESET edges=\(edges.filter { $0.flow != 0 }.count)")
424 zero()
425 owner = .none
426 case "CLEAR":
427 out.append("CLEAR nodes=\(names.count) edges=\(edges.count)")
428 names = []
429 demands = []
430 edges = []
431 owner = .none
432 case "SHOW":
433 let balance = balances()
434 out.append("NETWORK nodes=\(names.count) edges=\(edges.count)")
435 switch owner {
436 case .none:
437 out.append("STATE none")
438 case .circulation:
439 out.append("STATE circulation cost=\(cost())")
440 case let .flow(s,t):
441 out.append("STATE flow \(names[s]) \(names[t]) value=\(balance[s]) cost=\(cost())")
442 }
443 for (i,name) in names.enumerated() {
444 out.append("NODE \(name) demand=\(demands[i]) out=\(edges.filter { $0.from == i }.reduce(0) { $0+$1.flow }) in=\(edges.filter { $0.to == i }.reduce(0) { $0+$1.flow })")
445 }
446 for (i,e) in edges.enumerated() {
447 out.append("EDGE \(i+1) \(names[e.from]) \(names[e.to]) cap=\(e.capacity) cost=\(e.cost) low=\(e.lower) flow=\(e.flow)")
448 }
449 case "AUGMENT","MAXFLOW","DINIC","SCALING","MINCUT":
450 let (s,t) = try endpoints(a[0],a[1])
451 let initial = begin(s,t)
452 if c == "AUGMENT" {
453 if let p = augment(s,t) {
454 out.append("AUGMENT \(a[0]) \(a[1]) value=\(balances()[s]) amount=\(p.amount) \(notation(s,p.arcs))")
455 } else {
456 out.append("AUGMENT \(a[0]) \(a[1]) value=\(balances()[s]) none")
457 }
458 } else if c == "MINCUT" {
459 maxflow(s,t)
460 let (level,_) = bfs(s)
461 let cut = edges.indices.filter {
462 level[edges[$0].from] >= 0 && level[edges[$0].to] < 0
463 }
464 out.append("MINCUT \(a[0]) \(a[1]) value=\(balances()[s]) edges=\(cut.count)")
465 out.append("SOURCE "+names.indices.filter {
466 level[$0] >= 0
467 }.map {
468 names[$0]
469 }.joined(separator: " "))
470 out.append("SINK "+names.indices.filter {
471 level[$0] < 0
472 }.map {
473 names[$0]
474 }.joined(separator: " "))
475 for i in cut {
476 let e = edges[i]
477 out.append("CUT \(i+1) \(names[e.from]) \(names[e.to]) \(e.capacity)")
478 }
479 } else {
480 let (phases,paths) = c == "DINIC" ? dinic(s,t) : (0,c == "SCALING" ? scaling(s,t) : maxflow(s,t))
481 out.append("\(c) \(a[0]) \(a[1]) value=\(balances()[s]) start=\(initial)"+(c == "DINIC" ? " phases=\(phases)" : "")+" paths=\(paths.count)")
482 for (i,p) in paths.enumerated() {
483 out.append("AUG \(i+1)"+(c == "DINIC" ? " phase=\(p.phase)" : c == "SCALING" ? " delta=\(p.delta)" : "")+" amount=\(p.amount) \(notation(s,p.arcs))")
484 }
485 out += flows()
486 }
487 case "SENSITIVITY":
488 let (s,t) = try endpoints(a[0],a[1])
489 let base = copy()
490 base.zero()
491 base.maxflow(s,t)
492 let value = base.balances()[s]
493 var up = 0
494 var down = 0
495 var details = [String]()
496 for i in edges.indices {
497 let work = copy()
498 work.zero()
499 work.edges[i].capacity += 1
500 work.maxflow(s,t)
501 let effectiveUp = work.balances()[s] > value
502 var effectiveDown = false
503 if edges[i].capacity > 0 {
504 let reduced = copy()
505 reduced.zero()
506 reduced.edges[i].capacity -= 1
507 reduced.maxflow(s,t)
508 effectiveDown = reduced.balances()[s] < value
509 }
510 if effectiveUp {
511 up += 1
512 }
513 if effectiveDown {
514 down += 1
515 }
516 if effectiveUp || effectiveDown {
517 let e = edges[i]
518 details.append("CRITICAL \(i+1) \(names[e.from]) \(names[e.to]) up=\(effectiveUp ? "yes" : "no") down=\(effectiveDown ? "yes" : "no")")
519 }
520 }
521 out.append("SENSITIVITY \(a[0]) \(a[1]) value=\(value) up=\(up) down=\(down)")
522 out += details
523 case "MINCOST":
524 let (s,t) = try endpoints(a[0],a[1])
525 if negativeCycle(false) {
526 try fail("NEGATIVE CYCLE")
527 }
528 let target = a[2] == "MAX" ? nil : Int(a[2])
529 let work = copy()
530 work.zero()
531 let (amount,cost,paths) = work.mincost(s,t,target)
532 if let target = target, amount < target {
533 out.append("MINCOST \(a[0]) \(a[1]) k=\(a[2]) infeasible max=\(amount)")
534 } else {
535 for i in edges.indices {
536 edges[i].flow = work.edges[i].flow
537 }
538 owner = .flow(s,t)
539 out.append("MINCOST \(a[0]) \(a[1]) k=\(a[2]) flow=\(amount) cost=\(cost)")
540 for (i,p) in paths.enumerated() {
541 out.append("AUG \(i+1) amount=\(p.amount) cost=\(p.cost) \(notation(s,p.arcs))")
542 }
543 out += flows()
544 }
545 case "DISJOINT":
546 let (s,t) = try endpoints(a[0],a[1])
547 let original = edges.count
548 let work = copy()
549 work.zero()
550 var exits = Array(names.indices)
551 if a[2] == "VERTEX" {
552 for v in names.indices where v != s && v != t {
553 exits[v] = work.names.count
554 work.names.append(names[v])
555 work.demands.append(0)
556 }
557 }
558 for i in work.edges.indices {
559 work.edges[i].from = exits[work.edges[i].from]
560 work.edges[i].capacity = work.edges[i].capacity > 0 ? 1 : 0
561 work.edges[i].cost = 0
562 }
563 if a[2] == "VERTEX" {
564 for v in names.indices where v != s && v != t {
565 work.edges.append(Edge(from: v,to: exits[v],capacity: 1))
566 }
567 }
568 work.maxflow(s,t)
569 let paths = work.decompose().filter {
570 !$0.cycle
571 }
572 out.append("DISJOINT \(a[0]) \(a[1]) \(a[2]) count=\(paths.count)")
573 for (i,p) in paths.enumerated() {
574 out.append("PATH \(i+1) \(notation(s,p.edges.filter { $0 < original }.map { Arc(edge: $0,forward: true) }))")
575 }
576 case "DECOMPOSE":
577 let parts = decompose()
578 out.append("DECOMPOSE paths=\(parts.filter { !$0.cycle }.count) cycles=\(parts.filter { $0.cycle }.count)")
579 var counts = [0,0]
580 for p in parts {
581 let kind = p.cycle ? 1 : 0
582 counts[kind] += 1
583 out.append("\(p.cycle ? "CYCLE" : "PATH") \(counts[kind]) amount=\(p.amount) \(notation(p.start,p.edges.map { Arc(edge: $0,forward: true) }))")
584 }
585 case "VERIFY":
586 var violations = [String]()
587 for (i,e) in edges.enumerated() where e.flow < e.lower {
588 violations.append("LOW \(i+1) flow=\(e.flow) low=\(e.lower)")
589 }
590 let balance = balances()
591 for i in names.indices {
592 var need: Int? = demands[i]
593 if case let .flow(s,t) = owner {
594 need = i == s || i == t ? nil : 0
595 }
596 if let need = need, -balance[i] != need {
597 violations.append("BALANCE \(names[i]) net=\(-balance[i]) need=\(need)")
598 }
599 }
600 if violations.isEmpty {
601 out.append("VERIFY ok")
602 } else {
603 out.append("VERIFY violations=\(violations.count)")
604 out += violations
605 }
606 case "CIRCULATION":
607 if a[0] == "MINCOST" && negativeCycle(true) {
608 try fail("NEGATIVE CYCLE")
609 }
610 let sum = demands.reduce(0,+)
611 if sum != 0 {
612 return ["CIRCULATION infeasible unbalanced=\(sum)"]
613 }
614 let n = names.count
615 let work = copy()
616 work.names += ["super source","super sink"]
617 work.demands += [0,0]
618 var required = demands
619 for i in work.edges.indices {
620 let e = work.edges[i]
621 required[e.to] -= e.lower
622 required[e.from] += e.lower
623 work.edges[i].capacity -= e.lower
624 work.edges[i].flow = 0
625 }
626 for v in required.indices where required[v] < 0 {
627 work.edges.append(Edge(from: n,to: v,capacity: -required[v]))
628 }
629 let sinkStart = work.edges.count
630 for v in required.indices where required[v] > 0 {
631 work.edges.append(Edge(from: v,to: n+1,capacity: required[v]))
632 }
633 if a[0] == "ANY" {
634 work.maxflow(n,n+1)
635 } else {
636 _ = work.mincost(n,n+1)
637 }
638 let total = required.filter {
639 $0 > 0
640 }.reduce(0,+)
641 let achieved = work.edges.dropFirst(sinkStart).reduce(0) {
642 $0+$1.flow
643 }
644 if achieved < total {
645 let (level,_) = work.bfs(n)
646 out.append("CIRCULATION infeasible required=\(total) achieved=\(achieved)")
647 out.append("CUT "+names.indices.filter {
648 level[$0] >= 0
649 }.map {
650 names[$0]
651 }.joined(separator: " "))
652 } else {
653 for i in edges.indices {
654 edges[i].flow = edges[i].lower+work.edges[i].flow
655 }
656 owner = .circulation
657 out.append("CIRCULATION feasible required=\(total) cost=\(cost())")
658 out += flows()
659 }
660 case "MATCH":
661 out += try matching(a)
662 case "ASSIGN":
663 out += try assignment(a)
664 default:
665 break
666 }
667 return out
668 }

671func matching(_ a: [String]) throws -> [String] {
672 let left = a[0].split(separator: ",").map(String.init)
673 let right = a[1].split(separator: ",").map(String.init)
674 var seen = Set<String>()
675 for n in left+right {
676 if !seen.insert(n).inserted {
677 try fail("DUPLICATE VERTEX \(n)")
678 }
679 }
680 var pairs = [(Int,Int)]()
681 if a[2] != "-" {
682 for p in a[2].split(separator: ",").map(String.init) {
683 let words = p.split(separator: "-").map(String.init)
684 guard let u = left.firstIndex(of: words[0]) else {
685 try fail("UNKNOWN VERTEX \(words[0])")
686 }
687 guard let v = right.firstIndex(of: words[1]) else {
688 try fail("UNKNOWN VERTEX \(words[1])")
689 }
690 if pairs.contains(where: {
691 $0.0 == u && $0.1 == v
692 }) {
693 try fail("DUPLICATE PAIR \(p)")
694 }
695 pairs.append((u,v))
696 }
697 }
698 var lp = Array(repeating: -1,count: left.count)
699 var rp = Array(repeating: -1,count: right.count)
700 func attempt(_ u: Int, _ examined: inout [Bool]) -> Bool {
701 for (x,b) in pairs where x == u && !examined[b] {
702 examined[b] = true
703 if rp[b] < 0 || attempt(rp[b],&examined) {
704 lp[u] = b
705 rp[b] = u
706 return true
707 }
708 }
709 return false
710 }
711 for u in left.indices {
712 var examined = Array(repeating: false,count: right.count)
713 _ = attempt(u,&examined)
714 }
715 var zl = lp.map {
716 $0 < 0
717 }
718 var zr = Array(repeating: false,count: right.count)
719 while true {
720 var changed = false
721 for (u,v) in pairs where zl[u] && lp[u] != v && !zr[v] {
722 zr[v] = true
723 changed = true
724 }
725 for v in right.indices where zr[v] && rp[v] >= 0 && !zl[rp[v]] {

780 if best == nil || (a[0] == "MAX" ? sum > best! : sum < best!) {
