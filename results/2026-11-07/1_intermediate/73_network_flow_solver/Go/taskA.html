45func (g *graph) arcs(vertex int) []int {
46 result := []int{}
47 for i, e := range g.edges {
48 if e.from == vertex {
49 result = append(result, i+1)
50 }
51 }
52 for i, e := range g.edges {
53 if e.to == vertex {
54 result = append(result, -i-1)
55 }
56 }
57 return result
58}

60func (g *graph) endpoints(arc int) (int, int) {
61 if arc > 0 {
62 e := g.edges[arc-1]
63 return e.from, e.to
64 }
65 e := g.edges[-arc-1]
66 return e.to, e.from
67}

69func (g *graph) residual(arc int) int {
70 if arc > 0 {
71 e := g.edges[arc-1]
72 return e.capacity - e.flow
73 }
74 return g.edges[-arc-1].flow
75}

77func (g *graph) arcCost(arc int) int {
78 if arc > 0 {
79 return g.edges[arc-1].cost
80 }
81 return -g.edges[-arc-1].cost
82}

84func (g *graph) push(path []int, amount int) {
85 for _, arc := range path {
86 if arc > 0 {
87 g.edges[arc-1].flow += amount
88 } else {
89 g.edges[-arc-1].flow -= amount
90 }
91 }
92}

94func (g *graph) bottleneck(path []int) int {
95 result := g.residual(path[0])
96 for _, arc := range path[1:] {
97 result = min(result, g.residual(arc))
98 }
99 return result
100}

102func (g *graph) bfs(source, threshold int) ([]int, []int) {
103 levels, parents := make([]int, len(g.names)), make([]int, len(g.names))
104 for i := range levels {
105 levels[i] = -1
106 }
107 levels[source] = 0
108 queue := []int{source}
109 for at := 0; at < len(queue); at++ {
110 vertex := queue[at]
111 for _, arc := range g.arcs(vertex) {
112 _, target := g.endpoints(arc)
113 if g.residual(arc) >= threshold && levels[target] < 0 {
114 levels[target], parents[target] = levels[vertex]+1, arc
115 queue = append(queue, target)
116 }
117 }
118 }
119 return levels, parents
120}

122func (g *graph) path(source, target int, parents []int) []int {
123 result := []int{}
124 for target != source {
125 arc := parents[target]
126 result = append(result, arc)
127 target, _ = g.endpoints(arc)
128 }
129 for a, b := 0, len(result)-1; a < b; a, b = a+1, b-1 {
130 result[a], result[b] = result[b], result[a]
131 }
132 return result
133}

135func (g *graph) augment(source, target, threshold int) *augmentation {
136 levels, parents := g.bfs(source, threshold)
137 if levels[target] < 0 {
138 return nil
139 }
140 path := g.path(source, target, parents)
141 amount := g.bottleneck(path)
142 g.push(path, amount)
143 return &augmentation{path: path, amount: amount, stage: threshold}
144}

146func (g *graph) maximum(source, target int, mode string) ([]augmentation, int) {
147 result, phases := []augmentation{}, 0
148 if mode == "DINIC" {
149 adjacency := make([][]int, len(g.names))
150 for vertex := range adjacency {
151 adjacency[vertex] = g.arcs(vertex)
152 }
153 for {
154 levels, _ := g.bfs(source, 1)
155 if levels[target] < 0 {
156 break
157 }
158 phases++
159 pointers := make([]int, len(g.names))
160 path, vertex := []int{}, source
161 for {
162 if vertex == target {
163 amount := g.bottleneck(path)
164 g.push(path, amount)
165 result = append(result, augmentation{append([]int{}, path...), amount, phases, 0})
166 path, vertex = []int{}, source
167 } else if pointers[vertex] == len(adjacency[vertex]) {
168 if vertex == source {
169 break
170 }
171 arc := path[len(path)-1]
172 path = path[:len(path)-1]
173 vertex, _ = g.endpoints(arc)
174 pointers[vertex]++
175 } else {
176 arc := adjacency[vertex][pointers[vertex]]
177 _, next := g.endpoints(arc)
178 if g.residual(arc) > 0 && levels[next] == levels[vertex]+1 {
179 path = append(path, arc)
180 vertex = next
181 } else {
182 pointers[vertex]++
183 }
184 }
185 }
186 }
187 return result, phases
188 }
189 threshold := 1
190 if mode == "SCALING" {
191 largest := 0
192 for _, e := range g.edges {
193 largest = max(largest, e.capacity)
194 }
195 for threshold*2 <= largest {
196 threshold *= 2
197 }
198 }
199 for threshold > 0 {
200 for {
201 pushed := g.augment(source, target, threshold)
202 if pushed == nil {
203 break
204 }
205 result = append(result, *pushed)
206 }
207 threshold /= 2
208 }
209 return result, phases
210}

212func (g *graph) negativeCycle(useLower bool) bool {
213 distance := make([]int, len(g.names))
214 for pass := 0; pass < len(g.names); pass++ {
215 changed := false
216 for _, e := range g.edges {
217 capacity := e.capacity
218 if useLower {
219 capacity -= e.lower
220 }
221 if capacity > 0 && distance[e.to] > distance[e.from]+e.cost {
222 distance[e.to] = distance[e.from] + e.cost
223 changed = true
224 }
225 }
226 if !changed {
227 return false
228 }
229 }
230 return len(g.names) > 0
231}

233func (g *graph) shortest(source int) ([]int, []bool, []int) {
234 distance, defined, parents := make([]int, len(g.names)), make([]bool, len(g.names)), make([]int, len(g.names))
235 defined[source] = true
236 arcs := []int{}
237 for i := range g.edges {
238 if g.residual(i+1) > 0 {
239 arcs = append(arcs, i+1)
240 }
241 }
242 for i := range g.edges {
243 if g.residual(-i-1) > 0 {
244 arcs = append(arcs, -i-1)
245 }
246 }
247 for {
248 changed := false
249 for _, arc := range arcs {
250 a, b := g.endpoints(arc)
251 candidate := distance[a] + g.arcCost(arc)
252 if defined[a] && (!defined[b] || candidate < distance[b]) {
253 distance[b], defined[b], parents[b], changed = candidate, true, arc, true
254 }
255 }
256 if !changed {
257 return distance, defined, parents
258 }
259 }
260}

262func (g *graph) minimumCost(source, target, limit int) ([]augmentation, int, int) {
263 paths, total, cost := []augmentation{}, 0, 0
264 for limit < 0 || total < limit {
265 distance, defined, parents := g.shortest(source)
266 if !defined[target] {
267 break
268 }
269 path := g.path(source, target, parents)
270 amount := g.bottleneck(path)
271 if limit >= 0 {
272 amount = min(amount, limit-total)
273 }
274 g.push(path, amount)
275 total, cost = total+amount, cost+amount*distance[target]
276 paths = append(paths, augmentation{path: path, amount: amount, unit: distance[target]})
277 }
278 return paths, total, cost
279}

303func (g *graph) notation(path []int, originalEdges int) string {
304 source, _ := g.endpoints(path[0])
305 parts := []string{g.names[source]}
306 for _, arc := range path {
307 magnitude, sign := arc, "+"
308 if arc < 0 {
309 magnitude, sign = -arc, "-"
310 }
311 if originalEdges > 0 && magnitude > originalEdges {
312 continue
313 }
314 _, target := g.endpoints(arc)
315 parts = append(parts, sign+strconv.Itoa(magnitude), g.names[target])
316 }
317 return strings.Join(parts, " ")
318}

320func (g *graph) decompose() []piece {
321 residual := make([]int, len(g.edges))
322 balance := make([]int, len(g.names))
323 for i, e := range g.edges {
324 residual[i] = e.flow
325 balance[e.from] += e.flow
326 balance[e.to] -= e.flow
327 }
328 result := []piece{}
329 for {
330 first := -1
331 for i, flow := range residual {
332 if flow > 0 {
333 first = i
334 break
335 }
336 }
337 if first < 0 {
338 return result
339 }
340 source := g.edges[first].from
341 for vertex, amount := range balance {
342 if amount > 0 {
343 source = vertex
344 break
345 }
346 }
347 vertices, path := []int{source}, []int{}
348 for {
349 vertex, chosen := vertices[len(vertices)-1], -1
350 for i, e := range g.edges {
351 if e.from == vertex && residual[i] > 0 {
352 chosen = i
353 break
354 }
355 }
356 path = append(path, chosen+1)
357 target := g.edges[chosen].to
358 repeated := -1
359 for i, held := range vertices {
360 if held == target {
361 repeated = i
362 break
363 }
364 }
365 if repeated >= 0 {
366 cycle := append([]int{}, path[repeated:]...)
367 amount := residual[cycle[0]-1]
368 for _, id := range cycle {
369 amount = min(amount, residual[id-1])
370 }
371 for _, id := range cycle {
372 residual[id-1] -= amount
373 }
374 result = append(result, piece{cycle, amount, true})
375 break
376 }
377 vertices = append(vertices, target)
378 if balance[target] < 0 {
379 amount := min(balance[source], -balance[target])
380 for _, id := range path {
381 amount = min(amount, residual[id-1])
382 }
383 for _, id := range path {
384 residual[id-1] -= amount
385 }
386 balance[source] -= amount
387 balance[target] += amount
388 result = append(result, piece{path, amount, false})
389 break
390 }
391 }
392 }
393}

443func (n *network) matching(w []string) error {
444 left, right := strings.Split(w[1], ","), strings.Split(w[2], ",")
445 ids, leftID, rightID := map[string]bool{}, map[string]int{}, map[string]int{}
446 for i, name := range left {
447 if ids[name] {
448 return fmt.Errorf("DUPLICATE VERTEX %s", name)
449 }
450 ids[name], leftID[name] = true, i
451 }
452 for i, name := range right {
453 if ids[name] {
454 return fmt.Errorf("DUPLICATE VERTEX %s", name)
455 }
456 ids[name], rightID[name] = true, i
457 }
458 pairs := [][2]int{}
459 seen := map[string]bool{}
460 if w[3] != "-" {
461 for _, text := range strings.Split(w[3], ",") {
462 fields := strings.Split(text, "-")
463 a, ok := leftID[fields[0]]
464 if !ok {
465 return fmt.Errorf("UNKNOWN VERTEX %s", fields[0])
466 }
467 b, ok := rightID[fields[1]]
468 if !ok {
469 return fmt.Errorf("UNKNOWN VERTEX %s", fields[1])
470 }
471 if seen[text] {
472 return fmt.Errorf("DUPLICATE PAIR %s", text)
473 }
474 seen[text] = true
475 pairs = append(pairs, [2]int{a, b})
476 }
477 }
478 matesA, matesB := make([]int, len(left)), make([]int, len(right))
479 for i := range matesA {
480 matesA[i] = -1
481 }
482 for i := range matesB {
483 matesB[i] = -1
484 }
485 var attempt func(int, []bool) bool
486 attempt = func(a int, examined []bool) bool {
487 for _, pair := range pairs {
488 b := pair[1]
489 if pair[0] == a && !examined[b] {
490 examined[b] = true
491 if matesB[b] < 0 || attempt(matesB[b], examined) {
492 matesA[a], matesB[b] = b, a
493 return true
494 }
495 }
496 }
497 return false
498 }
499 for a := range left {
500 attempt(a, make([]bool, len(right)))
501 }
502 markedA, markedB := make([]bool, len(left)), make([]bool, len(right))
503 queue := []int{}
504 freeA, freeB, cover, hallA, hallB := []string{}, []string{}, []string{}, []string{}, []string{}
505 size := 0
506 for a := range left {
507 if matesA[a] < 0 {
508 markedA[a] = true
509 queue = append(queue, a)
510 freeA = append(freeA, left[a])
511 } else {
512 size++
513 }
514 }
515 for at := 0; at < len(queue); at++ {
516 a := queue[at]
517 for _, pair := range pairs {
518 b := pair[1]
519 if pair[0] == a && b != matesA[a] && !markedB[b] {
520 markedB[b] = true
521 if matesB[b] >= 0 && !markedA[matesB[b]] {
522 markedA[matesB[b]] = true
523 queue = append(queue, matesB[b])
524 }
525 }
526 }
527 }
528 for a, name := range left {
529 if markedA[a] {
530 hallA = append(hallA, name)
531 } else {
532 cover = append(cover, name)
533 }
534 }
535 for b, name := range right {
536 if matesB[b] < 0 {
537 freeB = append(freeB, name)
538 }
539 if markedB[b] {
540 cover = append(cover, name)
541 hallB = append(hallB, name)
542 }
543 }
544 fmt.Fprintf(n.out, "MATCH size=%d left=%d right=%d\n", size, len(left), len(right))
545 for a, b := range matesA {
546 if b >= 0 {
547 fmt.Fprintf(n.out, "PAIR %s %s\n", left[a], right[b])
548 }
549 }
550 joined := func(values []string) string {
551 if len(values) == 0 {
552 return "-"
553 }
554 return strings.Join(values, ",")
555 }
556 fmt.Fprintf(n.out, "FREE %s %s\n", joined(freeA), joined(freeB))
557 fmt.Fprintf(n.out, "COVER size=%d", len(cover))
558 for _, name := range cover {
559 fmt.Fprint(n.out, " "+name)
560 }
561 fmt.Fprintln(n.out)
562 if len(freeA) > 0 {
563 fmt.Fprintf(n.out, "HALL %s %s\n", joined(hallA), joined(hallB))
564 }
565 return nil
566}

568func (n *network) assignment(w []string) error {
569 size := number(w[2])
570 if size < 1 || size > 8 {
571 return fmt.Errorf("BAD SIZE")
572 }
573 if len(w)-3 != size*size {
574 return fmt.Errorf("BAD LENGTH")
575 }
576 mapping, best := make([]int, size), []int(nil)
577 used, optimum, count := make([]bool, size), 0, 0
578 var search func(int, int)
579 search = func(row, total int) {
580 if row == size {
581 better := best == nil || w[1] == "MIN" && total < optimum || w[1] == "MAX" && total > optimum
582 if better {
583 optimum, count, best = total, 0, append([]int{}, mapping...)
584 }
585 if total == optimum {
586 count++
587 }
588 return
589 }
590 for column := 0; column < size; column++ {
591 text := w[3+row*size+column]
592 if !used[column] && text != "x" {
593 mapping[row], used[column] = column, true
594 search(row+1, total+number(text))
595 used[column] = false
596 }
597 }
598 }
599 search(0, 0)
600 if best == nil {
601 fmt.Fprintf(n.out, "ASSIGN %s n=%d infeasible\n", w[1], size)
602 return nil
603 }
604 fmt.Fprintf(n.out, "ASSIGN %s n=%d total=%d optimal=%d\n", w[1], size, optimum, count)
605 for row, column := range best {
606 fmt.Fprintf(n.out, "MATE %d %d %s\n", row+1, column+1, w[3+row*size+column])
607 }
608 return nil
609}

611func (n *network) command(w []string) error {
612 cmd := w[0]
613 switch cmd {
614 case "NODE":
615 seen := map[string]bool{}
616 for _, name := range n.names {
617 seen[name] = true
618 }
619 for _, name := range w[1:] {
620 if seen[name] {
621 return fmt.Errorf("DUPLICATE NODE %s", name)
622 }
623 seen[name] = true
624 }
625 if len(n.names)+len(w)-1 > 20 {
626 return fmt.Errorf("TOO MANY NODES")
627 }
628 n.names = append(n.names, w[1:]...)
629 n.demands = append(n.demands, make([]int, len(w)-1)...)
630 fmt.Fprintf(n.out, "NODE added=%d total=%d\n", len(w)-1, len(n.names))
631 case "EDGE":
632 a, err := n.vertex(w[1])
633 if err != nil {
634 return err
635 }
636 b, err := n.vertex(w[2])
637 if err != nil {
638 return err
639 }
640 if a == b {
641 return fmt.Errorf("SELF LOOP")
642 }
643 if len(n.edges) == 60 {
644 return fmt.Errorf("TOO MANY EDGES")
645 }
646 cost := 0
647 if len(w) == 5 {
648 cost = number(w[4])
649 }
650 n.edges = append(n.edges, edge{from: a, to: b, capacity: number(w[3]), cost: cost})
651 fmt.Fprintf(n.out, "EDGE %d %s %s cap=%s cost=%d\n", len(n.edges), w[1], w[2], w[3], cost)
652 case "LOWER", "CAP", "COST":
653 id := number(w[1])
654 if id < 1 || id > len(n.edges) {
655 return fmt.Errorf("UNKNOWN EDGE %s", w[1])
656 }
657 e, value := &n.edges[id-1], number(w[2])
658 if cmd == "LOWER" {
659 if value > e.capacity {
660 return fmt.Errorf("BAD BOUND")
661 }
662 e.lower = value
663 fmt.Fprintf(n.out, "LOWER %d low=%d cap=%d\n", id, value, e.capacity)
664 } else if cmd == "CAP" {
665 if value < e.lower {
666 return fmt.Errorf("BAD BOUND")
667 }
668 if value < e.flow {
669 return fmt.Errorf("BELOW FLOW")
670 }
671 e.capacity = value
672 fmt.Fprintf(n.out, "CAP %d low=%d cap=%d\n", id, e.lower, value)
673 } else {
674 e.cost = value
675 fmt.Fprintf(n.out, "COST %d cost=%d\n", id, value)
676 }
677 case "DEMAND":
678 at, err := n.vertex(w[1])
679 if err != nil {
680 return err
681 }
682 n.demands[at] = number(w[2])
683 total := 0
684 for _, demand := range n.demands {
685 total += demand
686 }
687 fmt.Fprintf(n.out, "DEMAND %s %s total=%d\n", w[1], w[2], total)
688 case "RESET":
689 changed := 0
690 for i := range n.edges {
691 if n.edges[i].flow != 0 {
692 changed++
693 }
694 n.edges[i].flow = 0
695 }
696 n.owner = "none"
697 fmt.Fprintf(n.out, "RESET edges=%d\n", changed)
698 case "CLEAR":
699 fmt.Fprintf(n.out, "CLEAR nodes=%d edges=%d\n", len(n.names), len(n.edges))
700 n.graph, n.demands, n.owner = graph{}, nil, "none"
701 case "SHOW":
702 fmt.Fprintf(n.out, "NETWORK nodes=%d edges=%d\n", len(n.names), len(n.edges))
703 if n.owner == "none" {
704 fmt.Fprintln(n.out, "STATE none")
705 } else if n.owner == "circulation" {
706 fmt.Fprintf(n.out, "STATE circulation cost=%d\n", n.totalCost())
707 } else {
708 fmt.Fprintf(n.out, "STATE flow %s %s value=%d cost=%d\n", n.names[n.source], n.names[n.target], n.value(n.source), n.totalCost())
709 }
710 incoming, outgoing := n.balances()
711 for i, name := range n.names {
712 fmt.Fprintf(n.out, "NODE %s demand=%d out=%d in=%d\n", name, n.demands[i], outgoing[i], incoming[i])
713 }
714 for i, e := range n.edges {
715 fmt.Fprintf(n.out, "EDGE %d %s %s cap=%d cost=%d low=%d flow=%d\n", i+1, n.names[e.from], n.names[e.to], e.capacity, e.cost, e.lower, e.flow)
716 }
717 case "AUGMENT", "MAXFLOW", "DINIC", "SCALING", "MINCUT", "SENSITIVITY", "MINCOST", "DISJOINT":
718 source, target, err := n.endpointsCheck(w[1], w[2])
719 if err != nil {
720 return err
721 }
722 if cmd == "SENSITIVITY" {
723 base := n.copy()
724 base.maximum(source, target, "MAXFLOW")
725 value := base.value(source)
726 up, down := make([]bool, len(n.edges)), make([]bool, len(n.edges))
727 ups, downs := 0, 0
728 for i, e := range n.edges {
729 changed := n.copy()
730 changed.edges[i].capacity++
731 changed.maximum(source, target, "MAXFLOW")
732 up[i] = changed.value(source) > value
733 if up[i] {
734 ups++
735 }
736 if e.capacity > 0 {
737 changed = n.copy()
738 changed.edges[i].capacity--
739 changed.maximum(source, target, "MAXFLOW")
740 down[i] = changed.value(source) < value
741 if down[i] {
742 downs++
743 }
744 }
745 }
746 fmt.Fprintf(n.out, "SENSITIVITY %s %s value=%d up=%d down=%d\n", w[1], w[2], value, ups, downs)
747 yes := func(value bool) string {
748 if value {
749 return "yes"
750 }
751 return "no"
752 }
753 for i, e := range n.edges {
754 if up[i] || down[i] {
755 fmt.Fprintf(n.out, "CRITICAL %d %s %s up=%s down=%s\n", i+1, n.names[e.from], n.names[e.to], yes(up[i]), yes(down[i]))
756 }
757 }
758 } else if cmd == "MINCOST" {
759 if n.negativeCycle(false) {
760 return fmt.Errorf("NEGATIVE CYCLE")
761 }
762 limit := -1
763 if w[3] != "MAX" {
764 limit = number(w[3])
765 }
766 working := n.copy()
767 paths, value, cost := working.minimumCost(source, target, limit)
768 if limit >= 0 && value < limit {
769 fmt.Fprintf(n.out, "MINCOST %s %s k=%s infeasible max=%d\n", w[1], w[2], w[3], value)
770 return nil
771 }
772 for i := range n.edges {
773 n.edges[i].flow = working.edges[i].flow
774 }
775 n.owner, n.source, n.target = "flow", source, target
776 fmt.Fprintf(n.out, "MINCOST %s %s k=%s flow=%d cost=%d\n", w[1], w[2], w[3], value, cost)
777 for i, pushed := range paths {
778 fmt.Fprintf(n.out, "AUG %d amount=%d cost=%d %s\n", i+1, pushed.amount, pushed.unit, working.notation(pushed.path, 0))
779 }
780 n.flowList()
781 } else if cmd == "DISJOINT" {
782 working := &graph{}
783 entrances, exits := make([]int, len(n.names)), make([]int, len(n.names))
784 for i, name := range n.names {
785 entrances[i] = len(working.names)
786 working.names = append(working.names, name)
787 exits[i] = entrances[i]
788 if w[3] == "VERTEX" && i != source && i != target {
789 exits[i] = len(working.names)
790 working.names = append(working.names, name)
791 }
792 }
793 for _, e := range n.edges {
794 capacity := 0
795 if e.capacity > 0 {
796 capacity = 1
797 }
798 working.edges = append(working.edges, edge{from: exits[e.from], to: entrances[e.to], capacity: capacity})
799 }
800 if w[3] == "VERTEX" {
801 for i := range n.names {
802 if i != source && i != target {
803 working.edges = append(working.edges, edge{from: entrances[i], to: exits[i], capacity: 1})
804 }
805 }
806 }
807 working.maximum(entrances[source], entrances[target], "MAXFLOW")
808 paths := []piece{}
809 for _, part := range working.decompose() {
810 if !part.cycle {
811 paths = append(paths, part)
812 }
813 }
814 fmt.Fprintf(n.out, "DISJOINT %s %s %s count=%d\n", w[1], w[2], w[3], len(paths))
815 for i, part := range paths {
816 fmt.Fprintf(n.out, "PATH %d %s\n", i+1, working.notation(part.path, len(n.edges)))
817 }
818 } else {
819 start := n.start(source, target)
820 if cmd == "AUGMENT" {
821 pushed := n.augment(source, target, 1)
822 if pushed == nil {
823 fmt.Fprintf(n.out, "AUGMENT %s %s value=%d none\n", w[1], w[2], n.value(source))
824 } else {
825 fmt.Fprintf(n.out, "AUGMENT %s %s value=%d amount=%d %s\n", w[1], w[2], n.value(source), pushed.amount, n.notation(pushed.path, 0))
826 }
827 } else {
828 paths, phases := n.maximum(source, target, cmd)
829 if cmd == "MINCUT" {
830 reachable, _ := n.bfs(source, 1)
831 a, b, cut := []string{}, []string{}, []int{}
832 for i, name := range n.names {
833 if reachable[i] >= 0 {
834 a = append(a, name)
835 } else {
836 b = append(b, name)
837 }
838 }
839 for i, e := range n.edges {
840 if reachable[e.from] >= 0 && reachable[e.to] < 0 {
841 cut = append(cut, i)
842 }
843 }
844 fmt.Fprintf(n.out, "MINCUT %s %s value=%d edges=%d\nSOURCE %s\nSINK %s\n", w[1], w[2], n.value(source), len(cut), strings.Join(a, " "), strings.Join(b, " "))
845 for _, i := range cut {
846 e := n.edges[i]
847 fmt.Fprintf(n.out, "CUT %d %s %s %d\n", i+1, n.names[e.from], n.names[e.to], e.capacity)
848 }
849 } else {
850 fmt.Fprintf(n.out, "%s %s %s value=%d start=%d", cmd, w[1], w[2], n.value(source), start)
851 if cmd == "DINIC" {
852 fmt.Fprintf(n.out, " phases=%d", phases)
853 }
854 fmt.Fprintf(n.out, " paths=%d\n", len(paths))
855 for i, pushed := range paths {
856 fmt.Fprintf(n.out, "AUG %d", i+1)
857 if cmd == "DINIC" {
858 fmt.Fprintf(n.out, " phase=%d", pushed.stage)
859 }
860 if cmd == "SCALING" {
861 fmt.Fprintf(n.out, " delta=%d", pushed.stage)
862 }
863 fmt.Fprintf(n.out, " amount=%d %s\n", pushed.amount, n.notation(pushed.path, 0))
864 }
865 n.flowList()
866 }
867 }
868 }
869 case "DECOMPOSE":
870 parts := n.decompose()
871 paths, cycles := 0, 0
872 for _, part := range parts {
873 if part.cycle {
874 cycles++
875 } else {
876 paths++
877 }
878 }
879 fmt.Fprintf(n.out, "DECOMPOSE paths=%d cycles=%d\n", paths, cycles)
880 paths, cycles = 0, 0
881 for _, part := range parts {
882 kind, id := "PATH", paths+1
883 if part.cycle {
884 kind, id, cycles = "CYCLE", cycles+1, cycles+1
885 } else {
886 paths++
887 }
888 fmt.Fprintf(n.out, "%s %d amount=%d %s\n", kind, id, part.amount, n.notation(part.path, 0))
889 }
890 case "VERIFY":
891 violations := []string{}
892 for i, e := range n.edges {
893 if e.flow < e.lower {
894 violations = append(violations, fmt.Sprintf("LOW %d flow=%d low=%d", i+1, e.flow, e.lower))
895 }
896 }
897 incoming, outgoing := n.balances()
898 for i, name := range n.names {
899 if n.owner == "flow" && (i == n.source || i == n.target) {
900 continue
901 }
902 need := n.demands[i]
903 if n.owner == "flow" {
904 need = 0
905 }
906 if incoming[i]-outgoing[i] != need {
907 violations = append(violations, fmt.Sprintf("BALANCE %s net=%d need=%d", name, incoming[i]-outgoing[i], need))
908 }
909 }
910 if len(violations) == 0 {
911 fmt.Fprintln(n.out, "VERIFY ok")
912 } else {
913 fmt.Fprintf(n.out, "VERIFY violations=%d\n", len(violations))
914 for _, text := range violations {
915 fmt.Fprintln(n.out, text)
916 }
917 }
918 case "CIRCULATION":
919 if w[1] == "MINCOST" && n.negativeCycle(true) {
920 return fmt.Errorf("NEGATIVE CYCLE")
921 }
922 total := 0
923 for _, demand := range n.demands {
924 total += demand
925 }
926 if total != 0 {
927 fmt.Fprintf(n.out, "CIRCULATION infeasible unbalanced=%d\n", total)
928 return nil
929 }
930 working, residual := n.copy(), append([]int{}, n.demands...)
931 source, target := len(n.names), len(n.names)+1
932 working.names = append(working.names, "SS", "TT")
933 for i, e := range n.edges {
934 working.edges[i].capacity -= e.lower
935 residual[e.to] -= e.lower
936 residual[e.from] += e.lower
937 }
938 for i, value := range residual {
939 if value < 0 {
940 working.edges = append(working.edges, edge{from: source, to: i, capacity: -value})
941 }
942 }
943 required, sinkEdges := 0, []int{}
944 for i, value := range residual {
945 if value > 0 {
946 sinkEdges = append(sinkEdges, len(working.edges))
947 working.edges = append(working.edges, edge{from: i, to: target, capacity: value})
948 required += value
949 }
950 }
951 if w[1] == "ANY" {
952 working.maximum(source, target, "MAXFLOW")
953 } else {
954 working.minimumCost(source, target, -1)
955 }
956 achieved := 0
957 for _, i := range sinkEdges {
958 achieved += working.edges[i].flow
959 }
960 if achieved < required {
961 fmt.Fprintf(n.out, "CIRCULATION infeasible required=%d achieved=%d\nCUT", required, achieved)
962 reached, _ := working.bfs(source, 1)
963 for i, name := range n.names {
964 if reached[i] >= 0 {
965 fmt.Fprint(n.out, " "+name)
966 }
967 }
968 fmt.Fprintln(n.out)
969 } else {
970 for i := range n.edges {
971 n.edges[i].flow = n.edges[i].lower + working.edges[i].flow
972 }
973 n.owner = "circulation"
974 fmt.Fprintf(n.out, "CIRCULATION feasible required=%d cost=%d\n", required, n.totalCost())
975 n.flowList()
976 }
977 case "MATCH":
978 return n.matching(w)
979 case "ASSIGN":
980 return n.assignment(w)
981 }
982 return nil
983}
