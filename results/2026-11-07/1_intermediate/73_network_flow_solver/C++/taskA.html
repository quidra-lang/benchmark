51class Network {
52public:
53 std::vector<std::string> names;
54 std::vector<int> demands;
55 std::vector<Edge> edges;
56
57 int addNode(const std::string& name) {
58 names.push_back(name);
59 demands.push_back(0);
60 return static_cast<int>(names.size()) - 1;
61 }
62
63 int from(Arc arc) const {
64 return arc.forward ? edges[arc.edge].from : edges[arc.edge].to;
65 }
66
67 int to(Arc arc) const {
68 return arc.forward ? edges[arc.edge].to : edges[arc.edge].from;
69 }
70
71 int residual(Arc arc) const {
72 const Edge& edge = edges[arc.edge];
73 return arc.forward ? edge.capacity - edge.flow : edge.flow;
74 }
75
76 int cost(Arc arc) const {
77 return arc.forward ? edges[arc.edge].cost : -edges[arc.edge].cost;
78 }
79
80 Path arcList(int vertex) const {
81 Path arcs;
82 for (std::size_t i = 0; i < edges.size(); ++i) {
83 if (edges[i].from == vertex) arcs.push_back({static_cast<int>(i), true});
84 }
85 for (std::size_t i = 0; i < edges.size(); ++i) {
86 if (edges[i].to == vertex) arcs.push_back({static_cast<int>(i), false});
87 }
88 return arcs;
89 }
90
91 Path globalArcs() const {
92 Path arcs;
93 for (bool forward : {true, false}) {
94 for (std::size_t i = 0; i < edges.size(); ++i) arcs.push_back({static_cast<int>(i), forward});
95 }
96 return arcs;
97 }
98
99 void zero() {
100 for (Edge& edge : edges) edge.flow = 0;
101 }
102
103 std::vector<int> netOut() const {
104 std::vector<int> result(names.size(), 0);
105 for (const Edge& edge : edges) {
106 result[edge.from] += edge.flow;
107 result[edge.to] -= edge.flow;
108 }
109 return result;
110 }
111
112 int totalCost() const {
113 int result = 0;
114 for (const Edge& edge : edges) result += edge.flow * edge.cost;
115 return result;
116 }
117
118 Search breadthFirst(int source, int threshold = 1) const {
119 Search search{std::vector<int>(names.size(), -1), std::vector<std::optional<Arc>>(names.size())};
120 std::queue<int> queue;
121 search.levels[source] = 0;
122 queue.push(source);
123 while (!queue.empty()) {
124 int vertex = queue.front();
125 queue.pop();
126 for (Arc arc : arcList(vertex)) {
127 int target = to(arc);
128 if (residual(arc) >= threshold && search.levels[target] < 0) {
129 search.levels[target] = search.levels[vertex] + 1;
130 search.parents[target] = arc;
131 queue.push(target);
132 }
133 }
134 }
135 return search;
136 }
137
138 Path trace(const std::vector<std::optional<Arc>>& parents, int source, int target) const {
139 Path path;
140 while (target != source) {
141 Arc arc = *parents[target];
142 path.push_back(arc);
143 target = from(arc);
144 }
145 std::reverse(path.begin(), path.end());
146 return path;
147 }
148
149 int bottleneck(const Path& path) const {
150 int amount = std::numeric_limits<int>::max();
151 for (Arc arc : path) amount = std::min(amount, residual(arc));
152 return amount;
153 }
154
155 void push(const Path& path, int amount) {
156 for (Arc arc : path) edges[arc.edge].flow += arc.forward ? amount : -amount;
157 }
158
159 std::optional<Augmentation> augment(int source, int target, int threshold = 1) {
160 Search search = breadthFirst(source, threshold);
161 if (search.levels[target] < 0) return std::nullopt;
162 Path path = trace(search.parents, source, target);
163 int amount = bottleneck(path);
164 push(path, amount);
165 return Augmentation{path, amount, 0, threshold};
166 }
167
168 std::vector<Augmentation> maximumFlow(int source, int target, int threshold = 1) {
169 std::vector<Augmentation> result;
170 while (true) {
171 std::optional<Augmentation> next = augment(source, target, threshold);
172 if (!next) return result;
173 result.push_back(*next);
174 }
175 }
176
177 std::pair<int, std::vector<Augmentation>> dinic(int source, int target) {
178 int phases = 0;
179 std::vector<Augmentation> result;
180 std::vector<Path> arcs;
181 for (std::size_t v = 0; v < names.size(); ++v) arcs.push_back(arcList(static_cast<int>(v)));
182 while (true) {
183 Search search = breadthFirst(source);
184 if (search.levels[target] < 0) return {phases, result};
185 ++phases;
186 std::vector<std::size_t> pointers(names.size(), 0);
187 Path path;
188 int vertex = source;
189 while (true) {
190 if (vertex == target) {
191 int amount = bottleneck(path);
192 push(path, amount);
193 result.push_back({path, amount, phases});
194 path.clear();
195 vertex = source;
196 } else if (pointers[vertex] == arcs[vertex].size()) {
197 if (vertex == source) break;
198 Arc last = path.back();
199 path.pop_back();
200 vertex = from(last);
201 ++pointers[vertex];
202 } else {
203 Arc arc = arcs[vertex][pointers[vertex]];
204 int destination = to(arc);
205 if (residual(arc) > 0 && search.levels[destination] == search.levels[vertex] + 1) {
206 path.push_back(arc);
207 vertex = destination;
208 } else ++pointers[vertex];
209 }
210 }
211 }
212 }
213
214 bool negativeCycle(bool subtractLower = false) const {
215 std::vector<int> distance(names.size(), 0);
216 for (std::size_t pass = 0; pass < names.size(); ++pass) {
217 bool updated = false;
218 for (const Edge& edge : edges) {
219 if (edge.capacity - (subtractLower ? edge.lower : 0) < 1) continue;
220 if (distance[edge.to] > distance[edge.from] + edge.cost) {
221 distance[edge.to] = distance[edge.from] + edge.cost;
222 updated = true;
223 }
224 }
225 if (!updated) return false;
226 if (pass + 1 == names.size()) return true;
227 }
228 return false;
229 }

231 std::optional<Augmentation> cheapest(int source, int target, std::optional<int> limit) {
232 Path usable;
233 for (Arc arc : globalArcs()) {
234 if (residual(arc) > 0) usable.push_back(arc);
235 }
236 std::vector<std::optional<int>> distances(names.size());
237 std::vector<std::optional<Arc>> parents(names.size());
238 distances[source] = 0;
239 while (true) {
240 bool updated = false;
241 for (Arc arc : usable) {
242 int start = from(arc);
243 int end = to(arc);
244 if (!distances[start]) continue;
245 int candidate = *distances[start] + cost(arc);
246 if (!distances[end] || candidate < *distances[end]) {
247 distances[end] = candidate;
248 parents[end] = arc;
249 updated = true;
250 }
251 }
252 if (!updated) break;
253 }
254 if (!distances[target]) return std::nullopt;
255 Path path = trace(parents, source, target);
256 int amount = bottleneck(path);
257 if (limit) amount = std::min(amount, *limit);
258 push(path, amount);
259 return Augmentation{path, amount, 0, 1, *distances[target]};
260 }
261
262 std::vector<Augmentation> minimumCost(int source, int target, std::optional<int> limit) {
263 int pushed = 0;
264 std::vector<Augmentation> result;
265 while (!limit || pushed < *limit) {
266 std::optional<Augmentation> next = cheapest(source, target, limit ? std::optional<int>(*limit - pushed) : std::nullopt);
267 if (!next) break;
268 result.push_back(*next);
269 pushed += next->amount;
270 }
271 return result;
272 }

274 std::string pathText(const Path& path, int originalEdges = -1, const Network* original = nullptr) const {
275 std::ostringstream text;
276 text << names[from(path[0])];
277 for (Arc arc : path) {
278 if (originalEdges >= 0 && arc.edge >= originalEdges) continue;
279 text << ' ' << (arc.forward ? '+' : '-') << arc.edge + 1 << ' ';
280 if (original) text << original->names[original->edges[arc.edge].to];
281 else text << names[to(arc)];
282 }
283 return text.str();
284 }

286 std::vector<Component> decompose() const {
287 std::vector<int> remaining;
288 for (const Edge& edge : edges) remaining.push_back(edge.flow);
289 std::vector<int> balance = netOut();
290 std::vector<Component> result;
291 while (true) {
292 int firstEdge = -1;
293 for (std::size_t i = 0; i < edges.size(); ++i) {
294 if (remaining[i] > 0) {
295 firstEdge = static_cast<int>(i);
296 break;
297 }
298 }
299 if (firstEdge < 0) return result;
300 int start = edges[firstEdge].from;
301 for (std::size_t v = 0; v < balance.size(); ++v) {
302 if (balance[v] > 0) {
303 start = static_cast<int>(v);
304 break;
305 }
306 }
307 std::vector<int> vertices = {start};
308 Path path;
309 while (true) {
310 int next = 0;
311 while (edges[next].from != vertices.back() || remaining[next] == 0) ++next;
312 path.push_back({next, true});
313 int destination = edges[next].to;
314 auto repeated = std::find(vertices.begin(), vertices.end(), destination);
315 if (repeated != vertices.end()) {
316 std::size_t offset = static_cast<std::size_t>(repeated - vertices.begin());
317 Path cycle(path.begin() + static_cast<std::ptrdiff_t>(offset), path.end());
318 int amount = std::numeric_limits<int>::max();
319 for (Arc arc : cycle) amount = std::min(amount, remaining[arc.edge]);
320 for (Arc arc : cycle) remaining[arc.edge] -= amount;
321 result.push_back({true, amount, cycle});
322 break;
323 }
324 vertices.push_back(destination);
325 if (balance[destination] < 0) {
326 int amount = std::min(balance[start], -balance[destination]);
327 for (Arc arc : path) amount = std::min(amount, remaining[arc.edge]);
328 for (Arc arc : path) remaining[arc.edge] -= amount;
329 balance[start] -= amount;
330 balance[destination] += amount;
331 result.push_back({false, amount, path});
332 break;
333 }
334 }
335 }
336 }

367 auto found = std::find(network.names.begin(), network.names.end(), name);
368 if (found == network.names.end()) fail("UNKNOWN NODE " + name);
369 return static_cast<int>(found - network.names.begin());

373 int index = std::stoi(number) - 1;
374 if (index < 0 || index >= static_cast<int>(network.edges.size())) fail("UNKNOWN EDGE " + number);
375 return index;

379 int start = node(words[1]);

381 if (start == end) fail("SAME NODE");

386 if (owner != Owner::Flow || start != source || end != target) network.zero();

424 void maximum(const std::vector<std::string>& words) {
425 auto [start, end] = endpoints(words);
426 int initial = startFlow(start, end);
427 const std::string& command = words[0];
428 if (command == "AUGMENT") {
429 std::optional<Augmentation> pushed = network.augment(start, end);
430 std::cout << "AUGMENT " << words[1] << ' ' << words[2] << " value=" << network.netOut()[start];
431 if (pushed) std::cout << " amount=" << pushed->amount << ' ' << network.pathText(pushed->path) << '\n';
432 else std::cout << " none\n";
433 return;
434 }
435 std::vector<Augmentation> paths;
436 int phases = 0;
437 if (command == "DINIC") {
438 auto result = network.dinic(start, end);
439 phases = result.first;
440 paths = std::move(result.second);
441 } else if (command == "SCALING") {
442 int maximumCapacity = 0;
443 for (const Edge& item : network.edges) maximumCapacity = std::max(maximumCapacity, item.capacity);
444 int threshold = 1;
445 while (threshold <= maximumCapacity / 2) threshold *= 2;
446 for (; threshold >= 1; threshold /= 2) {
447 std::vector<Augmentation> next = network.maximumFlow(start, end, threshold);
448 paths.insert(paths.end(), next.begin(), next.end());
449 }
450 } else paths = network.maximumFlow(start, end);
451 if (command == "MINCUT") {
452 Search reachable = network.breadthFirst(start);
453 std::vector<int> cut;
454 for (std::size_t i = 0; i < network.edges.size(); ++i) {
455 const Edge& item = network.edges[i];
456 if (reachable.levels[item.from] >= 0 && reachable.levels[item.to] < 0) cut.push_back(static_cast<int>(i));
457 }
458 std::cout << "MINCUT " << words[1] << ' ' << words[2] << " value=" << network.netOut()[start]
459 << " edges=" << cut.size() << '\n';
460 for (bool inside : {true, false}) {
461 std::cout << (inside ? "SOURCE" : "SINK");
462 for (std::size_t i = 0; i < network.names.size(); ++i) {
463 if ((reachable.levels[i] >= 0) == inside) std::cout << ' ' << network.names[i];
464 }
465 std::cout << '\n';
466 }
467 for (int index : cut) {
468 const Edge& item = network.edges[index];
469 std::cout << "CUT " << index + 1 << ' ' << network.names[item.from] << ' '
470 << network.names[item.to] << ' ' << item.capacity << '\n';
471 }
472 return;
473 }
474 std::cout << command << ' ' << words[1] << ' ' << words[2] << " value=" << network.netOut()[start]
475 << " start=" << initial;
476 if (command == "DINIC") std::cout << " phases=" << phases;
477 std::cout << " paths=" << paths.size() << '\n';
478 for (std::size_t i = 0; i < paths.size(); ++i) {
479 const Augmentation& item = paths[i];
480 std::cout << "AUG " << i + 1;
481 if (command == "DINIC") std::cout << " phase=" << item.phase;
482 if (command == "SCALING") std::cout << " delta=" << item.threshold;
483 std::cout << " amount=" << item.amount << ' ' << network.pathText(item.path) << '\n';
484 }
485 flowList();
486 }

488 void sensitivity(const std::vector<std::string>& words) const {
489 auto [start, end] = endpoints(words);
490 auto value = [&](int changed, int delta) {
491 Network copy = network;
492 copy.zero();
493 if (changed >= 0) copy.edges[changed].capacity += delta;
494 copy.maximumFlow(start, end);
495 return copy.netOut()[start];
496 };
497 int baseline = value(-1, 0), upCount = 0, downCount = 0;
498 std::vector<std::pair<bool, bool>> flags;
499 for (std::size_t i = 0; i < network.edges.size(); ++i) {
500 bool up = value(static_cast<int>(i), 1) > baseline;
501 bool down = network.edges[i].capacity > 0 && value(static_cast<int>(i), -1) < baseline;
502 flags.emplace_back(up, down);
503 upCount += up;
504 downCount += down;
505 }
506 std::cout << "SENSITIVITY " << words[1] << ' ' << words[2] << " value=" << baseline
507 << " up=" << upCount << " down=" << downCount << '\n';
508 for (std::size_t i = 0; i < flags.size(); ++i) {
509 if (!flags[i].first && !flags[i].second) continue;
510 const Edge& item = network.edges[i];
511 std::cout << "CRITICAL " << i + 1 << ' ' << network.names[item.from] << ' ' << network.names[item.to]
512 << " up=" << (flags[i].first ? "yes" : "no") << " down=" << (flags[i].second ? "yes" : "no") << '\n';
513 }
514 }

516 void minimumCost(const std::vector<std::string>& words) {
517 auto [start, end] = endpoints(words);
518 if (network.negativeCycle()) fail("NEGATIVE CYCLE");
519 std::optional<int> limit;
520 if (words[3] != "MAX") limit = std::stoi(words[3]);
521 Network copy = network;
522 copy.zero();
523 std::vector<Augmentation> paths = copy.minimumCost(start, end, limit);
524 int pushed = copy.netOut()[start];
525 if (limit && pushed < *limit) {
526 std::cout << "MINCOST " << words[1] << ' ' << words[2] << " k=" << words[3]
527 << " infeasible max=" << pushed << '\n';
528 return;
529 }
530 for (std::size_t i = 0; i < network.edges.size(); ++i) network.edges[i].flow = copy.edges[i].flow;
531 owner = Owner::Flow;
532 source = start;
533 target = end;
534 std::cout << "MINCOST " << words[1] << ' ' << words[2] << " k=" << words[3]
535 << " flow=" << pushed << " cost=" << network.totalCost() << '\n';
536 for (std::size_t i = 0; i < paths.size(); ++i) {
537 std::cout << "AUG " << i + 1 << " amount=" << paths[i].amount << " cost=" << paths[i].unitCost
538 << ' ' << copy.pathText(paths[i].path) << '\n';
539 }
540 flowList();
541 }

543 void disjoint(const std::vector<std::string>& words) const {
544 auto [start, end] = endpoints(words);
545 Network copy;
546 std::vector<int> entrance(network.names.size()), exit(network.names.size());
547 for (std::size_t i = 0; i < network.names.size(); ++i) {
548 entrance[i] = copy.addNode(network.names[i]);
549 exit[i] = words[3] == "VERTEX" && static_cast<int>(i) != start && static_cast<int>(i) != end
550 ? copy.addNode(network.names[i]) : entrance[i];
551 }
552 for (const Edge& item : network.edges) copy.edges.push_back({exit[item.from], entrance[item.to], item.capacity > 0 ? 1 : 0});
553 if (words[3] == "VERTEX") {
554 for (std::size_t i = 0; i < entrance.size(); ++i) {
555 if (entrance[i] != exit[i]) copy.edges.push_back({entrance[i], exit[i], 1});
556 }
557 }
558 copy.maximumFlow(entrance[start], entrance[end]);
559 std::vector<Component> paths;
560 for (const Component& item : copy.decompose()) {
561 if (!item.cycle) paths.push_back(item);
562 }
563 std::cout << "DISJOINT " << words[1] << ' ' << words[2] << ' ' << words[3] << " count=" << paths.size() << '\n';
564 for (std::size_t i = 0; i < paths.size(); ++i) {
565 std::cout << "PATH " << i + 1 << ' ' << copy.pathText(paths[i].path, static_cast<int>(network.edges.size()), &network) << '\n';
566 }
567 }

581 void verify() const {
582 std::vector<std::string> violations;
583 for (std::size_t i = 0; i < network.edges.size(); ++i) {
584 const Edge& item = network.edges[i];
585 if (item.flow < item.lower) violations.push_back("LOW " + std::to_string(i + 1) + " flow="
586 + std::to_string(item.flow) + " low=" + std::to_string(item.lower));
587 }
588 std::vector<int> outgoing = network.netOut();
589 for (std::size_t i = 0; i < network.names.size(); ++i) {
590 if (owner == Owner::Flow && (static_cast<int>(i) == source || static_cast<int>(i) == target)) continue;
591 int needed = owner == Owner::Flow ? 0 : network.demands[i];
592 if (-outgoing[i] != needed) violations.push_back("BALANCE " + network.names[i]
593 + " net=" + std::to_string(-outgoing[i]) + " need=" + std::to_string(needed));
594 }
595 if (violations.empty()) std::cout << "VERIFY ok\n";
596 else {
597 std::cout << "VERIFY violations=" << violations.size() << '\n';
598 for (const std::string& violation : violations) std::cout << violation << '\n';
599 }
600 }

602 void circulation(const std::string& mode) {
603 if (mode == "MINCOST" && network.negativeCycle(true)) fail("NEGATIVE CYCLE");
604 int sum = std::accumulate(network.demands.begin(), network.demands.end(), 0);
605 if (sum != 0) {
606 std::cout << "CIRCULATION infeasible unbalanced=" << sum << '\n';
607 return;
608 }
609 Network copy = network;
610 copy.zero();
611 std::vector<int> remainder = network.demands;
612 for (std::size_t i = 0; i < network.edges.size(); ++i) {
613 const Edge& item = network.edges[i];
614 remainder[item.from] += item.lower;
615 remainder[item.to] -= item.lower;
616 copy.edges[i].capacity -= item.lower;
617 }
618 int superSource = copy.addNode("source"), superSink = copy.addNode("sink");
619 for (std::size_t i = 0; i < remainder.size(); ++i) {
620 if (remainder[i] < 0) copy.edges.push_back({superSource, static_cast<int>(i), -remainder[i]});
621 }
622 int positiveStart = static_cast<int>(copy.edges.size()), required = 0;
623 for (std::size_t i = 0; i < remainder.size(); ++i) {
624 if (remainder[i] > 0) {
625 copy.edges.push_back({static_cast<int>(i), superSink, remainder[i]});
626 required += remainder[i];
627 }
628 }
629 if (mode == "ANY") copy.maximumFlow(superSource, superSink);
630 else copy.minimumCost(superSource, superSink, std::nullopt);
631 int achieved = 0;
632 for (std::size_t i = positiveStart; i < copy.edges.size(); ++i) achieved += copy.edges[i].flow;
633 if (achieved < required) {
634 std::cout << "CIRCULATION infeasible required=" << required << " achieved=" << achieved << "\nCUT";
635 Search reachable = copy.breadthFirst(superSource);
636 for (std::size_t i = 0; i < network.names.size(); ++i) {
637 if (reachable.levels[i] >= 0) std::cout << ' ' << network.names[i];
638 }
639 std::cout << '\n';
640 return;
641 }
642 for (std::size_t i = 0; i < network.edges.size(); ++i) network.edges[i].flow = network.edges[i].lower + copy.edges[i].flow;
643 owner = Owner::Circulation;
644 std::cout << "CIRCULATION feasible required=" << required << " cost=" << network.totalCost() << '\n';
645 flowList();
646 }

648 static void match(const std::vector<std::string>& words) {
649 std::vector<std::string> left = split(words[1], ','), right = split(words[2], ',');
650 std::set<std::string> seen;
651 for (const std::vector<std::string>* side : {&left, &right}) {
652 for (const std::string& name : *side) {
653 if (!seen.insert(name).second) fail("DUPLICATE VERTEX " + name);
654 }
655 }
656 std::vector<std::pair<int, int>> pairs;
657 std::set<std::pair<int, int>> registered;
658 if (words[3] != "-") {
659 for (const std::string& item : split(words[3], ',')) {
660 std::vector<std::string> names = split(item, '-');
661 auto a = std::find(left.begin(), left.end(), names[0]);
662 if (a == left.end()) fail("UNKNOWN VERTEX " + names[0]);
663 auto b = std::find(right.begin(), right.end(), names[1]);
664 if (b == right.end()) fail("UNKNOWN VERTEX " + names[1]);
665 std::pair<int, int> pair{static_cast<int>(a - left.begin()), static_cast<int>(b - right.begin())};
666 if (!registered.insert(pair).second) fail("DUPLICATE PAIR " + item);
667 pairs.push_back(pair);
668 }
669 }
670 std::vector<int> leftPartner(left.size(), -1), rightPartner(right.size(), -1);
671 std::vector<bool> examined;
672 std::function<bool(int)> attempt = [&](int a) {
673 for (auto [from, to] : pairs) {
674 if (from != a || examined[to]) continue;
675 examined[to] = true;
676 if (rightPartner[to] < 0 || attempt(rightPartner[to])) {
677 rightPartner[to] = a;
678 leftPartner[a] = to;
679 return true;
680 }
681 }
682 return false;
683 };
684 for (std::size_t a = 0; a < left.size(); ++a) {
685 examined.assign(right.size(), false);
686 attempt(static_cast<int>(a));
687 }
688 std::vector<std::string> freeLeft, freeRight, hallLeft, hallRight, cover;
689 std::vector<bool> reachedLeft(left.size(), false), reachedRight(right.size(), false);
690 std::queue<int> queue;
691 int count = 0;
692 for (std::size_t a = 0; a < left.size(); ++a) {
693 if (leftPartner[a] < 0) {
694 freeLeft.push_back(left[a]);
695 reachedLeft[a] = true;
696 queue.push(static_cast<int>(a));
697 } else ++count;
698 }
699 for (std::size_t b = 0; b < right.size(); ++b) {
700 if (rightPartner[b] < 0) freeRight.push_back(right[b]);
701 }
702 while (!queue.empty()) {
703 int a = queue.front();
704 queue.pop();
705 for (auto [from, to] : pairs) {
706 if (from != a || leftPartner[a] == to || reachedRight[to]) continue;
707 reachedRight[to] = true;
708 int partner = rightPartner[to];
709 if (partner >= 0 && !reachedLeft[partner]) {
710 reachedLeft[partner] = true;
711 queue.push(partner);
712 }
713 }
714 }
715 for (std::size_t a = 0; a < left.size(); ++a) {
716 if (reachedLeft[a]) hallLeft.push_back(left[a]);
717 else cover.push_back(left[a]);
718 }
719 for (std::size_t b = 0; b < right.size(); ++b) {
720 if (reachedRight[b]) {
721 hallRight.push_back(right[b]);
722 cover.push_back(right[b]);
723 }
724 }
725 std::cout << "MATCH size=" << count << " left=" << left.size() << " right=" << right.size() << '\n';
726 for (std::size_t a = 0; a < left.size(); ++a) {
727 if (leftPartner[a] >= 0) std::cout << "PAIR " << left[a] << ' ' << right[leftPartner[a]] << '\n';
728 }
729 std::cout << "FREE " << commaList(freeLeft) << ' ' << commaList(freeRight) << "\nCOVER size=" << cover.size();
730 for (const std::string& name : cover) std::cout << ' ' << name;
731 std::cout << '\n';
732 if (!freeLeft.empty()) std::cout << "HALL " << commaList(hallLeft) << ' ' << commaList(hallRight) << '\n';
733 }

735 static void assign(const std::vector<std::string>& words) {
736 int size = std::stoi(words[2]);
737 if (size < 1 || size > 8) fail("BAD SIZE");
738 if (words.size() - 3 != static_cast<std::size_t>(size * size)) fail("BAD LENGTH");
739 std::vector<std::optional<int>> matrix;
740 for (std::size_t i = 3; i < words.size(); ++i) matrix.push_back(words[i] == "x" ? std::nullopt : std::optional<int>(std::stoi(words[i])));
741 std::vector<int> permutation(size), best;
742 std::iota(permutation.begin(), permutation.end(), 0);
743 int total = 0, optimal = 0;
744 do {
745 bool valid = true;
746 int sum = 0;
747 for (int row = 0; row < size; ++row) {
748 const std::optional<int>& item = matrix[row * size + permutation[row]];
749 if (!item) {
750 valid = false;
751 break;
752 }
753 sum += *item;
754 }
755 if (!valid) continue;
756 if (best.empty() || (words[1] == "MIN" ? sum < total : sum > total)) {
757 total = sum;
758 optimal = 1;
759 best = permutation;
760 } else if (sum == total) ++optimal;
761 } while (std::next_permutation(permutation.begin(), permutation.end()));
762 std::cout << "ASSIGN " << words[1] << " n=" << size;
763 if (best.empty()) std::cout << " infeasible\n";
764 else {
765 std::cout << " total=" << total << " optimal=" << optimal << '\n';
766 for (int row = 0; row < size; ++row) {
767 std::cout << "MATE " << row + 1 << ' ' << best[row] + 1 << ' ' << *matrix[row * size + best[row]] << '\n';
768 }
769 }
770 }

773 void execute(const std::string& line) {
774 std::istringstream input(line);
775 std::vector<std::string> words;
776 for (std::string word; input >> word;) words.push_back(word);
777 const std::string& command = words[0];
778 if (command == "NODE") {
779 std::set<std::string> names(network.names.begin(), network.names.end());
780 for (std::size_t i = 1; i < words.size(); ++i) {
781 if (!names.insert(words[i]).second) fail("DUPLICATE NODE " + words[i]);
782 }
783 if (network.names.size() + words.size() - 1 > 20) fail("TOO MANY NODES");
784 for (std::size_t i = 1; i < words.size(); ++i) network.addNode(words[i]);
785 std::cout << "NODE added=" << words.size() - 1 << " total=" << network.names.size() << '\n';
786 } else if (command == "EDGE") {
787 int from = node(words[1]), to = node(words[2]);
788 if (from == to) fail("SELF LOOP");
789 if (network.edges.size() == 60) fail("TOO MANY EDGES");
790 int capacity = std::stoi(words[3]), cost = words.size() == 5 ? std::stoi(words[4]) : 0;
791 network.edges.push_back({from, to, capacity, cost});
792 std::cout << "EDGE " << network.edges.size() << ' ' << words[1] << ' ' << words[2]
793 << " cap=" << capacity << " cost=" << cost << '\n';
794 } else if (command == "LOWER" || command == "CAP" || command == "COST") {
795 Edge& item = network.edges[edge(words[1])];
796 int value = std::stoi(words[2]);
797 if (command == "LOWER") {
798 if (value > item.capacity) fail("BAD BOUND");
799 item.lower = value;
800 std::cout << "LOWER " << words[1] << " low=" << value << " cap=" << item.capacity << '\n';
801 } else if (command == "CAP") {
802 if (value < item.lower) fail("BAD BOUND");
803 if (value < item.flow) fail("BELOW FLOW");
804 item.capacity = value;
805 std::cout << "CAP " << words[1] << " low=" << item.lower << " cap=" << value << '\n';
806 } else {
807 item.cost = value;
808 std::cout << "COST " << words[1] << " cost=" << value << '\n';
809 }
810 } else if (command == "DEMAND") {
811 int vertex = node(words[1]), demand = std::stoi(words[2]);
812 network.demands[vertex] = demand;
813 std::cout << "DEMAND " << words[1] << ' ' << demand << " total="
814 << std::accumulate(network.demands.begin(), network.demands.end(), 0) << '\n';
815 } else if (command == "RESET") {
816 int count = 0;
817 for (const Edge& item : network.edges) count += item.flow != 0;
818 network.zero();
819 owner = Owner::None;
820 std::cout << "RESET edges=" << count << '\n';
821 } else if (command == "CLEAR") {
822 std::cout << "CLEAR nodes=" << network.names.size() << " edges=" << network.edges.size() << '\n';
823 network = Network{};
824 owner = Owner::None;
825 source = target = -1;
826 } else if (command == "SHOW") show();
827 else if (command == "AUGMENT" || command == "MAXFLOW" || command == "DINIC"
828 || command == "SCALING" || command == "MINCUT") maximum(words);
829 else if (command == "SENSITIVITY") sensitivity(words);
830 else if (command == "MINCOST") minimumCost(words);
831 else if (command == "DISJOINT") disjoint(words);
832 else if (command == "DECOMPOSE") decompose();
833 else if (command == "VERIFY") verify();
834 else if (command == "CIRCULATION") circulation(words[1]);
835 else if (command == "MATCH") match(words);
836 else if (command == "ASSIGN") assign(words);
837 }
