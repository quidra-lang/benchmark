79const Graph = struct {
80 names: std.ArrayList(Text) = .empty,
81 edges: std.ArrayList(Edge) = .empty,
82 fn copy(self: *const Graph, gpa: Allocator) Allocator.Error!Graph {
83 var result: Graph = .{};
84 try result.names.appendSlice(gpa, self.names.items);
85 try result.edges.appendSlice(gpa, self.edges.items);
86 for (result.edges.items) |*edge| edge.flow = 0;
87 return result;
88 }
89 fn arcs(self: *const Graph, gpa: Allocator, vertex: usize) Allocator.Error![]i32 {
90 var result: std.ArrayList(i32) = .empty;
91 for (self.edges.items, 0..) |edge, i| {
92 if (edge.from == vertex) try result.append(gpa, @intCast(i + 1));
93 }
94 for (self.edges.items, 0..) |edge, i| {
95 if (edge.to == vertex) try result.append(gpa, -@as(i32, @intCast(i + 1)));
96 }
97 return result.items;
98 }
99 fn edgeIndex(arc: i32) usize {
100 return @intCast(@abs(arc) - 1);
101 }
102 fn endpoints(self: *const Graph, arc: i32) struct { from: usize, to: usize } {
103 const edge = self.edges.items[edgeIndex(arc)];
104 return if (arc > 0) .{ .from = edge.from, .to = edge.to } else .{ .from = edge.to, .to = edge.from };
105 }
106 fn residual(self: *const Graph, arc: i32) Int {
107 const edge = self.edges.items[edgeIndex(arc)];
108 return if (arc > 0) edge.capacity - edge.flow else edge.flow;
109 }
110 fn arcCost(self: *const Graph, arc: i32) Int {
111 const edge = self.edges.items[edgeIndex(arc)];
112 return if (arc > 0) edge.cost else -edge.cost;
113 }
114 fn push(self: *Graph, path: []const i32, amount: Int) void {
115 for (path) |arc| self.edges.items[edgeIndex(arc)].flow += if (arc > 0) amount else -amount;
116 }
117 fn bottleneck(self: *const Graph, path: []const i32) Int {
118 var result = self.residual(path[0]);
119 for (path[1..]) |arc| result = @min(result, self.residual(arc));
120 return result;
121 }
122 fn bfs(self: *const Graph, gpa: Allocator, source: usize, threshold: Int) Allocator.Error!Search {
123 const levels = try gpa.alloc(?usize, self.names.items.len);
124 const parents = try gpa.alloc(i32, self.names.items.len);
125 @memset(levels, null);
126 levels[source] = 0;
127 var queue: std.ArrayList(usize) = .empty;
128 try queue.append(gpa, source);
129 var at: usize = 0;
130 while (at < queue.items.len) : (at += 1) {
131 const vertex = queue.items[at];
132 for (try self.arcs(gpa, vertex)) |arc| {
133 const target = self.endpoints(arc).to;
134 if (self.residual(arc) >= threshold and levels[target] == null) {
135 levels[target] = levels[vertex].? + 1;
136 parents[target] = arc;
137 try queue.append(gpa, target);
138 }
139 }
140 }
141 return .{ .levels = levels, .parents = parents };
142 }
143 fn buildPath(self: *const Graph, gpa: Allocator, source: usize, target: usize, parents: []const i32) Allocator.Error![]i32 {
144 var result: std.ArrayList(i32) = .empty;
145 var current = target;
146 while (current != source) {
147 const arc = parents[current];
148 try result.append(gpa, arc);
149 current = self.endpoints(arc).from;
150 }
151 std.mem.reverse(i32, result.items);
152 return result.items;
153 }
154 fn augment(self: *Graph, gpa: Allocator, source: usize, target: usize, threshold: Int) Allocator.Error!?Augmentation {
155 const found = try self.bfs(gpa, source, threshold);
156 if (found.levels[target] == null) return null;
157 const route = try self.buildPath(gpa, source, target, found.parents);
158 const amount = self.bottleneck(route);
159 self.push(route, amount);
160 return .{ .path = route, .amount = amount, .stage = @intCast(threshold) };
161 }
162 fn maximum(self: *Graph, gpa: Allocator, source: usize, target: usize, mode: Text) Allocator.Error!Maximum {
163 var result: std.ArrayList(Augmentation) = .empty;
164 var phases: usize = 0;
165 if (equal(mode, "DINIC")) {
166 const adjacency = try gpa.alloc([]i32, self.names.items.len);
167 for (adjacency, 0..) |*list, vertex| list.* = try self.arcs(gpa, vertex);
168 while (true) {
169 const found = try self.bfs(gpa, source, 1);
170 if (found.levels[target] == null) break;
171 phases += 1;
172 const pointers = try gpa.alloc(usize, self.names.items.len);
173 @memset(pointers, 0);
174 var route: std.ArrayList(i32) = .empty;
175 var vertex = source;
176 while (true) {
177 if (vertex == target) {
178 const amount = self.bottleneck(route.items);
179 self.push(route.items, amount);
180 try result.append(gpa, .{ .path = try gpa.dupe(i32, route.items), .amount = amount, .stage = phases });
181 route.clearRetainingCapacity();
182 vertex = source;
183 } else if (pointers[vertex] == adjacency[vertex].len) {
184 if (vertex == source) break;
185 const arc = route.pop().?;
186 vertex = self.endpoints(arc).from;
187 pointers[vertex] += 1;
188 } else {
189 const arc = adjacency[vertex][pointers[vertex]];
190 const next = self.endpoints(arc).to;
191 if (self.residual(arc) > 0 and found.levels[next] != null and found.levels[next].? == found.levels[vertex].? + 1) {
192 try route.append(gpa, arc);
193 vertex = next;
194 } else pointers[vertex] += 1;
195 }
196 }
197 }
198 return .{ .paths = result.items, .phases = phases };
199 }
200 var threshold: Int = 1;
201 if (equal(mode, "SCALING")) {
202 var largest: Int = 0;
203 for (self.edges.items) |edge| largest = @max(largest, edge.capacity);
204 while (threshold * 2 <= largest) threshold *= 2;
205 }
206 while (threshold > 0) : (threshold = @divTrunc(threshold, 2)) {
207 while (try self.augment(gpa, source, target, threshold)) |pushed| try result.append(gpa, pushed);
208 }
209 return .{ .paths = result.items, .phases = phases };
210 }
211 fn negativeCycle(self: *const Graph, gpa: Allocator, use_lower: bool) Allocator.Error!bool {
212 const distance = try gpa.alloc(Int, self.names.items.len);
213 @memset(distance, 0);
214 for (0..self.names.items.len) |_| {
215 var changed = false;
216 for (self.edges.items) |edge| {
217 const capacity = edge.capacity - if (use_lower) edge.lower else 0;
218 if (capacity > 0 and distance[edge.to] > distance[edge.from] + edge.cost) {
219 distance[edge.to] = distance[edge.from] + edge.cost;
220 changed = true;
221 }
222 }
223 if (!changed) return false;
224 }
225 return self.names.items.len > 0;
226 }
227 fn shortest(self: *const Graph, gpa: Allocator, source: usize) Allocator.Error!struct { distance: []?Int, parents: []i32 } {
228 const distance = try gpa.alloc(?Int, self.names.items.len);
229 const parents = try gpa.alloc(i32, self.names.items.len);
230 @memset(distance, null);
231 distance[source] = 0;
232 var usable: std.ArrayList(i32) = .empty;
233 for (0..self.edges.items.len) |i| {
234 const arc: i32 = @intCast(i + 1);
235 if (self.residual(arc) > 0) try usable.append(gpa, arc);
236 }
237 for (0..self.edges.items.len) |i| {
238 const arc = -@as(i32, @intCast(i + 1));
239 if (self.residual(arc) > 0) try usable.append(gpa, arc);
240 }
241 while (true) {
242 var changed = false;
243 for (usable.items) |arc| {
244 const ends = self.endpoints(arc);
245 if (distance[ends.from]) |value| {
246 const candidate = value + self.arcCost(arc);
247 if (distance[ends.to] == null or candidate < distance[ends.to].?) {
248 distance[ends.to] = candidate;
249 parents[ends.to] = arc;
250 changed = true;
251 }
252 }
253 }
254 if (!changed) return .{ .distance = distance, .parents = parents };
255 }
256 }
257 fn minimumCost(self: *Graph, gpa: Allocator, source: usize, target: usize, limit: ?Int) Allocator.Error!Cheapest {
258 var paths: std.ArrayList(Augmentation) = .empty;
259 var total: Int = 0;
260 var cost: Int = 0;
261 while (limit == null or total < limit.?) {
262 const found = try self.shortest(gpa, source);
263 const unit = found.distance[target] orelse break;
264 const route = try self.buildPath(gpa, source, target, found.parents);
265 var amount = self.bottleneck(route);
266 if (limit) |value| amount = @min(amount, value - total);
267 self.push(route, amount);
268 total += amount;
269 cost += amount * unit;
270 try paths.append(gpa, .{ .path = route, .amount = amount, .unit = unit });
271 }
272 return .{ .paths = paths.items, .total = total, .cost = cost };
273 }
274 fn balances(self: *const Graph, gpa: Allocator) Allocator.Error!Balances {
275 const incoming = try gpa.alloc(Int, self.names.items.len);
276 const outgoing = try gpa.alloc(Int, self.names.items.len);
277 @memset(incoming, 0);
278 @memset(outgoing, 0);
279 for (self.edges.items) |edge| {
280 outgoing[edge.from] += edge.flow;
281 incoming[edge.to] += edge.flow;
282 }
283 return .{ .incoming = incoming, .outgoing = outgoing };
284 }
285 fn value(self: *const Graph, source: usize) Int {
286 var result: Int = 0;
287 for (self.edges.items) |edge| {
288 if (edge.from == source) result += edge.flow;
289 if (edge.to == source) result -= edge.flow;
290 }
291 return result;
292 }
293 fn totalCost(self: *const Graph) Int {
294 var result: Int = 0;
295 for (self.edges.items) |edge| result += edge.flow * edge.cost;
296 return result;
297 }
298 fn notation(self: *const Graph, gpa: Allocator, route: []const i32, original_edges: ?usize) Allocator.Error!Text {
299 var out: Builder = .{ .gpa = gpa };
300 try out.add(self.names.items[self.endpoints(route[0]).from]);
301 for (route) |arc| {
302 const magnitude: usize = @intCast(@abs(arc));
303 if (original_edges != null and magnitude > original_edges.?) continue;
304 try out.print(" {s}{d} {s}", .{ if (arc > 0) @as(Text, "+") else "-", magnitude, self.names.items[self.endpoints(arc).to] });
305 }

308 fn decompose(self: *const Graph, gpa: Allocator) Allocator.Error![]Piece {
309 const remaining = try gpa.alloc(Int, self.edges.items.len);
310 const balance = try gpa.alloc(Int, self.names.items.len);
311 @memset(balance, 0);
312 for (self.edges.items, 0..) |edge, i| {
313 remaining[i] = edge.flow;
314 balance[edge.from] += edge.flow;
315 balance[edge.to] -= edge.flow;
316 }
317 var result: std.ArrayList(Piece) = .empty;
318 while (true) {
319 var first: ?usize = null;
320 for (remaining, 0..) |flow, i| {
321 if (flow > 0) {
322 first = i;
323 break;
324 }
325 }
326 var source = self.edges.items[first orelse return result.items].from;
327 for (balance, 0..) |amount, vertex| {
328 if (amount > 0) {
329 source = vertex;
330 break;
331 }
332 }
333 var vertices: std.ArrayList(usize) = .empty;
334 var route: std.ArrayList(i32) = .empty;
335 try vertices.append(gpa, source);
336 while (true) {
337 const vertex = vertices.items[vertices.items.len - 1];
338 var chosen: usize = 0;
339 for (self.edges.items, 0..) |edge, i| {
340 if (edge.from == vertex and remaining[i] > 0) {
341 chosen = i;
342 break;
343 }
344 }
345 try route.append(gpa, @intCast(chosen + 1));
346 const target = self.edges.items[chosen].to;
347 if (std.mem.findScalar(usize, vertices.items, target)) |repeated| {
348 const cycle = try gpa.dupe(i32, route.items[repeated..]);
349 var amount = remaining[edgeIndex(cycle[0])];
350 for (cycle) |arc| amount = @min(amount, remaining[edgeIndex(arc)]);
351 for (cycle) |arc| remaining[edgeIndex(arc)] -= amount;
352 try result.append(gpa, .{ .path = cycle, .amount = amount, .cycle = true });
353 break;
354 }
355 try vertices.append(gpa, target);
356 if (balance[target] < 0) {
357 var amount = @min(balance[source], -balance[target]);
358 for (route.items) |arc| amount = @min(amount, remaining[edgeIndex(arc)]);
359 for (route.items) |arc| remaining[edgeIndex(arc)] -= amount;
360 balance[source] -= amount;
361 balance[target] += amount;
362 try result.append(gpa, .{ .path = route.items, .amount = amount, .cycle = false });
363 break;
364 }
365 }
366 }
367 }

370const MatchPair = struct { left: usize, right: usize };
371const Matching = struct {
372 pairs: []const MatchPair,
373 mates_a: []?usize,
374 mates_b: []?usize,
375 fn attempt(self: *Matching, a: usize, examined: []bool) bool {
376 for (self.pairs) |pair| {
377 const b = pair.right;
378 if (pair.left == a and !examined[b]) {
379 examined[b] = true;
380 if (self.mates_b[b] == null or self.attempt(self.mates_b[b].?, examined)) {
381 self.mates_a[a] = b;
382 self.mates_b[b] = a;
383 return true;
384 }
385 }
386 }
387 return false;
388 }

390const Assignment = struct {
391 gpa: Allocator,
392 words: []const Text,
393 size: usize,
394 maximize: bool,
395 mapping: []usize,
396 used: []bool,
397 best: ?[]usize = null,
398 optimum: Int = 0,
399 count: usize = 0,
400 fn visit(self: *Assignment, row: usize, total: Int) Allocator.Error!void {
401 if (row == self.size) {
402 if (self.best == null or (if (self.maximize) total > self.optimum else total < self.optimum)) {
403 self.optimum = total;
404 self.count = 0;
405 self.best = try self.gpa.dupe(usize, self.mapping);
406 }
407 if (total == self.optimum) self.count += 1;
408 return;
409 }
410 for (0..self.size) |column| {
411 const text = self.words[3 + row * self.size + column];
412 if (!self.used[column] and !equal(text, "x")) {
413 self.mapping[row] = column;
414 self.used[column] = true;
415 try self.visit(row + 1, total + number(text));
416 self.used[column] = false;
417 }
418 }
419 }

444 self.owner = .flow;
445 self.source = source;
446 self.target = target;

452 fn matching(self: *Network, scratch: Allocator, w: []const Text) Error!void {
453 const left = try split(scratch, w[1], ',');
454 const right = try split(scratch, w[2], ',');
455 var used_names: std.StringHashMapUnmanaged(void) = .empty;
456 for (left) |name| {
457 if (used_names.contains(name)) return bad(self.gpa, "DUPLICATE VERTEX {s}", .{name});
458 try used_names.put(scratch, name, {});
459 }
460 for (right) |name| {
461 if (used_names.contains(name)) return bad(self.gpa, "DUPLICATE VERTEX {s}", .{name});
462 try used_names.put(scratch, name, {});
463 }
464 var pairs: std.ArrayList(MatchPair) = .empty;
465 if (!equal(w[3], "-")) {
466 for (try split(scratch, w[3], ',')) |text| {
467 const fields = try split(scratch, text, '-');
468 var a: ?usize = null;
469 var b: ?usize = null;
470 for (left, 0..) |name, i| if (equal(name, fields[0])) {
471 a = i;
472 break;
473 };
474 if (a == null) return bad(self.gpa, "UNKNOWN VERTEX {s}", .{fields[0]});
475 for (right, 0..) |name, i| if (equal(name, fields[1])) {
476 b = i;
477 break;
478 };
479 if (b == null) return bad(self.gpa, "UNKNOWN VERTEX {s}", .{fields[1]});
480 for (pairs.items) |pair| {
481 if (pair.left == a.? and pair.right == b.?) return bad(self.gpa, "DUPLICATE PAIR {s}", .{text});
482 }
483 try pairs.append(scratch, .{ .left = a.?, .right = b.? });
484 }
485 }
486 var match: Matching = .{ .pairs = pairs.items, .mates_a = try scratch.alloc(?usize, left.len), .mates_b = try scratch.alloc(?usize, right.len) };
487 @memset(match.mates_a, null);
488 @memset(match.mates_b, null);
489 for (0..left.len) |a| {
490 const examined = try scratch.alloc(bool, right.len);
491 @memset(examined, false);
492 _ = match.attempt(a, examined);
493 }
494 const marked_a = try scratch.alloc(bool, left.len);
495 const marked_b = try scratch.alloc(bool, right.len);
496 @memset(marked_a, false);
497 @memset(marked_b, false);
498 var queue: std.ArrayList(usize) = .empty;
499 var size: usize = 0;
500 for (match.mates_a, 0..) |mate, a| {
501 if (mate == null) {
502 marked_a[a] = true;
503 try queue.append(scratch, a);
504 } else size += 1;
505 }
506 var at: usize = 0;
507 while (at < queue.items.len) : (at += 1) {
508 const a = queue.items[at];
509 for (pairs.items) |pair| {
510 const b = pair.right;
511 if (pair.left == a and (match.mates_a[a] == null or b != match.mates_a[a].?) and !marked_b[b]) {
512 marked_b[b] = true;
513 if (match.mates_b[b]) |mate| {
514 if (!marked_a[mate]) {
515 marked_a[mate] = true;
516 try queue.append(scratch, mate);
517 }
518 }
519 }
520 }
521 }
522 try self.out.line("MATCH size={d} left={d} right={d}", .{ size, left.len, right.len });
523 for (match.mates_a, 0..) |mate, a| {
524 if (mate) |b| try self.out.line("PAIR {s} {s}", .{ left[a], right[b] });
525 }
526 try self.out.add("FREE ");
527 try self.commaNames(left, match.mates_a, null);
528 try self.out.add(" ");
529 try self.commaNames(right, match.mates_b, null);
530 try self.out.add("\n");
531 var cover_count: usize = 0;
532 for (marked_a) |marked| if (!marked) {
533 cover_count += 1;
534 };
535 for (marked_b) |marked| if (marked) {
536 cover_count += 1;
537 };
538 try self.out.print("COVER size={d}", .{cover_count});
539 for (left, 0..) |name, a| if (!marked_a[a]) {
540 try self.out.print(" {s}", .{name});
541 };
542 for (right, 0..) |name, b| if (marked_b[b]) {
543 try self.out.print(" {s}", .{name});
544 };
545 try self.out.add("\n");
546 if (queue.items.len > 0) {
547 try self.out.add("HALL ");
548 try self.commaNames(left, null, marked_a);
549 try self.out.add(" ");
550 try self.commaNames(right, null, marked_b);
551 try self.out.add("\n");
552 }
553 }
554 fn commaNames(self: *Network, names: []const Text, mates: ?[]const ?usize, marked: ?[]const bool) Allocator.Error!void {
555 var count: usize = 0;
556 for (names, 0..) |name, i| {
557 const include = if (mates) |values| values[i] == null else marked.?[i];
558 if (include) {
559 if (count > 0) try self.out.add(",");
560 try self.out.add(name);
561 count += 1;
562 }
563 }
564 if (count == 0) try self.out.add("-");
565 }
566 fn assignment(self: *Network, scratch: Allocator, w: []const Text) Error!void {
567 const size = index(w[2]);
568 if (size < 1 or size > 8) return bad(self.gpa, "BAD SIZE", .{});
569 if (w.len - 3 != size * size) return bad(self.gpa, "BAD LENGTH", .{});
570 var search: Assignment = .{ .gpa = scratch, .words = w, .size = size, .maximize = equal(w[1], "MAX"), .mapping = try scratch.alloc(usize, size), .used = try scratch.alloc(bool, size) };
571 @memset(search.used, false);
572 try search.visit(0, 0);
573 const best = search.best orelse {
574 try self.out.line("ASSIGN {s} n={d} infeasible", .{ w[1], size });
575 return;
576 };
577 try self.out.line("ASSIGN {s} n={d} total={d} optimal={d}", .{ w[1], size, search.optimum, search.count });
578 for (best, 0..) |column, row| try self.out.line("MATE {d} {d} {s}", .{ row + 1, column + 1, w[3 + row * size + column] });
579 }
580 fn command(self: *Network, scratch: Allocator, w: []const Text) Error!void {
581 const cmd = w[0];
582 if (equal(cmd, "NODE")) {
583 var seen: std.StringHashMapUnmanaged(void) = .empty;
584 for (self.graph.names.items) |name| try seen.put(scratch, name, {});
585 for (w[1..]) |name| {
586 if (seen.contains(name)) return bad(self.gpa, "DUPLICATE NODE {s}", .{name});
587 try seen.put(scratch, name, {});
588 }
589 if (self.graph.names.items.len + w.len - 1 > 20) return bad(self.gpa, "TOO MANY NODES", .{});
590 for (w[1..]) |name| {
591 try self.graph.names.append(self.gpa, name);
592 try self.demands.append(self.gpa, 0);
593 }
594 try self.out.line("NODE added={d} total={d}", .{ w.len - 1, self.graph.names.items.len });
595 } else if (equal(cmd, "EDGE")) {
596 const a = try self.vertex(w[1]);
597 const b = try self.vertex(w[2]);
598 if (a == b) return bad(self.gpa, "SELF LOOP", .{});
599 if (self.graph.edges.items.len == 60) return bad(self.gpa, "TOO MANY EDGES", .{});
600 const cost: Int = if (w.len == 5) number(w[4]) else 0;
601 try self.graph.edges.append(self.gpa, .{ .from = a, .to = b, .capacity = number(w[3]), .cost = cost });
602 try self.out.line("EDGE {d} {s} {s} cap={s} cost={d}", .{ self.graph.edges.items.len, w[1], w[2], w[3], cost });
603 } else if (equal(cmd, "LOWER") or equal(cmd, "CAP") or equal(cmd, "COST")) {
604 const id = index(w[1]);
605 if (id < 1 or id > self.graph.edges.items.len) return bad(self.gpa, "UNKNOWN EDGE {s}", .{w[1]});
606 const edge = &self.graph.edges.items[id - 1];
607 const amount = number(w[2]);
608 if (equal(cmd, "LOWER")) {
609 if (amount > edge.capacity) return bad(self.gpa, "BAD BOUND", .{});
610 edge.lower = amount;
611 try self.out.line("LOWER {d} low={d} cap={d}", .{ id, amount, edge.capacity });
612 } else if (equal(cmd, "CAP")) {
613 if (amount < edge.lower) return bad(self.gpa, "BAD BOUND", .{});
614 if (amount < edge.flow) return bad(self.gpa, "BELOW FLOW", .{});
615 edge.capacity = amount;
616 try self.out.line("CAP {d} low={d} cap={d}", .{ id, edge.lower, amount });
617 } else {
618 edge.cost = amount;
619 try self.out.line("COST {d} cost={d}", .{ id, amount });
620 }
621 } else if (equal(cmd, "DEMAND")) {
622 const at = try self.vertex(w[1]);
623 self.demands.items[at] = number(w[2]);
624 var total: Int = 0;
625 for (self.demands.items) |demand| total += demand;
626 try self.out.line("DEMAND {s} {s} total={d}", .{ w[1], w[2], total });
627 } else if (equal(cmd, "RESET")) {
628 var changed: usize = 0;
629 for (self.graph.edges.items) |*edge| {
630 if (edge.flow != 0) changed += 1;
631 edge.flow = 0;
632 }
633 self.owner = .none;
634 try self.out.line("RESET edges={d}", .{changed});
635 } else if (equal(cmd, "CLEAR")) {
636 try self.out.line("CLEAR nodes={d} edges={d}", .{ self.graph.names.items.len, self.graph.edges.items.len });
637 self.graph = .{};
638 self.demands = .empty;
639 self.owner = .none;
640 } else if (equal(cmd, "SHOW")) {
641 const g = &self.graph;
642 try self.out.line("NETWORK nodes={d} edges={d}", .{ g.names.items.len, g.edges.items.len });
643 switch (self.owner) {
644 .none => try self.out.add("STATE none\n"),
645 .circulation => try self.out.line("STATE circulation cost={d}", .{g.totalCost()}),
646 .flow => try self.out.line("STATE flow {s} {s} value={d} cost={d}", .{ g.names.items[self.source], g.names.items[self.target], g.value(self.source), g.totalCost() }),
647 }
648 const balance = try g.balances(scratch);
649 for (g.names.items, 0..) |name, i| try self.out.line("NODE {s} demand={d} out={d} in={d}", .{ name, self.demands.items[i], balance.outgoing[i], balance.incoming[i] });
650 for (g.edges.items, 0..) |edge, i| try self.out.line("EDGE {d} {s} {s} cap={d} cost={d} low={d} flow={d}", .{ i + 1, g.names.items[edge.from], g.names.items[edge.to], edge.capacity, edge.cost, edge.lower, edge.flow });
651 } else if (equal(cmd, "AUGMENT") or equal(cmd, "MAXFLOW") or equal(cmd, "DINIC") or equal(cmd, "SCALING") or
652 equal(cmd, "MINCUT") or equal(cmd, "SENSITIVITY") or equal(cmd, "MINCOST") or equal(cmd, "DISJOINT"))
653 {
654 const ends = try self.endpointsCheck(w[1], w[2]);
655 const source = ends.source;
656 const target = ends.target;
657 const g = &self.graph;
658 if (equal(cmd, "SENSITIVITY")) {
659 var base = try g.copy(scratch);
660 _ = try base.maximum(scratch, source, target, "MAXFLOW");
661 const value = base.value(source);
662 const up = try scratch.alloc(bool, g.edges.items.len);
663 const down = try scratch.alloc(bool, g.edges.items.len);
664 @memset(down, false);
665 var ups: usize = 0;
666 var downs: usize = 0;
667 for (g.edges.items, 0..) |edge, i| {
668 var increased = try g.copy(scratch);
669 increased.edges.items[i].capacity += 1;
670 _ = try increased.maximum(scratch, source, target, "MAXFLOW");
671 up[i] = increased.value(source) > value;
672 if (up[i]) ups += 1;
673 if (edge.capacity > 0) {
674 var decreased = try g.copy(scratch);
675 decreased.edges.items[i].capacity -= 1;
676 _ = try decreased.maximum(scratch, source, target, "MAXFLOW");
677 down[i] = decreased.value(source) < value;
678 if (down[i]) downs += 1;
679 }
680 }
681 try self.out.line("SENSITIVITY {s} {s} value={d} up={d} down={d}", .{ w[1], w[2], value, ups, downs });
682 for (g.edges.items, 0..) |edge, i| {
683 if (up[i] or down[i]) try self.out.line("CRITICAL {d} {s} {s} up={s} down={s}", .{ i + 1, g.names.items[edge.from], g.names.items[edge.to], yes(up[i]), yes(down[i]) });
684 }
685 } else if (equal(cmd, "MINCOST")) {
686 if (try g.negativeCycle(scratch, false)) return bad(self.gpa, "NEGATIVE CYCLE", .{});
687 const limit: ?Int = if (equal(w[3], "MAX")) null else number(w[3]);
688 var working = try g.copy(scratch);
689 const found = try working.minimumCost(scratch, source, target, limit);
690 if (limit != null and found.total < limit.?) {
691 try self.out.line("MINCOST {s} {s} k={s} infeasible max={d}", .{ w[1], w[2], w[3], found.total });
692 return;
693 }
694 for (g.edges.items, working.edges.items) |*edge, computed| edge.flow = computed.flow;
695 self.owner = .flow;
696 self.source = source;
697 self.target = target;
698 try self.out.line("MINCOST {s} {s} k={s} flow={d} cost={d}", .{ w[1], w[2], w[3], found.total, found.cost });
699 for (found.paths, 0..) |pushed, i| try self.out.line("AUG {d} amount={d} cost={d} {s}", .{ i + 1, pushed.amount, pushed.unit, try working.notation(scratch, pushed.path, null) });
700 try self.flowList();
701 } else if (equal(cmd, "DISJOINT")) {
702 var working: Graph = .{};
703 const entrances = try scratch.alloc(usize, g.names.items.len);
704 const exits = try scratch.alloc(usize, g.names.items.len);
705 for (g.names.items, 0..) |name, i| {
706 entrances[i] = working.names.items.len;
707 try working.names.append(scratch, name);
708 exits[i] = entrances[i];
709 if (equal(w[3], "VERTEX") and i != source and i != target) {
710 exits[i] = working.names.items.len;
711 try working.names.append(scratch, name);
712 }
713 }
714 for (g.edges.items) |edge| try working.edges.append(scratch, .{ .from = exits[edge.from], .to = entrances[edge.to], .capacity = if (edge.capacity > 0) 1 else 0 });
715 if (equal(w[3], "VERTEX")) {
716 for (0..g.names.items.len) |i| {
717 if (i != source and i != target) try working.edges.append(scratch, .{ .from = entrances[i], .to = exits[i], .capacity = 1 });
718 }
719 }
720 _ = try working.maximum(scratch, entrances[source], entrances[target], "MAXFLOW");
721 const parts = try working.decompose(scratch);
722 var count: usize = 0;
723 for (parts) |part| if (!part.cycle) {
724 count += 1;
725 };
726 try self.out.line("DISJOINT {s} {s} {s} count={d}", .{ w[1], w[2], w[3], count });
727 count = 0;
728 for (parts) |part| {
729 if (!part.cycle) {
730 count += 1;
731 try self.out.line("PATH {d} {s}", .{ count, try working.notation(scratch, part.path, g.edges.items.len) });
732 }
733 }
734 } else {
735 const initial = self.start(source, target);
736 if (equal(cmd, "AUGMENT")) {
737 if (try g.augment(scratch, source, target, 1)) |pushed| {
738 try self.out.line("AUGMENT {s} {s} value={d} amount={d} {s}", .{ w[1], w[2], g.value(source), pushed.amount, try g.notation(scratch, pushed.path, null) });
739 } else try self.out.line("AUGMENT {s} {s} value={d} none", .{ w[1], w[2], g.value(source) });
740 } else {
741 const found = try g.maximum(scratch, source, target, cmd);
742 if (equal(cmd, "MINCUT")) {
743 const reached = try g.bfs(scratch, source, 1);
744 var count: usize = 0;
745 for (g.edges.items) |edge| {
746 if (reached.levels[edge.from] != null and reached.levels[edge.to] == null) count += 1;
747 }
748 try self.out.line("MINCUT {s} {s} value={d} edges={d}", .{ w[1], w[2], g.value(source), count });
749 try self.out.add("SOURCE");
750 for (g.names.items, 0..) |name, i| if (reached.levels[i] != null) {
751 try self.out.print(" {s}", .{name});
752 };
753 try self.out.add("\nSINK");
754 for (g.names.items, 0..) |name, i| if (reached.levels[i] == null) {
755 try self.out.print(" {s}", .{name});
756 };
757 try self.out.add("\n");
758 for (g.edges.items, 0..) |edge, i| {
759 if (reached.levels[edge.from] != null and reached.levels[edge.to] == null) try self.out.line("CUT {d} {s} {s} {d}", .{ i + 1, g.names.items[edge.from], g.names.items[edge.to], edge.capacity });
760 }
761 } else {
762 try self.out.print("{s} {s} {s} value={d} start={d}", .{ cmd, w[1], w[2], g.value(source), initial });
763 if (equal(cmd, "DINIC")) try self.out.print(" phases={d}", .{found.phases});
764 try self.out.line(" paths={d}", .{found.paths.len});
765 for (found.paths, 0..) |pushed, i| {
766 try self.out.print("AUG {d}", .{i + 1});
767 if (equal(cmd, "DINIC")) try self.out.print(" phase={d}", .{pushed.stage});
768 if (equal(cmd, "SCALING")) try self.out.print(" delta={d}", .{pushed.stage});
769 try self.out.line(" amount={d} {s}", .{ pushed.amount, try g.notation(scratch, pushed.path, null) });
770 }
771 try self.flowList();
772 }
773 }
774 }
775 } else if (equal(cmd, "DECOMPOSE")) {
776 const parts = try self.graph.decompose(scratch);
777 var paths: usize = 0;
778 var cycles: usize = 0;
779 for (parts) |part| {
780 if (part.cycle) cycles += 1 else paths += 1;
781 }
782 try self.out.line("DECOMPOSE paths={d} cycles={d}", .{ paths, cycles });
783 paths = 0;
784 cycles = 0;
785 for (parts) |part| {
786 if (part.cycle) cycles += 1 else paths += 1;
787 try self.out.line("{s} {d} amount={d} {s}", .{ if (part.cycle) @as(Text, "CYCLE") else "PATH", if (part.cycle) cycles else paths, part.amount, try self.graph.notation(scratch, part.path, null) });
788 }
789 } else if (equal(cmd, "VERIFY")) {
790 var violations: std.ArrayList(Text) = .empty;
791 const g = &self.graph;
792 for (g.edges.items, 0..) |edge, i| {
793 if (edge.flow < edge.lower) try violations.append(scratch, try std.fmt.allocPrint(scratch, "LOW {d} flow={d} low={d}", .{ i + 1, edge.flow, edge.lower }));
794 }
795 const balance = try g.balances(scratch);
796 for (g.names.items, 0..) |name, i| {
797 if (self.owner == .flow and (i == self.source or i == self.target)) continue;
798 const need = if (self.owner == .flow) @as(Int, 0) else self.demands.items[i];
799 const net = balance.incoming[i] - balance.outgoing[i];
800 if (net != need) try violations.append(scratch, try std.fmt.allocPrint(scratch, "BALANCE {s} net={d} need={d}", .{ name, net, need }));
801 }
802 if (violations.items.len == 0) try self.out.add("VERIFY ok\n") else {
803 try self.out.line("VERIFY violations={d}", .{violations.items.len});
804 for (violations.items) |text| try self.out.line("{s}", .{text});
805 }
806 } else if (equal(cmd, "CIRCULATION")) {
807 const g = &self.graph;
808 if (equal(w[1], "MINCOST") and try g.negativeCycle(scratch, true)) return bad(self.gpa, "NEGATIVE CYCLE", .{});
809 var total: Int = 0;
810 for (self.demands.items) |demand| total += demand;
811 if (total != 0) {
812 try self.out.line("CIRCULATION infeasible unbalanced={d}", .{total});
813 return;
814 }
815 var working = try g.copy(scratch);
816 const residual = try scratch.dupe(Int, self.demands.items);
817 const source = g.names.items.len;
818 const target = source + 1;
819 try working.names.appendSlice(scratch, &.{ "SS", "TT" });
820 for (g.edges.items, 0..) |edge, i| {
821 working.edges.items[i].capacity -= edge.lower;
822 residual[edge.to] -= edge.lower;
823 residual[edge.from] += edge.lower;
824 }
825 for (residual, 0..) |amount, i| {
826 if (amount < 0) try working.edges.append(scratch, .{ .from = source, .to = i, .capacity = -amount });
827 }
828 var required: Int = 0;
829 var sink_edges: std.ArrayList(usize) = .empty;
830 for (residual, 0..) |amount, i| {
831 if (amount > 0) {
832 try sink_edges.append(scratch, working.edges.items.len);
833 try working.edges.append(scratch, .{ .from = i, .to = target, .capacity = amount });
834 required += amount;
835 }
836 }
837 if (equal(w[1], "ANY")) {
838 _ = try working.maximum(scratch, source, target, "MAXFLOW");
839 } else _ = try working.minimumCost(scratch, source, target, null);
840 var achieved: Int = 0;
841 for (sink_edges.items) |i| achieved += working.edges.items[i].flow;
842 if (achieved < required) {
843 try self.out.line("CIRCULATION infeasible required={d} achieved={d}", .{ required, achieved });
844 try self.out.add("CUT");
845 const reached = try working.bfs(scratch, source, 1);
846 for (g.names.items, 0..) |name, i| if (reached.levels[i] != null) {
847 try self.out.print(" {s}", .{name});
848 };
849 try self.out.add("\n");
850 } else {
851 for (g.edges.items, 0..) |*edge, i| edge.flow = edge.lower + working.edges.items[i].flow;
852 self.owner = .circulation;
853 try self.out.line("CIRCULATION feasible required={d} cost={d}", .{ required, g.totalCost() });
854 try self.flowList();
855 }
856 } else if (equal(cmd, "MATCH")) {
857 try self.matching(scratch, w);
858 } else if (equal(cmd, "ASSIGN")) {
859 try self.assignment(scratch, w);
860 }
861 }
862};

864pub fn main(init: std.process.Init) !void {
865 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
866 defer arena.deinit();
867 const gpa = arena.allocator();
868 var input_buffer: [8192]u8 = undefined;
869 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
870 const text = try input.interface.allocRemaining(gpa, .unlimited);
871 var input_lines = std.mem.splitScalar(u8, text, '\n');
872 const count = index(input_lines.next().?);
873 var out: Builder = .{ .gpa = gpa };
874 var network: Network = .{ .gpa = gpa, .out = &out };
875 for (0..count) |_| {
876 var scratch_arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
877 defer scratch_arena.deinit();
878 const scratch = scratch_arena.allocator();
879 const words = try split(scratch, input_lines.next().?, ' ');
880 network.command(scratch, words) catch |err| switch (err) {
881 error.Invalid => try out.line("ERROR {s}", .{failure}),
882 else => return err,
883 };
884 }
885 var output_buffer: [8192]u8 = undefined;
886 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
887 try output.interface.writeAll(out.bytes.items);
888 try output.interface.flush();
889}
