78 Breadth breadth_first(nat source, threshold := 1)
79 levels := array(count, fill = -1)
80 parents := array(count, fill = 0)
81 levels[source] = 0
82 queue := [source]
83 nat at := 0
84 while at < #queue
85 vertex := queue[at]
86 at += 1
87 for arc := arcs[vertex][]
88 target := arc_end(edge_list, arc)
89 if arc_residual(edge_list, arc) >= threshold and levels[target] < 0
90 levels[target] = levels[vertex] + 1

92 queue = queue.append(target)

185 PathResult cheapest_path(nat source, nat sink)
186 usable := [for arc := global_arcs[] if arc_residual(edge_list, arc) > 0 then arc]
187 known := array(count, fill = false)
188 distance := array(count, fill = 0)
189 parents := array(count, fill = 0)

192 changed := false
193 for arc := usable[]

208 CostResult .minimum_cost(nat source, nat sink, target := -1)
209 amount := 0
210 cost := 0
211 Push[] result := []
212 while target < 0 or amount < target
213 PathResult search := cheapest_path(source, sink)
214 if not search.found
215 break
216 int delta := bottleneck(search.path)
217 if target >= 0 and delta > target - amount
218 delta = target - amount
219 .push_path(search.path, delta)
220 amount += delta
221 cost += delta * search.unit_cost
222 result = result.append(Push(true, delta, search.path, 0, search.unit_cost))
223 return CostResult(amount, cost, result)

253FlowPiece[] decompose(nat count, Edge[] edges)
254 remaining := [for edge := edges[] then edge.flow]
255 balance := [for vertex := (0:<nat(count))[] then flow_value(edges, vertex)]
256 FlowPiece[] pieces := []
257 while true
258 first_edge := #edges
259 for: i < #edges
260 if remaining[i] > 0
261 first_edge = i
262 break
263 if first_edge == #edges
264 return pieces
265 origin := edges[first_edge].u
266 for: vertex < count
267 if balance[vertex] > 0
268 origin = vertex
269 break
270 walk := [origin]
271 nat[] used := []
272 while true
273 vertex := walk[-1]
274 nat selected := 0
275 for: edge := edges[i]
276 if edge.u == vertex and remaining[i] > 0
277 selected = i
278 break
279 target := edges[selected].v
280 used = used.append(selected)
281 repeated := #walk
282 for: item := walk[i]
283 if item == target
284 repeated = i
285 break
286 if repeated < #walk
287 cycle := used[repeated:]
288 amount := remaining[cycle[0]]
289 for i := cycle[]
290 if remaining[i] < amount
291 amount = remaining[i]
292 for i := cycle[]
293 remaining[i] -= amount
294 pieces = pieces.append(FlowPiece(true, target, cycle, amount))
295 break
296 walk = walk.append(target)
297 if balance[target] < 0
298 amount := if balance[origin] < -balance[target] then balance[origin] else -balance[target]
299 for i := used[]
300 if remaining[i] < amount
301 amount = remaining[i]
302 for i := used[]
303 remaining[i] -= amount
304 balance[origin] -= amount
305 balance[target] += amount
306 pieces = pieces.append(FlowPiece(false, origin, used, amount))
307 break
