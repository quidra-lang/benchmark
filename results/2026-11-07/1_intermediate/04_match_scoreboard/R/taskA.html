14checkout_tables <- function() {
15 candidates <- list()
16 doubles <- list()
17 darts <- new.env(hash = TRUE, parent = emptyenv())
18 darts[["25"]] <- c(25L, 0L)
19 darts[["BULL"]] <- c(50L, 1L)
20 darts[["MISS"]] <- c(0L, 0L)
21 multipliers <- c(S = 1L, T = 3L, D = 2L)
22 for (prefix in names(multipliers)) {
23 for (value in 20:1) {
24 code <- paste0(prefix, value)
25 item <- list(code = code, value = value * multipliers[[prefix]])
26 candidates[[length(candidates) + 1L]] <- item
27 darts[[code]] <- c(item$value, as.integer(prefix == "D"))
28 if (prefix == "D") doubles[[length(doubles) + 1L]] <- item
29 }
30 }
31 candidates <- c(candidates, list(list(code = "25", value = 25L), list(code = "BULL", value = 50L)))
32 doubles[[length(doubles) + 1L]] <- list(code = "BULL", value = 50L)
33 tables <- lapply(1:3, function(i) new.env(hash = TRUE, parent = emptyenv()))
34 put <- function(count, total, codes) {
35 key <- as.character(total)
36 if (is.null(tables[[count]][[key]])) tables[[count]][[key]] <- codes
37 }
38 for (last in doubles) put(1L, last$value, last$code)
39 for (first in candidates) {
40 for (last in doubles) put(2L, first$value + last$value, c(first$code, last$code))
41 }
42 for (first in candidates) {
43 for (second in candidates) {
44 for (last in doubles) put(3L, first$value + second$value + last$value, c(first$code, second$code, last$code))
45 }
46 }
47 list(darts = darts, tables = tables)
48}

59create_match <- function(tokens) {
60 m <- new.env(parent = emptyenv())
61 m$id <- tokens[2L]
62 m$sport <- tokens[3L]
63 m$names <- tokens[4L]
64 if (m$sport != "BOWLING") m$names <- tokens[4:5]
65 m$initial <- list()
66 if (m$sport == "VOLLEYBALL") {
67 m$initial <- lapply(tokens[6:7], function(text) as.integer(strsplit(text, ",", fixed = TRUE)[[1L]]))
68 first_option <- 8L
69 } else if (m$sport == "BOWLING") {
70 first_option <- 5L
71 } else {
72 first_option <- 6L
73 }
74 rules <- option_rules[[m$sport]]
75 m$options <- lapply(rules, function(values) values[1L])
76 seen <- character()
77 if (first_option <= length(tokens)) {
78 for (token in tokens[first_option:length(tokens)]) {
79 parts <- regmatches(token, regexec("^([^=]+)=([^=]+)$", token))[[1L]]
80 if (length(parts) != 3L) score_error("invalid option")
81 key <- parts[2L]
82 value <- parts[3L]
83 if (is.null(rules[[key]]) || !(value %in% rules[[key]]) || key %in% seen) score_error("invalid option")
84 seen <- c(seen, key)
85 m$options[[key]] <- value
86 }
87 }
88 first <- if (is.null(m$options$FIRST)) 1L else side(m$options$FIRST)
89 stat_names <- c("points", "served", "servewins", "runmax", "holds", "games", "breaks", "opportunities", "converted",
90 "darts", "scored", "busts", "high", "tons", "maximums", "attempts", "hits")
91 stats <- setNames(lapply(stat_names, function(key) c(0L, 0L)), stat_names)
92 s <- list(p = c(0L, 0L), g = c(0L, 0L), won = c(0L, 0L), server = first, winner = 0L,
93 completed = list(), stats = stats, last = 0L, run = 0L)
94 if (m$sport == "VOLLEYBALL") {
95 s$lineups <- m$initial
96 s$subs <- c(0L, 0L)
97 s$timeouts <- c(0L, 0L)
98 } else if (m$sport == "BOWLING") {
99 s$frames <- rep(list(integer()), 10L)
100 s$marks <- rep(list(character()), 10L)
101 s$frame <- 1L
102 s$standing <- 10L
103 s$fresh <- TRUE
104 } else if (m$sport == "DARTS") {
105 start <- as.integer(m$options$START)
106 s$legs <- c(0L, 0L)
107 s$remaining <- c(start, start)
108 s$turn <- character()
109 s$turnscore <- 0L
110 s$turnstart <- start
111 s$legnumber <- 0L
112 }
113 m$state <- s
114 m$history <- new.env(hash = TRUE, parent = emptyenv())
115 m$history_size <- 0L
116 m

122tennis_mode <- function(m) {
123 s <- m$state
124 final <- s$won[1L] == as.integer(m$options$SETS) %/% 2L && s$won[1L] == s$won[2L]
125 if (final && m$options$FINAL == "SUPER") return("SUPER")
126 if (all(s$g == c(6L, 6L)) && !(final && m$options$FINAL == "ADV")) return("TB")
127 "NORMAL"

129rally_server <- function(m) {
130 s <- m$state
131 if (m$sport == "TENNIS") {
132 if (tennis_mode(m) != "NORMAL" && ((sum(s$p) + 1L) %/% 2L) %% 2L == 1L) return(other(s$server))
133 } else if (m$sport == "TABLETENNIS") {
134 first <- side(m$options$FIRST)
135 if (length(s$completed) %% 2L == 1L) first <- other(first)
136 total <- sum(s$p)
137 flip <- if (total < 20L) (total %/% 2L) %% 2L == 1L else total %% 2L == 1L
138 return(if (flip) other(first) else first)
139 }
140 s$server
141}
142rally_outcome <- function(m, winner) {
143 s <- m$state
144 points <- s$p
145 points[winner] <- points[winner] + 1L
146 loser <- other(winner)
147 game <- FALSE
148 if (m$sport == "TENNIS") {
149 mode <- tennis_mode(m)
150 if (mode == "NORMAL") {
151 game <- points[winner] >= 4L && (m$options$DEUCE == "NOAD" || points[winner] - points[loser] >= 2L)
152 games <- s$g[winner] + 1L
153 unit <- game && games >= 6L && games - s$g[loser] >= 2L
154 } else {
155 final <- s$won[1L] == s$won[2L] && s$won[1L] == as.integer(m$options$SETS) %/% 2L
156 target <- if (mode == "SUPER" || (final && m$options$FINAL == "TB10")) 10L else 7L
157 unit <- points[winner] >= target && points[winner] - points[loser] >= 2L
158 }
159 needed <- as.integer(m$options$SETS) %/% 2L + 1L
160 } else if (m$sport == "TABLETENNIS") {
161 unit <- points[winner] >= 11L && points[winner] - points[loser] >= 2L
162 needed <- as.integer(m$options$GAMES) %/% 2L + 1L
163 } else if (m$sport == "BADMINTON") {
164 unit <- points[winner] == 30L || (points[winner] >= 21L && points[winner] - points[loser] >= 2L)
165 needed <- 2L
166 } else {
167 target <- if (length(s$completed) == 4L) 15L else 25L
168 unit <- points[winner] >= target && points[winner] - points[loser] >= 2L
169 needed <- 3L
170 }
171 c(game, unit, unit && s$won[winner] + 1L == needed)
172}
173point_event <- function(m, winner) {
174 server <- rally_server(m)
175 outcome <- rally_outcome(m, winner)
176 game <- outcome[1L]
177 unit <- outcome[2L]
178 finished <- outcome[3L]
179 mode <- if (m$sport == "TENNIS") tennis_mode(m) else ""
180 opportunity <- mode == "NORMAL" && rally_outcome(m, other(server))[1L]
181 s <- m$state
182 record <- as.integer(c(1L, winner, server, opportunity, game, s$p, s$g, s$won, s$server,
183 length(s$completed), s$last, s$run, s$stats$runmax[winner], 0L, 0L))
184 s$stats$points[winner] <- s$stats$points[winner] + 1L
185 s$stats$served[server] <- s$stats$served[server] + 1L
186 if (winner == server) s$stats$servewins[server] <- s$stats$servewins[server] + 1L
187 s$run <- if (s$last == winner) s$run + 1L else 1L
188 s$last <- winner
189 s$stats$runmax[winner] <- max(s$stats$runmax[winner], s$run)
190 if (opportunity) {
191 s$stats$opportunities[other(server)] <- s$stats$opportunities[other(server)] + 1L
192 if (winner != server) s$stats$converted[winner] <- s$stats$converted[winner] + 1L
193 }
194 s$p[winner] <- s$p[winner] + 1L
195 if (m$sport == "TENNIS") {
196 if (game) {
197 s$stats$games[server] <- s$stats$games[server] + 1L
198 if (winner == server) {
199 s$stats$holds[winner] <- s$stats$holds[winner] + 1L
200 } else {
201 s$stats$breaks[winner] <- s$stats$breaks[winner] + 1L
202 }
203 s$g[winner] <- s$g[winner] + 1L
204 s$server <- other(s$server)
205 } else if (unit) {
206 s$server <- other(s$server)
207 }
208 if (unit) {
209 if (mode == "SUPER") {
210 record_score <- c(s$p, 2L, 0L)
211 } else if (mode == "TB") {
212 s$g[winner] <- s$g[winner] + 1L
213 record_score <- c(s$g, 1L, s$p[other(winner)])
214 } else {
215 record_score <- c(s$g, 0L, 0L)
216 }
217 s$completed[[length(s$completed) + 1L]] <- record_score
218 }
219 } else if (m$sport == "BADMINTON") {
220 s$server <- winner
221 } else if (m$sport == "VOLLEYBALL" && winner != server) {
222 s$server <- winner
223 lineup <- s$lineups[[winner]]
224 s$lineups[[winner]] <- c(lineup[2:6], lineup[1L])
225 record[18L] <- 1L
226 }
227 if (unit) {
228 if (m$sport != "TENNIS") s$completed[[length(s$completed) + 1L]] <- c(s$p, 0L, 0L)
229 s$won[winner] <- s$won[winner] + 1L
230 if (finished) {
231 s$winner <- winner
232 } else {
233 if (m$sport == "VOLLEYBALL") {
234 record[17L] <- 1L
235 record <- c(record, s$lineups[[1L]], s$lineups[[2L]], s$subs, s$timeouts)
236 s$lineups <- m$initial
237 s$subs <- c(0L, 0L)
238 s$timeouts <- c(0L, 0L)
239 first <- side(m$options$FIRST)
240 s$server <- if (length(s$completed) %% 2L == 0L) first else other(first)
241 }
242 s$p <- c(0L, 0L)
243 s$g <- c(0L, 0L)
244 }
245 } else if (game) {
246 s$p <- c(0L, 0L)
247 }
248 m$state <- s
249 remember(m, record)
250}
251roll_event <- function(s, pins) {
252 if (pins > s$standing) score_error("too many pins")
253 frame <- s$frame
254 if (s$fresh && pins == 10L) {
255 mark <- "X"
256 } else if (pins == s$standing) {
257 mark <- "/"
258 } else {
259 mark <- if (pins == 0L) "-" else as.character(pins)
260 }
261 s$frames[[frame]] <- c(s$frames[[frame]], pins)
262 s$marks[[frame]] <- c(s$marks[[frame]], mark)
263 s$standing <- s$standing - pins
264 s$fresh <- FALSE
265 rolls <- s$frames[[frame]]
266 if (frame < 10L) {
267 if ((length(rolls) == 1L && pins == 10L) || length(rolls) == 2L) {
268 s$frame <- s$frame + 1L
269 s$standing <- 10L
270 s$fresh <- TRUE
271 }
272 } else {
273 complete <- length(rolls) == 3L || (length(rolls) == 2L && rolls[1L] != 10L && sum(rolls) < 10L)
274 if (complete) {
275 s$winner <- 1L
276 } else if (s$standing == 0L) {
277 s$standing <- 10L
278 s$fresh <- TRUE
279 }
280 }
281 s
282}
283bowling_scores <- function(s) {
284 scores <- rep(NA_integer_, 10L)
285 for (frame in 1:10) {
286 rolls <- s$frames[[frame]]
287 if (!length(rolls)) next
288 later <- integer()
289 if (frame < 10L) later <- unlist(s$frames[(frame + 1L):10L], use.names = FALSE)
290 if (frame == 10L) {
291 if (s$winner != 0L) scores[frame] <- sum(rolls)
292 } else if (rolls[1L] == 10L) {
293 if (length(later) >= 2L) scores[frame] <- 10L + later[1L] + later[2L]
294 } else if (length(rolls) == 2L) {
295 if (sum(rolls) == 10L) {
296 if (length(later)) scores[frame] <- 10L + later[1L]
297 } else {
298 scores[frame] <- sum(rolls)
299 }
300 }
301 }
302 scores

318dart_event <- function(m, code) {
319 dart <- checkout_data$darts[[code]]
320 if (is.null(dart)) score_error("invalid dart")
321 s <- m$state
322 player <- s$server
323 value <- dart[1L]
324 double <- dart[2L] == 1L
325 before <- s$remaining[player]
326 s$stats$darts[player] <- s$stats$darts[player] + 1L
327 if (before == 50L || (before <= 40L && before %% 2L == 0L)) s$stats$attempts[player] <- s$stats$attempts[player] + 1L
328 remaining <- before - value
329 if (remaining < 0L || remaining == 1L || (remaining == 0L && !double)) {
330 s$stats$busts[player] <- s$stats$busts[player] + 1L
331 s$stats$scored[player] <- s$stats$scored[player] - s$turnscore
332 s$remaining[player] <- s$turnstart
333 m$state <- next_turn(s)
334 return(invisible(NULL))
335 }
336 s$stats$scored[player] <- s$stats$scored[player] + value
337 s$remaining[player] <- remaining
338 s$turnscore <- s$turnscore + value
339 s$turn <- c(s$turn, code)
340 if (remaining == 0L) {
341 s$stats$hits[player] <- s$stats$hits[player] + 1L
342 s <- complete_turn(s, player)
343 s$legs[player] <- s$legs[player] + 1L
344 set_over <- s$legs[player] == as.integer(m$options$LEGS)
345 if (set_over) {
346 s$completed[[length(s$completed) + 1L]] <- c(s$legs, 0L, 0L)
347 s$won[player] <- s$won[player] + 1L
348 }
349 if (set_over && s$won[player] == as.integer(m$options$SETS)) {
350 s$winner <- player
351 m$state <- s
352 return(invisible(NULL))
353 }
354 if (set_over) s$legs <- c(0L, 0L)
355 s$legnumber <- s$legnumber + 1L
356 first <- side(m$options$FIRST)
357 s$server <- if (s$legnumber %% 2L == 0L) first else other(first)
358 start <- as.integer(m$options$START)
359 s$remaining <- c(start, start)
360 s$turn <- character()
361 s$turnscore <- 0L
362 s$turnstart <- start
363 } else if (length(s$turn) == 3L) {
364 s <- next_turn(complete_turn(s, player))
365 }
366 m$state <- s
367}

369 if (m$history_size == 0L) score_error("nothing to undo")
370 key <- as.character(m$history_size)
371 data <- m$history[[key]]

375 m$state <- data

398 } else {
399 s$stats$breaks[winner] <- s$stats$breaks[winner] - 1L
400 }
401 }

403 s$lineups <- list(data[19:24], data[25:30])
404 s$subs <- data[31:32]
405 s$timeouts <- data[33:34]
406 }

408 lineup <- s$lineups[[winner]]
409 s$lineups[[winner]] <- c(lineup[6L], lineup[1:5])

423event <- function(m, command, args) {
424 correct <- (command == "POINT" && m$sport %in% rally_sports) || (command == "ROLL" && m$sport == "BOWLING") ||
425 (command == "THROW" && m$sport == "DARTS") || (command %in% c("SUB", "TIMEOUT") && m$sport == "VOLLEYBALL")
426 if (!correct) score_error("wrong event")
427 if (command == "POINT") {
428 events <- strsplit(args[1L], "", fixed = TRUE)[[1L]]
429 } else if (command %in% c("SUB", "TIMEOUT")) {
430 events <- ""
431 } else {
432 events <- args
433 }
434 for (value in events) {
435 if (m$state$winner != 0L) score_error("match over")
436 if (command == "POINT") {
437 point_event(m, side(value))
438 } else if (command == "ROLL") {
439 old <- m$state
440 m$state <- roll_event(old, as.integer(value))
441 remember(m, old)
442 } else if (command == "THROW") {
443 old <- m$state
444 dart_event(m, value)
445 remember(m, old)
446 } else if (command == "SUB") {
447 player <- side(args[1L])
448 outgoing <- as.integer(args[2L])
449 incoming <- as.integer(args[3L])
450 lineup <- m$state$lineups[[player]]
451 position <- match(outgoing, lineup)
452 if (is.na(position)) score_error("player not on court")
453 if (incoming %in% lineup) score_error("player already on court")
454 if (m$state$subs[player] >= 6L) score_error("no substitutions left")
455 remember(m, c(2L, player, position, outgoing))
456 m$state$lineups[[player]][position] <- incoming
457 m$state$subs[player] <- m$state$subs[player] + 1L
458 } else {
459 player <- side(args[1L])
460 if (m$state$timeouts[player] >= 2L) score_error("no timeouts left")
461 remember(m, c(3L, player))
462 m$state$timeouts[player] <- m$state$timeouts[player] + 1L
463 }
464 }
465}

474situation <- function(m, player) {
475 outcome <- rally_outcome(m, player)
476 if (outcome[3L]) return("match point")
477 if (outcome[2L]) return(if (m$sport %in% c("TENNIS", "VOLLEYBALL")) "set point" else "game point")
478 if (outcome[1L]) return(if (player == rally_server(m)) "game point" else "break point")
479 ""

481score_line <- function(m) {
482 s <- m$state
483 if (m$sport == "BOWLING") {
484 parts <- paste0(m$id, ": ", m$names[1L])
485 scores <- bowling_scores(s)
486 cumulative <- 0L
487 for (frame in 1:10) {
488 if (!length(s$frames[[frame]])) break
489 text <- paste(s$marks[[frame]], collapse = "")
490 if (!is.na(scores[frame])) {
491 cumulative <- cumulative + scores[frame]
492 text <- paste0(text, "(", cumulative, ")")
493 }
494 parts <- c(parts, text)
495 }
496 return(paste(parts, collapse = " "))
497 }
498 names <- m$names
499 if (s$winner == 0L) {
500 player <- if (m$sport %in% rally_sports) rally_server(m) else s$server
501 names[player] <- paste0(names[player], "*")
502 }
503 parts <- paste0(m$id, ": ", names[1L], " vs ", names[2L])
504 if (m$sport == "DARTS") {
505 parts <- c(parts, "sets", pair_text(s$won), "legs", pair_text(s$legs), "score", pair_text(s$remaining))
506 if (s$winner == 0L) {
507 if (length(s$turn)) parts <- c(parts, "turn", s$turn)
508 answer <- checkout(s$remaining[s$server], 3L - length(s$turn))
509 if (length(answer)) parts <- c(parts, "checkout", answer)
510 }
511 } else {
512 parts <- c(parts, vapply(s$completed, completed_text, character(1L)))
513 if (s$winner == 0L) {
514 if (m$sport == "TENNIS") {
515 mode <- tennis_mode(m)
516 if (mode == "SUPER") {
517 parts <- c(parts, paste0("[", pair_text(s$p), "]"))
518 } else {
519 parts <- c(parts, pair_text(s$g))
520 if (mode == "TB") {
521 parts <- c(parts, pair_text(s$p))
522 } else if (m$options$DEUCE == "AD" && min(s$p) >= 3L) {
523 if (s$p[1L] == s$p[2L]) {
524 label <- "40-40"
525 } else if (s$p[1L] > s$p[2L]) {
526 label <- "AD-40"
527 } else {
528 label <- "40-AD"
529 }
530 parts <- c(parts, label)
531 } else {
532 labels <- c("0", "15", "30", "40")
533 parts <- c(parts, pair_text(labels[s$p + 1L]))
534 }
535 }
536 } else {
537 parts <- c(parts, pair_text(s$p))
538 if (m$sport == "VOLLEYBALL") parts <- c(parts, "server", s$lineups[[s$server]][1L])
539 }
540 for (player in 1:2) {
541 label <- situation(m, player)
542 if (label != "") parts <- c(parts, paste0("(", label, " ", m$names[player], ")"))
543 }
544 }
545 }
546 paste(parts, collapse = " ")
547}
548stats_lines <- function(m) {
549 s <- m$state
550 stats <- s$stats
551 if (m$sport == "BOWLING") {
552 marks <- unlist(s$marks, use.names = FALSE)
553 open <- sum(vapply(s$frames, function(rolls) length(rolls) >= 2L && rolls[1L] != 10L && sum(rolls[1:2]) < 10L, logical(1L)))
554 future <- s
555 while (future$winner == 0L) future <- roll_event(future, future$standing)
556 maximum <- sum(bowling_scores(future))
557 return(paste0(m$id, " ", m$names[1L], ": strikes ", sum(marks == "X"), " spares ", sum(marks == "/"), " open ", open, " max ", maximum))
558 }
559 output <- character()
560 for (player in 1:2) {
561 if (m$sport == "DARTS") {
562 darts <- stats$darts[player]
563 average <- if (darts == 0L) 0L else (stats$scored[player] * 300L) %/% darts
564 text <- paste0("darts ", darts, " average ", sprintf("%d.%02d", as.integer(average %/% 100L), as.integer(average %% 100L)),
565 " busts ", stats$busts[player], " high ", stats$high[player], " tons ", stats$tons[player],
566 " 180s ", stats$maximums[player], " doubles ", stats$hits[player], "/", stats$attempts[player])
567 } else {
568 text <- paste0("points ", stats$points[player], " serve ", stats$servewins[player], "/", stats$served[player], " run ", stats$runmax[player])
569 if (m$sport == "TENNIS") {
570 text <- paste0(text, " holds ", stats$holds[player], "/", stats$games[player], " breaks ", stats$breaks[player],
571 " break points ", stats$converted[player], "/", stats$opportunities[player])
572 } else if (m$sport == "VOLLEYBALL") {
573 text <- paste0(text, " subs ", s$subs[player], " timeouts ", s$timeouts[player], " lineup ", paste(s$lineups[[player]], collapse = ","))
574 }
575 }
576 output <- c(output, paste0(m$id, " ", m$names[player], ": ", text))
577 }
578 output
579}
580result_line <- function(m) {
581 s <- m$state
582 if (s$winner == 0L) score_error("match not finished")
583 if (m$sport == "BOWLING") return(paste0(m$id, ": ", m$names[1L], " ", sum(bowling_scores(s))))
584 winner <- s$winner
585 paste0(m$id, ": ", m$names[winner], " def. ", m$names[other(winner)], " ",
586 paste(vapply(s$completed, completed_text, character(1L), winner = winner), collapse = " "))
587}
588execute <- function(line, matches) {
589 tokens <- strsplit(line, " ", fixed = TRUE)[[1L]]
590 command <- tokens[1L]
591 if (command == "MATCH") {
592 if (!is.null(matches[[tokens[2L]]])) score_error("duplicate match")
593 m <- create_match(tokens)
594 matches[[m$id]] <- m
595 return(character())
596 }
597 if (command == "CHECKOUT") {
598 answer <- checkout(as.integer(tokens[2L]))
599 text <- if (!length(answer)) "none" else paste(answer, collapse = " ")
600 return(paste0("checkout ", tokens[2L], ": ", text))
601 }
602 if (command == "STANDINGS") {
603 totals <- new.env(hash = TRUE, parent = emptyenv())
604 for (id in ls(matches, all.names = TRUE)) {
605 m <- matches[[id]]
606 if (m$sport != "BOWLING" && m$state$winner != 0L) {
607 for (player in 1:2) {
608 name <- m$names[player]
609 value <- totals[[name]]
610 if (is.null(value)) value <- c(0L, 0L)
611 index <- if (player == m$state$winner) 1L else 2L
612 value[index] <- value[index] + 1L
613 totals[[name]] <- value
614 }
615 }
616 }
617 names <- ls(totals, all.names = TRUE)
618 if (!length(names)) return("standings: none")
619 wins <- vapply(names, function(name) totals[[name]][1L], integer(1L))
620 losses <- vapply(names, function(name) totals[[name]][2L], integer(1L))
621 names <- names[order(-wins, losses, names, method = "radix")]
622 entries <- vapply(names, function(name) paste(name, pair_text(totals[[name]])), character(1L))
623 return(paste0("standings: ", paste(entries, collapse = ", ")))
624 }
625 m <- matches[[tokens[2L]]]
626 if (is.null(m)) score_error("unknown match")
627 if (command %in% c("POINT", "ROLL", "THROW", "SUB", "TIMEOUT")) {
628 event(m, command, tokens[seq.int(3L, length(tokens))])
629 } else if (command == "UNDO") {
630 undo_event(m)
631 } else if (command == "SCORE") {
632 return(score_line(m))
633 } else if (command == "STATS") {
634 return(stats_lines(m))
635 } else if (command == "RESULT") {
636 return(result_line(m))
637 }
638 character()
639}
640input <- file("stdin", open = "r")
641count <- as.integer(readLines(input, n = 1L))
642matches <- new.env(hash = TRUE, parent = emptyenv())
643for (i in seq_len(count)) {
644 line <- readLines(input, n = 1L)
645 output <- tryCatch(execute(line, matches), error = function(error) paste("ERROR", conditionMessage(error)))
646 if (length(output)) cat(paste(output, collapse = "\n"), "\n", sep = "")
647}
648close(input)
