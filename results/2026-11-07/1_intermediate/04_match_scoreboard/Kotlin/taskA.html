49class Scoreboard {
50 private val matches = linkedMapOf<String,Match>()
51 private val darts = mutableListOf<Dart>()
52 private val doubles = mutableListOf<Dart>()
53 private val dartIndex: Map<String,Int>
54 private val cache = mutableMapOf<Pair<Int,Int>,List<String>>()
55 init {
56 for (prefix in listOf("S","T","D")) {
57 for (n in 20 downTo 1) {
58 val multiplier = when (prefix) {
59 "S" -> 1
60 "T" -> 3
61 else -> 2
62 }
63 val dart = Dart("$prefix$n",n*multiplier,prefix == "D")
64 darts.add(dart)
65 if (dart.doubleOut) doubles.add(dart)
66 }
67 }
68 darts.add(Dart("25",25,false))
69 val bull = Dart("BULL",50,true)
70 darts.add(bull)
71 doubles.add(bull)
72 darts.add(Dart("MISS",0,false))
73 dartIndex = darts.withIndex().associate { it.value.name to it.index }
74 }
75 private fun fail(message: String): Nothing = throw Problem(message)
76 private fun checkout(score: Int, maximum: Int): List<String> = cache.getOrPut(score to maximum) {
77 fun search(): List<String> {
78 for (a in doubles) {
79 if (a.value == score) return listOf(a.name)
80 }
81 if (maximum >= 2) {
82 for (a in darts.take(62)) {
83 for (b in doubles) {
84 if (a.value+b.value == score) return listOf(a.name,b.name)
85 }
86 }
87 }
88 if (maximum >= 3) {
89 for (a in darts.take(62)) {
90 for (b in darts.take(62)) {
91 for (c in doubles) {
92 if (a.value+b.value+c.value == score) return listOf(a.name,b.name,c.name)
93 }
94 }
95 }
96 }
97 return emptyList()
98 }
99 search()
100 }
101 private inner class Match(t: List<String>) {
102 val id = t[1]
103 val sport = t[2]
104 val names = listOf(t[3],if (sport == "BOWLING") "" else t[4])
105 private val options = mutableMapOf<String,String>()
106 private val initial = Array(2) { IntArray(6) }
107 var s = IntArray(100)
108 private val finished = mutableListOf<Finished>()
109 private val history = ArrayDeque<IntArray>()
110 init {
111 val allowed = mutableMapOf<String,Set<String>>()
112 if (sport != "BOWLING") {
113 allowed["FIRST"] = setOf("A","B")
114 options["FIRST"] = "A"
115 }
116 if (sport == "TENNIS") {
117 allowed["SETS"] = setOf("3","5")
118 allowed["DEUCE"] = setOf("AD","NOAD")
119 allowed["FINAL"] = setOf("TB7","TB10","ADV","SUPER")
120 options.putAll(mapOf("SETS" to "3","DEUCE" to "AD","FINAL" to "TB7"))
121 }
122 if (sport == "TABLETENNIS") {
123 allowed["GAMES"] = setOf("5","7")
124 options["GAMES"] = "5"
125 }
126 if (sport == "DARTS") {
127 allowed["START"] = setOf("301","501","701")
128 allowed["SETS"] = setOf("1","2","3","4","5")
129 allowed["LEGS"] = setOf("1","2","3","4","5")
130 options.putAll(mapOf("START" to "501","SETS" to "1","LEGS" to "1"))
131 }
132 val firstOption = if (sport == "BOWLING") 4 else if (sport == "VOLLEYBALL") 7 else 5
133 val seen = mutableSetOf<String>()
134 for (token in t.drop(firstOption)) {
135 val p = token.split('=')
136 if (p.size != 2 || p[0] !in allowed || p[1] !in allowed.getValue(p[0]) || !seen.add(p[0])) fail("invalid option")
137 options[p[0]] = p[1]
138 }
139 s[GS] = first()
140 s[SERVE] = first()
141 s[LEG] = 1
142 s[STANDING] = 10
143 if (sport == "DARTS") {
144 s[R] = option("START")
145 s[R+1] = option("START")
146 s[TS] = option("START")
147 }
148 if (sport == "VOLLEYBALL") {
149 for (side in 0..1) {
150 val lineup = t[5+side].split(',').map { it.toInt() }
151 for (i in 0 until 6) {
152 initial[side][i] = lineup[i]
153 s[LINEUP+side*6+i] = lineup[i]
154 }
155 }
156 }
157 }
158 private fun first() = if (options.getOrDefault("FIRST","A") == "A") 0 else 1
159 private fun option(key: String) = options.getValue(key).toInt()
160 fun over() = s[DONE] != 0
161 private fun tennis() = sport == "TENNIS"
162 fun rally() = sport !in setOf("DARTS","BOWLING")
163 private fun finalSet() = tennis() && s[W] == option("SETS")/2 && s[W+1] == option("SETS")/2
164 private fun superTie() = finalSet() && options["FINAL"] == "SUPER"
165 private fun tieBreak() = superTie() || tennis() && s[G] == 6 && s[G+1] == 6 && !(finalSet() && options["FINAL"] == "ADV")
166 private fun server(): Int {
167 if (tennis()) {
168 val alternate = if (tieBreak()) (s[P]+s[P+1]+1)/2%2 else 0
169 return s[GS] xor alternate
170 }
171 if (sport == "TABLETENNIS") {
172 val played = s[P]+s[P+1]
173 val alternate = if (played < 20) played/2%2 else played%2
174 return first() xor (finished.size%2) xor alternate
175 }
176 return s[SERVE]
177 }
178 private fun normalWin(x: Int, extra: Int) = s[P+x]+extra >= 4 &&
179 (options["DEUCE"] == "NOAD" || s[P+x]+extra-s[P+1-x] >= 2)
180 private fun endSet(x: Int, score: Finished) {
181 finished.add(score)
182 s[W+x]++
183 val target = when (sport) {
184 "TENNIS" -> option("SETS")/2+1
185 "TABLETENNIS" -> option("GAMES")/2+1
186 "VOLLEYBALL" -> 3
187 else -> 2
188 }
189 if (s[W+x] == target) {
190 s[DONE] = 1
191 s[WINNER] = x
192 } else {
193 s[P] = 0
194 s[P+1] = 0
195 s[G] = 0
196 s[G+1] = 0
197 if (sport == "VOLLEYBALL") {
198 for (side in 0..1) {
199 s[SUBS+side] = 0
200 s[TIMEOUTS+side] = 0
201 initial[side].copyInto(s,LINEUP+side*6)
202 }
203 s[SERVE] = first() xor (finished.size%2)
204 }
205 }
206 }
207 private fun point(x: Int) {
208 val serve = server()
209 s[POINTS+x]++
210 s[SERVED+serve]++
211 if (serve == x) s[SERVEWON+x]++
212 s[CURRENT+1-x] = 0
213 s[CURRENT+x]++
214 s[RUN+x] = maxOf(s[RUN+x],s[CURRENT+x])
215 if (tennis()) {
216 val tie = tieBreak()
217 val superTie = superTie()
218 if (!tie && normalWin(1-serve,1)) {
219 s[BPO+1-serve]++
220 if (x != serve) s[BPW+x]++
221 }
222 s[P+x]++
223 if (tie) {
224 val target = if (superTie || finalSet() && options["FINAL"] == "TB10") 10 else 7
225 if (s[P+x] >= target && s[P+x]-s[P+1-x] >= 2) {
226 val score = if (superTie) Finished(s[P],s[P+1],2) else {
227 s[G+x]++
228 Finished(s[G],s[G+1],1,s[P+1-x])
229 }
230 s[GS] = s[GS] xor 1
231 endSet(x,score)
232 }
233 } else if (normalWin(x,0)) {
234 s[SERVICE+s[GS]]++
235 if (x == s[GS]) s[HOLDS+x]++ else s[BREAKS+x]++
236 s[G+x]++
237 s[P] = 0
238 s[P+1] = 0
239 s[GS] = s[GS] xor 1
240 if (s[G+x] >= 6 && s[G+x]-s[G+1-x] >= 2) endSet(x,Finished(s[G],s[G+1]))
241 }
242 return
243 }
244 s[P+x]++
245 if (sport == "BADMINTON") s[SERVE] = x
246 if (sport == "VOLLEYBALL" && s[SERVE] != x) {
247 val start = LINEUP+x*6
248 val old = s[start]
249 s.copyInto(s,start,start+1,start+6)
250 s[start+5] = old
251 s[SERVE] = x
252 }
253 val target = if (sport == "TABLETENNIS") 11 else if (sport == "BADMINTON") 21 else if (finished.size == 4) 15 else 25
254 if (s[P+x] >= target && s[P+x]-s[P+1-x] >= 2 || sport == "BADMINTON" && s[P+x] == 30) endSet(x,Finished(s[P],s[P+1]))
255 }
256 private fun completeTurn(x: Int) {
257 s[HIGH+x] = maxOf(s[HIGH+x],s[TV])
258 if (s[TV] >= 100) s[TONS+x]++
259 if (s[TV] == 180) s[MAXIMUMS+x]++
260 }
261 private fun nextTurn() {
262 s[SERVE] = s[SERVE] xor 1
263 s[TC] = 0
264 s[TV] = 0
265 s[TS] = s[R+s[SERVE]]
266 }
267 private fun dart(code: Int) {
268 val d = darts[code]
269 val x = s[SERVE]
270 val before = s[R+x]
271 if (s[TC] == 0) s[TS] = before
272 if (before == 50 || before <= 40 && before%2 == 0) s[ATTEMPTS+x]++
273 s[DARTS+x]++
274 s[TURN+s[TC]] = code
275 s[TC]++
276 s[TV] += d.value
277 val remaining = before-d.value
278 if (remaining < 0 || remaining == 1 || remaining == 0 && !d.doubleOut) {
279 s[SCORED+x] -= s[TV]-d.value
280 s[BUSTS+x]++
281 s[R+x] = s[TS]
282 nextTurn()
283 } else {
284 s[R+x] = remaining
285 s[SCORED+x] += d.value
286 if (remaining == 0) {
287 s[HITS+x]++
288 completeTurn(x)
289 s[L+x]++
290 if (s[L+x] == option("LEGS")) {
291 finished.add(Finished(s[L],s[L+1]))
292 s[W+x]++
293 if (s[W+x] == option("SETS")) {
294 s[DONE] = 1
295 s[WINNER] = x
296 return
297 }
298 s[L] = 0
299 s[L+1] = 0
300 }
301 s[LEG]++
302 s[R] = option("START")
303 s[R+1] = option("START")
304 s[SERVE] = first() xor ((s[LEG]-1)%2)
305 s[TC] = 0
306 s[TV] = 0
307 s[TS] = option("START")
308 } else if (s[TC] == 3) {
309 completeTurn(x)
310 nextTurn()
311 }
312 }
313 }
314 private fun roll(pins: Int) {
315 s[ROLLS+s[ROLLCOUNT]] = pins
316 s[ROLLCOUNT]++
317 val r = s[FRAMEROLLS]
318 val frame = s[FRAME]
319 if (r == 0) s[FIRSTROLL] = pins
320 if (r == 1) s[SECONDROLL] = pins
321 s[STANDING] -= pins
322 val end = if (frame < 9) r == 1 || r == 0 && pins == 10 else
323 r == 2 || r == 1 && s[FIRSTROLL] != 10 && s[FIRSTROLL]+pins < 10
324 if (end) {
325 if (frame == 9) s[DONE] = 1
326 else {
327 s[FRAME]++
328 s[FRAMEROLLS] = 0
329 s[STANDING] = 10
330 }
331 } else {
332 s[FRAMEROLLS]++
333 if (frame == 9 && s[STANDING] == 0) s[STANDING] = 10
334 }
335 }
336 private fun bowlingFrames(): List<BowlingFrame> {
337 val result = mutableListOf<BowlingFrame>()
338 var cursor = 0
339 var cumulative = 0
340 for (frame in 0 until 10) {
341 if (cursor >= s[ROLLCOUNT]) break
342 val first = s[ROLLS+cursor]
343 val count = if (frame < 9) if (first == 10) 1 else minOf(2,s[ROLLCOUNT]-cursor) else minOf(3,s[ROLLCOUNT]-cursor)
344 val marks = StringBuilder()
345 var standing = 10
346 var sum = 0
347 var fresh = true
348 for (i in 0 until count) {
349 val pins = s[ROLLS+cursor+i]
350 marks.append(if (pins == 10 && fresh) "X" else if (pins == standing) "/" else if (pins == 0) "-" else pins.toString())
351 sum += pins
352 standing -= pins
353 fresh = frame == 9 && standing == 0
354 if (fresh) standing = 10
355 }
356 val score = when {
357 frame == 9 -> if (over()) sum else null
358 first == 10 && cursor+2 < s[ROLLCOUNT] -> 10+s[ROLLS+cursor+1]+s[ROLLS+cursor+2]
359 first != 10 && count == 2 -> if (sum < 10) sum else if (cursor+2 < s[ROLLCOUNT]) 10+s[ROLLS+cursor+2] else null
360 else -> null
361 }
362 val total = score?.let {
363 cumulative += it
364 cumulative
365 }
366 val open = count >= 2 && first != 10 && first+s[ROLLS+cursor+1] < 10
367 result.add(BowlingFrame(marks.toString(),total,open))
368 cursor += count
369 }
370 return result
371 }
372 fun event(command: String, value: String, tokens: List<String>) {
373 if (over()) fail("match over")
374 var code = 0
375 var side = 0
376 var position = -1
377 if (command == "ROLL") {
378 code = value.toInt()
379 if (code > s[STANDING]) fail("too many pins")
380 }
381 if (command == "THROW") code = dartIndex[value] ?: fail("invalid dart")
382 if (command in setOf("SUB","TIMEOUT")) {
383 side = if (tokens[2] == "A") 0 else 1
384 if (command == "SUB") {
385 val from = tokens[3].toInt()
386 val to = tokens[4].toInt()
387 position = (0 until 6).firstOrNull { s[LINEUP+side*6+it] == from } ?: fail("player not on court")
388 if ((0 until 6).any { s[LINEUP+side*6+it] == to }) fail("player already on court")
389 if (s[SUBS+side] == 6) fail("no substitutions left")
390 } else if (s[TIMEOUTS+side] == 2) fail("no timeouts left")
391 }
392 val before = s.copyOf()
393 val count = finished.size
394 when (command) {
395 "POINT" -> point(if (value == "A") 0 else 1)
396 "ROLL" -> roll(code)
397 "THROW" -> dart(code)
398 "SUB" -> {
399 s[LINEUP+side*6+position] = tokens[4].toInt()
400 s[SUBS+side]++
401 }
402 "TIMEOUT" -> s[TIMEOUTS+side]++
403 }
404 val changed = s.indices.filter { before[it] != s[it] }
405 val undo = IntArray(1+changed.size*2)
406 undo[0] = count
407 for ((i,index) in changed.withIndex()) {
408 undo[1+i*2] = index
409 undo[2+i*2] = before[index]
410 }
411 history.addLast(undo)
412 }
413 fun undo() {
414 if (history.isEmpty()) fail("nothing to undo")
415 val undo = history.removeLast()
416 for (i in 1 until undo.size step 2) s[undo[i]] = undo[i+1]
417 finished.subList(undo[0],finished.size).clear()
418 }
419 private fun situation(x: Int): String {
420 val old = s.copyOf()
421 val count = finished.size
422 val games = s[G]+s[G+1]
423 val serve = server()
424 val normal = tennis() && !tieBreak()
425 point(x)
426 val result = when {
427 over() -> "match point"
428 finished.size > count -> if (tennis() || sport == "VOLLEYBALL") "set point" else "game point"
429 normal && s[G]+s[G+1] > games -> if (x == serve) "game point" else "break point"
430 else -> ""
431 }
432 s = old
433 finished.subList(count,finished.size).clear()
434 return result
435 }
436 private fun finishedScore(f: Finished, first: Int): String {
437 val pair = if (first == 0) "${f.a}-${f.b}" else "${f.b}-${f.a}"
438 return if (f.type == 2) "[$pair]" else pair+if (f.type == 1) "(${f.loser})" else ""
439 }
440 private fun pair(offset: Int) = "${s[offset]}-${s[offset+1]}"
441 fun score(): String {
442 if (sport == "BOWLING") {
443 val frames = bowlingFrames().map { it.marks+(it.cumulative?.let { total -> "($total)" } ?: "") }
444 return "$id: ${names[0]}"+if (frames.isEmpty()) "" else " ${frames.joinToString(" ")}"
445 }
446 val serve = if (over()) -1 else server()
447 val items = mutableListOf<String>()
448 if (sport == "DARTS") {
449 items.addAll(listOf("sets",pair(W),"legs",pair(L),"score",pair(R)))
450 if (!over()) {
451 if (s[TC] > 0) {
452 items.add("turn")
453 for (i in 0 until s[TC]) items.add(darts[s[TURN+i]].name)
454 }
455 val check = checkout(s[R+s[SERVE]],3-s[TC])
456 if (check.isNotEmpty()) {
457 items.add("checkout")
458 items.addAll(check)
459 }
460 }
461 } else {
462 items.addAll(finished.map { finishedScore(it,0) })
463 if (!over()) {
464 if (tennis()) {
465 if (superTie()) items.add("[${pair(P)}]")
466 else {
467 items.add(pair(G))
468 if (tieBreak()) items.add(pair(P))
469 else if (options["DEUCE"] == "AD" && s[P] >= 3 && s[P+1] >= 3) {
470 items.add(if (s[P] == s[P+1]) "40-40" else if (s[P] > s[P+1]) "AD-40" else "40-AD")
471 } else {
472 val points = listOf("0","15","30","40")
473 items.add("${points[s[P]]}-${points[s[P+1]]}")
474 }
475 }
476 } else {
477 items.add(pair(P))
478 if (sport == "VOLLEYBALL") items.addAll(listOf("server",s[LINEUP+s[SERVE]*6].toString()))
479 }
480 for (x in 0..1) {
481 val situation = situation(x)
482 if (situation.isNotEmpty()) items.add("($situation ${names[x]})")
483 }
484 }
485 }
486 return "$id: ${names[0]}${if (serve == 0) "*" else ""} vs ${names[1]}${if (serve == 1) "*" else ""} ${items.joinToString(" ")}"
487 }
488 fun stats(): List<String> {
489 if (sport == "BOWLING") {
490 val frames = bowlingFrames()
491 val strikes = frames.sumOf { it.marks.count { c -> c == 'X' } }
492 val spares = frames.sumOf { it.marks.count { c -> c == '/' } }
493 val open = frames.count { it.open }
494 val old = s.copyOf()
495 while (!over()) roll(s[STANDING])
496 val maximum = bowlingFrames().last().cumulative!!
497 s = old
498 return listOf("$id ${names[0]}: strikes $strikes spares $spares open $open max $maximum")
499 }
500 return (0..1).map { x ->
501 var line = "$id ${names[x]}: "
502 if (sport == "DARTS") {
503 val average = if (s[DARTS+x] == 0) 0L else s[SCORED+x].toLong()*300/s[DARTS+x]
504 line += "darts ${s[DARTS+x]} average ${average/100}.${(average%100).toString().padStart(2,'0')}"+
505 " busts ${s[BUSTS+x]} high ${s[HIGH+x]} tons ${s[TONS+x]} 180s ${s[MAXIMUMS+x]} doubles ${s[HITS+x]}/${s[ATTEMPTS+x]}"
506 } else {
507 line += "points ${s[POINTS+x]} serve ${s[SERVEWON+x]}/${s[SERVED+x]} run ${s[RUN+x]}"
508 if (tennis()) line += " holds ${s[HOLDS+x]}/${s[SERVICE+x]} breaks ${s[BREAKS+x]} break points ${s[BPW+x]}/${s[BPO+x]}"
509 if (sport == "VOLLEYBALL") line += " subs ${s[SUBS+x]} timeouts ${s[TIMEOUTS+x]} lineup "+
510 (0 until 6).joinToString(",") { s[LINEUP+x*6+it].toString() }
511 }
512 line
513 }
514 }
515 fun result(): String {
516 if (!over()) fail("match not finished")
517 if (sport == "BOWLING") return "$id: ${names[0]} ${bowlingFrames().last().cumulative}"
518 val winner = s[WINNER]
519 return "$id: ${names[winner]} def. ${names[1-winner]} ${finished.joinToString(" ") { finishedScore(it,winner) }}"
520 }
521 }
522 fun execute(line: String): List<String> {
523 val t = line.split(' ')
524 val command = t[0]
525 if (command == "MATCH") {
526 if (t[1] in matches) fail("duplicate match")
527 val match = Match(t)
528 matches[t[1]] = match
529 return emptyList()
530 }
531 if (command == "CHECKOUT") {
532 val answer = checkout(t[1].toInt(),3)
533 return listOf("checkout ${t[1]}: "+if (answer.isEmpty()) "none" else answer.joinToString(" "))
534 }
535 if (command == "STANDINGS") {
536 val totals = mutableMapOf<String,IntArray>()
537 for (m in matches.values) {
538 if (m.over() && m.sport != "BOWLING") {
539 val winner = m.s[WINNER]
540 totals.getOrPut(m.names[winner]) { IntArray(2) }[0]++
541 totals.getOrPut(m.names[1-winner]) { IntArray(2) }[1]++
542 }
543 }
544 val names = totals.keys.sortedWith(compareByDescending<String> { totals.getValue(it)[0] }.thenBy { totals.getValue(it)[1] }.thenBy { it })
545 val records = names.map { "$it ${totals.getValue(it)[0]}-${totals.getValue(it)[1]}" }
546 return listOf("standings: "+if (records.isEmpty()) "none" else records.joinToString(", "))
547 }
548 val match = matches[t[1]] ?: fail("unknown match")
549 when (command) {
550 "UNDO" -> match.undo()
551 "SCORE" -> return listOf(match.score())
552 "STATS" -> return match.stats()
553 "RESULT" -> return listOf(match.result())
554 else -> {
555 val correct = when (command) {
556 "POINT" -> match.rally()
557 "ROLL" -> match.sport == "BOWLING"
558 "THROW" -> match.sport == "DARTS"
559 else -> match.sport == "VOLLEYBALL"
560 }
561 if (!correct) fail("wrong event")
562 when (command) {
563 "POINT" -> {
564 for (c in t[2]) match.event(command,c.toString(),t)
565 }
566 "ROLL", "THROW" -> {
567 for (value in t.drop(2)) match.event(command,value,t)
568 }
569 else -> match.event(command,"",t)
570 }
571 }
572 }
573 return emptyList()
574 }
575}

577fun main() {
578 val input = System.`in`.bufferedReader()
579 val output = System.out.bufferedWriter()
580 val app = Scoreboard()
581 repeat(input.readLine().toInt()) {
582 val lines = try {
583 app.execute(input.readLine())
584 } catch (problem: Problem) {
585 listOf("ERROR ${problem.message}")
586 }
587 for (line in lines) {
588 output.write(line)
589 output.newLine()
590 }
591 }
592 output.flush()
593}
