64func fail(message string) { panic(message) }
65func number(s string) int {
66 n, _ := strconv.Atoi(s)

69func pairText(p [2]int, first int) string { return fmt.Sprintf("%d-%d", p[first], p[1-first]) }
70func initializeDarts() {
71 for _, multiplier := range []int{1, 3, 2} {
72 prefix := "S"
73 if multiplier == 2 { prefix = "D" }
74 if multiplier == 3 { prefix = "T" }
75 for n := 20; n >= 1; n-- {
76 dartKinds = append(dartKinds, dart{fmt.Sprintf("%s%d", prefix, n), multiplier*n, multiplier == 2})
77 }
78 }
79 dartKinds = append(dartKinds, dart{"25", 25, false}, dart{"BULL", 50, true}, dart{"MISS", 0, false})
80}
81func doubleDart(value int) int {
82 if value == 50 { return 61 }
83 if value >= 2 && value <= 40 && value%2 == 0 { return 60-value/2 }
84 return -1
85}
86func checkout(score, maximum int) []int {
87 if last := doubleDart(score); last >= 0 { return []int{last} }
88 if maximum >= 2 {
89 for i := 0; i < 62; i++ {
90 if last := doubleDart(score-dartKinds[i].value); last >= 0 { return []int{i, last} }
91 }
92 }
93 if maximum >= 3 && score <= 170 {
94 for i := 0; i < 62; i++ {
95 for j := 0; j < 62; j++ {
96 if last := doubleDart(score-dartKinds[i].value-dartKinds[j].value); last >= 0 { return []int{i, j, last} }
97 }
98 }
99 }
100 return nil
101}
102func dartNames(ids []int) string {
103 result := make([]string, len(ids))
104 for i, id := range ids { result[i] = dartKinds[id].name }
105 return strings.Join(result, " ")
106}
107func describeBowling(state bowlingState) bowlingInfo {
108 info := bowlingInfo{standing: 10, fresh: true}
109 for i := 0; i < state.count; i++ {
110 frame, pins := info.frame, state.rolls[i]
111 if info.count[frame] == 0 { info.start[frame] = i }
112 mark := byte('0'+pins)
113 switch {
114 case info.fresh && pins == 10: mark = 'X'
115 case pins == info.standing: mark = '/'
116 case pins == 0: mark = '-'
117 }
118 info.marks[i] = mark
119 info.count[frame]++
120 info.standing -= pins
121 info.fresh = false
122 ended := false
123 if frame < 9 { ended = pins == 10 && info.count[frame] == 1 || info.count[frame] == 2 } else if info.count[frame] == 2 {
124 ended = state.rolls[info.start[frame]]+pins < 10
125 } else if info.count[frame] == 3 { ended = true }
126 if ended {
127 info.closed[frame] = true
128 if frame == 9 { info.finished = true } else { info.frame++ }
129 info.standing, info.fresh = 10, true
130 } else if frame == 9 && info.standing == 0 { info.standing, info.fresh = 10, true }
131 }
132 for frame := 0; frame < 10; frame++ {
133 if info.count[frame] == 0 { continue }
134 start := info.start[frame]
135 if frame == 9 {
136 info.final[frame] = info.closed[frame]
137 for j := 0; j < info.count[frame]; j++ { info.score[frame] += state.rolls[start+j] }
138 } else if state.rolls[start] == 10 {
139 info.final[frame] = state.count >= start+3
140 if info.final[frame] { info.score[frame] = 10+state.rolls[start+1]+state.rolls[start+2] }
141 } else if info.count[frame] >= 2 {
142 total := state.rolls[start]+state.rolls[start+1]
143 info.final[frame] = total < 10 || state.count >= start+3
144 if info.final[frame] {
145 info.score[frame] = total
146 if total == 10 { info.score[frame] += state.rolls[start+2] }
147 }
148 }
149 }
150 return info
151}
152func bowlingTotal(info bowlingInfo) int {
153 total := 0
154 for _, score := range info.score { total += score }
155 return total
156}
157func finalSet(m *match, s rallyState) bool { return s.sets[0] == m.sets/2 && s.sets[1] == m.sets/2 }
158func tennisKind(m *match, s rallyState) int {
159 if finalSet(m, s) && m.final == 3 { return 2 }
160 if s.games[0] == 6 && s.games[1] == 6 && !(finalSet(m, s) && m.final == 2) { return 1 }
161 return 0
162}
163func rallyServer(m *match, s rallyState) int {
164 if m.sport == tennis && tennisKind(m, s) != 0 { return s.server ^ ((s.points[0]+s.points[1]+1)/2%2) }
165 if m.sport == tableTennis {
166 first, played := m.first ^ (s.segmentCount%2), s.points[0]+s.points[1]
167 if played < 20 { return first ^ (played/2%2) }
168 return first ^ (played%2)
169 }
170 return s.server
171}
172func normalGame(m *match, own, other int) bool { return own >= 4 && (m.noAd || own-other >= 2) }
173func addRally(m *match, s *rallyState, side int) {
174 other, server := 1-side, rallyServer(m, *s)
175 s.stats[side][0]++
176 s.stats[server][1]++
177 if side == server { s.stats[side][2]++ }
178 s.stats[side][3]++
179 s.stats[other][3] = 0
180 s.stats[side][4] = max(s.stats[side][4], s.stats[side][3])
181 kind := 0
182 if m.sport == tennis {
183 kind = tennisKind(m, *s)
184 receiver := 1-s.server
185 if kind == 0 && normalGame(m, s.points[receiver]+1, s.points[s.server]) {
186 s.stats[receiver][8]++
187 if side == receiver { s.stats[receiver][9]++ }
188 }
189 }
190 s.points[side]++
191 if m.sport == tennis {
192 target := 7
193 if kind == 2 || kind == 1 && finalSet(m, *s) && m.final == 1 { target = 10 }
194 ended := normalGame(m, s.points[side], s.points[other])
195 if kind != 0 { ended = s.points[side] >= target && s.points[side]-s.points[other] >= 2 }
196 if !ended { return }
197 if kind == 0 {
198 s.stats[s.server][5]++
199 if side == s.server { s.stats[side][6]++ } else { s.stats[side][7]++ }
200 }
201 var finished segment
202 if kind == 2 { finished = segment{s.points[0], s.points[1], 2, 0} } else {
203 s.games[side]++
204 loser := 0
205 if kind != 0 { loser = s.points[other] }
206 finished = segment{s.games[0], s.games[1], kind, loser}
207 }
208 s.server ^= 1
209 s.points = [2]int{}
210 if kind != 0 || s.games[side] >= 6 && s.games[side]-s.games[other] >= 2 {
211 s.segments[s.segmentCount] = finished
212 s.segmentCount++
213 s.sets[side]++
214 s.games = [2]int{}
215 if s.sets[side] == m.sets/2+1 { s.winner = side }
216 }
217 return
218 }
219 if m.sport == volleyball && side != s.server {
220 s.server = side
221 first := s.lineup[side][0]
222 copy(s.lineup[side][:5], s.lineup[side][1:])
223 s.lineup[side][5] = first
224 }
225 if m.sport == badminton { s.server = side }
226 target := 25
227 if m.sport == tableTennis { target = 11 }
228 if m.sport == badminton { target = 21 }
229 if m.sport == volleyball && s.segmentCount == 4 { target = 15 }
230 ended := s.points[side] >= target && s.points[side]-s.points[other] >= 2
231 if m.sport == badminton && s.points[side] == 30 { ended = true }
232 if !ended { return }
233 s.segments[s.segmentCount] = segment{s.points[0], s.points[1], 0, 0}
234 s.segmentCount++
235 s.sets[side]++
236 needed := 3
237 if m.sport == tableTennis { needed = m.games/2+1 }
238 if m.sport == badminton { needed = 2 }
239 if s.sets[side] == needed { s.winner = side } else {
240 s.points = [2]int{}
241 if m.sport == volleyball {
242 s.lineup = m.initialLineup
243 s.substitutions, s.timeouts = [2]int{}, [2]int{}
244 s.server = m.first ^ (s.segmentCount%2)
245 }
246 }
247}
248func endDartsTurn(s *dartsState) {
249 s.server ^= 1
250 s.turnStart = s.remaining[s.server]
251 s.turnCount, s.turnScore = 0, 0
252}
253func completedDartsTurn(s *dartsState, side int) {
254 s.stats[side][3] = max(s.stats[side][3], s.turnScore)
255 if s.turnScore >= 100 { s.stats[side][4]++ }
256 if s.turnScore == 180 { s.stats[side][5]++ }
257}
258func addDart(m *match, s *dartsState, id int) {
259 side := s.server
260 remaining, dart := s.remaining[side], dartKinds[id]
261 s.stats[side][0]++
262 if remaining == 50 || remaining <= 40 && remaining%2 == 0 { s.stats[side][6]++ }
263 s.turn[s.turnCount] = id
264 s.turnCount++
265 s.turnScore += dart.value
266 s.stats[side][1] += dart.value
267 after := remaining-dart.value
268 if after < 0 || after == 1 || after == 0 && !dart.double {
269 s.stats[side][1] -= s.turnScore
270 s.stats[side][2]++
271 s.remaining[side] = s.turnStart
272 endDartsTurn(s)
273 return
274 }
275 s.remaining[side] = after
276 if after == 0 {
277 s.stats[side][7]++
278 completedDartsTurn(s, side)
279 s.legs[side]++
280 if s.legs[side] == m.legs {
281 s.segments[s.segmentCount] = s.legs
282 s.segmentCount++
283 s.sets[side]++
284 if s.sets[side] == m.sets {
285 s.winner = side
286 return
287 }
288 s.legs = [2]int{}
289 }
290 s.legNumber++
291 s.remaining = [2]int{m.start, m.start}
292 s.server = m.first ^ ((s.legNumber-1)%2)
293 s.turnStart = m.start
294 s.turnCount, s.turnScore = 0, 0
295 } else if s.turnCount == 3 {
296 completedDartsTurn(s, side)
297 endDartsTurn(s)
298 }
299}
300func finished(m *match) bool {
301 switch s := m.state.(type) {
302 case bowlingState: return describeBowling(s).finished
303 case dartsState: return s.winner >= 0
304 default: return s.(rallyState).winner >= 0
305 }
306}
307func winningSide(m *match) int {
308 if s, ok := m.state.(dartsState); ok { return s.winner }
309 return m.state.(rallyState).winner
310}
311func segmentText(s segment, first int) string {
312 result := pairText([2]int{s.a, s.b}, first)
313 if s.kind == 1 { result += fmt.Sprintf("(%d)", s.loser) }
314 if s.kind == 2 { result = "["+result+"]" }
315 return result
316}
317func situation(m *match, s rallyState, side int) string {
318 next := s
319 addRally(m, &next, side)
320 if next.winner >= 0 { return "match point" }
321 if next.segmentCount != s.segmentCount {
322 if m.sport == tableTennis || m.sport == badminton { return "game point" }
323 return "set point"
324 }
325 if m.sport == tennis && tennisKind(m, s) == 0 && next.server != s.server {
326 if rallyServer(m, s) == side { return "game point" }
327 return "break point"
328 }
329 return ""
330}
331func score(m *match) string {
332 if s, ok := m.state.(bowlingState); ok {
333 info := describeBowling(s)
334 result := m.id+": "+m.names[0]
335 cumulative := 0
336 for frame := 0; frame < 10; frame++ {
337 if info.count[frame] == 0 { continue }
338 start := info.start[frame]
339 display := string(info.marks[start:start+info.count[frame]])
340 cumulative += info.score[frame]
341 if info.final[frame] { display += fmt.Sprintf("(%d)", cumulative) }
342 result += " "+display
343 }
344 return result
345 }
346 over, server := finished(m), 0
347 if s, ok := m.state.(dartsState); ok { server = s.server } else { server = rallyServer(m, m.state.(rallyState)) }
348 names := m.names
349 if !over { names[server] += "*" }
350 result := m.id+": "+names[0]+" vs "+names[1]
351 if s, ok := m.state.(dartsState); ok {
352 result += " sets "+pairText(s.sets, 0)+" legs "+pairText(s.legs, 0)+" score "+pairText(s.remaining, 0)
353 if !over {
354 if s.turnCount > 0 { result += " turn "+dartNames(s.turn[:s.turnCount]) }
355 if finish := checkout(s.remaining[s.server], 3-s.turnCount); len(finish) > 0 { result += " checkout "+dartNames(finish) }
356 }
357 return result
358 }
359 s := m.state.(rallyState)
360 for i := 0; i < s.segmentCount; i++ { result += " "+segmentText(s.segments[i], 0) }
361 if !over {
362 if m.sport == tennis {
363 kind := tennisKind(m, s)
364 if kind == 2 { result += " ["+pairText(s.points, 0)+"]" } else {
365 result += " "+pairText(s.games, 0)+" "
366 if kind != 0 { result += pairText(s.points, 0) } else if !m.noAd && s.points[0] >= 3 && s.points[1] >= 3 {
367 if s.points[0] == s.points[1] { result += "40-40" } else if s.points[0] > s.points[1] { result += "AD-40" } else { result += "40-AD" }
368 } else {
369 labels := []string{"0", "15", "30", "40"}
370 result += labels[s.points[0]]+"-"+labels[s.points[1]]
371 }
372 }
373 } else {
374 result += " "+pairText(s.points, 0)
375 if m.sport == volleyball { result += fmt.Sprintf(" server %d", s.lineup[s.server][0]) }
376 }
377 for side := 0; side < 2; side++ {
378 if label := situation(m, s, side); label != "" { result += " ("+label+" "+m.names[side]+")" }
379 }
380 }
381 return result
382}
383func statistics(m *match) []string {
384 if s, ok := m.state.(bowlingState); ok {
385 info := describeBowling(s)
386 strikes, spares, open := 0, 0, 0
387 for i := 0; i < s.count; i++ {
388 if info.marks[i] == 'X' { strikes++ }
389 if info.marks[i] == '/' { spares++ }
390 }
391 for i := 0; i < 10; i++ {
392 start := info.start[i]
393 if info.count[i] >= 2 && s.rolls[start] != 10 && s.rolls[start]+s.rolls[start+1] < 10 { open++ }
394 }
395 for !info.finished {
396 s.rolls[s.count] = info.standing
397 s.count++
398 info = describeBowling(s)
399 }
400 return []string{fmt.Sprintf("%s %s: strikes %d spares %d open %d max %d", m.id, m.names[0], strikes, spares, open, bowlingTotal(info))}
401 }
402 var result []string
403 for side := 0; side < 2; side++ {
404 line := m.id+" "+m.names[side]+": "
405 if state, ok := m.state.(dartsState); ok {
406 s := state.stats[side]
407 average := 0
408 if s[0] > 0 { average = s[1]*300/s[0] }
409 line += fmt.Sprintf("darts %d average %d.%02d busts %d high %d tons %d 180s %d doubles %d/%d", s[0], average/100, average%100, s[2], s[3], s[4], s[5], s[7], s[6])
410 } else {
411 r := m.state.(rallyState)
412 s := r.stats[side]
413 line += fmt.Sprintf("points %d serve %d/%d run %d", s[0], s[2], s[1], s[4])
414 if m.sport == tennis { line += fmt.Sprintf(" holds %d/%d breaks %d break points %d/%d", s[6], s[5], s[7], s[9], s[8]) }
415 if m.sport == volleyball {
416 var lineup []string
417 for _, value := range r.lineup[side] { lineup = append(lineup, strconv.Itoa(value)) }
418 line += fmt.Sprintf(" subs %d timeouts %d lineup %s", r.substitutions[side], r.timeouts[side], strings.Join(lineup, ","))
419 }
420 }
421 result = append(result, line)
422 }
423 return result
424}
425func result(m *match) string {
426 if !finished(m) { fail("match not finished") }
427 if s, ok := m.state.(bowlingState); ok { return fmt.Sprintf("%s: %s %d", m.id, m.names[0], bowlingTotal(describeBowling(s))) }
428 first := winningSide(m)
429 line := m.id+": "+m.names[first]+" def. "+m.names[1-first]
430 if s, ok := m.state.(dartsState); ok {
431 for i := 0; i < s.segmentCount; i++ { line += " "+pairText(s.segments[i], first) }
432 } else {
433 s := m.state.(rallyState)
434 for i := 0; i < s.segmentCount; i++ { line += " "+segmentText(s.segments[i], first) }
435 }
436 return line
437}
438func createMatch(a []string) {
439 if _, exists := matches[a[1]]; exists { fail("duplicate match") }
440 m := &match{id: a[1], sport: slices.Index([]string{"TENNIS", "TABLETENNIS", "BADMINTON", "VOLLEYBALL", "BOWLING", "DARTS"}, a[2]), sets: 3, games: 5, start: 501, legs: 1}
441 m.names[0] = a[3]
442 if m.sport != bowling { m.names[1] = a[4] }
443 if m.sport == dartsSport { m.sets = 1 }
444 optionStart := 5
445 if m.sport == bowling { optionStart = 4 }
446 if m.sport == volleyball { optionStart = 7 }
447 allowed := map[string][]string{}
448 switch m.sport {
449 case tennis:
450 allowed = map[string][]string{"SETS":{"3", "5"}, "DEUCE":{"AD", "NOAD"}, "FINAL":{"TB7", "TB10", "ADV", "SUPER"}, "FIRST":{"A", "B"}}
451 case tableTennis: allowed = map[string][]string{"GAMES":{"5", "7"}, "FIRST":{"A", "B"}}
452 case badminton, volleyball: allowed = map[string][]string{"FIRST":{"A", "B"}}
453 case dartsSport:
454 allowed = map[string][]string{"START":{"301", "501", "701"}, "SETS":{"1", "2", "3", "4", "5"}, "LEGS":{"1", "2", "3", "4", "5"}, "FIRST":{"A", "B"}}
455 }
456 seen := map[string]bool{}
457 for _, item := range a[optionStart:] {
458 pair := strings.Split(item, "=")
459 if len(pair) != 2 { fail("invalid option") }
460 values, exists := allowed[pair[0]]
461 if !exists || seen[pair[0]] || slices.Index(values, pair[1]) < 0 { fail("invalid option") }
462 seen[pair[0]] = true
463 switch pair[0] {
464 case "FIRST": m.first = slices.Index(values, pair[1])
465 case "SETS": m.sets = number(pair[1])
466 case "GAMES": m.games = number(pair[1])
467 case "START": m.start = number(pair[1])
468 case "LEGS": m.legs = number(pair[1])
469 case "DEUCE": m.noAd = pair[1] == "NOAD"
470 case "FINAL": m.final = slices.Index(values, pair[1])
471 }
472 }
473 switch m.sport {
474 case bowling: m.state = bowlingState{}
475 case dartsSport: m.state = dartsState{remaining: [2]int{m.start, m.start}, server: m.first, turnStart: m.start, legNumber: 1, winner: -1}
476 default:
477 s := rallyState{server: m.first, winner: -1}
478 if m.sport == volleyball {
479 for side := 0; side < 2; side++ {
480 for i, value := range strings.Split(a[5+side], ",") { m.initialLineup[side][i] = number(value) }
481 }
482 s.lineup = m.initialLineup
483 }
484 m.state = s
485 }
486 matches[m.id] = m
487}
488func execute(a []string) []string {
489 command := a[0]
490 if command == "MATCH" {
491 createMatch(a)
492 return nil
493 }
494 if command == "CHECKOUT" {
495 finish := checkout(number(a[1]), 3)
496 display := "none"
497 if len(finish) > 0 { display = dartNames(finish) }
498 return []string{"checkout "+a[1]+": "+display}
499 }
500 if command == "STANDINGS" {
501 totals := map[string][2]int{}
502 for _, m := range matches {
503 if m.sport == bowling || !finished(m) { continue }
504 first := winningSide(m)
505 for side := 0; side < 2; side++ {
506 value := totals[m.names[side]]
507 if side == first { value[0]++ } else { value[1]++ }
508 totals[m.names[side]] = value
509 }
510 }
511 var names []string
512 for name := range totals { names = append(names, name) }
513 sort.Slice(names, func(i, j int) bool {
514 x, y := totals[names[i]], totals[names[j]]
515 if x[0] != y[0] { return x[0] > y[0] }
516 if x[1] != y[1] { return x[1] < y[1] }
517 return names[i] < names[j]
518 })
519 var displays []string
520 for _, name := range names { displays = append(displays, name+" "+pairText(totals[name], 0)) }
521 display := "none"
522 if len(displays) > 0 { display = strings.Join(displays, ", ") }
523 return []string{"standings: "+display}
524 }
525 m, exists := matches[a[1]]
526 if !exists { fail("unknown match") }
527 switch command {
528 case "SCORE": return []string{score(m)}
529 case "STATS": return statistics(m)
530 case "RESULT": return []string{result(m)}
531 case "UNDO":
532 if len(m.history) == 0 { fail("nothing to undo") }
533 m.state = m.history[len(m.history)-1]
534 m.history = m.history[:len(m.history)-1]
535 return nil
536 }
537 allowed := command == "POINT" && m.sport <= volleyball || command == "ROLL" && m.sport == bowling || command == "THROW" && m.sport == dartsSport || (command == "SUB" || command == "TIMEOUT") && m.sport == volleyball
538 if !allowed { fail("wrong event") }
539 count := 1
540 if command == "POINT" { count = len(a[2]) }
541 if command == "ROLL" || command == "THROW" { count = len(a)-2 }
542 for i := 0; i < count; i++ {
543 if finished(m) { fail("match over") }
544 value, position, side := 0, -1, 0
545 switch command {
546 case "POINT":
547 if a[2][i] == 'B' { value = 1 }
548 case "ROLL":
549 value = number(a[i+2])
550 if value > describeBowling(m.state.(bowlingState)).standing { fail("too many pins") }
551 case "THROW":
552 value = slices.IndexFunc(dartKinds, func(d dart) bool { return d.name == a[i+2] })
553 if value < 0 { fail("invalid dart") }
554 case "SUB", "TIMEOUT":
555 if a[2] == "B" { side = 1 }
556 s := m.state.(rallyState)
557 if command == "TIMEOUT" {
558 if s.timeouts[side] >= 2 { fail("no timeouts left") }
559 } else {
560 leaving := number(a[3])
561 value = number(a[4])
562 position = slices.Index(s.lineup[side][:], leaving)
563 if position < 0 { fail("player not on court") }
564 if slices.Contains(s.lineup[side][:], value) { fail("player already on court") }
565 if s.substitutions[side] >= 6 { fail("no substitutions left") }
566 }
567 }
568 m.history = append(m.history, m.state)
569 switch command {
570 case "POINT":
571 s := m.state.(rallyState)
572 addRally(m, &s, value)
573 m.state = s
574 case "ROLL":
575 s := m.state.(bowlingState)
576 s.rolls[s.count] = value
577 s.count++
578 m.state = s
579 case "THROW":
580 s := m.state.(dartsState)
581 addDart(m, &s, value)
582 m.state = s
583 case "SUB", "TIMEOUT":
584 s := m.state.(rallyState)
585 if command == "SUB" {
586 s.lineup[side][position] = value
587 s.substitutions[side]++
588 } else { s.timeouts[side]++ }
589 m.state = s
590 }
591 }
592 return nil
593}
594func commandResult(a []string) (result []string) {
595 defer func() {
596 if message := recover(); message != nil { result = []string{"ERROR "+message.(string)} }
597 }()
598 return execute(a)
599}
