245 void dart(int code) {
246 Dart d = darts.get(code);
247 int x = s[SERVE], before = s[R+x];
248 if (s[TC] == 0) s[TS] = before;
249 if (before == 50 || before <= 40 && before%2 == 0) s[ATTEMPTS+x]++;
250 s[DARTS+x]++;

252 s[TC]++;

255 if (remaining < 0 || remaining == 1 || remaining == 0 && !d.doubleOut()) {
256 s[SCORED+x] -= s[TV]-d.value();
257 s[BUSTS+x]++;
258 s[R+x] = s[TS];
259 nextTurn();
260 } else {
261 s[R+x] = remaining;
262 s[SCORED+x] += d.value();

264 s[HITS+x]++;

266 s[L+x]++;

275 s[L] = 0;
276 s[L+1] = 0;
277 }
278 s[LEG]++;
279 s[R] = option("START");
280 s[R+1] = option("START");

282 s[TC] = 0;

313 List<BowlingFrame> bowlingFrames() {
314 List<BowlingFrame> result = new ArrayList<>();
315 int cursor = 0, cumulative = 0;
316 for (int frame = 0; frame < 10 && cursor < s[ROLLCOUNT]; frame++) {
317 int first = s[ROLLS+cursor], count;
318 if (frame < 9) count = first == 10 ? 1 : Math.min(2,s[ROLLCOUNT]-cursor);
319 else count = Math.min(3,s[ROLLCOUNT]-cursor);
320 StringBuilder marks = new StringBuilder();
321 int standing = 10, sum = 0;
322 boolean fresh = true;
323 for (int i = 0; i < count; i++) {
324 int pins = s[ROLLS+cursor+i];
325 marks.append(pins == 10 && fresh ? "X" : pins == standing ? "/" : pins == 0 ? "-" : Integer.toString(pins));
326 sum += pins;
327 standing -= pins;
328 fresh = frame == 9 && standing == 0;
329 if (fresh) standing = 10;
330 }
331 Integer score = null;
332 if (frame == 9) {
333 if (over()) score = sum;
334 } else if (first == 10 && cursor+2 < s[ROLLCOUNT]) {
335 score = 10+s[ROLLS+cursor+1]+s[ROLLS+cursor+2];
336 } else if (first != 10 && count == 2) {
337 if (sum < 10) score = sum;
338 else if (cursor+2 < s[ROLLCOUNT]) score = 10+s[ROLLS+cursor+2];
339 }
340 Integer total = null;
341 if (score != null) {
342 cumulative += score;
343 total = cumulative;
344 }
345 boolean open = count >= 2 && first != 10 && first+s[ROLLS+cursor+1] < 10;
346 result.add(new BowlingFrame(marks.toString(),total,open));
347 cursor += count;
348 }
349 return result;

351 void event(String command, String value, String[] tokens) {
352 if (over()) fail("match over");
353 int code = 0, side = 0, position = -1;
354 if (command.equals("ROLL")) {
355 code = Integer.parseInt(value);
356 if (code > s[STANDING]) fail("too many pins");
357 }
358 if (command.equals("THROW")) {
359 Integer index = dartIndex.get(value);
360 if (index == null) fail("invalid dart");
361 code = index;
362 }
363 if (command.equals("SUB") || command.equals("TIMEOUT")) {
364 side = tokens[2].equals("A") ? 0 : 1;
365 if (command.equals("SUB")) {
366 int from = Integer.parseInt(tokens[3]), to = Integer.parseInt(tokens[4]);
367 for (int i = 0; i < 6; i++) {
368 if (s[LINEUP+side*6+i] == from) position = i;
369 }
370 if (position < 0) fail("player not on court");
371 for (int i = 0; i < 6; i++) {
372 if (s[LINEUP+side*6+i] == to) fail("player already on court");
373 }
374 if (s[SUBS+side] == 6) fail("no substitutions left");
375 } else if (s[TIMEOUTS+side] == 2) fail("no timeouts left");
376 }
377 int[] before = s.clone();
378 int count = finished.size();
379 switch (command) {
380 case "POINT" -> point(value.equals("A") ? 0 : 1);
381 case "ROLL" -> roll(code);
382 case "THROW" -> dart(code);
383 case "SUB" -> {
384 s[LINEUP+side*6+position] = Integer.parseInt(tokens[4]);
385 s[SUBS+side]++;
386 }
387 case "TIMEOUT" -> s[TIMEOUTS+side]++;
388 }
389 int changed = 0;
390 for (int i = 0; i < s.length; i++) {
391 if (before[i] != s[i]) changed++;
392 }
393 int[] undo = new int[1+changed*2];
394 undo[0] = count;
395 int cursor = 1;
396 for (int i = 0; i < s.length; i++) {
397 if (before[i] != s[i]) {
398 undo[cursor++] = i;
399 undo[cursor++] = before[i];
400 }
401 }
402 history.push(undo);
403 }
404 void undo() {
405 if (history.isEmpty()) fail("nothing to undo");
406 int[] undo = history.pop();
407 for (int i = 1; i < undo.length; i += 2) s[undo[i]] = undo[i+1];
408 finished.subList(undo[0],finished.size()).clear();
409 }
410 String situation(int x) {

413 int serve = server();
414 boolean normal = tennis() && !tieBreak();

417 normal && s[G]+s[G+1] > games ? (x == serve ? "game point" : "break point") : "";
418 s = old;
419 finished.subList(count,finished.size()).clear();
420 return result;

514 List<String> execute(String line) {
515 String[] t = line.split(" ");
516 String command = t[0];
517 if (command.equals("MATCH")) {
518 if (matches.containsKey(t[1])) fail("duplicate match");
519 Match match = new Match(t);
520 matches.put(t[1],match);
521 return List.of();
522 }
523 if (command.equals("CHECKOUT")) {
524 List<String> answer = checkout(Integer.parseInt(t[1]),3);
525 return List.of("checkout "+t[1]+": "+(answer.isEmpty() ? "none" : String.join(" ",answer)));
526 }
527 if (command.equals("STANDINGS")) {
528 Map<String,int[]> totals = new HashMap<>();
529 for (Match match : matches.values()) {
530 if (match.over() && !match.sport.equals("BOWLING")) {
531 int winner = match.s[WINNER];
532 totals.computeIfAbsent(match.names[winner],name -> new int[2])[0]++;
533 totals.computeIfAbsent(match.names[1-winner],name -> new int[2])[1]++;
534 }
535 }
536 List<String> names = new ArrayList<>(totals.keySet());
537 names.sort(Comparator.<String>comparingInt(name -> totals.get(name)[0]).reversed().thenComparingInt(name -> totals.get(name)[1]).thenComparing(name -> name));
538 List<String> records = names.stream().map(name -> name+" "+totals.get(name)[0]+"-"+totals.get(name)[1]).toList();
539 return List.of("standings: "+(records.isEmpty() ? "none" : String.join(", ",records)));
540 }
541 Match match = matches.get(t[1]);
542 if (match == null) fail("unknown match");
543 if (command.equals("UNDO")) {
544 match.undo();
545 return List.of();
546 }
547 if (command.equals("SCORE")) return List.of(match.score());
548 if (command.equals("STATS")) return match.stats();
549 if (command.equals("RESULT")) return List.of(match.result());
550 boolean correct = switch (command) {
551 case "POINT" -> match.rally();
552 case "ROLL" -> match.sport.equals("BOWLING");
553 case "THROW" -> match.sport.equals("DARTS");
554 default -> match.sport.equals("VOLLEYBALL");
555 };
556 if (!correct) fail("wrong event");
557 if (command.equals("POINT")) {
558 for (char c : t[2].toCharArray()) match.event(command,String.valueOf(c),t);
559 } else if (command.equals("ROLL") || command.equals("THROW")) {
560 for (int i = 2; i < t.length; i++) match.event(command,t[i],t);
561 } else match.event(command,"",t);
562 return List.of();
563 }
