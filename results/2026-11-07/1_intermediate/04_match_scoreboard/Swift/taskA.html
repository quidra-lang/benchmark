29final class Scoreboard {
30 var matches: [String:Match] = [:]
31 var darts: [Dart] = []
32 var doubles: [Dart] = []
33 var dartIndex: [String:Int] = [:]
34 var cache: [Int:[String]] = [:]
35 init() {
36 for prefix in ["S","T","D"] {
37 for n in stride(from: 20,through: 1,by: -1) {
38 let multiplier = prefix == "S" ? 1 : (prefix == "T" ? 3 : 2)
39 let dart = Dart(name: prefix+String(n),value: n*multiplier,doubleOut: prefix == "D")
40 darts.append(dart)
41 if dart.doubleOut { doubles.append(dart) }
42 }
43 }
44 darts.append(Dart(name: "25",value: 25,doubleOut: false))
45 let bull = Dart(name: "BULL",value: 50,doubleOut: true)
46 darts.append(bull)
47 doubles.append(bull)
48 darts.append(Dart(name: "MISS",value: 0,doubleOut: false))
49 for (i,dart) in darts.enumerated() { dartIndex[dart.name] = i }
50 }
51 func checkout(_ score: Int, _ maximum: Int) -> [String] {
52 let key = score*4+maximum
53 if let answer = cache[key] { return answer }
54 func search() -> [String] {
55 for a in doubles where a.value == score { return [a.name] }
56 if maximum >= 2 {
57 for a in darts.prefix(62) {
58 for b in doubles where a.value+b.value == score { return [a.name,b.name] }
59 }
60 }
61 if maximum >= 3 {
62 for a in darts.prefix(62) {
63 for b in darts.prefix(62) {
64 for c in doubles where a.value+b.value+c.value == score { return [a.name,b.name,c.name] }
65 }
66 }
67 }
68 return []
69 }
70 let answer = search()
71 cache[key] = answer
72 return answer
73 }
74 func execute(_ line: String) throws -> [String] {
75 let t = line.split(separator: " ").map(String.init)
76 let command = t[0]
77 if command == "MATCH" {
78 guard matches[t[1]] == nil else { throw Problem(message: "duplicate match") }
79 let match = try Match(self,t)
80 matches[t[1]] = match
81 return []
82 }
83 if command == "CHECKOUT" {
84 let answer = checkout(Int(t[1])!,3)
85 return ["checkout "+t[1]+": "+(answer.isEmpty ? "none" : answer.joined(separator: " "))]
86 }
87 if command == "STANDINGS" {
88 var totals: [String:[Int]] = [:]
89 for match in matches.values where match.over() && match.sport != "BOWLING" {
90 let winner = match.s[WINNER]
91 totals[match.names[winner],default: [0,0]][0] += 1
92 totals[match.names[1-winner],default: [0,0]][1] += 1
93 }
94 let names = totals.keys.sorted {
95 if totals[$0]![0] != totals[$1]![0] { return totals[$0]![0] > totals[$1]![0] }
96 if totals[$0]![1] != totals[$1]![1] { return totals[$0]![1] < totals[$1]![1] }
97 return $0 < $1
98 }
99 let records = names.map { "\($0) \(totals[$0]![0])-\(totals[$0]![1])" }
100 return ["standings: "+(records.isEmpty ? "none" : records.joined(separator: ", "))]
101 }
102 guard let match = matches[t[1]] else { throw Problem(message: "unknown match") }
103 switch command {
104 case "UNDO": try match.undo()
105 case "SCORE": return [match.score()]
106 case "STATS": return match.stats()
107 case "RESULT": return [try match.result()]
108 default:
109 let correct: Bool
110 switch command {
111 case "POINT": correct = match.rally()
112 case "ROLL": correct = match.sport == "BOWLING"
113 case "THROW": correct = match.sport == "DARTS"
114 default: correct = match.sport == "VOLLEYBALL"
115 }
116 guard correct else { throw Problem(message: "wrong event") }
117 if command == "POINT" {
118 for c in t[2] { try match.event(command,String(c),t) }
119 } else if command == "ROLL" || command == "THROW" {
120 for value in t.dropFirst(2) { try match.event(command,value,t) }
121 } else { try match.event(command,"",t) }
122 }
123 return []
124 }
125}

127final class Match {
128 unowned let app: Scoreboard
129 let id: String
130 let sport: String
131 let names: [String]
132 var options: [String:String] = [:]
133 var initial = Array(repeating: Array(repeating: 0,count: 6),count: 2)
134 var s = Array(repeating: 0,count: 100)
135 var finished: [Finished] = []
136 var history: [[Int]] = []
137 init(_ owner: Scoreboard, _ t: [String]) throws {
138 app = owner
139 id = t[1]
140 sport = t[2]
141 names = [t[3],sport == "BOWLING" ? "" : t[4]]
142 var allowed: [String:Set<String>] = [:]
143 if sport != "BOWLING" {
144 allowed["FIRST"] = ["A","B"]
145 options["FIRST"] = "A"
146 }
147 if sport == "TENNIS" {
148 allowed["SETS"] = ["3","5"]
149 allowed["DEUCE"] = ["AD","NOAD"]
150 allowed["FINAL"] = ["TB7","TB10","ADV","SUPER"]
151 options["SETS"] = "3"
152 options["DEUCE"] = "AD"
153 options["FINAL"] = "TB7"
154 }
155 if sport == "TABLETENNIS" {
156 allowed["GAMES"] = ["5","7"]
157 options["GAMES"] = "5"
158 }
159 if sport == "DARTS" {
160 allowed["START"] = ["301","501","701"]
161 allowed["SETS"] = ["1","2","3","4","5"]
162 allowed["LEGS"] = ["1","2","3","4","5"]
163 options["START"] = "501"
164 options["SETS"] = "1"
165 options["LEGS"] = "1"
166 }
167 let firstOption = sport == "BOWLING" ? 4 : (sport == "VOLLEYBALL" ? 7 : 5)
168 var seen: Set<String> = []
169 for token in t.dropFirst(firstOption) {
170 let p = token.components(separatedBy: "=")
171 guard p.count == 2, let values = allowed[p[0]], values.contains(p[1]), seen.insert(p[0]).inserted else {
172 throw Problem(message: "invalid option")
173 }
174 options[p[0]] = p[1]
175 }
176 s[GS] = first()
177 s[SERVE] = first()
178 s[LEG] = 1
179 s[STANDING] = 10
180 if sport == "DARTS" {
181 s[R] = option("START")
182 s[R+1] = option("START")
183 s[TS] = option("START")
184 }
185 if sport == "VOLLEYBALL" {
186 for side in 0..<2 {
187 let lineup = t[5+side].split(separator: ",").map { Int($0)! }
188 for i in 0..<6 {
189 initial[side][i] = lineup[i]
190 s[LINEUP+side*6+i] = lineup[i]
191 }
192 }
193 }
194 }
195 func first() -> Int { options["FIRST",default: "A"] == "A" ? 0 : 1 }
196 func option(_ key: String) -> Int { Int(options[key]!)! }
197 func over() -> Bool { s[DONE] != 0 }
198 func tennis() -> Bool { sport == "TENNIS" }
199 func rally() -> Bool { sport != "DARTS" && sport != "BOWLING" }
200 func finalSet() -> Bool { tennis() && s[W] == option("SETS")/2 && s[W+1] == option("SETS")/2 }
201 func superTie() -> Bool { finalSet() && options["FINAL"] == "SUPER" }
202 func tieBreak() -> Bool { superTie() || (tennis() && s[G] == 6 && s[G+1] == 6 && !(finalSet() && options["FINAL"] == "ADV")) }
203 func server() -> Int {
204 if tennis() { return s[GS] ^ (tieBreak() ? (s[P]+s[P+1]+1)/2%2 : 0) }
205 if sport == "TABLETENNIS" {
206 let played = s[P]+s[P+1]
207 return first() ^ (finished.count%2) ^ (played < 20 ? played/2%2 : played%2)
208 }
209 return s[SERVE]
210 }
211 func normalWin(_ x: Int, _ extra: Int) -> Bool {
212 s[P+x]+extra >= 4 && (options["DEUCE"] == "NOAD" || s[P+x]+extra-s[P+1-x] >= 2)
213 }
214 func endSet(_ x: Int, _ score: Finished) {
215 finished.append(score)
216 s[W+x] += 1
217 let target: Int
218 switch sport {
219 case "TENNIS": target = option("SETS")/2+1
220 case "TABLETENNIS": target = option("GAMES")/2+1
221 case "VOLLEYBALL": target = 3
222 default: target = 2
223 }
224 if s[W+x] == target {
225 s[DONE] = 1
226 s[WINNER] = x
227 } else {
228 s[P] = 0
229 s[P+1] = 0
230 s[G] = 0
231 s[G+1] = 0
232 if sport == "VOLLEYBALL" {
233 for side in 0..<2 {
234 s[SUBS+side] = 0
235 s[TIMEOUTS+side] = 0
236 s.replaceSubrange((LINEUP+side*6)..<(LINEUP+side*6+6),with: initial[side])
237 }
238 s[SERVE] = first() ^ (finished.count%2)
239 }
240 }
241 }
242 func point(_ x: Int) {
243 let serve = server()
244 s[POINTS+x] += 1
245 s[SERVED+serve] += 1
246 if serve == x { s[SERVEWON+x] += 1 }
247 s[CURRENT+1-x] = 0
248 s[CURRENT+x] += 1
249 s[RUN+x] = max(s[RUN+x],s[CURRENT+x])
250 if tennis() {
251 let tie = tieBreak()
252 let superGame = superTie()
253 if !tie && normalWin(1-serve,1) {
254 s[BPO+1-serve] += 1
255 if x != serve { s[BPW+x] += 1 }
256 }
257 s[P+x] += 1
258 if tie {
259 let target = superGame || (finalSet() && options["FINAL"] == "TB10") ? 10 : 7
260 if s[P+x] >= target && s[P+x]-s[P+1-x] >= 2 {
261 let score: Finished
262 if superGame { score = Finished(a: s[P],b: s[P+1],type: 2) }
263 else {
264 s[G+x] += 1
265 score = Finished(a: s[G],b: s[G+1],type: 1,loser: s[P+1-x])
266 }
267 s[GS] ^= 1
268 endSet(x,score)
269 }
270 } else if normalWin(x,0) {
271 s[SERVICE+s[GS]] += 1
272 if x == s[GS] { s[HOLDS+x] += 1 }
273 else { s[BREAKS+x] += 1 }
274 s[G+x] += 1
275 s[P] = 0
276 s[P+1] = 0
277 s[GS] ^= 1
278 if s[G+x] >= 6 && s[G+x]-s[G+1-x] >= 2 { endSet(x,Finished(a: s[G],b: s[G+1])) }
279 }
280 return
281 }
282 s[P+x] += 1
283 if sport == "BADMINTON" { s[SERVE] = x }
284 if sport == "VOLLEYBALL" && s[SERVE] != x {
285 let start = LINEUP+x*6
286 let old = s[start]
287 for i in 0..<5 { s[start+i] = s[start+i+1] }
288 s[start+5] = old
289 s[SERVE] = x
290 }
291 let target = sport == "TABLETENNIS" ? 11 : (sport == "BADMINTON" ? 21 : (finished.count == 4 ? 15 : 25))
292 if (s[P+x] >= target && s[P+x]-s[P+1-x] >= 2) || (sport == "BADMINTON" && s[P+x] == 30) {
293 endSet(x,Finished(a: s[P],b: s[P+1]))
294 }
295 }
296 func completeTurn(_ x: Int) {
297 s[HIGH+x] = max(s[HIGH+x],s[TV])
298 if s[TV] >= 100 { s[TONS+x] += 1 }
299 if s[TV] == 180 { s[MAXIMUMS+x] += 1 }
300 }
301 func nextTurn() {
302 s[SERVE] ^= 1
303 s[TC] = 0
304 s[TV] = 0
305 s[TS] = s[R+s[SERVE]]
306 }
307 func dart(_ code: Int) {
308 let d = app.darts[code]
309 let x = s[SERVE]
310 let before = s[R+x]
311 if s[TC] == 0 { s[TS] = before }
312 if before == 50 || (before <= 40 && before%2 == 0) { s[ATTEMPTS+x] += 1 }
313 s[DARTS+x] += 1
314 s[TURN+s[TC]] = code
315 s[TC] += 1
316 s[TV] += d.value
317 let remaining = before-d.value
318 if remaining < 0 || remaining == 1 || (remaining == 0 && !d.doubleOut) {
319 s[SCORED+x] -= s[TV]-d.value
320 s[BUSTS+x] += 1
321 s[R+x] = s[TS]
322 nextTurn()
323 } else {
324 s[R+x] = remaining
325 s[SCORED+x] += d.value
326 if remaining == 0 {
327 s[HITS+x] += 1
328 completeTurn(x)
329 s[L+x] += 1
330 if s[L+x] == option("LEGS") {
331 finished.append(Finished(a: s[L],b: s[L+1]))
332 s[W+x] += 1
333 if s[W+x] == option("SETS") {
334 s[DONE] = 1
335 s[WINNER] = x
336 return
337 }
338 s[L] = 0
339 s[L+1] = 0
340 }
341 s[LEG] += 1
342 s[R] = option("START")
343 s[R+1] = option("START")
344 s[SERVE] = first() ^ ((s[LEG]-1)%2)
345 s[TC] = 0
346 s[TV] = 0
347 s[TS] = option("START")
348 } else if s[TC] == 3 {
349 completeTurn(x)
350 nextTurn()
351 }
352 }
353 }
354 func roll(_ pins: Int) {
355 s[ROLLS+s[ROLLCOUNT]] = pins
356 s[ROLLCOUNT] += 1
357 let r = s[FRAMEROLLS]
358 let frame = s[FRAME]
359 if r == 0 { s[FIRSTROLL] = pins }
360 if r == 1 { s[SECONDROLL] = pins }
361 s[STANDING] -= pins
362 let end = frame < 9 ? r == 1 || (r == 0 && pins == 10) :
363 r == 2 || (r == 1 && s[FIRSTROLL] != 10 && s[FIRSTROLL]+pins < 10)
364 if end {
365 if frame == 9 { s[DONE] = 1 }
366 else {
367 s[FRAME] += 1
368 s[FRAMEROLLS] = 0
369 s[STANDING] = 10
370 }
371 } else {
372 s[FRAMEROLLS] += 1
373 if frame == 9 && s[STANDING] == 0 { s[STANDING] = 10 }
374 }
375 }
376 func bowlingFrames() -> [BowlingFrame] {
377 var result: [BowlingFrame] = []
378 var cursor = 0
379 var cumulative = 0
380 for frame in 0..<10 {
381 if cursor >= s[ROLLCOUNT] { break }
382 let first = s[ROLLS+cursor]
383 let count = frame < 9 ? (first == 10 ? 1 : min(2,s[ROLLCOUNT]-cursor)) : min(3,s[ROLLCOUNT]-cursor)
384 var marks = ""
385 var standing = 10
386 var sum = 0
387 var fresh = true
388 for i in 0..<count {
389 let pins = s[ROLLS+cursor+i]
390 marks += pins == 10 && fresh ? "X" : (pins == standing ? "/" : (pins == 0 ? "-" : String(pins)))
391 sum += pins
392 standing -= pins
393 fresh = frame == 9 && standing == 0
394 if fresh { standing = 10 }
395 }
396 let score: Int?
397 if frame == 9 { score = over() ? sum : nil }
398 else if first == 10 && cursor+2 < s[ROLLCOUNT] { score = 10+s[ROLLS+cursor+1]+s[ROLLS+cursor+2] }
399 else if first != 10 && count == 2 {
400 score = sum < 10 ? sum : (cursor+2 < s[ROLLCOUNT] ? 10+s[ROLLS+cursor+2] : nil)
401 } else { score = nil }
402 var total: Int? = nil
403 if let score {
404 cumulative += score
405 total = cumulative
406 }
407 let open = count >= 2 && first != 10 && first+s[ROLLS+cursor+1] < 10
408 result.append(BowlingFrame(marks: marks,cumulative: total,open: open))
409 cursor += count
410 }
411 return result
412 }
413 func event(_ command: String, _ value: String, _ tokens: [String]) throws {
414 guard !over() else { throw Problem(message: "match over") }
415 var code = 0
416 var side = 0
417 var position = -1
418 if command == "ROLL" {
419 code = Int(value)!
420 guard code <= s[STANDING] else { throw Problem(message: "too many pins") }
421 }
422 if command == "THROW" {
423 guard let index = app.dartIndex[value] else { throw Problem(message: "invalid dart") }
424 code = index
425 }
426 if command == "SUB" || command == "TIMEOUT" {
427 side = tokens[2] == "A" ? 0 : 1
428 if command == "SUB" {
429 let from = Int(tokens[3])!
430 let to = Int(tokens[4])!
431 guard let found = s[(LINEUP+side*6)..<(LINEUP+side*6+6)].firstIndex(of: from) else {
432 throw Problem(message: "player not on court")
433 }
434 position = found-LINEUP-side*6
435 guard !(0..<6).contains(where: { s[LINEUP+side*6+$0] == to }) else { throw Problem(message: "player already on court") }
436 guard s[SUBS+side] < 6 else { throw Problem(message: "no substitutions left") }
437 } else if s[TIMEOUTS+side] == 2 { throw Problem(message: "no timeouts left") }
438 }
439 let before = s
440 let count = finished.count
441 switch command {
442 case "POINT": point(value == "A" ? 0 : 1)
443 case "ROLL": roll(code)
444 case "THROW": dart(code)
445 case "SUB":
446 s[LINEUP+side*6+position] = Int(tokens[4])!
447 s[SUBS+side] += 1
448 case "TIMEOUT": s[TIMEOUTS+side] += 1
449 default: preconditionFailure("Unreachable event")
450 }
451 var undo = [count]
452 for i in s.indices where s[i] != before[i] {
453 undo.append(i)
454 undo.append(before[i])
455 }
456 history.append(undo)
457 }
458 func undo() throws {
459 guard let undo = history.popLast() else { throw Problem(message: "nothing to undo") }
460 for i in stride(from: 1,to: undo.count,by: 2) { s[undo[i]] = undo[i+1] }
461 if finished.count > undo[0] { finished.removeLast(finished.count-undo[0]) }
462 }
463 func situation(_ x: Int) -> String {
464 let old = s
465 let count = finished.count
466 let games = s[G]+s[G+1]
467 let serve = server()
468 let normal = tennis() && !tieBreak()
469 point(x)
470 let result: String
471 if over() { result = "match point" }
472 else if finished.count > count { result = tennis() || sport == "VOLLEYBALL" ? "set point" : "game point" }
473 else if normal && s[G]+s[G+1] > games { result = x == serve ? "game point" : "break point" }
474 else { result = "" }
475 s = old
476 if finished.count > count { finished.removeLast(finished.count-count) }
477 return result
478 }
479 func finishedScore(_ f: Finished, _ first: Int) -> String {
480 let pair = first == 0 ? "\(f.a)-\(f.b)" : "\(f.b)-\(f.a)"
481 return f.type == 2 ? "["+pair+"]" : pair+(f.type == 1 ? "(\(f.loser))" : "")
482 }
483 func pair(_ offset: Int) -> String { "\(s[offset])-\(s[offset+1])" }
484 func score() -> String {
485 if sport == "BOWLING" {
486 let frames = bowlingFrames().map { $0.marks+($0.cumulative.map { "(\($0))" } ?? "") }
487 return id+": "+names[0]+(frames.isEmpty ? "" : " "+frames.joined(separator: " "))
488 }
489 let serve = over() ? -1 : server()
490 var items: [String] = []
491 if sport == "DARTS" {
492 items += ["sets",pair(W),"legs",pair(L),"score",pair(R)]
493 if !over() {
494 if s[TC] > 0 {
495 items.append("turn")
496 for i in 0..<s[TC] { items.append(app.darts[s[TURN+i]].name) }
497 }
498 let check = app.checkout(s[R+s[SERVE]],3-s[TC])
499 if !check.isEmpty {
500 items.append("checkout")
501 items += check
502 }
503 }
504 } else {
505 items += finished.map { finishedScore($0,0) }
506 if !over() {
507 if tennis() {
508 if superTie() { items.append("["+pair(P)+"]") }
509 else {
510 items.append(pair(G))
511 if tieBreak() { items.append(pair(P)) }
512 else if options["DEUCE"] == "AD" && s[P] >= 3 && s[P+1] >= 3 {
513 items.append(s[P] == s[P+1] ? "40-40" : (s[P] > s[P+1] ? "AD-40" : "40-AD"))
514 } else {
515 let points = ["0","15","30","40"]
516 items.append(points[s[P]]+"-"+points[s[P+1]])
517 }
518 }
519 } else {
520 items.append(pair(P))
521 if sport == "VOLLEYBALL" { items += ["server",String(s[LINEUP+s[SERVE]*6])] }
522 }
523 for x in 0..<2 {
524 let situation = situation(x)
525 if !situation.isEmpty { items.append("("+situation+" "+names[x]+")") }
526 }
527 }
528 }
529 return id+": "+names[0]+(serve == 0 ? "*" : "")+" vs "+names[1]+(serve == 1 ? "*" : "")+" "+items.joined(separator: " ")
530 }
531 func stats() -> [String] {
532 if sport == "BOWLING" {
533 let frames = bowlingFrames()
534 let strikes = frames.reduce(0) { $0+$1.marks.filter { $0 == "X" }.count }
535 let spares = frames.reduce(0) { $0+$1.marks.filter { $0 == "/" }.count }
536 let open = frames.filter { $0.open }.count
537 let old = s
538 while !over() { roll(s[STANDING]) }
539 let maximum = bowlingFrames().last!.cumulative!
540 s = old
541 return ["\(id) \(names[0]): strikes \(strikes) spares \(spares) open \(open) max \(maximum)"]
542 }
543 return (0..<2).map { x in
544 var line = id+" "+names[x]+": "
545 if sport == "DARTS" {
546 let average = s[DARTS+x] == 0 ? 0 : s[SCORED+x]*300/s[DARTS+x]
547 let formatted = String(format: "%ld.%02ld",locale: Locale(identifier: "en_US_POSIX"),average/100,average%100)
548 line += "darts \(s[DARTS+x]) average "+formatted+
549 " busts \(s[BUSTS+x]) high \(s[HIGH+x]) tons \(s[TONS+x]) 180s \(s[MAXIMUMS+x]) doubles \(s[HITS+x])/\(s[ATTEMPTS+x])"
550 } else {
551 line += "points \(s[POINTS+x]) serve \(s[SERVEWON+x])/\(s[SERVED+x]) run \(s[RUN+x])"
552 if tennis() { line += " holds \(s[HOLDS+x])/\(s[SERVICE+x]) breaks \(s[BREAKS+x]) break points \(s[BPW+x])/\(s[BPO+x])" }
553 if sport == "VOLLEYBALL" {
554 line += " subs \(s[SUBS+x]) timeouts \(s[TIMEOUTS+x]) lineup "+(0..<6).map { String(s[LINEUP+x*6+$0]) }.joined(separator: ",")
555 }
556 }
557 return line
558 }
559 }
560 func result() throws -> String {
561 guard over() else { throw Problem(message: "match not finished") }
562 if sport == "BOWLING" { return id+": "+names[0]+" \(bowlingFrames().last!.cumulative!)" }
563 let winner = s[WINNER]
564 return id+": "+names[winner]+" def. "+names[1-winner]+" "+finished.map { finishedScore($0,winner) }.joined(separator: " ")
565 }
