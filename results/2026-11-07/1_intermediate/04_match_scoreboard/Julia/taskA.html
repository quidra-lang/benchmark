4const OPTION_RULES = Dict(

13 sport::String
14 names::Vector{String}

21score_error(message) = throw(ArgumentError(message))
22side(text) = text == "A" ? 1 : 2
23other(player) = 3-player
24pair_text(values) = string(values[1],'-',values[2])
25average_text(value) = @sprintf("%d.%02d",div(value,100),value%100)
26option_int(m,key) = parse(Int,m.options[key])

28function checkout_tables()
29 candidates = Tuple{String,Int}[]
30 doubles = Tuple{String,Int}[]
31 darts = Dict("25"=>(25,false),"BULL"=>(50,true),"MISS"=>(0,false))
32 for (prefix,multiplier) in (("S",1),("T",3),("D",2))
33 for value in 20:-1:1
34 code = prefix*string(value)
35 push!(candidates,(code,value*multiplier))
36 darts[code] = (value*multiplier,prefix == "D")
37 prefix == "D" && push!(doubles,(code,value*multiplier))
38 end
39 end
40 append!(candidates,[("25",25),("BULL",50)])
41 push!(doubles,("BULL",50))
42 tables = [Dict{Int,Vector{String}}() for i in 1:3]
43 for (last,value) in doubles
44 !haskey(tables[1],value) && (tables[1][value] = [last])
45 end
46 for (first,a) in candidates
47 for (last,b) in doubles
48 !haskey(tables[2],a+b) && (tables[2][a+b] = [first,last])
49 end
50 end
51 for (first,a) in candidates
52 for (second,b) in candidates
53 for (last,c) in doubles
54 !haskey(tables[3],a+b+c) && (tables[3][a+b+c] = [first,second,last])
55 end
56 end
57 end
58 return darts,tables
59end

64function checkout(score,maximum=3)
65 for count in 1:maximum
66 haskey(CHECKOUTS[count],score) && return CHECKOUTS[count][score]
67 end
68 return String[]
69end

71function create_match(tokens)
72 id,sport = tokens[2:3]
73 names = sport == "BOWLING" ? tokens[4:4] : tokens[4:5]
74 initial = sport == "VOLLEYBALL" ? [parse.(Int,split(tokens[i],',')) for i in 6:7] : Vector{Int}[]
75 if sport == "BOWLING"
76 first_option = 5
77 elseif sport == "VOLLEYBALL"
78 first_option = 8
79 else
80 first_option = 6
81 end
82 options = Dict(key=>values[1] for (key,values) in OPTION_RULES[sport])
83 seen = Set{String}()
84 for token in tokens[first_option:end]
85 parts = split(token,'=';keepempty=true)
86 length(parts) == 2 || score_error("invalid option")
87 key,value = String.(parts)
88 haskey(OPTION_RULES[sport],key) && value in OPTION_RULES[sport][key] && !(key in seen) || score_error("invalid option")
89 push!(seen,key)
90 options[key] = value
91 end
92 first = haskey(options,"FIRST") ? side(options["FIRST"]) : 1
93 stats = Dict(key=>[0,0] for key in (:points,:served,:servewins,:runmax,:holds,:games,:breaks,:opportunities,:converted,:darts,:scored,:busts,:high,:tons,:maximums,:attempts,:hits))
94 state = Dict{Symbol,Any}(:p=>[0,0],:g=>[0,0],:won=>[0,0],:server=>first,:winner=>0,
95 :completed=>NTuple{4,Int}[],:stats=>stats,:last=>0,:run=>0)
96 if sport == "VOLLEYBALL"
97 state[:lineups] = deepcopy(initial)
98 state[:subs],state[:timeouts] = [0,0],[0,0]
99 elseif sport == "BOWLING"
100 state[:frames] = [Int[] for i in 1:10]
101 state[:marks] = [String[] for i in 1:10]
102 state[:frame],state[:standing],state[:fresh] = 1,10,true
103 elseif sport == "DARTS"
104 start = parse(Int,options["START"])
105 state[:legs],state[:remaining] = [0,0],[start,start]
106 state[:turn],state[:turnscore],state[:turnstart],state[:legnumber] = String[],0,start,0
107 end
108 return ScoreMatch(id,sport,names,options,initial,state,Any[])
109end

111function tennis_mode(m)
112 s = m.state
113 final = s[:won][1] == s[:won][2] == div(option_int(m,"SETS"),2)
114 final && m.options["FINAL"] == "SUPER" && return "SUPER"
115 s[:g] == [6,6] && !(final && m.options["FINAL"] == "ADV") && return "TB"
116 return "NORMAL"
117end

119function rally_server(m)
120 s = m.state
121 if m.sport == "TENNIS"
122 if tennis_mode(m) != "NORMAL" && isodd(div(sum(s[:p])+1,2))
123 return other(s[:server])
124 end
125 elseif m.sport == "TABLETENNIS"
126 first = side(m.options["FIRST"])
127 isodd(length(s[:completed])) && (first = other(first))
128 total = sum(s[:p])
129 flip = total < 20 ? isodd(div(total,2)) : isodd(total)
130 return flip ? other(first) : first
131 end
132 return s[:server]
133end

135function rally_outcome(m,winner)
136 s = m.state
137 points = copy(s[:p])
138 points[winner] += 1
139 loser = other(winner)
140 game = false
141 unit = false
142 if m.sport == "TENNIS"
143 mode = tennis_mode(m)
144 if mode == "NORMAL"
145 game = points[winner] >= 4 && (m.options["DEUCE"] == "NOAD" || points[winner]-points[loser] >= 2)
146 games = s[:g][winner]+1
147 unit = game && games >= 6 && games-s[:g][loser] >= 2
148 else
149 final = s[:won][1] == s[:won][2] == div(option_int(m,"SETS"),2)
150 target = mode == "SUPER" || (final && m.options["FINAL"] == "TB10") ? 10 : 7
151 unit = points[winner] >= target && points[winner]-points[loser] >= 2
152 end
153 needed = div(option_int(m,"SETS"),2)+1
154 elseif m.sport == "TABLETENNIS"
155 unit = points[winner] >= 11 && points[winner]-points[loser] >= 2
156 needed = div(option_int(m,"GAMES"),2)+1
157 elseif m.sport == "BADMINTON"
158 unit = points[winner] == 30 || (points[winner] >= 21 && points[winner]-points[loser] >= 2)
159 needed = 2
160 else
161 target = length(s[:completed]) == 4 ? 15 : 25
162 unit = points[winner] >= target && points[winner]-points[loser] >= 2
163 needed = 3
164 end
165 return game,unit,unit && s[:won][winner]+1 == needed
166end

168function point_event!(m,winner)
169 s = m.state
170 stats = s[:stats]
171 server = rally_server(m)
172 game,unit,finished = rally_outcome(m,winner)
173 mode = m.sport == "TENNIS" ? tennis_mode(m) : ""
174 opportunity = false
175 if mode == "NORMAL"
176 opportunity = rally_outcome(m,other(server))[1]
177 end
178 record = [1,winner,server,Int(opportunity),Int(game),s[:p]...,s[:g]...,s[:won]...,s[:server],length(s[:completed]),s[:last],s[:run],stats[:runmax][winner],0,0]
179 stats[:points][winner] += 1
180 stats[:served][server] += 1
181 winner == server && (stats[:servewins][server] += 1)
182 s[:run] = s[:last] == winner ? s[:run]+1 : 1
183 s[:last] = winner
184 stats[:runmax][winner] = max(stats[:runmax][winner],s[:run])
185 if opportunity
186 stats[:opportunities][other(server)] += 1
187 winner != server && (stats[:converted][winner] += 1)
188 end
189 s[:p][winner] += 1
190 if m.sport == "TENNIS"
191 if game
192 stats[:games][server] += 1
193 if winner == server
194 stats[:holds][winner] += 1
195 else
196 stats[:breaks][winner] += 1
197 end
198 s[:g][winner] += 1
199 s[:server] = other(s[:server])
200 elseif unit
201 s[:server] = other(s[:server])
202 end
203 if unit
204 if mode == "SUPER"
205 push!(s[:completed],(s[:p][1],s[:p][2],2,0))
206 elseif mode == "TB"
207 s[:g][winner] += 1
208 push!(s[:completed],(s[:g][1],s[:g][2],1,s[:p][other(winner)]))
209 else
210 push!(s[:completed],(s[:g][1],s[:g][2],0,0))
211 end
212 end
213 elseif m.sport == "BADMINTON"
214 s[:server] = winner
215 elseif m.sport == "VOLLEYBALL" && winner != server
216 s[:server] = winner
217 lineup = s[:lineups][winner]
218 s[:lineups][winner] = vcat(lineup[2:end],lineup[1])
219 record[18] = 1
220 end
221 if unit
222 m.sport != "TENNIS" && push!(s[:completed],(s[:p][1],s[:p][2],0,0))
223 s[:won][winner] += 1
224 if finished
225 s[:winner] = winner
226 else
227 if m.sport == "VOLLEYBALL"
228 record[17] = 1
229 append!(record,vcat(s[:lineups][1],s[:lineups][2],s[:subs],s[:timeouts]))
230 s[:lineups] = deepcopy(m.initial)
231 s[:subs],s[:timeouts] = [0,0],[0,0]
232 first = side(m.options["FIRST"])
233 s[:server] = iseven(length(s[:completed])) ? first : other(first)
234 end
235 s[:p],s[:g] = [0,0],[0,0]
236 end
237 elseif game
238 s[:p] = [0,0]
239 end
240 push!(m.history,Int32.(record))
241end

243function roll_event!(s,pins)
244 pins <= s[:standing] || score_error("too many pins")
245 frame = s[:frame]
246 if s[:fresh] && pins == 10
247 mark = "X"
248 elseif pins == s[:standing]
249 mark = "/"
250 else
251 mark = pins == 0 ? "-" : string(pins)
252 end
253 push!(s[:frames][frame],pins)
254 push!(s[:marks][frame],mark)
255 s[:standing] -= pins
256 s[:fresh] = false
257 rolls = s[:frames][frame]
258 if frame < 10
259 if (length(rolls) == 1 && pins == 10) || length(rolls) == 2
260 s[:frame] += 1
261 s[:standing],s[:fresh] = 10,true
262 end
263 else
264 complete = length(rolls) == 3 || (length(rolls) == 2 && rolls[1] != 10 && sum(rolls) < 10)
265 if complete
266 s[:winner] = 1
267 elseif s[:standing] == 0
268 s[:standing],s[:fresh] = 10,true
269 end
270 end
271end

273function bowling_scores(s)
274 scores = Union{Nothing,Int}[nothing for i in 1:10]
275 for frame in 1:10
276 rolls = s[:frames][frame]
277 isempty(rolls) && continue
278 later = [pin for i in frame+1:10 for pin in s[:frames][i]]
279 if frame == 10
280 s[:winner] != 0 && (scores[frame] = sum(rolls))
281 elseif rolls[1] == 10
282 length(later) >= 2 && (scores[frame] = 10+sum(later[1:2]))
283 elseif length(rolls) == 2
284 if sum(rolls) == 10
285 !isempty(later) && (scores[frame] = 10+later[1])
286 else
287 scores[frame] = sum(rolls)
288 end
289 end
290 end
291 return scores
292end

294function complete_turn!(s,player)
295 score = s[:turnscore]
296 stats = s[:stats]
297 stats[:high][player] = max(stats[:high][player],score)
298 score >= 100 && (stats[:tons][player] += 1)
299 score == 180 && (stats[:maximums][player] += 1)
300end

302function next_turn!(s)

304 s[:turn],s[:turnscore] = String[],0
305 s[:turnstart] = s[:remaining][s[:server]]

308function dart_event!(m,code)
309 haskey(DARTS,code) || score_error("invalid dart")
310 s = m.state
311 stats = s[:stats]
312 player = s[:server]
313 value,double = DARTS[code]
314 before = s[:remaining][player]
315 stats[:darts][player] += 1
316 (before == 50 || (before <= 40 && iseven(before))) && (stats[:attempts][player] += 1)
317 remaining = before-value
318 if remaining < 0 || remaining == 1 || (remaining == 0 && !double)
319 stats[:busts][player] += 1
320 stats[:scored][player] -= s[:turnscore]
321 s[:remaining][player] = s[:turnstart]
322 next_turn!(s)
323 return
324 end
325 stats[:scored][player] += value
326 s[:remaining][player] = remaining
327 s[:turnscore] += value
328 push!(s[:turn],code)
329 if remaining == 0
330 stats[:hits][player] += 1
331 complete_turn!(s,player)
332 s[:legs][player] += 1
333 set_over = s[:legs][player] == option_int(m,"LEGS")
334 if set_over
335 push!(s[:completed],(s[:legs][1],s[:legs][2],0,0))
336 s[:won][player] += 1
337 end
338 if set_over && s[:won][player] == option_int(m,"SETS")
339 s[:winner] = player
340 return
341 end
342 set_over && (s[:legs] = [0,0])
343 s[:legnumber] += 1
344 first = side(m.options["FIRST"])
345 s[:server] = iseven(s[:legnumber]) ? first : other(first)
346 start = option_int(m,"START")
347 s[:remaining] = [start,start]
348 s[:turn],s[:turnscore],s[:turnstart] = String[],0,start
349 elseif length(s[:turn]) == 3
350 complete_turn!(s,player)
351 next_turn!(s)
352 end
353end

355function undo_event!(m)
356 isempty(m.history) && score_error("nothing to undo")
357 record = pop!(m.history)
358 if record isa Dict
359 m.state = record
360 return
361 end
362 data = Int.(record)
363 s = m.state
364 if data[1] == 2
365 s[:lineups][data[2]][data[3]] = data[4]
366 s[:subs][data[2]] -= 1
367 elseif data[1] == 3
368 s[:timeouts][data[2]] -= 1
369 else
370 winner,server = data[2:3]
371 stats = s[:stats]
372 stats[:points][winner] -= 1
373 stats[:served][server] -= 1
374 winner == server && (stats[:servewins][server] -= 1)
375 if data[4] == 1
376 stats[:opportunities][other(server)] -= 1
377 winner != server && (stats[:converted][winner] -= 1)
378 end
379 if data[5] == 1
380 stats[:games][server] -= 1
381 if winner == server
382 stats[:holds][winner] -= 1
383 else
384 stats[:breaks][winner] -= 1
385 end
386 end
387 if data[17] == 1
388 s[:lineups] = [data[19:24],data[25:30]]
389 s[:subs],s[:timeouts] = data[31:32],data[33:34]
390 end
391 if data[18] == 1
392 lineup = s[:lineups][winner]
393 s[:lineups][winner] = vcat(lineup[end],lineup[1:end-1])
394 end
395 s[:p],s[:g],s[:won] = data[6:7],data[8:9],data[10:11]
396 s[:server],s[:winner] = data[12],0
397 resize!(s[:completed],data[13])
398 s[:last],s[:run] = data[14],data[15]
399 stats[:runmax][winner] = data[16]
400 end
401end

403function event!(m,command,args)
404 correct = (command == "POINT" && m.sport in RALLY_SPORTS) || (command == "ROLL" && m.sport == "BOWLING") ||
405 (command == "THROW" && m.sport == "DARTS") || (command in ("SUB","TIMEOUT") && m.sport == "VOLLEYBALL")
406 correct || score_error("wrong event")
407 if command == "POINT"
408 events = (string(c) for c in args[1])
409 elseif command in ("SUB","TIMEOUT")
410 events = [""]
411 else
412 events = args
413 end
414 for value in events
415 m.state[:winner] == 0 || score_error("match over")
416 if command == "POINT"
417 point_event!(m,side(value))
418 elseif command == "ROLL"
419 old = deepcopy(m.state)
420 roll_event!(m.state,parse(Int,value))
421 push!(m.history,old)
422 elseif command == "THROW"
423 old = deepcopy(m.state)
424 dart_event!(m,value)
425 push!(m.history,old)
426 elseif command == "SUB"
427 player = side(args[1])
428 outgoing,incoming = parse.(Int,args[2:3])
429 lineup = m.state[:lineups][player]
430 position = findfirst(==(outgoing),lineup)
431 position !== nothing || score_error("player not on court")
432 incoming in lineup && score_error("player already on court")
433 m.state[:subs][player] < 6 || score_error("no substitutions left")
434 push!(m.history,Int32[2,player,position,outgoing])
435 lineup[position] = incoming
436 m.state[:subs][player] += 1
437 else
438 player = side(args[1])
439 m.state[:timeouts][player] < 2 || score_error("no timeouts left")
440 push!(m.history,Int32[3,player])
441 m.state[:timeouts][player] += 1
442 end
443 end
444end

446function completed_text(record,winner=1)
447 a,b,kind,extra = record
448 if winner == 2
449 a,b = b,a
450 end
451 text = string(a,'-',b)
452 kind == 2 && return "["*text*"]"
453 return kind == 1 ? text*"($extra)" : text
454end

456function situation(m,player)
457 game,unit,finished = rally_outcome(m,player)
458 finished && return "match point"
459 if unit
460 return m.sport in ("TENNIS","VOLLEYBALL") ? "set point" : "game point"
461 end
462 if game
463 return player == rally_server(m) ? "game point" : "break point"
464 end
465 return ""
466end

468function score_line(m)
469 s = m.state
470 if m.sport == "BOWLING"
471 parts = [m.id*": "*m.names[1]]
472 scores = bowling_scores(s)
473 cumulative = 0
474 for frame in 1:10
475 isempty(s[:frames][frame]) && break
476 text = join(s[:marks][frame])
477 if scores[frame] !== nothing
478 cumulative += scores[frame]
479 text *= "($cumulative)"
480 end
481 push!(parts,text)
482 end
483 return join(parts," ")
484 end
485 names = copy(m.names)
486 if s[:winner] == 0
487 player = m.sport in RALLY_SPORTS ? rally_server(m) : s[:server]
488 names[player] *= "*"
489 end
490 parts = [m.id*": "*names[1]*" vs "*names[2]]
491 if m.sport == "DARTS"
492 append!(parts,["sets",pair_text(s[:won]),"legs",pair_text(s[:legs]),"score",pair_text(s[:remaining])])
493 if s[:winner] == 0
494 if !isempty(s[:turn])
495 push!(parts,"turn")
496 append!(parts,s[:turn])
497 end
498 answer = checkout(s[:remaining][s[:server]],3-length(s[:turn]))
499 if !isempty(answer)
500 push!(parts,"checkout")
501 append!(parts,answer)
502 end
503 end
504 else
505 append!(parts,completed_text.(s[:completed]))
506 if s[:winner] == 0
507 if m.sport == "TENNIS"
508 mode = tennis_mode(m)
509 if mode == "SUPER"
510 push!(parts,"["*pair_text(s[:p])*"]")
511 else
512 push!(parts,pair_text(s[:g]))
513 if mode == "TB"
514 push!(parts,pair_text(s[:p]))
515 elseif m.options["DEUCE"] == "AD" && minimum(s[:p]) >= 3
516 if s[:p][1] == s[:p][2]
517 push!(parts,"40-40")
518 else
519 push!(parts,s[:p][1] > s[:p][2] ? "AD-40" : "40-AD")
520 end
521 else
522 labels = ["0","15","30","40"]
523 push!(parts,labels[s[:p][1]+1]*"-"*labels[s[:p][2]+1])
524 end
525 end
526 else
527 push!(parts,pair_text(s[:p]))
528 if m.sport == "VOLLEYBALL"
529 append!(parts,["server",string(s[:lineups][s[:server]][1])])
530 end
531 end
532 for player in 1:2
533 label = situation(m,player)
534 !isempty(label) && push!(parts,"($label $(m.names[player]))")
535 end
536 end
537 end
538 return join(parts," ")
539end

541function stats_lines(m)
542 s = m.state
543 stats = s[:stats]
544 if m.sport == "BOWLING"
545 marks = [mark for frame in s[:marks] for mark in frame]
546 open = count(rolls -> length(rolls) >= 2 && rolls[1] != 10 && sum(rolls[1:2]) < 10,s[:frames])
547 future = deepcopy(s)
548 while future[:winner] == 0
549 roll_event!(future,future[:standing])
550 end
551 maximum = sum(bowling_scores(future))
552 return ["$(m.id) $(m.names[1]): strikes $(count(==("X"),marks)) spares $(count(==("/"),marks)) open $open max $maximum"]
553 end
554 output = String[]
555 for player in 1:2
556 prefix = "$(m.id) $(m.names[player]): "
557 if m.sport == "DARTS"
558 darts = stats[:darts][player]
559 average = darts == 0 ? 0 : div(stats[:scored][player]*300,darts)
560 text = "darts $darts average $(average_text(average)) busts $(stats[:busts][player]) high $(stats[:high][player]) tons $(stats[:tons][player]) 180s $(stats[:maximums][player]) doubles $(stats[:hits][player])/$(stats[:attempts][player])"
561 else
562 text = "points $(stats[:points][player]) serve $(stats[:servewins][player])/$(stats[:served][player]) run $(stats[:runmax][player])"
563 if m.sport == "TENNIS"
564 text *= " holds $(stats[:holds][player])/$(stats[:games][player]) breaks $(stats[:breaks][player]) break points $(stats[:converted][player])/$(stats[:opportunities][player])"
565 elseif m.sport == "VOLLEYBALL"
566 text *= " subs $(s[:subs][player]) timeouts $(s[:timeouts][player]) lineup $(join(s[:lineups][player],","))"
567 end
568 end
569 push!(output,prefix*text)
570 end
571 return output
572end

583function execute(line,matches)
584 tokens = String.(split(line,' '))
585 command = tokens[1]
586 if command == "MATCH"
587 haskey(matches,tokens[2]) && score_error("duplicate match")
588 m = create_match(tokens)
589 matches[m.id] = m
590 return String[]
591 elseif command == "CHECKOUT"
592 answer = checkout(parse(Int,tokens[2]))
593 return ["checkout $(tokens[2]): "*(isempty(answer) ? "none" : join(answer," "))]
594 elseif command == "STANDINGS"
595 totals = Dict{String,Vector{Int}}()
596 for m in values(matches)
597 if m.sport != "BOWLING" && m.state[:winner] != 0
598 winner = m.state[:winner]
599 for player in 1:2
600 record = get!(totals,m.names[player],[0,0])
601 record[player == winner ? 1 : 2] += 1
602 end
603 end
604 end
605 names = sort(collect(keys(totals));by=name -> (-totals[name][1],totals[name][2],name))
606 return ["standings: "*(isempty(names) ? "none" : join(["$name $(totals[name][1])-$(totals[name][2])" for name in names],", "))]
607 end
608 haskey(matches,tokens[2]) || score_error("unknown match")
609 m = matches[tokens[2]]
610 if command in ("POINT","ROLL","THROW","SUB","TIMEOUT")
611 event!(m,command,tokens[3:end])
612 elseif command == "UNDO"
613 undo_event!(m)
