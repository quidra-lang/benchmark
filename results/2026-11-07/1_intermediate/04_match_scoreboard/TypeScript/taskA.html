33const darts = new Map<string, Dart>();
34for (let n = 1; n <= 20; n++) {
35 darts.set("S" + n, { value: n, double: false });

43const candidatesB: string[] = [];
44for (const prefix of ["S", "T", "D"]) {
45 for (let n = 20; n >= 1; n--) {
46 candidatesA.push(prefix + n);

48 candidatesB.push(prefix + n);

54const checkoutCache = new Map<string, string[] | null>();

56function checkout(score: number, maximum: number = 3): string[] | null {
57 const key = score + ":" + maximum;
58 if (checkoutCache.has(key)) {
59 return checkoutCache.get(key)!;
60 }
61 let result: string[] | null = null;
62 if (maximum >= 1) {
63 const last = candidatesB.find(dart => darts.get(dart)!.value === score);
64 if (last) {
65 result = [last];
66 }
67 }
68 if (!result && maximum >= 2) {
69 for (const first of candidatesA) {
70 const last = candidatesB.find(dart => darts.get(first)!.value + darts.get(dart)!.value === score);
71 if (last) {
72 result = [first, last];
73 break;
74 }
75 }
76 }
77 if (!result && maximum >= 3) {
78 outer: for (const first of candidatesA) {
79 for (const second of candidatesA) {
80 const wanted = score - darts.get(first)!.value - darts.get(second)!.value;
81 const last = candidatesB.find(dart => darts.get(dart)!.value === wanted);
82 if (last) {
83 result = [first, second, last];
84 break outer;
85 }
86 }
87 }
88 }
89 checkoutCache.set(key, result);
90 return result;
91}

108 id: string;
109 sport: string;
110 names: string[];
111 options: Record<string, string>;
112 initialLineups: number[][];
113 first: number;

118 constructor(args: string[]) {
119 this.id = args[0];
120 this.sport = args[1];
121 this.names = this.sport === "BOWLING" ? [args[2]] : [args[2], args[3]];
122 this.initialLineups = this.sport === "VOLLEYBALL" ?
123 [args[4].split(",").map(Number), args[5].split(",").map(Number)] : [[], []];
124 const optionStart = this.sport === "BOWLING" ? 3 : this.sport === "VOLLEYBALL" ? 6 : 4;
125 this.options = { ...defaults[this.sport] };
126 const seen = new Set<string>();
127 for (const option of args.slice(optionStart)) {
128 const parts = option.split("=");
129 if (parts.length !== 2 || !Object.hasOwn(choices[this.sport], parts[0]) ||
130 !choices[this.sport][parts[0]].includes(parts[1]) || seen.has(parts[0])) {
131 fail("invalid option");
132 }
133 seen.add(parts[0]);
134 this.options[parts[0]] = parts[1];
135 }
136 this.first = this.options.FIRST === "B" ? 1 : 0;
137 const start = Number(this.options.START ?? 501);
138 this.state = {
139 done: false, winner: -1, sets: [0, 0], legs: [0, 0], points: [0, 0], games: [0, 0],
140 finished: [], gameServer: this.first, server: this.first,
141 lineups: this.initialLineups.map(lineup => [...lineup]), subs: [0, 0], timeouts: [0, 0],
142 rally: Array.from({ length: 2 }, () => ({
143 points: 0, served: 0, serveWon: 0, run: 0, current: 0,
144 holds: 0, serviceGames: 0, breaks: 0, opportunities: 0, conversions: 0
145 })),
146 frames: Array.from({ length: 10 }, () => []), marks: Array.from({ length: 10 }, () => []),
147 frame: 0, standing: 10, remaining: [start, start], player: this.first, legNumber: 0,
148 turn: [], turnStart: start, turnScore: 0,
149 darts: Array.from({ length: 2 }, () => ({
150 darts: 0, points: 0, busts: 0, high: 0, tons: 0, maximums: 0, attempts: 0, hits: 0
151 }))
152 };
153 this.initialState = structuredClone(this.state);
154 }

156 rallySport(): boolean {
157 return ["TENNIS", "TABLETENNIS", "BADMINTON", "VOLLEYBALL"].includes(this.sport);

160 finalSet(s: State): boolean {
161 const target = Math.floor(Number(this.options.SETS) / 2);
162 return s.sets[0] === target && s.sets[1] === target;

165 tennisKind(s: State): string {
166 if (this.finalSet(s) && this.options.FINAL === "SUPER") {
167 return "super";
168 }
169 if (s.games[0] === 6 && s.games[1] === 6 && !(this.finalSet(s) && this.options.FINAL === "ADV")) {
170 return "tie";
171 }
172 return "normal";

175 server(s: State): number {
176 if (this.sport === "TENNIS") {
177 if (this.tennisKind(s) === "normal") {
178 return s.gameServer;
179 }
180 const k = s.points[0] + s.points[1] + 1;
181 return s.gameServer ^ (Math.floor(k / 2) % 2);
182 }
183 if (this.sport === "TABLETENNIS") {
184 const first = this.first ^ (s.finished.length % 2);
185 const t = s.points[0] + s.points[1];
186 return first ^ (t < 20 ? Math.floor(t / 2) % 2 : t % 2);
187 }
188 return s.server;
189 }

191 normalGameWins(points: number[], winner: number): boolean {
192 return points[winner] >= 4 && (this.options.DEUCE === "NOAD" || points[winner] - points[1 - winner] >= 2);
193 }

195 finishRallyUnit(s: State, winner: number, record: Finished, target: number): void {
196 s.finished.push(record);
197 s.sets[winner]++;
198 if (s.sets[winner] === target) {
199 s.done = true;
200 s.winner = winner;
201 } else {
202 s.points = [0, 0];
203 s.games = [0, 0];
204 if (this.sport === "VOLLEYBALL") {
205 s.lineups = this.initialLineups.map(lineup => [...lineup]);
206 s.subs = [0, 0];
207 s.timeouts = [0, 0];
208 s.server = this.first ^ (s.finished.length % 2);
209 }
210 }

213 point(s: State, winner: number): void {
214 const server = this.server(s);
215 s.rally[winner].points++;
216 s.rally[server].served++;
217 if (winner === server) {
218 s.rally[server].serveWon++;
219 }
220 s.rally[winner].current++;
221 s.rally[winner].run = Math.max(s.rally[winner].run, s.rally[winner].current);
222 s.rally[1 - winner].current = 0;
223 if (this.sport === "TENNIS") {
224 const kind = this.tennisKind(s);
225 if (kind === "normal") {
226 const receiver = 1 - server;
227 const hypothetical = [...s.points];
228 hypothetical[receiver]++;
229 if (this.normalGameWins(hypothetical, receiver)) {
230 s.rally[receiver].opportunities++;
231 if (winner === receiver) {
232 s.rally[receiver].conversions++;
233 }
234 }
235 }
236 s.points[winner]++;
237 const targetSets = Math.floor(Number(this.options.SETS) / 2) + 1;
238 if (kind === "normal" && this.normalGameWins(s.points, winner)) {
239 s.rally[server].serviceGames++;
240 if (winner === server) {
241 s.rally[server].holds++;
242 } else {
243 s.rally[winner].breaks++;
244 }
245 s.games[winner]++;
246 s.points = [0, 0];
247 s.gameServer ^= 1;
248 if (s.games[winner] >= 6 && s.games[winner] - s.games[1 - winner] >= 2) {
249 this.finishRallyUnit(s, winner, { values: [...s.games], tie: -1, super: false }, targetSets);
250 }
251 } else if (kind !== "normal") {
252 const target = kind === "super" || (this.finalSet(s) && this.options.FINAL === "TB10") ? 10 : 7;
253 if (s.points[winner] >= target && s.points[winner] - s.points[1 - winner] >= 2) {
254 const record = kind === "super" ?
255 { values: [...s.points], tie: -1, super: true } :
256 { values: s.games.map((value, i) => value + (i === winner ? 1 : 0)), tie: s.points[1 - winner], super: false };
257 s.gameServer ^= 1;
258 this.finishRallyUnit(s, winner, record, targetSets);
259 }
260 }
261 } else {
262 s.points[winner]++;
263 if (this.sport === "BADMINTON") {
264 s.server = winner;
265 } else if (this.sport === "VOLLEYBALL" && winner !== s.server) {
266 s.server = winner;
267 const player = s.lineups[winner].shift()!;
268 s.lineups[winner].push(player);
269 }
270 const target = this.sport === "TABLETENNIS" ? 11 : this.sport === "BADMINTON" ? 21 :
271 s.finished.length === 4 ? 15 : 25;
272 const won = (s.points[winner] >= target && s.points[winner] - s.points[1 - winner] >= 2) ||
273 (this.sport === "BADMINTON" && s.points[winner] === 30);
274 if (won) {
275 const needed = this.sport === "TABLETENNIS" ? Math.floor(Number(this.options.GAMES) / 2) + 1 :
276 this.sport === "BADMINTON" ? 2 : 3;
277 this.finishRallyUnit(s, winner, { values: [...s.points], tie: -1, super: false }, needed);
278 }
279 }
280 }

282 standing(s: State): number {
283 return s.frame === 9 && s.standing === 0 ? 10 : s.standing;
284 }

286 roll(s: State, pins: number): void {
287 const standing = this.standing(s);
288 if (pins > standing) {
289 fail("too many pins");
290 }
291 const frame = s.frames[s.frame];
292 const freshRack = frame.length === 0 || (s.frame === 9 && s.standing === 0);
293 const mark = freshRack && pins === 10 ? "X" : pins === standing ? "/" : pins === 0 ? "-" : String(pins);
294 frame.push(pins);
295 s.marks[s.frame].push(mark);
296 s.standing = standing - pins;
297 if (s.frame < 9) {
298 if (pins === 10 && frame.length === 1 || frame.length === 2) {
299 s.frame++;
300 s.standing = 10;
301 }
302 } else if (frame.length === 3 || (frame.length === 2 && frame[0] !== 10 && frame[0] + frame[1] < 10)) {
303 s.done = true;
304 }
305 }

307 frameScores(s: State): (number | null)[] {
308 return s.frames.map((frame, i) => {
309 if (i === 9) {
310 return s.done ? frame.reduce((a, b) => a + b, 0) : null;
311 }
312 const later = s.frames.slice(i + 1).flat();
313 if (frame[0] === 10) {
314 return later.length >= 2 ? 10 + later[0] + later[1] : null;
315 }
316 if (frame.length < 2) {
317 return null;
318 }
319 if (frame[0] + frame[1] === 10) {
320 return later.length ? 10 + later[0] : null;
321 }
322 return frame[0] + frame[1];
323 });
324 }

326 maximum(s: State): number {
327 const hypothetical = structuredClone(s);
328 while (!hypothetical.done) {
329 this.roll(hypothetical, this.standing(hypothetical));
330 }
331 return this.frameScores(hypothetical).reduce<number>((sum, value) => sum + (value ?? 0), 0);
332 }

334 completedTurn(s: State, player: number): void {
335 const stats = s.darts[player];
336 stats.high = Math.max(stats.high, s.turnScore);
337 if (s.turnScore >= 100) {
338 stats.tons++;
339 }
340 if (s.turnScore === 180) {
341 stats.maximums++;
342 }
343 }

345 nextTurn(s: State): void {
346 s.player = 1 - s.player;
347 s.turn = [];
348 s.turnScore = 0;
349 s.turnStart = s.remaining[s.player];
350 }

352 throwDart(s: State, notation: string): void {
353 const dart = darts.get(notation);
354 if (!dart) {
355 fail("invalid dart");
356 }
357 const player = s.player;
358 const stats = s.darts[player];
359 const before = s.remaining[player];
360 stats.darts++;
361 if (before === 50 || before <= 40 && before % 2 === 0) {
362 stats.attempts++;
363 }
364 s.turn.push(notation);
365 s.turnScore += dart.value;
366 stats.points += dart.value;
367 const remaining = before - dart.value;
368 if (remaining < 0 || remaining === 1 || remaining === 0 && !dart.double) {
369 stats.busts++;
370 stats.points -= s.turnScore;
371 s.remaining[player] = s.turnStart;
372 this.nextTurn(s);
373 } else {
374 s.remaining[player] = remaining;
375 if (remaining === 0) {
376 stats.hits++;
377 this.completedTurn(s, player);
378 s.legs[player]++;
379 if (s.legs[player] === Number(this.options.LEGS)) {
380 s.sets[player]++;
381 s.finished.push({ values: [...s.legs], tie: -1, super: false });
382 if (s.sets[player] === Number(this.options.SETS)) {
383 s.done = true;
384 s.winner = player;
385 return;
386 }
387 s.legs = [0, 0];
388 }
389 s.legNumber++;
390 const start = Number(this.options.START);
391 s.remaining = [start, start];
392 s.player = this.first ^ (s.legNumber % 2);
393 s.turn = [];
394 s.turnScore = 0;
395 s.turnStart = start;
396 } else if (s.turn.length === 3) {
397 this.completedTurn(s, player);
398 this.nextTurn(s);
399 }
400 }

403 apply(name: string, args: string[]): void {
404 if (name === "POINT") {
405 this.point(this.state, args[0] === "A" ? 0 : 1);
406 } else if (name === "ROLL") {
407 this.roll(this.state, Number(args[0]));
408 } else if (name === "THROW") {
409 this.throwDart(this.state, args[0]);
410 } else {
411 const side = args[0] === "A" ? 0 : 1;
412 if (name === "SUB") {
413 const lineup = this.state.lineups[side];
414 const position = lineup.indexOf(Number(args[1]));
415 if (position < 0) {
416 fail("player not on court");
417 }
418 if (lineup.includes(Number(args[2]))) {
419 fail("player already on court");
420 }
421 if (this.state.subs[side] >= 6) {
422 fail("no substitutions left");
423 }
424 lineup[position] = Number(args[2]);
425 this.state.subs[side]++;
426 } else {
427 if (this.state.timeouts[side] >= 2) {
428 fail("no timeouts left");
429 }
430 this.state.timeouts[side]++;
431 }
432 }
433 }

435 event(name: string, args: string[]): void {
436 const allowed = name === "POINT" ? this.rallySport() : name === "ROLL" ? this.sport === "BOWLING" :
437 name === "THROW" ? this.sport === "DARTS" : this.sport === "VOLLEYBALL";
438 if (!allowed) {
439 fail("wrong event");
440 }
441 const events = name === "POINT" ? args[0].split("").map(value => [value]) :
442 name === "ROLL" || name === "THROW" ? args.map(value => [value]) : [args];
443 for (const values of events) {
444 if (this.state.done) {
445 fail("match over");
446 }
447 this.apply(name, values);
448 this.history.push({ name, args: values });
449 }
450 }

457 situation(side: number): string {
458 const s = this.state;
459 const next = structuredClone(s);
460 const server = this.server(s);
461 const normal = this.sport === "TENNIS" && this.tennisKind(s) === "normal";
462 this.point(next, side);
463 if (next.done) {
464 return "match point";
465 }
466 if (next.finished.length > s.finished.length) {
467 return ["TENNIS", "VOLLEYBALL"].includes(this.sport) ? "set point" : "game point";
468 }
469 if (normal && next.gameServer !== s.gameServer) {
470 return side === server ? "game point" : "break point";
471 }
472 return "";
473 }

475 score(): string {
476 const s = this.state;
477 if (this.sport === "BOWLING") {
478 const scores = this.frameScores(s);
479 let total = 0;
480 const items: string[] = [];
481 for (let i = 0; i < 10; i++) {
482 if (!s.frames[i].length) {
483 break;
484 }
485 let value = s.marks[i].join("");
486 if (scores[i] !== null) {
487 total += scores[i]!;
488 value += "(" + total + ")";
489 }
490 items.push(value);
491 }
492 return this.id + ": " + this.names[0] + (items.length ? " " + items.join(" ") : "");
493 }
494 const server = this.sport === "DARTS" ? s.player : this.server(s);
495 const names = this.names.map((name, i) => name + (!s.done && i === server ? "*" : ""));
496 let items = s.finished.map(record => this.scoreRecord(record));
497 if (this.sport === "DARTS") {
498 items = ["sets", pair(s.sets), "legs", pair(s.legs), "score", pair(s.remaining)];
499 if (!s.done) {
500 if (s.turn.length) {
501 items.push("turn", ...s.turn);
502 }
503 const finish = checkout(s.remaining[s.player], 3 - s.turn.length);
504 if (finish) {
505 items.push("checkout", ...finish);
506 }
507 }
508 } else if (!s.done) {
509 if (this.sport === "TENNIS") {
510 const kind = this.tennisKind(s);
511 if (kind === "super") {
512 items.push("[" + pair(s.points) + "]");
513 } else {
514 items.push(pair(s.games));
515 if (kind === "tie") {
516 items.push(pair(s.points));
517 } else if (this.options.DEUCE === "AD" && s.points.every(value => value >= 3)) {
518 items.push(s.points[0] === s.points[1] ? "40-40" : s.points[0] > s.points[1] ? "AD-40" : "40-AD");
519 } else {
520 items.push(s.points.map(value => ["0", "15", "30", "40"][value]).join("-"));
521 }
522 }
523 } else {
524 items.push(pair(s.points));
525 if (this.sport === "VOLLEYBALL") {
526 items.push("server", String(s.lineups[s.server][0]));
527 }
528 }
529 for (let side = 0; side < 2; side++) {
530 const situation = this.situation(side);
531 if (situation) {
532 items.push("(" + situation + " " + this.names[side] + ")");
533 }
534 }
535 }
536 return this.id + ": " + names[0] + " vs " + names[1] + " " + items.join(" ");
537 }
538
539 stats(): string[] {
540 const s = this.state;
541 if (this.sport === "BOWLING") {
542 const marks = s.marks.flat();
543 const open = s.frames.filter(frame => frame.length >= 2 && frame[0] !== 10 && frame[0] + frame[1] < 10).length;
544 return [this.id + " " + this.names[0] + ": strikes " + marks.filter(mark => mark === "X").length +
545 " spares " + marks.filter(mark => mark === "/").length + " open " + open + " max " + this.maximum(s)];
546 }
547 return this.names.map((name, i) => {
548 let result = this.id + " " + name + ": ";
549 if (this.sport === "DARTS") {
550 const stats = s.darts[i];
551 const average = stats.darts ? Math.floor(stats.points * 300 / stats.darts) : 0;
552 result += "darts " + stats.darts + " average " + Math.floor(average / 100) + "." +
553 String(average % 100).padStart(2, "0") + " busts " + stats.busts + " high " + stats.high +
554 " tons " + stats.tons + " 180s " + stats.maximums + " doubles " + stats.hits + "/" + stats.attempts;
555 } else {
556 const stats = s.rally[i];
557 result += "points " + stats.points + " serve " + stats.serveWon + "/" + stats.served + " run " + stats.run;
558 if (this.sport === "TENNIS") {
559 result += " holds " + stats.holds + "/" + stats.serviceGames + " breaks " + stats.breaks +
560 " break points " + stats.conversions + "/" + stats.opportunities;
561 } else if (this.sport === "VOLLEYBALL") {
562 result += " subs " + s.subs[i] + " timeouts " + s.timeouts[i] + " lineup " + s.lineups[i].join(",");
563 }
564 }
565 return result;
566 });
567 }
568
569 result(): string {
570 const s = this.state;
571 if (!s.done) {
572 fail("match not finished");
573 }
574 if (this.sport === "BOWLING") {
575 return this.id + ": " + this.names[0] + " " + this.frameScores(s).reduce<number>((sum, value) => sum + (value ?? 0), 0);
576 }
577 return this.id + ": " + this.names[s.winner] + " def. " + this.names[1 - s.winner] + " " +
578 s.finished.map(record => this.scoreRecord(record, s.winner === 1)).join(" ");
579 }

584function command(line: string): string[] {
585 const [name, ...args] = line.split(" ");
586 if (name === "MATCH") {
587 if (matches.has(args[0])) {
588 fail("duplicate match");
589 }
590 const match = new Match(args);
591 matches.set(match.id, match);
592 return [];
593 }
594 if (name === "CHECKOUT") {
595 const answer = checkout(Number(args[0]));
596 return ["checkout " + args[0] + ": " + (answer ? answer.join(" ") : "none")];
597 }
598 if (name === "STANDINGS") {
