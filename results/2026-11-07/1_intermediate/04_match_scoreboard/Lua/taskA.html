1local rally_sports = {TENNIS = true, TABLETENNIS = true, BADMINTON = true, VOLLEYBALL = true}
2local option_rules = {
3 TENNIS = {SETS = {'3', '5'}, DEUCE = {'AD', 'NOAD'}, FINAL = {'TB7', 'TB10', 'ADV', 'SUPER'}, FIRST = {'A', 'B'}},
4 TABLETENNIS = {GAMES = {'5', '7'}, FIRST = {'A', 'B'}},
5 BADMINTON = {FIRST = {'A', 'B'}}, VOLLEYBALL = {FIRST = {'A', 'B'}},
6 DARTS = {START = {'501', '301', '701'}, SETS = {'1', '2', '3', '4', '5'}, LEGS = {'1', '2', '3', '4', '5'}, FIRST = {'A', 'B'}},
7 BOWLING = {}
8}
9local function score_error(message) error(message, 0) end
10local function side(text) return text == 'A' and 1 or 2 end
11local function other(player) return 3 - player end
12local function pair_text(values) return values[1] .. '-' .. values[2] end
13local function clone(value)
14 if type(value) ~= 'table' then return value end
15 local result = {}
16 for key, item in pairs(value) do result[key] = clone(item) end
17 return result
18end
19local function append_all(target, items)
20 table.move(items, 1, #items, #target + 1, target)
21end
22local function sum(values)
23 local total = 0
24 for _, value in ipairs(values) do total = total + value end
25 return total
26end
27local function pack_record(record)
28 return string.pack(string.rep('i4', #record), table.unpack(record))
29end
30local function contains(values, needle)
31 for _, value in ipairs(values) do
32 if value == needle then return true end
33 end
34 return false
35end
36local function checkout_tables()
37 local candidates, doubles = {}, {}
38 local darts = {['25'] = {25, false}, BULL = {50, true}, MISS = {0, false}}
39 for _, item in ipairs({{'S', 1}, {'T', 3}, {'D', 2}}) do
40 local prefix, multiplier = item[1], item[2]
41 for value = 20, 1, -1 do
42 local code = prefix .. value
43 candidates[#candidates + 1] = {code, value * multiplier}
44 darts[code] = {value * multiplier, prefix == 'D'}
45 if prefix == 'D' then doubles[#doubles + 1] = {code, value * multiplier} end
46 end
47 end
48 candidates[#candidates + 1] = {'25', 25}
49 candidates[#candidates + 1] = {'BULL', 50}
50 doubles[#doubles + 1] = {'BULL', 50}
51 local tables = {{}, {}, {}}
52 for _, last in ipairs(doubles) do
53 if not tables[1][last[2]] then tables[1][last[2]] = {last[1]} end
54 end
55 for _, first in ipairs(candidates) do
56 for _, last in ipairs(doubles) do
57 local total = first[2] + last[2]
58 if not tables[2][total] then tables[2][total] = {first[1], last[1]} end
59 end
60 end
61 for _, first in ipairs(candidates) do
62 for _, second in ipairs(candidates) do
63 for _, last in ipairs(doubles) do
64 local total = first[2] + second[2] + last[2]
65 if not tables[3][total] then tables[3][total] = {first[1], second[1], last[1]} end
66 end
67 end
68 end
69 return darts, tables
70end
71local dart_data, checkout_data = checkout_tables()
72local function checkout(score, maximum)
73 for count = 1, maximum or 3 do
74 if checkout_data[count][score] then return checkout_data[count][score] end
75 end
76 return {}
77end
78local function create_match(tokens)
79 local m = {id = tokens[2], sport = tokens[3], names = {tokens[4]}, options = {}, initial = {}, history = {}}
80 if m.sport ~= 'BOWLING' then m.names[2] = tokens[5] end
81 local first_option
82 if m.sport == 'VOLLEYBALL' then
83 for player = 1, 2 do
84 m.initial[player] = {}
85 for number in tokens[player + 5]:gmatch('[^,]+') do
86 m.initial[player][#m.initial[player] + 1] = tonumber(number)
87 end
88 end
89 first_option = 8
90 elseif m.sport == 'BOWLING' then
91 first_option = 5
92 else
93 first_option = 6
94 end
95 local rules, seen = option_rules[m.sport], {}
96 for key, values in pairs(rules) do m.options[key] = values[1] end
97 for index = first_option, #tokens do
98 local key, value = tokens[index]:match('^([^=]+)=([^=]+)$')
99 if not key or not rules[key] or not contains(rules[key], value) or seen[key] then score_error('invalid option') end
100 seen[key] = true
101 m.options[key] = value
102 end
103 local first = side(m.options.FIRST or 'A')
104 local stats = {}
105 for _, key in ipairs({'points', 'served', 'servewins', 'runmax', 'holds', 'games', 'breaks', 'opportunities', 'converted',
106 'darts', 'scored', 'busts', 'high', 'tons', 'maximums', 'attempts', 'hits'}) do
107 stats[key] = {0, 0}
108 end
109 local s = {p = {0, 0}, g = {0, 0}, won = {0, 0}, server = first, winner = 0, completed = {}, stats = stats, last = 0, run = 0}
110 if m.sport == 'VOLLEYBALL' then
111 s.lineups = clone(m.initial)
112 s.subs = {0, 0}
113 s.timeouts = {0, 0}
114 elseif m.sport == 'BOWLING' then
115 s.frames, s.marks = {}, {}
116 for frame = 1, 10 do
117 s.frames[frame], s.marks[frame] = {}, {}
118 end
119 s.frame, s.standing, s.fresh = 1, 10, true
120 elseif m.sport == 'DARTS' then
121 local start = tonumber(m.options.START)
122 s.legs, s.remaining, s.turn = {0, 0}, {start, start}, {}
123 s.turnscore, s.turnstart, s.legnumber = 0, start, 0
124 end
125 m.state = s
126 return m
127end
128local function tennis_mode(m)
129 local s = m.state
130 local final = s.won[1] == tonumber(m.options.SETS) // 2 and s.won[1] == s.won[2]
131 if final and m.options.FINAL == 'SUPER' then return 'SUPER' end
132 if s.g[1] == 6 and s.g[2] == 6 and not (final and m.options.FINAL == 'ADV') then return 'TB' end
133 return 'NORMAL'
134end
135local function rally_server(m)
136 local s = m.state
137 if m.sport == 'TENNIS' then
138 if tennis_mode(m) ~= 'NORMAL' and ((sum(s.p) + 1) // 2) % 2 == 1 then return other(s.server) end
139 elseif m.sport == 'TABLETENNIS' then
140 local first = side(m.options.FIRST)
141 if #s.completed % 2 == 1 then first = other(first) end
142 local total = sum(s.p)
143 local flip
144 if total < 20 then flip = (total // 2) % 2 == 1 else flip = total % 2 == 1 end
145 if flip then return other(first) end
146 return first
147 end
148 return s.server
149end
150local function rally_outcome(m, winner)
151 local s = m.state
152 local points = {s.p[1], s.p[2]}
153 points[winner] = points[winner] + 1
154 local loser = other(winner)
155 local game, unit, needed = false, false, 0
156 if m.sport == 'TENNIS' then
157 local mode = tennis_mode(m)
158 if mode == 'NORMAL' then
159 game = points[winner] >= 4 and (m.options.DEUCE == 'NOAD' or points[winner] - points[loser] >= 2)
160 local games = s.g[winner] + 1
161 unit = game and games >= 6 and games - s.g[loser] >= 2
162 else
163 local final = s.won[1] == s.won[2] and s.won[1] == tonumber(m.options.SETS) // 2
164 local target = 7
165 if mode == 'SUPER' or (final and m.options.FINAL == 'TB10') then target = 10 end
166 unit = points[winner] >= target and points[winner] - points[loser] >= 2
167 end
168 needed = tonumber(m.options.SETS) // 2 + 1
169 elseif m.sport == 'TABLETENNIS' then
170 unit = points[winner] >= 11 and points[winner] - points[loser] >= 2
171 needed = tonumber(m.options.GAMES) // 2 + 1
172 elseif m.sport == 'BADMINTON' then
173 unit = points[winner] == 30 or (points[winner] >= 21 and points[winner] - points[loser] >= 2)
174 needed = 2
175 else
176 local target = #s.completed == 4 and 15 or 25
177 unit = points[winner] >= target and points[winner] - points[loser] >= 2
178 needed = 3
179 end
180 return game, unit, unit and s.won[winner] + 1 == needed
181end
182local function point_event(m, winner)
183 local server = rally_server(m)
184 local game, unit, finished = rally_outcome(m, winner)
185 local mode = m.sport == 'TENNIS' and tennis_mode(m) or ''
186 local opportunity = mode == 'NORMAL' and rally_outcome(m, other(server))
187 local s = m.state
188 local record = {1, winner, server, opportunity and 1 or 0, game and 1 or 0, s.p[1], s.p[2], s.g[1], s.g[2],
189 s.won[1], s.won[2], s.server, #s.completed, s.last, s.run, s.stats.runmax[winner], 0, 0}
190 s.stats.points[winner] = s.stats.points[winner] + 1
191 s.stats.served[server] = s.stats.served[server] + 1
192 if winner == server then s.stats.servewins[server] = s.stats.servewins[server] + 1 end
193 s.run = s.last == winner and s.run + 1 or 1
194 s.last = winner
195 s.stats.runmax[winner] = math.max(s.stats.runmax[winner], s.run)
196 if opportunity then
197 s.stats.opportunities[other(server)] = s.stats.opportunities[other(server)] + 1
198 if winner ~= server then s.stats.converted[winner] = s.stats.converted[winner] + 1 end
199 end
200 s.p[winner] = s.p[winner] + 1
201 if m.sport == 'TENNIS' then
202 if game then
203 s.stats.games[server] = s.stats.games[server] + 1
204 if winner == server then
205 s.stats.holds[winner] = s.stats.holds[winner] + 1
206 else
207 s.stats.breaks[winner] = s.stats.breaks[winner] + 1
208 end
209 s.g[winner] = s.g[winner] + 1
210 s.server = other(s.server)
211 elseif unit then
212 s.server = other(s.server)
213 end
214 if unit then
215 if mode == 'SUPER' then
216 s.completed[#s.completed + 1] = {s.p[1], s.p[2], 2, 0}
217 elseif mode == 'TB' then
218 s.g[winner] = s.g[winner] + 1
219 s.completed[#s.completed + 1] = {s.g[1], s.g[2], 1, s.p[other(winner)]}
220 else
221 s.completed[#s.completed + 1] = {s.g[1], s.g[2], 0, 0}
222 end
223 end
224 elseif m.sport == 'BADMINTON' then
225 s.server = winner
226 elseif m.sport == 'VOLLEYBALL' and winner ~= server then
227 s.server = winner
228 local lineup = s.lineups[winner]
229 local first = table.remove(lineup, 1)
230 lineup[#lineup + 1] = first
231 record[18] = 1
232 end
233 if unit then
234 if m.sport ~= 'TENNIS' then s.completed[#s.completed + 1] = {s.p[1], s.p[2], 0, 0} end
235 s.won[winner] = s.won[winner] + 1
236 if finished then
237 s.winner = winner
238 else
239 if m.sport == 'VOLLEYBALL' then
240 record[17] = 1
241 append_all(record, s.lineups[1])
242 append_all(record, s.lineups[2])
243 append_all(record, s.subs)
244 append_all(record, s.timeouts)
245 s.lineups = clone(m.initial)
246 s.subs, s.timeouts = {0, 0}, {0, 0}
247 local first = side(m.options.FIRST)
248 s.server = #s.completed % 2 == 0 and first or other(first)
249 end
250 s.p, s.g = {0, 0}, {0, 0}
251 end
252 elseif game then
253 s.p = {0, 0}
254 end
255 m.history[#m.history + 1] = pack_record(record)
256end
257local function roll_event(s, pins)
258 if pins > s.standing then score_error('too many pins') end
259 local frame, mark = s.frame, nil
260 if s.fresh and pins == 10 then
261 mark = 'X'
262 elseif pins == s.standing then
263 mark = '/'
264 else
265 mark = pins == 0 and '-' or tostring(pins)
266 end
267 local rolls = s.frames[frame]
268 rolls[#rolls + 1] = pins
269 s.marks[frame][#s.marks[frame] + 1] = mark
270 s.standing = s.standing - pins
271 s.fresh = false
272 if frame < 10 then
273 if (#rolls == 1 and pins == 10) or #rolls == 2 then
274 s.frame, s.standing, s.fresh = frame + 1, 10, true
275 end
276 else
277 local complete = #rolls == 3 or (#rolls == 2 and rolls[1] ~= 10 and sum(rolls) < 10)
278 if complete then
279 s.winner = 1
280 elseif s.standing == 0 then
281 s.standing, s.fresh = 10, true
282 end
283 end
284end
285local function bowling_scores(s)
286 local scores = {}
287 for frame = 1, 10 do
288 local rolls = s.frames[frame]
289 if #rolls > 0 then
290 local later = {}
291 for index = frame + 1, 10 do append_all(later, s.frames[index]) end
292 if frame == 10 then
293 if s.winner ~= 0 then scores[frame] = sum(rolls) end
294 elseif rolls[1] == 10 then
295 if #later >= 2 then scores[frame] = 10 + later[1] + later[2] end
296 elseif #rolls == 2 then
297 if sum(rolls) == 10 then
298 if #later > 0 then scores[frame] = 10 + later[1] end
299 else
300 scores[frame] = sum(rolls)
301 end
302 end
303 end
304 end
305 return scores
306end
307local function total_bowling(s)
308 local scores, total = bowling_scores(s), 0
309 for frame = 1, 10 do total = total + (scores[frame] or 0) end
310 return total
311end
312local function complete_turn(s, player)
313 local score = s.turnscore
314 s.stats.high[player] = math.max(s.stats.high[player], score)
315 if score >= 100 then s.stats.tons[player] = s.stats.tons[player] + 1 end
316 if score == 180 then s.stats.maximums[player] = s.stats.maximums[player] + 1 end
317end
318local function next_turn(s)
319 s.server = other(s.server)
320 s.turn, s.turnscore = {}, 0
321 s.turnstart = s.remaining[s.server]
322end
323local function dart_event(m, code)
324 local dart = dart_data[code]
325 if not dart then score_error('invalid dart') end
326 local s = m.state
327 local player, value, double = s.server, dart[1], dart[2]
328 local before = s.remaining[player]
329 s.stats.darts[player] = s.stats.darts[player] + 1
330 if before == 50 or (before <= 40 and before % 2 == 0) then s.stats.attempts[player] = s.stats.attempts[player] + 1 end
331 local remaining = before - value
332 if remaining < 0 or remaining == 1 or (remaining == 0 and not double) then
333 s.stats.busts[player] = s.stats.busts[player] + 1
334 s.stats.scored[player] = s.stats.scored[player] - s.turnscore
335 s.remaining[player] = s.turnstart
336 next_turn(s)
337 return
338 end
339 s.stats.scored[player] = s.stats.scored[player] + value
340 s.remaining[player] = remaining
341 s.turnscore = s.turnscore + value
342 s.turn[#s.turn + 1] = code
343 if remaining == 0 then
344 s.stats.hits[player] = s.stats.hits[player] + 1
345 complete_turn(s, player)
346 s.legs[player] = s.legs[player] + 1
347 local set_over = s.legs[player] == tonumber(m.options.LEGS)
348 if set_over then
349 s.completed[#s.completed + 1] = {s.legs[1], s.legs[2], 0, 0}
350 s.won[player] = s.won[player] + 1
351 end
352 if set_over and s.won[player] == tonumber(m.options.SETS) then
353 s.winner = player
354 return
355 end
356 if set_over then s.legs = {0, 0} end
357 s.legnumber = s.legnumber + 1
358 local first = side(m.options.FIRST)
359 s.server = s.legnumber % 2 == 0 and first or other(first)
360 local start = tonumber(m.options.START)
361 s.remaining, s.turn = {start, start}, {}
362 s.turnscore, s.turnstart = 0, start
363 elseif #s.turn == 3 then
364 complete_turn(s, player)
365 next_turn(s)
366 end
367end
368local function undo_event(m)
369 if #m.history == 0 then score_error('nothing to undo') end
370 local record = table.remove(m.history)
371 if type(record) == 'table' then
372 m.state = record
373 return
374 end
375 local data = {string.unpack(string.rep('i4', #record // 4), record)}
376 local s = m.state
377 if data[1] == 2 then
378 s.lineups[data[2]][data[3]] = data[4]
379 s.subs[data[2]] = s.subs[data[2]] - 1
380 elseif data[1] == 3 then
381 s.timeouts[data[2]] = s.timeouts[data[2]] - 1
382 else
383 local winner, server = data[2], data[3]
384 s.stats.points[winner] = s.stats.points[winner] - 1
385 s.stats.served[server] = s.stats.served[server] - 1
386 if winner == server then s.stats.servewins[server] = s.stats.servewins[server] - 1 end
387 if data[4] == 1 then
388 s.stats.opportunities[other(server)] = s.stats.opportunities[other(server)] - 1
389 if winner ~= server then s.stats.converted[winner] = s.stats.converted[winner] - 1 end
390 end
391 if data[5] == 1 then
392 s.stats.games[server] = s.stats.games[server] - 1
393 if winner == server then
394 s.stats.holds[winner] = s.stats.holds[winner] - 1
395 else
396 s.stats.breaks[winner] = s.stats.breaks[winner] - 1
397 end
398 end
399 if data[17] == 1 then
400 s.lineups = {{table.unpack(data, 19, 24)}, {table.unpack(data, 25, 30)}}
401 s.subs, s.timeouts = {data[31], data[32]}, {data[33], data[34]}
402 end
403 if data[18] == 1 then
404 local lineup = s.lineups[winner]
405 local last = table.remove(lineup)
406 table.insert(lineup, 1, last)
407 end
408 s.p, s.g, s.won = {data[6], data[7]}, {data[8], data[9]}, {data[10], data[11]}
409 s.server, s.winner = data[12], 0
410 while #s.completed > data[13] do table.remove(s.completed) end
411 s.last, s.run = data[14], data[15]
412 s.stats.runmax[winner] = data[16]
413 end
414end
415local function event(m, command, args)
416 local correct = (command == 'POINT' and rally_sports[m.sport]) or (command == 'ROLL' and m.sport == 'BOWLING') or
417 (command == 'THROW' and m.sport == 'DARTS') or ((command == 'SUB' or command == 'TIMEOUT') and m.sport == 'VOLLEYBALL')
418 if not correct then score_error('wrong event') end
419 local events
420 if command == 'POINT' then
421 events = {}
422 for index = 1, #args[1] do events[index] = args[1]:sub(index, index) end
423 elseif command == 'SUB' or command == 'TIMEOUT' then
424 events = {''}
425 else
426 events = args
427 end
428 for _, value in ipairs(events) do
429 if m.state.winner ~= 0 then score_error('match over') end
430 if command == 'POINT' then
431 point_event(m, side(value))
432 elseif command == 'ROLL' then
433 local old = clone(m.state)
434 roll_event(m.state, tonumber(value))
435 m.history[#m.history + 1] = old
436 elseif command == 'THROW' then
437 local old = clone(m.state)
438 dart_event(m, value)
439 m.history[#m.history + 1] = old
440 elseif command == 'SUB' then
441 local player = side(args[1])
442 local outgoing, incoming = tonumber(args[2]), tonumber(args[3])
443 local lineup, position = m.state.lineups[player], nil
444 for index, shirt in ipairs(lineup) do
445 if shirt == outgoing then position = index end
446 end
447 if not position then score_error('player not on court') end
448 if contains(lineup, incoming) then score_error('player already on court') end
449 if m.state.subs[player] >= 6 then score_error('no substitutions left') end
450 m.history[#m.history + 1] = pack_record({2, player, position, outgoing})
451 lineup[position] = incoming
452 m.state.subs[player] = m.state.subs[player] + 1
453 else
454 local player = side(args[1])
455 if m.state.timeouts[player] >= 2 then score_error('no timeouts left') end
456 m.history[#m.history + 1] = pack_record({3, player})
457 m.state.timeouts[player] = m.state.timeouts[player] + 1
458 end
459 end
460end
461local function completed_text(record, winner)
462 local a, b = record[1], record[2]
463 if winner == 2 then a, b = b, a end
464 local text = a .. '-' .. b
465 if record[3] == 2 then return '[' .. text .. ']' end
466 if record[3] == 1 then return text .. '(' .. record[4] .. ')' end
467 return text
468end
469local function situation(m, player)
470 local game, unit, finished = rally_outcome(m, player)
471 if finished then return 'match point' end
472 if unit then
473 if m.sport == 'TENNIS' or m.sport == 'VOLLEYBALL' then return 'set point' end
474 return 'game point'
475 end
476 if game then return player == rally_server(m) and 'game point' or 'break point' end
477 return ''
478end
479local function score_line(m)
480 local s = m.state
481 if m.sport == 'BOWLING' then
482 local parts, scores, cumulative = {m.id .. ': ' .. m.names[1]}, bowling_scores(s), 0
483 for frame = 1, 10 do
484 if #s.frames[frame] == 0 then break end
485 local text = table.concat(s.marks[frame])
486 if scores[frame] then
487 cumulative = cumulative + scores[frame]
488 text = text .. '(' .. cumulative .. ')'
489 end
490 parts[#parts + 1] = text
491 end
492 return table.concat(parts, ' ')
493 end
494 local names = {m.names[1], m.names[2]}
495 if s.winner == 0 then
496 local player = rally_sports[m.sport] and rally_server(m) or s.server
497 names[player] = names[player] .. '*'
498 end
499 local parts = {m.id .. ': ' .. names[1] .. ' vs ' .. names[2]}
500 if m.sport == 'DARTS' then
501 append_all(parts, {'sets', pair_text(s.won), 'legs', pair_text(s.legs), 'score', pair_text(s.remaining)})
502 if s.winner == 0 then
503 if #s.turn > 0 then
504 parts[#parts + 1] = 'turn'
505 append_all(parts, s.turn)
506 end
507 local answer = checkout(s.remaining[s.server], 3 - #s.turn)
508 if #answer > 0 then
509 parts[#parts + 1] = 'checkout'
510 append_all(parts, answer)
511 end
512 end
513 else
514 for _, record in ipairs(s.completed) do parts[#parts + 1] = completed_text(record) end
515 if s.winner == 0 then
516 if m.sport == 'TENNIS' then
517 local mode = tennis_mode(m)
518 if mode == 'SUPER' then
519 parts[#parts + 1] = '[' .. pair_text(s.p) .. ']'
520 else
521 parts[#parts + 1] = pair_text(s.g)
522 if mode == 'TB' then
523 parts[#parts + 1] = pair_text(s.p)
524 elseif m.options.DEUCE == 'AD' and math.min(s.p[1], s.p[2]) >= 3 then
525 local label
526 if s.p[1] == s.p[2] then
527 label = '40-40'
528 elseif s.p[1] > s.p[2] then
529 label = 'AD-40'
530 else
531 label = '40-AD'
532 end
533 parts[#parts + 1] = label
534 else
535 local labels = {'0', '15', '30', '40'}
536 parts[#parts + 1] = labels[s.p[1] + 1] .. '-' .. labels[s.p[2] + 1]
537 end
538 end
539 else
540 parts[#parts + 1] = pair_text(s.p)
541 if m.sport == 'VOLLEYBALL' then append_all(parts, {'server', tostring(s.lineups[s.server][1])}) end
542 end
543 for player = 1, 2 do
544 local label = situation(m, player)
545 if label ~= '' then parts[#parts + 1] = '(' .. label .. ' ' .. m.names[player] .. ')' end
546 end
547 end
548 end
549 return table.concat(parts, ' ')
550end
551local function stats_lines(m)
552 local s, output = m.state, {}
553 local stats = s.stats
554 if m.sport == 'BOWLING' then
555 local strikes, spares, open = 0, 0, 0
556 for frame = 1, 10 do
557 for _, mark in ipairs(s.marks[frame]) do
558 if mark == 'X' then strikes = strikes + 1 end
559 if mark == '/' then spares = spares + 1 end
560 end
561 local rolls = s.frames[frame]
562 if #rolls >= 2 and rolls[1] ~= 10 and rolls[1] + rolls[2] < 10 then open = open + 1 end
563 end
564 local future = clone(s)
565 while future.winner == 0 do roll_event(future, future.standing) end
566 return {m.id .. ' ' .. m.names[1] .. ': strikes ' .. strikes .. ' spares ' .. spares .. ' open ' .. open .. ' max ' .. total_bowling(future)}
567 end
568 for player = 1, 2 do
569 local text
570 if m.sport == 'DARTS' then
571 local darts = stats.darts[player]
572 local average = darts == 0 and 0 or (stats.scored[player] * 300) // darts
573 text = 'darts ' .. darts .. ' average ' .. string.format('%d.%02d', average // 100, average % 100) ..
574 ' busts ' .. stats.busts[player] .. ' high ' .. stats.high[player] .. ' tons ' .. stats.tons[player] ..
575 ' 180s ' .. stats.maximums[player] .. ' doubles ' .. stats.hits[player] .. '/' .. stats.attempts[player]
576 else
577 text = 'points ' .. stats.points[player] .. ' serve ' .. stats.servewins[player] .. '/' .. stats.served[player] .. ' run ' .. stats.runmax[player]
578 if m.sport == 'TENNIS' then
579 text = text .. ' holds ' .. stats.holds[player] .. '/' .. stats.games[player] .. ' breaks ' .. stats.breaks[player] ..
580 ' break points ' .. stats.converted[player] .. '/' .. stats.opportunities[player]
581 elseif m.sport == 'VOLLEYBALL' then
582 text = text .. ' subs ' .. s.subs[player] .. ' timeouts ' .. s.timeouts[player] .. ' lineup ' .. table.concat(s.lineups[player], ',')
583 end
584 end
585 output[#output + 1] = m.id .. ' ' .. m.names[player] .. ': ' .. text
586 end
587 return output
588end
589local function result_line(m)
590 local s = m.state
591 if s.winner == 0 then score_error('match not finished') end
592 if m.sport == 'BOWLING' then return m.id .. ': ' .. m.names[1] .. ' ' .. total_bowling(s) end
593 local records, winner = {}, s.winner
594 for _, record in ipairs(s.completed) do records[#records + 1] = completed_text(record, winner) end
595 return m.id .. ': ' .. m.names[winner] .. ' def. ' .. m.names[other(winner)] .. ' ' .. table.concat(records, ' ')
596end
597local matches = {}
598local function execute(line)
599 local tokens = {}
600 for token in line:gmatch('%S+') do tokens[#tokens + 1] = token end
601 local command = tokens[1]
602 if command == 'MATCH' then
603 if matches[tokens[2]] then score_error('duplicate match') end
604 local m = create_match(tokens)
605 matches[m.id] = m
606 return {}
607 end
608 if command == 'CHECKOUT' then
609 local answer = checkout(tonumber(tokens[2]))
610 local text = #answer == 0 and 'none' or table.concat(answer, ' ')
611 return {'checkout ' .. tokens[2] .. ': ' .. text}
612 end
613 if command == 'STANDINGS' then
614 local totals, names = {}, {}
615 for _, m in pairs(matches) do
616 if m.sport ~= 'BOWLING' and m.state.winner ~= 0 then
617 for player = 1, 2 do
618 local name = m.names[player]
619 if not totals[name] then
620 totals[name] = {0, 0}
621 names[#names + 1] = name
622 end
623 local index = player == m.state.winner and 1 or 2
624 totals[name][index] = totals[name][index] + 1
625 end
626 end
627 end
628 table.sort(names, function(a, b)
629 if totals[a][1] ~= totals[b][1] then return totals[a][1] > totals[b][1] end
630 if totals[a][2] ~= totals[b][2] then return totals[a][2] < totals[b][2] end
631 return a < b
632 end)
633 local entries = {}
634 for _, name in ipairs(names) do entries[#entries + 1] = name .. ' ' .. pair_text(totals[name]) end
635 return {'standings: ' .. (#entries == 0 and 'none' or table.concat(entries, ', '))}
636 end
637 local m = matches[tokens[2]]
638 if not m then score_error('unknown match') end
639 if command == 'POINT' or command == 'ROLL' or command == 'THROW' or command == 'SUB' or command == 'TIMEOUT' then
640 local args = {table.unpack(tokens, 3)}
641 event(m, command, args)
642 elseif command == 'UNDO' then
643 undo_event(m)
644 elseif command == 'SCORE' then
645 return {score_line(m)}
646 elseif command == 'STATS' then
647 return stats_lines(m)
648 elseif command == 'RESULT' then
649 return {result_line(m)}
650 end
651 return {}
652end
653local count = tonumber(io.read('*l'))
654for _ = 1, count do
655 local ok, result = pcall(execute, io.read('*l'))
656 if ok then
657 for _, line in ipairs(result) do io.write(line, '\n') end
658 else
659 io.write('ERROR ', result, '\n')
660 end
661end
