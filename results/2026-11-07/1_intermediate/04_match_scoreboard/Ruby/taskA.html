4module Darts
5 DATA = {}
6 (1..20).each do |n|
7 DATA["S#{n}"] = { value: n, double: false }
8 DATA["D#{n}"] = { value: 2 * n, double: true }
9 DATA["T#{n}"] = { value: 3 * n, double: false }
10 end
11 DATA["25"] = { value: 25, double: false }
12 DATA["BULL"] = { value: 50, double: true }
13 DATA["MISS"] = { value: 0, double: false }
14 A = %w[S T D].flat_map { |prefix| 20.downto(1).map { |n| "#{prefix}#{n}" } } + %w[25 BULL]
15 B = 20.downto(1).map { |n| "D#{n}" } + ["BULL"]
16 CACHE = {}
17
18 def self.checkout(score, maximum = 3)
19 key = [score, maximum]
20 return CACHE[key] if CACHE.key?(key)
21 (1..maximum).each do |count|
22 A.repeated_permutation(count - 1) do |prefix|
23 wanted = score - prefix.sum { |dart| DATA[dart][:value] }
24 last = B.find { |dart| DATA[dart][:value] == wanted }
25 if last
26 CACHE[key] = prefix + [last]
27 return CACHE[key]
28 end
29 end
30 end
31 CACHE[key] = nil
32 end
33end

35class Game
36 CHOICES = {
37 "TENNIS" => {
38 "SETS" => %w[3 5], "DEUCE" => %w[AD NOAD],
39 "FINAL" => %w[TB7 TB10 ADV SUPER], "FIRST" => %w[A B]
40 },
41 "TABLETENNIS" => { "GAMES" => %w[5 7], "FIRST" => %w[A B] },
42 "BADMINTON" => { "FIRST" => %w[A B] },
43 "VOLLEYBALL" => { "FIRST" => %w[A B] },
44 "DARTS" => { "START" => %w[301 501 701], "SETS" => %w[1 2 3 4 5],
45 "LEGS" => %w[1 2 3 4 5], "FIRST" => %w[A B] },
46 "BOWLING" => {}
47 }.freeze
48 DEFAULTS = {
49 "TENNIS" => { "SETS" => "3", "DEUCE" => "AD", "FINAL" => "TB7", "FIRST" => "A" },
50 "TABLETENNIS" => { "GAMES" => "5", "FIRST" => "A" },
51 "BADMINTON" => { "FIRST" => "A" }, "VOLLEYBALL" => { "FIRST" => "A" },
52 "DARTS" => { "START" => "501", "SETS" => "1", "LEGS" => "1", "FIRST" => "A" },
53 "BOWLING" => {}
54 }.freeze
55 attr_reader :id, :sport, :names, :state
56
57 def initialize(args)
58 @id, @sport = args[0, 2]
59 @names = @sport == "BOWLING" ? [args[2]] : args[2, 2]
60 @lineups = @sport == "VOLLEYBALL" ? args[4, 2].map { |lineup| lineup.split(",").map(&:to_i) } : [[], []]
61 option_start = @sport == "BOWLING" ? 3 : @sport == "VOLLEYBALL" ? 6 : 4
62 @options = DEFAULTS[@sport].dup
63 seen = {}
64 args.drop(option_start).each do |option|
65 parts = option.split("=", -1)
66 if parts.length != 2 || !CHOICES[@sport].key?(parts[0]) ||
67 !CHOICES[@sport][parts[0]].include?(parts[1]) || seen[parts[0]]
68 raise Invalid, "invalid option"
69 end
70 seen[parts[0]] = true
71 @options[parts[0]] = parts[1]
72 end
73 @first = @options["FIRST"] == "B" ? 1 : 0
74 start = @options.fetch("START", "501").to_i
75 @state = {
76 done: false, winner: -1, sets: [0, 0], legs: [0, 0], points: [0, 0], games: [0, 0],
77 finished: [], game_server: @first, server: @first,
78 lineups: @lineups.map(&:dup), subs: [0, 0], timeouts: [0, 0],
79 rally: Array.new(2) do
80 { points: 0, served: 0, serve_won: 0, run: 0, current: 0,
81 holds: 0, service_games: 0, breaks: 0, opportunities: 0, conversions: 0 }
82 end,
83 frames: Array.new(10) { [] }, marks: Array.new(10) { [] }, frame: 0, standing: 10,
84 remaining: [start, start], player: @first, leg_number: 0,
85 turn: [], turn_start: start, turn_score: 0,
86 darts: Array.new(2) do
87 { darts: 0, points: 0, busts: 0, high: 0, tons: 0, maximums: 0, attempts: 0, hits: 0 }
88 end
89 }
90 @initial = copy(@state)
91 @history = []
92 end

102 def rally_sport?
103 %w[TENNIS TABLETENNIS BADMINTON VOLLEYBALL].include?(@sport)
104 end

106 def final_set?(s)
107 target = @options["SETS"].to_i.div(2)
108 s[:sets][0] == target && s[:sets][1] == target
109 end
110
111 def tennis_kind(s)
112 return :super if final_set?(s) && @options["FINAL"] == "SUPER"
113 if s[:games] == [6, 6] && !(final_set?(s) && @options["FINAL"] == "ADV")
114 return :tie
115 end
116 :normal
117 end

119 def server(s)
120 if @sport == "TENNIS"
121 return s[:game_server] if tennis_kind(s) == :normal
122 k = s[:points].sum + 1
123 s[:game_server] ^ (k.div(2) % 2)
124 elsif @sport == "TABLETENNIS"
125 first = @first ^ (s[:finished].length % 2)
126 t = s[:points].sum
127 first ^ (t < 20 ? t.div(2) % 2 : t % 2)
128 else
129 s[:server]
130 end
131 end
132
133 def normal_game_wins?(points, winner)
134 points[winner] >= 4 && (@options["DEUCE"] == "NOAD" || points[winner] - points[1 - winner] >= 2)
135 end
136
137 def finish_rally_unit(s, winner, record, target)
138 s[:finished] << record
139 s[:sets][winner] += 1
140 if s[:sets][winner] == target
141 s[:done] = true
142 s[:winner] = winner
143 else
144 s[:points] = [0, 0]
145 s[:games] = [0, 0]
146 if @sport == "VOLLEYBALL"
147 s[:lineups] = @lineups.map(&:dup)
148 s[:subs] = [0, 0]
149 s[:timeouts] = [0, 0]
150 s[:server] = @first ^ (s[:finished].length % 2)
151 end
152 end
153 end

155 def point(s, winner)
156 serving = server(s)
157 s[:rally][winner][:points] += 1
158 s[:rally][serving][:served] += 1
159 s[:rally][serving][:serve_won] += 1 if winner == serving
160 s[:rally][winner][:current] += 1
161 s[:rally][winner][:run] = [s[:rally][winner][:run], s[:rally][winner][:current]].max
162 s[:rally][1 - winner][:current] = 0
163 if @sport == "TENNIS"
164 kind = tennis_kind(s)
165 if kind == :normal
166 receiver = 1 - serving
167 hypothetical = s[:points].dup
168 hypothetical[receiver] += 1
169 if normal_game_wins?(hypothetical, receiver)
170 s[:rally][receiver][:opportunities] += 1
171 s[:rally][receiver][:conversions] += 1 if winner == receiver
172 end
173 end
174 s[:points][winner] += 1
175 target_sets = @options["SETS"].to_i.div(2) + 1
176 if kind == :normal && normal_game_wins?(s[:points], winner)
177 s[:rally][serving][:service_games] += 1
178 if winner == serving
179 s[:rally][serving][:holds] += 1
180 else
181 s[:rally][winner][:breaks] += 1
182 end
183 s[:games][winner] += 1
184 s[:points] = [0, 0]
185 s[:game_server] ^= 1
186 if s[:games][winner] >= 6 && s[:games][winner] - s[:games][1 - winner] >= 2
187 finish_rally_unit(s, winner, { values: s[:games].dup, tie: -1, super: false }, target_sets)
188 end
189 elsif kind != :normal
190 target = kind == :super || (final_set?(s) && @options["FINAL"] == "TB10") ? 10 : 7
191 if s[:points][winner] >= target && s[:points][winner] - s[:points][1 - winner] >= 2
192 record = if kind == :super
193 { values: s[:points].dup, tie: -1, super: true }
194 else
195 { values: s[:games].each_with_index.map { |value, i| value + (i == winner ? 1 : 0) },
196 tie: s[:points][1 - winner], super: false }
197 end
198 s[:game_server] ^= 1
199 finish_rally_unit(s, winner, record, target_sets)
200 end
201 end
202 else
203 s[:points][winner] += 1
204 if @sport == "BADMINTON"
205 s[:server] = winner
206 elsif @sport == "VOLLEYBALL" && winner != s[:server]
207 s[:server] = winner
208 s[:lineups][winner].rotate!
209 end
210 target = if @sport == "TABLETENNIS"
211 11
212 elsif @sport == "BADMINTON"
213 21
214 else
215 s[:finished].length == 4 ? 15 : 25
216 end
217 won = (s[:points][winner] >= target && s[:points][winner] - s[:points][1 - winner] >= 2) ||
218 (@sport == "BADMINTON" && s[:points][winner] == 30)
219 if won
220 needed = @sport == "TABLETENNIS" ? @options["GAMES"].to_i.div(2) + 1 : @sport == "BADMINTON" ? 2 : 3
221 finish_rally_unit(s, winner, { values: s[:points].dup, tie: -1, super: false }, needed)
222 end
223 end
224 end

226 def standing(s)
227 s[:frame] == 9 && s[:standing].zero? ? 10 : s[:standing]
228 end
229
230 def roll(s, pins)
231 pins_standing = standing(s)
232 raise Invalid, "too many pins" if pins > pins_standing
233 frame = s[:frames][s[:frame]]
234 fresh_rack = frame.empty? || (s[:frame] == 9 && s[:standing].zero?)
235 mark = if fresh_rack && pins == 10
236 "X"
237 elsif pins == pins_standing
238 "/"
239 elsif pins.zero?
240 "-"
241 else
242 pins.to_s
243 end
244 frame << pins
245 s[:marks][s[:frame]] << mark
246 s[:standing] = pins_standing - pins
247 if s[:frame] < 9
248 if (pins == 10 && frame.length == 1) || frame.length == 2
249 s[:frame] += 1
250 s[:standing] = 10
251 end
252 elsif frame.length == 3 || (frame.length == 2 && frame[0] != 10 && frame[0] + frame[1] < 10)
253 s[:done] = true
254 end
255 end
256
257 def frame_scores(s)
258 s[:frames].each_with_index.map do |frame, i|
259 if i == 9
260 s[:done] ? frame.sum : nil
261 else
262 later = s[:frames].drop(i + 1).flatten
263 if frame[0] == 10
264 later.length >= 2 ? 10 + later[0] + later[1] : nil
265 elsif frame.length < 2
266 nil
267 elsif frame.sum == 10
268 later.empty? ? nil : 10 + later[0]
269 else
270 frame.sum
271 end
272 end
273 end
274 end

276 def maximum(s)
277 hypothetical = copy(s)
278 roll(hypothetical, standing(hypothetical)) until hypothetical[:done]
279 frame_scores(hypothetical).compact.sum
280 end

282 def completed_turn(s, player)
283 stats = s[:darts][player]
284 stats[:high] = [stats[:high], s[:turn_score]].max
285 stats[:tons] += 1 if s[:turn_score] >= 100
286 stats[:maximums] += 1 if s[:turn_score] == 180
287 end
288
289 def next_turn(s)
290 s[:player] = 1 - s[:player]
291 s[:turn] = []
292 s[:turn_score] = 0
293 s[:turn_start] = s[:remaining][s[:player]]
294 end

296 def throw_dart(s, notation)
297 dart = Darts::DATA[notation]
298 raise Invalid, "invalid dart" unless dart
299 player = s[:player]
300 stats = s[:darts][player]
301 before = s[:remaining][player]
302 stats[:darts] += 1
303 stats[:attempts] += 1 if before == 50 || (before <= 40 && before.even?)
304 s[:turn] << notation
305 s[:turn_score] += dart[:value]
306 stats[:points] += dart[:value]
307 remaining = before - dart[:value]
308 if remaining.negative? || remaining == 1 || (remaining.zero? && !dart[:double])
309 stats[:busts] += 1
310 stats[:points] -= s[:turn_score]
311 s[:remaining][player] = s[:turn_start]
312 next_turn(s)
313 else
314 s[:remaining][player] = remaining
315 if remaining.zero?
316 stats[:hits] += 1
317 completed_turn(s, player)
318 s[:legs][player] += 1
319 if s[:legs][player] == @options["LEGS"].to_i
320 s[:sets][player] += 1
321 s[:finished] << { values: s[:legs].dup, tie: -1, super: false }
322 if s[:sets][player] == @options["SETS"].to_i
323 s[:done] = true
324 s[:winner] = player
325 return
326 end
327 s[:legs] = [0, 0]
328 end
329 s[:leg_number] += 1
330 start = @options["START"].to_i
331 s[:remaining] = [start, start]
332 s[:player] = @first ^ (s[:leg_number] % 2)
333 s[:turn] = []
334 s[:turn_score] = 0
335 s[:turn_start] = start
336 elsif s[:turn].length == 3
337 completed_turn(s, player)
338 next_turn(s)
339 end
340 end
341 end

343 def apply(name, args)
344 case name
345 when "POINT"
346 point(@state, args[0] == "A" ? 0 : 1)
347 when "ROLL"
348 roll(@state, args[0].to_i)
349 when "THROW"
350 throw_dart(@state, args[0])
351 else
352 side = args[0] == "A" ? 0 : 1
353 if name == "SUB"
354 lineup = @state[:lineups][side]
355 position = lineup.index(args[1].to_i)
356 raise Invalid, "player not on court" unless position
357 raise Invalid, "player already on court" if lineup.include?(args[2].to_i)
358 raise Invalid, "no substitutions left" if @state[:subs][side] >= 6
359 lineup[position] = args[2].to_i
360 @state[:subs][side] += 1
361 else
362 raise Invalid, "no timeouts left" if @state[:timeouts][side] >= 2
363 @state[:timeouts][side] += 1
364 end
365 end
366 end

368 def event(name, args)
369 allowed = case name
370 when "POINT" then rally_sport?
371 when "ROLL" then @sport == "BOWLING"
372 when "THROW" then @sport == "DARTS"
373 else @sport == "VOLLEYBALL"
374 end
375 raise Invalid, "wrong event" unless allowed
376 events = if name == "POINT"
377 args[0].chars.map { |value| [value] }
378 elsif %w[ROLL THROW].include?(name)
379 args.map { |value| [value] }
380 else
381 [args]
382 end
383 events.each do |values|
384 raise Invalid, "match over" if @state[:done]
385 apply(name, values)
386 @history << { name: name, args: values }
387 end
388 end
389
390 def undo
391 raise Invalid, "nothing to undo" if @history.empty?
392 @history.pop
393 @state = copy(@initial)
394 @history.each { |event| apply(event[:name], event[:args]) }
395 end

397 def score_record(record, reverse = false)
398 value = pair(record[:values], reverse)
399 record[:super] ? "[#{value}]" : value + (record[:tie] >= 0 ? "(#{record[:tie]})" : "")
400 end

402 def situation(side)
403 s = @state
404 hypothetical = copy(s)
405 serving = server(s)
406 normal = @sport == "TENNIS" && tennis_kind(s) == :normal
407 point(hypothetical, side)
408 return "match point" if hypothetical[:done]
409 if hypothetical[:finished].length > s[:finished].length
410 return %w[TENNIS VOLLEYBALL].include?(@sport) ? "set point" : "game point"
411 end
412 return side == serving ? "game point" : "break point" if normal && hypothetical[:game_server] != s[:game_server]
413 nil
414 end

416 def score
417 s = @state
418 if @sport == "BOWLING"
419 scores = frame_scores(s)
420 total = 0
421 items = []
422 s[:frames].each_with_index do |frame, i|
423 break if frame.empty?
424 value = s[:marks][i].join
425 unless scores[i].nil?
426 total += scores[i]
427 value += "(#{total})"
428 end
429 items << value
430 end
431 return "#{@id}: #{@names[0]}" + (items.empty? ? "" : " " + items.join(" "))
432 end
433 serving = @sport == "DARTS" ? s[:player] : server(s)
434 displays = @names.each_with_index.map { |name, i| name + (!s[:done] && i == serving ? "*" : "") }
435 items = s[:finished].map { |record| score_record(record) }
436 if @sport == "DARTS"
437 items = ["sets", pair(s[:sets]), "legs", pair(s[:legs]), "score", pair(s[:remaining])]
438 unless s[:done]
439 items.concat(["turn", *s[:turn]]) unless s[:turn].empty?
440 finish = Darts.checkout(s[:remaining][s[:player]], 3 - s[:turn].length)
441 items.concat(["checkout", *finish]) if finish
442 end
443 elsif !s[:done]
444 if @sport == "TENNIS"
445 kind = tennis_kind(s)
446 if kind == :super
447 items << "[#{pair(s[:points])}]"
448 else
449 items << pair(s[:games])
450 if kind == :tie
451 items << pair(s[:points])
452 elsif @options["DEUCE"] == "AD" && s[:points].all? { |value| value >= 3 }
453 display = s[:points][0] == s[:points][1] ? "40-40" : s[:points][0] > s[:points][1] ? "AD-40" : "40-AD"
454 items << display
455 else
456 items << s[:points].map { |value| %w[0 15 30 40][value] }.join("-")
457 end
458 end
459 else
460 items << pair(s[:points])
461 items.concat(["server", s[:lineups][s[:server]][0].to_s]) if @sport == "VOLLEYBALL"
462 end
463 2.times do |side|
464 status = situation(side)
465 items << "(#{status} #{@names[side]})" if status
466 end
467 end
468 "#{@id}: #{displays[0]} vs #{displays[1]} " + items.join(" ")
469 end

471 def stats
472 s = @state
473 if @sport == "BOWLING"
474 marks = s[:marks].flatten
475 open_frames = s[:frames].count { |frame| frame.length >= 2 && frame[0] != 10 && frame[0] + frame[1] < 10 }
476 return ["#{@id} #{@names[0]}: strikes #{marks.count('X')} spares #{marks.count('/')} open #{open_frames} max #{maximum(s)}"]
477 end
478 @names.each_with_index.map do |name, i|
479 result = "#{@id} #{name}: "
480 if @sport == "DARTS"
481 stats = s[:darts][i]
482 average = stats[:darts].positive? ? (stats[:points] * 300).div(stats[:darts]) : 0
483 result += "darts #{stats[:darts]} average #{format('%d.%02d', average.div(100), average % 100)}"
484 result += " busts #{stats[:busts]} high #{stats[:high]} tons #{stats[:tons]} 180s #{stats[:maximums]}"
485 result += " doubles #{stats[:hits]}/#{stats[:attempts]}"
486 else
487 stats = s[:rally][i]
488 result += "points #{stats[:points]} serve #{stats[:serve_won]}/#{stats[:served]} run #{stats[:run]}"
489 if @sport == "TENNIS"
490 result += " holds #{stats[:holds]}/#{stats[:service_games]} breaks #{stats[:breaks]}"
491 result += " break points #{stats[:conversions]}/#{stats[:opportunities]}"
492 elsif @sport == "VOLLEYBALL"
493 result += " subs #{s[:subs][i]} timeouts #{s[:timeouts][i]} lineup " + s[:lineups][i].join(",")
494 end
495 end
496 result
497 end
498 end

500 def result
501 s = @state
502 raise Invalid, "match not finished" unless s[:done]
503 return "#{@id}: #{@names[0]} #{frame_scores(s).compact.sum}" if @sport == "BOWLING"
504 "#{@id}: #{@names[s[:winner]]} def. #{@names[1 - s[:winner]]} " +
505 s[:finished].map { |record| score_record(record, s[:winner] == 1) }.join(" ")
506 end

509class Scoreboard
510 def initialize
511 @matches = {}
512 end
513
514 def command(line)
515 name, *args = line.split(" ")
516 if name == "MATCH"
517 raise Invalid, "duplicate match" if @matches.key?(args[0])
518 match = Game.new(args)
519 @matches[match.id] = match
520 return []
521 end
522 if name == "CHECKOUT"
523 answer = Darts.checkout(args[0].to_i)
524 return ["checkout #{args[0]}: " + (answer ? answer.join(" ") : "none")]
525 end
526 if name == "STANDINGS"
527 totals = {}
528 @matches.each_value do |match|
529 next if !match.state[:done] || match.sport == "BOWLING"
530 match.names.each_with_index do |person, i|
531 totals[person] ||= [0, 0]
532 totals[person][i == match.state[:winner] ? 0 : 1] += 1
533 end
534 end
535 ordered = totals.sort_by { |person, values| [-values[0], values[1], person] }
536 display = ordered.map { |person, values| "#{person} #{values.join('-')}" }
537 return ["standings: " + (display.empty? ? "none" : display.join(", "))]
538 end
539 match = @matches[args[0]]
540 raise Invalid, "unknown match" unless match
541 case name
542 when "UNDO"
543 match.undo
544 []
545 when "SCORE"
546 [match.score]
547 when "STATS"
548 match.stats
549 when "RESULT"
550 [match.result]
551 else
552 match.event(name, args.drop(1))
553 []
554 end
555 end
556end

564 rescue Invalid => error
