4class MatchError implements Exception {
5 final String message;
6 MatchError(this.message);
7}
8Never fail(String message) => throw MatchError(message);
9String pair(List<int> values, [bool reverse = false]) => reverse ? '${values[1]}-${values[0]}' : '${values[0]}-${values[1]}';
10class DartValue {
11 final String name;
12 final int value;
13 final bool double;
14 DartValue(this.name, this.value, this.double);
15}
16final darts = <DartValue>[
17 for (final prefix in ['S', 'T', 'D'])
18 for (var n = 20; n >= 1; n--) DartValue('$prefix$n', n * (prefix == 'S' ? 1 : prefix == 'T' ? 3 : 2), prefix == 'D'),
19 DartValue('25', 25, false), DartValue('BULL', 50, true), DartValue('MISS', 0, false),
20];
21final doubles = [for (var i = 40; i < 60; i++) i, 61];
22List<int>? checkout(int score, int maximum) {
23 for (final last in doubles)
24 if (darts[last].value == score) return [last];
25 if (maximum >= 2) {
26 for (var a = 0; a < 62; a++) {
27 for (final b in doubles)
28 if (darts[a].value + darts[b].value == score) return [a, b];
29 }
30 }
31 if (maximum >= 3) {
32 for (var a = 0; a < 62; a++) {
33 for (var b = 0; b < 62; b++) {
34 for (final c in doubles)
35 if (darts[a].value + darts[b].value + darts[c].value == score) return [a, b, c];
36 }
37 }
38 }
39 return null;
40}

89 0];

154 bool get finalTennisSet => wins[0] == int.parse(match.options['SETS']!) ~/ 2 && wins[1] == wins[0];
155 String get tennisKind => finalTennisSet && match.options['FINAL'] == 'SUPER' ? 'SUPER' : games[0] == 6
156 && games[1] == 6 && !(finalTennisSet && match.options['FINAL'] == 'ADV') ? 'TIE' : 'NORMAL';
157 int nextServer() {
158 if (match.sport == 'TENNIS')
159 return tennisKind == 'NORMAL' ? server : server ^ (((points[0] + points[1] + 1) ~/ 2) % 2);
160 if (match.sport == 'TABLETENNIS') {
161 final t = points[0] + points[1], initial = match.first ^ (completed.length % 2);
162 return initial ^ ((t < 20 ? t ~/ 2 : t) % 2);
163 }
164 return server;
165 }
166
167 bool normalGameWin(int side, int extra) => points[side] + extra >= 4 && (match.options['DEUCE'] == 'NOAD'
168 || points[side] + extra - points[1 - side] >= 2);
169 void endSet(int side, FinishedSet result) {
170 completed.add(result);
171 wins[side]++;
172 final target = switch (match.sport) {
173 'TENNIS' => int.parse(match.options['SETS']!) ~/ 2 + 1,
174 'TABLETENNIS' => int.parse(match.options['GAMES']!) ~/ 2 + 1,
175 'BADMINTON' => 2,
176 _ => 3,
177 };
178 if (wins[side] == target) {
179 finished = true;
180 winner = side;
181 return;
182 }
183 points = [0, 0];
184 games = [0, 0];
185 if (match.sport == 'VOLLEYBALL') {
186 lineup = match.startingLineup.map((line) => List<int>.of(line)).toList();
187 substitutions = [0, 0];
188 timeouts = [0, 0];
189 server = match.first ^ (completed.length % 2);
190 }
191 }

193 void point(int side) {
194 final serving = nextServer(), kind = match.sport == 'TENNIS' ? tennisKind : '';
195 rallyPoints[side]++;
196 served[serving]++;
197 if (side == serving)
198 serveWins[side]++;
199 if (previousWinner == side)
200 currentRun++;
201 else {
202 previousWinner = side;
203 currentRun = 1;
204 }
205 runs[side] = max(runs[side], currentRun);
206 if (match.sport == 'TENNIS' && kind == 'NORMAL') {
207 final receiver = 1 - server;
208 if (normalGameWin(receiver, 1)) {
209 opportunities[receiver]++;
210 if (side == receiver)
211 convertedBreaks[receiver]++;
212 }
213 }
214 points[side]++;
215 if (match.sport == 'TENNIS') {
216 if (kind != 'NORMAL') {
217 final target = kind == 'SUPER' || finalTennisSet && match.options['FINAL'] == 'TB10' ? 10 : 7;
218 if (points[side] >= target && points[side] - points[1 - side] >= 2) {
219 server = 1 - server;
220 if (kind == 'SUPER')
221 endSet(side, FinishedSet(List.of(points), superTie: true));
222 else {
223 games[side] = 7;
224 endSet(side, FinishedSet(List.of(games), tieLoser: points[1 - side]));
225 }
226 }
227 } else if (normalGameWin(side, 0)) {
228 serviceGames[server]++;
229 if (side == server)
230 holds[side]++;
231 else breaks[side]++;
232 games[side]++;
233 points = [0, 0];
234 server = 1 - server;
235 if (games[side] >= 6 && games[side] - games[1 - side] >= 2)
236 endSet(side, FinishedSet(List.of(games)));
237 }
238 return;
239 }
240 if (match.sport == 'BADMINTON')
241 server = side;
242 if (match.sport == 'VOLLEYBALL' && server != side) {
243 server = side;
244 lineup[side].add(lineup[side].removeAt(0));
245 }
246 final target = match.sport == 'TABLETENNIS' ? 11 : match.sport == 'BADMINTON' ? 21 : completed.length == 4 ? 15 : 25;
247 if (points[side] >= target && points[side] - points[1 - side] >= 2 || match.sport == 'BADMINTON' && points[side] == 30) {
248 endSet(side, FinishedSet(List.of(points)));
249 }
250 }

252 void roll(int pins) {
253 if (pins > standing)
254 fail('too many pins');
255 final frame = frames.last, symbols = marks.last;
256 symbols.add(newPins && pins == 10 ? 'X' : pins == standing ? '/' : pins == 0 ? '-' : '$pins');
257 frame.add(pins);
258 standing -= pins;
259 newPins = false;
260 if (frames.length < 10) {
261 if (frame[0] == 10 || frame.length == 2) {
262 frames.add([]);
263 marks.add([]);
264 standing = 10;
265 newPins = true;
266 }
267 } else if (frame.length == 3 || frame.length == 2 && frame[0] != 10 && frame[0] + frame[1] < 10) {
268 finished = true;
269 } else if (standing == 0) {
270 standing = 10;
271 newPins = true;
272 }
273 }
274 List<int?> frameScores() {
275 final result = <int?>[];
276 for (var i = 0; i < frames.length; i++) {
277 final frame = frames[i], later = frames.skip(i + 1).expand((f) => f).toList();
278 int? value;
279 if (i == 9) {
280 if (finished)
281 value = frame.fold(0, (a, b) => a + b);
282 } else if (frame.isNotEmpty && frame[0] == 10) {
283 if (later.length >= 2)
284 value = 10 + later[0] + later[1];
285 } else if (frame.length == 2) {
286 final base = frame[0] + frame[1];
287 if (base < 10)
288 value = base;
289 else if (later.isNotEmpty) value = 10 + later[0];
290 }
291 result.add(value);
292 }
293 return result;
294 }

296 int finalBowlingScore() => frameScores().fold(0, (s, v) => s + (v ?? 0));
297 void endTurn() {
298 server = 1 - server;
299 turn.clear();
300 turnValue = 0;
301 turnStart = remaining[server];
302 }

304 void completedTurn(int side) {
305 high[side] = max(high[side], turnValue);
306 if (turnValue >= 100)
307 tons[side]++;
308 if (turnValue == 180)
309 maximums[side]++;
310 }

312 void throwDart(int index) {
313 final dart = darts[index], side = server, before = remaining[side];
314 if (before == 50 || before <= 40 && before % 2 == 0)
315 attempts[side]++;
316 dartCount[side]++;
317 final rest = before - dart.value;
318 if (rest < 0 || rest == 1 || rest == 0 && !dart.double) {
319 scored[side] -= turnValue;
320 remaining[side] = turnStart;
321 busts[side]++;
322 endTurn();
323 return;
324 }
325 remaining[side] = rest;
326 scored[side] += dart.value;
327 turnValue += dart.value;
328 turn.add(index);
329 if (rest == 0) {
330 hits[side]++;
331 completedTurn(side);
332 legs[side]++;
333 if (legs[side] == int.parse(match.options['LEGS']!)) {
334 completed.add(FinishedSet(List.of(legs)));
335 wins[side]++;
336 if (wins[side] == int.parse(match.options['SETS']!)) {
337 finished = true;
338 winner = side;
339 return;
340 }
341 legs = [0, 0];
342 }
343 legNumber++;
344 remaining = List.filled(2, int.parse(match.options['START']!));
345 server = match.first ^ ((legNumber - 1) % 2);
346 turn = [];
347 turnValue = 0;
348 turnStart = remaining[server];
349 } else if (turn.length == 3) {
350 completedTurn(side);
351 endTurn();
352 }
353 }

355 void apply(Event event) {
356 if (finished)
357 fail('match over');
358 switch (event.kind) {
359 case 'POINT':
360 point(event.side);
361 case 'ROLL':
362 roll(event.first);
363 case 'THROW':
364 throwDart(event.first);
365 case 'SUB':
366 final team = lineup[event.side], position = team.indexOf(event.first);
367 if (position < 0)
368 fail('player not on court');
369 if (team.contains(event.second))
370 fail('player already on court');
371 if (substitutions[event.side] == 6)
372 fail('no substitutions left');
373 team[position] = event.second;
374 substitutions[event.side]++;
375 case 'TIMEOUT':
376 if (timeouts[event.side] == 2)
377 fail('no timeouts left');
378 timeouts[event.side]++;
379 }
380 }
381 String? situation(int side) {
382 final trial = copy();
383 trial.point(side);
384 if (trial.finished)
385 return 'match point';
386 if (trial.completed.length > completed.length)
387 return match.sport == 'TABLETENNIS' || match.sport == 'BADMINTON' ? 'game point' : 'set point';
388 if (match.sport == 'TENNIS' && tennisKind == 'NORMAL' && trial.games[side] != games[side])
389 return side == nextServer() ? 'game point' : 'break point';
390 return null;
391 }

393 String score() {
394 if (match.sport == 'BOWLING') {
395 final scores = frameScores(), parts = <String>[];
396 var cumulative = 0;
397 for (var i = 0; i < frames.length; i++) {
398 if (frames[i].isEmpty)
399 continue;
400 if (scores[i] != null)
401 cumulative += scores[i]!;
402 parts.add('${marks[i].join()}${scores[i] == null ? '' : '($cumulative)'}');
403 }
404 return '${match.id}: ${match.names[0]}${parts.isEmpty ? '' : ' ${parts.join(' ')}'}';
405 }
406 final serving = nextServer(), display = [for (var i = 0; i < 2; i++) '${match.names[i]}${!finished && i == serving ? '*' : ''}'];
407 final items = <String>[];
408 if (match.sport == 'DARTS') {
409 items.addAll(['sets', pair(wins), 'legs', pair(legs), 'score', pair(remaining)]);
410 if (!finished) {
411 if (turn.isNotEmpty)
412 items.addAll(['turn', dartNames(turn)]);
413 final route = checkout(remaining[server], 3 - turn.length);
414 if (route != null)
415 items.addAll(['checkout', dartNames(route)]);
416 }
417 } else {
418 items.addAll(completed.map((s) => s.display()));
419 if (!finished) {
420 if (match.sport == 'TENNIS') {
421 if (tennisKind == 'SUPER')
422 items.add('[${pair(points)}]');
423 else {
424 items.add(pair(games));
425 if (tennisKind == 'TIE')
426 items.add(pair(points));
427 else if (match.options['DEUCE'] == 'AD' && points.every((p) => p >= 3)) items.add(points[0] == points[1] ? '40-40' : points[0] > points[1] ? 'AD-40' : '40-AD');
428 else items.add('${['0', '15', '30', '40'][points[0]]}-${['0', '15', '30', '40'][points[1]]}');
429 }
430 } else {
431 items.add(pair(points));
432 if (match.sport == 'VOLLEYBALL')
433 items.addAll(['server', '${lineup[server][0]}']);
434 }
435 for (var side = 0; side < 2; side++) {
436 final condition = situation(side);
437 if (condition != null)
438 items.add('($condition ${match.names[side]})');
439 }
440 }
441 }
442 return '${match.id}: ${display[0]} vs ${display[1]} ${items.join(' ')}';
443 }

445 String stats() {
446 final out = StringBuffer();
447 for (var side = 0; side < match.names.length; side++) {
448 out.write('${match.id} ${match.names[side]}: ');
449 if (match.rally) {
450 out.write('points ${rallyPoints[side]} serve ${serveWins[side]}/${served[side]} run ${runs[side]}');
451 if (match.sport == 'TENNIS')
452 out.write(' holds ${holds[side]}/${serviceGames[side]} breaks ${breaks[side]} break points ${convertedBreaks[side]}/${opportunities[side]}');
453 if (match.sport == 'VOLLEYBALL')
454 out.write(' subs ${substitutions[side]} timeouts ${timeouts[side]} lineup ${lineup[side].join(',')}');
455 } else if (match.sport == 'BOWLING') {
456 final allMarks = marks.expand((f) => f), trial = copy();
457 while (!trial.finished)
458 trial.roll(trial.standing);
459 final open = frames.where((f) => f.length >= 2 && f[0] != 10 && f[0] + f[1] < 10).length;
460 out.write('strikes ${allMarks.where((m) => m == 'X').length} spares ${allMarks.where((m) => m == '/').length} open $open max ${trial.finalBowlingScore()}');
461 } else {
462 final average = dartCount[side] == 0 ? 0 : scored[side] * 300 ~/ dartCount[side];
463 out.write('darts ${dartCount[side]} average ${average ~/ 100}.${(average % 100).toString().padLeft(2, '0')} busts ${busts[side]} high ${high[side]} tons ${tons[side]} 180s ${maximums[side]} doubles ${hits[side]}/${attempts[side]}');
464 }
465 out.writeln();
466 }
467 return out.toString();
468 }

470 String result() {
471 if (!finished)
472 fail('match not finished');
473 if (match.sport == 'BOWLING')
474 return '${match.id}: ${match.names[0]} ${finalBowlingScore()}';
475 return '${match.id}: ${match.names[winner]} def. ${match.names[1 - winner]} ${completed.map((s) => s.display(winner == 1)).join(' ')}';
476 }

478const allowed = <String, Map<String, List<String>>>{
479 'TENNIS': {
480 'SETS': ['3', '5'],
481 'DEUCE': ['AD', 'NOAD'],
482 'FINAL': ['TB7', 'TB10', 'ADV', 'SUPER'],
483 'FIRST': ['A', 'B']
484 },
485 'TABLETENNIS': { 'GAMES': ['5', '7'], 'FIRST': ['A', 'B'] },
486 'BADMINTON': { 'FIRST': ['A', 'B'] },
487 'VOLLEYBALL': { 'FIRST': ['A', 'B'] },
488 'BOWLING': {},
489 'DARTS': {
490 'START': ['501', '301', '701'],
491 'SETS': ['1', '2', '3', '4', '5'],
492 'LEGS': ['1', '2', '3', '4', '5'],
493 'FIRST': ['A', 'B']
494 },
495};
496void main() {
497 final matches = <String, Match>{}, output = StringBuffer();
498 final count = int.parse(stdin.readLineSync()!);
499 for (var command = 0; command < count; command++) {
500 final a = stdin.readLineSync()!.split(' '), op = a[0];
501 try {
502 if (op == 'MATCH') {
503 if (matches.containsKey(a[1]))
504 fail('duplicate match');
505 final sport = a[2], rules = allowed[sport]!;
506 final options = rules.map((key, values) => MapEntry(key, values.first));
507 final seen = <String>{}, start = sport == 'BOWLING' ? 4 : sport == 'VOLLEYBALL' ? 7 : 5;
508 for (final option in a.skip(start)) {
509 final pair = option.split('=');
510 if (pair.length != 2 || !(rules[pair[0]]?.contains(pair[1]) ?? false) || !seen.add(pair[0]))
511 fail('invalid option');
512 options[pair[0]] = pair[1];
513 }
514 final names = sport == 'BOWLING' ? [a[3]] : [a[3], a[4]];
515 final lineup = sport == 'VOLLEYBALL' ? [a[5].split(',').map(int.parse).toList(), a[6].split(',').map(int.parse).toList()] : <List<int>>[[],
516 []];
517 matches[a[1]] = Match(a[1], sport, names, options, lineup);
518 } else if (op == 'CHECKOUT') {
519 final score = int.parse(a[1]), route = checkout(score, 3);
520 output.writeln('checkout $score: ${route == null ? 'none' : dartNames(route)}');
521 } else if (op == 'STANDINGS') {
522 final records = <String, List<int>>{};
523 for (final m in matches.values.where((m) => m.sport != 'BOWLING' && m.state.finished)) {
524 (records[m.names[m.state.winner]] ??= [0, 0])[0]++;
525 (records[m.names[1 - m.state.winner]] ??= [0, 0])[1]++;
526 }
527 final ns = records.keys.toList()..sort((a, b) => records[a]![0] != records[b]![0] ? records[b]![0].compareTo(records[a]![0]) : records[a]![1] != records[b]![1] ? records[a]![1].compareTo(records[b]![1]) : a.compareTo(b));
528 output.writeln('standings: ${ns.isEmpty ? 'none' : ns.map((n) => '$n ${pair(records[n]!)}').join(', ')}');
529 } else {
530 final m = matches[a[1]];
531 if (m == null)
532 fail('unknown match');
533 if (op == 'UNDO')
534 m.undo();
535 else if (op == 'SCORE') output.writeln(m.state.score());
536 else if (op == 'STATS') output.write(m.state.stats());
537 else if (op == 'RESULT') output.writeln(m.state.result());
538 else {
539 if (op == 'POINT' && !m.rally || op == 'ROLL' && m.sport != 'BOWLING' || op == 'THROW' && m.sport != 'DARTS'
540 || ['SUB', 'TIMEOUT'].contains(op) && m.sport != 'VOLLEYBALL')
541 fail('wrong event');
542 final raw = op == 'POINT' ? a[2].split('') : ['ROLL', 'THROW'].contains(op) ? a.sublist(2) : [''];
543 for (final token in raw) {
544 if (m.state.finished)
545 fail('match over');
546 Event event;
547 if (op == 'POINT')
548 event = Event(op, token == 'A' ? 0 : 1);
549 else if (op == 'ROLL') event = Event(op, 0, int.parse(token));
550 else if (op == 'THROW') {
551 final index = darts.indexWhere((d) => d.name == token);
552 if (index < 0)
553 fail('invalid dart');
554 event = Event(op, 0, index);
555 } else event = Event(op, a[2] == 'A' ? 0 : 1, op == 'SUB' ? int.parse(a[3]) : 0, op == 'SUB' ? int.parse(a[4]) : 0);
556 m.state.apply(event);
557 m.history.add(event);
558 }
559 }
560 }
561 } on MatchError catch (error) {
562 output.writeln('ERROR ${error.message}');
563 }
564 }
565 stdout.write(output);
