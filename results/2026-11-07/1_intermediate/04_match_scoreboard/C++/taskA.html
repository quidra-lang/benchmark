56vector<string> split(const string &s, char delimiter = ' ') {
57 vector<string> result;
58 std::size_t start = 0;
59 for (std::size_t i = 0; i <= s.size(); ++i) {
60 if (i == s.size() || s[i] == delimiter) {
61 result.push_back(s.substr(start, i - start));
62 start = i + 1;
63 }
64 }
65 return result;
66}

68 string result;
69 for (std::size_t i = 0; i < items.size(); ++i) {
70 if (i) result += separator;

72 }
73 return result;

84 char prefix = value == 1 ? 'S' : value == 2 ? 'D' : 'T';
85 for (int n = 20; n >= 1; --n) darts.push_back({text(prefix, n), value * n, value == 2});

91int doubleDart(int value) {
92 if (value == 50) return 61;
93 if (value >= 2 && value <= 40 && value % 2 == 0) return 40 + 20 - value / 2;
94 return -1;

96vector<int> checkout(int score, int maximum) {
97 int last = doubleDart(score);
98 if (last >= 0) return {last};
99 if (maximum >= 2) {
100 for (int i = 0; i < 62; ++i) {
101 last = doubleDart(score - darts[i].value);
102 if (last >= 0) return {i, last};
103 }
104 }
105 if (maximum >= 3 && score <= 170) {
106 for (int i = 0; i < 62; ++i) {
107 for (int j = 0; j < 62; ++j) {
108 last = doubleDart(score - darts[i].value - darts[j].value);
109 if (last >= 0) return {i, j, last};
110 }
111 }
112 }
113 return {};

120BowlingInfo bowlingInfo(const BowlingState &state) {
121 BowlingInfo info;
122 for (int i = 0; i < state.count; ++i) {
123 int frame = info.frame, pins = state.rolls[i];
124 if (info.count[frame] == 0) info.start[frame] = i;
125 info.marks[i] = info.fresh && pins == 10 ? 'X' : pins == info.standing ? '/' : pins == 0 ? '-' : char('0' + pins);
126 ++info.count[frame];
127 info.standing -= pins;
128 info.fresh = false;
129 bool ended = false;
130 if (frame < 9) ended = pins == 10 && info.count[frame] == 1 || info.count[frame] == 2;
131 else if (info.count[frame] == 2) ended = state.rolls[info.start[frame]] + pins < 10;
132 else if (info.count[frame] == 3) ended = true;
133 if (ended) {
134 info.closed[frame] = true;
135 if (frame == 9) info.finished = true;
136 else ++info.frame;
137 info.standing = 10;
138 info.fresh = true;
139 } else if (frame == 9 && info.standing == 0) {
140 info.standing = 10;
141 info.fresh = true;
142 }
143 }
144 for (int frame = 0; frame < 10; ++frame) {
145 if (!info.count[frame]) continue;
146 int start = info.start[frame];
147 if (frame == 9) {
148 info.final[frame] = info.closed[frame];
149 for (int j = 0; j < info.count[frame]; ++j) info.score[frame] += state.rolls[start + j];
150 } else if (state.rolls[start] == 10) {
151 info.final[frame] = state.count >= start + 3;
152 if (info.final[frame]) info.score[frame] = 10 + state.rolls[start + 1] + state.rolls[start + 2];
153 } else if (info.count[frame] >= 2) {
154 int total = state.rolls[start] + state.rolls[start + 1];
155 info.final[frame] = total < 10 || state.count >= start + 3;
156 if (info.final[frame]) info.score[frame] = total + (total == 10 ? state.rolls[start + 2] : 0);
157 }
158 }
159 return info;
160}
161int bowlingTotal(const BowlingInfo &info) {
162 return std::accumulate(info.score.begin(), info.score.end(), 0);
163}
164bool finalSet(const Match &m, const Rally &s) { return s.sets[0] == m.sets / 2 && s.sets[1] == m.sets / 2; }
165int tennisKind(const Match &m, const Rally &s) {
166 if (finalSet(m, s) && m.final == 3) return 2;
167 if (s.games[0] == 6 && s.games[1] == 6 && !(finalSet(m, s) && m.final == 2)) return 1;
168 return 0;

170int rallyServer(const Match &m, const Rally &s) {
171 if (m.sport == Tennis && tennisKind(m, s)) return s.server ^ (((s.points[0] + s.points[1] + 1) / 2) % 2);
172 if (m.sport == TableTennis) {
173 int first = m.first ^ (s.segmentCount % 2), played = s.points[0] + s.points[1];
174 return first ^ (played < 20 ? played / 2 % 2 : played % 2);
175 }
176 return s.server;
177}
178bool normalGame(const Match &m, int own, int other) { return own >= 4 && (m.noAd || own - other >= 2); }
179void addRally(const Match &m, Rally &s, int side) {
180 int other = 1 - side, server = rallyServer(m, s);
181 ++s.stats[side][0];
182 ++s.stats[server][1];
183 if (side == server) ++s.stats[side][2];
184 ++s.stats[side][3];
185 s.stats[other][3] = 0;
186 s.stats[side][4] = std::max(s.stats[side][4], s.stats[side][3]);
187 int kind = m.sport == Tennis ? tennisKind(m, s) : 0;
188 if (m.sport == Tennis && kind == 0) {
189 int receiver = 1 - s.server;
190 if (normalGame(m, s.points[receiver] + 1, s.points[s.server])) {
191 ++s.stats[receiver][8];
192 if (side == receiver) ++s.stats[receiver][9];
193 }
194 }
195 ++s.points[side];
196 if (m.sport == Tennis) {
197 int target = kind == 2 || kind == 1 && finalSet(m, s) && m.final == 1 ? 10 : 7;
198 bool ended = kind ? s.points[side] >= target && s.points[side] - s.points[other] >= 2 : normalGame(m, s.points[side], s.points[other]);
199 if (!ended) return;
200 Segment segment;
201 if (!kind) {
202 ++s.stats[s.server][5];
203 if (side == s.server) ++s.stats[side][6];
204 else ++s.stats[side][7];
205 }
206 if (kind == 2) segment = {s.points[0], s.points[1], 2, 0};
207 else {
208 ++s.games[side];
209 segment = {s.games[0], s.games[1], kind, kind ? s.points[other] : 0};
210 }
211 s.server ^= 1;
212 s.points = {};
213 bool setEnded = kind != 0 || s.games[side] >= 6 && s.games[side] - s.games[other] >= 2;
214 if (setEnded) {
215 s.segments[s.segmentCount++] = segment;
216 ++s.sets[side];
217 s.games = {};
218 if (s.sets[side] == m.sets / 2 + 1) s.winner = side;
219 }
220 return;
221 }
222 if (m.sport == Volleyball && side != s.server) {
223 s.server = side;
224 std::rotate(s.lineup[side].begin(), s.lineup[side].begin() + 1, s.lineup[side].end());
225 }
226 if (m.sport == Badminton) s.server = side;
227 int target = m.sport == TableTennis ? 11 : m.sport == Badminton ? 21 : s.segmentCount == 4 ? 15 : 25;
228 bool ended = s.points[side] >= target && s.points[side] - s.points[other] >= 2;
229 if (m.sport == Badminton && s.points[side] == 30) ended = true;
230 if (!ended) return;
231 s.segments[s.segmentCount++] = {s.points[0], s.points[1], 0, 0};
232 ++s.sets[side];
233 int needed = m.sport == TableTennis ? m.games / 2 + 1 : m.sport == Badminton ? 2 : 3;
234 if (s.sets[side] == needed) s.winner = side;
235 else {
236 s.points = {};
237 if (m.sport == Volleyball) {
238 s.lineup = m.initialLineup;
239 s.substitutions = {};
240 s.timeouts = {};
241 s.server = m.first ^ (s.segmentCount % 2);
242 }
243 }
244}
245void endDartsTurn(DartsState &s) {
246 s.server ^= 1;
247 s.turnStart = s.remaining[s.server];
248 s.turnCount = s.turnScore = 0;
249}
250void completedDartsTurn(DartsState &s, int side) {
251 s.stats[side][3] = std::max(s.stats[side][3], s.turnScore);
252 if (s.turnScore >= 100) ++s.stats[side][4];
253 if (s.turnScore == 180) ++s.stats[side][5];
254}
255void addDart(const Match &m, DartsState &s, int id) {
256 int side = s.server, remaining = s.remaining[side];
257 const Dart &dart = darts[id];
258 ++s.stats[side][0];
259 if (remaining == 50 || remaining <= 40 && remaining % 2 == 0) ++s.stats[side][6];
260 s.turn[s.turnCount++] = id;
261 s.turnScore += dart.value;
262 s.stats[side][1] += dart.value;
263 int after = remaining - dart.value;
264 if (after < 0 || after == 1 || after == 0 && !dart.isDouble) {
265 s.stats[side][1] -= s.turnScore;
266 ++s.stats[side][2];
267 s.remaining[side] = s.turnStart;
268 endDartsTurn(s);
269 return;
270 }
271 s.remaining[side] = after;
272 if (after == 0) {
273 ++s.stats[side][7];
274 completedDartsTurn(s, side);
275 ++s.legs[side];
276 if (s.legs[side] == m.legs) {
277 s.segments[s.segmentCount++] = s.legs;
278 ++s.sets[side];
279 if (s.sets[side] == m.sets) {
280 s.winner = side;
281 return;
282 }
283 s.legs = {};
284 }
285 ++s.legNumber;
286 s.remaining = {m.start, m.start};
287 s.server = m.first ^ ((s.legNumber - 1) % 2);
288 s.turnStart = m.start;
289 s.turnCount = s.turnScore = 0;
290 } else if (s.turnCount == 3) {
291 completedDartsTurn(s, side);
292 endDartsTurn(s);
293 }
294}

307string situation(const Match &m, const Rally &s, int side) {
308 Rally next = s;
309 addRally(m, next, side);
310 if (next.winner >= 0) return "match point";
311 if (next.segmentCount != s.segmentCount) return m.sport == TableTennis || m.sport == Badminton ? "game point" : "set point";
312 if (m.sport == Tennis && tennisKind(m, s) == 0 && next.server != s.server) return rallyServer(m, s) == side ? "game point" : "break point";
313 return "";

315string score(const Match &m) {
316 if (m.sport == Bowling) {
317 const BowlingState &s = std::get<BowlingState>(m.state);
318 BowlingInfo info = bowlingInfo(s);
319 string result = m.id + ": " + m.names[0];
320 int cumulative = 0;
321 for (int frame = 0; frame < 10; ++frame) {
322 if (!info.count[frame]) continue;
323 string display(info.marks.data() + info.start[frame], info.count[frame]);
324 cumulative += info.score[frame];
325 if (info.final[frame]) display += text("(", cumulative, ")");
326 result += " " + display;
327 }
328 return result;
329 }
330 int server = m.sport == Darts ? std::get<DartsState>(m.state).server : rallyServer(m, std::get<Rally>(m.state));
331 bool over = finished(m);
332 string result = m.id + ": " + m.names[0] + (!over && server == 0 ? "*" : "") + " vs " + m.names[1] + (!over && server == 1 ? "*" : "");
333 if (m.sport == Darts) {
334 const DartsState &s = std::get<DartsState>(m.state);
335 result += " sets " + pairText(s.sets) + " legs " + pairText(s.legs) + " score " + pairText(s.remaining);
336 if (!over) {
337 if (s.turnCount) result += " turn " + dartNames(vector<int>(s.turn.begin(), s.turn.begin() + s.turnCount));
338 vector<int> finish = checkout(s.remaining[s.server], 3 - s.turnCount);
339 if (!finish.empty()) result += " checkout " + dartNames(finish);
340 }
341 return result;
342 }
343 const Rally &s = std::get<Rally>(m.state);
344 for (int i = 0; i < s.segmentCount; ++i) result += " " + segmentText(s.segments[i]);
345 if (!over) {
346 if (m.sport == Tennis) {
347 int kind = tennisKind(m, s);
348 if (kind == 2) result += " [" + pairText(s.points) + "]";
349 else {
350 result += " " + pairText(s.games) + " ";
351 if (kind) result += pairText(s.points);
352 else if (!m.noAd && s.points[0] >= 3 && s.points[1] >= 3) result += s.points[0] == s.points[1] ? "40-40" : s.points[0] > s.points[1] ? "AD-40" : "40-AD";
353 else {
354 const string names[] = {"0", "15", "30", "40"};
355 result += names[s.points[0]] + "-" + names[s.points[1]];
356 }
357 }
358 } else {
359 result += " " + pairText(s.points);
360 if (m.sport == Volleyball) result += text(" server ", s.lineup[s.server][0]);
361 }
362 for (int side = 0; side < 2; ++side) {
363 string label = situation(m, s, side);
364 if (!label.empty()) result += " (" + label + " " + m.names[side] + ")";
365 }
366 }
367 return result;
368}
369vector<string> statistics(const Match &m) {
370 if (m.sport == Bowling) {
371 BowlingState s = std::get<BowlingState>(m.state);
372 BowlingInfo info = bowlingInfo(s);
373 int strikes = 0, spares = 0, open = 0;
374 for (int i = 0; i < s.count; ++i) {
375 strikes += info.marks[i] == 'X';
376 spares += info.marks[i] == '/';
377 }
378 for (int i = 0; i < 10; ++i) {
379 int start = info.start[i];
380 if (info.count[i] >= 2 && s.rolls[start] != 10 && s.rolls[start] + s.rolls[start + 1] < 10) ++open;
381 }
382 while (!info.finished) {
383 s.rolls[s.count++] = info.standing;
384 info = bowlingInfo(s);
385 }
386 return {text(m.id, " ", m.names[0], ": strikes ", strikes, " spares ", spares, " open ", open, " max ", bowlingTotal(info))};
387 }
388 vector<string> result;
389 for (int side = 0; side < 2; ++side) {
390 string line = m.id + " " + m.names[side] + ": ";
391 if (m.sport == Darts) {
392 const std::array<int, 8> &s = std::get<DartsState>(m.state).stats[side];
393 int average = s[0] ? s[1] * 300 / s[0] : 0;
394 line += text("darts ", s[0], " average ", average / 100, ".", average % 100 < 10 ? "0" : "", average % 100, " busts ", s[2], " high ", s[3], " tons ", s[4], " 180s ", s[5], " doubles ", s[7], "/", s[6]);
395 } else {
396 const Rally &r = std::get<Rally>(m.state);
397 const std::array<int, 10> &s = r.stats[side];
398 line += text("points ", s[0], " serve ", s[2], "/", s[1], " run ", s[4]);
399 if (m.sport == Tennis) line += text(" holds ", s[6], "/", s[5], " breaks ", s[7], " break points ", s[9], "/", s[8]);
400 if (m.sport == Volleyball) {
401 vector<string> lineup;
402 for (int number : r.lineup[side]) lineup.push_back(std::to_string(number));
403 line += text(" subs ", r.substitutions[side], " timeouts ", r.timeouts[side], " lineup ", join(lineup, ","));
404 }
405 }
406 result.push_back(line);
407 }
408 return result;
409}

411 if (!finished(m)) fail("match not finished");
412 if (m.sport == Bowling) return text(m.id, ": ", m.names[0], " ", bowlingTotal(bowlingInfo(std::get<BowlingState>(m.state))));
413 int first = winner(m);

415 if (m.sport == Darts) {
416 const DartsState &s = std::get<DartsState>(m.state);
417 for (int i = 0; i < s.segmentCount; ++i) line += " " + pairText(s.segments[i], first);
418 } else {
419 const Rally &s = std::get<Rally>(m.state);
420 for (int i = 0; i < s.segmentCount; ++i) line += " " + segmentText(s.segments[i], first);
421 }
422 return line;

424void createMatch(const vector<string> &a) {
425 if (matches.count(a[1])) fail("duplicate match");
426 Match m;
427 m.id = a[1];
428 const vector<string> sports = {"TENNIS", "TABLETENNIS", "BADMINTON", "VOLLEYBALL", "BOWLING", "DARTS"};
429 m.sport = Sport(std::find(sports.begin(), sports.end(), a[2]) - sports.begin());
430 m.names[0] = a[3];
431 if (m.sport != Bowling) m.names[1] = a[4];
432 if (m.sport == Darts) m.sets = 1;
433 int optionStart = m.sport == Bowling ? 4 : m.sport == Volleyball ? 7 : 5;
434 std::map<string, vector<string>> allowed;
435 if (m.sport == Tennis) allowed = {{"SETS", {"3", "5"}}, {"DEUCE", {"AD", "NOAD"}}, {"FINAL", {"TB7", "TB10", "ADV", "SUPER"}}, {"FIRST", {"A", "B"}}};
436 if (m.sport == TableTennis) allowed = {{"GAMES", {"5", "7"}}, {"FIRST", {"A", "B"}}};
437 if (m.sport == Badminton || m.sport == Volleyball) allowed = {{"FIRST", {"A", "B"}}};
438 if (m.sport == Darts) allowed = {{"START", {"301", "501", "701"}}, {"SETS", {"1", "2", "3", "4", "5"}}, {"LEGS", {"1", "2", "3", "4", "5"}}, {"FIRST", {"A", "B"}}};
439 std::map<string, bool> seen;
440 for (int i = optionStart; i < int(a.size()); ++i) {
441 vector<string> option = split(a[i], '=');
442 if (option.size() != 2 || !allowed.count(option[0]) || seen[option[0]]) fail("invalid option");
443 const vector<string> &values = allowed[option[0]];
444 if (std::find(values.begin(), values.end(), option[1]) == values.end()) fail("invalid option");
445 seen[option[0]] = true;
446 if (option[0] == "FIRST") m.first = option[1] == "B";
447 if (option[0] == "SETS") m.sets = std::stoi(option[1]);
448 if (option[0] == "GAMES") m.games = std::stoi(option[1]);
449 if (option[0] == "START") m.start = std::stoi(option[1]);
450 if (option[0] == "LEGS") m.legs = std::stoi(option[1]);
451 if (option[0] == "DEUCE") m.noAd = option[1] == "NOAD";
452 if (option[0] == "FINAL") m.final = int(std::find(values.begin(), values.end(), option[1]) - values.begin());
453 }
454 if (m.sport == Bowling) m.state = BowlingState{};
455 else if (m.sport == Darts) {
456 DartsState s;
457 s.remaining = {m.start, m.start};
458 s.server = m.first;
459 s.turnStart = m.start;
460 m.state = s;
461 } else {
462 Rally s;
463 s.server = m.first;
464 if (m.sport == Volleyball) {
465 for (int side = 0; side < 2; ++side) {
466 vector<string> numbers = split(a[5 + side], ',');
467 for (int i = 0; i < 6; ++i) m.initialLineup[side][i] = std::stoi(numbers[i]);
468 }
469 s.lineup = m.initialLineup;
470 }
471 m.state = s;
472 }
473 matches.emplace(a[1], std::move(m));
474}
475vector<string> execute(const vector<string> &a) {
476 const string &command = a[0];
477 if (command == "MATCH") {
478 createMatch(a);
479 return {};
480 }
481 if (command == "CHECKOUT") {
482 vector<int> finish = checkout(std::stoi(a[1]), 3);
483 return {"checkout " + a[1] + ": " + (finish.empty() ? "none" : dartNames(finish))};
484 }
485 if (command == "STANDINGS") {
486 std::map<string, Pair> totals;
487 for (const auto &[id, m] : matches) {
488 if (m.sport == Bowling || !finished(m)) continue;
489 int side = winner(m);
490 ++totals[m.names[side]][0];
491 ++totals[m.names[1 - side]][1];
492 }
493 struct Standing { string name; int wins, losses; };
494 vector<Standing> rows;
495 for (const auto &[name, counts] : totals) rows.push_back({name, counts[0], counts[1]});
496 std::sort(rows.begin(), rows.end(), [](const Standing &a, const Standing &b) {
497 if (a.wins != b.wins) return a.wins > b.wins;
498 if (a.losses != b.losses) return a.losses < b.losses;
499 return a.name < b.name;
500 });
501 vector<string> displays;
502 for (const Standing &row : rows) displays.push_back(text(row.name, " ", row.wins, "-", row.losses));
503 return {"standings: " + (displays.empty() ? "none" : join(displays, ", "))};
504 }
505 std::map<string, Match>::iterator found = matches.find(a[1]);
506 if (found == matches.end()) fail("unknown match");
507 Match &m = found->second;
508 if (command == "SCORE") return {score(m)};
509 if (command == "STATS") return statistics(m);
510 if (command == "RESULT") return {result(m)};
511 if (command == "UNDO") {
512 if (m.history.empty()) fail("nothing to undo");
513 m.state = m.history.back();
514 m.history.pop_back();
515 return {};
516 }
517 bool allowed = command == "POINT" && m.sport <= Volleyball || command == "ROLL" && m.sport == Bowling || command == "THROW" && m.sport == Darts || (command == "SUB" || command == "TIMEOUT") && m.sport == Volleyball;
518 if (!allowed) fail("wrong event");
519 int count = command == "POINT" ? int(a[2].size()) : command == "ROLL" || command == "THROW" ? int(a.size()) - 2 : 1;
520 for (int i = 0; i < count; ++i) {
521 if (finished(m)) fail("match over");
522 int value = 0, position = -1, side = 0;
523 if (command == "POINT") value = a[2][i] == 'B';
524 if (command == "ROLL") {
525 value = std::stoi(a[i + 2]);
526 if (value > bowlingInfo(std::get<BowlingState>(m.state)).standing) fail("too many pins");
527 }
528 if (command == "THROW") {
529 value = -1;
530 for (int j = 0; j < int(darts.size()); ++j) if (darts[j].name == a[i + 2]) value = j;
531 if (value < 0) fail("invalid dart");
532 }
533 if (command == "SUB" || command == "TIMEOUT") {
534 side = a[2] == "B";
535 const Rally &s = std::get<Rally>(m.state);
536 if (command == "TIMEOUT") {
537 if (s.timeouts[side] >= 2) fail("no timeouts left");
538 } else {
539 int out = std::stoi(a[3]);
540 value = std::stoi(a[4]);
541 for (int j = 0; j < 6; ++j) if (s.lineup[side][j] == out) position = j;
542 if (position < 0) fail("player not on court");
543 if (std::find(s.lineup[side].begin(), s.lineup[side].end(), value) != s.lineup[side].end()) fail("player already on court");
544 if (s.substitutions[side] >= 6) fail("no substitutions left");
545 }
546 }
547 m.history.push_back(m.state);
548 if (command == "POINT") addRally(m, std::get<Rally>(m.state), value);
549 if (command == "ROLL") {
550 BowlingState &s = std::get<BowlingState>(m.state);
551 s.rolls[s.count++] = value;
552 }
553 if (command == "THROW") addDart(m, std::get<DartsState>(m.state), value);
554 if (command == "SUB") {
555 Rally &s = std::get<Rally>(m.state);
556 s.lineup[side][position] = value;
557 ++s.substitutions[side];
558 }
559 if (command == "TIMEOUT") ++std::get<Rally>(m.state).timeouts[side];
560 }
561 return {};
562}
