11DARTS = {}
12for number in range(1, 21):
13 DARTS[f"S{number}"] = (number, False)
14 DARTS[f"D{number}"] = (2 * number, True)
15 DARTS[f"T{number}"] = (3 * number, False)
16DARTS["25"] = (25, False)
17DARTS["BULL"] = (50, True)
18DARTS["MISS"] = (0, False)
19CANDIDATES_A = [
20 f"{prefix}{number}"
21 for prefix in ("S", "T", "D")
22 for number in range(20, 0, -1)
23] + ["25", "BULL"]
24CANDIDATES_B = [f"D{number}" for number in range(20, 0, -1)] + ["BULL"]

27@cache
28def checkout(score, maximum=3):
29 for count in range(1, maximum + 1):
30 for prefix in product(CANDIDATES_A, repeat=count - 1):
31 wanted = score - sum(DARTS[dart][0] for dart in prefix)
32 for last in CANDIDATES_B:
33 if DARTS[last][0] == wanted:
34 return prefix + (last,)
35 return None

40 values = values[::-1]

72class Game:
73 def __init__(self, args):
74 self.id, self.sport = args[:2]
75 if self.sport == "BOWLING":
76 self.names = [args[2]]
77 option_start = 3
78 else:
79 self.names = args[2:4]
80 option_start = 4
81 self.lineups = [[], []]
82 if self.sport == "VOLLEYBALL":
83 self.lineups = [
84 [int(number) for number in lineup.split(",")]
85 for lineup in args[4:6]
86 ]
87 option_start = 6
88 self.options = DEFAULTS[self.sport].copy()
89 seen = set()
90 for option in args[option_start:]:
91 parts = option.split("=")
92 if (
93 len(parts) != 2 or parts[0] not in CHOICES[self.sport]

100 self.first = int(self.options.get("FIRST", "A") == "B")
101 self.state = self.new_state()
102 self.initial_state = deepcopy(self.state)

147 def rally_sport(self):
148 return self.sport in ("TENNIS", "TABLETENNIS", "BADMINTON", "VOLLEYBALL")

150 def final_set(self, state):
151 target = int(self.options["SETS"]) // 2
152 return state["sets"] == [target, target]

154 def tennis_kind(self, state):
155 if self.final_set(state) and self.options["FINAL"] == "SUPER":
156 return "super"
157 advantage = self.final_set(state) and self.options["FINAL"] == "ADV"
158 if state["games"] == [6, 6] and not advantage:
159 return "tie"
160 return "normal"

162 def server(self, state):
163 if self.sport == "TENNIS":
164 if self.tennis_kind(state) == "normal":
165 return state["game_server"]
166 point_number = sum(state["points"]) + 1
167 return state["game_server"] ^ (point_number // 2 % 2)
168 if self.sport == "TABLETENNIS":
169 first = self.first ^ (len(state["finished"]) % 2)
170 played = sum(state["points"])
171 alternation = played // 2 % 2 if played < 20 else played % 2
172 return first ^ alternation
173 return state["server"]

175 def normal_game_wins(self, points, winner):
176 return points[winner] >= 4 and (
177 self.options["DEUCE"] == "NOAD"
178 or points[winner] - points[1 - winner] >= 2
179 )

181 def finish_rally_unit(self, state, winner, record, target):
182 state["finished"].append(record)
183 state["sets"][winner] += 1
184 if state["sets"][winner] == target:
185 state["done"] = True
186 state["winner"] = winner
187 return
188 state["points"] = [0, 0]
189 state["games"] = [0, 0]
190 if self.sport == "VOLLEYBALL":
191 state["lineups"] = [lineup.copy() for lineup in self.lineups]
192 state["subs"] = [0, 0]
193 state["timeouts"] = [0, 0]
194 state["server"] = self.first ^ (len(state["finished"]) % 2)

196 def tennis_point(self, state, winner, server):
197 kind = self.tennis_kind(state)
198 if kind == "normal":
199 receiver = 1 - server
200 hypothetical = state["points"].copy()
201 hypothetical[receiver] += 1
202 if self.normal_game_wins(hypothetical, receiver):
203 state["rally"][receiver]["opportunities"] += 1
204 if winner == receiver:
205 state["rally"][receiver]["conversions"] += 1
206 state["points"][winner] += 1
207 needed = int(self.options["SETS"]) // 2 + 1
208 if kind == "normal":
209 if not self.normal_game_wins(state["points"], winner):
210 return
211 state["rally"][server]["service_games"] += 1
212 if winner == server:
213 state["rally"][server]["holds"] += 1
214 else:
215 state["rally"][winner]["breaks"] += 1
216 state["games"][winner] += 1
217 state["points"] = [0, 0]
218 state["game_server"] ^= 1
219 if (
220 state["games"][winner] >= 6
221 and state["games"][winner] - state["games"][1 - winner] >= 2
222 ):
223 record = {
224 "values": state["games"].copy(), "tie": -1, "super": False
225 }
226 self.finish_rally_unit(state, winner, record, needed)
227 return
228 ten_point = kind == "super" or (
229 self.final_set(state) and self.options["FINAL"] == "TB10"
230 )
231 target = 10 if ten_point else 7
232 if (
233 state["points"][winner] < target
234 or state["points"][winner] - state["points"][1 - winner] < 2
235 ):
236 return
237 if kind == "super":
238 record = {
239 "values": state["points"].copy(), "tie": -1, "super": True
240 }
241 else:
242 values = state["games"].copy()
243 values[winner] += 1
244 record = {
245 "values": values,
246 "tie": state["points"][1 - winner],
247 "super": False,
248 }
249 state["game_server"] ^= 1
250 self.finish_rally_unit(state, winner, record, needed)

252 def point(self, state, winner):
253 serving = self.server(state)
254 stats = state["rally"]
255 stats[winner]["points"] += 1
256 stats[serving]["served"] += 1
257 if winner == serving:
258 stats[serving]["serve_won"] += 1
259 stats[winner]["current"] += 1
260 stats[winner]["run"] = max(
261 stats[winner]["run"], stats[winner]["current"]
262 )
263 stats[1 - winner]["current"] = 0
264 if self.sport == "TENNIS":
265 self.tennis_point(state, winner, serving)
266 return
267 state["points"][winner] += 1
268 if self.sport == "BADMINTON":
269 state["server"] = winner
270 elif self.sport == "VOLLEYBALL" and winner != state["server"]:
271 state["server"] = winner
272 lineup = state["lineups"][winner]
273 lineup.append(lineup.pop(0))
274 if self.sport == "TABLETENNIS":
275 target = 11
276 needed = int(self.options["GAMES"]) // 2 + 1
277 elif self.sport == "BADMINTON":
278 target = 21
279 needed = 2
280 else:
281 target = 15 if len(state["finished"]) == 4 else 25
282 needed = 3
283 won = (
284 state["points"][winner] >= target
285 and state["points"][winner] - state["points"][1 - winner] >= 2
286 ) or (self.sport == "BADMINTON" and state["points"][winner] == 30)
287 if won:
288 record = {
289 "values": state["points"].copy(), "tie": -1, "super": False
290 }
291 self.finish_rally_unit(state, winner, record, needed)

293 def standing(self, state):
294 if state["frame"] == 9 and state["standing"] == 0:
295 return 10
296 return state["standing"]

298 def roll(self, state, pins):
299 standing = self.standing(state)
300 if pins > standing:
301 raise Invalid("too many pins")
302 frame = state["frames"][state["frame"]]
303 fresh_rack = not frame or state["frame"] == 9 and state["standing"] == 0
304 if fresh_rack and pins == 10:
305 mark = "X"
306 elif pins == standing:
307 mark = "/"
308 elif pins == 0:
309 mark = "-"
310 else:
311 mark = str(pins)
312 frame.append(pins)
313 state["marks"][state["frame"]].append(mark)
314 state["standing"] = standing - pins
315 if state["frame"] < 9:
316 if frame == [10] or len(frame) == 2:
317 state["frame"] += 1
318 state["standing"] = 10
319 elif len(frame) == 3 or (
320 len(frame) == 2 and frame[0] != 10 and sum(frame) < 10
321 ):
322 state["done"] = True
323
324 def frame_scores(self, state):
325 scores = []
326 for i, frame in enumerate(state["frames"]):
327 value = None
328 if i == 9:
329 if state["done"]:
330 value = sum(frame)
331 else:
332 later = [
333 pins for following in state["frames"][i + 1:]
334 for pins in following
335 ]
336 if frame and frame[0] == 10:
337 if len(later) >= 2:
338 value = 10 + later[0] + later[1]
339 elif len(frame) == 2:
340 if sum(frame) < 10:
341 value = sum(frame)
342 elif later:
343 value = 10 + later[0]
344 scores.append(value)
345 return scores

347 def maximum(self, state):
348 hypothetical = deepcopy(state)
349 while not hypothetical["done"]:
350 self.roll(hypothetical, self.standing(hypothetical))
351 return sum(self.frame_scores(hypothetical))

353 def completed_turn(self, state, player):
354 stats = state["darts"][player]
355 stats["high"] = max(stats["high"], state["turn_score"])
356 if state["turn_score"] >= 100:
357 stats["tons"] += 1
358 if state["turn_score"] == 180:
359 stats["maximums"] += 1

361 def next_turn(self, state):
362 state["player"] = 1 - state["player"]
363 state["turn"] = []
364 state["turn_score"] = 0
365 state["turn_start"] = state["remaining"][state["player"]]

367 def win_leg(self, state, player):
368 state["legs"][player] += 1
369 if state["legs"][player] == int(self.options["LEGS"]):
370 state["sets"][player] += 1
371 state["finished"].append({
372 "values": state["legs"].copy(), "tie": -1, "super": False
373 })
374 if state["sets"][player] == int(self.options["SETS"]):
375 state["done"] = True
376 state["winner"] = player
377 return
378 state["legs"] = [0, 0]
379 state["leg_number"] += 1
380 start = int(self.options["START"])
381 state["remaining"] = [start, start]
382 state["player"] = self.first ^ (state["leg_number"] % 2)
383 state["turn"] = []
384 state["turn_score"] = 0
385 state["turn_start"] = start

387 def throw_dart(self, state, notation):
388 if notation not in DARTS:
389 raise Invalid("invalid dart")
390 value, double = DARTS[notation]
391 player = state["player"]
392 stats = state["darts"][player]
393 before = state["remaining"][player]
394 stats["darts"] += 1
395 if before == 50 or before <= 40 and before % 2 == 0:
396 stats["attempts"] += 1
397 state["turn"].append(notation)
398 state["turn_score"] += value
399 stats["points"] += value
400 remaining = before - value
401 if remaining < 0 or remaining == 1 or remaining == 0 and not double:
402 stats["busts"] += 1
403 stats["points"] -= state["turn_score"]
404 state["remaining"][player] = state["turn_start"]
405 self.next_turn(state)
406 return
407 state["remaining"][player] = remaining
408 if remaining == 0:
409 stats["hits"] += 1
410 self.completed_turn(state, player)
411 self.win_leg(state, player)
412 elif len(state["turn"]) == 3:
413 self.completed_turn(state, player)
414 self.next_turn(state)

416 def apply(self, name, args):
417 if name == "POINT":
418 self.point(self.state, int(args[0] == "B"))
419 elif name == "ROLL":
420 self.roll(self.state, int(args[0]))
421 elif name == "THROW":
422 self.throw_dart(self.state, args[0])
423 else:
424 side = int(args[0] == "B")
425 if name == "SUB":
426 lineup = self.state["lineups"][side]
427 outgoing, incoming = map(int, args[1:])
428 if outgoing not in lineup:
429 raise Invalid("player not on court")
430 if incoming in lineup:
431 raise Invalid("player already on court")
432 if self.state["subs"][side] >= 6:
433 raise Invalid("no substitutions left")
434 lineup[lineup.index(outgoing)] = incoming
435 self.state["subs"][side] += 1
436 else:
437 if self.state["timeouts"][side] >= 2:
438 raise Invalid("no timeouts left")
439 self.state["timeouts"][side] += 1

441 def event(self, name, args):
442 if name == "POINT":
443 allowed = self.rally_sport()
444 events = [(side,) for side in args[0]]
445 elif name == "ROLL":
446 allowed = self.sport == "BOWLING"
447 events = [(pins,) for pins in args]
448 elif name == "THROW":
449 allowed = self.sport == "DARTS"
450 events = [(dart,) for dart in args]
451 else:
452 allowed = self.sport == "VOLLEYBALL"
453 events = [tuple(args)]
454 if not allowed:
455 raise Invalid("wrong event")
456 for values in events:
457 if self.state["done"]:
458 raise Invalid("match over")
459 self.apply(name, values)
460 self.history.append((name, values))

462 def undo(self):
463 if not self.history:
464 raise Invalid("nothing to undo")
465 self.history.pop()
466 self.state = deepcopy(self.initial_state)
467 for name, args in self.history:
468 self.apply(name, args)

470 def score_record(self, record, reverse=False):
471 display = pair(record["values"], reverse)
472 if record["super"]:
473 return "[" + display + "]"
474 if record["tie"] >= 0:
475 display += f"({record['tie']})"

478 def situation(self, side):
479 state = self.state
480 hypothetical = deepcopy(state)
481 serving = self.server(state)
482 normal = self.sport == "TENNIS" and self.tennis_kind(state) == "normal"
483 self.point(hypothetical, side)
484 if hypothetical["done"]:
485 return "match point"
486 if len(hypothetical["finished"]) > len(state["finished"]):
487 if self.sport in ("TENNIS", "VOLLEYBALL"):
488 return "set point"
489 return "game point"
490 if normal and hypothetical["game_server"] != state["game_server"]:
491 return "game point" if side == serving else "break point"
492 return None

494 def bowling_score(self):
495 state = self.state
496 scores = self.frame_scores(state)
497 total = 0
498 items = []
499 for i, frame in enumerate(state["frames"]):
500 if not frame:
501 break
502 display = "".join(state["marks"][i])
503 if scores[i] is not None:
504 total += scores[i]
505 display += f"({total})"
506 items.append(display)
507 suffix = " " + " ".join(items) if items else ""
508 return f"{self.id}: {self.names[0]}{suffix}"

510 def tennis_current(self):
511 state = self.state
512 kind = self.tennis_kind(state)
513 if kind == "super":
514 return ["[" + pair(state["points"]) + "]"]
515 items = [pair(state["games"])]
516 if kind == "tie":
517 items.append(pair(state["points"]))
518 elif self.options["DEUCE"] == "AD" and min(state["points"]) >= 3:
519 a, b = state["points"]
520 if a == b:
521 items.append("40-40")
522 elif a > b:
523 items.append("AD-40")
524 else:
525 items.append("40-AD")
526 else:
527 display = ("0", "15", "30", "40")
528 items.append("-".join(display[value] for value in state["points"]))
529 return items

531 def score(self):
532 state = self.state
533 if self.sport == "BOWLING":
534 return self.bowling_score()
535 serving = state["player"] if self.sport == "DARTS" else self.server(state)
536 names = [
537 name + ("*" if not state["done"] and i == serving else "")
538 for i, name in enumerate(self.names)
539 ]
540 items = [self.score_record(record) for record in state["finished"]]
541 if self.sport == "DARTS":
542 items = [
543 "sets", pair(state["sets"]), "legs", pair(state["legs"]),
544 "score", pair(state["remaining"]),
545 ]
546 if not state["done"]:
547 if state["turn"]:
548 items.extend(["turn", *state["turn"]])
549 finish = checkout(
550 state["remaining"][state["player"]], 3 - len(state["turn"])
551 )
552 if finish:
553 items.extend(["checkout", *finish])
554 elif not state["done"]:
555 if self.sport == "TENNIS":
556 items.extend(self.tennis_current())
557 else:
558 items.append(pair(state["points"]))
559 if self.sport == "VOLLEYBALL":
560 number = state["lineups"][state["server"]][0]
561 items.extend(["server", str(number)])
562 for side in range(2):
563 status = self.situation(side)
564 if status:
565 items.append(f"({status} {self.names[side]})")
566 return f"{self.id}: {names[0]} vs {names[1]} " + " ".join(items)

568 def stats(self):
569 state = self.state
570 if self.sport == "BOWLING":
571 marks = [mark for frame in state["marks"] for mark in frame]
572 open_frames = sum(
573 len(frame) >= 2 and frame[0] != 10 and sum(frame[:2]) < 10
574 for frame in state["frames"]
575 )
576 return [
577 f"{self.id} {self.names[0]}: strikes {marks.count('X')}"
578 f" spares {marks.count('/')} open {open_frames}"
579 f" max {self.maximum(state)}"
580 ]
581 output = []
582 for i, name in enumerate(self.names):
583 line = f"{self.id} {name}: "
584 if self.sport == "DARTS":
585 stats = state["darts"][i]
586 average = (
587 stats["points"] * 300 // stats["darts"]
588 if stats["darts"] else 0
589 )
590 line += (
591 f"darts {stats['darts']}"
592 f" average {average // 100}.{average % 100:02d}"
593 f" busts {stats['busts']} high {stats['high']}"
594 f" tons {stats['tons']} 180s {stats['maximums']}"
595 f" doubles {stats['hits']}/{stats['attempts']}"
596 )
597 else:
598 stats = state["rally"][i]
599 line += (
600 f"points {stats['points']}"
601 f" serve {stats['serve_won']}/{stats['served']}"
602 f" run {stats['run']}"
603 )
604 if self.sport == "TENNIS":
605 line += (
606 f" holds {stats['holds']}/{stats['service_games']}"
607 f" breaks {stats['breaks']}"
608 f" break points {stats['conversions']}/{stats['opportunities']}"
609 )
610 elif self.sport == "VOLLEYBALL":
611 lineup = ",".join(map(str, state["lineups"][i]))
612 line += (
613 f" subs {state['subs'][i]} timeouts {state['timeouts'][i]}"
614 f" lineup {lineup}"
615 )
616 output.append(line)
617 return output

619 def result(self):
620 state = self.state
621 if not state["done"]:
622 raise Invalid("match not finished")
623 if self.sport == "BOWLING":
624 total = sum(self.frame_scores(state))
625 return f"{self.id}: {self.names[0]} {total}"
626 winner = state["winner"]
627 scores = " ".join(
628 self.score_record(record, reverse=winner == 1)
629 for record in state["finished"]
630 )
631 return (
632 f"{self.id}: {self.names[winner]} def. {self.names[1 - winner]}"
633 f" {scores}"
634 )

637class Scoreboard:
638 def __init__(self):
639 self.matches = {}
640
641 def standings(self):
642 totals = {}
643 for match in self.matches.values():
644 if not match.state["done"] or match.sport == "BOWLING":
645 continue
646 for side, name in enumerate(match.names):
647 totals.setdefault(name, [0, 0])
648 index = 0 if side == match.state["winner"] else 1
649 totals[name][index] += 1
650 names = sorted(
651 totals, key=lambda name: (-totals[name][0], totals[name][1], name)
652 )
653 display = ", ".join(
654 f"{name} {pair(totals[name])}" for name in names
655 )
656 return "standings: " + (display or "none")
657
658 def command(self, line):
659 name, *args = line.split()
660 if name == "MATCH":
661 if args[0] in self.matches:
662 raise Invalid("duplicate match")
663 match = Game(args)
664 self.matches[match.id] = match
665 return []
666 if name == "CHECKOUT":
667 answer = checkout(int(args[0]))
668 display = " ".join(answer) if answer else "none"
669 return [f"checkout {args[0]}: {display}"]
670 if name == "STANDINGS":
671 return [self.standings()]
672 if args[0] not in self.matches:
673 raise Invalid("unknown match")
674 match = self.matches[args[0]]
675 if name == "UNDO":
676 match.undo()
677 return []
678 if name == "SCORE":
679 return [match.score()]
680 if name == "STATS":
681 return match.stats()
682 if name == "RESULT":
683 return [match.result()]
684 match.event(name, args[1:])
685 return []
