105fn initialize_darts() -> Vec<Dart> {
106 let mut darts = Vec::new();
107 for multiplier in [1, 3, 2] {
108 let prefix = if multiplier == 1 {
109 'S'
110 } else if multiplier == 2 {
111 'D'
112 } else {
113 'T'
114 };
115 for n in (1..=20).rev() {
116 darts.push(Dart {
117 name: format!("{prefix}{n}"),
118 value: multiplier * n,
119 double: multiplier == 2,
120 });
121 }
122 }
123 darts.push(Dart {
124 name: "25".to_owned(),
125 value: 25,
126 double: false,
127 });
128 darts.push(Dart {
129 name: "BULL".to_owned(),
130 value: 50,
131 double: true,
132 });
133 darts.push(Dart {
134 name: "MISS".to_owned(),
135 value: 0,
136 double: false,
137 });
138 darts
139}
140fn double_dart(value: i32) -> Option<usize> {
141 if value == 50 {
142 Some(61)
143 } else if (2..=40).contains(&value) && value % 2 == 0 {
144 Some((60 - value / 2) as usize)
145 } else {
146 None
147 }
148}
149fn checkout(darts: &[Dart], score: i32, maximum: usize) -> Vec<usize> {
150 if let Some(last) = double_dart(score) {
151 return vec![last];
152 }
153 if maximum >= 2 {
154 for i in 0..62 {
155 if let Some(last) = double_dart(score - darts[i].value) {
156 return vec![i, last];
157 }
158 }
159 }
160 if maximum >= 3 && score <= 170 {
161 for i in 0..62 {
162 for j in 0..62 {
163 if let Some(last) = double_dart(score - darts[i].value - darts[j].value) {
164 return vec![i, j, last];
165 }
166 }
167 }
168 }
169 Vec::new()
170}
171fn dart_names(darts: &[Dart], ids: &[usize]) -> String {

173 .map(|&i| darts[i].name.as_str())
174 .collect::<Vec<_>>()
175 .join(" ")

177fn bowling_info(state: &BowlingState) -> BowlingInfo {
178 let mut info = BowlingInfo {
179 standing: 10,
180 fresh: true,
181 ..BowlingInfo::default()
182 };
183 for i in 0..state.count {
184 let frame = info.frame;
185 let pins = state.rolls[i];
186 if info.count[frame] == 0 {
187 info.start[frame] = i;
188 }
189 info.marks[i] = if info.fresh && pins == 10 {
190 b'X'
191 } else if pins == info.standing {
192 b'/'
193 } else if pins == 0 {
194 b'-'
195 } else {
196 b'0' + pins as u8
197 };
198 info.count[frame] += 1;
199 info.standing -= pins;
200 info.fresh = false;
201 let ended = if frame < 9 {
202 pins == 10 && info.count[frame] == 1 || info.count[frame] == 2
203 } else if info.count[frame] == 2 {
204 state.rolls[info.start[frame]] + pins < 10
205 } else {
206 info.count[frame] == 3
207 };
208 if ended {
209 info.closed[frame] = true;
210 if frame == 9 {
211 info.finished = true;
212 } else {
213 info.frame += 1;
214 }
215 info.standing = 10;
216 info.fresh = true;
217 } else if frame == 9 && info.standing == 0 {
218 info.standing = 10;
219 info.fresh = true;
220 }
221 }
222 for frame in 0..10 {
223 if info.count[frame] == 0 {
224 continue;
225 }
226 let start = info.start[frame];
227 if frame == 9 {
228 info.final_score[frame] = info.closed[frame];
229 info.score[frame] = state.rolls[start..start + info.count[frame]].iter().sum();
230 } else if state.rolls[start] == 10 {
231 info.final_score[frame] = state.count >= start + 3;
232 if info.final_score[frame] {
233 info.score[frame] = 10 + state.rolls[start + 1] + state.rolls[start + 2];
234 }
235 } else if info.count[frame] >= 2 {
236 let total = state.rolls[start] + state.rolls[start + 1];
237 info.final_score[frame] = total < 10 || state.count >= start + 3;
238 if info.final_score[frame] {
239 info.score[frame] = total
240 + if total == 10 {
241 state.rolls[start + 2]
242 } else {
243 0
244 };
245 }
246 }
247 }
248 info
249}
250fn final_set(c: &Config, s: &Rally) -> bool {
251 s.sets[0] == c.sets / 2 && s.sets[1] == c.sets / 2
252}
253fn tennis_kind(c: &Config, s: &Rally) -> usize {
254 if final_set(c, s) && c.final_rule == 3 {
255 2
256 } else if s.games == [6, 6] && !(final_set(c, s) && c.final_rule == 2) {
257 1
258 } else {
259 0
260 }
261}
262fn rally_server(c: &Config, s: &Rally) -> usize {
263 if c.sport == Sport::Tennis && tennis_kind(c, s) > 0 {
264 s.server ^ (((s.points[0] + s.points[1] + 1) / 2 % 2) as usize)
265 } else if c.sport == Sport::TableTennis {
266 let first = c.first ^ (s.segment_count % 2);
267 let played = s.points[0] + s.points[1];
268 first
269 ^ ((if played < 20 {
270 played / 2 % 2
271 } else {
272 played % 2
273 }) as usize)
274 } else {
275 s.server
276 }
277}
278fn normal_game(c: &Config, own: i32, other: i32) -> bool {
279 own >= 4 && (c.no_ad || own - other >= 2)
280}
281fn add_rally(c: &Config, s: &mut Rally, side: usize) {
282 let other = 1 - side;
283 let server = rally_server(c, s);
284 s.stats[side][0] += 1;
285 s.stats[server][1] += 1;
286 if side == server {
287 s.stats[side][2] += 1;
288 }
289 s.stats[side][3] += 1;
290 s.stats[other][3] = 0;
291 s.stats[side][4] = s.stats[side][4].max(s.stats[side][3]);
292 let kind = if c.sport == Sport::Tennis {
293 tennis_kind(c, s)
294 } else {
295 0
296 };
297 if c.sport == Sport::Tennis && kind == 0 {
298 let receiver = 1 - s.server;
299 if normal_game(c, s.points[receiver] + 1, s.points[s.server]) {
300 s.stats[receiver][8] += 1;
301 if side == receiver {
302 s.stats[receiver][9] += 1;
303 }
304 }
305 }
306 s.points[side] += 1;
307 if c.sport == Sport::Tennis {
308 let target = if kind == 2 || kind == 1 && final_set(c, s) && c.final_rule == 1 {
309 10
310 } else {
311 7
312 };
313 let ended = if kind > 0 {
314 s.points[side] >= target && s.points[side] - s.points[other] >= 2
315 } else {
316 normal_game(c, s.points[side], s.points[other])
317 };
318 if !ended {
319 return;
320 }
321 if kind == 0 {
322 s.stats[s.server][5] += 1;
323 if side == s.server {
324 s.stats[side][6] += 1;
325 } else {
326 s.stats[side][7] += 1;
327 }
328 }
329 let segment = if kind == 2 {
330 Segment {
331 a: s.points[0],
332 b: s.points[1],
333 kind: 2,
334 loser: 0,
335 }
336 } else {
337 s.games[side] += 1;
338 Segment {
339 a: s.games[0],
340 b: s.games[1],
341 kind,
342 loser: if kind > 0 { s.points[other] } else { 0 },
343 }
344 };
345 s.server ^= 1;
346 s.points = [0; 2];
347 if kind > 0 || s.games[side] >= 6 && s.games[side] - s.games[other] >= 2 {
348 s.segments[s.segment_count] = segment;
349 s.segment_count += 1;
350 s.sets[side] += 1;
351 s.games = [0; 2];
352 if s.sets[side] == c.sets / 2 + 1 {
353 s.winner = Some(side);
354 }
355 }
356 return;
357 }
358 if c.sport == Sport::Volleyball && side != s.server {
359 s.server = side;
360 s.lineup[side].rotate_left(1);
361 }
362 if c.sport == Sport::Badminton {
363 s.server = side;
364 }
365 let target = match c.sport {
366 Sport::TableTennis => 11,
367 Sport::Badminton => 21,
368 _ => {
369 if s.segment_count == 4 {
370 15
371 } else {
372 25
373 }
374 }
375 };
376 let ended = s.points[side] >= target && s.points[side] - s.points[other] >= 2
377 || c.sport == Sport::Badminton && s.points[side] == 30;
378 if !ended {
379 return;
380 }
381 s.segments[s.segment_count] = Segment {
382 a: s.points[0],
383 b: s.points[1],
384 kind: 0,
385 loser: 0,
386 };
387 s.segment_count += 1;
388 s.sets[side] += 1;
389 let needed = match c.sport {
390 Sport::TableTennis => c.games / 2 + 1,
391 Sport::Badminton => 2,
392 _ => 3,
393 };
394 if s.sets[side] == needed {
395 s.winner = Some(side);
396 } else {
397 s.points = [0; 2];
398 if c.sport == Sport::Volleyball {
399 s.lineup = c.initial_lineup;
400 s.substitutions = [0; 2];
401 s.timeouts = [0; 2];
402 s.server = c.first ^ (s.segment_count % 2);
403 }
404 }
405}
406fn end_darts_turn(s: &mut DartsState) {
407 s.server ^= 1;
408 s.turn_start = s.remaining[s.server];
409 s.turn_count = 0;
410 s.turn_score = 0;
411}
412fn completed_darts_turn(s: &mut DartsState, side: usize) {
413 s.stats[side][3] = s.stats[side][3].max(s.turn_score);
414 if s.turn_score >= 100 {
415 s.stats[side][4] += 1;
416 }
417 if s.turn_score == 180 {
418 s.stats[side][5] += 1;
419 }
420}
421fn add_dart(c: &Config, darts: &[Dart], s: &mut DartsState, id: usize) {
422 let side = s.server;
423 let remaining = s.remaining[side];
424 let dart = &darts[id];
425 s.stats[side][0] += 1;
426 if remaining == 50 || remaining <= 40 && remaining % 2 == 0 {
427 s.stats[side][6] += 1;
428 }
429 s.turn[s.turn_count] = id;
430 s.turn_count += 1;
431 s.turn_score += dart.value;
432 s.stats[side][1] += dart.value;
433 let after = remaining - dart.value;
434 if after < 0 || after == 1 || after == 0 && !dart.double {
435 s.stats[side][1] -= s.turn_score;
436 s.stats[side][2] += 1;
437 s.remaining[side] = s.turn_start;
438 end_darts_turn(s);
439 return;
440 }
441 s.remaining[side] = after;
442 if after == 0 {
443 s.stats[side][7] += 1;
444 completed_darts_turn(s, side);
445 s.legs[side] += 1;
446 if s.legs[side] == c.legs {
447 s.segments[s.segment_count] = s.legs;
448 s.segment_count += 1;
449 s.sets[side] += 1;
450 if s.sets[side] == c.sets {
451 s.winner = Some(side);
452 return;
453 }
454 s.legs = [0; 2];
455 }
456 s.leg_number += 1;
457 s.remaining = [c.start; 2];
458 s.server = c.first ^ ((s.leg_number - 1) % 2);
459 s.turn_start = c.start;
460 s.turn_count = 0;
461 s.turn_score = 0;
462 } else if s.turn_count == 3 {
463 completed_darts_turn(s, side);
464 end_darts_turn(s);
465 }
466}
467fn segment_text(s: Segment, first: usize) -> String {
468 let mut result = pair_text([s.a, s.b], first);
469 if s.kind == 1 {
470 result.push_str(&format!("({})", s.loser));
471 }
472 if s.kind == 2 {
473 result = format!("[{result}]");
474 }
475 result
476}
477fn situation(c: &Config, s: &Rally, side: usize) -> &'static str {
478 let mut next = *s;
479 add_rally(c, &mut next, side);
480 if next.winner.is_some() {
481 "match point"
482 } else if next.segment_count != s.segment_count {
483 if [Sport::TableTennis, Sport::Badminton].contains(&c.sport) {
484 "game point"
485 } else {
486 "set point"
487 }
488 } else if c.sport == Sport::Tennis && tennis_kind(c, s) == 0 && next.server != s.server {
489 if rally_server(c, s) == side {
490 "game point"
491 } else {
492 "break point"
493 }
494 } else {
495 ""
496 }
497}
498impl Match {
499 fn finished(&self) -> bool {
500 match self.state {
501 State::Bowling(s) => bowling_info(&s).finished,
502 State::Darts(s) => s.winner.is_some(),
503 State::Rally(s) => s.winner.is_some(),
504 }
505 }
506 fn winner(&self) -> usize {
507 match self.state {
508 State::Darts(s) => s.winner.unwrap(),
509 State::Rally(s) => s.winner.unwrap(),
510 _ => unreachable!(),
511 }
512 }
513 fn score(&self, darts: &[Dart]) -> String {
514 if let State::Bowling(s) = self.state {
515 let info = bowling_info(&s);
516 let mut line = format!("{}: {}", self.id, self.names[0]);
517 let mut cumulative = 0;
518 for frame in 0..10 {
519 if info.count[frame] == 0 {
520 continue;
521 }
522 let start = info.start[frame];
523 let marks =
524 std::str::from_utf8(&info.marks[start..start + info.count[frame]]).unwrap();
525 line.push_str(&format!(" {marks}"));
526 cumulative += info.score[frame];
527 if info.final_score[frame] {
528 line.push_str(&format!("({cumulative})"));
529 }
530 }
531 return line;
532 }
533 let over = self.finished();
534 let server = match self.state {
535 State::Darts(s) => s.server,
536 State::Rally(s) => rally_server(&self.config, &s),
537 _ => unreachable!(),
538 };
539 let mut line = format!(
540 "{}: {}{} vs {}{}",
541 self.id,
542 self.names[0],
543 if !over && server == 0 { "*" } else { "" },
544 self.names[1],
545 if !over && server == 1 { "*" } else { "" }
546 );
547 match self.state {
548 State::Darts(s) => {
549 line.push_str(&format!(
550 " sets {} legs {} score {}",
551 pair_text(s.sets, 0),
552 pair_text(s.legs, 0),
553 pair_text(s.remaining, 0)
554 ));
555 if !over {
556 if s.turn_count > 0 {
557 line.push_str(&format!(
558 " turn {}",
559 dart_names(darts, &s.turn[..s.turn_count])
560 ));
561 }
562 let finish = checkout(darts, s.remaining[s.server], 3 - s.turn_count);
563 if !finish.is_empty() {
564 line.push_str(&format!(" checkout {}", dart_names(darts, &finish)));
565 }
566 }
567 }
568 State::Rally(s) => {
569 for segment in &s.segments[..s.segment_count] {
570 line.push_str(&format!(" {}", segment_text(*segment, 0)));
571 }
572 if !over {
573 if self.config.sport == Sport::Tennis {
574 let kind = tennis_kind(&self.config, &s);
575 if kind == 2 {
576 line.push_str(&format!(" [{}]", pair_text(s.points, 0)));
577 } else {
578 line.push_str(&format!(" {} ", pair_text(s.games, 0)));
579 if kind > 0 {
580 line.push_str(&pair_text(s.points, 0));
581 } else if !self.config.no_ad && s.points[0] >= 3 && s.points[1] >= 3 {
582 line.push_str(if s.points[0] == s.points[1] {
583 "40-40"
584 } else if s.points[0] > s.points[1] {
585 "AD-40"
586 } else {
587 "40-AD"
588 });
589 } else {
590 let labels = ["0", "15", "30", "40"];
591 line.push_str(&format!(
592 "{}-{}",
593 labels[s.points[0] as usize], labels[s.points[1] as usize]
594 ));
595 }
596 }
597 } else {
598 line.push_str(&format!(" {}", pair_text(s.points, 0)));
599 if self.config.sport == Sport::Volleyball {
600 line.push_str(&format!(" server {}", s.lineup[s.server][0]));
601 }
602 }
603 for side in 0..2 {
604 let label = situation(&self.config, &s, side);
605 if !label.is_empty() {
606 line.push_str(&format!(" ({label} {})", self.names[side]));
607 }
608 }
609 }
610 }
611 _ => unreachable!(),
612 }
613 line
614 }
615 fn statistics(&self) -> Vec<String> {
616 if let State::Bowling(mut s) = self.state {
617 let mut info = bowling_info(&s);
618 let strikes = info.marks[..s.count].iter().filter(|&&c| c == b'X').count();
619 let spares = info.marks[..s.count].iter().filter(|&&c| c == b'/').count();
620 let mut open = 0;
621 for i in 0..10 {
622 let start = info.start[i];
623 if info.count[i] >= 2
624 && s.rolls[start] != 10
625 && s.rolls[start] + s.rolls[start + 1] < 10
626 {
627 open += 1;
628 }
629 }
630 while !info.finished {
631 s.rolls[s.count] = info.standing;
632 s.count += 1;
633 info = bowling_info(&s);
634 }
635 return vec![format!(
636 "{} {}: strikes {strikes} spares {spares} open {open} max {}",
637 self.id,
638 self.names[0],
639 info.score.iter().sum::<i32>()
640 )];
641 }
642 let mut result = Vec::new();
643 for side in 0..2 {
644 let mut line = format!("{} {}: ", self.id, self.names[side]);
645 match self.state {
646 State::Darts(state) => {
647 let s = state.stats[side];
648 let average = if s[0] > 0 { s[1] * 300 / s[0] } else { 0 };
649 line.push_str(&format!(
650 "darts {} average {}.{:02} busts {} high {} tons {} 180s {} doubles {}/{}",
651 s[0],
652 average / 100,
653 average % 100,
654 s[2],
655 s[3],
656 s[4],
657 s[5],
658 s[7],
659 s[6]
660 ));
661 }
662 State::Rally(r) => {
663 let s = r.stats[side];
664 line.push_str(&format!(
665 "points {} serve {}/{} run {}",
666 s[0], s[2], s[1], s[4]
667 ));
668 if self.config.sport == Sport::Tennis {
669 line.push_str(&format!(
670 " holds {}/{} breaks {} break points {}/{}",
671 s[6], s[5], s[7], s[9], s[8]
672 ));
673 }
674 if self.config.sport == Sport::Volleyball {
675 let lineup = r.lineup[side]
676 .iter()
677 .map(i32::to_string)
678 .collect::<Vec<_>>()
679 .join(",");
680 line.push_str(&format!(
681 " subs {} timeouts {} lineup {lineup}",
682 r.substitutions[side], r.timeouts[side]
683 ));
684 }
685 }
686 _ => unreachable!(),
687 }
688 result.push(line);
689 }
690 result
691 }
692 fn result(&self) -> Result<String, String> {
693 if !self.finished() {
694 return Err("match not finished".to_owned());
695 }
696 if let State::Bowling(s) = self.state {
697 return Ok(format!(
698 "{}: {} {}",
699 self.id,
700 self.names[0],
701 bowling_info(&s).score.iter().sum::<i32>()
702 ));
703 }
704 let first = self.winner();
705 let mut line = format!(
706 "{}: {} def. {}",
707 self.id,
708 self.names[first],
709 self.names[1 - first]
710 );
711 match self.state {
712 State::Darts(s) => {
713 for &segment in &s.segments[..s.segment_count] {
714 line.push_str(&format!(" {}", pair_text(segment, first)));
715 }
716 }
717 State::Rally(s) => {
718 for &segment in &s.segments[..s.segment_count] {
719 line.push_str(&format!(" {}", segment_text(segment, first)));
720 }
721 }
722 _ => unreachable!(),
723 }
724 Ok(line)
725 }
726}
727fn create_match(a: &[&str]) -> Result<Match, String> {
728 let sport = match a[2] {
729 "TENNIS" => Sport::Tennis,
730 "TABLETENNIS" => Sport::TableTennis,
731 "BADMINTON" => Sport::Badminton,
732 "VOLLEYBALL" => Sport::Volleyball,
733 "BOWLING" => Sport::Bowling,
734 _ => Sport::Darts,
735 };
736 let mut c = Config {
737 sport,
738 first: 0,
739 sets: if sport == Sport::Darts { 1 } else { 3 },
740 games: 5,
741 final_rule: 0,
742 start: 501,
743 legs: 1,
744 no_ad: false,
745 initial_lineup: [[0; 6]; 2],
746 };
747 let option_start = match sport {
748 Sport::Bowling => 4,
749 Sport::Volleyball => 7,
750 _ => 5,
751 };
752 let mut seen = HashSet::new();
753 for option in &a[option_start..] {
754 let (key, value) = option.split_once('=').ok_or("invalid option")?;
755 let allowed: &[&str] = match (sport, key) {
756 (Sport::Tennis, "SETS") => &["3", "5"],
757 (Sport::Tennis, "DEUCE") => &["AD", "NOAD"],
758 (Sport::Tennis, "FINAL") => &["TB7", "TB10", "ADV", "SUPER"],
759 (Sport::TableTennis, "GAMES") => &["5", "7"],
760 (Sport::Darts, "START") => &["301", "501", "701"],
761 (Sport::Darts, "SETS" | "LEGS") => &["1", "2", "3", "4", "5"],
762 (_, "FIRST") if sport != Sport::Bowling => &["A", "B"],
763 _ => return Err("invalid option".to_owned()),
764 };
765 let choice = allowed
766 .iter()
767 .position(|&s| s == value)
768 .ok_or("invalid option")?;
769 if !seen.insert(key) {
770 return Err("invalid option".to_owned());
771 }
772 match key {
773 "FIRST" => c.first = choice,
774 "SETS" => c.sets = number(value),
775 "GAMES" => c.games = number(value),
776 "START" => c.start = number(value),
777 "LEGS" => c.legs = number(value),
778 "DEUCE" => c.no_ad = value == "NOAD",
779 "FINAL" => c.final_rule = choice,
780 _ => unreachable!(),
781 }
782 }
783 let state = match sport {
784 Sport::Bowling => State::Bowling(BowlingState::default()),
785 Sport::Darts => State::Darts(DartsState {
786 remaining: [c.start; 2],
787 server: c.first,
788 turn_start: c.start,
789 leg_number: 1,
790 ..DartsState::default()
791 }),
792 _ => {
793 if sport == Sport::Volleyball {
794 for side in 0..2 {
795 for (i, s) in a[5 + side].split(',').enumerate() {
796 c.initial_lineup[side][i] = number(s);
797 }
798 }
799 }
800 State::Rally(Rally {
801 server: c.first,
802 lineup: c.initial_lineup,
803 ..Rally::default()
804 })
805 }
806 };
807 Ok(Match {
808 id: a[1].to_owned(),
809 names: [
810 a[3].to_owned(),
811 if sport == Sport::Bowling {
812 String::new()
813 } else {
814 a[4].to_owned()
815 },
816 ],
817 config: c,
818 state,
819 history: Vec::new(),
820 })
821}
822fn execute(
823 matches: &mut BTreeMap<String, Match>,
824 darts: &[Dart],
825 a: &[&str],
826) -> Result<Vec<String>, String> {
827 let command = a[0];
828 if command == "MATCH" {
829 if matches.contains_key(a[1]) {
830 return Err("duplicate match".to_owned());
831 }
832 let m = create_match(a)?;
833 matches.insert(m.id.clone(), m);
834 return Ok(Vec::new());
835 }
836 if command == "CHECKOUT" {
837 let finish = checkout(darts, number(a[1]), 3);
838 return Ok(vec![format!(
839 "checkout {}: {}",
840 a[1],
841 if finish.is_empty() {
842 "none".to_owned()
843 } else {
844 dart_names(darts, &finish)
845 }
846 )]);
847 }
848 if command == "STANDINGS" {
849 let mut totals: BTreeMap<&str, [i32; 2]> = BTreeMap::new();
850 for m in matches.values() {
851 if m.config.sport == Sport::Bowling || !m.finished() {
852 continue;
853 }
854 let first = m.winner();
855 for side in 0..2 {
856 totals.entry(&m.names[side]).or_default()[usize::from(side != first)] += 1;
857 }
858 }
859 let mut rows: Vec<(&str, [i32; 2])> = totals.into_iter().collect();
860 rows.sort_by(|a, b| {
861 b.1[0]
862 .cmp(&a.1[0])
863 .then(a.1[1].cmp(&b.1[1]))
864 .then(a.0.cmp(b.0))
865 });
866 let display = if rows.is_empty() {
867 "none".to_owned()
868 } else {
869 rows.iter()
870 .map(|(name, p)| format!("{name} {}", pair_text(*p, 0)))
871 .collect::<Vec<_>>()
872 .join(", ")
873 };
874 return Ok(vec![format!("standings: {display}")]);
875 }
876 let m = matches.get_mut(a[1]).ok_or("unknown match")?;
877 match command {
878 "SCORE" => return Ok(vec![m.score(darts)]),
879 "STATS" => return Ok(m.statistics()),
880 "RESULT" => return Ok(vec![m.result()?]),
881 "UNDO" => {
882 m.state = m.history.pop().ok_or("nothing to undo")?;
883 return Ok(Vec::new());
884 }
885 _ => (),
886 }
887 let c = m.config;
888 let allowed = match command {
889 "POINT" => ![Sport::Bowling, Sport::Darts].contains(&c.sport),
890 "ROLL" => c.sport == Sport::Bowling,
891 "THROW" => c.sport == Sport::Darts,
892 _ => c.sport == Sport::Volleyball,
893 };
894 if !allowed {
895 return Err("wrong event".to_owned());
896 }
897 let count = match command {
898 "POINT" => a[2].len(),
899 "ROLL" | "THROW" => a.len() - 2,
900 _ => 1,
901 };
902 for i in 0..count {
903 if m.finished() {
904 return Err("match over".to_owned());
905 }
906 let (mut value, mut position, mut side) = (0usize, None, 0usize);
907 match command {
908 "POINT" => value = usize::from(a[2].as_bytes()[i] == b'B'),
909 "ROLL" => {
910 value = number(a[i + 2]) as usize;
911 if let State::Bowling(s) = m.state {
912 if value as i32 > bowling_info(&s).standing {
913 return Err("too many pins".to_owned());
914 }
915 }
916 }
917 "THROW" => {
918 value = darts
919 .iter()
920 .position(|d| d.name == a[i + 2])
921 .ok_or("invalid dart")?
922 }
923 _ => {
924 side = usize::from(a[2] == "B");
925 if let State::Rally(s) = m.state {
926 if command == "TIMEOUT" {
927 if s.timeouts[side] >= 2 {
928 return Err("no timeouts left".to_owned());
929 }
930 } else {
931 let leaving = number(a[3]);
932 value = number(a[4]) as usize;
933 position = Some(
934 s.lineup[side]
935 .iter()
936 .position(|&n| n == leaving)
937 .ok_or("player not on court")?,
938 );
939 if s.lineup[side].contains(&(value as i32)) {
940 return Err("player already on court".to_owned());
941 }
942 if s.substitutions[side] >= 6 {
943 return Err("no substitutions left".to_owned());
944 }
945 }
946 }
947 }
948 }
949 m.history.push(m.state);
950 match &mut m.state {
951 State::Rally(s) => match command {
952 "POINT" => add_rally(&c, s, value),
953 "SUB" => {
954 s.lineup[side][position.unwrap()] = value as i32;
955 s.substitutions[side] += 1;
956 }
957 _ => s.timeouts[side] += 1,
958 },
959 State::Bowling(s) => {
960 s.rolls[s.count] = value as i32;
961 s.count += 1;
962 }
963 State::Darts(s) => add_dart(&c, darts, s, value),
964 }
965 }
966 Ok(Vec::new())
967}
968fn main() {
969 let darts = initialize_darts();
970 let mut matches = BTreeMap::new();
971 let stdin = io::stdin();
972 let mut lines = stdin.lock().lines();
973 let count: usize = lines.next().unwrap().unwrap().parse().unwrap();
974 let stdout = io::stdout();
975 let mut output = io::BufWriter::new(stdout.lock());
976 for line in lines.take(count) {
977 let line = line.unwrap();
978 let a: Vec<&str> = line.split(' ').collect();
979 let result = execute(&mut matches, &darts, &a)
980 .unwrap_or_else(|message| vec![format!("ERROR {message}")]);
981 for line in result {
982 writeln!(output, "{line}").unwrap();
983 }
984 }
985}
