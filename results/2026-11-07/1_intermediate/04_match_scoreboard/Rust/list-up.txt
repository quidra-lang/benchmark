#[derive(Clone, Copy, Default)]
struct Segment {
    a: i32,
    b: i32,
    kind: usize,
    loser: i32,
}
#[derive(Clone, Copy, Default)]
struct Rally {
    points: [i32; 2],
    games: [i32; 2],
    sets: [i32; 2],
    substitutions: [i32; 2],
    timeouts: [i32; 2],
    stats: [[i32; 10]; 2],
    lineup: [[i32; 6]; 2],
    segments: [Segment; 7],
    segment_count: usize,
    server: usize,
    winner: Option<usize>,
}
#[derive(Clone, Copy, Default)]
struct BowlingState {
    rolls: [i32; 21],
    count: usize,
}
#[derive(Clone, Copy, Default)]
struct DartsState {
    remaining: [i32; 2],
    legs: [i32; 2],
    sets: [i32; 2],
    stats: [[i32; 8]; 2],
    segments: [[i32; 2]; 9],
    turn: [usize; 3],
    segment_count: usize,
    server: usize,
    turn_start: i32,
    turn_count: usize,
    turn_score: i32,
    leg_number: usize,
    winner: Option<usize>,
}
#[derive(Clone, Copy)]
enum State {
    Rally(Rally),
    Bowling(BowlingState),
    Darts(DartsState),
}

    history: Vec<State>,

        "POINT" => ![Sport::Bowling, Sport::Darts].contains(&c.sport),
        "ROLL" => c.sport == Sport::Bowling,
        "THROW" => c.sport == Sport::Darts,
        _ => c.sport == Sport::Volleyball,
    };
    if !allowed {
        return Err("wrong event".to_owned());
    }
    let count = match command {
        "POINT" => a[2].len(),
        "ROLL" | "THROW" => a.len() - 2,
        _ => 1,
    };
    for i in 0..count {
        if m.finished() {
            return Err("match over".to_owned());
        }
        let (mut value, mut position, mut side) = (0usize, None, 0usize);
        match command {
            "POINT" => value = usize::from(a[2].as_bytes()[i] == b'B'),
            "ROLL" => {
                value = number(a[i + 2]) as usize;
                if let State::Bowling(s) = m.state {
                    if value as i32 > bowling_info(&s).standing {
                        return Err("too many pins".to_owned());
                    }
                }
            }
