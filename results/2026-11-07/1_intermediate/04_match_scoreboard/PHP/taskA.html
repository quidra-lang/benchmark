41function checkout_tables(): array
42{
43 $candidates = [];
44 $doubles = [];
45 $darts = ['25' => [25, false], 'BULL' => [50, true], 'MISS' => [0, false]];
46 foreach (['S' => 1, 'T' => 3, 'D' => 2] as $prefix => $multiplier) {
47 for ($value = 20; $value >= 1; --$value) {
48 $code = $prefix . $value;
49 $candidates[] = [$code, $value * $multiplier];
50 $darts[$code] = [$value * $multiplier, $prefix === 'D'];
51 if ($prefix === 'D') {
52 $doubles[] = [$code, $value * $multiplier];
53 }
54 }
55 }
56 $candidates[] = ['25', 25];
57 $candidates[] = ['BULL', 50];
58 $doubles[] = ['BULL', 50];
59 $tables = [1 => [], 2 => [], 3 => []];
60 foreach ($doubles as [$last, $value]) {
61 $tables[1][$value] ??= [$last];
62 }
63 foreach ($candidates as [$first, $a]) {
64 foreach ($doubles as [$last, $b]) {
65 $tables[2][$a + $b] ??= [$first, $last];
66 }
67 }
68 foreach ($candidates as [$first, $a]) {
69 foreach ($candidates as [$second, $b]) {
70 foreach ($doubles as [$last, $c]) {
71 $tables[3][$a + $b + $c] ??= [$first, $second, $last];
72 }
73 }
74 }
75 return [$darts, $tables];
76}
77[$dartData, $checkoutData] = checkout_tables();

90function create_match(array $tokens): array
91{
92 [$id, $sport] = array_slice($tokens, 1, 2);
93 $names = [1 => $tokens[3]];
94 if ($sport !== 'BOWLING') {
95 $names[2] = $tokens[4];
96 }
97 $initial = [];
98 if ($sport === 'VOLLEYBALL') {
99 $initial = [1 => array_map('intval', explode(',', $tokens[5])), 2 => array_map('intval', explode(',', $tokens[6]))];
100 $firstOption = 7;
101 } elseif ($sport === 'BOWLING') {
102 $firstOption = 4;
103 } else {
104 $firstOption = 5;
105 }
106 $options = array_map(fn($values) => $values[0], OPTION_RULES[$sport]);
107 $seen = [];
108 foreach (array_slice($tokens, $firstOption) as $token) {
109 $parts = explode('=', $token);
110 if (count($parts) !== 2) {
111 score_error('invalid option');
112 }
113 [$key, $value] = $parts;
114 if (!isset(OPTION_RULES[$sport][$key]) || !in_array($value, OPTION_RULES[$sport][$key], true) || isset($seen[$key])) {
115 score_error('invalid option');
116 }
117 $seen[$key] = true;
118 $options[$key] = $value;
119 }
120 $first = isset($options['FIRST']) ? side($options['FIRST']) : 1;
121 $stats = [];
122 foreach (['points', 'served', 'servewins', 'runmax', 'holds', 'games', 'breaks', 'opportunities', 'converted',
123 'darts', 'scored', 'busts', 'high', 'tons', 'maximums', 'attempts', 'hits'] as $key) {
124 $stats[$key] = pair();
125 }
126 $state = ['p' => pair(), 'g' => pair(), 'won' => pair(), 'server' => $first, 'winner' => 0,
127 'completed' => [], 'stats' => $stats, 'last' => 0, 'run' => 0];
128 if ($sport === 'VOLLEYBALL') {
129 $state['lineups'] = $initial;
130 $state['subs'] = pair();
131 $state['timeouts'] = pair();
132 } elseif ($sport === 'BOWLING') {
133 $state['frames'] = array_fill(1, 10, []);
134 $state['marks'] = array_fill(1, 10, []);
135 $state['frame'] = 1;
136 $state['standing'] = 10;
137 $state['fresh'] = true;
138 } elseif ($sport === 'DARTS') {
139 $start = (int) $options['START'];
140 $state['legs'] = pair();
141 $state['remaining'] = pair($start, $start);
142 $state['turn'] = [];
143 $state['turnscore'] = 0;
144 $state['turnstart'] = $start;
145 $state['legnumber'] = 0;
146 }
147 return ['id' => $id, 'sport' => $sport, 'names' => $names, 'options' => $options, 'initial' => $initial, 'state' => $state, 'history' => []];
148}

150function tennis_mode(array $m): string
151{
152 $s = $m['state'];
153 $final = $s['won'][1] === intdiv((int) $m['options']['SETS'], 2) && $s['won'][1] === $s['won'][2];
154 if ($final && $m['options']['FINAL'] === 'SUPER') {
155 return 'SUPER';
156 }
157 if ($s['g'] === pair(6, 6) && !($final && $m['options']['FINAL'] === 'ADV')) {
158 return 'TB';
159 }
160 return 'NORMAL';

163function rally_server(array $m): int
164{
165 $s = $m['state'];
166 if ($m['sport'] === 'TENNIS') {
167 if (tennis_mode($m) !== 'NORMAL' && intdiv(array_sum($s['p']) + 1, 2) % 2 === 1) {
168 return other($s['server']);
169 }
170 } elseif ($m['sport'] === 'TABLETENNIS') {
171 $first = side($m['options']['FIRST']);
172 if (count($s['completed']) % 2 === 1) {
173 $first = other($first);
174 }
175 $total = array_sum($s['p']);
176 $flip = $total < 20 ? intdiv($total, 2) % 2 === 1 : $total % 2 === 1;
177 return $flip ? other($first) : $first;
178 }
179 return $s['server'];
180}

182function rally_outcome(array $m, int $winner): array
183{
184 $s = $m['state'];
185 $points = $s['p'];
186 ++$points[$winner];
187 $loser = other($winner);
188 $game = false;
189 $unit = false;
190 if ($m['sport'] === 'TENNIS') {
191 $mode = tennis_mode($m);
192 if ($mode === 'NORMAL') {
193 $game = $points[$winner] >= 4 && ($m['options']['DEUCE'] === 'NOAD' || $points[$winner] - $points[$loser] >= 2);
194 $games = $s['g'][$winner] + 1;
195 $unit = $game && $games >= 6 && $games - $s['g'][$loser] >= 2;
196 } else {
197 $final = $s['won'][1] === $s['won'][2] && $s['won'][1] === intdiv((int) $m['options']['SETS'], 2);
198 $target = $mode === 'SUPER' || ($final && $m['options']['FINAL'] === 'TB10') ? 10 : 7;
199 $unit = $points[$winner] >= $target && $points[$winner] - $points[$loser] >= 2;
200 }
201 $needed = intdiv((int) $m['options']['SETS'], 2) + 1;
202 } elseif ($m['sport'] === 'TABLETENNIS') {
203 $unit = $points[$winner] >= 11 && $points[$winner] - $points[$loser] >= 2;
204 $needed = intdiv((int) $m['options']['GAMES'], 2) + 1;
205 } elseif ($m['sport'] === 'BADMINTON') {
206 $unit = $points[$winner] === 30 || ($points[$winner] >= 21 && $points[$winner] - $points[$loser] >= 2);
207 $needed = 2;
208 } else {
209 $target = count($s['completed']) === 4 ? 15 : 25;
210 $unit = $points[$winner] >= $target && $points[$winner] - $points[$loser] >= 2;
211 $needed = 3;
212 }
213 return [$game, $unit, $unit && $s['won'][$winner] + 1 === $needed];
214}

216function point_event(array &$m, int $winner): void
217{
218 $server = rally_server($m);
219 [$game, $unit, $finished] = rally_outcome($m, $winner);
220 $mode = $m['sport'] === 'TENNIS' ? tennis_mode($m) : '';
221 $opportunity = $mode === 'NORMAL' && rally_outcome($m, other($server))[0];
222 $s = $m['state'];
223 $record = [1, $winner, $server, (int) $opportunity, (int) $game, $s['p'][1], $s['p'][2], $s['g'][1], $s['g'][2],
224 $s['won'][1], $s['won'][2], $s['server'], count($s['completed']), $s['last'], $s['run'], $s['stats']['runmax'][$winner], 0, 0];
225 ++$s['stats']['points'][$winner];
226 ++$s['stats']['served'][$server];
227 if ($winner === $server) {
228 ++$s['stats']['servewins'][$server];
229 }
230 $s['run'] = $s['last'] === $winner ? $s['run'] + 1 : 1;
231 $s['last'] = $winner;
232 $s['stats']['runmax'][$winner] = max($s['stats']['runmax'][$winner], $s['run']);
233 if ($opportunity) {
234 ++$s['stats']['opportunities'][other($server)];
235 if ($winner !== $server) {
236 ++$s['stats']['converted'][$winner];
237 }
238 }
239 ++$s['p'][$winner];
240 if ($m['sport'] === 'TENNIS') {
241 if ($game) {
242 ++$s['stats']['games'][$server];
243 if ($winner === $server) {
244 ++$s['stats']['holds'][$winner];
245 } else {
246 ++$s['stats']['breaks'][$winner];
247 }
248 ++$s['g'][$winner];
249 $s['server'] = other($s['server']);
250 } elseif ($unit) {
251 $s['server'] = other($s['server']);
252 }
253 if ($unit) {
254 if ($mode === 'SUPER') {
255 $s['completed'][] = [$s['p'][1], $s['p'][2], 2, 0];
256 } elseif ($mode === 'TB') {
257 ++$s['g'][$winner];
258 $s['completed'][] = [$s['g'][1], $s['g'][2], 1, $s['p'][other($winner)]];
259 } else {
260 $s['completed'][] = [$s['g'][1], $s['g'][2], 0, 0];
261 }
262 }
263 } elseif ($m['sport'] === 'BADMINTON') {
264 $s['server'] = $winner;
265 } elseif ($m['sport'] === 'VOLLEYBALL' && $winner !== $server) {
266 $s['server'] = $winner;
267 $lineup = $s['lineups'][$winner];
268 $s['lineups'][$winner] = [...array_slice($lineup, 1), $lineup[0]];
269 $record[17] = 1;
270 }
271 if ($unit) {
272 if ($m['sport'] !== 'TENNIS') {
273 $s['completed'][] = [$s['p'][1], $s['p'][2], 0, 0];
274 }
275 ++$s['won'][$winner];
276 if ($finished) {
277 $s['winner'] = $winner;
278 } else {
279 if ($m['sport'] === 'VOLLEYBALL') {
280 $record[16] = 1;
281 $record = [...$record, ...$s['lineups'][1], ...$s['lineups'][2], ...array_values($s['subs']), ...array_values($s['timeouts'])];
282 $s['lineups'] = $m['initial'];
283 $s['subs'] = pair();
284 $s['timeouts'] = pair();
285 $first = side($m['options']['FIRST']);
286 $s['server'] = count($s['completed']) % 2 === 0 ? $first : other($first);
287 }
288 $s['p'] = pair();
289 $s['g'] = pair();
290 }
291 } elseif ($game) {
292 $s['p'] = pair();
293 }
294 $m['state'] = $s;
295 $m['history'][] = pack('l*', ...$record);
296}

298function roll_event(array &$s, int $pins): void
299{
300 if ($pins > $s['standing']) {
301 score_error('too many pins');
302 }
303 $frame = $s['frame'];
304 if ($s['fresh'] && $pins === 10) {
305 $mark = 'X';
306 } elseif ($pins === $s['standing']) {
307 $mark = '/';
308 } else {
309 $mark = $pins === 0 ? '-' : (string) $pins;
310 }
311 $s['frames'][$frame][] = $pins;
312 $s['marks'][$frame][] = $mark;
313 $s['standing'] -= $pins;
314 $s['fresh'] = false;
315 $rolls = $s['frames'][$frame];
316 if ($frame < 10) {
317 if ((count($rolls) === 1 && $pins === 10) || count($rolls) === 2) {
318 ++$s['frame'];
319 $s['standing'] = 10;
320 $s['fresh'] = true;
321 }
322 } else {
323 $complete = count($rolls) === 3 || (count($rolls) === 2 && $rolls[0] !== 10 && array_sum($rolls) < 10);
324 if ($complete) {
325 $s['winner'] = 1;
326 } elseif ($s['standing'] === 0) {
327 $s['standing'] = 10;
328 $s['fresh'] = true;
329 }
330 }
331}

333function bowling_scores(array $s): array
334{
335 $scores = array_fill(1, 10, null);
336 for ($frame = 1; $frame <= 10; ++$frame) {
337 $rolls = $s['frames'][$frame];
338 if ($rolls === []) {
339 continue;
340 }
341 $later = array_merge(...array_slice($s['frames'], $frame));
342 if ($frame === 10) {
343 if ($s['winner'] !== 0) {
344 $scores[$frame] = array_sum($rolls);
345 }
346 } elseif ($rolls[0] === 10) {
347 if (count($later) >= 2) {
348 $scores[$frame] = 10 + $later[0] + $later[1];
349 }
350 } elseif (count($rolls) === 2) {
351 if (array_sum($rolls) === 10) {
352 if ($later !== []) {
353 $scores[$frame] = 10 + $later[0];
354 }
355 } else {
356 $scores[$frame] = array_sum($rolls);
357 }
358 }
359 }
360 return $scores;
361}

383function dart_event(array &$m, string $code): void
384{
385 global $dartData;
386 if (!isset($dartData[$code])) {
387 score_error('invalid dart');
388 }
389 $s = $m['state'];
390 $player = $s['server'];
391 [$value, $double] = $dartData[$code];
392 $before = $s['remaining'][$player];
393 ++$s['stats']['darts'][$player];
394 if ($before === 50 || ($before <= 40 && $before % 2 === 0)) {
395 ++$s['stats']['attempts'][$player];
396 }
397 $remaining = $before - $value;
398 if ($remaining < 0 || $remaining === 1 || ($remaining === 0 && !$double)) {
399 ++$s['stats']['busts'][$player];
400 $s['stats']['scored'][$player] -= $s['turnscore'];
401 $s['remaining'][$player] = $s['turnstart'];
402 next_turn($s);
403 $m['state'] = $s;
404 return;
405 }
406 $s['stats']['scored'][$player] += $value;
407 $s['remaining'][$player] = $remaining;
408 $s['turnscore'] += $value;
409 $s['turn'][] = $code;
410 if ($remaining === 0) {
411 ++$s['stats']['hits'][$player];
412 complete_turn($s, $player);
413 ++$s['legs'][$player];
414 $setOver = $s['legs'][$player] === (int) $m['options']['LEGS'];
415 if ($setOver) {
416 $s['completed'][] = [$s['legs'][1], $s['legs'][2], 0, 0];
417 ++$s['won'][$player];
418 }
419 if ($setOver && $s['won'][$player] === (int) $m['options']['SETS']) {
420 $s['winner'] = $player;
421 $m['state'] = $s;
422 return;
423 }
424 if ($setOver) {
425 $s['legs'] = pair();
426 }
427 ++$s['legnumber'];
428 $first = side($m['options']['FIRST']);
429 $s['server'] = $s['legnumber'] % 2 === 0 ? $first : other($first);
430 $start = (int) $m['options']['START'];
431 $s['remaining'] = pair($start, $start);
432 $s['turn'] = [];
433 $s['turnscore'] = 0;
434 $s['turnstart'] = $start;
435 } elseif (count($s['turn']) === 3) {
436 complete_turn($s, $player);
437 next_turn($s);
438 }
439 $m['state'] = $s;
440}

442function undo_event(array &$m): void
443{
444 if ($m['history'] === []) {
445 score_error('nothing to undo');
446 }
447 $record = array_pop($m['history']);
448 if (is_array($record)) {
449 $m['state'] = $record;
450 return;
451 }
452 $data = array_values(unpack('l*', $record));
453 $s = $m['state'];
454 if ($data[0] === 2) {
455 $s['lineups'][$data[1]][$data[2]] = $data[3];
456 --$s['subs'][$data[1]];
457 } elseif ($data[0] === 3) {
458 --$s['timeouts'][$data[1]];
459 } else {
460 [$winner, $server] = array_slice($data, 1, 2);
461 --$s['stats']['points'][$winner];
462 --$s['stats']['served'][$server];
463 if ($winner === $server) {
464 --$s['stats']['servewins'][$server];
465 }
466 if ($data[3] === 1) {
467 --$s['stats']['opportunities'][other($server)];
468 if ($winner !== $server) {
469 --$s['stats']['converted'][$winner];
470 }
471 }
472 if ($data[4] === 1) {
473 --$s['stats']['games'][$server];
474 if ($winner === $server) {
475 --$s['stats']['holds'][$winner];
476 } else {
477 --$s['stats']['breaks'][$winner];
478 }
479 }
480 if ($data[16] === 1) {
481 $s['lineups'] = [1 => array_slice($data, 18, 6), 2 => array_slice($data, 24, 6)];
482 $s['subs'] = pair($data[30], $data[31]);
483 $s['timeouts'] = pair($data[32], $data[33]);
484 }
485 if ($data[17] === 1) {
486 $lineup = $s['lineups'][$winner];
487 $s['lineups'][$winner] = [$lineup[5], ...array_slice($lineup, 0, 5)];
488 }
489 $s['p'] = pair($data[5], $data[6]);
490 $s['g'] = pair($data[7], $data[8]);
491 $s['won'] = pair($data[9], $data[10]);
492 $s['server'] = $data[11];
493 $s['winner'] = 0;
494 $s['completed'] = array_slice($s['completed'], 0, $data[12]);
495 $s['last'] = $data[13];
496 $s['run'] = $data[14];
497 $s['stats']['runmax'][$winner] = $data[15];
498 }
499 $m['state'] = $s;
500}

502function event(array &$m, string $command, array $args): void
503{
504 $correct = ($command === 'POINT' && in_array($m['sport'], RALLY_SPORTS, true)) ||
505 ($command === 'ROLL' && $m['sport'] === 'BOWLING') || ($command === 'THROW' && $m['sport'] === 'DARTS') ||
506 (in_array($command, ['SUB', 'TIMEOUT'], true) && $m['sport'] === 'VOLLEYBALL');
507 if (!$correct) {
508 score_error('wrong event');
509 }
510 if ($command === 'POINT') {
511 $events = str_split($args[0]);
512 } elseif ($command === 'SUB' || $command === 'TIMEOUT') {
513 $events = [''];
514 } else {
515 $events = $args;
516 }
517 foreach ($events as $value) {
518 if ($m['state']['winner'] !== 0) {
519 score_error('match over');
520 }
521 if ($command === 'POINT') {
522 point_event($m, side($value));
523 } elseif ($command === 'ROLL') {
524 $old = $m['state'];
525 $next = $old;
526 roll_event($next, (int) $value);
527 $m['state'] = $next;
528 $m['history'][] = $old;
529 } elseif ($command === 'THROW') {
530 $old = $m['state'];
531 dart_event($m, $value);
532 $m['history'][] = $old;
533 } elseif ($command === 'SUB') {
534 $player = side($args[0]);
535 [$outgoing, $incoming] = array_map('intval', array_slice($args, 1, 2));
536 $lineup = $m['state']['lineups'][$player];
537 $position = array_search($outgoing, $lineup, true);
538 if ($position === false) {
539 score_error('player not on court');
540 }
541 if (in_array($incoming, $lineup, true)) {
542 score_error('player already on court');
543 }
544 if ($m['state']['subs'][$player] >= 6) {
545 score_error('no substitutions left');
546 }
547 $m['history'][] = pack('l*', 2, $player, $position, $outgoing);
548 $m['state']['lineups'][$player][$position] = $incoming;
549 ++$m['state']['subs'][$player];
550 } else {
551 $player = side($args[0]);
552 if ($m['state']['timeouts'][$player] >= 2) {
553 score_error('no timeouts left');
554 }
555 $m['history'][] = pack('l*', 3, $player);
556 ++$m['state']['timeouts'][$player];
557 }
558 }
559}

561function completed_text(array $record, int $winner = 1): string
562{
563 [$a, $b, $kind, $extra] = $record;
564 if ($winner === 2) {
565 [$a, $b] = [$b, $a];
566 }
567 $text = "$a-$b";
568 if ($kind === 2) {
569 return "[$text]";
570 }
571 return $kind === 1 ? "$text($extra)" : $text;

574function situation(array $m, int $player): string
575{
576 [$game, $unit, $finished] = rally_outcome($m, $player);
577 if ($finished) {
578 return 'match point';
579 }
580 if ($unit) {
581 return in_array($m['sport'], ['TENNIS', 'VOLLEYBALL'], true) ? 'set point' : 'game point';
582 }
583 if ($game) {
584 return $player === rally_server($m) ? 'game point' : 'break point';
585 }
586 return '';
587}

589function score_line(array $m): string
590{
591 $s = $m['state'];
592 if ($m['sport'] === 'BOWLING') {
593 $parts = ["$m[id]: " . $m['names'][1]];
594 $scores = bowling_scores($s);
595 $cumulative = 0;
596 for ($frame = 1; $frame <= 10; ++$frame) {
597 if ($s['frames'][$frame] === []) {
598 break;
599 }
600 $text = implode('', $s['marks'][$frame]);
601 if ($scores[$frame] !== null) {
602 $cumulative += $scores[$frame];
603 $text .= "($cumulative)";
604 }
605 $parts[] = $text;
606 }
607 return implode(' ', $parts);
608 }
609 $names = $m['names'];
610 if ($s['winner'] === 0) {
611 $player = in_array($m['sport'], RALLY_SPORTS, true) ? rally_server($m) : $s['server'];
612 $names[$player] .= '*';
613 }
614 $parts = ["$m[id]: $names[1] vs $names[2]"];
615 if ($m['sport'] === 'DARTS') {
616 $parts = [...$parts, 'sets', pair_text($s['won']), 'legs', pair_text($s['legs']), 'score', pair_text($s['remaining'])];
617 if ($s['winner'] === 0) {
618 if ($s['turn'] !== []) {
619 $parts = [...$parts, 'turn', ...$s['turn']];
620 }
621 $answer = checkout($s['remaining'][$s['server']], 3 - count($s['turn']));
622 if ($answer !== []) {
623 $parts = [...$parts, 'checkout', ...$answer];
624 }
625 }
626 } else {
627 $parts = [...$parts, ...array_map('completed_text', $s['completed'])];
628 if ($s['winner'] === 0) {
629 if ($m['sport'] === 'TENNIS') {
630 $mode = tennis_mode($m);
631 if ($mode === 'SUPER') {
632 $parts[] = '[' . pair_text($s['p']) . ']';
633 } else {
634 $parts[] = pair_text($s['g']);
635 if ($mode === 'TB') {
636 $parts[] = pair_text($s['p']);
637 } elseif ($m['options']['DEUCE'] === 'AD' && min($s['p']) >= 3) {
638 if ($s['p'][1] === $s['p'][2]) {
639 $parts[] = '40-40';
640 } elseif ($s['p'][1] > $s['p'][2]) {
641 $parts[] = 'AD-40';
642 } else {
643 $parts[] = '40-AD';
644 }
645 } else {
646 $labels = ['0', '15', '30', '40'];
647 $parts[] = $labels[$s['p'][1]] . '-' . $labels[$s['p'][2]];
648 }
649 }
650 } else {
651 $parts[] = pair_text($s['p']);
652 if ($m['sport'] === 'VOLLEYBALL') {
653 $parts = [...$parts, 'server', (string) $s['lineups'][$s['server']][0]];
654 }
655 }
656 for ($player = 1; $player <= 2; ++$player) {
657 $label = situation($m, $player);
658 if ($label !== '') {
659 $parts[] = "($label " . $m['names'][$player] . ')';
660 }
661 }
662 }
663 }
664 return implode(' ', $parts);
665}

667function stats_lines(array $m): array
668{
669 $s = $m['state'];
670 $stats = $s['stats'];
671 if ($m['sport'] === 'BOWLING') {
672 $marks = array_merge(...$s['marks']);
673 $open = count(array_filter($s['frames'], fn($rolls) =>
674 count($rolls) >= 2 && $rolls[0] !== 10 && $rolls[0] + $rolls[1] < 10));
675 $future = $s;
676 while ($future['winner'] === 0) {
677 roll_event($future, $future['standing']);
678 }
679 $maximum = array_sum(bowling_scores($future));
680 return ["$m[id] " . $m['names'][1] . ': strikes ' . count(array_filter($marks, fn($mark) => $mark === 'X')) . ' spares ' . count(array_filter($marks, fn($mark) => $mark === '/')) . " open $open max $maximum"];
681 }
682 $output = [];
683 for ($player = 1; $player <= 2; ++$player) {
684 if ($m['sport'] === 'DARTS') {
685 $darts = $stats['darts'][$player];
686 $average = $darts === 0 ? 0 : intdiv($stats['scored'][$player] * 300, $darts);
687 $text = "darts $darts average " . average_text($average) . ' busts ' . $stats['busts'][$player] . ' high ' . $stats['high'][$player] . ' tons ' . $stats['tons'][$player] . ' 180s ' . $stats['maximums'][$player] . ' doubles ' . $stats['hits'][$player] . '/' . $stats['attempts'][$player];
688 } else {
689 $text = 'points ' . $stats['points'][$player] . ' serve ' . $stats['servewins'][$player] . '/' . $stats['served'][$player] . ' run ' . $stats['runmax'][$player];
690 if ($m['sport'] === 'TENNIS') {
691 $text .= ' holds ' . $stats['holds'][$player] . '/' . $stats['games'][$player] . ' breaks ' . $stats['breaks'][$player] . ' break points ' . $stats['converted'][$player] . '/' . $stats['opportunities'][$player];
692 } elseif ($m['sport'] === 'VOLLEYBALL') {
693 $text .= ' subs ' . $s['subs'][$player] . ' timeouts ' . $s['timeouts'][$player] . ' lineup ' . implode(',', $s['lineups'][$player]);
694 }
695 }
696 $output[] = "$m[id] " . $m['names'][$player] . ': ' . $text;
697 }
698 return $output;
699}

701function result_line(array $m): string
702{
703 $s = $m['state'];
704 if ($s['winner'] === 0) {
705 score_error('match not finished');
706 }
707 if ($m['sport'] === 'BOWLING') {
708 return "$m[id]: " . $m['names'][1] . ' ' . array_sum(bowling_scores($s));
709 }
710 $winner = $s['winner'];
711 return "$m[id]: " . $m['names'][$winner] . ' def. ' . $m['names'][other($winner)] . ' ' . implode(' ', array_map(fn($record) => completed_text($record, $winner), $s['completed']));
712}

714function execute(string $line, array &$matches): array
715{
716 $tokens = explode(' ', $line);
717 $command = $tokens[0];
718 if ($command === 'MATCH') {
719 if (isset($matches[$tokens[1]])) {
720 score_error('duplicate match');
721 }
722 $m = create_match($tokens);
723 $matches[$m['id']] = $m;
724 return [];
725 }
726 if ($command === 'CHECKOUT') {
727 $answer = checkout((int) $tokens[1]);
728 return ["checkout $tokens[1]: " . ($answer === [] ? 'none' : implode(' ', $answer))];
729 }
730 if ($command === 'STANDINGS') {
731 $totals = [];
732 foreach ($matches as $m) {
733 if ($m['sport'] !== 'BOWLING' && $m['state']['winner'] !== 0) {
734 for ($player = 1; $player <= 2; ++$player) {
735 $name = $m['names'][$player];
736 $totals[$name] ??= [0, 0];
737 ++$totals[$name][$player === $m['state']['winner'] ? 0 : 1];
738 }
739 }
740 }
741 $names = array_keys($totals);
742 usort($names, fn($a, $b) => [-$totals[$a][0], $totals[$a][1], $a] <=> [-$totals[$b][0], $totals[$b][1], $b]);
743 return ['standings: ' . ($names === [] ? 'none' : implode(', ', array_map(fn($name) => "$name " . $totals[$name][0] . '-' . $totals[$name][1], $names)))];
744 }
745 if (!isset($matches[$tokens[1]])) {
746 score_error('unknown match');
747 }
748 $id = $tokens[1];
749 if (in_array($command, ['POINT', 'ROLL', 'THROW', 'SUB', 'TIMEOUT'], true)) {
750 event($matches[$id], $command, array_slice($tokens, 2));
751 } elseif ($command === 'UNDO') {
752 undo_event($matches[$id]);
753 } elseif ($command === 'SCORE') {
754 return [score_line($matches[$id])];
755 } elseif ($command === 'STATS') {
756 return stats_lines($matches[$id]);
757 } elseif ($command === 'RESULT') {
758 return [result_line($matches[$id])];
759 }
760 return [];
761}
762
763$matches = [];
764$count = (int) fgets(STDIN);
765for ($i = 0; $i < $count; ++$i) {
766 $line = rtrim(fgets(STDIN), "\r\n");
767 try {
768 foreach (execute($line, $matches) as $output) {
769 echo $output, "\n";
770 }
771 } catch (InvalidArgumentException $exception) {
772 echo 'ERROR ', $exception->getMessage(), "\n";
773 }
774}
