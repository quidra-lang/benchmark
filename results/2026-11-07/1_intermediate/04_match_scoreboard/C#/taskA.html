24 Program() {
25 foreach (string prefix in new List<string> { "S","T","D" }) {
26 for (int n = 20; n >= 1; n--) {
27 Dart d = new Dart(prefix+n,n*(prefix.Equals("S") ? 1 : prefix.Equals("T") ? 3 : 2),prefix.Equals("D"));
28 darts.Add(d);

34 darts.Add(bull);

39 List<string> Checkout(int score, int maximum) {
40 int key = score*4+maximum;
41 if (checkoutCache.ContainsKey(key)) return checkoutCache[key];
42 List<string> answer = new List<string> { };
43 {
44 foreach (Dart c in doubles) {
45 if (c.Value == score) {
46 answer = new List<string> { c.Name };
47 goto Found;
48 }
49 }
50 if (maximum >= 2) {
51 for (int i = 0; i < 62; i++) {
52 Dart a = darts[i];
53 foreach (Dart b in doubles) {
54 if (a.Value+b.Value == score) {
55 answer = new List<string> { a.Name,b.Name };
56 goto Found;
57 }
58 }
59 }
60 }
61 if (maximum >= 3) {
62 for (int i = 0; i < 62; i++) {
63 Dart a = darts[i];
64 for (int j = 0; j < 62; j++) {
65 Dart b = darts[j];
66 foreach (Dart c in doubles) {
67 if (a.Value+b.Value+c.Value == score) {
68 answer = new List<string> { a.Name,b.Name,c.Name };
69 goto Found;
70 }
71 }
72 }
73 }
74 }
75 }
76 Found:
77 checkoutCache[key] = answer;
78 return answer;

102 allowed["FINAL"] = new HashSet<string> { "TB7","TB10","ADV","SUPER" };

104 options["DEUCE"] = "AD";

111 if (sport.Equals("DARTS")) {

113 allowed["SETS"] = new HashSet<string> { "1","2","3","4","5" };

115 options["START"] = "501";

117 options["LEGS"] = "1";

145 public int First() { return options.GetValueOrDefault("FIRST","A").Equals("A") ? 0 : 1; }
146 public int Option(string name) { return int.Parse(options[name]); }
147 public bool Over() { return s[DONE] != 0; }
148 public bool Tennis() { return sport.Equals("TENNIS"); }
149 public bool Rally() { return !sport.Equals("DARTS") && !sport.Equals("BOWLING"); }
150 public bool FinalSet() { return Tennis() && s[W] == Option("SETS")/2 && s[W+1] == Option("SETS")/2; }
151 public bool SuperTie() { return FinalSet() && options["FINAL"].Equals("SUPER"); }
152 public bool TieBreak() { return SuperTie() || Tennis() && s[G] == 6 && s[G+1] == 6 && !(FinalSet() && options["FINAL"].Equals("ADV")); }
153 public int Server() {
154 if (Tennis()) {
155 int point = s[P]+s[P+1]+1;
156 return s[GS] ^ (TieBreak() ? point/2%2 : 0);
157 }
158 if (sport.Equals("TABLETENNIS")) {
159 int played = s[P]+s[P+1];
160 int alternate = played < 20 ? played/2%2 : played%2;
161 return First() ^ (finished.Count%2) ^ alternate;
162 }
163 return s[SERVE];
164 }
165 public bool NormalWin(int x, int extra) {
166 int a = s[P+x]+extra, b = s[P+1-x];
167 return a >= 4 && (options["DEUCE"].Equals("NOAD") || a-b >= 2);

169 public void EndSet(int x, Finished score) {
170 finished.Add(score);
171 s[W+x]++;
172 int target = Tennis() ? Option("SETS")/2+1 : sport.Equals("TABLETENNIS") ? Option("GAMES")/2+1 : sport.Equals("VOLLEYBALL") ? 3 : 2;
173 if (s[W+x] == target) {
174 s[DONE] = 1;
175 s[WINNER] = x;
176 } else {
177 s[P] = 0;
178 s[P+1] = 0;
179 s[G] = 0;
180 s[G+1] = 0;
181 if (sport.Equals("VOLLEYBALL")) {
182 for (int side = 0; side < 2; side++) {
183 s[SUBS+side] = 0;
184 s[TIMEOUTS+side] = 0;
185 Array.Copy(initial[side],0,s,LINEUP+side*6,6);
186 }
187 s[SERVE] = First() ^ (finished.Count%2);
188 }
189 }

191 public void Point(int x) {
192 int serve = Server();
193 s[POINTS+x]++;
194 s[SERVED+serve]++;
195 if (serve == x) s[SERVEWON+x]++;
196 s[CURRENT+1-x] = 0;
197 s[CURRENT+x]++;
198 s[RUN+x] = Math.Max(s[RUN+x],s[CURRENT+x]);
199 if (Tennis()) {
200 bool tie = TieBreak(), superTie = SuperTie();
201 if (!tie && NormalWin(1-serve,1)) {
202 s[BPO+1-serve]++;
203 if (x != serve) s[BPW+x]++;
204 }
205 s[P+x]++;
206 if (tie) {
207 int target = superTie || FinalSet() && options["FINAL"].Equals("TB10") ? 10 : 7;
208 if (s[P+x] >= target && s[P+x]-s[P+1-x] >= 2) {
209 Finished score;
210 if (superTie) score = new Finished(s[P],s[P+1],2,0);
211 else {
212 s[G+x]++;
213 score = new Finished(s[G],s[G+1],1,s[P+1-x]);
214 }
215 s[GS] ^= 1;
216 EndSet(x,score);
217 }
218 } else if (NormalWin(x,0)) {
219 s[SERVICE+s[GS]]++;
220 if (x == s[GS]) s[HOLDS+x]++;
221 else s[BREAKS+x]++;
222 s[G+x]++;
223 s[P] = 0;
224 s[P+1] = 0;
225 s[GS] ^= 1;
226 if (s[G+x] >= 6 && s[G+x]-s[G+1-x] >= 2) EndSet(x,new Finished(s[G],s[G+1],0,0));
227 }
228 return;
229 }
230 s[P+x]++;
231 if (sport.Equals("BADMINTON")) s[SERVE] = x;
232 if (sport.Equals("VOLLEYBALL") && s[SERVE] != x) {
233 int start = LINEUP+x*6, old = s[start];
234 Array.Copy(s,start+1,s,start,5);
235 s[start+5] = old;
236 s[SERVE] = x;
237 }
238 int target = sport.Equals("TABLETENNIS") ? 11 : sport.Equals("BADMINTON") ? 21 : finished.Count == 4 ? 15 : 25;
239 if (s[P+x] >= target && s[P+x]-s[P+1-x] >= 2 || sport.Equals("BADMINTON") && s[P+x] == 30) {
240 EndSet(x,new Finished(s[P],s[P+1],0,0));
241 }
242 }
243 public void CompleteTurn(int side) {
244 int score = s[TV];
245 s[HIGH+side] = Math.Max(s[HIGH+side],score);
246 if (score >= 100) s[TONS+side]++;
247 if (score == 180) s[MAXIMUMS+side]++;
248 }
249 public void NextTurn() {
250 s[SERVE] ^= 1;
251 s[TC] = 0;
252 s[TV] = 0;
253 s[TS] = s[R+s[SERVE]];
254 }
255 public void ApplyDart(int code) {
256 Dart d = app.darts[code];
257 int x = s[SERVE], before = s[R+x];
258 if (s[TC] == 0) s[TS] = before;
259 if (before == 50 || before <= 40 && before%2 == 0) s[ATTEMPTS+x]++;
260 s[DARTS+x]++;
261 s[TURN+s[TC]] = code;
262 s[TC]++;
263 s[TV] += d.Value;
264 int remaining = before-d.Value;
265 if (remaining < 0 || remaining == 1 || remaining == 0 && !d.DoubleOut) {
266 s[SCORED+x] -= s[TV]-d.Value;
267 s[BUSTS+x]++;
268 s[R+x] = s[TS];
269 NextTurn();
270 } else {
271 s[R+x] = remaining;
272 s[SCORED+x] += d.Value;
273 if (remaining == 0) {
274 s[HITS+x]++;
275 CompleteTurn(x);
276 s[L+x]++;
277 if (s[L+x] == Option("LEGS")) {
278 finished.Add(new Finished(s[L],s[L+1],0,0));
279 s[W+x]++;
280 if (s[W+x] == Option("SETS")) {
281 s[DONE] = 1;
282 s[WINNER] = x;
283 return;
284 }
285 s[L] = 0;
286 s[L+1] = 0;
287 }
288 s[LEG]++;
289 s[R] = Option("START");
290 s[R+1] = Option("START");
291 s[SERVE] = First() ^ ((s[LEG]-1)%2);
292 s[TC] = 0;
293 s[TV] = 0;
294 s[TS] = Option("START");
295 } else if (s[TC] == 3) {
296 CompleteTurn(x);
297 NextTurn();
298 }
299 }

301 public void Roll(int pins) {
302 s[ROLLS+s[ROLLCOUNT]] = pins;
303 s[ROLLCOUNT]++;
304 int r = s[FRAMEROLLS], frame = s[FRAME];
305 if (r == 0) s[FIRSTROLL] = pins;
306 if (r == 1) s[SECONDROLL] = pins;
307 s[STANDING] -= pins;
308 bool end;
309 if (frame < 9) end = r == 1 || r == 0 && pins == 10;
310 else end = r == 2 || r == 1 && s[FIRSTROLL] != 10 && s[FIRSTROLL]+pins < 10;
311 if (end) {
312 if (frame == 9) s[DONE] = 1;
313 else {
314 s[FRAME]++;
315 s[FRAMEROLLS] = 0;
316 s[STANDING] = 10;
317 }
318 } else {
319 s[FRAMEROLLS]++;
320 if (frame == 9 && s[STANDING] == 0) s[STANDING] = 10;
321 }

323 public List<BowlingFrame> BowlingFrames() {
324 List<BowlingFrame> result = new();
325 int cursor = 0, cumulative = 0;
326 for (int frame = 0; frame < 10 && cursor < s[ROLLCOUNT]; frame++) {
327 int first = s[ROLLS+cursor], count;
328 if (frame < 9) count = first == 10 ? 1 : Math.Min(2,s[ROLLCOUNT]-cursor);
329 else count = Math.Min(3,s[ROLLCOUNT]-cursor);
330 StringBuilder marks = new StringBuilder();
331 int standing = 10, sum = 0;
332 bool fresh = true;
333 for (int i = 0; i < count; i++) {
334 int pins = s[ROLLS+cursor+i];
335 marks.Append(pins == 10 && fresh ? "X" : pins == standing ? "/" : pins == 0 ? "-" : Convert.ToString(pins));
336 sum += pins;
337 standing -= pins;
338 fresh = frame == 9 && standing == 0;
339 if (fresh) standing = 10;
340 }
341 int? score = null;
342 if (frame == 9) {
343 if (Over()) score = sum;
344 } else if (first == 10 && cursor+2 < s[ROLLCOUNT]) {
345 score = 10+s[ROLLS+cursor+1]+s[ROLLS+cursor+2];
346 } else if (first != 10 && count == 2) {
347 if (sum < 10) score = sum;
348 else if (cursor+2 < s[ROLLCOUNT]) score = 10+s[ROLLS+cursor+2];
349 }
350 int? total = null;
351 if (score != null) {
352 cumulative += score.Value;
353 total = cumulative;
354 }
355 bool open = count >= 2 && first != 10 && first+s[ROLLS+cursor+1] < 10;
356 result.Add(new BowlingFrame(marks.ToString(),total,open));
357 cursor += count;
358 }
359 return result;

361 public void Event(string command, string value, string[] tokens) {
362 if (Over()) Fail("match over");
363 int code = 0, side = 0, position = -1;
364 if (command.Equals("ROLL")) {
365 code = int.Parse(value);
366 if (code > s[STANDING]) Fail("too many pins");
367 }
368 if (command.Equals("THROW")) {
369 if (!app.dartIndex.TryGetValue(value,out code)) Fail("invalid dart");
370 }
371 if (command.Equals("SUB") || command.Equals("TIMEOUT")) {
372 side = tokens[2].Equals("A") ? 0 : 1;
373 if (command.Equals("SUB")) {
374 int from = int.Parse(tokens[3]), to = int.Parse(tokens[4]);
375 int start = LINEUP+side*6;
376 position = Array.IndexOf(s,from,start,6)-start;
377 if (position < 0) Fail("player not on court");
378 if (Array.IndexOf(s,to,start,6) >= 0) Fail("player already on court");
379 if (s[SUBS+side] == 6) Fail("no substitutions left");
380 } else if (s[TIMEOUTS+side] == 2) Fail("no timeouts left");
381 }
382 int[] before = (int[])s.Clone();
383 int count = finished.Count;
384 switch (command) {
385 case "POINT":
386 Point(value.Equals("A") ? 0 : 1);
387 break;
388 case "ROLL":
389 Roll(code);
390 break;
391 case "THROW":
392 ApplyDart(code);
393 break;
394 case "SUB":
395 s[LINEUP+side*6+position] = int.Parse(tokens[4]);
396 s[SUBS+side]++;
397 break;
398 case "TIMEOUT":
399 s[TIMEOUTS+side]++;
400 break;
401 }
402 int changed = 0;
403 for (int i = 0; i < s.Length; i++) {
404 if (before[i] != s[i]) changed++;
405 }
406 int[] undo = new int[1+changed*2];
407 undo[0] = count;
408 int cursor = 1;
409 for (int i = 0; i < s.Length; i++) {
410 if (before[i] != s[i]) {
411 undo[cursor++] = i;
412 undo[cursor++] = before[i];
413 }
414 }
415 history.Push(undo);
416 }
417 public void Undo() {
418 if (history.Count == 0) Fail("nothing to undo");
419 int[] undo = history.Pop();
420 for (int i = 1; i < undo.Length; i += 2) s[undo[i]] = undo[i+1];
421 finished.RemoveRange(undo[0],finished.Count-undo[0]);
422 }
423 public string Situation(int x) {
424 int[] old = (int[])s.Clone();
425 int count = finished.Count, games = s[G]+s[G+1];
426 int serve = Server();
427 bool normal = Tennis() && !TieBreak();
428 Point(x);
429 string result = Over() ? "match point" : finished.Count > count ? (Tennis() || sport.Equals("VOLLEYBALL") ? "set point" : "game point") :
430 normal && s[G]+s[G+1] > games ? (x == serve ? "game point" : "break point") : "";
431 s = old;
432 finished.RemoveRange(count,finished.Count-count);
433 return result;
434 }

442 var frames = BowlingFrames().Select(f => f.Marks+(f.Cumulative == null ? "" : "("+f.Cumulative+")")).ToList();

469 items.Add(s[P] == s[P+1] ? "40-40" : s[P] > s[P+1] ? "AD-40" : "40-AD");

476 items.Add(Pair(P));

478 }
479 for (int x = 0; x < 2; x++) {
480 string situation = Situation(x);
481 if (situation.Length != 0) items.Add("("+situation+" "+names[x]+")");
482 }
483 }
484 }
485 return id+": "+names[0]+(serve == 0 ? "*" : "")+" vs "+names[1]+(serve == 1 ? "*" : "")+" "+string.Join(" ",items);

487 public List<string> Stats() {
488 List<string> output = new();
489 if (sport.Equals("BOWLING")) {
490 List<BowlingFrame> frames = BowlingFrames();

492 int spares = frames.Sum(f => f.Marks.Count(c => c == '/'));

500 }

502 string line = id+" "+names[x]+": ";

504 long average = s[DARTS+x] == 0 ? 0 : (long)s[SCORED+x]*300/s[DARTS+x];

514 }
515 output.Add(line);
516 }

527 List<string> Execute(string line) {
528 string[] t = line.Split(' ');
529 string command = t[0];
530 if (command.Equals("MATCH")) {
531 if (matches.ContainsKey(t[1])) Fail("duplicate match");
532 Match match = new Match(this,t);
533 matches[t[1]] = match;
534 return new List<string> { };
535 }
536 if (command.Equals("CHECKOUT")) {
537 List<string> answer = Checkout(int.Parse(t[1]),3);
538 return new List<string> { "checkout "+t[1]+": "+(answer.Count == 0 ? "none" : string.Join(" ",answer)) };
539 }
540 if (command.Equals("STANDINGS")) {
541 Dictionary<string,int[]> totals = new(StringComparer.Ordinal);
542 foreach (Match match in matches.Values) {
543 if (match.Over() && !match.sport.Equals("BOWLING")) {
544 int winner = match.s[WINNER];
545 if (!totals.ContainsKey(match.names[winner])) totals[match.names[winner]] = new int[2];
546 totals[match.names[winner]][0]++;
547 if (!totals.ContainsKey(match.names[1-winner])) totals[match.names[1-winner]] = new int[2];
548 totals[match.names[1-winner]][1]++;
549 }
550 }
551 List<string> names = totals.Keys.ToList();
552 names = names.OrderByDescending(name => totals[name][0]).ThenBy(name => totals[name][1]).ThenBy(name => name,StringComparer.Ordinal).ToList();
553 var records = names.Select(name => name+" "+totals[name][0]+"-"+totals[name][1]).ToList();
554 return new List<string> { "standings: "+(records.Count == 0 ? "none" : string.Join(", ",records)) };
555 }

558 if (command.Equals("UNDO")) {

560 return new List<string> { };

562 if (command.Equals("SCORE")) return new List<string> { target.Score() };
563 if (command.Equals("STATS")) return target.Stats();
564 if (command.Equals("RESULT")) return new List<string> { target.Result() };
