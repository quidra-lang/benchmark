55static char *format(const char *pattern, ...) {
56 va_list args, copy;
57 va_start(args, pattern);
58 va_copy(copy, args);
59 int length = vsnprintf(NULL, 0, pattern, copy);
60 va_end(copy);
61 char *result = allocate((size_t)length + 1);
62 vsnprintf(result, (size_t)length + 1, pattern, args);
63 va_end(args);
64 return result;
65}
66static void appendf(char **s, const char *pattern, ...) {
67 va_list args, copy;
68 va_start(args, pattern);
69 va_copy(copy, args);
70 int length = vsnprintf(NULL, 0, pattern, copy);
71 va_end(copy);
72 size_t old = strlen(*s);
73 *s = realloc(*s, old + (size_t)length + 1);
74 if (!*s) exit(EXIT_FAILURE);
75 vsnprintf(*s + old, (size_t)length + 1, pattern, args);
76 va_end(args);
77}
78static void push_owned(Strings *s, char *item) {
79 if (s->count == s->capacity) {
80 s->capacity = s->capacity ? s->capacity * 2 : 8;
81 s->items = realloc(s->items, (size_t)s->capacity * sizeof *s->items);
82 if (!s->items) exit(EXIT_FAILURE);
83 }
84 s->items[s->count++] = item;
85}
86static void free_strings(Strings *s) {
87 for (int i = 0; i < s->count; ++i) free(s->items[i]);
88 free(s->items);
89 *s = (Strings){0};
90}
91static Strings split(const char *s, char delimiter) {
92 Strings result = {0};
93 const char *start = s;
94 for (const char *p = s;; ++p) {
95 if (!*p || *p == delimiter) {
96 push_owned(&result, slice(start, (size_t)(p - start)));
97 start = p + 1;
98 if (!*p) break;
99 }
100 }
101 return result;
102}

107static int maximum(int a, int b) { return a > b ? a : b; }
108static void initialize_darts(void) {
109 int count = 0;
110 for (int group = 0; group < 3; ++group) {
111 int value = group == 0 ? 1 : group == 1 ? 3 : 2;
112 char prefix = value == 1 ? 'S' : value == 2 ? 'D' : 'T';
113 for (int n = 20; n >= 1; --n) {
114 snprintf(darts[count].name, sizeof darts[count].name, "%c%d", prefix, n);
115 darts[count].value = value * n;
116 darts[count].is_double = value == 2;
117 ++count;
118 }
119 }
120 darts[60] = (Dart){"25", 25, false};
121 darts[61] = (Dart){"BULL", 50, true};
122 darts[62] = (Dart){"MISS", 0, false};
123}
124static int double_dart(int value) {
125 if (value == 50) return 61;
126 if (value >= 2 && value <= 40 && value % 2 == 0) return 60 - value / 2;
127 return -1;
128}
129static int checkout(int score, int maximum_darts, int *result) {
130 int last = double_dart(score);
131 if (last >= 0) {
132 result[0] = last;
133 return 1;
134 }
135 if (maximum_darts >= 2) {
136 for (int i = 0; i < 62; ++i) {
137 last = double_dart(score - darts[i].value);
138 if (last >= 0) {
139 result[0] = i;
140 result[1] = last;
141 return 2;
142 }
143 }
144 }
145 if (maximum_darts >= 3 && score <= 170) {
146 for (int i = 0; i < 62; ++i) {
147 for (int j = 0; j < 62; ++j) {
148 last = double_dart(score - darts[i].value - darts[j].value);
149 if (last >= 0) {
150 result[0] = i;
151 result[1] = j;
152 result[2] = last;
153 return 3;
154 }
155 }
156 }
157 }
158 return 0;
159}
160static void append_darts(char **s, const int *ids, int count) {
161 for (int i = 0; i < count; ++i) appendf(s, "%s%s", i ? " " : "", darts[ids[i]].name);
162}
163static BowlingInfo bowling_info(const BowlingState *state) {
164 BowlingInfo info = {0};
165 info.standing = 10;
166 info.fresh = true;
167 for (int i = 0; i < state->count; ++i) {
168 int frame = info.frame, pins = state->rolls[i];
169 if (!info.count[frame]) info.start[frame] = i;
170 info.marks[i] = info.fresh && pins == 10 ? 'X' : pins == info.standing ? '/' : pins == 0 ? '-' : (char)('0' + pins);
171 ++info.count[frame];
172 info.standing -= pins;
173 info.fresh = false;
174 bool ended = false;
175 if (frame < 9) ended = (pins == 10 && info.count[frame] == 1) || info.count[frame] == 2;
176 else if (info.count[frame] == 2) ended = state->rolls[info.start[frame]] + pins < 10;
177 else if (info.count[frame] == 3) ended = true;
178 if (ended) {
179 info.closed[frame] = true;
180 if (frame == 9) info.finished = true;
181 else ++info.frame;
182 info.standing = 10;
183 info.fresh = true;
184 } else if (frame == 9 && info.standing == 0) {
185 info.standing = 10;
186 info.fresh = true;
187 }
188 }
189 for (int frame = 0; frame < 10; ++frame) {
190 if (!info.count[frame]) continue;
191 int start = info.start[frame];
192 if (frame == 9) {
193 info.final[frame] = info.closed[frame];
194 for (int j = 0; j < info.count[frame]; ++j) info.score[frame] += state->rolls[start + j];
195 } else if (state->rolls[start] == 10) {
196 info.final[frame] = state->count >= start + 3;
197 if (info.final[frame]) info.score[frame] = 10 + state->rolls[start + 1] + state->rolls[start + 2];
198 } else if (info.count[frame] >= 2) {
199 int total = state->rolls[start] + state->rolls[start + 1];
200 info.final[frame] = total < 10 || state->count >= start + 3;
201 if (info.final[frame]) info.score[frame] = total + (total == 10 ? state->rolls[start + 2] : 0);
202 }
203 }
204 return info;
205}
206static int bowling_total(const BowlingInfo *info) {
207 int total = 0;
208 for (int i = 0; i < 10; ++i) total += info->score[i];
209 return total;
210}
211static bool final_set(const Match *m, const Rally *s) { return s->sets[0] == m->sets / 2 && s->sets[1] == m->sets / 2; }
212static int tennis_kind(const Match *m, const Rally *s) {
213 if (final_set(m, s) && m->final == 3) return 2;
214 if (s->games[0] == 6 && s->games[1] == 6 && !(final_set(m, s) && m->final == 2)) return 1;
215 return 0;
216}
217static int rally_server(const Match *m, const Rally *s) {
218 if (m->sport == TENNIS && tennis_kind(m, s)) return s->server ^ (((s->points[0] + s->points[1] + 1) / 2) % 2);
219 if (m->sport == TABLETENNIS) {
220 int first = m->first ^ (s->segment_count % 2), played = s->points[0] + s->points[1];
221 return first ^ (played < 20 ? played / 2 % 2 : played % 2);
222 }
223 return s->server;
224}
225static bool normal_game(const Match *m, int own, int other) { return own >= 4 && (m->no_ad || own - other >= 2); }
226static void add_rally(const Match *m, Rally *s, int side) {
227 int other = 1 - side, server = rally_server(m, s);
228 ++s->stats[side][0];
229 ++s->stats[server][1];
230 if (side == server) ++s->stats[side][2];
231 ++s->stats[side][3];
232 s->stats[other][3] = 0;
233 s->stats[side][4] = maximum(s->stats[side][4], s->stats[side][3]);
234 int kind = m->sport == TENNIS ? tennis_kind(m, s) : 0;
235 if (m->sport == TENNIS && !kind) {
236 int receiver = 1 - s->server;
237 if (normal_game(m, s->points[receiver] + 1, s->points[s->server])) {
238 ++s->stats[receiver][8];
239 if (side == receiver) ++s->stats[receiver][9];
240 }
241 }
242 ++s->points[side];
243 if (m->sport == TENNIS) {
244 int target = kind == 2 || (kind == 1 && final_set(m, s) && m->final == 1) ? 10 : 7;
245 bool ended = kind ? s->points[side] >= target && s->points[side] - s->points[other] >= 2 : normal_game(m, s->points[side], s->points[other]);
246 if (!ended) return;
247 Segment segment;
248 if (!kind) {
249 ++s->stats[s->server][5];
250 if (side == s->server) ++s->stats[side][6];
251 else ++s->stats[side][7];
252 }
253 if (kind == 2) segment = (Segment){s->points[0], s->points[1], 2, 0};
254 else {
255 ++s->games[side];
256 segment = (Segment){s->games[0], s->games[1], kind, kind ? s->points[other] : 0};
257 }
258 s->server ^= 1;
259 memset(s->points, 0, sizeof s->points);
260 bool set_ended = kind || (s->games[side] >= 6 && s->games[side] - s->games[other] >= 2);
261 if (set_ended) {
262 s->segments[s->segment_count++] = segment;
263 ++s->sets[side];
264 memset(s->games, 0, sizeof s->games);
265 if (s->sets[side] == m->sets / 2 + 1) s->winner = side;
266 }
267 return;
268 }
269 if (m->sport == VOLLEYBALL && side != s->server) {
270 s->server = side;
271 int first = s->lineup[side][0];
272 memmove(s->lineup[side], s->lineup[side] + 1, 5 * sizeof(int));
273 s->lineup[side][5] = first;
274 }
275 if (m->sport == BADMINTON) s->server = side;
276 int target = m->sport == TABLETENNIS ? 11 : m->sport == BADMINTON ? 21 : s->segment_count == 4 ? 15 : 25;
277 bool ended = s->points[side] >= target && s->points[side] - s->points[other] >= 2;
278 if (m->sport == BADMINTON && s->points[side] == 30) ended = true;
279 if (!ended) return;
280 s->segments[s->segment_count++] = (Segment){s->points[0], s->points[1], 0, 0};
281 ++s->sets[side];
282 int needed = m->sport == TABLETENNIS ? m->games / 2 + 1 : m->sport == BADMINTON ? 2 : 3;
283 if (s->sets[side] == needed) s->winner = side;
284 else {
285 memset(s->points, 0, sizeof s->points);
286 if (m->sport == VOLLEYBALL) {
287 memcpy(s->lineup, m->initial_lineup, sizeof s->lineup);
288 memset(s->substitutions, 0, sizeof s->substitutions);
289 memset(s->timeouts, 0, sizeof s->timeouts);
290 s->server = m->first ^ (s->segment_count % 2);
291 }
292 }
293}
294static void end_darts_turn(DartsState *s) {
295 s->server ^= 1;
296 s->turn_start = s->remaining[s->server];
297 s->turn_count = s->turn_score = 0;
298}
299static void completed_darts_turn(DartsState *s, int side) {
300 s->stats[side][3] = maximum(s->stats[side][3], s->turn_score);
301 if (s->turn_score >= 100) ++s->stats[side][4];
302 if (s->turn_score == 180) ++s->stats[side][5];
303}
304static void add_dart(const Match *m, DartsState *s, int id) {
305 int side = s->server, remaining = s->remaining[side];
306 Dart dart = darts[id];
307 ++s->stats[side][0];
308 if (remaining == 50 || (remaining <= 40 && remaining % 2 == 0)) ++s->stats[side][6];
309 s->turn[s->turn_count++] = id;
310 s->turn_score += dart.value;
311 s->stats[side][1] += dart.value;
312 int after = remaining - dart.value;
313 if (after < 0 || after == 1 || (after == 0 && !dart.is_double)) {
314 s->stats[side][1] -= s->turn_score;
315 ++s->stats[side][2];
316 s->remaining[side] = s->turn_start;
317 end_darts_turn(s);
318 return;
319 }
320 s->remaining[side] = after;
321 if (!after) {
322 ++s->stats[side][7];
323 completed_darts_turn(s, side);
324 ++s->legs[side];
325 if (s->legs[side] == m->legs) {
326 memcpy(s->segments[s->segment_count++], s->legs, sizeof s->legs);
327 ++s->sets[side];
328 if (s->sets[side] == m->sets) {
329 s->winner = side;
330 return;
331 }
332 memset(s->legs, 0, sizeof s->legs);
333 }
334 ++s->leg_number;
335 s->remaining[0] = s->remaining[1] = m->start;
336 s->server = m->first ^ ((s->leg_number - 1) % 2);
337 s->turn_start = m->start;
338 s->turn_count = s->turn_score = 0;
339 } else if (s->turn_count == 3) {
340 completed_darts_turn(s, side);
341 end_darts_turn(s);
342 }
343}
344static bool finished(const Match *m) {
345 if (m->sport == BOWLING) return bowling_info(&m->state.bowling).finished;
346 return m->sport == DARTS ? m->state.darts.winner >= 0 : m->state.rally.winner >= 0;
347}
348static int winner(const Match *m) { return m->sport == DARTS ? m->state.darts.winner : m->state.rally.winner; }
349static void append_segment(char **line, const Segment *s, int first) {
350 int a = first ? s->b : s->a, b = first ? s->a : s->b;
351 appendf(line, s->kind == 2 ? " [%d-%d]" : " %d-%d", a, b);
352 if (s->kind == 1) appendf(line, "(%d)", s->loser);
353}
354static const char *situation(const Match *m, const Rally *s, int side) {
355 Rally next = *s;
356 add_rally(m, &next, side);
357 if (next.winner >= 0) return "match point";
358 if (next.segment_count != s->segment_count) return m->sport == TABLETENNIS || m->sport == BADMINTON ? "game point" : "set point";
359 if (m->sport == TENNIS && tennis_kind(m, s) == 0 && next.server != s->server) return rally_server(m, s) == side ? "game point" : "break point";
360 return "";
361}
362static char *score(const Match *m) {
363 if (m->sport == BOWLING) {
364 const BowlingState *s = &m->state.bowling;
365 BowlingInfo info = bowling_info(s);
366 char *line = format("%s: %s", m->id, m->names[0]);
367 int cumulative = 0;
368 for (int frame = 0; frame < 10; ++frame) {
369 if (!info.count[frame]) continue;
370 appendf(&line, " ");
371 for (int j = 0; j < info.count[frame]; ++j) appendf(&line, "%c", info.marks[info.start[frame] + j]);
372 cumulative += info.score[frame];
373 if (info.final[frame]) appendf(&line, "(%d)", cumulative);
374 }
375 return line;
376 }
377 int server = m->sport == DARTS ? m->state.darts.server : rally_server(m, &m->state.rally);
378 bool over = finished(m);
379 char *line = format("%s: %s%s vs %s%s", m->id, m->names[0], !over && server == 0 ? "*" : "", m->names[1], !over && server == 1 ? "*" : "");
380 if (m->sport == DARTS) {
381 const DartsState *s = &m->state.darts;
382 appendf(&line, " sets %d-%d legs %d-%d score %d-%d", s->sets[0], s->sets[1], s->legs[0], s->legs[1], s->remaining[0], s->remaining[1]);
383 if (!over) {
384 if (s->turn_count) {
385 appendf(&line, " turn ");
386 append_darts(&line, s->turn, s->turn_count);
387 }
388 int ids[3], count = checkout(s->remaining[s->server], 3 - s->turn_count, ids);
389 if (count) {
390 appendf(&line, " checkout ");
391 append_darts(&line, ids, count);
392 }
393 }
394 return line;
395 }
396 const Rally *s = &m->state.rally;
397 for (int i = 0; i < s->segment_count; ++i) append_segment(&line, &s->segments[i], 0);
398 if (!over) {
399 if (m->sport == TENNIS) {
400 int kind = tennis_kind(m, s);
401 if (kind == 2) appendf(&line, " [%d-%d]", s->points[0], s->points[1]);
402 else {
403 appendf(&line, " %d-%d ", s->games[0], s->games[1]);
404 if (kind) appendf(&line, "%d-%d", s->points[0], s->points[1]);
405 else if (!m->no_ad && s->points[0] >= 3 && s->points[1] >= 3) appendf(&line, "%s", s->points[0] == s->points[1] ? "40-40" : s->points[0] > s->points[1] ? "AD-40" : "40-AD");
406 else {
407 const char *names[] = {"0", "15", "30", "40"};
408 appendf(&line, "%s-%s", names[s->points[0]], names[s->points[1]]);
409 }
410 }
411 } else {
412 appendf(&line, " %d-%d", s->points[0], s->points[1]);
413 if (m->sport == VOLLEYBALL) appendf(&line, " server %d", s->lineup[s->server][0]);
414 }
415 for (int side = 0; side < 2; ++side) {
416 const char *label = situation(m, s, side);
417 if (*label) appendf(&line, " (%s %s)", label, m->names[side]);
418 }
419 }
420 return line;
421}
422static void statistics(const Match *m, Strings *out) {
423 if (m->sport == BOWLING) {
424 BowlingState s = m->state.bowling;
425 BowlingInfo info = bowling_info(&s);
426 int strikes = 0, spares = 0, open = 0;
427 for (int i = 0; i < s.count; ++i) {
428 strikes += info.marks[i] == 'X';
429 spares += info.marks[i] == '/';
430 }
431 for (int i = 0; i < 10; ++i) {
432 int start = info.start[i];
433 if (info.count[i] >= 2 && s.rolls[start] != 10 && s.rolls[start] + s.rolls[start + 1] < 10) ++open;
434 }
435 while (!info.finished) {
436 s.rolls[s.count++] = info.standing;
437 info = bowling_info(&s);
438 }
439 push_owned(out, format("%s %s: strikes %d spares %d open %d max %d", m->id, m->names[0], strikes, spares, open, bowling_total(&info)));
440 return;
441 }
442 for (int side = 0; side < 2; ++side) {
443 char *line = format("%s %s: ", m->id, m->names[side]);
444 if (m->sport == DARTS) {
445 const int *s = m->state.darts.stats[side];
446 int average = s[0] ? s[1] * 300 / s[0] : 0;
447 appendf(&line, "darts %d average %d.%02d busts %d high %d tons %d 180s %d doubles %d/%d", s[0], average / 100, average % 100, s[2], s[3], s[4], s[5], s[7], s[6]);
448 } else {
449 const Rally *r = &m->state.rally;
450 const int *s = r->stats[side];
451 appendf(&line, "points %d serve %d/%d run %d", s[0], s[2], s[1], s[4]);
452 if (m->sport == TENNIS) appendf(&line, " holds %d/%d breaks %d break points %d/%d", s[6], s[5], s[7], s[9], s[8]);
453 if (m->sport == VOLLEYBALL) {
454 appendf(&line, " subs %d timeouts %d lineup ", r->substitutions[side], r->timeouts[side]);
455 for (int i = 0; i < 6; ++i) appendf(&line, "%s%d", i ? "," : "", r->lineup[side][i]);
456 }
457 }
458 push_owned(out, line);
459 }
460}
461static char *result(const Match *m) {
462 if (!finished(m)) fail("match not finished");
463 if (m->sport == BOWLING) {
464 BowlingInfo info = bowling_info(&m->state.bowling);
465 return format("%s: %s %d", m->id, m->names[0], bowling_total(&info));
466 }
467 int first = winner(m);
468 char *line = format("%s: %s def. %s", m->id, m->names[first], m->names[1 - first]);
469 if (m->sport == DARTS) {
470 const DartsState *s = &m->state.darts;
471 for (int i = 0; i < s->segment_count; ++i) appendf(&line, " %d-%d", s->segments[i][first], s->segments[i][1 - first]);
472 } else {
473 const Rally *s = &m->state.rally;
474 for (int i = 0; i < s->segment_count; ++i) append_segment(&line, &s->segments[i], first);
475 }
476 return line;
477}
478static Match *find_match(const char *id) {
479 for (int i = 0; i < match_count; ++i) if (!strcmp(matches[i].id, id)) return &matches[i];
480 return NULL;
481}
482static void create_match(const Strings *a) {
483 if (find_match(a->items[1])) fail("duplicate match");
484 Match m = {0};
485 strcpy(m.id, a->items[1]);
486 const char *sports[] = {"TENNIS", "TABLETENNIS", "BADMINTON", "VOLLEYBALL", "BOWLING", "DARTS"};
487 for (int i = 0; i < 6; ++i) if (!strcmp(sports[i], a->items[2])) m.sport = i;
488 strcpy(m.names[0], a->items[3]);
489 if (m.sport != BOWLING) strcpy(m.names[1], a->items[4]);
490 m.sets = m.sport == DARTS ? 1 : 3;
491 m.games = 5;
492 m.start = 501;
493 m.legs = 1;
494 int option_start = m.sport == BOWLING ? 4 : m.sport == VOLLEYBALL ? 7 : 5;
495 const char *keys[] = {"FIRST", "SETS", "GAMES", "DEUCE", "FINAL", "START", "LEGS"};
496 const char *values[] = {"A B", "3 5", "5 7", "AD NOAD", "TB7 TB10 ADV SUPER", "301 501 701", "1 2 3 4 5"};
497 bool seen[7] = {false};
498 for (int i = option_start; i < a->count; ++i) {
499 Strings option = split(a->items[i], '=');
500 int key = -1;
501 if (option.count == 2) {
502 for (int j = 0; j < 7; ++j) if (!strcmp(keys[j], option.items[0])) key = j;
503 }
504 bool allowed = key == 0 && m.sport != BOWLING
505 || key == 1 && (m.sport == TENNIS || m.sport == DARTS)
506 || key == 2 && m.sport == TABLETENNIS
507 || (key == 3 || key == 4) && m.sport == TENNIS
508 || (key == 5 || key == 6) && m.sport == DARTS;
509 if (!allowed || seen[key]) {
510 free_strings(&option);
511 fail("invalid option");
512 }
513 Strings choices = split(key == 1 && m.sport == DARTS ? values[6] : values[key], ' ');
514 int choice = -1;
515 for (int j = 0; j < choices.count; ++j) if (!strcmp(choices.items[j], option.items[1])) choice = j;
516 if (choice < 0) {
517 free_strings(&choices);
518 free_strings(&option);
519 fail("invalid option");
520 }
521 seen[key] = true;
522 if (key == 0) m.first = choice;
523 if (key == 1) m.sets = atoi(option.items[1]);
524 if (key == 2) m.games = atoi(option.items[1]);
525 if (key == 3) m.no_ad = choice == 1;
526 if (key == 4) m.final = choice;
527 if (key == 5) m.start = atoi(option.items[1]);
528 if (key == 6) m.legs = atoi(option.items[1]);
529 free_strings(&choices);
530 free_strings(&option);
531 }
532 if (m.sport == DARTS) {
533 m.state.darts.remaining[0] = m.state.darts.remaining[1] = m.start;
534 m.state.darts.server = m.first;
535 m.state.darts.turn_start = m.start;
536 m.state.darts.leg_number = 1;
537 m.state.darts.winner = -1;
538 } else if (m.sport != BOWLING) {
539 m.state.rally.server = m.first;
540 m.state.rally.winner = -1;
541 if (m.sport == VOLLEYBALL) {
542 for (int side = 0; side < 2; ++side) {
543 Strings numbers = split(a->items[5 + side], ',');
544 for (int i = 0; i < 6; ++i) m.initial_lineup[side][i] = atoi(numbers.items[i]);
545 free_strings(&numbers);
546 }
547 memcpy(m.state.rally.lineup, m.initial_lineup, sizeof m.initial_lineup);
548 }
549 }
550 matches[match_count++] = m;
551}
552typedef struct { char name[13]; int wins, losses; } Standing;
553static int compare_standings(const void *left, const void *right) {
554 const Standing *a = left, *b = right;
555 if (a->wins != b->wins) return a->wins > b->wins ? -1 : 1;
556 if (a->losses != b->losses) return a->losses < b->losses ? -1 : 1;
557 return strcmp(a->name, b->name);
558}
559static char *standings(void) {
560 Standing rows[2000] = {0};
561 int count = 0;
562 for (int i = 0; i < match_count; ++i) {
563 Match *m = &matches[i];
564 if (m->sport == BOWLING || !finished(m)) continue;
565 int winning_side = winner(m);
566 for (int side = 0; side < 2; ++side) {
567 int id = 0;
568 while (id < count && strcmp(rows[id].name, m->names[side])) ++id;
569 if (id == count) strcpy(rows[count++].name, m->names[side]);
570 if (side == winning_side) ++rows[id].wins;
571 else ++rows[id].losses;
572 }
573 }
574 if (count > 1) qsort(rows, count, sizeof *rows, compare_standings);
575 char *line = format("standings: ");
576 if (!count) appendf(&line, "none");
577 for (int i = 0; i < count; ++i) appendf(&line, "%s%s %d-%d", i ? ", " : "", rows[i].name, rows[i].wins, rows[i].losses);
578 return line;
579}
580static void remember(Match *m) {
581 if (m->history_count == m->history_capacity) {
582 m->history_capacity = m->history_capacity ? m->history_capacity * 2 : 16;
583 m->history = realloc(m->history, (size_t)m->history_capacity * sizeof *m->history);
584 if (!m->history) exit(EXIT_FAILURE);
585 }
586 m->history[m->history_count++] = m->state;
587}
588static void execute(const Strings *a, Strings *out) {
589 const char *command = a->items[0];
590 if (!strcmp(command, "MATCH")) {
591 create_match(a);
592 return;
593 }
594 if (!strcmp(command, "CHECKOUT")) {
595 int ids[3], count = checkout(atoi(a->items[1]), 3, ids);
596 char *line = format("checkout %s: ", a->items[1]);
597 if (count) append_darts(&line, ids, count);
598 else appendf(&line, "none");
599 push_owned(out, line);
600 return;
601 }
602 if (!strcmp(command, "STANDINGS")) {
603 push_owned(out, standings());
604 return;
605 }
606 Match *m = find_match(a->items[1]);
607 if (!m) fail("unknown match");
608 if (!strcmp(command, "SCORE")) {
609 push_owned(out, score(m));
610 return;
611 }
612 if (!strcmp(command, "STATS")) {
613 statistics(m, out);
614 return;
615 }
616 if (!strcmp(command, "RESULT")) {
617 push_owned(out, result(m));
618 return;
619 }
620 if (!strcmp(command, "UNDO")) {
621 if (!m->history_count) fail("nothing to undo");
622 m->state = m->history[--m->history_count];
623 return;
624 }
625 bool point = !strcmp(command, "POINT"), roll = !strcmp(command, "ROLL"), dart = !strcmp(command, "THROW");
626 bool sub = !strcmp(command, "SUB"), timeout = !strcmp(command, "TIMEOUT");
627 if (!((point && m->sport <= VOLLEYBALL) || (roll && m->sport == BOWLING) || (dart && m->sport == DARTS) || ((sub || timeout) && m->sport == VOLLEYBALL))) fail("wrong event");
628 int count = point ? (int)strlen(a->items[2]) : roll || dart ? a->count - 2 : 1;
629 for (int i = 0; i < count; ++i) {
630 if (finished(m)) fail("match over");
631 int value = 0, position = -1, side = 0;
632 if (point) value = a->items[2][i] == 'B';
633 if (roll) {
634 value = atoi(a->items[i + 2]);
635 if (value > bowling_info(&m->state.bowling).standing) fail("too many pins");
636 }
637 if (dart) {
638 value = -1;
639 for (int j = 0; j < 63; ++j) if (!strcmp(darts[j].name, a->items[i + 2])) value = j;
640 if (value < 0) fail("invalid dart");
641 }
642 if (sub || timeout) {
643 side = !strcmp(a->items[2], "B");
644 const Rally *s = &m->state.rally;
645 if (timeout) {
646 if (s->timeouts[side] >= 2) fail("no timeouts left");
647 } else {
648 int leaving = atoi(a->items[3]);
649 value = atoi(a->items[4]);
650 for (int j = 0; j < 6; ++j) if (s->lineup[side][j] == leaving) position = j;
651 if (position < 0) fail("player not on court");
652 for (int j = 0; j < 6; ++j) if (s->lineup[side][j] == value) fail("player already on court");
653 if (s->substitutions[side] >= 6) fail("no substitutions left");
654 }
655 }
656 remember(m);
657 if (point) add_rally(m, &m->state.rally, value);
658 if (roll) {
659 BowlingState *s = &m->state.bowling;
660 s->rolls[s->count++] = value;
661 }
662 if (dart) add_dart(m, &m->state.darts, value);
663 if (sub) {
664 m->state.rally.lineup[side][position] = value;
665 ++m->state.rally.substitutions[side];
666 }
667 if (timeout) ++m->state.rally.timeouts[side];
668 }
669}
670int main(void) {
671 initialize_darts();
672 char line[400];
673 if (!fgets(line, sizeof line, stdin)) return 0;
674 int count = atoi(line);
675 for (int i = 0; i < count; ++i) {
676 if (!fgets(line, sizeof line, stdin)) break;
677 line[strcspn(line, "\n")] = '\0';
678 Strings a = split(line, ' ');
679 if (setjmp(recovery) == 0) {
680 Strings out = {0};
681 execute(&a, &out);
682 for (int j = 0; j < out.count; ++j) puts(out.items[j]);
683 free_strings(&out);
684 } else printf("ERROR %s\n", error_message);
685 free_strings(&a);
686 }
687 for (int i = 0; i < match_count; ++i) free(matches[i].history);
688 return 0;
689}
