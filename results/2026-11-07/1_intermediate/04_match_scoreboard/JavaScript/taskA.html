18const darts = new Map();
19for (let n = 1; n <= 20; n++) {
20 darts.set("S" + n, { value: n, double: false });
21 darts.set("D" + n, { value: 2 * n, double: true });
22 darts.set("T" + n, { value: 3 * n, double: false });
23}
24darts.set("25", { value: 25, double: false });
25darts.set("BULL", { value: 50, double: true });
26darts.set("MISS", { value: 0, double: false });
27const candidatesA = [];
28const candidatesB = [];
29for (const prefix of ["S", "T", "D"]) {
30 for (let n = 20; n >= 1; n--) {
31 candidatesA.push(prefix + n);
32 if (prefix === "D") {
33 candidatesB.push(prefix + n);
34 }
35 }
36}
37candidatesA.push("25", "BULL");
38candidatesB.push("BULL");
39const checkoutCache = new Map();
40
41function checkout(score, maximum = 3) {
42 const key = score + ":" + maximum;
43 if (checkoutCache.has(key)) {
44 return checkoutCache.get(key);
45 }
46 let result = null;
47 if (maximum >= 1) {
48 const last = candidatesB.find(dart => darts.get(dart).value === score);
49 if (last) {
50 result = [last];
51 }
52 }
53 if (!result && maximum >= 2) {
54 for (const first of candidatesA) {
55 const last = candidatesB.find(dart => darts.get(first).value + darts.get(dart).value === score);
56 if (last) {
57 result = [first, last];
58 break;
59 }
60 }
61 }
62 if (!result && maximum >= 3) {
63 outer: for (const first of candidatesA) {
64 for (const second of candidatesA) {
65 const wanted = score - darts.get(first).value - darts.get(second).value;
66 const last = candidatesB.find(dart => darts.get(dart).value === wanted);
67 if (last) {
68 result = [first, second, last];
69 break outer;
70 }
71 }
72 }
73 }
74 checkoutCache.set(key, result);
75 return result;
76}

103 constructor(args) {
104 this.id = args[0];
105 this.sport = args[1];
106 this.names = this.sport === "BOWLING" ? [args[2]] : [args[2], args[3]];
107 this.initialLineups = this.sport === "VOLLEYBALL" ?
108 [args[4].split(",").map(Number), args[5].split(",").map(Number)] : [[], []];
109 const optionStart = this.sport === "BOWLING" ? 3 : this.sport === "VOLLEYBALL" ? 6 : 4;
110 this.options = { ...defaults[this.sport] };
111 const seen = new Set();
112 for (const option of args.slice(optionStart)) {
113 const parts = option.split("=");
114 if (parts.length !== 2 || !Object.hasOwn(choices[this.sport], parts[0]) ||
115 !choices[this.sport][parts[0]].includes(parts[1]) || seen.has(parts[0])) {
116 fail("invalid option");
117 }
118 seen.add(parts[0]);
119 this.options[parts[0]] = parts[1];
120 }
121 this.first = this.options.FIRST === "B" ? 1 : 0;
122 const start = Number(this.options.START ?? 501);
123 this.state = {
124 done: false, winner: -1, sets: [0, 0], legs: [0, 0], points: [0, 0], games: [0, 0],
125 finished: [], gameServer: this.first, server: this.first,
126 lineups: this.initialLineups.map(lineup => [...lineup]), subs: [0, 0], timeouts: [0, 0],
127 rally: Array.from({ length: 2 }, () => ({

131 frames: Array.from({ length: 10 }, () => []), marks: Array.from({ length: 10 }, () => []),
132 frame: 0, standing: 10, remaining: [start, start], player: this.first, legNumber: 0,
133 turn: [], turnStart: start, turnScore: 0,

135 darts: 0, points: 0, busts: 0, high: 0, tons: 0, maximums: 0, attempts: 0, hits: 0

137 };

162 if (this.tennisKind(s) === "normal") {

198 point(s, winner) {
199 const server = this.server(s);
200 s.rally[winner].points++;
201 s.rally[server].served++;
202 if (winner === server) {
203 s.rally[server].serveWon++;
204 }
205 s.rally[winner].current++;
206 s.rally[winner].run = Math.max(s.rally[winner].run, s.rally[winner].current);
207 s.rally[1 - winner].current = 0;
208 if (this.sport === "TENNIS") {
209 const kind = this.tennisKind(s);
210 if (kind === "normal") {
211 const receiver = 1 - server;
212 const hypothetical = [...s.points];
213 hypothetical[receiver]++;
214 if (this.normalGameWins(hypothetical, receiver)) {
215 s.rally[receiver].opportunities++;
216 if (winner === receiver) {
217 s.rally[receiver].conversions++;
218 }
219 }
220 }
221 s.points[winner]++;
222 const targetSets = Math.floor(Number(this.options.SETS) / 2) + 1;
223 if (kind === "normal" && this.normalGameWins(s.points, winner)) {
224 s.rally[server].serviceGames++;
225 if (winner === server) {
226 s.rally[server].holds++;
227 } else {
228 s.rally[winner].breaks++;
229 }
230 s.games[winner]++;
231 s.points = [0, 0];
232 s.gameServer ^= 1;
233 if (s.games[winner] >= 6 && s.games[winner] - s.games[1 - winner] >= 2) {
234 this.finishRallyUnit(s, winner, { values: [...s.games], tie: -1, super: false }, targetSets);
235 }
236 } else if (kind !== "normal") {
237 const target = kind === "super" || (this.finalSet(s) && this.options.FINAL === "TB10") ? 10 : 7;
238 if (s.points[winner] >= target && s.points[winner] - s.points[1 - winner] >= 2) {
239 const record = kind === "super" ?
240 { values: [...s.points], tie: -1, super: true } :
241 { values: s.games.map((value, i) => value + (i === winner ? 1 : 0)), tie: s.points[1 - winner], super: false };
242 s.gameServer ^= 1;
243 this.finishRallyUnit(s, winner, record, targetSets);
244 }
245 }
246 } else {
247 s.points[winner]++;
248 if (this.sport === "BADMINTON") {
249 s.server = winner;
250 } else if (this.sport === "VOLLEYBALL" && winner !== s.server) {
251 s.server = winner;
252 const player = s.lineups[winner].shift();
253 s.lineups[winner].push(player);
254 }
255 const target = this.sport === "TABLETENNIS" ? 11 : this.sport === "BADMINTON" ? 21 :
256 s.finished.length === 4 ? 15 : 25;
257 const won = (s.points[winner] >= target && s.points[winner] - s.points[1 - winner] >= 2) ||
258 (this.sport === "BADMINTON" && s.points[winner] === 30);
259 if (won) {
260 const needed = this.sport === "TABLETENNIS" ? Math.floor(Number(this.options.GAMES) / 2) + 1 :
261 this.sport === "BADMINTON" ? 2 : 3;
262 this.finishRallyUnit(s, winner, { values: [...s.points], tie: -1, super: false }, needed);
263 }
264 }

267 standing(s) {
268 return s.frame === 9 && s.standing === 0 ? 10 : s.standing;

271 roll(s, pins) {
272 const standing = this.standing(s);
273 if (pins > standing) {
274 fail("too many pins");
275 }
276 const frame = s.frames[s.frame];
277 const freshRack = frame.length === 0 || (s.frame === 9 && s.standing === 0);
278 const mark = freshRack && pins === 10 ? "X" : pins === standing ? "/" : pins === 0 ? "-" : String(pins);
279 frame.push(pins);
280 s.marks[s.frame].push(mark);
281 s.standing = standing - pins;
282 if (s.frame < 9) {
283 if (pins === 10 && frame.length === 1 || frame.length === 2) {
284 s.frame++;
285 s.standing = 10;
286 }
287 } else if (frame.length === 3 || (frame.length === 2 && frame[0] !== 10 && frame[0] + frame[1] < 10)) {
288 s.done = true;
289 }
290 }

292 frameScores(s) {
293 return s.frames.map((frame, i) => {
294 if (i === 9) {
295 return s.done ? frame.reduce((a, b) => a + b, 0) : null;
296 }
297 const later = s.frames.slice(i + 1).flat();
298 if (frame[0] === 10) {
299 return later.length >= 2 ? 10 + later[0] + later[1] : null;
300 }
301 if (frame.length < 2) {
302 return null;
303 }
304 if (frame[0] + frame[1] === 10) {
305 return later.length ? 10 + later[0] : null;
306 }
307 return frame[0] + frame[1];
308 });
309 }

337 throwDart(s, notation) {
338 const dart = darts.get(notation);
339 if (!dart) {
340 fail("invalid dart");
341 }
342 const player = s.player;
343 const stats = s.darts[player];
344 const before = s.remaining[player];
345 stats.darts++;
346 if (before === 50 || before <= 40 && before % 2 === 0) {
347 stats.attempts++;
348 }
349 s.turn.push(notation);
350 s.turnScore += dart.value;
351 stats.points += dart.value;
352 const remaining = before - dart.value;
353 if (remaining < 0 || remaining === 1 || remaining === 0 && !dart.double) {
354 stats.busts++;
355 stats.points -= s.turnScore;
356 s.remaining[player] = s.turnStart;
357 this.nextTurn(s);
358 } else {
359 s.remaining[player] = remaining;
360 if (remaining === 0) {
361 stats.hits++;
362 this.completedTurn(s, player);
363 s.legs[player]++;
364 if (s.legs[player] === Number(this.options.LEGS)) {
365 s.sets[player]++;
366 s.finished.push({ values: [...s.legs], tie: -1, super: false });
367 if (s.sets[player] === Number(this.options.SETS)) {
368 s.done = true;
369 s.winner = player;
370 return;
371 }
372 s.legs = [0, 0];
373 }
374 s.legNumber++;
375 const start = Number(this.options.START);
376 s.remaining = [start, start];
377 s.player = this.first ^ (s.legNumber % 2);
378 s.turn = [];
379 s.turnScore = 0;
380 s.turnStart = start;
381 } else if (s.turn.length === 3) {
382 this.completedTurn(s, player);
383 this.nextTurn(s);
384 }
385 }

388 apply(name, args) {
389 if (name === "POINT") {
390 this.point(this.state, args[0] === "A" ? 0 : 1);
391 } else if (name === "ROLL") {
392 this.roll(this.state, Number(args[0]));
393 } else if (name === "THROW") {
394 this.throwDart(this.state, args[0]);
395 } else {
396 const side = args[0] === "A" ? 0 : 1;
397 if (name === "SUB") {
398 const lineup = this.state.lineups[side];
399 const position = lineup.indexOf(Number(args[1]));
400 if (position < 0) {
401 fail("player not on court");
402 }
403 if (lineup.includes(Number(args[2]))) {
404 fail("player already on court");
405 }
406 if (this.state.subs[side] >= 6) {
407 fail("no substitutions left");
408 }
409 lineup[position] = Number(args[2]);
410 this.state.subs[side]++;
411 } else {
412 if (this.state.timeouts[side] >= 2) {
413 fail("no timeouts left");
414 }
415 this.state.timeouts[side]++;
416 }
417 }

420 event(name, args) {
421 const allowed = name === "POINT" ? this.rallySport() : name === "ROLL" ? this.sport === "BOWLING" :
422 name === "THROW" ? this.sport === "DARTS" : this.sport === "VOLLEYBALL";
423 if (!allowed) {
424 fail("wrong event");

426 const events = name === "POINT" ? args[0].split("").map(value => [value]) :
427 name === "ROLL" || name === "THROW" ? args.map(value => [value]) : [args];
428 for (const values of events) {
429 if (this.state.done) {
430 fail("match over");
431 }
432 this.apply(name, values);
433 this.history.push({ name, args: values });
434 }

460 score() {
461 const s = this.state;
462 if (this.sport === "BOWLING") {
463 const scores = this.frameScores(s);
464 let total = 0;
465 const items = [];
466 for (let i = 0; i < 10; i++) {
467 if (!s.frames[i].length) {
468 break;
469 }
470 let value = s.marks[i].join("");
471 if (scores[i] !== null) {
472 total += scores[i];
473 value += "(" + total + ")";
474 }
475 items.push(value);
476 }
477 return this.id + ": " + this.names[0] + (items.length ? " " + items.join(" ") : "");
478 }
479 const server = this.sport === "DARTS" ? s.player : this.server(s);
480 const names = this.names.map((name, i) => name + (!s.done && i === server ? "*" : ""));
481 let items = s.finished.map(record => this.scoreRecord(record));
482 if (this.sport === "DARTS") {
483 items = ["sets", pair(s.sets), "legs", pair(s.legs), "score", pair(s.remaining)];
484 if (!s.done) {
485 if (s.turn.length) {
486 items.push("turn", ...s.turn);
487 }
488 const finish = checkout(s.remaining[s.player], 3 - s.turn.length);
489 if (finish) {
490 items.push("checkout", ...finish);
491 }
492 }
493 } else if (!s.done) {
494 if (this.sport === "TENNIS") {
495 const kind = this.tennisKind(s);
496 if (kind === "super") {
497 items.push("[" + pair(s.points) + "]");
498 } else {
499 items.push(pair(s.games));
500 if (kind === "tie") {
501 items.push(pair(s.points));
502 } else if (this.options.DEUCE === "AD" && s.points.every(value => value >= 3)) {
503 items.push(s.points[0] === s.points[1] ? "40-40" : s.points[0] > s.points[1] ? "AD-40" : "40-AD");
504 } else {
505 items.push(s.points.map(value => ["0", "15", "30", "40"][value]).join("-"));
506 }
507 }
508 } else {
509 items.push(pair(s.points));
510 if (this.sport === "VOLLEYBALL") {
511 items.push("server", String(s.lineups[s.server][0]));
512 }
513 }
514 for (let side = 0; side < 2; side++) {
515 const situation = this.situation(side);
516 if (situation) {
517 items.push("(" + situation + " " + this.names[side] + ")");
518 }
519 }
520 }
521 return this.id + ": " + names[0] + " vs " + names[1] + " " + items.join(" ");
522 }

524 stats() {
525 const s = this.state;
526 if (this.sport === "BOWLING") {
527 const marks = s.marks.flat();
528 const open = s.frames.filter(frame => frame.length >= 2 && frame[0] !== 10 && frame[0] + frame[1] < 10).length;
529 return [this.id + " " + this.names[0] + ": strikes " + marks.filter(mark => mark === "X").length +
530 " spares " + marks.filter(mark => mark === "/").length + " open " + open + " max " + this.maximum(s)];
531 }
532 return this.names.map((name, i) => {
533 let result = this.id + " " + name + ": ";
534 if (this.sport === "DARTS") {
535 const stats = s.darts[i];
536 const average = stats.darts ? Math.floor(stats.points * 300 / stats.darts) : 0;
537 result += "darts " + stats.darts + " average " + Math.floor(average / 100) + "." +
538 String(average % 100).padStart(2, "0") + " busts " + stats.busts + " high " + stats.high +
539 " tons " + stats.tons + " 180s " + stats.maximums + " doubles " + stats.hits + "/" + stats.attempts;
540 } else {
541 const stats = s.rally[i];
542 result += "points " + stats.points + " serve " + stats.serveWon + "/" + stats.served + " run " + stats.run;
543 if (this.sport === "TENNIS") {
544 result += " holds " + stats.holds + "/" + stats.serviceGames + " breaks " + stats.breaks +
545 " break points " + stats.conversions + "/" + stats.opportunities;
546 } else if (this.sport === "VOLLEYBALL") {
547 result += " subs " + s.subs[i] + " timeouts " + s.timeouts[i] + " lineup " + s.lineups[i].join(",");
548 }
549 }
550 return result;
551 });

569function command(line) {
570 const [name, ...args] = line.split(" ");
571 if (name === "MATCH") {
572 if (matches.has(args[0])) {
573 fail("duplicate match");
574 }
575 const match = new Match(args);
576 matches.set(match.id, match);
577 return [];
578 }
579 if (name === "CHECKOUT") {
580 const answer = checkout(Number(args[0]));
581 return ["checkout " + args[0] + ": " + (answer ? answer.join(" ") : "none")];
582 }
583 if (name === "STANDINGS") {
584 const totals = new Map();
585 for (const match of matches.values()) {
586 if (!match.state.done || match.sport === "BOWLING") {
587 continue;
588 }
589 match.names.forEach((person, i) => {
590 if (!totals.has(person)) {
591 totals.set(person, [0, 0]);
592 }
593 totals.get(person)[i === match.state.winner ? 0 : 1]++;
594 });
595 }
596 const ordered = [...totals].sort((a, b) => b[1][0] - a[1][0] || a[1][1] - b[1][1] || lex(a[0], b[0]));
597 return ["standings: " + (ordered.length ? ordered.map(([person, values]) => person + " " + pair(values)).join(", ") : "none")];
598 }
599 const match = matches.get(args[0]);
600 if (!match) {
601 fail("unknown match");
602 }
603 if (name === "UNDO") {
604 const previous = match.history.pop();
605 if (!previous) {
606 fail("nothing to undo");
607 }
608 match.state = structuredClone(match.initialState);
609 for (const event of match.history) {
610 match.apply(event.name, event.args);
611 }
612 return [];
613 }
614 if (name === "SCORE") {
615 return [match.score()];
616 }
617 if (name === "STATS") {
618 return match.stats();
619 }
620 if (name === "RESULT") {
621 return [match.result()];
622 }
623 match.event(name, args.slice(1));
624 return [];
625}
