Config settings(const vector<string> &tokens) {
    Config c;
    set<string> seen;
    const set<string> keys = {"page",  "frames", "tlb",  "sets", "policy",
                              "scope", "asid",   "tick", "tau",  "seed"};
    for (size_t i = 1; i < tokens.size(); i++) {
        size_t eq = tokens[i].find('=');
        string key = tokens[i].substr(0, eq), value = tokens[i].substr(eq + 1);
        if (!keys.count(key))
            throw runtime_error("unknown key " + key);
        if (!seen.insert(key).second)
            throw runtime_error("duplicate key " + key);
        bool valid = false;
        if (key == "policy") {
            valid = knownPolicy(value);
            if (valid)
                c.policy = value;
        } else if (key == "scope") {
            valid = value == "GLOBAL" || value == "LOCAL";
            if (valid)
                c.scope = value;
        } else if (key == "asid") {
            valid = value == "ON" || value == "OFF";
            if (valid)
                c.asid = value;
        } else {
            int64_t n = stoll(value);
            if (key == "page")
                valid = n >= 16 && n <= 4096 && (n & (n - 1)) == 0;
            else if (key == "frames")
                valid = n >= 1 && n <= 16;
            else if (key == "tlb" || key == "sets")
                valid = n >= 1 && n <= 8;
            else if (key == "tick")
                valid = n >= 0 && n <= 1000;
            else if (key == "tau")
                valid = n >= 1 && n <= 1000;
            else
                valid = n >= 1 && n <= 2147483646LL;
            if (valid) {
                if (key == "page")
                    c.page = int(n);
                else if (key == "frames")
                    c.frames = int(n);
                else if (key == "tlb")
                    c.tlb = int(n);
                else if (key == "sets")
                    c.sets = int(n);
                else if (key == "tick")
                    c.tick = int(n);
                else if (key == "tau")
                    c.tau = int(n);
                else
                    c.seed = n;
            }
        }
        if (!valid)
            throw runtime_error("invalid " + key);
    }
    if (c.tlb % c.sets != 0)
        throw runtime_error("invalid sets");
    return c;
}
