46struct Config {
47 int page = 256, frames = 4, tlb = 4, sets = 1, tick = 0, tau = 8;
48 int64_t seed = 1;
49 string policy = "FIFO", scope = "GLOBAL", asid = "OFF";
50 string description() const {
51 return "SYSTEM page=" + to_string(page) + " frames=" + to_string(frames) +
52 " tlb=" + to_string(tlb) + " sets=" + to_string(sets) + " policy=" + policy +
53 " scope=" + scope + " asid=" + asid + " tick=" + to_string(tick) +
54 " tau=" + to_string(tau) + " seed=" + to_string(seed);
55 }
56};
57Config settings(const vector<string> &tokens) {
58 Config c;
59 set<string> seen;
60 const set<string> keys = {"page", "frames", "tlb", "sets", "policy",
61 "scope", "asid", "tick", "tau", "seed"};
62 for (size_t i = 1; i < tokens.size(); i++) {
63 size_t eq = tokens[i].find('=');
64 string key = tokens[i].substr(0, eq), value = tokens[i].substr(eq + 1);
65 if (!keys.count(key))
66 throw runtime_error("unknown key " + key);
67 if (!seen.insert(key).second)
68 throw runtime_error("duplicate key " + key);
69 bool valid = false;
70 if (key == "policy") {
71 valid = knownPolicy(value);
72 if (valid)
73 c.policy = value;
74 } else if (key == "scope") {
75 valid = value == "GLOBAL" || value == "LOCAL";
76 if (valid)
77 c.scope = value;
78 } else if (key == "asid") {
79 valid = value == "ON" || value == "OFF";
80 if (valid)
81 c.asid = value;
82 } else {
83 int64_t n = stoll(value);
84 if (key == "page")
85 valid = n >= 16 && n <= 4096 && (n & (n - 1)) == 0;
86 else if (key == "frames")
87 valid = n >= 1 && n <= 16;
88 else if (key == "tlb" || key == "sets")
89 valid = n >= 1 && n <= 8;
90 else if (key == "tick")
91 valid = n >= 0 && n <= 1000;
92 else if (key == "tau")
93 valid = n >= 1 && n <= 1000;
94 else
95 valid = n >= 1 && n <= 2147483646LL;
96 if (valid) {
97 if (key == "page")
98 c.page = int(n);
99 else if (key == "frames")
100 c.frames = int(n);
101 else if (key == "tlb")
102 c.tlb = int(n);
103 else if (key == "sets")
104 c.sets = int(n);
105 else if (key == "tick")
106 c.tick = int(n);
107 else if (key == "tau")
108 c.tau = int(n);
109 else
110 c.seed = n;
111 }
112 }
113 if (!valid)
114 throw runtime_error("invalid " + key);
115 }
116 if (c.tlb % c.sets != 0)
117 throw runtime_error("invalid sets");
118 return c;
119}

172 void emit(const string &text) {
173 if (!silent)
174 output.push_back(text);

176 Process &process(int pid) {
177 auto it = processes.find(pid);
178 if (it == processes.end())
179 throw runtime_error("unknown process " + to_string(pid));
180 if (it->second.exited)
181 throw runtime_error("process " + to_string(pid) + " exited");
182 return it->second;
183 }
184 int quota(int pid) const {
185 int index = int(distance(processes.begin(), processes.find(pid)));
186 return c.frames / int(processes.size()) +
187 (index < c.frames % int(processes.size()) ? 1 : 0);
188 }
189 void declare(int pid, int count) {
190 if (fixed)
191 throw runtime_error("processes are fixed");
192 if (processes.count(pid))
193 throw runtime_error("duplicate process " + to_string(pid));
194 if (count < 1 || count > 256)
195 throw runtime_error("invalid page count");
196 if (processes.size() >= 8 || (c.scope == "LOCAL" && int(processes.size()) == c.frames))
197 throw runtime_error("too many processes");
198 processes.emplace(pid, Process(pid, count));
199 }
200 void invalidate(Page *page) {
201 for (vector<Page *> &set : tlb)
202 set.erase(remove(set.begin(), set.end(), page), set.end());
203 }
204 void write(Page *p) {
205 writebacks++;
206 processes.at(p->pid).writebacks++;
207 p->M = 0;
208 if (p->slot < 0) {
209 int slot = 0;
210 while (swap.count(slot))
211 slot++;
212 p->slot = slot;
213 swap[slot] = p;
214 }
215 }
216 int nextUse(const Page *p) const {
217 for (int i = cursor + 1; i < int(future.size()); i++) {
218 const Access &a = future[i];
219 if (a.pid == p->pid && a.address / c.page == p->number)
220 return i;
221 }
222 return numeric_limits<int>::max();
223 }
224 int choose(vector<int> candidates, Process &owner) {
225 vector<int> unlocked;
226 for (int f : candidates)
227 if (!frames[f]->locked)
228 unlocked.push_back(f);
229 if (!unlocked.empty())
230 candidates = unlocked;
231 if (c.policy == "RANDOM") {
232 random = random * 48271LL % 2147483647LL;
233 return candidates[random % candidates.size()];
234 }
235 if (c.policy != "CLOCK" && c.policy != "ECLOCK" && c.policy != "WSCLOCK") {
236 auto rank = [&](int f) -> pair<int, int> {
237 const Page &p = *frames[f];
238 if (c.policy == "FIFO")
239 return {p.load, 0};
240 if (c.policy == "LRU")
241 return {p.last, 0};
242 if (c.policy == "LFU")
243 return {p.uses, p.load};
244 if (c.policy == "NRU")
245 return {2 * p.R + p.M, f};
246 if (c.policy == "AGING")
247 return {p.age, p.load};
248 return {-nextUse(&p), f};
249 };
250 return *min_element(candidates.begin(), candidates.end(),
251 [&](int a, int b) { return rank(a) < rank(b); });
252 }
253 int h = c.scope == "GLOBAL" ? hand : owner.hand;
254 sort(candidates.begin(), candidates.end(), [&](int a, int b) {
255 return (a - h + c.frames) % c.frames < (b - h + c.frames) % c.frames;
256 });
257 int chosen = -1;
258 if (c.policy == "CLOCK") {
259 while (chosen < 0)
260 for (int f : candidates) {
261 if (frames[f]->R == 0) {
262 chosen = f;
263 break;
264 }
265 frames[f]->R = 0;
266 }
267 } else if (c.policy == "ECLOCK") {
268 while (chosen < 0) {
269 for (int f : candidates)
270 if (frames[f]->R == 0 && frames[f]->M == 0) {
271 chosen = f;
272 break;
273 }
274 if (chosen >= 0)
275 break;
276 for (int f : candidates) {
277 if (frames[f]->R == 0 && frames[f]->M == 1) {
278 chosen = f;
279 break;
280 }
281 frames[f]->R = 0;
282 }
283 }
284 } else {
285 for (int f : candidates) {
286 Page *p = frames[f];
287 if (p->R == 1) {
288 p->R = 0;
289 p->last = time;
290 } else if (time - p->last > c.tau) {
291 if (p->M == 0) {
292 chosen = f;
293 break;
294 }
295 write(p);
296 }
297 }
298 if (chosen < 0) {
299 for (int f : candidates)
300 if (frames[f]->M == 0) {
301 chosen = f;
302 break;
303 }
304 if (chosen < 0)
305 chosen = candidates[0];
306 }
307 }
308 if (c.scope == "GLOBAL")
309 hand = (chosen + 1) % c.frames;
310 else
311 owner.hand = (chosen + 1) % c.frames;
312 return chosen;
313 }
314 void access(int pid, const string &op, int addr) {
315 fixed = true;
316 cursor++;
317 Process &owner = process(pid);
318 if (addr >= int(owner.pages.size()) * c.page) {
319 owner.segfaults++;
320 emit("SEGFAULT P" + to_string(pid) + " " + hexAddress(addr));
321 return;
322 }
323 Page *p = &owner.pages[addr / c.page];
324 if (op == "W" && p->ro) {
325 owner.protfaults++;
326 emit("PROTFAULT P" + to_string(pid) + " " + hexAddress(addr));
327 return;
328 }
329 time++;
330 owner.accesses++;
331 bool switched = previous >= 0 && previous != pid;
332 if (switched) {
333 switches++;
334 if (c.asid == "OFF")
335 for (vector<Page *> &set : tlb)
336 set.clear();
337 }
338 previous = pid;
339 vector<Page *> &set = tlb[p->number % c.sets];
340 auto found = find(set.begin(), set.end(), p);
341 bool hit = found != set.end(), fault = false;
342 if (hit)
343 set.erase(found);
344 string result;
345 if (hit) {
346 tlbhits++;
347 result = "TLBHIT";
348 } else if (p->frame >= 0)
349 result = "HIT";
350 else {
351 fault = true;
352 faults++;
353 owner.faults++;
354 if (p->slot >= 0) {
355 swapin++;
356 result = "FAULT SWAPIN";
357 } else {
358 zero++;
359 result = "FAULT ZERO";
360 }
361 int frame = -1;
362 bool mayUseFree = c.scope == "GLOBAL" || owner.resident() < quota(pid);
363 if (mayUseFree)
364 for (int i = 0; i < c.frames; i++)
365 if (frames[i] == nullptr) {
366 frame = i;
367 break;
368 }
369 if (frame < 0) {
370 vector<int> candidates;
371 for (int i = 0; i < c.frames; i++)
372 if (frames[i] && (c.scope == "GLOBAL" || frames[i]->pid == pid))
373 candidates.push_back(i);
374 frame = choose(candidates, owner);
375 Page *victim = frames[frame];
376 result += " evict=" + victim->label();
377 invalidate(victim);
378 if (victim->M == 1) {
379 write(victim);
380 result += " writeback";
381 }
382 victim->frame = -1;
383 }
384 frames[frame] = p;
385 p->frame = frame;
386 p->load = time;
387 p->uses = 0;
388 p->M = 0;
389 p->age = 0;
390 owner.peak = max(owner.peak, owner.resident());
391 }
392 if (int(set.size()) == c.tlb / c.sets)
393 set.erase(set.begin());
394 set.push_back(p);
395 p->R = 1;
396 if (op == "W")
397 p->M = 1;
398 p->last = time;
399 p->uses++;
400 p->ever = true;
401 p->references++;
402 owner.history.push_back(p->number);
403 faultHistory.push_back(fault);
404 log.push_back({"ACCESS", pid, op, addr, 0});
405 bool interrupt = c.tick > 0 && time % c.tick == 0;
406 if (interrupt)
407 for (Page *page : frames)
408 if (page) {
409 page->age = page->age / 2 + 128 * page->R;
410 page->R = 0;
411 }
412 if (trace) {
413 emit("t=" + to_string(time) + " P" + to_string(pid) + " " + op + " " +
414 hexAddress(addr) + " page=" + to_string(p->number) + " frame=" +
415 to_string(p->frame) + " phys=" + hexAddress(p->frame * c.page + addr % c.page) +
416 " " + result + (switched ? " switch" : ""));
417 if (interrupt)
418 emit("t=" + to_string(time) + " tick");
419 }
420 }
421 void exitProcess(int pid) {
422 Process &p = process(pid);
423 int count = 0, slots = 0;
424 for (Page &page : p.pages) {
425 if (page.frame >= 0) {
426 frames[page.frame] = nullptr;
427 page.frame = -1;
428 count++;
429 }
430 invalidate(&page);
431 if (page.slot >= 0) {
432 swap.erase(page.slot);
433 page.slot = -1;
434 slots++;
435 }
436 }
437 p.exited = true;
438 log.push_back({"EXIT", pid, "", 0, 0});
439 emit("EXIT P" + to_string(pid) + " freed=" + to_string(count) +
440 " slots=" + to_string(slots));
441 }
442 void attributes(const string &cmd, int pid, int first, int last, const string &protection) {
443 Process &p = process(pid);
444 if (first > last || last >= int(p.pages.size()))
445 throw runtime_error("invalid page range");
446 for (int i = first; i <= last; i++) {
447 if (cmd == "PROTECT")
448 p.pages[i].ro = protection == "RO";
449 else
450 p.pages[i].locked = cmd == "LOCK";
451 }
452 if (cmd != "PROTECT")
453 log.push_back({cmd, pid, "", first, last});
454 }
455 void clean() {
456 int count = 0;
457 for (Page *p : frames)
458 if (p && p->M == 1) {
459 write(p);
460 count++;
461 }
462 log.push_back({"CLEAN", 0, "", 0, 0});
463 emit("CLEAN written=" + to_string(count));
464 }
465 unique_ptr<Simulation> rerun(const string &policy, int count) {
466 Config config = c;
467 config.policy = policy;
468 config.frames = count;
469 vector<Access> references;
470 for (const Log &e : log)
471 if (e.kind == "ACCESS")
472 references.push_back({e.pid, e.op, e.a});
473 auto replay = make_unique<Simulation>(config, references, true);
474 for (const auto &entry : processes)
475 replay->declare(entry.first, int(entry.second.pages.size()));
476 for (const Log &e : log) {
477 if (e.kind == "ACCESS")
478 replay->access(e.pid, e.op, e.a);
479 else if (e.kind == "EXIT")
480 replay->exitProcess(e.pid);
481 else if (e.kind == "CLEAN")
482 replay->clean();
483 else
484 replay->attributes(e.kind, e.pid, e.a, e.b, "");
485 }
486 return replay;
487 }
488 void summary() {
489 emit("SUMMARY policy=" + c.policy + " scope=" + c.scope + " time=" + to_string(time));
490 int seg = 0, prot = 0;
491 for (const auto &entry : processes) {
492 seg += entry.second.segfaults;
493 prot += entry.second.protfaults;
494 }
495 emit("total faults=" + to_string(faults) + " zero=" + to_string(zero) +
496 " swapin=" + to_string(swapin) + " faultrate=" + ratio(faults, time) +
497 " tlbhits=" + to_string(tlbhits) + " tlbrate=" + ratio(tlbhits, time) +
498 " writebacks=" + to_string(writebacks) + " switches=" + to_string(switches) +
499 " segfaults=" + to_string(seg) + " protfaults=" + to_string(prot));
500 for (const auto &entry : processes) {
501 const Process &p = entry.second;
502 emit("P" + to_string(p.pid) + " accesses=" + to_string(p.accesses) +
503 " faults=" + to_string(p.faults) + " writebacks=" + to_string(p.writebacks) +
504 " segfaults=" + to_string(p.segfaults) + " protfaults=" + to_string(p.protfaults) +
505 " resident=" + to_string(p.resident()) + " peak=" + to_string(p.peak) +
506 (c.scope == "LOCAL" ? " quota=" + to_string(quota(p.pid)) : "") +
507 (p.exited ? " exited" : ""));
508 }
509 }
510 void frameDump() {
511 int used =
512 int(count_if(frames.begin(), frames.end(), [](Page *p) { return p != nullptr; }));
513 emit("FRAMES used=" + to_string(used) + "/" + to_string(c.frames));
514 for (int i = 0; i < c.frames; i++) {
515 Page *p = frames[i];
516 if (!p)
517 emit(to_string(i) + " free");
518 else
519 emit(to_string(i) + " " + p->label() + " R=" + to_string(p->R) +
520 " M=" + to_string(p->M) + " load=" + to_string(p->load) +
521 " last=" + to_string(p->last) + " uses=" + to_string(p->uses) +
522 " age=" + to_string(p->age) + (p->locked ? " locked" : ""));
523 }
524 if (c.policy == "CLOCK" || c.policy == "ECLOCK" || c.policy == "WSCLOCK") {
525 if (c.scope == "GLOBAL")
526 emit("hand=" + to_string(hand));
527 else {
528 string text = "hand";
529 for (const auto &entry : processes)
530 text += " P" + to_string(entry.first) + "=" + to_string(entry.second.hand);
531 emit(text);
532 }
533 }
534 }
535 void pageTable(int pid) {
536 Process &p = process(pid);
537 emit("PAGETABLE P" + to_string(pid) + " pages=" + to_string(p.pages.size()) +
538 " resident=" + to_string(p.resident()));
539 for (const Page &page : p.pages)
540 if (page.ever) {
541 string info = page.frame >= 0
542 ? "frame=" + to_string(page.frame) + " R=" + to_string(page.R) +
543 " M=" + to_string(page.M)
544 : page.slot >= 0 ? "swap=" + to_string(page.slot)
545 : "zero";
546 emit(to_string(page.number) + " " + info + (page.ro ? " RO" : "") +
547 (page.locked ? " locked" : ""));
548 }
549 }
550 void tlbDump() {
551 int used = 0;
552 for (const auto &set : tlb)
553 used += int(set.size());
554 emit("TLB used=" + to_string(used) + "/" + to_string(c.tlb) + " sets=" + to_string(c.sets));
555 for (int i = 0; i < c.sets; i++) {
556 string text = "set " + to_string(i) + ":";
557 for (auto it = tlb[i].rbegin(); it != tlb[i].rend(); ++it)
558 text += " " + (*it)->label() + "->" + to_string((*it)->frame);
559 emit(text);
560 }
561 }
562 void compare() {
563 emit("COMPARE accesses=" + to_string(time) + " frames=" + to_string(c.frames));
564 map<string, pair<int, int>> scores;
565 for (const string &policy : policies) {
566 auto r = rerun(policy, c.frames);
567 scores[policy] = {r->faults, r->writebacks};
568 emit(policy + " faults=" + to_string(r->faults) +
569 " writebacks=" + to_string(r->writebacks) + " tlbhits=" + to_string(r->tlbhits));
570 }
571 vector<string> ranking = policies;
572 stable_sort(ranking.begin(), ranking.end(),
573 [&](const string &a, const string &b) { return scores[a] < scores[b]; });
574 string text = "ranking";
575 for (const string &p : ranking)
576 text += " " + p;
577 emit(text);
578 }
579 void execute(const vector<string> &t) {
580 string cmd = t[0];
581 if (isdigit(static_cast<unsigned char>(cmd[0]))) {
582 access(stoi(cmd), t[1], address(t[2]));
583 return;
584 }
585 if (cmd == "PROCESS")
586 declare(stoi(t[1]), stoi(t[2]));
587 else if (cmd == "EXIT")
588 exitProcess(stoi(t[1]));
589 else if (cmd == "PROTECT" || cmd == "LOCK" || cmd == "UNLOCK")
590 attributes(cmd, stoi(t[1]), stoi(t[2]), stoi(t[3]), t.size() > 4 ? t[4] : "");
591 else if (cmd == "TRACE")
592 trace = t[1] == "ON";
593 else if (cmd == "SUMMARY")
594 summary();
595 else if (cmd == "FRAMES")
596 frameDump();
597 else if (cmd == "PAGETABLE")
598 pageTable(stoi(t[1]));
599 else if (cmd == "TLB")
600 tlbDump();
601 else if (cmd == "SWAP") {
602 emit("SWAP used=" + to_string(swap.size()));
603 for (const auto &entry : swap)
604 emit(to_string(entry.first) + " " + entry.second->label());
605 } else if (cmd == "CLEAN")
606 clean();
607 else if (cmd == "COMPARE")
608 compare();
609 else if (cmd == "CURVE") {
610 string policy = t[1];
611 int lo = stoi(t[2]), hi = stoi(t[3]);
612 if (!knownPolicy(policy))
613 throw runtime_error("unknown policy");
614 if (lo < 1 || lo > hi || hi > 16 || (c.scope == "LOCAL" && lo < int(processes.size())))
615 throw runtime_error("invalid frame range");
616 emit("CURVE " + policy + " frames=" + to_string(lo) + ".." + to_string(hi));
617 int prior = numeric_limits<int>::max();
618 for (int f = lo; f <= hi; f++) {
619 int value = rerun(policy, f)->faults;
620 emit("F=" + to_string(f) + " faults=" + to_string(value) +
621 (value > prior ? " ANOMALY" : ""));
622 prior = value;
623 }
624 } else if (cmd == "EAT") {
625 int64_t total = stoll(t[1]) * time + stoll(t[2]) * (2LL * time - tlbhits) +
626 stoll(t[3]) * (swapin + writebacks);
627 emit(time == 0 ? "EAT accesses=0"
628 : "EAT accesses=" + to_string(time) + " total=" + to_string(total) +
629 " average=" + decimal(total * 100 / time));
630 } else if (cmd == "TIMELINE") {
631 int w = stoi(t[1]);
632 if (w < 1)
633 throw runtime_error("invalid window");
634 string text = "TIMELINE window=" + to_string(w);
635 for (int i = 0; i < int(faultHistory.size()); i += w) {
636 int count = 0;
637 for (int j = i; j < min(i + w, int(faultHistory.size())); j++)
638 if (faultHistory[j])
639 count++;
640 text += " " + to_string(count);
641 }
642 emit(text);
643 } else if (cmd == "WSET") {
644 int pid = stoi(t[1]);
645 Process &p = process(pid);
646 int w = stoi(t[2]);
647 if (w < 1)
648 throw runtime_error("invalid window");
649 set<int> pages(p.history.begin() + max(0, int(p.history.size()) - w), p.history.end());
650 string text = "WSET P" + to_string(pid) + " window=" + to_string(w) +
651 " size=" + to_string(pages.size());
652 for (int page : pages)
653 text += " " + to_string(page);
654 emit(text);
655 } else if (cmd == "HOT") {
656 int pid = stoi(t[1]);
657 Process &p = process(pid);
658 int k = stoi(t[2]);
659 if (k < 1)
660 throw runtime_error("invalid count");
661 vector<Page *> pages;
662 for (Page &page : p.pages)
663 if (page.references > 0)
664 pages.push_back(&page);
665 sort(pages.begin(), pages.end(), [](Page *a, Page *b) {
666 return a->references != b->references ? a->references > b->references
667 : a->number < b->number;
668 });
669 string text = "HOT P" + to_string(pid);
670 for (int i = 0; i < min(k, int(pages.size())); i++)
671 text += " " + to_string(pages[i]->number) + ":" + to_string(pages[i]->references);
672 emit(text);
673 }
674 }

676int main() {
677 int n;
678 cin >> n;
679 string line;
680 getline(cin, line);
681 vector<vector<string>> commands;
682 for (int i = 0; i < n; i++) {
683 getline(cin, line);
684 vector<string> t = words(line);
685 if (t[0] == "LOOP") {
686 int start = address(t[3]), interval = stoi(t[4]), length = stoi(t[5]),
687 repeats = stoi(t[6]);
688 for (int r = 0; r < repeats; r++)
689 for (int j = 0; j < length; j++)
690 commands.push_back({t[1], t[2], hexAddress(start + j * interval)});
691 } else if (t[0] == "RAND") {
692 int lo = address(t[2]), hi = address(t[3]), count = stoi(t[4]), rate = stoi(t[6]);
693 int64_t x = stoll(t[5]);
694 for (int j = 0; j < count; j++) {
695 x = x * 48271LL % 2147483647LL;
696 int addr = lo + int(x % (hi - lo));
697 x = x * 48271LL % 2147483647LL;
698 commands.push_back({t[1], x % 100 < rate ? "W" : "R", hexAddress(addr)});
699 }
700 } else
701 commands.push_back(t);
702 }
703 unique_ptr<Simulation> simulation;
704 for (int i = 0; i < int(commands.size()); i++) {
705 const vector<string> &t = commands[i];
706 try {
707 if (t[0] == "SYSTEM") {
708 simulation.reset();
709 Config c = settings(t);
710 vector<Access> future;
711 for (int j = i + 1; j < int(commands.size()) && commands[j][0] != "SYSTEM"; j++) {
712 const vector<string> &entry = commands[j];
713 if (isdigit(static_cast<unsigned char>(entry[0][0])))
714 future.push_back({stoi(entry[0]), entry[1], address(entry[2])});
715 }
716 simulation = make_unique<Simulation>(c, future, false);
717 output.push_back(c.description());
718 } else if (!simulation)
719 output.push_back("ERROR no system");
720 else
721 simulation->execute(t);
722 } catch (const runtime_error &e) {
723 output.push_back("ERROR " + string(e.what()));
724 }
725 }
726 for (const string &text : output)
727 cout << text << '\n';
