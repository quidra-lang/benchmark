7 static readonly string[] Policies = { "FIFO", "LRU", "LFU", "NRU", "AGING",
8 "CLOCK", "ECLOCK", "WSCLOCK", "RANDOM", "OPT" };
9 static string Hex(int x) => "0x" + x.ToString("X", CultureInfo.InvariantCulture);
10 static int Address(string s) => int.Parse(s[2..], NumberStyles.HexNumber,
11 CultureInfo.InvariantCulture);
12 static string Decimal(long x) => $"{x / 100}.{x % 100:00}";
13 static string Ratio(long x, long y) => Decimal(y == 0 ? 0 : 10000 * x / y) + "%";
14 sealed class CommandError : Exception {
15 public CommandError(string message) : base(message) {
16 }
17 }
18 sealed class Config {
19 public int Page = 256, Frames = 4, Tlb = 4, Sets = 1, Tick, Tau = 8;
20 public long Seed = 1;
21 public string Policy = "FIFO", Scope = "GLOBAL", Asid = "OFF";
22 public Config Copy() => (Config)MemberwiseClone();
23 public override string ToString() =>
24 $"SYSTEM page={Page} frames={Frames} tlb={Tlb} sets={Sets} policy={Policy} scope={Scope} asid={Asid} tick={Tick} tau={Tau} seed={Seed}";
25 }
26 static Config Settings(string[] tokens) {
27 Config c = new();
28 HashSet<string> seen = new();
29 string[] keys = { "page", "frames", "tlb", "sets", "policy",
30 "scope", "asid", "tick", "tau", "seed" };
31 foreach (string token in tokens.Skip(1)) {
32 string[] pair = token.Split('=');
33 string key = pair[0], value = pair[1];
34 if (!keys.Contains(key))
35 throw new CommandError("unknown key " + key);
36 if (!seen.Add(key))
37 throw new CommandError("duplicate key " + key);
38 bool valid;
39 if (key == "policy") {
40 valid = Policies.Contains(value);
41 if (valid)
42 c.Policy = value;
43 } else if (key == "scope") {
44 valid = value is "GLOBAL" or "LOCAL";
45 if (valid)
46 c.Scope = value;
47 } else if (key == "asid") {
48 valid = value is "ON" or "OFF";
49 if (valid)
50 c.Asid = value;
51 } else {
52 long n = long.Parse(value, CultureInfo.InvariantCulture);
53 valid = key switch { "page" => n >= 16 && n <= 4096 && (n & (n - 1)) == 0,
54 "frames" => n >= 1 && n <= 16,
55 "tlb" or "sets" => n >= 1 && n <= 8,
56 "tick" => n >= 0 && n <= 1000,
57 "tau" => n >= 1 && n <= 1000,
58 _ => n >= 1 && n <= 2147483646 };
59 if (valid)
60 switch (key) {
61 case "page":
62 c.Page = (int)n;
63 break;
64 case "frames":
65 c.Frames = (int)n;
66 break;
67 case "tlb":
68 c.Tlb = (int)n;
69 break;
70 case "sets":
71 c.Sets = (int)n;
72 break;
73 case "tick":
74 c.Tick = (int)n;
75 break;
76 case "tau":
77 c.Tau = (int)n;
78 break;
79 case "seed":
80 c.Seed = n;
81 break;
82 }
83 }
84 if (!valid)
85 throw new CommandError("invalid " + key);
86 }
87 if (c.Tlb % c.Sets != 0)
88 throw new CommandError("invalid sets");
89 return c;
90 }
91 sealed class Page {
92 public readonly int Pid, Number;
93 public int Frame = -1, Slot = -1, R, M, Load, Last, Uses, Age, References;
94 public bool Ro, Locked, Ever;
95 public Page(int pid, int number) {
96 Pid = pid;
97 Number = number;
98 }
99 public string Label => $"P{Pid}:{Number}";
100 }
101 sealed class Process {
102 public readonly int Pid;
103 public int Accesses, Faults, Writebacks, Segfaults, Protfaults, Peak, Hand;
104 public bool Exited;
105 public readonly Page[] Pages;
106 public readonly List<int> History = new();
107 public Process(int pid, int count) {
108 Pid = pid;
109 Pages = Enumerable.Range(0, count).Select(p => new Page(pid, p)).ToArray();
110 }
111 public int Resident => Pages.Count(p => p.Frame >= 0);
112 }
113 sealed record Access(int Pid, string Op, int Address);
114 sealed record Event(string Kind, int Pid = 0, string Op = "", int A = 0, int B = 0);
115 sealed class Simulation {
116 public readonly Config C;
117 public readonly SortedDictionary<int, Process> Processes = new();
118 readonly Page?[] frames;
119 readonly List<Page>[] tlb;
120 readonly SortedDictionary<int, Page> swap = new();
121 readonly List<Access> future;
122 readonly List<Event> log = new();
123 readonly List<bool> faultHistory = new();
124 readonly bool silent;
125 int cursor = -1, previous = -1, hand;
126 long random;
127 bool fixedProcesses, trace;
128 public int Time, Faults, Zero, Swapin, TlbHits, Writebacks, Switches;
129 public Simulation(Config c, List<Access> references, bool silent = false) {
130 C = c;
131 future = references;
132 this.silent = silent;
133 random = c.Seed;
134 frames = new Page?[c.Frames];
135 tlb = Enumerable.Range(0, c.Sets)
136 .Select(
137 _ => new List<Page>())
138 .ToArray();
139 }
140 void Emit(string s) {
141 if (!silent)
142 Console.WriteLine(s);
143 }
144 Process GetProcess(int pid) {
145 if (!Processes.TryGetValue(pid, out Process? p))
146 throw new CommandError("unknown process " + pid);
147 if (p.Exited)
148 throw new CommandError($"process {pid} exited");
149 return p;
150 }
151 int Quota(int pid) => C.Frames / Processes.Count +
152 (Processes.Keys.Take(C.Frames % Processes.Count).Contains(pid) ? 1
153 : 0);
154 public void Declare(int pid, int count) {
155 if (fixedProcesses)
156 throw new CommandError("processes are fixed");
157 if (Processes.ContainsKey(pid))
158 throw new CommandError("duplicate process " + pid);
159 if (count < 1 || count > 256)
160 throw new CommandError("invalid page count");
161 if (Processes.Count >= 8 || C.Scope == "LOCAL" && Processes.Count == C.Frames)
162 throw new CommandError("too many processes");
163 Processes.Add(pid, new Process(pid, count));
164 }
165 void Invalidate(Page p) {
166 foreach (List<Page> set in tlb)
167 set.Remove(p);
168 }
169 void Write(Page p) {
170 Writebacks++;
171 Processes[p.Pid].Writebacks++;
172 p.M = 0;
173 if (p.Slot >= 0)
174 return;
175 int slot = 0;
176 while (swap.ContainsKey(slot))
177 slot++;
178 p.Slot = slot;
179 swap.Add(slot, p);
180 }
181 int NextUse(Page p) {
182 for (int i = cursor + 1; i < future.Count; i++)
183 if (future[i].Pid == p.Pid && future[i].Address / C.Page == p.Number)
184 return i;
185 return int.MaxValue;
186 }
187 int Choose(List<int> candidates, Process owner) {
188 List<int> unlocked = candidates.Where(f => !frames[f]!.Locked).ToList();
189 if (unlocked.Count > 0)
190 candidates = unlocked;
191 IEnumerable<int>? order = C.Policy switch {
192 "FIFO" => candidates.OrderBy(f => frames[f]!.Load),
193 "LRU" => candidates.OrderBy(f => frames[f]!.Last),
194 "LFU" => candidates.OrderBy(f => frames[f]!.Uses).ThenBy(f => frames[f]!.Load),
195 "NRU" => candidates.OrderBy(f => 2 * frames[f]!.R + frames[f]!.M).ThenBy(f => f),
196 "AGING" => candidates.OrderBy(f => frames[f]!.Age).ThenBy(f => frames[f]!.Load),
197 "OPT" => candidates.OrderByDescending(f => NextUse(frames[f]!)).ThenBy(f => f),
198 _ => null
199 };
200 if (order != null)
201 return order.First();
202 if (C.Policy == "RANDOM") {
203 random = random * 48271 % 2147483647;
204 return candidates[(int)(random % candidates.Count)];
205 }
206 int h = C.Scope == "GLOBAL" ? hand : owner.Hand;
207 candidates = candidates.OrderBy(f => (f - h + C.Frames) % C.Frames).ToList();
208 int chosen = -1;
209 if (C.Policy == "CLOCK") {
210 while (chosen < 0)
211 foreach (int f in candidates) {
212 if (frames[f]!.R == 0) {
213 chosen = f;
214 break;
215 }
216 frames[f]!.R = 0;
217 }
218 } else if (C.Policy == "ECLOCK") {
219 while (chosen < 0) {
220 foreach (int f in candidates)
221 if (frames[f]!.R == 0 && frames[f]!.M == 0) {
222 chosen = f;
223 break;
224 }
225 if (chosen >= 0)
226 break;
227 foreach (int f in candidates) {
228 if (frames[f]!.R == 0 && frames[f]!.M == 1) {
229 chosen = f;
230 break;
231 }
232 frames[f]!.R = 0;
233 }
234 }
235 } else {
236 foreach (int f in candidates) {
237 Page p = frames[f]!;
238 if (p.R == 1) {
239 p.R = 0;
240 p.Last = Time;
241 } else if (Time - p.Last > C.Tau) {
242 if (p.M == 0) {
243 chosen = f;
244 break;
245 }
246 Write(p);
247 }
248 }
249 if (chosen < 0)
250 chosen = candidates.FirstOrDefault(f => frames[f]!.M == 0, candidates[0]);
251 }
252 if (C.Scope == "GLOBAL")
253 hand = (chosen + 1) % C.Frames;
254 else
255 owner.Hand = (chosen + 1) % C.Frames;
256 return chosen;
257 }
258 public void Access(int pid, string op, int address) {
259 fixedProcesses = true;
260 cursor++;
261 Process owner = GetProcess(pid);
262 if (address >= owner.Pages.Length * C.Page) {
263 owner.Segfaults++;
264 Emit($"SEGFAULT P{pid} {Hex(address)}");
265 return;
266 }
267 Page p = owner.Pages[address / C.Page];
268 if (op == "W" && p.Ro) {
269 owner.Protfaults++;
270 Emit($"PROTFAULT P{pid} {Hex(address)}");
271 return;
272 }
273 Time++;
274 owner.Accesses++;
275 bool switched = previous >= 0 && previous != pid;
276 if (switched) {
277 Switches++;
278 if (C.Asid == "OFF")
279 foreach (List<Page> s in tlb)
280 s.Clear();
281 }
282 previous = pid;
283 List<Page> set = tlb[p.Number % C.Sets];
284 bool hit = set.Remove(p), fault = false;
285 string result;
286 if (hit) {
287 TlbHits++;
288 result = "TLBHIT";
289 } else if (p.Frame >= 0)
290 result = "HIT";
291 else {
292 fault = true;
293 Faults++;
294 owner.Faults++;
295 if (p.Slot >= 0) {
296 Swapin++;
297 result = "FAULT SWAPIN";
298 } else {
299 Zero++;
300 result = "FAULT ZERO";
301 }
302 int frame = -1;
303 if (C.Scope == "GLOBAL" || owner.Resident < Quota(pid))
304 frame = Array.FindIndex(frames, x => x == null);
305 if (frame < 0) {
306 List<int> candidates =
307 Enumerable.Range(0, C.Frames)
308 .Where(f => frames[f] != null &&
309 (C.Scope == "GLOBAL" || frames[f]!.Pid == pid))
310 .ToList();
311 frame = Choose(candidates, owner);
312 Page victim = frames[frame]!;
313 result += " evict=" + victim.Label;
314 Invalidate(victim);
315 if (victim.M == 1) {
316 Write(victim);
317 result += " writeback";
318 }
319 victim.Frame = -1;
320 }
321 frames[frame] = p;
322 p.Frame = frame;
323 p.Load = Time;
324 p.Uses = 0;
325 p.M = 0;
326 p.Age = 0;
327 owner.Peak = Math.Max(owner.Peak, owner.Resident);
328 }
329 if (set.Count == C.Tlb / C.Sets)
330 set.RemoveAt(0);
331 set.Add(p);
332 p.R = 1;
333 if (op == "W")
334 p.M = 1;
335 p.Last = Time;
336 p.Uses++;
337 p.Ever = true;
338 p.References++;
339 owner.History.Add(p.Number);
340 faultHistory.Add(fault);
341 log.Add(new Event("ACCESS", pid, op, address));
342 bool interrupt = C.Tick > 0 && Time % C.Tick == 0;
343 if (interrupt) foreach (Page? page in frames)
344 if (page != null) {
345 page.Age = page.Age / 2 + 128 * page.R;
346 page.R = 0;
347 }
348 if (trace) {
349 Emit(
350 $"t={Time} P{pid} {op} {Hex(address)} page={p.Number} frame={p.Frame} phys={Hex(p.Frame * C.Page + address % C.Page)} {result}" +
351 (switched ? " switch" : ""));
352 if (interrupt)
353 Emit($"t={Time} tick");
354 }
355 }
356 void Exit(int pid) {
357 Process p = GetProcess(pid);
358 int count = 0, slots = 0;
359 foreach (Page page in p.Pages) {
360 if (page.Frame >= 0) {
361 frames[page.Frame] = null;
362 page.Frame = -1;
363 count++;
364 }
365 Invalidate(page);
366 if (page.Slot >= 0) {
367 swap.Remove(page.Slot);
368 page.Slot = -1;
369 slots++;
370 }
371 }
372 p.Exited = true;
373 log.Add(new Event("EXIT", pid));
374 Emit($"EXIT P{pid} freed={count} slots={slots}");
375 }
376 void Attributes(string cmd, int pid, int first, int last, string protection = "") {
377 Process p = GetProcess(pid);
378 if (first > last || last >= p.Pages.Length)
379 throw new CommandError("invalid page range");
380 for (int i = first; i <= last; i++)
381 if (cmd == "PROTECT")
382 p.Pages[i].Ro = protection == "RO";
383 else
384 p.Pages[i].Locked = cmd == "LOCK";
385 if (cmd != "PROTECT")
386 log.Add(new Event(cmd, pid, A: first, B: last));
387 }
388 void Clean() {
389 int count = 0;
390 foreach (Page? p in frames)
391 if (p != null && p.M == 1) {
392 Write(p);
393 count++;
394 }
395 log.Add(new Event("CLEAN"));
396 Emit("CLEAN written=" + count);
397 }
398 Simulation Rerun(string policy, int count) {
399 Config c = C.Copy();
400 c.Policy = policy;
401 c.Frames = count;
402 List<Access> references = log.Where(e => e.Kind == "ACCESS")
403 .Select(e => new Access(e.Pid, e.Op, e.A))
404 .ToList();
405 Simulation replay = new(c, references, true);
406 foreach (Process p in Processes.Values)
407 replay.Declare(p.Pid, p.Pages.Length);
408 foreach (Event e in log)
409 switch (e.Kind) {
410 case "ACCESS":
411 replay.Access(e.Pid, e.Op, e.A);
412 break;
413 case "EXIT":
414 replay.Exit(e.Pid);
415 break;
416 case "CLEAN":
417 replay.Clean();
418 break;
419 default:
420 replay.Attributes(e.Kind, e.Pid, e.A, e.B);
421 break;
422 }
423 return replay;
424 }
425 void Summary() {
426 Emit($"SUMMARY policy={C.Policy} scope={C.Scope} time={Time}");
427 Emit(
428 $"total faults={Faults} zero={Zero} swapin={Swapin} faultrate={Ratio(Faults, Time)} tlbhits={TlbHits} tlbrate={Ratio(TlbHits, Time)} writebacks={Writebacks} switches={Switches} segfaults={Processes.Values.Sum(p => p.Segfaults)} protfaults={Processes.Values.Sum(p => p.Protfaults)}");
429 foreach (Process p in Processes.Values)
430 Emit(
431 $"P{p.Pid} accesses={p.Accesses} faults={p.Faults} writebacks={p.Writebacks} segfaults={p.Segfaults} protfaults={p.Protfaults} resident={p.Resident} peak={p.Peak}" +
432 (C.Scope == "LOCAL" ? " quota=" + Quota(p.Pid) : "") +
433 (p.Exited ? " exited" : ""));
434 }
435 void FrameDump() {
436 Emit($"FRAMES used={frames.Count(p => p != null)}/{C.Frames}");
437 for (int i = 0; i < frames.Length; i++) {
438 Page? p = frames[i];
439 Emit(
440 p == null
441 ? $"{i} free"
442 : $"{i} {p.Label} R={p.R} M={p.M} load={p.Load} last={p.Last} uses={p.Uses} age={p.Age}" +
443 (p.Locked ? " locked" : ""));
444 }
445 if (C.Policy is "CLOCK" or "ECLOCK" or "WSCLOCK")
446 Emit(C.Scope == "GLOBAL"
447 ? "hand=" + hand
448 : "hand" +
449 string.Concat(Processes.Values.Select(p => $" P{p.Pid}={p.Hand}")));
450 }
451 void PageTable(int pid) {
452 Process p = GetProcess(pid);
453 Emit($"PAGETABLE P{pid} pages={p.Pages.Length} resident={p.Resident}");
454 foreach (Page page in p.Pages.Where(p => p.Ever)) {
455 string info = page.Frame >= 0 ? $"frame={page.Frame} R={page.R} M={page.M}"
456 : page.Slot >= 0 ? "swap=" + page.Slot
457 : "zero";
458 Emit($"{page.Number} {info}" + (page.Ro ? " RO" : "") +
459 (page.Locked ? " locked" : ""));
460 }
461 }
462 void Compare() {
463 Emit($"COMPARE accesses={Time} frames={C.Frames}");
464 Dictionary<string, Simulation> results = new();
465 foreach (string policy in Policies) {
466 Simulation r = Rerun(policy, C.Frames);
467 results[policy] = r;
468 Emit($"{policy} faults={r.Faults} writebacks={r.Writebacks} tlbhits={r.TlbHits}");
469 }
470 Emit("ranking " + string.Join(" ", Policies.OrderBy(p => results[p].Faults)
471 .ThenBy(p => results[p].Writebacks)));
472 }
473 public void Execute(string[] t) {
474 string cmd = t[0];
475 if (char.IsDigit(cmd[0])) {
476 Access(int.Parse(cmd), t[1], Address(t[2]));
477 return;
478 }
479 switch (cmd) {
480 case "PROCESS":
481 Declare(int.Parse(t[1]), int.Parse(t[2]));
482 break;
483 case "EXIT":
484 Exit(int.Parse(t[1]));
485 break;
486 case "PROTECT":
487 case "LOCK":
488 case "UNLOCK":
489 Attributes(cmd, int.Parse(t[1]), int.Parse(t[2]), int.Parse(t[3]),
490 t.Length > 4 ? t[4] : "");
491 break;
492 case "TRACE":
493 trace = t[1] == "ON";
494 break;
495 case "SUMMARY":
496 Summary();
497 break;
498 case "FRAMES":
499 FrameDump();
500 break;
501 case "PAGETABLE":
502 PageTable(int.Parse(t[1]));
503 break;
504 case "TLB":
505 Emit($"TLB used={tlb.Sum(s => s.Count)}/{C.Tlb} sets={C.Sets}");
506 for (int i = 0; i < tlb.Length; i++)
507 Emit($"set {i}:" + string.Concat(tlb[i].AsEnumerable().Reverse().Select(
508 p => $" {p.Label}->{p.Frame}")));
509 break;
510 case "SWAP":
511 Emit("SWAP used=" + swap.Count);
512 foreach (KeyValuePair<int, Page> entry in swap)
513 Emit(entry.Key + " " + entry.Value.Label);
514 break;
515 case "CLEAN":
516 Clean();
517 break;
518 case "COMPARE":
519 Compare();
520 break;
521 case "CURVE": {
522 string policy = t[1];
523 int lo = int.Parse(t[2]), hi = int.Parse(t[3]);
524 if (!Policies.Contains(policy))
525 throw new CommandError("unknown policy");
526 if (lo < 1 || lo > hi || hi > 16 || C.Scope == "LOCAL" && lo < Processes.Count)
527 throw new CommandError("invalid frame range");
528 Emit($"CURVE {policy} frames={lo}..{hi}");
529 int previousFaults = int.MaxValue;
530 for (int f = lo; f <= hi; f++) {
531 int value = Rerun(policy, f).Faults;
532 Emit($"F={f} faults={value}" + (value > previousFaults ? " ANOMALY" : ""));
533 previousFaults = value;
534 }
535 break;
536 }
537 case "EAT": {
538 long total = long.Parse(t[1]) * Time + long.Parse(t[2]) * (2L * Time - TlbHits) +
539 long.Parse(t[3]) * (Swapin + Writebacks);
540 Emit(
541 Time == 0
542 ? "EAT accesses=0"
543 : $"EAT accesses={Time} total={total} average={Decimal(total * 100 / Time)}");
544 break;
545 }
546 case "TIMELINE": {
547 int w = int.Parse(t[1]);
548 if (w < 1)
549 throw new CommandError("invalid window");
550 string text = "TIMELINE window=" + w;
551 for (int i = 0; i < faultHistory.Count; i += w)
552 text += " " + faultHistory.Skip(i).Take(w).Count(x => x);
553 Emit(text);
554 break;
555 }
556 case "WSET": {
557 int pid = int.Parse(t[1]);
558 Process p = GetProcess(pid);
559 int w = int.Parse(t[2]);
560 if (w < 1)
561 throw new CommandError("invalid window");
562 int[] pages = p.History.TakeLast(w).Distinct().OrderBy(x => x).ToArray();
563 Emit($"WSET P{pid} window={w} size={pages.Length}" +
564 string.Concat(pages.Select(x => " " + x)));
565 break;
566 }
567 case "HOT": {
568 int pid = int.Parse(t[1]);
569 Process p = GetProcess(pid);
570 int k = int.Parse(t[2]);
571 if (k < 1)
572 throw new CommandError("invalid count");
573 Emit($"HOT P{pid}" +
574 string.Concat(p.Pages.Where(p => p.References > 0)
575 .OrderByDescending(p => p.References)
576 .ThenBy(p => p.Number)
577 .Take(k)
578 .Select(p => $" {p.Number}:{p.References}")));
579 break;
580 }
581 }
582 }
583 }
584 static void Main() {
585 CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
586 int n = int.Parse(Console.ReadLine()!);
587 List<string[]> commands = new();
588 for (int i = 0; i < n; i++) {
589 string[] t = Console.ReadLine()!.Split(' ');
590 if (t[0] == "LOOP") {
591 int start = Address(t[3]), interval = int.Parse(t[4]), length = int.Parse(t[5]),
592 repeats = int.Parse(t[6]);
593 for (int r = 0; r < repeats; r++)
594 for (int j = 0; j < length; j++)
595 commands.Add(new[] { t[1], t[2], Hex(start + j * interval) });
596 } else if (t[0] == "RAND") {
597 int lo = Address(t[2]), hi = Address(t[3]), count = int.Parse(t[4]),
598 rate = int.Parse(t[6]);
599 long x = long.Parse(t[5]);
600 for (int j = 0; j < count; j++) {
601 x = x * 48271 % 2147483647;
602 int address = lo + (int)(x % (hi - lo));
603 x = x * 48271 % 2147483647;
604 commands.Add(new[] { t[1], x % 100 < rate ? "W" : "R", Hex(address) });
605 }
606 } else
607 commands.Add(t);
608 }
609 Simulation? simulation = null;
610 for (int i = 0; i < commands.Count; i++) {
611 string[] t = commands[i];
612 try {
613 if (t[0] == "SYSTEM") {
614 simulation = null;
615 Config c = Settings(t);
616 List<Access> future = new();
617 for (int j = i + 1; j < commands.Count && commands[j][0] != "SYSTEM"; j++) {
618 string[] line = commands[j];
619 if (char.IsDigit(line[0][0]))
620 future.Add(new Access(int.Parse(line[0]), line[1], Address(line[2])));
621 }
622 simulation = new Simulation(c, future);
623 Console.WriteLine(c);
624 } else if (simulation == null)
625 Console.WriteLine("ERROR no system");
626 else
627 simulation.Execute(t);
628 } catch (CommandError e) {
629 Console.WriteLine("ERROR " + e.Message);
630 }
631 }
632 }
