const policies = [
    "FIFO",
    "LRU",
    "LFU",
    "NRU",
    "AGING",
    "CLOCK",
    "ECLOCK",
    "WSCLOCK",
    "RANDOM",
    "OPT",
];
const defaults = {
    page: 256,
    frames: 4,
    tlb: 4,
    sets: 1,
    policy: "FIFO",
    scope: "GLOBAL",
    asid: "OFF",
    tick: 0,
    tau: 8,
    seed: 1,
};
const keys = Object.keys(defaults);
function parseSettings(fields) {
    const settings = { ...defaults };
    const seen = new Set();
    for (const pair of fields) {
        const [key, value] = pair.split("=");
        if (!keys.includes(key)) throw new Error(`unknown key ${key}`);
        if (seen.has(key)) throw new Error(`duplicate key ${key}`);
        seen.add(key);
        const n = Number(value);
        let valid = false;
        switch (key) {
            case "page":
                valid = n >= 16 && n <= 4096 && (n & (n - 1)) === 0;
                break;
            case "frames":
                valid = n >= 1 && n <= 16;
                break;
            case "tlb":
            case "sets":
                valid = n >= 1 && n <= 8;
                break;
            case "policy":
                valid = policies.includes(value);
                break;
            case "scope":
                valid = value === "GLOBAL" || value === "LOCAL";
                break;
            case "asid":
                valid = value === "ON" || value === "OFF";
                break;
            case "tick":
                valid = n >= 0 && n <= 1000;
                break;
            case "tau":
                valid = n >= 1 && n <= 1000;
                break;
            case "seed":
                valid = n >= 1 && n <= 2147483646;
                break;
        }
        if (!valid) throw new Error(`invalid ${key}`);
        switch (key) {
            case "policy":
                settings.policy = value;
                break;
            case "scope":
                settings.scope = value;
                break;
            case "asid":
                settings.asid = value;
                break;
            default:
                settings[key] = n;
        }
    }
    if (settings.tlb % settings.sets !== 0) throw new Error("invalid sets");
    return settings;
}
