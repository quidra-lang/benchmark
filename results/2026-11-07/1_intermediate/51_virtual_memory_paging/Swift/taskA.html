4func hex(_ value: Int) -> String {
5 "0x" + String(value, radix: 16, uppercase: true)
6}
7func address(_ text: String) -> Int {
8 Int(text.dropFirst(2), radix: 16)!
9}
10func decimal(_ value: Int) -> String {
11 "\(value / 100)." + (value % 100 < 10 ? "0" : "") + "\(value % 100)"
12}
13func ratio(_ x: Int, _ y: Int) -> String {
14 decimal(y == 0 ? 0 : 10000 * x / y) + "%"

16enum CommandError: Error { case message(String) }
17struct Config {
18 var page = 256, frames = 4, tlb = 4, sets = 1, tick = 0, tau = 8, seed = 1
19 var policy = "FIFO", scope = "GLOBAL", asid = "OFF"
20 var description: String {
21 "SYSTEM page=\(page) frames=\(frames) tlb=\(tlb) sets=\(sets) policy=\(policy) scope=\(scope) asid=\(asid) tick=\(tick) tau=\(tau) seed=\(seed)"
22 }
23}
24func settings(_ tokens: [String]) throws -> Config {
25 var c = Config()
26 let keys = ["page", "frames", "tlb", "sets", "policy", "scope", "asid", "tick", "tau", "seed"]
27 var seen = Set<String>()
28 for token in tokens.dropFirst() {
29 let pair = token.split(separator: "=").map(String.init)
30 let key = pair[0], value = pair[1]
31 guard keys.contains(key) else {
32 throw CommandError.message("unknown key " + key)
33 }
34 guard seen.insert(key).inserted else {
35 throw CommandError.message("duplicate key " + key)
36 }
37 var valid: Bool
38 switch key {
39 case "policy":
40 valid = policies.contains(value)
41 if valid {
42 c.policy = value
43 }
44 case "scope":
45 valid = ["GLOBAL", "LOCAL"].contains(value)
46 if valid {
47 c.scope = value
48 }
49 case "asid":
50 valid = ["ON", "OFF"].contains(value)
51 if valid {
52 c.asid = value
53 }
54 default:
55 let n = Int(value)!
56 switch key {
57 case "page":
58 valid = n >= 16 && n <= 4096 && (n & (n - 1)) == 0
59 if valid {
60 c.page = n
61 }
62 case "frames":
63 valid = n >= 1 && n <= 16
64 if valid {
65 c.frames = n
66 }
67 case "tlb":
68 valid = n >= 1 && n <= 8
69 if valid {
70 c.tlb = n
71 }
72 case "sets":
73 valid = n >= 1 && n <= 8
74 if valid {
75 c.sets = n
76 }
77 case "tick":
78 valid = n >= 0 && n <= 1000
79 if valid {
80 c.tick = n
81 }
82 case "tau":
83 valid = n >= 1 && n <= 1000
84 if valid {
85 c.tau = n
86 }
87 default:
88 valid = n >= 1 && n <= 2147483646
89 if valid {
90 c.seed = n
91 }
92 }
93 }
94 guard valid else {
95 throw CommandError.message("invalid " + key)
96 }
97 }
98 guard c.tlb % c.sets == 0 else {
99 throw CommandError.message("invalid sets")
100 }
101 return c
102}

161 func process(_ pid: Int) throws -> VMProcess {
162 guard let p = processes[pid] else {
163 throw CommandError.message("unknown process \(pid)")
164 }
165 guard !p.exited else {
166 throw CommandError.message("process \(pid) exited")
167 }
168 return p
169 }
170 func quota(_ pid: Int) -> Int {
171 c.frames / processes.count + (pids.firstIndex(of: pid)! < c.frames % processes.count ? 1 : 0)
172 }
173 func declare(_ pid: Int, _ count: Int) throws {
174 guard !fixed else {
175 throw CommandError.message("processes are fixed")
176 }
177 guard processes[pid] == nil else {
178 throw CommandError.message("duplicate process \(pid)")
179 }
180 guard count >= 1 && count <= 256 else {
181 throw CommandError.message("invalid page count")
182 }
183 guard processes.count < 8 && !(c.scope == "LOCAL" && processes.count == c.frames)
184 else {
185 throw CommandError.message("too many processes")
186 }
187 processes[pid] = VMProcess(pid, count)
188 }
189 func invalidate(_ page: Page) {
190 for i in tlb.indices {
191 tlb[i].removeAll { $0 === page }
192 }
193 }
194 func write(_ p: Page) {
195 writebacks += 1
196 processes[p.pid]!.writebacks += 1
197 p.m = 0
198 if p.slot < 0 {
199 var slot = 0
200 while swap[slot] != nil {
201 slot += 1
202 }
203 p.slot = slot
204 swap[slot] = p
205 }
206 }
207 func nextUse(_ p: Page) -> Int {
208 for i in (cursor + 1)..<future.count {
209 let a = future[i]
210 if a.pid == p.pid && a.address / c.page == p.number {
211 return i
212 }
213 }
214 return Int.max
215 }
216 func choose(_ input: [Int], _ owner: VMProcess) -> Int {
217 let unlocked = input.filter { !frames[$0]!.locked }
218 var candidates = unlocked.isEmpty ? input : unlocked
219 if ["FIFO", "LRU", "LFU", "NRU", "AGING", "OPT"].contains(c.policy) {
220 candidates.sort { a, b in
221 let x = frames[a]!, y = frames[b]!
222 switch c.policy {
223 case "FIFO":
224 return x.load < y.load
225 case "LRU":
226 return x.last < y.last
227 case "LFU":
228 return x.uses != y.uses ? x.uses < y.uses : x.load < y.load
229 case "NRU":
230 return 2 * x.r + x.m != 2 * y.r + y.m ? 2 * x.r + x.m < 2 * y.r + y.m : a < b
231 case "AGING":
232 return x.age != y.age ? x.age < y.age : x.load < y.load
233 default:
234 let nx = nextUse(x), ny = nextUse(y)
235 return nx != ny ? nx > ny : a < b
236 }
237 }
238 return candidates[0]
239 }
240 if c.policy == "RANDOM" {
241 random = random * 48271 % 2147483647
242 return candidates[random % candidates.count]
243 }
244 let h = c.scope == "GLOBAL" ? hand : owner.hand
245 candidates.sort { ($0 - h + c.frames) % c.frames < ($1 - h + c.frames) % c.frames }
246 var chosen: Int? = nil
247 if c.policy == "CLOCK" {
248 while chosen == nil {
249 for f in candidates {
250 if frames[f]!.r == 0 {
251 chosen = f
252 break
253 }
254 frames[f]!.r = 0
255 }
256 }
257 } else if c.policy == "ECLOCK" {
258 while chosen == nil {
259 for f in candidates where frames[f]!.r == 0 && frames[f]!.m == 0 {
260 chosen = f
261 break
262 }
263 if chosen != nil {
264 break
265 }
266 for f in candidates {
267 if frames[f]!.r == 0 && frames[f]!.m == 1 {
268 chosen = f
269 break
270 }
271 frames[f]!.r = 0
272 }
273 }
274 } else {
275 for f in candidates {
276 let p = frames[f]!
277 if p.r == 1 {
278 p.r = 0
279 p.last = time
280 } else if time - p.last > c.tau {
281 if p.m == 0 {
282 chosen = f
283 break
284 }
285 write(p)
286 }
287 }
288 if chosen == nil {
289 chosen = candidates.first { frames[$0]!.m == 0 } ?? candidates[0]
290 }
291 }
292 let result = chosen!
293 if c.scope == "GLOBAL" {
294 hand = (result + 1) % c.frames
295 } else {
296 owner.hand = (result + 1) % c.frames
297 }
298 return result
299 }
300 func access(_ a: Access) throws {
301 fixed = true
302 cursor += 1
303 let owner = try process(a.pid)
304 if a.address >= owner.pages.count * c.page {
305 owner.segfaults += 1
306 emit("SEGFAULT P\(a.pid) \(hex(a.address))")
307 return
308 }
309 let p = owner.pages[a.address / c.page]
310 if a.op == "W" && p.ro {
311 owner.protfaults += 1
312 emit("PROTFAULT P\(a.pid) \(hex(a.address))")
313 return
314 }
315 time += 1
316 owner.accesses += 1
317 let switched = previous >= 0 && previous != a.pid
318 if switched {
319 switches += 1
320 if c.asid == "OFF" {
321 for i in tlb.indices {
322 tlb[i].removeAll()
323 }
324 }
325 }
326 previous = a.pid
327 let set = p.number % c.sets
328 let hitIndex = tlb[set].firstIndex { $0 === p }
329 var fault = false, result: String
330 if let index = hitIndex {
331 tlb[set].remove(at: index)
332 tlbhits += 1
333 result = "TLBHIT"
334 } else if p.frame >= 0 {
335 result = "HIT"
336 } else {
337 fault = true
338 faults += 1
339 owner.faults += 1
340 if p.slot >= 0 {
341 swapin += 1
342 result = "FAULT SWAPIN"
343 } else {
344 zero += 1
345 result = "FAULT ZERO"
346 }
347 var frame: Int? = nil
348 if c.scope == "GLOBAL" || owner.resident < quota(a.pid) {
349 frame = frames.firstIndex { $0 == nil }
350 }
351 if frame == nil {
352 let candidates = frames.indices.filter { f in
353 guard let page = frames[f] else {
354 return false
355 }
356 return c.scope == "GLOBAL" || page.pid == a.pid
357 }
358 frame = choose(candidates, owner)
359 let victim = frames[frame!]!
360 result += " evict=" + victim.label
361 invalidate(victim)
362 if victim.m == 1 {
363 write(victim)
364 result += " writeback"
365 }
366 victim.frame = -1
367 }
368 let f = frame!
369 frames[f] = p
370 p.frame = f
371 p.load = time
372 p.uses = 0
373 p.m = 0
374 p.age = 0
375 owner.peak = max(owner.peak, owner.resident)
376 }
377 if tlb[set].count == c.tlb / c.sets {
378 tlb[set].removeFirst()
379 }
380 tlb[set].append(p)
381 p.r = 1
382 if a.op == "W" {
383 p.m = 1
384 }
385 p.last = time
386 p.uses += 1
387 p.ever = true
388 p.references += 1
389 owner.history.append(p.number)
390 faultHistory.append(fault)
391 log.append(.access(a))
392 let interrupt = c.tick > 0 && time % c.tick == 0
393 if interrupt {
394 for case let page? in frames {
395 page.age = page.age / 2 + 128 * page.r
396 page.r = 0
397 }
398 }
399 if trace {
400 emit("t=\(time) P\(a.pid) \(a.op) \(hex(a.address)) page=\(p.number) frame=\(p.frame) phys=\(hex(p.frame * c.page + a.address % c.page)) \(result)" +
401 (switched ? " switch" : ""))
402 if interrupt {
403 emit("t=\(time) tick")
404 }
405 }
406 }
407 func exit(_ pid: Int) throws {
408 let p = try process(pid)
409 var count = 0, slots = 0
410 for page in p.pages {
411 if page.frame >= 0 {
412 frames[page.frame] = nil
413 page.frame = -1
414 count += 1
415 }
416 invalidate(page)
417 if page.slot >= 0 {
418 swap.removeValue(forKey: page.slot)
419 page.slot = -1
420 slots += 1
421 }
422 }
423 p.exited = true
424 log.append(.exit(pid))
425 emit("EXIT P\(pid) freed=\(count) slots=\(slots)")
426 }
427 func attributes(_ cmd: String, _ pid: Int, _ first: Int, _ last: Int, _ protection: String = "") throws {
428 let p = try process(pid)
429 guard first <= last && last < p.pages.count else {
430 throw CommandError.message("invalid page range")
431 }
432 for i in first...last {
433 if cmd == "PROTECT" {
434 p.pages[i].ro = protection == "RO"
435 } else {
436 p.pages[i].locked = cmd == "LOCK"
437 }
438 }
439 if cmd != "PROTECT" {
440 log.append(.lock(pid, first, last, cmd == "LOCK"))
441 }
442 }
443 func clean() {
444 var count = 0
445 for case let p? in frames where p.m == 1 {
446 write(p)
447 count += 1
448 }
449 log.append(.clean)
450 emit("CLEAN written=\(count)")
451 }
452 func rerun(_ policy: String, _ count: Int) throws -> Simulation {
453 var config = c
454 config.policy = policy
455 config.frames = count
456 let references: [Access] = log.compactMap { event in
457 if case let .access(a) = event {
458 return a
459 }
460 return nil
461 }
462 let replay = Simulation(config, references, silent: true)
463 for pid in pids {
464 try replay.declare(pid, processes[pid]!.pages.count)
465 }
466 for event in log {
467 switch event {
468 case let .access(a):
469 try replay.access(a)
470 case let .exit(pid):
471 try replay.exit(pid)
472 case .clean:
473 replay.clean()
474 case let .lock(pid, first, last, locked):
475 try replay.attributes(
476 locked ? "LOCK" : "UNLOCK",
477 pid,
478 first,
479 last
480 )
481 }
482 }
483 return replay
484 }
485 func summary() {
486 emit("SUMMARY policy=\(c.policy) scope=\(c.scope) time=\(time)")
487 let seg = processes.values.reduce(0) { $0 + $1.segfaults },
488 prot = processes.values.reduce(0) { $0 + $1.protfaults }
489 emit(
490 "total faults=\(faults) zero=\(zero) swapin=\(swapin) faultrate=\(ratio(faults, time)) tlbhits=\(tlbhits) tlbrate=\(ratio(tlbhits, time)) writebacks=\(writebacks) switches=\(switches) segfaults=\(seg) protfaults=\(prot)"
491 )
492 for pid in pids {
493 let p = processes[pid]!
494 emit("P\(pid) accesses=\(p.accesses) faults=\(p.faults) writebacks=\(p.writebacks) segfaults=\(p.segfaults) protfaults=\(p.protfaults) resident=\(p.resident) peak=\(p.peak)" +
495 (c.scope == "LOCAL" ? " quota=\(quota(pid))" : "") + (p.exited ? " exited" : ""))
496 }
497 }
498 func frameDump() {
499 emit("FRAMES used=\(frames.compactMap { $0 }.count)/\(c.frames)")
500 for i in frames.indices {
501 if let p = frames[i] {
502 emit("\(i) \(p.label) R=\(p.r) M=\(p.m) load=\(p.load) last=\(p.last) uses=\(p.uses) age=\(p.age)" +
503 (p.locked ? " locked" : ""))
504 } else {
505 emit("\(i) free")
506 }
507 }
508 if ["CLOCK", "ECLOCK", "WSCLOCK"].contains(c.policy) {
509 emit(c.scope == "GLOBAL" ? "hand=\(hand)" : "hand" + pids.map { " P\($0)=\(processes[$0]!.hand)" }.joined())
510 }
511 }
512 func pageTable(_ pid: Int) throws {
513 let p = try process(pid)
514 emit("PAGETABLE P\(pid) pages=\(p.pages.count) resident=\(p.resident)")
515 for page in p.pages where page.ever {
516 let info = page.frame >= 0 ? "frame=\(page.frame) R=\(page.r) M=\(page.m)" : page
517 .slot >= 0 ? "swap=\(page.slot)" : "zero"
518 emit("\(page.number) \(info)" + (page.ro ? " RO" : "") + (page.locked ? " locked" : ""))
519 }
520 }
521 func compare() throws {
522 emit("COMPARE accesses=\(time) frames=\(c.frames)")
523 var results: [String: Simulation] = [:]
524 for policy in policies {
525 let r = try rerun(policy, c.frames)
526 results[policy] = r
527 emit("\(policy) faults=\(r.faults) writebacks=\(r.writebacks) tlbhits=\(r.tlbhits)")
528 }
529 let ranking = policies.sorted { a, b in
530 let x = results[a]!, y = results[b]!
531 if x.faults != y.faults {
532 return x.faults < y.faults
533 }
534 if x.writebacks != y.writebacks {
535 return x.writebacks < y.writebacks
536 }
537 return policies.firstIndex(of: a)! < policies.firstIndex(of: b)!
538 }
539 emit("ranking " + ranking.joined(separator: " "))
540 }
541 func execute(_ t: [String]) throws {
542 let cmd = t[0]
543 if cmd.first!.isNumber {
544 try access(Access(pid: Int(cmd)!, op: t[1], address: address(t[2])))
545 return
546 }
547 switch cmd {
548 case "PROCESS":
549 try declare(Int(t[1])!, Int(t[2])!)
550 case "EXIT":
551 try exit(Int(t[1])!)
552 case "PROTECT", "LOCK", "UNLOCK":
553 try attributes(
554 cmd,
555 Int(t[1])!,
556 Int(t[2])!,
557 Int(t[3])!,
558 t.count > 4 ? t[4] : ""
559 )
560 case "TRACE":
561 trace = t[1] == "ON"
562 case "SUMMARY":
563 summary()
564 case "FRAMES":
565 frameDump()
566 case "PAGETABLE":
567 try pageTable(Int(t[1])!)
568 case "TLB":
569 emit("TLB used=\(tlb.reduce(0) { $0 + $1.count })/\(c.tlb) sets=\(c.sets)")
570 for i in tlb.indices {
571 emit("set \(i):" + tlb[i].reversed().map { " \($0.label)->\($0.frame)" }.joined())
572 }
573 case "SWAP":
574 emit("SWAP used=\(swap.count)")
575 for slot in swap.keys.sorted() {
576 emit("\(slot) \(swap[slot]!.label)")
577 }
578 case "CLEAN":
579 clean()
580 case "COMPARE":
581 try compare()
582 case "CURVE":
583 let policy = t[1], lo = Int(t[2])!, hi = Int(t[3])!
584 guard policies.contains(policy) else {
585 throw CommandError.message("unknown policy")
586 }
587 guard lo >= 1 && lo <= hi && hi <= 16 && !(c.scope == "LOCAL" && lo < processes.count)
588 else {
589 throw CommandError.message("invalid frame range")
590 }
591 emit("CURVE \(policy) frames=\(lo)..\(hi)")
592 var previousFaults = Int.max
593 for f in lo...hi {
594 let value = try rerun(policy, f).faults
595 emit("F=\(f) faults=\(value)" + (value > previousFaults ? " ANOMALY" : ""))
596 previousFaults = value
597 }
598 case "EAT":
599 let total = Int(t[1])! * time + Int(t[2])! * (2 * time - tlbhits) + Int(t[3])! * (swapin + writebacks)
600 emit(time == 0 ? "EAT accesses=0" :
601 "EAT accesses=\(time) total=\(total) average=\(decimal(100 * total / time))")
602 case "TIMELINE":
603 let w = Int(t[1])!
604 guard w >= 1 else {
605 throw CommandError.message("invalid window")
606 }
607 var text = "TIMELINE window=\(w)"
608 for i in stride(from: 0, to: faultHistory.count, by: w) {
609 text += " \(faultHistory[i..<min(i + w, faultHistory.count)].filter { $0 }.count)"
610 }
611 emit(text)
612 case "WSET":
613 let pid = Int(t[1])!, p = try process(pid), w = Int(t[2])!
614 guard w >= 1 else {
615 throw CommandError.message("invalid window")
616 }
617 let pages = Set(p.history.suffix(w)).sorted()
618 emit("WSET P\(pid) window=\(w) size=\(pages.count)" + pages.map { " \($0)" }.joined())
619 case "HOT":
620 let pid = Int(t[1])!, p = try process(pid), k = Int(t[2])!
621 guard k >= 1 else {
622 throw CommandError.message("invalid count")
623 }
624 let pages = p.pages.filter { $0.references > 0 }.sorted { a, b in
625 a.references != b.references ? a.references > b.references : a.number < b.number
626 }
627 emit("HOT P\(pid)" + pages.prefix(k).map { " \($0.number):\($0.references)" }.joined())
628 default:
629 break
630 }
631 }
632}
633let n = Int(readLine()!)!
634var commands: [[String]] = []
635for _ in 0..<n {
636 let t = readLine()!.split(separator: " ").map(String.init)
637 if t[0] == "LOOP" {
638 let start = address(t[3]), interval = Int(t[4])!, length = Int(t[5])!, repeats = Int(t[6])!
639 for _ in 0..<repeats {
640 for j in 0..<length {
641 commands.append([t[1], t[2], hex(start + j * interval)])
642 }
643 }
644 } else if t[0] == "RAND" {
645 let lo = address(t[2]), hi = address(t[3]), count = Int(t[4])!, rate = Int(t[6])!
646 var x = Int(t[5])!
647 for _ in 0..<count {
648 x = x * 48271 % 2147483647
649 let a = lo + x % (hi - lo)
650 x = x * 48271 % 2147483647
651 commands.append([t[1], x % 100 < rate ? "W" : "R", hex(a)])
652 }
653 } else {
654 commands.append(t)
655 }
656}
657var simulation: Simulation? = nil
658for i in commands.indices {
659 let t = commands[i]
660 do {
661 if t[0] == "SYSTEM" {
662 simulation = nil
663 let c = try settings(t)
664 var future: [Access] = []
665 for j in (i + 1)..<commands.count {
666 let line = commands[j]
667 if line[0] == "SYSTEM" {
668 break
669 }
670 if line[0].first!.isNumber {
671 future.append(Access(
672 pid: Int(line[0])!,
673 op: line[1],
674 address: address(line[2])
675 ))
676 }
677 }
678 simulation = Simulation(c, future)
679 print(c.description)
680 } else if let current = simulation {
681 try current.execute(t)
682 } else {
683 print("ERROR no system")
684 }
685 } catch {
686 if case CommandError.message(let message) = error {
687 print("ERROR " + message)
688 }
689 }
690}
