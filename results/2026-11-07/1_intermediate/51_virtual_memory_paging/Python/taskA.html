55def loop_accesses(arguments):
56 pid, op = int(arguments[0]), arguments[1]
57 start = parse_address(arguments[2])
58 interval, length, repeats = (int(value) for value in arguments[3:])
59 return [
60 Access(pid, op, start + index * interval)
61 for _ in range(repeats)
62 for index in range(length)
63 ]
64
65
66def random_accesses(arguments):
67 pid = int(arguments[0])
68 low, high = parse_address(arguments[1]), parse_address(arguments[2])
69 count, state, write_rate = (int(value) for value in arguments[3:])
70 accesses = []
71 for _ in range(count):
72 state = next_random(state)
73 address = low + state % (high - low)
74 state = next_random(state)
75 op = "W" if state % 100 < write_rate else "R"
76 accesses.append(Access(pid, op, address))
77 return accesses

80def expand(lines):
81 items = []
82 for line in lines:
83 words = line.split()
84 if words[0][0].isdigit():
85 items.append(Access(int(words[0]), words[1], parse_address(words[2])))
86 elif words[0] == "LOOP":
87 items.extend(loop_accesses(words[1:]))
88 elif words[0] == "RAND":
89 items.extend(random_accesses(words[1:]))
90 else:
91 items.append(words)
92 return items

96 accesses = []

132def setting_value(key, text):
133 if key in WORD_CHOICES:
134 value = text
135 valid = value in WORD_CHOICES[key]
136 elif key == "page":
137 value = int(text)
138 valid = value in PAGE_SIZES
139 else:
140 value = int(text)
141 low, high = NUMBER_LIMITS[key]
142 valid = low <= value <= high
143 if not valid:
144 raise CommandError(f"invalid {key}")
145 return value

148def parse_settings(pairs):
149 values = {}
150 for pair in pairs:
151 key, text = pair.split("=")
152 if key not in DEFAULT_SETTINGS:
153 raise CommandError(f"unknown key {key}")
154 if key in values:
155 raise CommandError(f"duplicate key {key}")
156 values[key] = setting_value(key, text)
157 for key, value in DEFAULT_SETTINGS.items():
158 values.setdefault(key, value)
159 if values["tlb"] % values["sets"] != 0:
160 raise CommandError("invalid sets")
161 return Settings(values)

200class Tlb:
201 def __init__(self, entries, set_count):
202 self.capacity = entries // set_count
203 self.sets = [[] for _ in range(set_count)]
204
205 def set_of(self, page):
206 return self.sets[page.number % len(self.sets)]
207
208 def lookup(self, page):
209 entries = self.set_of(page)
210 if page not in entries:
211 return False
212 entries.remove(page)
213 entries.insert(0, page)
214 return True
215
216 def insert(self, page):
217 entries = self.set_of(page)
218 if len(entries) == self.capacity:
219 entries.pop()
220 entries.insert(0, page)
221
222 def remove(self, page):
223 entries = self.set_of(page)
224 if page in entries:
225 entries.remove(page)
226
227 def remove_process(self, pid):
228 for index, entries in enumerate(self.sets):
229 self.sets[index] = [page for page in entries if page.pid != pid]
230
231 def clear(self):
232 for entries in self.sets:
233 entries.clear()
234
235 def used(self):
236 return sum(len(entries) for entries in self.sets)

284 def write_back(self, page):
285 self.processes[page.pid].writebacks += 1
286 self.writebacks += 1
287 page.modified = 0

289 page.slot = self.free_slot()

292 def next_use(self, page):
293 key = (page.pid, page.number)
294 for index in range(self.position + 1, len(self.references)):
295 if self.references[index] == key:
296 return index
297 return len(self.references)

299 def clock(self, order):
300 for frame in order:
301 page = self.frames[frame]
302 if page.referenced == 0:
303 return frame
304 page.referenced = 0
305 return order[0]

307 def enhanced_clock(self, order):
308 while True:
309 for frame in order:
310 page = self.frames[frame]
311 if page.referenced == 0 and page.modified == 0:
312 return frame
313 for frame in order:
314 page = self.frames[frame]
315 if page.referenced == 0 and page.modified == 1:
316 return frame
317 page.referenced = 0
318
319 def wsclock(self, order):
320 for frame in order:
321 page = self.frames[frame]
322 if page.referenced == 1:
323 page.referenced = 0
324 page.last = self.time
325 elif self.time - page.last > self.settings.tau:
326 if page.modified == 0:
327 return frame
328 self.write_back(page)
329 for frame in order:
330 if self.frames[frame].modified == 0:
331 return frame
332 return order[0]

334 def select_clock(self, process, candidates):
335 local = self.settings.scope == "LOCAL"
336 hand = process.hand if local else self.hand
337 count = len(self.frames)
338 order = sorted(candidates, key=lambda frame: (frame - hand + count) % count)
339 if self.settings.policy == "CLOCK":
340 chosen = self.clock(order)
341 elif self.settings.policy == "ECLOCK":
342 chosen = self.enhanced_clock(order)
343 else:
344 chosen = self.wsclock(order)
345 if local:
346 process.hand = (chosen + 1) % count
347 else:
348 self.hand = (chosen + 1) % count
349 return chosen

351 def select(self, process, candidates):
352 policy = self.settings.policy
353 frames = self.frames
354 if policy == "FIFO":
355 return min(candidates, key=lambda frame: frames[frame].load)
356 if policy == "LRU":
357 return min(candidates, key=lambda frame: frames[frame].last)
358 if policy == "LFU":
359 return min(candidates, key=lambda frame: (frames[frame].uses, frames[frame].load))
360 if policy == "NRU":
361 return min(
362 candidates,
363 key=lambda frame: (2 * frames[frame].referenced + frames[frame].modified, frame),
364 )
365 if policy == "AGING":
366 return min(candidates, key=lambda frame: (frames[frame].age, frames[frame].load))
367 if policy == "RANDOM":
368 self.random_state = next_random(self.random_state)
369 return candidates[self.random_state % len(candidates)]
370 if policy == "OPT":
371 return min(candidates, key=lambda frame: (-self.next_use(frames[frame]), frame))
372 return self.select_clock(process, candidates)

374 def target_frame(self, process):
375 if self.settings.scope == "GLOBAL":
376 frame = self.free_frame()
377 if frame is not None:
378 return frame
379 candidates = list(range(len(self.frames)))
380 else:
381 if process.resident < self.quota(process):
382 return self.free_frame()
383 candidates = [
384 frame for frame, page in enumerate(self.frames)
385 if page is not None and page.pid == process.pid
386 ]
387 unlocked = [frame for frame in candidates if not self.frames[frame].locked]
388 if unlocked:
389 candidates = unlocked
390 return self.select(process, candidates)

392 def evict(self, frame):
393 victim = self.frames[frame]
394 self.frames[frame] = None
395 victim.frame = None
396 self.processes[victim.pid].resident -= 1
397 self.tlb.remove(victim)
398 if victim.modified == 0:
399 return False
400 self.write_back(victim)
401 return True

403 def fault(self, process, page):
404 process.faults += 1
405 self.faults += 1
406 if page.slot is None:
407 self.zeros += 1
408 result = "FAULT ZERO"
409 else:
410 self.swapins += 1
411 result = "FAULT SWAPIN"
412 frame = self.target_frame(process)
413 victim = self.frames[frame]
414 if victim is not None:
415 result += f" evict={victim.label()}"
416 if self.evict(frame):
417 result += " writeback"
418 self.frames[frame] = page
419 page.frame = frame
420 page.load = self.time
421 page.uses = 0
422 page.modified = 0
423 page.age = 0
424 process.resident += 1
425 process.peak = max(process.peak, process.resident)
426 return result

428 def timer_interrupt(self):
429 for page in self.frames:
430 if page is not None:
431 page.age = page.age // 2 + 128 * page.referenced
432 page.referenced = 0

434 def access(self, pid, op, address, position):
435 self.position = position
436 self.time += 1
437 process = self.processes[pid]
438 switched = self.last_pid is not None and self.last_pid != pid
439 if switched:
440 self.switches += 1
441 if self.settings.asid == "OFF":
442 self.tlb.clear()
443 self.last_pid = pid
444 number, offset = divmod(address, self.settings.page)
445 page = process.pages[number]
446 faulted = False
447 if self.tlb.lookup(page):
448 self.tlbhits += 1
449 result = "TLBHIT"
450 else:
451 if page.frame is not None:
452 result = "HIT"
453 else:
454 result = self.fault(process, page)
455 faulted = True
456 self.tlb.insert(page)
457 page.referenced = 1
458 if op == "W":
459 page.modified = 1
460 page.last = self.time
461 page.uses += 1
462 page.accessed = True
463 process.history.append(number)
464 self.fault_history.append(faulted)
465 physical = page.frame * self.settings.page + offset
466 line = (
467 f"t={self.time} P{pid} {op} {hex_text(address)} page={number} "
468 f"frame={page.frame} phys={hex_text(physical)} {result}"
469 )
470 if switched:
471 line += " switch"
472 lines = [line]
473 if self.settings.tick > 0 and self.time % self.settings.tick == 0:
474 self.timer_interrupt()
475 lines.append(f"t={self.time} tick")
476 return lines

478 def exit_process(self, pid):
479 freed = 0
480 for frame, page in enumerate(self.frames):
481 if page is not None and page.pid == pid:
482 self.frames[frame] = None
483 page.frame = None
484 freed += 1
485 self.tlb.remove_process(pid)
486 released = 0
487 for slot, page in enumerate(self.slots):
488 if page is not None and page.pid == pid:
489 self.slots[slot] = None
490 page.slot = None
491 released += 1
492 process = self.processes[pid]
493 process.resident = 0
494 process.exited = True
495 return freed, released

497 def clean(self):
498 written = 0
499 for page in self.frames:
500 if page is not None and page.modified == 1:
501 self.write_back(page)
502 written += 1
503 return written

505 def set_lock(self, pid, first, last, locked):
506 for page in self.processes[pid].pages[first:last + 1]:
507 page.locked = locked

543 def access_line(self, access):
544 self.require_system()
545 position = self.position
546 self.position += 1
547 process = self.require_process(access.pid)
548 page_size = self.simulation.settings.page
549 if access.address >= process.page_count * page_size:
550 process.segfaults += 1
551 self.emit("SEGFAULT", f"P{access.pid}", hex_text(access.address))
552 return
553 if access.op == "W" and process.pages[access.address // page_size].readonly:
554 process.protfaults += 1
555 self.emit("PROTFAULT", f"P{access.pid}", hex_text(access.address))
556 return
557 lines = self.simulation.access(access.pid, access.op, access.address, position)
558 self.events.append(("ACCESS", access.pid, access.op, access.address))
559 if self.trace:
560 self.output.extend(lines)

562 def declare_process(self, arguments):
563 pid, page_count = int(arguments[0]), int(arguments[1])
564 simulation = self.simulation
565 if self.position > 0:
566 raise CommandError("processes are fixed")
567 if pid in simulation.processes:
568 raise CommandError(f"duplicate process {pid}")
569 if not 1 <= page_count <= MAX_PAGE_COUNT:
570 raise CommandError("invalid page count")
571 declared = len(simulation.processes)
572 local = simulation.settings.scope == "LOCAL"
573 if declared >= MAX_PROCESSES or (local and declared >= simulation.settings.frames):
574 raise CommandError("too many processes")
575 simulation.declare(pid, page_count)

706 def rerun(self, policy, frames):
707 current = self.simulation
708 settings = current.settings.variant(policy, frames)
709 accesses = [values for kind, *values in self.events if kind == "ACCESS"]
710 references = [(pid, address // settings.page) for pid, _, address in accesses]
711 simulation = Simulation(settings, references)
712 for process in current.ordered_processes():
713 simulation.declare(process.pid, process.page_count)
714 position = 0
715 for kind, *values in self.events:
716 if kind == "ACCESS":
717 pid, op, address = values
718 simulation.access(pid, op, address, position)
719 position += 1
720 elif kind == "EXIT":
721 simulation.exit_process(values[0])
722 elif kind == "CLEAN":
723 simulation.clean()
724 else:
725 pid, first, last, locked = values
726 simulation.set_lock(pid, first, last, locked)
727 return simulation

729 def compare(self, arguments):
730 frames = self.simulation.settings.frames
731 self.emit("COMPARE", f"accesses={self.simulation.time}", f"frames={frames}")
732 results = []
733 for index, policy in enumerate(POLICIES):
734 rerun = self.rerun(policy, frames)
735 results.append((rerun.faults, rerun.writebacks, index, policy))

739 )

743 def curve(self, arguments):
744 policy, low, high = arguments[0], int(arguments[1]), int(arguments[2])
745 if policy not in POLICIES:
746 raise CommandError("unknown policy")
747 simulation = self.simulation
748 local = simulation.settings.scope == "LOCAL"
749 if not 1 <= low <= high <= MAX_FRAMES or (local and low < len(simulation.processes)):
750 raise CommandError("invalid frame range")
751 self.emit("CURVE", policy, f"frames={low}..{high}")
752 previous = None
753 for frames in range(low, high + 1):
754 faults = self.rerun(policy, frames).faults
755 line = f"F={frames} faults={faults}"
756 if previous is not None and faults > previous:
757 line += " ANOMALY"
758 self.emit(line)
759 previous = faults

761 def effective_access_time(self, arguments):
762 lookup, memory, disk = (int(value) for value in arguments)
763 simulation = self.simulation
764 accesses = simulation.time
765 if accesses == 0:
766 self.emit("EAT accesses=0")
767 return
768 total = (
769 lookup * accesses
770 + memory * (2 * accesses - simulation.tlbhits)
771 + disk * (simulation.swapins + simulation.writebacks)
772 )
773 average = 100 * total // accesses
774 self.emit(
775 "EAT", f"accesses={accesses}", f"total={total}",
776 f"average={average // 100}.{average % 100:02d}",
777 )

779 def timeline(self, arguments):
780 window = int(arguments[0])
781 if window < 1:
782 raise CommandError("invalid window")
783 history = self.simulation.fault_history
784 counts = [sum(history[start:start + window]) for start in range(0, len(history), window)]
785 self.emit(f"TIMELINE window={window}", *counts)

787 def working_set(self, arguments):
788 process = self.require_process(int(arguments[0]))
789 window = int(arguments[1])
790 if window < 1:
791 raise CommandError("invalid window")
792 pages = sorted(set(process.history[-window:]))
793 self.emit("WSET", f"P{process.pid}", f"window={window}", f"size={len(pages)}", *pages)

795 def hot_pages(self, arguments):
796 process = self.require_process(int(arguments[0]))
797 count = int(arguments[1])
798 if count < 1:
799 raise CommandError("invalid count")
800 ranked = sorted(Counter(process.history).items(), key=lambda item: (-item[1], item[0]))
801 hot = [f"{number}:{total}" for number, total in ranked[:count]]
802 self.emit("HOT", f"P{process.pid}", *hot)

828 def run(self, items):
829 for index, item in enumerate(items):
830 try:
831 if isinstance(item, Access):
832 self.access_line(item)
833 elif item[0] == "SYSTEM":
834 self.start_system(item[1:], segment_accesses(items, index + 1))
835 else:
836 self.execute(item)
837 except CommandError as error:
838 self.emit("ERROR", error)
