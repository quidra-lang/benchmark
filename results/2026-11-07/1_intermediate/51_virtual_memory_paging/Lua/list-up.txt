local policies = {
    "FIFO",
    "LRU",
    "LFU",
    "NRU",
    "AGING",
    "CLOCK",
    "ECLOCK",
    "WSCLOCK",
    "RANDOM",
    "OPT"
}
local defaults = {
    page = 256,
    frames = 4,
    tlb = 4,
    sets = 1,
    policy = "FIFO",
    scope = "GLOBAL",
    asid = "OFF",
    tick = 0,
    tau = 8,
    seed = 1
}
local keys = {"page", "frames", "tlb", "sets", "policy", "scope", "asid", "tick", "tau", "seed"}
function Sim:new(config, refs)
    local s =
        setmetatable(
        {
            c = config,
            refs = refs,
            procs = {},
            frames = {},
            tlb = {},
            swap = {},
            events = {},
            history = {},
            time = 0,
            faults = 0,
            writebacks = 0,
            tlbhits = 0,
            zero = 0,
            swapin = 0,
            switches = 0,
            cursor = 0,
            hand = 0,
            random = config.seed,
            fixed = false,
            trace = false
        },
        self
    )
    for i = 1, config.sets do
        s.tlb[i] = {}
    end
    return s
end
local function settings(w)
    local c, seen = {}, {}
    for k, v in pairs(defaults) do
        c[k] = v
    end
    for i = 2, #w do
        local k, value = w[i]:match("^([^=]+)=(.*)$")
        if c[k] == nil then
            fail("unknown key " .. k)
        end
        if seen[k] then
            fail("duplicate key " .. k)
        end
        seen[k] = true
        local valid, n = false, tonumber(value)
        if k == "policy" then
            valid = contains(policies, value) ~= nil
        elseif k == "scope" then
            valid = value == "GLOBAL" or value == "LOCAL"
        elseif k == "asid" then
            valid = value == "ON" or value == "OFF"
        elseif k == "page" then
            valid = n >= 16 and n <= 4096 and (n & (n - 1)) == 0
        else
            local ranges = {
                frames = {1, 16},
                tlb = {1, 8},
                sets = {1, 8},
                tick = {0, 1000},
                tau = {1, 1000},
                seed = {1, 2147483646}
            }
            valid = n >= ranges[k][1] and n <= ranges[k][2]
        end
        if not valid then
            fail("invalid " .. k)
        end
        c[k] = n or value
    end
    if c.tlb % c.sets ~= 0 then
        fail("invalid sets")
    end
    return c
end
