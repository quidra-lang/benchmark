Config settings(List<String> tokens) {
  Config c = Config();
  Set<String> seen = {};
  for (String token in tokens.skip(1)) {
    List<String> pair = token.split('=');
    String key = pair[0], value = pair[1];
    if (![
      'page',
      'frames',
      'tlb',
      'sets',
      'policy',
      'scope',
      'asid',
      'tick',
      'tau',
      'seed',
    ].contains(key))
      throw CommandError('unknown key $key');
    if (!seen.add(key)) throw CommandError('duplicate key $key');
    bool valid;
    if (key == 'policy') {
      valid = policies.contains(value);
      if (valid) c.policy = value;
    } else if (key == 'scope') {
      valid = ['GLOBAL', 'LOCAL'].contains(value);
      if (valid) c.scope = value;
    } else if (key == 'asid') {
      valid = ['ON', 'OFF'].contains(value);
      if (valid) c.asid = value;
    } else {
      int n = int.parse(value);
      valid = switch (key) {
        'page' => n >= 16 && n <= 4096 && (n & (n - 1)) == 0,
        'frames' => n >= 1 && n <= 16,
        'tlb' || 'sets' => n >= 1 && n <= 8,
        'tick' => n >= 0 && n <= 1000,
        'tau' => n >= 1 && n <= 1000,
        _ => n >= 1 && n <= 2147483646,
      };
      if (valid) {
        switch (key) {
          case 'page':
            {
              c.page = n;
            }
          case 'frames':
            {
              c.frames = n;
            }
          case 'tlb':
            {
              c.tlb = n;
            }
          case 'sets':
            {
              c.sets = n;
            }
          case 'tick':
            {
              c.tick = n;
            }
          case 'tau':
            {
              c.tau = n;
            }
          case 'seed':
            {
              c.seed = n;
            }
        }
      }
    }
    if (!valid) throw CommandError('invalid $key');
  }
  if (c.tlb % c.sets != 0) throw CommandError('invalid sets');
  return c;
}
