45Config settings(List<String> tokens) {
46 Config c = Config();
47 Set<String> seen = {};
48 for (String token in tokens.skip(1)) {
49 List<String> pair = token.split('=');
50 String key = pair[0], value = pair[1];
51 if (![
52 'page',
53 'frames',
54 'tlb',
55 'sets',
56 'policy',
57 'scope',
58 'asid',
59 'tick',
60 'tau',
61 'seed',
62 ].contains(key))
63 throw CommandError('unknown key $key');
64 if (!seen.add(key)) throw CommandError('duplicate key $key');
65 bool valid;
66 if (key == 'policy') {
67 valid = policies.contains(value);
68 if (valid) c.policy = value;
69 } else if (key == 'scope') {
70 valid = ['GLOBAL', 'LOCAL'].contains(value);
71 if (valid) c.scope = value;
72 } else if (key == 'asid') {
73 valid = ['ON', 'OFF'].contains(value);
74 if (valid) c.asid = value;
75 } else {
76 int n = int.parse(value);
77 valid = switch (key) {
78 'page' => n >= 16 && n <= 4096 && (n & (n - 1)) == 0,
79 'frames' => n >= 1 && n <= 16,
80 'tlb' || 'sets' => n >= 1 && n <= 8,
81 'tick' => n >= 0 && n <= 1000,
82 'tau' => n >= 1 && n <= 1000,
83 _ => n >= 1 && n <= 2147483646,
84 };
85 if (valid) {
86 switch (key) {
87 case 'page':
88 {
89 c.page = n;
90 }
91 case 'frames':
92 {
93 c.frames = n;
94 }
95 case 'tlb':
96 {
97 c.tlb = n;
98 }
99 case 'sets':
100 {
101 c.sets = n;
102 }
103 case 'tick':
104 {
105 c.tick = n;
106 }
107 case 'tau':
108 {
109 c.tau = n;
110 }
111 case 'seed':
112 {
113 c.seed = n;
114 }
115 }
116 }
117 }
118 if (!valid) throw CommandError('invalid $key');
119 }
120 if (c.tlb % c.sets != 0) throw CommandError('invalid sets');
121 return c;
122}

203 Process? p = processes[pid];
204 if (p == null) throw CommandError('unknown process $pid');

210 c.frames ~/ processes.length +
211 (pids.indexOf(pid) < c.frames % processes.length ? 1 : 0);
212 void declare(int pid, int count) {
213 if (fixed) throw CommandError('processes are fixed');
214 if (processes.containsKey(pid))
215 throw CommandError('duplicate process $pid');
216 if (count < 1 || count > 256) throw CommandError('invalid page count');
217 if (processes.length >= 8 ||
218 c.scope == 'LOCAL' && processes.length == c.frames)
219 throw CommandError('too many processes');
220 processes[pid] = Process(pid, count);
221 }

229 void write(Page p) {
230 writebacks++;
231 processes[p.pid]!.writebacks++;
232 p.m = 0;
233 if (p.slot < 0) {
234 int slot = 0;
235 while (swap.containsKey(slot)) {
236 slot++;
237 }
238 p.slot = slot;
239 swap[slot] = p;
240 }

243 int nextUse(Page p) {
244 for (int i = cursor + 1; i < future.length; i++) {
245 Access a = future[i];
246 if (a.pid == p.pid && a.address ~/ c.page == p.number) return i;
247 }
248 return 1 << 30;
249 }

251 int choose(List<int> candidates, Process owner) {
252 List<int> unlocked = candidates.where((f) => !frames[f]!.locked).toList();
253 if (unlocked.isNotEmpty) candidates = unlocked;
254 if (c.policy == 'RANDOM') {
255 random = random * 48271 % 2147483647;
256 return candidates[random % candidates.length];
257 }
258 if (!['CLOCK', 'ECLOCK', 'WSCLOCK'].contains(c.policy)) {
259 (int, int) rank(int f) {
260 Page p = frames[f]!;
261 return switch (c.policy) {
262 'FIFO' => (p.load, 0),
263 'LRU' => (p.last, 0),
264 'LFU' => (p.uses, p.load),
265 'NRU' => (2 * p.r + p.m, f),
266 'AGING' => (p.age, p.load),
267 _ => (-nextUse(p), f),
268 };
269 }
270
271 candidates.sort((a, b) {
272 final x = rank(a), y = rank(b);
273 int first = x.$1.compareTo(y.$1);
274 return first != 0 ? first : x.$2.compareTo(y.$2);
275 });
276 return candidates.first;
277 }
278 int h = c.scope == 'GLOBAL' ? hand : owner.hand;
279 candidates.sort(
280 (a, b) => ((a - h + c.frames) % c.frames).compareTo(
281 (b - h + c.frames) % c.frames,
282 ),
283 );
284 int chosen = -1;
285 if (c.policy == 'CLOCK') {
286 while (chosen < 0) {
287 for (int f in candidates) {
288 if (frames[f]!.r == 0) {
289 chosen = f;
290 break;
291 }
292 frames[f]!.r = 0;
293 }
294 }
295 } else if (c.policy == 'ECLOCK') {
296 while (chosen < 0) {
297 for (int f in candidates) {
298 if (frames[f]!.r == 0 && frames[f]!.m == 0) {
299 chosen = f;
300 break;
301 }
302 }
303 if (chosen >= 0) break;
304 for (int f in candidates) {
305 if (frames[f]!.r == 0 && frames[f]!.m == 1) {
306 chosen = f;
307 break;
308 }
309 frames[f]!.r = 0;
310 }
311 }
312 } else {
313 for (int f in candidates) {
314 Page p = frames[f]!;
315 if (p.r == 1) {
316 p.r = 0;
317 p.last = time;
318 } else if (time - p.last > c.tau) {
319 if (p.m == 0) {
320 chosen = f;
321 break;
322 }
323 write(p);
324 }
325 }
326 if (chosen < 0) {
327 for (int f in candidates) {
328 if (frames[f]!.m == 0) {
329 chosen = f;
330 break;
331 }
332 }
333 if (chosen < 0) chosen = candidates.first;
334 }
335 }
336 if (c.scope == 'GLOBAL') {
337 hand = (chosen + 1) % c.frames;
338 } else {
339 owner.hand = (chosen + 1) % c.frames;
340 }
341 return chosen;
342 }

344 void access(int pid, String op, int addr) {
345 fixed = true;
346 cursor++;
347 Process owner = process(pid);
348 if (addr >= owner.pages.length * c.page) {
349 owner.segfaults++;
350 emit('SEGFAULT P$pid ${hex(addr)}');
351 return;
352 }
353 Page p = owner.pages[addr ~/ c.page];
354 if (op == 'W' && p.ro) {
355 owner.protfaults++;
356 emit('PROTFAULT P$pid ${hex(addr)}');
357 return;
358 }
359 time++;
360 owner.accesses++;
361 bool switched = previous >= 0 && previous != pid;
362 if (switched) {
363 switches++;
364 if (c.asid == 'OFF') {
365 for (List<Page> set in tlb) {
366 set.clear();
367 }
368 }
369 }
370 previous = pid;
371 List<Page> set = tlb[p.number % c.sets];
372 bool hit = set.remove(p), fault = false;
373 String result;
374 if (hit) {
375 tlbhits++;
376 result = 'TLBHIT';
377 } else if (p.frame >= 0) {
378 result = 'HIT';
379 } else {
380 fault = true;
381 faults++;
382 owner.faults++;
383 if (p.slot >= 0) {
384 swapin++;
385 result = 'FAULT SWAPIN';
386 } else {
387 zero++;
388 result = 'FAULT ZERO';
389 }
390 int frame = -1;
391 bool mayUseFree = c.scope == 'GLOBAL' || owner.resident() < quota(pid);
392 if (mayUseFree) {
393 for (int i = 0; i < frames.length; i++) {
394 if (frames[i] == null) {
395 frame = i;
396 break;
397 }
398 }
399 }
400 if (frame < 0) {
401 List<int> candidates = [];
402 for (int i = 0; i < frames.length; i++) {
403 if (frames[i] != null &&
404 (c.scope == 'GLOBAL' || frames[i]!.pid == pid))
405 candidates.add(i);
406 }
407 frame = choose(candidates, owner);
408 Page victim = frames[frame]!;
409 result += ' evict=${victim.label()}';
410 invalidate(victim);
411 if (victim.m == 1) {
412 write(victim);
413 result += ' writeback';
414 }
415 victim.frame = -1;
416 }
417 frames[frame] = p;
418 p.frame = frame;
419 p.load = time;
420 p.uses = 0;
421 p.m = 0;
422 p.age = 0;
423 owner.peak = math.max(owner.peak, owner.resident());
424 }
425 if (set.length == c.tlb ~/ c.sets) set.removeAt(0);
426 set.add(p);
427 p.r = 1;
428 if (op == 'W') p.m = 1;
429 p.last = time;
430 p.uses++;
431 p.ever = true;
432 p.references++;
433 owner.history.add(p.number);
434 faultHistory.add(fault);
435 log.add(Log('ACCESS', pid, op, addr, 0));
436 bool interrupt = c.tick > 0 && time % c.tick == 0;
437 if (interrupt) {
438 for (Page? page in frames) {
439 if (page != null) {
440 page.age = page.age ~/ 2 + 128 * page.r;
441 page.r = 0;
442 }
443 }
444 }
445 if (trace) {
446 emit(
447 't=$time P$pid $op ${hex(addr)} page=${p.number} frame=${p.frame} phys=${hex(p.frame * c.page + addr % c.page)} $result${switched ? ' switch' : ''}',
448 );
449 if (interrupt) emit('t=$time tick');
450 }
451 }

453 void exit(int pid) {
454 Process p = process(pid);
455 int count = 0, slots = 0;
456 for (Page page in p.pages) {
457 if (page.frame >= 0) {
458 frames[page.frame] = null;
459 page.frame = -1;
460 count++;
461 }
462 invalidate(page);
463 if (page.slot >= 0) {
464 swap.remove(page.slot);
465 page.slot = -1;
466 slots++;
467 }
468 }
469 p.exited = true;
470 log.add(Log('EXIT', pid, '', 0, 0));
471 emit('EXIT P$pid freed=$count slots=$slots');
472 }

474 void attributes(String cmd, int pid, int first, int last, String protection) {
475 Process p = process(pid);
476 if (first > last || last >= p.pages.length)
477 throw CommandError('invalid page range');
478 for (int i = first; i <= last; i++) {
479 if (cmd == 'PROTECT') {
480 p.pages[i].ro = protection == 'RO';
481 } else {
482 p.pages[i].locked = cmd == 'LOCK';
483 }
484 }
485 if (cmd != 'PROTECT') log.add(Log(cmd, pid, '', first, last));
486 }

488 void clean() {
489 int count = 0;
490 for (Page? p in frames) {
491 if (p != null && p.m == 1) {
492 write(p);
493 count++;
494 }
495 }
496 log.add(Log('CLEAN', 0, '', 0, 0));
497 emit('CLEAN written=$count');
498 }

500 Simulation rerun(String policy, int count) {
501 Config config = c.copy()
502 ..policy = policy
503 ..frames = count;
504 List<Access> references = log
505 .where((e) => e.kind == 'ACCESS')
506 .map((e) => Access(e.pid, e.op, e.a))
507 .toList();
508 Simulation replay = Simulation(config, references, true);
509 for (int pid in pids) {
510 replay.declare(pid, processes[pid]!.pages.length);
511 }
512 for (Log e in log) {
513 switch (e.kind) {
514 case 'ACCESS':
515 {
516 replay.access(e.pid, e.op, e.a);
517 }
518 case 'EXIT':
519 {
520 replay.exit(e.pid);
521 }
522 case 'CLEAN':
523 {
524 replay.clean();
525 }
526 default:
527 {
528 replay.attributes(e.kind, e.pid, e.a, e.b, '');
529 }
530 }
531 }
532 return replay;
533 }

541 );

596 void compare() {
597 emit('COMPARE accesses=$time frames=${c.frames}');
598 Map<String, Simulation> results = {};
599 for (String policy in policies) {
600 Simulation r = rerun(policy, c.frames);
601 results[policy] = r;
602 emit(
603 '$policy faults=${r.faults} writebacks=${r.writebacks} tlbhits=${r.tlbhits}',
604 );
605 }
606 List<String> ranking = [...policies]
607 ..sort((a, b) {
608 int d = results[a]!.faults.compareTo(results[b]!.faults);
609 if (d != 0) return d;
610 d = results[a]!.writebacks.compareTo(results[b]!.writebacks);
611 return d != 0 ? d : policies.indexOf(a).compareTo(policies.indexOf(b));
612 });
613 emit('ranking ${ranking.join(' ')}');
614 }

616 void execute(List<String> t) {
617 String cmd = t[0];
618 int n(int i) => int.parse(t[i]);
619 if (int.tryParse(cmd) != null) {
620 access(n(0), t[1], address(t[2]));
621 return;
622 }
623 switch (cmd) {
624 case 'PROCESS':
625 {
626 declare(n(1), n(2));
627 }
628 case 'EXIT':
629 {
630 exit(n(1));
631 }
632 case 'PROTECT':
633 case 'LOCK':
634 case 'UNLOCK':
635 {
636 attributes(cmd, n(1), n(2), n(3), t.length > 4 ? t[4] : '');
637 }
638 case 'TRACE':
639 {
640 trace = t[1] == 'ON';
641 }
642 case 'SUMMARY':
643 {
644 summary();
645 }
646 case 'FRAMES':
647 {
648 frameDump();
649 }
650 case 'PAGETABLE':
651 {
652 pageTable(n(1));
653 }
654 case 'TLB':
655 {
656 tlbDump();
657 }
658 case 'SWAP':
659 {
660 emit('SWAP used=${swap.length}');
661 for (int slot in swap.keys.toList()..sort()) {
662 emit('$slot ${swap[slot]!.label()}');
663 }
664 }
665 case 'CLEAN':
666 {
667 clean();
668 }
669 case 'COMPARE':
670 {
671 compare();
672 }
673 case 'CURVE':
674 {
675 String policy = t[1];
676 int lo = n(2), hi = n(3);
677 if (!policies.contains(policy)) throw CommandError('unknown policy');
678 if (lo < 1 ||
679 lo > hi ||
680 hi > 16 ||
681 c.scope == 'LOCAL' && lo < processes.length)
682 throw CommandError('invalid frame range');
683 emit('CURVE $policy frames=$lo..$hi');
684 int prior = 1 << 30;
685 for (int f = lo; f <= hi; f++) {
686 int value = rerun(policy, f).faults;
687 emit('F=$f faults=$value${value > prior ? ' ANOMALY' : ''}');
688 prior = value;
689 }
690 }
691 case 'EAT':
692 {
693 int total =
694 n(1) * time +
695 n(2) * (2 * time - tlbhits) +
696 n(3) * (swapin + writebacks);
697 emit(
698 time == 0
699 ? 'EAT accesses=0'
700 : 'EAT accesses=$time total=$total average=${decimal(total * 100 ~/ time)}',
701 );
702 }
703 case 'TIMELINE':
704 {
705 int w = n(1);
706 if (w < 1) throw CommandError('invalid window');
707 StringBuffer text = StringBuffer('TIMELINE window=$w');
708 for (int i = 0; i < faultHistory.length; i += w) {
709 text.write(
710 ' ${faultHistory.sublist(i, math.min(i + w, faultHistory.length)).where((v) => v).length}',
711 );
712 }
713 emit(text.toString());
714 }
715 case 'WSET':
716 {
717 int pid = n(1);
718 Process p = process(pid);
719 int w = n(2);
720 if (w < 1) throw CommandError('invalid window');
721 List<int> pages =
722 p.history.skip(math.max(0, p.history.length - w)).toSet().toList()
723 ..sort();
724 emit(
725 'WSET P$pid window=$w size=${pages.length}${pages.map((p) => ' $p').join()}',
726 );
727 }
728 case 'HOT':
729 {
730 int pid = n(1);
731 Process p = process(pid);
732 int k = n(2);
733 if (k < 1) throw CommandError('invalid count');
734 List<Page> pages = p.pages.where((p) => p.references > 0).toList()
735 ..sort(
736 (a, b) => a.references != b.references
737 ? b.references.compareTo(a.references)
738 : a.number.compareTo(b.number),
739 );
740 emit(
741 'HOT P$pid${pages.take(k).map((p) => ' ${p.number}:${p.references}').join()}',
742 );
743 }
744 }
745 }
746}

748void main() {
749 int count = int.parse(stdin.readLineSync()!);
750 List<List<String>> commands = [];
751 for (int i = 0; i < count; i++) {
752 List<String> t = stdin.readLineSync()!.split(' ');
753 if (t[0] == 'LOOP') {
754 int start = address(t[3]),
755 interval = int.parse(t[4]),
756 length = int.parse(t[5]),
757 repeats = int.parse(t[6]);
758 for (int r = 0; r < repeats; r++) {
759 for (int j = 0; j < length; j++) {
760 commands.add([t[1], t[2], hex(start + j * interval)]);
761 }
762 }
763 } else if (t[0] == 'RAND') {
764 int lo = address(t[2]),
765 hi = address(t[3]),
766 count = int.parse(t[4]),
767 rate = int.parse(t[6]),
768 x = int.parse(t[5]);
769 for (int j = 0; j < count; j++) {
770 x = x * 48271 % 2147483647;
771 int addr = lo + x % (hi - lo);
772 x = x * 48271 % 2147483647;
773 commands.add([t[1], x % 100 < rate ? 'W' : 'R', hex(addr)]);
774 }
775 } else {
776 commands.add(t);
777 }
778 }
779 Simulation? simulation;
780 for (int i = 0; i < commands.length; i++) {
