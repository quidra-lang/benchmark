73 public function __construct(array $config, array $refs)
74 {
75 $this->c = $config;
76 $this->refs = $refs;
77 $this->random = $config["seed"];
78 $this->frames = array_fill(0, $config["frames"], null);

81 public function declareProcess(int $pid, int $count): void
82 {
83 if ($this->fixed) {
84 fail("processes are fixed");
85 }
86 if (isset($this->procs[$pid])) {
87 fail("duplicate process $pid");
88 }
89 if ($count < 1 || $count > 256) {
90 fail("invalid page count");
91 }
92 $n = count($this->procs);
93 if ($n >= 8 || ($this->c["scope"] === "LOCAL" && $n >= $this->c["frames"])) {
94 fail("too many processes");
95 }
96 $p = (object) [
97 "pages" => [],
98 "accesses" => 0,
99 "faults" => 0,
100 "writebacks" => 0,
101 "segfaults" => 0,
102 "protfaults" => 0,
103 "peak" => 0,
104 "hand" => 0,
105 "refs" => [],
106 "exited" => false,
107 ];
108 for ($i = 0; $i < $count; $i++) {
109 $p->pages[] = (object) [
110 "pid" => $pid,
111 "p" => $i,
112 "frame" => null,
113 "slot" => null,
114 "ro" => false,
115 "locked" => false,
116 "r" => 0,
117 "m" => 0,
118 "load" => 0,
119 "last" => 0,
120 "uses" => 0,
121 "age" => 0,
122 "ever" => false,
123 ];
124 }
125 $this->procs[$pid] = $p;
126 }
127 private function process(int $pid): object
128 {
129 if (!isset($this->procs[$pid])) {
130 fail("unknown process $pid");
131 }
132 $p = $this->procs[$pid];
133 if ($p->exited) {
134 fail("process $pid exited");
135 }
136 return $p;
137 }
138 private function resident(int $pid): int
139 {
140 return count(array_filter($this->frames, fn($p) => $p !== null && $p->pid === $pid));
141 }
142 private function quota(int $pid): int
143 {
144 $ids = sortedKeys($this->procs);
145 $n = count($ids);
146 return intdiv($this->c["frames"], $n) +
147 (array_search($pid, $ids, true) < $this->c["frames"] % $n ? 1 : 0);
148 }
149 private function invalidate(object $page): void
150 {
151 foreach ($this->tlb as &$set) {
152 $set = array_values(array_filter($set, fn($p) => $p !== $page));
153 }
154 unset($set);
155 }
156 private function writeback(object $page): void
157 {
158 $this->writebacks++;
159 $this->procs[$page->pid]->writebacks++;
160 $page->m = 0;
161 if ($page->slot === null) {
162 $slot = 0;
163 while (isset($this->swap[$slot])) {
164 $slot++;
165 }
166 $page->slot = $slot;
167 $this->swap[$slot] = $page;
168 }

170 private function victim(int $pid): int
171 {
172 $candidates = [];
173 $unlocked = [];
174 foreach ($this->frames as $i => $p) {
175 if ($p !== null && ($this->c["scope"] === "GLOBAL" || $p->pid === $pid)) {
176 $candidates[] = $i;
177 if (!$p->locked) {
178 $unlocked[] = $i;
179 }
180 }
181 }
182 if ($unlocked) {
183 $candidates = $unlocked;
184 }
185 $policy = $this->c["policy"];
186 if ($policy === "RANDOM") {
187 $this->random = (48271 * $this->random) % 2147483647;
188 return $candidates[$this->random % count($candidates)];
189 }
190 if ($policy === "OPT") {
191 $chosen = null;
192 $best = -1;
193 foreach ($candidates as $i) {
194 $p = $this->frames[$i];
195 $pos = PHP_INT_MAX;
196 for ($j = $this->cursor; $j < count($this->refs); $j++) {
197 $r = $this->refs[$j];
198 if ($r[0] === $p->pid && intdiv($r[2], $this->c["page"]) === $p->p) {
199 $pos = $j;
200 break;
201 }
202 }
203 if ($chosen === null || $pos > $best) {
204 $chosen = $i;
205 $best = $pos;
206 }
207 }
208 return $chosen;
209 }
210 $fields = ["FIFO" => "load", "LRU" => "last", "LFU" => "uses", "AGING" => "age"];
211 if (isset($fields[$policy]) || $policy === "NRU") {
212 $chosen = null;
213 $best = 0;
214 $bestTie = 0;
215 foreach ($candidates as $i) {
216 $p = $this->frames[$i];
217 $value = $policy === "NRU" ? 2 * $p->r + $p->m : $p->{$fields[$policy]};
218 $tie = $policy === "NRU" ? $i : $p->load;
219 if ($chosen === null || $value < $best || ($value === $best && $tie < $bestTie)) {
220 $chosen = $i;
221 $best = $value;
222 $bestTie = $tie;
223 }
224 }
225 return $chosen;
226 }
227 $hand = $this->c["scope"] === "GLOBAL" ? $this->hand : $this->procs[$pid]->hand;
228 $frames = $this->c["frames"];
229 usort(
230 $candidates,
231 fn($a, $b) => ($a - $hand + $frames) % $frames <=> ($b - $hand + $frames) % $frames,
232 );
233 $chosen = null;
234 if ($policy === "CLOCK") {
235 while ($chosen === null) {
236 foreach ($candidates as $i) {
237 if ($this->frames[$i]->r === 0) {
238 $chosen = $i;
239 break;
240 }
241 $this->frames[$i]->r = 0;
242 }
243 }
244 } elseif ($policy === "ECLOCK") {
245 while ($chosen === null) {
246 foreach ($candidates as $i) {
247 $p = $this->frames[$i];
248 if ($p->r === 0 && $p->m === 0) {
249 $chosen = $i;
250 break;
251 }
252 }
253 if ($chosen !== null) {
254 break;
255 }
256 foreach ($candidates as $i) {
257 $p = $this->frames[$i];
258 if ($p->r === 0 && $p->m === 1) {
259 $chosen = $i;
260 break;
261 }
262 $p->r = 0;
263 }
264 }
265 } else {
266 foreach ($candidates as $i) {
267 $p = $this->frames[$i];
268 if ($p->r === 1) {
269 $p->r = 0;
270 $p->last = $this->time;
271 } elseif ($this->time - $p->last > $this->c["tau"]) {
272 if ($p->m === 0) {
273 $chosen = $i;
274 break;
275 }
276 $this->writeback($p);
277 }
278 }
279 if ($chosen === null) {
280 foreach ($candidates as $i) {
281 if ($this->frames[$i]->m === 0) {
282 $chosen = $i;
283 break;
284 }
285 }
286 $chosen ??= $candidates[0];
287 }
288 }
289 if ($this->c["scope"] === "GLOBAL") {
290 $this->hand = ($chosen + 1) % $frames;
291 } else {
292 $this->procs[$pid]->hand = ($chosen + 1) % $frames;
293 }
294 return $chosen;
295 }
296 public function access(int $pid, string $op, int $address, bool $silent = false): void
297 {
298 $this->fixed = true;
299 $this->cursor++;
300 $proc = $this->process($pid);
301 if ($address >= count($proc->pages) * $this->c["page"]) {
302 $proc->segfaults++;
303 if (!$silent) {
304 out("SEGFAULT P$pid " . addr($address));
305 }
306 return;
307 }
308 $page = $proc->pages[intdiv($address, $this->c["page"])];
309 if ($op === "W" && $page->ro) {
310 $proc->protfaults++;
311 if (!$silent) {
312 out("PROTFAULT P$pid " . addr($address));
313 }
314 return;
315 }
316 $this->events[] = ["access", $pid, $op, $address];
317 $this->time++;
318 $proc->accesses++;
319 $proc->refs[] = $page->p;
320 $switched = $this->previous !== null && $this->previous !== $pid;
321 if ($switched) {
322 $this->switches++;
323 if ($this->c["asid"] === "OFF") {
324 $this->tlb = array_fill(0, $this->c["sets"], []);
325 }
326 }
327 $this->previous = $pid;
328 $set = &$this->tlb[$page->p % $this->c["sets"]];
329 $hit = array_search($page, $set, true);
330 $fault = false;
331 $victim = null;
332 $wrote = false;
333 if ($hit !== false) {
334 $result = "TLBHIT";
335 $this->tlbhits++;
336 array_splice($set, $hit, 1);
337 } else {
338 if ($page->frame !== null) {
339 $result = "HIT";
340 } else {
341 $fault = true;
342 $this->faults++;
343 $proc->faults++;
344 if ($page->slot !== null) {
345 $this->swapin++;
346 $result = "FAULT SWAPIN";
347 } else {
348 $this->zero++;
349 $result = "FAULT ZERO";
350 }
351 $free = array_search(null, $this->frames, true);
352 $useFree =
353 $this->c["scope"] === "GLOBAL"
354 ? $free !== false
355 : $this->resident($pid) < $this->quota($pid);
356 $frame = $useFree ? $free : $this->victim($pid);
357 $victim = $this->frames[$frame];
358 if ($victim !== null) {
359 $this->invalidate($victim);
360 $wrote = $victim->m === 1;
361 if ($wrote) {
362 $this->writeback($victim);
363 }
364 $victim->frame = null;
365 }
366 $this->frames[$frame] = $page;
367 $page->frame = $frame;
368 $page->load = $this->time;
369 $page->uses = $page->m = $page->age = 0;
370 $proc->peak = max($proc->peak, $this->resident($pid));
371 }
372 if (count($set) === intdiv($this->c["tlb"], $this->c["sets"])) {
373 array_shift($set);
374 }
375 }
376 $set[] = $page;
377 unset($set);
378 $page->r = 1;
379 if ($op === "W") {
380 $page->m = 1;
381 }
382 $page->last = $this->time;
383 $page->uses++;
384 $page->ever = true;
385 $this->history[] = $fault;
386 $tick = $this->c["tick"] > 0 && $this->time % $this->c["tick"] === 0;
387 if ($tick) {
388 foreach ($this->frames as $p) {
389 if ($p !== null) {
390 $p->age = intdiv($p->age, 2) + 128 * $p->r;
391 $p->r = 0;
392 }
393 }
394 }
395 if ($this->trace && !$silent) {
396 $line =
397 "t={$this->time} P$pid $op " .
398 addr($address) .
399 " page={$page->p} frame={$page->frame} phys=" .
400 addr($page->frame * $this->c["page"] + ($address % $this->c["page"])) .
401 " $result";
402 if ($victim !== null) {
403 $line .= " evict=P{$victim->pid}:{$victim->p}";
404 }
405 if ($wrote) {
406 $line .= " writeback";
407 }
408 if ($switched) {
409 $line .= " switch";
410 }
411 out($line);
412 if ($tick) {
413 out("t={$this->time} tick");
414 }
415 }
416 }
417 public function exitProcess(int $pid, bool $silent = false): void
418 {
419 $p = $this->process($pid);
420 $freed = $this->resident($pid);
421 $slots = 0;
422 foreach ($p->pages as $page) {
423 if ($page->frame !== null) {
424 $this->frames[$page->frame] = null;
425 }
426 $page->frame = null;
427 $this->invalidate($page);
428 if ($page->slot !== null) {
429 unset($this->swap[$page->slot]);
430 $page->slot = null;
431 $slots++;
432 }
433 }
434 $p->exited = true;
435 $this->events[] = ["exit", $pid];
436 if (!$silent) {
437 out("EXIT P$pid freed=$freed slots=$slots");
438 }

440 public function attributes(
441 string $cmd,
442 int $pid,
443 int $first,
444 int $last,
445 ?string $protection = null,
446 ): void {
447 $p = $this->process($pid);
448 if ($first > $last || $last >= count($p->pages)) {
449 fail("invalid page range");
450 }
451 for ($i = $first; $i <= $last; $i++) {
452 if ($cmd === "PROTECT") {
453 $p->pages[$i]->ro = $protection === "RO";
454 } else {
455 $p->pages[$i]->locked = $cmd === "LOCK";
456 }
457 }
458 if ($cmd !== "PROTECT") {
459 $this->events[] = ["attribute", $cmd, $pid, $first, $last];
460 }

462 public function clean(bool $silent = false): void
463 {
464 $count = 0;
465 foreach ($this->frames as $p) {
466 if ($p !== null && $p->m === 1) {
467 $this->writeback($p);
468 $count++;
469 }
470 }
471 $this->events[] = ["clean"];
472 if (!$silent) {
473 out("CLEAN written=$count");
474 }

476 private function rerun(string $policy, int $frames): Simulation
477 {
478 $config = $this->c;
479 $config["policy"] = $policy;
480 $config["frames"] = $frames;
481 $refs = [];
482 foreach ($this->events as $e) {
483 if ($e[0] === "access") {
484 $refs[] = array_slice($e, 1);
485 }
486 }
487 $s = new Simulation($config, $refs);
488 foreach ($this->procs as $pid => $p) {
489 $s->declareProcess($pid, count($p->pages));
490 }
491 foreach ($this->events as $e) {
492 switch ($e[0]) {
493 case "access":
494 $s->access($e[1], $e[2], $e[3], true);
495 break;
496 case "exit":
497 $s->exitProcess($e[1], true);
498 break;
499 case "clean":
500 $s->clean(true);
501 break;
502 case "attribute":
503 $s->attributes($e[1], $e[2], $e[3], $e[4]);
504 break;
505 }
506 }
507 return $s;
508 }
509 public function command(array $w): void
510 {
511 $cmd = $w[0];
512 $pid = (int) ($w[1] ?? 0);
513 switch ($cmd) {
514 case "PROCESS":
515 $this->declareProcess($pid, (int) $w[2]);
516 break;
517 case "EXIT":
518 $this->exitProcess($pid);
519 break;
520 case "PROTECT":
521 case "LOCK":
522 case "UNLOCK":
523 $this->attributes($cmd, $pid, (int) $w[2], (int) $w[3], $w[4] ?? null);
524 break;
525 case "TRACE":
526 $this->trace = $w[1] === "ON";
527 break;
528 case "CLEAN":
529 $this->clean();
530 break;
531 case "SUMMARY":
532 out(
533 "SUMMARY policy={$this->c["policy"]} scope={$this->c["scope"]} time={$this->time}",
534 );
535 $seg = $prot = 0;
536 foreach ($this->procs as $p) {
537 $seg += $p->segfaults;
538 $prot += $p->protfaults;
539 }
540 out(
541 "total faults={$this->faults} zero={$this->zero} swapin={$this->swapin} faultrate=" .
542 ratio($this->faults, $this->time) .
543 " tlbhits={$this->tlbhits} tlbrate=" .
544 ratio($this->tlbhits, $this->time) .
545 " writebacks={$this->writebacks} switches={$this->switches} segfaults=$seg protfaults=$prot",
546 );
547 foreach (sortedKeys($this->procs) as $id) {
548 $p = $this->procs[$id];
549 $line =
550 "P$id accesses={$p->accesses} faults={$p->faults} writebacks={$p->writebacks} segfaults={$p->segfaults} protfaults={$p->protfaults} resident=" .
551 $this->resident($id) .
552 " peak={$p->peak}";
553 if ($this->c["scope"] === "LOCAL") {
554 $line .= " quota=" . $this->quota($id);
555 }
556 if ($p->exited) {
557 $line .= " exited";
558 }
559 out($line);
560 }
561 break;
562 case "FRAMES":
563 out("FRAMES used=" . count(array_filter($this->frames)) . "/" . $this->c["frames"]);
564 foreach ($this->frames as $i => $p) {
565 if ($p === null) {
566 out("$i free");
567 } else {
568 out(
569 "$i P{$p->pid}:{$p->p} R={$p->r} M={$p->m} load={$p->load} last={$p->last} uses={$p->uses} age={$p->age}" .
570 ($p->locked ? " locked" : ""),
571 );
572 }
573 }
574 if (in_array($this->c["policy"], ["CLOCK", "ECLOCK", "WSCLOCK"], true)) {
575 if ($this->c["scope"] === "GLOBAL") {
576 out("hand={$this->hand}");
577 } else {
578 $line = "hand";
579 foreach (sortedKeys($this->procs) as $id) {
580 $line .= " P$id=" . $this->procs[$id]->hand;
581 }
582 out($line);
583 }
584 }
585 break;
586 case "PAGETABLE":
587 $p = $this->process($pid);
588 out(
589 "PAGETABLE P$pid pages=" .
590 count($p->pages) .
591 " resident=" .
592 $this->resident($pid),
593 );
594 foreach ($p->pages as $page) {
595 if (!$page->ever) {
596 continue;
597 }
598 if ($page->frame !== null) {
599 $state = "frame={$page->frame} R={$page->r} M={$page->m}";
600 } elseif ($page->slot !== null) {
601 $state = "swap={$page->slot}";
602 } else {
603 $state = "zero";
604 }
605 out(
606 "{$page->p} $state" .
607 ($page->ro ? " RO" : "") .
608 ($page->locked ? " locked" : ""),
609 );
610 }
611 break;
612 case "TLB":
613 $used = array_sum(array_map("count", $this->tlb));
614 out("TLB used=$used/{$this->c["tlb"]} sets={$this->c["sets"]}");
615 foreach ($this->tlb as $i => $set) {
616 $line = "set $i:";
617 foreach (array_reverse($set) as $p) {
618 $line .= " P{$p->pid}:{$p->p}->{$p->frame}";
619 }
620 out($line);
621 }
622 break;
623 case "SWAP":
624 out("SWAP used=" . count($this->swap));
625 foreach (sortedKeys($this->swap) as $slot) {
626 $p = $this->swap[$slot];
627 out("$slot P{$p->pid}:{$p->p}");
628 }
629 break;
630 case "COMPARE":
631 out("COMPARE accesses={$this->time} frames={$this->c["frames"]}");
632 $results = [];
633 foreach (POLICIES as $i => $policy) {
634 $s = $this->rerun($policy, $this->c["frames"]);
635 out(
636 "$policy faults={$s->faults} writebacks={$s->writebacks} tlbhits={$s->tlbhits}",
637 );
638 $results[] = [$s->faults, $s->writebacks, $i, $policy];
639 }
640 sort($results);
641 out("ranking " . implode(" ", array_column($results, 3)));
642 break;
643 case "CURVE":
644 [$policy, $lo, $hi] = [$w[1], (int) $w[2], (int) $w[3]];
645 if (!in_array($policy, POLICIES, true)) {
646 fail("unknown policy");
647 }
648 if (
649 $lo < 1 ||
650 $lo > $hi ||
651 $hi > 16 ||
652 ($this->c["scope"] === "LOCAL" && $lo < count($this->procs))
653 ) {
654 fail("invalid frame range");
655 }
656 out("CURVE $policy frames=$lo..$hi");
657 $previous = null;
658 for ($f = $lo; $f <= $hi; $f++) {
659 $faults = $this->rerun($policy, $f)->faults;
660 out(
661 "F=$f faults=$faults" .
662 ($previous !== null && $faults > $previous ? " ANOMALY" : ""),
663 );
664 $previous = $faults;
665 }
666 break;
667 case "EAT":
668 if ($this->time === 0) {
669 out("EAT accesses=0");
670 } else {
671 $total =
672 (int) $w[1] * $this->time +
673 (int) $w[2] * (2 * $this->time - $this->tlbhits) +
674 (int) $w[3] * ($this->swapin + $this->writebacks);
675 $v = intdiv(100 * $total, $this->time);
676 out(
677 "EAT accesses={$this->time} total=$total average=" .
678 sprintf("%d.%02d", intdiv($v, 100), $v % 100),
679 );
680 }
681 break;
682 case "TIMELINE":
683 if ($pid < 1) {
684 fail("invalid window");
685 }
686 $line = "TIMELINE window=$pid";
687 foreach (array_chunk($this->history, $pid) as $slice) {
688 $line .= " " . count(array_filter($slice));
689 }
690 out($line);
691 break;
692 case "WSET":
693 case "HOT":
694 $p = $this->process($pid);
695 $n = (int) $w[2];
696 if ($n < 1) {
697 fail($cmd === "WSET" ? "invalid window" : "invalid count");
698 }
699 $refs = $cmd === "WSET" ? array_slice($p->refs, -$n) : $p->refs;
700 $counts = array_count_values($refs);
701 $pages = array_keys($counts);
702 if ($cmd === "WSET") {
703 sort($pages, SORT_NUMERIC);
704 out(
705 "WSET P$pid window=$n size=" .
706 count($pages) .
707 ($pages ? " " . implode(" ", $pages) : ""),
708 );
709 } else {
710 usort($pages, fn($a, $b) => $counts[$b] <=> $counts[$a] ?: $a <=> $b);
711 $line = "HOT P$pid";
712 foreach (array_slice($pages, 0, $n) as $page) {
713 $line .= " $page:{$counts[$page]}";
714 }
715 out($line);
716 }
717 break;
718 default:
719 $this->access((int) $w[0], $w[1], intval($w[2], 16));
720 break;
721 }
722 }

724function settings(array $w): array
725{
726 $c = DEFAULTS;
727 $seen = [];
728 foreach (array_slice($w, 1) as $pair) {
729 [$key, $value] = explode("=", $pair);
730 if (!array_key_exists($key, $c)) {
731 fail("unknown key $key");
732 }
733 if (isset($seen[$key])) {
734 fail("duplicate key $key");
735 }
736 $seen[$key] = true;
737 $n = (int) $value;
738 $ranges = [
739 "frames" => [1, 16],
740 "tlb" => [1, 8],
741 "sets" => [1, 8],
742 "tick" => [0, 1000],
743 "tau" => [1, 1000],
744 "seed" => [1, 2147483646],
745 ];
746 if ($key === "policy") {
747 $valid = in_array($value, POLICIES, true);
748 } elseif ($key === "scope") {
749 $valid = in_array($value, ["GLOBAL", "LOCAL"], true);
750 } elseif ($key === "asid") {
