string[] policies()
    return "FIFO LRU LFU NRU AGING CLOCK ECLOCK WSCLOCK RANDOM OPT".split(" ")
int lookup(const string[] items, string target)
    for: j < #items
        if items[j] == target
            return int(j)
    return -1
class Settings()
    page_size := 256
    frame_count := 4
    tlb_size := 4
    sets := 1
    policy := "FIFO"
    scope := "GLOBAL"
    asid := false
    tick_interval := 0
    tau := 8
    seed := 1
Settings | error parse_settings(const string[] words)
    result := Settings()
    names := "page frames tlb sets policy scope asid tick tau seed".split(" ")
    seen := array(10, fill = false)
    for: 1 <= j < #words
        parts := words[j].split("=")
        key := lookup(names, parts[0])
        if key < 0
            return error("unknown key {parts[0]}")
        if seen[key]
            return error("duplicate key {parts[0]}")
        seen[key] = true
        value := parts[1]
        valid := true
        if key == 4
            valid = lookup(policies(), value) >= 0
            result.policy = value
        elif key == 5
            valid = value == "GLOBAL" or value == "LOCAL"
            result.scope = value
        elif key == 6
            valid = value == "ON" or value == "OFF"
            result.asid = value == "ON"
        else
            number := int(value)
            lower := [16, 1, 1, 1, 0, 0, 0, 0, 1, 1]
            upper := [4096, 16, 8, 8, 0, 0, 0, 1000, 1000, 2147483646]
            valid = lower[key] <= number <= upper[key]
            if key == 0
                reduced := number
                while reduced > 0 and reduced % 2 == 0
                    reduced /= 2
                valid = valid and reduced == 1
                result.page_size = number
            elif key == 1
                result.frame_count = number
            elif key == 2
                result.tlb_size = number
            elif key == 3
                result.sets = number
            elif key == 7
                result.tick_interval = number
            elif key == 8
                result.tau = number
            else
                result.seed = number
        if not valid
            return error("invalid {parts[0]}")
    if result.tlb_size % result.sets != 0
        return error("invalid sets")
    return result
