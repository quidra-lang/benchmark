54func number(s string) int { n, _ := strconv.Atoi(s); return n }
55func address(s string) int { n, _ := strconv.ParseInt(s[2:], 16, 64); return int(n) }
56func contains(words []string, s string) bool {
57 for _, word := range words {
58 if word == s {
59 return true
60 }
61 }
62 return false
63}
64func fail(s string) { panic(s) }
65func hex(n int) string { return fmt.Sprintf("0x%X", n) }
66func label(p *Page) string { return fmt.Sprintf("P%d:%d", p.PID, p.Number) }
67func ratio(x, y int) string {
68 v := 0
69 if y > 0 {
70 v = 10000 * x / y
71 }
72 return fmt.Sprintf("%d.%02d%%", v/100, v%100)
73}
74func settings(words []string) Settings {
75 c := Settings{256, 4, 4, 1, 0, 8, 1, "FIFO", "GLOBAL", "OFF"}
76 known := []string{"page", "frames", "tlb", "sets", "policy", "scope", "asid", "tick", "tau", "seed"}
77 seen := map[string]bool{}
78 for _, word := range words {
79 kv := strings.SplitN(word, "=", 2)
80 k, v := kv[0], kv[1]
81 if !contains(known, k) {
82 fail("unknown key " + k)
83 }
84 if seen[k] {
85 fail("duplicate key " + k)
86 }
87 seen[k] = true
88 n64, err := strconv.ParseInt(v, 10, 64)
89 n := int(n64)
90 valid := err == nil
91 switch k {
92 case "page":
93 valid = valid && n >= 16 && n <= 4096 && n&(n-1) == 0
94 c.Page = n
95 case "frames":
96 valid = valid && n >= 1 && n <= 16
97 c.Frames = n
98 case "tlb":
99 valid = valid && n >= 1 && n <= 8
100 c.TLB = n
101 case "sets":
102 valid = valid && n >= 1 && n <= 8
103 c.Sets = n
104 case "tick":
105 valid = valid && n >= 0 && n <= 1000
106 c.Tick = n
107 case "tau":
108 valid = valid && n >= 1 && n <= 1000
109 c.Tau = n
110 case "seed":
111 valid = valid && n64 >= 1 && n64 <= 2147483646
112 c.Seed = n64
113 case "policy":
114 valid = contains(policies, v)
115 c.Policy = v
116 case "scope":
117 valid = v == "GLOBAL" || v == "LOCAL"
118 c.Scope = v
119 case "asid":
120 valid = v == "ON" || v == "OFF"
121 c.ASID = v
122 }
123 if !valid {
124 fail("invalid " + k)
125 }
126 }
127 if c.TLB%c.Sets != 0 {
128 fail("invalid sets")
129 }
130 return c
131}
132func newSimulation(c Settings, refs [][2]int) *Simulation {
133 return &Simulation{Config: c, Processes: map[int]*Process{}, Frames: make([]*Page, c.Frames), TLB: make([][]*Page, c.Sets), Refs: refs, Random: c.Seed}
134}
135func (s *Simulation) pids() []int {
136 ids := []int{}
137 for id := range s.Processes {
138 ids = append(ids, id)
139 }
140 sort.Ints(ids)
141 return ids
142}
143func (s *Simulation) declare(pid, count int) {
144 p := &Process{PID: pid}
145 for n := 0; n < count; n++ {
146 p.Pages = append(p.Pages, &Page{PID: pid, Number: n, Frame: -1, Slot: -1})
147 }
148 s.Processes[pid] = p
149}
150func (s *Simulation) process(pid int) *Process {
151 p, ok := s.Processes[pid]
152 if !ok {
153 fail(fmt.Sprintf("unknown process %d", pid))
154 }
155 if p.Exited {
156 fail(fmt.Sprintf("process %d exited", pid))
157 }
158 return p
159}
160func (s *Simulation) quota(pid int) int {
161 ids := s.pids()
162 q, r := s.Config.Frames/len(ids), s.Config.Frames%len(ids)
163 for i, id := range ids {
164 if id == pid && i < r {
165 return q + 1
166 }
167 }
168 return q
169}
170func (s *Simulation) removeTLB(page *Page) {
171 k := page.Number % s.Config.Sets
172 entries := s.TLB[k][:0]
173 for _, p := range s.TLB[k] {
174 if p != page {
175 entries = append(entries, p)
176 }
177 }
178 s.TLB[k] = entries
179}
180func (s *Simulation) writeback(p *Page) {
181 s.Writes++
182 s.Processes[p.PID].Writes++
183 p.M = 0
184 if p.Slot < 0 {
185 slot := 0
186 for slot < len(s.Slots) && s.Slots[slot] != nil {
187 slot++
188 }
189 if slot == len(s.Slots) {
190 s.Slots = append(s.Slots, nil)
191 }
192 p.Slot = slot
193 s.Slots[slot] = p
194 }
195}
196func (s *Simulation) nextUse(p *Page) int {
197 for i := s.Position + 1; i < len(s.Refs); i++ {
198 if s.Refs[i] == [2]int{p.PID, p.Number} {
199 return i
200 }
201 }
202 return len(s.Refs)
203}
204func (s *Simulation) choose(proc *Process, candidates []int) int {
205 c := s.Config
206 if c.Policy == "RANDOM" {
207 s.Random = s.Random * 48271 % 2147483647
208 return candidates[int(s.Random%int64(len(candidates)))]
209 }
210 if contains([]string{"CLOCK", "ECLOCK", "WSCLOCK"}, c.Policy) {
211 hand := s.Hand
212 if c.Scope == "LOCAL" {
213 hand = proc.Hand
214 }
215 sort.Slice(candidates, func(i, j int) bool {
216 return (candidates[i]-hand+c.Frames)%c.Frames < (candidates[j]-hand+c.Frames)%c.Frames
217 })
218 chosen := -1
219 switch c.Policy {
220 case "CLOCK":
221 for chosen < 0 {
222 for _, f := range candidates {
223 p := s.Frames[f]
224 if p.R == 0 {
225 chosen = f
226 break
227 }
228 p.R = 0
229 }
230 }
231 case "ECLOCK":
232 for chosen < 0 {
233 for _, f := range candidates {
234 p := s.Frames[f]
235 if p.R == 0 && p.M == 0 {
236 chosen = f
237 break
238 }
239 }
240 if chosen >= 0 {
241 break
242 }
243 for _, f := range candidates {
244 p := s.Frames[f]
245 if p.R == 0 && p.M == 1 {
246 chosen = f
247 break
248 }
249 p.R = 0
250 }
251 }
252 case "WSCLOCK":
253 for _, f := range candidates {
254 p := s.Frames[f]
255 if p.R == 1 {
256 p.R = 0
257 p.Last = s.Time
258 } else if s.Time-p.Last > c.Tau {
259 if p.M == 0 {
260 chosen = f
261 break
262 }
263 s.writeback(p)
264 }
265 }
266 if chosen < 0 {
267 for _, f := range candidates {
268 if s.Frames[f].M == 0 {
269 chosen = f
270 break
271 }
272 }
273 }
274 if chosen < 0 {
275 chosen = candidates[0]
276 }
277 }
278 if c.Scope == "LOCAL" {
279 proc.Hand = (chosen + 1) % c.Frames
280 } else {
281 s.Hand = (chosen + 1) % c.Frames
282 }
283 return chosen
284 }
285 rank := func(f int) (int, int) {
286 p := s.Frames[f]
287 switch c.Policy {
288 case "FIFO":
289 return p.Load, f
290 case "LRU":
291 return p.Last, f
292 case "LFU":
293 return p.Uses, p.Load
294 case "NRU":
295 return 2*p.R + p.M, f
296 case "AGING":
297 return p.Age, p.Load
298 default:
299 return -s.nextUse(p), f
300 }
301 }
302 best := candidates[0]
303 for _, f := range candidates[1:] {
304 a, b := rank(f)
305 x, y := rank(best)
306 if a < x || (a == x && b < y) {
307 best = f
308 }
309 }
310 return best
311}
312func (s *Simulation) target(p *Process) int {
313 global := s.Config.Scope == "GLOBAL"
314 if global || p.Resident < s.quota(p.PID) {
315 for f, page := range s.Frames {
316 if page == nil {
317 return f
318 }
319 }
320 }
321 candidates, unlocked := []int{}, []int{}
322 for f, page := range s.Frames {
323 if page != nil && (global || page.PID == p.PID) {
324 candidates = append(candidates, f)
325 if !page.Locked {
326 unlocked = append(unlocked, f)
327 }
328 }
329 }
330 if len(unlocked) > 0 {
331 candidates = unlocked
332 }
333 return s.choose(p, candidates)
334}
335func (s *Simulation) access(pid int, op string, addr, pos int) []string {
336 s.Position = pos
337 s.Time++
338 p := s.Processes[pid]
339 switched := s.LastPID != 0 && s.LastPID != pid
340 if switched {
341 s.Switches++
342 if s.Config.ASID == "OFF" {
343 s.TLB = make([][]*Page, s.Config.Sets)
344 }
345 }
346 s.LastPID = pid
347 n, offset := addr/s.Config.Page, addr%s.Config.Page
348 page := p.Pages[n]
349 k := n % s.Config.Sets
350 hit := false
351 for _, entry := range s.TLB[k] {
352 if entry == page {
353 hit = true
354 break
355 }
356 }
357 result := ""
358 fault := 0
359 if hit {
360 s.Hits++
361 result = "TLBHIT"
362 s.removeTLB(page)
363 } else if page.Frame >= 0 {
364 result = "HIT"
365 } else {
366 fault = 1
367 s.Faults++
368 p.Faults++
369 result = "FAULT ZERO"
370 if page.Slot >= 0 {
371 s.Swapins++
372 result = "FAULT SWAPIN"
373 } else {
374 s.Zeros++
375 }
376 f := s.target(p)
377 victim := s.Frames[f]
378 if victim != nil {
379 result += " evict=" + label(victim)
380 victim.Frame = -1
381 s.Processes[victim.PID].Resident--
382 s.removeTLB(victim)
383 if victim.M == 1 {
384 s.writeback(victim)
385 result += " writeback"
386 }
387 }
388 s.Frames[f] = page
389 page.Frame = f
390 page.Load = s.Time
391 page.Uses, page.M, page.Age = 0, 0, 0
392 p.Resident++
393 if p.Resident > p.Peak {
394 p.Peak = p.Resident
395 }
396 }
397 entries := s.TLB[k]
398 if len(entries) == s.Config.TLB/s.Config.Sets {
399 entries = entries[:len(entries)-1]
400 }
401 s.TLB[k] = append([]*Page{page}, entries...)
402 page.R = 1
403 if op == "W" {
404 page.M = 1
405 }
406 page.Last = s.Time
407 page.Uses++
408 page.Seen = true
409 p.History = append(p.History, n)
410 s.FaultHistory = append(s.FaultHistory, fault)
411 line := fmt.Sprintf("t=%d P%d %s %s page=%d frame=%d phys=%s %s", s.Time, pid, op, hex(addr), n, page.Frame, hex(page.Frame*s.Config.Page+offset), result)
412 if switched {
413 line += " switch"
414 }
415 lines := []string{line}
416 if s.Config.Tick > 0 && s.Time%s.Config.Tick == 0 {
417 for _, pg := range s.Frames {
418 if pg != nil {
419 pg.Age = pg.Age/2 + 128*pg.R
420 pg.R = 0
421 }
422 }
423 lines = append(lines, fmt.Sprintf("t=%d tick", s.Time))
424 }
425 return lines
426}
427func (s *Simulation) exit(pid int) (int, int) {
428 freed, slots := 0, 0
429 for f, p := range s.Frames {
430 if p != nil && p.PID == pid {
431 s.Frames[f] = nil
432 p.Frame = -1
433 s.removeTLB(p)
434 freed++
435 }
436 }
437 for i, p := range s.Slots {
438 if p != nil && p.PID == pid {
439 s.Slots[i] = nil
440 p.Slot = -1
441 slots++
442 }
443 }
444 p := s.Processes[pid]
445 p.Resident = 0
446 p.Exited = true
447 return freed, slots
448}
449func (s *Simulation) clean() int {
450 n := 0
451 for _, p := range s.Frames {
452 if p != nil && p.M == 1 {
453 s.writeback(p)
454 n++
455 }
456 }
457 return n
458}
459func (s *Simulation) lock(pid, first, last int, locked bool) {
460 for _, p := range s.Processes[pid].Pages[first : last+1] {
461 p.Locked = locked
462 }
463}
464func rerun(s *Simulation, events []Event, policy string, frames int) *Simulation {
465 c := s.Config
466 c.Policy, c.Frames = policy, frames
467 refs := [][2]int{}
468 for _, e := range events {
469 if e.Kind == "ACCESS" {
470 refs = append(refs, [2]int{e.PID, e.Address / c.Page})
471 }
472 }
473 r := newSimulation(c, refs)
474 for _, id := range s.pids() {
475 r.declare(id, len(s.Processes[id].Pages))
476 }
477 pos := 0
478 for _, e := range events {
479 switch e.Kind {
480 case "ACCESS":
481 r.access(e.PID, e.Op, e.Address, pos)
482 pos++
483 case "EXIT":
484 r.exit(e.PID)
485 case "CLEAN":
486 r.clean()
487 case "LOCK":
488 r.lock(e.PID, e.First, e.Last, e.Locked)
489 }
490 }
491 return r
492}
493func command(s *Simulation, words []string, position int, trace *bool, events *[]Event) {
494 c := s.Config
495 cmd, v := words[0], words[1:]
496 n := func(i int) int { return number(v[i]) }
497 switch cmd {
498 case "PROCESS":
499 pid, count := n(0), n(1)
500 if position > 0 {
501 fail("processes are fixed")
502 }
503 if s.Processes[pid] != nil {
504 fail(fmt.Sprintf("duplicate process %d", pid))
505 }
506 if count < 1 || count > 256 {
507 fail("invalid page count")
508 }
509 if len(s.Processes) >= 8 || (c.Scope == "LOCAL" && len(s.Processes) >= c.Frames) {
510 fail("too many processes")
511 }
512 s.declare(pid, count)
513 case "EXIT":
514 p := s.process(n(0))
515 f, sl := s.exit(p.PID)
516 *events = append(*events, Event{Kind: "EXIT", PID: p.PID})
517 fmt.Printf("EXIT P%d freed=%d slots=%d\n", p.PID, f, sl)
518 case "PROTECT", "LOCK", "UNLOCK":
519 p := s.process(n(0))
520 first, last := n(1), n(2)
521 if first > last || last >= len(p.Pages) {
522 fail("invalid page range")
523 }
524 if cmd == "PROTECT" {
525 for _, pg := range p.Pages[first : last+1] {
526 pg.RO = v[3] == "RO"
527 }
528 } else {
529 s.lock(p.PID, first, last, cmd == "LOCK")
530 *events = append(*events, Event{Kind: "LOCK", PID: p.PID, First: first, Last: last, Locked: cmd == "LOCK"})
531 }
532 case "TRACE":
533 *trace = v[0] == "ON"
534 case "SUMMARY":
535 fmt.Printf("SUMMARY policy=%s scope=%s time=%d\n", c.Policy, c.Scope, s.Time)
536 seg, prot := 0, 0
537 for _, p := range s.Processes {
538 seg += p.Seg
539 prot += p.Prot
540 }
541 fmt.Printf("total faults=%d zero=%d swapin=%d faultrate=%s tlbhits=%d tlbrate=%s writebacks=%d switches=%d segfaults=%d protfaults=%d\n", s.Faults, s.Zeros, s.Swapins, ratio(s.Faults, s.Time), s.Hits, ratio(s.Hits, s.Time), s.Writes, s.Switches, seg, prot)
542 for _, id := range s.pids() {
543 p := s.Processes[id]
544 fmt.Printf("P%d accesses=%d faults=%d writebacks=%d segfaults=%d protfaults=%d resident=%d peak=%d", id, len(p.History), p.Faults, p.Writes, p.Seg, p.Prot, p.Resident, p.Peak)
545 if c.Scope == "LOCAL" {
546 fmt.Printf(" quota=%d", s.quota(id))
547 }
548 if p.Exited {
549 fmt.Print(" exited")
550 }
551 fmt.Println()
552 }
553 case "FRAMES":
554 used := 0
555 for _, p := range s.Frames {
556 if p != nil {
557 used++
558 }
559 }
560 fmt.Printf("FRAMES used=%d/%d\n", used, c.Frames)
561 for f, p := range s.Frames {
562 if p == nil {
563 fmt.Printf("%d free\n", f)
564 continue
565 }
566 fmt.Printf("%d %s R=%d M=%d load=%d last=%d uses=%d age=%d", f, label(p), p.R, p.M, p.Load, p.Last, p.Uses, p.Age)
567 if p.Locked {
568 fmt.Print(" locked")
569 }
570 fmt.Println()
571 }
572 if contains([]string{"CLOCK", "ECLOCK", "WSCLOCK"}, c.Policy) {
573 if c.Scope == "GLOBAL" {
574 fmt.Printf("hand=%d\n", s.Hand)
575 } else {
576 fmt.Print("hand")
577 for _, id := range s.pids() {
578 fmt.Printf(" P%d=%d", id, s.Processes[id].Hand)
579 }
580 fmt.Println()
581 }
582 }
583 case "PAGETABLE":
584 p := s.process(n(0))
585 fmt.Printf("PAGETABLE P%d pages=%d resident=%d\n", p.PID, len(p.Pages), p.Resident)
586 for _, pg := range p.Pages {
587 if !pg.Seen {
588 continue
589 }
590 if pg.Frame >= 0 {
591 fmt.Printf("%d frame=%d R=%d M=%d", pg.Number, pg.Frame, pg.R, pg.M)
592 } else if pg.Slot >= 0 {
593 fmt.Printf("%d swap=%d", pg.Number, pg.Slot)
594 } else {
595 fmt.Printf("%d zero", pg.Number)
596 }
597 if pg.RO {
598 fmt.Print(" RO")
599 }
600 if pg.Locked {
601 fmt.Print(" locked")
602 }
603 fmt.Println()
604 }
605 case "TLB":
606 used := 0
607 for _, set := range s.TLB {
608 used += len(set)
609 }
610 fmt.Printf("TLB used=%d/%d sets=%d\n", used, c.TLB, c.Sets)
611 for i, set := range s.TLB {
612 fmt.Printf("set %d:", i)
613 for _, p := range set {
614 fmt.Printf(" %s->%d", label(p), p.Frame)
615 }
616 fmt.Println()
617 }
618 case "SWAP":
619 used := 0
620 for _, p := range s.Slots {
621 if p != nil {
622 used++
623 }
624 }
625 fmt.Printf("SWAP used=%d\n", used)
626 for i, p := range s.Slots {
627 if p != nil {
628 fmt.Printf("%d %s\n", i, label(p))
629 }
630 }
631 case "CLEAN":
632 fmt.Printf("CLEAN written=%d\n", s.clean())
633 *events = append(*events, Event{Kind: "CLEAN"})
634 case "COMPARE":
635 fmt.Printf("COMPARE accesses=%d frames=%d\n", s.Time, c.Frames)
636 results := make([]*Simulation, len(policies))
637 order := make([]int, len(policies))
638 for i, p := range policies {
639 r := rerun(s, *events, p, c.Frames)
640 results[i], order[i] = r, i
641 fmt.Printf("%s faults=%d writebacks=%d tlbhits=%d\n", p, r.Faults, r.Writes, r.Hits)
642 }
643 sort.SliceStable(order, func(i, j int) bool {
644 a, b := results[order[i]], results[order[j]]
645 return a.Faults < b.Faults || (a.Faults == b.Faults && a.Writes < b.Writes)
646 })
647 fmt.Print("ranking")
648 for _, i := range order {
649 fmt.Print(" ", policies[i])
650 }
651 fmt.Println()
652 case "CURVE":
653 if !contains(policies, v[0]) {
654 fail("unknown policy")
655 }
656 lo, hi := n(1), n(2)
657 if lo < 1 || lo > hi || hi > 16 || (c.Scope == "LOCAL" && lo < len(s.Processes)) {
658 fail("invalid frame range")
659 }
660 fmt.Printf("CURVE %s frames=%d..%d\n", v[0], lo, hi)
661 previous := -1
662 for f := lo; f <= hi; f++ {
663 faults := rerun(s, *events, v[0], f).Faults
664 fmt.Printf("F=%d faults=%d", f, faults)
665 if previous >= 0 && faults > previous {
666 fmt.Print(" ANOMALY")
667 }
668 fmt.Println()
669 previous = faults
670 }
671 case "EAT":
672 if s.Time == 0 {
673 fmt.Println("EAT accesses=0")
674 return
675 }
676 total := int64(n(0))*int64(s.Time) + int64(n(1))*int64(2*s.Time-s.Hits) + int64(n(2))*int64(s.Swapins+s.Writes)
677 average := 100 * total / int64(s.Time)
678 fmt.Printf("EAT accesses=%d total=%d average=%d.%02d\n", s.Time, total, average/100, average%100)
679 case "TIMELINE":
680 w := n(0)
681 if w < 1 {
682 fail("invalid window")
683 }
684 fmt.Printf("TIMELINE window=%d", w)
685 for i := 0; i < len(s.FaultHistory); i += w {
686 total := 0
687 for j := i; j < i+w && j < len(s.FaultHistory); j++ {
688 total += s.FaultHistory[j]
689 }
690 fmt.Print(" ", total)
691 }
692 fmt.Println()
693 case "WSET", "HOT":
694 p := s.process(n(0))
695 k := n(1)
696 if k < 1 {
697 if cmd == "WSET" {
698 fail("invalid window")
699 }
700 fail("invalid count")
701 }
702 counts := map[int]int{}
703 history := p.History
704 if cmd == "WSET" && len(history) > k {
705 history = history[len(history)-k:]
706 }
707 for _, pg := range history {
708 counts[pg]++
709 }
710 pages := []int{}
711 for pg := range counts {
712 pages = append(pages, pg)
713 }
714 sort.Ints(pages)
715 if cmd == "WSET" {
716 fmt.Printf("WSET P%d window=%d size=%d", p.PID, k, len(pages))
717 for _, pg := range pages {
718 fmt.Print(" ", pg)
719 }
720 } else {
721 sort.SliceStable(pages, func(i, j int) bool { return counts[pages[i]] > counts[pages[j]] })
722 fmt.Printf("HOT P%d", p.PID)
723 for i, pg := range pages {
724 if i >= k {
725 break
726 }
727 fmt.Printf(" %d:%d", pg, counts[pg])
728 }
729 }
730 fmt.Println()
731 }
732}
733func expand(lines []string) []Item {
734 items := []Item{}
735 add := func(pid int, op string, addr int) {
736 items = append(items, Item{Kind: "ACCESS", PID: pid, Op: op, Address: addr})
737 }
738 for _, line := range lines {
739 w := strings.Fields(line)
740 switch w[0] {
741 case "LOOP":
742 for r := 0; r < number(w[6]); r++ {
743 for i := 0; i < number(w[5]); i++ {
744 add(number(w[1]), w[2], address(w[3])+i*number(w[4]))
745 }
746 }
747 case "RAND":
748 x := int64(number(w[5]))
749 lo, hi := address(w[2]), address(w[3])
750 for i := 0; i < number(w[4]); i++ {
751 x = x * 48271 % 2147483647
752 addr := lo + int(x%int64(hi-lo))
753 x = x * 48271 % 2147483647
754 op := "R"
755 if x%100 < int64(number(w[6])) {
756 op = "W"
757 }
758 add(number(w[1]), op, addr)
759 }
760 default:
761 if w[0][0] >= '0' && w[0][0] <= '9' {
762 add(number(w[0]), w[1], address(w[2]))
763 } else {
764 items = append(items, Item{Kind: w[0], Words: w})
765 }
766 }
767 }
768 return items
769}
770func main() {
771 reader := bufio.NewReader(os.Stdin)
772 readLine := func() string {
773 line, _ := reader.ReadString('\n')
774 return strings.TrimSuffix(strings.TrimSuffix(line, "\n"), "\r")
775 }
776 n := number(readLine())
777 lines := make([]string, n)
778 for i := range lines {
779 lines[i] = readLine()
780 }
781 items := expand(lines)
782 var s *Simulation
783 trace := false
784 position := 0
785 events := []Event{}
786 for index, item := range items {
787 func() {
788 defer func() {
789 if e := recover(); e != nil {
790 if msg, ok := e.(string); ok {
791 fmt.Println("ERROR", msg)
792 } else {
793 panic(e)
794 }
795 }
796 }()
797 if item.Kind == "SYSTEM" {
798 s = nil
799 c := settings(item.Words[1:])
800 refs := [][2]int{}
801 for _, future := range items[index+1:] {
802 if future.Kind == "SYSTEM" {
803 break
804 }
805 if future.Kind == "ACCESS" {
806 refs = append(refs, [2]int{future.PID, future.Address / c.Page})
807 }
808 }
809 s = newSimulation(c, refs)
810 trace, position, events = false, 0, []Event{}
811 fmt.Printf("SYSTEM page=%d frames=%d tlb=%d sets=%d policy=%s scope=%s asid=%s tick=%d tau=%d seed=%d\n", c.Page, c.Frames, c.TLB, c.Sets, c.Policy, c.Scope, c.ASID, c.Tick, c.Tau, c.Seed)
812 return
813 }
814 if s == nil {
815 fail("no system")
816 }
817 if item.Kind == "ACCESS" {
818 pos := position
819 position++
820 p := s.process(item.PID)
821 if item.Address >= len(p.Pages)*s.Config.Page {
822 p.Seg++
823 fmt.Printf("SEGFAULT P%d %s\n", item.PID, hex(item.Address))
824 return
825 }
826 if item.Op == "W" && p.Pages[item.Address/s.Config.Page].RO {
827 p.Prot++
828 fmt.Printf("PROTFAULT P%d %s\n", item.PID, hex(item.Address))
829 return
830 }
831 output := s.access(item.PID, item.Op, item.Address, pos)
832 events = append(events, Event{Kind: "ACCESS", PID: item.PID, Op: item.Op, Address: item.Address})
833 if trace {
834 for _, line := range output {
835 fmt.Println(line)
836 }
837 }
838 } else {
839 command(s, item.Words, position, &trace, &events)
840 }
841 }()
842 }
