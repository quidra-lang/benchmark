var policies = []string{"FIFO", "LRU", "LFU", "NRU", "AGING", "CLOCK", "ECLOCK", "WSCLOCK", "RANDOM", "OPT"}
type Settings struct {
	Page, Frames, TLB, Sets, Tick, Tau int
	Seed                               int64
	Policy, Scope, ASID                string
}
type Simulation struct {
	Config                                                                        Settings
	Processes                                                                     map[int]*Process
	Frames, Slots                                                                 []*Page
	TLB                                                                           [][]*Page
	Refs                                                                          [][2]int
	Position, Time, Faults, Zeros, Swapins, Hits, Writes, Switches, LastPID, Hand int
	Random                                                                        int64
	FaultHistory                                                                  []int
}
func settings(words []string) Settings {
	c := Settings{256, 4, 4, 1, 0, 8, 1, "FIFO", "GLOBAL", "OFF"}
	known := []string{"page", "frames", "tlb", "sets", "policy", "scope", "asid", "tick", "tau", "seed"}
	seen := map[string]bool{}
	for _, word := range words {
		kv := strings.SplitN(word, "=", 2)
		k, v := kv[0], kv[1]
		if !contains(known, k) {
			fail("unknown key " + k)
		}
		if seen[k] {
			fail("duplicate key " + k)
		}
		seen[k] = true
		n64, err := strconv.ParseInt(v, 10, 64)
		n := int(n64)
		valid := err == nil
		switch k {
		case "page":
			valid = valid && n >= 16 && n <= 4096 && n&(n-1) == 0
			c.Page = n
		case "frames":
			valid = valid && n >= 1 && n <= 16
			c.Frames = n
		case "tlb":
			valid = valid && n >= 1 && n <= 8
			c.TLB = n
		case "sets":
			valid = valid && n >= 1 && n <= 8
			c.Sets = n
		case "tick":
			valid = valid && n >= 0 && n <= 1000
			c.Tick = n
		case "tau":
			valid = valid && n >= 1 && n <= 1000
			c.Tau = n
		case "seed":
			valid = valid && n64 >= 1 && n64 <= 2147483646
			c.Seed = n64
		case "policy":
			valid = contains(policies, v)
			c.Policy = v
		case "scope":
			valid = v == "GLOBAL" || v == "LOCAL"
			c.Scope = v
		case "asid":
			valid = v == "ON" || v == "OFF"
			c.ASID = v
		}
		if !valid {
			fail("invalid " + k)
		}
	}
	if c.TLB%c.Sets != 0 {
		fail("invalid sets")
	}
	return c
}

	return &Simulation{Config: c, Processes: map[int]*Process{}, Frames: make([]*Page, c.Frames), TLB: make([][]*Page, c.Sets), Refs: refs, Random: c.Seed}
