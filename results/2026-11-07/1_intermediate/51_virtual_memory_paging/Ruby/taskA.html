2DEFAULTS = { 'page'=>256, 'frames'=>4, 'tlb'=>4, 'sets'=>1, 'policy'=>'FIFO',
3 'scope'=>'GLOBAL', 'asid'=>'OFF', 'tick'=>0, 'tau'=>8, 'seed'=>1 }.freeze

5def ratio(x, y)
6 v = y.zero? ? 0 : 10_000 * x / y
7 format('%d.%02d%%', v / 100, v % 100)
8end

10def address(n) = '0x' + n.to_s(16).upcase
11
12class Simulation
13 attr_reader :config, :processes, :time, :faults, :writebacks, :tlbhits, :events
14 def initialize(config, references)
15 @config, @references = config, references
16 @processes, @events, @history, @swap = {}, [], [], {}
17 @frames = Array.new(config['frames'])
18 @tlb = Array.new(config['sets']) { [] }
19 @time = @faults = @writebacks = @tlbhits = @zero = @swapin = @switches = 0
20 @cursor = @hand = 0
21 @random = config['seed']
22 @fixed = @trace = false
23 @previous = nil
24 end
25 def declare(pid, count)
26 raise 'processes are fixed' if @fixed
27 raise "duplicate process #{pid}" if @processes.key?(pid)
28 raise 'invalid page count' unless (1..256).cover?(count)
29 if @processes.length >= 8 || (@config['scope'] == 'LOCAL' && @processes.length >= @frames.length)
30 raise 'too many processes'
31 end
32 pages = Array.new(count) do |i|
33 { pid: pid, p: i, frame: nil, slot: nil, ro: false, locked: false, r: 0, m: 0,
34 load: 0, last: 0, uses: 0, age: 0, ever: false }
35 end
36 @processes[pid] = { pages: pages, accesses: 0, faults: 0, writebacks: 0,
37 segfaults: 0, protfaults: 0, peak: 0, exited: false, hand: 0, refs: [] }
38 end
39 def process(pid)
40 p = @processes[pid]
41 raise "unknown process #{pid}" unless p
42 raise "process #{pid} exited" if p[:exited]
43 p
44 end
45 def resident(pid) = @frames.count { |p| p && p[:pid] == pid }
46 def quota(pid)
47 ids = @processes.keys.sort
48 q, r = @frames.length.divmod(ids.length)
49 q + (ids.index(pid) < r ? 1 : 0)
50 end
51 def invalidate(page)
52 @tlb.each { |set| set.reject! { |p| p.equal?(page) } }
53 end
54 def writeback(page)
55 @writebacks += 1
56 @processes[page[:pid]][:writebacks] += 1
57 page[:m] = 0
58 unless page[:slot]
59 slot = 0
60 slot += 1 while @swap.key?(slot)
61 page[:slot] = slot
62 @swap[slot] = page
63 end
64 end
65 def victim(pid)
66 candidates = @frames.each_index.select do |i|
67 @frames[i] && (@config['scope'] == 'GLOBAL' || @frames[i][:pid] == pid)
68 end
69 unlocked = candidates.reject { |i| @frames[i][:locked] }
70 candidates = unlocked unless unlocked.empty?
71 policy = @config['policy']
72 if policy == 'RANDOM'
73 @random = 48_271 * @random % 2_147_483_647
74 return candidates[@random % candidates.length]
75 end
76 if policy == 'OPT'
77 return candidates.max_by do |i|
78 page = @frames[i]
79 next_use = (@cursor...@references.length).find do |j|
80 ref = @references[j]
81 ref[0] == page[:pid] && ref[2] / @config['page'] == page[:p]
82 end
83 [next_use || Float::INFINITY, -i]
84 end
85 end
86 keys = { 'FIFO'=>:load, 'LRU'=>:last, 'LFU'=>:uses, 'AGING'=>:age }
87 if keys.key?(policy)
88 return candidates.min_by do |i|
89 page = @frames[i]
90 [page[keys[policy]], page[:load]]
91 end
92 end
93 return candidates.min_by { |i| [2 * @frames[i][:r] + @frames[i][:m], i] } if policy == 'NRU'
94 hand = @config['scope'] == 'GLOBAL' ? @hand : @processes[pid][:hand]
95 circular = candidates.sort_by { |i| (i - hand) % @frames.length }
96 chosen = nil
97 case policy
98 when 'CLOCK'
99 until chosen
100 circular.each do |i|
101 if @frames[i][:r].zero?
102 chosen = i
103 break
104 end
105 @frames[i][:r] = 0
106 end
107 end
108 when 'ECLOCK'
109 until chosen
110 chosen = circular.find { |i| @frames[i][:r].zero? && @frames[i][:m].zero? }
111 break if chosen
112 circular.each do |i|
113 if @frames[i][:r].zero? && @frames[i][:m] == 1
114 chosen = i
115 break
116 end
117 @frames[i][:r] = 0
118 end
119 end
120 when 'WSCLOCK'
121 circular.each do |i|
122 page = @frames[i]
123 if page[:r] == 1
124 page[:r], page[:last] = 0, @time
125 elsif @time - page[:last] > @config['tau']
126 if page[:m].zero?
127 chosen = i
128 break
129 end
130 writeback(page)
131 end
132 end
133 chosen ||= circular.find { |i| @frames[i][:m].zero? } || circular.first
134 end
135 if @config['scope'] == 'GLOBAL'
136 @hand = (chosen + 1) % @frames.length
137 else
138 @processes[pid][:hand] = (chosen + 1) % @frames.length
139 end
140 chosen
141 end
142 def access(pid, op, addr, silent = false)
143 @fixed = true
144 @cursor += 1
145 proc = process(pid)
146 if addr >= proc[:pages].length * @config['page']
147 proc[:segfaults] += 1
148 puts "SEGFAULT P#{pid} #{address(addr)}" unless silent
149 return
150 end
151 page = proc[:pages][addr / @config['page']]
152 if op == 'W' && page[:ro]
153 proc[:protfaults] += 1
154 puts "PROTFAULT P#{pid} #{address(addr)}" unless silent
155 return
156 end
157 @events << [:access, pid, op, addr]
158 @time += 1
159 proc[:accesses] += 1
160 proc[:refs] << page[:p]
161 switched = !@previous.nil? && @previous != pid
162 if switched
163 @switches += 1
164 @tlb.each(&:clear) if @config['asid'] == 'OFF'
165 end
166 @previous = pid
167 set = @tlb[page[:p] % @config['sets']]
168 fault = false
169 victim_page = nil
170 wrote = false
171 if set.include?(page)
172 result = 'TLBHIT'
173 @tlbhits += 1
174 set.delete(page)
175 else
176 if page[:frame]
177 result = 'HIT'
178 else
179 fault = true
180 @faults += 1
181 proc[:faults] += 1
182 if page[:slot]
183 @swapin += 1
184 result = 'FAULT SWAPIN'
185 else
186 @zero += 1
187 result = 'FAULT ZERO'
188 end
189 free = @frames.index(nil)
190 use_free = @config['scope'] == 'GLOBAL' ? !free.nil? : resident(pid) < quota(pid)
191 frame = use_free ? free : victim(pid)
192 victim_page = @frames[frame]
193 if victim_page
194 invalidate(victim_page)
195 wrote = victim_page[:m] == 1
196 writeback(victim_page) if wrote
197 victim_page[:frame] = nil
198 end
199 @frames[frame] = page
200 page[:frame], page[:load], page[:uses], page[:m], page[:age] = frame, @time, 0, 0, 0
201 proc[:peak] = [proc[:peak], resident(pid)].max
202 end
203 set.shift if set.length == @config['tlb'] / @config['sets']
204 end
205 set << page
206 page[:r] = 1
207 page[:m] = 1 if op == 'W'
208 page[:last] = @time
209 page[:uses] += 1
210 page[:ever] = true
211 @history << fault
212 tick = @config['tick'].positive? && @time % @config['tick'] == 0
213 if tick
214 @frames.compact.each do |p|
215 p[:age] = p[:age] / 2 + 128 * p[:r]
216 p[:r] = 0
217 end
218 end
219 return unless @trace && !silent
220 text = "t=#{@time} P#{pid} #{op} #{address(addr)} page=#{page[:p]} frame=#{page[:frame]} phys=#{address(page[:frame] * @config['page'] + addr % @config['page'])} #{result}"
221 text += " evict=P#{victim_page[:pid]}:#{victim_page[:p]}" if victim_page
222 text += ' writeback' if wrote
223 text += ' switch' if switched
224 puts text
225 puts "t=#{@time} tick" if tick
226 end
227 def exit_process(pid, silent = false)
228 proc = process(pid)
229 freed, slots = resident(pid), 0
230 proc[:pages].each do |page|
231 @frames[page[:frame]] = nil if page[:frame]
232 page[:frame] = nil
233 invalidate(page)
234 unless page[:slot].nil?
235 @swap.delete(page[:slot])
236 page[:slot] = nil
237 slots += 1
238 end
239 end
240 proc[:exited] = true
241 @events << [:exit, pid]
242 puts "EXIT P#{pid} freed=#{freed} slots=#{slots}" unless silent
243 end
244 def attributes(cmd, pid, first, last, protection = nil)
245 proc = process(pid)
246 raise 'invalid page range' if first > last || last >= proc[:pages].length
247 (first..last).each do |i|
248 page = proc[:pages][i]
249 if cmd == 'PROTECT'
250 page[:ro] = protection == 'RO'
251 else
252 page[:locked] = cmd == 'LOCK'
253 end
254 end
255 @events << [:attribute, cmd, pid, first, last] unless cmd == 'PROTECT'
256 end
257 def clean(silent = false)
258 pages = @frames.compact.select { |p| p[:m] == 1 }
259 pages.each { |p| writeback(p) }
260 @events << [:clean]
261 puts "CLEAN written=#{pages.length}" unless silent
262 end
263 def rerun(policy, frames)
264 config = @config.merge('policy'=>policy, 'frames'=>frames)
265 refs = @events.select { |e| e[0] == :access }.map { |e| e[1..] }
266 sim = Simulation.new(config, refs)
267 @processes.each { |pid,p| sim.declare(pid, p[:pages].length) }
268 @events.each do |e|
269 case e[0]
270 when :access then sim.access(*e[1..], true)
271 when :exit then sim.exit_process(e[1], true)
272 when :clean then sim.clean(true)
273 when :attribute then sim.attributes(*e[1..])
274 end
275 end
276 sim
277 end
278 def command(words)
279 cmd = words[0]
280 pid = words[1].to_i
281 case cmd
282 when 'PROCESS' then declare(pid, words[2].to_i)
283 when 'EXIT' then exit_process(pid)
284 when 'PROTECT', 'LOCK', 'UNLOCK' then attributes(cmd, pid, words[2].to_i, words[3].to_i, words[4])
285 when 'TRACE' then @trace = words[1] == 'ON'
286 when 'CLEAN' then clean
287 when 'SUMMARY'
288 puts "SUMMARY policy=#{@config['policy']} scope=#{@config['scope']} time=#{@time}"
289 seg = @processes.values.sum { |p| p[:segfaults] }
290 prot = @processes.values.sum { |p| p[:protfaults] }
291 puts "total faults=#{@faults} zero=#{@zero} swapin=#{@swapin} faultrate=#{ratio(@faults,@time)} tlbhits=#{@tlbhits} tlbrate=#{ratio(@tlbhits,@time)} writebacks=#{@writebacks} switches=#{@switches} segfaults=#{seg} protfaults=#{prot}"
292 @processes.keys.sort.each do |id|
293 p = @processes[id]
294 line = "P#{id} accesses=#{p[:accesses]} faults=#{p[:faults]} writebacks=#{p[:writebacks]} segfaults=#{p[:segfaults]} protfaults=#{p[:protfaults]} resident=#{resident(id)} peak=#{p[:peak]}"
295 line += " quota=#{quota(id)}" if @config['scope'] == 'LOCAL'
296 line += ' exited' if p[:exited]
297 puts line
298 end
299 when 'FRAMES'
300 puts "FRAMES used=#{@frames.compact.length}/#{@frames.length}"
301 @frames.each_with_index do |p,i|
302 if p
303 line = "#{i} P#{p[:pid]}:#{p[:p]} R=#{p[:r]} M=#{p[:m]} load=#{p[:load]} last=#{p[:last]} uses=#{p[:uses]} age=#{p[:age]}"
304 puts line + (p[:locked] ? ' locked' : '')
305 else
306 puts "#{i} free"
307 end
308 end
309 if %w[CLOCK ECLOCK WSCLOCK].include?(@config['policy'])
310 puts(@config['scope'] == 'GLOBAL' ? "hand=#{@hand}" : 'hand' + @processes.keys.sort.map { |id| " P#{id}=#{@processes[id][:hand]}" }.join)
311 end
312 when 'PAGETABLE'
313 p = process(pid)
314 puts "PAGETABLE P#{pid} pages=#{p[:pages].length} resident=#{resident(pid)}"
315 p[:pages].select { |pg| pg[:ever] }.each do |pg|
316 state = if pg[:frame]
317 "frame=#{pg[:frame]} R=#{pg[:r]} M=#{pg[:m]}"
318 elsif pg[:slot]
319 "swap=#{pg[:slot]}"
320 else 'zero' end
321 puts "#{pg[:p]} #{state}" + (pg[:ro] ? ' RO' : '') + (pg[:locked] ? ' locked' : '')
322 end
323 when 'TLB'
324 puts "TLB used=#{@tlb.sum(&:length)}/#{@config['tlb']} sets=#{@config['sets']}"
325 @tlb.each_with_index { |set,i| puts "set #{i}:" + set.reverse.map { |p| " P#{p[:pid]}:#{p[:p]}->#{p[:frame]}" }.join }
326 when 'SWAP'
327 puts "SWAP used=#{@swap.length}"
328 @swap.keys.sort.each { |slot| p=@swap[slot]; puts "#{slot} P#{p[:pid]}:#{p[:p]}" }
329 when 'COMPARE'
330 puts "COMPARE accesses=#{@time} frames=#{@frames.length}"
331 results = POLICIES.each_with_index.map do |policy,i|
332 sim = rerun(policy, @frames.length)
333 puts "#{policy} faults=#{sim.faults} writebacks=#{sim.writebacks} tlbhits=#{sim.tlbhits}"
334 [sim.faults, sim.writebacks, i, policy]
335 end
336 puts 'ranking ' + results.sort.map(&:last).join(' ')
337 when 'CURVE'
338 policy, lo, hi = words[1], words[2].to_i, words[3].to_i
339 raise 'unknown policy' unless POLICIES.include?(policy)
340 if lo < 1 || lo > hi || hi > 16 || (@config['scope'] == 'LOCAL' && lo < @processes.length)
341 raise 'invalid frame range'
342 end
343 puts "CURVE #{policy} frames=#{lo}..#{hi}"
344 previous = nil
345 (lo..hi).each do |f|
346 faults = rerun(policy, f).faults
347 puts "F=#{f} faults=#{faults}" + (previous && faults > previous ? ' ANOMALY' : '')
348 previous = faults
349 end
350 when 'EAT'
351 if @time.zero?
352 puts 'EAT accesses=0'
353 else
354 a,b,c = words[1..].map(&:to_i)
355 total = a*@time+b*(2*@time-@tlbhits)+c*(@swapin+@writebacks)
356 v = 100*total/@time
357 puts "EAT accesses=#{@time} total=#{total} average=#{format('%d.%02d',v/100,v%100)}"
358 end
359 when 'TIMELINE'
360 w = pid
361 raise 'invalid window' if w < 1
362 puts "TIMELINE window=#{w}" + @history.each_slice(w).map { |slice| " #{slice.count(true)}" }.join
363 when 'WSET'
364 p = process(pid)
365 w = words[2].to_i
366 raise 'invalid window' if w < 1
367 pages = p[:refs].last(w).uniq.sort
368 puts "WSET P#{pid} window=#{w} size=#{pages.length}" + pages.map { |i| " #{i}" }.join
369 when 'HOT'
370 p = process(pid)
371 k = words[2].to_i
372 raise 'invalid count' if k < 1
373 pages = p[:refs].tally.sort_by { |page,count| [-count,page] }.first(k)
374 puts "HOT P#{pid}" + pages.map { |page,count| " #{page}:#{count}" }.join
375 else
376 access(words[0].to_i, words[1], words[2].to_i(16))
377 end
378 end
379end

381def system_settings(words)
382 config = DEFAULTS.dup
383 seen = {}
384 words.drop(1).each do |pair|
385 key,value = pair.split('=')
386 raise "unknown key #{key}" unless config.key?(key)
387 raise "duplicate key #{key}" if seen[key]
388 seen[key] = true
389 valid = case key
390 when 'policy' then POLICIES.include?(value)
391 when 'scope' then %w[GLOBAL LOCAL].include?(value)
392 when 'asid' then %w[ON OFF].include?(value)
393 else
394 n = value.to_i
395 ranges = { 'frames'=>1..16, 'tlb'=>1..8, 'sets'=>1..8, 'tick'=>0..1000,
396 'tau'=>1..1000, 'seed'=>1..2_147_483_646 }
397 key == 'page' ? n >= 16 && n <= 4096 && (n & (n-1)).zero? : ranges[key].cover?(n)
398 end
399 raise "invalid #{key}" unless valid
400 config[key] = %w[policy scope asid].include?(key) ? value : value.to_i
401 end
402 raise 'invalid sets' unless (config['tlb'] % config['sets']).zero?
403 config
404end

406commands = []
407STDIN.gets.to_i.times do
408 w = STDIN.gets.split
409 case w[0]
410 when 'LOOP'
411 pid,op,start,step,n,repeats = w[1].to_i,w[2],w[3].to_i(16),w[4].to_i,w[5].to_i,w[6].to_i
412 repeats.times { n.times { |i| commands << [pid.to_s,op,address(start+i*step)] } }
413 when 'RAND'
414 pid,lo,hi,count,x,rate = w[1].to_i,w[2].to_i(16),w[3].to_i(16),w[4].to_i,w[5].to_i,w[6].to_i
415 count.times do
416 x = 48_271*x % 2_147_483_647
417 addr = lo+x%(hi-lo)
418 x = 48_271*x % 2_147_483_647
419 commands << [pid.to_s,x%100 < rate ? 'W' : 'R',address(addr)]
420 end
421 else commands << w
422 end
423end
424simulation = nil
425commands.each_with_index do |words,i|
426 begin
427 if words[0] == 'SYSTEM'
428 simulation = nil
429 config = system_settings(words)
430 rest = commands[(i+1)..].take_while { |w| w[0] != 'SYSTEM' }
431 refs = rest.select { |w| w[0].match?(/\A\d/) }.map { |w| [w[0].to_i,w[1],w[2].to_i(16)] }
432 simulation = Simulation.new(config,refs)
433 puts 'SYSTEM ' + config.map { |key,value| "#{key}=#{value}" }.join(' ')
434 elsif simulation
435 simulation.command(words)
436 else
437 puts 'ERROR no system'
438 end
439 rescue RuntimeError => error
440 puts "ERROR #{error.message}"
441 end
