10}

17fn lookup(a: Text, names: []const Text) ?usize {
18 for (names, 0..) |b, j| if (eq(a, b)) return j;
19 return null;

38const Settings = struct {
39 page: I = 256,
40 frames: usize = 4,
41 tlb: usize = 4,
42 sets: usize = 1,
43 policy: usize = 0,
44 local: bool = false,
45 asid: bool = false,
46 tick: I = 0,
47 tau: I = 8,
48 seed: I = 1,
49 fn scope(c: Settings) Text {
50 return if (c.local) "LOCAL" else "GLOBAL";
51 }
52};
53const ConfigResult = union(enum) { settings: Settings, problem: Text };
54fn parseSettings(h: Host, tokens: []const Text) !ConfigResult {
55 const keys = [_]Text{ "page", "frames", "tlb", "sets", "policy", "scope", "asid", "tick", "tau", "seed" };
56 const lower = [_]I{ 16, 1, 1, 1, 0, 0, 0, 0, 1, 1 };
57 const upper = [_]I{ 4096, 16, 8, 8, 0, 0, 0, 1000, 1000, 2147483646 };
58 var c: Settings = .{};
59 var seen: [10]bool = @splat(false);
60 for (tokens[1..]) |word| {
61 var parts = std.mem.splitScalar(u8, word, '=');
62 const key = parts.next().?;
63 const value = parts.next().?;
64 const j = lookup(key, &keys) orelse return .{ .problem = try h.text("unknown key {s}", .{key}) };
65 if (seen[j]) return .{ .problem = try h.text("duplicate key {s}", .{key}) };
66 seen[j] = true;
67 var valid = true;
68 if (j == 4) {
69 if (lookup(value, &policy_names)) |policy| c.policy = policy else valid = false;
70 } else if (j == 5) {
71 valid = eq(value, "GLOBAL") or eq(value, "LOCAL");
72 c.local = eq(value, "LOCAL");
73 } else if (j == 6) {
74 valid = eq(value, "ON") or eq(value, "OFF");
75 c.asid = eq(value, "ON");
76 } else {
77 const n = num(value);
78 valid = lower[j] <= n and n <= upper[j];
79 if (j == 0) valid = valid and n > 0 and (n & (n - 1)) == 0;
80 if (valid) switch (j) {
81 0 => c.page = n,
82 1 => c.frames = ix(n),
83 2 => c.tlb = ix(n),
84 3 => c.sets = ix(n),
85 7 => c.tick = n,
86 8 => c.tau = n,
87 9 => c.seed = n,
88 else => unreachable,
89 };
90 }
91 if (!valid) return .{ .problem = try h.text("invalid {s}", .{key}) };
92 }
93 if (c.tlb % c.sets != 0) return .{ .problem = "invalid sets" };
94 return .{ .settings = c };
95}

161 fn processIndex(s: *Simulation, pid: I) ?usize {
162 for (s.processes.items, 0..) |p, j| if (p.pid == pid) return j;
163 return null;

165 fn live(s: *Simulation, pid: I) !?usize {
166 const owner = s.processIndex(pid) orelse {
167 try s.emit("ERROR unknown process {d}", .{pid});
168 return null;
169 };
170 if (s.processes.items[owner].exited) {
171 try s.emit("ERROR process {d} exited", .{pid});
172 return null;
173 }
174 return owner;
175 }
176 fn addProcess(s: *Simulation, pid: I, count: usize) !void {
177 const pages = try s.h.gpa.alloc(Page, count);
178 @memset(pages, .{});
179 try s.processes.append(s.h.gpa, .{ .pid = pid, .pages = pages });

181 fn page(s: *Simulation, ref: Reference) *Page {
182 return &s.processes.items[ref.process].pages[ref.page];
183 }
184 fn resident(s: *Simulation, owner: usize) I {
185 var count: I = 0;
186 for (s.frames) |frame| {
187 if (frame) |ref| {
188 if (ref.process == owner) count += 1;
189 }
190 }
191 return count;
192 }
193 fn quota(s: *Simulation, owner: usize) I {
194 var before: usize = 0;
195 for (s.processes.items) |p| if (p.pid < s.processes.items[owner].pid) {
196 before += 1;
197 };
198 const count = s.processes.items.len;
199 return @intCast(s.c.frames / count + @intFromBool(before < s.c.frames % count));
200 }
201 fn invalidate(s: *Simulation, ref: Reference) void {
202 var j: usize = 0;
203 while (j < s.entries.items.len) {
204 const entry = s.entries.items[j];
205 if (entry.process == ref.process and entry.page == ref.page) _ = s.entries.orderedRemove(j) else j += 1;
206 }
207 }
208 fn writeBack(s: *Simulation, ref: Reference) void {
209 s.processes.items[ref.process].writebacks += 1;
210 s.writebacks += 1;
211 s.page(ref).m = 0;
212 if (s.page(ref).slot != null) return;
213 var slot: usize = 0;
214 while (true) : (slot += 1) {
215 var used = false;
216 for (s.processes.items) |p| {
217 for (p.pages) |item| {
218 if (item.slot == slot) used = true;
219 }
220 }
221 if (!used) break;
222 }
223 s.page(ref).slot = slot;
224 }
225 fn nextUse(s: *Simulation, ref: Reference) usize {
226 for (s.cursor..s.future.items.len) |j| {
227 const request = s.future.items[j];
228 if (request.pid == s.processes.items[ref.process].pid and @divTrunc(request.address, s.c.page) == @as(I, @intCast(ref.page))) return j;
229 }
230 return s.future.items.len;
231 }
232 fn choose(s: *Simulation, owner: usize) usize {
233 var candidates: [16]usize = undefined;
234 var count: usize = 0;
235 var unlocked: [16]usize = undefined;
236 var unlocked_count: usize = 0;
237 for (s.frames, 0..) |maybe_ref, j| if (maybe_ref) |ref| {
238 if (!s.c.local or ref.process == owner) {
239 candidates[count] = j;
240 count += 1;
241 if (!s.page(ref).locked) {
242 unlocked[unlocked_count] = j;
243 unlocked_count += 1;
244 }
245 }
246 };
247 if (unlocked_count > 0) {
248 @memcpy(candidates[0..unlocked_count], unlocked[0..unlocked_count]);
249 count = unlocked_count;
250 }
251 if (s.c.policy == 8) {
252 s.random = @mod(48271 * s.random, 2147483647);
253 return candidates[ix(@mod(s.random, @as(I, @intCast(count))))];
254 }
255 if (s.c.policy >= 5 and s.c.policy <= 7) {
256 const hand = if (s.c.local) s.processes.items[owner].hand else s.hand;
257 var circular: [16]usize = undefined;
258 var length: usize = 0;
259 for (0..s.c.frames) |distance| {
260 const frame = (hand + distance) % s.c.frames;
261 for (candidates[0..count]) |candidate| if (candidate == frame) {
262 circular[length] = candidate;
263 length += 1;
264 };
265 }
266 var chosen: ?usize = null;
267 if (s.c.policy == 7) {
268 for (circular[0..length]) |candidate| {
269 const ref = s.frames[candidate].?;
270 const item = s.page(ref);
271 if (item.r == 1) {
272 item.r = 0;
273 item.last = s.time;
274 } else if (s.time - item.last > s.c.tau) {
275 if (item.m == 0) {
276 chosen = candidate;
277 break;
278 }
279 s.writeBack(ref);
280 }
281 }
282 if (chosen == null) {
283 for (circular[0..length]) |candidate| if (s.page(s.frames[candidate].?).m == 0) {
284 chosen = candidate;
285 break;
286 };
287 }
288 if (chosen == null) chosen = circular[0];
289 } else {
290 while (chosen == null) {
291 if (s.c.policy == 6) {
292 for (circular[0..length]) |candidate| {
293 const item = s.page(s.frames[candidate].?);
294 if (item.r == 0 and item.m == 0) {
295 chosen = candidate;
296 break;
297 }
298 }
299 }
300 if (chosen != null) break;
301 for (circular[0..length]) |candidate| {
302 const item = s.page(s.frames[candidate].?);
303 if (item.r == 0 and (s.c.policy == 5 or item.m == 1)) {
304 chosen = candidate;
305 break;
306 }
307 item.r = 0;
308 }
309 }
310 }
311 const result = chosen.?;
312 if (s.c.local) s.processes.items[owner].hand = (result + 1) % s.c.frames else s.hand = (result + 1) % s.c.frames;
313 return result;
314 }
315 var best = candidates[0];
316 for (candidates[0..count]) |candidate| {
317 const current_ref = s.frames[candidate].?;
318 const best_ref = s.frames[best].?;
319 const current = s.page(current_ref);
320 const previous = s.page(best_ref);
321 const better = switch (s.c.policy) {
322 0 => current.load < previous.load,
323 1 => current.last < previous.last,
324 2 => current.uses < previous.uses or (current.uses == previous.uses and current.load < previous.load),
325 3 => 2 * current.r + current.m < 2 * previous.r + previous.m,
326 4 => current.age < previous.age or (current.age == previous.age and current.load < previous.load),
327 9 => s.nextUse(current_ref) > s.nextUse(best_ref),
328 else => unreachable,
329 };
330 if (better) best = candidate;
331 }
332 return best;
333 }
334 fn access(s: *Simulation, request: Command) !void {
335 s.fixed_processes = true;
336 s.cursor += 1;
337 const owner = (try s.live(request.pid)) orelse return;
338 if (request.address >= @as(I, @intCast(s.processes.items[owner].pages.len)) * s.c.page) {
339 s.processes.items[owner].segfaults += 1;
340 return s.emit("SEGFAULT P{d} 0x{X}", .{ request.pid, request.address });
341 }
342 const page_number = ix(@divTrunc(request.address, s.c.page));
343 const ref: Reference = .{ .process = owner, .page = page_number };
344 if (eq(request.op, "W") and s.page(ref).readonly) {
345 s.processes.items[owner].protfaults += 1;
346 return s.emit("PROTFAULT P{d} 0x{X}", .{ request.pid, request.address });
347 }
348 s.time += 1;
349 const switched = s.previous != null and s.previous.? != owner;
350 if (switched) {
351 s.switches += 1;
352 if (!s.c.asid) s.entries.clearRetainingCapacity();
353 }
354 s.previous = owner;
355 var hit: ?usize = null;
356 for (s.entries.items, 0..) |entry, j| if (entry.process == owner and entry.page == page_number) {
357 hit = j;
358 break;
359 };
360 var fault = false;
361 var result: Text = "HIT";
362 var eviction: Text = "";
363 if (hit) |j| {
364 s.tlbhits += 1;
365 result = "TLBHIT";
366 const recent = s.entries.orderedRemove(j);
367 try s.entries.append(s.h.gpa, recent);
368 } else {
369 if (s.page(ref).frame == null) {
370 fault = true;
371 s.faults += 1;
372 s.processes.items[owner].faults += 1;
373 const swapped = s.page(ref).slot != null;
374 if (swapped) s.swapins += 1 else s.zeros += 1;
375 result = if (swapped) "FAULT SWAPIN" else "FAULT ZERO";
376 var chosen: ?usize = null;
377 if (!s.c.local or s.resident(owner) < s.quota(owner)) {
378 for (s.frames, 0..) |frame, j| if (frame == null) {
379 chosen = j;
380 break;
381 };
382 }
383 if (chosen == null) {
384 chosen = s.choose(owner);
385 const victim = s.frames[chosen.?].?;
386 const dirty = s.page(victim).m == 1;
387 eviction = try s.h.text(" evict=P{d}:{d}{s}", .{ s.processes.items[victim.process].pid, victim.page, if (dirty) @as(Text, " writeback") else "" });
388 if (dirty) s.writeBack(victim);
389 s.page(victim).frame = null;
390 s.invalidate(victim);
391 }
392 s.frames[chosen.?] = ref;
393 const item = s.page(ref);
394 item.frame = chosen;
395 item.load = s.time;
396 item.uses = 0;
397 item.m = 0;
398 item.age = 0;
399 s.processes.items[owner].peak = @max(s.processes.items[owner].peak, s.resident(owner));
400 }
401 var count: usize = 0;
402 var oldest: ?usize = null;
403 for (s.entries.items, 0..) |entry, j| {
404 if (entry.page % s.c.sets == page_number % s.c.sets) {
405 if (oldest == null) oldest = j;
406 count += 1;
407 }
408 }
409 if (count == s.c.tlb / s.c.sets) _ = s.entries.orderedRemove(oldest.?);
410 try s.entries.append(s.h.gpa, .{ .process = owner, .page = page_number, .frame = s.page(ref).frame.? });
411 }
412 const item = s.page(ref);
413 item.r = 1;
414 if (eq(request.op, "W")) item.m = 1;
415 item.last = s.time;
416 item.uses += 1;
417 item.ever = true;
418 s.processes.items[owner].accesses += 1;
419 try s.accesses.append(s.h.gpa, .{ .process = owner, .page = page_number, .fault = fault });
420 if (!s.silent) try s.actions.append(s.h.gpa, .{ .kind = .access, .pid = request.pid, .op = request.op, .address = request.address });
421 const timer = s.c.tick > 0 and @mod(s.time, s.c.tick) == 0;
422 if (timer) {
423 for (s.frames) |frame| if (frame) |loaded| {
424 const p = s.page(loaded);
425 p.age = @divTrunc(p.age, 2) + 128 * p.r;
426 p.r = 0;
427 };
428 }
429 if (s.trace) {
430 const physical = @as(I, @intCast(item.frame.?)) * s.c.page + @mod(request.address, s.c.page);
431 try s.emit("t={d} P{d} {s} 0x{X} page={d} frame={d} phys=0x{X} {s}{s}{s}", .{ s.time, request.pid, request.op, request.address, page_number, item.frame.?, physical, result, eviction, if (switched) @as(Text, " switch") else "" });
432 if (timer) try s.emit("t={d} tick", .{s.time});
433 }
434 }
435 fn exitProcess(s: *Simulation, owner: usize) !void {
436 const freed = s.resident(owner);
437 var slots: I = 0;
438 for (s.processes.items[owner].pages, 0..) |*item, j| {
439 if (item.slot != null) slots += 1;
440 item.slot = null;
441 item.frame = null;
442 s.invalidate(.{ .process = owner, .page = j });
443 }
444 for (s.frames) |*frame| {
445 if (frame.*) |ref| {
446 if (ref.process == owner) frame.* = null;
447 }
448 }
449 s.processes.items[owner].exited = true;
450 const pid = s.processes.items[owner].pid;
451 if (!s.silent) try s.actions.append(s.h.gpa, .{ .kind = .exit, .pid = pid });
452 try s.emit("EXIT P{d} freed={d} slots={d}", .{ pid, freed, slots });
453 }
454 fn clean(s: *Simulation) !void {
455 var written: I = 0;
456 for (s.frames) |frame| {
457 if (frame) |ref| {
458 if (s.page(ref).m == 1) {
459 s.writeBack(ref);
460 written += 1;
461 }
462 }
463 }
464 if (!s.silent) try s.actions.append(s.h.gpa, .{ .kind = .clean });
465 try s.emit("CLEAN written={d}", .{written});
466 }
467 fn rerun(s: *Simulation, policy: usize, frame_count: usize) !Simulation {
468 var c = s.c;
469 c.policy = policy;
470 c.frames = frame_count;
471 var replay = try Simulation.init(s.h, c);
472 replay.silent = true;
473 for (s.processes.items) |p| try replay.addProcess(p.pid, p.pages.len);
474 for (s.actions.items) |action| if (action.kind == .access) {
475 try replay.future.append(s.h.gpa, accessCommand(action.pid, action.op, action.address));
476 };
477 for (s.actions.items) |action| switch (action.kind) {
478 .access => try replay.access(accessCommand(action.pid, action.op, action.address)),
479 .exit => try replay.exitProcess(replay.processIndex(action.pid).?),
480 .clean => try replay.clean(),
481 .lock, .unlock => {
482 const owner = replay.processIndex(action.pid).?;
483 for (action.first..action.last + 1) |j| replay.processes.items[owner].pages[j].locked = action.kind == .lock;
484 },
485 };
486 return replay;
487 }
488 fn command(s: *Simulation, w: []const Text) !void {
489 const op = w[0];
490 if (eq(op, "PROCESS")) {
491 const pid = num(w[1]);
492 const count = num(w[2]);
493 if (s.fixed_processes) return s.emit("ERROR processes are fixed", .{});
494 if (s.processIndex(pid) != null) return s.emit("ERROR duplicate process {d}", .{pid});
495 if (count < 1 or count > 256) return s.emit("ERROR invalid page count", .{});
496 if (s.processes.items.len == 8 or (s.c.local and s.processes.items.len == s.c.frames)) return s.emit("ERROR too many processes", .{});
497 return s.addProcess(pid, ix(count));
498 }
499 var owner: usize = 0;
500 if (lookup(op, &.{ "EXIT", "PROTECT", "LOCK", "UNLOCK", "PAGETABLE", "WSET", "HOT" }) != null) owner = (try s.live(num(w[1]))) orelse return;
501 if (eq(op, "EXIT")) return s.exitProcess(owner);
502 if (eq(op, "PROTECT") or eq(op, "LOCK") or eq(op, "UNLOCK")) {
503 const first = ix(num(w[2]));
504 const last = ix(num(w[3]));
505 if (first > last or last >= s.processes.items[owner].pages.len) return s.emit("ERROR invalid page range", .{});
506 for (first..last + 1) |j| {
507 if (eq(op, "PROTECT")) {
508 s.processes.items[owner].pages[j].readonly = eq(w[4], "RO");
509 } else {
510 s.processes.items[owner].pages[j].locked = eq(op, "LOCK");
511 }
512 }
513 if (!eq(op, "PROTECT")) try s.actions.append(s.h.gpa, .{ .kind = if (eq(op, "LOCK")) .lock else .unlock, .pid = s.processes.items[owner].pid, .first = first, .last = last });
514 return;
515 }
516 if (eq(op, "TRACE")) {
517 s.trace = eq(w[1], "ON");
518 return;
519 }
520 if (eq(op, "CLEAN")) return s.clean();
521 if (eq(op, "SUMMARY")) {
522 try s.emit("SUMMARY policy={s} scope={s} time={d}", .{ policy_names[s.c.policy], s.c.scope(), s.time });
523 var segfaults: I = 0;
524 var protfaults: I = 0;
525 for (s.processes.items) |p| {
526 segfaults += p.segfaults;
527 protfaults += p.protfaults;
528 }
529 try s.emit("total faults={d} zero={d} swapin={d} faultrate={s} tlbhits={d} tlbrate={s} writebacks={d} switches={d} segfaults={d} protfaults={d}", .{ s.faults, s.zeros, s.swapins, try s.h.ratio(s.faults, s.time), s.tlbhits, try s.h.ratio(s.tlbhits, s.time), s.writebacks, s.switches, segfaults, protfaults });
530 for (1..100) |pid| if (s.processIndex(@intCast(pid))) |j| {
531 const p = s.processes.items[j];
532 const quota_text = if (s.c.local) try s.h.text(" quota={d}", .{s.quota(j)}) else "";
533 try s.emit("P{d} accesses={d} faults={d} writebacks={d} segfaults={d} protfaults={d} resident={d} peak={d}{s}{s}", .{ pid, p.accesses, p.faults, p.writebacks, p.segfaults, p.protfaults, s.resident(j), p.peak, quota_text, if (p.exited) @as(Text, " exited") else "" });
534 };
535 return;
536 }
537 if (eq(op, "FRAMES")) {
538 var used: usize = 0;
539 for (s.frames) |frame| if (frame != null) {
540 used += 1;
541 };
542 try s.emit("FRAMES used={d}/{d}", .{ used, s.c.frames });
543 for (s.frames, 0..) |frame, j| {
544 if (frame) |ref| {
545 const p = s.page(ref);
546 try s.emit("{d} P{d}:{d} R={d} M={d} load={d} last={d} uses={d} age={d}{s}", .{ j, s.processes.items[ref.process].pid, ref.page, p.r, p.m, p.load, p.last, p.uses, p.age, if (p.locked) @as(Text, " locked") else "" });
547 } else try s.emit("{d} free", .{j});
548 }
549 if (s.c.policy >= 5 and s.c.policy <= 7) {
550 if (!s.c.local) try s.emit("hand={d}", .{s.hand}) else {
551 try s.h.out.writeAll("hand");
552 for (1..100) |pid| if (s.processIndex(@intCast(pid))) |j| {
553 try s.h.out.print(" P{d}={d}", .{ pid, s.processes.items[j].hand });
554 };
555 try s.h.line("", .{});
556 }
557 }
558 return;
559 }
560 if (eq(op, "PAGETABLE")) {
561 const p = s.processes.items[owner];
562 try s.emit("PAGETABLE P{d} pages={d} resident={d}", .{ p.pid, p.pages.len, s.resident(owner) });
563 for (p.pages, 0..) |item, j| if (item.ever) {
564 const status = if (item.frame) |frame| try s.h.text("frame={d} R={d} M={d}", .{ frame, item.r, item.m }) else if (item.slot) |slot| try s.h.text("swap={d}", .{slot}) else "zero";
565 try s.emit("{d} {s}{s}{s}", .{ j, status, if (item.readonly) @as(Text, " RO") else "", if (item.locked) @as(Text, " locked") else "" });
566 };
567 return;
568 }
569 if (eq(op, "TLB")) {
570 try s.emit("TLB used={d}/{d} sets={d}", .{ s.entries.items.len, s.c.tlb, s.c.sets });
571 for (0..s.c.sets) |set| {
572 try s.h.out.print("set {d}:", .{set});
573 var j = s.entries.items.len;
574 while (j > 0) {
575 j -= 1;
576 const entry = s.entries.items[j];
577 if (entry.page % s.c.sets == set) {
578 try s.h.out.print(" P{d}:{d}->{d}", .{ s.processes.items[entry.process].pid, entry.page, entry.frame });
579 }
580 }
581 try s.h.line("", .{});
582 }
583 return;
584 }
585 if (eq(op, "SWAP")) {
586 var count: usize = 0;
587 var end: usize = 0;
588 for (s.processes.items) |p| {
589 for (p.pages) |item| {
590 if (item.slot) |slot| {
591 count += 1;
592 end = @max(end, slot + 1);
593 }
594 }
595 }
596 try s.emit("SWAP used={d}", .{count});
597 for (0..end) |slot| {
598 for (s.processes.items) |p| {
599 for (p.pages, 0..) |item, j| {
600 if (item.slot == slot) try s.emit("{d} P{d}:{d}", .{ slot, p.pid, j });
601 }
602 }
603 }
604 return;
605 }
606 if (eq(op, "COMPARE")) {
607 try s.emit("COMPARE accesses={d} frames={d}", .{ s.time, s.c.frames });
608 var fault_counts: [10]I = undefined;
609 var write_counts: [10]I = undefined;
610 var order: [10]usize = undefined;
611 for (policy_names, 0..) |name, j| {
612 const replay = try s.rerun(j, s.c.frames);
613 fault_counts[j] = replay.faults;
614 write_counts[j] = replay.writebacks;
615 order[j] = j;
616 try s.emit("{s} faults={d} writebacks={d} tlbhits={d}", .{ name, replay.faults, replay.writebacks, replay.tlbhits });
617 }
618 for (1..10) |j| {
619 const selected = order[j];
620 var at = j;
621 while (at > 0 and (fault_counts[selected] < fault_counts[order[at - 1]] or (fault_counts[selected] == fault_counts[order[at - 1]] and write_counts[selected] < write_counts[order[at - 1]]))) : (at -= 1) order[at] = order[at - 1];
622 order[at] = selected;
623 }
624 try s.h.out.writeAll("ranking");
625 for (order) |j| try s.h.out.print(" {s}", .{policy_names[j]});
626 return s.h.line("", .{});
627 }
628 if (eq(op, "CURVE")) {
629 const policy = lookup(w[1], &policy_names) orelse return s.emit("ERROR unknown policy", .{});
630 const lower = ix(num(w[2]));
631 const upper = ix(num(w[3]));
632 if (lower < 1 or lower > upper or upper > 16 or (s.c.local and lower < s.processes.items.len)) return s.emit("ERROR invalid frame range", .{});
633 try s.emit("CURVE {s} frames={d}..{d}", .{ w[1], lower, upper });
634 var previous: I = -1;
635 for (lower..upper + 1) |count| {
636 const replay = try s.rerun(policy, count);
637 try s.emit("F={d} faults={d}{s}", .{ count, replay.faults, if (previous >= 0 and replay.faults > previous) @as(Text, " ANOMALY") else "" });
638 previous = replay.faults;
639 }
640 return;
641 }
642 if (eq(op, "EAT")) {
643 if (s.time == 0) return s.emit("EAT accesses=0", .{});
644 const total = num(w[1]) * s.time + num(w[2]) * (2 * s.time - s.tlbhits) + num(w[3]) * (s.swapins + s.writebacks);
645 return s.emit("EAT accesses={d} total={d} average={s}", .{ s.time, total, try s.h.decimal(total, s.time) });
646 }
647 if (eq(op, "TIMELINE")) {
648 const width = ix(num(w[1]));
649 if (width < 1) return s.emit("ERROR invalid window", .{});
650 try s.h.out.print("TIMELINE window={d}", .{width});
651 var at: usize = 0;
652 while (at < s.accesses.items.len) : (at += width) {
653 var count: usize = 0;
654 for (s.accesses.items[at..@min(at + width, s.accesses.items.len)]) |record| if (record.fault) {
655 count += 1;
656 };
657 try s.h.out.print(" {d}", .{count});
658 }
659 return s.h.line("", .{});
660 }
661 if (eq(op, "WSET")) {
662 const width = ix(num(w[2]));
663 if (width < 1) return s.emit("ERROR invalid window", .{});
664 var seen: [256]bool = @splat(false);
665 var remaining = width;
666 var j = s.accesses.items.len;
667 while (j > 0 and remaining > 0) {
668 j -= 1;
669 const record = s.accesses.items[j];
670 if (record.process == owner) {
671 seen[record.page] = true;
672 remaining -= 1;
673 }
674 }
675 var count: usize = 0;
676 for (seen) |value| if (value) {
677 count += 1;
678 };
679 try s.h.out.print("WSET P{d} window={d} size={d}", .{ s.processes.items[owner].pid, width, count });
680 for (seen, 0..) |value, p| if (value) {
681 try s.h.out.print(" {d}", .{p});
682 };
683 return s.h.line("", .{});
684 }
685 if (eq(op, "HOT")) {
686 const count = ix(num(w[2]));
687 if (count < 1) return s.emit("ERROR invalid count", .{});
688 var frequencies: [256]I = @splat(0);
689 for (s.accesses.items) |record| if (record.process == owner) {
690 frequencies[record.page] += 1;
691 };
692 try s.h.out.print("HOT P{d}", .{s.processes.items[owner].pid});
693 for (0..count) |_| {
694 var best: ?usize = null;
695 for (frequencies, 0..) |frequency, j| if (frequency > 0 and (best == null or frequency > frequencies[best.?])) {
696 best = j;
697 };
698 const p = best orelse break;
699 try s.h.out.print(" {d}:{d}", .{ p, frequencies[p] });
700 frequencies[p] = 0;
701 }
702 return s.h.line("", .{});
703 }
704 }
705};
706fn readLine(r: *std.Io.Reader, a: std.mem.Allocator) !?Text {
707 var b: std.Io.Writer.Allocating = .init(a);
708 defer b.deinit();
709 _ = try r.streamDelimiterEnding(&b.writer, '\n');
710 if (r.bufferedLen() == 0 and b.written().len == 0) return null;
711 if (r.bufferedLen() > 0) r.toss(1);
712 return b.toOwnedSlice();
713}
714fn words(a: std.mem.Allocator, line: Text) ![]const Text {
715 var list: std.ArrayList(Text) = .empty;
716 var it = std.mem.splitScalar(u8, std.mem.trim(u8, line, "\r"), ' ');
717 while (it.next()) |word| try list.append(a, word);
718 return list.toOwnedSlice(a);
719}
720pub fn main(init: std.process.Init) !void {
721 var arena: std.heap.ArenaAllocator = .init(std.heap.page_allocator);
722 defer arena.deinit();
723 const a = arena.allocator();
724 var ib: [4096]u8 = undefined;
725 var ob: [4096]u8 = undefined;
726 var input = std.Io.File.stdin().readerStreaming(init.io, &ib);
727 var output = std.Io.File.stdout().writerStreaming(init.io, &ob);
728 const h: Host = .{ .gpa = a, .out = &output.interface };
729 const count = num(std.mem.trim(u8, (try readLine(&input.interface, a)).?, "\r"));
730 var commands: std.ArrayList(Command) = .empty;
731 for (0..ix(count)) |_| {
732 const w = try words(a, (try readLine(&input.interface, a)).?);
733 if (eq(w[0], "LOOP")) {
734 const pid = num(w[1]);
735 const start = hex(w[3]);
736 const interval = num(w[4]);
737 const length = num(w[5]);
738 const repeats = num(w[6]);
739 for (0..ix(repeats)) |_| {
740 for (0..ix(length)) |j| {
741 try commands.append(a, accessCommand(pid, w[2], start + @as(I, @intCast(j)) * interval));
742 }
743 }
744 } else if (eq(w[0], "RAND")) {
745 const pid = num(w[1]);
746 const lower = hex(w[2]);
747 const upper = hex(w[3]);
748 const n = num(w[4]);
749 var random = num(w[5]);
750 const rate = num(w[6]);
751 for (0..ix(n)) |_| {
752 random = @mod(48271 * random, 2147483647);
753 const address = lower + @mod(random, upper - lower);
754 random = @mod(48271 * random, 2147483647);
755 try commands.append(a, accessCommand(pid, if (@mod(random, 100) < rate) "W" else "R", address));
756 }
757 } else if (std.ascii.isDigit(w[0][0])) {
758 try commands.append(a, accessCommand(num(w[0]), w[1], hex(w[2])));
759 } else try commands.append(a, .{ .words = w });
760 }
761 var simulation: ?Simulation = null;
762 for (commands.items, 0..) |command, command_index| {
763 if (!command.access and eq(command.words[0], "SYSTEM")) {
764 simulation = null;
765 switch (try parseSettings(h, command.words)) {
766 .problem => |problem| try h.line("ERROR {s}", .{problem}),
767 .settings => |c| {
768 simulation = try Simulation.init(h, c);
769 for (commands.items[command_index + 1 ..]) |future| {
770 if (!future.access and eq(future.words[0], "SYSTEM")) break;
771 if (future.access) try simulation.?.future.append(a, future);
772 }
773 try h.line("SYSTEM page={d} frames={d} tlb={d} sets={d} policy={s} scope={s} asid={s} tick={d} tau={d} seed={d}", .{ c.page, c.frames, c.tlb, c.sets, policy_names[c.policy], c.scope(), if (c.asid) @as(Text, "ON") else "OFF", c.tick, c.tau, c.seed });
774 },
775 }
776 } else if (simulation) |*s| {
777 if (command.access) try s.access(command) else try s.command(command.words);
778 } else try h.line("ERROR no system", .{});
779 }
780 try output.interface.flush();
781}
