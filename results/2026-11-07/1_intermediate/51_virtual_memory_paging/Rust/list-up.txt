const POLICIES: [&str; 10] = [
    "FIFO", "LRU", "LFU", "NRU", "AGING", "CLOCK", "ECLOCK", "WSCLOCK", "RANDOM", "OPT",
];
    fn default() -> Self {
        Self {
            page: 256,
            frames: 4,
            tlb: 4,
            sets: 1,
            policy: "FIFO".into(),
            local: false,
            asid: false,
            tick: 0,
            tau: 8,
            seed: 1,
        }
    }
    fn scope(&self) -> &str {
        if self.local { "LOCAL" } else { "GLOBAL" }
    }
    fn description(&self) -> String {
        format!(
            "SYSTEM page={} frames={} tlb={} sets={} policy={} scope={} asid={} tick={} tau={} seed={}",
            self.page,
            self.frames,
            self.tlb,
            self.sets,
            self.policy,
            self.scope(),
            if self.asid { "ON" } else { "OFF" },
            self.tick,
            self.tau,
            self.seed
        )
    }
fn settings(tokens: &[String]) -> Result<Config, String> {
    let mut c = Config::default();
    let keys = [
        "page", "frames", "tlb", "sets", "policy", "scope", "asid", "tick", "tau", "seed",
    ];
    let mut seen = BTreeSet::new();
    for token in &tokens[1..] {
        let (key, value) = token.split_once('=').unwrap();
        if !keys.contains(&key) {
            return Err(format!("unknown key {}", key));
        }
        if !seen.insert(key) {
            return Err(format!("duplicate key {}", key));
        }
        let valid = match key {
            "policy" => {
                c.policy = value.into();
                POLICIES.contains(&value)
            }
            "scope" => {
                c.local = value == "LOCAL";
                ["GLOBAL", "LOCAL"].contains(&value)
            }
            "asid" => {
                c.asid = value == "ON";
                ["ON", "OFF"].contains(&value)
            }
            _ => {
                let n: u64 = value.parse().unwrap();
                let valid = match key {
                    "page" => (16..=4096).contains(&n) && n.is_power_of_two(),
                    "frames" => (1..=16).contains(&n),
                    "tlb" | "sets" => (1..=8).contains(&n),
                    "tick" => n <= 1000,
                    "tau" => (1..=1000).contains(&n),
                    _ => (1..=2147483646).contains(&n),
                };
                if valid {
                    match key {
                        "page" => c.page = n as usize,
                        "frames" => c.frames = n as usize,
                        "tlb" => c.tlb = n as usize,
                        "sets" => c.sets = n as usize,
                        "tick" => c.tick = n as usize,
                        "tau" => c.tau = n as usize,
                        _ => c.seed = n,
                    }
                }
                valid
            }
        };
        if !valid {
            return Err(format!("invalid {}", key));
        }
    }
    if c.tlb % c.sets != 0 {
        return Err("invalid sets".into());
    }
    Ok(c)
}
