71fn settings(tokens: &[String]) -> Result<Config, String> {
72 let mut c = Config::default();
73 let keys = [
74 "page", "frames", "tlb", "sets", "policy", "scope", "asid", "tick", "tau", "seed",
75 ];
76 let mut seen = BTreeSet::new();
77 for token in &tokens[1..] {
78 let (key, value) = token.split_once('=').unwrap();
79 if !keys.contains(&key) {
80 return Err(format!("unknown key {}", key));
81 }
82 if !seen.insert(key) {
83 return Err(format!("duplicate key {}", key));
84 }
85 let valid = match key {
86 "policy" => {
87 c.policy = value.into();
88 POLICIES.contains(&value)
89 }
90 "scope" => {
91 c.local = value == "LOCAL";
92 ["GLOBAL", "LOCAL"].contains(&value)
93 }
94 "asid" => {
95 c.asid = value == "ON";
96 ["ON", "OFF"].contains(&value)
97 }
98 _ => {
99 let n: u64 = value.parse().unwrap();
100 let valid = match key {
101 "page" => (16..=4096).contains(&n) && n.is_power_of_two(),
102 "frames" => (1..=16).contains(&n),
103 "tlb" | "sets" => (1..=8).contains(&n),
104 "tick" => n <= 1000,
105 "tau" => (1..=1000).contains(&n),
106 _ => (1..=2147483646).contains(&n),
107 };
108 if valid {
109 match key {
110 "page" => c.page = n as usize,
111 "frames" => c.frames = n as usize,
112 "tlb" => c.tlb = n as usize,
113 "sets" => c.sets = n as usize,
114 "tick" => c.tick = n as usize,
115 "tau" => c.tau = n as usize,
116 _ => c.seed = n,
117 }
118 }
119 valid
120 }
121 };
122 if !valid {
123 return Err(format!("invalid {}", key));
124 }
125 }
126 if c.tlb % c.sets != 0 {
127 return Err("invalid sets".into());
128 }
129 Ok(c)
130}

275 fn quota(&self, pid: usize) -> usize {
276 let rank = self.processes.keys().position(|&p| p == pid).unwrap();
277 self.c.frames / self.processes.len()
278 + usize::from(rank < self.c.frames % self.processes.len())
279 }
280 fn declare(&mut self, pid: usize, count: usize) -> Result<(), String> {
281 if self.fixed {
282 return Err("processes are fixed".into());
283 }
284 if self.processes.contains_key(&pid) {
285 return Err(format!("duplicate process {}", pid));
286 }
287 if !(1..=256).contains(&count) {
288 return Err("invalid page count".into());
289 }
290 if self.processes.len() >= 8 || self.c.local && self.processes.len() == self.c.frames {
291 return Err("too many processes".into());
292 }
293 self.processes.insert(pid, Process::new(count));
294 Ok(())
295 }
296 fn invalidate(&mut self, key: Key) {
297 for set in &mut self.tlb {
298 set.retain(|&p| p != key);
299 }
300 }
301 fn write(&mut self, key: Key) {
302 self.writebacks += 1;
303 self.processes.get_mut(&key.pid).unwrap().writebacks += 1;
304 self.page_mut(key).m = 0;
305 if self.page(key).slot.is_none() {
306 let mut slot = 0;
307 while self.swap.contains_key(&slot) {
308 slot += 1;
309 }
310 self.swap.insert(slot, key);
311 self.page_mut(key).slot = Some(slot);
312 }
313 }
314 fn next_use(&self, key: Key) -> usize {
315 self.future
316 .iter()
317 .enumerate()
318 .skip(self.cursor)
319 .find(|(_, a)| a.pid == key.pid && a.address / self.c.page == key.page)
320 .map_or(usize::MAX, |(i, _)| i)
321 }
322 fn choose(&mut self, mut candidates: Vec<usize>, pid: usize) -> usize {
323 let unlocked: Vec<usize> = candidates
324 .iter()
325 .copied()
326 .filter(|&f| !self.page(self.frames[f].unwrap()).locked)
327 .collect();
328 if !unlocked.is_empty() {
329 candidates = unlocked;
330 }
331 if ["FIFO", "LRU", "LFU", "NRU", "AGING", "OPT"].contains(&self.c.policy.as_str()) {
332 candidates.sort_by(|&a, &b| {
333 let x = self.page(self.frames[a].unwrap());
334 let y = self.page(self.frames[b].unwrap());
335 match self.c.policy.as_str() {
336 "FIFO" => x.load.cmp(&y.load),
337 "LRU" => x.last.cmp(&y.last),
338 "LFU" => (x.uses, x.load).cmp(&(y.uses, y.load)),
339 "NRU" => (2 * x.r + x.m, a).cmp(&(2 * y.r + y.m, b)),
340 "AGING" => (x.age, x.load).cmp(&(y.age, y.load)),
341 _ => self
342 .next_use(self.frames[b].unwrap())
343 .cmp(&self.next_use(self.frames[a].unwrap()))
344 .then(a.cmp(&b)),
345 }
346 });
347 return candidates[0];
348 }
349 if self.c.policy == "RANDOM" {
350 self.random = self.random * 48271 % 2147483647;
351 return candidates[(self.random % candidates.len() as u64) as usize];
352 }
353 let h = if self.c.local {
354 self.processes[&pid].hand
355 } else {
356 self.hand
357 };
358 candidates.sort_by_key(|&f| (f + self.c.frames - h) % self.c.frames);
359 let mut chosen = None;
360 if self.c.policy == "CLOCK" {
361 while chosen.is_none() {
362 for &f in &candidates {
363 let key = self.frames[f].unwrap();
364 if self.page(key).r == 0 {
365 chosen = Some(f);
366 break;
367 }
368 self.page_mut(key).r = 0;
369 }
370 }
371 } else if self.c.policy == "ECLOCK" {
372 while chosen.is_none() {
373 for &f in &candidates {
374 let p = self.page(self.frames[f].unwrap());
375 if p.r == 0 && p.m == 0 {
376 chosen = Some(f);
377 break;
378 }
379 }
380 if chosen.is_some() {
381 break;
382 }
383 for &f in &candidates {
384 let key = self.frames[f].unwrap();
385 if self.page(key).r == 0 && self.page(key).m == 1 {
386 chosen = Some(f);
387 break;
388 }
389 self.page_mut(key).r = 0;
390 }
391 }
392 } else {
393 for &f in &candidates {
394 let key = self.frames[f].unwrap();
395 if self.page(key).r == 1 {
396 let time = self.time;
397 let p = self.page_mut(key);
398 p.r = 0;
399 p.last = time;
400 } else if self.time - self.page(key).last > self.c.tau {
401 if self.page(key).m == 0 {
402 chosen = Some(f);
403 break;
404 }
405 self.write(key);
406 }
407 }
408 if chosen.is_none() {
409 chosen = Some(
410 candidates
411 .iter()
412 .copied()
413 .find(|&f| self.page(self.frames[f].unwrap()).m == 0)
414 .unwrap_or(candidates[0]),
415 );
416 }
417 }
418 let result = chosen.unwrap();
419 let next = (result + 1) % self.c.frames;
420 if self.c.local {
421 self.processes.get_mut(&pid).unwrap().hand = next;
422 } else {
423 self.hand = next;
424 }
425 result
426 }
427 fn access(&mut self, a: Access) -> Result<(), String> {
428 self.fixed = true;
429 self.cursor += 1;
430 let owner = self.process(a.pid)?;
431 if a.address >= owner.pages.len() * self.c.page {
432 self.processes.get_mut(&a.pid).unwrap().segfaults += 1;
433 self.emit(format!("SEGFAULT P{} {}", a.pid, hex(a.address)));
434 return Ok(());
435 }
436 let key = Key {
437 pid: a.pid,
438 page: a.address / self.c.page,
439 };
440 if a.op == 'W' && self.page(key).ro {
441 self.processes.get_mut(&a.pid).unwrap().protfaults += 1;
442 self.emit(format!("PROTFAULT P{} {}", a.pid, hex(a.address)));
443 return Ok(());
444 }
445 self.time += 1;
446 self.processes.get_mut(&a.pid).unwrap().accesses += 1;
447 let switched = self.previous.is_some_and(|pid| pid != a.pid);
448 if switched {
449 self.switches += 1;
450 if !self.c.asid {
451 for set in &mut self.tlb {
452 set.clear();
453 }
454 }
455 }
456 self.previous = Some(a.pid);
457 let set = key.page % self.c.sets;
458 let hit = self.tlb[set].iter().position(|&p| p == key);
459 let mut fault = false;
460 let mut result;
461 if let Some(index) = hit {
462 self.tlb[set].remove(index);
463 self.tlbhits += 1;
464 result = "TLBHIT".to_string();
465 } else if self.page(key).frame.is_some() {
466 result = "HIT".to_string();
467 } else {
468 fault = true;
469 self.faults += 1;
470 self.processes.get_mut(&a.pid).unwrap().faults += 1;
471 if self.page(key).slot.is_some() {
472 self.swapin += 1;
473 result = "FAULT SWAPIN".to_string();
474 } else {
475 self.zero += 1;
476 result = "FAULT ZERO".to_string();
477 }
478 let mut frame = None;
479 if !self.c.local || self.processes[&a.pid].resident() < self.quota(a.pid) {
480 frame = self.frames.iter().position(Option::is_none);
481 }
482 if frame.is_none() {
483 let candidates = self
484 .frames
485 .iter()
486 .enumerate()
487 .filter_map(|(f, key)| match key {
488 Some(p) if !self.c.local || p.pid == a.pid => Some(f),
489 _ => None,
490 })
491 .collect();
492 let f = self.choose(candidates, a.pid);
493 frame = Some(f);
494 let victim = self.frames[f].unwrap();
495 result += &format!(" evict={}", victim.label());
496 self.invalidate(victim);
497 if self.page(victim).m == 1 {
498 self.write(victim);
499 result += " writeback";
500 }
501 self.page_mut(victim).frame = None;
502 }
503 let f = frame.unwrap();
504 self.frames[f] = Some(key);
505 let time = self.time;
506 let p = self.page_mut(key);
507 p.frame = Some(f);
508 p.load = time;
509 p.uses = 0;
510 p.m = 0;
511 p.age = 0;
512 let owner = self.processes.get_mut(&a.pid).unwrap();
513 owner.peak = owner.peak.max(owner.resident());
514 }
515 if self.tlb[set].len() == self.c.tlb / self.c.sets {
516 self.tlb[set].remove(0);
517 }
518 self.tlb[set].push(key);
519 let time = self.time;
520 let p = self.page_mut(key);
521 p.r = 1;
522 if a.op == 'W' {
523 p.m = 1;
524 }
525 p.last = time;
526 p.uses += 1;
527 p.ever = true;
528 p.references += 1;
529 self.processes
530 .get_mut(&a.pid)
531 .unwrap()
532 .history
533 .push(key.page);
534 self.fault_history.push(fault);
535 self.log.push(Event::Access(a));
536 let interrupt = self.c.tick > 0 && self.time % self.c.tick == 0;
537 if interrupt {
538 for key in self.frames.clone().into_iter().flatten() {
539 let p = self.page_mut(key);
540 p.age = p.age / 2 + 128 * p.r;
541 p.r = 0;
542 }
543 }
544 if self.trace {
545 let f = self.page(key).frame.unwrap();
546 self.emit(format!(
547 "t={} P{} {} {} page={} frame={} phys={} {}{}",
548 self.time,
549 a.pid,
550 a.op,
551 hex(a.address),
552 key.page,
553 f,
554 hex(f * self.c.page + a.address % self.c.page),
555 result,
556 if switched { " switch" } else { "" }
557 ));
558 if interrupt {
559 self.emit(format!("t={} tick", self.time));
560 }
561 }
562 Ok(())
563 }
564 fn exit(&mut self, pid: usize) -> Result<(), String> {
565 let count = self.process(pid)?.pages.len();
566 let (mut freed, mut slots) = (0, 0);
567 for page in 0..count {
568 let key = Key { pid, page };
569 if let Some(f) = self.page(key).frame {
570 self.frames[f] = None;
571 self.page_mut(key).frame = None;
572 freed += 1;
573 }
574 self.invalidate(key);
575 if let Some(slot) = self.page(key).slot {
576 self.swap.remove(&slot);
577 self.page_mut(key).slot = None;
578 slots += 1;
579 }
580 }
581 self.processes.get_mut(&pid).unwrap().exited = true;
582 self.log.push(Event::Exit(pid));
583 self.emit(format!("EXIT P{} freed={} slots={}", pid, freed, slots));
584 Ok(())
585 }
586 fn attributes(
587 &mut self,
588 cmd: &str,
589 pid: usize,
590 first: usize,
591 last: usize,
592 ro: bool,
593 ) -> Result<(), String> {
594 let p = self.process(pid)?;
595 if first > last || last >= p.pages.len() {
596 return Err("invalid page range".into());
597 }
598 for page in first..=last {
599 let p = self.page_mut(Key { pid, page });
600 if cmd == "PROTECT" {
601 p.ro = ro;
602 } else {
603 p.locked = cmd == "LOCK";
604 }
605 }
606 if cmd != "PROTECT" {
607 self.log.push(Event::Lock(pid, first, last, cmd == "LOCK"));
608 }
609 Ok(())
610 }
611 fn clean(&mut self) {
612 let mut count = 0;
613 for key in self.frames.clone().into_iter().flatten() {
614 if self.page(key).m == 1 {
615 self.write(key);
616 count += 1;
617 }
618 }
619 self.log.push(Event::Clean);
620 self.emit(format!("CLEAN written={}", count));
621 }
622 fn rerun(&self, policy: &str, count: usize) -> Result<Self, String> {
623 let mut c = self.c.clone();
624 c.policy = policy.into();
625 c.frames = count;
626 let future = self
627 .log
628 .iter()
629 .filter_map(|e| {
630 if let Event::Access(a) = e {
631 Some(*a)
632 } else {
633 None
634 }
635 })
636 .collect();
637 let mut r = Self::new(c, future, true);
638 for (&pid, p) in &self.processes {
639 r.declare(pid, p.pages.len())?;
640 }
641 for event in &self.log {
642 match *event {
643 Event::Access(a) => r.access(a)?,
644 Event::Exit(pid) => r.exit(pid)?,
645 Event::Clean => r.clean(),
646 Event::Lock(pid, first, last, locked) => r.attributes(
647 if locked { "LOCK" } else { "UNLOCK" },
648 pid,
649 first,
650 last,
651 false,
652 )?,
653 }
654 }
655 Ok(r)
656 }
657 fn summary(&self) {
658 self.emit(format!(
659 "SUMMARY policy={} scope={} time={}",
660 self.c.policy,
661 self.c.scope(),
662 self.time
663 ));
664 let seg: usize = self.processes.values().map(|p| p.segfaults).sum();
665 let prot: usize = self.processes.values().map(|p| p.protfaults).sum();
666 self.emit(format!("total faults={} zero={} swapin={} faultrate={} tlbhits={} tlbrate={} writebacks={} switches={} segfaults={} protfaults={}",
667 self.faults, self.zero, self.swapin, ratio(self.faults, self.time), self.tlbhits, ratio(self.tlbhits, self.time), self.writebacks, self.switches, seg, prot));
668 for (&pid, p) in &self.processes {
669 self.emit(format!("P{} accesses={} faults={} writebacks={} segfaults={} protfaults={} resident={} peak={}{}{}", pid, p.accesses, p.faults,
670 p.writebacks, p.segfaults, p.protfaults, p.resident(), p.peak,
671 if self.c.local { format!(" quota={}", self.quota(pid)) } else { String::new() }, if p.exited { " exited" } else { "" }));
672 }
673 }

675 self.emit(format!(
676 "FRAMES used={}/{}",
677 self.frames.iter().flatten().count(),
678 self.c.frames
679 ));
680 for (f, key) in self.frames.iter().enumerate() {
681 if let Some(key) = key {
682 let p = self.page(*key);
683 self.emit(format!(
684 "{} {} R={} M={} load={} last={} uses={} age={}{}",
685 f,
686 key.label(),
687 p.r,
688 p.m,
689 p.load,
690 p.last,
691 p.uses,
692 p.age,
693 if p.locked { " locked" } else { "" }
694 ));
695 } else {
696 self.emit(format!("{} free", f));
697 }
698 }
699 if ["CLOCK", "ECLOCK", "WSCLOCK"].contains(&self.c.policy.as_str()) {
700 if !self.c.local {
701 self.emit(format!("hand={}", self.hand));
702 } else {
703 let mut text = "hand".to_string();
704 for (&pid, p) in &self.processes {
705 text += &format!(" P{}={}", pid, p.hand);
706 }
707 self.emit(text);
708 }
709 }

711 fn page_table(&self, pid: usize) -> Result<(), String> {
712 let p = self.process(pid)?;
713 self.emit(format!(
714 "PAGETABLE P{} pages={} resident={}",
715 pid,
716 p.pages.len(),
717 p.resident()
718 ));
719 for (number, p) in p.pages.iter().enumerate().filter(|(_, p)| p.ever) {
720 let info = if let Some(f) = p.frame {
721 format!("frame={} R={} M={}", f, p.r, p.m)
722 } else if let Some(slot) = p.slot {
723 format!("swap={}", slot)
724 } else {
725 "zero".into()
726 };
727 self.emit(format!(
728 "{} {}{}{}",
729 number,
730 info,
731 if p.ro { " RO" } else { "" },
732 if p.locked { " locked" } else { "" }
733 ));
734 }
735 Ok(())
736 }
737 fn compare(&self) -> Result<(), String> {
738 self.emit(format!(
739 "COMPARE accesses={} frames={}",
740 self.time, self.c.frames
741 ));
742 let mut results = Vec::new();
743 for (i, policy) in POLICIES.iter().enumerate() {
744 let r = self.rerun(policy, self.c.frames)?;
745 self.emit(format!(
746 "{} faults={} writebacks={} tlbhits={}",
747 policy, r.faults, r.writebacks, r.tlbhits
748 ));
749 results.push((r.faults, r.writebacks, i));
750 }
751 results.sort();
752 self.emit(
753 "ranking ".to_string()
754 + &results
755 .iter()
756 .map(|x| POLICIES[x.2])
757 .collect::<Vec<_>>()
758 .join(" "),
759 );
760 Ok(())
761 }
762 fn execute(&mut self, t: &[String]) -> Result<(), String> {
763 let cmd = t[0].as_str();
764 if cmd.as_bytes()[0].is_ascii_digit() {
765 return self.access(Access {
766 pid: number(cmd),
767 op: t[1].chars().next().unwrap(),
768 address: address(&t[2]),
769 });
770 }
771 match cmd {
772 "PROCESS" => self.declare(number(&t[1]), number(&t[2]))?,
773 "EXIT" => self.exit(number(&t[1]))?,
774 "PROTECT" | "LOCK" | "UNLOCK" => self.attributes(
775 cmd,
776 number(&t[1]),
777 number(&t[2]),
778 number(&t[3]),
779 t.get(4).is_some_and(|s| s == "RO"),
780 )?,
781 "TRACE" => self.trace = t[1] == "ON",
782 "SUMMARY" => self.summary(),
783 "FRAMES" => self.frame_dump(),
784 "PAGETABLE" => self.page_table(number(&t[1]))?,
785 "TLB" => {
786 self.emit(format!(
787 "TLB used={}/{} sets={}",
788 self.tlb.iter().map(Vec::len).sum::<usize>(),
789 self.c.tlb,
790 self.c.sets
791 ));
792 for (i, set) in self.tlb.iter().enumerate() {
793 let mut text = format!("set {}:", i);
794 for &key in set.iter().rev() {
795 text += &format!(" {}->{}", key.label(), self.page(key).frame.unwrap());
796 }
797 self.emit(text);
798 }
799 }
800 "SWAP" => {
801 self.emit(format!("SWAP used={}", self.swap.len()));
802 for (&slot, &key) in &self.swap {
803 self.emit(format!("{} {}", slot, key.label()));
804 }
805 }
806 "CLEAN" => self.clean(),
807 "COMPARE" => self.compare()?,
808 "CURVE" => {
809 let policy = t[1].as_str();
810 let lo = number(&t[2]);
811 let hi = number(&t[3]);
812 if !POLICIES.contains(&policy) {
813 return Err("unknown policy".into());
814 }
815 if lo < 1 || lo > hi || hi > 16 || self.c.local && lo < self.processes.len() {
816 return Err("invalid frame range".into());
817 }
818 self.emit(format!("CURVE {} frames={}..{}", policy, lo, hi));
819 let mut previous = usize::MAX;
820 for f in lo..=hi {
821 let value = self.rerun(policy, f)?.faults;
822 self.emit(format!(
823 "F={} faults={}{}",
824 f,
825 value,
826 if value > previous { " ANOMALY" } else { "" }
827 ));
828 previous = value;
829 }
830 }
831 "EAT" => {
832 let total = number(&t[1]) as u64 * self.time as u64
833 + number(&t[2]) as u64 * (2 * self.time - self.tlbhits) as u64
834 + number(&t[3]) as u64 * (self.swapin + self.writebacks) as u64;
835 self.emit(if self.time == 0 {
836 "EAT accesses=0".into()
837 } else {
838 format!(
839 "EAT accesses={} total={} average={}",
840 self.time,
841 total,
842 decimal(100 * total / self.time as u64)
843 )
844 });
845 }
846 "TIMELINE" => {
847 let w = number(&t[1]);
848 if w < 1 {
849 return Err("invalid window".into());
850 }
851 let mut text = format!("TIMELINE window={}", w);
852 for chunk in self.fault_history.chunks(w) {
853 text += &format!(" {}", chunk.iter().filter(|&&v| v).count());
854 }
855 self.emit(text);
856 }
857 "WSET" => {
858 let pid = number(&t[1]);
859 let p = self.process(pid)?;
860 let w = number(&t[2]);
861 if w < 1 {
862 return Err("invalid window".into());
863 }
864 let pages: BTreeSet<usize> = p.history.iter().rev().take(w).copied().collect();
865 let mut text = format!("WSET P{} window={} size={}", pid, w, pages.len());
866 for page in pages {
867 text += &format!(" {}", page);
868 }
869 self.emit(text);
870 }
871 "HOT" => {
872 let pid = number(&t[1]);
873 let p = self.process(pid)?;
874 let k = number(&t[2]);
875 if k < 1 {
876 return Err("invalid count".into());
877 }
878 let mut pages: Vec<_> = p
879 .pages
880 .iter()
881 .enumerate()
882 .filter(|(_, p)| p.references > 0)
883 .collect();
884 pages.sort_by(|(a, x), (b, y)| y.references.cmp(&x.references).then(a.cmp(b)));
885 let mut text = format!("HOT P{}", pid);
886 for (number, p) in pages.into_iter().take(k) {
887 text += &format!(" {}:{}", number, p.references);
888 }
889 self.emit(text);
890 }
891 _ => unreachable!(),
892 }
893 Ok(())
894 }
895}
896fn main() {
897 let mut input = String::new();
898 io::stdin().read_to_string(&mut input).unwrap();
899 let mut lines = input.lines();
900 let n = number(lines.next().unwrap());
901 let mut commands: Vec<Vec<String>> = Vec::new();
902 for _ in 0..n {
903 let t: Vec<String> = lines
904 .next()
905 .unwrap()
906 .split(' ')
907 .map(str::to_string)
908 .collect();
909 if t[0] == "LOOP" {
910 let start = address(&t[3]);
911 let interval = number(&t[4]);
912 let length = number(&t[5]);
913 let repeats = number(&t[6]);
914 for _ in 0..repeats {
915 for j in 0..length {
916 commands.push(vec![t[1].clone(), t[2].clone(), hex(start + j * interval)]);
917 }
918 }
919 } else if t[0] == "RAND" {
920 let lo = address(&t[2]);
921 let hi = address(&t[3]);
922 let count = number(&t[4]);
923 let rate = number(&t[6]) as u64;
924 let mut x: u64 = t[5].parse().unwrap();
925 for _ in 0..count {
926 x = x * 48271 % 2147483647;
927 let a = lo + (x % (hi - lo) as u64) as usize;
928 x = x * 48271 % 2147483647;
929 commands.push(vec![
930 t[1].clone(),
931 if x % 100 < rate {
932 "W".into()
933 } else {
934 "R".into()
935 },
936 hex(a),
937 ]);
938 }
939 } else {
940 commands.push(t);
941 }
942 }
943 let mut simulation: Option<Simulation> = None;
944 for (i, t) in commands.iter().enumerate() {
945 let result = if t[0] == "SYSTEM" {
946 simulation = None;
947 match settings(t) {
948 Err(error) => Err(error),
949 Ok(c) => {
950 let future = commands[i + 1..]
951 .iter()
952 .take_while(|t| t[0] != "SYSTEM")
953 .filter(|t| t[0].as_bytes()[0].is_ascii_digit())
954 .map(|t| Access {
955 pid: number(&t[0]),
956 op: t[1].chars().next().unwrap(),
957 address: address(&t[2]),
958 })
959 .collect();
960 println!("{}", c.description());
961 simulation = Some(Simulation::new(c, future, false));
962 Ok(())
963 }
964 }
965 } else if let Some(s) = &mut simulation {
966 s.execute(t)
967 } else {
968 Err("no system".into())
969 };
970 if let Err(error) = result {
971 println!("ERROR {}", error);
972 }
973 }
974}
