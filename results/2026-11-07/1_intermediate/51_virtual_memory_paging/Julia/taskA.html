5 text::String
6end

8num(s) = parse(Int, s)

11function ratio(x, y)
12 v = y == 0 ? 0 : div(10000x, y)
13 @sprintf("%d.%02d%%", div(v, 100), mod(v, 100))
14end
15function settings(words)
16 c = Dict{String,Union{Int,String}}("page"=>256, "frames"=>4, "tlb"=>4, "sets"=>1, "policy"=>"FIFO", "scope"=>"GLOBAL", "asid"=>"OFF", "tick"=>0, "tau"=>8, "seed"=>1)
17 seen = Set{String}()
18 limits = Dict("frames"=>(1,16), "tlb"=>(1,8), "sets"=>(1,8), "tick"=>(0,1000), "tau"=>(1,1000), "seed"=>(1,2147483646))
19 for word in words
20 k, text = String.(split(word, '='))
21 haskey(c, k) || fail("unknown key $k")
22 k in seen && fail("duplicate key $k")
23 push!(seen, k)
24 value = text
25 if k in ("policy", "scope", "asid")
26 choices = k == "policy" ? POLICIES : k == "scope" ? ["GLOBAL", "LOCAL"] : ["ON", "OFF"]
27 text in choices || fail("invalid $k")
28 else
29 value = num(text)
30 if k == "page"
31 (16 <= value <= 4096 && (value & (value - 1)) == 0) || fail("invalid $k")
32 else
33 lo, hi = limits[k]
34 lo <= value <= hi || fail("invalid $k")
35 end
36 end
37 c[k] = value
38 end
39 mod(c["tlb"], c["sets"]) == 0 || fail("invalid sets")
40 c
41end

103function Simulation(c, refs)
104 counts = Dict(k=>0 for k in ("faults", "zero", "swapin", "tlbhits", "writebacks", "switches"))
105 Simulation(c, Dict{Int,VMProcess}(), Union{Nothing,Page}[nothing for _ in 1:c["frames"]], Union{Nothing,Page}[], [Page[] for _ in 1:c["sets"]], refs, 0, 0, counts, 0, 0, c["seed"], Int[])

108function process(s, pid)
109 haskey(s.processes, pid) || fail("unknown process $pid")
110 p = s.processes[pid]
111 p.exited && fail("process $pid exited")
112 p
113end
114function quota(s, pid)
115 ids = pids(s)
116 q, r = divrem(s.config["frames"], length(ids))
117 q + (findfirst(==(pid), ids) <= r)
118end
119function removetlb!(s, p)
120 filter!(x -> x !== p, s.tlb[mod(p.number, s.config["sets"]) + 1])
121end
122function writeback!(s, p)
123 s.counts["writebacks"] += 1
124 s.processes[p.pid].writes += 1
125 p.m = 0
126 if p.slot < 0
127 i = findfirst(isnothing, s.slots)
128 if i === nothing
129 push!(s.slots, nothing)
130 i = length(s.slots)
131 end
132 p.slot = i - 1
133 s.slots[i] = p
134 end
135end
136function nextuse(s, p)
137 for i in (s.position + 1):length(s.refs)
138 s.refs[i] == (p.pid, p.number) && return i
139 end
140 length(s.refs) + 1
141end
142function choose!(s, proc, candidates)
143 c = s.config
144 policy = c["policy"]
145 if policy == "RANDOM"
146 s.random = mod(48271s.random, 2147483647)
147 return candidates[mod(s.random, length(candidates)) + 1]
148 elseif policy in CLOCKS
149 hand = c["scope"] == "LOCAL" ? proc.hand : s.hand
150 order = sort(candidates; by=f -> mod(f - 1 - hand, c["frames"]))
151 chosen = 0
152 if policy == "CLOCK"
153 while chosen == 0
154 for f in order
155 p = s.frames[f]
156 if p.r == 0
157 chosen = f
158 break
159 end
160 p.r = 0
161 end
162 end
163 elseif policy == "ECLOCK"
164 while chosen == 0
165 for f in order
166 p = s.frames[f]
167 if p.r == 0 && p.m == 0
168 chosen = f
169 break
170 end
171 end
172 chosen != 0 && break
173 for f in order
174 p = s.frames[f]
175 if p.r == 0 && p.m == 1
176 chosen = f
177 break
178 end
179 p.r = 0
180 end
181 end
182 else
183 for f in order
184 p = s.frames[f]
185 if p.r == 1
186 p.r, p.last = 0, s.time
187 elseif s.time - p.last > c["tau"]
188 if p.m == 0
189 chosen = f
190 break
191 end
192 writeback!(s, p)
193 end
194 end
195 if chosen == 0
196 i = findfirst(f -> s.frames[f].m == 0, order)
197 chosen = i === nothing ? order[1] : order[i]
198 end
199 end
200 if c["scope"] == "LOCAL"
201 proc.hand = mod(chosen, c["frames"])
202 else
203 s.hand = mod(chosen, c["frames"])
204 end
205 return chosen
206 end
207 function rank(f)
208 p = s.frames[f]
209 policy == "FIFO" && return (p.load, f)
210 policy == "LRU" && return (p.last, f)
211 policy == "LFU" && return (p.uses, p.load)
212 policy == "NRU" && return (2p.r + p.m, f)
213 policy == "AGING" && return (p.age, p.load)
214 (-nextuse(s, p), f)
215 end
216 first(sort(candidates; by=rank))
217end
218function target!(s, p)
219 globalmode = s.config["scope"] == "GLOBAL"
220 if globalmode || p.resident < quota(s, p.pid)
221 i = findfirst(isnothing, s.frames)
222 i !== nothing && return i
223 end
224 candidates = [f for f in eachindex(s.frames) if s.frames[f] !== nothing && (globalmode || s.frames[f].pid == p.pid)]
225 unlocked = filter(f -> !s.frames[f].locked, candidates)
226 choose!(s, p, isempty(unlocked) ? candidates : unlocked)
227end
228function access!(s, a::Access, position)
229 s.position = position
230 s.time += 1
231 c = s.config
232 p = s.processes[a.pid]
233 switched = s.lastpid != 0 && s.lastpid != a.pid
234 if switched
235 s.counts["switches"] += 1
236 c["asid"] == "OFF" && foreach(empty!, s.tlb)
237 end
238 s.lastpid = a.pid
239 n, off = divrem(a.address, c["page"])
240 page = p.pages[n + 1]
241 set = s.tlb[mod(n, c["sets"]) + 1]
242 hit = any(x -> x === page, set)
243 result = ""
244 fault = 0
245 if hit
246 s.counts["tlbhits"] += 1
247 result = "TLBHIT"
248 removetlb!(s, page)
249 elseif page.frame >= 0
250 result = "HIT"
251 else
252 fault = 1
253 p.faults += 1
254 s.counts["faults"] += 1
255 source = page.slot < 0 ? "zero" : "swapin"
256 s.counts[source] += 1
257 result = "FAULT " * uppercase(source)
258 f = target!(s, p)
259 victim = s.frames[f]
260 if victim !== nothing
261 result *= " evict=$(label(victim))"
262 victim.frame = -1
263 s.processes[victim.pid].resident -= 1
264 removetlb!(s, victim)
265 if victim.m == 1
266 writeback!(s, victim)
267 result *= " writeback"
268 end
269 end
270 s.frames[f] = page
271 page.frame, page.load = f - 1, s.time
272 page.uses = page.m = page.age = 0
273 p.resident += 1
274 p.peak = max(p.peak, p.resident)
275 end
276 length(set) == div(c["tlb"], c["sets"]) && pop!(set)
277 pushfirst!(set, page)
278 page.r = 1
279 a.op == "W" && (page.m = 1)
280 page.last = s.time
281 page.uses += 1
282 page.seen = true
283 push!(p.history, n)
284 push!(s.history, fault)
285 physical = page.frame * c["page"] + off
286 line = "t=$(s.time) P$(a.pid) $(a.op) $(hex(a.address)) page=$n frame=$(page.frame) phys=$(hex(physical)) $result"
287 switched && (line *= " switch")
288 lines = [line]
289 if c["tick"] > 0 && mod(s.time, c["tick"]) == 0
290 for pg in s.frames
291 pg === nothing && continue
292 pg.age = div(pg.age, 2) + 128pg.r
293 pg.r = 0
294 end
295 push!(lines, "t=$(s.time) tick")
296 end
297 lines
298end
299function exit!(s, pid)
300 freed = slots = 0
301 for (i, pg) in enumerate(s.frames)
302 if pg !== nothing && pg.pid == pid
303 s.frames[i] = nothing
304 pg.frame = -1
305 removetlb!(s, pg)
306 freed += 1
307 end
308 end
309 for (i, pg) in enumerate(s.slots)
310 if pg !== nothing && pg.pid == pid
311 s.slots[i] = nothing
312 pg.slot = -1
313 slots += 1
314 end
315 end
316 p = s.processes[pid]
317 p.resident, p.exited = 0, true
318 freed, slots
319end
320function clean!(s)
321 written = 0
322 for pg in s.frames
323 if pg !== nothing && pg.m == 1
324 writeback!(s, pg)
325 written += 1
326 end
327 end
328 written
329end
330function lock!(s, e::Event)
331 for pg in s.processes[e.pid].pages[(e.first + 1):(e.last + 1)]
332 pg.locked = e.locked
333 end
334end
335function rerun(s, events, policy, frames)
336 c = copy(s.config)
337 c["policy"], c["frames"] = policy, frames
338 refs = [(e.pid, div(e.address, c["page"])) for e in events if e.kind == "ACCESS"]
339 r = Simulation(c, refs)
340 for id in pids(s)
341 r.processes[id] = VMProcess(id, length(s.processes[id].pages))
342 end
343 pos = 0
344 for e in events
345 if e.kind == "ACCESS"
346 pos += 1
347 access!(r, Access(e.pid, e.op, e.address), pos)
348 elseif e.kind == "EXIT"
349 exit!(r, e.pid)
350 elseif e.kind == "CLEAN"
351 clean!(r)
352 else
353 lock!(r, e)
354 end
355 end
356 r
357end
358function command!(s, words, position, events, trace)
359 cmd, v = words[1], words[2:end]
360 n(i) = num(v[i])
361 c = s.config
362 if cmd == "PROCESS"
363 id, count = n(1), n(2)
364 position > 0 && fail("processes are fixed")
365 haskey(s.processes, id) && fail("duplicate process $id")
366 1 <= count <= 256 || fail("invalid page count")
367 (length(s.processes) >= 8 || (c["scope"] == "LOCAL" && length(s.processes) >= c["frames"])) && fail("too many processes")
368 s.processes[id] = VMProcess(id, count)
369 elseif cmd == "EXIT"
370 p = process(s, n(1))
371 f, slots = exit!(s, p.pid)
372 push!(events, Event("EXIT"; pid=p.pid))
373 println("EXIT P$(p.pid) freed=$f slots=$slots")
374 elseif cmd in ("PROTECT", "LOCK", "UNLOCK")
375 p = process(s, n(1))
376 first, last = n(2), n(3)
377 (first > last || last >= length(p.pages)) && fail("invalid page range")
378 if cmd == "PROTECT"
379 for pg in p.pages[(first + 1):(last + 1)]
380 pg.ro = v[4] == "RO"
381 end
382 else
383 e = Event("LOCK"; pid=p.pid, first=first, last=last, locked=cmd == "LOCK")
384 lock!(s, e)
385 push!(events, e)
386 end
387 elseif cmd == "TRACE"
388 return v[1] == "ON"
389 elseif cmd == "SUMMARY"
390 println("SUMMARY policy=$(c["policy"]) scope=$(c["scope"]) time=$(s.time)")
391 k = s.counts
392 seg = sum((p.seg for p in values(s.processes)); init=0)
393 prot = sum((p.prot for p in values(s.processes)); init=0)
394 println("total faults=$(k["faults"]) zero=$(k["zero"]) swapin=$(k["swapin"]) faultrate=$(ratio(k["faults"], s.time)) tlbhits=$(k["tlbhits"]) tlbrate=$(ratio(k["tlbhits"], s.time)) writebacks=$(k["writebacks"]) switches=$(k["switches"]) segfaults=$seg protfaults=$prot")
395 for id in pids(s)
396 p = s.processes[id]
397 tail = c["scope"] == "LOCAL" ? " quota=$(quota(s, id))" : ""
398 p.exited && (tail *= " exited")
399 println("P$id accesses=$(length(p.history)) faults=$(p.faults) writebacks=$(p.writes) segfaults=$(p.seg) protfaults=$(p.prot) resident=$(p.resident) peak=$(p.peak)$tail")
400 end
401 elseif cmd == "FRAMES"
402 println("FRAMES used=$(count(!isnothing, s.frames))/$(c["frames"])")
403 for (i, pg) in enumerate(s.frames)
404 if pg === nothing
405 println("$(i - 1) free")
406 else
407 tail = pg.locked ? " locked" : ""
408 println("$(i - 1) $(label(pg)) R=$(pg.r) M=$(pg.m) load=$(pg.load) last=$(pg.last) uses=$(pg.uses) age=$(pg.age)$tail")
409 end
410 end
411 if c["policy"] in CLOCKS
412 println(c["scope"] == "GLOBAL" ? "hand=$(s.hand)" : "hand" * join((" P$id=$(s.processes[id].hand)" for id in pids(s))))
413 end
414 elseif cmd == "PAGETABLE"
415 p = process(s, n(1))
416 println("PAGETABLE P$(p.pid) pages=$(length(p.pages)) resident=$(p.resident)")
417 for pg in p.pages
418 pg.seen || continue
419 text = pg.frame >= 0 ? "frame=$(pg.frame) R=$(pg.r) M=$(pg.m)" : pg.slot >= 0 ? "swap=$(pg.slot)" : "zero"
420 pg.ro && (text *= " RO")
421 pg.locked && (text *= " locked")
422 println("$(pg.number) $text")
423 end
424 elseif cmd == "TLB"
425 println("TLB used=$(sum(length, s.tlb))/$(c["tlb"]) sets=$(c["sets"])")
426 for (i, set) in enumerate(s.tlb)
427 println("set $(i - 1):", join((" $(label(pg))->$(pg.frame)" for pg in set)))
428 end
429 elseif cmd == "SWAP"
430 println("SWAP used=$(count(!isnothing, s.slots))")
431 for (i, pg) in enumerate(s.slots)
432 pg !== nothing && println("$(i - 1) $(label(pg))")
433 end
434 elseif cmd == "CLEAN"
435 println("CLEAN written=$(clean!(s))")
436 push!(events, Event("CLEAN"))
437 elseif cmd == "COMPARE"
438 println("COMPARE accesses=$(s.time) frames=$(c["frames"])")
439 results = Tuple{Int,Int,Int,String}[]
440 for (i, policy) in enumerate(POLICIES)
441 r = rerun(s, events, policy, c["frames"])
442 k = r.counts
443 println("$policy faults=$(k["faults"]) writebacks=$(k["writebacks"]) tlbhits=$(k["tlbhits"])")
444 push!(results, (k["faults"], k["writebacks"], i, policy))
445 end
446 println("ranking ", join((x[4] for x in sort(results)), " "))
447 elseif cmd == "CURVE"
448 policy, lo, hi = v[1], n(2), n(3)
449 policy in POLICIES || fail("unknown policy")
450 (1 <= lo <= hi <= 16 && !(c["scope"] == "LOCAL" && lo < length(s.processes))) || fail("invalid frame range")
451 println("CURVE $policy frames=$lo..$hi")
452 previous = -1
453 for f in lo:hi
454 faults = rerun(s, events, policy, f).counts["faults"]
455 println("F=$f faults=$faults", previous >= 0 && faults > previous ? " ANOMALY" : "")
456 previous = faults
457 end
458 elseif cmd == "EAT"
459 if s.time == 0
460 println("EAT accesses=0")
461 else
462 total = n(1) * s.time + n(2) * (2s.time - s.counts["tlbhits"]) + n(3) * (s.counts["swapin"] + s.counts["writebacks"])
463 average = div(100total, s.time)
464 @printf("EAT accesses=%d total=%d average=%d.%02d\n", s.time, total, div(average, 100), mod(average, 100))
465 end
466 elseif cmd == "TIMELINE"
467 w = n(1)
468 w >= 1 || fail("invalid window")
469 totals = [sum(s.history[i:min(i + w - 1, end)]) for i in 1:w:length(s.history)]
470 println("TIMELINE window=$w", join((" $x" for x in totals)))
471 elseif cmd in ("WSET", "HOT")
472 p = process(s, n(1))
473 k = n(2)
474 k >= 1 || fail(cmd == "WSET" ? "invalid window" : "invalid count")
475 if cmd == "WSET"
476 pages = sort(unique(p.history[max(1, length(p.history) - k + 1):end]))
477 println("WSET P$(p.pid) window=$k size=$(length(pages))", join((" $x" for x in pages)))
478 else
479 counts = Dict{Int,Int}()
480 for pg in p.history
481 counts[pg] = get(counts, pg, 0) + 1
482 end
483 pages = sort(collect(keys(counts)); by=x -> (-counts[x], x))
484 println("HOT P$(p.pid)", join((" $pg:$(counts[pg])" for pg in pages[1:min(k, end)])))
485 end
486 end
487 trace
488end
489function expand(lines)
490 items = Union{Access,Vector{String}}[]
491 for line in lines
492 w = String.(split(line))
493 if isdigit(w[1][1])
494 push!(items, Access(num(w[1]), w[2], addr(w[3])))
495 elseif w[1] == "LOOP"
496 for _ in 1:num(w[7]), i in 0:(num(w[6]) - 1)
497 push!(items, Access(num(w[2]), w[3], addr(w[4]) + i * num(w[5])))
498 end
499 elseif w[1] == "RAND"
500 x = num(w[6])
501 lo, hi = addr(w[3]), addr(w[4])
502 for _ in 1:num(w[5])
503 x = mod(48271x, 2147483647)
504 address = lo + mod(x, hi - lo)
505 x = mod(48271x, 2147483647)
506 push!(items, Access(num(w[2]), mod(x, 100) < num(w[7]) ? "W" : "R", address))
507 end
508 else
509 push!(items, w)
510 end
511 end
512 items
513end
514function main()
515 n = num(readline())
516 items = expand([readline() for _ in 1:n])
517 s = nothing
518 trace = false
519 position = 0
520 events = Event[]
521 for (index, item) in enumerate(items)
522 try
523 if item isa Vector{String} && item[1] == "SYSTEM"
524 s = nothing
525 c = settings(item[2:end])
526 refs = Tuple{Int,Int}[]
527 for future in items[(index + 1):end]
528 future isa Vector{String} && future[1] == "SYSTEM" && break
529 future isa Access && push!(refs, (future.pid, div(future.address, c["page"])))
530 end
531 s = Simulation(c, refs)
532 trace, position, events = false, 0, Event[]
533 println("SYSTEM ", join(("$k=$(c[k])" for k in ("page", "frames", "tlb", "sets", "policy", "scope", "asid", "tick", "tau", "seed")), " "))
534 else
535 s === nothing && fail("no system")
536 if item isa Access
537 position += 1
538 p = process(s, item.pid)
539 if item.address >= length(p.pages) * s.config["page"]
540 p.seg += 1
541 println("SEGFAULT P$(item.pid) $(hex(item.address))")
542 elseif item.op == "W" && p.pages[div(item.address, s.config["page"]) + 1].ro
543 p.prot += 1
544 println("PROTFAULT P$(item.pid) $(hex(item.address))")
545 else
546 lines = access!(s, item, position)
547 push!(events, Event("ACCESS"; pid=item.pid, op=item.op, address=item.address))
548 trace && foreach(println, lines)
549 end
550 else
551 trace = command!(s, item, position, events, trace)
552 end
553 end
554 catch e
555 e isa CommandError || rethrow()
556 println("ERROR ", e.text)
557 end
558 end
559end
