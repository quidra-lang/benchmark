18data class Config(
19 var page: Int = 256,
20 var frames: Int = 4,
21 var tlb: Int = 4,
22 var sets: Int = 1,
23 var tick: Int = 0,
24 var tau: Int = 8,

31 "SYSTEM page=$page frames=$frames tlb=$tlb sets=$sets policy=$policy scope=$scope asid=$asid tick=$tick tau=$tau seed=$seed"

34fun settings(tokens: List<String>): Config {
35 val c = Config()
36 val seen = mutableSetOf<String>()
37 for (token in tokens.drop(1)) {
38 val (key, value) = token.split('=')
39 if (
40 key !in
41 listOf(
42 "page",
43 "frames",
44 "tlb",
45 "sets",
46 "policy",
47 "scope",
48 "asid",
49 "tick",
50 "tau",
51 "seed",
52 )
53 )
54 throw CommandError("unknown key $key")
55 if (!seen.add(key)) throw CommandError("duplicate key $key")
56 val valid: Boolean
57 when (key) {
58 "policy" -> {
59 valid = value in policies
60 if (valid) c.policy = value
61 }
62 "scope" -> {
63 valid = value in listOf("GLOBAL", "LOCAL")
64 if (valid) c.scope = value
65 }
66 "asid" -> {
67 valid = value in listOf("ON", "OFF")
68 if (valid) c.asid = value
69 }
70 else -> {
71 val n = value.toLong()
72 valid =
73 when (key) {
74 "page" -> n in 16..4096L && (n and (n - 1)) == 0L
75 "frames" -> n in 1..16L
76 "tlb",
77 "sets" -> n in 1..8L
78 "tick" -> n in 0..1000L
79 "tau" -> n in 1..1000L
80 else -> n in 1..2147483646L
81 }
82 if (valid)
83 when (key) {
84 "page" -> c.page = n.toInt()
85 "frames" -> c.frames = n.toInt()
86 "tlb" -> c.tlb = n.toInt()
87 "sets" -> c.sets = n.toInt()
88 "tick" -> c.tick = n.toInt()
89 "tau" -> c.tau = n.toInt()
90 "seed" -> c.seed = n
91 }
92 }
93 }
94 if (!valid) throw CommandError("invalid $key")
95 }
96 if (c.tlb % c.sets != 0) throw CommandError("invalid sets")
97 return c
98}

136class Simulation(val c: Config, val future: List<Access>, val silent: Boolean) {
137 val processes = sortedMapOf<Int, Process>()
138 val frames = arrayOfNulls<Page>(c.frames)
139 val tlb = List(c.sets) { mutableListOf<Page>() }
140 val swap = sortedMapOf<Int, Page>()
141 var cursor = -1
142 var time = 0
143 var faults = 0
144 var zero = 0
145 var swapin = 0
146 var tlbhits = 0
147 var writebacks = 0
148 var switches = 0
149 var previous = -1
150 var hand = 0
151 var random = c.seed
152 var fixed = false
153 var trace = false
154 val faultHistory = mutableListOf<Boolean>()
155 val log = mutableListOf<Log>()

157 fun emit(text: String) {
158 if (!silent) output.add(text)

161 fun process(pid: Int): Process {
162 val p = processes[pid] ?: throw CommandError("unknown process $pid")
163 if (p.exited) throw CommandError("process $pid exited")
164 return p
165 }
166
167 fun quota(pid: Int) =
168 c.frames / processes.size +
169 if (processes.keys.indexOf(pid) < c.frames % processes.size) 1 else 0
170
171 fun declare(pid: Int, count: Int) {
172 if (fixed) throw CommandError("processes are fixed")
173 if (pid in processes) throw CommandError("duplicate process $pid")
174 if (count !in 1..256) throw CommandError("invalid page count")
175 if (processes.size >= 8 || c.scope == "LOCAL" && processes.size == c.frames)
176 throw CommandError("too many processes")
177 processes[pid] = Process(pid, count)
178 }
179
180 fun invalidate(page: Page) {
181 tlb.forEach { it.remove(page) }
182 }
183
184 fun write(p: Page) {
185 writebacks++
186 processes.getValue(p.pid).writebacks++
187 p.m = 0
188 if (p.slot < 0) {
189 var slot = 0
190 while (slot in swap) slot++
191 p.slot = slot
192 swap[slot] = p
193 }
194 }
195
196 fun nextUse(p: Page): Int {
197 for (i in cursor + 1 until future.size) {
198 val a = future[i]
199 if (a.pid == p.pid && a.address / c.page == p.number) return i
200 }
201 return Int.MAX_VALUE
202 }

204 fun choose(original: List<Int>, owner: Process): Int {
205 val unlocked = original.filter { !frames[it]!!.locked }
206 val candidates = (if (unlocked.isNotEmpty()) unlocked else original).toMutableList()
207 if (c.policy == "RANDOM") {
208 random = random * 48271L % 2147483647L
209 return candidates[(random % candidates.size).toInt()]
210 }
211 if (c.policy !in listOf("CLOCK", "ECLOCK", "WSCLOCK")) {
212 fun rank(f: Int): Pair<Int, Int> {
213 val p = frames[f]!!
214 return when (c.policy) {
215 "FIFO" -> p.load to 0
216 "LRU" -> p.last to 0
217 "LFU" -> p.uses to p.load
218 "NRU" -> (2 * p.r + p.m) to f
219 "AGING" -> p.age to p.load
220 else -> -nextUse(p) to f
221 }
222 }
223 return candidates.minWith(compareBy<Int> { rank(it).first }.thenBy { rank(it).second })
224 }
225 val h = if (c.scope == "GLOBAL") hand else owner.hand
226 candidates.sortBy { (it - h + c.frames) % c.frames }
227 var chosen = -1
228 when (c.policy) {
229 "CLOCK" -> {
230 while (chosen < 0) for (f in candidates) {
231 if (frames[f]!!.r == 0) {
232 chosen = f
233 break
234 }
235 frames[f]!!.r = 0
236 }
237 }
238 "ECLOCK" -> {
239 while (chosen < 0) {
240 for (f in candidates) if (frames[f]!!.r == 0 && frames[f]!!.m == 0) {
241 chosen = f
242 break
243 }
244 if (chosen >= 0) break
245 for (f in candidates) {
246 if (frames[f]!!.r == 0 && frames[f]!!.m == 1) {
247 chosen = f
248 break
249 }
250 frames[f]!!.r = 0
251 }
252 }
253 }
254 else -> {
255 for (f in candidates) {
256 val p = frames[f]!!
257 if (p.r == 1) {
258 p.r = 0
259 p.last = time
260 } else if (time - p.last > c.tau) {
261 if (p.m == 0) {
262 chosen = f
263 break
264 }
265 write(p)
266 }
267 }
268 if (chosen < 0)
269 chosen = candidates.firstOrNull { frames[it]!!.m == 0 } ?: candidates.first()
270 }
271 }
272 if (c.scope == "GLOBAL") hand = (chosen + 1) % c.frames
273 else owner.hand = (chosen + 1) % c.frames
274 return chosen
275 }

277 fun access(pid: Int, op: String, addr: Int) {
278 fixed = true
279 cursor++
280 val owner = process(pid)
281 if (addr >= owner.pages.size * c.page) {
282 owner.segfaults++
283 emit("SEGFAULT P$pid ${hex(addr)}")
284 return
285 }
286 val p = owner.pages[addr / c.page]
287 if (op == "W" && p.ro) {
288 owner.protfaults++
289 emit("PROTFAULT P$pid ${hex(addr)}")
290 return
291 }
292 time++
293 owner.accesses++
294 val switched = previous >= 0 && previous != pid
295 if (switched) {
296 switches++
297 if (c.asid == "OFF") tlb.forEach { it.clear() }
298 }
299 previous = pid
300 val set = tlb[p.number % c.sets]
301 val hit = set.remove(p)
302 var fault = false
303 var result: String
304 if (hit) {
305 tlbhits++
306 result = "TLBHIT"
307 } else if (p.frame >= 0) result = "HIT"
308 else {
309 fault = true
310 faults++
311 owner.faults++
312 if (p.slot >= 0) {
313 swapin++
314 result = "FAULT SWAPIN"
315 } else {
316 zero++
317 result = "FAULT ZERO"
318 }
319 var frame = -1
320 val mayUseFree = c.scope == "GLOBAL" || owner.resident() < quota(pid)
321 if (mayUseFree) frame = frames.indexOfFirst { it == null }
322 if (frame < 0) {
323 val candidates =
324 frames.indices.filter {
325 frames[it] != null && (c.scope == "GLOBAL" || frames[it]!!.pid == pid)
326 }
327 frame = choose(candidates, owner)
328 val victim = frames[frame]!!
329 result += " evict=${victim.label()}"
330 invalidate(victim)
331 if (victim.m == 1) {
332 write(victim)
333 result += " writeback"
334 }
335 victim.frame = -1
336 }
337 frames[frame] = p
338 p.frame = frame
339 p.load = time
340 p.uses = 0
341 p.m = 0
342 p.age = 0
343 owner.peak = max(owner.peak, owner.resident())
344 }
345 if (set.size == c.tlb / c.sets) set.removeAt(0)
346 set.add(p)
347 p.r = 1
348 if (op == "W") p.m = 1
349 p.last = time
350 p.uses++
351 p.ever = true
352 p.references++
353 owner.history.add(p.number)
354 faultHistory.add(fault)
355 log.add(Log("ACCESS", pid, op, addr, 0))
356 val interrupt = c.tick > 0 && time % c.tick == 0
357 if (interrupt)
358 for (page in frames.filterNotNull()) {
359 page.age = page.age / 2 + 128 * page.r
360 page.r = 0
361 }
362 if (trace) {
363 emit(
364 "t=$time P$pid $op ${hex(addr)} page=${p.number} frame=${p.frame} phys=${hex(p.frame * c.page + addr % c.page)} $result" +
365 if (switched) " switch" else "",
366 )
367 if (interrupt) emit("t=$time tick")
368 }
369 }

371 fun exit(pid: Int) {
372 val p = process(pid)
373 var count = 0
374 var slots = 0
375 for (page in p.pages) {
376 if (page.frame >= 0) {
377 frames[page.frame] = null
378 page.frame = -1
379 count++
380 }
381 invalidate(page)
382 if (page.slot >= 0) {
383 swap.remove(page.slot)
384 page.slot = -1
385 slots++
386 }
387 }
388 p.exited = true
389 log.add(Log("EXIT", pid, "", 0, 0))
390 emit("EXIT P$pid freed=$count slots=$slots")
391 }

393 fun attributes(cmd: String, pid: Int, first: Int, last: Int, protection: String) {
394 val p = process(pid)
395 if (first > last || last >= p.pages.size) throw CommandError("invalid page range")
396 for (i in first..last) {
397 if (cmd == "PROTECT") p.pages[i].ro = protection == "RO"
398 else p.pages[i].locked = cmd == "LOCK"
399 }
400 if (cmd != "PROTECT") log.add(Log(cmd, pid, "", first, last))
401 }

403 fun clean() {
404 var count = 0
405 for (p in frames.filterNotNull()) if (p.m == 1) {
406 write(p)
407 count++
408 }
409 log.add(Log("CLEAN", 0, "", 0, 0))
410 emit("CLEAN written=$count")
411 }

413 fun rerun(policy: String, count: Int): Simulation {
414 val config = c.copy(policy = policy, frames = count)
415 val references = log.filter { it.kind == "ACCESS" }.map { Access(it.pid, it.op, it.a) }
416 val replay = Simulation(config, references, true)
417 for (p in processes.values) replay.declare(p.pid, p.pages.size)
418 for (e in log) when (e.kind) {
419 "ACCESS" -> replay.access(e.pid, e.op, e.a)
420 "EXIT" -> replay.exit(e.pid)
421 "CLEAN" -> replay.clean()
422 else -> replay.attributes(e.kind, e.pid, e.a, e.b, "")
423 }
424 return replay
425 }

427 fun summary() {
428 emit("SUMMARY policy=${c.policy} scope=${c.scope} time=$time")
429 val seg = processes.values.sumOf { it.segfaults }
430 val prot = processes.values.sumOf { it.protfaults }
431 emit(
432 "total faults=$faults zero=$zero swapin=$swapin faultrate=${ratio(faults, time)} tlbhits=$tlbhits tlbrate=${ratio(tlbhits, time)} writebacks=$writebacks switches=$switches segfaults=$seg protfaults=$prot",
433 )
434 for (p in processes.values) emit(
435 "P${p.pid} accesses=${p.accesses} faults=${p.faults} writebacks=${p.writebacks} segfaults=${p.segfaults} protfaults=${p.protfaults} resident=${p.resident()} peak=${p.peak}" +
436 (if (c.scope == "LOCAL") " quota=${quota(p.pid)}" else "") +
437 if (p.exited) " exited" else "",
438 )
439 }

441 fun frameDump() {
442 emit("FRAMES used=${frames.count { it != null }}/${c.frames}")
443 for ((i, p) in frames.withIndex()) {
444 if (p == null) emit("$i free")
445 else
446 emit(

448 if (p.locked) " locked" else "",
449 )
450 }
451 if (c.policy in listOf("CLOCK", "ECLOCK", "WSCLOCK")) {
452 if (c.scope == "GLOBAL") emit("hand=$hand")
453 else emit("hand" + processes.values.joinToString("") { " P${it.pid}=${it.hand}" })
454 }

457 fun pageTable(pid: Int) {
458 val p = process(pid)
459 emit("PAGETABLE P$pid pages=${p.pages.size} resident=${p.resident()}")
460 for (page in p.pages.filter { it.ever }) {
461 val info =
462 if (page.frame >= 0) "frame=${page.frame} R=${page.r} M=${page.m}"
463 else if (page.slot >= 0) "swap=${page.slot}" else "zero"

466 (if (page.ro) " RO" else "") +
467 if (page.locked) " locked" else "",
468 )

479 fun compare() {
480 emit("COMPARE accesses=$time frames=${c.frames}")
481 val results = mutableMapOf<String, Simulation>()
482 for (policy in policies) {
483 val r = rerun(policy, c.frames)
484 results[policy] = r
485 emit("$policy faults=${r.faults} writebacks=${r.writebacks} tlbhits=${r.tlbhits}")
486 }
487 val ranking =
488 policies.sortedWith(
489 compareBy<String> { results.getValue(it).faults }
490 .thenBy { results.getValue(it).writebacks },
491 )
492 emit("ranking " + ranking.joinToString(" "))
493 }

495 fun execute(t: List<String>) {
496 val cmd = t[0]
497 fun n(i: Int) = t[i].toInt()
498 if (cmd[0].isDigit()) {
499 access(n(0), t[1], address(t[2]))
500 return
501 }
502 when (cmd) {
503 "PROCESS" -> declare(n(1), n(2))
504 "EXIT" -> exit(n(1))
505 "PROTECT",
506 "LOCK",
507 "UNLOCK" -> attributes(cmd, n(1), n(2), n(3), t.getOrElse(4) { "" })
508 "TRACE" -> trace = t[1] == "ON"
509 "SUMMARY" -> summary()
510 "FRAMES" -> frameDump()
511 "PAGETABLE" -> pageTable(n(1))
512 "TLB" -> tlbDump()
513 "SWAP" -> {
514 emit("SWAP used=${swap.size}")
515 for ((slot, page) in swap) emit("$slot ${page.label()}")
516 }
517 "CLEAN" -> clean()
518 "COMPARE" -> compare()
519 "CURVE" -> {
520 val policy = t[1]
521 val lo = n(2)
522 val hi = n(3)
523 if (policy !in policies) throw CommandError("unknown policy")
524 if (lo < 1 || lo > hi || hi > 16 || c.scope == "LOCAL" && lo < processes.size)
525 throw CommandError("invalid frame range")
526 emit("CURVE $policy frames=$lo..$hi")
527 var prior = Int.MAX_VALUE
528 for (f in lo..hi) {
529 val value = rerun(policy, f).faults
530 emit("F=$f faults=$value" + if (value > prior) " ANOMALY" else "")
531 prior = value
532 }
533 }
534 "EAT" -> {
535 val total =
536 t[1].toLong() * time +
537 t[2].toLong() * (2L * time - tlbhits) +
538 t[3].toLong() * (swapin + writebacks)
539 emit(
540 if (time == 0) "EAT accesses=0"
541 else "EAT accesses=$time total=$total average=${decimal(total * 100 / time)}",
542 )
543 }
544 "TIMELINE" -> {
545 val w = n(1)
546 if (w < 1) throw CommandError("invalid window")
547 val text = StringBuilder("TIMELINE window=$w")
548 for (i in faultHistory.indices step w) text
549 .append(' ')
550 .append(faultHistory.subList(i, min(i + w, faultHistory.size)).count { it })
551 emit(text.toString())
552 }
553 "WSET" -> {
554 val pid = n(1)
555 val p = process(pid)
556 val w = n(2)
557 if (w < 1) throw CommandError("invalid window")
558 val pages = p.history.takeLast(w).toSortedSet()
559 emit("WSET P$pid window=$w size=${pages.size}" + pages.joinToString("") { " $it" })
560 }
561 "HOT" -> {
562 val pid = n(1)
563 val p = process(pid)
564 val k = n(2)
565 if (k < 1) throw CommandError("invalid count")
566 val pages =
567 p.pages
568 .filter { it.references > 0 }
569 .sortedWith(
570 compareByDescending<Page> { it.references }.thenBy { it.number },
571 )
572 .take(k)
573 emit("HOT P$pid" + pages.joinToString("") { " ${it.number}:${it.references}" })
574 }
575 }
576 }
577}

579fun main() {
580 val input = System.`in`.bufferedReader()
581 val commands = mutableListOf<List<String>>()
582 repeat(input.readLine().toInt()) {
583 val t = input.readLine().split(' ')
584 when (t[0]) {
585 "LOOP" -> {
586 val start = address(t[3])
587 val interval = t[4].toInt()
588 val length = t[5].toInt()
589 val repeats = t[6].toInt()
590 repeat(repeats) {
591 for (j in 0 until length) commands.add(
592 listOf(t[1], t[2], hex(start + j * interval)),
593 )
594 }
595 }
596 "RAND" -> {
597 val lo = address(t[2])
598 val hi = address(t[3])
599 val count = t[4].toInt()
600 val rate = t[6].toInt()
601 var x = t[5].toLong()
602 repeat(count) {
603 x = x * 48271L % 2147483647L
604 val addr = lo + (x % (hi - lo)).toInt()
605 x = x * 48271L % 2147483647L
606 commands.add(listOf(t[1], if (x % 100 < rate) "W" else "R", hex(addr)))
607 }
608 }
609 else -> commands.add(t)
610 }
611 }
612 var simulation: Simulation? = null
613 for ((i, t) in commands.withIndex()) {
614 try {
615 if (t[0] == "SYSTEM") {
616 simulation = null
617 val c = settings(t)
618 val future = mutableListOf<Access>()
619 for (j in i + 1 until commands.size) {
620 val entry = commands[j]
621 if (entry[0] == "SYSTEM") break
622 if (entry[0][0].isDigit())
623 future.add(Access(entry[0].toInt(), entry[1], address(entry[2])))
624 }
625 simulation = Simulation(c, future, false)
626 output.add(c.description())
627 } else if (simulation == null) output.add("ERROR no system") else simulation.execute(t)
628 } catch (e: CommandError) {
629 output.add("ERROR ${e.message}")
630 }
631 }
632 if (output.isNotEmpty()) print(output.joinToString("\n", postfix = "\n"))
633}
