val policies =
    listOf("FIFO", "LRU", "LFU", "NRU", "AGING", "CLOCK", "ECLOCK", "WSCLOCK", "RANDOM", "OPT")
data class Config(
    var page: Int = 256,
    var frames: Int = 4,
    var tlb: Int = 4,
    var sets: Int = 1,
    var tick: Int = 0,
    var tau: Int = 8,
    var seed: Long = 1,
    var policy: String = "FIFO",
    var scope: String = "GLOBAL",
    var asid: String = "OFF",
) {
    fun description() =
        "SYSTEM page=$page frames=$frames tlb=$tlb sets=$sets policy=$policy scope=$scope asid=$asid tick=$tick tau=$tau seed=$seed"
fun settings(tokens: List<String>): Config {
    val c = Config()
    val seen = mutableSetOf<String>()
    for (token in tokens.drop(1)) {
        val (key, value) = token.split('=')
        if (
            key !in
                listOf(
                    "page",
                    "frames",
                    "tlb",
                    "sets",
                    "policy",
                    "scope",
                    "asid",
                    "tick",
                    "tau",
                    "seed",
                )
        )
            throw CommandError("unknown key $key")
        if (!seen.add(key)) throw CommandError("duplicate key $key")
        val valid: Boolean
        when (key) {
            "policy" -> {
                valid = value in policies
                if (valid) c.policy = value
            }
            "scope" -> {
                valid = value in listOf("GLOBAL", "LOCAL")
                if (valid) c.scope = value
            }
            "asid" -> {
                valid = value in listOf("ON", "OFF")
                if (valid) c.asid = value
            }
            else -> {
                val n = value.toLong()
                valid =
                    when (key) {
                        "page" -> n in 16..4096L && (n and (n - 1)) == 0L
                        "frames" -> n in 1..16L
                        "tlb",
                        "sets" -> n in 1..8L
                        "tick" -> n in 0..1000L
                        "tau" -> n in 1..1000L
                        else -> n in 1..2147483646L
                    }
                if (valid)
                    when (key) {
                        "page" -> c.page = n.toInt()
                        "frames" -> c.frames = n.toInt()
                        "tlb" -> c.tlb = n.toInt()
                        "sets" -> c.sets = n.toInt()
                        "tick" -> c.tick = n.toInt()
                        "tau" -> c.tau = n.toInt()
                        "seed" -> c.seed = n
                    }
            }
        }
        if (!valid) throw CommandError("invalid $key")
    }
    if (c.tlb % c.sets != 0) throw CommandError("invalid sets")
    return c
}
