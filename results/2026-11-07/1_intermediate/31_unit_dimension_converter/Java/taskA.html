20 record Fraction(long n, long d) implements Comparable<Fraction> {
21 static final Fraction ONE = new Fraction(1, 1);
22 Fraction {
23 if (d < 0) { n = -n; d = -d; }
24 long a = Math.abs(n), b = d;
25 while (b != 0) { long remainder = a % b; a = b; b = remainder; }
26 n /= a; d /= a;
27 }
28 Fraction(long n) { this(n, 1); }
29 Fraction add(Fraction b) { return new Fraction(n*b.d + b.n*d, d*b.d); }
30 Fraction sub(Fraction b) { return new Fraction(n*b.d - b.n*d, d*b.d); }
31 Fraction mul(Fraction b) { return new Fraction(n*b.n, d*b.d); }
32 Fraction div(Fraction b) { return new Fraction(n*b.d, d*b.n); }
33 Fraction neg() { return new Fraction(-n, d); }
34 Fraction abs() { return new Fraction(Math.abs(n), d); }
35 Fraction power(int exponent) {
36 Fraction factor = exponent < 0 ? new Fraction(d, n) : this;
37 Fraction result = ONE;
38 for (int i = 0; i < Math.abs(exponent); i++) result = result.mul(factor);
39 return result;
40 }
41 public int compareTo(Fraction b) { return Long.compare(n*b.d, b.n*d); }
42 public String toString() { return d == 1 ? Long.toString(n) : n + "/" + d; }
43 String decimal() {
44 long magnitude = Math.abs(n);
45 return String.format(Locale.ROOT, "%s%d.%06d", n < 0 ? "-" : "", magnitude/d, magnitude%d*1_000_000/d);
46 }
47 String notation(String unit) { return this + " " + unit + " (" + decimal() + ")"; }
48 }

99 static boolean letter(char c) { return c >= 'A' && c <= 'Z' || c >= 'a' && c <= 'z'; }
100 static boolean digit(char c) { return c >= '0' && c <= '9'; }
101 static boolean digits(String text) {
102 if (text.isEmpty()) return false;
103 for (int i = 0; i < text.length(); i++) if (!digit(text.charAt(i))) return false;
104 return true;
105 }
106 static Fraction number(String text) {
107 String body = text.startsWith("-") ? text.substring(1) : text;
108 Fraction value;
109 int dot = body.indexOf('.'), slash = body.indexOf('/');
110 if (digits(body)) value = new Fraction(Long.parseLong(body));
111 else if (dot >= 0) {
112 String whole = body.substring(0, dot), tail = body.substring(dot+1);
113 if (!digits(whole) || !digits(tail)) throw new CommandError("BAD NUMBER " + text);
114 long scale = 1;
115 for (int i = 0; i < tail.length(); i++) scale *= 10;
116 value = new Fraction(Long.parseLong(whole)*scale + Long.parseLong(tail), scale);
117 } else if (slash >= 0) {
118 String n = body.substring(0, slash), d = body.substring(slash+1);
119 if (!digits(n) || !digits(d) || Long.parseLong(d) == 0) throw new CommandError("BAD NUMBER " + text);
120 value = new Fraction(Long.parseLong(n), Long.parseLong(d));
121 } else throw new CommandError("BAD NUMBER " + text);
122 return text.startsWith("-") ? value.neg() : value;
123 }
124 static int exponent(String text) {
125 long value = Long.parseLong(text);
126 if (value == 0 || value < -3 || value > 3) throw new CommandError("BAD EXPONENT " + text);
127 return (int)value;
128 }
129 static void checkName(String name, boolean variable) {
130 if (name.isEmpty() || name.length() > 12 || !name.chars().allMatch(c -> letter((char)c))
131 || variable && List.of("abs", "sqrt", "min", "max").contains(name)) throw new CommandError("BAD NAME " + name);
132 }
133 static long integerRoot(long n) {
134 if (n < 2) return n;
135 long x = n, next = (n+1)/2;
136 while (next < x) { x = next; next = (x+n/x)/2; }
137 return x;
138 }

156 static class Parser {
157 final String source;
158 final List<Token> tokens = new ArrayList<>();
159 int position;
160 Parser(String source, boolean quantity) {
161 this.source = source;
162 for (int i = 0; i < source.length();) {
163 int start = i, kind;
164 char c = source.charAt(i);
165 if (letter(c)) {
166 kind = IDENTIFIER;
167 while (i < source.length() && letter(source.charAt(i))) i++;
168 } else if (digit(c)) {
169 kind = NUMBER;
170 while (i < source.length() && digit(source.charAt(i))) i++;
171 if (quantity && i+1 < source.length() && source.charAt(i) == '.' && digit(source.charAt(i+1))) {
172 i++;
173 while (i < source.length() && digit(source.charAt(i))) i++;
174 }
175 } else if (quantity && c == '[' && source.indexOf(']', i+1) > i+1) {
176 kind = UNIT; i = source.indexOf(']', i+1)+1;
177 } else {
178 kind = (quantity ? "+-*/^()," : "*/^()-").indexOf(c) >= 0 ? c : INVALID;
179 i++;
180 }
181 tokens.add(new Token(kind, source.substring(start, i), start));
182 }
183 tokens.add(new Token(END, "", source.length()));
184 }
185 Token current() { return tokens.get(position); }
186 boolean take(int kind) {
187 if (current().kind != kind) return false;
188 position++; return true;
189 }
190 CommandError syntax() {
191 return new CommandError("SYNTAX " + source + " AT " + (current().kind == END ? "END" : current().position+1));
192 }
193 void expect(int kind) { if (!take(kind)) throw syntax(); }
194 String readExponent() {
195 int start = current().position;
196 take('-');
197 Token token = current();
198 if (token.kind != NUMBER || token.text.contains(".")) throw syntax();
199 position++;
200 return source.substring(start, token.position+token.text.length());
201 }
202 UnitNode unitTerm() {
203 Token token = current();
204 UnitNode result;
205 if (take(IDENTIFIER)) result = new UnitName(token.text);
206 else if (token.kind == NUMBER && token.text.charAt(0) != '0') { position++; result = new UnitNumber(token.text); }
207 else if (take('(')) { result = unitExpression(); expect(')'); }
208 else throw syntax();
209 return take('^') ? new UnitPower(result, readExponent()) : result;
210 }
211 UnitNode unitExpression() {
212 List<UnitTerm> terms = new ArrayList<>();
213 terms.add(new UnitTerm('*', unitTerm()));
214 while (current().kind == '*' || current().kind == '/') {
215 char op = (char)current().kind; position++;
216 terms.add(new UnitTerm(op, unitTerm()));
217 }
218 return new UnitProduct(List.copyOf(terms));
219 }
220 UnitNode completeUnit() {
221 UnitNode result = unitExpression();
222 if (current().kind != END) throw syntax();
223 return result;
224 }
225 Expr primary() {
226 Token token = current();
227 Expr result;
228 if (take(NUMBER)) {
229 String unit = null;
230 if (current().kind == UNIT) {
231 String text = current().text; position++;
232 unit = text.substring(1, text.length()-1);
233 }
234 result = new Literal(token.text, unit);
235 } else if (take(UNIT)) result = new Literal("1", token.text.substring(1, token.text.length()-1));
236 else if (take(IDENTIFIER)) {
237 if (take('(')) {
238 List<Expr> args = new ArrayList<>();
239 args.add(expression());
240 while (take(',')) args.add(expression());
241 expect(')'); result = new Call(token.text, List.copyOf(args));
242 } else result = new Variable(token.text);
243 } else if (take('(')) { result = expression(); expect(')'); }
244 else throw syntax();
245 return take('^') ? new Power(result, readExponent()) : result;
246 }
247 Expr unary() { return take('-') ? new Negative(unary()) : primary(); }
248 Expr product() {
249 Expr result = unary();
250 while (current().kind == '*' || current().kind == '/') {
251 char op = (char)current().kind; position++;
252 result = new Binary(op, result, unary());
253 }
254 return result;
255 }
256 Expr expression() {
257 Expr result = product();
258 while (current().kind == '+' || current().kind == '-') {
259 char op = (char)current().kind; position++;
260 result = new Binary(op, result, product());
261 }
262 return result;
263 }
264 Expr complete() {
265 Expr result = expression();
266 if (current().kind != END) throw syntax();
267 return result;
268 }
269 }

271 record Reduced(Fraction coefficient, LinkedHashMap<String, Long> names) {}
272 static Reduced reduce(UnitNode node) {
273 return switch (node) {
274 case UnitName atom -> {
275 LinkedHashMap<String, Long> names = new LinkedHashMap<>();
276 names.put(atom.name, 1L);
277 yield new Reduced(Fraction.ONE, names);
278 }
279 case UnitNumber atom -> new Reduced(new Fraction(Long.parseLong(atom.text)), new LinkedHashMap<>());
280 case UnitPower power -> {
281 Reduced inner = reduce(power.child);
282 int k = Integer.parseInt(power.exponent);
283 inner.names.replaceAll((name, count) -> count*k);
284 yield new Reduced(inner.coefficient.power(k), inner.names);
285 }
286 case UnitProduct product -> {
287 Fraction coefficient = Fraction.ONE;
288 LinkedHashMap<String, Long> names = new LinkedHashMap<>();
289 for (UnitTerm term : product.terms) {
290 Reduced right = reduce(term.node);
291 long sign = term.op == '*' ? 1 : -1;
292 coefficient = sign == 1 ? coefficient.mul(right.coefficient) : coefficient.div(right.coefficient);
293 for (var entry : right.names.entrySet()) names.merge(entry.getKey(), sign*entry.getValue(), Long::sum);
294 }
295 yield new Reduced(coefficient, names);
296 }
297 };
298 }
299 static String reducedText(UnitNode tree) {
300 Reduced reduced = reduce(tree);
301 List<String> parts = new ArrayList<>();
302 if (!reduced.coefficient.equals(Fraction.ONE)) parts.add(reduced.coefficient.toString());
303 for (var entry : reduced.names.entrySet()) {
304 long k = entry.getValue();
305 if (k != 0) parts.add(k == 1 ? entry.getKey() : entry.getKey() + "^" + k);
306 }
307 return parts.isEmpty() ? "1" : String.join(" ", parts);
308 }

354 void validateUnit(UnitNode node) {
355 switch (node) {
356 case UnitName atom -> { if (resolve(atom.name) == null) throw new CommandError("UNKNOWN UNIT " + atom.name); }
357 case UnitNumber atom -> {}
358 case UnitPower power -> { validateUnit(power.child); exponent(power.exponent); }
359 case UnitProduct product -> { for (UnitTerm term : product.terms) validateUnit(term.node); }
360 }
361 }
362 Quantity unitValue(UnitNode node) {
363 return switch (node) {
364 case UnitName atom -> resolve(atom.name);
365 case UnitNumber atom -> new Quantity(new Fraction(Long.parseLong(atom.text)), Dimension.NONE);
366 case UnitPower power -> {
367 Quantity inner = unitValue(power.child);
368 int k = Integer.parseInt(power.exponent);
369 yield new Quantity(inner.value.power(k), inner.dim.scale(k));
370 }
371 case UnitProduct product -> {
372 Quantity result = new Quantity(Fraction.ONE, Dimension.NONE);
373 for (UnitTerm term : product.terms) {
374 Quantity right = unitValue(term.node);
375 result = term.op == '*'
376 ? new Quantity(result.value.mul(right.value), result.dim.combine(right.dim, 1))
377 : new Quantity(result.value.div(right.value), result.dim.combine(right.dim, -1));
378 }
379 yield result;
380 }

390 Quantity evaluate(Expr expression) {
391 return switch (expression) {
392 case Literal literal -> {
393 if (literal.unit == null) yield new Quantity(number(literal.number), Dimension.NONE);
394 Quantity unit = unit(literal.unit);
395 yield new Quantity(number(literal.number).mul(unit.value), unit.dim);
396 }
397 case Variable variable -> {
398 Quantity value = variables.get(variable.name);
399 if (value == null) throw new CommandError("UNKNOWN VARIABLE " + variable.name);
400 yield value;
401 }
402 case Negative negative -> {
403 Quantity value = evaluate(negative.child);
404 yield new Quantity(value.value.neg(), value.dim);
405 }
406 case Power power -> {
407 Quantity value = evaluate(power.child);
408 int k = exponent(power.exponent);
409 if (value.value.n == 0 && k < 0) throw new CommandError("DIVISION BY ZERO");
410 yield new Quantity(value.value.power(k), value.dim.scale(k));
411 }
412 case Binary binary -> {
413 Quantity a = evaluate(binary.left), b = evaluate(binary.right);
414 if (binary.op == '+' || binary.op == '-') {
415 a.dim.match(b.dim);
416 yield new Quantity(binary.op == '+' ? a.value.add(b.value) : a.value.sub(b.value), a.dim);
417 }
418 if (binary.op == '/' && b.value.n == 0) throw new CommandError("DIVISION BY ZERO");
419 yield binary.op == '*' ? new Quantity(a.value.mul(b.value), a.dim.combine(b.dim, 1))
420 : new Quantity(a.value.div(b.value), a.dim.combine(b.dim, -1));
421 }
422 case Call call -> {
423 int arity = switch (call.name) {
424 case "abs", "sqrt" -> 1;
425 case "min", "max" -> 2;
426 default -> throw new CommandError("UNKNOWN FUNCTION " + call.name);
427 };
428 if (call.args.size() != arity) throw new CommandError("BAD ARGUMENTS " + call.name);
429 Quantity a = evaluate(call.args.getFirst());
430 if (call.name.equals("abs")) yield new Quantity(a.value.abs(), a.dim);
431 if (call.name.equals("sqrt")) {
432 if (a.value.n < 0) throw new CommandError("NEGATIVE ROOT");
433 if (!a.dim.even()) throw new CommandError("ODD DIMENSION " + a.dim);
434 long n = integerRoot(a.value.n), d = integerRoot(a.value.d);
435 if (n*n != a.value.n || d*d != a.value.d) throw new CommandError("NOT EXACT");
436 yield new Quantity(new Fraction(n, d), a.dim.half());
437 }
438 Quantity b = evaluate(call.args.get(1));
439 a.dim.match(b.dim);
440 int comparison = a.value.compareTo(b.value);
441 yield call.name.equals("min") ? (comparison <= 0 ? a : b) : (comparison >= 0 ? a : b);
442 }
443 };

459 List<String> command(String line) {
460 String[] w = line.split(" ");
461 return switch (w[0]) {
462 case "DEFINE" -> {
463 checkName(w[1], false);
464 Fraction value = number(w[3]);
465 Quantity unit = unit(w[4]);
466 if (kindDimension(w[1]) != null || resolve(w[1]) != null) throw new CommandError("NAME TAKEN " + w[1]);
467 if (value.n <= 0) throw new CommandError("NOT POSITIVE " + w[3]);
468 Fraction factor = value.mul(unit.value);
469 units.add(new Unit(w[1], new Quantity(factor, unit.dim), false, new Definition(value, w[4])));
470 yield List.of("DEFINE " + w[1] + " factor=" + factor + " dim=" + unit.dim + " kind=" + kindName(unit.dim));
471 }
472 case "UNDEF" -> {
473 int index = -1;
474 for (int i = 0; i < units.size(); i++) if (units.get(i).symbol.equals(w[1])) { index = i; break; }
475 if (index < 0) throw new CommandError("NOT DEFINED " + w[1]);
476 if (index < 42) throw new CommandError("BUILTIN UNIT " + w[1]);
477 units.remove(index);
478 yield List.of("UNDEF " + w[1]);
479 }
480 case "DEFS" -> {
481 List<String> output = new ArrayList<>();
482 output.add("DEFS count=" + (units.size()-42));
483 for (Unit unit : units.subList(42, units.size())) {
484 output.add("DEF " + unit.symbol + " = " + unit.definition.value + " " + unit.definition.expression
485 + " factor=" + unit.quantity.value + " dim=" + unit.quantity.dim);
486 }
487 yield output;
488 }
489 case "CONVERT", "INVERT" -> {
490 Fraction value = number(w[1]);
491 Quantity from = unit(w[2]), to = unit(w[4]);
492 boolean inverse = w[0].equals("INVERT");
493 if (inverse) {
494 if (from.dim.equals(Dimension.NONE) || !from.dim.combine(to.dim, 1).equals(Dimension.NONE))
495 throw new CommandError("NOT RECIPROCAL " + from.dim + " vs " + to.dim);
496 if (value.n == 0) throw new CommandError("DIVISION BY ZERO");
497 } else from.dim.match(to.dim);
498 Fraction result = value.mul(from.value);
499 if (inverse) result = Fraction.ONE.div(result);
500 result = result.div(to.value);
501 yield List.of(w[0] + " " + value + " " + w[2] + " = " + result.notation(w[4]));
502 }
503 case "DIM", "REDUCE", "SIMPLIFY" -> {
504 ParsedUnit parsed = parseUnit(w[1]);
505 Quantity unit = parsed.quantity;
506 String result = switch (w[0]) {
507 case "DIM" -> unit.dim + " factor=" + unit.value + " kind=" + kindName(unit.dim);
508 case "REDUCE" -> reducedText(parsed.tree);
509 default -> notation(unit);
510 };
511 yield List.of(w[0] + " " + w[1] + " = " + result);
512 }
513 case "ADD", "SUB", "MUL", "DIV", "COMPARE" -> {
514 Fraction a = number(w[1]);
515 Quantity first = unit(w[2]);
516 Fraction b = number(w[3]);
517 Quantity second = unit(w[4]);
518 boolean target = w.length == 7;
519 Quantity destination = target ? unit(w[6]) : first;
520 if (w[0].equals("ADD") || w[0].equals("SUB") || w[0].equals("COMPARE")) {
521 first.dim.match(second.dim);
522 if (target) first.dim.match(destination.dim);
523 Fraction left = a.mul(first.value), right = b.mul(second.value);
524 if (w[0].equals("COMPARE")) {
525 String relation = new String[]{"<", "=", ">"}[left.compareTo(right)+1];
526 String ratio = right.n == 0 ? "undefined" : left.div(right).toString();
527 yield List.of("COMPARE " + a + " " + w[2] + " " + relation + " " + b + " " + w[4] + " ratio=" + ratio);
528 }
529 boolean add = w[0].equals("ADD");
530 Fraction result = (add ? left.add(right) : left.sub(right)).div(destination.value);
531 yield List.of(w[0] + " " + a + " " + w[2] + (add ? " + " : " - ") + b + " " + w[4]
532 + " = " + result.notation(target ? w[6] : w[2]));
533 }
534 boolean multiply = w[0].equals("MUL");
535 Dimension dim = first.dim.combine(second.dim, multiply ? 1 : -1);
536 if (!multiply && b.n == 0) throw new CommandError("DIVISION BY ZERO");
537 if (target) dim.match(destination.dim);
538 Fraction raw = multiply ? a.mul(b) : a.div(b);
539 Fraction left = a.mul(first.value), right = b.mul(second.value);
540 Fraction coherent = multiply ? left.mul(right) : left.div(right);
541 String op = multiply ? "*" : "/";
542 String prefix = w[0] + " " + a + " " + w[2] + " " + op + " " + b + " " + w[4] + " = ";
543 if (target) yield List.of(prefix + coherent.div(destination.value).notation(w[6]));
544 String rightUnit = !multiply && (w[4].contains("*") || w[4].contains("/")) ? "(" + w[4] + ")" : w[4];
545 yield List.of(prefix + raw + " " + w[2] + op + rightUnit + " = " + coherent.notation(simplified(dim)));
546 }
547 case "TEMP" -> {
548 Fraction value = number(w[1]);
549 for (String scale : List.of(w[2], w[4])) {
550 if (!List.of("C", "F", "K", "R").contains(scale)) throw new CommandError("UNKNOWN SCALE " + scale);
551 }
552 Fraction kelvin = switch (w[2]) {
553 case "C" -> value.add(new Fraction(5463, 20));
554 case "F" -> value.add(new Fraction(45967, 100)).mul(new Fraction(5)).div(new Fraction(9));
555 case "R" -> value.mul(new Fraction(5)).div(new Fraction(9));
556 default -> value;
557 };
558 if (kelvin.n < 0) throw new CommandError("BELOW ABSOLUTE ZERO");
559 Fraction result = switch (w[4]) {
560 case "C" -> kelvin.sub(new Fraction(5463, 20));
561 case "F" -> kelvin.mul(new Fraction(9)).div(new Fraction(5)).sub(new Fraction(45967, 100));
562 case "R" -> kelvin.mul(new Fraction(9)).div(new Fraction(5));
563 default -> kelvin;
564 };
565 yield List.of("TEMP " + value + " " + w[2] + " = " + result.notation(w[4]));
566 }
567 case "LIST" -> {
568 Dimension dim = kindDimension(w[1]);
569 if (dim == null) dim = unit(w[1]).dim;
570 List<Unit> selected = selected(dim);
571 List<String> output = new ArrayList<>();
572 output.add("LIST " + w[1] + " dim=" + dim + " kind=" + kindName(dim) + " count=" + selected.size());
573 for (Unit unit : selected) {
574 output.add("UNIT " + unit.symbol + " factor=" + unit.quantity.value + " origin="
575 + (unit.definition == null ? "builtin" : "user") + " prefix=" + (unit.prefix ? "yes" : "no"));
576 }
577 yield output;
578 }
579 case "TABLE" -> {
580 Fraction value = number(w[1]);
581 Quantity from = unit(w[2]);
582 List<Unit> selected = selected(from.dim);
583 List<String> output = new ArrayList<>();
584 output.add("TABLE " + value + " " + w[2] + " dim=" + from.dim + " count=" + selected.size());
585 for (Unit unit : selected) output.add("ROW " + value.mul(from.value).div(unit.quantity.value).notation(unit.symbol));
586 yield output;
587 }
588 case "AUTO" -> {
589 Fraction value = number(w[1]);
590 Quantity unit = unit(w[2]);
591 Unit reference = null;
592 for (Unit candidate : units.subList(0, 42)) {
593 if (candidate.prefix && candidate.quantity.dim.equals(unit.dim)) { reference = candidate; break; }
594 }
595 if (reference == null) throw new CommandError("NO PREFIXABLE UNIT " + unit.dim);
596 Fraction scaled = value.mul(unit.value).div(reference.quantity.value), magnitude = scaled.abs();
597 int k = 0;
598 if (magnitude.n != 0) {
599 if (magnitude.compareTo(Fraction.ONE) >= 0) {
600 while (k < 9 && magnitude.compareTo(new Fraction(10).power(k+3)) >= 0) k += 3;
601 } else {
602 while (k > -9 && magnitude.compareTo(new Fraction(10).power(k)) < 0) k -= 3;
603 }
604 }
605 String prefix = new String[]{"n", "u", "m", "", "k", "M", "G"}[(k+9)/3];
606 yield List.of("AUTO " + value + " " + w[2] + " = " + scaled.div(new Fraction(10).power(k)).notation(prefix+reference.symbol));
607 }
608 case "SPLIT" -> {
609 Fraction value = number(w[1]);
610 Quantity unit = unit(w[2]);
611 String[] parts = w[4].split(",");
612 List<Quantity> destinations = new ArrayList<>();
613 for (String part : parts) destinations.add(unit(part));
614 if (value.n < 0) throw new CommandError("NEGATIVE VALUE");
615 for (Quantity destination : destinations) unit.dim.match(destination.dim);
616 for (int i = 1; i < destinations.size(); i++) {
617 if (destinations.get(i-1).value.compareTo(destinations.get(i).value) <= 0) throw new CommandError("UNIT ORDER");
618 }
619 Fraction remaining = value.mul(unit.value);
620 List<String> pieces = new ArrayList<>();
621 for (int i = 0; i < parts.length-1; i++) {
622 Fraction quotient = remaining.div(destinations.get(i).value);
623 long count = quotient.n/quotient.d;
624 remaining = remaining.sub(new Fraction(count).mul(destinations.get(i).value));
625 pieces.add(count + " " + parts[i]);
626 }
627 pieces.add(remaining.div(destinations.getLast().value).notation(parts[parts.length-1]));
628 yield List.of("SPLIT " + value + " " + w[2] + " = " + String.join(" ", pieces));
629 }
630 case "LET" -> {
631 checkName(w[1], true);
632 Quantity quantity = quantity(w[3]);
633 variables.put(w[1], quantity);
634 yield List.of("LET " + w[1] + " = " + notation(quantity));
635 }
636 case "EVAL" -> {
637 Quantity quantity = quantity(w[1]);
638 String result;
639 if (w.length == 4) {
640 Quantity unit = unit(w[3]);
641 quantity.dim.match(unit.dim);
642 result = quantity.value.div(unit.value).notation(w[3]);
643 } else result = notation(quantity);
644 yield List.of("EVAL " + w[1] + " = " + result);
645 }
646 case "VARS" -> {
647 List<String> output = new ArrayList<>();
648 output.add("VARS count=" + variables.size());
649 for (Map.Entry<String, Quantity> entry : variables.entrySet()) output.add("VAR " + entry.getKey() + " = " + notation(entry.getValue()));
650 yield output;
651 }
652 default -> throw new CommandError("UNKNOWN COMMAND " + w[0]);
653 };
654 }
655 }

657 public static void main(String[] args) throws IOException {
658 BufferedReader input = new BufferedReader(new InputStreamReader(System.in, StandardCharsets.UTF_8));
659 PrintWriter output = new PrintWriter(new BufferedWriter(new OutputStreamWriter(System.out, StandardCharsets.UTF_8)));
660 int count = Integer.parseInt(input.readLine());
661 Database database = new Database();
662 for (int i = 0; i < count; i++) {
663 String line = input.readLine();
664 try { for (String result : database.command(line)) output.print(result + "\n"); }
665 catch (CommandError error) { output.print("ERROR " + error.getMessage() + "\n"); }
666 }
667 output.flush();
668 }
