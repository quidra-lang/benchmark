class Quantity(value := 1.0, dimension := [0, 0, 0, 0, 0, 0, 0])
class Unit(string symbol, Quantity quantity, bool prefixable, Definition | none definition := none)
Quantity | none resolve(const Database database, string name)
    for item := database.units[]
        if item.symbol == name
            return item.quantity
    prefixes := [
        Prefix("G", 1000000000.0), Prefix("M", 1000000.0), Prefix("k", 1000.0),
        Prefix("h", 100.0), Prefix("da", 10.0), Prefix("d", 1.0 / 10.0),
        Prefix("c", 1.0 / 100.0), Prefix("m", 1.0 / 1000.0),
        Prefix("u", 1.0 / 1000000.0), Prefix("n", 1.0 / 1000000000.0),
    ]
    for prefix := prefixes[]
        if name.starts_with(prefix.symbol)
            rest := name[#prefix.symbol:]
            for item := database.units[0:<42][]
                if item.prefixable and item.symbol == rest
                    return Quantity(prefix.factor * item.quantity.value, item.quantity.dimension)
