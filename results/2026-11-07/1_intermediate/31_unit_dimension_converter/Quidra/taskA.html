276Parser tokenize(string source, bool quantity)
277 Token[] tokens := []
278 index := 0
279 count := int(#source)
280 while index < count
281 start := index
282 character := source[index]
283 index += 1
284 TokenKind kind := TokenKind.Invalid
285 if letter(character)
286 kind = TokenKind.Identifier
287 while index < count and letter(source[index])
288 index += 1
289 elif digit(character)
290 kind = TokenKind.Number
291 while index < count and digit(source[index])
292 index += 1
293 if quantity and index + 1 < count and source[index] == "." and digit(source[index + 1])
294 index += 1
295 while index < count and digit(source[index])
296 index += 1
297 elif quantity and character == "["
298 closing := index
299 while closing < count and source[closing] != "]"
300 closing += 1
301 if index < closing < count
302 kind = TokenKind.Unit
303 index = closing + 1
304 elif (if quantity then "+-*/^()," else "*/^()-").contains(character)
305 kind = TokenKind.Punctuation
306 tokens = tokens.append(Token(kind, source[start:<index], start + 1))
307 return Parser(source, tokens.append(Token(TokenKind.End, "", count + 1)))

528Quantity | error evaluate_quantity(const Database database, const Node[] nodes, int index)
529 item := nodes[index]
530 if item.kind == Operation.Number
531 return Quantity(try parse_number(item.text))
532 if item.kind == Operation.UnitValue
533 unit := try unit_quantity(database, item.unit)
534 value := try parse_number(item.text)
535 return Quantity(value * unit.value, unit.dimension)
536 if item.kind == Operation.Variable
537 match database.variables.get(item.text)
538 Quantity stored
539 return stored
540 none
541 return error("UNKNOWN VARIABLE " + item.text)
542 if item.kind == Operation.Call
543 if not includes(["abs", "sqrt", "min", "max"], item.text)
544 return error("UNKNOWN FUNCTION " + item.text)
545 required := if item.text == "abs" or item.text == "sqrt" then 1 else 2
546 if int(#item.arguments) != required
547 return error("BAD ARGUMENTS " + item.text)
548 first := try evaluate_quantity(database, nodes, item.arguments[0])
549 if item.text == "abs"
550 return Quantity(math.abs(first.value), first.dimension)
551 if item.text == "sqrt"
552 if first.value < 0.0
553 return error("NEGATIVE ROOT")
554 for exponent := first.dimension[]
555 if exponent % 2 != 0
556 return error("ODD DIMENSION " + dimension_text(first.dimension))
557 parts := ratio_parts(first.value)
558 numerator := integer_sqrt(parts.numerator)
559 denominator := integer_sqrt(parts.denominator)
560 if numerator * numerator != parts.numerator or denominator * denominator != parts.denominator
561 return error("NOT EXACT")
562 return Quantity(real(numerator) / real(denominator), first.dimension / 2)
563 second := try evaluate_quantity(database, nodes, item.arguments[1])
564 try match_dimensions(first.dimension, second.dimension)
565 take_first := if item.text == "min" then first.value <= second.value else first.value >= second.value
566 return if take_first then first else second
567 left := try evaluate_quantity(database, nodes, item.left)
568 if item.kind == Operation.Negative
569 return Quantity(-left.value, left.dimension)
570 if item.kind == Operation.Power
571 exponent := try checked_exponent(item.text)
572 if left.value == 0.0 and exponent < 0
573 return error("DIVISION BY ZERO")
574 return Quantity(scalar_power(left.value, exponent), left.dimension * exponent)
575 right := try evaluate_quantity(database, nodes, item.right)
576 if item.kind == Operation.Add or item.kind == Operation.Subtract
577 try match_dimensions(left.dimension, right.dimension)
578 value := if item.kind == Operation.Add then left.value + right.value else left.value - right.value
579 return Quantity(value, left.dimension)
580 if item.kind == Operation.Multiply
581 return Quantity(left.value * right.value, left.dimension + right.dimension)
582 if right.value == 0.0
583 return error("DIVISION BY ZERO")
584 return Quantity(left.value / right.value, left.dimension - right.dimension)
