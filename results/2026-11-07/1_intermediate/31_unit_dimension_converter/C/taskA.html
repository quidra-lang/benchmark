19static void *allocate(size_t size) {
20 void *value = calloc(1, size);
21 Allocation *item = malloc(sizeof *item);
22 if (!value || !item) exit(EXIT_FAILURE);
23 *item = (Allocation){value, temporary}; temporary = item;
24 return value;
25}
26static void release_temporary(void) {
27 while (temporary) {
28 Allocation *next = temporary->next;
29 free(temporary->value); free(temporary); temporary = next;
30 }
31}
32static char *slice(const char *text, size_t start, size_t end) {
33 char *result = allocate(end - start + 1);
34 memcpy(result, text + start, end - start); return result;

36static char *format(const char *pattern, ...) {
37 va_list args, copy;
38 va_start(args, pattern); va_copy(copy, args);
39 int length = vsnprintf(NULL, 0, pattern, copy); va_end(copy);
40 char *result = allocate((size_t)length + 1);
41 vsnprintf(result, (size_t)length + 1, pattern, args); va_end(args); return result;
42}
43static void fail(const char *pattern, ...) {
44 va_list args; va_start(args, pattern);
45 fputs("ERROR ", stdout); vprintf(pattern, args); putchar('\n'); va_end(args);
46 longjmp(failure, 1);
47}
48static bool equal(const char *a, const char *b) { return strcmp(a, b) == 0; }
49static bool letter(char c) { return (c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z'); }
50static bool digit(char c) { return c >= '0' && c <= '9'; }
51static int64_t gcd(int64_t a, int64_t b) {
52 if (a < 0) a = -a;
53 while (b) { int64_t remainder = a % b; a = b; b = remainder; }
54 return a;
55}
56static Rational rational(int64_t n, int64_t d) {
57 if (d < 0) { n = -n; d = -d; }
58 int64_t common = gcd(n, d); return (Rational){n / common, d / common};
59}
60static Rational integer(int64_t n) { return (Rational){n, 1}; }
61static Rational add(Rational a, Rational b) { return rational(a.n * b.d + b.n * a.d, a.d * b.d); }
62static Rational subtract(Rational a, Rational b) { return rational(a.n * b.d - b.n * a.d, a.d * b.d); }
63static Rational multiply(Rational a, Rational b) { return rational(a.n * b.n, a.d * b.d); }
64static Rational divide(Rational a, Rational b) { return rational(a.n * b.d, a.d * b.n); }
65static Rational negate(Rational a) { a.n = -a.n; return a; }
66static Rational power(Rational a, int exponent) {
67 if (exponent < 0) { a = rational(a.d, a.n); exponent = -exponent; }
68 Rational result = ONE;
69 for (int i = 0; i < exponent; i++) result = multiply(result, a);
70 return result;
71}
72static int compare(Rational a, Rational b) {
73 int64_t left = a.n * b.d, right = b.n * a.d;
74 return (left > right) - (left < right);
75}
76static char *fraction(Rational value) {
77 return value.d == 1 ? format("%" PRId64, value.n) : format("%" PRId64 "/%" PRId64, value.n, value.d);
78}
79static char *decimal(Rational value) {
80 int64_t magnitude = value.n < 0 ? -value.n : value.n;
81 return format("%s%" PRId64 ".%06" PRId64, value.n < 0 ? "-" : "", magnitude / value.d, (magnitude % value.d) * 1000000 / value.d);
82}
83static char *notation(Rational value, const char *unit) { return format("%s %s (%s)", fraction(value), unit, decimal(value)); }
84static Dimension dimension_add(Dimension a, Dimension b, int sign) {
85 for (int i = 0; i < 7; i++) a.power[i] += sign * b.power[i];
86 return a;

88static Dimension dimension_scale(Dimension dim, int64_t factor) {
89 for (int i = 0; i < 7; i++) dim.power[i] *= factor;
90 return dim;

92static bool same_dimension(Dimension a, Dimension b) {
93 for (int i = 0; i < 7; i++) if (a.power[i] != b.power[i]) return false;
94 return true;

96static bool dimensionless(Dimension dim) { return same_dimension(dim, (Dimension){{0}}); }
97static char *dimension_text(Dimension dim) {
98 char *result = "";
99 for (int i = 0; i < 7; i++) if (dim.power[i]) {
100 const char *term = dim.power[i] == 1 ? base_names[i] : format("%s^%" PRId64, base_names[i], dim.power[i]);
101 result = format("%s%s%s", result, *result ? " " : "", term);
102 }
103 return *result ? result : "1";
104}
105static void match_dimensions(Dimension a, Dimension b) {
106 if (!same_dimension(a, b)) fail("DIMENSION MISMATCH %s vs %s", dimension_text(a), dimension_text(b));
107}
108static Rational number(const char *text) {
109 size_t i = text[0] == '-' ? 1 : 0;
110 bool negative = i != 0;
111 if (!digit(text[i])) fail("BAD NUMBER %s", text);
112 int64_t whole = 0;
113 while (digit(text[i])) whole = whole * 10 + text[i++] - '0';
114 int64_t denominator = 1, numerator = whole;
115 if (text[i] == '.' || text[i] == '/') {
116 char operation = text[i++];
117 if (!digit(text[i])) fail("BAD NUMBER %s", text);
118 int64_t tail = 0, scale = 1;
119 while (digit(text[i])) { tail = tail * 10 + text[i++] - '0'; scale *= 10; }
120 if (operation == '.') { numerator = whole * scale + tail; denominator = scale; }
121 else { denominator = tail; if (!denominator) fail("BAD NUMBER %s", text); }
122 }
123 if (text[i]) fail("BAD NUMBER %s", text);
124 return rational(negative ? -numerator : numerator, denominator);
125}
126static int exponent_value(const char *text) {
127 int64_t value = strtoll(text, NULL, 10);
128 if (!value || value < -3 || value > 3) fail("BAD EXPONENT %s", text);
129 return (int)value;
130}
131static void check_name(const char *name, bool variable) {
132 size_t length = strlen(name);
133 bool valid = length > 0 && length <= 12;
134 for (size_t i = 0; i < length; i++) if (!letter(name[i])) valid = false;
135 if (variable && (equal(name, "abs") || equal(name, "sqrt") || equal(name, "min") || equal(name, "max"))) valid = false;
136 if (!valid) fail("BAD NAME %s", name);
137}

186static bool resolve(const char *name, Quantity *result) {
187 for (size_t i = 0; i < unit_count; i++) if (equal(name, units[i].symbol)) { *result = units[i].quantity; return true; }
188 static const char *prefixes[] = {"G", "M", "k", "h", "da", "d", "c", "m", "u", "n"};
189 static const Rational factors[] = {{1000000000,1}, {1000000,1}, {1000,1}, {100,1}, {10,1}, {1,10}, {1,100}, {1,1000}, {1,1000000}, {1,1000000000}};
190 for (size_t p = 0; p < 10; p++) {
191 size_t length = strlen(prefixes[p]);
192 if (strncmp(name, prefixes[p], length) != 0) continue;
193 for (size_t i = 0; i < 42; i++) if (units[i].prefix && equal(name + length, units[i].symbol)) {
194 *result = units[i].quantity; result->value = multiply(factors[p], result->value); return true;
195 }
196 }
197 return false;
198}

211enum { TOKEN_END = 256, TOKEN_ID, TOKEN_NUMBER, TOKEN_UNIT, TOKEN_INVALID };
212typedef struct { int kind; size_t start, end; char *text; } Token;
213typedef struct { const char *source; Token tokens[201]; size_t count, position; } Parser;
214typedef enum { LITERAL, UNIT_NAME, UNIT_VALUE, VARIABLE, BINARY, NEGATIVE, POWER, CALL } NodeKind;
215typedef struct Node {
216 NodeKind kind; char operation; char *text, *unit, *exponent;
217 struct Node *left, *right, *args, *next;
218 size_t argument_count; int exponent_number; Quantity resolved;
219} Node;
220static Node *node(NodeKind kind) { Node *result = allocate(sizeof *result); result->kind = kind; return result; }
221static Node *binary_node(char operation, Node *left, Node *right) {
222 Node *result = node(BINARY); result->operation = operation; result->left = left; result->right = right; return result;
223}
224static void tokenize(Parser *parser, const char *text, bool quantity) {
225 *parser = (Parser){.source = text};
226 size_t length = strlen(text), i = 0;
227 while (i < length) {
228 size_t start = i;
229 int kind;
230 if (letter(text[i])) { kind = TOKEN_ID; while (letter(text[i])) i++; }
231 else if (digit(text[i])) {
232 kind = TOKEN_NUMBER; while (digit(text[i])) i++;
233 if (quantity && text[i] == '.' && digit(text[i + 1])) { i++; while (digit(text[i])) i++; }
234 } else if (quantity && text[i] == '[' && text[i + 1] && text[i + 1] != ']' && strchr(text + i + 1, ']')) {
235 kind = TOKEN_UNIT; i = (size_t)(strchr(text + i + 1, ']') - text) + 1;
236 } else {
237 kind = strchr(quantity ? "+-*/^()," : "*/^()-", text[i]) ? (unsigned char)text[i] : TOKEN_INVALID; i++;
238 }
239 parser->tokens[parser->count++] = (Token){kind, start, i, slice(text, start, i)};
240 }
241 parser->tokens[parser->count++] = (Token){TOKEN_END, length, length, ""};
242}
243static Token *current(Parser *parser) { return &parser->tokens[parser->position]; }
244static void syntax(Parser *parser) {
245 Token *token = current(parser);
246 fail("SYNTAX %s AT %s", parser->source, token->kind == TOKEN_END ? "END" : format("%zu", token->start + 1));
247}
248static bool accept(Parser *parser, int kind) {
249 if (current(parser)->kind != kind) return false;
250 parser->position++; return true;
251}
252static void expect(Parser *parser, int kind) { if (!accept(parser, kind)) syntax(parser); }
253static char *parse_exponent(Parser *parser) {
254 size_t start = current(parser)->start;
255 accept(parser, '-');
256 Token *token = current(parser);
257 if (token->kind != TOKEN_NUMBER || strchr(token->text, '.')) syntax(parser);
258 parser->position++; return slice(parser->source, start, token->end);
259}
260static Node *unit_expression(Parser *parser);
261static Node *unit_term(Parser *parser) {
262 Token *token = current(parser);
263 Node *result;
264 if (accept(parser, TOKEN_ID)) { result = node(UNIT_NAME); result->text = token->text; }
265 else if (token->kind == TOKEN_NUMBER && token->text[0] != '0') {
266 parser->position++; result = node(LITERAL); result->text = token->text;
267 } else if (accept(parser, '(')) { result = unit_expression(parser); expect(parser, ')'); }
268 else { syntax(parser); return NULL; }
269 if (accept(parser, '^')) {
270 Node *power_node = node(POWER); power_node->left = result; power_node->exponent = parse_exponent(parser); result = power_node;
271 }
272 return result;
273}
274static Node *unit_expression(Parser *parser) {
275 Node *one = node(LITERAL); one->text = "1";
276 Node *result = binary_node('*', one, unit_term(parser));
277 while (current(parser)->kind == '*' || current(parser)->kind == '/') {
278 char operation = (char)current(parser)->kind; parser->position++;
279 Node *right = unit_term(parser); result = binary_node(operation, result, right);
280 }
281 return result;
282}
283static void validate_units(Node *expression) {
284 if (expression->kind == UNIT_NAME) {
285 if (!resolve(expression->text, &expression->resolved)) fail("UNKNOWN UNIT %s", expression->text);
286 } else if (expression->kind == POWER) {
287 validate_units(expression->left); expression->exponent_number = exponent_value(expression->exponent);
288 } else if (expression->kind == BINARY) {
289 validate_units(expression->left); validate_units(expression->right);
290 }
291}
292static Quantity evaluate_unit(Node *expression) {
293 if (expression->kind == LITERAL) return (Quantity){number(expression->text), {{0}}};
294 if (expression->kind == UNIT_NAME) return expression->resolved;
295 Quantity left = evaluate_unit(expression->left);
296 if (expression->kind == POWER) return (Quantity){power(left.value, expression->exponent_number), dimension_scale(left.dim, expression->exponent_number)};
297 Quantity right = evaluate_unit(expression->right);
298 return (Quantity){expression->operation == '*' ? multiply(left.value, right.value) : divide(left.value, right.value), dimension_add(left.dim, right.dim, expression->operation == '*' ? 1 : -1)};
299}
300static Node *parse_unit(const char *text) {
301 Parser parser; tokenize(&parser, text, false);
302 Node *expression = unit_expression(&parser);
303 if (current(&parser)->kind != TOKEN_END) syntax(&parser);
304 validate_units(expression); return expression;
305}
306static Quantity unit_value(const char *text) { return evaluate_unit(parse_unit(text)); }

308typedef struct { Rational coefficient; int64_t exponents[201]; } Reduced;
309typedef struct { const char *names[201]; size_t count; } NameOrder;
310static Reduced reduce_node(Node *expression, NameOrder *order) {
311 Reduced result = {.coefficient = ONE};
312 if (expression->kind == LITERAL) { result.coefficient = number(expression->text); return result; }
313 if (expression->kind == UNIT_NAME) {
314 size_t index = 0;
315 while (index < order->count && !equal(order->names[index], expression->text)) index++;
316 if (index == order->count) order->names[order->count++] = expression->text;
317 result.exponents[index] = 1; return result;
318 }
319 result = reduce_node(expression->left, order);
320 if (expression->kind == POWER) {
321 result.coefficient = power(result.coefficient, expression->exponent_number);
322 for (size_t i = 0; i < order->count; i++) result.exponents[i] *= expression->exponent_number;
323 return result;
324 }
325 Reduced right = reduce_node(expression->right, order);
326 int sign = expression->operation == '*' ? 1 : -1;
327 result.coefficient = sign == 1 ? multiply(result.coefficient, right.coefficient) : divide(result.coefficient, right.coefficient);
328 for (size_t i = 0; i < order->count; i++) result.exponents[i] += sign * right.exponents[i];
329 return result;
330}
331static char *reduced_text(Node *expression) {
332 NameOrder order = {0}; Reduced reduced = reduce_node(expression, &order);
333 char *result = reduced.coefficient.n == 1 && reduced.coefficient.d == 1 ? "" : fraction(reduced.coefficient);
334 for (size_t i = 0; i < order.count; i++) if (reduced.exponents[i]) {
335 const char *term = reduced.exponents[i] == 1 ? order.names[i] : format("%s^%" PRId64, order.names[i], reduced.exponents[i]);
336 result = format("%s%s%s", result, *result ? " " : "", term);
337 }
338 return *result ? result : "1";
339}

342static Node *quantity_primary(Parser *parser) {
343 Token *token = current(parser);
344 Node *result;
345 if (accept(parser, TOKEN_NUMBER)) {
346 result = node(LITERAL); result->text = token->text;
347 if (current(parser)->kind == TOKEN_UNIT) {
348 token = current(parser); parser->position++;
349 result->kind = UNIT_VALUE; result->unit = slice(token->text, 1, strlen(token->text) - 1);
350 }
351 } else if (accept(parser, TOKEN_UNIT)) {
352 result = node(UNIT_VALUE); result->text = "1"; result->unit = slice(token->text, 1, strlen(token->text) - 1);
353 } else if (accept(parser, TOKEN_ID)) {
354 result = node(VARIABLE); result->text = token->text;
355 if (accept(parser, '(')) {
356 result->kind = CALL;
357 Node **next = &result->args;
358 do {
359 *next = quantity_expression(parser); next = &(*next)->next; result->argument_count++;
360 } while (accept(parser, ','));
361 expect(parser, ')');
362 }
363 } else if (accept(parser, '(')) { result = quantity_expression(parser); expect(parser, ')'); }
364 else { syntax(parser); return NULL; }
365 if (accept(parser, '^')) {
366 Node *power_node = node(POWER); power_node->left = result; power_node->exponent = parse_exponent(parser); result = power_node;
367 }
368 return result;
369}
370static Node *quantity_unary(Parser *parser) {
371 if (!accept(parser, '-')) return quantity_primary(parser);
372 Node *result = node(NEGATIVE); result->left = quantity_unary(parser); return result;
373}

377 char operation = (char)current(parser)->kind; parser->position++;
378 Node *right = quantity_unary(parser); result = binary_node(operation, result, right);

383 Node *result = quantity_product(parser);

385 char operation = (char)current(parser)->kind; parser->position++;
386 Node *right = quantity_product(parser); result = binary_node(operation, result, right);

390static int64_t integer_sqrt(int64_t n) {
391 if (n < 2) return n;
392 int64_t x = n, next = (n + 1) / 2;
393 while (next < x) { x = next; next = (x + n / x) / 2; }
394 return x;
395}
396static Quantity evaluate_quantity(Node *expression) {
397 if (expression->kind == LITERAL) return (Quantity){number(expression->text), {{0}}};
398 if (expression->kind == UNIT_VALUE) {
399 Quantity result = unit_value(expression->unit);
400 result.value = multiply(number(expression->text), result.value); return result;
401 }
402 if (expression->kind == VARIABLE) {
403 Variable *variable = find_variable(expression->text);
404 if (!variable) fail("UNKNOWN VARIABLE %s", expression->text);
405 return variable->quantity;
406 }
407 if (expression->kind == CALL) {
408 bool absolute = equal(expression->text, "abs"), root = equal(expression->text, "sqrt");
409 bool minimum = equal(expression->text, "min"), maximum = equal(expression->text, "max");
410 if (!absolute && !root && !minimum && !maximum) fail("UNKNOWN FUNCTION %s", expression->text);
411 if (expression->argument_count != (absolute || root ? 1u : 2u)) fail("BAD ARGUMENTS %s", expression->text);
412 Quantity left = evaluate_quantity(expression->args);
413 if (absolute) { if (left.value.n < 0) left.value = negate(left.value); return left; }
414 if (root) {
415 if (left.value.n < 0) fail("NEGATIVE ROOT");
416 for (int i = 0; i < 7; i++) if (left.dim.power[i] % 2) fail("ODD DIMENSION %s", dimension_text(left.dim));
417 int64_t n = integer_sqrt(left.value.n), d = integer_sqrt(left.value.d);
418 if (n * n != left.value.n || d * d != left.value.d) fail("NOT EXACT");
419 left.value = rational(n, d);
420 for (int i = 0; i < 7; i++) left.dim.power[i] /= 2;
421 return left;
422 }
423 Quantity right = evaluate_quantity(expression->args->next);
424 match_dimensions(left.dim, right.dim);
425 int order = compare(left.value, right.value);
426 return (minimum ? order <= 0 : order >= 0) ? left : right;
427 }
428 Quantity left = evaluate_quantity(expression->left);
429 if (expression->kind == NEGATIVE) { left.value = negate(left.value); return left; }
430 if (expression->kind == POWER) {
431 int exponent = exponent_value(expression->exponent);
432 if (!left.value.n && exponent < 0) fail("DIVISION BY ZERO");
433 return (Quantity){power(left.value, exponent), dimension_scale(left.dim, exponent)};
434 }
435 Quantity right = evaluate_quantity(expression->right);
436 if (expression->operation == '+' || expression->operation == '-') {
437 match_dimensions(left.dim, right.dim);
438 left.value = expression->operation == '+' ? add(left.value, right.value) : subtract(left.value, right.value);
439 return left;
440 }
441 if (expression->operation == '/' && !right.value.n) fail("DIVISION BY ZERO");
442 return (Quantity){expression->operation == '*' ? multiply(left.value, right.value) : divide(left.value, right.value), dimension_add(left.dim, right.dim, expression->operation == '*' ? 1 : -1)};
443}
444static Quantity quantity_value(const char *text) {
445 Parser parser; tokenize(&parser, text, true);
446 Node *expression = quantity_expression(&parser);
447 if (current(&parser)->kind != TOKEN_END) syntax(&parser);
448 return evaluate_quantity(expression);
449}

451static int unit_order(const void *left, const void *right) {
452 const Unit *a = *(Unit *const *)left, *b = *(Unit *const *)right;
453 int order = compare(a->quantity.value, b->quantity.value);
454 return order ? order : strcmp(a->symbol, b->symbol);
455}
456static size_t select_units(Dimension dim, Unit **selected) {
457 size_t count = 0;
458 for (size_t i = 0; i < unit_count; i++) if (same_dimension(dim, units[i].quantity.dim)) selected[count++] = &units[i];
459 qsort(selected, count, sizeof *selected, unit_order); return count;
460}
461static int variable_order(const void *left, const void *right) {
462 const Variable *a = *(Variable *const *)left, *b = *(Variable *const *)right;
463 return strcmp(a->name, b->name);
464}
465static void command(char **words, size_t count) {
466 const char *name = words[0];
467 if (equal(name, "DEFINE")) {
468 check_name(words[1], false);
469 Rational value = number(words[3]); Quantity unit = unit_value(words[4]), existing;
470 if (find_kind(words[1]) || resolve(words[1], &existing)) fail("NAME TAKEN %s", words[1]);
471 if (value.n <= 0) fail("NOT POSITIVE %s", words[3]);
472 Rational factor = multiply(value, unit.value);
473 Unit *created = &units[unit_count++]; *created = (Unit){0};
474 strcpy(created->symbol, words[1]); created->quantity = (Quantity){factor, unit.dim};
475 created->definition_value = value; strcpy(created->definition_unit, words[4]);
476 printf("DEFINE %s factor=%s dim=%s kind=%s\n", words[1], fraction(factor), dimension_text(unit.dim), kind_name(unit.dim));
477 } else if (equal(name, "UNDEF")) {
478 size_t index = 0;
479 while (index < unit_count && !equal(units[index].symbol, words[1])) index++;
480 if (index < 42) fail("BUILTIN UNIT %s", words[1]);
481 if (index == unit_count) fail("NOT DEFINED %s", words[1]);
482 memmove(&units[index], &units[index + 1], (unit_count - index - 1) * sizeof *units);
483 unit_count--; printf("UNDEF %s\n", words[1]);
484 } else if (equal(name, "DEFS")) {
485 printf("DEFS count=%zu\n", unit_count - 42);
486 for (size_t i = 42; i < unit_count; i++) {
487 Unit *unit = &units[i];
488 printf("DEF %s = %s %s factor=%s dim=%s\n", unit->symbol, fraction(unit->definition_value), unit->definition_unit, fraction(unit->quantity.value), dimension_text(unit->quantity.dim));
489 }
490 } else if (equal(name, "CONVERT") || equal(name, "INVERT")) {
491 Rational value = number(words[1]); Quantity from = unit_value(words[2]); Quantity to = unit_value(words[4]);
492 bool inverse = equal(name, "INVERT");
493 if (inverse) {
494 if (dimensionless(from.dim) || !dimensionless(dimension_add(from.dim, to.dim, 1))) fail("NOT RECIPROCAL %s vs %s", dimension_text(from.dim), dimension_text(to.dim));
495 if (!value.n) fail("DIVISION BY ZERO");
496 } else match_dimensions(from.dim, to.dim);
497 Rational result = multiply(value, from.value);
498 if (inverse) result = divide(ONE, result);
499 result = divide(result, to.value);
500 printf("%s %s %s = %s\n", name, fraction(value), words[2], notation(result, words[4]));
501 } else if (equal(name, "DIM") || equal(name, "REDUCE") || equal(name, "SIMPLIFY")) {
502 Node *expression = parse_unit(words[1]); Quantity unit = evaluate_unit(expression);
503 if (equal(name, "DIM")) printf("DIM %s = %s factor=%s kind=%s\n", words[1], dimension_text(unit.dim), fraction(unit.value), kind_name(unit.dim));
504 else if (equal(name, "REDUCE")) printf("REDUCE %s = %s\n", words[1], reduced_text(expression));
505 else printf("SIMPLIFY %s = %s\n", words[1], notation(unit.value, simplified(unit.dim)));
506 } else if (equal(name, "ADD") || equal(name, "SUB") || equal(name, "MUL") || equal(name, "DIV") || equal(name, "COMPARE")) {
507 Rational a = number(words[1]); Quantity first = unit_value(words[2]);
508 Rational b = number(words[3]); Quantity second = unit_value(words[4]);
509 bool target = count == 7;
510 Quantity destination = target ? unit_value(words[6]) : first;
511 if (equal(name, "ADD") || equal(name, "SUB") || equal(name, "COMPARE")) {
512 match_dimensions(first.dim, second.dim);
513 if (target) match_dimensions(first.dim, destination.dim);
514 Rational left = multiply(a, first.value); Rational right = multiply(b, second.value);
515 if (equal(name, "COMPARE")) {
516 int relation = compare(left, right);
517 const char *ratio = right.n ? fraction(divide(left, right)) : "undefined";
518 printf("COMPARE %s %s %c %s %s ratio=%s\n", fraction(a), words[2], relation < 0 ? '<' : relation > 0 ? '>' : '=', fraction(b), words[4], ratio);
519 } else {
520 bool addition = equal(name, "ADD");
521 Rational sum = addition ? add(left, right) : subtract(left, right);
522 Rational result = divide(sum, destination.value);
523 printf("%s %s %s %c %s %s = %s\n", name, fraction(a), words[2], addition ? '+' : '-', fraction(b), words[4], notation(result, target ? words[6] : words[2]));
524 }
525 } else {
526 bool product = equal(name, "MUL");
527 Dimension dim = dimension_add(first.dim, second.dim, product ? 1 : -1);
528 if (!product && !b.n) fail("DIVISION BY ZERO");
529 if (target) match_dimensions(dim, destination.dim);
530 Rational raw = product ? multiply(a, b) : divide(a, b);
531 Rational left = multiply(a, first.value); Rational right = multiply(b, second.value);
532 Rational coherent = product ? multiply(left, right) : divide(left, right);
533 printf("%s %s %s %c %s %s = ", name, fraction(a), words[2], product ? '*' : '/', fraction(b), words[4]);
534 if (target) puts(notation(divide(coherent, destination.value), words[6]));
535 else {
536 const char *right_unit = !product && (strchr(words[4], '*') || strchr(words[4], '/')) ? format("(%s)", words[4]) : words[4];
537 printf("%s %s%c%s = %s\n", fraction(raw), words[2], product ? '*' : '/', right_unit, notation(coherent, simplified(dim)));
538 }
539 }
540 } else if (equal(name, "TEMP")) {
541 Rational value = number(words[1]);
542 if (strlen(words[2]) != 1 || !strchr("CFKR", words[2][0])) fail("UNKNOWN SCALE %s", words[2]);
543 if (strlen(words[4]) != 1 || !strchr("CFKR", words[4][0])) fail("UNKNOWN SCALE %s", words[4]);
544 Rational kelvin = value;
545 if (equal(words[2], "C")) kelvin = add(value, rational(5463,20));
546 else if (equal(words[2], "F")) { kelvin = add(value, rational(45967,100)); kelvin = multiply(kelvin, rational(5,9)); }
547 else if (equal(words[2], "R")) kelvin = multiply(value, rational(5,9));
548 if (kelvin.n < 0) fail("BELOW ABSOLUTE ZERO");
549 Rational result = kelvin;
550 if (equal(words[4], "C")) result = subtract(kelvin, rational(5463,20));
551 else if (equal(words[4], "F")) { result = multiply(kelvin, rational(9,5)); result = subtract(result, rational(45967,100)); }
552 else if (equal(words[4], "R")) result = multiply(kelvin, rational(9,5));
553 printf("TEMP %s %s = %s\n", fraction(value), words[2], notation(result, words[4]));
554 } else if (equal(name, "LIST") || equal(name, "TABLE")) {
555 bool table = equal(name, "TABLE");
556 Rational value = ONE; Quantity unit = {ONE, {{0}}}; Dimension dim;
557 if (table) { value = number(words[1]); unit = unit_value(words[2]); dim = unit.dim; }
558 else {
559 const Kind *kind = find_kind(words[1]);
560 dim = kind ? kind->dim : unit_value(words[1]).dim;
561 }
562 Unit *selected[72]; size_t length = select_units(dim, selected);
563 if (table) printf("TABLE %s %s dim=%s count=%zu\n", fraction(value), words[2], dimension_text(dim), length);
564 else printf("LIST %s dim=%s kind=%s count=%zu\n", words[1], dimension_text(dim), kind_name(dim), length);
565 for (size_t i = 0; i < length; i++) {
566 Unit *item = selected[i];
567 if (table) {
568 Rational coherent = multiply(value, unit.value); Rational result = divide(coherent, item->quantity.value);
569 printf("ROW %s\n", notation(result, item->symbol));
570 } else printf("UNIT %s factor=%s origin=%s prefix=%s\n", item->symbol, fraction(item->quantity.value), item->builtin ? "builtin" : "user", item->prefix ? "yes" : "no");
571 }
572 } else if (equal(name, "AUTO")) {
573 Rational value = number(words[1]); Quantity unit = unit_value(words[2]);
574 size_t index = 0;
575 while (index < 42 && (!units[index].prefix || !same_dimension(unit.dim, units[index].quantity.dim))) index++;
576 if (index == 42) fail("NO PREFIXABLE UNIT %s", dimension_text(unit.dim));
577 Rational coherent = multiply(value, unit.value); Rational scaled = divide(coherent, units[index].quantity.value);
578 Rational magnitude = scaled.n < 0 ? negate(scaled) : scaled;
579 int exponent = 0;
580 if (magnitude.n) {
581 if (compare(magnitude, ONE) >= 0) {
582 while (exponent < 9 && compare(magnitude, power(integer(10), exponent + 3)) >= 0) exponent += 3;
583 } else {
584 while (exponent > -9 && compare(magnitude, power(integer(10), exponent)) < 0) exponent -= 3;
585 }
586 }
587 static const char *prefixes[] = {"n", "u", "m", "", "k", "M", "G"};
588 const char *symbol = format("%s%s", prefixes[(exponent + 9) / 3], units[index].symbol);
589 Rational result = divide(scaled, power(integer(10), exponent));
590 printf("AUTO %s %s = %s\n", fraction(value), words[2], notation(result, symbol));
591 } else if (equal(name, "SPLIT")) {
592 Rational value = number(words[1]); Quantity unit = unit_value(words[2]);
593 char *list = slice(words[4], 0, strlen(words[4])), *parts[5];
594 size_t length = 1; parts[0] = list;
595 for (char *p = list; *p; p++) if (*p == ',') { *p = '\0'; parts[length++] = p + 1; }
596 Quantity destinations[5];
597 for (size_t i = 0; i < length; i++) destinations[i] = unit_value(parts[i]);
598 if (value.n < 0) fail("NEGATIVE VALUE");
599 for (size_t i = 0; i < length; i++) match_dimensions(unit.dim, destinations[i].dim);
600 for (size_t i = 0; i + 1 < length; i++) if (compare(destinations[i].value, destinations[i + 1].value) <= 0) fail("UNIT ORDER");
601 Rational remaining = multiply(value, unit.value);
602 printf("SPLIT %s %s = ", fraction(value), words[2]);
603 for (size_t i = 0; i + 1 < length; i++) {
604 Rational quotient = divide(remaining, destinations[i].value); int64_t count_unit = quotient.n / quotient.d;
605 remaining = subtract(remaining, multiply(integer(count_unit), destinations[i].value));
606 printf("%" PRId64 " %s ", count_unit, parts[i]);
607 }
608 puts(notation(divide(remaining, destinations[length - 1].value), parts[length - 1]));
609 } else if (equal(name, "LET")) {
610 check_name(words[1], true); Quantity quantity = quantity_value(words[3]);
611 Variable *variable = find_variable(words[1]);
612 if (!variable) { variable = &variables[variable_count++]; strcpy(variable->name, words[1]); }
613 variable->quantity = quantity;
614 printf("LET %s = %s\n", words[1], notation(quantity.value, simplified(quantity.dim)));
615 } else if (equal(name, "EVAL")) {
616 Quantity quantity = quantity_value(words[1]);
617 if (count == 4) {
618 Quantity unit = unit_value(words[3]); match_dimensions(quantity.dim, unit.dim);
619 printf("EVAL %s = %s\n", words[1], notation(divide(quantity.value, unit.value), words[3]));
620 } else printf("EVAL %s = %s\n", words[1], notation(quantity.value, simplified(quantity.dim)));
621 } else if (equal(name, "VARS")) {
622 Variable *ordered[50];
623 for (size_t i = 0; i < variable_count; i++) ordered[i] = &variables[i];
624 qsort(ordered, variable_count, sizeof *ordered, variable_order);
625 printf("VARS count=%zu\n", variable_count);
626 for (size_t i = 0; i < variable_count; i++) printf("VAR %s = %s\n", ordered[i]->name, notation(ordered[i]->quantity.value, simplified(ordered[i]->quantity.dim)));
627 } else fail("UNKNOWN COMMAND %s", name);
628}

630int main(void) {
631 initialize_units();
632 char line[256];
633 if (!fgets(line, sizeof line, stdin)) return 0;
634 int count = atoi(line);
635 for (int i = 0; i < count; i++) {
636 if (!fgets(line, sizeof line, stdin)) break;
637 line[strcspn(line, "\r\n")] = '\0';
638 char *words[201]; size_t word_count = 0;
639 for (char *word = strtok(line, " "); word; word = strtok(NULL, " ")) words[word_count++] = word;
640 if (!setjmp(failure)) command(words, word_count);
641 release_temporary();
642 }
643 return 0;
