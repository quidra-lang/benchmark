12class Fraction {
13 constructor(n, d = 1n) {
14 if (d < 0n) { n = -n; d = -d; }
15 const divisor = gcd(n, d);
16 this.n = n / divisor;
17 this.d = d / divisor;
18 }
19 add(b) { return new Fraction(this.n * b.d + b.n * this.d, this.d * b.d); }
20 sub(b) { return new Fraction(this.n * b.d - b.n * this.d, this.d * b.d); }
21 mul(b) { return new Fraction(this.n * b.n, this.d * b.d); }
22 div(b) { return new Fraction(this.n * b.d, this.d * b.n); }
23 neg() { return new Fraction(-this.n, this.d); }
24 abs() { return this.n < 0n ? this.neg() : this; }
25 compare(b) {
26 const left = this.n * b.d, right = b.n * this.d;
27 return left < right ? -1 : left > right ? 1 : 0;
28 }
29 pow(exponent) {
30 const factor = exponent < 0n ? ONE.div(this) : this;
31 let result = ONE;
32 for (let i = 0n; i < abs(exponent); i++) result = result.mul(factor);
33 return result;
34 }
35 toString() { return this.d === 1n ? `${this.n}` : `${this.n}/${this.d}`; }
36 decimal() {
37 const magnitude = abs(this.n);
38 const whole = magnitude / this.d, tail = magnitude % this.d * 1_000_000n / this.d;
39 return `${this.n < 0n ? '-' : ''}${whole}.${String(tail).padStart(6, '0')}`;
40 }
41 notation(unit) { return `${this} ${unit} (${this.decimal()})`; }
42}

45function numberValue(text) {
46 if (!/^-?[0-9]+([./][0-9]+)?$/.test(text)) fail(`BAD NUMBER ${text}`);
47 const negative = text.startsWith('-'), body = negative ? text.slice(1) : text;
48 let value;
49 if (body.includes('/')) {
50 const [n, d] = body.split('/').map(BigInt);
51 if (d === 0n) fail(`BAD NUMBER ${text}`);
52 value = new Fraction(n, d);
53 } else if (body.includes('.')) {
54 const [whole, digits] = body.split('.');
55 const scale = 10n ** BigInt(digits.length);
56 value = new Fraction(BigInt(whole) * scale + BigInt(digits), scale);
57 } else value = new Fraction(BigInt(body));
58 return negative ? value.neg() : value;
59}
60function exponentValue(text) {
61 const value = BigInt(text);
62 if (value === 0n || value < -3n || value > 3n) fail(`BAD EXPONENT ${text}`);
63 return value;
64}

70class Dimension {
71 constructor(exponents = Array(7).fill(0n)) { this.exponents = exponents; }
72 add(other) { return new Dimension(this.exponents.map((value, i) => value + other.exponents[i])); }
73 sub(other) { return new Dimension(this.exponents.map((value, i) => value - other.exponents[i])); }
74 scale(factor) { return new Dimension(this.exponents.map(value => value * factor)); }
75 equals(other) { return this.exponents.every((value, i) => value === other.exponents[i]); }
76 match(other) { if (!this.equals(other)) fail(`DIMENSION MISMATCH ${this} vs ${other}`); }
77 toString() {
78 const symbols = ['kg', 'm', 's', 'A', 'K', 'mol', 'cd'];
79 return this.exponents.flatMap((value, i) => value === 0n ? [] : [value === 1n ? symbols[i] : `${symbols[i]}^${value}`]).join(' ') || '1';
80 }

150const letter = c => c !== undefined && ((c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z'));
151const digit = c => c !== undefined && c >= '0' && c <= '9';

153class Parser {
154 tokens = [];
155 position = 0;
156
157 constructor(source, quantityMode) {
158 this.source = source;
159 for (let i = 0; i < source.length;) {
160 const start = i, character = source[i++];
161 let kind = 'invalid';
162 if (letter(character)) {
163 kind = 'name';
164 while (letter(source[i])) i++;
165 } else if (digit(character)) {
166 kind = 'number';
167 while (digit(source[i])) i++;
168 if (quantityMode && source[i] === '.' && digit(source[i + 1])) {
169 i++;
170 while (digit(source[i])) i++;
171 }
172 } else if (quantityMode && character === '[') {
173 const closing = source.indexOf(']', i);
174 if (closing > i) { kind = 'unit'; i = closing + 1; }
175 } else if ((quantityMode ? '+-*/^(),' : '*/^()-').includes(character)) kind = 'symbol';
176 this.tokens.push({ kind, text: source.slice(start, i), position: start + 1 });
177 }
178 this.tokens.push({ kind: 'end', text: '', position: source.length + 1 });
179 }
180 get current() { return this.tokens[this.position]; }
181 syntax() { fail(`SYNTAX ${this.source} AT ${this.current.kind === 'end' ? 'END' : this.current.position}`); }
182 advance() { return this.tokens[this.position++]; }
183 take(text) {
184 if (this.current.text !== text) return false;
185 this.position++;
186 return true;
187 }
188 expect(text) { if (!this.take(text)) this.syntax(); }
189 exponent() {
190 const sign = this.take('-') ? '-' : '';
191 if (this.current.kind !== 'number' || this.current.text.includes('.')) this.syntax();
192 return sign + this.advance().text;
193 }
194 unitTerm() {
195 let node;
196 if (this.current.kind === 'name') node = { kind: 'name', name: this.advance().text };
197 else if (this.current.kind === 'number' && this.current.text[0] !== '0') node = { kind: 'number', value: BigInt(this.advance().text) };
198 else if (this.take('(')) { node = this.unitExpression(); this.expect(')'); }
199 else this.syntax();
200 return this.take('^') ? { kind: 'power', child: node, exponent: this.exponent() } : node;
201 }
202 unitExpression() {
203 const terms = [{ operation: '*', node: this.unitTerm() }];
204 while (this.current.text === '*' || this.current.text === '/') {
205 const operation = this.advance().text;
206 terms.push({ operation, node: this.unitTerm() });
207 }
208 return { kind: 'product', terms };
209 }
210 unit() {
211 const node = this.unitExpression();
212 if (this.current.kind !== 'end') this.syntax();
213 return node;
214 }
215 primary() {
216 const token = this.current;
217 let node;
218 if (token.kind === 'number') {
219 this.advance();
220 const unit = this.current.kind === 'unit' ? this.advance().text.slice(1, -1) : null;
221 node = { kind: 'literal', number: token.text, unit };
222 } else if (token.kind === 'unit') {
223 this.advance();
224 node = { kind: 'literal', number: '1', unit: token.text.slice(1, -1) };
225 } else if (token.kind === 'name') {
226 this.advance();
227 if (this.take('(')) {
228 const args = [this.expression()];
229 while (this.take(',')) args.push(this.expression());
230 this.expect(')');
231 node = { kind: 'call', name: token.text, args };
232 } else node = { kind: 'variable', name: token.text };
233 } else if (this.take('(')) { node = this.expression(); this.expect(')'); }
234 else this.syntax();
235 return this.take('^') ? { kind: 'power', child: node, exponent: this.exponent() } : node;
236 }
237 unary() { return this.take('-') ? { kind: 'negative', child: this.unary() } : this.primary(); }
238 product() {
239 let node = this.unary();
240 while (this.current.text === '*' || this.current.text === '/') {
241 const operation = this.advance().text;
242 node = { kind: 'binary', operation, left: node, right: this.unary() };
243 }
244 return node;
245 }
246 expression() {
247 let node = this.product();
248 while (this.current.text === '+' || this.current.text === '-') {
249 const operation = this.advance().text;
250 node = { kind: 'binary', operation, left: node, right: this.product() };
251 }
252 return node;
253 }
254 quantity() {
255 const node = this.expression();
256 if (this.current.kind !== 'end') this.syntax();
257 return node;
258 }
259}

261function validateUnit(database, node) {
262 switch (node.kind) {
263 case 'name': if (!database.resolve(node.name)) fail(`UNKNOWN UNIT ${node.name}`); break;
264 case 'power': validateUnit(database, node.child); exponentValue(node.exponent); break;
265 case 'product': for (const term of node.terms) validateUnit(database, term.node); break;
266 }
267}
268function evaluateUnit(database, node) {
269 switch (node.kind) {
270 case 'name': return database.resolve(node.name);
271 case 'number': return new Quantity(new Fraction(node.value));
272 case 'power': {
273 const value = evaluateUnit(database, node.child), exponent = BigInt(node.exponent);
274 return new Quantity(value.value.pow(exponent), value.dimension.scale(exponent));
275 }
276 case 'product': {
277 let result = new Quantity();
278 for (const term of node.terms) {
279 const right = evaluateUnit(database, term.node);
280 result = term.operation === '*' ? new Quantity(result.value.mul(right.value), result.dimension.add(right.dimension)) :
281 new Quantity(result.value.div(right.value), result.dimension.sub(right.dimension));
282 }
283 return result;
284 }
285 }

294function reduceUnit(node) {
295 switch (node.kind) {
296 case 'name': return { coefficient: ONE, names: new Map([[node.name, 1n]]) };
297 case 'number': return { coefficient: new Fraction(node.value), names: new Map() };
298 case 'power': {
299 const value = reduceUnit(node.child), exponent = BigInt(node.exponent);
300 return { coefficient: value.coefficient.pow(exponent), names: new Map([...value.names].map(([name, power]) => [name, power * exponent])) };
301 }
302 case 'product': {
303 let coefficient = ONE;
304 const names = new Map();
305 for (const term of node.terms) {
306 const right = reduceUnit(term.node), multiply = term.operation === '*';
307 coefficient = multiply ? coefficient.mul(right.coefficient) : coefficient.div(right.coefficient);
308 for (const [name, power] of right.names) names.set(name, (names.get(name) ?? 0n) + (multiply ? power : -power));
309 }
310 return { coefficient, names };
311 }
312 }

314function reducedText(tree) {
315 const value = reduceUnit(tree), parts = value.coefficient.compare(ONE) === 0 ? [] : [String(value.coefficient)];
316 for (const [name, power] of value.names) if (power !== 0n) parts.push(power === 1n ? name : `${name}^${power}`);
317 return parts.join(' ') || '1';
318}
319function integerSqrt(value) {
320 if (value < 2n) return value;
321 let current = value, next = (value + 1n) / 2n;
322 while (next < current) { current = next; next = (current + value / current) / 2n; }
323 return current;
324}
325function evaluateQuantity(database, node) {
326 switch (node.kind) {
327 case 'literal': {
328 const unit = node.unit === null ? new Quantity() : unitValue(database, node.unit);
329 return new Quantity(numberValue(node.number).mul(unit.value), unit.dimension);
330 }
331 case 'variable': {
332 const value = database.variables.get(node.name);
333 if (!value) fail(`UNKNOWN VARIABLE ${node.name}`);
334 return value;
335 }
336 case 'negative': {
337 const value = evaluateQuantity(database, node.child);
338 return new Quantity(value.value.neg(), value.dimension);
339 }
340 case 'power': {
341 const value = evaluateQuantity(database, node.child), exponent = exponentValue(node.exponent);
342 if (value.value.n === 0n && exponent < 0n) fail('DIVISION BY ZERO');
343 return new Quantity(value.value.pow(exponent), value.dimension.scale(exponent));
344 }
345 case 'binary': {
346 const left = evaluateQuantity(database, node.left), right = evaluateQuantity(database, node.right);
347 if (node.operation === '+' || node.operation === '-') {
348 left.dimension.match(right.dimension);
349 return new Quantity(node.operation === '+' ? left.value.add(right.value) : left.value.sub(right.value), left.dimension);
350 }
351 if (node.operation === '*') return new Quantity(left.value.mul(right.value), left.dimension.add(right.dimension));
352 if (right.value.n === 0n) fail('DIVISION BY ZERO');
353 return new Quantity(left.value.div(right.value), left.dimension.sub(right.dimension));
354 }
355 case 'call': {
356 if (!FUNCTIONS.has(node.name)) fail(`UNKNOWN FUNCTION ${node.name}`);
357 if (node.args.length !== FUNCTIONS.get(node.name)) fail(`BAD ARGUMENTS ${node.name}`);
358 const first = evaluateQuantity(database, node.args[0]);
359 if (node.name === 'abs') return new Quantity(first.value.abs(), first.dimension);
360 if (node.name === 'sqrt') {
361 if (first.value.n < 0n) fail('NEGATIVE ROOT');
362 if (first.dimension.exponents.some(value => value % 2n !== 0n)) fail(`ODD DIMENSION ${first.dimension}`);
363 const n = integerSqrt(first.value.n), d = integerSqrt(first.value.d);
364 if (n * n !== first.value.n || d * d !== first.value.d) fail('NOT EXACT');
365 return new Quantity(new Fraction(n, d), new Dimension(first.dimension.exponents.map(value => value / 2n)));
366 }
367 const second = evaluateQuantity(database, node.args[1]);
368 first.dimension.match(second.dimension);
369 const comparison = first.value.compare(second.value);
370 return (node.name === 'min' ? comparison <= 0 : comparison >= 0) ? first : second;
371 }
372 }

376function execute(database, line) {
377 const words = line.split(' '), command = words[0];
378 switch (command) {
379 case 'DEFINE': {
380 const name = words[1];
381 checkName(name);
382 const value = numberValue(words[3]), unit = unitValue(database, words[4]);
383 if (KINDS.has(name) || database.resolve(name)) fail(`NAME TAKEN ${name}`);
384 if (value.n <= 0n) fail(`NOT POSITIVE ${words[3]}`);
385 const result = new Quantity(value.mul(unit.value), unit.dimension);
386 database.units.push({ symbol: name, quantity: result, prefixable: false, definition: { value, expression: words[4] } });
387 return [`DEFINE ${name} factor=${result.value} dim=${result.dimension} kind=${kindName(result.dimension)}`];
388 }
389 case 'UNDEF': {
390 const index = database.units.findIndex(unit => unit.symbol === words[1]);
391 if (index < 0) fail(`NOT DEFINED ${words[1]}`);
392 if (index < BUILTINS.length) fail(`BUILTIN UNIT ${words[1]}`);
393 database.units.splice(index, 1);
394 return [`UNDEF ${words[1]}`];
395 }
396 case 'DEFS': return [`DEFS count=${database.units.length - BUILTINS.length}`, ...database.units.slice(BUILTINS.length).map(unit =>
397 `DEF ${unit.symbol} = ${unit.definition.value} ${unit.definition.expression} factor=${unit.quantity.value} dim=${unit.quantity.dimension}`)];
398 case 'CONVERT': case 'INVERT': {
399 const value = numberValue(words[1]), source = unitValue(database, words[2]), target = unitValue(database, words[4]);
400 if (command === 'INVERT') {
401 if (source.dimension.equals(DIMENSIONLESS) || !source.dimension.add(target.dimension).equals(DIMENSIONLESS)) {
402 fail(`NOT RECIPROCAL ${source.dimension} vs ${target.dimension}`);
403 }
404 if (value.n === 0n) fail('DIVISION BY ZERO');
405 } else source.dimension.match(target.dimension);
406 let result = value.mul(source.value);
407 if (command === 'INVERT') result = ONE.div(result);
408 return [`${command} ${value} ${words[2]} = ${result.div(target.value).notation(words[4])}`];
409 }
410 case 'DIM': case 'REDUCE': case 'SIMPLIFY': {
411 const tree = parseUnit(database, words[1]), unit = evaluateUnit(database, tree);
412 const result = command === 'DIM' ? `${unit.dimension} factor=${unit.value} kind=${kindName(unit.dimension)}` :
413 command === 'REDUCE' ? reducedText(tree) : unit.value.notation(database.simplified(unit.dimension));
414 return [`${command} ${words[1]} = ${result}`];
415 }
416 case 'ADD': case 'SUB': case 'MUL': case 'DIV': case 'COMPARE': {
417 const a = numberValue(words[1]), first = unitValue(database, words[2]);
418 const b = numberValue(words[3]), second = unitValue(database, words[4]);
419 const hasTarget = words.length === 7, target = hasTarget ? unitValue(database, words[6]) : first;
420 const heading = `${command} ${a} ${words[2]}`, suffix = `${b} ${words[4]}`;
421 if (command === 'ADD' || command === 'SUB' || command === 'COMPARE') {
422 first.dimension.match(second.dimension);
423 if (hasTarget) first.dimension.match(target.dimension);
424 const left = a.mul(first.value), right = b.mul(second.value);
425 if (command === 'COMPARE') {
426 const relation = ['<', '=', '>'][left.compare(right) + 1];
427 return [`${heading} ${relation} ${suffix} ratio=${right.n === 0n ? 'undefined' : left.div(right)}`];
428 }
429 const add = command === 'ADD', result = add ? left.add(right) : left.sub(right);
430 return [`${heading} ${add ? '+' : '-'} ${suffix} = ${result.div(target.value).notation(hasTarget ? words[6] : words[2])}`];
431 }
432 const multiply = command === 'MUL', dimension = multiply ? first.dimension.add(second.dimension) : first.dimension.sub(second.dimension);
433 if (!multiply && b.n === 0n) fail('DIVISION BY ZERO');
434 if (hasTarget) dimension.match(target.dimension);
435 const raw = multiply ? a.mul(b) : a.div(b), left = a.mul(first.value), right = b.mul(second.value);
436 const coherent = multiply ? left.mul(right) : left.div(right), operation = multiply ? '*' : '/';
437 const prefix = `${heading} ${operation} ${suffix} =`;
438 if (hasTarget) return [`${prefix} ${coherent.div(target.value).notation(words[6])}`];
439 const rightUnit = !multiply && /[*/]/.test(words[4]) ? `(${words[4]})` : words[4];
440 return [`${prefix} ${raw} ${words[2]}${operation}${rightUnit} = ${coherent.notation(database.simplified(dimension))}`];
441 }
442 case 'TEMP': {
443 const value = numberValue(words[1]);
444 for (const scale of [words[2], words[4]]) if (!['C', 'F', 'K', 'R'].includes(scale)) fail(`UNKNOWN SCALE ${scale}`);
445 const celsius = new Fraction(5463n, 20n), fahrenheit = new Fraction(45967n, 100n);
446 const kelvin = words[2] === 'C' ? value.add(celsius) : words[2] === 'F' ? value.add(fahrenheit).mul(new Fraction(5n, 9n)) :
447 words[2] === 'R' ? value.mul(new Fraction(5n, 9n)) : value;
448 if (kelvin.n < 0n) fail('BELOW ABSOLUTE ZERO');
449 const result = words[4] === 'C' ? kelvin.sub(celsius) : words[4] === 'F' ? kelvin.mul(new Fraction(9n, 5n)).sub(fahrenheit) :
450 words[4] === 'R' ? kelvin.mul(new Fraction(9n, 5n)) : kelvin;
451 return [`TEMP ${value} ${words[2]} = ${result.notation(words[4])}`];
452 }
453 case 'LIST': {
454 const dimension = KINDS.get(words[1]) ?? unitValue(database, words[1]).dimension;
455 const units = database.selected(dimension);
456 return [`LIST ${words[1]} dim=${dimension} kind=${kindName(dimension)} count=${units.length}`,
457 ...units.map(unit => `UNIT ${unit.symbol} factor=${unit.quantity.value} origin=${unit.definition ? 'user' : 'builtin'} prefix=${unit.prefixable ? 'yes' : 'no'}`)];
458 }
459 case 'TABLE': {
460 const value = numberValue(words[1]), unit = unitValue(database, words[2]), units = database.selected(unit.dimension);
461 return [`TABLE ${value} ${words[2]} dim=${unit.dimension} count=${units.length}`,
462 ...units.map(item => `ROW ${value.mul(unit.value).div(item.quantity.value).notation(item.symbol)}`)];
463 }
464 case 'AUTO': {
465 const value = numberValue(words[1]), unit = unitValue(database, words[2]);
466 const reference = BUILTINS.find(item => item.prefixable && item.quantity.dimension.equals(unit.dimension));
467 if (!reference) fail(`NO PREFIXABLE UNIT ${unit.dimension}`);
468 const scaled = value.mul(unit.value).div(reference.quantity.value), magnitude = scaled.abs();
469 let exponent = 0n;
470 if (magnitude.compare(ONE) >= 0) {
471 while (exponent < 9n && magnitude.compare(TEN.pow(exponent + 3n)) >= 0) exponent += 3n;
472 } else if (magnitude.n > 0n) {
473 while (exponent > -9n && magnitude.compare(TEN.pow(exponent)) < 0) exponent -= 3n;
474 }
475 const prefix = ['n', 'u', 'm', '', 'k', 'M', 'G'][Number((exponent + 9n) / 3n)];
476 return [`AUTO ${value} ${words[2]} = ${scaled.div(TEN.pow(exponent)).notation(prefix + reference.symbol)}`];
477 }
478 case 'SPLIT': {
479 const value = numberValue(words[1]), unit = unitValue(database, words[2]);
480 const parts = words[4].split(','), targets = parts.map(part => unitValue(database, part));
481 if (value.n < 0n) fail('NEGATIVE VALUE');
482 for (const target of targets) unit.dimension.match(target.dimension);
483 for (let i = 0; i + 1 < targets.length; i++) if (targets[i].value.compare(targets[i + 1].value) <= 0) fail('UNIT ORDER');
484 let remaining = value.mul(unit.value);
485 const pieces = [];
486 for (let i = 0; i + 1 < targets.length; i++) {
487 const quotient = remaining.div(targets[i].value), whole = quotient.n / quotient.d;
488 remaining = remaining.sub(new Fraction(whole).mul(targets[i].value));
489 pieces.push(`${whole} ${parts[i]}`);
490 }
491 pieces.push(remaining.div(targets.at(-1).value).notation(parts.at(-1)));
492 return [`SPLIT ${value} ${words[2]} = ${pieces.join(' ')}`];
493 }
494 case 'LET': {
495 checkName(words[1], true);
496 const value = quantityValue(database, words[3]);
497 database.variables.set(words[1], value);
498 return [`LET ${words[1]} = ${value.value.notation(database.simplified(value.dimension))}`];
499 }
500 case 'EVAL': {
501 const value = quantityValue(database, words[1]);
502 let result = value.value, label = database.simplified(value.dimension);
503 if (words.length === 4) {
504 const target = unitValue(database, words[3]);
505 value.dimension.match(target.dimension);
506 result = result.div(target.value);
507 label = words[3];
508 }
509 return [`EVAL ${words[1]} = ${result.notation(label)}`];
510 }
511 case 'VARS': return [`VARS count=${database.variables.size}`, ...[...database.variables.keys()].sort().map(name => {
512 const value = database.variables.get(name);
513 return `VAR ${name} = ${value.value.notation(database.simplified(value.dimension))}`;
514 })];
515 default: fail(`UNKNOWN COMMAND ${command}`);
516 }
517}

519const lines = readFileSync(0, 'utf8').split(/\r?\n/), count = Number(lines[0]);
520const database = new Database(), output = [];
521for (let i = 1; i <= count; i++) {
522 try { output.push(...execute(database, lines[i])); }
523 catch (error) {
524 if (!(error instanceof CommandError)) throw error;
525 output.push(`ERROR ${error.message}`);
526 }
527}
528process.stdout.write(output.join('\n') + '\n');
