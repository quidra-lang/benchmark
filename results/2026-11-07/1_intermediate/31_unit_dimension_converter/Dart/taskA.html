8final class Fraction implements Comparable<Fraction> {
9 final int numerator, denominator;
10 const Fraction._(this.numerator, this.denominator);
11 static const one = Fraction._(1, 1);
12
13 factory Fraction(int numerator, [int denominator = 1]) {
14 if (denominator < 0) {
15 numerator = -numerator;
16 denominator = -denominator;
17 }
18 final divisor = numerator.abs().gcd(denominator);
19 return Fraction._(numerator ~/ divisor, denominator ~/ divisor);
20 }
21
22 factory Fraction.parse(String text) {
23 final negative = text.startsWith('-');
24 final body = negative ? text.substring(1) : text;
25 final separator = RegExp(r'[./]').firstMatch(body)?.start ?? -1;
26 final first = separator < 0 ? body : body.substring(0, separator);
27 final second = separator < 0 ? '1' : body.substring(separator + 1);
28 final digits = RegExp(r'^[0-9]+$');
29 if (!digits.hasMatch(first) || !digits.hasMatch(second)) {
30 throw CommandError('BAD NUMBER $text');
31 }
32 var numerator = int.parse(first, radix: 10), denominator = 1;
33 if (separator >= 0 && body[separator] == '/') {
34 denominator = int.parse(second, radix: 10);
35 if (denominator == 0) throw CommandError('BAD NUMBER $text');
36 } else if (separator >= 0) {
37 for (var i = 0; i < second.length; i++) denominator *= 10;
38 numerator = numerator * denominator + int.parse(second, radix: 10);
39 }
40 return Fraction(negative ? -numerator : numerator, denominator);
41 }
42
43 bool get isZero => numerator == 0;
44 Fraction operator +(Fraction other) => Fraction(
45 numerator * other.denominator + other.numerator * denominator,
46 denominator * other.denominator);
47 Fraction operator -(Fraction other) => Fraction(
48 numerator * other.denominator - other.numerator * denominator,
49 denominator * other.denominator);
50 Fraction operator *(Fraction other) => Fraction(numerator * other.numerator, denominator * other.denominator);
51 Fraction operator /(Fraction other) => Fraction(numerator * other.denominator, denominator * other.numerator);
52 Fraction operator -() => Fraction(-numerator, denominator);
53 Fraction abs() => Fraction(numerator.abs(), denominator);
54 Fraction power(int exponent) {
55 final factor = exponent < 0 ? one / this : this;
56 var result = one;
57 for (var i = 0; i < exponent.abs(); i++) result = result * factor;
58 return result;
59 }
60 String decimal() {
61 final magnitude = numerator.abs();
62 final places = (magnitude % denominator * 1000000 ~/ denominator).toString().padLeft(6, '0');
63 return '${numerator < 0 ? '-' : ''}${magnitude ~/ denominator}.$places';
64 }
65 @override
66 int compareTo(Fraction other) => (numerator * other.denominator).compareTo(other.numerator * denominator);
67 @override
68 bool operator ==(Object other) => other is Fraction && numerator == other.numerator && denominator == other.denominator;
69 @override
70 int get hashCode => Object.hash(numerator, denominator);
71 @override
72 String toString() => denominator == 1 ? '$numerator' : '$numerator/$denominator';
73}

75final class Dimension {
76 final int kg, m, s, a, k, mol, cd;
77 const Dimension({this.kg = 0, this.m = 0, this.s = 0, this.a = 0, this.k = 0, this.mol = 0, this.cd = 0});
78 static const zero = Dimension();
79 List<int> get exponents => [kg, m, s, a, k, mol, cd];
80 bool get isZero => this == zero;
81 bool get isEven => exponents.every((value) => value % 2 == 0);
82 Dimension operator +(Dimension other) => Dimension(
83 kg: kg + other.kg, m: m + other.m, s: s + other.s, a: a + other.a,
84 k: k + other.k, mol: mol + other.mol, cd: cd + other.cd);
85 Dimension operator -(Dimension other) => Dimension(
86 kg: kg - other.kg, m: m - other.m, s: s - other.s, a: a - other.a,
87 k: k - other.k, mol: mol - other.mol, cd: cd - other.cd);
88 Dimension scale(int factor) => Dimension(
89 kg: kg * factor, m: m * factor, s: s * factor, a: a * factor,
90 k: k * factor, mol: mol * factor, cd: cd * factor);
91 Dimension half() => Dimension(kg: kg ~/ 2, m: m ~/ 2, s: s ~/ 2, a: a ~/ 2, k: k ~/ 2, mol: mol ~/ 2, cd: cd ~/ 2);
92 @override
93 bool operator ==(Object other) => other is Dimension &&
94 kg == other.kg && m == other.m && s == other.s && a == other.a &&
95 k == other.k && mol == other.mol && cd == other.cd;
96 @override
97 int get hashCode => Object.hash(kg, m, s, a, k, mol, cd);
98 @override
99 String toString() {
100 const symbols = ['kg', 'm', 's', 'A', 'K', 'mol', 'cd'];
101 final values = exponents;
102 final parts = <String>[];
103 for (var i = 0; i < values.length; i++) {
104 if (values[i] != 0) parts.add(values[i] == 1 ? symbols[i] : '${symbols[i]}^${values[i]}');
105 }
106 return parts.isEmpty ? '1' : parts.join(' ');
107 }
108}

136abstract class Parser {
137 final String source;
138 final tokens = <Token>[];
139 var index = 0;
140 Token get current => tokens[index];
141
142 Parser(this.source, bool quantity) {
143 for (var i = 0; i < source.length;) {
144 final start = i++;
145 final c = source.codeUnitAt(start);
146 TokenKind kind;
147 if (isLetter(c)) {
148 while (i < source.length && isLetter(source.codeUnitAt(i))) i++;
149 kind = TokenKind.identifier;
150 } else if (isDigit(c)) {
151 while (i < source.length && isDigit(source.codeUnitAt(i))) i++;
152 if (quantity && i + 1 < source.length && source[i] == '.' && isDigit(source.codeUnitAt(i + 1))) {
153 i += 2;
154 while (i < source.length && isDigit(source.codeUnitAt(i))) i++;
155 }
156 kind = TokenKind.number;
157 } else {
158 final end = quantity && c == 91 ? source.indexOf(']', i) : -1;
159 if (end > i) {
160 i = end + 1;
161 kind = TokenKind.unit;
162 } else {
163 kind = (quantity ? '+-*/^(),' : '*/^()-').contains(source[start]) ? TokenKind.symbol : TokenKind.invalid;
164 }
165 }
166 tokens.add(Token(kind, source.substring(start, i), start + 1));
167 }
168 tokens.add(Token(TokenKind.end, '', source.length + 1));
169 }
170
171 static bool isLetter(int c) => c >= 65 && c <= 90 || c >= 97 && c <= 122;
172 static bool isDigit(int c) => c >= 48 && c <= 57;
173 Token advance() => tokens[index++];
174 bool take(String text) {
175 if (current.text != text) return false;
176 index++;
177 return true;
178 }
179 void expect(String text) { if (!take(text)) throw syntax(); }
180 CommandError syntax() => CommandError('SYNTAX $source AT ${current.kind == TokenKind.end ? 'END' : current.position}');
181 void finish() { if (current.kind != TokenKind.end) throw syntax(); }
182 String exponent() {
183 final sign = take('-') ? '-' : '';
184 if (current.kind != TokenKind.number || current.text.contains('.')) throw syntax();
185 return sign + advance().text;
186 }
187}

208final class UnitParser extends Parser {
209 UnitParser(String source) : super(source, false);
210 UnitNode parse() { final node = expression(); finish(); return node; }
211 UnitNode expression() {
212 final terms = <(String, UnitNode)>[('*', term())];
213 while (current.text == '*' || current.text == '/') {
214 final op = advance().text;
215 terms.add((op, term()));
216 }
217 return UnitProduct(terms);
218 }
219 UnitNode term() {
220 UnitNode atom;
221 if (current.kind == TokenKind.identifier) {
222 atom = UnitName(advance().text);
223 } else if (current.kind == TokenKind.number && current.text[0] != '0') {
224 atom = UnitNumber(advance().text);
225 } else if (take('(')) {
226 atom = expression();
227 expect(')');
228 } else {
229 throw syntax();
230 }
231 return take('^') ? UnitPower(atom, exponent()) : atom;
232 }
233}

265final class QuantityParser extends Parser {
266 QuantityParser(String source) : super(source, true);
267 QuantityNode parse() { final node = expression(); finish(); return node; }
268 QuantityNode expression() {
269 var node = product();
270 while (current.text == '+' || current.text == '-') {
271 final op = advance().text;
272 node = Binary(op, node, product());
273 }
274 return node;
275 }
276 QuantityNode product() {
277 var node = unary();
278 while (current.text == '*' || current.text == '/') {
279 final op = advance().text;
280 node = Binary(op, node, unary());
281 }
282 return node;
283 }
284 QuantityNode unary() {
285 if (take('-')) return Negative(unary());
286 final node = primary();
287 return take('^') ? Power(node, exponent()) : node;
288 }
289 QuantityNode primary() {
290 if (current.kind == TokenKind.number) {
291 final number = advance().text;
292 return Literal(number, current.kind == TokenKind.unit ? unitString() : null);
293 }
294 if (current.kind == TokenKind.unit) return Literal('1', unitString());
295 if (current.kind == TokenKind.identifier) {
296 final name = advance().text;
297 if (!take('(')) return Variable(name);
298 final arguments = <QuantityNode>[expression()];
299 while (take(',')) arguments.add(expression());
300 expect(')');
301 return Call(name, arguments);
302 }
303 if (take('(')) {
304 final node = expression();
305 expect(')');
306 return node;
307 }
308 throw syntax();
309 }
310 String unitString() {
311 final text = advance().text;
312 return text.substring(1, text.length - 1);
313 }

407 ParsedUnit parseUnit(String text) {
408 final tree = UnitParser(text).parse();
409 final names = <String>[], seen = <String>{};
410 void validate(UnitNode node) {
411 switch (node) {
412 case UnitName name:
413 if (resolve(name.name) == null) throw CommandError('UNKNOWN UNIT ${name.name}');
414 if (seen.add(name.name)) names.add(name.name);
415 case UnitNumber _:
416 break;
417 case UnitPower power:
418 validate(power.atom);
419 checkExponent(power.exponent);
420 case UnitProduct product:
421 for (final (_, term) in product.terms) validate(term);
422 }
423 }
424 validate(tree);
425 return (tree: tree, quantity: unitValue(tree), names: names);
426 }
427 Quantity unitValue(UnitNode node) {
428 switch (node) {
429 case UnitName name:
430 return resolve(name.name)!;
431 case UnitNumber number:
432 return Quantity(Fraction(int.parse(number.text, radix: 10)), Dimension.zero);
433 case UnitPower power:
434 final atom = unitValue(power.atom);
435 final exponent = int.parse(power.exponent, radix: 10);
436 return Quantity(atom.value.power(exponent), atom.dimension.scale(exponent));
437 case UnitProduct product:
438 var result = const Quantity(Fraction.one, Dimension.zero);
439 for (final (op, term) in product.terms) {
440 final value = unitValue(term);
441 result = op == '*'
442 ? Quantity(result.value * value.value, result.dimension + value.dimension)
443 : Quantity(result.value / value.value, result.dimension - value.dimension);
444 }
445 return result;
446 }
447 }
448 static Reduced reduce(UnitNode node) {
449 switch (node) {
450 case UnitName name:
451 return (coefficient: Fraction.one, exponents: {name.name: 1});
452 case UnitNumber number:
453 return (coefficient: Fraction(int.parse(number.text, radix: 10)), exponents: <String, int>{});
454 case UnitPower power:
455 final atom = reduce(power.atom);
456 final exponent = int.parse(power.exponent, radix: 10);
457 return (coefficient: atom.coefficient.power(exponent),
458 exponents: {for (final entry in atom.exponents.entries) entry.key: entry.value * exponent});
459 case UnitProduct product:
460 var coefficient = Fraction.one;
461 final exponents = <String, int>{};
462 for (final (op, term) in product.terms) {
463 final value = reduce(term);
464 coefficient = op == '*' ? coefficient * value.coefficient : coefficient / value.coefficient;
465 for (final entry in value.exponents.entries) {
466 exponents[entry.key] = (exponents[entry.key] ?? 0) + (op == '*' ? entry.value : -entry.value);
467 }
468 }
469 return (coefficient: coefficient, exponents: exponents);
470 }

472 static String reducedText(ParsedUnit parsed) {
473 final reduced = reduce(parsed.tree);
474 final parts = <String>[];
475 if (reduced.coefficient != Fraction.one) parts.add(reduced.coefficient.toString());
476 for (final name in parsed.names) {
477 final exponent = reduced.exponents[name]!;

482 static int integerRoot(int number) {
483 var low = 0, high = number + 1;
484 while (high - low > 1) {
485 final middle = low + (high - low) ~/ 2;
486 if (middle <= number ~/ middle) {
487 low = middle;
488 } else {
489 high = middle;
490 }
491 }
492 return low;
493 }
494 Quantity evaluate(QuantityNode node) {
495 switch (node) {
496 case Literal literal:
497 final unitText = literal.unit;
498 if (unitText == null) return Quantity(Fraction.parse(literal.number), Dimension.zero);
499 final unit = parseUnit(unitText).quantity;
500 return Quantity(Fraction.parse(literal.number) * unit.value, unit.dimension);
501 case Variable variable:
502 return variables[variable.name] ?? (throw CommandError('UNKNOWN VARIABLE ${variable.name}'));
503 case Negative negative:
504 final operand = evaluate(negative.operand);
505 return Quantity(-operand.value, operand.dimension);
506 case Power power:
507 final basis = evaluate(power.base);
508 final exponent = checkExponent(power.exponent);
509 if (basis.value.isZero && exponent < 0) throw CommandError('DIVISION BY ZERO');
510 return Quantity(basis.value.power(exponent), basis.dimension.scale(exponent));
511 case Binary binary:
512 final left = evaluate(binary.left);
513 final right = evaluate(binary.right);
514 if (binary.op == '+' || binary.op == '-') {
515 match(left.dimension, right.dimension);
516 return Quantity(binary.op == '+' ? left.value + right.value : left.value - right.value, left.dimension);
517 }
518 if (binary.op == '*') return Quantity(left.value * right.value, left.dimension + right.dimension);
519 if (right.value.isZero) throw CommandError('DIVISION BY ZERO');
520 return Quantity(left.value / right.value, left.dimension - right.dimension);
521 case Call call:
522 if (!const ['abs', 'sqrt', 'min', 'max'].contains(call.name)) throw CommandError('UNKNOWN FUNCTION ${call.name}');
523 final count = call.name == 'abs' || call.name == 'sqrt' ? 1 : 2;
524 if (call.arguments.length != count) throw CommandError('BAD ARGUMENTS ${call.name}');
525 final a = evaluate(call.arguments[0]);
526 if (call.name == 'abs') return Quantity(a.value.abs(), a.dimension);
527 if (call.name == 'sqrt') {
528 if (a.value.numerator < 0) throw CommandError('NEGATIVE ROOT');
529 if (!a.dimension.isEven) throw CommandError('ODD DIMENSION ${a.dimension}');
530 final p = integerRoot(a.value.numerator), q = integerRoot(a.value.denominator);
531 if (p * p != a.value.numerator || q * q != a.value.denominator) throw CommandError('NOT EXACT');
532 return Quantity(Fraction(p, q), a.dimension.half());
533 }
534 final b = evaluate(call.arguments[1]);
535 match(a.dimension, b.dimension);
536 final comparison = a.value.compareTo(b.value);
537 return call.name == 'min' ? (comparison <= 0 ? a : b) : (comparison >= 0 ? a : b);
538 }
539 }

541 final selected = units.where((unit) => unit.quantity.dimension == dimension).toList();
542 selected.sort((a, b) {
543 final comparison = a.quantity.value.compareTo(b.quantity.value);
544 return comparison != 0 ? comparison : a.symbol.compareTo(b.symbol);
545 });
546 return selected;
547 }

549 String execute(String line) {
550 final words = line.split(' '), command = words[0];
551 switch (command) {
552 case 'DEFINE': {
553 checkName(words[1], false);
554 final value = Fraction.parse(words[3]);
555 final unit = parseUnit(words[4]).quantity;
556 if (kinds.containsKey(words[1]) || resolve(words[1]) != null) throw CommandError('NAME TAKEN ${words[1]}');
557 if (value.numerator <= 0) throw CommandError('NOT POSITIVE ${words[3]}');
558 final defined = Quantity(value * unit.value, unit.dimension);
559 units.add(Unit(words[1], defined, false, Definition(value, words[4])));
560 return 'DEFINE ${words[1]} factor=${defined.value} dim=${defined.dimension} kind=${kind(defined.dimension)}';
561 }
562 case 'UNDEF': {
563 final index = units.indexWhere((unit) => unit.symbol == words[1]);
564 if (index >= 0 && units[index].definition == null) throw CommandError('BUILTIN UNIT ${words[1]}');
565 if (index < 0) throw CommandError('NOT DEFINED ${words[1]}');
566 units.removeAt(index);
567 return 'UNDEF ${words[1]}';
568 }
569 case 'DEFS': {
570 final users = units.where((unit) => unit.definition != null).toList();
571 return [
572 'DEFS count=${users.length}',
573 for (final unit in users)
574 'DEF ${unit.symbol} = ${unit.definition!.value} ${unit.definition!.expression} factor=${unit.quantity.value} dim=${unit.quantity.dimension}',
575 ].join('\n');
576 }
577 case 'CONVERT': case 'INVERT': {
578 final value = Fraction.parse(words[1]);
579 final a = parseUnit(words[2]).quantity;
580 final b = parseUnit(words[4]).quantity;
581 if (command == 'CONVERT') {
582 match(a.dimension, b.dimension);
583 } else {
584 if (a.dimension.isZero || !(a.dimension + b.dimension).isZero) {
585 throw CommandError('NOT RECIPROCAL ${a.dimension} vs ${b.dimension}');
586 }
587 if (value.isZero) throw CommandError('DIVISION BY ZERO');
588 }
589 final result = command == 'CONVERT' ? value * a.value / b.value : (Fraction.one / (value * a.value)) / b.value;
590 return '$command $value ${words[2]} = ${format(result, words[4])}';
591 }
592 case 'DIM': case 'REDUCE': case 'SIMPLIFY': {
593 final parsed = parseUnit(words[1]), unit = parsed.quantity;
594 return switch (command) {
595 'DIM' => 'DIM ${words[1]} = ${unit.dimension} factor=${unit.value} kind=${kind(unit.dimension)}',
596 'REDUCE' => 'REDUCE ${words[1]} = ${reducedText(parsed)}',
597 _ => 'SIMPLIFY ${words[1]} = ${formatQuantity(unit)}',
598 };
599 }
600 case 'ADD': case 'SUB': case 'MUL': case 'DIV': case 'COMPARE': {
601 final x = Fraction.parse(words[1]);
602 final a = parseUnit(words[2]).quantity;
603 final y = Fraction.parse(words[3]);
604 final b = parseUnit(words[4]).quantity;
605 final target = words.length == 7 ? parseUnit(words[6]).quantity : null;
606 if (command == 'COMPARE') {
607 match(a.dimension, b.dimension);
608 final left = x * a.value, right = y * b.value;
609 final comparison = left.compareTo(right);
610 final relation = comparison < 0 ? '<' : comparison > 0 ? '>' : '=';
611 final ratio = right.isZero ? 'undefined' : (left / right).toString();
612 return 'COMPARE $x ${words[2]} $relation $y ${words[4]} ratio=$ratio';
613 }
614 final op = switch (command) { 'ADD' => '+', 'SUB' => '-', 'MUL' => '*', _ => '/' };
615 final prefix = '$command $x ${words[2]} $op $y ${words[4]} = ';
616 if (command == 'ADD' || command == 'SUB') {
617 match(a.dimension, b.dimension);
618 if (target != null) match(a.dimension, target.dimension);
619 final left = x * a.value, right = y * b.value;
620 final result = (command == 'ADD' ? left + right : left - right) / (target?.value ?? a.value);
621 return prefix + format(result, target == null ? words[2] : words[6]);
622 }
623 final dimension = command == 'MUL' ? a.dimension + b.dimension : a.dimension - b.dimension;
624 if (command == 'DIV' && y.isZero) throw CommandError('DIVISION BY ZERO');
625 if (target != null) match(dimension, target.dimension);
626 final product = command == 'MUL' ? x * y : x / y;
627 final first = x * a.value, second = y * b.value;
628 final coherent = command == 'MUL' ? first * second : first / second;
629 if (target != null) return prefix + format(coherent / target.value, words[6]);
630 final rightUnit = command == 'DIV' && (words[4].contains('*') || words[4].contains('/')) ? '(${words[4]})' : words[4];
631 return '$prefix$product ${words[2]}$op$rightUnit = ${format(coherent, simplified(dimension))}';
632 }
633 case 'TEMP': {
634 final value = Fraction.parse(words[1]);
635 for (final scale in [words[2], words[4]]) {
636 if (!const ['C', 'F', 'K', 'R'].contains(scale)) throw CommandError('UNKNOWN SCALE $scale');
637 }
638 final kelvin = switch (words[2]) {
639 'C' => value + Fraction(5463, 20),
640 'F' => (value + Fraction(45967, 100)) * Fraction(5) / Fraction(9),
641 'R' => value * Fraction(5) / Fraction(9), _ => value,
642 };
643 if (kelvin.numerator < 0) throw CommandError('BELOW ABSOLUTE ZERO');
644 final result = switch (words[4]) {
645 'C' => kelvin - Fraction(5463, 20),
646 'F' => kelvin * Fraction(9) / Fraction(5) - Fraction(45967, 100),
647 'R' => kelvin * Fraction(9) / Fraction(5), _ => kelvin,
648 };
649 return 'TEMP $value ${words[2]} = ${format(result, words[4])}';
650 }
651 case 'LIST': {
652 final dimension = kinds[words[1]] ?? parseUnit(words[1]).quantity.dimension;
653 final selected = matching(dimension);
654 return [
655 'LIST ${words[1]} dim=$dimension kind=${kind(dimension)} count=${selected.length}',
656 for (final unit in selected)
657 'UNIT ${unit.symbol} factor=${unit.quantity.value} origin=${unit.definition == null ? 'builtin' : 'user'} prefix=${unit.prefix ? 'yes' : 'no'}',
658 ].join('\n');
659 }
660 case 'TABLE': case 'AUTO': case 'SPLIT': {
661 final value = Fraction.parse(words[1]);
662 final unit = parseUnit(words[2]).quantity;
663 if (command == 'TABLE') {
664 final selected = matching(unit.dimension);
665 return [
666 'TABLE $value ${words[2]} dim=${unit.dimension} count=${selected.length}',
667 for (final row in selected) 'ROW ${format(value * unit.value / row.quantity.value, row.symbol)}',
668 ].join('\n');
669 }
670 if (command == 'AUTO') {
671 Unit? basis;
672 for (final candidate in units) {
673 if (candidate.definition == null && candidate.prefix && candidate.quantity.dimension == unit.dimension) {
674 basis = candidate;
675 break;
676 }
677 }
678 if (basis == null) throw CommandError('NO PREFIXABLE UNIT ${unit.dimension}');
679 final scaled = value * unit.value / basis.quantity.value, magnitude = scaled.abs();
680 var exponent = 0;
681 if (magnitude.compareTo(Fraction.one) >= 0) {
682 while (exponent < 9 && magnitude.compareTo(Fraction(10).power(exponent + 3)) >= 0) exponent += 3;
683 } else if (!magnitude.isZero) {
684 while (exponent > -9 && magnitude.compareTo(Fraction(10).power(exponent)) < 0) exponent -= 3;
685 }
686 final prefix = switch (exponent) { 9 => 'G', 6 => 'M', 3 => 'k', -3 => 'm', -6 => 'u', -9 => 'n', _ => '' };
687 return 'AUTO $value ${words[2]} = ${format(scaled / Fraction(10).power(exponent), prefix + basis.symbol)}';
688 }
689 final parts = words[4].split(',');
690 final destinations = <Quantity>[];
691 for (final part in parts) destinations.add(parseUnit(part).quantity);
692 if (value.numerator < 0) throw CommandError('NEGATIVE VALUE');
693 for (final destination in destinations) match(unit.dimension, destination.dimension);
694 for (var i = 1; i < destinations.length; i++) {
695 if (destinations[i - 1].value.compareTo(destinations[i].value) <= 0) throw CommandError('UNIT ORDER');
696 }
697 var remaining = value * unit.value;
698 final pieces = <String>[];
699 for (var i = 0; i < parts.length - 1; i++) {
700 final quotient = remaining / destinations[i].value;
701 final whole = quotient.numerator ~/ quotient.denominator;
702 remaining = remaining - Fraction(whole) * destinations[i].value;
703 pieces.add('$whole ${parts[i]}');
704 }
705 pieces.add(format(remaining / destinations.last.value, parts.last));
706 return 'SPLIT $value ${words[2]} = ${pieces.join(' ')}';
707 }
708 case 'LET': {
709 checkName(words[1], true);
710 final value = evaluate(QuantityParser(words[3]).parse());
711 variables[words[1]] = value;
712 return 'LET ${words[1]} = ${formatQuantity(value)}';
713 }
714 case 'EVAL': {
715 final value = evaluate(QuantityParser(words[1]).parse());
716 if (words.length == 4) {
717 final unit = parseUnit(words[3]).quantity;
718 match(value.dimension, unit.dimension);
719 return 'EVAL ${words[1]} = ${format(value.value / unit.value, words[3])}';
720 }
721 return 'EVAL ${words[1]} = ${formatQuantity(value)}';
722 }
723 case 'VARS': {
724 final names = variables.keys.toList()..sort();
725 return ['VARS count=${variables.length}', for (final name in names) 'VAR $name = ${formatQuantity(variables[name]!)}'].join('\n');
726 }
727 default:
728 throw CommandError('UNKNOWN COMMAND $command');
729 }
730 }
731}

742 }
