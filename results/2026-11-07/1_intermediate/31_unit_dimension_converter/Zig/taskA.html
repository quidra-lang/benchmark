29fn equal(a: []const u8, b: []const u8) bool {
30 return std.mem.eql(u8, a, b);
31}
32fn letter(c: u8) bool {
33 return (c >= 'A' and c <= 'Z') or (c >= 'a' and c <= 'z');
34}
35fn digit(c: u8) bool {
36 return c >= '0' and c <= '9';
37}
38fn digits(text: []const u8) bool {
39 if (text.len == 0) return false;
40 for (text) |c| if (!digit(c)) return false;
41 return true;
42}
43fn integer(text: []const u8) i64 {
44 return std.fmt.parseInt(i64, text, 10) catch unreachable;
45}
46fn absolute(value: i64) i64 {
47 return if (value < 0) -value else value;
48}
49fn line(out: *Writer, comptime fmt: []const u8, args: anytype) Writer.Error!void {

52fn gcd(a: i64, b: i64) i64 {
53 var x = absolute(a);
54 var y = absolute(b);
55 while (y != 0) {
56 const remainder = @mod(x, y);
57 x = y;
58 y = remainder;
59 }
60 return x;
61}

63const Fraction = struct {
64 n: i64 = 0,
65 d: i64 = 1,
66 const one: Fraction = .{ .n = 1 };
67
68 fn init(n: i64, d: i64) Fraction {
69 const divisor = gcd(n, d) * @as(i64, if (d < 0) -1 else 1);
70 return .{ .n = @divTrunc(n, divisor), .d = @divTrunc(d, divisor) };
71 }
72 fn whole(n: i64) Fraction {
73 return .{ .n = n };
74 }
75 fn add(a: Fraction, b: Fraction) Fraction {
76 return init(a.n * b.d + b.n * a.d, a.d * b.d);
77 }
78 fn sub(a: Fraction, b: Fraction) Fraction {
79 return init(a.n * b.d - b.n * a.d, a.d * b.d);
80 }
81 fn mul(a: Fraction, b: Fraction) Fraction {
82 return init(a.n * b.n, a.d * b.d);
83 }
84 fn div(a: Fraction, b: Fraction) Fraction {
85 return init(a.n * b.d, a.d * b.n);
86 }
87 fn neg(a: Fraction) Fraction {
88 return .{ .n = -a.n, .d = a.d };
89 }
90 fn abs(a: Fraction) Fraction {
91 return .{ .n = absolute(a.n), .d = a.d };
92 }
93 fn eq(a: Fraction, b: Fraction) bool {
94 return a.n == b.n and a.d == b.d;
95 }
96 fn less(a: Fraction, b: Fraction) bool {
97 return a.n * b.d < b.n * a.d;
98 }
99 fn power(a: Fraction, exponent: i64) Fraction {
100 const factor = if (exponent < 0) one.div(a) else a;
101 var result = one;
102 var i: i64 = 0;
103 while (i < absolute(exponent)) : (i += 1) result = result.mul(factor);
104 return result;
105 }
106 pub fn format(value: Fraction, out: *Writer) Writer.Error!void {
107 try out.print("{d}", .{value.n});
108 if (value.d != 1) try out.print("/{d}", .{value.d});
109 }

133const Dimension = struct {
134 exponents: [7]i64 = @splat(0),
135 const zero: Dimension = .{};
136 fn eq(a: Dimension, b: Dimension) bool {
137 return std.mem.eql(i64, &a.exponents, &b.exponents);
138 }
139 fn add(a: Dimension, b: Dimension) Dimension {
140 var result: Dimension = .{};
141 for (&result.exponents, a.exponents, b.exponents) |*value, x, y| value.* = x + y;
142 return result;
143 }
144 fn sub(a: Dimension, b: Dimension) Dimension {
145 var result: Dimension = .{};
146 for (&result.exponents, a.exponents, b.exponents) |*value, x, y| value.* = x - y;
147 return result;
148 }
149 fn scale(a: Dimension, k: i64) Dimension {
150 var result: Dimension = .{};
151 for (&result.exponents, a.exponents) |*value, x| value.* = x * k;
152 return result;
153 }
154 fn half(a: Dimension) Dimension {
155 var result: Dimension = .{};
156 for (&result.exponents, a.exponents) |*value, x| value.* = @divTrunc(x, 2);
157 return result;
158 }
159 fn even(a: Dimension) bool {
160 for (a.exponents) |value| if (@mod(value, 2) != 0) return false;
161 return true;
162 }
163 pub fn format(dimension: Dimension, out: *Writer) Writer.Error!void {
164 const symbols = [_][]const u8{ "kg", "m", "s", "A", "K", "mol", "cd" };
165 var first = true;
166 for (dimension.exponents, symbols) |value, symbol| {
167 if (value == 0) continue;
168 if (!first) try out.writeByte(' ');
169 first = false;
170 try out.writeAll(symbol);
171 if (value != 1) try out.print("^{d}", .{value});
172 }
173 if (first) try out.writeByte('1');
174 }

274 fn resolve(calculator: *const Calculator, name: []const u8) ?Quantity {
275 for (calculator.units[0..calculator.unit_count]) |unit| if (equal(unit.symbol, name)) return unit.quantity;
276 for (prefixes) |prefix| {
277 if (!std.mem.startsWith(u8, name, prefix.symbol)) continue;
278 const suffix = name[prefix.symbol.len..];
279 for (builtins) |unit| {
280 if (unit.prefix and equal(unit.symbol, suffix))
281 return .{ .value = prefix.factor.mul(unit.quantity.value), .dimension = unit.quantity.dimension };
282 }
283 }
284 return null;
285 }
286 fn variableIndex(calculator: *const Calculator, name: []const u8) ?usize {
287 for (calculator.variables[0..calculator.variable_count], 0..) |variable, i| if (equal(variable.name, name)) return i;
288 return null;
289 }

291 const a = calculator.units[left];
292 const b = calculator.units[right];
293 return if (a.quantity.value.eq(b.quantity.value)) std.mem.order(u8, a.symbol, b.symbol) == .lt else a.quantity.value.less(b.quantity.value);

317const Parser = struct {
318 session: *Session,
319 source: []const u8,
320 tokens: [201]Token = undefined,
321 length: usize = 0,
322 position: usize = 0,
323
324 fn init(session: *Session, source: []const u8, quantity: bool) Parser {
325 var parser: Parser = .{ .session = session, .source = source };
326 var i: usize = 0;
327 while (i < source.len) {
328 const start = i;
329 const c = source[i];
330 i += 1;
331 var kind: TokenKind = .invalid;
332 if (letter(c)) {
333 while (i < source.len and letter(source[i])) i += 1;
334 kind = .identifier;
335 } else if (digit(c)) {
336 while (i < source.len and digit(source[i])) i += 1;
337 if (quantity and i + 1 < source.len and source[i] == '.' and digit(source[i + 1])) {
338 i += 2;
339 while (i < source.len and digit(source[i])) i += 1;
340 }
341 kind = .number;
342 } else if (quantity and c == '[') {
343 if (std.mem.findScalar(u8, source[i..], ']')) |offset| {
344 if (offset > 0) {
345 i += offset + 1;
346 kind = .unit;
347 }
348 }
349 } else if (std.mem.findScalar(u8, if (quantity) "+-*/^()," else "*/^()-", c) != null) kind = .symbol;
350 parser.tokens[parser.length] = .{ .kind = kind, .text = source[start..i], .position = start + 1 };
351 parser.length += 1;
352 }
353 parser.tokens[parser.length] = .{ .kind = .end, .text = "", .position = source.len + 1 };
354 parser.length += 1;
355 return parser;
356 }
357 fn current(parser: *const Parser) Token {
358 return parser.tokens[parser.position];
359 }
360 fn advance(parser: *Parser) Token {
361 const token = parser.current();
362 parser.position += 1;
363 return token;
364 }
365 fn take(parser: *Parser, text: []const u8) bool {
366 if (!equal(parser.current().text, text)) return false;
367 parser.position += 1;
368 return true;
369 }
370 fn syntax(parser: *Parser) Error {
371 const token = parser.current();
372 return if (token.kind == .end) parser.session.fail("SYNTAX {s} AT END", .{parser.source}) else parser.session.fail("SYNTAX {s} AT {d}", .{ parser.source, token.position });
373 }
374 fn expect(parser: *Parser, text: []const u8) Error!void {
375 if (!parser.take(text)) return parser.syntax();
376 }
377 fn finish(parser: *Parser) Error!void {
378 if (parser.current().kind != .end) return parser.syntax();
379 }
380 fn exponent(parser: *Parser) Error![]const u8 {
381 const start = parser.current().position - 1;
382 _ = parser.take("-");
383 const token = parser.current();
384 if (token.kind != .number or std.mem.findScalar(u8, token.text, '.') != null) return parser.syntax();
385 _ = parser.advance();
386 return parser.source[start .. token.position - 1 + token.text.len];
387 }

398 fn unitExpression(parser: *Parser) Error!*const UnitNode {
399 var terms: std.ArrayList(UnitTerm) = .empty;
400 try terms.append(parser.session.gpa, .{ .op = '*', .node = try parser.unitTerm() });
401 while (equal(parser.current().text, "*") or equal(parser.current().text, "/")) {
402 const op = parser.advance().text[0];
403 try terms.append(parser.session.gpa, .{ .op = op, .node = try parser.unitTerm() });
404 }
405 return parser.unitNode(.{ .product = try terms.toOwnedSlice(parser.session.gpa) });
406 }
407 fn unitTerm(parser: *Parser) Error!*const UnitNode {
408 const token = parser.current();
409 var atom: *const UnitNode = undefined;
410 if (token.kind == .identifier) {
411 _ = parser.advance();
412 atom = try parser.unitNode(.{ .name = token.text });
413 } else if (token.kind == .number and token.text[0] != '0') {
414 _ = parser.advance();
415 atom = try parser.unitNode(.{ .number = integer(token.text) });
416 } else if (parser.take("(")) {
417 atom = try parser.unitExpression();
418 try parser.expect(")");
419 } else return parser.syntax();
420 return if (parser.take("^")) parser.unitNode(.{ .power = .{ .atom = atom, .exponent = try parser.exponent() } }) else atom;
421 }
422 fn quantityExpression(parser: *Parser) Error!*const QuantityNode {
423 var node = try parser.quantityProduct();
424 while (equal(parser.current().text, "+") or equal(parser.current().text, "-")) {
425 const op = parser.advance().text[0];
426 const right = try parser.quantityProduct();
427 node = try parser.quantityNode(.{ .binary = .{ .op = op, .left = node, .right = right } });
428 }
429 return node;
430 }
431 fn quantityProduct(parser: *Parser) Error!*const QuantityNode {
432 var node = try parser.quantityUnary();
433 while (equal(parser.current().text, "*") or equal(parser.current().text, "/")) {
434 const op = parser.advance().text[0];
435 const right = try parser.quantityUnary();
436 node = try parser.quantityNode(.{ .binary = .{ .op = op, .left = node, .right = right } });
437 }
438 return node;
439 }
440 fn quantityUnary(parser: *Parser) Error!*const QuantityNode {
441 if (parser.take("-")) return parser.quantityNode(.{ .negative = try parser.quantityUnary() });
442 const atom = try parser.quantityPrimary();
443 return if (parser.take("^")) parser.quantityNode(.{ .power = .{ .atom = atom, .exponent = try parser.exponent() } }) else atom;
444 }
445 fn quantityPrimary(parser: *Parser) Error!*const QuantityNode {
446 const token = parser.current();
447 if (token.kind == .number) {
448 _ = parser.advance();
449 const unit_text: ?[]const u8 = if (parser.current().kind == .unit) blk: {
450 const unit_token = parser.advance();
451 break :blk unit_token.text[1 .. unit_token.text.len - 1];
452 } else null;
453 return parser.quantityNode(.{ .literal = .{ .number = token.text, .unit = unit_text } });
454 }
455 if (token.kind == .unit) {
456 _ = parser.advance();
457 return parser.quantityNode(.{ .literal = .{ .number = "1", .unit = token.text[1 .. token.text.len - 1] } });
458 }
459 if (token.kind == .identifier) {
460 _ = parser.advance();
461 if (!parser.take("(")) return parser.quantityNode(.{ .variable = token.text });
462 var arguments: std.ArrayList(*const QuantityNode) = .empty;
463 try arguments.append(parser.session.gpa, try parser.quantityExpression());
464 while (parser.take(",")) try arguments.append(parser.session.gpa, try parser.quantityExpression());
465 try parser.expect(")");
466 return parser.quantityNode(.{ .call = .{ .name = token.text, .arguments = try arguments.toOwnedSlice(parser.session.gpa) } });
467 }
468 if (parser.take("(")) {
469 const node = try parser.quantityExpression();
470 try parser.expect(")");
471 return node;
472 }
473 return parser.syntax();
474 }

482fn integerRoot(number: i64) i64 {
483 var low: i64 = 0;
484 var high = number + 1;
485 while (high - low > 1) {
486 const middle = low + @divTrunc(high - low, 2);
487 if (middle <= @divTrunc(number, middle)) low = middle else high = middle;
488 }
489 return low;
490}

492const Session = struct {
493 gpa: Allocator,
494 calculator: *Calculator,
495 out: *Writer,

498 fn fail(session: *Session, comptime fmt: []const u8, args: anytype) Error {
499 session.problem = std.fmt.allocPrint(session.gpa, fmt, args) catch return error.OutOfMemory;
500 return error.Invalid;

502 fn number(session: *Session, text: []const u8) Error!Fraction {
503 const negative = text.len > 0 and text[0] == '-';
504 const body = if (negative) text[1..] else text;
505 var separator: ?usize = null;
506 for (body, 0..) |c, i| {
507 if (c == '.' or c == '/') {
508 separator = i;
509 break;
510 }
511 }
512 const first = if (separator) |i| body[0..i] else body;
513 const second = if (separator) |i| body[i + 1 ..] else "1";
514 if (!digits(first) or !digits(second)) return session.fail("BAD NUMBER {s}", .{text});
515 var n = integer(first);
516 var d: i64 = 1;
517 if (separator) |i| {
518 if (body[i] == '/') {
519 d = integer(second);
520 if (d == 0) return session.fail("BAD NUMBER {s}", .{text});
521 } else {
522 for (second) |_| d *= 10;
523 n = n * d + integer(second);
524 }
525 }
526 return Fraction.init(if (negative) -n else n, d);
527 }
528 fn checkExponent(session: *Session, text: []const u8) Error!i64 {
529 const value = integer(text);

550 fn validateUnit(session: *Session, node: *const UnitNode, names: *std.ArrayList([]const u8)) Error!void {
551 switch (node.*) {
552 .number => {},
553 .name => |name| {
554 if (session.calculator.resolve(name) == null) return session.fail("UNKNOWN UNIT {s}", .{name});
555 for (names.items) |existing| if (equal(name, existing)) return;
556 try names.append(session.gpa, name);
557 },
558 .power => |power| {
559 try session.validateUnit(power.atom, names);
560 _ = try session.checkExponent(power.exponent);
561 },
562 .product => |terms| for (terms) |term| try session.validateUnit(term.node, names),
563 }

565 fn unitValue(session: *Session, node: *const UnitNode) Quantity {
566 switch (node.*) {
567 .number => |value| return .{ .value = Fraction.whole(value), .dimension = Dimension.zero },
568 .name => |name| return session.calculator.resolve(name).?,
569 .power => |power| {
570 const atom = session.unitValue(power.atom);
571 const exponent = integer(power.exponent);
572 return .{ .value = atom.value.power(exponent), .dimension = atom.dimension.scale(exponent) };
573 },
574 .product => |terms| {
575 var result: Quantity = .{ .value = Fraction.one, .dimension = Dimension.zero };
576 for (terms) |term| {
577 const value = session.unitValue(term.node);
578 result = if (term.op == '*') .{ .value = result.value.mul(value.value), .dimension = result.dimension.add(value.dimension) } else .{ .value = result.value.div(value.value), .dimension = result.dimension.sub(value.dimension) };
579 }
580 return result;
581 },
582 }

584 fn parseUnit(session: *Session, text: []const u8) Error!ParsedUnit {
585 var parser = Parser.init(session, text, false);
586 const tree = try parser.unitExpression();
587 try parser.finish();
588 var names: std.ArrayList([]const u8) = .empty;
589 try session.validateUnit(tree, &names);
590 return .{ .tree = tree, .quantity = session.unitValue(tree), .names = names.items };
591 }
592 fn reduce(session: *Session, node: *const UnitNode, names: []const []const u8) Allocator.Error!Reduced {
593 switch (node.*) {
594 .power => |power| {
595 var result = try session.reduce(power.atom, names);
596 const exponent = integer(power.exponent);
597 result.coefficient = result.coefficient.power(exponent);
598 for (result.exponents) |*value| value.* *= exponent;
599 return result;
600 },
601 else => {},
602 }
603 const exponents = try session.gpa.alloc(i64, names.len);
604 @memset(exponents, 0);
605 var result: Reduced = .{ .coefficient = Fraction.one, .exponents = exponents };
606 switch (node.*) {
607 .name => |name| {
608 for (names, 0..) |existing, i| {
609 if (equal(existing, name)) {
610 exponents[i] = 1;
611 break;
612 }
613 }
614 },
615 .number => |value| result.coefficient = Fraction.whole(value),
616 .product => |terms| {
617 for (terms) |term| {
618 const value = try session.reduce(term.node, names);
619 result.coefficient = if (term.op == '*') result.coefficient.mul(value.coefficient) else result.coefficient.div(value.coefficient);
620 for (result.exponents, value.exponents) |*target, source| target.* += if (term.op == '*') source else -source;
621 }
622 },
623 .power => unreachable,
624 }
625 return result;
626 }
627 fn reducedText(session: *Session, parsed: ParsedUnit) Error![]const u8 {
628 const reduced = try session.reduce(parsed.tree, parsed.names);
629 var output: Writer.Allocating = .init(session.gpa);
630 defer output.deinit();
631 var first = true;
632 if (!reduced.coefficient.eq(Fraction.one)) {
633 try output.writer.print("{f}", .{reduced.coefficient});
634 first = false;
635 }
636 for (parsed.names, reduced.exponents) |name, exponent| {
637 if (exponent == 0) continue;
638 if (!first) try output.writer.writeByte(' ');
639 first = false;
640 try output.writer.writeAll(name);
641 if (exponent != 1) try output.writer.print("^{d}", .{exponent});
642 }
643 if (first) try output.writer.writeByte('1');
644 return output.toOwnedSlice();
645 }
646 fn evaluate(session: *Session, node: *const QuantityNode) Error!Quantity {
647 switch (node.*) {
648 .literal => |literal| {
649 if (literal.unit) |text| {
650 const unit = (try session.parseUnit(text)).quantity;
651 return .{ .value = (try session.number(literal.number)).mul(unit.value), .dimension = unit.dimension };
652 }
653 return .{ .value = try session.number(literal.number), .dimension = Dimension.zero };
654 },
655 .variable => |name| {
656 const index = session.calculator.variableIndex(name) orelse return session.fail("UNKNOWN VARIABLE {s}", .{name});
657 return session.calculator.variables[index].quantity;
658 },
659 .negative => |operand| {
660 const value = try session.evaluate(operand);
661 return .{ .value = value.value.neg(), .dimension = value.dimension };
662 },
663 .power => |power| {
664 const value = try session.evaluate(power.atom);
665 const exponent = try session.checkExponent(power.exponent);
666 if (value.value.n == 0 and exponent < 0) return session.fail("DIVISION BY ZERO", .{});
667 return .{ .value = value.value.power(exponent), .dimension = value.dimension.scale(exponent) };
668 },
669 .binary => |binary| {
670 const a = try session.evaluate(binary.left);
671 const b = try session.evaluate(binary.right);
672 if (binary.op == '+' or binary.op == '-') {
673 try session.match(a.dimension, b.dimension);
674 return .{ .value = if (binary.op == '+') a.value.add(b.value) else a.value.sub(b.value), .dimension = a.dimension };
675 }
676 if (binary.op == '*') return .{ .value = a.value.mul(b.value), .dimension = a.dimension.add(b.dimension) };
677 if (b.value.n == 0) return session.fail("DIVISION BY ZERO", .{});
678 return .{ .value = a.value.div(b.value), .dimension = a.dimension.sub(b.dimension) };
679 },
680 .call => |call| {
681 const function = std.meta.stringToEnum(Function, call.name) orelse return session.fail("UNKNOWN FUNCTION {s}", .{call.name});
682 const count: usize = if (function == .abs or function == .sqrt) 1 else 2;
683 if (call.arguments.len != count) return session.fail("BAD ARGUMENTS {s}", .{call.name});
684 const a = try session.evaluate(call.arguments[0]);
685 if (function == .abs) return .{ .value = a.value.abs(), .dimension = a.dimension };
686 if (function == .sqrt) {
687 if (a.value.n < 0) return session.fail("NEGATIVE ROOT", .{});
688 if (!a.dimension.even()) return session.fail("ODD DIMENSION {f}", .{a.dimension});
689 const p = integerRoot(a.value.n);
690 const q = integerRoot(a.value.d);
691 if (p * p != a.value.n or q * q != a.value.d) return session.fail("NOT EXACT", .{});
692 return .{ .value = Fraction.init(p, q), .dimension = a.dimension.half() };
693 }
694 const b = try session.evaluate(call.arguments[1]);
695 try session.match(a.dimension, b.dimension);
696 return if (function == .min) (if (b.value.less(a.value)) b else a) else (if (a.value.less(b.value)) b else a);
697 },
698 }
699 }

702 const tree = try parser.quantityExpression();
703 try parser.finish();

715 fn execute(session: *Session, text: []const u8) Error!void {
716 var words: std.ArrayList([]const u8) = .empty;
717 var tokens = std.mem.splitScalar(u8, text, ' ');
718 while (tokens.next()) |word| try words.append(session.gpa, word);
719 const w = words.items;
720 const command = std.meta.stringToEnum(Command, w[0]) orelse return session.fail("UNKNOWN COMMAND {s}", .{w[0]});
721 const calculator = session.calculator;
722 const out = session.out;
723 switch (command) {
724 .DEFINE => {
725 try session.checkName(w[1], false);
726 const value = try session.number(w[3]);
727 const unit = (try session.parseUnit(w[4])).quantity;
728 if (std.meta.stringToEnum(Kind, w[1]) != null or calculator.resolve(w[1]) != null)
729 return session.fail("NAME TAKEN {s}", .{w[1]});
730 if (value.n <= 0) return session.fail("NOT POSITIVE {s}", .{w[3]});
731 const defined: Quantity = .{ .value = value.mul(unit.value), .dimension = unit.dimension };
732 const name = try calculator.storage.dupe(u8, w[1]);
733 const expression = try calculator.storage.dupe(u8, w[4]);
734 calculator.units[calculator.unit_count] = .{
735 .symbol = name,
736 .quantity = defined,
737 .prefix = false,
738 .definition = .{ .value = value, .expression = expression },
739 };
740 calculator.unit_count += 1;
741 try line(out, "DEFINE {s} factor={f} dim={f} kind={s}", .{ name, defined.value, defined.dimension, kindName(defined.dimension) });
742 },
743 .UNDEF => {
744 var found: ?usize = null;
745 for (calculator.units[0..calculator.unit_count], 0..) |unit, i| {
746 if (equal(unit.symbol, w[1])) {
747 found = i;
748 break;
749 }
750 }
751 const index = found orelse return session.fail("NOT DEFINED {s}", .{w[1]});
752 if (index < builtins.len) return session.fail("BUILTIN UNIT {s}", .{w[1]});
753 for (index + 1..calculator.unit_count) |i| calculator.units[i - 1] = calculator.units[i];
754 calculator.unit_count -= 1;
755 try line(out, "UNDEF {s}", .{w[1]});
756 },
757 .DEFS => {
758 try line(out, "DEFS count={d}", .{calculator.unit_count - builtins.len});
759 for (calculator.units[builtins.len..calculator.unit_count]) |unit| {
760 const definition = unit.definition.?;
761 try line(out, "DEF {s} = {f} {s} factor={f} dim={f}", .{ unit.symbol, definition.value, definition.expression, unit.quantity.value, unit.quantity.dimension });
762 }
763 },
764 .CONVERT, .INVERT => {
765 const value = try session.number(w[1]);
766 const a = (try session.parseUnit(w[2])).quantity;
767 const b = (try session.parseUnit(w[4])).quantity;
768 if (command == .CONVERT) try session.match(a.dimension, b.dimension) else {
769 if (a.dimension.eq(Dimension.zero) or !a.dimension.add(b.dimension).eq(Dimension.zero))
770 return session.fail("NOT RECIPROCAL {f} vs {f}", .{ a.dimension, b.dimension });
771 if (value.n == 0) return session.fail("DIVISION BY ZERO", .{});
772 }
773 const result = if (command == .CONVERT) value.mul(a.value).div(b.value) else Fraction.one.div(value.mul(a.value)).div(b.value);
774 try line(out, "{s} {f} {s} = {f}", .{ w[0], value, w[2], notation(result, w[4]) });
775 },
776 .DIM, .REDUCE, .SIMPLIFY => {
777 const parsed = try session.parseUnit(w[1]);
778 const unit = parsed.quantity;
779 if (command == .DIM) try line(out, "DIM {s} = {f} factor={f} kind={s}", .{ w[1], unit.dimension, unit.value, kindName(unit.dimension) }) else if (command == .REDUCE) try line(out, "REDUCE {s} = {s}", .{ w[1], try session.reducedText(parsed) }) else try line(out, "SIMPLIFY {s} = {f}", .{ w[1], try session.quantityNotation(unit) });
780 },
781 .ADD, .SUB, .MUL, .DIV, .COMPARE => {
782 const x = try session.number(w[1]);
783 const a = (try session.parseUnit(w[2])).quantity;
784 const y = try session.number(w[3]);
785 const b = (try session.parseUnit(w[4])).quantity;
786 const target: ?Quantity = if (w.len == 7) (try session.parseUnit(w[6])).quantity else null;
787 if (command == .COMPARE) {
788 try session.match(a.dimension, b.dimension);
789 const left = x.mul(a.value);
790 const right = y.mul(b.value);
791 const relation: u8 = if (left.less(right)) '<' else if (right.less(left)) '>' else '=';
792 try out.print("COMPARE {f} {s} {c} {f} {s} ratio=", .{ x, w[2], relation, y, w[4] });
793 if (right.n == 0) try line(out, "undefined", .{}) else try line(out, "{f}", .{left.div(right)});
794 return;
795 }
796 const op: u8 = switch (command) {
797 .ADD => '+',
798 .SUB => '-',
799 .MUL => '*',
800 else => '/',
801 };
802 if (command == .ADD or command == .SUB) {
803 try session.match(a.dimension, b.dimension);
804 if (target) |destination| try session.match(a.dimension, destination.dimension);
805 const left = x.mul(a.value);
806 const right = y.mul(b.value);
807 const result = (if (command == .ADD) left.add(right) else left.sub(right)).div(if (target) |destination| destination.value else a.value);
808 try line(out, "{s} {f} {s} {c} {f} {s} = {f}", .{ w[0], x, w[2], op, y, w[4], notation(result, if (target != null) w[6] else w[2]) });
809 return;
810 }
811 const dimension = if (command == .MUL) a.dimension.add(b.dimension) else a.dimension.sub(b.dimension);
812 if (command == .DIV and y.n == 0) return session.fail("DIVISION BY ZERO", .{});
813 if (target) |destination| try session.match(dimension, destination.dimension);
814 const product = if (command == .MUL) x.mul(y) else x.div(y);
815 const first = x.mul(a.value);
816 const second = y.mul(b.value);
817 const coherent = if (command == .MUL) first.mul(second) else first.div(second);
818 try out.print("{s} {f} {s} {c} {f} {s} = ", .{ w[0], x, w[2], op, y, w[4] });
819 if (target) |destination| try line(out, "{f}", .{notation(coherent.div(destination.value), w[6])}) else {
820 try out.print("{f} {s}{c}", .{ product, w[2], op });
821 if (command == .DIV and (std.mem.findScalar(u8, w[4], '*') != null or std.mem.findScalar(u8, w[4], '/') != null))
822 try out.print("({s})", .{w[4]})
823 else
824 try out.writeAll(w[4]);
825 try line(out, " = {f}", .{notation(coherent, try session.simplified(dimension))});
826 }
827 },
828 .TEMP => {
829 const value = try session.number(w[1]);
830 const source = std.meta.stringToEnum(Scale, w[2]) orelse return session.fail("UNKNOWN SCALE {s}", .{w[2]});
831 const target = std.meta.stringToEnum(Scale, w[4]) orelse return session.fail("UNKNOWN SCALE {s}", .{w[4]});
832 const kelvin = switch (source) {
833 .C => value.add(Fraction.init(5463, 20)),
834 .F => value.add(Fraction.init(45967, 100)).mul(Fraction.whole(5)).div(Fraction.whole(9)),
835 .R => value.mul(Fraction.whole(5)).div(Fraction.whole(9)),
836 .K => value,
837 };
838 if (kelvin.n < 0) return session.fail("BELOW ABSOLUTE ZERO", .{});
839 const result = switch (target) {
840 .C => kelvin.sub(Fraction.init(5463, 20)),
841 .F => kelvin.mul(Fraction.whole(9)).div(Fraction.whole(5)).sub(Fraction.init(45967, 100)),
842 .R => kelvin.mul(Fraction.whole(9)).div(Fraction.whole(5)),
843 .K => kelvin,
844 };
845 try line(out, "TEMP {f} {s} = {f}", .{ value, w[2], notation(result, w[4]) });
846 },
847 .LIST => {
848 const dimension = if (std.meta.stringToEnum(Kind, w[1])) |kind| kind_dimensions[@backingInt(kind)] else (try session.parseUnit(w[1])).quantity.dimension;
849 const selected = try session.matching(dimension);
850 try line(out, "LIST {s} dim={f} kind={s} count={d}", .{ w[1], dimension, kindName(dimension), selected.len });
851 for (selected) |i| {
852 const unit = calculator.units[i];
853 try line(out, "UNIT {s} factor={f} origin={s} prefix={s}", .{
854 unit.symbol, unit.quantity.value, @as([]const u8, if (unit.definition == null) "builtin" else "user"),
855 @as([]const u8, if (unit.prefix) "yes" else "no"),
856 });
857 }
858 },
859 .TABLE, .AUTO, .SPLIT => {
860 const value = try session.number(w[1]);
861 const unit = (try session.parseUnit(w[2])).quantity;
862 if (command == .TABLE) {
863 const selected = try session.matching(unit.dimension);
864 try line(out, "TABLE {f} {s} dim={f} count={d}", .{ value, w[2], unit.dimension, selected.len });
865 for (selected) |i| {
866 const row = calculator.units[i];
867 try line(out, "ROW {f}", .{notation(value.mul(unit.value).div(row.quantity.value), row.symbol)});
868 }
869 return;
870 }
871 if (command == .AUTO) {
872 var basis: ?Unit = null;
873 for (builtins) |candidate| {
874 if (candidate.prefix and candidate.quantity.dimension.eq(unit.dimension)) {
875 basis = candidate;
876 break;
877 }
878 }
879 const reference = basis orelse return session.fail("NO PREFIXABLE UNIT {f}", .{unit.dimension});
880 const scaled = value.mul(unit.value).div(reference.quantity.value);
881 const magnitude = scaled.abs();
882 var exponent: i64 = 0;
883 if (!magnitude.less(Fraction.one)) {
884 while (exponent < 9 and !magnitude.less(Fraction.whole(10).power(exponent + 3))) exponent += 3;
885 } else if (magnitude.n != 0) {
886 while (exponent > -9 and magnitude.less(Fraction.whole(10).power(exponent))) exponent -= 3;
887 }
888 const prefix: []const u8 = switch (exponent) {
889 9 => "G",
890 6 => "M",
891 3 => "k",
892 -3 => "m",
893 -6 => "u",
894 -9 => "n",
895 else => "",
896 };
897 const symbol = try std.fmt.allocPrint(session.gpa, "{s}{s}", .{ prefix, reference.symbol });
898 try line(out, "AUTO {f} {s} = {f}", .{ value, w[2], notation(scaled.div(Fraction.whole(10).power(exponent)), symbol) });
899 return;
900 }
901 var parts: std.ArrayList([]const u8) = .empty;
902 var destinations: std.ArrayList(Quantity) = .empty;
903 var part_tokens = std.mem.splitScalar(u8, w[4], ',');
904 while (part_tokens.next()) |part| {
905 try parts.append(session.gpa, part);
906 try destinations.append(session.gpa, (try session.parseUnit(part)).quantity);
907 }
908 if (value.n < 0) return session.fail("NEGATIVE VALUE", .{});
909 for (destinations.items) |destination| try session.match(unit.dimension, destination.dimension);
910 for (1..destinations.items.len) |i|
911 if (!destinations.items[i].value.less(destinations.items[i - 1].value)) return session.fail("UNIT ORDER", .{});
912 var remaining = value.mul(unit.value);
913 try out.print("SPLIT {f} {s} = ", .{ value, w[2] });
914 for (0..parts.items.len - 1) |i| {
915 const quotient = remaining.div(destinations.items[i].value);
916 const whole = @divTrunc(quotient.n, quotient.d);
917 remaining = remaining.sub(Fraction.whole(whole).mul(destinations.items[i].value));
918 try out.print("{d} {s} ", .{ whole, parts.items[i] });
919 }
920 const last = parts.items.len - 1;
921 try line(out, "{f}", .{notation(remaining.div(destinations.items[last].value), parts.items[last])});
922 },
923 .LET => {
924 try session.checkName(w[1], true);
925 const value = try session.parseQuantity(w[3]);
926 if (calculator.variableIndex(w[1])) |index| calculator.variables[index].quantity = value else {
927 const name = try calculator.storage.dupe(u8, w[1]);
928 calculator.variables[calculator.variable_count] = .{ .name = name, .quantity = value };
929 calculator.variable_count += 1;
930 }
931 try line(out, "LET {s} = {f}", .{ w[1], try session.quantityNotation(value) });
932 },
933 .EVAL => {
934 const value = try session.parseQuantity(w[1]);
935 if (w.len == 4) {
936 const unit = (try session.parseUnit(w[3])).quantity;
937 try session.match(value.dimension, unit.dimension);
938 try line(out, "EVAL {s} = {f}", .{ w[1], notation(value.value.div(unit.value), w[3]) });
939 } else try line(out, "EVAL {s} = {f}", .{ w[1], try session.quantityNotation(value) });
940 },
941 .VARS => {
942 var indices: [50]usize = undefined;
943 for (indices[0..calculator.variable_count], 0..) |*index, i| index.* = i;
944 std.sort.heap(usize, indices[0..calculator.variable_count], @as(*const Calculator, calculator), Calculator.variableLess);
945 try line(out, "VARS count={d}", .{calculator.variable_count});
946 for (indices[0..calculator.variable_count]) |index| {
947 const variable = calculator.variables[index];
948 try line(out, "VAR {s} = {f}", .{ variable.name, try session.quantityNotation(variable.quantity) });
949 }
950 },
951 }
952 }

955fn readLine(gpa: Allocator, input: *std.Io.Reader) ![]const u8 {
956 var buffer: Writer.Allocating = .init(gpa);
957 defer buffer.deinit();
958 _ = try input.streamDelimiterEnding(&buffer.writer, '\n');
959 if (input.seek < input.end) input.toss(1) else if (buffer.written().len == 0) return error.EndOfStream;
960 const text = try buffer.toOwnedSlice();
961 return if (text.len > 0 and text[text.len - 1] == '\r') text[0 .. text.len - 1] else text;
962}
963pub fn main(init: std.process.Init) !void {
964 var storage = std.heap.ArenaAllocator.init(std.heap.page_allocator);
965 defer storage.deinit();
966 var calculator = Calculator.init(storage.allocator());
967 var input_buffer: [4096]u8 = undefined;
968 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
969 var output_buffer: [4096]u8 = undefined;
970 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
971 const count: usize = @intCast(integer(try readLine(storage.allocator(), &input.interface)));
972 for (0..count) |_| {
973 var scratch = std.heap.ArenaAllocator.init(std.heap.page_allocator);
974 defer scratch.deinit();
975 const gpa = scratch.allocator();
976 const text = try readLine(gpa, &input.interface);
977 var response: Writer.Allocating = .init(gpa);
978 defer response.deinit();
979 var session: Session = .{ .gpa = gpa, .calculator = &calculator, .out = &response.writer };
980 session.execute(text) catch |err| {
