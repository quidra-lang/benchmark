19struct Fraction {
20 Integer n, d;
21
22 Fraction(Integer numerator = 0, Integer denominator = 1) : n(numerator), d(denominator) {
23 if (d < 0) { n = -n; d = -d; }
24 const auto divisor = std::gcd(n, d);
25 n /= divisor;
26 d /= divisor;
27 }
28 Fraction operator+(Fraction b) const { return {n * b.d + b.n * d, d * b.d}; }
29 Fraction operator-(Fraction b) const { return {n * b.d - b.n * d, d * b.d}; }
30 Fraction operator*(Fraction b) const { return {n * b.n, d * b.d}; }
31 Fraction operator/(Fraction b) const { return {n * b.d, d * b.n}; }
32 Fraction operator-() const { return {-n, d}; }
33 Fraction absolute() const { return n < 0 ? -*this : *this; }
34 int compare(Fraction b) const {
35 const auto left = n * b.d, right = b.n * d;
36 return left < right ? -1 : left > right ? 1 : 0;
37 }
38 Fraction power(Integer exponent) const {
39 const Fraction factor = exponent < 0 ? Fraction(1) / *this : *this;
40 Fraction result(1);
41 if (exponent < 0) exponent = -exponent;
42 for (Integer i = 0; i < exponent; ++i) result = result * factor;
43 return result;
44 }
45 std::string str() const {
46 return std::to_string(n) + (d == 1 ? "" : "/" + std::to_string(d));
47 }
48 std::string decimal() const {
49 const auto magnitude = n < 0 ? -n : n;
50 const auto whole = magnitude / d, tail = magnitude % d * 1000000 / d;
51 const auto digits = std::to_string(tail);
52 return std::string(n < 0 ? "-" : "") + std::to_string(whole) + "." + std::string(6 - digits.size(), '0') + digits;
53 }
54 std::string notation(const std::string& unit) const { return str() + " " + unit + " (" + decimal() + ")"; }
55};

57std::vector<std::string> split(const std::string& text, char separator) {
58 std::vector<std::string> result;
59 std::size_t start = 0;
60 for (;;) {
61 const auto end = text.find(separator, start);
62 result.push_back(text.substr(start, end == std::string::npos ? end : end - start));
63 if (end == std::string::npos) return result;
64 start = end + 1;
65 }
66}
67std::string join(const std::vector<std::string>& values, const std::string& separator = " ") {
68 std::string result;
69 for (std::size_t i = 0; i < values.size(); ++i) {
70 if (i) result += separator;
71 result += values[i];
72 }
73 return result;
74}
75Fraction numberValue(const std::string& text) {
76 static const std::regex pattern(R"(-?[0-9]+([./][0-9]+)?)");
77 if (!std::regex_match(text, pattern)) fail("BAD NUMBER " + text);
78 const bool negative = text.front() == '-';
79 const auto body = negative ? text.substr(1) : text;
80 Fraction result;
81 if (const auto slash = body.find('/'); slash != std::string::npos) {
82 const Integer denominator = std::stoll(body.substr(slash + 1));
83 if (denominator == 0) fail("BAD NUMBER " + text);
84 result = Fraction(std::stoll(body.substr(0, slash)), denominator);
85 } else if (const auto dot = body.find('.'); dot != std::string::npos) {
86 Integer scale = 1;
87 for (std::size_t i = dot + 1; i < body.size(); ++i) scale *= 10;
88 result = Fraction(std::stoll(body.substr(0, dot)) * scale + std::stoll(body.substr(dot + 1)), scale);
89 } else result = Fraction(std::stoll(body));
90 return negative ? -result : result;
91}
92Integer checkedExponent(const std::string& text) {
93 const Integer value = std::stoll(text);
94 if (value == 0 || value < -3 || value > 3) fail("BAD EXPONENT " + text);
95 return value;
96}

103struct Dimension {
104 std::array<Integer, 7> exponents{};
105
106 Dimension operator+(const Dimension& other) const {
107 Dimension result;
108 for (std::size_t i = 0; i < 7; ++i) result.exponents[i] = exponents[i] + other.exponents[i];
109 return result;
110 }
111 Dimension operator-(const Dimension& other) const {
112 Dimension result;
113 for (std::size_t i = 0; i < 7; ++i) result.exponents[i] = exponents[i] - other.exponents[i];
114 return result;
115 }
116 Dimension scaled(Integer factor) const {
117 Dimension result;
118 for (std::size_t i = 0; i < 7; ++i) result.exponents[i] = exponents[i] * factor;
119 return result;
120 }
121 bool operator==(const Dimension& other) const { return exponents == other.exponents; }
122 void match(const Dimension& other) const {
123 if (!(*this == other)) fail("DIMENSION MISMATCH " + str() + " vs " + other.str());
124 }
125 std::string str() const {
126 const std::array<std::string, 7> symbols{"kg", "m", "s", "A", "K", "mol", "cd"};
127 std::vector<std::string> parts;
128 for (std::size_t i = 0; i < 7; ++i) if (exponents[i] != 0) {
129 parts.push_back(symbols[i] + (exponents[i] == 1 ? "" : "^" + std::to_string(exponents[i])));
130 }
131 return parts.empty() ? "1" : join(parts);
132 }

189 std::vector<Unit> units;

192 Database() {
193 for (const auto& row : BUILTINS) units.push_back({row.symbol, {row.factor, *kindDimension(row.kind)}, row.prefixable, std::nullopt});
194 }
195 std::optional<Quantity> resolve(const std::string& name) const {
196 for (const auto& unit : units) if (unit.symbol == name) return unit.quantity;
197 for (const auto& [prefix, factor] : PREFIXES) {
198 if (name.compare(0, prefix.size(), prefix) != 0) continue;
199 const auto suffix = name.substr(prefix.size());
200 for (std::size_t i = 0; i < BUILTINS.size(); ++i) {
201 const auto& unit = units[i];
202 if (unit.prefixable && unit.symbol == suffix) return Quantity{factor * unit.quantity.value, unit.quantity.dimension};
203 }
204 }
205 return std::nullopt;
206 }
207 std::string simplified(const Dimension& dimension) const {
208 for (const auto& unit : units) if (unit.quantity.value.compare(1) == 0 && unit.quantity.dimension == dimension) return unit.symbol;
209 return dimension.str();
210 }
211 std::vector<Unit> selected(const Dimension& dimension) const {
212 std::vector<Unit> result;
213 for (const auto& unit : units) if (unit.quantity.dimension == dimension) result.push_back(unit);
214 std::sort(result.begin(), result.end(), [](const Unit& a, const Unit& b) {
215 const auto comparison = a.quantity.value.compare(b.quantity.value);
216 return comparison ? comparison < 0 : a.symbol < b.symbol;
217 });
218 return result;
219 }

239class Parser {
240 std::string source;
241 std::vector<Token> tokens;
242 std::size_t position = 0;
243
244 const Token& current() const { return tokens[position]; }
245 Token advance() { return tokens[position++]; }
246 [[noreturn]] void syntax() const {
247 fail("SYNTAX " + source + " AT " + (current().kind == TokenKind::End ? "END" : std::to_string(current().position)));
248 }
249 bool take(const std::string& text) {
250 if (current().text != text) return false;
251 ++position;
252 return true;
253 }
254 void expect(const std::string& text) { if (!take(text)) syntax(); }
255 std::string exponent() {
256 const std::string sign = take("-") ? "-" : "";
257 if (current().kind != TokenKind::Number || current().text.find('.') != std::string::npos) syntax();
258 return sign + advance().text;
259 }
260 UnitExpr unitTerm() {
261 UnitExpr node;
262 if (current().kind == TokenKind::Name) node = {UnitExpr::Kind::Name, advance().text, {}, {}};
263 else if (current().kind == TokenKind::Number && current().text.front() != '0') node = {UnitExpr::Kind::Number, advance().text, {}, {}};
264 else if (take("(")) { node = unitExpression(); expect(")"); }
265 else syntax();
266 if (take("^")) {
267 auto power = exponent();
268 node = {UnitExpr::Kind::Power, power, {std::move(node)}, {}};
269 }
270 return node;
271 }
272 UnitExpr unitExpression() {
273 UnitExpr node{UnitExpr::Kind::Product, "", {unitTerm()}, {'*'}};
274 while (current().text == "*" || current().text == "/") {
275 node.operations.push_back(advance().text.front());
276 node.children.push_back(unitTerm());
277 }
278 return node;
279 }
280 QuantityExpr primary() {
281 const auto token = current();
282 QuantityExpr node;
283 if (token.kind == TokenKind::Number) {
284 advance();
285 const auto unit = current().kind == TokenKind::Unit ? advance().text : "";
286 node = {QuantityExpr::Kind::Literal, token.text, unit.empty() ? "" : unit.substr(1, unit.size() - 2), 0, {}};
287 } else if (token.kind == TokenKind::Unit) {
288 advance();
289 node = {QuantityExpr::Kind::Literal, "1", token.text.substr(1, token.text.size() - 2), 0, {}};
290 } else if (token.kind == TokenKind::Name) {
291 advance();
292 if (take("(")) {
293 node = {QuantityExpr::Kind::Call, token.text, "", 0, {expression()}};
294 while (take(",")) node.children.push_back(expression());
295 expect(")");
296 } else node = {QuantityExpr::Kind::Variable, token.text, "", 0, {}};
297 } else if (take("(")) { node = expression(); expect(")"); }
298 else syntax();
299 if (take("^")) {
300 auto power = exponent();
301 node = {QuantityExpr::Kind::Power, power, "", 0, {std::move(node)}};
302 }
303 return node;
304 }
305 QuantityExpr unary() {
306 if (take("-")) return {QuantityExpr::Kind::Negative, "", "", 0, {unary()}};
307 return primary();
308 }
309 QuantityExpr product() {
310 auto node = unary();
311 while (current().text == "*" || current().text == "/") {
312 const auto operation = advance().text.front();
313 auto right = unary();
314 node = {QuantityExpr::Kind::Binary, "", "", operation, {std::move(node), std::move(right)}};
315 }
316 return node;
317 }
318 QuantityExpr expression() {
319 auto node = product();
320 while (current().text == "+" || current().text == "-") {
321 const auto operation = advance().text.front();
322 auto right = product();
323 node = {QuantityExpr::Kind::Binary, "", "", operation, {std::move(node), std::move(right)}};
324 }
325 return node;
326 }
327public:
328 Parser(std::string text, bool quantityMode) : source(std::move(text)) {
329 for (std::size_t i = 0; i < source.size();) {
330 const auto start = i;
331 const auto character = source[i++];
332 auto kind = TokenKind::Invalid;
333 if (letter(character)) {
334 kind = TokenKind::Name;
335 while (i < source.size() && letter(source[i])) ++i;
336 } else if (digit(character)) {
337 kind = TokenKind::Number;
338 while (i < source.size() && digit(source[i])) ++i;
339 if (quantityMode && i + 1 < source.size() && source[i] == '.' && digit(source[i + 1])) {
340 ++i;
341 while (i < source.size() && digit(source[i])) ++i;
342 }
343 } else if (quantityMode && character == '[') {
344 const auto closing = source.find(']', i);
345 if (closing != std::string::npos && closing > i) { kind = TokenKind::Unit; i = closing + 1; }
346 } else if (std::string(quantityMode ? "+-*/^()," : "*/^()-").find(character) != std::string::npos) kind = TokenKind::Symbol;
347 tokens.push_back({kind, source.substr(start, i - start), start + 1});
348 }
349 tokens.push_back({TokenKind::End, "", source.size() + 1});
350 }
351 UnitExpr unit() {
352 auto node = unitExpression();
353 if (current().kind != TokenKind::End) syntax();
354 return node;
355 }
356 QuantityExpr quantity() {
357 auto node = expression();
358 if (current().kind != TokenKind::End) syntax();
359 return node;
360 }
361};

363void validateUnit(const Database& database, const UnitExpr& node) {
364 if (node.kind == UnitExpr::Kind::Name) {
365 if (!database.resolve(node.text)) fail("UNKNOWN UNIT " + node.text);
366 } else {
367 for (const auto& child : node.children) validateUnit(database, child);
368 if (node.kind == UnitExpr::Kind::Power) checkedExponent(node.text);
369 }
370}
371Quantity evaluateUnit(const Database& database, const UnitExpr& node) {
372 if (node.kind == UnitExpr::Kind::Name) return *database.resolve(node.text);
373 if (node.kind == UnitExpr::Kind::Number) return {Fraction(std::stoll(node.text)), {}};
374 if (node.kind == UnitExpr::Kind::Power) {
375 const auto value = evaluateUnit(database, node.children.front());
376 const Integer exponent = std::stoll(node.text);
377 return {value.value.power(exponent), value.dimension.scaled(exponent)};
378 }
379 Quantity result;
380 for (std::size_t i = 0; i < node.children.size(); ++i) {
381 const auto right = evaluateUnit(database, node.children[i]);
382 result = node.operations[i] == '*' ? Quantity{result.value * right.value, result.dimension + right.dimension} :
383 Quantity{result.value / right.value, result.dimension - right.dimension};
384 }
385 return result;
386}
387UnitExpr parseUnit(const Database& database, const std::string& text) {
388 auto tree = Parser(text, false).unit();
389 validateUnit(database, tree);
390 return tree;
391}

394struct Reduced { Fraction coefficient{1}; std::vector<std::pair<std::string, Integer>> powers; };
395Reduced reduceUnit(const UnitExpr& node) {
396 if (node.kind == UnitExpr::Kind::Name) return {1, {{node.text, 1}}};
397 if (node.kind == UnitExpr::Kind::Number) return {Fraction(std::stoll(node.text)), {}};
398 if (node.kind == UnitExpr::Kind::Power) {
399 auto result = reduceUnit(node.children.front());
400 const Integer exponent = std::stoll(node.text);
401 result.coefficient = result.coefficient.power(exponent);
402 for (auto& item : result.powers) item.second *= exponent;
403 return result;
404 }
405 Reduced result;
406 for (std::size_t i = 0; i < node.children.size(); ++i) {
407 const auto right = reduceUnit(node.children[i]);
408 const bool multiply = node.operations[i] == '*';
409 result.coefficient = multiply ? result.coefficient * right.coefficient : result.coefficient / right.coefficient;
410 for (const auto& [name, power] : right.powers) {
411 auto found = std::find_if(result.powers.begin(), result.powers.end(), [&](const auto& item) { return item.first == name; });
412 const auto amount = multiply ? power : -power;
413 if (found == result.powers.end()) result.powers.emplace_back(name, amount);
414 else found->second += amount;
415 }
416 }
417 return result;
418}
419std::string reducedText(const UnitExpr& tree) {
420 const auto result = reduceUnit(tree);
421 std::vector<std::string> parts;
422 if (result.coefficient.compare(1) != 0) parts.push_back(result.coefficient.str());
423 for (const auto& [name, power] : result.powers) if (power != 0) {
424 parts.push_back(name + (power == 1 ? "" : "^" + std::to_string(power)));
425 }
426 return parts.empty() ? "1" : join(parts);
427}
428Integer integerSqrt(Integer value) {
429 if (value < 2) return value;
430 Integer current = value, next = (value + 1) / 2;
431 while (next < current) {
432 current = next;
433 next = (current + value / current) / 2;
434 }
435 return current;
436}
437Quantity evaluateQuantity(const Database& database, const QuantityExpr& node) {
438 if (node.kind == QuantityExpr::Kind::Literal) {
439 const auto unit = node.unit.empty() ? Quantity{} : unitValue(database, node.unit);
440 return {numberValue(node.text) * unit.value, unit.dimension};
441 }
442 if (node.kind == QuantityExpr::Kind::Variable) {
443 const auto found = database.variables.find(node.text);
444 if (found == database.variables.end()) fail("UNKNOWN VARIABLE " + node.text);
445 return found->second;
446 }
447 if (node.kind == QuantityExpr::Kind::Call) {
448 const auto function = FUNCTIONS.find(node.text);
449 if (function == FUNCTIONS.end()) fail("UNKNOWN FUNCTION " + node.text);
450 if (node.children.size() != function->second) fail("BAD ARGUMENTS " + node.text);
451 const auto first = evaluateQuantity(database, node.children.front());
452 if (node.text == "abs") return {first.value.absolute(), first.dimension};
453 if (node.text == "sqrt") {
454 if (first.value.n < 0) fail("NEGATIVE ROOT");
455 for (const auto exponent : first.dimension.exponents) if (exponent % 2 != 0) fail("ODD DIMENSION " + first.dimension.str());
456 const auto n = integerSqrt(first.value.n), d = integerSqrt(first.value.d);
457 if (n * n != first.value.n || d * d != first.value.d) fail("NOT EXACT");
458 auto dimension = first.dimension;
459 for (auto& exponent : dimension.exponents) exponent /= 2;
460 return {{n, d}, dimension};
461 }
462 const auto second = evaluateQuantity(database, node.children[1]);
463 first.dimension.match(second.dimension);
464 const auto comparison = first.value.compare(second.value);
465 return (node.text == "min" ? comparison <= 0 : comparison >= 0) ? first : second;
466 }
467 const auto left = evaluateQuantity(database, node.children.front());
468 if (node.kind == QuantityExpr::Kind::Negative) return {-left.value, left.dimension};
469 if (node.kind == QuantityExpr::Kind::Power) {
470 const auto exponent = checkedExponent(node.text);
471 if (left.value.n == 0 && exponent < 0) fail("DIVISION BY ZERO");
472 return {left.value.power(exponent), left.dimension.scaled(exponent)};
473 }
474 const auto right = evaluateQuantity(database, node.children[1]);
475 if (node.operation == '+' || node.operation == '-') {
476 left.dimension.match(right.dimension);
477 return {node.operation == '+' ? left.value + right.value : left.value - right.value, left.dimension};
478 }
479 if (node.operation == '*') return {left.value * right.value, left.dimension + right.dimension};
480 if (right.value.n == 0) fail("DIVISION BY ZERO");
481 return {left.value / right.value, left.dimension - right.dimension};
482}
483Quantity quantityValue(const Database& database, const std::string& text) {
484 return evaluateQuantity(database, Parser(text, true).quantity());
485}

487std::vector<std::string> execute(Database& database, const std::string& line) {
488 const auto words = split(line, ' ');
489 const auto& command = words[0];
490 if (command == "DEFINE") {
491 const auto& name = words[1];
492 checkName(name);
493 const auto value = numberValue(words[3]);
494 const auto unit = unitValue(database, words[4]);
495 if (kindDimension(name) || database.resolve(name)) fail("NAME TAKEN " + name);
496 if (value.n <= 0) fail("NOT POSITIVE " + words[3]);
497 const Quantity result{value * unit.value, unit.dimension};
498 database.units.push_back({name, result, false, Definition{value, words[4]}});
499 return {"DEFINE " + name + " factor=" + result.value.str() + " dim=" + result.dimension.str() + " kind=" + kindName(result.dimension)};
500 }
501 if (command == "UNDEF") {
502 const auto found = std::find_if(database.units.begin(), database.units.end(), [&](const Unit& unit) { return unit.symbol == words[1]; });
503 if (found == database.units.end()) fail("NOT DEFINED " + words[1]);
504 if (std::size_t(found - database.units.begin()) < BUILTINS.size()) fail("BUILTIN UNIT " + words[1]);
505 database.units.erase(found);
506 return {"UNDEF " + words[1]};
507 }
508 if (command == "DEFS") {
509 std::vector<std::string> rows{"DEFS count=" + std::to_string(database.units.size() - BUILTINS.size())};
510 for (std::size_t i = BUILTINS.size(); i < database.units.size(); ++i) {
511 const auto& unit = database.units[i];
512 rows.push_back("DEF " + unit.symbol + " = " + unit.definition->value.str() + " " + unit.definition->expression +
513 " factor=" + unit.quantity.value.str() + " dim=" + unit.quantity.dimension.str());
514 }
515 return rows;
516 }
517 if (command == "CONVERT" || command == "INVERT") {
518 const auto value = numberValue(words[1]);
519 const auto source = unitValue(database, words[2]);
520 const auto target = unitValue(database, words[4]);
521 if (command == "INVERT") {
522 if (source.dimension == Dimension{} || !(source.dimension + target.dimension == Dimension{})) {
523 fail("NOT RECIPROCAL " + source.dimension.str() + " vs " + target.dimension.str());
524 }
525 if (value.n == 0) fail("DIVISION BY ZERO");
526 } else source.dimension.match(target.dimension);
527 auto result = value * source.value;
528 if (command == "INVERT") result = Fraction(1) / result;
529 return {command + " " + value.str() + " " + words[2] + " = " + (result / target.value).notation(words[4])};
530 }
531 if (command == "DIM" || command == "REDUCE" || command == "SIMPLIFY") {
532 const auto tree = parseUnit(database, words[1]);
533 const auto unit = evaluateUnit(database, tree);
534 const auto result = command == "DIM" ? unit.dimension.str() + " factor=" + unit.value.str() + " kind=" + kindName(unit.dimension) :
535 command == "REDUCE" ? reducedText(tree) : unit.value.notation(database.simplified(unit.dimension));
536 return {command + " " + words[1] + " = " + result};
537 }
538 if (command == "ADD" || command == "SUB" || command == "MUL" || command == "DIV" || command == "COMPARE") {
539 const auto a = numberValue(words[1]);
540 const auto first = unitValue(database, words[2]);
541 const auto b = numberValue(words[3]);
542 const auto second = unitValue(database, words[4]);
543 const bool hasTarget = words.size() == 7;
544 const auto target = hasTarget ? unitValue(database, words[6]) : first;
545 const auto prefix = command + " " + a.str() + " " + words[2];
546 const auto suffix = b.str() + " " + words[4];
547 if (command == "ADD" || command == "SUB" || command == "COMPARE") {
548 first.dimension.match(second.dimension);
549 if (hasTarget) first.dimension.match(target.dimension);
550 const auto left = a * first.value, right = b * second.value;
551 if (command == "COMPARE") {
552 const auto comparison = left.compare(right);
553 const std::string relation = comparison < 0 ? "<" : comparison > 0 ? ">" : "=";
554 return {prefix + " " + relation + " " + suffix + " ratio=" + (right.n == 0 ? "undefined" : (left / right).str())};
555 }
556 const auto result = command == "ADD" ? left + right : left - right;
557 return {prefix + (command == "ADD" ? " + " : " - ") + suffix + " = " +
558 (result / target.value).notation(hasTarget ? words[6] : words[2])};
559 }
560 const bool multiply = command == "MUL";
561 const auto dimension = multiply ? first.dimension + second.dimension : first.dimension - second.dimension;
562 if (!multiply && b.n == 0) fail("DIVISION BY ZERO");
563 if (hasTarget) dimension.match(target.dimension);
564 const auto raw = multiply ? a * b : a / b;
565 const auto left = a * first.value, right = b * second.value;
566 const auto coherent = multiply ? left * right : left / right;
567 const std::string operation = multiply ? "*" : "/";
568 const auto heading = prefix + " " + operation + " " + suffix + " = ";
569 if (hasTarget) return {heading + (coherent / target.value).notation(words[6])};
570 const auto rightUnit = !multiply && words[4].find_first_of("*/") != std::string::npos ? "(" + words[4] + ")" : words[4];
571 return {heading + raw.str() + " " + words[2] + operation + rightUnit + " = " + coherent.notation(database.simplified(dimension))};
572 }
573 if (command == "TEMP") {
574 const auto value = numberValue(words[1]);
575 for (const auto index : {2, 4}) if (words[index] != "C" && words[index] != "F" && words[index] != "K" && words[index] != "R") {
576 fail("UNKNOWN SCALE " + words[index]);
577 }
578 const Fraction celsius(5463, 20), fahrenheit(45967, 100);
579 const auto kelvin = words[2] == "C" ? value + celsius : words[2] == "F" ? (value + fahrenheit) * Fraction(5, 9) :
580 words[2] == "R" ? value * Fraction(5, 9) : value;
581 if (kelvin.n < 0) fail("BELOW ABSOLUTE ZERO");
582 const auto result = words[4] == "C" ? kelvin - celsius : words[4] == "F" ? kelvin * Fraction(9, 5) - fahrenheit :
583 words[4] == "R" ? kelvin * Fraction(9, 5) : kelvin;
584 return {"TEMP " + value.str() + " " + words[2] + " = " + result.notation(words[4])};
585 }
586 if (command == "LIST") {
587 const auto known = kindDimension(words[1]);
588 const auto dimension = known ? *known : unitValue(database, words[1]).dimension;
589 const auto units = database.selected(dimension);
590 std::vector<std::string> rows{"LIST " + words[1] + " dim=" + dimension.str() + " kind=" + kindName(dimension) + " count=" + std::to_string(units.size())};
591 for (const auto& unit : units) rows.push_back("UNIT " + unit.symbol + " factor=" + unit.quantity.value.str() +
592 " origin=" + (unit.definition ? "user" : "builtin") + " prefix=" + (unit.prefixable ? "yes" : "no"));
593 return rows;
594 }
595 if (command == "TABLE") {
596 const auto value = numberValue(words[1]);
597 const auto unit = unitValue(database, words[2]);
598 const auto selected = database.selected(unit.dimension);
599 std::vector<std::string> rows{"TABLE " + value.str() + " " + words[2] + " dim=" + unit.dimension.str() + " count=" + std::to_string(selected.size())};
600 for (const auto& item : selected) rows.push_back("ROW " + (value * unit.value / item.quantity.value).notation(item.symbol));
601 return rows;
602 }
603 if (command == "AUTO") {
604 const auto value = numberValue(words[1]);
605 const auto unit = unitValue(database, words[2]);
606 std::size_t index = 0;
607 while (index < BUILTINS.size() && !(database.units[index].prefixable && database.units[index].quantity.dimension == unit.dimension)) ++index;
608 if (index == BUILTINS.size()) fail("NO PREFIXABLE UNIT " + unit.dimension.str());
609 const auto& reference = database.units[index];
610 const auto scaled = value * unit.value / reference.quantity.value;
611 const auto magnitude = scaled.absolute();
612 Integer exponent = 0;
613 if (magnitude.compare(1) >= 0) {
614 while (exponent < 9 && magnitude.compare(Fraction(10).power(exponent + 3)) >= 0) exponent += 3;
615 } else if (magnitude.n > 0) {
616 while (exponent > -9 && magnitude.compare(Fraction(10).power(exponent)) < 0) exponent -= 3;
617 }
618 const std::array<std::string, 7> prefixes{"n", "u", "m", "", "k", "M", "G"};
619 return {"AUTO " + value.str() + " " + words[2] + " = " + (scaled / Fraction(10).power(exponent)).notation(prefixes[(exponent + 9) / 3] + reference.symbol)};
620 }
621 if (command == "SPLIT") {
622 const auto value = numberValue(words[1]);
623 const auto unit = unitValue(database, words[2]);
624 const auto parts = split(words[4], ',');
625 std::vector<Quantity> targets;
626 for (const auto& part : parts) targets.push_back(unitValue(database, part));
627 if (value.n < 0) fail("NEGATIVE VALUE");
628 for (const auto& target : targets) unit.dimension.match(target.dimension);
629 for (std::size_t i = 0; i + 1 < targets.size(); ++i) if (targets[i].value.compare(targets[i + 1].value) <= 0) fail("UNIT ORDER");
630 auto remaining = value * unit.value;
631 std::vector<std::string> pieces;
632 for (std::size_t i = 0; i + 1 < targets.size(); ++i) {
633 const auto quotient = remaining / targets[i].value;
634 const auto whole = quotient.n / quotient.d;
635 remaining = remaining - Fraction(whole) * targets[i].value;
636 pieces.push_back(std::to_string(whole) + " " + parts[i]);
637 }
638 pieces.push_back((remaining / targets.back().value).notation(parts.back()));
639 return {"SPLIT " + value.str() + " " + words[2] + " = " + join(pieces)};
640 }
641 if (command == "LET") {
642 checkName(words[1], true);
643 const auto value = quantityValue(database, words[3]);
644 database.variables[words[1]] = value;
645 return {"LET " + words[1] + " = " + value.value.notation(database.simplified(value.dimension))};
646 }
647 if (command == "EVAL") {
648 const auto value = quantityValue(database, words[1]);
649 auto result = value.value;
650 auto label = database.simplified(value.dimension);
651 if (words.size() == 4) {
652 const auto target = unitValue(database, words[3]);
653 value.dimension.match(target.dimension);
654 result = result / target.value;
655 label = words[3];
656 }
657 return {"EVAL " + words[1] + " = " + result.notation(label)};
658 }
659 if (command == "VARS") {
660 std::vector<std::string> rows{"VARS count=" + std::to_string(database.variables.size())};
661 for (const auto& [name, value] : database.variables) rows.push_back("VAR " + name + " = " + value.value.notation(database.simplified(value.dimension)));
662 return rows;
663 }
664 fail("UNKNOWN COMMAND " + command);
665}
