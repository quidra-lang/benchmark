11class Fraction {
12 readonly n: bigint;
13 readonly d: bigint;
14
15 constructor(n: bigint, d: bigint = 1n) {
16 if (d < 0n) { n = -n; d = -d; }
17 const divisor = gcd(n, d);
18 this.n = n / divisor;
19 this.d = d / divisor;
20 }
21 add(other: Fraction): Fraction { return new Fraction(this.n * other.d + other.n * this.d, this.d * other.d); }
22 sub(other: Fraction): Fraction { return new Fraction(this.n * other.d - other.n * this.d, this.d * other.d); }
23 mul(other: Fraction): Fraction { return new Fraction(this.n * other.n, this.d * other.d); }
24 div(other: Fraction): Fraction { return new Fraction(this.n * other.d, this.d * other.n); }
25 neg(): Fraction { return new Fraction(-this.n, this.d); }
26 abs(): Fraction { return this.n < 0n ? this.neg() : this; }
27 compare(other: Fraction): -1 | 0 | 1 {
28 const a = this.n * other.d, b = other.n * this.d;
29 return a < b ? -1 : a > b ? 1 : 0;
30 }
31 pow(exponent: bigint): Fraction {
32 const factor = exponent < 0n ? ONE.div(this) : this;
33 let result = ONE;
34 for (let i = 0n; i < abs(exponent); i++) result = result.mul(factor);
35 return result;
36 }
37 toString(): string { return this.d === 1n ? String(this.n) : `${this.n}/${this.d}`; }
38 decimal(): string {
39 const n = abs(this.n), whole = n / this.d, tail = (n % this.d) * 1_000_000n / this.d;
40 return `${this.n < 0n ? "-" : ""}${whole}.${String(tail).padStart(6, "0")}`;
41 }
42 notation(unit: string): string { return `${this} ${unit} (${this.decimal()})`; }
43}
44const integer = (value: bigint): Fraction => new Fraction(value);
45const ONE = integer(1n), TEN = integer(10n);

47function numberValue(text: string): Fraction {
48 if (!/^-?[0-9]+([./][0-9]+)?$/.test(text)) fail(`BAD NUMBER ${text}`);
49 const negative = text.startsWith("-"), body = negative ? text.slice(1) : text;
50 let value: Fraction;
51 if (body.includes("/")) {
52 const [n, d] = body.split("/");
53 if (BigInt(d) === 0n) fail(`BAD NUMBER ${text}`);
54 value = new Fraction(BigInt(n), BigInt(d));
55 } else if (body.includes(".")) {
56 const [whole, tail] = body.split(".");
57 let scale = 1n;
58 for (const _ of tail) scale *= 10n;
59 value = new Fraction(BigInt(whole) * scale + BigInt(tail), scale);
60 } else value = integer(BigInt(body));
61 return negative ? value.neg() : value;
62}
63function checkedExponent(text: string): bigint {
64 const value = BigInt(text);
65 if (value === 0n || value < -3n || value > 3n) fail(`BAD EXPONENT ${text}`);
66 return value;
67}

173type TokenKind = "name" | "number" | "unit" | "symbol" | "invalid" | "end";
174interface Token { readonly kind: TokenKind; readonly text: string; readonly position: number; }
175type UnitExpr =
176 | { readonly kind: "name"; readonly name: string }
177 | { readonly kind: "number"; readonly value: bigint }
178 | { readonly kind: "power"; readonly child: UnitExpr; readonly exponent: string }
179 | { readonly kind: "product"; readonly terms: readonly UnitTerm[] };
180interface UnitTerm { readonly operation: "*" | "/"; readonly node: UnitExpr; }
181type QuantityExpr =
182 | { readonly kind: "literal"; readonly number: string; readonly unit: string | null }
183 | { readonly kind: "variable"; readonly name: string }
184 | { readonly kind: "negative"; readonly child: QuantityExpr }
185 | { readonly kind: "power"; readonly child: QuantityExpr; readonly exponent: string }
186 | { readonly kind: "binary"; readonly operation: "+" | "-" | "*" | "/"; readonly left: QuantityExpr; readonly right: QuantityExpr }
187 | { readonly kind: "call"; readonly name: string; readonly arguments: readonly QuantityExpr[] };
188const letter = (c: string | undefined): boolean => c !== undefined && ((c >= "A" && c <= "Z") || (c >= "a" && c <= "z"));
189const digit = (c: string | undefined): boolean => c !== undefined && c >= "0" && c <= "9";

191class Parser {
192 private readonly tokens: Token[] = [];
193 private position = 0;
194
195 constructor(private readonly source: string, quantityMode: boolean) {
196 for (let i = 0; i < source.length;) {
197 const start = i, character = source[i++];
198 let kind: TokenKind = "invalid";
199 if (letter(character)) {
200 kind = "name";
201 while (letter(source[i])) i++;
202 } else if (digit(character)) {
203 kind = "number";
204 while (digit(source[i])) i++;
205 if (quantityMode && source[i] === "." && digit(source[i + 1])) {
206 i++;
207 while (digit(source[i])) i++;
208 }
209 } else if (quantityMode && character === "[") {
210 const closing = source.indexOf("]", i);
211 if (closing > i) { kind = "unit"; i = closing + 1; }
212 } else if ((quantityMode ? "+-*/^()," : "*/^()-").includes(character)) kind = "symbol";
213 this.tokens.push({ kind, text: source.slice(start, i), position: start + 1 });
214 }
215 this.tokens.push({ kind: "end", text: "", position: source.length + 1 });
216 }
217 private get current(): Token { return this.tokens[this.position]; }
218 private syntax(): never { fail(`SYNTAX ${this.source} AT ${this.current.kind === "end" ? "END" : this.current.position}`); }
219 private advance(): Token { return this.tokens[this.position++]; }
220 private take(text: string): boolean {
221 if (this.current.text !== text) return false;
222 this.position++;
223 return true;
224 }
225 private expect(text: string): void { if (!this.take(text)) this.syntax(); }
226 private exponent(): string {
227 const sign = this.take("-") ? "-" : "";
228 if (this.current.kind !== "number" || this.current.text.includes(".")) this.syntax();
229 return sign + this.advance().text;
230 }
231 private unitTerm(): UnitExpr {
232 let result: UnitExpr;
233 if (this.current.kind === "name") result = { kind: "name", name: this.advance().text };
234 else if (this.current.kind === "number" && this.current.text[0] !== "0") result = { kind: "number", value: BigInt(this.advance().text) };
235 else if (this.take("(")) { result = this.unitExpression(); this.expect(")"); }
236 else this.syntax();
237 return this.take("^") ? { kind: "power", child: result, exponent: this.exponent() } : result;
238 }
239 private unitExpression(): UnitExpr {
240 const terms: UnitTerm[] = [{ operation: "*", node: this.unitTerm() }];
241 while (this.current.text === "*" || this.current.text === "/") {
242 const operation = this.current.text;
243 this.advance();
244 terms.push({ operation, node: this.unitTerm() });
245 }
246 return { kind: "product", terms };
247 }
248 unit(): UnitExpr {
249 const result = this.unitExpression();
250 if (this.current.kind !== "end") this.syntax();
251 return result;
252 }
253 private primary(): QuantityExpr {
254 let result: QuantityExpr;
255 const token = this.current;
256 if (token.kind === "number") {
257 this.advance();
258 const unit = this.current.kind === "unit" ? this.advance().text.slice(1, -1) : null;
259 result = { kind: "literal", number: token.text, unit };
260 } else if (token.kind === "unit") {
261 this.advance();
262 result = { kind: "literal", number: "1", unit: token.text.slice(1, -1) };
263 } else if (token.kind === "name") {
264 this.advance();
265 if (this.take("(")) {
266 const arguments_: QuantityExpr[] = [this.expression()];
267 while (this.take(",")) arguments_.push(this.expression());
268 this.expect(")");
269 result = { kind: "call", name: token.text, arguments: arguments_ };
270 } else result = { kind: "variable", name: token.text };
271 } else if (this.take("(")) { result = this.expression(); this.expect(")"); }
272 else this.syntax();
273 return this.take("^") ? { kind: "power", child: result, exponent: this.exponent() } : result;
274 }
275 private unary(): QuantityExpr { return this.take("-") ? { kind: "negative", child: this.unary() } : this.primary(); }
276 private product(): QuantityExpr {
277 let result = this.unary();
278 while (this.current.text === "*" || this.current.text === "/") {
279 const operation = this.current.text;
280 this.advance();
281 result = { kind: "binary", operation, left: result, right: this.unary() };
282 }
283 return result;
284 }
285 private expression(): QuantityExpr {
286 let result = this.product();
287 while (this.current.text === "+" || this.current.text === "-") {
288 const operation = this.current.text;
289 this.advance();
290 result = { kind: "binary", operation, left: result, right: this.product() };
291 }
292 return result;
293 }
294 quantity(): QuantityExpr {
295 const result = this.expression();
296 if (this.current.kind !== "end") this.syntax();
297 return result;
298 }
299}

301function validateUnit(database: Database, node: UnitExpr): void {
302 switch (node.kind) {
303 case "name": if (!database.resolve(node.name)) fail(`UNKNOWN UNIT ${node.name}`); break;
304 case "power": validateUnit(database, node.child); checkedExponent(node.exponent); break;
305 case "product": for (const term of node.terms) validateUnit(database, term.node); break;
306 case "number": break;
307 }
308}
309function evaluateUnit(database: Database, node: UnitExpr): Quantity {
310 switch (node.kind) {
311 case "name": return database.resolve(node.name)!;
312 case "number": return quantity(integer(node.value));
313 case "power": {
314 const value = evaluateUnit(database, node.child), exponent = BigInt(node.exponent);
315 return quantity(value.value.pow(exponent), value.dimension.scale(exponent));
316 }
317 case "product": {
318 let result = quantity();
319 for (const term of node.terms) {
320 const right = evaluateUnit(database, term.node);
321 result = term.operation === "*" ? quantity(result.value.mul(right.value), result.dimension.add(right.dimension)) :
322 quantity(result.value.div(right.value), result.dimension.sub(right.dimension));
323 }
324 return result;
325 }
326 }

335interface Reduced { readonly coefficient: Fraction; readonly names: ReadonlyMap<string, bigint>; }
336function reduceUnit(node: UnitExpr): Reduced {
337 switch (node.kind) {
338 case "name": return { coefficient: ONE, names: new Map([[node.name, 1n]]) };
339 case "number": return { coefficient: integer(node.value), names: new Map() };
340 case "power": {
341 const result = reduceUnit(node.child), exponent = BigInt(node.exponent);
342 return { coefficient: result.coefficient.pow(exponent), names: new Map([...result.names].map(([name, power]) => [name, power * exponent])) };
343 }
344 case "product": {
345 let coefficient = ONE;
346 const names = new Map<string, bigint>();
347 for (const term of node.terms) {
348 const right = reduceUnit(term.node), multiply = term.operation === "*";
349 coefficient = multiply ? coefficient.mul(right.coefficient) : coefficient.div(right.coefficient);
350 for (const [name, power] of right.names) names.set(name, (names.get(name) ?? 0n) + (multiply ? power : -power));
351 }
352 return { coefficient, names };
353 }
354 }

361function integerSqrt(value: bigint): bigint {
362 if (value < 2n) return value;
363 let current = value, next = (value + 1n) / 2n;
364 while (next < current) { current = next; next = (current + value / current) / 2n; }
365 return current;
366}
367function evaluateQuantity(database: Database, node: QuantityExpr): Quantity {
368 switch (node.kind) {
369 case "literal": {
370 if (node.unit === null) return quantity(numberValue(node.number));
371 const unit = unitValue(database, node.unit);
372 return quantity(numberValue(node.number).mul(unit.value), unit.dimension);
373 }
374 case "variable": {
375 const value = database.variables.get(node.name);
376 if (!value) fail(`UNKNOWN VARIABLE ${node.name}`);
377 return value;
378 }
379 case "negative": {
380 const value = evaluateQuantity(database, node.child);
381 return quantity(value.value.neg(), value.dimension);
382 }
383 case "power": {
384 const value = evaluateQuantity(database, node.child), exponent = checkedExponent(node.exponent);
385 if (value.value.n === 0n && exponent < 0n) fail("DIVISION BY ZERO");
386 return quantity(value.value.pow(exponent), value.dimension.scale(exponent));
387 }
388 case "binary": {
389 const left = evaluateQuantity(database, node.left), right = evaluateQuantity(database, node.right);
390 if (node.operation === "+" || node.operation === "-") {
391 left.dimension.match(right.dimension);
392 return quantity(node.operation === "+" ? left.value.add(right.value) : left.value.sub(right.value), left.dimension);
393 }
394 if (node.operation === "*") return quantity(left.value.mul(right.value), left.dimension.add(right.dimension));
395 if (right.value.n === 0n) fail("DIVISION BY ZERO");
396 return quantity(left.value.div(right.value), left.dimension.sub(right.dimension));
397 }
398 case "call": {
399 if (!["abs", "sqrt", "min", "max"].includes(node.name)) fail(`UNKNOWN FUNCTION ${node.name}`);
400 const arity = node.name === "abs" || node.name === "sqrt" ? 1 : 2;
401 if (node.arguments.length !== arity) fail(`BAD ARGUMENTS ${node.name}`);
402 const first = evaluateQuantity(database, node.arguments[0]);
403 if (node.name === "abs") return quantity(first.value.abs(), first.dimension);
404 if (node.name === "sqrt") {
405 if (first.value.n < 0n) fail("NEGATIVE ROOT");
406 if (first.dimension.exponents.some(value => value % 2n !== 0n)) fail(`ODD DIMENSION ${first.dimension}`);
407 const n = integerSqrt(first.value.n), d = integerSqrt(first.value.d);
408 if (n * n !== first.value.n || d * d !== first.value.d) fail("NOT EXACT");
409 return quantity(new Fraction(n, d), first.dimension.map(value => value / 2n));
410 }
411 const second = evaluateQuantity(database, node.arguments[1]);
412 first.dimension.match(second.dimension);
413 const comparison = first.value.compare(second.value);
414 return (node.name === "min" ? comparison <= 0 : comparison >= 0) ? first : second;
415 }
416 }

423const isCommand = (name: string): name is CommandName => COMMANDS.some(command => command === name);

425function execute(database: Database, line: string): string[] {
426 const words = line.split(" "), command = words[0];
427 if (!isCommand(command)) fail(`UNKNOWN COMMAND ${command}`);
428 switch (command) {
429 case "DEFINE": {
430 const name = words[1];
431 checkName(name);
432 const value = numberValue(words[3]), unit = unitValue(database, words[4]);
433 if (kindDimension(name) || database.resolve(name)) fail(`NAME TAKEN ${name}`);
434 if (value.n <= 0n) fail(`NOT POSITIVE ${words[3]}`);
435 const result = quantity(value.mul(unit.value), unit.dimension);
436 database.units.push({ symbol: name, quantity: result, prefixable: false, definition: { value, expression: words[4] } });
437 return [`DEFINE ${name} factor=${result.value} dim=${result.dimension} kind=${kindName(result.dimension)}`];
438 }
439 case "UNDEF": {
440 const index = database.units.findIndex(unit => unit.symbol === words[1]);
441 if (index < 0) fail(`NOT DEFINED ${words[1]}`);
442 if (index < BUILTINS.length) fail(`BUILTIN UNIT ${words[1]}`);
443 database.units.splice(index, 1);
444 return [`UNDEF ${words[1]}`];
445 }
446 case "DEFS": return [`DEFS count=${database.units.length - BUILTINS.length}`, ...database.units.slice(BUILTINS.length).map(unit =>
447 `DEF ${unit.symbol} = ${unit.definition!.value} ${unit.definition!.expression} factor=${unit.quantity.value} dim=${unit.quantity.dimension}`)];
448 case "CONVERT": case "INVERT": {
449 const value = numberValue(words[1]), source = unitValue(database, words[2]), target = unitValue(database, words[4]);
450 if (command === "INVERT") {
451 if (source.dimension.equals(DIMENSIONLESS) || !source.dimension.add(target.dimension).equals(DIMENSIONLESS)) {
452 fail(`NOT RECIPROCAL ${source.dimension} vs ${target.dimension}`);
453 }
454 if (value.n === 0n) fail("DIVISION BY ZERO");
455 } else source.dimension.match(target.dimension);
456 let result = value.mul(source.value);
457 if (command === "INVERT") result = ONE.div(result);
458 return [`${command} ${value} ${words[2]} = ${result.div(target.value).notation(words[4])}`];
459 }
460 case "DIM": case "REDUCE": case "SIMPLIFY": {
461 const tree = parseUnit(database, words[1]), unit = evaluateUnit(database, tree);
462 const result = command === "DIM" ? `${unit.dimension} factor=${unit.value} kind=${kindName(unit.dimension)}` :
463 command === "REDUCE" ? reducedText(tree) : unit.value.notation(database.simplified(unit.dimension));
464 return [`${command} ${words[1]} = ${result}`];
465 }
466 case "ADD": case "SUB": case "MUL": case "DIV": case "COMPARE": {
467 const a = numberValue(words[1]), first = unitValue(database, words[2]);
468 const b = numberValue(words[3]), second = unitValue(database, words[4]);
469 const hasTarget = words.length === 7, target = hasTarget ? unitValue(database, words[6]) : first;
470 const prefix = `${command} ${a} ${words[2]}`, suffix = `${b} ${words[4]}`;
471 if (command === "ADD" || command === "SUB" || command === "COMPARE") {
472 first.dimension.match(second.dimension);
473 if (hasTarget) first.dimension.match(target.dimension);
474 const left = a.mul(first.value), right = b.mul(second.value);
475 if (command === "COMPARE") {
476 const relation = ["<", "=", ">"][left.compare(right) + 1];
477 return [`${prefix} ${relation} ${suffix} ratio=${right.n === 0n ? "undefined" : left.div(right)}`];
478 }
479 const add = command === "ADD", result = add ? left.add(right) : left.sub(right);
480 return [`${prefix} ${add ? "+" : "-"} ${suffix} = ${result.div(target.value).notation(hasTarget ? words[6] : words[2])}`];
481 }
482 const multiply = command === "MUL", dimension = multiply ? first.dimension.add(second.dimension) : first.dimension.sub(second.dimension);
483 if (!multiply && b.n === 0n) fail("DIVISION BY ZERO");
484 if (hasTarget) dimension.match(target.dimension);
485 const raw = multiply ? a.mul(b) : a.div(b), left = a.mul(first.value), right = b.mul(second.value);
486 const coherent = multiply ? left.mul(right) : left.div(right), operation = multiply ? "*" : "/";
487 const heading = `${prefix} ${operation} ${suffix} =`;
488 if (hasTarget) return [`${heading} ${coherent.div(target.value).notation(words[6])}`];
489 const rightUnit = !multiply && /[*/]/.test(words[4]) ? `(${words[4]})` : words[4];
490 return [`${heading} ${raw} ${words[2]}${operation}${rightUnit} = ${coherent.notation(database.simplified(dimension))}`];
491 }
492 case "TEMP": {
493 const value = numberValue(words[1]);
494 for (const scale of [words[2], words[4]]) if (!["C", "F", "K", "R"].includes(scale)) fail(`UNKNOWN SCALE ${scale}`);
495 const celsius = new Fraction(5463n, 20n), fahrenheit = new Fraction(45967n, 100n);
496 const kelvin = words[2] === "C" ? value.add(celsius) : words[2] === "F" ? value.add(fahrenheit).mul(integer(5n)).div(integer(9n)) :
497 words[2] === "R" ? value.mul(integer(5n)).div(integer(9n)) : value;
498 if (kelvin.n < 0n) fail("BELOW ABSOLUTE ZERO");
499 const result = words[4] === "C" ? kelvin.sub(celsius) : words[4] === "F" ? kelvin.mul(integer(9n)).div(integer(5n)).sub(fahrenheit) :
500 words[4] === "R" ? kelvin.mul(integer(9n)).div(integer(5n)) : kelvin;
501 return [`TEMP ${value} ${words[2]} = ${result.notation(words[4])}`];
502 }
503 case "LIST": {
504 const dimension = kindDimension(words[1]) ?? unitValue(database, words[1]).dimension;
505 const selected = database.selected(dimension);
506 return [`LIST ${words[1]} dim=${dimension} kind=${kindName(dimension)} count=${selected.length}`,
507 ...selected.map(unit => `UNIT ${unit.symbol} factor=${unit.quantity.value} origin=${unit.definition ? "user" : "builtin"} prefix=${unit.prefixable ? "yes" : "no"}`)];
508 }
509 case "TABLE": {
510 const value = numberValue(words[1]), unit = unitValue(database, words[2]), selected = database.selected(unit.dimension);
511 return [`TABLE ${value} ${words[2]} dim=${unit.dimension} count=${selected.length}`,
512 ...selected.map(item => `ROW ${value.mul(unit.value).div(item.quantity.value).notation(item.symbol)}`)];
513 }
514 case "AUTO": {
515 const value = numberValue(words[1]), unit = unitValue(database, words[2]);
516 const reference = database.units.slice(0, BUILTINS.length).find(item => item.prefixable && item.quantity.dimension.equals(unit.dimension));
517 if (!reference) fail(`NO PREFIXABLE UNIT ${unit.dimension}`);
518 const scaled = value.mul(unit.value).div(reference.quantity.value), magnitude = scaled.abs();
519 let exponent = 0n;
520 if (magnitude.compare(ONE) >= 0) {
521 while (exponent < 9n && magnitude.compare(TEN.pow(exponent + 3n)) >= 0) exponent += 3n;
522 } else if (magnitude.n > 0n) {
523 while (exponent > -9n && magnitude.compare(TEN.pow(exponent)) < 0) exponent -= 3n;
524 }
525 const prefix = ["n", "u", "m", "", "k", "M", "G"][Number((exponent + 9n) / 3n)];
526 return [`AUTO ${value} ${words[2]} = ${scaled.div(TEN.pow(exponent)).notation(prefix + reference.symbol)}`];
527 }
528 case "SPLIT": {
529 const value = numberValue(words[1]), unit = unitValue(database, words[2]);
530 const parts = words[4].split(","), targets = parts.map(part => unitValue(database, part));
531 if (value.n < 0n) fail("NEGATIVE VALUE");
532 for (const target of targets) unit.dimension.match(target.dimension);
533 for (let i = 0; i + 1 < targets.length; i++) if (targets[i].value.compare(targets[i + 1].value) <= 0) fail("UNIT ORDER");
534 let remaining = value.mul(unit.value);
535 const pieces: string[] = [];
536 for (let i = 0; i + 1 < targets.length; i++) {
537 const quotient = remaining.div(targets[i].value), whole = quotient.n / quotient.d;
538 remaining = remaining.sub(integer(whole).mul(targets[i].value));
539 pieces.push(`${whole} ${parts[i]}`);
540 }
541 pieces.push(remaining.div(targets.at(-1)!.value).notation(parts.at(-1)!));
542 return [`SPLIT ${value} ${words[2]} = ${pieces.join(" ")}`];
543 }
544 case "LET": {
545 checkName(words[1], true);
546 const value = quantityValue(database, words[3]);
547 database.variables.set(words[1], value);
548 return [`LET ${words[1]} = ${value.value.notation(database.simplified(value.dimension))}`];
549 }
550 case "EVAL": {
551 const value = quantityValue(database, words[1]);
552 let result = value.value, label = database.simplified(value.dimension);
553 if (words.length === 4) {
554 const target = unitValue(database, words[3]);
555 value.dimension.match(target.dimension);
556 result = result.div(target.value);
557 label = words[3];
558 }
559 return [`EVAL ${words[1]} = ${result.notation(label)}`];
560 }
561 case "VARS": return [`VARS count=${database.variables.size}`, ...[...database.variables.keys()].sort().map(name => {
562 const value = database.variables.get(name)!;
563 return `VAR ${name} = ${value.value.notation(database.simplified(value.dimension))}`;
564 })];
565 }
566}
567
568const lines = readFileSync(0, "utf8").split(/\r?\n/), commandCount = Number(lines[0]);
569const database = new Database(), output: string[] = [];
570for (let i = 1; i <= commandCount; i++) {
571 try { output.push(...execute(database, lines[i])); }
572 catch (error) {
573 if (!(error instanceof CommandError)) throw error;
574 output.push(`ERROR ${error.message}`);
575 }
576}
577process.stdout.write(output.join("\n") + "\n");
