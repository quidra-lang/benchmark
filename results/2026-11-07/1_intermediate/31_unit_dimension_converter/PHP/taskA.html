8final readonly class Fraction {
9 public int $n;
10 public int $d;
11 public function __construct(int $n, int $d = 1) {
12 if ($d < 0) { $n = -$n; $d = -$d; }
13 $a = abs($n); $b = $d;
14 while ($b !== 0) [$a, $b] = [$b, $a % $b];
15 $this->n = intdiv($n, $a);
16 $this->d = intdiv($d, $a);
17 }
18 public static function parse(string $text): self {
19 if (!preg_match('/\A-?[0-9]+(?:[.\/][0-9]+)?\z/', $text)) throw new CommandError("BAD NUMBER $text");
20 $negative = str_starts_with($text, '-');
21 $body = $negative ? substr($text, 1) : $text;
22 $separator = strcspn($body, './');
23 $first = substr($body, 0, $separator);
24 $n = (int)$first; $d = 1;
25 if ($separator < strlen($body)) {
26 $second = substr($body, $separator + 1);
27 if ($body[$separator] === '/') {
28 $d = (int)$second;
29 if ($d === 0) throw new CommandError("BAD NUMBER $text");
30 } else {
31 for ($i = 0; $i < strlen($second); $i++) $d *= 10;
32 $n = $n * $d + (int)$second;
33 }
34 }
35 return new self($negative ? -$n : $n, $d);
36 }
37 public function add(self $b): self { return new self($this->n * $b->d + $b->n * $this->d, $this->d * $b->d); }
38 public function sub(self $b): self { return new self($this->n * $b->d - $b->n * $this->d, $this->d * $b->d); }
39 public function mul(self $b): self { return new self($this->n * $b->n, $this->d * $b->d); }
40 public function div(self $b): self { return new self($this->n * $b->d, $this->d * $b->n); }
41 public function neg(): self { return new self(-$this->n, $this->d); }
42 public function magnitude(): self { return new self(abs($this->n), $this->d); }
43 public function same(self $b): bool { return $this->n === $b->n && $this->d === $b->d; }
44 public function compare(self $b): int { return ($this->n * $b->d) <=> ($b->n * $this->d); }
45 public function power(int $exponent): self {
46 $factor = $exponent < 0 ? (new self(1))->div($this) : $this;
47 $result = new self(1);
48 for ($i = 0; $i < abs($exponent); $i++) $result = $result->mul($factor);
49 return $result;
50 }
51 public function decimal(): string {
52 $magnitude = abs($this->n);
53 $whole = intdiv($magnitude, $this->d);
54 $places = str_pad((string)intdiv($magnitude % $this->d * 1000000, $this->d), 6, '0', STR_PAD_LEFT);
55 return ($this->n < 0 ? '-' : '') . "$whole.$places";
56 }
57 public function __toString(): string { return $this->d === 1 ? (string)$this->n : "$this->n/$this->d"; }
58}

60final readonly class Dimension {
61 public function __construct(public array $exponents = [0, 0, 0, 0, 0, 0, 0]) {}
62 public function same(self $b): bool { return $this->exponents === $b->exponents; }
63 public function isZero(): bool { return $this->exponents === [0, 0, 0, 0, 0, 0, 0]; }
64 public function isEven(): bool {
65 foreach ($this->exponents as $value) if ($value % 2 !== 0) return false;
66 return true;
67 }
68 public function add(self $b): self {
69 $result = [];
70 foreach ($this->exponents as $i => $value) $result[] = $value + $b->exponents[$i];
71 return new self($result);
72 }
73 public function sub(self $b): self {
74 $result = [];
75 foreach ($this->exponents as $i => $value) $result[] = $value - $b->exponents[$i];
76 return new self($result);
77 }
78 public function scale(int $factor): self { return new self(array_map(fn(int $x): int => $x * $factor, $this->exponents)); }
79 public function half(): self { return new self(array_map(fn(int $x): int => intdiv($x, 2), $this->exponents)); }
80 public function __toString(): string {
81 $symbols = ['kg', 'm', 's', 'A', 'K', 'mol', 'cd'];
82 $parts = [];
83 foreach ($this->exponents as $i => $value) {
84 if ($value !== 0) $parts[] = $value === 1 ? $symbols[$i] : $symbols[$i] . "^$value";
85 }
86 return $parts === [] ? '1' : implode(' ', $parts);
87 }

100enum TokenKind { case Number; case Identifier; case Unit; case Symbol; case Invalid; case End; }
101final readonly class Token {
102 public function __construct(public TokenKind $kind, public string $text, public int $position) {}
103}
104abstract class Parser {
105 private array $tokens = [];
106 private int $index = 0;
107 public function __construct(private readonly string $source, bool $quantity) {
108 for ($i = 0; $i < strlen($source);) {
109 $start = $i++;
110 $c = $source[$start];
111 if (self::letter($c)) {
112 while ($i < strlen($source) && self::letter($source[$i])) $i++;
113 $kind = TokenKind::Identifier;
114 } elseif (self::digit($c)) {
115 while ($i < strlen($source) && self::digit($source[$i])) $i++;
116 if ($quantity && $i + 1 < strlen($source) && $source[$i] === '.' && self::digit($source[$i + 1])) {
117 $i += 2;
118 while ($i < strlen($source) && self::digit($source[$i])) $i++;
119 }
120 $kind = TokenKind::Number;
121 } else {
122 $end = $quantity && $c === '[' ? strpos($source, ']', $i) : false;
123 if ($end !== false && $end > $i) {
124 $i = $end + 1;
125 $kind = TokenKind::Unit;
126 } else {
127 $kind = str_contains($quantity ? '+-*/^(),' : '*/^()-', $c) ? TokenKind::Symbol : TokenKind::Invalid;
128 }
129 }
130 $this->tokens[] = new Token($kind, substr($source, $start, $i - $start), $start + 1);
131 }
132 $this->tokens[] = new Token(TokenKind::End, '', strlen($source) + 1);
133 }
134 public static function letter(string $c): bool {
135 $n = ord($c);
136 return ($n >= 65 && $n <= 90) || ($n >= 97 && $n <= 122);
137 }
138 private static function digit(string $c): bool { $n = ord($c); return $n >= 48 && $n <= 57; }
139 protected function current(): Token { return $this->tokens[$this->index]; }
140 protected function advance(): Token { return $this->tokens[$this->index++]; }
141 protected function take(string $text): bool {
142 if ($this->current()->text !== $text) return false;
143 $this->index++;
144 return true;
145 }
146 protected function syntax(): CommandError {
147 $token = $this->current();
148 $position = $token->kind === TokenKind::End ? 'END' : (string)$token->position;
149 return new CommandError("SYNTAX $this->source AT $position");
150 }
151 protected function expect(string $text): void { if (!$this->take($text)) throw $this->syntax(); }
152 protected function finish(): void { if ($this->current()->kind !== TokenKind::End) throw $this->syntax(); }
153 protected function exponent(): string {
154 $sign = $this->take('-') ? '-' : '';
155 if ($this->current()->kind !== TokenKind::Number || str_contains($this->current()->text, '.')) throw $this->syntax();
156 return $sign . $this->advance()->text;
157 }
158}

173final class UnitParser extends Parser {
174 public function __construct(string $source) { parent::__construct($source, false); }
175 public function parse(): UnitNode { $node = $this->expression(); $this->finish(); return $node; }
176 private function expression(): UnitNode {
177 $terms = [['*', $this->term()]];
178 while (in_array($this->current()->text, ['*', '/'], true)) {
179 $op = $this->advance()->text;
180 $terms[] = [$op, $this->term()];
181 }
182 return new UnitProduct($terms);
183 }
184 private function term(): UnitNode {
185 $token = $this->current();
186 if ($token->kind === TokenKind::Identifier) {
187 $atom = new UnitName($this->advance()->text);
188 } elseif ($token->kind === TokenKind::Number && $token->text[0] !== '0') {
189 $atom = new UnitNumber($this->advance()->text);
190 } elseif ($this->take('(')) {
191 $atom = $this->expression();
192 $this->expect(')');
193 } else throw $this->syntax();
194 return $this->take('^') ? new UnitPower($atom, $this->exponent()) : $atom;
195 }
196}

217final class QuantityParser extends Parser {
218 public function __construct(string $source) { parent::__construct($source, true); }
219 public function parse(): QuantityNode { $node = $this->expression(); $this->finish(); return $node; }
220 private function expression(): QuantityNode {
221 $node = $this->product();
222 while (in_array($this->current()->text, ['+', '-'], true)) {
223 $op = $this->advance()->text;
224 $node = new Binary($op, $node, $this->product());
225 }
226 return $node;
227 }
228 private function product(): QuantityNode {
229 $node = $this->unary();
230 while (in_array($this->current()->text, ['*', '/'], true)) {
231 $op = $this->advance()->text;
232 $node = new Binary($op, $node, $this->unary());
233 }
234 return $node;
235 }
236 private function unary(): QuantityNode {
237 if ($this->take('-')) return new Negative($this->unary());
238 $node = $this->primary();
239 return $this->take('^') ? new Power($node, $this->exponent()) : $node;
240 }
241 private function primary(): QuantityNode {
242 if ($this->current()->kind === TokenKind::Number) {
243 $number = $this->advance()->text;
244 return new Literal($number, $this->current()->kind === TokenKind::Unit ? $this->unitString() : null);
245 }
246 if ($this->current()->kind === TokenKind::Unit) return new Literal('1', $this->unitString());
247 if ($this->current()->kind === TokenKind::Identifier) {
248 $name = $this->advance()->text;
249 if (!$this->take('(')) return new Variable($name);
250 $arguments = [$this->expression()];
251 while ($this->take(',')) $arguments[] = $this->expression();
252 $this->expect(')');
253 return new Call($name, $arguments);
254 }
255 if ($this->take('(')) {
256 $node = $this->expression();
257 $this->expect(')');
258 return $node;
259 }
260 throw $this->syntax();
261 }
262 private function unitString(): string { return substr($this->advance()->text, 1, -1); }
263}

373 private function unitValue(UnitNode $node): Quantity {
374 if ($node instanceof UnitName) return $this->resolve($node->name);
375 if ($node instanceof UnitNumber) return new Quantity(new Fraction((int)$node->text), new Dimension());
376 if ($node instanceof UnitPower) {

393 private static function reduce(UnitNode $node): Reduced {
394 if ($node instanceof UnitName) return new Reduced(new Fraction(1), [$node->name => 1]);
395 if ($node instanceof UnitNumber) return new Reduced(new Fraction((int)$node->text), []);
396 if ($node instanceof UnitPower) {

398 $exponent = (int)$node->exponent;
399 $powers = [];
400 foreach ($atom->exponents as $name => $value) $powers[$name] = $value * $exponent;
401 return new Reduced($atom->coefficient->power($exponent), $powers);
402 }
403 if ($node instanceof UnitProduct) {
404 $coefficient = new Fraction(1); $exponents = [];
405 foreach ($node->terms as [$op, $term]) {

424 private static function integerRoot(int $number): int {
425 $low = 0; $high = $number + 1;
426 while ($high - $low > 1) {
427 $middle = $low + intdiv($high - $low, 2);
428 if ($middle <= intdiv($number, $middle)) $low = $middle;
429 else $high = $middle;
430 }
431 return $low;
432 }
433 private function evaluate(QuantityNode $node): Quantity {
434 if ($node instanceof Literal) {
435 if ($node->unit === null) return new Quantity(Fraction::parse($node->number), new Dimension());
436 $unit = $this->parseUnit($node->unit)->quantity;
437 return new Quantity(Fraction::parse($node->number)->mul($unit->value), $unit->dimension);
438 }
439 if ($node instanceof Variable) return $this->variables[$node->name] ?? throw new CommandError("UNKNOWN VARIABLE $node->name");
440 if ($node instanceof Negative) {
441 $operand = $this->evaluate($node->operand);
442 return new Quantity($operand->value->neg(), $operand->dimension);
443 }
444 if ($node instanceof Power) {
445 $base = $this->evaluate($node->base);
446 $exponent = self::checkExponent($node->exponent);
447 if ($base->value->n === 0 && $exponent < 0) throw new CommandError('DIVISION BY ZERO');
448 return new Quantity($base->value->power($exponent), $base->dimension->scale($exponent));
449 }
450 if ($node instanceof Binary) {
451 $a = $this->evaluate($node->left);
452 $b = $this->evaluate($node->right);
453 if ($node->op === '+' || $node->op === '-') {
454 self::matchDimensions($a->dimension, $b->dimension);
455 return new Quantity($node->op === '+' ? $a->value->add($b->value) : $a->value->sub($b->value), $a->dimension);
456 }
457 if ($node->op === '*') return new Quantity($a->value->mul($b->value), $a->dimension->add($b->dimension));
458 if ($b->value->n === 0) throw new CommandError('DIVISION BY ZERO');
459 return new Quantity($a->value->div($b->value), $a->dimension->sub($b->dimension));
460 }
461 if ($node instanceof Call) {
462 if (!in_array($node->name, ['abs', 'sqrt', 'min', 'max'], true)) throw new CommandError("UNKNOWN FUNCTION $node->name");
463 $count = in_array($node->name, ['abs', 'sqrt'], true) ? 1 : 2;
464 if (count($node->arguments) !== $count) throw new CommandError("BAD ARGUMENTS $node->name");
465 $a = $this->evaluate($node->arguments[0]);
466 if ($node->name === 'abs') return new Quantity($a->value->magnitude(), $a->dimension);
467 if ($node->name === 'sqrt') {
468 if ($a->value->n < 0) throw new CommandError('NEGATIVE ROOT');
469 if (!$a->dimension->isEven()) throw new CommandError("ODD DIMENSION $a->dimension");
470 $p = self::integerRoot($a->value->n); $q = self::integerRoot($a->value->d);
471 if ($p * $p !== $a->value->n || $q * $q !== $a->value->d) throw new CommandError('NOT EXACT');
472 return new Quantity(new Fraction($p, $q), $a->dimension->half());
473 }
474 $b = $this->evaluate($node->arguments[1]);
475 self::matchDimensions($a->dimension, $b->dimension);
476 $comparison = $a->value->compare($b->value);
477 return $node->name === 'min' ? ($comparison <= 0 ? $a : $b) : ($comparison >= 0 ? $a : $b);
478 }
479 throw new LogicException('Unknown quantity node');
480 }
481 private function matching(Dimension $dimension): array {
482 $selected = array_values(array_filter($this->units, fn(Unit $unit): bool => $unit->quantity->dimension->same($dimension)));
483 usort($selected, function(Unit $a, Unit $b): int {
484 $comparison = $a->quantity->value->compare($b->quantity->value);
485 return $comparison !== 0 ? $comparison : strcmp($a->symbol, $b->symbol);
486 });
487 return $selected;
488 }

490 public function execute(string $line): string {
491 $w = explode(' ', $line); $command = $w[0];
492 switch ($command) {
493 case 'DEFINE':
494 self::checkName($w[1], false);
495 $value = Fraction::parse($w[3]);
496 $unit = $this->parseUnit($w[4])->quantity;
497 if (array_key_exists($w[1], self::KIND_EXPONENTS) || $this->resolve($w[1]) !== null) throw new CommandError("NAME TAKEN $w[1]");
498 if ($value->n <= 0) throw new CommandError("NOT POSITIVE $w[3]");
499 $defined = new Quantity($value->mul($unit->value), $unit->dimension);
500 $this->units[] = new Unit($w[1], $defined, false, new Definition($value, $w[4]));
501 return "DEFINE $w[1] factor=$defined->value dim=$defined->dimension kind=" . self::kind($defined->dimension);
502 case 'UNDEF':
503 $index = null;
504 foreach ($this->units as $i => $unit) if ($unit->symbol === $w[1]) { $index = $i; break; }
505 if ($index !== null && $this->units[$index]->definition === null) throw new CommandError("BUILTIN UNIT $w[1]");
506 if ($index === null) throw new CommandError("NOT DEFINED $w[1]");
507 array_splice($this->units, $index, 1);
508 return "UNDEF $w[1]";
509 case 'DEFS':
510 $users = array_values(array_filter($this->units, fn(Unit $unit): bool => $unit->definition !== null));
511 $lines = ['DEFS count=' . count($users)];
512 foreach ($users as $unit) {
513 $definition = $unit->definition; $quantity = $unit->quantity;
514 $lines[] = "DEF $unit->symbol = $definition->value $definition->expression factor=$quantity->value dim=$quantity->dimension";
515 }
516 return implode("\n", $lines);
517 case 'CONVERT': case 'INVERT':
518 $value = Fraction::parse($w[1]);
519 $a = $this->parseUnit($w[2])->quantity;
520 $b = $this->parseUnit($w[4])->quantity;
521 if ($command === 'CONVERT') self::matchDimensions($a->dimension, $b->dimension);
522 else {
523 if ($a->dimension->isZero() || !$a->dimension->add($b->dimension)->isZero()) throw new CommandError("NOT RECIPROCAL $a->dimension vs $b->dimension");
524 if ($value->n === 0) throw new CommandError('DIVISION BY ZERO');
525 }
526 $result = $command === 'CONVERT' ? $value->mul($a->value)->div($b->value) : (new Fraction(1))->div($value->mul($a->value))->div($b->value);
527 return "$command $value $w[2] = " . self::format($result, $w[4]);
528 case 'DIM': case 'REDUCE': case 'SIMPLIFY':
529 $parsed = $this->parseUnit($w[1]); $unit = $parsed->quantity;
530 return match ($command) {
531 'DIM' => "DIM $w[1] = $unit->dimension factor=$unit->value kind=" . self::kind($unit->dimension),
532 'REDUCE' => "REDUCE $w[1] = " . self::reducedText($parsed),
533 default => "SIMPLIFY $w[1] = " . $this->formatQuantity($unit),
534 };
535 case 'ADD': case 'SUB': case 'MUL': case 'DIV': case 'COMPARE':
536 $x = Fraction::parse($w[1]);
537 $a = $this->parseUnit($w[2])->quantity;
538 $y = Fraction::parse($w[3]);
539 $b = $this->parseUnit($w[4])->quantity;
540 $target = count($w) === 7 ? $this->parseUnit($w[6])->quantity : null;
541 if ($command === 'COMPARE') {
542 self::matchDimensions($a->dimension, $b->dimension);
543 $left = $x->mul($a->value); $right = $y->mul($b->value);
544 $comparison = $left->compare($right);
545 $relation = $comparison < 0 ? '<' : ($comparison > 0 ? '>' : '=');
546 $ratio = $right->n === 0 ? 'undefined' : (string)$left->div($right);
547 return "COMPARE $x $w[2] $relation $y $w[4] ratio=$ratio";
548 }
549 $op = match ($command) { 'ADD' => '+', 'SUB' => '-', 'MUL' => '*', default => '/' };
550 $prefix = "$command $x $w[2] $op $y $w[4] = ";
551 if ($command === 'ADD' || $command === 'SUB') {
552 self::matchDimensions($a->dimension, $b->dimension);
553 if ($target !== null) self::matchDimensions($a->dimension, $target->dimension);
554 $left = $x->mul($a->value); $right = $y->mul($b->value);
555 $sum = $command === 'ADD' ? $left->add($right) : $left->sub($right);
556 return $prefix . self::format($sum->div($target?->value ?? $a->value), $target === null ? $w[2] : $w[6]);
557 }
558 $dimension = $command === 'MUL' ? $a->dimension->add($b->dimension) : $a->dimension->sub($b->dimension);
559 if ($command === 'DIV' && $y->n === 0) throw new CommandError('DIVISION BY ZERO');
560 if ($target !== null) self::matchDimensions($dimension, $target->dimension);
561 $product = $command === 'MUL' ? $x->mul($y) : $x->div($y);
562 $first = $x->mul($a->value); $second = $y->mul($b->value);
563 $coherent = $command === 'MUL' ? $first->mul($second) : $first->div($second);
564 if ($target !== null) return $prefix . self::format($coherent->div($target->value), $w[6]);
565 $rightUnit = $command === 'DIV' && (str_contains($w[4], '*') || str_contains($w[4], '/')) ? "($w[4])" : $w[4];
566 return "$prefix$product $w[2]$op$rightUnit = " . self::format($coherent, $this->simplified($dimension));
567 case 'TEMP':
568 $value = Fraction::parse($w[1]);
569 foreach ([$w[2], $w[4]] as $scale) if (!in_array($scale, ['C', 'F', 'K', 'R'], true)) throw new CommandError("UNKNOWN SCALE $scale");
570 $kelvin = match ($w[2]) {
571 'C' => $value->add(new Fraction(5463, 20)),
572 'F' => $value->add(new Fraction(45967, 100))->mul(new Fraction(5))->div(new Fraction(9)),
573 'R' => $value->mul(new Fraction(5))->div(new Fraction(9)), default => $value,
574 };
575 if ($kelvin->n < 0) throw new CommandError('BELOW ABSOLUTE ZERO');
576 $result = match ($w[4]) {
577 'C' => $kelvin->sub(new Fraction(5463, 20)),
578 'F' => $kelvin->mul(new Fraction(9))->div(new Fraction(5))->sub(new Fraction(45967, 100)),
579 'R' => $kelvin->mul(new Fraction(9))->div(new Fraction(5)), default => $kelvin,
580 };
581 return "TEMP $value $w[2] = " . self::format($result, $w[4]);
582 case 'LIST':
583 $dimension = array_key_exists($w[1], self::KIND_EXPONENTS) ? new Dimension(self::KIND_EXPONENTS[$w[1]]) : $this->parseUnit($w[1])->quantity->dimension;
584 $selected = $this->matching($dimension);
585 $lines = ["LIST $w[1] dim=$dimension kind=" . self::kind($dimension) . ' count=' . count($selected)];
586 foreach ($selected as $unit) {
587 $origin = $unit->definition === null ? 'builtin' : 'user'; $prefix = $unit->prefix ? 'yes' : 'no';
588 $factor = $unit->quantity->value;
589 $lines[] = "UNIT $unit->symbol factor=$factor origin=$origin prefix=$prefix";
590 }
591 return implode("\n", $lines);
592 case 'TABLE': case 'AUTO': case 'SPLIT':
593 $value = Fraction::parse($w[1]);
594 $unit = $this->parseUnit($w[2])->quantity;
595 if ($command === 'TABLE') {
596 $selected = $this->matching($unit->dimension);
597 $lines = ["TABLE $value $w[2] dim=$unit->dimension count=" . count($selected)];
598 foreach ($selected as $row) $lines[] = 'ROW ' . self::format($value->mul($unit->value)->div($row->quantity->value), $row->symbol);
599 return implode("\n", $lines);
600 }
601 if ($command === 'AUTO') {
602 $basis = null;
603 foreach ($this->units as $candidate) {
604 if ($candidate->definition === null && $candidate->prefix && $candidate->quantity->dimension->same($unit->dimension)) { $basis = $candidate; break; }
605 }
606 if ($basis === null) throw new CommandError("NO PREFIXABLE UNIT $unit->dimension");
607 $scaled = $value->mul($unit->value)->div($basis->quantity->value); $magnitude = $scaled->magnitude();
608 $exponent = 0;
609 if ($magnitude->compare(new Fraction(1)) >= 0) {
610 while ($exponent < 9 && $magnitude->compare((new Fraction(10))->power($exponent + 3)) >= 0) $exponent += 3;
611 } elseif ($magnitude->n !== 0) {
612 while ($exponent > -9 && $magnitude->compare((new Fraction(10))->power($exponent)) < 0) $exponent -= 3;
613 }
614 $prefix = match ($exponent) { 9 => 'G', 6 => 'M', 3 => 'k', -3 => 'm', -6 => 'u', -9 => 'n', default => '' };
615 return "AUTO $value $w[2] = " . self::format($scaled->div((new Fraction(10))->power($exponent)), $prefix . $basis->symbol);
616 }
617 $parts = explode(',', $w[4]); $destinations = [];
618 foreach ($parts as $part) $destinations[] = $this->parseUnit($part)->quantity;
619 if ($value->n < 0) throw new CommandError('NEGATIVE VALUE');
620 foreach ($destinations as $destination) self::matchDimensions($unit->dimension, $destination->dimension);
621 for ($i = 1; $i < count($destinations); $i++) {
622 if ($destinations[$i - 1]->value->compare($destinations[$i]->value) <= 0) throw new CommandError('UNIT ORDER');
623 }
624 $remaining = $value->mul($unit->value); $pieces = [];
625 for ($i = 0; $i < count($parts) - 1; $i++) {
626 $quotient = $remaining->div($destinations[$i]->value);
627 $whole = intdiv($quotient->n, $quotient->d);
628 $remaining = $remaining->sub((new Fraction($whole))->mul($destinations[$i]->value));
629 $pieces[] = "$whole $parts[$i]";
630 }
631 $last = count($parts) - 1;
632 $pieces[] = self::format($remaining->div($destinations[$last]->value), $parts[$last]);
633 return "SPLIT $value $w[2] = " . implode(' ', $pieces);
634 case 'LET':
635 self::checkName($w[1], true);
636 $value = $this->evaluate((new QuantityParser($w[3]))->parse());
637 $this->variables[$w[1]] = $value;
638 return "LET $w[1] = " . $this->formatQuantity($value);
639 case 'EVAL':
640 $value = $this->evaluate((new QuantityParser($w[1]))->parse());
641 if (count($w) === 4) {
642 $unit = $this->parseUnit($w[3])->quantity;
643 self::matchDimensions($value->dimension, $unit->dimension);
644 return "EVAL $w[1] = " . self::format($value->value->div($unit->value), $w[3]);
645 }
646 return "EVAL $w[1] = " . $this->formatQuantity($value);
647 case 'VARS':
648 $names = array_keys($this->variables);
649 sort($names, SORT_STRING);
650 $lines = ['VARS count=' . count($names)];
651 foreach ($names as $name) $lines[] = "VAR $name = " . $this->formatQuantity($this->variables[$name]);
652 return implode("\n", $lines);
653 default:
654 throw new CommandError("UNKNOWN COMMAND $command");
655 }
656 }
657}
658
659$calculator = new Calculator();
660$count = (int)trim(fgets(STDIN));
661for ($i = 0; $i < $count; $i++) {
662 $line = rtrim(fgets(STDIN), "\r\n");
663 try { echo $calculator->execute($line), "\n"; }
664 catch (CommandError $error) { echo $error->getMessage(), "\n"; }
665}
