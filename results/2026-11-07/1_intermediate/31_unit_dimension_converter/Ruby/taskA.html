5Fraction = Data.define(:n, :d) do
6 include(Comparable)
7
8 def self.make(n, d = 1)
9 n, d = -n, -d if d.negative?
10 divisor = n.gcd(d)
11 new(n / divisor, d / divisor)
12 end
13
14 def +(b) = Fraction.make(n * b.d + b.n * d, d * b.d)
15 def -(b) = Fraction.make(n * b.d - b.n * d, d * b.d)
16 def *(b) = Fraction.make(n * b.n, d * b.d)
17 def /(b) = Fraction.make(n * b.d, d * b.n)
18 def -@ = Fraction.new(-n, d)
19 def abs = Fraction.new(n.abs, d)
20 def <=>(b) = (n * b.d) <=> (b.n * d)
21 def power(exponent)
22 factor = exponent.negative? ? Fraction.make(d, n) : self
23 result = Fraction::ONE
24 exponent.abs.times { result *= factor }
25 result
26 end

28 def to_s = d == 1 ? n.to_s : "#{n}/#{d}"
29 def decimal
30 magnitude = n.abs
31 format("%s%d.%06d", n.negative? ? "-" : "", magnitude / d, magnitude % d * 1_000_000 / d)
32 end

34 def notation(unit) = "#{self} #{unit} (#{decimal})"
35end

40Dimension = Data.define(:kg, :m, :s, :a, :k, :mol, :cd) do

45 def scale(factor) = Dimension.new(*exponents.map { |value| value * factor })
46 def half = Dimension.new(*exponents.map { |value| value / 2 })

49 raise CommandError, "DIMENSION MISMATCH #{self} vs #{other}" if self != other

54 terms = exponents.each_with_index.filter_map do |value, i|
55 next if value.zero?
56 value == 1 ? symbols[i] : "#{symbols[i]}^#{value}"
57 end
58
59 terms.empty? ? "1" : terms.join(" ")
60 end

99def letter?(c) = c && (c.between?(65, 90) || c.between?(97, 122))

103def number(text)
104 body = text.delete_prefix("-")
105 value = if digits?(body)
106 whole(Integer(body, 10))
107 elsif body.include?(".")
108 first, second = body.split(".", 2)
109 raise CommandError, "BAD NUMBER #{text}" unless digits?(first) && digits?(second)
110 scale = 1
111 second.length.times { scale *= 10 }
112 Fraction.make(Integer(first, 10) * scale + Integer(second, 10), scale)
113 elsif body.include?("/")
114 first, second = body.split("/", 2)
115 unless digits?(first) && digits?(second) && !Integer(second, 10).zero?
116 raise CommandError, "BAD NUMBER #{text}"
117 end
118
119 Fraction.make(Integer(first, 10), Integer(second, 10))
120 else
121 raise CommandError, "BAD NUMBER #{text}"
122 end
123
124 text.start_with?("-") ? -value : value
125end

152class Parser
153 def initialize(source, quantity)
154 @source, @tokens, @position = source, [], 0
155 i = 0
156 while i < source.length
157 start, c = i, source.getbyte(i)
158 kind = if letter?(c)
159 i += 1 while letter?(source.getbyte(i))
160 :identifier
161 elsif digit?(c)
162 i += 1 while digit?(source.getbyte(i))
163 if quantity && source.getbyte(i) == 46 && digit?(source.getbyte(i + 1))
164 i += 1
165 i += 1 while digit?(source.getbyte(i))
166 end
167
168 :number
169 elsif quantity && c == 91 && (finish = source.index("]", i + 1)) && finish > i + 1
170 i = finish + 1
171 :unit
172 else
173 i += 1
174 (quantity ? "+-*/^()," : "*/^()-").include?(c.chr) ? :symbol : :invalid
175 end
176
177 @tokens << Token.new(kind, source[start...i], start)
178 end
179
180 @tokens << Token.new(:end, "", source.length)
181 end
182
183 def current = @tokens[@position]
184 def advance
185 token = current
186 @position += 1
187 token
188 end
189
190 def take(text)
191 return false unless current.text == text
192 @position += 1
193 true
194 end
195
196 def syntax
197 raise CommandError, "SYNTAX #{@source} AT #{current.kind == :end ? "END" : current.position + 1}"
198 end
199
200 def expect(text)
201 syntax unless take(text)
202 end
203
204 def read_exponent
205 start = current.position
206 take("-")
207 syntax unless current.kind == :number && !current.text.include?(".")
208 token = advance
209 @source[start...(token.position + token.text.length)]
210 end
211
212 def unit_term
213 atom = if current.kind == :identifier
214 UnitName.new(advance.text)
215 elsif current.kind == :number && current.text[0] != "0"
216 UnitNumber.new(advance.text)
217 elsif take("(")
218 unit_expression.tap { expect(")") }
219 else
220 syntax
221 end
222
223 take("^") ? UnitPower.new(atom, read_exponent) : atom
224 end
225
226 def unit_expression
227 terms = [UnitTerm.new("*", unit_term)]
228 while %w[* /].include?(current.text)
229 op = advance.text
230 terms << UnitTerm.new(op, unit_term)
231 end
232
233 UnitProduct.new(terms.freeze)
234 end
235
236 def complete_unit
237 result = unit_expression
238 syntax unless current.kind == :end
239 result
240 end
241
242 def unit_string = advance.text[1...-1]
243 def primary
244 result = if current.kind == :number
245 text = advance.text
246 Literal.new(text, current.kind == :unit ? unit_string : nil)
247 elsif current.kind == :unit
248 Literal.new("1", unit_string)
249 elsif current.kind == :identifier
250 name = advance.text
251 if take("(")
252 args = [expression]
253 args << expression while take(",")
254 expect(")")
255 Call.new(name, args.freeze)
256 else
257 Variable.new(name)
258 end
259 elsif take("(")
260 expression.tap { expect(")") }
261 else
262 syntax
263 end
264
265 take("^") ? Power.new(result, read_exponent) : result
266 end
267
268 def unary = take("-") ? Negative.new(unary) : primary
269 def product
270 result = unary
271 while %w[* /].include?(current.text)
272 op = advance.text
273 result = Binary.new(op, result, unary)
274 end
275
276 result
277 end
278
279 def expression
280 result = product
281 while %w[+ -].include?(current.text)
282 op = advance.text
283 result = Binary.new(op, result, product)
284 end
285
286 result
287 end
288
289 def complete
290 result = expression
291 syntax unless current.kind == :end
292 result
293 end
294end

296Reduced = Data.define(:coefficient, :names)
297def reduce(node)
298 case node
299 when UnitName
300 Reduced.new(Fraction::ONE, {node.name => 1})
301 when UnitNumber
302 Reduced.new(whole(Integer(node.text, 10)), {})
303 when UnitPower
304 inner = reduce(node.child)
305 k = Integer(node.exponent, 10)
306 Reduced.new(inner.coefficient.power(k), inner.names.transform_values { |count| count * k })
307 when UnitProduct
308 coefficient, names = Fraction::ONE, {}
309 node.terms.each do |term|
310 right = reduce(term.node)
311 coefficient = term.op == "*" ? coefficient * right.coefficient : coefficient / right.coefficient
312 right.names.each { |name, count| names[name] = names.fetch(name, 0) + (term.op == "*" ? count : -count) }
313 end
314
315 Reduced.new(coefficient, names)
316 end
317end

319def reduced_text(tree)
320 result = reduce(tree)
321 parts = []
322 parts << result.coefficient.to_s if result.coefficient != Fraction::ONE
323 result.names.each do |name, count|
324 parts << (count == 1 ? name : "#{name}^#{count}") unless count.zero?
325 end
326
327 parts.empty? ? "1" : parts.join(" ")
328end

398 def resolve(name)
399 exact = @units.find { |entry| entry.symbol == name }
400 return exact.quantity if exact
401 PREFIXES.each do |prefix, factor|
402 next unless name.start_with?(prefix)
403 suffix = name[prefix.length..]
404 entry = @units.take(42).find { |candidate| candidate.prefix && candidate.symbol == suffix }
405 return Quantity.new(factor * entry.quantity.value, entry.quantity.dim) if entry

412 case node

415 when UnitPower

423 def unit_value(node)
424 case node
425 when UnitName
426 resolve(node.name)
427 when UnitNumber
428 Quantity.new(whole(Integer(node.text, 10)), Dimension::ZERO)
429 when UnitPower
430 value = unit_value(node.child)
431 k = Integer(node.exponent, 10)

433 when UnitProduct

435 node.terms.each do |term|
436 right = unit_value(term.node)
437 result = if term.op == "*"
438 Quantity.new(result.value * right.value, result.dim + right.dim)
439 else
440 Quantity.new(result.value / right.value, result.dim - right.dim)
441 end
442 end

444 result

456 def evaluate(expression)
457 case expression
458 when Literal
459 return Quantity.new(number(expression.number), Dimension::ZERO) unless expression.unit
460 from = unit(expression.unit)
461 Quantity.new(number(expression.number) * from.value, from.dim)
462 when Variable
463 @variables.fetch(expression.name) { raise CommandError, "UNKNOWN VARIABLE #{expression.name}" }
464 when Negative
465 value = evaluate(expression.child)
466 Quantity.new(-value.value, value.dim)
467 when Power
468 value = evaluate(expression.child)
469 k = exponent(expression.exponent)
470 raise CommandError, "DIVISION BY ZERO" if value.value.n.zero? && k.negative?
471 Quantity.new(value.value.power(k), value.dim.scale(k))
472 when Binary
473 a = evaluate(expression.left)
474 b = evaluate(expression.right)
475 case expression.op
476 when "+", "-"
477 a.dim.match(b.dim)
478 Quantity.new(expression.op == "+" ? a.value + b.value : a.value - b.value, a.dim)
479 when "*"
480 Quantity.new(a.value * b.value, a.dim + b.dim)
481 else
482 raise CommandError, "DIVISION BY ZERO" if b.value.n.zero?
483 Quantity.new(a.value / b.value, a.dim - b.dim)
484 end
485
486 when Call
487 arity = case expression.name
488 when "abs", "sqrt"
489 1
490 when "min", "max"
491 2
492 else
493 raise CommandError, "UNKNOWN FUNCTION #{expression.name}"
494 end
495
496 raise CommandError, "BAD ARGUMENTS #{expression.name}" unless expression.args.length == arity
497 a = evaluate(expression.args[0])
498 case expression.name
499 when "abs"
500 Quantity.new(a.value.abs, a.dim)
501 when "sqrt"
502 raise CommandError, "NEGATIVE ROOT" if a.value.n.negative?
503 raise CommandError, "ODD DIMENSION #{a.dim}" unless a.dim.even?
504 n, d = Integer.sqrt(a.value.n), Integer.sqrt(a.value.d)
505 raise CommandError, "NOT EXACT" if n * n != a.value.n || d * d != a.value.d
506 Quantity.new(Fraction.make(n, d), a.dim.half)
507 else
508 b = evaluate(expression.args[1])
509 a.dim.match(b.dim)
510 if expression.name == "min"
511 a.value <= b.value ? a : b
512 else
513 a.value >= b.value ? a : b
514 end
515 end
516 end
517 end

528 def command(line)
529 w = line.split(" ")
530 case w[0]
531 when "DEFINE"
532 check_name(w[1], false)
533 value = number(w[3])
534 from = unit(w[4])
535 raise CommandError, "NAME TAKEN #{w[1]}" if KINDS.key?(w[1]) || resolve(w[1])
536 raise CommandError, "NOT POSITIVE #{w[3]}" unless value.n.positive?
537 factor = value * from.value
538 @units << Unit.new(w[1].freeze, Quantity.new(factor, from.dim), false, Definition.new(value, w[4].freeze))
539 ["DEFINE #{w[1]} factor=#{factor} dim=#{from.dim} kind=#{kind_name(from.dim)}"]
540 when "UNDEF"
541 index = @units.index { |entry| entry.symbol == w[1] }
542 raise CommandError, "NOT DEFINED #{w[1]}" unless index
543 raise CommandError, "BUILTIN UNIT #{w[1]}" if index < 42
544 @units.delete_at(index)
545 ["UNDEF #{w[1]}"]
546 when "DEFS"
547 rows = @units.drop(42).map do |entry|
548 "DEF #{entry.symbol} = #{entry.definition.value} #{entry.definition.expression} factor=#{entry.quantity.value} dim=#{entry.quantity.dim}"
549 end
550
551 ["DEFS count=#{@units.length - 42}"] + rows
552 when "CONVERT", "INVERT"
553 value = number(w[1])
554 from = unit(w[2])
555 to = unit(w[4])
556 inverse = w[0] == "INVERT"
557 if inverse
558 if from.dim == Dimension::ZERO || from.dim + to.dim != Dimension::ZERO
559 raise CommandError, "NOT RECIPROCAL #{from.dim} vs #{to.dim}"
560 end
561
562 raise CommandError, "DIVISION BY ZERO" if value.n.zero?
563 else
564 from.dim.match(to.dim)
565 end
566
567 result = value * from.value
568 result = Fraction::ONE / result if inverse
569 result /= to.value
570 ["#{w[0]} #{value} #{w[2]} = #{result.notation(w[4])}"]
571 when "DIM", "REDUCE", "SIMPLIFY"
572 parsed = parse_unit(w[1])
573 from = parsed.quantity
574 result = case w[0]
575 when "DIM"
576 "#{from.dim} factor=#{from.value} kind=#{kind_name(from.dim)}"
577 when "REDUCE"
578 reduced_text(parsed.tree)
579 else
580 notation(from)
581 end
582
583 ["#{w[0]} #{w[1]} = #{result}"]
584 when "ADD", "SUB", "MUL", "DIV", "COMPARE"
585 a = number(w[1])
586 first = unit(w[2])
587 b = number(w[3])
588 second = unit(w[4])
589 target = w.length == 7
590 destination = target ? unit(w[6]) : first
591 if %w[ADD SUB COMPARE].include?(w[0])
592 first.dim.match(second.dim)
593 first.dim.match(destination.dim) if target
594 left = a * first.value
595 right = b * second.value
596 if w[0] == "COMPARE"
597 relation = ["<", "=", ">"][(left <=> right) + 1]
598 ratio = right.n.zero? ? "undefined" : (left / right).to_s
599 ["COMPARE #{a} #{w[2]} #{relation} #{b} #{w[4]} ratio=#{ratio}"]
600 else
601 add = w[0] == "ADD"
602 result = (add ? left + right : left - right) / destination.value
603 ["#{w[0]} #{a} #{w[2]} #{add ? "+" : "-"} #{b} #{w[4]} = #{result.notation(target ? w[6] : w[2])}"]
604 end
605 else
606 multiply = w[0] == "MUL"
607 dim = multiply ? first.dim + second.dim : first.dim - second.dim
608 raise CommandError, "DIVISION BY ZERO" if !multiply && b.n.zero?
609 dim.match(destination.dim) if target
610 raw = multiply ? a * b : a / b
611 left = a * first.value
612 right = b * second.value
613 coherent = multiply ? left * right : left / right
614 op = multiply ? "*" : "/"
615 prefix = "#{w[0]} #{a} #{w[2]} #{op} #{b} #{w[4]} = "
616 if target
617 [prefix + (coherent / destination.value).notation(w[6])]
618 else
619 right_unit = !multiply && (w[4].include?("*") || w[4].include?("/")) ? "(#{w[4]})" : w[4]
620 ["#{prefix}#{raw} #{w[2]}#{op}#{right_unit} = #{coherent.notation(simplified(dim))}"]
621 end
622 end
623
624 when "TEMP"
625 value = number(w[1])
626 [w[2], w[4]].each do |scale|
627 raise CommandError, "UNKNOWN SCALE #{scale}" unless %w[C F K R].include?(scale)
628 end
629
630 kelvin = case w[2]
631 when "C"
632 value + Fraction.make(5463, 20)
633 when "F"
634 (value + Fraction.make(45967, 100)) * whole(5) / whole(9)
635 when "R"
636 value * whole(5) / whole(9)
637 else
638 value
639 end
640
641 raise CommandError, "BELOW ABSOLUTE ZERO" if kelvin.n.negative?
642 result = case w[4]
643 when "C"
644 kelvin - Fraction.make(5463, 20)
645 when "F"
646 kelvin * whole(9) / whole(5) - Fraction.make(45967, 100)
647 when "R"
648 kelvin * whole(9) / whole(5)
649 else
650 kelvin
651 end
652
653 ["TEMP #{value} #{w[2]} = #{result.notation(w[4])}"]
654 when "LIST"
655 dim = KINDS[w[1]] || unit(w[1]).dim
656 matching = selected(dim)
657 rows = matching.map do |entry|
658 "UNIT #{entry.symbol} factor=#{entry.quantity.value} origin=#{entry.definition ? "user" : "builtin"} prefix=#{entry.prefix ? "yes" : "no"}"
659 end
660
661 ["LIST #{w[1]} dim=#{dim} kind=#{kind_name(dim)} count=#{matching.length}"] + rows
662 when "TABLE"
663 value = number(w[1])
664 from = unit(w[2])
665 matching = selected(from.dim)
666 rows = matching.map do |entry|
667 "ROW #{(value * from.value / entry.quantity.value).notation(entry.symbol)}"
668 end
669
670 ["TABLE #{value} #{w[2]} dim=#{from.dim} count=#{matching.length}"] + rows
671 when "AUTO"
672 value = number(w[1])
673 from = unit(w[2])
674 reference = @units.take(42).find { |entry| entry.prefix && entry.quantity.dim == from.dim }
675 raise CommandError, "NO PREFIXABLE UNIT #{from.dim}" unless reference
676 scaled = value * from.value / reference.quantity.value
677 magnitude, k = scaled.abs, 0
678 unless magnitude.n.zero?
679 if magnitude >= Fraction::ONE
680 k += 3 while k < 9 && magnitude >= whole(10).power(k + 3)
681 else
682 k -= 3 while k > -9 && magnitude < whole(10).power(k)
683 end
684 end
685
686 prefix = ["n", "u", "m", "", "k", "M", "G"][(k + 9) / 3]
687 ["AUTO #{value} #{w[2]} = #{(scaled / whole(10).power(k)).notation(prefix + reference.symbol)}"]
688 when "SPLIT"
689 value = number(w[1])
690 from = unit(w[2])
691 parts = w[4].split(",")
692 destinations = parts.map { |part| unit(part) }
693 raise CommandError, "NEGATIVE VALUE" if value.n.negative?
694 destinations.each { |entry| from.dim.match(entry.dim) }
695 raise CommandError, "UNIT ORDER" if destinations.each_cons(2).any? { |a, b| a.value <= b.value }
696 remaining = value * from.value
697 pieces = []
698 (parts.length - 1).times do |i|
699 quotient = remaining / destinations[i].value
700 count = quotient.n / quotient.d
701 remaining -= whole(count) * destinations[i].value
702 pieces << "#{count} #{parts[i]}"
703 end
704
705 pieces << (remaining / destinations.last.value).notation(parts.last)
706 ["SPLIT #{value} #{w[2]} = #{pieces.join(" ")}"]
707 when "LET"
708 check_name(w[1], true)
709 value = quantity(w[3])
710 @variables[w[1]] = value
711 ["LET #{w[1]} = #{notation(value)}"]
712 when "EVAL"
713 value = quantity(w[1])
714 result = if w.length == 4
715 destination = unit(w[3])
716 value.dim.match(destination.dim)
717 (value.value / destination.value).notation(w[3])
718 else
719 notation(value)
720 end
721
722 ["EVAL #{w[1]} = #{result}"]
723 when "VARS"
724 ["VARS count=#{@variables.length}"] +
725 @variables.keys.sort.map { |name| "VAR #{name} = #{notation(@variables.fetch(name))}" }
726 else
727 raise CommandError, "UNKNOWN COMMAND #{w[0]}"
728 end
729 end
730end
