6private class Fraction(numerator: Long, denominator: Long = 1) : Comparable<Fraction> {
7 val n: Long
8 val d: Long
9
10 init {
11 var a = abs(numerator)
12 var b = abs(denominator)
13 while (b != 0L) {
14 val remainder = a % b
15 a = b
16 b = remainder
17 }
18 val divisor = if (denominator < 0) -a else a
19 n = numerator / divisor
20 d = denominator / divisor
21 }
22
23 companion object {
24 val ONE = Fraction(1)
25 }
26
27 operator fun plus(b: Fraction) = Fraction(n * b.d + b.n * d, d * b.d)
28
29 operator fun minus(b: Fraction) = Fraction(n * b.d - b.n * d, d * b.d)
30
31 operator fun times(b: Fraction) = Fraction(n * b.n, d * b.d)
32
33 operator fun div(b: Fraction) = Fraction(n * b.d, d * b.n)
34
35 operator fun unaryMinus() = Fraction(-n, d)
36
37 fun absolute() = Fraction(abs(n), d)
38
39 fun power(exponent: Int): Fraction {
40 val factor = if (exponent < 0) Fraction(d, n) else this
41 var result = ONE
42 repeat(abs(exponent)) { result *= factor }
43 return result
44 }
45
46 override fun compareTo(other: Fraction) = (n * other.d).compareTo(other.n * d)
47
48 override fun equals(other: Any?) = other is Fraction && n == other.n && d == other.d
49
50 override fun hashCode() = 31 * n.hashCode() + d.hashCode()
51
52 override fun toString() = if (d == 1L) "$n" else "$n/$d"
53
54 fun decimal(): String {
55 val magnitude = abs(n)
56 val places = (magnitude % d * 1_000_000 / d).toString().padStart(6, '0')
57 return "${if (n < 0) "-" else ""}${magnitude / d}.$places"
58 }
59
60 fun notation(unit: String) = "$this $unit (${decimal()})"
61}

152private fun number(text: String): Fraction {
153 val body = text.removePrefix("-")
154 val value =
155 when {
156 digits(body) -> Fraction(body.toLong())
157 '.' in body -> {
158 val parts = body.split('.', limit = 2)
159 if (!digits(parts[0]) || !digits(parts[1])) throw CommandError("BAD NUMBER $text")
160 var scale = 1L
161 repeat(parts[1].length) { scale *= 10 }
162 Fraction(parts[0].toLong() * scale + parts[1].toLong(), scale)
163 }
164 '/' in body -> {
165 val parts = body.split('/', limit = 2)
166 if (!digits(parts[0]) || !digits(parts[1]) || parts[1].toLong() == 0L)
167 throw CommandError("BAD NUMBER $text")
168 Fraction(parts[0].toLong(), parts[1].toLong())
169 }
170 else -> throw CommandError("BAD NUMBER $text")
171 }
172 return if (text.startsWith('-')) -value else value
173}
174
175private fun exponent(text: String): Int {
176 val value = text.toLong()
177 if (value == 0L || value !in -3L..3L) throw CommandError("BAD EXPONENT $text")
178 return value.toInt()
179}

190private fun integerRoot(n: Long): Long {
191 if (n < 2) return n
192 var x = n
193 var next = (n + 1) / 2
194 while (next < x) {
195 x = next
196 next = (x + n / x) / 2
197 }
198 return x
199}

238private class Parser(private val source: String, quantity: Boolean) {
239 private val tokens = mutableListOf<Token>()
240 private var position = 0
241 private val current
242 get() = tokens[position]
243
244 init {
245 var i = 0
246 while (i < source.length) {
247 val start = i
248 val c = source[i]
249 val kind =
250 when {
251 letter(c) -> {
252 while (i < source.length && letter(source[i])) i++
253 TokenKind.IDENTIFIER
254 }
255 c in '0'..'9' -> {
256 while (i < source.length && source[i] in '0'..'9') i++
257 if (
258 quantity &&
259 i + 1 < source.length &&
260 source[i] == '.' &&
261 source[i + 1] in '0'..'9'
262 ) {
263 i++
264 while (i < source.length && source[i] in '0'..'9') i++
265 }
266 TokenKind.NUMBER
267 }
268 quantity && c == '[' && source.indexOf(']', i + 1) > i + 1 -> {
269 i = source.indexOf(']', i + 1) + 1
270 TokenKind.UNIT
271 }
272 else -> {
273 i++
274 if (c in (if (quantity) "+-*/^()," else "*/^()-")) TokenKind.SYMBOL
275 else TokenKind.INVALID
276 }
277 }
278 tokens += Token(kind, source.substring(start, i), start)
279 }
280 tokens += Token(TokenKind.END, "", source.length)
281 }
282
283 private fun take(text: String): Boolean {
284 if (current.text != text) return false
285 position++
286 return true
287 }
288
289 private fun advance() = tokens[position++]
290
291 private fun syntax() =
292 CommandError(
293 "SYNTAX $source AT ${if (current.kind == TokenKind.END) "END" else current.position+1}",
294 )
295
296 private fun expect(text: String) {
297 if (!take(text)) throw syntax()
298 }
299
300 private fun readExponent(): String {
301 val start = current.position
302 take("-")
303 if (current.kind != TokenKind.NUMBER || '.' in current.text) throw syntax()
304 val token = advance()
305 return source.substring(start, token.position + token.text.length)
306 }
307
308 private fun unitTerm(): UnitNode {
309 val atom =
310 when {
311 current.kind == TokenKind.IDENTIFIER -> UnitName(advance().text)
312 current.kind == TokenKind.NUMBER && current.text[0] != '0' ->
313 UnitNumber(advance().text)
314 take("(") -> unitExpression().also { expect(")") }
315 else -> throw syntax()
316 }
317 return if (take("^")) UnitPower(atom, readExponent()) else atom
318 }
319
320 private fun unitExpression(): UnitNode {
321 val terms = mutableListOf(UnitTerm('*', unitTerm()))
322 while (current.text == "*" || current.text == "/") {
323 val op = advance().text[0]
324 terms += UnitTerm(op, unitTerm())
325 }
326 return UnitProduct(terms)
327 }
328
329 fun completeUnit(): UnitNode {
330 val result = unitExpression()
331 if (current.kind != TokenKind.END) throw syntax()
332 return result
333 }
334
335 private fun unitString(): String = advance().text.let { it.substring(1, it.length - 1) }
336
337 private fun primary(): Expr {
338 val result =
339 when {
340 current.kind == TokenKind.NUMBER -> {
341 val text = advance().text
342 Literal(text, if (current.kind == TokenKind.UNIT) unitString() else null)
343 }
344 current.kind == TokenKind.UNIT -> Literal("1", unitString())
345 current.kind == TokenKind.IDENTIFIER -> {
346 val name = advance().text
347 if (take("(")) {
348 val args = mutableListOf(expression())
349 while (take(",")) args += expression()
350 expect(")")
351 Call(name, args)
352 } else Variable(name)
353 }
354 take("(") -> expression().also { expect(")") }
355 else -> throw syntax()
356 }
357 return if (take("^")) Power(result, readExponent()) else result
358 }
359
360 private fun unary(): Expr = if (take("-")) Negative(unary()) else primary()
361
362 private fun product(): Expr {
363 var result = unary()
364 while (current.text == "*" || current.text == "/") {
365 val op = advance().text[0]
366 result = Binary(op, result, unary())
367 }
368 return result
369 }
370
371 private fun expression(): Expr {
372 var result = product()
373 while (current.text == "+" || current.text == "-") {
374 val op = advance().text[0]
375 result = Binary(op, result, product())
376 }
377 return result
378 }
379
380 fun complete(): Expr {
381 val result = expression()
382 if (current.kind != TokenKind.END) throw syntax()
383 return result
384 }
385}

387private data class Reduced(val coefficient: Fraction, val names: LinkedHashMap<String, Long>)
388
389private fun reduce(node: UnitNode): Reduced =
390 when (node) {
391 is UnitName -> Reduced(Fraction.ONE, linkedMapOf(node.name to 1L))
392 is UnitNumber -> Reduced(Fraction(node.text.toLong()), linkedMapOf())
393 is UnitPower -> {
394 val result = reduce(node.child)
395 val k = node.exponent.toInt()
396 val coefficient = result.coefficient.power(k)
397 result.names.replaceAll { _, value -> value * k }
398 Reduced(coefficient, result.names)
399 }
400 is UnitProduct -> {
401 var coefficient = Fraction.ONE
402 val names = linkedMapOf<String, Long>()
403 for (term in node.terms) {
404 val right = reduce(term.node)
405 coefficient =
406 if (term.op == '*') coefficient * right.coefficient
407 else coefficient / right.coefficient
408 for ((name, count) in right.names) names[name] =
409 (names[name] ?: 0L) + if (term.op == '*') count else -count
410 }
411 Reduced(coefficient, names)
412 }
413 }
414
415private fun reducedText(tree: UnitNode): String {
416 val result = reduce(tree)
417 val parts = mutableListOf<String>()
418 if (result.coefficient != Fraction.ONE) parts += result.coefficient.toString()
419 for ((name, count) in result.names) if (count != 0L)
420 parts += if (count == 1L) name else "$name^$count"
421 return if (parts.isEmpty()) "1" else parts.joinToString(" ")
422}

500 private fun resolve(name: String): Quantity? {
501 units
502 .firstOrNull { it.symbol == name }
503 ?.let {
504 return it.quantity
505 }
506 for ((prefix, factor) in prefixes) {
507 if (!name.startsWith(prefix)) continue
508 val symbol = name.substring(prefix.length)
509 units
510 .take(42)
511 .firstOrNull { it.prefix && it.symbol == symbol }
512 ?.let {
513 return Quantity(factor * it.quantity.value, it.quantity.dim)
514 }
515 }
516 return null

519 private fun validateUnit(node: UnitNode) {
520 when (node) {
521 is UnitName ->
522 if (resolve(node.name) == null) throw CommandError("UNKNOWN UNIT ${node.name}")
523 is UnitNumber -> {}
524 is UnitPower -> {

526 exponent(node.exponent)

534 is UnitName -> resolve(node.name)!!
535 is UnitNumber -> Quantity(Fraction(node.text.toLong()), Dimension())
536 is UnitPower -> {

564 private fun evaluate(expression: Expr): Quantity =
565 when (expression) {
566 is Literal -> {
567 val text = expression.unit
568 if (text == null) Quantity(number(expression.number), Dimension())
569 else {
570 val parsed = unit(text)
571 Quantity(number(expression.number) * parsed.value, parsed.dim)
572 }
573 }
574 is Variable ->
575 variables[expression.name]
576 ?: throw CommandError("UNKNOWN VARIABLE ${expression.name}")
577 is Negative -> evaluate(expression.child).let { Quantity(-it.value, it.dim) }
578 is Power -> {
579 val value = evaluate(expression.child)
580 val k = exponent(expression.exponent)
581 if (value.value.n == 0L && k < 0) throw CommandError("DIVISION BY ZERO")
582 Quantity(value.value.power(k), value.dim.scale(k))
583 }
584 is Binary -> {
585 val a = evaluate(expression.left)
586 val b = evaluate(expression.right)
587 when (expression.op) {
588 '+',
589 '-' -> {
590 a.dim.match(b.dim)
591 Quantity(
592 if (expression.op == '+') a.value + b.value else a.value - b.value,
593 a.dim,
594 )
595 }
596 '*' -> Quantity(a.value * b.value, a.dim + b.dim)
597 else -> {
598 if (b.value.n == 0L) throw CommandError("DIVISION BY ZERO")
599 Quantity(a.value / b.value, a.dim - b.dim)
600 }
601 }
602 }
603 is Call -> {
604 val arity =
605 when (expression.name) {
606 "abs",
607 "sqrt" -> 1
608 "min",
609 "max" -> 2
610 else -> throw CommandError("UNKNOWN FUNCTION ${expression.name}")
611 }
612 if (expression.args.size != arity)
613 throw CommandError("BAD ARGUMENTS ${expression.name}")
614 val a = evaluate(expression.args[0])
615 when (expression.name) {
616 "abs" -> Quantity(a.value.absolute(), a.dim)
617 "sqrt" -> {
618 if (a.value.n < 0) throw CommandError("NEGATIVE ROOT")
619 if (!a.dim.even()) throw CommandError("ODD DIMENSION ${a.dim}")
620 val n = integerRoot(a.value.n)
621 val d = integerRoot(a.value.d)
622 if (n * n != a.value.n || d * d != a.value.d)
623 throw CommandError("NOT EXACT")
624 Quantity(Fraction(n, d), a.dim.half())
625 }
626 else -> {
627 val b = evaluate(expression.args[1])
628 a.dim.match(b.dim)
629 if (expression.name == "min") {
630 if (a.value <= b.value) a else b
631 } else {
632 if (a.value >= b.value) a else b
633 }
634 }
635 }
636 }

650 fun command(line: String): List<String> {
651 val w = line.split(' ')
652 return when (w[0]) {
653 "DEFINE" -> {
654 checkName(w[1], false)
655 val value = number(w[3])
656 val from = unit(w[4])
657 if (w[1] in kinds || resolve(w[1]) != null) throw CommandError("NAME TAKEN ${w[1]}")
658 if (value.n <= 0) throw CommandError("NOT POSITIVE ${w[3]}")
659 val factor = value * from.value
660 units += Unit(w[1], Quantity(factor, from.dim), false, Definition(value, w[4]))
661 listOf("DEFINE ${w[1]} factor=$factor dim=${from.dim} kind=${kindName(from.dim)}")
662 }
663 "UNDEF" -> {
664 val index = units.indexOfFirst { it.symbol == w[1] }
665 if (index < 0) throw CommandError("NOT DEFINED ${w[1]}")
666 if (index < 42) throw CommandError("BUILTIN UNIT ${w[1]}")
667 units.removeAt(index)
668 listOf("UNDEF ${w[1]}")
669 }
670 "DEFS" ->
671 listOf("DEFS count=${units.size-42}") +
672 units.drop(42).map {
673 val definition = it.definition!!
674 "DEF ${it.symbol} = ${definition.value} ${definition.expression} factor=${it.quantity.value} dim=${it.quantity.dim}"
675 }
676 "CONVERT",
677 "INVERT" -> {
678 val value = number(w[1])
679 val from = unit(w[2])
680 val to = unit(w[4])
681 val inverse = w[0] == "INVERT"
682 if (inverse) {
683 if (from.dim == Dimension() || from.dim + to.dim != Dimension())
684 throw CommandError("NOT RECIPROCAL ${from.dim} vs ${to.dim}")
685 if (value.n == 0L) throw CommandError("DIVISION BY ZERO")
686 } else from.dim.match(to.dim)
687 var result = value * from.value
688 if (inverse) result = Fraction.ONE / result
689 result /= to.value
690 listOf("${w[0]} $value ${w[2]} = ${result.notation(w[4])}")
691 }
692 "DIM",
693 "REDUCE",
694 "SIMPLIFY" -> {
695 val parsed = parseUnit(w[1])
696 val from = parsed.quantity
697 val result =
698 when (w[0]) {
699 "DIM" -> "${from.dim} factor=${from.value} kind=${kindName(from.dim)}"
700 "REDUCE" -> reducedText(parsed.tree)
701 else -> notation(from)
702 }
703 listOf("${w[0]} ${w[1]} = $result")
704 }
705 "ADD",
706 "SUB",
707 "MUL",
708 "DIV",
709 "COMPARE" -> {
710 val a = number(w[1])
711 val first = unit(w[2])
712 val b = number(w[3])
713 val second = unit(w[4])
714 val target = w.size == 7
715 val destination = if (target) unit(w[6]) else first
716 if (w[0] in listOf("ADD", "SUB", "COMPARE")) {
717 first.dim.match(second.dim)
718 if (target) first.dim.match(destination.dim)
719 val left = a * first.value
720 val right = b * second.value
721 if (w[0] == "COMPARE") {
722 val relation = listOf("<", "=", ">")[left.compareTo(right) + 1]
723 val ratio = if (right.n == 0L) "undefined" else (left / right).toString()
724 listOf("COMPARE $a ${w[2]} $relation $b ${w[4]} ratio=$ratio")
725 } else {
726 val add = w[0] == "ADD"
727 val result = (if (add) left + right else left - right) / destination.value
728 listOf(
729 "${w[0]} $a ${w[2]} ${if (add) "+" else "-"} $b ${w[4]} = ${result.notation(if (target) w[6] else w[2])}",
730 )
731 }
732 } else {
733 val multiply = w[0] == "MUL"
734 val dim = if (multiply) first.dim + second.dim else first.dim - second.dim
735 if (!multiply && b.n == 0L) throw CommandError("DIVISION BY ZERO")
736 if (target) dim.match(destination.dim)
737 val raw = if (multiply) a * b else a / b
738 val left = a * first.value
739 val right = b * second.value
740 val coherent = if (multiply) left * right else left / right
741 val op = if (multiply) "*" else "/"
742 val prefix = "${w[0]} $a ${w[2]} $op $b ${w[4]} = "
743 if (target) listOf(prefix + (coherent / destination.value).notation(w[6]))
744 else {
745 val rightUnit =
746 if (!multiply && ('*' in w[4] || '/' in w[4])) "(${w[4]})" else w[4]
747 listOf(
748 "$prefix$raw ${w[2]}$op$rightUnit = ${coherent.notation(simplified(dim))}",
749 )
750 }
751 }
752 }
753 "TEMP" -> {
754 val value = number(w[1])
755 for (scale in listOf(w[2], w[4])) if (scale !in listOf("C", "F", "K", "R"))
756 throw CommandError("UNKNOWN SCALE $scale")
757 val kelvin =
758 when (w[2]) {
759 "C" -> value + Fraction(5463, 20)
760 "F" -> (value + Fraction(45967, 100)) * Fraction(5) / Fraction(9)
761 "R" -> value * Fraction(5) / Fraction(9)
762 else -> value
763 }
764 if (kelvin.n < 0) throw CommandError("BELOW ABSOLUTE ZERO")
765 val result =
766 when (w[4]) {
767 "C" -> kelvin - Fraction(5463, 20)
768 "F" -> kelvin * Fraction(9) / Fraction(5) - Fraction(45967, 100)
769 "R" -> kelvin * Fraction(9) / Fraction(5)
770 else -> kelvin
771 }
772 listOf("TEMP $value ${w[2]} = ${result.notation(w[4])}")
773 }
774 "LIST" -> {
775 val dim = kinds[w[1]] ?: unit(w[1]).dim
776 val matching = selected(dim)
777 listOf("LIST ${w[1]} dim=$dim kind=${kindName(dim)} count=${matching.size}") +
778 matching.map {
779 "UNIT ${it.symbol} factor=${it.quantity.value} origin=${if (it.definition == null) "builtin" else "user"} prefix=${if (it.prefix) "yes" else "no"}"
780 }
781 }
782 "TABLE" -> {
783 val value = number(w[1])
784 val from = unit(w[2])
785 val matching = selected(from.dim)
786 listOf("TABLE $value ${w[2]} dim=${from.dim} count=${matching.size}") +
787 matching.map {
788 "ROW ${(value * from.value / it.quantity.value).notation(it.symbol)}"
789 }
790 }
791 "AUTO" -> {
792 val value = number(w[1])
793 val from = unit(w[2])
794 val reference =
795 units.take(42).firstOrNull { it.prefix && it.quantity.dim == from.dim }
796 ?: throw CommandError("NO PREFIXABLE UNIT ${from.dim}")
797 val scaled = value * from.value / reference.quantity.value
798 val magnitude = scaled.absolute()
799 var k = 0
800 if (magnitude.n != 0L) {
801 if (magnitude >= Fraction.ONE) {
802 while (k < 9 && magnitude >= Fraction(10).power(k + 3)) k += 3
803 } else {
804 while (k > -9 && magnitude < Fraction(10).power(k)) k -= 3
805 }
806 }
807 val prefix = listOf("n", "u", "m", "", "k", "M", "G")[(k + 9) / 3]
808 listOf(
809 "AUTO $value ${w[2]} = ${(scaled / Fraction(10).power(k)).notation(prefix+reference.symbol)}",
810 )
811 }
812 "SPLIT" -> {
813 val value = number(w[1])
814 val from = unit(w[2])
815 val parts = w[4].split(',')
816 val destinations = parts.map(::unit)
817 if (value.n < 0) throw CommandError("NEGATIVE VALUE")
818 destinations.forEach { from.dim.match(it.dim) }
819 if (destinations.zipWithNext().any { (a, b) -> a.value <= b.value })
820 throw CommandError("UNIT ORDER")
821 var remaining = value * from.value
822 val pieces = mutableListOf<String>()
823 for (i in 0 until parts.lastIndex) {
824 val quotient = remaining / destinations[i].value
825 val count = quotient.n / quotient.d
826 remaining -= Fraction(count) * destinations[i].value
827 pieces += "$count ${parts[i]}"
828 }
829 pieces += (remaining / destinations.last().value).notation(parts.last())
830 listOf("SPLIT $value ${w[2]} = ${pieces.joinToString(" ")}")
831 }
832 "LET" -> {
833 checkName(w[1], true)
834 val value = quantity(w[3])
835 variables[w[1]] = value
836 listOf("LET ${w[1]} = ${notation(value)}")
837 }
838 "EVAL" -> {
839 val value = quantity(w[1])
840 val result =
841 if (w.size == 4) {
842 val destination = unit(w[3])
843 value.dim.match(destination.dim)
844 (value.value / destination.value).notation(w[3])
845 } else notation(value)
846 listOf("EVAL ${w[1]} = $result")
847 }
848 "VARS" ->
849 listOf("VARS count=${variables.size}") +
850 variables.map { (name, value) -> "VAR $name = ${notation(value)}" }
851 else -> throw CommandError("UNKNOWN COMMAND ${w[0]}")
852 }
853 }
854}

856fun main() {
857 val input = System.`in`.bufferedReader()
858 val output = System.out.bufferedWriter()
859 val database = Database()
860 repeat(input.readLine().toInt()) {
861 val line = input.readLine()
862 try {
863 for (result in database.command(line)) output.write("$result\n")
864 } catch (error: CommandError) {
865 output.write("${error.message}\n")
866 }
867 }
868 output.flush()
869}
