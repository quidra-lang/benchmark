29decimal <- function(value) {
30 n <- abs(numerator(value))
31 d <- denominator(value)
32 whole <- n %/% d
33 tail <- ((n %% d) * 1000000L) %/% d
34 digits <- as.character(tail)
35 paste0(if (value < ZERO) "-" else "", as.character(whole), ".", strrep("0", 6L - nchar(digits)), digits)
36}

40parse_number <- function(text) {
41 if (!grepl("^-?[0-9]+([./][0-9]+)?$", text, perl = TRUE)) fail(paste("BAD NUMBER", text))
42 negative <- startsWith(text, "-")
43 body <- if (negative) substring(text, 2L) else text
44 if (grepl("/", body, fixed = TRUE)) {
45 parts <- strsplit(body, "/", fixed = TRUE)[[1L]]
46 divisor <- integer_text(parts[2L])
47 if (divisor == 0L) fail(paste("BAD NUMBER", text))
48 value <- as.bigq(integer_text(parts[1L]), divisor)
49 } else if (grepl(".", body, fixed = TRUE)) {
50 parts <- strsplit(body, ".", fixed = TRUE)[[1L]]
51 divisor <- integer_power(as.bigz(10L), nchar(parts[2L]))
52 value <- as.bigq(integer_text(parts[1L]) * divisor + integer_text(parts[2L]), divisor)
53 } else value <- as.bigq(integer_text(body))
54 if (negative) -value else value
55}

164tokenize <- function(source, quantity_mode) {
165 characters <- strsplit(source, "", fixed = TRUE)[[1L]]
166 count <- length(characters)
167 tokens <- list()
168 position <- 1L
169 while (position <= count) {
170 start <- position
171 character <- characters[position]
172 position <- position + 1L
173 kind <- "invalid"
174 if (is_letter(character)) {
175 kind <- "name"
176 while (position <= count && is_letter(characters[position])) position <- position + 1L
177 } else if (is_digit(character)) {
178 kind <- "number"
179 while (position <= count && is_digit(characters[position])) position <- position + 1L
180 if (quantity_mode && position < count && characters[position] == "." && is_digit(characters[position + 1L])) {
181 position <- position + 1L
182 while (position <= count && is_digit(characters[position])) position <- position + 1L
183 }
184 } else if (quantity_mode && character == "[") {
185 closing <- position
186 while (closing <= count && characters[closing] != "]") closing <- closing + 1L
187 if (closing > position && closing <= count) {
188 kind <- "unit"
189 position <- closing + 1L
190 }
191 } else if (character %in% strsplit(if (quantity_mode) "+-*/^()," else "*/^()-", "", fixed = TRUE)[[1L]]) kind <- "punctuation"
192 tokens[[length(tokens) + 1L]] <- list(kind = kind, text = substring(source, start, position - 1L), position = start)
193 }
194 tokens[[length(tokens) + 1L]] <- list(kind = "end", text = "", position = count + 1L)
195 parser <- new.env(parent = emptyenv())
196 parser$source <- source
197 parser$tokens <- tokens
198 parser$position <- 1L
199 parser
200}

220unit_term <- function(parser) {
221 token <- current(parser)
222 if (token$kind == "name") {
223 parser$position <- parser$position + 1L
224 result <- node("name", text = token$text)
225 } else if (token$kind == "number" && !startsWith(token$text, "0")) {
226 parser$position <- parser$position + 1L
227 result <- node("number", text = token$text)
228 } else if (take(parser, "(")) {
229 result <- unit_expression(parser)
230 expect(parser, ")")
231 } else syntax_error(parser)
232 if (take(parser, "^")) result <- node("power", left = result, text = parse_exponent(parser))
233 result

235unit_expression <- function(parser) {
236 first <- unit_term(parser)
237 result <- node("*", left = node("number", text = "1"), right = first)
238 while (current(parser)$text %in% c("*", "/")) {
239 operation <- current(parser)$text
240 parser$position <- parser$position + 1L
241 following <- unit_term(parser)
242 result <- node(operation, left = result, right = following)
243 }
244 result

246validate_unit <- function(database, tree) {
247 if (tree$kind == "name") {
248 resolved <- resolve(database, tree$text)
249 if (is.null(resolved)) fail(paste("UNKNOWN UNIT", tree$text))
250 tree$resolved <- resolved
251 } else if (tree$kind == "power") {
252 tree$left <- validate_unit(database, tree$left)
253 tree$exponent <- checked_exponent(tree$text)
254 } else if (tree$kind %in% c("*", "/")) {
255 tree$left <- validate_unit(database, tree$left)
256 tree$right <- validate_unit(database, tree$right)
257 }
258 tree
259}

266evaluate_unit <- function(tree) {
267 if (tree$kind == "number") return(quantity(as.bigq(integer_text(tree$text))))
268 if (tree$kind == "name") return(tree$resolved)
269 left <- evaluate_unit(tree$left)
270 if (tree$kind == "power") return(quantity(scalar_power(left$value, tree$exponent), left$dimension * tree$exponent))
271 right <- evaluate_unit(tree$right)
272 if (tree$kind == "*") quantity(left$value * right$value, left$dimension + right$dimension)
273 else quantity(left$value / right$value, left$dimension - right$dimension)
274}

276reduced <- function(tree) {
277 if (tree$kind == "number") return(list(coefficient = as.bigq(integer_text(tree$text)), names = character(), powers = as.bigz(integer())))
278 if (tree$kind == "name") return(list(coefficient = ONE, names = tree$text, powers = as.bigz(1L)))
279 left <- reduced(tree$left)
280 if (tree$kind == "power") {
281 left$coefficient <- scalar_power(left$coefficient, tree$exponent)
282 left$powers <- left$powers * tree$exponent
283 return(left)
284 }
285 right <- reduced(tree$right)
286 sign <- if (tree$kind == "*") 1L else -1L
287 left$coefficient <- if (sign == 1L) left$coefficient * right$coefficient else left$coefficient / right$coefficient
288 for (i in seq_along(right$names)) {
289 index <- match(right$names[i], left$names)
290 if (is.na(index)) {
291 left$names <- c(left$names, right$names[i])
292 left$powers <- c(left$powers, sign * right$powers[i])
293 } else left$powers[index] <- left$powers[index] + sign * right$powers[i]
294 }
295 left
296}
297reduced_text <- function(tree) {
298 result <- reduced(tree)
299 parts <- if (result$coefficient != ONE) fraction(result$coefficient) else character()
300 for (i in seq_along(result$names)) {
301 exponent <- result$powers[i]
302 if (exponent != 0L) parts <- c(parts, if (exponent == 1L) result$names[i] else paste0(result$names[i], "^", as.character(exponent)))
303 }
304 if (length(parts)) paste(parts, collapse = " ") else "1"
305}
306quantity_primary <- function(parser) {
307 token <- current(parser)
308 if (token$kind == "number") {
309 parser$position <- parser$position + 1L
310 if (current(parser)$kind == "unit") {
311 unit <- current(parser)$text
312 parser$position <- parser$position + 1L
313 result <- node("unit", text = token$text, unit = substring(unit, 2L, nchar(unit) - 1L))
314 } else result <- node("number", text = token$text)
315 } else if (token$kind == "unit") {
316 parser$position <- parser$position + 1L
317 result <- node("unit", text = "1", unit = substring(token$text, 2L, nchar(token$text) - 1L))
318 } else if (token$kind == "name") {
319 parser$position <- parser$position + 1L
320 if (take(parser, "(")) {
321 arguments <- list(quantity_expression(parser))
322 while (take(parser, ",")) arguments[[length(arguments) + 1L]] <- quantity_expression(parser)
323 expect(parser, ")")
324 result <- node("call", text = token$text, arguments = arguments)
325 } else result <- node("variable", text = token$text)
326 } else if (take(parser, "(")) {
327 result <- quantity_expression(parser)
328 expect(parser, ")")
329 } else syntax_error(parser)
330 if (take(parser, "^")) result <- node("power", left = result, text = parse_exponent(parser))
331 result
332}

356integer_sqrt <- function(value) {
357 if (value < 2L) return(value)
358 current_value <- value
359 following <- (value + 1L) %/% 2L
360 while (following < current_value) {
361 current_value <- following
362 following <- (current_value + value %/% current_value) %/% 2L
363 }
364 current_value
365}
366evaluate_quantity <- function(database, tree) {
367 if (tree$kind == "number") return(quantity(parse_number(tree$text)))
368 if (tree$kind == "unit") {
369 unit <- unit_value(database, tree$unit)
370 return(quantity(parse_number(tree$text) * unit$value, unit$dimension))
371 }
372 if (tree$kind == "variable") {
373 if (!exists(tree$text, database$variables, inherits = FALSE)) fail(paste("UNKNOWN VARIABLE", tree$text))
374 return(database$variables[[tree$text]])
375 }
376 if (tree$kind == "call") {
377 if (!tree$text %in% c("abs", "sqrt", "min", "max")) fail(paste("UNKNOWN FUNCTION", tree$text))
378 required <- if (tree$text %in% c("abs", "sqrt")) 1L else 2L
379 if (length(tree$arguments) != required) fail(paste("BAD ARGUMENTS", tree$text))
380 first <- evaluate_quantity(database, tree$arguments[[1L]])
381 if (tree$text == "abs") return(quantity(abs(first$value), first$dimension))
382 if (tree$text == "sqrt") {
383 if (first$value < ZERO) fail("NEGATIVE ROOT")
384 if (any(first$dimension %% 2L != 0L)) fail(paste("ODD DIMENSION", dimension_text(first$dimension)))
385 n <- numerator(first$value)
386 d <- denominator(first$value)
387 root_n <- integer_sqrt(n)
388 root_d <- integer_sqrt(d)
389 if (root_n * root_n != n || root_d * root_d != d) fail("NOT EXACT")
390 return(quantity(as.bigq(root_n, root_d), first$dimension %/% 2L))
391 }
392 second <- evaluate_quantity(database, tree$arguments[[2L]])
393 match_dimensions(first$dimension, second$dimension)
394 take_first <- if (tree$text == "min") first$value <= second$value else first$value >= second$value
395 return(if (take_first) first else second)
396 }
397 left <- evaluate_quantity(database, tree$left)
398 if (tree$kind == "negative") return(quantity(-left$value, left$dimension))
399 if (tree$kind == "power") {
400 exponent <- checked_exponent(tree$text)
401 if (left$value == ZERO && exponent < 0L) fail("DIVISION BY ZERO")
402 return(quantity(scalar_power(left$value, exponent), left$dimension * exponent))
403 }
404 right <- evaluate_quantity(database, tree$right)
405 if (tree$kind %in% c("+", "-")) {
406 match_dimensions(left$dimension, right$dimension)
407 value <- if (tree$kind == "+") left$value + right$value else left$value - right$value
408 return(quantity(value, left$dimension))
409 }
410 if (tree$kind == "*") return(quantity(left$value * right$value, left$dimension + right$dimension))
411 if (right$value == ZERO) fail("DIVISION BY ZERO")
412 quantity(left$value / right$value, left$dimension - right$dimension)
413}
414quantity_value <- function(database, source) {
415 parser <- tokenize(source, TRUE)
416 tree <- quantity_expression(parser)
417 if (current(parser)$kind != "end") syntax_error(parser)
418 evaluate_quantity(database, tree)
419}

421execute <- function(database, line) {
422 words <- strsplit(line, " ", fixed = TRUE)[[1L]]
423 command <- words[1L]
424 if (command == "DEFINE") {
425 name <- words[2L]
426 check_name(name)
427 value <- parse_number(words[4L])
428 unit <- unit_value(database, words[5L])
429 if (name %in% names(KINDS) || !is.null(resolve(database, name))) fail(paste("NAME TAKEN", name))
430 if (value <= ZERO) fail(paste("NOT POSITIVE", words[4L]))
431 factor <- value * unit$value
432 database$units[[length(database$units) + 1L]] <- list(
433 symbol = name, quantity = quantity(factor, unit$dimension), prefixable = FALSE,
434 definition = list(value = value, expression = words[5L]))
435 return(paste0("DEFINE ", name, " factor=", fraction(factor), " dim=", dimension_text(unit$dimension), " kind=", kind_name(unit$dimension)))
436 }
437 if (command == "UNDEF") {
438 name <- words[2L]
439 symbols <- vapply(database$units, function(unit) unit$symbol, character(1L))
440 index <- match(name, symbols)
441 if (!is.na(index) && index <= 42L) fail(paste("BUILTIN UNIT", name))
442 if (is.na(index)) fail(paste("NOT DEFINED", name))
443 database$units <- database$units[-index]
444 return(paste("UNDEF", name))
445 }
446 if (command == "DEFS") {
447 output <- paste0("DEFS count=", length(database$units) - 42L)
448 for (unit in database$units[-seq_len(42L)]) {
449 definition <- unit$definition
450 output <- c(output, paste0("DEF ", unit$symbol, " = ", fraction(definition$value), " ", definition$expression,
451 " factor=", fraction(unit$quantity$value), " dim=", dimension_text(unit$quantity$dimension)))
452 }
453 return(output)
454 }
455 if (command %in% c("CONVERT", "INVERT")) {
456 value <- parse_number(words[2L])
457 source <- unit_value(database, words[3L])
458 target <- unit_value(database, words[5L])
459 if (command == "INVERT") {
460 if (same_dimension(source$dimension, DIMENSIONLESS) || !same_dimension(source$dimension + target$dimension, DIMENSIONLESS)) {
461 fail(paste("NOT RECIPROCAL", dimension_text(source$dimension), "vs", dimension_text(target$dimension)))
462 }
463 if (value == ZERO) fail("DIVISION BY ZERO")
464 } else match_dimensions(source$dimension, target$dimension)
465 result <- value * source$value
466 if (command == "INVERT") result <- ONE / result
467 return(paste(command, fraction(value), words[3L], "=", notation(result / target$value, words[5L])))
468 }
469 if (command %in% c("DIM", "REDUCE", "SIMPLIFY")) {
470 tree <- parse_unit(database, words[2L])
471 unit <- evaluate_unit(tree)
472 result <- switch(command,
473 DIM = paste0(dimension_text(unit$dimension), " factor=", fraction(unit$value), " kind=", kind_name(unit$dimension)),
474 REDUCE = reduced_text(tree),
475 SIMPLIFY = notation(unit$value, simplified(database, unit$dimension)))
476 return(paste(command, words[2L], "=", result))
477 }
478 if (command %in% c("ADD", "SUB", "MUL", "DIV", "COMPARE")) {
479 a <- parse_number(words[2L])
480 first <- unit_value(database, words[3L])
481 b <- parse_number(words[4L])
482 second <- unit_value(database, words[5L])
483 has_target <- length(words) == 7L
484 target <- if (has_target) unit_value(database, words[7L]) else first
485 prefix <- paste(command, fraction(a), words[3L])
486 suffix <- paste(fraction(b), words[5L])
487 if (command %in% c("ADD", "SUB", "COMPARE")) {
488 match_dimensions(first$dimension, second$dimension)
489 if (has_target) match_dimensions(first$dimension, target$dimension)
490 left <- a * first$value
491 right <- b * second$value
492 if (command == "COMPARE") {
493 relation <- if (left < right) "<" else if (left > right) ">" else "="
494 ratio <- if (right == ZERO) "undefined" else fraction(left / right)
495 return(paste(prefix, relation, suffix, paste0("ratio=", ratio)))
496 }
497 operation <- if (command == "ADD") "+" else "-"
498 result <- if (command == "ADD") left + right else left - right
499 label <- if (has_target) words[7L] else words[3L]
500 return(paste(prefix, operation, suffix, "=", notation(result / target$value, label)))
501 }
502 multiply <- command == "MUL"
503 dimension <- first$dimension + second$dimension * (if (multiply) 1L else -1L)
504 if (!multiply && b == ZERO) fail("DIVISION BY ZERO")
505 if (has_target) match_dimensions(dimension, target$dimension)
506 raw <- if (multiply) a * b else a / b
507 left <- a * first$value
508 right <- b * second$value
509 coherent <- if (multiply) left * right else left / right
510 operation <- if (multiply) "*" else "/"
511 prefix <- paste(prefix, operation, suffix, "=")
512 if (has_target) return(paste(prefix, notation(coherent / target$value, words[7L])))
513 right_unit <- words[5L]
514 if (!multiply && grepl("[*/]", right_unit)) right_unit <- paste0("(", right_unit, ")")
515 return(paste(prefix, fraction(raw), paste0(words[3L], operation, right_unit), "=", notation(coherent, simplified(database, dimension))))
516 }
517 if (command == "TEMP") {
518 value <- parse_number(words[2L])
519 for (scale in words[c(3L, 5L)]) if (!scale %in% c("C", "F", "K", "R")) fail(paste("UNKNOWN SCALE", scale))
520 kelvin <- switch(words[3L], C = value + as.bigq(5463L, 20L),
521 F = (value + as.bigq(45967L, 100L)) * 5L / 9L,
522 R = value * 5L / 9L, K = value)
523 if (kelvin < ZERO) fail("BELOW ABSOLUTE ZERO")
524 result <- switch(words[5L], C = kelvin - as.bigq(5463L, 20L),
525 F = kelvin * 9L / 5L - as.bigq(45967L, 100L),
526 R = kelvin * 9L / 5L, K = kelvin)
527 return(paste("TEMP", fraction(value), words[3L], "=", notation(result, words[5L])))
528 }
529 if (command %in% c("LIST", "TABLE")) {
530 value <- ONE
531 unit <- quantity()
532 if (command == "TABLE") {
533 value <- parse_number(words[2L])
534 unit <- unit_value(database, words[3L])
535 dimension <- unit$dimension
536 } else if (words[2L] %in% names(KINDS)) dimension <- KINDS[[words[2L]]]
537 else dimension <- unit_value(database, words[2L])$dimension
538 selected <- selected_units(database, dimension)
539 if (command == "TABLE") {
540 output <- paste0("TABLE ", fraction(value), " ", words[3L], " dim=", dimension_text(dimension), " count=", length(selected))
541 } else output <- paste0("LIST ", words[2L], " dim=", dimension_text(dimension), " kind=", kind_name(dimension), " count=", length(selected))
542 for (item in selected) {
543 if (command == "TABLE") output <- c(output, paste("ROW", notation((value * unit$value) / item$quantity$value, item$symbol)))
544 else output <- c(output, paste0("UNIT ", item$symbol, " factor=", fraction(item$quantity$value),
545 " origin=", if (is.null(item$definition)) "builtin" else "user",
546 " prefix=", if (item$prefixable) "yes" else "no"))
547 }
548 return(output)
549 }
550 if (command == "AUTO") {
551 value <- parse_number(words[2L])
552 unit <- unit_value(database, words[3L])
553 reference <- NULL
554 for (item in database$units[seq_len(42L)]) {
555 if (item$prefixable && same_dimension(item$quantity$dimension, unit$dimension)) {
556 reference <- item
557 break
558 }
559 }
560 if (is.null(reference)) fail(paste("NO PREFIXABLE UNIT", dimension_text(unit$dimension)))
561 scaled <- (value * unit$value) / reference$quantity$value
562 magnitude <- abs(scaled)
563 exponent <- 0L
564 if (magnitude >= ONE) {
565 while (exponent < 9L && magnitude >= scalar_power(as.bigq(10L), exponent + 3L)) exponent <- exponent + 3L
566 } else if (magnitude > ZERO) {
567 while (exponent > -9L && magnitude < scalar_power(as.bigq(10L), exponent)) exponent <- exponent - 3L
568 }
569 prefix <- c("n", "u", "m", "", "k", "M", "G")[(exponent + 9L) %/% 3L + 1L]
570 return(paste("AUTO", fraction(value), words[3L], "=", notation(scaled / scalar_power(as.bigq(10L), exponent), paste0(prefix, reference$symbol))))
571 }
572 if (command == "SPLIT") {
573 value <- parse_number(words[2L])
574 unit <- unit_value(database, words[3L])
575 parts <- strsplit(words[5L], ",", fixed = TRUE)[[1L]]
576 targets <- lapply(parts, function(part) unit_value(database, part))
577 if (value < ZERO) fail("NEGATIVE VALUE")
578 for (target in targets) match_dimensions(unit$dimension, target$dimension)
579 for (i in seq_len(length(targets) - 1L)) if (targets[[i]]$value <= targets[[i + 1L]]$value) fail("UNIT ORDER")
580 remaining <- value * unit$value
581 pieces <- character()
582 for (i in seq_len(length(parts) - 1L)) {
583 quotient <- remaining / targets[[i]]$value
584 count <- numerator(quotient) %/% denominator(quotient)
585 remaining <- remaining - as.bigq(count) * targets[[i]]$value
586 pieces <- c(pieces, paste(as.character(count), parts[i]))
587 }
588 last <- length(parts)
589 pieces <- c(pieces, notation(remaining / targets[[last]]$value, parts[last]))
590 return(paste("SPLIT", fraction(value), words[3L], "=", paste(pieces, collapse = " ")))
591 }
592 if (command == "LET") {
593 check_name(words[2L], TRUE)
594 result <- quantity_value(database, words[4L])
595 database$variables[[words[2L]]] <- result
596 return(paste("LET", words[2L], "=", notation(result$value, simplified(database, result$dimension))))
597 }
598 if (command == "EVAL") {
599 result <- quantity_value(database, words[2L])
600 value <- result$value
601 label <- simplified(database, result$dimension)
602 if (length(words) == 4L) {
603 unit <- unit_value(database, words[4L])
604 match_dimensions(result$dimension, unit$dimension)
605 value <- value / unit$value
606 label <- words[4L]
607 }
608 return(paste("EVAL", words[2L], "=", notation(value, label)))
609 }
610 if (command == "VARS") {
611 names <- sort(ls(database$variables, all.names = TRUE), method = "radix")
612 output <- paste0("VARS count=", length(names))
613 for (name in names) {
614 result <- database$variables[[name]]
615 output <- c(output, paste("VAR", name, "=", notation(result$value, simplified(database, result$dimension))))
616 }
617 return(output)
618 }
619 fail(paste("UNKNOWN COMMAND", command))
620}

623 input <- file("stdin", open = "r")
624 on.exit(close(input))

629 output <- tryCatch(execute(database, line), command_error = function(problem) paste("ERROR", conditionMessage(problem)))
630 cat(paste0(output, "\n"), sep = "")
631 }
