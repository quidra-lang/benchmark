5local Fraction = {}
6Fraction.__index = Fraction
7local function fraction(n, d)
8 d = d or 1
9 if d < 0 then n, d = -n, -d end
10 local a, b = math.abs(n), d
11 while b ~= 0 do a, b = b, a % b end
12 return setmetatable({ n = n // a, d = d // a }, Fraction)
13end
14local ONE = fraction(1)
15function Fraction.__add(a, b) return fraction(a.n * b.d + b.n * a.d, a.d * b.d) end
16function Fraction.__sub(a, b) return fraction(a.n * b.d - b.n * a.d, a.d * b.d) end
17function Fraction.__mul(a, b) return fraction(a.n * b.n, a.d * b.d) end
18function Fraction.__div(a, b) return fraction(a.n * b.d, a.d * b.n) end
19function Fraction.__unm(a) return fraction(-a.n, a.d) end
20function Fraction.__eq(a, b) return a.n == b.n and a.d == b.d end
21function Fraction.__lt(a, b) return a.n * b.d < b.n * a.d end
22function Fraction.__le(a, b) return a.n * b.d <= b.n * a.d end
23function Fraction.__tostring(a)
24 return a.d == 1 and tostring(a.n) or string.format("%d/%d", a.n, a.d)
25end
26function Fraction:power(exponent)
27 local factor = exponent < 0 and ONE / self or self
28 local result = ONE
29 for _ = 1, math.abs(exponent) do result = result * factor end
30 return result
31end
32function Fraction:magnitude() return fraction(math.abs(self.n), self.d) end
33function Fraction:decimal()
34 local n = math.abs(self.n)
35 return string.format("%s%d.%06d", self.n < 0 and "-" or "", n // self.d, (n % self.d * 1000000) // self.d)
36end
37local function parse_number(text)
38 local sign, first, separator, second = text:match("^(%-?)([0-9]+)([./]?)([0-9]*)$")
39 if not sign or (separator == "") ~= (second == "") then fail("BAD NUMBER " .. text) end
40 local n, d = tonumber(first, 10), 1
41 if separator == "/" then
42 d = tonumber(second, 10)
43 if d == 0 then fail("BAD NUMBER " .. text) end
44 elseif separator == "." then
45 for _ = 1, #second do d = d * 10 end
46 n = n * d + tonumber(second, 10)
47 end
48 return fraction(sign == "-" and -n or n, d)
49end

161local Parser = {}
162Parser.__index = Parser
163local function letter(c) return c and (c >= 65 and c <= 90 or c >= 97 and c <= 122) end
164local function digit(c) return c and c >= 48 and c <= 57 end
165function Parser.new(source, quantity_mode)
166 local tokens, i = {}, 1
167 while i <= #source do
168 local start, c = i, source:byte(i)
169 i = i + 1
170 local kind
171 if letter(c) then
172 while letter(source:byte(i)) do i = i + 1 end
173 kind = "name"
174 elseif digit(c) then
175 while digit(source:byte(i)) do i = i + 1 end
176 if quantity_mode and source:sub(i, i) == "." and digit(source:byte(i + 1)) then
177 i = i + 2
178 while digit(source:byte(i)) do i = i + 1 end
179 end
180 kind = "number"
181 else
182 local ending = quantity_mode and c == 91 and source:find("]", i, true)
183 if ending and ending > i then
184 i, kind = ending + 1, "unit"
185 else
186 local symbols = quantity_mode and "+-*/^()," or "*/^()-"
187 kind = symbols:find(string.char(c), 1, true) and "symbol" or "invalid"
188 end
189 end
190 tokens[#tokens + 1] = { kind = kind, text = source:sub(start, i - 1), position = start }
191 end
192 tokens[#tokens + 1] = { kind = "end", text = "", position = #source + 1 }
193 return setmetatable({ source = source, tokens = tokens, index = 1 }, Parser)
194end
195function Parser:current() return self.tokens[self.index] end
196function Parser:advance()
197 local token = self:current()
198 self.index = self.index + 1
199 return token
200end
201function Parser:take(text)
202 if self:current().text ~= text then return false end
203 self.index = self.index + 1
204 return true
205end
206function Parser:syntax()
207 local token = self:current()
208 fail(string.format("SYNTAX %s AT %s", self.source, token.kind == "end" and "END" or token.position))
209end
210function Parser:expect(text) if not self:take(text) then self:syntax() end end
211function Parser:finish() if self:current().kind ~= "end" then self:syntax() end end
212function Parser:exponent()
213 local sign = self:take("-") and "-" or ""
214 local token = self:current()
215 if token.kind ~= "number" or token.text:find(".", 1, true) then self:syntax() end
216 return sign .. self:advance().text
217end
218function Parser:unit_expression()
219 local terms = { { op = "*", node = self:unit_term() } }
220 while self:current().text == "*" or self:current().text == "/" do
221 local op = self:advance().text
222 terms[#terms + 1] = { op = op, node = self:unit_term() }
223 end
224 return { kind = "product", terms = terms }
225end
226function Parser:unit_term()
227 local token, atom = self:current()
228 if token.kind == "name" then
229 atom = { kind = "name", name = self:advance().text }
230 elseif token.kind == "number" and token.text:sub(1, 1) ~= "0" then
231 atom = { kind = "number", text = self:advance().text }
232 elseif self:take("(") then
233 atom = self:unit_expression()
234 self:expect(")")
235 else self:syntax() end
236 if self:take("^") then return { kind = "power", atom = atom, exponent = self:exponent() } end
237 return atom
238end
239function Parser:quantity_expression()
240 local node = self:product()
241 while self:current().text == "+" or self:current().text == "-" do
242 local op = self:advance().text
243 node = { kind = "binary", op = op, left = node, right = self:product() }
244 end
245 return node
246end
247function Parser:product()
248 local node = self:unary()
249 while self:current().text == "*" or self:current().text == "/" do
250 local op = self:advance().text
251 node = { kind = "binary", op = op, left = node, right = self:unary() }
252 end
253 return node
254end
255function Parser:unary()
256 if self:take("-") then return { kind = "negative", operand = self:unary() } end
257 local node = self:primary()
258 if self:take("^") then return { kind = "power", base = node, exponent = self:exponent() } end
259 return node
260end
261function Parser:primary()
262 local token = self:current()
263 if token.kind == "number" then
264 local number, unit = self:advance().text
265 if self:current().kind == "unit" then unit = self:advance().text:sub(2, -2) end
266 return { kind = "literal", number = number, unit = unit }
267 elseif token.kind == "unit" then
268 return { kind = "literal", number = "1", unit = self:advance().text:sub(2, -2) }
269 elseif token.kind == "name" then
270 local name = self:advance().text
271 if not self:take("(") then return { kind = "variable", name = name } end
272 local arguments = { self:quantity_expression() }
273 while self:take(",") do arguments[#arguments + 1] = self:quantity_expression() end
274 self:expect(")")
275 return { kind = "call", name = name, arguments = arguments }
276 elseif self:take("(") then
277 local node = self:quantity_expression()
278 self:expect(")")
279 return node
280 end
281 self:syntax()
282end

286function Calculator.new()
287 local units = {}
288 for _, row in ipairs(BUILTIN_ROWS) do
289 local fields = words(row)
290 units[#units + 1] = { symbol = fields[1], quantity = quantity(parse_number(fields[2]), KINDS[fields[3]]), prefix = fields[4] == "yes" }
291 end
292 return setmetatable({ units = units, variables = {} }, Calculator)
293end
294function Calculator:resolve(name)
295 for _, unit in ipairs(self.units) do if unit.symbol == name then return unit.quantity end end
296 for _, prefix in ipairs(PREFIXES) do
297 for _, unit in ipairs(self.units) do
298 if not unit.definition and unit.prefix and name == (prefix[1] .. unit.symbol) then
299 return quantity(prefix[2] * unit.quantity.value, unit.quantity.dimension)
300 end
301 end
302 end
303end
304function Calculator:simplified(dim)
305 for _, unit in ipairs(self.units) do
306 if unit.quantity.value == ONE and unit.quantity.dimension == dim then return unit.symbol end
307 end
308 return tostring(dim)
309end
310function Calculator:format_quantity(value) return notation(value.value, self:simplified(value.dimension)) end
311function Calculator:validate_unit(node, names, seen)
312 if node.kind == "name" then
313 node.quantity = self:resolve(node.name)
314 if not node.quantity then fail("UNKNOWN UNIT " .. node.name) end
315 if not seen[node.name] then names[#names + 1], seen[node.name] = node.name, true end
316 elseif node.kind == "power" then
317 self:validate_unit(node.atom, names, seen)
318 check_exponent(node.exponent)
319 elseif node.kind == "product" then
320 for _, term in ipairs(node.terms) do self:validate_unit(term.node, names, seen) end
321 end
322end
323local function unit_value(node)
324 if node.kind == "name" then return node.quantity end
325 if node.kind == "number" then return quantity(fraction(tonumber(node.text, 10)), DIMENSIONLESS) end
326 if node.kind == "power" then
327 local atom, exponent = unit_value(node.atom), tonumber(node.exponent, 10)
328 return quantity(atom.value:power(exponent), atom.dimension:scale(exponent))
329 end
330 local value, dim = ONE, DIMENSIONLESS
331 for _, term in ipairs(node.terms) do
332 local next_value = unit_value(term.node)
333 if term.op == "*" then value, dim = value * next_value.value, dim + next_value.dimension
334 else value, dim = value / next_value.value, dim - next_value.dimension end
335 end
336 return quantity(value, dim)
337end
338function Calculator:parse_unit(text)
339 local parser = Parser.new(text, false)
340 local tree = parser:unit_expression()
341 parser:finish()
342 local names = {}
343 self:validate_unit(tree, names, {})
344 return { tree = tree, names = names, quantity = unit_value(tree) }
345end
346local function reduce(node)
347 if node.kind == "name" then return { coefficient = ONE, exponents = { [node.name] = 1 } } end
348 if node.kind == "number" then return { coefficient = fraction(tonumber(node.text, 10)), exponents = {} } end
349 if node.kind == "power" then
350 local atom, exponent = reduce(node.atom), tonumber(node.exponent, 10)
351 local powers = {}
352 for name, value in pairs(atom.exponents) do powers[name] = value * exponent end
353 return { coefficient = atom.coefficient:power(exponent), exponents = powers }
354 end
355 local coefficient, powers = ONE, {}
356 for _, term in ipairs(node.terms) do
357 local value = reduce(term.node)
358 if term.op == "*" then coefficient = coefficient * value.coefficient
359 else coefficient = coefficient / value.coefficient end
360 for name, power in pairs(value.exponents) do
361 powers[name] = (powers[name] or 0) + (term.op == "*" and power or -power)
362 end
363 end
364 return { coefficient = coefficient, exponents = powers }
365end
366local function reduced_text(parsed)
367 local value, parts = reduce(parsed.tree), {}
368 if value.coefficient ~= ONE then parts[#parts + 1] = tostring(value.coefficient) end
369 for _, name in ipairs(parsed.names) do
370 local exponent = value.exponents[name]
371 if exponent ~= 0 then parts[#parts + 1] = exponent == 1 and name or name .. "^" .. exponent end
372 end
373 return #parts == 0 and "1" or table.concat(parts, " ")
374end
375local function integer_root(number)
376 local low, high = 0, number + 1
377 while high - low > 1 do
378 local middle = low + (high - low) // 2
379 if middle <= number // middle then low = middle else high = middle end
380 end
381 return low
382end
383function Calculator:evaluate(node)
384 if node.kind == "literal" then
385 if not node.unit then return quantity(parse_number(node.number), DIMENSIONLESS) end
386 local unit = self:parse_unit(node.unit).quantity
387 return quantity(parse_number(node.number) * unit.value, unit.dimension)
388 elseif node.kind == "variable" then
389 local value = self.variables[node.name]
390 if not value then fail("UNKNOWN VARIABLE " .. node.name) end
391 return value
392 elseif node.kind == "negative" then
393 local operand = self:evaluate(node.operand)
394 return quantity(-operand.value, operand.dimension)
395 elseif node.kind == "power" then
396 local base = self:evaluate(node.base)
397 local exponent = check_exponent(node.exponent)
398 if base.value.n == 0 and exponent < 0 then fail("DIVISION BY ZERO") end
399 return quantity(base.value:power(exponent), base.dimension:scale(exponent))
400 elseif node.kind == "binary" then
401 local a = self:evaluate(node.left)
402 local b = self:evaluate(node.right)
403 if node.op == "+" or node.op == "-" then
404 match_dimensions(a.dimension, b.dimension)
405 return quantity(node.op == "+" and a.value + b.value or a.value - b.value, a.dimension)
406 elseif node.op == "*" then return quantity(a.value * b.value, a.dimension + b.dimension) end
407 if b.value.n == 0 then fail("DIVISION BY ZERO") end
408 return quantity(a.value / b.value, a.dimension - b.dimension)
409 end
410 local count = FUNCTIONS[node.name]
411 if not count then fail("UNKNOWN FUNCTION " .. node.name) end
412 if #node.arguments ~= count then fail("BAD ARGUMENTS " .. node.name) end
413 local a = self:evaluate(node.arguments[1])
414 if node.name == "abs" then return quantity(a.value:magnitude(), a.dimension) end
415 if node.name == "sqrt" then
416 if a.value.n < 0 then fail("NEGATIVE ROOT") end
417 if not a.dimension:is_even() then fail("ODD DIMENSION " .. tostring(a.dimension)) end
418 local p, q = integer_root(a.value.n), integer_root(a.value.d)
419 if p * p ~= a.value.n or q * q ~= a.value.d then fail("NOT EXACT") end
420 return quantity(fraction(p, q), a.dimension:half())
421 end
422 local b = self:evaluate(node.arguments[2])
423 match_dimensions(a.dimension, b.dimension)
424 if node.name == "min" then return a.value <= b.value and a or b end
425 return a.value >= b.value and a or b
426end
427function Calculator:parse_quantity(text)
428 local parser = Parser.new(text, true)
429 local tree = parser:quantity_expression()
430 parser:finish()
431 return self:evaluate(tree)
432end
433local function ascii_less(a, b)
434 for i = 1, math.min(#a, #b) do
435 if a:byte(i) ~= b:byte(i) then return a:byte(i) < b:byte(i) end
436 end
437 return #a < #b
438end
439function Calculator:matching(dim)
440 local selected = {}
441 for _, unit in ipairs(self.units) do
442 if unit.quantity.dimension == dim then selected[#selected + 1] = unit end
443 end
444 table.sort(selected, function(a, b)
445 if a.quantity.value ~= b.quantity.value then return a.quantity.value < b.quantity.value end
446 return ascii_less(a.symbol, b.symbol)
447 end)
448 return selected
449end

451function Calculator:execute(line)
452 local w = words(line)
453 local command = w[1]
454 if command == "DEFINE" then
455 check_name(w[2], false)
456 local value = parse_number(w[4])
457 local unit = self:parse_unit(w[5]).quantity
458 if KINDS[w[2]] or self:resolve(w[2]) then fail("NAME TAKEN " .. w[2]) end
459 if value.n <= 0 then fail("NOT POSITIVE " .. w[4]) end
460 local defined = quantity(value * unit.value, unit.dimension)
461 self.units[#self.units + 1] = { symbol = w[2], quantity = defined, prefix = false, definition = { value = value, expression = w[5] } }
462 return string.format("DEFINE %s factor=%s dim=%s kind=%s", w[2], defined.value, defined.dimension, kind_name(defined.dimension))
463 elseif command == "UNDEF" then
464 local index
465 for i, unit in ipairs(self.units) do if unit.symbol == w[2] then index = i break end end
466 if index and not self.units[index].definition then fail("BUILTIN UNIT " .. w[2]) end
467 if not index then fail("NOT DEFINED " .. w[2]) end
468 table.remove(self.units, index)
469 return "UNDEF " .. w[2]
470 elseif command == "DEFS" then
471 local lines = { "DEFS count=" .. (#self.units - #BUILTIN_ROWS) }
472 for _, unit in ipairs(self.units) do
473 if unit.definition then
474 lines[#lines + 1] = string.format("DEF %s = %s %s factor=%s dim=%s", unit.symbol, unit.definition.value,
475 unit.definition.expression, unit.quantity.value, unit.quantity.dimension)
476 end
477 end
478 return table.concat(lines, "\n")
479 elseif command == "CONVERT" or command == "INVERT" then
480 local value = parse_number(w[2])
481 local a = self:parse_unit(w[3]).quantity
482 local b = self:parse_unit(w[5]).quantity
483 if command == "CONVERT" then match_dimensions(a.dimension, b.dimension)
484 else
485 if a.dimension == DIMENSIONLESS or a.dimension + b.dimension ~= DIMENSIONLESS then
486 fail(string.format("NOT RECIPROCAL %s vs %s", a.dimension, b.dimension))
487 end
488 if value.n == 0 then fail("DIVISION BY ZERO") end
489 end
490 local result = command == "CONVERT" and (value * a.value) / b.value or (ONE / (value * a.value)) / b.value
491 return string.format("%s %s %s = %s", command, value, w[3], notation(result, w[5]))
492 elseif command == "DIM" or command == "REDUCE" or command == "SIMPLIFY" then
493 local parsed = self:parse_unit(w[2])
494 local unit = parsed.quantity
495 if command == "DIM" then return string.format("DIM %s = %s factor=%s kind=%s", w[2], unit.dimension, unit.value, kind_name(unit.dimension)) end
496 if command == "REDUCE" then return "REDUCE " .. w[2] .. " = " .. reduced_text(parsed) end
497 return "SIMPLIFY " .. w[2] .. " = " .. self:format_quantity(unit)
498 elseif command == "ADD" or command == "SUB" or command == "MUL" or command == "DIV" or command == "COMPARE" then
499 local x = parse_number(w[2])
500 local a = self:parse_unit(w[3]).quantity
501 local y = parse_number(w[4])
502 local b = self:parse_unit(w[5]).quantity
503 local target = #w == 7 and self:parse_unit(w[7]).quantity or nil
504 if command == "COMPARE" then
505 match_dimensions(a.dimension, b.dimension)
506 local left, right = x * a.value, y * b.value
507 local relation = left < right and "<" or left > right and ">" or "="
508 local ratio = right.n == 0 and "undefined" or tostring(left / right)
509 return string.format("COMPARE %s %s %s %s %s ratio=%s", x, w[3], relation, y, w[5], ratio)
510 end
511 local op = ({ ADD = "+", SUB = "-", MUL = "*", DIV = "/" })[command]
512 local prefix = string.format("%s %s %s %s %s %s = ", command, x, w[3], op, y, w[5])
513 if command == "ADD" or command == "SUB" then
514 match_dimensions(a.dimension, b.dimension)
515 if target then match_dimensions(a.dimension, target.dimension) end
516 local left, right = x * a.value, y * b.value
517 local sum = command == "ADD" and left + right or left - right
518 return prefix .. notation(sum / (target and target.value or a.value), target and w[7] or w[3])
519 end
520 local dim = command == "MUL" and a.dimension + b.dimension or a.dimension - b.dimension
521 if command == "DIV" and y.n == 0 then fail("DIVISION BY ZERO") end
522 if target then match_dimensions(dim, target.dimension) end
523 local product = command == "MUL" and x * y or x / y
524 local first, second = x * a.value, y * b.value
525 local coherent = command == "MUL" and first * second or first / second
526 if target then return prefix .. notation(coherent / target.value, w[7]) end
527 local right_unit = w[5]
528 if command == "DIV" and (right_unit:find("*", 1, true) or right_unit:find("/", 1, true)) then right_unit = "(" .. right_unit .. ")" end
529 return string.format("%s%s %s%s%s = %s", prefix, product, w[3], op, right_unit, notation(coherent, self:simplified(dim)))
530 elseif command == "TEMP" then
531 local value = parse_number(w[2])
532 local scales = { C = true, F = true, K = true, R = true }
533 if not scales[w[3]] then fail("UNKNOWN SCALE " .. w[3]) end
534 if not scales[w[5]] then fail("UNKNOWN SCALE " .. w[5]) end
535 local kelvin = value
536 if w[3] == "C" then kelvin = value + fraction(5463, 20)
537 elseif w[3] == "F" then kelvin = ((value + fraction(45967, 100)) * fraction(5)) / fraction(9)
538 elseif w[3] == "R" then kelvin = (value * fraction(5)) / fraction(9) end
539 if kelvin.n < 0 then fail("BELOW ABSOLUTE ZERO") end
540 local result = kelvin
541 if w[5] == "C" then result = kelvin - fraction(5463, 20)
542 elseif w[5] == "F" then result = ((kelvin * fraction(9)) / fraction(5)) - fraction(45967, 100)
543 elseif w[5] == "R" then result = (kelvin * fraction(9)) / fraction(5) end
544 return string.format("TEMP %s %s = %s", value, w[3], notation(result, w[5]))
545 elseif command == "LIST" then
546 local dim = KINDS[w[2]] or self:parse_unit(w[2]).quantity.dimension
547 local selected = self:matching(dim)
548 local lines = { string.format("LIST %s dim=%s kind=%s count=%d", w[2], dim, kind_name(dim), #selected) }
549 for _, unit in ipairs(selected) do
550 lines[#lines + 1] = string.format("UNIT %s factor=%s origin=%s prefix=%s", unit.symbol, unit.quantity.value,
551 unit.definition and "user" or "builtin", unit.prefix and "yes" or "no")
552 end
553 return table.concat(lines, "\n")
554 elseif command == "TABLE" or command == "AUTO" or command == "SPLIT" then
555 local value = parse_number(w[2])
556 local unit = self:parse_unit(w[3]).quantity
557 if command == "TABLE" then
558 local selected = self:matching(unit.dimension)
559 local lines = { string.format("TABLE %s %s dim=%s count=%d", value, w[3], unit.dimension, #selected) }
560 for _, row in ipairs(selected) do lines[#lines + 1] = "ROW " .. notation((value * unit.value) / row.quantity.value, row.symbol) end
561 return table.concat(lines, "\n")
562 elseif command == "AUTO" then
563 local basis
564 for _, candidate in ipairs(self.units) do
565 if not candidate.definition and candidate.prefix and candidate.quantity.dimension == unit.dimension then basis = candidate break end
566 end
567 if not basis then fail("NO PREFIXABLE UNIT " .. tostring(unit.dimension)) end
568 local scaled = (value * unit.value) / basis.quantity.value
569 local magnitude, exponent, ten = scaled:magnitude(), 0, fraction(10)
570 if magnitude >= ONE then
571 while exponent < 9 and magnitude >= ten:power(exponent + 3) do exponent = exponent + 3 end
572 elseif magnitude.n ~= 0 then
573 while exponent > -9 and magnitude < ten:power(exponent) do exponent = exponent - 3 end
574 end
575 local prefixes = { [9] = "G", [6] = "M", [3] = "k", [0] = "", [-3] = "m", [-6] = "u", [-9] = "n" }
576 return string.format("AUTO %s %s = %s", value, w[3], notation(scaled / ten:power(exponent), prefixes[exponent] .. basis.symbol))
577 end
578 local parts, destinations = words(w[5], "[^,]+"), {}
579 for _, part in ipairs(parts) do destinations[#destinations + 1] = self:parse_unit(part).quantity end
580 if value.n < 0 then fail("NEGATIVE VALUE") end
581 for _, destination in ipairs(destinations) do match_dimensions(unit.dimension, destination.dimension) end
582 for i = 2, #destinations do
583 if destinations[i - 1].value <= destinations[i].value then fail("UNIT ORDER") end
584 end
585 local remaining, pieces = value * unit.value, {}
586 for i = 1, #parts - 1 do
587 local quotient = remaining / destinations[i].value
588 local whole = quotient.n // quotient.d
589 remaining = remaining - fraction(whole) * destinations[i].value
590 pieces[#pieces + 1] = string.format("%d %s", whole, parts[i])
591 end
592 pieces[#pieces + 1] = notation(remaining / destinations[#parts].value, parts[#parts])
593 return string.format("SPLIT %s %s = %s", value, w[3], table.concat(pieces, " "))
594 elseif command == "LET" then
595 check_name(w[2], true)
596 local value = self:parse_quantity(w[4])
597 self.variables[w[2]] = value
598 return "LET " .. w[2] .. " = " .. self:format_quantity(value)
599 elseif command == "EVAL" then
600 local value = self:parse_quantity(w[2])
601 if #w == 4 then
602 local unit = self:parse_unit(w[4]).quantity
603 match_dimensions(value.dimension, unit.dimension)
604 return "EVAL " .. w[2] .. " = " .. notation(value.value / unit.value, w[4])
605 end
606 return "EVAL " .. w[2] .. " = " .. self:format_quantity(value)
607 elseif command == "VARS" then
608 local names = {}
609 for name in pairs(self.variables) do names[#names + 1] = name end
610 table.sort(names, ascii_less)
611 local lines = { "VARS count=" .. #names }
612 for _, name in ipairs(names) do lines[#lines + 1] = "VAR " .. name .. " = " .. self:format_quantity(self.variables[name]) end
613 return table.concat(lines, "\n")
614 end
615 fail("UNKNOWN COMMAND " .. command)
616end
