8readonly record struct Fraction : IComparable<Fraction>
9{
10 public long Numerator { get; }
11 public long Denominator { get; }
12 public static Fraction One => new(1);
13 public bool IsZero => Numerator == 0;
14
15 public Fraction(long numerator, long denominator = 1)
16 {
17 if (denominator < 0) { numerator = -numerator; denominator = -denominator; }
18 long a = Math.Abs(numerator), b = denominator;
19 while (b != 0) (a, b) = (b, a % b);
20 Numerator = numerator / a;
21 Denominator = denominator / a;
22 }
23
24 public static Fraction Parse(string text)
25 {
26 bool negative = text.StartsWith('-');
27 string body = negative ? text[1..] : text;
28 int separator = body.IndexOfAny(['.', '/']);
29 string first = separator < 0 ? body : body[..separator];
30 string second = separator < 0 ? "1" : body[(separator + 1)..];
31 if (!Digits(first) || !Digits(second)) throw new CommandError($"BAD NUMBER {text}");
32 long numerator = long.Parse(first), denominator = 1;
33 if (separator >= 0 && body[separator] == '/') {
34 denominator = long.Parse(second);
35 if (denominator == 0) throw new CommandError($"BAD NUMBER {text}");
36 } else if (separator >= 0) {
37 for (int i = 0; i < second.Length; i++) denominator *= 10;
38 numerator = numerator * denominator + long.Parse(second);
39 }
40 return new(negative ? -numerator : numerator, denominator);
41 }
42
43 private static bool Digits(string text) => text.Length > 0 && text.All(c => c is >= '0' and <= '9');
44 public static Fraction operator +(Fraction a, Fraction b) =>
45 new(a.Numerator * b.Denominator + b.Numerator * a.Denominator, a.Denominator * b.Denominator);
46 public static Fraction operator -(Fraction a, Fraction b) =>
47 new(a.Numerator * b.Denominator - b.Numerator * a.Denominator, a.Denominator * b.Denominator);
48 public static Fraction operator *(Fraction a, Fraction b) =>
49 new(a.Numerator * b.Numerator, a.Denominator * b.Denominator);
50 public static Fraction operator /(Fraction a, Fraction b) =>
51 new(a.Numerator * b.Denominator, a.Denominator * b.Numerator);
52 public static Fraction operator -(Fraction a) => new(-a.Numerator, a.Denominator);
53 public Fraction Abs() => new(Math.Abs(Numerator), Denominator);
54 public int CompareTo(Fraction other) =>
55 (Numerator * other.Denominator).CompareTo(other.Numerator * Denominator);
56 public Fraction Power(int exponent)
57 {
58 Fraction factor = exponent < 0 ? One / this : this, result = One;
59 for (int i = 0; i < Math.Abs(exponent); i++) result *= factor;
60 return result;
61 }
62 public string Decimal()
63 {
64 long magnitude = Math.Abs(Numerator);
65 return $"{(Numerator < 0 ? "-" : "")}{magnitude / Denominator}.{magnitude % Denominator * 1_000_000 / Denominator:D6}";
66 }
67 public override string ToString() => Denominator == 1 ? $"{Numerator}" : $"{Numerator}/{Denominator}";
68}

70readonly record struct Dimension(long Kg = 0, long M = 0, long S = 0, long A = 0,
71 long K = 0, long Mol = 0, long Cd = 0)
72{
73 public bool IsZero => this == new Dimension();
74 public bool IsEven => Exponents().All(value => value % 2 == 0);
75 private long[] Exponents() => [Kg, M, S, A, K, Mol, Cd];
76 public static Dimension operator +(Dimension a, Dimension b) =>
77 new(a.Kg + b.Kg, a.M + b.M, a.S + b.S, a.A + b.A, a.K + b.K, a.Mol + b.Mol, a.Cd + b.Cd);
78 public static Dimension operator -(Dimension a, Dimension b) =>
79 new(a.Kg - b.Kg, a.M - b.M, a.S - b.S, a.A - b.A, a.K - b.K, a.Mol - b.Mol, a.Cd - b.Cd);
80 public Dimension Scale(long k) => new(Kg * k, M * k, S * k, A * k, K * k, Mol * k, Cd * k);
81 public Dimension Half() => new(Kg / 2, M / 2, S / 2, A / 2, K / 2, Mol / 2, Cd / 2);
82 public override string ToString()
83 {
84 string[] symbols = ["kg", "m", "s", "A", "K", "mol", "cd"];
85 long[] values = Exponents();
86 List<string> parts = [];
87 for (int i = 0; i < values.Length; i++)
88 if (values[i] != 0) parts.Add(values[i] == 1 ? symbols[i] : $"{symbols[i]}^{values[i]}");
89 return parts.Count == 0 ? "1" : string.Join(' ', parts);
90 }
91}

97enum TokenKind { Number, Identifier, Unit, Symbol, Invalid, End }
98readonly record struct Token(TokenKind Kind, string Text, int Position);
99
100abstract class Parser
101{
102 private readonly string source;
103 private readonly List<Token> tokens = [];
104 private int index;
105 protected Token Current => tokens[index];
106
107 protected Parser(string source, bool quantity)
108 {
109 this.source = source;
110 for (int i = 0; i < source.Length;) {
111 int start = i++;
112 char c = source[start];
113 TokenKind kind;
114 if (IsLetter(c)) {
115 while (i < source.Length && IsLetter(source[i])) i++;
116 kind = TokenKind.Identifier;
117 } else if (IsDigit(c)) {
118 while (i < source.Length && IsDigit(source[i])) i++;
119 if (quantity && i + 1 < source.Length && source[i] == '.' && IsDigit(source[i + 1])) {
120 i += 2;
121 while (i < source.Length && IsDigit(source[i])) i++;
122 }
123 kind = TokenKind.Number;
124 } else if (quantity && c == '[' && source.IndexOf(']', i) is int end && end > i) {
125 i = end + 1;
126 kind = TokenKind.Unit;
127 } else kind = (quantity ? "+-*/^()," : "*/^()-").Contains(c) ? TokenKind.Symbol : TokenKind.Invalid;
128 tokens.Add(new(kind, source[start..i], start + 1));
129 }
130 tokens.Add(new(TokenKind.End, "", source.Length + 1));
131 }
132
133 public static bool IsLetter(char c) => c is >= 'A' and <= 'Z' or >= 'a' and <= 'z';
134 private static bool IsDigit(char c) => c is >= '0' and <= '9';
135 protected Token Advance() => tokens[index++];
136 protected bool Take(string text)
137 {
138 if (Current.Text != text) return false;
139 index++;
140 return true;
141 }
142 protected void Expect(string text) { if (!Take(text)) throw Syntax(); }
143 protected CommandError Syntax() => new($"SYNTAX {source} AT {(Current.Kind == TokenKind.End ? "END" : Current.Position.ToString())}");
144 protected void Finish() { if (Current.Kind != TokenKind.End) throw Syntax(); }
145 protected string Exponent()
146 {
147 string sign = Take("-") ? "-" : "";
148 if (Current.Kind != TokenKind.Number || Current.Text.Contains('.')) throw Syntax();
149 return sign + Advance().Text;
150 }
151}

153abstract record UnitNode;
154sealed record UnitName(string Name) : UnitNode;
155sealed record UnitNumber(string Text) : UnitNode;
156sealed record UnitPower(UnitNode Atom, string Exponent) : UnitNode;
157sealed record UnitProduct(IReadOnlyList<(char Operator, UnitNode Term)> Terms) : UnitNode;
158
159sealed class UnitParser(string source) : Parser(source, false)
160{
161 public UnitNode Parse() { UnitNode result = Expression(); Finish(); return result; }
162 private UnitNode Expression()
163 {
164 List<(char, UnitNode)> terms = [('*', Term())];
165 while (Current.Text is "*" or "/") {
166 char op = Advance().Text[0];
167 terms.Add((op, Term()));
168 }
169 return new UnitProduct(terms);
170 }
171 private UnitNode Term()
172 {
173 UnitNode atom;
174 if (Current.Kind == TokenKind.Identifier) atom = new UnitName(Advance().Text);
175 else if (Current.Kind == TokenKind.Number && Current.Text[0] != '0') atom = new UnitNumber(Advance().Text);
176 else if (Take("(")) { atom = Expression(); Expect(")"); }
177 else throw Syntax();
178 return Take("^") ? new UnitPower(atom, Exponent()) : atom;
179 }
180}

182abstract record QuantityNode;
183sealed record Literal(string Number, string? Unit) : QuantityNode;
184sealed record Variable(string Name) : QuantityNode;
185sealed record Binary(char Operator, QuantityNode Left, QuantityNode Right) : QuantityNode;
186sealed record Power(QuantityNode Base, string Exponent) : QuantityNode;
187sealed record Negative(QuantityNode Operand) : QuantityNode;
188sealed record Call(string Name, IReadOnlyList<QuantityNode> Arguments) : QuantityNode;
189
190sealed class QuantityParser(string source) : Parser(source, true)
191{
192 public QuantityNode Parse() { QuantityNode result = Expression(); Finish(); return result; }
193 private QuantityNode Expression()
194 {
195 QuantityNode node = Product();
196 while (Current.Text is "+" or "-") {
197 char op = Advance().Text[0];
198 node = new Binary(op, node, Product());
199 }
200 return node;
201 }
202 private QuantityNode Product()
203 {
204 QuantityNode node = Unary();
205 while (Current.Text is "*" or "/") {
206 char op = Advance().Text[0];
207 node = new Binary(op, node, Unary());
208 }
209 return node;
210 }
211 private QuantityNode Unary()
212 {
213 if (Take("-")) return new Negative(Unary());
214 QuantityNode node = Primary();
215 return Take("^") ? new Power(node, Exponent()) : node;
216 }
217 private QuantityNode Primary()
218 {
219 if (Current.Kind == TokenKind.Number) {
220 string number = Advance().Text;
221 return new Literal(number, Current.Kind == TokenKind.Unit ? Advance().Text[1..^1] : null);
222 }
223 if (Current.Kind == TokenKind.Unit) return new Literal("1", Advance().Text[1..^1]);
224 if (Current.Kind == TokenKind.Identifier) {
225 string name = Advance().Text;
226 if (!Take("(")) return new Variable(name);
227 List<QuantityNode> arguments = [Expression()];
228 while (Take(",")) arguments.Add(Expression());
229 Expect(")");
230 return new Call(name, arguments);
231 }
232 if (Take("(")) { QuantityNode node = Expression(); Expect(")"); return node; }
233 throw Syntax();
234 }
235}

289 private Quantity? Resolve(string name)
290 {
291 Unit? exact = units.Find(unit => unit.Symbol == name);
292 if (exact is not null) return exact.Quantity;
293 foreach (var prefix in Prefixes)
294 foreach (Unit unit in units)
295 if (unit.Definition is null && unit.Prefix && name == prefix.Symbol + unit.Symbol)
296 return new(prefix.Factor * unit.Quantity.Value, unit.Quantity.Dimension);
297 return null;
298 }

320 private ParsedUnit ParseUnit(string text)
321 {
322 UnitNode tree = new UnitParser(text).Parse();
323 List<string> names = [];
324 HashSet<string> seen = new(StringComparer.Ordinal);
325 void Validate(UnitNode node)
326 {
327 switch (node) {
328 case UnitName name:
329 if (Resolve(name.Name) is null) throw new CommandError($"UNKNOWN UNIT {name.Name}");
330 if (seen.Add(name.Name)) names.Add(name.Name);
331 break;
332 case UnitPower power: Validate(power.Atom); CheckExponent(power.Exponent); break;
333 case UnitProduct product: foreach (var term in product.Terms) Validate(term.Term); break;
334 }
335 }
336 Validate(tree);
337 return new(tree, UnitValue(tree), names);
338 }
339 private Quantity UnitValue(UnitNode node)
340 {
341 switch (node) {
342 case UnitName name: return Resolve(name.Name)!.Value;
343 case UnitNumber number: return new(new(long.Parse(number.Text)), new());
344 case UnitPower power:
345 Quantity atom = UnitValue(power.Atom);
346 int exponent = (int)long.Parse(power.Exponent);
347 return new(atom.Value.Power(exponent), atom.Dimension.Scale(exponent));
348 case UnitProduct product:
349 Quantity result = new(Fraction.One, new());
350 foreach (var term in product.Terms) {
351 Quantity value = UnitValue(term.Term);
352 result = term.Operator == '*'
353 ? new(result.Value * value.Value, result.Dimension + value.Dimension)
354 : new(result.Value / value.Value, result.Dimension - value.Dimension);
355 }
356 return result;
357 default: throw new InvalidOperationException();
358 }
359 }
360 private static Reduced Reduce(UnitNode node)
361 {
362 switch (node) {
363 case UnitName name: return new(Fraction.One, new(StringComparer.Ordinal) { [name.Name] = 1 });
364 case UnitNumber number: return new(new(long.Parse(number.Text)), new(StringComparer.Ordinal));
365 case UnitPower power:
366 Reduced atom = Reduce(power.Atom);
367 int exponent = (int)long.Parse(power.Exponent);
368 return new(atom.Coefficient.Power(exponent), atom.Exponents.ToDictionary(pair => pair.Key, pair => pair.Value * exponent, StringComparer.Ordinal));
369 case UnitProduct product:
370 Fraction coefficient = Fraction.One;
371 Dictionary<string, long> exponents = new(StringComparer.Ordinal);
372 foreach (var term in product.Terms) {
373 Reduced value = Reduce(term.Term);
374 coefficient = term.Operator == '*' ? coefficient * value.Coefficient : coefficient / value.Coefficient;
375 foreach (var pair in value.Exponents)
376 exponents[pair.Key] = exponents.GetValueOrDefault(pair.Key) + (term.Operator == '*' ? pair.Value : -pair.Value);
377 }
378 return new(coefficient, exponents);
379 default: throw new InvalidOperationException();
380 }
381 }

393 private static long IntegerRoot(long number)
394 {
395 long low = 0, high = number + 1;
396 while (high - low > 1) {
397 long middle = low + (high - low) / 2;
398 if (middle <= number / middle) low = middle;
399 else high = middle;
400 }
401 return low;
402 }
403 private Quantity Evaluate(QuantityNode node)
404 {
405 switch (node) {
406 case Literal literal:
407 if (literal.Unit is null) return new(Fraction.Parse(literal.Number), new());
408 Quantity unit = ParseUnit(literal.Unit).Quantity;
409 return new(Fraction.Parse(literal.Number) * unit.Value, unit.Dimension);
410 case Variable variable:
411 if (!variables.TryGetValue(variable.Name, out Quantity stored)) throw new CommandError($"UNKNOWN VARIABLE {variable.Name}");
412 return stored;
413 case Negative negative:
414 Quantity operand = Evaluate(negative.Operand);
415 return new(-operand.Value, operand.Dimension);
416 case Power power:
417 Quantity basis = Evaluate(power.Base);
418 int exponent = CheckExponent(power.Exponent);
419 if (basis.Value.IsZero && exponent < 0) throw new CommandError("DIVISION BY ZERO");
420 return new(basis.Value.Power(exponent), basis.Dimension.Scale(exponent));
421 case Binary binary:
422 Quantity left = Evaluate(binary.Left);
423 Quantity right = Evaluate(binary.Right);
424 if (binary.Operator is '+' or '-') {
425 Match(left.Dimension, right.Dimension);
426 return new(binary.Operator == '+' ? left.Value + right.Value : left.Value - right.Value, left.Dimension);
427 }
428 if (binary.Operator == '*') return new(left.Value * right.Value, left.Dimension + right.Dimension);
429 if (right.Value.IsZero) throw new CommandError("DIVISION BY ZERO");
430 return new(left.Value / right.Value, left.Dimension - right.Dimension);
431 case Call call:
432 if (call.Name is not ("abs" or "sqrt" or "min" or "max")) throw new CommandError($"UNKNOWN FUNCTION {call.Name}");
433 int count = call.Name is "abs" or "sqrt" ? 1 : 2;
434 if (call.Arguments.Count != count) throw new CommandError($"BAD ARGUMENTS {call.Name}");
435 Quantity a = Evaluate(call.Arguments[0]);
436 if (call.Name == "abs") return new(a.Value.Abs(), a.Dimension);
437 if (call.Name == "sqrt") {
438 if (a.Value.Numerator < 0) throw new CommandError("NEGATIVE ROOT");
439 if (!a.Dimension.IsEven) throw new CommandError($"ODD DIMENSION {a.Dimension}");
440 long p = IntegerRoot(a.Value.Numerator), q = IntegerRoot(a.Value.Denominator);
441 if (p * p != a.Value.Numerator || q * q != a.Value.Denominator) throw new CommandError("NOT EXACT");
442 return new(new(p, q), a.Dimension.Half());
443 }
444 Quantity b = Evaluate(call.Arguments[1]);
445 Match(a.Dimension, b.Dimension);
446 int comparison = a.Value.CompareTo(b.Value);
447 return call.Name == "min" ? (comparison <= 0 ? a : b) : (comparison >= 0 ? a : b);
448 default: throw new InvalidOperationException();
449 }
450 }
451 private List<Unit> Matching(Dimension dimension)
452 {
453 List<Unit> selected = units.FindAll(unit => unit.Quantity.Dimension == dimension);
454 selected.Sort((a, b) => {
455 int comparison = a.Quantity.Value.CompareTo(b.Quantity.Value);
456 return comparison != 0 ? comparison : StringComparer.Ordinal.Compare(a.Symbol, b.Symbol);
457 });
458 return selected;
459 }

461 public string Execute(string line)
462 {
463 string[] words = line.Split(' ');
464 string command = words[0];
465 switch (command) {
466 case "DEFINE": {
467 CheckName(words[1], false);
468 Fraction value = Fraction.Parse(words[3]);
469 Quantity unit = ParseUnit(words[4]).Quantity;
470 if (Kinds.ContainsKey(words[1]) || Resolve(words[1]) is not null) throw new CommandError($"NAME TAKEN {words[1]}");
471 if (value.Numerator <= 0) throw new CommandError($"NOT POSITIVE {words[3]}");
472 Quantity defined = new(value * unit.Value, unit.Dimension);
473 units.Add(new(words[1], defined, false, new(value, words[4])));
474 return $"DEFINE {words[1]} factor={defined.Value} dim={defined.Dimension} kind={Kind(defined.Dimension)}";
475 }
476 case "UNDEF": {
477 int index = units.FindIndex(unit => unit.Symbol == words[1]);
478 if (index >= 0 && units[index].Definition is null) throw new CommandError($"BUILTIN UNIT {words[1]}");
479 if (index < 0) throw new CommandError($"NOT DEFINED {words[1]}");
480 units.RemoveAt(index);
481 return $"UNDEF {words[1]}";
482 }
483 case "DEFS": {
484 List<Unit> users = units.FindAll(unit => unit.Definition is not null);
485 List<string> lines = [$"DEFS count={users.Count}"];
486 foreach (Unit unit in users)
487 lines.Add($"DEF {unit.Symbol} = {unit.Definition!.Value} {unit.Definition.Expression} factor={unit.Quantity.Value} dim={unit.Quantity.Dimension}");
488 return string.Join('\n', lines);
489 }
490 case "CONVERT": case "INVERT": {
491 Fraction value = Fraction.Parse(words[1]);
492 Quantity a = ParseUnit(words[2]).Quantity;
493 Quantity b = ParseUnit(words[4]).Quantity;
494 if (command == "CONVERT") Match(a.Dimension, b.Dimension);
495 else {
496 if (a.Dimension.IsZero || !(a.Dimension + b.Dimension).IsZero)
497 throw new CommandError($"NOT RECIPROCAL {a.Dimension} vs {b.Dimension}");
498 if (value.IsZero) throw new CommandError("DIVISION BY ZERO");
499 }
500 Fraction result = command == "CONVERT" ? value * a.Value / b.Value : (Fraction.One / (value * a.Value)) / b.Value;
501 return $"{command} {value} {words[2]} = {Format(result, words[4])}";
502 }
503 case "DIM": case "REDUCE": case "SIMPLIFY": {
504 ParsedUnit parsed = ParseUnit(words[1]);
505 Quantity unit = parsed.Quantity;
506 return command switch {
507 "DIM" => $"DIM {words[1]} = {unit.Dimension} factor={unit.Value} kind={Kind(unit.Dimension)}",
508 "REDUCE" => $"REDUCE {words[1]} = {ReducedText(parsed)}",
509 _ => $"SIMPLIFY {words[1]} = {Format(unit)}"
510 };
511 }
512 case "ADD": case "SUB": case "MUL": case "DIV": case "COMPARE": {
513 Fraction x = Fraction.Parse(words[1]);
514 Quantity a = ParseUnit(words[2]).Quantity;
515 Fraction y = Fraction.Parse(words[3]);
516 Quantity b = ParseUnit(words[4]).Quantity;
517 Quantity? target = words.Length == 7 ? ParseUnit(words[6]).Quantity : null;
518 if (command == "COMPARE") {
519 Match(a.Dimension, b.Dimension);
520 Fraction left = x * a.Value, right = y * b.Value;
521 int comparison = left.CompareTo(right);
522 string relation = comparison < 0 ? "<" : comparison > 0 ? ">" : "=";
523 string ratio = right.IsZero ? "undefined" : (left / right).ToString();
524 return $"COMPARE {x} {words[2]} {relation} {y} {words[4]} ratio={ratio}";
525 }
526 char op = command switch { "ADD" => '+', "SUB" => '-', "MUL" => '*', _ => '/' };
527 string prefix = $"{command} {x} {words[2]} {op} {y} {words[4]} = ";
528 if (command is "ADD" or "SUB") {
529 Match(a.Dimension, b.Dimension);
530 if (target is Quantity destination) Match(a.Dimension, destination.Dimension);
531 Fraction left = x * a.Value, right = y * b.Value;
532 Fraction result = (command == "ADD" ? left + right : left - right) / (target?.Value ?? a.Value);
533 return prefix + Format(result, target is null ? words[2] : words[6]);
534 }
535 Dimension dimension = command == "MUL" ? a.Dimension + b.Dimension : a.Dimension - b.Dimension;
536 if (command == "DIV" && y.IsZero) throw new CommandError("DIVISION BY ZERO");
537 if (target is Quantity output) Match(dimension, output.Dimension);
538 Fraction product = command == "MUL" ? x * y : x / y;
539 Fraction first = x * a.Value, second = y * b.Value;
540 Fraction coherent = command == "MUL" ? first * second : first / second;
541 if (target is Quantity converted) return prefix + Format(coherent / converted.Value, words[6]);
542 string rightUnit = command == "DIV" && (words[4].Contains('*') || words[4].Contains('/')) ? $"({words[4]})" : words[4];
543 return prefix + $"{product} {words[2]}{op}{rightUnit} = {Format(coherent, Simplified(dimension))}";
544 }
545 case "TEMP": {
546 Fraction value = Fraction.Parse(words[1]);
547 foreach (string scale in new[] { words[2], words[4] })
548 if (scale is not ("C" or "F" or "K" or "R")) throw new CommandError($"UNKNOWN SCALE {scale}");
549 Fraction kelvin = words[2] switch {
550 "C" => value + new Fraction(5463, 20),
551 "F" => (value + new Fraction(45967, 100)) * new Fraction(5) / new Fraction(9),
552 "R" => value * new Fraction(5) / new Fraction(9), _ => value
553 };
554 if (kelvin.Numerator < 0) throw new CommandError("BELOW ABSOLUTE ZERO");
555 Fraction result = words[4] switch {
556 "C" => kelvin - new Fraction(5463, 20),
557 "F" => kelvin * new Fraction(9) / new Fraction(5) - new Fraction(45967, 100),
558 "R" => kelvin * new Fraction(9) / new Fraction(5), _ => kelvin
559 };
560 return $"TEMP {value} {words[2]} = {Format(result, words[4])}";
561 }
562 case "LIST": {
563 Dimension dimension = Kinds.TryGetValue(words[1], out Dimension kind) ? kind : ParseUnit(words[1]).Quantity.Dimension;
564 List<Unit> selected = Matching(dimension);
565 List<string> lines = [$"LIST {words[1]} dim={dimension} kind={Kind(dimension)} count={selected.Count}"];
566 foreach (Unit unit in selected)
567 lines.Add($"UNIT {unit.Symbol} factor={unit.Quantity.Value} origin={(unit.Definition is null ? "builtin" : "user")} prefix={(unit.Prefix ? "yes" : "no")}");
568 return string.Join('\n', lines);
569 }
570 case "TABLE": case "AUTO": case "SPLIT": {
571 Fraction value = Fraction.Parse(words[1]);
572 Quantity unit = ParseUnit(words[2]).Quantity;
573 if (command == "TABLE") {
574 List<Unit> selected = Matching(unit.Dimension);
575 List<string> lines = [$"TABLE {value} {words[2]} dim={unit.Dimension} count={selected.Count}"];
576 foreach (Unit row in selected) lines.Add("ROW " + Format(value * unit.Value / row.Quantity.Value, row.Symbol));
577 return string.Join('\n', lines);
578 }
579 if (command == "AUTO") {
580 Unit? basis = units.Find(candidate => candidate.Definition is null && candidate.Prefix && candidate.Quantity.Dimension == unit.Dimension);
581 if (basis is null) throw new CommandError($"NO PREFIXABLE UNIT {unit.Dimension}");
582 Fraction scaled = value * unit.Value / basis.Quantity.Value, magnitude = scaled.Abs();
583 int exponent = 0;
584 if (magnitude.CompareTo(Fraction.One) >= 0) {
585 while (exponent < 9 && magnitude.CompareTo(new Fraction(10).Power(exponent + 3)) >= 0) exponent += 3;
586 } else if (!magnitude.IsZero) {
587 while (exponent > -9 && magnitude.CompareTo(new Fraction(10).Power(exponent)) < 0) exponent -= 3;
588 }
589 string prefix = exponent switch { 9 => "G", 6 => "M", 3 => "k", -3 => "m", -6 => "u", -9 => "n", _ => "" };
590 return $"AUTO {value} {words[2]} = {Format(scaled / new Fraction(10).Power(exponent), prefix + basis.Symbol)}";
591 }
592 string[] parts = words[4].Split(',');
593 List<Quantity> destinations = [];
594 foreach (string part in parts) destinations.Add(ParseUnit(part).Quantity);
595 if (value.Numerator < 0) throw new CommandError("NEGATIVE VALUE");
596 foreach (Quantity destination in destinations) Match(unit.Dimension, destination.Dimension);
597 for (int i = 1; i < destinations.Count; i++)
598 if (destinations[i - 1].Value.CompareTo(destinations[i].Value) <= 0) throw new CommandError("UNIT ORDER");
599 Fraction remaining = value * unit.Value;
600 List<string> pieces = [];
601 for (int i = 0; i < parts.Length - 1; i++) {
602 Fraction quotient = remaining / destinations[i].Value;
603 long whole = quotient.Numerator / quotient.Denominator;
604 remaining -= new Fraction(whole) * destinations[i].Value;
605 pieces.Add($"{whole} {parts[i]}");
606 }
607 pieces.Add(Format(remaining / destinations[^1].Value, parts[^1]));
608 return $"SPLIT {value} {words[2]} = {string.Join(' ', pieces)}";
609 }
610 case "LET": {
611 CheckName(words[1], true);
612 Quantity value = Evaluate(new QuantityParser(words[3]).Parse());
613 variables[words[1]] = value;
614 return $"LET {words[1]} = {Format(value)}";
615 }
616 case "EVAL": {
617 Quantity value = Evaluate(new QuantityParser(words[1]).Parse());
618 if (words.Length == 4) {
619 Quantity unit = ParseUnit(words[3]).Quantity;
620 Match(value.Dimension, unit.Dimension);
621 return $"EVAL {words[1]} = {Format(value.Value / unit.Value, words[3])}";
622 }
623 return $"EVAL {words[1]} = {Format(value)}";
624 }
625 case "VARS": {
626 List<string> lines = [$"VARS count={variables.Count}"];
627 foreach (string name in variables.Keys.Order(StringComparer.Ordinal)) lines.Add($"VAR {name} = {Format(variables[name])}");
628 return string.Join('\n', lines);
629 }
630 default: throw new CommandError($"UNKNOWN COMMAND {command}");
631 }
632 }
633}
