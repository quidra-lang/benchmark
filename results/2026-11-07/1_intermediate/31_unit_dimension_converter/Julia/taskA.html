19function fraction_text(value::Fraction)
20 n, d = numerator(value), denominator(value)
21 d == 1 ? string(n) : "$n/$d"
22end
23function decimal_text(value::Fraction)
24 n, d = abs(numerator(value)), denominator(value)
25 whole = div(n, d)
26 tail = div(rem(n, d) * 1_000_000, d)
27 string(value < ZERO ? "-" : "", whole, '.', lpad(string(tail), 6, '0'))
28end
29notation(value::Fraction, unit::AbstractString) = "$(fraction_text(value)) $unit ($(decimal_text(value)))"

31function scalar_power(value::Fraction, exponent::Integer)
32 factor = exponent < 0 ? ONE / value : value
33 result = ONE
34 for _ in 1:abs(exponent)
35 result *= factor
36 end
37 result
38end
39function parse_number(text::AbstractString)::Fraction
40 occursin(r"^-?[0-9]+([./][0-9]+)?$", text) || fail("BAD NUMBER $text")
41 negative = startswith(text, "-")
42 body = negative ? text[2:end] : text
43 value = if occursin('/', body)
44 parts = split(body, '/')
45 divisor = parse(Int64, parts[2]; base = 10)
46 divisor != 0 || fail("BAD NUMBER $text")
47 parse(Int64, parts[1]; base = 10) // divisor
48 elseif occursin('.', body)
49 parts = split(body, '.')
50 divisor = Int64(1)
51 for _ in 1:length(parts[2])
52 divisor *= 10
53 end
54 (parse(Int64, parts[1]; base = 10) * divisor + parse(Int64, parts[2]; base = 10)) // divisor
55 else
56 parse(Int64, body; base = 10) // Int64(1)
57 end
58 negative ? -value : value
59end
60function checked_exponent(text::AbstractString)
61 value = parse(Int64, text; base = 10)
62 value in (-3, -2, -1, 1, 2, 3) || fail("BAD EXPONENT $text")
63 value
64end

67 fail("BAD NAME $name")

70combine(a::Dimension, b::Dimension, sign::Int64 = Int64(1)) = ntuple(i -> a[i] + sign * b[i], 7)
71scaled(dimension::Dimension, multiplier::Int64) = ntuple(i -> dimension[i] * multiplier, 7)
72function dimension_text(dimension::Dimension)
73 terms = String[]
74 for i in eachindex(dimension)
75 exponent = dimension[i]
76 exponent == 0 && continue
77 push!(terms, exponent == 1 ? BASE_NAMES[i] : "$(BASE_NAMES[i])^$exponent")
78 end
79 isempty(terms) ? "1" : join(terms, ' ')
80end
81function match_dimensions(first::Dimension, second::Dimension)
82 first == second || fail("DIMENSION MISMATCH $(dimension_text(first)) vs $(dimension_text(second))")
83end

176@enum TokenKind NAME_TOKEN NUMBER_TOKEN UNIT_TOKEN SYMBOL_TOKEN INVALID_TOKEN END_TOKEN
177struct Token
178 kind::TokenKind
179 text::String
180 position::Int
181end
182mutable struct Parser
183 source::String
184 tokens::Vector{Token}
185 position::Int
186end
187letter(c::Char) = 'A' <= c <= 'Z' || 'a' <= c <= 'z'
188digit(c::Char) = '0' <= c <= '9'
189function tokenize(source::String, quantity_mode::Bool)
190 characters = collect(source)
191 tokens = Token[]
192 count = length(characters)
193 position = 1
194 while position <= count
195 start = position
196 character = characters[position]
197 position += 1
198 kind = INVALID_TOKEN
199 if letter(character)
200 kind = NAME_TOKEN
201 while position <= count && letter(characters[position])
202 position += 1
203 end
204 elseif digit(character)
205 kind = NUMBER_TOKEN
206 while position <= count && digit(characters[position])
207 position += 1
208 end
209 if quantity_mode && position < count && characters[position] == '.' && digit(characters[position+1])
210 position += 1
211 while position <= count && digit(characters[position])
212 position += 1
213 end
214 end
215 elseif quantity_mode && character == '['
216 closing = position
217 while closing <= count && characters[closing] != ']'
218 closing += 1
219 end
220 if position < closing <= count
221 kind = UNIT_TOKEN
222 position = closing + 1
223 end
224 elseif character in (quantity_mode ? "+-*/^()," : "*/^()-")
225 kind = SYMBOL_TOKEN
226 end
227 push!(tokens, Token(kind, source[start:position-1], start))
228 end
229 push!(tokens, Token(END_TOKEN, "", count + 1))
230 Parser(source, tokens, 1)
231end
232current(parser::Parser) = parser.tokens[parser.position]
233function syntax_error(parser::Parser)
234 token = current(parser)
235 position = token.kind == END_TOKEN ? "END" : string(token.position)
236 fail("SYNTAX $(parser.source) AT $position")
237end
238function take_token!(parser::Parser, text::String)
239 current(parser).text == text || return false
240 parser.position += 1
241 true
242end
243expect!(parser::Parser, text::String) = take_token!(parser, text) || syntax_error(parser)
244function parse_exponent!(parser::Parser)
245 sign = take_token!(parser, "-") ? "-" : ""
246 token = current(parser)
247 token.kind == NUMBER_TOKEN && !occursin('.', token.text) || syntax_error(parser)
248 parser.position += 1
249 sign * token.text
250end

266function unit_term!(parser::Parser)::UnitExpr
267 token = current(parser)
268 result = if token.kind == NAME_TOKEN
269 parser.position += 1
270 UnitName(token.text)
271 elseif token.kind == NUMBER_TOKEN && first(token.text) != '0'
272 parser.position += 1
273 UnitNumber(parse(Int64, token.text; base = 10))
274 elseif take_token!(parser, "(")
275 inner = unit_expression!(parser)
276 expect!(parser, ")")
277 inner
278 else
279 syntax_error(parser)
280 end
281 take_token!(parser, "^") ? UnitPower(result, parse_exponent!(parser)) : result
282end
283function unit_expression!(parser::Parser)
284 terms = Pair{Char,UnitExpr}['*' => unit_term!(parser)]
285 while current(parser).text in ("*", "/")
286 operation = first(current(parser).text)
287 parser.position += 1
288 push!(terms, operation => unit_term!(parser))
289 end
290 UnitProduct(terms)
291end
292function validate_unit(database::Database, tree::UnitExpr)
293 if tree isa UnitName
294 isnothing(resolve(database, tree.name)) && fail("UNKNOWN UNIT $(tree.name)")
295 elseif tree isa UnitPower
296 validate_unit(database, tree.child)
297 checked_exponent(tree.exponent)
298 elseif tree isa UnitProduct
299 for (_, term) in tree.terms
300 validate_unit(database, term)
301 end
302 end
303 nothing
304end

307function evaluate_unit(database::Database, tree::UnitPower)
308 value = evaluate_unit(database, tree.child)

311end
312function evaluate_unit(database::Database, tree::UnitProduct)
313 result = Quantity()
314 for (operation, term) in tree.terms
315 right = evaluate_unit(database, term)
316 result = operation == '*' ? Quantity(result.value * right.value, combine(result.dimension, right.dimension)) :
317 Quantity(result.value / right.value, combine(result.dimension, right.dimension, Int64(-1)))
318 end
319 result
320end

325 validate_unit(database, tree)
326 tree
327end

330mutable struct Reduced
331 coefficient::Fraction
332 names::Vector{String}
333 powers::Vector{Int64}
334end
335reduce_unit(tree::UnitName) = Reduced(ONE, [tree.name], Int64[1])
336reduce_unit(tree::UnitNumber) = Reduced(tree.value // Int64(1), String[], Int64[])
337function reduce_unit(tree::UnitPower)
338 result = reduce_unit(tree.child)
339 exponent = parse(Int64, tree.exponent; base = 10)
340 result.coefficient = scalar_power(result.coefficient, exponent)
341 result.powers .*= exponent
342 result
343end
344function reduce_unit(tree::UnitProduct)
345 result = Reduced(ONE, String[], Int64[])
346 for (operation, term) in tree.terms
347 right = reduce_unit(term)
348 sign = operation == '*' ? Int64(1) : Int64(-1)
349 result.coefficient = sign == 1 ? result.coefficient * right.coefficient : result.coefficient / right.coefficient
350 for i in eachindex(right.names)
351 index = findfirst(==(right.names[i]), result.names)
352 if isnothing(index)
353 push!(result.names, right.names[i])
354 push!(result.powers, sign * right.powers[i])
355 else
356 result.powers[index] += sign * right.powers[i]
357 end
358 end
359 end
360 result
361end
362function reduced_text(tree::UnitExpr)
363 result = reduce_unit(tree)
364 parts = result.coefficient == ONE ? String[] : [fraction_text(result.coefficient)]
365 for i in eachindex(result.names)
366 exponent = result.powers[i]
367 exponent == 0 && continue
368 push!(parts, exponent == 1 ? result.names[i] : "$(result.names[i])^$exponent")
369 end
370 isempty(parts) ? "1" : join(parts, ' ')
371end

397function quantity_primary!(parser::Parser)::QuantityExpr
398 token = current(parser)
399 result = if token.kind == NUMBER_TOKEN
400 parser.position += 1
401 unit = nothing
402 if current(parser).kind == UNIT_TOKEN
403 text = current(parser).text
404 parser.position += 1
405 unit = text[2:end-1]
406 end
407 Literal(token.text, unit)
408 elseif token.kind == UNIT_TOKEN
409 parser.position += 1
410 Literal("1", token.text[2:end-1])
411 elseif token.kind == NAME_TOKEN
412 parser.position += 1
413 if take_token!(parser, "(")
414 arguments = QuantityExpr[quantity_expression!(parser)]
415 while take_token!(parser, ",")
416 push!(arguments, quantity_expression!(parser))
417 end
418 expect!(parser, ")")
419 Call(token.text, arguments)
420 else
421 Variable(token.text)
422 end
423 elseif take_token!(parser, "(")
424 inner = quantity_expression!(parser)
425 expect!(parser, ")")
426 inner
427 else
428 syntax_error(parser)
429 end
430 take_token!(parser, "^") ? Power(result, parse_exponent!(parser)) : result
431end
432quantity_unary!(parser::Parser)::QuantityExpr = take_token!(parser, "-") ? Negative(quantity_unary!(parser)) : quantity_primary!(parser)

437 parser.position += 1

439 result = Binary(operation, result, right)
440 end
441 result

447 parser.position += 1

449 result = Binary(operation, result, right)

453function integer_sqrt(value::Int64)
454 value < 2 && return value
455 current_value = value
456 following = div(value + 1, 2)
457 while following < current_value
458 current_value = following
459 following = div(current_value + div(value, current_value), 2)
460 end
461 current_value
462end

482function evaluate_quantity(database::Database, tree::Binary)
483 left = evaluate_quantity(database, tree.left)
484 right = evaluate_quantity(database, tree.right)
485 if tree.operation in ('+', '-')
486 match_dimensions(left.dimension, right.dimension)
487 value = tree.operation == '+' ? left.value + right.value : left.value - right.value
488 return Quantity(value, left.dimension)
489 end
490 tree.operation == '*' && return Quantity(left.value * right.value, combine(left.dimension, right.dimension))
491 right.value == ZERO && fail("DIVISION BY ZERO")
492 Quantity(left.value / right.value, combine(left.dimension, right.dimension, Int64(-1)))
493end
494function evaluate_quantity(database::Database, tree::Call)
495 tree.name in ("abs", "sqrt", "min", "max") || fail("UNKNOWN FUNCTION $(tree.name)")
496 required = tree.name in ("abs", "sqrt") ? 1 : 2
497 length(tree.arguments) == required || fail("BAD ARGUMENTS $(tree.name)")
498 first_value = evaluate_quantity(database, tree.arguments[1])
499 tree.name == "abs" && return Quantity(abs(first_value.value), first_value.dimension)
500 if tree.name == "sqrt"
501 first_value.value < ZERO && fail("NEGATIVE ROOT")
502 all(iseven, first_value.dimension) || fail("ODD DIMENSION $(dimension_text(first_value.dimension))")
503 n, d = numerator(first_value.value), denominator(first_value.value)
504 root_n, root_d = integer_sqrt(n), integer_sqrt(d)
505 root_n * root_n == n && root_d * root_d == d || fail("NOT EXACT")
506 return Quantity(root_n // root_d, ntuple(i -> div(first_value.dimension[i], 2), 7))
507 end
508 second_value = evaluate_quantity(database, tree.arguments[2])
509 match_dimensions(first_value.dimension, second_value.dimension)
510 take_first = tree.name == "min" ? first_value.value <= second_value.value : first_value.value >= second_value.value
511 take_first ? first_value : second_value
512end
513function quantity_value(database::Database, source::String)
514 parser = tokenize(source, true)
515 tree = quantity_expression!(parser)
516 current(parser).kind == END_TOKEN || syntax_error(parser)
517 evaluate_quantity(database, tree)
518end

520const COMMANDS = ("DEFINE", "UNDEF", "DEFS", "CONVERT", "INVERT", "DIM", "REDUCE", "SIMPLIFY",
521 "ADD", "SUB", "MUL", "DIV", "TEMP", "COMPARE", "LIST", "TABLE", "AUTO", "SPLIT",
522 "LET", "EVAL", "VARS")
523function execute!(database::Database, line::String)::Vector{String}
524 words = String.(split(line))
525 command = words[1]
526 command in COMMANDS || fail("UNKNOWN COMMAND $command")
527 if command == "DEFINE"
528 name = words[2]
529 check_name(name)
530 value = parse_number(words[4])
531 unit = unit_value(database, words[5])
532 if !isnothing(kind_dimension(name)) || !isnothing(resolve(database, name))
533 fail("NAME TAKEN $name")
534 end
535 value > ZERO || fail("NOT POSITIVE $(words[4])")
536 result = Quantity(value * unit.value, unit.dimension)
537 push!(database.units, UnitEntry(name, result, false, Definition(value, words[5])))
538 return ["DEFINE $name factor=$(fraction_text(result.value)) dim=$(dimension_text(result.dimension)) kind=$(kind_name(result.dimension))"]
539 elseif command == "UNDEF"
540 name = words[2]
541 index = findfirst(unit -> unit.symbol == name, database.units)
542 isnothing(index) && fail("NOT DEFINED $name")
543 index <= 42 && fail("BUILTIN UNIT $name")
544 deleteat!(database.units, index)
545 return ["UNDEF $name"]
546 elseif command == "DEFS"
547 output = ["DEFS count=$(length(database.units) - 42)"]
548 for unit in @view database.units[43:end]
549 definition = unit.definition::Definition
550 push!(output, "DEF $(unit.symbol) = $(fraction_text(definition.value)) $(definition.expression) factor=$(fraction_text(unit.quantity.value)) dim=$(dimension_text(unit.quantity.dimension))")
551 end
552 return output
553 elseif command in ("CONVERT", "INVERT")
554 value = parse_number(words[2])
555 source = unit_value(database, words[3])
556 target = unit_value(database, words[5])
557 if command == "INVERT"
558 if source.dimension == DIMENSIONLESS || combine(source.dimension, target.dimension) != DIMENSIONLESS
559 fail("NOT RECIPROCAL $(dimension_text(source.dimension)) vs $(dimension_text(target.dimension))")
560 end
561 value == ZERO && fail("DIVISION BY ZERO")
562 else
563 match_dimensions(source.dimension, target.dimension)
564 end
565 result = value * source.value
566 command == "INVERT" && (result = ONE / result)
567 return ["$command $(fraction_text(value)) $(words[3]) = $(notation(result / target.value, words[5]))"]
568 elseif command in ("DIM", "REDUCE", "SIMPLIFY")
569 tree = parse_unit(database, words[2])
570 unit = evaluate_unit(database, tree)
571 result_text = if command == "DIM"
572 "$(dimension_text(unit.dimension)) factor=$(fraction_text(unit.value)) kind=$(kind_name(unit.dimension))"
573 elseif command == "REDUCE"
574 reduced_text(tree)
575 else
576 notation(unit.value, simplified(database, unit.dimension))
577 end
578 return ["$command $(words[2]) = $result_text"]
579 elseif command in ("ADD", "SUB", "MUL", "DIV", "COMPARE")
580 a = parse_number(words[2])
581 first_unit = unit_value(database, words[3])
582 b = parse_number(words[4])
583 second_unit = unit_value(database, words[5])
584 has_target = length(words) == 7
585 target = has_target ? unit_value(database, words[7]) : first_unit
586 prefix = "$command $(fraction_text(a)) $(words[3])"
587 suffix = "$(fraction_text(b)) $(words[5])"
588 if command in ("ADD", "SUB", "COMPARE")
589 match_dimensions(first_unit.dimension, second_unit.dimension)
590 has_target && match_dimensions(first_unit.dimension, target.dimension)
591 left = a * first_unit.value
592 right = b * second_unit.value
593 if command == "COMPARE"
594 relation = left < right ? "<" : left > right ? ">" : "="
595 ratio = right == ZERO ? "undefined" : fraction_text(left / right)
596 return ["$prefix $relation $suffix ratio=$ratio"]
597 end
598 operation = command == "ADD" ? "+" : "-"
599 result = command == "ADD" ? left + right : left - right
600 label = has_target ? words[7] : words[3]
601 return ["$prefix $operation $suffix = $(notation(result / target.value, label))"]
602 end
603 multiply = command == "MUL"
604 dimension = combine(first_unit.dimension, second_unit.dimension, multiply ? Int64(1) : Int64(-1))
605 !multiply && b == ZERO && fail("DIVISION BY ZERO")
606 has_target && match_dimensions(dimension, target.dimension)
607 raw = multiply ? a * b : a / b
608 left = a * first_unit.value
609 right = b * second_unit.value
610 coherent = multiply ? left * right : left / right
611 operation = multiply ? "*" : "/"
612 prefix = "$prefix $operation $suffix ="
613 has_target && return ["$prefix $(notation(coherent / target.value, words[7]))"]
614 right_unit = words[5]
615 if !multiply && (occursin('*', right_unit) || occursin('/', right_unit))
616 right_unit = "($right_unit)"
617 end
618 combined = words[3] * operation * right_unit
619 return ["$prefix $(fraction_text(raw)) $combined = $(notation(coherent, simplified(database, dimension)))"]
620 elseif command == "TEMP"
621 value = parse_number(words[2])
622 for scale in (words[3], words[5])
623 scale in ("C", "F", "K", "R") || fail("UNKNOWN SCALE $scale")
624 end
625 kelvin = if words[3] == "C"
626 value + 5463//20
627 elseif words[3] == "F"
628 (value + 45967//100) * 5 / 9
629 elseif words[3] == "R"
630 value * 5 / 9
631 else
632 value
633 end
634 kelvin < ZERO && fail("BELOW ABSOLUTE ZERO")
635 result = if words[5] == "C"
636 kelvin - 5463//20
637 elseif words[5] == "F"
638 kelvin * 9 / 5 - 45967//100
639 elseif words[5] == "R"
640 kelvin * 9 / 5
641 else
642 kelvin
643 end
644 return ["TEMP $(fraction_text(value)) $(words[3]) = $(notation(result, words[5]))"]
645 elseif command in ("LIST", "TABLE")
646 value = ONE
647 unit = Quantity()
648 dimension = if command == "TABLE"
649 value = parse_number(words[2])
650 unit = unit_value(database, words[3])
651 unit.dimension
652 else
653 known = kind_dimension(words[2])
654 isnothing(known) ? unit_value(database, words[2]).dimension : known
655 end
656 selected = selected_units(database, dimension)
657 output = if command == "TABLE"
658 ["TABLE $(fraction_text(value)) $(words[3]) dim=$(dimension_text(dimension)) count=$(length(selected))"]
659 else
660 ["LIST $(words[2]) dim=$(dimension_text(dimension)) kind=$(kind_name(dimension)) count=$(length(selected))"]
661 end
662 for item in selected
663 if command == "TABLE"
664 push!(output, "ROW $(notation((value * unit.value) / item.quantity.value, item.symbol))")
665 else
666 origin = isnothing(item.definition) ? "builtin" : "user"
667 prefix_flag = item.prefixable ? "yes" : "no"
668 push!(output, "UNIT $(item.symbol) factor=$(fraction_text(item.quantity.value)) origin=$origin prefix=$prefix_flag")
669 end
670 end
671 return output
672 elseif command == "AUTO"
673 value = parse_number(words[2])
674 unit = unit_value(database, words[3])
675 index = findfirst(item -> item.prefixable && item.quantity.dimension == unit.dimension, @view database.units[1:42])
676 isnothing(index) && fail("NO PREFIXABLE UNIT $(dimension_text(unit.dimension))")
677 reference = database.units[index]
678 scaled_value = (value * unit.value) / reference.quantity.value
679 magnitude = abs(scaled_value)
680 exponent = 0
681 if magnitude >= ONE
682 while exponent < 9 && magnitude >= scalar_power(10//1, exponent + 3)
683 exponent += 3
684 end
685 elseif magnitude > ZERO
686 while exponent > -9 && magnitude < scalar_power(10//1, exponent)
687 exponent -= 3
688 end
689 end
690 prefix = ("n", "u", "m", "", "k", "M", "G")[div(exponent + 9, 3) + 1]
691 label = prefix * reference.symbol
692 return ["AUTO $(fraction_text(value)) $(words[3]) = $(notation(scaled_value / scalar_power(10//1, exponent), label))"]
693 elseif command == "SPLIT"
694 value = parse_number(words[2])
695 unit = unit_value(database, words[3])
696 parts = String.(split(words[5], ','))
697 targets = [unit_value(database, part) for part in parts]
698 value < ZERO && fail("NEGATIVE VALUE")
699 for target in targets
700 match_dimensions(unit.dimension, target.dimension)
701 end
702 for i in 1:length(targets)-1
703 targets[i].value > targets[i+1].value || fail("UNIT ORDER")
704 end
705 remaining = value * unit.value
706 pieces = String[]
707 for i in 1:length(parts)-1
708 quotient = remaining / targets[i].value
709 whole = div(numerator(quotient), denominator(quotient))
710 remaining -= whole * targets[i].value
711 push!(pieces, "$whole $(parts[i])")
712 end
713 push!(pieces, notation(remaining / targets[end].value, parts[end]))
714 return ["SPLIT $(fraction_text(value)) $(words[3]) = $(join(pieces, ' '))"]
715 elseif command == "LET"
716 check_name(words[2], true)
717 result = quantity_value(database, words[4])
718 database.variables[words[2]] = result
719 return ["LET $(words[2]) = $(notation(result.value, simplified(database, result.dimension)))"]
720 elseif command == "EVAL"
721 result = quantity_value(database, words[2])
722 value = result.value
723 label = simplified(database, result.dimension)
724 if length(words) == 4
725 unit = unit_value(database, words[4])
726 match_dimensions(result.dimension, unit.dimension)
727 value /= unit.value
728 label = words[4]
729 end
730 return ["EVAL $(words[2]) = $(notation(value, label))"]
731 else
732 names = sort!(collect(keys(database.variables)))
733 output = ["VARS count=$(length(names))"]
734 for name in names
735 result = database.variables[name]
736 push!(output, "VAR $name = $(notation(result.value, simplified(database, result.dimension)))")
737 end
738 return output
739 end
740end

743 command_count = parse(Int, readline())
744 database = new_database()
745 for _ in 1:command_count
746 line = readline()
747 output = try
748 execute!(database, line)
749 catch problem
750 problem isa CommandError || rethrow()
751 ["ERROR $(problem.message)"]
752 end
753 for line in output
754 println(line)
755 end
756 end
757end
