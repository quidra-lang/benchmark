161def decimal_text(value):
162 whole, remainder = divmod(abs(value.numerator), value.denominator)

210def parse_number(text):
211 negative = text.startswith("-")
212 body = text[1:] if negative else text
213 value = None
214 if body.isdigit():
215 value = Fraction(int(body))
216 elif "." in body:
217 whole, _, fraction = body.partition(".")
218 if whole.isdigit() and fraction.isdigit():
219 value = decimal_value(body)
220 elif "/" in body:
221 numerator, _, denominator = body.partition("/")
222 if numerator.isdigit() and denominator.isdigit() and int(denominator) != 0:
223 value = Fraction(int(numerator), int(denominator))
224 if value is None:
225 raise CommandError(f"BAD NUMBER {text}")
226 return -value if negative else value

241def square_root(value, dimension):
242 if value < 0:
243 raise CommandError("NEGATIVE ROOT")
244 if any(exponent % 2 != 0 for exponent in dimension):
245 raise CommandError(f"ODD DIMENSION {dimension_text(dimension)}")
246 numerator_root = isqrt(value.numerator)
247 denominator_root = isqrt(value.denominator)
248 exact_numerator = numerator_root * numerator_root == value.numerator
249 exact_denominator = denominator_root * denominator_root == value.denominator
250 if not exact_numerator or not exact_denominator:
251 raise CommandError("NOT EXACT")
252 root_dimension = tuple(exponent // 2 for exponent in dimension)
253 return Fraction(numerator_root, denominator_root), root_dimension

256def to_kelvin(value, scale):
257 if scale == "C":
258 return value + CELSIUS_OFFSET
259 if scale == "F":
260 return (value + FAHRENHEIT_OFFSET) * 5 / 9
261 if scale == "R":
262 return value * 5 / 9

276def prefix_exponent(value):
277 size = abs(value)
278 k = 0
279 if size >= 1:
280 while k < 9 and size >= Fraction(10) ** (k + 3):
281 k += 3
282 elif size > 0:
283 while k > -9 and size < Fraction(10) ** k:
284 k -= 3
285 return k

289 while index < len(text) and predicate(text[index]):

294def tokenize(text, quantity):
295 symbols = QUANTITY_SYMBOLS if quantity else UNIT_SYMBOLS
296 tokens = []
297 index = 0
298 while index < len(text):
299 char = text[index]
300 kind = "invalid"
301 end = index + 1
302 if char.isalpha():
303 kind = "name"
304 end = run_end(text, index, str.isalpha)
305 elif char.isdigit():
306 kind = "digits"
307 end = run_end(text, index, str.isdigit)
308 if quantity and end + 1 < len(text) and text[end] == "." and text[end + 1].isdigit():
309 kind = "decimal"
310 end = run_end(text, end + 1, str.isdigit)
311 elif char in symbols:
312 kind = "symbol"
313 elif quantity and char == "[":
314 close = text.find("]", index + 1)
315 if close > index + 1:
316 kind = "unit"
317 end = close + 1
318 tokens.append(Token(kind, text[index:end], index + 1))
319 index = end
320 return tokens

323class Parser:
324 def __init__(self, text, quantity):
325 self.text = text
326 self.tokens = tokenize(text, quantity)
327 self.index = 0
328
329 def peek(self):
330 if self.index < len(self.tokens):
331 return self.tokens[self.index]
332 return None
333
334 def at_kind(self, kind):
335 token = self.peek()
336 return token is not None and token.kind == kind
337
338 def at_symbol(self, symbol):
339 return self.at_kind("symbol") and self.peek().text == symbol
340
341 def advance(self):
342 token = self.tokens[self.index]
343 self.index += 1
344 return token
345
346 def fail(self):
347 token = self.peek()
348 position = "END" if token is None else token.position
349 raise CommandError(f"SYNTAX {self.text} AT {position}")
350
351 def expect_kind(self, kind):
352 if not self.at_kind(kind):
353 self.fail()
354 return self.advance()
355
356 def expect_symbol(self, symbol):
357 if not self.at_symbol(symbol):
358 self.fail()
359 self.advance()
360
361 def exponent(self):
362 sign = ""
363 if self.at_symbol("-"):
364 self.advance()
365 sign = "-"
366 return sign + self.expect_kind("digits").text
367
368 def parse(self):
369 tree = self.expression()
370 if self.peek() is not None:
371 self.fail()
372 return tree

375class UnitParser(Parser):
376 def __init__(self, text):
377 super().__init__(text, False)

379 def expression(self):
380 terms = [("*", self.term())]
381 while self.at_symbol("*") or self.at_symbol("/"):
382 operator = self.advance().text
383 terms.append((operator, self.term()))
384 return terms
385
386 def term(self):
387 atom = self.atom()
388 exponent = None
389 if self.at_symbol("^"):
390 self.advance()
391 exponent = self.exponent()
392 return UnitTerm(atom, exponent)
393
394 def atom(self):
395 if self.at_kind("name"):
396 return UnitAtom("name", self.advance().text)
397 if self.at_kind("digits") and not self.peek().text.startswith("0"):
398 return UnitAtom("number", self.advance().text)
399 self.expect_symbol("(")
400 group = self.expression()
401 self.expect_symbol(")")
402 return UnitAtom("group", None, group)

409 def expression(self):
410 node = self.product()
411 while self.at_symbol("+") or self.at_symbol("-"):
412 operator = self.advance().text
413 node = Node("binary", operator, (node, self.product()))
414 return node

416 def product(self):
417 node = self.unary()
418 while self.at_symbol("*") or self.at_symbol("/"):
419 operator = self.advance().text
420 node = Node("binary", operator, (node, self.unary()))
421 return node
422
423 def unary(self):
424 if self.at_symbol("-"):
425 self.advance()
426 return Node("negate", None, (self.unary(),))
427 return self.power()
428
429 def power(self):
430 node = self.primary()
431 if self.at_symbol("^"):
432 self.advance()
433 return Node("power", self.exponent(), (node,))
434 return node
435
436 def primary(self):
437 if self.at_kind("digits") or self.at_kind("decimal"):
438 numeral = self.advance().text
439 unit = None
440 if self.at_kind("unit"):
441 unit = self.advance().text[1:-1]
442 return Node("numeral", numeral, unit=unit)
443 if self.at_kind("unit"):
444 return Node("numeral", "1", unit=self.advance().text[1:-1])
445 if self.at_kind("name"):
446 name = self.advance().text
447 if not self.at_symbol("("):
448 return Node("variable", name)
449 self.advance()
450 arguments = [self.expression()]
451 while self.at_symbol(","):
452 self.advance()
453 arguments.append(self.expression())
454 self.expect_symbol(")")
455 return Node("call", name, arguments)
456 self.expect_symbol("(")
457 node = self.expression()
458 self.expect_symbol(")")
459 return node

462def reduce_terms(terms):
463 coefficient = Fraction(1)
464 exponents = {}
465 for operator, term in terms:
466 term_coefficient, term_exponents = reduce_term(term)
467 sign = -1 if operator == "/" else 1
468 if operator == "/":
469 coefficient /= term_coefficient
470 else:
471 coefficient *= term_coefficient
472 for name, exponent in term_exponents.items():
473 exponents[name] = exponents.get(name, 0) + sign * exponent
474 return coefficient, exponents
475
476
477def reduce_term(term):
478 atom = term.atom
479 if atom.kind == "name":
480 coefficient, exponents = Fraction(1), {atom.text: 1}
481 elif atom.kind == "number":
482 coefficient, exponents = Fraction(int(atom.text)), {}
483 else:
484 coefficient, exponents = reduce_terms(atom.group)
485 if term.exponent is not None:
486 k = int(term.exponent)
487 coefficient = coefficient ** k
488 exponents = {name: exponent * k for name, exponent in exponents.items()}
489 return coefficient, exponents

492def reduced_text(coefficient, exponents):
493 parts = []
494 for name, exponent in exponents.items():
495 if exponent == 1:
496 parts.append(name)
497 elif exponent != 0:
498 parts.append(f"{name}^{exponent}")
499 if coefficient != 1:
500 parts.insert(0, fraction_text(coefficient))
501 return " ".join(parts) if parts else "1"

504class UnitRegistry:
505 def __init__(self):
506 self.units = [
507 Unit(symbol, Fraction(numerator, denominator), KIND_DIMENSIONS[kind], prefixable, True)
508 for symbol, numerator, denominator, kind, prefixable in BUILTIN_UNITS
509 ]

511 def find(self, symbol):
512 for unit in self.units:
513 if unit.symbol == symbol:
514 return unit
515 return None

517 def resolve(self, name):
518 unit = self.find(name)
519 if unit is not None:
520 return unit.factor, unit.dimension
521 for prefix, exponent in PREFIXES:
522 if name.startswith(prefix):
523 unit = self.find(name[len(prefix):])
524 if unit is not None and unit.prefixable:
525 return Fraction(10) ** exponent * unit.factor, unit.dimension
526 return None
527
528 def evaluate(self, terms):
529 factor = Fraction(1)
530 dimension = DIMENSIONLESS
531 for operator, term in terms:
532 term_factor, term_dimension = self.evaluate_term(term)
533 if operator == "/":
534 factor /= term_factor
535 dimension = subtract_dimensions(dimension, term_dimension)
536 else:
537 factor *= term_factor
538 dimension = add_dimensions(dimension, term_dimension)
539 return factor, dimension

541 def evaluate_term(self, term):
542 atom = term.atom
543 if atom.kind == "name":
544 resolved = self.resolve(atom.text)
545 if resolved is None:
546 raise CommandError(f"UNKNOWN UNIT {atom.text}")
547 factor, dimension = resolved
548 elif atom.kind == "number":
549 factor, dimension = Fraction(int(atom.text)), DIMENSIONLESS
550 else:
551 factor, dimension = self.evaluate(atom.group)
552 if term.exponent is not None:
553 k = exponent_value(term.exponent)
554 factor = factor ** k
555 dimension = scale_dimension(dimension, k)
556 return factor, dimension

611 def evaluate(self, node):
612 if node.kind == "numeral":
613 value = decimal_value(node.text)
614 if node.unit is None:
615 return value, DIMENSIONLESS
616 factor, dimension = self.registry.measure(node.unit)
617 return value * factor, dimension
618 if node.kind == "variable":
619 if node.text not in self.variables:
620 raise CommandError(f"UNKNOWN VARIABLE {node.text}")
621 return self.variables[node.text]
622 if node.kind == "call":
623 return self.call(node)
624 if node.kind == "negate":
625 value, dimension = self.evaluate(node.children[0])
626 return -value, dimension
627 if node.kind == "power":
628 value, dimension = self.evaluate(node.children[0])
629 k = exponent_value(node.text)
630 if value == 0 and k < 0:
631 raise CommandError("DIVISION BY ZERO")
632 return value ** k, scale_dimension(dimension, k)
633 return self.binary(node)
634
635 def binary(self, node):
636 left_value, left_dimension = self.evaluate(node.children[0])
637 right_value, right_dimension = self.evaluate(node.children[1])
638 if node.text == "+" or node.text == "-":
639 check_dimensions(left_dimension, right_dimension)
640 if node.text == "+":
641 return left_value + right_value, left_dimension
642 return left_value - right_value, left_dimension
643 if node.text == "*":
644 return left_value * right_value, add_dimensions(left_dimension, right_dimension)
645 if right_value == 0:
646 raise CommandError("DIVISION BY ZERO")
647 return left_value / right_value, subtract_dimensions(left_dimension, right_dimension)

649 def call(self, node):
650 name = node.text
651 if name not in FUNCTION_ARITY:
652 raise CommandError(f"UNKNOWN FUNCTION {name}")
653 if len(node.children) != FUNCTION_ARITY[name]:
654 raise CommandError(f"BAD ARGUMENTS {name}")
655 arguments = [self.evaluate(child) for child in node.children]
656 if name == "abs":
657 value, dimension = arguments[0]
658 return abs(value), dimension
659 if name == "sqrt":
660 value, dimension = arguments[0]
661 return square_root(value, dimension)
662 first, second = arguments
663 check_dimensions(first[1], second[1])
664 if name == "min":
665 return first if first[0] <= second[0] else second
666 return first if first[0] >= second[0] else second

668 def define(self, arguments):
669 name, _, number_text, expression = arguments
670 check_name(name)
671 number = parse_number(number_text)
672 factor, dimension = self.registry.measure(expression)
673 if name in KIND_DIMENSIONS or self.registry.resolve(name) is not None:
674 raise CommandError(f"NAME TAKEN {name}")
675 if number <= 0:
676 raise CommandError(f"NOT POSITIVE {number_text}")
677 unit_factor = number * factor
678 unit = Unit(name, unit_factor, dimension, False, False, (number, expression))
679 self.registry.units.append(unit)
680 self.emit(
681 f"DEFINE {name} factor={fraction_text(unit_factor)} "
682 f"dim={dimension_text(dimension)} kind={kind_name(dimension)}"
683 )

686 name = arguments[0]

690 if unit is None:
691 raise CommandError(f"NOT DEFINED {name}")
692 self.registry.units.remove(unit)

764 first, second, target = self.operands(arguments)

786 def multiply_or_divide(self, command, symbol, arguments):
787 first, second, target = self.operands(arguments)
788 if command == "MUL":
789 dimension = add_dimensions(first.dimension, second.dimension)
790 else:
791 dimension = subtract_dimensions(first.dimension, second.dimension)
792 if second.number == 0:
793 raise CommandError("DIVISION BY ZERO")
794 if target is not None:
795 target_text, target_factor, target_dimension = target
796 check_dimensions(dimension, target_dimension)
797 if command == "MUL":
798 product = first.number * second.number
799 total = first.base_value() * second.base_value()
800 combined = f"{first.expression}*{second.expression}"
801 else:
802 product = first.number / second.number
803 total = first.base_value() / second.base_value()
804 denominator = second.expression
805 if "*" in denominator or "/" in denominator:
806 denominator = f"({denominator})"
807 combined = f"{first.expression}/{denominator}"
808 prefix = f"{command} {first.label()} {symbol} {second.label()} = "
809 if target is not None:
810 self.emit(prefix + quantity_text(total / target_factor, target_text))
811 else:
812 unit = self.registry.simplified(dimension)
813 self.emit(
814 prefix + f"{fraction_text(product)} {combined} = {quantity_text(total, unit)}"

883 def split(self, arguments):
884 source = self.amount(arguments[0], arguments[1])
885 parts = arguments[3].split(",")
886 measures = [self.registry.measure(part) for part in parts]
887 if source.number < 0:
888 raise CommandError("NEGATIVE VALUE")
889 for _, dimension in measures:
890 check_dimensions(source.dimension, dimension)
891 for index in range(len(measures) - 1):
892 if measures[index][0] <= measures[index + 1][0]:
893 raise CommandError("UNIT ORDER")
894 remaining = source.base_value()
895 pieces = []
896 for part, (factor, _) in zip(parts[:-1], measures[:-1]):
897 count = remaining // factor
898 remaining = remaining - count * factor
899 pieces.append(f"{count} {part}")
900 pieces.append(quantity_text(remaining / measures[-1][0], parts[-1]))
901 self.emit(f"SPLIT {source.label()} = " + " ".join(pieces))
