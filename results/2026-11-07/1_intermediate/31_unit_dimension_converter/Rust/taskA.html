9#[derive(Clone, Copy, PartialEq, Eq)]
10struct Rational { n: i64, d: i64 }
11impl Rational {
12 const ONE: Self = Self { n: 1, d: 1 };
13 fn new(mut n: i64, mut d: i64) -> Self {
14 if d < 0 { n = -n; d = -d; }
15 let (mut a, mut b) = (n.abs(), d);
16 while b != 0 { (a, b) = (b, a % b); }
17 Self { n: n / a, d: d / a }
18 }
19 fn integer(n: i64) -> Self { Self { n, d: 1 } }
20 fn pow(mut self, mut exponent: i32) -> Self {
21 if exponent < 0 { self = Self::new(self.d, self.n); exponent = -exponent; }
22 let mut result = Self::ONE;
23 for _ in 0..exponent { result = result * self; }
24 result
25 }
26 fn abs(self) -> Self { Self { n: self.n.abs(), d: self.d } }
27 fn decimal(self) -> String {
28 let magnitude = self.n.abs();
29 format!("{}{}.{:06}", if self.n < 0 { "-" } else { "" }, magnitude / self.d, magnitude % self.d * 1_000_000 / self.d)
30 }
31 fn notation(self, unit: &str) -> String { format!("{self} {unit} ({})", self.decimal()) }
32}
33impl Display for Rational {
34 fn fmt(&self, out: &mut fmt::Formatter<'_>) -> fmt::Result {
35 if self.d == 1 { write!(out, "{}", self.n) } else { write!(out, "{}/{}", self.n, self.d) }
36 }
37}
38impl Add for Rational { type Output = Self; fn add(self, rhs: Self) -> Self { Self::new(self.n * rhs.d + rhs.n * self.d, self.d * rhs.d) } }
39impl Sub for Rational { type Output = Self; fn sub(self, rhs: Self) -> Self { Self::new(self.n * rhs.d - rhs.n * self.d, self.d * rhs.d) } }
40impl Mul for Rational { type Output = Self; fn mul(self, rhs: Self) -> Self { Self::new(self.n * rhs.n, self.d * rhs.d) } }
41impl Div for Rational { type Output = Self; fn div(self, rhs: Self) -> Self { Self::new(self.n * rhs.d, self.d * rhs.n) } }
42impl Neg for Rational { type Output = Self; fn neg(self) -> Self { Self { n: -self.n, d: self.d } } }
43impl Ord for Rational { fn cmp(&self, rhs: &Self) -> Ordering { (self.n * rhs.d).cmp(&(rhs.n * self.d)) } }
44impl PartialOrd for Rational { fn partial_cmp(&self, rhs: &Self) -> Option<Ordering> { Some(self.cmp(rhs)) } }

46#[derive(Clone, Copy, Default, PartialEq, Eq)]
47struct Dimension([i64; 7]);
48impl Dimension {
49 fn combine(self, rhs: Self, sign: i64) -> Self { Self(std::array::from_fn(|i| self.0[i] + sign * rhs.0[i])) }
50 fn scale(self, factor: i64) -> Self { Self(self.0.map(|x| x * factor)) }
51 fn dimensionless(self) -> bool { self.0 == [0; 7] }
52 fn matches(self, rhs: Self) -> Result<()> {
53 if self == rhs { Ok(()) } else { Err(format!("DIMENSION MISMATCH {self} vs {rhs}")) }
54 }
55}
56impl Display for Dimension {
57 fn fmt(&self, out: &mut fmt::Formatter<'_>) -> fmt::Result {
58 let mut first = true;
59 for (name, exponent) in ["kg", "m", "s", "A", "K", "mol", "cd"].iter().zip(self.0) {
60 if exponent == 0 { continue; }
61 if !first { write!(out, " ")?; }
62 first = false;
63 write!(out, "{name}")?;
64 if exponent != 1 { write!(out, "^{exponent}")?; }
65 }
66 if first { write!(out, "1") } else { Ok(()) }
67 }

72fn number(text: &str) -> Result<Rational> {
73 let unsigned = text.strip_prefix('-').unwrap_or(text);
74 let digits = |part: &str| !part.is_empty() && part.bytes().all(|c| c.is_ascii_digit());
75 let parse = |part: &str| part.parse::<i64>().unwrap();
76 let value = if digits(unsigned) {
77 Rational::integer(parse(unsigned))
78 } else if let Some((whole, tail)) = unsigned.split_once('.') {
79 if !digits(whole) || !digits(tail) { return Err(format!("BAD NUMBER {text}")); }
80 let mut scale = 1;
81 for _ in tail.bytes() { scale *= 10; }
82 Rational::new(parse(whole) * scale + parse(tail), scale)
83 } else if let Some((n, d)) = unsigned.split_once('/') {
84 if !digits(n) || !digits(d) || parse(d) == 0 { return Err(format!("BAD NUMBER {text}")); }
85 Rational::new(parse(n), parse(d))
86 } else { return Err(format!("BAD NUMBER {text}")); };
87 Ok(if text.starts_with('-') { -value } else { value })
88}
89fn exponent(text: &str) -> Result<i32> {
90 let value = text.parse::<i32>().unwrap();
91 if value == 0 || !(-3..=3).contains(&value) { Err(format!("BAD EXPONENT {text}")) } else { Ok(value) }
92}
93fn check_name(name: &str, variable: bool) -> Result<()> {
94 if name.is_empty() || name.len() > 12 || !name.bytes().all(|c| c.is_ascii_alphabetic()) || variable && ["abs", "sqrt", "min", "max"].contains(&name) {
95 Err(format!("BAD NAME {name}"))
96 } else { Ok(()) }
97}
98fn integer_sqrt(n: i64) -> i64 {
99 if n < 2 { return n; }
100 let (mut x, mut next) = (n, (n + 1) / 2);
101 while next < x { x = next; next = (x + n / x) / 2; }
102 x
103}

125#[derive(Clone, Copy, PartialEq, Eq)]
126enum TokenKind { Identifier, Number, Unit, Punctuation(u8), Invalid, End }
127#[derive(Clone, Copy)]
128struct Token<'a> { kind: TokenKind, start: usize, end: usize, text: &'a str }
129struct Parser<'a> { source: &'a str, tokens: Vec<Token<'a>>, position: usize }
130enum RawUnit<'a> {
131 Number(&'a str), Name(&'a str), Power(Box<RawUnit<'a>>, &'a str), Product(Vec<(u8, RawUnit<'a>)>),
132}
133enum UnitExpr<'a> {
134 Number(i64), Name(&'a str, Quantity), Power(Box<UnitExpr<'a>>, i32), Product(Vec<(u8, UnitExpr<'a>)>),
135}
136enum Expr<'a> {
137 Number(&'a str), Unit(&'a str, &'a str), Variable(&'a str), Power(Box<Expr<'a>>, &'a str),
138 Negative(Box<Expr<'a>>), Binary(u8, Box<Expr<'a>>, Box<Expr<'a>>), Call(&'a str, Vec<Expr<'a>>),
139}
140impl<'a> Parser<'a> {
141 fn new(source: &'a str, quantity: bool) -> Self {
142 let bytes = source.as_bytes();
143 let mut tokens = Vec::new();
144 let mut i = 0;
145 while i < bytes.len() {
146 let start = i;
147 let kind;
148 if bytes[i].is_ascii_alphabetic() {
149 kind = TokenKind::Identifier;
150 while bytes.get(i).is_some_and(u8::is_ascii_alphabetic) { i += 1; }
151 } else if bytes[i].is_ascii_digit() {
152 kind = TokenKind::Number;
153 while bytes.get(i).is_some_and(u8::is_ascii_digit) { i += 1; }
154 if quantity && bytes.get(i) == Some(&b'.') && bytes.get(i + 1).is_some_and(u8::is_ascii_digit) {
155 i += 1;
156 while bytes.get(i).is_some_and(u8::is_ascii_digit) { i += 1; }
157 }
158 } else if quantity && bytes[i] == b'[' && bytes[i + 1..].iter().position(|c| *c == b']').is_some_and(|end| end > 0) {
159 kind = TokenKind::Unit;
160 i += bytes[i + 1..].iter().position(|c| *c == b']').unwrap() + 2;
161 } else {
162 let punctuation = if quantity { b"+-*/^(),".as_slice() } else { b"*/^()-".as_slice() };
163 kind = if punctuation.contains(&bytes[i]) { TokenKind::Punctuation(bytes[i]) } else { TokenKind::Invalid };
164 i += 1;
165 }
166 tokens.push(Token { kind, start, end: i, text: &source[start..i] });
167 }
168 tokens.push(Token { kind: TokenKind::End, start: i, end: i, text: "" });
169 Self { source, tokens, position: 0 }
170 }
171 fn current(&self) -> Token<'a> { self.tokens[self.position] }
172 fn syntax(&self) -> String {
173 let at = if self.current().kind == TokenKind::End { "END".to_owned() } else { (self.current().start + 1).to_string() };
174 format!("SYNTAX {} AT {at}", self.source)
175 }
176 fn accept(&mut self, kind: TokenKind) -> bool {
177 if self.current().kind != kind { return false; }
178 self.position += 1; true
179 }
180 fn eat(&mut self, c: u8) -> bool { self.accept(TokenKind::Punctuation(c)) }
181 fn expect(&mut self, c: u8) -> Result<()> { if self.eat(c) { Ok(()) } else { Err(self.syntax()) } }
182 fn exponent(&mut self) -> Result<&'a str> {
183 let start = self.current().start;
184 self.eat(b'-');
185 let token = self.current();
186 if token.kind != TokenKind::Number || token.text.contains('.') { return Err(self.syntax()); }
187 self.position += 1;
188 Ok(&self.source[start..token.end])
189 }
190 fn unit_term(&mut self) -> Result<RawUnit<'a>> {
191 let token = self.current();
192 let mut result = if self.accept(TokenKind::Identifier) {
193 RawUnit::Name(token.text)
194 } else if token.kind == TokenKind::Number && !token.text.starts_with('0') {
195 self.position += 1; RawUnit::Number(token.text)
196 } else if self.eat(b'(') {
197 let inner = self.unit_expression()?; self.expect(b')')?; inner
198 } else { return Err(self.syntax()); };
199 if self.eat(b'^') { result = RawUnit::Power(Box::new(result), self.exponent()?); }
200 Ok(result)
201 }
202 fn unit_expression(&mut self) -> Result<RawUnit<'a>> {
203 let mut terms = vec![(b'*', self.unit_term()?)];
204 while let TokenKind::Punctuation(op @ (b'*' | b'/')) = self.current().kind {
205 self.position += 1; terms.push((op, self.unit_term()?));
206 }
207 Ok(RawUnit::Product(terms))
208 }
209 fn unit_complete(mut self) -> Result<RawUnit<'a>> {
210 let result = self.unit_expression()?;
211 if self.current().kind != TokenKind::End { return Err(self.syntax()); }
212 Ok(result)
213 }
214 fn primary(&mut self) -> Result<Expr<'a>> {
215 let token = self.current();
216 let mut result = if self.accept(TokenKind::Number) {
217 if self.current().kind == TokenKind::Unit {
218 let unit = self.current().text; self.position += 1;
219 Expr::Unit(token.text, &unit[1..unit.len() - 1])
220 } else { Expr::Number(token.text) }
221 } else if self.accept(TokenKind::Unit) {
222 Expr::Unit("1", &token.text[1..token.text.len() - 1])
223 } else if self.accept(TokenKind::Identifier) {
224 if self.eat(b'(') {
225 let mut args = vec![self.expression()?];
226 while self.eat(b',') { args.push(self.expression()?); }
227 self.expect(b')')?; Expr::Call(token.text, args)
228 } else { Expr::Variable(token.text) }
229 } else if self.eat(b'(') {
230 let inner = self.expression()?; self.expect(b')')?; inner
231 } else { return Err(self.syntax()); };
232 if self.eat(b'^') { result = Expr::Power(Box::new(result), self.exponent()?); }
233 Ok(result)
234 }
235 fn unary(&mut self) -> Result<Expr<'a>> {
236 if self.eat(b'-') { Ok(Expr::Negative(Box::new(self.unary()?))) } else { self.primary() }
237 }
238 fn product(&mut self) -> Result<Expr<'a>> {
239 let mut result = self.unary()?;
240 while let TokenKind::Punctuation(op @ (b'*' | b'/')) = self.current().kind {
241 self.position += 1;
242 let right = self.unary()?; result = Expr::Binary(op, Box::new(result), Box::new(right));
243 }
244 Ok(result)
245 }
246 fn expression(&mut self) -> Result<Expr<'a>> {
247 let mut result = self.product()?;
248 while let TokenKind::Punctuation(op @ (b'+' | b'-')) = self.current().kind {
249 self.position += 1;
250 let right = self.product()?; result = Expr::Binary(op, Box::new(result), Box::new(right));
251 }
252 Ok(result)
253 }
254 fn complete(mut self) -> Result<Expr<'a>> {
255 let result = self.expression()?;
256 if self.current().kind != TokenKind::End { return Err(self.syntax()); }
257 Ok(result)
258 }
259}
260impl<'a> RawUnit<'a> {
261 fn resolve(self, db: &Database) -> Result<UnitExpr<'a>> {
262 Ok(match self {
263 Self::Number(text) => UnitExpr::Number(text.parse().unwrap()),
264 Self::Name(name) => UnitExpr::Name(name, db.resolve(name).ok_or_else(|| format!("UNKNOWN UNIT {name}"))?),
265 Self::Power(inner, text) => {
266 let resolved = inner.resolve(db)?;
267 UnitExpr::Power(Box::new(resolved), exponent(text)?)
268 }
269 Self::Product(terms) => {
270 let mut resolved = Vec::new();
271 for (op, term) in terms { resolved.push((op, term.resolve(db)?)); }
272 UnitExpr::Product(resolved)
273 }
274 })
275 }

277struct Reduced<'a> { coefficient: Rational, names: Vec<(&'a str, i64)> }
278impl<'a> UnitExpr<'a> {
279 fn value(&self) -> Quantity {
280 match self {
281 Self::Number(n) => Quantity { value: Rational::integer(*n), dim: Dimension::default() },
282 Self::Name(_, quantity) => *quantity,
283 Self::Power(inner, k) => {
284 let quantity = inner.value();
285 Quantity { value: quantity.value.pow(*k), dim: quantity.dim.scale(i64::from(*k)) }
286 }
287 Self::Product(terms) => {
288 let mut result = Quantity { value: Rational::ONE, dim: Dimension::default() };
289 for (op, term) in terms {
290 let right = term.value();
291 result.value = if *op == b'*' { result.value * right.value } else { result.value / right.value };
292 result.dim = result.dim.combine(right.dim, if *op == b'*' { 1 } else { -1 });
293 }
294 result
295 }
296 }
297 }
298 fn reduce(&self) -> Reduced<'a> {
299 match self {
300 Self::Number(n) => Reduced { coefficient: Rational::integer(*n), names: Vec::new() },
301 Self::Name(name, _) => Reduced { coefficient: Rational::ONE, names: vec![(*name, 1)] },
302 Self::Power(inner, k) => {
303 let mut result = inner.reduce(); result.coefficient = result.coefficient.pow(*k);
304 for (_, count) in &mut result.names { *count *= i64::from(*k); }
305 result
306 }
307 Self::Product(terms) => {
308 let mut result = Reduced { coefficient: Rational::ONE, names: Vec::new() };
309 for (op, term) in terms {
310 let right = term.reduce();
311 let sign = if *op == b'*' { 1 } else { -1 };
312 result.coefficient = if sign == 1 { result.coefficient * right.coefficient } else { result.coefficient / right.coefficient };
313 for (name, count) in right.names {
314 if let Some((_, existing)) = result.names.iter_mut().find(|(key, _)| *key == name) { *existing += sign * count; }
315 else { result.names.push((name, sign * count)); }
316 }
317 }
318 result
319 }
320 }
321 }
322 fn reduced_text(&self) -> String {
323 let result = self.reduce();
324 let mut parts = Vec::new();
325 if result.coefficient != Rational::ONE { parts.push(result.coefficient.to_string()); }
326 for (name, count) in result.names {
327 if count != 0 { parts.push(if count == 1 { name.to_owned() } else { format!("{name}^{count}") }); }
328 }
329 if parts.is_empty() { "1".to_owned() } else { parts.join(" ") }
330 }

332impl Expr<'_> {
333 fn evaluate(&self, db: &Database) -> Result<Quantity> {
334 match self {
335 Self::Number(text) => Ok(Quantity { value: number(text)?, dim: Dimension::default() }),
336 Self::Unit(text, unit) => {
337 let mut result = db.unit(unit)?; result.value = number(text)? * result.value; Ok(result)
338 }
339 Self::Variable(name) => db.variables.get(*name).copied().ok_or_else(|| format!("UNKNOWN VARIABLE {name}")),
340 Self::Power(inner, text) => {
341 let quantity = inner.evaluate(db)?; let k = exponent(text)?;
342 if quantity.value.n == 0 && k < 0 { return Err("DIVISION BY ZERO".to_owned()); }
343 Ok(Quantity { value: quantity.value.pow(k), dim: quantity.dim.scale(i64::from(k)) })
344 }
345 Self::Negative(inner) => { let mut result = inner.evaluate(db)?; result.value = -result.value; Ok(result) }
346 Self::Binary(op, left, right) => {
347 let a = left.evaluate(db)?; let b = right.evaluate(db)?;
348 if *op == b'+' || *op == b'-' {
349 a.dim.matches(b.dim)?;
350 Ok(Quantity { value: if *op == b'+' { a.value + b.value } else { a.value - b.value }, dim: a.dim })
351 } else {
352 if *op == b'/' && b.value.n == 0 { return Err("DIVISION BY ZERO".to_owned()); }
353 Ok(Quantity { value: if *op == b'*' { a.value * b.value } else { a.value / b.value }, dim: a.dim.combine(b.dim, if *op == b'*' { 1 } else { -1 }) })
354 }
355 }
356 Self::Call(name, args) => {
357 let arity = match *name { "abs" | "sqrt" => 1, "min" | "max" => 2, _ => return Err(format!("UNKNOWN FUNCTION {name}")) };
358 if args.len() != arity { return Err(format!("BAD ARGUMENTS {name}")); }
359 let mut a = args[0].evaluate(db)?;
360 match *name {
361 "abs" => { a.value = a.value.abs(); Ok(a) }
362 "sqrt" => {
363 if a.value.n < 0 { return Err("NEGATIVE ROOT".to_owned()); }
364 if a.dim.0.iter().any(|x| x % 2 != 0) { return Err(format!("ODD DIMENSION {}", a.dim)); }
365 let n = integer_sqrt(a.value.n); let d = integer_sqrt(a.value.d);
366 if n * n != a.value.n || d * d != a.value.d { return Err("NOT EXACT".to_owned()); }
367 Ok(Quantity { value: Rational::new(n, d), dim: Dimension(a.dim.0.map(|x| x / 2)) })
368 }
369 _ => {
370 let b = args[1].evaluate(db)?; a.dim.matches(b.dim)?;
371 let take_left = if *name == "min" { a.value <= b.value } else { a.value >= b.value };
372 Ok(if take_left { a } else { b })
373 }
374 }
375 }
376 }
377 }

418 }
419 fn parse_unit<'a>(&self, text: &'a str) -> Result<UnitExpr<'a>> { Parser::new(text, false).unit_complete()?.resolve(self) }
420 fn unit(&self, text: &str) -> Result<Quantity> { Ok(self.parse_unit(text)?.value()) }
421 fn quantity(&self, text: &str) -> Result<Quantity> { Parser::new(text, true).complete()?.evaluate(self) }
422 fn simplified(&self, dim: Dimension) -> String {
423 self.units.iter().find(|unit| unit.quantity.value == Rational::ONE && unit.quantity.dim == dim).map_or_else(|| dim.to_string(), |unit| unit.symbol.clone())
424 }
425 fn selected(&self, dim: Dimension) -> Vec<&Unit> {
426 let mut units: Vec<_> = self.units.iter().filter(|unit| unit.quantity.dim == dim).collect();
427 units.sort_by(|a, b| a.quantity.value.cmp(&b.quantity.value).then_with(|| a.symbol.cmp(&b.symbol)));
428 units
429 }
430 fn command(&mut self, line: &str) -> Result<Vec<String>> {
431 let w: Vec<_> = line.split(' ').collect();
432 let mut output = Vec::new();
433 match w[0] {
434 "DEFINE" => {
435 check_name(w[1], false)?;
436 let value = number(w[3])?; let unit = self.unit(w[4])?;
437 if kind_dimension(w[1]).is_some() || self.resolve(w[1]).is_some() { return Err(format!("NAME TAKEN {}", w[1])); }
438 if value.n <= 0 { return Err(format!("NOT POSITIVE {}", w[3])); }
439 let factor = value * unit.value;
440 self.units.push(Unit { symbol: w[1].to_owned(), quantity: Quantity { value: factor, dim: unit.dim }, prefix: false,
441 definition: Some(Definition { value, unit: w[4].to_owned() }) });
442 output.push(format!("DEFINE {} factor={factor} dim={} kind={}", w[1], unit.dim, kind_name(unit.dim)));
443 }
444 "UNDEF" => {
445 let index = self.units.iter().position(|unit| unit.symbol == w[1]).ok_or_else(|| format!("NOT DEFINED {}", w[1]))?;
446 if index < 42 { return Err(format!("BUILTIN UNIT {}", w[1])); }
447 self.units.remove(index); output.push(format!("UNDEF {}", w[1]));
448 }
449 "DEFS" => {
450 output.push(format!("DEFS count={}", self.units.len() - 42));
451 for unit in &self.units[42..] {
452 let definition = unit.definition.as_ref().unwrap();
453 output.push(format!("DEF {} = {} {} factor={} dim={}", unit.symbol, definition.value, definition.unit, unit.quantity.value, unit.quantity.dim));
454 }
455 }
456 "CONVERT" | "INVERT" => {
457 let value = number(w[1])?; let from = self.unit(w[2])?; let to = self.unit(w[4])?;
458 let inverse = w[0] == "INVERT";
459 if inverse {
460 if from.dim.dimensionless() || !from.dim.combine(to.dim, 1).dimensionless() { return Err(format!("NOT RECIPROCAL {} vs {}", from.dim, to.dim)); }
461 if value.n == 0 { return Err("DIVISION BY ZERO".to_owned()); }
462 } else { from.dim.matches(to.dim)?; }
463 let mut result = value * from.value;
464 if inverse { result = Rational::ONE / result; }
465 result = result / to.value;
466 output.push(format!("{} {value} {} = {}", w[0], w[2], result.notation(w[4])));
467 }
468 "DIM" | "REDUCE" | "SIMPLIFY" => {
469 let expression = self.parse_unit(w[1])?; let unit = expression.value();
470 output.push(match w[0] {
471 "DIM" => format!("DIM {} = {} factor={} kind={}", w[1], unit.dim, unit.value, kind_name(unit.dim)),
472 "REDUCE" => format!("REDUCE {} = {}", w[1], expression.reduced_text()),
473 _ => format!("SIMPLIFY {} = {}", w[1], unit.value.notation(&self.simplified(unit.dim))),
474 });
475 }
476 "ADD" | "SUB" | "MUL" | "DIV" | "COMPARE" => {
477 let a = number(w[1])?; let first = self.unit(w[2])?;
478 let b = number(w[3])?; let second = self.unit(w[4])?;
479 let target = w.len() == 7;
480 let destination = if target { self.unit(w[6])? } else { first };
481 if ["ADD", "SUB", "COMPARE"].contains(&w[0]) {
482 first.dim.matches(second.dim)?;
483 if target { first.dim.matches(destination.dim)?; }
484 let left = a * first.value; let right = b * second.value;
485 if w[0] == "COMPARE" {
486 let relation = match left.cmp(&right) { Ordering::Less => '<', Ordering::Equal => '=', Ordering::Greater => '>' };
487 let ratio = if right.n == 0 { "undefined".to_owned() } else { (left / right).to_string() };
488 output.push(format!("COMPARE {a} {} {relation} {b} {} ratio={ratio}", w[2], w[4]));
489 } else {
490 let op = if w[0] == "ADD" { '+' } else { '-' };
491 let sum = if op == '+' { left + right } else { left - right };
492 let result = sum / destination.value;
493 output.push(format!("{} {a} {} {op} {b} {} = {}", w[0], w[2], w[4], result.notation(if target { w[6] } else { w[2] })));
494 }
495 } else {
496 let product = w[0] == "MUL";
497 let dim = first.dim.combine(second.dim, if product { 1 } else { -1 });
498 if !product && b.n == 0 { return Err("DIVISION BY ZERO".to_owned()); }
499 if target { dim.matches(destination.dim)?; }
500 let raw = if product { a * b } else { a / b };
501 let left = a * first.value; let right = b * second.value;
502 let coherent = if product { left * right } else { left / right };
503 let op = if product { '*' } else { '/' };
504 let prefix = format!("{} {a} {} {op} {b} {} = ", w[0], w[2], w[4]);
505 if target { output.push(prefix + &(coherent / destination.value).notation(w[6])); }
506 else {
507 let right_unit = if !product && w[4].contains(['*', '/']) { format!("({})", w[4]) } else { w[4].to_owned() };
508 output.push(format!("{prefix}{raw} {}{op}{right_unit} = {}", w[2], coherent.notation(&self.simplified(dim))));
509 }
510 }
511 }
512 "TEMP" => {
513 let value = number(w[1])?;
514 for scale in [w[2], w[4]] { if !["C", "F", "K", "R"].contains(&scale) { return Err(format!("UNKNOWN SCALE {scale}")); } }
515 let kelvin = match w[2] {
516 "C" => value + Rational::new(5463, 20),
517 "F" => (value + Rational::new(45967, 100)) * Rational::new(5, 9),
518 "R" => value * Rational::new(5, 9), _ => value,
519 };
520 if kelvin.n < 0 { return Err("BELOW ABSOLUTE ZERO".to_owned()); }
521 let result = match w[4] {
522 "C" => kelvin - Rational::new(5463, 20),
523 "F" => kelvin * Rational::new(9, 5) - Rational::new(45967, 100),
524 "R" => kelvin * Rational::new(9, 5), _ => kelvin,
525 };
526 output.push(format!("TEMP {value} {} = {}", w[2], result.notation(w[4])));
527 }
528 "LIST" => {
529 let dim = match kind_dimension(w[1]) { Some(dim) => dim, None => self.unit(w[1])?.dim };
530 let units = self.selected(dim);
531 output.push(format!("LIST {} dim={dim} kind={} count={}", w[1], kind_name(dim), units.len()));
532 for unit in units { output.push(format!("UNIT {} factor={} origin={} prefix={}", unit.symbol, unit.quantity.value,
533 if unit.definition.is_none() { "builtin" } else { "user" }, if unit.prefix { "yes" } else { "no" })); }
534 }
535 "TABLE" => {
536 let value = number(w[1])?; let from = self.unit(w[2])?; let units = self.selected(from.dim);
537 output.push(format!("TABLE {value} {} dim={} count={}", w[2], from.dim, units.len()));
538 for unit in units { output.push(format!("ROW {}", ((value * from.value) / unit.quantity.value).notation(&unit.symbol))); }
539 }
540 "AUTO" => {
541 let value = number(w[1])?; let unit = self.unit(w[2])?;
542 let reference = self.units[..42].iter().find(|item| item.prefix && item.quantity.dim == unit.dim)
543 .ok_or_else(|| format!("NO PREFIXABLE UNIT {}", unit.dim))?;
544 let scaled = (value * unit.value) / reference.quantity.value;
545 let magnitude = scaled.abs(); let mut k = 0;
546 if magnitude.n != 0 {
547 if magnitude >= Rational::ONE {
548 while k < 9 && magnitude >= Rational::integer(10).pow(k + 3) { k += 3; }
549 } else {
550 while k > -9 && magnitude < Rational::integer(10).pow(k) { k -= 3; }
551 }
552 }
553 let prefix = ["n", "u", "m", "", "k", "M", "G"][((k + 9) / 3) as usize];
554 let symbol = format!("{prefix}{}", reference.symbol);
555 output.push(format!("AUTO {value} {} = {}", w[2], (scaled / Rational::integer(10).pow(k)).notation(&symbol)));
556 }
557 "SPLIT" => {
558 let value = number(w[1])?; let unit = self.unit(w[2])?;
559 let parts: Vec<_> = w[4].split(',').collect();
560 let mut destinations = Vec::new();
561 for part in &parts { destinations.push(self.unit(part)?); }
562 if value.n < 0 { return Err("NEGATIVE VALUE".to_owned()); }
563 for destination in &destinations { unit.dim.matches(destination.dim)?; }
564 if destinations.windows(2).any(|pair| pair[0].value <= pair[1].value) { return Err("UNIT ORDER".to_owned()); }
565 let mut remaining = value * unit.value; let mut pieces = Vec::new();
566 for i in 0..parts.len() - 1 {
567 let quotient = remaining / destinations[i].value; let count = quotient.n / quotient.d;
568 remaining = remaining - Rational::integer(count) * destinations[i].value;
569 pieces.push(format!("{count} {}", parts[i]));
570 }
571 pieces.push((remaining / destinations.last().unwrap().value).notation(parts.last().unwrap()));
572 output.push(format!("SPLIT {value} {} = {}", w[2], pieces.join(" ")));
573 }
574 "LET" => {
575 check_name(w[1], true)?; let quantity = self.quantity(w[3])?;
576 self.variables.insert(w[1].to_owned(), quantity);
577 output.push(format!("LET {} = {}", w[1], quantity.value.notation(&self.simplified(quantity.dim))));
578 }
579 "EVAL" => {
580 let quantity = self.quantity(w[1])?;
581 let result = if w.len() == 4 {
582 let unit = self.unit(w[3])?; quantity.dim.matches(unit.dim)?;
583 (quantity.value / unit.value).notation(w[3])
584 } else { quantity.value.notation(&self.simplified(quantity.dim)) };
585 output.push(format!("EVAL {} = {result}", w[1]));
586 }
587 "VARS" => {
588 output.push(format!("VARS count={}", self.variables.len()));
589 for (name, quantity) in &self.variables { output.push(format!("VAR {name} = {}", quantity.value.notation(&self.simplified(quantity.dim)))); }
590 }
591 _ => return Err(format!("UNKNOWN COMMAND {}", w[0])),
592 }
593 Ok(output)
594 }
595}
596
597fn main() -> io::Result<()> {
598 let mut input = String::new(); io::stdin().read_to_string(&mut input)?;
599 let mut lines = input.lines(); let count: usize = lines.next().unwrap().parse().unwrap();
600 let mut database = Database::new(); let mut out = io::BufWriter::new(io::stdout().lock());
601 for line in lines.take(count) {
602 match database.command(line) {
603 Ok(output) => for text in output { writeln!(out, "{text}")?; },
604 Err(error) => writeln!(out, "ERROR {error}")?,
605 }
606 }
607 out.flush()
608}
