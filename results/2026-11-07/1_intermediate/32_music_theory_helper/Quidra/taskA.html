48int | error parse_frequency(string text, bool tuning)
49 parts := text.split(".")
50 if #parts > 2 or #parts[0] > 5 or not digits_only(parts[0])

153int quality_delta(int quality, int number)
154 remainder := (number - 1) % 7

158class Interval self(int quality, int number)
159 string text()
160 return interval_qualities()[quality] + string(number)
161
162 bool forbidden()
163 return number == 1 and (quality == 4 or quality == 6)

165 Tone tone()
166 letters := number - 1
167 return Tone(letters, 12 * (letters / 7) + natural_pitch(letters % 7) + quality_delta(quality, number))

169 Interval simple()
170 reduced := number

175 string consonance()
176 Interval reduced := simple()
177 if quality == 0 and (reduced.number == 1 or reduced.number == 5 or reduced.number == 8)
178 return "perfect"
179 if (quality == 1 or quality == 2) and (reduced.number == 3 or reduced.number == 6)

183Interval | error parse_interval(string text)
184 qualities := interval_qualities()
185 for quality := qualities.indices()[]
186 prefix := qualities[quality]
187 if text.starts_with(prefix)
188 rest := text[#prefix:]
189 if rest == "" or #rest > 2 or rest[0] == "0" or not digits_only(rest)
190 continue
191 number := int(rest)
192 interval := Interval(int(quality), number)
193 if quality_delta(int(quality), number) != -99 and not interval.forbidden()
194 return interval
195 return error("BAD INTERVAL " + text)

199Motion | error between(Note first, Note second)
200 distance := second.position() - first.position()
201 semitones := second.midi() - first.midi()
202 down := distance < 0 or (distance == 0 and semitones < 0)
203 if down
204 distance = -distance
205 semitones = -semitones
206 difference := semitones - 12 * (distance / 7) - natural_pitch(distance % 7)
207 for: 0 <= quality < 7
208 if quality_delta(quality, distance + 1) == difference
209 return Motion(Interval(quality, distance + 1), down, semitones)
210 return error("UNNAMEABLE INTERVAL")

399string identify(const Note[] notes, const Catalog catalog)
400 width := int(#notes)
401 int[] order := []
402 for: 0 <= index < width
403 order = order.append(notes[index].midi() * width + index)
404 order = order.sorted()
405 Note[] candidates := []
406 present := array(12, fill = false)
407 for rank := order[]
408 note := notes[rank % width]
409 pitch_class := note.pitch.pitch_class()
410 if not present[pitch_class]
411 candidates = candidates.append(note)
412 present[pitch_class] = true
413 bass := candidates[0].pitch
414 for root := candidates[]
415 for chord := catalog.chords[]
416 possible := array(12, fill = false)
417 for tone := chord.tones[]
418 possible[modulo(root.midi() + tone.semitones, 12)] = true
419 if possible != present
420 continue
421 inversion := 0
422 while modulo(root.midi() + chord.tones[inversion].semitones, 12) != bass.pitch_class()
423 inversion += 1
424 Pitch | none slash := if inversion == 0 then none else bass
425 symbol := chord_symbol(root.pitch, chord, slash)
426 return "IDENTIFY {symbol} root={root.pitch.text()} quality={chord.name} inversion={inversion}"
427 return "IDENTIFY UNKNOWN"

429string | error melody(const Note[] notes, const string[] words)
430 int[] values := []
431 for note := notes[]
432 values = values.append(note.midi())
433 low := first_index(values, values.min())
434 high := first_index(values, values.max())
435 range := try between(notes[low], notes[high])
436 leaps := 0
437 string[] steps := []
438 for: 1 <= index < int(#notes)
439 motion := try between(notes[index - 1], notes[index])
440 steps = steps.append((if motion.down then "-" else "+") + motion.interval.text())
441 if math.abs(values[index] - values[index - 1]) >= 3
442 leaps += 1
443 return "MELODY n={#notes} low={words[low]} high={words[high]} range={range.interval.text()} leaps={leaps} steps=" + steps.join(",")

459string[] diatonic(Key key, const Catalog catalog)
460 rows := ["DIATONIC " + key.text()]
461 for: 3 <= size <= 4
462 string[] items := []
463 for: 0 <= index < 7
464 root := key.scale[index]
465 third := modulo(key.scale[(index + 2) % 7].pitch_class() - root.pitch_class(), 12)
466 fifth := modulo(key.scale[(index + 4) % 7].pitch_class() - root.pitch_class(), 12)

468 if size == 3

470 else
471 seventh := modulo(key.scale[(index + 6) % 7].pitch_class() - root.pitch_class(), 12)
472 quality = if third == 4 then (if seventh == 11 then 9 else 8) elif fifth == 7 then 10 else 12
473 chord := catalog.chords[quality]
474 items = items.append(roman(index, chord) + "=" + chord_symbol(root, chord))
475 rows = rows.append((if size == 3 then "TRIADS " else "SEVENTHS ") + items.join(" "))

480string[] | error execute(Calculator &calculator, const Catalog catalog, const string[] words)
481 command := words[0]
482 commands := "MIDI NOTE FREQ TUNING PITCH INTERVAL INVERT TRANSPOSE SCALE CHORD VOICE IDENTIFY MELODY ENHARMONIC KEYSIG RELATIVE CIRCLE DIATONIC PROGRESSION ANALYZE FINDKEY DEGREE SPELL KEYDIST".split(" ")
483 minimum := [1,1,1,1,1,2,1,3,2,2,2,2,2,1,2,2,1,2,3,3,1,3,3,4]
484 maximum := [1,1,1,1,1,2,1,3,2,3,3,15,15,1,2,2,1,2,15,15,15,3,3,4]
485 command_index := -1
486 for index := commands.indices()[]
487 if command == commands[index]
488 command_index = int(index)
489 break
490 if command_index < 0
491 return error("UNKNOWN COMMAND " + command)
492 count := int(#words) - 1
493 if count < minimum[command_index] or count > maximum[command_index]
494 return error("ARGUMENT COUNT")
495 arguments := words[1:]
496 if command == "MIDI" or command == "FREQ" or command == "ENHARMONIC"
497 note := try parse_note(arguments[0])
498 value := note.midi()
499 if command == "MIDI"
500 return ["MIDI {arguments[0]} {value}"]
501 if command == "FREQ"
502 exponent := real64(value - 69) / 12.0
503 cents := math.floor(real64(calculator.tuning) * (2.0 ^ exponent) + 0.5)
504 return ["FREQ {arguments[0]} " + frequency_text(cents)]
505 Note[] equivalents := []
506 for: -1 <= octave <= 9
507 for: 0 <= letter < 7
508 alteration := value - 12 * (octave + 1) - natural_pitch(letter)
509 if -2 <= alteration <= 2
510 equivalents = equivalents.append(Note(Pitch(letter, alteration), octave))
511 return ["ENHARMONIC {arguments[0]} " + notes_text(equivalents)]
512 elif command == "NOTE"
513 value := try bounded_integer(arguments[0], "NUMBER", 0, 127)
514 return ["NOTE {value} {spelling(value).text()} {spelling(value, true).text()}"]
515 elif command == "TUNING" or command == "PITCH"
516 value := try parse_frequency(arguments[0], command == "TUNING")
517 if command == "TUNING"
518 calculator.tuning = value
519 return ["TUNING " + frequency_text(value)]
520 ratio := real64(value) / real64(calculator.tuning)
521 real64 height := 69.0 + 12.0 * math.log(ratio) / math.log(real64(2))
522 nearest := math.floor(height + 0.5)
523 if nearest < 0 or nearest > 127
524 return error("OUT OF RANGE")
525 cents := math.floor(100.0 * (height - real64(nearest)) + 0.5)
526 sign := if cents >= 0 then "+" else ""
527 return ["PITCH " + frequency_text(value) + " {spelling(nearest).text()} {sign}{cents}"]
528 elif command == "INTERVAL"
529 first := try parse_note(arguments[0])
530 second := try parse_note(arguments[1])
531 motion := try between(first, second)
532 direction := if motion.down then "down" else "up"
533 return ["INTERVAL {arguments[0]} {arguments[1]} {motion.interval.text()} {direction} semitones={motion.semitones} simple={motion.interval.simple().text()} class={motion.interval.consonance()}"]
534 elif command == "INVERT"
535 interval := try parse_interval(arguments[0])
536 simple := interval.simple()
537 inverses := [0, 2, 1, 4, 3, 6, 5]
538 inverse := Interval(inverses[simple.quality], 9 - simple.number)
539 if inverse.forbidden()
540 return error("CANNOT INVERT")
541 return ["INVERT {arguments[0]} {simple.text()} {inverse.text()}"]
542 elif command == "TRANSPOSE"
543 note := try parse_note(arguments[0])
544 interval := try parse_interval(arguments[1])
545 if arguments[2] != "UP" and arguments[2] != "DOWN"
546 return error("BAD DIRECTION " + arguments[2])
547 sign := if arguments[2] == "UP" then 1 else -1
548 tone := interval.tone()
549 target := note.position() + sign * tone.letters
550 value := note.midi() + sign * tone.semitones
551 letter := modulo(target, 7)
552 octave := floor_div(target, 7)
553 alteration := value - 12 * (octave + 1) - natural_pitch(letter)
554 if alteration < -2 or alteration > 2
555 return error("CANNOT SPELL")
556 result := Note(Pitch(letter, alteration), octave)
557 if not result.valid()
558 return error("OUT OF RANGE")
559 return ["TRANSPOSE {arguments[0]} {arguments[1]} {arguments[2]} {result.text()}"]
560 elif command == "SCALE"
561 root := try parse_pitch(arguments[0])
562 index := try named_index(arguments[1], "SCALE", catalog)
563 scale := catalog.scales[index]
564 pitches := try build(root, scale.tones)
565 int[] steps := []
566 for tone_index := scale.tones.indices()[]
567 next := if tone_index + 1 < #scale.tones then scale.tones[tone_index + 1].semitones else 12
568 steps = steps.append(next - scale.tones[tone_index].semitones)
569 return ["SCALE {arguments[0]} {arguments[1]} " + pitches_text(pitches) + " steps=" + integer_list(steps)]
570 elif command == "CHORD" or command == "VOICE"
571 Pitch root
572 initial := 0
573 if command == "VOICE"
574 note := try parse_note(arguments[0])
575 root = note.pitch
576 initial = note.midi()
577 else
578 root = try parse_pitch(arguments[0])
579 quality := try named_index(arguments[1], "QUALITY", catalog)
580 chord := catalog.chords[quality]
581 inversion := 0
582 if count == 3
583 inversion = try bounded_integer(arguments[2], "INVERSION", 0, int(#chord.tones) - 1)
584 tones := try build(root, chord.tones)
585 Pitch[] ordered := []
586 for: 0 <= index < int(#tones)
587 ordered = ordered.append(tones[(index + inversion) % int(#tones)])
588 Pitch | none bass := if inversion == 0 then none else ordered[0]
589 symbol := chord_symbol(root, chord, bass)
590 if command == "CHORD"
591 return ["CHORD {symbol} " + pitches_text(ordered)]
592 Note[] notes := []
593 int[] values := []
594 value := initial
595 for index := ordered.indices()[]
596 pitch := ordered[index]
597 if index > 0 or inversion > 0
598 difference := modulo(pitch.pitch_class() - value, 12)
599 value += if difference == 0 then 12 else difference
600 note := make_note(pitch, value)
601 if not note.valid()
602 return error("OUT OF RANGE")
603 notes = notes.append(note)
604 values = values.append(value)
605 return ["VOICE {symbol} " + notes_text(notes) + " midi=" + integer_list(values)]
606 elif command == "IDENTIFY" or command == "MELODY"
607 notes := try parse_notes(arguments)
608 if command == "IDENTIFY"
609 return [identify(notes, catalog)]
610 result := try melody(notes, arguments)
611 return [result]
612 elif command == "CIRCLE"
613 mode := try named_index(arguments[0], "MODE", catalog)
614 Pitch[] pitches := []
615 for: -7 <= fifths <= 7
616 pitches = pitches.append(fifth_pitch(fifths - catalog.modes[mode].offset))
617 return ["CIRCLE {arguments[0]} " + pitches_text(pitches)]
618 elif command == "FINDKEY"
619 pitches := try parse_pitches(arguments)
620 string[] keys := []
621 for: -7 <= fifths <= 7
622 for: 0 <= mode < 2
623 key := try make_key(fifth_pitch(fifths - catalog.modes[mode].offset), catalog.modes[mode], catalog)
624 contains_all := true
625 for pitch := pitches[]
626 expected := key.scale[key.degree(pitch)]
627 if pitch.alteration != expected.alteration
628 contains_all = false
629 if contains_all
630 keys = keys.append(key.text())
631 return ["FINDKEY {#keys}" + (if #keys == 0 then "" else " " + keys.join(", "))]
632 elif command == "SPELL"
633 value := try bounded_integer(arguments[0], "NUMBER", 0, 127)
634 key := try parse_key(arguments, 1, catalog)
635 pitch := spelling(value, key.fifths < 0).pitch
636 for candidate := key.scale[]
637 if candidate.pitch_class() == value % 12
638 pitch = candidate
639 break
640 note := make_note(pitch, value)
641 if not note.valid()
642 return error("OUT OF RANGE")
643 return ["SPELL {value} {key.text()} {note.text()} {key.label(pitch)}"]
644 else
645 key := try parse_key(arguments, 0, catalog)
646 if command == "KEYSIG"
647 value := key.fifths
648 if value == 0
649 return ["KEYSIG {key.text()} none"]
650 string[] names := if value > 0 then ["F#", "C#", "G#", "D#", "A#", "E#", "B#"] else ["Bb", "Eb", "Ab", "Db", "Gb", "Cb", "Fb"]
651 kind := if value > 0 then "sharps" else "flats"
652 return ["KEYSIG {key.text()} {kind}={math.abs(value)} " + names[0:<math.abs(value)].join(" ")]
653 elif command == "RELATIVE"
654 return ["RELATIVE {key.text()} major={fifth_pitch(key.fifths).text()} minor={fifth_pitch(key.fifths + 3).text()}"]
655 elif command == "DIATONIC"
656 return diatonic(key, catalog)
657 elif command == "DEGREE"
658 pitch := try parse_pitch(arguments[2])
659 return ["DEGREE {key.text()} {pitch.text()} {key.label(pitch)}"]
660 elif command == "KEYDIST"
661 other := try parse_key(arguments, 2, catalog)
662 distance := modulo(other.fifths - key.fifths, 12)
663 if distance > 6
664 distance -= 12
665 first_set := key.classes()
666 second_set := other.classes()
667 common := 0
668 for index := first_set.indices()[]
669 if first_set[index] and second_set[index]
670 common += 1
671 same_tonic := key.tonic.pitch_class() == other.tonic.pitch_class()
672 string relation
673 if first_set == second_set
674 relation = if same_tonic then "same" else "relative"
675 elif same_tonic
676 relation = "parallel"
677 else
678 relation = if math.abs(distance) == 1 then "close" else "distant"
679 sign := if distance > 0 then "+" else ""
680 return ["KEYDIST {key.text()} {other.text()} fifths={sign}{distance} common={common} relation={relation}"]
681 elif command == "ANALYZE"
682 Symbol[] symbols := []
683 for word := arguments[2:][]
684 parsed := try parse_symbol(word, catalog)
685 symbols = symbols.append(parsed)
686 string[] numerals := []
687 for symbol := symbols[]
688 numerals = numerals.append(key.prefix(symbol.root) + roman(key.degree(symbol.root), catalog.chords[symbol.quality]))
689 return ["ANALYZE {key.text()} " + numerals.join(" ")]
690 else
691 Numeral[] numerals := []
692 for word := arguments[2:][]
693 parsed := try parse_numeral(word, catalog)
694 numerals = numerals.append(parsed)
695 rows := ["PROGRESSION {key.text()} {#numerals}"]
696 for index := numerals.indices()[]
697 numeral := numerals[index]
698 root := key.scale[numeral.degree]
699 root.alteration += numeral.alteration
700 chord := catalog.chords[numeral.quality]
701 tones := try build(root, chord.tones)
702 rows = rows.append(arguments[index + 2] + " " + chord_symbol(root, chord) + " " + pitches_text(tones))
703 rows = rows.append("CADENCE " + cadence(numerals))
704 return rows
