3NATURALS = [0, 2, 4, 5, 7, 9, 11].freeze
4FIFTHS = [0, 2, 4, -1, 1, 3, 5].freeze
5ACCIDENTALS = ['bb', 'b', '', '#', 'x'].freeze
6SHARPS = %w[C C# D D# E F F# G G# A A# B].freeze
7FLATS = %w[C Db D Eb E F Gb G Ab A Bb B].freeze
8ROMANS = %w[I II III IV V VI VII].freeze
9PERFECT_DELTAS = { dd: -2, d: -1, P: 0, A: 1, AA: 2 }.freeze
10MAJOR_DELTAS = { dd: -3, d: -2, m: -1, M: 0, A: 1, AA: 2 }.freeze
11INVERSE_QUALITIES = { P: :P, M: :m, m: :M, A: :d, d: :A, AA: :dd, dd: :AA }.freeze

17def canonical_integer(text, low, high)
18 return unless /\A(?:0|[1-9][0-9]*)\z/.match?(text)
19 value = text.to_i
20 value if value.between?(low, high)
21end

23def parse_number(text)
24 canonical_integer(text, 0, 127) || raise(CommandError, "BAD NUMBER #{text}")
25end

27def parse_frequency(text, tuning)
28 parts = /\A([0-9]{1,5})(?:\.([0-9]{1,2}))?\z/.match(text)
29 raise CommandError, "BAD FREQUENCY #{text}" unless parts
30 value = parts[1].to_i * 100
31 value += parts[2].to_i * (parts[2].length == 1 ? 10 : 1) if parts[2]
32 if value.zero? || (tuning && !value.between?(40_000, 48_000))
33 raise CommandError, "BAD FREQUENCY #{text}"
34 end
35 value
36end

40Pitch = Data.define(:letter, :alteration) do
41 def pitch_class = (NATURALS[letter] + alteration) % 12
42 def to_s = LETTERS[letter] + ACCIDENTALS[alteration + 2]
43
44 def self.prefix(text)
45 match = /\A([A-G])(bb|b|#|x)?/.match(text)
46 return unless match
47 [new(LETTERS.index(match[1]), ACCIDENTALS.index(match[2] || '') - 2), match[0].length]
48 end
49
50 def self.parse(text)
51 prefix = self.prefix(text)
52 raise CommandError, "BAD PITCH #{text}" unless prefix && prefix[1] == text.length
53 prefix[0]
54 end
55
56 def self.at_fifths(fifths)
57 new(LETTERS.index('FCGDAEB'[(fifths + 1) % 7]), (fifths + 1).div(7))
58 end

60 def raise_by(interval)
61 target = letter + interval.distance
62 change = NATURALS[letter] + alteration + interval.semitones - NATURALS[target % 7] - 12 * target.div(7)
63 raise CommandError, 'CANNOT SPELL' unless change.between?(-2, 2)
64 Pitch.new(target % 7, change)
65 end
66end

68Note = Data.define(:pitch, :octave) do
69 def midi = 12 * (octave + 1) + NATURALS[pitch.letter] + pitch.alteration
70 def position = 7 * octave + pitch.letter
71 def valid? = octave.between?(-1, 9) && midi.between?(0, 127)
72 def to_s = "#{pitch}#{octave}"
73
74 def self.parse(text)
75 prefix = Pitch.prefix(text)
76 raise CommandError, "BAD NOTE #{text}" unless prefix
77 rest = text[prefix[1]..]
78 raise CommandError, "BAD NOTE #{text}" unless /\A(?:-1|[0-9])\z/.match?(rest)
79 note = new(prefix[0], rest.to_i)
80 raise CommandError, "BAD NOTE #{text}" unless note.valid?
81 note
82 end
83
84 def self.from_pitch(pitch, midi)
85 new(pitch, (midi - NATURALS[pitch.letter] - pitch.alteration).div(12) - 1)
86 end
87
88 def self.spelling(midi, flat = false)
89 new(Pitch.parse((flat ? FLATS : SHARPS)[midi % 12]), midi.div(12) - 1)
90 end
91end

93Interval = Data.define(:quality, :number) do
94 def distance = number - 1
95 def perfect_type? = [0, 3, 4].include?(distance % 7)
96 def forbidden? = number == 1 && %i[d dd].include?(quality)
97 def semitones = 12 * distance.div(7) + NATURALS[distance % 7] + (perfect_type? ? PERFECT_DELTAS : MAJOR_DELTAS).fetch(quality)
98 def to_s = "#{quality}#{number}"
99
100 def simple
101 n = number
102 n -= 7 while n > 8
103 Interval.new(quality, n)
104 end
105
106 def consonance
107 small = simple
108 return 'perfect' if quality == :P && [1, 5, 8].include?(small.number)
109 return 'imperfect' if %i[M m].include?(quality) && [3, 6].include?(small.number)
110 'dissonant'
111 end
112
113 def self.parse(text)
114 parts = /\A(P|M|m|A|d|AA|dd)([1-9][0-9]?)\z/.match(text)
115 raise CommandError, "BAD INTERVAL #{text}" unless parts
116 interval = new(parts[1].to_sym, parts[2].to_i)
117 deltas = interval.perfect_type? ? PERFECT_DELTAS : MAJOR_DELTAS
118 raise CommandError, "BAD INTERVAL #{text}" if !deltas.key?(interval.quality) || interval.forbidden?
119 interval
120 end

122 def self.between(first, second)
123 distance = second.position - first.position
124 semitones = second.midi - first.midi
125 down = distance.negative? || (distance.zero? && semitones.negative?)
126 distance, semitones = -distance, -semitones if down
127 difference = semitones - 12 * distance.div(7) - NATURALS[distance % 7]
128 quality = ([0, 3, 4].include?(distance % 7) ? PERFECT_DELTAS : MAJOR_DELTAS).key(difference)
129 raise CommandError, 'UNNAMEABLE INTERVAL' unless quality
130 Motion.new(new(quality, distance + 1), down, semitones)
131 end
132end

187def parse_quality(text) = CHORDS.find { |chord| chord.name == text } || raise(CommandError, "BAD QUALITY #{text}")
188def parse_mode(text) = MODES.find { |mode| mode.name == text } || raise(CommandError, "BAD MODE #{text}")
189def build_pitches(root, pattern) = pattern.map { |interval| root.raise_by(interval) }.freeze

191Key = Data.define(:tonic, :mode, :fifths, :scale) do
192 def self.make(tonic, mode)
193 fifths = FIFTHS[tonic.letter] + 7 * tonic.alteration + mode.offset
194 raise CommandError, 'THEORETICAL KEY' unless fifths.between?(-7, 7)
195 new(tonic, mode, fifths, build_pitches(tonic, SCALES.fetch(mode.scale)))
196 end

198 def self.parse(words, start = 0)
199 tonic = Pitch.parse(words[start])
200 mode = parse_mode(words[start + 1])
201 make(tonic, mode)
202 end

204 def degree(pitch) = (pitch.letter - tonic.letter) % 7
205 def prefix(pitch)
206 difference = pitch.alteration - scale[degree(pitch)].alteration
207 (difference.negative? ? 'b' : '#') * difference.abs
208 end
209 def label(pitch) = "#{prefix(pitch)}#{degree(pitch) + 1}"
210 def pitch_set = scale.reduce(0) { |mask, pitch| mask | (1 << pitch.pitch_class) }
211 def to_s = "#{tonic} #{mode.name}"
212end

214Numeral = Data.define(:alteration, :degree, :quality)
215ChordSymbol = Data.define(:root, :quality)
216def parse_numeral(text)
217 start = ['b', '#'].include?(text[0]) ? 1 : 0
218 alteration = start.zero? ? 0 : (text[0] == 'b' ? -1 : 1)
219 ending = start
220 ending += 1 while ending < text.length && 'IViv'.include?(text[ending])
221 part = text[start...ending]
222 suffix = text[ending..]
223 ROMANS.each_with_index do |roman, degree|
224 CHORDS.each do |quality|
225 expected = quality.lower ? roman.downcase : roman
226 return Numeral.new(alteration, degree, quality) if part == expected && suffix == quality.numeral_suffix
227 end
228 end
229 raise CommandError, "BAD NUMERAL #{text}"
230end

232def parse_symbol(text)
233 prefix = Pitch.prefix(text)
234 if prefix
235 quality = CHORDS.find { |chord| chord.suffix == text[prefix[1]..] }
236 return ChordSymbol.new(prefix[0], quality) if quality
237 end
238 raise CommandError, "BAD SYMBOL #{text}"
239end

241def chord_command(command, args)
242 input = Note.parse(args[0]) if command == 'VOICE'
243 root = input ? input.pitch : Pitch.parse(args[0])
244 quality = parse_quality(args[1])
245 inversion = args.length == 3 ? canonical_integer(args[2], 0, quality.intervals.length - 1) : 0
246 raise CommandError, "BAD INVERSION #{args[2]}" unless inversion
247 ordered = build_pitches(root, quality.intervals).rotate(inversion)
248 name = quality.symbol(root, inversion.zero? ? nil : ordered.first)
249 return "CHORD #{name} #{ordered.join(' ')}" unless input
250 value = input.midi
251 notes = ordered.each_with_index.map do |pitch, i|
252 unless i.zero? && inversion.zero?
253 difference = (pitch.pitch_class - value) % 12
254 value += difference.zero? ? 12 : difference
255 end
256 note = Note.from_pitch(pitch, value)
257 raise CommandError, 'OUT OF RANGE' unless note.valid?
258 note
259 end
260 "VOICE #{name} #{notes.join(' ')} midi=#{notes.map(&:midi).join(',')}"
261end

263def identify(notes)
264 candidates = notes.each_with_index.sort_by { |note, i| [note.midi, i] }.map(&:first).uniq { |note| note.pitch.pitch_class }
265 bass = candidates.first.pitch
266 mask = candidates.reduce(0) { |set, note| set | (1 << note.pitch.pitch_class) }
267 candidates.each do |root|
268 CHORDS.each do |quality|
269 set = quality.intervals.reduce(0) { |bits, interval| bits | (1 << ((root.midi + interval.semitones) % 12)) }
270 next unless set == mask
271 inversion = quality.intervals.index { |interval| (root.midi + interval.semitones) % 12 == bass.pitch_class }
272 name = quality.symbol(root.pitch, inversion.zero? ? nil : bass)
273 return "IDENTIFY #{name} root=#{root.pitch} quality=#{quality.name} inversion=#{inversion}"
274 end
275 end
276 'IDENTIFY UNKNOWN'
277end

279def melody(notes)
280 low = high = notes.first
281 notes.each do |note|
282 low = note if note.midi < low.midi
283 high = note if note.midi > high.midi
284 end
285 range = Interval.between(low, high).interval
286 leaps = 0
287 steps = notes.each_cons(2).map do |first, second|
288 motion = Interval.between(first, second)
289 leaps += 1 if (second.midi - first.midi).abs >= 3
290 "#{motion.down ? '-' : '+'}#{motion.interval}"
291 end
292 "MELODY n=#{notes.length} low=#{low} high=#{high} range=#{range} leaps=#{leaps} steps=#{steps.join(',')}"
293end

295def cadence(numerals)
296 return 'none' if numerals.length < 2
297 x, y = numerals.last(2)
298 if x.alteration.zero? && y.alteration.zero?
299 return 'authentic' if x.degree == 4 && y.degree.zero?
300 return 'plagal' if x.degree == 3 && y.degree.zero?
301 return 'deceptive' if x.degree == 4 && y.degree == 5
302 end
303 y.alteration.zero? && y.degree == 4 ? 'half' : 'none'
304end

306def diatonic(key)
307 rows = ["DIATONIC #{key}"]
308 [3, 4].each do |size|
309 items = key.scale.each_with_index.map do |root, i|
310 third = (key.scale[(i + 2) % 7].pitch_class - root.pitch_class) % 12
311 fifth = (key.scale[(i + 4) % 7].pitch_class - root.pitch_class) % 12
312 index = if size == 3
313 third == 4 ? 0 : (fifth == 7 ? 1 : 2)
314 else
315 seventh = (key.scale[(i + 6) % 7].pitch_class - root.pitch_class) % 12
316 third == 4 ? (seventh == 11 ? 9 : 8) : (fifth == 7 ? 10 : 12)
317 end
318 quality = CHORDS[index]
319 "#{quality.roman(i)}=#{quality.symbol(root)}"
320 end
321 rows << (size == 3 ? 'TRIADS ' : 'SEVENTHS ') + items.join(' ')
322 end
323 rows.join("\n")
324end

326def key_command(command, args)
327 key = Key.parse(args)
328 case command
329 when 'KEYSIG'
330 p = key.fifths
331 return "KEYSIG #{key} none" if p.zero?
332 names = p.positive? ? %w[F# C# G# D# A# E# B#] : %w[Bb Eb Ab Db Gb Cb Fb]
333 "KEYSIG #{key} #{p.positive? ? 'sharps' : 'flats'}=#{p.abs} #{names.take(p.abs).join(' ')}"
334 when 'RELATIVE'
335 "RELATIVE #{key} major=#{Pitch.at_fifths(key.fifths)} minor=#{Pitch.at_fifths(key.fifths + 3)}"
336 when 'DEGREE'
337 pitch = Pitch.parse(args[2])
338 "DEGREE #{key} #{args[2]} #{key.label(pitch)}"
339 when 'KEYDIST'
340 other = Key.parse(args, 2)
341 distance = (other.fifths - key.fifths) % 12
342 distance -= 12 if distance > 6
343 same_tonic = key.tonic.pitch_class == other.tonic.pitch_class
344 relation = if key.pitch_set == other.pitch_set
345 same_tonic ? 'same' : 'relative'
346 elsif same_tonic
347 'parallel'
348 else
349 distance.abs == 1 ? 'close' : 'distant'
350 end
351 common = (key.pitch_set & other.pitch_set).digits(2).sum
352 "KEYDIST #{key} #{other} fifths=#{distance.positive? ? '+' : ''}#{distance} common=#{common} relation=#{relation}"
353 when 'DIATONIC'
354 diatonic(key)
355 when 'ANALYZE'
356 symbols = args.drop(2).map { |word| parse_symbol(word) }
357 items = symbols.map { |chord| key.prefix(chord.root) + chord.quality.roman(key.degree(chord.root)) }
358 "ANALYZE #{key} #{items.join(' ')}"
359 when 'PROGRESSION'
360 numerals = args.drop(2).map { |word| parse_numeral(word) }
361 rows = ["PROGRESSION #{key} #{numerals.length}"]
362 numerals.each_with_index do |numeral, i|
363 degree = key.scale[numeral.degree]
364 root = Pitch.new(degree.letter, degree.alteration + numeral.alteration)
365 pitches = build_pitches(root, numeral.quality.intervals)
366 rows << "#{args[i + 2]} #{numeral.quality.symbol(root)} #{pitches.join(' ')}"
367 end
368 rows << "CADENCE #{cadence(numerals)}"
369 rows.join("\n")
370 end
371end

373ARITIES = {
374 'MIDI' => 1..1, 'NOTE' => 1..1, 'FREQ' => 1..1, 'TUNING' => 1..1, 'PITCH' => 1..1,
375 'INTERVAL' => 2..2, 'INVERT' => 1..1, 'TRANSPOSE' => 3..3, 'SCALE' => 2..2,
376 'CHORD' => 2..3, 'VOICE' => 2..3, 'IDENTIFY' => 2..15, 'MELODY' => 2..15,
377 'ENHARMONIC' => 1..1, 'KEYSIG' => 2..2, 'RELATIVE' => 2..2, 'CIRCLE' => 1..1,
378 'DIATONIC' => 2..2, 'PROGRESSION' => 3..15, 'ANALYZE' => 3..15, 'FINDKEY' => 1..15,
379 'DEGREE' => 3..3, 'SPELL' => 3..3, 'KEYDIST' => 4..4
380}.freeze
381
382class Calculator
383 def initialize
384 @tuning = 44_000
385 end
386
387 def execute(line)
388 command, *args = line.split(' ')
389 arity = ARITIES[command] || raise(CommandError, "UNKNOWN COMMAND #{command}")
390 raise CommandError, 'ARGUMENT COUNT' unless arity.cover?(args.length)
391 case command
392 when 'MIDI', 'FREQ', 'ENHARMONIC'
393 note = Note.parse(args[0])
394 value = note.midi
395 return "MIDI #{args[0]} #{value}" if command == 'MIDI'
396 if command == 'FREQ'
397 cents = (@tuning * 2.0**((value - 69) / 12.0) + 0.5).floor
398 return "FREQ #{args[0]} #{frequency_text(cents)}"
399 end
400 equivalents = []
401 (-1..9).each do |octave|
402 (0..6).each do |letter|
403 alteration = value - 12 * (octave + 1) - NATURALS[letter]
404 equivalents << Note.new(Pitch.new(letter, alteration), octave) if alteration.between?(-2, 2)
405 end
406 end
407 "ENHARMONIC #{args[0]} #{equivalents.join(' ')}"
408 when 'NOTE'
409 value = parse_number(args[0])
410 "NOTE #{value} #{Note.spelling(value)} #{Note.spelling(value, true)}"
411 when 'TUNING', 'PITCH'
412 value = parse_frequency(args[0], command == 'TUNING')
413 if command == 'TUNING'
414 @tuning = value
415 return "TUNING #{frequency_text(value)}"
416 end
417 x = 69 + 12 * Math.log2(value.fdiv(@tuning))
418 nearest = (x + 0.5).floor
419 raise CommandError, 'OUT OF RANGE' unless nearest.between?(0, 127)
420 cents = (100 * (x - nearest) + 0.5).floor
421 "PITCH #{frequency_text(value)} #{Note.spelling(nearest)} #{cents >= 0 ? '+' : ''}#{cents}"
422 when 'INTERVAL'
423 first = Note.parse(args[0])
424 second = Note.parse(args[1])
425 motion = Interval.between(first, second)
426 "INTERVAL #{args[0]} #{args[1]} #{motion.interval} #{motion.down ? 'down' : 'up'} semitones=#{motion.semitones} simple=#{motion.interval.simple} class=#{motion.interval.consonance}"
427 when 'INVERT'
428 simple = Interval.parse(args[0]).simple
429 inverse = Interval.new(INVERSE_QUALITIES.fetch(simple.quality), 9 - simple.number)
430 raise CommandError, 'CANNOT INVERT' if inverse.forbidden?
431 "INVERT #{args[0]} #{simple} #{inverse}"
432 when 'TRANSPOSE'
433 note = Note.parse(args[0])
434 interval = Interval.parse(args[1])
435 direction = args[2]
436 raise CommandError, "BAD DIRECTION #{direction}" unless %w[UP DOWN].include?(direction)
437 sign = direction == 'UP' ? 1 : -1
438 position = note.position + sign * interval.distance
439 value = note.midi + sign * interval.semitones
440 letter, octave = position % 7, position.div(7)
441 alteration = value - 12 * (octave + 1) - NATURALS[letter]
442 raise CommandError, 'CANNOT SPELL' unless alteration.between?(-2, 2)
443 result = Note.new(Pitch.new(letter, alteration), octave)
444 raise CommandError, 'OUT OF RANGE' unless result.valid?
445 "TRANSPOSE #{args[0]} #{args[1]} #{direction} #{result}"
446 when 'SCALE'
447 root = Pitch.parse(args[0])
448 pattern = SCALES[args[1]] || raise(CommandError, "BAD SCALE #{args[1]}")
449 pitches = build_pitches(root, pattern)
450 steps = pattern.each_with_index.map { |interval, i| (pattern[i + 1]&.semitones || 12) - interval.semitones }
451 "SCALE #{args[0]} #{args[1]} #{pitches.join(' ')} steps=#{steps.join(',')}"
452 when 'CHORD', 'VOICE'
453 chord_command(command, args)
454 when 'IDENTIFY'
455 identify(args.map { |word| Note.parse(word) })
456 when 'MELODY'
457 melody(args.map { |word| Note.parse(word) })
458 when 'CIRCLE'
459 mode = parse_mode(args[0])
460 pitches = (-7..7).map { |p| Pitch.at_fifths(p - mode.offset) }
461 "CIRCLE #{args[0]} #{pitches.join(' ')}"
462 when 'FINDKEY'
463 pitches = args.map { |word| Pitch.parse(word) }
464 keys = []
465 (-7..7).each do |p|
466 MODES.first(2).each do |mode|
467 key = Key.make(Pitch.at_fifths(p - mode.offset), mode)
468 keys << key if pitches.all? { |pitch| key.scale.include?(pitch) }
469 end
470 end
471 "FINDKEY #{keys.length}#{keys.empty? ? '' : ' ' + keys.join(', ')}"
472 when 'SPELL'
473 value = parse_number(args[0])
474 key = Key.parse(args, 1)
475 pitch = key.scale.find { |candidate| candidate.pitch_class == value % 12 } || Note.spelling(value, key.fifths.negative?).pitch
476 note = Note.from_pitch(pitch, value)
477 raise CommandError, 'OUT OF RANGE' unless note.valid?
478 "SPELL #{value} #{key} #{note} #{key.label(pitch)}"
479 else
480 key_command(command, args)
481 end
482 end
483end
