18bounded_integer <- function(text, kind, low, high) {
19 if (nchar(text) > 3L || !grepl("^(0|[1-9][0-9]*)$", text)) bad(kind, text)
20 value <- as.integer(text)
21 if (value < low || value > high) bad(kind, text)
22 value

24parse_frequency <- function(text, tuning) {
25 if (!grepl("^[0-9]{1,5}(\\.[0-9]{1,2})?$", text)) bad("FREQUENCY", text)
26 parts <- strsplit(text, ".", fixed = TRUE)[[1L]]
27 value <- as.integer(parts[[1L]]) * 100L
28 if (length(parts) == 2L) {
29 value <- value + as.integer(parts[[2L]]) * if (nchar(parts[[2L]]) == 1L) 10L else 1L
30 }
31 if (value == 0L || (tuning && (value < 40000L || value > 48000L))) bad("FREQUENCY", text)
32 value
33}

39pitch_prefix <- function(text, kind) {
40 first <- substr(text, 1L, 1L)
41 letter <- match(first, letters)
42 if (is.na(letter)) bad(kind, text)
43 end <- 2L
44 alteration <- 0L
45 next_char <- substr(text, end, end)
46 if (next_char == "b") {
47 alteration <- -1L
48 end <- end + 1L
49 if (substr(text, end, end) == "b") {
50 alteration <- -2L
51 end <- end + 1L
52 }
53 } else if (next_char == "#") {
54 alteration <- 1L
55 end <- end + 1L
56 } else if (next_char == "x") {
57 alteration <- 2L
58 end <- end + 1L
59 }
60 list(pitch = new_pitch(letter - 1L, alteration), rest = substring(text, end))
61}

92quality_delta <- function(quality, number) {
93 if ((number - 1L) %% 7L %in% c(0L, 3L, 4L)) {
94 c(0L, NA_integer_, NA_integer_, 1L, -1L, 2L, -2L)[quality]
95 } else c(NA_integer_, 0L, -1L, 1L, -2L, 2L, -3L)[quality]

97new_interval <- function(quality, number) list(quality = quality, number = number)
98forbidden_interval <- function(interval) interval$number == 1L && interval$quality %in% c(5L, 7L)
99parse_interval <- function(text) {
100 for (quality in seq_along(interval_qualities)) {
101 prefix <- interval_qualities[quality]
102 if (!startsWith(text, prefix)) next
103 rest <- substring(text, nchar(prefix) + 1L)
104 if (!grepl("^[1-9][0-9]?$", rest)) next
105 number <- as.integer(rest)
106 interval <- new_interval(quality, number)
107 if (!is.na(quality_delta(quality, number)) && !forbidden_interval(interval)) return(interval)
108 }
109 bad("INTERVAL", text)
110}
111interval_text <- function(interval) paste0(interval_qualities[interval$quality], interval$number)
112simple_interval <- function(interval) {
113 while (interval$number > 8L) interval$number <- interval$number - 7L
114 interval
115}
116interval_tone <- function(interval) {
117 distance <- interval$number - 1L
118 c(letters = distance,
119 semitones = 12L * (distance %/% 7L) + naturals[distance %% 7L + 1L] + quality_delta(interval$quality, interval$number))
120}
121consonance <- function(interval) {
122 simple <- simple_interval(interval)
123 if (simple$quality == 1L && simple$number %in% c(1L, 5L, 8L)) return("perfect")
124 if (simple$quality %in% c(2L, 3L) && simple$number %in% c(3L, 6L)) return("imperfect")
125 "dissonant"
126}
127between <- function(first, second) {
128 distance <- second$position - first$position
129 semitones <- second$midi - first$midi
130 down <- distance < 0L || (distance == 0L && semitones < 0L)
131 if (down) {
132 distance <- -distance
133 semitones <- -semitones
134 }
135 difference <- semitones - 12L * (distance %/% 7L) - naturals[distance %% 7L + 1L]
136 for (quality in seq_along(interval_qualities)) {
137 delta <- quality_delta(quality, distance + 1L)
138 if (!is.na(delta) && delta == difference) {
139 return(list(interval = new_interval(quality, distance + 1L), down = down, semitones = semitones))
140 }
141 }
142 fail("UNNAMEABLE INTERVAL")
143}
144raise_pitch <- function(root, tone) {
145 target <- root[["letter"]] + tone[["letters"]]
146 alteration <- naturals[root[["letter"]] + 1L] + root[["alteration"]] + tone[["semitones"]] -
147 naturals[target %% 7L + 1L] - 12L * (target %/% 7L)
148 if (alteration < -2L || alteration > 2L) fail("CANNOT SPELL")
149 new_pitch(target %% 7L, alteration)
150}

214parse_numeral <- function(text) {
215 prefix <- substr(text, 1L, 1L)
216 alteration <- if (prefix == "b") -1L else if (prefix == "#") 1L else 0L
217 rest <- if (alteration == 0L) text else substring(text, 2L)
218 end <- 1L
219 while (end <= nchar(rest) && substr(rest, end, end) %in% c("I", "V", "i", "v")) end <- end + 1L
220 part <- substr(rest, 1L, end - 1L)
221 suffix <- substring(rest, end)
222 for (degree in seq_along(romans)) {
223 for (quality in seq_along(chords)) {
224 chord <- chords[[quality]]
225 expected <- if (chord$lower) tolower(romans[degree]) else romans[degree]
226 if (part == expected && suffix == chord$numeral_suffix) {
227 return(list(alteration = alteration, degree = degree, quality = quality))
228 }
229 }
230 }
231 bad("NUMERAL", text)

240identify <- function(notes) {
241 values <- vapply(notes, function(note) note$midi, integer(1L))
242 ordered <- notes[order(values, method = "radix")]
243 classes <- vapply(ordered, function(note) pitch_class(note$pitch), integer(1L))
244 candidates <- ordered[!duplicated(classes)]
245 present <- sort(unique(classes))
246 bass <- candidates[[1L]]$pitch
247 for (root in candidates) {
248 for (chord in chords) {
249 semitones <- vapply(chord$tones, function(tone) tone[["semitones"]], integer(1L))
250 possible <- (root$midi + semitones) %% 12L
251 if (!identical(sort(unique(possible)), present)) next
252 inversion <- match(pitch_class(bass), possible) - 1L
253 symbol <- chord_symbol(root$pitch, chord, if (inversion == 0L) NULL else bass)
254 return(paste0("IDENTIFY ", symbol, " root=", pitch_text(root$pitch), " quality=", chord$name, " inversion=", inversion))
255 }
256 }
257 "IDENTIFY UNKNOWN"
258}
259melody <- function(notes, arguments) {
260 values <- vapply(notes, function(note) note$midi, integer(1L))
261 low <- which.min(values)
262 high <- which.max(values)
263 range <- between(notes[[low]], notes[[high]])$interval
264 steps <- vapply(seq.int(2L, length(notes)), function(index) {
265 motion <- between(notes[[index - 1L]], notes[[index]])
266 paste0(if (motion$down) "-" else "+", interval_text(motion$interval))
267 }, character(1L))
268 paste0("MELODY n=", length(notes), " low=", arguments[low], " high=", arguments[high],
269 " range=", interval_text(range), " leaps=", sum(abs(diff(values)) >= 3L), " steps=", joined(steps, ","))
270}
271cadence <- function(numerals) {
272 if (length(numerals) < 2L) return("none")
273 first <- numerals[[length(numerals) - 1L]]
274 second <- numerals[[length(numerals)]]
275 if (first$alteration == 0L && second$alteration == 0L) {
276 if (first$degree == 5L && second$degree == 1L) return("authentic")
277 if (first$degree == 4L && second$degree == 1L) return("plagal")
278 if (first$degree == 5L && second$degree == 6L) return("deceptive")
279 }
280 if (second$alteration == 0L && second$degree == 5L) "half" else "none"
281}
282diatonic <- function(key) {
283 rows <- paste("DIATONIC", key_text(key))
284 classes <- vapply(key$scale, pitch_class, integer(1L))
285 for (size in 3L:4L) {
286 items <- vapply(seq_len(7L), function(index) {
287 third <- (classes[(index + 1L) %% 7L + 1L] - classes[index]) %% 12L
288 fifth <- (classes[(index + 3L) %% 7L + 1L] - classes[index]) %% 12L
289 if (size == 3L) {
290 quality <- if (third == 4L) 1L else if (fifth == 7L) 2L else 3L
291 } else {
292 seventh <- (classes[(index + 5L) %% 7L + 1L] - classes[index]) %% 12L
293 quality <- if (third == 4L) { if (seventh == 11L) 10L else 9L } else if (fifth == 7L) 11L else 13L
294 }
295 chord <- chords[[quality]]
296 paste0(roman(index, chord), "=", chord_symbol(key$scale[[index]], chord))
297 }, character(1L))
298 rows <- c(rows, paste(if (size == 3L) "TRIADS" else "SEVENTHS", joined(items)))
299 }
300 rows
301}

311execute <- function(line, calculator) {
312 tokens <- words(line)
313 command <- tokens[[1L]]
314 arity <- arities[[command]]
315 if (is.null(arity)) fail(paste("UNKNOWN COMMAND", command))
316 arguments <- tokens[-1L]
317 count <- length(arguments)
318 if (count < arity[1L] || count > arity[2L]) fail("ARGUMENT COUNT")
319 if (command %in% c("MIDI", "FREQ", "ENHARMONIC")) {
320 note <- parse_note(arguments[[1L]])
321 value <- note$midi
322 if (command == "MIDI") return(paste("MIDI", arguments[[1L]], value))
323 if (command == "FREQ") {
324 cents <- as.integer(floor(calculator$tuning * 2^((value - 69L) / 12) + 0.5))
325 return(paste("FREQ", arguments[[1L]], frequency_text(cents)))
326 }
327 equivalents <- list()
328 for (octave in -1L:9L) for (letter in 0L:6L) {
329 alteration <- value - 12L * (octave + 1L) - naturals[letter + 1L]
330 if (alteration >= -2L && alteration <= 2L) equivalents[[length(equivalents) + 1L]] <- new_note(new_pitch(letter, alteration), octave)
331 }
332 return(paste("ENHARMONIC", arguments[[1L]], notes_text(equivalents)))
333 }
334 if (command == "NOTE") {
335 value <- bounded_integer(arguments[[1L]], "NUMBER", 0L, 127L)
336 return(paste("NOTE", value, note_text(spelling(value)), note_text(spelling(value, TRUE))))
337 }
338 if (command %in% c("TUNING", "PITCH")) {
339 value <- parse_frequency(arguments[[1L]], command == "TUNING")
340 if (command == "TUNING") {
341 calculator$tuning <- value
342 return(paste("TUNING", frequency_text(value)))
343 }
344 height <- 69 + 12 * log2(value / calculator$tuning)
345 nearest <- as.integer(floor(height + 0.5))
346 if (nearest < 0L || nearest > 127L) fail("OUT OF RANGE")
347 cents <- as.integer(floor(100 * (height - nearest) + 0.5))
348 return(paste("PITCH", frequency_text(value), note_text(spelling(nearest)), sprintf("%+d", cents)))
349 }
350 if (command == "INTERVAL") {
351 first <- parse_note(arguments[[1L]])
352 second <- parse_note(arguments[[2L]])
353 motion <- between(first, second)
354 return(paste0("INTERVAL ", arguments[[1L]], " ", arguments[[2L]], " ", interval_text(motion$interval),
355 if (motion$down) " down" else " up", " semitones=", motion$semitones,
356 " simple=", interval_text(simple_interval(motion$interval)), " class=", consonance(motion$interval)))
357 }
358 if (command == "INVERT") {
359 interval <- parse_interval(arguments[[1L]])
360 simple <- simple_interval(interval)
361 inverse <- new_interval(c(1L, 3L, 2L, 5L, 4L, 7L, 6L)[simple$quality], 9L - simple$number)
362 if (forbidden_interval(inverse)) fail("CANNOT INVERT")
363 return(paste("INVERT", arguments[[1L]], interval_text(simple), interval_text(inverse)))
364 }
365 if (command == "TRANSPOSE") {
366 note <- parse_note(arguments[[1L]])
367 interval <- parse_interval(arguments[[2L]])
368 direction <- arguments[[3L]]
369 if (!direction %in% c("UP", "DOWN")) bad("DIRECTION", direction)
370 sign <- if (direction == "UP") 1L else -1L
371 tone <- interval_tone(interval)
372 position <- note$position + sign * tone[["letters"]]
373 value <- note$midi + sign * tone[["semitones"]]
374 letter <- position %% 7L
375 octave <- position %/% 7L
376 alteration <- value - 12L * (octave + 1L) - naturals[letter + 1L]
377 if (alteration < -2L || alteration > 2L) fail("CANNOT SPELL")
378 result <- new_note(new_pitch(letter, alteration), octave)
379 if (!valid_note(result)) fail("OUT OF RANGE")
380 return(paste("TRANSPOSE", arguments[[1L]], arguments[[2L]], direction, note_text(result)))
381 }
382 if (command == "SCALE") {
383 root <- parse_pitch(arguments[[1L]])
384 index <- named_index(arguments[[2L]], "SCALE", names(scales))
385 tones <- scales[[index]]
386 pitches <- build(root, tones)
387 semitones <- vapply(tones, function(tone) tone[["semitones"]], integer(1L))
388 return(paste0("SCALE ", arguments[[1L]], " ", arguments[[2L]], " ", pitches_text(pitches),
389 " steps=", joined(diff(c(semitones, 12L)), ",")))
390 }
391 if (command %in% c("CHORD", "VOICE")) {
392 if (command == "VOICE") {
393 note <- parse_note(arguments[[1L]])
394 root <- note$pitch
395 } else root <- parse_pitch(arguments[[1L]])
396 quality <- named_index(arguments[[2L]], "QUALITY", names(chords))
397 chord <- chords[[quality]]
398 inversion <- if (count == 3L) bounded_integer(arguments[[3L]], "INVERSION", 0L, length(chord$tones) - 1L) else 0L
399 tones <- build(root, chord$tones)
400 ordered <- tones[(seq_along(tones) - 1L + inversion) %% length(tones) + 1L]
401 symbol <- chord_symbol(root, chord, if (inversion == 0L) NULL else ordered[[1L]])
402 if (command == "CHORD") return(paste("CHORD", symbol, pitches_text(ordered)))
403 notes <- list()
404 values <- integer()
405 value <- note$midi
406 for (index in seq_along(ordered)) {
407 pitch <- ordered[[index]]
408 if (index > 1L || inversion > 0L) {
409 difference <- (pitch_class(pitch) - value) %% 12L
410 value <- value + if (difference == 0L) 12L else difference
411 }
412 next_note <- make_note(pitch, value)
413 if (!valid_note(next_note)) fail("OUT OF RANGE")
414 notes[[index]] <- next_note
415 values <- c(values, value)
416 }
417 return(paste0("VOICE ", symbol, " ", notes_text(notes), " midi=", joined(values, ",")))
418 }
419 if (command %in% c("IDENTIFY", "MELODY")) {
420 notes <- lapply(arguments, parse_note)
421 return(if (command == "IDENTIFY") identify(notes) else melody(notes, arguments))
422 }
423 if (command == "CIRCLE") {
424 mode <- named_index(arguments[[1L]], "MODE", mode_names)
425 pitches <- lapply(-7L:7L, function(value) fifth_pitch(value - mode_offsets[mode]))
426 return(paste("CIRCLE", arguments[[1L]], pitches_text(pitches)))
427 }
428 if (command == "FINDKEY") {
429 pitches <- lapply(arguments, parse_pitch)
430 keys <- character()
431 for (fifths in -7L:7L) for (mode in 1L:2L) {
432 key <- make_key(fifth_pitch(fifths - mode_offsets[mode]), mode)
433 included <- vapply(pitches, function(pitch) identical(pitch, key$scale[[key_degree(key, pitch)]]), logical(1L))
434 if (all(included)) keys <- c(keys, key_text(key))
435 }
436 return(paste0("FINDKEY ", length(keys), if (length(keys)) paste0(" ", joined(keys, ", ")) else ""))
437 }
438 if (command == "SPELL") {
439 value <- bounded_integer(arguments[[1L]], "NUMBER", 0L, 127L)
440 key <- parse_key(arguments, 2L)
441 classes <- vapply(key$scale, pitch_class, integer(1L))
442 index <- match(value %% 12L, classes)

475 return(paste0("KEYDIST ", key_text(key), " ", key_text(other), " fifths=", if (distance > 0L) "+" else "", distance,
476 " common=", length(intersect(first_set, second_set)), " relation=", relation))
477 }
