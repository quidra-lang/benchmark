7sealed class CommandError(string message) : Exception("ERROR " + message);

9readonly record struct Pitch(int Letter, int Alteration)
10{
11 public static readonly int[] Naturals = [0, 2, 4, 5, 7, 9, 11];
12 public int Natural => Naturals[Letter];
13 public int Class => Music.Mod(Natural + Alteration, 12);
14 public override string ToString() => "CDEFGAB"[Letter] + new[] { "bb", "b", "", "#", "x" }[Alteration + 2];
15}
16
17readonly record struct Note(Pitch Pitch, int Octave)
18{
19 public int Midi => 12 * (Octave + 1) + Pitch.Natural + Pitch.Alteration;
20 public int Position => 7 * Octave + Pitch.Letter;
21 public bool Valid => Octave is >= -1 and <= 9 && Midi is >= 0 and <= 127;
22 public override string ToString() => $"{Pitch}{Octave}";
23}
24
25readonly record struct Interval(string Quality, int Number)
26{
27 public int Letters => Number - 1;
28 public bool PerfectType => Letters % 7 is 0 or 3 or 4;
29 public int Semitones => 12 * (Letters / 7) + Pitch.Naturals[Letters % 7] + Difference(Quality, PerfectType);
30 public Interval Simple
31 {
32 get
33 {
34 int number = Number;
35 while (number > 8) number -= 7;
36 return new(Quality, number);
37 }
38 }
39 public string Consonance => Simple.ToString() switch
40 {
41 "P1" or "P5" or "P8" => "perfect",
42 "m3" or "M3" or "m6" or "M6" => "imperfect",
43 _ => "dissonant"
44 };
45 public static int Difference(string quality, bool perfect) => quality switch
46 {
47 "dd" => perfect ? -2 : -3, "d" => perfect ? -1 : -2,
48 "m" => -1, "A" => 1, "AA" => 2, _ => 0
49 };
50 public override string ToString() => Quality + Number;
51}

53sealed record Quality(string Name, Interval[] Intervals, string Suffix, string Numeral)
54{
55 public bool Lower => Numeral[0] == 'i';
56 public string NumeralSuffix => Numeral[1..];
57}
58sealed record Mode(string Name, string Scale, int Offset);
59readonly record struct Key(Pitch Tonic, Mode Mode)
60{
61 public int Fifths => new[] { 0, 2, 4, -1, 1, 3, 5 }[Tonic.Letter] + 7 * Tonic.Alteration + Mode.Offset;
62 public override string ToString() => $"{Tonic} {Mode.Name}";
63}
64readonly record struct Numeral(int Degree, int Alteration, Quality Quality);
65readonly record struct Chord(Pitch Root, Quality Quality)
66{
67 public override string ToString() => Root + Quality.Suffix;
68}
69readonly record struct Motion(Interval Interval, bool Down, int Semitones);
70readonly record struct Degree(int Index, string Prefix)
71{
72 public override string ToString() => Prefix + (Index + 1);
73}

77 private static readonly string[] Sharps = "C C# D D# E F F# G G# A A# B".Split(' ');
78 private static readonly string[] Flats = "C Db D Eb E F Gb G Ab A Bb B".Split(' ');
79 private static readonly string[] Romans = ["I", "II", "III", "IV", "V", "VI", "VII"];
80 private static readonly string[] LowerRomans = ["i", "ii", "iii", "iv", "v", "vi", "vii"];

125 public static int Mod(int value, int divisor) => (value % divisor + divisor) % divisor;
126 public static int FloorDiv(int value, int divisor) => (value - Mod(value, divisor)) / divisor;
127 public static string Join<T>(IEnumerable<T> values) => string.Join(' ', values);
128 public static string Frequency(int cents) => $"{cents / 100}.{cents % 100:D2}";
129 public static string Signed(int value, bool positiveZero = false) => value > 0 || value == 0 && positiveZero ? "+" + value : value.ToString(CultureInfo.InvariantCulture);
130 private static Interval[] Intervals(string text) => text.Split(' ').Select(ParseInterval).ToArray();
131 private static int Alteration(string text) => text switch { "bb" => -2, "b" => -1, "#" => 1, "x" => 2, _ => 0 };
132 public static Pitch ParsePitch(string text)
133 {
134 Match match = Regex.Match(text, @"\A([A-G])(bb|b|#|x)?\z");
135 if (!match.Success) throw new CommandError("BAD PITCH " + text);
136 return new("CDEFGAB".IndexOf(match.Groups[1].Value[0]), Alteration(match.Groups[2].Value));
137 }
138 public static Note ParseNote(string text)
139 {
140 Match match = Regex.Match(text, @"\A([A-G])(bb|b|#|x)?(-1|[0-9])\z");
141 if (!match.Success) throw new CommandError("BAD NOTE " + text);
142 Note note = new(new("CDEFGAB".IndexOf(match.Groups[1].Value[0]), Alteration(match.Groups[2].Value)),
143 int.Parse(match.Groups[3].Value, CultureInfo.InvariantCulture));
144 if (!note.Valid) throw new CommandError("BAD NOTE " + text);
145 return note;
146 }
147 public static int ParseNumber(string text)
148 {
149 if (!Regex.IsMatch(text, @"\A(?:0|[1-9][0-9]*)\z") || !int.TryParse(text, out int number) || number > 127)
150 throw new CommandError("BAD NUMBER " + text);
151 return number;
152 }
153 public static int ParseFrequency(string text, bool tuning)
154 {
155 Match match = Regex.Match(text, @"\A([0-9]{1,5})(?:\.([0-9]{1,2}))?\z");
156 if (!match.Success) throw new CommandError("BAD FREQUENCY " + text);
157 int cents = int.Parse(match.Groups[1].Value, CultureInfo.InvariantCulture) * 100;
158 string tail = match.Groups[2].Value;
159 if (tail.Length > 0) cents += int.Parse(tail.PadRight(2, '0'), CultureInfo.InvariantCulture);
160 if (tuning ? cents is < 40000 or > 48000 : cents == 0) throw new CommandError("BAD FREQUENCY " + text);
161 return cents;
162 }
163 public static Interval ParseInterval(string text)
164 {
165 Match match = Regex.Match(text, @"\A(AA|dd|P|M|m|A|d)([1-9][0-9]?)\z");
166 if (!match.Success) throw new CommandError("BAD INTERVAL " + text);
167 Interval interval = new(match.Groups[1].Value, int.Parse(match.Groups[2].Value, CultureInfo.InvariantCulture));
168 if ((interval.PerfectType ? interval.Quality is "M" or "m" : interval.Quality == "P") ||
169 interval.Number == 1 && (interval.Quality is "d" or "dd")) throw new CommandError("BAD INTERVAL " + text);
170 return interval;
171 }
172 public static Quality ParseQuality(string text) => Qualities.FirstOrDefault(q => q.Name == text) ?? throw new CommandError("BAD QUALITY " + text);
173 public static Mode ParseMode(string text) => Modes.FirstOrDefault(m => m.Name == text) ?? throw new CommandError("BAD MODE " + text);
174 public static Key ParseKey(string tonic, string mode)
175 {
176 Pitch pitch = ParsePitch(tonic);
177 Mode parsedMode = ParseMode(mode);
178 Key key = new(pitch, parsedMode);
179 if (key.Fifths is < -7 or > 7) throw new CommandError("THEORETICAL KEY");
180 return key;
181 }
182 public static Interval[] ParseScale(string text) => Scales.TryGetValue(text, out Interval[]? scale) ? scale : throw new CommandError("BAD SCALE " + text);
183 public static int ParseInversion(string text, Quality quality)
184 {
185 if (text.Length != 1 || text[0] < '0' || text[0] >= '0' + quality.Intervals.Length) throw new CommandError("BAD INVERSION " + text);
186 return text[0] - '0';
187 }
188 public static Numeral ParseNumeral(string text)
189 {
190 int start = text[0] is 'b' or '#' ? 1 : 0;
191 int alteration = start == 0 ? 0 : text[0] == 'b' ? -1 : 1;
192 int end = start;
193 while (end < text.Length && "IViv".Contains(text[end])) end++;
194 string roman = text[start..end], suffix = text[end..];
195 int degree = Array.IndexOf(Romans, roman);
196 bool lower = false;
197 if (degree < 0) { degree = Array.IndexOf(LowerRomans, roman); lower = true; }
198 Quality? quality = Qualities.FirstOrDefault(q => q.Lower == lower && q.NumeralSuffix == suffix);
199 if (degree < 0 || quality is null) throw new CommandError("BAD NUMERAL " + text);
200 return new(degree, alteration, quality);
201 }
202 public static Chord ParseSymbol(string text)
203 {
204 Match match = Regex.Match(text, @"\A([A-G])(bb|b|#|x)?(.*)\z");
205 if (!match.Success) throw new CommandError("BAD SYMBOL " + text);
206 Quality? quality = Qualities.FirstOrDefault(q => q.Suffix == match.Groups[3].Value);
207 if (quality is null) throw new CommandError("BAD SYMBOL " + text);
208 Pitch root = new("CDEFGAB".IndexOf(match.Groups[1].Value[0]), Alteration(match.Groups[2].Value));
209 return new(root, quality);
210 }
211 public static Pitch Raise(Pitch root, Interval interval)
212 {
213 int letter = Mod(root.Letter + interval.Letters, 7);
214 int alteration = root.Natural + root.Alteration + interval.Semitones - Pitch.Naturals[letter] - 12 * FloorDiv(root.Letter + interval.Letters, 7);
215 if (alteration is < -2 or > 2) throw new CommandError("CANNOT SPELL");
216 return new(letter, alteration);
217 }
218 public static Pitch[] Build(Pitch root, Interval[] intervals) => intervals.Select(interval => Raise(root, interval)).ToArray();
219 public static Pitch[] Scale(Key key) => Build(key.Tonic, Scales[key.Mode.Scale]);
220 public static Note FromMidi(Pitch pitch, int midi) => new(pitch, (midi - pitch.Natural - pitch.Alteration) / 12 - 1);
221 public static Note StandardNote(int midi, bool flat = false) => new(ParsePitch((flat ? Flats : Sharps)[Mod(midi, 12)]), midi / 12 - 1);
222 public static Pitch FifthsPitch(int fifths)
223 {
224 int index = Mod(fifths + 1, 7);
225 return new("CDEFGAB".IndexOf("FCGDAEB"[index]), FloorDiv(fifths + 1, 7));
226 }
227 public static Key KeyAt(int fifths, Mode mode) => new(FifthsPitch(fifths - mode.Offset), mode);
228 public static Degree Label(Key key, Pitch pitch)
229 {
230 int index = Mod(pitch.Letter - key.Tonic.Letter, 7);
231 int alteration = pitch.Alteration - Scale(key)[index].Alteration;
232 return new(index, new string(alteration < 0 ? 'b' : '#', Math.Abs(alteration)));
233 }
234 public static string Roman(int degree, Quality quality) => (quality.Lower ? LowerRomans : Romans)[degree] + quality.NumeralSuffix;
235 public static Motion Between(Note first, Note second)
236 {
237 int letters = second.Position - first.Position, semitones = second.Midi - first.Midi;
238 bool down = letters < 0 || letters == 0 && semitones < 0;
239 if (down) { letters = -letters; semitones = -semitones; }
240 int difference = semitones - (12 * (letters / 7) + Pitch.Naturals[letters % 7]);
241 bool perfect = letters % 7 is 0 or 3 or 4;
242 string? quality = perfect ? difference switch { -2 => "dd", -1 => "d", 0 => "P", 1 => "A", 2 => "AA", _ => null }
243 : difference switch { -3 => "dd", -2 => "d", -1 => "m", 0 => "M", 1 => "A", 2 => "AA", _ => null };
244 if (quality is null) throw new CommandError("UNNAMEABLE INTERVAL");
245 return new(new(quality, letters + 1), down, semitones);
246 }
247 public static int PitchMask(IEnumerable<Pitch> pitches)
248 {
249 int mask = 0;
250 foreach (Pitch pitch in pitches) mask |= 1 << pitch.Class;
251 return mask;
252 }
253 public static int ChordMask(Note root, Quality quality)
254 {
255 int mask = 0;
256 foreach (Interval interval in quality.Intervals) mask |= 1 << Mod(root.Midi + interval.Semitones, 12);
257 return mask;
258 }
259 public static string Cadence(Numeral[] numerals)
260 {
261 if (numerals.Length < 2) return "none";
262 Numeral a = numerals[^2], b = numerals[^1];
263 if (a.Alteration == 0 && b.Alteration == 0)
264 {
265 if (a.Degree == 4 && b.Degree == 0) return "authentic";
266 if (a.Degree == 3 && b.Degree == 0) return "plagal";
267 if (a.Degree == 4 && b.Degree == 5) return "deceptive";
268 }
269 return b.Alteration == 0 && b.Degree == 4 ? "half" : "none";
270 }
271}

273sealed class Calculator
274{
275 private int tuning = 44000;
276 private static readonly Dictionary<string, (int Min, int Max)> Arity = new(StringComparer.Ordinal)
277 {
278 ["MIDI"] = (1, 1), ["NOTE"] = (1, 1), ["FREQ"] = (1, 1), ["TUNING"] = (1, 1), ["PITCH"] = (1, 1),
279 ["INTERVAL"] = (2, 2), ["INVERT"] = (1, 1), ["TRANSPOSE"] = (3, 3), ["SCALE"] = (2, 2),
280 ["CHORD"] = (2, 3), ["VOICE"] = (2, 3), ["IDENTIFY"] = (2, 15), ["MELODY"] = (2, 15),
281 ["ENHARMONIC"] = (1, 1), ["KEYSIG"] = (2, 2), ["RELATIVE"] = (2, 2), ["CIRCLE"] = (1, 1),
282 ["DIATONIC"] = (2, 2), ["PROGRESSION"] = (3, 15), ["ANALYZE"] = (3, 15), ["FINDKEY"] = (1, 15),
283 ["DEGREE"] = (3, 3), ["SPELL"] = (3, 3), ["KEYDIST"] = (4, 4)
284 };
285 public string Execute(string line)
286 {
287 string[] w = line.Split(' ');
288 string command = w[0];
289 if (!Arity.TryGetValue(command, out var arity)) throw new CommandError("UNKNOWN COMMAND " + command);
290 if (w.Length - 1 < arity.Min || w.Length - 1 > arity.Max) throw new CommandError("ARGUMENT COUNT");
291 switch (command)
292 {
293 case "MIDI": return $"MIDI {w[1]} {Music.ParseNote(w[1]).Midi}";
294 case "NOTE":
295 {
296 int number = Music.ParseNumber(w[1]);
297 return $"NOTE {number} {Music.StandardNote(number)} {Music.StandardNote(number, true)}";
298 }
299 case "FREQ":
300 {
301 Note note = Music.ParseNote(w[1]);
302 double value = (tuning / 100.0) * Math.Pow(2, (note.Midi - 69) / 12.0);
303 return $"FREQ {w[1]} {Music.Frequency((int)Math.Floor(value * 100 + 0.5))}";
304 }
305 case "TUNING":
306 {
307 int value = Music.ParseFrequency(w[1], true);
308 tuning = value;
309 return "TUNING " + Music.Frequency(value);
310 }
311 case "PITCH":
312 {
313 int value = Music.ParseFrequency(w[1], false);
314 double x = 69 + 12 * Math.Log2((double)value / tuning);
315 int number = (int)Math.Floor(x + 0.5);
316 if (number is < 0 or > 127) throw new CommandError("OUT OF RANGE");
317 int cents = (int)Math.Floor(100 * (x - number) + 0.5);
318 return $"PITCH {Music.Frequency(value)} {Music.StandardNote(number)} {Music.Signed(cents, true)}";
319 }
320 case "INTERVAL":
321 {
322 Note a = Music.ParseNote(w[1]);
323 Note b = Music.ParseNote(w[2]);
324 Motion motion = Music.Between(a, b);
325 return $"INTERVAL {w[1]} {w[2]} {motion.Interval} {(motion.Down ? "down" : "up")} semitones={motion.Semitones} simple={motion.Interval.Simple} class={motion.Interval.Consonance}";
326 }
327 case "INVERT":
328 {
329 Interval original = Music.ParseInterval(w[1]), simple = original.Simple;
330 string quality = simple.Quality switch { "P" => "P", "M" => "m", "m" => "M", "A" => "d", "d" => "A", "AA" => "dd", _ => "AA" };
331 Interval inverted = new(quality, 9 - simple.Number);
332 if (inverted.Number == 1 && (quality is "d" or "dd")) throw new CommandError("CANNOT INVERT");
333 return $"INVERT {w[1]} {simple} {inverted}";
334 }
335 case "TRANSPOSE":
336 {
337 Note note = Music.ParseNote(w[1]);
338 Interval interval = Music.ParseInterval(w[2]);
339 if (w[3] is not ("UP" or "DOWN")) throw new CommandError("BAD DIRECTION " + w[3]);
340 int sign = w[3] == "UP" ? 1 : -1;
341 int position = note.Position + sign * interval.Letters, midi = note.Midi + sign * interval.Semitones;
342 int letter = Music.Mod(position, 7), octave = Music.FloorDiv(position, 7);
343 int alteration = midi - (12 * (octave + 1) + Pitch.Naturals[letter]);
344 if (alteration is < -2 or > 2) throw new CommandError("CANNOT SPELL");
345 Note result = new(new(letter, alteration), octave);
346 if (!result.Valid) throw new CommandError("OUT OF RANGE");
347 return $"TRANSPOSE {w[1]} {w[2]} {w[3]} {result}";
348 }
349 case "SCALE":
350 {
351 Pitch root = Music.ParsePitch(w[1]);
352 Interval[] intervals = Music.ParseScale(w[2]);
353 Pitch[] notes = Music.Build(root, intervals);
354 int[] steps = intervals.Select((interval, i) => (i + 1 == intervals.Length ? 12 : intervals[i + 1].Semitones) - interval.Semitones).ToArray();
355 return $"SCALE {w[1]} {w[2]} {Music.Join(notes)} steps={string.Join(',', steps)}";
356 }
357 case "CHORD": case "VOICE":
358 {
359 Note? input = null;
360 Pitch root;
361 if (command == "VOICE") { input = Music.ParseNote(w[1]); root = input.Value.Pitch; }
362 else root = Music.ParsePitch(w[1]);
363 Quality quality = Music.ParseQuality(w[2]);
364 int inversion = w.Length == 4 ? Music.ParseInversion(w[3], quality) : 0;
365 Pitch[] tones = Music.Build(root, quality.Intervals);
366 Pitch[] reordered = tones.Skip(inversion).Concat(tones.Take(inversion)).ToArray();
367 string symbol = new Chord(root, quality).ToString() + (inversion == 0 ? "" : "/" + reordered[0]);
368 if (command == "CHORD") return $"CHORD {symbol} {Music.Join(reordered)}";
369 var notes = new List<Note>();
370 int previous = input!.Value.Midi;
371 for (int i = 0; i < reordered.Length; i++)
372 {
373 int midi = previous;
374 if (i != 0 || inversion != 0)
375 {
376 midi += Music.Mod(reordered[i].Class - previous, 12);
377 if (midi == previous) midi += 12;
378 }
379 Note made = Music.FromMidi(reordered[i], midi);
380 if (!made.Valid) throw new CommandError("OUT OF RANGE");
381 notes.Add(made);
382 previous = midi;
383 }
384 return $"VOICE {symbol} {Music.Join(notes)} midi={string.Join(',', notes.Select(n => n.Midi))}";
385 }
386 case "IDENTIFY":
387 {
388 Note[] notes = w.Skip(1).Select(Music.ParseNote).ToArray();
389 var seen = new HashSet<int>();
390 Note[] candidates = notes.OrderBy(note => note.Midi).Where(note => seen.Add(note.Pitch.Class)).ToArray();
391 int mask = Music.PitchMask(notes.Select(note => note.Pitch));
392 Note bass = candidates[0];
393 foreach (Note root in candidates)
394 foreach (Quality quality in Music.Qualities)
395 {
396 if (Music.ChordMask(root, quality) != mask) continue;
397 int inversion = Array.FindIndex(quality.Intervals, interval => Music.Mod(root.Midi + interval.Semitones, 12) == bass.Pitch.Class);
398 string symbol = new Chord(root.Pitch, quality).ToString() + (inversion == 0 ? "" : "/" + bass.Pitch);
399 return $"IDENTIFY {symbol} root={root.Pitch} quality={quality.Name} inversion={inversion}";
400 }
401 return "IDENTIFY UNKNOWN";
402 }
403 case "MELODY":
404 {
405 Note[] notes = w.Skip(1).Select(Music.ParseNote).ToArray();
406 Note low = notes[0], high = notes[0];
407 foreach (Note note in notes)
408 {
409 if (note.Midi < low.Midi) low = note;
410 if (note.Midi > high.Midi) high = note;
411 }
412 Interval range = Music.Between(low, high).Interval;
413 var steps = new List<string>();
414 int leaps = 0;
415 for (int i = 1; i < notes.Length; i++)
416 {
417 Motion motion = Music.Between(notes[i - 1], notes[i]);
418 steps.Add((motion.Down ? "-" : "+") + motion.Interval);
419 if (Math.Abs(notes[i].Midi - notes[i - 1].Midi) >= 3) leaps++;
420 }
421 return $"MELODY n={notes.Length} low={low} high={high} range={range} leaps={leaps} steps={string.Join(',', steps)}";
422 }
423 case "ENHARMONIC":
424 {
425 Note input = Music.ParseNote(w[1]);
426 var notes = new List<Note>();
427 for (int letter = 0; letter < 7; letter++)
428 for (int alteration = -2; alteration <= 2; alteration++)
429 {
430 Pitch pitch = new(letter, alteration);
431 if (pitch.Class != input.Pitch.Class) continue;
432 Note note = Music.FromMidi(pitch, input.Midi);
433 if (note.Valid) notes.Add(note);
434 }
435 return $"ENHARMONIC {w[1]} {Music.Join(notes.OrderBy(note => note.Position))}";
436 }
437 case "KEYSIG": case "RELATIVE": case "DIATONIC": case "PROGRESSION": case "ANALYZE": case "DEGREE":
438 {
439 Key key = Music.ParseKey(w[1], w[2]);
440 if (command == "KEYSIG")
441 {
442 int fifths = key.Fifths;
443 if (fifths == 0) return $"KEYSIG {key} none";
444 string[] names = (fifths > 0 ? "F# C# G# D# A# E# B#" : "Bb Eb Ab Db Gb Cb Fb").Split(' ');
445 return $"KEYSIG {key} {(fifths > 0 ? "sharps" : "flats")}={Math.Abs(fifths)} {Music.Join(names.Take(Math.Abs(fifths)))}";
446 }
447 if (command == "RELATIVE") return $"RELATIVE {key} major={Music.KeyAt(key.Fifths, Music.Modes[0]).Tonic} minor={Music.KeyAt(key.Fifths, Music.Modes[1]).Tonic}";
448 if (command == "DEGREE")
449 {
450 Pitch pitch = Music.ParsePitch(w[3]);
451 return $"DEGREE {key} {w[3]} {Music.Label(key, pitch)}";
452 }
453 if (command == "ANALYZE")
454 {
455 Chord[] chords = w.Skip(3).Select(Music.ParseSymbol).ToArray();
456 string[] results = chords.Select(chord =>
457 {
458 Degree degree = Music.Label(key, chord.Root);
459 return degree.Prefix + Music.Roman(degree.Index, chord.Quality);
460 }).ToArray();
461 return $"ANALYZE {key} {Music.Join(results)}";
462 }
463 if (command == "PROGRESSION")
464 {
465 Numeral[] numerals = w.Skip(3).Select(Music.ParseNumeral).ToArray();
466 Pitch[] keyScale = Music.Scale(key);
467 var lines = new List<string> { $"PROGRESSION {key} {numerals.Length}" };
468 for (int i = 0; i < numerals.Length; i++)
469 {
470 Numeral numeral = numerals[i];
471 Pitch degree = keyScale[numeral.Degree];
472 Pitch root = new(degree.Letter, degree.Alteration + numeral.Alteration);
473 Pitch[] tones = Music.Build(root, numeral.Quality.Intervals);
474 lines.Add($"{w[i + 3]} {new Chord(root, numeral.Quality)} {Music.Join(tones)}");
475 }
476 lines.Add("CADENCE " + Music.Cadence(numerals));
477 return string.Join('\n', lines);
478 }
479 Pitch[] scale = Music.Scale(key);
480 var triads = new List<string>();
481 var sevenths = new List<string>();
482 for (int i = 0; i < 7; i++)
483 {
484 int third = Music.Mod(scale[(i + 2) % 7].Class - scale[i].Class, 12);
485 int fifth = Music.Mod(scale[(i + 4) % 7].Class - scale[i].Class, 12);
486 int seventh = Music.Mod(scale[(i + 6) % 7].Class - scale[i].Class, 12);
487 Quality triad = Music.ParseQuality(third == 4 ? "maj" : fifth == 7 ? "min" : "dim");
488 Quality tetrad = Music.ParseQuality(third == 4 ? seventh == 11 ? "maj7" : "7" : fifth == 7 ? "m7" : "m7b5");
489 triads.Add($"{Music.Roman(i, triad)}={new Chord(scale[i], triad)}");
490 sevenths.Add($"{Music.Roman(i, tetrad)}={new Chord(scale[i], tetrad)}");
491 }
492 return $"DIATONIC {key}\nTRIADS {Music.Join(triads)}\nSEVENTHS {Music.Join(sevenths)}";
493 }
494 case "CIRCLE":
495 {
496 Mode mode = Music.ParseMode(w[1]);
497 return $"CIRCLE {w[1]} {Music.Join(Enumerable.Range(-7, 15).Select(p => Music.KeyAt(p, mode).Tonic))}";
498 }
499 case "FINDKEY":
500 {
501 Pitch[] pitches = w.Skip(1).Select(Music.ParsePitch).ToArray();
502 var keys = new List<Key>();
503 for (int fifths = -7; fifths <= 7; fifths++)
504 for (int m = 0; m < 2; m++)
505 {
506 Key key = Music.KeyAt(fifths, Music.Modes[m]);
507 Pitch[] scale = Music.Scale(key);
508 if (pitches.All(scale.Contains)) keys.Add(key);
509 }
510 return "FINDKEY " + keys.Count + (keys.Count == 0 ? "" : " " + string.Join(", ", keys));
511 }
512 case "SPELL":
513 {
514 int midi = Music.ParseNumber(w[1]);
515 Key key = Music.ParseKey(w[2], w[3]);
516 Pitch pitch = Music.StandardNote(midi, key.Fifths < 0).Pitch;
517 foreach (Pitch candidate in Music.Scale(key))
518 if (candidate.Class == midi % 12) { pitch = candidate; break; }
519 Note note = Music.FromMidi(pitch, midi);
520 if (!note.Valid) throw new CommandError("OUT OF RANGE");
521 return $"SPELL {midi} {key} {note} {Music.Label(key, pitch)}";
522 }
523 case "KEYDIST":
524 {
525 Key a = Music.ParseKey(w[1], w[2]);
526 Key b = Music.ParseKey(w[3], w[4]);
527 int difference = Music.Mod(b.Fifths - a.Fifths, 12);
528 if (difference > 6) difference -= 12;
529 int first = Music.PitchMask(Music.Scale(a)), second = Music.PitchMask(Music.Scale(b));
530 int common = Enumerable.Range(0, 12).Count(i => ((first & second) & (1 << i)) != 0);
531 bool sameTonic = a.Tonic.Class == b.Tonic.Class;
532 string relation = first == second ? sameTonic ? "same" : "relative" : sameTonic ? "parallel" : Math.Abs(difference) == 1 ? "close" : "distant";
533 return $"KEYDIST {a} {b} fifths={Music.Signed(difference)} common={common} relation={relation}";
534 }
535 default: throw new InvalidOperationException();
536 }
537 }
538}
