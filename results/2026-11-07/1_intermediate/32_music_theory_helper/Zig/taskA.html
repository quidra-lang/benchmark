95const IntervalQuality = enum { P, M, m, A, d, AA, dd };
96const Interval = struct {
97 quality: IntervalQuality,
98 number: i32,
99 fn distance(interval: Interval) i32 {
100 return interval.number - 1;
101 }
102 fn perfect(interval: Interval) bool {
103 return switch (@mod(interval.distance(), 7)) {
104 0, 3, 4 => true,
105 else => false,
106 };
107 }
108 fn semitones(interval: Interval) i32 {
109 const difference: i32 = switch (interval.quality) {
110 .dd => if (interval.perfect()) -2 else -3,
111 .d => if (interval.perfect()) -1 else -2,
112 .m => -1,
113 .A => 1,
114 .AA => 2,
115 .P, .M => 0,
116 };
117 return 12 * @divTrunc(interval.distance(), 7) + naturals[@intCast(@mod(interval.distance(), 7))] + difference;
118 }
119 fn simple(interval: Interval) Interval {
120 var result = interval;
121 while (result.number > 8) result.number -= 7;
122 return result;
123 }
124 fn consonance(interval: Interval) []const u8 {
125 const small = interval.simple();
126 if (small.quality == .P and (small.number == 1 or small.number == 5 or small.number == 8)) return "perfect";
127 if ((small.quality == .m or small.quality == .M) and (small.number == 3 or small.number == 6)) return "imperfect";
128 return "dissonant";
129 }
130 pub fn format(interval: Interval, out: *Writer) Writer.Error!void {
131 try out.print("{s}{d}", .{ @tagName(interval.quality), interval.number });
132 }

134fn rawInterval(text: []const u8) ?Interval {
135 var start: usize = 0;
136 while (start < text.len and !digit(text[start])) start += 1;
137 const quality = std.meta.stringToEnum(IntervalQuality, text[0..start]) orelse return null;
138 const number = text[start..];
139 if (number.len == 0 or number.len > 2 or number[0] == '0' or !digits(number)) return null;
140 const result: Interval = .{ .quality = quality, .number = integer(number) };
141 if (result.perfect()) {
142 if (quality == .M or quality == .m) return null;
143 } else if (quality == .P) return null;
144 if (result.number == 1 and (quality == .d or quality == .dd)) return null;
145 return result;
146}
147const Pattern = struct {
148 items: [7]Interval = @splat(Interval{ .quality = .P, .number = 1 }),
149 len: usize = 0,
150 fn list(pattern: *const Pattern) []const Interval {
151 return pattern.items[0..pattern.len];
152 }

154fn makePattern(comptime text: []const u8) Pattern {
155 @setEvalBranchQuota(20000);
156 var result: Pattern = .{};
157 var words = std.mem.tokenizeScalar(u8, text, ' ');
158 while (words.next()) |word| {
159 result.items[result.len] = rawInterval(word).?;
160 result.len += 1;
161 }
162 return result;
163}

323const PitchPrefix = struct { pitch: Pitch, len: usize };
324fn pitchPrefix(text: []const u8) ?PitchPrefix {
325 if (text.len == 0) return null;
326 const letter = std.mem.findScalar(u8, letters, text[0]) orelse return null;
327 var len: usize = 1;
328 var alteration: i32 = 0;
329 if (std.mem.startsWith(u8, text[1..], "bb")) {
330 len = 3;
331 alteration = -2;
332 } else if (text.len > 1) {
333 switch (text[1]) {
334 'b' => {
335 len = 2;
336 alteration = -1;
337 },
338 '#' => {
339 len = 2;
340 alteration = 1;
341 },
342 'x' => {
343 len = 2;
344 alteration = 2;
345 },
346 else => {},
347 }
348 }
349 return .{ .pitch = .{ .letter = letter, .alteration = alteration }, .len = len };
350}
351fn rawPitch(text: []const u8) ?Pitch {
352 const prefix = pitchPrefix(text) orelse return null;
353 return if (prefix.len == text.len) prefix.pitch else null;
354}
355fn rawNote(text: []const u8) ?Note {
356 const prefix = pitchPrefix(text) orelse return null;
357 const tail = text[prefix.len..];
358 const octave: i32 = if (equal(tail, "-1")) -1 else if (tail.len == 1 and digit(tail[0])) tail[0] - '0' else return null;
359 const note: Note = .{ .pitch = prefix.pitch, .octave = octave };
360 return if (note.valid()) note else null;
361}
362fn fromMidi(pitch: Pitch, midi: i32) Note {
363 return .{ .pitch = pitch, .octave = @divExact(midi - pitch.natural() - pitch.alteration, 12) - 1 };
364}
365fn standardNote(midi: i32, flat: bool) Note {
366 return .{ .pitch = rawPitch((if (flat) flat_names else sharp_names)[@intCast(@mod(midi, 12))]).?, .octave = @divFloor(midi, 12) - 1 };
367}

380fn cadence(numerals: []const Numeral) []const u8 {
381 if (numerals.len < 2) return "none";
382 const a = numerals[numerals.len - 2];
383 const b = numerals[numerals.len - 1];
384 if (a.alteration == 0 and b.alteration == 0) {
385 if (a.degree == 4 and b.degree == 0) return "authentic";
386 if (a.degree == 3 and b.degree == 0) return "plagal";
387 if (a.degree == 4 and b.degree == 5) return "deceptive";
388 }
389 return if (b.alteration == 0 and b.degree == 4) "half" else "none";
390}

429 fn number(session: *Session, text: []const u8) Error!i32 {
430 if (!digits(text) or (text.len > 1 and text[0] == '0')) return session.fail("BAD NUMBER {s}", .{text});
431 const value = std.fmt.parseInt(i32, text, 10) catch return session.fail("BAD NUMBER {s}", .{text});
432 if (value > 127) return session.fail("BAD NUMBER {s}", .{text});
433 return value;
434 }
435 fn frequency(session: *Session, text: []const u8, tuning: bool) Error!i32 {
436 const separator = std.mem.findScalar(u8, text, '.');
437 const first = if (separator) |i| text[0..i] else text;
438 const tail = if (separator) |i| text[i + 1 ..] else "";
439 if (first.len > 5 or !digits(first) or (separator != null and (tail.len > 2 or !digits(tail))))
440 return session.fail("BAD FREQUENCY {s}", .{text});
441 var value = integer(first) * 100;
442 if (tail.len > 0) value += integer(tail) * @as(i32, if (tail.len == 1) 10 else 1);
443 if (if (tuning) value < 40000 or value > 48000 else value == 0) return session.fail("BAD FREQUENCY {s}", .{text});
444 return value;
445 }

462 fn inversion(session: *Session, text: []const u8, quality_value: QualityName) Error!usize {
463 if (text.len != 1 or !digit(text[0]) or text[0] - '0' >= quality_value.data().pattern.len)
464 return session.fail("BAD INVERSION {s}", .{text});
465 return text[0] - '0';
466 }
467 fn numeral(session: *Session, text: []const u8) Error!Numeral {
468 const start: usize = if (text[0] == 'b' or text[0] == '#') 1 else 0;
469 const alteration: i32 = if (start == 0) 0 else if (text[0] == 'b') -1 else 1;
470 var end = start;
471 while (end < text.len and std.mem.findScalar(u8, "IViv", text[end]) != null) end += 1;
472 const roman = text[start..end];
473 for (romans, 0..) |name, index| {
474 const lower = equal(roman, lower_romans[index]);
475 if (!lower and !equal(roman, name)) continue;
476 for (quality_table, 0..) |entry, q| {
477 if (entry.lower() == lower and equal(entry.numeral[1..], text[end..]))
478 return .{ .degree = index, .alteration = alteration, .quality = @fromBackingInt(@intCast(q)) };
479 }
480 }
481 return session.fail("BAD NUMERAL {s}", .{text});
482 }
483 fn symbol(session: *Session, text: []const u8) Error!Chord {
484 const prefix = pitchPrefix(text) orelse return session.fail("BAD SYMBOL {s}", .{text});
485 for (quality_table, 0..) |entry, index| {
486 if (equal(entry.suffix, text[prefix.len..])) return .{ .root = prefix.pitch, .quality = @fromBackingInt(@intCast(index)) };
487 }
488 return session.fail("BAD SYMBOL {s}", .{text});
489 }
490 fn raise(session: *Session, root: Pitch, interval_value: Interval) Error!Pitch {
491 const position = @as(i32, @intCast(root.letter)) + interval_value.distance();
492 const target: usize = @intCast(@mod(position, 7));
493 const alteration = root.natural() + root.alteration + interval_value.semitones() - naturals[target] - 12 * @divFloor(position, 7);
494 if (alteration < -2 or alteration > 2) return session.fail("CANNOT SPELL", .{});
495 return .{ .letter = target, .alteration = alteration };
496 }
497 fn build(session: *Session, root: Pitch, pattern: *const Pattern) Error!Pitches {
498 var result: Pitches = .{ .len = pattern.len };
499 for (pattern.list(), 0..) |interval_value, i| result.items[i] = try session.raise(root, interval_value);
500 return result;
501 }
502 fn scale(session: *Session, key_value: Key) Error!Pitches {
503 return session.build(key_value.tonic, key_value.mode.scale().pattern());
504 }
505 fn label(session: *Session, key_value: Key, pitch_value: Pitch) Error!Degree {
506 const index: usize = @intCast(@mod(@as(i32, @intCast(pitch_value.letter)) - @as(i32, @intCast(key_value.tonic.letter)), 7));
507 const pitches = try session.scale(key_value);
508 return .{ .index = index, .alteration = pitch_value.alteration - pitches.items[index].alteration };
509 }
510 fn between(session: *Session, first: Note, second: Note) Error!Motion {
511 var distance = second.position() - first.position();
512 var semitones = second.midi() - first.midi();
513 const down = distance < 0 or (distance == 0 and semitones < 0);
514 if (down) {
515 distance = -distance;
516 semitones = -semitones;
517 }
518 const difference = semitones - (12 * @divTrunc(distance, 7) + naturals[@intCast(@mod(distance, 7))]);
519 const perfect = switch (@mod(distance, 7)) {
520 0, 3, 4 => true,
521 else => false,
522 };
523 const quality_value: ?IntervalQuality = if (perfect) switch (difference) {
524 -2 => .dd,
525 -1 => .d,
526 0 => .P,
527 1 => .A,
528 2 => .AA,
529 else => null,
530 } else switch (difference) {
531 -3 => .dd,
532 -2 => .d,
533 -1 => .m,
534 0 => .M,
535 1 => .A,
536 2 => .AA,
537 else => null,
538 };
539 return .{ .interval = .{ .quality = quality_value orelse return session.fail("UNNAMEABLE INTERVAL", .{}), .number = distance + 1 }, .down = down, .semitones = semitones };
540 }
541 fn writePitches(session: *Session, pitches: []const Pitch) Writer.Error!void {
542 for (pitches) |pitch_value| try session.out.print(" {f}", .{pitch_value});
543 }
544 fn execute(session: *Session, text: []const u8) Error!void {
545 var words: [16][]const u8 = undefined;
546 var count: usize = 0;
547 var tokens = std.mem.tokenizeScalar(u8, text, ' ');
548 while (tokens.next()) |word| {
549 words[count] = word;
550 count += 1;
551 }
552 const w = words[0..count];
553 const command = std.meta.stringToEnum(Command, w[0]) orelse return session.fail("UNKNOWN COMMAND {s}", .{w[0]});
554 const limits = arity(command);
555 if (count - 1 < limits.min or count - 1 > limits.max) return session.fail("ARGUMENT COUNT", .{});
556 const out = session.out;
557 switch (command) {
558 .MIDI => try line(out, "MIDI {s} {d}", .{ w[1], (try session.note(w[1])).midi() }),
559 .NOTE => {
560 const number_value = try session.number(w[1]);
561 try line(out, "NOTE {d} {f} {f}", .{ number_value, standardNote(number_value, false), standardNote(number_value, true) });
562 },
563 .FREQ => {
564 const input = try session.note(w[1]);
565 const value = (@as(f64, @floatFromInt(session.tuning.*)) / 100.0) * @exp2(@as(f64, @floatFromInt(input.midi() - 69)) / 12.0);
566 try line(out, "FREQ {s} {f}", .{ w[1], Frequency{ .cents = @intFromFloat(@floor(value * 100.0 + 0.5)) } });
567 },
568 .TUNING => {
569 const value = try session.frequency(w[1], true);
570 session.tuning.* = value;
571 try line(out, "TUNING {f}", .{Frequency{ .cents = value }});
572 },
573 .PITCH => {
574 const value = try session.frequency(w[1], false);
575 const x = 69.0 + 12.0 * @log2(@as(f64, @floatFromInt(value)) / @as(f64, @floatFromInt(session.tuning.*)));
576 const midi: i32 = @intFromFloat(@floor(x + 0.5));
577 if (midi < 0 or midi > 127) return session.fail("OUT OF RANGE", .{});
578 const cents: i32 = @intFromFloat(@floor(100.0 * (x - @as(f64, @floatFromInt(midi))) + 0.5));
579 try line(out, "PITCH {f} {f} {f}", .{ Frequency{ .cents = value }, standardNote(midi, false), Signed{ .value = cents, .positive_zero = true } });
580 },
581 .INTERVAL => {
582 const a = try session.note(w[1]);
583 const b = try session.note(w[2]);
584 const motion = try session.between(a, b);
585 try line(out, "INTERVAL {s} {s} {f} {s} semitones={d} simple={f} class={s}", .{
586 w[1], w[2], motion.interval, if (motion.down) "down" else "up", motion.semitones, motion.interval.simple(), motion.interval.consonance(),
587 });
588 },
589 .INVERT => {
590 const original = try session.interval(w[1]);
591 const simple = original.simple();
592 const quality_value: IntervalQuality = switch (simple.quality) {
593 .P => .P,
594 .M => .m,
595 .m => .M,
596 .A => .d,
597 .d => .A,
598 .AA => .dd,
599 .dd => .AA,
600 };
601 const inverted: Interval = .{ .quality = quality_value, .number = 9 - simple.number };
602 if (inverted.number == 1 and (quality_value == .d or quality_value == .dd)) return session.fail("CANNOT INVERT", .{});
603 try line(out, "INVERT {s} {f} {f}", .{ w[1], simple, inverted });
604 },
605 .TRANSPOSE => {
606 const input = try session.note(w[1]);
607 const interval_value = try session.interval(w[2]);
608 if (!equal(w[3], "UP") and !equal(w[3], "DOWN")) return session.fail("BAD DIRECTION {s}", .{w[3]});
609 const sign: i32 = if (equal(w[3], "UP")) 1 else -1;
610 const position = input.position() + sign * interval_value.distance();
611 const midi = input.midi() + sign * interval_value.semitones();
612 const letter: usize = @intCast(@mod(position, 7));
613 const octave = @divFloor(position, 7);
614 const alteration = midi - (12 * (octave + 1) + naturals[letter]);
615 if (alteration < -2 or alteration > 2) return session.fail("CANNOT SPELL", .{});
616 const result: Note = .{ .pitch = .{ .letter = letter, .alteration = alteration }, .octave = octave };
617 if (!result.valid()) return session.fail("OUT OF RANGE", .{});
618 try line(out, "TRANSPOSE {s} {s} {s} {f}", .{ w[1], w[2], w[3], result });
619 },
620 .SCALE => {
621 const root = try session.pitch(w[1]);
622 const name = try session.scaleName(w[2]);
623 const pattern = name.pattern();
624 const pitches = try session.build(root, pattern);
625 try out.print("SCALE {s} {s}", .{ w[1], w[2] });
626 try session.writePitches(pitches.list());
627 try out.writeAll(" steps=");
628 for (pattern.list(), 0..) |interval_value, i| {
629 if (i != 0) try out.writeByte(',');
630 const next: i32 = if (i + 1 == pattern.len) 12 else pattern.items[i + 1].semitones();
631 try out.print("{d}", .{next - interval_value.semitones()});
632 }
633 try out.writeByte('\n');
634 },
635 .CHORD, .VOICE => {
636 var input: ?Note = null;
637 const root = if (command == .VOICE) blk: {
638 input = try session.note(w[1]);
639 break :blk input.?.pitch;
640 } else try session.pitch(w[1]);
641 const quality_value = try session.quality(w[2]);
642 const inversion_value = if (count == 4) try session.inversion(w[3], quality_value) else 0;
643 const tones = try session.build(root, &quality_value.data().pattern);
644 var reordered: Pitches = .{ .len = tones.len };
645 for (0..tones.len) |i| reordered.items[i] = tones.items[(i + inversion_value) % tones.len];
646 try out.print("{s} {f}", .{ w[0], Chord{ .root = root, .quality = quality_value } });
647 if (inversion_value != 0) try out.print("/{f}", .{reordered.items[0]});
648 if (command == .CHORD) {
649 try session.writePitches(reordered.list());
650 } else {
651 var notes: [7]Note = undefined;
652 var previous = input.?.midi();
653 for (reordered.list(), 0..) |pitch_value, i| {
654 var midi = previous;
655 if (i != 0 or inversion_value != 0) {
656 midi += @mod(pitch_value.class() - previous, 12);
657 if (midi == previous) midi += 12;
658 }
659 notes[i] = fromMidi(pitch_value, midi);
660 if (!notes[i].valid()) return session.fail("OUT OF RANGE", .{});
661 previous = midi;
662 }
663 for (notes[0..tones.len]) |note_value| try out.print(" {f}", .{note_value});
664 try out.writeAll(" midi=");
665 for (notes[0..tones.len], 0..) |note_value, i| {
666 if (i != 0) try out.writeByte(',');
667 try out.print("{d}", .{note_value.midi()});
668 }
669 }
670 try out.writeByte('\n');
671 },
672 .IDENTIFY => {
673 var ordered: [15]IndexedNote = undefined;
674 var mask: u16 = 0;
675 for (w[1..], 0..) |word, i| {
676 const note_value = try session.note(word);
677 ordered[i] = .{ .note = note_value, .index = i };
678 mask |= bit(note_value.pitch.class());
679 }
680 std.sort.heap(IndexedNote, ordered[0 .. count - 1], {}, IndexedNote.less);
681 var candidates: [15]Note = undefined;
682 var candidate_count: usize = 0;
683 var seen: u16 = 0;
684 for (ordered[0 .. count - 1]) |entry| {
685 const note_bit = bit(entry.note.pitch.class());
686 if ((seen & note_bit) != 0) continue;
687 seen |= note_bit;
688 candidates[candidate_count] = entry.note;
689 candidate_count += 1;
690 }
691 const bass = candidates[0];
692 for (candidates[0..candidate_count]) |root| {
693 for (quality_table, 0..) |entry, q| {
694 const quality_value: QualityName = @fromBackingInt(@intCast(q));
695 if (chordMask(root, quality_value) != mask) continue;
696 var inversion_value: usize = 0;
697 for (entry.pattern.list(), 0..) |interval_value, i| {
698 if (@mod(root.midi() + interval_value.semitones(), 12) == bass.pitch.class()) {
699 inversion_value = i;
700 break;
701 }
702 }
703 try out.print("IDENTIFY {f}", .{Chord{ .root = root.pitch, .quality = quality_value }});
704 if (inversion_value != 0) try out.print("/{f}", .{bass.pitch});
705 try line(out, " root={f} quality={s} inversion={d}", .{ root.pitch, @tagName(quality_value), inversion_value });
706 return;
707 }
708 }
709 try line(out, "IDENTIFY UNKNOWN", .{});
710 },
711 .MELODY => {
712 var notes: [15]Note = undefined;
713 for (w[1..], 0..) |word, i| notes[i] = try session.note(word);
714 const values = notes[0 .. count - 1];
715 var low = values[0];
716 var high = values[0];
717 for (values) |note_value| {
718 if (note_value.midi() < low.midi()) low = note_value;
719 if (note_value.midi() > high.midi()) high = note_value;
720 }
721 const range = (try session.between(low, high)).interval;
722 var steps: [14]Motion = undefined;
723 var leaps: usize = 0;
724 for (1..values.len) |i| {
725 steps[i - 1] = try session.between(values[i - 1], values[i]);
726 if (absolute(values[i].midi() - values[i - 1].midi()) >= 3) leaps += 1;
727 }
728 try out.print("MELODY n={d} low={f} high={f} range={f} leaps={d} steps=", .{ values.len, low, high, range, leaps });
729 for (steps[0 .. values.len - 1], 0..) |motion, i| {
730 if (i != 0) try out.writeByte(',');
731 try out.print("{c}{f}", .{ @as(u8, if (motion.down) '-' else '+'), motion.interval });
732 }
733 try out.writeByte('\n');
734 },
735 .ENHARMONIC => {
736 const input = try session.note(w[1]);
737 var notes: [7]Note = undefined;
738 var len: usize = 0;
739 for (0..7) |letter| {
740 var alteration: i32 = -2;
741 while (alteration <= 2) : (alteration += 1) {
742 const pitch_value: Pitch = .{ .letter = letter, .alteration = alteration };
743 if (pitch_value.class() != input.pitch.class()) continue;
744 const note_value = fromMidi(pitch_value, input.midi());
745 if (note_value.valid()) {
746 notes[len] = note_value;
747 len += 1;
748 }
749 }
750 }
751 std.sort.heap(Note, notes[0..len], {}, struct {
752 fn less(_: void, a: Note, b: Note) bool {
753 return a.position() < b.position();
754 }
755 }.less);
756 try out.print("ENHARMONIC {s}", .{w[1]});
757 for (notes[0..len]) |note_value| try out.print(" {f}", .{note_value});
758 try out.writeByte('\n');
759 },
760 .KEYSIG, .RELATIVE, .DIATONIC, .PROGRESSION, .ANALYZE, .DEGREE => {
761 const key_value = try session.key(w[1], w[2]);
762 if (command == .KEYSIG) {
763 const fifths = key_value.fifths();
764 if (fifths == 0) return line(out, "KEYSIG {f} none", .{key_value});
765 const names = if (fifths > 0) [_][]const u8{ "F#", "C#", "G#", "D#", "A#", "E#", "B#" } else [_][]const u8{ "Bb", "Eb", "Ab", "Db", "Gb", "Cb", "Fb" };
766 try out.print("KEYSIG {f} {s}={d}", .{ key_value, if (fifths > 0) "sharps" else "flats", absolute(fifths) });
767 for (names[0..@intCast(absolute(fifths))]) |name| try out.print(" {s}", .{name});
768 try out.writeByte('\n');
769 return;
770 }
771 if (command == .RELATIVE) return line(out, "RELATIVE {f} major={f} minor={f}", .{ key_value, keyAt(key_value.fifths(), .major).tonic, keyAt(key_value.fifths(), .minor).tonic });
772 if (command == .DEGREE) {
773 const pitch_value = try session.pitch(w[3]);
774 return line(out, "DEGREE {f} {s} {f}", .{ key_value, w[3], try session.label(key_value, pitch_value) });
775 }
776 if (command == .ANALYZE) {
777 var chords: [13]Chord = undefined;
778 for (w[3..], 0..) |word, i| chords[i] = try session.symbol(word);
779 try out.print("ANALYZE {f}", .{key_value});
780 for (chords[0 .. count - 3]) |chord| {
781 const degree = try session.label(key_value, chord.root);
782 try out.writeByte(' ');
783 try degree.prefix(out);
784 try out.print("{f}", .{Roman{ .degree = degree.index, .quality = chord.quality }});
785 }
786 try out.writeByte('\n');
787 return;
788 }
789 if (command == .PROGRESSION) {
790 var numerals: [13]Numeral = undefined;
791 for (w[3..], 0..) |word, i| numerals[i] = try session.numeral(word);
792 const key_scale = try session.scale(key_value);
793 try line(out, "PROGRESSION {f} {d}", .{ key_value, count - 3 });
794 for (numerals[0 .. count - 3], 0..) |numeral_value, i| {
795 const degree = key_scale.items[numeral_value.degree];
796 const root: Pitch = .{ .letter = degree.letter, .alteration = degree.alteration + numeral_value.alteration };
797 const tones = try session.build(root, &numeral_value.quality.data().pattern);
798 try out.print("{s} {f}", .{ w[i + 3], Chord{ .root = root, .quality = numeral_value.quality } });
799 try session.writePitches(tones.list());
800 try out.writeByte('\n');
801 }
802 try line(out, "CADENCE {s}", .{cadence(numerals[0 .. count - 3])});
803 return;
804 }
805 const key_scale = try session.scale(key_value);
806 try line(out, "DIATONIC {f}", .{key_value});
807 for (0..2) |kind| {
808 try out.writeAll(if (kind == 0) "TRIADS" else "SEVENTHS");
809 for (0..7) |i| {
810 const third = @mod(key_scale.items[(i + 2) % 7].class() - key_scale.items[i].class(), 12);
811 const fifth = @mod(key_scale.items[(i + 4) % 7].class() - key_scale.items[i].class(), 12);
812 const seventh = @mod(key_scale.items[(i + 6) % 7].class() - key_scale.items[i].class(), 12);
813 const quality_value: QualityName = if (kind == 0)
814 (if (third == 4) .maj else if (fifth == 7) .min else .dim)
815 else
816 (if (third == 4) (if (seventh == 11) .maj7 else .@"7") else if (fifth == 7) .m7 else .m7b5);
817 try out.print(" {f}={f}", .{ Roman{ .degree = i, .quality = quality_value }, Chord{ .root = key_scale.items[i], .quality = quality_value } });
818 }
819 try out.writeByte('\n');
820 }
821 },
822 .CIRCLE => {
823 const mode_value = try session.mode(w[1]);
824 try out.print("CIRCLE {s}", .{w[1]});
825 var fifths: i32 = -7;
826 while (fifths <= 7) : (fifths += 1) try out.print(" {f}", .{keyAt(fifths, mode_value).tonic});
827 try out.writeByte('\n');
828 },
829 .FINDKEY => {
830 var pitches: [15]Pitch = undefined;
831 for (w[1..], 0..) |word, i| pitches[i] = try session.pitch(word);
832 var keys: [30]Key = undefined;
833 var len: usize = 0;
834 var fifths: i32 = -7;
835 while (fifths <= 7) : (fifths += 1) {
836 for ([_]Mode{ .major, .minor }) |mode_value| {
837 const key_value = keyAt(fifths, mode_value);
838 const key_scale = try session.scale(key_value);
839 var matches = true;
840 for (pitches[0 .. count - 1]) |pitch_value| {
841 var found = false;
842 for (key_scale.list()) |candidate| {
843 if (candidate.same(pitch_value)) {
844 found = true;
845 break;
846 }
847 }
848 if (!found) {
849 matches = false;
850 break;
851 }
852 }
853 if (matches) {
854 keys[len] = key_value;
855 len += 1;
856 }
857 }
858 }
859 try out.print("FINDKEY {d}", .{len});
860 for (keys[0..len], 0..) |key_value, i| try out.print("{s}{f}", .{ if (i == 0) " " else ", ", key_value });
861 try out.writeByte('\n');
862 },
863 .SPELL => {
864 const midi = try session.number(w[1]);
865 const key_value = try session.key(w[2], w[3]);
866 const key_scale = try session.scale(key_value);
867 var pitch_value = standardNote(midi, key_value.fifths() < 0).pitch;
868 for (key_scale.list()) |candidate| {
869 if (candidate.class() == @mod(midi, 12)) {
870 pitch_value = candidate;
871 break;
872 }
873 }
874 const note_value = fromMidi(pitch_value, midi);
875 if (!note_value.valid()) return session.fail("OUT OF RANGE", .{});
876 try line(out, "SPELL {d} {f} {f} {f}", .{ midi, key_value, note_value, try session.label(key_value, pitch_value) });
877 },
878 .KEYDIST => {
879 const a = try session.key(w[1], w[2]);
880 const b = try session.key(w[3], w[4]);
881 var difference = @mod(b.fifths() - a.fifths(), 12);
882 if (difference > 6) difference -= 12;
883 const first_scale = try session.scale(a);
884 const second_scale = try session.scale(b);
885 const first = first_scale.mask();
886 const second = second_scale.mask();
887 const same_tonic = a.tonic.class() == b.tonic.class();
888 const relation: []const u8 = if (first == second) (if (same_tonic) "same" else "relative") else if (same_tonic) "parallel" else if (absolute(difference) == 1) "close" else "distant";
889 try line(out, "KEYDIST {f} {f} fifths={f} common={d} relation={s}", .{ a, b, Signed{ .value = difference }, @popCount(first & second), relation });
890 },
891 }
892 }
893};

895fn readLine(gpa: Allocator, input: *std.Io.Reader) ![]const u8 {
896 var buffer: Writer.Allocating = .init(gpa);
897 defer buffer.deinit();
898 _ = try input.streamDelimiterEnding(&buffer.writer, '\n');
899 if (input.seek < input.end) input.toss(1) else if (buffer.written().len == 0) return error.EndOfStream;
900 const text = try buffer.toOwnedSlice();
901 return if (text.len > 0 and text[text.len - 1] == '\r') text[0 .. text.len - 1] else text;
902}
903pub fn main(init: std.process.Init) !void {
904 var header = std.heap.ArenaAllocator.init(std.heap.page_allocator);
905 defer header.deinit();
906 var input_buffer: [1024]u8 = undefined;
907 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
908 var output_buffer: [4096]u8 = undefined;
909 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
910 var tuning: i32 = 44000;
911 const count: usize = @intCast(integer(try readLine(header.allocator(), &input.interface)));
912 for (0..count) |_| {
913 var scratch = std.heap.ArenaAllocator.init(std.heap.page_allocator);
914 defer scratch.deinit();
915 const gpa = scratch.allocator();
916 const text = try readLine(gpa, &input.interface);
917 var response: Writer.Allocating = .init(gpa);
918 defer response.deinit();
919 var session: Session = .{ .gpa = gpa, .tuning = &tuning, .out = &response.writer };
920 session.execute(text) catch |err| {
921 if (err != error.Invalid) return err;
922 try line(&output.interface, "ERROR {s}", .{session.problem});
923 continue;
924 };
925 try output.interface.writeAll(response.written());
926 }
927 try output.interface.flush();
928}
