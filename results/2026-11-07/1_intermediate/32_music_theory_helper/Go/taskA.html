25func mod(n, divisor int) int {
26 r := n % divisor
27 if r < 0 {
28 r += divisor

32func floorDiv(n, divisor int) int { return (n - mod(n, divisor)) / divisor }

34 if n < 0 {
35 return -n

39func digits(text string) bool {
40 if text == "" {
41 return false
42 }
43 for _, c := range text {
44 if c < '0' || c > '9' {
45 return false
46 }
47 }
48 return true
49}
50func canonical(text string, low, high int) (int, bool) {
51 if !digits(text) || len(text) > 1 && text[0] == '0' {
52 return 0, false
53 }
54 n, err := strconv.Atoi(text)
55 return n, err == nil && n >= low && n <= high
56}
57func parseNumber(text string) (int, error) {
58 if value, ok := canonical(text, 0, 127); ok {
59 return value, nil
60 }
61 return 0, bad("NUMBER", text)
62}
63func parseFrequency(text string, tuning bool) (int, error) {
64 parts := strings.Split(text, ".")
65 if len(parts) > 2 || len(parts[0]) > 5 || !digits(parts[0]) {
66 return 0, bad("FREQUENCY", text)
67 }
68 whole, _ := strconv.Atoi(parts[0])
69 cents := whole * 100
70 if len(parts) == 2 {
71 if len(parts[1]) > 2 || !digits(parts[1]) {
72 return 0, bad("FREQUENCY", text)
73 }
74 fraction, _ := strconv.Atoi(parts[1])
75 if len(parts[1]) == 1 {
76 fraction *= 10
77 }
78 cents += fraction
79 }
80 if cents == 0 || tuning && (cents < 40000 || cents > 48000) {
81 return 0, bad("FREQUENCY", text)
82 }
83 return cents, nil
84}
85func frequencyText(cents int) string { return fmt.Sprintf("%d.%02d", cents/100, cents%100) }
86func join[T any](values []T, separator string) string {
87 parts := make([]string, len(values))
88 for i, value := range values {
89 parts[i] = fmt.Sprint(value)
90 }
91 return strings.Join(parts, separator)
92}

94type Pitch struct{ Letter, Alteration int }
95
96func pitchPrefix(text string) (Pitch, int, bool) {
97 if text == "" {
98 return Pitch{}, 0, false
99 }
100 letter := strings.IndexByte("CDEFGAB", text[0])
101 if letter < 0 {
102 return Pitch{}, 0, false
103 }
104 pitch, end := Pitch{letter, 0}, 1
105 if end < len(text) {
106 switch text[end] {
107 case 'b':
108 pitch.Alteration = -1
109 end++
110 if end < len(text) && text[end] == 'b' {
111 pitch.Alteration = -2
112 end++
113 }
114 case '#':
115 pitch.Alteration = 1
116 end++
117 case 'x':
118 pitch.Alteration = 2
119 end++
120 }
121 }
122 return pitch, end, true
123}
124func parsePitch(text string) (Pitch, error) {
125 pitch, end, ok := pitchPrefix(text)
126 if !ok || end != len(text) {
127 return Pitch{}, bad("PITCH", text)
128 }
129 return pitch, nil
130}
131func (p Pitch) Class() int { return mod(naturals[p.Letter]+p.Alteration, 12) }
132func (p Pitch) String() string { return string("CDEFGAB"[p.Letter]) + accidentals[p.Alteration+2] }
133func (p Pitch) Raise(tone Tone) (Pitch, error) {
134 target := p.Letter + tone.Letters
135 alteration := naturals[p.Letter] + p.Alteration + tone.Semitones - naturals[target%7] - 12*(target/7)
136 if alteration < -2 || alteration > 2 {
137 return Pitch{}, failure("CANNOT SPELL")
138 }
139 return Pitch{target % 7, alteration}, nil
140}
141func fifthPitch(fifths int) Pitch {
142 return Pitch{[...]int{3, 0, 4, 1, 5, 2, 6}[mod(fifths+1, 7)], floorDiv(fifths+1, 7)}
143}

145type Note struct {
146 Pitch Pitch
147 Octave int
148}

150func parseNote(text string) (Note, error) {
151 pitch, end, ok := pitchPrefix(text)
152 if !ok {
153 return Note{}, bad("NOTE", text)
154 }
155 rest := text[end:]
156 var octave int
157 if rest == "-1" {
158 octave = -1
159 } else if len(rest) == 1 && rest[0] >= '0' && rest[0] <= '9' {
160 octave = int(rest[0] - '0')
161 } else {
162 return Note{}, bad("NOTE", text)
163 }
164 note := Note{pitch, octave}
165 if !note.Valid() {
166 return Note{}, bad("NOTE", text)
167 }
168 return note, nil
169}
170func (n Note) Midi() int { return 12*(n.Octave+1) + naturals[n.Pitch.Letter] + n.Pitch.Alteration }
171func (n Note) Position() int { return 7*n.Octave + n.Pitch.Letter }
172func (n Note) Valid() bool { return n.Octave >= -1 && n.Octave <= 9 && n.Midi() >= 0 && n.Midi() <= 127 }
173func (n Note) String() string { return fmt.Sprintf("%s%d", n.Pitch, n.Octave) }
174func noteFromPitch(pitch Pitch, value int) Note {
175 return Note{pitch, (value-naturals[pitch.Letter]-pitch.Alteration)/12 - 1}
176}
177func spelling(value int, flat bool) Note {
178 name := sharpNames[value%12]
179 if flat {
180 name = flatNames[value%12]
181 }
182 pitch, _ := parsePitch(name)
183 return Note{pitch, value/12 - 1}
184}
185func parseNotes(words []string) ([]Note, error) {
186 notes := make([]Note, len(words))
187 for i, word := range words {
188 note, err := parseNote(word)
189 if err != nil {
190 return nil, err
191 }
192 notes[i] = note
193 }
194 return notes, nil
195}

197type IntervalQuality int

199const (
200 Perfect IntervalQuality = iota
201 Major
202 Minor
203 Augmented
204 Diminished
205 DoubleAugmented
206 DoubleDiminished

209func (q IntervalQuality) String() string { return [...]string{"P", "M", "m", "A", "d", "AA", "dd"}[q] }
210func (q IntervalQuality) Delta(perfect bool) (int, bool) {
211 switch q {
212 case Perfect:
213 return 0, perfect
214 case Major:
215 return 0, !perfect
216 case Minor:
217 return -1, !perfect
218 case Augmented:
219 return 1, true
220 case DoubleAugmented:
221 return 2, true
222 case Diminished:
223 if perfect {
224 return -1, true
225 }
226 return -2, true
227 default:
228 if perfect {
229 return -2, true
230 }
231 return -3, true
232 }
233}
234func (q IntervalQuality) Inverse() IntervalQuality {
235 return [...]IntervalQuality{Perfect, Minor, Major, Diminished, Augmented, DoubleDiminished, DoubleAugmented}[q]
236}

238type Interval struct {

243type Motion struct {

249func perfectType(number int) bool {
250 r := (number - 1) % 7
251 return r == 0 || r == 3 || r == 4
252}
253func (i Interval) Forbidden() bool { return i.Number == 1 && (i.Quality == Diminished || i.Quality == DoubleDiminished) }
254func parseInterval(text string) (Interval, error) {
255 for q := Perfect; q <= DoubleDiminished; q++ {
256 rest, found := strings.CutPrefix(text, q.String())
257 if !found {
258 continue
259 }
260 number, ok := canonical(rest, 1, 99)
261 if !ok {
262 continue
263 }
264 interval := Interval{q, number}
265 _, permitted := q.Delta(perfectType(number))
266 if permitted && !interval.Forbidden() {
267 return interval, nil
268 }
269 }
270 return Interval{}, bad("INTERVAL", text)
271}
272func (i Interval) String() string { return fmt.Sprintf("%s%d", i.Quality, i.Number) }
273func (i Interval) Tone() Tone {
274 s := i.Number - 1
275 delta, _ := i.Quality.Delta(perfectType(i.Number))
276 return Tone{s, 12*(s/7) + naturals[s%7] + delta}
277}
278func (i Interval) Simple() Interval {
279 for i.Number > 8 {
280 i.Number -= 7
281 }
282 return i
283}
284func (i Interval) Class() string {
285 simple := i.Simple()
286 if simple.Quality == Perfect && (simple.Number == 1 || simple.Number == 5 || simple.Number == 8) {
287 return "perfect"
288 }
289 if (simple.Quality == Major || simple.Quality == Minor) && (simple.Number == 3 || simple.Number == 6) {
290 return "imperfect"
291 }
292 return "dissonant"
293}
294func between(a, b Note) (Motion, error) {
295 distance, semitones := b.Position()-a.Position(), b.Midi()-a.Midi()
296 down := distance < 0 || distance == 0 && semitones < 0
297 if down {
298 distance, semitones = -distance, -semitones
299 }
300 difference := semitones - 12*(distance/7) - naturals[distance%7]
301 for q := Perfect; q <= DoubleDiminished; q++ {
302 if delta, ok := q.Delta(perfectType(distance+1)); ok && delta == difference {
303 return Motion{Interval{q, distance + 1}, down, semitones}, nil
304 }
305 }
306 return Motion{}, failure("UNNAMEABLE INTERVAL")
307}

419func makeKey(tonic Pitch, mode int) (Key, error) {
420 key := Key{Tonic: tonic, Mode: mode, Fifths: fifthsValues[tonic.Letter] + 7*tonic.Alteration + modes[mode].Offset}
421 if key.Fifths < -7 || key.Fifths > 7 {
422 return Key{}, failure("THEORETICAL KEY")
423 }
424 scale, err := build(tonic, scales[modes[mode].Scale].Tones)
425 if err != nil {
426 return Key{}, err
427 }
428 copy(key.Scale[:], scale)
429 return key, nil
430}

502func chordCommand(command string, a []string) (string, error) {
503 var note Note
504 var root Pitch
505 var err error
506 if command == "VOICE" {
507 note, err = parseNote(a[0])
508 root = note.Pitch
509 } else {
510 root, err = parsePitch(a[0])
511 }
512 if err != nil {
513 return "", err
514 }
515 quality, err := parseQuality(a[1])
516 if err != nil {
517 return "", err
518 }
519 chord, inversion := chords[quality], 0
520 if len(a) == 3 {
521 var ok bool
522 inversion, ok = canonical(a[2], 0, len(chord.Tones)-1)
523 if !ok {
524 return "", bad("INVERSION", a[2])
525 }
526 }
527 tones, err := build(root, chord.Tones)
528 if err != nil {
529 return "", err
530 }
531 ordered := make([]Pitch, len(tones))
532 for i := range ordered {
533 ordered[i] = tones[(i+inversion)%len(tones)]
534 }
535 var bass *Pitch
536 if inversion != 0 {
537 bass = &ordered[0]
538 }
539 name := symbol(root, quality, bass)
540 if command == "CHORD" {
541 return fmt.Sprintf("CHORD %s %s", name, join(ordered, " ")), nil
542 }
543 notes := make([]Note, len(ordered))
544 values := make([]int, len(ordered))
545 value := note.Midi()
546 for i, pitch := range ordered {
547 if i != 0 || inversion != 0 {
548 difference := mod(pitch.Class()-value, 12)
549 if difference == 0 {
550 difference = 12
551 }
552 value += difference
553 }
554 notes[i] = noteFromPitch(pitch, value)
555 if !notes[i].Valid() {
556 return "", failure("OUT OF RANGE")
557 }
558 values[i] = value
559 }
560 return fmt.Sprintf("VOICE %s %s midi=%s", name, join(notes, " "), join(values, ",")), nil
561}
562func identify(notes []Note) string {
563 slices.SortStableFunc(notes, func(a, b Note) int { return a.Midi() - b.Midi() })
564 candidates := notes[:0]
565 var set uint
566 for _, note := range notes {
567 bit := uint(1) << note.Pitch.Class()
568 if set&bit == 0 {
569 set |= bit
570 candidates = append(candidates, note)
571 }
572 }
573 bass := candidates[0].Pitch
574 for _, root := range candidates {
575 for quality, chord := range chords {
576 var mask uint
577 for _, tone := range chord.Tones {
578 mask |= 1 << mod(root.Midi()+tone.Semitones, 12)
579 }
580 if mask != set {
581 continue
582 }
583 inversion := 0
584 for mod(root.Midi()+chord.Tones[inversion].Semitones, 12) != bass.Class() {
585 inversion++
586 }
587 var slash *Pitch
588 if inversion != 0 {
589 slash = &bass
590 }
591 return fmt.Sprintf("IDENTIFY %s root=%s quality=%s inversion=%d", symbol(root.Pitch, quality, slash), root.Pitch, chord.Name, inversion)
592 }
593 }
594 return "IDENTIFY UNKNOWN"
595}
596func melody(notes []Note, words []string) (string, error) {
597 low, high := 0, 0
598 for i, note := range notes {
599 if note.Midi() < notes[low].Midi() {
600 low = i
601 }
602 if note.Midi() > notes[high].Midi() {
603 high = i
604 }
605 }
606 rangeMotion, err := between(notes[low], notes[high])
607 if err != nil {
608 return "", err
609 }
610 steps := make([]string, len(notes)-1)
611 leaps := 0
612 for i := 1; i < len(notes); i++ {
613 motion, err := between(notes[i-1], notes[i])
614 if err != nil {
615 return "", err
616 }
617 sign := "+"
618 if motion.Down {
619 sign = "-"
620 }
621 steps[i-1] = sign + motion.Interval.String()
622 if abs(notes[i].Midi()-notes[i-1].Midi()) >= 3 {
623 leaps++
624 }
625 }
626 return fmt.Sprintf("MELODY n=%d low=%s high=%s range=%s leaps=%d steps=%s", len(notes), words[low], words[high], rangeMotion.Interval, leaps, strings.Join(steps, ",")), nil
627}
628func cadence(numerals []Numeral) string {
629 if len(numerals) < 2 {
630 return "none"
631 }
632 x, y := numerals[len(numerals)-2], numerals[len(numerals)-1]
633 if x.Alteration == 0 && y.Alteration == 0 {
634 if x.Degree == 4 && y.Degree == 0 {
635 return "authentic"
636 }
637 if x.Degree == 3 && y.Degree == 0 {
638 return "plagal"
639 }
640 if x.Degree == 4 && y.Degree == 5 {
641 return "deceptive"
642 }
643 }
644 if y.Alteration == 0 && y.Degree == 4 {
645 return "half"
646 }
647 return "none"
648}
649func diatonic(key Key) string {
650 rows := []string{"DIATONIC " + key.String()}
651 for size := 3; size <= 4; size++ {
652 items := make([]string, 7)
653 for i, root := range key.Scale {
654 third := mod(key.Scale[(i+2)%7].Class()-root.Class(), 12)
655 fifth := mod(key.Scale[(i+4)%7].Class()-root.Class(), 12)
656 var quality int
657 if size == 3 {
658 if third == 4 {
659 quality = 0
660 } else if fifth == 7 {
661 quality = 1
662 } else {
663 quality = 2
664 }
665 } else {
666 seventh := mod(key.Scale[(i+6)%7].Class()-root.Class(), 12)
667 if third == 4 {
668 quality = 8
669 if seventh == 11 {
670 quality = 9
671 }
672 } else if fifth == 7 {
673 quality = 10
674 } else {
675 quality = 12
676 }
677 }
678 items[i] = roman(i, quality) + "=" + symbol(root, quality, nil)
679 }
680 prefix := "TRIADS "
681 if size == 4 {
682 prefix = "SEVENTHS "
683 }
684 rows = append(rows, prefix+strings.Join(items, " "))
685 }
686 return strings.Join(rows, "\n")
687}
688func keyCommand(command string, a []string) (string, error) {
689 key, err := parseKey(a)
690 if err != nil {
691 return "", err
692 }
693 switch command {
694 case "KEYSIG":
695 p := key.Fifths
696 if p == 0 {
697 return fmt.Sprintf("KEYSIG %s none", key), nil
698 }
699 names, kind := []string{"F#", "C#", "G#", "D#", "A#", "E#", "B#"}, "sharps"
700 if p < 0 {
701 names, kind = []string{"Bb", "Eb", "Ab", "Db", "Gb", "Cb", "Fb"}, "flats"
702 }
703 return fmt.Sprintf("KEYSIG %s %s=%d %s", key, kind, abs(p), strings.Join(names[:abs(p)], " ")), nil
704 case "RELATIVE":
705 return fmt.Sprintf("RELATIVE %s major=%s minor=%s", key, fifthPitch(key.Fifths), fifthPitch(key.Fifths+3)), nil
706 case "DEGREE":
707 pitch, err := parsePitch(a[2])
708 if err != nil {
709 return "", err
710 }
711 return fmt.Sprintf("DEGREE %s %s %s", key, a[2], key.Label(pitch)), nil
712 case "KEYDIST":
713 other, err := parseKey(a[2:])
714 if err != nil {
715 return "", err
716 }
717 distance := mod(other.Fifths-key.Fifths, 12)
718 if distance > 6 {
719 distance -= 12
720 }
721 sameTonic := key.Tonic.Class() == other.Tonic.Class()
722 relation := "distant"
723 if key.Set() == other.Set() {
724 relation = "relative"
725 if sameTonic {
726 relation = "same"
727 }
728 } else if sameTonic {
729 relation = "parallel"
730 } else if abs(distance) == 1 {
731 relation = "close"
732 }
733 distanceText := strconv.Itoa(distance)
734 if distance > 0 {
735 distanceText = "+" + distanceText
736 }
737 return fmt.Sprintf("KEYDIST %s %s fifths=%s common=%d relation=%s", key, other, distanceText, bits.OnesCount(key.Set()&other.Set()), relation), nil
738 case "DIATONIC":
739 return diatonic(key), nil
740 case "ANALYZE":
741 symbols := make([]Symbol, len(a)-2)
742 for i, word := range a[2:] {
743 parsed, err := parseSymbol(word)
744 if err != nil {
745 return "", err
746 }
747 symbols[i] = parsed
748 }
749 items := make([]string, len(symbols))
750 for i, chord := range symbols {
751 items[i] = key.Prefix(chord.Root) + roman(key.Degree(chord.Root), chord.Quality)
752 }
753 return fmt.Sprintf("ANALYZE %s %s", key, strings.Join(items, " ")), nil
754 case "PROGRESSION":
755 numerals := make([]Numeral, len(a)-2)
756 for i, word := range a[2:] {
757 parsed, err := parseNumeral(word)
758 if err != nil {
759 return "", err
760 }
761 numerals[i] = parsed
762 }
763 rows := []string{fmt.Sprintf("PROGRESSION %s %d", key, len(numerals))}
764 for i, numeral := range numerals {
765 root := key.Scale[numeral.Degree]
766 root.Alteration += numeral.Alteration
767 tones, err := build(root, chords[numeral.Quality].Tones)
768 if err != nil {
769 return "", err
770 }
771 rows = append(rows, fmt.Sprintf("%s %s %s", a[i+2], symbol(root, numeral.Quality, nil), join(tones, " ")))
772 }
773 rows = append(rows, "CADENCE "+cadence(numerals))
774 return strings.Join(rows, "\n"), nil
775 }
776 return "", failure("UNKNOWN COMMAND " + command)
777}

779var arities = map[string][2]int{
780 "MIDI": {1,1}, "NOTE": {1,1}, "FREQ": {1,1}, "TUNING": {1,1}, "PITCH": {1,1},
781 "INTERVAL": {2,2}, "INVERT": {1,1}, "TRANSPOSE": {3,3}, "SCALE": {2,2},
782 "CHORD": {2,3}, "VOICE": {2,3}, "IDENTIFY": {2,15}, "MELODY": {2,15},
783 "ENHARMONIC": {1,1}, "KEYSIG": {2,2}, "RELATIVE": {2,2}, "CIRCLE": {1,1},
784 "DIATONIC": {2,2}, "PROGRESSION": {3,15}, "ANALYZE": {3,15}, "FINDKEY": {1,15},
785 "DEGREE": {3,3}, "SPELL": {3,3}, "KEYDIST": {4,4},
786}
787
788type Calculator struct{ Tuning int }
789
790func (c *Calculator) Execute(line string) (string, error) {
791 words := strings.Split(line, " ")
792 command, a := words[0], words[1:]
793 arity, found := arities[command]
794 if !found {
795 return "", failure("UNKNOWN COMMAND " + command)
796 }
797 if len(a) < arity[0] || len(a) > arity[1] {
798 return "", failure("ARGUMENT COUNT")
799 }
800 switch command {
801 case "MIDI", "FREQ", "ENHARMONIC":
802 note, err := parseNote(a[0])
803 if err != nil {
804 return "", err
805 }
806 value := note.Midi()
807 if command == "MIDI" {
808 return fmt.Sprintf("MIDI %s %d", a[0], value), nil
809 }
810 if command == "FREQ" {
811 cents := int(math.Floor(float64(c.Tuning)*math.Exp2(float64(value-69)/12) + 0.5))
812 return fmt.Sprintf("FREQ %s %s", a[0], frequencyText(cents)), nil
813 }
814 var notes []Note
815 for octave := -1; octave <= 9; octave++ {
816 for letter := range 7 {
817 alteration := value - 12*(octave+1) - naturals[letter]
818 if alteration >= -2 && alteration <= 2 {
819 notes = append(notes, Note{Pitch{letter, alteration}, octave})
820 }
821 }
822 }
823 return fmt.Sprintf("ENHARMONIC %s %s", a[0], join(notes, " ")), nil
824 case "NOTE":
825 value, err := parseNumber(a[0])
826 if err != nil {
827 return "", err
828 }
829 return fmt.Sprintf("NOTE %d %s %s", value, spelling(value, false), spelling(value, true)), nil
830 case "TUNING", "PITCH":
831 value, err := parseFrequency(a[0], command == "TUNING")
832 if err != nil {
833 return "", err
834 }
835 if command == "TUNING" {
836 c.Tuning = value
837 return "TUNING " + frequencyText(value), nil
838 }
839 x := 69 + 12*math.Log2(float64(value)/float64(c.Tuning))
840 nearest := int(math.Floor(x + 0.5))
841 if nearest < 0 || nearest > 127 {
842 return "", failure("OUT OF RANGE")
843 }
844 cents := int(math.Floor(100*(x-float64(nearest)) + 0.5))
845 return fmt.Sprintf("PITCH %s %s %+d", frequencyText(value), spelling(nearest, false), cents), nil
846 case "INTERVAL":
847 first, err := parseNote(a[0])
848 if err != nil {
849 return "", err
850 }
851 second, err := parseNote(a[1])
852 if err != nil {
853 return "", err
854 }
855 motion, err := between(first, second)
856 if err != nil {
857 return "", err
858 }
859 direction := "up"
860 if motion.Down {
861 direction = "down"
862 }
863 return fmt.Sprintf("INTERVAL %s %s %s %s semitones=%d simple=%s class=%s", a[0], a[1], motion.Interval, direction,
864 motion.Semitones, motion.Interval.Simple(), motion.Interval.Class()), nil
865 case "INVERT":
866 interval, err := parseInterval(a[0])
867 if err != nil {
868 return "", err
869 }
870 simple := interval.Simple()
871 inverse := Interval{simple.Quality.Inverse(), 9 - simple.Number}
872 if inverse.Forbidden() {
873 return "", failure("CANNOT INVERT")
874 }
875 return fmt.Sprintf("INVERT %s %s %s", a[0], simple, inverse), nil
876 case "TRANSPOSE":
877 note, err := parseNote(a[0])
878 if err != nil {
879 return "", err
880 }
881 interval, err := parseInterval(a[1])
882 if err != nil {
883 return "", err
884 }
885 if a[2] != "UP" && a[2] != "DOWN" {
886 return "", bad("DIRECTION", a[2])
887 }
888 sign := 1
889 if a[2] == "DOWN" {
890 sign = -1
891 }
892 tone := interval.Tone()
893 position, value := note.Position()+sign*tone.Letters, note.Midi()+sign*tone.Semitones
894 letter, octave := mod(position, 7), floorDiv(position, 7)
895 alteration := value - 12*(octave+1) - naturals[letter]
896 if alteration < -2 || alteration > 2 {
897 return "", failure("CANNOT SPELL")
898 }
899 result := Note{Pitch{letter, alteration}, octave}
900 if !result.Valid() {
901 return "", failure("OUT OF RANGE")
902 }
903 return fmt.Sprintf("TRANSPOSE %s %s %s %s", a[0], a[1], a[2], result), nil
904 case "SCALE":
905 root, err := parsePitch(a[0])
906 if err != nil {
907 return "", err
908 }
909 scale, err := parseScale(a[1])
910 if err != nil {
911 return "", err
912 }
913 pitches, err := build(root, scale.Tones)
914 if err != nil {
915 return "", err
916 }
917 steps := make([]int, len(scale.Tones))
918 for i, tone := range scale.Tones {
919 next := 12
920 if i+1 < len(scale.Tones) {
921 next = scale.Tones[i+1].Semitones
922 }
923 steps[i] = next - tone.Semitones
924 }
925 return fmt.Sprintf("SCALE %s %s %s steps=%s", a[0], a[1], join(pitches, " "), join(steps, ",")), nil
926 case "CHORD", "VOICE":
927 return chordCommand(command, a)
928 case "IDENTIFY", "MELODY":
929 notes, err := parseNotes(a)
930 if err != nil {
931 return "", err
932 }
933 if command == "IDENTIFY" {
934 return identify(notes), nil
935 }
936 return melody(notes, a)
937 case "CIRCLE":
938 mode, err := parseMode(a[0])
939 if err != nil {
940 return "", err
941 }
942 pitches := make([]Pitch, 15)
943 for i := range pitches {
944 pitches[i] = fifthPitch(i - 7 - modes[mode].Offset)
945 }
946 return fmt.Sprintf("CIRCLE %s %s", a[0], join(pitches, " ")), nil
947 case "FINDKEY":
948 pitches := make([]Pitch, len(a))
949 for i, word := range a {
950 pitch, err := parsePitch(word)
951 if err != nil {
952 return "", err
953 }
954 pitches[i] = pitch
955 }
956 var keys []Key
957 for p := -7; p <= 7; p++ {
958 for mode := range 2 {
959 key, err := makeKey(fifthPitch(p-modes[mode].Offset), mode)
960 if err != nil {
961 return "", err
962 }
963 match := true
964 for _, pitch := range pitches {
965 if !slices.Contains(key.Scale[:], pitch) {
966 match = false
967 }
968 }
969 if match {
970 keys = append(keys, key)
971 }
972 }
973 }
974 result := fmt.Sprintf("FINDKEY %d", len(keys))
975 if len(keys) != 0 {
976 result += " " + join(keys, ", ")
977 }
978 return result, nil
979 case "SPELL":
980 value, err := parseNumber(a[0])
981 if err != nil {
982 return "", err
983 }
984 key, err := parseKey(a[1:])
985 if err != nil {
986 return "", err
987 }
988 pitch := spelling(value, key.Fifths < 0).Pitch
989 for _, candidate := range key.Scale {
990 if candidate.Class() == value%12 {
991 pitch = candidate
992 break
993 }
994 }
995 note := noteFromPitch(pitch, value)
996 if !note.Valid() {
997 return "", failure("OUT OF RANGE")
998 }
999 return fmt.Sprintf("SPELL %d %s %s %s", value, key, note, key.Label(pitch)), nil
1000 default:
1001 return keyCommand(command, a)
1002 }
1003}
