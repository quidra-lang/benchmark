7private val naturals = intArrayOf(0, 2, 4, 5, 7, 9, 11)
8private val fifthsValues = intArrayOf(0, 2, 4, -1, 1, 3, 5)
9private val accidentals = listOf("bb", "b", "", "#", "x")

11private val flatNames = "C Db D Eb E F Gb G Ab A Bb B".split(' ')

13private val integerPattern = Regex("0|[1-9][0-9]*")

16private class CommandError(message: String) : RuntimeException("ERROR $message")
17private fun bad(kind: String, word: String): Nothing = throw CommandError("BAD $kind $word")
18private fun mod(value: Int, divisor: Int): Int {
19 val remainder = value % divisor
20 return if (remainder < 0) remainder + divisor else remainder
21}
22private fun canonical(text: String, range: IntRange): Int? =
23 if (integerPattern.matches(text)) text.toIntOrNull()?.takeIf { it in range } else null
24private fun parseNumber(text: String): Int = canonical(text, 0..127) ?: bad("NUMBER", text)
25private fun frequencyCents(text: String, tuning: Boolean): Int {
26 if (!frequencyPattern.matches(text)) bad("FREQUENCY", text)
27 val parts = text.split('.')
28 var result = parts[0].toInt() * 100
29 if (parts.size == 2) result += parts[1].toInt() * if (parts[1].length == 1) 10 else 1
30 if (result == 0 || tuning && result !in 40000..48000) bad("FREQUENCY", text)
31 return result
32}
33private fun frequencyText(cents: Int) = "${cents / 100}.${(cents % 100).toString().padStart(2, '0')}"

35private data class Pitch(val letter: Int, val alteration: Int) {
36 val pitchClass: Int get() = mod(naturals[letter] + alteration, 12)
37 fun raise(tone: Tone): Pitch {
38 val target = letter + tone.letters
39 val change = naturals[letter] + alteration + tone.semitones - naturals[target % 7] - 12 * (target / 7)
40 if (change !in -2..2) throw CommandError("CANNOT SPELL")
41 return Pitch(target % 7, change)
42 }
43 override fun toString() = "${LETTERS[letter]}${accidentals[alteration + 2]}"
44}
45private data class PitchPrefix(val pitch: Pitch, val end: Int)
46private fun pitchPrefix(text: String): PitchPrefix? {
47 val letter = LETTERS.indexOf(text[0])
48 if (letter < 0) return null
49 var end = 1
50 var alteration = 0
51 when (text.getOrNull(end)) {
52 'b' -> {
53 alteration = -1
54 end++
55 if (text.getOrNull(end) == 'b') { alteration = -2; end++ }
56 }
57 '#' -> { alteration = 1; end++ }
58 'x' -> { alteration = 2; end++ }
59 }
60 return PitchPrefix(Pitch(letter, alteration), end)
61}
62private fun parsePitch(text: String): Pitch {
63 val prefix = pitchPrefix(text)
64 if (prefix == null || prefix.end != text.length) bad("PITCH", text)
65 return prefix.pitch
66}
67private fun fifthPitch(fifths: Int) =
68 Pitch(LETTERS.indexOf("FCGDAEB"[mod(fifths + 1, 7)]), Math.floorDiv(fifths + 1, 7))
69private data class Note(val pitch: Pitch, val octave: Int) {
70 val midi: Int get() = 12 * (octave + 1) + naturals[pitch.letter] + pitch.alteration
71 val position: Int get() = 7 * octave + pitch.letter
72 val valid: Boolean get() = octave in -1..9 && midi in 0..127
73 override fun toString() = "$pitch$octave"
74}
75private fun parseNote(text: String): Note {
76 val prefix = pitchPrefix(text) ?: bad("NOTE", text)
77 val rest = text.substring(prefix.end)
78 val octave = when {
79 rest == "-1" -> -1
80 rest.length == 1 && rest[0] in '0'..'9' -> rest[0] - '0'
81 else -> bad("NOTE", text)
82 }
83 return Note(prefix.pitch, octave).also { if (!it.valid) bad("NOTE", text) }
84}
85private fun fromPitch(pitch: Pitch, midi: Int) =
86 Note(pitch, (midi - naturals[pitch.letter] - pitch.alteration) / 12 - 1)
87private fun spelling(midi: Int, flat: Boolean) =
88 Note(parsePitch((if (flat) flatNames else sharpNames)[midi % 12]), midi / 12 - 1)

90private enum class Quality(val text: String) {
91 PERFECT("P"), MAJOR("M"), MINOR("m"), AUGMENTED("A"), DIMINISHED("d"), DOUBLE_AUGMENTED("AA"), DOUBLE_DIMINISHED("dd");
92 fun delta(perfect: Boolean): Int? = when (this) {
93 PERFECT -> if (perfect) 0 else null
94 MAJOR -> if (perfect) null else 0
95 MINOR -> if (perfect) null else -1
96 AUGMENTED -> 1
97 DIMINISHED -> if (perfect) -1 else -2
98 DOUBLE_AUGMENTED -> 2
99 DOUBLE_DIMINISHED -> if (perfect) -2 else -3
100 }
101 fun inverse(): Quality = when (this) {
102 PERFECT -> PERFECT
103 MAJOR -> MINOR
104 MINOR -> MAJOR
105 AUGMENTED -> DIMINISHED
106 DIMINISHED -> AUGMENTED
107 DOUBLE_AUGMENTED -> DOUBLE_DIMINISHED
108 DOUBLE_DIMINISHED -> DOUBLE_AUGMENTED
109 }
110}
111private fun perfectType(number: Int) = (number - 1) % 7 in listOf(0, 3, 4)
112private data class Tone(val letters: Int, val semitones: Int)
113private data class Interval(val quality: Quality, val number: Int) {
114 val forbidden: Boolean get() = number == 1 && quality in listOf(Quality.DIMINISHED, Quality.DOUBLE_DIMINISHED)
115 val tone: Tone get() {
116 val distance = number - 1
117 return Tone(distance, 12 * (distance / 7) + naturals[distance % 7] + quality.delta(perfectType(number))!!)
118 }
119 fun simple(): Interval {
120 var n = number
121 while (n > 8) n -= 7
122 return copy(number = n)
123 }
124 fun consonance(): String {
125 val simple = simple()
126 if (quality == Quality.PERFECT && simple.number in listOf(1, 5, 8)) return "perfect"
127 if (quality in listOf(Quality.MAJOR, Quality.MINOR) && simple.number in listOf(3, 6)) return "imperfect"
128 return "dissonant"
129 }
130 override fun toString() = "${quality.text}$number"
131}
132private fun parseInterval(text: String): Interval {
133 for (quality in Quality.entries) {
134 if (!text.startsWith(quality.text)) continue
135 val number = canonical(text.substring(quality.text.length), 1..99) ?: continue
136 if (quality.delta(perfectType(number)) == null) continue
137 val result = Interval(quality, number)
138 if (!result.forbidden) return result
139 }
140 bad("INTERVAL", text)
141}
142private data class Motion(val interval: Interval, val down: Boolean, val semitones: Int)
143private fun between(first: Note, second: Note): Motion {
144 var distance = second.position - first.position
145 var semitones = second.midi - first.midi
146 val down = distance < 0 || distance == 0 && semitones < 0
147 if (down) { distance = -distance; semitones = -semitones }
148 val difference = semitones - 12 * (distance / 7) - naturals[distance % 7]
149 for (quality in Quality.entries) {
150 if (quality.delta(perfectType(distance + 1)) == difference) return Motion(Interval(quality, distance + 1), down, semitones)
151 }
152 throw CommandError("UNNAMEABLE INTERVAL")
153}

201private fun build(root: Pitch, tones: List<Tone>) = tones.map(root::raise)
202private fun symbol(root: Pitch, quality: Int, bass: Pitch? = null) =

204private fun roman(degree: Int, quality: Int): String {

208private data class Key(val tonic: Pitch, val mode: Int, val fifths: Int, val scale: List<Pitch>) {
209 fun degree(pitch: Pitch) = mod(pitch.letter - tonic.letter, 7)
210 fun prefix(pitch: Pitch): String {
211 val difference = pitch.alteration - scale[degree(pitch)].alteration
212 return if (difference < 0) "b".repeat(-difference) else "#".repeat(difference)
213 }
214 fun label(pitch: Pitch) = "${prefix(pitch)}${degree(pitch) + 1}"
215 fun pitchSet(): Int = scale.fold(0) { mask, pitch -> mask or (1 shl pitch.pitchClass) }
216 override fun toString() = "$tonic ${modes[mode].name}"
217}
218private fun makeKey(tonic: Pitch, mode: Int): Key {
219 val fifths = fifthsValues[tonic.letter] + 7 * tonic.alteration + modes[mode].offset
220 if (fifths !in -7..7) throw CommandError("THEORETICAL KEY")
221 return Key(tonic, mode, fifths, build(tonic, scales[modes[mode].scale].tones))
222}
223private fun parseKey(words: List<String>, start: Int = 0): Key {
224 val tonic = parsePitch(words[start])
225 val mode = parseMode(words[start + 1])
226 return makeKey(tonic, mode)
227}
228private data class Numeral(val alteration: Int, val degree: Int, val quality: Int)
229private data class ChordSymbol(val root: Pitch, val quality: Int)
230private fun parseNumeral(text: String): Numeral {
231 val start = if (text[0] == 'b' || text[0] == '#') 1 else 0
232 val alteration = if (start == 0) 0 else if (text[0] == 'b') -1 else 1
233 var end = start
234 while (end < text.length && text[end] in "IViv") end++
235 val part = text.substring(start, end)
236 val suffix = text.substring(end)
237 for (degree in 0..6) {
238 for ((quality, chord) in chords.withIndex()) {
239 val expected = if (chord.lower) romans[degree].lowercase() else romans[degree]
240 if (part == expected && suffix == chord.numeralSuffix) return Numeral(alteration, degree, quality)
241 }
242 }
243 bad("NUMERAL", text)
244}
245private fun parseSymbol(text: String): ChordSymbol {
246 val prefix = pitchPrefix(text)
247 if (prefix != null) {
248 val quality = chords.indexOfFirst { it.suffix == text.substring(prefix.end) }
249 if (quality >= 0) return ChordSymbol(prefix.pitch, quality)
250 }
251 bad("SYMBOL", text)
252}

254private fun chordCommand(command: String, args: List<String>): String {
255 val input = if (command == "VOICE") parseNote(args[0]) else null
256 val root = input?.pitch ?: parsePitch(args[0])
257 val quality = parseQuality(args[1])
258 val chord = chords[quality]
259 val inversion = if (args.size == 3) canonical(args[2], chord.tones.indices) ?: bad("INVERSION", args[2]) else 0
260 val tones = build(root, chord.tones)
261 val ordered = tones.drop(inversion) + tones.take(inversion)
262 val name = symbol(root, quality, if (inversion == 0) null else ordered.first())
263 if (input == null) return "CHORD $name ${ordered.joinToString(" ")}"
264 var value = input.midi
265 val notes = ordered.mapIndexed { i, pitch ->
266 if (i != 0 || inversion != 0) {
267 val difference = mod(pitch.pitchClass - value, 12)
268 value += if (difference == 0) 12 else difference
269 }
270 fromPitch(pitch, value).also { if (!it.valid) throw CommandError("OUT OF RANGE") }
271 }
272 return "VOICE $name ${notes.joinToString(" ")} midi=${notes.joinToString(",") { it.midi.toString() }}"
273}
274private fun identify(notes: List<Note>): String {
275 val candidates = notes.sortedBy { it.midi }.distinctBy { it.pitch.pitchClass }
276 val bass = candidates.first().pitch
277 val set = candidates.map { it.pitch.pitchClass }.toSet()
278 for (root in candidates) {
279 for ((quality, chord) in chords.withIndex()) {
280 if (chord.tones.map { mod(root.midi + it.semitones, 12) }.toSet() != set) continue
281 val inversion = chord.tones.indexOfFirst { mod(root.midi + it.semitones, 12) == bass.pitchClass }
282 return "IDENTIFY ${symbol(root.pitch, quality, if (inversion == 0) null else bass)} root=${root.pitch} quality=${chord.name} inversion=$inversion"
283 }
284 }
285 return "IDENTIFY UNKNOWN"
286}
287private fun melody(notes: List<Note>): String {
288 var low = notes.first()
289 var high = low
290 for (note in notes) {
291 if (note.midi < low.midi) low = note
292 if (note.midi > high.midi) high = note
293 }
294 val range = between(low, high).interval
295 var leaps = 0
296 val steps = (1 until notes.size).map { i ->
297 val motion = between(notes[i - 1], notes[i])
298 if (abs(notes[i].midi - notes[i - 1].midi) >= 3) leaps++
299 "${if (motion.down) "-" else "+"}${motion.interval}"
300 }
301 return "MELODY n=${notes.size} low=$low high=$high range=$range leaps=$leaps steps=${steps.joinToString(",")}"
302}
303private fun cadence(numerals: List<Numeral>): String {
304 if (numerals.size < 2) return "none"
305 val x = numerals[numerals.size - 2]
306 val y = numerals.last()
307 if (x.alteration == 0 && y.alteration == 0) {
308 if (x.degree == 4 && y.degree == 0) return "authentic"
309 if (x.degree == 3 && y.degree == 0) return "plagal"
310 if (x.degree == 4 && y.degree == 5) return "deceptive"
311 }
312 return if (y.alteration == 0 && y.degree == 4) "half" else "none"
313}
314private fun diatonic(key: Key): String {
315 val rows = mutableListOf("DIATONIC $key")
316 for (size in 3..4) {
317 val items = key.scale.mapIndexed { i, root ->
318 val third = mod(key.scale[(i + 2) % 7].pitchClass - root.pitchClass, 12)
319 val fifth = mod(key.scale[(i + 4) % 7].pitchClass - root.pitchClass, 12)
320 val quality = if (size == 3) {
321 if (third == 4) 0 else if (fifth == 7) 1 else 2
322 } else {
323 val seventh = mod(key.scale[(i + 6) % 7].pitchClass - root.pitchClass, 12)
324 if (third == 4) (if (seventh == 11) 9 else 8) else if (fifth == 7) 10 else 12
325 }
326 "${roman(i, quality)}=${symbol(root, quality)}"
327 }
328 rows += (if (size == 3) "TRIADS " else "SEVENTHS ") + items.joinToString(" ")
329 }
330 return rows.joinToString("\n")
331}

333private fun keyCommand(command: String, args: List<String>): String {
334 val key = parseKey(args)
335 return when (command) {
336 "KEYSIG" -> {
337 val p = key.fifths
338 if (p == 0) "KEYSIG $key none"
339 else {
340 val names = (if (p > 0) "F# C# G# D# A# E# B#" else "Bb Eb Ab Db Gb Cb Fb").split(' ')
341 "KEYSIG $key ${if (p > 0) "sharps" else "flats"}=${abs(p)} ${names.take(abs(p)).joinToString(" ")}"
342 }
343 }
344 "RELATIVE" -> "RELATIVE $key major=${fifthPitch(key.fifths)} minor=${fifthPitch(key.fifths + 3)}"
345 "DEGREE" -> {
346 val pitch = parsePitch(args[2])
347 "DEGREE $key ${args[2]} ${key.label(pitch)}"
348 }
349 "KEYDIST" -> {
350 val other = parseKey(args, 2)
351 var distance = mod(other.fifths - key.fifths, 12)
352 if (distance > 6) distance -= 12
353 val sameTonic = key.tonic.pitchClass == other.tonic.pitchClass
354 val relation = when {
355 key.pitchSet() == other.pitchSet() -> if (sameTonic) "same" else "relative"
356 sameTonic -> "parallel"
357 abs(distance) == 1 -> "close"
358 else -> "distant"
359 }
360 val common = Integer.bitCount(key.pitchSet() and other.pitchSet())
361 "KEYDIST $key $other fifths=${if (distance > 0) "+" else ""}$distance common=$common relation=$relation"
362 }
363 "DIATONIC" -> diatonic(key)
364 "ANALYZE" -> {
365 val symbols = args.drop(2).map(::parseSymbol)
366 val items = symbols.map { key.prefix(it.root) + roman(key.degree(it.root), it.quality) }
367 "ANALYZE $key ${items.joinToString(" ")}"
368 }
369 "PROGRESSION" -> {
370 val numerals = args.drop(2).map(::parseNumeral)
371 val rows = mutableListOf("PROGRESSION $key ${numerals.size}")
372 for ((i, numeral) in numerals.withIndex()) {
373 val degree = key.scale[numeral.degree]
374 val root = degree.copy(alteration = degree.alteration + numeral.alteration)
375 val tones = build(root, chords[numeral.quality].tones)
376 rows += "${args[i + 2]} ${symbol(root, numeral.quality)} ${tones.joinToString(" ")}"
377 }
378 rows += "CADENCE ${cadence(numerals)}"
379 rows.joinToString("\n")
380 }
381 else -> throw CommandError("UNKNOWN COMMAND $command")
382 }
383}

385private class Calculator {
386 private var tuning = 44000
387
388 fun execute(line: String): String {
389 val words = line.split(' ')
390 val command = words[0]
391 val args = words.drop(1)
392 val arity = when (command) {
393 "MIDI", "NOTE", "FREQ", "TUNING", "PITCH", "INVERT", "ENHARMONIC", "CIRCLE" -> 1..1
394 "INTERVAL", "SCALE", "KEYSIG", "RELATIVE", "DIATONIC" -> 2..2
395 "TRANSPOSE", "DEGREE", "SPELL" -> 3..3
396 "KEYDIST" -> 4..4
397 "CHORD", "VOICE" -> 2..3
398 "IDENTIFY", "MELODY" -> 2..15
399 "PROGRESSION", "ANALYZE" -> 3..15
400 "FINDKEY" -> 1..15
401 else -> throw CommandError("UNKNOWN COMMAND $command")
402 }
403 if (args.size !in arity) throw CommandError("ARGUMENT COUNT")
404 return when (command) {
405 "MIDI", "FREQ", "ENHARMONIC" -> {
406 val note = parseNote(args[0])
407 val value = note.midi
408 when (command) {
409 "MIDI" -> "MIDI ${args[0]} $value"
410 "FREQ" -> {
411 val cents = floor(tuning * 2.0.pow((value - 69) / 12.0) + 0.5).toInt()
412 "FREQ ${args[0]} ${frequencyText(cents)}"
413 }
414 else -> {
415 val equivalents = buildList {
416 for (octave in -1..9) {
417 for (letter in 0..6) {
418 val alteration = value - 12 * (octave + 1) - naturals[letter]
419 if (alteration in -2..2) add(Note(Pitch(letter, alteration), octave))
420 }
421 }
422 }
423 "ENHARMONIC ${args[0]} ${equivalents.joinToString(" ")}"
424 }
425 }
426 }
427 "NOTE" -> {
428 val value = parseNumber(args[0])
429 "NOTE $value ${spelling(value, false)} ${spelling(value, true)}"
430 }
431 "TUNING", "PITCH" -> {
432 val value = frequencyCents(args[0], command == "TUNING")
433 if (command == "TUNING") {
434 tuning = value
435 "TUNING ${frequencyText(value)}"
436 } else {
437 val x = 69 + 12 * log2(value.toDouble() / tuning)
438 val nearest = floor(x + 0.5).toInt()
439 if (nearest !in 0..127) throw CommandError("OUT OF RANGE")
440 val cents = floor(100 * (x - nearest) + 0.5).toInt()
441 "PITCH ${frequencyText(value)} ${spelling(nearest, false)} ${if (cents >= 0) "+" else ""}$cents"
442 }
443 }
444 "INTERVAL" -> {
445 val first = parseNote(args[0])
446 val second = parseNote(args[1])
447 val motion = between(first, second)
448 "INTERVAL ${args[0]} ${args[1]} ${motion.interval} ${if (motion.down) "down" else "up"} semitones=${motion.semitones} simple=${motion.interval.simple()} class=${motion.interval.consonance()}"
449 }
450 "INVERT" -> {
451 val simple = parseInterval(args[0]).simple()
452 val inverse = Interval(simple.quality.inverse(), 9 - simple.number)
453 if (inverse.forbidden) throw CommandError("CANNOT INVERT")
454 "INVERT ${args[0]} $simple $inverse"
455 }
456 "TRANSPOSE" -> {
457 val note = parseNote(args[0])
458 val interval = parseInterval(args[1])
459 val direction = args[2]
460 if (direction != "UP" && direction != "DOWN") bad("DIRECTION", direction)
461 val sign = if (direction == "UP") 1 else -1
462 val tone = interval.tone
463 val position = note.position + sign * tone.letters
464 val value = note.midi + sign * tone.semitones
465 val letter = mod(position, 7)
466 val octave = Math.floorDiv(position, 7)
467 val alteration = value - 12 * (octave + 1) - naturals[letter]
468 if (alteration !in -2..2) throw CommandError("CANNOT SPELL")
469 val result = Note(Pitch(letter, alteration), octave)
470 if (!result.valid) throw CommandError("OUT OF RANGE")
471 "TRANSPOSE ${args[0]} ${args[1]} $direction $result"
472 }
473 "SCALE" -> {
474 val root = parsePitch(args[0])
475 val scale = parseScale(args[1])
476 val pitches = build(root, scale.tones)
477 val steps = scale.tones.mapIndexed { i, tone -> (scale.tones.getOrNull(i + 1)?.semitones ?: 12) - tone.semitones }
478 "SCALE ${args[0]} ${args[1]} ${pitches.joinToString(" ")} steps=${steps.joinToString(",")}"
479 }
480 "CHORD", "VOICE" -> chordCommand(command, args)
481 "IDENTIFY" -> identify(args.map(::parseNote))
482 "MELODY" -> melody(args.map(::parseNote))
483 "CIRCLE" -> {
484 val mode = parseMode(args[0])
485 val pitches = (-7..7).map { fifthPitch(it - modes[mode].offset) }
486 "CIRCLE ${args[0]} ${pitches.joinToString(" ")}"
487 }
488 "FINDKEY" -> {
489 val pitches = args.map(::parsePitch)
490 val keys = buildList {
491 for (p in -7..7) {
492 for (mode in 0..1) {
493 val key = makeKey(fifthPitch(p - modes[mode].offset), mode)
494 if (key.scale.containsAll(pitches)) add(key)
495 }
496 }
497 }
498 "FINDKEY ${keys.size}${if (keys.isEmpty()) "" else " " + keys.joinToString(", ")}"
499 }
500 "SPELL" -> {
501 val value = parseNumber(args[0])
502 val key = parseKey(args, 1)
503 val pitch = key.scale.find { it.pitchClass == value % 12 } ?: spelling(value, key.fifths < 0).pitch
504 val note = fromPitch(pitch, value)
505 if (!note.valid) throw CommandError("OUT OF RANGE")
506 "SPELL $value $key $note ${key.label(pitch)}"
507 }
508 else -> keyCommand(command, args)
509 }
510 }
511}

513fun main() {
514 val input = System.`in`.bufferedReader()
515 val output = System.out.bufferedWriter()
516 val count = input.readLine().toInt()
517 val calculator = Calculator()
518 repeat(count) {
519 val response = try { calculator.execute(input.readLine()) }
520 catch (error: CommandError) { error.message!! }
521 output.append(response).append('\n')
522 }
523 output.flush()
524}
