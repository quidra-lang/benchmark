77class Pitch:
78 def __init__(self, letter, alteration):
79 self.letter = letter
80 self.alteration = alteration
81
82 def name(self):
83 return LETTERS[self.letter] + ACCIDENTALS[self.alteration + 2]
84
85 def pitch_class(self):
86 return (NATURALS[self.letter] + self.alteration) % 12
87
88 def altered(self, change):
89 return Pitch(self.letter, self.alteration + change)
90
91 def raised(self, interval):
92 total = self.letter + interval.distance()
93 letter = total % 7
94 alteration = (
95 NATURALS[self.letter] + self.alteration + interval.semitones()
96 - NATURALS[letter] - 12 * (total // 7)
97 )
98 if not -2 <= alteration <= 2:
99 return None
100 return Pitch(letter, alteration)

103class Note:
104 def __init__(self, pitch, octave):
105 self.pitch = pitch
106 self.octave = octave
107
108 def name(self):
109 return self.pitch.name() + str(self.octave)
110
111 def midi(self):
112 return 12 * (self.octave + 1) + NATURALS[self.pitch.letter] + self.pitch.alteration
113
114 def position(self):
115 return 7 * self.octave + self.pitch.letter
116
117 def valid(self):
118 return -1 <= self.octave <= 9 and 0 <= self.midi() <= 127

121class Interval:
122 def __init__(self, quality, number):
123 self.quality = quality
124 self.number = number
125
126 def name(self):
127 return self.quality + str(self.number)
128
129 def distance(self):
130 return self.number - 1
131
132 def semitones(self):
133 return base_semitones(self.distance()) + quality_offsets(self.number)[self.quality]
134
135 def simple(self):
136 number = self.number
137 while number > 8:
138 number -= 7
139 return Interval(self.quality, number)
140
141 def consonance(self):
142 name = self.simple().name()
143 if name in PERFECT_INTERVALS:
144 return "perfect"
145 if name in IMPERFECT_INTERVALS:
146 return "imperfect"
147 return "dissonant"

150class Key:
151 def __init__(self, tonic, mode):
152 self.tonic = tonic
153 self.mode = mode
154
155 def name(self):
156 return self.tonic.name() + " " + self.mode
157
158 def fifths(self):
159 offset = MODES[self.mode][1]
160 return FIFTHS_VALUES[self.tonic.letter] + 7 * self.tonic.alteration + offset
161
162 def theoretical(self):
163 return not -7 <= self.fifths() <= 7
164
165 def scale(self):
166 return build_pitches(self.tonic, SCALES[MODES[self.mode][0]])

168 def pitch_classes(self):
169 return {pitch.pitch_class() for pitch in self.scale()}

171 def degree(self, pitch):
172 index = (pitch.letter - self.tonic.letter) % 7
173 difference = pitch.alteration - self.scale()[index].alteration
174 if difference < 0:
175 marks = "b" * -difference
176 else:
177 marks = "#" * difference
178 return marks, index + 1

180 def degree_label(self, pitch):
181 marks, degree = self.degree(pitch)
182 return marks + str(degree)

196def base_semitones(distance):
197 return 12 * (distance // 7) + NATURALS[distance % 7]

200def quality_offsets(number):
201 if (number - 1) % 7 in (0, 3, 4):
202 return PERFECT_OFFSETS
203 return MAJOR_OFFSETS

206def named_interval(distance, semitones):
207 difference = semitones - base_semitones(distance)
208 for quality, offset in quality_offsets(distance + 1).items():
209 if offset == difference:
210 return Interval(quality, distance + 1)
211 return None

214def interval_between(first, second):
215 distance = second.position() - first.position()
216 semitones = second.midi() - first.midi()
217 direction = "up"
218 if distance < 0 or (distance == 0 and semitones < 0):
219 direction = "down"
220 distance, semitones = -distance, -semitones
221 interval = named_interval(distance, semitones)
222 if interval is None:
223 raise CommandError("UNNAMEABLE INTERVAL")
224 return interval, direction

239def parse_note(text):
240 if text.endswith("-1"):
241 pitch = parse_pitch(text[:-2])
242 octave = -1
243 elif text[-1:].isdigit():
244 pitch = parse_pitch(text[:-1])
245 octave = int(text[-1])
246 else:
247 return None
248 if pitch is None:
249 return None
250 note = Note(pitch, octave)
251 return note if note.valid() else None

254def parse_frequency(text):
255 whole, dot, fraction = text.partition(".")
256 if not whole.isdigit() or len(whole) > 5:
257 return None
258 if dot and (not fraction.isdigit() or len(fraction) > 2):
259 return None
260 return 100 * int(whole) + int(fraction.ljust(2, "0"))

263def parse_interval(text):
264 split = 0
265 while split < len(text) and not text[split].isdigit():
266 split += 1
267 quality = text[:split]
268 number = decimal_integer(text[split:])
269 if number is None or not 1 <= number <= 99:
270 return None
271 if quality not in quality_offsets(number):
272 return None
273 if number == 1 and quality in ("d", "dd"):
274 return None
275 return Interval(quality, number)

278def parse_numeral(text):
279 shift = 0
280 if text[0] == "b":
281 shift = -1
282 elif text[0] == "#":
283 shift = 1
284 start = 0 if shift == 0 else 1
285 end = start
286 while end < len(text) and text[end] in "IViv":
287 end += 1
288 numeral = text[start:end]
289 suffix = text[end:]
290 if numeral in ROMAN_NUMERALS:
291 uppercase = True
292 elif numeral.islower() and numeral.upper() in ROMAN_NUMERALS:
293 uppercase = False
294 else:
295 return None
296 degree = ROMAN_NUMERALS.index(numeral.upper()) + 1
297 for quality, (_, _, example) in CHORDS.items():
298 if (example[0] == "I") == uppercase and example[1:] == suffix:
299 return Numeral(text, shift, degree, quality)
300 return None

470def cadence(numerals):
471 if len(numerals) == 1:
472 return "none"
473 first, last = numerals[-2], numerals[-1]
474 if first.is_plain(5) and last.is_plain(1):
475 return "authentic"
476 if first.is_plain(4) and last.is_plain(1):
477 return "plagal"

479 return "deceptive"
480 if last.is_plain(5):
481 return "half"
482 return "none"

528 note = note_argument(words[0])

532 def tuning(self, words):
533 value = parse_frequency(words[0])
534 if value is None or not 40000 <= value <= 48000:
535 raise CommandError("BAD FREQUENCY " + words[0])
536 self.reference_hundredths = value

539 def nearest_pitch(self, words):
540 value = parse_frequency(words[0])
541 if value is None or value <= 0:
542 raise CommandError("BAD FREQUENCY " + words[0])
543 exact = 69 + 12 * math.log2(value / self.reference_hundredths)
544 nearest = round(exact)
545 if not 0 <= nearest <= 127:
546 raise CommandError("OUT OF RANGE")
547 cents = round(100 * (exact - nearest))
548 self.emit("PITCH", hertz_text(value), spelling(nearest, SHARP_NAMES), f"{cents:+d}")

568 def transpose(self, words):
569 note = note_argument(words[0])
570 interval = interval_argument(words[1])
571 direction = direction_argument(words[2])
572 sign = 1 if direction == "UP" else -1
573 position = note.position() + sign * interval.distance()
574 midi = note.midi() + sign * interval.semitones()
575 letter = position % 7
576 octave = position // 7
577 alteration = midi - (12 * (octave + 1) + NATURALS[letter])
578 if not -2 <= alteration <= 2:
579 raise CommandError("CANNOT SPELL")
580 result = Note(Pitch(letter, alteration), octave)
581 if not result.valid():
582 raise CommandError("OUT OF RANGE")
583 self.emit("TRANSPOSE", note.name(), interval.name(), direction, result.name())

585 def scale(self, words):
586 root = pitch_argument(words[0])
587 name = scale_argument(words[1])
588 pitches = build_pitches(root, SCALES[name])
589 semitones = semitone_counts(SCALES[name]) + [12]
590 steps = [str(upper - lower) for lower, upper in zip(semitones, semitones[1:])]
591 self.emit("SCALE", root.name(), name, *pitch_names(pitches), "steps=" + ",".join(steps))

593 def chord(self, words):
594 root = pitch_argument(words[0])
595 quality = quality_argument(words[1])
596 inversion = inversion_argument(words, 2, quality)
597 tones = chord_tones(root, quality, inversion)

601 def voice(self, words):
602 given = note_argument(words[0])
603 quality = quality_argument(words[1])
604 inversion = inversion_argument(words, 2, quality)
605 tones = chord_tones(given.pitch, quality, inversion)
606 notes = []
607 previous = given.midi()
608 for index, pitch in enumerate(tones):
609 if index == 0 and inversion == 0:
610 current = given
611 else:
612 current = lowest_note_above(pitch, previous)
613 notes.append(current)
614 previous = current.midi()
615 if not all(note.valid() for note in notes):
616 raise CommandError("OUT OF RANGE")
617 symbol = inverted_symbol(given.pitch, quality, tones, inversion)
618 midis = ",".join(str(note.midi()) for note in notes)
619 self.emit("VOICE", symbol, *(note.name() for note in notes), "midi=" + midis)

621 def identify(self, words):
622 notes = [note_argument(word) for word in words]
623 classes = {note.midi() % 12 for note in notes}
624 candidates = []
625 for note in sorted(notes, key=Note.midi):
626 if all(other.midi() % 12 != note.midi() % 12 for other in candidates):
627 candidates.append(note)
628 bass = candidates[0]
629 for root in candidates:
630 for quality, (intervals, suffix, _) in CHORDS.items():
631 tones = [(root.midi() + count) % 12 for count in semitone_counts(intervals)]
632 if set(tones) != classes:
633 continue
634 inversion = tones.index(bass.midi() % 12)
635 symbol = root.pitch.name() + suffix
636 if inversion > 0:
637 symbol += "/" + bass.pitch.name()
638 self.emit(
639 "IDENTIFY", symbol, "root=" + root.pitch.name(),
640 "quality=" + quality, f"inversion={inversion}",
641 )
642 return
643 self.emit("IDENTIFY", "UNKNOWN")

645 def melody(self, words):
646 notes = [note_argument(word) for word in words]
647 low = min(notes, key=Note.midi)
648 high = max(notes, key=Note.midi)
649 span = interval_between(low, high)[0]
650 steps = []
651 leaps = 0
652 for first, second in zip(notes, notes[1:]):
653 interval, direction = interval_between(first, second)
654 steps.append(("+" if direction == "up" else "-") + interval.name())
655 if abs(second.midi() - first.midi()) >= 3:
656 leaps += 1
657 self.emit(
658 "MELODY", f"n={len(notes)}", "low=" + low.name(), "high=" + high.name(),
659 "range=" + span.name(), f"leaps={leaps}", "steps=" + ",".join(steps),
660 )

662 def enharmonic(self, words):
663 note = note_argument(words[0])
664 midi = note.midi()
665 found = []
666 for letter in range(7):
667 for alteration in range(-2, 3):
668 pitch = Pitch(letter, alteration)
669 if pitch.pitch_class() != midi % 12:
670 continue
671 candidate = note_from_midi(pitch, midi)
672 if candidate.valid():
673 found.append(candidate)
674 found.sort(key=Note.position)
675 self.emit("ENHARMONIC", note.name(), *(candidate.name() for candidate in found))

699 def diatonic(self, words):
700 key = key_argument(words, 0)
701 scale = key.scale()
702 triads = []
703 sevenths = []
704 for index, root in enumerate(scale):
705 gaps = tuple(
706 (scale[(index + step) % 7].pitch_class() - root.pitch_class()) % 12
707 for step in (2, 4, 6)
708 )
709 triads.append(diatonic_item(index + 1, root, TRIAD_QUALITIES[gaps[:2]]))
710 sevenths.append(diatonic_item(index + 1, root, SEVENTH_QUALITIES[gaps]))
711 self.emit("DIATONIC", key.name())
712 self.emit("TRIADS", *triads)
713 self.emit("SEVENTHS", *sevenths)

715 def progression(self, words):
716 key = key_argument(words, 0)
717 numerals = [numeral_argument(word) for word in words[2:]]
718 scale = key.scale()
719 chords = []
720 for numeral in numerals:
721 root = scale[numeral.degree - 1].altered(numeral.shift)
722 chords.append((root, chord_tones(root, numeral.quality)))
723 self.emit("PROGRESSION", key.name(), len(numerals))
724 for numeral, (root, tones) in zip(numerals, chords):
725 self.emit(numeral.text, chord_symbol(root, numeral.quality), *pitch_names(tones))
726 self.emit("CADENCE", cadence(numerals))

732 for root, quality in symbols:
733 marks, degree = key.degree(root)

737 def find_keys(self, words):
738 wanted = [pitch_argument(word).name() for word in words]

740 for fifths in range(-7, 8):
741 for mode in ("major", "minor"):

756 def spell(self, words):
757 number = number_argument(words[0])
758 key = key_argument(words, 1)
759 chosen = None
760 for pitch in key.scale():
761 if pitch.pitch_class() == number % 12:
762 chosen = pitch
763 if chosen is None:
764 names = SHARP_NAMES if key.fifths() >= 0 else FLAT_NAMES
765 chosen = parse_pitch(names[number % 12])
766 note = note_from_midi(chosen, number)
767 if not note.valid():
768 raise CommandError("OUT OF RANGE")
769 self.emit("SPELL", number, key.name(), note.name(), key.degree_label(chosen))

771 def key_distance(self, words):
772 first = key_argument(words, 0)
773 second = key_argument(words, 2)
774 shift = (second.fifths() - first.fifths()) % 12
775 if shift > 6:
776 shift -= 12
777 first_classes = first.pitch_classes()
778 second_classes = second.pitch_classes()
779 common = len(first_classes & second_classes)
780 same_set = first_classes == second_classes
781 same_tonic = first.tonic.pitch_class() == second.tonic.pitch_class()
782 if same_set and same_tonic:
783 relation = "same"
784 elif same_set:
785 relation = "relative"
786 elif same_tonic:
787 relation = "parallel"
788 elif shift in (1, -1):
789 relation = "close"
790 else:
791 relation = "distant"
792 fifths = f"{shift:+d}" if shift != 0 else "0"
793 self.emit(
794 "KEYDIST", first.name(), second.name(), "fifths=" + fifths,
795 f"common={common}", "relation=" + relation,
796 )

798 def execute(self, line):
799 command, *arguments = line.split(" ")
800 try:
801 if command not in self.handlers:
802 raise CommandError("UNKNOWN COMMAND " + command)
803 minimum, maximum, handler = self.handlers[command]
804 if len(arguments) < minimum or (maximum is not None and len(arguments) > maximum):
805 raise CommandError("ARGUMENT COUNT")
806 handler(arguments)
807 except CommandError as error:
808 self.emit("ERROR", error)
