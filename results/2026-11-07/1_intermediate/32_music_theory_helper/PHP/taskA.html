8function modulo(int $value, int $base): int { return ($value % $base + $base) % $base; }
9function floorDivide(int $value, int $base): int { return intdiv($value - modulo($value, $base), $base); }
10function frequencyText(int $cents): string { return intdiv($cents, 100) . '.' . str_pad((string)($cents % 100), 2, '0', STR_PAD_LEFT); }
11function signedText(int $value, bool $positiveZero = false): string { return ($value > 0 || ($value === 0 && $positiveZero) ? '+' : '') . $value; }

13final readonly class Pitch {
14 public const LETTERS = 'CDEFGAB';
15 public const NATURALS = [0, 2, 4, 5, 7, 9, 11];
16 private const ACCIDENTALS = ['bb', 'b', '', '#', 'x'];
17 public function __construct(public int $letter, public int $alteration = 0) {}
18 public function natural(): int { return self::NATURALS[$this->letter]; }
19 public function pitchClass(): int { return modulo($this->natural() + $this->alteration, 12); }
20 public function same(self $other): bool { return $this->letter === $other->letter && $this->alteration === $other->alteration; }
21 public function __toString(): string { return self::LETTERS[$this->letter] . self::ACCIDENTALS[$this->alteration + 2]; }
22 public static function prefix(string $text): ?array {
23 if ($text === '' || ($letter = strpos(self::LETTERS, $text[0])) === false) return null;
24 $end = 1; $alteration = 0;
25 if (str_starts_with(substr($text, 1), 'bb')) { $end = 3; $alteration = -2; }
26 elseif (strlen($text) > 1) {
27 [$end, $alteration] = match ($text[1]) {
28 'b' => [2, -1], '#' => [2, 1], 'x' => [2, 2], default => [1, 0],
29 };
30 }
31 return [new self($letter, $alteration), $end];
32 }
33 public static function parse(string $text): self {
34 $prefix = self::prefix($text);
35 if ($prefix === null || $prefix[1] !== strlen($text)) throw new CommandError("BAD PITCH $text");
36 return $prefix[0];
37 }
38 public static function atFifths(int $fifths): self {
39 return new self(strpos(self::LETTERS, 'FCGDAEB'[modulo($fifths + 1, 7)]), floorDivide($fifths + 1, 7));
40 }
41 public function raise(Interval $interval): self {
42 $position = $this->letter + $interval->distance();
43 $letter = $position % 7;
44 $alteration = $this->natural() + $this->alteration + $interval->semitones()
45 - self::NATURALS[$letter] - 12 * intdiv($position, 7);
46 if ($alteration < -2 || $alteration > 2) throw new CommandError('CANNOT SPELL');
47 return new self($letter, $alteration);
48 }
49}

51final readonly class Note {
52 private const SHARPS = ['C', 'C#', 'D', 'D#', 'E', 'F', 'F#', 'G', 'G#', 'A', 'A#', 'B'];
53 private const FLATS = ['C', 'Db', 'D', 'Eb', 'E', 'F', 'Gb', 'G', 'Ab', 'A', 'Bb', 'B'];
54 public function __construct(public Pitch $pitch, public int $octave) {}
55 public function midi(): int { return 12 * ($this->octave + 1) + $this->pitch->natural() + $this->pitch->alteration; }
56 public function position(): int { return 7 * $this->octave + $this->pitch->letter; }
57 public function valid(): bool { return $this->octave >= -1 && $this->octave <= 9 && $this->midi() >= 0 && $this->midi() <= 127; }
58 public function __toString(): string { return $this->pitch . $this->octave; }
59 public static function parse(string $text): self {
60 $prefix = Pitch::prefix($text);
61 if ($prefix === null) throw new CommandError("BAD NOTE $text");
62 $tail = substr($text, $prefix[1]);
63 if (!preg_match('/\A(?:-1|[0-9])\z/', $tail)) throw new CommandError("BAD NOTE $text");
64 $note = new self($prefix[0], (int)$tail);
65 if (!$note->valid()) throw new CommandError("BAD NOTE $text");
66 return $note;
67 }
68 public static function fromMidi(Pitch $pitch, int $midi): self {
69 return new self($pitch, intdiv($midi - $pitch->natural() - $pitch->alteration, 12) - 1);
70 }
71 public static function spelling(int $midi, bool $flat = false): self {
72 $names = $flat ? self::FLATS : self::SHARPS;
73 return new self(Pitch::parse($names[modulo($midi, 12)]), floorDivide($midi, 12) - 1);
74 }
75}

77enum IntervalQuality: string {
78 case Perfect = 'P'; case Major = 'M'; case Minor = 'm'; case Augmented = 'A';
79 case Diminished = 'd'; case DoubleAugmented = 'AA'; case DoubleDiminished = 'dd';
80 public function inverse(): self {
81 return match ($this) {
82 self::Perfect => self::Perfect, self::Major => self::Minor, self::Minor => self::Major,
83 self::Augmented => self::Diminished, self::Diminished => self::Augmented,
84 self::DoubleAugmented => self::DoubleDiminished, self::DoubleDiminished => self::DoubleAugmented,
85 };
86 }

88final readonly class Interval {
89 public function __construct(public IntervalQuality $quality, public int $number) {}
90 public function distance(): int { return $this->number - 1; }
91 public function perfectType(): bool { return in_array($this->distance() % 7, [0, 3, 4], true); }
92 public function semitones(): int {
93 $difference = match ($this->quality) {
94 IntervalQuality::DoubleDiminished => $this->perfectType() ? -2 : -3,
95 IntervalQuality::Diminished => $this->perfectType() ? -1 : -2,
96 IntervalQuality::Minor => -1, IntervalQuality::Augmented => 1,
97 IntervalQuality::DoubleAugmented => 2, default => 0,
98 };
99 return 12 * intdiv($this->distance(), 7) + Pitch::NATURALS[$this->distance() % 7] + $difference;
100 }
101 public function simple(): self {
102 $number = $this->number;
103 while ($number > 8) $number -= 7;
104 return new self($this->quality, $number);
105 }
106 public function consonance(): string {
107 $simple = $this->simple();
108 if ($simple->quality === IntervalQuality::Perfect && in_array($simple->number, [1, 5, 8], true)) return 'perfect';
109 if (in_array($simple->quality, [IntervalQuality::Minor, IntervalQuality::Major], true)
110 && in_array($simple->number, [3, 6], true)) return 'imperfect';
111 return 'dissonant';
112 }
113 public function __toString(): string { return $this->quality->value . $this->number; }
114 public static function parse(string $text): self {
115 if (!preg_match('/\A(P|M|m|A|d|AA|dd)([1-9][0-9]?)\z/', $text, $parts)) throw new CommandError("BAD INTERVAL $text");
116 $quality = IntervalQuality::from($parts[1]);
117 $result = new self($quality, (int)$parts[2]);
118 $invalid = $result->perfectType()
119 ? in_array($quality, [IntervalQuality::Major, IntervalQuality::Minor], true)
120 : $quality === IntervalQuality::Perfect;
121 if ($invalid || ($result->number === 1 && in_array($quality, [IntervalQuality::Diminished, IntervalQuality::DoubleDiminished], true)))
122 throw new CommandError("BAD INTERVAL $text");
123 return $result;
124 }
125 public static function between(Note $a, Note $b): Motion {
126 $distance = $b->position() - $a->position();
127 $semitones = $b->midi() - $a->midi();
128 $down = $distance < 0 || ($distance === 0 && $semitones < 0);
129 if ($down) { $distance = -$distance; $semitones = -$semitones; }
130 $difference = $semitones - (12 * intdiv($distance, 7) + Pitch::NATURALS[$distance % 7]);
131 $quality = in_array($distance % 7, [0, 3, 4], true)
132 ? match ($difference) {
133 -2 => IntervalQuality::DoubleDiminished, -1 => IntervalQuality::Diminished,
134 0 => IntervalQuality::Perfect, 1 => IntervalQuality::Augmented, 2 => IntervalQuality::DoubleAugmented, default => null,
135 }
136 : match ($difference) {
137 -3 => IntervalQuality::DoubleDiminished, -2 => IntervalQuality::Diminished, -1 => IntervalQuality::Minor,
138 0 => IntervalQuality::Major, 1 => IntervalQuality::Augmented, 2 => IntervalQuality::DoubleAugmented, default => null,
139 };
140 if ($quality === null) throw new CommandError('UNNAMEABLE INTERVAL');
141 return new Motion(new self($quality, $distance + 1), $down, $semitones);
142 }
143}

228 private static function number(string $text): int {
229 if (!preg_match('/\A(?:0|[1-9][0-9]{0,2})\z/', $text) || (int)$text > 127) throw new CommandError("BAD NUMBER $text");
230 return (int)$text;
231 }
232 private static function frequency(string $text, bool $tuning): int {
233 if (!preg_match('/\A([0-9]{1,5})(?:\.([0-9]{1,2}))?\z/', $text, $parts)) throw new CommandError("BAD FREQUENCY $text");
234 $fraction = $parts[2] ?? '';
235 $value = (int)$parts[1] * 100 + (int)$fraction * (strlen($fraction) === 1 ? 10 : 1);
236 if ($tuning ? $value < 40000 || $value > 48000 : $value === 0) throw new CommandError("BAD FREQUENCY $text");
237 return $value;
238 }
239 private function quality(string $text): Quality {
240 foreach ($this->qualities as $quality) if ($quality->name === $text) return $quality;
241 throw new CommandError("BAD QUALITY $text");
242 }
243 private function scalePattern(string $text): array { return $this->scales[$text] ?? throw new CommandError("BAD SCALE $text"); }
244 private static function inversion(string $text, Quality $quality): int {
245 if (!preg_match('/\A[0-9]\z/', $text) || (int)$text >= count($quality->pattern)) throw new CommandError("BAD INVERSION $text");
246 return (int)$text;
247 }
248 private function numeral(string $text): Numeral {
249 $start = str_contains('b#', $text[0]) ? 1 : 0;
250 $alteration = $start === 0 ? 0 : ($text[0] === 'b' ? -1 : 1);
251 $end = $start;
252 while ($end < strlen($text) && str_contains('IViv', $text[$end])) $end++;
253 $roman = substr($text, $start, $end - $start);
254 $degree = array_search(strtoupper($roman), ['I', 'II', 'III', 'IV', 'V', 'VI', 'VII'], true);
255 if ($degree !== false && ($roman === strtoupper($roman) || $roman === strtolower($roman))) {
256 foreach ($this->qualities as $quality) {
257 if ($quality->lower() === ($roman === strtolower($roman)) && substr($quality->numeral, 1) === substr($text, $end))
258 return new Numeral($degree, $alteration, $quality);
259 }
260 }
261 throw new CommandError("BAD NUMERAL $text");
262 }

267 if ($quality->suffix === substr($text, $prefix[1])) return new Chord($prefix[0], $quality);

271 }

274 private function degree(Key $key, Pitch $pitch): Degree {

283 private static function cadence(array $numerals): string {
284 if (count($numerals) < 2) return 'none';
285 $a = $numerals[count($numerals) - 2]; $b = $numerals[count($numerals) - 1];
286 if ($a->alteration === 0 && $b->alteration === 0) {
287 if ($a->degree === 4 && $b->degree === 0) return 'authentic';
288 if ($a->degree === 3 && $b->degree === 0) return 'plagal';
289 if ($a->degree === 4 && $b->degree === 5) return 'deceptive';
290 }
291 return $b->alteration === 0 && $b->degree === 4 ? 'half' : 'none';
292 }
293 public function execute(string $line): string {
294 $words = explode(' ', $line);
295 $command = array_shift($words);
296 $limits = self::LIMITS[$command] ?? throw new CommandError("UNKNOWN COMMAND $command");
297 $count = count($words);
298 if ($count < $limits[0] || $count > $limits[1]) throw new CommandError('ARGUMENT COUNT');
299 switch ($command) {
300 case 'MIDI':
301 $note = Note::parse($words[0]);
302 return "MIDI $note " . $note->midi();
303 case 'NOTE':
304 $number = self::number($words[0]);
305 return "NOTE $number " . Note::spelling($number) . ' ' . Note::spelling($number, true);
306 case 'FREQ':
307 $note = Note::parse($words[0]);
308 $cents = (int)floor($this->tuning * 2.0 ** (($note->midi() - 69) / 12.0) + 0.5);
309 return "FREQ $note " . frequencyText($cents);
310 case 'TUNING':
311 $value = self::frequency($words[0], true);
312 $this->tuning = $value;
313 return 'TUNING ' . frequencyText($value);
314 case 'PITCH':
315 $value = self::frequency($words[0], false);
316 $height = 69.0 + 12.0 * log($value / $this->tuning, 2.0);
317 $number = (int)floor($height + 0.5);
318 if ($number < 0 || $number > 127) throw new CommandError('OUT OF RANGE');
319 $cents = (int)floor(100.0 * ($height - $number) + 0.5);
320 return 'PITCH ' . frequencyText($value) . ' ' . Note::spelling($number) . ' ' . signedText($cents, true);
321 case 'INTERVAL':
322 $a = Note::parse($words[0]);
323 $b = Note::parse($words[1]);
324 $motion = Interval::between($a, $b);
325 return "INTERVAL $a $b $motion->interval " . ($motion->down ? 'down' : 'up')
326 . " semitones=$motion->semitones simple=" . $motion->interval->simple() . ' class=' . $motion->interval->consonance();
327 case 'INVERT':
328 $original = Interval::parse($words[0]);
329 $simple = $original->simple();
330 $inverse = new Interval($simple->quality->inverse(), 9 - $simple->number);
331 if ($inverse->number === 1 && in_array($inverse->quality, [IntervalQuality::Diminished, IntervalQuality::DoubleDiminished], true))
332 throw new CommandError('CANNOT INVERT');
333 return "INVERT $original $simple $inverse";
334 case 'TRANSPOSE':
335 $note = Note::parse($words[0]);
336 $interval = Interval::parse($words[1]);
337 $direction = $words[2];
338 if ($direction !== 'UP' && $direction !== 'DOWN') throw new CommandError("BAD DIRECTION $direction");
339 $sign = $direction === 'UP' ? 1 : -1;
340 $position = $note->position() + $sign * $interval->distance();
341 $midi = $note->midi() + $sign * $interval->semitones();
342 $letter = modulo($position, 7); $octave = floorDivide($position, 7);
343 $alteration = $midi - (12 * ($octave + 1) + Pitch::NATURALS[$letter]);
344 if ($alteration < -2 || $alteration > 2) throw new CommandError('CANNOT SPELL');
345 $result = new Note(new Pitch($letter, $alteration), $octave);
346 if (!$result->valid()) throw new CommandError('OUT OF RANGE');
347 return "TRANSPOSE $note $interval $direction $result";
348 case 'SCALE':
349 $root = Pitch::parse($words[0]);
350 $pattern = $this->scalePattern($words[1]);
351 $pitches = self::build($root, $pattern);
352 $steps = [];
353 foreach ($pattern as $i => $interval) $steps[] = ($i + 1 === count($pattern) ? 12 : $pattern[$i + 1]->semitones()) - $interval->semitones();
354 return "SCALE $root $words[1] " . implode(' ', $pitches) . ' steps=' . implode(',', $steps);
355 case 'CHORD': case 'VOICE':
356 $input = $command === 'VOICE' ? Note::parse($words[0]) : null;
357 $root = $input === null ? Pitch::parse($words[0]) : $input->pitch;
358 $quality = $this->quality($words[1]);
359 $inversion = $count === 3 ? self::inversion($words[2], $quality) : 0;
360 $pitches = self::build($root, $quality->pattern);
361 $pitches = array_merge(array_slice($pitches, $inversion), array_slice($pitches, 0, $inversion));
362 $symbol = $quality->symbol($root) . ($inversion === 0 ? '' : '/' . $pitches[0]);
363 if ($command === 'CHORD') return "CHORD $symbol " . implode(' ', $pitches);
364 $notes = []; $midis = []; $previous = $input->midi();
365 foreach ($pitches as $i => $pitch) {
366 $midi = $previous;
367 if ($i !== 0 || $inversion !== 0) {
368 $midi += modulo($pitch->pitchClass() - $previous, 12);
369 if ($midi === $previous) $midi += 12;
370 }
371 $note = Note::fromMidi($pitch, $midi);
372 if (!$note->valid()) throw new CommandError('OUT OF RANGE');
373 $notes[] = $note; $midis[] = $midi; $previous = $midi;
374 }
375 return "VOICE $symbol " . implode(' ', $notes) . ' midi=' . implode(',', $midis);
376 case 'IDENTIFY':
377 $notes = array_map(Note::parse(...), $words);
378 usort($notes, fn(Note $a, Note $b): int => $a->midi() <=> $b->midi());
379 $candidates = []; $mask = 0;
380 foreach ($notes as $note) {
381 $bit = 1 << $note->pitch->pitchClass();
382 if (($mask & $bit) === 0) $candidates[] = $note;
383 $mask |= $bit;
384 }
385 $bass = $candidates[0];
386 foreach ($candidates as $root) {
387 foreach ($this->qualities as $quality) {
388 $candidate = 0; $inversion = 0;
389 foreach ($quality->pattern as $i => $interval) {
390 $pitchClass = modulo($root->midi() + $interval->semitones(), 12);
391 $candidate |= 1 << $pitchClass;
392 if ($pitchClass === $bass->pitch->pitchClass()) $inversion = $i;
393 }
394 if ($candidate === $mask) {
395 $symbol = $quality->symbol($root->pitch) . ($inversion === 0 ? '' : '/' . $bass->pitch);
396 return "IDENTIFY $symbol root=$root->pitch quality=$quality->name inversion=$inversion";
397 }
398 }
399 }
400 return 'IDENTIFY UNKNOWN';
401 case 'MELODY':
402 $notes = array_map(Note::parse(...), $words);
403 $low = $high = $notes[0];
404 foreach ($notes as $note) {
405 if ($note->midi() < $low->midi()) $low = $note;
406 if ($note->midi() > $high->midi()) $high = $note;
407 }
408 $range = Interval::between($low, $high)->interval;
409 $steps = []; $leaps = 0;
410 for ($i = 1; $i < $count; $i++) {
411 $motion = Interval::between($notes[$i - 1], $notes[$i]);
412 $steps[] = ($motion->down ? '-' : '+') . $motion->interval;
413 if (abs($notes[$i]->midi() - $notes[$i - 1]->midi()) >= 3) $leaps++;
414 }
415 return "MELODY n=$count low=$low high=$high range=$range leaps=$leaps steps=" . implode(',', $steps);
416 case 'ENHARMONIC':
417 $input = Note::parse($words[0]);
418 $notes = [];
419 for ($letter = 0; $letter < 7; $letter++) {
420 for ($alteration = -2; $alteration <= 2; $alteration++) {

452 $seventh = modulo($scale[($i + 6) % 7]->pitchClass() - $scale[$i]->pitchClass(), 12);

492 foreach ($pitches as $pitch) {
493 $found = false;

495 if (!$found) { $matches = false; break; }
496 }

498 }

508 $pitch = Note::spelling($midi, $key->fifths() < 0)->pitch;
509 foreach ($this->scale($key) as $candidate) if ($candidate->pitchClass() === $midi % 12) { $pitch = $candidate; break; }
510 $note = Note::fromMidi($pitch, $midi);
511 if (!$note->valid()) throw new CommandError('OUT OF RANGE');
512 return "SPELL $midi $key $note " . $this->degree($key, $pitch);
513 case 'KEYDIST':
514 $a = Key::parse($words[0], $words[1]);
515 $b = Key::parse($words[2], $words[3]);
516 $distance = modulo($b->fifths() - $a->fifths(), 12);
517 if ($distance > 6) $distance -= 12;
518 $first = self::mask($this->scale($a)); $second = self::mask($this->scale($b));
519 $sameTonic = $a->tonic->pitchClass() === $b->tonic->pitchClass();
520 $relation = $first === $second ? ($sameTonic ? 'same' : 'relative')
521 : ($sameTonic ? 'parallel' : (abs($distance) === 1 ? 'close' : 'distant'));
522 $common = 0;
523 for ($i = 0; $i < 12; $i++) if (($first & $second & (1 << $i)) !== 0) $common++;
524 return "KEYDIST $a $b fifths=" . signedText($distance) . " common=$common relation=$relation";
525 }
526 throw new LogicException('Unreachable command');
527 }
