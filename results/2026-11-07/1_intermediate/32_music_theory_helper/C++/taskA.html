22int mod(int value, int divisor) { return (value % divisor + divisor) % divisor; }
23int floor_div(int value, int divisor) { return (value - mod(value, divisor)) / divisor; }

31vector<string> words(const string& text) {
32 std::istringstream stream(text);
33 vector<string> result;
34 for (string word; stream >> word;) result.push_back(word);
35 return result;
36}
37string join(const vector<string>& items, const string& separator = " ") {
38 string result;
39 for (std::size_t i = 0; i < items.size(); ++i) {
40 if (i) result += separator;
41 result += items[i];
42 }
43 return result;
44}
45int integer(const string& text, const string& kind, int low, int high) {
46 static const std::regex form("0|[1-9][0-9]*");
47 if (text.size() > 3 || !std::regex_match(text, form)) bad(kind, text);
48 int value = std::stoi(text);
49 if (value < low || value > high) bad(kind, text);
50 return value;
51}
52int frequency(const string& text, bool tuning) {
53 static const std::regex form("[0-9]{1,5}(\\.[0-9]{1,2})?");
54 if (!std::regex_match(text, form)) bad("FREQUENCY", text);
55 auto dot = text.find('.');
56 string fraction = dot == string::npos ? "" : text.substr(dot + 1);
57 fraction.append(2 - fraction.size(), '0');
58 int value = std::stoi(text.substr(0, dot)) * 100 + std::stoi(fraction);
59 if (value == 0 || (tuning && (value < 40000 || value > 48000))) bad("FREQUENCY", text);
60 return value;
61}
62string frequency_text(int cents) {
63 return to_string(cents / 100) + "." + (cents % 100 < 10 ? "0" : "") + to_string(cents % 100);
64}

66struct Tone { int letters, semitones; };
67struct Pitch {
68 int letter, alteration;
69 int pitch_class() const { return mod(naturals[letter] + alteration, 12); }
70 string text() const { return string(1, letters[letter]) + accidentals[alteration + 2]; }
71 bool operator==(const Pitch& other) const { return letter == other.letter && alteration == other.alteration; }
72 Pitch raise(Tone tone) const {
73 int target = letter + tone.letters;
74 int value = naturals[letter] + alteration + tone.semitones - naturals[mod(target, 7)] - 12 * floor_div(target, 7);
75 if (value < -2 || value > 2) fail("CANNOT SPELL");
76 return {mod(target, 7), value};
77 }
78};
79std::pair<Pitch, string> pitch_prefix(const string& text, const string& kind) {
80 static const std::regex form("^([A-G])(bb|b|#|x)?");
81 std::smatch match;
82 if (!std::regex_search(text, match, form)) bad(kind, text);
83 int letter = static_cast<int>(letters.find(match[1].str()));
84 int alteration = static_cast<int>(std::find(accidentals.begin(), accidentals.end(), match[2].str()) - accidentals.begin()) - 2;
85 return {{letter, alteration}, text.substr(match.length())};
86}
87Pitch parse_pitch(const string& text) {
88 auto [pitch, rest] = pitch_prefix(text, "PITCH");
89 if (!rest.empty()) bad("PITCH", text);
90 return pitch;
91}
92Pitch fifth_pitch(int value) {
93 const std::array<int, 7> order{3, 0, 4, 1, 5, 2, 6};
94 return {order[mod(value + 1, 7)], floor_div(value + 1, 7)};
95}
96struct Note {
97 Pitch pitch;
98 int octave;
99 int midi() const { return 12 * (octave + 1) + naturals[pitch.letter] + pitch.alteration; }
100 int position() const { return 7 * octave + pitch.letter; }
101 bool valid() const { return octave >= -1 && octave <= 9 && midi() >= 0 && midi() <= 127; }
102 string text() const { return pitch.text() + to_string(octave); }
103};
104Note parse_note(const string& text) {
105 auto [pitch, rest] = pitch_prefix(text, "NOTE");
106 static const std::regex form("-1|[0-9]");
107 if (!std::regex_match(rest, form)) bad("NOTE", text);
108 Note note{pitch, std::stoi(rest)};
109 if (!note.valid()) bad("NOTE", text);
110 return note;
111}
112Note make_note(Pitch pitch, int midi) { return {pitch, (midi - naturals[pitch.letter] - pitch.alteration) / 12 - 1}; }
113Note spelling(int midi, bool flat = false) { return {parse_pitch((flat ? flats : sharps)[mod(midi, 12)]), floor_div(midi, 12) - 1}; }

115const std::array<string, 7> interval_qualities{"P", "M", "m", "A", "d", "AA", "dd"};
116std::optional<int> quality_delta(const string& quality, int number) {
117 int remainder = (number - 1) % 7;
118 bool perfect = remainder == 0 || remainder == 3 || remainder == 4;
119 if (quality == "P" && perfect) return 0;
120 if (quality == "M" && !perfect) return 0;
121 if (quality == "m" && !perfect) return -1;
122 if (quality == "A") return 1;
123 if (quality == "AA") return 2;
124 if (quality == "d") return perfect ? -1 : -2;
125 if (quality == "dd") return perfect ? -2 : -3;
126 return std::nullopt;
127}
128struct Interval {
129 string quality;
130 int number;
131 string text() const { return quality + to_string(number); }
132 bool forbidden() const { return number == 1 && (quality == "d" || quality == "dd"); }
133 Tone tone() const { return {number - 1, 12 * ((number - 1) / 7) + naturals[(number - 1) % 7] + *quality_delta(quality, number)}; }
134 Interval simple() const {
135 int result = number;
136 while (result > 8) result -= 7;
137 return {quality, result};
138 }
139 string consonance() const {
140 int n = simple().number;
141 if (quality == "P" && (n == 1 || n == 5 || n == 8)) return "perfect";
142 if ((quality == "m" || quality == "M") && (n == 3 || n == 6)) return "imperfect";
143 return "dissonant";
144 }
145};
146Interval parse_interval(const string& text) {
147 static const std::regex form("(AA|dd|P|M|m|A|d)([1-9][0-9]?)");
148 std::smatch match;
149 if (!std::regex_match(text, match, form)) bad("INTERVAL", text);
150 Interval interval{match[1].str(), std::stoi(match[2].str())};
151 if (!quality_delta(interval.quality, interval.number) || interval.forbidden()) bad("INTERVAL", text);
152 return interval;
153}
154struct Motion { Interval interval; bool down; int semitones; };
155Motion between(Note first, Note second) {
156 int distance = second.position() - first.position(), semitones = second.midi() - first.midi();
157 bool down = distance < 0 || (distance == 0 && semitones < 0);
158 if (down) { distance = -distance; semitones = -semitones; }
159 int difference = semitones - 12 * (distance / 7) - naturals[distance % 7];
160 for (const auto& quality : interval_qualities) {
161 if (quality_delta(quality, distance + 1) == difference) return {{quality, distance + 1}, down, semitones};
162 }
163 fail("UNNAMEABLE INTERVAL");
164}

197 Chord(string name, string suffix, const string& numeral, const string& pattern)
198 : name(std::move(name)), suffix(std::move(suffix)), lower(numeral[0] == 'i'),

200 string symbol(Pitch root, std::optional<Pitch> bass = std::nullopt) const {

203 string roman(int degree) const {
204 string result = romans[degree];

229struct Key {
230 Pitch tonic;
231 Mode mode;
232 int fifths;
233 vector<Pitch> scale;
234 Key(Pitch tonic, Mode mode) : tonic(tonic), mode(std::move(mode)),
235 fifths(fifth_values[tonic.letter] + 7 * tonic.alteration + this->mode.offset) {
236 if (fifths < -7 || fifths > 7) fail("THEORETICAL KEY");
237 scale = build(tonic, named(this->mode.scale, "SCALE", scales).tones);
238 }
239 string text() const { return tonic.text() + " " + mode.name; }
240 int degree(Pitch pitch) const { return mod(pitch.letter - tonic.letter, 7); }
241 string prefix(Pitch pitch) const {
242 int difference = pitch.alteration - scale[degree(pitch)].alteration;
243 return string(std::abs(difference), difference < 0 ? 'b' : '#');
244 }
245 string label(Pitch pitch) const { return prefix(pitch) + to_string(degree(pitch) + 1); }
246 std::set<int> pitch_classes() const {
247 std::set<int> result;
248 for (auto pitch : scale) result.insert(pitch.pitch_class());
249 return result;
250 }
251};
252Key parse_key(const vector<string>& args, std::size_t start = 0) {
253 Pitch tonic = parse_pitch(args[start]);
254 const Mode& mode = named(args[start + 1], "MODE", modes);
255 return {tonic, mode};
256}
257struct Numeral { int alteration, degree; const Chord* chord; };
258Numeral parse_numeral(const string& text) {
259 std::size_t start = text[0] == 'b' || text[0] == '#' ? 1 : 0, end = start;
260 int alteration = start == 0 ? 0 : text[0] == 'b' ? -1 : 1;
261 while (end < text.size() && string("IViv").find(text[end]) != string::npos) ++end;
262 string part = text.substr(start, end - start), suffix = text.substr(end);
263 for (int degree = 0; degree < 7; ++degree) {
264 for (const Chord& chord : chords) {
265 string expected = romans[degree];
266 if (chord.lower) for (char& letter : expected) letter = letter == 'I' ? 'i' : 'v';
267 if (part == expected && suffix == chord.numeral_suffix) return {alteration, degree, &chord};
268 }
269 }
270 bad("NUMERAL", text);
271}
272struct Symbol { Pitch root; const Chord* chord; };
273Symbol parse_symbol(const string& text) {
274 auto [pitch, rest] = pitch_prefix(text, "SYMBOL");
275 for (const Chord& chord : chords) if (chord.suffix == rest) return {pitch, &chord};
276 bad("SYMBOL", text);
277}
278string identify(vector<Note> notes) {
279 std::stable_sort(notes.begin(), notes.end(), [](Note a, Note b) { return a.midi() < b.midi(); });
280 std::set<int> present;
281 vector<Note> candidates;
282 for (auto note : notes) if (present.insert(note.pitch.pitch_class()).second) candidates.push_back(note);
283 Pitch bass = candidates.front().pitch;
284 for (auto root : candidates) {
285 for (const Chord& chord : chords) {
286 vector<int> classes;
287 for (auto tone : chord.tones) classes.push_back(mod(root.midi() + tone.semitones, 12));
288 if (std::set<int>(classes.begin(), classes.end()) != present) continue;
289 int inversion = static_cast<int>(std::find(classes.begin(), classes.end(), bass.pitch_class()) - classes.begin());
290 std::optional<Pitch> slash = inversion == 0 ? std::nullopt : std::optional<Pitch>(bass);
291 return "IDENTIFY " + chord.symbol(root.pitch, slash) + " root=" + root.pitch.text()
292 + " quality=" + chord.name + " inversion=" + to_string(inversion);
293 }
294 }
295 return "IDENTIFY UNKNOWN";
296}
297string melody(const vector<Note>& notes, const vector<string>& args) {
298 std::size_t low = 0, high = 0;
299 for (std::size_t i = 1; i < notes.size(); ++i) {
300 if (notes[i].midi() < notes[low].midi()) low = i;
301 if (notes[i].midi() > notes[high].midi()) high = i;
302 }
303 Interval range = between(notes[low], notes[high]).interval;
304 vector<string> steps;
305 int leaps = 0;
306 for (std::size_t i = 1; i < notes.size(); ++i) {
307 Motion motion = between(notes[i - 1], notes[i]);
308 steps.push_back(string(motion.down ? "-" : "+") + motion.interval.text());
309 if (std::abs(notes[i].midi() - notes[i - 1].midi()) >= 3) ++leaps;
310 }
311 return "MELODY n=" + to_string(notes.size()) + " low=" + args[low] + " high=" + args[high]
312 + " range=" + range.text() + " leaps=" + to_string(leaps) + " steps=" + join(steps, ",");
313}
314string cadence(const vector<Numeral>& numerals) {
315 if (numerals.size() < 2) return "none";
316 const Numeral& first = numerals[numerals.size() - 2];
317 const Numeral& last = numerals.back();
318 if (first.alteration == 0 && last.alteration == 0) {
319 if (first.degree == 4 && last.degree == 0) return "authentic";
320 if (first.degree == 3 && last.degree == 0) return "plagal";
321 if (first.degree == 4 && last.degree == 5) return "deceptive";
322 }
323 return last.alteration == 0 && last.degree == 4 ? "half" : "none";
324}

326vector<string> diatonic(const Key& key) {
327 vector<string> result{"DIATONIC " + key.text()};
328 for (int size : {3, 4}) {
329 vector<string> items;
330 for (int degree = 0; degree < 7; ++degree) {
331 int root = key.scale[degree].pitch_class();
332 int third = mod(key.scale[(degree + 2) % 7].pitch_class() - root, 12);
333 int fifth = mod(key.scale[(degree + 4) % 7].pitch_class() - root, 12);
334 string quality;
335 if (size == 3) quality = third == 4 ? "maj" : fifth == 7 ? "min" : "dim";
336 else {
337 int seventh = mod(key.scale[(degree + 6) % 7].pitch_class() - root, 12);
338 quality = third == 4 ? (seventh == 11 ? "maj7" : "7") : fifth == 7 ? "m7" : "m7b5";
339 }
340 const Chord& chord = named(quality, "QUALITY", chords);
341 items.push_back(chord.roman(degree) + "=" + chord.symbol(key.scale[degree]));
342 }
343 result.push_back(string(size == 3 ? "TRIADS " : "SEVENTHS ") + join(items));
344 }
345 return result;
346}
347vector<string> execute_key(const string& command, const vector<string>& args) {
348 Key key = parse_key(args);
349 if (command == "KEYSIG") {
350 if (key.fifths == 0) return {"KEYSIG " + key.text() + " none"};
351 vector<string> names = words(key.fifths > 0 ? "F# C# G# D# A# E# B#" : "Bb Eb Ab Db Gb Cb Fb");
352 names.resize(std::abs(key.fifths));
353 return {"KEYSIG " + key.text() + (key.fifths > 0 ? " sharps=" : " flats=") + to_string(std::abs(key.fifths)) + " " + join(names)};
354 }
355 if (command == "RELATIVE") return {"RELATIVE " + key.text() + " major=" + fifth_pitch(key.fifths).text() + " minor=" + fifth_pitch(key.fifths + 3).text()};
356 if (command == "DIATONIC") return diatonic(key);
357 if (command == "DEGREE") {
358 Pitch pitch = parse_pitch(args[2]);
359 return {"DEGREE " + key.text() + " " + args[2] + " " + key.label(pitch)};
360 }
361 if (command == "KEYDIST") {
362 Key other = parse_key(args, 2);
363 int distance = mod(other.fifths - key.fifths, 12);
364 if (distance > 6) distance -= 12;
365 std::set<int> first = key.pitch_classes(), second = other.pitch_classes();
366 bool same_tonic = key.tonic.pitch_class() == other.tonic.pitch_class();
367 string relation = first == second ? (same_tonic ? "same" : "relative")
368 : same_tonic ? "parallel" : std::abs(distance) == 1 ? "close" : "distant";
369 int common = 0;
370 for (int pitch : first) if (second.count(pitch)) ++common;
371 return {"KEYDIST " + key.text() + " " + other.text() + " fifths=" + (distance > 0 ? "+" : "")
372 + to_string(distance) + " common=" + to_string(common) + " relation=" + relation};
373 }
374 if (command == "ANALYZE") {
375 vector<Symbol> symbols;
376 for (std::size_t i = 2; i < args.size(); ++i) symbols.push_back(parse_symbol(args[i]));
377 vector<string> results;
378 for (const Symbol& symbol : symbols) results.push_back(key.prefix(symbol.root) + symbol.chord->roman(key.degree(symbol.root)));
379 return {"ANALYZE " + key.text() + " " + join(results)};
380 }
381 vector<Numeral> numerals;
382 for (std::size_t i = 2; i < args.size(); ++i) numerals.push_back(parse_numeral(args[i]));
383 vector<string> result{"PROGRESSION " + key.text() + " " + to_string(numerals.size())};
384 for (std::size_t i = 0; i < numerals.size(); ++i) {
385 const Numeral& numeral = numerals[i];
386 Pitch root = key.scale[numeral.degree];
387 root.alteration += numeral.alteration;
388 vector<Pitch> pitches = build(root, numeral.chord->tones);
389 result.push_back(args[i + 2] + " " + numeral.chord->symbol(root) + " " + join(pitch_names(pitches)));
390 }
391 result.push_back("CADENCE " + cadence(numerals));
392 return result;
393}

396 {"MIDI", {1, 1}}, {"NOTE", {1, 1}}, {"FREQ", {1, 1}}, {"TUNING", {1, 1}}, {"PITCH", {1, 1}},

399 {"ENHARMONIC", {1, 1}}, {"KEYSIG", {2, 2}}, {"RELATIVE", {2, 2}}, {"CIRCLE", {1, 1}},
400 {"DIATONIC", {2, 2}}, {"PROGRESSION", {3, 15}}, {"ANALYZE", {3, 15}}, {"FINDKEY", {1, 15}},

403class Calculator {
404 int tuning_ = 44000;
405public:
406 vector<string> execute(const string& line) {
407 vector<string> args = words(line);
408 string command = args.front();
409 auto entry = arities.find(command);
410 if (entry == arities.end()) fail("UNKNOWN COMMAND " + command);
411 args.erase(args.begin());
412 auto [low, high] = entry->second;
413 if (args.size() < static_cast<std::size_t>(low) || args.size() > static_cast<std::size_t>(high)) fail("ARGUMENT COUNT");
414 if (command == "MIDI" || command == "FREQ" || command == "ENHARMONIC") {
415 Note note = parse_note(args[0]);
416 int value = note.midi();
417 if (command == "MIDI") return {"MIDI " + args[0] + " " + to_string(value)};
418 if (command == "FREQ") {
419 int cents = static_cast<int>(std::floor(tuning_ * std::exp2((value - 69) / 12.0) + 0.5));
420 return {"FREQ " + args[0] + " " + frequency_text(cents)};
421 }
422 vector<string> equivalents;
423 for (int octave = -1; octave <= 9; ++octave) {
424 for (int letter = 0; letter < 7; ++letter) {
425 int alteration = value - 12 * (octave + 1) - naturals[letter];
426 if (alteration >= -2 && alteration <= 2) equivalents.push_back(Note{{letter, alteration}, octave}.text());
427 }
428 }
429 return {"ENHARMONIC " + args[0] + " " + join(equivalents)};
430 }
431 if (command == "NOTE") {
432 int value = integer(args[0], "NUMBER", 0, 127);
433 return {"NOTE " + to_string(value) + " " + spelling(value).text() + " " + spelling(value, true).text()};
434 }
435 if (command == "TUNING" || command == "PITCH") {
436 int value = frequency(args[0], command == "TUNING");
437 if (command == "TUNING") {
438 tuning_ = value;
439 return {"TUNING " + frequency_text(value)};
440 }
441 double height = 69 + 12 * std::log2(static_cast<double>(value) / tuning_);
442 int nearest = static_cast<int>(std::floor(height + 0.5));
443 if (nearest < 0 || nearest > 127) fail("OUT OF RANGE");
444 int cents = static_cast<int>(std::floor(100 * (height - nearest) + 0.5));
445 return {"PITCH " + frequency_text(value) + " " + spelling(nearest).text() + " " + (cents >= 0 ? "+" : "") + to_string(cents)};
446 }
447 if (command == "INTERVAL") {
448 Note first = parse_note(args[0]);
449 Note second = parse_note(args[1]);
450 Motion motion = between(first, second);
451 return {"INTERVAL " + args[0] + " " + args[1] + " " + motion.interval.text() + (motion.down ? " down" : " up")
452 + " semitones=" + to_string(motion.semitones) + " simple=" + motion.interval.simple().text() + " class=" + motion.interval.consonance()};
453 }
454 if (command == "INVERT") {
455 Interval simple = parse_interval(args[0]).simple();
456 const std::map<string, string> inverse_qualities{{"P", "P"}, {"M", "m"}, {"m", "M"}, {"A", "d"}, {"d", "A"}, {"AA", "dd"}, {"dd", "AA"}};
457 Interval inverse{inverse_qualities.at(simple.quality), 9 - simple.number};
458 if (inverse.forbidden()) fail("CANNOT INVERT");
459 return {"INVERT " + args[0] + " " + simple.text() + " " + inverse.text()};
460 }
461 if (command == "TRANSPOSE") {
462 Note note = parse_note(args[0]);
463 Interval interval = parse_interval(args[1]);
464 const string& direction = args[2];
465 if (direction != "UP" && direction != "DOWN") bad("DIRECTION", direction);
466 int sign = direction == "UP" ? 1 : -1;
467 Tone tone = interval.tone();
468 int target = note.position() + sign * tone.letters, value = note.midi() + sign * tone.semitones;
469 int letter = mod(target, 7), octave = floor_div(target, 7), alteration = value - 12 * (octave + 1) - naturals[letter];
470 if (alteration < -2 || alteration > 2) fail("CANNOT SPELL");
471 Note result{{letter, alteration}, octave};
472 if (!result.valid()) fail("OUT OF RANGE");
473 return {"TRANSPOSE " + args[0] + " " + args[1] + " " + direction + " " + result.text()};
474 }
475 if (command == "SCALE") {
476 Pitch root = parse_pitch(args[0]);
477 const Scale& scale = named(args[1], "SCALE", scales);
478 vector<Pitch> pitches = build(root, scale.tones);
479 vector<string> steps;
480 for (std::size_t i = 0; i < scale.tones.size(); ++i) {
481 int next = i + 1 == scale.tones.size() ? 12 : scale.tones[i + 1].semitones;
482 steps.push_back(to_string(next - scale.tones[i].semitones));
483 }
484 return {"SCALE " + args[0] + " " + args[1] + " " + join(pitch_names(pitches)) + " steps=" + join(steps, ",")};
485 }
486 if (command == "CHORD" || command == "VOICE") {
487 std::optional<Note> input;
488 if (command == "VOICE") input = parse_note(args[0]);
489 Pitch root = input ? input->pitch : parse_pitch(args[0]);
490 const Chord& chord = named(args[1], "QUALITY", chords);
491 int inversion = args.size() == 3 ? integer(args[2], "INVERSION", 0, static_cast<int>(chord.tones.size()) - 1) : 0;
492 vector<Pitch> pitches = build(root, chord.tones);
493 std::rotate(pitches.begin(), pitches.begin() + inversion, pitches.end());
494 std::optional<Pitch> bass = inversion == 0 ? std::nullopt : std::optional<Pitch>(pitches.front());
495 string symbol = chord.symbol(root, bass);
496 if (command == "CHORD") return {"CHORD " + symbol + " " + join(pitch_names(pitches))};
497 int value = input->midi();
498 vector<string> notes, numbers;
499 for (std::size_t i = 0; i < pitches.size(); ++i) {
500 if (i > 0 || inversion > 0) {
501 int difference = mod(pitches[i].pitch_class() - value, 12);
502 value += difference == 0 ? 12 : difference;
503 }
504 Note note = make_note(pitches[i], value);
505 if (!note.valid()) fail("OUT OF RANGE");
506 notes.push_back(note.text());
507 numbers.push_back(to_string(value));
508 }
509 return {"VOICE " + symbol + " " + join(notes) + " midi=" + join(numbers, ",")};
510 }
511 if (command == "IDENTIFY" || command == "MELODY") {
512 vector<Note> notes;
513 for (const string& arg : args) notes.push_back(parse_note(arg));
514 return {command == "IDENTIFY" ? identify(notes) : melody(notes, args)};
515 }
516 if (command == "CIRCLE") {
517 const Mode& mode = named(args[0], "MODE", modes);
518 vector<string> names;
519 for (int fifths = -7; fifths <= 7; ++fifths) names.push_back(fifth_pitch(fifths - mode.offset).text());
520 return {"CIRCLE " + args[0] + " " + join(names)};
521 }
522 if (command == "FINDKEY") {
523 vector<Pitch> pitches;
524 for (const string& arg : args) pitches.push_back(parse_pitch(arg));
525 vector<string> keys;
526 for (int fifths = -7; fifths <= 7; ++fifths) {
527 for (int i = 0; i < 2; ++i) {
528 Key key(fifth_pitch(fifths - modes[i].offset), modes[i]);
529 bool contains = std::all_of(pitches.begin(), pitches.end(), [&](Pitch pitch) {
530 return std::find(key.scale.begin(), key.scale.end(), pitch) != key.scale.end();
531 });
532 if (contains) keys.push_back(key.text());
533 }
534 }
535 return {"FINDKEY " + to_string(keys.size()) + (keys.empty() ? "" : " " + join(keys, ", "))};
536 }
537 if (command == "SPELL") {
538 int value = integer(args[0], "NUMBER", 0, 127);
539 Key key = parse_key(args, 1);
540 Pitch pitch = spelling(value, key.fifths < 0).pitch;
541 for (auto candidate : key.scale) if (candidate.pitch_class() == value % 12) { pitch = candidate; break; }
542 Note note = make_note(pitch, value);
543 if (!note.valid()) fail("OUT OF RANGE");
544 return {"SPELL " + to_string(value) + " " + key.text() + " " + note.text() + " " + key.label(pitch)};
545 }
546 return execute_key(command, args);
547 }
548};
