21 static int canonical(String text, int low, int high) {
22 if (!text.matches("0|[1-9][0-9]*")) return -1;
23 try {
24 int value = Integer.parseInt(text);
25 return value >= low && value <= high ? value : -1;
26 } catch (NumberFormatException e) {
27 return -1;
28 }
29 }
30 static int number(String text) {

35 static int frequency(String text, boolean tuning) {
36 if (!text.matches("[0-9]{1,5}(\\.[0-9]{1,2})?")) throw bad("FREQUENCY", text);
37 String[] parts = text.split("\\.");
38 int cents = Integer.parseInt(parts[0]) * 100;
39 if (parts.length == 2) cents += Integer.parseInt(parts[1]) * (parts[1].length() == 1 ? 10 : 1);
40 if (cents == 0 || tuning && (cents < 40000 || cents > 48000)) throw bad("FREQUENCY", text);
41 return cents;
42 }

45 record Pitch(int letter, int alteration) {
46 int pitchClass() { return Math.floorMod(NATURALS[letter] + alteration, 12); }
47 Pitch raise(Tone tone) {
48 int target = letter + tone.letters;
49 int change = NATURALS[letter] + alteration + tone.semitones - NATURALS[target % 7] - 12 * (target / 7);
50 if (change < -2 || change > 2) throw new CommandError("CANNOT SPELL");
51 return new Pitch(target % 7, change);
52 }
53 @Override public String toString() { return LETTERS.charAt(letter) + ACCIDENTALS[alteration + 2]; }
54 }
55 record PitchPrefix(Pitch pitch, int end) {}
56 static PitchPrefix pitchPrefix(String text) {
57 int letter = LETTERS.indexOf(text.charAt(0));
58 if (letter < 0) return null;
59 int end = 1, alteration = 0;
60 if (end < text.length()) {
61 switch (text.charAt(end)) {
62 case 'b' -> {
63 alteration = -1;
64 end++;
65 if (end < text.length() && text.charAt(end) == 'b') { alteration = -2; end++; }
66 }
67 case '#' -> { alteration = 1; end++; }
68 case 'x' -> { alteration = 2; end++; }
69 }
70 }
71 return new PitchPrefix(new Pitch(letter, alteration), end);
72 }
73 static Pitch pitch(String text) {
74 PitchPrefix prefix = pitchPrefix(text);
75 if (prefix == null || prefix.end != text.length()) throw bad("PITCH", text);
76 return prefix.pitch;
77 }
78 static Pitch fifthPitch(int fifths) {
79 return new Pitch(LETTERS.indexOf("FCGDAEB".charAt(Math.floorMod(fifths + 1, 7))), Math.floorDiv(fifths + 1, 7));
80 }
81 record Note(Pitch pitch, int octave) {
82 int midi() { return 12 * (octave + 1) + NATURALS[pitch.letter] + pitch.alteration; }
83 int position() { return 7 * octave + pitch.letter; }
84 boolean valid() { return octave >= -1 && octave <= 9 && midi() >= 0 && midi() <= 127; }
85 @Override public String toString() { return pitch.toString() + octave; }
86 }
87 static Note note(String text) {
88 PitchPrefix prefix = pitchPrefix(text);
89 if (prefix == null) throw bad("NOTE", text);
90 String rest = text.substring(prefix.end);
91 int octave;
92 if (rest.equals("-1")) octave = -1;
93 else if (rest.length() == 1 && rest.charAt(0) >= '0' && rest.charAt(0) <= '9') octave = rest.charAt(0) - '0';
94 else throw bad("NOTE", text);
95 Note result = new Note(prefix.pitch, octave);
96 if (!result.valid()) throw bad("NOTE", text);
97 return result;
98 }

100 return new Note(pitch, (midi - NATURALS[pitch.letter] - pitch.alteration) / 12 - 1);

103 return new Note(pitch((flat ? FLATS : SHARPS)[midi % 12]), midi / 12 - 1);
104 }

111 enum Quality {
112 PERFECT("P"), MAJOR("M"), MINOR("m"), AUGMENTED("A"), DIMINISHED("d"), DOUBLE_AUGMENTED("AA"), DOUBLE_DIMINISHED("dd");
113 final String text;
114 Quality(String text) { this.text = text; }
115 Integer delta(boolean perfect) {
116 return switch (this) {
117 case PERFECT -> perfect ? 0 : null;
118 case MAJOR -> perfect ? null : 0;
119 case MINOR -> perfect ? null : -1;
120 case AUGMENTED -> 1;
121 case DIMINISHED -> perfect ? -1 : -2;
122 case DOUBLE_AUGMENTED -> 2;
123 case DOUBLE_DIMINISHED -> perfect ? -2 : -3;
124 };
125 }
126 Quality inverse() {
127 return switch (this) {
128 case PERFECT -> PERFECT;
129 case MAJOR -> MINOR;
130 case MINOR -> MAJOR;
131 case AUGMENTED -> DIMINISHED;
132 case DIMINISHED -> AUGMENTED;
133 case DOUBLE_AUGMENTED -> DOUBLE_DIMINISHED;
134 case DOUBLE_DIMINISHED -> DOUBLE_AUGMENTED;
135 };
136 }
137 }
138 static boolean perfectType(int number) {
139 int remainder = (number - 1) % 7;
140 return remainder == 0 || remainder == 3 || remainder == 4;
141 }
142 record Tone(int letters, int semitones) {}
143 record Interval(Quality quality, int number) {
144 boolean forbidden() { return number == 1 && (quality == Quality.DIMINISHED || quality == Quality.DOUBLE_DIMINISHED); }
145 Tone tone() {
146 int distance = number - 1;
147 return new Tone(distance, 12 * (distance / 7) + NATURALS[distance % 7] + quality.delta(perfectType(number)));
148 }
149 Interval simple() {
150 int n = number;
151 while (n > 8) n -= 7;
152 return new Interval(quality, n);
153 }
154 String consonance() {
155 Interval simple = simple();
156 if (quality == Quality.PERFECT && (simple.number == 1 || simple.number == 5 || simple.number == 8)) return "perfect";
157 if ((quality == Quality.MAJOR || quality == Quality.MINOR) && (simple.number == 3 || simple.number == 6)) return "imperfect";
158 return "dissonant";
159 }
160 @Override public String toString() { return quality.text + number; }
161 }
162 static Interval interval(String text) {
163 for (Quality quality : Quality.values()) {
164 if (!text.startsWith(quality.text)) continue;
165 int n = canonical(text.substring(quality.text.length()), 1, 99);
166 if (n < 0 || quality.delta(perfectType(n)) == null) continue;
167 Interval result = new Interval(quality, n);
168 if (!result.forbidden()) return result;
169 }
170 throw bad("INTERVAL", text);
171 }
172 record Motion(Interval interval, boolean down, int semitones) {}
173 static Motion between(Note first, Note second) {
174 int distance = second.position() - first.position(), semitones = second.midi() - first.midi();
175 boolean down = distance < 0 || distance == 0 && semitones < 0;
176 if (down) { distance = -distance; semitones = -semitones; }
177 int difference = semitones - 12 * (distance / 7) - NATURALS[distance % 7];
178 for (Quality quality : Quality.values()) {
179 Integer delta = quality.delta(perfectType(distance + 1));
180 if (delta != null && delta == difference) return new Motion(new Interval(quality, distance + 1), down, semitones);
181 }
182 throw new CommandError("UNNAMEABLE INTERVAL");
183 }

280 static Numeral numeral(String text) {
281 int start = 0, alteration = 0;
282 if (text.charAt(0) == 'b') { start = 1; alteration = -1; }
283 else if (text.charAt(0) == '#') { start = 1; alteration = 1; }
284 int end = start;
285 while (end < text.length() && "IViv".indexOf(text.charAt(end)) >= 0) end++;
286 String part = text.substring(start, end), suffix = text.substring(end);
287 for (int degree = 0; degree < 7; degree++) {
288 for (int quality = 0; quality < CHORDS.size(); quality++) {
289 Chord chord = CHORDS.get(quality);
290 String expected = chord.lower ? ROMANS[degree].toLowerCase(Locale.ROOT) : ROMANS[degree];
291 if (part.equals(expected) && suffix.equals(chord.numeralSuffix)) return new Numeral(alteration, degree, quality);
292 }
293 }
294 throw bad("NUMERAL", text);
295 }

306 static String chordCommand(String command, List<String> args) {
307 Note input = command.equals("VOICE") ? note(args.get(0)) : null;
308 Pitch root = input == null ? pitch(args.get(0)) : input.pitch;
309 int quality = chordQuality(args.get(1));
310 Chord chord = CHORDS.get(quality);
311 int inversion = args.size() == 3 ? canonical(args.get(2), 0, chord.tones.size() - 1) : 0;
312 if (inversion < 0) throw bad("INVERSION", args.get(2));
313 List<Pitch> tones = build(root, chord.tones), ordered = new ArrayList<>();
314 for (int i = 0; i < tones.size(); i++) ordered.add(tones.get((i + inversion) % tones.size()));
315 String name = symbol(root, quality, inversion == 0 ? null : ordered.getFirst());
316 if (input == null) return "CHORD " + name + " " + join(ordered, " ");
317 List<Note> result = new ArrayList<>();
318 List<Integer> values = new ArrayList<>();
319 int value = input.midi();
320 for (int i = 0; i < ordered.size(); i++) {
321 Pitch pitch = ordered.get(i);
322 if (i != 0 || inversion != 0) {
323 int difference = Math.floorMod(pitch.pitchClass() - value, 12);
324 value += difference == 0 ? 12 : difference;
325 }
326 Note next = fromPitch(pitch, value);
327 if (!next.valid()) throw new CommandError("OUT OF RANGE");
328 result.add(next);
329 values.add(value);
330 }
331 return "VOICE " + name + " " + join(result, " ") + " midi=" + join(values, ",");
332 }
333 static String identify(List<Note> notes) {
334 notes.sort(Comparator.comparingInt(Note::midi));
335 List<Note> candidates = new ArrayList<>();
336 int set = 0;
337 for (Note note : notes) {
338 int bit = 1 << note.pitch.pitchClass();
339 if ((set & bit) == 0) { set |= bit; candidates.add(note); }
340 }
341 Pitch bass = candidates.getFirst().pitch;
342 for (Note root : candidates) {
343 for (int quality = 0; quality < CHORDS.size(); quality++) {
344 Chord chord = CHORDS.get(quality);
345 int mask = 0;
346 for (Tone tone : chord.tones) mask |= 1 << Math.floorMod(root.midi() + tone.semitones, 12);
347 if (mask != set) continue;
348 int inversion = 0;
349 while (Math.floorMod(root.midi() + chord.tones.get(inversion).semitones, 12) != bass.pitchClass()) inversion++;
350 return "IDENTIFY " + symbol(root.pitch, quality, inversion == 0 ? null : bass)
351 + " root=" + root.pitch + " quality=" + chord.name + " inversion=" + inversion;
352 }
353 }
354 return "IDENTIFY UNKNOWN";
355 }
356 static String melody(List<Note> notes) {
357 Note low = notes.getFirst(), high = low;
358 for (Note note : notes) {
359 if (note.midi() < low.midi()) low = note;
360 if (note.midi() > high.midi()) high = note;
361 }
362 Interval range = between(low, high).interval;
363 List<String> steps = new ArrayList<>();
364 int leaps = 0;
365 for (int i = 1; i < notes.size(); i++) {
366 Motion motion = between(notes.get(i - 1), notes.get(i));
367 steps.add((motion.down ? "-" : "+") + motion.interval);
368 if (Math.abs(notes.get(i).midi() - notes.get(i - 1).midi()) >= 3) leaps++;
369 }
370 return "MELODY n=" + notes.size() + " low=" + low + " high=" + high + " range=" + range
371 + " leaps=" + leaps + " steps=" + join(steps, ",");
372 }
373 static String cadence(List<Numeral> numerals) {
374 if (numerals.size() < 2) return "none";
375 Numeral x = numerals.get(numerals.size() - 2), y = numerals.getLast();
376 if (x.alteration == 0 && y.alteration == 0) {
377 if (x.degree == 4 && y.degree == 0) return "authentic";
378 if (x.degree == 3 && y.degree == 0) return "plagal";
379 if (x.degree == 4 && y.degree == 5) return "deceptive";
380 }
381 return y.alteration == 0 && y.degree == 4 ? "half" : "none";
382 }
383 static String diatonic(Key key) {
384 List<String> rows = new ArrayList<>();
385 rows.add("DIATONIC " + key);
386 for (int size = 3; size <= 4; size++) {
387 List<String> items = new ArrayList<>();
388 for (int i = 0; i < 7; i++) {
389 Pitch root = key.scale.get(i);
390 int third = Math.floorMod(key.scale.get((i + 2) % 7).pitchClass() - root.pitchClass(), 12);
391 int fifth = Math.floorMod(key.scale.get((i + 4) % 7).pitchClass() - root.pitchClass(), 12);
392 int quality;
393 if (size == 3) quality = third == 4 ? 0 : fifth == 7 ? 1 : 2;
394 else {
395 int seventh = Math.floorMod(key.scale.get((i + 6) % 7).pitchClass() - root.pitchClass(), 12);
396 quality = third == 4 ? (seventh == 11 ? 9 : 8) : fifth == 7 ? 10 : 12;
397 }
398 items.add(roman(i, quality) + "=" + symbol(root, quality, null));
399 }
400 rows.add((size == 3 ? "TRIADS " : "SEVENTHS ") + join(items, " "));
401 }
402 return join(rows, "\n");
403 }
404 static String keyCommand(String command, List<String> args) {
405 Key key = key(args, 0);
406 return switch (command) {
407 case "KEYSIG" -> {
408 int p = key.fifths;
409 if (p == 0) yield "KEYSIG " + key + " none";
410 List<String> names = List.of((p > 0 ? "F# C# G# D# A# E# B#" : "Bb Eb Ab Db Gb Cb Fb").split(" "));
411 yield "KEYSIG " + key + (p > 0 ? " sharps=" : " flats=") + Math.abs(p) + " " + join(names.subList(0, Math.abs(p)), " ");
412 }
413 case "RELATIVE" -> "RELATIVE " + key + " major=" + fifthPitch(key.fifths) + " minor=" + fifthPitch(key.fifths + 3);
414 case "DEGREE" -> {
415 Pitch pitch = pitch(args.get(2));
416 yield "DEGREE " + key + " " + args.get(2) + " " + key.label(pitch);
417 }
418 case "KEYDIST" -> {
419 Key other = key(args, 2);
420 int distance = Math.floorMod(other.fifths - key.fifths, 12);
421 if (distance > 6) distance -= 12;
422 boolean sameTonic = key.tonic.pitchClass() == other.tonic.pitchClass();
423 String relation;
424 if (key.pitchSet() == other.pitchSet()) relation = sameTonic ? "same" : "relative";
425 else if (sameTonic) relation = "parallel";
426 else relation = Math.abs(distance) == 1 ? "close" : "distant";
427 yield "KEYDIST " + key + " " + other + " fifths=" + (distance > 0 ? "+" : "") + distance
428 + " common=" + Integer.bitCount(key.pitchSet() & other.pitchSet()) + " relation=" + relation;
429 }
430 case "DIATONIC" -> diatonic(key);
431 case "ANALYZE" -> {
432 List<ChordSymbol> symbols = new ArrayList<>();
433 for (String word : args.subList(2, args.size())) symbols.add(chordSymbol(word));
434 List<String> items = new ArrayList<>();
435 for (ChordSymbol chord : symbols) items.add(key.prefix(chord.root) + roman(key.degree(chord.root), chord.quality));
436 yield "ANALYZE " + key + " " + join(items, " ");
437 }
438 case "PROGRESSION" -> {
439 List<Numeral> numerals = new ArrayList<>();
440 for (String word : args.subList(2, args.size())) numerals.add(numeral(word));
441 List<String> rows = new ArrayList<>();
442 rows.add("PROGRESSION " + key + " " + numerals.size());
443 for (int i = 0; i < numerals.size(); i++) {
444 Numeral numeral = numerals.get(i);
445 Pitch degree = key.scale.get(numeral.degree);
446 Pitch root = new Pitch(degree.letter, degree.alteration + numeral.alteration);
447 List<Pitch> tones = build(root, CHORDS.get(numeral.quality).tones);
448 rows.add(args.get(i + 2) + " " + symbol(root, numeral.quality, null) + " " + join(tones, " "));
449 }
450 rows.add("CADENCE " + cadence(numerals));
451 yield join(rows, "\n");
452 }
453 default -> throw new CommandError("UNKNOWN COMMAND " + command);
454 };
455 }

459 String execute(String line) {
460 List<String> words = Arrays.asList(line.split(" "));
461 String command = words.getFirst();
462 List<String> args = words.subList(1, words.size());
463 int minimum = switch (command) {
464 case "MIDI", "NOTE", "FREQ", "TUNING", "PITCH", "INVERT", "ENHARMONIC", "CIRCLE", "FINDKEY" -> 1;
465 case "INTERVAL", "SCALE", "CHORD", "VOICE", "IDENTIFY", "MELODY", "KEYSIG", "RELATIVE", "DIATONIC" -> 2;
466 case "TRANSPOSE", "PROGRESSION", "ANALYZE", "DEGREE", "SPELL" -> 3;
467 case "KEYDIST" -> 4;
468 default -> throw new CommandError("UNKNOWN COMMAND " + command);
469 };
470 int maximum = switch (command) {
471 case "CHORD", "VOICE" -> 3;
472 case "IDENTIFY", "MELODY", "PROGRESSION", "ANALYZE", "FINDKEY" -> 15;
473 default -> minimum;
474 };
475 if (args.size() < minimum || args.size() > maximum) throw new CommandError("ARGUMENT COUNT");
476 return switch (command) {
477 case "MIDI", "FREQ", "ENHARMONIC" -> {
478 Note note = note(args.getFirst());
479 int value = note.midi();
480 if (command.equals("MIDI")) yield "MIDI " + args.getFirst() + " " + value;
481 if (command.equals("FREQ")) {
482 int cents = (int) Math.floor(tuning * Math.pow(2, (value - 69) / 12.0) + 0.5);
483 yield "FREQ " + args.getFirst() + " " + frequencyText(cents);
484 }
485 List<Note> equivalents = new ArrayList<>();
486 for (int octave = -1; octave <= 9; octave++) {
487 for (int letter = 0; letter < 7; letter++) {
488 int alteration = value - 12 * (octave + 1) - NATURALS[letter];
489 if (alteration >= -2 && alteration <= 2) equivalents.add(new Note(new Pitch(letter, alteration), octave));
490 }
491 }
492 yield "ENHARMONIC " + args.getFirst() + " " + join(equivalents, " ");
493 }
494 case "NOTE" -> {
495 int value = number(args.getFirst());
496 yield "NOTE " + value + " " + spelling(value, false) + " " + spelling(value, true);
497 }
498 case "TUNING", "PITCH" -> {
499 int value = frequency(args.getFirst(), command.equals("TUNING"));
500 if (command.equals("TUNING")) { tuning = value; yield "TUNING " + frequencyText(value); }
501 double x = 69 + 12 * Math.log((double) value / tuning) / Math.log(2);
502 int nearest = (int) Math.floor(x + 0.5);
503 if (nearest < 0 || nearest > 127) throw new CommandError("OUT OF RANGE");
504 int cents = (int) Math.floor(100 * (x - nearest) + 0.5);
505 yield "PITCH " + frequencyText(value) + " " + spelling(nearest, false) + " " + (cents >= 0 ? "+" : "") + cents;
506 }
507 case "INTERVAL" -> {
508 Note first = note(args.get(0)), second = note(args.get(1));
509 Motion motion = between(first, second);
510 yield "INTERVAL " + args.get(0) + " " + args.get(1) + " " + motion.interval + (motion.down ? " down" : " up")
511 + " semitones=" + motion.semitones + " simple=" + motion.interval.simple() + " class=" + motion.interval.consonance();
512 }
513 case "INVERT" -> {
514 Interval simple = interval(args.getFirst()).simple();
515 Interval inverse = new Interval(simple.quality.inverse(), 9 - simple.number);
516 if (inverse.forbidden()) throw new CommandError("CANNOT INVERT");
517 yield "INVERT " + args.getFirst() + " " + simple + " " + inverse;
518 }
519 case "TRANSPOSE" -> {
520 Note note = note(args.get(0));
521 Interval interval = interval(args.get(1));
522 String direction = args.get(2);
523 if (!direction.equals("UP") && !direction.equals("DOWN")) throw bad("DIRECTION", direction);
524 int sign = direction.equals("UP") ? 1 : -1;
525 Tone tone = interval.tone();
526 int position = note.position() + sign * tone.letters, value = note.midi() + sign * tone.semitones;
527 int letter = Math.floorMod(position, 7), octave = Math.floorDiv(position, 7);
528 int alteration = value - 12 * (octave + 1) - NATURALS[letter];
529 if (alteration < -2 || alteration > 2) throw new CommandError("CANNOT SPELL");
530 Note result = new Note(new Pitch(letter, alteration), octave);
531 if (!result.valid()) throw new CommandError("OUT OF RANGE");
532 yield "TRANSPOSE " + args.get(0) + " " + args.get(1) + " " + direction + " " + result;
533 }
534 case "SCALE" -> {
535 Pitch root = pitch(args.get(0));
536 Scale scale = scale(args.get(1));
537 List<Pitch> pitches = build(root, scale.tones);
538 List<Integer> steps = new ArrayList<>();
539 for (int i = 0; i < scale.tones.size(); i++) {
540 int next = i + 1 < scale.tones.size() ? scale.tones.get(i + 1).semitones : 12;
541 steps.add(next - scale.tones.get(i).semitones);
542 }
543 yield "SCALE " + args.get(0) + " " + args.get(1) + " " + join(pitches, " ") + " steps=" + join(steps, ",");
544 }
545 case "CHORD", "VOICE" -> chordCommand(command, args);
546 case "IDENTIFY" -> identify(notes(args));
547 case "MELODY" -> melody(notes(args));
548 case "CIRCLE" -> {
549 int mode = mode(args.getFirst());
550 List<Pitch> pitches = new ArrayList<>();
551 for (int p = -7; p <= 7; p++) pitches.add(fifthPitch(p - MODES.get(mode).offset));
552 yield "CIRCLE " + args.getFirst() + " " + join(pitches, " ");
553 }
554 case "FINDKEY" -> {
555 List<Pitch> pitches = new ArrayList<>();
556 for (String word : args) pitches.add(pitch(word));
557 List<Key> keys = new ArrayList<>();
558 for (int p = -7; p <= 7; p++) {
559 for (int mode = 0; mode < 2; mode++) {
560 Key key = key(fifthPitch(p - MODES.get(mode).offset), mode);
561 if (key.scale.containsAll(pitches)) keys.add(key);
562 }
563 }
564 yield "FINDKEY " + keys.size() + (keys.isEmpty() ? "" : " " + join(keys, ", "));
565 }
566 case "SPELL" -> {
567 int value = number(args.getFirst());
568 Key key = key(args, 1);
569 Pitch pitch = spelling(value, key.fifths < 0).pitch;
570 for (Pitch candidate : key.scale) if (candidate.pitchClass() == value % 12) { pitch = candidate; break; }
571 Note note = fromPitch(pitch, value);
572 if (!note.valid()) throw new CommandError("OUT OF RANGE");
573 yield "SPELL " + value + " " + key + " " + note + " " + key.label(pitch);
574 }
575 default -> keyCommand(command, args);
576 };
577 }

579 public static void main(String[] args) throws IOException {
580 BufferedReader input = new BufferedReader(new InputStreamReader(System.in));
581 BufferedWriter output = new BufferedWriter(new OutputStreamWriter(System.out));
582 int count = Integer.parseInt(input.readLine());
583 Main calculator = new Main();
584 for (int i = 0; i < count; i++) {
585 String response;
586 try { response = calculator.execute(input.readLine()); }
587 catch (CommandError error) { response = error.getMessage(); }
588 output.write(response);
589 output.write('\n');
590 }
591 output.flush();
592 }
