3class CommandError extends Error {}
4function fail(message) { throw new CommandError(message); }
5function bad(kind, text) { fail(`BAD ${kind} ${text}`); }
6const mod = (value, divisor) => (value % divisor + divisor) % divisor;
7const LETTERS = 'CDEFGAB';
8const NATURALS = [0, 2, 4, 5, 7, 9, 11];
9const FIFTHS = [0, 2, 4, -1, 1, 3, 5];
10const ACCIDENTALS = ['bb', 'b', '', '#', 'x'];
11const SHARPS = 'C C# D D# E F F# G G# A A# B'.split(' ');
12const FLATS = 'C Db D Eb E F Gb G Ab A Bb B'.split(' ');
13const ROMANS = ['I', 'II', 'III', 'IV', 'V', 'VI', 'VII'];

15function integer(text, kind, low, high) {
16 if (!/^(0|[1-9][0-9]*)$/.test(text) || text.length > 3) bad(kind, text);
17 const value = Number(text);
18 if (value < low || value > high) bad(kind, text);
19 return value;
20}
21function frequency(text, tuning) {
22 if (!/^[0-9]{1,5}(\.[0-9]{1,2})?$/.test(text)) bad('FREQUENCY', text);
23 const [whole, digits = ''] = text.split('.');
24 const cents = Number(whole) * 100 + Number(digits.padEnd(2, '0'));
25 if (cents === 0 || (tuning && (cents < 40000 || cents > 48000))) bad('FREQUENCY', text);
26 return cents;
27}
28const frequencyText = cents => `${Math.trunc(cents / 100)}.${String(cents % 100).padStart(2, '0')}`;

30class Pitch {
31 constructor(letter, alteration) { this.letter = letter; this.alteration = alteration; }
32 get pitchClass() { return mod(NATURALS[this.letter] + this.alteration, 12); }
33 equals(other) { return this.letter === other.letter && this.alteration === other.alteration; }
34 toString() { return LETTERS[this.letter] + ACCIDENTALS[this.alteration + 2]; }
35 raise(tone) {
36 const target = this.letter + tone.letters;
37 const alteration = NATURALS[this.letter] + this.alteration + tone.semitones - NATURALS[mod(target, 7)] - 12 * Math.floor(target / 7);
38 if (alteration < -2 || alteration > 2) fail('CANNOT SPELL');
39 return new Pitch(mod(target, 7), alteration);
40 }
41}
42function pitchPrefix(text, kind) {
43 const match = /^([A-G])(bb|b|#|x)?/.exec(text);
44 if (!match) bad(kind, text);
45 return { pitch: new Pitch(LETTERS.indexOf(match[1]), ACCIDENTALS.indexOf(match[2] ?? '') - 2), rest: text.slice(match[0].length) };
46}
47function parsePitch(text) {
48 const result = pitchPrefix(text, 'PITCH');
49 if (result.rest) bad('PITCH', text);
50 return result.pitch;
51}
52function fifthPitch(value) {
53 return new Pitch([3, 0, 4, 1, 5, 2, 6][mod(value + 1, 7)], Math.floor((value + 1) / 7));
54}
55class Note {
56 constructor(pitch, octave) { this.pitch = pitch; this.octave = octave; }
57 get midi() { return 12 * (this.octave + 1) + NATURALS[this.pitch.letter] + this.pitch.alteration; }
58 get position() { return 7 * this.octave + this.pitch.letter; }
59 get valid() { return this.octave >= -1 && this.octave <= 9 && this.midi >= 0 && this.midi <= 127; }
60 toString() { return `${this.pitch}${this.octave}`; }
61}
62function parseNote(text) {
63 const parsed = pitchPrefix(text, 'NOTE');
64 if (!/^(-1|[0-9])$/.test(parsed.rest)) bad('NOTE', text);
65 const note = new Note(parsed.pitch, Number(parsed.rest));
66 if (!note.valid) bad('NOTE', text);
67 return note;
68}
69const makeNote = (pitch, midi) => new Note(pitch, (midi - NATURALS[pitch.letter] - pitch.alteration) / 12 - 1);
70const spelling = (midi, flat = false) => new Note(parsePitch((flat ? FLATS : SHARPS)[mod(midi, 12)]), Math.floor(midi / 12) - 1);

72const INTERVAL_QUALITIES = ['P', 'M', 'm', 'A', 'd', 'AA', 'dd'];
73const PERFECT = { P: 0, A: 1, d: -1, AA: 2, dd: -2 };
74const MAJOR = { M: 0, m: -1, A: 1, d: -2, AA: 2, dd: -3 };
75const qualityDelta = (quality, number) => ([0, 3, 4].includes((number - 1) % 7) ? PERFECT : MAJOR)[quality];
76class Interval {
77 constructor(quality, number) { this.quality = quality; this.number = number; }
78 toString() { return this.quality + this.number; }
79 get forbidden() { return this.number === 1 && (this.quality === 'd' || this.quality === 'dd'); }
80 get tone() {
81 const letters = this.number - 1;
82 return { letters, semitones: 12 * Math.floor(letters / 7) + NATURALS[letters % 7] + qualityDelta(this.quality, this.number) };
83 }
84 get simple() {
85 let number = this.number;
86 while (number > 8) number -= 7;
87 return new Interval(this.quality, number);
88 }
89 get consonance() {
90 const simple = this.simple;
91 if (simple.quality === 'P' && [1, 5, 8].includes(simple.number)) return 'perfect';
92 if (['m', 'M'].includes(simple.quality) && [3, 6].includes(simple.number)) return 'imperfect';
93 return 'dissonant';
94 }
95}
96function parseInterval(text) {
97 const match = /^(AA|dd|P|M|m|A|d)([1-9][0-9]?)$/.exec(text);
98 if (!match) bad('INTERVAL', text);
99 const interval = new Interval(match[1], Number(match[2]));
100 if (qualityDelta(interval.quality, interval.number) === undefined || interval.forbidden) bad('INTERVAL', text);
101 return interval;
102}
103function between(first, second) {
104 let distance = second.position - first.position, semitones = second.midi - first.midi;
105 const down = distance < 0 || (distance === 0 && semitones < 0);
106 if (down) { distance = -distance; semitones = -semitones; }
107 const difference = semitones - 12 * Math.floor(distance / 7) - NATURALS[distance % 7];
108 const quality = INTERVAL_QUALITIES.find(item => qualityDelta(item, distance + 1) === difference);
109 if (quality === undefined) fail('UNNAMEABLE INTERVAL');
110 return { interval: new Interval(quality, distance + 1), down, semitones };
111}
112const tones = pattern => pattern.split(' ').map(text => parseInterval(text).tone);
113const build = (root, pattern) => pattern.map(tone => root.raise(tone));

115const SCALES = Object.entries({
116 major: 'P1 M2 M3 P4 P5 M6 M7', natural_minor: 'P1 M2 m3 P4 P5 m6 m7',
117 harmonic_minor: 'P1 M2 m3 P4 P5 m6 M7', melodic_minor: 'P1 M2 m3 P4 P5 M6 M7',
118 ionian: 'P1 M2 M3 P4 P5 M6 M7', dorian: 'P1 M2 m3 P4 P5 M6 m7',
119 phrygian: 'P1 m2 m3 P4 P5 m6 m7', lydian: 'P1 M2 M3 A4 P5 M6 M7',
120 mixolydian: 'P1 M2 M3 P4 P5 M6 m7', aeolian: 'P1 M2 m3 P4 P5 m6 m7',
121 locrian: 'P1 m2 m3 P4 d5 m6 m7', major_pentatonic: 'P1 M2 M3 P5 M6',
122 minor_pentatonic: 'P1 m3 P4 P5 m7', blues: 'P1 m3 P4 A4 P5 m7',
123 whole_tone: 'P1 M2 M3 A4 A5 A6',
124}).map(([name, pattern]) => ({ name, tones: tones(pattern) }));
125class Chord {
126 constructor(name, suffix, numeral, pattern) {
127 this.name = name;
128 this.suffix = suffix;
129 this.lower = numeral[0] === 'i';
130 this.numeralSuffix = numeral.slice(1);
131 this.tones = tones(pattern);
132 }
133 symbol(root, bass = null) { return `${root}${this.suffix}${bass === null ? '' : '/' + bass}`; }
134 roman(degree) { return (this.lower ? ROMANS[degree].toLowerCase() : ROMANS[degree]) + this.numeralSuffix; }
135}
136const CHORDS = [
137 new Chord('maj', '', 'I', 'P1 M3 P5'), new Chord('min', 'm', 'i', 'P1 m3 P5'),
138 new Chord('dim', 'dim', 'io', 'P1 m3 d5'), new Chord('aug', 'aug', 'I+', 'P1 M3 A5'),
139 new Chord('sus2', 'sus2', 'Isus2', 'P1 M2 P5'), new Chord('sus4', 'sus4', 'Isus4', 'P1 P4 P5'),
140 new Chord('6', '6', 'I6', 'P1 M3 P5 M6'), new Chord('m6', 'm6', 'i6', 'P1 m3 P5 M6'),
141 new Chord('7', '7', 'I7', 'P1 M3 P5 m7'), new Chord('maj7', 'maj7', 'Imaj7', 'P1 M3 P5 M7'),
142 new Chord('m7', 'm7', 'i7', 'P1 m3 P5 m7'), new Chord('mMaj7', 'mMaj7', 'imaj7', 'P1 m3 P5 M7'),
143 new Chord('m7b5', 'm7b5', 'ih7', 'P1 m3 d5 m7'), new Chord('dim7', 'dim7', 'io7', 'P1 m3 d5 d7'),
144 new Chord('9', '9', 'I9', 'P1 M3 P5 m7 M9'), new Chord('add9', 'add9', 'Iadd9', 'P1 M3 P5 M9'),
145];
146const MODES = [
147 ['major', 'major', 0], ['minor', 'natural_minor', -3], ['ionian', 'ionian', 0],
148 ['dorian', 'dorian', -2], ['phrygian', 'phrygian', -4], ['lydian', 'lydian', 1],
149 ['mixolydian', 'mixolydian', -1], ['aeolian', 'aeolian', -3], ['locrian', 'locrian', -5],
150].map(([name, scale, offset]) => ({ name, scale, offset }));
151function named(text, kind, choices) { return choices.find(item => item.name === text) ?? bad(kind, text); }
152class Key {
153 constructor(tonic, mode) {
154 this.tonic = tonic;
155 this.mode = mode;
156 this.fifths = FIFTHS[tonic.letter] + 7 * tonic.alteration + mode.offset;
157 if (this.fifths < -7 || this.fifths > 7) fail('THEORETICAL KEY');
158 this.scale = build(tonic, named(mode.scale, 'SCALE', SCALES).tones);
159 }
160 toString() { return `${this.tonic} ${this.mode.name}`; }
161 degree(pitch) { return mod(pitch.letter - this.tonic.letter, 7); }
162 prefix(pitch) {
163 const difference = pitch.alteration - this.scale[this.degree(pitch)].alteration;
164 return (difference < 0 ? 'b' : '#').repeat(Math.abs(difference));
165 }
166 label(pitch) { return this.prefix(pitch) + (this.degree(pitch) + 1); }
167 get pitchClasses() { return new Set(this.scale.map(pitch => pitch.pitchClass)); }
168}
169function parseKey(args, start = 0) {
170 const tonic = parsePitch(args[start]), mode = named(args[start + 1], 'MODE', MODES);
171 return new Key(tonic, mode);
172}
173function parseNumeral(text) {
174 const start = text[0] === 'b' || text[0] === '#' ? 1 : 0;
175 const alteration = start === 0 ? 0 : text[0] === 'b' ? -1 : 1;
176 let end = start;
177 while (end < text.length && 'IViv'.includes(text[end])) end++;
178 const part = text.slice(start, end), suffix = text.slice(end);
179 for (let degree = 0; degree < 7; degree++) {
180 for (const chord of CHORDS) {
181 const expected = chord.lower ? ROMANS[degree].toLowerCase() : ROMANS[degree];
182 if (part === expected && suffix === chord.numeralSuffix) return { alteration, degree, chord };
183 }
184 }
185 bad('NUMERAL', text);
186}
187function parseSymbol(text) {
188 const parsed = pitchPrefix(text, 'SYMBOL'), chord = CHORDS.find(item => item.suffix === parsed.rest);
189 if (!chord) bad('SYMBOL', text);
190 return { root: parsed.pitch, chord };
191}
192const equalSets = (a, b) => a.size === b.size && [...a].every(value => b.has(value));

194function identify(notes) {
195 const ordered = notes.map((note, index) => ({ note, index })).sort((a, b) => a.note.midi - b.note.midi || a.index - b.index);
196 const present = new Set(), candidates = [];
197 for (const { note } of ordered) {
198 if (present.has(note.pitch.pitchClass)) continue;
199 present.add(note.pitch.pitchClass);
200 candidates.push(note);
201 }
202 const bass = candidates[0].pitch;
203 for (const root of candidates) {
204 for (const chord of CHORDS) {
205 const classes = chord.tones.map(tone => mod(root.midi + tone.semitones, 12));
206 if (!equalSets(new Set(classes), present)) continue;
207 const inversion = classes.indexOf(bass.pitchClass);
208 return `IDENTIFY ${chord.symbol(root.pitch, inversion === 0 ? null : bass)} root=${root.pitch} quality=${chord.name} inversion=${inversion}`;
209 }
210 }
211 return 'IDENTIFY UNKNOWN';
212}
213function melody(notes, args) {
214 const values = notes.map(note => note.midi), low = values.indexOf(Math.min(...values)), high = values.indexOf(Math.max(...values));
215 const range = between(notes[low], notes[high]).interval, steps = [];
216 let leaps = 0;
217 for (let i = 1; i < notes.length; i++) {
218 const motion = between(notes[i - 1], notes[i]);
219 steps.push(`${motion.down ? '-' : '+'}${motion.interval}`);
220 if (Math.abs(values[i] - values[i - 1]) >= 3) leaps++;
221 }
222 return `MELODY n=${notes.length} low=${args[low]} high=${args[high]} range=${range} leaps=${leaps} steps=${steps.join(',')}`;
223}
224function cadence(numerals) {
225 if (numerals.length < 2) return 'none';
226 const first = numerals.at(-2), last = numerals.at(-1);
227 if (first.alteration === 0 && last.alteration === 0) {
228 if (first.degree === 4 && last.degree === 0) return 'authentic';
229 if (first.degree === 3 && last.degree === 0) return 'plagal';
230 if (first.degree === 4 && last.degree === 5) return 'deceptive';
231 }
232 return last.alteration === 0 && last.degree === 4 ? 'half' : 'none';
233}
234function diatonic(key) {
235 const rows = [`DIATONIC ${key}`], classes = key.scale.map(pitch => pitch.pitchClass);
236 for (const size of [3, 4]) {
237 const items = [];
238 for (let degree = 0; degree < 7; degree++) {
239 const third = mod(classes[(degree + 2) % 7] - classes[degree], 12), fifth = mod(classes[(degree + 4) % 7] - classes[degree], 12);
240 let quality;
241 if (size === 3) quality = third === 4 ? 'maj' : fifth === 7 ? 'min' : 'dim';
242 else {
243 const seventh = mod(classes[(degree + 6) % 7] - classes[degree], 12);
244 quality = third === 4 ? (seventh === 11 ? 'maj7' : '7') : fifth === 7 ? 'm7' : 'm7b5';
245 }
246 const chord = named(quality, 'QUALITY', CHORDS);
247 items.push(`${chord.roman(degree)}=${chord.symbol(key.scale[degree])}`);
248 }
249 rows.push(`${size === 3 ? 'TRIADS' : 'SEVENTHS'} ${items.join(' ')}`);
250 }
251 return rows;
252}

254function executeKey(command, args) {
255 const key = parseKey(args);
256 switch (command) {
257 case 'KEYSIG': {
258 if (key.fifths === 0) return [`KEYSIG ${key} none`];
259 const names = (key.fifths > 0 ? 'F# C# G# D# A# E# B#' : 'Bb Eb Ab Db Gb Cb Fb').split(' ');
260 return [`KEYSIG ${key} ${key.fifths > 0 ? 'sharps' : 'flats'}=${Math.abs(key.fifths)} ${names.slice(0, Math.abs(key.fifths)).join(' ')}`];
261 }
262 case 'RELATIVE': return [`RELATIVE ${key} major=${fifthPitch(key.fifths)} minor=${fifthPitch(key.fifths + 3)}`];
263 case 'DIATONIC': return diatonic(key);
264 case 'DEGREE': {
265 const pitch = parsePitch(args[2]);
266 return [`DEGREE ${key} ${args[2]} ${key.label(pitch)}`];
267 }
268 case 'KEYDIST': {
269 const other = parseKey(args, 2);
270 let distance = mod(other.fifths - key.fifths, 12);
271 if (distance > 6) distance -= 12;
272 const first = key.pitchClasses, second = other.pitchClasses, sameTonic = key.tonic.pitchClass === other.tonic.pitchClass;
273 const relation = equalSets(first, second) ? (sameTonic ? 'same' : 'relative') : sameTonic ? 'parallel' : Math.abs(distance) === 1 ? 'close' : 'distant';
274 const common = [...first].filter(value => second.has(value)).length;
275 return [`KEYDIST ${key} ${other} fifths=${distance > 0 ? '+' : ''}${distance} common=${common} relation=${relation}`];
276 }
277 case 'ANALYZE': {
278 const symbols = args.slice(2).map(parseSymbol);
279 return [`ANALYZE ${key} ${symbols.map(symbol => key.prefix(symbol.root) + symbol.chord.roman(key.degree(symbol.root))).join(' ')}`];
280 }
281 case 'PROGRESSION': {
282 const numerals = args.slice(2).map(parseNumeral), rows = [`PROGRESSION ${key} ${numerals.length}`];
283 for (const [index, numeral] of numerals.entries()) {
284 const degree = key.scale[numeral.degree], root = new Pitch(degree.letter, degree.alteration + numeral.alteration);
285 const pitches = build(root, numeral.chord.tones);
286 rows.push(`${args[index + 2]} ${numeral.chord.symbol(root)} ${pitches.join(' ')}`);
287 }
288 rows.push(`CADENCE ${cadence(numerals)}`);
289 return rows;
290 }
291 }

294const ARITIES = {
295 MIDI: [1,1], NOTE: [1,1], FREQ: [1,1], TUNING: [1,1], PITCH: [1,1], INTERVAL: [2,2],
296 INVERT: [1,1], TRANSPOSE: [3,3], SCALE: [2,2], CHORD: [2,3], VOICE: [2,3],
297 IDENTIFY: [2,15], MELODY: [2,15], ENHARMONIC: [1,1], KEYSIG: [2,2], RELATIVE: [2,2],
298 CIRCLE: [1,1], DIATONIC: [2,2], PROGRESSION: [3,15], ANALYZE: [3,15], FINDKEY: [1,15],
299 DEGREE: [3,3], SPELL: [3,3], KEYDIST: [4,4],
300};
301class Calculator {
302 #tuning = 44000;
303
304 execute(line) {
305 const words = line.split(' '), command = words[0];
306 if (!Object.hasOwn(ARITIES, command)) fail(`UNKNOWN COMMAND ${command}`);
307 const args = words.slice(1), [low, high] = ARITIES[command];
308 if (args.length < low || args.length > high) fail('ARGUMENT COUNT');
309 switch (command) {
310 case 'MIDI': case 'FREQ': case 'ENHARMONIC': {
311 const note = parseNote(args[0]), value = note.midi;
312 if (command === 'MIDI') return [`MIDI ${args[0]} ${value}`];
313 if (command === 'FREQ') return [`FREQ ${args[0]} ${frequencyText(Math.floor(this.#tuning * 2 ** ((value - 69) / 12) + 0.5))}`];
314 const equivalents = [];
315 for (let octave = -1; octave <= 9; octave++) {
316 for (let letter = 0; letter < 7; letter++) {
317 const alteration = value - 12 * (octave + 1) - NATURALS[letter];
318 if (alteration >= -2 && alteration <= 2) equivalents.push(new Note(new Pitch(letter, alteration), octave));
319 }
320 }
321 return [`ENHARMONIC ${args[0]} ${equivalents.join(' ')}`];
322 }
323 case 'NOTE': {
324 const value = integer(args[0], 'NUMBER', 0, 127);
325 return [`NOTE ${value} ${spelling(value)} ${spelling(value, true)}`];
326 }
327 case 'TUNING': case 'PITCH': {
328 const value = frequency(args[0], command === 'TUNING');
329 if (command === 'TUNING') {
330 this.#tuning = value;
331 return [`TUNING ${frequencyText(value)}`];
332 }
333 const height = 69 + 12 * Math.log2(value / this.#tuning), nearest = Math.floor(height + 0.5);
334 if (nearest < 0 || nearest > 127) fail('OUT OF RANGE');
335 const cents = Math.floor(100 * (height - nearest) + 0.5);
336 return [`PITCH ${frequencyText(value)} ${spelling(nearest)} ${cents >= 0 ? '+' : ''}${cents}`];
337 }
338 case 'INTERVAL': {
339 const first = parseNote(args[0]), second = parseNote(args[1]), motion = between(first, second);
340 return [`INTERVAL ${args[0]} ${args[1]} ${motion.interval} ${motion.down ? 'down' : 'up'} semitones=${motion.semitones} simple=${motion.interval.simple} class=${motion.interval.consonance}`];
341 }
342 case 'INVERT': {
343 const interval = parseInterval(args[0]), simple = interval.simple;
344 const inverses = { P: 'P', M: 'm', m: 'M', A: 'd', d: 'A', AA: 'dd', dd: 'AA' };
345 const inverse = new Interval(inverses[simple.quality], 9 - simple.number);
346 if (inverse.forbidden) fail('CANNOT INVERT');
347 return [`INVERT ${args[0]} ${simple} ${inverse}`];
348 }
349 case 'TRANSPOSE': {
350 const note = parseNote(args[0]), interval = parseInterval(args[1]), direction = args[2];
351 if (direction !== 'UP' && direction !== 'DOWN') bad('DIRECTION', direction);
352 const sign = direction === 'UP' ? 1 : -1, tone = interval.tone;
353 const target = note.position + sign * tone.letters, value = note.midi + sign * tone.semitones;
354 const letter = mod(target, 7), octave = Math.floor(target / 7), alteration = value - 12 * (octave + 1) - NATURALS[letter];
355 if (alteration < -2 || alteration > 2) fail('CANNOT SPELL');
356 const result = new Note(new Pitch(letter, alteration), octave);
357 if (!result.valid) fail('OUT OF RANGE');
358 return [`TRANSPOSE ${args[0]} ${args[1]} ${direction} ${result}`];
359 }
360 case 'SCALE': {
361 const root = parsePitch(args[0]), scale = named(args[1], 'SCALE', SCALES), pitches = build(root, scale.tones);
362 const semitones = [...scale.tones.map(tone => tone.semitones), 12];
363 const steps = scale.tones.map((_, i) => semitones[i + 1] - semitones[i]);
364 return [`SCALE ${args[0]} ${args[1]} ${pitches.join(' ')} steps=${steps.join(',')}`];
365 }
366 case 'CHORD': case 'VOICE': {
367 const input = command === 'VOICE' ? parseNote(args[0]) : null, root = input?.pitch ?? parsePitch(args[0]);
368 const chord = named(args[1], 'QUALITY', CHORDS);
369 const inversion = args.length === 3 ? integer(args[2], 'INVERSION', 0, chord.tones.length - 1) : 0;
370 const pitches = build(root, chord.tones), ordered = pitches.slice(inversion).concat(pitches.slice(0, inversion));
371 const symbol = chord.symbol(root, inversion === 0 ? null : ordered[0]);
372 if (command === 'CHORD') return [`CHORD ${symbol} ${ordered.join(' ')}`];
373 let value = input.midi;
374 const notes = [];
375 for (const [index, pitch] of ordered.entries()) {
376 if (index > 0 || inversion > 0) value += mod(pitch.pitchClass - value, 12) || 12;
377 const note = makeNote(pitch, value);
378 if (!note.valid) fail('OUT OF RANGE');
379 notes.push(note);
380 }
381 return [`VOICE ${symbol} ${notes.join(' ')} midi=${notes.map(note => note.midi).join(',')}`];
382 }
383 case 'IDENTIFY': case 'MELODY': {
384 const notes = args.map(parseNote);
385 return [command === 'IDENTIFY' ? identify(notes) : melody(notes, args)];
386 }
387 case 'CIRCLE': {
388 const mode = named(args[0], 'MODE', MODES);
389 const pitches = Array.from({ length: 15 }, (_, index) => fifthPitch(index - 7 - mode.offset));
390 return [`CIRCLE ${args[0]} ${pitches.join(' ')}`];
391 }
392 case 'FINDKEY': {
393 const pitches = args.map(parsePitch), keys = [];
394 for (let fifths = -7; fifths <= 7; fifths++) {
395 for (const mode of MODES.slice(0, 2)) {
396 const key = new Key(fifthPitch(fifths - mode.offset), mode);
397 if (pitches.every(pitch => key.scale.some(candidate => candidate.equals(pitch)))) keys.push(key);
398 }
399 }
400 return [`FINDKEY ${keys.length}${keys.length === 0 ? '' : ' ' + keys.join(', ')}`];
401 }
402 case 'SPELL': {
403 const value = integer(args[0], 'NUMBER', 0, 127), key = parseKey(args, 1);
404 const pitch = key.scale.find(item => item.pitchClass === value % 12) ?? spelling(value, key.fifths < 0).pitch;
405 const note = makeNote(pitch, value);
406 if (!note.valid) fail('OUT OF RANGE');
407 return [`SPELL ${value} ${key} ${note} ${key.label(pitch)}`];
408 }
409 default: return executeKey(command, args);
410 }
411 }
412}
