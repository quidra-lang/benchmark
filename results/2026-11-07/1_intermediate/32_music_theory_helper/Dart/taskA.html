12const naturals = [0, 2, 4, 5, 7, 9, 11];

16final class Pitch {
17 final int letter;
18 final int alteration;
19 const Pitch(this.letter, this.alteration);
20 int get natural => naturals[letter];
21 int get pitchClass => (natural + alteration) % 12;
22 @override
23 bool operator ==(Object other) => other is Pitch && letter == other.letter && alteration == other.alteration;
24 @override
25 int get hashCode => Object.hash(letter, alteration);
26 @override
27 String toString() => '${letters[letter]}${accidentals[alteration + 2]}';
28}

30final class Note {
31 final Pitch pitch;
32 final int octave;
33 const Note(this.pitch, this.octave);
34 int get midi => 12 * (octave + 1) + pitch.natural + pitch.alteration;
35 int get position => 7 * octave + pitch.letter;
36 bool get valid => octave >= -1 && octave <= 9 && midi >= 0 && midi <= 127;
37 @override
38 String toString() => '$pitch$octave';
39}

41final class Interval {
42 final String quality;
43 final int number;
44 const Interval(this.quality, this.number);
45 int get letterDistance => number - 1;
46 bool get perfectType => const {0, 3, 4}.contains(letterDistance % 7);
47 int get semitones {
48 final difference = switch (quality) {
49 'dd' => perfectType ? -2 : -3,
50 'd' => perfectType ? -1 : -2,
51 'm' => -1,
52 'A' => 1,
53 'AA' => 2,
54 _ => 0,
55 };
56 return 12 * (letterDistance ~/ 7) + naturals[letterDistance % 7] + difference;
57 }
58 Interval get simple {
59 var n = number;
60 while (n > 8) {
61 n -= 7;
62 }
63 return Interval(quality, n);
64 }
65 String get consonance => switch (simple.toString()) {
66 'P1' || 'P5' || 'P8' => 'perfect',
67 'm3' || 'M3' || 'm6' || 'M6' => 'imperfect',
68 _ => 'dissonant',
69 };
70 @override
71 String toString() => '$quality$number';
72}

170int alteration(String? text) => switch (text) {'bb' => -2, 'b' => -1, '#' => 1, 'x' => 2, _ => 0};
171Pitch parsePitch(String text) {
172 final match = RegExp(r'^([A-G])(bb|b|#|x)?$').firstMatch(text);
173 if (match == null) throw CommandError('BAD PITCH $text');
174 return Pitch(letters.indexOf(match[1]!), alteration(match[2]));
175}
176Note parseNote(String text) {
177 final match = RegExp(r'^([A-G])(bb|b|#|x)?(-1|[0-9])$').firstMatch(text);
178 if (match == null) throw CommandError('BAD NOTE $text');
179 final note = Note(Pitch(letters.indexOf(match[1]!), alteration(match[2])), int.parse(match[3]!));
180 if (!note.valid) throw CommandError('BAD NOTE $text');
181 return note;
182}
183int parseNumber(String text) {
184 final value = int.tryParse(text);
185 if (!RegExp(r'^(0|[1-9][0-9]*)$').hasMatch(text) || value == null || value > 127) {
186 throw CommandError('BAD NUMBER $text');
187 }
188 return value;
189}
190int parseFrequency(String text, bool tuning) {
191 final match = RegExp(r'^([0-9]{1,5})(?:\.([0-9]{1,2}))?$').firstMatch(text);
192 if (match == null) throw CommandError('BAD FREQUENCY $text');
193 var cents = int.parse(match[1]!) * 100;
194 final tail = match[2];
195 if (tail != null) cents += int.parse(tail.padRight(2, '0'));
196 if (tuning ? cents < 40000 || cents > 48000 : cents == 0) throw CommandError('BAD FREQUENCY $text');
197 return cents;
198}
199Interval parseInterval(String text) {
200 final match = RegExp(r'^(AA|dd|P|M|m|A|d)([1-9][0-9]?)$').firstMatch(text);
201 if (match == null) throw CommandError('BAD INTERVAL $text');
202 final interval = Interval(match[1]!, int.parse(match[2]!));
203 if ((interval.perfectType ? const {'M', 'm'}.contains(interval.quality) : interval.quality == 'P') ||
204 interval.number == 1 && const {'d', 'dd'}.contains(interval.quality)) {
205 throw CommandError('BAD INTERVAL $text');
206 }
207 return interval;
208}
209Quality parseQuality(String text) {
210 for (final quality in qualities) {
211 if (quality.name == text) return quality;
212 }
213 throw CommandError('BAD QUALITY $text');
214}
215Mode parseMode(String text) {
216 for (final mode in modes) {
217 if (mode.name == text) return mode;
218 }
219 throw CommandError('BAD MODE $text');
220}
221Key parseKey(String tonic, String mode) {
222 final pitch = parsePitch(tonic);
223 final parsedMode = parseMode(mode);
224 final key = Key(pitch, parsedMode);
225 if (key.fifths < -7 || key.fifths > 7) throw const CommandError('THEORETICAL KEY');
226 return key;
227}
228List<Interval> parseScale(String text) => scales[text] ?? (throw CommandError('BAD SCALE $text'));
229int parseInversion(String text, Quality quality) {
230 if (!RegExp(r'^[0-9]$').hasMatch(text) || int.parse(text) >= quality.intervals.length) {
231 throw CommandError('BAD INVERSION $text');
232 }
233 return int.parse(text);
234}
235Numeral parseNumeral(String text) {
236 final start = text.startsWith('b') || text.startsWith('#') ? 1 : 0;
237 final change = start == 0 ? 0 : text[0] == 'b' ? -1 : 1;
238 var end = start;
239 while (end < text.length && 'IViv'.contains(text[end])) {
240 end++;
241 }
242 final roman = text.substring(start, end);
243 final suffix = text.substring(end);
244 var degree = romans.indexOf(roman);
245 var lower = false;
246 if (degree < 0) {
247 degree = lowerRomans.indexOf(roman);
248 lower = true;
249 }
250 if (degree >= 0) {
251 for (final quality in qualities) {
252 if (quality.lower == lower && quality.numeralSuffix == suffix) return Numeral(degree, change, quality);
253 }
254 }
255 throw CommandError('BAD NUMERAL $text');
256}
257Chord parseSymbol(String text) {
258 final match = RegExp(r'^([A-G])(bb|b|#|x)?(.*)$').firstMatch(text);
259 if (match != null) {
260 for (final quality in qualities) {
261 if (quality.suffix == match[3]) return Chord(Pitch(letters.indexOf(match[1]!), alteration(match[2])), quality);
262 }
263 }
264 throw CommandError('BAD SYMBOL $text');
265}

267Pitch raisePitch(Pitch root, Interval interval) {
268 final letter = (root.letter + interval.letterDistance) % 7;
269 final change = root.natural + root.alteration + interval.semitones - naturals[letter] -
270 12 * floorDiv(root.letter + interval.letterDistance, 7);
271 if (change < -2 || change > 2) throw const CommandError('CANNOT SPELL');
272 return Pitch(letter, change);
273}
274List<Pitch> build(Pitch root, List<Interval> pattern) => pattern.map((interval) => raisePitch(root, interval)).toList();
275List<Pitch> keyScale(Key key) => build(key.tonic, scales[key.mode.scale]!);
276Note fromMidi(Pitch pitch, int midi) => Note(pitch, (midi - pitch.natural - pitch.alteration) ~/ 12 - 1);
277Note standardNote(int midi, [bool flat = false]) => Note(parsePitch((flat ? flatNames : sharpNames)[midi % 12]), midi ~/ 12 - 1);
278Pitch fifthsPitch(int fifths) => Pitch(letters.indexOf('FCGDAEB'[(fifths + 1) % 7]), floorDiv(fifths + 1, 7));
279Key keyAt(int fifths, Mode mode) => Key(fifthsPitch(fifths - mode.offset), mode);
280Degree label(Key key, Pitch pitch) {
281 final index = (pitch.letter - key.tonic.letter) % 7;
282 final change = pitch.alteration - keyScale(key)[index].alteration;
283 return Degree(index, (change < 0 ? 'b' : '#') * change.abs());
284}
285String romanNotation(int degree, Quality quality) => (quality.lower ? lowerRomans : romans)[degree] + quality.numeralSuffix;
286Motion between(Note first, Note second) {
287 var distance = second.position - first.position;
288 var semitones = second.midi - first.midi;
289 final down = distance < 0 || distance == 0 && semitones < 0;
290 if (down) {
291 distance = -distance;
292 semitones = -semitones;
293 }
294 final difference = semitones - (12 * (distance ~/ 7) + naturals[distance % 7]);
295 final perfect = const {0, 3, 4}.contains(distance % 7);
296 final quality = perfect
297 ? const {-2: 'dd', -1: 'd', 0: 'P', 1: 'A', 2: 'AA'}[difference]
298 : const {-3: 'dd', -2: 'd', -1: 'm', 0: 'M', 1: 'A', 2: 'AA'}[difference];
299 if (quality == null) throw const CommandError('UNNAMEABLE INTERVAL');
300 return Motion(Interval(quality, distance + 1), down, semitones);
301}
302int pitchMask(Iterable<Pitch> pitches) {
303 var mask = 0;
304 for (final pitch in pitches) {
305 mask |= 1 << pitch.pitchClass;
306 }
307 return mask;
308}
309int chordMask(Note root, Quality quality) {
310 var mask = 0;
311 for (final interval in quality.intervals) {
312 mask |= 1 << ((root.midi + interval.semitones) % 12);
313 }
314 return mask;
315}
316String cadence(List<Numeral> numerals) {
317 if (numerals.length < 2) return 'none';
318 final a = numerals[numerals.length - 2], b = numerals.last;
319 if (a.alteration == 0 && b.alteration == 0) {
320 if (a.degree == 4 && b.degree == 0) return 'authentic';
321 if (a.degree == 3 && b.degree == 0) return 'plagal';
322 if (a.degree == 4 && b.degree == 5) return 'deceptive';
323 }
324 return b.alteration == 0 && b.degree == 4 ? 'half' : 'none';
325}
326String frequency(int cents) => '${cents ~/ 100}.${(cents % 100).toString().padLeft(2, '0')}';
327String signed(int value, [bool positiveZero = false]) => value > 0 || value == 0 && positiveZero ? '+$value' : '$value';

338final class Calculator {
339 int tuning = 44000;
340 String execute(String line) {
341 final w = line.split(' ');
342 final command = w[0];
343 final arity = arities[command];
344 if (arity == null) throw CommandError('UNKNOWN COMMAND $command');
345 if (w.length - 1 < arity.$1 || w.length - 1 > arity.$2) throw const CommandError('ARGUMENT COUNT');
346 switch (command) {
347 case 'MIDI':
348 return 'MIDI ${w[1]} ${parseNote(w[1]).midi}';
349 case 'NOTE':
350 final number = parseNumber(w[1]);
351 return 'NOTE $number ${standardNote(number)} ${standardNote(number, true)}';
352 case 'FREQ':
353 final note = parseNote(w[1]);
354 final value = (tuning / 100) * math.pow(2, (note.midi - 69) / 12);
355 return 'FREQ ${w[1]} ${frequency((value * 100 + 0.5).floor())}';
356 case 'TUNING':
357 final value = parseFrequency(w[1], true);
358 tuning = value;
359 return 'TUNING ${frequency(value)}';
360 case 'PITCH':
361 final value = parseFrequency(w[1], false);
362 final x = 69 + 12 * math.log(value / tuning) / math.ln2;
363 final number = (x + 0.5).floor();
364 if (number < 0 || number > 127) throw const CommandError('OUT OF RANGE');
365 final cents = (100 * (x - number) + 0.5).floor();
366 return 'PITCH ${frequency(value)} ${standardNote(number)} ${signed(cents, true)}';
367 case 'INTERVAL':
368 final a = parseNote(w[1]);
369 final b = parseNote(w[2]);
370 final motion = between(a, b);
371 return 'INTERVAL ${w[1]} ${w[2]} ${motion.interval} ${motion.down ? 'down' : 'up'} '
372 'semitones=${motion.semitones} simple=${motion.interval.simple} class=${motion.interval.consonance}';
373 case 'INVERT':
374 final interval = parseInterval(w[1]);
375 final simple = interval.simple;
376 final quality = const {'P': 'P', 'M': 'm', 'm': 'M', 'A': 'd', 'd': 'A', 'AA': 'dd', 'dd': 'AA'}[simple.quality]!;
377 final inverted = Interval(quality, 9 - simple.number);
378 if (inverted.number == 1 && const {'d', 'dd'}.contains(quality)) throw const CommandError('CANNOT INVERT');
379 return 'INVERT ${w[1]} $simple $inverted';
380 case 'TRANSPOSE':
381 final note = parseNote(w[1]);
382 final interval = parseInterval(w[2]);
383 if (w[3] != 'UP' && w[3] != 'DOWN') throw CommandError('BAD DIRECTION ${w[3]}');
384 final sign = w[3] == 'UP' ? 1 : -1;
385 final position = note.position + sign * interval.letterDistance;
386 final midi = note.midi + sign * interval.semitones;
387 final letter = position % 7, octave = floorDiv(position, 7);
388 final change = midi - (12 * (octave + 1) + naturals[letter]);
389 if (change < -2 || change > 2) throw const CommandError('CANNOT SPELL');
390 final result = Note(Pitch(letter, change), octave);
391 if (!result.valid) throw const CommandError('OUT OF RANGE');
392 return 'TRANSPOSE ${w[1]} ${w[2]} ${w[3]} $result';
393 case 'SCALE':
394 final root = parsePitch(w[1]);
395 final pattern = parseScale(w[2]);
396 final notes = build(root, pattern);
397 final steps = List.generate(pattern.length, (i) => (i + 1 == pattern.length ? 12 : pattern[i + 1].semitones) - pattern[i].semitones);
398 return 'SCALE ${w[1]} ${w[2]} ${notes.join(' ')} steps=${steps.join(',')}';
399 case 'CHORD':
400 case 'VOICE':
401 final input = command == 'VOICE' ? parseNote(w[1]) : null;
402 final root = input?.pitch ?? parsePitch(w[1]);
403 final quality = parseQuality(w[2]);
404 final inversion = w.length == 4 ? parseInversion(w[3], quality) : 0;
405 final tones = build(root, quality.intervals);
406 final reordered = [...tones.skip(inversion), ...tones.take(inversion)];
407 final symbol = '${Chord(root, quality)}${inversion == 0 ? '' : '/${reordered.first}'}';
408 if (command == 'CHORD') return 'CHORD $symbol ${reordered.join(' ')}';
409 final notes = <Note>[];
410 var previous = input!.midi;
411 for (var i = 0; i < reordered.length; i++) {
412 var midi = previous;
413 if (i != 0 || inversion != 0) {
414 midi += (reordered[i].pitchClass - previous) % 12;
415 if (midi == previous) midi += 12;
416 }
417 final made = fromMidi(reordered[i], midi);
418 if (!made.valid) throw const CommandError('OUT OF RANGE');
419 notes.add(made);
420 previous = midi;
421 }
422 return 'VOICE $symbol ${notes.join(' ')} midi=${notes.map((n) => n.midi).join(',')}';
423 case 'IDENTIFY':
424 final notes = w.skip(1).map(parseNote).toList();
425 final ordered = List.generate(notes.length, (i) => (i, notes[i]));
426 ordered.sort((a, b) {
427 final comparison = a.$2.midi.compareTo(b.$2.midi);
428 return comparison != 0 ? comparison : a.$1.compareTo(b.$1);
429 });
430 final seen = <int>{};
431 final candidates = ordered.map((entry) => entry.$2).where((note) => seen.add(note.pitch.pitchClass)).toList();
432 final mask = pitchMask(notes.map((note) => note.pitch));
433 final bass = candidates.first;
434 for (final root in candidates) {
435 for (final quality in qualities) {
436 if (chordMask(root, quality) != mask) continue;
437 final inversion = quality.intervals.indexWhere((interval) => (root.midi + interval.semitones) % 12 == bass.pitch.pitchClass);
438 final symbol = '${Chord(root.pitch, quality)}${inversion == 0 ? '' : '/${bass.pitch}'}';
439 return 'IDENTIFY $symbol root=${root.pitch} quality=${quality.name} inversion=$inversion';
440 }
441 }
442 return 'IDENTIFY UNKNOWN';
443 case 'MELODY':
444 final notes = w.skip(1).map(parseNote).toList();
445 var low = notes.first, high = notes.first;
446 for (final note in notes) {
447 if (note.midi < low.midi) low = note;
448 if (note.midi > high.midi) high = note;
449 }
450 final range = between(low, high).interval;
451 final steps = <String>[];
452 var leaps = 0;
453 for (var i = 1; i < notes.length; i++) {
454 final motion = between(notes[i - 1], notes[i]);
455 steps.add('${motion.down ? '-' : '+'}${motion.interval}');
456 if ((notes[i].midi - notes[i - 1].midi).abs() >= 3) leaps++;
457 }
458 return 'MELODY n=${notes.length} low=$low high=$high range=$range leaps=$leaps steps=${steps.join(',')}';
459 case 'ENHARMONIC':
460 final input = parseNote(w[1]);
461 final notes = <Note>[];
462 for (var letter = 0; letter < 7; letter++) {
463 for (var change = -2; change <= 2; change++) {
464 final pitch = Pitch(letter, change);
465 if (pitch.pitchClass != input.pitch.pitchClass) continue;
466 final note = fromMidi(pitch, input.midi);
467 if (note.valid) notes.add(note);
468 }
469 }
470 notes.sort((a, b) => a.position.compareTo(b.position));
471 return 'ENHARMONIC ${w[1]} ${notes.join(' ')}';
472 case 'KEYSIG':
473 case 'RELATIVE':
474 case 'DIATONIC':
475 case 'PROGRESSION':
476 case 'ANALYZE':
477 case 'DEGREE':
478 final key = parseKey(w[1], w[2]);
479 if (command == 'KEYSIG') {
480 final fifths = key.fifths;
481 if (fifths == 0) return 'KEYSIG $key none';
482 final names = (fifths > 0 ? 'F# C# G# D# A# E# B#' : 'Bb Eb Ab Db Gb Cb Fb').split(' ');
483 return 'KEYSIG $key ${fifths > 0 ? 'sharps' : 'flats'}=${fifths.abs()} ${names.take(fifths.abs()).join(' ')}';
484 }
485 if (command == 'RELATIVE') return 'RELATIVE $key major=${keyAt(key.fifths, modes[0]).tonic} minor=${keyAt(key.fifths, modes[1]).tonic}';
486 if (command == 'DEGREE') {
487 final pitch = parsePitch(w[3]);
488 return 'DEGREE $key ${w[3]} ${label(key, pitch)}';
489 }
490 if (command == 'ANALYZE') {
491 final chords = w.skip(3).map(parseSymbol).toList();
492 final results = chords.map((chord) {
493 final degree = label(key, chord.root);
494 return degree.prefix + romanNotation(degree.index, chord.quality);
495 });
496 return 'ANALYZE $key ${results.join(' ')}';
497 }
498 if (command == 'PROGRESSION') {
499 final numerals = w.skip(3).map(parseNumeral).toList();
500 final scale = keyScale(key);
501 final lines = ['PROGRESSION $key ${numerals.length}'];
502 for (var i = 0; i < numerals.length; i++) {
503 final numeral = numerals[i];
504 final degree = scale[numeral.degree];
505 final root = Pitch(degree.letter, degree.alteration + numeral.alteration);
506 final tones = build(root, numeral.quality.intervals);
507 lines.add('${w[i + 3]} ${Chord(root, numeral.quality)} ${tones.join(' ')}');
508 }
509 lines.add('CADENCE ${cadence(numerals)}');
510 return lines.join('\n');
511 }
512 final scale = keyScale(key);
513 final triads = <String>[], sevenths = <String>[];
514 for (var i = 0; i < 7; i++) {
515 final third = (scale[(i + 2) % 7].pitchClass - scale[i].pitchClass) % 12;
516 final fifth = (scale[(i + 4) % 7].pitchClass - scale[i].pitchClass) % 12;
517 final seventh = (scale[(i + 6) % 7].pitchClass - scale[i].pitchClass) % 12;
518 final triad = parseQuality(third == 4 ? 'maj' : fifth == 7 ? 'min' : 'dim');
519 final tetrad = parseQuality(third == 4 ? seventh == 11 ? 'maj7' : '7' : fifth == 7 ? 'm7' : 'm7b5');
520 triads.add('${romanNotation(i, triad)}=${Chord(scale[i], triad)}');
521 sevenths.add('${romanNotation(i, tetrad)}=${Chord(scale[i], tetrad)}');
522 }
523 return 'DIATONIC $key\nTRIADS ${triads.join(' ')}\nSEVENTHS ${sevenths.join(' ')}';
524 case 'CIRCLE':
525 final mode = parseMode(w[1]);
526 return 'CIRCLE ${w[1]} ${List.generate(15, (i) => keyAt(i - 7, mode).tonic).join(' ')}';
527 case 'FINDKEY':
528 final pitches = w.skip(1).map(parsePitch).toList();
529 final keys = <Key>[];
530 for (var fifths = -7; fifths <= 7; fifths++) {
531 for (var m = 0; m < 2; m++) {
532 final key = keyAt(fifths, modes[m]);
533 final scale = keyScale(key);
534 if (pitches.every(scale.contains)) keys.add(key);
535 }
536 }
537 return 'FINDKEY ${keys.length}${keys.isEmpty ? '' : ' ${keys.join(', ')}'}';
538 case 'SPELL':
539 final midi = parseNumber(w[1]);
540 final key = parseKey(w[2], w[3]);
541 var pitch = standardNote(midi, key.fifths < 0).pitch;
542 for (final candidate in keyScale(key)) {
543 if (candidate.pitchClass == midi % 12) {
544 pitch = candidate;
545 break;
546 }
547 }
548 final note = fromMidi(pitch, midi);
549 if (!note.valid) throw const CommandError('OUT OF RANGE');
550 return 'SPELL $midi $key $note ${label(key, pitch)}';
551 case 'KEYDIST':
552 final a = parseKey(w[1], w[2]);
553 final b = parseKey(w[3], w[4]);
554 var difference = (b.fifths - a.fifths) % 12;
555 if (difference > 6) difference -= 12;
556 final first = pitchMask(keyScale(a)), second = pitchMask(keyScale(b));
557 var common = 0;
558 for (var i = 0; i < 12; i++) {
559 if ((first & second & (1 << i)) != 0) common++;
560 }
561 final sameTonic = a.tonic.pitchClass == b.tonic.pitchClass;
562 final relation = first == second ? sameTonic ? 'same' : 'relative' : sameTonic ? 'parallel' : difference.abs() == 1 ? 'close' : 'distant';
563 return 'KEYDIST $a $b fifths=${signed(difference)} common=$common relation=$relation';
564 default:
565 throw StateError('Unknown checked command');
566 }
567 }
568}
