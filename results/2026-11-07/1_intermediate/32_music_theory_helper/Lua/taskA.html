5local function join(values, separator)
6 local text = {}
7 for i, value in ipairs(values) do text[i] = tostring(value) end

10local function frequency_text(cents) return format("%d.%02d", cents // 100, cents % 100) end
11local function signed_text(value, positive_zero)
12 return (value > 0 or (value == 0 and positive_zero)) and "+" .. value or tostring(value)
13end

15local letters, naturals = "CDEFGAB", {0, 2, 4, 5, 7, 9, 11}
16local accidentals = {"bb", "b", "", "#", "x"}
17local sharp_names = {"C", "C#", "D", "D#", "E", "F", "F#", "G", "G#", "A", "A#", "B"}
18local flat_names = {"C", "Db", "D", "Eb", "E", "F", "Gb", "G", "Ab", "A", "Bb", "B"}
19local romans = {"I", "II", "III", "IV", "V", "VI", "VII"}
20local Pitch = {}
21Pitch.__index = Pitch
22function Pitch.new(letter, alteration) return setmetatable({letter = letter, alteration = alteration or 0}, Pitch) end
23function Pitch:natural() return naturals[self.letter] end
24function Pitch:class() return (self:natural() + self.alteration) % 12 end
25function Pitch:__tostring() return letters:sub(self.letter, self.letter) .. accidentals[self.alteration + 3] end
26function Pitch.__eq(a, b) return a.letter == b.letter and a.alteration == b.alteration end
27function Pitch.prefix(text)
28 if text == "" then return nil end
29 local letter = letters:find(text:sub(1, 1), 1, true)
30 if not letter then return nil end
31 local finish, alteration = 2, 0
32 if text:sub(2, 3) == "bb" then finish, alteration = 4, -2
33 elseif text:sub(2, 2) == "b" then finish, alteration = 3, -1
34 elseif text:sub(2, 2) == "#" then finish, alteration = 3, 1
35 elseif text:sub(2, 2) == "x" then finish, alteration = 3, 2 end
36 return Pitch.new(letter, alteration), finish
37end
38function Pitch.parse(text)
39 local pitch, finish = Pitch.prefix(text)
40 if not pitch or finish ~= #text + 1 then fail("BAD PITCH", text) end
41 return pitch
42end
43function Pitch.at_fifths(fifths)
44 local index = (fifths + 1) % 7 + 1
45 local letter = letters:find(("FCGDAEB"):sub(index, index), 1, true)
46 return Pitch.new(letter, (fifths + 1) // 7)
47end
48function Pitch:raise(interval)
49 local position = self.letter - 1 + interval:distance()
50 local letter = position % 7 + 1
51 local alteration = self:natural() + self.alteration + interval:semitones() - naturals[letter] - 12 * (position // 7)
52 if alteration < -2 or alteration > 2 then fail("CANNOT SPELL") end
53 return Pitch.new(letter, alteration)
54end

56local Note = {}
57Note.__index = Note
58function Note.new(pitch, octave) return setmetatable({pitch = pitch, octave = octave}, Note) end
59function Note:midi() return 12 * (self.octave + 1) + self.pitch:natural() + self.pitch.alteration end
60function Note:position() return 7 * self.octave + self.pitch.letter - 1 end
61function Note:valid() return self.octave >= -1 and self.octave <= 9 and self:midi() >= 0 and self:midi() <= 127 end
62function Note:__tostring() return tostring(self.pitch) .. self.octave end
63function Note.parse(text)
64 local pitch, finish = Pitch.prefix(text)
65 if not pitch then fail("BAD NOTE", text) end
66 local tail = text:sub(finish)
67 if tail ~= "-1" and not tail:match("^[0-9]$") then fail("BAD NOTE", text) end
68 local note = Note.new(pitch, tonumber(tail, 10))
69 if not note:valid() then fail("BAD NOTE", text) end
70 return note
71end
72function Note.from_midi(pitch, midi)
73 return Note.new(pitch, (midi - pitch:natural() - pitch.alteration) // 12 - 1)
74end
75function Note.spelling(midi, flat)
76 return Note.new(Pitch.parse((flat and flat_names or sharp_names)[midi % 12 + 1]), midi // 12 - 1)
77end

79local perfect_deltas = {dd = -2, d = -1, P = 0, A = 1, AA = 2}
80local major_deltas = {dd = -3, d = -2, m = -1, M = 0, A = 1, AA = 2}
81local Interval = {}
82Interval.__index = Interval
83function Interval.new(quality, number) return setmetatable({quality = quality, number = number}, Interval) end
84function Interval:distance() return self.number - 1 end
85function Interval:perfect()
86 local remainder = self:distance() % 7
87 return remainder == 0 or remainder == 3 or remainder == 4
88end
89function Interval:semitones()
90 local deltas = self:perfect() and perfect_deltas or major_deltas
91 return 12 * (self:distance() // 7) + naturals[self:distance() % 7 + 1] + deltas[self.quality]
92end
93function Interval:simple()
94 local number = self.number
95 while number > 8 do number = number - 7 end
96 return Interval.new(self.quality, number)
97end
98function Interval:consonance()
99 local simple = self:simple()
100 if simple.quality == "P" and (simple.number == 1 or simple.number == 5 or simple.number == 8) then return "perfect" end
101 if (simple.quality == "M" or simple.quality == "m") and (simple.number == 3 or simple.number == 6) then return "imperfect" end
102 return "dissonant"
103end
104function Interval:__tostring() return self.quality .. self.number end
105function Interval.parse(text)
106 local quality, number = text:match("^([A-Za-z]+)([0-9]+)$")
107 if not quality or #number > 2 or number:sub(1, 1) == "0" then fail("BAD INTERVAL", text) end
108 local result = Interval.new(quality, tonumber(number, 10))
109 local deltas = result:perfect() and perfect_deltas or major_deltas
110 if deltas[quality] == nil or (result.number == 1 and (quality == "d" or quality == "dd")) then fail("BAD INTERVAL", text) end
111 return result
112end
113local function between(a, b)
114 local distance, semitones = b:position() - a:position(), b:midi() - a:midi()
115 local down = distance < 0 or (distance == 0 and semitones < 0)
116 if down then distance, semitones = -distance, -semitones end
117 local difference = semitones - (12 * (distance // 7) + naturals[distance % 7 + 1])
118 local qualities = (distance % 7 == 0 or distance % 7 == 3 or distance % 7 == 4)
119 and {[-2] = "dd", [-1] = "d", [0] = "P", [1] = "A", [2] = "AA"}
120 or {[-3] = "dd", [-2] = "d", [-1] = "m", [0] = "M", [1] = "A", [2] = "AA"}
121 if not qualities[difference] then fail("UNNAMEABLE INTERVAL") end
122 return {interval = Interval.new(qualities[difference], distance + 1), down = down, semitones = semitones}
123end
124local function pattern(text)
125 local result = {}
126 for word in text:gmatch("[^ ]+") do result[#result + 1] = Interval.parse(word) end
127 return result
128end

182local Key = {}
183Key.__index = Key
184function Key.new(tonic, mode) return setmetatable({tonic = tonic, mode = mode}, Key) end
185function Key:fifths() return ({0, 2, 4, -1, 1, 3, 5})[self.tonic.letter] + 7 * self.tonic.alteration + mode_info[self.mode].offset end
186function Key:scale() return build(self.tonic, scales[mode_info[self.mode].scale]) end
187function Key:__tostring() return tostring(self.tonic) .. " " .. self.mode end
188function Key.parse(tonic, mode)
189 local pitch = Pitch.parse(tonic)
190 local mode_value = parse_mode(mode)
191 local key = Key.new(pitch, mode_value)
192 if key:fifths() < -7 or key:fifths() > 7 then fail("THEORETICAL KEY") end
193 return key
194end
195function Key.at_fifths(fifths, mode) return Key.new(Pitch.at_fifths(fifths - mode_info[mode].offset), mode) end
196function Key:degree(pitch)
197 local degree = (pitch.letter - self.tonic.letter) % 7 + 1
198 local difference = pitch.alteration - self:scale()[degree].alteration
199 local prefix = (difference < 0 and "b" or "#"):rep(math.abs(difference))
200 return degree, prefix
201end
202local function degree_label(key, pitch)
203 local degree, prefix = key:degree(pitch)
204 return prefix .. degree
205end
206local function parse_number(text)
207 if not text:match("^[0-9]+$") or #text > 3 or (#text > 1 and text:sub(1, 1) == "0") then fail("BAD NUMBER", text) end
208 local value = tonumber(text, 10)
209 if value > 127 then fail("BAD NUMBER", text) end
210 return value
211end
212local function parse_frequency(text, tuning)
213 local whole, tail = text:match("^([0-9]+)%.([0-9]+)$")
214 if not whole then whole, tail = text, "" end
215 if not whole:match("^[0-9]+$") or #whole > 5 or #tail > 2 then fail("BAD FREQUENCY", text) end
216 local value = tonumber(whole, 10) * 100
217 if tail ~= "" then value = value + tonumber(tail, 10) * (#tail == 1 and 10 or 1) end
218 if value == 0 or (tuning and (value < 40000 or value > 48000)) then fail("BAD FREQUENCY", text) end
219 return value
220end
221local function parse_inversion(text, quality)
222 if not text:match("^[0-9]$") then fail("BAD INVERSION", text) end
223 local value = tonumber(text, 10)
224 if value >= #quality.pattern then fail("BAD INVERSION", text) end
225 return value
226end
227local function parse_numeral(text)
228 local first = text:sub(1, 1)
229 local start = (first == "b" or first == "#") and 2 or 1
230 local alteration = start == 1 and 0 or (first == "b" and -1 or 1)
231 local finish = start
232 while finish <= #text and ("IViv"):find(text:sub(finish, finish), 1, true) do finish = finish + 1 end
233 local part, suffix = text:sub(start, finish - 1), text:sub(finish)
234 for degree, roman in ipairs(romans) do
235 local lower = part == roman:lower()
236 if lower or part == roman then
237 for _, quality in ipairs(qualities) do
238 if quality:lower() == lower and quality.numeral:sub(2) == suffix then
239 return {degree = degree, alteration = alteration, quality = quality}
240 end
241 end
242 end
243 end
244 fail("BAD NUMERAL", text)
245end
246local function parse_symbol(text)
247 local root, finish = Pitch.prefix(text)
248 if root then
249 for _, quality in ipairs(qualities) do
250 if quality.suffix == text:sub(finish) then return {root = root, quality = quality} end
251 end
252 end
253 fail("BAD SYMBOL", text)
254end
255local function parse_values(words, first, parser)
256 local result = {}
257 for i = first, #words do result[#result + 1] = parser(words[i]) end
258 return result
259end
260local function mask(pitches)
261 local result = 0
262 for _, pitch in ipairs(pitches) do result = result | (1 << pitch:class()) end
263 return result
264end
265local function cadence(numerals)
266 if #numerals < 2 then return "none" end
267 local a, b = numerals[#numerals - 1], numerals[#numerals]
268 if a.alteration == 0 and b.alteration == 0 then
269 if a.degree == 5 and b.degree == 1 then return "authentic" end
270 if a.degree == 4 and b.degree == 1 then return "plagal" end
271 if a.degree == 5 and b.degree == 6 then return "deceptive" end
272 end
273 return b.alteration == 0 and b.degree == 5 and "half" or "none"
274end

276local limits = {
277 MIDI = {1, 1}, NOTE = {1, 1}, FREQ = {1, 1}, TUNING = {1, 1}, PITCH = {1, 1},
278 INTERVAL = {2, 2}, INVERT = {1, 1}, TRANSPOSE = {3, 3}, SCALE = {2, 2},
279 CHORD = {2, 3}, VOICE = {2, 3}, IDENTIFY = {2, 15}, MELODY = {2, 15}, ENHARMONIC = {1, 1},
280 KEYSIG = {2, 2}, RELATIVE = {2, 2}, CIRCLE = {1, 1}, DIATONIC = {2, 2},
281 PROGRESSION = {3, 15}, ANALYZE = {3, 15}, FINDKEY = {1, 15}, DEGREE = {3, 3}, SPELL = {3, 3}, KEYDIST = {4, 4},
282}
283local tuning = 44000
284local function execute(line)
285 local words = {}
286 for word in line:gmatch("[^ ]+") do words[#words + 1] = word end
287 local command, count = words[1], #words - 1
288 local arity = limits[command]
289 if not arity then fail("UNKNOWN COMMAND", command) end
290 if count < arity[1] or count > arity[2] then fail("ARGUMENT COUNT") end
291 if command == "MIDI" then
292 local note = Note.parse(words[2])
293 return format("MIDI %s %d", note, note:midi())
294 elseif command == "NOTE" then
295 local number = parse_number(words[2])
296 return format("NOTE %d %s %s", number, Note.spelling(number), Note.spelling(number, true))
297 elseif command == "FREQ" then
298 local note = Note.parse(words[2])
299 local cents = math.floor(tuning * 2 ^ ((note:midi() - 69) / 12) + 0.5)
300 return format("FREQ %s %s", note, frequency_text(cents))
301 elseif command == "TUNING" then
302 local value = parse_frequency(words[2], true)
303 tuning = value
304 return "TUNING " .. frequency_text(value)
305 elseif command == "PITCH" then
306 local value = parse_frequency(words[2], false)
307 local height = 69 + 12 * math.log(value / tuning, 2)
308 local number = math.floor(height + 0.5)
309 if number < 0 or number > 127 then fail("OUT OF RANGE") end
310 local cents = math.floor(100 * (height - number) + 0.5)
311 return format("PITCH %s %s %s", frequency_text(value), Note.spelling(number), signed_text(cents, true))
312 elseif command == "INTERVAL" then
313 local a = Note.parse(words[2])
314 local b = Note.parse(words[3])
315 local motion = between(a, b)
316 return format("INTERVAL %s %s %s %s semitones=%d simple=%s class=%s", a, b, motion.interval,
317 motion.down and "down" or "up", motion.semitones, motion.interval:simple(), motion.interval:consonance())
318 elseif command == "INVERT" then
319 local original = Interval.parse(words[2])
320 local simple = original:simple()
321 local inverse_quality = {P = "P", M = "m", m = "M", A = "d", d = "A", AA = "dd", dd = "AA"}
322 local inverse = Interval.new(inverse_quality[simple.quality], 9 - simple.number)
323 if inverse.number == 1 and (inverse.quality == "d" or inverse.quality == "dd") then fail("CANNOT INVERT") end
324 return format("INVERT %s %s %s", original, simple, inverse)
325 elseif command == "TRANSPOSE" then
326 local note = Note.parse(words[2])
327 local interval = Interval.parse(words[3])
328 local direction = words[4]
329 if direction ~= "UP" and direction ~= "DOWN" then fail("BAD DIRECTION", direction) end
330 local sign = direction == "UP" and 1 or -1
331 local position = note:position() + sign * interval:distance()
332 local midi = note:midi() + sign * interval:semitones()
333 local letter, octave = position % 7 + 1, position // 7
334 local alteration = midi - (12 * (octave + 1) + naturals[letter])
335 if alteration < -2 or alteration > 2 then fail("CANNOT SPELL") end
336 local result = Note.new(Pitch.new(letter, alteration), octave)
337 if not result:valid() then fail("OUT OF RANGE") end
338 return format("TRANSPOSE %s %s %s %s", note, interval, direction, result)
339 elseif command == "SCALE" then
340 local root = Pitch.parse(words[2])
341 local intervals = scales[words[3]]
342 if not intervals then fail("BAD SCALE", words[3]) end
343 local pitches = build(root, intervals)
344 local steps = {}
345 for i, interval in ipairs(intervals) do
346 local next_tone = i == #intervals and 12 or intervals[i + 1]:semitones()
347 steps[i] = next_tone - interval:semitones()
348 end
349 return format("SCALE %s %s %s steps=%s", root, words[3], join(pitches), join(steps, ","))
350 elseif command == "CHORD" or command == "VOICE" then
351 local input, root
352 if command == "VOICE" then input = Note.parse(words[2]); root = input.pitch
353 else root = Pitch.parse(words[2]) end
354 local quality = parse_quality(words[3])
355 local inversion = count == 3 and parse_inversion(words[4], quality) or 0
356 local tones = build(root, quality.pattern)
357 local pitches = {}
358 for i = 1, #tones do pitches[i] = tones[(i - 1 + inversion) % #tones + 1] end
359 local symbol = quality:symbol(root) .. (inversion > 0 and "/" .. tostring(pitches[1]) or "")
360 if command == "CHORD" then return "CHORD " .. symbol .. " " .. join(pitches) end
361 local notes, midis = {}, {}
362 local previous = input:midi()
363 for i, pitch in ipairs(pitches) do
364 local midi = previous
365 if i > 1 or inversion > 0 then
366 midi = midi + (pitch:class() - previous) % 12
367 if midi == previous then midi = midi + 12 end
368 end
369 local note = Note.from_midi(pitch, midi)
370 if not note:valid() then fail("OUT OF RANGE") end
371 notes[i], midis[i], previous = note, midi, midi
372 end
373 return format("VOICE %s %s midi=%s", symbol, join(notes), join(midis, ","))
374 elseif command == "IDENTIFY" then
375 local notes = parse_values(words, 2, Note.parse)
376 local ordered = {}
377 for i, note in ipairs(notes) do ordered[i] = {note = note, index = i} end
378 table.sort(ordered, function(a, b)
379 if a.note:midi() ~= b.note:midi() then return a.note:midi() < b.note:midi() end
380 return a.index < b.index
381 end)
382 local candidates, target = {}, 0
383 for _, entry in ipairs(ordered) do
384 local bit = 1 << entry.note.pitch:class()
385 if target & bit == 0 then candidates[#candidates + 1] = entry.note end
386 target = target | bit
387 end
388 local bass = candidates[1]
389 for _, root in ipairs(candidates) do
390 for _, quality in ipairs(qualities) do
391 local candidate, inversion = 0, 0
392 for i, interval in ipairs(quality.pattern) do
393 local pitch_class = (root:midi() + interval:semitones()) % 12
394 candidate = candidate | (1 << pitch_class)
395 if pitch_class == bass.pitch:class() then inversion = i - 1 end
396 end
397 if candidate == target then
398 local symbol = quality:symbol(root.pitch) .. (inversion > 0 and "/" .. tostring(bass.pitch) or "")
399 return format("IDENTIFY %s root=%s quality=%s inversion=%d", symbol, root.pitch, quality.name, inversion)
400 end
401 end
402 end
403 return "IDENTIFY UNKNOWN"
404 elseif command == "MELODY" then
405 local notes = parse_values(words, 2, Note.parse)
406 local low, high = notes[1], notes[1]
407 for _, note in ipairs(notes) do
408 if note:midi() < low:midi() then low = note end
409 if note:midi() > high:midi() then high = note end
410 end
411 local range = between(low, high).interval
412 local steps, leaps = {}, 0
413 for i = 2, #notes do
414 local motion = between(notes[i - 1], notes[i])
415 steps[#steps + 1] = (motion.down and "-" or "+") .. tostring(motion.interval)
416 if math.abs(notes[i]:midi() - notes[i - 1]:midi()) >= 3 then leaps = leaps + 1 end
417 end
418 return format("MELODY n=%d low=%s high=%s range=%s leaps=%d steps=%s", count, low, high, range, leaps, join(steps, ","))
419 elseif command == "ENHARMONIC" then
420 local input = Note.parse(words[2])
421 local notes = {}
422 for letter = 1, 7 do
423 for alteration = -2, 2 do
424 local pitch = Pitch.new(letter, alteration)
425 if pitch:class() == input.pitch:class() then
426 local note = Note.from_midi(pitch, input:midi())
427 if note:valid() then notes[#notes + 1] = note end
428 end
429 end
430 end
431 table.sort(notes, function(a, b) return a:position() < b:position() end)
432 return format("ENHARMONIC %s %s", input, join(notes))
433 elseif command == "CIRCLE" then
434 local mode = parse_mode(words[2])
435 local tonics = {}
436 for fifths = -7, 7 do tonics[#tonics + 1] = Key.at_fifths(fifths, mode).tonic end
437 return "CIRCLE " .. mode .. " " .. join(tonics)
438 elseif command == "FINDKEY" then
439 local pitches = parse_values(words, 2, Pitch.parse)
440 local keys = {}
441 for fifths = -7, 7 do
442 for _, mode in ipairs({"major", "minor"}) do
443 local key = Key.at_fifths(fifths, mode)
444 local scale, matches = key:scale(), true
445 for _, pitch in ipairs(pitches) do
446 local found = false
447 for _, candidate in ipairs(scale) do if candidate == pitch then found = true; break end end
448 if not found then matches = false; break end
449 end
450 if matches then keys[#keys + 1] = key end
451 end
452 end
453 return "FINDKEY " .. #keys .. (#keys > 0 and " " .. join(keys, ", ") or "")
454 elseif command == "SPELL" then
455 local midi = parse_number(words[2])
456 local key = Key.parse(words[3], words[4])
457 local pitch = Note.spelling(midi, key:fifths() < 0).pitch
458 for _, candidate in ipairs(key:scale()) do
459 if candidate:class() == midi % 12 then pitch = candidate; break end
460 end
461 local note = Note.from_midi(pitch, midi)
462 if not note:valid() then fail("OUT OF RANGE") end
463 return format("SPELL %d %s %s %s", midi, key, note, degree_label(key, pitch))
464 end
465
466 local key = Key.parse(words[2], words[3])
467 if command == "KEYSIG" then
468 local fifths = key:fifths()
469 if fifths == 0 then return format("KEYSIG %s none", key) end
470 local names = fifths > 0 and {"F#", "C#", "G#", "D#", "A#", "E#", "B#"} or {"Bb", "Eb", "Ab", "Db", "Gb", "Cb", "Fb"}
471 local chosen = {}
472 for i = 1, math.abs(fifths) do chosen[i] = names[i] end
473 return format("KEYSIG %s %s=%d %s", key, fifths > 0 and "sharps" or "flats", math.abs(fifths), join(chosen))
474 elseif command == "RELATIVE" then
475 return format("RELATIVE %s major=%s minor=%s", key, Key.at_fifths(key:fifths(), "major").tonic, Key.at_fifths(key:fifths(), "minor").tonic)
476 elseif command == "DEGREE" then
477 local pitch = Pitch.parse(words[4])
478 return format("DEGREE %s %s %s", key, pitch, degree_label(key, pitch))
479 elseif command == "KEYDIST" then
480 local other = Key.parse(words[4], words[5])
481 local distance = (other:fifths() - key:fifths()) % 12
482 if distance > 6 then distance = distance - 12 end
483 local a, b = mask(key:scale()), mask(other:scale())
484 local same_tonic = key.tonic:class() == other.tonic:class()
485 local relation
486 if a == b then relation = same_tonic and "same" or "relative"
487 elseif same_tonic then relation = "parallel"
488 elseif math.abs(distance) == 1 then relation = "close"
489 else relation = "distant" end
490 local common = 0
491 for i = 0, 11 do if a & b & (1 << i) ~= 0 then common = common + 1 end end
492 return format("KEYDIST %s %s fifths=%s common=%d relation=%s", key, other, signed_text(distance), common, relation)
493 elseif command == "ANALYZE" then
494 local chords = parse_values(words, 4, parse_symbol)
495 local results = {}
496 for i, chord in ipairs(chords) do
497 local degree, prefix = key:degree(chord.root)
498 results[i] = prefix .. chord.quality:roman(degree)
499 end
500 return format("ANALYZE %s %s", key, join(results))
501 elseif command == "PROGRESSION" then
502 local numerals = parse_values(words, 4, parse_numeral)
503 local scale = key:scale()
504 local lines = {format("PROGRESSION %s %d", key, #numerals)}
505 for i, numeral in ipairs(numerals) do
506 local degree = scale[numeral.degree]
507 local root = Pitch.new(degree.letter, degree.alteration + numeral.alteration)
508 local pitches = build(root, numeral.quality.pattern)
509 lines[#lines + 1] = format("%s %s %s", words[i + 3], numeral.quality:symbol(root), join(pitches))
510 end
511 lines[#lines + 1] = "CADENCE " .. cadence(numerals)
512 return join(lines, "\n")
513 else
514 local scale = key:scale()
515 local lines = {format("DIATONIC %s", key)}
516 for size = 3, 4 do
517 local items = {}
518 for i = 1, 7 do
519 local third = (scale[(i + 1) % 7 + 1]:class() - scale[i]:class()) % 12
520 local fifth = (scale[(i + 3) % 7 + 1]:class() - scale[i]:class()) % 12
521 local seventh = (scale[(i + 5) % 7 + 1]:class() - scale[i]:class()) % 12
522 local name
523 if size == 3 then name = third == 4 and "maj" or (fifth == 7 and "min" or "dim")
524 else name = third == 4 and (seventh == 11 and "maj7" or "7") or (fifth == 7 and "m7" or "m7b5") end
525 local quality = quality_names[name]
526 items[i] = quality:roman(i) .. "=" .. quality:symbol(scale[i])
527 end
528 lines[#lines + 1] = (size == 3 and "TRIADS " or "SEVENTHS ") .. join(items)
529 end
530 return join(lines, "\n")
531 end
532end
