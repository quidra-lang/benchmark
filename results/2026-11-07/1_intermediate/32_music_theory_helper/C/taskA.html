98static int mod(int value, int divisor) { int r = value % divisor; return r < 0 ? r + divisor : r; }
99static int floor_div(int value, int divisor) { return (value - mod(value, divisor)) / divisor; }
100static bool canonical(const char *text, int low, int high, int *value) {
101 if (!*text || (text[0] == '0' && text[1])) return false;
102 int n = 0;
103 for (const char *p = text; *p; ++p) {
104 if (*p < '0' || *p > '9' || n > (high - (*p - '0')) / 10) return false;
105 n = n * 10 + *p - '0';
106 if (n > high) return false;
107 }
108 if (n < low) return false;
109 *value = n;
110 return true;
111}
112static int pitch_prefix(const char *text, Pitch *pitch) {
113 if (!text[0]) return 0;
114 const char *letter = strchr(letter_names, text[0]);
115 if (!letter) return 0;
116 *pitch = (Pitch){(int)(letter - letter_names), 0};
117 int end = 1;
118 if (text[end] == 'b') {
119 pitch->alteration = -1;
120 if (text[++end] == 'b') { pitch->alteration = -2; ++end; }
121 } else if (text[end] == '#') { pitch->alteration = 1; ++end; }
122 else if (text[end] == 'x') { pitch->alteration = 2; ++end; }
123 return end;
124}
125static Pitch read_pitch(const char *text) {
126 Pitch pitch;
127 int end = pitch_prefix(text, &pitch);
128 if (!end || text[end]) fail("BAD PITCH %s", text);
129 return pitch;
130}
131static int midi(Note note) { return 12 * (note.octave + 1) + natural[note.pitch.letter] + note.pitch.alteration; }
132static int position(Note note) { return 7 * note.octave + note.pitch.letter; }
133static int pitch_class(Pitch pitch) { return mod(natural[pitch.letter] + pitch.alteration, 12); }
134static bool valid_note(Note note) { return note.octave >= -1 && note.octave <= 9 && midi(note) >= 0 && midi(note) <= 127; }
135static Note read_note(const char *text) {
136 Note note;
137 int end = pitch_prefix(text, &note.pitch);
138 if (!end) fail("BAD NOTE %s", text);
139 const char *octave = text + end;
140 if (!strcmp(octave, "-1")) note.octave = -1;
141 else if (octave[0] >= '0' && octave[0] <= '9' && !octave[1]) note.octave = octave[0] - '0';
142 else fail("BAD NOTE %s", text);
143 if (!valid_note(note)) fail("BAD NOTE %s", text);
144 return note;
145}
146static int read_number(const char *text) {
147 int result;
148 if (!canonical(text, 0, 127, &result)) fail("BAD NUMBER %s", text);
149 return result;
150}
151static int read_frequency(const char *text, bool is_tuning) {
152 int i = 0, whole = 0;
153 while (text[i] >= '0' && text[i] <= '9') {
154 if (i == 5) fail("BAD FREQUENCY %s", text);
155 whole = whole * 10 + text[i++] - '0';
156 }
157 if (!i) fail("BAD FREQUENCY %s", text);
158 int cents = whole * 100;
159 if (text[i] == '.') {
160 ++i;
161 int start = i, fraction = 0;
162 while (text[i] >= '0' && text[i] <= '9') {
163 if (i - start == 2) fail("BAD FREQUENCY %s", text);
164 fraction = fraction * 10 + text[i++] - '0';
165 }
166 if (i == start) fail("BAD FREQUENCY %s", text);
167 cents += i - start == 1 ? fraction * 10 : fraction;
168 }
169 if (text[i] || cents == 0 || (is_tuning && (cents < 40000 || cents > 48000))) fail("BAD FREQUENCY %s", text);
170 return cents;
171}

190static bool quality_delta(Quality quality, bool perfect, int *delta) {
191 if (quality == PERFECT) { if (!perfect) return false; *delta = 0; }
192 else if (quality == MAJOR || quality == MINOR) { if (perfect) return false; *delta = quality == MAJOR ? 0 : -1; }
193 else if (quality == AUGMENTED || quality == DOUBLE_AUGMENTED) *delta = quality == AUGMENTED ? 1 : 2;
194 else *delta = (quality == DIMINISHED ? -1 : -2) - !perfect;
195 return true;
196}
197static bool perfect_type(int number) { int r = (number - 1) % 7; return r == 0 || r == 3 || r == 4; }
198static Tone interval_tone(Interval interval) {
199 int s = interval.number - 1, delta;
200 quality_delta(interval.quality, perfect_type(interval.number), &delta);
201 return (Tone){s, 12 * (s / 7) + natural[s % 7] + delta};
202}
203static Interval read_interval(const char *text) {
204 for (int q = PERFECT; q <= DOUBLE_DIMINISHED; ++q) {
205 size_t length = strlen(quality_names[q]);
206 int number, delta;
207 if (!strncmp(text, quality_names[q], length) && canonical(text + length, 1, 99, &number)
208 && quality_delta((Quality)q, perfect_type(number), &delta)
209 && !(number == 1 && (q == DIMINISHED || q == DOUBLE_DIMINISHED))) return (Interval){(Quality)q, number};
210 }
211 fail("BAD INTERVAL %s", text);
212}
213static Text interval_text(Interval interval) {
214 Text result;
215 snprintf(result.text, sizeof result.text, "%s%d", quality_names[interval.quality], interval.number);
216 return result;
217}
218static Interval simple_interval(Interval interval) {
219 while (interval.number > 8) interval.number -= 7;
220 return interval;
221}
222static IntervalResult between(Note a, Note b) {
223 int distance = position(b) - position(a), semitones = midi(b) - midi(a);
224 bool down = distance < 0 || (distance == 0 && semitones < 0);
225 if (down) { distance = -distance; semitones = -semitones; }
226 int difference = semitones - 12 * (distance / 7) - natural[distance % 7];
227 for (int q = PERFECT; q <= DOUBLE_DIMINISHED; ++q) {
228 int delta;
229 if (quality_delta((Quality)q, perfect_type(distance + 1), &delta) && delta == difference)
230 return (IntervalResult){{(Quality)q, distance + 1}, down, semitones};
231 }
232 fail("UNNAMEABLE INTERVAL");
233}
234static Pitch raise_pitch(Pitch root, Tone tone) {
235 int target = root.letter + tone.letters;
236 int alteration = natural[root.letter] + root.alteration + tone.semitones - natural[target % 7] - 12 * (target / 7);
237 if (alteration < -2 || alteration > 2) fail("CANNOT SPELL");
238 return (Pitch){target % 7, alteration};
239}
240static void build(Pitch root, const Tone *tones, int count, Pitch *result) {
241 for (int i = 0; i < count; ++i) result[i] = raise_pitch(root, tones[i]);
242}

270static Numeral read_numeral(const char *text) {
271 int start = text[0] == 'b' || text[0] == '#' ? 1 : 0;
272 int end = start;
273 while (text[end] && strchr("IViv", text[end])) ++end;
274 for (int degree = 0; degree < 7; ++degree) {
275 for (int q = 0; q < 16; ++q) {
276 const char *part = (chords[q].lower ? roman_lower : roman_upper)[degree];
277 if (end - start == (int)strlen(part) && !strncmp(text + start, part, (size_t)(end - start))
278 && !strcmp(text + end, chords[q].numeral_suffix))
279 return (Numeral){start ? (text[0] == 'b' ? -1 : 1) : 0, degree, q};
280 }
281 }
282 fail("BAD NUMERAL %s", text);
283}

340static void execute(int word_count, char **words) {
341 int command = -1;
342 for (int i = 0; i < 24; ++i) if (!strcmp(words[0], commands[i].name)) { command = i; break; }
343 if (command < 0) fail("UNKNOWN COMMAND %s", words[0]);
344 int count = word_count - 1;
345 char **a = words + 1;
346 if (count < commands[command].minimum || count > commands[command].maximum) fail("ARGUMENT COUNT");
347 switch ((Command)command) {
348 case MIDI: case FREQ: case ENHARMONIC: {
349 Note note = read_note(a[0]);
350 int value = midi(note);
351 if (command == MIDI) out("MIDI %s %d\n", a[0], value);
352 else if (command == FREQ) {
353 int cents = (int)floor(tuning * exp2((value - 69) / 12.0) + 0.5);
354 out("FREQ %s %d.%02d\n", a[0], cents / 100, cents % 100);
355 } else {
356 out("ENHARMONIC %s", a[0]);
357 for (int octave = -1; octave <= 9; ++octave) for (int letter = 0; letter < 7; ++letter) {
358 int alteration = value - 12 * (octave + 1) - natural[letter];
359 if (alteration >= -2 && alteration <= 2) out(" %s", note_text((Note){{letter, alteration}, octave}).text);
360 }
361 out("\n");
362 }
363 break;
364 }
365 case NOTE: {
366 int value = read_number(a[0]);
367 out("NOTE %d %s %s\n", value, spelling(value, false).text, spelling(value, true).text);
368 break;
369 }
370 case TUNING: case PITCH: {
371 int frequency = read_frequency(a[0], command == TUNING);
372 if (command == TUNING) {
373 tuning = frequency;
374 out("TUNING %d.%02d\n", frequency / 100, frequency % 100);
375 } else {
376 double x = 69 + 12 * log2((double)frequency / tuning);
377 int value = (int)floor(x + 0.5);
378 if (value < 0 || value > 127) fail("OUT OF RANGE");
379 int cents = (int)floor(100 * (x - value) + 0.5);
380 out("PITCH %d.%02d %s %+d\n", frequency / 100, frequency % 100, spelling(value, false).text, cents);
381 }
382 break;
383 }
384 case INTERVAL: {
385 Note first = read_note(a[0]);
386 Note second = read_note(a[1]);
387 IntervalResult result = between(first, second);
388 Interval simple = simple_interval(result.interval);
389 const char *class = "dissonant";
390 if (simple.quality == PERFECT && (simple.number == 1 || simple.number == 5 || simple.number == 8)) class = "perfect";
391 else if ((simple.quality == MAJOR || simple.quality == MINOR) && (simple.number == 3 || simple.number == 6)) class = "imperfect";
392 out("INTERVAL %s %s %s %s semitones=%d simple=%s class=%s\n", a[0], a[1], interval_text(result.interval).text,
393 result.down ? "down" : "up", result.semitones, interval_text(simple).text, class);
394 break;
395 }
396 case INVERT: {
397 Interval simple = simple_interval(read_interval(a[0]));
398 static const Quality inversions[] = {PERFECT, MINOR, MAJOR, DIMINISHED, AUGMENTED, DOUBLE_DIMINISHED, DOUBLE_AUGMENTED};
399 Interval inverse = {inversions[simple.quality], 9 - simple.number};
400 if (inverse.number == 1 && (inverse.quality == DIMINISHED || inverse.quality == DOUBLE_DIMINISHED)) fail("CANNOT INVERT");
401 out("INVERT %s %s %s\n", a[0], interval_text(simple).text, interval_text(inverse).text);
402 break;
403 }
404 case TRANSPOSE: {
405 Note note = read_note(a[0]);
406 Tone tone = interval_tone(read_interval(a[1]));
407 if (strcmp(a[2], "UP") && strcmp(a[2], "DOWN")) fail("BAD DIRECTION %s", a[2]);
408 int sign = !strcmp(a[2], "UP") ? 1 : -1;
409 int target = position(note) + sign * tone.letters, value = midi(note) + sign * tone.semitones;
410 int letter = mod(target, 7), octave = floor_div(target, 7);
411 int alteration = value - 12 * (octave + 1) - natural[letter];
412 if (alteration < -2 || alteration > 2) fail("CANNOT SPELL");
413 Note result = {{letter, alteration}, octave};
414 if (!valid_note(result)) fail("OUT OF RANGE");
415 out("TRANSPOSE %s %s %s %s\n", a[0], a[1], a[2], note_text(result).text);
416 break;
417 }
418 case SCALE: {
419 Pitch root = read_pitch(a[0]);
420 const Scale *scale = &scales[read_scale(a[1])];
421 Pitch pitches[7];
422 build(root, scale->tones, scale->count, pitches);
423 out("SCALE %s %s", a[0], a[1]);
424 print_pitches(pitches, scale->count);
425 out(" steps=");
426 for (int i = 0; i < scale->count; ++i) {
427 int next = i + 1 == scale->count ? 12 : scale->tones[i + 1].semitones;
428 out("%s%d", i ? "," : "", next - scale->tones[i].semitones);
429 }
430 out("\n");
431 break;
432 }
433 case CHORD: case VOICE: {
434 Note note = {{0,0}, 0};
435 Pitch root;
436 if (command == VOICE) { note = read_note(a[0]); root = note.pitch; }
437 else root = read_pitch(a[0]);
438 int quality = read_quality(a[1]), inversion = 0;
439 const Chord *chord = &chords[quality];
440 if (count == 3 && !canonical(a[2], 0, chord->count - 1, &inversion)) fail("BAD INVERSION %s", a[2]);
441 Pitch pitches[5], ordered[5];
442 build(root, chord->tones, chord->count, pitches);
443 for (int i = 0; i < chord->count; ++i) ordered[i] = pitches[(i + inversion) % chord->count];
444 Text symbol = symbol_text(root, quality, inversion ? &ordered[0] : NULL);
445 if (command == CHORD) {
446 out("CHORD %s", symbol.text);
447 print_pitches(ordered, chord->count);
448 out("\n");
449 } else {
450 Note voiced[5];
451 int value = midi(note);
452 for (int i = 0; i < chord->count; ++i) {
453 if (i || inversion) {
454 int difference = mod(pitch_class(ordered[i]) - value, 12);
455 value += difference ? difference : 12;
456 }
457 voiced[i] = make_note(ordered[i], value);
458 if (!valid_note(voiced[i])) fail("OUT OF RANGE");
459 }
460 out("VOICE %s", symbol.text);
461 for (int i = 0; i < chord->count; ++i) out(" %s", note_text(voiced[i]).text);
462 out(" midi=");
463 for (int i = 0; i < chord->count; ++i) out("%s%d", i ? "," : "", midi(voiced[i]));
464 out("\n");
465 }
466 break;
467 }
468 case IDENTIFY: case MELODY: {
469 Note notes[15];
470 for (int i = 0; i < count; ++i) notes[i] = read_note(a[i]);
471 if (command == MELODY) {
472 int low = 0, high = 0, leaps = 0;
473 for (int i = 1; i < count; ++i) {
474 if (midi(notes[i]) < midi(notes[low])) low = i;
475 if (midi(notes[i]) > midi(notes[high])) high = i;
476 if (abs(midi(notes[i]) - midi(notes[i - 1])) >= 3) ++leaps;
477 }
478 Interval range = between(notes[low], notes[high]).interval;
479 IntervalResult steps[14];
480 for (int i = 1; i < count; ++i) steps[i - 1] = between(notes[i - 1], notes[i]);
481 out("MELODY n=%d low=%s high=%s range=%s leaps=%d steps=", count, a[low], a[high], interval_text(range).text, leaps);
482 for (int i = 0; i < count - 1; ++i) out("%s%c%s", i ? "," : "", steps[i].down ? '-' : '+', interval_text(steps[i].interval).text);
483 out("\n");
484 } else {
485 for (int i = 1; i < count; ++i) {
486 Note value = notes[i];
487 int j = i;
488 while (j > 0 && midi(notes[j - 1]) > midi(value)) { notes[j] = notes[j - 1]; --j; }
489 notes[j] = value;
490 }
491 unsigned set = 0;
492 int unique = 0;
493 for (int i = 0; i < count; ++i) {
494 unsigned bit = 1u << pitch_class(notes[i].pitch);
495 if (!(set & bit)) { set |= bit; notes[unique++] = notes[i]; }
496 }
497 for (int i = 0; i < unique; ++i) for (int q = 0; q < 16; ++q) {
498 unsigned candidate = 0;
499 for (int j = 0; j < chords[q].count; ++j) candidate |= 1u << mod(midi(notes[i]) + chords[q].tones[j].semitones, 12);
500 if (candidate != set) continue;
501 int inversion = 0;
502 while (mod(midi(notes[i]) + chords[q].tones[inversion].semitones, 12) != pitch_class(notes[0].pitch)) ++inversion;
503 out("IDENTIFY %s root=%s quality=%s inversion=%d\n", symbol_text(notes[i].pitch, q, inversion ? &notes[0].pitch : NULL).text,
504 pitch_text(notes[i].pitch).text, chords[q].name, inversion);
505 return;
506 }
507 out("IDENTIFY UNKNOWN\n");
508 }
509 break;
510 }
511 case CIRCLE: {
512 int mode = read_mode(a[0]);
513 out("CIRCLE %s", a[0]);
514 for (int p = -7; p <= 7; ++p) out(" %s", pitch_text(fifth_pitch(p - modes[mode].offset)).text);
515 out("\n");
516 break;
517 }
518 case FINDKEY: {
519 Pitch pitches[15];
520 for (int i = 0; i < count; ++i) pitches[i] = read_pitch(a[i]);
521 Key matches[30];
522 int matched = 0;
523 for (int p = -7; p <= 7; ++p) for (int mode = 0; mode < 2; ++mode) {
524 Key key = make_key(fifth_pitch(p - modes[mode].offset), mode);
525 bool match = true;
526 for (int i = 0; i < count; ++i) {
527 Pitch degree = key.scale[degree_of(key, pitches[i])];
528 if (degree.alteration != pitches[i].alteration) match = false;
529 }
530 if (match) matches[matched++] = key;
531 }
532 out("FINDKEY %d", matched);
533 for (int i = 0; i < matched; ++i) out("%s%s %s", i ? ", " : " ", pitch_text(matches[i].tonic).text, modes[matches[i].mode].name);
534 out("\n");
535 break;
536 }
537 case SPELL: {
538 int value = read_number(a[0]);
539 Key key = read_key(a + 1);
540 Pitch pitch = read_pitch((key.fifths < 0 ? flat_names : sharp_names)[value % 12]);
541 for (int i = 0; i < 7; ++i) if (pitch_class(key.scale[i]) == value % 12) { pitch = key.scale[i]; break; }
542 Note note = make_note(pitch, value);
543 if (!valid_note(note)) fail("OUT OF RANGE");
544 out("SPELL %d %s %s %s %s\n", value, a[1], a[2], note_text(note).text, degree_label(key, pitch).text);
545 break;
546 }
547 case KEYSIG: case RELATIVE: case DIATONIC: case PROGRESSION: case ANALYZE: case DEGREE: case KEYDIST: {
548 Key key = read_key(a);
549 if (command == KEYSIG) {
550 out("KEYSIG %s %s", a[0], a[1]);
551 if (!key.fifths) out(" none");
552 else {
553 static const char *sharps[] = {"F#", "C#", "G#", "D#", "A#", "E#", "B#"};
554 static const char *flats[] = {"Bb", "Eb", "Ab", "Db", "Gb", "Cb", "Fb"};
555 out(" %s=%d", key.fifths > 0 ? "sharps" : "flats", abs(key.fifths));
556 for (int i = 0; i < abs(key.fifths); ++i) out(" %s", (key.fifths > 0 ? sharps : flats)[i]);
557 }
558 out("\n");
559 } else if (command == RELATIVE) {
560 out("RELATIVE %s %s major=%s minor=%s\n", a[0], a[1], pitch_text(fifth_pitch(key.fifths)).text, pitch_text(fifth_pitch(key.fifths + 3)).text);
561 } else if (command == DEGREE) {
562 Pitch pitch = read_pitch(a[2]);
563 out("DEGREE %s %s %s %s\n", a[0], a[1], a[2], degree_label(key, pitch).text);
564 } else if (command == KEYDIST) {
565 Key other = read_key(a + 2);
566 unsigned first = key_set(key), second = key_set(other), common = first & second;
567 int r = mod(other.fifths - key.fifths, 12), fifths = r > 6 ? r - 12 : r, count_common = 0;
568 while (common) { count_common += common & 1u; common >>= 1; }
569 bool same_tonic = pitch_class(key.tonic) == pitch_class(other.tonic);
570 const char *relation = first == second ? (same_tonic ? "same" : "relative")
571 : same_tonic ? "parallel" : abs(fifths) == 1 ? "close" : "distant";
572 out("KEYDIST %s %s %s %s fifths=", a[0], a[1], a[2], a[3]);
573 out(fifths > 0 ? "+%d" : "%d", fifths);
574 out(" common=%d relation=%s\n", count_common, relation);
575 } else if (command == DIATONIC) {
576 out("DIATONIC %s %s\n", a[0], a[1]);
577 for (int size = 3; size <= 4; ++size) {
578 out(size == 3 ? "TRIADS" : "SEVENTHS");
579 for (int i = 0; i < 7; ++i) {
580 int root = pitch_class(key.scale[i]);
581 int third = mod(pitch_class(key.scale[(i + 2) % 7]) - root, 12);
582 int fifth = mod(pitch_class(key.scale[(i + 4) % 7]) - root, 12);
583 int quality;
584 if (size == 3) quality = third == 4 ? 0 : fifth == 7 ? 1 : 2;
585 else {
586 int seventh = mod(pitch_class(key.scale[(i + 6) % 7]) - root, 12);
587 quality = third == 4 ? (seventh == 11 ? 9 : 8) : fifth == 7 ? 10 : 12;
588 }
589 out(" %s=%s", roman_text(i, quality).text, symbol_text(key.scale[i], quality, NULL).text);
590 }
591 out("\n");
592 }
593 } else if (command == ANALYZE) {
594 Symbol symbols[13];
595 for (int i = 2; i < count; ++i) symbols[i - 2] = read_symbol(a[i]);
596 out("ANALYZE %s %s", a[0], a[1]);
597 for (int i = 0; i < count - 2; ++i) out(" %s%s", degree_prefix(key, symbols[i].root).text,
598 roman_text(degree_of(key, symbols[i].root), symbols[i].quality).text);
599 out("\n");
600 } else {
601 int length = count - 2;
602 Numeral numerals[13];
603 Pitch roots[13], tones[13][5];
604 for (int i = 0; i < length; ++i) numerals[i] = read_numeral(a[i + 2]);
605 for (int i = 0; i < length; ++i) {
606 Numeral numeral = numerals[i];
607 roots[i] = key.scale[numeral.degree];
608 roots[i].alteration += numeral.alteration;
609 build(roots[i], chords[numeral.quality].tones, chords[numeral.quality].count, tones[i]);
610 }
611 out("PROGRESSION %s %s %d\n", a[0], a[1], length);
612 for (int i = 0; i < length; ++i) {
613 out("%s %s", a[i + 2], symbol_text(roots[i], numerals[i].quality, NULL).text);
614 print_pitches(tones[i], chords[numerals[i].quality].count);
615 out("\n");
616 }
617 const char *cadence = "none";
618 if (length >= 2) {
619 Numeral x = numerals[length - 2], y = numerals[length - 1];
620 if (!x.alteration && !y.alteration && x.degree == 4 && y.degree == 0) cadence = "authentic";
621 else if (!x.alteration && !y.alteration && x.degree == 3 && y.degree == 0) cadence = "plagal";
622 else if (!x.alteration && !y.alteration && x.degree == 4 && y.degree == 5) cadence = "deceptive";
623 else if (!y.alteration && y.degree == 4) cadence = "half";
624 }
625 out("CADENCE %s\n", cadence);
626 }
627 break;
628 }
629 }
630}

634 if (!fgets(line, sizeof line, stdin)) return 0;
635 int count = (int)strtol(line, NULL, 10);

637 if (!fgets(line, sizeof line, stdin)) break;
638 char *words[16];
639 int length = 0;
640 for (char *word = strtok(line, " \r\n"); word; word = strtok(NULL, " \r\n")) words[length++] = word;
641 response_length = 0;
642 if (!setjmp(command_error)) execute(length, words);
643 fputs(response, stdout);
644 }
