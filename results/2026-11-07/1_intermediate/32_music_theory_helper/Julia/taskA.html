15function bounded_integer(text::AbstractString, kind::String, low::Int, high::Int)
16 length(text) <= 3 && occursin(r"^(0|[1-9][0-9]*)$", text) || bad(kind, text)
17 value = parse(Int, text; base = 10)
18 low <= value <= high || bad(kind, text)
19 value
20end
21function frequency(text::AbstractString, tuning::Bool)
22 occursin(r"^[0-9]{1,5}(\.[0-9]{1,2})?$", text) || bad("FREQUENCY", text)
23 parts = split(text, '.')
24 value = parse(Int, parts[1]; base = 10) * 100
25 if length(parts) == 2
26 value += parse(Int, parts[2]; base = 10) * (length(parts[2]) == 1 ? 10 : 1)
27 end
28 (value > 0 && (!tuning || 40000 <= value <= 48000)) || bad("FREQUENCY", text)
29 value
30end
31frequency_text(cents::Int) = string(div(cents, 100), '.', lpad(string(rem(cents, 100)), 2, '0'))

41function pitch_prefix(text::AbstractString, kind::String)
42 isempty(text) && bad(kind, text)
43 index = findfirst(==(text[1]), LETTERS)
44 isnothing(index) && bad(kind, text)
45 position = 2
46 alteration = 0
47 if position <= lastindex(text)
48 if text[position] == 'b'
49 alteration = -1
50 position += 1
51 if position <= lastindex(text) && text[position] == 'b'
52 alteration = -2
53 position += 1
54 end
55 elseif text[position] == '#'
56 alteration = 1
57 position += 1
58 elseif text[position] == 'x'
59 alteration = 2
60 position += 1
61 end
62 end
63 (pitch = Pitch(index - 1, alteration), rest = text[position:end])
64end

68 parsed.pitch

93@enum IntervalQuality PERFECT MAJOR MINOR AUGMENTED DIMINISHED DOUBLE_AUGMENTED DOUBLE_DIMINISHED
94const QUALITY_TEXT = ("P", "M", "m", "A", "d", "AA", "dd")
95perfect_type(number::Int) = mod(number - 1, 7) in (0, 3, 4)
96function quality_delta(quality::IntervalQuality, perfect::Bool)::Union{Int,Nothing}
97 quality == PERFECT && return (perfect ? 0 : nothing)
98 quality == MAJOR && return (perfect ? nothing : 0)
99 quality == MINOR && return (perfect ? nothing : -1)
100 quality == AUGMENTED && return 1
101 quality == DOUBLE_AUGMENTED && return 2
102 quality == DIMINISHED && return (perfect ? -1 : -2)
103 perfect ? -2 : -3
104end

109struct Interval
110 quality::IntervalQuality
111 number::Int
112end
113Base.show(io::IO, interval::Interval) = print(io, QUALITY_TEXT[Int(interval.quality) + 1], interval.number)
114forbidden(interval::Interval) = interval.number == 1 && interval.quality in (DIMINISHED, DOUBLE_DIMINISHED)
115function parse_interval(text::AbstractString)
116 for quality in instances(IntervalQuality)
117 prefix = QUALITY_TEXT[Int(quality) + 1]
118 startswith(text, prefix) || continue
119 rest = text[length(prefix) + 1:end]
120 occursin(r"^[1-9][0-9]?$", rest) || continue
121 number = parse(Int, rest; base = 10)
122 interval = Interval(quality, number)
123 if !isnothing(quality_delta(quality, perfect_type(number))) && !forbidden(interval)
124 return interval
125 end
126 end
127 bad("INTERVAL", text)
128end
129function simple_interval(interval::Interval)
130 number = interval.number
131 while number > 8
132 number -= 7
133 end
134 Interval(interval.quality, number)
135end
136function interval_tone(interval::Interval)
137 distance = interval.number - 1
138 Tone(distance, 12 * div(distance, 7) + NATURALS[mod(distance, 7) + 1] + something(quality_delta(interval.quality, perfect_type(interval.number))))
139end
140function consonance(interval::Interval)
141 simple = simple_interval(interval)
142 simple.quality == PERFECT && simple.number in (1, 5, 8) && return "perfect"
143 simple.quality in (MAJOR, MINOR) && simple.number in (3, 6) && return "imperfect"
144 "dissonant"
145end
146struct Motion
147 interval::Interval
148 down::Bool
149 semitones::Int
150end
151function between(first::Note, second::Note)
152 distance = note_position(second) - note_position(first)
153 semitones = midi(second) - midi(first)
154 down = distance < 0 || (distance == 0 && semitones < 0)
155 if down
156 distance = -distance
157 semitones = -semitones
158 end
159 difference = semitones - 12 * div(distance, 7) - NATURALS[mod(distance, 7) + 1]
160 for quality in instances(IntervalQuality)
161 quality_delta(quality, perfect_type(distance + 1)) == difference && return Motion(Interval(quality, distance + 1), down, semitones)
162 end
163 fail("UNNAMEABLE INTERVAL")
164end
165function raise_pitch(root::Pitch, tone::Tone)
166 target = root.letter + tone.letters
167 alteration = NATURALS[root.letter + 1] + root.alteration + tone.semitones - NATURALS[mod(target, 7) + 1] - 12 * fld(target, 7)
168 -2 <= alteration <= 2 || fail("CANNOT SPELL")
169 Pitch(mod(target, 7), alteration)
170end

269function parse_numeral(text::AbstractString)
270 start = text[1] in ('b', '#') ? 2 : 1
271 alteration = start == 1 ? 0 : text[1] == 'b' ? -1 : 1
272 finish = start
273 while finish <= lastindex(text) && text[finish] in ('I', 'V', 'i', 'v')
274 finish += 1
275 end
276 part = text[start:finish - 1]
277 suffix = text[finish:end]
278 for degree in eachindex(ROMANS), quality in eachindex(CHORDS)
279 chord = CHORDS[quality]
280 expected = chord.lower ? lowercase(ROMANS[degree]) : ROMANS[degree]
281 part == expected && suffix == chord.numeral_suffix && return Numeral(alteration, degree, quality)
282 end
283 bad("NUMERAL", text)
284end

289 ChordSymbol(parsed.pitch, quality)

292function identify(notes::Vector{Note})
293 ordered = notes[sortperm(midi.(notes))]
294 present = Set{Int}()
295 candidates = Note[]
296 for note in ordered
297 pitch_class(note.pitch) in present && continue
298 push!(present, pitch_class(note.pitch))
299 push!(candidates, note)
300 end
301 bass = candidates[1].pitch
302 for root in candidates, chord in CHORDS
303 possible = [mod(midi(root) + tone.semitones, 12) for tone in chord.tones]
304 Set(possible) == present || continue
305 inversion = something(findfirst(==(pitch_class(bass)), possible)) - 1
306 symbol = chord_symbol(root.pitch, chord, inversion == 0 ? nothing : bass)
307 return "IDENTIFY $symbol root=$(root.pitch) quality=$(chord.name) inversion=$inversion"
308 end
309 "IDENTIFY UNKNOWN"
310end
311function melody(notes::Vector{Note}, arguments::Vector{String})
312 values = midi.(notes)
313 low, high = argmin(values), argmax(values)
314 range = between(notes[low], notes[high]).interval
315 steps = String[]
316 for index in 2:length(notes)
317 motion = between(notes[index - 1], notes[index])
318 push!(steps, string(motion.down ? '-' : '+', motion.interval))
319 end
320 leaps = count(value -> abs(value) >= 3, diff(values))
321 "MELODY n=$(length(notes)) low=$(arguments[low]) high=$(arguments[high]) range=$range leaps=$leaps steps=$(join(steps, ','))"
322end
323function cadence(numerals::Vector{Numeral})
324 length(numerals) < 2 && return "none"
325 first, second = numerals[end - 1], numerals[end]
326 if first.alteration == 0 && second.alteration == 0
327 first.degree == 5 && second.degree == 1 && return "authentic"
328 first.degree == 4 && second.degree == 1 && return "plagal"
329 first.degree == 5 && second.degree == 6 && return "deceptive"
330 end
331 second.alteration == 0 && second.degree == 5 ? "half" : "none"
332end
333function diatonic(key::Key)
334 rows = ["DIATONIC $key"]
335 classes = pitch_class.(key.scale)
336 for size in 3:4
337 items = String[]
338 for index in 1:7
339 third = mod(classes[mod(index + 1, 7) + 1] - classes[index], 12)
340 fifth = mod(classes[mod(index + 3, 7) + 1] - classes[index], 12)
341 quality = if size == 3
342 third == 4 ? 1 : fifth == 7 ? 2 : 3
343 else
344 seventh = mod(classes[mod(index + 5, 7) + 1] - classes[index], 12)
345 third == 4 ? (seventh == 11 ? 10 : 9) : fifth == 7 ? 11 : 13
346 end
347 chord = CHORDS[quality]
348 push!(items, "$(roman(index, chord))=$(chord_symbol(key.scale[index], chord))")
349 end
350 push!(rows, string(size == 3 ? "TRIADS " : "SEVENTHS ", join(items, ' ')))
351 end
352 rows
353end

355const ARITIES = Dict(
356 "MIDI" => (1,1), "NOTE" => (1,1), "FREQ" => (1,1), "TUNING" => (1,1), "PITCH" => (1,1),
357 "INTERVAL" => (2,2), "INVERT" => (1,1), "TRANSPOSE" => (3,3), "SCALE" => (2,2),
358 "CHORD" => (2,3), "VOICE" => (2,3), "IDENTIFY" => (2,15), "MELODY" => (2,15),
359 "ENHARMONIC" => (1,1), "KEYSIG" => (2,2), "RELATIVE" => (2,2), "CIRCLE" => (1,1),
360 "DIATONIC" => (2,2), "PROGRESSION" => (3,15), "ANALYZE" => (3,15), "FINDKEY" => (1,15),
361 "DEGREE" => (3,3), "SPELL" => (3,3), "KEYDIST" => (4,4),
362)
363mutable struct Calculator
364 tuning::Int
365end

367function execute(calculator::Calculator, line::String)::Vector{String}
368 tokens = String.(split(line, ' '))
369 command = tokens[1]
370 haskey(ARITIES, command) || fail("UNKNOWN COMMAND $command")
371 arguments = tokens[2:end]
372 lower, upper = ARITIES[command]
373 lower <= length(arguments) <= upper || fail("ARGUMENT COUNT")
374 if command in ("MIDI", "FREQ", "ENHARMONIC")
375 note = parse_note(arguments[1])
376 value = midi(note)
377 command == "MIDI" && return ["MIDI $(arguments[1]) $value"]
378 if command == "FREQ"
379 cents = floor(Int, calculator.tuning * exp2((value - 69) / 12) + 0.5)
380 return ["FREQ $(arguments[1]) $(frequency_text(cents))"]
381 end
382 equivalents = Note[]
383 for octave in -1:9, letter in 0:6
384 alteration = value - 12 * (octave + 1) - NATURALS[letter + 1]
385 -2 <= alteration <= 2 && push!(equivalents, Note(Pitch(letter, alteration), octave))
386 end
387 return ["ENHARMONIC $(arguments[1]) $(join(equivalents, ' '))"]
388 elseif command == "NOTE"
389 value = bounded_integer(arguments[1], "NUMBER", 0, 127)
390 return ["NOTE $value $(spelling(value)) $(spelling(value, true))"]
391 elseif command in ("TUNING", "PITCH")
392 value = frequency(arguments[1], command == "TUNING")
393 if command == "TUNING"
394 calculator.tuning = value
395 return ["TUNING $(frequency_text(value))"]
396 end
397 height = 69 + 12 * log2(value / calculator.tuning)
398 nearest = floor(Int, height + 0.5)
399 0 <= nearest <= 127 || fail("OUT OF RANGE")
400 cents = floor(Int, 100 * (height - nearest) + 0.5)
401 sign = cents >= 0 ? "+" : ""
402 return ["PITCH $(frequency_text(value)) $(spelling(nearest)) $sign$cents"]
403 elseif command == "INTERVAL"
404 first = parse_note(arguments[1])
405 second = parse_note(arguments[2])
406 motion = between(first, second)
407 direction = motion.down ? "down" : "up"
408 return ["INTERVAL $(arguments[1]) $(arguments[2]) $(motion.interval) $direction semitones=$(motion.semitones) simple=$(simple_interval(motion.interval)) class=$(consonance(motion.interval))"]
409 elseif command == "INVERT"
410 interval = parse_interval(arguments[1])
411 simple = simple_interval(interval)
412 inverses = (PERFECT, MINOR, MAJOR, DIMINISHED, AUGMENTED, DOUBLE_DIMINISHED, DOUBLE_AUGMENTED)
413 inverse = Interval(inverses[Int(simple.quality) + 1], 9 - simple.number)
414 forbidden(inverse) && fail("CANNOT INVERT")
415 return ["INVERT $(arguments[1]) $simple $inverse"]
416 elseif command == "TRANSPOSE"
417 note = parse_note(arguments[1])
418 interval = parse_interval(arguments[2])
419 direction = arguments[3]
420 direction in ("UP", "DOWN") || bad("DIRECTION", direction)
421 sign = direction == "UP" ? 1 : -1
422 tone = interval_tone(interval)
423 target = note_position(note) + sign * tone.letters
424 value = midi(note) + sign * tone.semitones
425 letter, octave = mod(target, 7), fld(target, 7)
426 alteration = value - 12 * (octave + 1) - NATURALS[letter + 1]
427 -2 <= alteration <= 2 || fail("CANNOT SPELL")
428 result = Note(Pitch(letter, alteration), octave)
429 valid_note(result) || fail("OUT OF RANGE")
430 return ["TRANSPOSE $(arguments[1]) $(arguments[2]) $direction $result"]
431 elseif command == "SCALE"
432 root = parse_pitch(arguments[1])
433 scale = SCALES[named_index(arguments[2], "SCALE", SCALES)]
434 pitches = build(root, scale.tones)
435 semitones = [tone.semitones for tone in scale.tones]
436 steps = diff(vcat(semitones, 12))
437 return ["SCALE $(arguments[1]) $(arguments[2]) $(join(pitches, ' ')) steps=$(join(steps, ','))"]
438 elseif command in ("CHORD", "VOICE")
439 input = command == "VOICE" ? parse_note(arguments[1]) : nothing
440 root = isnothing(input) ? parse_pitch(arguments[1]) : input.pitch
441 quality = named_index(arguments[2], "QUALITY", CHORDS)
442 chord = CHORDS[quality]
443 inversion = length(arguments) == 3 ? bounded_integer(arguments[3], "INVERSION", 0, length(chord.tones) - 1) : 0
444 tones = build(root, chord.tones)
445 ordered = vcat(tones[inversion + 1:end], tones[1:inversion])
446 symbol = chord_symbol(root, chord, inversion == 0 ? nothing : ordered[1])
447 command == "CHORD" && return ["CHORD $symbol $(join(ordered, ' '))"]
448 value = midi(something(input))
449 notes = Note[]
450 for (index, pitch) in enumerate(ordered)
451 if index > 1 || inversion > 0
452 difference = mod(pitch_class(pitch) - value, 12)
453 value += difference == 0 ? 12 : difference
454 end
455 note = make_note(pitch, value)
456 valid_note(note) || fail("OUT OF RANGE")
457 push!(notes, note)
458 end
459 return ["VOICE $symbol $(join(notes, ' ')) midi=$(join(midi.(notes), ','))"]
460 elseif command in ("IDENTIFY", "MELODY")
461 notes = parse_note.(arguments)
462 return [command == "IDENTIFY" ? identify(notes) : melody(notes, arguments)]
463 elseif command == "CIRCLE"
464 mode = named_index(arguments[1], "MODE", MODES)
465 pitches = [fifth_pitch(fifths - MODES[mode].offset) for fifths in -7:7]
466 return ["CIRCLE $(arguments[1]) $(join(pitches, ' '))"]
467 elseif command == "FINDKEY"
468 pitches = parse_pitch.(arguments)
469 keys = Key[]
470 for fifths in -7:7, mode in 1:2
471 key = make_key(fifth_pitch(fifths - MODES[mode].offset), mode)
472 all(pitch -> pitch in key.scale, pitches) && push!(keys, key)
473 end
474 return ["FINDKEY $(length(keys))" * (isempty(keys) ? "" : " " * join(keys, ", "))]
475 elseif command == "SPELL"
476 value = bounded_integer(arguments[1], "NUMBER", 0, 127)
477 key = parse_key(arguments, 2)
478 index = findfirst(pitch -> pitch_class(pitch) == mod(value, 12), key.scale)
479 pitch = isnothing(index) ? spelling(value, key.fifths < 0).pitch : key.scale[index]
480 note = make_note(pitch, value)
481 valid_note(note) || fail("OUT OF RANGE")
482 return ["SPELL $value $key $note $(key_label(key, pitch))"]
483 end
484 key = parse_key(arguments)
485 if command == "KEYSIG"
486 value = key.fifths
487 value == 0 && return ["KEYSIG $key none"]
488 names = split(value > 0 ? "F# C# G# D# A# E# B#" : "Bb Eb Ab Db Gb Cb Fb")
489 kind = value > 0 ? "sharps" : "flats"
490 return ["KEYSIG $key $kind=$(abs(value)) $(join(names[1:abs(value)], ' '))"]
491 elseif command == "RELATIVE"
492 return ["RELATIVE $key major=$(fifth_pitch(key.fifths)) minor=$(fifth_pitch(key.fifths + 3))"]
493 elseif command == "DIATONIC"
494 return diatonic(key)
495 elseif command == "DEGREE"
496 pitch = parse_pitch(arguments[3])
497 return ["DEGREE $key $(arguments[3]) $(key_label(key, pitch))"]
498 elseif command == "KEYDIST"
499 other = parse_key(arguments, 3)
500 distance = mod(other.fifths - key.fifths, 12)
501 distance > 6 && (distance -= 12)
502 first_set, second_set = key_set(key), key_set(other)
503 same_tonic = pitch_class(key.tonic) == pitch_class(other.tonic)
504 relation = if first_set == second_set
505 same_tonic ? "same" : "relative"
506 elseif same_tonic
507 "parallel"
508 else
509 abs(distance) == 1 ? "close" : "distant"
510 end
511 sign = distance > 0 ? "+" : ""
512 return ["KEYDIST $key $other fifths=$sign$distance common=$(length(intersect(first_set, second_set))) relation=$relation"]
513 elseif command == "ANALYZE"
514 symbols = parse_symbol.(arguments[3:end])
515 numerals = [key_prefix(key, symbol.root) * roman(key_degree(key, symbol.root), CHORDS[symbol.quality]) for symbol in symbols]
516 return ["ANALYZE $key $(join(numerals, ' '))"]
517 else
518 numerals = parse_numeral.(arguments[3:end])
519 rows = ["PROGRESSION $key $(length(numerals))"]
520 for (index, numeral) in enumerate(numerals)
521 degree = key.scale[numeral.degree]
522 root = Pitch(degree.letter, degree.alteration + numeral.alteration)
523 chord = CHORDS[numeral.quality]
524 tones = build(root, chord.tones)
525 push!(rows, "$(arguments[index + 2]) $(chord_symbol(root, chord)) $(join(tones, ' '))")
526 end
527 push!(rows, "CADENCE $(cadence(numerals))")
528 return rows
529 end
530end
