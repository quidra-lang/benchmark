109function simulate(route: number[], print = false): Run {
110 let time = departure,
111 previous = 0,
112 waiting = 0,
113 service = 0,
114 violations = 0,
115 lateness = 0;
116 for (const id of route) {
117 const stop = getStop(id),
118 arrival = time + distance(previous, id),
119 wait = stop.open === null ? 0 : Math.max(0, stop.open - arrival),
120 begin = arrival + wait;
121 const late = stop.close === null ? 0 : Math.max(0, arrival - stop.close);
122 time = begin + stop.service;
123 waiting += wait;
124 service += stop.service;
125 if (late > 0) violations++;
126 lateness += late;
127 if (print)
128 output.push(
129 `${id} arrive ${clock(arrival)} wait ${wait} begin ${clock(begin)} leave ${clock(time)} ${late ? "late " + late : "ok"}`,
130 );
131 previous = id;
132 }
133 time += distance(previous, 0);
134 if (print)
135 output.push(
136 `return ${clock(time)}, travel ${length(route)}, wait ${waiting}, service ${service}, violations ${violations}, lateness ${lateness}`,
137 );
138 return { time, violations, lateness };

140function twoOpt(route: number[], announce: boolean): number {
141 let moves = 0;
142 while (true) {
143 let changed = false;
144 scan: for (let i = 0; i + 1 < route.length; i++)
145 for (let j = i + 1; j < route.length; j++) {
146 const before = route[i - 1] ?? 0,
147 after = route[j + 1] ?? 0;
148 const gain =
149 distance(before, route[i]) +
150 distance(route[j], after) -
151 distance(before, route[j]) -
152 distance(route[i], after);
153 if (gain > 0) {
154 if (announce) output.push(`reverse ${route[i]}..${route[j]} gain ${gain}`);
155 route.splice(i, j - i + 1, ...route.slice(i, j + 1).reverse());
156 moves++;
157 changed = true;
158 break scan;
159 }
160 }
161 if (!changed) return moves;
162 }

164function exact(): number[] {
165 const names = ids(),
166 memo = new Map<string, number>();
167 function best(previous: number, mask: number): number {
168 if (mask === 0) return distance(previous, 0);
169 const key = previous + ":" + mask;
170 if (memo.has(key)) return memo.get(key)!;
171 let value = Infinity;
172 for (let i = 0; i < names.length; i++)
173 if (mask & (1 << i))
174 value = Math.min(
175 value,
176 distance(previous, names[i]) + best(names[i], mask ^ (1 << i)),
177 );
178 memo.set(key, value);
179 return value;
180 }
181 const result: number[] = [];
182 let previous = 0,
183 mask = (1 << names.length) - 1;
184 while (mask) {
185 const target = best(previous, mask);
186 for (let i = 0; i < names.length; i++)
187 if (
188 mask & (1 << i) &&
189 distance(previous, names[i]) + best(names[i], mask ^ (1 << i)) === target
190 ) {
191 result.push(names[i]);
192 previous = names[i];
193 mask ^= 1 << i;
194 break;
195 }
196 }
197 return result;
198}
199function spanning(names: number[], first: number): { total: number; edges: string[] } {
200 const tree = [first],
201 remaining = names.filter((id) => id !== first),
202 edges: string[] = [];
203 let total = 0;
204 while (remaining.length) {
205 let chosen = -1,
206 parent = -1,
207 weight = Infinity;
208 for (const id of remaining)
209 for (const from of tree)
210 if (distance(from, id) < weight) {
211 chosen = id;
212 parent = from;
213 weight = distance(from, id);
214 }
215 total += weight;
216 edges.push(location(parent) + "-" + chosen);
217 tree.push(chosen);
218 remaining.splice(remaining.indexOf(chosen), 1);
219 }
220 return { total, edges };
221}
222function improvePlan(command: string): void {
223 const value = requirePlan(),
224 oldLength = planLength(value),
225 oldCount = value.routes.length;
226 let moves = 0;
227 function step(): boolean {
228 const routes = value.routes,
229 capacity = value.capacity;
230 if (command === "RELOCATE") {
231 for (let r = 0; r < routes.length; r++)
232 for (let p = 0; p < routes[r].length; p++)
233 for (let t = 0; t < routes.length; t++)
234 if (t !== r) {
235 const a = routes[r],
236 b = routes[t],
237 c = a[p];
238 if (load(b) + getStop(c).demand > capacity) continue;
239 for (let j = 0; j <= b.length; j++) {
240 const before = a[p - 1] ?? 0,
241 after = a[p + 1] ?? 0,
242 u = b[j - 1] ?? 0,
243 v = b[j] ?? 0;
244 const gain =
245 distance(before, c) +
246 distance(c, after) -
247 distance(before, after) -
248 distance(u, c) -
249 distance(c, v) +
250 distance(u, v);
251 if (gain > 0) {
252 output.push(
253 `relocate ${c} from route ${r + 1} to route ${t + 1} after ${location(u)} gain ${gain}`,
254 );
255 a.splice(p, 1);
256 b.splice(j, 0, c);
257 value.routes = normalize(routes);
258 return true;
259 }
260 }
261 }
262 } else if (command === "EXCHANGE") {
263 for (let r = 0; r < routes.length; r++)
264 for (let t = r + 1; t < routes.length; t++)
265 for (let p = 0; p < routes[r].length; p++)
266 for (let j = 0; j < routes[t].length; j++) {
267 const a = routes[r],
268 b = routes[t],
269 c = a[p],
270 e = b[j];
271 if (
272 load(a) - getStop(c).demand + getStop(e).demand > capacity ||
273 load(b) - getStop(e).demand + getStop(c).demand > capacity
274 )
275 continue;
276 const nextA = [...a],
277 nextB = [...b];
278 nextA[p] = e;
279 nextB[j] = c;
280 const gain = length(a) + length(b) - length(nextA) - length(nextB);
281 if (gain > 0) {
282 output.push(
283 `exchange ${c} (route ${r + 1}) with ${e} (route ${t + 1}) gain ${gain}`,
284 );
285 routes[r] = nextA;
286 routes[t] = nextB;
287 value.routes = normalize(routes);
288 return true;
289 }
290 }
291 } else {
292 for (let r = 0; r < routes.length; r++)
293 for (let t = r + 1; t < routes.length; t++)
294 for (let i = 0; i <= routes[r].length; i++)
295 for (let j = 0; j <= routes[t].length; j++) {
296 const a = routes[r],
297 b = routes[t],
298 nextA = [...a.slice(0, i), ...b.slice(j)],
299 nextB = [...b.slice(0, j), ...a.slice(i)];
300 if (load(nextA) > capacity || load(nextB) > capacity) continue;
301 const x = a[i - 1] ?? 0,
302 y = b[j - 1] ?? 0,
303 afterA = a[i] ?? 0,
304 afterB = b[j] ?? 0;
305 const gain =
306 distance(x, afterA) +
307 distance(y, afterB) -
308 distance(x, afterB) -
309 distance(y, afterA);
310 if (gain > 0) {
311 output.push(
312 `cross route ${r + 1} after ${location(x)} with route ${t + 1} after ${location(y)} gain ${gain}`,
313 );
314 routes[r] = nextA;
315 routes[t] = nextB;
316 value.routes = normalize(routes);
317 return true;
318 }
319 }
320 }
321 return false;
322 }
323 while (step()) moves++;
324 output.push(
325 `${command.toLowerCase()}: moves ${moves}${command === "EXCHANGE" ? "" : `, routes ${oldCount} -> ${value.routes.length}`}, length ${oldLength} -> ${planLength(value)}`,
326 );
327}
328function execute(line: string): void {
329 const [command, ...args] = line.split(" "),
330 a = Number(args[0]),
331 b = Number(args[1]);
332 if (command === "DEPOT") {
333 depot = { x: a, y: b };
334 output.push(`depot at (${a},${b})`);
335 return;
336 }
337 if (command === "START") {
338 departure = minute(args[0]);
339 output.push("start " + clock(departure));
340 return;
341 }
342 if (command === "STOP") {
343 if (stops.has(a)) fail(`stop ${a} already exists`);
344 const open = args.length > 5 ? minute(args[5]) : null,
345 close = args.length > 5 ? minute(args[6]) : null;
346 if (open !== null && open > close!) fail("invalid window");
347 stops.set(a, {
348 x: b,
349 y: Number(args[2]),
350 demand: Number(args[3]),
351 service: Number(args[4]),
352 open,
353 close,
354 });
355 tour = null;
356 plan = null;
357 output.push(`stop ${a} added`);
358 return;
359 }
360 if (command === "REMOVE") {
361 if (!stops.delete(a)) fail("unknown stop " + a);
362 if (!stops.size) {
363 tour = null;
364 plan = null;
365 } else {
366 if (tour !== null) tour = tour.filter((id) => id !== a);
367 if (plan !== null)
368 plan.routes = normalize(plan.routes.map((route) => route.filter((id) => id !== a)));
369 }
370 output.push(`stop ${a} removed`);
371 return;
372 }
373 if (command === "LIST") {
374 output.push(
375 depot === null ? "depot none" : `depot (${depot!.x},${depot!.y})`,
376 "start " + clock(departure),
377 "stops " + stops.size,
378 );
379 for (const id of ids()) {
380 const stop = getStop(id);
381 output.push(
382 `${id} at (${stop.x},${stop.y}) demand ${stop.demand} service ${stop.service} window ${stop.open === null ? "none" : clock(stop.open) + "-" + clock(stop.close!)}`,
383 );
384 }
385 return;
386 }
387 if (command === "MATRIX") {
388 if (depot === null) fail("no depot");
389 const locations = [0, ...ids()],
390 rows = [
391 ["", ...locations.map(location)],
392 ...locations.map((id) => [
393 location(id),
394 ...locations.map((other) => String(distance(id, other))),
395 ]),
396 ];
397 const width = Math.max(...rows.flat().map((cell) => cell.length));
398 output.push(...rows.map((row) => row.map((cell) => cell.padStart(width)).join(" ")));
399 return;
400 }
401 if (["ORDER", "NEAREST", "INSERTION", "EXACT", "BOUND", "VRP", "SWEEP"].includes(command))
402 requireStops();
403 if (command === "ORDER") {
404 const order = args.map(Number);
405 for (const id of order) if (!stops.has(id)) fail("unknown stop " + id);
406 const seen = new Set<number>();
407 for (const id of order) {
408 if (seen.has(id)) fail("duplicate stop " + id);
409 seen.add(id);
410 }
411 if (order.length !== stops.size)
412 fail(`order has ${order.length} stops, expected ${stops.size}`);
413 tour = order;
414 } else if (command === "NEAREST") {
415 const remaining = ids(),
416 route: number[] = [];
417 let previous = 0;
418 while (remaining.length) {
419 remaining.sort((a, b) => distance(previous, a) - distance(previous, b) || a - b);
420 previous = remaining.shift()!;
421 route.push(previous);
422 }
423 tour = route;
424 } else if (command === "INSERTION") {
425 const remaining = ids(),
426 route: number[] = [];
427 while (remaining.length) {
428 let best = Infinity,
429 chosen = -1,
430 position = -1;
431 for (const id of remaining)
432 for (let j = 0; j <= route.length; j++) {
433 const before = route[j - 1] ?? 0,
434 after = route[j] ?? 0,
435 increase =
436 distance(before, id) + distance(id, after) - distance(before, after);
437 if (increase < best) {
438 best = increase;
439 chosen = id;
440 position = j;
441 }
442 }
443 route.splice(position, 0, chosen);
444 remaining.splice(remaining.indexOf(chosen), 1);
445 }
446 tour = route;
447 } else if (command === "EXACT") {
448 if (stops.size > 10) fail(`too many stops for exact (${stops.size})`);
449 tour = exact();
450 }
451 if (["ORDER", "NEAREST", "INSERTION", "EXACT"].includes(command)) {
452 output.push(`${command.toLowerCase()}: ${notation(tour!)} length ${length(tour!)}`);
453 return;
454 }
455 if (command === "TOUR") {
456 const route = requireTour();
457 output.push(`tour: ${notation(route)} length ${length(route)}`);
458 return;
459 }
460 if (["TWOOPT", "OROPT", "SWAP"].includes(command)) {
461 const route = requireTour(),
462 old = length(route);
463 let moves = 0;
464 if (command === "TWOOPT") moves = twoOpt(route, true);
465 else
466 while (true) {
467 let changed = false;
468 if (command === "SWAP") {
469 scan: for (let i = 0; i + 1 < route.length; i++)
470 for (let j = i + 1; j < route.length; j++) {
471 const next = [...route];
472 [next[i], next[j]] = [next[j], next[i]];
473 const gain = length(route) - length(next);
474 if (gain > 0) {
475 output.push(`swap ${route[i]} and ${route[j]} gain ${gain}`);
476 route.splice(0, route.length, ...next);
477 moves++;
478 changed = true;
479 break scan;
480 }
481 }
482 } else {
483 scan: for (let size = 1; size <= 3; size++)
484 for (let i = 0; i + size <= route.length; i++) {
485 const segment = route.slice(i, i + size),
486 rest = [...route.slice(0, i), ...route.slice(i + size)];
487 for (let j = 0; j <= rest.length; j++)
488 if (j !== i) {
489 const next = [
490 ...rest.slice(0, j),
491 ...segment,
492 ...rest.slice(j),
493 ],
494 gain = length(route) - length(next);
495 if (gain > 0) {
496 output.push(
497 `move ${segment.join("-")} after ${location(rest[j - 1] ?? 0)} gain ${gain}`,
498 );
499 route.splice(0, route.length, ...next);
500 moves++;
501 changed = true;
502 break scan;
503 }
504 }
505 }
506 }
507 if (!changed) break;
508 }
509 output.push(`${command.toLowerCase()}: moves ${moves}, length ${old} -> ${length(route)}`);
510 return;
511 }
512 if (command === "TIMECHECK") {
513 const route = args.length ? requirePlan().routes[a - 1] : requireTour();
514 if (route === undefined) fail("no route " + a);
515 output.push(`timecheck ${args.length ? "route " + a : "tour"} depart ${clock(departure)}`);
516 simulate(route, true);
517 return;
518 }
519 if (command === "BOUND") {
520 const names = ids(),
521 mst = spanning(names, 0),
522 near = names.map((id) => distance(0, id)).sort((a, b) => a - b);
523 const oneTree =
524 names.length === 1
525 ? 2 * near[0]
526 : spanning(names, names[0]).total + near[0] + near[1],
527 bound = Math.max(mst.total, oneTree);
528 output.push(
529 `mst: ${mst.edges.join(" ")} (length ${mst.total})`,
530 `bound: mst ${mst.total}, one-tree ${oneTree}, best ${bound}`,
531 tour === null
532 ? "gap: no tour"
533 : `gap: tour ${length(tour)}, ${decimal(bound > 0 ? Math.floor((2000 * (length(tour) - bound) + bound) / (2 * bound)) : 0)}%`,
534 );
535 return;
536 }
537 if (command === "VRP" || command === "SWEEP" || command === "SPLIT") {
538 if (command === "SPLIT") requireTour();
539 checkCapacity(a);
540 let routes: number[][],
541 merges = 0;
542 if (command === "VRP") {
543 const limit = args.length > 1 ? b : null;
544 if (limit !== null)
545 for (const id of ids())
546 if (2 * distance(0, id) > limit) fail(`stop ${id} too far for limit ${limit}`);
547 routes = ids().map((id) => [id]);
548 const pairs: Saving[] = [];
549 for (const i of ids())
550 for (const j of ids())
551 if (i < j) {
552 const saving = distance(0, i) + distance(0, j) - distance(i, j);
553 if (saving > 0) pairs.push({ i, j, saving });
554 }
555 pairs.sort((a, b) => b.saving - a.saving || a.i - b.i || a.j - b.j);
556 for (const pair of pairs) {
557 const x = routes.findIndex((route) => route.includes(pair.i)),
558 y = routes.findIndex((route) => route.includes(pair.j));
559 const first = routes[x],
560 second = routes[y];
561 if (
562 x === y ||
563 ![first[0], first.at(-1)].includes(pair.i) ||
564 ![second[0], second.at(-1)].includes(pair.j) ||
565 load(first) + load(second) > a
566 )
567 continue;
568 const joined = [
569 ...(first.at(-1) === pair.i ? first : [...first].reverse()),
570 ...(second[0] === pair.j ? second : [...second].reverse()),
571 ];
572 if (limit !== null && length(joined) > limit) continue;
573 routes = routes.filter((_, i) => i !== x && i !== y);
574 routes.push(joined);
575 merges++;
576 }
577 } else if (command === "SPLIT") {
578 const route = requireTour(),
579 values = Array<number>(route.length + 1).fill(Infinity),
580 previous = Array<number>(route.length + 1).fill(-1);
581 values[0] = 0;
582 for (let j = 1; j <= route.length; j++)
583 for (let i = 0; i < j; i++) {
584 const segment = route.slice(i, j),
585 candidate = values[i] + length(segment);
586 if (load(segment) <= a && candidate < values[j]) {
587 values[j] = candidate;
588 previous[j] = i;
589 }
590 }
591 routes = [];
592 for (let j = route.length; j > 0; j = previous[j])
593 routes.push(route.slice(previous[j], j));
594 } else {
595 const ordered = ids().sort((a, b) => {
596 const x = getStop(a),
597 y = getStop(b),
598 dx = x.x - depot!.x,
599 dy = x.y - depot!.y,
600 ex = y.x - depot!.x,
601 ey = y.y - depot!.y;
602 const zeroA = dx === 0 && dy === 0,
603 zeroB = ex === 0 && ey === 0;
604 if (zeroA || zeroB) return zeroA && zeroB ? a - b : zeroA ? -1 : 1;
605 const halfA = dy > 0 || (dy === 0 && dx > 0) ? 0 : 1,
606 halfB = ey > 0 || (ey === 0 && ex > 0) ? 0 : 1,
607 cross = dx * ey - dy * ex;
608 return (
609 halfA - halfB ||
610 (cross === 0 ? distance(0, a) - distance(0, b) || a - b : cross > 0 ? -1 : 1)
611 );
612 });
613 routes = [[]];
614 for (const id of ordered) {
615 if (load(routes.at(-1)!) + getStop(id).demand > a) routes.push([]);
616 routes.at(-1)!.push(id);
617 }
618 }
619 plan = { capacity: a, routes: normalize(routes) };
620 output.push(
621 `${command.toLowerCase()}: capacity ${a}${command === "VRP" ? `, merges ${merges}` : ""}, routes ${plan.routes.length}, length ${planLength(plan)}`,
622 );
623 routeLines(plan);
624 return;
625 }
626 if (command === "ROUTES") {
627 const value = requirePlan();
628 output.push(
629 `plan: capacity ${value.capacity}, routes ${value.routes.length}, length ${planLength(value)}`,
630 );
631 routeLines(value);
632 return;
633 }
634 if (["RELOCATE", "EXCHANGE", "CROSS"].includes(command)) {
635 improvePlan(command);
636 return;
637 }
638 if (command === "ROUTEOPT") {
639 const value = requirePlan(),
640 old = planLength(value);
641 value.routes.forEach((route, i) => {
642 const before = length(route),
643 moves = twoOpt(route, false);
644 output.push(`route ${i + 1}: moves ${moves}, length ${before} -> ${length(route)}`);
645 });
646 value.routes = normalize(value.routes);
647 output.push(`routeopt: length ${old} -> ${planLength(value)}`);
648 return;
649 }
650 if (command === "REPORT") {
651 const all = [...stops.values()];
652 output.push(
653 `report: stops ${stops.size}, demand ${all.reduce((sum, stop) => sum + stop.demand, 0)}, service ${all.reduce((sum, stop) => sum + stop.service, 0)}, windows ${all.filter((stop) => stop.open !== null).length}`,
654 );
655 if (tour === null) output.push("tour: none");
656 else {
657 const result = simulate(tour);
658 output.push(
659 `tour: length ${length(tour)}, return ${clock(result.time)}, violations ${result.violations}, lateness ${result.lateness}`,
660 );
661 }
662 if (plan === null) output.push("plan: none");
663 else {
664 const results = plan.routes.map((route) => simulate(route)),
665 denominator = plan.routes.length * plan.capacity,
666 utilization = Math.floor(
667 (2000 * plan.routes.reduce((sum, route) => sum + load(route), 0) +
668 denominator) /
669 (2 * denominator),
670 );
671 output.push(
672 `plan: capacity ${plan.capacity}, routes ${plan.routes.length}, length ${planLength(plan)}, utilization ${decimal(utilization)}%, return ${clock(Math.max(...results.map((result) => result.time)))}, violations ${results.reduce((sum, result) => sum + result.violations, 0)}, lateness ${results.reduce((sum, result) => sum + result.lateness, 0)}`,
673 );
674 }
675 }
