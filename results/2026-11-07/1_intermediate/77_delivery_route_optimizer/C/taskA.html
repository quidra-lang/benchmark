76static int distance(int a, int b) {
77 int ax = a ? stops[a].x : depot_x, ay = a ? stops[a].y : depot_y;
78 int bx = b ? stops[b].x : depot_x, by = b ? stops[b].y : depot_y;
79 return abs(ax - bx) + abs(ay - by);
80}

82static int at(const Route *r, int position) {
83 return position < 0 || position >= r->count ? 0 : r->ids[position];

86static int length(const Route *r) {
87 int total = 0, previous = 0;
88 for (int i = 0; i < r->count; i++) {
89 total += distance(previous, r->ids[i]);
90 previous = r->ids[i];
91 }
92 return total + distance(previous, 0);
93}

129static void reverse(Route *r, int first, int last) {
130 while (first < last) {
131 int id = r->ids[first];
132 r->ids[first++] = r->ids[last];
133 r->ids[last--] = id;
134 }
135}

138 Route result = {0};
139 result.count = last - first;

141 return result;

145 Route result = *a;
146 memcpy(result.ids + result.count, b->ids, (size_t)b->count * sizeof *b->ids);
147 result.count += b->count;

156static int smallest(const Route *r) {
157 int result = r->ids[0];
158 for (int i = 1; i < r->count; i++) result = minimum(result, r->ids[i]);
159 return result;

162static int route_compare(const void *a, const void *b) {

166static void normalize(void) {
167 for (int i = plan.count - 1; i >= 0; i--) if (!plan.routes[i].count) remove_route(&plan, i);
168 for (int i = 0; i < plan.count; i++) {
169 Route *r = &plan.routes[i];
170 if (r->ids[0] > r->ids[r->count - 1]) reverse(r, 0, r->count - 1);
171 }
172 qsort(plan.routes, (size_t)plan.count, sizeof *plan.routes, route_compare);
173}

200static Run simulate(const Route *r, bool print) {
201 Run run = {0};
202 int t = departure, previous = 0;
203 for (int i = 0; i < r->count; i++) {
204 int id = r->ids[i];
205 Stop *p = &stops[id];
206 int arrival = t + distance(previous, id), wait = p->open < 0 ? 0 : maximum(0, p->open - arrival);
207 int begin = arrival + wait, late = p->close < 0 ? 0 : maximum(0, arrival - p->close);
208 t = begin + p->service;
209 run.waiting += wait;
210 run.service += p->service;
211 run.lateness += late;
212 run.violations += late > 0;
213 if (print) {
214 printf("%d arrive ", id);
215 print_time(arrival);
216 printf(" wait %d begin ", wait);
217 print_time(begin);
218 fputs(" leave ", stdout);
219 print_time(t);
220 if (late) printf(" late %d\n", late);
221 else puts(" ok");
222 }
223 previous = id;
224 }
225 run.returned = t + distance(previous, 0);
226 if (print) {
227 fputs("return ", stdout);
228 print_time(run.returned);
229 printf(", travel %d, wait %d, service %d, violations %d, lateness %d\n", length(r), run.waiting, run.service, run.violations, run.lateness);
230 }
231 return run;
232}

234static int two_opt(Route *r, bool print) {
235 int moves = 0;
236 for (;;) {
237 bool moved = false;
238 for (int i = 0; i + 1 < r->count; i++) for (int j = i + 1; j < r->count; j++) {
239 int a = r->ids[i], b = r->ids[j];
240 int gain = distance(at(r, i - 1), a) + distance(b, at(r, j + 1)) - distance(at(r, i - 1), b) - distance(a, at(r, j + 1));
241 if (gain > 0) {
242 if (print) printf("reverse %d..%d gain %d\n", a, b, gain);
243 reverse(r, i, j);
244 moves++;
245 moved = true;
246 goto scan_done;
247 }
248 }
249 scan_done: if (!moved) return moves;
250 }
251}

253typedef struct {
254 int count, ids[10], *memo;
255} Exact;
256
257static int exact_best(Exact *e, int mask, int last) {
258 if (mask == (1 << e->count) - 1) return distance(last == e->count ? 0 : e->ids[last], 0);
259 int *entry = &e->memo[mask * (e->count + 1) + last];
260 if (*entry >= 0) return *entry;
261 int value = INT_MAX, previous = last == e->count ? 0 : e->ids[last];
262 for (int i = 0; i < e->count; i++) if (!(mask & (1 << i))) value = minimum(value, distance(previous, e->ids[i]) + exact_best(e, mask | (1 << i), i));
263 return *entry = value;
264}
265
266static Route exact(void) {
267 Exact e = {0};
268 e.count = ids(e.ids);
269 size_t cells = (size_t)(1 << e.count) * (e.count + 1);
270 e.memo = malloc(cells * sizeof *e.memo);
271 if (!e.memo) exit(EXIT_FAILURE);
272 for (size_t i = 0; i < cells; i++) e.memo[i] = -1;
273 Route result = {0};
274 int mask = 0, last = e.count, full = (1 << e.count) - 1;
275 while (mask != full) for (int i = 0; i < e.count; i++) if (!(mask & (1 << i)) && distance(last == e.count ? 0 : e.ids[last], e.ids[i]) + exact_best(&e, mask | (1 << i), i) == exact_best(&e, mask, last)) {
276 result.ids[result.count++] = e.ids[i];
277 mask |= 1 << i;
278 last = i;
279 break;
280 }
281 free(e.memo);
282 return result;
283}

285static int angle_compare(const void *pa, const void *pb) {
286 int a = *(const int *)pa, b = *(const int *)pb;
287 int ax = stops[a].x - depot_x, ay = stops[a].y - depot_y, bx = stops[b].x - depot_x, by = stops[b].y - depot_y;
288 bool az = ax == 0 && ay == 0, bz = bx == 0 && by == 0;
289 if (az || bz) return az == bz ? a - b : az ? -1 : 1;
290 int ah = ay > 0 || (ay == 0 && ax > 0) ? 0 : 1, bh = by > 0 || (by == 0 && bx > 0) ? 0 : 1;
291 if (ah != bh) return ah - bh;
292 int cross = ax * by - ay * bx;
293 if (cross) return cross > 0 ? -1 : 1;
294 int order = distance(0, a) - distance(0, b);
295 return order ? order : a - b;
296}

298static int saving_compare(const void *pa, const void *pb) {
299 const Saving *a = pa, *b = pb;
300 return a->value != b->value ? b->value - a->value : a->a != b->a ? a->a - b->a : a->b - b->b;

303static void bound(void) {
304 int all[MAX_STOPS], n = ids(all), entered[MAX_STOPS + 1] = {0}, entered_count = 1, mst = 0, edges[MAX_STOPS][2];
305 bool included[MAX_STOPS] = {false};
306 for (int step = 0; step < n; step++) {
307 int best = INT_MAX, chosen = -1, parent = 0;
308 for (int i = 0; i < n; i++) if (!included[i]) for (int j = 0; j < entered_count; j++) if (distance(entered[j], all[i]) < best) {
309 best = distance(entered[j], all[i]);
310 chosen = i;
311 parent = entered[j];
312 }
313 included[chosen] = true;
314 entered[entered_count++] = all[chosen];
315 mst += best;
316 edges[step][0] = parent;
317 edges[step][1] = all[chosen];
318 }
319 int one_tree;
320 if (n == 1) one_tree = 2 * distance(0, all[0]);
321 else {
322 memset(included, 0, sizeof included);
323 included[0] = true;
324 int tree = 0;
325 for (int step = 1; step < n; step++) {
326 int best = INT_MAX, chosen = -1;
327 for (int i = 0; i < n; i++) if (!included[i]) for (int j = 0; j < n; j++) if (included[j] && distance(all[j], all[i]) < best) {
328 best = distance(all[j], all[i]);
329 chosen = i;
330 }
331 included[chosen] = true;
332 tree += best;
333 }
334 int first = INT_MAX, second = INT_MAX;
335 for (int i = 0; i < n; i++) {
336 int d = distance(0, all[i]);
337 if (d < first) {
338 second = first;
339 first = d;
340 } else if (d < second) second = d;
341 }
342 one_tree = tree + first + second;
343 }
344 fputs("mst:", stdout);
345 for (int i = 0; i < n; i++) {
346 putchar(' ');
347 location(edges[i][0]);
348 printf("-%d", edges[i][1]);
349 }
350 int lower = maximum(mst, one_tree);
351 printf(" (length %d)\nbound: mst %d, one-tree %d, best %d\n", mst, mst, one_tree, lower);
352 if (!has_tour) puts("gap: no tour");
353 else {
354 int value = length(&tour);
355 long long rounded = lower ? (2000LL * (value - lower) + lower) / (2LL * lower) : 0;
356 printf("gap: tour %d, %lld.%lld%%\n", value, rounded / 10, rounded % 10);
357 }
358}

360static void improve_plan(const char *command) {
361 int initial = plan_length(), old_count = plan.count, moves = 0;
362 for (;;) {
363 bool moved = false;
364 for (int r = 0; r < plan.count; r++) {
365 Route *a = &plan.routes[r];
366 if (!strcmp(command, "RELOCATE")) {
367 for (int p = 0; p < a->count; p++) for (int t = 0; t < plan.count; t++) if (t != r) {
368 Route *b = &plan.routes[t];
369 int c = a->ids[p];
370 if (load(b) + stops[c].demand > capacity) continue;
371 for (int j = 0; j <= b->count; j++) {
372 int u = at(b, j - 1), v = at(b, j);
373 int gain = distance(at(a, p - 1), c) + distance(c, at(a, p + 1)) - distance(at(a, p - 1), at(a, p + 1)) - distance(u, c) - distance(c, v) + distance(u, v);
374 if (gain > 0) {
375 printf("relocate %d from route %d to route %d after ", c, r + 1, t + 1);
376 location(u);
377 printf(" gain %d\n", gain);
378 remove_at(a, p);
379 insert(b, j, c);
380 moved = true;
381 goto scan_done;
382 }
383 }
384 }
385 } else for (int t = r + 1; t < plan.count; t++) {
386 Route *b = &plan.routes[t];
387 if (!strcmp(command, "EXCHANGE")) {
388 for (int p = 0; p < a->count; p++) for (int j = 0; j < b->count; j++) {
389 int c = a->ids[p], e = b->ids[j];
390 if (load(a) - stops[c].demand + stops[e].demand > capacity || load(b) - stops[e].demand + stops[c].demand > capacity) continue;
391 Route x = *a, y = *b;
392 x.ids[p] = e;
393 y.ids[j] = c;
394 int gain = length(a) + length(b) - length(&x) - length(&y);
395 if (gain > 0) {
396 printf("exchange %d (route %d) with %d (route %d) gain %d\n", c, r + 1, e, t + 1, gain);
397 *a = x;
398 *b = y;
399 moved = true;
400 goto scan_done;
401 }
402 }
403 } else {
404 for (int i = 0; i <= a->count; i++) for (int j = 0; j <= b->count; j++) {
405 Route a_prefix = slice(a, 0, i), a_tail = slice(a, i, a->count), b_prefix = slice(b, 0, j), b_tail = slice(b, j, b->count);
406 Route x = concatenate(&a_prefix, &b_tail), y = concatenate(&b_prefix, &a_tail);
407 if (load(&x) > capacity || load(&y) > capacity) continue;
408 int gain = distance(at(a, i - 1), at(a, i)) + distance(at(b, j - 1), at(b, j)) - distance(at(a, i - 1), at(b, j)) - distance(at(b, j - 1), at(a, i));
409 if (gain > 0) {
410 printf("cross route %d after ", r + 1);
411 location(at(a, i - 1));
412 printf(" with route %d after ", t + 1);
413 location(at(b, j - 1));
414 printf(" gain %d\n", gain);
415 *a = x;
416 *b = y;
417 moved = true;
418 goto scan_done;
419 }
420 }
421 }
422 }
423 }
424 scan_done: if (!moved) break;
425 moves++;
426 normalize();
427 }
428 const char *name = !strcmp(command, "RELOCATE") ? "relocate" : !strcmp(command, "EXCHANGE") ? "exchange" : "cross";
429 printf("%s: moves %d", name, moves);
430 if (strcmp(command, "EXCHANGE")) printf(", routes %d -> %d", old_count, plan.count);
431 printf(", length %d -> %d\n", initial, plan_length());
432}

434static void command(char *line) {
435 char *w[MAX_STOPS + 2];
436 int count = 0;
437 for (char *p = strtok(line, " "); p; p = strtok(NULL, " ")) w[count++] = p;
438 const char *c = w[0];
439 if (!strcmp(c, "DEPOT")) {
440 depot = true;
441 depot_x = atoi(w[1]);
442 depot_y = atoi(w[2]);
443 printf("depot at (%d,%d)\n", depot_x, depot_y);
444 } else if (!strcmp(c, "START")) {
445 departure = time_value(w[1]);
446 fputs("start ", stdout);
447 print_time(departure);
448 putchar('\n');
449 } else if (!strcmp(c, "STOP")) {
450 int id = atoi(w[1]);
451 if (stops[id].present) fail("stop %d already exists", id);
452 int open = count == 8 ? time_value(w[6]) : -1, close = count == 8 ? time_value(w[7]) : -1;
453 if (open > close) fail("invalid window");
454 stops[id] = (Stop){true, atoi(w[2]), atoi(w[3]), atoi(w[4]), atoi(w[5]), open, close};
455 stop_count++;
456 has_tour = has_plan = false;
457 printf("stop %d added\n", id);
458 } else if (!strcmp(c, "REMOVE")) {
459 int id = atoi(w[1]);
460 if (!stops[id].present) fail("unknown stop %d", id);
461 stops[id].present = false;
462 stop_count--;
463 if (!stop_count) has_tour = has_plan = false;
464 else {
465 if (has_tour) for (int i = 0; i < tour.count; i++) if (tour.ids[i] == id) {
466 remove_at(&tour, i);
467 break;
468 }
469 if (has_plan) {
470 for (int r = 0; r < plan.count; r++) for (int i = 0; i < plan.routes[r].count; i++) if (plan.routes[r].ids[i] == id) {
471 remove_at(&plan.routes[r], i);
472 break;
473 }
474 normalize();
475 }
476 }
477 printf("stop %d removed\n", id);
478 } else if (!strcmp(c, "LIST")) {
479 if (depot) printf("depot (%d,%d)\n", depot_x, depot_y);
480 else puts("depot none");
481 fputs("start ", stdout);
482 print_time(departure);
483 printf("\nstops %d\n", stop_count);
484 for (int id = 1; id <= MAX_ID; id++) if (stops[id].present) {
485 Stop *p = &stops[id];
486 printf("%d at (%d,%d) demand %d service %d window ", id, p->x, p->y, p->demand, p->service);
487 if (p->open < 0) fputs("none", stdout);
488 else {
489 print_time(p->open);
490 putchar('-');
491 print_time(p->close);
492 }
493 putchar('\n');
494 }
495 } else if (!strcmp(c, "MATRIX")) {
496 if (!depot) fail("no depot");
497 int all[MAX_STOPS + 1] = {0}, n = ids(all + 1) + 1, width = 1;
498 char text[16];
499 for (int i = 0; i < n; i++) for (int j = 0; j < n; j++) {
500 int size = snprintf(text, sizeof text, "%d", distance(all[i], all[j]));
501 width = maximum(width, size);
502 if (all[i]) width = maximum(width, snprintf(text, sizeof text, "%d", all[i]));
503 }
504 printf("%*s", width, "");
505 for (int i = 0; i < n; i++) {
506 if (all[i]) snprintf(text, sizeof text, "%d", all[i]);
507 else strcpy(text, "D");
508 printf(" %*s", width, text);
509 }
510 putchar('\n');
511 for (int i = 0; i < n; i++) {
512 if (all[i]) snprintf(text, sizeof text, "%d", all[i]);
513 else strcpy(text, "D");
514 printf("%*s", width, text);
515 for (int j = 0; j < n; j++) printf(" %*d", width, distance(all[i], all[j]));
516 putchar('\n');
517 }
518 } else if (!strcmp(c, "ORDER")) {
519 network_check();
520 Route next = {0};
521 next.count = count - 1;
522 for (int i = 0; i < next.count; i++) {
523 next.ids[i] = atoi(w[i + 1]);
524 if (!stops[next.ids[i]].present) fail("unknown stop %d", next.ids[i]);
525 }
526 for (int i = 0; i < next.count; i++) for (int j = 0; j < i; j++) if (next.ids[i] == next.ids[j]) fail("duplicate stop %d", next.ids[i]);
527 if (next.count != stop_count) fail("order has %d stops, expected %d", next.count, stop_count);
528 tour = next;
529 has_tour = true;
530 fputs("order: ", stdout);
531 sequence(&tour);
532 printf(" length %d\n", length(&tour));
533 } else if (!strcmp(c, "NEAREST") || !strcmp(c, "INSERTION") || !strcmp(c, "EXACT")) {
534 network_check();
535 if (!strcmp(c, "EXACT")) {
536 if (stop_count > 10) fail("too many stops for exact (%d)", stop_count);
537 tour = exact();
538 } else {
539 tour.count = 0;
540 while (tour.count < stop_count) {
541 int best = INT_MAX, chosen = 0, position = 0;
542 for (int id = 1; id <= MAX_ID; id++) if (stops[id].present && !contains(&tour, id)) {
543 if (!strcmp(c, "NEAREST")) {
544 int value = distance(tour.count ? tour.ids[tour.count - 1] : 0, id);
545 if (value < best) {
546 best = value;
547 chosen = id;
548 position = tour.count;
549 }
550 } else for (int j = 0; j <= tour.count; j++) {
551 int value = distance(at(&tour, j - 1), id) + distance(id, at(&tour, j)) - distance(at(&tour, j - 1), at(&tour, j));
552 if (value < best) {
553 best = value;
554 chosen = id;
555 position = j;
556 }
557 }
558 }
559 insert(&tour, position, chosen);
560 }
561 }
562 has_tour = true;
563 fputs(!strcmp(c, "NEAREST") ? "nearest: " : !strcmp(c, "INSERTION") ? "insertion: " : "exact: ", stdout);
564 sequence(&tour);
565 printf(" length %d\n", length(&tour));
566 } else if (!strcmp(c, "TOUR")) {
567 tour_check();
568 fputs("tour: ", stdout);
569 sequence(&tour);
570 printf(" length %d\n", length(&tour));
571 } else if (!strcmp(c, "TWOOPT") || !strcmp(c, "OROPT") || !strcmp(c, "SWAP")) {
572 tour_check();
573 int initial = length(&tour), moves = 0;
574 if (!strcmp(c, "TWOOPT")) moves = two_opt(&tour, true);
575 else for (;;) {
576 bool moved = false;
577 if (!strcmp(c, "OROPT")) {
578 for (int size = 1; size <= 3; size++) for (int i = 0; i + size <= tour.count; i++) {
579 Route segment = slice(&tour, i, i + size), rest = tour;
580 memmove(rest.ids + i, rest.ids + i + size, (size_t)(rest.count - i - size) * sizeof *rest.ids);
581 rest.count -= size;
582 for (int j = 0; j <= rest.count; j++) if (j != i) {
583 Route prefix = slice(&rest, 0, j), tail = slice(&rest, j, rest.count), first = concatenate(&prefix, &segment), next = concatenate(&first, &tail);
584 int gain = length(&tour) - length(&next);
585 if (gain > 0) {
586 fputs("move ", stdout);
587 for (int p = 0; p < segment.count; p++) {
588 if (p) putchar('-');
589 printf("%d", segment.ids[p]);
590 }
591 fputs(" after ", stdout);
592 location(at(&rest, j - 1));
593 printf(" gain %d\n", gain);
594 tour = next;
595 moves++;
596 moved = true;
597 goto tour_scan_done;
598 }
599 }
600 }
601 } else {
602 for (int i = 0; i + 1 < tour.count; i++) for (int j = i + 1; j < tour.count; j++) {
603 Route next = tour;
604 next.ids[i] = tour.ids[j];
605 next.ids[j] = tour.ids[i];
606 int gain = length(&tour) - length(&next);
607 if (gain > 0) {
608 printf("swap %d and %d gain %d\n", tour.ids[i], tour.ids[j], gain);
609 tour = next;
610 moves++;
611 moved = true;
612 goto tour_scan_done;
613 }
614 }
615 }
616 tour_scan_done: if (!moved) break;
617 }
618 printf("%s: moves %d, length %d -> %d\n", !strcmp(c, "TWOOPT") ? "twoopt" : !strcmp(c, "OROPT") ? "oropt" : "swap", moves, initial, length(&tour));
619 } else if (!strcmp(c, "TIMECHECK")) {
620 Route *route;
621 if (count == 1) {
622 tour_check();
623 route = &tour;
624 fputs("timecheck tour depart ", stdout);
625 } else {
626 plan_check();
627 int r = atoi(w[1]);
628 if (r > plan.count) fail("no route %d", r);
629 route = &plan.routes[r - 1];
630 printf("timecheck route %d depart ", r);
631 }
632 print_time(departure);
633 putchar('\n');
634 simulate(route, true);
635 } else if (!strcmp(c, "BOUND")) {
636 network_check();
637 bound();
638 } else if (!strcmp(c, "VRP") || !strcmp(c, "SPLIT") || !strcmp(c, "SWEEP")) {
639 if (!strcmp(c, "SPLIT")) tour_check();
640 else network_check();
641 int q = atoi(w[1]);
642 capacity_check(q);
643 if (!strcmp(c, "VRP") && count == 3) for (int id = 1; id <= MAX_ID; id++) if (stops[id].present && 2 * distance(0, id) > atoi(w[2])) fail("stop %d too far for limit %s", id, w[2]);
644 Plan next = {0};
645 int merges = 0;
646 if (!strcmp(c, "VRP")) {
647 int all[MAX_STOPS], n = ids(all);
648 Saving savings[MAX_STOPS * (MAX_STOPS - 1) / 2];
649 int saving_count = 0;
650 for (int i = 0; i < n; i++) {
651 next.routes[next.count++] = (Route) {
652 1, {all[i]}
653 };
654 for (int j = i + 1; j < n; j++) {
655 int value = distance(0, all[i]) + distance(0, all[j]) - distance(all[i], all[j]);
656 if (value > 0) savings[saving_count++] = (Saving){all[i], all[j], value};
657 }
658 }
659 qsort(savings, (size_t)saving_count, sizeof *savings, saving_compare);
660 for (int i = 0; i < saving_count; i++) {
661 Saving s = savings[i];
662 int r = 0, t = 0;
663 while (!contains(&next.routes[r], s.a)) r++;
664 while (!contains(&next.routes[t], s.b)) t++;
665 Route a = next.routes[r], b = next.routes[t];
666 if (r == t || (a.ids[0] != s.a && a.ids[a.count - 1] != s.a) || (b.ids[0] != s.b && b.ids[b.count - 1] != s.b) || load(&a) + load(&b) > q) continue;
667 if (a.ids[0] == s.a) reverse(&a, 0, a.count - 1);
668 if (b.ids[b.count - 1] == s.b) reverse(&b, 0, b.count - 1);
669 Route joined = concatenate(&a, &b);
670 if (count == 3 && length(&joined) > atoi(w[2])) continue;
671 remove_route(&next, maximum(r, t));
672 remove_route(&next, minimum(r, t));
673 next.routes[next.count++] = joined;
674 merges++;
675 }
676 } else if (!strcmp(c, "SPLIT")) {
677 int values[MAX_STOPS + 1], previous[MAX_STOPS + 1];
678 values[0] = 0;
679 for (int j = 1; j <= tour.count; j++) {
680 values[j] = INT_MAX;
681 for (int i = 0; i < j; i++) {
682 Route part = slice(&tour, i, j);
683 int candidate = values[i] + length(&part);
684 if (load(&part) <= q && candidate < values[j]) {
685 values[j] = candidate;
686 previous[j] = i;
687 }
688 }
689 }
690 for (int j = tour.count; j > 0; j = previous[j]) next.routes[next.count++] = slice(&tour, previous[j], j);
691 } else {
692 int all[MAX_STOPS], n = ids(all);
693 qsort(all, (size_t)n, sizeof *all, angle_compare);
694 next.count = 1;
695 for (int i = 0; i < n; i++) {
696 Route *r = &next.routes[next.count - 1];
697 if (load(r) + stops[all[i]].demand > q) r = &next.routes[next.count++];
698 r->ids[r->count++] = all[i];
699 }
700 }
701 plan = next;
702 capacity = q;
703 has_plan = true;
704 normalize();
705 printf("%s: capacity %d", !strcmp(c, "VRP") ? "vrp" : !strcmp(c, "SPLIT") ? "split" : "sweep", q);
706 if (!strcmp(c, "VRP")) printf(", merges %d", merges);
707 printf(", routes %d, length %d\n", plan.count, plan_length());
708 route_lines();
709 } else if (!strcmp(c, "ROUTES")) {
710 plan_check();
711 printf("plan: capacity %d, routes %d, length %d\n", capacity, plan.count, plan_length());
712 route_lines();
713 } else if (!strcmp(c, "RELOCATE") || !strcmp(c, "EXCHANGE") || !strcmp(c, "CROSS")) {
714 plan_check();
715 improve_plan(c);
716 } else if (!strcmp(c, "ROUTEOPT")) {
717 plan_check();
718 int initial = plan_length();
719 for (int i = 0; i < plan.count; i++) {
720 int before = length(&plan.routes[i]), moves = two_opt(&plan.routes[i], false);
721 printf("route %d: moves %d, length %d -> %d\n", i + 1, moves, before, length(&plan.routes[i]));
722 }
723 normalize();
724 printf("routeopt: length %d -> %d\n", initial, plan_length());
725 } else if (!strcmp(c, "REPORT")) {
726 int demand = 0, service = 0, windows = 0;
727 for (int id = 1; id <= MAX_ID; id++) if (stops[id].present) {
728 demand += stops[id].demand;
729 service += stops[id].service;
730 windows += stops[id].open >= 0;
731 }
732 printf("report: stops %d, demand %d, service %d, windows %d\n", stop_count, demand, service, windows);
733 if (!has_tour) puts("tour: none");
734 else {
735 Run run = simulate(&tour, false);
736 printf("tour: length %d, return ", length(&tour));
737 print_time(run.returned);
738 printf(", violations %d, lateness %d\n", run.violations, run.lateness);
739 }
740 if (!has_plan) puts("plan: none");
741 else {
742 int returned = 0, violations = 0, lateness = 0;
743 for (int i = 0; i < plan.count; i++) {
744 Run run = simulate(&plan.routes[i], false);
745 returned = maximum(returned, run.returned);
746 violations += run.violations;
747 lateness += run.lateness;
748 }
749 long long denominator = (long long)plan.count * capacity, utilization = (2000LL * demand + denominator) / (2 * denominator);
750 printf("plan: capacity %d, routes %d, length %d, utilization %lld.%lld%%, return ", capacity, plan.count, plan_length(), utilization / 10, utilization % 10);
751 print_time(returned);
752 printf(", violations %d, lateness %d\n", violations, lateness);
753 }
754 }
755}

757static char *read_line(void) {
758 size_t length = 0, capacity = 128;
759 char *line = malloc(capacity);
760 int ch;
761 if (!line) exit(EXIT_FAILURE);
762 while ((ch = getchar()) != EOF && ch != '\n') {
763 if (length + 1 == capacity) {
764 capacity *= 2;
765 char *p = realloc(line, capacity);
766 if (!p) exit(EXIT_FAILURE);
767 line = p;
768 }
769 line[length++] = (char)ch;
770 }
771 line[length] = '\0';
772 return line;
773}
