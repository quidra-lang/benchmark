26 def distance(a, b)
27 x, y = point(a)
28 u, v = point(b)
29 (x - u).abs + (y - v).abs

85 def simulate(route, print_lines = false)
86 time = @departure
87 previous = 0
88 waiting = 0
89 service = 0
90 violations = 0
91 lateness = 0
92 route.each do |id|
93 stop = @stops[id]
94 arrival = time + distance(previous, id)
95 wait = stop.open ? [0, stop.open - arrival].max : 0
96 late = stop.close ? [0, arrival - stop.close].max : 0
97 begin_time = arrival + wait
98 time = begin_time + stop.service
99 waiting += wait
100 service += stop.service
101 violations += 1 if late.positive?
102 lateness += late
103 puts "#{id} arrive #{clock(arrival)} wait #{wait} begin #{clock(begin_time)} leave #{clock(time)} #{late.positive? ? "late #{late}" : 'ok'}" if print_lines
104 previous = id
105 end
106 time += distance(previous, 0)
107 puts "return #{clock(time)}, travel #{length(route)}, wait #{waiting}, service #{service}, violations #{violations}, lateness #{lateness}" if print_lines
108 [time, violations, lateness]
109 end

111 def exact
112 stops = ids
113 full = (1 << stops.length) - 1
114 memo = {}
115 solve = lambda do |mask, previous|
116 return distance(previous, 0) if mask == full
117 memo[[mask, previous]] ||= stops.each_index.reject { |i| mask[i] == 1 }.map do |i|
118 distance(previous, stops[i]) + solve.call(mask | (1 << i), stops[i])
119 end.min
120 end
121 route = []
122 mask = 0
123 previous = 0
124 until mask == full
125 best = solve.call(mask, previous)
126 i = stops.each_index.find { |j| mask[j].zero? && distance(previous, stops[j]) + solve.call(mask | (1 << j), stops[j]) == best }
127 route << stops[i]
128 mask |= 1 << i
129 previous = stops[i]
130 end
131 route
132 end

134 def tour_move(route, method)
135 old = length(route)
136 n = route.length
137 if method == 'OROPT'
138 (1..3).each do |size|
139 next if size > n
140 (0..(n - size)).each do |i|
141 segment = route[i, size]
142 remaining = route.take(i) + route.drop(i + size)
143 (0..remaining.length).each do |j|
144 next if j == i
145 candidate = remaining.take(j) + segment + remaining.drop(j)
146 gain = old - length(candidate)
147 if gain.positive?
148 after = j.zero? ? 'D' : remaining[j - 1]
149 return [candidate, "move #{segment.join('-')} after #{after} gain #{gain}"]
150 end
151 end
152 end
153 end
154 else
155 (0...n).each do |i|
156 ((i + 1)...n).each do |j|
157 candidate = route.dup
158 if method == 'TWOOPT'
159 candidate[i..j] = candidate[i..j].reverse
160 else
161 candidate[i], candidate[j] = candidate[j], candidate[i]
162 end
163 gain = old - length(candidate)
164 next unless gain.positive?
165 message = method == 'TWOOPT' ? "reverse #{route[i]}..#{route[j]} gain #{gain}" : "swap #{route[i]} and #{route[j]} gain #{gain}"
166 return [candidate, message]
167 end
168 end
169 end
170 nil
171 end

173 def improve_tour(route, method, output = true)
174 count = 0
175 loop do
176 move = tour_move(route, method)
177 break unless move
178 route, message = move
179 puts message if output
180 count += 1
181 end
182 [route, count]
183 end

185 def plan_move(method)
186 before = plan_length
187 @routes.each_index do |r|
188 a = @routes[r]
189 if method == 'RELOCATE'
190 a.each_with_index do |c, p|
191 @routes.each_index do |t|
192 next if r == t || load(@routes[t]) + @stops[c].demand > @capacity
193 (0..@routes[t].length).each do |j|
194 candidate = @routes.map(&:dup)
195 candidate[r].delete_at(p)
196 candidate[t].insert(j, c)
197 gain = before - plan_length(candidate)
198 next unless gain.positive?
199 after = j.zero? ? 'D' : @routes[t][j - 1]
200 return [candidate, "relocate #{c} from route #{r + 1} to route #{t + 1} after #{after} gain #{gain}"]
201 end
202 end
203 end
204 else
205 ((r + 1)...@routes.length).each do |t|
206 b = @routes[t]
207 if method == 'EXCHANGE'
208 a.each_with_index do |c, p|
209 b.each_with_index do |e, j|
210 next if load(a) - @stops[c].demand + @stops[e].demand > @capacity || load(b) - @stops[e].demand + @stops[c].demand > @capacity
211 candidate = @routes.map(&:dup)
212 candidate[r][p], candidate[t][j] = e, c
213 gain = before - plan_length(candidate)
214 return [candidate, "exchange #{c} (route #{r + 1}) with #{e} (route #{t + 1}) gain #{gain}"] if gain.positive?
215 end
216 end
217 else
218 (0..a.length).each do |i|
219 (0..b.length).each do |j|
220 left = a.take(i) + b.drop(j)
221 right = b.take(j) + a.drop(i)
222 next if load(left) > @capacity || load(right) > @capacity
223 candidate = @routes.map(&:dup)
224 candidate[r] = left
225 candidate[t] = right
226 gain = before - plan_length(candidate)
227 next unless gain.positive?
228 x = i.zero? ? 'D' : a[i - 1]
229 y = j.zero? ? 'D' : b[j - 1]
230 return [candidate, "cross route #{r + 1} after #{x} with route #{t + 1} after #{y} gain #{gain}"]
231 end
232 end
233 end
234 end
235 end
236 end
237 nil
238 end

240 def bounds
241 tree = [0]
242 remaining = ids
243 edges = []
244 mst = 0
245 until remaining.empty?
246 weight, c, order = remaining.flat_map { |id| tree.each_index.map { |i| [distance(tree[i], id), id, i] } }.min
247 edges << "#{name(tree[order])}-#{c}"
248 mst += weight
249 tree << c
250 remaining.delete(c)
251 end
252 puts "mst: #{edges.join(' ')} (length #{mst})"
253 if ids.length == 1
254 one_tree = 2 * distance(0, ids.first)
255 else
256 tree = [ids.first]
257 remaining = ids.drop(1)
258 cost = 0
259 until remaining.empty?
260 weight, c = remaining.flat_map { |id| tree.map { |p| [distance(p, id), id] } }.min
261 cost += weight
262 tree << c
263 remaining.delete(c)
264 end
265 one_tree = cost + ids.map { |id| distance(0, id) }.sort.take(2).sum
266 end
267 best = [mst, one_tree].max
268 puts "bound: mst #{mst}, one-tree #{one_tree}, best #{best}"
269 if @tour
270 value = length(@tour)
271 u = best.positive? ? (2000 * (value - best) + best) / (2 * best) : 0
272 puts "gap: tour #{value}, #{u / 10}.#{u % 10}%"
273 else
274 puts 'gap: no tour'
275 end
276 end

278 def create_plan(command, args)
279 capacity = args[0].to_i
280 command == 'SPLIT' ? require_tour : require_data
281 check_capacity(capacity)
282 if command == 'VRP' && args[1]
283 limit = args[1].to_i
284 id = ids.find { |c| 2 * distance(0, c) > limit }
285 raise RouteError, "stop #{id} too far for limit #{limit}" if id
286 end
287 merges = 0
288 if command == 'VRP'
289 routes = ids.map { |id| [id] }
290 savings = ids.combination(2).map { |i, j| [distance(0, i) + distance(0, j) - distance(i, j), i, j] }.select { |s, _, _| s.positive? }.sort_by { |s, i, j| [-s, i, j] }
291 savings.each do |_, i, j|
292 r = routes.index { |route| route.include?(i) }
293 t = routes.index { |route| route.include?(j) }
294 a = routes[r]
295 b = routes[t]
296 next if r == t || ![a.first, a.last].include?(i) || ![b.first, b.last].include?(j) || load(a) + load(b) > capacity
297 joined = (a.first == i ? a.reverse : a) + (b.last == j ? b.reverse : b)
298 next if limit && length(joined) > limit
299 routes[r] = joined
300 routes.delete_at(t)
301 merges += 1
302 end
303 elsif command == 'SPLIT'
304 values = [0]
305 parents = []
306 (1..@tour.length).each do |j|
307 options = (0...j).select { |i| load(@tour[i...j]) <= capacity }.map { |i| [values[i] + length(@tour[i...j]), i] }
308 values[j], parents[j] = options.min
309 end
310 routes = []
311 j = @tour.length
312 while j.positive?
313 routes << @tour[parents[j]...j]
314 j = parents[j]
315 end
316 else
317 sweep = ids.sort do |a, b|
318 ax, ay = point(a).zip(@depot).map { |x, d| x - d }
319 bx, by = point(b).zip(@depot).map { |x, d| x - d }
320 az = ax.zero? && ay.zero?
321 bz = bx.zero? && by.zero?
322 if az || bz
323 az == bz ? a <=> b : az ? -1 : 1
324 else
325 ah = ay.positive? || (ay.zero? && ax.positive?) ? 0 : 1
326 bh = by.positive? || (by.zero? && bx.positive?) ? 0 : 1
327 cross = ax * by - ay * bx
328 ah != bh ? ah <=> bh : (cross.zero? ? [distance(0, a), a] <=> [distance(0, b), b] : -cross <=> 0)
329 end
330 end
331 routes = [[]]
332 sweep.each do |id|
333 routes << [] if load(routes.last) + @stops[id].demand > capacity
334 routes.last << id
335 end
336 end
337 @capacity = capacity
338 @routes = routes
339 normalize
340 extra = command == 'VRP' ? " merges #{merges}," : ''
341 puts "#{command.downcase}: capacity #{capacity},#{extra} routes #{@routes.length}, length #{plan_length}"
342 route_lines
343 end

345 def execute(line)
346 command, *args = line.split(' ')
347 case command
348 when 'DEPOT'
349 @depot = args.map(&:to_i)
350 puts "depot at (#{@depot.join(',')})"
351 when 'START'
352 @departure = minutes(args[0])
353 puts "start #{clock(@departure)}"
354 when 'STOP'
355 id, x, y, demand, service = args.take(5).map(&:to_i)
356 raise RouteError, "stop #{id} already exists" if @stops.key?(id)
357 opening = args[5] ? minutes(args[5]) : nil
358 closing = args[6] ? minutes(args[6]) : nil
359 raise RouteError, 'invalid window' if opening && opening > closing
360 @stops[id] = Stop.new(x, y, demand, service, opening, closing)
361 @tour = @routes = nil
362 puts "stop #{id} added"
363 when 'REMOVE'
364 id = args[0].to_i
365 raise RouteError, "unknown stop #{id}" unless @stops.key?(id)
366 @stops.delete(id)
367 if @stops.empty?
368 @tour = @routes = nil
369 else
370 @tour.delete(id) if @tour
371 if @routes
372 @routes.each { |route| route.delete(id) }
373 normalize
374 end
375 end
376 puts "stop #{id} removed"
377 when 'LIST'
378 puts(@depot ? "depot (#{@depot.join(',')})" : 'depot none')
379 puts "start #{clock(@departure)}", "stops #{@stops.length}"
380 ids.each do |id|
381 s = @stops[id]
382 puts "#{id} at (#{s.x},#{s.y}) demand #{s.demand} service #{s.service} window #{s.open ? clock(s.open) + '-' + clock(s.close) : 'none'}"
383 end
384 when 'MATRIX'
385 raise RouteError, 'no depot' unless @depot
386 locations = [0] + ids
387 cells = [[''] + locations.map { |id| name(id) }] + locations.map { |a| [name(a)] + locations.map { |b| distance(a, b).to_s } }
388 width = cells.flatten.map(&:length).max
389 cells.each { |row| puts row.map { |s| s.rjust(width) }.join(' ') }
390 when 'ORDER', 'NEAREST', 'INSERTION', 'EXACT'
391 require_data
392 if command == 'ORDER'
393 route = args.map(&:to_i)
394 route.each { |id| raise RouteError, "unknown stop #{id}" unless @stops.key?(id) }
395 seen = []
396 route.each do |id|
397 raise RouteError, "duplicate stop #{id}" if seen.include?(id)
398 seen << id
399 end
400 raise RouteError, "order has #{route.length} stops, expected #{@stops.length}" unless route.length == @stops.length
401 elsif command == 'EXACT'
402 raise RouteError, "too many stops for exact (#{@stops.length})" if @stops.length > 10
403 route = exact
404 else
405 route = []
406 until route.length == @stops.length
407 remaining = ids - route
408 if command == 'NEAREST'
409 route << remaining.min_by { |id| [distance(route.last || 0, id), id] }
410 else
411 _, id, j = remaining.flat_map do |c|
412 (0..route.length).map { |i| [length(route.take(i) + [c] + route.drop(i)) - length(route), c, i] }
413 end.min
414 route.insert(j, id)
415 end
416 end
417 end
418 @tour = route
419 puts "#{command.downcase}: #{notation(route)} length #{length(route)}"
420 when 'TOUR'
421 require_tour
422 puts "tour: #{notation(@tour)} length #{length(@tour)}"
423 when 'TWOOPT', 'OROPT', 'SWAP'
424 require_tour
425 before = length(@tour)
426 @tour, count = improve_tour(@tour, command)
427 puts "#{command.downcase}: moves #{count}, length #{before} -> #{length(@tour)}"
428 when 'TIMECHECK'
429 if args.empty?
430 require_tour
431 puts "timecheck tour depart #{clock(@departure)}"
432 simulate(@tour, true)
433 else
434 require_plan
435 r = args[0].to_i
436 raise RouteError, "no route #{r}" if r > @routes.length
437 puts "timecheck route #{r} depart #{clock(@departure)}"
438 simulate(@routes[r - 1], true)
439 end
440 when 'BOUND'
441 require_data
442 bounds
443 when 'VRP', 'SPLIT', 'SWEEP'
444 create_plan(command, args)
445 when 'ROUTES'
446 require_plan
447 puts "plan: capacity #{@capacity}, routes #{@routes.length}, length #{plan_length}"
448 route_lines
449 when 'RELOCATE', 'EXCHANGE', 'CROSS'
450 require_plan
451 before = plan_length
452 old_count = @routes.length
453 count = 0
454 loop do
455 move = plan_move(command)
456 break unless move
457 @routes, message = move
458 puts message
459 normalize
460 count += 1
461 end
462 extra = command == 'EXCHANGE' ? '' : " routes #{old_count} -> #{@routes.length},"
463 puts "#{command.downcase}: moves #{count},#{extra} length #{before} -> #{plan_length}"
464 when 'ROUTEOPT'
465 require_plan
466 before = plan_length
467 @routes.each_index do |r|
468 old = length(@routes[r])
469 @routes[r], count = improve_tour(@routes[r], 'TWOOPT', false)
470 puts "route #{r + 1}: moves #{count}, length #{old} -> #{length(@routes[r])}"
471 end
472 normalize
473 puts "routeopt: length #{before} -> #{plan_length}"
474 when 'REPORT'
475 puts "report: stops #{@stops.length}, demand #{@stops.values.sum(&:demand)}, service #{@stops.values.sum(&:service)}, windows #{@stops.values.count { |s| s.open }}"
476 if @tour
477 time, violations, late = simulate(@tour)
478 puts "tour: length #{length(@tour)}, return #{clock(time)}, violations #{violations}, lateness #{late}"
479 else
480 puts 'tour: none'
481 end
482 if @routes
483 runs = @routes.map { |route| simulate(route) }
484 denominator = @routes.length * @capacity
485 u = (2000 * @routes.sum { |r| load(r) } + denominator) / (2 * denominator)
486 puts "plan: capacity #{@capacity}, routes #{@routes.length}, length #{plan_length}, utilization #{u / 10}.#{u % 10}%, return #{clock(runs.map(&:first).max)}, violations #{runs.sum { |r| r[1] }}, lateness #{runs.sum { |r| r[2] }}"
487 else
488 puts 'plan: none'
489 end
490 end
491 rescue RouteError => error
492 puts "error: #{error.message}"
493 end
494end
