6class Problem(string message) : Exception("error: " + message) { }
7
8record Stop(int X, int Y, int Demand, int Service, int Open, int Close);
9
10record Saving(int A, int B, int Value);
11
12record Run(int Returned, int Waiting, int Service, int Violations, int Lateness, List<string> Lines);
13
14class Program
15{
16 readonly SortedDictionary<int, Stop> stops = [];
17 bool depot;
18 int depotX, depotY, departure = 480, capacity;
19 List<int>? tour;
20 List<List<int>>? routes;
21
22 static void Fail(string message) => throw new Problem(message);
23
24 static int Time(string text) => int.Parse(text[..2]) * 60 + int.Parse(text[3..]);
25
26 static string Clock(int value) => $"{value / 60:00}:{value % 60:00}";
27
28 static string Location(int id) => id == 0 ? "D" : id.ToString();
29
30 static string Sequence(List<int> route) => "D-" + string.Join("-", route) + "-D";
31
32 static int At(List<int> route, int position) => position < 0 || position >= route.Count ? 0 : route[position];
33
34 int Distance(int a, int b)
35 {
36 int ax = a == 0 ? depotX : stops[a].X, ay = a == 0 ? depotY : stops[a].Y;
37 int bx = b == 0 ? depotX : stops[b].X, by = b == 0 ? depotY : stops[b].Y;
38 return Math.Abs(ax - bx) + Math.Abs(ay - by);
39 }
40
41 int Length(List<int> route)
42 {
43 int total = 0, previous = 0;
44 foreach (int id in route)
45 {
46 total += Distance(previous, id);
47 previous = id;
48 }
49 return total + Distance(previous, 0);
50 }
51 int Load(List<int> route) => route.Sum(id => stops[id].Demand);
52 int PlanLength() => routes!.Sum(Length);
53
54 void Normalize()
55 {
56 routes!.RemoveAll(r => r.Count == 0);
57 foreach (List<int> r in routes) if (r[0] > r[^1]) r.Reverse();
58 routes.Sort((a, b) => a.Min().CompareTo(b.Min()));
59 }

61 void NetworkCheck()
62 {
63 if (!depot) Fail("no depot");
64 if (stops.Count == 0) Fail("no stops");
65 }
66
67 void TourCheck()
68 {
69 if (tour == null) Fail("no tour");
70 }
71
72 void PlanCheck()
73 {
74 if (routes == null) Fail("no plan");
75 }

77 void CapacityCheck(int q)
78 {
79 foreach ((int id, Stop p) in stops) if (p.Demand > q) Fail($"stop {id} demand {p.Demand} exceeds capacity {q}");
80 }
81 List<string> RouteLines() => routes!.Select((r, i) => $"route {i + 1}: {Sequence(r)} load {Load(r)} length {Length(r)}").ToList();
82
83 Run Simulate(List<int> route)
84 {
85 int t = departure, previous = 0, waiting = 0, service = 0, violations = 0, lateness = 0;
86 List<string> lines = [];
87 foreach (int id in route)
88 {
89 Stop p = stops[id];
90 int arrival = t + Distance(previous, id), wait = p.Open < 0 ? 0 : Math.Max(0, p.Open - arrival);
91 int begin = arrival + wait, late = p.Close < 0 ? 0 : Math.Max(0, arrival - p.Close);
92 t = begin + p.Service;
93 waiting += wait;
94 service += p.Service;
95 lateness += late;
96 if (late > 0) violations++;
97 lines.Add($"{id} arrive {Clock(arrival)} wait {wait} begin {Clock(begin)} leave {Clock(t)} " + (late > 0 ? $"late {late}" : "ok"));
98 previous = id;
99 }
100 t += Distance(previous, 0);
101 lines.Add($"return {Clock(t)}, travel {Length(route)}, wait {waiting}, service {service}, violations {violations}, lateness {lateness}");
102 return new(t, waiting, service, violations, lateness, lines);
103 }

105 int TwoOpt(List<int> route, List<string>? output)
106 {
107 int moves = 0;
108 while (true)
109 {
110 bool moved = false;
111 for (int i = 0; i + 1 < route.Count; i++) for (int j = i + 1; j < route.Count; j++)
112 {
113 int a = route[i], b = route[j];
114 int gain = Distance(At(route, i - 1), a) + Distance(b, At(route, j + 1)) - Distance(At(route, i - 1), b) - Distance(a, At(route, j + 1));
115 if (gain > 0)
116 {
117 output?.Add($"reverse {a}..{b} gain {gain}");
118 route.Reverse(i, j - i + 1);
119 moves++;
120 moved = true;
121 goto ScanDone;
122 }
123 }
124 ScanDone: if (!moved) return moves;
125 }
126 }

128 List<int> Exact()
129 {
130 List<int> ids = [.. stops.Keys];
131 int n = ids.Count, full = (1 << n) - 1;
132 int[,] memo = new int[1 << n, n + 1];
133 for (int mask = 0; mask <= full; mask++) for (int last = 0; last <= n; last++) memo[mask, last] = -1;
134 int Best(int mask, int last)
135 {
136 if (mask == full) return Distance(last == n ? 0 : ids[last], 0);
137 if (memo[mask, last] >= 0) return memo[mask, last];
138 int value = int.MaxValue, previous = last == n ? 0 : ids[last];
139 for (int i = 0; i < n; i++) if ((mask & (1 << i)) == 0) value = Math.Min(value, Distance(previous, ids[i]) + Best(mask | (1 << i), i));
140 return memo[mask, last] = value;
141 }
142 List<int> result = [];
143 int used = 0, current = n;
144 while (used != full) for (int i = 0; i < n; i++) if ((used & (1 << i)) == 0 && Distance(current == n ? 0 : ids[current], ids[i]) + Best(used | (1 << i), i) == Best(used, current))
145 {
146 result.Add(ids[i]);
147 used |= 1 << i;
148 current = i;
149 break;
150 }
151 return result;
152 }

154 int AngleCompare(int a, int b)
155 {
156 int ax = stops[a].X - depotX, ay = stops[a].Y - depotY, bx = stops[b].X - depotX, by = stops[b].Y - depotY;
157 bool az = ax == 0 && ay == 0, bz = bx == 0 && by == 0;
158 if (az || bz) return az == bz ? a.CompareTo(b) : az ? -1 : 1;
159 int ah = ay > 0 || ay == 0 && ax > 0 ? 0 : 1, bh = by > 0 || by == 0 && bx > 0 ? 0 : 1;
160 if (ah != bh) return ah.CompareTo(bh);
161 int cross = ax * by - ay * bx;
162 if (cross != 0) return cross > 0 ? -1 : 1;
163 int order = Distance(0, a).CompareTo(Distance(0, b));
164 return order == 0 ? a.CompareTo(b) : order;
165 }

167 List<string> Bound()
168 {
169 List<int> entered = [0];
170 SortedSet<int> remaining = new(stops.Keys);
171 List<string> edges = [];
172 int mst = 0;
173 while (remaining.Count != 0)
174 {
175 int best = int.MaxValue, selected = 0, parent = 0;
176 foreach (int c in remaining) foreach (int p in entered) if (Distance(p, c) < best)
177 {
178 best = Distance(p, c);
179 selected = c;
180 parent = p;
181 }
182 entered.Add(selected);
183 remaining.Remove(selected);
184 mst += best;
185 edges.Add($"{Location(parent)}-{selected}");
186 }
187 int oneTree;
188 if (stops.Count == 1) oneTree = 2 * Distance(0, stops.Keys.First());
189 else
190 {
191 int tree = 0;
192 HashSet<int> included = [stops.Keys.First()];
193 while (included.Count < stops.Count)
194 {
195 int best = int.MaxValue, chosen = 0;
196 foreach (int c in stops.Keys) if (!included.Contains(c)) foreach (int p in included) if (Distance(p, c) < best)
197 {
198 best = Distance(p, c);
199 chosen = c;
200 }
201 included.Add(chosen);
202 tree += best;
203 }
204 int[] distances = stops.Keys.Select(p => Distance(0, p)).Order().ToArray();
205 oneTree = tree + distances[0] + distances[1];
206 }
207 int lower = Math.Max(mst, oneTree);
208 List<string> output = [$"mst: {string.Join(" ", edges)} (length {mst})", $"bound: mst {mst}, one-tree {oneTree}, best {lower}"];
209 if (tour == null) output.Add("gap: no tour");
210 else
211 {
212 int value = Length(tour);
213 long rounded = lower == 0 ? 0 : (2000L * (value - lower) + lower) / (2L * lower);
214 output.Add($"gap: tour {value}, {rounded / 10}.{rounded % 10}%");
215 }
216 return output;
217 }

219 List<string> ImprovePlan(string command)
220 {
221 List<string> output = [];
222 int initial = PlanLength(), oldCount = routes!.Count, moves = 0;
223 while (true)
224 {
225 bool moved = false;
226 for (int r = 0; r < routes.Count; r++)
227 {
228 List<int> a = routes[r];
229 if (command == "RELOCATE")
230 {
231 for (int p = 0; p < a.Count; p++) for (int t = 0; t < routes.Count; t++) if (t != r)
232 {
233 List<int> b = routes[t];
234 int c = a[p];
235 if (Load(b) + stops[c].Demand > capacity) continue;
236 for (int j = 0; j <= b.Count; j++)
237 {
238 int u = At(b, j - 1), v = At(b, j);
239 int gain = Distance(At(a, p - 1), c) + Distance(c, At(a, p + 1)) - Distance(At(a, p - 1), At(a, p + 1)) - Distance(u, c) - Distance(c, v) + Distance(u, v);
240 if (gain > 0)
241 {
242 output.Add($"relocate {c} from route {r + 1} to route {t + 1} after {Location(u)} gain {gain}");
243 a.RemoveAt(p);
244 b.Insert(j, c);
245 moved = true;
246 goto ScanDone;
247 }
248 }
249 }
250 }
251 else for (int t = r + 1; t < routes.Count; t++)
252 {
253 List<int> b = routes[t];
254 if (command == "EXCHANGE")
255 {
256 for (int p = 0; p < a.Count; p++) for (int j = 0; j < b.Count; j++)
257 {
258 int c = a[p], e = b[j];
259 if (Load(a) - stops[c].Demand + stops[e].Demand > capacity || Load(b) - stops[e].Demand + stops[c].Demand > capacity) continue;
260 List<int> x = [.. a], y = [.. b];
261 x[p] = e;
262 y[j] = c;
263 int gain = Length(a) + Length(b) - Length(x) - Length(y);
264 if (gain > 0)
265 {
266 output.Add($"exchange {c} (route {r + 1}) with {e} (route {t + 1}) gain {gain}");
267 routes[r] = x;
268 routes[t] = y;
269 moved = true;
270 goto ScanDone;
271 }
272 }
273 }
274 else
275 {
276 for (int i = 0; i <= a.Count; i++) for (int j = 0; j <= b.Count; j++)
277 {
278 List<int> x = a.Take(i).Concat(b.Skip(j)).ToList(), y = b.Take(j).Concat(a.Skip(i)).ToList();
279 if (Load(x) > capacity || Load(y) > capacity) continue;
280 int gain = Distance(At(a, i - 1), At(a, i)) + Distance(At(b, j - 1), At(b, j)) - Distance(At(a, i - 1), At(b, j)) - Distance(At(b, j - 1), At(a, i));
281 if (gain > 0)
282 {
283 output.Add($"cross route {r + 1} after {Location(At(a, i - 1))} with route {t + 1} after {Location(At(b, j - 1))} gain {gain}");
284 routes[r] = x;
285 routes[t] = y;
286 moved = true;
287 goto ScanDone;
288 }
289 }
290 }
291 }
292 }
293 ScanDone: if (!moved) break;
294 moves++;
295 Normalize();
296 }
297 output.Add(command.ToLowerInvariant() + $": moves {moves}" + (command == "EXCHANGE" ? "" : $", routes {oldCount} -> {routes.Count}") + $", length {initial} -> {PlanLength()}");
298 return output;
299 }

301 List<string> Command(string line)
302 {
303 string[] w = line.Split(' ');
304 string c = w[0];
305 List<string> output = [];
306 switch (c)
307 {
308 case "DEPOT":
309 depot = true;
310 depotX = int.Parse(w[1]);
311 depotY = int.Parse(w[2]);
312 output.Add($"depot at ({depotX},{depotY})");
313 break;
314 case "START":
315 departure = Time(w[1]);
316 output.Add("start " + Clock(departure));
317 break;
318 case "STOP":
319 {
320 int id = int.Parse(w[1]);
321 if (stops.ContainsKey(id)) Fail($"stop {id} already exists");
322 int open = w.Length == 8 ? Time(w[6]) : -1, close = w.Length == 8 ? Time(w[7]) : -1;
323 if (open > close) Fail("invalid window");
324 stops[id] = new(int.Parse(w[2]), int.Parse(w[3]), int.Parse(w[4]), int.Parse(w[5]), open, close);
325 tour = null;
326 routes = null;
327 output.Add($"stop {id} added");
328 break;
329 }
330 case "REMOVE":
331 {
332 int id = int.Parse(w[1]);
333 if (!stops.ContainsKey(id)) Fail("unknown stop " + id);
334 stops.Remove(id);
335 if (stops.Count == 0)
336 {
337 tour = null;
338 routes = null;
339 }
340 else
341 {
342 tour?.Remove(id);
343 if (routes != null)
344 {
345 foreach (List<int> r in routes) r.Remove(id);
346 Normalize();
347 }
348 }
349 output.Add($"stop {id} removed");
350 break;
351 }
352 case "LIST":
353 output.Add(depot ? $"depot ({depotX},{depotY})" : "depot none");
354 output.Add("start " + Clock(departure));
355 output.Add("stops " + stops.Count);
356 foreach ((int id, Stop p) in stops) output.Add($"{id} at ({p.X},{p.Y}) demand {p.Demand} service {p.Service} window " + (p.Open < 0 ? "none" : Clock(p.Open) + "-" + Clock(p.Close)));
357 break;
358 case "MATRIX":
359 {
360 if (!depot) Fail("no depot");
361 List<int> ids = [0, .. stops.Keys];
362 int width = 1;
363 foreach (int a in ids) foreach (int b in ids) width = Math.Max(width, Math.Max(Location(a).Length, Distance(a, b).ToString().Length));
364 output.Add(new string(' ', width) + string.Concat(ids.Select(id => " " + Location(id).PadLeft(width))));
365 foreach (int a in ids) output.Add(Location(a).PadLeft(width) + string.Concat(ids.Select(b => " " + Distance(a, b).ToString().PadLeft(width))));
366 break;
367 }
368 case "ORDER":
369 {
370 NetworkCheck();
371 List<int> order = w.Skip(1).Select(int.Parse).ToList();
372 foreach (int id in order) if (!stops.ContainsKey(id)) Fail("unknown stop " + id);
373 HashSet<int> seen = [];
374 foreach (int id in order) if (!seen.Add(id)) Fail("duplicate stop " + id);
375 if (order.Count != stops.Count) Fail($"order has {order.Count} stops, expected {stops.Count}");
376 tour = [.. order];
377 output.Add($"order: {Sequence(tour)} length {Length(tour)}");
378 break;
379 }
380 case "NEAREST":
381 case "INSERTION":
382 case "EXACT":
383 {
384 NetworkCheck();
385 if (c == "EXACT")
386 {
387 if (stops.Count > 10) Fail($"too many stops for exact ({stops.Count})");
388 tour = Exact();
389 }
390 else
391 {
392 tour = [];
393 while (tour.Count < stops.Count)
394 {
395 int best = int.MaxValue, chosen = 0, position = 0;
396 foreach (int id in stops.Keys) if (!tour.Contains(id))
397 {
398 if (c == "NEAREST")
399 {
400 int value = Distance(tour.Count == 0 ? 0 : tour[^1], id);
401 if (value < best)
402 {
403 best = value;
404 chosen = id;
405 position = tour.Count;
406 }
407 }
408 else for (int j = 0; j <= tour.Count; j++)
409 {
410 int value = Distance(At(tour, j - 1), id) + Distance(id, At(tour, j)) - Distance(At(tour, j - 1), At(tour, j));
411 if (value < best)
412 {
413 best = value;
414 chosen = id;
415 position = j;
416 }
417 }
418 }
419 tour.Insert(position, chosen);
420 }
421 }
422 output.Add(c.ToLowerInvariant() + $": {Sequence(tour)} length {Length(tour)}");
423 break;
424 }
425 case "TOUR":
426 TourCheck();
427 output.Add($"tour: {Sequence(tour!)} length {Length(tour!)}");
428 break;
429 case "TWOOPT":
430 case "OROPT":
431 case "SWAP":
432 {
433 TourCheck();
434 int initial = Length(tour!), moves = 0;
435 if (c == "TWOOPT") moves = TwoOpt(tour!, output);
436 else while (true)
437 {
438 bool moved = false;
439 if (c == "OROPT")
440 {
441 for (int size = 1; size <= 3; size++) for (int i = 0; i + size <= tour!.Count; i++)
442 {
443 List<int> segment = tour.GetRange(i, size), rest = [.. tour];
444 rest.RemoveRange(i, size);
445 for (int j = 0; j <= rest.Count; j++) if (j != i)
446 {
447 List<int> next = [.. rest];
448 next.InsertRange(j, segment);
449 int gain = Length(tour) - Length(next);
450 if (gain > 0)
451 {
452 output.Add($"move {string.Join("-", segment)} after {Location(At(rest, j - 1))} gain {gain}");
453 tour = next;
454 moves++;
455 moved = true;
456 goto ScanDone;
457 }
458 }
459 }
460 }
461 else
462 {
463 for (int i = 0; i + 1 < tour!.Count; i++) for (int j = i + 1; j < tour.Count; j++)
464 {
465 List<int> next = [.. tour];
466 (next[i], next[j]) = (next[j], next[i]);
467 int gain = Length(tour) - Length(next);
468 if (gain > 0)
469 {
470 output.Add($"swap {tour[i]} and {tour[j]} gain {gain}");
471 tour = next;
472 moves++;
473 moved = true;
474 goto ScanDone;
475 }
476 }
477 }
478 ScanDone: if (!moved) break;
479 }
480 string title = c == "TWOOPT" ? "twoopt" : c == "OROPT" ? "oropt" : "swap";
481 output.Add($"{title}: moves {moves}, length {initial} -> {Length(tour!)}");
482 break;
483 }
484 case "TIMECHECK":
485 {
486 List<int> route;
487 if (w.Length == 1)
488 {
489 TourCheck();
490 route = tour!;
491 output.Add("timecheck tour depart " + Clock(departure));
492 }
493 else
494 {
495 PlanCheck();
496 int r = int.Parse(w[1]);
497 if (r > routes!.Count) Fail("no route " + r);
498 route = routes[r - 1];
499 output.Add($"timecheck route {r} depart {Clock(departure)}");
500 }
501 output.AddRange(Simulate(route).Lines);
502 break;
503 }
504 case "BOUND":
505 NetworkCheck();
506 output = Bound();
507 break;
508 case "VRP":
509 case "SPLIT":
510 case "SWEEP":
511 {
512 if (c == "SPLIT") TourCheck();
513 else NetworkCheck();
514 int q = int.Parse(w[1]);
515 CapacityCheck(q);
516 if (c == "VRP" && w.Length == 3) foreach (int id in stops.Keys) if (2 * Distance(0, id) > int.Parse(w[2])) Fail($"stop {id} too far for limit {w[2]}");
517 List<List<int>> next = [];
518 int merges = 0;
519 if (c == "VRP")
520 {
521 foreach (int id in stops.Keys) next.Add([id]);
522 List<Saving> savings = [];
523 foreach (int a in stops.Keys) foreach (int b in stops.Keys.Where(b => b > a))
524 {
525 int saving = Distance(0, a) + Distance(0, b) - Distance(a, b);
526 if (saving > 0) savings.Add(new(a, b, saving));
527 }
528 foreach (Saving saving in savings.OrderByDescending(s => s.Value).ThenBy(s => s.A).ThenBy(s => s.B))
529 {
530 List<int> a = next.First(r => r.Contains(saving.A)), b = next.First(r => r.Contains(saving.B));
531 if (a == b || a[0] != saving.A && a[^1] != saving.A || b[0] != saving.B && b[^1] != saving.B || Load(a) + Load(b) > q) continue;
532 List<int> x = [.. a], y = [.. b];
533 if (x[0] == saving.A) x.Reverse();
534 if (y[^1] == saving.B) y.Reverse();
535 x.AddRange(y);
536 if (w.Length == 3 && Length(x) > int.Parse(w[2])) continue;
537 next.Remove(a);
538 next.Remove(b);
539 next.Add(x);
540 merges++;
541 }
542 }
543 else if (c == "SPLIT")
544 {
545 int n = tour!.Count;
546 int[] values = Enumerable.Repeat(int.MaxValue, n + 1).ToArray(), previous = new int[n + 1];
547 values[0] = 0;
548 for (int j = 1; j <= n; j++) for (int i = 0; i < j; i++)
549 {
550 List<int> part = tour.GetRange(i, j - i);
551 if (Load(part) <= q && values[i] + Length(part) < values[j])
552 {
553 values[j] = values[i] + Length(part);
554 previous[j] = i;
555 }
556 }
557 for (int j = n; j > 0; j = previous[j]) next.Add(tour.GetRange(previous[j], j - previous[j]));
558 }
559 else
560 {
561 List<int> ids = [.. stops.Keys];
562 ids.Sort(AngleCompare);
563 List<int> route = [];
564 next.Add(route);
565 foreach (int id in ids)
566 {
567 if (Load(route) + stops[id].Demand > q)
568 {
569 route = [];
570 next.Add(route);
571 }
572 route.Add(id);
573 }
574 }
575 routes = next;
576 capacity = q;
577 Normalize();
578 output.Add(c.ToLowerInvariant() + $": capacity {q}" + (c == "VRP" ? $", merges {merges}" : "") + $", routes {routes.Count}, length {PlanLength()}");
579 output.AddRange(RouteLines());
580 break;
581 }
582 case "ROUTES":
583 PlanCheck();
584 output.Add($"plan: capacity {capacity}, routes {routes!.Count}, length {PlanLength()}");
585 output.AddRange(RouteLines());
586 break;
587 case "RELOCATE":
588 case "EXCHANGE":
589 case "CROSS":
590 PlanCheck();
591 output = ImprovePlan(c);
592 break;
593 case "ROUTEOPT":
594 {
595 PlanCheck();
596 int initial = PlanLength();
597 for (int i = 0; i < routes!.Count; i++)
598 {
599 List<int> route = routes[i];
600 int before = Length(route), moves = TwoOpt(route, null);
601 output.Add($"route {i + 1}: moves {moves}, length {before} -> {Length(route)}");
602 }
603 Normalize();
604 output.Add($"routeopt: length {initial} -> {PlanLength()}");
605 break;
606 }
607 case "REPORT":
608 {
609 int demand = stops.Values.Sum(p => p.Demand), service = stops.Values.Sum(p => p.Service);
610 output.Add($"report: stops {stops.Count}, demand {demand}, service {service}, windows {stops.Values.Count(p => p.Open >= 0)}");
611 if (tour == null) output.Add("tour: none");
612 else
613 {
614 Run run = Simulate(tour);
615 output.Add($"tour: length {Length(tour)}, return {Clock(run.Returned)}, violations {run.Violations}, lateness {run.Lateness}");
616 }
617 if (routes == null) output.Add("plan: none");
618 else
619 {
620 int returned = 0, violations = 0, lateness = 0;
621 foreach (List<int> route in routes)
622 {
623 Run run = Simulate(route);
624 returned = Math.Max(returned, run.Returned);
625 violations += run.Violations;
626 lateness += run.Lateness;
627 }
628 long denominator = (long)routes.Count * capacity, utilization = (2000L * demand + denominator) / (2 * denominator);
629 output.Add($"plan: capacity {capacity}, routes {routes.Count}, length {PlanLength()}, utilization {utilization / 10}.{utilization % 10}%, return {Clock(returned)}, violations {violations}, lateness {lateness}");
630 }
631 break;
632 }
633 }
634 return output;
635 }
