1struct CommandError <: Exception
2 message::String
3end
4fail(s) = throw(CommandError(s))
5check(b, s) = b ? nothing : fail(s)
6parsetime(s) = 60parse(Int, s[1:2])+parse(Int, s[4:5])
7timetext(t) = lpad(string(div(t, 60)), 2, '0')*":"*lpad(string(t%60), 2, '0')
8placename(i) = i==0 ? "D" : string(i)
9notation(path) = join(["D"; string.(path); "D"], '-')
10tenths(i) = "$(div(i,10)).$(i%10)"

27Workbench() = Workbench(nothing, 480, Dict{Int,Stop}(), nothing, nothing, 0)
28stopids(w) = sort!(collect(keys(w.stops)))
29coordinates(w, i) = i==0 ? w.depot : (w.stops[i].x, w.stops[i].y)
30function distance(w, a, b)
31 x, y=coordinates(w, a)
32 u, v=coordinates(w, b)
33 abs(x-u)+abs(y-v)
34end
35function pathlength(w, path)
36 route=[0; path; 0]
37 sum(distance(w, route[i], route[i+1]) for i in 1:(length(route)-1))
38end
39pathload(w, path) = sum((w.stops[i].demand for i in path); init = 0)
40planlength(w) = sum(pathlength(w, path) for path in w.routes)
41function normalize!(w)
42 routes=[copy(path) for path in w.routes if !isempty(path)]
43 for path in routes
44 first(path)>last(path) && reverse!(path)
45 end
46 sort!(routes; by = minimum)
47 w.routes=routes
48end

76function simulate(w, path)
77 visits=NTuple{6,Int}[]
78 timestamp=w.start
79 previous=0
80 waiting=0
81 service=0
82 violations=0
83 lateness=0
84 for id in path
85 stop=w.stops[id]
86 arrival=timestamp+distance(w, previous, id)
87 wait=stop.opening<0 ? 0 : max(0, stop.opening-arrival)
88 late=stop.closing<0 ? 0 : max(0, arrival-stop.closing)
89 beginning=arrival+wait
90 timestamp=beginning+stop.service
91 push!(visits, (id, arrival, wait, beginning, timestamp, late))
92 waiting+=wait
93 service+=stop.service
94 violations+=late>0
95 lateness+=late
96 previous=id
97 end
98 (
99 visits = visits,
100 finish = timestamp+distance(w, previous, 0),
101 travel = pathlength(w, path),
102 wait = waiting,
103 service = service,
104 violations = violations,
105 lateness = lateness,
106 )
107end
108function firstreversal(w, path)
109 around=[0; path; 0]
110 for i in 1:(length(path)-1), j in (i+1):length(path)
111 gain=distance(w, around[i], around[i+1])+distance(w, around[j+1], around[j+2])-distance(
112 w,
113 around[i],
114 around[j+1],
115 )-distance(w, around[i+1], around[j+2])
116 gain>0 && return (i, j, gain)
117 end
118 nothing
119end
120function twoopt!(w, path; display = true)
121 moves=0
122 while true
123 move=firstreversal(w, path)
124 move===nothing && break
125 i, j, gain=move
126 display && println("reverse $(path[i])..$(path[j]) gain $gain")
127 path[i:j]=reverse(path[i:j])
128 moves+=1
129 end
130 moves
131end
132function firstsegment(w, path)
133 n=length(path)
134 around=[0; path; 0]
135 for size in 1:3, i in 1:(n-size+1)
136 rest=[path[1:(i-1)]; path[(i+size):end]]
137 remaining=[0; rest; 0]
138 firstid=path[i]
139 lastid=path[i+size-1]
140 removal=distance(w, around[i], firstid)+distance(w, lastid, around[i+size+1])-distance(
141 w,
142 around[i],
143 around[i+size+1],
144 )
145 for j in 0:length(rest)
146 j==i-1 && continue
147 gain=removal-distance(w, remaining[j+1], firstid)-distance(w, lastid, remaining[j+2])+distance(
148 w,
149 remaining[j+1],
150 remaining[j+2],
151 )
152 gain>0 && return (i, size, j, gain)
153 end
154 end
155 nothing
156end
157function firstswap(w, path)
158 before=pathlength(w, path)
159 for i in 1:(length(path)-1), j in (i+1):length(path)
160 changed=copy(path)
161 changed[i], changed[j]=changed[j], changed[i]
162 gain=before-pathlength(w, changed)
163 gain>0 && return (i, j, gain)
164 end
165 nothing
166end
167function exacttour(w)
168 ids=stopids(w)
169 n=length(ids)
170 full=(1<<n)-1
171 costs=zeros(Int, full+1, n)
172 for i in 1:n
173 costs[1, i]=distance(w, ids[i], 0)
174 end
175 for mask in 1:(full-1), i in 1:n
176 (mask & (1<<(i-1)))!=0 && continue
177 costs[mask+1, i]=minimum(
178 distance(w, ids[i], ids[j])+costs[(mask⊻(1<<(j-1)))+1, j] for
179 j in 1:n if (mask & (1<<(j-1)))!=0
180 )
181 end
182 cost=minimum(distance(w, 0, ids[i])+costs[(full⊻(1<<(i-1)))+1, i] for i in 1:n)
183 path=Int[]
184 current=0
185 mask=full
186 while mask!=0
187 choice=first(
188 i for i in 1:n if (mask & (1<<(i-1)))!=0 &&
189 distance(w, current, ids[i])+costs[(mask⊻(1<<(i-1)))+1, i]==cost
190 )
191 mask ⊻= 1<<(choice-1)
192 cost=costs[mask+1, choice]
193 current=ids[choice]
194 push!(path, current)
195 end
196 path
197end
198function spanningtree(w, vertices)
199 tree=vertices[1:1]
200 remaining=vertices[2:end]
201 edges=NTuple{3,Int}[]
202 while !isempty(remaining)
203 weight, child, at=minimum(
204 (distance(w, parent, child), child, at) for child in remaining for
205 (at, parent) in enumerate(tree)
206 )
207 parent=tree[at]
208 push!(edges, (parent, child, weight))
209 push!(tree, child)
210 filter!(!=(child), remaining)
211 end
212 edges
213end

215 function less(a, b)
216 x, y=coordinates(w, a) .- w.depot
217 u, v=coordinates(w, b) .- w.depot
218 az=x==0 && y==0
219 bz=u==0 && v==0
220 az!=bz && return az
221 az && return a<b
222 ah=(y>0 || (y==0 && x>0)) ? 0 : 1
223 bh=(v>0 || (v==0 && u>0)) ? 0 : 1
224 ah!=bh && return ah<bh
225 cross=x*v-y*u
226 cross!=0 && return cross>0
227 d=abs(x)+abs(y)
228 e=abs(u)+abs(v)
229 d!=e ? d<e : a<b
230 end
231 sort!(stopids(w); lt = less)
232end
233function savings!(w, q, limit)
234 ids=stopids(w)
235 if limit!==nothing
236 for id in ids
237 check(2distance(w, 0, id)<=limit, "stop $id too far for limit $limit")
238 end
239 end
240 pairs=Tuple{Int,Int,Int}[]
241 for (i, a) in enumerate(ids), b in ids[(i+1):end]
242 saving=distance(w, 0, a)+distance(w, 0, b)-distance(w, a, b)
243 saving>0 && push!(pairs, (-saving, a, b))
244 end
245 sort!(pairs)
246 routes=[[id] for id in ids]
247 merges=0
248 for (_, a, b) in pairs
249 r=findfirst(path->a in path, routes)
250 t=findfirst(path->b in path, routes)
251 r==t && continue
252 firstpath=routes[r]
253 secondpath=routes[t]
254 a in (first(firstpath), last(firstpath)) && b in (first(secondpath), last(secondpath)) ||
255 continue
256 pathload(w, firstpath)+pathload(w, secondpath)<=q || continue
257 merged=[
258 first(firstpath)==a ? reverse(firstpath) : firstpath;
259 last(secondpath)==b ? reverse(secondpath) : secondpath
260 ]
261 limit!==nothing && pathlength(w, merged)>limit && continue
262 routes=[path for (i, path) in enumerate(routes) if i!=r && i!=t]
263 push!(routes, merged)
264 merges+=1
265 end
266 w.capacity=q
267 w.routes=routes
268 normalize!(w)
269 planreport(w, "vrp: capacity $q, merges $merges")
270end
271function splitplan!(w, q)
272 path=w.tour
273 n=length(path)
274 best=fill(typemax(Int), n+1)
275 best[1]=0
276 previous=zeros(Int, n+1)
277 for j in 1:n, i in 0:(j-1)
278 segment=path[(i+1):j]
279 pathload(w, segment)>q && continue
280 cost=best[i+1]+pathlength(w, segment)
281 if cost<best[j+1]
282 best[j+1]=cost
283 previous[j+1]=i
284 end
285 end
286 routes=Vector{Int}[]
287 j=n
288 while j>0
289 i=previous[j+1]
290 push!(routes, path[(i+1):j])
291 j=i
292 end
293 w.capacity=q
294 w.routes=routes
295 normalize!(w)
296 planreport(w, "split: capacity $q")
297end
298function firstrelocation(w)
299 routes=w.routes
300 for (r, source) in enumerate(routes)
301 around=[0; source; 0]
302 for (p, id) in enumerate(source), (t, target) in enumerate(routes)
303 t==r && continue
304 pathload(w, target)+w.stops[id].demand<=w.capacity || continue
305 near=[0; target; 0]
306 removal=distance(w, around[p], id)+distance(w, id, around[p+2])-distance(
307 w,
308 around[p],
309 around[p+2],
310 )
311 for j in 0:length(target)
312 gain=removal-distance(w, near[j+1], id)-distance(w, id, near[j+2])+distance(
313 w,
314 near[j+1],
315 near[j+2],
316 )
317 gain>0 && return (r, p, t, j, gain)
318 end
319 end
320 end
321 nothing
322end
323function firstexchange(w)
324 for r in 1:(length(w.routes)-1), t in (r+1):length(w.routes)
325 a=w.routes[r]
326 b=w.routes[t]
327 total=pathlength(w, a)+pathlength(w, b)
328 for p in eachindex(a), j in eachindex(b)
329 shift=w.stops[b[j]].demand-w.stops[a[p]].demand
330 pathload(w, a)+shift<=w.capacity && pathload(w, b)-shift<=w.capacity || continue
331 firstpath=copy(a)
332 secondpath=copy(b)
333 firstpath[p]=b[j]
334 secondpath[j]=a[p]
335 gain=total-pathlength(w, firstpath)-pathlength(w, secondpath)
336 gain>0 && return (r, p, t, j, gain)
337 end
338 end
339 nothing
340end
341function firstcross(w)
342 for r in 1:(length(w.routes)-1), t in (r+1):length(w.routes)
343 a=w.routes[r]
344 b=w.routes[t]
345 arounda=[0; a; 0]
346 aroundb=[0; b; 0]
347 for i in 0:length(a), j in 0:length(b)
348 pathload(w, [a[1:i]; b[(j+1):end]])<=w.capacity &&
349 pathload(w, [b[1:j]; a[(i+1):end]])<=w.capacity || continue
350 gain=distance(w, arounda[i+1], arounda[i+2])+distance(w, aroundb[j+1], aroundb[j+2])-distance(
351 w,
352 arounda[i+1],
353 aroundb[j+2],
354 )-distance(w, aroundb[j+1], arounda[i+2])
355 gain>0 && return (r, i, t, j, gain)
356 end
357 end
358 nothing
359end
360function improveplan!(w, kind)
361 before=planlength(w)
362 countbefore=length(w.routes)
363 moves=0
364 while true
365 move=kind=="RELOCATE" ? firstrelocation(w) :
366 kind=="EXCHANGE" ? firstexchange(w) : firstcross(w)
367 move===nothing && break
368 r, p, t, j, gain=move
369 if kind=="RELOCATE"
370 id=w.routes[r][p]
371 after=j==0 ? "D" : string(w.routes[t][j])
372 println("relocate $id from route $r to route $t after $after gain $gain")
373 deleteat!(w.routes[r], p)
374 insert!(w.routes[t], j+1, id)
375 elseif kind=="EXCHANGE"
376 a=w.routes[r][p]
377 b=w.routes[t][j]
378 println("exchange $a (route $r) with $b (route $t) gain $gain")
379 w.routes[r][p]=b
380 w.routes[t][j]=a
381 else
382 a=w.routes[r]
383 b=w.routes[t]
384 println(
385 "cross route $r after $(p==0 ? "D" : string(a[p])) with route $t after $(j==0 ? "D" : string(b[j])) gain $gain",
386 )
387 w.routes[r]=[a[1:p]; b[(j+1):end]]
388 w.routes[t]=[b[1:j]; a[(p+1):end]]
389 end
390 normalize!(w)
391 moves+=1
392 end
393 counts=kind=="EXCHANGE" ? "" : "routes $countbefore -> $(length(w.routes)), "
394 println("$(lowercase(kind)): moves $moves, $(counts)length $before -> $(planlength(w))")
395end
396function execute!(w, line)
397 parts=String.(split(line, ' '))
398 command=parts[1]
399 args=parts[2:end]
400 if command=="DEPOT"
401 x, y=parse.(Int, args)
402 w.depot=(x, y)
403 println("depot at ($x,$y)")
404 elseif command=="START"
405 w.start=parsetime(args[1])
406 println("start $(timetext(w.start))")
407 elseif command=="STOP"
408 id, x, y, demand, service=parse.(Int, args[1:5])
409 check(!haskey(w.stops, id), "stop $id already exists")
410 opening=-1
411 closing=-1
412 if length(args)==7
413 opening=parsetime(args[6])
414 closing=parsetime(args[7])
415 check(opening<=closing, "invalid window")
416 end
417 w.stops[id]=Stop(x, y, demand, service, opening, closing)
418 w.tour=nothing
419 w.routes=nothing
420 println("stop $id added")
421 elseif command=="REMOVE"
422 id=parse(Int, args[1])
423 check(haskey(w.stops, id), "unknown stop $id")
424 delete!(w.stops, id)
425 w.tour!==nothing && filter!(!=(id), w.tour)
426 if w.routes!==nothing
427 for path in w.routes
428 filter!(!=(id), path)
429 end
430 normalize!(w)
431 end
432 if isempty(w.stops)
433 w.tour=nothing
434 w.routes=nothing
435 end
436 println("stop $id removed")
437 elseif command=="LIST"
438 println(w.depot===nothing ? "depot none" : "depot ($(w.depot[1]),$(w.depot[2]))")
439 println("start $(timetext(w.start))\nstops $(length(w.stops))")
440 for id in stopids(w)
441 stop=w.stops[id]
442 window=stop.opening<0 ? "none" : timetext(stop.opening)*"-"*timetext(stop.closing)
443 println(
444 "$id at ($(stop.x),$(stop.y)) demand $(stop.demand) service $(stop.service) window $window",
445 )
446 end
447 elseif command=="MATRIX"
448 requiredepot(w)
449 ids=[0; stopids(w)]
450 table=[[placename(id); [string(distance(w, id, other)) for other in ids]] for id in ids]
451 header=placename.(ids)
452 width=maximum(length, [header; reduce(vcat, table)])
453 println(join([repeat(" ", width); lpad.(header, width)], ' '))
454 for row in table
455 println(join(lpad.(row, width), ' '))
456 end
457 elseif command in ("ORDER", "NEAREST", "INSERTION", "EXACT", "BOUND", "VRP", "SWEEP")
458 requiredepot(w)
459 requirestops(w)
460 if command=="ORDER"
461 path=parse.(Int, args)
462 for id in path
463 check(haskey(w.stops, id), "unknown stop $id")
464 end
465 seen=Set{Int}()
466 for id in path
467 check(!(id in seen), "duplicate stop $id")
468 push!(seen, id)
469 end
470 check(
471 length(path)==length(w.stops),
472 "order has $(length(path)) stops, expected $(length(w.stops))",
473 )
474 installtour!(w, "order", path)
475 elseif command=="NEAREST"
476 remaining=stopids(w)
477 path=Int[]
478 current=0
479 while !isempty(remaining)
480 chosen=remaining[argmin([(distance(w, current, id), id) for id in remaining])]
481 push!(path, chosen)
482 filter!(!=(chosen), remaining)
483 current=chosen
484 end
485 installtour!(w, "nearest", path)
486 elseif command=="INSERTION"
487 remaining=stopids(w)
488 path=Int[]
489 while !isempty(remaining)
490 around=[0; path; 0]
491 _, id, j=minimum(
492 (
493 distance(w, around[j+1], id)+distance(w, id, around[j+2])-distance(
494 w,
495 around[j+1],
496 around[j+2],
497 ),
498 id,
499 j,
500 ) for id in remaining for j in 0:length(path)
501 )
502 insert!(path, j+1, id)
503 filter!(!=(id), remaining)
504 end
505 installtour!(w, "insertion", path)
506 elseif command=="EXACT"
507 check(length(w.stops)<=10, "too many stops for exact ($(length(w.stops)))")
508 installtour!(w, "exact", exacttour(w))
509 elseif command=="BOUND"
510 ids=stopids(w)
511 edges=spanningtree(w, [0; ids])
512 mst=sum(e[3] for e in edges)
513 println(
514 "mst: ",
515 join(["$(placename(a))-$b" for (a, b, _) in edges], ' '),
516 " (length $mst)",
517 )
518 distances=sort([distance(w, 0, id) for id in ids])
519 one=length(ids)==1 ? 2distances[1] :
520 sum(e[3] for e in spanningtree(w, ids))+distances[1]+distances[2]
521 best=max(mst, one)
522 println("bound: mst $mst, one-tree $one, best $best")
523 if w.tour===nothing
524 println("gap: no tour")
525 else
526 total=pathlength(w, w.tour)
527 gap=best==0 ? 0 : div(2000*(total-best)+best, 2best)
528 println("gap: tour $total, $(tenths(gap))%")
529 end
530 else
531 q=parse(Int, args[1])
532 requirecapacity(w, q)
533 if command=="VRP"
534 savings!(w, q, length(args)==2 ? parse(Int, args[2]) : nothing)
535 else
536 routes=[Int[]]
537 used=0
538 for id in sweeporder(w)
539 demand=w.stops[id].demand
540 if used+demand>q
541 push!(routes, Int[])
542 used=0
543 end
544 push!(last(routes), id)
545 used+=demand
546 end
547 w.capacity=q
548 w.routes=routes
549 normalize!(w)
550 planreport(w, "sweep: capacity $q")
551 end
552 end
553 elseif command=="TOUR"
554 requiretour(w)
555 println("tour: $(notation(w.tour)) length $(pathlength(w,w.tour))")
556 elseif command in ("TWOOPT", "OROPT", "SWAP")
557 requiretour(w)
558 before=pathlength(w, w.tour)
559 moves=0
560 if command=="TWOOPT"
561 moves=twoopt!(w, w.tour)
562 else
563 while true
564 move=command=="OROPT" ? firstsegment(w, w.tour) : firstswap(w, w.tour)
565 move===nothing && break
566 if command=="OROPT"
567 i, size, j, gain=move
568 segment=w.tour[i:(i+size-1)]
569 rest=[w.tour[1:(i-1)]; w.tour[(i+size):end]]
570 println(
571 "move $(join(segment,'-')) after $(j==0 ? "D" : string(rest[j])) gain $gain",
572 )
573 w.tour=[rest[1:j]; segment; rest[(j+1):end]]
574 else
575 i, j, gain=move
576 println("swap $(w.tour[i]) and $(w.tour[j]) gain $gain")
577 w.tour[i], w.tour[j]=w.tour[j], w.tour[i]
578 end
579 moves+=1
580 end
581 end
582 println("$(lowercase(command)): moves $moves, length $before -> $(pathlength(w,w.tour))")
583 elseif command=="TIMECHECK"
584 if isempty(args)
585 requiretour(w)
586 path=w.tour
587 println("timecheck tour depart $(timetext(w.start))")
588 else
589 requireplan(w)
590 r=parse(Int, args[1])
591 check(r<=length(w.routes), "no route $r")
592 path=w.routes[r]
593 println("timecheck route $r depart $(timetext(w.start))")
594 end
595 run=simulate(w, path)
596 for (id, arrival, wait, beginning, leave, late) in run.visits
597 println(
598 "$id arrive $(timetext(arrival)) wait $wait begin $(timetext(beginning)) leave $(timetext(leave)) $(late>0 ? "late $late" : "ok")",
599 )
600 end
601 println(
602 "return $(timetext(run.finish)), travel $(run.travel), wait $(run.wait), service $(run.service), violations $(run.violations), lateness $(run.lateness)",
603 )
604 elseif command=="SPLIT"
605 requiretour(w)
606 q=parse(Int, args[1])
607 requirecapacity(w, q)
608 splitplan!(w, q)
609 elseif command=="ROUTES"
610 requireplan(w)
611 planreport(w, "plan: capacity $(w.capacity)")
612 elseif command in ("RELOCATE", "EXCHANGE", "CROSS")
613 requireplan(w)
614 improveplan!(w, command)
615 elseif command=="ROUTEOPT"
616 requireplan(w)
617 before=planlength(w)
618 for (r, path) in enumerate(w.routes)
619 old=pathlength(w, path)
620 moves=twoopt!(w, path; display = false)
621 println("route $r: moves $moves, length $old -> $(pathlength(w,path))")
622 end
623 normalize!(w)
624 println("routeopt: length $before -> $(planlength(w))")
625 elseif command=="REPORT"
626 println(
627 "report: stops $(length(w.stops)), demand $(sum((s.demand for s in values(w.stops));init=0)), service $(sum((s.service for s in values(w.stops));init=0)), windows $(count(s->s.opening>=0,values(w.stops)))",
628 )
629 if w.tour===nothing
630 println("tour: none")
631 else
632 run=simulate(w, w.tour)
633 println(
634 "tour: length $(run.travel), return $(timetext(run.finish)), violations $(run.violations), lateness $(run.lateness)",
635 )
636 end
637 if w.routes===nothing
638 println("plan: none")
639 else
640 runs=[simulate(w, path) for path in w.routes]
641 k=length(w.routes)
642 total=sum(pathload(w, path) for path in w.routes)
643 utilization=div(2000total+k*w.capacity, 2k*w.capacity)
644 println(
645 "plan: capacity $(w.capacity), routes $k, length $(planlength(w)), utilization $(tenths(utilization))%, return $(timetext(maximum(r.finish for r in runs))), violations $(sum(r.violations for r in runs)), lateness $(sum(r.lateness for r in runs))",
646 )
647 end
648 end
649end
