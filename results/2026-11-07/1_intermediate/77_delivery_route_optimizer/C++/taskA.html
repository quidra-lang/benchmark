49 void normalize() {
50 routes.erase(std::remove_if(routes.begin(), routes.end(), [](const Route& route) { return route.empty(); }), routes.end());
51 for (Route& route : routes) {
52 if (route.front() > route.back()) std::reverse(route.begin(), route.end());
53 }
54 std::sort(routes.begin(), routes.end(), [](const Route& a, const Route& b) {
55 return *std::min_element(a.begin(), a.end()) < *std::min_element(b.begin(), b.end());
56 });
57 }

76 int distance(int a, int b) const {
77 std::pair<int, int> first = a == 0 ? *depot : std::make_pair(stops.at(a).x, stops.at(a).y);
78 std::pair<int, int> second = b == 0 ? *depot : std::make_pair(stops.at(b).x, stops.at(b).y);
79 return std::abs(first.first - second.first) + std::abs(first.second - second.second);
80 }

82 static int at(const Route& route, int position) {
83 return position < 0 || position >= static_cast<int>(route.size()) ? 0 : route[position];

86 int length(const Route& route) const {

89 total += distance(previous, stop);

91 }
92 return total + distance(previous, 0);

134 Run simulate(const Route& route, bool output = false) const {
135 Run run{departure, length(route), 0, 0, 0, 0};
136 int previous = 0;
137 for (int number : route) {
138 const Stop& stop = stops.at(number);
139 int arrival = run.returned + distance(previous, number);
140 int wait = stop.window ? std::max(0, stop.window->first - arrival) : 0;
141 int late = stop.window ? std::max(0, arrival - stop.window->second) : 0;
142 int begin = arrival + wait;
143 run.returned = begin + stop.service;
144 run.wait += wait;
145 run.service += stop.service;
146 run.violations += late > 0;
147 run.lateness += late;
148 if (output) {
149 std::cout << number << " arrive " << timeText(arrival) << " wait " << wait << " begin " << timeText(begin)
150 << " leave " << timeText(run.returned) << (late > 0 ? " late " + std::to_string(late) : " ok") << '\n';
151 }
152 previous = number;
153 }
154 run.returned += distance(previous, 0);
155 if (output) std::cout << "return " << timeText(run.returned) << ", travel " << run.travel << ", wait " << run.wait
156 << ", service " << run.service << ", violations " << run.violations << ", lateness " << run.lateness << '\n';
157 return run;
158 }

160 void construct(const std::string& command) {
161 requireStops();
162 if (command == "EXACT" && stops.size() > 10) fail("too many stops for exact (" + std::to_string(stops.size()) + ')');
163 Route route, remaining = numbers();
164 if (command == "EXACT") {
165 int best = std::numeric_limits<int>::max();
166 do {
167 int total = length(remaining);
168 if (total < best) {
169 best = total;
170 route = remaining;
171 }
172 } while (std::next_permutation(remaining.begin(), remaining.end()));
173 } else while (!remaining.empty()) {
174 int best = std::numeric_limits<int>::max();
175 std::size_t selected = 0, position = 0;
176 for (std::size_t i = 0; i < remaining.size(); ++i) {
177 int stop = remaining[i];
178 if (command == "NEAREST") {
179 int cost = distance(route.empty() ? 0 : route.back(), stop);
180 if (cost < best) {
181 best = cost;
182 selected = i;
183 position = route.size();
184 }
185 } else {
186 for (std::size_t j = 0; j <= route.size(); ++j) {
187 int before = at(route, static_cast<int>(j) - 1), after = at(route, static_cast<int>(j));
188 int cost = distance(before, stop) + distance(stop, after) - distance(before, after);
189 if (cost < best) {
190 best = cost;
191 selected = i;
192 position = j;
193 }
194 }
195 }
196 }
197 route.insert(route.begin() + static_cast<std::ptrdiff_t>(position), remaining[selected]);
198 remaining.erase(remaining.begin() + static_cast<std::ptrdiff_t>(selected));
199 }
200 tour = route;
201 std::string label = command == "NEAREST" ? "nearest" : (command == "INSERTION" ? "insertion" : "exact");
202 std::cout << label << ": " << sequence(route) << " length " << length(route) << '\n';
203 }

205 int twoOpt(Route& route, bool output) const {
206 int moves = 0;
207 while (true) {
208 bool improved = false;
209 for (int i = 0; i + 1 < static_cast<int>(route.size()) && !improved; ++i) {
210 for (int j = i + 1; j < static_cast<int>(route.size()); ++j) {
211 int gain = distance(at(route, i - 1), route[i]) + distance(route[j], at(route, j + 1))
212 - distance(at(route, i - 1), route[j]) - distance(route[i], at(route, j + 1));
213 if (gain <= 0) continue;
214 if (output) std::cout << "reverse " << route[i] << ".." << route[j] << " gain " << gain << '\n';
215 std::reverse(route.begin() + i, route.begin() + j + 1);
216 ++moves;
217 improved = true;
218 break;
219 }
220 }
221 if (!improved) return moves;
222 }

225 void improveTour(const std::string& command) {
226 requireTour();
227 int before = length(*tour), moves = 0;
228 if (command == "TWOOPT") moves = twoOpt(*tour, true);
229 else while (true) {
230 bool improved = false;
231 Route& route = *tour;
232 int size = static_cast<int>(route.size());
233 if (command == "SWAP") {
234 int oldLength = length(route);
235 for (int i = 0; i + 1 < size && !improved; ++i) {
236 for (int j = i + 1; j < size; ++j) {
237 Route changed = route;
238 std::swap(changed[i], changed[j]);
239 int gain = oldLength - length(changed);
240 if (gain <= 0) continue;
241 std::cout << "swap " << route[i] << " and " << route[j] << " gain " << gain << '\n';
242 route = std::move(changed);
243 improved = true;
244 break;
245 }
246 }
247 } else {
248 for (int count = 1; count <= 3 && !improved; ++count) {
249 for (int i = 0; i + count <= size && !improved; ++i) {
250 Route segment(route.begin() + i, route.begin() + i + count), remainder = route;
251 remainder.erase(remainder.begin() + i, remainder.begin() + i + count);
252 int removal = distance(at(route, i - 1), segment.front()) + distance(segment.back(), at(route, i + count))
253 - distance(at(route, i - 1), at(route, i + count));
254 for (int j = 0; j <= static_cast<int>(remainder.size()); ++j) {
255 if (j == i) continue;
256 int a = at(remainder, j - 1), b = at(remainder, j);
257 int gain = removal - distance(a, segment.front()) - distance(segment.back(), b) + distance(a, b);
258 if (gain <= 0) continue;
259 std::cout << "move ";
260 for (std::size_t k = 0; k < segment.size(); ++k) std::cout << (k ? "-" : "") << segment[k];
261 std::cout << " after " << name(a) << " gain " << gain << '\n';
262 remainder.insert(remainder.begin() + j, segment.begin(), segment.end());
263 route = std::move(remainder);
264 improved = true;
265 break;
266 }
267 }
268 }
269 }
270 if (!improved) break;
271 ++moves;
272 }
273 std::string label = command == "TWOOPT" ? "twoopt" : (command == "OROPT" ? "oropt" : "swap");
274 std::cout << label << ": moves " << moves << ", length " << before << " -> " << length(*tour) << '\n';
275 }

277 std::pair<int, std::vector<std::string>> mst(Route tree, Route remaining) const {
278 int total = 0;
279 std::vector<std::string> edges;
280 while (!remaining.empty()) {
281 int best = std::numeric_limits<int>::max();
282 std::size_t child = 0, parent = 0;
283 for (std::size_t c = 0; c < remaining.size(); ++c) {
284 for (std::size_t p = 0; p < tree.size(); ++p) {
285 int cost = distance(tree[p], remaining[c]);
286 if (cost < best) {
287 best = cost;
288 child = c;
289 parent = p;
290 }
291 }
292 }
293 total += best;
294 edges.push_back(name(tree[parent]) + '-' + name(remaining[child]));
295 tree.push_back(remaining[child]);
296 remaining.erase(remaining.begin() + static_cast<std::ptrdiff_t>(child));
297 }
298 return {total, edges};
299 }

301 void bound() const {
302 requireStops();
303 Route all = numbers();
304 auto [minimum, edges] = mst({0}, all);
305 std::cout << "mst:";
306 for (const std::string& edge : edges) std::cout << ' ' << edge;
307 std::cout << " (length " << minimum << ")\n";
308 int oneTree = 2 * distance(0, all[0]);
309 if (all.size() >= 2) {
310 Route remaining(all.begin() + 1, all.end());
311 oneTree = mst({all[0]}, remaining).first;
312 std::vector<int> distances;
313 for (int stop : all) distances.push_back(distance(0, stop));
314 std::sort(distances.begin(), distances.end());
315 oneTree += distances[0] + distances[1];
316 }
317 int best = std::max(minimum, oneTree);
318 std::cout << "bound: mst " << minimum << ", one-tree " << oneTree << ", best " << best << '\n';
319 if (!tour) std::cout << "gap: no tour\n";
320 else {
321 int total = length(*tour), tenths = best > 0 ? (2000 * (total - best) + best) / (2 * best) : 0;
322 std::cout << "gap: tour " << total << ", " << tenths / 10 << '.' << tenths % 10 << "%\n";
323 }

326 void constructPlan(const std::vector<std::string>& words) {
327 const std::string& command = words[0];
328 if (command == "SPLIT") requireTour();
329 else requireStops();
330 int capacity = std::stoi(words[1]);
331 checkCapacity(capacity);
332 std::optional<int> limit;
333 if (words.size() == 3) {
334 limit = std::stoi(words[2]);
335 for (const auto& [number, stop] : stops) {
336 if (2 * distance(0, number) > *limit) fail("stop " + std::to_string(number) + " too far for limit " + words[2]);
337 }
338 }
339 Plan result{capacity, {}};
340 int merges = 0;
341 if (command == "VRP") {
342 Route all = numbers();
343 for (int number : all) result.routes.push_back({number});
344 std::vector<std::tuple<int, int, int>> pairs;
345 for (std::size_t i = 0; i < all.size(); ++i) {
346 for (std::size_t j = i + 1; j < all.size(); ++j) {
347 int saving = distance(0, all[i]) + distance(0, all[j]) - distance(all[i], all[j]);
348 if (saving > 0) pairs.emplace_back(-saving, all[i], all[j]);
349 }
350 }
351 std::sort(pairs.begin(), pairs.end());
352 for (auto [saving, first, second] : pairs) {
353 std::size_t a = 0, b = 0;
354 for (std::size_t i = 0; i < result.routes.size(); ++i) {
355 if (std::find(result.routes[i].begin(), result.routes[i].end(), first) != result.routes[i].end()) a = i;
356 if (std::find(result.routes[i].begin(), result.routes[i].end(), second) != result.routes[i].end()) b = i;
357 }
358 if (a == b) continue;
359 Route left = result.routes[a], right = result.routes[b];
360 if ((left.front() != first && left.back() != first) || (right.front() != second && right.back() != second)) continue;
361 if (load(left) + load(right) > capacity) continue;
362 if (left.front() == first) std::reverse(left.begin(), left.end());
363 if (right.back() == second) std::reverse(right.begin(), right.end());
364 left.insert(left.end(), right.begin(), right.end());
365 if (limit && length(left) > *limit) continue;
366 result.routes[a] = std::move(left);
367 result.routes.erase(result.routes.begin() + static_cast<std::ptrdiff_t>(b));
368 ++merges;
369 }
370 } else if (command == "SPLIT") {
371 int size = static_cast<int>(tour->size());
372 std::vector<int> values(size + 1, 0), previous(size + 1, 0);
373 for (int j = 1; j <= size; ++j) {
374 values[j] = std::numeric_limits<int>::max();
375 for (int i = 0; i < j; ++i) {
376 Route segment(tour->begin() + i, tour->begin() + j);
377 if (load(segment) > capacity) continue;
378 int cost = values[i] + length(segment);
379 if (cost < values[j]) {
380 values[j] = cost;
381 previous[j] = i;
382 }
383 }
384 }
385 for (int j = size; j > 0; j = previous[j]) result.routes.emplace_back(tour->begin() + previous[j], tour->begin() + j);
386 } else {
387 Route all = numbers();
388 std::sort(all.begin(), all.end(), [&](int a, int b) {
389 int ax = stops.at(a).x - depot->first, ay = stops.at(a).y - depot->second;
390 int bx = stops.at(b).x - depot->first, by = stops.at(b).y - depot->second;
391 bool zeroA = ax == 0 && ay == 0, zeroB = bx == 0 && by == 0;
392 if (zeroA != zeroB) return zeroA;
393 if (zeroA) return a < b;
394 int halfA = ay > 0 || (ay == 0 && ax > 0) ? 0 : 1;
395 int halfB = by > 0 || (by == 0 && bx > 0) ? 0 : 1;
396 if (halfA != halfB) return halfA < halfB;
397 int cross = ax * by - ay * bx;
398 if (cross != 0) return cross > 0;
399 return std::make_pair(distance(0, a), a) < std::make_pair(distance(0, b), b);
400 });
401 result.routes.push_back({});
402 for (int number : all) {
403 if (load(result.routes.back()) + stops.at(number).demand > capacity) result.routes.push_back({});
404 result.routes.back().push_back(number);
405 }
406 }
407 result.normalize();
408 plan = std::move(result);
409 std::cout << (command == "VRP" ? "vrp" : (command == "SPLIT" ? "split" : "sweep")) << ": capacity " << capacity;
410 if (command == "VRP") std::cout << ", merges " << merges;
411 std::cout << ", routes " << plan->routes.size() << ", length " << planLength() << '\n';
412 routeLines();
413 }

415 void improvePlan(const std::string& command) {
416 requirePlan();
417 int before = planLength(), oldCount = static_cast<int>(plan->routes.size()), moves = 0;
418 while (true) {
419 bool improved = false;
420 auto& routes = plan->routes;
421 int size = static_cast<int>(routes.size());
422 if (command == "RELOCATE") {
423 for (int r = 0; r < size && !improved; ++r) {
424 for (int p = 0; p < static_cast<int>(routes[r].size()) && !improved; ++p) {
425 int stop = routes[r][p];
426 int removal = distance(at(routes[r], p - 1), stop) + distance(stop, at(routes[r], p + 1)) - distance(at(routes[r], p - 1), at(routes[r], p + 1));
427 for (int t = 0; t < size && !improved; ++t) {
428 if (r == t || load(routes[t]) + stops.at(stop).demand > plan->capacity) continue;
429 for (int j = 0; j <= static_cast<int>(routes[t].size()); ++j) {
430 int a = at(routes[t], j - 1), b = at(routes[t], j);
431 int gain = removal - distance(a, stop) - distance(stop, b) + distance(a, b);
432 if (gain <= 0) continue;
433 std::cout << "relocate " << stop << " from route " << r + 1 << " to route " << t + 1 << " after " << name(a) << " gain " << gain << '\n';
434 routes[r].erase(routes[r].begin() + p);
435 routes[t].insert(routes[t].begin() + j, stop);
436 improved = true;
437 break;
438 }
439 }
440 }
441 }
442 } else {
443 for (int r = 0; r < size && !improved; ++r) {
444 for (int t = r + 1; t < size && !improved; ++t) {
445 int a = static_cast<int>(routes[r].size()), b = static_cast<int>(routes[t].size());
446 for (int i = 0; i < a + (command == "CROSS" ? 1 : 0) && !improved; ++i) {
447 for (int j = 0; j < b + (command == "CROSS" ? 1 : 0); ++j) {
448 Route left, right;
449 int gain;
450 if (command == "EXCHANGE") {
451 left = routes[r];
452 right = routes[t];
453 std::swap(left[i], right[j]);
454 if (load(left) > plan->capacity || load(right) > plan->capacity) continue;
455 gain = length(routes[r]) + length(routes[t]) - length(left) - length(right);
456 } else {
457 left.assign(routes[r].begin(), routes[r].begin() + i);
458 left.insert(left.end(), routes[t].begin() + j, routes[t].end());
459 right.assign(routes[t].begin(), routes[t].begin() + j);
460 right.insert(right.end(), routes[r].begin() + i, routes[r].end());
461 if (load(left) > plan->capacity || load(right) > plan->capacity) continue;
462 gain = distance(at(routes[r], i - 1), at(routes[r], i)) + distance(at(routes[t], j - 1), at(routes[t], j))
463 - distance(at(routes[r], i - 1), at(routes[t], j)) - distance(at(routes[t], j - 1), at(routes[r], i));
464 }
465 if (gain <= 0) continue;
466 if (command == "EXCHANGE") std::cout << "exchange " << routes[r][i] << " (route " << r + 1 << ") with " << routes[t][j] << " (route " << t + 1 << ") gain " << gain << '\n';
467 else std::cout << "cross route " << r + 1 << " after " << name(at(routes[r], i - 1)) << " with route " << t + 1 << " after " << name(at(routes[t], j - 1)) << " gain " << gain << '\n';
468 routes[r] = std::move(left);
469 routes[t] = std::move(right);
470 improved = true;
471 break;
472 }
473 }
474 }
475 }
476 }
477 if (!improved) break;
478 ++moves;
479 plan->normalize();
480 }
481 std::string label = command == "RELOCATE" ? "relocate" : (command == "EXCHANGE" ? "exchange" : "cross");
482 std::cout << label << ": moves " << moves;
483 if (command != "EXCHANGE") std::cout << ", routes " << oldCount << " -> " << plan->routes.size();
484 std::cout << ", length " << before << " -> " << planLength() << '\n';
485 }

487public:
488 void execute(const std::string& line) {
489 std::istringstream input(line);
490 std::vector<std::string> words;
491 for (std::string word; input >> word;) words.push_back(word);
492 const std::string& command = words[0];
493 if (command == "DEPOT") {
494 depot = std::make_pair(std::stoi(words[1]), std::stoi(words[2]));
495 std::cout << "depot at (" << depot->first << ',' << depot->second << ")\n";
496 } else if (command == "START") {
497 departure = parseTime(words[1]);
498 std::cout << "start " << timeText(departure) << '\n';
499 } else if (command == "STOP") {
500 int number = std::stoi(words[1]);
501 if (stops.count(number)) fail("stop " + words[1] + " already exists");
502 Stop stop{std::stoi(words[2]), std::stoi(words[3]), std::stoi(words[4]), std::stoi(words[5]), std::nullopt};
503 if (words.size() == 8) {
504 int opening = parseTime(words[6]), closing = parseTime(words[7]);
505 if (opening > closing) fail("invalid window");
506 stop.window = std::make_pair(opening, closing);
507 }
508 stops.emplace(number, stop);
509 tour.reset();
510 plan.reset();
511 std::cout << "stop " << number << " added\n";
512 } else if (command == "REMOVE") {
513 int number = std::stoi(words[1]);
514 if (!stops.erase(number)) fail("unknown stop " + words[1]);
515 if (stops.empty()) {
516 tour.reset();
517 plan.reset();
518 }
519 else {
520 if (tour) tour->erase(std::remove(tour->begin(), tour->end(), number), tour->end());
521 if (plan) {
522 for (Route& route : plan->routes) route.erase(std::remove(route.begin(), route.end(), number), route.end());
523 plan->normalize();
524 }
525 }
526 std::cout << "stop " << number << " removed\n";
527 } else if (command == "LIST") {
528 if (depot) std::cout << "depot (" << depot->first << ',' << depot->second << ")\n";
529 else std::cout << "depot none\n";
530 std::cout << "start " << timeText(departure) << "\nstops " << stops.size() << '\n';
531 for (const auto& [number, stop] : stops) {
532 std::cout << number << " at (" << stop.x << ',' << stop.y << ") demand " << stop.demand << " service " << stop.service
533 << " window " << (stop.window ? timeText(stop.window->first) + '-' + timeText(stop.window->second) : "none") << '\n';
534 }
535 } else if (command == "MATRIX") {
536 if (!depot) fail("no depot");
537 Route locations = numbers();
538 locations.insert(locations.begin(), 0);
539 std::size_t width = 1;
540 for (int a : locations) {
541 width = std::max(width, name(a).size());
542 for (int b : locations) width = std::max(width, std::to_string(distance(a, b)).size());
543 }
544 std::cout << std::string(width, ' ');
545 for (int location : locations) std::cout << ' ' << std::setw(static_cast<int>(width)) << name(location);
546 std::cout << '\n';
547 for (int a : locations) {
548 std::cout << std::setw(static_cast<int>(width)) << name(a);
549 for (int b : locations) std::cout << ' ' << std::setw(static_cast<int>(width)) << distance(a, b);
550 std::cout << '\n';
551 }
552 } else if (command == "ORDER") {
553 requireStops();
554 Route route;
555 for (std::size_t i = 1; i < words.size(); ++i) {
556 int number = std::stoi(words[i]);
557 if (!stops.count(number)) fail("unknown stop " + words[i]);
558 route.push_back(number);
559 }
560 std::set<int> seen;
561 for (int number : route) {
562 if (!seen.insert(number).second) fail("duplicate stop " + std::to_string(number));
563 }
564 if (route.size() != stops.size()) fail("order has " + std::to_string(route.size()) + " stops, expected " + std::to_string(stops.size()));
565 tour = route;
566 std::cout << "order: " << sequence(route) << " length " << length(route) << '\n';
567 } else if (command == "NEAREST" || command == "INSERTION" || command == "EXACT") construct(command);
568 else if (command == "TOUR") {
569 requireTour();
570 std::cout << "tour: " << sequence(*tour) << " length " << length(*tour) << '\n';
571 } else if (command == "TWOOPT" || command == "OROPT" || command == "SWAP") improveTour(command);
572 else if (command == "TIMECHECK") {
573 const Route* route;
574 if (words.size() == 1) {
575 requireTour();
576 route = &*tour;
577 std::cout << "timecheck tour depart " << timeText(departure) << '\n';
578 } else {
579 requirePlan();
580 int number = std::stoi(words[1]);
581 if (number > static_cast<int>(plan->routes.size())) fail("no route " + words[1]);
582 route = &plan->routes[number - 1];
583 std::cout << "timecheck route " << number << " depart " << timeText(departure) << '\n';
584 }
585 simulate(*route, true);
586 } else if (command == "BOUND") bound();
587 else if (command == "VRP" || command == "SPLIT" || command == "SWEEP") constructPlan(words);
588 else if (command == "ROUTES") {
589 requirePlan();
590 std::cout << "plan: capacity " << plan->capacity << ", routes " << plan->routes.size() << ", length " << planLength() << '\n';
591 routeLines();
592 } else if (command == "RELOCATE" || command == "EXCHANGE" || command == "CROSS") improvePlan(command);
593 else if (command == "ROUTEOPT") {
594 requirePlan();
595 int before = planLength();
596 for (std::size_t i = 0; i < plan->routes.size(); ++i) {
597 int oldLength = length(plan->routes[i]);
598 int moves = twoOpt(plan->routes[i], false);
599 std::cout << "route " << i + 1 << ": moves " << moves << ", length " << oldLength << " -> " << length(plan->routes[i]) << '\n';
600 }
601 plan->normalize();
602 std::cout << "routeopt: length " << before << " -> " << planLength() << '\n';
603 } else if (command == "REPORT") {
604 int demand = 0, service = 0, windows = 0;
605 for (const auto& [number, stop] : stops) {
606 demand += stop.demand;
607 service += stop.service;
608 windows += stop.window.has_value();
609 }
610 std::cout << "report: stops " << stops.size() << ", demand " << demand << ", service " << service << ", windows " << windows << '\n';
611 if (!tour) std::cout << "tour: none\n";
612 else {
613 Run run = simulate(*tour);
614 std::cout << "tour: length " << run.travel << ", return " << timeText(run.returned) << ", violations " << run.violations << ", lateness " << run.lateness << '\n';
615 }
616 if (!plan) std::cout << "plan: none\n";
617 else {
618 int returned = departure, violations = 0, lateness = 0;
619 for (const Route& route : plan->routes) {
620 Run run = simulate(route);
621 returned = std::max(returned, run.returned);
622 violations += run.violations;
623 lateness += run.lateness;
624 }
625 int denominator = static_cast<int>(plan->routes.size()) * plan->capacity;
626 int utilization = (2000 * demand + denominator) / (2 * denominator);
627 std::cout << "plan: capacity " << plan->capacity << ", routes " << plan->routes.size() << ", length " << planLength()
628 << ", utilization " << utilization / 10 << '.' << utilization % 10 << "%, return " << timeText(returned)
629 << ", violations " << violations << ", lateness " << lateness << '\n';
630 }
631 }
632 }
