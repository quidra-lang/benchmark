74 func point(_ n: Int) -> (Int,Int) {
75 if n == 0 {
76 return depot!
77 }
78 let s = stops[n]!
79 return (s.x,s.y)

81 func d(_ a: Int,_ b: Int) -> Int {
82 let x = point(a),y = point(b)
83 return abs(x.0-y.0)+abs(x.1-y.1)

85 func length(_ p: [Int]) -> Int {

143 func run(_ p: [Int]) -> Run {
144 var at = 0
145 var t = start
146 var wait = 0
147 var service = 0
148 var violations = 0
149 var late = 0
150 var out = [String]()
151 for n in p {
152 let s = stops[n]!
153 let arrival = t+d(at,n)
154 let w = max(0,(s.window?.0 ?? arrival)-arrival)
155 let begin = arrival+w
156 let e = max(0,arrival-(s.window?.1 ?? arrival))
157 t = begin+s.service
158 wait += w
159 service += s.service
160 if e > 0 {
161 violations += 1
162 }
163 late += e
164 out.append("\(n) arrive \(timeText(arrival)) wait \(w) begin \(timeText(begin)) leave \(timeText(t)) "+(e > 0 ? "late \(e)" : "ok"))
165 at = n
166 }
167 t += d(at,0)
168 out.append("return \(timeText(t)), travel \(length(p)), wait \(wait), service \(service), violations \(violations), lateness \(late)")
169 return Run(lines: out,returned: t,violations: violations,lateness: late)
170 }
171 func firstTourMove(_ p: [Int],_ kind: String) -> TourMove? {
172 let before = length(p),n = p.count
173 if kind == "OROPT" {
174 for size in 1...3 where size <= n {
175 for i in 0...(n-size) {
176 let segment = Array(p[i..<(i+size)])
177 let rest = Array(p.prefix(i))+Array(p.dropFirst(i+size))
178 for j in 0...rest.count where j != i {
179 let next = Array(rest.prefix(j))+segment+Array(rest.dropFirst(j))
180 let gain = before-length(next)
181 if gain > 0 {
182 return TourMove(path: next,text: "move "+segment.map(String.init).joined(separator: "-")+" after \(name(j == 0 ? 0 : rest[j-1])) gain \(gain)")
183 }
184 }
185 }
186 }
187 } else if n >= 2 {
188 for i in 0..<(n-1) {
189 for j in (i+1)..<n {
190 var next = p
191 if kind == "TWOOPT" {
192 next.replaceSubrange(i...j,with: p[i...j].reversed())
193 } else {
194 next.swapAt(i,j)
195 }
196 let gain = before-length(next)
197 if gain > 0 {
198 return TourMove(path: next,text: kind == "TWOOPT" ? "reverse \(p[i])..\(p[j]) gain \(gain)" : "swap \(p[i]) and \(p[j]) gain \(gain)")
199 }
200 }
201 }
202 }
203 return nil
204 }
205 func firstPlanMove(_ p: Plan,_ kind: String) -> PlanMove? {
206 let rs = p.routes,k = rs.count
207 func replaced(_ r: Int,_ a: [Int],_ t: Int,_ b: [Int]) -> [[Int]] {
208 var next = rs
209 next[r] = a
210 next[t] = b
211 return next
212 }
213 if kind == "RELOCATE" {
214 for r in 0..<k {
215 for pos in rs[r].indices {
216 for t in 0..<k where t != r {
217 let c = rs[r][pos]
218 if load(rs[t])+stops[c]!.demand > p.capacity {
219 continue
220 }
221 for j in 0...rs[t].count {
222 var a = rs[r]
223 a.remove(at: pos)
224 var b = rs[t]
225 b.insert(c,at: j)
226 let gain = length(rs[r])+length(rs[t])-length(a)-length(b)
227 if gain > 0 {
228 return PlanMove(routes: replaced(r,a,t,b),text: "relocate \(c) from route \(r+1) to route \(t+1) after \(name(j == 0 ? 0 : rs[t][j-1])) gain \(gain)")
229 }
230 }
231 }
232 }
233 }
234 } else {
235 for r in 0..<k {
236 for t in (r+1)..<k {
237 if kind == "EXCHANGE" {
238 for pos in rs[r].indices {
239 for j in rs[t].indices {
240 let c = rs[r][pos],e = rs[t][j]
241 var a = rs[r],b = rs[t]
242 a[pos] = e
243 b[j] = c
244 if load(a) > p.capacity || load(b) > p.capacity {
245 continue
246 }
247 let gain = length(rs[r])+length(rs[t])-length(a)-length(b)
248 if gain > 0 {
249 return PlanMove(routes: replaced(r,a,t,b),text: "exchange \(c) (route \(r+1)) with \(e) (route \(t+1)) gain \(gain)")
250 }
251 }
252 }
253 } else {
254 for i in 0...rs[r].count {
255 for j in 0...rs[t].count {
256 let a = Array(rs[r].prefix(i))+Array(rs[t].dropFirst(j))
257 let b = Array(rs[t].prefix(j))+Array(rs[r].dropFirst(i))
258 if load(a) > p.capacity || load(b) > p.capacity {
259 continue
260 }
261 let gain = length(rs[r])+length(rs[t])-length(a)-length(b)
262 if gain > 0 {
263 return PlanMove(routes: replaced(r,a,t,b),text: "cross route \(r+1) after \(name(i == 0 ? 0 : rs[r][i-1])) with route \(t+1) after \(name(j == 0 ? 0 : rs[t][j-1])) gain \(gain)")
264 }
265 }
266 }
267 }
268 }
269 }
270 }
271 return nil

273 func mst(_ initial: Int,_ vertices: [Int]) -> (Int,[String]) {
274 var entered = [initial]
275 var left = Set(vertices)
276 left.remove(initial)
277 var sum = 0
278 var edges = [String]()
279 while !left.isEmpty {
280 var best: (Int,Int,Int)?
281 for c in left.sorted() {
282 for (i,p) in entered.enumerated() {
283 let candidate = (d(p,c),c,i)
284 if best == nil || candidate < best! {
285 best = candidate
286 }
287 }
288 }
289 let (weight,c,i) = best!
290 let p = entered[i]
291 entered.append(c)
292 left.remove(c)
293 sum += weight
294 edges.append("\(name(p))-\(c)")
295 }
296 return (sum,edges)
297 }
298 func exactCost(_ ids: [Int],_ mask: Int,_ at: Int,_ memo: inout [State:Int]) -> Int {
299 if mask == (1 << ids.count)-1 {
300 return d(at,0)
301 }
302 let key = State(mask: mask,at: at)
303 if let value = memo[key] {
304 return value
305 }
306 var best = Int.max
307 for (i,n) in ids.enumerated() where mask & (1 << i) == 0 {
308 let cost = d(at,n)+exactCost(ids,mask | (1 << i),n,&memo)
309 best = min(best,cost)
310 }
311 memo[key] = best
312 return best
313 }

328 case "STOP":

332 }

344 try fail("unknown stop \(id)")
345 }
346 if stops.isEmpty {

348 plan = nil
349 } else {
350 tour = tour?.filter {
351 $0 != id

371 out.append("\(id) at (\(s.x),\(s.y)) demand \(s.demand) service \(s.service) window "+(s.window.map {
372 "\(timeText($0.0))-\(timeText($0.1))"

374 }

377 if depot == nil {
378 try fail("no depot")
379 }
380 let locations = [0]+ids
381 let rows = [[""]+locations.map {
382 name($0)
383 }]+locations.map {
384 x in [name(x)]+locations.map {
385 String(d(x,$0))
386 }
387 }
388 let width = rows.flatMap {
389 $0
390 }.map(\.count).max()!

392 $0.map {
393 String(repeating: " ",count: width-$0.count)+$0

395 }
396 case "TOUR":
397 let p = try getTour()

399 case "TIMECHECK":
400 let p: [Int]
401 if a.isEmpty {
402 p = try getTour()

404 } else {

432 var count = 0

437 }

445 for i in p.routes.indices {

447 var route = original

462 $0+$1.demand
463 }
464 out.append("report: stops \(stops.count), demand \(demand), service \(stops.values.reduce(0) { $0+$1.service }), windows \(stops.values.filter { $0.window != nil }.count)")

466 let r = run(tour)

472 let runs = p.routes.map {

476 let u = (2000*demand+capacity)/(2*capacity)
477 out.append("plan: capacity \(p.capacity), routes \(p.routes.count), length \(total(p)), utilization \(u/10).\(u%10)%, return \(timeText(runs.map(\.returned).max()!)), violations \(runs.reduce(0) { $0+$1.violations }), lateness \(runs.reduce(0) { $0+$1.lateness })")
478 } else {
479 out.append("plan: none")
480 }
481 return out
482 case "SPLIT":
483 let p = try getTour()

485 try capacity(q)
486 var values = Array(repeating: Int.max,count: p.count+1)
487 var previous = Array(repeating: 0,count: p.count+1)
488 values[0] = 0
489 for j in 1...p.count {
490 for i in 0..<j {
491 let part = Array(p[i..<j])
492 if load(part) <= q {
493 let candidate = values[i]+length(part)
494 if candidate < values[j] {
495 values[j] = candidate
496 previous[j] = i
497 }
498 }
499 }
500 }
501 var routes = [[Int]]()
502 var j = p.count
503 while j > 0 {
504 let i = previous[j]
505 routes.append(Array(p[i..<j]))
506 j = i
507 }
508 let result = Plan(capacity: q,routes: normalize(routes))
509 plan = result
510 return ["split: capacity \(q), routes \(result.routes.count), length \(total(result))"]+routeLines(result)
511 default:

513 }
514 try construction()
515 switch c {
516 case "ORDER":
517 let values = a.map {
518 Int($0)!
519 }
520 for id in values where stops[id] == nil {
521 try fail("unknown stop \(id)")
522 }
523 var seen = Set<Int>()
524 for id in values {
525 if !seen.insert(id).inserted {
526 try fail("duplicate stop \(id)")
527 }
528 }
529 if values.count != stops.count {
530 try fail("order has \(values.count) stops, expected \(stops.count)")
531 }

533 case "NEAREST":

535 var p = [Int]()
536 var at = 0
537 while !remaining.isEmpty {
538 let next = remaining.min {
539 (d(at,$0),$0) < (d(at,$1),$1)
540 }!
541 p.append(next)
542 remaining.remove(next)
543 at = next
544 }

546 case "INSERTION":

548 while p.count < stops.count {

550 for id in ids where !p.contains(id) {
551 for j in 0...p.count {
552 let left = j == 0 ? 0 : p[j-1],right = j == p.count ? 0 : p[j]
553 let candidate = (d(left,id)+d(id,right)-d(left,right),id,j)

555 best = candidate
556 }
557 }
558 }
559 let (_,id,j) = best!

566 }
567 let ids = self.ids
568 let full = (1 << ids.count)-1
569 var memo = [State:Int]()
570 var mask = 0
571 var at = 0
572 var p = [Int]()
573 while mask != full {
574 let target = exactCost(ids,mask,at,&memo)
575 let i = ids.indices.first {
576 mask & (1 << $0) == 0 && d(at,ids[$0])+exactCost(ids,mask | (1 << $0),ids[$0],&memo) == target
577 }!

580 mask |= 1 << i
581 }

583 case "BOUND":

585 let (m,edges) = mst(0,ids)
586 let tree = ids.count == 1 ? 2*d(0,ids[0]) : mst(ids[0],ids).0+ids.map {
587 d(0,$0)

595 out.append("gap: tour \(l), \(u/10).\(u%10)%")
596 } else {
597 out.append("gap: no tour")
598 }
599 return out
600 case "VRP","SWEEP":
601 let q = n(0)
602 try capacity(q)
603 let limit = a.count == 2 ? n(1) : nil
604 if let limit = limit {
605 for id in ids where 2*d(0,id) > limit {
606 try fail("stop \(id) too far for limit \(limit)")
607 }
608 }
609 var routes = [[Int]]()
610 var merges = 0
611 if c == "VRP" {
612 let ids = self.ids
613 routes = ids.map {
614 [$0]
615 }
616 var savings = [(Int,Int,Int)]()
617 for (i,x) in ids.enumerated() {
618 for y in ids.dropFirst(i+1) {
619 let saving = d(0,x)+d(0,y)-d(x,y)
620 if saving > 0 {
621 savings.append((-saving,x,y))
622 }
623 }
624 }
625 savings.sort {
626 $0 < $1
627 }
628 for (_,x,y) in savings {
629 let ia = routes.firstIndex {
630 $0.contains(x)
631 }!,ib = routes.firstIndex {
632 $0.contains(y)
633 }!
634 if ia == ib {
635 continue
636 }
637 var left = routes[ia],right = routes[ib]
638 if (x != left.first! && x != left.last!) || (y != right.first! && y != right.last!) {
639 continue
640 }
641 if load(left)+load(right) > q {
642 continue
643 }
644 if left.last! != x {
645 left.reverse()
646 }
647 if right.first! != y {
648 right.reverse()
649 }
650 let joined = left+right
651 if let limit = limit,length(joined) > limit {
652 continue
653 }
654 routes[ia] = joined
655 routes.remove(at: ib)
656 merges += 1
657 }
658 } else {
659 let center = depot!
660 let ordered = ids.sorted {
661 x,y in let a = point(x),b = point(y)
662 let ax = a.0-center.0,ay = a.1-center.1,bx = b.0-center.0,by = b.1-center.1
663 let az = ax == 0 && ay == 0,bz = bx == 0 && by == 0
664 if az || bz {
665 return az != bz ? az : x < y
666 }
667 let ah = ay > 0 || ay == 0 && ax > 0 ? 0 : 1,bh = by > 0 || by == 0 && bx > 0 ? 0 : 1
668 if ah != bh {
669 return ah < bh
670 }
671 let cross = ax*by-ay*bx
672 if cross != 0 {
673 return cross > 0
674 }
675 return d(0,x) == d(0,y) ? x < y : d(0,x) < d(0,y)
676 }
677 var route = [Int]()
678 var load = 0
679 for id in ordered {
680 let demand = stops[id]!.demand
681 if load+demand > q {
682 routes.append(route)
683 route = []
684 load = 0
685 }
686 route.append(id)
687 load += demand
688 }
689 routes.append(route)
690 }
691 let result = Plan(capacity: q,routes: normalize(routes))
692 plan = result
693 out.append(c == "VRP" ? "vrp: capacity \(q), merges \(merges), routes \(result.routes.count), length \(total(result))" : "sweep: capacity \(q), routes \(result.routes.count), length \(total(result))")
694 out += routeLines(result)
695 return out
696 default:
697 break
698 }
699 let p = try getTour()
700 return ["\(c.lowercased()): \(notation(p)) length \(length(p))"]
701 }
