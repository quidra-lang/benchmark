71 simulate <- function(route, output = FALSE) {
72 time <- state$departure
73 previous <- 0
74 waiting <- service <- violations <- lateness <- 0
75 for (id in route) {
76 s <- stop_at(id)
77 arrival <- time + distance(previous, id)
78 wait <- if (is.null(s$open)) 0 else max(0, s$open - arrival)
79 late <- if (is.null(s$close)) 0 else max(0, arrival - s$close)
80 begin <- arrival + wait
81 time <- begin + s$service
82 waiting <- waiting + wait
83 service <- service + s$service
84 violations <- violations + as.integer(late > 0)
85 lateness <- lateness + late
86 if (output) {
87 emit(id, " arrive ", clock(arrival), " wait ", wait, " begin ", clock(begin), " leave ", clock(time), " ", if (late > 0) paste0("late ", late) else "ok")
88 }
89 previous <- id
90 }
91 time <- time + distance(previous, 0)
92 if (output) {
93 emit("return ", clock(time), ", travel ", route_length(route), ", wait ", waiting, ", service ", service, ", violations ", violations, ", lateness ", lateness)
94 }
95 c(time, violations, lateness)
96 }
97 exact <- function() {
98 stops <- ids()
99 full <- bitwShiftL(1L, length(stops)) - 1L
100 memo <- new.env(hash = TRUE, parent = emptyenv())
101 solve <- function(mask, previous) {
102 if (mask == full) return(distance(previous, 0))
103 key <- paste(mask, previous, sep = ",")
104 cached <- get0(key, envir = memo, inherits = FALSE)
105 if (!is.null(cached)) return(cached)
106 best <- Inf
107 for (i in seq_along(stops)) {
108 bit <- bitwShiftL(1L, i - 1L)
109 if (bitwAnd(mask, bit) == 0L) {
110 best <- min(best, distance(previous, stops[i]) + solve(bitwOr(mask, bit), stops[i]))
111 }
112 }
113 assign(key, best, envir = memo)
114 best
115 }
116 route <- integer()
117 mask <- previous <- 0L
118 while (mask != full) {
119 best <- solve(mask, previous)
120 for (i in seq_along(stops)) {
121 bit <- bitwShiftL(1L, i - 1L)
122 if (bitwAnd(mask, bit) == 0L && distance(previous, stops[i]) + solve(bitwOr(mask, bit), stops[i]) == best) {
123 route <- c(route, stops[i])
124 mask <- bitwOr(mask, bit)
125 previous <- stops[i]
126 break
127 }
128 }
129 }
130 route
131 }
132 tour_move <- function(route, method) {
133 old <- route_length(route)
134 n <- length(route)
135 if (method == "OROPT") {
136 for (size in 1:3) {
137 if (size > n) next
138 for (i in 0:(n - size)) {
139 part <- segment(route, i, i + size)
140 remaining <- c(take(route, i), drop(route, i + size))
141 for (j in 0:length(remaining)) {
142 if (j == i) next
143 candidate <- append(remaining, part, after = j)
144 gain <- old - route_length(candidate)
145 if (gain > 0) {
146 after <- if (j == 0) "D" else remaining[j]
147 return(list(route = candidate, message = paste0("move ", paste(part, collapse = "-"), " after ", after, " gain ", gain)))
148 }
149 }
150 }
151 }
152 } else if (n >= 2) {
153 for (i in seq_len(n - 1L)) {
154 for (j in seq.int(i + 1L, n)) {
155 candidate <- route
156 if (method == "TWOOPT") {
157 candidate[i:j] <- rev(candidate[i:j])
158 } else {
159 candidate[c(i, j)] <- candidate[c(j, i)]
160 }
161 gain <- old - route_length(candidate)
162 if (gain > 0) {
163 message <- if (method == "TWOOPT") paste0("reverse ", route[i], "..", route[j], " gain ", gain) else paste0("swap ", route[i], " and ", route[j], " gain ", gain)
164 return(list(route = candidate, message = message))
165 }
166 }
167 }
168 }
169 NULL
170 }
171 improve_tour <- function(route, method, output = TRUE) {
172 count <- 0
173 repeat {
174 move <- tour_move(route, method)
175 if (is.null(move)) break
176 route <- move$route
177 if (output) emit(move$message)
178 count <- count + 1
179 }
180 list(route = route, count = count)
181 }
182 plan_move <- function(method) {
183 before <- plan_length()
184 k <- length(state$routes)
185 for (r in seq_len(k)) {
186 a <- state$routes[[r]]
187 if (method == "RELOCATE") {
188 for (p in seq_along(a)) {
189 c <- a[p]
190 for (t in seq_len(k)) {
191 b <- state$routes[[t]]
192 if (r == t || load(b) + stop_at(c)$demand > state$capacity) next
193 for (j in 0:length(b)) {
194 candidate <- state$routes
195 candidate[[r]] <- a[-p]
196 candidate[[t]] <- append(b, c, after = j)
197 gain <- before - plan_length(candidate)
198 if (gain > 0) {
199 after <- if (j == 0) "D" else b[j]
200 return(list(routes = candidate, message = paste0("relocate ", c, " from route ", r, " to route ", t, " after ", after, " gain ", gain)))
201 }
202 }
203 }
204 }
205 } else if (r < k) {
206 for (t in seq.int(r + 1L, k)) {
207 b <- state$routes[[t]]
208 if (method == "EXCHANGE") {
209 for (p in seq_along(a)) {
210 c <- a[p]
211 for (j in seq_along(b)) {
212 e <- b[j]
213 if (load(a) - stop_at(c)$demand + stop_at(e)$demand > state$capacity || load(b) - stop_at(e)$demand + stop_at(c)$demand > state$capacity) next
214 candidate <- state$routes
215 candidate[[r]][p] <- e
216 candidate[[t]][j] <- c
217 gain <- before - plan_length(candidate)
218 if (gain > 0) return(list(routes = candidate, message = paste0("exchange ", c, " (route ", r, ") with ", e, " (route ", t, ") gain ", gain)))
219 }
220 }
221 } else {
222 for (i in 0:length(a)) {
223 for (j in 0:length(b)) {
224 left <- c(take(a, i), drop(b, j))
225 right <- c(take(b, j), drop(a, i))
226 if (load(left) > state$capacity || load(right) > state$capacity) next
227 candidate <- state$routes
228 candidate[[r]] <- left
229 candidate[[t]] <- right
230 gain <- before - plan_length(candidate)
231 if (gain > 0) {
232 x <- if (i == 0) "D" else a[i]
233 y <- if (j == 0) "D" else b[j]
234 return(list(routes = candidate, message = paste0("cross route ", r, " after ", x, " with route ", t, " after ", y, " gain ", gain)))
235 }
236 }
237 }
238 }
239 }
240 }
241 }
242 NULL

244 bounds <- function() {
245 tree <- 0
246 remaining <- ids()
247 edges <- character()
248 mst <- 0
249 while (length(remaining) > 0) {
250 best <- NULL
251 for (id in remaining) {
252 for (i in seq_along(tree)) {
253 choice <- c(distance(tree[i], id), id, i)
254 if (lex_less(choice, best)) best <- choice
255 }
256 }
257 weight <- best[1]
258 c <- best[2]
259 edges <- c(edges, paste0(location_name(tree[best[3]]), "-", c))
260 mst <- mst + weight
261 tree <- c(tree, c)
262 remaining <- remaining[remaining != c]
263 }
264 emit("mst: ", paste(edges, collapse = " "), " (length ", mst, ")")
265 stops <- ids()
266 if (length(stops) == 1) {
267 one_tree <- 2 * distance(0, stops[1])
268 } else {
269 tree <- stops[1]
270 remaining <- stops[-1]
271 cost <- 0
272 while (length(remaining) > 0) {
273 best <- NULL
274 for (id in remaining) {
275 for (parent in tree) {
276 choice <- c(distance(parent, id), id)
277 if (lex_less(choice, best)) best <- choice
278 }
279 }
280 cost <- cost + best[1]
281 tree <- c(tree, best[2])
282 remaining <- remaining[remaining != best[2]]
283 }
284 one_tree <- cost + sum(head(sort(vapply(stops, function(id) distance(0, id), numeric(1))), 2))
285 }
286 best <- max(mst, one_tree)
287 emit("bound: mst ", mst, ", one-tree ", one_tree, ", best ", best)
288 if (is.null(state$tour)) {
289 emit("gap: no tour")
290 } else {
291 value <- route_length(state$tour)
292 u <- if (best > 0) (2000 * (value - best) + best) %/% (2 * best) else 0
293 emit("gap: tour ", value, ", ", u %/% 10, ".", u %% 10, "%")
294 }

296 create_plan <- function(command, args) {
297 capacity <- as.integer(args[1])
298 if (command == "SPLIT") require_tour() else require_data()
299 check_capacity(capacity)
300 limit <- if (command == "VRP" && length(args) > 1) as.integer(args[2]) else NULL
301 if (!is.null(limit)) {
302 for (id in ids()) {
303 if (2 * distance(0, id) > limit) fail(paste0("stop ", id, " too far for limit ", limit))
304 }
305 }
306 merges <- 0
307 if (command == "VRP") {
308 stops <- ids()
309 routes <- lapply(stops, function(id) id)
310 savings <- list()
311 if (length(stops) >= 2) {
312 for (p in seq_len(length(stops) - 1L)) {
313 for (q in seq.int(p + 1L, length(stops))) {
314 i <- stops[p]
315 j <- stops[q]
316 saving <- distance(0, i) + distance(0, j) - distance(i, j)
317 if (saving > 0) savings[[length(savings) + 1L]] <- c(-saving, i, j)
318 }
319 }
320 }
321 if (length(savings) > 0) {
322 keys <- do.call(rbind, savings)
323 savings <- savings[order(keys[, 1], keys[, 2], keys[, 3])]
324 }
325 for (pair in savings) {
326 i <- pair[2]
327 j <- pair[3]
328 r <- which(vapply(routes, function(route) i %in% route, logical(1)))[1]
329 t <- which(vapply(routes, function(route) j %in% route, logical(1)))[1]
330 a <- routes[[r]]
331 b <- routes[[t]]
332 if (r == t || !(i %in% c(a[1], tail(a, 1))) || !(j %in% c(b[1], tail(b, 1))) || load(a) + load(b) > capacity) next
333 joined <- c(if (a[1] == i) rev(a) else a, if (tail(b, 1) == j) rev(b) else b)
334 if (!is.null(limit) && route_length(joined) > limit) next
335 routes[[r]] <- joined
336 routes <- routes[-t]
337 merges <- merges + 1
338 }
339 } else if (command == "SPLIT") {
340 values <- c(0, rep(0, length(state$tour)))
341 parents <- integer(length(state$tour))
342 for (j in seq_along(state$tour)) {
343 best <- NULL
344 for (i in 0:(j - 1L)) {
345 part <- segment(state$tour, i, j)
346 if (load(part) <= capacity) {
347 choice <- c(values[i + 1L] + route_length(part), i)
348 if (lex_less(choice, best)) best <- choice
349 }
350 }
351 values[j + 1L] <- best[1]
352 parents[j] <- best[2]
353 }
354 routes <- list()
355 j <- length(state$tour)
356 while (j > 0) {
357 routes[[length(routes) + 1L]] <- segment(state$tour, parents[j], j)
358 j <- parents[j]
359 }
360 } else {
361 angle_less <- function(a, b) {
362 av <- point(a) - state$depot
363 bv <- point(b) - state$depot
364 az <- all(av == 0)
365 bz <- all(bv == 0)
366 if (az || bz) return(if (az == bz) a < b else az)
367 ah <- if (av[2] > 0 || (av[2] == 0 && av[1] > 0)) 0 else 1
368 bh <- if (bv[2] > 0 || (bv[2] == 0 && bv[1] > 0)) 0 else 1
369 if (ah != bh) return(ah < bh)
370 cross <- av[1] * bv[2] - av[2] * bv[1]
371 if (cross != 0) return(cross > 0)
372 lex_less(c(distance(0, a), a), c(distance(0, b), b))
373 }
374 sweep <- integer()
375 for (id in ids()) {
376 j <- 0
377 while (j < length(sweep) && !angle_less(id, sweep[j + 1L])) j <- j + 1L
378 sweep <- append(sweep, id, after = j)
379 }
380 routes <- list(integer())
381 for (id in sweep) {
382 last <- length(routes)
383 if (load(routes[[last]]) + stop_at(id)$demand > capacity) {
384 routes[[last + 1L]] <- integer()
385 last <- last + 1L
386 }
387 routes[[last]] <- c(routes[[last]], id)
388 }
389 }
390 state$capacity <- capacity
391 state$routes <- routes
392 normalize()
393 extra <- if (command == "VRP") paste0(" merges ", merges, ",") else ""
394 emit(tolower(command), ": capacity ", capacity, ",", extra, " routes ", length(state$routes), ", length ", plan_length())
395 route_lines()
396 }
397 execute <- function(line) {
398 words <- strsplit(line, " ", fixed = TRUE)[[1]]
399 command <- words[1]
400 args <- words[-1]
401 if (command == "DEPOT") {
402 state$depot <- as.integer(args)
403 emit("depot at (", paste(state$depot, collapse = ","), ")")
404 } else if (command == "START") {
405 state$departure <- minutes(args[1])
406 emit("start ", clock(state$departure))
407 } else if (command == "STOP") {
408 numbers <- as.integer(args[1:5])
409 id <- numbers[1]
410 if (!is.null(stop_at(id))) fail(paste0("stop ", id, " already exists"))
411 opening <- if (length(args) > 5) minutes(args[6]) else NULL
412 closing <- if (length(args) > 5) minutes(args[7]) else NULL
413 if (!is.null(opening) && opening > closing) fail("invalid window")
414 state$stops[[as.character(id)]] <- list(x = numbers[2], y = numbers[3], demand = numbers[4], service = numbers[5], open = opening, close = closing)
415 state$tour <- state$routes <- NULL
416 emit("stop ", id, " added")
417 } else if (command == "REMOVE") {
418 id <- as.integer(args[1])
419 if (is.null(stop_at(id))) fail(paste0("unknown stop ", id))
420 state$stops[[as.character(id)]] <- NULL
421 if (length(state$stops) == 0) {
422 state$tour <- state$routes <- NULL
423 } else {
424 if (!is.null(state$tour)) state$tour <- state$tour[state$tour != id]
425 if (!is.null(state$routes)) {
426 state$routes <- lapply(state$routes, function(route) route[route != id])
427 normalize()
428 }
429 }
430 emit("stop ", id, " removed")
431 } else if (command == "LIST") {
432 emit(if (is.null(state$depot)) "depot none" else paste0("depot (", paste(state$depot, collapse = ","), ")"))
433 emit("start ", clock(state$departure))
434 emit("stops ", length(state$stops))
435 for (id in ids()) {
436 s <- stop_at(id)
437 window <- if (is.null(s$open)) "none" else paste0(clock(s$open), "-", clock(s$close))
438 emit(id, " at (", s$x, ",", s$y, ") demand ", s$demand, " service ", s$service, " window ", window)
439 }
440 } else if (command == "MATRIX") {
441 if (is.null(state$depot)) fail("no depot")
442 locations <- c(0, ids())
443 cells <- list(c("", vapply(locations, location_name, character(1))))
444 for (a in locations) cells[[length(cells) + 1L]] <- c(location_name(a), as.character(vapply(locations, function(b) distance(a, b), numeric(1))))
445 width <- max(nchar(unlist(cells), type = "bytes"))
446 for (row in cells) emit(paste(sprintf(paste0("%", width, "s"), row), collapse = " "))
447 } else if (command %in% c("ORDER", "NEAREST", "INSERTION", "EXACT")) {
448 require_data()
449 if (command == "ORDER") {
450 route <- as.integer(args)
451 for (id in route) if (is.null(stop_at(id))) fail(paste0("unknown stop ", id))
452 seen <- integer()
453 for (id in route) {
454 if (id %in% seen) fail(paste0("duplicate stop ", id))
455 seen <- c(seen, id)
456 }
457 if (length(route) != length(state$stops)) fail(paste0("order has ", length(route), " stops, expected ", length(state$stops)))
458 } else if (command == "EXACT") {
459 if (length(state$stops) > 10) fail(paste0("too many stops for exact (", length(state$stops), ")"))
460 route <- exact()
461 } else {
462 route <- integer()
463 while (length(route) < length(state$stops)) {
464 remaining <- setdiff(ids(), route)
465 best <- NULL
466 for (c in remaining) {
467 if (command == "NEAREST") {
468 previous <- if (length(route) == 0) 0 else tail(route, 1)
469 choice <- c(distance(previous, c), c)
470 if (lex_less(choice, best)) best <- choice
471 } else {
472 for (j in 0:length(route)) {
473 candidate <- append(route, c, after = j)
474 choice <- c(route_length(candidate) - route_length(route), c, j)
475 if (lex_less(choice, best)) best <- choice
476 }
477 }
478 }
479 route <- if (command == "NEAREST") c(route, best[2]) else append(route, best[2], after = best[3])
480 }
481 }
482 state$tour <- route
483 emit(tolower(command), ": ", notation(route), " length ", route_length(route))
484 } else if (command == "TOUR") {
485 require_tour()
486 emit("tour: ", notation(state$tour), " length ", route_length(state$tour))
487 } else if (command %in% c("TWOOPT", "OROPT", "SWAP")) {
488 require_tour()
489 before <- route_length(state$tour)
490 result <- improve_tour(state$tour, command)
491 state$tour <- result$route
492 emit(tolower(command), ": moves ", result$count, ", length ", before, " -> ", route_length(state$tour))
493 } else if (command == "TIMECHECK") {
494 if (length(args) == 0) {
495 require_tour()
496 emit("timecheck tour depart ", clock(state$departure))
497 simulate(state$tour, TRUE)
498 } else {
499 require_plan()
500 r <- as.integer(args[1])
501 if (r > length(state$routes)) fail(paste0("no route ", r))
502 emit("timecheck route ", r, " depart ", clock(state$departure))
503 simulate(state$routes[[r]], TRUE)
504 }
505 } else if (command == "BOUND") {
506 require_data()
507 bounds()
508 } else if (command %in% c("VRP", "SPLIT", "SWEEP")) {
509 create_plan(command, args)
510 } else if (command == "ROUTES") {
511 require_plan()
512 emit("plan: capacity ", state$capacity, ", routes ", length(state$routes), ", length ", plan_length())
513 route_lines()
514 } else if (command %in% c("RELOCATE", "EXCHANGE", "CROSS")) {
515 require_plan()
516 before <- plan_length()
517 old_count <- length(state$routes)
518 count <- 0
519 repeat {
520 move <- plan_move(command)
521 if (is.null(move)) break
522 state$routes <- move$routes
523 emit(move$message)
524 normalize()
525 count <- count + 1
526 }
527 extra <- if (command == "EXCHANGE") "" else paste0(" routes ", old_count, " -> ", length(state$routes), ",")
528 emit(tolower(command), ": moves ", count, ",", extra, " length ", before, " -> ", plan_length())
529 } else if (command == "ROUTEOPT") {
530 require_plan()
531 before <- plan_length()
532 for (r in seq_along(state$routes)) {
533 old <- route_length(state$routes[[r]])
534 result <- improve_tour(state$routes[[r]], "TWOOPT", FALSE)
535 state$routes[[r]] <- result$route
536 emit("route ", r, ": moves ", result$count, ", length ", old, " -> ", route_length(result$route))
537 }
538 normalize()
539 emit("routeopt: length ", before, " -> ", plan_length())
540 } else if (command == "REPORT") {
541 emit("report: stops ", length(state$stops), ", demand ", sum(vapply(state$stops, function(s) s$demand, numeric(1))), ", service ", sum(vapply(state$stops, function(s) s$service, numeric(1))), ", windows ", sum(vapply(state$stops, function(s) !is.null(s$open), logical(1))))
542 if (is.null(state$tour)) {
543 emit("tour: none")
544 } else {
545 run <- simulate(state$tour)
546 emit("tour: length ", route_length(state$tour), ", return ", clock(run[1]), ", violations ", run[2], ", lateness ", run[3])
547 }
548 if (is.null(state$routes)) {
549 emit("plan: none")
550 } else {
551 runs <- do.call(rbind, lapply(state$routes, simulate))
552 denominator <- length(state$routes) * state$capacity
553 weight <- sum(vapply(state$routes, load, numeric(1)))
554 u <- (2000 * weight + denominator) %/% (2 * denominator)
555 emit("plan: capacity ", state$capacity, ", routes ", length(state$routes), ", length ", plan_length(), ", utilization ", u %/% 10, ".", u %% 10, "%, return ", clock(max(runs[, 1])), ", violations ", sum(runs[, 2]), ", lateness ", sum(runs[, 3]))
556 }
557 }
558 invisible(NULL)
559 }
