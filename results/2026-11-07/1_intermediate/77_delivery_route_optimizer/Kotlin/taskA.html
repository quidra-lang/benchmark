31 fun point(n: Int) = if (n == 0) depot!! else stops.getValue(n).let {

34 fun d(a: Int,b: Int): Int {
35 val x = point(a)
36 val y = point(b)
37 return kotlin.math.abs(x.first-y.first)+kotlin.math.abs(x.second-y.second)

39 fun length(p: List<Int>): Int {
40 val all = listOf(0)+p+0
41 return all.zipWithNext().sumOf {
42 (a,b) -> d(a,b)

45 fun load(p: List<Int>) = p.sumOf {

48 fun normalize(routes: List<List<Int>>) = routes.filter {

67 fun capacity(q: Int) {

70 fun run(p: List<Int>): Run {
71 var at = 0
72 var t = start
73 var wait = 0
74 var service = 0
75 var violations = 0
76 var late = 0
77 val out = mutableListOf<String>()
78 for (n in p) {
79 val s = stops.getValue(n)
80 val arrival = t+d(at,n)
81 val w = maxOf(0,(s.window?.first ?: arrival)-arrival)
82 val begin = arrival+w
83 val e = maxOf(0,arrival-(s.window?.second ?: arrival))
84 t = begin+s.service
85 wait += w
86 service += s.service
87 if (e > 0) violations++
88 late += e
89 out.add("$n arrive ${timeText(arrival)} wait $w begin ${timeText(begin)} leave ${timeText(t)} "+if (e > 0) "late $e" else "ok")
90 at = n
91 }
92 t += d(at,0)
93 out.add("return ${timeText(t)}, travel ${length(p)}, wait $wait, service $service, violations $violations, lateness $late")
94 return Run(out,t,violations,late)
95 }
96 fun firstTourMove(p: List<Int>,kind: String): TourMove? {
97 val before = length(p)
98 val n = p.size
99 if (kind == "OROPT") {
100 for (s in 1..3) for (i in 0..n-s) {
101 val segment = p.subList(i,i+s)
102 val rest = p.take(i)+p.drop(i+s)
103 for (j in 0..rest.size) if (j != i) {
104 val next = rest.take(j)+segment+rest.drop(j)
105 val gain = before-length(next)
106 if (gain > 0) return TourMove(next,"move ${segment.joinToString("-")} after ${name(if (j == 0) 0 else rest[j-1])} gain $gain")
107 }
108 }
109 } else for (i in 0 until n-1) for (j in i+1 until n) {
110 val next = p.toMutableList()
111 if (kind == "TWOOPT") for (k in i..j) next[k] = p[j-(k-i)] else {
112 next[i] = p[j]
113 next[j] = p[i]
114 }
115 val gain = before-length(next)
116 if (gain > 0) return TourMove(next,if (kind == "TWOOPT") "reverse ${p[i]}..${p[j]} gain $gain" else "swap ${p[i]} and ${p[j]} gain $gain")
117 }
118 return null
119 }
120 fun firstPlanMove(p: Plan,kind: String): PlanMove? {
121 val rs = p.routes
122 val k = rs.size
123 fun replaced(r: Int,a: List<Int>,t: Int,b: List<Int>) = rs.mapIndexed {
124 i,old -> if (i == r) a else if (i == t) b else old
125 }
126 if (kind == "RELOCATE") {
127 for (r in 0 until k) for (pos in rs[r].indices) for (t in 0 until k) if (t != r) {
128 val c = rs[r][pos]
129 if (load(rs[t])+stops.getValue(c).demand > p.capacity) continue
130 for (j in 0..rs[t].size) {
131 val a = rs[r].filterIndexed {
132 i,_ -> i != pos
133 }
134 val b = rs[t].take(j)+c+rs[t].drop(j)
135 val gain = length(rs[r])+length(rs[t])-length(a)-length(b)
136 if (gain > 0) return PlanMove(replaced(r,a,t,b),"relocate $c from route ${r+1} to route ${t+1} after ${name(if (j == 0) 0 else rs[t][j-1])} gain $gain")
137 }
138 }
139 } else for (r in 0 until k) for (t in r+1 until k) {
140 if (kind == "EXCHANGE") {
141 for (pos in rs[r].indices) for (j in rs[t].indices) {
142 val c = rs[r][pos]
143 val e = rs[t][j]
144 val a = rs[r].toMutableList()
145 val b = rs[t].toMutableList()
146 a[pos] = e
147 b[j] = c
148 if (load(a) > p.capacity || load(b) > p.capacity) continue
149 val gain = length(rs[r])+length(rs[t])-length(a)-length(b)
150 if (gain > 0) return PlanMove(replaced(r,a,t,b),"exchange $c (route ${r+1}) with $e (route ${t+1}) gain $gain")
151 }
152 } else for (i in 0..rs[r].size) for (j in 0..rs[t].size) {
153 val a = rs[r].take(i)+rs[t].drop(j)
154 val b = rs[t].take(j)+rs[r].drop(i)
155 if (load(a) > p.capacity || load(b) > p.capacity) continue
156 val gain = length(rs[r])+length(rs[t])-length(a)-length(b)
157 if (gain > 0) return PlanMove(replaced(r,a,t,b),"cross route ${r+1} after ${name(if (i == 0) 0 else rs[r][i-1])} with route ${t+1} after ${name(if (j == 0) 0 else rs[t][j-1])} gain $gain")
158 }
159 }
160 return null
161 }
162 fun mst(initial: Int,vertices: List<Int>): Pair<Int,List<String>> {
163 val entered = mutableListOf(initial)
164 val left = vertices.toMutableSet()
165 left.remove(initial)
166 var total = 0
167 val edges = mutableListOf<String>()
168 while (left.isNotEmpty()) {
169 var best: Triple<Int,Int,Int>? = null
170 for (c in left.sorted()) for ((i,p) in entered.withIndex()) {
171 val candidate = Triple(d(p,c),c,i)
172 val old = best
173 if (old == null || candidate.first < old.first || (candidate.first == old.first && (candidate.second < old.second || (candidate.second == old.second && candidate.third < old.third)))) best = candidate
174 }
175 val b = best!!
176 val p = entered[b.third]
177 entered.add(b.second)
178 left.remove(b.second)
179 total += b.first
180 edges.add("${name(p)}-${b.second}")
181 }
182 return total to edges
183 }
184 fun command(w: List<String>): List<String> {
185 val c = w[0]
186 val a = w.drop(1)
187 fun n(i: Int) = a[i].toInt()
188 val out = mutableListOf<String>()
189 when(c) {
190 "DEPOT" -> {
191 depot = n(0) to n(1)
192 return listOf("depot at (${a[0]},${a[1]})")
193 }
194 "START" -> {
195 start = time(a[0])
196 return listOf("start ${timeText(start)}")
197 }
198 "STOP" -> {
199 val id = n(0)
200 if (id in stops) fail("stop $id already exists")
201 val window = if (a.size == 7) time(a[5]) to time(a[6]) else null
202 if (window != null && window.first > window.second) fail("invalid window")
203 stops[id] = Stop(n(1),n(2),n(3),n(4),window)
204 tour = null
205 plan = null
206 return listOf("stop $id added")
207 }
208 "REMOVE" -> {
209 val id = n(0)
210 if (stops.remove(id) == null) fail("unknown stop $id")
211 if (stops.isEmpty()) {
212 tour = null
213 plan = null
214 } else {
215 tour = tour?.filter {
216 it != id
217 }
218 plan?.let {
219 it.routes = normalize(it.routes.map {
220 r -> r.filter {
221 it != id
222 }
223 })
224 }
225 }
226 return listOf("stop $id removed")
227 }
228 "LIST" -> {
229 out.add(depot?.let {
230 "depot (${it.first},${it.second})"
231 }?: "depot none")
232 out.add("start ${timeText(start)}")
233 out.add("stops ${stops.size}")
234 for ((id,s) in stops) out.add("$id at (${s.x},${s.y}) demand ${s.demand} service ${s.service} window "+(s.window?.let {
235 "${timeText(it.first)}-${timeText(it.second)}"
236 }?: "none"))
237 return out
238 }
239 "MATRIX" -> {
240 if (depot == null) fail("no depot")
241 val ids = listOf(0)+stops.keys
242 val rows = listOf(listOf("")+ids.map {
243 name(it)
244 })+ids.map {
245 x -> listOf(name(x))+ids.map {
246 d(x,it).toString()
247 }
248 }
249 val width = rows.flatten().maxOf {
250 it.length
251 }
252 return rows.map {
253 row -> row.joinToString(" ") {
254 it.padStart(width)
255 }
256 }
257 }
258 "TOUR" -> {
259 val p = tour()
260 return listOf("tour: ${notation(p)} length ${length(p)}")
261 }
262 "TIMECHECK" -> {
263 val p: List<Int>
264 if (a.isEmpty()) {
265 p = tour()
266 out.add("timecheck tour depart ${timeText(start)}")
267 } else {
268 val plan = plan()
269 val r = n(0)
270 if (r > plan.routes.size) fail("no route $r")
271 p = plan.routes[r-1]
272 out.add("timecheck route $r depart ${timeText(start)}")
273 }
274 return out+run(p).lines
275 }
276 "TWOOPT","OROPT","SWAP" -> {
277 var p = tour()
278 val before = length(p)
279 var count = 0
280 while (true) {
281 val move = firstTourMove(p,c) ?: break
282 out.add(move.text)
283 p = move.path
284 count++
285 }
286 tour = p
287 out.add("${c.lowercase()}: moves $count, length $before -> ${length(p)}")
288 return out

290 "ROUTES" -> {
291 val p = plan()
292 return listOf("plan: capacity ${p.capacity}, routes ${p.routes.size}, length ${total(p)}")+routeLines(p)
293 }
294 "RELOCATE","EXCHANGE","CROSS" -> {
295 val p = plan()
296 val before = total(p)
297 val countBefore = p.routes.size
298 var count = 0
299 while (true) {
300 val move = firstPlanMove(p,c) ?: break
301 out.add(move.text)
302 p.routes = normalize(move.routes)
303 count++
304 }
305 val routePart = if (c == "EXCHANGE") "" else "routes $countBefore -> ${p.routes.size}, "
306 out.add("${c.lowercase()}: moves $count, ${routePart}length $before -> ${total(p)}")
307 return out
308 }
309 "ROUTEOPT" -> {
310 val p = plan()
311 val before = total(p)
312 val routes = mutableListOf<List<Int>>()
313 for ((i,original) in p.routes.withIndex()) {
314 var route = original
315 var count = 0
316 while (true) {
317 val move = firstTourMove(route,"TWOOPT") ?: break
318 route = move.path
319 count++
320 }
321 out.add("route ${i+1}: moves $count, length ${length(original)} -> ${length(route)}")
322 routes.add(route)
323 }
324 p.routes = normalize(routes)
325 out.add("routeopt: length $before -> ${total(p)}")
326 return out
327 }
328 "REPORT" -> {
329 out.add("report: stops ${stops.size}, demand ${stops.values.sumOf { it.demand }}, service ${stops.values.sumOf { it.service }}, windows ${stops.values.count { it.window != null }}")
330 val tour = tour
331 if (tour == null) out.add("tour: none") else {
332 val run = run(tour)
333 out.add("tour: length ${length(tour)}, return ${timeText(run.returned)}, violations ${run.violations}, lateness ${run.lateness}")
334 }
335 val p = plan
336 if (p == null) out.add("plan: none") else {
337 val runs = p.routes.map {
338 run(it)
339 }
340 val capacity = p.routes.size*p.capacity
341 val u = (2000*stops.values.sumOf {
342 it.demand
343 }+capacity)/(2*capacity)
344 out.add("plan: capacity ${p.capacity}, routes ${p.routes.size}, length ${total(p)}, utilization ${u/10}.${u%10}%, return ${timeText(runs.maxOf { it.returned })}, violations ${runs.sumOf { it.violations }}, lateness ${runs.sumOf { it.lateness }}")
345 }
346 return out
347 }
348 "SPLIT" -> {
349 val p = tour()
350 val q = n(0)
351 capacity(q)
352 val values = IntArray(p.size+1) {
353 Int.MAX_VALUE
354 }
355 val previous = IntArray(p.size+1)
356 values[0] = 0
357 for (j in 1..p.size) for (i in 0 until j) {
358 val part = p.subList(i,j)
359 if (load(part) <= q) {
360 val candidate = values[i]+length(part)
361 if (candidate < values[j]) {
362 values[j] = candidate
363 previous[j] = i
364 }
365 }
366 }
367 val routes = mutableListOf<List<Int>>()
368 var j = p.size
369 while (j > 0) {
370 val i = previous[j]
371 routes.add(p.subList(i,j).toList())
372 j = i
373 }
374 val result = Plan(q,normalize(routes))
375 plan = result
376 return listOf("split: capacity $q, routes ${result.routes.size}, length ${total(result)}")+routeLines(result)
377 }
378 }
379 construction()
380 when(c) {
381 "ORDER" -> {
382 val ids = a.map {
383 it.toInt()
384 }
385 for (id in ids) if (id !in stops) fail("unknown stop $id")
386 val seen = mutableSetOf<Int>()
387 for (id in ids) if (!seen.add(id)) fail("duplicate stop $id")
388 if (ids.size != stops.size) fail("order has ${ids.size} stops, expected ${stops.size}")
389 tour = ids
390 }
391 "NEAREST" -> {
392 val remaining = stops.keys.toMutableSet()
393 val p = mutableListOf<Int>()
394 var at = 0
395 while (remaining.isNotEmpty()) {
396 val next = remaining.minWith(compareBy<Int> {
397 d(at,it)
398 }.thenBy {
399 it
400 })
401 p.add(next)
402 remaining.remove(next)
403 at = next
404 }
405 tour = p
406 }
407 "INSERTION" -> {
408 val p = mutableListOf<Int>()
409 while (p.size < stops.size) {
410 var best: Triple<Int,Int,Int>? = null
411 for (id in stops.keys) if (id !in p) for (j in 0..p.size) {
412 val left = if (j == 0) 0 else p[j-1]
413 val right = if (j == p.size) 0 else p[j]
414 val gain = d(left,id)+d(id,right)-d(left,right)
415 val old = best
416 if (old == null || gain < old.first || (gain == old.first && (id < old.second || (id == old.second && j < old.third)))) best = Triple(gain,id,j)
417 }
418 val b = best!!
419 p.add(b.third,b.second)
420 }
421 tour = p
422 }
423 "EXACT" -> {
424 if (stops.size > 10) fail("too many stops for exact (${stops.size})")
425 val ids = stops.keys.toList()
426 val memo = mutableMapOf<Pair<Int,Int>,Int>()
427 val full = (1 shl ids.size)-1
428 fun cost(mask: Int,at: Int): Int {
429 if (mask == full) return d(at,0)
430 return memo.getOrPut(mask to at) {
431 ids.indices.filter {
432 mask and (1 shl it) == 0
433 }.minOf {
434 d(at,ids[it])+cost(mask or (1 shl it),ids[it])
435 }
436 }
437 }
438 val p = mutableListOf<Int>()
439 var mask = 0
440 var at = 0
441 while (mask != full) {
442 val target = cost(mask,at)
443 val i = ids.indices.first {
444 mask and (1 shl it) == 0 && d(at,ids[it])+cost(mask or (1 shl it),ids[it]) == target
445 }
446 at = ids[i]
447 p.add(at)
448 mask = mask or (1 shl i)
449 }
450 tour = p
451 }
452 "BOUND" -> {
453 val ids = stops.keys.toList()
454 val (m,edges) = mst(0,ids)
455 val tree = if (ids.size == 1) 2*d(0,ids[0]) else mst(ids[0],ids).first+ids.map {
456 d(0,it)
457 }.sorted().take(2).sum()
458 val bound = maxOf(m,tree)
459 out.add("mst: ${edges.joinToString(" ")} (length $m)")
460 out.add("bound: mst $m, one-tree $tree, best $bound")
461 val p = tour
462 if (p == null) out.add("gap: no tour") else {
463 val l = length(p)
464 val u = if (bound == 0) 0 else (2000*(l-bound)+bound)/(2*bound)
465 out.add("gap: tour $l, ${u/10}.${u%10}%")
466 }
467 return out
468 }
469 "VRP","SWEEP" -> {
470 val q = n(0)
471 capacity(q)
472 val limit = if (a.size == 2) n(1) else null
473 if (limit != null) for (id in stops.keys) if (2*d(0,id) > limit) fail("stop $id too far for limit $limit")
474 var merges = 0
475 val routes = mutableListOf<List<Int>>()
476 if (c == "VRP") {
477 routes.addAll(stops.keys.map {
478 listOf(it)
479 })
480 val ids = stops.keys.toList()
481 val savings = ids.flatMapIndexed {
482 i,x -> ids.drop(i+1).map {
483 y -> Triple(d(0,x)+d(0,y)-d(x,y),x,y)
484 }
485 }.filter {
486 it.first > 0
487 }.sortedWith(compareByDescending<Triple<Int,Int,Int>> {
488 it.first
489 }.thenBy {
490 it.second
491 }.thenBy {
492 it.third
493 })
494 for ((_,x,y) in savings) {
495 val ia = routes.indexOfFirst {
496 x in it
497 }
498 val ib = routes.indexOfFirst {
499 y in it
500 }
501 if (ia == ib) continue
502 var left = routes[ia]
503 var right = routes[ib]
504 if (x != left.first() && x != left.last() || y != right.first() && y != right.last()) continue
505 if (load(left)+load(right) > q) continue
506 if (left.last() != x) left = left.asReversed()
507 if (right.first() != y) right = right.asReversed()
508 val joined = left+right
509 if (limit != null && length(joined) > limit) continue
510 routes[ia] = joined
511 routes.removeAt(ib)
512 merges++
513 }
514 } else {
515 val center = depot!!
516 val ids = stops.keys.sortedWith(Comparator {
517 x,y -> val a = point(x)
518 val b = point(y)
519 val ax = a.first-center.first
520 val ay = a.second-center.second
521 val bx = b.first-center.first
522 val by = b.second-center.second
523 val az = ax == 0 && ay == 0
524 val bz = bx == 0 && by == 0
525 if (az || bz) {
526 if (az != bz) {
527 if (az) -1 else 1
528 } else x.compareTo(y)
529 } else {
530 val ah = if (ay > 0 || ay == 0 && ax > 0) 0 else 1
531 val bh = if (by > 0 || by == 0 && bx > 0) 0 else 1
532 val cross = ax*by-ay*bx
533 if (ah != bh) ah-bh else if (cross != 0) {
534 if (cross > 0) -1 else 1
535 } else if (d(0,x) != d(0,y)) d(0,x)-d(0,y) else x-y
536 }
537 })
538 var route = mutableListOf<Int>()
539 var load = 0
540 for (id in ids) {
541 val demand = stops.getValue(id).demand
542 if (load+demand > q) {
543 routes.add(route)
544 route = mutableListOf()
545 load = 0
546 }
547 route.add(id)
548 load += demand
549 }
550 routes.add(route)
551 }
552 val result = Plan(q,normalize(routes))
553 plan = result
554 out.add(if (c == "VRP") "vrp: capacity $q, merges $merges, routes ${result.routes.size}, length ${total(result)}" else "sweep: capacity $q, routes ${result.routes.size}, length ${total(result)}")
555 out.addAll(routeLines(result))
556 return out
557 }
558 }
559 val p = tour()
560 return listOf("${c.lowercase()}: ${notation(p)} length ${length(p)}")
561 }
562}
