151 def simulate(self, sequence):
152 run = Run()
153 time = self.start
154 previous = DEPOT
155 for number in sequence:
156 stop = self.stops[number]
157 arrival = time + self.distance(previous, number)
158 wait = 0
159 lateness = 0
160 if stop.window is not None:
161 opening, closing = stop.window
162 wait = max(opening - arrival, 0)
163 lateness = max(arrival - closing, 0)
164 begin = arrival + wait
165 time = begin + stop.service
166 run.visits.append((number, arrival, wait, begin, time, lateness))
167 run.wait += wait
168 run.service += stop.service
169 run.lateness += lateness
170 if lateness > 0:
171 run.violations += 1
172 previous = number
173 run.return_time = time + self.distance(previous, DEPOT)
174 run.travel = self.length(sequence)
175 return run

322 def exact_tour(self):
323 numbers = sorted(self.stops)
324 count = len(numbers)
325 distance = self.distance
326 full = (1 << count) - 1
327 rest = [[0] * count for _ in range(full + 1)]
328 for i, number in enumerate(numbers):
329 rest[0][i] = distance(number, DEPOT)
330 for mask in range(1, full):
331 for i in range(count):
332 if not mask & (1 << i):
333 rest[mask][i] = min(
334 distance(numbers[i], numbers[j]) + rest[mask ^ (1 << j)][j]
335 for j in range(count)
336 if mask & (1 << j)
337 )
338 cost = min(distance(DEPOT, numbers[i]) + rest[full ^ (1 << i)][i] for i in range(count))
339 tour = []
340 current = DEPOT
341 mask = full
342 while mask:
343 choice = next(
344 j for j in range(count)
345 if mask & (1 << j)
346 and distance(current, numbers[j]) + rest[mask ^ (1 << j)][j] == cost
347 )
348 mask ^= 1 << choice
349 cost = rest[mask][choice]
350 current = numbers[choice]
351 tour.append(current)
352 return tour

358 def two_opt(self, sequence):
359 moves = []
360 while (move := self.first_reversal(sequence)) is not None:
361 start, end, gain = move
362 moves.append((sequence[start], sequence[end], gain))
363 sequence[start:end + 1] = reversed(sequence[start:end + 1])
364 return moves

366 def first_reversal(self, sequence):
367 distance = self.distance
368 path = [DEPOT, *sequence, DEPOT]
369 for i in range(1, len(sequence)):
370 for j in range(i + 1, len(sequence) + 1):
371 gain = (
372 distance(path[i - 1], path[i]) + distance(path[j], path[j + 1])
373 - distance(path[i - 1], path[j]) - distance(path[i], path[j + 1])
374 )
375 if gain > 0:
376 return i - 1, j - 1, gain
377 return None

387 def first_segment_move(self, tour):
388 count = len(tour)
389 path = [DEPOT, *tour, DEPOT]
390 for size in SEGMENT_SIZES:
391 for i in range(1, count - size + 2):
392 first, last = path[i], path[i + size - 1]
393 removal = self.detour(path[i - 1], first, last, path[i + size])
394 rest = [DEPOT, *tour[:i - 1], *tour[i + size - 1:], DEPOT]
395 for j in range(count - size + 1):
396 if j == i - 1:
397 continue
398 gain = removal - self.detour(rest[j], first, last, rest[j + 1])
399 if gain > 0:
400 return i - 1, size, j, gain
401 return None

403 def or_opt_tour(self, arguments):

419 def first_swap(self, tour):
420 length = self.length(tour)
421 for i in range(len(tour) - 1):
422 for j in range(i + 1, len(tour)):
423 swapped = tour.copy()
424 swapped[i], swapped[j] = tour[j], tour[i]
425 gain = length - self.length(swapped)
426 if gain > 0:
427 return i, j, gain

466 def spanning_tree(self, vertices):
467 tree = vertices[:1]
468 remaining = vertices[1:]
469 edges = []
470 while remaining:
471 weight, child, _, parent = min(
472 (self.distance(parent, child), child, index, parent)
473 for child in remaining
474 for index, parent in enumerate(tree)
475 )
476 edges.append((parent, child, weight))
477 tree.append(child)
478 remaining.remove(child)
479 return edges

481 def bound(self, arguments):
482 self.require_depot()
483 self.require_stops()
484 numbers = sorted(self.stops)
485 edges = self.spanning_tree([DEPOT, *numbers])
486 tree_length = sum(weight for _, _, weight in edges)
487 listing = " ".join(f"{place_name(parent)}-{child}" for parent, child, _ in edges)
488 self.emit(f"mst: {listing} (length {tree_length})")
489 depot_distances = sorted(self.distance(DEPOT, number) for number in numbers)
490 if len(numbers) == 1:
491 one_tree = 2 * depot_distances[0]
492 else:
493 stop_tree = sum(weight for _, _, weight in self.spanning_tree(numbers))
494 one_tree = stop_tree + depot_distances[0] + depot_distances[1]
495 best = max(tree_length, one_tree)
496 self.emit(f"bound: mst {tree_length}, one-tree {one_tree}, best {best}")
497 if self.tour is None:
498 self.emit("gap: no tour")
499 return
500 length = self.length(self.tour)
501 gap = (2000 * (length - best) + best) // (2 * best) if best > 0 else 0
502 self.emit(f"gap: tour {length}, {tenths(gap)}%")

513 def savings(self, arguments):
514 capacity = int(arguments[0])
515 limit = int(arguments[1]) if len(arguments) == 2 else None
516 self.require_depot()
517 self.require_stops()
518 self.require_capacity(capacity)
519 numbers = sorted(self.stops)
520 if limit is not None:
521 for number in numbers:
522 if 2 * self.distance(DEPOT, number) > limit:
523 raise CommandError(f"stop {number} too far for limit {limit}")
524 pairs = []
525 for index, i in enumerate(numbers):
526 for j in numbers[index + 1:]:
527 saving = self.distance(DEPOT, i) + self.distance(DEPOT, j) - self.distance(i, j)
528 if saving > 0:
529 pairs.append((-saving, i, j))
530 pairs.sort()
531 routes = [[number] for number in numbers]
532 route_of = {route[0]: route for route in routes}
533 merges = 0
534 for _, i, j in pairs:
535 first, second = route_of[i], route_of[j]
536 if (
537 first is second
538 or i not in (first[0], first[-1])
539 or j not in (second[0], second[-1])
540 ):
541 continue
542 if self.load(first) + self.load(second) > capacity:
543 continue
544 merged = (first[::-1] if first[0] == i else first) + (
545 second[::-1] if second[-1] == j else second
546 )
547 if limit is not None and self.length(merged) > limit:
548 continue
549 routes.remove(first)
550 routes.remove(second)
551 routes.append(merged)
552 for number in merged:
553 route_of[number] = merged
554 merges += 1
555 self.plan = Plan(capacity, routes)
556 self.emit_plan(f"vrp: capacity {capacity}, merges {merges}")

558 def split(self, arguments):
559 capacity = int(arguments[0])
560 self.require_tour()
561 self.require_capacity(capacity)
562 tour = self.tour
563 count = len(tour)
564 best = [0] + [None] * count
565 previous = [0] * (count + 1)
566 for end in range(1, count + 1):
567 for start in range(end):
568 segment = tour[start:end]
569 if self.load(segment) > capacity:
570 continue
571 value = best[start] + self.length(segment)
572 if best[end] is None or value < best[end]:
573 best[end] = value
574 previous[end] = start
575 routes = []
576 end = count
577 while end > 0:
578 routes.append(tour[previous[end]:end])
579 end = previous[end]
580 self.plan = Plan(capacity, routes)
581 self.emit_plan(f"split: capacity {capacity}")

583 def sweep_order(self):
584 depot_x, depot_y = self.depot
585 offsets = {
586 number: (stop.x - depot_x, stop.y - depot_y) for number, stop in self.stops.items()
587 }
588
589 def compare(first, second):
590 (x1, y1), (x2, y2) = offsets[first], offsets[second]
591 half1, half2 = half_plane(x1, y1), half_plane(x2, y2)
592 if half1 != half2:
593 return half1 - half2
594 cross = x1 * y2 - y1 * x2
595 if cross != 0:
596 return -1 if cross > 0 else 1
597 distance1, distance2 = abs(x1) + abs(y1), abs(x2) + abs(y2)
598 if distance1 != distance2:
599 return distance1 - distance2
600 return first - second
601
602 numbers = sorted(offsets)
603 at_depot = [number for number in numbers if offsets[number] == (0, 0)]
604 others = [number for number in numbers if offsets[number] != (0, 0)]
605 return at_depot + sorted(others, key=cmp_to_key(compare))

628 def first_relocation(self):
629 routes = self.plan.routes
630 loads = [self.load(route) for route in routes]
631 for r, source in enumerate(routes):
632 path = [DEPOT, *source, DEPOT]
633 for p in range(1, len(source) + 1):
634 stop = path[p]
635 removal = self.detour(path[p - 1], stop, stop, path[p + 1])
636 demand = self.stops[stop].demand
637 for t, target in enumerate(routes):
638 if t == r or loads[t] + demand > self.plan.capacity:
639 continue
640 around = [DEPOT, *target, DEPOT]
641 for j in range(len(target) + 1):
642 gain = removal - self.detour(around[j], stop, stop, around[j + 1])
643 if gain > 0:
644 return r, p - 1, t, j, gain
645 return None

669 def first_exchange(self):
670 routes = self.plan.routes
671 capacity = self.plan.capacity
672 for r, first in enumerate(routes):
673 for t in range(r + 1, len(routes)):
674 second = routes[t]
675 first_load = self.load(first)
676 second_load = self.load(second)
677 total = self.length(first) + self.length(second)
678 for p, stop in enumerate(first):
679 for j, other in enumerate(second):
680 shift = self.stops[other].demand - self.stops[stop].demand
681 if first_load + shift > capacity or second_load - shift > capacity:
682 continue
683 changed_first = first[:p] + [other] + first[p + 1:]
684 changed_second = second[:j] + [stop] + second[j + 1:]
685 gain = total - self.length(changed_first) - self.length(changed_second)
686 if gain > 0:
687 return r, p, t, j, gain
688 return None

690 def exchange(self, arguments):
691 self.require_plan()
692 plan = self.plan
693 length_before = self.plan_length()
694 moves = 0
695 while (move := self.first_exchange()) is not None:
696 r, p, t, j, gain = move
697 routes = plan.routes
698 stop, other = routes[r][p], routes[t][j]
699 self.emit(f"exchange {stop} (route {r + 1}) with {other} (route {t + 1}) gain {gain}")
700 routes[r][p], routes[t][j] = other, stop
701 plan.normalize()
702 moves += 1
703 self.emit(f"exchange: moves {moves}, length {length_before} -> {self.plan_length()}")

705 def first_cross(self):
706 distance = self.distance
707 routes = self.plan.routes
708 capacity = self.plan.capacity
709 for r, first in enumerate(routes):
710 for t in range(r + 1, len(routes)):
711 second = routes[t]
712 a = [DEPOT, *first, DEPOT]
713 b = [DEPOT, *second, DEPOT]
714 for i in range(len(first) + 1):
715 for j in range(len(second) + 1):
716 if (
717 self.load(first[:i] + second[j:]) > capacity
718 or self.load(second[:j] + first[i:]) > capacity
719 ):
720 continue
721 gain = (
722 distance(a[i], a[i + 1]) + distance(b[j], b[j + 1])
723 - distance(a[i], b[j + 1]) - distance(b[j], a[i + 1])
724 )
725 if gain > 0:
726 return r, i, t, j, gain
727 return None

729 def cross(self, arguments):
730 self.require_plan()
731 plan = self.plan
732 routes_before = len(plan.routes)
733 length_before = self.plan_length()
734 moves = 0
735 while (move := self.first_cross()) is not None:
736 r, i, t, j, gain = move
737 routes = plan.routes
738 first, second = routes[r], routes[t]
739 self.emit(
740 f"cross route {r + 1} after {name_at(first, i)} "
741 f"with route {t + 1} after {name_at(second, j)} gain {gain}"
742 )
743 routes[r], routes[t] = first[:i] + second[j:], second[:j] + first[i:]
744 plan.normalize()
745 moves += 1
746 self.emit(
747 f"cross: moves {moves}, routes {routes_before} -> {len(plan.routes)}, "
748 f"length {length_before} -> {self.plan_length()}"
749 )

764 stops = self.stops.values()
765 demand = sum(stop.demand for stop in stops)
766 service = sum(stop.service for stop in stops)
767 windows = sum(stop.window is not None for stop in stops)
