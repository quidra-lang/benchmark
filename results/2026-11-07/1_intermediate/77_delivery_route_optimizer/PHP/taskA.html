140 private function simulate(array $route, bool $output = false): array
141 {
142 $time = $this->departure;
143 $previous = 0;
144 $waiting = $service = $violations = $lateness = 0;
145 foreach ($route as $id) {
146 $stop = $this->stops[$id];
147 $arrival = $time + $this->distance($previous, $id);
148 $wait = $stop['open'] === null ? 0 : max(0, $stop['open'] - $arrival);
149 $late = $stop['close'] === null ? 0 : max(0, $arrival - $stop['close']);
150 $begin = $arrival + $wait;
151 $time = $begin + $stop['service'];
152 $waiting += $wait;
153 $service += $stop['service'];
154 $violations += $late > 0 ? 1 : 0;
155 $lateness += $late;
156 if ($output) {
157 echo "$id arrive ", $this->clock($arrival), " wait $wait begin ", $this->clock($begin), ' leave ', $this->clock($time), ' ', $late > 0 ? "late $late" : 'ok', "\n";
158 }
159 $previous = $id;
160 }
161 $time += $this->distance($previous, 0);
162 if ($output) {
163 echo 'return ', $this->clock($time), ', travel ', $this->length($route), ", wait $waiting, service $service, violations $violations, lateness $lateness\n";
164 }
165 return [$time, $violations, $lateness];
166 }

168 private function exact(): array
169 {
170 $stops = $this->ids();
171 $full = (1 << count($stops)) - 1;
172 $memo = [];
173 $solve = function (int $mask, int $previous) use (&$solve, &$memo, $stops, $full): int {
174 if ($mask === $full) {
175 return $this->distance($previous, 0);
176 }
177 $key = "$mask,$previous";
178 if (isset($memo[$key])) {
179 return $memo[$key];
180 }
181 $best = null;
182 foreach ($stops as $i => $id) {
183 if (($mask & (1 << $i)) === 0) {
184 $cost = $this->distance($previous, $id) + $solve($mask | (1 << $i), $id);
185 $best = $best === null ? $cost : min($best, $cost);
186 }
187 }
188 return $memo[$key] = $best;
189 };
190 $route = [];
191 $mask = $previous = 0;
192 while ($mask !== $full) {
193 $best = $solve($mask, $previous);
194 foreach ($stops as $i => $id) {
195 if (($mask & (1 << $i)) === 0 && $this->distance($previous, $id) + $solve($mask | (1 << $i), $id) === $best) {
196 $route[] = $id;
197 $mask |= 1 << $i;
198 $previous = $id;
199 break;
200 }
201 }
202 }
203 return $route;
204 }

206 private function tourMove(array $route, string $method): ?array
207 {
208 $old = $this->length($route);
209 $n = count($route);
210 if ($method === 'OROPT') {
211 for ($size = 1; $size <= 3; $size++) {
212 for ($i = 0; $i + $size <= $n; $i++) {
213 $segment = array_slice($route, $i, $size);
214 $remaining = $route;
215 array_splice($remaining, $i, $size);
216 for ($j = 0; $j <= count($remaining); $j++) {
217 if ($j === $i) {
218 continue;
219 }
220 $candidate = $remaining;
221 array_splice($candidate, $j, 0, $segment);
222 $gain = $old - $this->length($candidate);
223 if ($gain > 0) {
224 $after = $j === 0 ? 'D' : $remaining[$j - 1];
225 return [$candidate, 'move ' . implode('-', $segment) . " after $after gain $gain"];
226 }
227 }
228 }
229 }
230 } else {
231 for ($i = 0; $i < $n; $i++) {
232 for ($j = $i + 1; $j < $n; $j++) {
233 $candidate = $route;
234 if ($method === 'TWOOPT') {
235 array_splice($candidate, $i, $j - $i + 1, array_reverse(array_slice($route, $i, $j - $i + 1)));
236 } else {
237 [$candidate[$i], $candidate[$j]] = [$candidate[$j], $candidate[$i]];
238 }
239 $gain = $old - $this->length($candidate);
240 if ($gain > 0) {
241 $message = $method === 'TWOOPT' ? "reverse {$route[$i]}..{$route[$j]} gain $gain" : "swap {$route[$i]} and {$route[$j]} gain $gain";
242 return [$candidate, $message];
243 }
244 }
245 }
246 }
247 return null;

263 private function planMove(string $method): ?array
264 {
265 $before = $this->planLength();
266 $k = count($this->routes);
267 foreach ($this->routes as $r => $a) {
268 if ($method === 'RELOCATE') {
269 foreach ($a as $p => $c) {
270 foreach ($this->routes as $t => $b) {
271 if ($r === $t || $this->load($b) + $this->stops[$c]['demand'] > $this->capacity) {
272 continue;
273 }
274 for ($j = 0; $j <= count($b); $j++) {
275 $candidate = $this->routes;
276 array_splice($candidate[$r], $p, 1);
277 array_splice($candidate[$t], $j, 0, [$c]);
278 $gain = $before - $this->planLength($candidate);
279 if ($gain > 0) {
280 $after = $j === 0 ? 'D' : $b[$j - 1];
281 return [$candidate, 'relocate ' . $c . ' from route ' . ($r + 1) . ' to route ' . ($t + 1) . " after $after gain $gain"];
282 }
283 }
284 }
285 }
286 } else {
287 for ($t = $r + 1; $t < $k; $t++) {
288 $b = $this->routes[$t];
289 if ($method === 'EXCHANGE') {
290 foreach ($a as $p => $c) {
291 foreach ($b as $j => $e) {
292 if ($this->load($a) - $this->stops[$c]['demand'] + $this->stops[$e]['demand'] > $this->capacity || $this->load($b) - $this->stops[$e]['demand'] + $this->stops[$c]['demand'] > $this->capacity) {
293 continue;
294 }
295 $candidate = $this->routes;
296 $candidate[$r][$p] = $e;
297 $candidate[$t][$j] = $c;
298 $gain = $before - $this->planLength($candidate);
299 if ($gain > 0) {
300 return [$candidate, "exchange $c (route " . ($r + 1) . ") with $e (route " . ($t + 1) . ") gain $gain"];
301 }
302 }
303 }
304 } else {
305 for ($i = 0; $i <= count($a); $i++) {
306 for ($j = 0; $j <= count($b); $j++) {
307 $left = array_merge(array_slice($a, 0, $i), array_slice($b, $j));
308 $right = array_merge(array_slice($b, 0, $j), array_slice($a, $i));
309 if ($this->load($left) > $this->capacity || $this->load($right) > $this->capacity) {
310 continue;
311 }
312 $candidate = $this->routes;
313 $candidate[$r] = $left;
314 $candidate[$t] = $right;
315 $gain = $before - $this->planLength($candidate);
316 if ($gain > 0) {
317 $x = $i === 0 ? 'D' : $a[$i - 1];
318 $y = $j === 0 ? 'D' : $b[$j - 1];
319 return [$candidate, 'cross route ' . ($r + 1) . " after $x with route " . ($t + 1) . " after $y gain $gain"];
320 }
321 }
322 }
323 }
324 }
325 }
326 }
327 return null;
328 }

330 private function bounds(): void
331 {
332 $tree = [0];
333 $remaining = $this->ids();
334 $edges = [];
335 $mst = 0;
336 while ($remaining !== []) {
337 $best = null;
338 foreach ($remaining as $id) {
339 foreach ($tree as $i => $parent) {
340 $choice = [$this->distance($parent, $id), $id, $i];
341 if ($best === null || $choice < $best) {
342 $best = $choice;
343 }
344 }
345 }
346 [$weight, $c, $order] = $best;
347 $edges[] = $this->name($tree[$order]) . "-$c";
348 $mst += $weight;
349 $tree[] = $c;
350 $remaining = array_values(array_diff($remaining, [$c]));
351 }
352 echo 'mst: ', implode(' ', $edges), " (length $mst)\n";
353 $ids = $this->ids();
354 if (count($ids) === 1) {
355 $oneTree = 2 * $this->distance(0, $ids[0]);
356 } else {
357 $tree = [$ids[0]];
358 $remaining = array_slice($ids, 1);
359 $cost = 0;
360 while ($remaining !== []) {
361 $best = null;
362 foreach ($remaining as $id) {
363 foreach ($tree as $parent) {
364 $choice = [$this->distance($parent, $id), $id];
365 if ($best === null || $choice < $best) {
366 $best = $choice;
367 }
368 }
369 }
370 [$weight, $c] = $best;
371 $cost += $weight;
372 $tree[] = $c;
373 $remaining = array_values(array_diff($remaining, [$c]));
374 }
375 $distances = array_map(fn($id) => $this->distance(0, $id), $ids);
376 sort($distances, SORT_NUMERIC);
377 $oneTree = $cost + $distances[0] + $distances[1];
378 }
379 $best = max($mst, $oneTree);
380 echo "bound: mst $mst, one-tree $oneTree, best $best\n";
381 if ($this->tour === null) {
382 echo "gap: no tour\n";
383 } else {
384 $value = $this->length($this->tour);
385 $u = $best > 0 ? intdiv(2000 * ($value - $best) + $best, 2 * $best) : 0;
386 echo "gap: tour $value, ", intdiv($u, 10), '.', $u % 10, "%\n";
387 }

390 private function createPlan(string $command, array $args): void
391 {
392 $capacity = (int)$args[0];
393 if ($command === 'SPLIT') {
394 $this->requireTour();
395 } else {
396 $this->requireData();
397 }
398 $this->checkCapacity($capacity);
399 $limit = $command === 'VRP' && isset($args[1]) ? (int)$args[1] : null;
400 if ($limit !== null) {
401 foreach ($this->ids() as $id) {
402 if (2 * $this->distance(0, $id) > $limit) {
403 throw new RouteError("stop $id too far for limit $limit");
404 }
405 }
406 }
407 $merges = 0;
408 if ($command === 'VRP') {
409 $ids = $this->ids();
410 $routes = array_map(fn($id) => [$id], $ids);
411 $savings = [];
412 foreach ($ids as $p => $i) {
413 for ($q = $p + 1; $q < count($ids); $q++) {
414 $j = $ids[$q];
415 $saving = $this->distance(0, $i) + $this->distance(0, $j) - $this->distance($i, $j);
416 if ($saving > 0) {
417 $savings[] = [-$saving, $i, $j];
418 }
419 }
420 }
421 usort($savings, fn($a, $b) => $a <=> $b);
422 foreach ($savings as [, $i, $j]) {
423 $r = $t = null;
424 foreach ($routes as $index => $route) {
425 if (in_array($i, $route, true)) {
426 $r = $index;
427 }
428 if (in_array($j, $route, true)) {
429 $t = $index;
430 }
431 }
432 $a = $routes[$r];
433 $b = $routes[$t];
434 if ($r === $t || !in_array($i, [$a[0], $a[count($a) - 1]], true) || !in_array($j, [$b[0], $b[count($b) - 1]], true) || $this->load($a) + $this->load($b) > $capacity) {
435 continue;
436 }
437 $joined = array_merge($a[0] === $i ? array_reverse($a) : $a, $b[count($b) - 1] === $j ? array_reverse($b) : $b);
438 if ($limit !== null && $this->length($joined) > $limit) {
439 continue;
440 }
441 $routes[$r] = $joined;
442 array_splice($routes, $t, 1);
443 $merges++;
444 }
445 } elseif ($command === 'SPLIT') {
446 $values = [0];
447 $parents = [];
448 for ($j = 1; $j <= count($this->tour); $j++) {
449 $best = null;
450 for ($i = 0; $i < $j; $i++) {
451 $segment = array_slice($this->tour, $i, $j - $i);
452 if ($this->load($segment) <= $capacity) {
453 $choice = [$values[$i] + $this->length($segment), $i];
454 if ($best === null || $choice < $best) {
455 $best = $choice;
456 }
457 }
458 }
459 [$values[$j], $parents[$j]] = $best;
460 }
461 $routes = [];
462 for ($j = count($this->tour); $j > 0; $j = $parents[$j]) {
463 $routes[] = array_slice($this->tour, $parents[$j], $j - $parents[$j]);
464 }
465 } else {
466 $sweep = $this->ids();
467 usort($sweep, function ($a, $b) {
468 [$ax, $ay] = $this->point($a);
469 [$bx, $by] = $this->point($b);
470 $ax -= $this->depot[0];
471 $ay -= $this->depot[1];
472 $bx -= $this->depot[0];
473 $by -= $this->depot[1];
474 $az = $ax === 0 && $ay === 0;
475 $bz = $bx === 0 && $by === 0;
476 if ($az || $bz) {
477 return $az === $bz ? $a <=> $b : ($az ? -1 : 1);
478 }
479 $ah = $ay > 0 || ($ay === 0 && $ax > 0) ? 0 : 1;
480 $bh = $by > 0 || ($by === 0 && $bx > 0) ? 0 : 1;
481 $cross = $ax * $by - $ay * $bx;
482 if ($ah !== $bh) {
483 return $ah <=> $bh;
484 }
485 return $cross !== 0 ? 0 <=> $cross : [$this->distance(0, $a), $a] <=> [$this->distance(0, $b), $b];
486 });
487 $routes = [[]];
488 foreach ($sweep as $id) {
489 $last = count($routes) - 1;
490 if ($this->load($routes[$last]) + $this->stops[$id]['demand'] > $capacity) {
491 $routes[] = [];
492 $last++;
493 }
494 $routes[$last][] = $id;
495 }
496 }
497 $this->capacity = $capacity;
498 $this->routes = $routes;
499 $this->normalize();
500 echo strtolower($command), ": capacity $capacity,", $command === 'VRP' ? " merges $merges," : '', ' routes ', count($this->routes), ', length ', $this->planLength(), "\n";
501 $this->routeLines();
502 }

504 public function execute(string $line): void
505 {
506 $args = explode(' ', $line);
507 $command = array_shift($args);
508 try {
509 switch ($command) {
510 case 'DEPOT':
511 $this->depot = array_map('intval', $args);
512 echo 'depot at (', implode(',', $this->depot), ")\n";
513 break;
514 case 'START':
515 $this->departure = $this->minutes($args[0]);
516 echo 'start ', $this->clock($this->departure), "\n";
517 break;
518 case 'STOP':
519 [$id, $x, $y, $demand, $service] = array_map('intval', array_slice($args, 0, 5));
520 if (isset($this->stops[$id])) {
521 throw new RouteError("stop $id already exists");
522 }
523 $opening = isset($args[5]) ? $this->minutes($args[5]) : null;
524 $closing = isset($args[6]) ? $this->minutes($args[6]) : null;
525 if ($opening !== null && $opening > $closing) {
526 throw new RouteError('invalid window');
527 }
528 $this->stops[$id] = ['x' => $x, 'y' => $y, 'demand' => $demand, 'service' => $service, 'open' => $opening, 'close' => $closing];
529 $this->tour = $this->routes = null;
530 echo "stop $id added\n";
531 break;
532 case 'REMOVE':
533 $id = (int)$args[0];
534 if (!isset($this->stops[$id])) {
535 throw new RouteError("unknown stop $id");
536 }
537 unset($this->stops[$id]);
538 if ($this->stops === []) {
539 $this->tour = $this->routes = null;
540 } else {
541 if ($this->tour !== null) {
542 $this->tour = array_values(array_diff($this->tour, [$id]));
543 }
544 if ($this->routes !== null) {
545 $this->routes = array_map(fn($route) => array_values(array_diff($route, [$id])), $this->routes);
546 $this->normalize();
547 }
548 }
549 echo "stop $id removed\n";
550 break;
551 case 'LIST':
552 echo $this->depot === null ? 'depot none' : 'depot (' . implode(',', $this->depot) . ')', "\n";
553 echo 'start ', $this->clock($this->departure), "\nstops ", count($this->stops), "\n";
554 foreach ($this->ids() as $id) {
555 $s = $this->stops[$id];
556 $window = $s['open'] === null ? 'none' : $this->clock($s['open']) . '-' . $this->clock($s['close']);
557 echo "$id at ({$s['x']},{$s['y']}) demand {$s['demand']} service {$s['service']} window $window\n";
558 }
559 break;
560 case 'MATRIX':
561 if ($this->depot === null) {
562 throw new RouteError('no depot');
563 }
564 $locations = array_merge([0], $this->ids());
565 $cells = [array_merge([''], array_map(fn($id) => $this->name($id), $locations))];
566 foreach ($locations as $a) {
567 $cells[] = array_merge([$this->name($a)], array_map(fn($b) => (string)$this->distance($a, $b), $locations));
568 }
569 $width = 0;
570 foreach ($cells as $row) {
571 foreach ($row as $cell) {
572 $width = max($width, strlen($cell));
573 }
574 }
575 foreach ($cells as $row) {
576 echo implode(' ', array_map(fn($s) => str_pad($s, $width, ' ', STR_PAD_LEFT), $row)), "\n";
577 }
578 break;
579 case 'ORDER':
580 case 'NEAREST':
581 case 'INSERTION':
582 case 'EXACT':
583 $this->requireData();
584 if ($command === 'ORDER') {
585 $route = array_map('intval', $args);
586 foreach ($route as $id) {
587 if (!isset($this->stops[$id])) {
588 throw new RouteError("unknown stop $id");
589 }
590 }
591 $seen = [];
592 foreach ($route as $id) {
593 if (isset($seen[$id])) {
594 throw new RouteError("duplicate stop $id");
595 }
596 $seen[$id] = true;
597 }
598 if (count($route) !== count($this->stops)) {
599 throw new RouteError('order has ' . count($route) . ' stops, expected ' . count($this->stops));
600 }
601 } elseif ($command === 'EXACT') {
602 if (count($this->stops) > 10) {
603 throw new RouteError('too many stops for exact (' . count($this->stops) . ')');
604 }
605 $route = $this->exact();
606 } else {
607 $route = [];
608 while (count($route) < count($this->stops)) {
609 $remaining = array_values(array_diff($this->ids(), $route));
610 $best = null;
611 foreach ($remaining as $c) {
612 if ($command === 'NEAREST') {
613 $choice = [$this->distance($route === [] ? 0 : $route[count($route) - 1], $c), $c];
614 if ($best === null || $choice < $best) {
615 $best = $choice;
616 }
617 } else {
618 for ($j = 0; $j <= count($route); $j++) {
619 $candidate = $route;
620 array_splice($candidate, $j, 0, [$c]);
621 $choice = [$this->length($candidate) - $this->length($route), $c, $j];
622 if ($best === null || $choice < $best) {
623 $best = $choice;
624 }
625 }
626 }
627 }
628 if ($command === 'NEAREST') {
629 $route[] = $best[1];
630 } else {
631 array_splice($route, $best[2], 0, [$best[1]]);
632 }
633 }
634 }
635 $this->tour = $route;
636 echo strtolower($command), ': ', $this->notation($route), ' length ', $this->length($route), "\n";
637 break;
638 case 'TOUR':
639 $this->requireTour();
640 echo 'tour: ', $this->notation($this->tour), ' length ', $this->length($this->tour), "\n";
641 break;
642 case 'TWOOPT':
643 case 'OROPT':
644 case 'SWAP':
645 $this->requireTour();
646 $before = $this->length($this->tour);
647 [$this->tour, $count] = $this->improveTour($this->tour, $command);
648 echo strtolower($command), ": moves $count, length $before -> ", $this->length($this->tour), "\n";
649 break;
650 case 'TIMECHECK':
651 if ($args === []) {
652 $this->requireTour();
653 echo 'timecheck tour depart ', $this->clock($this->departure), "\n";
654 $this->simulate($this->tour, true);
655 } else {
656 $this->requirePlan();
657 $r = (int)$args[0];
658 if ($r > count($this->routes)) {
659 throw new RouteError("no route $r");
660 }
661 echo "timecheck route $r depart ", $this->clock($this->departure), "\n";
662 $this->simulate($this->routes[$r - 1], true);
663 }
664 break;
665 case 'BOUND':
666 $this->requireData();
667 $this->bounds();
668 break;
669 case 'VRP':
670 case 'SPLIT':
671 case 'SWEEP':
672 $this->createPlan($command, $args);
673 break;
674 case 'ROUTES':
675 $this->requirePlan();
676 echo 'plan: capacity ', $this->capacity, ', routes ', count($this->routes), ', length ', $this->planLength(), "\n";
677 $this->routeLines();
678 break;
679 case 'RELOCATE':
680 case 'EXCHANGE':
681 case 'CROSS':
682 $this->requirePlan();
683 $before = $this->planLength();
684 $oldCount = count($this->routes);
685 $count = 0;
686 while (($move = $this->planMove($command)) !== null) {
687 [$this->routes, $message] = $move;
688 echo $message, "\n";
689 $this->normalize();
690 $count++;
691 }
692 echo strtolower($command), ": moves $count,", $command === 'EXCHANGE' ? '' : " routes $oldCount -> " . count($this->routes) . ',', " length $before -> ", $this->planLength(), "\n";
693 break;
694 case 'ROUTEOPT':
695 $this->requirePlan();
696 $before = $this->planLength();
697 foreach ($this->routes as $r => $route) {
698 $old = $this->length($route);
699 [$this->routes[$r], $count] = $this->improveTour($route, 'TWOOPT', false);
700 echo 'route ', $r + 1, ": moves $count, length $old -> ", $this->length($this->routes[$r]), "\n";
701 }
702 $this->normalize();
703 echo "routeopt: length $before -> ", $this->planLength(), "\n";
704 break;
705 case 'REPORT':
706 $demand = $service = $windows = 0;
707 foreach ($this->stops as $stop) {
708 $demand += $stop['demand'];
709 $service += $stop['service'];
710 $windows += $stop['open'] !== null ? 1 : 0;
711 }
712 echo 'report: stops ', count($this->stops), ", demand $demand, service $service, windows $windows\n";
713 if ($this->tour === null) {
714 echo "tour: none\n";
715 } else {
716 [$time, $violations, $late] = $this->simulate($this->tour);
717 echo 'tour: length ', $this->length($this->tour), ', return ', $this->clock($time), ", violations $violations, lateness $late\n";
718 }
719 if ($this->routes === null) {
720 echo "plan: none\n";
721 } else {
722 $runs = array_map(fn($route) => $this->simulate($route), $this->routes);
723 $denominator = count($this->routes) * $this->capacity;
724 $weight = array_sum(array_map(fn($route) => $this->load($route), $this->routes));
725 $u = intdiv(2000 * $weight + $denominator, 2 * $denominator);
726 echo 'plan: capacity ', $this->capacity, ', routes ', count($this->routes), ', length ', $this->planLength(), ', utilization ', intdiv($u, 10), '.', $u % 10, '%, return ', $this->clock(max(array_column($runs, 0))), ', violations ', array_sum(array_column($runs, 1)), ', lateness ', array_sum(array_column($runs, 2)), "\n";
727 }
728 break;
729 }
730 } catch (RouteError $error) {
731 echo 'error: ', $error->getMessage(), "\n";
732 }
733 }
734}
