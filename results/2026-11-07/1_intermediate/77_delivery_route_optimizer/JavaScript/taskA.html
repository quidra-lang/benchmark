12function ids() {
13 return [...stops.keys()].sort((a, b) => a - b);

15function point(id) {

18function distance(a, b) {
19 const x = point(a),
20 y = point(b);
21 return Math.abs(x.x - y.x) + Math.abs(x.y - y.y);
22}
23function length(route) {
24 const path = [0, ...route, 0];
25 return path.slice(1).reduce((sum, id, i) => sum + distance(path[i], id), 0);
26}
27function load(route) {
28 return route.reduce((sum, id) => sum + stops.get(id).demand, 0);

30function notation(route) {

65function normalize(routes) {
66 return routes
67 .filter((route) => route.length)
68 .map((route) => (route[0] > route.at(-1) ? [...route].reverse() : route))
69 .sort((a, b) => Math.min(...a) - Math.min(...b));
70}

81function simulate(route, print = false) {
82 let time = departure,
83 previous = 0,
84 waiting = 0,
85 service = 0,
86 violations = 0,
87 lateness = 0;
88 for (const id of route) {
89 const stop = stops.get(id),
90 arrival = time + distance(previous, id),
91 wait = stop.open === null ? 0 : Math.max(0, stop.open - arrival),
92 begin = arrival + wait;
93 const late = stop.close === null ? 0 : Math.max(0, arrival - stop.close);
94 time = begin + stop.service;
95 waiting += wait;
96 service += stop.service;
97 if (late > 0) violations++;
98 lateness += late;
99 if (print)
100 output.push(
101 `${id} arrive ${clock(arrival)} wait ${wait} begin ${clock(begin)} leave ${clock(time)} ${late ? "late " + late : "ok"}`,
102 );
103 previous = id;
104 }
105 time += distance(previous, 0);
106 if (print)
107 output.push(
108 `return ${clock(time)}, travel ${length(route)}, wait ${waiting}, service ${service}, violations ${violations}, lateness ${lateness}`,
109 );
110 return { time, violations, lateness };
111}
112function twoOpt(route, announce) {
113 let moves = 0;
114 while (true) {
115 let changed = false;
116 scan: for (let i = 0; i + 1 < route.length; i++)
117 for (let j = i + 1; j < route.length; j++) {
118 const before = route[i - 1] ?? 0,
119 after = route[j + 1] ?? 0;
120 const gain =
121 distance(before, route[i]) +
122 distance(route[j], after) -
123 distance(before, route[j]) -
124 distance(route[i], after);
125 if (gain > 0) {
126 if (announce) output.push(`reverse ${route[i]}..${route[j]} gain ${gain}`);
127 route.splice(i, j - i + 1, ...route.slice(i, j + 1).reverse());
128 moves++;
129 changed = true;
130 break scan;
131 }
132 }
133 if (!changed) return moves;
134 }

136function exact() {
137 const names = ids(),
138 memo = new Map();
139 function best(previous, mask) {
140 if (mask === 0) return distance(previous, 0);
141 const key = previous + ":" + mask;
142 if (memo.has(key)) return memo.get(key);
143 let value = Infinity;
144 for (let i = 0; i < names.length; i++)
145 if (mask & (1 << i))
146 value = Math.min(
147 value,
148 distance(previous, names[i]) + best(names[i], mask ^ (1 << i)),
149 );
150 memo.set(key, value);
151 return value;
152 }
153 const result = [];
154 let previous = 0,
155 mask = (1 << names.length) - 1;
156 while (mask) {
157 const target = best(previous, mask);
158 for (let i = 0; i < names.length; i++)
159 if (
160 mask & (1 << i) &&
161 distance(previous, names[i]) + best(names[i], mask ^ (1 << i)) === target
162 ) {
163 result.push(names[i]);
164 previous = names[i];
165 mask ^= 1 << i;
166 break;
167 }
168 }
169 return result;
170}
171function spanning(names, first) {
172 const tree = [first],
173 remaining = names.filter((id) => id !== first),
174 edges = [];
175 let total = 0;
176 while (remaining.length) {
177 let chosen = -1,
178 parent = -1,
179 weight = Infinity;
180 for (const id of remaining)
181 for (const from of tree)
182 if (distance(from, id) < weight) {
183 chosen = id;
184 parent = from;
185 weight = distance(from, id);
186 }
187 total += weight;
188 edges.push(location(parent) + "-" + chosen);
189 tree.push(chosen);
190 remaining.splice(remaining.indexOf(chosen), 1);
191 }
192 return { total, edges };
193}
194function improvePlan(command) {
195 const value = requirePlan(),
196 oldLength = planLength(value),
197 oldCount = value.routes.length;
198 let moves = 0;
199 function step() {
200 const routes = value.routes,
201 capacity = value.capacity;
202 if (command === "RELOCATE") {
203 for (let r = 0; r < routes.length; r++)
204 for (let p = 0; p < routes[r].length; p++)
205 for (let t = 0; t < routes.length; t++)
206 if (t !== r) {
207 const a = routes[r],
208 b = routes[t],
209 c = a[p];
210 if (load(b) + stops.get(c).demand > capacity) continue;
211 for (let j = 0; j <= b.length; j++) {
212 const before = a[p - 1] ?? 0,
213 after = a[p + 1] ?? 0,
214 u = b[j - 1] ?? 0,
215 v = b[j] ?? 0;
216 const gain =
217 distance(before, c) +
218 distance(c, after) -
219 distance(before, after) -
220 distance(u, c) -
221 distance(c, v) +
222 distance(u, v);
223 if (gain > 0) {
224 output.push(
225 `relocate ${c} from route ${r + 1} to route ${t + 1} after ${location(u)} gain ${gain}`,
226 );
227 a.splice(p, 1);
228 b.splice(j, 0, c);
229 value.routes = normalize(routes);
230 return true;
231 }
232 }
233 }
234 } else if (command === "EXCHANGE") {
235 for (let r = 0; r < routes.length; r++)
236 for (let t = r + 1; t < routes.length; t++)
237 for (let p = 0; p < routes[r].length; p++)
238 for (let j = 0; j < routes[t].length; j++) {
239 const a = routes[r],
240 b = routes[t],
241 c = a[p],
242 e = b[j];
243 if (
244 load(a) - stops.get(c).demand + stops.get(e).demand > capacity ||
245 load(b) - stops.get(e).demand + stops.get(c).demand > capacity
246 )
247 continue;
248 const nextA = [...a],
249 nextB = [...b];
250 nextA[p] = e;
251 nextB[j] = c;
252 const gain = length(a) + length(b) - length(nextA) - length(nextB);
253 if (gain > 0) {
254 output.push(
255 `exchange ${c} (route ${r + 1}) with ${e} (route ${t + 1}) gain ${gain}`,
256 );
257 routes[r] = nextA;
258 routes[t] = nextB;
259 value.routes = normalize(routes);
260 return true;
261 }
262 }
263 } else {
264 for (let r = 0; r < routes.length; r++)
265 for (let t = r + 1; t < routes.length; t++)
266 for (let i = 0; i <= routes[r].length; i++)
267 for (let j = 0; j <= routes[t].length; j++) {
268 const a = routes[r],
269 b = routes[t],
270 nextA = [...a.slice(0, i), ...b.slice(j)],
271 nextB = [...b.slice(0, j), ...a.slice(i)];
272 if (load(nextA) > capacity || load(nextB) > capacity) continue;
273 const x = a[i - 1] ?? 0,
274 y = b[j - 1] ?? 0,
275 afterA = a[i] ?? 0,
276 afterB = b[j] ?? 0;
277 const gain =
278 distance(x, afterA) +
279 distance(y, afterB) -
280 distance(x, afterB) -
281 distance(y, afterA);
282 if (gain > 0) {
283 output.push(
284 `cross route ${r + 1} after ${location(x)} with route ${t + 1} after ${location(y)} gain ${gain}`,
285 );
286 routes[r] = nextA;
287 routes[t] = nextB;
288 value.routes = normalize(routes);
289 return true;
290 }
291 }
292 }
293 return false;
294 }
295 while (step()) moves++;
296 output.push(
297 `${command.toLowerCase()}: moves ${moves}${command === "EXCHANGE" ? "" : `, routes ${oldCount} -> ${value.routes.length}`}, length ${oldLength} -> ${planLength(value)}`,
298 );
299}
300function execute(line) {
301 const [command, ...args] = line.split(" "),
302 a = Number(args[0]),
303 b = Number(args[1]);
304 if (command === "DEPOT") {
305 depot = { x: a, y: b };
306 output.push(`depot at (${a},${b})`);
307 return;
308 }
309 if (command === "START") {
310 departure = minute(args[0]);
311 output.push("start " + clock(departure));
312 return;
313 }
314 if (command === "STOP") {
315 if (stops.has(a)) fail(`stop ${a} already exists`);
316 const open = args.length > 5 ? minute(args[5]) : null,
317 close = args.length > 5 ? minute(args[6]) : null;
318 if (open !== null && open > close) fail("invalid window");
319 stops.set(a, {
320 x: b,
321 y: Number(args[2]),
322 demand: Number(args[3]),
323 service: Number(args[4]),
324 open,
325 close,
326 });
327 tour = null;
328 plan = null;
329 output.push(`stop ${a} added`);
330 return;
331 }
332 if (command === "REMOVE") {
333 if (!stops.delete(a)) fail("unknown stop " + a);
334 if (!stops.size) {
335 tour = null;
336 plan = null;
337 } else {
338 if (tour !== null) tour = tour.filter((id) => id !== a);
339 if (plan !== null)
340 plan.routes = normalize(plan.routes.map((route) => route.filter((id) => id !== a)));
341 }
342 output.push(`stop ${a} removed`);
343 return;
344 }
345 if (command === "LIST") {
346 output.push(
347 depot === null ? "depot none" : `depot (${depot.x},${depot.y})`,
348 "start " + clock(departure),
349 "stops " + stops.size,
350 );
351 for (const id of ids()) {
352 const stop = stops.get(id);
353 output.push(
354 `${id} at (${stop.x},${stop.y}) demand ${stop.demand} service ${stop.service} window ${stop.open === null ? "none" : clock(stop.open) + "-" + clock(stop.close)}`,
355 );
356 }
357 return;
358 }
359 if (command === "MATRIX") {
360 if (depot === null) fail("no depot");
361 const locations = [0, ...ids()],
362 rows = [
363 ["", ...locations.map(location)],
364 ...locations.map((id) => [
365 location(id),
366 ...locations.map((other) => String(distance(id, other))),
367 ]),
368 ];
369 const width = Math.max(...rows.flat().map((cell) => cell.length));
370 output.push(...rows.map((row) => row.map((cell) => cell.padStart(width)).join(" ")));
371 return;
372 }
373 if (["ORDER", "NEAREST", "INSERTION", "EXACT", "BOUND", "VRP", "SWEEP"].includes(command))
374 requireStops();
375 if (command === "ORDER") {
376 const order = args.map(Number);
377 for (const id of order) if (!stops.has(id)) fail("unknown stop " + id);
378 const seen = new Set();
379 for (const id of order) {
380 if (seen.has(id)) fail("duplicate stop " + id);
381 seen.add(id);
382 }
383 if (order.length !== stops.size)
384 fail(`order has ${order.length} stops, expected ${stops.size}`);
385 tour = order;
386 } else if (command === "NEAREST") {
387 const remaining = ids(),
388 route = [];
389 let previous = 0;
390 while (remaining.length) {
391 remaining.sort((a, b) => distance(previous, a) - distance(previous, b) || a - b);
392 previous = remaining.shift();
393 route.push(previous);
394 }
395 tour = route;
396 } else if (command === "INSERTION") {
397 const remaining = ids(),
398 route = [];
399 while (remaining.length) {
400 let best = Infinity,
401 chosen = -1,
402 position = -1;
403 for (const id of remaining)
404 for (let j = 0; j <= route.length; j++) {
405 const before = route[j - 1] ?? 0,
406 after = route[j] ?? 0,
407 increase =
408 distance(before, id) + distance(id, after) - distance(before, after);
409 if (increase < best) {
410 best = increase;
411 chosen = id;
412 position = j;
413 }
414 }
415 route.splice(position, 0, chosen);
416 remaining.splice(remaining.indexOf(chosen), 1);
417 }
418 tour = route;
419 } else if (command === "EXACT") {
420 if (stops.size > 10) fail(`too many stops for exact (${stops.size})`);
421 tour = exact();
422 }
423 if (["ORDER", "NEAREST", "INSERTION", "EXACT"].includes(command)) {
424 output.push(`${command.toLowerCase()}: ${notation(tour)} length ${length(tour)}`);
425 return;
426 }
427 if (command === "TOUR") {
428 const route = requireTour();
429 output.push(`tour: ${notation(route)} length ${length(route)}`);
430 return;
431 }
432 if (["TWOOPT", "OROPT", "SWAP"].includes(command)) {
433 const route = requireTour(),
434 old = length(route);
435 let moves = 0;
436 if (command === "TWOOPT") moves = twoOpt(route, true);
437 else
438 while (true) {
439 let changed = false;
440 if (command === "SWAP") {
441 scan: for (let i = 0; i + 1 < route.length; i++)
442 for (let j = i + 1; j < route.length; j++) {
443 const next = [...route];
444 [next[i], next[j]] = [next[j], next[i]];
445 const gain = length(route) - length(next);
446 if (gain > 0) {
447 output.push(`swap ${route[i]} and ${route[j]} gain ${gain}`);
448 route.splice(0, route.length, ...next);
449 moves++;
450 changed = true;
451 break scan;
452 }
453 }
454 } else {
455 scan: for (let size = 1; size <= 3; size++)
456 for (let i = 0; i + size <= route.length; i++) {
457 const segment = route.slice(i, i + size),
458 rest = [...route.slice(0, i), ...route.slice(i + size)];
459 for (let j = 0; j <= rest.length; j++)
460 if (j !== i) {
461 const next = [
462 ...rest.slice(0, j),
463 ...segment,
464 ...rest.slice(j),
465 ],
466 gain = length(route) - length(next);
467 if (gain > 0) {
468 output.push(
469 `move ${segment.join("-")} after ${location(rest[j - 1] ?? 0)} gain ${gain}`,
470 );
471 route.splice(0, route.length, ...next);
472 moves++;
473 changed = true;
474 break scan;
475 }
476 }
477 }
478 }
479 if (!changed) break;
480 }
481 output.push(`${command.toLowerCase()}: moves ${moves}, length ${old} -> ${length(route)}`);
482 return;
483 }
484 if (command === "TIMECHECK") {
485 const route = args.length ? requirePlan().routes[a - 1] : requireTour();
486 if (route === undefined) fail("no route " + a);
487 output.push(`timecheck ${args.length ? "route " + a : "tour"} depart ${clock(departure)}`);
488 simulate(route, true);
489 return;
490 }
491 if (command === "BOUND") {
492 const names = ids(),
493 mst = spanning(names, 0),
494 near = names.map((id) => distance(0, id)).sort((a, b) => a - b);
495 const oneTree =
496 names.length === 1
497 ? 2 * near[0]
498 : spanning(names, names[0]).total + near[0] + near[1],
499 bound = Math.max(mst.total, oneTree);
500 output.push(
501 `mst: ${mst.edges.join(" ")} (length ${mst.total})`,
502 `bound: mst ${mst.total}, one-tree ${oneTree}, best ${bound}`,
503 tour === null
504 ? "gap: no tour"
505 : `gap: tour ${length(tour)}, ${decimal(bound > 0 ? Math.floor((2000 * (length(tour) - bound) + bound) / (2 * bound)) : 0)}%`,
506 );
507 return;
508 }
509 if (command === "VRP" || command === "SWEEP" || command === "SPLIT") {
510 if (command === "SPLIT") requireTour();
511 checkCapacity(a);
512 let routes,
513 merges = 0;
514 if (command === "VRP") {
515 const limit = args.length > 1 ? b : null;
516 if (limit !== null)
517 for (const id of ids())
518 if (2 * distance(0, id) > limit) fail(`stop ${id} too far for limit ${limit}`);
519 routes = ids().map((id) => [id]);
520 const pairs = [];
521 for (const i of ids())
522 for (const j of ids())
523 if (i < j) {
524 const saving = distance(0, i) + distance(0, j) - distance(i, j);
525 if (saving > 0) pairs.push({ i, j, saving });
526 }
527 pairs.sort((a, b) => b.saving - a.saving || a.i - b.i || a.j - b.j);
528 for (const pair of pairs) {
529 const x = routes.findIndex((route) => route.includes(pair.i)),
530 y = routes.findIndex((route) => route.includes(pair.j));
531 const first = routes[x],
532 second = routes[y];
533 if (
534 x === y ||
535 ![first[0], first.at(-1)].includes(pair.i) ||
536 ![second[0], second.at(-1)].includes(pair.j) ||
537 load(first) + load(second) > a
538 )
539 continue;
540 const joined = [
541 ...(first.at(-1) === pair.i ? first : [...first].reverse()),
542 ...(second[0] === pair.j ? second : [...second].reverse()),
543 ];
544 if (limit !== null && length(joined) > limit) continue;
545 routes = routes.filter((_, i) => i !== x && i !== y);
546 routes.push(joined);
547 merges++;
548 }
549 } else if (command === "SPLIT") {
550 const route = requireTour(),
551 values = Array(route.length + 1).fill(Infinity),
552 previous = Array(route.length + 1).fill(-1);
553 values[0] = 0;
554 for (let j = 1; j <= route.length; j++)
555 for (let i = 0; i < j; i++) {
556 const segment = route.slice(i, j),
557 candidate = values[i] + length(segment);
558 if (load(segment) <= a && candidate < values[j]) {
559 values[j] = candidate;
560 previous[j] = i;
561 }
562 }
563 routes = [];
564 for (let j = route.length; j > 0; j = previous[j])
565 routes.push(route.slice(previous[j], j));
566 } else {
567 const ordered = ids().sort((a, b) => {
568 const x = stops.get(a),
569 y = stops.get(b),
570 dx = x.x - depot.x,
571 dy = x.y - depot.y,
572 ex = y.x - depot.x,
573 ey = y.y - depot.y;
574 const zeroA = dx === 0 && dy === 0,
575 zeroB = ex === 0 && ey === 0;
576 if (zeroA || zeroB) return zeroA && zeroB ? a - b : zeroA ? -1 : 1;
577 const halfA = dy > 0 || (dy === 0 && dx > 0) ? 0 : 1,
578 halfB = ey > 0 || (ey === 0 && ex > 0) ? 0 : 1,
579 cross = dx * ey - dy * ex;
580 return (
581 halfA - halfB ||
582 (cross === 0 ? distance(0, a) - distance(0, b) || a - b : cross > 0 ? -1 : 1)
583 );
584 });
585 routes = [[]];
586 for (const id of ordered) {
587 if (load(routes.at(-1)) + stops.get(id).demand > a) routes.push([]);
588 routes.at(-1).push(id);
589 }
590 }
591 plan = { capacity: a, routes: normalize(routes) };
592 output.push(
593 `${command.toLowerCase()}: capacity ${a}${command === "VRP" ? `, merges ${merges}` : ""}, routes ${plan.routes.length}, length ${planLength(plan)}`,
594 );
595 routeLines(plan);
596 return;
597 }
598 if (command === "ROUTES") {
599 const value = requirePlan();
600 output.push(
601 `plan: capacity ${value.capacity}, routes ${value.routes.length}, length ${planLength(value)}`,
602 );
603 routeLines(value);
604 return;
605 }
606 if (["RELOCATE", "EXCHANGE", "CROSS"].includes(command)) {
607 improvePlan(command);
608 return;
609 }
610 if (command === "ROUTEOPT") {
611 const value = requirePlan(),
612 old = planLength(value);
613 value.routes.forEach((route, i) => {
614 const before = length(route),
615 moves = twoOpt(route, false);
616 output.push(`route ${i + 1}: moves ${moves}, length ${before} -> ${length(route)}`);
617 });
618 value.routes = normalize(value.routes);
619 output.push(`routeopt: length ${old} -> ${planLength(value)}`);
620 return;
621 }
622 if (command === "REPORT") {
623 const all = [...stops.values()];
624 output.push(
625 `report: stops ${stops.size}, demand ${all.reduce((sum, stop) => sum + stop.demand, 0)}, service ${all.reduce((sum, stop) => sum + stop.service, 0)}, windows ${all.filter((stop) => stop.open !== null).length}`,
626 );
627 if (tour === null) output.push("tour: none");
628 else {
629 const result = simulate(tour);
630 output.push(
631 `tour: length ${length(tour)}, return ${clock(result.time)}, violations ${result.violations}, lateness ${result.lateness}`,
632 );
633 }
634 if (plan === null) output.push("plan: none");
635 else {
636 const results = plan.routes.map((route) => simulate(route)),
637 denominator = plan.routes.length * plan.capacity,
638 utilization = Math.floor(
639 (2000 * plan.routes.reduce((sum, route) => sum + load(route), 0) +
640 denominator) /
641 (2 * denominator),
642 );
643 output.push(
644 `plan: capacity ${plan.capacity}, routes ${plan.routes.length}, length ${planLength(plan)}, utilization ${decimal(utilization)}%, return ${clock(Math.max(...results.map((result) => result.time)))}, violations ${results.reduce((sum, result) => sum + result.violations, 0)}, lateness ${results.reduce((sum, result) => sum + result.lateness, 0)}`,
645 );
646 }
647 }
648}
649const input = fs.readFileSync(0, "utf8").split(/\r?\n/),
650 count = Number(input[0]);
651for (const line of input.slice(1, count + 1)) {
652 try {
653 execute(line);
654 } catch (error) {
655 output.push(error.message);
656 }
657}
658if (output.length) process.stdout.write(output.join("\n") + "\n");
659
