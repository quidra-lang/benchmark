function point(id) {
    return id === 0 ? depot : stops.get(id);
}
function distance(a, b) {
    const x = point(a),
        y = point(b);
    return Math.abs(x.x - y.x) + Math.abs(x.y - y.y);
}
function length(route) {
    const path = [0, ...route, 0];
    return path.slice(1).reduce((sum, id, i) => sum + distance(path[i], id), 0);
