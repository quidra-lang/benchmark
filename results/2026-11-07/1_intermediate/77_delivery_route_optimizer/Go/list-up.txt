func abs(n int) int {
	if n < 0 {
		return -n
	}
	return n
}
func (w *workbench) distance(a, b int) int {
	x, y := w.stops[a], w.stops[b]
	if a == 0 {
		x = *w.depot
	}
	if b == 0 {
		y = *w.depot
	}
	return abs(x.x-y.x) + abs(x.y-y.y)
}
func (w *workbench) length(p []int) int {
	total, last := 0, 0
	for _, id := range p {
		total += w.distance(last, id)
		last = id
	}
	return total + w.distance(last, 0)
