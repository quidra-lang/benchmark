1const std = @import("std");
2const Allocator = std.mem.Allocator;
3const Text = []const u8;
4const Int = i64;
5const Error = Allocator.Error || error{Invalid};
6var failure: Text = "";
7fn equal(a: Text, b: Text) bool {
8 return std.mem.eql(u8, a, b);
9}
10fn number(s: Text) Int {
11 return std.fmt.parseInt(Int, s, 10) catch unreachable;
12}
13fn natural(s: Text) usize {
14 return @intCast(number(s));
15}
16fn abs(n: Int) Int {
17 return if (n < 0) -n else n;
18}
19fn clock(s: Text) Int {
20 return number(s[0..2]) * 60 + number(s[3..]);
21}
22fn timeText(gpa: Allocator, t: Int) Allocator.Error!Text {
23 return std.fmt.allocPrint(gpa, "{d:0>2}:{d:0>2}", .{ @divTrunc(t, 60), @mod(t, 60) });
24}
25fn label(gpa: Allocator, id: usize) Allocator.Error!Text {
26 return if (id == 0) "D" else std.fmt.allocPrint(gpa, "{d}", .{id});
27}
28const Builder = struct {
29 gpa: Allocator,
30 bytes: std.ArrayList(u8) = .empty,
31 fn add(self: *Builder, s: Text) Allocator.Error!void {
32 try self.bytes.appendSlice(self.gpa, s);
33 }
34 fn print(self: *Builder, comptime format: Text, args: anytype) Allocator.Error!void {
35 try self.add(try std.fmt.allocPrint(self.gpa, format, args));
36 }
37 fn line(self: *Builder, comptime format: Text, args: anytype) Allocator.Error!void {
38 try self.print(format ++ "\n", args);
39 }
40};
41fn words(gpa: Allocator, s: Text) Allocator.Error![]Text {
42 var list: std.ArrayList(Text) = .empty;
43 var iterator = std.mem.tokenizeScalar(u8, s, ' ');
44 while (iterator.next()) |word| try list.append(gpa, word);
45 return list.items;
46}
47fn intLess(_: void, a: usize, b: usize) bool {
48 return a < b;
49}
50fn reverse(p: []usize) void {
51 std.mem.reverse(usize, p);
52}
53fn concatenate(gpa: Allocator, a: []const usize, b: []const usize) Allocator.Error![]usize {
54 const result = try gpa.alloc(usize, a.len + b.len);
55 @memcpy(result[0..a.len], a);
56 @memcpy(result[a.len..], b);
57 return result;
58}
59fn place(gpa: Allocator, p: []const usize, index: usize, values: []const usize) Allocator.Error![]usize {
60 const result = try gpa.alloc(usize, p.len + values.len);
61 @memcpy(result[0..index], p[0..index]);
62 @memcpy(result[index .. index + values.len], values);
63 @memcpy(result[index + values.len ..], p[index..]);
64 return result;
65}
66fn notation(gpa: Allocator, p: []const usize) Allocator.Error!Text {
67 var out: Builder = .{ .gpa = gpa };
68 try out.add("D");
69 for (p) |id| try out.print("-{d}", .{id});
70 try out.add("-D");
71 return out.bytes.items;
72}
73fn minimum(p: []const usize) usize {
74 var value = p[0];
75 for (p) |id| value = @min(value, id);
76 return value;
77}
78fn routeLess(_: void, a: []usize, b: []usize) bool {
79 return minimum(a) < minimum(b);
80}
81fn at(p: []const usize, index: usize) usize {
82 return if (index == 0 or index > p.len) 0 else p[index - 1];
83}
84const Stop = struct { x: Int = 0, y: Int = 0, demand: Int = 0, service: Int = 0, open: ?Int = null, close: Int = 0 };
85const Run = struct { returned: Int = 0, travel: Int = 0, wait: Int = 0, service: Int = 0, violations: Int = 0, lateness: Int = 0 };
86const Workbench = struct {
87 gpa: Allocator,
88 out: *Builder,
89 depot: ?Stop = null,
90 start: Int = 480,
91 stops: std.AutoHashMapUnmanaged(usize, Stop) = .empty,
92 tour: ?[]usize = null,
93 routes: ?std.ArrayList([]usize) = null,
94 capacity: Int = 0,
95 fn fail(self: *Workbench, comptime format: Text, args: anytype) Error {
96 failure = std.fmt.allocPrint(self.gpa, "error: " ++ format, args) catch return error.OutOfMemory;
97 return error.Invalid;
98 }
99 fn ids(self: *Workbench) Error![]usize {
100 var result: std.ArrayList(usize) = .empty;
101 var iterator = self.stops.keyIterator();
102 while (iterator.next()) |id| try result.append(self.gpa, id.*);
103 std.mem.sort(usize, result.items, {}, intLess);
104 return result.items;
105 }
106 fn distance(self: *Workbench, a: usize, b: usize) Int {
107 const p = if (a == 0) self.depot.? else self.stops.get(a).?;
108 const q = if (b == 0) self.depot.? else self.stops.get(b).?;
109 return abs(p.x - q.x) + abs(p.y - q.y);
110 }
111 fn length(self: *Workbench, p: []const usize) Int {
112 var total: Int = 0;
113 var last: usize = 0;
114 for (p) |id| {
115 total += self.distance(last, id);
116 last = id;
117 }
118 return total + self.distance(last, 0);
119 }
120 fn load(self: *Workbench, p: []const usize) Int {
121 var total: Int = 0;
122 for (p) |id| total += self.stops.get(id).?.demand;
123 return total;
124 }
125 fn total(self: *Workbench) Int {
126 var result: Int = 0;
127 for (self.routes.?.items) |p| result += self.length(p);
128 return result;
129 }
130 fn normalize(self: *Workbench) Error!void {
131 var result: std.ArrayList([]usize) = .empty;
132 for (self.routes.?.items) |p| {
133 if (p.len == 0) continue;
134 if (p[0] > p[p.len - 1]) reverse(p);
135 try result.append(self.gpa, p);
136 }
137 std.mem.sort([]usize, result.items, {}, routeLess);
138 self.routes = result;
139 }
140 fn routeLines(self: *Workbench) Error!void {
141 for (self.routes.?.items, 0..) |p, i| try self.out.line("route {d}: {s} load {d} length {d}", .{ i + 1, try notation(self.gpa, p), self.load(p), self.length(p) });
142 }
143 fn simulate(self: *Workbench, p: []const usize, display: bool) Error!Run {
144 var result: Run = .{ .travel = self.length(p) };
145 var t = self.start;
146 var last: usize = 0;
147 for (p) |id| {
148 const item = self.stops.get(id).?;
149 const arrival = t + self.distance(last, id);
150 const wait = if (item.open) |open| @max(0, open - arrival) else 0;
151 const late = if (item.open != null) @max(0, arrival - item.close) else 0;
152 const begin = arrival + wait;
153 t = begin + item.service;
154 last = id;
155 result.wait += wait;
156 result.service += item.service;
157 result.lateness += late;
158 if (late > 0) result.violations += 1;
159 if (display) try self.out.line("{d} arrive {s} wait {d} begin {s} leave {s} {s}", .{ id, try timeText(self.gpa, arrival), wait, try timeText(self.gpa, begin), try timeText(self.gpa, t), if (late > 0) try std.fmt.allocPrint(self.gpa, "late {d}", .{late}) else "ok" });
160 }
161 result.returned = t + self.distance(last, 0);
162 if (display) try self.out.line("return {s}, travel {d}, wait {d}, service {d}, violations {d}, lateness {d}", .{ try timeText(self.gpa, result.returned), result.travel, result.wait, result.service, result.violations, result.lateness });
163 return result;
164 }
165 fn twoopt(self: *Workbench, p: []usize, display: bool) Error!usize {
166 var moves: usize = 0;
167 while (true) {
168 var changed = false;
169 scan: for (0..p.len) |i| {
170 for (i + 1..p.len) |j| {
171 const a = at(p, i);
172 const b = at(p, j + 2);
173 const gain = self.distance(a, p[i]) + self.distance(p[j], b) - self.distance(a, p[j]) - self.distance(p[i], b);
174 if (gain > 0) {
175 if (display) try self.out.line("reverse {d}..{d} gain {d}", .{ p[i], p[j], gain });
176 reverse(p[i .. j + 1]);
177 moves += 1;
178 changed = true;
179 break :scan;
180 }
181 }
182 }
183 if (!changed) return moves;
184 }
185 }
186 fn baseCheck(self: *Workbench) Error!void {
187 if (self.depot == null) return self.fail("no depot", .{});
188 if (self.stops.count() == 0) return self.fail("no stops", .{});
189 }
190 fn capacityCheck(self: *Workbench, q: Int) Error!void {
191 for (try self.ids()) |id| if (self.stops.get(id).?.demand > q) return self.fail("stop {d} demand {d} exceeds capacity {d}", .{ id, self.stops.get(id).?.demand, q });
192 }
193 fn exact(self: *Workbench) Error![]usize {
194 const list = try self.ids();
195 var search: Exact = .{ .w = self, .ids = list };
196 const full = (@as(usize, 1) << @intCast(list.len)) - 1;
197 var mask = full;
198 var last: usize = 0;
199 var result: std.ArrayList(usize) = .empty;
200 while (mask != 0) {
201 const best = try search.cost(mask, last);
202 for (list, 0..) |id, i| {
203 const bit = @as(usize, 1) << @intCast(i);
204 if (mask & bit != 0 and self.distance(last, id) + try search.cost(mask ^ bit, id) == best) {
205 try result.append(self.gpa, id);
206 mask ^= bit;
207 last = id;
208 break;
209 }
210 }
211 }
212 return result.items;
213 }
214 fn bound(self: *Workbench) Error!void {
215 const list = try self.ids();
216 var tree: std.ArrayList(usize) = .empty;
217 try tree.append(self.gpa, 0);
218 var used: std.AutoHashMapUnmanaged(usize, void) = .empty;
219 var edges: Builder = .{ .gpa = self.gpa };
220 var mst: Int = 0;
221 for (0..list.len) |round| {
222 var best: Int = std.math.maxInt(Int);
223 var candidate: usize = 0;
224 var parent: usize = 0;
225 for (list) |id| {
226 if (used.contains(id)) continue;
227 for (tree.items) |p| {
228 const d = self.distance(p, id);
229 if (d < best) {
230 best = d;
231 candidate = id;
232 parent = p;
233 }
234 }
235 }
236 if (round > 0) try edges.add(" ");
237 try edges.print("{s}-{d}", .{ try label(self.gpa, parent), candidate });
238 mst += best;
239 try used.put(self.gpa, candidate, {});
240 try tree.append(self.gpa, candidate);
241 }
242 var one = 2 * self.distance(0, list[0]);
243 if (list.len > 1) {
244 const distances = try self.gpa.alloc(Int, list.len);
245 for (list, 0..) |id, i| distances[i] = self.distance(0, id);
246 std.mem.sort(Int, distances, {}, struct {
247 fn less(_: void, a: Int, b: Int) bool {
248 return a < b;
249 }
250 }.less);
251 one = distances[0] + distances[1];
252 used = .empty;
253 try used.put(self.gpa, list[0], {});
254 while (used.count() < list.len) {
255 var best: Int = std.math.maxInt(Int);
256 var candidate: usize = 0;
257 for (list) |a| {
258 if (!used.contains(a)) continue;
259 for (list) |b| {
260 if (!used.contains(b) and self.distance(a, b) < best) {
261 best = self.distance(a, b);
262 candidate = b;
263 }
264 }
265 }
266 one += best;
267 try used.put(self.gpa, candidate, {});
268 }
269 }
270 const best = @max(mst, one);
271 try self.out.line("mst: {s} (length {d})", .{ edges.bytes.items, mst });
272 try self.out.line("bound: mst {d}, one-tree {d}, best {d}", .{ mst, one, best });
273 if (self.tour) |p| {
274 const lengthValue = self.length(p);
275 const u = if (best > 0) @divTrunc(2000 * (lengthValue - best) + best, 2 * best) else 0;
276 try self.out.line("gap: tour {d}, {d}.{d}%", .{ lengthValue, @divTrunc(u, 10), @mod(u, 10) });
277 } else try self.out.line("gap: no tour", .{});
278 }
279 fn improve(self: *Workbench, command: Text) Error!void {
280 const initial = self.total();
281 const k0 = self.routes.?.items.len;
282 var moves: usize = 0;
283 while (true) {
284 var changed = false;
285 if (equal(command, "RELOCATE")) {
286 scan: for (self.routes.?.items, 0..) |a, r| {
287 for (a, 0..) |c, p| {
288 for (self.routes.?.items, 0..) |b, t| {
289 if (r == t or self.load(b) + self.stops.get(c).?.demand > self.capacity) continue;
290 for (0..b.len + 1) |j| {
291 const before = at(a, p);
292 const after = at(a, p + 2);
293 const u = at(b, j);
294 const v = at(b, j + 1);
295 const gain = self.distance(before, c) + self.distance(c, after) - self.distance(before, after) - self.distance(u, c) - self.distance(c, v) + self.distance(u, v);
296 if (gain > 0) {
297 try self.out.line("relocate {d} from route {d} to route {d} after {s} gain {d}", .{ c, r + 1, t + 1, try label(self.gpa, u), gain });
298 self.routes.?.items[r] = try concatenate(self.gpa, a[0..p], a[p + 1 ..]);
299 self.routes.?.items[t] = try place(self.gpa, b, j, &.{c});
300 changed = true;
301 break :scan;
302 }
303 }
304 }
305 }
306 }
307 } else if (equal(command, "EXCHANGE")) {
308 scan: for (self.routes.?.items, 0..) |a, r| {
309 for (r + 1..self.routes.?.items.len) |t| {
310 const b = self.routes.?.items[t];
311 for (a, 0..) |c, p| {
312 for (b, 0..) |e, j| {
313 if (self.load(a) - self.stops.get(c).?.demand + self.stops.get(e).?.demand > self.capacity or self.load(b) - self.stops.get(e).?.demand + self.stops.get(c).?.demand > self.capacity) continue;
314 const before = self.length(a) + self.length(b);
315 a[p] = e;
316 b[j] = c;
317 const gain = before - self.length(a) - self.length(b);
318 if (gain > 0) {
319 try self.out.line("exchange {d} (route {d}) with {d} (route {d}) gain {d}", .{ c, r + 1, e, t + 1, gain });
320 changed = true;
321 break :scan;
322 }
323 a[p] = c;
324 b[j] = e;
325 }
326 }
327 }
328 }
329 } else {
330 scan: for (self.routes.?.items, 0..) |a, r| {
331 for (r + 1..self.routes.?.items.len) |t| {
332 const b = self.routes.?.items[t];
333 for (0..a.len + 1) |i| {
334 for (0..b.len + 1) |j| {
335 if (self.load(a[0..i]) + self.load(b[j..]) > self.capacity or self.load(b[0..j]) + self.load(a[i..]) > self.capacity) continue;
336 const x = at(a, i);
337 const y = at(b, j);
338 const gain = self.distance(x, at(a, i + 1)) + self.distance(y, at(b, j + 1)) - self.distance(x, at(b, j + 1)) - self.distance(y, at(a, i + 1));
339 if (gain > 0) {
340 try self.out.line("cross route {d} after {s} with route {d} after {s} gain {d}", .{ r + 1, try label(self.gpa, x), t + 1, try label(self.gpa, y), gain });
341 self.routes.?.items[r] = try concatenate(self.gpa, a[0..i], b[j..]);
342 self.routes.?.items[t] = try concatenate(self.gpa, b[0..j], a[i..]);
343 changed = true;
344 break :scan;
345 }
346 }
347 }
348 }
349 }
350 }
351 if (!changed) break;
352 moves += 1;
353 try self.normalize();
354 }
355 const lower = if (equal(command, "EXCHANGE")) "exchange" else if (equal(command, "CROSS")) "cross" else "relocate";
356 if (equal(command, "EXCHANGE")) try self.out.line("{s}: moves {d}, length {d} -> {d}", .{ lower, moves, initial, self.total() }) else try self.out.line("{s}: moves {d}, routes {d} -> {d}, length {d} -> {d}", .{ lower, moves, k0, self.routes.?.items.len, initial, self.total() });
357 }
358 fn command(self: *Workbench, line: Text) Error!void {
359 const tokens = try words(self.gpa, line);
360 const commandName = tokens[0];
361 const args = tokens[1..];
362 if (equal(commandName, "DEPOT")) {
363 self.depot = .{ .x = number(args[0]), .y = number(args[1]) };
364 try self.out.line("depot at ({d},{d})", .{ self.depot.?.x, self.depot.?.y });
365 return;
366 }
367 if (equal(commandName, "START")) {
368 self.start = clock(args[0]);
369 try self.out.line("start {s}", .{try timeText(self.gpa, self.start)});
370 return;
371 }
372 if (equal(commandName, "STOP")) {
373 const id = natural(args[0]);
374 if (self.stops.contains(id)) return self.fail("stop {d} already exists", .{id});
375 var item: Stop = .{ .x = number(args[1]), .y = number(args[2]), .demand = number(args[3]), .service = number(args[4]) };
376 if (args.len > 5) {
377 item.open = clock(args[5]);
378 item.close = clock(args[6]);
379 if (item.open.? > item.close) return self.fail("invalid window", .{});
380 }
381 try self.stops.put(self.gpa, id, item);
382 self.tour = null;
383 self.routes = null;
384 try self.out.line("stop {d} added", .{id});
385 return;
386 }
387 if (equal(commandName, "REMOVE")) {
388 const id = natural(args[0]);
389 if (!self.stops.remove(id)) return self.fail("unknown stop {d}", .{id});
390 if (self.stops.count() == 0) {
391 self.tour = null;
392 self.routes = null;
393 } else {
394 if (self.tour) |p| self.tour = try without(self.gpa, p, id);
395 if (self.routes != null) {
396 for (self.routes.?.items) |*p| p.* = try without(self.gpa, p.*, id);
397 try self.normalize();
398 }
399 }
400 try self.out.line("stop {d} removed", .{id});
401 return;
402 }
403 if (equal(commandName, "LIST")) {
404 if (self.depot) |depot| {
405 try self.out.line("depot ({d},{d})", .{ depot.x, depot.y });
406 } else try self.out.line("depot none", .{});
407 try self.out.line("start {s}", .{try timeText(self.gpa, self.start)});
408 try self.out.line("stops {d}", .{self.stops.count()});
409 for (try self.ids()) |id| {
410 const item = self.stops.get(id).?;
411 const window = if (item.open) |open| try std.fmt.allocPrint(self.gpa, "{s}-{s}", .{ try timeText(self.gpa, open), try timeText(self.gpa, item.close) }) else "none";
412 try self.out.line("{d} at ({d},{d}) demand {d} service {d} window {s}", .{ id, item.x, item.y, item.demand, item.service, window });
413 }
414 return;
415 }
416 if (equal(commandName, "MATRIX")) {
417 if (self.depot == null) return self.fail("no depot", .{});
418 const list = try concatenate(self.gpa, &.{0}, try self.ids());
419 var width: usize = 1;
420 for (list) |a| {
421 width = @max(width, (try label(self.gpa, a)).len);
422 for (list) |b| width = @max(width, (try std.fmt.allocPrint(self.gpa, "{d}", .{self.distance(a, b)})).len);
423 }
424 try self.cell("", width);
425 for (list) |id| {
426 try self.out.add(" ");
427 try self.cell(try label(self.gpa, id), width);
428 }
429 try self.out.add("\n");
430 for (list) |a| {
431 try self.cell(try label(self.gpa, a), width);
432 for (list) |b| {
433 try self.out.add(" ");
434 try self.cell(try std.fmt.allocPrint(self.gpa, "{d}", .{self.distance(a, b)}), width);
435 }
436 try self.out.add("\n");
437 }
438 return;
439 }
440 if (equal(commandName, "REPORT")) {
441 try self.report();
442 return;
443 }
444 if (equal(commandName, "TOUR") or equal(commandName, "TWOOPT") or equal(commandName, "OROPT") or equal(commandName, "SWAP") or equal(commandName, "SPLIT") or equal(commandName, "TIMECHECK") and args.len == 0) {
445 if (self.tour == null) return self.fail("no tour", .{});
446 } else if (equal(commandName, "ROUTES") or equal(commandName, "RELOCATE") or equal(commandName, "EXCHANGE") or equal(commandName, "CROSS") or equal(commandName, "ROUTEOPT") or equal(commandName, "TIMECHECK")) {
447 if (self.routes == null) return self.fail("no plan", .{});
448 } else try self.baseCheck();
449 if (equal(commandName, "ORDER")) {
450 const p = try self.gpa.alloc(usize, args.len);
451 for (args, 0..) |arg, i| {
452 p[i] = natural(arg);
453 if (!self.stops.contains(p[i])) return self.fail("unknown stop {d}", .{p[i]});
454 }
455 var seen: std.AutoHashMapUnmanaged(usize, void) = .empty;
456 for (p) |id| {
457 if (seen.contains(id)) return self.fail("duplicate stop {d}", .{id});
458 try seen.put(self.gpa, id, {});
459 }
460 if (p.len != self.stops.count()) return self.fail("order has {d} stops, expected {d}", .{ p.len, self.stops.count() });
461 self.tour = p;
462 try self.out.line("order: {s} length {d}", .{ try notation(self.gpa, p), self.length(p) });
463 } else if (equal(commandName, "NEAREST") or equal(commandName, "INSERTION") or equal(commandName, "EXACT")) {
464 if (equal(commandName, "EXACT") and self.stops.count() > 10) return self.fail("too many stops for exact ({d})", .{self.stops.count()});
465 var p: []usize = try self.gpa.alloc(usize, 0);
466 if (equal(commandName, "EXACT")) {
467 p = try self.exact();
468 } else {
469 const list = try self.ids();
470 var used: std.AutoHashMapUnmanaged(usize, void) = .empty;
471 while (p.len < list.len) {
472 var best: Int = std.math.maxInt(Int);
473 var candidate: usize = 0;
474 var position: usize = 0;
475 for (list) |id| {
476 if (used.contains(id)) continue;
477 if (equal(commandName, "NEAREST")) {
478 const d = self.distance(at(p, p.len), id);
479 if (d < best) {
480 best = d;
481 candidate = id;
482 position = p.len;
483 }
484 } else {
485 for (0..p.len + 1) |j| {
486 const a = at(p, j);
487 const b = at(p, j + 1);
488 const increase = self.distance(a, id) + self.distance(id, b) - self.distance(a, b);
489 if (increase < best) {
490 best = increase;
491 candidate = id;
492 position = j;
493 }
494 }
495 }
496 }
497 p = try place(self.gpa, p, position, &.{candidate});
498 try used.put(self.gpa, candidate, {});
499 }
500 }
501 self.tour = p;
502 const lower = if (equal(commandName, "NEAREST")) "nearest" else if (equal(commandName, "INSERTION")) "insertion" else "exact";
503 try self.out.line("{s}: {s} length {d}", .{ lower, try notation(self.gpa, p), self.length(p) });
504 } else if (equal(commandName, "TOUR")) {
505 try self.out.line("tour: {s} length {d}", .{ try notation(self.gpa, self.tour.?), self.length(self.tour.?) });
506 } else if (equal(commandName, "TWOOPT")) {
507 const before = self.length(self.tour.?);
508 const moves = try self.twoopt(self.tour.?, true);
509 try self.out.line("twoopt: moves {d}, length {d} -> {d}", .{ moves, before, self.length(self.tour.?) });
510 } else if (equal(commandName, "SWAP") or equal(commandName, "OROPT")) {
511 try self.improveTour(commandName);
512 } else if (equal(commandName, "TIMECHECK")) {
513 if (args.len == 0) {
514 try self.out.line("timecheck tour depart {s}", .{try timeText(self.gpa, self.start)});
515 _ = try self.simulate(self.tour.?, true);
516 } else {
517 const r = natural(args[0]);
518 if (r > self.routes.?.items.len) return self.fail("no route {d}", .{r});
519 try self.out.line("timecheck route {d} depart {s}", .{ r, try timeText(self.gpa, self.start) });
520 _ = try self.simulate(self.routes.?.items[r - 1], true);
521 }
522 } else if (equal(commandName, "BOUND")) {
523 try self.bound();
524 } else if (equal(commandName, "VRP") or equal(commandName, "SPLIT") or equal(commandName, "SWEEP")) {
525 try self.construct(commandName, args);
526 } else if (equal(commandName, "ROUTES")) {
527 try self.out.line("plan: capacity {d}, routes {d}, length {d}", .{ self.capacity, self.routes.?.items.len, self.total() });
528 try self.routeLines();
529 } else if (equal(commandName, "RELOCATE") or equal(commandName, "EXCHANGE") or equal(commandName, "CROSS")) {
530 try self.improve(commandName);
531 } else if (equal(commandName, "ROUTEOPT")) {
532 const before = self.total();
533 for (self.routes.?.items, 0..) |p, i| {
534 const old = self.length(p);
535 const moves = try self.twoopt(p, false);
536 try self.out.line("route {d}: moves {d}, length {d} -> {d}", .{ i + 1, moves, old, self.length(p) });
537 }
538 try self.normalize();
539 try self.out.line("routeopt: length {d} -> {d}", .{ before, self.total() });
540 }
541 }
542 fn cell(self: *Workbench, text: Text, width: usize) Error!void {
543 for (text.len..width) |_| try self.out.add(" ");
544 try self.out.add(text);
545 }
546 fn improveTour(self: *Workbench, commandName: Text) Error!void {
547 const before = self.length(self.tour.?);
548 var moves: usize = 0;
549 while (true) {
550 const p = self.tour.?;
551 const old = self.length(p);
552 var changed = false;
553 if (equal(commandName, "SWAP")) {
554 scan: for (0..p.len) |i| {
555 for (i + 1..p.len) |j| {
556 const a = p[i];
557 const b = p[j];
558 p[i] = b;
559 p[j] = a;
560 const gain = old - self.length(p);
561 if (gain > 0) {
562 try self.out.line("swap {d} and {d} gain {d}", .{ a, b, gain });
563 changed = true;
564 break :scan;
565 }
566 p[i] = a;
567 p[j] = b;
568 }
569 }
570 } else {
571 scan: for (1..4) |size| {
572 if (size > p.len) continue;
573 for (0..p.len - size + 1) |i| {
574 const segment = p[i .. i + size];
575 const rest = try concatenate(self.gpa, p[0..i], p[i + size ..]);
576 const removed = self.distance(at(p, i), segment[0]) + self.distance(segment[size - 1], at(p, i + size + 1)) - self.distance(at(p, i), at(p, i + size + 1));
577 for (0..rest.len + 1) |j| {
578 if (j == i) continue;
579 const u = at(rest, j);
580 const v = at(rest, j + 1);
581 const gain = removed - self.distance(u, segment[0]) - self.distance(segment[size - 1], v) + self.distance(u, v);
582 if (gain > 0) {
583 var names: Builder = .{ .gpa = self.gpa };
584 for (segment, 0..) |id, k| {
585 if (k > 0) try names.add("-");
586 try names.print("{d}", .{id});
587 }
588 try self.out.line("move {s} after {s} gain {d}", .{ names.bytes.items, try label(self.gpa, u), gain });
589 self.tour = try place(self.gpa, rest, j, segment);
590 changed = true;
591 break :scan;
592 }
593 }
594 }
595 }
596 }
597 if (!changed) break;
598 moves += 1;
599 }
600 try self.out.line("{s}: moves {d}, length {d} -> {d}", .{ if (equal(commandName, "SWAP")) "swap" else "oropt", moves, before, self.length(self.tour.?) });
601 }
602 fn construct(self: *Workbench, commandName: Text, args: []const Text) Error!void {
603 const q = number(args[0]);
604 try self.capacityCheck(q);
605 const list = try self.ids();
606 const limit: ?Int = if (args.len > 1) number(args[1]) else null;
607 if (equal(commandName, "VRP")) {
608 if (limit) |boundValue| for (list) |id| {
609 if (2 * self.distance(0, id) > boundValue) return self.fail("stop {d} too far for limit {d}", .{ id, boundValue });
610 };
611 }
612 var routes: std.ArrayList([]usize) = .empty;
613 var merges: usize = 0;
614 if (equal(commandName, "VRP")) {
615 for (list) |id| try routes.append(self.gpa, try self.gpa.dupe(usize, &.{id}));
616 const Saving = struct { a: usize, b: usize, value: Int };
617 var savings: std.ArrayList(Saving) = .empty;
618 for (list, 0..) |a, i| {
619 for (list[i + 1 ..]) |b| {
620 const value = self.distance(0, a) + self.distance(0, b) - self.distance(a, b);
621 if (value > 0) try savings.append(self.gpa, .{ .a = a, .b = b, .value = value });
622 }
623 }
624 std.mem.sort(Saving, savings.items, {}, struct {
625 fn less(_: void, a: Saving, b: Saving) bool {
626 return a.value > b.value or a.value == b.value and (a.a < b.a or a.a == b.a and a.b < b.b);
627 }
628 }.less);
629 for (savings.items) |saving| {
630 var r: usize = 0;
631 var t: usize = 0;
632 for (routes.items, 0..) |p, k| {
633 for (p) |id| {
634 if (id == saving.a) r = k;
635 if (id == saving.b) t = k;
636 }
637 }
638 if (r == t) continue;
639 const originalA = routes.items[r];
640 const originalB = routes.items[t];
641 if ((originalA[0] != saving.a and originalA[originalA.len - 1] != saving.a) or (originalB[0] != saving.b and originalB[originalB.len - 1] != saving.b) or self.load(originalA) + self.load(originalB) > q) continue;
642 const a = try self.gpa.dupe(usize, originalA);
643 const b = try self.gpa.dupe(usize, originalB);
644 if (a[0] == saving.a) reverse(a);
645 if (b[b.len - 1] == saving.b) reverse(b);
646 const combined = try concatenate(self.gpa, a, b);
647 if (limit != null and self.length(combined) > limit.?) continue;
648 routes.items[r] = combined;
649 _ = routes.orderedRemove(t);
650 merges += 1;
651 }
652 } else if (equal(commandName, "SPLIT")) {
653 const p = self.tour.?;
654 const costs = try self.gpa.alloc(Int, p.len + 1);
655 const previous = try self.gpa.alloc(usize, p.len + 1);
656 costs[0] = 0;
657 for (1..p.len + 1) |j| {
658 costs[j] = std.math.maxInt(Int);
659 for (0..j) |i| {
660 if (self.load(p[i..j]) > q) continue;
661 const value = costs[i] + self.length(p[i..j]);
662 if (value < costs[j]) {
663 costs[j] = value;
664 previous[j] = i;
665 }
666 }
667 }
668 var j = p.len;
669 while (j > 0) {
670 const i = previous[j];
671 try routes.append(self.gpa, try self.gpa.dupe(usize, p[i..j]));
672 j = i;
673 }
674 } else {
675 std.mem.sort(usize, list, self, angleLess);
676 var current: std.ArrayList(usize) = .empty;
677 for (list) |id| {
678 if (self.load(current.items) + self.stops.get(id).?.demand > q) {
679 try routes.append(self.gpa, current.items);
680 current = .empty;
681 }
682 try current.append(self.gpa, id);
683 }
684 try routes.append(self.gpa, current.items);
685 }
686 self.routes = routes;
687 self.capacity = q;
688 try self.normalize();
689 if (equal(commandName, "VRP")) try self.out.line("vrp: capacity {d}, merges {d}, routes {d}, length {d}", .{ q, merges, self.routes.?.items.len, self.total() }) else try self.out.line("{s}: capacity {d}, routes {d}, length {d}", .{ if (equal(commandName, "SPLIT")) "split" else "sweep", q, self.routes.?.items.len, self.total() });
690 try self.routeLines();
691 }
692 fn report(self: *Workbench) Error!void {
693 var demand: Int = 0;
694 var service: Int = 0;
695 var windows: usize = 0;
696 var iterator = self.stops.valueIterator();
697 while (iterator.next()) |item| {
698 demand += item.demand;
699 service += item.service;
700 if (item.open != null) windows += 1;
701 }
702 try self.out.line("report: stops {d}, demand {d}, service {d}, windows {d}", .{ self.stops.count(), demand, service, windows });
703 if (self.tour) |p| {
704 const r = try self.simulate(p, false);
705 try self.out.line("tour: length {d}, return {s}, violations {d}, lateness {d}", .{ r.travel, try timeText(self.gpa, r.returned), r.violations, r.lateness });
706 } else try self.out.line("tour: none", .{});
707 if (self.routes) |routes| {
708 var returned: Int = 0;
709 var violations: Int = 0;
710 var lateness: Int = 0;
711 for (routes.items) |p| {
712 const r = try self.simulate(p, false);
713 returned = @max(returned, r.returned);
714 violations += r.violations;
715 lateness += r.lateness;
716 }
717 const denominator = @as(Int, @intCast(routes.items.len)) * self.capacity;
718 const u = @divTrunc(2000 * demand + denominator, 2 * denominator);
719 try self.out.line("plan: capacity {d}, routes {d}, length {d}, utilization {d}.{d}%, return {s}, violations {d}, lateness {d}", .{ self.capacity, routes.items.len, self.total(), @divTrunc(u, 10), @mod(u, 10), try timeText(self.gpa, returned), violations, lateness });
720 } else try self.out.line("plan: none", .{});
721 }
722};
723fn angleLess(w: *Workbench, a: usize, b: usize) bool {
724 const p = w.stops.get(a).?;
725 const q = w.stops.get(b).?;
726 const depot = w.depot.?;
727 const ax = p.x - depot.x;
728 const ay = p.y - depot.y;
729 const bx = q.x - depot.x;
730 const by = q.y - depot.y;
731 const az = ax == 0 and ay == 0;
732 const bz = bx == 0 and by == 0;
733 if (az != bz) return az;
734 if (az) return a < b;
735 const ah = ay > 0 or ay == 0 and ax > 0;
736 const bh = by > 0 or by == 0 and bx > 0;
737 if (ah != bh) return ah;
738 const cross = ax * by - ay * bx;
739 if (cross != 0) return cross > 0;
740 const da = abs(ax) + abs(ay);
741 const db = abs(bx) + abs(by);
742 return da < db or da == db and a < b;
743}
744fn without(gpa: Allocator, p: []const usize, id: usize) Allocator.Error![]usize {
745 var result: std.ArrayList(usize) = .empty;
746 for (p) |value| if (value != id) try result.append(gpa, value);
747 return result.items;
748}
749const Exact = struct {
750 w: *Workbench,
751 ids: []const usize,
752 memo: std.AutoHashMapUnmanaged(struct { mask: usize, last: usize }, Int) = .empty,
753 fn cost(self: *Exact, mask: usize, last: usize) Error!Int {
754 if (mask == 0) return self.w.distance(last, 0);
755 const key = .{ .mask = mask, .last = last };
756 if (self.memo.get(key)) |value| return value;
757 var best: Int = std.math.maxInt(Int);
758 for (self.ids, 0..) |id, i| {
759 const bit = @as(usize, 1) << @intCast(i);
760 if (mask & bit != 0) best = @min(best, self.w.distance(last, id) + try self.cost(mask ^ bit, id));
761 }
762 try self.memo.put(self.w.gpa, key, best);
763 return best;
764 }
765};
766pub fn main(init: std.process.Init) !void {
767 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
768 defer arena.deinit();
769 const gpa = arena.allocator();
770 var inputBuffer: [8192]u8 = undefined;
771 var input = std.Io.File.stdin().readerStreaming(init.io, &inputBuffer);
772 const text = try input.interface.allocRemaining(gpa, .unlimited);
773 var lines = std.mem.splitScalar(u8, text, '\n');
774 const count = natural(std.mem.trimEnd(u8, lines.next().?, "\r"));
775 var out: Builder = .{ .gpa = gpa };
776 var w: Workbench = .{ .gpa = gpa, .out = &out };
777 for (0..count) |_| {
778 const line = std.mem.trimEnd(u8, lines.next().?, "\r");
779 w.command(line) catch |err| {
780 if (err != error.Invalid) return err;
781 try out.line("{s}", .{failure});
782 };
783 }
784 var outputBuffer: [8192]u8 = undefined;
785 var output = std.Io.File.stdout().writerStreaming(init.io, &outputBuffer);
786 try output.interface.writeAll(out.bytes.items);
787 try output.interface.flush();
788}
