int absolute(int value)
    return if value < 0 then -value else value
    Stop get_stop(nat id)
        match stop_table.get(id)
            Stop value
                return value
    Point coordinates(nat id)
        if id == 0
            return depot
        Stop value := get_stop(id)
        return Point(value.x, value.y)

    int distance(nat first, nat second)
        Point a := coordinates(first)
        Point b := coordinates(second)
        return absolute(a.x - b.x) + absolute(a.y - b.y)
    int path_length(nat[] path)
        result := 0
        nat previous := 0
        for id := path[]
            result += distance(previous, id)
            previous = id
        return result + distance(previous, 0)
