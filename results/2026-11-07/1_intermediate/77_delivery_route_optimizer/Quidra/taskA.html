231 nat[] exact_tour()
232 nat[] ids := stop_ids()
233 n := #ids
234 powers := [for i := (0:<nat(n))[] then nat(2) ^ i]
235 states := nat(2) ^ n
236 full := states - 1
237 int[][] costs := array(states, fill = array(n, fill = 0))
238 for: i < n
239 costs[0][i] = distance(ids[i], 0)
240 for: 1 <= mask < full
241 for: i < n
242 if mask / powers[i] % 2 == 1
243 continue
244 set_cost := false
245 best := 0
246 for: j < n
247 if mask / powers[j] % 2 == 1
248 int value := distance(ids[i], ids[j]) + costs[mask - powers[j]][j]
249 if not set_cost or value < best
250 best = value
251 set_cost = true
252 costs[mask][i] = best
253 cost := 0
254 cost_set := false
255 for: i < n
256 int value := distance(0, ids[i]) + costs[full - powers[i]][i]
257 if not cost_set or value < cost
258 cost = value
259 cost_set = true
260 nat[] path := []
261 nat current := 0
262 mask := full
263 while mask > 0
264 for: i < n
265 if mask / powers[i] % 2 == 1 and distance(current, ids[i]) + costs[mask - powers[i]][i] == cost
266 mask -= powers[i]
267 cost = costs[mask][i]
268 current = ids[i]
269 path = path.append(current)
270 break
271 return path

291 bool sweep_less(nat first, nat second)
292 Point a := coordinates(first)
293 Point b := coordinates(second)
294 int x := a.x - depot.x
295 int y := a.y - depot.y
296 int u := b.x - depot.x
297 int v := b.y - depot.y
298 az := x == 0 and y == 0
299 bz := u == 0 and v == 0
300 if az != bz
301 return az
302 if az
303 return first < second
304 ah := y > 0 or (y == 0 and x > 0)
305 bh := v > 0 or (v == 0 and u > 0)
306 if ah != bh
307 return ah
308 cross := x * v - y * u
309 if cross != 0
310 return cross > 0
311 da := absolute(x) + absolute(y)
312 db := absolute(u) + absolute(v)
313 return if da != db then da < db else first < second

324 void | error .savings(int q, int limit)
325 nat[] ids := stop_ids()
326 if limit >= 0
327 for id := ids[]
328 if 2 * distance(0, id) > limit
329 return error("stop {id} too far for limit {limit}")
330 Saving[] pairs := []
331 for: first := ids[i]
332 for second := ids[i + 1:][]
333 int value := distance(0, first) + distance(0, second) - distance(first, second)
334 if value > 0
335 nat at := 0
336 while at < #pairs and pairs[at].value >= value
337 at += 1
338 pairs = pairs[:<at].append(Saving(first, second, value)).concat(pairs[at:])
339 nat[][] paths := [for id := ids[] then [id]]
340 nat merges := 0
341 for pair := pairs[]
342 nat r := 0
343 nat t := 0
344 for: path := paths[i]
345 if contains_id(path, pair.first)
346 r = i
347 if contains_id(path, pair.second)
348 t = i
349 if r == t
350 continue
351 a := paths[r]
352 b := paths[t]
353 if not (pair.first == a[0] or pair.first == a[-1]) or not (pair.second == b[0] or pair.second == b[-1])
354 continue
355 if path_load(a) + path_load(b) > q
356 continue
357 if a[0] == pair.first
358 a = a[::-1]
359 if b[-1] == pair.second
360 b = b[::-1]
361 nat[] merged := a.concat(b)
362 if limit >= 0 and path_length(merged) > limit
363 continue
364 paths = [for i := paths.indices()[] if i != r and i != t then paths[i]].append(merged)
365 merges += 1
366 capacity = q
367 routes = paths
