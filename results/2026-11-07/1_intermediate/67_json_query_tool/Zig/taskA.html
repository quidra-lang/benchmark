1const std = @import("std");
2const Allocator = std.mem.Allocator;
3const Text = []const u8;
4const Error = Allocator.Error || error{Invalid};
5const Kind = enum { object, array, string, number, boolean, null };
6const Failure = struct { reason: Text, position: usize = 0, json: bool = false };
7var failure: Failure = undefined;
8fn equal(a: Text, b: Text) bool {
9 return std.mem.eql(u8, a, b);
10}
11fn starts(s: Text, p: Text) bool {
12 return std.mem.startsWith(u8, s, p);
13}
14fn bad(gpa: Allocator, comptime pattern: Text, args: anytype) Error {
15 failure = .{ .reason = std.fmt.allocPrint(gpa, pattern, args) catch return error.OutOfMemory };
16 return error.Invalid;
17}
18fn jsonFail(offset: usize, reason: Text) Error {
19 failure = .{ .reason = reason, .position = offset, .json = true };
20 return error.Invalid;

22const Builder = struct {
23 gpa: Allocator,
24 bytes: std.ArrayList(u8) = .empty,
25 fn add(self: *Builder, s: Text) Allocator.Error!void {
26 try self.bytes.appendSlice(self.gpa, s);
27 }
28 fn byte(self: *Builder, c: u8) Allocator.Error!void {
29 try self.bytes.append(self.gpa, c);
30 }
31 fn print(self: *Builder, comptime pattern: Text, args: anytype) Allocator.Error!void {
32 try self.add(try std.fmt.allocPrint(self.gpa, pattern, args));
33 }
34 fn line(self: *Builder, comptime pattern: Text, args: anytype) Allocator.Error!void {
35 try self.print(pattern ++ "\n", args);
36 }

38 return self.bytes.items;

41fn join(gpa: Allocator, parts: []const Text, separator: Text) Allocator.Error!Text {
42 var out: Builder = .{ .gpa = gpa };
43 for (parts, 0..) |p, i| {
44 if (i > 0) try out.add(separator);
45 try out.add(p);
46 }
47 return out.result();
48}
49fn split(gpa: Allocator, text: Text, separator: u8, limit: usize) Allocator.Error![]Text {
50 var parts: std.ArrayList(Text) = .empty;
51 var start: usize = 0;
52 for (text, 0..) |c, i| {
53 if (c == separator and (limit == 0 or parts.items.len + 1 < limit)) {
54 try parts.append(gpa, text[start..i]);
55 start = i + 1;
56 }
57 }
58 try parts.append(gpa, text[start..]);
59 return parts.items;
60}
61fn identifier(s: Text) bool {
62 if (s.len == 0 or (!std.ascii.isAlphabetic(s[0]) and s[0] != '_')) return false;
63 for (s) |c| if (!std.ascii.isAlphanumeric(c) and c != '_') return false;
64 return true;
65}
66fn integerNotation(s: Text) bool {
67 if (equal(s, "0")) return true;
68 const ds = if (starts(s, "-")) s[1..] else s;
69 if (ds.len == 0 or ds.len > 9 or ds[0] == '0') return false;
70 for (ds) |c| if (!std.ascii.isDigit(c)) return false;
71 return true;
72}
73fn quote(gpa: Allocator, s: Text, mark: u8) Allocator.Error!Text {
74 var out: Builder = .{ .gpa = gpa };
75 try out.byte(mark);
76 for (s) |c| {
77 if (c == mark or c == '\\') {
78 try out.byte('\\');
79 try out.byte(c);
80 } else switch (c) {
81 8 => try out.add("\\b"),
82 12 => try out.add("\\f"),
83 10 => try out.add("\\n"),
84 13 => try out.add("\\r"),
85 9 => try out.add("\\t"),
86 else => if (c < 32 or c == 127) {
87 try out.print("\\u{x:0>4}", .{c});
88 } else try out.byte(c),
89 }
90 }
91 try out.byte(mark);
92 return out.result();
93}
94const Value = struct {
95 kind: Kind,
96 text: Text = "",
97 number: i64 = 0,
98 boolean: bool = false,
99 keys: std.ArrayList(Text) = .empty,
100 items: std.ArrayList(*Value) = .empty,
101 fn find(self: *const Value, key: Text) ?usize {
102 for (self.keys.items, 0..) |k, i| if (equal(k, key)) return i;
103 return null;
104 }
105 fn copy(self: *const Value, gpa: Allocator) Error!*Value {
106 const result = try make(gpa, self.kind);
107 result.text = self.text;
108 result.number = self.number;
109 result.boolean = self.boolean;
110 try result.keys.appendSlice(gpa, self.keys.items);
111 for (self.items.items) |item| try result.items.append(gpa, try item.copy(gpa));
112 return result;
113 }
114 fn size(self: *const Value) [2]usize {
115 var count: usize = 1;
116 var depth: usize = 1;
117 for (self.items.items) |item| {
118 const child = item.size();
119 count += child[0];
120 depth = @max(depth, child[1] + 1);
121 }
122 return .{ count, depth };
123 }

125fn make(gpa: Allocator, kind: Kind) Allocator.Error!*Value {
126 const v = try gpa.create(Value);
127 v.* = .{ .kind = kind };
128 return v;
129}
130fn compact(gpa: Allocator, v: *Value) Error!Text {
131 switch (v.kind) {
132 .null => return "null",
133 .boolean => return if (v.boolean) "true" else "false",
134 .number => return std.fmt.allocPrint(gpa, "{d}", .{v.number}),
135 .string => return quote(gpa, v.text, '"'),
136 else => {},
137 }
138 var out: Builder = .{ .gpa = gpa };
139 try out.byte(if (v.kind == .array) '[' else '{');
140 for (v.items.items, 0..) |item, i| {
141 if (i > 0) try out.byte(',');
142 if (v.kind == .object) {
143 try out.add(try quote(gpa, v.keys.items[i], '"'));
144 try out.byte(':');
145 }
146 try out.add(try compact(gpa, item));
147 }
148 try out.byte(if (v.kind == .array) ']' else '}');
149 return out.result();
150}
151fn pretty(gpa: Allocator, v: *Value, indent: Text) Error!Text {
152 if (v.items.items.len == 0) return compact(gpa, v);
153 const inner = try std.fmt.allocPrint(gpa, "{s} ", .{indent});
154 var out: Builder = .{ .gpa = gpa };
155 try out.byte(if (v.kind == .array) '[' else '{');
156 try out.byte('\n');
157 for (v.items.items, 0..) |item, i| {
158 if (i > 0) try out.add(",\n");
159 try out.add(inner);
160 if (v.kind == .object) {
161 try out.add(try quote(gpa, v.keys.items[i], '"'));
162 try out.add(": ");
163 }
164 try out.add(try pretty(gpa, item, inner));
165 }
166 try out.byte('\n');
167 try out.add(indent);
168 try out.byte(if (v.kind == .array) ']' else '}');
169 return out.result();
170}
171fn same(a: *Value, b: *Value) bool {
172 if (a.kind != b.kind) return false;
173 switch (a.kind) {
174 .null => return true,
175 .string => return equal(a.text, b.text),
176 .number => return a.number == b.number,
177 .boolean => return a.boolean == b.boolean,
178 else => {},
179 }
180 if (a.items.items.len != b.items.items.len) return false;
181 for (a.items.items, 0..) |item, i| {
182 const j = if (a.kind == .object) b.find(a.keys.items[i]) orelse return false else i;
183 if (!same(item, b.items.items[j])) return false;
184 }
185 return true;
186}
187const Reader = struct {
188 gpa: Allocator,
189 text: Text,
190 pos: usize = 0,
191 fn peek(self: *const Reader) u8 {
192 return if (self.pos < self.text.len) self.text[self.pos] else 0;
193 }
194 fn skip(self: *Reader) void {
195 while (self.peek() == ' ' or self.peek() == '\n') self.pos += 1;
196 }
197 fn require(self: *Reader, c: u8, reason: Text) Error!void {
198 if (self.pos == self.text.len) return jsonFail(self.pos, "unexpected end of input");
199 if (self.peek() != c) return jsonFail(self.pos, reason);
200 }
201 fn readString(self: *Reader, mark: u8) Error!Text {
202 self.pos += 1;
203 var out: Builder = .{ .gpa = self.gpa };
204 while (true) {
205 const c = self.peek();
206 if (c == 0 or c == '\n') return jsonFail(self.pos, "unterminated string");
207 if (c == mark) {
208 self.pos += 1;
209 return out.result();
210 }
211 if (c != '\\') {
212 try out.byte(c);
213 self.pos += 1;
214 continue;
215 }
216 const start = self.pos;
217 self.pos += 1;
218 const e = self.peek();
219 if (e == 0 or e == '\n') return jsonFail(self.pos, "unterminated string");
220 if (e == 'u') {
221 if (start + 6 > self.text.len) return jsonFail(start, "bad unicode escape");
222 const digits = self.text[start + 2 .. start + 6];
223 for (digits) |d| if (!std.ascii.isHex(d)) return jsonFail(start, "bad unicode escape");
224 const n = std.fmt.parseInt(u16, digits, 16) catch unreachable;
225 if (n > 127) return jsonFail(start, "non-ascii escape");
226 try out.byte(@intCast(n));
227 self.pos = start + 6;
228 } else {
229 const decoded: u8 = if (e == mark) mark else switch (e) {
230 '\\' => '\\',
231 '/' => '/',
232 'b' => 8,
233 'f' => 12,
234 'n' => 10,
235 'r' => 13,
236 't' => 9,
237 else => return jsonFail(start, try std.fmt.allocPrint(self.gpa, "invalid escape '\\{c}'", .{e})),
238 };
239 try out.byte(decoded);
240 self.pos += 1;
241 }
242 }
243 }
244 fn containerEnd(self: *Reader, close: u8, reason: Text) Error!bool {
245 self.skip();
246 if (self.peek() == close) {
247 self.pos += 1;
248 return true;
249 }
250 try self.require(',', reason);
251 self.pos += 1;
252 self.skip();
253 if (self.peek() == close) return jsonFail(self.pos, "trailing comma");
254 return false;
255 }
256 fn readValue(self: *Reader) Error!*Value {
257 self.skip();
258 const c = self.peek();
259 if (c == 0) return jsonFail(self.pos, "unexpected end of input");
260 if (c == '"') {
261 const v = try make(self.gpa, .string);
262 v.text = try self.readString('"');
263 return v;
264 }
265 if (c == '{' or c == '[') {
266 const kind: Kind = if (c == '{') .object else .array;
267 const close: u8 = if (c == '{') '}' else ']';
268 const reason: Text = if (c == '{') "expected ',' or '}'" else "expected ',' or ']'";
269 const v = try make(self.gpa, kind);
270 self.pos += 1;
271 self.skip();
272 if (self.peek() == close) {
273 self.pos += 1;
274 return v;
275 }
276 while (true) {
277 if (kind == .object) {
278 self.skip();
279 try self.require('"', "expected key");
280 const start = self.pos;
281 const key = try self.readString('"');
282 if (v.find(key) != null) return jsonFail(start, try std.fmt.allocPrint(self.gpa, "duplicate key {s}", .{try quote(self.gpa, key, '"')}));
283 self.skip();
284 try self.require(':', "expected ':'");
285 self.pos += 1;
286 try v.keys.append(self.gpa, key);
287 }
288 try v.items.append(self.gpa, try self.readValue());
289 if (try self.containerEnd(close, reason)) return v;
290 }
291 }
292 if (c == '-' or std.ascii.isDigit(c)) {
293 const start = self.pos;
294 const negative = c == '-';
295 if (negative) self.pos += 1;
296 const first = self.pos;
297 while (std.ascii.isDigit(self.peek())) self.pos += 1;
298 const ds = self.text[first..self.pos];
299 if (ds.len == 0) return jsonFail(start, "invalid number");
300 if (ds.len > 1 and ds[0] == '0') return jsonFail(first, "leading zero");
301 if (self.peek() == '.') return jsonFail(self.pos, "fraction not allowed");
302 if (self.peek() == 'e' or self.peek() == 'E') return jsonFail(self.pos, "exponent not allowed");
303 if (ds.len >= 10) return jsonFail(start, "number out of range");
304 const v = try make(self.gpa, .number);
305 const n = std.fmt.parseInt(i64, ds, 10) catch unreachable;
306 v.number = if (negative) -n else n;
307 return v;
308 }
309 if (std.ascii.isAlphabetic(c)) {
310 const start = self.pos;
311 while (std.ascii.isAlphabetic(self.peek())) self.pos += 1;
312 const word = self.text[start..self.pos];
313 if (equal(word, "null")) return make(self.gpa, .null);
314 if (equal(word, "true") or equal(word, "false")) {
315 const v = try make(self.gpa, .boolean);
316 v.boolean = equal(word, "true");
317 return v;
318 }
319 return jsonFail(start, try std.fmt.allocPrint(self.gpa, "invalid literal '{s}'", .{word}));
320 }
321 return jsonFail(self.pos, try std.fmt.allocPrint(self.gpa, "unexpected character '{c}'", .{c}));
322 }
323 fn parse(self: *Reader) Error!*Value {
324 const v = try self.readValue();
325 self.skip();
326 if (self.pos < self.text.len) return jsonFail(self.pos, "trailing characters");
327 return v;
328 }
329};
330fn parseJson(gpa: Allocator, text: Text) Error!*Value {
331 var p: Reader = .{ .gpa = gpa, .text = text };
332 return p.parse();

334const SelectorKind = enum { key, index, all, slice };
335const Selector = struct { kind: SelectorKind, key: Text = "", index: i64 = 0, start: ?i64 = null, end: ?i64 = null, descendant: bool = false };
336const PathReader = struct {
337 source: Reader,
338 reason: Text,
339 fn invalid(self: *PathReader) Error {
340 return bad(self.source.gpa, "{s}", .{self.reason});
341 }
342 fn readIdentifier(self: *PathReader) Error!Text {
343 const start = self.source.pos;
344 const c = self.source.peek();
345 if (!std.ascii.isAlphabetic(c) and c != '_') return self.invalid();
346 while (std.ascii.isAlphanumeric(self.source.peek()) or self.source.peek() == '_') self.source.pos += 1;
347 return self.source.text[start..self.source.pos];
348 }
349 fn quoted(self: *PathReader) Error!Text {
350 return self.source.readString('\'') catch |err| {
351 if (err != error.Invalid) return err;
352 return self.invalid();
353 };
354 }
355 fn selection(self: *PathReader) Error!Selector {
356 if (self.source.peek() == '\'') {
357 const key = try self.quoted();
358 if (self.source.peek() != ']') return self.invalid();
359 self.source.pos += 1;
360 return .{ .kind = .key, .key = key };
361 }
362 const after = self.source.text[self.source.pos..];
363 const end = std.mem.findScalar(u8, after, ']') orelse return self.invalid();
364 const text = after[0..end];
365 self.source.pos += end + 1;
366 if (equal(text, "*")) return .{ .kind = .all };
367 if (integerNotation(text)) return .{ .kind = .index, .index = std.fmt.parseInt(i64, text, 10) catch unreachable };
368 const colon = std.mem.findScalar(u8, text, ':') orelse return self.invalid();
369 const a = text[0..colon];
370 const b = text[colon + 1 ..];
371 if ((a.len > 0 and !integerNotation(a)) or (b.len > 0 and !integerNotation(b))) return self.invalid();
372 return .{ .kind = .slice, .start = if (a.len == 0) null else std.fmt.parseInt(i64, a, 10) catch unreachable, .end = if (b.len == 0) null else std.fmt.parseInt(i64, b, 10) catch unreachable };
373 }
374};
375fn parsePath(gpa: Allocator, text: Text, field: bool) Error![]Selector {
376 var p: PathReader = .{ .source = .{ .gpa = gpa, .text = text }, .reason = if (field) "bad field" else "bad path" };
377 if (p.source.peek() != (if (field) @as(u8, '@') else '$')) return p.invalid();
378 p.source.pos += 1;
379 var out: std.ArrayList(Selector) = .empty;
380 while (p.source.pos < text.len) {
381 var s: Selector = undefined;
382 if (p.source.peek() == '[') {
383 p.source.pos += 1;
384 s = try p.selection();
385 } else {
386 if (p.source.peek() != '.') return p.invalid();
387 p.source.pos += 1;
388 const descendant = !field and p.source.peek() == '.';
389 if (descendant) p.source.pos += 1;
390 if (!field and p.source.peek() == '*') {
391 p.source.pos += 1;
392 s = .{ .kind = .all };
393 } else if (descendant and p.source.peek() == '[') {
394 p.source.pos += 1;
395 s = try p.selection();
396 } else s = .{ .kind = .key, .key = try p.readIdentifier() };
397 s.descendant = descendant;
398 }
399 if (field and s.kind != .key and s.kind != .index) return p.invalid();
400 try out.append(gpa, s);
401 }
402 return out.items;
403}
404fn parseKey(gpa: Allocator, text: Text) Error!Text {
405 var p: PathReader = .{ .source = .{ .gpa = gpa, .text = text }, .reason = "bad key" };
406 const key = if (p.source.peek() == '\'') try p.quoted() else try p.readIdentifier();
407 if (p.source.pos != text.len) return p.invalid();
408 return key;
409}
410const Step = struct { key: ?Text = null, position: usize };
411const Match = struct {
412 node: ?*Value,
413 steps: []const Step = &.{},
414 fn child(self: Match, gpa: Allocator, i: usize) Allocator.Error!Match {
415 const v = self.node.?;
416 const steps = try gpa.alloc(Step, self.steps.len + 1);
417 @memcpy(steps[0..self.steps.len], self.steps);
418 steps[self.steps.len] = .{ .key = if (v.kind == .object) v.keys.items[i] else null, .position = i };
419 return .{ .node = v.items.items[i], .steps = steps };
420 }
421 fn location(self: Match, gpa: Allocator) Allocator.Error!Text {
422 var out: Builder = .{ .gpa = gpa };
423 try out.byte('$');
424 for (self.steps) |s| {
425 if (s.key) |key| {
426 if (identifier(key)) try out.print(".{s}", .{key}) else try out.print("[{s}]", .{try quote(gpa, key, '\'')});
427 } else try out.print("[{d}]", .{s.position});
428 }
429 return out.result();
430 }

432fn position(v: *Value, s: Selector) ?usize {
433 if (s.kind == .key) return if (v.kind == .object) v.find(s.key) else null;
434 if (v.kind != .array) return null;
435 var i = s.index;
436 if (i < 0) i += @intCast(v.items.items.len);
437 if (i < 0 or i >= @as(i64, @intCast(v.items.items.len))) return null;
438 return @intCast(i);
439}
440fn selectValue(gpa: Allocator, m: Match, s: Selector, out: *std.ArrayList(Match)) Error!void {
441 const v = m.node.?;
442 if (s.kind == .all) {
443 for (0..v.items.items.len) |i| try out.append(gpa, try m.child(gpa, i));
444 } else if (s.kind == .slice) {
445 if (v.kind != .array) return;
446 const n: i64 = @intCast(v.items.items.len);
447 var a = s.start orelse 0;
448 var b = s.end orelse n;
449 if (a < 0) a += n;
450 if (b < 0) b += n;
451 a = @max(0, @min(n, a));
452 b = @max(0, @min(n, b));
453 var i = a;
454 while (i < b) : (i += 1) try out.append(gpa, try m.child(gpa, @intCast(i)));
455 } else if (position(v, s)) |i| try out.append(gpa, try m.child(gpa, i));
456}
457fn selectDescendants(gpa: Allocator, m: Match, s: Selector, out: *std.ArrayList(Match)) Error!void {
458 try selectValue(gpa, m, s, out);
459 for (0..m.node.?.items.items.len) |i| try selectDescendants(gpa, try m.child(gpa, i), s, out);
460}
461fn applySegment(gpa: Allocator, matches: []const Match, s: Selector) Error![]Match {
462 var out: std.ArrayList(Match) = .empty;
463 for (matches) |m| {
464 if (s.descendant) try selectDescendants(gpa, m, s, &out) else try selectValue(gpa, m, s, &out);
465 }
466 return out.items;
467}
468fn evaluate(gpa: Allocator, root: *Value, path: []const Selector) Error![]Match {
469 var result = try gpa.alloc(Match, 1);
470 result[0] = .{ .node = root };
471 for (path) |s| result = try applySegment(gpa, result, s);
472 return result;
473}
474fn fieldValue(v: *Value, path: []const Selector) ?*Value {
475 var current = v;
476 for (path) |s| {
477 const i = position(current, s) orelse return null;
478 current = current.items.items[i];
479 }
480 return current;
481}
482fn rank(v: ?*Value) u8 {
483 const x = v orelse return 3;
484 return switch (x.kind) {
485 .null => 0,
486 .number => 1,
487 .string => 2,
488 else => 3,
489 };
490}
491fn compare(a: ?*Value, b: ?*Value) i8 {
492 const x = rank(a);
493 const y = rank(b);
494 if (x < y) return -1;
495 if (x > y) return 1;
496 if (x == 1) {
497 if (a.?.number < b.?.number) return -1;
498 if (a.?.number > b.?.number) return 1;
499 }
500 if (x == 2) return switch (std.mem.order(u8, a.?.text, b.?.text)) {
501 .lt => -1,
502 .eq => 0,
503 .gt => 1,
504 };
505 return 0;
506}
507fn holds(operator: Text, v: ?*Value, reference: ?*Value) bool {
508 if (equal(operator, "EXISTS")) return v != null;
509 if (equal(operator, "==") or equal(operator, "!=")) {
510 const matches = v != null and same(v.?, reference.?);
511 return if (equal(operator, "==")) matches else !matches;
512 }
513 if (v == null or v.?.kind != reference.?.kind or (v.?.kind != .number and v.?.kind != .string)) return false;
514 const c = compare(v, reference);
515 if (equal(operator, "<")) return c < 0;
516 if (equal(operator, "<=")) return c <= 0;
517 if (equal(operator, ">")) return c > 0;
518 return c >= 0;
519}
520fn documentOrder(a: Match, b: Match) i8 {
521 for (0..@min(a.steps.len, b.steps.len)) |i| {
522 if (a.steps[i].position < b.steps[i].position) return -1;
523 if (a.steps[i].position > b.steps[i].position) return 1;
524 }
525 if (a.steps.len < b.steps.len) return -1;
526 if (a.steps.len > b.steps.len) return 1;
527 return 0;
528}
529fn targets(gpa: Allocator, matches: []const Match) Allocator.Error![]Match {
530 const sorted = try gpa.dupe(Match, matches);
531 std.mem.sort(Match, sorted, {}, struct {
532 fn less(_: void, a: Match, b: Match) bool {
533 return documentOrder(a, b) > 0;
534 }
535 }.less);
536 var out: std.ArrayList(Match) = .empty;
537 for (sorted) |m| if (out.items.len == 0 or documentOrder(out.items[out.items.len - 1], m) != 0) {
538 try out.append(gpa, m);
539 };
540 return out.items;
541}
542fn resolve(root: *Value, steps: []const Step) *Value {
543 var v = root;
544 for (steps) |s| v = v.items.items[s.position];
545 return v;
546}
547fn merge(gpa: Allocator, target: *Value, patch: *Value) Error!void {
548 for (patch.keys.items, patch.items.items) |key, item| {
549 if (target.find(key)) |i| {
550 if (target.items.items[i].kind == .object and item.kind == .object) try merge(gpa, target.items.items[i], item) else target.items.items[i] = try item.copy(gpa);
551 } else {
552 try target.keys.append(gpa, key);
553 try target.items.append(gpa, try item.copy(gpa));
554 }
555 }
556}
557fn shape(gpa: Allocator, args: []const Text, minimum: usize, maximum: usize) Error!void {
558 if (args.len < minimum or args.len > maximum) return bad(gpa, "bad arguments", .{});
559}
560fn requireKind(gpa: Allocator, matches: []const Match, kind: Kind, reason: Text) Error!void {
561 for (matches) |m| if (m.node.?.kind != kind) return bad(gpa, "{s} at {s}", .{ reason, try m.location(gpa) });
562}
563const Tool = struct {
564 gpa: Allocator,
565 document: *Value,
566 history: std.ArrayList(*Value) = .empty,
567 out: Builder,
568 fn commit(self: *Tool, v: *Value) Allocator.Error!void {
569 try self.history.append(self.gpa, self.document);
570 self.document = v;
571 }
572 fn load(self: *Tool, lines: []const Text) Error!void {
573 const text = try join(self.gpa, lines, "\n");
574 const v = parseJson(self.gpa, text) catch |err| {
575 if (err != error.Invalid) return err;
576 var line: usize = 1;
577 var column: usize = 1;
578 for (text[0..failure.position]) |c| {
579 if (c == '\n') {
580 line += 1;
581 column = 1;
582 } else column += 1;
583 }
584 try self.out.line("PARSE ERROR line {d} col {d}: {s}", .{ line, column, failure.reason });
585 return;
586 };
587 try self.commit(v);
588 const counts = v.size();
589 try self.out.line("LOADED nodes={d} depth={d}", .{ counts[0], counts[1] });
590 }
591 fn leaves(self: *Tool, m: Match) Error!void {
592 if (m.node.?.items.items.len == 0) {
593 try self.out.line("{s} = {s}", .{ try m.location(self.gpa), try compact(self.gpa, m.node.?) });
594 return;
595 }
596 for (0..m.node.?.items.items.len) |i| try self.leaves(try m.child(self.gpa, i));
597 }
598 fn execute(self: *Tool, line: Text) Error!void {
599 const gpa = self.gpa;
600 const separator = std.mem.findScalar(u8, line, ' ');
601 const command = if (separator) |p| line[0..p] else line;
602 const argument = if (separator) |p| line[p + 1 ..] else "";
603 const known = " GET COMPACT PRETTY TYPE LENGTH KEYS FLATTEN COUNT SUM MIN MAX AVG FILTER SET DELETE INSERT MERGE SORT RENAME UNDO ";
604 const needle = try std.fmt.allocPrint(gpa, " {s} ", .{command});
605 if (std.mem.find(u8, known, needle) == null) return bad(gpa, "unknown command '{s}'", .{command});
606 var args = if (separator != null) try split(gpa, argument, ' ', 0) else try gpa.alloc(Text, 0);
607 if (equal(command, "UNDO")) {
608 try shape(gpa, args, 0, 0);
609 if (self.history.items.len == 0) return bad(gpa, "nothing to undo", .{});
610 self.document = self.history.items[self.history.items.len - 1];
611 self.history.items.len -= 1;
612 try self.out.line("UNDO {d}", .{self.history.items.len});
613 return;
614 }
615 if (equal(command, "SET") or equal(command, "MERGE") or equal(command, "INSERT")) {
616 const count: usize = if (equal(command, "INSERT")) 3 else 2;
617 args = try split(gpa, argument, ' ', count);
618 if (separator == null or args.len != count) return bad(gpa, "bad arguments", .{});
619 } else if (equal(command, "FILTER")) {
620 args = try split(gpa, argument, ' ', 4);
621 if (separator == null or args.len < 3 or (equal(args[2], "EXISTS") and args.len == 4) or (!equal(args[2], "EXISTS") and args.len == 3)) return bad(gpa, "bad arguments", .{});
622 } else if (equal(command, "SORT")) {
623 try shape(gpa, args, 1, 3);
624 if (args.len == 3 and (equal(args[1], "DESC") or !equal(args[2], "DESC"))) return bad(gpa, "bad arguments", .{});
625 } else {
626 const optional = equal(command, "COMPACT") or equal(command, "PRETTY") or equal(command, "FLATTEN");
627 const rename = equal(command, "RENAME");
628 try shape(gpa, args, if (rename) 2 else if (optional) 0 else 1, if (rename) 2 else 1);
629 }
630 const path = try parsePath(gpa, if (args.len == 0) "$" else args[0], false);
631 var field: []Selector = &.{};
632 var patch: ?*Value = null;
633 var index: i64 = 0;
634 var new_key: Text = "";
635 var operator: Text = "";
636 var descending = false;
637 if (equal(command, "FILTER")) {
638 field = try parsePath(gpa, args[1], true);
639 operator = args[2];
640 if (!equal(operator, "==") and !equal(operator, "!=") and !equal(operator, "<") and !equal(operator, "<=") and !equal(operator, ">") and !equal(operator, ">=") and !equal(operator, "EXISTS")) return bad(gpa, "bad operator", .{});
641 if (!equal(operator, "EXISTS")) patch = try parseJson(gpa, args[3]);
642 } else if (equal(command, "SET") or equal(command, "MERGE")) patch = try parseJson(gpa, args[1]) else if (equal(command, "INSERT")) {
643 if (!integerNotation(args[1])) return bad(gpa, "bad index", .{});
644 index = std.fmt.parseInt(i64, args[1], 10) catch unreachable;
645 patch = try parseJson(gpa, args[2]);
646 } else if (equal(command, "SORT")) {
647 var field_text: Text = "@";
648 if (args.len > 1) {
649 descending = equal(args[args.len - 1], "DESC");
650 if (!equal(args[1], "DESC")) field_text = args[1];
651 }
652 field = try parsePath(gpa, field_text, true);
653 } else if (equal(command, "RENAME")) new_key = try parseKey(gpa, args[1]);
654 if (equal(command, "DELETE") and path.len == 0) return bad(gpa, "cannot delete root", .{});
655 if (equal(command, "MERGE") and patch.?.kind != .object) return bad(gpa, "merge value is not an object", .{});
656 var matches = try evaluate(gpa, self.document, path);
657 if (equal(command, "LENGTH")) for (matches) |m| if (m.node.?.kind != .object and m.node.?.kind != .array and m.node.?.kind != .string) return bad(gpa, "no length at {s}", .{try m.location(gpa)});
658 if (equal(command, "KEYS") or equal(command, "MERGE")) try requireKind(gpa, matches, .object, "not an object");
659 if (equal(command, "FILTER") or equal(command, "INSERT") or equal(command, "SORT")) try requireKind(gpa, matches, .array, "not an array");
660 if (equal(command, "INSERT")) for (matches) |m| {
661 const n: i64 = @intCast(m.node.?.items.items.len);
662 if (index < -n - 1 or index > n) return bad(gpa, "index out of range at {s}", .{try m.location(gpa)});
663 };
664 if (equal(command, "RENAME")) for (matches) |m| if (m.steps.len == 0 or m.steps[m.steps.len - 1].key == null) return bad(gpa, "not a member at {s}", .{try m.location(gpa)});
665 if (equal(command, "COUNT")) {
666 try self.out.line("COUNT {d}", .{matches.len});
667 return;
668 }
669 if (equal(command, "SUM") or equal(command, "MIN") or equal(command, "MAX") or equal(command, "AVG")) {
670 var total: i64 = 0;
671 var minimum: i64 = 0;
672 var maximum: i64 = 0;
673 var count: i64 = 0;
674 for (matches) |m| if (m.node.?.kind == .number) {
675 const n = m.node.?.number;
676 total += n;
677 if (count == 0 or n < minimum) minimum = n;
678 if (count == 0 or n > maximum) maximum = n;
679 count += 1;
680 };
681 var text: Text = "null";
682 if (equal(command, "SUM")) text = try std.fmt.allocPrint(gpa, "{d}", .{total}) else if (equal(command, "MIN") and count > 0) text = try std.fmt.allocPrint(gpa, "{d}", .{minimum}) else if (equal(command, "MAX") and count > 0) text = try std.fmt.allocPrint(gpa, "{d}", .{maximum}) else if (equal(command, "AVG") and count > 0) {
683 const magnitude = if (total < 0) -total else total;
684 const q = @divTrunc(magnitude * 200 + count, 2 * count);
685 text = try std.fmt.allocPrint(gpa, "{s}{d}.{d:0>2}", .{ if (total < 0 and q >= 1) @as(Text, "-") else "", @divTrunc(q, 100), @mod(q, 100) });
686 }
687 try self.out.line("{s} {s}", .{ command, text });
688 return;
689 }
690 const modifying = equal(command, "SET") or equal(command, "DELETE") or equal(command, "INSERT") or equal(command, "MERGE") or equal(command, "SORT") or equal(command, "RENAME");
691 if (modifying) {
692 if (equal(command, "SET") and path.len == 0) {
693 try self.commit(try patch.?.copy(gpa));
694 try self.out.add("SET 1\n");
695 return;
696 }
697 if (equal(command, "SET")) {
698 var selected: std.ArrayList(Match) = .empty;
699 const last = path[path.len - 1];
700 for (try evaluate(gpa, self.document, path[0 .. path.len - 1])) |parent| {
701 if (last.kind == .key and !last.descendant and parent.node.?.kind == .object) {
702 const steps = try gpa.alloc(Step, parent.steps.len + 1);
703 @memcpy(steps[0..parent.steps.len], parent.steps);
704 steps[parent.steps.len] = .{ .key = last.key, .position = parent.node.?.find(last.key) orelse parent.node.?.items.items.len };
705 try selected.append(gpa, .{ .node = null, .steps = steps });
706 } else try selected.appendSlice(gpa, try applySegment(gpa, &.{parent}, last));
707 }
708 matches = selected.items;
709 }
710 const selected = try targets(gpa, matches);
711 const document = try self.document.copy(gpa);
712 for (selected) |target| {
713 if (equal(command, "SET") or equal(command, "DELETE") or equal(command, "RENAME")) {
714 const parent = resolve(document, target.steps[0 .. target.steps.len - 1]);
715 const i = target.steps[target.steps.len - 1].position;
716 if (equal(command, "SET")) {
717 if (i == parent.items.items.len) {
718 try parent.keys.append(gpa, target.steps[target.steps.len - 1].key.?);
719 try parent.items.append(gpa, try patch.?.copy(gpa));
720 } else parent.items.items[i] = try patch.?.copy(gpa);
721 } else if (equal(command, "DELETE")) {
722 _ = parent.items.orderedRemove(i);
723 if (parent.kind == .object) _ = parent.keys.orderedRemove(i);
724 } else {
725 if (!equal(parent.keys.items[i], new_key) and parent.find(new_key) != null) return bad(gpa, "key exists at {s}", .{try target.location(gpa)});
726 parent.keys.items[i] = new_key;
727 }
728 } else {
729 const v = resolve(document, target.steps);
730 if (equal(command, "INSERT")) {
731 const j = if (index >= 0) index else @as(i64, @intCast(v.items.items.len)) + 1 + index;
732 try v.items.insert(gpa, @intCast(j), try patch.?.copy(gpa));
733 } else if (equal(command, "MERGE")) try merge(gpa, v, patch.?) else {
734 var at: usize = 1;
735 while (at < v.items.items.len) : (at += 1) {
736 const item = v.items.items[at];
737 const key = fieldValue(item, field);
738 var destination = at;
739 while (destination > 0) {
740 const comparison = compare(key, fieldValue(v.items.items[destination - 1], field));
741 if (!(if (descending) comparison > 0 else comparison < 0)) break;
742 v.items.items[destination] = v.items.items[destination - 1];
743 destination -= 1;
744 }
745 v.items.items[destination] = item;
746 }
747 }
748 }
749 }
750 try self.commit(document);
751 try self.out.line("{s} {d}", .{ command, selected.len });
752 return;
753 }
754 var emitted = false;
755 for (matches) |m| {
756 const v = m.node.?;
757 if (equal(command, "GET")) try self.out.line("{s} = {s}", .{ try m.location(gpa), try compact(gpa, v) }) else if (equal(command, "COMPACT")) try self.out.line("{s}", .{try compact(gpa, v)}) else if (equal(command, "PRETTY")) try self.out.line("{s}", .{try pretty(gpa, v, "")}) else if (equal(command, "TYPE")) try self.out.line("{s} {s}", .{ try m.location(gpa), @tagName(v.kind) }) else if (equal(command, "LENGTH")) try self.out.line("{s} {d}", .{ try m.location(gpa), if (v.kind == .string) v.text.len else v.items.items.len }) else if (equal(command, "KEYS")) {
758 var quoted_keys: std.ArrayList(Text) = .empty;
759 for (v.keys.items) |key| try quoted_keys.append(gpa, try quote(gpa, key, '"'));
760 try self.out.line("{s} [{s}]", .{ try m.location(gpa), try join(gpa, quoted_keys.items, ",") });
761 } else if (equal(command, "FLATTEN")) try self.leaves(m) else if (equal(command, "FILTER")) {
762 for (0..v.items.items.len) |i| {
763 const element = try m.child(gpa, i);
764 if (holds(operator, fieldValue(element.node.?, field), patch)) {
765 try self.out.line("{s} = {s}", .{ try element.location(gpa), try compact(gpa, element.node.?) });
766 emitted = true;
767 }
768 }
769 continue;
770 }
771 emitted = true;
772 }
773 if (!emitted) try self.out.add("no match\n");
774 }
775 fn run(self: *Tool, lines: []const Text) Error!void {
776 var at: usize = 0;
777 while (true) {
778 const start = at;
779 while (!equal(lines[at], "%%")) at += 1;
780 try self.load(lines[start..at]);
781 at += 1;
782 while (!equal(lines[at], "END") and !equal(lines[at], "LOAD")) {
783 self.execute(lines[at]) catch |err| {
784 if (err != error.Invalid) return err;
785 if (failure.json) try self.out.line("PARSE ERROR col {d}: {s}", .{ failure.position + 1, failure.reason }) else try self.out.line("ERROR {s}", .{failure.reason});
786 };
787 at += 1;
788 }
789 if (equal(lines[at], "END")) return;
790 at += 1;
791 }
792 }
793};
794pub fn main(init: std.process.Init) !void {
795 var storage = std.heap.ArenaAllocator.init(std.heap.page_allocator);
796 defer storage.deinit();
797 const gpa = storage.allocator();
798 var input_buffer: [4096]u8 = undefined;
799 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
800 var output_buffer: [4096]u8 = undefined;
801 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
802 const text = try input.interface.allocRemaining(gpa, .unlimited);
803 var tool: Tool = .{ .gpa = gpa, .document = try make(gpa, .null), .out = .{ .gpa = gpa } };
804 try tool.run(try split(gpa, text, '\n', 0));
805 try output.interface.writeAll(tool.out.result());
806 try output.interface.flush();
807}
