39 read_string <- function() {
40 position <<- position + 1
41 output <- ""
42 repeat {
43 c <- char()
44 if (c == "" || c == "\n") error("unterminated string")
45 if (c == '"') {
46 position <<- position + 1
47 return(output)
48 }
49 if (c == "\\") {
50 start <- position
51 escaped <- slice(text, position + 1, position + 1)
52 if (escaped == "" || escaped == "\n") error("unterminated string", start + 1)
53 if (escaped == "u") {
54 hex <- slice(text, position + 2, position + 5)
55 if (!grepl("^[0-9A-Fa-f]{4}$", hex)) error("bad unicode escape", start)
56 if (strtoi(hex, 16) > 127) error("non-ascii escape", start)
57 output <- paste0(output, intToUtf8(strtoi(hex, 16)))
58 position <<- position + 6
59 } else {
60 escapes <- c('"' = '"', "\\" = "\\", "/" = "/", "b" = "\b", "f" = "\f", "n" = "\n", "r" = "\r", "t" = "\t")
61 if (!(escaped %in% names(escapes))) error(paste0("invalid escape '\\", escaped, "'"), start)
62 output <- paste0(output, escapes[[escaped]])
63 position <<- position + 2
64 }
65 } else {
66 output <- paste0(output, c)
67 position <<- position + 1
68 }
69 }

71 read_value <- function() {
72 whitespace()
73 c <- char()
74 if (c == "") error("unexpected end of input")
75 if (c %in% c("{", "[")) {
76 object <- c == "{"
77 ending <- if (object) "}" else "]"
78 result <- value(if (object) "object" else "array", list())
79 position <<- position + 1
80 whitespace()
81 if (char() == ending) {
82 position <<- position + 1
83 return(result)
84 }
85 repeat {
86 if (object) {
87 whitespace()
88 if (char() == "") error("unexpected end of input")
89 if (char() != '"') error("expected key")
90 start <- position
91 key <- read_string()
92 if (member(result, key)) error(paste0("duplicate key ", quoted(key)), start)
93 whitespace()
94 if (char() == "") error("unexpected end of input")
95 if (char() != ":") error("expected ':'")
96 position <<- position + 1
97 result$data[[length(result$data) + 1]] <- list(key, read_value())
98 } else result$data[[length(result$data) + 1]] <- read_value()
99 whitespace()
100 if (char() == ending) {
101 position <<- position + 1
102 return(result)
103 }
104 if (char() == "") error("unexpected end of input")
105 if (char() != ",") error(paste0("expected ',' or '", ending, "'"))
106 position <<- position + 1
107 whitespace()
108 if (char() == ending) error("trailing comma")
109 }
110 }
111 if (c == '"') return(value("string", read_string()))
112 if (c == "-" || grepl("^[0-9]$", c)) {
113 start <- position
114 if (c == "-") position <<- position + 1
115 digit_start <- position
116 while (grepl("^[0-9]$", char())) position <<- position + 1
117 digits <- slice(text, digit_start, position - 1)
118 if (digits == "") error("invalid number", start)
119 if (nchar(digits) > 1 && startsWith(digits, "0")) error("leading zero", digit_start)
120 if (char() == ".") error("fraction not allowed")
121 if (char() %in% c("e", "E")) error("exponent not allowed")
122 if (nchar(digits) >= 10) error("number out of range", start)
123 return(value("number", as.numeric(slice(text, start, position - 1))))
124 }
125 if (grepl("^[A-Za-z]$", c)) {
126 start <- position
127 while (grepl("^[A-Za-z]$", char())) position <<- position + 1
128 word <- slice(text, start, position - 1)
129 if (!(word %in% c("true", "false", "null"))) error(paste0("invalid literal '", word, "'"), start)
130 return(if (word == "null") value() else value("boolean", word == "true"))
131 }
132 error(paste0("unexpected character '", c, "'"))
133 }
134 result <- read_value()
135 whitespace()
136 if (position <= nchar(text)) error("trailing characters")

168parse_path <- function(text, field = FALSE) {
169 tryCatch({
170 if (slice(text, 1, 1) != if (field) "@" else "$") stop("bad")
171 position <- 2
172 selectors <- list()
173 word <- function() {
174 rest <- slice(text, position)
175 match <- regexpr("^([A-Za-z_][A-Za-z0-9_]*|\\*)", rest)
176 if (match[1] < 0) stop("bad")
177 regmatches(rest, match)
178 }
179 while (position <= nchar(text)) {
180 descendant <- FALSE
181 if (!field && slice(text, position, position + 1) == "..") {
182 descendant <- TRUE
183 position <- position + 2
184 } else if (substr(text, position, position) == ".") {
185 position <- position + 1
186 key <- word()
187 if (field && key == "*") stop("bad")
188 selectors[[length(selectors) + 1]] <- list(descendant = FALSE, kind = if (key == "*") "all" else "key", key = key)
189 position <- position + nchar(key)
190 next
191 } else if (substr(text, position, position) != "[") stop("bad")
192 if (slice(text, position, position) == "[") {
193 position <- position + 1
194 if (slice(text, position, position) == "'") {
195 parsed <- quoted_key(text, position)
196 key <- parsed$text
197 position <- parsed$position
198 kind <- "key"
199 } else {
200 at <- regexpr("]", slice(text, position), fixed = TRUE)[1]
201 if (at < 0) stop("bad")
202 finish <- position + at - 1
203 token <- slice(text, position, finish - 1)
204 position <- finish
205 if (grepl(integer_pattern, token)) {
206 kind <- "index"
207 key <- as.numeric(token)
208 } else if (!field && token == "*") {
209 kind <- "all"
210 key <- NULL
211 } else if (!field && sum(letters_of(token) == ":") == 1) {
212 at <- regexpr(":", token, fixed = TRUE)[1]
213 bounds <- c(slice(token, 1, at - 1), slice(token, at + 1))
214 if (any(bounds != "" & !grepl(integer_pattern, bounds))) stop("bad")
215 kind <- "slice"
216 key <- lapply(bounds, function(bound) if (bound == "") NULL else as.numeric(bound))
217 } else stop("bad")
218 }
219 if (slice(text, position, position) != "]") stop("bad")
220 position <- position + 1
221 } else if (descendant) {
222 key <- word()
223 kind <- if (key == "*") "all" else "key"
224 position <- position + nchar(key)
225 } else stop("bad")
226 selectors[[length(selectors) + 1]] <- list(descendant = descendant, kind = kind, key = key)
227 }
228 selectors
229 }, error = function(e) fail(if (field) "ERROR bad field" else "ERROR bad path"))

347 SORT = count >= 1 && count <= 3 && (count < 3 || (args[2] != "DESC" && args[3] == "DESC")),

452 else if (name == "COMPACT") output <- c(output, compact(v))
453 else if (name == "PRETTY") output <- c(output, pretty(v))
454 else if (name == "TYPE") output <- c(output, paste0(where, " ", v$kind))
455 else if (name == "LENGTH") output <- c(output, paste0(where, " ", if (v$kind == "string") nchar(v$data) else length(v$data)))

459 } else if (name == "FILTER") {

472 }

474 }
475 if (length(output)) cat(paste0(output, "\n"), sep = "") else emit("no match")
476 }, command_error = function(e) emit(e$message))
