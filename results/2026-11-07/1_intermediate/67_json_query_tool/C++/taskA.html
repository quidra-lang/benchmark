13enum class Kind { Object, Array, String, Number, Boolean, Null };
14struct Value {
15 Kind kind = Kind::Null;
16 std::string text;
17 std::int64_t number = 0;
18 bool boolean = false;
19 std::vector<std::string> keys;
20 std::vector<Value> items;
21
22 explicit Value(Kind kind = Kind::Null) : kind(kind) {}
23 int find(const std::string& key) const {
24 auto found = std::find(keys.begin(), keys.end(), key);
25 return found == keys.end() ? -1 : static_cast<int>(found - keys.begin());
26 }
27 int count() const {
28 int result = 1;
29 for (const Value& item : items) result += item.count();
30 return result;
31 }
32 int depth() const {
33 int result = 1;
34 for (const Value& item : items) result = std::max(result, 1 + item.depth());
35 return result;
36 }

39bool letter(char c) { return (c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z'); }
40bool digit(char c) { return c >= '0' && c <= '9'; }
41bool identifier_start(char c) { return letter(c) || c == '_'; }
42bool identifier_char(char c) { return identifier_start(c) || digit(c); }
43bool identifier(const std::string& text) {
44 return !text.empty() && identifier_start(text[0]) &&
45 std::all_of(text.begin(), text.end(), identifier_char);
46}
47bool integer_notation(const std::string& text) {
48 if (text == "0") return true;
49 std::size_t start = !text.empty() && text[0] == '-' ? 1 : 0;
50 return text.size() > start && text.size() - start <= 9 && text[start] != '0' &&
51 std::all_of(text.begin() + start, text.end(), digit);
52}
53int hexadecimal(char c) {
54 if (digit(c)) return c - '0';
55 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
56 if (c >= 'A' && c <= 'F') return c - 'A' + 10;
57 return -1;
58}

60std::string quote(const std::string& text, char mark = '"') {
61 std::string result(1, mark);
62 static const char* digits = "0123456789abcdef";
63 for (unsigned char c : text) {
64 if (c == static_cast<unsigned char>(mark) || c == '\\') {
65 result += '\\';
66 result += static_cast<char>(c);
67 }
68 else if (c == 8) result += "\\b";
69 else if (c == 12) result += "\\f";
70 else if (c == 10) result += "\\n";
71 else if (c == 13) result += "\\r";
72 else if (c == 9) result += "\\t";
73 else if (c < 32 || c == 127) {
74 result += "\\u00";
75 result += digits[c >> 4];
76 result += digits[c & 15];
77 } else result += static_cast<char>(c);
78 }
79 return result + mark;
80}

82std::string compact(const Value& value) {
83 if (value.kind == Kind::Null) return "null";
84 if (value.kind == Kind::Boolean) return value.boolean ? "true" : "false";
85 if (value.kind == Kind::Number) return std::to_string(value.number);
86 if (value.kind == Kind::String) return quote(value.text);
87 std::string result = value.kind == Kind::Object ? "{" : "[";
88 for (std::size_t i = 0; i < value.items.size(); ++i) {
89 if (i) result += ',';
90 if (value.kind == Kind::Object) result += quote(value.keys[i]) + ":";
91 result += compact(value.items[i]);
92 }
93 return result + (value.kind == Kind::Object ? '}' : ']');
94}
95
96std::string pretty(const Value& value, const std::string& indent = "") {
97 if (value.items.empty()) return compact(value);
98 std::string inner = indent + " ";
99 std::string result = value.kind == Kind::Object ? "{\n" : "[\n";
100 for (std::size_t i = 0; i < value.items.size(); ++i) {
101 if (i) result += ",\n";
102 result += inner;
103 if (value.kind == Kind::Object) result += quote(value.keys[i]) + ": ";
104 result += pretty(value.items[i], inner);
105 }
106 return result + "\n" + indent + (value.kind == Kind::Object ? "}" : "]");
107}

109bool equal(const Value& a, const Value& b) {
110 if (a.kind != b.kind) return false;
111 if (a.kind == Kind::String) return a.text == b.text;
112 if (a.kind == Kind::Number) return a.number == b.number;
113 if (a.kind == Kind::Boolean) return a.boolean == b.boolean;
114 if (a.items.size() != b.items.size()) return false;
115 for (std::size_t i = 0; i < a.items.size(); ++i) {
116 int j = a.kind == Kind::Object ? b.find(a.keys[i]) : static_cast<int>(i);
117 if (j < 0 || !equal(a.items[i], b.items[static_cast<std::size_t>(j)])) return false;
118 }
119 return true;
120}

127class JsonReader {
128 const std::string& text;
129 std::size_t position = 0;
130 char peek() const { return position < text.size() ? text[position] : '\0'; }
131 [[noreturn]] void fail(const std::string& reason) const { throw JsonError(reason, position); }
132 void whitespace() { while (peek() == ' ' || peek() == '\n') ++position; }
133 void require(char c, const std::string& reason) const {
134 if (position == text.size()) fail("unexpected end of input");
135 if (peek() != c) fail(reason);
136 }
137 std::string string() {
138 ++position;
139 std::string result;
140 while (true) {
141 char c = peek();
142 if (c == '\0' || c == '\n') fail("unterminated string");
143 if (c == '"') {
144 ++position;
145 return result;
146 }
147 if (c != '\\') {
148 result += c;
149 ++position;
150 continue;
151 }
152 std::size_t slash = position++;
153 char escape = peek();
154 if (escape == '\0' || escape == '\n') fail("unterminated string");
155 if (escape == 'u') {
156 int code = 0;
157 if (position + 4 >= text.size()) throw JsonError("bad unicode escape", slash);
158 for (std::size_t i = position + 1; i <= position + 4; ++i) {
159 int h = hexadecimal(text[i]);
160 if (h < 0) throw JsonError("bad unicode escape", slash);
161 code = code * 16 + h;
162 }
163 if (code > 127) throw JsonError("non-ascii escape", slash);
164 result += static_cast<char>(code);
165 position += 5;
166 } else {
167 switch (escape) {
168 case '"': case '\\': case '/': result += escape;
169 break;
170 case 'b': result += '\b';
171 break;
172 case 'f': result += '\f';
173 break;
174 case 'n': result += '\n';
175 break;
176 case 'r': result += '\r';
177 break;
178 case 't': result += '\t';
179 break;
180 default: throw JsonError("invalid escape '\\" + std::string(1, escape) + "'", slash);
181 }
182 ++position;
183 }
184 }
185 }
186 Value number() {
187 std::size_t start = position;
188 bool negative = peek() == '-';
189 if (negative) ++position;
190 std::size_t digits = position;
191 while (digit(peek())) ++position;
192 std::size_t length = position - digits;
193 if (!length) throw JsonError("invalid number", start);
194 if (length > 1 && text[digits] == '0') throw JsonError("leading zero", digits);
195 if (peek() == '.') fail("fraction not allowed");
196 if (peek() == 'e' || peek() == 'E') fail("exponent not allowed");
197 if (length > 9) throw JsonError("number out of range", start);
198 Value value(Kind::Number);
199 value.number = std::stoll(text.substr(start, position - start));
200 return value;
201 }
202 bool end_container(char closing, const std::string& reason) {
203 whitespace();
204 if (peek() == closing) {
205 ++position;
206 return true;
207 }
208 require(',', reason);
209 ++position;
210 whitespace();
211 if (peek() == closing) fail("trailing comma");
212 return false;
213 }
214 Value object() {
215 ++position;
216 whitespace();
217 Value object(Kind::Object);
218 if (peek() == '}') {
219 ++position;
220 return object;
221 }
222 while (true) {
223 whitespace();
224 require('"', "expected key");
225 std::size_t key_start = position;
226 std::string key = string();
227 if (object.find(key) >= 0) throw JsonError("duplicate key " + quote(key), key_start);
228 whitespace();
229 require(':', "expected ':'");
230 ++position;
231 Value child = value();
232 object.keys.push_back(std::move(key));
233 object.items.push_back(std::move(child));
234 if (end_container('}', "expected ',' or '}'")) return object;
235 }
236 }
237 Value array() {
238 ++position;
239 whitespace();
240 Value array(Kind::Array);
241 if (peek() == ']') {
242 ++position;
243 return array;
244 }
245 while (true) {
246 array.items.push_back(value());
247 if (end_container(']', "expected ',' or ']'")) return array;
248 }
249 }
250 Value value() {
251 whitespace();
252 char c = peek();
253 if (c == '\0') fail("unexpected end of input");
254 if (c == '{') return object();
255 if (c == '[') return array();
256 if (c == '"') {
257 Value result(Kind::String);
258 result.text = string();
259 return result;
260 }
261 if (c == '-' || digit(c)) return number();
262 if (letter(c)) {
263 std::size_t start = position;
264 while (letter(peek())) ++position;
265 std::string word = text.substr(start, position - start);
266 if (word == "null") return Value();
267 if (word == "true" || word == "false") {
268 Value result(Kind::Boolean);
269 result.boolean = word == "true";
270 return result;
271 }
272 throw JsonError("invalid literal '" + word + "'", start);
273 }
274 fail("unexpected character '" + std::string(1, c) + "'");
275 }
276public:
277 explicit JsonReader(const std::string& text) : text(text) {}
278 Value parse() {
279 Value result = value();
280 whitespace();
281 if (position != text.size()) fail("trailing characters");
282 return result;
283 }
284};

295class PathReader {
296 const std::string& text;
297 std::string reason;
298 std::size_t position = 0;
299 char peek() const { return position < text.size() ? text[position] : '\0'; }
300 [[noreturn]] void fail() const { throw std::runtime_error(reason); }
301 bool accept(char c) {
302 if (peek() != c) return false;
303 ++position;
304 return true;
305 }
306 void expect(char c) { if (!accept(c)) fail(); }
307 std::string name() {
308 if (!identifier_start(peek())) fail();
309 std::size_t start = position++;
310 while (identifier_char(peek())) ++position;
311 return text.substr(start, position - start);
312 }
313 std::string quoted_key() {
314 expect('\'');
315 std::string result;
316 while (true) {
317 char c = peek();
318 if (c == '\0') fail();
319 ++position;
320 if (c == '\'') return result;
321 if (c != '\\') {
322 result += c;
323 continue;
324 }
325 char escape = peek();
326 if (escape == '\0') fail();
327 ++position;
328 if (escape == 'u') {
329 if (position + 4 > text.size()) fail();
330 int code = 0;
331 for (int i = 0; i < 4; ++i) {
332 int h = hexadecimal(text[position++]);
333 if (h < 0) fail();
334 code = code * 16 + h;
335 }
336 if (code > 127) fail();
337 result += static_cast<char>(code);
338 } else {
339 switch (escape) {
340 case '\'': case '\\': case '/': result += escape;
341 break;
342 case 'b': result += '\b';
343 break;
344 case 'f': result += '\f';
345 break;
346 case 'n': result += '\n';
347 break;
348 case 'r': result += '\r';
349 break;
350 case 't': result += '\t';
351 break;
352 default: fail();
353 }
354 }
355 }
356 }
357 Selector selection() {
358 Selector selector;
359 if (peek() == '\'') {
360 selector.key = quoted_key();
361 expect(']');
362 return selector;
363 }
364 std::size_t end = text.find(']', position);
365 if (end == std::string::npos) fail();
366 std::string content = text.substr(position, end - position);
367 position = end + 1;
368 if (content == "*") {
369 selector.kind = SelectorKind::All;
370 return selector;
371 }
372 if (integer_notation(content)) {
373 selector.kind = SelectorKind::Index;
374 selector.index = std::stoi(content);
375 return selector;
376 }
377 std::size_t colon = content.find(':');
378 if (colon == std::string::npos || content.find(':', colon + 1) != std::string::npos) fail();
379 std::string first = content.substr(0, colon), last = content.substr(colon + 1);
380 if ((!first.empty() && !integer_notation(first)) || (!last.empty() && !integer_notation(last))) fail();
381 selector.kind = SelectorKind::Slice;
382 if (!first.empty()) selector.first = std::stoi(first);
383 if (!last.empty()) selector.last = std::stoi(last);
384 return selector;
385 }
386public:
387 PathReader(const std::string& text, std::string reason) : text(text), reason(std::move(reason)) {}
388 std::vector<Selector> path(bool field = false) {
389 expect(field ? '@' : '$');
390 std::vector<Selector> selectors;
391 while (position != text.size()) {
392 Selector selector;
393 if (accept('[')) selector = selection();
394 else {
395 expect('.');
396 bool descendant = !field && accept('.');
397 if (!field && accept('*')) selector.kind = SelectorKind::All;
398 else if (descendant && accept('[')) selector = selection();
399 else selector.key = name();
400 selector.descendant = descendant;
401 }
402 if (field && selector.kind != SelectorKind::Key && selector.kind != SelectorKind::Index) fail();
403 selectors.push_back(std::move(selector));
404 }
405 return selectors;
406 }
407 std::string key() {
408 std::string result = peek() == '\'' ? quoted_key() : name();
409 if (position != text.size()) fail();
410 return result;
411 }
412};

414struct Step {
415 std::optional<std::string> key;
416 int position;
417 std::string text() const {
418 if (!key) return "[" + std::to_string(position) + "]";
419 return identifier(*key) ? "." + *key : "[" + quote(*key, '\'') + "]";
420 }
421};
422struct Match {
423 const Value* value;
424 std::vector<Step> steps;
425 Match child(int position) const {
426 Match result{&value->items[static_cast<std::size_t>(position)], steps};
427 result.steps.push_back({value->kind == Kind::Object ? std::optional<std::string>(value->keys[position]) : std::nullopt, position});
428 return result;
429 }
430 std::string location() const {
431 std::string result = "$";
432 for (const Step& step : steps) result += step.text();
433 return result;
434 }
435 std::vector<int> positions() const {
436 std::vector<int> result;
437 for (const Step& step : steps) result.push_back(step.position);
438 return result;
439 }
440};

442int child_position(const Value& value, const Selector& selector) {
443 if (selector.kind == SelectorKind::Key) return value.kind == Kind::Object ? value.find(selector.key) : -1;
444 if (value.kind != Kind::Array) return -1;
445 int count = static_cast<int>(value.items.size());
446 int index = selector.index < 0 ? count + selector.index : selector.index;
447 return index >= 0 && index < count ? index : -1;
448}
449
450void select(const Selector& selector, const Match& match, std::vector<Match>& output) {
451 const Value& value = *match.value;
452 if (selector.kind == SelectorKind::Key || selector.kind == SelectorKind::Index) {
453 int position = child_position(value, selector);
454 if (position >= 0) output.push_back(match.child(position));
455 } else if (selector.kind == SelectorKind::All) {
456 for (int i = 0; i < static_cast<int>(value.items.size()); ++i) output.push_back(match.child(i));
457 } else if (value.kind == Kind::Array) {
458 int count = static_cast<int>(value.items.size());
459 auto clamp = [count](int bound) {
460 if (bound < 0) bound += count;
461 return std::clamp(bound, 0, count);
462 };
463 int begin = selector.first ? clamp(*selector.first) : 0;
464 int end = selector.last ? clamp(*selector.last) : count;
465 for (int i = begin; i < end; ++i) output.push_back(match.child(i));
466 }
467}
468
469void select_descendants(const Selector& selector, const Match& match, std::vector<Match>& output) {
470 select(selector, match, output);
471 for (int i = 0; i < static_cast<int>(match.value->items.size()); ++i)
472 select_descendants(selector, match.child(i), output);
473}

475std::vector<Match> apply(const Selector& selector, const std::vector<Match>& matches) {
476 std::vector<Match> result;
477 for (const Match& match : matches) {
478 if (selector.descendant) select_descendants(selector, match, result);
479 else select(selector, match, result);
480 }
481 return result;
482}

484std::vector<Match> evaluate(const Value& root, const std::vector<Selector>& selectors) {
485 std::vector<Match> matches{{&root, {}}};
486 for (const Selector& selector : selectors) matches = apply(selector, matches);
487 return matches;
488}

501 std::sort(matches.begin(), matches.end(), [](const Match& a, const Match& b) { return a.positions() > b.positions(); });

508Value& resolve(Value& root, const Match& match, bool parent = false) {
509 Value* value = &root;
510 std::size_t count = match.steps.size() - (parent ? 1 : 0);
511 for (std::size_t i = 0; i < count; ++i) value = &value->items[static_cast<std::size_t>(match.steps[i].position)];
512 return *value;
513}

519void merge(Value& target, const Value& patch) {
520 for (std::size_t i = 0; i < patch.items.size(); ++i) {
521 int position = target.find(patch.keys[i]);
522 if (position < 0) {
523 target.keys.push_back(patch.keys[i]);
524 target.items.push_back(patch.items[i]);
525 } else if (target.items[position].kind == Kind::Object && patch.items[i].kind == Kind::Object) {
526 merge(target.items[position], patch.items[i]);
527 } else target.items[position] = patch.items[i];
528 }
529}

531bool filter_holds(const std::string& op, const Value* value, const Value& reference) {
532 if (op == "EXISTS") return value != nullptr;
533 if (op == "==" || op == "!=") {
534 bool same = value && equal(*value, reference);
535 return op == "==" ? same : !same;
536 }
537 if (!value || value->kind != reference.kind || (value->kind != Kind::Number && value->kind != Kind::String)) return false;
538 int comparison = value->kind == Kind::Number ? (value->number < reference.number ? -1 : value->number > reference.number ? 1 : 0)
539 : value->text.compare(reference.text);
540 if (op == "<") return comparison < 0;
541 if (op == "<=") return comparison <= 0;
542 if (op == ">") return comparison > 0;
543 return comparison >= 0;
544}

546int rank(const Value* value) {
547 if (!value) return 3;
548 if (value->kind == Kind::Null) return 0;
549 if (value->kind == Kind::Number) return 1;
550 if (value->kind == Kind::String) return 2;
551 return 3;
552}

554bool less(const Value* a, const Value* b) {
555 int ar = rank(a), br = rank(b);
556 if (ar != br) return ar < br;
557 if (ar == 1) return a->number < b->number;
558 if (ar == 2) return a->text < b->text;
559 return false;
560}

562std::vector<std::string> split_arguments(const std::optional<std::string>& arguments, std::size_t maximum) {
563 std::vector<std::string> result;
564 if (!arguments) return result;
565 std::size_t position = 0;
566 while (result.size() + 1 < maximum) {
567 std::size_t space = arguments->find(' ', position);
568 if (space == std::string::npos) break;
569 result.push_back(arguments->substr(position, space - position));
570 position = space + 1;
571 }
572 result.push_back(arguments->substr(position));
573 return result;
574}

576std::vector<std::string> words(const std::optional<std::string>& arguments, std::size_t minimum, std::size_t maximum) {
577 std::vector<std::string> result = split_arguments(arguments, std::string::npos);
578 if (result.size() < minimum || result.size() > maximum) throw std::runtime_error("bad arguments");
579 return result;
580}

582class QueryTool {
583 Value document;
584 std::vector<Value> history;
585
586 void commit(Value value) {
587 history.push_back(std::move(document));
588 document = std::move(value);
589 }
590 void flatten(const Match& match) const {
591 if (match.value->items.empty()) {
592 std::cout << match.location() << " = " << compact(*match.value) << '\n';
593 } else {
594 for (int i = 0; i < static_cast<int>(match.value->items.size()); ++i) flatten(match.child(i));
595 }

597 void modify(const std::string& command, const std::optional<std::string>& arguments) {
598 std::vector<std::string> args;
599 if (command == "SET" || command == "MERGE" || command == "INSERT") {
600 std::size_t expected = command == "INSERT" ? 3 : 2;
601 args = split_arguments(arguments, expected);
602 if (args.size() != expected) throw std::runtime_error("bad arguments");
603 } else args = words(arguments, command == "RENAME" ? 2 : 1, command == "SORT" ? 3 : command == "RENAME" ? 2 : 1);
604 bool descending = false;
605 std::string field_text = "@";
606 if (command == "SORT" && args.size() > 1) {
607 descending = args.back() == "DESC";
608 std::size_t remaining = args.size() - 1 - (descending ? 1 : 0);
609 if (remaining > 1 || (remaining == 1 && args[1] == "DESC")) throw std::runtime_error("bad arguments");
610 if (remaining) field_text = args[1];
611 }
612 std::vector<Selector> path = PathReader(args[0], "bad path").path();
613 std::vector<Selector> field;
614 std::string new_key;
615 int insertion = 0;
616 Value value;
617 if (command == "INSERT") {
618 if (!integer_notation(args[1])) throw std::runtime_error("bad index");
619 insertion = std::stoi(args[1]);
620 }
621 if (command == "SET" || command == "MERGE" || command == "INSERT") value = JsonReader(args.back()).parse();
622 if (command == "SORT") field = PathReader(field_text, "bad field").path(true);
623 if (command == "RENAME") new_key = PathReader(args[1], "bad key").key();
624 if (command == "DELETE" && path.empty()) throw std::runtime_error("cannot delete root");
625 if (command == "MERGE" && value.kind != Kind::Object) throw std::runtime_error("merge value is not an object");
626 if (command == "SET" && path.empty()) {
627 commit(std::move(value));
628 std::cout << "SET 1\n";
629 return;
630 }
631 std::vector<Match> matches;
632 if (command == "SET") {
633 Selector last = path.back();
634 path.pop_back();
635 for (const Match& parent : evaluate(document, path)) {
636 if (!last.descendant && last.kind == SelectorKind::Key && parent.value->kind == Kind::Object) {
637 int position = parent.value->find(last.key);
638 if (position < 0) position = static_cast<int>(parent.value->items.size());
639 Match target{nullptr, parent.steps};
640 target.steps.push_back({last.key, position});
641 matches.push_back(std::move(target));
642 } else {
643 std::vector<Match> selected = apply(last, {parent});
644 matches.insert(matches.end(), selected.begin(), selected.end());
645 }
646 }
647 } else matches = evaluate(document, path);
648 if (command == "INSERT" || command == "SORT") require_kind(matches, Kind::Array, "not an array");
649 if (command == "MERGE") require_kind(matches, Kind::Object, "not an object");
650 if (command == "INSERT") {
651 for (const Match& match : matches) {
652 int count = static_cast<int>(match.value->items.size());
653 if (insertion < -count - 1 || insertion > count)
654 throw std::runtime_error("index out of range at " + match.location());
655 }
656 }
657 if (command == "RENAME") {
658 for (const Match& match : matches)
659 if (match.steps.empty() || !match.steps.back().key)
660 throw std::runtime_error("not a member at " + match.location());
661 }
662 std::vector<Match> selected = targets(std::move(matches));
663 Value updated = document;
664 for (const Match& target : selected) {
665 if (command == "SET" || command == "DELETE" || command == "RENAME") {
666 Value& parent = resolve(updated, target, true);
667 int position = target.steps.back().position;
668 if (command == "SET") {
669 if (position == static_cast<int>(parent.items.size())) {
670 parent.keys.push_back(*target.steps.back().key);
671 parent.items.push_back(value);
672 } else parent.items[position] = value;
673 } else if (command == "DELETE") {
674 parent.items.erase(parent.items.begin() + position);
675 if (parent.kind == Kind::Object) parent.keys.erase(parent.keys.begin() + position);
676 } else {
677 if (parent.keys[position] != new_key && parent.find(new_key) >= 0)
678 throw std::runtime_error("key exists at " + target.location());
679 parent.keys[position] = new_key;
680 }
681 } else {
682 Value& array = resolve(updated, target);
683 if (command == "INSERT") {
684 int position = insertion < 0 ? static_cast<int>(array.items.size()) + 1 + insertion : insertion;
685 array.items.insert(array.items.begin() + position, value);
686 } else if (command == "MERGE") merge(array, value);
687 else {
688 std::stable_sort(array.items.begin(), array.items.end(), [&](const Value& a, const Value& b) {
689 const Value* av = field_value(a, field), *bv = field_value(b, field);
690 return descending ? less(bv, av) : less(av, bv);
691 });
692 }
693 }
694 }
695 commit(std::move(updated));
696 std::cout << command << ' ' << selected.size() << '\n';
697 }
698
699 void execute(const std::string& command, const std::optional<std::string>& arguments) {
700 static const std::set<std::string> modifying{"SET", "DELETE", "INSERT", "MERGE", "SORT", "RENAME"};
701 static const std::set<std::string> ordinary{"GET", "COMPACT", "PRETTY", "TYPE", "LENGTH", "KEYS", "FLATTEN",
702 "COUNT", "SUM", "MIN", "MAX", "AVG"};
703 if (modifying.count(command)) {
704 modify(command, arguments);
705 return;
706 }
707 if (command == "UNDO") {
708 words(arguments, 0, 0);
709 if (history.empty()) throw std::runtime_error("nothing to undo");
710 document = std::move(history.back());
711 history.pop_back();
712 std::cout << "UNDO " << history.size() << '\n';
713 return;
714 }
715 if (command == "FILTER") {
716 std::vector<std::string> args = split_arguments(arguments, 4);
717 if (args.size() < 3 || (args.size() == 4) == (args[2] == "EXISTS")) throw std::runtime_error("bad arguments");
718 std::vector<Selector> path = PathReader(args[0], "bad path").path();
719 std::vector<Selector> field = PathReader(args[1], "bad field").path(true);
720 static const std::set<std::string> operators{"==", "!=", "<", "<=", ">", ">=", "EXISTS"};
721 if (!operators.count(args[2])) throw std::runtime_error("bad operator");
722 Value reference;
723 if (args.size() == 4) reference = JsonReader(args[3]).parse();
724 std::vector<Match> matches = evaluate(document, path);
725 require_kind(matches, Kind::Array, "not an array");
726 bool found = false;
727 for (const Match& match : matches) {
728 for (int i = 0; i < static_cast<int>(match.value->items.size()); ++i) {
729 Match element = match.child(i);
730 if (!filter_holds(args[2], field_value(*element.value, field), reference)) continue;
731 std::cout << element.location() << " = " << compact(*element.value) << '\n';
732 found = true;
733 }
734 }
735 if (!found) std::cout << "no match\n";
736 return;
737 }
738 if (!ordinary.count(command)) throw std::runtime_error("unknown command '" + command + "'");
739 bool optional = command == "COMPACT" || command == "PRETTY" || command == "FLATTEN";
740 std::vector<std::string> args = words(arguments, optional ? 0 : 1, 1);
741 std::vector<Match> matches = evaluate(document, PathReader(args.empty() ? "$" : args[0], "bad path").path());
742 if (command == "COUNT") {
743 std::cout << "COUNT " << matches.size() << '\n';
744 return;
745 }
746 if (command == "SUM" || command == "MIN" || command == "MAX" || command == "AVG") {
747 std::vector<std::int64_t> numbers;
748 std::int64_t sum = 0;
749 for (const Match& match : matches) if (match.value->kind == Kind::Number) {
750 numbers.push_back(match.value->number);
751 sum += match.value->number;
752 }
753 std::cout << command << ' ';
754 if (command == "SUM") std::cout << sum;
755 else if (numbers.empty()) std::cout << "null";
756 else if (command == "MIN") std::cout << *std::min_element(numbers.begin(), numbers.end());
757 else if (command == "MAX") std::cout << *std::max_element(numbers.begin(), numbers.end());
758 else {
759 std::int64_t count = static_cast<std::int64_t>(numbers.size());
760 std::int64_t q = ((sum < 0 ? -sum : sum) * 200 + count) / (2 * count);
761 if (sum < 0 && q) std::cout << '-';
762 std::cout << q / 100 << '.' << std::setw(2) << std::setfill('0') << q % 100 << std::setfill(' ');
763 }
764 std::cout << '\n';
765 return;
766 }
767 if (command == "KEYS") require_kind(matches, Kind::Object, "not an object");
768 if (command == "LENGTH") for (const Match& match : matches) {
769 Kind kind = match.value->kind;
770 if (kind != Kind::Object && kind != Kind::Array && kind != Kind::String)
771 throw std::runtime_error("no length at " + match.location());
772 }
773 if (matches.empty()) std::cout << "no match\n";
774 static const std::vector<std::string> type_names{"object", "array", "string", "number", "boolean", "null"};
775 for (const Match& match : matches) {
776 if (command == "COMPACT") std::cout << compact(*match.value) << '\n';
777 else if (command == "PRETTY") std::cout << pretty(*match.value) << '\n';
778 else if (command == "FLATTEN") flatten(match);
779 else if (command == "GET") std::cout << match.location() << " = " << compact(*match.value) << '\n';
780 else if (command == "TYPE") std::cout << match.location() << ' ' << type_names[static_cast<int>(match.value->kind)] << '\n';
781 else if (command == "LENGTH")
782 std::cout << match.location() << ' ' << (match.value->kind == Kind::String ? match.value->text.size() : match.value->items.size()) << '\n';
783 else if (command == "KEYS") {
784 std::cout << match.location() << " [";
785 for (std::size_t i = 0; i < match.value->keys.size(); ++i) {
786 if (i) std::cout << ',';
787 std::cout << quote(match.value->keys[i]);
788 }
789 std::cout << "]\n";
790 }
791 }
792 }
793public:
794 void load(const std::vector<std::string>& lines) {
795 std::string text;
796 for (std::size_t i = 0; i < lines.size(); ++i) {
797 if (i) text += '\n';
798 text += lines[i];
799 }
800 try {
801 Value value = JsonReader(text).parse();
802 int count = value.count(), depth = value.depth();
803 commit(std::move(value));
804 std::cout << "LOADED nodes=" << count << " depth=" << depth << '\n';
805 } catch (const JsonError& error) {
806 int line = 1, column = 1;
807 for (std::size_t i = 0; i < error.position; ++i) {
808 if (text[i] == '\n') {
809 ++line;
810 column = 1;
811 }
812 else ++column;
813 }
814 std::cout << "PARSE ERROR line " << line << " col " << column << ": " << error.what() << '\n';
815 }
816 }
817 void command(const std::string& line) {
818 std::size_t space = line.find(' ');
819 std::optional<std::string> arguments;
820 if (space != std::string::npos) arguments = line.substr(space + 1);
821 try { execute(line.substr(0, space), arguments); }
822 catch (const JsonError& error) { std::cout << "PARSE ERROR col " << error.position + 1 << ": " << error.what() << '\n'; }
823 catch (const std::runtime_error& error) { std::cout << "ERROR " << error.what() << '\n'; }
824 }
825};

829 std::string line;
830 while (std::getline(std::cin, line) && line != "%%") lines.push_back(line);
831 return lines;
832}
