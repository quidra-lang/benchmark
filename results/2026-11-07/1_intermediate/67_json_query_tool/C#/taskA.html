25 static bool Matches(string text, string pattern) => Regex.IsMatch(text, @"\A(?:" + pattern + @")\z");

27 static bool Letter(char c) => c is >= 'A' and <= 'Z' or >= 'a' and <= 'z';
28
29 static bool Digit(char c) => c is >= '0' and <= '9';

31 static bool IntegerNotation(string text) => Matches(text, "0|-?[1-9][0-9]{0,8}");

33 static string Quote(string text, char delimiter)
34 {
35 StringBuilder output = new();
36 output.Append(delimiter);
37 foreach (char c in text) if (c == delimiter || c == '\\') output.Append('\\').Append(c);
38 else output.Append(c switch { '\b' => "\\b", '\f' => "\\f", '\n' => "\\n", '\r' => "\\r", '\t' => "\\t", _ => c < 32 || c == 127 ? "\\u" + ((int)c).ToString("x4", CultureInfo.InvariantCulture) : c.ToString() });
39 return output.Append(delimiter).ToString();
40 }

42 class Parser(string text)
43 {
44 protected readonly string Text = text;
45 public int Position;
46 public char Peek => Position == Text.Length ? '\0' : Text[Position];
47 void Fail(string reason) => throw new ParseError(Position, reason);
48 void Whitespace()
49 {
50 while (Peek is ' ' or '\n') Position++;
51 }
52 void NeedCharacter()
53 {
54 if (Position == Text.Length) Fail("unexpected end of input");
55 }
56 public string String(char delimiter)
57 {
58 Position++;
59 StringBuilder output = new();
60 while (true)
61 {
62 if (Position == Text.Length || Peek == '\n') Fail("unterminated string");
63 char c = Peek;
64 if (c == delimiter)
65 {
66 Position++;
67 return output.ToString();
68 }
69 if (c != '\\')
70 {
71 output.Append(c);
72 Position++;
73 continue;
74 }
75 int start = Position++;
76 if (Position == Text.Length || Peek == '\n') Fail("unterminated string");
77 char escaped = Peek;
78 if (escaped == 'u')
79 {
80 if (Position + 4 >= Text.Length || !Matches(Text.Substring(Position + 1, 4), "[0-9A-Fa-f]{4}")) throw new ParseError(start, "bad unicode escape");
81 int value = int.Parse(Text.AsSpan(Position + 1, 4), NumberStyles.HexNumber, CultureInfo.InvariantCulture);
82 if (value > 127) throw new ParseError(start, "non-ascii escape");
83 output.Append((char)value);
84 Position += 5;
85 }
86 else
87 {
88 char value = escaped switch { '\\' or '/' => escaped, 'b' => '\b', 'f' => '\f', 'n' => '\n', 'r' => '\r', 't' => '\t', _ => escaped == delimiter ? delimiter : '\0' };
89 if (value == '\0') throw new ParseError(start, $"invalid escape '\\{escaped}'");
90 output.Append(value);
91 Position++;
92 }
93 }
94 }
95 object? Value()
96 {
97 Whitespace();
98 NeedCharacter();
99 char c = Peek;
100 if (c == '"') return String('"');
101 if (c == '{')
102 {
103 Position++;
104 Whitespace();
105 JsonObject map = new(StringComparer.Ordinal);
106 if (Peek == '}')
107 {
108 Position++;
109 return map;
110 }
111 while (true)
112 {
113 Whitespace();
114 NeedCharacter();
115 if (Peek != '"') Fail("expected key");
116 int start = Position;
117 string key = String('"');
118 if (map.ContainsKey(key)) throw new ParseError(start, "duplicate key " + Quote(key, '"'));
119 Whitespace();
120 NeedCharacter();
121 if (Peek != ':') Fail("expected ':'");
122 Position++;
123 map.Add(key, Value());
124 Whitespace();
125 if (Peek == '}')
126 {
127 Position++;
128 return map;
129 }
130 NeedCharacter();
131 if (Peek != ',') Fail("expected ',' or '}'");
132 Position++;
133 Whitespace();
134 if (Peek == '}') Fail("trailing comma");
135 }
136 }
137 if (c == '[')
138 {
139 Position++;
140 Whitespace();
141 List<object?> list = [];
142 if (Peek == ']')
143 {
144 Position++;
145 return list;
146 }
147 while (true)
148 {
149 list.Add(Value());
150 Whitespace();
151 if (Peek == ']')
152 {
153 Position++;
154 return list;
155 }
156 NeedCharacter();
157 if (Peek != ',') Fail("expected ',' or ']'");
158 Position++;
159 Whitespace();
160 if (Peek == ']') Fail("trailing comma");
161 }
162 }
163 if (c == '-' || Digit(c))
164 {
165 int start = Position;
166 if (c == '-') Position++;
167 int digits = Position;
168 while (Digit(Peek)) Position++;
169 if (Position == digits) throw new ParseError(start, "invalid number");
170 if (Position - digits >= 2 && Text[digits] == '0') throw new ParseError(digits, "leading zero");
171 if (Peek == '.') Fail("fraction not allowed");
172 if (Peek is 'e' or 'E') Fail("exponent not allowed");
173 if (Position - digits >= 10) throw new ParseError(start, "number out of range");
174 return int.Parse(Text[start..Position], CultureInfo.InvariantCulture);
175 }
176 if (Letter(c))
177 {
178 int start = Position;
179 while (Letter(Peek)) Position++;
180 string word = Text[start..Position];
181 return word switch { "true" => true, "false" => false, "null" => null, _ => throw new ParseError(start, $"invalid literal '{word}'") };
182 }
183 Fail($"unexpected character '{c}'");
184 return null;
185 }
186 public object? Whole()
187 {
188 object? value = Value();
189 Whitespace();
190 if (Position != Text.Length) Fail("trailing characters");
191 return value;
192 }
193 }

196 {
197 if (value is JsonObject map)
198 {
199 JsonObject result = new(StringComparer.Ordinal);
200 foreach (var entry in map) result.Add(entry.Key, Copy(entry.Value));

206 static string Compact(object? value) => value switch { null => "null", string text => Quote(text, '"'), bool b => b ? "true" : "false", int n => n.ToString(CultureInfo.InvariantCulture), JsonObject map => "{" + string.Join(",", map.Select(entry => Quote(entry.Key, '"') + ":" + Compact(entry.Value))) + "}", List<object?> list => "[" + string.Join(",", list.Select(Compact)) + "]", _ => throw new InvalidOperationException() };

208 static string Pretty(object? value, string indent)
209 {
210 if (value is JsonObject { Count: > 0 } || value is List<object?> { Count: > 0 })
211 {
212 string child = indent + " ";
213 List<string> parts = [];
214 if (value is JsonObject map) foreach (var entry in map) parts.Add(child + Quote(entry.Key, '"') + ": " + Pretty(entry.Value, child));
215 else foreach (object? item in (List<object?>)value) parts.Add(child + Pretty(item, child));
216 return (value is JsonObject ? "{" : "[") + "\n" + string.Join(",\n", parts) + "\n" + indent + (value is JsonObject ? "}" : "]");
217 }
218 return Compact(value);
219 }

225 class PathReader(string text) : Parser(text)
226 {
227 string IdentifierName()
228 {
229 int start = Position;
230 if (!Letter(Peek) && Peek != '_') throw new Problem("");
231 Position++;
232 while (Letter(Peek) || Digit(Peek) || Peek == '_') Position++;
233 return Text[start..Position];
234 }
235 public List<Selector> Read(bool field)
236 {
237 if (Peek != (field ? '@' : '$')) throw new Problem("");
238 Position++;
239 List<Selector> selectors = [];
240 while (Position < Text.Length)
241 {
242 bool descendant = false;
243 if (Peek == '.')
244 {
245 Position++;
246 if (Peek == '.')
247 {
248 if (field) throw new Problem("");
249 descendant = true;
250 Position++;
251 }
252 if (Peek == '*')
253 {
254 if (field) throw new Problem("");
255 Position++;
256 selectors.Add(new('A', null, null, null, descendant));
257 continue;
258 }
259 if (!(descendant && Peek == '['))
260 {
261 selectors.Add(new('K', IdentifierName(), null, null, descendant));
262 continue;
263 }
264 }
265 else if (Peek != '[') throw new Problem("");
266 if (Peek != '[') throw new Problem("");
267 Position++;
268 if (Peek == '\'')
269 {
270 string key = String('\'');
271 if (Peek != ']') throw new Problem("");
272 Position++;
273 selectors.Add(new('K', key, null, null, descendant));
274 continue;
275 }
276 int end = Text.IndexOf(']', Position);
277 if (end < 0) throw new Problem("");
278 string content = Text[Position..end];
279 Position = end + 1;
280 if (content == "*" && !field) selectors.Add(new('A', null, null, null, descendant));
281 else if (IntegerNotation(content)) selectors.Add(new('I', null, int.Parse(content), null, descendant));
282 else if (!field && content.Contains(':'))
283 {
284 string[] parts = content.Split(':');
285 if (parts.Length != 2 || parts[0].Length > 0 && !IntegerNotation(parts[0]) || parts[1].Length > 0 && !IntegerNotation(parts[1])) throw new Problem("");
286 selectors.Add(new('S', null, parts[0].Length == 0 ? null : int.Parse(parts[0]), parts[1].Length == 0 ? null : int.Parse(parts[1]), descendant));
287 }
288 else throw new Problem("");
289 }
290 return selectors;
291 }
292 }

306 static string Key(string text)
307 {
308 if (Matches(text, Identifier)) return text;
309 try
310 {
311 Parser reader = new(text);
312 if (reader.Peek != '\'') throw new Problem("");
313 string key = reader.String('\'');
314 if (reader.Position != text.Length) throw new Problem("");
315 return key;
316 }
317 catch (Exception error) when (error is Problem or ParseError)

319 throw new Problem("ERROR bad key");
320 }

327 static Match Child(Match parent, object key, int position)
328 {
329 List<Step> location = [.. parent.Location, new(key, position)];
330 return new(key is string name ? ((JsonObject)parent.Value!)[name] : ((List<object?>)parent.Value!)[(int)key], location);

333 static List<Match> Children(Match parent)
334 {
335 List<Match> result = [];
336 if (parent.Value is JsonObject map)
337 {
338 int i = 0;
339 foreach (string key in map.Keys) result.Add(Child(parent, key, i++));
340 }
341 else if (parent.Value is List<object?> list) for (int i = 0; i < list.Count; i++) result.Add(Child(parent, i, i));
342 return result;
343 }

345 static void Preorder(Match match, List<Match> result)
346 {
347 result.Add(match);
348 foreach (Match child in Children(match)) Preorder(child, result);
349 }

351 static List<Match> Select(Match match, Selector selector)
352 {
353 List<Match> result = [];
354 if (selector.Kind == 'K' && match.Value is JsonObject map && map.ContainsKey(selector.Key!)) result.Add(Child(match, selector.Key!, map.Keys.ToList().IndexOf(selector.Key!)));
355 else if (selector.Kind == 'A') result.AddRange(Children(match));
356 else if (match.Value is List<object?> list && selector.Kind is 'I' or 'S')
357 {
358 int count = list.Count;
359 if (selector.Kind == 'I')
360 {
361 int i = selector.First!.Value < 0 ? count + selector.First.Value : selector.First.Value;
362 if (i >= 0 && i < count) result.Add(Child(match, i, i));
363 }
364 else
365 {
366 int first = selector.First ?? 0, last = selector.Last ?? count;
367 if (first < 0) first += count;
368 if (last < 0) last += count;
369 first = Math.Clamp(first, 0, count);
370 last = Math.Clamp(last, 0, count);
371 for (int i = first; i < last; i++) result.Add(Child(match, i, i));
372 }
373 }
374 return result;
375 }

377 static List<Match> Apply(List<Match> matches, Selector selector)
378 {
379 List<Match> result = [];
380 foreach (Match match in matches)
381 {
382 List<Match> bases = [];
383 if (selector.Descendant) Preorder(match, bases);
384 else bases.Add(match);
385 foreach (Match basis in bases) result.AddRange(Select(basis, selector));
386 }
387 return result;
388 }

390 static List<Match> Evaluate(IEnumerable<Selector> selectors)
391 {
392 List<Match> matches = [new(document, [])];
393 foreach (Selector selector in selectors) matches = Apply(matches, selector);
394 return matches;

398 {
399 StringBuilder output = new("$");
400 foreach (Step step in steps) if (step.Key is string key) output.Append(Matches(key, Identifier) ? "." + key : "[" + Quote(key, '\'') + "]");

423 static bool Equal(object? left, object? right)
424 {
425 if (left is JsonObject a && right is JsonObject b) return a.Count == b.Count && a.All(entry => b.ContainsKey(entry.Key) && Equal(entry.Value, b[entry.Key]));
426 if (left is List<object?> x && right is List<object?> y) return x.Count == y.Count && x.Zip(y).All(pair => Equal(pair.First, pair.Second));
427 return Equals(left, right);
428 }

430 static int Rank(object? value) => value switch { null => 0, int => 1, string => 2, _ => 3 };
431
432 static int Compare(object? left, object? right)
433 {
434 int order = Rank(left).CompareTo(Rank(right));
435 if (order != 0) return order;
436 return Rank(left) switch { 1 => ((int)left!).CompareTo((int)right!), 2 => StringComparer.Ordinal.Compare((string)left!, (string)right!), _ => 0 };
437 }

439 static void Merge(JsonObject target, JsonObject source)
440 {
441 foreach (var entry in source) if (target.TryGetValue(entry.Key, out object? old) && old is JsonObject oldMap && entry.Value is JsonObject newMap) Merge(oldMap, newMap);
442 else target[entry.Key] = Copy(entry.Value);
443 }

445 static int CompareLocations(Match left, Match right)
446 {
447 for (int i = 0; i < Math.Min(left.Location.Count, right.Location.Count); i++)
448 {
449 int order = left.Location[i].Position.CompareTo(right.Location[i].Position);
450 if (order != 0) return order;
451 }
452 return left.Location.Count.CompareTo(right.Location.Count);
453 }

455 static List<Match> Targets(List<Match> matches)
456 {
457 Dictionary<string, Match> unique = new(StringComparer.Ordinal);
458 foreach (Match match in matches) unique.TryAdd(Location(match.Location), match);
459 return unique.Values.OrderByDescending(match => match, Comparer<Match>.Create(CompareLocations)).ToList();

484 static List<string> Execute(string line)
485 {
486 int space = line.IndexOf(' ');
487 string command = space < 0 ? line : line[..space];
488 string? argument = space < 0 ? null : line[(space + 1)..];
489 Require(Commands.Contains(command), $"ERROR unknown command '{command}'");
490 int limit = command is "SET" or "MERGE" ? 2 : command == "INSERT" ? 3 : command == "FILTER" ? 4 : int.MaxValue;
491 string[] args = argument is null ? [] : argument.Split(' ', limit);
492 bool shape = command switch { "SET" or "MERGE" or "RENAME" => args.Length == 2, "INSERT" => args.Length == 3, "FILTER" => args.Length >= 3 && (args[2] == "EXISTS" ? args.Length == 3 : args.Length == 4), "COMPACT" or "PRETTY" or "FLATTEN" => args.Length <= 1, "UNDO" => args.Length == 0, "SORT" => args.Length is >= 1 and <= 3 && (args.Length < 3 || args[1] != "DESC" && args[2] == "DESC"), _ => args.Length == 1 };
493 Require(shape, "ERROR bad arguments");
494 if (command == "UNDO")
495 {
496 Require(History.Count > 0, "ERROR nothing to undo");
497 document = History[^1];
498 History.RemoveAt(History.Count - 1);
499 return [$"UNDO {History.Count}"];
500 }
501 var selectors = Path(args.Length == 0 ? "$" : args[0], false);
502 List<Selector> fields = [];
503 object? value = null;
504 int index = 0;
505 string? rename = null, op = null;
506 if (command == "FILTER")
507 {
508 fields = Path(args[1], true);
509 op = args[2];
510 Require(op is "==" or "!=" or "<" or "<=" or ">" or ">=" or "EXISTS", "ERROR bad operator");
511 if (op != "EXISTS") value = JsonValue(args[3]);
512 }
513 else if (command == "SORT") fields = Path(args.Length >= 2 && args[1] != "DESC" ? args[1] : "@", true);
514 else if (command == "RENAME") rename = Key(args[1]);
515 else if (command == "INSERT")
516 {
517 Require(IntegerNotation(args[1]), "ERROR bad index");
518 index = int.Parse(args[1]);
519 value = JsonValue(args[2]);
520 }
521 else if (command is "SET" or "MERGE") value = JsonValue(args[1]);
522 if (command == "MERGE") Require(value is JsonObject, "ERROR merge value is not an object");
523 if (command == "DELETE") Require(selectors.Count > 0, "ERROR cannot delete root");
524 List<Match> matches = Evaluate(selectors);
525 if (command is "FILTER" or "INSERT" or "SORT") RequireType(matches, "array");
526 if (command is "KEYS" or "MERGE") RequireType(matches, "object");
527 if (command == "LENGTH") foreach (Match match in matches) Require(match.Value is JsonObject or List<object?> or string, "ERROR no length at " + Location(match.Location));
528 if (command == "INSERT") foreach (Match match in matches) Require(index >= -((List<object?>)match.Value!).Count - 1 && index <= ((List<object?>)match.Value!).Count, "ERROR index out of range at " + Location(match.Location));
529 if (command == "RENAME") foreach (Match match in matches) Require(match.Location.Count > 0 && match.Location[^1].Key is string, "ERROR not a member at " + Location(match.Location));
530 List<string> output = [];
531 if (command is "SET" or "DELETE" or "INSERT" or "MERGE" or "SORT" or "RENAME")
532 {
533 if (command == "SET" && selectors.Count > 0)
534 {
535 Selector last = selectors[^1];
536 if (last.Kind == 'K' && !last.Descendant)
537 {
538 matches = [];
539 foreach (Match parent in Evaluate(selectors.Take(selectors.Count - 1))) if (parent.Value is JsonObject map)
540 {
541 int position = map.Keys.ToList().IndexOf(last.Key!);
542 matches.Add(new(null, [.. parent.Location, new Step(last.Key!, position < 0 ? map.Count : position)]));
543 }
544 }
545 }
546 var targets = Targets(matches);
547 object? changed = Copy(document);
548 foreach (Match target in targets)
549 {
550 var place = target.Location;
551 object? current = command == "SET" ? null : At(changed, place);
552 if (command == "SET")
553 {
554 if (place.Count == 0) changed = Copy(value);
555 else
556 {
557 object? parent = At(changed, place.Take(place.Count - 1));
558 if (place[^1].Key is string key) ((JsonObject)parent!)[key] = Copy(value);
559 else ((List<object?>)parent!)[(int)place[^1].Key] = Copy(value);
560 }
561 }
562 else if (command == "DELETE")
563 {
564 object? parent = At(changed, place.Take(place.Count - 1));
565 if (place[^1].Key is string key) ((JsonObject)parent!).Remove(key);
566 else ((List<object?>)parent!).RemoveAt((int)place[^1].Key);
567 }
568 else if (command == "INSERT")
569 {
570 var list = (List<object?>)current!;
571 list.Insert(index < 0 ? list.Count + 1 + index : index, Copy(value));
572 }
573 else if (command == "MERGE") Merge((JsonObject)current!, (JsonObject)value!);
574 else if (command == "SORT")
575 {
576 bool descending = args[^1] == "DESC";
577 var list = (List<object?>)current!;
578 var sorted = list.OrderBy(item => Field(item, fields), Comparer<object?>.Create((left, right) => (descending ? -1 : 1) * Compare(left, right))).ToList();
579 list.Clear();
580 list.AddRange(sorted);
581 }
582 else
583 {
584 var parent = (JsonObject)At(changed, place.Take(place.Count - 1))!;
585 string old = (string)place[^1].Key;
586 Require(old == rename || !parent.ContainsKey(rename!), "ERROR key exists at " + Location(place));
587 var entries = parent.Select(entry => KeyValuePair.Create(entry.Key == old ? rename! : entry.Key, entry.Value)).ToList();
588 parent.Clear();
589 foreach (var entry in entries) parent.Add(entry.Key, entry.Value);
590 }
591 }
592 History.Add(Copy(document));
593 document = changed;
594 return [$"{command} {targets.Count}"];
595 }
596 if (command == "COUNT") return [$"COUNT {matches.Count}"];
597 if (command is "SUM" or "MIN" or "MAX" or "AVG")
598 {
599 long total = 0;
600 int? minimum = null, maximum = null;
601 int count = 0;
602 foreach (Match match in matches) if (match.Value is int number)
603 {
604 total += number;
605 count++;
606 minimum = minimum is null ? number : Math.Min(minimum.Value, number);
607 maximum = maximum is null ? number : Math.Max(maximum.Value, number);
608 }
609 string result = command == "SUM" ? total.ToString(CultureInfo.InvariantCulture) : command == "MIN" ? minimum?.ToString(CultureInfo.InvariantCulture) ?? "null" : command == "MAX" ? maximum?.ToString(CultureInfo.InvariantCulture) ?? "null" : "null";
610 if (command == "AVG" && count > 0)
611 {
612 long q = (Math.Abs(total) * 200 + count) / (2L * count);
613 result = (total < 0 && q > 0 ? "-" : "") + (q / 100).ToString(CultureInfo.InvariantCulture) + "." + (q % 100).ToString("D2", CultureInfo.InvariantCulture);
614 }
615 return [$"{command} {result}"];
616 }
617 if (command == "FILTER")
618 {
619 foreach (Match match in matches) foreach (Match child in Children(match))
620 {
621 object? f = Field(child.Value, fields);
622 bool equality = f != Missing && Equal(f, value), pass;
623 if (op == "EXISTS") pass = f != Missing;
624 else if (op is "==" or "!=") pass = (op == "==") == equality;
625 else
626 {
627 bool comparable = f is int && value is int || f is string && value is string;
628 int order = comparable ? Compare(f, value) : 0;
629 pass = comparable && (op switch { "<" => order < 0, "<=" => order <= 0, ">" => order > 0, _ => order >= 0 });
630 }
631 if (pass) output.Add(Location(child.Location) + " = " + Compact(child.Value));
632 }
633 }
634 else foreach (Match match in matches)
635 {
636 string place = Location(match.Location);
637 object? item = match.Value;
638 switch (command)
639 {
640 case "GET":
641 output.Add(place + " = " + Compact(item));
642 break;
643 case "COMPACT":
644 output.Add(Compact(item));
645 break;
646 case "PRETTY":
647 output.Add(Pretty(item, ""));
648 break;
649 case "TYPE":
650 output.Add(place + " " + TypeName(item));
651 break;
652 case "LENGTH":
653 output.Add(place + " " + (item is JsonObject map ? map.Count : item is List<object?> list ? list.Count : ((string)item!).Length));
654 break;
655 case "KEYS": output.Add(place + " " + Compact(((JsonObject)item!).Keys.Select(key => (object?)key).ToList()));
656 break;
657 case "FLATTEN":
658 List<Match> descendants = [];
659 Preorder(match, descendants);
660 foreach (Match leaf in descendants) if (Children(leaf).Count == 0) output.Add(Location(leaf.Location) + " = " + Compact(leaf.Value));
661 break;
662 }
663 }
664 return output.Count == 0 ? ["no match"] : output;
665 }

667 static (int Count, int Depth) Size(object? value)
668 {
669 int count = 1, depth = 1;
670 foreach (Match child in Children(new(value, [])))
671 {
672 var size = Size(child.Value);
673 count += size.Count;
674 depth = Math.Max(depth, size.Depth + 1);
675 }
676 return (count, depth);
677 }

679 static void Load(List<string> lines)
680 {
681 string text = string.Join("\n", lines);
682 try
683 {
684 object? value = new Parser(text).Whole();
685 History.Add(Copy(document));
686 document = value;
687 var size = Size(value);
688 Console.WriteLine($"LOADED nodes={size.Count} depth={size.Depth}");
689 }
690 catch (ParseError error)
691 {
692 int line = 1, column = 1;
693 for (int i = 0; i < error.Position; i++) if (text[i] == '\n')
694 {
695 line++;
696 column = 1;
697 }
698 else column++;
699 Console.WriteLine($"PARSE ERROR line {line} col {column}: {error.Message}");
700 }
701 }
