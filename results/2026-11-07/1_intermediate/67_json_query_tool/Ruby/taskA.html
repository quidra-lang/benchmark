9class StrictJson
10 attr_reader :position
11 def initialize(text)
12 @text = text
13 @position = 0
14 end
15 def fail_at(reason, position = @position)
16 raise JsonFault.new(reason, position)
17 end
18 def whitespace
19 @position += 1 while [' ', "\n"].include?(@text[@position])
20 end
21 def string
22 @position += 1
23 output = +''
24 loop do
25 char = @text[@position]
26 fail_at('unterminated string') if char.nil? || char == "\n"
27 if char == '"'
28 @position += 1
29 return output
30 end
31 if char == '\\'
32 start = @position
33 escaped = @text[@position + 1]
34 fail_at('unterminated string', start + 1) if escaped.nil? || escaped == "\n"
35 if escaped == 'u'
36 hex = @text[@position + 2, 4]
37 fail_at('bad unicode escape', start) unless hex && hex.match?(/\A[0-9A-Fa-f]{4}\z/)
38 fail_at('non-ascii escape', start) if hex.to_i(16) > 127
39 output << hex.to_i(16).chr
40 @position += 6
41 else
42 escapes = { '"' => '"', '\\' => '\\', '/' => '/', 'b' => "\b", 'f' => "\f", 'n' => "\n", 'r' => "\r", 't' => "\t" }
43 fail_at("invalid escape '\\#{escaped}'", start) unless escapes.key?(escaped)
44 output << escapes[escaped]
45 @position += 2
46 end
47 else
48 output << char
49 @position += 1
50 end
51 end
52 end
53 def value
54 whitespace
55 char = @text[@position]
56 fail_at('unexpected end of input') if char.nil?
57 if char == '{' || char == '['
58 object = char == '{'
59 result = object ? {} : []
60 ending = object ? '}' : ']'
61 @position += 1
62 whitespace
63 if @text[@position] == ending
64 @position += 1
65 return result
66 end
67 loop do
68 if object
69 whitespace
70 fail_at('unexpected end of input') if @text[@position].nil?
71 fail_at('expected key') unless @text[@position] == '"'
72 key_start = @position
73 key = string
74 fail_at('duplicate key ' + JsonTool.quote(key), key_start) if result.key?(key)
75 whitespace
76 fail_at('unexpected end of input') if @text[@position].nil?
77 fail_at("expected ':'") unless @text[@position] == ':'
78 @position += 1
79 result[key] = value
80 else
81 result << value
82 end
83 whitespace
84 if @text[@position] == ending
85 @position += 1
86 return result
87 end
88 fail_at('unexpected end of input') if @text[@position].nil?
89 fail_at("expected ',' or '#{ending}'") unless @text[@position] == ','
90 @position += 1
91 whitespace
92 fail_at('trailing comma') if @text[@position] == ending
93 end
94 elsif char == '"'
95 string
96 elsif char == '-' || char.match?(/[0-9]/)
97 start = @position
98 @position += 1 if char == '-'
99 digits_start = @position
100 @position += 1 while @text[@position]&.match?(/[0-9]/)
101 digits = @text[digits_start...@position]
102 fail_at('invalid number', start) if digits.empty?
103 fail_at('leading zero', digits_start) if digits.length > 1 && digits[0] == '0'
104 fail_at('fraction not allowed') if @text[@position] == '.'
105 fail_at('exponent not allowed') if ['e', 'E'].include?(@text[@position])
106 fail_at('number out of range', start) if digits.length >= 10
107 @text[start...@position].to_i
108 elsif char.match?(/[A-Za-z]/)
109 start = @position
110 @position += 1 while @text[@position]&.match?(/[A-Za-z]/)
111 word = @text[start...@position]
112 fail_at("invalid literal '#{word}'", start) unless %w[true false null].include?(word)
113 { 'true' => true, 'false' => false, 'null' => nil }[word]
114 else
115 fail_at("unexpected character '#{char}'")
116 end
117 end
118 def read
119 result = value
120 whitespace
121 fail_at('trailing characters') if @position < @text.length
122 result
123 end
124end

126class JsonTool
127 INTEGER = /\A(?:0|-?[1-9][0-9]{0,8})\z/
128 IDENTIFIER = /\A[A-Za-z_][A-Za-z0-9_]*\z/
129 MISSING = Object.new.freeze
130 COMMANDS = %w[LOAD GET COMPACT PRETTY TYPE LENGTH KEYS FLATTEN COUNT SUM MIN MAX AVG FILTER SET DELETE INSERT MERGE SORT RENAME UNDO].freeze
131 def self.quote(text, delimiter = '"')
132 escapes = { "\b" => '\\b', "\f" => '\\f', "\n" => '\\n', "\r" => '\\r', "\t" => '\\t' }
133 delimiter + text.each_char.map do |char|
134 if char == delimiter || char == '\\'
135 '\\' + char
136 elsif escapes.key?(char)
137 escapes[char]
138 elsif char.ord < 32 || char.ord == 127
139 format('\\u%04x', char.ord)
140 else
141 char
142 end
143 end.join + delimiter
144 end

146 @document = nil
147 @history = []

149 def copy(value)
150 Marshal.load(Marshal.dump(value))
151 end
152 def type(value)
153 case value
154 when Hash then 'object'
155 when Array then 'array'
156 when String then 'string'
157 when Integer then 'number'
158 when true, false then 'boolean'
159 else 'null'
160 end
161 end
162 def compact(value)
163 case value
164 when Hash then '{' + value.map { |key, item| self.class.quote(key) + ':' + compact(item) }.join(',') + '}'
165 when Array then '[' + value.map { |item| compact(item) }.join(',') + ']'
166 when String then self.class.quote(value)
167 when nil then 'null'
168 else value.to_s
169 end
170 end
171 def pretty(value, indent = '')
172 return compact(value) unless (value.is_a?(Hash) || value.is_a?(Array)) && !value.empty?
173 next_indent = indent + ' '
174 if value.is_a?(Hash)
175 items = value.map { |key, item| next_indent + self.class.quote(key) + ': ' + pretty(item, next_indent) }
176 "{\n" + items.join(",\n") + "\n#{indent}}"
177 else
178 "[\n" + value.map { |item| next_indent + pretty(item, next_indent) }.join(",\n") + "\n#{indent}]"
179 end
180 end
181 def quoted_key(text, position)
182 raise 'bad' unless text[position] == "'"
183 position += 1
184 output = +''
185 while position < text.length
186 char = text[position]
187 return [output, position + 1] if char == "'"
188 if char == '\\'
189 escaped = text[position + 1]
190 if escaped == 'u'
191 hex = text[position + 2, 4]
192 raise 'bad' unless hex && hex.match?(/\A[0-9a-fA-F]{4}\z/) && hex.to_i(16) <= 127
193 output << hex.to_i(16).chr
194 position += 6
195 else
196 escapes = { "'" => "'", '\\' => '\\', '/' => '/', 'b' => "\b", 'f' => "\f", 'n' => "\n", 'r' => "\r", 't' => "\t" }
197 raise 'bad' unless escapes.key?(escaped)
198 output << escapes[escaped]
199 position += 2
200 end
201 else
202 output << char
203 position += 1
204 end
205 end
206 raise 'bad'
207 end
208 def path(text, field = false)
209 begin
210 raise 'bad' unless text && text[0] == (field ? '@' : '$')
211 position = 1
212 selectors = []
213 while position < text.length
214 descendant = false
215 if text[position, 2] == '..' && !field
216 descendant = true
217 position += 2
218 elsif text[position] == '.'
219 position += 1
220 match = /\A(?:[A-Za-z_][A-Za-z0-9_]*|\*)/.match(text[position..])
221 raise 'bad' unless match && (!field || match[0] != '*')
222 word = match[0]
223 selectors << [false, word == '*' ? :all : :key, word]
224 position += word.length
225 next
226 elsif text[position] != '['
227 raise 'bad'
228 end
229 if text[position] == '['
230 position += 1
231 if text[position] == "'"
232 key, position = quoted_key(text, position)
233 kind = :key
234 else
235 finish = text.index(']', position)
236 raise 'bad' unless finish
237 word = text[position...finish]
238 position = finish
239 if INTEGER.match?(word)
240 kind = :index
241 key = word.to_i
242 elsif word == '*' && !field
243 kind = :all
244 key = nil
245 elsif !field && word.count(':') == 1 && word.split(':', -1).all? { |part| part.empty? || INTEGER.match?(part) }
246 kind = :slice
247 key = word.split(':', -1).map { |part| part.empty? ? nil : part.to_i }
248 else
249 raise 'bad'
250 end
251 end
252 raise 'bad' unless text[position] == ']'
253 position += 1
254 elsif descendant
255 match = /\A(?:[A-Za-z_][A-Za-z0-9_]*|\*)/.match(text[position..])
256 raise 'bad' unless match
257 key = match[0]
258 kind = key == '*' ? :all : :key
259 position += key.length
260 else
261 raise 'bad'
262 end
263 selectors << [descendant, kind, key]
264 end
265 selectors
266 rescue RuntimeError
267 raise(field ? 'ERROR bad field' : 'ERROR bad path')
268 end
269 end
270 def location(steps)
271 '$' + steps.map { |step| step.is_a?(Integer) ? "[#{step}]" : (IDENTIFIER.match?(step) ? '.' + step : '[' + self.class.quote(step, "'") + ']') }.join
272 end
273 def children(match)
274 value, steps, positions = match
275 if value.is_a?(Hash)
276 value.each_with_index.map { |(key, item), index| [item, steps + [key], positions + [index]] }
277 elsif value.is_a?(Array)
278 value.each_with_index.map { |item, index| [item, steps + [index], positions + [index]] }
279 else
280 []
281 end
282 end
283 def preorder(match)
284 [match] + children(match).flat_map { |child| preorder(child) }
285 end
286 def select(match, selector)
287 _, kind, key = selector
288 value = match[0]
289 all = children(match)
290 case kind
291 when :key then value.is_a?(Hash) ? all.select { |item| item[1].last == key } : []
292 when :all then all
293 when :index
294 return [] unless value.is_a?(Array)
295 index = key < 0 ? value.length + key : key
296 index.between?(0, value.length - 1) ? [all[index]] : []
297 when :slice
298 return [] unless value.is_a?(Array)
299 bounds = [key[0] || 0, key[1] || value.length].map { |n| [[n < 0 ? n + value.length : n, 0].max, value.length].min }
300 bounds[0] < bounds[1] ? all[bounds[0]...bounds[1]] : []
301 end
302 end
303 def evaluate(selectors, initial = [[@document, [], []]])
304 selectors.reduce(initial) do |matches, selector|
305 matches.flat_map { |match| (selector[0] ? preorder(match) : [match]).flat_map { |item| select(item, selector) } }
306 end
307 end
308 def field(value, selectors)
309 result = evaluate(selectors, [[value, [], []]])
310 result.empty? ? MISSING : result[0][0]
311 end
312 def at(steps)
313 steps.reduce(@document) { |value, step| value[step] }
314 end
315 def assign(steps, value)
316 if steps.empty?
317 @document = value
318 else
319 at(steps[0...-1])[steps[-1]] = value
320 end
321 end
322 def merge(target, source)
323 source.each do |key, value|
324 if target[key].is_a?(Hash) && value.is_a?(Hash)
325 merge(target[key], value)
326 else
327 target[key] = copy(value)
328 end
329 end
330 end
331 def parse(text)
332 StrictJson.new(text).read
333 rescue JsonFault => error
334 raise "PARSE ERROR col #{error.position + 1}: #{error.message}"
335 end
336 def ensure_kind(matches, wanted, message)
337 bad = matches.find { |item| !wanted.include?(type(item[0])) }
338 raise "ERROR #{message} at #{location(bad[1])}" if bad
339 end
340 def command(line)
341 command, argument = line.split(' ', 2)
342 raise "ERROR unknown command '#{command}'" unless COMMANDS.include?(command)
343 count = { 'SET' => 2, 'MERGE' => 2, 'INSERT' => 3, 'FILTER' => 4 }[command]
344 args = argument ? argument.split(/ /, count || -1) : []
345 if %w[SET MERGE INSERT].include?(command)
346 raise 'ERROR bad arguments' unless args.length == count
347 elsif command == 'FILTER'
348 raise 'ERROR bad arguments' unless args.length >= 3 && (args[2] == 'EXISTS' ? args.length == 3 : args.length == 4)
349 elsif command == 'SORT'
350 raise 'ERROR bad arguments' unless args.length.between?(1, 3) && (args.length < 3 || (args[1] != 'DESC' && args[2] == 'DESC'))
351 else
352 expected = %w[COMPACT PRETTY FLATTEN].include?(command) ? [0, 1] : (command == 'UNDO' ? [0] : command == 'RENAME' ? [2] : [1])
353 raise 'ERROR bad arguments' unless expected.include?(args.length)
354 end
355 if command == 'UNDO'
356 raise 'ERROR nothing to undo' if @history.empty?
357 @document = @history.pop
358 puts "UNDO #{@history.length}"
359 return
360 end
361 selectors = path(args[0] || '$')
362 field_selectors = nil
363 if command == 'FILTER'
364 field_selectors = path(args[1], true)
365 raise 'ERROR bad operator' unless %w[== != < <= > >= EXISTS].include?(args[2])
366 elsif command == 'SORT'
367 field_selectors = path(args.length >= 2 && args[1] != 'DESC' ? args[1] : '@', true)
368 elsif command == 'INSERT'
369 raise 'ERROR bad index' unless INTEGER.match?(args[1])
370 elsif command == 'RENAME'
371 begin
372 if IDENTIFIER.match?(args[1])
373 new_key = args[1]
374 else
375 new_key, ending = quoted_key(args[1], 0)
376 raise 'bad' unless ending == args[1].length
377 end
378 rescue RuntimeError
379 raise 'ERROR bad key'
380 end
381 elsif command == 'DELETE'
382 raise 'ERROR cannot delete root' if selectors.empty?
383 end
384 replacement = parse(args[-1]) if %w[SET MERGE INSERT].include?(command) || (command == 'FILTER' && args[2] != 'EXISTS')
385 raise 'ERROR merge value is not an object' if command == 'MERGE' && !replacement.is_a?(Hash)
386 matches = evaluate(selectors)
387 case command
388 when 'LENGTH' then ensure_kind(matches, %w[object array string], 'no length')
389 when 'KEYS', 'MERGE' then ensure_kind(matches, ['object'], 'not an object')
390 when 'FILTER', 'INSERT', 'SORT' then ensure_kind(matches, ['array'], 'not an array')
391 when 'RENAME'
392 bad = matches.find { |item| item[1].empty? || !item[1].last.is_a?(String) }
393 raise "ERROR not a member at #{location(bad[1])}" if bad
394 end
395 if command == 'INSERT'
396 index = args[1].to_i
397 bad = matches.find { |item| !index.between?(-item[0].length - 1, item[0].length) }
398 raise "ERROR index out of range at #{location(bad[1])}" if bad
399 end
400 if %w[SET DELETE INSERT MERGE SORT RENAME].include?(command)
401 if command == 'SET' && !selectors.empty? && selectors[-1][0] == false && selectors[-1][1] == :key
402 key = selectors[-1][2]
403 matches = evaluate(selectors[0...-1]).filter_map do |parent|
404 next unless parent[0].is_a?(Hash)
405 index = parent[0].keys.index(key) || parent[0].length
406 [parent[0][key], parent[1] + [key], parent[2] + [index]]
407 end
408 end
409 targets = matches.uniq { |item| item[1] }.sort { |a, b| b[2] <=> a[2] }
410 original = copy(@document)
411 begin
412 targets.each do |_, steps, _|
413 value = at(steps) unless command == 'SET'
414 case command
415 when 'SET' then assign(steps, copy(replacement))
416 when 'DELETE'
417 parent = at(steps[0...-1])
418 parent.is_a?(Array) ? parent.delete_at(steps[-1]) : parent.delete(steps[-1])
419 when 'INSERT' then value.insert(index < 0 ? value.length + 1 + index : index, copy(replacement))
420 when 'MERGE' then merge(value, replacement)
421 when 'SORT'
422 descending = args[-1] == 'DESC'
423 decorated = value.each_with_index.map { |item, i| [field(item, field_selectors), i, item] }
424 decorated.sort! do |a, b|
425 key = lambda do |v|
426 case v
427 when nil then [0, 0]
428 when Integer then [1, v]
429 when String then [2, v]
430 else [3, 0]
431 end
432 end
433 comparison = key.call(a[0]) <=> key.call(b[0])
434 comparison = -comparison if descending
435 comparison == 0 ? a[1] <=> b[1] : comparison
436 end
437 value.replace(decorated.map { |item| item[2] })
438 when 'RENAME'
439 parent = at(steps[0...-1])
440 key = steps[-1]
441 raise "ERROR key exists at #{location(steps)}" if key != new_key && parent.key?(new_key)
442 assign(steps[0...-1], parent.to_h { |name, item| [name == key ? new_key : name, item] })
443 end
444 end
445 rescue RuntimeError
446 @document = original
447 raise
448 end
449 @history << original
450 puts "#{command} #{targets.length}"
451 elsif %w[COUNT SUM MIN MAX AVG].include?(command)
452 numbers = matches.map(&:first).select { |value| value.is_a?(Integer) }
453 if command == 'COUNT'
454 puts "COUNT #{matches.length}"
455 elsif command == 'AVG'
456 if numbers.empty?
457 puts 'AVG null'
458 else
459 sum = numbers.sum
460 rounded = (sum.abs * 200 + numbers.length) / (2 * numbers.length)
461 puts format('AVG %s%d.%02d', sum < 0 && rounded > 0 ? '-' : '', rounded / 100, rounded % 100)
462 end
463 else
464 result = command == 'SUM' ? numbers.sum : numbers.public_send(command.downcase)
465 puts "#{command} #{result.nil? ? 'null' : result}"
466 end
467 else
468 output = []
469 matches.each do |match|
470 value, steps, = match
471 case command
472 when 'GET' then output << "#{location(steps)} = #{compact(value)}"
473 when 'COMPACT' then output << compact(value)
474 when 'PRETTY' then output << pretty(value)
475 when 'TYPE' then output << "#{location(steps)} #{type(value)}"
476 when 'LENGTH' then output << "#{location(steps)} #{value.length}"
477 when 'KEYS' then output << "#{location(steps)} #{compact(value.keys)}"
478 when 'FLATTEN'
479 preorder(match).each { |item| output << "#{location(item[1])} = #{compact(item[0])}" if children(item).empty? }
480 when 'FILTER'
481 children(match).each do |item|
482 f = field(item[0], field_selectors)
483 operator = args[2]
484 equal = !f.equal?(MISSING) && type(f) == type(replacement) && f == replacement
485 pass = if operator == 'EXISTS'
486 !f.equal?(MISSING)
487 elsif operator == '==' then equal
488 elsif operator == '!=' then !equal
489 elsif (f.is_a?(Integer) && replacement.is_a?(Integer)) || (f.is_a?(String) && replacement.is_a?(String))
490 f.public_send(operator, replacement)
491 else false
492 end
493 output << "#{location(item[1])} = #{compact(item[0])}" if pass
494 end
495 end
496 end
497 puts(output.empty? ? ['no match'] : output)
498 end
499 rescue RuntimeError => error
500 puts error.message
501 end
502 def load_document(lines)
503 text = lines.join("\n")
504 begin
505 value = StrictJson.new(text).read
506 @history << copy(@document)
507 @document = value
508 stats = lambda do |item|
509 children = item.is_a?(Hash) ? item.values : item.is_a?(Array) ? item : []
510 nested = children.map { |child| stats.call(child) }
511 [1 + nested.sum(&:first), 1 + (nested.map(&:last).max || 0)]
512 end
513 count, depth = stats.call(value)
514 puts "LOADED nodes=#{count} depth=#{depth}"
515 rescue JsonFault => error
516 prefix = text[0...error.position]
517 number = prefix.count("\n") + 1
518 column = error.position - (prefix.rindex("\n") || -1)
519 puts "PARSE ERROR line #{number} col #{column}: #{error.message}"
520 end
521 end
522 def run
523 input = STDIN.each_line.map { |line| line.chomp("\n").delete_suffix("\r") }
524 position = 0
525 loop do
526 block = []
527 while input[position] != '%%'
528 block << input[position]
529 position += 1
530 end
531 position += 1
532 load_document(block)
533 while input[position] != 'LOAD' && input[position] != 'END'
534 command(input[position])
535 position += 1
536 end
537 break if input[position] == 'END'
538 position += 1
539 end
540 end
