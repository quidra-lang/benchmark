7function isIdentifier(text: string): boolean {
8 return /^[A-Za-z_]/.test(text) && !/[^A-Za-z0-9_]/.test(text);
9}
10const integerNotation = /^(?:0|-?[1-9][0-9]{0,8})$/;
11const escapeCodes: Record<string, string> = { b: '\b', f: '\f', n: '\n', r: '\r', t: '\t', '/': '/', '\\': '\\' };

13 constructor(public position: number, message: string) { super(message); }

15function quote(text: string, delimiter = '"'): string {
16 let result = delimiter;
17 const escapes: Record<string, string> = { '\b': '\\b', '\f': '\\f', '\n': '\\n', '\r': '\\r', '\t': '\\t' };
18 for (const char of text) {
19 if (char === delimiter || char === '\\') result += '\\' + char;
20 else if (Object.hasOwn(escapes, char)) result += escapes[char];
21 else if (char.charCodeAt(0) < 32 || char.charCodeAt(0) === 127) result += '\\u' + char.charCodeAt(0).toString(16).padStart(4, '0');
22 else result += char;
23 }
24 return result + delimiter;
25}
26function compact(value: Value): string {
27 if (value instanceof Map) return '{' + [...value].map(([key, child]) => quote(key) + ':' + compact(child)).join(',') + '}';
28 if (Array.isArray(value)) return '[' + value.map(compact).join(',') + ']';
29 return typeof value === 'string' ? quote(value) : String(value);
30}
31function pretty(value: Value, indentation = ''): string {
32 const next = indentation + ' ';
33 if (value instanceof Map && value.size > 0) return '{\n' + [...value].map(([key, child]) => next + quote(key) + ': ' + pretty(child, next)).join(',\n') + '\n' + indentation + '}';
34 if (Array.isArray(value) && value.length > 0) return '[\n' + value.map(child => next + pretty(child, next)).join(',\n') + '\n' + indentation + ']';
35 return compact(value);
36}
37function clone(value: Value): Value {
38 if (value instanceof Map) return new Map<string, Value>([...value].map(([key, child]) => [key, clone(child)]));
39 return Array.isArray(value) ? value.map(clone) : value;
40}
41function typeName(value: Value): string {
42 return value === null ? 'null' : value instanceof Map ? 'object' : Array.isArray(value) ? 'array' : typeof value;
43}

45class JsonReader {
46 position = 0;
47 constructor(public text: string) {}
48 current(): string { return this.text[this.position] ?? ''; }
49 fail(reason: string, position = this.position): never { throw new JsonFailure(position, reason); }
50 whitespace(): void { while (this.current() === ' ' || this.current() === '\n') this.position++; }
51 string(delimiter = '"'): string {
52 this.position++;
53 let result = '';
54 while (true) {
55 const char = this.current();
56 if (char === '' || char === '\n') this.fail('unterminated string');
57 if (char === delimiter) { this.position++; return result; }
58 if (char !== '\\') { result += char; this.position++; continue; }
59 const start = this.position;
60 const escaped = this.text[start + 1] ?? '';
61 if (escaped === '' || escaped === '\n') this.fail('unterminated string', start + 1);
62 if (escaped === 'u') {
63 const digits = this.text.slice(start + 2, start + 6);
64 if (!/^[0-9A-Fa-f]{4}$/.test(digits)) this.fail('bad unicode escape', start);
65 const code = parseInt(digits, 16);
66 if (code > 127) this.fail('non-ascii escape', start);
67 result += String.fromCharCode(code); this.position += 6;
68 } else if (escaped === delimiter || Object.hasOwn(escapeCodes, escaped)) {
69 result += escaped === delimiter ? delimiter : escapeCodes[escaped]; this.position += 2;
70 } else this.fail(`invalid escape '\\${escaped}'`, start);
71 }
72 }
73 number(): number {
74 const start = this.position;
75 const negative = this.current() === '-';
76 if (negative) this.position++;
77 const digitsStart = this.position;
78 while (/^[0-9]$/.test(this.current())) this.position++;
79 const digits = this.text.slice(digitsStart, this.position);
80 if (digits.length === 0) this.fail('invalid number', start);
81 if (digits.length > 1 && digits[0] === '0') this.fail('leading zero', digitsStart);
82 if (this.current() === '.') this.fail('fraction not allowed');
83 if (this.current() === 'e' || this.current() === 'E') this.fail('exponent not allowed');
84 if (digits.length >= 10) this.fail('number out of range', start);
85 return (negative ? -1 : 1) * Number(digits);
86 }
87 value(): Value {
88 this.whitespace();
89 const char = this.current();
90 if (char === '') this.fail('unexpected end of input');
91 if (char === '"') return this.string();
92 if (char === '-' || /^[0-9]$/.test(char)) return this.number();
93 if (/^[A-Za-z]$/.test(char)) {
94 const start = this.position;
95 while (/^[A-Za-z]$/.test(this.current())) this.position++;
96 const word = this.text.slice(start, this.position);
97 if (word === 'null') return null;
98 if (word === 'true' || word === 'false') return word === 'true';
99 this.fail(`invalid literal '${word}'`, start);
100 }
101 if (char === '{') {
102 this.position++; this.whitespace();
103 const object = new Map<string, Value>();
104 if (this.current() === '}') { this.position++; return object; }
105 while (true) {
106 this.whitespace();
107 if (this.current() === '') this.fail('unexpected end of input');
108 if (this.current() !== '"') this.fail('expected key');
109 const start = this.position;
110 const key = this.string();
111 if (object.has(key)) this.fail('duplicate key ' + quote(key), start);
112 this.whitespace();
113 if (this.current() === '') this.fail('unexpected end of input');
114 if (this.current() !== ':') this.fail("expected ':'");
115 this.position++; object.set(key, this.value()); this.whitespace();
116 if (this.current() === '}') { this.position++; return object; }
117 if (this.current() === '') this.fail('unexpected end of input');
118 if (this.current() !== ',') this.fail("expected ',' or '}'");
119 this.position++; this.whitespace();
120 if (this.current() === '}') this.fail('trailing comma');
121 }
122 }
123 if (char === '[') {
124 this.position++; this.whitespace();
125 const array: Value[] = [];
126 if (this.current() === ']') { this.position++; return array; }
127 while (true) {
128 array.push(this.value()); this.whitespace();
129 if (this.current() === ']') { this.position++; return array; }
130 if (this.current() === '') this.fail('unexpected end of input');
131 if (this.current() !== ',') this.fail("expected ',' or ']'");
132 this.position++; this.whitespace();
133 if (this.current() === ']') this.fail('trailing comma');
134 }
135 }
136 this.fail(`unexpected character '${char}'`);
137 }
138 parse(): Value {
139 const value = this.value(); this.whitespace();
140 if (this.current() !== '') this.fail('trailing characters');
141 return value;
142 }
143}
144function path(text: string, field = false): Selector[] {
145 try {
146 if (text[0] !== (field ? '@' : '$')) throw new Error();
147 const result: Selector[] = [];
148 let position = 1;
149 while (position < text.length) {
150 let descendant = false;
151 let bracket = false;
152 if (!field && text.startsWith('..', position)) { descendant = true; position += 2; bracket = text[position] === '['; }
153 else if (text[position] === '.') position++;
154 else if (text[position] === '[') bracket = true;
155 else throw new Error();
156 let selector: Selector;
157 if (bracket) {
158 position++;
159 if (text[position] === "'") {
160 const reader = new JsonReader(text); reader.position = position;
161 const key = reader.string("'"); position = reader.position;
162 if (text[position++] !== ']') throw new Error();
163 selector = { kind: 'key', key, descendant };
164 } else {
165 const end = text.indexOf(']', position);
166 if (end < 0) throw new Error();
167 const selection = text.slice(position, end); position = end + 1;
168 if (integerNotation.test(selection)) selector = { kind: 'index', index: Number(selection), descendant };
169 else if (!field && selection === '*') selector = { kind: 'all', descendant };
170 else if (!field && selection.includes(':')) {
171 const parts = selection.split(':');
172 if (parts.length !== 2 || parts.some(part => part !== '' && !integerNotation.test(part))) throw new Error();
173 selector = { kind: 'slice', start: parts[0] === '' ? undefined : Number(parts[0]), end: parts[1] === '' ? undefined : Number(parts[1]), descendant };
174 } else throw new Error();
175 }
176 } else if (!field && text[position] === '*') { position++; selector = { kind: 'all', descendant }; }
177 else {
178 const name = /^[A-Za-z_][A-Za-z0-9_]*/.exec(text.slice(position));
179 if (name === null) throw new Error();
180 position += name[0].length; selector = { kind: 'key', key: name[0], descendant };
181 }
182 result.push(selector);
183 }
184 return result;
185 } catch { throw new Error(field ? 'bad field' : 'bad path'); }
186}
187function children(match: Match): Match[] {
188 if (match.value instanceof Map) return [...match.value].map(([key, value], index) => ({ value, path: [...match.path, key], order: [...match.order, index] }));
189 if (Array.isArray(match.value)) return match.value.map((value, index) => ({ value, path: [...match.path, index], order: [...match.order, index] }));
190 return [];
191}
192function preorder(match: Match): Match[] {
193 const result: Match[] = [];
194 const pending = [match];
195 while (pending.length > 0) {
196 const next = pending.pop()!; result.push(next); pending.push(...children(next).reverse());
197 }
198 return result;
199}
200function select(match: Match, selector: Selector): Match[] {
201 const all = children(match);
202 if (selector.kind === 'all') return all;
203 if (selector.kind === 'key') return match.value instanceof Map ? all.filter(child => child.path[child.path.length - 1] === selector.key) : [];
204 if (!Array.isArray(match.value)) return [];
205 const length = match.value.length;
206 if (selector.kind === 'index') {
207 const index = selector.index! < 0 ? length + selector.index! : selector.index!;
208 return index >= 0 && index < length ? [all[index]] : [];
209 }
210 const bound = (value: number): number => Math.min(length, Math.max(0, value < 0 ? length + value : value));
211 return all.slice(bound(selector.start ?? 0), bound(selector.end ?? length));
212}
213function applySegment(matches: Match[], selector: Selector): Match[] {
214 return matches.flatMap(match => (selector.descendant ? preorder(match) : [match]).flatMap(value => select(value, selector)));
215}
216function evaluate(document: Value, selectors: Selector[]): Match[] {
217 return selectors.reduce(applySegment, [{ value: document, path: [], order: [] }]);
218}
219function location(steps: Step[]): string {
220 return '$' + steps.map(step => typeof step === 'number' ? `[${step}]` : isIdentifier(step) ? '.' + step : '[' + quote(step, "'") + ']').join('');
221}
222function fieldValue(value: Value, selectors: Selector[]): Value | undefined {
223 let matches: Match[] = [{ value, path: [], order: [] }];
224 for (const selector of selectors) matches = applySegment(matches, selector);
225 return matches.length === 0 ? undefined : matches[0].value;
226}
227function equal(left: Value, right: Value): boolean {
228 if (typeName(left) !== typeName(right)) return false;
229 if (left instanceof Map && right instanceof Map) return left.size === right.size && [...left].every(([key, value]) => right.has(key) && equal(value, right.get(key)!));
230 if (Array.isArray(left) && Array.isArray(right)) return left.length === right.length && left.every((value, index) => equal(value, right[index]));
231 return left === right;
232}
233function compareKeys(left: Value | undefined, right: Value | undefined): number {
234 const rank = (value: Value | undefined): number => value === null ? 0 : typeof value === 'number' ? 1 : typeof value === 'string' ? 2 : 3;
235 const difference = rank(left) - rank(right);
236 if (difference !== 0) return difference;
237 if (typeof left === 'number' && typeof right === 'number') return left - right;
238 if (typeof left === 'string' && typeof right === 'string') return left === right ? 0 : left < right ? -1 : 1;
239 return 0;
240}
241function targets(matches: Match[]): Match[] {
242 const unique = new Map<string, Match>();
243 for (const match of matches) unique.set(JSON.stringify(match.path), match);
244 return [...unique.values()].sort((a, b) => {
245 for (let i = 0; i < Math.min(a.order.length, b.order.length); i++) if (a.order[i] !== b.order[i]) return b.order[i] - a.order[i];
246 return b.order.length - a.order.length;
247 });
248}
249function getAt(document: Value, steps: Step[]): Value {
250 let value = document;
251 for (const step of steps) value = typeof step === 'string' ? (value as Map<string, Value>).get(step)! : (value as Value[])[step];
252 return value;
253}
254function putAt(document: Value, steps: Step[], value: Value): Value {
255 if (steps.length === 0) return value;
256 const parent = getAt(document, steps.slice(0, -1));
257 const last = steps[steps.length - 1];
258 if (typeof last === 'string') (parent as Map<string, Value>).set(last, value);
259 else (parent as Value[])[last] = value;
260 return document;
261}
262function merge(target: Map<string, Value>, source: Map<string, Value>): void {
263 for (const [key, value] of source) {
264 const previous = target.get(key);
265 if (previous instanceof Map && value instanceof Map) merge(previous, value);
266 else target.set(key, clone(value));
267 }
268}
269function splitLimited(text: string, maximum: number): string[] {
270 const result: string[] = [];
271 let start = 0;
272 while (result.length < maximum) {
273 const space = text.indexOf(' ', start);
274 if (space < 0) break;
275 result.push(text.slice(start, space)); start = space + 1;
276 }
277 result.push(text.slice(start)); return result;
278}

280class QueryTool {
281 document: Value = null;
282 history: Value[] = [];
283 output: string[] = [];
284 load(lines: string[]): void {
285 const text = lines.join('\n');
286 try {
287 const document = new JsonReader(text).parse();
288 this.history.push(clone(this.document)); this.document = clone(document);
289 const all = preorder({ value: document, path: [], order: [] });
290 this.output.push(`LOADED nodes=${all.length} depth=${Math.max(...all.map(match => match.path.length + 1))}`);
291 } catch (error) {
292 const failure = error as JsonFailure;
293 const before = text.slice(0, failure.position);
294 const line = before.split('\n').length;
295 const column = failure.position - before.lastIndexOf('\n');
296 this.output.push(`PARSE ERROR line ${line} col ${column}: ${failure.message}`);
297 }
298 }
299 execute(line: string): void {
300 const space = line.indexOf(' ');
301 const command = space < 0 ? line : line.slice(0, space);
302 const argument = space < 0 ? null : line.slice(space + 1);
303 const commands = new Set(['LOAD', 'GET', 'COMPACT', 'PRETTY', 'TYPE', 'LENGTH', 'KEYS', 'FLATTEN', 'COUNT', 'SUM', 'MIN', 'MAX', 'AVG', 'FILTER', 'SET', 'DELETE', 'INSERT', 'MERGE', 'SORT', 'RENAME', 'UNDO']);
304 if (!commands.has(command)) throw new Error(`unknown command '${command}'`);
305 let args = argument === null ? [] : argument.split(' ');
306 if (command === 'SET' || command === 'MERGE') args = argument === null ? [] : splitLimited(argument, 1);
307 if (command === 'INSERT') args = argument === null ? [] : splitLimited(argument, 2);
308 if (command === 'FILTER') args = argument === null ? [] : splitLimited(argument, 3);
309 const optional = ['COMPACT', 'PRETTY', 'FLATTEN'].includes(command);
310 let shape = optional ? args.length <= 1 : command === 'UNDO' || command === 'LOAD' ? args.length === 0 : command === 'SET' || command === 'MERGE' || command === 'RENAME' ? args.length === 2 : command === 'INSERT' ? args.length === 3 : command === 'FILTER' ? args.length >= 3 && (args[2] === 'EXISTS' ? args.length === 3 : args.length === 4) : command === 'SORT' ? args.length >= 1 && args.length <= 3 && (args.length < 3 || args[1] !== 'DESC' && args[2] === 'DESC') : args.length === 1;
311 if (!shape) throw new Error('bad arguments');
312 if (command === 'UNDO') {
313 if (this.history.length === 0) throw new Error('nothing to undo');
314 this.document = clone(this.history.pop()!); this.output.push(`UNDO ${this.history.length}`); return;
315 }
316 const selectors = path(args[0] ?? '$');
317 let field: Selector[] = [];
318 let key = '';
319 let value: Value = null;
320 if (command === 'FILTER') {
321 field = path(args[1], true);
322 if (!['==', '!=', '<', '<=', '>', '>=', 'EXISTS'].includes(args[2])) throw new Error('bad operator');
323 if (args[2] !== 'EXISTS') value = new JsonReader(args[3]).parse();
324 } else if (command === 'SORT') field = path(args.length > 1 && args[1] !== 'DESC' ? args[1] : '@', true);
325 else if (command === 'RENAME') {
326 try {
327 if (isIdentifier(args[1])) key = args[1];
328 else { const reader = new JsonReader(args[1]); if (reader.current() !== "'") throw new Error(); key = reader.string("'"); if (reader.position !== args[1].length) throw new Error(); }
329 } catch { throw new Error('bad key'); }
330 } else if (command === 'SET' || command === 'MERGE') value = new JsonReader(args[1]).parse();
331 else if (command === 'INSERT') {
332 if (!integerNotation.test(args[1])) throw new Error('bad index');
333 value = new JsonReader(args[2]).parse();
334 }
335 if (command === 'MERGE' && !(value instanceof Map)) throw new Error('merge value is not an object');
336 if (command === 'DELETE' && selectors.length === 0) throw new Error('cannot delete root');
337 let matches = evaluate(this.document, selectors);
338 if (['FILTER', 'INSERT', 'SORT'].includes(command)) {
339 const invalid = matches.find(match => !Array.isArray(match.value));
340 if (invalid !== undefined) throw new Error('not an array at ' + location(invalid.path));
341 }
342 if (command === 'MERGE' || command === 'KEYS') {
343 const invalid = matches.find(match => !(match.value instanceof Map));
344 if (invalid !== undefined) throw new Error('not an object at ' + location(invalid.path));
345 }
346 if (command === 'LENGTH') {
347 const invalid = matches.find(match => !(match.value instanceof Map) && !Array.isArray(match.value) && typeof match.value !== 'string');
348 if (invalid !== undefined) throw new Error('no length at ' + location(invalid.path));
349 }
350 if (command === 'RENAME') {
351 const invalid = matches.find(match => typeof match.path[match.path.length - 1] !== 'string');
352 if (invalid !== undefined) throw new Error('not a member at ' + location(invalid.path));
353 }
354 if (command === 'INSERT') {
355 const index = Number(args[1]);
356 const invalid = matches.find(match => index < -(match.value as Value[]).length - 1 || index > (match.value as Value[]).length);
357 if (invalid !== undefined) throw new Error('index out of range at ' + location(invalid.path));
358 }
359 if (command === 'FILTER') {
360 const output: string[] = [];
361 for (const match of matches) for (const child of children(match)) {
362 const left = fieldValue(child.value, field);
363 const operator = args[2];
364 let accepted = operator === 'EXISTS' ? left !== undefined : operator === '==' ? left !== undefined && equal(left, value) : operator === '!=' ? left === undefined || !equal(left, value) : false;
365 if (!['EXISTS', '==', '!='].includes(operator) && (typeof left === 'number' && typeof value === 'number' || typeof left === 'string' && typeof value === 'string')) {
366 const comparison = compareKeys(left, value);
367 accepted = operator === '<' ? comparison < 0 : operator === '<=' ? comparison <= 0 : operator === '>' ? comparison > 0 : comparison >= 0;
368 }
369 if (accepted) output.push(location(child.path) + ' = ' + compact(child.value));
370 }
371 this.output.push(...(output.length === 0 ? ['no match'] : output)); return;
372 }
373 if (['SET', 'DELETE', 'INSERT', 'MERGE', 'SORT', 'RENAME'].includes(command)) {
374 if (command === 'SET' && selectors.length > 0) {
375 const last = selectors[selectors.length - 1];
376 if (last.kind === 'key' && !last.descendant) {
377 matches = [];
378 for (const parent of evaluate(this.document, selectors.slice(0, -1))) {
379 if (!(parent.value instanceof Map)) continue;
380 const keys = [...parent.value.keys()];
381 const index = keys.indexOf(last.key!);
382 matches.push({ value: parent.value.get(last.key!) ?? null, path: [...parent.path, last.key!], order: [...parent.order, index < 0 ? keys.length : index] });
383 }
384 }
385 }
386 const selected = targets(matches);
387 let document = clone(this.document);
388 for (const target of selected) {
389 if (command === 'SET') document = putAt(document, target.path, clone(value));
390 else if (command === 'DELETE') {
391 const parent = getAt(document, target.path.slice(0, -1)); const last = target.path[target.path.length - 1];
392 if (typeof last === 'string') (parent as Map<string, Value>).delete(last); else (parent as Value[]).splice(last, 1);
393 } else if (command === 'INSERT') {
394 const array = getAt(document, target.path) as Value[];
395 const index = Number(args[1]); array.splice(index < 0 ? array.length + 1 + index : index, 0, clone(value));
396 } else if (command === 'MERGE') merge(getAt(document, target.path) as Map<string, Value>, value as Map<string, Value>);
397 else if (command === 'SORT') {
398 const array = getAt(document, target.path) as Value[];
399 const direction = args[args.length - 1] === 'DESC' ? -1 : 1;
400 const sorted = array.map((value, index) => ({ value, index, key: fieldValue(value, field) })).sort((a, b) => direction * compareKeys(a.key, b.key) || a.index - b.index).map(item => item.value);
401 document = putAt(document, target.path, sorted);
402 } else {
403 const parentPath = target.path.slice(0, -1); const old = target.path[target.path.length - 1] as string;
404 const object = getAt(document, parentPath) as Map<string, Value>;
405 if (old !== key && object.has(key)) throw new Error('key exists at ' + location(target.path));
406 document = putAt(document, parentPath, new Map<string, Value>([...object].map(([name, child]) => [name === old ? key : name, child])));
407 }
408 }
409 this.history.push(clone(this.document)); this.document = document;
410 this.output.push(`${command} ${selected.length}`); return;
411 }
412 if (command === 'COUNT') { this.output.push(`COUNT ${matches.length}`); return; }
413 if (['SUM', 'MIN', 'MAX', 'AVG'].includes(command)) {
414 const numbers = matches.map(match => match.value).filter((value): value is number => typeof value === 'number');
415 const sum = numbers.reduce((total, number) => total + number, 0);
416 let answer = 'null';
417 if (command === 'SUM') answer = String(sum);
418 else if (numbers.length > 0) {
419 if (command === 'AVG') { const rounded = Math.floor((Math.abs(sum) * 200 + numbers.length) / (2 * numbers.length)); answer = (sum < 0 && rounded > 0 ? '-' : '') + Math.floor(rounded / 100) + '.' + String(rounded % 100).padStart(2, '0'); }
420 else answer = String(numbers.reduce((best, number) => command === 'MIN' ? Math.min(best, number) : Math.max(best, number)));
421 }
422 this.output.push(command + ' ' + answer); return;
423 }
424 if (matches.length === 0) { this.output.push('no match'); return; }
425 for (const match of matches) {
426 if (command === 'GET') this.output.push(location(match.path) + ' = ' + compact(match.value));
427 else if (command === 'COMPACT') this.output.push(compact(match.value));
428 else if (command === 'PRETTY') this.output.push(pretty(match.value));
429 else if (command === 'TYPE') this.output.push(location(match.path) + ' ' + typeName(match.value));
430 else if (command === 'LENGTH') this.output.push(location(match.path) + ' ' + (match.value instanceof Map ? match.value.size : (match.value as string | Value[]).length));
431 else if (command === 'KEYS') this.output.push(location(match.path) + ' ' + compact([...(match.value as Map<string, Value>).keys()]));
432 else if (command === 'FLATTEN') for (const leaf of preorder(match)) if (children(leaf).length === 0) this.output.push(location(leaf.path) + ' = ' + compact(leaf.value));
433 }
434 }
435 run(lines: string[]): void {
436 let position = 0;
437 const block = (): string[] => { const body: string[] = []; while (lines[position] !== '%%') body.push(lines[position++]); position++; return body; };
438 this.load(block());
439 while (lines[position] !== 'END') {
440 const line = lines[position++];
441 if (line === 'LOAD') { this.load(block()); continue; }
442 try { this.execute(line); }
443 catch (error) {
444 this.output.push(error instanceof JsonFailure ? `PARSE ERROR col ${error.position + 1}: ${error.message}` : `ERROR ${(error as Error).message}`);
445 }
446 }
447 }
