38def is_letter(char):
39 return "a" <= char <= "z" or "A" <= char <= "Z"

46def is_hex_digit(char):
47 return is_digit(char) or "a" <= char <= "f" or "A" <= char <= "F"

50def is_identifier_start(char):
51 return is_letter(char) or char == "_"

54def is_identifier_char(char):
55 return is_identifier_start(char) or is_digit(char)

58def is_identifier(text):
59 return text != "" and is_identifier_start(text[0]) and all(
60 is_identifier_char(char) for char in text

64def is_integer_notation(text):
65 if text == "0":
66 return True
67 digits = text[1:] if text.startswith("-") else text
68 return 1 <= len(digits) <= 9 and digits[0] != "0" and all(is_digit(char) for char in digits)

71def hex_value(digits):
72 if len(digits) != 4 or not all(is_hex_digit(char) for char in digits):
73 return -1
74 return int(digits, 16)

77def quote(text, mark):
78 parts = [mark]
79 for char in text:
80 code = ord(char)
81 if char == mark:
82 parts.append("\\" + char)
83 elif char in QUOTE_ESCAPES:
84 parts.append(QUOTE_ESCAPES[char])
85 elif code < 32 or code == 127:
86 parts.append(f"\\u{code:04x}")
87 else:
88 parts.append(char)
89 parts.append(mark)
90 return "".join(parts)

94 def __init__(self, kind, scalar=None):

106 duplicate.items = [item.copy() for item in self.items]

116def compact(value):
117 if value.kind == NULL:
118 return "null"
119 if value.kind == BOOLEAN:
120 return "true" if value.scalar else "false"
121 if value.kind == NUMBER:
122 return str(value.scalar)
123 if value.kind == STRING:
124 return quote(value.scalar, '"')
125 if value.kind == ARRAY:
126 return "[" + ",".join(compact(item) for item in value.items) + "]"
127 members = [quote(key, '"') + ":" + compact(item) for key, item in zip(value.keys, value.items)]
128 return "{" + ",".join(members) + "}"

131def pretty(value, indent=""):
132 if value.kind not in (OBJECT, ARRAY) or not value.items:
133 return compact(value)
134 inner = indent + " "
135 lines = []
136 for position, item in enumerate(value.items):
137 label = quote(value.keys[position], '"') + ": " if value.kind == OBJECT else ""
138 lines.append(inner + label + pretty(item, inner))
139 opening, closing = ("{", "}") if value.kind == OBJECT else ("[", "]")
140 return opening + "\n" + ",\n".join(lines) + "\n" + indent + closing

143def equal(first, second):
144 if first.kind != second.kind:
145 return False
146 if first.kind == ARRAY:
147 return len(first.items) == len(second.items) and all(
148 equal(a, b) for a, b in zip(first.items, second.items)
149 )
150 if first.kind == OBJECT:
151 if len(first.keys) != len(second.keys):
152 return False
153 for key, item in zip(first.keys, first.items):
154 position = second.find(key)
155 if position < 0 or not equal(item, second.items[position]):
156 return False
157 return True
158 return first.scalar == second.scalar

161class JsonReader:
162 def __init__(self, text):
163 self.text = text
164 self.pos = 0
165
166 def peek(self):
167 return self.text[self.pos] if self.pos < len(self.text) else ""
168
169 def fail(self, reason, position=None):
170 raise JsonError(reason, self.pos if position is None else position)
171
172 def skip_whitespace(self):
173 while self.peek() in (" ", "\n"):
174 self.pos += 1
175
176 def require(self, char, reason):
177 if self.pos == len(self.text):
178 self.fail("unexpected end of input")
179 if self.text[self.pos] != char:
180 self.fail(reason)
181
182 def parse(self):
183 value = self.value()
184 self.skip_whitespace()
185 if self.pos < len(self.text):
186 self.fail("trailing characters")
187 return value
188
189 def value(self):
190 self.skip_whitespace()
191 char = self.peek()
192 if char == "":
193 self.fail("unexpected end of input")
194 if char == "{":
195 return self.object()
196 if char == "[":
197 return self.array()
198 if char == '"':
199 return JsonValue(STRING, self.string())
200 if char == "-" or is_digit(char):
201 return self.number()
202 if is_letter(char):
203 return self.literal()
204 self.fail(f"unexpected character '{char}'")
205
206 def container_end(self, closing, reason):
207 self.skip_whitespace()
208 if self.peek() == closing:
209 self.pos += 1
210 return True
211 self.require(",", reason)
212 self.pos += 1
213 self.skip_whitespace()
214 if self.peek() == closing:
215 self.fail("trailing comma")
216 return False
217
218 def object(self):
219 node = JsonValue(OBJECT)
220 self.pos += 1
221 self.skip_whitespace()
222 if self.peek() == "}":
223 self.pos += 1
224 return node
225 while True:
226 self.skip_whitespace()
227 self.require('"', "expected key")
228 start = self.pos
229 key = self.string()
230 if key in node.keys:
231 self.fail("duplicate key " + quote(key, '"'), start)
232 self.skip_whitespace()
233 self.require(":", "expected ':'")
234 self.pos += 1
235 node.keys.append(key)
236 node.items.append(self.value())
237 if self.container_end("}", "expected ',' or '}'"):
238 return node
239
240 def array(self):
241 node = JsonValue(ARRAY)
242 self.pos += 1
243 self.skip_whitespace()
244 if self.peek() == "]":
245 self.pos += 1
246 return node
247 while True:
248 node.items.append(self.value())
249 if self.container_end("]", "expected ',' or ']'"):
250 return node
251
252 def string(self):
253 self.pos += 1
254 chars = []
255 while True:
256 char = self.peek()
257 if char == "" or char == "\n":
258 self.fail("unterminated string")
259 if char == '"':
260 self.pos += 1
261 return "".join(chars)
262 if char == "\\":
263 chars.append(self.escape())
264 else:
265 chars.append(char)
266 self.pos += 1
267
268 def escape(self):
269 start = self.pos
270 self.pos += 1
271 escape = self.peek()
272 if escape == "" or escape == "\n":
273 self.fail("unterminated string")
274 if escape == "u":
275 code = hex_value(self.text[start + 2:start + 6])
276 if code < 0:
277 self.fail("bad unicode escape", start)
278 if code > 127:
279 self.fail("non-ascii escape", start)
280 self.pos = start + 6
281 return chr(code)
282 if escape not in JSON_ESCAPES:
283 self.fail(f"invalid escape '\\{escape}'", start)
284 self.pos += 1
285 return JSON_ESCAPES[escape]
286
287 def number(self):
288 start = self.pos
289 negative = self.peek() == "-"
290 if negative:
291 self.pos += 1
292 digits_start = self.pos
293 while is_digit(self.peek()):
294 self.pos += 1
295 digits = self.text[digits_start:self.pos]
296 if digits == "":
297 self.fail("invalid number", start)
298 if len(digits) >= 2 and digits[0] == "0":
299 self.fail("leading zero", digits_start)
300 if self.peek() == ".":
301 self.fail("fraction not allowed")
302 if self.peek() in ("e", "E"):
303 self.fail("exponent not allowed")
304 if len(digits) >= 10:
305 self.fail("number out of range", start)
306 value = int(digits)
307 return JsonValue(NUMBER, -value if negative else value)
308
309 def literal(self):
310 start = self.pos
311 while is_letter(self.peek()):
312 self.pos += 1
313 word = self.text[start:self.pos]
314 if word == "true":
315 return JsonValue(BOOLEAN, True)
316 if word == "false":
317 return JsonValue(BOOLEAN, False)
318 if word == "null":
319 return JsonValue(NULL)
320 self.fail(f"invalid literal '{word}'", start)

333class PathReader:
334 def __init__(self, text, error):
335 self.text = text
336 self.pos = 0
337 self.error = error
338
339 def fail(self):
340 raise CommandError(self.error)
341
342 def at_end(self):
343 return self.pos == len(self.text)
344
345 def peek(self):
346 return "" if self.at_end() else self.text[self.pos]
347
348 def accept(self, char):
349 if self.peek() != char:
350 return False
351 self.pos += 1
352 return True
353
354 def expect(self, char):
355 if not self.accept(char):
356 self.fail()
357
358 def identifier(self):
359 start = self.pos
360 if not is_identifier_start(self.peek()):
361 self.fail()
362 while is_identifier_char(self.peek()):
363 self.pos += 1
364 return self.text[start:self.pos]
365
366 def quoted_key(self):
367 self.expect("'")
368 chars = []
369 while True:
370 char = self.peek()
371 if char == "":
372 self.fail()
373 self.pos += 1
374 if char == "'":
375 return "".join(chars)
376 chars.append(self.escape() if char == "\\" else char)
377
378 def escape(self):
379 escape = self.peek()
380 self.pos += 1
381 if escape == "u":
382 code = hex_value(self.text[self.pos:self.pos + 4])
383 if code < 0 or code > 127:
384 self.fail()
385 self.pos += 4
386 return chr(code)
387 if escape not in KEY_ESCAPES:
388 self.fail()
389 return KEY_ESCAPES[escape]
390
391 def selection(self):
392 if self.peek() == "'":
393 key = self.quoted_key()
394 self.expect("]")
395 return Segment(KEY, key=key)
396 end = self.text.find("]", self.pos)
397 if end < 0:
398 self.fail()
399 content = self.text[self.pos:end]
400 self.pos = end + 1
401 if content == "*":
402 return Segment(ALL)
403 if is_integer_notation(content):
404 return Segment(INDEX, index=int(content))
405 start, colon, stop = content.partition(":")
406 for bound in (start, stop):
407 if bound != "" and not is_integer_notation(bound):
408 self.fail()
409 if colon == "":
410 self.fail()
411 return Segment(
412 SLICE,
413 start=int(start) if start else None,
414 end=int(stop) if stop else None,
415 )

418def parse_path(text):
419 reader = PathReader(text, "bad path")
420 reader.expect("$")
421 segments = []
422 while not reader.at_end():
423 if reader.accept("["):
424 segments.append(reader.selection())
425 continue
426 reader.expect(".")
427 descendant = reader.accept(".")
428 if reader.accept("*"):
429 segment = Segment(ALL)
430 elif descendant and reader.accept("["):
431 segment = reader.selection()
432 else:
433 segment = Segment(KEY, key=reader.identifier())
434 segment.descendant = descendant
435 segments.append(segment)
436 return segments
437
438
439def parse_field(text):
440 reader = PathReader(text, "bad field")
441 reader.expect("@")
442 segments = []
443 while not reader.at_end():
444 if reader.accept("["):
445 segment = reader.selection()
446 if segment.kind not in (KEY, INDEX):
447 reader.fail()
448 else:
449 reader.expect(".")
450 segment = Segment(KEY, key=reader.identifier())
451 segments.append(segment)
452 return segments
453
454
455def parse_key(text):
456 reader = PathReader(text, "bad key")
457 if reader.peek() == "'":
458 key = reader.quoted_key()
459 else:
460 key = reader.identifier()
461 if not reader.at_end():
462 reader.fail()
463 return key

479class Match:
480 def __init__(self, node, steps):
481 self.node = node
482 self.steps = steps
483
484 def child(self, position):
485 key = self.node.keys[position] if self.node.kind == OBJECT else None
486 return Match(self.node.items[position], self.steps + [Step(key, position)])
487
488 def location(self):
489 return "$" + "".join(step.text() for step in self.steps)
490
491 def positions(self):
492 return [step.position for step in self.steps]

495def child_position(value, segment):
496 if segment.kind == KEY:
497 return value.find(segment.key) if value.kind == OBJECT else -1
498 if value.kind != ARRAY:
499 return -1
500 size = len(value.items)
501 index = segment.index if segment.index >= 0 else size + segment.index
502 return index if 0 <= index < size else -1

504
505def clamp_bound(bound, size):
506 if bound < 0:
507 bound += size
508 return min(max(bound, 0), size)
509
510
511def slice_positions(value, segment):
512 if value.kind != ARRAY:
513 return range(0)
514 size = len(value.items)
515 start = 0 if segment.start is None else clamp_bound(segment.start, size)
516 end = size if segment.end is None else clamp_bound(segment.end, size)
517 return range(start, end)

520def select(segment, match, results):
521 if segment.kind == ALL:
522 positions = range(len(match.node.items))
523 elif segment.kind == SLICE:
524 positions = slice_positions(match.node, segment)
525 else:
526 position = child_position(match.node, segment)
527 positions = [position] if position >= 0 else []
528 for position in positions:
529 results.append(match.child(position))
530

532def preorder(match, results):
533 results.append(match)
534 for position in range(len(match.node.items)):
535 preorder(match.child(position), results)
536 return results

539def apply_segment(segment, matches):
540 results = []
541 for match in matches:
542 sources = preorder(match, []) if segment.descendant else [match]
543 for source in sources:
544 select(segment, source, results)
545 return results

549 matches = [Match(root, [])]

551 matches = apply_segment(segment, matches)

564def filter_holds(operator, value, reference):
565 if operator == "EXISTS":
566 return value is not None
567 if operator in ("==", "!="):
568 same = value is not None and equal(value, reference)
569 return same if operator == "==" else not same
570 if value is None or value.kind != reference.kind or value.kind not in (NUMBER, STRING):
571 return False
572 if operator == "<":
573 return value.scalar < reference.scalar
574 if operator == "<=":
575 return value.scalar <= reference.scalar
576 if operator == ">":
577 return value.scalar > reference.scalar
578 return value.scalar >= reference.scalar

589def average_text(numbers):
590 total = sum(numbers)
591 count = len(numbers)
592 rounded = (abs(total) * 200 + count) // (2 * count)
593 sign = "-" if total < 0 and rounded >= 1 else ""
594 return f"{sign}{rounded // 100}.{rounded % 100:02d}"

685 def load(self, lines):

779 def filter_elements(self, argument_part):
780 parts = [] if argument_part is None else argument_part.split(" ", 3)
781 has_value = len(parts) == 4
782 if len(parts) < 3 or has_value == (parts[2] == "EXISTS"):
783 raise CommandError("bad arguments")
784 segments = parse_path(parts[0])
785 field = parse_field(parts[1])
786 operator = parts[2]
787 if operator not in OPERATORS:
788 raise CommandError("bad operator")
789 reference = JsonReader(parts[3]).parse() if has_value else None
790 matches = evaluate(self.document, segments)
791 require_kind(matches, ARRAY, "not an array")
792 found = False
793 for match in matches:
794 for position in range(len(match.node.items)):
795 element = match.child(position)
796 if filter_holds(operator, field_value(element.node, field), reference):
797 self.emit(element.location(), "=", compact(element.node))
798 found = True
799 if not found:
800 self.emit("no match")

802 def set_value(self, argument_part):
803 path, value_text = split_arguments(argument_part, 2)
804 segments = parse_path(path)
805 value = JsonReader(value_text).parse()
806 if not segments:
807 self.commit(value)
808 self.emit("SET", 1)
809 return
810 last = segments[-1]
811 targets = []
812 for parent in evaluate(self.document, segments[:-1]):
813 if last.kind == KEY and not last.descendant and parent.node.kind == OBJECT:
814 position = parent.node.find(last.key)
815 if position < 0:
816 position = len(parent.node.items)
817 targets.append(Match(None, parent.steps + [Step(last.key, position)]))
818 else:
819 targets.extend(apply_segment(last, [parent]))
820 targets = reverse_document_order(targets)
821 document = self.document.copy()
822 for target in targets:
823 container, position = locate(document, target)
824 if position == len(container.items):
825 container.keys.append(target.steps[-1].key)
826 container.items.append(value.copy())
827 else:
828 container.items[position] = value.copy()
829 self.commit(document)
830 self.emit("SET", len(targets))

847 def insert(self, argument_part):
848 path, index_text, value_text = split_arguments(argument_part, 3)
849 segments = parse_path(path)
850 if not is_integer_notation(index_text):
851 raise CommandError("bad index")
852 index = int(index_text)
853 value = JsonReader(value_text).parse()
854 matches = evaluate(self.document, segments)
855 require_kind(matches, ARRAY, "not an array")

857 size = len(match.node.items)

860 targets = reverse_document_order(matches)
861 document = self.document.copy()
862 for target in targets:
863 array = resolve(document, target.positions())
864 position = index if index >= 0 else len(array.items) + 1 + index

866 self.commit(document)

884 def sort(self, argument_part):
885 arguments = words(argument_part, 1, 3)
886 options = arguments[1:]
887 descending = len(options) > 0 and options[-1] == "DESC"
888 if descending:
889 options.pop()
890 if len(options) > 1 or "DESC" in options:
891 raise CommandError("bad arguments")

893 field = parse_field(options[0] if options else "@")

899 array = resolve(document, target.positions())
900 array.items.sort(
901 key=lambda item: sort_key(field_value(item, field)), reverse=descending
902 )

906 def rename(self, argument_part):
907 path, key_text = words(argument_part, 2, 2)
908 segments = parse_path(path)
909 new_key = parse_key(key_text)
910 matches = evaluate(self.document, segments)
911 for match in matches:
912 if not match.steps or match.steps[-1].key is None:
913 raise CommandError(f"not a member at {match.location()}")
914 targets = reverse_document_order(matches)
915 document = self.document.copy()
916 for target in targets:
917 container, position = locate(document, target)
918 if container.keys[position] != new_key and new_key in container.keys:
919 raise CommandError(f"key exists at {target.location()}")
920 container.keys[position] = new_key
921 self.commit(document)
922 self.emit("RENAME", len(targets))
