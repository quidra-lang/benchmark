87 va_list args, saved;
88 va_start(args, pattern);
89 va_copy(saved, args);
90 int n = vsnprintf(NULL, 0, pattern, args);
91 va_end(args);
92 char *s = allocate((size_t)n + 1);
93 vsnprintf(s, (size_t)n + 1, pattern, saved);

119static void append_n(Buffer *b, const char *text, size_t n) {
120 if (b->length + n + 1 > b->capacity) {
121 b->capacity = (b->length + n + 1) * 2;
122 b->data = resize(b->data, b->capacity);
123 }
124 memcpy(b->data + b->length, text, n);
125 b->length += n;
126 b->data[b->length] = 0;

149static bool identifier(const char *s) {
150 if (!letter(*s) && *s != '_') return false;
151 for (s++; *s; s++) if (!letter(*s) && !digit(*s) && *s != '_') return false;

155static bool integer_notation(const char *s) {
156 if (!strcmp(s, "0")) return true;
157 if (*s == '-') s++;
158 size_t n = strlen(s);
159 if (!n || n > 9 || *s == '0') return false;
160 for (; *s; s++) if (!digit(*s)) return false;
161 return true;
162}

165 return digit(c) ? c - '0' : c >= 'A' && c <= 'F' ? c - 'A' + 10 : c >= 'a' && c <= 'f' ? c - 'a' + 10 : -1;

168static char *quote(const char *s, char delimiter) {
169 Buffer b = {0};
170 character(&b, delimiter);
171 for (; *s; s++) {
172 if (*s == delimiter || *s == '\\') {
173 character(&b, '\\');
174 character(&b, *s);
175 } else switch (*s) {
176 case '\b':
177 append(&b, "\\b");
178 break;
179 case '\f':
180 append(&b, "\\f");
181 break;
182 case '\n':
183 append(&b, "\\n");
184 break;
185 case '\r':
186 append(&b, "\\r");
187 break;
188 case '\t':
189 append(&b, "\\t");
190 break;
191 default:
192 if ((unsigned char)*s < 32 || *s == 127) append(&b, format("\\u%04x", (unsigned char)*s));
193 else character(&b, *s);
194 }
195 }
196 character(&b, delimiter);
197 return finish(&b);
198}

245static char *read_string(Parser *p, char delimiter) {
246 p->position++;
247 Buffer b = {0};
248 while (true) {
249 if (p->position == p->length || peek(p) == '\n') parser_fail(p, p->position, "unterminated string");
250 char c = peek(p);
251 if (c == delimiter) {
252 p->position++;
253 return finish(&b);
254 }
255 if (c != '\\') {
256 character(&b, c);
257 p->position++;
258 continue;
259 }
260 size_t start = p->position++;
261 if (p->position == p->length || peek(p) == '\n') parser_fail(p, p->position, "unterminated string");
262 char escaped = peek(p);
263 if (escaped == 'u') {
264 if (p->position + 4 >= p->length) parser_fail(p, start, "bad unicode escape");
265 int value = 0;
266 for (size_t i = 1; i <= 4; i++) {
267 int h = hex(p->text[p->position + i]);
268 if (h < 0) parser_fail(p, start, "bad unicode escape");
269 value = value * 16 + h;
270 }
271 if (value > 127) parser_fail(p, start, "non-ascii escape");
272 character(&b, (char)value);
273 p->position += 5;
274 } else {
275 char value = 0;
276 switch (escaped) {
277 case '\\':
278 case '/':
279 value = escaped;
280 break;
281 case 'b':
282 value = '\b';
283 break;
284 case 'f':
285 value = '\f';
286 break;
287 case 'n':
288 value = '\n';
289 break;
290 case 'r':
291 value = '\r';
292 break;
293 case 't':
294 value = '\t';
295 break;
296 default:
297 if (escaped == delimiter) value = delimiter;
298 }
299 if (!value) parser_fail(p, start, format("invalid escape '\\%c'", escaped));
300 character(&b, value);
301 p->position++;
302 }
303 }
304}

306static Value *read_value(Parser *p) {
307 whitespace(p);
308 need_character(p);
309 char c = peek(p);
310 if (c == '"') return string_value(read_string(p, '"'));
311 if (c == '{') {
312 p->position++;
313 whitespace(p);
314 Value *v = make(OBJECT);
315 if (peek(p) == '}') {
316 p->position++;
317 return v;
318 }
319 while (true) {
320 whitespace(p);
321 need_character(p);
322 if (peek(p) != '"') parser_fail(p, p->position, "expected key");
323 size_t start = p->position;
324 char *key = read_string(p, '"');
325 if (entry_index(v, key) >= 0) parser_fail(p, start, format("duplicate key %s", quote(key, '"')));
326 whitespace(p);
327 need_character(p);
328 if (peek(p) != ':') parser_fail(p, p->position, "expected ':'");
329 p->position++;
330 put(v, key, read_value(p));
331 whitespace(p);
332 if (peek(p) == '}') {
333 p->position++;
334 return v;
335 }
336 need_character(p);
337 if (peek(p) != ',') parser_fail(p, p->position, "expected ',' or '}'");
338 p->position++;
339 whitespace(p);
340 if (peek(p) == '}') parser_fail(p, p->position, "trailing comma");
341 }
342 }
343 if (c == '[') {
344 p->position++;
345 whitespace(p);
346 Value *v = make(ARRAY);
347 if (peek(p) == ']') {
348 p->position++;
349 return v;
350 }
351 while (true) {
352 push(v, read_value(p));
353 whitespace(p);
354 if (peek(p) == ']') {
355 p->position++;
356 return v;
357 }
358 need_character(p);
359 if (peek(p) != ',') parser_fail(p, p->position, "expected ',' or ']'");
360 p->position++;
361 whitespace(p);
362 if (peek(p) == ']') parser_fail(p, p->position, "trailing comma");
363 }
364 }
365 if (c == '-' || digit(c)) {
366 size_t start = p->position;
367 if (c == '-') p->position++;
368 size_t digits = p->position;
369 while (digit(peek(p))) p->position++;
370 size_t n = p->position - digits;
371 if (!n) parser_fail(p, start, "invalid number");
372 if (n >= 2 && p->text[digits] == '0') parser_fail(p, digits, "leading zero");
373 if (peek(p) == '.') parser_fail(p, p->position, "fraction not allowed");
374 if (peek(p) == 'e' || peek(p) == 'E') parser_fail(p, p->position, "exponent not allowed");
375 if (n >= 10) parser_fail(p, start, "number out of range");
376 return number_value(atoi(slice(p->text + start, p->position - start)));
377 }
378 if (letter(c)) {
379 size_t start = p->position;
380 while (letter(peek(p))) p->position++;
381 char *word = slice(p->text + start, p->position - start);
382 if (!strcmp(word, "null")) return make(NIL);
383 if (!strcmp(word, "true") || !strcmp(word, "false")) {
384 Value *v = make(BOOLEAN);
385 v->number = !strcmp(word, "true");
386 return v;
387 }
388 parser_fail(p, start, format("invalid literal '%s'", word));
389 }
390 parser_fail(p, p->position, format("unexpected character '%c'", c));
391 return NULL;
392}

403 Value *out = make(v->kind);

411static char *compact(Value *v) {
412 if (v->kind == NIL) return "null";
413 if (v->kind == BOOLEAN) return v->number ? "true" : "false";
414 if (v->kind == NUMBER) return format("%d", v->number);
415 if (v->kind == STRING) return quote(v->text, '"');
416 Buffer b = {0};
417 character(&b, v->kind == OBJECT ? '{' : '[');
418 for (size_t i = 0; i < v->count; i++) {
419 if (i) character(&b, ',');
420 if (v->kind == OBJECT) {
421 append(&b, quote(v->entries[i].key, '"'));
422 character(&b, ':');
423 }
424 append(&b, compact(v->kind == OBJECT ? v->entries[i].value : v->items[i]));
425 }
426 character(&b, v->kind == OBJECT ? '}' : ']');
427 return finish(&b);
428}

430static char *pretty(Value *v, const char *indent) {
431 if ((v->kind != ARRAY && v->kind != OBJECT) || !v->count) return compact(v);
432 Buffer b = {0};
433 char *child = format("%s ", indent);
434 character(&b, v->kind == OBJECT ? '{' : '[');
435 character(&b, '\n');
436 for (size_t i = 0; i < v->count; i++) {
437 if (i) append(&b, ",\n");
438 append(&b, child);
439 if (v->kind == OBJECT) append(&b, format("%s: ", quote(v->entries[i].key, '"')));
440 append(&b, pretty(v->kind == OBJECT ? v->entries[i].value : v->items[i], child));
441 }
442 append(&b, format("\n%s%c", indent, v->kind == OBJECT ? '}' : ']'));
443 return finish(&b);
444}

471static Path path(const char *text, bool field) {
472 Parser p = {text, 0, strlen(text), field ? "ERROR bad field" : "ERROR bad path"};
473 Path out = {0};
474 if (peek(&p) != (field ? '@' : '$')) fail("%s", p.path_error);
475 p.position++;
476 while (p.position < p.length) {
477 Selector s = {0};
478 if (peek(&p) == '.') {
479 p.position++;
480 if (peek(&p) == '.') {
481 if (field) fail("%s", p.path_error);
482 s.descendant = true;
483 p.position++;
484 }
485 if (peek(&p) == '*') {
486 if (field) fail("%s", p.path_error);
487 p.position++;
488 s.kind = 'A';
489 APPEND(out.items, out.count, s);
490 continue;
491 }
492 if (!(s.descendant && peek(&p) == '[')) {
493 s.kind = 'K';
494 s.key = read_identifier(&p);
495 APPEND(out.items, out.count, s);
496 continue;
497 }
498 } else if (peek(&p) != '[') fail("%s", p.path_error);
499 if (peek(&p) != '[') fail("%s", p.path_error);
500 p.position++;
501 if (peek(&p) == '\'') {
502 s.kind = 'K';
503 s.key = read_string(&p, '\'');
504 if (peek(&p) != ']') fail("%s", p.path_error);
505 p.position++;
506 APPEND(out.items, out.count, s);
507 continue;
508 }
509 const char *end = strchr(text + p.position, ']');
510 if (!end) fail("%s", p.path_error);
511 char *content = slice(text + p.position, (size_t)(end - text) - p.position);
512 p.position = (size_t)(end - text) + 1;
513 if (!strcmp(content, "*") && !field) s.kind = 'A';
514 else if (integer_notation(content)) {
515 s.kind = 'I';
516 s.first = atoi(content);
517 s.has_first = true;
518 } else if (!field && strchr(content, ':')) {
519 char *colon = strchr(content, ':'), *first = slice(content, (size_t)(colon - content)), *last = colon + 1;
520 if (strchr(last, ':') || (*first && !integer_notation(first)) || (*last && !integer_notation(last))) fail("%s", p.path_error);
521 s.kind = 'S';
522 s.has_first = *first;
523 s.has_last = *last;
524 s.first = atoi(first);
525 s.last = atoi(last);
526 } else fail("%s", p.path_error);
527 APPEND(out.items, out.count, s);
528 }
529 return out;
530}

541static Match child(Match parent, Step step) {
542 Match out = { .count = parent.count + 1 };
543 out.steps = allocate(out.count * sizeof(*out.steps));
544 if (parent.count) memcpy(out.steps, parent.steps, parent.count * sizeof(*out.steps));
545 out.steps[parent.count] = step;
546 out.value = step.member ? parent.value->entries[step.position].value : parent.value->items[step.index];
547 return out;
548}

550static Matches children(Match parent) {
551 Matches out = {0};
552 Value *v = parent.value;
553 if (v->kind == OBJECT || v->kind == ARRAY) for (size_t i = 0; i < v->count; i++) {
554 Step step = {.key = v->kind == OBJECT ? v->entries[i].key : NULL, .index = i, .position = i, .member = v->kind == OBJECT};
555 APPEND(out.items, out.count, child(parent, step));
556 }
557 return out;
558}

560static void preorder(Match match, Matches *out) {
561 APPEND(out->items, out->count, match);
562 Matches kids = children(match);
563 for (size_t i = 0; i < kids.count; i++) preorder(kids.items[i], out);
564}

566static Matches select_values(Match match, Selector s) {
567 Matches out = {0};
568 Value *v = match.value;
569 if (s.kind == 'K' && v->kind == OBJECT) {
570 int index = entry_index(v, s.key);
571 if (index >= 0) APPEND(out.items, out.count, child(match, ((Step){s.key, 0, (size_t)index, true})));
572 } else if (s.kind == 'A') out = children(match);
573 else if (v->kind == ARRAY && (s.kind == 'I' || s.kind == 'S')) {
574 int n = (int)v->count;
575 if (s.kind == 'I') {
576 int i = s.first < 0 ? n + s.first : s.first;
577 if (i >= 0 && i < n) APPEND(out.items, out.count, child(match, ((Step){NULL, (size_t)i, (size_t)i, false})));
578 } else {
579 int a = s.has_first ? s.first : 0, b = s.has_last ? s.last : n;
580 if (a < 0) a += n;
581 if (b < 0) b += n;
582 if (a < 0) a = 0;
583 if (a > n) a = n;
584 if (b < 0) b = 0;
585 if (b > n) b = n;
586 for (int i = a; i < b; i++) APPEND(out.items, out.count, child(match, ((Step){NULL, (size_t)i, (size_t)i, false})));
587 }
588 }
589 return out;
590}

592static Matches apply(Matches matches, Selector s) {
593 Matches out = {0};
594 for (size_t i = 0; i < matches.count; i++) {
595 Matches bases = {0};
596 if (s.descendant) preorder(matches.items[i], &bases);
597 else APPEND(bases.items, bases.count, matches.items[i]);
598 for (size_t j = 0; j < bases.count; j++) {
599 Matches values = select_values(bases.items[j], s);
600 for (size_t k = 0; k < values.count; k++) APPEND(out.items, out.count, values.items[k]);
601 }
602 }
603 return out;
604}

639static bool equal(Value *a, Value *b) {
640 if (a->kind != b->kind) return false;
641 if (a->kind == NIL) return true;
642 if (a->kind == NUMBER || a->kind == BOOLEAN) return a->number == b->number;
643 if (a->kind == STRING) return !strcmp(a->text, b->text);
644 if (a->count != b->count) return false;
645 for (size_t i = 0; i < a->count; i++) {
646 if (a->kind == ARRAY) {
647 if (!equal(a->items[i], b->items[i])) return false;
648 } else {
649 int j = entry_index(b, a->entries[i].key);
650 if (j < 0 || !equal(a->entries[i].value, b->entries[j].value)) return false;
651 }
652 }
653 return true;
654}

667static void merge(Value *target, Value *source) {
668 for (size_t i = 0; i < source->count; i++) {
669 Entry e = source->entries[i];
670 int j = entry_index(target, e.key);
671 if (j >= 0 && target->entries[j].value->kind == OBJECT && e.value->kind == OBJECT) merge(target->entries[j].value, e.value);
672 else put(target, e.key, clone(e.value));
673 }
674}

676static int compare_locations(const Match *a, const Match *b) {
677 size_t n = a->count < b->count ? a->count : b->count;
678 for (size_t i = 0; i < n; i++) if (a->steps[i].position != b->steps[i].position) return a->steps[i].position > b->steps[i].position ? 1 : -1;
679 return (a->count > b->count) - (a->count < b->count);
680}
681
682static int reverse_locations(const void *a, const void *b) {
683 return compare_locations(b, a);
684}

686static Matches targets(Matches matches) {
687 Matches out = {0};
688 if (matches.count) qsort(matches.items, matches.count, sizeof(*matches.items), reverse_locations);
689 for (size_t i = 0; i < matches.count; i++) if (!i || compare_locations(matches.items + i, matches.items + i - 1)) APPEND(out.items, out.count, matches.items[i]);
690 return out;
691}

693static void require_type(Matches matches, Kind kind) {
694 for (size_t i = 0; i < matches.count; i++) if (matches.items[i].value->kind != kind) fail("ERROR not an %s at %s", kind == ARRAY ? "array" : "object", location(matches.items[i]));
695}

697static bool one_of(const char *word, const char *words) {
698 if (!*word || strchr(word, ' ')) return false;
699 return strstr(words, format(" %s ", word)) != NULL;

702static char **arguments(const char *text, size_t limit, size_t *count) {
703 char **out = NULL;
704 *count = 0;
705 if (!text) return out;
706 size_t start = 0;
707 for (size_t i = 0;; i++) if (!text[i] || (text[i] == ' ' && *count + 1 < limit)) {
708 APPEND(out, *count, slice(text + start, i - start));
709 start = i + 1;
710 if (!text[i]) break;
711 }
712 return out;
713}

720static char *execute(const char *line) {
721 const char *space = strchr(line, ' '), *argument = space ? space + 1 : NULL;
722 char *command = space ? slice(line, (size_t)(space - line)) : copy_text(line);
723 if (!one_of(command, " GET COMPACT PRETTY TYPE LENGTH KEYS FLATTEN COUNT SUM MIN MAX AVG FILTER SET DELETE INSERT MERGE SORT RENAME UNDO ")) fail("ERROR unknown command '%s'", command);
724 size_t argc, limit = one_of(command, " SET MERGE ") ? 2 : !strcmp(command, "INSERT") ? 3 : !strcmp(command, "FILTER") ? 4 : SIZE_MAX;
725 char **args = arguments(argument, limit, &argc);
726 bool shape = one_of(command, " SET MERGE RENAME ") ? argc == 2 : !strcmp(command, "INSERT") ? argc == 3 : !strcmp(command, "FILTER") ? argc >= 3 && (!strcmp(args[2], "EXISTS") ? argc == 3 : argc == 4) : one_of(command, " COMPACT PRETTY FLATTEN ") ? argc <= 1 : !strcmp(command, "UNDO") ? argc == 0 : !strcmp(command, "SORT") ? argc >= 1 && argc <= 3 && (argc < 3 || (strcmp(args[1], "DESC") && !strcmp(args[2], "DESC"))) : argc == 1;
727 if (!shape) fail("ERROR bad arguments");
728 if (!strcmp(command, "UNDO")) {
729 if (!history_count) fail("ERROR nothing to undo");
730 document = history[--history_count];
731 return format("UNDO %zu\n", history_count);
732 }
733 Path p = path(argc ? args[0] : "$", false), fields = {0};
734 Value *value = &nil;
735 int index = 0;
736 char *rename = NULL, *op = NULL;
737 if (!strcmp(command, "FILTER")) {
738 fields = path(args[1], true);
739 op = args[2];
740 if (!one_of(op, " == != < <= > >= EXISTS ")) fail("ERROR bad operator");
741 if (strcmp(op, "EXISTS")) value = json(args[3]);
742 } else if (!strcmp(command, "SORT")) fields = path(argc >= 2 && strcmp(args[1], "DESC") ? args[1] : "@", true);
743 else if (!strcmp(command, "RENAME")) rename = rename_key(args[1]);
744 else if (!strcmp(command, "INSERT")) {
745 if (!integer_notation(args[1])) fail("ERROR bad index");
746 index = atoi(args[1]);
747 value = json(args[2]);
748 } else if (one_of(command, " SET MERGE ")) value = json(args[1]);
749 if (!strcmp(command, "MERGE") && value->kind != OBJECT) fail("ERROR merge value is not an object");
750 if (!strcmp(command, "DELETE") && !p.count) fail("ERROR cannot delete root");
751 Matches matches = evaluate(p);
752 if (one_of(command, " FILTER INSERT SORT ")) require_type(matches, ARRAY);
753 if (one_of(command, " KEYS MERGE ")) require_type(matches, OBJECT);
754 if (!strcmp(command, "LENGTH")) for (size_t i = 0; i < matches.count; i++) {
755 Kind kind = matches.items[i].value->kind;
756 if (kind != OBJECT && kind != ARRAY && kind != STRING) fail("ERROR no length at %s", location(matches.items[i]));
757 }
758 if (!strcmp(command, "INSERT")) for (size_t i = 0; i < matches.count; i++) {
759 int n = (int)matches.items[i].value->count;
760 if (index < -n - 1 || index > n) fail("ERROR index out of range at %s", location(matches.items[i]));
761 }
762 if (!strcmp(command, "RENAME")) for (size_t i = 0; i < matches.count; i++) if (!matches.items[i].count || !matches.items[i].steps[matches.items[i].count - 1].member) fail("ERROR not a member at %s", location(matches.items[i]));
763 if (one_of(command, " SET DELETE INSERT MERGE SORT RENAME ")) {
764 if (!strcmp(command, "SET") && p.count) {
765 Selector last = p.items[p.count - 1];
766 if (last.kind == 'K' && !last.descendant) {
767 Path parents_path = p;
768 parents_path.count--;
769 Matches parents = evaluate(parents_path);
770 matches = (Matches){0};
771 for (size_t i = 0; i < parents.count; i++) {
772 Match parent = parents.items[i];
773 if (parent.value->kind != OBJECT) continue;
774 int position = entry_index(parent.value, last.key);
775 Match target = { .value = &nil, .count = parent.count + 1 };
776 target.steps = allocate(target.count * sizeof(*target.steps));
777 if (parent.count) memcpy(target.steps, parent.steps, parent.count * sizeof(*target.steps));
778 target.steps[parent.count] = (Step){last.key, 0, position < 0 ? parent.value->count : (size_t)position, true};
779 APPEND(matches.items, matches.count, target);
780 }
781 }
782 }
783 Matches places = targets(matches);
784 Value *changed = clone(document);
785 for (size_t i = 0; i < places.count; i++) {
786 Match target = places.items[i];
787 Value *current = !strcmp(command, "SET") ? NULL : at(changed, target.steps, target.count);
788 if (!strcmp(command, "SET")) {
789 if (!target.count) changed = clone(value);
790 else {
791 Value *parent = at(changed, target.steps, target.count - 1);
792 Step s = target.steps[target.count - 1];
793 if (s.member) put(parent, s.key, clone(value));
794 else parent->items[s.index] = clone(value);
795 }
796 } else if (!strcmp(command, "DELETE")) {
797 Value *parent = at(changed, target.steps, target.count - 1);
798 Step s = target.steps[target.count - 1];
799 size_t j = s.member ? (size_t)entry_index(parent, s.key) : s.index;
800 if (s.member) memmove(parent->entries + j, parent->entries + j + 1, (parent->count - j - 1) * sizeof(*parent->entries));
801 else memmove(parent->items + j, parent->items + j + 1, (parent->count - j - 1) * sizeof(*parent->items));
802 parent->count--;
803 } else if (!strcmp(command, "INSERT")) {
804 size_t j = index < 0 ? (size_t)((int)current->count + 1 + index) : (size_t)index;
805 current->items = resize(current->items, (current->count + 1) * sizeof(*current->items));
806 memmove(current->items + j + 1, current->items + j, (current->count - j) * sizeof(*current->items));
807 current->items[j] = clone(value);
808 current->count++;
809 } else if (!strcmp(command, "MERGE")) merge(current, value);
810 else if (!strcmp(command, "SORT")) {
811 int direction = !strcmp(args[argc - 1], "DESC") ? -1 : 1;
812 for (size_t j = 1; j < current->count; j++) {
813 Value *item = current->items[j], *key = field(item, fields);
814 size_t k = j;
815 while (k && direction * compare(field(current->items[k - 1], fields), key) > 0) {
816 current->items[k] = current->items[k - 1];
817 k--;
818 }
819 current->items[k] = item;
820 }
821 } else {
822 Value *parent = at(changed, target.steps, target.count - 1);
823 char *old = target.steps[target.count - 1].key;
824 if (strcmp(old, rename) && entry_index(parent, rename) >= 0) fail("ERROR key exists at %s", location(target));
825 parent->entries[entry_index(parent, old)].key = rename;
826 }
827 }
828 APPEND(history, history_count, clone(document));
829 document = changed;
830 return format("%s %zu\n", command, places.count);
831 }
832 if (!strcmp(command, "COUNT")) return format("COUNT %zu\n", matches.count);
833 if (one_of(command, " SUM MIN MAX AVG ")) {
834 int64_t total = 0;
835 int minimum = 0, maximum = 0;
836 size_t count = 0;
837 for (size_t i = 0; i < matches.count; i++) if (matches.items[i].value->kind == NUMBER) {
838 int n = matches.items[i].value->number;
839 total += n;
840 if (!count || n < minimum) minimum = n;
841 if (!count || n > maximum) maximum = n;
842 count++;
843 }
844 char *result = !strcmp(command, "SUM") ? format("%lld", (long long)total) : !count ? "null" : format("%d", !strcmp(command, "MIN") ? minimum : maximum);
845 if (!strcmp(command, "AVG")) {
846 if (!count) result = "null";
847 else {
848 int64_t q = ((total < 0 ? -total : total) * 200 + (int64_t)count) / (2 * (int64_t)count);
849 result = format("%s%lld.%02lld", total < 0 && q ? "-" : "", (long long)(q / 100), (long long)(q % 100));
850 }
851 }
852 return format("%s %s\n", command, result);
853 }
854 Buffer out = {0};
855 if (!strcmp(command, "FILTER")) {
856 for (size_t i = 0; i < matches.count; i++) {
857 Matches kids = children(matches.items[i]);
858 for (size_t j = 0; j < kids.count; j++) {
859 Match child = kids.items[j];
860 Value *f = field(child.value, fields);
861 bool equality = f != &missing && equal(f, value), pass;
862 if (!strcmp(op, "EXISTS")) pass = f != &missing;
863 else if (!strcmp(op, "==") || !strcmp(op, "!=")) pass = (!strcmp(op, "==")) == equality;
864 else {
865 bool comparable = (f->kind == NUMBER && value->kind == NUMBER) || (f->kind == STRING && value->kind == STRING);
866 int order = comparable ? compare(f, value) : 0;
867 pass = comparable && (!strcmp(op, "<") ? order < 0 : !strcmp(op, "<=") ? order <= 0 : !strcmp(op, ">") ? order > 0 : order >= 0);
868 }
869 if (pass) add_output(&out, format("%s = %s", location(child), compact(child.value)));
870 }
871 }
872 } else for (size_t i = 0; i < matches.count; i++) {
873 Match match = matches.items[i];
874 Value *v = match.value;
875 char *place = location(match);
876 if (!strcmp(command, "GET")) add_output(&out, format("%s = %s", place, compact(v)));
877 else if (!strcmp(command, "COMPACT")) add_output(&out, compact(v));
878 else if (!strcmp(command, "PRETTY")) add_output(&out, pretty(v, ""));
879 else if (!strcmp(command, "TYPE")) add_output(&out, format("%s %s", place, type(v)));
880 else if (!strcmp(command, "LENGTH")) add_output(&out, format("%s %zu", place, v->kind == STRING ? strlen(v->text) : v->count));
881 else if (!strcmp(command, "KEYS")) {
882 Value *keys = make(ARRAY);
883 for (size_t j = 0; j < v->count; j++) push(keys, string_value(v->entries[j].key));
884 add_output(&out, format("%s %s", place, compact(keys)));
885 } else {
886 Matches leaves = {0};
887 preorder(match, &leaves);
888 for (size_t j = 0; j < leaves.count; j++) if (!children(leaves.items[j]).count) add_output(&out, format("%s = %s", location(leaves.items[j]), compact(leaves.items[j].value)));
889 }
890 }
891 return out.length ? finish(&out) : "no match\n";
892}

894static void size(Value *v, size_t *count, size_t *depth, size_t level) {
895 (*count)++;
896 if (level > *depth) *depth = level;
897 if (v->kind == ARRAY || v->kind == OBJECT) for (size_t i = 0; i < v->count; i++) size(v->kind == ARRAY ? v->items[i] : v->entries[i].value, count, depth, level + 1);
898}

900static char *read_line(void) {
901 Buffer b = {0};
902 int c;
903 while ((c = getchar()) != EOF && c != '\n') character(&b, (char)c);
904 if (c == EOF && !b.length) return NULL;
905 return finish(&b);

908static void load(void) {
909 Buffer text = {0};
910 char *line;
911 bool first = true;
912 while ((line = read_line()) && strcmp(line, "%%")) {
913 if (!first) character(&text, '\n');
914 append(&text, line);
915 first = false;
916 }
917 char *source = finish(&text);
918 if (setjmp(failure)) {
919 size_t line_number = 1, column = 1;
920 for (size_t i = 0; i < error_position; i++) if (source[i] == '\n') {
921 line_number++;
922 column = 1;
923 } else column++;
924 printf("PARSE ERROR line %zu col %zu: %s\n", line_number, column, error_reason);
925 return;
926 }
927 Value *value = json(source);
928 APPEND(history, history_count, clone(document));
929 document = value;
930 size_t count = 0, depth = 0;
931 size(value, &count, &depth, 1);
932 printf("LOADED nodes=%zu depth=%zu\n", count, depth);
933}

935int main(void) {
936 load();
937 char *line;
938 while ((line = read_line()) && strcmp(line, "END")) {
939 if (!strcmp(line, "LOAD")) load();
940 else if (setjmp(failure)) {
941 if (parse_error) printf("PARSE ERROR col %zu: %s\n", error_position + 1, error_reason);
942 else puts(error_reason);
943 } else fputs(execute(line), stdout);
944 }
945 return 0;
946}
