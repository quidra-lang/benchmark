56String quote(String text, [String delimiter = '"']) {
57 final result = StringBuffer(delimiter);
58 const escapes = {
59 '\b': r'\b',
60 '\f': r'\f',
61 '\n': r'\n',
62 '\r': r'\r',
63 '\t': r'\t',
64 };
65 for (final char in text.split('')) {
66 if (char == delimiter || char == r'\') {
67 result.write(r'\');
68 result.write(char);
69 } else if (escapes.containsKey(char)) {
70 result.write(escapes[char]);
71 } else if (char.codeUnitAt(0) < 32 || char.codeUnitAt(0) == 127) {
72 result.write(
73 '\\u${char.codeUnitAt(0).toRadixString(16).padLeft(4, '0')}',
74 );
75 } else {
76 result.write(char);
77 }
78 }
79 result.write(delimiter);
80 return result.toString();
81}

118 final String text;

121 String current() => position < text.length ? text[position] : '';
122 Never fail(String reason, [int? at]) =>
123 throw JsonFailure(at ?? position, reason);

125 while (current() == ' ' || current() == '\n') position++;

128 String string([String delimiter = '"']) {
129 position++;
130 final result = StringBuffer();
131 while (true) {
132 final char = current();
133 if (char.isEmpty || char == '\n') fail('unterminated string');
134 if (char == delimiter) {
135 position++;
136 return result.toString();
137 }
138 if (char != r'\') {
139 result.write(char);
140 position++;
141 continue;
142 }
143 final start = position;
144 final escaped = start + 1 < text.length ? text[start + 1] : '';
145 if (escaped.isEmpty || escaped == '\n')
146 fail('unterminated string', start + 1);
147 if (escaped == 'u') {
148 if (start + 6 > text.length) fail('bad unicode escape', start);
149 final digits = text.substring(start + 2, start + 6);
150 if (!RegExp(r'^[0-9A-Fa-f]{4}$').hasMatch(digits))
151 fail('bad unicode escape', start);
152 final code = int.parse(digits, radix: 16);
153 if (code > 127) fail('non-ascii escape', start);
154 result.writeCharCode(code);
155 position += 6;
156 } else if (escaped == delimiter || escapeCodes.containsKey(escaped)) {
157 result.write(escaped == delimiter ? delimiter : escapeCodes[escaped]);
158 position += 2;
159 } else {
160 fail("invalid escape '\\$escaped'", start);
161 }
162 }
163 }

165 int number() {
166 final start = position;
167 final negative = current() == '-';
168 if (negative) position++;
169 final digitsStart = position;
170 while (RegExp(r'^[0-9]$').hasMatch(current())) position++;
171 final digits = text.substring(digitsStart, position);
172 if (digits.isEmpty) fail('invalid number', start);
173 if (digits.length > 1 && digits[0] == '0')
174 fail('leading zero', digitsStart);
175 if (current() == '.') fail('fraction not allowed');
176 if (current() == 'e' || current() == 'E') fail('exponent not allowed');
177 if (digits.length >= 10) fail('number out of range', start);
178 return (negative ? -1 : 1) * int.parse(digits);
179 }

181 Value value() {
182 whitespace();
183 final char = current();
184 if (char.isEmpty) fail('unexpected end of input');
185 if (char == '"') return string();
186 if (char == '-' || RegExp(r'^[0-9]$').hasMatch(char)) return number();
187 if (RegExp(r'^[A-Za-z]$').hasMatch(char)) {
188 final start = position;
189 while (RegExp(r'^[A-Za-z]$').hasMatch(current())) position++;
190 final word = text.substring(start, position);
191 if (word == 'null') return null;
192 if (word == 'true' || word == 'false') return word == 'true';
193 fail("invalid literal '$word'", start);
194 }
195 if (char == '{') {
196 position++;
197 whitespace();
198 final object = <String, Value>{};
199 if (current() == '}') {
200 position++;
201 return object;
202 }
203 while (true) {
204 whitespace();
205 if (current().isEmpty) fail('unexpected end of input');
206 if (current() != '"') fail('expected key');
207 final start = position;
208 final key = string();
209 if (object.containsKey(key)) fail('duplicate key ${quote(key)}', start);
210 whitespace();
211 if (current().isEmpty) fail('unexpected end of input');
212 if (current() != ':') fail("expected ':'");
213 position++;
214 object[key] = value();
215 whitespace();
216 if (current() == '}') {
217 position++;
218 return object;
219 }
220 if (current().isEmpty) fail('unexpected end of input');
221 if (current() != ',') fail("expected ',' or '}'");
222 position++;
223 whitespace();
224 if (current() == '}') fail('trailing comma');
225 }
226 }
227 if (char == '[') {
228 position++;
229 whitespace();
230 final array = <Value>[];
231 if (current() == ']') {
232 position++;
233 return array;
234 }
235 while (true) {
236 array.add(value());
237 whitespace();
238 if (current() == ']') {
239 position++;
240 return array;
241 }
242 if (current().isEmpty) fail('unexpected end of input');
243 if (current() != ',') fail("expected ',' or ']'");
244 position++;
245 whitespace();
246 if (current() == ']') fail('trailing comma');
247 }
248 }
249 fail("unexpected character '$char'");
250 }
251
252 Value parse() {
253 final result = value();
254 whitespace();
255 if (current().isNotEmpty) fail('trailing characters');
256 return result;
257 }

260List<Selector> parsePath(String text, {bool field = false}) {
261 try {
262 if (text.isEmpty || text[0] != (field ? '@' : r'$')) throw QueryFailure('');
263 final result = <Selector>[];
264 var position = 1;
265 while (position < text.length) {
266 var descendant = false;
267 var bracket = false;
268 if (!field && text.startsWith('..', position)) {
269 descendant = true;
270 position += 2;
271 bracket = position < text.length && text[position] == '[';
272 } else if (text[position] == '.') {
273 position++;
274 } else if (text[position] == '[') {
275 bracket = true;
276 } else {
277 throw QueryFailure('');
278 }
279 final Selector selector;
280 if (bracket) {
281 position++;
282 if (position < text.length && text[position] == "'") {
283 final reader = JsonReader(text)..position = position;
284 final key = reader.string("'");
285 position = reader.position;
286 if (position == text.length || text[position++] != ']')
287 throw QueryFailure('');
288 selector = Selector('key', key: key, descendant: descendant);
289 } else {
290 final end = text.indexOf(']', position);
291 if (end < 0) throw QueryFailure('');
292 final selection = text.substring(position, end);
293 position = end + 1;
294 if (integerNotation.hasMatch(selection)) {
295 selector = Selector(
296 'index',
297 index: int.parse(selection),
298 descendant: descendant,
299 );
300 } else if (!field && selection == '*') {
301 selector = Selector('all', descendant: descendant);
302 } else if (!field && selection.contains(':')) {
303 final parts = selection.split(':');
304 if (parts.length != 2 ||
305 parts.any(
306 (part) => part.isNotEmpty && !integerNotation.hasMatch(part),
307 ))
308 throw QueryFailure('');
309 selector = Selector(
310 'slice',
311 start: parts[0].isEmpty ? null : int.parse(parts[0]),
312 end: parts[1].isEmpty ? null : int.parse(parts[1]),
313 descendant: descendant,
314 );
315 } else {
316 throw QueryFailure('');
317 }
318 }
319 } else if (!field && position < text.length && text[position] == '*') {
320 position++;
321 selector = Selector('all', descendant: descendant);
322 } else {
323 final name = RegExp(r'^[A-Za-z_][A-Za-z0-9_]*')
324 .firstMatch(text.substring(position));
325 if (name == null) throw QueryFailure('');
326 position += name[0]!.length;
327 selector = Selector('key', key: name[0]!, descendant: descendant);
328 }
329 result.add(selector);
330 }
331 return result;
332 } catch (_) {
333 throw QueryFailure(field ? 'bad field' : 'bad path');
334 }
335}

337List<JsonMatch> children(JsonMatch match) {
338 final value = match.value;

340 var index = 0;

352 return List.generate(
353 value.length,

355 value[index],
356 [...match.path, index],
357 [...match.order, index],

359 );
360 return [];

363List<JsonMatch> preorder(JsonMatch match) {
364 final result = <JsonMatch>[];
365 final pending = [match];
366 while (pending.isNotEmpty) {
367 final next = pending.removeLast();
368 result.add(next);
369 pending.addAll(children(next).reversed);
370 }
371 return result;
372}

374List<JsonMatch> select(JsonMatch match, Selector selector) {
375 final all = children(match);
376 if (selector.kind == 'all') return all;
377 if (selector.kind == 'key')
378 return match.value is Map<String, Value>
379 ? all.where((child) => child.path.last == selector.key).toList()
380 : [];
381 if (match.value is! List<Value>) return [];
382 final length = (match.value as List<Value>).length;
383 if (selector.kind == 'index') {
384 final index = selector.index! < 0
385 ? length + selector.index!
386 : selector.index!;
387 return index >= 0 && index < length ? [all[index]] : [];
388 }
389 int bound(int value) {
390 final adjusted = value < 0 ? length + value : value;
391 return adjusted < 0
392 ? 0
393 : adjusted > length
394 ? length
395 : adjusted;
396 }
397
398 final start = bound(selector.start ?? 0);
399 final end = bound(selector.end ?? length);
400 return start >= end ? [] : all.sublist(start, end);
401}

403List<JsonMatch> applySegment(List<JsonMatch> matches, Selector selector) =>
404 matches
405 .expand(
406 (match) => (selector.descendant ? preorder(match) : [match]).expand(
407 (value) => select(value, selector),
408 ),
409 )
410 .toList();
411List<JsonMatch> evaluate(Value document, List<Selector> selectors) => selectors
412 .fold<List<JsonMatch>>([JsonMatch(document, [], [])], applySegment);
413String location(List<Step> steps) =>
414 r'$' +
415 steps.map((step) {
416 if (step is int) return '[$step]';
417 final key = step as String;
418 return isIdentifier(key) ? '.$key' : '[${quote(key, "'")}]';
419 }).join();
420Value fieldValue(Value value, List<Selector> selectors) {
421 var matches = [JsonMatch(value, [], [])];
422 for (final selector in selectors) matches = applySegment(matches, selector);
423 return matches.isEmpty ? missing : matches[0].value;
424}

426bool equal(Value left, Value right) {
427 if (typeName(left) != typeName(right)) return false;
428 if (left is Map<String, Value> && right is Map<String, Value>)
429 return left.length == right.length &&
430 left.entries.every(
431 (entry) =>
432 right.containsKey(entry.key) &&
433 equal(entry.value, right[entry.key]),
434 );
435 if (left is List<Value> && right is List<Value>)
436 return left.length == right.length &&
437 List.generate(
438 left.length,
439 (i) => equal(left[i], right[i]),
440 ).every((same) => same);
441 return left == right;
442}

444int compareKeys(Value left, Value right) {
445 int rank(Value value) => value == null
446 ? 0
447 : value is int
448 ? 1
449 : value is String
450 ? 2
451 : 3;
452 final difference = rank(left) - rank(right);
453 if (difference != 0) return difference;
454 if (left is int && right is int) return left.compareTo(right);
455 if (left is String && right is String) return left.compareTo(right);
456 return 0;
457}

459List<JsonMatch> targets(List<JsonMatch> matches) {
460 final unique = <String, JsonMatch>{};
461 for (final match in matches) unique[jsonEncode(match.path)] = match;
462 return unique.values.toList()..sort((a, b) {
463 final length = a.order.length < b.order.length
464 ? a.order.length
465 : b.order.length;
466 for (var i = 0; i < length; i++) {
467 if (a.order[i] != b.order[i]) return b.order[i].compareTo(a.order[i]);
468 }
469 return b.order.length.compareTo(a.order.length);
470 });
471}

506 final result = <String>[];
507 var start = 0;
508 while (result.length < maximum) {
509 final space = text.indexOf(' ', start);
510 if (space < 0) break;
511 result.add(text.substring(start, space));
512 start = space + 1;

514 result.add(text.substring(start));

541 void execute(String line) {
542 final space = line.indexOf(' ');
543 final command = space < 0 ? line : line.substring(0, space);
544 final argument = space < 0 ? null : line.substring(space + 1);
545 const commands = {
546 'LOAD',
547 'GET',
548 'COMPACT',
549 'PRETTY',
550 'TYPE',
551 'LENGTH',
552 'KEYS',
553 'FLATTEN',
554 'COUNT',
555 'SUM',
556 'MIN',
557 'MAX',
558 'AVG',
559 'FILTER',
560 'SET',
561 'DELETE',
562 'INSERT',
563 'MERGE',
564 'SORT',
565 'RENAME',
566 'UNDO',
567 };
568 if (!commands.contains(command))
569 throw QueryFailure("unknown command '$command'");
570 var args = argument == null ? <String>[] : argument.split(' ');
571 if (command == 'SET' || command == 'MERGE')
572 args = argument == null ? [] : splitLimited(argument, 1);
573 if (command == 'INSERT')
574 args = argument == null ? [] : splitLimited(argument, 2);
575 if (command == 'FILTER')
576 args = argument == null ? [] : splitLimited(argument, 3);
577 final optional = ['COMPACT', 'PRETTY', 'FLATTEN'].contains(command);
578 final shape = optional
579 ? args.length <= 1
580 : command == 'UNDO' || command == 'LOAD'
581 ? args.isEmpty
582 : command == 'SET' || command == 'MERGE' || command == 'RENAME'
583 ? args.length == 2
584 : command == 'INSERT'
585 ? args.length == 3
586 : command == 'FILTER'
587 ? args.length >= 3 &&
588 (args[2] == 'EXISTS' ? args.length == 3 : args.length == 4)
589 : command == 'SORT'
590 ? args.isNotEmpty &&
591 args.length <= 3 &&
592 (args.length < 3 || args[1] != 'DESC' && args[2] == 'DESC')
593 : args.length == 1;
594 if (!shape) throw QueryFailure('bad arguments');
595 if (command == 'UNDO') {
596 if (history.isEmpty) throw QueryFailure('nothing to undo');
597 document = clone(history.removeLast());
598 output.add('UNDO ${history.length}');
599 return;
600 }
601 final selectors = parsePath(args.isEmpty ? r'$' : args[0]);
602 List<Selector> field = [];
603 var key = '';
604 Value value;
605 if (command == 'FILTER') {
606 field = parsePath(args[1], field: true);
607 if (!['==', '!=', '<', '<=', '>', '>=', 'EXISTS'].contains(args[2]))
608 throw QueryFailure('bad operator');
609 if (args[2] != 'EXISTS') value = JsonReader(args[3]).parse();
610 } else if (command == 'SORT') {
611 field = parsePath(
612 args.length > 1 && args[1] != 'DESC' ? args[1] : '@',
613 field: true,
614 );
615 } else if (command == 'RENAME') {
616 try {
617 if (isIdentifier(args[1])) {
618 key = args[1];
619 } else {
620 final reader = JsonReader(args[1]);
621 if (reader.current() != "'") throw QueryFailure('');
622 key = reader.string("'");
623 if (reader.position != args[1].length) throw QueryFailure('');
624 }
625 } catch (_) {
626 throw QueryFailure('bad key');
627 }
628 } else if (command == 'SET' || command == 'MERGE') {
629 value = JsonReader(args[1]).parse();
630 } else if (command == 'INSERT') {
631 if (!integerNotation.hasMatch(args[1])) throw QueryFailure('bad index');
632 value = JsonReader(args[2]).parse();
633 }
634 if (command == 'MERGE' && value is! Map<String, Value>)
635 throw QueryFailure('merge value is not an object');
636 if (command == 'DELETE' && selectors.isEmpty)
637 throw QueryFailure('cannot delete root');
638 var matches = evaluate(document, selectors);
639 for (final match in matches) {
640 if (['FILTER', 'INSERT', 'SORT'].contains(command) &&
641 match.value is! List<Value>)
642 throw QueryFailure('not an array at ${location(match.path)}');
643 if (['MERGE', 'KEYS'].contains(command) &&
644 match.value is! Map<String, Value>)
645 throw QueryFailure('not an object at ${location(match.path)}');
646 if (command == 'LENGTH' &&
647 match.value is! Map<String, Value> &&
648 match.value is! List<Value> &&
649 match.value is! String)
650 throw QueryFailure('no length at ${location(match.path)}');
651 if (command == 'RENAME' &&
652 (match.path.isEmpty || match.path.last is! String))
653 throw QueryFailure('not a member at ${location(match.path)}');
654 }
655 if (command == 'INSERT') {
656 final index = int.parse(args[1]);
657 for (final match in matches) {
658 final n = (match.value as List<Value>).length;
659 if (index < -n - 1 || index > n)
660 throw QueryFailure('index out of range at ${location(match.path)}');
661 }
662 }
663 if (command == 'FILTER') {
664 final lines = <String>[];
665 for (final match in matches) {
666 for (final child in children(match)) {
667 final left = fieldValue(child.value, field);
668 final operator = args[2];
669 var accepted = operator == 'EXISTS'
670 ? !identical(left, missing)
671 : operator == '=='
672 ? !identical(left, missing) && equal(left, value)
673 : operator == '!='
674 ? identical(left, missing) || !equal(left, value)
675 : false;
676 if (!['EXISTS', '==', '!='].contains(operator) &&
677 (left is int && value is int ||
678 left is String && value is String)) {
679 final comparison = compareKeys(left, value);
680 accepted = operator == '<'
681 ? comparison < 0
682 : operator == '<='
683 ? comparison <= 0
684 : operator == '>'
685 ? comparison > 0
686 : comparison >= 0;
687 }
688 if (accepted)
689 lines.add('${location(child.path)} = ${compact(child.value)}');
690 }
691 }
692 output.addAll(lines.isEmpty ? ['no match'] : lines);
693 return;
694 }
695 if ([
696 'SET',
697 'DELETE',
698 'INSERT',
699 'MERGE',
700 'SORT',
701 'RENAME',
702 ].contains(command)) {
703 if (command == 'SET' && selectors.isNotEmpty) {
704 final last = selectors.last;
705 if (last.kind == 'key' && !last.descendant) {
706 matches = [];
707 for (final parent in evaluate(
708 document,
709 selectors.sublist(0, selectors.length - 1),
710 )) {
711 if (parent.value is! Map<String, Value>) continue;
712 final object = parent.value as Map<String, Value>;
713 final keys = object.keys.toList();
714 final index = keys.indexOf(last.key!);
715 matches.add(
716 JsonMatch(
717 object[last.key],
718 [...parent.path, last.key!],
719 [...parent.order, index < 0 ? keys.length : index],
720 ),
721 );
722 }
723 }
724 }
725 final selected = targets(matches);
726 Value candidate = clone(document);
727 for (final target in selected) {
728 if (command == 'SET') {
729 candidate = putAt(candidate, target.path, clone(value));
730 } else if (command == 'DELETE') {
731 final parent = getAt(
732 candidate,
733 target.path.sublist(0, target.path.length - 1),
734 );
735 final last = target.path.last;
736 if (last is String) {
737 (parent as Map<String, Value>).remove(last);
738 } else {
739 (parent as List<Value>).removeAt(last as int);
740 }
741 } else if (command == 'INSERT') {
742 final array = getAt(candidate, target.path) as List<Value>;
743 final index = int.parse(args[1]);
744 array.insert(
745 index < 0 ? array.length + 1 + index : index,
746 clone(value),
747 );
748 } else if (command == 'MERGE') {
749 merge(
750 getAt(candidate, target.path) as Map<String, Value>,
751 value as Map<String, Value>,
752 );
753 } else if (command == 'SORT') {
754 final array = getAt(candidate, target.path) as List<Value>;
755 final direction = args.last == 'DESC' ? -1 : 1;
756 final decorated = List.generate(
757 array.length,
758 (i) =>
759 (value: array[i], index: i, key: fieldValue(array[i], field)),
760 );
761 decorated.sort((a, b) {
762 final comparison = direction * compareKeys(a.key, b.key);
763 return comparison == 0 ? a.index.compareTo(b.index) : comparison;
764 });
765 candidate = putAt(
766 candidate,
767 target.path,
768 decorated.map((item) => item.value).toList(),
769 );
770 } else {
771 final parentPath = target.path.sublist(0, target.path.length - 1);
772 final old = target.path.last as String;
773 final object = getAt(candidate, parentPath) as Map<String, Value>;
774 if (old != key && object.containsKey(key))
775 throw QueryFailure('key exists at ${location(target.path)}');
776 candidate = putAt(
777 candidate,
778 parentPath,
779 Map<String, Value>.fromEntries(
780 object.entries.map(
781 (entry) =>
782 MapEntry(entry.key == old ? key : entry.key, entry.value),
783 ),
784 ),
785 );
786 }
787 }
788 history.add(clone(document));
789 document = candidate;
790 output.add('$command ${selected.length}');
791 return;
792 }
793 if (command == 'COUNT') {
794 output.add('COUNT ${matches.length}');
795 return;
796 }
797 if (['SUM', 'MIN', 'MAX', 'AVG'].contains(command)) {
798 final numbers = matches
799 .map((match) => match.value)
800 .whereType<int>()
801 .toList();
802 final sum = numbers.fold<int>(0, (total, number) => total + number);
803 var answer = 'null';
804 if (command == 'SUM') {
805 answer = '$sum';
806 } else if (numbers.isNotEmpty) {
807 if (command == 'AVG') {
808 final rounded =
809 (sum.abs() * 200 + numbers.length) ~/ (2 * numbers.length);
810 answer =
811 '${sum < 0 && rounded > 0 ? '-' : ''}${rounded ~/ 100}.${(rounded % 100).toString().padLeft(2, '0')}';
812 } else {
813 answer = numbers
814 .reduce(
815 (best, number) => command == 'MIN'
816 ? best < number
817 ? best
818 : number
819 : best > number
820 ? best
821 : number,
822 )
823 .toString();
824 }
825 }
826 output.add('$command $answer');
827 return;
828 }
829 if (matches.isEmpty) {
830 output.add('no match');
831 return;
832 }
833 for (final match in matches) {
834 final value = match.value;
835 if (command == 'GET') {
836 output.add('${location(match.path)} = ${compact(value)}');
837 } else if (command == 'COMPACT') {
838 output.add(compact(value));
839 } else if (command == 'PRETTY') {
840 output.add(pretty(value));
841 } else if (command == 'TYPE') {
842 output.add('${location(match.path)} ${typeName(value)}');
843 } else if (command == 'LENGTH') {
844 final length = value is Map<String, Value>
845 ? value.length
846 : value is List<Value>
847 ? value.length
848 : (value as String).length;
849 output.add('${location(match.path)} $length');
850 } else if (command == 'KEYS') {
851 output.add(
852 '${location(match.path)} ${compact((value as Map<String, Value>).keys.toList())}',
853 );
854 } else if (command == 'FLATTEN') {
855 for (final leaf in preorder(match)) {
856 if (children(leaf).isEmpty)
857 output.add('${location(leaf.path)} = ${compact(leaf.value)}');
858 }
859 }
860 }
861 }

863 void run(List<String> lines) {
864 var position = 0;
865 List<String> block() {
866 final body = <String>[];
867 while (lines[position] != '%%') body.add(lines[position++]);
868 position++;
869 return body;
870 }
871
872 load(block());
873 while (lines[position] != 'END') {
874 final line = lines[position++];
875 if (line == 'LOAD') {
876 load(block());
877 continue;
878 }
879 try {
880 execute(line);
881 } on JsonFailure catch (error) {
882 output.add('PARSE ERROR col ${error.position + 1}: ${error.message}');
883 } on QueryFailure catch (error) {
884 output.add('ERROR ${error.message}');
885 }
886 }
887 }
888}
