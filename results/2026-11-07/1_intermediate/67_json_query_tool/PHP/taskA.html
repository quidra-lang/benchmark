13function quoted(string $text, string $quote = '"'): string
14{
15 $output = $quote;
16 $escapes = [8 => '\\b', 12 => '\\f', 10 => '\\n', 13 => '\\r', 9 => '\\t'];
17 foreach (str_split($text) as $char) {
18 $code = ord($char);
19 if ($char === $quote || $char === '\\') $output .= '\\' . $char;
20 elseif (isset($escapes[$code])) $output .= $escapes[$code];
21 elseif ($code < 32 || $code === 127) $output .= sprintf('\\u%04x', $code);
22 else $output .= $char;
23 }
24 return $output . $quote;
25}
26function member(JsonValue $value, string $key): int
27{
28 if ($value->kind === 'object') foreach ($value->data as $index => $pair) if ($pair[0] === $key) return $index;
29 return -1;
30}
31function compact_json(JsonValue $value): string
32{
33 return match ($value->kind) {
34 'null' => 'null', 'number' => (string)$value->data, 'boolean' => $value->data ? 'true' : 'false', 'string' => quoted($value->data),
35 'array' => '[' . implode(',', array_map('compact_json', $value->data)) . ']',
36 'object' => '{' . implode(',', array_map(fn($pair) => quoted($pair[0]) . ':' . compact_json($pair[1]), $value->data)) . '}'
37 };
38}
39function pretty(JsonValue $value, string $indent = ''): string
40{
41 if (!in_array($value->kind, ['object', 'array'], true) || $value->data === []) return compact_json($value);
42 $next = $indent . ' ';
43 $items = $value->kind === 'object' ? array_map(fn($pair) => $next . quoted($pair[0]) . ': ' . pretty($pair[1], $next), $value->data)
44 : array_map(fn($item) => $next . pretty($item, $next), $value->data);
45 return ($value->kind === 'object' ? '{' : '[') . "\n" . implode(",\n", $items) . "\n" . $indent . ($value->kind === 'object' ? '}' : ']');
46}
47class StrictJson
48{
49 private int $position = 0;
50 public function __construct(private string $text) {
51 }
52 private function char(): string {
53 return $this->text[$this->position] ?? '';
54 }
55 private function fail(string $reason, ?int $position = null): never {
56 throw new ParseFault($reason, $position ?? $this->position);
57 }
58 private function whitespace(): void {
59 while (in_array($this->char(), [' ', "\n"], true)) $this->position++;
60 }
61 private function string(): string
62 {
63 $this->position++;
64 $output = '';
65 while (true) {
66 $char = $this->char();
67 if ($char === '' || $char === "\n") $this->fail('unterminated string');
68 if ($char === '"') {
69 $this->position++;
70 return $output;
71 }
72 if ($char === '\\') {
73 $start = $this->position;
74 $escaped = $this->text[$start + 1] ?? '';
75 if ($escaped === '' || $escaped === "\n") $this->fail('unterminated string', $start + 1);
76 if ($escaped === 'u') {
77 $hex = substr($this->text, $start + 2, 4);
78 if (!preg_match('/\A[0-9A-Fa-f]{4}\z/', $hex)) $this->fail('bad unicode escape', $start);
79 if (hexdec($hex) > 127) $this->fail('non-ascii escape', $start);
80 $output .= chr(hexdec($hex));
81 $this->position += 6;
82 }
83 else {
84 $escapes = ['"' => '"', '\\' => '\\', '/' => '/', 'b' => "\x08", 'f' => "\x0c", 'n' => "\n", 'r' => "\r", 't' => "\t"];
85 if (!isset($escapes[$escaped])) $this->fail("invalid escape '\\$escaped'", $start);
86 $output .= $escapes[$escaped];
87 $this->position += 2;
88 }
89 }
90 else {
91 $output .= $char;
92 $this->position++;
93 }
94 }
95 }
96 private function value(): JsonValue
97 {
98 $this->whitespace();
99 $char = $this->char();
100 if ($char === '') $this->fail('unexpected end of input');
101 if ($char === '{' || $char === '[') {
102 $object = $char === '{';
103 $ending = $object ? '}' : ']';
104 $result = new JsonValue($object ? 'object' : 'array', []);
105 $this->position++;
106 $this->whitespace();
107 if ($this->char() === $ending) {
108 $this->position++;
109 return $result;
110 }
111 while (true) {
112 if ($object) {
113 $this->whitespace();
114 if ($this->char() === '') $this->fail('unexpected end of input');
115 if ($this->char() !== '"') $this->fail('expected key');
116 $start = $this->position;
117 $key = $this->string();
118 if (member($result, $key) >= 0) $this->fail('duplicate key ' . quoted($key), $start);
119 $this->whitespace();
120 if ($this->char() === '') $this->fail('unexpected end of input');
121 if ($this->char() !== ':') $this->fail("expected ':'");
122 $this->position++;
123 $result->data[] = [$key, $this->value()];
124 }
125 else $result->data[] = $this->value();
126 $this->whitespace();
127 if ($this->char() === $ending) {
128 $this->position++;
129 return $result;
130 }
131 if ($this->char() === '') $this->fail('unexpected end of input');
132 if ($this->char() !== ',') $this->fail("expected ',' or '$ending'");
133 $this->position++;
134 $this->whitespace();
135 if ($this->char() === $ending) $this->fail('trailing comma');
136 }
137 }
138 if ($char === '"') return new JsonValue('string', $this->string());
139 if ($char === '-' || ctype_digit($char)) {
140 $start = $this->position;
141 if ($char === '-') $this->position++;
142 $digitsStart = $this->position;
143 while ($this->char() !== '' && ctype_digit($this->char())) $this->position++;
144 $digits = substr($this->text, $digitsStart, $this->position - $digitsStart);
145 if ($digits === '') $this->fail('invalid number', $start);
146 if (strlen($digits) > 1 && $digits[0] === '0') $this->fail('leading zero', $digitsStart);
147 if ($this->char() === '.') $this->fail('fraction not allowed');
148 if (in_array($this->char(), ['e', 'E'], true)) $this->fail('exponent not allowed');
149 if (strlen($digits) >= 10) $this->fail('number out of range', $start);
150 return new JsonValue('number', (int)substr($this->text, $start, $this->position - $start));
151 }
152 if (preg_match('/[A-Za-z]/', $char)) {
153 $start = $this->position;
154 while ($this->char() !== '' && preg_match('/[A-Za-z]/', $this->char())) $this->position++;
155 $word = substr($this->text, $start, $this->position - $start);
156 if (!in_array($word, ['true', 'false', 'null'], true)) $this->fail("invalid literal '$word'", $start);
157 return $word === 'null' ? new JsonValue() : new JsonValue('boolean', $word === 'true');
158 }
159 $this->fail("unexpected character '$char'");
160 }
161 public function read(): JsonValue
162 {
163 $result = $this->value();
164 $this->whitespace();
165 if ($this->position < strlen($this->text)) $this->fail('trailing characters');
166 return $result;
167 }

171 private JsonValue $document;
172 private array $history = [];
173 private const INTEGER = '/\A(?:0|-?[1-9][0-9]{0,8})\z/';
174 private const IDENTIFIER = '/\A[A-Za-z_][A-Za-z0-9_]*\z/';

179 return unserialize(serialize($value));

184 private function quotedKey(string $text, int &$position): string
185 {
186 if (($text[$position] ?? '') !== "'") $this->fail('bad');
187 $position++;
188 $output = '';
189 while ($position < strlen($text)) {
190 $char = $text[$position];
191 if ($char === "'") {
192 $position++;
193 return $output;
194 }
195 if ($char === '\\') {
196 $escaped = $text[$position + 1] ?? '';
197 if ($escaped === 'u') {
198 $hex = substr($text, $position + 2, 4);
199 if (!preg_match('/\A[0-9A-Fa-f]{4}\z/', $hex) || hexdec($hex) > 127) $this->fail('bad');
200 $output .= chr(hexdec($hex));
201 $position += 6;
202 }
203 else {
204 $escapes = ["'" => "'", '\\' => '\\', '/' => '/', 'b' => "\x08", 'f' => "\x0c", 'n' => "\n", 'r' => "\r", 't' => "\t"];
205 if (!isset($escapes[$escaped])) $this->fail('bad');
206 $output .= $escapes[$escaped];
207 $position += 2;
208 }
209 }
210 else {
211 $output .= $char;
212 $position++;
213 }
214 }
215 $this->fail('bad');
216 }
217 private function path(string $text, bool $field = false): array
218 {
219 try {
220 if (($text[0] ?? '') !== ($field ? '@' : '$')) $this->fail('bad');
221 $position = 1;
222 $selectors = [];
223 while ($position < strlen($text)) {
224 $descendant = false;
225 if (!$field && substr($text, $position, 2) === '..') {
226 $descendant = true;
227 $position += 2;
228 }
229 elseif ($text[$position] === '.') {
230 $position++;
231 if (!preg_match('/\A(?:[A-Za-z_][A-Za-z0-9_]*|\*)/', substr($text, $position), $match) || ($field && $match[0] === '*')) $this->fail('bad');
232 $selectors[] = [false, $match[0] === '*' ? 'all' : 'key', $match[0]];
233 $position += strlen($match[0]);
234 continue;
235 }
236 elseif ($text[$position] !== '[') $this->fail('bad');
237 if (($text[$position] ?? '') === '[') {
238 $position++;
239 if (($text[$position] ?? '') === "'") {
240 $key = $this->quotedKey($text, $position);
241 $kind = 'key';
242 }
243 else {
244 $finish = strpos($text, ']', $position);
245 if ($finish === false) $this->fail('bad');
246 $word = substr($text, $position, $finish - $position);
247 $position = $finish;
248 if (preg_match(self::INTEGER, $word)) {
249 $kind = 'index';
250 $key = (int)$word;
251 }
252 elseif (!$field && $word === '*') {
253 $kind = 'all';
254 $key = null;
255 }
256 elseif (!$field && substr_count($word, ':') === 1) {
257 $bounds = explode(':', $word);
258 foreach ($bounds as $bound) if ($bound !== '' && !preg_match(self::INTEGER, $bound)) $this->fail('bad');
259 $kind = 'slice';
260 $key = array_map(fn($bound) => $bound === '' ? null : (int)$bound, $bounds);
261 }
262 else $this->fail('bad');
263 }
264 if (($text[$position] ?? '') !== ']') $this->fail('bad');
265 $position++;
266 }
267 elseif ($descendant) {
268 if (!preg_match('/\A(?:[A-Za-z_][A-Za-z0-9_]*|\*)/', substr($text, $position), $match)) $this->fail('bad');
269 $key = $match[0];
270 $kind = $key === '*' ? 'all' : 'key';
271 $position += strlen($key);
272 }
273 else $this->fail('bad');
274 $selectors[] = [$descendant, $kind, $key];
275 }
276 return $selectors;
277 }
278 catch (RuntimeException $error) {
279 $this->fail($field ? 'ERROR bad field' : 'ERROR bad path');
280 }
281 }
282 private function location(array $steps): string
283 {
284 $output = '$';
285 foreach ($steps as $step) $output .= is_int($step) ? "[$step]" : (preg_match(self::IDENTIFIER, $step) ? '.' . $step : '[' . quoted($step, "'") . ']');
286 return $output;
287 }
288 private function children(array $match): array
289 {
290 [$value, $steps, $positions] = $match;
291 $output = [];
292 if ($value->kind === 'object') foreach ($value->data as $i => [$key, $child]) $output[] = [$child, [...$steps, $key], [...$positions, $i]];
293 elseif ($value->kind === 'array') foreach ($value->data as $i => $child) $output[] = [$child, [...$steps, $i], [...$positions, $i]];
294 return $output;
295 }
296 private function preorder(array $match): array
297 {
298 $output = [$match];
299 foreach ($this->children($match) as $child) array_push($output, ...$this->preorder($child));
300 return $output;
301 }
302 private function select(array $match, array $selector): array
303 {
304 [, $kind, $key] = $selector;
305 $value = $match[0];
306 $all = $this->children($match);
307 if ($kind === 'all') return $all;
308 if ($kind === 'key') return $value->kind === 'object' ? array_values(array_filter($all, fn($item) => $item[1][array_key_last($item[1])] === $key)) : [];
309 if ($value->kind !== 'array') return [];
310 $n = count($value->data);
311 if ($kind === 'index') {
312 $index = $key < 0 ? $n + $key : $key;
313 return $index >= 0 && $index < $n ? [$all[$index]] : [];
314 }
315 $bounds = [$key[0] ?? 0, $key[1] ?? $n];
316 $bounds = array_map(fn($bound) => min($n, max(0, $bound < 0 ? $n + $bound : $bound)), $bounds);
317 return $bounds[0] < $bounds[1] ? array_slice($all, $bounds[0], $bounds[1] - $bounds[0]) : [];
318 }
319 private function evaluate(array $selectors, ?array $initial = null): array
320 {
321 $matches = $initial ?? [[$this->document, [], []]];
322 foreach ($selectors as $selector) {
323 $next = [];
324 foreach ($matches as $match) foreach ($selector[0] ? $this->preorder($match) : [$match] as $item) array_push($next, ...$this->select($item, $selector));
325 $matches = $next;
326 }
327 return $matches;
328 }
329 private function field(JsonValue $value, array $selectors): ?JsonValue
330 {
331 $result = $this->evaluate($selectors, [[$value, [], []]]);
332 return $result === [] ? null : $result[0][0];
333 }
334 private function at(array $steps): JsonValue
335 {
336 $value = $this->document;
337 foreach ($steps as $step) $value = is_int($step) ? $value->data[$step] : $value->data[member($value, $step)][1];
338 return $value;
339 }
340 private function assign(array $steps, JsonValue $value): void
341 {
342 if ($steps === []) {
343 $this->document = $value;
344 return;
345 }
346 $last = array_pop($steps);
347 $parent = $this->at($steps);
348 if (is_int($last)) $parent->data[$last] = $value;
349 else {
350 $index = member($parent, $last);
351 if ($index < 0) $parent->data[] = [$last, $value];
352 else $parent->data[$index][1] = $value;
353 }
354 }
355 private function merge(JsonValue $target, JsonValue $source): void
356 {
357 foreach ($source->data as [$key, $child]) {
358 $index = member($target, $key);
359 if ($index >= 0 && $target->data[$index][1]->kind === 'object' && $child->kind === 'object') $this->merge($target->data[$index][1], $child);
360 elseif ($index >= 0) $target->data[$index][1] = $this->copy($child);
361 else $target->data[] = [$key, $this->copy($child)];
362 }
363 }
364 private function equal(?JsonValue $left, JsonValue $right): bool
365 {
366 if ($left === null || $left->kind !== $right->kind) return false;
367 if ($left->kind === 'object') {
368 if (count($left->data) !== count($right->data)) return false;
369 foreach ($left->data as [$key, $item]) {
370 $index = member($right, $key);
371 if ($index < 0 || !$this->equal($item, $right->data[$index][1])) return false;
372 }
373 return true;
374 }
375 if ($left->kind === 'array') {
376 if (count($left->data) !== count($right->data)) return false;
377 foreach ($left->data as $index => $item) if (!$this->equal($item, $right->data[$index])) return false;
378 return true;
379 }
380 return $left->data === $right->data;
381 }
382 private function checkKinds(array $matches, array $kinds, string $reason): void
383 {
384 foreach ($matches as $match) if (!in_array($match[0]->kind, $kinds, true)) $this->fail('ERROR ' . $reason . ' at ' . $this->location($match[1]));
385 }

388 try {

395 private function comparePositions(array $left, array $right): int
396 {
397 for ($i = 0; $i < min(count($left), count($right)); $i++) if ($left[$i] !== $right[$i]) return $left[$i] <=> $right[$i];
398 return count($left) <=> count($right);
399 }
400 public function command(string $line): void
401 {
402 [$command, $argument] = array_pad(explode(' ', $line, 2), 2, null);
403 try {
404 $names = ['GET', 'COMPACT', 'PRETTY', 'TYPE', 'LENGTH', 'KEYS', 'FLATTEN', 'COUNT', 'SUM', 'MIN', 'MAX', 'AVG', 'FILTER', 'SET', 'DELETE', 'INSERT', 'MERGE', 'SORT', 'RENAME', 'UNDO'];
405 if (!in_array($command, $names, true)) $this->fail("ERROR unknown command '$command'");
406 $limit = ['SET' => 2, 'MERGE' => 2, 'INSERT' => 3, 'FILTER' => 4][$command] ?? PHP_INT_MAX;
407 $args = $argument === null ? [] : explode(' ', $argument, $limit);
408 $count = count($args);
409 $shape = match ($command) {
410 'SET', 'MERGE' => $count === 2, 'INSERT' => $count === 3,
411 'FILTER' => $count >= 3 && ($args[2] === 'EXISTS' ? $count === 3 : $count === 4),
412 'SORT' => $count >= 1 && $count <= 3 && ($count < 3 || ($args[1] !== 'DESC' && $args[2] === 'DESC')),
413 'COMPACT', 'PRETTY', 'FLATTEN' => $count <= 1, 'UNDO' => $count === 0, 'RENAME' => $count === 2, default => $count === 1
414 };
415 if (!$shape) $this->fail('ERROR bad arguments');
416 if ($command === 'UNDO') {
417 if ($this->history === []) $this->fail('ERROR nothing to undo');
418 $this->document = array_pop($this->history);
419 echo 'UNDO ' . count($this->history) . "\n";
420 return;
421 }
422 $selectors = $this->path($args[0] ?? '$');
423 $field = [];
424 if ($command === 'FILTER') {
425 $field = $this->path($args[1], true);
426 if (!in_array($args[2], ['==', '!=', '<', '<=', '>', '>=', 'EXISTS'], true)) $this->fail('ERROR bad operator');
427 }
428 elseif ($command === 'SORT') $field = $this->path($count >= 2 && $args[1] !== 'DESC' ? $args[1] : '@', true);
429 elseif ($command === 'INSERT' && !preg_match(self::INTEGER, $args[1])) $this->fail('ERROR bad index');
430 elseif ($command === 'RENAME') {
431 try {
432 if (preg_match(self::IDENTIFIER, $args[1])) $newKey = $args[1];
433 else {
434 $position = 0;
435 $newKey = $this->quotedKey($args[1], $position);
436 if ($position !== strlen($args[1])) $this->fail('bad');
437 }
438 }
439 catch (RuntimeException $error) {
440 $this->fail('ERROR bad key');
441 }
442 }
443 elseif ($command === 'DELETE' && $selectors === []) $this->fail('ERROR cannot delete root');
444 $replacement = new JsonValue();
445 if (in_array($command, ['SET', 'MERGE', 'INSERT'], true) || ($command === 'FILTER' && $args[2] !== 'EXISTS')) $replacement = $this->parse($args[$count - 1]);
446 if ($command === 'MERGE' && $replacement->kind !== 'object') $this->fail('ERROR merge value is not an object');
447 $matches = $this->evaluate($selectors);
448 if ($command === 'LENGTH') $this->checkKinds($matches, ['object', 'array', 'string'], 'no length');
449 if (in_array($command, ['KEYS', 'MERGE'], true)) $this->checkKinds($matches, ['object'], 'not an object');
450 if (in_array($command, ['FILTER', 'INSERT', 'SORT'], true)) $this->checkKinds($matches, ['array'], 'not an array');
451 if ($command === 'RENAME') foreach ($matches as $match) if ($match[1] === [] || !is_string($match[1][array_key_last($match[1])])) $this->fail('ERROR not a member at ' . $this->location($match[1]));
452 if ($command === 'INSERT') {
453 $index = (int)$args[1];
454 foreach ($matches as $match) if ($index < -count($match[0]->data) - 1 || $index > count($match[0]->data)) $this->fail('ERROR index out of range at ' . $this->location($match[1]));
455 }
456 if (in_array($command, ['SET', 'DELETE', 'INSERT', 'MERGE', 'SORT', 'RENAME'], true)) {
457 if ($command === 'SET' && $selectors !== [] && !$selectors[array_key_last($selectors)][0] && $selectors[array_key_last($selectors)][1] === 'key') {
458 $key = $selectors[array_key_last($selectors)][2];
459 $matches = [];
460 foreach ($this->evaluate(array_slice($selectors, 0, -1)) as $parent) {
461 if ($parent[0]->kind !== 'object') continue;
462 $position = member($parent[0], $key);
463 $matches[] = [null, [...$parent[1], $key], [...$parent[2], $position < 0 ? count($parent[0]->data) : $position]];
464 }
465 }
466 $targets = [];
467 $seen = [];
468 foreach ($matches as $match) {
469 $key = serialize($match[1]);
470 if (!isset($seen[$key])) $targets[] = $match;
471 $seen[$key] = true;
472 }
473 usort($targets, fn($a, $b) => $this->comparePositions($b[2], $a[2]));
474 $original = $this->copy($this->document);
475 try {
476 foreach ($targets as [, $steps]) {
477 $value = $command === 'SET' ? null : $this->at($steps);
478 switch ($command) {
479 case 'SET':
480 $this->assign($steps, $this->copy($replacement));
481 break;
482 case 'DELETE':
483 $last = array_pop($steps);
484 $parent = $this->at($steps);
485 array_splice($parent->data, is_int($last) ? $last : member($parent, $last), 1);
486 break;
487 case 'INSERT':
488 array_splice($value->data, $index < 0 ? count($value->data) + 1 + $index : $index, 0, [$this->copy($replacement)]);
489 break;
490 case 'MERGE':
491 $this->merge($value, $replacement);
492 break;
493 case 'SORT':
494 $decorated = [];
495 foreach ($value->data as $i => $item) $decorated[] = [$this->field($item, $field), $i, $item];
496 $descending = $args[$count - 1] === 'DESC';
497 usort($decorated, function ($a, $b) use ($descending) {
498 $rank = fn($v) => $v === null ? 3 : match ($v->kind) {
499 'null' => 0, 'number' => 1, 'string' => 2, default => 3
500 };
501 $ra = $rank($a[0]);
502 $rb = $rank($b[0]);
503 $comparison = $ra <=> $rb;
504 if ($comparison === 0 && $ra === 1) $comparison = $a[0]->data <=> $b[0]->data;
505 if ($comparison === 0 && $ra === 2) $comparison = strcmp($a[0]->data, $b[0]->data);
506 if ($descending) $comparison = -$comparison;
507 return $comparison === 0 ? $a[1] <=> $b[1] : $comparison;
508 });
509 $value->data = array_map(fn($item) => $item[2], $decorated);
510 break;
511 case 'RENAME':
512 $last = array_pop($steps);
513 $parent = $this->at($steps);
514 if ($last !== $newKey && member($parent, $newKey) >= 0) $this->fail('ERROR key exists at ' . $this->location([...$steps, $last]));
515 $parent->data[member($parent, $last)][0] = $newKey;
516 break;
517 }
518 }
519 }
520 catch (RuntimeException $error) {
521 $this->document = $original;
522 throw $error;
523 }
524 $this->history[] = $original;
525 echo "$command " . count($targets) . "\n";
526 return;
527 }
528 if (in_array($command, ['COUNT', 'SUM', 'MIN', 'MAX', 'AVG'], true)) {
529 $numbers = array_map(fn($match) => $match[0]->data, array_filter($matches, fn($match) => $match[0]->kind === 'number'));
530 if ($command === 'COUNT') $result = (string)count($matches);
531 elseif ($command === 'SUM') $result = (string)array_sum($numbers);
532 elseif ($numbers === []) $result = 'null';
533 elseif ($command === 'MIN') $result = (string)min($numbers);
534 elseif ($command === 'MAX') $result = (string)max($numbers);
535 else {
536 $sum = array_sum($numbers);
537 $rounded = intdiv(abs($sum) * 200 + count($numbers), 2 * count($numbers));
538 $result = sprintf('%s%d.%02d', $sum < 0 && $rounded > 0 ? '-' : '', intdiv($rounded, 100), $rounded % 100);
539 }
540 echo "$command $result\n";
541 return;
542 }
543 $output = [];
544 foreach ($matches as $match) {
545 [$value, $steps] = $match;
546 $where = $this->location($steps);
547 switch ($command) {
548 case 'GET':
549 $output[] = "$where = " . compact_json($value);
550 break;
551 case 'COMPACT':
552 $output[] = compact_json($value);
553 break;
554 case 'PRETTY':
555 $output[] = pretty($value);
556 break;
557 case 'TYPE':
558 $output[] = "$where $value->kind";
559 break;
560 case 'LENGTH':
561 $output[] = "$where " . ($value->kind === 'string' ? strlen($value->data) : count($value->data));
562 break;
563 case 'KEYS':
564 $output[] = "$where [" . implode(',', array_map(fn($pair) => quoted($pair[0]), $value->data)) . ']';
565 break;
566 case 'FLATTEN':
567 foreach ($this->preorder($match) as $item) if ($this->children($item) === []) $output[] = $this->location($item[1]) . ' = ' . compact_json($item[0]);
568 break;
569 case 'FILTER':
570 foreach ($this->children($match) as $item) {
571 $f = $this->field($item[0], $field);
572 $operator = $args[2];
573 $equal = $this->equal($f, $replacement);
574 $pass = false;
575 if ($operator === 'EXISTS') $pass = $f !== null;
576 elseif ($operator === '==') $pass = $equal;
577 elseif ($operator === '!=') $pass = !$equal;
578 elseif ($f !== null && $f->kind === $replacement->kind && in_array($f->kind, ['number', 'string'], true)) {
579 $comparison = $f->kind === 'string' ? strcmp($f->data, $replacement->data) : ($f->data <=> $replacement->data);
580 $pass = match ($operator) {
581 '<' => $comparison < 0, '<=' => $comparison <= 0, '>' => $comparison > 0, '>=' => $comparison >= 0
582 };
583 }
584 if ($pass) $output[] = $this->location($item[1]) . ' = ' . compact_json($item[0]);
585 }
586 break;
587 }
588 }
589 echo $output === [] ? "no match\n" : implode("\n", $output) . "\n";
590 }
591 catch (RuntimeException $error) {
592 echo $error->getMessage() . "\n";
593 }
594 }
595 private function load(array $lines): void
596 {
597 $text = implode("\n", $lines);
598 try {
599 $value = (new StrictJson($text))->read();
600 $this->history[] = $this->copy($this->document);
601 $this->document = $value;
602 $stats = function ($item) use (&$stats) {
603 $children = $item->kind === 'object' ? array_column($item->data, 1) : ($item->kind === 'array' ? $item->data : []);
604 $count = 1;
605 $depth = 1;
606 foreach ($children as $child) {
607 [$n, $d] = $stats($child);
608 $count += $n;
609 $depth = max($depth, $d + 1);
610 }
611 return [$count, $depth];
612 };
613 [$count, $depth] = $stats($value);
614 echo "LOADED nodes=$count depth=$depth\n";
615 }
616 catch (ParseFault $error) {
617 $prefix = substr($text, 0, $error->position);
618 $line = substr_count($prefix, "\n") + 1;
619 $last = strrpos($prefix, "\n");
620 $column = $error->position - ($last === false ? -1 : $last);
621 echo "PARSE ERROR line $line col $column: {$error->getMessage()}\n";
622 }
623 }
624 public function run(): void
625 {
626 $input = [];
627 while (($line = fgets(STDIN)) !== false) $input[] = rtrim($line, "\r\n");
628 $position = 0;
629 while (true) {
630 $block = [];
631 while ($input[$position] !== '%%') $block[] = $input[$position++];
632 $position++;
633 $this->load($block);
634 while ($input[$position] !== 'LOAD' && $input[$position] !== 'END') $this->command($input[$position++]);
635 if ($input[$position] === 'END') break;
636 $position++;
637 }
638 }
