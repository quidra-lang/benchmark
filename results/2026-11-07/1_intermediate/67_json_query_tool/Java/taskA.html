44 static String quote(String s, char delimiter) {
45 StringBuilder out = new StringBuilder().append(delimiter);
46 for (char c : s.toCharArray()) {
47 if (c == delimiter || c == '\\') out.append('\\').append(c);
48 else out.append(switch (c) {
49 case '\b' -> "\\b"; case '\f' -> "\\f"; case '\n' -> "\\n"; case '\r' -> "\\r"; case '\t' -> "\\t"; default -> c < 32 || c == 127 ? String.format(Locale.ROOT, "\\u%04x", (int)c) : "" + c;
50 });
51 }
52 return out.append(delimiter).toString();
53 }

55 static class Parser {
56 final String text;
57 int at;
58 Parser(String text) {
59 this.text = text;
60 }
61 char peek() {
62 return at == text.length() ? '\0' : text.charAt(at);
63 }
64 void fail(String reason) {
65 throw new ParseError(at, reason);
66 }
67 void whitespace() {
68 while (peek() == ' ' || peek() == '\n') at++;
69 }
70 void needCharacter() {
71 if (at == text.length()) fail("unexpected end of input");
72 }
73 String string(char delimiter) {
74 at++;
75 StringBuilder out = new StringBuilder();
76 while (true) {
77 if (at == text.length() || peek() == '\n') fail("unterminated string");
78 char c = peek();
79 if (c == delimiter) {
80 at++;
81 return out.toString();
82 }
83 if (c != '\\') {
84 out.append(c);
85 at++;
86 continue;
87 }
88 int start = at;
89 at++;
90 if (at == text.length() || peek() == '\n') fail("unterminated string");
91 char escaped = peek();
92 if (escaped == 'u') {
93 if (at + 4 >= text.length() || !text.substring(at + 1, at + 5).matches("[0-9A-Fa-f]{4}")) throw new ParseError(start, "bad unicode escape");
94 int value = Integer.parseInt(text.substring(at + 1, at + 5), 16);
95 if (value > 127) throw new ParseError(start, "non-ascii escape");
96 out.append((char)value);
97 at += 5;
98 } else {
99 char value = switch (escaped) {
100 case '\\', '/' -> escaped;
101 case 'b' -> '\b';
102 case 'f' -> '\f';
103 case 'n' -> '\n';
104 case 'r' -> '\r';
105 case 't' -> '\t';
106 default -> escaped == delimiter ? delimiter : '\0';
107 };
108 if (value == '\0') throw new ParseError(start, "invalid escape '\\" + escaped + "'");
109 out.append(value);
110 at++;
111 }
112 }
113 }
114 Object value() {
115 whitespace();
116 needCharacter();
117 char c = peek();
118 if (c == '"') return string('"');
119 if (c == '{') {
120 at++;
121 whitespace();
122 LinkedHashMap<String, Object> map = new LinkedHashMap<>();
123 if (peek() == '}') {
124 at++;
125 return map;
126 }
127 while (true) {
128 whitespace();
129 needCharacter();
130 if (peek() != '"') fail("expected key");
131 int start = at;
132 String key = string('"');
133 if (map.containsKey(key)) throw new ParseError(start, "duplicate key " + quote(key, '"'));
134 whitespace();
135 needCharacter();
136 if (peek() != ':') fail("expected ':'");
137 at++;
138 map.put(key, value());
139 whitespace();
140 if (peek() == '}') {
141 at++;
142 return map;
143 }
144 needCharacter();
145 if (peek() != ',') fail("expected ',' or '}'");
146 at++;
147 whitespace();
148 if (peek() == '}') fail("trailing comma");
149 }
150 }
151 if (c == '[') {
152 at++;
153 whitespace();
154 List<Object> list = new ArrayList<>();
155 if (peek() == ']') {
156 at++;
157 return list;
158 }
159 while (true) {
160 list.add(value());
161 whitespace();
162 if (peek() == ']') {
163 at++;
164 return list;
165 }
166 needCharacter();
167 if (peek() != ',') fail("expected ',' or ']'");
168 at++;
169 whitespace();
170 if (peek() == ']') fail("trailing comma");
171 }
172 }
173 if (c == '-' || digit(c)) {
174 int start = at;
175 if (c == '-') at++;
176 int digits = at;
177 while (digit(peek())) at++;
178 if (at == digits) throw new ParseError(start, "invalid number");
179 if (at - digits >= 2 && text.charAt(digits) == '0') throw new ParseError(digits, "leading zero");
180 if (peek() == '.') fail("fraction not allowed");
181 if (peek() == 'e' || peek() == 'E') fail("exponent not allowed");
182 if (at - digits >= 10) throw new ParseError(start, "number out of range");
183 return Integer.parseInt(text.substring(start, at));
184 }
185 if (letter(c)) {
186 int start = at;
187 while (letter(peek())) at++;
188 String word = text.substring(start, at);
189 return switch (word) {
190 case "true" -> true;
191 case "false" -> false;
192 case "null" -> null;
193 default -> throw new ParseError(start, "invalid literal '" + word + "'");
194 };
195 }
196 fail("unexpected character '" + c + "'");
197 return null;
198 }
199 Object whole() {
200 Object result = value();
201 whitespace();
202 if (at != text.length()) fail("trailing characters");
203 return result;
204 }
205 }

221 static String compact(Object v) {
222 if (v == null) return "null";
223 if (v instanceof String s) return quote(s, '"');
224 if (v instanceof Map) {
225 List<String> parts = new ArrayList<>();
226 object(v).forEach((key, value) -> parts.add(quote(key, '"') + ":" + compact(value)));
227 return "{" + String.join(",", parts) + "}";
228 }
229 if (v instanceof List) {
230 List<String> parts = new ArrayList<>();
231 for (Object item : array(v)) parts.add(compact(item));
232 return "[" + String.join(",", parts) + "]";
233 }
234 return v.toString();

237 static String pretty(Object v, String indent) {
238 if (v instanceof Map && !object(v).isEmpty() || v instanceof List && !array(v).isEmpty()) {
239 String child = indent + " ";
240 List<String> parts = new ArrayList<>();
241 if (v instanceof Map) object(v).forEach((key, value) -> parts.add(child + quote(key, '"') + ": " + pretty(value, child)));
242 else for (Object item : array(v)) parts.add(child + pretty(item, child));
243 return (v instanceof Map ? "{" : "[") + "\n" + String.join(",\n", parts) + "\n" + indent + (v instanceof Map ? "}" : "]");
244 }
245 return compact(v);

256 String identifier() {
257 int start = at;
258 if (!letter(peek()) && peek() != '_') throw new Problem("");
259 at++;
260 while (letter(peek()) || digit(peek()) || peek() == '_') at++;
261 return text.substring(start, at);
262 }
263 List<Selector> read(boolean field) {
264 if (peek() != (field ? '@' : '$')) throw new Problem("");
265 at++;
266 List<Selector> selectors = new ArrayList<>();
267 while (at < text.length()) {
268 boolean descendant = false;
269 if (peek() == '.') {
270 at++;
271 if (peek() == '.') {
272 if (field) throw new Problem("");
273 descendant = true;
274 at++;
275 }
276 if (peek() == '*') {
277 if (field) throw new Problem("");
278 at++;
279 selectors.add(new Selector('A', null, null, null, descendant));
280 continue;
281 }
282 if (!(descendant && peek() == '[')) {
283 selectors.add(new Selector('K', identifier(), null, null, descendant));
284 continue;
285 }
286 } else if (peek() != '[') throw new Problem("");
287 if (peek() != '[') throw new Problem("");
288 at++;
289 if (peek() == '\'') {
290 String key = string('\'');
291 if (peek() != ']') throw new Problem("");
292 at++;
293 selectors.add(new Selector('K', key, null, null, descendant));
294 continue;
295 }
296 int end = text.indexOf(']', at);
297 if (end < 0) throw new Problem("");
298 String content = text.substring(at, end);
299 at = end + 1;
300 if (content.equals("*") && !field) selectors.add(new Selector('A', null, null, null, descendant));
301 else if (integerNotation(content)) selectors.add(new Selector('I', null, Integer.valueOf(content), null, descendant));
302 else if (!field && content.contains(":")) {
303 String[] parts = content.split(":", -1);
304 if (parts.length != 2 || !parts[0].isEmpty() && !integerNotation(parts[0]) || !parts[1].isEmpty() && !integerNotation(parts[1])) throw new Problem("");
305 selectors.add(new Selector('S', null, parts[0].isEmpty() ? null : Integer.valueOf(parts[0]), parts[1].isEmpty() ? null : Integer.valueOf(parts[1]), descendant));
306 } else throw new Problem("");
307 }
308 return selectors;
309 }
310 }

336 List<Step> location = new ArrayList<>(parent.location());

342 static List<Match> children(Match parent) {
343 List<Match> out = new ArrayList<>();
344 if (parent.value() instanceof Map) {
345 int i = 0;
346 for (String key : object(parent.value()).keySet()) out.add(child(parent, key, i++));
347 } else if (parent.value() instanceof List) for (int i = 0; i < array(parent.value()).size(); i++) out.add(child(parent, i, i));
348 return out;

351 static void preorder(Match match, List<Match> out) {
352 out.add(match);
353 for (Match child : children(match)) preorder(child, out);

356 static List<Match> select(Match match, Selector selector) {
357 List<Match> out = new ArrayList<>();
358 Object value = match.value();
359 if (selector.kind() == 'K' && value instanceof Map && object(value).containsKey(selector.key())) {
360 int pos = new ArrayList<>(object(value).keySet()).indexOf(selector.key());
361 out.add(child(match, selector.key(), pos));
362 } else if (selector.kind() == 'A') out.addAll(children(match));
363 else if (value instanceof List && (selector.kind() == 'I' || selector.kind() == 'S')) {
364 int n = array(value).size();
365 if (selector.kind() == 'I') {
366 int i = selector.first() < 0 ? n + selector.first() : selector.first();
367 if (i >= 0 && i < n) out.add(child(match, i, i));
368 } else {
369 int a = selector.first() == null ? 0 : selector.first(), b = selector.last() == null ? n : selector.last();
370 if (a < 0) a += n;
371 if (b < 0) b += n;
372 a = Math.clamp(a, 0, n);
373 b = Math.clamp(b, 0, n);
374 for (int i = a; i < b; i++) out.add(child(match, i, i));
375 }
376 }
377 return out;
378 }

380 static List<Match> apply(List<Match> matches, Selector selector) {
381 List<Match> out = new ArrayList<>();
382 for (Match match : matches) {
383 List<Match> bases = new ArrayList<>();
384 if (selector.descendant()) preorder(match, bases);
385 else bases.add(match);
386 for (Match base : bases) out.addAll(select(base, selector));
387 }
388 return out;
389 }

474 static List<String> execute(String line) {
475 int space = line.indexOf(' ');
476 String command = space < 0 ? line : line.substring(0, space), argument = space < 0 ? null : line.substring(space + 1);
477 require(COMMANDS.contains(command), "ERROR unknown command '" + command + "'");
478 String[] args = argument == null ? new String[0] : argument.split(" ", command.equals("SET") || command.equals("MERGE") ? 2 : command.equals("INSERT") ? 3 : command.equals("FILTER") ? 4 : -1);
479 boolean shape = switch (command) {
480 case "SET", "MERGE", "RENAME" -> args.length == 2;
481 case "INSERT" -> args.length == 3;
482 case "FILTER" -> args.length >= 3 && (args[2].equals("EXISTS") ? args.length == 3 : args.length == 4);
483 case "COMPACT", "PRETTY", "FLATTEN" -> args.length <= 1;
484 case "UNDO" -> args.length == 0;
485 case "SORT" -> args.length >= 1 && args.length <= 3 && (args.length < 3 || !args[1].equals("DESC") && args[2].equals("DESC"));
486 default -> args.length == 1;
487 };
488 require(shape, "ERROR bad arguments");
489 if (command.equals("UNDO")) {
490 require(!history.isEmpty(), "ERROR nothing to undo");
491 document = history.removeLast();
492 return List.of("UNDO " + history.size());
493 }
494 List<Selector> selectors = path(args.length == 0 ? "$" : args[0], false), fields = List.of();
495 Object value = null;
496 int index = 0;
497 String rename = null, operator = null;
498 if (command.equals("FILTER")) {
499 fields = path(args[1], true);
500 operator = args[2];
501 require(Set.of("==", "!=", "<", "<=", ">", ">=", "EXISTS").contains(operator), "ERROR bad operator");
502 if (!operator.equals("EXISTS")) value = jsonValue(args[3]);
503 } else if (command.equals("SORT")) fields = path(args.length >= 2 && !args[1].equals("DESC") ? args[1] : "@", true);
504 else if (command.equals("RENAME")) rename = key(args[1]);
505 else if (command.equals("INSERT")) {
506 require(integerNotation(args[1]), "ERROR bad index");
507 index = Integer.parseInt(args[1]);
508 value = jsonValue(args[2]);
509 } else if (command.equals("SET") || command.equals("MERGE")) value = jsonValue(args[1]);
510 if (command.equals("MERGE")) require(value instanceof Map, "ERROR merge value is not an object");
511 if (command.equals("DELETE")) require(!selectors.isEmpty(), "ERROR cannot delete root");
512 List<Match> matches = evaluate(selectors);
513 if (Set.of("FILTER", "INSERT", "SORT").contains(command)) requireType(matches, "array");
514 if (Set.of("KEYS", "MERGE").contains(command)) requireType(matches, "object");
515 if (command.equals("LENGTH")) for (Match match : matches) require(match.value() instanceof Map || match.value() instanceof List || match.value() instanceof String, "ERROR no length at " + location(match.location()));
516 if (command.equals("INSERT")) for (Match match : matches) require(index >= -array(match.value()).size() - 1 && index <= array(match.value()).size(), "ERROR index out of range at " + location(match.location()));
517 if (command.equals("RENAME")) for (Match match : matches) require(!match.location().isEmpty() && match.location().getLast().key() instanceof String, "ERROR not a member at " + location(match.location()));
518 List<String> out = new ArrayList<>();
519 if (Set.of("SET", "DELETE", "INSERT", "MERGE", "SORT", "RENAME").contains(command)) {
520 if (command.equals("SET") && !selectors.isEmpty()) {
521 Selector last = selectors.getLast();
522 if (last.kind() == 'K' && !last.descendant()) {
523 matches = new ArrayList<>();
524 for (Match parent : evaluate(selectors.subList(0, selectors.size() - 1))) if (parent.value() instanceof Map) {
525 List<Step> place = new ArrayList<>(parent.location());
526 int pos = new ArrayList<>(object(parent.value()).keySet()).indexOf(last.key());
527 place.add(new Step(last.key(), pos < 0 ? object(parent.value()).size() : pos));
528 matches.add(new Match(null, place));
529 }
530 }
531 }
532 List<Match> targets = targets(matches);
533 Object changed = copy(document);
534 for (Match target : targets) {
535 List<Step> place = target.location();
536 Object current = command.equals("SET") ? null : at(changed, place);
537 if (command.equals("SET")) {
538 if (place.isEmpty()) changed = copy(value);
539 else {
540 Object parent = at(changed, place.subList(0, place.size() - 1));
541 Object part = place.getLast().key();
542 if (part instanceof String s) object(parent).put(s, copy(value));
543 else array(parent).set((int)part, copy(value));
544 }
545 } else if (command.equals("DELETE")) {
546 Object parent = at(changed, place.subList(0, place.size() - 1));
547 Object part = place.getLast().key();
548 if (part instanceof String) object(parent).remove(part);
549 else array(parent).remove((int)part);
550 } else if (command.equals("INSERT")) array(current).add(index < 0 ? array(current).size() + 1 + index : index, copy(value));
551 else if (command.equals("MERGE")) merge(current, value);
552 else if (command.equals("SORT")) {
553 List<Selector> sortFields = fields;
554 boolean descending = args[args.length - 1].equals("DESC");
555 array(current).sort((a, b) -> (descending ? -1 : 1) * compare(field(a, sortFields), field(b, sortFields)));
556 } else {
557 Object parent = at(changed, place.subList(0, place.size() - 1));
558 String old = (String)place.getLast().key();
559 require(old.equals(rename) || !object(parent).containsKey(rename), "ERROR key exists at " + location(place));
560 LinkedHashMap<String, Object> renamed = new LinkedHashMap<>();
561 for (var entry : object(parent).entrySet()) renamed.put(entry.getKey().equals(old) ? rename : entry.getKey(), entry.getValue());
562 object(parent).clear();
563 object(parent).putAll(renamed);
564 }
565 }
566 history.add(copy(document));
567 document = changed;
568 return List.of(command + " " + targets.size());
569 }
570 if (command.equals("COUNT")) return List.of("COUNT " + matches.size());
571 if (Set.of("SUM", "MIN", "MAX", "AVG").contains(command)) {
572 long total = 0;
573 Integer minimum = null, maximum = null;
574 int count = 0;
575 for (Match match : matches) if (match.value() instanceof Integer n) {
576 total += n;
577 count++;
578 minimum = minimum == null ? n : Math.min(minimum, n);
579 maximum = maximum == null ? n : Math.max(maximum, n);
580 }
581 String result = command.equals("SUM") ? "" + total : command.equals("MIN") ? "" + minimum : command.equals("MAX") ? "" + maximum : "null";
582 if (command.equals("AVG") && count > 0) {
583 long q = (Math.abs(total) * 200 + count) / (2L * count);
584 result = (total < 0 && q > 0 ? "-" : "") + q / 100 + "." + String.format(Locale.ROOT, "%02d", q % 100);
585 }
586 return List.of(command + " " + result);
587 }
588 if (command.equals("FILTER")) {
589 for (Match match : matches) for (Match child : children(match)) {
590 Object f = field(child.value(), fields);
591 boolean equality = f != MISSING && equal(f, value), pass;
592 if (operator.equals("EXISTS")) pass = f != MISSING;
593 else if (operator.equals("==") || operator.equals("!=")) pass = operator.equals("==") == equality;
594 else {
595 boolean comparable = f instanceof Integer && value instanceof Integer || f instanceof String && value instanceof String;
596 int order = comparable ? compare(f, value) : 0;
597 pass = comparable && switch (operator) {
598 case "<" -> order < 0;
599 case "<=" -> order <= 0;
600 case ">" -> order > 0;
601 default -> order >= 0;
602 };
603 }
604 if (pass) out.add(location(child.location()) + " = " + compact(child.value()));
605 }
606 } else for (Match match : matches) {
607 String place = location(match.location());
608 Object item = match.value();
609 switch (command) {
610 case "GET":
611 out.add(place + " = " + compact(item));
612 break;
613 case "COMPACT":
614 out.add(compact(item));
615 break;
616 case "PRETTY":
617 out.add(pretty(item, ""));
618 break;
619 case "TYPE":
620 out.add(place + " " + type(item));
621 break;
622 case "LENGTH":
623 out.add(place + " " + (item instanceof Map ? object(item).size() : item instanceof List ? array(item).size() : ((String)item).length()));
624 break;
625 case "KEYS":
626 out.add(place + " " + compact(new ArrayList<>(object(item).keySet())));
627 break;
628 case "FLATTEN":
629 List<Match> descendants = new ArrayList<>();
630 preorder(match, descendants);
631 for (Match leaf : descendants) if (children(leaf).isEmpty()) out.add(location(leaf.location()) + " = " + compact(leaf.value()));
632 break;
633 }
634 }
635 return out.isEmpty() ? List.of("no match") : out;
636 }

638 static int[] size(Object v) {
639 int count = 1, depth = 1;
640 for (Match child : children(new Match(v, List.of()))) {
641 int[] childSize = size(child.value());
642 count += childSize[0];
643 depth = Math.max(depth, childSize[1] + 1);
644 }
645 return new int[]{count, depth};
646 }

648 static void load(List<String> body) {
649 String text = String.join("\n", body);
650 try {
651 Object value = new Parser(text).whole();
652 history.add(copy(document));
653 document = value;
654 int[] size = size(value);
655 System.out.println("LOADED nodes=" + size[0] + " depth=" + size[1]);
656 } catch (ParseError ex) {
657 int line = 1, column = 1;
658 for (int i = 0; i < ex.position; i++) {
659 if (text.charAt(i) == '\n') {
660 line++;
661 column = 1;
662 } else column++;
663 }
664 System.out.println("PARSE ERROR line " + line + " col " + column + ": " + ex.getMessage());
665 }
666 }
