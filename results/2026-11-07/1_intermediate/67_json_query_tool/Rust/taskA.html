4#[derive(Clone)]
5enum Value {

7 Array(Vec<Value>),

13fn identifier(s: &str) -> bool {
14 let mut b = s.bytes();
15 b.next().is_some_and(|c| c.is_ascii_alphabetic() || c == b'_')
16 && b.all(|c| c.is_ascii_alphanumeric() || c == b'_')
17}
18fn integer(s: &str) -> Option<i64> {
19 if s == "0" {
20 return Some(0);
21 }
22 let digits = s.strip_prefix('-').unwrap_or(s);
23 if (1..=9).contains(&digits.len()) && digits.as_bytes()[0] != b'0' && digits.bytes().all(|c| c.is_ascii_digit()) {
24 s.parse().ok()
25 } else {
26 None
27 }
28}
29fn quote(s: &str, q: char) -> String {
30 let mut out = q.to_string();
31 for c in s.chars() {
32 if c == q || c == '\\' {
33 out.push('\\');
34 out.push(c);
35 }
36 else {
37 match c {
38 '\x08' => out.push_str("\\b"), '\x0c' => out.push_str("\\f"), '\n' => out.push_str("\\n"),
39 '\r' => out.push_str("\\r"), '\t' => out.push_str("\\t"),
40 c if (c as u32) < 32 || c == '\x7f' => out.push_str(&format!("\\u{:04x}", c as u32)),
41 _ => out.push(c),
42 }
43 }
44 }
45 out.push(q);
46 out
47}
48impl Value {
49 fn kind(&self) -> &str {
50 match self {
51 Self::Object(_) => "object", Self::Array(_) => "array", Self::String(_) => "string", Self::Number(_) => "number", Self::Boolean(_) => "boolean", Self::Null => "null"
52 }
53 }
54 fn compact(&self) -> String {
55 match self {
56 Self::Object(o) => format!("{{{}}}", o.iter().map(|(k,v)| format!("{}:{}", quote(k,'"'), v.compact())).collect::<Vec<_>>().join(",")),
57 Self::Array(a) => format!("[{}]", a.iter().map(Value::compact).collect::<Vec<_>>().join(",")),
58 Self::String(s) => quote(s,'"'), Self::Number(n) => n.to_string(), Self::Boolean(b) => b.to_string(), Self::Null => "null".into(),
59 }
60 }
61 fn pretty(&self, indent: &str) -> String {
62 let next = format!("{indent} ");
63 match self {
64 Self::Object(o) if !o.is_empty() => format!("{{\n{}\n{indent}}}", o.iter().map(|(k,v)| format!("{next}{}: {}", quote(k,'"'), v.pretty(&next))).collect::<Vec<_>>().join(",\n")),
65 Self::Array(a) if !a.is_empty() => format!("[\n{}\n{indent}]", a.iter().map(|v| format!("{next}{}",v.pretty(&next))).collect::<Vec<_>>().join(",\n")),
66 _ => self.compact(),
67 }
68 }
69 fn children(&self) -> usize {
70 match self {
71 Self::Object(o) => o.len(), Self::Array(a) => a.len(), _ => 0
72 }
73 }
74 fn child(&self, i: usize) -> &Value {
75 match self {
76 Self::Object(o) => &o[i].1, Self::Array(a) => &a[i], _ => unreachable!()
77 }
78 }
79 fn length(&self) -> Option<usize> {
80 match self {
81 Self::Object(o) => Some(o.len()), Self::Array(a) => Some(a.len()), Self::String(s) => Some(s.len()), _ => None
82 }
83 }
84 fn size_depth(&self) -> (usize,usize) {
85 let mut count = 1;
86 let mut depth = 1;
87 for i in 0..self.children() {
88 let (n,d) = self.child(i).size_depth();
89 count += n;
90 depth = depth.max(d+1);
91 }
92 (count,depth)
93 }
94 fn equals(&self, other: &Value) -> bool {
95 match (self,other) {
96 (Self::Null,Self::Null) => true, (Self::Boolean(a),Self::Boolean(b)) => a==b,
97 (Self::Number(a),Self::Number(b)) => a==b, (Self::String(a),Self::String(b)) => a==b,
98 (Self::Array(a),Self::Array(b)) => a.len()==b.len() && a.iter().zip(b).all(|(x,y)| x.equals(y)),
99 (Self::Object(a),Self::Object(b)) => a.len()==b.len() && a.iter().all(|(k,v)| b.iter().find(|(key,_)| key==k).is_some_and(|(_,x)| v.equals(x))),
100 _ => false,
101 }
102 }

104struct ParseError {
105 position: usize, reason: String
106}
107struct Parser<'a> {
108 text: &'a str, at: usize
109}
110impl<'a> Parser<'a> {
111 fn error(&self, reason: impl Into<String>) -> ParseError {
112 ParseError {
113 position:self.at, reason:reason.into()
114 }
115 }
116 fn current(&self) -> Option<u8> {
117 self.text.as_bytes().get(self.at).copied()
118 }
119 fn skip(&mut self) {
120 while matches!(self.current(),Some(b' '|b'\n')) {
121 self.at+=1;
122 }
123 }
124 fn string(&mut self) -> Result<String,ParseError> {
125 self.at+=1;
126 let mut out=String::new();
127 loop {
128 match self.current() {
129 None | Some(b'\n') => return Err(self.error("unterminated string")),
130 Some(b'"') => {
131 self.at+=1;
132 return Ok(out);
133 }
134 Some(b'\\') => {
135 let start=self.at;
136 let e=self.text.as_bytes().get(self.at+1).copied();
137 match e {
138 None | Some(b'\n') => {
139 self.at+=1;
140 return Err(self.error("unterminated string"));
141 }
142 Some(b'u') => {
143 let digits=self.text.as_bytes().get(self.at+2..self.at+6).filter(|d|d.iter().all(|c|c.is_ascii_hexdigit())).ok_or_else(||self.error("bad unicode escape"))?;
144 let n=u32::from_str_radix(std::str::from_utf8(digits).unwrap(),16).unwrap();
145 if n>127 {
146 return Err(self.error("non-ascii escape"));
147 }
148 out.push(char::from_u32(n).unwrap());
149 self.at+=6;
150 }
151 Some(c) => {
152 let decoded=match c {
153 b'"'=> '"', b'\\'=>'\\', b'/'=>'/', b'b'=>'\x08', b'f'=>'\x0c', b'n'=>'\n', b'r'=>'\r', b't'=>'\t', _=>return Err(ParseError {
154 position:start,reason:format!("invalid escape '\\{}'",c as char)
155 })
156 };
157 out.push(decoded);
158 self.at+=2;
159 }
160 }
161 }
162 Some(c) => {
163 out.push(c as char);
164 self.at+=1;
165 }
166 }
167 }
168 }
169 fn value(&mut self) -> Result<Value,ParseError> {
170 self.skip();
171 match self.current() {
172 None => Err(self.error("unexpected end of input")),
173 Some(b'"') => self.string().map(Value::String),
174 Some(b'{') => {
175 self.at+=1;
176 self.skip();
177 let mut object=Vec::new();
178 if self.current()==Some(b'}') {
179 self.at+=1;
180 return Ok(Value::Object(object));
181 }
182 loop {
183 self.skip();
184 let start=self.at;
185 match self.current() {
186 None=>return Err(self.error("unexpected end of input")),Some(b'"')=>{
187 },_=>return Err(self.error("expected key"))
188 }
189 let key=self.string()?;
190 if object.iter().any(|(k,_)| k==&key) {
191 return Err(ParseError {
192 position:start,reason:format!("duplicate key {}",quote(&key,'"'))
193 });
194 }
195 self.skip();
196 match self.current() {
197 None=>return Err(self.error("unexpected end of input")),Some(b':')=>self.at+=1,_=>return Err(self.error("expected ':'"))
198 }
199 let value=self.value()?;
200 object.push((key,value));
201 self.skip();
202 match self.current() {
203 Some(b'}')=>{
204 self.at+=1;
205 break
206 },None=>return Err(self.error("unexpected end of input")),Some(b',')=>self.at+=1,_=>return Err(self.error("expected ',' or '}'"))
207 }
208 self.skip();
209 if self.current()==Some(b'}') {
210 return Err(self.error("trailing comma"));
211 }
212 }
213 Ok(Value::Object(object))
214 }
215 Some(b'[') => {
216 self.at+=1;
217 self.skip();
218 let mut array=Vec::new(); if self.current()==Some(b']') {
219 self.at+=1;
220 return Ok(Value::Array(array));
221 }
222 loop {
223 array.push(self.value()?);
224 self.skip();
225 match self.current() {
226 Some(b']')=>{
227 self.at+=1;
228 break
229 },None=>return Err(self.error("unexpected end of input")),Some(b',')=>self.at+=1,_=>return Err(self.error("expected ',' or ']'"))
230 }
231 self.skip(); if self.current()==Some(b']') {
232 return Err(self.error("trailing comma"));
233 }
234 }
235 Ok(Value::Array(array))
236 }
237 Some(c) if c==b'-' || c.is_ascii_digit() => {
238 let start=self.at;
239 if c==b'-' {
240 self.at+=1;
241 }
242 let digits=self.at;
243 while self.current().is_some_and(|c|c.is_ascii_digit()) {
244 self.at+=1;
245 }
246 if self.at==digits {
247 return Err(ParseError {
248 position:start,reason:"invalid number".into()
249 });
250 }
251 if self.at-digits>=2 && self.text.as_bytes()[digits]==b'0' {
252 return Err(ParseError {
253 position:digits,reason:"leading zero".into()
254 });
255 }
256 if self.current()==Some(b'.') {
257 return Err(self.error("fraction not allowed"));
258 }
259 if matches!(self.current(),Some(b'e'|b'E')) {
260 return Err(self.error("exponent not allowed"));
261 }
262 if self.at-digits>=10 {
263 return Err(ParseError {
264 position:start,reason:"number out of range".into()
265 });
266 }
267 Ok(Value::Number(self.text[start..self.at].parse().unwrap()))
268 }
269 Some(c) if c.is_ascii_alphabetic() => {
270 let start=self.at;
271 while self.current().is_some_and(|c|c.is_ascii_alphabetic()) {
272 self.at+=1;
273 }
274 match &self.text[start..self.at] {
275 "null"=>Ok(Value::Null),"true"=>Ok(Value::Boolean(true)),"false"=>Ok(Value::Boolean(false)),word=>Err(ParseError {
276 position:start,reason:format!("invalid literal '{word}'")
277 })
278 }
279 }
280 Some(c) => Err(self.error(format!("unexpected character '{}'",c as char))),
281 }
282 }
283 fn whole(text: &'a str) -> Result<Value,ParseError> {
284 let mut parser=Self {
285 text,at:0
286 };
287 let value=parser.value()?;
288 parser.skip();
289 if parser.at!=text.len() {
290 return Err(parser.error("trailing characters"));
291 }
292 Ok(value)
293 }
294}
295#[derive(Clone)]
296enum Selector {
297 Key(String), Index(i64), All, Slice(Option<i64>,Option<i64>)
298}
299#[derive(Clone)]
300struct Segment {
301 descendant: bool, selector: Selector
302}
303fn quoted_key(s: &str, at: &mut usize) -> Option<String> {
304 let b=s.as_bytes();
305 if b.get(*at)!=Some(&b'\'') {
306 return None;
307 }
308 *at+=1;
309 let mut out=String::new();
310 while *at<b.len() {
311 let c=b[*at];
312 *at+=1;
313 if c==b'\'' {
314 return Some(out);
315 }
316 if c!=b'\\' {
317 out.push(c as char);
318 continue;
319 }
320 let c=*b.get(*at)?;
321 *at+=1;
322 if c==b'u' {
323 let digits=b.get(*at..*at+4)?;
324 if !digits.iter().all(|c|c.is_ascii_hexdigit()) {
325 return None;
326 }
327 let n=u32::from_str_radix(std::str::from_utf8(digits).ok()?,16).ok()?;
328 if n>127 {
329 return None;
330 }
331 out.push(char::from_u32(n)?);
332 *at+=4;
333 } else {
334 out.push(match c {
335 b'\''=>'\'',b'\\'=>'\\',b'/'=>'/',b'b'=>'\x08',b'f'=>'\x0c',b'n'=>'\n',b'r'=>'\r',b't'=>'\t',_=>return None
336 });
337 }
338 }
339 None
340}
341fn path(s: &str, field: bool) -> Option<Vec<Segment>> {
342 let b=s.as_bytes();
343 if b.first()!=Some(&if field {
344 b'@'
345 } else {
346 b'$'
347 }) {
348 return None;
349 }
350 let mut at=1;
351 let mut segments=Vec::new();
352 while at<b.len() {
353 let mut descendant=false;
354 if b[at]==b'.' {
355 at+=1;
356 if b.get(at)==Some(&b'.') {
357 if field {
358 return None;
359 }
360 descendant=true;
361 at+=1;
362 }
363 else if b.get(at)==Some(&b'[') {
364 return None;
365 }
366 }
367 else if b[at]!=b'[' {
368 return None;
369 }
370 else {
371 }
372 let selector=if b.get(at)==Some(&b'[') {
373 at+=1;
374 if b.get(at)==Some(&b'\'') {
375 let key=quoted_key(s,&mut at)?; if b.get(at)!=Some(&b']') {
376 return None;
377 }
378 at+=1;
379 Selector::Key(key)
380 }
381 else {
382 let end=at+s[at..].find(']')?;
383 let item=&s[at..end];
384 at=end+1;
385 if item=="*" && !field {
386 Selector::All
387 }
388 else if let Some((a,b))=item.split_once(':') {
389 if field {
390 return None;
391 }
392 Selector::Slice(if a.is_empty() {
393 None
394 } else {
395 Some(integer(a)?)
396 },if b.is_empty() {
397 None
398 } else {
399 Some(integer(b)?)
400 })
401 }
402 else {
403 Selector::Index(integer(item)?)
404 }
405 }
406 } else if b.get(at)==Some(&b'*') && !field {
407 at+=1;
408 Selector::All
409 }
410 else {
411 let start=at;
412 while at<b.len() && (b[at].is_ascii_alphanumeric() || b[at]==b'_') {
413 at+=1;
414 }
415 let key=&s[start..at];
416 if !identifier(key) {
417 return None;
418 }
419 Selector::Key(key.into())
420 };
421 segments.push(Segment {
422 descendant,selector
423 });
424 }
425 Some(segments)
426}
427type Location = Vec<usize>;
428fn value_at<'a>(root: &'a Value, location: &[usize]) -> &'a Value {
429 let mut v=root;
430 for &i in location {
431 v=v.child(i);
432 }
433 v
434}
435fn value_at_mut<'a>(root: &'a mut Value, location: &[usize]) -> &'a mut Value {
436 if location.is_empty() {
437 return root;
438 }
439 let next=match root {
440 Value::Object(o)=>&mut o[location[0]].1,Value::Array(a)=>&mut a[location[0]],_=>unreachable!()
441 };
442 value_at_mut(next,&location[1..])
443}
444fn notation(root: &Value, location: &[usize]) -> String {
445 let mut out="$".to_string();
446 let mut value=root;
447 for &i in location {
448 match value {
449 Value::Object(o)=>{
450 let key=&o[i].0;
451 if identifier(key) {
452 out.push('.');
453 out+=key;
454 } else {
455 out+=&format!("[{}]",quote(key,'\''));
456 }
457 },_=>out+=&format!("[{i}]")
458 }
459 value=value.child(i);
460 }
461 out
462}
463fn preorder(root: &Value, location: &Location, out: &mut Vec<Location>) {
464 out.push(location.clone());
465 for i in 0..value_at(root,location).children() {
466 let mut child=location.clone();
467 child.push(i);
468 preorder(root,&child,out);
469 }
470}
471fn selection(value: &Value, selector: &Selector) -> Vec<usize> {
472 match (value,selector) {
473 (Value::Object(o),Selector::Key(k)) => o.iter().position(|(key,_)|key==k).into_iter().collect(),
474 (Value::Object(_)|Value::Array(_),Selector::All) => (0..value.children()).collect(),
475 (Value::Array(a),Selector::Index(i)) => {
476 let j=if *i<0 {
477 a.len() as i64+i
478 } else {
479 *i
480 };
481 if j>=0 && j<a.len() as i64 {
482 vec![j as usize]
483 } else {
484 Vec::new()
485 }
486 },
487 (Value::Array(a),Selector::Slice(start,end)) => {
488 let n=a.len() as i64;
489 let bound=|x:i64| (if x<0 {
490 x+n
491 } else {
492 x
493 }).clamp(0,n);
494 let a=bound(start.unwrap_or(0));
495 let b=bound(end.unwrap_or(n));
496 (a..b).map(|i|i as usize).collect()
497 },
498 _=>Vec::new(),
499 }
500}
501fn apply(root: &Value, matches: Vec<Location>, segment: &Segment) -> Vec<Location> {
502 let mut out=Vec::new();
503 for location in matches {
504 let mut parents=Vec::new();
505 if segment.descendant {
506 preorder(root,&location,&mut parents);
507 } else {
508 parents.push(location);
509 }
510 for parent in parents {
511 for i in selection(value_at(root,&parent),&segment.selector) {
512 let mut next=parent.clone();
513 next.push(i);
514 out.push(next);
515 }
516 }
517 }
518 out
519}
520fn evaluate(root: &Value, path: &[Segment]) -> Vec<Location> {
521 let mut matches=vec![Vec::new()];
522 for segment in path {
523 matches=apply(root,matches,segment);
524 }
525 matches
526}
527fn field_value<'a>(root: &'a Value, field: &[Segment]) -> Option<&'a Value> {
528 let mut value=root;
529 for segment in field {
530 let i=*selection(value,&segment.selector).first()?;
531 value=value.child(i);
532 }
533 Some(value)
534}
535fn sort_order(a: Option<&Value>, b: Option<&Value>) -> Ordering {
536 let rank=|v:Option<&Value>|match v {
537 Some(Value::Null)=>0,Some(Value::Number(_))=>1,Some(Value::String(_))=>2,_=>3
538 };
539 let order=rank(a).cmp(&rank(b));
540 if order!=Ordering::Equal {
541 return order;
542 }
543 match (a,b) {
544 (Some(Value::Number(a)),Some(Value::Number(b)))=>a.cmp(b),(Some(Value::String(a)),Some(Value::String(b)))=>a.cmp(b),_=>Ordering::Equal
545 }
546}
547fn merge(target: &mut Value, source: &Value) {
548 if let (Value::Object(t),Value::Object(m))=(target,source) {
549 for (key,value) in m {
550 if let Some((_,old))=t.iter_mut().find(|(k,_)|k==key) {
551 if matches!((&*old,value),(Value::Object(_),Value::Object(_))) {
552 merge(old,value);
553 } else {
554 *old=value.clone();
555 }
556 }
557 else {
558 t.push((key.clone(),value.clone()));
559 }
560 }
561 }

563struct Tool {
564 root: Value, history: Vec<Value>
565}
566impl Tool {
567 fn load(&mut self, text: &str) {
568 match Parser::whole(text) {
569 Ok(value)=>{
570 let (nodes,depth)=value.size_depth();
571 self.history.push(self.root.clone());
572 self.root=value;
573 println!("LOADED nodes={nodes} depth={depth}");
574 }
575 Err(error)=>{
576 let prefix=&text[..error.position];
577 let line=prefix.bytes().filter(|&c|c==b'\n').count()+1;
578 let col=prefix.rsplit('\n').next().unwrap().len()+1;
579 println!("PARSE ERROR line {line} col {col}: {}",error.reason);
580 }
581 }
582 }
583 fn command(&mut self, line: &str) -> Result<Vec<String>,String> {
584 let (command,arg)=line.split_once(' ').map_or((line,None),|(c,a)|(c,Some(a)));
585 let known=["GET","COMPACT","PRETTY","TYPE","LENGTH","KEYS","FLATTEN","COUNT","SUM","MIN","MAX","AVG","FILTER","SET","DELETE","INSERT","MERGE","SORT","RENAME","UNDO"];
586 if !known.contains(&command) {
587 return Err(format!("ERROR unknown command '{command}'"));
588 }
589 let words:Vec<_>=arg.map_or(Vec::new(),|a|if ["SET","MERGE"].contains(&command) {
590 a.splitn(2,' ').collect()
591 } else if command=="INSERT" {
592 a.splitn(3,' ').collect()
593 } else if command=="FILTER" {
594 a.splitn(4,' ').collect()
595 } else {
596 a.split(' ').collect()
597 });
598 let shape=match command {
599 "UNDO"=>words.is_empty(),"COMPACT"|"PRETTY"|"FLATTEN"=>words.len()<=1,"SET"|"MERGE"|"RENAME"=>words.len()==2,"INSERT"=>words.len()==3,"FILTER"=>words.len()>=3 && (if words[2]=="EXISTS" {
600 words.len()==3
601 } else {
602 words.len()==4
603 }),"SORT"=>(1..=3).contains(&words.len()) && (words.len()<3 || (words[1]!="DESC" && words[2]=="DESC")),_=>words.len()==1
604 };
605 if !shape {
606 return Err("ERROR bad arguments".into());
607 }
608 if command=="UNDO" {
609 self.root=self.history.pop().ok_or("ERROR nothing to undo")?;
610 return Ok(vec![format!("UNDO {}",self.history.len())]);
611 }
612 let parsed=path(words.first().copied().unwrap_or("$"),false).ok_or("ERROR bad path")?;
613 let mut field=Vec::new();
614 let mut index=0;
615 let mut rename=String::new();
616 let mut supplied=None;
617 if command=="FILTER" || command=="SORT" {
618 let f=if command=="FILTER" {
619 words[1]
620 } else {
621 words.get(1).filter(|&&s|s!="DESC").copied().unwrap_or("@")
622 };
623 field=path(f,true).ok_or("ERROR bad field")?;
624 }
625 if command=="FILTER" && !["==","!=","<","<=",">",">=","EXISTS"].contains(&words[2]) {
626 return Err("ERROR bad operator".into());
627 }
628 if command=="INSERT" {
629 index=integer(words[1]).ok_or("ERROR bad index")?;
630 }
631 if command=="RENAME" {
632 rename=if identifier(words[1]) {
633 words[1].into()
634 } else {
635 let mut at=0;
636 let key=quoted_key(words[1],&mut at).ok_or("ERROR bad key")?;
637 if at!=words[1].len() {
638 return Err("ERROR bad key".into());
639 }
640 key
641 };
642 }
643 let json=match command {
644 "SET"|"MERGE"=>Some(words[1]),"INSERT"=>Some(words[2]),"FILTER" if words[2]!="EXISTS"=>Some(words[3]),_=>None
645 };
646 if let Some(text)=json {
647 supplied=Some(Parser::whole(text).map_err(|e|format!("PARSE ERROR col {}: {}",e.position+1,e.reason))?);
648 }
649 if command=="MERGE" && !matches!(supplied,Some(Value::Object(_))) {
650 return Err("ERROR merge value is not an object".into());
651 }
652 let matches=evaluate(&self.root,&parsed);
653 let required=match command {
654 "FILTER"|"INSERT"|"SORT"=>Some("array"),"MERGE"|"KEYS"=>Some("object"),_=>None
655 };
656 for location in &matches {
657 let value=value_at(&self.root,location);
658 let loc=notation(&self.root,location);
659 if let Some(kind)=required {
660 if value.kind()!=kind {
661 return Err(format!("ERROR not an {kind} at {loc}"));
662 }
663 }
664 if command=="LENGTH" && value.length().is_none() {
665 return Err(format!("ERROR no length at {loc}"));
666 }
667 if command=="RENAME" && (location.is_empty() || !matches!(value_at(&self.root,&location[..location.len()-1]),Value::Object(_))) {
668 return Err(format!("ERROR not a member at {loc}"));
669 }
670 }
671 if command=="INSERT" {
672 for location in &matches {
673 let n=value_at(&self.root,location).children() as i64;
674 if index < -n-1 || index>n {
675 return Err(format!("ERROR index out of range at {}",notation(&self.root,location)));
676 }
677 }
678 }
679 let mut output=Vec::new();
680 match command {
681 "COUNT"=>return Ok(vec![format!("COUNT {}",matches.len())]),
682 "SUM"|"MIN"|"MAX"|"AVG"=>{
683 let nums:Vec<_>=matches.iter().filter_map(|l|if let Value::Number(n)=value_at(&self.root,l) {
684 Some(*n)
685 } else {
686 None
687 }).collect();
688 let sum:i64=nums.iter().sum();
689 let text=match command {
690 "SUM"=>sum.to_string(),"MIN"=>nums.iter().min().map_or("null".into(),|n|n.to_string()),"MAX"=>nums.iter().max().map_or("null".into(),|n|n.to_string()),_=>if nums.is_empty() {
691 "null".into()
692 } else {
693 let n=nums.len() as i64;
694 let q=(sum.abs()*200+n)/(2*n);
695 format!("{}{}.{:02}",if sum<0 && q>0 {
696 "-"
697 } else {
698 ""
699 },q/100,q%100)
700 }
701 };
702 return Ok(vec![format!("{command} {text}")]);
703 }
704 "GET"|"COMPACT"|"PRETTY"|"TYPE"|"LENGTH"|"KEYS"|"FLATTEN"|"FILTER"=>{
705 for location in &matches {
706 let value=value_at(&self.root,location);
707 let loc=notation(&self.root,location);
708 match command {
709 "GET"=>output.push(format!("{loc} = {}",value.compact())),"COMPACT"=>output.push(value.compact()),"PRETTY"=>output.push(value.pretty("")),"TYPE"=>output.push(format!("{loc} {}",value.kind())),"LENGTH"=>output.push(format!("{loc} {}",value.length().unwrap())),
710 "KEYS"=>if let Value::Object(o)=value {
711 output.push(format!("{loc} {}",Value::Array(o.iter().map(|(k,_)|Value::String(k.clone())).collect()).compact()));
712 },
713 "FLATTEN"=>{
714 let mut descendants=Vec::new();
715 preorder(&self.root,location,&mut descendants);
716 for leaf in descendants {
717 let v=value_at(&self.root,&leaf);
718 if v.children()==0 {
719 output.push(format!("{} = {}",notation(&self.root,&leaf),v.compact()));
720 }
721 }
722 },
723 _=>for i in 0..value.children() {
724 let element=value.child(i);
725 let f=field_value(element,&field);
726 let r=supplied.as_ref();
727 let equal=f.zip(r).is_some_and(|(a,b)|a.equals(b));
728 let comparison=f.zip(r).and_then(|(a,b)|match (a,b) {
729 (Value::Number(a),Value::Number(b))=>Some(a.cmp(b)),(Value::String(a),Value::String(b))=>Some(a.cmp(b)),_=>None
730 });
731 let good=match words[2] {
732 "EXISTS"=>f.is_some(),"=="=>equal,"!="=>!equal,"<"=>comparison==Some(Ordering::Less),"<="=>comparison.is_some_and(|c|c!=Ordering::Greater),">"=>comparison==Some(Ordering::Greater),_=>comparison.is_some_and(|c|c!=Ordering::Less)
733 };
734 if good {
735 let mut child=location.clone();
736 child.push(i);
737 output.push(format!("{} = {}",notation(&self.root,&child),element.compact()));
738 }
739 },
740 }
741 }
742 if output.is_empty() {
743 output.push("no match".into());
744 }
745 return Ok(output);
746 }
747 _=>{
748 }
749 }
750 if command=="DELETE" && parsed.is_empty() {
751 return Err("ERROR cannot delete root".into());
752 }
753 let mut targets=matches;
754 let new_key=if command=="SET" {
755 parsed.last().and_then(|s|if !s.descendant {
756 if let Selector::Key(k)=&s.selector {
757 Some(k.clone())
758 } else {
759 None
760 }
761 } else {
762 None
763 })
764 } else {
765 None
766 };
767 if command=="SET" && !parsed.is_empty() {
768 if let Some(key)=&new_key {
769 targets=Vec::new();
770 for parent in evaluate(&self.root,&parsed[..parsed.len()-1]) {
771 if let Value::Object(o)=value_at(&self.root,&parent) {
772 let mut target=parent;
773 target.push(o.iter().position(|(k,_)|k==key).unwrap_or(o.len()));
774 targets.push(target);
775 }
776 }
777 }
778 }
779 let targets:BTreeSet<_>=targets.into_iter().collect();
780 let count=targets.len();
781 let mut changed=self.root.clone();
782 for location in targets.iter().rev() {
783 if command=="SET" {
784 if location.is_empty() {
785 changed=supplied.as_ref().unwrap().clone();
786 }
787 else {
788 let (&i,parent)=location.split_last().unwrap();
789 match value_at_mut(&mut changed,parent) {
790 Value::Object(o)=>{
791 if i==o.len() {
792 o.push((new_key.as_ref().unwrap().clone(),supplied.as_ref().unwrap().clone()));
793 } else {
794 o[i].1=supplied.as_ref().unwrap().clone();
795 }
796 },Value::Array(a)=>a[i]=supplied.as_ref().unwrap().clone(),_=>unreachable!()
797 }
798 }
799 } else if command=="DELETE" {
800 let (&i,parent)=location.split_last().unwrap();
801 match value_at_mut(&mut changed,parent) {
802 Value::Object(o)=>{
803 o.remove(i);
804 },Value::Array(a)=>{
805 a.remove(i);
806 },_=>unreachable!()
807 }
808 }
809 else if command=="RENAME" {
810 let (&i,parent)=location.split_last().unwrap();
811 if let Value::Object(o)=value_at_mut(&mut changed,parent) {
812 if o[i].0!=rename && o.iter().any(|(k,_)|k==&rename) {
813 return Err(format!("ERROR key exists at {}",notation(&self.root,location)));
814 }
815 o[i].0=rename.clone();
816 }
817 } else {
818 let target=value_at_mut(&mut changed,location);
819 match command {
820 "MERGE"=>merge(target,supplied.as_ref().unwrap()),
821 "INSERT"=>if let Value::Array(a)=target {
822 let j=if index<0 {
823 a.len() as i64+1+index
824 } else {
825 index
826 };
827 a.insert(j as usize,supplied.as_ref().unwrap().clone());
828 },
829 "SORT"=>if let Value::Array(a)=target {
830 let desc=words.last()==Some(&"DESC");
831 a.sort_by(|a,b| {
832 let order=sort_order(field_value(a,&field),field_value(b,&field));
833 if desc {
834 order.reverse()
835 } else {
836 order
837 }
838 });
839 },
840 _=>unreachable!(),
841 }
842 }
843 }
844 self.history.push(self.root.clone());
845 self.root=changed;
846 Ok(vec![format!("{command} {count}")])
847 }
848}
849fn main() {

853 let mut at=0;

856 };

864 while lines[at]!="LOAD" && lines[at]!="END" {
865 match tool.command(lines[at]) {
866 Ok(output)=>for line in output {
867 println!("{line}");
868 },Err(error)=>println!("{error}")
869 }
870 at+=1;
871 }
872 if lines[at]=="END" {
873 break;
874 }
875 at+=1;
876 }
