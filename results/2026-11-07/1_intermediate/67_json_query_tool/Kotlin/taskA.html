1private sealed interface Value
2private data class ObjectValue(val members: List<Pair<String, Value>>) : Value
3private data class ArrayValue(val elements: List<Value>) : Value
4private data class StringValue(val text: String) : Value
5private data class NumberValue(val number: Long) : Value
6private data class BooleanValue(val boolean: Boolean) : Value
7private data object NullValue : Value
8private class Problem(val text: String) : RuntimeException(text)
9private class ParseProblem(val position: Int, val reason: String) : RuntimeException(reason)
10private fun identifier(s: String) = s.matches(Regex("[A-Za-z_][A-Za-z0-9_]*"))
11private fun integer(s: String): Long? = if (s.matches(Regex("0|-?[1-9][0-9]{0,8}"))) s.toLong() else null
12private fun quote(s: String, q: Char): String = buildString {
13 append(q)
14 for (c in s) {
15 if (c == q || c == '\\') append('\\').append(c)
16 else append(when (c) {
17 '\b' -> "\\b"
18 '\u000c' -> "\\f"
19 '\n' -> "\\n"
20 '\r' -> "\\r"
21 '\t' -> "\\t"
22 else -> if (c.code < 32 || c.code == 127) "\\u%04x".format(c.code) else c.toString()
23 })
24 }
25 append(q)
26}
27private fun kind(v: Value): String = when (v) {
28 is ObjectValue -> "object"
29 is ArrayValue -> "array"
30 is StringValue -> "string"
31 is NumberValue -> "number"
32 is BooleanValue -> "boolean"
33 NullValue -> "null"
34}
35private fun compact(v: Value): String = when (v) {
36 is ObjectValue -> v.members.joinToString(",", "{", "}") {
37 quote(it.first, '"')+":"+compact(it.second)
38 }
39 is ArrayValue -> v.elements.joinToString(",", "[", "]", transform = ::compact)
40 is StringValue -> quote(v.text, '"')
41 is NumberValue -> v.number.toString()
42 is BooleanValue -> v.boolean.toString()
43 NullValue -> "null"
44}
45private fun pretty(v: Value, indent: String = ""): String {
46 val next = "$indent "
47 return when {
48 v is ObjectValue && v.members.isNotEmpty() -> "{\n"+v.members.joinToString(",\n") {
49 next+quote(it.first, '"')+": "+pretty(it.second, next)
50 }+"\n$indent}"
51 v is ArrayValue && v.elements.isNotEmpty() -> "[\n"+v.elements.joinToString(",\n") {
52 next+pretty(it, next)
53 }+"\n$indent]"
54 else -> compact(v)
55 }
56}
57private fun children(v: Value): Int = when (v) {
58 is ObjectValue -> v.members.size
59 is ArrayValue -> v.elements.size
60 else -> 0
61}
62private fun child(v: Value, i: Int): Value = when (v) {
63 is ObjectValue -> v.members[i].second
64 is ArrayValue -> v.elements[i]
65 else -> error("not a container")
66}
67private fun length(v: Value): Int? = when (v) {
68 is ObjectValue -> v.members.size
69 is ArrayValue -> v.elements.size
70 is StringValue -> v.text.length
71 else -> null
72}
73private fun sizeDepth(v: Value): Pair<Int, Int> {
74 var size = 1
75 var depth = 1
76 for (i in 0 until children(v)) {
77 val (n, d) = sizeDepth(child(v, i))
78 size += n
79 depth = maxOf(depth, d+1)
80 }
81 return size to depth
82}
83private fun equal(a: Value, b: Value): Boolean = when {
84 a is ObjectValue && b is ObjectValue -> a.members.size == b.members.size && a.members.all {
85 (key, value) -> b.members.firstOrNull {
86 it.first == key
87 }?.let {
88 equal(value, it.second)
89 } == true
90 }
91 a is ArrayValue && b is ArrayValue -> a.elements.size == b.elements.size && a.elements.zip(b.elements).all {
92 equal(it.first, it.second)
93 }
94 else -> a == b
95}
96private class JsonParser(private val text: String) {
97 private var at = 0
98 private fun current() = text.getOrNull(at)
99 private fun fail(reason: String, position: Int = at): Nothing = throw ParseProblem(position, reason)
100 private fun skip() {
101 while (current() == ' ' || current() == '\n') at++
102 }
103 private fun string(): String {
104 at++
105 val out = StringBuilder()
106 while (true) {
107 when (val c = current()) {
108 null, '\n' -> fail("unterminated string")
109 '"' -> {
110 at++
111 return out.toString()
112 }
113 '\\' -> {
114 val start = at
115 when (val e = text.getOrNull(at+1)) {
116 null, '\n' -> fail("unterminated string", at+1)
117 'u' -> {
118 if (at+6 > text.length || text.substring(at+2, at+6).any {
119 it.digitToIntOrNull(16) == null
120 }) fail("bad unicode escape")
121 val n = text.substring(at+2, at+6).toInt(16)
122 if (n > 127) fail("non-ascii escape")
123 out.append(n.toChar())
124 at += 6
125 }
126 else -> {
127 out.append(when (e) {
128 '"' -> '"'
129 '\\' -> '\\'
130 '/' -> '/'
131 'b' -> '\b'
132 'f' -> '\u000c'
133 'n' -> '\n'
134 'r' -> '\r'
135 't' -> '\t'
136 else -> fail("invalid escape '\\$e'", start)
137 })
138 at += 2
139 }
140 }
141 }
142 else -> {
143 out.append(c)
144 at++
145 }
146 }
147 }
148 }
149 private fun value(): Value {
150 skip()
151 return when (val c = current()) {
152 null -> fail("unexpected end of input")
153 '"' -> StringValue(string())
154 '{' -> {
155 at++
156 skip()
157 val members = mutableListOf<Pair<String, Value>>()
158 if (current() == '}') {
159 at++
160 return ObjectValue(members)
161 }
162 while (true) {
163 skip()
164 val start = at
165 if (current() == null) fail("unexpected end of input")
166 if (current() != '"') fail("expected key")
167 val key = string()
168 if (members.any {
169 it.first == key
170 }) fail("duplicate key ${quote(key, '"')}", start)
171 skip()
172 if (current() == null) fail("unexpected end of input")
173 if (current() != ':') fail("expected ':'")
174 at++
175 members.add(key to value())
176 skip()
177 when (current()) {
178 '}' -> {
179 at++
180 break
181 }
182 null -> fail("unexpected end of input")
183 ',' -> at++
184 else -> fail("expected ',' or '}'")
185 }
186 skip()
187 if (current() == '}') fail("trailing comma")
188 }
189 ObjectValue(members)
190 }
191 '[' -> {
192 at++
193 skip()
194 val elements = mutableListOf<Value>()
195 if (current() == ']') {
196 at++
197 return ArrayValue(elements)
198 }
199 while (true) {
200 elements.add(value())
201 skip()
202 when (current()) {
203 ']' -> {
204 at++
205 break
206 }
207 null -> fail("unexpected end of input")
208 ',' -> at++
209 else -> fail("expected ',' or ']'")
210 }
211 skip()
212 if (current() == ']') fail("trailing comma")
213 }
214 ArrayValue(elements)
215 }
216 else -> when {
217 c == '-' || c in '0'..'9' -> {
218 val start = at
219 if (c == '-') at++
220 val digits = at
221 while (current()?.let {
222 it in '0'..'9'
223 } == true) at++
224 if (at == digits) fail("invalid number", start)
225 if (at-digits >= 2 && text[digits] == '0') fail("leading zero", digits)
226 if (current() == '.') fail("fraction not allowed")
227 if (current() == 'e' || current() == 'E') fail("exponent not allowed")
228 if (at-digits >= 10) fail("number out of range", start)
229 NumberValue(text.substring(start, at).toLong())
230 }
231 c in 'A'..'Z' || c in 'a'..'z' -> {
232 val start = at
233 while (current()?.let {
234 it in 'A'..'Z' || it in 'a'..'z'
235 } == true) at++
236 when (val word = text.substring(start, at)) {
237 "null" -> NullValue
238 "true" -> BooleanValue(true)
239 "false" -> BooleanValue(false)
240 else -> fail("invalid literal '$word'", start)
241 }
242 }
243 else -> fail("unexpected character '$c'")
244 }
245 }
246 }
247 fun parse(): Value {
248 val value = value()
249 skip()
250 if (at != text.length) fail("trailing characters")
251 return value
252 }
253}
254private sealed interface Selector
255private data class Key(val key: String) : Selector
256private data class Index(val index: Long) : Selector
257private data object All : Selector
258private data class Slice(val start: Long?, val end: Long?) : Selector
259private data class Segment(val descendant: Boolean, val selector: Selector)
260private class PathParser(private val text: String) {
261 var at = 0
262 fun quoted(): String? {
263 if (text.getOrNull(at) != '\'') return null
264 at++
265 val out = StringBuilder()
266 while (at < text.length) {
267 val c = text[at++]
268 if (c == '\'') return out.toString()
269 if (c != '\\') {
270 out.append(c)
271 continue
272 }
273 val e = text.getOrNull(at++) ?: return null
274 if (e == 'u') {
275 if (at+4 > text.length || text.substring(at, at+4).any {
276 it.digitToIntOrNull(16) == null
277 }) return null
278 val n = text.substring(at, at+4).toInt(16)
279 if (n > 127) return null
280 out.append(n.toChar())
281 at += 4
282 } else out.append(when (e) {
283 '\'' -> '\''
284 '\\' -> '\\'
285 '/' -> '/'
286 'b' -> '\b'
287 'f' -> '\u000c'
288 'n' -> '\n'
289 'r' -> '\r'
290 't' -> '\t'
291 else -> return null
292 })
293 }
294 return null
295 }
296 fun parse(field: Boolean): List<Segment>? {
297 if (text.firstOrNull() != if (field) '@' else '$') return null
298 at = 1
299 val segments = mutableListOf<Segment>()
300 while (at < text.length) {
301 var descendant = false
302 if (text[at] == '.') {
303 at++
304 if (text.getOrNull(at) == '.') {
305 if (field) return null
306 descendant = true
307 at++
308 }
309 else if (text.getOrNull(at) == '[') return null
310 } else if (text[at] != '[') return null
311 val selector: Selector
312 if (text.getOrNull(at) == '[') {
313 at++
314 if (text.getOrNull(at) == '\'') {
315 val key = quoted() ?: return null
316 if (text.getOrNull(at++) != ']') return null
317 selector = Key(key)
318 } else {
319 val end = text.indexOf(']', at)
320 if (end < 0) return null
321 val item = text.substring(at, end)
322 at = end+1
323 selector = if (item == "*" && !field) All else if (':' in item) {
324 if (field || item.count {
325 it == ':'
326 } != 1) return null
327 val a = item.substringBefore(':')
328 val b = item.substringAfter(':')
329 Slice(if (a.isEmpty()) null else integer(a) ?: return null, if (b.isEmpty()) null else integer(b) ?: return null)
330 } else Index(integer(item) ?: return null)
331 }
332 } else if (text.getOrNull(at) == '*' && !field) {
333 at++
334 selector = All
335 }
336 else {
337 val start = at
338 while (at < text.length && (text[at] in 'A'..'Z' || text[at] in 'a'..'z' || text[at] in '0'..'9' || text[at] == '_')) at++
339 val key = text.substring(start, at)
340 if (!identifier(key)) return null
341 selector = Key(key)
342 }
343 segments.add(Segment(descendant, selector))
344 }
345 return segments
346 }
347}
348private typealias Location = List<Int>
349private fun valueAt(root: Value, location: Location): Value = location.fold(root, ::child)
350private fun update(root: Value, location: Location, change: (Value) -> Value): Value {
351 if (location.isEmpty()) return change(root)
352 val i = location[0]
353 val next = update(child(root, i), location.drop(1), change)
354 return when (root) {
355 is ObjectValue -> ObjectValue(root.members.toMutableList().also {
356 it[i] = it[i].first to next
357 })
358 is ArrayValue -> ArrayValue(root.elements.toMutableList().also {
359 it[i] = next
360 })
361 else -> error("not a container")
362 }
363}
364private fun notation(root: Value, location: Location): String = buildString {
365 append('$')
366 var value = root
367 for (i in location) {
368 if (value is ObjectValue) {
369 val key = value.members[i].first
370 if (identifier(key)) append('.').append(key) else append('[').append(quote(key, '\'')).append(']')
371 }
372 else append('[').append(i).append(']')
373 value = child(value, i)
374 }
375}
376private fun preorder(root: Value, location: Location, out: MutableList<Location>) {
377 out.add(location)
378 for (i in 0 until children(valueAt(root, location))) preorder(root, location+i, out)
379}
380private fun selection(v: Value, s: Selector): List<Int> = when {
381 v is ObjectValue && s is Key -> v.members.indexOfFirst {
382 it.first == s.key
383 }.takeIf {
384 it >= 0
385 }?.let(::listOf) ?: emptyList()
386 (v is ObjectValue || v is ArrayValue) && s == All -> (0 until children(v)).toList()
387 v is ArrayValue && s is Index -> {
388 val j = if (s.index < 0) v.elements.size+s.index else s.index
389 if (j >= 0 && j < v.elements.size) listOf(j.toInt()) else emptyList()
390 }
391 v is ArrayValue && s is Slice -> {
392 val n = v.elements.size.toLong()
393 fun bound(x: Long) = (if (x < 0) x+n else x).coerceIn(0, n)
394 (bound(s.start ?: 0).toInt() until bound(s.end ?: n).toInt()).toList()
395 }
396 else -> emptyList()
397}
398private fun apply(root: Value, matches: List<Location>, segment: Segment): List<Location> = buildList {
399 for (location in matches) {
400 val parents = mutableListOf<Location>()
401 if (segment.descendant) preorder(root, location, parents) else parents.add(location)
402 for (parent in parents) for (i in selection(valueAt(root, parent), segment.selector)) add(parent+i)
403 }
404}
405private fun evaluate(root: Value, path: List<Segment>): List<Location> = path.fold(listOf(emptyList<Int>()), {
406 matches, segment -> apply(root, matches, segment)
407})
408private fun fieldValue(root: Value, field: List<Segment>): Value? {
409 var value = root
410 for (segment in field) value = child(value, selection(value, segment.selector).firstOrNull() ?: return null)
411 return value
412}
413private fun sortOrder(a: Value?, b: Value?): Int {
414 fun rank(v: Value?) = when (v) {
415 NullValue -> 0
416 is NumberValue -> 1
417 is StringValue -> 2
418 else -> 3
419 }
420 val order = rank(a).compareTo(rank(b))
421 if (order != 0) return order
422 return when {
423 a is NumberValue && b is NumberValue -> a.number.compareTo(b.number)
424 a is StringValue && b is StringValue -> a.text.compareTo(b.text)
425 else -> 0
426 }
427}
428private fun merge(target: Value, source: Value): Value {
429 val members = (target as ObjectValue).members.toMutableList()
430 for ((key, value) in (source as ObjectValue).members) {
431 val i = members.indexOfFirst {
432 it.first == key
433 }
434 if (i < 0) members.add(key to value)
435 else members[i] = key to if (members[i].second is ObjectValue && value is ObjectValue) merge(members[i].second, value) else value
436 }
437 return ObjectValue(members)
438}
439private class Tool {
440 var root: Value = NullValue
441 val history = mutableListOf<Value>()
442 fun load(text: String) {
443 try {
444 val value = JsonParser(text).parse()
445 val (nodes, depth) = sizeDepth(value)
446 history.add(root)
447 root = value
448 println("LOADED nodes=$nodes depth=$depth")
449 } catch (p: ParseProblem) {
450 val prefix = text.substring(0, p.position)
451 println("PARSE ERROR line ${prefix.count { it == '\n' }+1} col ${prefix.substringAfterLast('\n').length+1}: ${p.reason}")
452 }

454 fun command(line: String): List<String> {
455 val command = line.substringBefore(' ')
456 val arg = if (' ' in line) line.substringAfter(' ') else null
457 val known = setOf("GET", "COMPACT", "PRETTY", "TYPE", "LENGTH", "KEYS", "FLATTEN", "COUNT", "SUM", "MIN", "MAX", "AVG", "FILTER", "SET", "DELETE", "INSERT", "MERGE", "SORT", "RENAME", "UNDO")
458 if (command !in known) throw Problem("ERROR unknown command '$command'")
459 val words = arg?.split(' ', limit = when (command) {
460 "SET", "MERGE" -> 2
461 "INSERT" -> 3
462 "FILTER" -> 4
463 else -> 0
464 }) ?: emptyList()
465 val shape = when (command) {
466 "UNDO" -> words.isEmpty()
467 "COMPACT", "PRETTY", "FLATTEN" -> words.size <= 1
468 "SET", "MERGE", "RENAME" -> words.size == 2
469 "INSERT" -> words.size == 3
470 "FILTER" -> words.size >= 3 && if (words[2] == "EXISTS") words.size == 3 else words.size == 4
471 "SORT" -> words.size in 1..3 && (words.size < 3 || (words[1] != "DESC" && words[2] == "DESC"))
472 else -> words.size == 1
473 }
474 if (!shape) throw Problem("ERROR bad arguments")
475 if (command == "UNDO") {
476 if (history.isEmpty()) throw Problem("ERROR nothing to undo")
477 root = history.removeAt(history.lastIndex)
478 return listOf("UNDO ${history.size}")
479 }
480 val path = PathParser(words.firstOrNull() ?: "$").parse(false) ?: throw Problem("ERROR bad path")
481 val field = if (command == "FILTER" || command == "SORT") PathParser(if (command == "FILTER") words[1] else words.getOrNull(1)?.takeIf {
482 it != "DESC"
483 } ?: "@").parse(true) ?: throw Problem("ERROR bad field") else emptyList()
484 if (command == "FILTER" && words[2] !in setOf("==", "!=", "<", "<=", ">", ">=", "EXISTS")) throw Problem("ERROR bad operator")
485 val index = if (command == "INSERT") integer(words[1]) ?: throw Problem("ERROR bad index") else 0L
486 val rename = if (command == "RENAME") {
487 if (identifier(words[1])) words[1] else {
488 val p = PathParser(words[1])
489 val key = p.quoted() ?: throw Problem("ERROR bad key")
490 if (p.at != words[1].length) throw Problem("ERROR bad key")
491 key
492 }
493 } else ""
494 val json = when (command) {
495 "SET", "MERGE" -> words[1]
496 "INSERT" -> words[2]
497 "FILTER" -> if (words[2] != "EXISTS") words[3] else null
498 else -> null
499 }
500 val supplied = json?.let {
501 try {
502 JsonParser(it).parse()
503 } catch (p: ParseProblem) {
504 throw Problem("PARSE ERROR col ${p.position+1}: ${p.reason}")
505 }
506 }
507 if (command == "MERGE" && supplied !is ObjectValue) throw Problem("ERROR merge value is not an object")
508 val matches = evaluate(root, path)
509 val required = when (command) {
510 "FILTER", "INSERT", "SORT" -> "array"
511 "MERGE", "KEYS" -> "object"
512 else -> null
513 }
514 for (location in matches) {
515 val value = valueAt(root, location)
516 val loc = notation(root, location)
517 if (required != null && kind(value) != required) throw Problem("ERROR not an $required at $loc")
518 if (command == "LENGTH" && length(value) == null) throw Problem("ERROR no length at $loc")
519 if (command == "RENAME" && (location.isEmpty() || valueAt(root, location.dropLast(1)) !is ObjectValue)) throw Problem("ERROR not a member at $loc")
520 }
521 if (command == "INSERT") for (location in matches) {
522 val n = children(valueAt(root, location)).toLong()
523 if (index < -n-1 || index > n) throw Problem("ERROR index out of range at ${notation(root, location)}")
524 }
525 if (command == "COUNT") return listOf("COUNT ${matches.size}")
526 if (command in setOf("SUM", "MIN", "MAX", "AVG")) {
527 val nums = matches.mapNotNull {
528 (valueAt(root, it) as? NumberValue)?.number
529 }
530 val sum = nums.sum()
531 val result = when (command) {
532 "SUM" -> sum.toString()
533 "MIN" -> nums.minOrNull()?.toString() ?: "null"
534 "MAX" -> nums.maxOrNull()?.toString() ?: "null"
535 else -> if (nums.isEmpty()) "null" else {
536 val n = nums.size.toLong()
537 val q = (kotlin.math.abs(sum)*200+n)/(2*n)
538 (if (sum < 0 && q > 0) "-" else "")+"${q/100}."+(q%100).toString().padStart(2, '0')
539 }
540 }
541 return listOf("$command $result")
542 }
543 if (command in setOf("GET", "COMPACT", "PRETTY", "TYPE", "LENGTH", "KEYS", "FLATTEN", "FILTER")) {
544 val output = mutableListOf<String>()
545 for (location in matches) {
546 val value = valueAt(root, location)
547 val loc = notation(root, location)
548 when (command) {
549 "GET" -> output.add("$loc = ${compact(value)}")
550 "COMPACT" -> output.add(compact(value))
551 "PRETTY" -> output.add(pretty(value))
552 "TYPE" -> output.add("$loc ${kind(value)}")
553 "LENGTH" -> output.add("$loc ${length(value)}")
554 "KEYS" -> output.add("$loc ${compact(ArrayValue((value as ObjectValue).members.map { StringValue(it.first) }))}")
555 "FLATTEN" -> {
556 val leaves = mutableListOf<Location>()
557 preorder(root, location, leaves)
558 for (leaf in leaves) if (children(valueAt(root, leaf)) == 0) output.add("${notation(root, leaf)} = ${compact(valueAt(root, leaf))}")
559 }
560 else -> for (i in 0 until children(value)) {
561 val element = child(value, i)
562 val f = fieldValue(element, field)
563 val eq = f != null && supplied != null && equal(f, supplied)
564 val cmp = when {
565 f is NumberValue && supplied is NumberValue -> f.number.compareTo(supplied.number)
566 f is StringValue && supplied is StringValue -> f.text.compareTo(supplied.text)
567 else -> null
568 }
569 val good = when (words[2]) {
570 "EXISTS" -> f != null
571 "==" -> eq
572 "!=" -> !eq
573 "<" -> cmp != null && cmp < 0
574 "<=" -> cmp != null && cmp <= 0
575 ">" -> cmp != null && cmp > 0
576 else -> cmp != null && cmp >= 0
577 }
578 if (good) output.add("${notation(root, location+i)} = ${compact(element)}")
579 }
580 }
581 }
582 return output.ifEmpty {
583 listOf("no match")
584 }
585 }
586 if (command == "DELETE" && path.isEmpty()) throw Problem("ERROR cannot delete root")
587 val newKey = if (command == "SET" && path.lastOrNull()?.descendant == false) (path.last().selector as? Key)?.key else null
588 val targets = if (command == "SET" && newKey != null) evaluate(root, path.dropLast(1)).mapNotNull {
589 parent ->
590 val objectValue = valueAt(root, parent) as? ObjectValue ?: return@mapNotNull null
591 parent+(objectValue.members.indexOfFirst {
592 it.first == newKey
593 }.takeIf {
594 it >= 0
595 } ?: objectValue.members.size)
596 } else matches
597 val order = Comparator<Location> {
598 a, b -> var result = 0
599 for (i in 0 until minOf(a.size, b.size)) {
600 result = a[i].compareTo(b[i])
601 if (result != 0) break
602 }
603 if (result == 0) a.size.compareTo(b.size) else result
604 }
605 val unique = targets.distinct().sortedWith(order.reversed())
606 var changed = root
607 for (location in unique) {
608 if (command in setOf("SET", "DELETE", "RENAME") && location.isNotEmpty()) {
609 val i = location.last()
610 changed = update(changed, location.dropLast(1)) {
611 parent ->
612 when (parent) {
613 is ObjectValue -> {
614 val members = parent.members.toMutableList()
615 when (command) {
616 "SET" -> if (i == members.size) members.add(newKey!! to supplied!!) else members[i] = members[i].first to supplied!!
617 "DELETE" -> members.removeAt(i)
618 else -> {
619 if (members[i].first != rename && members.any {
620 it.first == rename
621 }) throw Problem("ERROR key exists at ${notation(root, location)}")
622 members[i] = rename to members[i].second
623 }
624 }
625 ObjectValue(members)
626 }
627 is ArrayValue -> {
628 val elements = parent.elements.toMutableList()
629 if (command == "SET") elements[i] = supplied!! else elements.removeAt(i)
630 ArrayValue(elements)
631 }
632 else -> error("not a container")
633 }
634 }
635 } else changed = update(changed, location) {
636 target ->
637 when (command) {
638 "SET" -> supplied!!
639 "MERGE" -> merge(target, supplied!!)
640 "INSERT" -> {
641 val elements = (target as ArrayValue).elements.toMutableList()
642 val j = if (index < 0) elements.size+1+index else index
643 elements.add(j.toInt(), supplied!!)
644 ArrayValue(elements)
645 }
646 "SORT" -> {
647 val desc = words.last() == "DESC"
648 ArrayValue((target as ArrayValue).elements.sortedWith {
649 a, b -> val cmp = sortOrder(fieldValue(a, field), fieldValue(b, field))
650 if (desc) -cmp else cmp
651 })
652 }
653 else -> error("invalid command")
654 }
655 }
656 }
657 history.add(root)
658 root = changed
659 return listOf("$command ${unique.size}")
660 }
661}

664 var at = 0
