bool one_of(string[] words, string word)
    for item := words[]
        if item == word
            return true
    return false
bool all_chars(string text, string alphabet)
    for: i < #text
        if not alphabet.contains(text[i])
            return false
    return true
bool identifier(string text)
    return text != "" and all_chars(text[0], "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ_") and all_chars(text, "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ_0123456789")
nat find_at(string text, string needle)
    match text.find(needle)
        nat at
            return at
        none
            return #text
string ascii(nat code)
    return string.from_utf8(bin([nat8(code)]))
string quote_text(string text, string mark := DQ)
    result := mark
    for code := text.codepoints()[]
        ch := ascii(nat(code))
        if ch == mark or ch == "\"
            result += "\" + ch
        elif code < 32 or code == 127
        else
            result += ch
    return result + mark
class Value(string kind, text := "", number := 0)
    string[] keys := []
    nat[] items := []
class Step(bool key_step, string key, nat position)
class Match(nat node, Step[] steps)
class Segment(string kind, key := "", index := 0)
    int | none start := none
    int | none stop := none
    descendant := false
class Tree(Value[] nodes, nat root)
    nat .add(Value value)
        index := #nodes
        nodes = nodes.append(value)
        return index
    int key_index(nat index, string key)
        for: item := nodes[index].keys[i]
            if item == key
                return int(i)
        return -1
    nat count_nodes(nat index)
        nat total := 1
        for item := nodes[index].items[]
            total += count_nodes(item)
        return total
    nat depth(nat index)
        nat longest := 0
        for item := nodes[index].items[]
            nat child_depth := depth(item)
            if child_depth > longest
                longest = child_depth
        return longest + 1
    string compact(nat index)
        value := nodes[index]
        if value.kind == "null" or value.kind == "boolean"
            return value.text
    string pretty(nat index, indent := "")
        value := nodes[index]
        if #value.items == 0
            return compact(index)
        inner := indent + "  "
        string[] parts := []
        for: item := value.items[i]
            parts = parts.append(inner + (if value.kind == "object" then quote_text(value.keys[i]) + ": " else "") + pretty(item, inner))
        return (if value.kind == "object" then "{{" else "[") + NL + parts.join("," + NL) + NL + indent + (if value.kind == "object" then "}}" else "]")
    Match child(Match parent, nat position)
        value := nodes[parent.node]
        step := Step(value.kind == "object", if value.kind == "object" then value.keys[position] else "", position)
        return Match(value.items[position], parent.steps.append(step))
    int child_position(nat index, Segment selector)
        value := nodes[index]
        if selector.kind == "key"
            return if value.kind == "object" then key_index(index, selector.key) else -1
    Match[] select(Segment selector, Match[] matches)
        Match[] result := []
        for parent := matches[]
            Match[] sources := if selector.descendant then preorder(parent) else [parent]
            for source := sources[]
                value := nodes[source.node]
                if selector.kind == "all"
                    for: i < #value.items
                        result = result.append(child(source, i))
                elif selector.kind == "slice"
                else
                    int position := child_position(source.node, selector)
                    if position >= 0
                        result = result.append(child(source, nat(position)))
        return result
    Match[] evaluate(Segment[] selectors)
        result := [Match(root, [])]
        for selector := selectors[]
            result = select(selector, result)
        return result
    nat resolve(Step[] steps)
        current := root
        for step := steps[]
            current = nodes[current].items[step.position]
        return current
class JsonReader(string text)
    nat position := 0
    tree := Tree([], 0)
    reason := ""
    nat error_at := 0
    string current()
        return if position < #text then text[position] else ""
    error .bad(string message, nat at)
        reason = message
        error_at = at
        return error(message)
    void .skip()
        while current() == " " or current() == NL
            position += 1
    void | error .expect(string ch, string message)
        if current() == ""
            return .bad("unexpected end of input", position)
        if current() != ch
            return .bad(message, position)
    string | error .read_string()
        position += 1
        result := ""
        while true
            string ch := current()
            if ch == "" or ch == NL
                return .bad("unterminated string", position)
            position += 1
            if ch == DQ
                return result
            if ch != "\"
                result += ch
                continue
    nat | error .read_value()
        .skip()
        string ch := current()
        if ch == ""
            return .bad("unexpected end of input", position)
        if "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ".contains(ch)
            start := position
            while current() != "" and "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ".contains(current())
                position += 1
            word := text[start:<position]
            if word == "null" or word == "true" or word == "false"
                return .tree.add(Value(if word == "null" then "null" else "boolean", word))
            return .bad("invalid literal '{word}'", start)
        if ch != "{{" and ch != "["
            return .bad("unexpected character '{ch}'", position)
        object := ch == "{{"
        closing := if object then "}}" else "]"
        value := Value(if object then "object" else "array")
        position += 1
        .skip()
        if current() == closing
            position += 1
            return .tree.add(value)
        while true
            if object
                .skip()
                try .expect(DQ, "expected key")
                start := position
                string key := try .read_string()
                if one_of(value.keys, key)
                    return .bad("duplicate key " + quote_text(key), start)
                .skip()
                try .expect(":", "expected ':'")
                position += 1
                value.keys = value.keys.append(key)
            nat item := try .read_value()
            value.items = value.items.append(item)
            .skip()
            if current() == closing
                position += 1
                return .tree.add(value)
            try .expect(",", if object then "expected ',' or '}}'" else "expected ',' or ']'")
            position += 1
            .skip()
            if current() == closing
                return .bad("trailing comma", position)
    Tree | error .parse()
        nat index := try .read_value()
        .skip()
        if position < #text
            return .bad("trailing characters", position)
        tree.root = index
        return tree
class PathReader(string text, string message)
    nat position := 0
    string current()
        return if position < #text then text[position] else ""
    bool .accept(string ch)
        if current() != ch
            return false
        position += 1
        return true
    void | error .expect(string ch)
        if not .accept(ch)
            return error(message)
        return void
    string | error .read_identifier()
        start := position
        if current() == "" or not "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ_".contains(current())
            return error(message)
        while current() != "" and "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ_0123456789".contains(current())
            position += 1
        return text[start:<position]
    Segment | error .selection()
        if current() == "'"
        start := position
        while current() != "" and current() != "]"
            position += 1
        content := text[start:<position]
        try .expect("]")
        if content == "*"
            return Segment("all")
    Segment[] | error .parse(field := false)
        try .expect(if field then "@" else "$")
        Segment[] result := []
        while position < #text
            selector := Segment("key")
            if .accept("[")
                selector = try .selection()
                if field and selector.kind != "key" and selector.kind != "index"
                    return error(message)
            else
                try .expect(".")
                descendant := not field and .accept(".")
                if not field and .accept("*")
                    selector = Segment("all")
                elif descendant and .accept("[")
                    selector = try .selection()
                else
                    string key := try .read_identifier()
                    selector = Segment("key", key)
                selector.descendant = descendant
            result = result.append(selector)
        return result
Segment[] | error parse_path(string text, field := false)
    reader := PathReader(text, if field then "bad field" else "bad path")
    return .reader.parse(field)
string | error parse_key(string text)
    reader := PathReader(text, "bad key")
    result := if reader.current() == "'" then try .reader.read_key() else try .reader.read_identifier()
    return if reader.position != #text then error("bad key") else result
string location(Match selected)
    result := "$"
    for step := selected.steps[]
        result += if not step.key_step then "[{step.position}]" elif identifier(step.key) then "." + step.key else "[" + quote_text(step.key, "'") + "]"
    return result
bool earlier(Match first, Match second)
    n := if #first.steps < #second.steps then #first.steps else #second.steps
    for: i < n
        if first.steps[i].position != second.steps[i].position
            return first.steps[i].position < second.steps[i].position
    return #first.steps < #second.steps
Match[] targets(Match[] matches)
    Match[] result := []
    for selected := matches[]
        present := false
        at := #result
        for: other := result[i]
            if not earlier(selected, other) and not earlier(other, selected)
                present = true
                break
            if earlier(other, selected)
                at = i
                break
        if not present
            result = result[:<at].append(selected).concat(result[at:])
    return result
string[] | error words(string argument, bool present, nat minimum, nat maximum)
    string[] result := if present then argument.split(" ") else []
    return if #result < minimum or #result > maximum then error("bad arguments") else result
class QueryTool()
    document := Tree([Value("null", "null")], 0)
    Tree[] history := []
    void .commit(Tree next)
        history = history.append(document)
        document = next
    string .load(string text)
        reader := JsonReader(text)
        match .reader.parse()
            Tree next
                .commit(next)
                return "LOADED nodes={document.count_nodes(document.root)} depth={document.depth(document.root)}"
    string | error .execute(string line)
        space := find_at(line, " ")
        command := line[:<space]
        present := space < #line
        argument := if present then line[space + 1:] else ""
        if not one_of(["GET", "COMPACT", "PRETTY", "TYPE", "LENGTH", "KEYS", "FLATTEN", "COUNT", "SUM", "MIN", "MAX", "AVG", "FILTER", "SET", "DELETE", "INSERT", "MERGE", "SORT", "RENAME", "UNDO"], command)
            return error("unknown command '{command}'")
        if command == "UNDO"
            args := try words(argument, present, 0, 0)
            if #history == 0
                return error("nothing to undo")
            document = history[-1]
            history = history[:<#history - 1]
            return "UNDO {#history}"
        Segment[] path := []
        new_key := ""
        if command == "SET" or command == "MERGE"
        elif command == "INSERT"
        elif command == "FILTER"
        elif command == "SORT"
        elif command == "RENAME"
            args := try words(argument, present, 2, 2)
            path = try parse_path(args[0])
            new_key = try parse_key(args[1])
        else
            optional := one_of(["COMPACT", "PRETTY", "FLATTEN"], command)
            args := try words(argument, present, if optional then 0 else 1, 1)
            path = try parse_path(if #args == 0 then "$" else args[0])
        Match[] matches := document.evaluate(path)
        for selected := matches[]
            if command == "RENAME" and (#selected.steps == 0 or not selected.steps[-1].key_step)
                return error("not a member at {location(selected)}")
        if one_of(["SET", "DELETE", "INSERT", "MERGE", "SORT", "RENAME"], command)
            chosen := matches
            chosen = targets(chosen)
            Tree next := document
            for selected := chosen[]
                container := if #selected.steps == 0 then next.root else next.resolve(selected.steps[:<#selected.steps - 1])
                position := if #selected.steps == 0 then 0 else selected.steps[-1].position
                if command == "SET"
                elif command == "DELETE"
                elif command == "RENAME"
                    if next.nodes[container].keys[position] != new_key and next.key_index(container, new_key) >= 0
                        return error("key exists at {location(selected)}")
                    next.nodes[container].keys[position] = new_key
            .commit(next)
            return "{command} {#chosen}"
        string[] output := []
        for selected := matches[]
            if command == "GET"
            elif command == "COMPACT" or command == "PRETTY"
                output = output.append(if command == "COMPACT" then document.compact(selected.node) else document.pretty(selected.node))
        return if #output == 0 then "no match" else output.join(NL)
tool := QueryTool()
