145 func spelling(_ system: String, _ id: String) -> String {
146 let hepburn = system == "HEPBURN" || system == "TRADITIONAL"
147 let wapuro = system == "WAPURO"
148 if hepburn || wapuro {
149 let result: String
150 switch id {
151 case "si": result = "shi"
152 case "ti": result = "chi"
153 case "tu": result = "tsu"
154 case "hu": result = "fu"
155 case "zi": result = "ji"
156 case "di": result = hepburn ? "ji" : id
157 case "du": result = hepburn ? "zu" : id
158 case "wi": result = hepburn ? "i" : id
159 case "we": result = hepburn ? "e" : id
160 case "wo": result = hepburn ? "o" : id
161 default: result = id
162 }
163 if id.hasPrefix("sy") {
164 return "sh" + String(id.last!)
165 }
166 if id.hasPrefix("ty") {
167 return "ch" + String(id.last!)
168 }
169 if id.hasPrefix("zy") || (hepburn && id.hasPrefix("dy")) {
170 return "j" + String(id.last!)
171 }
172 return result
173 }
174 if system == "KUNREI" {
175 switch id {
176 case "di": return "zi"
177 case "du": return "zu"
178 case "wi": return "i"
179 case "we": return "e"
180 case "wo": return "o"
181 default: return id.hasPrefix("dy") ? "z" + slice(id, 1) : id
182 }
183 }
184 return id
185 }

187 func match(_ text: String, _ position: Int, _ system: String) -> SyllableMatch? {
188 let maximum = min(3, text.count - position)
189 if maximum < 1 {
190 return nil
191 }
192 for length in stride(from: maximum, through: 1, by: -1) {
193 if let id = readers[system]![slice(text, position, position + length)] {
194 return SyllableMatch(id: id, length: length)
195 }
196 }
197 return nil
198 }

200 func parse(_ word: Word, _ system: String) throws -> [Mora] {
201 let text = word.text.lowercased()
202 let characters = Array(text)
203 let normalize = system == "ANY"
204 var result: [Mora] = []
205 var position = 0
206 while position < characters.count {
207 let c = characters[position]
208 let next: Character = position + 1 < characters.count ? characters[position + 1] : "\0"
209 let at = word.start + position + 1
210 let id: String
211 let advance: Int
212 if c == "-" {
213 guard let previous = result.last, regular(previous.id) else {
214 throw column(at, "misplaced long mark")
215 }
216 id = "L"
217 advance = 1
218 } else if c == "'" {
219 throw column(at, "misplaced apostrophe")
220 } else if let syllable = match(text, position, system) {
221 id = syllable.id
222 advance = syllable.length
223 } else if (normalize || system == "WAPURO") && slice(text, position).hasPrefix("xtu") {
224 id = "Q"
225 advance = 3
226 } else if c == "n" {
227 id = "N"
228 if normalize {
229 let third: Character = position + 2 < characters.count ? characters[position + 2] : "\0"
230 advance = next == "'" || (next == "n" && !"aiueoy'".contains(third)) ? 2 : 1
231 } else if system == "WAPURO" {
232 advance = next == "n" ? 2 : 1
233 } else {
234 advance = next == "'" ? 2 : 1
235 }
236 } else if (normalize || system == "TRADITIONAL") && c == "m" && "bmp".contains(next) {
237 id = "N"
238 advance = 1
239 } else if letter(c) && !"aiueon".contains(c) && c == next && match(text, position + 1, system) != nil {
240 id = "Q"
241 advance = 1
242 } else if (normalize || system == "HEPBURN" || system == "TRADITIONAL") && c == "t" &&
243 slice(text, position + 1).hasPrefix("ch") && match(text, position + 1, system) != nil {
244 id = "Q"
245 advance = 1
246 } else {
247 throw column(at, "unknown syllable")
248 }
249 result.append(Mora(id: id, column: at))
250 position += advance
251 }
252 return result
253 }

255 func scan(_ text: String) -> [Word] {
256 text.matches(of: #/[A-Za-z'-]+/#).map {
257 Word(text: String($0.output), start: text.distance(from: text.startIndex, to: $0.range.lowerBound))
258 }
259 }

261 func pattern(_ text: String) -> Int {
262 let letters = text.filter { letter($0) }
263 if letters.allSatisfy({ $0 >= "a" && $0 <= "z" }) {
264 return 0
265 }
266 if text.first! >= "A" && text.first! <= "Z" &&
267 letters.dropFirst().allSatisfy({ $0 >= "a" && $0 <= "z" }) {
268 return 1
269 }
270 if letters.count >= 2 && letters.allSatisfy({ $0 >= "A" && $0 <= "Z" }) {
271 return 2
272 }
273 return -1
274 }

276 func longInfo(_ morae: [Mora]) -> LongInfo {
277 var info = LongInfo(
278 paired: Array(repeating: false, count: morae.count),
279 elements: Array(repeating: false, count: morae.count),
280 vowels: Array(repeating: "", count: morae.count)
281 )
282 for index in morae.indices {
283 if morae[index].id == "L" {
284 info.elements[index] = true
285 info.vowels[index] = String(morae[index - 1].id.last!)
286 }
287 }
288 var index = 0
289 while index + 1 < morae.count {
290 let first = morae[index].id
291 let second = morae[index + 1].id
292 let v = String(first.last!)
293 if regular(first) && vowel(second) && (v == second || (v == "o" && second == "u") || (v == "e" && second == "i")) {
294 info.paired[index + 1] = true
295 if (v == "o" && (second == "o" || second == "u")) || (v == "u" && second == "u") {
296 info.elements[index + 1] = true
297 info.vowels[index + 1] = v
298 }
299 index += 2
300 } else {
301 index += 1
302 }
303 }
304 return info
305 }

307 func parts(_ word: Word, _ system: String, _ longMode: String) throws -> [String] {
308 let info = longInfo(word.morae)
309 var result: [String] = []
310 for index in word.morae.indices {
311 let mora = word.morae[index]
312 let id = mora.id
313 let nextId = index + 1 < word.morae.count ? word.morae[index + 1].id : ""
314 let next = regular(nextId) ? spelling(system, nextId) : ""
315 let part: String
316 if info.elements[index] {
317 let v = info.vowels[index]
318 switch longMode {
319 case "KEEP": part = id == "L" ? "-" : spelling(system, id)
320 case "DROP": part = ""
321 case "PASSPORT": part = v == "o" && !begins(next, "aiueo") ? "h" : ""
322 case "HYPHEN": part = "-"
323 default: part = v
324 }
325 } else if regular(id) {
326 part = spelling(system, id)
327 } else if id == "N" {
328 if system == "WAPURO" {
329 part = nextId == "N" || begins(next, "aiueoyn") ? "nn" : "n"
330 } else if system == "TRADITIONAL" && begins(next, "bmp") {
331 part = "m"
332 } else {
333 part = begins(next, "aiueoy") ? "n'" : "n"
334 }
335 } else if !next.isEmpty && !begins(next, "aiueon") {
336 part = (system == "HEPBURN" || system == "TRADITIONAL") && next.hasPrefix("ch") ? "t" : String(next.first!)
337 } else if system == "WAPURO" {
338 part = "xtu"
339 } else {
340 throw column(mora.column, "small tsu cannot be written")
341 }
342 result.append(part)
343 }
344 return result
345 }

347 func write(_ word: Word, _ system: String, _ longMode: String, _ caseMode: String) throws -> String {
348 var result = try parts(word, system, longMode).joined()
349 let outputPattern = caseMode == "KEEP" ? word.pattern : (caseMode == "LOWER" ? 0 : (caseMode == "TITLE" ? 1 : 2))
350 if outputPattern == 1 {
351 result = String(result.first!).uppercased() + slice(result, 1)
352 }
353 if outputPattern == 2 {
354 result = result.uppercased()
355 }
356 return result
357 }

372 func elements(_ word: Word) -> [Element] {
373 var result: [Element] = []
374 for mora in word.morae {
375 let id = mora.id
376 if id == "L" {
377 continue
378 }
379 if id == "N" {
380 result.append(Element(base: "N", voice: 0, size: 0))
381 } else if id == "Q" {
382 result.append(Element(base: "tu", voice: 0, size: 1))
383 } else {
384 let meta = metadata[id]!
385 if meta.contracted {
386 result.append(Element(base: devoice(String(meta.row.first!) + "i"), voice: voice(meta.row), size: 0))
387 result.append(Element(base: "y" + String(id.last!), voice: 0, size: 1))
388 } else {
389 result.append(Element(base: devoice(id), voice: voice(meta.row), size: 0))
390 }
391 }
392 }
393 return result

396 func sorted(_ words: [Word]) -> [Word] {
397 let bases = (orderSetting == "GOJUON" ? gojuon : iroha).split(separator: " ").map(String.init)
398 let ranks = Dictionary(uniqueKeysWithValues: bases.enumerated().map { ($0.element, $0.offset) })
399 func key(_ word: Word) -> [[Int]] {
400 let collation = elements(word)
401 return [
402 collation.map { ranks[$0.base]! },
403 collation.map { $0.voice },
404 collation.map { $0.size },
405 word.morae.map { $0.id == "L" ? 1 : 0 }
406 ]
407 }
408 return words.map { (word: $0, key: key($0)) }.sorted {
409 $0.key.lexicographicallyPrecedes($1.key) { $0.lexicographicallyPrecedes($1) }
410 }.map { $0.word }
411 }

413 func patternMorae(_ pattern: String, _ system: String) throws -> [String] {
414 let characters = Array(pattern)
415 guard !characters.isEmpty && characters.allSatisfy({ ($0 >= "a" && $0 <= "z") || "'-?*".contains($0) }) else {
416 throw Problem.message("invalid pattern")
417 }
418 var result: [String] = []
419 var start = 0
420 do {
421 for index in 0...characters.count {
422 if index == characters.count || characters[index] == "?" || characters[index] == "*" {
423 if index > start {
424 let chunk = Word(text: String(characters[start..<index]), start: 0)
425 result.append(contentsOf: try parse(chunk, system).map { $0.id })
426 }
427 if index < characters.count {
428 result.append(String(characters[index]))
429 }
430 start = index + 1
431 }
432 }
433 } catch {
434 throw Problem.message("invalid pattern")
435 }
436 return result
437 }

439 func matches(_ pattern: [String], _ word: [String]) -> Bool {
440 var dp = Array(repeating: false, count: word.count + 1)
441 dp[0] = true
442 for token in pattern {
443 var next = Array(repeating: false, count: word.count + 1)
444 next[0] = token == "*" && dp[0]
445 if !word.isEmpty {
446 for index in 1...word.count {
447 next[index] = token == "*" ? dp[index] || next[index - 1] :
448 dp[index - 1] && (token == "?" || token == word[index - 1])
449 }
450 }
451 dp = next
452 }
453 return dp[word.count]
454 }

456 func detect(_ words: [Word]) -> String {
457 var result: [String] = []
458 for system in systems {
459 var verdict = "canonical"
460 do {
461 let parsed = try words.map { original in
462 var word = original
463 word.morae = try parse(word, system)
464 return word
465 }
466 for word in parsed {
467 let expected = try canonical(word, system)
468 if word.text != expected {
469 verdict = "readable"
470 }
471 }
472 } catch {
473 verdict = "error"
474 }
475 result.append(system + "=" + verdict)
476 }
477 return result.joined(separator: " ")
478 }

480 func execute(_ line: String) throws -> String {
481 let command = String(line.split(separator: " ", maxSplits: 1)[0])
482 guard counts[command] != nil || command == "SET" || command == "TABLE" else {
483 throw Problem.message("unknown command " + command)
484 }
485 if command == "SET" || command == "TABLE" {
486 let tokens = line.split(separator: " ", omittingEmptySubsequences: false).map(String.init)
487 if tokens.count < 3 {
488 throw Problem.message("missing argument")
489 }
490 if tokens.count > 3 {
491 throw Problem.message("too many arguments")
492 }
493 if command == "SET" {
494 let allowed: [String]
495 switch tokens[1] {
496 case "LONG": allowed = ["KEEP", "DROP", "PASSPORT", "HYPHEN", "DOUBLE"]
497 case "CASE": allowed = ["KEEP", "LOWER", "UPPER", "TITLE"]
498 case "ORDER": allowed = ["GOJUON", "IROHA"]
499 default: throw Problem.message("unknown option " + tokens[1])
500 }
501 guard allowed.contains(tokens[2]) else {
502 throw Problem.message("unknown value " + tokens[2])
503 }
504 if tokens[1] == "LONG" {
505 longSetting = tokens[2]
506 }
507 if tokens[1] == "CASE" {
508 caseSetting = tokens[2]
509 }
510 if tokens[1] == "ORDER" {
511 orderSetting = tokens[2]
512 }
513 return tokens[1] + "=" + tokens[2]
514 }
515 guard systems.contains(tokens[1]) else {
516 throw Problem.message("unknown system " + tokens[1])
517 }
518 guard rows.contains(tokens[2]) else {
519 throw Problem.message("unknown row " + tokens[2])
520 }
521 return table.filter { $0.row == tokens[2] }.map { spelling(tokens[1], $0.id) }.joined(separator: " ")
522 }
523 let fixed = counts[command]!
524 let tokens = line.split(separator: " ", maxSplits: fixed + 1, omittingEmptySubsequences: false).map(String.init)
525 guard tokens.count >= fixed + 2 else {
526 throw Problem.message("missing argument")
527 }
528 let arguments = Array(tokens[1..<(fixed + 1)])
529 let text = tokens[fixed + 1]
530 let systemCount = command == "NORMALIZE" || command == "DETECT" ? 0 :
531 (["CONVERT", "DIFF", "SAME"].contains(command) ? 2 : 1)
532 for index in 0..<systemCount {
533 guard systems.contains(arguments[index]) else {
534 throw Problem.message("unknown system " + arguments[index])
535 }
536 }
537 let searchPattern: [String]
538 if command == "FIND" {
539 searchPattern = try patternMorae(arguments[1], arguments[0])
540 } else {
541 searchPattern = []
542 }
543 let slash = command == "SAME" || command == "HAIKU"
544 for (index, c) in text.enumerated() {
545 if !letter(c) && c != "'" && c != "-" && !" 0123456789.,!?;:()\"".contains(c) && !(slash && c == "/") {
546 throw column(index + 1, "invalid character")
547 }
548 }
549 let poem = text.split(separator: "/", omittingEmptySubsequences: false).map(String.init)
550 if command == "SAME" && (poem.count != 2 || scan(poem[0]).isEmpty || scan(poem[1]).isEmpty) {
551 throw Problem.message("expected two texts")
552 }
553 if command == "HAIKU" {
554 guard (poem.count == 3 || poem.count == 5) && poem.allSatisfy({ !scan($0).isEmpty }) else {
555 throw Problem.message("expected 3 or 5 lines")
556 }
557 }
558 var words = scan(text)
559 guard !words.isEmpty else {
560 throw Problem.message("no words")
561 }
562 for index in words.indices {
563 let casePattern = pattern(words[index].text)
564 if casePattern < 0 {
565 throw column(words[index].start + 1, "mixed case")
566 }
567 words[index].pattern = casePattern
568 }
569 if command == "DETECT" {
570 return detect(words)
571 }
572 let slashPosition = Array(text).firstIndex(of: "/") ?? -1
573 for index in words.indices {
574 let source = command == "NORMALIZE" ? "ANY" :
575 (command == "SAME" && words[index].start > slashPosition ? arguments[1] : arguments[0])
576 words[index].morae = try parse(words[index], source)
577 }
578 let system = command == "NORMALIZE" ? "HEPBURN" : arguments[0]
579 switch command {
580 case "CONVERT", "NORMALIZE", "REVERSE":
581 let target = command == "CONVERT" ? arguments[1] : system
582 var result = ""
583 var cursor = 0
584 for word in words {
585 result += slice(text, cursor, word.start)
586 var written = word
587 if command == "REVERSE" {
588 var reversed: [Mora] = []
589 for index in word.morae.indices {
590 let mora = word.morae[index]
591 let id = mora.id == "L" ? String(word.morae[index - 1].id.last!) : mora.id
592 reversed.append(Mora(id: id, column: mora.column))
593 }
594 written.morae = Array(reversed.reversed())
595 }
596 result += try write(written, target, longSetting, caseSetting)
597 cursor = word.start + word.text.count
598 }
599 return result + slice(text, cursor)
600 case "VALID":
601 for word in words {
602 let expected = try canonical(word, system)
603 if word.text != expected {
604 let actualCharacters = Array(word.text)
605 let expectedCharacters = Array(expected)
606 var index = 0
607 while index < min(actualCharacters.count, expectedCharacters.count) && actualCharacters[index] == expectedCharacters[index] {
608 index += 1
609 }
610 return "INVALID col \(word.start + index + 1): expected \(expected)"
611 }
612 }
613 return "VALID"
614 case "DIFF":
615 var changes: [String] = []
616 for word in words {
617 let first = try write(word, system, longSetting, caseSetting)
618 let second = try write(word, arguments[1], longSetting, caseSetting)
619 if first != second {
620 changes.append(first + ">" + second)
621 }
622 }
623 return "\(changes.count) of \(words.count)" + (changes.isEmpty ? "" : ": " + changes.joined(separator: " "))
624 case "SAME":
625 let left = words.filter { $0.start < slashPosition }
626 let right = words.filter { $0.start > slashPosition }
627 var exact = true
628 for index in 0..<min(left.count, right.count) {
629 if loose(left[index]) != loose(right[index]) {
630 return "DIFFERENT at word \(index + 1)"
631 }
632 if ids(left[index]) != ids(right[index]) {
633 exact = false
634 }
635 }
636 if left.count != right.count {
637 return "DIFFERENT at word \(min(left.count, right.count) + 1)"
638 }
639 return exact ? "EXACT" : "LOOSE"
640 case "MORAE":
641 var result: [String] = []
642 var count = 0
643 for word in words {
644 let pieces = word.morae.map { mora in
645 regular(mora.id) ? spelling(system, mora.id) : (mora.id == "L" ? "-" : mora.id)
646 }
647 count += pieces.count
648 result.append(pieces.joined(separator: "."))
649 }
650 return "\(count): " + result.joined(separator: " ")
651 case "SYLLABLES":
652 var result: [String] = []
653 var count = 0
654 for word in words {
655 let pieces = try parts(word, system, "KEEP")
656 let info = longInfo(word.morae)
657 var syllables: [String] = []
658 for index in pieces.indices {
659 if index == 0 || (regular(word.morae[index].id) && !info.paired[index]) {
660 syllables.append(pieces[index])
661 } else {
662 syllables[syllables.count - 1] += pieces[index]
663 }
664 }
665 count += syllables.count
666 result.append(syllables.joined(separator: "."))
667 }
668 return "\(count): " + result.joined(separator: " ")
669 case "ROW":
670 var result: [String] = []
671 for word in words {
672 for mora in word.morae {
673 if let meta = metadata[mora.id] {
674 result.append(spelling(system, meta.id) + ": " + meta.row + " " + meta.dan)
675 } else {
676 result.append((mora.id == "L" ? "-" : mora.id) + ": special")
677 }
678 }
679 }
680 return result.joined(separator: ", ")
681 case "STATS":
682 let morae = words.flatMap { $0.morae }
683 let regular = morae.compactMap { metadata[$0.id] }
684 let rowCounts = Dictionary(grouping: regular, by: { baseRow($0.row) }).mapValues { $0.count }
685 var result = ["morae=\(morae.count)"]
686 result += rows.prefix(10).compactMap { row in rowCounts[row].map { row + "=\($0)" } }
687 result += [
688 "voiced=\(regular.filter { voice($0.row) == 1 }.count)",
689 "semivoiced=\(regular.filter { voice($0.row) == 2 }.count)",
690 "contracted=\(regular.filter { $0.contracted }.count)",
691 "N=\(morae.filter { $0.id == "N" }.count)",
692 "Q=\(morae.filter { $0.id == "Q" }.count)",
693 "L=\(morae.filter { $0.id == "L" }.count)"
694 ]
695 return result.joined(separator: " ")
696 case "SORT", "INDEX":
697 let ordered = sorted(words)
698 if command == "SORT" {
699 return ordered.map { $0.text }.joined(separator: " ")
700 }
701 var groups: [String] = []
702 var group: [String] = []
703 var previous = ""
704 for word in ordered {
705 let base = elements(word)[0].base
706 let heading = orderSetting == "IROHA" || base == "N" ? base : metadata[base]!.row
707 if heading != previous && !group.isEmpty {
708 groups.append(previous + ": " + group.joined(separator: " "))
709 group.removeAll()
710 }
711 previous = heading
712 group.append(word.text)
713 }
714 groups.append(previous + ": " + group.joined(separator: " "))
715 return groups.joined(separator: "; ")
716 case "FIND":
717 let found = words.filter { matches(searchPattern, ids($0)) }.map { $0.text }
718 return "\(found.count) of \(words.count)" + (found.isEmpty ? "" : ": " + found.joined(separator: " "))
719 case "HAIKU":
720 var totals = Array(repeating: 0, count: poem.count)
721 for word in words {
722 let index = slice(text, 0, word.start).filter { $0 == "/" }.count
723 totals[index] += word.morae.count
724 }
725 let expected = poem.count == 3 ? [5, 7, 5] : [5, 7, 5, 7, 7]
726 return (totals == expected ? "OK " : "NG ") + (poem.count == 3 ? "haiku " : "tanka ") +
727 totals.map(String.init).joined(separator: "-")
728 case "KAIBUN":
729 let bases = words.flatMap { elements($0).map { $0.base } }
730 return (bases == Array(bases.reversed()) ? "YES " : "NO ") + bases.joined(separator: ".")
731 case "SHIRITORI":
732 var seen: Set<String> = []
733 var previous = ""
734 for (index, word) in words.enumerated() {
735 let collation = elements(word)
736 let head = collation.first!.base
737 let tail = collation.last!.base
738 let prefix = "NG \(index + 1): " + word.text
739 if index > 0 && head != previous {
740 return prefix + " must start with " + spelling(system, previous)
741 }
742 let key = ids(word).joined(separator: ".")
743 if !seen.insert(key).inserted {
744 return prefix + " is repeated"
745 }
746 if tail == "N" {
747 return prefix + " ends with N"
748 }
749 previous = tail
750 }
751 return "OK \(words.count)"
752 default:
753 preconditionFailure("Unreachable command")
754 }
755 }
756}
