68 private fun spelling(system: String, id: String): String {
69 val hepburn = system in setOf("HEPBURN","TRADITIONAL")
70 if (hepburn || system == "WAPURO") {
71 val result = when (id) {
72 "si" -> "shi"
73 "ti" -> "chi"
74 "tu" -> "tsu"
75 "hu" -> "fu"
76 "zi" -> "ji"
77 "di" -> if (hepburn) "ji" else id
78 "du" -> if (hepburn) "zu" else id
79 "wi" -> if (hepburn) "i" else id
80 "we" -> if (hepburn) "e" else id
81 "wo" -> if (hepburn) "o" else id
82 else -> id
83 }
84 if (id.startsWith("sy")) {
85 return "sh"+id.last()
86 }
87 if (id.startsWith("ty")) {
88 return "ch"+id.last()
89 }
90 if (id.startsWith("zy") || hepburn && id.startsWith("dy")) {
91 return "j"+id.last()
92 }
93 return result
94 }
95 if (system == "KUNREI") {
96 return when (id) {
97 "di" -> "zi"
98 "du" -> "zu"
99 "wi" -> "i"
100 "we" -> "e"
101 "wo" -> "o"
102 else -> if (id.startsWith("dy")) "z"+id.substring(1) else id
103 }
104 }
105 return id

116 private fun parse(word: Word, system: String): List<Mora> {
117 val text = word.text.lowercase()
118 val result = mutableListOf<Mora>()
119 val normalize = system == "ANY"
120 var p = 0
121 while (p < text.length) {
122 val c = text[p]
123 val next = text.getOrElse(p+1) { '\u0000' }
124 val at = word.start+p+1
125 val id: String
126 val advance: Int
127 if (c == '-') {
128 if (result.isEmpty() || !regular(result.last().id)) {
129 column(at,"misplaced long mark")
130 }
131 id = "L"
132 advance = 1
133 } else if (c == '\'') {
134 column(at,"misplaced apostrophe")
135 } else {
136 val syllable = match(text,p,system)
137 if (syllable != null) {
138 id = syllable.id
139 advance = syllable.length
140 } else if ((normalize || system == "WAPURO") && text.startsWith("xtu",p)) {
141 id = "Q"
142 advance = 3
143 } else if (c == 'n') {
144 id = "N"
145 advance = if (normalize) {
146 val third = text.getOrElse(p+2) { '\u0000' }
147 if (next == '\'' || next == 'n' && third !in "aiueoy'") 2 else 1
148 } else if (system == "WAPURO") {
149 if (next == 'n') 2 else 1
150 } else if (next == '\'') 2 else 1
151 } else if ((normalize || system == "TRADITIONAL") && c == 'm' && next in "bmp") {
152 id = "N"
153 advance = 1
154 } else if (letter(c) && c !in "aiueon" && c == next && match(text,p+1,system) != null) {
155 id = "Q"
156 advance = 1
157 } else if ((normalize || system in setOf("HEPBURN","TRADITIONAL")) &&
158 c == 't' && text.startsWith("ch",p+1) && match(text,p+1,system) != null) {
159 id = "Q"
160 advance = 1
161 } else {
162 column(at,"unknown syllable")
163 }
164 }
165 result.add(Mora(id,at))
166 p += advance
167 }
168 return result
169 }
170 private fun scan(text: String): List<Word> = Regex("[A-Za-z'-]+").findAll(text)
171 .map { Word(it.value,it.range.first) }.toList()
172 private fun pattern(text: String): Int {
173 val letters = text.filter(::letter)
174 if (letters.all { it in 'a'..'z' }) {
175 return 0
176 }
177 if (text[0] in 'A'..'Z' && letters.drop(1).all { it in 'a'..'z' }) {
178 return 1
179 }
180 if (letters.length >= 2 && letters.all { it in 'A'..'Z' }) {
181 return 2
182 }
183 return -1
184 }
185 private fun longInfo(morae: List<Mora>): LongInfo {
186 val paired = BooleanArray(morae.size)
187 val elements = BooleanArray(morae.size)
188 val vowels = CharArray(morae.size)
189 for (i in morae.indices) {
190 if (morae[i].id == "L") {
191 elements[i] = true
192 vowels[i] = morae[i-1].id.last()
193 }
194 }
195 var p = 0
196 while (p+1 < morae.size) {
197 val a = morae[p].id
198 val b = morae[p+1].id
199 val v = a.last()
200 if (regular(a) && vowel(b) && (v == b.last() || v == 'o' && b == "u" || v == 'e' && b == "i")) {
201 paired[p+1] = true
202 if (v == 'o' && b in setOf("o","u") || v == 'u' && b == "u") {
203 elements[p+1] = true
204 vowels[p+1] = v
205 }
206 p += 2
207 } else {
208 p++
209 }
210 }
211 return LongInfo(paired,elements,vowels)
212 }
213 private fun parts(word: Word, system: String, longMode: String): List<String> {
214 val info = longInfo(word.morae)
215 val result = mutableListOf<String>()
216 for ((i,mora) in word.morae.withIndex()) {
217 val id = mora.id
218 val nextId = word.morae.getOrNull(i+1)?.id ?: ""
219 val next = if (regular(nextId)) spelling(system,nextId) else ""
220 val part: String
221 if (info.elements[i]) {
222 val v = info.vowels[i]
223 part = when (longMode) {
224 "KEEP" -> if (id == "L") "-" else spelling(system,id)
225 "DROP" -> ""
226 "PASSPORT" -> if (v == 'o' && !begins(next,"aiueo")) "h" else ""
227 "HYPHEN" -> "-"
228 else -> v.toString()
229 }
230 } else if (regular(id)) {
231 part = spelling(system,id)
232 } else if (id == "N") {
233 part = if (system == "WAPURO") {
234 if (nextId == "N" || begins(next,"aiueoyn")) "nn" else "n"
235 } else if (system == "TRADITIONAL" && begins(next,"bmp")) "m"
236 else if (begins(next,"aiueoy")) "n'" else "n"
237 } else {
238 part = if (next.isNotEmpty() && !begins(next,"aiueon")) {
239 if (system in setOf("HEPBURN","TRADITIONAL") && next.startsWith("ch")) "t" else next.take(1)
240 } else if (system == "WAPURO") "xtu"
241 else column(mora.column,"small tsu cannot be written")
242 }
243 result.add(part)
244 }
245 return result
246 }
247 private fun write(word: Word, system: String, longMode: String, caseMode: String): String {
248 var result = parts(word,system,longMode).joinToString("")
249 val pattern = when (caseMode) {
250 "KEEP" -> word.pattern
251 "LOWER" -> 0
252 "TITLE" -> 1
253 else -> 2
254 }
255 if (pattern == 1) {
256 result = result.take(1).uppercase()+result.drop(1)
257 }
258 if (pattern == 2) {
259 result = result.uppercase()
260 }
261 return result
262 }

292 private fun compare(a: List<Int>, b: List<Int>): Int {
293 for (i in 0 until minOf(a.size,b.size)) {
294 if (a[i] != b[i]) {
295 return a[i].compareTo(b[i])
296 }
297 }
298 return a.size.compareTo(b.size)
299 }
300 private fun sorted(words: List<Word>): List<Word> {
301 val ranks = (if (orderSetting == "GOJUON") gojuon else iroha).split(' ').withIndex().associate { it.value to it.index }
302 fun key(word: Word): List<List<Int>> {
303 val collation = elements(word)
304 return listOf(collation.map { ranks.getValue(it.base) },collation.map { it.voice },
305 collation.map { it.size },word.morae.map { if (it.id == "L") 1 else 0 })
306 }
307 return words.map { it to key(it) }.sortedWith { a,b ->
308 val left = a.second
309 val right = b.second
310 var result = 0
311 for (i in 0 until 4) {
312 result = compare(left[i],right[i])
313 if (result != 0) {
314 break
315 }
316 }
317 result
318 }.map { it.first }

320 private fun patternMorae(pattern: String, system: String): List<String> {
321 if (!Regex("[a-z'?*\\-]+").matches(pattern)) {
322 fail("invalid pattern")
323 }
324 val result = mutableListOf<String>()
325 var start = 0
326 try {
327 for (i in 0..pattern.length) {
328 if (i == pattern.length || pattern[i] == '?' || pattern[i] == '*') {
329 if (i > start) {
330 result.addAll(parse(Word(pattern.substring(start,i),0),system).map { it.id })
331 }
332 if (i < pattern.length) {
333 result.add(pattern[i].toString())
334 }
335 start = i+1
336 }
337 }
338 } catch (problem: Problem) {
339 fail("invalid pattern")
340 }
341 return result
342 }
343 private fun matches(pattern: List<String>, word: List<String>): Boolean {
344 var dp = BooleanArray(word.size+1)
345 dp[0] = true
346 for (token in pattern) {
347 val next = BooleanArray(word.size+1)
348 next[0] = token == "*" && dp[0]
349 for (j in 1..word.size) {
350 next[j] = if (token == "*") dp[j] || next[j-1] else dp[j-1] && (token == "?" || token == word[j-1])
351 }
352 dp = next
353 }
354 return dp[word.size]
355 }
356 private fun detect(words: List<Word>): String = systems.joinToString(" ") { system ->
357 val verdict = try {
358 val parsed = words.map { it.copy(morae = parse(it,system)) }
359 if (parsed.all { it.text == canonical(it,system) }) "canonical" else "readable"
360 } catch (problem: Problem) {
361 "error"
362 }
363 "$system=$verdict"
364 }
365 fun execute(line: String): String {
366 val command = line.substringBefore(' ')
367 if (command !in counts && command !in setOf("SET","TABLE")) {
368 fail("unknown command $command")
369 }
370 if (command in setOf("SET","TABLE")) {
371 val tokens = line.split(' ')
372 if (tokens.size < 3) {
373 fail("missing argument")
374 }
375 if (tokens.size > 3) {
376 fail("too many arguments")
377 }
378 if (command == "SET") {
379 val allowed = when (tokens[1]) {
380 "LONG" -> setOf("KEEP","DROP","PASSPORT","HYPHEN","DOUBLE")
381 "CASE" -> setOf("KEEP","LOWER","UPPER","TITLE")
382 "ORDER" -> setOf("GOJUON","IROHA")
383 else -> fail("unknown option ${tokens[1]}")
384 }
385 if (tokens[2] !in allowed) {
386 fail("unknown value ${tokens[2]}")
387 }
388 when (tokens[1]) {
389 "LONG" -> longSetting = tokens[2]
390 "CASE" -> caseSetting = tokens[2]
391 "ORDER" -> orderSetting = tokens[2]
392 }
393 return "${tokens[1]}=${tokens[2]}"
394 }
395 if (tokens[1] !in systems) {
396 fail("unknown system ${tokens[1]}")
397 }
398 if (tokens[2] !in rows) {
399 fail("unknown row ${tokens[2]}")
400 }
401 return metadata.values.filter { it.row == tokens[2] }.joinToString(" ") { spelling(tokens[1],it.id) }
402 }
403 val fixed = counts.getValue(command)
404 val chunks = line.split(' ',limit = fixed+2)
405 if (chunks.size < fixed+2) {
406 fail("missing argument")
407 }
408 val arguments = chunks.subList(1,fixed+1)
409 val text = chunks[fixed+1]
410 val systemCount = if (command in setOf("NORMALIZE","DETECT")) 0 else if (command in setOf("CONVERT","DIFF","SAME")) 2 else 1
411 for (i in 0 until systemCount) {
412 if (arguments[i] !in systems) {
413 fail("unknown system ${arguments[i]}")
414 }
415 }
416 val searchPattern = if (command == "FIND") patternMorae(arguments[1],arguments[0]) else emptyList()
417 val slash = command in setOf("SAME","HAIKU")
418 for ((i,c) in text.withIndex()) {
419 if (!letter(c) && c != '\'' && c != '-' && c !in " 0123456789.,!?;:()\"" && !(slash && c == '/')) {
420 column(i+1,"invalid character")
421 }
422 }
423 val poem = text.split('/')
424 if (command == "SAME" && (poem.size != 2 || scan(poem[0]).isEmpty() || scan(poem[1]).isEmpty())) {
425 fail("expected two texts")
426 }
427 if (command == "HAIKU") {
428 if (poem.size != 3 && poem.size != 5) {
429 fail("expected 3 or 5 lines")
430 }
431 if (poem.any { scan(it).isEmpty() }) {
432 fail("expected 3 or 5 lines")
433 }
434 }
435 var words = scan(text)
436 if (words.isEmpty()) {
437 fail("no words")
438 }
439 words = words.map { word ->
440 val pattern = pattern(word.text)
441 if (pattern < 0) {
442 column(word.start+1,"mixed case")
443 }
444 word.copy(pattern = pattern)
445 }
446 if (command == "DETECT") {
447 return detect(words)
448 }
449 words = words.map { word ->
450 val source = if (command == "NORMALIZE") "ANY" else
451 if (command == "SAME" && word.start > text.indexOf('/')) arguments[1] else arguments[0]
452 word.copy(morae = parse(word,source))
453 }
454 val system = if (command == "NORMALIZE") "HEPBURN" else arguments[0]
455 when (command) {
456 "CONVERT", "NORMALIZE", "REVERSE" -> {
457 val target = if (command == "CONVERT") arguments[1] else system
458 val result = StringBuilder()
459 var cursor = 0
460 for (word in words) {
461 result.append(text.substring(cursor,word.start))
462 val written = if (command == "REVERSE") {
463 val reversed = word.morae.mapIndexed { i,mora ->
464 val id = if (mora.id == "L") word.morae[i-1].id.last().toString() else mora.id
465 Mora(id,mora.column)
466 }.reversed()
467 word.copy(morae = reversed)
468 } else {
469 word
470 }
471 result.append(write(written,target,longSetting,caseSetting))
472 cursor = word.start+word.text.length
473 }
474 result.append(text.substring(cursor))
475 return result.toString()
476 }
477 "VALID" -> {
478 for (word in words) {
479 val expected = canonical(word,system)
480 if (word.text != expected) {
481 var i = 0
482 while (i < minOf(word.text.length,expected.length) && word.text[i] == expected[i]) {
483 i++
484 }
485 return "INVALID col ${word.start+i+1}: expected $expected"
486 }
487 }
488 return "VALID"
489 }
490 "DIFF" -> {
491 val changes = mutableListOf<String>()
492 for (word in words) {
493 val a = write(word,system,longSetting,caseSetting)
494 val b = write(word,arguments[1],longSetting,caseSetting)
495 if (a != b) {
496 changes.add("$a>$b")
497 }
498 }
499 return "${changes.size} of ${words.size}" + if (changes.isEmpty()) "" else ": ${changes.joinToString(" ")}"
500 }
501 "SAME" -> {
502 val left = words.filter { it.start < text.indexOf('/') }
503 val right = words.filter { it.start > text.indexOf('/') }
504 var exact = true
505 for (i in 0 until minOf(left.size,right.size)) {
506 if (loose(left[i]) != loose(right[i])) {
507 return "DIFFERENT at word ${i+1}"
508 }
509 if (ids(left[i]) != ids(right[i])) {
510 exact = false
511 }
512 }
513 if (left.size != right.size) {
514 return "DIFFERENT at word ${minOf(left.size,right.size)+1}"
515 }
516 return if (exact) "EXACT" else "LOOSE"
517 }
518 "MORAE" -> {
519 var count = 0
520 val result = words.map { word ->
521 val pieces = word.morae.map { if (regular(it.id)) spelling(system,it.id) else if (it.id == "L") "-" else it.id }
522 count += pieces.size
523 pieces.joinToString(".")
524 }
525 return "$count: ${result.joinToString(" ")}"
526 }
527 "SYLLABLES" -> {
528 var count = 0
529 val result = words.map { word ->
530 val pieces = parts(word,system,"KEEP")
531 val syllables = mutableListOf<String>()
532 val info = longInfo(word.morae)
533 for (i in pieces.indices) {
534 if (i == 0 || regular(word.morae[i].id) && !info.paired[i]) {
535 syllables.add(pieces[i])
536 } else {
537 syllables[syllables.lastIndex] += pieces[i]
538 }
539 }
540 count += syllables.size
541 syllables.joinToString(".")
542 }
543 return "$count: ${result.joinToString(" ")}"
544 }
545 "ROW" -> return words.flatMap { it.morae }.joinToString(", ") { mora ->
546 val meta = metadata[mora.id]
547 if (meta == null) (if (mora.id == "L") "-" else mora.id)+": special"
548 else "${spelling(system,meta.id)}: ${meta.row} ${meta.dan}"
549 }
550 "STATS" -> {
551 val morae = words.flatMap { it.morae }
552 val regular = morae.mapNotNull { metadata[it.id] }
553 val rowCounts = regular.groupingBy { baseRow(it.row) }.eachCount()
554 val result = mutableListOf("morae=${morae.size}")
555 result += rows.take(10).filter { it in rowCounts }.map { "$it=${rowCounts.getValue(it)}" }
556 result += listOf(
557 "voiced=${regular.count { voice(it.row) == 1 }}",
558 "semivoiced=${regular.count { voice(it.row) == 2 }}",
559 "contracted=${regular.count { it.contracted }}",
560 "N=${morae.count { it.id == "N" }}",
561 "Q=${morae.count { it.id == "Q" }}",
562 "L=${morae.count { it.id == "L" }}"
563 )
564 return result.joinToString(" ")
565 }
566 "SORT", "INDEX" -> {
567 val sorted = sorted(words)
568 if (command == "SORT") {
569 return sorted.joinToString(" ") { it.text }
570 }
571 val groups = mutableListOf<String>()
572 val group = mutableListOf<String>()
573 var previous = ""
574 for (word in sorted) {
575 val base = elements(word).first().base
576 val heading = if (orderSetting == "IROHA" || base == "N") base else metadata.getValue(base).row
577 if (heading != previous && group.isNotEmpty()) {
578 groups.add("$previous: ${group.joinToString(" ")}")
579 group.clear()
580 }
581 previous = heading
582 group.add(word.text)
583 }
584 groups.add("$previous: ${group.joinToString(" ")}")
585 return groups.joinToString("; ")
586 }
587 "FIND" -> {
588 val found = words.filter { matches(searchPattern,ids(it)) }.map { it.text }
589 return "${found.size} of ${words.size}" + if (found.isEmpty()) "" else ": ${found.joinToString(" ")}"
590 }
591 "HAIKU" -> {
592 val totals = IntArray(poem.size)
593 for (word in words) {
594 totals[text.take(word.start).count { it == '/' }] += word.morae.size
595 }
596 val expected = if (poem.size == 3) listOf(5,7,5) else listOf(5,7,5,7,7)
597 return (if (totals.toList() == expected) "OK " else "NG ") +
598 (if (poem.size == 3) "haiku " else "tanka ") + totals.joinToString("-")
599 }
600 "KAIBUN" -> {
601 val bases = words.flatMap { elements(it) }.map { it.base }
602 return (if (bases == bases.reversed()) "YES " else "NO ") + bases.joinToString(".")
603 }
604 "SHIRITORI" -> {
605 val seen = mutableSetOf<List<String>>()
606 var previous = ""
607 for ((i,word) in words.withIndex()) {
608 val collation = elements(word)
609 val head = collation.first().base
610 val tail = collation.last().base
611 val prefix = "NG ${i+1}: ${word.text}"
612 if (i > 0 && head != previous) {
613 return "$prefix must start with ${spelling(system,previous)}"
614 }
615 if (!seen.add(ids(word))) {
616 return "$prefix is repeated"
617 }
618 if (tail == "N") {
619 return "$prefix ends with N"
620 }
621 previous = tail
622 }
623 return "OK ${words.size}"
624 }
625 else -> error("Unreachable command")
626 }
627 }
628}

630fun main() {
631 val input = System.`in`.bufferedReader()
632 val output = System.out.bufferedWriter()
633 val count = input.readLine().toInt()
634 val converter = Converter()
635 repeat(count) {
636 val result = try {
637 converter.execute(input.readLine())
638 } catch (problem: Problem) {
639 "ERROR ${problem.message}"
640 }
641 output.write(result)
642 output.newLine()
643 }
644 output.flush()
645}
