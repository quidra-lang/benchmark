62function spelling(string $system, string $id): string
63{
64 if (in_array($system, ['HEPBURN', 'TRADITIONAL', 'WAPURO'], true)) {
65 $changes = ['si' => 'shi', 'ti' => 'chi', 'tu' => 'tsu', 'hu' => 'fu', 'zi' => 'ji'];
66 if ($system !== 'WAPURO') {
67 $changes += ['di' => 'ji', 'du' => 'zu', 'wi' => 'i', 'we' => 'e', 'wo' => 'o'];
68 }
69 if (isset($changes[$id])) {
70 return $changes[$id];
71 }
72 if (str_starts_with($id, 'sy')) {
73 return 'sh' . substr($id, 2);
74 }
75 if (str_starts_with($id, 'ty')) {
76 return 'ch' . substr($id, 2);
77 }
78 if (str_starts_with($id, 'zy') || ($system !== 'WAPURO' && str_starts_with($id, 'dy'))) {
79 return 'j' . substr($id, 2);
80 }
81 } elseif ($system === 'KUNREI') {
82 $changes = ['di' => 'zi', 'du' => 'zu', 'wi' => 'i', 'we' => 'e', 'wo' => 'o',
83 'dya' => 'zya', 'dyu' => 'zyu', 'dyo' => 'zyo'];
84 return $changes[$id] ?? $id;
85 }
86 return $id;
87}

90foreach ([...SYSTEMS, 'NORMAL'] as $system) {

92 $styles = $system === 'NORMAL' ? ['HEPBURN', 'KUNREI', 'NIHON', 'WAPURO'] : [$system];

104 }
105 usort($entries, fn($a, $b) => strlen($b['text']) <=> strlen($a['text']));
106 $readings[$system] = $entries;
107}

118function parse_word(string $text, int $start, string $system): array
119{
120 $word = strtolower($text);
121 $ids = [];
122 $columns = [];
123 $p = 0;
124 while ($p < strlen($word)) {
125 $c = $word[$p];
126 $c2 = $word[$p + 1] ?? '';
127 $c3 = $word[$p + 2] ?? '';
128 $matched = match_syllable($word, $p, $system);
129 $advance = 1;
130 if ($c === '-') {
131 if ($ids === [] || !is_syllabic($ids[count($ids) - 1])) {
132 column_error($start + $p, 'misplaced long mark');
133 }
134 $id = '-';
135 } elseif ($c === "'") {
136 column_error($start + $p, 'misplaced apostrophe');
137 } elseif ($matched !== null) {
138 $id = $matched['id'];
139 $advance = strlen($matched['text']);
140 } elseif (in_array($system, ['WAPURO', 'NORMAL'], true) && substr($word, $p, 3) === 'xtu') {
141 $id = 'Q';
142 $advance = 3;
143 } elseif ($c === 'n') {
144 $id = 'N';
145 if ($system === 'WAPURO') {
146 $advance = $c2 === 'n' ? 2 : 1;
147 } elseif ($system === 'NORMAL') {
148 if ($c2 === "'" || ($c2 === 'n' && !in_array($c3, ['a', 'i', 'u', 'e', 'o', 'y', "'"], true))) {
149 $advance = 2;
150 }
151 } elseif ($c2 === "'") {
152 $advance = 2;
153 }
154 } elseif (in_array($system, ['TRADITIONAL', 'NORMAL'], true) && $c === 'm' && in_array($c2, ['b', 'm', 'p'], true)) {
155 $id = 'N';
156 } elseif ($c >= 'a' && $c <= 'z' && !in_array($c, ['a', 'i', 'u', 'e', 'o', 'n'], true) && $c === $c2 && match_syllable($word, $p + 1, $system) !== null) {
157 $id = 'Q';
158 } elseif (in_array($system, ['HEPBURN', 'TRADITIONAL', 'NORMAL'], true) && $c === 't' && $c2 === 'c' && $c3 === 'h' && match_syllable($word, $p + 1, $system) !== null) {
159 $id = 'Q';
160 } else {
161 column_error($start + $p, 'unknown syllable');
162 }
163 $ids[] = $id;
164 $columns[] = $start + $p;
165 $p += $advance;
166 }
167 return [$ids, $columns];
168}

170function long_flags(array $ids): array
171{
172 $paired = array_fill(0, count($ids), false);
173 $long = array_map(fn($id) => $id === '-', $ids);
174 $i = 0;
175 while ($i + 1 < count($ids)) {
176 $first = $ids[$i];
177 $second = $ids[$i + 1];
178 if (is_syllabic($first) && is_vowel($second)) {
179 $vowel = substr($first, -1);
180 if ($vowel === $second || ($vowel === 'o' && $second === 'u') || ($vowel === 'e' && $second === 'i')) {
181 $paired[$i + 1] = true;
182 $long[$i + 1] = ($vowel === 'o' && in_array($second, ['o', 'u'], true)) || ($vowel === 'u' && $second === 'u');
183 $i += 2;
184 continue;
185 }
186 }
187 ++$i;
188 }
189 return [$paired, $long];
190}

192function write_parts(array $ids, array $columns, string $system, string $longSetting): array
193{
194 [, $long] = long_flags($ids);
195 $parts = [];
196 foreach ($ids as $i => $id) {
197 $nextId = $ids[$i + 1] ?? '';
198 $nextText = is_syllabic($nextId) ? spelling($system, $nextId) : '';
199 $first = $nextText[0] ?? '';
200 if ($long[$i]) {
201 $vowel = substr($ids[$i - 1], -1);
202 if ($longSetting === 'KEEP') {
203 $part = $id === '-' ? '-' : spelling($system, $id);
204 } elseif ($longSetting === 'DROP') {
205 $part = '';
206 } elseif ($longSetting === 'PASSPORT') {
207 $part = $vowel === 'o' && !is_vowel($first) ? 'h' : '';
208 } elseif ($longSetting === 'HYPHEN') {
209 $part = '-';
210 } else {
211 $part = $vowel;
212 }
213 } elseif (is_syllabic($id)) {
214 $part = spelling($system, $id);
215 } elseif ($id === 'N') {
216 if ($system === 'WAPURO') {
217 $part = $nextId === 'N' || in_array($first, ['a', 'i', 'u', 'e', 'o', 'y', 'n'], true) ? 'nn' : 'n';
218 } elseif ($system === 'TRADITIONAL' && in_array($first, ['b', 'm', 'p'], true)) {
219 $part = 'm';
220 } else {
221 $part = in_array($first, ['a', 'i', 'u', 'e', 'o', 'y'], true) ? "n'" : 'n';
222 }
223 } else {
224 if ($nextText !== '' && !in_array($first, ['a', 'i', 'u', 'e', 'o', 'n'], true)) {
225 $part = in_array($system, ['HEPBURN', 'TRADITIONAL'], true) && str_starts_with($nextText, 'ch') ? 't' : $first;
226 } elseif ($system === 'WAPURO') {
227 $part = 'xtu';
228 } else {
229 column_error($columns[$i], 'small tsu cannot be written');
230 }
231 }
232 $parts[] = $part;
233 }
234 return $parts;
235}

237function word_pattern(string $text): string
238{
239 if ($text === strtolower($text)) {
240 return 'LOWER';
241 }
242 if ($text[0] >= 'A' && $text[0] <= 'Z' && substr($text, 1) === strtolower(substr($text, 1))) {
243 return 'TITLE';
244 }
245 if (preg_match_all('/[A-Z]/', $text) >= 2 && $text === strtoupper($text)) {
246 return 'UPPER';
247 }
248 return '';
249}

251function written(array $word, string $system, array $settings, bool $canonical = false, ?array $ids = null, ?array $columns = null): string
252{
253 $mode = $canonical ? 'KEEP' : $settings['LONG'];
254 $text = implode('', write_parts($ids ?? $word['ids'], $columns ?? $word['columns'], $system, $mode));
255 $pattern = $canonical || $settings['CASE'] === 'KEEP' ? $word['pattern'] : $settings['CASE'];
256 if ($pattern === 'UPPER') {
257 return strtoupper($text);
258 }
259 return $pattern === 'TITLE' ? ucfirst($text) : $text;
260}

262function words_in(string $text): array
263{
264 preg_match_all("/[A-Za-z'-]+/", $text, $matches, PREG_OFFSET_CAPTURE);
265 $words = [];
266 foreach ($matches[0] as [$word, $offset]) {
267 $words[] = ['text' => $word, 'start' => $offset + 1, 'pattern' => '', 'ids' => [], 'columns' => [],
268 'part' => substr_count(substr($text, 0, $offset), '/') + 1, 'index' => count($words)];
269 }
270 return $words;
271}

273function replace_words(string $text, array $words, array $replacements): string
274{
275 $output = '';
276 $position = 0;
277 foreach ($words as $i => $word) {
278 $offset = $word['start'] - 1;
279 $output .= substr($text, $position, $offset - $position) . $replacements[$i];
280 $position = $offset + strlen($word['text']);
281 }
282 return $output . substr($text, $position);
283}

290function elements(array $ids): array
291{
292 global $rowOf, $danOf, $iOf;
293 $items = [];
294 foreach ($ids as $id) {
295 if ($id === '-') {
296 continue;
297 }
298 if ($id === 'N') {
299 $items[] = ['N', 0, 0];
300 } elseif ($id === 'Q') {
301 $items[] = ['tu', 0, 1];
302 } else {
303 $row = $rowOf[$id];
304 $voice = 0;
305 if (in_array($row, ['ga', 'za', 'da', 'ba'], true)) {
306 $voice = 1;
307 } elseif ($row === 'pa') {
308 $voice = 2;
309 }
310 if (strlen($danOf[$id]) === 2) {
311 $items[] = [devoice($iOf[$row]), $voice, 0];
312 $items[] = [$danOf[$id], 0, 1];
313 } else {
314 $items[] = [devoice($id), $voice, 0];
315 }
316 }
317 }
318 return $items;
319}

321function collation(array $word, string $order): array
322{
323 $ranks = array_flip(explode(' ', $order === 'GOJUON' ? GOJUON : IROHA));
324 $items = elements($word['ids']);
325 return [array_map(fn($item) => $ranks[$item[0]], $items), array_column($items, 1), array_column($items, 2),
326 array_map(fn($id) => $id === '-' ? 1 : 0, $word['ids'])];

329function vector_compare(array $a, array $b): int
330{
331 for ($i = 0; $i < min(count($a), count($b)); ++$i) {
332 if ($a[$i] !== $b[$i]) {
333 return $a[$i] <=> $b[$i];
334 }
335 }
336 return count($a) <=> count($b);
337}

339function pattern_tokens(string $pattern, string $system): array
340{
341 if (preg_match("/^[a-z'\\-?*]+$/D", $pattern) !== 1) {
342 fail('invalid pattern');
343 }
344 $tokens = [];
345 $chunk = '';
346 try {
347 foreach (str_split($pattern) as $c) {
348 if ($c === '?' || $c === '*') {
349 $tokens = [...$tokens, ...parse_word($chunk, 1, $system)[0], $c];
350 $chunk = '';
351 } else {
352 $chunk .= $c;
353 }
354 }
355 return [...$tokens, ...parse_word($chunk, 1, $system)[0]];
356 } catch (InvalidArgumentException $exception) {
357 fail('invalid pattern');
358 }
359}

361function wildcard_matches(array $tokens, array $ids): bool
362{
363 $previous = array_fill(0, count($ids) + 1, false);
364 $previous[0] = true;
365 foreach ($tokens as $token) {
366 $current = array_fill(0, count($ids) + 1, false);
367 if ($token === '*') {
368 $current[0] = $previous[0];
369 }
370 foreach ($ids as $j => $id) {
371 if ($token === '*') {
372 $current[$j + 1] = $previous[$j + 1] || $current[$j];
373 } else {
374 $current[$j + 1] = $previous[$j] && ($token === '?' || $token === $id);
375 }
376 }
377 $previous = $current;
378 }
379 return $previous[count($ids)];
380}

382function execute(string $line, array &$settings): string
383{
384 global $identifiers, $rowOf, $danOf;
385 $command = explode(' ', $line, 2)[0];
386 if (!array_key_exists($command, ARG_COUNTS)) {
387 fail("unknown command $command");
388 }
389 $k = ARG_COUNTS[$command];
390 if ($command === 'SET' || $command === 'TABLE') {
391 $parts = explode(' ', $line);
392 if (count($parts) < 3) {
393 fail('missing argument');
394 }
395 if (count($parts) > 3) {
396 fail('too many arguments');
397 }
398 $args = array_slice($parts, 1);
399 $text = '';
400 } else {
401 if (substr_count($line, ' ') < $k + 1) {
402 fail('missing argument');
403 }
404 $parts = explode(' ', $line, $k + 2);
405 $args = array_slice($parts, 1, $k);
406 $text = $parts[$k + 1];
407 }
408 $systemCount = 1;
409 if (in_array($command, ['SET', 'NORMALIZE', 'DETECT'], true)) {
410 $systemCount = 0;
411 } elseif (in_array($command, ['CONVERT', 'DIFF', 'SAME'], true)) {
412 $systemCount = 2;
413 }
414 foreach (array_slice($args, 0, $systemCount) as $system) {
415 if (!in_array($system, SYSTEMS, true)) {
416 fail("unknown system $system");
417 }
418 }
419 if ($command === 'SET') {
420 [$option, $value] = $args;
421 if (!isset(SET_VALUES[$option])) {
422 fail("unknown option $option");
423 }
424 if (!in_array($value, SET_VALUES[$option], true)) {
425 fail("unknown value $value");
426 }
427 $settings[$option] = $value;
428 return "$option=$value";
429 }
430 if ($command === 'TABLE') {
431 [$system, $row] = $args;
432 if (!in_array($row, ROWS, true)) {
433 fail("unknown row $row");
434 }
435 $ids = array_filter($identifiers, fn($id) => $rowOf[$id] === $row);
436 return implode(' ', array_map(fn($id) => spelling($system, $id), $ids));
437 }
438 $tokens = $command === 'FIND' ? pattern_tokens($args[1], $args[0]) : [];
439 $slashAllowed = $command === 'HAIKU' || $command === 'SAME';
440 foreach (str_split($text) as $i => $c) {
441 $allowed = ctype_alpha($c) || ctype_digit($c) || str_contains("'- .,!?;:()\"", $c) || ($slashAllowed && $c === '/');
442 if (!$allowed) {
443 column_error($i + 1, 'invalid character');
444 }
445 }
446 $words = words_in($text);
447 if ($command === 'SAME') {
448 $left = array_filter($words, fn($w) => $w['part'] === 1);
449 $right = array_filter($words, fn($w) => $w['part'] === 2);
450 if (substr_count($text, '/') !== 1 || $left === [] || $right === []) {
451 fail('expected two texts');
452 }
453 } elseif ($command === 'HAIKU') {
454 $lines = substr_count($text, '/') + 1;
455 if (!in_array($lines, [3, 5], true)) {
456 fail('expected 3 or 5 lines');
457 }
458 for ($i = 1; $i <= $lines; ++$i) {
459 if (!array_any($words, fn($w) => $w['part'] === $i)) {
460 fail('expected 3 or 5 lines');
461 }
462 }
463 }
464 if ($words === []) {
465 fail('no words');
466 }
467 foreach ($words as &$word) {
468 $word['pattern'] = word_pattern($word['text']);
469 if ($word['pattern'] === '') {
470 column_error($word['start'], 'mixed case');
471 }
472 }
473 unset($word);
474 if ($command === 'DETECT') {
475 $output = [];
476 foreach (SYSTEMS as $system) {
477 $verdict = 'canonical';
478 try {
479 $parsedWords = $words;
480 foreach ($parsedWords as &$word) {
481 [$word['ids'], $word['columns']] = parse_word($word['text'], $word['start'], $system);
482 }
483 unset($word);
484 foreach ($parsedWords as $word) {
485 if (written($word, $system, $settings, true) !== $word['text']) {
486 $verdict = 'readable';
487 }
488 }
489 } catch (InvalidArgumentException $exception) {
490 unset($word);
491 $verdict = 'error';
492 }
493 $output[] = "$system=$verdict";
494 }
495 return implode(' ', $output);
496 }
497 foreach ($words as &$word) {
498 if ($command === 'NORMALIZE') {
499 $system = 'NORMAL';
500 } elseif ($command === 'SAME') {
501 $system = $args[$word['part'] - 1];
502 } else {
503 $system = $args[0];
504 }
505 [$word['ids'], $word['columns']] = parse_word($word['text'], $word['start'], $system);
506 }
507 unset($word);
508 $system = $command === 'NORMALIZE' ? 'HEPBURN' : $args[0];
509 if (in_array($command, ['CONVERT', 'NORMALIZE', 'REVERSE'], true)) {
510 $target = $command === 'CONVERT' ? $args[1] : $system;
511 $output = [];
512 foreach ($words as $word) {
513 $ids = $word['ids'];
514 $columns = $word['columns'];
515 if ($command === 'REVERSE') {
516 foreach ($ids as $i => $id) {
517 if ($id === '-') {
518 $ids[$i] = substr($word['ids'][$i - 1], -1);
519 }
520 }
521 $ids = array_reverse($ids);
522 $columns = array_reverse($columns);
523 }
524 $output[] = written($word, $target, $settings, false, $ids, $columns);
525 }
526 return replace_words($text, $words, $output);
527 }
528 if ($command === 'VALID') {
529 foreach ($words as $word) {
530 $expected = written($word, $system, $settings, true);
531 if ($expected !== $word['text']) {
532 $i = 0;
533 while ($i < min(strlen($expected), strlen($word['text'])) && $expected[$i] === $word['text'][$i]) {
534 ++$i;
535 }
536 return 'INVALID col ' . ($word['start'] + $i) . ': expected ' . $expected;
537 }
538 }
539 return 'VALID';
540 }
541 if ($command === 'DIFF') {
542 $output = [];
543 foreach ($words as $word) {
544 $a = written($word, $args[0], $settings);
545 $b = written($word, $args[1], $settings);
546 if ($a !== $b) {
547 $output[] = "$a>$b";
548 }
549 }
550 $prefix = count($output) . ' of ' . count($words);
551 return $output === [] ? $prefix : $prefix . ': ' . implode(' ', $output);
552 }
553 if ($command === 'SAME') {
554 $left = array_values(array_filter($words, fn($w) => $w['part'] === 1));
555 $right = array_values(array_filter($words, fn($w) => $w['part'] === 2));
556 $exact = true;
557 for ($i = 0; $i < min(count($left), count($right)); ++$i) {
558 $a = $left[$i]['ids'];
559 $b = $right[$i]['ids'];
560 $aLong = long_flags($a)[1];
561 $bLong = long_flags($b)[1];
562 $aLoose = array_values(array_filter($a, fn($id, $j) => !$aLong[$j], ARRAY_FILTER_USE_BOTH));
563 $bLoose = array_values(array_filter($b, fn($id, $j) => !$bLong[$j], ARRAY_FILTER_USE_BOTH));
564 if ($aLoose !== $bLoose) {
565 return 'DIFFERENT at word ' . ($i + 1);
566 }
567 $exact = $exact && $a === $b;
568 }
569 if (count($left) !== count($right)) {
570 return 'DIFFERENT at word ' . (min(count($left), count($right)) + 1);
571 }
572 return $exact ? 'EXACT' : 'LOOSE';
573 }
574 if ($command === 'MORAE') {
575 $total = 0;
576 $output = [];
577 foreach ($words as $word) {
578 $total += count($word['ids']);
579 $output[] = implode('.', array_map(fn($id) => spelling($system, $id), $word['ids']));
580 }
581 return "$total: " . implode(' ', $output);
582 }
583 if ($command === 'SYLLABLES') {
584 $total = 0;
585 $output = [];
586 foreach ($words as $word) {
587 [$paired] = long_flags($word['ids']);
588 $parts = write_parts($word['ids'], $word['columns'], $system, 'KEEP');
589 $syllables = [];
590 foreach ($word['ids'] as $i => $id) {
591 if ($i > 0 && (!is_syllabic($id) || $paired[$i])) {
592 $syllables[count($syllables) - 1] .= $parts[$i];
593 } else {
594 $syllables[] = $parts[$i];
595 }
596 }
597 $total += count($syllables);
598 $output[] = implode('.', $syllables);
599 }
600 return "$total: " . implode(' ', $output);
601 }
