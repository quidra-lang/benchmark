19function metadata()
20 identifiers = String[]
21 row_of = Dict{String,String}()
22 dan_of = Dict{String,String}()
23 i_of = Dict{String,String}()
24 for (row,text) in zip(ROWS,ROW_TEXT)
25 for id in split(text)
26 push!(identifiers,id)
27 row_of[id] = row
28 dan_of[id] = string(last(id))
29 last(id) == 'i' && (i_of[row] = id)
30 end
31 end
32 for row in CONTRACT_ROWS
33 stem = i_of[row][1:end-1]
34 for dan in ("ya","yu","yo")
35 id = stem*dan
36 push!(identifiers,id)
37 row_of[id] = row
38 dan_of[id] = dan
39 end
40 end
41 return identifiers,row_of,dan_of,i_of
42end

49function spelling(system,id)
50 id in ("N","Q","-") && return id
51 if system in ("HEPBURN","TRADITIONAL","WAPURO")
52 changes = Dict("si"=>"shi", "ti"=>"chi", "tu"=>"tsu", "hu"=>"fu", "zi"=>"ji")
53 if system != "WAPURO"
54 merge!(changes,Dict("di"=>"ji", "du"=>"zu", "wi"=>"i", "we"=>"e", "wo"=>"o"))
55 end
56 haskey(changes,id) && return changes[id]
57 if startswith(id,"sy")
58 return "sh"*id[3:end]
59 elseif startswith(id,"ty")
60 return "ch"*id[3:end]
61 elseif startswith(id,"zy") || (system != "WAPURO" && startswith(id,"dy"))
62 return "j"*id[3:end]
63 end
64 elseif system == "KUNREI"
65 changes = Dict("di"=>"zi", "du"=>"zu", "wi"=>"i", "we"=>"e", "wo"=>"o",
66 "dya"=>"zya", "dyu"=>"zyu", "dyo"=>"zyo")
67 return get(changes,id,id)
68 end
69 return id
70end

72function reading_tables()
73 tables = Dict{String,Vector{Pair{String,String}}}()
74 for system in vcat(String.(SYSTEMS),["NORMAL"])
75 seen = Dict{String,String}()
76 styles = system == "NORMAL" ? ["HEPBURN","KUNREI","NIHON","WAPURO"] : [system]
77 for id in IDS
78 for style in styles
79 text = spelling(style,id)
80 !haskey(seen,text) && (seen[text] = id)
81 end
82 end
83 tables[system] = sort(collect(seen);by=pair -> -length(pair.first))
84 end
85 return tables
86end

89fail(message) = throw(ArgumentError(message))
90column_error(column,message) = fail("col $column: $message")
91is_syllabic(id) = haskey(ROW_OF,id)
92is_vowel(id) = id in ("a","i","u","e","o")

94function match_syllable(word,position,system)
95 position > length(word) && return nothing
96 index = findfirst(pair -> startswith(SubString(word,position),pair.first),READINGS[system])
97 return index === nothing ? nothing : READINGS[system][index]
98end

100function parse_word(text,start,system)
101 word = lowercase(text)
102 ids = String[]
103 columns = Int[]
104 p = 1
105 while p <= length(word)
106 c = word[p]
107 c2 = p < length(word) ? word[p+1] : '\0'
108 c3 = p+2 <= length(word) ? word[p+2] : '\0'
109 matched = match_syllable(word,p,system)
110 advance = 1
111 if c == '-'
112 (isempty(ids) || !is_syllabic(ids[end])) && column_error(start+p-1,"misplaced long mark")
113 id = "-"
114 elseif c == '\''
115 column_error(start+p-1,"misplaced apostrophe")
116 elseif matched !== nothing
117 id = matched.second
118 advance = length(matched.first)
119 elseif system in ("WAPURO","NORMAL") && startswith(SubString(word,p),"xtu")
120 id = "Q"
121 advance = 3
122 elseif c == 'n'
123 id = "N"
124 if system == "WAPURO"
125 advance = c2 == 'n' ? 2 : 1
126 elseif system == "NORMAL"
127 if c2 == '\'' || (c2 == 'n' && !(c3 in ('a','i','u','e','o','y','\'')))
128 advance = 2
129 end
130 elseif c2 == '\''
131 advance = 2
132 end
133 elseif system in ("TRADITIONAL","NORMAL") && c == 'm' && c2 in ('b','m','p')
134 id = "N"
135 elseif 'a' <= c <= 'z' && !(c in ('a','i','u','e','o','n')) && c == c2 && match_syllable(word,p+1,system) !== nothing
136 id = "Q"
137 elseif system in ("HEPBURN","TRADITIONAL","NORMAL") && c == 't' && c2 == 'c' && c3 == 'h' && match_syllable(word,p+1,system) !== nothing
138 id = "Q"
139 else
140 column_error(start+p-1,"unknown syllable")
141 end
142 push!(ids,id)
143 push!(columns,start+p-1)
144 p += advance
145 end
146 return ids,columns
147end

149function long_flags(ids)
150 paired = falses(length(ids))
151 long = [id == "-" for id in ids]
152 i = 1
153 while i < length(ids)
154 first,second = ids[i],ids[i+1]
155 if is_syllabic(first) && is_vowel(second)
156 vowel = string(last(first))
157 if vowel == second || (vowel == "o" && second == "u") || (vowel == "e" && second == "i")
158 paired[i+1] = true
159 long[i+1] = (vowel == "o" && second in ("o","u")) || (vowel == "u" && second == "u")
160 i += 2
161 continue
162 end
163 end
164 i += 1
165 end
166 return paired,long
167end

169function write_parts(ids,columns,system,long_setting)
170 paired,long = long_flags(ids)
171 parts = String[]
172 for i in eachindex(ids)
173 id = ids[i]
174 next_id = i < length(ids) ? ids[i+1] : ""
175 next_text = is_syllabic(next_id) ? spelling(system,next_id) : ""
176 first = isempty(next_text) ? '\0' : next_text[1]
177 if long[i]
178 vowel = last(ids[i-1])
179 if long_setting == "KEEP"
180 part = id == "-" ? "-" : spelling(system,id)
181 elseif long_setting == "DROP"
182 part = ""
183 elseif long_setting == "PASSPORT"
184 part = vowel == 'o' && !(first in ('a','i','u','e','o')) ? "h" : ""
185 elseif long_setting == "HYPHEN"
186 part = "-"
187 else
188 part = string(vowel)
189 end
190 elseif is_syllabic(id)
191 part = spelling(system,id)
192 elseif id == "N"
193 if system == "WAPURO"
194 part = next_id == "N" || first in ('a','i','u','e','o','y','n') ? "nn" : "n"
195 elseif system == "TRADITIONAL" && first in ('b','m','p')
196 part = "m"
197 else
198 part = first in ('a','i','u','e','o','y') ? "n'" : "n"
199 end
200 else
201 if !isempty(next_text) && !(first in ('a','i','u','e','o','n'))
202 part = system in ("HEPBURN","TRADITIONAL") && startswith(next_text,"ch") ? "t" : string(first)
203 elseif system == "WAPURO"
204 part = "xtu"
205 else
206 column_error(columns[i],"small tsu cannot be written")
207 end
208 end
209 push!(parts,part)
210 end
211 return parts
212end

224function word_pattern(text)
225 text == lowercase(text) && return "LOWER"
226 if 'A' <= text[1] <= 'Z' && text[2:end] == lowercase(text[2:end])
227 return "TITLE"
228 end
229 if count(c -> 'A' <= c <= 'Z',text) >= 2 && text == uppercase(text)
230 return "UPPER"
231 end
232 return ""
233end
234
235function written(word,system,settings;canonical=false,ids=word.ids,columns=word.columns)
236 mode = canonical ? "KEEP" : settings["LONG"]
237 text = join(write_parts(ids,columns,system,mode))
238 pattern = canonical || settings["CASE"] == "KEEP" ? word.pattern : settings["CASE"]
239 pattern == "UPPER" && return uppercase(text)
240 pattern == "TITLE" && return uppercasefirst(text)
241 return text
242end

244function words_in(text)
245 result = RomajiWord[]
246 for matched in eachmatch(r"[A-Za-z'-]+",text)
247 start = matched.offset
248 part = count(==('/'),text[1:start-1])+1
249 push!(result,RomajiWord(String(matched.match),start,"",String[],Int[],part,length(result)+1))
250 end
251 return result
252end

254function replace_words(text,words,replacements)
255 result = IOBuffer()
256 position = 1
257 for (word,replacement) in zip(words,replacements)
258 print(result,text[position:word.start-1],replacement)
259 position = word.start+length(word.text)
260 end
261 print(result,text[position:end])
262 return String(take!(result))
263end

269function elements(ids)
270 result = Tuple{String,Int,Int}[]
271 for id in ids
272 if id == "-"
273 continue
274 elseif id == "N"
275 push!(result,("N",0,0))
276 elseif id == "Q"
277 push!(result,("tu",0,1))
278 else
279 row = ROW_OF[id]
280 voice = 0
281 if row in ("ga","za","da","ba")
282 voice = 1
283 elseif row == "pa"
284 voice = 2
285 end
286 if length(DAN_OF[id]) == 2
287 push!(result,(devoice(I_OF[row]),voice,0))
288 push!(result,(DAN_OF[id],0,1))
289 else
290 push!(result,(devoice(id),voice,0))
291 end
292 end
293 end
294 return result
295end

297function collation(word,order)
298 ranks = order == "GOJUON" ? GOJUON : IROHA
299 items = elements(word.ids)
300 return (Tuple(findfirst(==(item[1]),ranks) for item in items),
301 Tuple(item[2] for item in items),Tuple(item[3] for item in items),
302 Tuple(id == "-" ? 1 : 0 for id in word.ids),word.index)
303end

305function pattern_tokens(pattern,system)
306 occursin(r"^[a-z'\-?*]+$",pattern) || fail("invalid pattern")
307 tokens = String[]
308 chunk = ""
309 try
310 for c in pattern
311 if c in ('?','*')
312 append!(tokens,parse_word(chunk,1,system)[1])
313 push!(tokens,string(c))
314 chunk = ""
315 else
316 chunk *= string(c)
317 end
318 end
319 append!(tokens,parse_word(chunk,1,system)[1])
320 catch exception
321 exception isa ArgumentError || rethrow()
322 fail("invalid pattern")
323 end
324 return tokens
325end

327function wildcard_matches(tokens,ids)
328 previous = falses(length(ids)+1)
329 previous[1] = true
330 for token in tokens
331 current = falses(length(ids)+1)
332 token == "*" && (current[1] = previous[1])
333 for j in eachindex(ids)
334 if token == "*"
335 current[j+1] = previous[j+1] || current[j]
336 else
337 current[j+1] = previous[j] && (token == "?" || token == ids[j])
338 end
339 end
340 previous = current
341 end
342 return previous[end]
343end

345function execute(line,settings)
346 command = split(line,' ';limit=2)[1]
347 haskey(ARG_COUNTS,command) || fail("unknown command $command")
348 k = ARG_COUNTS[command]
349 if command in ("SET","TABLE")
350 parts = split(line,' ')
351 length(parts) < 3 && fail("missing argument")
352 length(parts) > 3 && fail("too many arguments")
353 args = parts[2:3]
354 text = ""
355 else
356 count(==(' '),line) >= k+1 || fail("missing argument")
357 parts = split(line,' ';limit=k+2)
358 args = parts[2:k+1]
359 text = String(parts[end])
360 end
361 system_count = 1
362 if command in ("SET","NORMALIZE","DETECT")
363 system_count = 0
364 elseif command in ("CONVERT","DIFF","SAME")
365 system_count = 2
366 end
367 for system in args[1:system_count]
368 system in SYSTEMS || fail("unknown system $system")
369 end
370 if command == "SET"
371 option,value = args
372 haskey(SET_VALUES,option) || fail("unknown option $option")
373 value in SET_VALUES[option] || fail("unknown value $value")
374 settings[option] = value
375 return "$option=$value"
376 elseif command == "TABLE"
377 system,row = args
378 row in ROWS || fail("unknown row $row")
379 return join((spelling(system,id) for id in IDS if ROW_OF[id] == row)," ")
380 end
381 tokens = command == "FIND" ? pattern_tokens(args[2],args[1]) : String[]
382 slash_allowed = command in ("HAIKU","SAME")
383 for (column,c) in enumerate(text)
384 allowed = 'A' <= c <= 'Z' || 'a' <= c <= 'z' || '0' <= c <= '9' || c in "'- .,!?;:()\"" || (slash_allowed && c == '/')
385 allowed || column_error(column,"invalid character")
386 end
387 words = words_in(text)
388 if command == "SAME"
389 (count(==('/'),text) == 1 && any(w.part == 1 for w in words) && any(w.part == 2 for w in words)) || fail("expected two texts")
390 elseif command == "HAIKU"
391 lines = count(==('/'),text)+1
392 (lines in (3,5) && all(any(w.part == i for w in words) for i in 1:lines)) || fail("expected 3 or 5 lines")
393 end
394 isempty(words) && fail("no words")
395 for word in words
396 word.pattern = word_pattern(word.text)
397 isempty(word.pattern) && column_error(word.start,"mixed case")
398 end
399 if command == "DETECT"
400 output = String[]
401 for system in SYSTEMS
402 verdict = "canonical"
403 try
404 for word in words
405 word.ids,word.columns = parse_word(word.text,word.start,system)
406 end
407 for word in words
408 if written(word,system,settings;canonical=true) != word.text
409 verdict = "readable"
410 end
411 end
412 catch exception
413 exception isa ArgumentError || rethrow()
414 verdict = "error"
415 end
416 push!(output,"$system=$verdict")
417 end
418 return join(output," ")
419 end
420 for word in words
421 if command == "NORMALIZE"
422 system = "NORMAL"
423 elseif command == "SAME"
424 system = args[word.part]
425 else
426 system = args[1]
427 end
428 word.ids,word.columns = parse_word(word.text,word.start,system)
429 end
430 system = command == "NORMALIZE" ? "HEPBURN" : args[1]
431 if command in ("CONVERT","NORMALIZE","REVERSE")
432 target = command == "CONVERT" ? args[2] : system
433 output = String[]
434 for word in words
435 ids = word.ids
436 columns = word.columns
437 if command == "REVERSE"
438 ids = [id == "-" ? string(last(word.ids[i-1])) : id for (i,id) in enumerate(word.ids)]
439 ids = reverse(ids)
440 columns = reverse(columns)
441 end
442 push!(output,written(word,target,settings;ids=ids,columns=columns))
443 end
444 return replace_words(text,words,output)
445 elseif command == "VALID"
446 for word in words
447 expected = written(word,system,settings;canonical=true)
448 if expected != word.text
449 i = 1
450 while i <= min(length(expected),length(word.text)) && expected[i] == word.text[i]
451 i += 1
452 end
453 return "INVALID col $(word.start+i-1): expected $expected"
454 end
455 end
456 return "VALID"
457 elseif command == "DIFF"
458 output = String[]
459 for word in words
460 first = written(word,args[1],settings)
461 second = written(word,args[2],settings)
462 first != second && push!(output,first*">"*second)
463 end
464 prefix = "$(length(output)) of $(length(words))"
465 return isempty(output) ? prefix : prefix*": "*join(output," ")
466 elseif command == "SAME"
467 left = filter(w -> w.part == 1,words)
468 right = filter(w -> w.part == 2,words)
469 exact = true
470 for i in 1:min(length(left),length(right))
471 a,b = left[i].ids,right[i].ids
472 a_loose = a[.!long_flags(a)[2]]
473 b_loose = b[.!long_flags(b)[2]]
474 a_loose != b_loose && return "DIFFERENT at word $i"
475 exact &= a == b
476 end
477 length(left) != length(right) && return "DIFFERENT at word $(min(length(left),length(right))+1)"
478 return exact ? "EXACT" : "LOOSE"
479 elseif command == "MORAE"
480 output = [join(spelling.(Ref(system),word.ids),".") for word in words]
481 return "$(sum(length(w.ids) for w in words)): "*join(output," ")
482 elseif command == "SYLLABLES"
483 output = String[]
484 total = 0
485 for word in words
486 paired,long = long_flags(word.ids)
487 parts = write_parts(word.ids,word.columns,system,"KEEP")
488 syllables = String[]
489 for i in eachindex(word.ids)
490 if i > 1 && (!is_syllabic(word.ids[i]) || paired[i])
491 syllables[end] *= parts[i]
492 else
493 push!(syllables,parts[i])
494 end
495 end
496 total += length(syllables)
497 push!(output,join(syllables,"."))
498 end
499 return "$total: "*join(output," ")
500 elseif command == "ROW"
501 output = String[]
502 for word in words
503 for id in word.ids
504 push!(output,is_syllabic(id) ? "$(spelling(system,id)): $(ROW_OF[id]) $(DAN_OF[id])" : "$id: special")
505 end
506 end
507 return join(output,", ")
508 elseif command == "STATS"
509 counts = Dict(name=>0 for name in vcat(String.(BASE_ROWS),split("voiced semivoiced contracted N Q L")))
510 total = 0
511 for word in words
512 for id in word.ids
513 total += 1
514 if is_syllabic(id)
515 row = ROW_OF[id]
516 counts[get(BASE_ROW,row,row)] += 1
517 row in ("ga","za","da","ba") && (counts["voiced"] += 1)
518 row == "pa" && (counts["semivoiced"] += 1)
519 length(DAN_OF[id]) == 2 && (counts["contracted"] += 1)
520 else
521 counts[id == "-" ? "L" : id] += 1
522 end
523 end
524 end
525 output = ["morae=$total"]
526 append!(output,["$row=$(counts[row])" for row in BASE_ROWS if counts[row] > 0])
527 append!(output,["$name=$(counts[name])" for name in split("voiced semivoiced contracted N Q L")])
528 return join(output," ")
529 elseif command in ("SORT","INDEX")
530 ordered = sort(words;by=w -> collation(w,settings["ORDER"]))
531 command == "SORT" && return join((word.text for word in ordered)," ")
532 groups = String[]
533 previous = ""
534 for word in ordered
535 base = elements(word.ids)[1][1]
536 heading = settings["ORDER"] == "IROHA" || base == "N" ? base : ROW_OF[base]
537 if heading == previous
538 groups[end] *= " "*word.text
539 else
540 push!(groups,heading*": "*word.text)
541 previous = heading
542 end
543 end
544 return join(groups,"; ")
545 elseif command == "FIND"
546 output = [word.text for word in words if wildcard_matches(tokens,word.ids)]
547 prefix = "$(length(output)) of $(length(words))"
548 return isempty(output) ? prefix : prefix*": "*join(output," ")
549 elseif command == "HAIKU"
550 counts = [sum(length(w.ids) for w in words if w.part == i) for i in 1:count(==('/'),text)+1]
551 expected = length(counts) == 3 ? [5,7,5] : [5,7,5,7,7]
552 verdict = counts == expected ? "OK" : "NG"
553 form = length(counts) == 3 ? "haiku" : "tanka"
554 return "$verdict $form "*join(counts,"-")
555 elseif command == "KAIBUN"
556 bases = [item[1] for word in words for item in elements(word.ids)]
557 return (bases == reverse(bases) ? "YES " : "NO ")*join(bases,".")
558 elseif command == "SHIRITORI"
559 seen = Vector{String}[]
560 previous = ""
561 for (i,word) in enumerate(words)
562 items = elements(word.ids)
563 head,tail = items[1][1],items[end][1]
564 if i > 1 && head != previous
565 return "NG $i: $(word.text) must start with $(spelling(system,previous))"
566 elseif word.ids in seen
567 return "NG $i: $(word.text) is repeated"
568 elseif tail == "N"
569 return "NG $i: $(word.text) ends with N"
570 end
571 push!(seen,word.ids)
572 previous = tail
573 end
574 return "OK $(length(words))"
575 end
576end
