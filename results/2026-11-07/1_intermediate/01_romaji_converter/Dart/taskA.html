32String spelling(String id, int system) {
33 if (system == 3)
34 return id;
35 if (system == 2) {
36 const changes = {
37 'di': 'zi',
38 'du': 'zu',
39 'wi': 'i',
40 'we': 'e',
41 'wo': 'o',
42 'dya': 'zya',
43 'dyu': 'zyu',
44 'dyo': 'zyo'
45 };
46 return changes[id] ?? id;
47 }
48 final hepburn = system == 0 || system == 1;
49 const common = { 'si': 'shi', 'ti': 'chi', 'tu': 'tsu', 'hu': 'fu', 'zi': 'ji' };
50 final commonSpelling = common[id];
51 if (commonSpelling != null)
52 return commonSpelling;
53 if (hepburn) {
54 const extra = { 'di': 'ji', 'du': 'zu', 'wi': 'i', 'we': 'e', 'wo': 'o' };
55 final extraSpelling = extra[id];
56 if (extraSpelling != null)
57 return extraSpelling;
58 }
59 if (id.length == 3) {
60 final prefix = id.substring(0, 2), last = id.substring(2);
61 if (prefix == 'sy')
62 return 'sh$last';
63 if (prefix == 'ty')
64 return 'ch$last';
65 if (prefix == 'zy' || hepburn && prefix == 'dy')
66 return 'j$last';
67 }
68 return id;
69}

127 void checkCase(Word word) {
128 final codes = word.original.codeUnits, letters = codes.where(letter).toList();
129 if (letters.every((c) => !upper(c))) {
130 word.pattern = 0;
131 return;
132 }
133 if (upper(codes.first) && codes.skip(1).where(letter).every((c) => !upper(c))) {
134 word.pattern = 1;
135 return;
136 }
137 if (letters.length >= 2 && letters.every(upper)) {
138 word.pattern = 2;
139 return;
140 }
141 columnError(word.start + 1, 'mixed case');
142 }

144 (int, int)? match(String text, int position, int system) {
145 (int, int)? best;
146 final choices = system == 5 ? [0, 2, 3, 4] : [system];
147 for (var id = 0; id < table.length; id++) {
148 for (final s in choices) {
149 final candidate = spellings[s][id];

151 best = (id, candidate.length);

157 List<Mora> parse(String original, int start, int system) {
158 final text = original.toLowerCase(), result = <Mora>[];
159 var p = 0;
160 while (p < text.length) {
161 final c = text[p], next = p + 1 < text.length ? text[p + 1] : '';
162 final column = start + p + 1;
163 if (c == '-') {
164 if (result.isEmpty || result.last.id >= moraN)
165 columnError(column, 'misplaced long mark');
166 result.add(Mora(moraL, column));
167 p++;
168 continue;
169 }
170 if (c == "'")
171 columnError(column, 'misplaced apostrophe');
172 final syllable = match(text, p, system);
173 if (syllable != null) {
174 result.add(Mora(syllable.$1, column));
175 p += syllable.$2;
176 continue;
177 }
178 if ((system == 4 || system == 5) && text.startsWith('xtu', p)) {
179 result.add(Mora(moraQ, column));
180 p += 3;
181 continue;
182 }
183 if (c == 'n') {
184 var advance = 1;
185 if (system == 5) {
186 final third = p + 2 < text.length ? text[p + 2] : '';
187 if (next == "'" || next == 'n' && !vowel(third) && third != 'y' && third != "'")
188 advance = 2;
189 } else if (system == 4) {
190 if (next == 'n')
191 advance = 2;
192 } else if (next == "'") {
193 advance = 2;
194 }
195 result.add(Mora(moraN, column));
196 p += advance;
197 continue;
198 }
199 if ((system == 1 || system == 5) && c == 'm' && next.isNotEmpty && 'bmp'.contains(next)) {
200 result.add(Mora(moraN, column));
201 p++;
202 continue;
203 }
204 if (letter(c.codeUnitAt(0)) && !'aiueon'.contains(c) && next == c && match(text, p + 1, system) != null) {
205 result.add(Mora(moraQ, column));
206 p++;
207 continue;
208 }
209 if ((system == 0 || system == 1 || system == 5) && c == 't' && text.startsWith('ch', p + 1) && match(text,
210 p + 1, system) != null) {
211 result.add(Mora(moraQ, column));
212 p++;
213 continue;
214 }
215 columnError(column, 'unknown syllable');
216 }
217 return result;
218 }

220 String moraVowel(int id) => table[id].id.substring(table[id].id.length - 1);
221 List<bool> pairs(List<Mora> morae) {
222 final result = List.filled(morae.length, false);
223 var i = 0;
224 while (i + 1 < morae.length) {
225 final a = morae[i].id, b = morae[i + 1].id;
226 if (a < moraN && b < 5) {
227 final v = moraVowel(a), following = table[b].id;
228 if (v == following || v == 'o' && following == 'u' || v == 'e' && following == 'i') {
229 result[i + 1] = true;
230 i += 2;
231 continue;
232 }
233 }
234 i++;
235 }
236 return result;
237 }
238 List<bool> longElements(List<Mora> morae) {
239 final paired = pairs(morae), result = List.filled(morae.length, false);
240 for (var i = 0; i < morae.length; i++) {
241 if (morae[i].id == moraL) {
242 result[i] = true;
243 continue;
244 }
245 if (!paired[i])
246 continue;
247 final v = moraVowel(morae[i - 1].id), following = table[morae[i].id].id;
248 result[i] = v == 'o' && (following == 'o' || following == 'u') || v == 'u' && following == 'u';
249 }
250 return result;
251 }
252 List<String> parts(List<Mora> morae, int target, String longSetting) {
253 final long = longElements(morae), result = <String>[];
254 for (var i = 0; i < morae.length; i++) {
255 final mora = morae[i], nextId = i + 1 < morae.length ? morae[i + 1].id : -1;
256 final next = nextId >= 0 && nextId < moraN ? spellings[target][nextId] : '';
257 final first = next.isEmpty ? '' : next[0];
258 if (long[i]) {
259 final v = moraVowel(morae[i - 1].id);
260 result.add(switch (longSetting) {
261 'KEEP' => mora.id == moraL ? '-' : spellings[target][mora.id],
262 'DROP' => '',
263 'PASSPORT' => v == 'o' && !vowel(first) ? 'h' : '',
264 'HYPHEN' => '-',
265 _ => v,
266 });
267 } else if (mora.id < moraN) {
268 result.add(spellings[target][mora.id]);
269 } else if (mora.id == moraN) {
270 if (target == 4) {
271 result.add(nextId == moraN || vowel(first) || first == 'y' || first == 'n' ? 'nn' : 'n');
272 } else if (target == 1 && first.isNotEmpty && 'bmp'.contains(first)) {
273 result.add('m');
274 } else {
275 result.add(vowel(first) || first == 'y' ? "n'" : 'n');
276 }
277 } else {
278 if (first.isNotEmpty && !'aiueon'.contains(first)) {
279 result.add((target == 0 || target == 1) && next.startsWith('ch') ? 't' : first);
280 } else if (target == 4) {
281 result.add('xtu');
282 } else {
283 columnError(mora.column, 'small tsu cannot be written');
284 }
285 }
286 }
287 return result;
288 }

293 return pattern == 0 ? raw : pattern == 1 ? raw[0].toUpperCase() + raw.substring(1) : raw.toUpperCase();
294 }

352 List<int> pattern(String text, int system) {
353 if (text.codeUnits.any((c) => !(c >= 97 && c <= 122 || c == 39 || c == 45 || c == 63 || c == 42)))
354 throw LanguageError('invalid pattern');
355 final result = <int>[];
356 var start = 0;
357 try {
358 for (var i = 0; i <= text.length; i++) {
359 if (i == text.length || text[i] == '?' || text[i] == '*') {
360 if (i > start)
361 result.addAll(parse(text.substring(start, i), start, system).map((m) => m.id));
362 if (i < text.length)
363 result.add(text[i] == '?' ? -1 : -2);
364 start = i + 1;
365 }
366 }
367 } on LanguageError {
368 throw LanguageError('invalid pattern');
369 }
370 return result;
371 }

373 bool matches(List<int> pattern, List<Mora> morae) {
374 var previous = List.filled(morae.length + 1, false);
375 previous[0] = true;
376 for (final token in pattern) {
377 final current = List.filled(morae.length + 1, false);
378 for (var j = 0; j <= morae.length; j++) {
379 current[j] = token == -2 ? previous[j] || j > 0 && current[j - 1] : j > 0 && previous[j - 1] && (token == -1
380 || token == morae[j - 1].id);
381 }
382 previous = current;
383 }
384 return previous.last;
385 }

387 String execute(String line) {
388 const commands = ['SET', 'TABLE', 'CONVERT', 'NORMALIZE', 'REVERSE', 'VALID', 'DETECT', 'DIFF',
389 'SAME', 'MORAE', 'SYLLABLES', 'ROW', 'STATS', 'SORT', 'INDEX', 'FIND', 'HAIKU', 'KAIBUN', 'SHIRITORI'];
390 final firstSpace = line.indexOf(' '), op = firstSpace < 0 ? line : line.substring(0, firstSpace);
391 if (!commands.contains(op))
392 throw LanguageError('unknown command $op');
393 final fixed = op == 'SET' || op == 'TABLE';
394 final argumentCount = ['CONVERT', 'DIFF', 'SAME', 'FIND'].contains(op) ? 2 : ['NORMALIZE', 'DETECT'].contains(op) ? 0 : 1;
395 List<String> args;
396 String text = '';
397 if (fixed) {
398 final split = line.split(' ');
399 if (split.length < 3)
400 throw LanguageError('missing argument');
401 if (split.length > 3)
402 throw LanguageError('too many arguments');
403 args = split.skip(1).toList();
404 } else {
405 args = [];
406 var cursor = firstSpace;
407 if (cursor < 0)
408 throw LanguageError('missing argument');
409 for (var i = 0; i < argumentCount; i++) {
410 final end = line.indexOf(' ', cursor + 1);
411 if (end < 0)
412 throw LanguageError('missing argument');
413 args.add(line.substring(cursor + 1, end));
414 cursor = end;
415 }
416 text = line.substring(cursor + 1);
417 }
418 var system = 0, target = 0;
419 final systemCount = op == 'TABLE' ? 1 : fixed || ['NORMALIZE', 'DETECT'].contains(op) ? 0 : ['CONVERT',
420 'DIFF', 'SAME'].contains(op) ? 2 : 1;
421 for (var i = 0; i < systemCount; i++) {
422 final value = systems.indexOf(args[i]);
423 if (value < 0)
424 throw LanguageError('unknown system ${args[i]}');
425 if (i == 0)
426 system = value;
427 else target = value;
428 }
429 if (op == 'SET') {
430 const allowed = {
431 'LONG': ['KEEP', 'DROP', 'PASSPORT', 'HYPHEN', 'DOUBLE'],
432 'CASE': ['KEEP', 'LOWER', 'UPPER', 'TITLE'],
433 'ORDER': ['GOJUON', 'IROHA']
434 };
435 if (!allowed.containsKey(args[0]))
436 throw LanguageError('unknown option ${args[0]}');
437 if (!allowed[args[0]]!.contains(args[1]))
438 throw LanguageError('unknown value ${args[1]}');
439 if (args[0] == 'LONG')
440 longMode = args[1];
441 else if (args[0] == 'CASE') caseMode = args[1];
442 else order = args[1];
443 return '${args[0]}=${args[1]}';
444 }
445 if (op == 'TABLE') {
446 final row = rowNames.indexOf(args[1]);
447 if (row < 0)
448 throw LanguageError('unknown row ${args[1]}');
449 return [for (var i = 0; i < table.length; i++) if (table[i].row == row) spellings[system][i]].join(' ');
450 }
451 final findPattern = op == 'FIND' ? pattern(args[1], system) : <int>[];
452 final slash = op == 'SAME' || op == 'HAIKU';
453 const separators = ' 0123456789.,!?;:()"';
454 for (var i = 0; i < text.length; i++) {
455 final c = text.codeUnitAt(i);
456 if (!wordCharacter(c) && !separators.contains(text[i]) && !(slash && text[i] == '/'))
457 columnError(i + 1, 'invalid character');
458 }
459 final parsed = words(text), lines = text.split('/');
460 if (op == 'SAME' && (lines.length != 2 || lines.any((part) => words(part).isEmpty)))
461 throw LanguageError('expected two texts');
462 if (op == 'HAIKU' && (!(lines.length == 3 || lines.length == 5) || lines.any((part) => words(part).isEmpty)))
463 throw LanguageError('expected 3 or 5 lines');
464 if (parsed.isEmpty)
465 throw LanguageError('no words');
466 for (final word in parsed)
467 checkCase(word);
468 if (op == 'DETECT') {
469 final results = <String>[];
470 for (var s = 0; s < 5; s++) {
471 var verdict = 'canonical';
472 try {
473 for (final word in parsed)
474 word.morae = parse(word.original, word.start, s);
475 if (parsed.any((word) => write(word, s, canonical: true) != word.original))
476 verdict = 'readable';
477 } on LanguageError {
478 verdict = 'error';
479 }
480 results.add('${systems[s]}=$verdict');
481 }
482 return results.join(' ');
483 }
484 final slashPosition = op == 'SAME' ? text.indexOf('/') : -1;
485 for (final word in parsed)
486 word.morae = parse(word.original, word.start, op == 'NORMALIZE' ? 5 : op == 'SAME' && word.start > slashPosition ? target : system);
487 if (['CONVERT', 'NORMALIZE', 'REVERSE'].contains(op)) {
488 final output = StringBuffer();
489 var cursor = 0;
490 for (final word in parsed) {
491 output.write(text.substring(cursor, word.start));
492 List<Mora>? reversed;
493 if (op == 'REVERSE') {
494 reversed = [for (var i = 0; i < word.morae.length; i++) word.morae[i].id == moraL ? Mora('aiueo'.indexOf(moraVowel(word.morae[i - 1].id)),
495 word.morae[i].column): word.morae[i]].reversed.toList();
496 }
497 output.write(write(word, op == 'NORMALIZE' ? 0 : op == 'REVERSE' ? system : target, morae: reversed));
498 cursor = word.end;
499 }
500 output.write(text.substring(cursor));
501 return output.toString();
502 }
503 if (op == 'VALID') {
504 for (final word in parsed) {
505 final canonical = write(word, system, canonical: true);
506 if (canonical == word.original)
507 continue;
508 var i = 0;
509 while (i < canonical.length && i < word.original.length && canonical[i] == word.original[i])
510 i++;
511 return 'INVALID col ${word.start + i + 1}: expected $canonical';
512 }
513 return 'VALID';
514 }
515 if (op == 'DIFF') {
516 final differences = <String>[];
517 for (final word in parsed) {
518 final a = write(word, system), b = write(word, target);
519 if (a != b)
520 differences.add('$a>$b');
521 }
522 return '${differences.length} of ${parsed.length}${differences.isEmpty ? '' : ': ${differences.join(' ')}'}';
523 }
524 if (op == 'SAME') {
525 final left = parsed.where((w) => w.start < slashPosition).toList(), right = parsed.where((w) => w.start > slashPosition).toList();
526 var exact = true;
527 for (var i = 0; i < left.length && i < right.length; i++) {
528 if (!sameMorae(loose(left[i]), loose(right[i])))
529 return 'DIFFERENT at word ${i + 1}';
530 if (!sameMorae(left[i].morae, right[i].morae))
531 exact = false;
532 }
533 if (left.length != right.length)
534 return 'DIFFERENT at word ${(left.length < right.length ? left.length : right.length) + 1}';
535 return exact ? 'EXACT' : 'LOOSE';
536 }
537 if (op == 'MORAE') {
538 final total = parsed.fold(0, (s, w) => s + w.morae.length);
539 String display(int id) => id < moraN ? spellings[system][id] : id == moraN ? 'N' : id == moraQ ? 'Q' : '-';
540 return '$total: ${parsed.map((w) => w.morae.map((m) => display(m.id)).join('.')).join(' ')}';
541 }
542 if (op == 'SYLLABLES') {
543 var total = 0;
544 final result = <String>[];
545 for (final word in parsed) {
546 final paired = pairs(word.morae), written = parts(word.morae, system, 'KEEP'), out = StringBuffer();
547 for (var i = 0; i < word.morae.length; i++) {
548 if (i == 0 || word.morae[i].id < moraN && !paired[i]) {
549 if (i > 0)
550 out.write('.');
551 total++;
552 }
553 out.write(written[i]);
554 }
555 result.add(out.toString());
556 }
557 return '$total: ${result.join(' ')}';
558 }
559 if (op == 'ROW') {
560 final result = <String>[];
561 for (final word in parsed) {
562 for (final mora in word.morae) {
563 final id = mora.id;
564 result.add(id < moraN ? '${spellings[system][id]}: ${rowNames[table[id].row]} ${table[id].dan}' : '${id == moraN ? 'N' : id == moraQ ? 'Q' : '-'}: special');
565 }
566 }
567 return result.join(', ');
568 }
569 if (op == 'STATS') {
570 var total = 0, voiced = 0, semi = 0, contracted = 0;
571 final rows = List.filled(10, 0), special = List.filled(3, 0);
572 for (final word in parsed) {
573 for (final mora in word.morae) {
574 total++;
575 if (mora.id >= moraN) {
576 special[mora.id - moraN]++;
577 continue;
578 }
579 final row = table[mora.id].row;
580 rows[baseRows[row]]++;
581 if (row >= 10 && row <= 13)
582 voiced++;
583 if (row == 14)
584 semi++;
585 if (mora.id >= 72)
586 contracted++;
587 }
588 }
589 return ['morae=$total', for (var i = 0; i < 10; i++) if (rows[i] > 0) '${rowNames[i]}=${rows[i]}',
590 'voiced=$voiced', 'semivoiced=$semi', 'contracted=$contracted', 'N=${special[0]}', 'Q=${special[1]}',
591 'L=${special[2]}'].join(' ');
592 }
593 if (op == 'FIND') {
594 final found = parsed.where((word) => matches(findPattern, word.morae)).toList();
595 return '${found.length} of ${parsed.length}${found.isEmpty ? '' : ': ${found.map((w) => w.original).join(' ')}'}';
596 }
597 if (op == 'HAIKU') {
598 final counts = List.filled(lines.length, 0);
599 for (final word in parsed)
600 counts[text.substring(0, word.start).split('/').length - 1] += word.morae.length;
601 const expected = [5, 7, 5, 7, 7];
602 final ok = counts.indexed.every((entry) => entry.$2 == expected[entry.$1]);
603 return '${ok ? 'OK' : 'NG'} ${counts.length == 3 ? 'haiku' : 'tanka'} ${counts.join('-')}';
604 }
605 for (final word in parsed)
606 collate(word);
607 if (op == 'SORT' || op == 'INDEX') {
608 final ranks = order == 'GOJUON' ? List.generate(gojuon.length, (i) => i) :
609 gojuon.map((base) => iroha.indexOf(base)).toList();
610 for (final word in parsed) {
611 word.sortKey = [
612 word.elements.map((e) => ranks[e.base]).toList(),
613 word.elements.map((e) => e.voice).toList(),
614 word.elements.map((e) => e.size).toList(),
615 word.morae.map((m) => m.id == moraL ? 1 : 0).toList(),
616 ];
617 }
618 parsed.sort(compareWords);
619 if (op == 'SORT')
620 return parsed.map((w) => w.original).join(' ');
621 final groups = <String>[];
622 String? previous;
623 var group = <String>[];
624 for (final word in parsed) {
625 final base = gojuon[word.elements.first.base];
626 final heading = order == 'IROHA' || base == 'N' ? base : rowNames[table.firstWhere((m) => m.id == base).row];
627 if (heading != previous) {
628 if (previous != null)
629 groups.add('$previous: ${group.join(' ')}');
630 previous = heading;
631 group = [];
632 }
633 group.add(word.original);
634 }
635 groups.add('$previous: ${group.join(' ')}');
636 return groups.join('; ');
637 }
638 if (op == 'KAIBUN') {
639 final sequence = parsed.expand((w) => w.elements).map((e) => e.base).toList();
640 final palindrome = sequence.indexed.take(sequence.length ~/ 2).every((entry) =>
641 entry.$2 == sequence[sequence.length - 1 - entry.$1]);
642 return '${palindrome ? 'YES' : 'NO'} ${sequence.map((b) => gojuon[b]).join('.')}';
643 }
644 for (var i = 0; i < parsed.length; i++) {
645 final word = parsed[i], head = word.elements.first.base, tail = word.elements.last.base;
646 if (i > 0 && head != parsed[i - 1].elements.last.base) {
647 final previous = gojuon[parsed[i - 1].elements.last.base];
648 final expected = previous == 'N' ? 'N' : spellings[system][table.indexWhere((m) => m.id == previous)];
649 return 'NG ${i + 1}: ${word.original} must start with $expected';
650 }
651 if (parsed.take(i).any((earlier) => sameMorae(earlier.morae, word.morae)))
652 return 'NG ${i + 1}: ${word.original} is repeated';
653 if (gojuon[tail] == 'N')
654 return 'NG ${i + 1}: ${word.original} ends with N';
655 }
656 return 'OK ${parsed.length}';
657 }
658}
