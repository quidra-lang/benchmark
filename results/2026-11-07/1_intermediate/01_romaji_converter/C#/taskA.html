70 static bool Letter(char c) => c >= 'a' && c <= 'z' || c >= 'A' && c <= 'Z';
71 static bool Begins(string text, string choices) => text.Length > 0 && choices.Contains(text[0]);

81 static int Voice(string row) => row == "pa" ? 2 : new[] { "ga","za","da","ba" }.Contains(row) ? 1 : 0;
82 static string Spelling(string system, string id) {
83 bool hepburn = system is "HEPBURN" or "TRADITIONAL", wapuro = system == "WAPURO";
84 if (hepburn || wapuro) {
85 string result = id switch {
86 "si" => "shi", "ti" => "chi", "tu" => "tsu", "hu" => "fu", "zi" => "ji",
87 "di" => hepburn ? "ji" : id, "du" => hepburn ? "zu" : id,
88 "wi" => hepburn ? "i" : id, "we" => hepburn ? "e" : id, "wo" => hepburn ? "o" : id,
89 _ => id
90 };
91 if (id.StartsWith("sy",StringComparison.Ordinal)) {
92 return "sh"+id[^1];
93 }
94 if (id.StartsWith("ty",StringComparison.Ordinal)) {
95 return "ch"+id[^1];
96 }
97 if (id.StartsWith("zy",StringComparison.Ordinal) || hepburn && id.StartsWith("dy",StringComparison.Ordinal)) {
98 return "j"+id[^1];
99 }
100 return result;
101 }
102 if (system == "KUNREI") {
103 return id switch {
104 "di" => "zi", "du" => "zu", "wi" => "i", "we" => "e", "wo" => "o",
105 _ => id.StartsWith("dy",StringComparison.Ordinal) ? "z"+id[1..] : id
106 };
107 }
108 return id;
109 }
110 MatchResult? Match(string text, int position, string system) {
111 for (int length = Math.Min(3,text.Length-position); length >= 1; length--) {
112 if (readers[system].TryGetValue(text.Substring(position,length),out string? id)) {
113 return new MatchResult(id,length);
114 }
115 }
116 return null;
117 }
118 List<Mora> Parse(Word word, string system) {
119 string text = word.Text.ToLowerInvariant();
120 var result = new List<Mora>();
121 bool normalize = system == "ANY";
122 int p = 0;
123 while (p < text.Length) {
124 char c = text[p], next = p+1 < text.Length ? text[p+1] : '\0';
125 int at = word.Start+p+1;
126 string id;
127 int advance;
128 if (c == '-') {
129 if (result.Count == 0 || !Regular(result[^1].Id)) {
130 Column(at,"misplaced long mark");
131 }
132 id = "L";
133 advance = 1;
134 } else if (c == '\'') {
135 Column(at,"misplaced apostrophe");
136 return result;
137 } else {
138 MatchResult? syllable = Match(text,p,system);
139 if (syllable is not null) {
140 id = syllable.Id;
141 advance = syllable.Length;
142 } else if ((normalize || system == "WAPURO") && text[p..].StartsWith("xtu",StringComparison.Ordinal)) {
143 id = "Q";
144 advance = 3;
145 } else if (c == 'n') {
146 id = "N";
147 advance = 1;
148 if (normalize) {
149 char third = p+2 < text.Length ? text[p+2] : '\0';
150 if (next == '\'' || next == 'n' && !"aiueoy'".Contains(third)) {
151 advance = 2;
152 }
153 } else if (system == "WAPURO") {
154 if (next == 'n') {
155 advance = 2;
156 }
157 } else if (next == '\'') {
158 advance = 2;
159 }
160 } else if ((normalize || system == "TRADITIONAL") && c == 'm' && "bmp".Contains(next)) {
161 id = "N";
162 advance = 1;
163 } else if (Letter(c) && !"aiueon".Contains(c) && c == next && Match(text,p+1,system) is not null) {
164 id = "Q";
165 advance = 1;
166 } else if ((normalize || system == "HEPBURN" || system == "TRADITIONAL") && c == 't' &&
167 text[(p+1)..].StartsWith("ch",StringComparison.Ordinal) && Match(text,p+1,system) is not null) {
168 id = "Q";
169 advance = 1;
170 } else {
171 Column(at,"unknown syllable");
172 return result;
173 }
174 }
175 result.Add(new Mora(id,at));
176 p += advance;
177 }
178 return result;
179 }
180 static List<Word> Scan(string text) => Regex.Matches(text,"[A-Za-z'-]+")
181 .Select(match => new Word(match.Value,match.Index,0,new List<Mora>())).ToList();
182 static int Pattern(string text) {
183 char[] letters = text.Where(Letter).ToArray();
184 if (letters.All(c => c >= 'a' && c <= 'z')) {
185 return 0;
186 }
187 if (text[0] >= 'A' && text[0] <= 'Z' && letters.Skip(1).All(c => c >= 'a' && c <= 'z')) {
188 return 1;
189 }
190 if (letters.Length >= 2 && letters.All(c => c >= 'A' && c <= 'Z')) {
191 return 2;
192 }
193 return -1;
194 }
195 LongInfo GetLongInfo(List<Mora> morae) {
196 bool[] paired = new bool[morae.Count], elements = new bool[morae.Count];
197 char[] vowels = new char[morae.Count];
198 for (int i = 0; i < morae.Count; i++) {
199 if (morae[i].Id == "L") {
200 elements[i] = true;
201 vowels[i] = morae[i-1].Id[^1];
202 }
203 }
204 int p = 0;
205 while (p+1 < morae.Count) {
206 string a = morae[p].Id, b = morae[p+1].Id;
207 char v = a[^1];
208 if (Regular(a) && Vowel(b) && (v == b[^1] || v == 'o' && b == "u" || v == 'e' && b == "i")) {
209 paired[p+1] = true;
210 if (v == 'o' && (b == "o" || b == "u") || v == 'u' && b == "u") {
211 elements[p+1] = true;
212 vowels[p+1] = v;
213 }
214 p += 2;
215 } else {
216 p++;
217 }
218 }
219 return new LongInfo(paired,elements,vowels);
220 }
221 List<string> Parts(Word word, string system, string longMode) {
222 LongInfo info = GetLongInfo(word.Morae);
223 var result = new List<string>();
224 for (int i = 0; i < word.Morae.Count; i++) {
225 Mora mora = word.Morae[i];
226 string id = mora.Id, nextId = i+1 < word.Morae.Count ? word.Morae[i+1].Id : "";
227 string next = Regular(nextId) ? Spelling(system,nextId) : "";
228 string part;
229 if (info.Elements[i]) {
230 char v = info.Vowels[i];
231 part = longMode switch {
232 "KEEP" => id == "L" ? "-" : Spelling(system,id),
233 "DROP" => "",
234 "PASSPORT" => v == 'o' && !Begins(next,"aiueo") ? "h" : "",
235 "HYPHEN" => "-",
236 _ => v.ToString()
237 };
238 } else if (Regular(id)) {
239 part = Spelling(system,id);
240 } else if (id == "N") {
241 if (system == "WAPURO") {
242 part = nextId == "N" || Begins(next,"aiueoyn") ? "nn" : "n";
243 } else if (system == "TRADITIONAL" && Begins(next,"bmp")) {
244 part = "m";
245 } else {
246 part = Begins(next,"aiueoy") ? "n'" : "n";
247 }
248 } else {
249 if (next.Length > 0 && !Begins(next,"aiueon")) {
250 part = (system is "HEPBURN" or "TRADITIONAL") && next.StartsWith("ch",StringComparison.Ordinal) ? "t" : next[..1];
251 } else if (system == "WAPURO") {
252 part = "xtu";
253 } else {
254 Column(mora.Column,"small tsu cannot be written");
255 return result;
256 }
257 }
258 result.Add(part);
259 }
260 return result;
261 }
262 string Write(Word word, string system, string longMode, string caseMode) {
263 string result = string.Concat(Parts(word,system,longMode));
264 int pattern = caseMode == "KEEP" ? word.Pattern : caseMode == "LOWER" ? 0 : caseMode == "TITLE" ? 1 : 2;
265 if (pattern == 1) {
266 result = result[..1].ToUpperInvariant()+result[1..];
267 }
268 if (pattern == 2) {
269 result = result.ToUpperInvariant();
270 }
271 return result;
272 }
273 string Canonical(Word word, string system) => Write(word,system,"KEEP","KEEP");
274 static List<string> Ids(Word word) => word.Morae.Select(m => m.Id).ToList();

276 LongInfo info = GetLongInfo(word.Morae);

279 List<Element> Elements(Word word) {
280 var result = new List<Element>();
281 foreach (Mora mora in word.Morae) {
282 string id = mora.Id;
283 if (id == "L") {
284 continue;
285 }
286 if (id == "N") {
287 result.Add(new Element("N",0,0));
288 } else if (id == "Q") {
289 result.Add(new Element("tu",0,1));
290 } else {
291 Meta meta = metadata[id];
292 if (meta.Contracted) {
293 result.Add(new Element(Devoice(meta.Row[..1]+"i"),Voice(meta.Row),0));
294 result.Add(new Element("y"+id[^1],0,1));
295 } else {
296 result.Add(new Element(Devoice(id),Voice(meta.Row),0));
297 }
298 }
299 }
300 return result;
301 }
302 List<Word> Sorted(List<Word> words) {
303 var ranks = (orderSetting == "GOJUON" ? Gojuon : Iroha).Split(' ')
304 .Select((value,index) => (value,index)).ToDictionary(x => x.value,x => x.index);
305 var comparer = Comparer<int[]>.Create((a,b) => a.AsSpan().SequenceCompareTo(b));
306 var keyed = words.Select(word => {
307 var elements = Elements(word);
308 return new {
309 Word = word,
310 Ranks = elements.Select(e => ranks[e.Base]).ToArray(),
311 Voices = elements.Select(e => e.Voice).ToArray(),
312 Sizes = elements.Select(e => e.Size).ToArray(),
313 Longs = word.Morae.Select(m => m.Id == "L" ? 1 : 0).ToArray()
314 };
315 });
316 return keyed.OrderBy(k => k.Ranks,comparer).ThenBy(k => k.Voices,comparer)
317 .ThenBy(k => k.Sizes,comparer).ThenBy(k => k.Longs,comparer).Select(k => k.Word).ToList();
318 }
319 List<string> PatternMorae(string pattern, string system) {
320 if (!Regex.IsMatch(pattern,@"\A[a-z'?*\-]+\z")) {
321 Fail("invalid pattern");
322 }
323 var result = new List<string>();
324 try {
325 foreach (string part in Regex.Split(pattern,"([?*])")) {
326 if (part is "?" or "*") {
327 result.Add(part);
328 } else if (part.Length > 0) {
329 result.AddRange(Parse(new Word(part,0,0,new List<Mora>()),system).Select(m => m.Id));
330 }
331 }
332 } catch (Problem) {
333 Fail("invalid pattern");
334 }
335 return result;
336 }
337 static bool Matches(List<string> pattern, List<string> word) {
338 bool[] dp = new bool[word.Count+1];
339 dp[0] = true;
340 foreach (string token in pattern) {
341 bool[] next = new bool[word.Count+1];
342 next[0] = token == "*" && dp[0];
343 for (int j = 1; j <= word.Count; j++) {
344 next[j] = token == "*" ? dp[j] || next[j-1] : dp[j-1] && (token == "?" || token == word[j-1]);
345 }
346 dp = next;
347 }
348 return dp[word.Count];
349 }
350 string Detect(List<Word> words) {
351 var result = new List<string>();
352 foreach (string system in Systems) {
353 string verdict = "canonical";
354 try {
355 var parsed = words.Select(w => w with { Morae = Parse(w,system) }).ToList();
356 foreach (Word word in parsed) {
357 if (word.Text != Canonical(word,system)) {
358 verdict = "readable";
359 }
360 }
361 } catch (Problem) {
362 verdict = "error";
363 }
364 result.Add(system+"="+verdict);
365 }
366 return string.Join(" ",result);
367 }
368 string Execute(string line) {
369 string command = line.Split(new[] { ' ' },2,StringSplitOptions.None)[0];
370 if (!counts.ContainsKey(command) && command is not "SET" and not "TABLE") {
371 Fail("unknown command "+command);
372 }
373 if (command is "SET" or "TABLE") {
374 string[] tokens = line.Split(' ');
375 if (tokens.Length < 3) {
376 Fail("missing argument");
377 }
378 if (tokens.Length > 3) {
379 Fail("too many arguments");
380 }
381 if (command == "SET") {
382 string[] allowed = tokens[1] switch {
383 "LONG" => new[] { "KEEP","DROP","PASSPORT","HYPHEN","DOUBLE" },
384 "CASE" => new[] { "KEEP","LOWER","UPPER","TITLE" },
385 "ORDER" => new[] { "GOJUON","IROHA" },
386 _ => throw new Problem("unknown option "+tokens[1])
387 };
388 if (!allowed.Contains(tokens[2])) {
389 Fail("unknown value "+tokens[2]);
390 }
391 if (tokens[1] == "LONG") {
392 longSetting = tokens[2];
393 }
394 if (tokens[1] == "CASE") {
395 caseSetting = tokens[2];
396 }
397 if (tokens[1] == "ORDER") {
398 orderSetting = tokens[2];
399 }
400 return tokens[1]+"="+tokens[2];
401 }
402 if (!Systems.Contains(tokens[1])) {
403 Fail("unknown system "+tokens[1]);
404 }
405 if (!Rows.Contains(tokens[2])) {
406 Fail("unknown row "+tokens[2]);
407 }
408 return string.Join(" ",table.Where(m => m.Row == tokens[2]).Select(m => Spelling(tokens[1],m.Id)));
409 }
410 int fixedArgs = counts[command];
411 string[] chunks = line.Split(new[] { ' ' },fixedArgs+2,StringSplitOptions.None);
412 if (chunks.Length < fixedArgs+2) {
413 Fail("missing argument");
414 }
415 string[] arguments = chunks.Skip(1).Take(fixedArgs).ToArray();
416 string text = chunks[fixedArgs+1];
417 int systemCount = command is "NORMALIZE" or "DETECT" ? 0 : command is "CONVERT" or "DIFF" or "SAME" ? 2 : 1;
418 for (int i = 0; i < systemCount; i++) {
419 if (!Systems.Contains(arguments[i])) {
420 Fail("unknown system "+arguments[i]);
421 }
422 }
423 var searchPattern = command == "FIND" ? PatternMorae(arguments[1],arguments[0]) : new List<string>();
424 bool slash = command is "SAME" or "HAIKU";
425 for (int i = 0; i < text.Length; i++) {
426 char c = text[i];
427 if (!Letter(c) && c != '\'' && c != '-' && !" 0123456789.,!?;:()\"".Contains(c) && !(slash && c == '/')) {
428 Column(i+1,"invalid character");
429 }
430 }
431 string[] poem = text.Split('/');
432 if (command == "SAME" && (poem.Length != 2 || Scan(poem[0]).Count == 0 || Scan(poem[1]).Count == 0)) {
433 Fail("expected two texts");
434 }
435 if (command == "HAIKU") {
436 if (poem.Length != 3 && poem.Length != 5) {
437 Fail("expected 3 or 5 lines");
438 }
439 foreach (string part in poem) {
440 if (Scan(part).Count == 0) {
441 Fail("expected 3 or 5 lines");
442 }
443 }
444 }
445 var words = Scan(text);
446 if (words.Count == 0) {
447 Fail("no words");
448 }
449 for (int i = 0; i < words.Count; i++) {
450 int pattern = Pattern(words[i].Text);
451 if (pattern < 0) {
452 Column(words[i].Start+1,"mixed case");
453 }
454 words[i] = words[i] with { Pattern = pattern };
455 }
456 if (command == "DETECT") {
457 return Detect(words);
458 }
459 for (int i = 0; i < words.Count; i++) {
460 Word word = words[i];
461 string source = command == "NORMALIZE" ? "ANY" : command == "SAME" && word.Start > text.IndexOf('/') ? arguments[1] : arguments[0];
462 words[i] = word with { Morae = Parse(word,source) };
463 }
464 string system = command == "NORMALIZE" ? "HEPBURN" : arguments[0];
465 switch (command) {
466 case "CONVERT":
467 case "NORMALIZE":
468 case "REVERSE": {
469 string target = command == "CONVERT" ? arguments[1] : system;
470 var result = new StringBuilder();
471 int cursor = 0;
472 foreach (Word word in words) {
473 result.Append(text[cursor..word.Start]);
474 Word written = word;
475 if (command == "REVERSE") {
476 var reversed = new List<Mora>();
477 for (int i = 0; i < word.Morae.Count; i++) {
478 Mora mora = word.Morae[i];
479 string id = mora.Id == "L" ? word.Morae[i-1].Id[^1].ToString() : mora.Id;
480 reversed.Add(new Mora(id,mora.Column));
481 }
482 reversed.Reverse();
483 written = word with { Morae = reversed };
484 }
485 result.Append(Write(written,target,longSetting,caseSetting));
486 cursor = word.Start+word.Text.Length;
487 }
488 result.Append(text[cursor..]);
489 return result.ToString();
490 }
491 case "VALID": {
492 foreach (Word word in words) {
493 string expected = Canonical(word,system);
494 if (word.Text != expected) {
495 int i = 0;
496 while (i < Math.Min(word.Text.Length,expected.Length) && word.Text[i] == expected[i]) {
497 i++;
498 }
499 return $"INVALID col {word.Start+i+1}: expected {expected}";
500 }
501 }
502 return "VALID";
503 }
504 case "DIFF": {
505 var changes = new List<string>();
506 foreach (Word word in words) {
507 string a = Write(word,system,longSetting,caseSetting);
508 string b = Write(word,arguments[1],longSetting,caseSetting);
509 if (a != b) {
510 changes.Add(a+">"+b);
511 }
512 }
513 return $"{changes.Count} of {words.Count}"+(changes.Count == 0 ? "" : ": "+string.Join(" ",changes));
514 }
515 case "SAME": {
516 var left = words.Where(w => w.Start < text.IndexOf('/')).ToList();
517 var right = words.Where(w => w.Start > text.IndexOf('/')).ToList();
518 bool exact = true;
519 for (int i = 0; i < Math.Min(left.Count,right.Count); i++) {
520 if (!Loose(left[i]).SequenceEqual(Loose(right[i]))) {
521 return $"DIFFERENT at word {i+1}";
522 }
523 if (!Ids(left[i]).SequenceEqual(Ids(right[i]))) {
524 exact = false;
525 }
526 }
527 if (left.Count != right.Count) {
528 return $"DIFFERENT at word {Math.Min(left.Count,right.Count)+1}";
529 }
530 return exact ? "EXACT" : "LOOSE";
531 }
532 case "MORAE": {
533 var result = new List<string>();
534 int count = 0;
535 foreach (Word word in words) {
536 var pieces = word.Morae.Select(m => Regular(m.Id) ? Spelling(system,m.Id) : m.Id == "L" ? "-" : m.Id).ToList();
537 count += pieces.Count;
538 result.Add(string.Join(".",pieces));
539 }
540 return $"{count}: {string.Join(" ",result)}";
541 }
542 case "SYLLABLES": {
543 var result = new List<string>();
544 int count = 0;
545 foreach (Word word in words) {
546 var pieces = Parts(word,system,"KEEP");
547 var syllables = new List<string>();
548 LongInfo info = GetLongInfo(word.Morae);
549 for (int i = 0; i < pieces.Count; i++) {
550 if (i == 0 || Regular(word.Morae[i].Id) && !info.Paired[i]) {
551 syllables.Add(pieces[i]);
552 } else {
553 syllables[^1] += pieces[i];
554 }
555 }
556 count += syllables.Count;
557 result.Add(string.Join(".",syllables));
558 }
559 return $"{count}: {string.Join(" ",result)}";
560 }
561 case "ROW": {
562 var result = new List<string>();
563 foreach (Word word in words) {
564 foreach (Mora mora in word.Morae) {
565 result.Add(metadata.TryGetValue(mora.Id,out Meta? meta) ?
566 $"{Spelling(system,meta.Id)}: {meta.Row} {meta.Dan}" : (mora.Id == "L" ? "-" : mora.Id)+": special");
567 }
568 }
569 return string.Join(", ",result);
570 }
571 case "STATS": {
572 var morae = words.SelectMany(w => w.Morae).ToList();
573 var regular = morae.Where(m => Regular(m.Id)).Select(m => metadata[m.Id]).ToList();
574 var rowCounts = regular.GroupBy(m => BaseRow(m.Row)).ToDictionary(g => g.Key,g => g.Count());
575 var result = new List<string> { $"morae={morae.Count}" };
576 result.AddRange(Rows.Take(10).Where(rowCounts.ContainsKey).Select(row => $"{row}={rowCounts[row]}"));
577 result.AddRange(new[] {
578 $"voiced={regular.Count(m => Voice(m.Row) == 1)}",
579 $"semivoiced={regular.Count(m => Voice(m.Row) == 2)}",
580 $"contracted={regular.Count(m => m.Contracted)}",
581 $"N={morae.Count(m => m.Id == "N")}",
582 $"Q={morae.Count(m => m.Id == "Q")}",
583 $"L={morae.Count(m => m.Id == "L")}"
584 });
585 return string.Join(" ",result);
586 }
587 case "SORT":
588 case "INDEX": {
589 var sorted = Sorted(words);
590 if (command == "SORT") {
591 return string.Join(" ",sorted.Select(w => w.Text));
592 }
593 var groups = new List<string>();
594 var group = new List<string>();
595 string previous = "";
596 foreach (Word word in sorted) {
597 string baseKana = Elements(word)[0].Base;
598 string heading = orderSetting == "IROHA" || baseKana == "N" ? baseKana : metadata[baseKana].Row;
599 if (heading != previous && group.Count > 0) {
600 groups.Add(previous+": "+string.Join(" ",group));
601 group.Clear();
602 }
603 previous = heading;
604 group.Add(word.Text);
605 }
606 groups.Add(previous+": "+string.Join(" ",group));
607 return string.Join("; ",groups);
608 }
609 case "FIND": {
610 var found = words.Where(w => Matches(searchPattern,Ids(w))).Select(w => w.Text).ToList();
611 return $"{found.Count} of {words.Count}"+(found.Count == 0 ? "" : ": "+string.Join(" ",found));
612 }
613 case "HAIKU": {
614 int[] totals = new int[poem.Length];
615 foreach (Word word in words) {
616 totals[text[..word.Start].Count(c => c == '/')] += word.Morae.Count;
617 }
618 int[] expected = poem.Length == 3 ? new[] { 5,7,5 } : new[] { 5,7,5,7,7 };
619 return (totals.SequenceEqual(expected) ? "OK " : "NG ")+(poem.Length == 3 ? "haiku " : "tanka ")+string.Join("-",totals);
620 }
621 case "KAIBUN": {
622 var bases = words.SelectMany(Elements).Select(e => e.Base).ToList();
623 return (bases.SequenceEqual(bases.AsEnumerable().Reverse()) ? "YES " : "NO ")+string.Join(".",bases);
624 }
625 case "SHIRITORI": {
626 var seen = new HashSet<string>(StringComparer.Ordinal);
627 string previous = "";
628 for (int i = 0; i < words.Count; i++) {
629 Word word = words[i];
630 var elements = Elements(word);
631 string head = elements[0].Base, tail = elements[^1].Base;
632 string prefix = $"NG {i+1}: {word.Text}";
633 if (i > 0 && head != previous) {
634 return prefix+" must start with "+Spelling(system,previous);
635 }
636 string key = string.Join(".",Ids(word));
637 if (!seen.Add(key)) {
638 return prefix+" is repeated";
639 }
640 if (tail == "N") {
641 return prefix+" ends with N";
642 }
643 previous = tail;
644 }
645 return $"OK {words.Count}";
646 }
647 default: throw new InvalidOperationException("Unreachable command");
648 }
649 }
