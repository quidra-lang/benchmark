69local function spelling(system, id)
70 if system == "HEPBURN" or system == "TRADITIONAL" or system == "WAPURO" then
71 local changes = {si = "shi", ti = "chi", tu = "tsu", hu = "fu", zi = "ji"}
72 if system ~= "WAPURO" then
73 changes.di, changes.du, changes.wi, changes.we, changes.wo = "ji", "zu", "i", "e", "o"
74 end
75 if changes[id] then
76 return changes[id]
77 end
78 local prefix = id:sub(1, 2)
79 if prefix == "sy" then
80 return "sh" .. id:sub(3)
81 elseif prefix == "ty" then
82 return "ch" .. id:sub(3)
83 elseif prefix == "zy" or (system ~= "WAPURO" and prefix == "dy") then
84 return "j" .. id:sub(3)
85 end
86 elseif system == "KUNREI" then
87 local changes = {di = "zi", du = "zu", wi = "i", we = "e", wo = "o", dya = "zya", dyu = "zyu", dyo = "zyo"}
88 return changes[id] or id
89 end
90 return id
91end

115local function match_syllable(word, position, system)
116 if position > #word then
117 return nil
118 end

127local function parse_word(text, start, system)
128 local word = text:lower()
129 local ids, columns = {}, {}
130 local p = 1
131 while p <= #word do
132 local c, c2, c3 = word:sub(p, p), word:sub(p + 1, p + 1), word:sub(p + 2, p + 2)
133 local matched = match_syllable(word, p, system)
134 local id, advance = nil, 1
135 if c == "-" then
136 if #ids == 0 or not is_syllabic(ids[#ids]) then
137 column_error(start + p - 1, "misplaced long mark")
138 end
139 id = "-"
140 elseif c == "'" then
141 column_error(start + p - 1, "misplaced apostrophe")
142 elseif matched then
143 id, advance = matched.id, #matched.text
144 elseif (system == "WAPURO" or system == "NORMAL") and word:sub(p, p + 2) == "xtu" then
145 id, advance = "Q", 3
146 elseif c == "n" then
147 id = "N"
148 if system == "WAPURO" then
149 if c2 == "n" then
150 advance = 2
151 end
152 elseif system == "NORMAL" then
153 if c2 == "'" or (c2 == "n" and not contains({"a", "i", "u", "e", "o", "y", "'"}, c3)) then
154 advance = 2
155 end
156 elseif c2 == "'" then
157 advance = 2
158 end
159 elseif (system == "TRADITIONAL" or system == "NORMAL") and c == "m" and contains({"b", "m", "p"}, c2) then
160 id = "N"
161 elseif c:match("^[a-z]$") and not contains({"a", "i", "u", "e", "o", "n"}, c) and c == c2 and match_syllable(word, p + 1, system) then
162 id = "Q"
163 elseif contains({"HEPBURN", "TRADITIONAL", "NORMAL"}, system) and c == "t" and c2 == "c" and c3 == "h" and match_syllable(word, p + 1, system) then
164 id = "Q"
165 else
166 column_error(start + p - 1, "unknown syllable")
167 end
168 ids[#ids + 1] = id
169 columns[#columns + 1] = start + p - 1
170 p = p + advance
171 end
172 return ids, columns
173end

175local function long_flags(ids)
176 local paired, long = {}, {}
177 for i, id in ipairs(ids) do
178 paired[i], long[i] = false, id == "-"
179 end
180 local i = 1
181 while i < #ids do
182 local first, second = ids[i], ids[i + 1]
183 local pair = false
184 if is_syllabic(first) and is_vowel(second) then
185 local vowel = first:sub(-1)
186 if vowel == second or (vowel == "o" and second == "u") or (vowel == "e" and second == "i") then
187 paired[i + 1] = true
188 long[i + 1] = (vowel == "o" and (second == "o" or second == "u")) or (vowel == "u" and second == "u")
189 pair = true
190 end
191 end
192 i = i + (pair and 2 or 1)
193 end
194 return paired, long
195end

197local function write_parts(ids, columns, system, long_setting)
198 local _, long = long_flags(ids)
199 local parts = {}
200 for i, id in ipairs(ids) do
201 local next_id = ids[i + 1] or ""
202 local next_text = is_syllabic(next_id) and spelling(system, next_id) or ""
203 local first = next_text:sub(1, 1)
204 local part
205 if long[i] then
206 local vowel = ids[i - 1]:sub(-1)
207 if long_setting == "KEEP" then
208 part = id == "-" and "-" or spelling(system, id)
209 elseif long_setting == "DROP" then
210 part = ""
211 elseif long_setting == "PASSPORT" then
212 part = vowel == "o" and not is_vowel(first) and "h" or ""
213 elseif long_setting == "HYPHEN" then
214 part = "-"
215 else
216 part = vowel
217 end
218 elseif is_syllabic(id) then
219 part = spelling(system, id)
220 elseif id == "N" then
221 if system == "WAPURO" then
222 part = (next_id == "N" or contains({"a", "i", "u", "e", "o", "y", "n"}, first)) and "nn" or "n"
223 elseif system == "TRADITIONAL" and contains({"b", "m", "p"}, first) then
224 part = "m"
225 else
226 part = contains({"a", "i", "u", "e", "o", "y"}, first) and "n'" or "n"
227 end
228 else
229 if next_text ~= "" and not contains({"a", "i", "u", "e", "o", "n"}, first) then
230 part = (system == "HEPBURN" or system == "TRADITIONAL") and next_text:sub(1, 2) == "ch" and "t" or first
231 elseif system == "WAPURO" then
232 part = "xtu"
233 else
234 column_error(columns[i], "small tsu cannot be written")
235 end
236 end
237 parts[i] = part
238 end
239 return parts
240end

242local function word_pattern(text)
243 if text == text:lower() then
244 return "LOWER"
245 elseif text:sub(1, 1):match("^[A-Z]$") and text:sub(2) == text:sub(2):lower() then
246 return "TITLE"
247 end
248 local _, count = text:gsub("[A-Z]", "")
249 if count >= 2 and text == text:upper() then
250 return "UPPER"
251 end
252 return ""
253end

255local function written(word, system, settings, canonical, ids, columns)
256 local mode = canonical and "KEEP" or settings.LONG
257 local text = table.concat(write_parts(ids or word.ids, columns or word.columns, system, mode))
258 local pattern = word.pattern
259 if not canonical and settings.CASE ~= "KEEP" then
260 pattern = settings.CASE
261 end
262 if pattern == "UPPER" then
263 return text:upper()
264 elseif pattern == "TITLE" then
265 return text:sub(1, 1):upper() .. text:sub(2)
266 end
267 return text
268end

332local function collation(word, order)
333 local ranks = {}
334 for i, id in ipairs(order == "GOJUON" and gojuon or iroha) do
335 ranks[id] = i
336 end
337 local key = {{}, {}, {}, {}}
338 for i, item in ipairs(elements(word.ids)) do

340 end
341 for i, id in ipairs(word.ids) do
342 key[4][i] = id == "-" and 1 or 0

347local function vector_compare(a, b)
348 for i = 1, math.min(#a, #b) do
349 if a[i] ~= b[i] then
350 return a[i] < b[i] and -1 or 1
351 end
352 end
353 if #a == #b then
354 return 0
355 end
356 return #a < #b and -1 or 1
357end

359local function pattern_tokens(pattern, system)
360 if pattern:find("[^a-z'%-?*]") then
361 fail("invalid pattern")
362 end
363 local tokens, chunk = {}, ""
364 local function flush()
365 local valid, ids = pcall(parse_word, chunk, 1, system)
366 if not valid then
367 fail("invalid pattern")
368 end
369 table.move(ids,1,#ids,#tokens+1,tokens)
370 chunk = ""
371 end
372 for i = 1, #pattern do
373 local c = pattern:sub(i, i)
374 if c == "?" or c == "*" then
375 flush()
376 tokens[#tokens + 1] = c
377 else
378 chunk = chunk .. c
379 end
380 end
381 flush()
382 return tokens
383end

385local function wildcard_matches(tokens, ids)
386 local previous = {[0] = true}
387 for j = 1, #ids do
388 previous[j] = false
389 end
390 for _, token in ipairs(tokens) do
391 local current = {[0] = token == "*" and previous[0] or false}
392 for j, id in ipairs(ids) do
393 if token == "*" then
394 current[j] = previous[j] or current[j - 1]
395 else
396 current[j] = previous[j - 1] and (token == "?" or token == id)
397 end
398 end
399 previous = current
400 end
401 return previous[#ids]
402end

404local function equal_ids(a, b)
405 return vector_compare(a, b) == 0
406end

408local function execute(line, settings)
409 local command = line:match("^[^ ]+")
410 local k = arg_counts[command]
411 if k == nil then
412 fail("unknown command " .. command)
413 end
414 local args, text = {}, ""
415 if command == "SET" or command == "TABLE" then
416 local parts = split_words(line)
417 if #parts < 3 then
418 fail("missing argument")
419 elseif #parts > 3 then
420 fail("too many arguments")
421 end
422 args = {parts[2], parts[3]}
423 else
424 if count_character(line, " ") < k + 1 then
425 fail("missing argument")
426 end
427 local position = #command + 2
428 for i = 1, k do
429 local ending = line:find(" ", position, true)
430 args[i] = line:sub(position, ending - 1)
431 position = ending + 1
432 end
433 text = line:sub(position)
434 end
435 local system_count = 1
436 if command == "SET" or command == "NORMALIZE" or command == "DETECT" then
437 system_count = 0
438 elseif command == "CONVERT" or command == "DIFF" or command == "SAME" then
439 system_count = 2
440 end
441 for i = 1, system_count do
442 if not contains(systems, args[i]) then
443 fail("unknown system " .. args[i])
444 end
445 end
446 if command == "SET" then
447 local option, value = table.unpack(args)
448 if not set_values[option] then
449 fail("unknown option " .. option)
450 elseif not contains(set_values[option], value) then
451 fail("unknown value " .. value)
452 end
453 settings[option] = value
454 return option .. "=" .. value
455 elseif command == "TABLE" then
456 local system, row = table.unpack(args)
457 if not contains(rows, row) then
458 fail("unknown row " .. row)
459 end
460 local output = {}
461 for _, id in ipairs(identifiers) do
462 if row_of[id] == row then
463 output[#output + 1] = spelling(system, id)
464 end
465 end
466 return table.concat(output, " ")
467 end
468 local tokens = command == "FIND" and pattern_tokens(args[2], args[1]) or {}
469 local slash_allowed = command == "HAIKU" or command == "SAME"
470 for i = 1, #text do
471 local c = text:sub(i, i)
472 local allowed = c:match("^[A-Za-z0-9]$") or ("'- .,!?;:()\""):find(c, 1, true) or (slash_allowed and c == "/")
473 if not allowed then
474 column_error(i, "invalid character")
475 end
476 end
477 local words = words_in(text)
478 if command == "SAME" or command == "HAIKU" then
479 local parts = count_character(text, "/") + 1
480 local expected = command == "SAME" and "expected two texts" or "expected 3 or 5 lines"
481 if (command == "SAME" and parts ~= 2) or (command == "HAIKU" and parts ~= 3 and parts ~= 5) then
482 fail(expected)
483 end
484 for i = 1, parts do
485 local found = false
486 for _, word in ipairs(words) do
487 if word.part == i then
488 found = true
489 end
490 end
491 if not found then
492 fail(expected)
493 end
494 end
495 end
496 if #words == 0 then
497 fail("no words")
498 end
499 for _, word in ipairs(words) do
500 word.pattern = word_pattern(word.text)
501 if word.pattern == "" then
502 column_error(word.start, "mixed case")
503 end
504 end
505 if command == "DETECT" then
506 local output = {}
507 for _, system in ipairs(systems) do
508 local verdict = "canonical"
509 for _, word in ipairs(words) do
510 local valid, ids, columns = pcall(parse_word, word.text, word.start, system)
511 if not valid then
512 verdict = "error"
513 break
514 end
515 word.ids, word.columns = ids, columns
516 end
517 if verdict ~= "error" then
518 for _, word in ipairs(words) do
519 if written(word, system, settings, true) ~= word.text then
520 verdict = "readable"
521 end
522 end
523 end
524 output[#output + 1] = system .. "=" .. verdict
525 end
526 return table.concat(output, " ")
527 end
528 for _, word in ipairs(words) do
529 local system = args[1]
530 if command == "NORMALIZE" then
531 system = "NORMAL"
532 elseif command == "SAME" then
533 system = args[word.part]
534 end
535 word.ids, word.columns = parse_word(word.text, word.start, system)
536 end
537 local system = command == "NORMALIZE" and "HEPBURN" or args[1]
538 if command == "CONVERT" or command == "NORMALIZE" or command == "REVERSE" then
539 local target = command == "CONVERT" and args[2] or system
540 local output = {}
541 for i, word in ipairs(words) do
542 local ids, columns = word.ids, word.columns
543 if command == "REVERSE" then
544 ids, columns = {}, {}
545 for j = #word.ids, 1, -1 do
546 local id = word.ids[j]
547 ids[#ids + 1] = id == "-" and word.ids[j - 1]:sub(-1) or id
548 columns[#columns + 1] = word.columns[j]
549 end
550 end
551 output[i] = written(word, target, settings, false, ids, columns)
552 end
553 return replace_words(text, words, output)
554 elseif command == "VALID" then
555 for _, word in ipairs(words) do
556 local expected = written(word, system, settings, true)
557 if expected ~= word.text then
558 local i = 1
559 while i <= math.min(#expected, #word.text) and expected:sub(i, i) == word.text:sub(i, i) do
560 i = i + 1
561 end
562 return "INVALID col " .. (word.start + i - 1) .. ": expected " .. expected
563 end
564 end
565 return "VALID"
566 elseif command == "DIFF" then
567 local output = {}
568 for _, word in ipairs(words) do
569 local a = written(word, args[1], settings)
570 local b = written(word, args[2], settings)
571 if a ~= b then
572 output[#output + 1] = a .. ">" .. b
573 end
574 end
575 local prefix = #output .. " of " .. #words
576 return #output == 0 and prefix or prefix .. ": " .. table.concat(output, " ")
577 elseif command == "SAME" then
578 local left, right = {}, {}
579 for _, word in ipairs(words) do
580 local side = word.part == 1 and left or right
581 side[#side + 1] = word
582 end
583 local exact = true
584 for i = 1, math.min(#left, #right) do
585 local a, b = left[i].ids, right[i].ids
586 local _, a_long = long_flags(a)
587 local _, b_long = long_flags(b)
588 local a_loose, b_loose = {}, {}
589 for j, id in ipairs(a) do
590 if not a_long[j] then
591 a_loose[#a_loose + 1] = id
592 end
593 end
594 for j, id in ipairs(b) do
595 if not b_long[j] then
596 b_loose[#b_loose + 1] = id
597 end
598 end
599 if not equal_ids(a_loose, b_loose) then
600 return "DIFFERENT at word " .. i
601 end
602 exact = exact and equal_ids(a, b)
603 end
604 if #left ~= #right then
605 return "DIFFERENT at word " .. (math.min(#left, #right) + 1)
606 end
607 return exact and "EXACT" or "LOOSE"
608 elseif command == "MORAE" or command == "SYLLABLES" then
609 local output, total = {}, 0
610 for _, word in ipairs(words) do
611 local parts = {}
612 if command == "MORAE" then
613 for i, id in ipairs(word.ids) do
614 parts[i] = spelling(system, id)
615 end
616 else
617 local paired = long_flags(word.ids)
618 local written_parts = write_parts(word.ids, word.columns, system, "KEEP")
619 for i, id in ipairs(word.ids) do
620 if i > 1 and (not is_syllabic(id) or paired[i]) then
621 parts[#parts] = parts[#parts] .. written_parts[i]
622 else
623 parts[#parts + 1] = written_parts[i]
624 end
625 end
626 end
627 total = total + #parts
628 output[#output + 1] = table.concat(parts, ".")
629 end
630 return total .. ": " .. table.concat(output, " ")
631 elseif command == "ROW" then
632 local output = {}
633 for _, word in ipairs(words) do
634 for _, id in ipairs(word.ids) do
635 if is_syllabic(id) then
636 output[#output + 1] = spelling(system, id) .. ": " .. row_of[id] .. " " .. dan_of[id]
637 else
638 output[#output + 1] = id .. ": special"
639 end
640 end
641 end
642 return table.concat(output, ", ")
643 elseif command == "STATS" then
644 local counts, total = {}, 0
645 for i = 1, 10 do
646 counts[rows[i]] = 0
647 end
648 for _, name in ipairs(split_words("voiced semivoiced contracted N Q L")) do
649 counts[name] = 0
650 end
651 for _, word in ipairs(words) do
652 for _, id in ipairs(word.ids) do
653 total = total + 1
654 if is_syllabic(id) then
655 local row = row_of[id]
656 local base = base_row[row] or row
657 counts[base] = counts[base] + 1
658 if contains({"ga", "za", "da", "ba"}, row) then
659 counts.voiced = counts.voiced + 1
660 elseif row == "pa" then
661 counts.semivoiced = counts.semivoiced + 1
662 end
663 if #dan_of[id] == 2 then
664 counts.contracted = counts.contracted + 1
665 end
666 else
667 local key = id == "-" and "L" or id
668 counts[key] = counts[key] + 1
669 end
670 end
671 end
672 local output = {"morae=" .. total}
673 for i = 1, 10 do
674 local row = rows[i]
675 if counts[row] > 0 then
676 output[#output + 1] = row .. "=" .. counts[row]
