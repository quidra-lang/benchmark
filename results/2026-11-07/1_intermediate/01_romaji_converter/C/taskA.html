48static bool letter(char c) {

51static bool vowel(char c) { return c && strchr("aiueo", c); }
52static bool word_char(char c) { return letter(c) || c == '\'' || c == '-'; }
53static char lower(char c) { return c >= 'A' && c <= 'Z' ? c + ('a' - 'A') : c; }
54static char upper(char c) { return c >= 'a' && c <= 'z' ? c - ('a' - 'A') : c; }

139static int lex(const char *text, Word *words) {
140 int count = 0, section = 0;
141 for (int i = 0; text[i];) {
142 if (!word_char(text[i])) {
143 if (text[i] == '/') ++section;
144 ++i;
145 continue;
146 }
147 Word *w = &words[count++];
148 memset(w, 0, sizeof *w);
149 w->start = i;
150 w->section = section;
151 while (word_char(text[i])) ++i;
152 w->end = i;
153 memcpy(w->original, text + w->start, i - w->start);
154 }
155 return count;
156}
157static void check_case(Word *w) {
158 bool all_lower = true, all_upper = true;
159 bool title = w->original[0] >= 'A' && w->original[0] <= 'Z';
160 int letters = 0;
161 for (int i = 0; w->original[i]; ++i) {
162 char c = w->original[i];
163 if (!letter(c)) continue;
164 ++letters;
165 if (c != lower(c)) all_lower = false;
166 if (c != upper(c)) all_upper = false;
167 if (i && c != lower(c)) title = false;
168 }
169 if (all_lower) w->pattern = 0;
170 else if (title) w->pattern = 1;
171 else if (letters >= 2 && all_upper) w->pattern = 2;
172 else column_error(w->start + 1, "mixed case");
173}
174static int match(const char *s, int pos, int system, int *length) {
175 int id = -1;
176 *length = 0;
177 for (int i = 0; i < N; ++i) {
178 for (int t = 0; t < 5; ++t) {
179 if (system >= 0 && system != t) continue;
180 const char *p = table[i].spelling[t];
181 int n = (int)strlen(p);
182 if (n > *length && !strncmp(s + pos, p, n)) {
183 id = i;
184 *length = n;
185 }
186 }
187 }
188 return id;
189}
190static bool parse(Word *w, int system) {
191 char s[CAP];
192 strcpy(s, w->original);
193 int n = (int)strlen(s);
194 for (int i = 0; i < n; ++i) s[i] = lower(s[i]);
195 w->count = 0;
196 for (int p = 0; p < n;) {
197 char c = s[p], c2 = s[p + 1];
198 int id = -1, advance = 1, length;
199 if (c == '-') {
200 if (!w->count || w->tokens[w->count - 1].id >= N) return column_problem(w->start + p + 1, "misplaced long mark");
201 id = L;
202 } else if (c == '\'') return column_problem(w->start + p + 1, "misplaced apostrophe");
203 else {
204 id = match(s, p, system, &length);
205 if (id >= 0) advance = length;
206 else if ((system == 4 || system < 0) && !strncmp(s + p, "xtu", 3)) {
207 id = Q;
208 advance = 3;
209 } else if (c == 'n') {
210 id = N;
211 if (system < 0) {
212 char c3 = p + 2 < n ? s[p + 2] : '\0';
213 if (c2 == '\'' || (c2 == 'n' && !vowel(c3) && c3 != 'y' && c3 != '\'')) advance = 2;
214 } else if ((system == 4 && c2 == 'n') || (system != 4 && c2 == '\'')) advance = 2;
215 } else if ((system == 1 || system < 0) && c == 'm' && c2 && strchr("bmp", c2)) id = N;
216 else if (letter(c) && !vowel(c) && c != 'n' && c2 == c && match(s, p + 1, system, &length) >= 0) id = Q;
217 else if (system <= 1 && c == 't' && !strncmp(s + p + 1, "ch", 2) && match(s, p + 1, system, &length) >= 0) id = Q;
218 else return column_problem(w->start + p + 1, "unknown syllable");
219 }
220 w->tokens[w->count++] = (Token){id, w->start + p + 1};
221 p += advance;
222 }
223 return true;
224}
225static void pairs(const Word *w, bool *result, bool long_only) {
226 memset(result, 0, CAP * sizeof *result);
227 for (int i = 0; i + 1 < w->count;) {
228 int a = w->tokens[i].id, b = w->tokens[i + 1].id;
229 if (a < N && b < 5) {
230 char x = table[a].id[strlen(table[a].id) - 1], y = table[b].id[0];
231 if (x == y || (x == 'o' && y == 'u') || (x == 'e' && y == 'i')) {
232 result[i + 1] = !long_only || (x == 'o' && (y == 'o' || y == 'u')) || (x == 'u' && y == 'u');
233 i += 2;
234 continue;
235 }
236 }
237 ++i;
238 }
239 if (long_only) {
240 for (int i = 0; i < w->count; ++i) if (w->tokens[i].id == L) result[i] = true;
241 }
242}
243static void parts(const Word *w, int system, int mode, char result[CAP][8]) {
244 bool long_elements[CAP];
245 pairs(w, long_elements, true);
246 for (int i = 0; i < w->count; ++i) {
247 char *s = result[i];
248 s[0] = '\0';
249 int id = w->tokens[i].id, next_id = i + 1 < w->count ? w->tokens[i + 1].id : -1;
250 const char *next = next_id >= 0 && next_id < N ? table[next_id].spelling[system] : "";
251 char first = next[0];
252 if (long_elements[i]) {
253 const char *previous = table[w->tokens[i - 1].id].id;
254 char v = previous[strlen(previous) - 1];
255 if (mode == 0) strcpy(s, id == L ? "-" : table[id].spelling[system]);
256 if (mode == 2 && v == 'o' && !vowel(first)) strcpy(s, "h");
257 if (mode == 3) strcpy(s, "-");
258 if (mode == 4) {
259 s[0] = v;
260 s[1] = '\0';
261 }
262 } else if (id < N) strcpy(s, table[id].spelling[system]);
263 else if (id == N) {
264 if (system == 4) strcpy(s, next_id == N || vowel(first) || first == 'y' || first == 'n' ? "nn" : "n");
265 else if (system == 1 && first && strchr("bmp", first)) strcpy(s, "m");
266 else strcpy(s, vowel(first) || first == 'y' ? "n'" : "n");
267 } else if (id == Q) {
268 if (first && !vowel(first) && first != 'n') {
269 s[0] = system <= 1 && !strncmp(next, "ch", 2) ? 't' : first;
270 s[1] = '\0';
271 } else if (system == 4) strcpy(s, "xtu");
272 else column_error(w->tokens[i].col, "small tsu cannot be written");
273 }
274 }
275}
276static void write_word(const Word *w, int system, bool canonical, char *output) {
277 char pieces[CAP][8];
278 parts(w, system, canonical ? 0 : long_mode, pieces);
279 output[0] = '\0';
280 for (int i = 0; i < w->count; ++i) strcat(output, pieces[i]);
281 int pattern = canonical || case_mode == 0 ? w->pattern : case_mode == 1 ? 0 : case_mode == 2 ? 2 : 1;
282 if (pattern == 1) output[0] = upper(output[0]);
283 if (pattern == 2) for (int i = 0; output[i]; ++i) output[i] = upper(output[i]);
284}
285static bool same_sequence(const Word *a, const Word *b, bool loose) {
286 bool omit_a[CAP], omit_b[CAP];
287 pairs(a, omit_a, true);
288 pairs(b, omit_b, true);
289 int i = 0, j = 0;
290 for (;;) {
291 while (loose && i < a->count && omit_a[i]) ++i;
292 while (loose && j < b->count && omit_b[j]) ++j;
293 if (i == a->count || j == b->count) return i == a->count && j == b->count;
294 if (a->tokens[i++].id != b->tokens[j++].id) return false;
295 }
296}
297static int elements(const Word *w, Element *result) {
298 int count = 0;
299 for (int i = 0; i < w->count; ++i) {
300 int id = w->tokens[i].id;
301 if (id == L) continue;
302 if (id == N) result[count++] = (Element){N, 0, 0};
303 else if (id == Q) result[count++] = (Element){mora_id("tu"), 0, 1};
304 else {
305 const Mora *m = &table[id];
306 char base[5];
307 strcpy(base, m->id);
308 if (m->voice) base[0] = "ksthh"[m->row - 10];
309 if (strlen(m->dan) == 2) {
310 base[1] = 'i';
311 base[2] = '\0';
312 result[count++] = (Element){mora_id(base), m->voice, 0};
313 result[count++] = (Element){mora_id(m->dan), 0, 1};
314 } else result[count++] = (Element){mora_id(base), m->voice, 0};
315 }
316 }
317 return count;
318}
319static int rank_of(int id) {
320 const char *order = order_mode == 0
321 ? "a i u e o ka ki ku ke ko sa si su se so ta ti tu te to na ni nu ne no ha hi hu he ho ma mi mu me mo ya yu yo ra ri ru re ro wa wi we wo N"
322 : "i ro ha ni ho he to ti ri nu ru wo wa ka yo ta re so tu ne na ra mu u wi no o ku ya ma ke hu ko e te a sa ki yu me mi si we hi mo se su N";
323 const char *name = id == N ? "N" : table[id].id;
324 int rank = 0;
325 for (const char *p = order; *p;) {
326 const char *end = strchr(p, ' ');
327 int n = end ? (int)(end - p) : (int)strlen(p);
328 if ((int)strlen(name) == n && !strncmp(p, name, n)) return rank;
329 ++rank;
330 if (!end) break;
331 p = end + 1;
332 }
333 return -1;
334}
335static int compare_words(const void *left, const void *right) {
336 const Word *a = left, *b = right;
337 Element ea[CAP * 2], eb[CAP * 2];
338 int na = elements(a, ea), nb = elements(b, eb);
339 for (int field = 0; field < 3; ++field) {
340 for (int i = 0; i < na && i < nb; ++i) {
341 int x = field == 0 ? rank_of(ea[i].id) : field == 1 ? ea[i].voice : ea[i].size;
342 int y = field == 0 ? rank_of(eb[i].id) : field == 1 ? eb[i].voice : eb[i].size;
343 if (x != y) return x < y ? -1 : 1;
344 }
345 if (na != nb) return na < nb ? -1 : 1;
346 }
347 for (int i = 0; i < a->count && i < b->count; ++i) {
348 int x = a->tokens[i].id == L, y = b->tokens[i].id == L;
349 if (x != y) return x < y ? -1 : 1;
350 }
351 if (a->count != b->count) return a->count < b->count ? -1 : 1;
352 return a->start < b->start ? -1 : a->start > b->start;
353}
354static int pattern_tokens(const char *pattern, int system, int *result) {
355 int count = 0, start = 0, n = (int)strlen(pattern);
356 for (int i = 0; i <= n; ++i) {
357 char c = pattern[i];
358 if (!c || c == '?' || c == '*') {
359 if (i > start) {
360 Word w = {0};
361 memcpy(w.original, pattern + start, i - start);
362 if (!parse(&w, system)) fail("invalid pattern");
363 for (int j = 0; j < w.count; ++j) result[count++] = w.tokens[j].id;
364 }
365 if (c) result[count++] = c == '?' ? -1 : -2;
366 start = i + 1;
367 } else if (!((c >= 'a' && c <= 'z') || c == '\'' || c == '-')) fail("invalid pattern");
368 }
369 return count;
370}
371static bool glob(const int *pattern, int length, const Word *w) {
372 bool dp[CAP] = {true};
373 for (int i = 0; i < length; ++i) {
374 bool next[CAP] = {false};
375 if (pattern[i] == -2) {
376 next[0] = dp[0];
377 for (int j = 1; j <= w->count; ++j) next[j] = dp[j] || next[j - 1];
378 } else {
379 for (int j = 1; j <= w->count; ++j) next[j] = dp[j - 1] && (pattern[i] == -1 || pattern[i] == w->tokens[j - 1].id);
380 }
381 memcpy(dp, next, sizeof dp);
382 }
383 return dp[w->count];
384}
385static void process(const char *line, Buffer *out) {
386 const char *names[] = {"SET", "TABLE", "CONVERT", "NORMALIZE", "REVERSE", "VALID", "DETECT", "DIFF", "SAME", "MORAE", "SYLLABLES", "ROW", "STATS", "SORT", "INDEX", "FIND", "HAIKU", "KAIBUN", "SHIRITORI"};
387 const int fixed[] = {2, 2, 2, 0, 1, 1, 0, 2, 2, 1, 1, 1, 1, 1, 1, 2, 1, 1, 1};
388 char command[CAP], args[2][CAP] = {{0}}, text[CAP] = {0};
389 const char *space = strchr(line, ' ');
390 int name_length = space ? (int)(space - line) : (int)strlen(line);
391 memcpy(command, line, name_length);
392 command[name_length] = '\0';
393 int cmd = lookup(names, 19, command);
394 if (cmd < 0) named_error("unknown command", command);
395 if (cmd <= 1) {
396 int spaces = 0;
397 for (const char *p = line; *p; ++p) if (*p == ' ') ++spaces;
398 if (spaces < 2) fail("missing argument");
399 if (spaces > 2) fail("too many arguments");
400 const char *second = strchr(space + 1, ' ');
401 memcpy(args[0], space + 1, second - space - 1);
402 strcpy(args[1], second + 1);
403 } else {
404 const char *p = line;
405 for (int i = 0; i <= fixed[cmd]; ++i) {
406 const char *end = strchr(p, ' ');
407 if (!end) fail("missing argument");
408 if (i) memcpy(args[i - 1], p, end - p);
409 p = end + 1;
410 }
411 strcpy(text, p);
412 }
413 if (cmd == 0) {
414 const char *options[] = {"LONG", "CASE", "ORDER"};
415 const char *values[3][5] = {{"KEEP", "DROP", "PASSPORT", "HYPHEN", "DOUBLE"}, {"KEEP", "LOWER", "UPPER", "TITLE"}, {"GOJUON", "IROHA"}};
416 const int sizes[] = {5, 4, 2};
417 int option = lookup(options, 3, args[0]);
418 if (option < 0) named_error("unknown option", args[0]);
419 int value = lookup(values[option], sizes[option], args[1]);
420 if (value < 0) named_error("unknown value", args[1]);
421 if (option == 0) long_mode = value;
422 if (option == 1) case_mode = value;
423 if (option == 2) order_mode = value;
424 append(out, args[0]);
425 append(out, "=");
426 append(out, args[1]);
427 return;
428 }
429 int system_count = cmd == 3 || cmd == 6 ? 0 : cmd == 2 || cmd == 7 || cmd == 8 ? 2 : 1;
430 int sys[2] = {-1, -1};
431 for (int i = 0; i < system_count; ++i) {
432 sys[i] = lookup(systems, 5, args[i]);
433 if (sys[i] < 0) named_error("unknown system", args[i]);
434 }
435 int source = sys[0];
436 if (cmd == 1) {
437 int row = lookup(rows, 15, args[1]);
438 if (row < 0) named_error("unknown row", args[1]);
439 for (int i = 0; i < N; ++i) if (table[i].row == row) item(out, table[i].spelling[source], " ");
440 return;
441 }
442 int pattern[CAP], pattern_length = 0;
443 if (cmd == 15) pattern_length = pattern_tokens(args[1], source, pattern);
444 for (int i = 0; text[i]; ++i) {
445 char c = text[i];
446 if (!word_char(c) && !(c >= '0' && c <= '9') && !strchr(" .,!?;:()\"", c) && !((cmd == 8 || cmd == 16) && c == '/')) column_error(i + 1, "invalid character");
447 }
448 Word words[CAP];
449 int count = lex(text, words), sections = 1, section_words[CAP] = {0};
450 for (int i = 0; text[i]; ++i) if (text[i] == '/') ++sections;
451 for (int i = 0; i < count; ++i) ++section_words[words[i].section];
452 if (cmd == 8 && (sections != 2 || !section_words[0] || !section_words[1])) fail("expected two texts");
453 if (cmd == 16) {
454 if (sections != 3 && sections != 5) fail("expected 3 or 5 lines");
455 for (int i = 0; i < sections; ++i) if (!section_words[i]) fail("expected 3 or 5 lines");
456 }
457 if (!count) fail("no words");
458 for (int i = 0; i < count; ++i) check_case(&words[i]);
459 if (cmd == 6) {
460 for (int s = 0; s < 5; ++s) {
461 bool readable = true, canonical = true;
462 for (int i = 0; i < count; ++i) {
463 if (!parse(&words[i], s)) {
464 readable = false;
465 break;
466 }
467 }
468 if (readable) {
469 for (int i = 0; i < count; ++i) {
470 char written[CAP * 4];
471 write_word(&words[i], s, true, written);
472 if (strcmp(written, words[i].original)) canonical = false;
473 }
474 }
475 if (s) append(out, " ");
476 append(out, systems[s]);
477 append(out, "=");
478 append(out, !readable ? "error" : canonical ? "canonical" : "readable");
479 }
480 return;
481 }
482 for (int i = 0; i < count; ++i) {
483 if (!parse(&words[i], cmd == 8 && words[i].section == 1 ? sys[1] : source)) fail(error_message);
484 }
485 if (cmd == 2 || cmd == 3 || cmd == 4) {
486 int pos = 0;
487 for (int j = 0; j < count; ++j) {
488 Word *w = &words[j];
489 char separators[CAP] = {0}, written[CAP * 4];
490 memcpy(separators, text + pos, w->start - pos);
491 append(out, separators);
492 if (cmd == 4) {
493 for (int i = 0; i < w->count; ++i) {
494 if (w->tokens[i].id == L) {
495 const char *id = table[w->tokens[i - 1].id].id;
496 char v[] = {id[strlen(id) - 1], '\0'};
497 w->tokens[i].id = mora_id(v);
498 }
499 }
500 for (int i = 0; i < w->count / 2; ++i) {
501 Token t = w->tokens[i];
502 w->tokens[i] = w->tokens[w->count - 1 - i];
503 w->tokens[w->count - 1 - i] = t;
504 }
505 }
506 write_word(w, cmd == 2 ? sys[1] : cmd == 3 ? 0 : source, false, written);
507 append(out, written);
508 pos = w->end;
509 }
510 append(out, text + pos);
511 return;
512 }
513 if (cmd == 5) {
514 for (int i = 0; i < count; ++i) {
515 char canonical[CAP * 4];
516 write_word(&words[i], source, true, canonical);
517 if (strcmp(canonical, words[i].original)) {
518 int p = 0;
519 while (canonical[p] && words[i].original[p] && canonical[p] == words[i].original[p]) ++p;
520 append(out, "INVALID col ");
521 number(out, words[i].start + p + 1);
522 append(out, ": expected ");
523 append(out, canonical);
524 return;
525 }
526 }
527 append(out, "VALID");
528 return;
529 }
530 if (cmd == 7 || cmd == 15) {
531 Buffer selected = {0};
532 int matches = 0;
533 for (int i = 0; i < count; ++i) {
534 if (cmd == 7) {
535 char a[CAP * 4], b[CAP * 4];
536 write_word(&words[i], source, false, a);
537 write_word(&words[i], sys[1], false, b);
538 if (strcmp(a, b)) {
539 if (matches++) append(&selected, " ");
540 append(&selected, a);
541 append(&selected, ">");
542 append(&selected, b);
543 }
544 } else if (glob(pattern, pattern_length, &words[i])) {
545 if (matches++) append(&selected, " ");
546 append(&selected, words[i].original);
547 }
548 }
549 number(out, matches);
550 append(out, " of ");
551 number(out, count);
552 if (matches) {
553 append(out, ": ");
554 append(out, selected.data);
555 }
556 return;
557 }
558 if (cmd == 8) {
559 int left = section_words[0], right = section_words[1];
560 bool exact = true;
561 for (int i = 0; i < left && i < right; ++i) {
562 if (!same_sequence(&words[i], &words[left + i], true)) {
563 append(out, "DIFFERENT at word ");
564 number(out, i + 1);
565 return;
566 }
567 if (!same_sequence(&words[i], &words[left + i], false)) exact = false;
568 }
569 if (left != right) {
570 append(out, "DIFFERENT at word ");
571 number(out, (left < right ? left : right) + 1);
572 } else append(out, exact ? "EXACT" : "LOOSE");
573 return;
574 }
575 if (cmd == 9 || cmd == 10) {
576 Buffer body = {0};
577 int total = 0;
578 for (int j = 0; j < count; ++j) {
579 const Word *w = &words[j];
580 char canonical[CAP][8];
581 bool paired[CAP];
582 if (cmd == 10) {
583 parts(w, source, 0, canonical);
584 pairs(w, paired, false);
585 }
586 if (j) append(&body, " ");
587 for (int i = 0; i < w->count; ++i) {
588 int id = w->tokens[i].id;
589 bool starts = cmd == 9 || i == 0 || (id < N && !paired[i]);
590 if (starts) {
591 if (i) append(&body, ".");
592 ++total;
593 }
594 append(&body, cmd == 10 ? canonical[i] : id < N ? table[id].spelling[source] : id == N ? "N" : id == Q ? "Q" : "-");
595 }
596 }
597 number(out, total);
598 append(out, ": ");
599 append(out, body.data);
600 return;
601 }
602 if (cmd == 11) {
603 for (int j = 0; j < count; ++j) {
604 for (int i = 0; i < words[j].count; ++i) {
605 int id = words[j].tokens[i].id;
606 if (out->size) append(out, ", ");
607 append(out, id < N ? table[id].spelling[source] : id == N ? "N" : id == Q ? "Q" : "-");
608 append(out, ": ");
609 if (id < N) {
610 append(out, rows[table[id].row]);
611 append(out, " ");
612 append(out, table[id].dan);
613 } else append(out, "special");
614 }
615 }
616 return;
617 }
618 if (cmd == 12) {
619 int row_counts[10] = {0}, other[6] = {0}, total = 0;
620 for (int j = 0; j < count; ++j) {
621 for (int i = 0; i < words[j].count; ++i) {
622 int id = words[j].tokens[i].id;
623 ++total;
624 if (id < N) {
625 ++row_counts[table[id].base];
626 if (table[id].voice) ++other[table[id].voice - 1];
627 if (strlen(table[id].dan) == 2) ++other[2];
628 } else ++other[id - N + 3];
629 }
630 }
631 append(out, "morae=");
632 number(out, total);
633 for (int r = 0; r < 10; ++r) {
634 if (row_counts[r]) {
635 append(out, " ");
636 append(out, rows[r]);
637 append(out, "=");
638 number(out, row_counts[r]);
639 }
640 }
641 const char *labels[] = {"voiced", "semivoiced", "contracted", "N", "Q", "L"};
642 for (int i = 0; i < 6; ++i) {
643 append(out, " ");
644 append(out, labels[i]);
645 append(out, "=");
646 number(out, other[i]);
647 }
648 return;
649 }
650 if (cmd == 13 || cmd == 14) {
651 qsort(words, count, sizeof *words, compare_words);
652 char previous[8] = "";
653 for (int i = 0; i < count; ++i) {
654 if (cmd == 13) item(out, words[i].original, " ");
655 else {
656 Element e[CAP * 2];
657 elements(&words[i], e);
658 int id = e[0].id;
659 const char *heading = id == N ? "N" : order_mode == 1 ? table[id].id : rows[table[id].row];
660 if (!strcmp(previous, heading)) append(out, " ");
661 else {
662 if (i) append(out, "; ");
663 append(out, heading);
664 append(out, ": ");
665 }
666 append(out, words[i].original);
667 strcpy(previous, heading);
668 }
669 }
670 return;
671 }
672 if (cmd == 16) {
673 int counts[5] = {0}, expected[] = {5, 7, 5, 7, 7};
674 bool ok = true;
675 for (int i = 0; i < count; ++i) counts[words[i].section] += words[i].count;
676 for (int i = 0; i < sections; ++i) if (counts[i] != expected[i]) ok = false;
677 append(out, ok ? "OK " : "NG ");
678 append(out, sections == 3 ? "haiku " : "tanka ");
679 for (int i = 0; i < sections; ++i) {
680 if (i) append(out, "-");
681 number(out, counts[i]);
682 }
683 return;
684 }
685 if (cmd == 17) {
686 int kana[CAP * 2], length = 0;
687 for (int i = 0; i < count; ++i) {
688 Element e[CAP * 2];
689 int n = elements(&words[i], e);
690 for (int j = 0; j < n; ++j) kana[length++] = e[j].id;
691 }
692 bool palindrome = true;
693 for (int i = 0; i < length / 2; ++i) if (kana[i] != kana[length - 1 - i]) palindrome = false;
694 append(out, palindrome ? "YES " : "NO ");
695 for (int i = 0; i < length; ++i) {
696 if (i) append(out, ".");
697 append(out, kana[i] == N ? "N" : table[kana[i]].id);
698 }
699 return;
700 }
701 int previous_tail = -1;
702 for (int i = 0; i < count; ++i) {
703 Element e[CAP * 2];
704 int length = elements(&words[i], e), reason = 0;
705 if (i && e[0].id != previous_tail) reason = 1;
706 if (!reason) {
707 for (int j = 0; j < i; ++j) {
708 if (same_sequence(&words[i], &words[j], false)) {
709 reason = 2;
710 break;
711 }
712 }
713 }
714 if (!reason && e[length - 1].id == N) reason = 3;
715 if (reason) {
716 append(out, "NG ");
717 number(out, i + 1);
718 append(out, ": ");
719 append(out, words[i].original);
720 if (reason == 1) {
721 append(out, " must start with ");
722 append(out, table[previous_tail].spelling[source]);
723 }
724 if (reason == 2) append(out, " is repeated");
725 if (reason == 3) append(out, " ends with N");
726 return;
727 }
728 previous_tail = e[length - 1].id;
729 }
730 append(out, "OK ");
731 number(out, count);
732}

746 }
