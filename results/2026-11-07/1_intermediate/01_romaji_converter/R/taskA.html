92parse_word <- function(text,start,system) {
93 word <- tolower(text)
94 ids <- character(0)
95 columns <- numeric(0)
96 p <- 1
97 while (p <= nchar(word)) {
98 c <- character_at(word,p)
99 c2 <- character_at(word,p+1)
100 c3 <- character_at(word,p+2)
101 matched <- match_syllable(word,p,system)
102 advance <- 1
103 if (c == "-") {
104 if (length(ids) == 0 || !is_syllabic(tail(ids,1))) {
105 column_error(start+p-1,"misplaced long mark")
106 }
107 id <- "-"
108 } else if (c == "'") {
109 column_error(start+p-1,"misplaced apostrophe")
110 } else if (!is.null(matched)) {
111 id <- matched$id
112 advance <- nchar(matched$text)
113 } else if (system %in% c("WAPURO","NORMAL") && startsWith(substring(word,p),"xtu")) {
114 id <- "Q"
115 advance <- 3
116 } else if (c == "n") {
117 id <- "N"
118 if (system == "WAPURO") {
119 advance <- if (c2 == "n") 2 else 1
120 } else if (system == "NORMAL") {
121 if (c2 == "'" || (c2 == "n" && !c3 %in% c("a","i","u","e","o","y","'"))) {
122 advance <- 2
123 }
124 } else if (c2 == "'") {
125 advance <- 2
126 }
127 } else if (system %in% c("TRADITIONAL","NORMAL") && c == "m" && c2 %in% c("b","m","p")) {
128 id <- "N"
129 } else if (grepl("^[a-z]$",c) && !c %in% c("a","i","u","e","o","n") && c == c2 && !is.null(match_syllable(word,p+1,system))) {
130 id <- "Q"
131 } else if (system %in% c("HEPBURN","TRADITIONAL","NORMAL") && c == "t" && c2 == "c" && c3 == "h" && !is.null(match_syllable(word,p+1,system))) {
132 id <- "Q"
133 } else {
134 column_error(start+p-1,"unknown syllable")
135 }
136 ids <- c(ids,id)
137 columns <- c(columns,start+p-1)
138 p <- p+advance
139 }
140 list(ids=ids,columns=columns)
141}

143long_flags <- function(ids) {
144 paired <- rep(FALSE,length(ids))
145 long <- ids == "-"
146 i <- 1
147 while (i < length(ids)) {
148 first <- ids[i]
149 second <- ids[i+1]
150 if (is_syllabic(first) && is_vowel(second)) {
151 vowel <- last_letter(first)
152 if (vowel == second || (vowel == "o" && second == "u") || (vowel == "e" && second == "i")) {
153 paired[i+1] <- TRUE
154 long[i+1] <- (vowel == "o" && second %in% c("o","u")) || (vowel == "u" && second == "u")
155 i <- i+2
156 next
157 }
158 }
159 i <- i+1
160 }
161 list(paired=paired,long=long)
162}

164write_parts <- function(ids,columns,system,long_setting) {
165 long <- long_flags(ids)$long
166 parts <- character(0)
167 for (i in seq_along(ids)) {
168 id <- ids[i]
169 next_id <- if (i < length(ids)) ids[i+1] else ""
170 next_text <- if (is_syllabic(next_id)) spelling(system,next_id) else ""
171 first <- substr(next_text,1,1)
172 if (long[i]) {
173 vowel <- last_letter(ids[i-1])
174 if (long_setting == "KEEP") {
175 part <- if (id == "-") "-" else spelling(system,id)
176 } else if (long_setting == "DROP") {
177 part <- ""
178 } else if (long_setting == "PASSPORT") {
179 part <- if (vowel == "o" && !is_vowel(first)) "h" else ""
180 } else if (long_setting == "HYPHEN") {
181 part <- "-"
182 } else {
183 part <- vowel
184 }
185 } else if (is_syllabic(id)) {
186 part <- spelling(system,id)
187 } else if (id == "N") {
188 if (system == "WAPURO") {
189 part <- if (next_id == "N" || first %in% c("a","i","u","e","o","y","n")) "nn" else "n"
190 } else if (system == "TRADITIONAL" && first %in% c("b","m","p")) {
191 part <- "m"
192 } else {
193 part <- if (first %in% c("a","i","u","e","o","y")) "n'" else "n"
194 }
195 } else {
196 if (next_text != "" && !first %in% c("a","i","u","e","o","n")) {
197 part <- if (system %in% c("HEPBURN","TRADITIONAL") && startsWith(next_text,"ch")) "t" else first
198 } else if (system == "WAPURO") {
199 part <- "xtu"
200 } else {
201 column_error(columns[i],"small tsu cannot be written")
202 }
203 }
204 parts <- c(parts,part)
205 }
206 parts
207}

209word_pattern <- function(text) {
210 if (text == tolower(text)) return("LOWER")
211 if (grepl("^[A-Z]",text) && substring(text,2) == tolower(substring(text,2))) return("TITLE")
212 letters <- strsplit(text,"",fixed=TRUE)[[1]]
213 if (sum(grepl("^[A-Z]$",letters)) >= 2 && text == toupper(text)) return("UPPER")
214 ""
215}

308pattern_tokens <- function(pattern,system) {
309 if (!grepl("^[a-z'\\-?*]+$",pattern)) fail("invalid pattern")
310 tokens <- character(0)
311 chunk <- ""
312 tryCatch({
313 for (c in strsplit(pattern,"",fixed=TRUE)[[1]]) {
314 if (c %in% c("?","*")) {
315 tokens <- c(tokens,parse_word(chunk,1,system)$ids,c)
316 chunk <- ""
317 } else {
318 chunk <- paste0(chunk,c)
319 }
320 }
321 c(tokens,parse_word(chunk,1,system)$ids)
322 },error=function(e) fail("invalid pattern"))
323}

325wildcard_matches <- function(tokens,ids) {
326 previous <- c(TRUE,rep(FALSE,length(ids)))
327 for (token in tokens) {
328 current <- rep(FALSE,length(ids)+1)
329 if (token == "*") current[1] <- previous[1]
330 for (j in seq_along(ids)) {
331 if (token == "*") {
332 current[j+1] <- previous[j+1] || current[j]
333 } else {
334 current[j+1] <- previous[j] && (token == "?" || token == ids[j])
335 }
336 }
337 previous <- current
338 }
339 tail(previous,1)
340}

342execute <- function(line,settings) {
343 command <- strsplit(line," ",fixed=TRUE)[[1]][1]
344 if (!command %in% names(arg_counts)) fail(paste("unknown command",command))
345 k <- arg_counts[[command]]
346 if (command %in% c("SET","TABLE")) {
347 parts <- strsplit(line," ",fixed=TRUE)[[1]]
348 if (length(parts) < 3) fail("missing argument")
349 if (length(parts) > 3) fail("too many arguments")
350 args <- parts[2:3]
351 text <- ""
352 } else {
353 if (count_character(line," ") < k+1) fail("missing argument")
354 args <- character(0)
355 position <- nchar(command)+2
356 for (i in seq_len(k)) {
357 rest <- substring(line,position)
358 ending <- regexpr(" ",rest,fixed=TRUE)[1]
359 args <- c(args,substr(rest,1,ending-1))
360 position <- position+ending
361 }
362 text <- substring(line,position)
363 }
364 system_count <- 1
365 if (command %in% c("SET","NORMALIZE","DETECT")) {
366 system_count <- 0
367 } else if (command %in% c("CONVERT","DIFF","SAME")) {
368 system_count <- 2
369 }
370 for (system in head(args,system_count)) {
371 if (!system %in% systems) fail(paste("unknown system",system))
372 }
373 if (command == "SET") {
374 option <- args[1]
375 value <- args[2]
376 if (!option %in% names(set_values)) fail(paste("unknown option",option))
377 if (!value %in% set_values[[option]]) fail(paste("unknown value",value))
378 settings[[option]] <- value
379 return(paste0(option,"=",value))
380 }
381 if (command == "TABLE") {
382 system <- args[1]
383 row <- args[2]
384 if (!row %in% rows) fail(paste("unknown row",row))
385 return(paste(vapply(identifiers[row_of[identifiers] == row],function(id) spelling(system,id),character(1)),collapse=" "))
386 }
387 tokens <- if (command == "FIND") pattern_tokens(args[2],args[1]) else character(0)
388 slash_allowed <- command %in% c("HAIKU","SAME")
389 characters <- strsplit(text,"",fixed=TRUE)[[1]]
390 for (i in seq_along(characters)) {
391 c <- characters[i]
392 allowed <- grepl("^[A-Za-z0-9]$",c) || c %in% strsplit("'- .,!?;:()\"","",fixed=TRUE)[[1]] || (slash_allowed && c == "/")
393 if (!allowed) column_error(i,"invalid character")
394 }
395 word_list <- words_in(text)
396 if (command %in% c("SAME","HAIKU")) {
397 parts <- count_character(text,"/")+1
398 message <- if (command == "SAME") "expected two texts" else "expected 3 or 5 lines"
399 if ((command == "SAME" && parts != 2) || (command == "HAIKU" && !parts %in% c(3,5))) fail(message)
400 for (i in seq_len(parts)) {
401 if (!any(vapply(word_list,function(word) word$part == i,logical(1)))) fail(message)
402 }
403 }
404 if (length(word_list) == 0) fail("no words")
405 for (i in seq_along(word_list)) {
406 word_list[[i]]$pattern <- word_pattern(word_list[[i]]$text)
407 if (word_list[[i]]$pattern == "") column_error(word_list[[i]]$start,"mixed case")
408 }
409 if (command == "DETECT") {
410 output <- character(0)
411 for (system in systems) {
412 verdict <- tryCatch({
413 result <- "canonical"
414 parsed_words <- lapply(word_list,function(word) {
415 parsed <- parse_word(word$text,word$start,system)
416 word$ids <- parsed$ids
417 word$columns <- parsed$columns
418 word
419 })
420 for (word in parsed_words) {
421 if (written(word,system,settings,TRUE) != word$text) result <- "readable"
422 }
423 result
424 },error=function(e) "error")
425 output <- c(output,paste0(system,"=",verdict))
426 }
427 return(paste(output,collapse=" "))
428 }
429 for (i in seq_along(word_list)) {
430 word <- word_list[[i]]
431 if (command == "NORMALIZE") {
432 system <- "NORMAL"
433 } else if (command == "SAME") {
434 system <- args[word$part]
435 } else {
436 system <- args[1]
437 }
438 parsed <- parse_word(word$text,word$start,system)
439 word_list[[i]]$ids <- parsed$ids
440 word_list[[i]]$columns <- parsed$columns
441 }
442 system <- if (command == "NORMALIZE") "HEPBURN" else args[1]
443 if (command %in% c("CONVERT","NORMALIZE","REVERSE")) {
444 target <- if (command == "CONVERT") args[2] else system
445 output <- vapply(word_list,function(word) {
446 ids <- word$ids
447 columns <- word$columns
448 if (command == "REVERSE") {
449 for (i in seq_along(ids)) {
450 if (ids[i] == "-") ids[i] <- last_letter(word$ids[i-1])
451 }
452 ids <- rev(ids)
453 columns <- rev(columns)
454 }
455 written(word,target,settings,FALSE,ids,columns)
456 },character(1))
457 return(replace_words(text,word_list,output))
458 }
459 if (command == "VALID") {
460 for (word in word_list) {
461 expected <- written(word,system,settings,TRUE)
462 if (expected != word$text) {
463 i <- 1
464 while (i <= min(nchar(expected),nchar(word$text)) && character_at(expected,i) == character_at(word$text,i)) {
465 i <- i+1
466 }
467 return(paste0("INVALID col ",word$start+i-1,": expected ",expected))
468 }
469 }
470 return("VALID")
471 }
472 if (command == "DIFF") {
473 output <- character(0)
474 for (word in word_list) {
475 a <- written(word,args[1],settings)
476 b <- written(word,args[2],settings)
477 if (a != b) output <- c(output,paste0(a,">",b))
478 }
479 prefix <- paste(length(output),"of",length(word_list))
480 return(if (length(output) == 0) prefix else paste0(prefix,": ",paste(output,collapse=" ")))
481 }
482 if (command == "SAME") {
483 left <- Filter(function(word) word$part == 1,word_list)
484 right <- Filter(function(word) word$part == 2,word_list)
485 exact <- TRUE
486 for (i in seq_len(min(length(left),length(right)))) {
487 a <- left[[i]]$ids
488 b <- right[[i]]$ids
489 if (!identical(a[!long_flags(a)$long],b[!long_flags(b)$long])) return(paste("DIFFERENT at word",i))
490 exact <- exact && identical(a,b)
491 }
492 if (length(left) != length(right)) return(paste("DIFFERENT at word",min(length(left),length(right))+1))
493 return(if (exact) "EXACT" else "LOOSE")
494 }
495 if (command %in% c("MORAE","SYLLABLES")) {
496 output <- character(0)
497 total <- 0
498 for (word in word_list) {
499 if (command == "MORAE") {
500 parts <- vapply(word$ids,function(id) spelling(system,id),character(1))
501 } else {
502 paired <- long_flags(word$ids)$paired
503 written_parts <- write_parts(word$ids,word$columns,system,"KEEP")
504 parts <- character(0)
505 for (i in seq_along(word$ids)) {
506 if (i > 1 && (!is_syllabic(word$ids[i]) || paired[i])) {
507 parts[length(parts)] <- paste0(parts[length(parts)],written_parts[i])
508 } else {
509 parts <- c(parts,written_parts[i])
510 }
511 }
512 }
513 total <- total+length(parts)
514 output <- c(output,paste(parts,collapse="."))
515 }
516 return(paste0(total,": ",paste(output,collapse=" ")))
517 }
518 if (command == "ROW") {
519 output <- character(0)
520 for (word in word_list) {
521 for (id in word$ids) {
522 item <- if (is_syllabic(id)) paste0(spelling(system,id),": ",row_of[[id]]," ",dan_of[[id]]) else paste0(id,": special")
523 output <- c(output,item)
524 }
525 }
526 return(paste(output,collapse=", "))
527 }
528 if (command == "STATS") {
529 counts <- setNames(rep(0,16),c(rows[1:10],words("voiced semivoiced contracted N Q L")))
530 total <- 0
531 for (word in word_list) {
532 for (id in word$ids) {
533 total <- total+1
534 if (is_syllabic(id)) {
535 row <- row_of[[id]]
536 base <- if (row %in% names(base_row)) base_row[[row]] else row
537 counts[base] <- counts[base]+1
538 if (row %in% c("ga","za","da","ba")) counts["voiced"] <- counts["voiced"]+1
539 if (row == "pa") counts["semivoiced"] <- counts["semivoiced"]+1
540 if (nchar(dan_of[[id]]) == 2) counts["contracted"] <- counts["contracted"]+1
541 } else {
542 key <- if (id == "-") "L" else id
543 counts[key] <- counts[key]+1
544 }
545 }
546 }
547 output <- paste0("morae=",total)
548 for (row in rows[1:10]) {
549 if (counts[[row]] > 0) output <- c(output,paste0(row,"=",counts[[row]]))
550 }
551 for (name in words("voiced semivoiced contracted N Q L")) output <- c(output,paste0(name,"=",counts[[name]]))
552 return(paste(output,collapse=" "))
553 }
554 if (command %in% c("SORT","INDEX")) {
555 keys <- lapply(word_list,collation,settings$ORDER)
556 indices <- seq_along(word_list)
557 for (i in seq_along(indices)) {
558 index <- indices[i]
559 position <- i
560 while (position > 1 && key_less(keys[[index]],keys[[indices[position-1]]])) {
561 indices[position] <- indices[position-1]
562 position <- position-1
563 }
564 indices[position] <- index
565 }
566 ordered <- word_list[indices]
567 if (command == "SORT") return(paste(vapply(ordered,function(word) word$text,character(1)),collapse=" "))
568 groups <- character(0)
569 previous <- ""
570 for (word in ordered) {
571 base <- elements(word$ids)[[1]]$base
572 heading <- if (settings$ORDER == "IROHA" || base == "N") base else row_of[[base]]
573 if (heading == previous) {
574 groups[length(groups)] <- paste(groups[length(groups)],word$text)
575 } else {
576 groups <- c(groups,paste0(heading,": ",word$text))
577 previous <- heading
578 }
579 }
580 return(paste(groups,collapse="; "))
581 }
582 if (command == "FIND") {
583 matches <- Filter(function(word) wildcard_matches(tokens,word$ids),word_list)
584 prefix <- paste(length(matches),"of",length(word_list))
585 return(if (length(matches) == 0) prefix else paste0(prefix,": ",paste(vapply(matches,function(word) word$text,character(1)),collapse=" ")))
586 }
587 if (command == "HAIKU") {
588 counts <- rep(0,count_character(text,"/")+1)
589 for (word in word_list) counts[word$part] <- counts[word$part]+length(word$ids)
590 expected <- if (length(counts) == 3) c(5,7,5) else c(5,7,5,7,7)
591 verdict <- if (identical(counts,expected)) "OK" else "NG"
592 form <- if (length(counts) == 3) "haiku" else "tanka"
593 return(paste(verdict,form,paste(counts,collapse="-")))
594 }
595 if (command == "KAIBUN") {
596 bases <- unlist(lapply(word_list,function(word) vapply(elements(word$ids),function(item) item$base,character(1))),use.names=FALSE)
597 return(paste(if (identical(bases,rev(bases))) "YES" else "NO",paste(bases,collapse=".")))
598 }
599 seen <- list()
600 previous <- ""
601 for (i in seq_along(word_list)) {
602 word <- word_list[[i]]
603 items <- elements(word$ids)
604 head <- items[[1]]$base
605 tail <- items[[length(items)]]$base
606 prefix <- paste0("NG ",i,": ",word$text)
607 if (i > 1 && head != previous) return(paste(prefix,"must start with",spelling(system,previous)))
608 if (any(vapply(seen,function(ids) identical(ids,word$ids),logical(1)))) return(paste(prefix,"is repeated"))
609 if (tail == "N") return(paste(prefix,"ends with N"))
610 seen[[length(seen)+1]] <- word$ids
611 previous <- tail
612 }
613 paste("OK",length(word_list))
614}
