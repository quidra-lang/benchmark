100def case_pattern(text, column):
101 letters = [char for char in text if char in LETTERS]
102 if all(char in LOWER for char in letters):
103 return "LOWER"
104 if text[0] in UPPER and all(char in LOWER for char in text[1:] if char in LETTERS):
105 return "TITLE"
106 if len(letters) >= 2 and all(char in UPPER for char in letters):
107 return "UPPER"
108 raise Problem("mixed case", column)

122def syllable_at(text, position, readings):
123 for length in (3, 2, 1):
124 spelling = text[position:position + length]
125 if len(spelling) == length and spelling in readings:
126 return readings[spelling], length
127 return None

130def parse_word(original, column, system, normalize=False):
131 text = original.lower()
132 readings = NORMAL_READINGS if normalize else READINGS[system]
133 morae = []
134 position = 0
135 while position < len(text):
136 char = text[position]
137 next_char = text[position + 1] if position + 1 < len(text) else ""
138 third = text[position + 2] if position + 2 < len(text) else ""
139 current_column = column + position
140 if char == "-":
141 if not morae or morae[-1].identifier not in MORA_INFO:
142 raise Problem("misplaced long mark", current_column)
143 identifier, length = "L", 1
144 elif char == "'":
145 raise Problem("misplaced apostrophe", current_column)
146 else:
147 match = syllable_at(text, position, readings)
148 if match is not None:
149 identifier, length = match
150 elif (system == "WAPURO" or normalize) and text.startswith("xtu", position):
151 identifier, length = "Q", 3
152 elif char == "n":
153 identifier, length = "N", 1
154 if normalize:
155 if next_char == "'" or (next_char == "n" and (not third or third not in "aiueoy'")):
156 length = 2
157 elif system == "WAPURO":
158 if next_char == "n":
159 length = 2
160 elif next_char == "'":
161 length = 2
162 elif (system == "TRADITIONAL" or normalize) and char == "m" and next_char in ("b", "m", "p"):
163 identifier, length = "N", 1
164 elif char in LOWER and char not in "aiueon" and next_char == char and syllable_at(text, position + 1, readings):
165 identifier, length = "Q", 1
166 elif (system in ("HEPBURN", "TRADITIONAL") or normalize) and text.startswith("tch", position) and syllable_at(text, position + 1, readings):
167 identifier, length = "Q", 1
168 else:
169 raise Problem("unknown syllable", current_column)
170 morae.append(Mora(identifier, current_column))
171 position += length
172 return morae

175def long_pairs(morae):
176 latter = set()
177 position = 0
178 while position + 1 < len(morae):
179 first = morae[position].identifier
180 second = morae[position + 1].identifier
181 if first in MORA_INFO and second in VOWELS and len(second) == 1:
182 vowel = first[-1]
183 if vowel == second or (vowel, second) in (("o", "u"), ("e", "i")):
184 latter.add(position + 1)
185 position += 2
186 continue
187 position += 1
188 return latter

191def long_elements(morae):
192 pairs = long_pairs(morae)
193 elements = {}
194 for index, mora in enumerate(morae):
195 if mora.identifier == "L":
196 elements[index] = morae[index - 1].identifier[-1]
197 elif index in pairs:
198 previous = morae[index - 1].identifier[-1]
199 if (previous, mora.identifier) in (("o", "o"), ("o", "u"), ("u", "u")):
200 elements[index] = previous
201 return elements

204def write_parts(morae, system, long_mode):
205 elements = long_elements(morae)
206 parts = []
207 for index, mora in enumerate(morae):
208 identifier = mora.identifier
209 following = morae[index + 1].identifier if index + 1 < len(morae) else None
210 next_spelling = SPELLINGS[system].get(following, "")
211 if index in elements:
212 vowel = elements[index]
213 if long_mode == "KEEP":
214 part = "-" if identifier == "L" else SPELLINGS[system][identifier]
215 elif long_mode == "DROP":
216 part = ""
217 elif long_mode == "PASSPORT":
218 part = "h" if vowel == "o" and (not next_spelling or next_spelling[0] not in VOWELS) else ""
219 elif long_mode == "HYPHEN":
220 part = "-"
221 else:
222 part = vowel
223 elif identifier in MORA_INFO:
224 part = SPELLINGS[system][identifier]
225 elif identifier == "N":
226 initial = next_spelling[:1]
227 if system == "WAPURO":
228 part = "nn" if following == "N" or initial in tuple("aiueoyn") else "n"
229 elif system == "TRADITIONAL" and initial in ("b", "m", "p"):
230 part = "m"
231 else:
232 part = "n'" if initial in tuple("aiueoy") else "n"
233 else:
234 if next_spelling and next_spelling[0] not in "aiueon":
235 part = "t" if system in ("HEPBURN", "TRADITIONAL") and next_spelling.startswith("ch") else next_spelling[0]
236 elif system == "WAPURO":
237 part = "xtu"
238 else:
239 raise Problem("small tsu cannot be written", mora.column)
240 parts.append(part)
241 return parts

244def apply_case(text, pattern):
245 if pattern == "UPPER":
246 return text.upper()
247 if pattern == "TITLE":
248 return text.capitalize()
249 return text

253 return apply_case("".join(write_parts(word.morae, system, "KEEP")), word.pattern)

260def collation_elements(morae):
261 elements = []
262 for mora in morae:
263 identifier = mora.identifier
264 if identifier == "L":
265 continue
266 if identifier == "N":
267 elements.append(("N", 0, 0))
268 elif identifier == "Q":
269 elements.append(("tu", 0, 1))
270 else:
271 row, dan = MORA_INFO[identifier]
272 voice = 1 if row in ("ga", "za", "da", "ba") else 2 if row == "pa" else 0
273 if dan in ("ya", "yu", "yo"):
274 elements.append((devoiced(row[0] + "i"), voice, 0))
275 elements.append((dan, 0, 1))
276 else:
277 elements.append((devoiced(identifier), voice, 0))
278 return elements

281def wildcard_match(pattern, morae):
282 identifiers = [mora.identifier for mora in morae]
283 previous = [True] + [False] * len(identifiers)
284 for item in pattern:
285 current = [False] * (len(identifiers) + 1)
286 if item == "*":
287 current[0] = previous[0]
288 for index in range(1, len(current)):
289 current[index] = previous[index] or current[index - 1]
290 else:
291 for index, identifier in enumerate(identifiers, 1):
292 current[index] = previous[index - 1] and (item == "?" or item == identifier)
293 previous = current
294 return previous[-1]

321def read_pattern(text, system):
322 if any(char not in LOWER + "'-?*" for char in text):
323 raise Problem("invalid pattern")
324 result = []
325 try:
326 for part in re.split(r"([?*])", text):
327 if part in ("?", "*"):
328 result.append(part)
329 elif part:
330 result.extend(mora.identifier for mora in parse_word(part, 1, system))
331 except Problem:
332 raise Problem("invalid pattern")
333 return result

336def prepare_request(line):
337 command = line.split(" ", 1)[0]
338 if command not in FORMATS:
339 raise Problem(f"unknown command {command}")
340 count, system_count = FORMATS[command]
341 if command in ("SET", "TABLE"):
342 pieces = line.split(" ")
343 if len(pieces) < 3:
344 raise Problem("missing argument")
345 if len(pieces) > 3:
346 raise Problem("too many arguments")
347 args, text = pieces[1:], ""
348 else:
349 pieces = line.split(" ", count + 1)
350 if len(pieces) < count + 2:
351 raise Problem("missing argument")
352 args, text = pieces[1:count + 1], pieces[-1]
353 for system in args[:system_count]:
354 if system not in SYSTEMS:
355 raise Problem(f"unknown system {system}")
356 if command in ("SET", "TABLE"):
357 return Request(command, args)
358 pattern = read_pattern(args[1], args[0]) if command == "FIND" else None
359 allowed = LETTERS + "'-" + SEPARATORS + ("/" if command in ("SAME", "HAIKU") else "")
360 for column, char in enumerate(text, 1):
361 if char not in allowed:
362 raise Problem("invalid character", column)
363 if command == "SAME":
364 parts = text.split("/")
365 if len(parts) != 2 or any(WORD_PATTERN.search(part) is None for part in parts):
366 raise Problem("expected two texts")
367 if command == "HAIKU":
368 parts = text.split("/")
369 if len(parts) not in (3, 5) or any(WORD_PATTERN.search(part) is None for part in parts):
370 raise Problem("expected 3 or 5 lines")
371 matches = list(WORD_PATTERN.finditer(text))
372 if not matches:
373 raise Problem("no words")
374 words = [Word(match.group(), match.start() + 1) for match in matches]
375 if command != "DETECT":
376 slash_column = text.index("/") + 1 if command == "SAME" else None
377 for word in words:
378 if command == "NORMALIZE":
379 system = "HEPBURN"
380 elif command == "SAME" and word.column > slash_column:
381 system = args[1]
382 else:
383 system = args[0]
384 word.morae = parse_word(word.text, word.column, system, command == "NORMALIZE")
385 return Request(command, args, text, words, pattern)

388class Converter:
389 def __init__(self):
390 self.settings = {"LONG": "KEEP", "CASE": "KEEP", "ORDER": "GOJUON"}
391 self.handlers = {
392 "SET": self.set_option, "TABLE": self.table, "CONVERT": self.convert,
393 "NORMALIZE": self.normalize, "REVERSE": self.reverse, "VALID": self.valid,
394 "DETECT": self.detect, "DIFF": self.diff, "SAME": self.same,
395 "MORAE": self.morae, "SYLLABLES": self.syllables, "ROW": self.row,
396 "STATS": self.stats, "SORT": self.sort, "INDEX": self.index,
397 "FIND": self.find, "HAIKU": self.haiku, "KAIBUN": self.kaibun,
398 "SHIRITORI": self.shiritori,
399 }

401 def execute(self, line):
402 try:
403 request = prepare_request(line)
404 return self.handlers[request.command](request)
405 except Problem as problem:
406 return "ERROR " + problem.message

408 def write(self, word, system, morae=None):
409 sequence = word.morae if morae is None else morae
410 text = "".join(write_parts(sequence, system, self.settings["LONG"]))
411 pattern = word.pattern if self.settings["CASE"] == "KEEP" else self.settings["CASE"]
412 return apply_case(text, pattern)

414 def rewrite_text(self, request, render):
415 words = iter(request.words)
416 return WORD_PATTERN.sub(lambda match: render(next(words)), request.text)

452 def valid(self, request):
453 for word in request.words:
454 expected = canonical(word, request.args[0])
455 if word.text != expected:
456 index = 0
457 while index < min(len(word.text), len(expected)) and word.text[index] == expected[index]:
458 index += 1
459 return f"INVALID col {word.column + index}: expected {expected}"

462 def detect(self, request):
463 results = []
464 for system in SYSTEMS:
465 try:
466 for word in request.words:

474 def diff(self, request):
475 differences = []
476 for word in request.words:
477 first = self.write(word, request.args[0])
478 second = self.write(word, request.args[1])
479 if first != second:
480 differences.append(f"{first}>{second}")
481 counts = f"{len(differences)} of {len(request.words)}"
482 return counts + (": " + " ".join(differences) if differences else "")

484 def same(self, request):
485 slash_column = request.text.index("/") + 1
486 left = [word for word in request.words if word.column < slash_column]
487 right = [word for word in request.words if word.column > slash_column]
488 exact = True
489 for index, (first, second) in enumerate(zip(left, right), 1):
490 first_long = long_elements(first.morae)
491 second_long = long_elements(second.morae)
492 first_loose = [m.identifier for i, m in enumerate(first.morae) if i not in first_long]
493 second_loose = [m.identifier for i, m in enumerate(second.morae) if i not in second_long]
494 if first_loose != second_loose:
495 return f"DIFFERENT at word {index}"
496 exact = exact and first.identifiers() == second.identifiers()
497 if len(left) != len(right):
498 return f"DIFFERENT at word {min(len(left), len(right)) + 1}"
499 return "EXACT" if exact else "LOOSE"

501 def morae(self, request):
502 system = request.args[0]
503 count = sum(len(word.morae) for word in request.words)
504 words = [".".join(SPELLINGS[system][m.identifier] if m.identifier in MORA_INFO
505 else SPECIAL_DISPLAY[m.identifier] for m in word.morae)
506 for word in request.words]
507 return f"{count}: " + " ".join(words)

509 def syllables(self, request):
510 count = 0
511 words = []
512 for word in request.words:
513 parts = write_parts(word.morae, request.args[0], "KEEP")
514 pairs = long_pairs(word.morae)
515 syllables = []
516 for index, (mora, part) in enumerate(zip(word.morae, parts)):
517 if index == 0 or (mora.identifier in MORA_INFO and index not in pairs):
518 syllables.append(part)
519 else:
520 syllables[-1] += part
521 count += len(syllables)
522 words.append(".".join(syllables))
523 return f"{count}: " + " ".join(words)

529 identifier = mora.identifier

537 def stats(self, request):
538 base_counts = dict.fromkeys(BASE_ROWS, 0)
539 other_counts = dict.fromkeys(("voiced", "semivoiced", "contracted", "N", "Q", "L"), 0)
540 total = 0
541 for word in request.words:
542 for mora in word.morae:
543 total += 1
544 identifier = mora.identifier
545 if identifier in MORA_INFO:
546 row, dan = MORA_INFO[identifier]
547 base_counts[BASE_ROW[row]] += 1
548 if row in ("ga", "za", "da", "ba"):
549 other_counts["voiced"] += 1
550 if row == "pa":
551 other_counts["semivoiced"] += 1
552 if dan in ("ya", "yu", "yo"):
553 other_counts["contracted"] += 1
554 else:
555 other_counts[identifier] += 1
556 items = [f"morae={total}"]
557 items.extend(f"{row}={count}" for row, count in base_counts.items() if count)
558 items.extend(f"{name}={count}" for name, count in other_counts.items())
559 return " ".join(items)

561 def collation_key(self, word):
562 elements = collation_elements(word.morae)
563 ranks = COLLATION_RANKS[self.settings["ORDER"]]
564 return (
565 tuple(ranks[base] for base, voice, size in elements),
566 tuple(voice for base, voice, size in elements),
567 tuple(size for base, voice, size in elements),
568 tuple(1 if mora.identifier == "L" else 0 for mora in word.morae),
569 )

577 def index(self, request):
578 groups = []
579 for word in self.sorted_words(request):
580 first = collation_elements(word.morae)[0][0]
581 if self.settings["ORDER"] == "IROHA" or first == "N":
582 heading = first
583 else:
584 heading = MORA_INFO[first][0]
585 if groups and groups[-1][0] == heading:
586 groups[-1][1].append(word.text)
587 else:
588 groups.append((heading, [word.text]))
589 return "; ".join(heading + ": " + " ".join(words) for heading, words in groups)

591 def find(self, request):
592 matches = [word.text for word in request.words if wildcard_match(request.pattern, word.morae)]
593 counts = f"{len(matches)} of {len(request.words)}"
594 return counts + (": " + " ".join(matches) if matches else "")

596 def haiku(self, request):
597 line_count = request.text.count("/") + 1
598 counts = [0] * line_count
599 for word in request.words:
600 index = request.text[:word.column - 1].count("/")
601 counts[index] += len(word.morae)
602 expected = [5, 7, 5] if line_count == 3 else [5, 7, 5, 7, 7]
603 verdict = "OK" if counts == expected else "NG"
604 form = "haiku" if line_count == 3 else "tanka"
605 return f"{verdict} {form} " + "-".join(map(str, counts))

607 def kaibun(self, request):
608 bases = [base for word in request.words for base, voice, size in collation_elements(word.morae)]
609 verdict = "YES" if bases == bases[::-1] else "NO"
610 return verdict + " " + ".".join(bases)

612 def shiritori(self, request):
613 previous_tail = None
