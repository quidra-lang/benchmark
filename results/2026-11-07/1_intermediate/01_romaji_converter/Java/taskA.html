145 List<Mora> parse(Word word, String system) {
146 String text = word.text().toLowerCase(Locale.ROOT);
147 List<Mora> result = new ArrayList<>();
148 boolean normalize = system.equals("ANY");
149 int p = 0;
150 while (p < text.length()) {
151 char c = text.charAt(p), next = p+1 < text.length() ? text.charAt(p+1) : 0;
152 int at = word.start()+p+1;
153 String id;
154 int advance;
155 if (c == '-') {
156 if (result.isEmpty() || !regular(result.getLast().id())) {
157 column(at,"misplaced long mark");
158 }
159 id = "L";
160 advance = 1;
161 } else if (c == '\'') {
162 column(at,"misplaced apostrophe");
163 return result;
164 } else {
165 Match syllable = match(text,p,system);
166 if (syllable != null) {
167 id = syllable.id();
168 advance = syllable.length();
169 } else if ((normalize || system.equals("WAPURO")) && text.startsWith("xtu",p)) {
170 id = "Q";
171 advance = 3;
172 } else if (c == 'n') {
173 id = "N";
174 advance = 1;
175 if (normalize) {
176 char third = p+2 < text.length() ? text.charAt(p+2) : 0;
177 if (next == '\'' || next == 'n' && "aiueoy'".indexOf(third) < 0) {
178 advance = 2;
179 }
180 } else if (system.equals("WAPURO")) {
181 if (next == 'n') {
182 advance = 2;
183 }
184 } else if (next == '\'') {
185 advance = 2;
186 }
187 } else if ((normalize || system.equals("TRADITIONAL")) && c == 'm' && "bmp".indexOf(next) >= 0) {
188 id = "N";
189 advance = 1;
190 } else if (letter(c) && "aiueon".indexOf(c) < 0 && c == next && match(text,p+1,system) != null) {
191 id = "Q";
192 advance = 1;
193 } else if ((normalize || system.equals("HEPBURN") || system.equals("TRADITIONAL")) &&
194 c == 't' && text.startsWith("ch",p+1) && match(text,p+1,system) != null) {
195 id = "Q";
196 advance = 1;
197 } else {
198 column(at,"unknown syllable");
199 return result;
200 }
201 }
202 result.add(new Mora(id,at));
203 p += advance;
204 }
205 return result;
206 }

215 static int pattern(String text) {
216 List<Character> letters = new ArrayList<>();
217 for (char c : text.toCharArray()) {
218 if (letter(c)) {
219 letters.add(c);
220 }
221 }
222 if (letters.stream().allMatch(c -> c >= 'a' && c <= 'z')) {
223 return 0;
224 }
225 if (text.charAt(0) >= 'A' && text.charAt(0) <= 'Z' &&
226 letters.subList(1,letters.size()).stream().allMatch(c -> c >= 'a' && c <= 'z')) {
227 return 1;
228 }
229 if (letters.size() >= 2 && letters.stream().allMatch(c -> c >= 'A' && c <= 'Z')) {
230 return 2;
231 }
232 return -1;
233 }
234 LongInfo longInfo(List<Mora> morae) {
235 int count = morae.size();
236 boolean[] paired = new boolean[count], elements = new boolean[count];
237 char[] vowels = new char[count];
238 for (int i = 0; i < count; i++) {
239 if (morae.get(i).id().equals("L")) {
240 elements[i] = true;
241 vowels[i] = last(morae.get(i-1).id());
242 }
243 }
244 int i = 0;
245 while (i+1 < count) {
246 String a = morae.get(i).id(), b = morae.get(i+1).id();
247 char v = last(a);
248 if (regular(a) && vowel(b) && (v == last(b) || v == 'o' && b.equals("u") || v == 'e' && b.equals("i"))) {
249 paired[i+1] = true;
250 if (v == 'o' && (b.equals("o") || b.equals("u")) || v == 'u' && b.equals("u")) {
251 elements[i+1] = true;
252 vowels[i+1] = v;
253 }
254 i += 2;
255 } else {
256 i++;
257 }
258 }
259 return new LongInfo(paired,elements,vowels);
260 }
261 List<String> parts(Word word, String system, String longMode) {
262 List<Mora> morae = word.morae();
263 LongInfo info = longInfo(morae);
264 List<String> result = new ArrayList<>();
265 for (int i = 0; i < morae.size(); i++) {
266 Mora mora = morae.get(i);
267 String id = mora.id();
268 String nextId = i+1 < morae.size() ? morae.get(i+1).id() : "";
269 String next = regular(nextId) ? spelling(system,nextId) : "";
270 String part;
271 if (info.elements()[i]) {
272 char v = info.vowels()[i];
273 part = switch (longMode) {
274 case "KEEP" -> id.equals("L") ? "-" : spelling(system,id);
275 case "DROP" -> "";
276 case "PASSPORT" -> v == 'o' && !begins(next,"aiueo") ? "h" : "";
277 case "HYPHEN" -> "-";
278 default -> String.valueOf(v);
279 };
280 } else if (regular(id)) {
281 part = spelling(system,id);
282 } else if (id.equals("N")) {
283 if (system.equals("WAPURO")) {
284 part = nextId.equals("N") || begins(next,"aiueoyn") ? "nn" : "n";
285 } else if (system.equals("TRADITIONAL") && begins(next,"bmp")) {
286 part = "m";
287 } else {
288 part = begins(next,"aiueoy") ? "n'" : "n";
289 }
290 } else {
291 if (!next.isEmpty() && !begins(next,"aiueon")) {
292 part = (system.equals("HEPBURN") || system.equals("TRADITIONAL")) && next.startsWith("ch") ? "t" : next.substring(0,1);
293 } else if (system.equals("WAPURO")) {
294 part = "xtu";
295 } else {
296 column(mora.column(),"small tsu cannot be written");
297 return result;
298 }
299 }
300 result.add(part);
301 }
302 return result;
303 }

371 List<String> patternMorae(String pattern, String system) {
372 if (!pattern.matches("[a-z'?*\\-]+")) {
373 fail("invalid pattern");
374 }
375 List<String> result = new ArrayList<>();
376 try {
377 for (String part : pattern.splitWithDelimiters("[?*]",0)) {
378 if (part.equals("?") || part.equals("*")) {
379 result.add(part);
380 } else if (!part.isEmpty()) {
381 result.addAll(parse(new Word(part,0,0,List.of()),system).stream().map(Mora::id).toList());
382 }
383 }
384 } catch (Problem problem) {
385 fail("invalid pattern");
386 }
387 return result;
388 }
389 static boolean matches(List<String> pattern, List<String> word) {
390 boolean[] dp = new boolean[word.size()+1];
391 dp[0] = true;
392 for (String token : pattern) {
393 boolean[] next = new boolean[word.size()+1];
394 next[0] = token.equals("*") && dp[0];
395 for (int j = 1; j <= word.size(); j++) {
396 next[j] = token.equals("*") ? dp[j] || next[j-1] : dp[j-1] && (token.equals("?") || token.equals(word.get(j-1)));
397 }
398 dp = next;
399 }
400 return dp[word.size()];
401 }

423 String execute(String line) {
424 String command = line.split(" ",2)[0];
425 if (!counts.containsKey(command) && !command.equals("SET") && !command.equals("TABLE")) {
426 fail("unknown command "+command);
427 }
428 if (command.equals("SET") || command.equals("TABLE")) {
429 String[] tokens = line.split(" ");
430 if (tokens.length < 3) {
431 fail("missing argument");
432 }
433 if (tokens.length > 3) {
434 fail("too many arguments");
435 }
436 if (command.equals("SET")) {
437 Set<String> allowed = switch (tokens[1]) {
438 case "LONG" -> Set.of("KEEP","DROP","PASSPORT","HYPHEN","DOUBLE");
439 case "CASE" -> Set.of("KEEP","LOWER","UPPER","TITLE");
440 case "ORDER" -> Set.of("GOJUON","IROHA");
441 default -> throw new Problem("unknown option "+tokens[1]);
442 };
443 if (!allowed.contains(tokens[2])) {
444 fail("unknown value "+tokens[2]);
445 }
446 if (tokens[1].equals("LONG")) {
447 longSetting = tokens[2];
448 }
449 if (tokens[1].equals("CASE")) {
450 caseSetting = tokens[2];
451 }
452 if (tokens[1].equals("ORDER")) {
453 orderSetting = tokens[2];
454 }
455 return tokens[1]+"="+tokens[2];
456 }
457 if (!SYSTEMS.contains(tokens[1])) {
458 fail("unknown system "+tokens[1]);
459 }
460 if (!ROWS.contains(tokens[2])) {
461 fail("unknown row "+tokens[2]);
462 }
463 List<String> result = new ArrayList<>();
464 for (Meta meta : metadata.values()) {
465 if (meta.row().equals(tokens[2])) {
466 result.add(spelling(tokens[1],meta.id()));
467 }
468 }
469 return String.join(" ",result);
470 }
471 int fixed = counts.get(command);
472 String[] tokens = line.split(" ",fixed+2);
473 if (tokens.length < fixed+2) {
474 fail("missing argument");
475 }
476 String[] arguments = Arrays.copyOfRange(tokens,1,fixed+1);
477 String text = tokens[fixed+1];
478 int systemCount = command.equals("NORMALIZE") || command.equals("DETECT") ? 0 :
479 Set.of("CONVERT","DIFF","SAME").contains(command) ? 2 : 1;
480 for (int i = 0; i < systemCount; i++) {
481 if (!SYSTEMS.contains(arguments[i])) {
482 fail("unknown system "+arguments[i]);
483 }
484 }
485 List<String> searchPattern = command.equals("FIND") ? patternMorae(arguments[1],arguments[0]) : List.of();
486 boolean slash = command.equals("SAME") || command.equals("HAIKU");
487 for (int i = 0; i < text.length(); i++) {
488 char c = text.charAt(i);
489 if (!letter(c) && c != '\'' && c != '-' && " 0123456789.,!?;:()\"".indexOf(c) < 0 && !(slash && c == '/')) {
490 column(i+1,"invalid character");
491 }
492 }
493 String[] poem = text.split("/",-1);
494 if (command.equals("SAME") && (poem.length != 2 || scan(poem[0]).isEmpty() || scan(poem[1]).isEmpty())) {
495 fail("expected two texts");
496 }
497 if (command.equals("HAIKU")) {
498 if (poem.length != 3 && poem.length != 5) {
499 fail("expected 3 or 5 lines");
500 }
501 for (String part : poem) {
502 if (scan(part).isEmpty()) {
503 fail("expected 3 or 5 lines");
504 }
505 }
506 }
507 List<Word> words = scan(text);
508 if (words.isEmpty()) {
509 fail("no words");
510 }
511 for (int i = 0; i < words.size(); i++) {
512 Word word = words.get(i);
513 int pattern = pattern(word.text());
514 if (pattern < 0) {
515 column(word.start()+1,"mixed case");
516 }
517 words.set(i,new Word(word.text(),word.start(),pattern,List.of()));
518 }
519 if (command.equals("DETECT")) {
520 return detect(words);
521 }
522 for (int i = 0; i < words.size(); i++) {
523 Word word = words.get(i);
524 String source = command.equals("NORMALIZE") ? "ANY" :
525 command.equals("SAME") && word.start() > text.indexOf('/') ? arguments[1] : arguments[0];
526 words.set(i,new Word(word.text(),word.start(),word.pattern(),parse(word,source)));
527 }
528 String system = command.equals("NORMALIZE") ? "HEPBURN" : arguments[0];
529 switch (command) {
530 case "CONVERT", "NORMALIZE", "REVERSE": {
531 String target = command.equals("CONVERT") ? arguments[1] : system;
532 StringBuilder result = new StringBuilder();
533 int cursor = 0;
534 for (Word word : words) {
535 result.append(text,cursor,word.start());
536 Word written = word;
537 if (command.equals("REVERSE")) {
538 List<Mora> reversed = new ArrayList<>();
539 for (int i = 0; i < word.morae().size(); i++) {
540 Mora mora = word.morae().get(i);
541 reversed.add(mora.id().equals("L") ? new Mora(String.valueOf(last(word.morae().get(i-1).id())),mora.column()) : mora);
542 }
543 Collections.reverse(reversed);
544 written = new Word(word.text(),word.start(),word.pattern(),reversed);
545 }
546 result.append(write(written,target,longSetting,caseSetting));
547 cursor = word.start()+word.text().length();
548 }
549 result.append(text.substring(cursor));
550 return result.toString();
551 }
552 case "VALID": {
553 for (Word word : words) {
554 String expected = canonical(word,system);
555 if (!word.text().equals(expected)) {
556 int i = 0;
557 while (i < Math.min(word.text().length(),expected.length()) && word.text().charAt(i) == expected.charAt(i)) {
558 i++;
559 }
560 return "INVALID col "+(word.start()+i+1)+": expected "+expected;
561 }
562 }
563 return "VALID";
564 }
565 case "DIFF": {
566 List<String> changes = new ArrayList<>();
567 for (Word word : words) {
568 String a = write(word,system,longSetting,caseSetting);
569 String b = write(word,arguments[1],longSetting,caseSetting);
570 if (!a.equals(b)) {
571 changes.add(a+">"+b);
572 }
573 }
574 return changes.size()+" of "+words.size()+(changes.isEmpty() ? "" : ": "+String.join(" ",changes));
575 }
576 case "SAME": {
577 List<Word> left = words.stream().filter(w -> w.start() < text.indexOf('/')).toList();
578 List<Word> right = words.stream().filter(w -> w.start() > text.indexOf('/')).toList();
579 boolean exact = true;
580 for (int i = 0; i < Math.min(left.size(),right.size()); i++) {
581 if (!loose(left.get(i)).equals(loose(right.get(i)))) {
582 return "DIFFERENT at word "+(i+1);
583 }
584 if (!ids(left.get(i)).equals(ids(right.get(i)))) {
585 exact = false;
586 }
587 }
588 if (left.size() != right.size()) {
589 return "DIFFERENT at word "+(Math.min(left.size(),right.size())+1);
590 }
591 return exact ? "EXACT" : "LOOSE";
592 }
593 case "MORAE": {
594 List<String> result = new ArrayList<>();
595 int count = 0;
596 for (Word word : words) {
597 List<String> pieces = new ArrayList<>();
598 for (Mora mora : word.morae()) {
599 pieces.add(regular(mora.id()) ? spelling(system,mora.id()) : mora.id().equals("L") ? "-" : mora.id());
600 }
601 count += pieces.size();
602 result.add(String.join(".",pieces));
603 }
604 return count+": "+String.join(" ",result);
605 }
606 case "SYLLABLES": {
607 List<String> result = new ArrayList<>();
608 int count = 0;
609 for (Word word : words) {
610 List<String> pieces = parts(word,system,"KEEP"), syllables = new ArrayList<>();
611 LongInfo info = longInfo(word.morae());
612 for (int i = 0; i < pieces.size(); i++) {
613 if (i == 0 || regular(word.morae().get(i).id()) && !info.paired()[i]) {
614 syllables.add(pieces.get(i));
615 } else {
616 syllables.set(syllables.size()-1,syllables.getLast()+pieces.get(i));
617 }
618 }
619 count += syllables.size();
620 result.add(String.join(".",syllables));
621 }
622 return count+": "+String.join(" ",result);
623 }
624 case "ROW": {
625 List<String> result = new ArrayList<>();
626 for (Word word : words) {
627 for (Mora mora : word.morae()) {
628 Meta meta = metadata.get(mora.id());
629 result.add(meta == null ? (mora.id().equals("L") ? "-" : mora.id())+": special" :
630 spelling(system,meta.id())+": "+meta.row()+" "+meta.dan());
631 }
632 }
633 return String.join(", ",result);
634 }
635 case "STATS": {
636 Map<String,Integer> rowCounts = new HashMap<>();
637 int count = 0, voiced = 0, semivoiced = 0, contracted = 0, n = 0, q = 0, l = 0;
638 for (Word word : words) {
639 for (Mora mora : word.morae()) {
640 count++;
641 Meta meta = metadata.get(mora.id());
642 if (meta == null) {
643 if (mora.id().equals("N")) {
644 n++;
645 }
646 if (mora.id().equals("Q")) {
647 q++;
648 }
649 if (mora.id().equals("L")) {
650 l++;
651 }
652 } else {
653 rowCounts.merge(baseRow(meta.row()),1,Integer::sum);
654 if (voice(meta.row()) == 1) {
655 voiced++;
656 }
657 if (voice(meta.row()) == 2) {
658 semivoiced++;
659 }
660 if (meta.contracted()) {
661 contracted++;
662 }
663 }
664 }
665 }
666 List<String> result = new ArrayList<>(List.of("morae="+count));
667 for (String row : ROWS.subList(0,10)) {
668 if (rowCounts.containsKey(row)) {
669 result.add(row+"="+rowCounts.get(row));
670 }
671 }
672 result.addAll(List.of("voiced="+voiced,"semivoiced="+semivoiced,"contracted="+contracted,"N="+n,"Q="+q,"L="+l));
673 return String.join(" ",result);
674 }
675 case "SORT", "INDEX": {
676 List<Word> sorted = sorted(words);
677 if (command.equals("SORT")) {
678 return String.join(" ",sorted.stream().map(Word::text).toList());
679 }
680 List<String> groups = new ArrayList<>(), group = new ArrayList<>();
681 String previous = "";
682 for (Word word : sorted) {
683 String base = elements(word).getFirst().base();
684 String heading = orderSetting.equals("IROHA") || base.equals("N") ? base : metadata.get(base).row();
685 if (!heading.equals(previous) && !group.isEmpty()) {
686 groups.add(previous+": "+String.join(" ",group));
687 group.clear();
688 }
689 previous = heading;
690 group.add(word.text());
691 }
692 groups.add(previous+": "+String.join(" ",group));
693 return String.join("; ",groups);
694 }
695 case "FIND": {
696 List<String> found = words.stream().filter(w -> matches(searchPattern,ids(w))).map(Word::text).toList();
697 return found.size()+" of "+words.size()+(found.isEmpty() ? "" : ": "+String.join(" ",found));
698 }
699 case "HAIKU": {
700 int[] totals = new int[poem.length];
701 for (Word word : words) {
702 int index = (int)text.substring(0,word.start()).chars().filter(c -> c == '/').count();
703 totals[index] += word.morae().size();
704 }
705 int[] expected = poem.length == 3 ? new int[]{5,7,5} : new int[]{5,7,5,7,7};
706 return (Arrays.equals(totals,expected) ? "OK " : "NG ")+(poem.length == 3 ? "haiku " : "tanka ")+
707 String.join("-",Arrays.stream(totals).mapToObj(String::valueOf).toList());
708 }
709 case "KAIBUN": {
710 List<String> bases = words.stream().flatMap(w -> elements(w).stream()).map(Element::base).toList();
711 List<String> reversed = bases.reversed();
712 return (bases.equals(reversed) ? "YES " : "NO ")+String.join(".",bases);
713 }
714 case "SHIRITORI": {
715 Set<List<String>> seen = new HashSet<>();
716 String previous = "";
717 for (int i = 0; i < words.size(); i++) {
718 Word word = words.get(i);
719 List<Element> elements = elements(word);
720 String head = elements.getFirst().base(), tail = elements.getLast().base();
721 String prefix = "NG "+(i+1)+": "+word.text();
722 if (i > 0 && !head.equals(previous)) {
723 return prefix+" must start with "+spelling(system,previous);
724 }
725 if (!seen.add(ids(word))) {
726 return prefix+" is repeated";
727 }
728 if (tail.equals("N")) {
729 return prefix+" ends with N";
730 }
731 previous = tail;
732 }
733 return "OK "+words.size();
734 }
735 default: throw new IllegalStateException("Unreachable command");
736 }
737 }

748 }
