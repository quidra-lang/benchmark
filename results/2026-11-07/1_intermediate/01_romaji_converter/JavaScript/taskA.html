42function spelling(system, id) {
43 const hepburn = {
44 si: "shi", ti: "chi", tu: "tsu", hu: "fu", zi: "ji", di: "ji",
45 du: "zu", wi: "i", we: "e", wo: "o"
46 };
47 const kunrei = {
48 di: "zi", du: "zu", wi: "i", we: "e", wo: "o",
49 dya: "zya", dyu: "zyu", dyo: "zyo"
50 };
51 const wapuro = {
52 si: "shi", ti: "chi", tu: "tsu", hu: "fu", zi: "ji"
53 };
54 if (system === "KUNREI") {
55 return kunrei[id] ?? id;
56 }
57 if (system === "NIHON") {
58 return id;
59 }
60 const replacements = system === "WAPURO" ? wapuro : hepburn;
61 if (replacements[id] !== undefined) {
62 return replacements[id];
63 }
64 if (id.startsWith("sy")) {
65 return "sh" + id.slice(2);
66 }
67 if (id.startsWith("ty")) {
68 return "ch" + id.slice(2);
69 }
70 if (id.startsWith("zy") || (system !== "WAPURO" && id.startsWith("dy"))) {
71 return "j" + id.slice(2);
72 }
73 return id;
74}

97function casePattern(word) {
98 if (!/[A-Z]/.test(word)) {
99 return "LOWER";
100 }
101 if (/^[A-Z]/.test(word) && !/[A-Z]/.test(word.slice(1))) {
102 return "TITLE";
103 }
104 const letters = word.replace(/[^A-Za-z]/g, "");
105 if (letters.length >= 2 && !/[a-z]/.test(letters)) {
106 return "UPPER";
107 }
108 return "";
109}

118function parseWord(word, system) {
119 const text = word.original.toLowerCase();
120 const result = [];
121 let position = 0;
122 while (position < text.length) {
123 const c = text[position];
124 const c2 = text[position + 1] ?? "";
125 const column = word.start + position + 1;
126 let id = "";
127 let length = 1;
128 if (c === "-") {
129 if (result.length === 0 || !info.has(result[result.length - 1].id)) {
130 at(column, "misplaced long mark");
131 }
132 id = "-";
133 } else if (c === "'") {
134 at(column, "misplaced apostrophe");
135 } else {
136 const match = matchAt(text, position, system);
137 if (match) {
138 id = match.id;
139 length = match.spelling.length;
140 } else if ((system === "WAPURO" || system === "NORMALIZE") && text.startsWith("xtu", position)) {
141 id = "Q";
142 length = 3;
143 } else if (c === "n") {
144 id = "N";
145 if (system === "NORMALIZE") {
146 const third = text[position + 2] ?? "";
147 if (c2 === "'" || (c2 === "n" && !/^[aiueoy']$/.test(third))) {
148 length = 2;
149 }
150 } else if ((system === "WAPURO" && c2 === "n") || (system !== "WAPURO" && c2 === "'")) {
151 length = 2;
152 }
153 } else if ((system === "TRADITIONAL" || system === "NORMALIZE") && c === "m" && /^[bmp]$/.test(c2)) {
154 id = "N";
155 } else if (/^[a-z]$/.test(c) && !/[aiueon]/.test(c) && c2 === c && matchAt(text, position + 1, system)) {
156 id = "Q";
157 } else if (["HEPBURN", "TRADITIONAL", "NORMALIZE"].includes(system) && c === "t" &&
158 text.startsWith("ch", position + 1) && matchAt(text, position + 1, system)) {
159 id = "Q";
160 } else {
161 at(column, "unknown syllable");
162 }
163 }
164 result.push({ id, column });
165 position += length;
166 }
167 return result;
168}

170function longData(morae) {
171 const pairs = new Set();
172 const elements = new Map();
173 for (let i = 0; i < morae.length; i++) {
174 if (morae[i].id === "-") {
175 elements.set(i, morae[i - 1].id.slice(-1));
176 }
177 }
178 let i = 0;
179 while (i + 1 < morae.length) {
180 const current = morae[i].id;
181 const next = morae[i + 1].id;
182 const vowel = current.slice(-1);
183 if (info.has(current) && /^[aiueo]$/.test(next) &&
184 (vowel === next || (vowel === "o" && next === "u") || (vowel === "e" && next === "i"))) {
185 pairs.add(i + 1);
186 if ((vowel === "o" && (next === "o" || next === "u")) || (vowel === "u" && next === "u")) {
187 elements.set(i + 1, vowel);
188 }
189 i += 2;
190 } else {
191 i++;
192 }
193 }
194 return { pairs, elements };
195}

197function writtenParts(morae, system, longMode) {
198 const data = longData(morae);
199 const parts = [];
200 for (let i = 0; i < morae.length; i++) {
201 const mora = morae[i];
202 const next = morae[i + 1]?.id ?? "";
203 const nextSpelling = info.has(next) ? spelling(system, next) : "";
204 const first = nextSpelling[0] ?? "";
205 let part = "";
206 if (data.elements.has(i)) {
207 const vowel = data.elements.get(i);
208 if (longMode === "KEEP") {
209 part = mora.id === "-" ? "-" : spelling(system, mora.id);
210 } else if (longMode === "PASSPORT") {
211 part = vowel === "o" && !/^[aiueo]$/.test(first) ? "h" : "";
212 } else if (longMode === "HYPHEN") {
213 part = "-";
214 } else if (longMode === "DOUBLE") {
215 part = vowel;
216 }
217 } else if (info.has(mora.id)) {
218 part = spelling(system, mora.id);
219 } else if (mora.id === "N") {
220 if (system === "WAPURO") {
221 part = next === "N" || /^[aiueoyn]$/.test(first) ? "nn" : "n";
222 } else if (system === "TRADITIONAL" && /^[bmp]$/.test(first)) {
223 part = "m";
224 } else {
225 part = /^[aiueoy]$/.test(first) ? "n'" : "n";
226 }
227 } else if (mora.id === "Q") {
228 if (first && !/[aiueon]/.test(first)) {
229 part = ["HEPBURN", "TRADITIONAL"].includes(system) && nextSpelling.startsWith("ch") ? "t" : first;
230 } else if (system === "WAPURO") {
231 part = "xtu";
232 } else {
233 at(mora.column, "small tsu cannot be written");
234 }
235 }
236 parts.push(part);
237 }
238 return parts;
239}

241function writeWord(word, system, settings, canonical = false, morae = word.morae) {
242 const value = writtenParts(morae, system, canonical ? "KEEP" : settings.LONG).join("");
243 const pattern = canonical || settings.CASE === "KEEP" ? word.pattern : settings.CASE;
244 if (pattern === "UPPER") {
245 return value.toUpperCase();
246 }
247 if (pattern === "TITLE") {
248 return value[0].toUpperCase() + value.slice(1);
249 }
250 return value;
251}

263function elements(morae) {
264 const result = [];
265 for (const mora of morae) {
266 if (mora.id === "-") {
267 continue;
268 }
269 if (mora.id === "N" || mora.id === "Q") {
270 result.push({ base: mora.id === "N" ? "N" : "tu", voice: 0, size: mora.id === "Q" ? 1 : 0 });
271 continue;
272 }
273 const item = info.get(mora.id);
274 const voice = ["ga", "za", "da", "ba"].includes(item.row) ? 1 : item.row === "pa" ? 2 : 0;
275 const base = item.contracted ? devoice(item.row[0] + "i") : devoice(mora.id);
276 result.push({ base, voice, size: 0 });
277 if (item.contracted) {
278 result.push({ base: item.dan, voice: 0, size: 1 });
279 }
280 }
281 return result;
282}

289function compareSequence(a, b) {
290 for (let i = 0; i < Math.min(a.length, b.length); i++) {
291 if (a[i] !== b[i]) {
292 return a[i] - b[i];
293 }
294 }
295 return a.length - b.length;
296}

298function sortedWords(words, order) {
299 const ranks = new Map(orders[order].map((base, rank) => [base, rank]));
300 const entries = words.map(word => {
301 const cols = elements(word.morae);
302 const keys = [
303 cols.map(item => ranks.get(item.base)),
304 cols.map(item => item.voice),
305 cols.map(item => item.size),
306 word.morae.map(mora => mora.id === "-" ? 1 : 0)
307 ];
308 return { word, keys };
309 });
310 entries.sort((a, b) => {
311 for (let i = 0; i < a.keys.length; i++) {
312 const difference = compareSequence(a.keys[i], b.keys[i]);
313 if (difference !== 0) {
314 return difference;
315 }
316 }
317 return 0;
318 });
319 return entries.map(entry => entry.word);
320}

322function parsePattern(pattern, system) {
323 if (!/^[a-z'?*-]+$/.test(pattern)) {
324 fail("invalid pattern");
325 }
326 const result = [];
327 try {
328 for (const part of pattern.split(/([?*])/)) {
329 if (part === "?" || part === "*") {
330 result.push(part);
331 } else if (part) {
332 const word = { original: part, start: 0, pattern: "LOWER", morae: [] };
333 result.push(...parseWord(word, system).map(mora => mora.id));
334 }
335 }
336 } catch (error) {
337 if (!(error instanceof Invalid)) {
338 throw error;
339 }
340 fail("invalid pattern");
341 }
342 return result;
343}

345function matchesPattern(pattern, morae) {
346 let previous = new Array(morae.length + 1).fill(false);
347 previous[0] = true;
348 for (const token of pattern) {
349 const current = new Array(morae.length + 1).fill(false);
350 if (token === "*") {
351 current[0] = previous[0];
352 }
353 for (let j = 1; j <= morae.length; j++) {
354 current[j] = token === "*" ? previous[j] || current[j - 1] :
355 previous[j - 1] && (token === "?" || token === morae[j - 1].id);
356 }
357 previous = current;
358 }
359 return previous[morae.length];
360}

377function command(line) {
378 const name = line.split(" ", 1)[0];
379 if (!Object.hasOwn(argumentCounts, name)) {
380 fail("unknown command " + name);
381 }
382 const count = argumentCounts[name];
383 const args = [];
384 let text = "";
385 if (name === "SET" || name === "TABLE") {
386 const parts = line.split(" ");
387 if (parts.length < 3) {
388 fail("missing argument");
389 }
390 if (parts.length > 3) {
391 fail("too many arguments");
392 }
393 args.push(...parts.slice(1));
394 } else {
395 let position = 0;
396 for (let i = 0; i <= count; i++) {
397 const space = line.indexOf(" ", position);
398 if (space < 0) {
399 fail("missing argument");
400 }
401 if (i > 0) {
402 args.push(line.slice(position, space));
403 }
404 position = space + 1;
405 }
406 text = line.slice(position);
407 }
408 const systemCount = ["NORMALIZE", "DETECT", "SET"].includes(name) ? 0 :
409 ["CONVERT", "DIFF", "SAME"].includes(name) ? 2 : 1;
410 for (const system of args.slice(0, systemCount)) {
411 if (!systems.includes(system)) {
412 fail("unknown system " + system);
413 }
414 }
415 const system = name === "NORMALIZE" ? "NORMALIZE" : args[0];
416 if (name === "SET") {
417 const choices = {
418 LONG: ["KEEP", "DROP", "PASSPORT", "HYPHEN", "DOUBLE"],
419 CASE: ["KEEP", "LOWER", "UPPER", "TITLE"], ORDER: ["GOJUON", "IROHA"]
420 };
421 if (!Object.hasOwn(choices, args[0])) {
422 fail("unknown option " + args[0]);
423 }
424 if (!choices[args[0]].includes(args[1])) {
425 fail("unknown value " + args[1]);
426 }
427 settings[args[0]] = args[1];
428 return args[0] + "=" + args[1];
429 }
430 if (name === "TABLE") {
431 if (!rowData.some(([row]) => row === args[1])) {
432 fail("unknown row " + args[1]);
433 }
434 return table.filter(item => item.row === args[1]).map(item => spelling(system, item.id)).join(" ");
435 }
436 const pattern = name === "FIND" ? parsePattern(args[1], system) : [];
437 const allowSlash = name === "HAIKU" || name === "SAME";
438 for (let i = 0; i < text.length; i++) {
439 if (!/[A-Za-z0-9' .,! ?;:()"-]/.test(text[i]) && !(allowSlash && text[i] === "/")) {
440 at(i + 1, "invalid character");
441 }
442 }
443 const pieces = text.split("/");
444 if (name === "SAME" && (pieces.length !== 2 || pieces.some(piece => scanWords(piece).length === 0))) {
445 fail("expected two texts");
446 }
447 if (name === "HAIKU" && (![3, 5].includes(pieces.length) || pieces.some(piece => scanWords(piece).length === 0))) {
448 fail("expected 3 or 5 lines");
449 }
450 const words = scanWords(text);
451 if (words.length === 0) {
452 fail("no words");
453 }
454 for (const word of words) {
455 if (!word.pattern) {
456 at(word.start + 1, "mixed case");
457 }
458 }
459 if (name === "DETECT") {
460 return systems.map(source => {
461 let verdict = "canonical";
462 try {
463 for (const word of words) {
464 word.morae = parseWord(word, source);
465 }
466 if (words.some(word => word.original !== writeWord(word, source, settings, true))) {
467 verdict = "readable";
468 }
469 } catch (error) {
470 if (!(error instanceof Invalid)) {
471 throw error;
472 }
473 verdict = "error";
474 }
475 return source + "=" + verdict;
476 }).join(" ");
477 }
478 const slash = text.indexOf("/");
479 for (const word of words) {
480 const source = name === "SAME" && word.start > slash ? args[1] : system;
481 word.morae = parseWord(word, source);
482 }
483 if (name === "CONVERT" || name === "NORMALIZE" || name === "REVERSE") {
484 const target = name === "NORMALIZE" ? "HEPBURN" : name === "CONVERT" ? args[1] : system;
485 return replaceWords(text, words, word => {
486 let morae = word.morae;
487 if (name === "REVERSE") {
488 morae = morae.map((mora, i) => ({
489 id: mora.id === "-" ? word.morae[i - 1].id.slice(-1) : mora.id,
490 column: mora.column
491 })).reverse();
492 }
493 return writeWord(word, target, settings, false, morae);
494 });
495 }
496 if (name === "VALID") {
497 for (const word of words) {
498 const canonical = writeWord(word, system, settings, true);
499 if (canonical !== word.original) {
500 let i = 0;
501 while (i < Math.min(canonical.length, word.original.length) && canonical[i] === word.original[i]) {
502 i++;
503 }
504 return "INVALID col " + (word.start + i + 1) + ": expected " + canonical;
505 }
506 }
507 return "VALID";
508 }
509 if (name === "DIFF") {
510 const differences = [];
511 for (const word of words) {
512 const a = writeWord(word, system, settings);
513 const b = writeWord(word, args[1], settings);
514 if (a !== b) {
515 differences.push(a + ">" + b);
516 }
517 }
518 return counted(differences, words.length);
519 }
520 if (name === "SAME") {
521 const left = words.filter(word => word.start < slash);
522 const right = words.filter(word => word.start > slash);
523 for (let i = 0; i < Math.min(left.length, right.length); i++) {
524 const loose = (word) => {
525 const data = longData(word.morae);
526 return ids(word.morae.filter((_, index) => !data.elements.has(index)));
527 };
528 if (loose(left[i]) !== loose(right[i])) {
529 return "DIFFERENT at word " + (i + 1);
530 }
531 }
532 if (left.length !== right.length) {
533 return "DIFFERENT at word " + (Math.min(left.length, right.length) + 1);
534 }
535 return left.every((word, i) => ids(word.morae) === ids(right[i].morae)) ? "EXACT" : "LOOSE";
536 }
537 if (name === "MORAE") {
538 const total = words.reduce((sum, word) => sum + word.morae.length, 0);
539 return total + ": " + words.map(word => word.morae.map(mora =>
540 info.has(mora.id) ? spelling(system, mora.id) : mora.id).join(".")).join(" ");
541 }
542 if (name === "SYLLABLES") {
543 let total = 0;
544 const displays = words.map(word => {
545 const data = longData(word.morae);
546 const parts = writtenParts(word.morae, system, "KEEP");
547 const groups = [];
548 for (let i = 0; i < word.morae.length; i++) {
549 if (i === 0 || (info.has(word.morae[i].id) && !data.pairs.has(i))) {
550 groups.push(parts[i]);
551 } else {
552 groups[groups.length - 1] += parts[i];
553 }
554 }
555 total += groups.length;
556 return groups.join(".");
557 });
558 return total + ": " + displays.join(" ");
559 }
560 if (name === "ROW") {
561 return words.flatMap(word => word.morae.map(mora => {
562 const item = info.get(mora.id);
563 return item ? spelling(system, mora.id) + ": " + item.row + " " + item.dan : mora.id + ": special";
564 })).join(", ");
565 }
566 if (name === "STATS") {
567 const counts = new Map();
568 const increment = (key) => {
569 counts.set(key, (counts.get(key) ?? 0) + 1);
570 };
571 const all = words.flatMap(word => word.morae);
572 for (const mora of all) {
573 const item = info.get(mora.id);
574 if (!item) {
575 increment(mora.id === "-" ? "L" : mora.id);
576 } else {
577 increment(devoice(item.row));
578 if (["ga", "za", "da", "ba"].includes(item.row)) {
579 increment("voiced");
580 }
581 if (item.row === "pa") {
582 increment("semivoiced");
583 }
584 if (item.contracted) {
585 increment("contracted");
586 }
587 }
588 }
589 const output = ["morae=" + all.length];
590 for (const row of baseRows) {
591 if (counts.has(row)) {
592 output.push(row + "=" + counts.get(row));
593 }
594 }
595 for (const key of ["voiced", "semivoiced", "contracted", "N", "Q", "L"]) {
596 output.push(key + "=" + (counts.get(key) ?? 0));
597 }
598 return output.join(" ");
599 }
600 if (name === "SORT" || name === "INDEX") {
601 const sorted = sortedWords(words, settings.ORDER);
602 if (name === "SORT") {
603 return sorted.map(word => word.original).join(" ");
604 }
605 const groups = [];
606 for (const word of sorted) {
607 const base = elements(word.morae)[0].base;
608 const heading = settings.ORDER === "IROHA" || base === "N" ? base : info.get(base).row;
609 if (groups.length === 0 || groups[groups.length - 1].heading !== heading) {
610 groups.push({ heading, words: [] });
611 }
612 groups[groups.length - 1].words.push(word.original);
613 }
614 return groups.map(group => group.heading + ": " + group.words.join(" ")).join("; ");
615 }
616 if (name === "FIND") {
617 return counted(words.filter(word => matchesPattern(pattern, word.morae)).map(word => word.original), words.length);
618 }
619 if (name === "HAIKU") {
620 const counts = new Array(pieces.length).fill(0);
621 for (const word of words) {
622 const lineIndex = text.slice(0, word.start).split("/").length - 1;
623 counts[lineIndex] += word.morae.length;
624 }
625 const expected = pieces.length === 3 ? [5, 7, 5] : [5, 7, 5, 7, 7];
626 const verdict = counts.every((count, i) => count === expected[i]) ? "OK" : "NG";
627 return verdict + (pieces.length === 3 ? " haiku " : " tanka ") + counts.join("-");
628 }
629 if (name === "KAIBUN") {
630 const sequence = words.flatMap(word => elements(word.morae).map(item => item.base));
631 const palindrome = sequence.every((base, i) => base === sequence[sequence.length - 1 - i]);
