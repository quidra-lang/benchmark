4class Romaji
5 SYSTEMS = %w[HEPBURN TRADITIONAL KUNREI NIHON WAPURO].freeze
6 BASE_ROWS = %w[a ka sa ta na ha ma ya ra wa].freeze
7 ROW_DATA = [
8 ["a", %w[a i u e o]], ["ka", %w[ka ki ku ke ko]],
9 ["sa", %w[sa si su se so]], ["ta", %w[ta ti tu te to]],
10 ["na", %w[na ni nu ne no]], ["ha", %w[ha hi hu he ho]],
11 ["ma", %w[ma mi mu me mo]], ["ya", %w[ya yu yo]],
12 ["ra", %w[ra ri ru re ro]], ["wa", %w[wa wi we wo]],
13 ["ga", %w[ga gi gu ge go]], ["za", %w[za zi zu ze zo]],
14 ["da", %w[da di du de do]], ["ba", %w[ba bi bu be bo]],
15 ["pa", %w[pa pi pu pe po]]
16 ].freeze
17 CONTRACTED_ROWS = %w[ka sa ta na ha ma ra ga za da ba pa].freeze
18 HEPBURN = {
19 "si" => "shi", "ti" => "chi", "tu" => "tsu", "hu" => "fu",
20 "zi" => "ji", "di" => "ji", "du" => "zu", "wi" => "i",
21 "we" => "e", "wo" => "o"

23 KUNREI = {
24 "di" => "zi", "du" => "zu", "wi" => "i", "we" => "e",
25 "wo" => "o", "dya" => "zya", "dyu" => "zyu", "dyo" => "zyo"
26 }.freeze
27 WAPURO = {
28 "si" => "shi", "ti" => "chi", "tu" => "tsu", "hu" => "fu", "zi" => "ji"
29 }.freeze
30 ORDERS = {
31 "GOJUON" => %w[a i u e o ka ki ku ke ko sa si su se so ta ti tu te to
32 na ni nu ne no ha hi hu he ho ma mi mu me mo ya yu yo
33 ra ri ru re ro wa wi we wo N],
34 "IROHA" => %w[i ro ha ni ho he to ti ri nu ru wo wa ka yo ta re so tu ne
35 na ra mu u wi no o ku ya ma ke hu ko e te a sa ki yu me mi
36 si we hi mo se su N]
37 }.freeze
38 ARGUMENT_COUNTS = {
39 "SET" => 2, "TABLE" => 2, "CONVERT" => 2, "NORMALIZE" => 0,
40 "REVERSE" => 1, "VALID" => 1, "DETECT" => 0, "DIFF" => 2,
41 "SAME" => 2, "MORAE" => 1, "SYLLABLES" => 1, "ROW" => 1,

45 CHOICES = {
46 "LONG" => %w[KEEP DROP PASSPORT HYPHEN DOUBLE],
47 "CASE" => %w[KEEP LOWER UPPER TITLE],
48 "ORDER" => %w[GOJUON IROHA]

51 def initialize
52 @table = []
53 @info = {}
54 ROW_DATA.each do |row, ids|
55 ids.each do |id|
56 item = { id: id, row: row, dan: id[-1], contracted: false }
57 @table << item
58 @info[id] = item
59 end
60 end
61 CONTRACTED_ROWS.each do |row|
62 %w[ya yu yo].each do |dan|
63 id = row[0] + dan
64 item = { id: id, row: row, dan: dan, contracted: true }
65 @table << item
66 @info[id] = item
67 end
68 end
69 @lexicons = {}
70 (SYSTEMS + ["NORMALIZE"]).each do |system|
71 sources = system == "NORMALIZE" ? %w[HEPBURN KUNREI NIHON WAPURO] : [system]
72 entries = {}
73 @table.each do |item|
74 sources.each do |source|
75 text = spelling(source, item[:id])
76 entries[text] = item[:id] unless entries.key?(text)
77 end
78 end
79 @lexicons[system] = entries.to_a.sort_by { |text, _| -text.length }
80 end
81 @settings = { "LONG" => "KEEP", "CASE" => "KEEP", "ORDER" => "GOJUON" }
82 end

92 def spelling(system, id)
93 return KUNREI.fetch(id, id) if system == "KUNREI"
94 return id if system == "NIHON"
95
96 replacements = system == "WAPURO" ? WAPURO : HEPBURN
97 return replacements[id] if replacements.key?(id)
98 return "sh" + id[2..] if id.start_with?("sy")
99 return "ch" + id[2..] if id.start_with?("ty")
100 if id.start_with?("zy") || (system != "WAPURO" && id.start_with?("dy"))
101 return "j" + id[2..]
102 end
103 id
104 end

106 def match_at(text, position, system)
107 @lexicons[system].find { |spelling, _| text[position, spelling.length] == spelling }
108 end

110 def case_pattern(word)
111 return "LOWER" unless word.match?(/[A-Z]/)
112 return "TITLE" if word.match?(/\A[A-Z]/) && !word[1..].match?(/[A-Z]/)
113
114 letters = word.delete("^A-Za-z")
115 return "UPPER" if letters.length >= 2 && !letters.match?(/[a-z]/)
116
117 nil
118 end

120 def scan_words(text)
121 words = []
122 position = 0
123 while (match = /[A-Za-z'-]+/.match(text, position))
124 original = match[0]
125 words << {
126 original: original, start: match.begin(0),
127 pattern: case_pattern(original), morae: []
128 }
129 position = match.end(0)
130 end
131 words
132 end

134 def parse_word(word, system)
135 text = word[:original].downcase
136 result = []
137 position = 0
138 while position < text.length
139 c = text[position]
140 c2 = text[position + 1] || ""
141 column = word[:start] + position + 1
142 length = 1
143 if c == "-"
144 if result.empty? || !@info.key?(result[-1][:id])
145 at(column, "misplaced long mark")
146 end
147 id = "-"
148 elsif c == "'"
149 at(column, "misplaced apostrophe")
150 else
151 match = match_at(text, position, system)
152 if match
153 id = match[1]
154 length = match[0].length
155 elsif %w[WAPURO NORMALIZE].include?(system) && text[position, 3] == "xtu"
156 id = "Q"
157 length = 3
158 elsif c == "n"
159 id = "N"
160 if system == "NORMALIZE"
161 third = text[position + 2] || ""
162 if c2 == "'" || (c2 == "n" && !third.match?(/\A[aiueoy']\z/))
163 length = 2
164 end
165 elsif (system == "WAPURO" && c2 == "n") || (system != "WAPURO" && c2 == "'")
166 length = 2
167 end
168 elsif %w[TRADITIONAL NORMALIZE].include?(system) && c == "m" && c2.match?(/\A[bmp]\z/)
169 id = "N"
170 elsif c.match?(/\A[a-z]\z/) && !c.match?(/[aiueon]/) &&
171 c2 == c && match_at(text, position + 1, system)
172 id = "Q"
173 elsif %w[HEPBURN TRADITIONAL NORMALIZE].include?(system) && c == "t" &&
174 text[position + 1, 2] == "ch" && match_at(text, position + 1, system)
175 id = "Q"
176 else
177 at(column, "unknown syllable")
178 end
179 end
180 result << { id: id, column: column }
181 position += length
182 end
183 result
184 end

186 def long_data(morae)
187 pairs = {}
188 elements = {}
189 morae.each_with_index do |mora, i|
190 elements[i] = morae[i - 1][:id][-1] if mora[:id] == "-"
191 end
192 i = 0
193 while i + 1 < morae.length
194 current = morae[i][:id]
195 following = morae[i + 1][:id]
196 vowel = current[-1]
197 if @info.key?(current) && following.match?(/\A[aiueo]\z/) &&
198 (vowel == following || (vowel == "o" && following == "u") ||
199 (vowel == "e" && following == "i"))
200 pairs[i + 1] = true
201 if (vowel == "o" && %w[o u].include?(following)) || (vowel == "u" && following == "u")
202 elements[i + 1] = vowel
203 end
204 i += 2
205 else
206 i += 1
207 end
208 end
209 [pairs, elements]
210 end

212 def written_parts(morae, system, long_mode)
213 _, elements = long_data(morae)
214 morae.each_with_index.map do |mora, i|
215 following = morae[i + 1]&.fetch(:id) || ""
216 next_spelling = @info.key?(following) ? spelling(system, following) : ""
217 first = next_spelling[0] || ""
218 if elements.key?(i)
219 vowel = elements[i]
220 case long_mode
221 when "KEEP"
222 mora[:id] == "-" ? "-" : spelling(system, mora[:id])
223 when "PASSPORT"
224 vowel == "o" && !first.match?(/\A[aiueo]\z/) ? "h" : ""
225 when "HYPHEN"
226 "-"
227 when "DOUBLE"
228 vowel
229 else
230 ""
231 end
232 elsif @info.key?(mora[:id])
233 spelling(system, mora[:id])
234 elsif mora[:id] == "N"
235 if system == "WAPURO"
236 following == "N" || first.match?(/\A[aiueoyn]\z/) ? "nn" : "n"
237 elsif system == "TRADITIONAL" && first.match?(/\A[bmp]\z/)
238 "m"
239 else
240 first.match?(/\A[aiueoy]\z/) ? "n'" : "n"
241 end
242 elsif mora[:id] == "Q"
243 if !first.empty? && !first.match?(/[aiueon]/)
244 if %w[HEPBURN TRADITIONAL].include?(system) && next_spelling.start_with?("ch")
245 "t"
246 else
247 first
248 end
249 elsif system == "WAPURO"
250 "xtu"
251 else
252 at(mora[:column], "small tsu cannot be written")
253 end
254 end
255 end
256 end

258 def write_word(word, system, canonical: false, morae: word[:morae])
259 value = written_parts(morae, system, canonical ? "KEEP" : @settings["LONG"]).join
260 pattern = canonical || @settings["CASE"] == "KEEP" ? word[:pattern] : @settings["CASE"]
261 case pattern
262 when "UPPER"
263 value.upcase
264 when "TITLE"
265 value.capitalize
266 else
267 value
268 end
269 end

272 iterator = words.each
273 text.gsub(/[A-Za-z'-]+/) { yield(iterator.next) }

276 def devoice(id)
277 id[0].tr("gzdbp", "ksthh") + id[1..]
278 end

280 def collation_elements(morae)
281 result = []
282 morae.each do |mora|
283 id = mora[:id]
284 next if id == "-"
285
286 if id == "N" || id == "Q"
287 result << { base: id == "N" ? "N" : "tu", voice: 0, size: id == "Q" ? 1 : 0 }
288 next
289 end
290 item = @info[id]
291 voice = if %w[ga za da ba].include?(item[:row])
292 1
293 elsif item[:row] == "pa"
294 2
295 else
296 0
297 end
298 base = item[:contracted] ? devoice(item[:row][0] + "i") : devoice(id)
299 result << { base: base, voice: voice, size: 0 }
300 if item[:contracted]
301 result << { base: item[:dan], voice: 0, size: 1 }
302 end
303 end
304 result
305 end

307 def sorted_words(words)
308 ranks = ORDERS[@settings["ORDER"]].each_with_index.to_h
309 words.each_with_index.sort_by do |word, index|
310 elements = collation_elements(word[:morae])
311 [
312 elements.map { |item| ranks[item[:base]] },
313 elements.map { |item| item[:voice] },
314 elements.map { |item| item[:size] },
315 word[:morae].map { |mora| mora[:id] == "-" ? 1 : 0 },
316 index
317 ]
318 end.map(&:first)
319 end

321 def parse_pattern(pattern, system)
322 fail_with("invalid pattern") unless pattern.match?(/\A[a-z'?*-]+\z/)
323
324 result = []
325 begin
326 pattern.split(/([?*])/, -1).each do |part|
327 if part == "?" || part == "*"
328 result << part
329 elsif !part.empty?
330 word = { original: part, start: 0, pattern: "LOWER", morae: [] }
331 result.concat(parse_word(word, system).map { |mora| mora[:id] })
332 end
333 end
334 rescue Invalid
335 fail_with("invalid pattern")
336 end
337 result
338 end

340 def matches_pattern?(pattern, morae)
341 previous = Array.new(morae.length + 1, false)
342 previous[0] = true
343 pattern.each do |token|
344 current = Array.new(morae.length + 1, false)
345 current[0] = previous[0] if token == "*"
346 (1..morae.length).each do |j|
347 current[j] = if token == "*"
348 previous[j] || current[j - 1]
349 else
350 previous[j - 1] && (token == "?" || token == morae[j - 1][:id])
351 end
352 end
353 previous = current
354 end
355 previous[morae.length]
356 end

366 def command(line)
367 name = line.split(" ", 2)[0]
368 fail_with("unknown command #{name}") unless ARGUMENT_COUNTS.key?(name)
369
370 count = ARGUMENT_COUNTS[name]
371 args = []
372 text = ""
373 if name == "SET" || name == "TABLE"
374 parts = line.split(" ")
375 fail_with("missing argument") if parts.length < 3
376 fail_with("too many arguments") if parts.length > 3
377 args = parts[1..]
378 else
379 position = 0
380 (0..count).each do |i|
381 space = line.index(" ", position)
382 fail_with("missing argument") unless space
383 args << line[position...space] if i.positive?
384 position = space + 1
385 end
386 text = line[position..]
387 end
388 system_count = if %w[NORMALIZE DETECT SET].include?(name)
389 0
390 elsif %w[CONVERT DIFF SAME].include?(name)
391 2
392 else
393 1
394 end
395 args.first(system_count).each do |system|
396 fail_with("unknown system #{system}") unless SYSTEMS.include?(system)
397 end
398 system = name == "NORMALIZE" ? "NORMALIZE" : args[0]
399 if name == "SET"
400 fail_with("unknown option #{args[0]}") unless CHOICES.key?(args[0])
401 fail_with("unknown value #{args[1]}") unless CHOICES[args[0]].include?(args[1])
402 @settings[args[0]] = args[1]
403 return "#{args[0]}=#{args[1]}"
404 end
405 if name == "TABLE"
406 fail_with("unknown row #{args[1]}") unless ROW_DATA.any? { |row, _| row == args[1] }
407 return @table.select { |item| item[:row] == args[1] }
408 .map { |item| spelling(system, item[:id]) }.join(" ")
409 end
410 pattern = name == "FIND" ? parse_pattern(args[1], system) : []
411 allow_slash = name == "HAIKU" || name == "SAME"
412 text.each_char.with_index do |char, i|
413 unless char.match?(/[A-Za-z0-9' .,! ?;:()"-]/) || (allow_slash && char == "/")
414 at(i + 1, "invalid character")
415 end
416 end
417 pieces = text.split("/", -1)
418 if name == "SAME" && (pieces.length != 2 || pieces.any? { |piece| scan_words(piece).empty? })
419 fail_with("expected two texts")
420 end
421 if name == "HAIKU" && (![3, 5].include?(pieces.length) || pieces.any? { |piece| scan_words(piece).empty? })
422 fail_with("expected 3 or 5 lines")
423 end
424 words = scan_words(text)
425 fail_with("no words") if words.empty?
426 words.each do |word|
427 at(word[:start] + 1, "mixed case") unless word[:pattern]
428 end
429 if name == "DETECT"
430 return SYSTEMS.map do |source|
431 verdict = "canonical"
432 begin
433 words.each { |word| word[:morae] = parse_word(word, source) }
434 if words.any? { |word| word[:original] != write_word(word, source, canonical: true) }
435 verdict = "readable"
436 end
437 rescue Invalid
438 verdict = "error"
439 end
440 "#{source}=#{verdict}"
441 end.join(" ")
442 end
443 slash = text.index("/")
444 words.each do |word|
445 source = name == "SAME" && word[:start] > slash ? args[1] : system
446 word[:morae] = parse_word(word, source)
447 end
448 case name
449 when "CONVERT", "NORMALIZE", "REVERSE"
450 target = if name == "NORMALIZE"
451 "HEPBURN"
452 elsif name == "CONVERT"
453 args[1]
454 else
455 system
456 end
457 replace_words(text, words) do |word|
458 morae = word[:morae]
459 if name == "REVERSE"
460 morae = morae.each_with_index.map do |mora, i|
461 id = mora[:id] == "-" ? word[:morae][i - 1][:id][-1] : mora[:id]
462 { id: id, column: mora[:column] }
463 end.reverse
464 end
465 write_word(word, target, morae: morae)
466 end
467 when "VALID"
468 words.each do |word|
469 canonical = write_word(word, system, canonical: true)
470 next if canonical == word[:original]
471
472 i = 0
473 while i < [canonical.length, word[:original].length].min && canonical[i] == word[:original][i]
474 i += 1
475 end
476 return "INVALID col #{word[:start] + i + 1}: expected #{canonical}"
477 end
478 "VALID"
479 when "DIFF"
480 differences = []
481 words.each do |word|
482 a = write_word(word, system)
483 b = write_word(word, args[1])
484 differences << "#{a}>#{b}" if a != b
485 end
486 counted(differences, words.length)
487 when "SAME"
488 left, right = words.partition { |word| word[:start] < slash }
489 [left.length, right.length].min.times do |i|
490 loose = [left[i], right[i]].map do |word|
491 _, long_elements = long_data(word[:morae])
492 kept = word[:morae].each_with_index.filter_map do |mora, index|
493 mora unless long_elements.key?(index)
494 end
495 ids(kept)
496 end
497 return "DIFFERENT at word #{i + 1}" if loose[0] != loose[1]
498 end
499 if left.length != right.length
500 return "DIFFERENT at word #{[left.length, right.length].min + 1}"
501 end
502 left.each_with_index.all? { |word, i| ids(word[:morae]) == ids(right[i][:morae]) } ? "EXACT" : "LOOSE"
503 when "MORAE"
504 total = words.sum { |word| word[:morae].length }
505 displays = words.map do |word|
506 word[:morae].map { |mora| @info.key?(mora[:id]) ? spelling(system, mora[:id]) : mora[:id] }.join(".")
507 end
508 "#{total}: " + displays.join(" ")
509 when "SYLLABLES"
510 total = 0
511 displays = words.map do |word|
512 pairs, = long_data(word[:morae])
513 parts = written_parts(word[:morae], system, "KEEP")
514 groups = []
515 word[:morae].each_with_index do |mora, i|
516 if i.zero? || (@info.key?(mora[:id]) && !pairs.key?(i))
517 groups << parts[i]
518 else
519 groups[-1] += parts[i]
520 end
521 end
522 total += groups.length
523 groups.join(".")
524 end
525 "#{total}: " + displays.join(" ")
526 when "ROW"
527 words.flat_map do |word|
528 word[:morae].map do |mora|
529 item = @info[mora[:id]]
530 if item
531 "#{spelling(system, mora[:id])}: #{item[:row]} #{item[:dan]}"
532 else
533 "#{mora[:id]}: special"
534 end
535 end
536 end.join(", ")
537 when "STATS"
538 counts = Hash.new(0)
539 all = words.flat_map { |word| word[:morae] }
540 all.each do |mora|
541 item = @info[mora[:id]]
542 if item
543 counts[devoice(item[:row])] += 1
544 counts["voiced"] += 1 if %w[ga za da ba].include?(item[:row])
545 counts["semivoiced"] += 1 if item[:row] == "pa"
546 counts["contracted"] += 1 if item[:contracted]
547 else
548 counts[mora[:id] == "-" ? "L" : mora[:id]] += 1
549 end
550 end
551 output = ["morae=#{all.length}"]
552 BASE_ROWS.each do |row|
553 output << "#{row}=#{counts[row]}" if counts[row].positive?
554 end
555 %w[voiced semivoiced contracted N Q L].each do |key|
556 output << "#{key}=#{counts[key]}"
557 end
558 output.join(" ")
559 when "SORT", "INDEX"
560 sorted = sorted_words(words)
561 return sorted.map { |word| word[:original] }.join(" ") if name == "SORT"
562
563 groups = []
564 sorted.each do |word|
565 base = collation_elements(word[:morae])[0][:base]
566 heading = @settings["ORDER"] == "IROHA" || base == "N" ? base : @info[base][:row]
567 if groups.empty? || groups[-1][:heading] != heading
568 groups << { heading: heading, words: [] }
569 end
570 groups[-1][:words] << word[:original]
571 end
572 groups.map { |group| group[:heading] + ": " + group[:words].join(" ") }.join("; ")
573 when "FIND"
574 matches = words.select { |word| matches_pattern?(pattern, word[:morae]) }
575 counted(matches.map { |word| word[:original] }, words.length)
576 when "HAIKU"
577 counts = Array.new(pieces.length, 0)
578 words.each do |word|
579 line_index = text[0...word[:start]].count("/")
580 counts[line_index] += word[:morae].length
581 end
582 expected = pieces.length == 3 ? [5, 7, 5] : [5, 7, 5, 7, 7]
583 verdict = counts == expected ? "OK" : "NG"
584 "#{verdict} #{pieces.length == 3 ? 'haiku' : 'tanka'} " + counts.join("-")
585 when "KAIBUN"
586 sequence = words.flat_map { |word| collation_elements(word[:morae]).map { |item| item[:base] } }
587 (sequence == sequence.reverse ? "YES " : "NO ") + sequence.join(".")
588 when "SHIRITORI"
589 seen = {}
590 previous_tail = nil
591 words.each_with_index do |word, i|
592 sequence = collation_elements(word[:morae])
593 head = sequence[0][:base]
594 tail = sequence[-1][:base]
595 prefix = "NG #{i + 1}: #{word[:original]}"
596 if i.positive? && head != previous_tail
597 return prefix + " must start with " + spelling(system, previous_tail)
598 end
599 key = ids(word[:morae])
600 return prefix + " is repeated" if seen.key?(key)
601 return prefix + " ends with N" if tail == "N"
602
603 seen[key] = true
604 previous_tail = tail
605 end
606 "OK #{words.length}"
607 end
608 end
609end

614 line = STDIN.readline.chomp
