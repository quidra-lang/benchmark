49function spelling(system: string, id: string): string {
50 const hepburn: Record<string, string> = {
51 si: "shi", ti: "chi", tu: "tsu", hu: "fu", zi: "ji", di: "ji",
52 du: "zu", wi: "i", we: "e", wo: "o"
53 };
54 const kunrei: Record<string, string> = {
55 di: "zi", du: "zu", wi: "i", we: "e", wo: "o",
56 dya: "zya", dyu: "zyu", dyo: "zyo"
57 };
58 const wapuro: Record<string, string> = {
59 si: "shi", ti: "chi", tu: "tsu", hu: "fu", zi: "ji"
60 };
61 if (system === "KUNREI") {
62 return kunrei[id] ?? id;
63 }
64 if (system === "NIHON") {
65 return id;
66 }
67 const replacements = system === "WAPURO" ? wapuro : hepburn;
68 if (replacements[id] !== undefined) {
69 return replacements[id];
70 }
71 if (id.startsWith("sy")) {
72 return "sh" + id.slice(2);
73 }
74 if (id.startsWith("ty")) {
75 return "ch" + id.slice(2);
76 }
77 if (id.startsWith("zy") || (system !== "WAPURO" && id.startsWith("dy"))) {
78 return "j" + id.slice(2);
79 }
80 return id;
81}

104function casePattern(word: string): string {
105 if (!/[A-Z]/.test(word)) {
106 return "LOWER";
107 }
108 if (/^[A-Z]/.test(word) && !/[A-Z]/.test(word.slice(1))) {
109 return "TITLE";
110 }
111 const letters = word.replace(/[^A-Za-z]/g, "");
112 if (letters.length >= 2 && !/[a-z]/.test(letters)) {
113 return "UPPER";
114 }
115 return "";
116}

125function parseWord(word: Word, system: string): Mora[] {
126 const text = word.original.toLowerCase();
127 const result: Mora[] = [];
128 let position = 0;
129 while (position < text.length) {
130 const c = text[position];
131 const c2 = text[position + 1] ?? "";
132 const column = word.start + position + 1;
133 let id = "";
134 let length = 1;
135 if (c === "-") {
136 if (result.length === 0 || !info.has(result[result.length - 1].id)) {
137 at(column, "misplaced long mark");
138 }
139 id = "-";
140 } else if (c === "'") {
141 at(column, "misplaced apostrophe");
142 } else {
143 const match = matchAt(text, position, system);
144 if (match) {
145 id = match.id;
146 length = match.spelling.length;
147 } else if ((system === "WAPURO" || system === "NORMALIZE") && text.startsWith("xtu", position)) {
148 id = "Q";
149 length = 3;
150 } else if (c === "n") {
151 id = "N";
152 if (system === "NORMALIZE") {
153 const third = text[position + 2] ?? "";
154 if (c2 === "'" || (c2 === "n" && !/^[aiueoy']$/.test(third))) {
155 length = 2;
156 }
157 } else if ((system === "WAPURO" && c2 === "n") || (system !== "WAPURO" && c2 === "'")) {
158 length = 2;
159 }
160 } else if ((system === "TRADITIONAL" || system === "NORMALIZE") && c === "m" && /^[bmp]$/.test(c2)) {
161 id = "N";
162 } else if (/^[a-z]$/.test(c) && !/[aiueon]/.test(c) && c2 === c && matchAt(text, position + 1, system)) {
163 id = "Q";
164 } else if (["HEPBURN", "TRADITIONAL", "NORMALIZE"].includes(system) && c === "t" &&
165 text.startsWith("ch", position + 1) && matchAt(text, position + 1, system)) {
166 id = "Q";
167 } else {
168 at(column, "unknown syllable");
169 }
170 }
171 result.push({ id, column });
172 position += length;
173 }
174 return result;
175}

177function longData(morae: Mora[]): LongData {
178 const pairs = new Set<number>();
179 const elements = new Map<number, string>();
180 for (let i = 0; i < morae.length; i++) {
181 if (morae[i].id === "-") {
182 elements.set(i, morae[i - 1].id.slice(-1));
183 }
184 }
185 let i = 0;
186 while (i + 1 < morae.length) {
187 const current = morae[i].id;
188 const next = morae[i + 1].id;
189 const vowel = current.slice(-1);
190 if (info.has(current) && /^[aiueo]$/.test(next) &&
191 (vowel === next || (vowel === "o" && next === "u") || (vowel === "e" && next === "i"))) {
192 pairs.add(i + 1);
193 if ((vowel === "o" && (next === "o" || next === "u")) || (vowel === "u" && next === "u")) {
194 elements.set(i + 1, vowel);
195 }
196 i += 2;
197 } else {
198 i++;
199 }
200 }
201 return { pairs, elements };
202}

204function writtenParts(morae: Mora[], system: string, longMode: string): string[] {
205 const data = longData(morae);
206 const parts: string[] = [];
207 for (let i = 0; i < morae.length; i++) {
208 const mora = morae[i];
209 const next = morae[i + 1]?.id ?? "";
210 const nextSpelling = info.has(next) ? spelling(system, next) : "";
211 const first = nextSpelling[0] ?? "";
212 let part = "";
213 if (data.elements.has(i)) {
214 const vowel = data.elements.get(i)!;
215 if (longMode === "KEEP") {
216 part = mora.id === "-" ? "-" : spelling(system, mora.id);
217 } else if (longMode === "PASSPORT") {
218 part = vowel === "o" && !/^[aiueo]$/.test(first) ? "h" : "";
219 } else if (longMode === "HYPHEN") {
220 part = "-";
221 } else if (longMode === "DOUBLE") {
222 part = vowel;
223 }
224 } else if (info.has(mora.id)) {
225 part = spelling(system, mora.id);
226 } else if (mora.id === "N") {
227 if (system === "WAPURO") {
228 part = next === "N" || /^[aiueoyn]$/.test(first) ? "nn" : "n";
229 } else if (system === "TRADITIONAL" && /^[bmp]$/.test(first)) {
230 part = "m";
231 } else {
232 part = /^[aiueoy]$/.test(first) ? "n'" : "n";
233 }
234 } else if (mora.id === "Q") {
235 if (first && !/[aiueon]/.test(first)) {
236 part = ["HEPBURN", "TRADITIONAL"].includes(system) && nextSpelling.startsWith("ch") ? "t" : first;
237 } else if (system === "WAPURO") {
238 part = "xtu";
239 } else {
240 at(mora.column, "small tsu cannot be written");
241 }
242 }
243 parts.push(part);
244 }
245 return parts;
246}

248function writeWord(word: Word, system: string, settings: Settings, canonical: boolean = false, morae: Mora[] = word.morae): string {
249 const value = writtenParts(morae, system, canonical ? "KEEP" : settings.LONG).join("");
250 const pattern = canonical || settings.CASE === "KEEP" ? word.pattern : settings.CASE;
251 if (pattern === "UPPER") {
252 return value.toUpperCase();
253 }
254 if (pattern === "TITLE") {
255 return value[0].toUpperCase() + value.slice(1);
256 }
257 return value;
258}

270function elements(morae: Mora[]): Element[] {
271 const result: Element[] = [];
272 for (const mora of morae) {
273 if (mora.id === "-") {
274 continue;
275 }
276 if (mora.id === "N" || mora.id === "Q") {
277 result.push({ base: mora.id === "N" ? "N" : "tu", voice: 0, size: mora.id === "Q" ? 1 : 0 });
278 continue;
279 }
280 const item = info.get(mora.id)!;
281 const voice = ["ga", "za", "da", "ba"].includes(item.row) ? 1 : item.row === "pa" ? 2 : 0;
282 const base = item.contracted ? devoice(item.row[0] + "i") : devoice(mora.id);
283 result.push({ base, voice, size: 0 });
284 if (item.contracted) {
285 result.push({ base: item.dan, voice: 0, size: 1 });
286 }
287 }
288 return result;
289}

296function compareSequence(a: number[], b: number[]): number {
297 for (let i = 0; i < Math.min(a.length, b.length); i++) {
298 if (a[i] !== b[i]) {
299 return a[i] - b[i];
300 }
301 }
302 return a.length - b.length;
303}

305function sortedWords(words: Word[], order: string): Word[] {
306 const ranks = new Map<string, number>(orders[order].map((base, rank): [string, number] => [base, rank]));
307 const entries = words.map(word => {
308 const cols = elements(word.morae);
309 const keys = [
310 cols.map(item => ranks.get(item.base)!),
311 cols.map(item => item.voice),
312 cols.map(item => item.size),
313 word.morae.map(mora => mora.id === "-" ? 1 : 0)
314 ];
315 return { word, keys };
316 });
317 entries.sort((a, b) => {
318 for (let i = 0; i < a.keys.length; i++) {
319 const difference = compareSequence(a.keys[i], b.keys[i]);
320 if (difference !== 0) {
321 return difference;
322 }
323 }
324 return 0;
325 });
326 return entries.map(entry => entry.word);
327}

329function parsePattern(pattern: string, system: string): string[] {
330 if (!/^[a-z'?*-]+$/.test(pattern)) {
331 fail("invalid pattern");
332 }
333 const result: string[] = [];
334 try {
335 for (const part of pattern.split(/([?*])/)) {
336 if (part === "?" || part === "*") {
337 result.push(part);
338 } else if (part) {
339 const word = { original: part, start: 0, pattern: "LOWER", morae: [] };
340 result.push(...parseWord(word, system).map(mora => mora.id));
341 }
342 }
343 } catch (error) {
344 if (!(error instanceof Invalid)) {
345 throw error;
346 }
347 fail("invalid pattern");
348 }
349 return result;
350}

352function matchesPattern(pattern: string[], morae: Mora[]): boolean {
353 let previous = new Array<boolean>(morae.length + 1).fill(false);
354 previous[0] = true;
355 for (const token of pattern) {
356 const current = new Array<boolean>(morae.length + 1).fill(false);
357 if (token === "*") {
358 current[0] = previous[0];
359 }
360 for (let j = 1; j <= morae.length; j++) {
361 current[j] = token === "*" ? previous[j] || current[j - 1] :
362 previous[j - 1] && (token === "?" || token === morae[j - 1].id);
363 }
364 previous = current;
365 }
366 return previous[morae.length];
367}

377const settings: Settings = { LONG: "KEEP", CASE: "KEEP", ORDER: "GOJUON" };
378const argumentCounts: Record<string, number> = {
379 SET: 2, TABLE: 2, CONVERT: 2, NORMALIZE: 0, REVERSE: 1, VALID: 1,
380 DETECT: 0, DIFF: 2, SAME: 2, MORAE: 1, SYLLABLES: 1, ROW: 1,

384function command(line: string): string {
385 const name = line.split(" ", 1)[0];
386 if (!Object.hasOwn(argumentCounts, name)) {
387 fail("unknown command " + name);
388 }
389 const count = argumentCounts[name];
390 const args: string[] = [];
391 let text = "";
392 if (name === "SET" || name === "TABLE") {
393 const parts = line.split(" ");
394 if (parts.length < 3) {
395 fail("missing argument");
396 }
397 if (parts.length > 3) {
398 fail("too many arguments");
399 }
400 args.push(...parts.slice(1));
401 } else {
402 let position = 0;
403 for (let i = 0; i <= count; i++) {
404 const space = line.indexOf(" ", position);
405 if (space < 0) {
406 fail("missing argument");
407 }
408 if (i > 0) {
409 args.push(line.slice(position, space));
410 }
411 position = space + 1;
412 }
413 text = line.slice(position);
414 }
415 const systemCount = ["NORMALIZE", "DETECT", "SET"].includes(name) ? 0 :
416 ["CONVERT", "DIFF", "SAME"].includes(name) ? 2 : 1;
417 for (const system of args.slice(0, systemCount)) {
418 if (!systems.includes(system)) {
419 fail("unknown system " + system);
420 }
421 }
422 const system = name === "NORMALIZE" ? "NORMALIZE" : args[0];
423 if (name === "SET") {
424 const choices: Record<string, string[]> = {
425 LONG: ["KEEP", "DROP", "PASSPORT", "HYPHEN", "DOUBLE"],
426 CASE: ["KEEP", "LOWER", "UPPER", "TITLE"], ORDER: ["GOJUON", "IROHA"]
427 };
428 if (!Object.hasOwn(choices, args[0])) {
429 fail("unknown option " + args[0]);
430 }
431 if (!choices[args[0]].includes(args[1])) {
432 fail("unknown value " + args[1]);
433 }
434 settings[args[0] as keyof Settings] = args[1];
435 return args[0] + "=" + args[1];
436 }
437 if (name === "TABLE") {
438 if (!rowData.some(([row]) => row === args[1])) {
439 fail("unknown row " + args[1]);
440 }
441 return table.filter(item => item.row === args[1]).map(item => spelling(system, item.id)).join(" ");
442 }
443 const pattern = name === "FIND" ? parsePattern(args[1], system) : [];
444 const allowSlash = name === "HAIKU" || name === "SAME";
445 for (let i = 0; i < text.length; i++) {
446 if (!/[A-Za-z0-9' .,! ?;:()"-]/.test(text[i]) && !(allowSlash && text[i] === "/")) {
447 at(i + 1, "invalid character");
448 }
449 }
450 const pieces = text.split("/");
451 if (name === "SAME" && (pieces.length !== 2 || pieces.some(piece => scanWords(piece).length === 0))) {
452 fail("expected two texts");
453 }
454 if (name === "HAIKU" && (![3, 5].includes(pieces.length) || pieces.some(piece => scanWords(piece).length === 0))) {
455 fail("expected 3 or 5 lines");
456 }
457 const words = scanWords(text);
458 if (words.length === 0) {
459 fail("no words");
460 }
461 for (const word of words) {
462 if (!word.pattern) {
463 at(word.start + 1, "mixed case");
464 }
465 }
466 if (name === "DETECT") {
467 return systems.map(source => {
468 let verdict = "canonical";
469 try {
470 for (const word of words) {
471 word.morae = parseWord(word, source);
472 }
473 if (words.some(word => word.original !== writeWord(word, source, settings, true))) {
474 verdict = "readable";
475 }
476 } catch (error) {
477 if (!(error instanceof Invalid)) {
478 throw error;
479 }
480 verdict = "error";
481 }
482 return source + "=" + verdict;
483 }).join(" ");
484 }
485 const slash = text.indexOf("/");
486 for (const word of words) {
487 const source = name === "SAME" && word.start > slash ? args[1] : system;
488 word.morae = parseWord(word, source);
489 }
490 if (name === "CONVERT" || name === "NORMALIZE" || name === "REVERSE") {
491 const target = name === "NORMALIZE" ? "HEPBURN" : name === "CONVERT" ? args[1] : system;
492 return replaceWords(text, words, word => {
493 let morae = word.morae;
494 if (name === "REVERSE") {
495 morae = morae.map((mora, i) => ({
496 id: mora.id === "-" ? word.morae[i - 1].id.slice(-1) : mora.id,
497 column: mora.column
498 })).reverse();
499 }
500 return writeWord(word, target, settings, false, morae);
501 });
502 }
503 if (name === "VALID") {
504 for (const word of words) {
505 const canonical = writeWord(word, system, settings, true);
506 if (canonical !== word.original) {
507 let i = 0;
508 while (i < Math.min(canonical.length, word.original.length) && canonical[i] === word.original[i]) {
509 i++;
510 }
511 return "INVALID col " + (word.start + i + 1) + ": expected " + canonical;
512 }
513 }
514 return "VALID";
515 }
516 if (name === "DIFF") {
517 const differences: string[] = [];
518 for (const word of words) {
519 const a = writeWord(word, system, settings);
520 const b = writeWord(word, args[1], settings);
521 if (a !== b) {
522 differences.push(a + ">" + b);
523 }
524 }
525 return counted(differences, words.length);
526 }
527 if (name === "SAME") {
528 const left = words.filter(word => word.start < slash);
529 const right = words.filter(word => word.start > slash);
530 for (let i = 0; i < Math.min(left.length, right.length); i++) {
531 const loose = (word: Word): string => {
532 const data = longData(word.morae);
533 return ids(word.morae.filter((_, index) => !data.elements.has(index)));
534 };
535 if (loose(left[i]) !== loose(right[i])) {
536 return "DIFFERENT at word " + (i + 1);
537 }
538 }
539 if (left.length !== right.length) {
540 return "DIFFERENT at word " + (Math.min(left.length, right.length) + 1);
541 }
542 return left.every((word, i) => ids(word.morae) === ids(right[i].morae)) ? "EXACT" : "LOOSE";
543 }
544 if (name === "MORAE") {
545 const total = words.reduce((sum, word) => sum + word.morae.length, 0);
546 return total + ": " + words.map(word => word.morae.map(mora =>
547 info.has(mora.id) ? spelling(system, mora.id) : mora.id).join(".")).join(" ");
548 }
549 if (name === "SYLLABLES") {
550 let total = 0;
551 const displays = words.map(word => {
552 const data = longData(word.morae);
553 const parts = writtenParts(word.morae, system, "KEEP");
554 const groups: string[] = [];
555 for (let i = 0; i < word.morae.length; i++) {
556 if (i === 0 || (info.has(word.morae[i].id) && !data.pairs.has(i))) {
557 groups.push(parts[i]);
558 } else {
559 groups[groups.length - 1] += parts[i];
560 }
561 }
562 total += groups.length;
563 return groups.join(".");
564 });
565 return total + ": " + displays.join(" ");
566 }
567 if (name === "ROW") {
568 return words.flatMap(word => word.morae.map(mora => {
569 const item = info.get(mora.id);
570 return item ? spelling(system, mora.id) + ": " + item.row + " " + item.dan : mora.id + ": special";
571 })).join(", ");
572 }
573 if (name === "STATS") {
574 const counts = new Map<string, number>();
575 const increment = (key: string): void => {
576 counts.set(key, (counts.get(key) ?? 0) + 1);
577 };
578 const all = words.flatMap(word => word.morae);
579 for (const mora of all) {
580 const item = info.get(mora.id);
581 if (!item) {
582 increment(mora.id === "-" ? "L" : mora.id);
583 } else {
584 increment(devoice(item.row));
585 if (["ga", "za", "da", "ba"].includes(item.row)) {
586 increment("voiced");
587 }
588 if (item.row === "pa") {
589 increment("semivoiced");
590 }
591 if (item.contracted) {
592 increment("contracted");
593 }
594 }
595 }
596 const output = ["morae=" + all.length];
597 for (const row of baseRows) {
598 if (counts.has(row)) {
599 output.push(row + "=" + counts.get(row));
600 }
601 }
602 for (const key of ["voiced", "semivoiced", "contracted", "N", "Q", "L"]) {
603 output.push(key + "=" + (counts.get(key) ?? 0));
604 }
605 return output.join(" ");
606 }
607 if (name === "SORT" || name === "INDEX") {
608 const sorted = sortedWords(words, settings.ORDER);
609 if (name === "SORT") {
610 return sorted.map(word => word.original).join(" ");
611 }
612 const groups: { heading: string; words: string[] }[] = [];
613 for (const word of sorted) {
614 const base = elements(word.morae)[0].base;
