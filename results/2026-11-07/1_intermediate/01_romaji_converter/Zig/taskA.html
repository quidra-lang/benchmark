48fn choice(value: []const u8, values: []const []const u8) ?usize {
49 for (values, 0..) |candidate, i| {

52 }
53 return null;

82fn spelling(gpa: Allocator, id: []const u8, system: usize) Allocator.Error![]const u8 {
83 if (system == 3)
84 return id;
85 if (system == 2) {
86 const from = [_][]const u8{ "di", "du", "wi", "we", "wo", "dya", "dyu", "dyo" };
87 const to = [_][]const u8{ "zi", "zu", "i", "e", "o", "zya", "zyu", "zyo" };
88 if (choice(id, &from)) |i|
89 return to[i];
90 return id;
91 }
92 const common = [_][]const u8{ "si", "ti", "tu", "hu", "zi" };
93 const replacements = [_][]const u8{ "shi", "chi", "tsu", "fu", "ji" };
94 if (choice(id, &common)) |i|
95 return replacements[i];
96 const hepburn = system == 0 or system == 1;
97 if (hepburn) {
98 const from = [_][]const u8{ "di", "du", "wi", "we", "wo" };
99 const to = [_][]const u8{ "ji", "zu", "i", "e", "o" };
100 if (choice(id, &from)) |i|
101 return to[i];
102 }
103 if (id.len == 3) {
104 const prefix = id[0..2];
105 if (std.mem.eql(u8, prefix, "sy"))
106 return std.fmt.allocPrint(gpa, "sh{s}", .{id[2..]});
107 if (std.mem.eql(u8, prefix, "ty"))
108 return std.fmt.allocPrint(gpa, "ch{s}", .{id[2..]});
109 if (std.mem.eql(u8, prefix, "zy") or (hepburn and std.mem.eql(u8, prefix, "dy")))
110 return std.fmt.allocPrint(gpa, "j{s}", .{id[2..]});
111 }
112 return id;
113}

115fn tokenizeWords(gpa: Allocator, text: []const u8) Allocator.Error![]Word {
116 var result: std.ArrayList(Word) = .empty;
117 var i: usize = 0;
118 while (i < text.len) {
119 if (!wordCharacter(text[i])) {
120 i += 1;
121 continue;
122 }
123 const start = i;
124 while (i < text.len and wordCharacter(text[i]))
125 i += 1;
126 try result.append(gpa, .{ .original = text[start..i], .start = start, .end = i, .ordinal = result.items.len });
127 }
128 return result.items;
129}

141 fn init(gpa: Allocator) Allocator.Error!Engine {
142 var self = Engine{};
143 var len: usize = 0;
144 for (row_text, 0..) |text, row| {
145 var ids = std.mem.splitScalar(u8, text, ' ');
146 while (ids.next()) |id| {
147 self.table[len] = .{ .id = id, .row = row, .dan = id[id.len - 1 ..] };
148 len += 1;
149 }
150 }
151 for ([_]usize{ 1, 2, 3, 4, 5, 6, 8, 10, 11, 12, 13, 14 }) |row| {
152 var stem: []const u8 = undefined;
153 for (self.table[0..72]) |m| {
154 if (m.row == row and std.mem.eql(u8, m.dan, "i")) {
155 stem = m.id;
156 break;
157 }
158 }
159 for ([_][]const u8{ "ya", "yu", "yo" }) |dan| {
160 const id = try std.fmt.allocPrint(gpa, "{s}{s}", .{ stem[0 .. stem.len - 1], dan });
161 self.table[len] = .{ .id = id, .row = row, .dan = dan };
162 len += 1;
163 }
164 }
165 for (0..5) |s| {
166 for (self.table, 0..) |m, i|
167 self.spellings[s][i] = try spelling(gpa, m.id, s);
168 }
169 var ranks = std.mem.splitScalar(u8, iroha_text, ' ');
170 var rank: usize = 0;
171 while (ranks.next()) |id| : (rank += 1)
172 self.iroha_rank[self.baseIndex(id)] = rank;
173 return self;
174 }

213 fn checkCase(self: *Engine, word: *Word) error{BadCommand}!void {
214 var letters: usize = 0;
215 var uppers: usize = 0;
216 for (word.original) |c| {
217 if (std.ascii.isAlphabetic(c)) {
218 letters += 1;
219 if (std.ascii.isUpper(c))
220 uppers += 1;
221 }
222 }
223 if (uppers == 0) {
224 word.pattern = 0;
225 return;
226 }
227 if (std.ascii.isUpper(word.original[0]) and uppers == 1) {
228 word.pattern = 1;
229 return;
230 }
231 if (letters >= 2 and uppers == letters) {
232 word.pattern = 2;
233 return;
234 }
235 return self.columnError(word.start + 1, "mixed case");
236 }

238 fn matchAt(self: *const Engine, text: []const u8, position: usize, system: usize) ?Match {
239 var best: ?Match = null;
240 const choices: []const usize = if (system == 5) &.{ 0, 2, 3, 4 } else &.{system};
241 for (0..108) |id| {
242 for (choices) |s| {
243 const candidate = self.spellings[s][id];
244 if (std.mem.startsWith(u8, text[position..], candidate) and (best == null or candidate.len > best.?.len))
245 best = .{ .id = id, .len = candidate.len };
246 }
247 }
248 return best;
249 }

251 fn parse(self: *Engine, gpa: Allocator, original: []const u8, start: usize, system: usize) CommandError![]Mora {
252 const text = try std.ascii.allocLowerString(gpa, original);
253 var result: std.ArrayList(Mora) = .empty;
254 var p: usize = 0;
255 while (p < text.len) {
256 const c = text[p];
257 const next: u8 = if (p + 1 < text.len) text[p + 1] else 0;
258 const column = start + p + 1;
259 if (c == '-') {
260 if (result.items.len == 0 or result.items[result.items.len - 1].id >= mora_n)
261 return self.columnError(column, "misplaced long mark");
262 try result.append(gpa, .{ .id = mora_l, .column = column });
263 p += 1;
264 continue;
265 }
266 if (c == '\'')
267 return self.columnError(column, "misplaced apostrophe");
268 if (self.matchAt(text, p, system)) |matched| {
269 try result.append(gpa, .{ .id = matched.id, .column = column });
270 p += matched.len;
271 continue;
272 }
273 if ((system == 4 or system == 5) and std.mem.startsWith(u8, text[p..], "xtu")) {
274 try result.append(gpa, .{ .id = mora_q, .column = column });
275 p += 3;
276 continue;
277 }
278 if (c == 'n') {
279 var advance: usize = 1;
280 if (system == 5) {
281 const third: u8 = if (p + 2 < text.len) text[p + 2] else 0;
282 if (next == '\'' or (next == 'n' and !vowel(third) and third != 'y' and third != '\''))
283 advance = 2;
284 } else if (system == 4) {
285 if (next == 'n')
286 advance = 2;
287 } else if (next == '\'') {
288 advance = 2;
289 }
290 try result.append(gpa, .{ .id = mora_n, .column = column });
291 p += advance;
292 continue;
293 }
294 if ((system == 1 or system == 5) and c == 'm' and next != 0 and std.mem.findScalar(u8, "bmp", next) != null) {
295 try result.append(gpa, .{ .id = mora_n, .column = column });
296 p += 1;
297 continue;
298 }
299 if (std.ascii.isAlphabetic(c) and std.mem.findScalar(u8, "aiueon", c) == null and next == c and self.matchAt(text, p + 1, system) != null) {
300 try result.append(gpa, .{ .id = mora_q, .column = column });
301 p += 1;
302 continue;
303 }
304 if ((system == 0 or system == 1 or system == 5) and c == 't' and std.mem.startsWith(u8, text[p + 1 ..], "ch") and self.matchAt(text, p + 1, system) != null) {
305 try result.append(gpa, .{ .id = mora_q, .column = column });
306 p += 1;
307 continue;
308 }
309 return self.columnError(column, "unknown syllable");
310 }
311 return result.items;
312 }

314 fn pairs(self: *const Engine, gpa: Allocator, morae: []const Mora) Allocator.Error![]bool {
315 const result = try gpa.alloc(bool, morae.len);
316 @memset(result, false);
317 var i: usize = 0;
318 while (i + 1 < morae.len) {
319 const a = morae[i].id;
320 const b = morae[i + 1].id;
321 if (a < mora_n and b < 5) {
322 const v = self.moraVowel(a);
323 const following = self.table[b].id[0];
324 if (v == following or (v == 'o' and following == 'u') or (v == 'e' and following == 'i')) {
325 result[i + 1] = true;
326 i += 2;
327 continue;
328 }
329 }
330 i += 1;
331 }
332 return result;
333 }

335 fn longElements(self: *const Engine, gpa: Allocator, morae: []const Mora) Allocator.Error![]bool {
336 const paired = try self.pairs(gpa, morae);
337 const result = try gpa.alloc(bool, morae.len);
338 @memset(result, false);
339 for (morae, 0..) |m, i| {
340 if (m.id == mora_l) {
341 result[i] = true;
342 continue;
343 }
344 if (!paired[i])
345 continue;
346 const v = self.moraVowel(morae[i - 1].id);
347 const following = self.table[m.id].id[0];
348 result[i] = (v == 'o' and (following == 'o' or following == 'u')) or (v == 'u' and following == 'u');
349 }
350 return result;
351 }

353 fn parts(self: *Engine, gpa: Allocator, morae: []const Mora, target: usize, long_setting: usize) CommandError![][]const u8 {
354 const long = try self.longElements(gpa, morae);
355 const result = try gpa.alloc([]const u8, morae.len);
356 for (morae, 0..) |m, i| {
357 const next_id: usize = if (i + 1 < morae.len) morae[i + 1].id else 111;
358 const next = if (next_id < mora_n) self.spellings[target][next_id] else "";
359 const first: u8 = if (next.len > 0) next[0] else 0;
360 if (long[i]) {
361 const v = self.moraVowel(morae[i - 1].id);
362 result[i] = switch (long_setting) {
363 0 => if (m.id == mora_l) "-" else self.spellings[target][m.id],
364 1 => "",
365 2 => if (v == 'o' and !vowel(first)) "h" else "",
366 3 => "-",
367 else => self.table[morae[i - 1].id].id[self.table[morae[i - 1].id].id.len - 1 ..],
368 };
369 } else if (m.id < mora_n) {
370 result[i] = self.spellings[target][m.id];
371 } else if (m.id == mora_n) {
372 if (target == 4) {
373 result[i] = if (next_id == mora_n or vowel(first) or first == 'y' or first == 'n') "nn" else "n";
374 } else if (target == 1 and first != 0 and std.mem.findScalar(u8, "bmp", first) != null) {
375 result[i] = "m";
376 } else {
377 result[i] = if (vowel(first) or first == 'y') "n'" else "n";
378 }
379 } else {
380 if (first != 0 and std.mem.findScalar(u8, "aiueon", first) == null) {
381 result[i] = if ((target == 0 or target == 1) and std.mem.startsWith(u8, next, "ch")) "t" else next[0..1];
382 } else if (target == 4) {
383 result[i] = "xtu";
384 } else {
385 return self.columnError(m.column, "small tsu cannot be written");
386 }
387 }
388 }
389 return result;
390 }

392 fn writeWord(self: *Engine, gpa: Allocator, word: Word, target: usize, canonical: bool, morae: []const Mora) CommandError![]const u8 {
393 const pieces = try self.parts(gpa, morae, target, if (canonical) 0 else self.long_mode);
394 const result = try std.mem.concat(gpa, u8, pieces);
395 const pattern: usize = if (canonical or self.case_mode == 0) word.pattern else if (self.case_mode == 1) 0 else if (self.case_mode == 3) 1 else 2;
396 if (pattern == 1)
397 result[0] = std.ascii.toUpper(result[0]);
398 if (pattern == 2)
399 _ = std.ascii.upperString(result, result);
400 return result;
401 }

424 fn collate(self: *const Engine, gpa: Allocator, word: *Word) Allocator.Error!void {
425 var result: std.ArrayList(Element) = .empty;
426 for (word.morae) |m| {
427 if (m.id == mora_l)
428 continue;
429 if (m.id == mora_n) {
430 try result.append(gpa, .{ .base = 47, .voice = 0, .size = 0 });
431 continue;
432 }
433 if (m.id == mora_q) {
434 try result.append(gpa, .{ .base = self.baseIndex("tu"), .voice = 0, .size = 1 });
435 continue;
436 }
437 const syllable = self.table[m.id];
438 const voice: usize = if (syllable.row >= 10 and syllable.row <= 13) 1 else if (syllable.row == 14) 2 else 0;
439 if (m.id < 72) {
440 try result.append(gpa, .{ .base = self.devoiced(syllable.id), .voice = voice, .size = 0 });
441 } else {
442 const stem = [_]u8{ syllable.id[0], 'i' };
443 try result.append(gpa, .{ .base = self.devoiced(&stem), .voice = voice, .size = 0 });
444 try result.append(gpa, .{ .base = self.baseIndex(syllable.id[1..]), .voice = 0, .size = 1 });
445 }
446 }
447 word.elements = result.items;
448 }

450 fn prepareSortKey(self: *const Engine, gpa: Allocator, word: *Word) Allocator.Error!void {
451 for (0..3) |criterion| {
452 const key = try gpa.alloc(usize, word.elements.len);
453 for (word.elements, 0..) |e, i| {
454 key[i] = switch (criterion) {
455 0 => if (self.iroha) self.iroha_rank[e.base] else e.base,
456 1 => e.voice,
457 else => e.size,
458 };
459 }
460 word.sort_key[criterion] = key;
461 }
462 const long = try gpa.alloc(usize, word.morae.len);
463 for (word.morae, 0..) |m, i|
464 long[i] = if (m.id == mora_l) 1 else 0;
465 word.sort_key[3] = long;
466 }

468 fn wordLess(_: void, a: Word, b: Word) bool {
469 for (a.sort_key, b.sort_key) |left, right| {
470 const order = std.mem.order(usize, left, right);
471 if (order != .eq)
472 return order == .lt;
473 }
474 return a.ordinal < b.ordinal;
475 }

477 fn parsePattern(self: *Engine, gpa: Allocator, text: []const u8, system: usize) CommandError![]i32 {
478 for (text) |c| {
479 if (!((c >= 'a' and c <= 'z') or c == '\'' or c == '-' or c == '?' or c == '*'))
480 return self.fail("invalid pattern");
481 }
482 var result: std.ArrayList(i32) = .empty;
483 var start: usize = 0;
484 var i: usize = 0;
485 while (i <= text.len) : (i += 1) {
486 if (i == text.len or text[i] == '?' or text[i] == '*') {
487 if (i > start) {
488 const part = self.parse(gpa, text[start..i], start, system) catch |err| {
489 if (err == error.BadCommand)
490 return self.fail("invalid pattern");
491 return err;
492 };
493 for (part) |m|
494 try result.append(gpa, @intCast(m.id));
495 }
496 if (i < text.len)
497 try result.append(gpa, if (text[i] == '?') -1 else -2);
498 start = i + 1;
499 }
500 }
501 return result.items;
502 }

504 fn matches(gpa: Allocator, pattern: []const i32, morae: []const Mora) Allocator.Error!bool {
505 var previous = try gpa.alloc(bool, morae.len + 1);
506 @memset(previous, false);
507 previous[0] = true;
508 for (pattern) |token| {
509 const current = try gpa.alloc(bool, morae.len + 1);
510 @memset(current, false);
511 for (0..morae.len + 1) |j|
512 current[j] = if (token == -2) previous[j] or (j > 0 and current[j - 1]) else j > 0 and previous[j - 1] and (token == -1 or token == @as(i32, @intCast(morae[j - 1].id)));
513 previous = current;
514 }
515 return previous[morae.len];
516 }

518 fn execute(self: *Engine, gpa: Allocator, line: []const u8, out: *std.Io.Writer) CommandError!void {
519 const commands = [_][]const u8{ "SET", "TABLE", "CONVERT", "NORMALIZE", "REVERSE", "VALID", "DETECT", "DIFF", "SAME", "MORAE", "SYLLABLES", "ROW", "STATS", "SORT", "INDEX", "FIND", "HAIKU", "KAIBUN", "SHIRITORI" };
520 const first_space = std.mem.findScalar(u8, line, ' ');
521 const op = if (first_space) |position| line[0..position] else line;
522 if (choice(op, &commands) == null)
523 return self.failFmt("unknown command {s}", .{op});
524 const fixed = std.mem.eql(u8, op, "SET") or std.mem.eql(u8, op, "TABLE");
525 const argument_count: usize = if (choice(op, &.{ "CONVERT", "DIFF", "SAME", "FIND" }) != null) 2 else if (choice(op, &.{ "NORMALIZE", "DETECT" }) != null) 0 else 1;
526 var args: [2][]const u8 = undefined;
527 var text: []const u8 = "";
528 if (fixed) {
529 var tokens = std.mem.splitScalar(u8, line, ' ');
530 var len: usize = 0;
531 while (tokens.next()) |token| {
532 if (len > 0 and len <= 2)
533 args[len - 1] = token;
534 len += 1;
535 }
536 if (len < 3)
537 return self.fail("missing argument");
538 if (len > 3)
539 return self.fail("too many arguments");
540 } else {
541 var cursor = first_space orelse return self.fail("missing argument");
542 for (0..argument_count) |i| {
543 const offset = std.mem.findScalar(u8, line[cursor + 1 ..], ' ') orelse return self.fail("missing argument");
544 const end = cursor + 1 + offset;
545 args[i] = line[cursor + 1 .. end];
546 cursor = end;
547 }
548 text = line[cursor + 1 ..];
549 }
550 var system: usize = 0;
551 var target: usize = 0;
552 const system_count: usize = if (std.mem.eql(u8, op, "TABLE")) 1 else if (fixed or choice(op, &.{ "NORMALIZE", "DETECT" }) != null) 0 else if (choice(op, &.{ "CONVERT", "DIFF", "SAME" }) != null) 2 else 1;
553 for (0..system_count) |i| {
554 const value = choice(args[i], &systems) orelse return self.failFmt("unknown system {s}", .{args[i]});
555 if (i == 0) {
556 system = value;
557 } else {
558 target = value;
559 }
560 }
561 if (std.mem.eql(u8, op, "SET")) {
562 const options = [_][]const u8{ "LONG", "CASE", "ORDER" };
563 const option = choice(args[0], &options) orelse return self.failFmt("unknown option {s}", .{args[0]});
564 const values: []const []const u8 = switch (option) {
565 0 => &.{ "KEEP", "DROP", "PASSPORT", "HYPHEN", "DOUBLE" },
566 1 => &.{ "KEEP", "LOWER", "UPPER", "TITLE" },
567 else => &.{ "GOJUON", "IROHA" },
568 };
569 const value = choice(args[1], values) orelse return self.failFmt("unknown value {s}", .{args[1]});
570 switch (option) {
571 0 => self.long_mode = value,
572 1 => self.case_mode = value,
573 else => self.iroha = value == 1,
574 }
575 try out.print("{s}={s}", .{ args[0], args[1] });
576 return;
577 }
578 if (std.mem.eql(u8, op, "TABLE")) {
579 const row = choice(args[1], &row_names) orelse return self.failFmt("unknown row {s}", .{args[1]});
580 var first = true;
581 for (self.table, 0..) |m, i| {
582 if (m.row == row) {
583 if (!first)
584 try out.writeByte(' ');
585 first = false;
586 try out.writeAll(self.spellings[system][i]);
587 }
588 }
589 return;
590 }
591 const find_pattern: []const i32 = if (std.mem.eql(u8, op, "FIND")) try self.parsePattern(gpa, args[1], system) else &.{};
592 const allow_slash = std.mem.eql(u8, op, "SAME") or std.mem.eql(u8, op, "HAIKU");
593 for (text, 0..) |c, i| {
594 if (!wordCharacter(c) and std.mem.findScalar(u8, " 0123456789.,!?;:()\"", c) == null and !(allow_slash and c == '/'))
595 return self.columnError(i + 1, "invalid character");
596 }
597 const words = try tokenizeWords(gpa, text);
598 var line_count: usize = 0;
599 var empty_line = false;
600 var segments = std.mem.splitScalar(u8, text, '/');
601 while (segments.next()) |segment| {
602 line_count += 1;
603 if (!hasWord(segment))
604 empty_line = true;
605 }
606 if (std.mem.eql(u8, op, "SAME") and (line_count != 2 or empty_line))
607 return self.fail("expected two texts");
608 if (std.mem.eql(u8, op, "HAIKU") and (!(line_count == 3 or line_count == 5) or empty_line))
609 return self.fail("expected 3 or 5 lines");
610 if (words.len == 0)
611 return self.fail("no words");
612 for (words) |*word|
613 try self.checkCase(word);
614 if (std.mem.eql(u8, op, "DETECT")) {
615 for (0..5) |s| {
616 var verdict: []const u8 = "canonical";
617 for (words) |*word| {
618 word.morae = self.parse(gpa, word.original, word.start, s) catch |err| {
619 if (err == error.BadCommand) {
620 verdict = "error";
621 break;
622 }
623 return err;
624 };
625 }
626 if (!std.mem.eql(u8, verdict, "error")) {
627 for (words) |word| {
628 const canonical = try self.writeWord(gpa, word, s, true, word.morae);
629 if (!std.mem.eql(u8, canonical, word.original))
630 verdict = "readable";
631 }
632 }
633 if (s > 0)
634 try out.writeByte(' ');
635 try out.print("{s}={s}", .{ systems[s], verdict });
636 }
637 return;
638 }
639 const slash_position = if (std.mem.eql(u8, op, "SAME")) std.mem.findScalar(u8, text, '/').? else text.len;
640 for (words) |*word|
641 word.morae = try self.parse(gpa, word.original, word.start, if (std.mem.eql(u8, op, "NORMALIZE")) 5 else if (std.mem.eql(u8, op, "SAME") and word.start > slash_position) target else system);
642 if (choice(op, &.{ "CONVERT", "NORMALIZE", "REVERSE" }) != null) {
643 var cursor: usize = 0;
644 for (words) |word| {
645 try out.writeAll(text[cursor..word.start]);
646 var morae: []const Mora = word.morae;
647 if (std.mem.eql(u8, op, "REVERSE")) {
648 const reversed = try gpa.alloc(Mora, word.morae.len);
649 for (word.morae, 0..) |m, i| {
650 var replacement = m;
651 if (m.id == mora_l) {
652 const v = self.moraVowel(word.morae[i - 1].id);
653 replacement.id = std.mem.findScalar(u8, "aiueo", v).?;
654 }
655 reversed[word.morae.len - 1 - i] = replacement;
656 }
657 morae = reversed;
658 }
659 const written = try self.writeWord(gpa, word, if (std.mem.eql(u8, op, "NORMALIZE")) 0 else if (std.mem.eql(u8, op, "REVERSE")) system else target, false, morae);
660 try out.writeAll(written);
661 cursor = word.end;
662 }
663 try out.writeAll(text[cursor..]);
664 return;
665 }
666 if (std.mem.eql(u8, op, "VALID")) {
667 for (words) |word| {
668 const canonical = try self.writeWord(gpa, word, system, true, word.morae);
669 if (std.mem.eql(u8, canonical, word.original))
670 continue;
671 var i: usize = 0;
672 while (i < canonical.len and i < word.original.len and canonical[i] == word.original[i])
673 i += 1;
674 try out.print("INVALID col {d}: expected {s}", .{ word.start + i + 1, canonical });
675 return;
676 }
677 try out.writeAll("VALID");
678 return;
679 }
680 if (std.mem.eql(u8, op, "DIFF")) {
681 const Difference = struct { a: []const u8, b: []const u8 };
682 var differences: std.ArrayList(Difference) = .empty;
683 for (words) |word| {
684 const a = try self.writeWord(gpa, word, system, false, word.morae);
685 const b = try self.writeWord(gpa, word, target, false, word.morae);
686 if (!std.mem.eql(u8, a, b))
687 try differences.append(gpa, .{ .a = a, .b = b });
688 }
689 try out.print("{d} of {d}", .{ differences.items.len, words.len });
690 if (differences.items.len > 0) {
691 try out.writeAll(": ");
692 for (differences.items, 0..) |d, i| {
693 if (i > 0)
694 try out.writeByte(' ');
695 try out.print("{s}>{s}", .{ d.a, d.b });
696 }
697 }
698 return;
699 }
700 if (std.mem.eql(u8, op, "SAME")) {
701 var split: usize = 0;
702 while (split < words.len and words[split].start < slash_position)
703 split += 1;
704 const left = words[0..split];
705 const right = words[split..];
706 var exact = true;
707 for (0..@min(left.len, right.len)) |i| {
708 if (!equalMorae(try self.loose(gpa, left[i]), try self.loose(gpa, right[i]))) {
709 try out.print("DIFFERENT at word {d}", .{i + 1});
710 return;
711 }
712 if (!equalMorae(left[i].morae, right[i].morae))
713 exact = false;
714 }
715 if (left.len != right.len) {
716 try out.print("DIFFERENT at word {d}", .{@min(left.len, right.len) + 1});
717 return;
718 }
719 try out.writeAll(if (exact) "EXACT" else "LOOSE");
720 return;
721 }
722 if (std.mem.eql(u8, op, "MORAE")) {
723 var total: usize = 0;
724 for (words) |word|
725 total += word.morae.len;
726 try out.print("{d}: ", .{total});
727 for (words, 0..) |word, i| {
728 if (i > 0)
729 try out.writeByte(' ');
730 for (word.morae, 0..) |m, j| {
731 if (j > 0)
732 try out.writeByte('.');
733 try out.writeAll(if (m.id < mora_n) self.spellings[system][m.id] else if (m.id == mora_n) "N" else if (m.id == mora_q) "Q" else "-");
734 }
735 }
736 return;
737 }
738 if (std.mem.eql(u8, op, "SYLLABLES")) {
739 var total: usize = 0;
740 const result = try gpa.alloc([]const u8, words.len);
741 for (words, 0..) |word, w| {
742 const paired = try self.pairs(gpa, word.morae);
743 const written = try self.parts(gpa, word.morae, system, 0);
744 var bytes: std.ArrayList(u8) = .empty;
745 for (word.morae, 0..) |m, i| {
746 if (i == 0 or (m.id < mora_n and !paired[i])) {
747 if (i > 0)
748 try bytes.append(gpa, '.');
749 total += 1;
750 }
751 try bytes.appendSlice(gpa, written[i]);
752 }
753 result[w] = bytes.items;
754 }
755 try out.print("{d}: ", .{total});
756 for (result, 0..) |s, i| {
757 if (i > 0)
758 try out.writeByte(' ');
759 try out.writeAll(s);
760 }
761 return;
762 }
763 if (std.mem.eql(u8, op, "ROW")) {
764 var first = true;
765 for (words) |word| {
766 for (word.morae) |m| {
767 if (!first)
768 try out.writeAll(", ");
769 first = false;
770 if (m.id < mora_n) {
771 const syllable = self.table[m.id];
772 try out.print("{s}: {s} {s}", .{ self.spellings[system][m.id], row_names[syllable.row], syllable.dan });
773 } else {
774 const display: []const u8 = if (m.id == mora_n) "N" else if (m.id == mora_q) "Q" else "-";
775 try out.print("{s}: special", .{display});
776 }
777 }
778 }
779 return;
780 }
781 if (std.mem.eql(u8, op, "STATS")) {
782 var total: usize = 0;
783 var voiced: usize = 0;
784 var semi: usize = 0;
785 var contracted: usize = 0;
786 var rows: [10]usize = @splat(0);
787 var special: [3]usize = @splat(0);
788 for (words) |word| {
789 for (word.morae) |m| {
790 total += 1;
791 if (m.id >= mora_n) {
792 special[m.id - mora_n] += 1;
793 continue;
794 }
795 const row = self.table[m.id].row;
796 rows[base_rows[row]] += 1;
797 if (row >= 10 and row <= 13)
798 voiced += 1;
799 if (row == 14)
800 semi += 1;
801 if (m.id >= 72)
802 contracted += 1;
803 }
804 }
805 try out.print("morae={d}", .{total});
806 for (rows, 0..) |n, i| {
807 if (n > 0)
808 try out.print(" {s}={d}", .{ row_names[i], n });
809 }
810 try out.print(" voiced={d} semivoiced={d} contracted={d} N={d} Q={d} L={d}", .{ voiced, semi, contracted, special[0], special[1], special[2] });
811 return;
812 }
813 if (std.mem.eql(u8, op, "FIND")) {
814 var found: std.ArrayList(Word) = .empty;
815 for (words) |word| {
816 if (try matches(gpa, find_pattern, word.morae))
817 try found.append(gpa, word);
818 }
819 try out.print("{d} of {d}", .{ found.items.len, words.len });
820 if (found.items.len > 0) {
821 try out.writeAll(": ");
822 for (found.items, 0..) |word, i| {
823 if (i > 0)
824 try out.writeByte(' ');
825 try out.writeAll(word.original);
826 }
827 }
828 return;
829 }
830 if (std.mem.eql(u8, op, "HAIKU")) {
831 var counts: [5]usize = @splat(0);
832 for (words) |word| {
833 const part = std.mem.countScalar(u8, text[0..word.start], '/');
834 counts[part] += word.morae.len;
835 }
836 const expected = [_]usize{ 5, 7, 5, 7, 7 };
837 const ok = std.mem.eql(usize, counts[0..line_count], expected[0..line_count]);
838 const verdict: []const u8 = if (ok) "OK" else "NG";
839 const poem: []const u8 = if (line_count == 3) "haiku" else "tanka";
840 try out.print("{s} {s} ", .{ verdict, poem });
841 for (counts[0..line_count], 0..) |n, i| {
842 if (i > 0)
843 try out.writeByte('-');
844 try out.print("{d}", .{n});
845 }
846 return;
847 }
848 for (words) |*word|
849 try self.collate(gpa, word);
850 if (std.mem.eql(u8, op, "SORT") or std.mem.eql(u8, op, "INDEX")) {
851 for (words) |*word|
852 try self.prepareSortKey(gpa, word);
853 std.mem.sort(Word, words, {}, wordLess);
854 if (std.mem.eql(u8, op, "SORT")) {
855 for (words, 0..) |word, i| {
856 if (i > 0)
857 try out.writeByte(' ');
858 try out.writeAll(word.original);
859 }
860 return;
861 }
862 var previous: []const u8 = "";
863 for (words, 0..) |word, i| {
864 const base = word.elements[0].base;
865 const heading = if (self.iroha or base == 47) self.baseName(base) else row_names[self.table[base].row];
866 if (!std.mem.eql(u8, previous, heading)) {
867 if (i > 0)
868 try out.writeAll("; ");
869 try out.print("{s}:", .{heading});
870 previous = heading;
871 }
872 try out.print(" {s}", .{word.original});
873 }
874 return;
875 }
876 if (std.mem.eql(u8, op, "KAIBUN")) {
877 var sequence: std.ArrayList(usize) = .empty;
878 for (words) |word| {
879 for (word.elements) |e|
880 try sequence.append(gpa, e.base);
881 }
882 var palindrome = true;
883 for (0..sequence.items.len / 2) |i| {
884 if (sequence.items[i] != sequence.items[sequence.items.len - 1 - i])
885 palindrome = false;
886 }
887 try out.writeAll(if (palindrome) "YES " else "NO ");
888 for (sequence.items, 0..) |base, i| {
889 if (i > 0)
890 try out.writeByte('.');
891 try out.writeAll(self.baseName(base));
892 }
893 return;
894 }
895 for (words, 0..) |word, i| {
896 const head = word.elements[0].base;
897 const tail = word.elements[word.elements.len - 1].base;
898 if (i > 0 and head != words[i - 1].elements[words[i - 1].elements.len - 1].base) {
899 const previous = words[i - 1].elements[words[i - 1].elements.len - 1].base;
900 try out.print("NG {d}: {s} must start with {s}", .{ i + 1, word.original, if (previous == 47) @as([]const u8, "N") else self.spellings[system][previous] });
901 return;
902 }
903 for (words[0..i]) |earlier| {
904 if (equalMorae(earlier.morae, word.morae)) {
905 try out.print("NG {d}: {s} is repeated", .{ i + 1, word.original });
906 return;
907 }
908 }
909 if (tail == 47) {
910 try out.print("NG {d}: {s} ends with N", .{ i + 1, word.original });
911 return;
912 }
913 }
914 try out.print("OK {d}", .{words.len});
915 }
916};

918pub fn main(init: std.process.Init) !void {
919 var table_arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
920 defer table_arena.deinit();
921 var engine = try Engine.init(table_arena.allocator());
922 var input_buffer: [1024]u8 = undefined;
923 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
924 var output_buffer: [8192]u8 = undefined;
925 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
926 const out = &output.interface;
927 const n = try std.fmt.parseInt(usize, (try input.interface.takeDelimiter('\n')).?, 10);
928 for (0..n) |_| {
929 const line = (try input.interface.takeDelimiter('\n')).?;
930 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
931 defer arena.deinit();
932 var draft_buffer: [8192]u8 = undefined;
933 var draft: std.Io.Writer = .fixed(&draft_buffer);
934 engine.execute(arena.allocator(), line, &draft) catch |err| {
935 if (err == error.BadCommand) {
936 try out.print("ERROR {s}\n", .{engine.error_buffer[0..engine.error_len]});
937 continue;
938 }
939 return err;
940 };
941 try out.writeAll(draft.buffered());
942 try out.writeByte('\n');
943 }
944 try out.flush();
945}
