37func letter(c byte) bool { return c >= 'a' && c <= 'z' || c >= 'A' && c <= 'Z' }
38func vowel(c byte) bool { return c != 0 && strings.ContainsRune("aiueo", rune(c)) }
39func wordChar(c byte) bool { return letter(c) || c == '\'' || c == '-' }
40func lower(c byte) byte {
41 if c >= 'A' && c <= 'Z' { return c + 'a' - 'A' }
42 return c
43}
44func upper(c byte) byte {
45 if c >= 'a' && c <= 'z' { return c - 'a' + 'A' }
46 return c
47}
48func moraID(id string) int {
49 for i, m := range table {
50 if m.id == id { return i }
51 }
52 return moraN

103func fail(message string) { panic(message) }
104func columnError(col int, message string) { fail(fmt.Sprintf("col %d: %s", col, message)) }
105func lex(text string) []word {
106 var words []word
107 section := 0
108 for i := 0; i < len(text); {
109 if !wordChar(text[i]) {
110 if text[i] == '/' { section++ }
111 i++
112 continue
113 }
114 start := i
115 for i < len(text) && wordChar(text[i]) { i++ }
116 words = append(words, word{original: text[start:i], start: start, end: i, section: section})
117 }
118 return words
119}
120func checkCase(w *word) {
121 allLower, allUpper := true, true
122 title := w.original[0] >= 'A' && w.original[0] <= 'Z'
123 letters := 0
124 for i := 0; i < len(w.original); i++ {
125 c := w.original[i]
126 if !letter(c) { continue }
127 letters++
128 if c != lower(c) { allLower = false }
129 if c != upper(c) { allUpper = false }
130 if i > 0 && c != lower(c) { title = false }
131 }
132 switch {
133 case allLower: w.pattern = 0
134 case title: w.pattern = 1
135 case letters >= 2 && allUpper: w.pattern = 2
136 default: columnError(w.start+1, "mixed case")
137 }
138}
139func match(s string, pos, system int) (int, int) {
140 id, length := -1, 0
141 for i, m := range table {
142 for t, p := range m.spelling {
143 if system >= 0 && t != system { continue }
144 if len(p) > length && strings.HasPrefix(s[pos:], p) {
145 id, length = i, len(p)
146 }
147 }
148 }
149 return id, length
150}
151func parse(w *word, system int) string {
152 w.tokens = nil
153 s := strings.ToLower(w.original)
154 for p := 0; p < len(s); {
155 c, c2 := s[p], byte(0)
156 if p+1 < len(s) { c2 = s[p+1] }
157 id, advance := -1, 1
158 if c == '-' {
159 if len(w.tokens) == 0 || w.tokens[len(w.tokens)-1].id >= moraN {
160 return fmt.Sprintf("col %d: misplaced long mark", w.start+p+1)
161 }
162 id = moraL
163 } else if c == '\'' {
164 return fmt.Sprintf("col %d: misplaced apostrophe", w.start+p+1)
165 } else {
166 found, length := match(s, p, system)
167 switch {
168 case found >= 0:
169 id, advance = found, length
170 case (system == 4 || system < 0) && strings.HasPrefix(s[p:], "xtu"):
171 id, advance = moraQ, 3
172 case c == 'n':
173 id = moraN
174 if system < 0 {
175 c3 := byte(0)
176 if p+2 < len(s) { c3 = s[p+2] }
177 if c2 == '\'' || c2 == 'n' && !vowel(c3) && c3 != 'y' && c3 != '\'' { advance = 2 }
178 } else if system == 4 && c2 == 'n' || system != 4 && c2 == '\'' { advance = 2 }
179 case (system == 1 || system < 0) && c == 'm' && c2 != 0 && strings.ContainsRune("bmp", rune(c2)):
180 id = moraN
181 default:
182 next, _ := match(s, p+1, system)
183 if letter(c) && !vowel(c) && c != 'n' && c2 == c && next >= 0 {
184 id = moraQ
185 } else if system <= 1 && c == 't' && strings.HasPrefix(s[p+1:], "ch") && next >= 0 {
186 id = moraQ
187 } else { return fmt.Sprintf("col %d: unknown syllable", w.start+p+1) }
188 }
189 }
190 w.tokens = append(w.tokens, token{id, w.start+p+1})
191 p += advance
192 }
193 return ""
194}
195func pairs(w word, longOnly bool) []bool {
196 result := make([]bool, len(w.tokens))
197 for i := 0; i+1 < len(w.tokens); {
198 a, b := w.tokens[i].id, w.tokens[i+1].id
199 if a < moraN && b < 5 {
200 x, y := table[a].id[len(table[a].id)-1], table[b].id[0]
201 if x == y || x == 'o' && y == 'u' || x == 'e' && y == 'i' {
202 result[i+1] = !longOnly || x == 'o' && (y == 'o' || y == 'u') || x == 'u' && y == 'u'
203 i += 2
204 continue
205 }
206 }
207 i++
208 }
209 if longOnly {
210 for i, t := range w.tokens {
211 if t.id == moraL { result[i] = true }
212 }
213 }
214 return result
215}
216func parts(w word, system, mode int) []string {
217 longElements := pairs(w, true)
218 result := make([]string, len(w.tokens))
219 for i, t := range w.tokens {
220 nextID, next := -1, ""
221 if i+1 < len(w.tokens) {
222 nextID = w.tokens[i+1].id
223 if nextID < moraN { next = table[nextID].spelling[system] }
224 }
225 first := byte(0)
226 if next != "" { first = next[0] }
227 s := ""
228 if longElements[i] {
229 previous := table[w.tokens[i-1].id].id
230 v := previous[len(previous)-1]
231 switch mode {
232 case 0:
233 if t.id == moraL { s = "-" } else { s = table[t.id].spelling[system] }
234 case 2:
235 if v == 'o' && !vowel(first) { s = "h" }
236 case 3: s = "-"
237 case 4: s = string(v)
238 }
239 } else if t.id < moraN {
240 s = table[t.id].spelling[system]
241 } else if t.id == moraN {
242 s = "n"
243 if system == 4 {
244 if nextID == moraN || vowel(first) || first == 'y' || first == 'n' { s = "nn" }
245 } else if system == 1 && first != 0 && strings.ContainsRune("bmp", rune(first)) {
246 s = "m"
247 } else if vowel(first) || first == 'y' { s = "n'" }
248 } else if t.id == moraQ {
249 if first != 0 && !vowel(first) && first != 'n' {
250 s = string(first)
251 if system <= 1 && strings.HasPrefix(next, "ch") { s = "t" }
252 } else if system == 4 { s = "xtu" } else { columnError(t.col, "small tsu cannot be written") }
253 }
254 result[i] = s
255 }
256 return result
257}
258func writeWord(w word, system int, canonical bool, config settings) string {
259 mode := config.long
260 pattern := w.pattern
261 if canonical { mode = 0 } else if config.casing != 0 {
262 pattern = []int{0, 0, 2, 1}[config.casing]
263 }
264 result := strings.Join(parts(w, system, mode), "")
265 if pattern == 1 && len(result) > 0 { result = string(upper(result[0])) + result[1:] }
266 if pattern == 2 { result = strings.ToUpper(result) }
267 return result
268}
269func sequence(w word, loose bool) []int {
270 omit := pairs(w, true)
271 var result []int
272 for i, t := range w.tokens {
273 if !loose || !omit[i] { result = append(result, t.id) }
274 }
275 return result
276}
277func elements(w word) []element {
278 var result []element
279 for _, t := range w.tokens {
280 switch t.id {
281 case moraL: continue
282 case moraN: result = append(result, element{moraN, 0, 0})
283 case moraQ: result = append(result, element{moraID("tu"), 0, 1})
284 default:
285 m := table[t.id]
286 base := m.id
287 if m.voice != 0 {
288 r := slices.Index([]string{"ga", "za", "da", "ba", "pa"}, m.row)
289 base = string("ksthh"[r]) + base[1:]
290 }
291 if len(m.dan) == 2 {
292 result = append(result, element{moraID(base[:1]+"i"), m.voice, 0}, element{moraID(m.dan), 0, 1})
293 } else { result = append(result, element{moraID(base), m.voice, 0}) }
294 }
295 }
296 return result
297}

299 ranks := strings.Fields("a i u e o ka ki ku ke ko sa si su se so ta ti tu te to na ni nu ne no ha hi hu he ho ma mi mu me mo ya yu yo ra ri ru re ro wa wi we wo N")
300 if order == 1 { ranks = strings.Fields("i ro ha ni ho he to ti ri nu ru wo wa ka yo ta re so tu ne na ra mu u wi no o ku ya ma ke hu ko e te a sa ki yu me mi si we hi mo se su N") }
301 result := [4][]int{}
302 for _, e := range elements(w) {
303 id := "N"

306 result[1] = append(result[1], e.voice)

309 for _, t := range w.tokens {
310 n := 0

316func less(a, b [4][]int) bool {
317 for k := range a {
318 if comparison := slices.Compare(a[k], b[k]); comparison != 0 {
319 return comparison < 0
320 }
321 }
322 return false
323}
324func patternTokens(pattern string, system int) []int {
325 var result []int
326 start := 0
327 for i := 0; i <= len(pattern); i++ {
328 c := byte(0)
329 if i < len(pattern) { c = pattern[i] }
330 if c == 0 || c == '?' || c == '*' {
331 if i > start {
332 w := word{original: pattern[start:i]}
333 if parse(&w, system) != "" { fail("invalid pattern") }
334 result = append(result, sequence(w, false)...)
335 }
336 if c == '?' { result = append(result, -1) }
337 if c == '*' { result = append(result, -2) }
338 start = i+1
339 } else if !(c >= 'a' && c <= 'z' || c == '\'' || c == '-') { fail("invalid pattern") }
340 }
341 return result
342}
343func glob(pattern []int, w word) bool {
344 dp := make([]bool, len(w.tokens)+1)
345 dp[0] = true
346 for _, p := range pattern {
347 next := make([]bool, len(dp))
348 if p == -2 {
349 next[0] = dp[0]
350 for j := 1; j < len(dp); j++ { next[j] = dp[j] || next[j-1] }
351 } else {
352 for j := 1; j < len(dp); j++ { next[j] = dp[j-1] && (p == -1 || p == w.tokens[j-1].id) }
353 }
354 dp = next
355 }
356 return dp[len(w.tokens)]
357}
358func process(line string, config *settings) string {
359 names := []string{"SET", "TABLE", "CONVERT", "NORMALIZE", "REVERSE", "VALID", "DETECT", "DIFF", "SAME", "MORAE", "SYLLABLES", "ROW", "STATS", "SORT", "INDEX", "FIND", "HAIKU", "KAIBUN", "SHIRITORI"}
360 fixed := []int{2, 2, 2, 0, 1, 1, 0, 2, 2, 1, 1, 1, 1, 1, 1, 2, 1, 1, 1}

363 if cmd < 0 { fail("unknown command " + command) }
364 var args []string
365 text := ""
366 if cmd <= 1 {

381 if cmd == 0 {
382 option := slices.Index([]string{"LONG", "CASE", "ORDER"}, args[0])

385 value := slices.Index(values[option], args[1])
386 if value < 0 { fail("unknown value " + args[1]) }

388 case 0: config.long = value
389 case 1: config.casing = value
390 case 2: config.order = value

394 systemCount := 1

402 source := sys[0]
403 if cmd == 1 {

424 if cmd == 16 {

430 if len(words) == 0 { fail("no words") }

581 }
582 if cmd == 13 || cmd == 14 {

584 var output []string
585 previous := ""
586 for _, w := range words {
587 if cmd == 13 {
588 output = append(output, w.original)

590 }

592 heading := "N"

594 heading = table[id].row

603 }
604 if cmd == 16 {
605 clear(counts)
606 for _, w := range words { counts[w.section] += len(w.tokens) }
607 expected := []int{5, 7, 5, 7, 7}[:sections]
608 verdict, kind := "NG", "haiku"
609 if slices.Equal(counts, expected) { verdict = "OK" }
610 if sections == 5 { kind = "tanka" }
611 var numbers []string
612 for _, c := range counts { numbers = append(numbers, strconv.Itoa(c)) }
613 return verdict+" "+kind+" "+strings.Join(numbers, "-")
614 }
615 if cmd == 17 {
616 var kana []int
617 var output []string
618 for _, w := range words {
619 for _, e := range elements(w) {
620 kana = append(kana, e.id)
621 s := "N"
622 if e.id != moraN { s = table[e.id].id }
623 output = append(output, s)
624 }
625 }
626 verdict := "YES"
627 for i := 0; i < len(kana)/2; i++ {
628 if kana[i] != kana[len(kana)-1-i] {
629 verdict = "NO"
630 }
631 }
632 return verdict+" "+strings.Join(output, ".")
633 }
634 previousTail := -1
635 for i, w := range words {
636 e := elements(w)
637 prefix := fmt.Sprintf("NG %d: %s", i+1, w.original)
638 if i > 0 && e[0].id != previousTail { return prefix+" must start with "+table[previousTail].spelling[source] }
639 for j := 0; j < i; j++ {
640 if slices.Equal(sequence(w, false), sequence(words[j], false)) {
641 return prefix+" is repeated"
642 }
643 }
644 previousTail = e[len(e)-1].id
645 if previousTail == moraN { return prefix+" ends with N" }
646 }
647 return fmt.Sprintf("OK %d", len(words))
648}
