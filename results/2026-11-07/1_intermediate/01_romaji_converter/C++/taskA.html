30bool letter(char c) { return (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z'); }
31bool vowel(char c) { return c != '\0' && string("aiueo").find(c) != string::npos; }
32bool wordChar(char c) { return letter(c) || c == '\'' || c == '-'; }
33char lower(char c) { return c >= 'A' && c <= 'Z' ? c + ('a' - 'A') : c; }
34char upper(char c) { return c >= 'a' && c <= 'z' ? c - ('a' - 'A') : c; }
35string lowerText(string s) {
36 for (char &c : s) c = lower(c);
37 return s;
38}
39vector<string> split(const string &s, char delimiter = ' ') {
40 vector<string> result;
41 std::size_t start = 0;
42 for (std::size_t i = 0; i <= s.size(); ++i) {
43 if (i == s.size() || s[i] == delimiter) {
44 result.push_back(s.substr(start, i - start));
45 start = i + 1;
46 }
47 }
48 return result;
49}
50string join(const vector<string> &items, const string &separator) {
51 string result;
52 for (std::size_t i = 0; i < items.size(); ++i) {
53 if (i) result += separator;
54 result += items[i];
55 }
56 return result;
57}
58int lookup(const vector<string> &items, const string &s) {
59 vector<string>::const_iterator it = std::find(items.begin(), items.end(), s);
60 return it == items.end() ? -1 : int(it - items.begin());

116vector<Word> lex(const string &text) {
117 vector<Word> words;
118 int section = 0;
119 for (int i = 0; i < int(text.size());) {
120 if (!wordChar(text[i])) {
121 if (text[i] == '/') ++section;
122 ++i;
123 continue;
124 }
125 int start = i;
126 while (i < int(text.size()) && wordChar(text[i])) ++i;
127 words.push_back({text.substr(start, i - start), start, i, section, 0, {}});
128 }
129 return words;
130}
131void checkCase(Word &w) {
132 bool allLower = true, allUpper = true, title = w.original[0] >= 'A' && w.original[0] <= 'Z';
133 int letters = 0;
134 for (int i = 0; i < int(w.original.size()); ++i) {
135 char c = w.original[i];
136 if (!letter(c)) continue;
137 ++letters;
138 if (c != lower(c)) allLower = false;
139 if (c != upper(c)) allUpper = false;
140 if (i > 0 && c != lower(c)) title = false;
141 }
142 if (allLower) w.pattern = 0;
143 else if (title) w.pattern = 1;
144 else if (letters >= 2 && allUpper) w.pattern = 2;
145 else columnError(w.start + 1, "mixed case");
146}
147std::pair<int, int> match(const string &s, int pos, int system) {
148 int id = -1, length = 0;
149 for (int i = 0; i < 108; ++i) {
150 for (int t = 0; t < 5; ++t) {
151 if (system >= 0 && t != system) continue;
152 const string &p = table[i].spelling[t];
153 if (int(p.size()) > length && s.compare(pos, p.size(), p) == 0) {
154 id = i;
155 length = int(p.size());
156 }
157 }
158 }
159 return {id, length};
160}
161void parse(Word &w, int system) {
162 w.tokens.clear();
163 string s = lowerText(w.original);
164 for (int p = 0; p < int(s.size());) {
165 char c = s[p], c2 = p + 1 < int(s.size()) ? s[p + 1] : '\0';
166 int id = -1, advance = 1;
167 if (c == '-') {
168 if (w.tokens.empty() || w.tokens.back().id >= N) columnError(w.start + p + 1, "misplaced long mark");
169 id = L;
170 } else if (c == '\'') {
171 columnError(w.start + p + 1, "misplaced apostrophe");
172 } else {
173 std::pair<int, int> found = match(s, p, system);
174 if (found.first >= 0) {
175 id = found.first;
176 advance = found.second;
177 } else if ((system == 4 || system < 0) && s.compare(p, 3, "xtu") == 0) {
178 id = Q;
179 advance = 3;
180 } else if (c == 'n') {
181 id = N;
182 if (system < 0) {
183 char c3 = p + 2 < int(s.size()) ? s[p + 2] : '\0';
184 if (c2 == '\'' || (c2 == 'n' && !vowel(c3) && c3 != 'y' && c3 != '\'')) advance = 2;
185 } else if ((system == 4 && c2 == 'n') || (system != 4 && c2 == '\'')) advance = 2;
186 } else if ((system == 1 || system < 0) && c == 'm' && c2 && string("bmp").find(c2) != string::npos) {
187 id = N;
188 } else if (letter(c) && !vowel(c) && c != 'n' && c2 == c && match(s, p + 1, system).first >= 0) {
189 id = Q;
190 } else if (system <= 1 && c == 't' && s.compare(p + 1, 2, "ch") == 0 && match(s, p + 1, system).first >= 0) {
191 id = Q;
192 } else columnError(w.start + p + 1, "unknown syllable");
193 }
194 w.tokens.push_back({id, w.start + p + 1});
195 p += advance;
196 }
197}
198vector<bool> pairs(const vector<Token> &tokens) {
199 vector<bool> result(tokens.size(), false);
200 for (int i = 0; i + 1 < int(tokens.size());) {
201 int a = tokens[i].id, b = tokens[i + 1].id;
202 if (a < N && b < 5) {
203 char x = table[a].id.back(), y = table[b].id[0];
204 if (x == y || (x == 'o' && y == 'u') || (x == 'e' && y == 'i')) {
205 result[i + 1] = true;
206 i += 2;
207 continue;
208 }
209 }
210 ++i;
211 }
212 return result;
213}
214vector<bool> longs(const vector<Token> &tokens) {
215 vector<bool> result = pairs(tokens);
216 for (int i = 0; i < int(tokens.size()); ++i) {
217 if (tokens[i].id == L) result[i] = true;
218 else if (result[i]) {
219 char x = table[tokens[i - 1].id].id.back(), y = table[tokens[i].id].id[0];
220 result[i] = (x == 'o' && (y == 'o' || y == 'u')) || (x == 'u' && y == 'u');
221 }
222 }
223 return result;
224}
225vector<string> parts(const Word &w, int system, int mode) {
226 vector<bool> longElements = longs(w.tokens);
227 vector<string> result;
228 for (int i = 0; i < int(w.tokens.size()); ++i) {
229 int id = w.tokens[i].id;
230 int nextId = i + 1 < int(w.tokens.size()) ? w.tokens[i + 1].id : -1;
231 string next = nextId >= 0 && nextId < N ? table[nextId].spelling[system] : "";
232 char first = next.empty() ? '\0' : next[0];
233 string s;
234 if (longElements[i]) {
235 char v = table[w.tokens[i - 1].id].id.back();
236 if (mode == 0) s = id == L ? "-" : table[id].spelling[system];
237 if (mode == 2 && v == 'o' && !vowel(first)) s = "h";
238 if (mode == 3) s = "-";
239 if (mode == 4) s = string(1, v);
240 } else if (id < N) s = table[id].spelling[system];
241 else if (id == N) {
242 if (system == 4) s = nextId == N || vowel(first) || first == 'y' || first == 'n' ? "nn" : "n";
243 else if (system == 1 && first && string("bmp").find(first) != string::npos) s = "m";
244 else s = vowel(first) || first == 'y' ? "n'" : "n";
245 } else if (id == Q) {
246 if (first && !vowel(first) && first != 'n') s = system <= 1 && next.compare(0, 2, "ch") == 0 ? "t" : string(1, first);
247 else if (system == 4) s = "xtu";
248 else columnError(w.tokens[i].col, "small tsu cannot be written");
249 }
250 result.push_back(s);
251 }
252 return result;
253}
254string write(const Word &w, int system, bool canonical = false) {
255 string result = join(parts(w, system, canonical ? 0 : longMode), "");
256 int pattern = canonical || caseMode == 0 ? w.pattern : caseMode == 1 ? 0 : caseMode == 2 ? 2 : 1;
257 if (pattern == 1 && !result.empty()) result[0] = upper(result[0]);
258 if (pattern == 2) for (char &c : result) c = upper(c);
259 return result;
260}
261vector<int> sequence(const Word &w, bool loose = false) {
262 vector<bool> omit = longs(w.tokens);
263 vector<int> result;
264 for (int i = 0; i < int(w.tokens.size()); ++i) {
265 if (!loose || !omit[i]) result.push_back(w.tokens[i].id);
266 }
267 return result;
268}
269vector<Element> elements(const Word &w) {
270 vector<Element> result;
271 for (Token t : w.tokens) {
272 if (t.id == L) continue;
273 if (t.id == N) result.push_back({N, 0, 0});
274 else if (t.id == Q) result.push_back({moraId("tu"), 0, 1});
275 else {
276 const Mora &m = table[t.id];
277 string base = m.id;
278 if (m.voice) base[0] = string("ksthh")[lookup(vector<string>{"ga", "za", "da", "ba", "pa"}, m.row)];
279 if (m.dan.size() == 2) {
280 result.push_back({moraId(base.substr(0, 1) + "i"), m.voice, 0});
281 result.push_back({moraId(m.dan), 0, 1});
282 } else result.push_back({moraId(base), m.voice, 0});
283 }
284 }
285 return result;
286}
287std::array<vector<int>, 4> key(const Word &w) {
288 string order = orderMode == 0
289 ? "a i u e o ka ki ku ke ko sa si su se so ta ti tu te to na ni nu ne no ha hi hu he ho ma mi mu me mo ya yu yo ra ri ru re ro wa wi we wo N"
290 : "i ro ha ni ho he to ti ri nu ru wo wa ka yo ta re so tu ne na ra mu u wi no o ku ya ma ke hu ko e te a sa ki yu me mi si we hi mo se su N";
291 vector<string> rank = split(order);
292 std::array<vector<int>, 4> result;
293 for (Element e : elements(w)) {
294 result[0].push_back(lookup(rank, e.id == N ? "N" : table[e.id].id));
295 result[1].push_back(e.voice);
296 result[2].push_back(e.size);
297 }
298 for (Token t : w.tokens) result[3].push_back(t.id == L ? 1 : 0);
299 return result;
300}
301vector<int> patternTokens(const string &pattern, int system) {
302 vector<int> result;
303 string fragment;
304 for (int i = 0; i <= int(pattern.size()); ++i) {
305 char c = i == int(pattern.size()) ? '\0' : pattern[i];
306 if (!c || c == '?' || c == '*') {
307 if (!fragment.empty()) {
308 Word w{fragment, 0, int(fragment.size()), 0, 0, {}};
309 try { parse(w, system); }
310 catch (const std::runtime_error &) { fail("invalid pattern"); }
311 for (Token t : w.tokens) result.push_back(t.id);
312 fragment.clear();
313 }
314 if (c) result.push_back(c == '?' ? -1 : -2);
315 } else {
316 if (!((c >= 'a' && c <= 'z') || c == '\'' || c == '-')) fail("invalid pattern");
317 fragment += c;
318 }
319 }
320 return result;
321}
322bool glob(const vector<int> &pattern, const Word &w) {
323 vector<bool> dp(w.tokens.size() + 1, false);
324 dp[0] = true;
325 for (int p : pattern) {
326 vector<bool> next(dp.size(), false);
327 if (p == -2) {
328 next[0] = dp[0];
329 for (int j = 1; j < int(dp.size()); ++j) next[j] = dp[j] || next[j - 1];
330 } else {
331 for (int j = 1; j < int(dp.size()); ++j) next[j] = dp[j - 1] && (p == -1 || p == w.tokens[j - 1].id);
332 }
333 dp.swap(next);
334 }
335 return dp.back();
336}
337string process(const string &line) {
338 const vector<string> names = {"SET", "TABLE", "CONVERT", "NORMALIZE", "REVERSE", "VALID", "DETECT", "DIFF", "SAME", "MORAE", "SYLLABLES", "ROW", "STATS", "SORT", "INDEX", "FIND", "HAIKU", "KAIBUN", "SHIRITORI"};
339 const int fixed[] = {2, 2, 2, 0, 1, 1, 0, 2, 2, 1, 1, 1, 1, 1, 1, 2, 1, 1, 1};
340 std::size_t firstSpace = line.find(' ');
341 string command = line.substr(0, firstSpace);
342 int cmd = lookup(names, command);
343 if (cmd < 0) fail("unknown command " + command);
344 vector<string> args;
345 string text;
346 if (cmd <= 1) {
347 vector<string> fields = split(line);
348 if (fields.size() < 3) fail("missing argument");
349 if (fields.size() > 3) fail("too many arguments");
350 args.assign(fields.begin() + 1, fields.end());
351 } else {
352 std::size_t pos = 0;
353 for (int i = 0; i <= fixed[cmd]; ++i) {
354 std::size_t end = line.find(' ', pos);
355 if (end == string::npos) fail("missing argument");
356 if (i > 0) args.push_back(line.substr(pos, end - pos));
357 pos = end + 1;
358 }
359 text = line.substr(pos);
360 }
361 if (cmd == 0) {
362 int option = lookup(vector<string>{"LONG", "CASE", "ORDER"}, args[0]);
363 if (option < 0) fail("unknown option " + args[0]);
364 const vector<vector<string>> values = {{"KEEP", "DROP", "PASSPORT", "HYPHEN", "DOUBLE"}, {"KEEP", "LOWER", "UPPER", "TITLE"}, {"GOJUON", "IROHA"}};
365 int value = lookup(values[option], args[1]);
366 if (value < 0) fail("unknown value " + args[1]);
367 if (option == 0) longMode = value;
368 if (option == 1) caseMode = value;
369 if (option == 2) orderMode = value;
370 return args[0] + "=" + args[1];
371 }
372 int systemCount = cmd == 3 || cmd == 6 ? 0 : cmd == 2 || cmd == 7 || cmd == 8 ? 2 : 1;
373 vector<int> sys;
374 for (int i = 0; i < systemCount; ++i) {
375 int s = lookup(systems, args[i]);
376 if (s < 0) fail("unknown system " + args[i]);
377 sys.push_back(s);
378 }
379 int source = sys.empty() ? -1 : sys[0];
380 if (cmd == 1) {
381 if (lookup(rows, args[1]) < 0) fail("unknown row " + args[1]);
382 vector<string> result;
383 for (const Mora &m : table) if (m.row == args[1]) result.push_back(m.spelling[source]);
384 return join(result, " ");
385 }
386 vector<int> pattern;
387 if (cmd == 15) pattern = patternTokens(args[1], source);
388 bool slash = cmd == 8 || cmd == 16;
389 for (int i = 0; i < int(text.size()); ++i) {
390 char c = text[i];
391 if (!wordChar(c) && !(c >= '0' && c <= '9') && string(" .,!?;:()\"").find(c) == string::npos && !(slash && c == '/')) columnError(i + 1, "invalid character");
392 }
393 vector<Word> words = lex(text);
394 int sections = int(std::count(text.begin(), text.end(), '/')) + 1;
395 vector<int> counts(sections, 0);
396 for (const Word &w : words) ++counts[w.section];
397 if (cmd == 8 && (sections != 2 || counts[0] == 0 || counts[1] == 0)) fail("expected two texts");
398 if (cmd == 16 && ((sections != 3 && sections != 5) || std::find(counts.begin(), counts.end(), 0) != counts.end())) fail("expected 3 or 5 lines");
399 if (words.empty()) fail("no words");
400 for (Word &w : words) checkCase(w);
401 if (cmd == 6) {
402 vector<string> verdicts;
403 for (int s = 0; s < 5; ++s) {
404 string verdict = "canonical";
405 try {
406 for (Word &w : words) parse(w, s);
407 for (const Word &w : words) if (write(w, s, true) != w.original) verdict = "readable";
408 } catch (const std::runtime_error &) { verdict = "error"; }
409 verdicts.push_back(systems[s] + "=" + verdict);
410 }
411 return join(verdicts, " ");
412 }
413 for (Word &w : words) parse(w, cmd == 8 && w.section == 1 ? sys[1] : source);
414 if (cmd == 2 || cmd == 3 || cmd == 4) {
415 string result;
416 int pos = 0;
417 for (Word &w : words) {
418 result += text.substr(pos, w.start - pos);
419 if (cmd == 4) {
420 for (int i = 0; i < int(w.tokens.size()); ++i) {
421 if (w.tokens[i].id == L) w.tokens[i].id = moraId(string(1, table[w.tokens[i - 1].id].id.back()));
422 }
423 std::reverse(w.tokens.begin(), w.tokens.end());
424 }
425 result += write(w, cmd == 2 ? sys[1] : cmd == 3 ? 0 : source);
426 pos = w.end;
427 }
428 return result + text.substr(pos);
429 }
430 if (cmd == 5) {
431 for (const Word &w : words) {
432 string canonical = write(w, source, true);
433 if (canonical != w.original) {
434 int i = 0;
435 while (i < int(canonical.size()) && i < int(w.original.size()) && canonical[i] == w.original[i]) ++i;
436 return "INVALID col " + std::to_string(w.start + i + 1) + ": expected " + canonical;
437 }
438 }
439 return "VALID";
440 }
441 if (cmd == 7 || cmd == 15) {
442 vector<string> selected;
443 for (const Word &w : words) {
444 if (cmd == 7) {
445 string a = write(w, source), b = write(w, sys[1]);
446 if (a != b) selected.push_back(a + ">" + b);
447 } else if (glob(pattern, w)) selected.push_back(w.original);
448 }
449 string result = std::to_string(selected.size()) + " of " + std::to_string(words.size());
450 if (!selected.empty()) result += ": " + join(selected, " ");
451 return result;
452 }
453 if (cmd == 8) {
454 vector<Word> left, right;
455 for (const Word &w : words) (w.section == 0 ? left : right).push_back(w);
456 bool exact = true;
457 for (int i = 0; i < int(std::min(left.size(), right.size())); ++i) {
458 if (sequence(left[i], true) != sequence(right[i], true)) return "DIFFERENT at word " + std::to_string(i + 1);
459 if (sequence(left[i]) != sequence(right[i])) exact = false;
460 }
461 if (left.size() != right.size()) return "DIFFERENT at word " + std::to_string(std::min(left.size(), right.size()) + 1);
462 return exact ? "EXACT" : "LOOSE";
463 }
464 if (cmd == 9 || cmd == 10) {
465 vector<string> output;
466 int total = 0;
467 for (const Word &w : words) {
468 vector<string> pieces;
469 if (cmd == 9) {
470 for (Token t : w.tokens) pieces.push_back(t.id < N ? table[t.id].spelling[source] : t.id == N ? "N" : t.id == Q ? "Q" : "-");
471 } else {
472 vector<string> canonical = parts(w, source, 0);
473 vector<bool> paired = pairs(w.tokens);
474 for (int i = 0; i < int(w.tokens.size()); ++i) {
475 if (i > 0 && (w.tokens[i].id >= N || paired[i])) pieces.back() += canonical[i];
476 else pieces.push_back(canonical[i]);
477 }
478 }
479 total += int(pieces.size());
480 output.push_back(join(pieces, "."));
481 }
482 return std::to_string(total) + ": " + join(output, " ");
483 }
484 if (cmd == 11) {
485 vector<string> output;
486 for (const Word &w : words) {
487 for (Token t : w.tokens) {
488 if (t.id < N) {
489 const Mora &m = table[t.id];
490 output.push_back(m.spelling[source] + ": " + m.row + " " + m.dan);
491 } else output.push_back(string(t.id == N ? "N" : t.id == Q ? "Q" : "-") + ": special");
492 }
493 }
494 return join(output, ", ");
495 }
496 if (cmd == 12) {
497 std::array<int, 10> rowCounts{};
498 std::array<int, 6> other{};
499 int total = 0;
500 for (const Word &w : words) {
501 for (Token t : w.tokens) {
502 ++total;
503 if (t.id < N) {
504 const Mora &m = table[t.id];
505 ++rowCounts[m.base];
506 if (m.voice) ++other[m.voice - 1];
507 if (m.dan.size() == 2) ++other[2];
508 } else ++other[t.id - N + 3];
509 }
510 }
511 vector<string> output = {"morae=" + std::to_string(total)};
512 for (int r = 0; r < 10; ++r) if (rowCounts[r]) output.push_back(rows[r] + "=" + std::to_string(rowCounts[r]));
