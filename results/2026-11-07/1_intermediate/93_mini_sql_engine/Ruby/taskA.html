58def statements(source)
59 source
60 .scan(/'[^']*'|--[^\n]*|[^;'\-]+|-(?!-)|;/)
61 .each_with_object([""]) do |part, a|
62 a[-1] << part
63 a << "" if part == ";"
64 end
65 .reject { |s| !s.end_with?(";") }
66end

68def tokenize(text)
69 tokens = []
70 i = 0
71 while i < text.length
72 rest = text[i..]
73 if (m = /\A[ \n]+|\A--[^\n]*/.match(rest))
74 i += m[0].length
75 next
76 elsif (m = /\A'([^']*)'/.match(rest))
77 kind, value = :string, m[1]
78 elsif (m = /\A[0-9]+/.match(rest))
79 kind, value = :integer, m[0].to_i
80 elsif (m = /\A[A-Za-z_][A-Za-z0-9_]*/.match(rest))
81 kind = RESERVED.include?(m[0].upcase) ? :keyword : :name
82 value = kind == :keyword ? m[0].upcase : m[0]
83 elsif (m = /\A(?:<>|!=|<=|>=|\|\||[(),.;*\/%+\-=<>])/.match(rest))
84 kind, value = :symbol, m[0]
85 else
86 raise "unexpected character '#{text[i]}'"
87 end
88
89 tokens << Token.new(kind, value, m[0])
90 i += m[0].length
91 end
92
93 tokens
94end

96class Parser
97 def initialize(tokens)
98 @tokens, @i = tokens, 0
99 end
100
101 def peek = @tokens[@i]
102 def take
103 t = peek
104 @i += 1
105 t
106 end
107
108 def at?(*words) = [:keyword, :symbol].include?(peek.kind) && words.include?(peek.value)
109 def accept(word)
110 return false unless at?(word)
111 @i += 1
112 true
113 end
114
115 def syntax = raise "syntax error at '#{peek.text}'"
116 def expect(word)
117 syntax unless accept(word)
118 end
119
120 def kind(k)
121 syntax unless peek.kind == k
122 take.value
123 end
124
125 def comma
126 a = [yield]
127 a << yield while accept(",")
128 a
129 end
130
131 def parens
132 expect("(")
133 a = comma { yield }
134 expect(")")
135 a
136 end
137
138 def alias_name
139 return kind(:name) if accept("AS")
140 peek.kind == :name ? take.value : nil
141 end
142
143 def clause(word) = accept(word) ? expression : nil
144 def table_ref
145 name = kind(:name)
146 [name, alias_name || name]
147 end
148
149 def reference
150 name = kind(:name)
151 if accept(".")
152 col = kind(:name)
153 Expr.new(kind: :ref, table: name, column: col, text: "#{name}.#{col}")
154 else
155 Expr.new(kind: :ref, column: name, text: name)
156 end
157 end
158
159 def expression = binary(%w[OR], :and_expr)
160 def and_expr = binary(%w[AND], :not_expr)
161 def not_expr = accept("NOT") ? Expr.new(kind: :unary, op: "NOT", args: [not_expr]) : comparison
162 def binary(ops, method)
163 e = send(method)
164 while at?(*ops)
165 op = take.value
166 e = Expr.new(kind: :binary, op: op, args: [e, send(method)])
167 end
168
169 e
170 end
171
172 def comparison
173 e = additive
174 if at?(*COMPARISONS)
175 op = take.value
176 return Expr.new(kind: :binary, op: op, args: [e, additive])
177 end
178
179 if accept("IS")
180 neg = accept("NOT")
181 expect("NULL")
182 return Expr.new(kind: :isnull, args: [e], negated: neg)
183 end
184
185 neg = accept("NOT")
186 if accept("IN")
187 return Expr.new(kind: :in, args: [e, *parens { expression }], negated: neg)
188 elsif accept("BETWEEN")
189 low = additive
190 expect("AND")
191 return Expr.new(kind: :between, args: [e, low, additive], negated: neg)
192 elsif accept("LIKE")
193 return Expr.new(kind: :like, args: [e, additive], negated: neg)
194 end
195
196 syntax if neg
197 e
198 end
199
200 def additive = binary(%w[+ - ||], :multiplicative)
201 def multiplicative = binary(%w[* / %], :unary)
202 def unary = accept("-") ? Expr.new(kind: :unary, op: "-", args: [unary]) : primary
203 def primary
204 return Expr.new(kind: :literal, value: take.value) if [:integer, :string].include?(peek.kind)
205 return Expr.new(kind: :literal, value: nil) if accept("NULL")
206 return reference if peek.kind == :name
207 if at?(*AGGREGATES)
208 name = take.value
209 expect("(")
210 if name == "COUNT" && accept("*")
211 args, distinct = [], false
212 else
213 distinct = accept("DISTINCT")
214 args = [expression]
215 end
216
217 expect(")")
218 return Expr.new(kind: :aggregate, op: name, args: args, distinct: distinct)
219 end
220
221 if accept("(")
222 e = expression
223 expect(")")
224 return e
225 end
226
227 syntax
228 end

230 def expr_tokens
231 start = @i
232 e = expression
233 [e, @tokens[start...@i]]
234 end
235
236 def select_item
237 e, tokens = expr_tokens
238 name = alias_name
239 heading = name ||
240 (e.kind == :ref && tokens[0].kind == :name ? e.column : tokens
241 .each_with_index
242 .map { |t, i| (i > 0 && t.kind != :symbol && tokens[i - 1].kind != :symbol ? " " : "") + t.text }
243 .join)
244 {expr: e, alias: name, heading: heading}
245 end
246
247 def order_item
248 e, tokens = expr_tokens
249 desc = accept("DESC")
250 accept("ASC") unless desc
251 {expr: e, tokens: tokens, desc: desc}
252 end
253
254 def select_query
255 q = {kind: :select, distinct: accept("DISTINCT")}
256 q[:items] = accept("*") ? nil : comma { select_item }
257 expect("FROM")
258 q[:source] = table_ref
259 q[:joins] = []
260 while at?("JOIN", "INNER", "LEFT")
261 left = accept("LEFT")
262 left ? accept("OUTER") : accept("INNER")
263 expect("JOIN")
264 ref = table_ref
265 expect("ON")
266 q[:joins] << {table: ref, left: left, condition: expression}
267 end
268
269 q[:where] = clause("WHERE")
270 q[:group] = []
271 if accept("GROUP")
272 expect("BY")
273 q[:group] = comma { reference }
274 end
275
276 q[:having] = clause("HAVING")
277 q[:order] = []
278 if accept("ORDER")
279 expect("BY")
280 q[:order] = comma { order_item }
281 end
282
283 if accept("LIMIT")
284 q[:limit] = kind(:integer)
285 q[:offset] = kind(:integer) if accept("OFFSET")
286 end
287
288 q
289 end

291 def statement
292 result = nil
293 if accept("CREATE")
294 expect("TABLE")
295 name = kind(:name)
296 columns = parens do
297 n = kind(:name)
298 syntax unless at?("INT", "TEXT")
299 type = take.value
300 pk = 0
301 nn = false
302 loop do
303 if accept("PRIMARY")
304 expect("KEY")
305 pk += 1
306 elsif accept("NOT")
307 expect("NULL")
308 nn = true
309 else
310 break
311 end
312 end
313
314 Column.new(n, type, nn, pk)
315 end
316
317 result = {kind: :create, name: name, columns: columns}
318 elsif accept("DROP")
319 expect("TABLE")
320 result = {kind: :drop, name: kind(:name)}
321 elsif accept("INSERT")
322 expect("INTO")
323 name = kind(:name)
324 cols = at?("(") ? parens { kind(:name) } : nil
325 expect("VALUES")
326 result = {kind: :insert, name: name, columns: cols, tuples: comma { parens { expression } }}
327 elsif accept("UPDATE")
328 name = kind(:name)
329 expect("SET")
330 assignments = comma {
331 n = kind(:name)
332 expect("=")
333 [n, expression]
334 }
335 result = {kind: :update, name: name, assignments: assignments, where: clause("WHERE")}
336 elsif accept("DELETE")
337 expect("FROM")
338 result = {kind: :delete, name: kind(:name), where: clause("WHERE")}
339 elsif accept("SELECT")
340 result = select_query
341 elsif accept("EXIT")
342 result = {kind: :exit}
343 end
344
345 expect(";")
346 result
347 end
348end

350class Scope
351 attr_reader :entries, :width
352 def initialize
353 @entries, @width = [], 0
354 end
355
356 def add(name, table)
357 raise "duplicate table name #{name}" if @entries.any? { |entry| entry[0] == name }
358 @entries << [name, table, @width]
359 @width += table.columns.length
360 end

362 def resolve(ref)
363 entries = ref.table ? @entries.select { |entry| entry[0] == ref.table } : @entries
364 raise "unknown table #{ref.table}" if ref.table && entries.empty?
365 matches = entries.filter_map do |_, table, offset|
366 i = table.column(ref.column)
367 [offset + i, table.columns[i].type] unless i.nil?
368 end
369
370 raise "unknown column #{ref.text}" if matches.empty?
371 raise "ambiguous column #{ref.column}" if matches.length > 1
372 ref.index = matches[0][0]
373 matches[0][1]
374 end
375end

377def compatible(a, b) = a == b || a == "NULL" || b == "NULL"
378def require_type(a, b, op)
379 raise "type mismatch for #{op}" unless compatible(a, b)
380end
381
382def check(e, scope, aggregates = false, grouped = nil, inside = false)
383 case e.kind
384 when :literal
385 e.value.nil? ? "NULL" : e.value.is_a?(Integer) ? "INT" : "TEXT"
386 when :ref
387 type = scope.resolve(e)
388 raise "column #{e.text} is not grouped" if grouped && !inside && !grouped.include?(e.index)
389 type
390 when :aggregate
391 raise "misuse of aggregate #{e.op}" if !aggregates || inside
392 return "INT" if e.args.empty?
393 type = check(e.args[0], scope, aggregates, grouped, true)
394 require_type(type, "INT", e.op) if %w[SUM AVG].include?(e.op)
395 %w[MIN MAX].include?(e.op) ? type : "INT"
396 else
397 types = e.args.map { |arg| check(arg, scope, aggregates, grouped, inside) }
398 case e.kind
399 when :unary
400 require_type(types[0], "INT", e.op)
401 "INT"
402 when :binary
403 if COMPARISONS.include?(e.op)
404 require_type(types[0], types[1], e.op)
405 "INT"
406 else
407 type = e.op == "||" ? "TEXT" : "INT"
408 types.each { |t| require_type(t, type, e.op) }
409 type
410 end
411
412 when :isnull
413 "INT"
414 when :in, :between
415 types.drop(1).each { |t| require_type(t, types[0], e.kind == :in ? "IN" : "BETWEEN") }
416 "INT"
417 when :like
418 types.each { |t| require_type(t, "TEXT", "LIKE") }
419 "INT"
420 end
421 end
422end

424def condition_check(e, scope, clause, aggregates = false, grouped = nil)
425 require_type(check(e, scope, aggregates, grouped), "INT", clause) if e
426end

428def has_aggregate(e) = e.kind == :aggregate || (e.args || []).any? { |a| has_aggregate(a) }
429def true_value(v) = !v.nil? && v != 0
430def logical_not(v) = v.nil? ? nil : v == 0 ? 1 : 0
431def quotient(a, b) = (a < 0) == (b < 0) ? a.abs / b.abs : -(a.abs / b.abs)
432def operate(op, a, b)
433 return a == 0 || b == 0 ? 0 : a.nil? || b.nil? ? nil : 1 if op == "AND"
434 return true_value(a) || true_value(b) ? 1 : a.nil? || b.nil? ? nil : 0 if op == "OR"
435 return nil if a.nil? || b.nil?
436 case op
437 when "+", "||"
438 a + b
439 when "-"
440 a - b
441 when "*"
442 a * b
443 when "/"
444 b == 0 ? nil : quotient(a, b)
445 when "%"
446 b == 0 ? nil : a - b * quotient(a, b)
447 when "="
448 a == b ? 1 : 0
449 when "<>", "!="
450 a != b ? 1 : 0
451 else
452 a.public_send(op, b) ? 1 : 0
453 end
454end
455
456def like(text, pattern)
457 dp = [true] + [false] * text.length
458 pattern.each_char do |char|
459 if char == "%"
460 (1..text.length).each { |i| dp[i] ||= dp[i - 1] }
461 else
462 text.length.downto(1) { |i| dp[i] = dp[i - 1] && (char == "_" || char == text[i - 1]) }
463 dp[0] = false
464 end
465 end
466
467 dp[-1] ? 1 : 0
468end

470def evaluate(e, row, group = nil)
471 return e.value if e.kind == :literal
472 return row[e.index] if e.kind == :ref
473 if e.kind == :aggregate
474 return group.length if e.args.empty?
475 values = group.map { |r| evaluate(e.args[0], r) }.compact
476 values = values.uniq if e.distinct
477 return values.length if e.op == "COUNT"
478 return nil if values.empty?
479 return case e.op
480 when "SUM"
481 values.sum
482 when "MIN"
483 values.min
484 when "MAX"
485 values.max
486 when "AVG"
487 quotient(values.sum, values.length)
488 end
489 end
490
491 a = e.args.map { |arg| evaluate(arg, row, group) }
492 case e.kind
493 when :unary
494 e.op == "NOT" ? logical_not(a[0]) : a[0].nil? ? nil : -a[0]
495 when :binary
496 operate(e.op, *a)
497 when :isnull
498 a[0].nil? != e.negated ? 1 : 0
499 else
500 result = case e.kind
501 when :in
502 a[0].nil? ? nil : a.drop(1).include?(a[0]) ? 1 : a.drop(1).include?(nil) ? nil : 0
503 when :between
504 operate("AND", operate(">=", a[0], a[1]), operate("<=", a[0], a[2]))
505 when :like
506 a.any?(&:nil?) ? nil : like(*a)
507 end
508
509 e.negated ? logical_not(result) : result
510 end
511end
512
513def satisfies(e, row, group = nil) = e.nil? || true_value(evaluate(e, row, group))
514def row_count(n) = "#{n} #{n == 1 ? "row" : "rows"}"
515def value_text(v) = v.nil? ? "NULL" : v.to_s
516
517def print_table(items, types, rows)
518 headings = items.map { |i| i[:heading] }
519 cells = rows.map { |r| r.map { |v| value_text(v) } }
520 widths = headings.each_index.map { |i| ([headings[i].length] + cells.map { |r| r[i].length }).max }
521 separator = "+" + widths.map { |w| "-" * (w + 2) + "+" }.join
522 puts(separator)
523 puts("|" + headings.each_index.map { |i| " #{headings[i].ljust(widths[i])} |" }.join)
524 puts(separator)
525 cells.each { |r|
526 puts("|" + r.each_index.map { |i| " #{types[i] == "INT" ? r[i].rjust(widths[i]) : r[i].ljust(widths[i])} |" }.join)
527 }
528 puts(separator)
529 puts("(#{row_count(rows.length)})")
530end

532class Database
533 def initialize = @tables = {}
534 def table(name) = @tables.fetch(name) { raise "unknown table #{name}" }
535 def indices(t, names)
536 result = []
537 names.each do |n|
538 i = t.column(n)
539 raise "unknown column #{n}" if i.nil?
540 raise "duplicate column #{n}" if result.include?(i)
541 result << i
542 end
543
544 result
545 end
546
547 def constraints(t, rows, checked)
548 checked.each do |i|
549 t.columns.each_with_index { |c, j| raise "NULL value in column #{c.name}" if c.not_null && rows[i][j].nil? }
550 if t.key && rows.count { |r| r[t.key] == rows[i][t.key] } > 1
551 raise "duplicate primary key #{value_text(rows[i][t.key])}"
552 end
553 end
554 end
555
556 def execute(s)
557 case s[:kind]
558 when :create
559 name = s[:name]
560 raise "table #{name} already exists" if @tables.key?(name)
561 columns = []
562 key = nil
563 primary = 0
564 s[:columns].each do |c|
565 raise "duplicate column #{c.name}" if columns.any? { |p| p.name == c.name }
566 primary += c.primary
567 raise "multiple primary keys" if primary >= 2
568 key = columns.length if c.primary > 0
569 c.not_null ||= c.primary > 0
570 columns << c
571 end
572
573 @tables[name] = Table.new(name, columns, key, [])
574 puts("Table #{name} created.")
575 when :drop
576 table(s[:name])
577 @tables.delete(s[:name])
578 puts("Table #{s[:name]} dropped.")
579 when :insert
580 t = table(s[:name])
581 targets = s[:columns] ? indices(t, s[:columns]) : (0...t.columns.length).to_a
582 s[:tuples].each do |tuple|
583 raise "expected #{targets.length} values, got #{tuple.length}" if tuple.length != targets.length
584 tuple.zip(targets).each { |e, i|
585 require_type(check(e, Scope.new), t.columns[i].type, "column #{t.columns[i].name}")
586 }
587 end
588
589 additions = s[:tuples].map do |tuple|
590 row = [nil] * t.columns.length
591 tuple.zip(targets).each { |e, i| row[i] = evaluate(e, []) }
592 row
593 end
594
595 rows = t.rows + additions
596 constraints(t, rows, (t.rows.length...rows.length).to_a)
597 t.rows = rows
598 puts("#{row_count(additions.length)} inserted.")
599 when :update, :delete
600 t = table(s[:name])
601 scope = Scope.new
602 scope.add(t.name, t)
603 if s[:kind] == :update
604 targets = indices(t, s[:assignments].map(&:first))
605 s[:assignments].zip(targets).each { |(_, e), i|
606 require_type(check(e, scope), t.columns[i].type, "column #{t.columns[i].name}")
607 }
608 condition_check(s[:where], scope, "WHERE")
609 rows = t.rows.dup
610 changed = []
611 t.rows.each_with_index do |row, i|
612 next unless satisfies(s[:where], row)
613 replacement = row.dup
614 s[:assignments].zip(targets).each { |(_, e), j| replacement[j] = evaluate(e, row) }
615 rows[i] = replacement
616 changed << i
617 end
618
619 constraints(t, rows, changed)
620 t.rows = rows
621 puts("#{row_count(changed.length)} updated.")
622 else
623 condition_check(s[:where], scope, "WHERE")
624 rows = t.rows.reject { |r| satisfies(s[:where], r) }
625 count = t.rows.length - rows.length
626 t.rows = rows
627 puts("#{row_count(count)} deleted.")
628 end
629
630 when :select
631 select(s)
632 end
633 end

635 def select(q)
636 scope = Scope.new
637 scope.add(q[:source][1], table(q[:source][0]))
638 q[:joins].each { |j|
639 scope.add(j[:table][1], table(j[:table][0]))
640 condition_check(j[:condition], scope, "ON")
641 }
642 condition_check(q[:where], scope, "WHERE")
643 q[:group].each { |ref| scope.resolve(ref) }
644 items = q[:items] ||
645 scope.entries.flat_map { |name, t, _|
646 t.columns.map { |c| {expr: Expr.new(kind: :ref, table: name, column: c.name, text: c.name), heading: c.name} }
647 }
648 aggregate = !q[:group].empty? || q[:having] || (items + q[:order]).any? { |i| has_aggregate(i[:expr]) }
649 grouped = aggregate ? q[:group].map(&:index) : nil
650 types = items.map { |i| check(i[:expr], scope, true, grouped) }
651 condition_check(q[:having], scope, "HAVING", true, grouped)
652 keys = q[:order].map do |o|
653 ts = o[:tokens]
654 if ts.length == 1 && ts[0].kind == :integer
655 k = ts[0].value
656 raise "ORDER BY position #{k} is out of range" unless k >= 1 && k <= items.length
657 items[k - 1][:expr]
658 elsif ts.length == 1 && ts[0].kind == :name && (item = items.find { |i| i[:alias] == ts[0].value })
659 item[:expr]
660 else
661 check(o[:expr], scope, true, grouped)
662 o[:expr]
663 end
664 end
665
666 rows = scope.entries[0][1].rows
667 q[:joins].zip(scope.entries.drop(1)).each do |join, (_, t, _)|
668 rows = rows.flat_map do |left|
669 joined = t.rows.filter_map { |right|
670 combined = left + right
671 combined if satisfies(join[:condition], combined)
672 }
673 joined.empty? && join[:left] ? [left + [nil] * t.columns.length] : joined
674 end
675 end
676
677 rows = rows.select { |r| satisfies(q[:where], r) }
678 sources = if aggregate
679 groups = q[:group].empty? ? [rows] : rows.group_by { |r| grouped.map { |i| r[i] } }.values
680 groups.select { |g| satisfies(q[:having], g[0], g) }.map { |g| [g[0], g] }
681 else
682 rows.map { |r| [r, nil] }
683 end
684
685 results = sources.each_with_index.map { |(r, g), i|
686 [items.map { |it| evaluate(it[:expr], r, g) }, keys.map { |key| evaluate(key, r, g) }, i]
687 }
688 results.sort! do |a, b|
689 comparison = 0
690 keys.each_index do |i|
691 x, y = a[1][i], b[1][i]
692 comparison = x.nil? ? y.nil? ? 0 : -1 : y.nil? ? 1 : x <=> y
693 comparison = -comparison if q[:order][i][:desc]
694 break unless comparison == 0
695 end
696
697 comparison == 0 ? a[2] <=> b[2] : comparison
698 end
699
700 results = results.uniq { |r| r[0] } if q[:distinct]
701 results = results.drop(q[:offset] || 0)
702 results = results.take(q[:limit]) if q.key?(:limit)
703 print_table(items, types, results.map(&:first))
704 end
705end
