16List<Token> tokenize(String source) {
17 final pattern = RegExp(r"[ \n]+|--[^\n]*|'[^']*'|[0-9]+|[A-Za-z_][A-Za-z0-9_]*|<>|!=|<=|>=|\|\||[(),.;*/%+\-=<>]");
18 final tokens = <Token>[];
19 var i = 0;
20 while (i < source.length) {
21 final match = pattern.matchAsPrefix(source, i);
22 if (match == null) fail("unexpected character '${source[i]}'");
23 final text = match.group(0)!;
24 i = match.end;
25 if (text.startsWith(' ') || text.startsWith('\n') || text.startsWith('--')) continue;
26 final kind = text.startsWith("'") ? 'string' : RegExp(r'^[0-9]').hasMatch(text) ? 'integer' : RegExp(r'^[A-Za-z_]').hasMatch(text) ? (reserved.contains(text.toUpperCase()) ? 'keyword' : 'name') : 'symbol';
27 tokens.add(Token(kind, text, kind == 'keyword' ? text.toUpperCase() : text));
28 }
29 return tokens;
30}

109class Parser {
110 final List<Token> tokens;
111 int pos = 0;
112 Parser(this.tokens);
113 Token get peek => tokens[pos < tokens.length ? pos : tokens.length - 1];
114 bool at(List<String> keys) => (peek.kind == 'keyword' || peek.kind == 'symbol') && keys.contains(peek.key);
115 Token take() {
116 final t = peek;
117 pos++;
118 return t;
119 }
120 bool accept(String key) {
121 if (at([key])) {
122 pos++;
123 return true;
124 }
125 return false;
126 }
127 Never syntax() => fail("syntax error at '${peek.text}'");
128 void expect(String key) {
129 if (!accept(key)) syntax();
130 }
131 String name() {
132 if (peek.kind != 'name') syntax();
133 return take().text;
134 }
135 int integer() {
136 if (peek.kind != 'integer') syntax();
137 return int.parse(take().text);
138 }
139 List<T> list<T>(T Function() parse) {
140 final a = [parse()];
141 while (accept(',')) {
142 a.add(parse());
143 }
144 return a;
145 }
146 List<T> parenthesized<T>(T Function() parse) {
147 expect('(');
148 final a = list(parse);
149 expect(')');
150 return a;
151 }
152 String? alias() {
153 if (accept('AS')) return name();
154 return peek.kind == 'name' ? take().text : null;
155 }
156 Expr ref() {
157 final first = name();
158 if (accept('.')) {
159 final c = name();
160 return Expr('column', table: first, column: c, written: '$first.$c');
161 }
162 return Expr('column', column: first, written: first);
163 }
164 TableRef tableRef() {
165 final n = name();
166 return TableRef(n, alias());
167 }
168 Expr? clause(String key) => accept(key) ? expr() : null;
169 Statement parse() {
170 final s = Statement();
171 if (accept('CREATE')) {
172 s.command = 'CREATE';
173 expect('TABLE');
174 s.name = name();
175 s.columns = parenthesized(() {
176 final n = name();
177 if (!at(['INT', 'TEXT'])) syntax();
178 final type = take().key;
179 var keys = 0;
180 var nn = false;
181 while (at(['PRIMARY', 'NOT'])) {
182 if (accept('PRIMARY')) {
183 expect('KEY');
184 keys++;
185 } else {
186 expect('NOT');
187 expect('NULL');
188 nn = true;
189 }
190 }
191 return Column(n, type, nn || keys > 0, keys);
192 });
193 } else if (accept('DROP')) {
194 s.command = 'DROP';
195 expect('TABLE');
196 s.name = name();
197 }
198 else if (accept('INSERT')) {
199 s.command = 'INSERT';
200 expect('INTO');
201 s.name = name();
202 if (at(['('])) s.names = parenthesized(name);
203 expect('VALUES');
204 s.tuples = list(() => parenthesized(expr));
205 }
206 else if (accept('UPDATE')) {
207 s.command = 'UPDATE';
208 s.name = name();
209 expect('SET');
210 s.assignments = list(() {
211 final n = name();
212 expect('=');
213 return (n, expr());
214 });
215 s.whereExpr = clause('WHERE');
216 }
217 else if (accept('DELETE')) {
218 s.command = 'DELETE';
219 expect('FROM');
220 s.name = name();
221 s.whereExpr = clause('WHERE');
222 }
223 else if (accept('SELECT')) {
224 s.command = 'SELECT';
225 s.distinct = accept('DISTINCT');
226 if (!accept('*')) s.items = list(() {
227 final start = pos;
228 final e = expr();
229 final t = tokens.sublist(start, pos);
230 final a = alias();
231 var heading = '';
232 for (var i = 0; i < t.length; i++) {
233 if (i > 0 && t[i].kind != 'symbol' && t[i-1].kind != 'symbol') heading += ' ';
234 heading += t[i].text;
235 }
236 return Item(e, a, a ?? (e.kind == 'column' && t[0].kind == 'name' ? e.column : heading));
237 });
238 expect('FROM');
239 s.source = tableRef();
240 while (at(['JOIN', 'LEFT', 'INNER'])) {
241 final outer = accept('LEFT');
242 if (outer) {
243 accept('OUTER');
244 } else {
245 accept('INNER');
246 }
247 expect('JOIN');
248 final t = tableRef();
249 expect('ON');
250 s.joins.add(Join(t, outer, expr()));
251 }
252 s.whereExpr = clause('WHERE');
253 if (accept('GROUP')) {
254 expect('BY');
255 s.grouping = list(ref);
256 }
257 s.having = clause('HAVING');
258 if (accept('ORDER')) {
259 expect('BY');
260 s.orders = list(() {
261 final start = pos;
262 final e = expr();
263 final t = tokens.sublist(start, pos);
264 final desc = accept('DESC');
265 if (!desc) accept('ASC');
266 return Order(e, t, desc);
267 });
268 }
269 if (accept('LIMIT')) {
270 s.limit = integer();
271 if (accept('OFFSET')) s.offset = integer();
272 }
273 } else if (accept('EXIT')) {
274 s.command = 'EXIT';
275 }
276 expect(';');
277 return s;
278 }
279 Expr binary(List<String> ops, Expr Function() operand) {
280 var e = operand();
281 while (at(ops)) {
282 final op = take().key;
283 e = Expr('binary', op: op, children: [e, operand()]);
284 }
285 return e;
286 }
287 Expr expr() => binary(['OR'], andExpr);
288 Expr andExpr() => binary(['AND'], notExpr);
289 Expr notExpr() => accept('NOT') ? Expr('unary', op: 'NOT', children: [notExpr()]) : comparison();
290 Expr comparison() {
291 final e = additive();
292 if (at(comparisons)) {
293 final op = take().key;
294 return Expr('binary', op: op, children: [e, additive()]);
295 }
296 if (accept('IS')) {
297 final n = accept('NOT');
298 expect('NULL');
299 return Expr('isnull', children: [e])..negated = n;
300 }
301 final n = accept('NOT');
302 Expr? r;
303 if (accept('IN')) {
304 r = Expr('in', children: [e, ...parenthesized(expr)]);
305 }
306 else if (accept('BETWEEN')) {
307 final low = additive();
308 expect('AND');
309 r = Expr('between', children: [e, low, additive()]);
310 }
311 else if (accept('LIKE')) {
312 r = Expr('like', children: [e, additive()]);
313 }
314 else if (n) {
315 syntax();
316 }
317 if (r != null) {
318 r.negated = n;
319 return r;
320 }
321 return e;
322 }
323 Expr additive() => binary(['+', '-', '||'], multiplicative);
324 Expr multiplicative() => binary(['*', '/', '%'], unary);
325 Expr unary() => accept('-') ? Expr('unary', op: '-', children: [unary()]) : primary();
326 Expr primary() {
327 if (peek.kind == 'integer') return Expr('literal', literal: int.parse(take().text));
328 if (peek.kind == 'string') {
329 final t = take().text;
330 return Expr('literal', literal: t.substring(1, t.length-1));
331 }
332 if (accept('NULL')) return Expr('literal');
333 if (peek.kind == 'name') return ref();
334 if (at(aggregateNames)) {
335 final n = take().key;
336 expect('(');
337 final e = Expr('aggregate', op: n);
338 if (n == 'COUNT' && accept('*')) {
339 } else {
340 e.distinct = accept('DISTINCT');
341 e.children = [expr()];
342 }
343 expect(')');
344 return e;
345 }
346 if (accept('(')) {
347 final e = expr();
348 expect(')');
349 return e;
350 }
351 syntax();
352 }
353}

370 String resolve(Expr e) {
371 var selected = entries;
372 if (e.table != null) {
373 selected = entries.where((entry) => entry.name == e.table).toList();
374 if (selected.isEmpty) fail('unknown table ${e.table}');
375 }
376 final candidates = <(int, String)>[];
377 for (final entry in selected) {
378 for (var i = 0; i < entry.table.columns.length; i++) {
379 final c = entry.table.columns[i];
380 if (c.name == e.column) candidates.add((entry.offset+i, c.type));
381 }
382 }
383 if (candidates.isEmpty) fail('unknown column ${e.written}');
384 if (candidates.length > 1) fail('ambiguous column ${e.column}');
385 e.index = candidates[0].$1;
386 return candidates[0].$2;
387 }

389String type(Object? value) => value == null ? 'NULL' : value is int ? 'INT' : 'TEXT';
390String written(Object? value) => value?.toString() ?? 'NULL';
391void require(String a, String b, String op) {
392 if (a != b && a != 'NULL' && b != 'NULL') fail('type mismatch for $op');
393}
394String check(Expr e, Scope scope, [bool allowed = false, Set<int>? grouped, bool inside = false]) {
395 if (e.kind == 'literal') return type(e.literal);
396 if (e.kind == 'column') {
397 final t = scope.resolve(e);
398 if (grouped != null && !inside && !grouped.contains(e.index)) fail('column ${e.written} is not grouped');
399 return t;
400 }
401 if (e.kind == 'aggregate') {
402 if (!allowed || inside) fail('misuse of aggregate ${e.op}');
403 if (e.children.isEmpty) return 'INT';
404 final t = check(e.children[0], scope, allowed, grouped, true);
405 if (['SUM', 'AVG'].contains(e.op)) require(t, 'INT', e.op);
406 return ['MIN', 'MAX'].contains(e.op) ? t : 'INT';
407 }
408 final types = e.children.map((c) => check(c, scope, allowed, grouped, inside)).toList();
409 switch (e.kind) {
410 case 'unary': require(types[0], 'INT', e.op);
411 case 'binary':
412 if (comparisons.contains(e.op)) {
413 require(types[0], types[1], e.op);
414 }
415 else {
416 final t = e.op == '||' ? 'TEXT' : 'INT';
417 for (final a in types) {
418 require(a, t, e.op);
419 }
420 return t;
421 }
422 case 'in': case 'between': for (final t in types.skip(1)) {
423 require(t, types[0], e.kind == 'in' ? 'IN' : 'BETWEEN');
424 }
425 case 'like': for (final t in types) {
426 require(t, 'TEXT', 'LIKE');
427 }
428 }
429 return 'INT';
430}
431void condition(Expr? e, Scope scope, String clause, [bool allowed = false, Set<int>? grouped]) {
432 if (e != null) require(check(e, scope, allowed, grouped), 'INT', clause);
433}
434bool truth(Object? v) => v != null && v != 0;
435Object? logicalNot(Object? v) => v == null ? null : truth(v) ? 0 : 1;
436int compare(Object? a, Object? b) {
437 if (a == b) return 0;
438 if (a == null) return -1;
439 if (b == null) return 1;
440 return a is int ? a.compareTo(b as int) : (a as String).compareTo(b as String);
441}
442Object? operation(String op, Object? a, Object? b) {
443 if (op == 'AND') return a == 0 || b == 0 ? 0 : a == null || b == null ? null : 1;
444 if (op == 'OR') return truth(a) || truth(b) ? 1 : a == null || b == null ? null : 0;
445 if (a == null || b == null) return null;
446 switch (op) {
447 case '+': return (a as int) + (b as int);
448 case '-': return (a as int) - (b as int);
449 case '*': return (a as int) * (b as int);
450 case '/': return b == 0 ? null : (a as int) ~/ (b as int);
451 case '%': return b == 0 ? null : (a as int).remainder(b as int);
452 case '||': return (a as String) + (b as String);
453 case '=': return a == b ? 1 : 0;
454 case '!=': case '<>': return a != b ? 1 : 0;
455 case '<': return compare(a, b) < 0 ? 1 : 0;
456 case '<=': return compare(a, b) <= 0 ? 1 : 0;
457 case '>': return compare(a, b) > 0 ? 1 : 0;
458 default: return compare(a, b) >= 0 ? 1 : 0;
459 }
460}
461bool like(String text, String pattern) {
462 final matched = [true, ...List.filled(text.length, false)];
463 for (final c in pattern.split('')) {
464 if (c == '%') {
465 for (var i = 1; i <= text.length; i++) {
466 matched[i] = matched[i] || matched[i-1];
467 }
468 }
469 else {
470 for (var i = text.length; i > 0; i--) {
471 matched[i] = matched[i-1] && (c == '_' || c == text[i-1]);
472 }
473 matched[0] = false;
474 }
475 }
476 return matched.last;
477}
478Object? evaluate(Expr e, List<Object?> row, [List<List<Object?>> group = const []]) {
479 if (e.kind == 'literal') return e.literal;
480 if (e.kind == 'column') return row[e.index];
481 if (e.kind == 'aggregate') {
482 if (e.children.isEmpty) return group.length;
483 var values = group.map((r) => evaluate(e.children[0], r)).where((v) => v != null).toList();
484 if (e.distinct) values = values.toSet().toList();
485 if (e.op == 'COUNT') return values.length;
486 if (values.isEmpty) return null;
487 if (e.op == 'MIN') return values.reduce((a,b) => compare(a,b) < 0 ? a : b);
488 if (e.op == 'MAX') return values.reduce((a,b) => compare(a,b) > 0 ? a : b);
489 final sum = values.fold<int>(0, (a,b) => a + (b as int));
490 return e.op == 'AVG' ? sum ~/ values.length : sum;
491 }
492 final v = e.children.map((c) => evaluate(c, row, group)).toList();
493 Object? result;
494 switch (e.kind) {
495 case 'unary': return e.op == 'NOT' ? logicalNot(v[0]) : v[0] == null ? null : -(v[0] as int);
496 case 'binary': return operation(e.op, v[0], v[1]);
497 case 'isnull': result = v[0] == null ? 1 : 0;
498 case 'in': result = v[0] == null ? null : v.skip(1).contains(v[0]) ? 1 : v.skip(1).contains(null) ? null : 0;
499 case 'between': result = operation('AND', operation('>=',v[0],v[1]), operation('<=',v[0],v[2]));
500 case 'like': result = v.contains(null) ? null : like(v[0] as String, v[1] as String) ? 1 : 0;
501 }
502 return e.negated ? logicalNot(result) : result;
503}

564void executeSelect(Statement s) {
565 final scope = Scope();
566 final tables = <Table>[];
567 final refs = [s.source!, ...s.joins.map((j) => j.table)];
568 for (var i = 0; i < refs.length; i++) {
569 final r = refs[i];
570 final t = lookup(r.name);
571 scope.add(r.scopeName,t);
572 tables.add(t);
573 if (i > 0) condition(s.joins[i-1].condition,scope,'ON');
574 }
575 condition(s.whereExpr,scope,'WHERE');
576 for (final e in s.grouping) {
577 scope.resolve(e);
578 }
579 final aggregate = s.grouping.isNotEmpty || s.having != null || (s.items ?? []).any((i) => i.expr.hasAggregate) || s.orders.any((o) => o.expr.hasAggregate);
580 final grouped = aggregate ? s.grouping.map((e) => e.index).toSet() : null;
581 final items = s.items ?? [for (final entry in scope.entries) for (final c in entry.table.columns) Item(Expr('column',table:entry.name,column:c.name,written:c.name),null,c.name)];
582 final types = items.map((i) => check(i.expr,scope,true,grouped)).toList();
583 condition(s.having,scope,'HAVING',true,grouped);
584 final keys = <Expr>[];
585 for (final order in s.orders) {
586 if (order.tokens.length == 1 && order.tokens[0].kind == 'integer') {
587 final k = int.parse(order.tokens[0].text);
588 if (k < 1 || k > items.length) fail('ORDER BY position $k is out of range');
589 keys.add(items[k-1].expr);
590 }
591 else {
592 final i = order.tokens.length == 1 && order.tokens[0].kind == 'name' ? items.indexWhere((i) => i.alias == order.tokens[0].text) : -1;
593 if (i >= 0) {
594 keys.add(items[i].expr);
595 } else {
596 check(order.expr,scope,true,grouped);
597 keys.add(order.expr);
598 }
599 }
600 }
601 var rows = tables[0].rows;
602 for (var j = 0; j < s.joins.length; j++) {
603 final join = s.joins[j];
604 final next = <List<Object?>>[];
605 for (final l in rows) {
606 var matched = false;
607 for (final r in tables[j+1].rows) {
608 final combined = [...l,...r];
609 if (truth(evaluate(join.condition,combined))) {
610 next.add(combined);
611 matched = true;
612 }
613 }
614 if (!matched && join.outer) next.add([...l,...List<Object?>.filled(tables[j+1].columns.length,null)]);
615 }
616 rows = next;
617 }
618 if (s.whereExpr != null) rows = rows.where((r) => truth(evaluate(s.whereExpr!,r))).toList();
619 var groups = <List<List<Object?>>>[];
620 if (aggregate) {
621 if (s.grouping.isEmpty) {
622 groups = [rows];
623 }
624 else {
625 final indices = <String, int>{};
626 for (final r in rows) {
627 final key = rowKey(s.grouping.map((e) => r[e.index]).toList());
628 final i = indices[key];
629 if (i != null) {
630 groups[i].add(r);
631 } else {
632 indices[key] = groups.length;
633 groups.add([r]);
634 }
635 }
636 }
637 } else {
638 groups = rows.map((r) => [r]).toList();
639 }
640 var results = <Result>[];
641 for (final g in groups) {
642 final row = g.isEmpty ? List<Object?>.filled(scope.width,null) : g[0];
643 if (s.having != null && !truth(evaluate(s.having!,row,g))) continue;
644 results.add(Result(items.map((i) => evaluate(i.expr,row,g)).toList(),keys.map((k) => evaluate(k,row,g)).toList(),results.length));
645 }
646 if (keys.isNotEmpty) results.sort((a,b) {
647 for (var i = 0; i < keys.length; i++) {
648 final c = compare(a.keys[i],b.keys[i]);
649 if (c != 0) return s.orders[i].descending ? -c : c;
650 }
651 return a.ordinal.compareTo(b.ordinal);
652 });
653 if (s.distinct) {
654 final seen = <String>{};
655 results = results.where((r) => seen.add(rowKey(r.values))).toList();
656 }
657 printTable(items,types,results.skip(s.offset).take(s.limit ?? results.length).map((r) => r.values).toList());
658}
659void execute(Statement s) {
660 if (s.command.isEmpty || s.command == 'EXIT') return;
661 if (s.command == 'SELECT') {
662 executeSelect(s);
663 return;
664 }
665 if (s.command == 'CREATE') {
666 if (database.containsKey(s.name)) fail('table ${s.name} already exists');
667 final seen = <String>{};
668 var keys = 0;
669 for (final c in s.columns) {
670 if (!seen.add(c.name)) fail('duplicate column ${c.name}');
671 keys += c.keys;
672 if (keys > 1) fail('multiple primary keys');
673 }
674 database[s.name] = Table(s.name,s.columns);
675 print('Table ${s.name} created.');
676 return;
677 }
678 final original = lookup(s.name);
679 final t = Table(original.name,original.columns,original.rows.map((r) => List<Object?>.of(r)).toList());
680 if (s.command == 'DROP') {
681 database.remove(s.name);
682 print('Table ${s.name} dropped.');
683 return;
684 }
685 final scope = Scope()..add(t.name,t);
686 if (s.command == 'INSERT') {
687 final columns = targets(s.names ?? t.columns.map((c) => c.name).toList(),t);
688 for (final tuple in s.tuples) {
689 if (tuple.length != columns.length) fail('expected ${columns.length} values, got ${tuple.length}');
690 for (var j = 0; j < tuple.length; j++) {
691 final c = t.columns[columns[j]];
692 require(check(tuple[j],Scope()),c.type,'column ${c.name}');
693 }
694 }
695 final start = t.rows.length;
696 for (final tuple in s.tuples) {
697 final row = List<Object?>.filled(t.columns.length,null);
698 for (var j = 0; j < tuple.length; j++) {
699 row[columns[j]] = evaluate(tuple[j],[]);
700 }
701 t.rows.add(row);
702 }
703 constraints(t,List.generate(t.rows.length-start,(i) => start+i));
704 database[s.name] = t;
705 countMessage(s.tuples.length,'inserted');
706 return;
707 }
708 var assigned = <int>[];
709 if (s.command == 'UPDATE') {
710 assigned = targets(s.assignments.map((a) => a.$1).toList(),t);
711 for (var j = 0; j < assigned.length; j++) {
712 final c = t.columns[assigned[j]];
713 require(check(s.assignments[j].$2,scope),c.type,'column ${c.name}');
714 }
715 }
716 condition(s.whereExpr,scope,'WHERE');
717 final affected = List.generate(t.rows.length,(i) => i).where((i) => s.whereExpr == null || truth(evaluate(s.whereExpr!,t.rows[i]))).toList();
718 if (s.command == 'DELETE') {
719 final removed = affected.toSet();
720 t.rows = [
721 for (var i = 0; i < t.rows.length; i++)
722 if (!removed.contains(i)) t.rows[i]
723 ];
724 }
725 else {
726 for (final i in affected) {
727 final before = List<Object?>.of(t.rows[i]);
728 for (var j = 0; j < assigned.length; j++) {
729 t.rows[i][assigned[j]] = evaluate(s.assignments[j].$2,before);
730 }
731 }
732 constraints(t,affected);
733 }
734 database[s.name] = t;
735 countMessage(affected.length,s.command == 'DELETE' ? 'deleted' : 'updated');
736}
737void main() {
738 final input = stdin.readLineSync;
739 final buffer = StringBuffer();
740 String? line;
741 while ((line = input()) != null) {
742 buffer.writeln(line);
743 }
744 final source = buffer.toString();
745 var start = 0;
746 var quoted = false;
747 var comment = false;
748 for (var i = 0; i < source.length; i++) {
749 final c = source[i];
750 if (comment) {
751 if (c == '\n') comment = false;
752 }
753 else if (c == "'") {
754 quoted = !quoted;
755 }
756 else if (!quoted && c == '-' && i+1 < source.length && source[i+1] == '-') {
757 comment = true;
758 }
759 else if (!quoted && c == ';') {
760 try {
761 final s = Parser(tokenize(source.substring(start,i+1))).parse();
762 if (s.command == 'EXIT') break;
763 execute(s);
764 } on SqlError catch (e) {
765 print('ERROR: ${e.reason}');
766 }
767 start = i+1;
768 }
769 }
