38 if (b.kind == Cell::Null) return 1;

58vector<Token>tokenize(const string&s) {
59 static const unordered_set<string>reserved = {
60 "SELECT", "DISTINCT", "FROM", "AS", "JOIN", "INNER", "LEFT", "OUTER", "ON", "WHERE", "GROUP", "BY", "HAVING", "ORDER", "ASC", "DESC", "LIMIT", "OFFSET", "INSERT", "INTO", "VALUES", "UPDATE", "SET", "DELETE", "CREATE", "TABLE", "DROP", "PRIMARY", "KEY", "NOT", "NULL", "INT", "TEXT", "AND", "OR", "IS", "IN", "BETWEEN", "LIKE", "COUNT", "SUM", "MIN", "MAX", "AVG", "EXIT"
61 };
62 vector<Token>a;
63 size_t i = 0;
64 while (i<s.size()) {
65 char c = s[i];
66 size_t start = i;
67 if (c == ' ' || c == '\n') {
68 i++;
69 continue;
70 }
71 if (s.compare(i, 2, "--") == 0) {
72 i = s.find('\n', i);
73 if (i == string::npos) break;
74 continue;
75 }
76 if (c == '\'') {
77 i = s.find('\'', i+1)+1;
78 a.push_back({
79 "string", s.substr(start, i-start), Cell(s.substr(start+1, i-start-2))
80 });
81 }
82 else if (isdigit(static_cast<unsigned char>(c))) {
83 while (i<s.size() && isdigit(static_cast<unsigned char>(s[i]))) i++;
84 string t = s.substr(start, i-start);
85 a.push_back({
86 "integer", t, Cell(stoll(t))
87 });
88 }
89 else if (isalpha(static_cast<unsigned char>(c)) || c == '_') {
90 while (i<s.size() && (isalnum(static_cast<unsigned char>(s[i])) || s[i] == '_')) i++;
91 string t = s.substr(start, i-start), up = t;
92 for (char&ch:up) ch = static_cast<char>(toupper(static_cast<unsigned char>(ch)));
93 bool keyword = reserved.count(up);
94 a.push_back({
95 keyword?"keyword":"name", t, Cell(keyword?up:t)
96 });
97 }
98 else {
99 string t = s.substr(i, 2);
100 if (t != "<>" && t != "!=" && t != "<=" && t != ">=" && t != "||") t = string(1, c);
101 if (t.size() == 1 && string("(),.;*/%+-=<>").find(c) == string::npos) fail(string("unexpected character '")+c+"'");
102 a.push_back({
103 "symbol", t, Cell(t)
104 });
105 i += t.size();
106 }
107 }
108 return a;
109}

205 template<class F>auto comma(F read) {

219 string alias() {
220 if (accept("AS")) return name();
221 return peek().kind == "name"?name():"";

229 E reference() {
230 string n = name();
231 E e = node("ref");
232 if (accept(".")) {
233 e->table = n;
234 e->column = name();
235 e->text = n+"."+e->column;
236 } else e->column = e->text = n;
237 return e;
238 }
239 E expr(int level = 0) {
240 if (level == 2 && accept("NOT")) return node("unary", "NOT", {
241 expr(2)
242 });
243 if (level == 6 && accept("-")) return node("unary", "-", {
244 expr(6)
245 });
246 if (level == 7) return primary();
247 E left = expr(level+1);
248 if (level == 3) {
249 if (peek().kind == "symbol" && comparisons.count(peek().value.text)) {
250 string op = take().value.text;
251 return node("binary", op, {
252 left, expr(4)
253 });
254 }
255 if (accept("IS")) {
256 bool neg = accept("NOT");
257 expect("NULL");
258 E e = node("isnull", "", {
259 left
260 });
261 e->negated = neg;
262 return e;
263 }
264 bool neg = accept("NOT");
265 E e;
266 if (accept("IN")) {
267 vector<E>a {
268 left
269 }, b = parens([&] {
270 return expr();
271 });
272 a.insert(a.end(), b.begin(), b.end());
273 e = node("in", "", a);
274 } else if (accept("BETWEEN")) {
275 E low = expr(4);
276 expect("AND");
277 e = node("between", "", {
278 left, low, expr(4)
279 });
280 } else if (accept("LIKE")) e = node("like", "", {
281 left, expr(4)
282 });
283 else if (neg) syntax();
284 if (e) {
285 e->negated = neg;
286 return e;
287 }
288 return left;
289 }
290 static const vector<vector<string>>ops = {
291 {
292 "OR"
293 }, {
294 "AND"
295 }, {
296 }, {
297 }, {
298 "+", "-", "||"
299 }, {
300 "*", "/", "%"
301 }, {
302 }
303 };
304 while (any_of(ops[level].begin(), ops[level].end(), [&](const string&s) {
305 return at(s);
306 })) {
307 string op = take().value.text;
308 left = node("binary", op, {
309 left, expr(level+1)
310 });
311 }
312 return left;
313 }
314 E primary() {
315 Token t = peek();
316 if (t.kind == "integer" || t.kind == "string") {
317 take();
318 return node("literal", "", {
319 }, t.value);
320 }
321 if (accept("NULL")) return node("literal");
322 if (t.kind == "name") return reference();
323 if (t.kind == "keyword" && aggregates.count(t.value.text)) {
324 string n = take().value.text;
325 expect("(");
326 E e = node("aggregate", n);
327 if (!(n == "COUNT" && accept("*"))) {
328 e->distinct = accept("DISTINCT");
329 e->args.push_back(expr());
330 }
331 expect(")");
332 return e;
333 }
334 if (accept("(")) {
335 E e = expr();
336 expect(")");
337 return node("group", "", {
338 e
339 });
340 }
341 syntax();
342 }
343 E clause(const string&w) {
344 return accept(w)?expr():nullptr;
345 }
346 Item item() {
347 size_t start = pos;
348 E e = expr();
349 size_t stop = pos;
350 string a = alias(), h = a;
351 if (h.empty()) {
352 if (e->kind == "ref") h = e->column;
353 else for (size_t i = start; i<stop; i++) {
354 if (i>start && ts[i-1].kind != "symbol" && ts[i].kind != "symbol") h += ' ';
355 h += ts[i].text;
356 }
357 }
358 return {
359 e, a, h
360 };
361 }
362 Order order_item() {
363 size_t start = pos;
364 E e = expr();
365 bool single = pos == start+1, desc = accept("DESC");
366 if (!desc) accept("ASC");
367 Order o;
368 o.expr = e;
369 o.descending = desc;
370 if (single && ts[start].kind == "integer") o.ordinal = ts[start].value.number;
371 if (single && ts[start].kind == "name") o.alias = ts[start].value.text;
372 return o;
373 }
374 Query query() {
375 Query q;
376 q.distinct = accept("DISTINCT");
377 q.star = accept("*");
378 if (!q.star) q.items = comma([&] {
379 return item();
380 });
381 expect("FROM");
382 q.source = table();
383 while (at("JOIN") || at("INNER") || at("LEFT")) {
384 bool left = accept("LEFT");
385 if (left) accept("OUTER");
386 else accept("INNER");
387 expect("JOIN");
388 Ref t = table();
389 expect("ON");
390 q.joins.push_back({
391 t, left, expr()
392 });
393 }
394 q.where = clause("WHERE");
395 if (accept("GROUP")) {
396 q.grouped = true;
397 expect("BY");
398 q.group = comma([&] {
399 return reference();
400 });
401 }
402 q.having = clause("HAVING");
403 if (accept("ORDER")) {
404 expect("BY");
405 q.order = comma([&] {
406 return order_item();
407 });
408 }
409 if (accept("LIMIT")) {
410 q.limit = integer();
411 if (accept("OFFSET")) q.offset = integer();
412 }
413 return q;
414 }
415 Column column() {
416 Column c;
417 c.name = name();
418 if (!at("INT") && !at("TEXT")) syntax();
419 c.type = take().value.text;
420 while (true) {
421 if (accept("PRIMARY")) {
422 expect("KEY");
423 c.primary++;
424 c.not_null = true;
425 } else if (accept("NOT")) {
426 expect("NULL");
427 c.not_null = true;
428 } else break;
429 }
430 return c;
431 }

434 if (accept("CREATE")) {
435 c.kind = "create";
436 expect("TABLE");
437 c.name = name();

440 });
441 } else if (accept("DROP")) {

443 expect("TABLE");
444 c.name = name();
445 } else if (accept("INSERT")) {
446 c.kind = "insert";
447 expect("INTO");
448 c.name = name();

450 c.named_targets = true;
451 c.names = parens([&] {
452 return name();
453 });
454 }
455 expect("VALUES");
456 c.tuples = comma([&] {
457 return parens([&] {
458 return expr();
459 });
460 });
461 } else if (accept("UPDATE")) {
462 c.kind = "update";
463 c.name = name();
464 expect("SET");
465 c.assignments = comma([&] {

467 expect("=");

479 c.query = query();
480 } else if (accept("EXIT")) c.kind = "exit";

510 string resolve(E e) {

512 bool seen = false;
513 for (const Entry&part:entries) if (e->table.empty() || e->table == part.name) {
514 seen = true;
515 int i = part.table->column(e->column);
516 if (i >= 0) found.emplace_back(part.offset+i, part.table->columns[i].type);

519 if (found.empty()) fail("unknown column "+e->text);
520 if (found.size()>1) fail("ambiguous column "+e->column);
521 e->index = found[0].first;

531string check(E e, Scope&s, bool allow = false, const vector<int>*grouped = nullptr, bool inside = false) {
532 if (e->kind == "literal") return e->value.kind == Cell::Null?"NULL":e->value.kind == Cell::Int?"INT":"TEXT";
533 if (e->kind == "ref") {
534 string t = s.resolve(e);
535 if (grouped && !inside && find(grouped->begin(), grouped->end(), e->index) == grouped->end()) fail("column "+e->text+" is not grouped");
536 return t;
537 }
538 if (e->kind == "aggregate") {
539 if (!allow || inside) fail("misuse of aggregate "+e->op);
540 if (e->args.empty()) return "INT";
541 string t = check(e->args[0], s, allow, grouped, true);
542 if (e->op == "SUM" || e->op == "AVG") require_type(t, "INT", e->op);
543 return e->op == "MIN" || e->op == "MAX"?t:"INT";
544 }
545 vector<string>types;
546 for (E a:e->args) types.push_back(check(a, s, allow, grouped, inside));
547 if (e->kind == "group") return types[0];
548 if (e->kind == "isnull") return "INT";
549 if (e->kind == "binary" && comparisons.count(e->op)) {
550 require_type(types[0], types[1], e->op);
551 return "INT";
552 }
553 if (e->kind == "in" || e->kind == "between") {
554 for (size_t i = 1; i<types.size(); i++) require_type(types[i], types[0], e->kind == "in"?"IN":"BETWEEN");
555 return "INT";
556 }
557 string op = e->kind == "like"?"LIKE":e->op, t = op == "LIKE" || op == "||"?"TEXT":"INT";
558 for (const string&a:types) require_type(a, t, op);
559 return op == "||"?"TEXT":"INT";
560}
561void condition(E e, Scope&s, const string&name, bool allow = false, const vector<int>*grouped = nullptr) {
562 if (e) require_type(check(e, s, allow, grouped), "INT", name);

569Cell operate(const string&op, const Cell&a, const Cell&b) {
570 if (op == "AND") {
571 if ((a.kind == Cell::Int && a.number == 0) || (b.kind == Cell::Int && b.number == 0)) return Cell(0LL);
572 return a.kind == Cell::Null || b.kind == Cell::Null?Cell():Cell(1LL);
573 }
574 if (op == "OR") {
575 if (truth(a) || truth(b)) return Cell(1LL);
576 return a.kind == Cell::Null || b.kind == Cell::Null?Cell():Cell(0LL);
577 }
578 if (a.kind == Cell::Null || b.kind == Cell::Null) return Cell();
579 long long x = a.number, y = b.number;
580 if (op == "+") return Cell(x+y);
581 if (op == "-") return Cell(x-y);
582 if (op == "*") return Cell(x*y);
583 if (op == "/") return y == 0?Cell():Cell(x/y);
584 if (op == "%") return y == 0?Cell():Cell(x%y);
585 if (op == "||") return Cell(a.text+b.text);
586 int c = compare(a, b);
587 return Cell((op == "="?c == 0:op == "<>" || op == "!="?c != 0:op == "<"?c<0:op == "<="?c <= 0:op == ">"?c>0:c >= 0)?1LL:0LL);
588}
589bool like(const string&text, const string&pattern) {
590 vector<bool>dp(text.size()+1);
591 dp[0] = true;
592 for (char c:pattern) {
593 if (c == '%') {
594 for (size_t i = 1; i <= text.size(); i++) dp[i] = dp[i] || dp[i-1];
595 } else {
596 for (size_t i = text.size(); i>0; i--) dp[i] = dp[i-1] && (c == '_' || c == text[i-1]);
597 dp[0] = false;
598 }
599 }
600 return dp.back();
601}
602Cell evaluate(E e, const Row&row, const Rows*group = nullptr) {
603 if (e->kind == "literal") return e->value;
604 if (e->kind == "ref") return row[e->index];
605 if (e->kind == "aggregate") {
606 if (e->args.empty()) return Cell(static_cast<long long>(group->size()));
607 vector<Cell>vs;
608 for (const Row&r:*group) {
609 Cell v = evaluate(e->args[0], r);
610 if (v.kind != Cell::Null && (!e->distinct || find(vs.begin(), vs.end(), v) == vs.end())) vs.push_back(v);
611 }
612 if (e->op == "COUNT") return Cell(static_cast<long long>(vs.size()));
613 if (vs.empty()) return Cell();
614 if (e->op == "MIN" || e->op == "MAX") {
615 Cell v = vs[0];
616 for (const Cell&x:vs) if (e->op == "MIN"?compare(x, v)<0:compare(x, v)>0) v = x;
617 return v;
618 }
619 long long sum = 0;
620 for (const Cell&v:vs) sum += v.number;
621 return Cell(e->op == "AVG"?sum/static_cast<long long>(vs.size()):sum);
622 }
623 vector<Cell>a;
624 for (E arg:e->args) a.push_back(evaluate(arg, row, group));
625 if (e->kind == "group") return a[0];
626 if (e->kind == "unary") return e->op == "NOT"?inverse(a[0]):a[0].kind == Cell::Null?Cell():Cell(-a[0].number);
627 if (e->kind == "binary") return operate(e->op, a[0], a[1]);
628 if (e->kind == "isnull") return Cell((a[0].kind == Cell::Null) != e->negated?1LL:0LL);
629 Cell result;
630 if (e->kind == "in") {
631 if (a[0].kind != Cell::Null) {
632 bool found = false, unknown = false;
633 for (size_t i = 1; i<a.size(); i++) {
634 if (a[i].kind == Cell::Null) unknown = true;
635 else if (a[i] == a[0]) found = true;
636 }
637 result = found?Cell(1LL):unknown?Cell():Cell(0LL);
638 }
639 }
640 else if (e->kind == "between") result = operate("AND", operate(">=", a[0], a[1]), operate("<=", a[0], a[2]));
641 else if (a[0].kind != Cell::Null && a[1].kind != Cell::Null) result = Cell(like(a[0].text, a[1].text)?1LL:0LL);
642 return e->negated?inverse(result):result;
643}
644bool satisfies(E e, const Row&r, const Rows*g = nullptr) {
645 return !e || truth(evaluate(e, r, g));
646}

650void print_table(const vector<Item>&items, const vector<string>&types, const Rows&rows) {
651 vector<size_t>width;
652 for (const Item&i:items) width.push_back(i.heading.size());
653 vector<vector<string>>cells;
654 for (const Row&r:rows) {
655 vector<string>a;
656 for (size_t i = 0; i<r.size(); i++) {
657 a.push_back(r[i].written());
658 width[i] = max(width[i], a.back().size());
659 }
660 cells.push_back(a);
661 }
662 string sep = "+";
663 for (size_t w:width) sep += string(w+2, '-')+"+";
664 auto render = [&](const vector<string>&a, bool header) {
665 string s = "|";
666 for (size_t i = 0; i<a.size(); i++) {
667 string pad(width[i]-a[i].size(), ' ');
668 s += " "+(!header && types[i] == "INT"?pad+a[i]:a[i]+pad)+" |";
669 }
670 cout<<s<<'\n';
671 };
672 vector<string>heads;
673 for (const Item&i:items) heads.push_back(i.heading);
674 cout<<sep<<'\n';
675 render(heads, true);
676 cout<<sep<<'\n';
677 for (const auto&r:cells) render(r, false);
678 cout<<sep<<"\n("<<count_rows(rows.size())<<")\n";
679}

697 void constraints(const Table&t, const Rows&rows, const vector<size_t>&checked) {
698 for (size_t i:checked) {
699 for (size_t j = 0; j<t.columns.size(); j++) if (t.columns[j].not_null && rows[i][j].kind == Cell::Null) fail("NULL value in column "+t.columns[j].name);
700 if (t.key >= 0) for (size_t j = 0; j<rows.size(); j++) if (i != j && rows[i][t.key] == rows[j][t.key]) fail("duplicate primary key "+rows[i][t.key].written());
701 }
702 }
703 void select(Query q) {
704 Scope scope;
705 scope.add(q.source.alias, table(q.source.name));
706 for (const Join&j:q.joins) {
707 scope.add(j.table.alias, table(j.table.name));
708 condition(j.condition, scope, "ON");
709 }
710 condition(q.where, scope, "WHERE");
711 vector<int>grouping;
712 for (E ref:q.group) {
713 scope.resolve(ref);
714 grouping.push_back(ref->index);
715 }
716 if (q.star) for (const Entry&part:scope.entries) for (const Column&c:part.table->columns) {
717 E e = node("ref");
718 e->table = part.name;
719 e->column = e->text = c.name;
720 q.items.push_back({
721 e, "", c.name
722 });
723 }
724 bool aggregate = q.grouped || bool(q.having);
725 for (const Item&i:q.items) aggregate = aggregate || has_aggregate(i.expr);
726 for (const Order&o:q.order) aggregate = aggregate || has_aggregate(o.expr);
727 const vector<int>*grouped = aggregate?&grouping:nullptr;
728 vector<string>types;
729 for (const Item&i:q.items) types.push_back(check(i.expr, scope, true, grouped));
730 condition(q.having, scope, "HAVING", true, grouped);
731 vector<E>keys;
732 for (const Order&o:q.order) {
733 E key;
734 if (o.ordinal >= 0) {
735 if (o.ordinal<1 || o.ordinal>static_cast<long long>(q.items.size())) fail("ORDER BY position "+to_string(o.ordinal)+" is out of range");
736 key = q.items[o.ordinal-1].expr;
737 } else if (!o.alias.empty()) {
738 for (const Item&i:q.items) if (i.alias == o.alias) {
739 key = i.expr;
740 break;
741 }
742 }
743 if (!key) {
744 check(o.expr, scope, true, grouped);
745 key = o.expr;
746 }
747 keys.push_back(key);
748 }
749 Rows rows = scope.entries[0].table->rows;
750 for (size_t ji = 0; ji<q.joins.size(); ji++) {
751 const Join&j = q.joins[ji];
752 const Table&t = *scope.entries[ji+1].table;
753 Rows next;
754 for (const Row&left:rows) {
755 bool found = false;
756 for (const Row&right:t.rows) {
757 Row combined = left;
758 combined.insert(combined.end(), right.begin(), right.end());
759 if (satisfies(j.condition, combined)) {
760 next.push_back(combined);
761 found = true;
762 }
763 }
764 if (j.left && !found) {
765 Row combined = left;
766 combined.resize(left.size()+t.columns.size());
767 next.push_back(combined);
768 }
769 }
770 rows = move(next);
771 }
772 rows.erase(remove_if(rows.begin(), rows.end(), [&](const Row&r) {
773 return !satisfies(q.where, r);
774 }), rows.end());
775 vector<Rows>groups;
776 if (!aggregate) {
777 for (const Row&r:rows) groups.push_back({
778 r
779 });
780 } else if (!q.grouped) groups.push_back(rows);
781 else {
782 vector<Row>groupkeys;
783 for (const Row&r:rows) {
784 Row key;
785 for (int i:grouping) key.push_back(r[i]);
786 auto it = find(groupkeys.begin(), groupkeys.end(), key);
787 size_t at = it-groupkeys.begin();
788 if (it == groupkeys.end()) {
789 groupkeys.push_back(key);
790 groups.push_back({
791 });
792 }
793 groups[at].push_back(r);
794 }
795 }
796 struct Result {
797 Row values, keys;
798 };
799 vector<Result>results;
800 for (const Rows&g:groups) {
801 Row empty;
802 const Row&r = g.empty()?empty:g[0];
803 const Rows*context = aggregate?&g:nullptr;
804 if (!satisfies(q.having, r, context)) continue;
805 Result result;
806 for (const Item&i:q.items) result.values.push_back(evaluate(i.expr, r, context));
807 for (E key:keys) result.keys.push_back(evaluate(key, r, context));
808 results.push_back(move(result));
809 }
810 stable_sort(results.begin(), results.end(), [&](const Result&a, const Result&b) {
811 for (size_t i = 0; i<keys.size(); i++) {
812 int c = compare(a.keys[i], b.keys[i]);
813 if (c) return q.order[i].descending?c>0:c<0;
814 }
815 return false;
816 });
817 Rows unique;
818 for (const Result&r:results) if (!q.distinct || find(unique.begin(), unique.end(), r.values) == unique.end()) unique.push_back(r.values);
819 size_t first = min(static_cast<size_t>(q.offset), unique.size()), last = q.limit<0?unique.size():min(unique.size(), first+static_cast<size_t>(q.limit));
820 Rows output(unique.begin()+first, unique.begin()+last);
821 print_table(q.items, types, output);
822 }
823 void execute(const Command&c) {
824 if (c.kind == "select") {
825 select(c.query);
826 return;
827 }
828 if (c.kind == "create") {
829 if (tables.count(c.name)) fail("table "+c.name+" already exists");
830 unordered_set<string>seen;
831 int primary = 0, key = -1;
832 for (size_t i = 0; i<c.columns.size(); i++) {
833 const Column&column = c.columns[i];
834 if (!seen.insert(column.name).second) fail("duplicate column "+column.name);
835 primary += column.primary;
836 if (primary >= 2) fail("multiple primary keys");
837 if (column.primary) key = static_cast<int>(i);
838 }
839 tables[c.name] = Table {
840 c.name, c.columns, key, {
841 }
842 };
843 cout<<"Table "<<c.name<<" created.\n";
844 return;
845 }
846 Table&t = table(c.name);
847 if (c.kind == "drop") {
848 tables.erase(c.name);
849 cout<<"Table "<<c.name<<" dropped.\n";
850 return;
851 }
852 Rows rows;
853 vector<size_t>checked;
854 size_t count = 0;
855 if (c.kind == "insert") {
856 vector<int>target;
857 if (c.named_targets) target = targets(t, c.names);
858 else for (size_t i = 0; i<t.columns.size(); i++) target.push_back(static_cast<int>(i));
859 Scope empty;
860 for (const auto&tuple:c.tuples) {
861 if (tuple.size() != target.size()) fail("expected "+to_string(target.size())+" values, got "+to_string(tuple.size()));
862 for (size_t i = 0; i<tuple.size(); i++) {
863 const Column&column = t.columns[target[i]];
864 require_type(check(tuple[i], empty), column.type, "column "+column.name);
865 }
866 }
867 rows = t.rows;
868 for (const auto&tuple:c.tuples) {
869 Row row(t.columns.size());
870 for (size_t i = 0; i<tuple.size(); i++) row[target[i]] = evaluate(tuple[i], {
871 });
872 checked.push_back(rows.size());
873 rows.push_back(row);
874 }
875 constraints(t, rows, checked);
876 count = c.tuples.size();
877 } else if (c.kind == "update") {
878 vector<string>names;
879 for (const auto&a:c.assignments) names.push_back(a.first);
880 vector<int>target = targets(t, names);
881 Scope scope;
882 scope.add(t.name, t);
883 for (size_t i = 0; i<c.assignments.size(); i++) {
884 const Column&column = t.columns[target[i]];
885 require_type(check(c.assignments[i].second, scope), column.type, "column "+column.name);
886 }
887 condition(c.where, scope, "WHERE");
888 rows = t.rows;
889 for (size_t i = 0; i<t.rows.size(); i++) if (satisfies(c.where, t.rows[i])) {
890 Row row = t.rows[i];
891 for (size_t j = 0; j<c.assignments.size(); j++) row[target[j]] = evaluate(c.assignments[j].second, t.rows[i]);
892 rows[i] = row;
893 checked.push_back(i);
894 count++;
895 }
896 constraints(t, rows, checked);
897 } else {
898 Scope scope;
899 scope.add(t.name, t);
900 condition(c.where, scope, "WHERE");
901 for (const Row&r:t.rows) if (satisfies(c.where, r)) count++;
902 else rows.push_back(r);
903 }
904 t.rows = move(rows);
905 cout<<count_rows(count)<<(c.kind == "insert"?" inserted.\n":c.kind == "update"?" updated.\n":" deleted.\n");
906 }
907};

912 for (size_t i = 0; i<source.size(); i++) {

921 }
922 }
923 Database db;
924 for (const string&s:statements) {

926 Command c = Parser(tokenize(s)).statement();

929 } catch (const runtime_error&error) {
930 cout<<"ERROR: "<<error.what()<<'\n';
931 }
