[[noreturn]] void fail(const string&s) {
    throw runtime_error(s);
}
struct Cell {
    enum Kind {
        Null, Int, Text
    }
    kind = Null;
    long long number = 0;
    string text;
    Cell() = default;
    Cell(long long n):kind(Int), number(n) {
    }
    Cell(string s):kind(Text), text(move(s)) {
    }
    bool operator == (const Cell&b) const {
        return kind == b.kind && (kind == Null || (kind == Int?number == b.number:text == b.text));
    }
    string written() const {
        return kind == Null?"NULL":kind == Int?to_string(number):text;
    }
};
using Row = vector<Cell>;
using Rows = vector<Row>;
bool truth(const Cell&v) {
    return v.kind != Cell::Null && v.number != 0;
}

struct Table {
    string name;
    vector<Column>columns;
    int key = -1;
    Rows rows;
    int column(const string&n) const {
        for (size_t i = 0; i<columns.size(); i++) if (columns[i].name == n) return static_cast<int>(i);
        return -1;
    }
};
struct Entry {
    string name;
    Table*table;
    int offset;
};
struct Scope {
    vector<Entry>entries;
    int width = 0;
    void add(string name, Table&t) {
        for (const Entry&e:entries) if (e.name == name) fail("duplicate table name "+name);
        entries.push_back({
            name, &t, width
        });
        width += static_cast<int>(t.columns.size());
    }
    string resolve(E e) {
        vector<pair<int, string>>found;
        bool seen = false;
        for (const Entry&part:entries) if (e->table.empty() || e->table == part.name) {
            seen = true;
            int i = part.table->column(e->column);
            if (i >= 0) found.emplace_back(part.offset+i, part.table->columns[i].type);
        }
        if (!e->table.empty() && !seen) fail("unknown table "+e->table);
        if (found.empty()) fail("unknown column "+e->text);
        if (found.size()>1) fail("ambiguous column "+e->column);
        e->index = found[0].first;
        return found[0].second;
    }
};
bool compatible(const string&a, const string&b) {
    return a == b || a == "NULL" || b == "NULL";
}
void require_type(const string&a, const string&b, const string&op) {
    if (!compatible(a, b)) fail("type mismatch for "+op);
}

void condition(E e, Scope&s, const string&name, bool allow = false, const vector<int>*grouped = nullptr) {

Cell evaluate(E e, const Row&row, const Rows*group = nullptr) {
    if (e->kind == "literal") return e->value;

bool satisfies(E e, const Row&r, const Rows*g = nullptr) {
    return !e || truth(evaluate(e, r, g));
}
string count_rows(size_t n) {
    return to_string(n)+(n == 1?" row":" rows");
}
