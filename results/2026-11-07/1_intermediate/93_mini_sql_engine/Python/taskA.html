187def split_statements(source):
188 statements = []
189 start = 0
190 for found in STATEMENT_PATTERN.finditer(source):
191 if found.group() == ';':
192 statements.append(source[start:found.end()])
193 start = found.end()
194 return statements

197def tokenize(text):
198 tokens = []
199 position = 0
200 while position < len(text):
201 found = TOKEN_PATTERN.match(text, position)
202 if found is None:
203 raise SqlError(f"unexpected character '{text[position]}'")
204 position = found.end()
205 written = found.group()
206 match found.lastgroup:
207 case 'string':
208 tokens.append(Token('string', found.group('string'), written))
209 case 'integer':
210 tokens.append(Token('integer', int(written), written))
211 case 'word' if written.upper() in RESERVED_WORDS:
212 tokens.append(Token('keyword', written.upper(), written))
213 case 'word':
214 tokens.append(Token('name', written, written))
215 case 'symbol':
216 tokens.append(Token('symbol', written, written))
217 return tokens

220def written_form(tokens):
221 text = tokens[0].text
222 for previous, token in zip(tokens, tokens[1:]):
223 if previous.kind != 'symbol' and token.kind != 'symbol':
224 text += ' '
225 text += token.text
226 return text

276 def parse_statement(self):
277 statement = None
278 if self.accept('CREATE'):
279 statement = self.parse_create_table()
280 elif self.accept('DROP'):
281 self.expect('TABLE')
282 statement = DropTable(self.expect_kind('name'))
283 elif self.accept('INSERT'):
284 statement = self.parse_insert()
285 elif self.accept('UPDATE'):
286 statement = self.parse_update()
287 elif self.accept('DELETE'):
288 self.expect('FROM')
289 table = self.expect_kind('name')
290 statement = Delete(table, self.parse_clause('WHERE'))
291 elif self.accept('SELECT'):
292 statement = self.parse_select()
293 elif self.accept('EXIT'):
294 statement = Exit()
295 self.expect(';')

298 def parse_create_table(self):
299 self.expect('TABLE')
300 name = self.expect_kind('name')

303 def parse_column_def(self):
304 name = self.expect_kind('name')
305 if not self.at('INT', 'TEXT'):
306 raise self.syntax_error()
307 column_type = self.advance().value
308 primary_keys = 0
309 not_null = False
310 while True:
311 if self.accept('PRIMARY'):
312 self.expect('KEY')
313 primary_keys += 1
314 elif self.accept('NOT'):
315 self.expect('NULL')
316 not_null = True
317 else:
318 return ColumnDef(name, column_type, primary_keys, not_null)

320 def parse_insert(self):

344 def parse_select(self):
345 distinct = self.accept('DISTINCT')
346 items = None if self.accept('*') else self.comma_list(self.parse_select_item)
347 self.expect('FROM')
348 source = self.parse_table_ref()
349 joins = []

351 joins.append(self.parse_join())
352 where = self.parse_clause('WHERE')

355 self.expect('BY')

357 having = self.parse_clause('HAVING')
358 order_by = []

360 self.expect('BY')

363 if self.accept('LIMIT'):
364 limit = self.expect_kind('integer')

366 offset = self.expect_kind('integer')

422 def parse_binary(self, operators, parse_operand):
423 expr = parse_operand()
424 while self.at(*operators):
425 operator = self.advance().value
426 expr = Binary(operator, expr, parse_operand())
427 return expr

429 def parse_expr(self):
430 return self.parse_binary(('OR',), self.parse_and)

432 def parse_and(self):
433 return self.parse_binary(('AND',), self.parse_not)

435 def parse_not(self):
436 if self.accept('NOT'):
437 return Unary('NOT', self.parse_not())
438 return self.parse_comparison()

440 def parse_comparison(self):
441 left = self.parse_additive()
442 if self.at(*COMPARISON_OPERATORS):
443 operator = self.advance().value
444 return Binary(operator, left, self.parse_additive())
445 if self.accept('IS'):
446 negated = self.accept('NOT')
447 self.expect('NULL')
448 return IsNull(left, negated)
449 negated = self.accept('NOT')
450 if self.accept('IN'):
451 return InList(left, self.parenthesized_list(self.parse_expr), negated)
452 if self.accept('BETWEEN'):
453 low = self.parse_additive()
454 self.expect('AND')
455 return Between(left, low, self.parse_additive(), negated)
456 if self.accept('LIKE'):
457 return Like(left, self.parse_additive(), negated)
458 if negated:
459 raise self.syntax_error()
460 return left

462 def parse_additive(self):
463 return self.parse_binary(('+', '-', '||'), self.parse_multiplicative)
464
465 def parse_multiplicative(self):
466 return self.parse_binary(('*', '/', '%'), self.parse_unary)
467
468 def parse_unary(self):
469 if self.accept('-'):
470 return Unary('-', self.parse_unary())
471 return self.parse_primary()

473 def parse_primary(self):
474 token = self.peek()
475 if token.kind in ('integer', 'string'):
476 self.advance()
477 return Literal(token.value)
478 if self.accept('NULL'):
479 return Literal(None)
480 if token.kind == 'name':
481 return self.parse_column_ref()
482 if self.at(*AGGREGATE_NAMES):
483 return self.parse_aggregate()
484 if self.accept('('):
485 expr = self.parse_expr()
486 self.expect(')')
487 return expr
488 raise self.syntax_error()

490 def parse_aggregate(self):
491 name = self.advance().value
492 self.expect('(')
493 if name == 'COUNT' and self.accept('*'):
494 argument = None
495 distinct = False
496 else:
497 distinct = self.accept('DISTINCT')
498 argument = self.parse_expr()
499 self.expect(')')
500 return Aggregate(name, argument, distinct)

532 def __init__(self):
533 self.entries = []

536 def add(self, name, table):
537 if any(entry.name == name for entry in self.entries):
538 raise SqlError(f'duplicate table name {name}')
539 self.entries.append(ScopeEntry(name, table, self.width))

542 def resolve(self, ref):
543 entries = self.entries
544 if ref.table is not None:
545 entries = [entry for entry in entries if entry.name == ref.table]
546 if not entries:
547 raise SqlError(f'unknown table {ref.table}')
548 candidates = []
549 for entry in entries:
550 index = entry.table.column_index(ref.column)
551 if index is not None:
552 candidates.append((entry, index))
553 if not candidates:
554 raise SqlError(f'unknown column {ref.text}')
555 if len(candidates) > 1:
556 raise SqlError(f'ambiguous column {ref.column}')
557 entry, index = candidates[0]
558 ref.index = entry.offset + index
559 return entry.table.columns[index].type

584def check(expr, context, in_aggregate=False):
585 match expr:
586 case Literal(value=None):
587 return 'NULL'
588 case Literal(value=int()):
589 return 'INT'
590 case Literal():
591 return 'TEXT'
592 case ColumnRef():
593 column_type = context.scope.resolve(expr)
594 grouping = context.grouping
595 if grouping is not None and not in_aggregate and expr.index not in grouping:
596 raise SqlError(f'column {expr.text} is not grouped')
597 return column_type
598 case Unary():
599 require(check(expr.operand, context, in_aggregate), 'INT', expr.operator)
600 return 'INT'
601 case Binary():
602 left = check(expr.left, context, in_aggregate)
603 right = check(expr.right, context, in_aggregate)
604 if expr.operator in COMPARISON_OPERATORS:
605 require(left, right, expr.operator)
606 return 'INT'
607 operand_type = 'TEXT' if expr.operator == '||' else 'INT'
608 require(left, operand_type, expr.operator)
609 require(right, operand_type, expr.operator)
610 return operand_type
611 case IsNull():
612 check(expr.operand, context, in_aggregate)
613 return 'INT'
614 case InList():
615 operand_type = check(expr.operand, context, in_aggregate)
616 item_types = [check(item, context, in_aggregate) for item in expr.items]
617 for item_type in item_types:
618 require(item_type, operand_type, 'IN')
619 return 'INT'
620 case Between():
621 operand_type = check(expr.operand, context, in_aggregate)
622 low_type = check(expr.low, context, in_aggregate)
623 high_type = check(expr.high, context, in_aggregate)
624 require(low_type, operand_type, 'BETWEEN')
625 require(high_type, operand_type, 'BETWEEN')
626 return 'INT'
627 case Like():
628 operand_type = check(expr.operand, context, in_aggregate)
629 pattern_type = check(expr.pattern, context, in_aggregate)
630 require(operand_type, 'TEXT', 'LIKE')
631 require(pattern_type, 'TEXT', 'LIKE')
632 return 'INT'
633 case Aggregate():
634 if not context.allow_aggregates or in_aggregate:
635 raise SqlError(f'misuse of aggregate {expr.name}')
636 if expr.argument is None:
637 return 'INT'
638 argument_type = check(expr.argument, context, True)
639 if expr.name in ('SUM', 'AVG'):
640 require(argument_type, 'INT', expr.name)
641 return argument_type if expr.name in ('MIN', 'MAX') else 'INT'

711def divide(left, right):
712 quotient = abs(left) // abs(right)
713 return quotient if (left < 0) == (right < 0) else -quotient

716def apply_operator(operator, left, right):
717 if operator == 'AND':
718 return logical_and(left, right)
719 if operator == 'OR':
720 return logical_or(left, right)
721 if left is None or right is None:
722 return None
723 match operator:
724 case '+' | '||':
725 return left + right
726 case '-':
727 return left - right
728 case '*':
729 return left * right
730 case '/':
731 return None if right == 0 else divide(left, right)
732 case '%':
733 return None if right == 0 else left - right * divide(left, right)
734 case '=':
735 return int(left == right)
736 case '<>' | '!=':
737 return int(left != right)
738 case '<':
739 return int(left < right)
740 case '<=':
741 return int(left <= right)
742 case '>':
743 return int(left > right)
744 case '>=':
745 return int(left >= right)

748def like_match(text, pattern):
749 matched = [True] + [False] * len(text)
750 for symbol in pattern:
751 if symbol == '%':
752 for index in range(1, len(text) + 1):
753 matched[index] = matched[index] or matched[index - 1]
754 else:
755 for index in range(len(text), 0, -1):
756 matched[index] = matched[index - 1] and symbol in ('_', text[index - 1])
757 matched[0] = False
758 return matched[-1]

761def evaluate(expr, row, group=None):
762 match expr:
763 case Literal():
764 return expr.value
765 case ColumnRef():
766 return row[expr.index]
767 case Unary(operator='-'):
768 value = evaluate(expr.operand, row, group)
769 return None if value is None else -value
770 case Unary():
771 return logical_not(evaluate(expr.operand, row, group))
772 case Binary():
773 left = evaluate(expr.left, row, group)
774 right = evaluate(expr.right, row, group)
775 return apply_operator(expr.operator, left, right)
776 case IsNull():
777 is_null = evaluate(expr.operand, row, group) is None
778 return int(is_null != expr.negated)
779 case InList():
780 value = evaluate(expr.operand, row, group)
781 items = [evaluate(item, row, group) for item in expr.items]
782 if value is None:
783 result = None
784 elif value in items:
785 result = 1
786 else:
787 result = None if None in items else 0
788 return negate_if(expr.negated, result)
789 case Between():
790 value = evaluate(expr.operand, row, group)
791 low = apply_operator('>=', value, evaluate(expr.low, row, group))
792 high = apply_operator('<=', value, evaluate(expr.high, row, group))
793 return negate_if(expr.negated, logical_and(low, high))
794 case Like():
795 text = evaluate(expr.operand, row, group)
796 pattern = evaluate(expr.pattern, row, group)
797 if text is None or pattern is None:
798 return None
799 return negate_if(expr.negated, int(like_match(text, pattern)))
800 case Aggregate():
801 return evaluate_aggregate(expr, group)

804def evaluate_aggregate(aggregate, group):
805 if aggregate.argument is None:
806 return len(group)
807 values = [evaluate(aggregate.argument, row) for row in group]
808 values = [value for value in values if value is not None]
809 if aggregate.distinct:
810 values = list(dict.fromkeys(values))
811 if aggregate.name == 'COUNT':
812 return len(values)
813 if not values:
814 return None
815 match aggregate.name:
816 case 'SUM':
817 return sum(values)
818 case 'MIN':
819 return min(values)
820 case 'MAX':
821 return max(values)
822 case 'AVG':
823 return divide(sum(values), len(values))

830def join_rows(rows, table, join):
831 joined = []
832 padding = [None] * len(table.columns)
833 for left in rows:
834 matched = False
835 for right in table.rows:
836 combined = left + right
837 if is_true(evaluate(join.condition, combined)):
838 joined.append(combined)
839 matched = True
840 if join.left_outer and not matched:
841 joined.append(left + padding)

853 if first is None or second is None:
854 return (first is not None) - (second is not None)
855 return (first > second) - (first < second)

858def compare_keys(first, second, descending):
859 for first_key, second_key, reverse in zip(first, second, descending):
860 result = compare_values(first_key, second_key)
861 if result != 0:
862 return -result if reverse else result
863 return 0

896def column_indices(table, names):
897 indices = []
898 for name in names:
899 index = table.column_index(name)
900 if index is None:
901 raise SqlError(f'unknown column {name}')
902 if index in indices:
903 raise SqlError(f'duplicate column {name}')
904 indices.append(index)
905 return indices

908def check_constraints(table, rows, checked):
909 key = table.primary_key
910 key_counts = Counter(row[key] for row in rows) if key is not None else Counter()
911 for position in checked:
912 row = rows[position]
913 for column, value in zip(table.columns, row):
914 if column.not_null and value is None:
915 raise SqlError(f'NULL value in column {column.name}')
916 if key is not None and key_counts[row[key]] > 1:
917 raise SqlError(f'duplicate primary key {format_value(row[key])}')

922 self.tables = {}

929 def execute(self, statement):
930 match statement:
931 case CreateTable():
932 self.create_table(statement)
933 case DropTable():
934 self.table(statement.name)
935 del self.tables[statement.name]
936 print(f'Table {statement.name} dropped.')
937 case Insert():
938 self.insert(statement)
939 case Update():
940 self.update(statement)
941 case Delete():
942 self.delete(statement)
943 case Select():
944 self.select(statement)

946 def create_table(self, statement):
947 if statement.name in self.tables:
948 raise SqlError(f'table {statement.name} already exists')
949 columns = []
950 primary_key = None
951 primary_key_count = 0
952 for definition in statement.columns:
953 if any(column.name == definition.name for column in columns):
954 raise SqlError(f'duplicate column {definition.name}')
955 primary_key_count += definition.primary_keys
956 if primary_key_count >= 2:
957 raise SqlError('multiple primary keys')
958 if definition.primary_keys:
959 primary_key = len(columns)
960 not_null = definition.not_null or definition.primary_keys > 0
961 columns.append(Column(definition.name, definition.type, not_null))
962 self.tables[statement.name] = Table(statement.name, columns, primary_key)
963 print(f'Table {statement.name} created.')

965 def insert(self, statement):
966 table = self.table(statement.table)
967 if statement.columns is None:
968 targets = list(range(len(table.columns)))
969 else:
970 targets = column_indices(table, statement.columns)
971 context = CheckContext(Scope())
972 for values in statement.tuples:
973 if len(values) != len(targets):
974 raise SqlError(f'expected {len(targets)} values, got {len(values)}')
975 for expr, index in zip(values, targets):
976 column = table.columns[index]
977 require(check(expr, context), column.type, f'column {column.name}')
978 new_rows = []
979 for values in statement.tuples:
980 row = [None] * len(table.columns)
981 for expr, index in zip(values, targets):
982 row[index] = evaluate(expr, [])
983 new_rows.append(row)
984 rows = table.rows + new_rows
985 check_constraints(table, rows, range(len(table.rows), len(rows)))
986 table.rows = rows
987 print(f'{count_rows(len(new_rows))} inserted.')

989 def update(self, statement):
990 table = self.table(statement.table)
991 targets = column_indices(table, [column for column, _ in statement.assignments])
992 context = CheckContext(table_scope(table))
993 for (_, expr), index in zip(statement.assignments, targets):
994 column = table.columns[index]
995 require(check(expr, context), column.type, f'column {column.name}')
996 check_condition(statement.where, context, 'WHERE')
997 rows = list(table.rows)
998 updated = []
999 for position, row in enumerate(table.rows):
1000 if satisfies(statement.where, row):
1001 new_row = list(row)
1002 for (_, expr), index in zip(statement.assignments, targets):
1003 new_row[index] = evaluate(expr, row)
1004 rows[position] = new_row
1005 updated.append(position)
1006 check_constraints(table, rows, updated)
1007 table.rows = rows
1008 print(f'{count_rows(len(updated))} updated.')

1010 def delete(self, statement):
1011 table = self.table(statement.table)
1012 check_condition(statement.where, CheckContext(table_scope(table)), 'WHERE')
1013 kept = [row for row in table.rows if not satisfies(statement.where, row)]
1014 deleted = len(table.rows) - len(kept)
1015 table.rows = kept
1016 print(f'{count_rows(deleted)} deleted.')

1018 def select(self, query):
1019 scope = Scope()
1020 scope.add(query.source.scope_name, self.table(query.source.name))
1021 for join in query.joins:
1022 scope.add(join.table.scope_name, self.table(join.table.name))
1023 check_condition(join.condition, CheckContext(scope), 'ON')
1024 check_condition(query.where, CheckContext(scope), 'WHERE')
1025 for ref in query.group_by:
1026 scope.resolve(ref)
1027 items = query.items
1028 if items is None:
1029 items = [SelectItem(ColumnRef(entry.name, column.name, column.name), None, column.name)
1030 for entry in scope.entries for column in entry.table.columns]
1031 aggregated = (bool(query.group_by) or query.having is not None
1032 or any(contains_aggregate(item.expr) for item in items + query.order_by))
1033 grouping = [ref.index for ref in query.group_by]
1034 context = CheckContext(scope, True, grouping if aggregated else None)
1035 types = [check(item.expr, context) for item in items]
1036 check_condition(query.having, context, 'HAVING')
1037 sort_keys = [resolve_order_key(item, items, context) for item in query.order_by]
1038
1039 rows = scope.entries[0].table.rows
1040 for join, entry in zip(query.joins, scope.entries[1:]):
1041 rows = join_rows(rows, entry.table, join)
1042 rows = [row for row in rows if satisfies(query.where, row)]
1043 if aggregated:
1044 groups = group_rows(rows, grouping) if query.group_by else [rows]
1045 sources = [(group[0] if group else None, group) for group in groups]
1046 sources = [(row, group) for row, group in sources
1047 if satisfies(query.having, row, group)]
1048 else:
1049 sources = [(row, None) for row in rows]
1050 results = [ResultRow([evaluate(item.expr, row, group) for item in items],
1051 [evaluate(key, row, group) for key in sort_keys])
1052 for row, group in sources]
1053 if query.order_by:
1054 descending = [item.descending for item in query.order_by]
1055 results.sort(key=cmp_to_key(
1056 lambda first, second: compare_keys(first.keys, second.keys, descending)))
1057 if query.distinct:
1058 unique = {}
1059 for result in results:
1060 unique.setdefault(tuple(result.values), result)
1061 results = list(unique.values())
1062 if query.offset is not None:
1063 results = results[query.offset:]
1064 if query.limit is not None:
1065 results = results[:query.limit]
1066 headings = [item.heading for item in items]
1067 print_table(headings, types, [result.values for result in results])
