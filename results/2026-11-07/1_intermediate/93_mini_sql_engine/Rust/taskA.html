1use std::collections::{BTreeMap, BTreeSet};
2use std::io::{self, Read};
3#[derive(Clone, PartialEq, Eq, PartialOrd, Ord)]
4enum Value {
5 Null,
6 Int(i64),
7 Text(String),
8}
9impl Value {
10 fn text(&self) -> String {
11 match self {
12 Self::Null => "NULL".into(),
13 Self::Int(n) => n.to_string(),
14 Self::Text(s) => s.clone(),
15 }
16 }
17 fn truth(&self) -> Option<bool> {
18 match self {
19 Self::Int(n) => Some(*n != 0),
20 _ => None,
21 }
22 }
23 fn boolean(v: Option<bool>) -> Self {
24 v.map_or(Self::Null, |v| Self::Int(v as i64))
25 }
26 fn ty(&self) -> u8 {
27 match self {
28 Self::Null => 0,
29 Self::Int(_) => 1,
30 Self::Text(_) => 2,
31 }
32 }
33}
34#[derive(Clone)]
35struct Token {
36 raw: String,
37 text: String,
38 kind: u8,
39}
40fn lex(s: &str) -> Result<Vec<Token>, String> {
41 let b = s.as_bytes();
42 let mut i = 0;
43 let mut out = Vec::new();
44 let reserved = "SELECT DISTINCT FROM AS JOIN INNER LEFT OUTER ON WHERE GROUP BY HAVING ORDER ASC DESC LIMIT OFFSET INSERT INTO VALUES UPDATE SET DELETE CREATE TABLE DROP PRIMARY KEY NOT NULL INT TEXT AND OR IS IN BETWEEN LIKE COUNT SUM MIN MAX AVG EXIT";
45 while i < b.len() {
46 if b[i].is_ascii_whitespace() {
47 i += 1;
48 continue;
49 }
50 let start = i;
51 let kind;
52 if b[i] == b'\'' {
53 i += 1;
54 while i < b.len() && b[i] != b'\'' {
55 i += 1
56 }
57 i += 1;
58 kind = 3
59 } else if b[i].is_ascii_digit() {
60 i += 1;
61 while i < b.len() && b[i].is_ascii_digit() {
62 i += 1
63 }
64 kind = 2
65 } else if b[i].is_ascii_alphabetic() || b[i] == b'_' {
66 i += 1;
67 while i < b.len() && (b[i].is_ascii_alphanumeric() || b[i] == b'_') {
68 i += 1
69 }
70 kind = if reserved
71 .split_whitespace()
72 .any(|w| w == s[start..i].to_ascii_uppercase())
73 {
74 1
75 } else {
76 0
77 }
78 } else {
79 if i + 1 < b.len() && ["<>", "!=", "<=", ">=", "||"].contains(&&s[i..i + 2]) {
80 i += 2
81 } else if b"(),.;*/%+-=<>".contains(&b[i]) {
82 i += 1
83 } else {
84 return Err(format!("unexpected character '{}'", b[i] as char));
85 }
86 kind = 4
87 }
88 let raw = s[start..i].to_string();
89 let text = if kind == 1 {
90 raw.to_ascii_uppercase()
91 } else if kind == 3 {
92 raw[1..raw.len() - 1].into()
93 } else {
94 raw.clone()
95 };
96 out.push(Token { raw, text, kind })
97 }
98 Ok(out)
99}
100#[derive(Clone)]
101enum Expr {
102 Literal(Value),
103 Column(Option<String>, String),
104 Op(String, Vec<Expr>),
105 Aggregate(String, bool, Option<Box<Expr>>),

107}
108impl Expr {
109 fn aggregate(&self) -> bool {
110 match self {
111 Self::Aggregate(..) => true,
112 Self::Op(_, v) => v.iter().any(Self::aggregate),
113 Self::Paren(e) => e.aggregate(),
114 _ => false,
115 }
116 }
117}

162impl Parser {
163 fn peek(&self) -> &str {
164 self.t.get(self.i).map_or("", |t| {
165 if t.kind == 3 {
166 t.raw.as_str()
167 } else {
168 t.text.as_str()
169 }
170 })
171 }
172 fn eat(&mut self, s: &str) -> bool {
173 if self.peek() == s {
174 self.i += 1;
175 true
176 } else {
177 false
178 }
179 }
180 fn error(&self) -> String {
181 format!(
182 "syntax error at '{}'",
183 self.t.get(self.i).map_or(";", |t| t.raw.as_str())
184 )
185 }
186 fn expect(&mut self, s: &str) -> Result<(), String> {
187 if self.eat(s) {
188 Ok(())
189 } else {
190 Err(self.error())
191 }
192 }
193 fn name(&mut self) -> Result<String, String> {
194 if self.t.get(self.i).is_some_and(|t| t.kind == 0) {
195 let s = self.t[self.i].text.clone();
196 self.i += 1;
197 Ok(s)
198 } else {
199 Err(self.error())
200 }
201 }
202 fn integer(&mut self) -> Result<usize, String> {
203 if self.t.get(self.i).is_some_and(|t| t.kind == 2) {
204 let s = self.t[self.i].text.parse().unwrap();
205 self.i += 1;
206 Ok(s)
207 } else {
208 Err(self.error())
209 }
210 }
211 fn expr(&mut self) -> Result<Expr, String> {
212 self.binary(0)
213 }
214 fn binary(&mut self, level: usize) -> Result<Expr, String> {
215 if level == 2 {
216 if self.eat("NOT") {
217 return Ok(Expr::Op("NOT".into(), vec![self.binary(2)?]));
218 }
219 return self.binary(3);
220 }
221 if level == 6 {
222 if self.eat("-") {
223 return Ok(Expr::Op("NEG".into(), vec![self.binary(6)?]));
224 }
225 return self.primary();
226 }
227 let mut left = self.binary(level + 1)?;
228 if level == 3 {
229 let op = self.peek().to_string();
230 if ["=", "!=", "<>", "<", ">", "<=", ">="].contains(&op.as_str()) {
231 self.i += 1;
232 return Ok(Expr::Op(op, vec![left, self.binary(4)?]));
233 }
234 if self.eat("IS") {
235 let neg = self.eat("NOT");
236 self.expect("NULL")?;
237 return Ok(Expr::Op(
238 if neg { "IS NOT NULL" } else { "IS NULL" }.into(),
239 vec![left],
240 ));
241 }
242 let neg = self.eat("NOT");
243 let result = if self.eat("IN") {
244 self.expect("(")?;
245 let mut args = vec![left];
246 args.push(self.expr()?);
247 while self.eat(",") {
248 args.push(self.expr()?)
249 }
250 self.expect(")")?;
251 Some(Expr::Op("IN".into(), args))
252 } else if self.eat("BETWEEN") {
253 let low = self.binary(4)?;
254 self.expect("AND")?;
255 let high = self.binary(4)?;
256 Some(Expr::Op("BETWEEN".into(), vec![left, low, high]))
257 } else if self.eat("LIKE") {
258 Some(Expr::Op("LIKE".into(), vec![left, self.binary(4)?]))
259 } else {
260 if neg {
261 return Err(self.error());
262 }
263 return Ok(left);
264 };
265 let result = result.unwrap();
266 return Ok(if neg {
267 Expr::Op("NOT".into(), vec![result])
268 } else {
269 result
270 });
271 }
272 loop {
273 let op = self.peek().to_string();
274 let matches = match level {
275 0 => op == "OR",
276 1 => op == "AND",
277 4 => ["+", "-", "||"].contains(&op.as_str()),
278 5 => ["*", "/", "%"].contains(&op.as_str()),
279 _ => false,
280 };
281 if !matches {
282 break;
283 }
284 self.i += 1;
285 left = Expr::Op(op, vec![left, self.binary(level + 1)?])
286 }
287 Ok(left)
288 }
289 fn column(&mut self) -> Result<Expr, String> {
290 let s = self.name()?;
291 if self.eat(".") {
292 Ok(Expr::Column(Some(s), self.name()?))
293 } else {
294 Ok(Expr::Column(None, s))
295 }
296 }
297 fn primary(&mut self) -> Result<Expr, String> {
298 let Some(t) = self.t.get(self.i).cloned() else {
299 return Err(self.error());
300 };
301 if t.kind == 2 {
302 self.i += 1;
303 return Ok(Expr::Literal(Value::Int(t.text.parse().unwrap())));
304 }
305 if t.kind == 3 {
306 self.i += 1;
307 return Ok(Expr::Literal(Value::Text(t.text)));
308 }
309 if self.eat("NULL") {
310 return Ok(Expr::Literal(Value::Null));
311 }
312 if self.eat("(") {
313 let e = self.expr()?;
314 self.expect(")")?;
315 return Ok(Expr::Paren(Box::new(e)));
316 }
317 if ["COUNT", "SUM", "MIN", "MAX", "AVG"].contains(&t.text.as_str()) {
318 self.i += 1;
319 self.expect("(")?;
320 if t.text == "COUNT" && self.eat("*") {
321 self.expect(")")?;
322 return Ok(Expr::Aggregate(t.text, false, None));
323 }
324 let distinct = self.eat("DISTINCT");
325 let e = self.expr()?;
326 self.expect(")")?;
327 return Ok(Expr::Aggregate(t.text, distinct, Some(Box::new(e))));
328 }
329 self.column()
330 }
331 fn optional_where(&mut self) -> Result<Option<Expr>, String> {
332 if self.eat("WHERE") {
333 Ok(Some(self.expr()?))
334 } else {
335 Ok(None)
336 }
337 }
338 fn table(&mut self) -> Result<(String, String), String> {
339 let n = self.name()?;
340 let alias = if self.eat("AS") || self.t.get(self.i).is_some_and(|t| t.kind == 0) {
341 self.name()?
342 } else {
343 n.clone()
344 };
345 Ok((n, alias))
346 }
347 fn parse(&mut self) -> Result<Command, String> {
348 let command = if self.peek() == ";" {
349 Command::Empty
350 } else if self.eat("EXIT") {
351 Command::Exit
352 } else if self.eat("CREATE") {
353 self.expect("TABLE")?;
354 let n = self.name()?;
355 self.expect("(")?;
356 let mut cols = Vec::new();
357 loop {
358 let name = self.name()?;
359 let ty = if self.eat("INT") {
360 1
361 } else {
362 self.expect("TEXT")?;
363 2
364 };
365 let mut not_null = false;
366 let mut pk = 0;
367 loop {
368 if self.eat("PRIMARY") {
369 self.expect("KEY")?;
370 pk += 1;
371 not_null = true
372 } else if self.eat("NOT") {
373 self.expect("NULL")?;
374 not_null = true
375 } else {
376 break;
377 }
378 }
379 cols.push(Column {
380 name,
381 ty,
382 not_null,
383 pk,
384 });
385 if !self.eat(",") {
386 break;
387 }
388 }
389 self.expect(")")?;
390 Command::Create(n, cols)
391 } else if self.eat("DROP") {
392 self.expect("TABLE")?;
393 Command::Drop(self.name()?)
394 } else if self.eat("INSERT") {
395 self.expect("INTO")?;
396 let n = self.name()?;
397 let columns = if self.eat("(") {
398 let mut c = vec![self.name()?];
399 while self.eat(",") {
400 c.push(self.name()?)
401 }
402 self.expect(")")?;
403 Some(c)
404 } else {
405 None
406 };
407 self.expect("VALUES")?;
408 let mut tuples = Vec::new();
409 loop {
410 self.expect("(")?;
411 let mut es = vec![self.expr()?];
412 while self.eat(",") {
413 es.push(self.expr()?)
414 }
415 self.expect(")")?;
416 tuples.push(es);
417 if !self.eat(",") {
418 break;
419 }
420 }
421 Command::Insert(n, columns, tuples)
422 } else if self.eat("UPDATE") {
423 let n = self.name()?;
424 self.expect("SET")?;
425 let mut assignments = Vec::new();
426 loop {
427 let c = self.name()?;
428 self.expect("=")?;
429 assignments.push((c, self.expr()?));
430 if !self.eat(",") {
431 break;
432 }
433 }
434 let filter = self.optional_where()?;
435 Command::Update(n, assignments, filter)
436 } else if self.eat("DELETE") {
437 self.expect("FROM")?;
438 let n = self.name()?;
439 let filter = self.optional_where()?;
440 Command::Delete(n, filter)
441 } else if self.eat("SELECT") {
442 let distinct = self.eat("DISTINCT");
443 let items = if self.eat("*") {
444 None
445 } else {
446 let mut items = Vec::new();
447 loop {
448 let start = self.i;
449 let expr = self.expr()?;
450 let end = self.i;
451 let alias = if self.eat("AS") || self.t.get(self.i).is_some_and(|t| t.kind == 0)
452 {
453 Some(self.name()?)
454 } else {
455 None
456 };
457 let heading = if let Some(a) = &alias {
458 a.clone()
459 } else if let Expr::Column(_, c) = &expr {
460 c.clone()
461 } else {
462 let mut h = String::new();
463 for j in start..end {
464 if j > start && self.t[j - 1].kind != 4 && self.t[j].kind != 4 {
465 h.push(' ')
466 }
467 h.push_str(&self.t[j].raw)
468 }
469 h
470 };
471 items.push(Item {
472 expr,
473 alias,
474 heading,
475 });
476 if !self.eat(",") {
477 break;
478 }
479 }
480 Some(items)
481 };
482 self.expect("FROM")?;
483 let from = self.table()?;
484 let mut joins = Vec::new();
485 while ["JOIN", "INNER", "LEFT"].contains(&self.peek()) {
486 let left = if self.eat("LEFT") {
487 self.eat("OUTER");
488 true
489 } else {
490 self.eat("INNER");
491 false
492 };
493 self.expect("JOIN")?;
494 let t = self.table()?;
495 self.expect("ON")?;
496 joins.push((left, t, self.expr()?));
497 }
498 let filter = self.optional_where()?;
499 let groups = if self.eat("GROUP") {
500 self.expect("BY")?;
501 let mut g = vec![self.column()?];
502 while self.eat(",") {
503 g.push(self.column()?)
504 }
505 Some(g)
506 } else {
507 None
508 };
509 let having = if self.eat("HAVING") {
510 Some(self.expr()?)
511 } else {
512 None
513 };
514 let mut order = Vec::new();
515 if self.eat("ORDER") {
516 self.expect("BY")?;
517 loop {
518 let e = self.expr()?;
519 let desc = if self.eat("DESC") {
520 true
521 } else {
522 self.eat("ASC");
523 false
524 };
525 order.push((e, desc));
526 if !self.eat(",") {
527 break;
528 }
529 }
530 }
531 let mut offset = 0;
532 let limit = if self.eat("LIMIT") {
533 let n = self.integer()?;
534 if self.eat("OFFSET") {
535 offset = self.integer()?
536 }
537 Some(n)
538 } else {
539 None
540 };
541 Command::Select(Query {
542 distinct,
543 items,
544 from,
545 joins,
546 filter,
547 groups,
548 having,
549 order,
550 limit,
551 offset,
552 })
553 } else {
554 return Err(self.error());
555 };
556 self.expect(";")?;
557 if self.i != self.t.len() {
558 return Err(self.error());
559 }
560 Ok(command)
561 }
562}

570fn scope(table: &Table, n: &str) -> Scope {
571 table
572 .columns
573 .iter()
574 .map(|c| ScopeColumn {
575 table: n.into(),
576 name: c.name.clone(),
577 ty: c.ty,
578 })
579 .collect()
580}
581fn resolve(r: &Option<String>, c: &str, s: &Scope) -> Result<usize, String> {
582 let raw = r.as_ref().map_or(c.to_string(), |r| format!("{}.{}", r, c));
583 if let Some(r) = r {
584 if !s.iter().any(|x| x.table == *r) {
585 return Err(format!("unknown table {}", r));
586 }
587 }
588 let matches: Vec<_> = s
589 .iter()
590 .enumerate()
591 .filter(|(_, x)| x.name == c && r.as_ref().is_none_or(|r| x.table == *r))
592 .map(|(i, _)| i)
593 .collect();
594 if matches.is_empty() {
595 Err(format!("unknown column {}", raw))
596 } else if matches.len() > 1 {
597 Err(format!("ambiguous column {}", c))
598 } else {
599 Ok(matches[0])
600 }
601}
602fn compatible(a: u8, b: u8) -> bool {
603 a == 0 || b == 0 || a == b
604}
605fn check(
606 e: &Expr,
607 s: &Scope,
608 aggregates: bool,
609 groups: Option<&[usize]>,
610 inside: bool,
611) -> Result<u8, String> {
612 match e {
613 Expr::Literal(v) => Ok(v.ty()),
614 Expr::Column(r, c) => {
615 let i = resolve(r, c, s)?;
616 if !inside && groups.is_some_and(|g| !g.contains(&i)) {
617 return Err(format!(
618 "column {} is not grouped",
619 r.as_ref().map_or(c.clone(), |r| format!("{}.{}", r, c))
620 ));
621 }
622 Ok(s[i].ty)
623 }
624 Expr::Paren(e) => check(e, s, aggregates, groups, inside),
625 Expr::Aggregate(n, _, arg) => {
626 if !aggregates || inside {
627 return Err(format!("misuse of aggregate {}", n));
628 }
629 let ty = if let Some(arg) = arg {
630 check(arg, s, aggregates, groups, true)?
631 } else {
632 1
633 };
634 if (n == "SUM" || n == "AVG") && !compatible(ty, 1) {
635 return Err(format!("type mismatch for {}", n));
636 }
637 Ok(if ["COUNT", "SUM", "AVG"].contains(&n.as_str()) {
638 1
639 } else {
640 ty
641 })
642 }
643 Expr::Op(op, args) => {
644 let types = args
645 .iter()
646 .map(|e| check(e, s, aggregates, groups, inside))
647 .collect::<Result<Vec<_>, _>>()?;
648 let error = || format!("type mismatch for {}", if op == "NEG" { "-" } else { op });
649 let required = match op.as_str() {
650 "||" | "LIKE" => 2,
651 "IS NULL" | "IS NOT NULL" => return Ok(1),
652 "IN" | "BETWEEN" | "=" | "!=" | "<>" | "<" | ">" | "<=" | ">=" => {
653 if types.iter().skip(1).any(|&t| !compatible(types[0], t)) {
654 return Err(error());
655 }
656 return Ok(1);
657 }
658 _ => 1,
659 };
660 if types.iter().any(|&t| !compatible(t, required)) {
661 return Err(error());
662 }
663 Ok(if op == "||" { 2 } else { 1 })
664 }
665 }
666}
667fn condition(
668 e: &Option<Expr>,
669 s: &Scope,
670 ag: bool,
671 g: Option<&[usize]>,
672 kind: &str,
673) -> Result<(), String> {
674 if let Some(e) = e {
675 if !compatible(check(e, s, ag, g, false)?, 1) {
676 return Err(format!("type mismatch for {}", kind));
677 }
678 }
679 Ok(())
680}
681fn logical(a: &Value, b: &Value, and: bool) -> Value {
682 let a = a.truth();
683 let b = b.truth();
684 Value::boolean(if and {
685 if a == Some(false) || b == Some(false) {
686 Some(false)
687 } else if a.is_none() || b.is_none() {
688 None
689 } else {
690 Some(true)
691 }
692 } else if a == Some(true) || b == Some(true) {
693 Some(true)
694 } else if a.is_none() || b.is_none() {
695 None
696 } else {
697 Some(false)
698 })
699}
700fn compare(a: &Value, b: &Value, op: &str) -> Value {
701 if *a == Value::Null || *b == Value::Null {
702 return Value::Null;
703 }
704 let ord = a.cmp(b);
705 Value::Int(match op {
706 "=" => a == b,
707 "!=" | "<>" => a != b,
708 "<" => ord.is_lt(),
709 ">" => ord.is_gt(),
710 "<=" => !ord.is_gt(),
711 _ => !ord.is_lt(),
712 } as i64)
713}
714fn like(s: &str, p: &str) -> bool {
715 let mut dp = vec![false; s.len() + 1];
716 dp[0] = true;
717 for c in p.bytes() {
718 let mut next = vec![false; s.len() + 1];
719 if c == b'%' {
720 next[0] = dp[0]
721 }
722 for i in 1..=s.len() {
723 next[i] = if c == b'%' {
724 dp[i] || next[i - 1]
725 } else {
726 dp[i - 1] && (c == b'_' || c == s.as_bytes()[i - 1])
727 }
728 }
729 dp = next
730 }
731 dp[s.len()]
732}
733fn evaluate(e: &Expr, s: &Scope, row: &[Value], group: &[Vec<Value>]) -> Value {
734 match e {
735 Expr::Literal(v) => v.clone(),
736 Expr::Column(r, c) => row
737 .get(resolve(r, c, s).unwrap())
738 .cloned()
739 .unwrap_or(Value::Null),
740 Expr::Paren(e) => evaluate(e, s, row, group),
741 Expr::Aggregate(n, distinct, arg) => {
742 if arg.is_none() {
743 return Value::Int(group.len() as i64);
744 }
745 let arg = arg.as_ref().unwrap();
746 let mut values = Vec::new();
747 for r in group {
748 let v = evaluate(arg, s, r, &[]);
749 if v != Value::Null && (!distinct || !values.contains(&v)) {
750 values.push(v)
751 }
752 }
753 if n == "COUNT" {
754 return Value::Int(values.len() as i64);
755 }
756 if values.is_empty() {
757 return Value::Null;
758 }
759 match n.as_str() {
760 "MIN" => values.into_iter().min().unwrap(),
761 "MAX" => values.into_iter().max().unwrap(),
762 _ => {
763 let sum: i64 = values
764 .iter()
765 .map(|v| if let Value::Int(n) = v { *n } else { 0 })
766 .sum();
767 Value::Int(if n == "AVG" {
768 sum / values.len() as i64
769 } else {
770 sum
771 })
772 }
773 }
774 }
775 Expr::Op(op, args) => {
776 let v: Vec<_> = args.iter().map(|e| evaluate(e, s, row, group)).collect();
777 match op.as_str() {
778 "NOT" => Value::boolean(v[0].truth().map(|b| !b)),
779 "AND" => logical(&v[0], &v[1], true),
780 "OR" => logical(&v[0], &v[1], false),
781 "IS NULL" => Value::Int((v[0] == Value::Null) as i64),
782 "IS NOT NULL" => Value::Int((v[0] != Value::Null) as i64),
783 "IN" => {
784 if v[0] == Value::Null {
785 Value::Null
786 } else if v[1..].contains(&v[0]) {
787 Value::Int(1)
788 } else if v[1..].contains(&Value::Null) {
789 Value::Null
790 } else {
791 Value::Int(0)
792 }
793 }
794 "BETWEEN" => logical(
795 &compare(&v[0], &v[1], ">="),
796 &compare(&v[0], &v[2], "<="),
797 true,
798 ),
799 "=" | "!=" | "<>" | "<" | ">" | "<=" | ">=" => compare(&v[0], &v[1], op),
800 _ => {
801 if v.contains(&Value::Null) {
802 return Value::Null;
803 }
804 if op == "LIKE" || op == "||" {
805 if let (Value::Text(a), Value::Text(b)) = (&v[0], &v[1]) {
806 if op == "LIKE" {
807 Value::Int(like(a, b) as i64)
808 } else {
809 Value::Text(format!("{}{}", a, b))
810 }
811 } else {
812 Value::Null
813 }
814 } else {
815 let a = if let Value::Int(a) = v[0] { a } else { 0 };
816 let b = if v.len() > 1 {
817 if let Value::Int(b) = v[1] { b } else { 0 }
818 } else {
819 0
820 };
821 match op.as_str() {
822 "NEG" => Value::Int(-a),
823 "+" => Value::Int(a + b),
824 "-" => Value::Int(a - b),
825 "*" => Value::Int(a * b),
826 "/" => {
827 if b == 0 {
828 Value::Null
829 } else {
830 Value::Int(a / b)
831 }
832 }
833 "%" => {
834 if b == 0 {
835 Value::Null
836 } else {
837 Value::Int(a % b)
838 }
839 }
840 _ => Value::Null,
841 }
842 }
843 }
844 }
845 }
846 }
847}
848fn table<'a>(db: &'a BTreeMap<String, Table>, n: &str) -> Result<&'a Table, String> {
849 db.get(n).ok_or_else(|| format!("unknown table {}", n))
850}
851fn targets(t: &Table, names: &[String]) -> Result<Vec<usize>, String> {
852 let mut ids = Vec::new();
853 for n in names {
854 let i = t
855 .columns
856 .iter()
857 .position(|c| c.name == *n)
858 .ok_or_else(|| format!("unknown column {}", n))?;
859 if ids.contains(&i) {
860 return Err(format!("duplicate column {}", n));
861 }
862 ids.push(i)
863 }
864 Ok(ids)
865}
866fn constraints(t: &Table, ids: &[usize]) -> Result<(), String> {
867 let primary = t.columns.iter().position(|c| c.pk > 0);
868 for &i in ids {
869 let row = &t.rows[i];
870 for (j, c) in t.columns.iter().enumerate() {
871 if c.not_null && row[j] == Value::Null {
872 return Err(format!("NULL value in column {}", c.name));
873 }
874 }
875 if let Some(pk) = primary {
876 if t.rows
877 .iter()
878 .enumerate()
879 .any(|(j, r)| j != i && r[pk] == row[pk])
880 {
881 return Err(format!("duplicate primary key {}", row[pk].text()));
882 }
883 }
884 }
885 Ok(())
886}
887fn result_table(items: &[Item], types: &[u8], rows: &[Vec<Value>]) -> Vec<String> {
888 let strings: Vec<Vec<_>> = rows
889 .iter()
890 .map(|r| r.iter().map(Value::text).collect())
891 .collect();
892 let widths: Vec<_> = items
893 .iter()
894 .enumerate()
895 .map(|(i, item)| {
896 strings
897 .iter()
898 .map(|r| r[i].len())
899 .chain(std::iter::once(item.heading.len()))
900 .max()
901 .unwrap()
902 })
903 .collect();
904 let separator = format!(
905 "+{}",
906 widths
907 .iter()
908 .map(|w| format!("{}+", "-".repeat(w + 2)))
909 .collect::<String>()
910 );
911 let line = |cells: &[String], header: bool| {
912 format!(
913 "|{}",
914 cells
915 .iter()
916 .enumerate()
917 .map(|(i, c)| format!(
918 " {} |",
919 if !header && types[i] == 1 {
920 format!("{:>width$}", c, width = widths[i])
921 } else {
922 format!("{:<width$}", c, width = widths[i])
923 }
924 ))
925 .collect::<String>()
926 )
927 };
928 let mut out = vec![
929 separator.clone(),
930 line(
931 &items.iter().map(|i| i.heading.clone()).collect::<Vec<_>>(),
932 true,
933 ),
934 separator.clone(),
935 ];
936 for r in strings {
937 out.push(line(&r, false))
938 }
939 out.push(separator);
940 out.push(format!(
941 "({} row{})",
942 rows.len(),
943 if rows.len() == 1 { "" } else { "s" }
944 ));
945 out
946}
947fn execute(command: Command, db: &mut BTreeMap<String, Table>) -> Result<Vec<String>, String> {
948 let message = |n: usize, verb: &str| {
949 vec![format!(
950 "{} row{} {}.",
951 n,
952 if n == 1 { "" } else { "s" },
953 verb
954 )]
955 };
956 match command {
957 Command::Empty | Command::Exit => Ok(Vec::new()),
958 Command::Create(n, cols) => {
959 if db.contains_key(&n) {
960 return Err(format!("table {} already exists", n));
961 }
962 let mut seen = BTreeSet::new();
963 let mut pk = 0;
964 for c in &cols {
965 if !seen.insert(&c.name) {
966 return Err(format!("duplicate column {}", c.name));
967 }
968 pk += c.pk;
969 if pk > 1 {
970 return Err("multiple primary keys".into());
971 }
972 }
973 db.insert(
974 n.clone(),
975 Table {
976 columns: cols,
977 rows: Vec::new(),
978 },
979 );
980 Ok(vec![format!("Table {} created.", n)])
981 }
982 Command::Drop(n) => {
983 table(db, &n)?;
984 db.remove(&n);
985 Ok(vec![format!("Table {} dropped.", n)])
986 }
987 Command::Insert(n, names, tuples) => {
988 let mut t = table(db, &n)?.clone();
989 let ids = if let Some(names) = names {
990 targets(&t, &names)?
991 } else {
992 (0..t.columns.len()).collect()
993 };
994 for es in &tuples {
995 if es.len() != ids.len() {
996 return Err(format!("expected {} values, got {}", ids.len(), es.len()));
997 }
998 for (e, &i) in es.iter().zip(&ids) {
999 let ty = check(e, &vec![], false, None, false)?;
1000 if !compatible(ty, t.columns[i].ty) {
1001 return Err(format!("type mismatch for column {}", t.columns[i].name));
1002 }
1003 }
1004 }
1005 let old = t.rows.len();
1006 for es in &tuples {
1007 let mut row = vec![Value::Null; t.columns.len()];
1008 for (e, &i) in es.iter().zip(&ids) {
1009 row[i] = evaluate(e, &vec![], &[], &[])
1010 }
1011 t.rows.push(row)
1012 }
1013 constraints(&t, &(old..t.rows.len()).collect::<Vec<_>>())?;
1014 let count = tuples.len();
1015 db.insert(n, t);
1016 Ok(message(count, "inserted"))
1017 }
1018 Command::Update(n, assignments, filter) => {
1019 let mut t = table(db, &n)?.clone();
1020 let ids = targets(
1021 &t,
1022 &assignments
1023 .iter()
1024 .map(|(n, _)| n.clone())
1025 .collect::<Vec<_>>(),
1026 )?;
1027 let s = scope(&t, &n);
1028 for ((_, e), &i) in assignments.iter().zip(&ids) {
1029 if !compatible(check(e, &s, false, None, false)?, t.columns[i].ty) {
1030 return Err(format!("type mismatch for column {}", t.columns[i].name));
1031 }
1032 }
1033 condition(&filter, &s, false, None, "WHERE")?;
1034 let mut modified = Vec::new();
1035 for i in 0..t.rows.len() {
1036 if filter
1037 .as_ref()
1038 .is_none_or(|e| evaluate(e, &s, &t.rows[i], &[]).truth() == Some(true))
1039 {
1040 let old = t.rows[i].clone();
1041 for ((_, e), &j) in assignments.iter().zip(&ids) {
1042 t.rows[i][j] = evaluate(e, &s, &old, &[])
1043 }
1044 modified.push(i)
1045 }
1046 }
1047 constraints(&t, &modified)?;
1048 db.insert(n, t);
1049 Ok(message(modified.len(), "updated"))
1050 }
1051 Command::Delete(n, filter) => {
1052 let mut t = table(db, &n)?.clone();
1053 let s = scope(&t, &n);
1054 condition(&filter, &s, false, None, "WHERE")?;
1055 let old = t.rows.len();
1056 t.rows.retain(|r| {
1057 !filter
1058 .as_ref()
1059 .is_none_or(|e| evaluate(e, &s, r, &[]).truth() == Some(true))
1060 });
1061 let count = old - t.rows.len();
1062 db.insert(n, t);
1063 Ok(message(count, "deleted"))
1064 }
1065 Command::Select(q) => {
1066 let first = table(db, &q.from.0)?;
1067 let mut s = scope(first, &q.from.1);
1068 let mut seen = BTreeSet::from([q.from.1.clone()]);
1069 let mut scopes = vec![s.clone()];
1070 for (_, t, on) in &q.joins {
1071 let table = table(db, &t.0)?;
1072 if !seen.insert(t.1.clone()) {
1073 return Err(format!("duplicate table name {}", t.1));
1074 }
1075 s.extend(scope(table, &t.1));
1076 condition(&Some(on.clone()), &s, false, None, "ON")?;
1077 scopes.push(s.clone())
1078 }
1079 condition(&q.filter, &s, false, None, "WHERE")?;
1080 let mut groups = Vec::new();
1081 if let Some(es) = &q.groups {
1082 for e in es {
1083 if let Expr::Column(r, c) = e {
1084 groups.push(resolve(r, c, &s)?);
1085 }
1086 }
1087 }
1088 let expanded = q.items.is_none();
1089 let items = q.items.unwrap_or_else(|| {
1090 s.iter()
1091 .map(|c| Item {
1092 expr: Expr::Column(Some(c.table.clone()), c.name.clone()),
1093 alias: None,
1094 heading: c.name.clone(),
1095 })
1096 .collect()
1097 });
1098 let ag = q.groups.is_some()
1099 || q.having.is_some()
1100 || items.iter().any(|i| i.expr.aggregate())
1101 || q.order.iter().any(|(e, _)| e.aggregate());
1102 let grouping = if ag { Some(groups.as_slice()) } else { None };
1103 let mut types = Vec::new();
1104 for item in &items {
1105 let ty = check(&item.expr, &s, true, grouping, false).map_err(|error| {
1106 if expanded
1107 && error.starts_with("column ")
1108 && error.ends_with(" is not grouped")
1109 {
1110 format!("column {} is not grouped", item.heading)
1111 } else {
1112 error
1113 }
1114 })?;
1115 types.push(ty);
1116 }
1117 condition(&q.having, &s, true, grouping, "HAVING")?;
1118 let mut keys = Vec::new();
1119 for (e, desc) in &q.order {
1120 let position = match e {
1121 Expr::Literal(Value::Int(n)) => {
1122 if *n < 1 || *n as usize > items.len() {
1123 return Err(format!("ORDER BY position {} is out of range", n));
1124 }
1125 Some(*n as usize - 1)
1126 }
1127 Expr::Column(None, n) => items.iter().position(|i| i.alias.as_ref() == Some(n)),
1128 _ => None,
1129 };
1130 if position.is_none() {
1131 check(e, &s, true, grouping, false)?;
1132 }
1133 keys.push((e.clone(), *desc, position));
1134 }
1135 let mut rows = first.rows.clone();
1136 for (j, (left, t, on)) in q.joins.iter().enumerate() {
1137 let other = table(db, &t.0)?;
1138 let mut next = Vec::new();
1139 for r in &rows {
1140 let mut matched = false;
1141 for right in &other.rows {
1142 let mut row = r.clone();
1143 row.extend(right.clone());
1144 if evaluate(on, &scopes[j + 1], &row, &[]).truth() == Some(true) {
1145 next.push(row);
1146 matched = true
1147 }
1148 }
1149 if *left && !matched {
1150 let mut row = r.clone();
1151 row.extend(vec![Value::Null; other.columns.len()]);
1152 next.push(row)
1153 }
1154 }
1155 rows = next
1156 }
1157 rows.retain(|r| {
1158 q.filter
1159 .as_ref()
1160 .is_none_or(|e| evaluate(e, &s, r, &[]).truth() == Some(true))
1161 });
1162 let mut grouped: Vec<Vec<Vec<Value>>> = Vec::new();
1163 if ag {
1164 if q.groups.is_some() {
1165 let mut indices: BTreeMap<Vec<Value>, usize> = BTreeMap::new();
1166 for row in rows {
1167 let key = groups.iter().map(|&i| row[i].clone()).collect::<Vec<_>>();
1168 let i = if let Some(&i) = indices.get(&key) {
1169 i
1170 } else {
1171 let i = grouped.len();
1172 indices.insert(key, i);
1173 grouped.push(Vec::new());
1174 i
1175 };
1176 grouped[i].push(row)
1177 }
1178 } else {
1179 grouped.push(rows)
1180 }
1181 } else {
1182 for row in rows {
1183 grouped.push(vec![row])
1184 }
1185 }
1186 let mut result = Vec::new();
1187 for group in grouped {
1188 let empty = Vec::new();
1189 let row = group.first().unwrap_or(&empty);
1190 if q.having
1191 .as_ref()
1192 .is_some_and(|e| evaluate(e, &s, row, &group).truth() != Some(true))
1193 {
1194 continue;
1195 }
1196 let values = items
1197 .iter()
1198 .map(|i| evaluate(&i.expr, &s, row, &group))
1199 .collect::<Vec<_>>();
1200 let order = keys
1201 .iter()
1202 .map(|(e, _, p)| {
1203 p.map_or_else(|| evaluate(e, &s, row, &group), |i| values[i].clone())
1204 })
1205 .collect::<Vec<_>>();
1206 result.push((values, order))
1207 }
1208 result.sort_by(|a, b| {
1209 for (i, (_, desc, _)) in keys.iter().enumerate() {
1210 let ord = a.1[i].cmp(&b.1[i]);
1211 if !ord.is_eq() {
1212 return if *desc { ord.reverse() } else { ord };
1213 }
1214 }
1215 std::cmp::Ordering::Equal
1216 });
1217 if q.distinct {
1218 let mut unique = BTreeSet::new();
1219 result.retain(|r| unique.insert(r.0.clone()))
1220 }
1221 let output = result
1222 .into_iter()
1223 .skip(q.offset)
1224 .take(q.limit.unwrap_or(usize::MAX))
1225 .map(|r| r.0)
1226 .collect::<Vec<_>>();
1227 Ok(result_table(&items, &types, &output))
1228 }
1229 }
1230}
