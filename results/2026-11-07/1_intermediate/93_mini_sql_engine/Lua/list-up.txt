local NULL = {}
local function fail(s)
    error(s, 0)
end
local function concat(a, b)
    local z = {}
    for _, v in ipairs(a) do
        z[#z + 1] = v
    end
    for _, v in ipairs(b) do
        z[#z + 1] = v
    end
    return z
end

local function column(t, name)
    for i, c in ipairs(t.columns) do
        if c.name == name then
            return i
        end
    end
end
local function resolve(e, scope)
    local found, kind, table_seen
    for _, entry in ipairs(scope) do
        if not e.table or e.table == entry.name then
            table_seen = true
            local i = column(entry.table, e.column)
            if i then
                if found then
                    fail("ambiguous column " .. e.column)
                end
                found = entry.offset + i
                kind = entry.table.columns[i].type
            end
        end
    end
    if e.table and not table_seen then
        fail("unknown table " .. e.table)
    end
    if not found then
        fail("unknown column " .. e.text)
    end
    e.index = found
    return kind
end
local function compatible(a, b)
    return a == b or a == "NULL" or b == "NULL"
end
local function require_type(a, b, op)
    if not compatible(a, b) then
        fail("type mismatch for " .. op)
    end
end
local check
check = function(e, scope, allow, grouped, inside)
    if e.kind == "literal" then
        return e.value == NULL and "NULL" or type(e.value) == "number" and "INT" or "TEXT"
    end
    if e.kind == "ref" then
        local t = resolve(e, scope)
        if grouped and not inside and not grouped[e.index] then
            fail("column " .. e.text .. " is not grouped")
        end
        return t
    end
    if e.kind == "aggregate" then
        if not allow or inside then
            fail("misuse of aggregate " .. e.op)
        end
local function condition(e, scope, name, allow, grouped)
    if e then
        require_type(check(e, scope, allow, grouped), "INT", name)
    end
end
local function truth(v)
    return v ~= NULL and v ~= 0
end
local evaluate
evaluate = function(e, row, group)
    if e.kind == "literal" then
        return e.value
    elseif e.kind == "ref" then
        return row[e.index]
    end
local function satisfies(e, row, group)
    return not e or truth(evaluate(e, row, group))
end
local function count(n)
    return n .. (n == 1 and " row" or " rows")
end
local function written(v)
    return v == NULL and "NULL" or type(v) == "number" and string.format("%d", v) or v
end
local Database = {}
Database.__index = Database
function Database:table(n)
    if not self.tables[n] then
        fail("unknown table " .. n)
    end
    return self.tables[n]
end
function Database:targets(t, names)
    local a = {}
    local seen = {}
    for _, name in ipairs(names) do
        local i = column(t, name)
        if not i then
            fail("unknown column " .. name)
        end
        if seen[i] then
            fail("duplicate column " .. name)
        end
        seen[i] = true
        a[#a + 1] = i
    end
    return a
end
function Database:constraints(t, rows, checked)
    for _, i in ipairs(checked) do
        for j, c in ipairs(t.columns) do
            if c.notnull and rows[i][j] == NULL then
                fail("NULL value in column " .. c.name)
            end
        end
        if t.key then
            for j, r in ipairs(rows) do
                if i ~= j and r[t.key] == rows[i][t.key] then
                    fail("duplicate primary key " .. written(r[t.key]))
                end
            end
        end
    end

        end
        local targets = self:targets(t, names)
        local scope = {{name = t.name, table = t, offset = 0}}
        for i, a in ipairs(s.assignments) do
            local c = t.columns[targets[i]]
            require_type(check(a[2], scope), c.type, "column " .. c.name)
        end
