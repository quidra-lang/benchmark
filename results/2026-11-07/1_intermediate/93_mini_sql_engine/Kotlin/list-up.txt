private data class Entry(val name:String, val table:Table, val offset:Int)
private class Scope {
    val entries = mutableListOf<Entry>();
    var width = 0
    fun add(n:String, t:Table) {
        if (entries.any {
            it.name == n
        }) fail("duplicate table name $n");
        entries.add(Entry(n, t, width));
        width += t.columns.size
    }
    fun resolve(e:Expr):String {
        val found = mutableListOf<Pair<Int, String>>();
        var seen = false;
        for (p in entries) if (e.table == null || e.table == p.name) {
            seen = true;
            val i = p.table.column(e.column);
            if (i >= 0) found.add(p.offset+i to p.table.columns[i].type)
        };
        if (e.table != null && !seen) fail("unknown table ${e.table}");
        if (found.isEmpty()) fail("unknown column ${e.text}");
        if (found.size>1) fail("ambiguous column ${e.column}");
        e.index = found[0].first;
        return found[0].second
    }
}
