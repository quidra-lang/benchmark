9private fun tokenize(s:String):List<Token> {
10 val a = mutableListOf<Token>();
11 var i = 0
12 while (i<s.length) {
13 val c = s[i];
14 val start = i;
15 if (c == ' ' || c == '\n') {
16 i++;
17 continue
18 };
19 if (s.startsWith("--", i)) {
20 i = s.indexOf('\n', i).let {
21 if (it<0) s.length else it
22 };
23 continue
24 }
25 if (c == '\'') {
26 i = s.indexOf('\'', i+1)+1;
27 a.add(Token("string", s.substring(start, i), s.substring(start+1, i-1)))
28 }
29 else if (c in '0'..'9') {
30 while (i<s.length && s[i] in '0'..'9') i++;
31 val t = s.substring(start, i);
32 a.add(Token("integer", t, t.toLong()))
33 }
34 else if (c.isLetter() || c == '_') {
35 while (i<s.length && (s[i].isLetterOrDigit() || s[i] == '_')) i++;
36 val t = s.substring(start, i);
37 val up = t.uppercase(Locale.ROOT);
38 a.add(Token(if (up in reserved)"keyword" else "name", t, if (up in reserved) up else t))
39 }
40 else {
41 var t = s.substring(i, minOf(i+2, s.length));
42 if (t !in setOf("<>", "!=", "<=", ">=", "||")) t = c.toString();
43 if (t.length == 1 && c !in "(),.;*/%+-=<>") fail("unexpected character '$c'");
44 a.add(Token("symbol", t, t));
45 i += t.length
46 }
47 };
48 return a
49}

86private class Parser(val tokens:List<Token>) {
87 var position = 0
88 fun peek() = tokens[position]
89 fun take() = tokens[position++]
90 fun at(w:String):Boolean {
91 val t = peek();
92 return t.kind in setOf("keyword", "symbol") && t.value == w
93 }
94 fun accept(w:String):Boolean {
95 if (!at(w)) return false;
96 position++;
97 return true
98 }
99 fun syntax():Nothing = fail("syntax error at '${peek().text}'")
100 fun expect(w:String) {
101 if (!accept(w)) syntax()
102 }
103 fun name():String {
104 if (peek().kind != "name") syntax();
105 return take().value as String
106 }
107 fun integer():Long {
108 if (peek().kind != "integer") syntax();
109 return take().value as Long
110 }
111 fun <T> comma(read:() -> T):List<T> {
112 val a = mutableListOf(read());
113 while (accept(",")) a.add(read());
114 return a
115 }
116 fun <T> parens(read:() -> T):List<T> {
117 expect("(");
118 val a = comma(read);
119 expect(")");
120 return a
121 }
122 fun alias():String? = if (accept("AS")) name() else if (peek().kind == "name") name() else null
123 fun table():TableRef {
124 val n = name();
125 return TableRef(n, alias() ?: n)
126 }
127 fun reference():Expr {
128 val n = name();
129 val e = Expr("ref");
130 if (accept(".")) {
131 e.table = n;
132 e.column = name();
133 e.text = n+"."+e.column
134 } else {
135 e.column = n;
136 e.text = n
137 };
138 return e
139 }
140 fun expr(level:Int = 0):Expr {
141 if (level == 2 && accept("NOT")) return Expr("unary", "NOT", listOf(expr(2)));
142 if (level == 6 && accept("-")) return Expr("unary", "-", listOf(expr(6)));
143 if (level == 7) return primary();
144 var left = expr(level+1)
145 if (level == 3) {
146 if (peek().kind == "symbol" && (peek().value as String) in comparisons) {
147 val op = take().value as String;
148 return Expr("binary", op, listOf(left, expr(4)))
149 };
150 if (accept("IS")) {
151 val neg = accept("NOT");
152 expect("NULL");
153 return Expr("isnull", args = listOf(left)).also {
154 it.negated = neg
155 }
156 };
157 val neg = accept("NOT");
158 var e:Expr? = null
159 if (accept("IN")) e = Expr("in", args = listOf(left)+parens {
160 expr()
161 }) else if (accept("BETWEEN")) {
162 val low = expr(4);
163 expect("AND");
164 e = Expr("between", args = listOf(left, low, expr(4)))
165 } else if (accept("LIKE")) e = Expr("like", args = listOf(left, expr(4))) else if (neg) syntax();
166 if (e != null) {
167 e.negated = neg;
168 return e
169 };
170 return left
171 }
172 val ops = listOf(listOf("OR"), listOf("AND"), emptyList(), emptyList(), listOf("+", "-", "||"), listOf("*", "/", "%"), emptyList())
173 while (ops[level].any {
174 at(it)
175 }) {
176 val op = take().value as String;
177 left = Expr("binary", op, listOf(left, expr(level+1)))
178 };
179 return left
180 }
181 fun primary():Expr {
182 val t = peek();
183 if (t.kind == "integer" || t.kind == "string") {
184 take();
185 return Expr("literal", value = t.value)
186 };
187 if (accept("NULL")) return Expr("literal");
188 if (t.kind == "name") return reference();
189 if (t.kind == "keyword" && (t.value as String) in aggregates) {
190 val n = take().value as String;
191 expect("(");
192 var distinct = false;
193 val args = if (n == "COUNT" && accept("*")) emptyList() else {
194 distinct = accept("DISTINCT");
195 listOf(expr())
196 };
197 expect(")");
198 return Expr("aggregate", n, args).also {
199 it.distinct = distinct
200 }
201 };
202 if (accept("(")) {
203 val e = expr();
204 expect(")");
205 return Expr("group", args = listOf(e))
206 };
207 syntax()
208 }
209 fun clause(w:String):Expr? = if (accept(w)) expr() else null
210 fun item():Item {
211 val start = position;
212 val e = expr();
213 val stop = position;
214 val alias = alias();
215 val heading = alias ?: if (e.kind == "ref") e.column else buildString {
216 for (i in start until stop) {
217 if (i>start && tokens[i].kind != "symbol" && tokens[i-1].kind != "symbol") append(' ');
218 append(tokens[i].text)
219 }
220 };
221 return Item(e, alias, heading)
222 }
223 fun orderItem():Order {
224 val start = position;
225 val e = expr();
226 val single = position == start+1;
227 val t = tokens[start];
228 val desc = accept("DESC");
229 if (!desc) accept("ASC");
230 return Order(e, desc, if (single && t.kind == "integer") t.value as Long else null, if (single && t.kind == "name") t.value as String else null)
231 }
232 fun query():Query {
233 val q = Query();
234 q.distinct = accept("DISTINCT");
235 q.star = accept("*");
236 if (!q.star) q.items.addAll(comma {
237 item()
238 });
239 expect("FROM");
240 q.source = table();
241 while (at("JOIN") || at("INNER") || at("LEFT")) {
242 val left = accept("LEFT");
243 if (left) accept("OUTER") else accept("INNER");
244 expect("JOIN");
245 val t = table();
246 expect("ON");
247 q.joins.add(Join(t, left, expr()))
248 };
249 q.where = clause("WHERE");
250 if (accept("GROUP")) {
251 expect("BY");
252 q.group = comma {
253 reference()
254 }
255 };
256 q.having = clause("HAVING");
257 if (accept("ORDER")) {
258 expect("BY");
259 q.order.addAll(comma {
260 orderItem()
261 })
262 };
263 if (accept("LIMIT")) {
264 q.limit = integer();
265 if (accept("OFFSET")) q.offset = integer()
266 };
267 return q
268 }
269 fun column():Column {
270 val n = name();
271 if (!at("INT") && !at("TEXT")) syntax();
272 val type = take().value as String;
273 var pk = 0;
274 var nn = false;
275 while (true) {
276 if (accept("PRIMARY")) {
277 expect("KEY");
278 pk++;
279 nn = true
280 } else if (accept("NOT")) {
281 expect("NULL");
282 nn = true
283 } else break
284 };
285 return Column(n, type, pk, nn)
286 }
287 fun statement():Command {
288 val c = Command();
289 if (accept("CREATE")) {
290 c.kind = "create";
291 expect("TABLE");
292 c.name = name();
293 c.columns = parens {
294 column()
295 }
296 } else if (accept("DROP")) {
297 c.kind = "drop";
298 expect("TABLE");
299 c.name = name()
300 } else if (accept("INSERT")) {
301 c.kind = "insert";
302 expect("INTO");
303 c.name = name();
304 if (at("(")) c.names = parens {
305 name()
306 };
307 expect("VALUES");
308 c.tuples = comma {
309 parens {
310 expr()
311 }
312 }
313 } else if (accept("UPDATE")) {
314 c.kind = "update";
315 c.name = name();
316 expect("SET");
317 c.assignments = comma {
318 val n = name();
319 expect("=");
320 n to expr()
321 };
322 c.where = clause("WHERE")
323 } else if (accept("DELETE")) {
324 c.kind = "delete";
325 expect("FROM");
326 c.name = name();
327 c.where = clause("WHERE")
328 } else if (accept("SELECT")) {
329 c.kind = "select";
330 c.query = query()
331 } else if (accept("EXIT")) c.kind = "exit";
332 expect(";");
333 return c
334 }

354 fun resolve(e:Expr):String {
355 val found = mutableListOf<Pair<Int, String>>();
356 var seen = false;
357 for (p in entries) if (e.table == null || e.table == p.name) {
358 seen = true;
359 val i = p.table.column(e.column);
360 if (i >= 0) found.add(p.offset+i to p.table.columns[i].type)
361 };
362 if (e.table != null && !seen) fail("unknown table ${e.table}");
363 if (found.isEmpty()) fail("unknown column ${e.text}");
364 if (found.size>1) fail("ambiguous column ${e.column}");
365 e.index = found[0].first;
366 return found[0].second
367 }

369private fun compatible(a:String, b:String) = a == b || a == "NULL" || b == "NULL"
370private fun requireType(a:String, b:String, op:String) {
371 if (!compatible(a, b)) fail("type mismatch for $op")
372}
373private fun check(e:Expr, s:Scope, allow:Boolean = false, grouped:List<Int>? = null, inside:Boolean = false):String {
374 if (e.kind == "literal") return if (e.value == null)"NULL" else if (e.value is Long)"INT" else "TEXT";
375 if (e.kind == "ref") {
376 val t = s.resolve(e);
377 if (grouped != null && !inside && e.index !in grouped) fail("column ${e.text} is not grouped");
378 return t
379 }
380 if (e.kind == "aggregate") {
381 if (!allow || inside) fail("misuse of aggregate ${e.op}");
382 if (e.args.isEmpty()) return "INT";
383 val t = check(e.args[0], s, allow, grouped, true);
384 if (e.op == "SUM" || e.op == "AVG") requireType(t, "INT", e.op);
385 return if (e.op == "MIN" || e.op == "MAX") t else "INT"
386 }
387 val ts = e.args.map {
388 check(it, s, allow, grouped, inside)
389 };
390 if (e.kind == "group") return ts[0];
391 if (e.kind == "isnull") return "INT";
392 if (e.kind == "binary" && e.op in comparisons) {
393 requireType(ts[0], ts[1], e.op);
394 return "INT"
395 };
396 if (e.kind == "in" || e.kind == "between") {
397 for (t in ts.drop(1)) requireType(t, ts[0], if (e.kind == "in")"IN" else "BETWEEN");
398 return "INT"
399 };
400 val op = if (e.kind == "like")"LIKE" else e.op;
401 val t = if (op == "LIKE" || op == "||")"TEXT" else "INT";
402 for (actual in ts) requireType(actual, t, op);
403 return if (op == "||")"TEXT" else "INT"
404}
405private fun condition(e:Expr?, s:Scope, name:String, allow:Boolean = false, grouped:List<Int>? = null) {
406 if (e != null) requireType(check(e, s, allow, grouped), "INT", name)
407}

411private fun truth(v:Any?) = v != null && (v as Long) != 0L
412private fun flag(b:Boolean) = if (b) 1L else 0L
413private fun inverse(v:Any?):Long? = if (v == null) null else flag(v == 0L)
414private fun compare(a:Any?, b:Any?):Int {
415 if (a == b) return 0;
416 if (a == null) return -1;
417 if (b == null) return 1;
418 return if (a is Long) a.compareTo(b as Long) else(a as String).compareTo(b as String)
419}
420private fun operate(op:String, a:Any?, b:Any?):Any? {
421 if (op == "AND") return if (a == 0L || b == 0L) 0L else if (a == null || b == null) null else 1L;
422 if (op == "OR") return if (truth(a) || truth(b)) 1L else if (a == null || b == null) null else 0L;
423 if (a == null || b == null) return null;
424 if (op in comparisons) {
425 val c = compare(a, b);
426 return flag(when (op) {
427 "=" -> c == 0;
428 "<>", "!=" -> c != 0;
429 "<" -> c<0;
430 "<=" -> c <= 0;
431 ">" -> c>0;
432 else -> c >= 0
433 })
434 };
435 if (op == "||") return (a as String)+(b as String);
436 val x = a as Long;
437 val y = b as Long;
438 return when (op) {
439 "+" -> x+y;
440 "-" -> x-y;
441 "*" -> x*y;
442 "/" -> if (y == 0L) null else x/y;
443 "%" -> if (y == 0L) null else x%y;
444 else -> error("operator")
445 }
446}
447private fun like(text:String, pattern:String):Boolean {
448 val dp = BooleanArray(text.length+1);
449 dp[0] = true;
450 for (c in pattern) {
451 if (c == '%') {
452 for (i in 1..text.length) dp[i] = dp[i] || dp[i-1]
453 } else {
454 for (i in text.length downTo 1) dp[i] = dp[i-1] && (c == '_' || c == text[i-1]);
455 dp[0] = false
456 }
457 };
458 return dp[text.length]
459}
460private fun evaluate(e:Expr, row:Row, group:List<Row>? = null):Any? {
461 if (e.kind == "literal") return e.value;
462 if (e.kind == "ref") return row[e.index];
463 if (e.kind == "aggregate") {
464 if (e.args.isEmpty()) return group!!.size.toLong();
465 var values = group!!.map {
466 evaluate(e.args[0], it)
467 }.filterNotNull();
468 if (e.distinct) values = values.distinct();
469 if (e.op == "COUNT") return values.size.toLong();
470 if (values.isEmpty()) return null;
471 if (e.op == "MIN" || e.op == "MAX") return values.reduce {
472 a, b -> if ((compare(a, b) <= 0) == (e.op == "MIN")) a else b
473 };
474 val sum = values.sumOf {
475 it as Long
476 };
477 return if (e.op == "AVG") sum/values.size else sum
478 }
479 val a = e.args.map {
480 evaluate(it, row, group)
481 };
482 if (e.kind == "group") return a[0];
483 if (e.kind == "unary") return if (e.op == "NOT") inverse(a[0]) else if (a[0] == null) null else -(a[0] as Long);
484 if (e.kind == "binary") return operate(e.op, a[0], a[1]);
485 if (e.kind == "isnull") return flag((a[0] == null) != e.negated)
486 val result = when (e.kind) {
487 "in" -> if (a[0] == null) null else if (a.drop(1).contains(a[0])) 1L else if (a.drop(1).contains(null)) null else 0L;
488 "between" -> operate("AND", operate(">=", a[0], a[1]), operate("<=", a[0], a[2]));
489 else -> if (a[0] == null || a[1] == null) null else flag(like(a[0] as String, a[1] as String))
490 };
491 return if (e.negated) inverse(result) else result
492}
493private fun satisfies(e:Expr?, r:Row, g:List<Row>? = null) = e == null || truth(evaluate(e, r, g))

521private class Database {
522 val tables = mutableMapOf<String, Table>()
523 fun table(n:String) = tables[n] ?: fail("unknown table $n")
524 fun targets(t:Table, names:List<String>):List<Int> {
525 val a = mutableListOf<Int>();
526 for (n in names) {
527 val i = t.column(n);
528 if (i<0) fail("unknown column $n");
529 if (i in a) fail("duplicate column $n");
530 a.add(i)
531 };
532 return a
533 }
534 fun constraints(t:Table, rows:List<Row>, checked:List<Int>) {
535 for (i in checked) {
536 for ((j, c) in t.columns.withIndex()) if (c.notNull && rows[i][j] == null) fail("NULL value in column ${c.name}");
537 if (t.key >= 0) for (j in rows.indices) if (i != j && rows[i][t.key] == rows[j][t.key]) fail("duplicate primary key ${written(rows[i][t.key])}")
538 }
539 }
540 fun execute(c:Command) {
541 if (c.kind == "select") {
542 select(c.query!!);
543 return
544 };
545 if (c.kind == "create") {
546 if (tables.containsKey(c.name)) fail("table ${c.name} already exists");
547 val seen = mutableSetOf<String>();
548 var primary = 0;
549 var key = -1;
550 for ((i, col) in c.columns.withIndex()) {
551 if (!seen.add(col.name)) fail("duplicate column ${col.name}");
552 primary += col.primary;
553 if (primary >= 2) fail("multiple primary keys");
554 if (col.primary>0) key = i
555 };
556 tables[c.name] = Table(c.name, c.columns, key);
557 println("Table ${c.name} created.");
558 return
559 }
560 val t = table(c.name);
561 if (c.kind == "drop") {
562 tables.remove(c.name);
563 println("Table ${c.name} dropped.");
564 return
565 };
566 val rows = mutableListOf<Row>();
567 val checked = mutableListOf<Int>();
568 var count = 0
569 if (c.kind == "insert") {
570 val target = c.names?.let {
571 targets(t, it)
572 } ?: t.columns.indices.toList();
573 val empty = Scope();
574 for (tuple in c.tuples) {
575 if (tuple.size != target.size) fail("expected ${target.size} values, got ${tuple.size}");
576 for ((i, e) in tuple.withIndex()) {
577 val col = t.columns[target[i]];
578 requireType(check(e, empty), col.type, "column ${col.name}")
579 }
580 };
581 rows.addAll(t.rows)
582 for (tuple in c.tuples) {
583 val r = MutableList<Any?>(t.columns.size) {
584 null
585 };
586 for ((i, e) in tuple.withIndex()) r[target[i]] = evaluate(e, emptyList());
587 checked.add(rows.size);
588 rows.add(r)
589 };
590 constraints(t, rows, checked);
591 count = c.tuples.size
592 } else if (c.kind == "update") {
593 val target = targets(t, c.assignments.map {
594 it.first
595 });
596 val scope = Scope();
597 scope.add(t.name, t);
598 for ((i, a) in c.assignments.withIndex()) {
599 val col = t.columns[target[i]];
600 requireType(check(a.second, scope), col.type, "column ${col.name}")
601 };
602 condition(c.where, scope, "WHERE")
603 for ((i, r) in t.rows.withIndex()) {
604 var next = r;
605 if (satisfies(c.where, r)) {
606 val replacement = r.toMutableList();
607 for ((j, a) in c.assignments.withIndex()) replacement[target[j]] = evaluate(a.second, r);
608 next = replacement;
609 checked.add(i);
610 count++
611 };
612 rows.add(next)
613 };
614 constraints(t, rows, checked)
615 } else {
616 val scope = Scope();
617 scope.add(t.name, t);
618 condition(c.where, scope, "WHERE");
619 for (r in t.rows) if (satisfies(c.where, r)) count++ else rows.add(r)
620 }
621 t.rows = rows;
622 println(countRows(count)+when (c.kind) {
623 "insert" -> " inserted.";
624 "update" -> " updated.";
625 else -> " deleted."
626 })
627 }
628 private data class Result(val values:Row, val keys:Row, val position:Int)
629 fun select(q:Query) {
630 val scope = Scope();
631 scope.add(q.source.alias, table(q.source.name));
632 for (j in q.joins) {
633 scope.add(j.table.alias, table(j.table.name));
634 condition(j.condition, scope, "ON")
635 };
636 condition(q.where, scope, "WHERE");
637 val grouping = q.group?.map {
638 scope.resolve(it);
639 it.index
640 } ?: emptyList()
641 if (q.star) for (p in scope.entries) for (c in p.table.columns) {
642 val e = Expr("ref");
643 e.table = p.name;
644 e.column = c.name;
645 e.text = c.name;
646 q.items.add(Item(e, null, c.name))
647 };
648 val aggregate = q.group != null || q.having != null || q.items.any {
649 hasAggregate(it.expr)
650 } || q.order.any {
651 hasAggregate(it.expr)
652 };
653 val grouped = if (aggregate) grouping else null;
654 val types = q.items.map {
655 check(it.expr, scope, true, grouped)
656 };
657 condition(q.having, scope, "HAVING", true, grouped)
658 val keys = q.order.map {
659 o -> var key:Expr? = null;
660 if (o.ordinal != null) {
661 val k = o.ordinal;
662 if (k<1 || k>q.items.size) fail("ORDER BY position $k is out of range");
663 key = q.items[k.toInt()-1].expr
664 } else if (o.alias != null) key = q.items.firstOrNull {
665 it.alias == o.alias
666 }?.expr;
667 if (key == null) {
668 check(o.expr, scope, true, grouped);
669 key = o.expr
670 };
671 key
672 }
673 var rows = scope.entries[0].table.rows;
674 for ((ji, j) in q.joins.withIndex()) {
675 val t = scope.entries[ji+1].table;
676 val next = mutableListOf<Row>();
677 for (left in rows) {
678 var found = false;
679 for (right in t.rows) {
680 val combined = left+right;
681 if (satisfies(j.condition, combined)) {
682 next.add(combined);
683 found = true
684 }
685 };
686 if (j.left && !found) next.add(left+List(t.columns.size) {
687 null
688 })
689 };
690 rows = next
691 };
692 rows = rows.filter {
693 satisfies(q.where, it)
694 }
695 val groups = if (!aggregate) rows.map {
696 listOf(it)
697 } else if (q.group == null) listOf(rows) else rows.groupBy {
698 r -> grouping.map {
699 r[it]
700 }
701 }.values.toList();
702 val results = mutableListOf<Result>()
703 for (g in groups) {
704 val row = g.firstOrNull() ?: emptyList();
705 val context = if (aggregate) g else null;
706 if (!satisfies(q.having, row, context)) continue;
707 results.add(Result(q.items.map {
708 evaluate(it.expr, row, context)
709 }, keys.map {
710 evaluate(it!!, row, context)
711 }, results.size))
712 }
713 results.sortWith {
714 a, b -> var result = 0;
715 for (i in keys.indices) {
716 result = compare(a.keys[i], b.keys[i]);
717 if (result != 0) {
718 if (q.order[i].descending) result = -result;
719 break
720 }
721 };
722 if (result == 0) a.position.compareTo(b.position) else result
723 };
724 val unique = if (q.distinct) results.distinctBy {
725 it.values
726 } else results;
727 val first = minOf(q.offset, unique.size.toLong()).toInt();
728 val end = q.limit?.let {
729 minOf(unique.size.toLong(), first+it).toInt()
730 } ?: unique.size;
731 printTable(q.items, types, unique.subList(first, end).map {
732 it.values
733 })
734 }
735}

749 };

751 };

760 }
