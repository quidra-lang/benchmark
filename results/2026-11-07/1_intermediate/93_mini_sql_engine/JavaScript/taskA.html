71function tokenize(text) {
72 const result = [];
73 for (let p = 0; p < text.length;) {
74 const ch = text[p];
75 if (ch === " " || ch === "\n" || ch === "\r") {
76 p++;
77 continue;
78 }
79 if (text.startsWith("--", p)) {
80 const end = text.indexOf("\n", p);
81 p = end < 0 ? text.length : end + 1;
82 continue;
83 }
84 if (ch === "'") {
85 const end = text.indexOf("'", p + 1);
86 result.push({
87 kind: "string",
88 text: text.slice(p, end + 1),
89 value: text.slice(p + 1, end),
90 });
91 p = end + 1;
92 } else if (/[0-9]/.test(ch)) {
93 const start = p;
94 while (p < text.length && /[0-9]/.test(text[p])) p++;
95 const word = text.slice(start, p);
96 result.push({ kind: "integer", text: word, value: Number(word) });
97 } else if (/[A-Za-z_]/.test(ch)) {
98 const start = p;
99 while (p < text.length && /[A-Za-z0-9_]/.test(text[p])) p++;
100 const word = text.slice(start, p),
101 upper = word.toUpperCase();
102 result.push({
103 kind: reserved.has(upper) ? "reserved" : "name",
104 text: word,
105 value: reserved.has(upper) ? upper : word,
106 });
107 } else {
108 let word = text.slice(p, p + 2);
109 if (!["<>", "!=", "<=", ">=", "||"].includes(word)) word = ch;
110 if (!["<>", "!=", "<=", ">=", "||"].includes(word) && !"(),.;*/%+-=<>".includes(ch))
111 fail("unexpected character '" + ch + "'");
112 result.push({ kind: "symbol", text: word, value: word });
113 p += word.length;
114 }
115 }
116 result.push({ kind: "end", text: "", value: "" });
117 return result;
118}

122 if (i > 0 && tokens[i - 1].kind !== "symbol" && token.kind !== "symbol") text += " ";
123 text += token.text;
124 });
125 return text;

399function resolve(e, scope) {
400 if (e.table !== null) {
401 const part = scope.find((p) => p.name === e.table);
402 if (!part) fail("unknown table " + e.table);
403 const index = part.table.columns.findIndex((c) => c.name === e.column);
404 if (index < 0) fail("unknown column " + e.text);
405 return (e.index = part.offset + index);
406 }
407 const matches = scope.flatMap((p) =>
408 p.table.columns.flatMap((c, i) => (c.name === e.column ? [p.offset + i] : [])),
409 );
410 if (matches.length === 0) fail("unknown column " + e.column);
411 if (matches.length > 1) fail("ambiguous column " + e.column);
412 return (e.index = matches[0]);
413}
414function check(e, scope, allow, grouping, inside = false) {
415 if (e.kind === "literal")
416 return e.value === null ? "NULL" : typeof e.value === "number" ? "INT" : "TEXT";
417 if (e.kind === "reference") {
418 const index = resolve(e, scope);
419 if (grouping !== null && !inside && !grouping.has(index))
420 fail("column " + e.text + " is not grouped");
421 for (const part of scope)
422 if (index >= part.offset && index < part.offset + part.table.columns.length)
423 return part.table.columns[index - part.offset].type;
424 }
425 if (e.kind === "aggregate") {
426 if (!allow || inside) fail("misuse of aggregate " + e.op);
427 if (e.args.length === 0) return "INT";
428 const type = check(e.args[0], scope, true, grouping, true);
429 if ((e.op === "SUM" || e.op === "AVG") && !compatible(type, "INT"))
430 fail("type mismatch for " + e.op);
431 return e.op === "MIN" || e.op === "MAX" ? type : "INT";
432 }
433 const types = e.args.map((a) => check(a, scope, allow, grouping, inside));
434 if (e.kind === "group") return types[0];
435 if (e.kind === "isnull") return "INT";
436 let op = e.op,
437 result = "INT";
438 if (e.kind === "in" || e.kind === "between") {
439 op = e.kind === "in" ? "IN" : "BETWEEN";
440 if (types.slice(1).some((t) => !compatible(t, types[0]))) fail("type mismatch for " + op);
441 } else if (e.kind === "binary" && comparison.has(op)) {
442 if (!compatible(types[0], types[1])) fail("type mismatch for " + op);
443 } else {
444 if (e.kind === "like") op = "LIKE";
445 const required = op === "||" || op === "LIKE" ? "TEXT" : "INT";
446 if (types.some((t) => !compatible(t, required))) fail("type mismatch for " + op);
447 result = op === "||" ? "TEXT" : "INT";
448 }
449 return result;
450}
451function condition(e, scope, allow, grouping, name) {
452 if (!compatible(check(e, scope, allow, grouping), "INT")) fail("type mismatch for " + name);
453}

457function relation(op, a, b) {
458 if (a === null || b === null) return null;
459 const c = compare(a, b);
460 return {
461 "=": c === 0,
462 "<>": c !== 0,
463 "!=": c !== 0,
464 "<": c < 0,
465 "<=": c <= 0,
466 ">": c > 0,
467 ">=": c >= 0,
468 }[op]
469 ? 1
470 : 0;
471}
472function like(text, pattern) {
473 let previous = Array(text.length + 1).fill(false);
474 previous[0] = true;
475 for (const ch of pattern) {
476 const current = Array(text.length + 1).fill(false);
477 current[0] = ch === "%" && previous[0];
478 for (let i = 1; i <= text.length; i++)
479 current[i] =
480 ch === "%"
481 ? previous[i] || current[i - 1]
482 : previous[i - 1] && (ch === "_" || ch === text[i - 1]);
483 previous = current;
484 }
485 return previous[text.length];
486}
487function evaluate(e, row, group) {
488 if (e.kind === "literal") return e.value;
489 if (e.kind === "reference") return row[e.index];
490 if (e.kind === "aggregate") {
491 const rows = group;
492 if (e.args.length === 0) return rows.length;
493 let values = rows.map((r) => evaluate(e.args[0], r, null)).filter((v) => v !== null);
494 if (e.distinct) values = [...new Set(values)];
495 if (e.op === "COUNT") return values.length;
496 if (values.length === 0) return null;
497 if (e.op === "MIN" || e.op === "MAX")
498 return values.reduce((a, b) => (compare(a, b) <= 0 === (e.op === "MIN") ? a : b));
499 const sum = values.reduce((a, b) => a + b, 0);
500 return e.op === "SUM" ? sum : Math.trunc(sum / values.length);
501 }
502 const values = e.args.map((a) => evaluate(a, row, group)),
503 a = values[0],
504 b = values[1];
505 if (e.kind === "group") return a;
506 if (e.kind === "unary") return e.op === "NOT" ? inverse(a) : a === null ? null : -a;
507 if (e.kind === "isnull") return e.op === "NOT" ? (a === null ? 0 : 1) : a === null ? 1 : 0;
508 let result;
509 if (e.kind === "in")
510 result =
511 a === null
512 ? null
513 : values.slice(1).some((v) => v !== null && v === a)
514 ? 1
515 : values.slice(1).includes(null)
516 ? null
517 : 0;
518 else if (e.kind === "between")
519 result = conjunction(relation(">=", a, b), relation("<=", a, values[2]));
520 else if (e.kind === "like") result = a === null || b === null ? null : like(a, b) ? 1 : 0;
521 else {
522 if (comparison.has(e.op)) return relation(e.op, a, b);
523 if (e.op === "AND") return conjunction(a, b);
524 if (e.op === "OR") return disjunction(a, b);
525 if (a === null || b === null) return null;
526 if (e.op === "||") return a + b;
527 const x = a,
528 y = b;
529 switch (e.op) {
530 case "+":
531 return x + y;
532 case "-":
533 return x - y;
534 case "*":
535 return x * y;
536 case "/":
537 return y === 0 ? null : Math.trunc(x / y);
538 case "%":
539 return y === 0 ? null : x % y;
540 }
541 throw new Error("invalid expression");
542 }
543 return e.op === "NOT" ? inverse(result) : result;
544}

587 }

590 const aggregate =
591 q.group !== null ||
592 q.having !== null ||

604 }

606 if (q.having) condition(q.having, scope, true, grouped, "HAVING");
607 for (const item of q.order) {

660 };

668 return 0;
669 });
670 if (q.distinct) {
671 const seen = new Set();
672 results = results.filter((row) => {
673 const id = rowKey(row.values);
674 if (seen.has(id)) return false;
675 seen.add(id);
676 return true;
677 });
678 }
679 results = results.slice(q.offset, q.limit === null ? undefined : q.offset + q.limit);
680 const headings = q.items.map((i) => i.heading),
681 strings = results.map((row) => row.values.map(written));
682 const widths = headings.map((h, i) =>
683 Math.max(h.length, ...strings.map((row) => row[i].length)),
684 );
685 const separator = "+" + widths.map((w) => "-".repeat(w + 2) + "+").join("");
686 const format = (row, header) =>
687 "|" +
688 row
689 .map(
690 (v, i) =>
691 " " +
692 (!header && types[i] === "INT"
693 ? v.padStart(widths[i])
694 : v.padEnd(widths[i])) +
695 " |",
696 )
697 .join("");
698 return [
699 separator,
700 format(headings, true),
701 separator,
702 ...strings.map((row) => format(row, false)),
703 separator,
704 `(${results.length} ${results.length === 1 ? "row" : "rows"})`,
705 ];
706 }
707 execute(c) {
708 if (c.kind === "empty" || c.kind === "exit") return [];
709 if (c.kind === "select") return this.select(c.query);
710 if (c.kind === "create") {
711 if (this.tables.has(c.name)) fail("table " + c.name + " already exists");
712 const seen = new Set();
713 let primary = 0;
714 for (const column of c.columns) {
715 if (seen.has(column.name)) fail("duplicate column " + column.name);
716 seen.add(column.name);
717 primary += column.primary;
718 if (primary >= 2) fail("multiple primary keys");
719 }
720 this.tables.set(c.name, { columns: c.columns, rows: [] });
721 return ["Table " + c.name + " created."];
722 }
723 const table = this.table(c.name);
724 if (c.kind === "drop") {
725 this.tables.delete(c.name);
726 return ["Table " + c.name + " dropped."];
727 }
728 const scope = [{ name: c.name, table, offset: 0 }];
729 let rows, count;
730 if (c.kind === "insert") {
731 const targets =
732 c.names === null ? table.columns.map((_, i) => i) : this.targets(table, c.names);
733 for (const tuple of c.tuples) {
734 if (tuple.length !== targets.length)
735 fail(`expected ${targets.length} values, got ${tuple.length}`);
736 tuple.forEach((expr, i) => {
737 if (!compatible(check(expr, [], false, null), table.columns[targets[i]].type))
738 fail("type mismatch for column " + table.columns[targets[i]].name);
739 });
740 }
741 const appended = c.tuples.map((tuple) => {
742 const row = Array(table.columns.length).fill(null);
743 tuple.forEach((e, i) => {
744 row[targets[i]] = evaluate(e, [], null);
745 });
746 return row;
747 });
748 rows = table.rows.concat(appended);
749 this.constraints(
750 table,
751 rows,
752 appended.map((_, i) => table.rows.length + i),
753 );
754 count = appended.length;
755 } else if (c.kind === "update") {
756 const targets = this.targets(
757 table,
758 c.assignments.map(([name]) => name),
759 );
760 c.assignments.forEach(([, e], i) => {
761 if (!compatible(check(e, scope, false, null), table.columns[targets[i]].type))
762 fail("type mismatch for column " + table.columns[targets[i]].name);
763 });
764 if (c.where) condition(c.where, scope, false, null, "WHERE");
765 const changed = [];
766 rows = table.rows.map((row, i) => {
767 if (c.where && !truth(evaluate(c.where, row, null))) return row;
768 const result = [...row],
769 values = c.assignments.map(([, e]) => evaluate(e, row, null));
770 targets.forEach((column, j) => {
771 result[column] = values[j];
772 });
773 changed.push(i);
774 return result;
775 });
776 this.constraints(table, rows, changed);
777 count = changed.length;
778 } else {
779 if (c.where) condition(c.where, scope, false, null, "WHERE");
780 rows = table.rows.filter(
781 (row) => c.where !== null && !truth(evaluate(c.where, row, null)),
782 );
783 count = table.rows.length - rows.length;
784 }
785 table.rows = rows;
786 return [
787 `${count} ${count === 1 ? "row" : "rows"} ${c.kind === "insert" ? "inserted" : c.kind === "update" ? "updated" : "deleted"}.`,
788 ];
789 }
790}
