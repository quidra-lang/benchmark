25function tokenize(string $text): array
26{
27 $reserved = array_flip(explode(" ", RESERVED));
28 $tokens = [];
29 $i = 0;
30 while ($i < strlen($text)) {
31 $rest = substr($text, $i);
32 if (preg_match('/^(?:[ \n]+|--[^\n]*)/', $rest, $m)) {
33 $i += strlen($m[0]);
34 continue;
35 }
36 if (preg_match("/^'([^']*)'/", $rest, $m)) {
37 $kind = "string";
38 $value = $m[1];
39 } elseif (preg_match("/^[0-9]+/", $rest, $m)) {
40 $kind = "integer";
41 $value = (int) $m[0];
42 } elseif (preg_match("/^[A-Za-z_][A-Za-z0-9_]*/", $rest, $m)) {
43 $upper = strtoupper($m[0]);
44 $kind = isset($reserved[$upper]) ? "keyword" : "name";
45 $value = $kind === "keyword" ? $upper : $m[0];
46 } elseif (preg_match("/^(?:<>|!=|<=|>=|\|\||[(),.;*\/%+\-=<>])/", $rest, $m)) {
47 $kind = "symbol";
48 $value = $m[0];
49 } else {
50 fail("unexpected character '" . $text[$i] . "'");
51 }
52 $tokens[] = ["kind" => $kind, "value" => $value, "text" => $m[0]];
53 $i += strlen($m[0]);
54 }
55 return $tokens;
56}
57final class Parser
58{
59 private int $position = 0;
60 public function __construct(private array $tokens) {}
61 private function peek(): array
62 {
63 return $this->tokens[$this->position];
64 }
65 private function take(): array
66 {
67 return $this->tokens[$this->position++];
68 }
69 private function at(string $word): bool
70 {
71 $t = $this->peek();
72 return in_array($t["kind"], ["symbol", "keyword"], true) && $t["value"] === $word;
73 }
74 private function accept(string $word): bool
75 {
76 if (!$this->at($word)) {
77 return false;
78 }
79 $this->position++;
80 return true;
81 }
82 private function syntax(): never
83 {
84 fail("syntax error at '" . $this->peek()["text"] . "'");
85 }
86 private function expect(string $word): void
87 {
88 if (!$this->accept($word)) {
89 $this->syntax();
90 }

92 private function kind(string $kind): mixed
93 {
94 if ($this->peek()["kind"] !== $kind) {
95 $this->syntax();

97 return $this->take()["value"];

99 private function comma(Closure $read): array
100 {
101 $a = [$read()];
102 while ($this->accept(",")) {
103 $a[] = $read();
104 }
105 return $a;

107 private function parens(Closure $read): array

114 private function alias(): ?string
115 {
116 if ($this->accept("AS")) {
117 return $this->kind("name");

119 return $this->peek()["kind"] === "name" ? $this->take()["value"] : null;

126 private function reference(): Expr

128 $name = $this->kind("name");
129 $e = new Expr("ref");
130 if ($this->accept(".")) {
131 $e->table = $name;
132 $e->column = $this->kind("name");

139 private function expression(int $level = 0): Expr
140 {
141 if ($level === 2 && $this->accept("NOT")) {
142 return new Expr("unary", "NOT", [$this->expression(2)]);
143 }
144 if ($level === 6 && $this->accept("-")) {
145 return new Expr("unary", "-", [$this->expression(6)]);
146 }
147 if ($level === 7) {
148 return $this->primary();
149 }
150 $left = $this->expression($level + 1);
151 if ($level === 3) {
152 if (
153 $this->peek()["kind"] === "symbol" &&
154 in_array($this->peek()["value"], COMPARISONS, true)
155 ) {
156 $op = $this->take()["value"];
157 return new Expr("binary", $op, [$left, $this->expression(4)]);
158 }
159 if ($this->accept("IS")) {
160 $neg = $this->accept("NOT");
161 $this->expect("NULL");
162 $e = new Expr("isnull", "", [$left]);
163 $e->negated = $neg;
164 return $e;
165 }
166 $neg = $this->accept("NOT");
167 $e = null;
168 if ($this->accept("IN")) {
169 $e = new Expr("in", "", [$left, ...$this->parens(fn() => $this->expression())]);
170 } elseif ($this->accept("BETWEEN")) {
171 $low = $this->expression(4);
172 $this->expect("AND");
173 $e = new Expr("between", "", [$left, $low, $this->expression(4)]);
174 } elseif ($this->accept("LIKE")) {
175 $e = new Expr("like", "", [$left, $this->expression(4)]);
176 } elseif ($neg) {
177 $this->syntax();
178 }
179 if ($e) {
180 $e->negated = $neg;
181 return $e;
182 }
183 return $left;
184 }
185 $ops = [
186 0 => ["OR"],
187 1 => ["AND"],
188 2 => [],
189 4 => ["+", "-", "||"],
190 5 => ["*", "/", "%"],
191 6 => [],
192 ][$level];
193 while (
194 in_array($this->peek()["kind"], ["keyword", "symbol"], true) &&
195 in_array($this->peek()["value"], $ops, true)
196 ) {
197 $op = $this->take()["value"];
198 $left = new Expr("binary", $op, [$left, $this->expression($level + 1)]);
199 }
200 return $left;
201 }
202 private function primary(): Expr
203 {
204 $t = $this->peek();
205 if (in_array($t["kind"], ["integer", "string"], true)) {
206 $this->take();
207 return new Expr("literal", value: $t["value"]);
208 }
209 if ($this->accept("NULL")) {
210 return new Expr("literal");
211 }
212 if ($t["kind"] === "name") {
213 return $this->reference();
214 }
215 if ($t["kind"] === "keyword" && in_array($t["value"], AGGREGATES, true)) {
216 $name = $this->take()["value"];
217 $e = new Expr("aggregate", $name);
218 $this->expect("(");
219 if (!($name === "COUNT" && $this->accept("*"))) {
220 $e->distinct = $this->accept("DISTINCT");
221 $e->args = [$this->expression()];
222 }
223 $this->expect(")");
224 return $e;
225 }
226 if ($this->accept("(")) {
227 $e = $this->expression();
228 $this->expect(")");
229 return new Expr("group", "", [$e]);
230 }
231 $this->syntax();
232 }
233 private function clause(string $word): ?Expr
234 {
235 return $this->accept($word) ? $this->expression() : null;
236 }
237 private function item(): array
238 {
239 $start = $this->position;
240 $e = $this->expression();
241 $ts = array_slice($this->tokens, $start, $this->position - $start);
242 $alias = $this->alias();
243 $heading = $alias;
244 if ($heading === null) {
245 if ($e->kind === "ref") {
246 $heading = $e->column;
247 } else {
248 $heading = "";
249 foreach ($ts as $i => $t) {
250 if ($i > 0 && $t["kind"] !== "symbol" && $ts[$i - 1]["kind"] !== "symbol") {
251 $heading .= " ";
252 }
253 $heading .= $t["text"];
254 }
255 }
256 }
257 return ["expr" => $e, "alias" => $alias, "heading" => $heading];
258 }
259 private function orderItem(): array
260 {
261 $start = $this->position;
262 $e = $this->expression();
263 $single = $this->position === $start + 1;
264 $t = $this->tokens[$start];
265 $desc = $this->accept("DESC");
266 if (!$desc) {
267 $this->accept("ASC");
268 }
269 return [
270 "expr" => $e,
271 "desc" => $desc,
272 "ordinal" => $single && $t["kind"] === "integer" ? $t["value"] : null,
273 "alias" => $single && $t["kind"] === "name" ? $t["value"] : null,
274 ];
275 }
276 private function query(): array
277 {
278 $q = [
279 "kind" => "select",
280 "distinct" => $this->accept("DISTINCT"),
281 "star" => $this->accept("*"),
282 "joins" => [],
283 "group" => null,
284 "order" => [],
285 "limit" => null,
286 "offset" => 0,
287 ];
288 $q["items"] = $q["star"] ? [] : $this->comma(fn() => $this->item());
289 $this->expect("FROM");
290 $q["source"] = $this->table();
291 while ($this->at("JOIN") || $this->at("INNER") || $this->at("LEFT")) {
292 $left = $this->accept("LEFT");
293 if ($left) {
294 $this->accept("OUTER");
295 } else {
296 $this->accept("INNER");
297 }
298 $this->expect("JOIN");
299 $t = $this->table();
300 $this->expect("ON");
301 $q["joins"][] = ["table" => $t, "left" => $left, "condition" => $this->expression()];
302 }
303 $q["where"] = $this->clause("WHERE");
304 if ($this->accept("GROUP")) {
305 $this->expect("BY");
306 $q["group"] = $this->comma(fn() => $this->reference());
307 }
308 $q["having"] = $this->clause("HAVING");
309 if ($this->accept("ORDER")) {
310 $this->expect("BY");
311 $q["order"] = $this->comma(fn() => $this->orderItem());
312 }
313 if ($this->accept("LIMIT")) {
314 $q["limit"] = $this->kind("integer");
315 if ($this->accept("OFFSET")) {
316 $q["offset"] = $this->kind("integer");
317 }
318 }
319 return $q;
320 }
321 public function statement(): ?array
322 {
323 $s = null;
324 if ($this->accept("CREATE")) {
325 $this->expect("TABLE");
326 $name = $this->kind("name");
327 $columns = $this->parens(function () {
328 $n = $this->kind("name");
329 if (!$this->at("INT") && !$this->at("TEXT")) {
330 $this->syntax();
331 }
332 $type = $this->take()["value"];
333 $pk = 0;
334 $nn = false;
335 while (true) {
336 if ($this->accept("PRIMARY")) {
337 $this->expect("KEY");
338 $pk++;
339 $nn = true;
340 } elseif ($this->accept("NOT")) {
341 $this->expect("NULL");
342 $nn = true;
343 } else {
344 break;
345 }
346 }
347 return ["name" => $n, "type" => $type, "primary" => $pk, "notNull" => $nn];
348 });
349 $s = ["kind" => "create", "name" => $name, "columns" => $columns];
350 } elseif ($this->accept("DROP")) {
351 $this->expect("TABLE");
352 $s = ["kind" => "drop", "name" => $this->kind("name")];
353 } elseif ($this->accept("INSERT")) {
354 $this->expect("INTO");
355 $name = $this->kind("name");
356 $names = $this->at("(") ? $this->parens(fn() => $this->kind("name")) : null;
357 $this->expect("VALUES");
358 $s = [
359 "kind" => "insert",
360 "name" => $name,
361 "names" => $names,
362 "tuples" => $this->comma(fn() => $this->parens(fn() => $this->expression())),
363 ];
364 } elseif ($this->accept("UPDATE")) {
365 $name = $this->kind("name");
366 $this->expect("SET");
367 $as = $this->comma(function () {
368 $n = $this->kind("name");
369 $this->expect("=");
370 return [$n, $this->expression()];
371 });
372 $s = [
373 "kind" => "update",
374 "name" => $name,
375 "assignments" => $as,
376 "where" => $this->clause("WHERE"),
377 ];
378 } elseif ($this->accept("DELETE")) {
379 $this->expect("FROM");
380 $s = [
381 "kind" => "delete",
382 "name" => $this->kind("name"),
383 "where" => $this->clause("WHERE"),
384 ];
385 } elseif ($this->accept("SELECT")) {
386 $s = $this->query();
387 } elseif ($this->accept("EXIT")) {
388 $s = ["kind" => "exit"];
389 }
390 $this->expect(";");
391 return $s;
392 }

426 public function resolve(Expr $e): string
427 {
428 $found = [];
429 $seen = false;
430 foreach ($this->entries as $entry) {

462function check(
463 Expr $e,
464 Scope $scope,
465 bool $allow = false,
466 ?array $grouped = null,
467 bool $inside = false,
468): string {
469 if ($e->kind === "literal") {
470 return $e->value === null ? "NULL" : (is_int($e->value) ? "INT" : "TEXT");
471 }
472 if ($e->kind === "ref") {
473 $t = $scope->resolve($e);
474 if ($grouped !== null && !$inside && !in_array($e->index, $grouped, true)) {
475 fail("column " . $e->text . " is not grouped");
476 }
477 return $t;
478 }
479 if ($e->kind === "aggregate") {
480 if (!$allow || $inside) {
481 fail("misuse of aggregate " . $e->op);
482 }
483 if (!$e->args) {
484 return "INT";
485 }
486 $t = check($e->args[0], $scope, $allow, $grouped, true);
487 if (in_array($e->op, ["SUM", "AVG"], true)) {
488 requireType($t, "INT", $e->op);
489 }
490 return in_array($e->op, ["MIN", "MAX"], true) ? $t : "INT";
491 }
492 $types = [];
493 foreach ($e->args as $a) {
494 $types[] = check($a, $scope, $allow, $grouped, $inside);
495 }
496 if ($e->kind === "group") {
497 return $types[0];
498 }
499 if ($e->kind === "isnull") {
500 return "INT";
501 }
502 if ($e->kind === "binary" && in_array($e->op, COMPARISONS, true)) {
503 requireType($types[0], $types[1], $e->op);
504 return "INT";
505 }
506 if ($e->kind === "in" || $e->kind === "between") {
507 foreach (array_slice($types, 1) as $t) {
508 requireType($t, $types[0], $e->kind === "in" ? "IN" : "BETWEEN");
509 }
510 return "INT";
511 }
512 $op = $e->kind === "like" ? "LIKE" : $e->op;
513 $type = in_array($op, ["LIKE", "||"], true) ? "TEXT" : "INT";
514 foreach ($types as $t) {
515 requireType($t, $type, $op);
516 }
517 return $op === "||" ? "TEXT" : "INT";
518}

550function operate(string $op, int|string|null $a, int|string|null $b): int|string|null
551{
552 if ($op === "AND") {
553 return $a === 0 || $b === 0 ? 0 : ($a === null || $b === null ? null : 1);
554 }
555 if ($op === "OR") {
556 return truth($a) || truth($b) ? 1 : ($a === null || $b === null ? null : 0);
557 }
558 if ($a === null || $b === null) {
559 return null;
560 }
561 return match ($op) {
562 "+" => $a + $b,
563 "-" => $a - $b,
564 "*" => $a * $b,
565 "||" => $a . $b,
566 "/" => $b === 0 ? null : intdiv($a, $b),
567 "%" => $b === 0 ? null : $a - $b * intdiv($a, $b),
568 "=" => $a === $b ? 1 : 0,
569 "<>", "!=" => $a !== $b ? 1 : 0,
570 "<" => compareValues($a, $b) < 0 ? 1 : 0,
571 "<=" => compareValues($a, $b) <= 0 ? 1 : 0,
572 ">" => compareValues($a, $b) > 0 ? 1 : 0,
573 ">=" => compareValues($a, $b) >= 0 ? 1 : 0,
574 };

576function likeMatch(string $text, string $pattern): int
577{
578 $dp = array_fill(0, strlen($text) + 1, false);
579 $dp[0] = true;
580 for ($j = 0; $j < strlen($pattern); $j++) {
581 $c = $pattern[$j];
582 if ($c === "%") {
583 for ($i = 1; $i <= strlen($text); $i++) {
584 $dp[$i] = $dp[$i] || $dp[$i - 1];
585 }
586 } else {
587 for ($i = strlen($text); $i > 0; $i--) {
588 $dp[$i] = $dp[$i - 1] && ($c === "_" || $c === $text[$i - 1]);
589 }
590 $dp[0] = false;
591 }
592 }
593 return $dp[strlen($text)] ? 1 : 0;
594}
595function evaluate(Expr $e, array $row, ?array $group = null): int|string|null
596{
597 if ($e->kind === "literal") {
598 return $e->value;
599 }
600 if ($e->kind === "ref") {
601 return $row[$e->index];
602 }
603 if ($e->kind === "aggregate") {
604 if (!$e->args) {
605 return count($group);
606 }
607 $values = [];
608 foreach ($group as $r) {
609 $v = evaluate($e->args[0], $r);
610 if ($v !== null && (!$e->distinct || !in_array($v, $values, true))) {
611 $values[] = $v;
612 }
613 }
614 if ($e->op === "COUNT") {
615 return count($values);
616 }
617 if (!$values) {
618 return null;
619 }
620 return match ($e->op) {
621 "SUM" => array_sum($values),
622 "AVG" => intdiv(array_sum($values), count($values)),
623 "MIN" => array_reduce(
624 array_slice($values, 1),
625 fn($a, $b) => compareValues($a, $b) <= 0 ? $a : $b,
626 $values[0],
627 ),
628 "MAX" => array_reduce(
629 array_slice($values, 1),
630 fn($a, $b) => compareValues($a, $b) >= 0 ? $a : $b,
631 $values[0],
632 ),
633 };
634 }
635 $a = [];
636 foreach ($e->args as $arg) {
637 $a[] = evaluate($arg, $row, $group);
638 }
639 if ($e->kind === "group") {
640 return $a[0];
641 }
642 if ($e->kind === "unary") {
643 return $e->op === "NOT" ? inverse($a[0]) : ($a[0] === null ? null : -$a[0]);
644 }
645 if ($e->kind === "binary") {
646 return operate($e->op, $a[0], $a[1]);
647 }
648 if ($e->kind === "isnull") {
649 return ($a[0] === null) !== $e->negated ? 1 : 0;
650 }
651 $result = match ($e->kind) {
652 "in" => $a[0] === null
653 ? null
654 : (in_array($a[0], array_slice($a, 1), true)
655 ? 1
656 : (in_array(null, array_slice($a, 1), true)
657 ? null
658 : 0)),
659 "between" => operate("AND", operate(">=", $a[0], $a[1]), operate("<=", $a[0], $a[2])),
660 "like" => $a[0] === null || $a[1] === null ? null : likeMatch($a[0], $a[1]),
661 };
662 return $e->negated ? inverse($result) : $result;
663}

689function printTable(array $items, array $types, array $rows): void
690{
691 $headings = array_column($items, "heading");
692 $cells = array_map(fn($r) => array_map("written", $r), $rows);
693 $widths = array_map("strlen", $headings);
694 foreach ($cells as $r) {
695 foreach ($r as $i => $v) {
696 $widths[$i] = max($widths[$i], strlen($v));
697 };
698 }
699 $sep = "+";
700 foreach ($widths as $w) {
701 $sep .= str_repeat("-", $w + 2) . "+";
702 }
703 $render = function ($row, $header) use ($widths, $types) {
704 $s = "|";
705 foreach ($row as $i => $v) {
706 $s .=
707 " " .
708 str_pad(
709 $v,
710 $widths[$i],
711 " ",
712 !$header && $types[$i] === "INT" ? STR_PAD_LEFT : STR_PAD_RIGHT,
713 ) .
714 " |";
715 }
716 return $s;
717 };
718 echo $sep, "\n", $render($headings, true), "\n", $sep, "\n";
719 foreach ($cells as $r) {
720 echo $render($r, false), "\n";
721 }
722 echo $sep, "\n(", rowCount(count($rows)), ")\n";
723}

729 if (!isset($this->tables[$name])) {
730 fail("unknown table " . $name);
731 }
732 return $this->tables[$name];
733 }
734 private function targets(Table $t, array $names): array
735 {
736 $result = [];
737 foreach ($names as $n) {
738 $i = $t->column($n);
739 if ($i === null) {
740 fail("unknown column " . $n);
741 }
742 if (in_array($i, $result, true)) {
743 fail("duplicate column " . $n);
744 }
745 $result[] = $i;
746 }
747 return $result;
748 }
749 private function constraints(Table $t, array $rows, array $checked): void
750 {
751 foreach ($checked as $i) {
752 foreach ($t->columns as $j => $c) {
753 if ($c["notNull"] && $rows[$i][$j] === null) {
754 fail("NULL value in column " . $c["name"]);
755 }
756 }
757 if ($t->primary !== null) {
758 foreach ($rows as $j => $r) {
759 if ($i !== $j && $r[$t->primary] === $rows[$i][$t->primary]) {
760 fail("duplicate primary key " . written($r[$t->primary]));
761 }
762 };
763 }
764 }

766 public function execute(array $s): void
767 {
768 if ($s["kind"] === "select") {
769 $this->select($s);
770 return;
771 }
772 if ($s["kind"] === "create") {
773 if (isset($this->tables[$s["name"]])) {
774 fail("table " . $s["name"] . " already exists");
775 }
776 $seen = [];
777 $primary = 0;
778 $key = null;
779 foreach ($s["columns"] as $i => $c) {
780 if (isset($seen[$c["name"]])) {
781 fail("duplicate column " . $c["name"]);
782 }
783 $seen[$c["name"]] = true;
784 $primary += $c["primary"];
785 if ($primary >= 2) {
786 fail("multiple primary keys");
787 }
788 if ($c["primary"] > 0) {
789 $key = $i;
790 }
791 }
792 $this->tables[$s["name"]] = new Table($s["name"], $s["columns"], $key);
793 echo "Table ", $s["name"], " created.\n";
794 return;
795 }
796 $t = $this->table($s["name"]);
797 if ($s["kind"] === "drop") {
798 unset($this->tables[$s["name"]]);
799 echo "Table ", $s["name"], " dropped.\n";
800 return;
801 }
802 $rows = [];
803 $checked = [];
804 $count = 0;
805 if ($s["kind"] === "insert") {
806 $targets =
807 $s["names"] === null ? array_keys($t->columns) : $this->targets($t, $s["names"]);
808 foreach ($s["tuples"] as $tuple) {
809 if (count($tuple) !== count($targets)) {
810 fail("expected " . count($targets) . " values, got " . count($tuple));
811 }
812 foreach ($tuple as $i => $e) {
813 $c = $t->columns[$targets[$i]];
814 requireType(check($e, new Scope()), $c["type"], "column " . $c["name"]);
815 }
816 }
817 $rows = $t->rows;
818 foreach ($s["tuples"] as $tuple) {
819 $row = array_fill(0, count($t->columns), null);
820 foreach ($tuple as $i => $e) {
821 $row[$targets[$i]] = evaluate($e, []);
822 }
823 $checked[] = count($rows);
824 $rows[] = $row;
825 }
826 $this->constraints($t, $rows, $checked);
827 $count = count($s["tuples"]);
828 } elseif ($s["kind"] === "update") {
829 $targets = $this->targets($t, array_column($s["assignments"], 0));
830 $scope = new Scope();
831 $scope->add($t->name, $t);
832 foreach ($s["assignments"] as $i => $a) {
833 $c = $t->columns[$targets[$i]];
834 requireType(check($a[1], $scope), $c["type"], "column " . $c["name"]);
835 }
836 condition($s["where"], $scope, "WHERE");
837 foreach ($t->rows as $i => $row) {
838 $next = $row;
839 if (satisfies($s["where"], $row)) {
840 foreach ($s["assignments"] as $j => $a) {
841 $next[$targets[$j]] = evaluate($a[1], $row);
842 }
843 $checked[] = $i;
844 $count++;
845 }
846 $rows[] = $next;
847 }
848 $this->constraints($t, $rows, $checked);
849 } else {
850 $scope = new Scope();
851 $scope->add($t->name, $t);
852 condition($s["where"], $scope, "WHERE");
853 foreach ($t->rows as $row) {
854 if (satisfies($s["where"], $row)) {
855 $count++;
856 } else {
857 $rows[] = $row;
858 }
859 }
860 }
861 $t->rows = $rows;
862 echo rowCount($count),
863 match ($s["kind"]) {
864 "insert" => " inserted.",
865 "update" => " updated.",
866 "delete" => " deleted.",
867 },
868 "\n";
869 }
870 private function select(array $q): void
871 {
872 $scope = new Scope();
873 $scope->add($q["source"]["alias"], $this->table($q["source"]["name"]));
874 foreach ($q["joins"] as $j) {
875 $scope->add($j["table"]["alias"], $this->table($j["table"]["name"]));
876 condition($j["condition"], $scope, "ON");
877 }
878 condition($q["where"], $scope, "WHERE");
879 $grouping = [];
880 foreach ($q["group"] ?? [] as $ref) {
881 $scope->resolve($ref);
882 $grouping[] = $ref->index;
883 }
884 $items = $q["items"];
885 if ($q["star"]) {
886 foreach ($scope->entries as $entry) {
887 foreach ($entry["table"]->columns as $c) {
888 $e = new Expr("ref");
889 $e->table = $entry["name"];
890 $e->column = $e->text = $c["name"];
891 $items[] = ["expr" => $e, "heading" => $c["name"], "alias" => null];
892 };
893 };
894 }
895 $aggregate = $q["group"] !== null || $q["having"] !== null;
896 foreach ([...$items, ...$q["order"]] as $i) {
897 $aggregate = $aggregate || hasAggregate($i["expr"]);
898 }
899 $grouped = $aggregate ? $grouping : null;
900 $types = [];
901 foreach ($items as $i) {
902 $types[] = check($i["expr"], $scope, true, $grouped);
903 }
904 condition($q["having"], $scope, "HAVING", true, $grouped);
905 $keys = [];
906 foreach ($q["order"] as $o) {
907 $key = null;
908 if ($o["ordinal"] !== null) {
909 $k = $o["ordinal"];
910 if ($k < 1 || $k > count($items)) {
911 fail("ORDER BY position " . $k . " is out of range");
912 }
913 $key = $items[$k - 1]["expr"];
914 } elseif ($o["alias"] !== null) {
915 foreach ($items as $i) {
916 if ($i["alias"] === $o["alias"]) {
917 $key = $i["expr"];
918 break;
919 }
920 };
921 }
922 if ($key === null) {
923 check($o["expr"], $scope, true, $grouped);
924 $key = $o["expr"];
925 }
926 $keys[] = $key;
927 }
928 $rows = $scope->entries[0]["table"]->rows;
929 foreach ($q["joins"] as $ji => $join) {
930 $t = $scope->entries[$ji + 1]["table"];
931 $next = [];
932 foreach ($rows as $left) {
933 $found = false;
934 foreach ($t->rows as $right) {
935 $combined = [...$left, ...$right];
936 if (satisfies($join["condition"], $combined)) {
937 $next[] = $combined;
938 $found = true;
939 }
940 }
941 if ($join["left"] && !$found) {
942 $next[] = [...$left, ...array_fill(0, count($t->columns), null)];
943 }
944 }
945 $rows = $next;
946 }
947 $rows = array_values(array_filter($rows, fn($r) => satisfies($q["where"], $r)));
948 $groups = [];
949 if (!$aggregate) {
950 foreach ($rows as $r) {
951 $groups[] = [$r];
952 };
953 } elseif ($q["group"] === null) {
954 $groups = [$rows];
955 } else {
956 $groupKeys = [];
957 foreach ($rows as $r) {
958 $key = array_map(fn($i) => $r[$i], $grouping);
959 $at = array_search($key, $groupKeys, true);
960 if ($at === false) {
961 $at = count($groups);
962 $groups[] = [];
963 $groupKeys[] = $key;
964 }
965 $groups[$at][] = $r;
966 }
967 }
968 $results = [];
969 foreach ($groups as $g) {
970 $row = $g[0] ?? [];
971 $context = $aggregate ? $g : null;
972 if (!satisfies($q["having"], $row, $context)) {
973 continue;
974 }
975 $values = [];
976 $sort = [];
977 foreach ($items as $i) {
978 $values[] = evaluate($i["expr"], $row, $context);
979 }
980 foreach ($keys as $key) {
981 $sort[] = evaluate($key, $row, $context);
982 }
983 $results[] = ["values" => $values, "keys" => $sort, "position" => count($results)];
984 }
985 usort($results, function ($a, $b) use ($q) {
986 foreach ($q["order"] as $i => $o) {
987 $c = compareValues($a["keys"][$i], $b["keys"][$i]);
988 if ($c !== 0) {
989 return $o["desc"] ? -$c : $c;
990 }
991 }
992 return $a["position"] <=> $b["position"];
993 });
994 $unique = [];
995 foreach ($results as $r) {
996 if (!$q["distinct"] || !in_array($r["values"], $unique, true)) {
997 $unique[] = $r["values"];
998 }
999 }
1000 $rows = array_slice($unique, $q["offset"], $q["limit"]);
1001 printTable($items, $types, $rows);
1002 }
1003}
1004$source = stream_get_contents(STDIN);
1005$start = 0;
1006$segments = [];
1007for ($i = 0; $i < strlen($source); $i++) {
1008 if ($source[$i] === "'") {
1009 $i = strpos($source, "'", $i + 1);
1010 } elseif (substr($source, $i, 2) === "--") {

1013 } elseif ($source[$i] === ";") {
1014 $segments[] = substr($source, $start, $i - $start + 1);
1015 $start = $i + 1;
1016 }
1017}

1029 echo "ERROR: ", $err->getMessage(), "\n";
