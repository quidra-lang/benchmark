18 static List<Token> tokenize(String s) {
19 List<Token>a = new ArrayList<>();
20 int i = 0;
21 while (i<s.length()) {
22 char c = s.charAt(i);
23 int start = i;
24 if (c == ' ' || c == '\n') {
25 i++;
26 continue;
27 }
28 if (s.startsWith("--", i)) {
29 int end = s.indexOf('\n', i);
30 i = end<0?s.length():end;
31 continue;
32 }
33 if (c == '\'') {
34 i = s.indexOf('\'', i+1)+1;
35 a.add(new Token("string", s.substring(start, i), s.substring(start+1, i-1)));
36 }
37 else if (c >= '0' && c <= '9') {
38 while (i<s.length() && s.charAt(i) >= '0' && s.charAt(i) <= '9') i++;
39 String t = s.substring(start, i);
40 a.add(new Token("integer", t, Long.parseLong(t)));
41 }
42 else if (Character.isLetter(c) || c == '_') {
43 while (i<s.length() && (Character.isLetterOrDigit(s.charAt(i)) || s.charAt(i) == '_')) i++;
44 String t = s.substring(start, i), up = t.toUpperCase(Locale.ROOT);
45 a.add(new Token(RESERVED.contains(up)?"keyword":"name", t, RESERVED.contains(up)?up:t));
46 }
47 else {
48 String t = s.substring(i, Math.min(i+2, s.length()));
49 if (!Set.of("<>", "!=", "<=", ">=", "||").contains(t)) t = String.valueOf(c);
50 if (t.length() == 1 && "(),.;*/%+-=<>".indexOf(c)<0) throw error("unexpected character '"+c+"'");
51 a.add(new Token("symbol", t, t));
52 i += t.length();
53 }
54 }
55 return a;
56 }

164 Expr reference() {
165 String name = name();
166 Expr e = node("ref");
167 if (accept(".")) {
168 e.table = name;
169 e.column = name();
170 e.text = name+"."+e.column;
171 } else e.column = e.text = name;
172 return e;
173 }
174 Expr expr() {
175 return expr(0);
176 }
177 Expr expr(int level) {
178 if (level == 2 && accept("NOT")) return node("unary", "NOT", List.of(expr(2)));
179 if (level == 6 && accept("-")) return node("unary", "-", List.of(expr(6)));
180 if (level == 7) return primary();
181 Expr left = expr(level+1);
182 if (level == 3) {
183 if (peek().kind.equals("symbol") && COMPARISONS.contains(peek().value)) {
184 String op = (String) take().value;
185 return node("binary", op, List.of(left, expr(4)));
186 }
187 if (accept("IS")) {
188 boolean neg = accept("NOT");
189 expect("NULL");
190 Expr e = node("isnull", "", List.of(left));
191 e.negated = neg;
192 return e;
193 }
194 boolean neg = accept("NOT");
195 Expr e = null;
196 if (accept("IN")) {
197 List<Expr>a = new ArrayList<>();
198 a.add(left);
199 a.addAll(parens(this::expr));
200 e = node("in", "", a);
201 } else if (accept("BETWEEN")) {
202 Expr low = expr(4);
203 expect("AND");
204 e = node("between", "", List.of(left, low, expr(4)));
205 } else if (accept("LIKE")) e = node("like", "", List.of(left, expr(4)));
206 else if (neg) throw syntax();
207 if (e != null) {
208 e.negated = neg;
209 return e;
210 }
211 return left;
212 }
213 List<List<String>>ops = List.of(List.of("OR"), List.of("AND"), List.of(), List.of(), List.of("+", "-", "||"), List.of("*", "/", "%"), List.of());
214 while (ops.get(level).stream().anyMatch(this::at)) {
215 String op = (String) take().value;
216 left = node("binary", op, List.of(left, expr(level+1)));
217 }
218 return left;
219 }
220 Expr primary() {
221 Token t = peek();
222 if (t.kind.equals("integer") || t.kind.equals("string")) {
223 take();
224 return node("literal", "", List.of(), t.value);
225 }
226 if (accept("NULL")) return node("literal");
227 if (t.kind.equals("name")) return reference();
228 if (t.kind.equals("keyword") && AGGREGATES.contains(t.value)) {
229 String n = (String) take().value;
230 expect("(");
231 Expr e = node("aggregate", n, List.of());
232 if (!(n.equals("COUNT") && accept("*"))) {
233 e.distinct = accept("DISTINCT");
234 e.args.add(expr());
235 }
236 expect(")");
237 return e;
238 }
239 if (accept("(")) {
240 Expr e = expr();
241 expect(")");
242 return node("group", "", List.of(e));
243 }
244 throw syntax();
245 }
246 Expr clause(String w) {
247 return accept(w)?expr():null;
248 }
249 Item item() {
250 int start = position;
251 Expr e = expr();
252 int stop = position;
253 String alias = alias(), heading = alias;
254 if (heading == null) {
255 if (e.kind.equals("ref")) heading = e.column;
256 else {
257 StringBuilder s = new StringBuilder();
258 for (int i = start; i<stop; i++) {
259 if (i>start && !tokens.get(i).kind.equals("symbol") && !tokens.get(i-1).kind.equals("symbol")) s.append(' ');
260 s.append(tokens.get(i).text);
261 }
262 heading = s.toString();
263 }
264 }
265 return new Item(e, alias, heading);
266 }
267 Order orderItem() {
268 int start = position;
269 Expr e = expr();
270 boolean single = position == start+1;
271 Token t = tokens.get(start);
272 boolean desc = accept("DESC");
273 if (!desc) accept("ASC");
274 return new Order(e, desc, single && t.kind.equals("integer")?(Long) t.value:null, single && t.kind.equals("name")?(String) t.value:null);
275 }
276 Query query() {
277 Query q = new Query();
278 q.distinct = accept("DISTINCT");
279 q.star = accept("*");
280 if (!q.star) q.items = comma(this::item);
281 expect("FROM");
282 q.source = table();
283 while (at("JOIN") || at("INNER") || at("LEFT")) {
284 boolean left = accept("LEFT");
285 if (left) accept("OUTER");
286 else accept("INNER");
287 expect("JOIN");
288 TableRef t = table();
289 expect("ON");
290 q.joins.add(new Join(t, left, expr()));
291 }
292 q.where = clause("WHERE");
293 if (accept("GROUP")) {
294 expect("BY");
295 q.group = comma(this::reference);
296 }
297 q.having = clause("HAVING");
298 if (accept("ORDER")) {
299 expect("BY");
300 q.order = comma(this::orderItem);
301 }
302 if (accept("LIMIT")) {
303 q.limit = integer();
304 if (accept("OFFSET")) q.offset = integer();
305 }
306 return q;
307 }
308 Column column() {
309 String n = name();
310 if (!at("INT") && !at("TEXT")) throw syntax();
311 String type = (String) take().value;
312 int pk = 0;
313 boolean nn = false;
314 while (true) {
315 if (accept("PRIMARY")) {
316 expect("KEY");
317 pk++;
318 nn = true;
319 } else if (accept("NOT")) {
320 expect("NULL");
321 nn = true;
322 } else break;
323 }
324 return new Column(n, type, pk, nn);
325 }

327 Command c = new Command();
328 if (accept("CREATE")) {
329 c.kind = "create";
330 expect("TABLE");
331 c.name = name();
332 c.columns = parens(this::column);
333 } else if (accept("DROP")) {
334 c.kind = "drop";
335 expect("TABLE");
336 c.name = name();
337 } else if (accept("INSERT")) {
338 c.kind = "insert";
339 expect("INTO");
340 c.name = name();
341 if (at("(")) c.names = parens(this::name);
342 expect("VALUES");
343 c.tuples = comma(() -> parens(this::expr));
344 } else if (accept("UPDATE")) {

346 c.name = name();
347 expect("SET");
348 c.assignments = comma(() -> {
349 String n = name();
350 expect("=");
351 return Map.entry(n, expr());
352 });
353 c.where = clause("WHERE");
354 } else if (accept("DELETE")) {
355 c.kind = "delete";
356 expect("FROM");
357 c.name = name();
358 c.where = clause("WHERE");
359 } else if (accept("SELECT")) {
360 c.kind = "select";
361 c.query = query();
362 } else if (accept("EXIT")) c.kind = "exit";
363 expect(";");
364 return c;
365 }

413 static String check(Expr e, Scope s) {
414 return check(e, s, false, null, false);

416 static String check(Expr e, Scope s, boolean allow, List<Integer>grouped, boolean inside) {
417 if (e.kind.equals("literal")) return e.value == null?"NULL":e.value instanceof Long?"INT":"TEXT";
418 if (e.kind.equals("ref")) {
419 String type = s.resolve(e);
420 if (grouped != null && !inside && !grouped.contains(e.index)) throw error("column "+e.text+" is not grouped");
421 return type;
422 }
423 if (e.kind.equals("aggregate")) {
424 if (!allow || inside) throw error("misuse of aggregate "+e.op);
425 if (e.args.isEmpty()) return "INT";
426 String type = check(e.args.get(0), s, allow, grouped, true);
427 if (e.op.equals("SUM") || e.op.equals("AVG")) require(type, "INT", e.op);
428 return e.op.equals("MIN") || e.op.equals("MAX")?type:"INT";
429 }
430 List<String>types = new ArrayList<>();
431 for (Expr a:e.args) types.add(check(a, s, allow, grouped, inside));
432 if (e.kind.equals("group")) return types.get(0);
433 if (e.kind.equals("isnull")) return "INT";
434 if (e.kind.equals("binary") && COMPARISONS.contains(e.op)) {
435 require(types.get(0), types.get(1), e.op);
436 return "INT";
437 }
438 if (e.kind.equals("in") || e.kind.equals("between")) {
439 for (String t:types.subList(1, types.size())) require(t, types.get(0), e.kind.equals("in")?"IN":"BETWEEN");
440 return "INT";
441 }
442 String op = e.kind.equals("like")?"LIKE":e.op, type = op.equals("LIKE") || op.equals("||")?"TEXT":"INT";
443 for (String t:types) require(t, type, op);
444 return op.equals("||")?"TEXT":"INT";
445 }

461 static int compare(Object a, Object b) {
462 if (Objects.equals(a, b)) return 0;
463 if (a == null) return -1;
464 if (b == null) return 1;
465 return a instanceof Long?Long.compare((Long) a, (Long) b):((String) a).compareTo((String) b);
466 }
467 static Object operate(String op, Object a, Object b) {
468 if (op.equals("AND")) {
469 if (Objects.equals(a, 0L) || Objects.equals(b, 0L)) return 0L;
470 return a == null || b == null?null:1L;
471 }
472 if (op.equals("OR")) {
473 if (truth(a) || truth(b)) return 1L;
474 return a == null || b == null?null:0L;
475 }
476 if (a == null || b == null) return null;
477 if (COMPARISONS.contains(op)) {
478 int c = compare(a, b);
479 return (switch (op) {
480 case "=" -> c == 0;
481 case "<>", "!=" -> c != 0;
482 case "<" -> c<0;
483 case "<=" -> c <= 0;
484 case ">" -> c>0;
485 default -> c >= 0;
486 })?1L:0L;
487 }
488 if (op.equals("||")) return (String) a+(String) b;
489 long x = (Long) a, y = (Long) b;
490 return switch (op) {
491 case "+" -> x+y;
492 case "-" -> x-y;
493 case "*" -> x*y;
494 case "/" -> y == 0?null:x/y;
495 case "%" -> y == 0?null:x%y;
496 default -> throw new IllegalStateException();
497 };
498 }
499 static boolean like(String text, String pattern) {
500 boolean[]dp = new boolean[text.length()+1];
501 dp[0] = true;
502 for (char c:pattern.toCharArray()) {
503 if (c == '%') {
504 for (int i = 1; i <= text.length(); i++) dp[i] = dp[i] || dp[i-1];
505 } else {
506 for (int i = text.length(); i>0; i--) dp[i] = dp[i-1] && (c == '_' || c == text.charAt(i-1));
507 dp[0] = false;
508 }
509 }
510 return dp[text.length()];
511 }
512 static Object evaluate(Expr e, List<Object>row) {
513 return evaluate(e, row, null);
514 }
515 static Object evaluate(Expr e, List<Object>row, List<List<Object>>group) {
516 if (e.kind.equals("literal")) return e.value;
517 if (e.kind.equals("ref")) return row.get(e.index);
518 if (e.kind.equals("aggregate")) {
519 if (e.args.isEmpty()) return (long) group.size();
520 List<Object>values = new ArrayList<>();
521 for (List<Object>r:group) {
522 Object v = evaluate(e.args.get(0), r);
523 if (v != null && (!e.distinct || !values.contains(v))) values.add(v);
524 }
525 if (e.op.equals("COUNT")) return (long) values.size();
526 if (values.isEmpty()) return null;
527 if (e.op.equals("MIN") || e.op.equals("MAX")) {
528 Object result = values.get(0);
529 for (Object v:values) if (e.op.equals("MIN")?compare(v, result)<0:compare(v, result)>0) result = v;
530 return result;
531 }
532 long sum = 0;
533 for (Object v:values) sum += (Long) v;
534 return e.op.equals("AVG")?sum/values.size():sum;
535 }
536 List<Object>a = new ArrayList<>();
537 for (Expr arg:e.args) a.add(evaluate(arg, row, group));
538 if (e.kind.equals("group")) return a.get(0);
539 if (e.kind.equals("unary")) return e.op.equals("NOT")?inverse(a.get(0)):a.get(0) == null?null:-(Long) a.get(0);
540 if (e.kind.equals("binary")) return operate(e.op, a.get(0), a.get(1));
541 if (e.kind.equals("isnull")) return (a.get(0) == null) != e.negated?1L:0L;
542 Object result = null;
543 if (e.kind.equals("in")) {
544 if (a.get(0) != null) result = a.subList(1, a.size()).contains(a.get(0))?1L:a.subList(1, a.size()).contains(null)?null:0L;
545 }
546 else if (e.kind.equals("between")) result = operate("AND", operate(">=", a.get(0), a.get(1)), operate("<=", a.get(0), a.get(2)));
547 else if (a.get(0) != null && a.get(1) != null) result = like((String) a.get(0), (String) a.get(1))?1L:0L;
548 return e.negated?inverse(result):result;
549 }

581 static String formatRow(List<String>row, int[]width, List<String>types, boolean header) {

612 void execute(Command c) {
613 if (c.kind.equals("select")) {
614 select(c.query);
615 return;
616 }
617 if (c.kind.equals("create")) {
618 if (tables.containsKey(c.name)) throw error("table "+c.name+" already exists");
619 Set<String>seen = new HashSet<>();
620 int primary = 0, key = -1;
621 for (int i = 0; i<c.columns.size(); i++) {
622 Column column = c.columns.get(i);
623 if (!seen.add(column.name)) throw error("duplicate column "+column.name);
624 primary += column.primary;
625 if (primary >= 2) throw error("multiple primary keys");
626 if (column.primary>0) key = i;
627 }
628 tables.put(c.name, new Table(c.name, c.columns, key));
629 System.out.println("Table "+c.name+" created.");
630 return;
631 }
632 Table t = table(c.name);
633 if (c.kind.equals("drop")) {
634 tables.remove(c.name);
635 System.out.println("Table "+c.name+" dropped.");
636 return;
637 }
638 List<List<Object>>rows = new ArrayList<>();
639 List<Integer>checked = new ArrayList<>();
640 int count = 0;
641 if (c.kind.equals("insert")) {
642 List<Integer>target;
643 if (c.names != null) target = targets(t, c.names);
644 else {
645 target = new ArrayList<>();
646 for (int i = 0; i<t.columns.size(); i++) target.add(i);
647 }
648 Scope empty = new Scope();
649 for (List<Expr>tuple:c.tuples) {
650 if (tuple.size() != target.size()) throw error("expected "+target.size()+" values, got "+tuple.size());
651 for (int i = 0; i<tuple.size(); i++) {
652 Column column = t.columns.get(target.get(i));
653 require(check(tuple.get(i), empty), column.type, "column "+column.name);
654 }
655 }
656 rows.addAll(t.rows);
657 for (List<Expr>tuple:c.tuples) {
658 List<Object>row = new ArrayList<>(Collections.nCopies(t.columns.size(), null));
659 for (int i = 0; i<tuple.size(); i++) row.set(target.get(i), evaluate(tuple.get(i), List.of()));
660 checked.add(rows.size());
661 rows.add(row);
662 }
663 constraints(t, rows, checked);
664 count = c.tuples.size();
665 } else if (c.kind.equals("update")) {
666 List<Integer>target = targets(t, c.assignments.stream().map(Map.Entry::getKey).toList());
667 Scope scope = new Scope();
668 scope.add(t.name, t);
669 for (int i = 0; i<c.assignments.size(); i++) {
670 Column column = t.columns.get(target.get(i));
671 require(check(c.assignments.get(i).getValue(), scope), column.type, "column "+column.name);
672 }
673 condition(c.where, scope, "WHERE");
674 for (int i = 0; i<t.rows.size(); i++) {
675 List<Object>row = t.rows.get(i), next = row;
676 if (satisfies(c.where, row, null)) {
677 next = new ArrayList<>(row);
678 for (int j = 0; j<c.assignments.size(); j++) next.set(target.get(j), evaluate(c.assignments.get(j).getValue(), row));
679 checked.add(i);
680 count++;
681 }
682 rows.add(next);
683 }
684 constraints(t, rows, checked);
685 } else {
686 Scope scope = new Scope();
687 scope.add(t.name, t);
688 condition(c.where, scope, "WHERE");
689 for (List<Object>row:t.rows) if (satisfies(c.where, row, null)) count++;
690 else rows.add(row);
691 }
692 t.rows = rows;
693 System.out.println(count(count)+(c.kind.equals("insert")?" inserted.":c.kind.equals("update")?" updated.":" deleted."));
694 }
695 void select(Query q) {
696 Scope scope = new Scope();
697 scope.add(q.source.alias, table(q.source.name));
698 for (Join j:q.joins) {
699 scope.add(j.table.alias, table(j.table.name));
700 condition(j.condition, scope, "ON");
701 }
702 condition(q.where, scope, "WHERE");
703 List<Integer>grouping = new ArrayList<>();
704 if (q.group != null) for (Expr e:q.group) {
705 scope.resolve(e);
706 grouping.add(e.index);
707 }
708 if (q.star) for (Entry part:scope.entries) for (Column c:part.table.columns) {
709 Expr e = node("ref");
710 e.table = part.name;
711 e.column = e.text = c.name;
712 q.items.add(new Item(e, null, c.name));
713 }
714 boolean aggregate = q.group != null || q.having != null;
715 for (Item i:q.items) aggregate = aggregate || hasAggregate(i.expr);
716 for (Order o:q.order) aggregate = aggregate || hasAggregate(o.expr);
717 List<Integer>grouped = aggregate?grouping:null;
718 List<String>types = new ArrayList<>();
719 for (Item i:q.items) types.add(check(i.expr, scope, true, grouped, false));
720 condition(q.having, scope, "HAVING", true, grouped);
721 List<Expr>keys = new ArrayList<>();
722 for (Order o:q.order) {
723 Expr key = null;
724 if (o.ordinal != null) {
725 long k = o.ordinal;
726 if (k<1 || k>q.items.size()) throw error("ORDER BY position "+k+" is out of range");
727 key = q.items.get((int) k-1).expr;
728 } else if (o.alias != null) for (Item i:q.items) if (o.alias.equals(i.alias)) {
729 key = i.expr;
730 break;
731 }
732 if (key == null) {
733 check(o.expr, scope, true, grouped, false);
734 key = o.expr;
735 }
736 keys.add(key);
737 }
738 List<List<Object>>rows = scope.entries.get(0).table.rows;
739 for (int ji = 0; ji<q.joins.size(); ji++) {
740 Join j = q.joins.get(ji);
741 Table t = scope.entries.get(ji+1).table;
742 List<List<Object>>next = new ArrayList<>();
743 for (List<Object>left:rows) {
744 boolean found = false;
745 for (List<Object>right:t.rows) {
746 List<Object>combined = new ArrayList<>(left);
747 combined.addAll(right);
748 if (satisfies(j.condition, combined, null)) {
749 next.add(combined);
750 found = true;
751 }
752 }
753 if (j.left && !found) {
754 List<Object>combined = new ArrayList<>(left);
755 combined.addAll(Collections.nCopies(t.columns.size(), null));
756 next.add(combined);
757 }
758 }
759 rows = next;
760 }
761 List<List<Object>>filtered = new ArrayList<>();
762 for (List<Object>r:rows) if (satisfies(q.where, r, null)) filtered.add(r);
763 rows = filtered;
764 List<List<List<Object>>>groups = new ArrayList<>();
765 if (!aggregate) {
766 for (List<Object>r:rows) groups.add(Collections.singletonList(r));
767 } else if (q.group == null) groups.add(rows);
768 else {
769 Map<List<Object>, List<List<Object>>>byKey = new LinkedHashMap<>();
770 for (List<Object>r:rows) {
771 List<Object>key = new ArrayList<>();
772 for (int i:grouping) key.add(r.get(i));
773 byKey.computeIfAbsent(key, k -> new ArrayList<>()).add(r);
774 }
775 groups.addAll(byKey.values());
776 }
777 record Result(List<Object>values, List<Object>keys) {
778 }
779 List<Result>results = new ArrayList<>();
780 for (List<List<Object>>g:groups) {
781 List<Object>row = g.isEmpty()?List.of():g.get(0);
782 List<List<Object>>context = aggregate?g:null;
783 if (!satisfies(q.having, row, context)) continue;
784 List<Object>values = new ArrayList<>(), sort = new ArrayList<>();
785 for (Item i:q.items) values.add(evaluate(i.expr, row, context));
786 for (Expr key:keys) sort.add(evaluate(key, row, context));
787 results.add(new Result(values, sort));
788 }
789 results.sort((a, b) -> {
790 for (int i = 0; i<keys.size(); i++) {
791 int c = compare(a.keys.get(i), b.keys.get(i));
792 if (c != 0) return q.order.get(i).descending?-c:c;
793 }
794 return 0;
795 });
796 List<List<Object>>unique = new ArrayList<>();
797 Set<List<Object>>seen = new HashSet<>();
798 for (Result r:results) if (!q.distinct || seen.add(r.values)) unique.add(r.values);
799 int first = (int) Math.min(q.offset, unique.size()), last = q.limit == null?unique.size():(int) Math.min(unique.size(), first+q.limit);
800 printTable(q.items, types, new ArrayList<>(unique.subList(first, last)));
801 }
802 }
