20 static List<Token> Tokenize(string s)
21 {
22 List<Token>a = new();
23 int i = 0;
24 while (i<s.Length) {
25 char c = s[i];
26 int start = i;
27 if (c == ' ' || c == '\n') {
28 i++;
29 continue;
30 }
31 if (s.AsSpan(i).StartsWith("--")) {
32 int end = s.IndexOf('\n', i);
33 i = end<0?s.Length:end;
34 continue;
35 }
36 if (c == '\'') {
37 i = s.IndexOf('\'', i+1)+1;
38 a.Add(new("string", s[start..i], s[(start+1)..(i-1)]));
39 }
40 else if (char.IsAsciiDigit(c)) {
41 while (i<s.Length && char.IsAsciiDigit(s[i])) i++;
42 string t = s[start..i];
43 a.Add(new("integer", t, long.Parse(t)));
44 }
45 else if (char.IsAsciiLetter(c) || c == '_') {
46 while (i<s.Length && (char.IsAsciiLetterOrDigit(s[i]) || s[i] == '_')) i++;
47 string t = s[start..i], up = t.ToUpperInvariant();
48 a.Add(new(Reserved.Contains(up)?"keyword":"name", t, Reserved.Contains(up)?up:t));
49 }
50 else {
51 string t = s.Substring(i, Math.Min(2, s.Length-i));
52 if (!new[] {
53 "<>", "!=", "<=", ">=", "||"
54 }.Contains(t)) t = c.ToString();
55 if (t.Length == 1 && !"(),.;*/%+-=<>".Contains(c)) throw Error($"unexpected character '{c}'");
56 a.Add(new("symbol", t, t));
57 i += t.Length;
58 }
59 }
60 return a;
61 }

109 Token Peek() => tokens[position];

163 Expr Expression(int level = 0)
164 {
165 if (level == 2 && Accept("NOT")) return new("unary", "NOT", new[] {
166 Expression(2)
167 });
168 if (level == 6 && Accept("-")) return new("unary", "-", new[] {
169 Expression(6)
170 });
171 if (level == 7) return Primary();
172 Expr left = Expression(level+1);
173 if (level == 3) {
174 if (Peek().Kind == "symbol" && Comparisons.Contains((string) Peek().Value!)) {
175 string op = (string) Take().Value!;
176 return new("binary", op, new[] {
177 left, Expression(4)
178 });
179 }
180 if (Accept("IS")) {
181 bool neg = Accept("NOT");
182 Expect("NULL");
183 return new("isnull", "", new[] {
184 left
185 }) {
186 Negated = neg
187 };
188 }
189 bool not = Accept("NOT");
190 Expr? result = null;
191 if (Accept("IN")) result = new("in", "", new[] {
192 left
193 }.Concat(Parens(() => Expression())));
194 else if (Accept("BETWEEN")) {
195 Expr low = Expression(4);
196 Expect("AND");
197 result = new("between", "", new[] {
198 left, low, Expression(4)
199 });
200 } else if (Accept("LIKE")) result = new("like", "", new[] {
201 left, Expression(4)
202 });
203 else if (not) throw Syntax();
204 if (result != null) {
205 result.Negated = not;
206 return result;
207 }
208 return left;
209 }
210 string[][]ops = {
211 new[] {
212 "OR"
213 }, new[] {
214 "AND"
215 }, Array.Empty<string>(), Array.Empty<string>(), new[] {
216 "+", "-", "||"
217 }, new[] {
218 "*", "/", "%"
219 }, Array.Empty<string>()
220 };
221 while (ops[level].Any(At)) {
222 string op = (string) Take().Value!;
223 left = new("binary", op, new[] {
224 left, Expression(level+1)
225 });
226 }
227 return left;
228 }
229 Expr Primary() {
230 Token t = Peek();
231 if (t.Kind == "integer" || t.Kind == "string") {
232 Take();
233 return new("literal", value:t.Value);
234 }
235 if (Accept("NULL")) return new("literal");
236 if (t.Kind == "name") return Reference();
237 if (t.Kind == "keyword" && Aggregates.Contains((string) t.Value!)) {
238 string n = (string) Take().Value!;
239 Expect("(");
240 Expr e = new("aggregate", n);
241 if (!(n == "COUNT" && Accept("*"))) {
242 e.Distinct = Accept("DISTINCT");
243 e.Args.Add(Expression());
244 }
245 Expect(")");
246 return e;
247 }
248 if (Accept("(")) {
249 Expr e = Expression();
250 Expect(")");
251 return new("group", "", new[] {
252 e
253 });
254 }
255 throw Syntax();
256 }
257 Expr? Clause(string word) => Accept(word)?Expression():null;
258 Item SelectItem() {
259 int start = position;
260 Expr e = Expression();
261 int stop = position;
262 string? alias = Alias();
263 string heading = alias ?? "";
264 if (alias == null) {
265 if (e.Kind == "ref") heading = e.Column;
266 else {
267 StringBuilder s = new();
268 for (int i = start; i<stop; i++) {
269 if (i>start && tokens[i].Kind != "symbol" && tokens[i-1].Kind != "symbol") s.Append(' ');
270 s.Append(tokens[i].Text);
271 }
272 heading = s.ToString();
273 }
274 }
275 return new(e, alias, heading);
276 }
277 Order OrderItem() {
278 int start = position;
279 Expr e = Expression();
280 bool single = position == start+1;
281 Token t = tokens[start];
282 bool desc = Accept("DESC");
283 if (!desc) Accept("ASC");
284 return new(e, desc, single && t.Kind == "integer"?(long) t.Value!:null, single && t.Kind == "name"?(string) t.Value!:null);
285 }
286 Query Select() {
287 Query q = new() {
288 Distinct = Accept("DISTINCT"), Star = Accept("*")
289 };
290 if (!q.Star) q.Items = Comma(SelectItem);
291 Expect("FROM");
292 q.Source = Table();
293 while (At("JOIN") || At("INNER") || At("LEFT")) {
294 bool left = Accept("LEFT");
295 if (left) Accept("OUTER");
296 else Accept("INNER");
297 Expect("JOIN");
298 TableRef t = Table();
299 Expect("ON");
300 q.Joins.Add(new(t, left, Expression()));
301 }
302 q.Where = Clause("WHERE");
303 if (Accept("GROUP")) {
304 Expect("BY");
305 q.Group = Comma(Reference);
306 }
307 q.Having = Clause("HAVING");
308 if (Accept("ORDER")) {
309 Expect("BY");
310 q.Order = Comma(OrderItem);
311 }
312 if (Accept("LIMIT")) {
313 q.Limit = Integer();
314 if (Accept("OFFSET")) q.Offset = Integer();
315 }
316 return q;
317 }
318 Column ColumnDefinition() {
319 string n = Name();
320 if (!At("INT") && !At("TEXT")) throw Syntax();
321 string t = (string) Take().Value!;
322 int pk = 0;
323 bool nn = false;
324 while (true) {
325 if (Accept("PRIMARY")) {
326 Expect("KEY");
327 pk++;
328 nn = true;
329 } else if (Accept("NOT")) {
330 Expect("NULL");
331 nn = true;
332 } else break;
333 }
334 return new(n, t, pk, nn);
335 }

375 }

415 static bool Compatible(string a, string b) => a == b || a == "NULL" || b == "NULL";
416 static void Require(string a, string b, string op) {
417 if (!Compatible(a, b)) throw Error("type mismatch for "+op);
418 }
419 static string Check(Expr e, Scope s, bool allow = false, List<int>? grouped = null, bool inside = false)
420 {
421 if (e.Kind == "literal") return e.Value == null?"NULL":e.Value is long?"INT":"TEXT";
422 if (e.Kind == "ref") {
423 string type = s.Resolve(e);
424 if (grouped != null && !inside && !grouped.Contains(e.Index)) throw Error("column "+e.Text+" is not grouped");
425 return type;
426 }
427 if (e.Kind == "aggregate") {
428 if (!allow || inside) throw Error("misuse of aggregate "+e.Op);
429 if (e.Args.Count == 0) return "INT";
430 string type = Check(e.Args[0], s, allow, grouped, true);
431 if (e.Op == "SUM" || e.Op == "AVG") Require(type, "INT", e.Op);
432 return e.Op == "MIN" || e.Op == "MAX"?type:"INT";
433 }
434 List<string>types = e.Args.Select(a => Check(a, s, allow, grouped, inside)).ToList();
435 if (e.Kind == "group") return types[0];
436 if (e.Kind == "isnull") return "INT";
437 if (e.Kind == "binary" && Comparisons.Contains(e.Op)) {
438 Require(types[0], types[1], e.Op);
439 return "INT";
440 }
441 if (e.Kind == "in" || e.Kind == "between") {
442 foreach (string t in types.Skip(1)) Require(t, types[0], e.Kind == "in"?"IN":"BETWEEN");
443 return "INT";
444 }
445 string op = e.Kind == "like"?"LIKE":e.Op, typeRequired = op == "LIKE" || op == "||"?"TEXT":"INT";
446 foreach (string t in types) Require(t, typeRequired, op);
447 return op == "||"?"TEXT":"INT";
448 }
449 static void Condition(Expr? e, Scope s, string name, bool allow = false, List<int>? grouped = null) {
450 if (e != null) Require(Check(e, s, allow, grouped), "INT", name);
451 }
452 static bool HasAggregate(Expr e) => e.Kind == "aggregate" || e.Args.Any(HasAggregate);
453 static bool Truth(object? v) => v != null && (long) v != 0;
454 static object? Inverse(object? v) => v == null?null:(long) v == 0?1L:0L;
455 static int Compare(object? a, object? b) {
456 if (Equals(a, b)) return 0;
457 if (a == null) return -1;
458 if (b == null) return 1;
459 return a is long x?x.CompareTo((long) b):string.CompareOrdinal((string) a, (string) b);
460 }
461 static object? Operate(string op, object? a, object? b)
462 {
463 if (op == "AND") {
464 if (Equals(a, 0L) || Equals(b, 0L)) return 0L;
465 return a == null || b == null?null:1L;
466 }
467 if (op == "OR") {
468 if (Truth(a) || Truth(b)) return 1L;
469 return a == null || b == null?null:0L;
470 }
471 if (a == null || b == null) return null;
472 if (Comparisons.Contains(op)) {
473 int c = Compare(a, b);
474 bool ok = op switch {
475 "=" => c == 0, "<>" or "!=" => c != 0, "<" => c<0, "<=" => c <= 0, ">" => c>0, _ => c >= 0
476 };
477 return ok?1L:0L;
478 }
479 if (op == "||") return (string) a+(string) b;
480 long x = (long) a, y = (long) b;
481 return op switch {
482 "+" => x+y, "-" => x-y, "*" => x*y, "/" => y == 0?null:x/y, "%" => y == 0?null:x%y, _ => throw new InvalidOperationException()
483 };
484 }
485 static bool Like(string text, string pattern) {
486 bool[]dp = new bool[text.Length+1];
487 dp[0] = true;
488 foreach (char c in pattern) {
489 if (c == '%') {
490 for (int i = 1; i <= text.Length; i++) dp[i]|=dp[i-1];
491 } else {
492 for (int i = text.Length; i>0; i--) dp[i] = dp[i-1] && (c == '_' || c == text[i-1]);
493 dp[0] = false;
494 }
495 }
496 return dp[text.Length];
497 }
498 static object? Evaluate(Expr e, List<object?>row, List<List<object?>>? group = null)
499 {
500 if (e.Kind == "literal") return e.Value;
501 if (e.Kind == "ref") return row[e.Index];
502 if (e.Kind == "aggregate") {
503 if (e.Args.Count == 0) return (long) group!.Count;
504 List<object?>values = new();
505 foreach (List<object?>r in group!) {
506 object? v = Evaluate(e.Args[0], r);
507 if (v != null && (!e.Distinct || !values.Contains(v))) values.Add(v);
508 }
509 if (e.Op == "COUNT") return (long) values.Count;
510 if (values.Count == 0) return null;
511 if (e.Op == "MIN" || e.Op == "MAX") {
512 object? result = values[0];
513 foreach (object? v in values) if (e.Op == "MIN"?Compare(v, result)<0:Compare(v, result)>0) result = v;
514 return result;
515 }
516 long sum = values.Sum(v => (long) v!);
517 return e.Op == "AVG"?sum/values.Count:sum;
518 }
519 List<object?>a = e.Args.Select(arg => Evaluate(arg, row, group)).ToList();
520 if (e.Kind == "group") return a[0];
521 if (e.Kind == "unary") return e.Op == "NOT"?Inverse(a[0]):a[0] == null?null:-(long) a[0]!;
522 if (e.Kind == "binary") return Operate(e.Op, a[0], a[1]);
523 if (e.Kind == "isnull") return (a[0] == null) != e.Negated?1L:0L;
524 object? resultValue = null;
525 if (e.Kind == "in") {
526 if (a[0] != null) resultValue = a.Skip(1).Contains(a[0])?1L:a.Skip(1).Contains(null)?null:0L;
527 } else if (e.Kind == "between") resultValue = Operate("AND", Operate(">=", a[0], a[1]), Operate("<=", a[0], a[2]));
528 else if (a[0] != null && a[1] != null) resultValue = Like((string) a[0]!, (string) a[1]!)?1L:0L;
529 return e.Negated?Inverse(resultValue):resultValue;
530 }
531 static bool Satisfies(Expr? e, List<object?>row, List<List<object?>>? group = null) => e == null || Truth(Evaluate(e, row, group));

571 public void Execute(Command c)
572 {
573 if (c.Kind == "select") {
574 Select(c.Query!);
575 return;
576 }
577 if (c.Kind == "create") {
578 if (tables.ContainsKey(c.Name)) throw Error("table "+c.Name+" already exists");
579 HashSet<string>seen = new();
580 int primary = 0, key = -1;
581 for (int i = 0; i<c.Columns.Count; i++) {
582 Column column = c.Columns[i];
583 if (!seen.Add(column.Name)) throw Error("duplicate column "+column.Name);
584 primary += column.Primary;
585 if (primary >= 2) throw Error("multiple primary keys");
586 if (column.Primary>0) key = i;
587 }
588 tables[c.Name] = new(c.Name, c.Columns, key);
589 Console.WriteLine("Table "+c.Name+" created.");
590 return;
591 }
592 Table t = GetTable(c.Name);
593 if (c.Kind == "drop") {
594 tables.Remove(c.Name);
595 Console.WriteLine("Table "+c.Name+" dropped.");
596 return;
597 }
598 List<List<object?>>rows = new();
599 List<int>checkedRows = new();
600 int count = 0;
601 if (c.Kind == "insert") {
602 List<int>target = c.Names == null?Enumerable.Range(0, t.Columns.Count).ToList():Targets(t, c.Names);
603 Scope empty = new();
604 foreach (List<Expr>tuple in c.Tuples) {
605 if (tuple.Count != target.Count) throw Error($"expected {target.Count} values, got {tuple.Count}");
606 for (int i = 0; i<tuple.Count; i++) {
607 Column column = t.Columns[target[i]];
608 Require(Check(tuple[i], empty), column.Type, "column "+column.Name);
609 }
610 }
611 rows.AddRange(t.Rows);
612 foreach (List<Expr>tuple in c.Tuples) {
613 List<object?>row = Enumerable.Repeat<object?>(null, t.Columns.Count).ToList();
614 for (int i = 0; i<tuple.Count; i++) row[target[i]] = Evaluate(tuple[i], new());
615 checkedRows.Add(rows.Count);
616 rows.Add(row);
617 }
618 Constraints(t, rows, checkedRows);
619 count = c.Tuples.Count;
620 } else if (c.Kind == "update") {
621 List<int>target = Targets(t, c.Assignments.Select(a => a.Name).ToList());
622 Scope scope = new();
623 scope.Add(t.Name, t);
624 for (int i = 0; i<c.Assignments.Count; i++) {
625 Column column = t.Columns[target[i]];
626 Require(Check(c.Assignments[i].Expression, scope), column.Type, "column "+column.Name);
627 }
628 Condition(c.Where, scope, "WHERE");
629 for (int i = 0; i<t.Rows.Count; i++) {
630 List<object?>row = t.Rows[i], next = row;
631 if (Satisfies(c.Where, row)) {
632 next = new(row);
633 for (int j = 0; j<c.Assignments.Count; j++) next[target[j]] = Evaluate(c.Assignments[j].Expression, row);
634 checkedRows.Add(i);
635 count++;
636 }
637 rows.Add(next);
638 }
639 Constraints(t, rows, checkedRows);
640 } else {
641 Scope scope = new();
642 scope.Add(t.Name, t);
643 Condition(c.Where, scope, "WHERE");
644 foreach (List<object?>row in t.Rows) if (Satisfies(c.Where, row)) count++;
645 else rows.Add(row);
646 }
647 t.Rows = rows;

651 void Select(Query q)
652 {
653 Scope scope = new();
654 scope.Add(q.Source.Alias, GetTable(q.Source.Name));
655 foreach (Join j in q.Joins) {
656 scope.Add(j.Table.Alias, GetTable(j.Table.Name));
657 Condition(j.Condition, scope, "ON");
658 }
659 Condition(q.Where, scope, "WHERE");
660 List<int>grouping = new();
661 if (q.Group != null) foreach (Expr e in q.Group) {
662 scope.Resolve(e);
663 grouping.Add(e.Index);
664 }
665 if (q.Star) foreach (Entry p in scope.Entries) foreach (Column c in p.Table.Columns) {
666 Expr e = new("ref") {
667 Table = p.Name, Column = c.Name, Text = c.Name
668 };
669 q.Items.Add(new(e, null, c.Name));
670 }
671 bool aggregate = q.Group != null || q.Having != null || q.Items.Any(i => HasAggregate(i.Expression)) || q.Order.Any(o => HasAggregate(o.Expression));
672 List<int>? grouped = aggregate?grouping:null;
673 List<string>types = q.Items.Select(i => Check(i.Expression, scope, true, grouped)).ToList();
674 Condition(q.Having, scope, "HAVING", true, grouped);
675 List<Expr>keys = new();
676 foreach (Order o in q.Order) {
677 Expr? key = null;
678 if (o.Ordinal != null) {
679 long k = o.Ordinal.Value;
680 if (k<1 || k>q.Items.Count) throw Error("ORDER BY position "+k+" is out of range");
681 key = q.Items[(int) k-1].Expression;
682 } else if (o.Alias != null) key = q.Items.FirstOrDefault(i => i.Alias == o.Alias)?.Expression;
683 if (key == null) {
684 Check(o.Expression, scope, true, grouped);
685 key = o.Expression;
686 }
687 keys.Add(key);
688 }
689 List<List<object?>>rows = scope.Entries[0].Table.Rows;
690 for (int ji = 0; ji<q.Joins.Count; ji++) {
691 Join j = q.Joins[ji];
692 Table t = scope.Entries[ji+1].Table;
693 List<List<object?>>next = new();
694 foreach (List<object?>left in rows) {
695 bool found = false;
696 foreach (List<object?>right in t.Rows) {
697 List<object?>combined = left.Concat(right).ToList();
698 if (Satisfies(j.Condition, combined)) {
699 next.Add(combined);
700 found = true;
701 }
702 }
703 if (j.Left && !found) next.Add(left.Concat(Enumerable.Repeat<object?>(null, t.Columns.Count)).ToList());
704 }
705 rows = next;
706 }
707 rows = rows.Where(r => Satisfies(q.Where, r)).ToList();
708 List<List<List<object?>>>groups = new();
709 if (!aggregate) {
710 foreach (List<object?>r in rows) groups.Add(new() {
711 r
712 });
713 } else if (q.Group == null) groups.Add(rows);
714 else {
715 List<List<object?>>groupKeys = new();
716 foreach (List<object?>r in rows) {
717 List<object?>key = grouping.Select(i => r[i]).ToList();
718 int at = groupKeys.FindIndex(k => k.SequenceEqual(key));
719 if (at<0) {
720 at = groups.Count;
721 groups.Add(new());
722 groupKeys.Add(key);
723 }
724 groups[at].Add(r);
725 }
726 }
727 List<Result>results = new();
728 foreach (List<List<object?>>g in groups) {
729 List<object?>row = g.Count == 0?new():g[0];
730 List<List<object?>>? context = aggregate?g:null;
731 if (!Satisfies(q.Having, row, context)) continue;
732 List<object?>values = q.Items.Select(i => Evaluate(i.Expression, row, context)).ToList(), sort = keys.Select(key => Evaluate(key, row, context)).ToList();
733 results.Add(new(values, sort, results.Count));
734 }
735 results.Sort((a, b) => {
736 for (int i = 0; i<keys.Count; i++) {
737 int c = Compare(a.Keys[i], b.Keys[i]);
738 if (c != 0) return q.Order[i].Descending?-c:c;
739 }
740 return a.Position.CompareTo(b.Position);
741 });
742 List<List<object?>>unique = new();
743 foreach (Result r in results) if (!q.Distinct || !unique.Any(row => row.SequenceEqual(r.Values))) unique.Add(r.Values);
744 int first = (int) Math.Min(q.Offset, unique.Count), last = q.Limit == null?unique.Count:(int) Math.Min(unique.Count, first+q.Limit.Value);
745 PrintTable(q.Items, types, unique.GetRange(first, last-first));
746 }
