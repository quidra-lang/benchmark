99static int compare(Value a, Value b) {
100 if (a.type == NULL_VALUE || b.type == NULL_VALUE) return (a.type != NULL_VALUE) - (b.type != NULL_VALUE);
101 if (a.type == INTEGER_VALUE) return (a.integer > b.integer) - (a.integer < b.integer);
102 int c = strcmp(a.text,b.text);
103 return (c > 0) - (c < 0);
104}

122static int member(const char *s, const char *list) {
123 size_t n = strlen(s);
124 for (const char *p = list; *p;) {
125 const char *q = strchr(p,' ');
126 size_t len = q ? (size_t)(q-p) : strlen(p);
127 if (len == n && !strncmp(s,p,n)) return 1;
128 p += len;
129 if (*p) p++;
130 }
131 return 0;
132}

150static void tokenize(const char *s) {
151 tokens = (Tokens){
152 0
153 };
154 for (size_t i = 0; s[i];) {
155 if (s[i] == ' ' || s[i] == '\n') {
156 i++;
157 continue;
158 }
159 if (s[i] == '-' && s[i+1] == '-') {
160 while (s[i] && s[i] != '\n') i++;
161 continue;
162 }
163 size_t start = i;
164 char kind = 's';
165 if (s[i] == '\'') {
166 i++;
167 while (s[i] != '\'') i++;
168 i++;
169 kind = 't';
170 }
171 else if (isdigit((unsigned char)s[i])) {
172 while (isdigit((unsigned char)s[i])) i++;
173 kind = 'i';
174 }
175 else if (isalpha((unsigned char)s[i]) || s[i] == '_') {
176 while (isalnum((unsigned char)s[i]) || s[i] == '_') i++;
177 kind = 'n';
178 }
179 else if (s[i+1] && ((s[i] == '<' && (s[i+1] == '>' || s[i+1] == '=')) || (s[i] == '!' && s[i+1] == '=') || (s[i] == '>' && s[i+1] == '=') || (s[i] == '|' && s[i+1] == '|'))) i += 2;
180 else if (strchr("(),.;*/%+-=<>",s[i])) i++;
181 else error("unexpected character '%c'",s[i]);
182 char *raw = slice(s+start,i-start), *key = raw;
183 if (kind == 'n') {
184 char *upper = copy(raw);
185 for (char *p = upper; *p; p++) *p = (char)toupper((unsigned char)*p);
186 if (member(upper,reserved)) {
187 kind = 'k';
188 key = upper;
189 }
190 }
191 APPEND(tokens,((Token){
192 kind,raw,key
193 }));
194 }
195 position = 0;
196}

299static Expr *parse_primary(void) {
300 if (peek().kind == 'i') {
301 Expr *e=expression('l',NULL);
302 e->literal=integer(strtoll(take().text,NULL,10));
303 return e;
304 }
305 if (peek().kind == 't') {
306 Expr *e=expression('l',NULL);
307 char *s=take().text;
308 e->literal=text(slice(s+1,strlen(s)-2));
309 return e;
310 }
311 if (accept("NULL")) return expression('l',NULL);
312 if (peek().kind == 'n') return parse_column();
313 if (at_any(aggregates)) {
314 char *name=take().key;
315 Expr *e=expression('a',name);
316 expect("(");
317 if (!(is(name,"COUNT") && accept("*"))) {
318 e->distinct=accept("DISTINCT");
319 APPEND(e->children,parse_expr());
320 }
321 expect(")");
322 return e;
323 }
324 if (accept("(")) {
325 Expr *e=parse_expr();
326 expect(")");
327 return e;
328 }
329 syntax();
330 return NULL;
331}

337static Expr *parse_binary(const char *ops, Expr *(*operand)(void)) {
338 Expr *e=operand();
339 while (at_any(ops)) {
340 char *op=take().key;
341 Expr *right=operand();
342 e=binary(op,e,right);
343 }
344 return e;
345}

355static Expr *parse_comparison(void) {
356 Expr *left=parse_additive(), *e=NULL;
357 if (at_any(comparisons)) {
358 char *op=take().key;
359 return binary(op,left,parse_additive());
360 }
361 if (accept("IS")) {
362 e=expression('z',NULL);
363 e->negated=accept("NOT");
364 expect("NULL");
365 APPEND(e->children,left);
366 return e;
367 }
368 int negated=accept("NOT");
369 if (accept("IN")) {
370 e=expression('i',NULL);
371 APPEND(e->children,left);
372 expect("(");
373 Expressions items=expression_list();
374 for (int i=0;i<items.n;i++) APPEND(e->children,items.v[i]);
375 expect(")");
376 } else if (accept("BETWEEN")) {
377 e=expression('r',NULL);
378 APPEND(e->children,left);
379 APPEND(e->children,parse_additive());
380 expect("AND");
381 APPEND(e->children,parse_additive());
382 } else if (accept("LIKE")) {
383 e=expression('p',NULL);
384 APPEND(e->children,left);
385 APPEND(e->children,parse_additive());
386 }
387 else if (negated) syntax();
388 if (!e) return left;
389 e->negated=negated;
390 return e;
391}

485static Statement parse_statement(void) {
486 Statement s={
487 0
488 };
489 s.command="";
490 s.limit=-1;
491 if (accept("CREATE")) {
492 s.command="CREATE";
493 expect("TABLE");
494 s.name=parse_name();
495 expect("(");
496 do {
497 char *name=parse_name();
498 if (!at_any("INT TEXT")) syntax();
499 Type type=is(take().key,"INT") ? INTEGER_VALUE : TEXT_VALUE;
500 int keys=0, nn=0;
501 while (at_any("PRIMARY NOT")) {
502 if (accept("PRIMARY")) {
503 expect("KEY");
504 keys++;
505 } else {
506 expect("NOT");
507 expect("NULL");
508 nn=1;
509 }
510 }
511 APPEND(s.columns,((Column){
512 name,type,nn || keys>0,keys
513 }));
514 } while (accept(","));
515 expect(")");
516 } else if (accept("DROP")) {
517 s.command="DROP";
518 expect("TABLE");
519 s.name=parse_name();
520 }
521 else if (accept("INSERT")) {
522 s.command="INSERT";
523 expect("INTO");
524 s.name=parse_name();
525 if (accept("(")) {
526 s.specified_names=1;
527 do {
528 APPEND(s.names,parse_name());
529 } while (accept(","));
530 expect(")");
531 }
532 expect("VALUES");
533 do {
534 expect("(");
535 APPEND(s.tuples,expression_list());
536 expect(")");
537 } while (accept(","));
538 } else if (accept("UPDATE")) {
539 s.command="UPDATE";
540 s.name=parse_name();
541 expect("SET");
542 do {
543 char *name=parse_name();
544 expect("=");
545 Expr *e=parse_expr();
546 APPEND(s.assignments,((Assignment){
547 name,e
548 }));
549 } while (accept(","));
550 s.where=parse_clause("WHERE");
551 } else if (accept("DELETE")) {
552 s.command="DELETE";
553 expect("FROM");
554 s.name=parse_name();
555 s.where=parse_clause("WHERE");
556 }
557 else if (accept("SELECT")) {
558 s.command="SELECT";
559 s.distinct=accept("DISTINCT");
560 s.star=accept("*");
561 if (!s.star) do {
562 int start=position;
563 Expr *e=parse_expr();
564 int end=position;
565 char *alias=parse_alias();
566 char *h=alias ? alias : e->kind == 'c' && tokens.v[start].kind == 'n' ? e->column : heading(start,end);
567 APPEND(s.items,((Item){
568 e,alias,h
569 }));
570 } while (accept(","));
571 expect("FROM");
572 s.source=parse_table_ref();
573 while (at_any("JOIN INNER LEFT")) {
574 int outer=accept("LEFT");
575 accept(outer ? "OUTER" : "INNER");
576 expect("JOIN");
577 TableRef ref=parse_table_ref();
578 expect("ON");
579 Expr *condition=parse_expr();
580 APPEND(s.joins,((Join){
581 ref,outer,condition
582 }));
583 }
584 s.where=parse_clause("WHERE");
585 if (accept("GROUP")) {
586 expect("BY");
587 do {
588 APPEND(s.grouping,parse_column());
589 } while (accept(","));
590 }
591 s.having=parse_clause("HAVING");
592 if (accept("ORDER")) {
593 expect("BY");
594 do {
595 int start=position;
596 Expr *e=parse_expr();
597 int count=position-start, desc=accept("DESC");
598 if (!desc) accept("ASC");
599 APPEND(s.orders,((Order){
600 e,start,count,desc
601 }));
602 } while (accept(","));
603 }
604 if (accept("LIMIT")) {
605 s.limit=parse_integer();
606 if (accept("OFFSET")) s.offset=parse_integer();
607 }
608 } else if (accept("EXIT")) s.command="EXIT";
609 expect(";");
610 return s;
611}

643static Table *persist_table(Table *t) {
644 Table *p=calloc(1,sizeof(*p));
645 if (!p) exit(1);
646 p->name=persist_string(t->name);
647 p->columns.n=t->columns.n;
648 p->columns.v=calloc((size_t)p->columns.n,sizeof(Column));
649 p->rows.n=t->rows.n;
650 p->rows.v=calloc((size_t)p->rows.n,sizeof(Value *));
651 if (!p->columns.v || (!p->rows.v && p->rows.n)) exit(1);
652 for (int i=0;i<p->columns.n;i++) {
653 p->columns.v[i]=t->columns.v[i];
654 p->columns.v[i].name=persist_string(t->columns.v[i].name);
655 }
656 for (int i=0;i<p->rows.n;i++) {
657 p->rows.v[i]=malloc((size_t)p->columns.n*sizeof(Value));
658 if (!p->rows.v[i]) exit(1);
659 for (int j=0;j<p->columns.n;j++) {
660 p->rows.v[i][j]=t->rows.v[i][j];
661 if (p->rows.v[i][j].type == TEXT_VALUE) p->rows.v[i][j].text=persist_string(p->rows.v[i][j].text);
662 }
663 }
664 return p;
665}

708static Type resolve(Expr *e, Scope *scope) {
709 int count=0, found_table=0;
710 Type type=NULL_VALUE;
711 for (int i=0;i<scope->entries.n;i++) {
712 Entry entry=scope->entries.v[i];
713 if (e->table && !is(e->table,entry.name)) continue;
714 found_table=1;
715 int j=column_index(entry.table,e->column);
716 if (j>=0) {
717 count++;
718 e->index=entry.offset+j;
719 type=entry.table->columns.v[j].type;
720 }
721 }
722 if (e->table && !found_table) error("unknown table %s",e->table);
723 if (!count) error("unknown column %s",e->written);
724 if (count>1) error("ambiguous column %s",e->column);
725 return type;
726}

728static void require(Type a, Type b, const char *op) {
729 if (a != b && a != NULL_VALUE && b != NULL_VALUE) error("type mismatch for %s",op);
730}

732static Type check(Expr *e, Scope *scope, int allowed, Expressions *grouped, int inside) {
733 if (e->kind == 'l') return e->literal.type;
734 if (e->kind == 'c') {
735 Type t=resolve(e,scope);
736 if (grouped && !inside) {
737 int found=0;
738 for (int i=0;i<grouped->n;i++) if (grouped->v[i]->index == e->index) found=1;
739 if (!found) error("column %s is not grouped",e->written);
740 }
741 return t;
742 }
743 if (e->kind == 'a') {
744 if (!allowed || inside) error("misuse of aggregate %s",e->op);
745 if (!e->children.n) return INTEGER_VALUE;
746 Type t=check(e->children.v[0],scope,allowed,grouped,1);
747 if (member(e->op,"SUM AVG")) require(t,INTEGER_VALUE,e->op);
748 return member(e->op,"MIN MAX") ? t : INTEGER_VALUE;
749 }
750 Type *types=allocate((size_t)e->children.n*sizeof(Type));
751 for (int i=0;i<e->children.n;i++) types[i]=check(e->children.v[i],scope,allowed,grouped,inside);
752 if (e->kind == 'u') require(types[0],INTEGER_VALUE,e->op);
753 else if (e->kind == 'b') {
754 if (member(e->op,comparisons)) require(types[0],types[1],e->op);
755 else {
756 Type t=is(e->op,"||") ? TEXT_VALUE : INTEGER_VALUE;
757 require(types[0],t,e->op);
758 require(types[1],t,e->op);
759 return t;
760 }
761 } else if (e->kind == 'i' || e->kind == 'r') {
762 for (int i=1;i<e->children.n;i++) require(types[i],types[0],e->kind == 'i' ? "IN" : "BETWEEN");
763 } else if (e->kind == 'p') {
764 require(types[0],TEXT_VALUE,"LIKE");
765 require(types[1],TEXT_VALUE,"LIKE");
766 }
767 return INTEGER_VALUE;
768}

780static Value operation(char *op, Value a, Value b) {
781 if (is(op,"AND")) return (a.type != NULL_VALUE && !a.integer) || (b.type != NULL_VALUE && !b.integer) ? integer(0) : a.type == NULL_VALUE || b.type == NULL_VALUE ? null_value() : integer(1);
782 if (is(op,"OR")) return truth(a) || truth(b) ? integer(1) : a.type == NULL_VALUE || b.type == NULL_VALUE ? null_value() : integer(0);
783 if (a.type == NULL_VALUE || b.type == NULL_VALUE) return null_value();
784 if (is(op,"+")) return integer(a.integer+b.integer);
785 if (is(op,"-")) return integer(a.integer-b.integer);
786 if (is(op,"*")) return integer(a.integer*b.integer);
787 if (is(op,"/")) return b.integer ? integer(a.integer/b.integer) : null_value();
788 if (is(op,"%")) return b.integer ? integer(a.integer%b.integer) : null_value();
789 if (is(op,"||")) return text(format("%s%s",a.text,b.text));
790 int c=compare(a,b);
791 if (is(op,"=")) return integer(c==0);
792 if (is(op,"!=") || is(op,"<>")) return integer(c!=0);
793 if (is(op,"<")) return integer(c<0);
794 if (is(op,"<=")) return integer(c<=0);
795 if (is(op,">")) return integer(c>0);
796 return integer(c>=0);
797}

799static int like_match(char *s, char *pattern) {
800 size_t n=strlen(s);
801 unsigned char *matched=allocate(n+1);
802 matched[0]=1;
803 for (char *p=pattern; *p; p++) {
804 if (*p == '%') {
805 for (size_t i=1;i<=n;i++) matched[i]=matched[i] || matched[i-1];
806 }
807 else {
808 for (size_t i=n;i>0;i--) matched[i]=matched[i-1] && (*p == '_' || *p == s[i-1]);
809 matched[0]=0;
810 }
811 }
812 return matched[n];
813}

815static Value evaluate(Expr *e, Value *row, Rows *group) {
816 if (e->kind == 'l') return e->literal;
817 if (e->kind == 'c') return row[e->index];
818 if (e->kind == 'a') {
819 if (!e->children.n) return integer(group->n);
820 Value *values=allocate((size_t)group->n*sizeof(Value));
821 int n=0;
822 for (int i=0;i<group->n;i++) {
823 Value v=evaluate(e->children.v[0],group->v[i],NULL);
824 if (v.type == NULL_VALUE) continue;
825 int seen=0;
826 if (e->distinct) for (int j=0;j<n;j++) if (equal(v,values[j])) seen=1;
827 if (!seen) values[n++]=v;
828 }
829 if (is(e->op,"COUNT")) return integer(n);
830 if (!n) return null_value();
831 Value result=values[0];
832 int64_t sum=0;
833 for (int i=0;i<n;i++) {
834 sum+=values[i].integer;
835 if ((is(e->op,"MIN") && compare(values[i],result)<0) || (is(e->op,"MAX") && compare(values[i],result)>0)) result=values[i];
836 }
837 if (member(e->op,"MIN MAX")) return result;
838 return integer(is(e->op,"AVG") ? sum/n : sum);
839 }
840 Value *v=allocate((size_t)e->children.n*sizeof(Value));
841 for (int i=0;i<e->children.n;i++) v[i]=evaluate(e->children.v[i],row,group);
842 if (e->kind == 'u') return is(e->op,"NOT") ? logical_not(v[0]) : v[0].type == NULL_VALUE ? null_value() : integer(-v[0].integer);
843 if (e->kind == 'b') return operation(e->op,v[0],v[1]);
844 Value result=null_value();
845 if (e->kind == 'z') result=integer(v[0].type == NULL_VALUE);
846 else if (e->kind == 'i' && v[0].type != NULL_VALUE) {
847 result=integer(0);
848 int unknown=0, matched=0;
849 for (int i=1;i<e->children.n;i++) {
850 if (v[i].type == NULL_VALUE) unknown=1;
851 else if (equal(v[0],v[i])) matched=1;
852 }
853 result=matched ? integer(1) : unknown ? null_value() : integer(0);
854 } else if (e->kind == 'r') result=operation("AND",operation(">=",v[0],v[1]),operation("<=",v[0],v[2]));
855 else if (e->kind == 'p' && v[0].type != NULL_VALUE && v[1].type != NULL_VALUE) result=integer(like_match(v[0].text,v[1].text));
856 return e->negated ? logical_not(result) : result;
857}

870static void constraints(Table *t, int *indices, int n) {
871 int key=-1;
872 for (int j=0;j<t->columns.n;j++) if (t->columns.v[j].keys) key=j;
873 for (int a=0;a<n;a++) {
874 int i=indices[a];
875 for (int j=0;j<t->columns.n;j++) if (t->columns.v[j].not_null && t->rows.v[i][j].type == NULL_VALUE) error("NULL value in column %s",t->columns.v[j].name);
876 if (key>=0) for (int j=0;j<t->rows.n;j++) if (i!=j && equal(t->rows.v[i][key],t->rows.v[j][key])) error("duplicate primary key %s",written(t->rows.v[i][key]));
877 }

933static void execute_select(Statement *s) {
934 Scope scope={
935 0
936 };
937 Table **tables=allocate((size_t)(s->joins.n+1)*sizeof(Table *));
938 for (int i=0;i<=s->joins.n;i++) {
939 TableRef r=i ? s->joins.v[i-1].table : s->source;
940 tables[i]=lookup(r.name);
941 add_scope(&scope,r.alias ? r.alias : r.name,tables[i]);
942 if (i) condition(s->joins.v[i-1].condition,&scope,"ON",0,NULL);
943 }
944 condition(s->where,&scope,"WHERE",0,NULL);
945 for (int i=0;i<s->grouping.n;i++) resolve(s->grouping.v[i],&scope);
946 int aggregate=s->grouping.n>0 || s->having;
947 for (int i=0;i<s->items.n;i++) if (has_aggregate(s->items.v[i].expr)) aggregate=1;
948 for (int i=0;i<s->orders.n;i++) if (has_aggregate(s->orders.v[i].expr)) aggregate=1;
949 Expressions *grouped=aggregate ? &s->grouping : NULL;
950 if (s->star) for (int i=0;i<scope.entries.n;i++) {
951 Entry entry=scope.entries.v[i];
952 for (int j=0;j<entry.table->columns.n;j++) {
953 Column c=entry.table->columns.v[j];
954 Expr *e=expression('c',NULL);
955 e->table=entry.name;
956 e->column=c.name;
957 e->written=c.name;
958 APPEND(s->items,((Item){
959 e,NULL,c.name
960 }));
961 }
962 }
963 Type *types=allocate((size_t)s->items.n*sizeof(Type));
964 for (int i=0;i<s->items.n;i++) types[i]=check(s->items.v[i].expr,&scope,1,grouped,0);
965 condition(s->having,&scope,"HAVING",1,grouped);
966 Expressions keys={
967 0
968 };
969 for (int i=0;i<s->orders.n;i++) {
970 Order order=s->orders.v[i];
971 Token first=tokens.v[order.start];
972 Expr *key=NULL;
973 if (order.count==1 && first.kind=='i') {
974 int k=(int)strtol(first.text,NULL,10);
975 if (k<1 || k>s->items.n) error("ORDER BY position %d is out of range",k);
976 key=s->items.v[k-1].expr;
977 }
978 else if (order.count==1 && first.kind=='n') for (int j=0;j<s->items.n;j++) if (s->items.v[j].alias && is(s->items.v[j].alias,first.text)) {
979 key=s->items.v[j].expr;
980 break;
981 }
982 if (!key) {
983 check(order.expr,&scope,1,grouped,0);
984 key=order.expr;
985 }
986 APPEND(keys,key);
987 }
988 Rows rows=tables[0]->rows;
989 int width=tables[0]->columns.n;
990 for (int j=0;j<s->joins.n;j++) {
991 Rows next={
992 0
993 };
994 Join join=s->joins.v[j];
995 Table *right=tables[j+1];
996 for (int l=0;l<rows.n;l++) {
997 int matched=0;
998 for (int r=0;r<right->rows.n;r++) {
999 Value *combined=allocate((size_t)(width+right->columns.n)*sizeof(Value));
1000 memcpy(combined,rows.v[l],(size_t)width*sizeof(Value));
1001 memcpy(combined+width,right->rows.v[r],(size_t)right->columns.n*sizeof(Value));
1002 if (truth(evaluate(join.condition,combined,NULL))) {
1003 APPEND(next,combined);
1004 matched=1;
1005 }
1006 }
1007 if (!matched && join.outer) {
1008 Value *combined=allocate((size_t)(width+right->columns.n)*sizeof(Value));
1009 memcpy(combined,rows.v[l],(size_t)width*sizeof(Value));
1010 APPEND(next,combined);
1011 }
1012 }
1013 rows=next;
1014 width+=right->columns.n;
1015 }
1016 if (s->where) {
1017 Rows filtered={
1018 0
1019 };
1020 for (int i=0;i<rows.n;i++) if (truth(evaluate(s->where,rows.v[i],NULL))) APPEND(filtered,rows.v[i]);
1021 rows=filtered;
1022 }
1023 VECTOR(Rows, Groups);
1024 Groups groups={
1025 0
1026 };
1027 Rows group_keys={
1028 0
1029 };
1030 if (aggregate && !s->grouping.n) APPEND(groups,rows);
1031 else if (aggregate) {
1032 for (int i=0;i<rows.n;i++) {
1033 Value *key=allocate((size_t)s->grouping.n*sizeof(Value));
1034 for (int j=0;j<s->grouping.n;j++) key[j]=rows.v[i][s->grouping.v[j]->index];
1035 int group=-1;
1036 for (int g=0;g<group_keys.n;g++) if (equal_rows(key,group_keys.v[g],s->grouping.n)) {
1037 group=g;
1038 break;
1039 }
1040 if (group<0) {
1041 group=groups.n;
1042 APPEND(group_keys,key);
1043 APPEND(groups,((Rows){
1044 0
1045 }));
1046 }
1047 APPEND(groups.v[group],rows.v[i]);
1048 }
1049 } else for (int i=0;i<rows.n;i++) {
1050 Rows group={
1051 0
1052 };
1053 APPEND(group,rows.v[i]);
1054 APPEND(groups,group);
1055 }
1056 Results results={
1057 0
1058 };
1059 for (int i=0;i<groups.n;i++) {
1060 Rows group=groups.v[i];
1061 Value *row=group.n ? group.v[0] : allocate((size_t)scope.width*sizeof(Value));
1062 if (s->having && !truth(evaluate(s->having,row,&group))) continue;
1063 Result result={
1064 allocate((size_t)s->items.n*sizeof(Value)),allocate((size_t)keys.n*sizeof(Value)),results.n
1065 };
1066 for (int j=0;j<s->items.n;j++) result.values[j]=evaluate(s->items.v[j].expr,row,&group);
1067 for (int j=0;j<keys.n;j++) result.keys[j]=evaluate(keys.v[j],row,&group);
1068 APPEND(results,result);
1069 }
1070 if (keys.n) {
1071 sort_orders=s->orders;
1072 qsort(results.v,(size_t)results.n,sizeof(Result),result_compare);
1073 }
1074 if (s->distinct) {
1075 Results unique={
1076 0
1077 };
1078 for (int i=0;i<results.n;i++) {
1079 int seen=0;
1080 for (int j=0;j<unique.n;j++) if (equal_rows(results.v[i].values,unique.v[j].values,s->items.n)) {
1081 seen=1;
1082 break;
1083 } if (!seen) APPEND(unique,results.v[i]);
1084 } results=unique;
1085 }
1086 print_table(s->items,types,results,s->offset,s->limit);
1087}

1102 for (int i=0;i<s->columns.n;i++) {

1104 keys+=s->columns.v[i].keys;

1130 };

1174 }

1201static int process_statement(char *source) {
1202 volatile int exit_requested=0;
1203 if (setjmp(failure)==0) {
1204 tokenize(source);
1205 Statement s=parse_statement();
1206 exit_requested=is(s.command,"EXIT");
1207 if (!exit_requested) execute(&s);
1208 }
1209 clear_arena();

1211}
1212int main(void) {
1213 size_t capacity=1024, length=0;
1214 char *statement=malloc(capacity);
1215 if (!statement) return 1;
1216 int c, previous=0, quoted=0, comment=0;
1217 while ((c=getchar()) != EOF) {
1218 if (length+2>capacity) {
1219 capacity*=2;
1220 statement=realloc(statement,capacity);
1221 if (!statement) return 1;
1222 }
1223 statement[length++]=(char)c;
1224 if (comment) {
1225 if (c=='\n') comment=0;
1226 }
1227 else if (c=='\'') quoted=!quoted;
1228 else if (!quoted && c=='-' && previous=='-') comment=1;
1229 else if (!quoted && c==';') {
1230 statement[length]=0;
1231 if (process_statement(statement)) break;
1232 length=0;
1233 }
1234 previous=c;
1235 }
1236 for (int i=0;i<table_count;i++) destroy_table(database[i]);
1237 free(statement);
1238 free(arena);
