11enum Value: Hashable {
12 case null, int(Int64), text(String)
13 var type: String {
14 switch self {
15 case .null: return "NULL"
16 case .int: return "INT"
17 case .text: return "TEXT"

21 switch self {
22 case .null: return "NULL"
23 case .int(let n): return String(n)
24 case .text(let s): return s
25 }
26 }
27 var integer: Int64 {
28 if case .int(let n) = self {
29 return n
30 }

32 }

34 self != .null && integer != 0

47func tokenize(_ text: String) throws -> [Token] {
48 let a = Array(text)
49 let n = a.count
50 var i = 0
51 var tokens: [Token] = []
52 func letter(_ c: Character) -> Bool {
53 c.isASCII && (c.isLetter || c == "_")
54 }
55 while i < n {
56 let c = a[i]
57 let start = i
58 if c == " " || c == "\n" {
59 i += 1
60 continue
61 }
62 if c == "-" && i+1 < n && a[i+1] == "-" {
63 while i < n && a[i] != "\n" {
64 i += 1
65 }
66 continue
67 }
68 var kind = "symbol"
69 if c == "'" {
70 i += 1
71 while i < n && a[i] != "'" {
72 i += 1
73 }
74 i += 1
75 kind = "string"
76 }
77 else if c.isNumber {
78 i += 1
79 while i < n && a[i].isNumber {
80 i += 1
81 }
82 kind = "integer"
83 }
84 else if letter(c) {
85 i += 1
86 while i < n && (letter(a[i]) || a[i].isNumber) {
87 i += 1
88 }
89 kind = "name"
90 }
91 else if i+1 < n && ["<>","!=","<=",">=","||"].contains(String(a[i...i+1])) {
92 i += 2
93 }
94 else if "(),.;*/%+-=<>".contains(c) {
95 i += 1
96 }
97 else {
98 try fail("unexpected character '\(c)'")
99 }
100 let s = String(a[start..<i])
101 if kind == "name" && reserved.contains(s.uppercased()) {
102 kind = "keyword"
103 }
104 tokens.append(Token(kind:kind,text:s,key:kind == "keyword" ? s.uppercased() : s))
105 }
106 return tokens
107}

199final class Parser {
200 let tokens: [Token]
201 var pos = 0
202 init(_ t: [Token]) {
203 tokens=t
204 }
205 var peek: Token {
206 tokens[min(pos,tokens.count-1)]
207 }
208 func at(_ keys: [String]) -> Bool {
209 (peek.kind == "keyword" || peek.kind == "symbol") && keys.contains(peek.key)
210 }
211 @discardableResult func take() -> Token {
212 let t = peek
213 pos += 1
214 return t
215 }
216 func accept(_ key: String) -> Bool {
217 if at([key]) {
218 pos += 1
219 return true
220 }
221 return false
222 }
223 func syntax() throws -> Never {
224 try fail("syntax error at '\(peek.text)'")
225 }
226 func expect(_ key: String) throws {
227 if !accept(key) {
228 try syntax()
229 }
230 }
231 func name() throws -> String {
232 if peek.kind != "name" {
233 try syntax()
234 }
235 return take().text
236 }
237 func integer() throws -> Int {
238 if peek.kind != "integer" {
239 try syntax()
240 }
241 return Int(take().text)!
242 }
243 func list<T>(_ parse: () throws -> T) throws -> [T] {
244 var a = [try parse()]
245 while accept(",") {
246 a.append(try parse())
247 }
248 return a
249 }
250 func parenthesized<T>(_ parse: () throws -> T) throws -> [T] {
251 try expect("(")
252 let a = try list(parse)
253 try expect(")")
254 return a
255 }
256 func alias() throws -> String? {
257 if accept("AS") {
258 return try name()
259 }
260 return peek.kind == "name" ? take().text : nil
261 }
262 func ref() throws -> Expr {
263 let first = try name()
264 if accept(".") {
265 let c = try name()
266 return Expr("column",table:first,column:c,written:first+"."+c)
267 }
268 return Expr("column",column:first,written:first)
269 }
270 func tableRef() throws -> TableRef {
271 let n = try name()
272 return TableRef(name:n,alias:try alias())
273 }
274 func clause(_ key: String) throws -> Expr? {
275 accept(key) ? try expr() : nil
276 }
277 func parse() throws -> Statement {
278 let s = Statement()
279 if accept("CREATE") {
280 s.command="CREATE"
281 try expect("TABLE")
282 s.name=try name()
283 s.columns=try parenthesized {
284 let n = try self.name()
285 if !self.at(["INT","TEXT"]) {
286 try self.syntax()
287 }
288 let type = self.take().key
289 var keys=0
290 var nn=false
291 while self.at(["PRIMARY","NOT"]) {
292 if self.accept("PRIMARY") {
293 try self.expect("KEY")
294 keys += 1
295 } else {
296 try self.expect("NOT")
297 try self.expect("NULL")
298 nn=true
299 }
300 }
301 return Column(name:n,type:type,notNull:nn || keys > 0,keys:keys)
302 }
303 } else if accept("DROP") {
304 s.command="DROP"
305 try expect("TABLE")
306 s.name=try name()
307 }
308 else if accept("INSERT") {
309 s.command="INSERT"
310 try expect("INTO")
311 s.name=try name()
312 if at(["("]) {
313 s.names=try parenthesized {
314 try self.name()
315 }
316 }
317 try expect("VALUES")
318 s.tuples=try list {
319 try self.parenthesized {
320 try self.expr()
321 }
322 }
323 }
324 else if accept("UPDATE") {
325 s.command="UPDATE"
326 s.name=try name()
327 try expect("SET")
328 s.assignments=try list {
329 let n = try self.name()
330 try self.expect("=")
331 return (n,try self.expr())
332 }
333 s.whereExpr=try clause("WHERE")
334 }
335 else if accept("DELETE") {
336 s.command="DELETE"
337 try expect("FROM")
338 s.name=try name()
339 s.whereExpr=try clause("WHERE")
340 }
341 else if accept("SELECT") {
342 s.command="SELECT"
343 s.distinct=accept("DISTINCT")
344 if !accept("*") {
345 s.items=try list {
346 let start=self.pos
347 let e=try self.expr()
348 let t=Array(self.tokens[start..<self.pos])
349 let alias=try self.alias()
350 var h=""
351 for (i,v) in t.enumerated() {
352 if i > 0 && v.kind != "symbol" && t[i-1].kind != "symbol" {
353 h += " "
354 }
355 h += v.text
356 }
357 return Item(expr:e,alias:alias,heading:alias ?? (e.kind == "column" && t[0].kind == "name" ? e.column : h))
358 }
359 }
360 try expect("FROM")
361 s.source=try tableRef()
362 while at(["JOIN","LEFT","INNER"]) {
363 let outer=accept("LEFT")
364 if outer {
365 _=accept("OUTER")
366 } else {
367 _=accept("INNER")
368 }
369 try expect("JOIN")
370 let t=try tableRef()
371 try expect("ON")
372 s.joins.append(Join(table:t,outer:outer,condition:try expr()))
373 }
374 s.whereExpr=try clause("WHERE")
375 if accept("GROUP") {
376 try expect("BY")
377 s.grouping=try list {
378 try self.ref()
379 }
380 }
381 s.having=try clause("HAVING")
382 if accept("ORDER") {
383 try expect("BY")
384 s.orders=try list {
385 let start=self.pos
386 let e=try self.expr()
387 let t=Array(self.tokens[start..<self.pos])
388 let desc=self.accept("DESC")
389 if !desc {
390 _=self.accept("ASC")
391 }
392 return Order(expr:e,tokens:t,descending:desc)
393 }
394 }
395 if accept("LIMIT") {
396 s.limit=try integer()
397 if accept("OFFSET") {
398 s.offset=try integer()
399 }
400 }
401 } else if accept("EXIT") {
402 s.command="EXIT"
403 }
404 try expect(";")
405 return s
406 }
407 func binary(_ ops: [String], _ operand: () throws -> Expr) throws -> Expr {
408 var e=try operand()
409 while at(ops) {
410 let op=take().key
411 e=Expr("binary",op,[e,try operand()])
412 }
413 return e
414 }
415 func expr() throws -> Expr {
416 try binary(["OR"]) {
417 try self.andExpr()
418 }
419 }
420 func andExpr() throws -> Expr {
421 try binary(["AND"]) {
422 try self.notExpr()
423 }
424 }
425 func notExpr() throws -> Expr {
426 if accept("NOT") {
427 return Expr("unary","NOT",[try notExpr()])
428 }
429 return try comparison()
430 }
431 func comparison() throws -> Expr {
432 let e=try additive()
433 if at(comparisons) {
434 let op=take().key
435 return Expr("binary",op,[e,try additive()])
436 }
437 if accept("IS") {
438 let n=accept("NOT")
439 try expect("NULL")
440 let r=Expr("isnull","",[e])
441 r.negated=n
442 return r
443 }
444 let n=accept("NOT")
445 var r: Expr?
446 if accept("IN") {
447 r=Expr("in","",[e]+(try parenthesized {
448 try self.expr()
449 }))
450 }
451 else if accept("BETWEEN") {
452 let low=try additive()
453 try expect("AND")
454 r=Expr("between","",[e,low,try additive()])
455 }
456 else if accept("LIKE") {
457 r=Expr("like","",[e,try additive()])
458 }
459 else if n {
460 try syntax()
461 }
462 if let r=r {
463 r.negated=n
464 return r
465 }
466 return e
467 }
468 func additive() throws -> Expr {
469 try binary(["+","-","||"]) {
470 try self.multiplicative()
471 }
472 }
473 func multiplicative() throws -> Expr {
474 try binary(["*","/","%"]) {
475 try self.unary()
476 }
477 }
478 func unary() throws -> Expr {
479 if accept("-") {
480 return Expr("unary","-",[try unary()])
481 }
482 return try primary()
483 }
484 func primary() throws -> Expr {
485 if peek.kind == "integer" {
486 return Expr("literal","",[],.int(Int64(take().text)!))
487 }
488 if peek.kind == "string" {
489 return Expr("literal","",[],.text(String(take().text.dropFirst().dropLast())))
490 }
491 if accept("NULL") {
492 return Expr("literal")
493 }
494 if peek.kind == "name" {
495 return try ref()
496 }
497 if at(aggregates) {
498 let n=take().key
499 try expect("(")
500 let e=Expr("aggregate",n)
501 if n == "COUNT" && accept("*") {
502 } else {
503 e.distinct=accept("DISTINCT")
504 e.children=[try expr()]
505 }
506 try expect(")")
507 return e
508 }
509 if accept("(") {
510 let e=try expr()
511 try expect(")")
512 return e
513 }
514 try syntax()
515 }
516}

520 let table: Table
521 let offset: Int

524struct Scope {
525 var entries: [Entry] = []
526 var width=0
527 mutating func add(_ name: String, _ table: Table) throws {
528 if entries.contains(where: {
529 $0.name == name
530 }) {
531 try fail("duplicate table name \(name)")
532 }
533 entries.append(Entry(name:name,table:table,offset:width))
534 width += table.columns.count
535 }
536 func resolve(_ e: Expr) throws -> String {
537 var candidates: [(Int,String)] = []
538 var selected=entries
539 if let table=e.table {
540 selected=entries.filter {
541 $0.name == table
542 }
543 if selected.isEmpty {
544 try fail("unknown table \(table)")
545 }
546 }
547 for entry in selected {
548 for (i,c) in entry.table.columns.enumerated() where c.name == e.column {
549 candidates.append((entry.offset+i,c.type))
550 }
551 }
552 if candidates.isEmpty {
553 try fail("unknown column \(e.written)")
554 }
555 if candidates.count > 1 {
556 try fail("ambiguous column \(e.column)")
557 }
558 e.index=candidates[0].0
559 return candidates[0].1
560 }
561}

569func check(_ e: Expr, _ scope: Scope, _ allowed: Bool = false, _ grouped: Set<Int>? = nil, _ inside: Bool = false) throws -> String {
570 if e.kind == "literal" {
571 return e.literal.type
572 }
573 if e.kind == "column" {
574 let t=try scope.resolve(e)
575 if let g=grouped, !inside && !g.contains(e.index) {
576 try fail("column \(e.written) is not grouped")
577 }
578 return t
579 }
580 if e.kind == "aggregate" {
581 if !allowed || inside {
582 try fail("misuse of aggregate \(e.op)")
583 }
584 if e.children.isEmpty {
585 return "INT"
586 }
587 let t=try check(e.children[0],scope,allowed,grouped,true)
588 if ["SUM","AVG"].contains(e.op) {
589 try require(t,"INT",e.op)
590 }
591 return ["MIN","MAX"].contains(e.op) ? t : "INT"
592 }
593 let types=try e.children.map {
594 try check($0,scope,allowed,grouped,inside)
595 }
596 switch e.kind {
597 case "unary": try require(types[0],"INT",e.op)
598 case "binary":
599 if comparisons.contains(e.op) {
600 try require(types[0],types[1],e.op)
601 }
602 else {
603 let t=e.op == "||" ? "TEXT" : "INT"
604 for a in types {
605 try require(a,t,e.op)
606 }
607 return t
608 }
609 case "in","between": for t in types.dropFirst() {
610 try require(t,types[0],e.kind == "in" ? "IN" : "BETWEEN")
611 }
612 case "like": for t in types {
613 try require(t,"TEXT","LIKE")
614 }
615 default: break
616 }
617 return "INT"
618}

620func condition(_ e: Expr?, _ scope: Scope, _ clause: String, _ allowed: Bool = false, _ grouped: Set<Int>? = nil) throws {
621 if let e=e {
622 try require(check(e,scope,allowed,grouped),"INT",clause)
623 }

626func compare(_ a: Value, _ b: Value) -> Int {
627 if a == b {
628 return 0
629 }
630 if a == .null {
631 return -1
632 }
633 if b == .null {
634 return 1
635 }
636 if case .int(let x)=a, case .int(let y)=b {
637 return x < y ? -1 : 1
638 }
639 return a.written < b.written ? -1 : 1
640}

650func operation(_ op: String, _ a: Value, _ b: Value) -> Value {
651 if op == "AND" {
652 return a == .int(0) || b == .int(0) ? .int(0) : (a == .null || b == .null ? .null : .int(1))
653 }
654 if op == "OR" {
655 return a.truth || b.truth ? .int(1) : (a == .null || b == .null ? .null : .int(0))
656 }
657 if a == .null || b == .null {
658 return .null
659 }
660 let x=a.integer
661 let y=b.integer
662 switch op {
663 case "+": return .int(x+y)
664 case "-": return .int(x-y)
665 case "*": return .int(x*y)
666 case "/": return y == 0 ? .null : .int(x/y)
667 case "%": return y == 0 ? .null : .int(x%y)
668 case "||": return .text(a.written+b.written)
669 case "=": return boolean(a == b)
670 case "!=","<>": return boolean(a != b)
671 case "<": return boolean(compare(a,b) < 0)
672 case "<=": return boolean(compare(a,b) <= 0)
673 case ">": return boolean(compare(a,b) > 0)
674 default: return boolean(compare(a,b) >= 0)
675 }
676}

678func like(_ text: String, _ pattern: String) -> Bool {
679 let a=Array(text)
680 var matched=[true]+Array(repeating:false,count:a.count)
681 for c in pattern {
682 if c == "%" {
683 for i in 0..<a.count {
684 matched[i+1] = matched[i+1] || matched[i]
685 }
686 }
687 else {
688 for i in stride(from:a.count,through:1,by:-1) {
689 matched[i] = matched[i-1] && (c == "_" || c == a[i-1])
690 }
691 matched[0]=false
692 }
693 }
694 return matched.last!
695}

697func evaluate(_ e: Expr, _ row: [Value], _ group: [[Value]] = []) -> Value {
698 if e.kind == "literal" {
699 return e.literal
700 }
701 if e.kind == "column" {
702 return row[e.index]
703 }
704 if e.kind == "aggregate" {
705 if e.children.isEmpty {
706 return .int(Int64(group.count))
707 }
708 var values=group.map {
709 evaluate(e.children[0],$0)
710 }.filter {
711 $0 != .null
712 }
713 if e.distinct {
714 var seen=Set<Value>()
715 values=values.filter {
716 seen.insert($0).inserted
717 }
718 }
719 if e.op == "COUNT" {
720 return .int(Int64(values.count))
721 }
722 if values.isEmpty {
723 return .null
724 }
725 if e.op == "MIN" {
726 return values.min {
727 compare($0,$1) < 0
728 }!
729 }
730 if e.op == "MAX" {
731 return values.max {
732 compare($0,$1) < 0
733 }!
734 }
735 let sum=values.reduce(Int64(0)) {
736 $0+$1.integer
737 }
738 return .int(e.op == "AVG" ? sum/Int64(values.count) : sum)
739 }
740 let v=e.children.map {
741 evaluate($0,row,group)
742 }
743 var result=Value.null
744 switch e.kind {
745 case "unary": return e.op == "NOT" ? logicalNot(v[0]) : (v[0] == .null ? .null : .int(-v[0].integer))
746 case "binary": return operation(e.op,v[0],v[1])
747 case "isnull": result=boolean(v[0] == .null)
748 case "in": result=v[0] == .null ? .null : (v.dropFirst().contains(v[0]) ? .int(1) : (v.dropFirst().contains(.null) ? .null : .int(0)))
749 case "between": result=operation("AND",operation(">=",v[0],v[1]),operation("<=",v[0],v[2]))
750 case "like": result=v.contains(.null) ? .null : boolean(like(v[0].written,v[1].written))
751 default: break
752 }
753 return e.negated ? logicalNot(result) : result
754}

820 print(separator)

822 print(line(r.map {
823 $0.written
824 },true))

830func executeSelect(_ s: Statement) throws {
831 var scope=Scope()
832 var tables: [Table] = []
833 let refs=[s.source!]+s.joins.map {
834 $0.table
835 }
836 for (i,r) in refs.enumerated() {
837 let t=try lookup(r.name)
838 try scope.add(r.scopeName,t)
839 tables.append(t)
840 if i > 0 {
841 try condition(s.joins[i-1].condition,scope,"ON")
842 }
843 }
844 try condition(s.whereExpr,scope,"WHERE")
845 for e in s.grouping {
846 _=try scope.resolve(e)
847 }
848 let aggregate = !s.grouping.isEmpty || s.having != nil || (s.items ?? []).contains {
849 $0.expr.hasAggregate
850 } || s.orders.contains {
851 $0.expr.hasAggregate
852 }
853 let grouped: Set<Int>? = aggregate ? Set(s.grouping.map {
854 $0.index
855 }) : nil
856 var items=s.items ?? []
857 if s.items == nil {
858 for entry in scope.entries {
859 for c in entry.table.columns {
860 items.append(Item(expr:Expr("column",table:entry.name,column:c.name,written:c.name),alias:nil,heading:c.name))
861 }
862 }
863 }
864 let types=try items.map {
865 try check($0.expr,scope,true,grouped)
866 }
867 try condition(s.having,scope,"HAVING",true,grouped)
868 var keys: [Expr] = []
869 for order in s.orders {
870 if order.tokens.count == 1 && order.tokens[0].kind == "integer" {
871 let k=Int(order.tokens[0].text)!
872 if k < 1 || k > items.count {
873 try fail("ORDER BY position \(k) is out of range")
874 }
875 keys.append(items[k-1].expr)
876 } else if order.tokens.count == 1 && order.tokens[0].kind == "name", let item=items.first(where: {
877 $0.alias == order.tokens[0].text
878 }) {
879 keys.append(item.expr)
880 }
881 else {
882 _=try check(order.expr,scope,true,grouped)
883 keys.append(order.expr)
884 }
885 }
886 var rows=tables[0].rows
887 for (j,join) in s.joins.enumerated() {
888 var next: [[Value]] = []
889 for l in rows {
890 var matched=false
891 for r in tables[j+1].rows {
892 let combined=l+r
893 if evaluate(join.condition,combined).truth {
894 next.append(combined)
895 matched=true
896 }
897 }
898 if !matched && join.outer {
899 next.append(l+Array(repeating:.null,count:tables[j+1].columns.count))
900 }
901 }
902 rows=next
903 }
904 if let e=s.whereExpr {
905 rows=rows.filter {
906 evaluate(e,$0).truth
907 }
908 }
909 var groups: [[[Value]]] = []
910 if aggregate {
911 if s.grouping.isEmpty {
912 groups=[rows]
913 }
914 else {
915 var indices: [[Value]:Int] = [:]
916 for r in rows {
917 let key=s.grouping.map {
918 r[$0.index]
919 }
920 if let i=indices[key] {
921 groups[i].append(r)
922 } else {
923 indices[key]=groups.count
924 groups.append([r])
925 }
926 }
927 }
928 } else {
929 groups=rows.map {
930 [$0]
931 }
932 }
933 var results: [(values:[Value],keys:[Value],ordinal:Int)] = []
934 for g in groups {
935 let row=g.first ?? Array(repeating:.null,count:scope.width)
936 if let e=s.having, !evaluate(e,row,g).truth {
937 continue
938 }
939 results.append((items.map {
940 evaluate($0.expr,row,g)
941 },keys.map {
942 evaluate($0,row,g)
943 },results.count))
944 }
945 if !keys.isEmpty {
946 results.sort {
947 a,b in for i in keys.indices {
948 let c=compare(a.keys[i],b.keys[i])
949 if c != 0 {
950 return s.orders[i].descending ? c > 0 : c < 0
951 }
952 }
953 return a.ordinal < b.ordinal
954 }
955 }
956 if s.distinct {
957 var seen=Set<[Value]>()
958 results=results.filter {
959 seen.insert($0.values).inserted
960 }
961 }
962 let selected=results.dropFirst(min(s.offset,results.count)).prefix(s.limit ?? results.count).map {
963 $0.values
964 }
965 printTable(items,types,selected)
966}

968func execute(_ s: Statement) throws {
969 if s.command == "" || s.command == "EXIT" {
970 return
971 }
972 if s.command == "SELECT" {
973 try executeSelect(s)
974 return
975 }
976 if s.command == "CREATE" {
977 if database[s.name] != nil {
978 try fail("table \(s.name) already exists")
979 }
980 var seen=Set<String>()
981 var keys=0
982 for c in s.columns {
983 if seen.contains(c.name) {
984 try fail("duplicate column \(c.name)")
985 }
986 seen.insert(c.name)
987 keys += c.keys
988 if keys > 1 {
989 try fail("multiple primary keys")
990 }
991 }
992 database[s.name]=Table(name:s.name,columns:s.columns)
993 print("Table \(s.name) created.")
994 return
995 }
996 var t=try lookup(s.name)
997 if s.command == "DROP" {
998 database.removeValue(forKey:s.name)
999 print("Table \(s.name) dropped.")
1000 return
1001 }
1002 var scope=Scope()
1003 try scope.add(t.name,t)
1004 if s.command == "INSERT" {
1005 let columns=try targets(s.names ?? t.columns.map {
1006 $0.name
1007 },t)
1008 for tuple in s.tuples {
1009 if tuple.count != columns.count {
1010 try fail("expected \(columns.count) values, got \(tuple.count)")
1011 }
1012 for (e,i) in zip(tuple,columns) {
1013 try require(check(e,Scope()),t.columns[i].type,"column \(t.columns[i].name)")
1014 }
1015 }
1016 let start=t.rows.count
1017 for tuple in s.tuples {
1018 var row=Array(repeating:Value.null,count:t.columns.count)
1019 for (e,i) in zip(tuple,columns) {
1020 row[i]=evaluate(e,[])
1021 }
1022 t.rows.append(row)
1023 }
1024 try constraints(t,Array(start..<t.rows.count))
1025 database[s.name]=t
1026 countMessage(s.tuples.count,"inserted")
1027 return
1028 }
1029 var assigned: [Int] = []
1030 if s.command == "UPDATE" {
1031 assigned=try targets(s.assignments.map {
1032 $0.0
1033 },t)
1034 for (assignment,i) in zip(s.assignments,assigned) {
1035 try require(check(assignment.1,scope),t.columns[i].type,"column \(t.columns[i].name)")
1036 }
1037 }
1038 try condition(s.whereExpr,scope,"WHERE")
1039 let affected=t.rows.indices.filter {
1040 s.whereExpr == nil || evaluate(s.whereExpr!,t.rows[$0]).truth
1041 }
1042 if s.command == "DELETE" {
1043 let removed=Set(affected)
1044 t.rows=t.rows.enumerated().filter {
1045 !removed.contains($0.offset)
1046 }.map {
1047 $0.element
1048 }
1049 }
1050 else {
1051 for i in affected {
1052 let original=t.rows[i]
1053 for (assignment,j) in zip(s.assignments,assigned) {
1054 t.rows[i][j]=evaluate(assignment.1,original)
1055 }
1056 }
1057 try constraints(t,affected)
1058 }
1059 database[s.name]=t
1060 countMessage(affected.count,s.command == "DELETE" ? "deleted" : "updated")
1061}
1062let input=Array(String(data:FileHandle.standardInput.readDataToEndOfFile(),encoding:.utf8)!)
1063var start=0
1064var i=0
1065var quoted=false
1066var comment=false
1067while i < input.count {
1068 let c=input[i]
1069 if comment {
1070 if c == "\n" {
1071 comment=false
1072 }
1073 }
1074 else if c == "'" {
1075 quoted.toggle()
1076 }
1077 else if !quoted && c == "-" && i+1 < input.count && input[i+1] == "-" {
1078 comment=true
1079 }
1080 else if !quoted && c == ";" {
1081 do {
1082 let s=try Parser(tokenize(String(input[start...i]))).parse()
1083 if s.command == "EXIT" {
1084 break
1085 }
1086 try execute(s)
1087 } catch let e as SqlError {
1088 print("ERROR: \(e.reason)")
1089 } catch {
1090 }
1091 start=i+1
1092 }
1093 i += 1
1094}
