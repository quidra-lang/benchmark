6enum NodeKind { Directory, File, Link }
7abstract class Node(int inode, int modified)
8{
9 public int Inode { get; } = inode;
10 public int Modified { get; set; } = modified;
11 public DirectoryNode Parent { get; set; } = null!;
12 public string Name { get; set; } = "";
13 public abstract NodeKind Kind { get; }
14 public abstract long Size { get; }
15 public virtual int Links => 1;
16 public char Tag => Kind switch { NodeKind.Directory => 'd', NodeKind.File => 'f', _ => 'l' };
17}
18sealed class DirectoryNode(int inode, int modified) : Node(inode, modified)
19{
20 public SortedDictionary<string, Node> Entries { get; } = new(StringComparer.Ordinal);
21 public override NodeKind Kind => NodeKind.Directory;
22 public override long Size => Entries.Count;
23 public override int Links => 2 + Entries.Values.Count(node => node is DirectoryNode);
24}
25sealed class FileNode(int inode, int modified, IEnumerable<string>? lines = null) : Node(inode, modified)
26{
27 public List<string> Lines { get; set; } = lines?.ToList() ?? [];
28 public override NodeKind Kind => NodeKind.File;
29 public override long Size => Lines.Sum(line => (long)line.Length + 1);
30}
31sealed class LinkNode(int inode, int modified, string target) : Node(inode, modified)
32{
33 public string Target { get; } = target;
34 public override NodeKind Kind => NodeKind.Link;
35 public override long Size => Target.Length;
36}
37enum Failure { None, Missing, NotDirectory, TooManyLinks }
38sealed record Resolution(Node? Node, Failure Error = Failure.None, DirectoryNode? CreationParent = null, string CreationName = "")
39{
40 public string Reason => Error switch
41 {
42 Failure.Missing => "No such file or directory",
43 Failure.NotDirectory => "Not a directory",
44 Failure.TooManyLinks => "Too many levels of symbolic links",
45 _ => ""
46 };
47}
48sealed record Options(HashSet<char> Flags, string[] Paths)
49{
50 public bool Has(char option) => Flags.Contains(option);
51 public static Options Parse(string[] arguments)
52 {
53 var flags = new HashSet<char>();
54 int position = 0;
55 while (position < arguments.Length && arguments[position].StartsWith('-'))
56 foreach (char flag in arguments[position++][1..]) flags.Add(flag);
57 return new(flags, arguments[position..]);
58 }
59}

76 string Absolute(Node node)
77 {
78 var names = new List<string>();
79 while (node != root) { names.Add(node.Name); node = node.Parent; }
80 names.Reverse();
81 return "/" + string.Join("/", names);
82 }
83 bool Descendant(Node node, Node ancestor)
84 {
85 while (true)
86 {
87 if (node == ancestor) return true;
88 if (node == root) return false;
89 node = node.Parent;
90 }

92 static string Join(string path, string name) => path == "/" ? "/" + name : path + "/" + name;
93 static string Last(string path) => path[(path.LastIndexOf('/') + 1)..];
94 static bool InvalidSource(string name) => name is "" or "." or "..";
95 void Add(DirectoryNode parent, string name, Node node)

97 parent.Entries[name] = node;

124 Resolution Resolve(string path, bool follow = true, bool makeParents = false)
125 {
126 int expansions = 0;
127 Resolution Walk(string text, DirectoryNode basis, bool finalLink, bool original)
128 {
129 Node node = text.StartsWith('/') ? root : basis;
130 string[] components = text.Split('/', StringSplitOptions.RemoveEmptyEntries);
131 for (int i = 0; i < components.Length; i++)
132 {
133 if (node is not DirectoryNode directory) return new(null, Failure.NotDirectory);
134 string name = components[i];
135 if (name == ".") continue;
136 if (name == "..") { node = node.Parent; continue; }
137 if (!directory.Entries.TryGetValue(name, out Node? child))
138 {
139 if (original && makeParents && i + 1 < components.Length) child = Create(directory, name, NodeKind.Directory);
140 else return new(null, Failure.Missing, original && i + 1 == components.Length ? directory : null, name);
141 }
142 if (child is LinkNode link && (i + 1 < components.Length || finalLink))
143 {
144 if (++expansions > 8) return new(null, Failure.TooManyLinks);
145 Resolution expanded = Walk(link.Target, directory, true, false);
146 if (expanded.Node is null) return expanded;
147 child = expanded.Node;
148 }
149 node = child;
150 }
151 return new(node);
152 }
153 return Walk(path, current, follow, true);
154 }
155 void Clone(Node source, DirectoryNode parent, string name)
156 {
157 Node clone = Create(parent, name, source.Kind, (source as FileNode)?.Lines, (source as LinkNode)?.Target ?? "");
158 if (source is DirectoryNode directory)
159 foreach (var (childName, child) in directory.Entries) Clone(child, (DirectoryNode)clone, childName);
160 }
161 static void WalkNodes(Node node, string path, Action<Node, string, int> visit, int depth = 0, int limit = 1000)
162 {
163 visit(node, path, depth);
164 if (node is DirectoryNode directory && depth < limit)
165 foreach (var (name, child) in directory.Entries) WalkNodes(child, Join(path, name), visit, depth + 1, limit);
166 }
167 void ReadInputs(string command, string[] paths, IReadOnlyList<string> input, Action<IReadOnlyList<string>, string?> consume)
168 {
169 if (paths.Length == 0) { consume(input, null); return; }
170 foreach (string path in paths)
171 {
172 Resolution result = Resolve(path);
173 if (result.Node is null) terminal($"{command}: {path}: {result.Reason}");
174 else if (result.Node is DirectoryNode) terminal($"{command}: {path}: Is a directory");
175 else consume(((FileNode)result.Node).Lines, path);
176 }

178 void Move(string[] paths)
179 {
180 string destination = paths[^1];
181 if (paths.Length > 2 && Resolve(destination).Node is not DirectoryNode)
182 {
183 terminal($"mv: target '{destination}' is not a directory");
184 return;
185 }
186 foreach (string sourcePath in paths[..^1])
187 {
188 string basename = Last(sourcePath);
189 void Problem(string displayPath, string reason) => terminal($"mv: cannot move '{sourcePath}' to '{displayPath}': {reason}");
190 if (InvalidSource(basename)) { Problem(destination, "Invalid argument"); continue; }
191 Resolution sourceResult = Resolve(sourcePath, false);
192 Node? source = sourceResult.Node;
193 if (source is null) { terminal($"mv: cannot stat '{sourcePath}': {sourceResult.Reason}"); continue; }
194 DirectoryNode parent;
195 string name, target;
196 if (Resolve(destination).Node is DirectoryNode directory)
197 {
198 parent = directory;
199 name = basename;
200 target = Join(destination, basename);
201 }
202 else
203 {
204 Resolution result = Resolve(destination, false);
205 if (result.Node is { } node) { parent = node.Parent; name = node.Name; }
206 else if (result.CreationParent is { } creation) { parent = creation; name = result.CreationName; }
207 else { Problem(destination, result.Reason); continue; }
208 target = destination;
209 }
210 parent.Entries.TryGetValue(name, out Node? existing);
211 if (source is DirectoryNode && Descendant(parent, source))
212 terminal($"mv: cannot move '{sourcePath}' to a subdirectory of itself, '{target}'");
213 else if (existing == source) terminal($"mv: '{sourcePath}' and '{target}' are the same file");
214 else if (existing is not null && source is DirectoryNode && existing is not DirectoryNode)
215 terminal($"mv: cannot overwrite non-directory '{target}' with directory '{sourcePath}'");
216 else if (source is not DirectoryNode && existing is DirectoryNode)
217 terminal($"mv: cannot overwrite directory '{target}' with non-directory '{sourcePath}'");
218 else if (existing is DirectoryNode occupied && occupied.Entries.Count > 0) Problem(target, "Directory not empty");
219 else if (existing == current) Problem(target, "Device or resource busy");
220 else
221 {
222 if (existing is not null) Detach(existing);
223 Detach(source);
224 Add(parent, name, source);
225 }
226 }

228 void Copy(Options options)
229 {
230 string[] paths = options.Paths;
231 string destination = paths[^1];
232 if (paths.Length > 2 && Resolve(destination).Node is not DirectoryNode)
233 {
234 terminal($"cp: target '{destination}' is not a directory");
235 return;
236 }
237 foreach (string sourcePath in paths[..^1])
238 {
239 string basename = Last(sourcePath);
240 void Problem(string displayPath, string reason) => terminal($"cp: cannot copy '{sourcePath}' to '{displayPath}': {reason}");
241 if (InvalidSource(basename)) { Problem(destination, "Invalid argument"); continue; }
242 Resolution sourceResult = Resolve(sourcePath);
243 Node? source = sourceResult.Node;
244 if (source is null) { terminal($"cp: cannot stat '{sourcePath}': {sourceResult.Reason}"); continue; }
245 if (source is DirectoryNode && !options.Has('r'))
246 {
247 terminal($"cp: -r not specified; omitting directory '{sourcePath}'");
248 continue;
249 }
250 Resolution destinationResult = Resolve(destination);
251 if (destinationResult.Node is null && destinationResult.CreationParent is null) { Problem(destination, destinationResult.Reason); continue; }
252 string target = destinationResult.Node is DirectoryNode ? Join(destination, basename) : destination;
253 Resolution result = Resolve(target, source is not DirectoryNode);
254 if (source is DirectoryNode)
255 {
256 if (result.Node is not null) Problem(target, "File exists");
257 else if (result.CreationParent is { } parent)
258 {
259 if (Descendant(parent, source)) terminal($"cp: cannot copy a directory, '{sourcePath}', into itself, '{target}'");
260 else Clone(source, parent, result.CreationName);
261 }
262 else Problem(target, result.Reason);
263 }
264 else if (result.CreationParent is { } creation) Create(creation, result.CreationName, NodeKind.File, ((FileNode)source).Lines);
265 else if (result.Node is null) Problem(target, result.Reason);
266 else if (result.Node == source) terminal($"cp: '{sourcePath}' and '{target}' are the same file");
267 else if (result.Node is DirectoryNode) terminal($"cp: cannot overwrite directory '{target}' with non-directory '{sourcePath}'");
268 else Write((FileNode)result.Node, ((FileNode)source).Lines);
269 }
270 }
271 void Listing(Options options, Action<string> output)
272 {
273 string[] paths = options.Paths.Length == 0 ? ["."] : options.Paths;
274 foreach (string path in paths)
275 {
276 Resolution result = Resolve(path, !options.Has('l'));
277 if (result.Node is null) { terminal($"ls: cannot access '{path}': {result.Reason}"); continue; }
278 Node node = result.Node;
279 var entries = new List<(string Name, Node Node)>();
280 if (node is not DirectoryNode directory) entries.Add((path, node));
281 else
282 {
283 if (paths.Length > 1) output(path + ":");
284 if (options.Has('a')) { entries.Add((".", directory)); entries.Add(("..", directory.Parent)); }
285 entries.AddRange(directory.Entries.Where(pair => options.Has('a') || !pair.Key.StartsWith('.')).Select(pair => (pair.Key, pair.Value)));
286 if (options.Has('t')) entries = entries.OrderByDescending(entry => entry.Node.Modified).ToList();
287 if (options.Has('r')) entries.Reverse();
288 }
289 static string[] Numbers(Node item) => [item.Links.ToString(), item.Size.ToString(), item.Modified.ToString()];
290 var widths = new int[3];
291 foreach (var entry in entries)
292 {
293 string[] values = Numbers(entry.Node);
294 for (int i = 0; i < 3; i++) widths[i] = Math.Max(widths[i], values[i].Length);
295 }
296 foreach (var (name, item) in entries)
297 {
298 if (!options.Has('l')) { output(name); continue; }
299 string line = item is FileNode ? "-" : item.Tag.ToString();
300 string[] values = Numbers(item);
301 for (int i = 0; i < 3; i++) line += " " + values[i].PadLeft(widths[i]);
302 line += " " + name;
303 if (item is LinkNode link) line += " -> " + link.Target;
304 output(line);
305 }
306 }

308 static bool GlobMatches(string pattern, string text)
309 {
310 var previous = new bool[text.Length + 1];
311 previous[0] = true;
312 foreach (char c in pattern)
313 {
314 var next = new bool[previous.Length];
315 if (c == '*')
316 {
317 next[0] = previous[0];
318 for (int i = 0; i < text.Length; i++) next[i + 1] = previous[i + 1] || next[i];
319 }
320 else
321 for (int i = 0; i < text.Length; i++) next[i + 1] = previous[i] && (c == '?' || c == text[i]);
322 previous = next;
323 }
324 return previous[^1];
325 }
326 void Find(string[] arguments, Action<string> output)
327 {
328 int position = arguments.Length > 0 && !arguments[0].StartsWith('-') ? 1 : 0;
329 string path = position == 1 ? arguments[0] : ".";
330 string? pattern = null, type = null;
331 int limit = 1000;
332 for (; position < arguments.Length; position += 2)
333 {
334 switch (arguments[position])
335 {
336 case "-name": pattern = arguments[position + 1]; break;
337 case "-type": type = arguments[position + 1]; break;
338 case "-maxdepth": limit = int.Parse(arguments[position + 1]); break;
339 }
340 }
341 Resolution result = Resolve(path);
342 if (result.Node is null) { terminal($"find: '{path}': {result.Reason}"); return; }
343 WalkNodes(result.Node, path, (node, display, depth) =>
344 {
345 string name = depth == 0 ? (path == "/" ? "/" : Last(path)) : node.Name;
346 if ((pattern is null || GlobMatches(pattern, name)) && (type is null || type[0] == node.Tag)) output(display);
347 }, limit: limit);
348 }
349 void DiskUsage(Options options, Action<string> output)
350 {
351 long Report(Node node, string path, bool show)
352 {
353 long usage = node is DirectoryNode ? 0 : node.Size;
354 if (node is DirectoryNode directory)
355 foreach (var (name, child) in directory.Entries)
356 usage += Report(child, Join(path, name), show && !options.Has('s') && (child is DirectoryNode || options.Has('a')));
357 if (show) output($"{usage} {path}");
358 return usage;
359 }
360 string[] paths = options.Paths.Length == 0 ? ["."] : options.Paths;
361 foreach (string path in paths)
362 {
363 Resolution result = Resolve(path);
364 if (result.Node is null) terminal($"du: cannot access '{path}': {result.Reason}");
365 else Report(result.Node, path, true);
366 }

368 void Tree(Options options, Action<string> output)
369 {
370 string path = options.Paths.FirstOrDefault() ?? ".";
371 Resolution result = Resolve(path);
372 if (result.Node is null) { terminal($"tree: {path}: {result.Reason}"); return; }
373 if (result.Node is not DirectoryNode directory) { terminal($"tree: {path}: Not a directory"); return; }
374 int directories = 0, files = 0;
375 void Draw(DirectoryNode parent, string prefix)
376 {
377 var entries = parent.Entries.Where(pair => options.Has('a') || !pair.Key.StartsWith('.')).ToArray();
378 for (int i = 0; i < entries.Length; i++)
379 {
380 var (name, child) = entries[i];
381 bool last = i + 1 == entries.Length;
382 string line = prefix + (last ? "`-- " : "|-- ") + name;
383 if (child is LinkNode link) line += " -> " + link.Target;
384 output(line);
385 if (child is DirectoryNode nested) { directories++; Draw(nested, prefix + (last ? " " : "| ")); }
386 else files++;
387 }
388 }
389 output(path);
390 Draw(directory, "");
391 output($"{directories} {(directories == 1 ? "directory" : "directories")}, {files} {(files == 1 ? "file" : "files")}");
392 }
393 void Execute(string text, IReadOnlyList<string> input, Action<string> output)
394 {
395 string[] words = text.Split(' ', StringSplitOptions.RemoveEmptyEntries);
396 string command = words[0];
397 string[] arguments = words[1..];
398 Options options = Options.Parse(arguments);
399 switch (command)
400 {
401 case "pwd": output(Absolute(current)); break;
402 case "echo": output(text[5..]); break;
403 case "cd":
404 {
405 string path = arguments.FirstOrDefault() ?? "/";
406 bool old = path == "-";
407 if (old)
408 {
409 if (previous is null) { terminal("cd: OLDPWD not set"); return; }
410 path = previous;
411 }
412 Resolution result = Resolve(path);
413 if (result.Node is null) terminal($"cd: {path}: {result.Reason}");
414 else if (result.Node is not DirectoryNode directory) terminal($"cd: {path}: Not a directory");
415 else
416 {
417 previous = Absolute(current);
418 current = directory;
419 if (old) output(Absolute(current));
420 }
421 break;
422 }
423 case "mkdir":
424 foreach (string path in options.Paths)
425 {
426 Resolution result = Resolve(path, false, options.Has('p'));
427 if (result.CreationParent is { } parent) Create(parent, result.CreationName, NodeKind.Directory);
428 else if (result.Node is null) terminal($"mkdir: cannot create directory '{path}': {result.Reason}");
429 else if (!options.Has('p') || Resolve(path).Node is not DirectoryNode)
430 terminal($"mkdir: cannot create directory '{path}': File exists");
431 }
432 break;
433 case "touch":
434 foreach (string path in arguments)
435 {
436 Resolution result = Resolve(path);
437 if (result.CreationParent is { } parent) Create(parent, result.CreationName, NodeKind.File);
438 else if (result.Node is { } node) node.Modified = clock;
439 else terminal($"touch: cannot touch '{path}': {result.Reason}");
440 }
441 break;
442 case "ls": Listing(options, output); break;
443 case "stat":
444 foreach (string path in arguments)
445 {
446 Resolution result = Resolve(path, false);
447 if (result.Node is null) { terminal($"stat: cannot stat '{path}': {result.Reason}"); continue; }
448 Node node = result.Node;
449 string type = node switch
450 {
451 DirectoryNode => "directory",
452 LinkNode => "symbolic link",
453 FileNode file => file.Lines.Count == 0 ? "regular empty file" : "regular file",
454 _ => throw new InvalidOperationException()
455 };
456 output($"{path}: {type}, size {node.Size}, inode {node.Inode}, links {node.Links}, modified {node.Modified}");
457 }
458 break;
459 case "rm":
460 foreach (string path in options.Paths)
461 {
462 void Problem(string reason) => terminal($"rm: cannot remove '{path}': {reason}");
463 if (Last(path) is "." or "..") { terminal($"rm: refusing to remove '.' or '..' directory: skipping '{path}'"); continue; }
464 Resolution result = Resolve(path, false);
465 if (result.Node is null)
466 {
467 if (!options.Has('f') || result.Error != Failure.Missing) Problem(result.Reason);
468 }
469 else if (result.Node is DirectoryNode && !options.Has('r')) Problem("Is a directory");
470 else if (Descendant(current, result.Node)) Problem("Device or resource busy");
471 else Detach(result.Node);
472 }
473 break;
474 case "rmdir":
475 foreach (string path in arguments)
476 {
477 void Problem(string reason) => terminal($"rmdir: failed to remove '{path}': {reason}");
478 if (Last(path) is "." or "..") { Problem("Invalid argument"); continue; }
479 Resolution result = Resolve(path, false);
480 if (result.Node is null) Problem(result.Reason);
481 else if (result.Node is not DirectoryNode directory) Problem("Not a directory");
482 else if (Descendant(current, directory)) Problem("Device or resource busy");
483 else if (directory.Entries.Count > 0) Problem("Directory not empty");
484 else Detach(directory);
485 }
486 break;
487 case "mv": Move(arguments); break;
488 case "cp": Copy(options); break;
489 case "ln":
490 {
491 string path = arguments[2];
492 Resolution result = Resolve(path, false);
493 if (result.CreationParent is { } parent) Create(parent, result.CreationName, NodeKind.Link, target: arguments[1]);
494 else terminal($"ln: failed to create symbolic link '{path}': {(result.Node is null ? result.Reason : "File exists")}");
495 break;
496 }
497 case "readlink":
498 case "realpath":
499 {
500 string path = arguments[0];
501 Resolution result = Resolve(path, command == "realpath");
502 if (result.Node is null) terminal($"{command}: {path}: {result.Reason}");
503 else if (command == "realpath") output(Absolute(result.Node));
504 else if (result.Node is LinkNode link) output(link.Target);
505 else terminal($"readlink: {path}: Invalid argument");
506 break;
507 }
508 case "find": Find(arguments, output); break;
509 case "du": DiskUsage(options, output); break;
510 case "tree": Tree(options, output); break;
511 case "cat":
512 case "wc":
513 case "head":
514 case "tail":
515 case "grep":
516 case "sort":
517 case "uniq": TextCommand(command, arguments, input, output); break;
518 default: terminal(command + ": command not found"); break;
519 }
520 }
521 static int NumericValue(string text)
522 {
523 int position = text.StartsWith('-') ? 1 : 0;
524 int value = 0;
525 while (position < text.Length && text[position] is >= '0' and <= '9') value = value * 10 + text[position++] - '0';
526 return text.StartsWith('-') ? -value : value;
527 }
528 void TextCommand(string command, string[] arguments, IReadOnlyList<string> input, Action<string> output)
529 {
530 Options options = Options.Parse(arguments);
531 string[] paths = options.Paths;
532 if (command == "grep")
533 {
534 string pattern = paths[0];
535 paths = paths[1..];
536 if (options.Has('i')) pattern = pattern.ToLowerInvariant();
537 bool named = paths.Length > 1 || options.Has('r');
538 void Consume(IReadOnlyList<string> lines, string? name)
539 {
540 string prefix = named ? name + ":" : "";
541 long count = 0;
542 for (int i = 0; i < lines.Count; i++)
543 {
544 string comparison = options.Has('i') ? lines[i].ToLowerInvariant() : lines[i];
545 if (comparison.Contains(pattern, StringComparison.Ordinal) == options.Has('v')) continue;
546 count++;
547 if (!options.Has('c')) output(prefix + (options.Has('n') ? $"{i + 1}:" : "") + lines[i]);
548 }
549 if (options.Has('c')) output(prefix + count);
550 }
551 if (!options.Has('r')) ReadInputs(command, paths, input, Consume);
552 else
553 {
554 if (paths.Length == 0) paths = ["."];
555 foreach (string path in paths)
556 {
557 Resolution result = Resolve(path);
558 if (result.Node is null) { terminal($"grep: {path}: {result.Reason}"); continue; }
559 WalkNodes(result.Node, path, (node, display, _) =>
560 {
561 if (node is FileNode file) Consume(file.Lines, display);
562 });
563 }
564 }
565 return;
566 }
567 if (command == "sort")
568 {
569 var lines = new List<string>();
570 ReadInputs(command, paths, input, (part, _) => lines.AddRange(part));
571 lines.Sort((a, b) =>
572 {
573 if (options.Has('n'))
574 {
575 int difference = NumericValue(a).CompareTo(NumericValue(b));
576 if (difference != 0) return difference;
577 }
578 return StringComparer.Ordinal.Compare(a, b);
579 });
580 if (options.Has('r')) lines.Reverse();
581 foreach (string line in lines) output(line);
582 return;
583 }
584 string countText = "10";
585 if (command is "head" or "tail")
586 {
587 int offset = arguments.Length > 0 && arguments[0] == "-n" ? 2 : 0;
588 if (offset > 0) countText = arguments[1];
589 paths = arguments[offset..];
590 }
591 var totals = new long[3];
592 void ReportCounts(long[] counts, string? name)
593 {
594 var values = new List<string>();
595 for (int i = 0; i < 3; i++)
596 if (options.Flags.Count == 0 || options.Has("lwc"[i])) values.Add(counts[i].ToString());
597 output(string.Join(" ", values) + (name is null ? "" : " " + name));
598 }
599 ReadInputs(command, paths, input, (lines, name) =>
600 {
601 if (command == "cat")
602 {
603 foreach (string line in lines) output(line);
604 }
605 else if (command == "wc")
606 {
607 long[] counts = [lines.Count, 0, 0];
608 foreach (string line in lines)
609 {
610 counts[2] += (long)line.Length + 1;
611 bool inside = false;
612 foreach (char c in line)
613 {
614 if (c != ' ' && !inside) counts[1]++;
615 inside = c != ' ';
616 }
617 }
618 for (int i = 0; i < 3; i++) totals[i] += counts[i];
619 ReportCounts(counts, name);
620 }
621 else if (command is "head" or "tail")
622 {
623 int count = int.Parse(countText[0] is '-' or '+' ? countText[1..] : countText);
624 int first = 0, last = lines.Count;
625 if (command == "head") last = countText[0] == '-' ? Math.Max(0, last - count) : Math.Min(last, count);
626 else first = countText[0] == '+' ? Math.Min(last, Math.Max(0, count - 1)) : Math.Max(0, last - count);
627 for (int i = first; i < last; i++) output(lines[i]);
628 }
