26enum class Kind { Directory, File, Link };
27struct Node {
28 int inode, modified;
29 std::weak_ptr<Node> parent;
30 std::string name;
31 Content content;
32 Node(int number, int time, Content data) : inode(number), modified(time), content(std::move(data)) {}
33 Kind kind() const { return static_cast<Kind>(content.index()); }
34 const std::map<std::string, NodePtr>& entries() const { return std::get<Directory>(content).entries; }
35 const Strings& lines() const { return std::get<File>(content).lines; }
36 const std::string& target() const { return std::get<Link>(content).target; }
37 Int size() const {
38 if (kind() == Kind::Directory) return static_cast<Int>(entries().size());
39 if (kind() == Kind::Link) return static_cast<Int>(target().size());
40 return std::accumulate(lines().begin(), lines().end(), Int{0}, [](Int total, const auto& line) { return total + static_cast<Int>(line.size()) + 1; });
41 }
42 int links() const {
43 if (kind() != Kind::Directory) return 1;
44 return 2 + static_cast<int>(std::count_if(entries().begin(), entries().end(), [](const auto& entry) { return entry.second->kind() == Kind::Directory; }));
45 }
46 char tag() const { return kind() == Kind::Directory ? 'd' : kind() == Kind::File ? 'f' : 'l'; }
47};
48enum class Reason { None, Missing, NotDirectory, TooManyLinks };
49std::string reasonText(Reason reason) {
50 switch (reason) {
51 case Reason::None: return "";
52 case Reason::Missing: return "No such file or directory";
53 case Reason::NotDirectory: return "Not a directory";
54 case Reason::TooManyLinks: return "Too many levels of symbolic links";
55 }
56 return "";
57}

64struct Options { std::set<char> flags; Strings paths; };
65Options options(const Strings& arguments) {
66 Options result;
67 std::size_t position = 0;
68 while (position < arguments.size() && !arguments[position].empty() && arguments[position][0] == '-') {
69 result.flags.insert(arguments[position].begin() + 1, arguments[position].end()); ++position;
70 }
71 result.paths.assign(arguments.begin() + position, arguments.end());
72 return result;
73}
74Strings split(const std::string& text, const std::string& separator) {
75 Strings result;
76 std::size_t position = 0;
77 while (true) {
78 auto end = text.find(separator, position);
79 result.push_back(text.substr(position, end == std::string::npos ? end : end - position));
80 if (end == std::string::npos) return result;
81 position = end + separator.size();
82 }
83}
84std::string join(const std::string& path, const std::string& name) { return path == "/" ? "/" + name : path + "/" + name; }
85std::string lastComponent(const std::string& path) {
86 auto position = path.find_last_of('/');
87 return position == std::string::npos ? path : path.substr(position + 1);
88}
89bool invalidSource(const std::string& name) { return name.empty() || name == "." || name == ".."; }
90bool globMatches(const std::string& pattern, const std::string& name) {
91 std::vector<bool> previous(name.size() + 1, false);
92 previous[0] = true;
93 for (char character : pattern) {
94 std::vector<bool> current(name.size() + 1, false);
95 if (character == '*') {
96 current[0] = previous[0];
97 for (std::size_t index = 1; index <= name.size(); ++index) current[index] = previous[index] || current[index - 1];
98 } else {
99 for (std::size_t index = 0; index < name.size(); ++index) current[index + 1] = previous[index] && (character == '?' || character == name[index]);
100 }
101 previous = std::move(current);
102 }
103 return previous.back();
104}
105std::string lower(std::string text) {
106 for (char& character : text) if ('A' <= character && character <= 'Z') character = static_cast<char>(character - 'A' + 'a');
107 return text;
108}

117 std::string absolute(NodePtr node) const {
118 Strings names;
119 while (node != root) { names.push_back(node->name); node = node->parent.lock(); }
120 std::string result;
121 for (auto name = names.rbegin(); name != names.rend(); ++name) result += "/" + *name;
122 return result.empty() ? "/" : result;
123 }
124 bool descendant(NodePtr node, const NodePtr& ancestor) const {
125 while (true) {
126 if (node == ancestor) return true;
127 if (node == root) return false;
128 node = node->parent.lock();
129 }
130 }
131 void add(const NodePtr& parent, const std::string& name, const NodePtr& node) {
132 std::get<Directory>(parent->content).entries[name] = node;
133 parent->modified = clock;
134 node->parent = parent; node->name = name;
135 }
136 NodePtr create(const NodePtr& parent, const std::string& name, Content content) {
137 auto node = std::make_shared<Node>(nextInode++, clock, std::move(content));
138 add(parent, name, node);
139 return node;
140 }
141 void detach(const NodePtr& node) {
142 auto parent = node->parent.lock();
143 std::get<Directory>(parent->content).entries.erase(node->name);
144 parent->modified = clock;
145 }
146 void write(const NodePtr& node, const Strings& lines, bool append = false) {
147 auto& content = std::get<File>(node->content).lines;
148 if (append) content.insert(content.end(), lines.begin(), lines.end());
149 else content = lines;
150 node->modified = clock;
151 }
152 Resolution resolve(const std::string& path, bool follow = true, bool makeParents = false) {
153 int expansions = 0;
154 std::function<Resolution(const std::string&, NodePtr, bool, bool)> walk;
155 walk = [&](const std::string& text, NodePtr base, bool finalLink, bool original) -> Resolution {
156 auto node = text[0] == '/' ? root : base;
157 auto components = split(text, "/");
158 components.erase(std::remove(components.begin(), components.end(), ""), components.end());
159 for (std::size_t index = 0; index < components.size(); ++index) {
160 if (node->kind() != Kind::Directory) return {nullptr, Reason::NotDirectory, nullptr, ""};
161 const auto& name = components[index];
162 if (name == ".") continue;
163 if (name == "..") { node = node->parent.lock(); continue; }
164 auto found = node->entries().find(name);
165 NodePtr child;
166 if (found != node->entries().end()) child = found->second;
167 else if (original && makeParents && index + 1 < components.size()) child = create(node, name, Directory{});
168 else return {nullptr, Reason::Missing, original && index + 1 == components.size() ? node : nullptr, name};
169 if (child->kind() == Kind::Link && (index + 1 < components.size() || finalLink)) {
170 if (++expansions > 8) return {nullptr, Reason::TooManyLinks, nullptr, ""};
171 auto expanded = walk(child->target(), node, true, false);
172 if (!expanded.node) return expanded;
173 child = expanded.node;
174 }
175 node = child;
176 }
177 return {node, Reason::None, nullptr, ""};
178 };
179 return walk(path, current, follow, true);
180 }
181 NodePtr clone(const NodePtr& source, const NodePtr& parent, const std::string& name) {
182 Content content = source->kind() == Kind::Directory ? Content{Directory{}} : source->content;
183 auto copy = create(parent, name, std::move(content));
184 if (source->kind() == Kind::Directory) {
185 for (const auto& [childName, child] : source->entries()) clone(child, copy, childName);
186 }
187 return copy;
188 }
189 template<class Visitor>
190 void walkNodes(const NodePtr& node, const std::string& path, int depth, int limit, const Visitor& visit) {
191 visit(node, path, depth);
192 if (node->kind() == Kind::Directory && depth < limit) {
193 for (const auto& [name, child] : node->entries()) walkNodes(child, join(path, name), depth + 1, limit, visit);
194 }
195 }
196 template<class Consumer>
197 void inputs(const std::string& command, const Strings& paths, const Strings& input, const Consumer& consume) {
198 if (paths.empty()) { consume(input, std::nullopt); return; }
199 for (const auto& path : paths) {
200 auto result = resolve(path);
201 if (!result.node) terminal(command + ": " + path + ": " + reasonText(result.reason));
202 else if (result.node->kind() == Kind::Directory) terminal(command + ": " + path + ": Is a directory");
203 else consume(result.node->lines(), std::optional<std::string>(path));
204 }
205 }
206 void move(const Strings& paths) {
207 const auto& destination = paths.back();
208 if (paths.size() > 2) {
209 auto result = resolve(destination);
210 if (!result.node || result.node->kind() != Kind::Directory) {
211 terminal("mv: target '" + destination + "' is not a directory"); return;
212 }
213 }
214 for (std::size_t index = 0; index + 1 < paths.size(); ++index) {
215 const auto& sourcePath = paths[index];
216 auto basename = lastComponent(sourcePath);
217 auto problem = [&](const std::string& target, const std::string& reason) { terminal("mv: cannot move '" + sourcePath + "' to '" + target + "': " + reason); };
218 if (invalidSource(basename)) { problem(destination, "Invalid argument"); continue; }
219 auto sourceResult = resolve(sourcePath, false);
220 if (!sourceResult.node) { terminal("mv: cannot stat '" + sourcePath + "': " + reasonText(sourceResult.reason)); continue; }
221 auto source = sourceResult.node, targetResult = resolve(destination).node;
222 NodePtr parent;
223 std::string name, target;
224 if (targetResult && targetResult->kind() == Kind::Directory) {
225 parent = targetResult; name = basename; target = join(destination, name);
226 } else {
227 auto result = resolve(destination, false);
228 if (result.node) { parent = result.node->parent.lock(); name = result.node->name; }
229 else if (result.creationParent) { parent = result.creationParent; name = result.creationName; }
230 else { problem(destination, reasonText(result.reason)); continue; }
231 target = destination;
232 }
233 auto found = parent->entries().find(name);
234 NodePtr existing = found == parent->entries().end() ? nullptr : found->second;
235 if (source->kind() == Kind::Directory && descendant(parent, source)) {
236 terminal("mv: cannot move '" + sourcePath + "' to a subdirectory of itself, '" + target + "'");
237 } else if (existing == source) {
238 terminal("mv: '" + sourcePath + "' and '" + target + "' are the same file");
239 } else if (existing && source->kind() == Kind::Directory && existing->kind() != Kind::Directory) {
240 terminal("mv: cannot overwrite non-directory '" + target + "' with directory '" + sourcePath + "'");
241 } else if (existing && source->kind() != Kind::Directory && existing->kind() == Kind::Directory) {
242 terminal("mv: cannot overwrite directory '" + target + "' with non-directory '" + sourcePath + "'");
243 } else if (existing && existing->kind() == Kind::Directory && !existing->entries().empty()) {
244 problem(target, "Directory not empty");
245 } else if (existing == current) {
246 problem(target, "Device or resource busy");
247 } else {
248 if (existing) detach(existing);
249 detach(source); add(parent, name, source);
250 }
251 }
252 }
253 void copy(const Options& parsed) {
254 const auto& paths = parsed.paths;
255 const auto& destination = paths.back();
256 if (paths.size() > 2) {
257 auto result = resolve(destination);
258 if (!result.node || result.node->kind() != Kind::Directory) {
259 terminal("cp: target '" + destination + "' is not a directory"); return;
260 }
261 }
262 for (std::size_t index = 0; index + 1 < paths.size(); ++index) {
263 const auto& sourcePath = paths[index];
264 auto basename = lastComponent(sourcePath);
265 auto problem = [&](const std::string& target, const std::string& reason) { terminal("cp: cannot copy '" + sourcePath + "' to '" + target + "': " + reason); };
266 if (invalidSource(basename)) { problem(destination, "Invalid argument"); continue; }
267 auto sourceResult = resolve(sourcePath);
268 if (!sourceResult.node) { terminal("cp: cannot stat '" + sourcePath + "': " + reasonText(sourceResult.reason)); continue; }
269 auto source = sourceResult.node;
270 if (source->kind() == Kind::Directory && !parsed.flags.count('r')) {
271 terminal("cp: -r not specified; omitting directory '" + sourcePath + "'"); continue;
272 }
273 auto destinationResult = resolve(destination);
274 if (!destinationResult.node && !destinationResult.creationParent) { problem(destination, reasonText(destinationResult.reason)); continue; }
275 auto target = destinationResult.node && destinationResult.node->kind() == Kind::Directory ? join(destination, basename) : destination;
276 auto result = resolve(target, source->kind() != Kind::Directory);
277 if (source->kind() == Kind::Directory) {
278 if (result.node) problem(target, "File exists");
279 else if (!result.creationParent) problem(target, reasonText(result.reason));
280 else if (descendant(result.creationParent, source)) terminal("cp: cannot copy a directory, '" + sourcePath + "', into itself, '" + target + "'");
281 else clone(source, result.creationParent, result.creationName);
282 } else {
283 if (result.creationParent) create(result.creationParent, result.creationName, File{source->lines()});
284 else if (!result.node) problem(target, reasonText(result.reason));
285 else if (result.node == source) terminal("cp: '" + sourcePath + "' and '" + target + "' are the same file");
286 else if (result.node->kind() == Kind::Directory) terminal("cp: cannot overwrite directory '" + target + "' with non-directory '" + sourcePath + "'");
287 else write(result.node, source->lines());
288 }
289 }
290 }
291 void listing(const Options& parsed, const Output& output) {
292 auto paths = parsed.paths.empty() ? Strings{"."} : parsed.paths;
293 for (const auto& path : paths) {
294 auto result = resolve(path, !parsed.flags.count('l'));
295 if (!result.node) { terminal("ls: cannot access '" + path + "': " + reasonText(result.reason)); continue; }
296 auto node = result.node;
297 std::vector<std::pair<std::string, NodePtr>> list;
298 if (node->kind() != Kind::Directory) list.emplace_back(path, node);
299 else {
300 if (paths.size() > 1) output(path + ":");
301 if (parsed.flags.count('a')) { list.emplace_back(".", node); list.emplace_back("..", node->parent.lock()); }
302 for (const auto& entry : node->entries()) if (parsed.flags.count('a') || entry.first[0] != '.') list.push_back(entry);
303 if (parsed.flags.count('t')) std::stable_sort(list.begin(), list.end(), [](const auto& left, const auto& right) { return left.second->modified > right.second->modified; });
304 if (parsed.flags.count('r')) std::reverse(list.begin(), list.end());
305 }
306 std::array<std::size_t, 3> widths{};
307 auto values = [](const NodePtr& item) { return std::array<std::string, 3>{std::to_string(item->links()), std::to_string(item->size()), std::to_string(item->modified)}; };
308 for (const auto& [name, item] : list) {
309 auto numbers = values(item);
310 for (std::size_t index = 0; index < widths.size(); ++index) widths[index] = std::max(widths[index], numbers[index].size());
311 }
312 for (const auto& [name, item] : list) {
313 if (!parsed.flags.count('l')) { output(name); continue; }
314 std::string line(1, item->kind() == Kind::File ? '-' : item->tag());
315 auto numbers = values(item);
316 for (std::size_t index = 0; index < widths.size(); ++index) line += " " + std::string(widths[index] - numbers[index].size(), ' ') + numbers[index];
317 line += " " + name;
318 if (item->kind() == Kind::Link) line += " -> " + item->target();
319 output(line);
320 }
321 }
322 }
323 void find(const Strings& arguments, const Output& output) {
324 std::size_t position = !arguments.empty() && arguments[0][0] != '-' ? 1 : 0;
325 std::string path = position ? arguments[0] : ".";
326 std::optional<std::string> pattern, type;
327 int limit = 1000;
328 for (; position < arguments.size(); position += 2) {
329 if (arguments[position] == "-name") pattern = arguments[position + 1];
330 else if (arguments[position] == "-type") type = arguments[position + 1];
331 else limit = std::stoi(arguments[position + 1]);
332 }
333 auto result = resolve(path);
334 if (!result.node) { terminal("find: '" + path + "': " + reasonText(result.reason)); return; }
335 walkNodes(result.node, path, 0, limit, [&](const NodePtr& node, const std::string& display, int depth) {
336 auto name = depth == 0 ? (path == "/" ? "/" : lastComponent(path)) : node->name;
337 if (pattern && !globMatches(*pattern, name)) return;
338 if (type && (*type)[0] != node->tag()) return;
339 output(display);
340 });
341 }
342 void diskUsage(const Options& parsed, const Output& output) {
343 auto paths = parsed.paths.empty() ? Strings{"."} : parsed.paths;
344 std::function<Int(const NodePtr&, const std::string&, bool)> report;
345 report = [&](const NodePtr& node, const std::string& path, bool show) {
346 Int usage = node->kind() == Kind::Directory ? 0 : node->size();
347 if (node->kind() == Kind::Directory) {
348 for (const auto& [name, child] : node->entries()) {
349 usage += report(child, join(path, name), show && !parsed.flags.count('s') && (child->kind() == Kind::Directory || parsed.flags.count('a')));
350 }
351 }
352 if (show) output(std::to_string(usage) + " " + path);
353 return usage;
354 };
355 for (const auto& path : paths) {
356 auto result = resolve(path);
357 if (!result.node) terminal("du: cannot access '" + path + "': " + reasonText(result.reason));
358 else report(result.node, path, true);
359 }
360 }
361 void tree(const Options& parsed, const Output& output) {
362 auto path = parsed.paths.empty() ? "." : parsed.paths[0];
363 auto result = resolve(path);
364 if (!result.node) { terminal("tree: " + path + ": " + reasonText(result.reason)); return; }
365 if (result.node->kind() != Kind::Directory) { terminal("tree: " + path + ": Not a directory"); return; }
366 int directories = 0, files = 0;
367 std::function<void(const NodePtr&, const std::string&)> draw;
368 draw = [&](const NodePtr& node, const std::string& prefix) {
369 std::vector<std::pair<std::string, NodePtr>> list;
370 for (const auto& entry : node->entries()) if (parsed.flags.count('a') || entry.first[0] != '.') list.push_back(entry);
371 for (std::size_t index = 0; index < list.size(); ++index) {
372 const auto& [name, child] = list[index];
373 bool last = index + 1 == list.size();
374 output(prefix + (last ? "`-- " : "|-- ") + name + (child->kind() == Kind::Link ? " -> " + child->target() : ""));
375 if (child->kind() == Kind::Directory) { ++directories; draw(child, prefix + (last ? " " : "| ")); }
376 else ++files;
377 }
378 };
379 output(path); draw(result.node, "");
380 output(std::to_string(directories) + (directories == 1 ? " directory, " : " directories, ") + std::to_string(files) + (files == 1 ? " file" : " files"));
381 }
382 void execute(const std::string& text, const Strings& input, const Output& output) {
383 std::istringstream words(text);
384 std::string command;
385 words >> command;
386 Strings arguments;
387 for (std::string word; words >> word;) arguments.push_back(word);
388 auto parsed = options(arguments);
389 if (command == "pwd") output(absolute(current));
390 else if (command == "echo") output(text.substr(text.find(' ') + 1));
391 else if (command == "cd") {
392 std::string path = arguments.empty() ? "/" : arguments[0];
393 bool old = path == "-";
394 if (old) {
395 if (!previous) { terminal("cd: OLDPWD not set"); return; }
396 path = *previous;
397 }
398 auto result = resolve(path);
399 if (!result.node) terminal("cd: " + path + ": " + reasonText(result.reason));
400 else if (result.node->kind() != Kind::Directory) terminal("cd: " + path + ": Not a directory");
401 else {
402 previous = absolute(current); current = result.node;
403 if (old) output(absolute(current));
404 }
405 } else if (command == "mkdir") {
406 for (const auto& path : parsed.paths) {
407 auto result = resolve(path, false, parsed.flags.count('p') != 0);
408 if (result.creationParent) create(result.creationParent, result.creationName, Directory{});
409 else if (!result.node) terminal("mkdir: cannot create directory '" + path + "': " + reasonText(result.reason));
410 else {
411 auto followed = parsed.flags.count('p') ? resolve(path).node : nullptr;
412 if (!followed || followed->kind() != Kind::Directory) terminal("mkdir: cannot create directory '" + path + "': File exists");
413 }
414 }
415 } else if (command == "touch") {
416 for (const auto& path : arguments) {
417 auto result = resolve(path);
418 if (result.creationParent) create(result.creationParent, result.creationName, File{});
419 else if (result.node) result.node->modified = clock;
420 else terminal("touch: cannot touch '" + path + "': " + reasonText(result.reason));
421 }
422 } else if (command == "ls") listing(parsed, output);
423 else if (command == "stat") {
424 for (const auto& path : arguments) {
425 auto result = resolve(path, false);
426 if (!result.node) { terminal("stat: cannot stat '" + path + "': " + reasonText(result.reason)); continue; }
427 auto node = result.node;
428 std::string kind = node->kind() == Kind::Directory ? "directory" : node->kind() == Kind::Link ? "symbolic link"
429 : node->lines().empty() ? "regular empty file" : "regular file";
430 output(path + ": " + kind + ", size " + std::to_string(node->size()) + ", inode " + std::to_string(node->inode)
431 + ", links " + std::to_string(node->links()) + ", modified " + std::to_string(node->modified));
432 }
433 } else if (command == "rm") {
434 for (const auto& path : parsed.paths) {
435 auto problem = [&](const std::string& reason) { terminal("rm: cannot remove '" + path + "': " + reason); };
436 auto last = lastComponent(path);
437 if (last == "." || last == "..") { terminal("rm: refusing to remove '.' or '..' directory: skipping '" + path + "'"); continue; }
438 auto result = resolve(path, false);
439 if (!result.node) { if (!(parsed.flags.count('f') && result.reason == Reason::Missing)) problem(reasonText(result.reason)); }
440 else if (result.node->kind() == Kind::Directory && !parsed.flags.count('r')) problem("Is a directory");
441 else if (descendant(current, result.node)) problem("Device or resource busy");
442 else detach(result.node);
443 }
444 } else if (command == "rmdir") {
445 for (const auto& path : arguments) {
446 auto problem = [&](const std::string& reason) { terminal("rmdir: failed to remove '" + path + "': " + reason); };
447 auto last = lastComponent(path);
448 if (last == "." || last == "..") { problem("Invalid argument"); continue; }
449 auto result = resolve(path, false);
450 if (!result.node) problem(reasonText(result.reason));
451 else if (result.node->kind() != Kind::Directory) problem("Not a directory");
452 else if (descendant(current, result.node)) problem("Device or resource busy");
453 else if (!result.node->entries().empty()) problem("Directory not empty");
454 else detach(result.node);
455 }
456 } else if (command == "mv") move(arguments);
457 else if (command == "cp") copy(parsed);
458 else if (command == "ln") {
459 const auto& path = arguments[2];
460 auto result = resolve(path, false);
461 if (result.creationParent) create(result.creationParent, result.creationName, Link{arguments[1]});
462 else terminal("ln: failed to create symbolic link '" + path + "': " + (result.node ? "File exists" : reasonText(result.reason)));
463 } else if (command == "readlink" || command == "realpath") {
464 const auto& path = arguments[0];
465 auto result = resolve(path, command == "realpath");
466 if (!result.node) terminal(command + ": " + path + ": " + reasonText(result.reason));
467 else if (command == "realpath") output(absolute(result.node));
468 else if (result.node->kind() == Kind::Link) output(result.node->target());
469 else terminal("readlink: " + path + ": Invalid argument");
470 } else if (command == "find") find(arguments, output);
471 else if (command == "du") diskUsage(parsed, output);
472 else if (command == "tree") tree(parsed, output);
473 else if (command == "cat" || command == "wc" || command == "head" || command == "tail" || command == "grep" || command == "sort" || command == "uniq") textCommand(command, arguments, input, output);
474 else terminal(command + ": command not found");
475 }
476 static int numericValue(const std::string& line) {
477 std::size_t position = !line.empty() && line[0] == '-' ? 1 : 0;
478 int value = 0;
479 while (position < line.size() && '0' <= line[position] && line[position] <= '9') value = value * 10 + line[position++] - '0';
480 return !line.empty() && line[0] == '-' ? -value : value;
481 }
482 void textCommand(const std::string& command, const Strings& arguments, const Strings& input, const Output& output) {
483 auto parsed = options(arguments);
484 const auto& flags = parsed.flags;
485 auto paths = parsed.paths;
486 if (command == "grep") {
487 auto pattern = paths.front();
488 paths.erase(paths.begin());
489 if (flags.count('i')) pattern = lower(pattern);
490 bool named = paths.size() > 1 || flags.count('r');
491 auto consume = [&](const Strings& lines, const std::optional<std::string>& name) {
492 auto prefix = named ? *name + ":" : "";
493 int count = 0;
494 for (std::size_t index = 0; index < lines.size(); ++index) {
495 auto comparison = flags.count('i') ? lower(lines[index]) : lines[index];
496 bool matches = comparison.find(pattern) != std::string::npos;
497 if (matches == (flags.count('v') != 0)) continue;
498 ++count;
499 if (!flags.count('c')) output(prefix + (flags.count('n') ? std::to_string(index + 1) + ":" : "") + lines[index]);
500 }
501 if (flags.count('c')) output(prefix + std::to_string(count));
502 };
503 if (!flags.count('r')) inputs(command, paths, input, consume);
504 else {
505 if (paths.empty()) paths.push_back(".");
506 for (const auto& path : paths) {
507 auto result = resolve(path);
508 if (!result.node) { terminal("grep: " + path + ": " + reasonText(result.reason)); continue; }
509 walkNodes(result.node, path, 0, 1000, [&](const NodePtr& node, const std::string& display, int) {
510 if (node->kind() == Kind::File) consume(node->lines(), display);
511 });
512 }
513 }
514 return;
515 }
516 if (command == "sort") {
517 Strings lines;
518 inputs(command, paths, input, [&](const Strings& part, const std::optional<std::string>&) { lines.insert(lines.end(), part.begin(), part.end()); });
519 std::sort(lines.begin(), lines.end(), [&](const auto& left, const auto& right) {
520 if (flags.count('n')) {
521 int a = numericValue(left), b = numericValue(right);
522 if (a != b) return a < b;
523 }
524 return left < right;
525 });
526 if (flags.count('r')) std::reverse(lines.begin(), lines.end());
527 for (const auto& line : lines) output(line);
528 return;
529 }
530 std::string countText = "10";
531 if (command == "head" || command == "tail") {
532 std::size_t position = !arguments.empty() && arguments[0] == "-n" ? 2 : 0;
533 if (position) countText = arguments[1];
534 paths.assign(arguments.begin() + position, arguments.end());
535 }
536 std::array<Int, 3> totals{};
537 auto reportCounts = [&](const std::array<Int, 3>& counts, const std::optional<std::string>& name) {
538 std::string line;
539 for (std::size_t index = 0; index < counts.size(); ++index) {
540 if (flags.empty() || flags.count("lwc"[index])) {
541 if (!line.empty()) line += " ";
542 line += std::to_string(counts[index]);
543 }
544 }
545 if (name) line += " " + *name;
546 output(line);
547 };
548 inputs(command, paths, input, [&](const Strings& lines, const std::optional<std::string>& name) {
549 if (command == "cat") {
550 for (const auto& line : lines) output(line);
551 } else if (command == "wc") {
552 std::array<Int, 3> counts{static_cast<Int>(lines.size()), 0, 0};
553 for (const auto& line : lines) {
554 counts[2] += static_cast<Int>(line.size()) + 1;
555 bool inside = false;
556 for (char character : line) {
557 if (character != ' ' && !inside) ++counts[1];
558 inside = character != ' ';
559 }
560 }
561 for (std::size_t index = 0; index < counts.size(); ++index) totals[index] += counts[index];
562 reportCounts(counts, name);
563 } else if (command == "head" || command == "tail") {
564 int count = std::stoi(countText[0] == '-' || countText[0] == '+' ? countText.substr(1) : countText);
565 int size = static_cast<int>(lines.size()), begin = 0, end = size;
566 if (command == "head") end = countText[0] == '-' ? std::max(0, size - count) : std::min(size, count);
567 else begin = countText[0] == '+' ? std::min(size, std::max(0, count - 1)) : std::max(0, size - count);
568 for (int index = begin; index < end; ++index) output(lines[index]);
569 } else {
570 for (std::size_t begin = 0; begin < lines.size();) {
571 auto end = begin + 1;
572 while (end < lines.size() && lines[end] == lines[begin]) ++end;
573 output((flags.count('c') ? std::to_string(end - begin) + " " : "") + lines[begin]);
574 begin = end;
575 }
576 }
577 });
578 if (command == "wc" && paths.size() > 1) reportCounts(totals, "total");
579 }
580
581public:
582 explicit Shell(Output print) : terminal(std::move(print)) { root->parent = root; }
583 void process(const std::string& line) {
584 ++clock;
585 terminal(absolute(current) + "$ " + line);
586 auto redirect = line.find(" >> ");
587 bool append = redirect != std::string::npos;
588 if (!append) redirect = line.find(" > ");
589 NodePtr redirectFile;
590 std::string pipeline = line;
591 if (redirect != std::string::npos) {
592 auto path = line.substr(redirect + (append ? 4 : 3));
593 pipeline = line.substr(0, redirect);
594 auto result = resolve(path);
595 if (result.creationParent) redirectFile = create(result.creationParent, result.creationName, File{});
596 else if (!result.node) { terminal("sh: " + path + ": " + reasonText(result.reason)); return; }
597 else if (result.node->kind() == Kind::Directory) { terminal("sh: " + path + ": Is a directory"); return; }
598 else {
599 redirectFile = result.node;
600 if (!append) write(redirectFile, {});
601 }
602 }
603 auto commands = split(pipeline, " | ");
604 Strings input;
605 for (std::size_t index = 0; index < commands.size(); ++index) {
606 Strings output;
607 execute(commands[index], input, [&](const std::string& text) {
608 if (index + 1 == commands.size() && !redirectFile) terminal(text);
609 else output.push_back(text);
610 });
611 input = std::move(output);
612 }
613 if (redirectFile && !input.empty()) write(redirectFile, input, true);
614 }
615};
