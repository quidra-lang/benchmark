101 static final class Shell {
102 final Directory root = new Directory(1, 0);
103 Directory current = root;
104 String previous;
105 int clock, nextInode = 2;

107 Shell() { root.parent = root; }
108 String absolute(Node node) {
109 List<String> parts = new ArrayList<>();
110 while (node != root) { parts.add(node.name); node = node.parent; }
111 Collections.reverse(parts);
112 return "/" + String.join("/", parts);
113 }
114 boolean descendant(Node node, Node ancestor) {
115 while (true) {
116 if (node == ancestor) return true;
117 if (node == root) return false;
118 node = node.parent;
119 }

122 parent.entries.put(name, node);
123 parent.modified = clock;
124 node.parent = parent;

147 final class Resolver {
148 final boolean parents;
149 int expansions;
150 Resolver(boolean parents) { this.parents = parents; }
151 Resolution walk(String path, Directory base, boolean follow, boolean original) {
152 Node node = path.startsWith("/") ? root : base;
153 String[] parts = path.split("/");
154 for (int i = 0; i < parts.length; i++) {
155 String part = parts[i];
156 if (part.isEmpty()) continue;
157 if (!(node instanceof Directory directory)) return Resolution.failed(Failure.NOT_DIRECTORY);
158 if (part.equals(".")) continue;
159 if (part.equals("..")) { node = node.parent; continue; }
160 boolean last = i + 1 == parts.length;
161 Node child = directory.entries.get(part);
162 if (child == null) {
163 if (original && last) return new Resolution(null, Failure.MISSING, directory, part);
164 if (original && parents) child = makeDirectory(directory, part);
165 else return Resolution.failed(Failure.MISSING);
166 }
167 if (child instanceof Link link && (!last || follow)) {
168 if (++expansions > 8) return Resolution.failed(Failure.LOOP);
169 Resolution result = walk(link.target, directory, true, false);
170 if (result.node == null) return result;
171 child = result.node;
172 }
173 node = child;
174 }
175 return Resolution.found(node);
176 }
177 }
178 Resolution resolve(String path, boolean follow, boolean parents) {
179 return new Resolver(parents).walk(path, current, follow, true);
180 }
181 Resolution resolve(String path, boolean follow) { return resolve(path, follow, false); }
182 Resolution resolve(String path) { return resolve(path, true); }

184 void cloneNode(Node source, Directory parent, String name) {
185 switch (source) {
186 case Directory directory -> {
187 Directory copy = makeDirectory(parent, name);
188 for (Node child : directory.entries.values()) cloneNode(child, copy, child.name);
189 }
190 case FileNode file -> makeFile(parent, name, file.lines);
191 case Link link -> add(parent, name, new Link(nextInode++, clock, link.target));
192 }
193 }
194 void move(String sourcePath, String destination, Output out) {
195 if (invalidSource(sourcePath)) {
196 out.error("mv: cannot move '%s' to '%s': Invalid argument", sourcePath, destination);
197 return;
198 }
199 Resolution result = resolve(sourcePath, false);
200 Node source = result.node;
201 if (source == null) { out.error("mv: cannot stat '%s': %s", sourcePath, result.reason()); return; }
202 Directory parent;
203 String name, target;
204 if (resolve(destination).node instanceof Directory directory) {
205 parent = directory;
206 name = lastComponent(sourcePath);
207 target = joinPath(destination, name);
208 } else {
209 result = resolve(destination, false);
210 if (result.absent()) { parent = result.creationParent; name = result.creationName; }
211 else if (result.node != null) { parent = result.node.parent; name = result.node.name; }
212 else { out.error("mv: cannot move '%s' to '%s': %s", sourcePath, destination, result.reason()); return; }
213 target = destination;
214 }
215 Node existing = parent.entries.get(name);
216 if (source instanceof Directory && descendant(parent, source)) {
217 out.error("mv: cannot move '%s' to a subdirectory of itself, '%s'", sourcePath, target);
218 } else if (existing == source) {
219 out.error("mv: '%s' and '%s' are the same file", sourcePath, target);
220 } else if (existing != null && source instanceof Directory && !(existing instanceof Directory)) {
221 out.error("mv: cannot overwrite non-directory '%s' with directory '%s'", target, sourcePath);
222 } else if (!(source instanceof Directory) && existing instanceof Directory) {
223 out.error("mv: cannot overwrite directory '%s' with non-directory '%s'", target, sourcePath);
224 } else if (existing instanceof Directory directory && !directory.entries.isEmpty()) {
225 out.error("mv: cannot move '%s' to '%s': Directory not empty", sourcePath, target);
226 } else if (existing == current) {
227 out.error("mv: cannot move '%s' to '%s': Device or resource busy", sourcePath, target);
228 } else {
229 if (existing != null) detach(existing);
230 detach(source);
231 add(parent, name, source);
232 }
233 }
234 void copy(String sourcePath, String destination, boolean recursive, Output out) {
235 if (invalidSource(sourcePath)) {
236 out.error("cp: cannot copy '%s' to '%s': Invalid argument", sourcePath, destination);
237 return;
238 }
239 Resolution result = resolve(sourcePath);
240 Node source = result.node;
241 if (source == null) { out.error("cp: cannot stat '%s': %s", sourcePath, result.reason()); return; }
242 if (source instanceof Directory && !recursive) {
243 out.error("cp: -r not specified; omitting directory '%s'", sourcePath);
244 return;
245 }
246 result = resolve(destination);
247 if (result.node == null && !result.absent()) {
248 out.error("cp: cannot copy '%s' to '%s': %s", sourcePath, destination, result.reason());
249 return;
250 }
251 String target = result.node instanceof Directory ? joinPath(destination, lastComponent(sourcePath)) : destination;
252 result = resolve(target, !(source instanceof Directory));
253 if (source instanceof Directory) {
254 if (result.node != null) out.error("cp: cannot copy '%s' to '%s': File exists", sourcePath, target);
255 else if (descendant(result.creationParent, source)) {
256 out.error("cp: cannot copy a directory, '%s', into itself, '%s'", sourcePath, target);
257 } else cloneNode(source, result.creationParent, result.creationName);
258 } else if (result.absent()) {
259 makeFile(result.creationParent, result.creationName, ((FileNode) source).lines);
260 } else if (result.node == source) {
261 out.error("cp: '%s' and '%s' are the same file", sourcePath, target);
262 } else if (result.node instanceof Directory) {
263 out.error("cp: cannot overwrite directory '%s' with non-directory '%s'", target, sourcePath);
264 } else if (result.node instanceof FileNode file) {
265 write(file, ((FileNode) source).lines, false);
266 } else out.error("cp: cannot copy '%s' to '%s': %s", sourcePath, target, result.reason());
267 }

269 void listing(Options options, Output out) {
270 List<String> paths = options.paths.isEmpty() ? List.of(".") : options.paths;
271 for (String path : paths) {
272 Resolution result = resolve(path, !options.has('l'));
273 if (result.node == null) { out.error("ls: cannot access '%s': %s", path, result.reason()); continue; }
274 List<Listed> entries = new ArrayList<>();
275 if (result.node instanceof Directory directory) {
276 if (paths.size() > 1) out.emit(path + ":");
277 if (options.has('a')) {
278 entries.add(new Listed(".", directory));
279 entries.add(new Listed("..", directory.parent));
280 }
281 for (Node child : directory.entries.values()) {
282 if (options.has('a') || !child.name.startsWith(".")) entries.add(new Listed(child.name, child));
283 }
284 } else entries.add(new Listed(path, result.node));
285 if (options.has('t')) entries.sort(Comparator.comparingInt((Listed entry) -> entry.node.modified).reversed());
286 if (options.has('r')) Collections.reverse(entries);
287 int[] widths = {1, 1, 1};
288 for (Listed entry : entries) {
289 long[] numbers = {entry.node.links(), entry.node.size(), entry.node.modified};
290 for (int i = 0; i < 3; i++) widths[i] = Math.max(widths[i], Long.toString(numbers[i]).length());
291 }
292 for (Listed entry : entries) {
293 if (!options.has('l')) { out.emit(entry.name); continue; }
294 Node node = entry.node;
295 String format = "%c %" + widths[0] + "d %" + widths[1] + "d %" + widths[2] + "d %s%s";
296 out.print(format, node instanceof FileNode ? '-' : node.tag(), node.links(), node.size(), node.modified,
297 entry.name, node instanceof Link link ? " -> " + link.target : "");
298 }
299 }

301 void walk(Node node, String path, int depth, int limit, Visitor visit) {
302 visit.accept(node, path, depth);
303 if (node instanceof Directory directory && depth < limit) {
304 for (Node child : directory.entries.values()) walk(child, joinPath(path, child.name), depth + 1, limit, visit);
305 }
306 }
307 void find(List<String> args, Output out) {
308 boolean explicit = !args.isEmpty() && !args.getFirst().startsWith("-");
309 String path = explicit ? args.getFirst() : ".";
310 Map<String, String> tests = new HashMap<>();
311 for (int i = explicit ? 1 : 0; i < args.size(); i += 2) tests.put(args.get(i), args.get(i + 1));
312 Resolution result = resolve(path);
313 if (result.node == null) { out.error("find: '%s': %s", path, result.reason()); return; }
314 int limit = Integer.parseInt(tests.getOrDefault("-maxdepth", "1000"));
315 walk(result.node, path, 0, limit, (node, display, depth) -> {
316 String name = depth == 0 ? (path.equals("/") ? "/" : lastComponent(path)) : node.name;
317 if ((!tests.containsKey("-name") || globMatch(tests.get("-name"), name)) &&
318 (!tests.containsKey("-type") || tests.get("-type").charAt(0) == node.tag())) out.emit(display);
319 });
320 }

340 void drawTree(Directory directory, String prefix, boolean all, int[] counts, Output out) {
341 List<Node> children = directory.entries.values().stream().filter(node -> all || !node.name.startsWith(".")).toList();
342 for (int i = 0; i < children.size(); i++) {
343 Node child = children.get(i);
344 boolean last = i + 1 == children.size();
345 out.emit(prefix + (last ? "`-- " : "|-- ") + child.name + (child instanceof Link link ? " -> " + link.target : ""));
346 if (child instanceof Directory nested) {
347 counts[0]++;
348 drawTree(nested, prefix + (last ? " " : "| "), all, counts, out);
349 } else counts[1]++;
350 }

352 void tree(Options options, Output out) {
353 String path = options.paths.isEmpty() ? "." : options.paths.getFirst();
354 Resolution result = resolve(path);
355 if (result.node == null) { out.error("tree: %s: %s", path, result.reason()); return; }
356 if (!(result.node instanceof Directory directory)) { out.error("tree: %s: Not a directory", path); return; }
357 out.emit(path);
358 int[] counts = {0, 0};
359 drawTree(directory, "", options.has('a'), counts, out);

361 counts[1], counts[1] == 1 ? "file" : "files");
362 }
363 void readInputs(String command, List<String> paths, List<String> input, Output out, Consumer<Input> consume) {
364 if (paths.isEmpty()) { consume.accept(new Input(input, null)); return; }
365 for (String path : paths) {
366 Resolution result = resolve(path);
367 if (result.node == null) out.error("%s: %s: %s", command, path, result.reason());
368 else if (result.node instanceof FileNode file) consume.accept(new Input(file.lines, path));
369 else out.error("%s: %s: Is a directory", command, path);
370 }

372 void textCommand(String command, List<String> args, List<String> input, Output out) {
373 Options options = options(args);
374 switch (command) {
375 case "grep" -> {
376 String pattern = options.has('i') ? options.paths.getFirst().toLowerCase(Locale.ROOT) : options.paths.getFirst();
377 List<String> paths = options.paths.subList(1, options.paths.size());
378 boolean named = options.has('r') || paths.size() > 1;
379 Consumer<Input> consume = part -> {
380 String prefix = named ? part.name + ":" : "";
381 long selected = 0;
382 for (int i = 0; i < part.lines.size(); i++) {
383 String line = part.lines.get(i);
384 String comparison = options.has('i') ? line.toLowerCase(Locale.ROOT) : line;
385 if (comparison.contains(pattern) == options.has('v')) continue;
386 selected++;
387 if (!options.has('c')) out.emit(prefix + (options.has('n') ? (i + 1) + ":" : "") + line);
388 }
389 if (options.has('c')) out.emit(prefix + selected);
390 };
391 if (!options.has('r')) readInputs(command, paths, input, out, consume);
392 else for (String path : paths.isEmpty() ? List.of(".") : paths) {
393 Resolution result = resolve(path);
394 if (result.node == null) out.error("grep: %s: %s", path, result.reason());
395 else walk(result.node, path, 0, 1000, (node, display, depth) -> {
396 if (node instanceof FileNode file) consume.accept(new Input(file.lines, display));
397 });
398 }
399 }
400 case "sort" -> {
401 List<String> lines = new ArrayList<>();
402 readInputs(command, options.paths, input, out, part -> lines.addAll(part.lines));
403 lines.sort((a, b) -> {
404 int comparison = options.has('n') ? Long.compare(numericPrefix(a), numericPrefix(b)) : 0;
405 return comparison != 0 ? comparison : a.compareTo(b);
406 });
407 if (options.has('r')) Collections.reverse(lines);
408 lines.forEach(out::emit);
409 }
410 case "wc" -> {
411 boolean all = !options.has('l') && !options.has('w') && !options.has('c');
412 long[] totals = new long[3];
413 java.util.function.BiConsumer<long[], String> report = (counts, name) -> {
414 StringJoiner numbers = new StringJoiner(" ");
415 for (int i = 0; i < 3; i++) if (all || options.has("lwc".charAt(i))) numbers.add(Long.toString(counts[i]));
416 out.emit(numbers + (name == null ? "" : " " + name));
417 };
418 readInputs(command, options.paths, input, out, part -> {
419 long[] counts = {part.lines.size(), 0, 0};
420 for (String line : part.lines) {
421 counts[2] += (long) line.length() + 1;
422 boolean inside = false;
423 for (int i = 0; i < line.length(); i++) {
424 if (line.charAt(i) != ' ' && !inside) counts[1]++;
425 inside = line.charAt(i) != ' ';
426 }
427 }
428 for (int i = 0; i < 3; i++) totals[i] += counts[i];
429 report.accept(counts, part.name);
430 });
431 if (options.paths.size() > 1) report.accept(totals, "total");
432 }
433 case "head", "tail" -> {
434 boolean explicit = !args.isEmpty() && args.getFirst().equals("-n");
435 String raw = explicit ? args.get(1) : "10";
436 boolean special = raw.startsWith("-") || raw.startsWith("+");
437 int count = Integer.parseInt(special ? raw.substring(1) : raw);
438 readInputs(command, args.subList(explicit ? 2 : 0, args.size()), input, out, part -> {
439 int first = 0, end = part.lines.size();
440 if (command.equals("head")) end = special ? Math.max(0, end - count) : Math.min(end, count);
441 else first = special ? Math.min(end, Math.max(0, count - 1)) : Math.max(0, end - count);
442 for (int i = first; i < end; i++) out.emit(part.lines.get(i));
443 });
444 }
445 default -> readInputs(command, options.paths, input, out, part -> {
446 if (command.equals("cat")) { part.lines.forEach(out::emit); return; }
447 for (int first = 0; first < part.lines.size();) {
448 int end = first + 1;
449 while (end < part.lines.size() && part.lines.get(first).equals(part.lines.get(end))) end++;
450 out.emit((options.has('c') ? (end - first) + " " : "") + part.lines.get(first));
451 first = end;
452 }
453 });
454 }

456 void execute(String text, List<String> input, Output out) {
457 List<String> words = Arrays.asList(text.split(" "));
458 String command = words.getFirst();
459 List<String> args = words.subList(1, words.size());
460 Options options = options(args);
461 switch (command) {
462 case "pwd" -> out.emit(absolute(current));
463 case "echo" -> out.emit(text.substring(5));
464 case "cd" -> {
465 String path = args.isEmpty() ? "/" : args.getFirst();
466 boolean back = path.equals("-");
467 if (back) {
468 if (previous == null) { out.error("cd: OLDPWD not set"); return; }
469 path = previous;
470 }
471 Resolution result = resolve(path);
472 if (result.node == null) out.error("cd: %s: %s", path, result.reason());
473 else if (!(result.node instanceof Directory directory)) out.error("cd: %s: Not a directory", path);
474 else {
475 previous = absolute(current);
476 current = directory;
477 if (back) out.emit(absolute(current));
478 }
479 }
480 case "mkdir" -> {
481 for (String path : options.paths) {
482 Resolution result = resolve(path, false, options.has('p'));
483 if (result.absent()) makeDirectory(result.creationParent, result.creationName);
484 else if (result.node == null) out.error("mkdir: cannot create directory '%s': %s", path, result.reason());
485 else if (!options.has('p') || !(resolve(path).node instanceof Directory)) {
486 out.error("mkdir: cannot create directory '%s': File exists", path);
487 }
488 }
489 }
490 case "touch" -> {
491 for (String path : args) {
492 Resolution result = resolve(path);
493 if (result.absent()) makeFile(result.creationParent, result.creationName, List.of());
494 else if (result.node != null) result.node.modified = clock;
495 else out.error("touch: cannot touch '%s': %s", path, result.reason());
496 }
497 }
498 case "ls" -> listing(options, out);
499 case "stat" -> {
500 for (String path : args) {
501 Resolution result = resolve(path, false);
502 Node node = result.node;
503 if (node == null) { out.error("stat: cannot stat '%s': %s", path, result.reason()); continue; }
504 String kind = switch (node) {
505 case Directory directory -> "directory";
506 case Link link -> "symbolic link";
507 case FileNode file -> file.lines.isEmpty() ? "regular empty file" : "regular file";
508 };
509 out.print("%s: %s, size %d, inode %d, links %d, modified %d", path, kind, node.size(), node.inode, node.links(), node.modified);
510 }
511 }
512 case "rm", "rmdir" -> {
513 boolean directoryOnly = command.equals("rmdir");
514 for (String path : options.paths) {
515 String last = lastComponent(path);
516 boolean dot = last.equals(".") || last.equals("..");
517 if (dot && !directoryOnly) {
518 out.error("rm: refusing to remove '.' or '..' directory: skipping '%s'", path);
519 continue;
520 }
521 String problem = "";
522 if (dot) problem = "Invalid argument";
523 else {
524 Resolution result = resolve(path, false);
525 Node node = result.node;
526 if (node == null) {
527 if (!directoryOnly && options.has('f') && result.failure == Failure.MISSING) continue;
528 problem = result.reason();
529 } else if (directoryOnly && !(node instanceof Directory)) problem = "Not a directory";
530 else if (!directoryOnly && node instanceof Directory && !options.has('r')) problem = "Is a directory";
531 else if (descendant(current, node)) problem = "Device or resource busy";
532 else if (directoryOnly && !((Directory) node).entries.isEmpty()) problem = "Directory not empty";
533 else detach(node);
534 }
535 if (!problem.isEmpty()) out.error(directoryOnly ? "rmdir: failed to remove '%s': %s" : "rm: cannot remove '%s': %s", path, problem);
536 }
537 }
538 case "mv", "cp" -> {
539 String destination = options.paths.getLast();
540 if (options.paths.size() > 2 && !(resolve(destination).node instanceof Directory)) {
541 out.error("%s: target '%s' is not a directory", command, destination);
542 return;
543 }
544 for (String source : options.paths.subList(0, options.paths.size() - 1)) {
545 if (command.equals("mv")) move(source, destination, out);
546 else copy(source, destination, options.has('r'), out);
547 }
548 }
549 case "ln" -> {
550 String path = args.get(2);
551 Resolution result = resolve(path, false);
552 if (result.absent()) add(result.creationParent, result.creationName, new Link(nextInode++, clock, args.get(1)));
553 else out.error("ln: failed to create symbolic link '%s': %s", path, result.node == null ? result.reason() : "File exists");
554 }
555 case "readlink", "realpath" -> {
556 String path = args.getFirst();
557 Resolution result = resolve(path, command.equals("realpath"));
558 if (result.node == null) out.error("%s: %s: %s", command, path, result.reason());
559 else if (command.equals("realpath")) out.emit(absolute(result.node));
560 else if (result.node instanceof Link link) out.emit(link.target);
561 else out.error("readlink: %s: Invalid argument", path);
562 }
563 case "find" -> find(args, out);
564 case "du" -> diskUsage(options, out);
565 case "tree" -> tree(options, out);
566 case "cat", "wc", "head", "tail", "grep", "sort", "uniq" -> textCommand(command, args, input, out);
567 default -> out.error("%s: command not found", command);
568 }
569 }
570 void process(String line, PrintWriter terminal) {
571 clock++;
572 terminal.print(absolute(current) + "$ " + line + "\n");
573 String pipeline = line;
574 FileNode redirect = null;
575 int marker = line.indexOf(" >");
576 if (marker >= 0) {
577 pipeline = line.substring(0, marker);
578 boolean append = line.startsWith(" >> ", marker);
579 String path = line.substring(marker + (append ? 4 : 3));
580 Resolution result = resolve(path);
581 if (result.absent()) redirect = makeFile(result.creationParent, result.creationName, List.of());
582 else if (result.node instanceof FileNode file) {
583 redirect = file;
584 if (!append) write(file, List.of(), false);
585 } else {
586 terminal.print("sh: " + path + ": " + (result.node == null ? result.reason() : "Is a directory") + "\n");
587 return;
588 }
589 }
590 String[] commands = pipeline.split(" \\| ");
591 List<String> input = List.of();
592 for (int i = 0; i < commands.length; i++) {
593 Output out = new Output(terminal, i + 1 == commands.length && redirect == null);
594 execute(commands[i], input, out);
595 input = out.lines;
596 }
597 if (redirect != null && !input.isEmpty()) write(redirect, input, true);
598 }
599 }

601 static boolean globMatch(String pattern, String text) {
602 boolean[] previous = new boolean[text.length() + 1];
603 previous[0] = true;
604 for (char character : pattern.toCharArray()) {
605 boolean[] next = new boolean[previous.length];
606 next[0] = character == '*' && previous[0];
607 for (int i = 0; i < text.length(); i++) {
608 next[i + 1] = character == '*' ? previous[i + 1] || next[i] :
609 previous[i] && (character == '?' || character == text.charAt(i));
610 }
611 previous = next;
612 }
613 return previous[text.length()];
614 }
615 static long numericPrefix(String text) {
616 boolean negative = text.startsWith("-");
617 int at = negative ? 1 : 0;
618 long value = 0;
619 while (at < text.length() && text.charAt(at) >= '0' && text.charAt(at) <= '9') value = value * 10 + text.charAt(at++) - '0';
620 return negative ? -value : value;
621 }
