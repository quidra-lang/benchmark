5private sealed class Node(val inode: Int, var modified: Int) {
6 lateinit var parent: Directory
7 var name = ""
8 abstract val size: Long
9 abstract val tag: Char
10 open val links: Int get() = 1
11}

13private class Directory(inode: Int, modified: Int) : Node(inode, modified) {
14 val entries = TreeMap<String, Node>()
15 override val size: Long get() = entries.size.toLong()
16 override val tag = 'd'
17 override val links: Int get() = 2 + entries.values.count { it is Directory }
18}

20private class FileNode(inode: Int, modified: Int, lines: List<String> = emptyList()) : Node(inode, modified) {
21 var lines = lines.toMutableList()
22 override val size: Long get() = lines.sumOf { it.length.toLong() + 1 }
23 override val tag = 'f'
24}

26private class Link(inode: Int, modified: Int, val target: String) : Node(inode, modified) {
27 override val size: Long get() = target.length.toLong()
28 override val tag = 'l'
29}

69private fun lastComponent(path: String) = path.substringAfterLast('/')
70private fun invalidSource(path: String) = lastComponent(path) in listOf("", ".", "..")
71private fun joinPath(path: String, name: String) = if (path == "/") "/$name" else "$path/$name"

80 fun absolute(node: Node): String {
81 val names = mutableListOf<String>()
82 var cursor = node
83 while (cursor !== root) { names.add(cursor.name); cursor = cursor.parent }
84 return "/" + names.asReversed().joinToString("/")
85 }
86 fun descendant(node: Node, ancestor: Node): Boolean {
87 var cursor = node
88 while (true) {
89 if (cursor === ancestor) return true
90 if (cursor === root) return false
91 cursor = cursor.parent
92 }
93 }
94 fun add(parent: Directory, name: String, node: Node) {
95 parent.entries[name] = node
96 parent.modified = clock
97 node.parent = parent
98 node.name = name
99 }
100 fun makeDirectory(parent: Directory, name: String): Directory = Directory(nextInode++, clock).also { add(parent, name, it) }
101 fun makeFile(parent: Directory, name: String, lines: List<String> = emptyList()): FileNode =
102 FileNode(nextInode++, clock, lines).also { add(parent, name, it) }
103 fun detach(node: Node) {
104 node.parent.entries.remove(node.name)
105 node.parent.modified = clock
106 }
107 fun write(file: FileNode, lines: List<String>, append: Boolean = false) {
108 if (append) file.lines.addAll(lines) else file.lines = lines.toMutableList()
109 file.modified = clock
110 }

112 private inner class Resolver(val parents: Boolean) {
113 var expansions = 0
114 fun walk(path: String, base: Directory, follow: Boolean, original: Boolean): Resolution {
115 var node: Node = if (path.startsWith('/')) root else base
116 val parts = path.split('/')
117 for ((i, part) in parts.withIndex()) {
118 if (part.isEmpty()) continue
119 val directory = node as? Directory ?: return Resolution(failure = Failure.NOT_DIRECTORY)
120 if (part == ".") continue
121 if (part == "..") { node = node.parent; continue }
122 val last = i + 1 == parts.size
123 var child = directory.entries[part]
124 if (child == null) {
125 if (original && last) return Resolution(failure = Failure.MISSING, creationParent = directory, creationName = part)
126 if (original && parents) child = makeDirectory(directory, part)
127 else return Resolution(failure = Failure.MISSING)
128 }
129 if (child is Link && (!last || follow)) {
130 if (++expansions > 8) return Resolution(failure = Failure.LOOP)
131 val result = walk(child.target, directory, true, false)
132 child = result.node ?: return result
133 }
134 node = child
135 }
136 return Resolution(node)
137 }
138 }
139 fun resolve(path: String, follow: Boolean = true, parents: Boolean = false): Resolution =
140 Resolver(parents).walk(path, current, follow, true)

142 fun cloneNode(source: Node, parent: Directory, name: String) {
143 when (source) {
144 is Directory -> {
145 val copy = makeDirectory(parent, name)
146 for (child in source.entries.values) cloneNode(child, copy, child.name)
147 }
148 is FileNode -> makeFile(parent, name, source.lines)
149 is Link -> add(parent, name, Link(nextInode++, clock, source.target))
150 }
151 }
152 fun move(sourcePath: String, destination: String, out: Output) {
153 fun problem(path: String, reason: String) = out.error("mv: cannot move '%s' to '%s': %s", sourcePath, path, reason)
154 if (invalidSource(sourcePath)) { problem(destination, "Invalid argument"); return }
155 var result = resolve(sourcePath, false)
156 val source = result.node
157 if (source == null) { out.error("mv: cannot stat '%s': %s", sourcePath, result.reason); return }
158 val parent: Directory
159 val name: String
160 val target: String
161 val destinationNode = resolve(destination).node
162 if (destinationNode is Directory) {
163 parent = destinationNode
164 name = lastComponent(sourcePath)
165 target = joinPath(destination, name)
166 } else {
167 result = resolve(destination, false)
168 val existing = result.node
169 if (result.absent) { parent = result.creationParent!!; name = result.creationName }
170 else if (existing != null) { parent = existing.parent; name = existing.name }
171 else { problem(destination, result.reason); return }
172 target = destination
173 }
174 val existing = parent.entries[name]
175 when {
176 source is Directory && descendant(parent, source) -> out.error("mv: cannot move '%s' to a subdirectory of itself, '%s'", sourcePath, target)
177 existing === source -> out.error("mv: '%s' and '%s' are the same file", sourcePath, target)
178 existing != null && source is Directory && existing !is Directory -> out.error("mv: cannot overwrite non-directory '%s' with directory '%s'", target, sourcePath)
179 source !is Directory && existing is Directory -> out.error("mv: cannot overwrite directory '%s' with non-directory '%s'", target, sourcePath)
180 existing is Directory && existing.entries.isNotEmpty() -> problem(target, "Directory not empty")
181 existing === current -> problem(target, "Device or resource busy")
182 else -> {
183 if (existing != null) detach(existing)
184 detach(source)
185 add(parent, name, source)
186 }
187 }
188 }
189 fun copy(sourcePath: String, destination: String, recursive: Boolean, out: Output) {
190 fun problem(path: String, reason: String) = out.error("cp: cannot copy '%s' to '%s': %s", sourcePath, path, reason)
191 if (invalidSource(sourcePath)) { problem(destination, "Invalid argument"); return }
192 var result = resolve(sourcePath)
193 val source = result.node
194 if (source == null) { out.error("cp: cannot stat '%s': %s", sourcePath, result.reason); return }
195 if (source is Directory && !recursive) { out.error("cp: -r not specified; omitting directory '%s'", sourcePath); return }
196 result = resolve(destination)
197 if (result.node == null && !result.absent) { problem(destination, result.reason); return }
198 val target = if (result.node is Directory) joinPath(destination, lastComponent(sourcePath)) else destination
199 result = resolve(target, source !is Directory)
200 val targetNode = result.node
201 when {
202 source is Directory -> {
203 if (targetNode != null) problem(target, "File exists")
204 else if (descendant(result.creationParent!!, source)) out.error("cp: cannot copy a directory, '%s', into itself, '%s'", sourcePath, target)
205 else cloneNode(source, result.creationParent!!, result.creationName)
206 }
207 result.absent -> makeFile(result.creationParent!!, result.creationName, (source as FileNode).lines)
208 targetNode === source -> out.error("cp: '%s' and '%s' are the same file", sourcePath, target)
209 targetNode is Directory -> out.error("cp: cannot overwrite directory '%s' with non-directory '%s'", target, sourcePath)
210 targetNode is FileNode -> write(targetNode, (source as FileNode).lines)
211 else -> problem(target, result.reason)
212 }
213 }

215 fun listing(options: Options, out: Output) {
216 val paths = options.paths.ifEmpty { listOf(".") }
217 for (path in paths) {
218 val result = resolve(path, 'l' !in options)
219 val node = result.node
220 if (node == null) { out.error("ls: cannot access '%s': %s", path, result.reason); continue }
221 val entries = mutableListOf<Listed>()
222 if (node is Directory) {
223 if (paths.size > 1) out.emit("$path:")
224 if ('a' in options) { entries.add(Listed(".", node)); entries.add(Listed("..", node.parent)) }
225 for (child in node.entries.values) if ('a' in options || !child.name.startsWith('.')) entries.add(Listed(child.name, child))
226 } else entries.add(Listed(path, node))
227 if ('t' in options) entries.sortByDescending { it.node.modified }
228 if ('r' in options) entries.reverse()
229 val widths = intArrayOf(1, 1, 1)
230 for (entry in entries) {
231 val numbers = longArrayOf(entry.node.links.toLong(), entry.node.size, entry.node.modified.toLong())
232 for (i in 0..2) widths[i] = maxOf(widths[i], numbers[i].toString().length)
233 }
234 for (entry in entries) {
235 if ('l' !in options) { out.emit(entry.name); continue }
236 val item = entry.node
237 val numbers = longArrayOf(item.links.toLong(), item.size, item.modified.toLong())
238 val fields = numbers.mapIndexed { i, value -> value.toString().padStart(widths[i]) }.joinToString(" ")
239 val suffix = if (item is Link) " -> ${item.target}" else ""
240 out.emit("${if (item is FileNode) '-' else item.tag} $fields ${entry.name}$suffix")
241 }
242 }
243 }
244 fun walk(node: Node, path: String, depth: Int, limit: Int, visit: (Node, String, Int) -> Unit) {
245 visit(node, path, depth)
246 if (node is Directory && depth < limit) {
247 for (child in node.entries.values) walk(child, joinPath(path, child.name), depth + 1, limit, visit)
248 }
249 }
250 fun find(args: List<String>, out: Output) {
251 val explicit = args.isNotEmpty() && !args.first().startsWith('-')
252 val path = if (explicit) args.first() else "."
253 val tests = mutableMapOf<String, String>()
254 var at = if (explicit) 1 else 0
255 while (at < args.size) { tests[args[at]] = args[at + 1]; at += 2 }
256 val result = resolve(path)
257 val node = result.node
258 if (node == null) { out.error("find: '%s': %s", path, result.reason); return }
259 walk(node, path, 0, tests["-maxdepth"]?.toInt() ?: 1000) { item, display, depth ->
260 val name = if (depth == 0) { if (path == "/") "/" else lastComponent(path) } else item.name
261 if ((tests["-name"]?.let { globMatch(it, name) } != false) &&
262 (tests["-type"]?.first()?.let { it == item.tag } != false)) out.emit(display)
263 }
264 }
265 fun usage(node: Node, path: String, show: Boolean, options: Options, out: Output): Long {
266 var total = if (node is Directory) 0L else node.size
267 if (node is Directory) for (child in node.entries.values) {
268 total += usage(child, joinPath(path, child.name),
269 show && 's' !in options && (child is Directory || 'a' in options), options, out)
270 }
271 if (show) out.print("%d %s", total, path)
272 return total
273 }

282 fun drawTree(directory: Directory, prefix: String, all: Boolean, counts: IntArray, out: Output) {
283 val children = directory.entries.values.filter { all || !it.name.startsWith('.') }
284 for ((i, child) in children.withIndex()) {
285 val last = i + 1 == children.size
286 out.emit(prefix + (if (last) "`-- " else "|-- ") + child.name + (if (child is Link) " -> ${child.target}" else ""))
287 if (child is Directory) {
288 counts[0]++
289 drawTree(child, prefix + (if (last) " " else "| "), all, counts, out)
290 } else counts[1]++
291 }

293 fun tree(options: Options, out: Output) {
294 val path = options.paths.firstOrNull() ?: "."
295 val result = resolve(path)
296 val node = result.node
297 if (node == null) { out.error("tree: %s: %s", path, result.reason); return }
298 if (node !is Directory) { out.error("tree: %s: Not a directory", path); return }
299 out.emit(path)
300 val counts = intArrayOf(0, 0)
301 drawTree(node, "", 'a' in options, counts, out)
302 out.print("%d %s, %d %s", counts[0], if (counts[0] == 1) "directory" else "directories",
303 counts[1], if (counts[1] == 1) "file" else "files")
304 }
305 fun readInputs(command: String, paths: List<String>, input: List<String>, out: Output, consume: (Input) -> Unit) {
306 if (paths.isEmpty()) { consume(Input(input)); return }
307 for (path in paths) {
308 val result = resolve(path)
309 when (val node = result.node) {
310 null -> out.error("%s: %s: %s", command, path, result.reason)
311 is FileNode -> consume(Input(node.lines, path))
312 else -> out.error("%s: %s: Is a directory", command, path)
313 }
314 }

316 fun textCommand(command: String, args: List<String>, input: List<String>, out: Output) {
317 val options = options(args)
318 when (command) {
319 "grep" -> {
320 val pattern = options.paths.first().let { if ('i' in options) it.lowercase(Locale.ROOT) else it }
321 val paths = options.paths.drop(1)
322 val named = 'r' in options || paths.size > 1
323 val consume: (Input) -> Unit = { part ->
324 val prefix = if (named) "${part.name!!}:" else ""
325 var selected = 0L
326 for ((i, line) in part.lines.withIndex()) {
327 val comparison = if ('i' in options) line.lowercase(Locale.ROOT) else line
328 if (comparison.contains(pattern) != ('v' in options)) {
329 selected++
330 if ('c' !in options) out.emit(prefix + (if ('n' in options) "${i + 1}:" else "") + line)
331 }
332 }
333 if ('c' in options) out.emit("$prefix$selected")
334 }
335 if ('r' !in options) readInputs(command, paths, input, out, consume)
336 else for (path in paths.ifEmpty { listOf(".") }) {
337 val result = resolve(path)
338 val node = result.node
339 if (node == null) out.error("grep: %s: %s", path, result.reason)
340 else walk(node, path, 0, 1000) { item, display, _ ->
341 if (item is FileNode) consume(Input(item.lines, display))
342 }
343 }
344 }
345 "sort" -> {
346 val lines = mutableListOf<String>()
347 readInputs(command, options.paths, input, out) { lines.addAll(it.lines) }
348 lines.sortWith { a, b ->
349 val comparison = if ('n' in options) numericPrefix(a).compareTo(numericPrefix(b)) else 0
350 if (comparison != 0) comparison else a.compareTo(b)
351 }
352 if ('r' in options) lines.reverse()
353 lines.forEach(out::emit)
354 }
355 "wc" -> {
356 val all = "lwc".none { it in options }
357 val report: (LongArray, String?) -> Unit = { counts, name ->
358 val values = (0..2).filter { all || "lwc"[it] in options }.joinToString(" ") { counts[it].toString() }
359 out.emit(values + (if (name != null) " $name" else ""))
360 }
361 val totals = LongArray(3)
362 readInputs(command, options.paths, input, out) { part ->
363 val counts = longArrayOf(part.lines.size.toLong(), 0, 0)
364 for (line in part.lines) {
365 counts[2] += line.length.toLong() + 1
366 var inside = false
367 for (character in line) {
368 if (character != ' ' && !inside) counts[1]++
369 inside = character != ' '
370 }
371 }
372 for (i in 0..2) totals[i] += counts[i]
373 report(counts, part.name)
374 }
375 if (options.paths.size > 1) report(totals, "total")
376 }
377 "head", "tail" -> {
378 val explicit = args.firstOrNull() == "-n"
379 val raw = if (explicit) args[1] else "10"
380 val special = raw.startsWith('-') || raw.startsWith('+')
381 val count = (if (special) raw.drop(1) else raw).toInt()
382 readInputs(command, args.drop(if (explicit) 2 else 0), input, out) { part ->
383 var first = 0
384 var end = part.lines.size
385 if (command == "head") end = if (special) maxOf(0, end - count) else minOf(end, count)
386 else first = if (special) minOf(end, maxOf(0, count - 1)) else maxOf(0, end - count)
387 for (i in first until end) out.emit(part.lines[i])
388 }
389 }
390 else -> readInputs(command, options.paths, input, out) { part ->
391 if (command == "cat") part.lines.forEach(out::emit)
392 else {
393 var first = 0
394 while (first < part.lines.size) {
395 var end = first + 1
396 while (end < part.lines.size && part.lines[first] == part.lines[end]) end++
397 out.emit((if ('c' in options) "${end - first} " else "") + part.lines[first])
398 first = end
399 }
400 }
401 }
402 }

404 fun execute(text: String, input: List<String>, out: Output) {
405 val words = text.split(' ')
406 val command = words.first()
407 val args = words.drop(1)
408 val options = options(args)
409 when (command) {
410 "pwd" -> out.emit(absolute(current))
411 "echo" -> out.emit(text.substring(5))
412 "cd" -> {
413 var path = args.firstOrNull() ?: "/"
414 val back = path == "-"
415 if (back) {
416 val old = previous
417 if (old == null) { out.error("cd: OLDPWD not set"); return }
418 path = old
419 }
420 val result = resolve(path)
421 when (val node = result.node) {
422 null -> out.error("cd: %s: %s", path, result.reason)
423 is Directory -> {
424 previous = absolute(current)
425 current = node
426 if (back) out.emit(absolute(current))
427 }
428 else -> out.error("cd: %s: Not a directory", path)
429 }
430 }
431 "mkdir" -> for (path in options.paths) {
432 val result = resolve(path, false, 'p' in options)
433 when {
434 result.absent -> makeDirectory(result.creationParent!!, result.creationName)
435 result.node == null -> out.error("mkdir: cannot create directory '%s': %s", path, result.reason)
436 'p' !in options || resolve(path).node !is Directory -> out.error("mkdir: cannot create directory '%s': File exists", path)
437 }
438 }
439 "touch" -> for (path in args) {
440 val result = resolve(path)
441 when {
442 result.absent -> makeFile(result.creationParent!!, result.creationName)
443 result.node != null -> result.node.modified = clock
444 else -> out.error("touch: cannot touch '%s': %s", path, result.reason)
445 }
446 }
447 "ls" -> listing(options, out)
448 "stat" -> for (path in args) {
449 val result = resolve(path, false)
450 val node = result.node
451 if (node == null) { out.error("stat: cannot stat '%s': %s", path, result.reason); continue }
452 val kind = when (node) {
453 is Directory -> "directory"
454 is Link -> "symbolic link"
455 is FileNode -> if (node.lines.isEmpty()) "regular empty file" else "regular file"
456 }
457 out.print("%s: %s, size %d, inode %d, links %d, modified %d", path, kind, node.size, node.inode, node.links, node.modified)
458 }
459 "rm", "rmdir" -> {
460 val directoryOnly = command == "rmdir"
461 for (path in options.paths) {
462 val dot = lastComponent(path) in listOf(".", "..")
463 if (dot && !directoryOnly) {
464 out.error("rm: refusing to remove '.' or '..' directory: skipping '%s'", path)
465 continue
466 }
467 val problem = if (dot) "Invalid argument" else {
468 val result = resolve(path, false)
469 val node = result.node
470 when {
471 node == null -> {
472 if (!directoryOnly && 'f' in options && result.failure == Failure.MISSING) continue
473 result.reason
474 }
475 directoryOnly && node !is Directory -> "Not a directory"
476 !directoryOnly && node is Directory && 'r' !in options -> "Is a directory"
477 descendant(current, node) -> "Device or resource busy"
478 directoryOnly && (node as Directory).entries.isNotEmpty() -> "Directory not empty"
479 else -> { detach(node); "" }
480 }
481 }
482 if (problem.isNotEmpty()) out.error(if (directoryOnly) "rmdir: failed to remove '%s': %s" else "rm: cannot remove '%s': %s", path, problem)
483 }
484 }
485 "mv", "cp" -> {
486 val destination = options.paths.last()
487 if (options.paths.size > 2 && resolve(destination).node !is Directory) {
488 out.error("%s: target '%s' is not a directory", command, destination)
489 return
490 }
491 for (source in options.paths.dropLast(1)) {
492 if (command == "mv") move(source, destination, out)
493 else copy(source, destination, 'r' in options, out)
494 }
495 }
496 "ln" -> {
497 val path = args[2]
498 val result = resolve(path, false)
499 if (result.absent) add(result.creationParent!!, result.creationName, Link(nextInode++, clock, args[1]))
500 else out.error("ln: failed to create symbolic link '%s': %s", path, if (result.node == null) result.reason else "File exists")
501 }
502 "readlink", "realpath" -> {
503 val path = args.first()
504 val result = resolve(path, command == "realpath")
505 val node = result.node
506 when {
507 node == null -> out.error("%s: %s: %s", command, path, result.reason)
508 command == "realpath" -> out.emit(absolute(node))
509 node is Link -> out.emit(node.target)
510 else -> out.error("readlink: %s: Invalid argument", path)
511 }
512 }
513 "find" -> find(args, out)
514 "du" -> diskUsage(options, out)
515 "tree" -> tree(options, out)
516 "cat", "wc", "head", "tail", "grep", "sort", "uniq" -> textCommand(command, args, input, out)
517 else -> out.error("%s: command not found", command)
518 }
519 }
520 fun process(line: String, terminal: PrintWriter) {
521 clock++
522 terminal.print("${absolute(current)}\$ $line\n")
523 var pipeline = line
524 var redirect: FileNode? = null
525 val marker = line.indexOf(" >")
526 if (marker >= 0) {
527 pipeline = line.substring(0, marker)
528 val append = line.startsWith(" >> ", marker)
529 val path = line.substring(marker + (if (append) 4 else 3))
530 val result = resolve(path)
531 val node = result.node
532 if (result.absent) redirect = makeFile(result.creationParent!!, result.creationName)
533 else if (node is FileNode) {
534 redirect = node
535 if (!append) write(node, emptyList())
536 } else {
537 terminal.print("sh: $path: ${if (node == null) result.reason else "Is a directory"}\n")
538 return
539 }
540 }
541 val commands = pipeline.split(" | ")
542 var input: List<String> = emptyList()
543 for ((i, command) in commands.withIndex()) {
544 val out = Output(terminal, i + 1 == commands.size && redirect == null)
545 execute(command, input, out)
546 input = out.lines
547 }
548 if (redirect != null && input.isNotEmpty()) write(redirect, input, true)
549 }
550}

552private fun globMatch(pattern: String, text: String): Boolean {
553 var previous = BooleanArray(text.length + 1)
554 previous[0] = true
555 for (character in pattern) {
556 val next = BooleanArray(previous.size)
557 next[0] = character == '*' && previous[0]
558 for (i in text.indices) next[i + 1] = if (character == '*') previous[i + 1] || next[i]
559 else previous[i] && (character == '?' || character == text[i])
560 previous = next
561 }
562 return previous[text.length]
563}
564private fun numericPrefix(text: String): Long {
565 val negative = text.startsWith('-')
566 var at = if (negative) 1 else 0
567 var value = 0L
