8type NodeRef = Rc<RefCell<Node>>;
9type Lines = Vec<Rc<str>>;
10
11#[derive(Clone)]
12enum Data {
13 Directory(BTreeMap<String, NodeRef>),
14 File(Lines),
15 Link(String),
16}
17struct Node {
18 inode: usize,
19 modified: usize,
20 parent: Weak<RefCell<Node>>,
21 name: String,
22 data: Data,
23}
24impl Node {
25 fn tag(&self) -> char {
26 match &self.data { Data::Directory(_) => 'd', Data::File(_) => 'f', Data::Link(_) => 'l' }
27 }
28 fn size(&self) -> u64 {
29 match &self.data {
30 Data::Directory(entries) => entries.len() as u64,
31 Data::File(lines) => lines.iter().map(|line| line.len() as u64 + 1).sum(),
32 Data::Link(target) => target.len() as u64,
33 }
34 }
35 fn links(&self) -> u64 {
36 match &self.data {
37 Data::Directory(entries) => 2 + entries.values().filter(|node| is_directory(node)).count() as u64,
38 _ => 1,
39 }
40 }
41}
42fn is_directory(node: &NodeRef) -> bool { matches!(&node.borrow().data, Data::Directory(_)) }
43fn parent(node: &NodeRef) -> NodeRef { node.borrow().parent.upgrade().unwrap() }
44fn children(node: &NodeRef) -> Vec<(String, NodeRef)> {
45 match &node.borrow().data {
46 Data::Directory(entries) => entries.iter().map(|(name, child)| (name.clone(), child.clone())).collect(),
47 _ => Vec::new(),
48 }
49}
50fn child(node: &NodeRef, name: &str) -> Option<NodeRef> {
51 match &node.borrow().data { Data::Directory(entries) => entries.get(name).cloned(), _ => None }
52}
53fn last(path: &str) -> &str { path.rsplit('/').next().unwrap() }
54fn join(path: &str, name: &str) -> String {
55 if path == "/" { format!("/{name}") } else { format!("{path}/{name}") }
56}
57fn invalid_source(path: &str) -> bool { matches!(last(path), "" | "." | "..") }

59#[derive(Clone, Copy)]
60enum Failure { Missing, NotDirectory, TooManyLinks }
61struct ResolutionError {
62 failure: Failure,
63 creation: Option<(NodeRef, String)>,
64}
65impl ResolutionError {
66 fn new(failure: Failure) -> Self { Self { failure, creation: None } }
67 fn reason(&self) -> &'static str {
68 match self.failure {
69 Failure::Missing => "No such file or directory",
70 Failure::NotDirectory => "Not a directory",
71 Failure::TooManyLinks => "Too many levels of symbolic links",
72 }
73 }
74}
75struct Output<'a> {
76 terminal: &'a mut dyn Write,
77 direct: bool,
78 lines: Lines,
79}
80impl Output<'_> {
81 fn emit(&mut self, line: impl Into<Rc<str>>) {
82 let line = line.into();
83 if self.direct { writeln!(self.terminal, "{line}").unwrap(); }
84 else { self.lines.push(line); }
85 }
86 fn error(&mut self, message: impl Display) { writeln!(self.terminal, "{message}").unwrap(); }
87}
88struct Options<'a> {
89 flags: [bool; 128],
90 paths: &'a [&'a str],
91}
92impl Options<'_> {
93 fn has(&self, option: u8) -> bool { self.flags[option as usize] }
94}
95fn options<'a>(arguments: &'a [&'a str]) -> Options<'a> {
96 let mut flags = [false; 128];
97 let mut position = 0;
98 while position < arguments.len() && arguments[position].starts_with('-') {
99 for flag in arguments[position][1..].bytes() { flags[flag as usize] = true; }
100 position += 1;
101 }
102 Options { flags, paths: &arguments[position..] }
103}

105struct Shell {
106 root: NodeRef,
107 current: NodeRef,
108 previous: Option<String>,
109 clock: usize,
110 next_inode: usize,
111}
112impl Shell {
113 fn new() -> Self {
114 let root = Rc::new_cyclic(|parent| RefCell::new(Node {
115 inode: 1, modified: 0, parent: parent.clone(), name: String::new(), data: Data::Directory(BTreeMap::new()),
116 }));
117 Self { current: root.clone(), root, previous: None, clock: 0, next_inode: 2 }
118 }
119 fn absolute(&self, node: &NodeRef) -> String {
120 let mut node = node.clone();
121 let mut names = Vec::new();
122 while !Rc::ptr_eq(&node, &self.root) {
123 names.push(node.borrow().name.clone());
124 node = parent(&node);
125 }
126 names.reverse();
127 format!("/{}", names.join("/"))
128 }
129 fn descendant(&self, node: &NodeRef, ancestor: &NodeRef) -> bool {
130 let mut node = node.clone();
131 loop {
132 if Rc::ptr_eq(&node, ancestor) { return true; }
133 if Rc::ptr_eq(&node, &self.root) { return false; }
134 node = parent(&node);
135 }
136 }
137 fn add(&self, directory: &NodeRef, name: &str, node: NodeRef) {
138 {
139 let mut item = node.borrow_mut();
140 item.name = name.to_owned();
141 item.parent = Rc::downgrade(directory);
142 }
143 let mut directory = directory.borrow_mut();
144 if let Data::Directory(entries) = &mut directory.data { entries.insert(name.to_owned(), node); }
145 directory.modified = self.clock;
146 }
147 fn create(&mut self, directory: &NodeRef, name: &str, data: Data) -> NodeRef {
148 let node = Rc::new(RefCell::new(Node {
149 inode: self.next_inode, modified: self.clock, parent: Weak::new(), name: String::new(), data,
150 }));
151 self.next_inode += 1;
152 self.add(directory, name, node.clone());
153 node
154 }
155 fn detach(&self, node: &NodeRef) {
156 let directory = parent(node);
157 let name = node.borrow().name.clone();
158 let mut directory = directory.borrow_mut();
159 if let Data::Directory(entries) = &mut directory.data { entries.remove(&name); }
160 directory.modified = self.clock;
161 }
162 fn write(&self, file: &NodeRef, lines: Lines, append: bool) {
163 let mut file = file.borrow_mut();
164 if let Data::File(content) = &mut file.data {
165 if append { content.extend(lines); } else { *content = lines; }
166 }
167 file.modified = self.clock;
168 }
169 fn walk(&mut self, path: &str, base: &NodeRef, follow: bool, original: bool,
170 parents: bool, expansions: &mut usize) -> Result<NodeRef, ResolutionError> {
171 let mut node = if path.starts_with('/') { self.root.clone() } else { base.clone() };
172 let mut parts = path.split('/').filter(|part| !part.is_empty()).peekable();
173 while let Some(name) = parts.next() {
174 if !is_directory(&node) { return Err(ResolutionError::new(Failure::NotDirectory)); }
175 if name == "." { continue; }
176 if name == ".." { node = parent(&node); continue; }
177 let last = parts.peek().is_none();
178 let mut next = match child(&node, name) {
179 Some(next) => next,
180 None if original && parents && !last => self.create(&node, name, Data::Directory(BTreeMap::new())),
181 None => return Err(ResolutionError {
182 failure: Failure::Missing,
183 creation: (original && last).then(|| (node.clone(), name.to_owned())),
184 }),
185 };
186 let target = match &next.borrow().data { Data::Link(target) => Some(target.clone()), _ => None };
187 if let Some(target) = target.filter(|_| !last || follow) {
188 *expansions += 1;
189 if *expansions > 8 { return Err(ResolutionError::new(Failure::TooManyLinks)); }
190 next = self.walk(&target, &node, true, false, false, expansions)?;
191 }
192 node = next;
193 }
194 Ok(node)
195 }
196 fn resolve(&mut self, path: &str, follow: bool, parents: bool) -> Result<NodeRef, ResolutionError> {
197 let current = self.current.clone();
198 self.walk(path, &current, follow, true, parents, &mut 0)
199 }
200 fn clone_node(&mut self, source: &NodeRef, directory: &NodeRef, name: &str) {
201 let data = source.borrow().data.clone();
202 match data {
203 Data::Directory(entries) => {
204 let copy = self.create(directory, name, Data::Directory(BTreeMap::new()));
205 for (name, node) in entries { self.clone_node(&node, &copy, &name); }
206 }
207 data => { self.create(directory, name, data); }
208 }
209 }
210 fn move_one(&mut self, source_path: &str, destination: &str, output: &mut Output<'_>) {
211 if invalid_source(source_path) {
212 output.error(format!("mv: cannot move '{source_path}' to '{destination}': Invalid argument"));
213 return;
214 }
215 let source = match self.resolve(source_path, false, false) {
216 Ok(source) => source,
217 Err(error) => { output.error(format!("mv: cannot stat '{source_path}': {}", error.reason())); return; }
218 };
219 let destination_node = self.resolve(destination, true, false).ok().filter(is_directory);
220 let (directory, name, target) = if let Some(directory) = destination_node {
221 (directory, last(source_path).to_owned(), join(destination, last(source_path)))
222 } else {
223 match self.resolve(destination, false, false) {
224 Ok(node) => {
225 let name = node.borrow().name.clone();
226 (parent(&node), name, destination.to_owned())
227 }
228 Err(error) => {
229 if let Some((directory, name)) = error.creation.as_ref() {
230 (directory.clone(), name.clone(), destination.to_owned())
231 } else {
232 output.error(format!("mv: cannot move '{source_path}' to '{destination}': {}", error.reason()));
233 return;
234 }
235 }
236 }
237 };
238 let existing = child(&directory, &name);
239 let existing_directory = existing.as_ref().is_some_and(is_directory);
240 if is_directory(&source) && self.descendant(&directory, &source) {
241 output.error(format!("mv: cannot move '{source_path}' to a subdirectory of itself, '{target}'"));
242 } else if existing.as_ref().is_some_and(|node| Rc::ptr_eq(node, &source)) {
243 output.error(format!("mv: '{source_path}' and '{target}' are the same file"));
244 } else if existing.is_some() && is_directory(&source) && !existing_directory {
245 output.error(format!("mv: cannot overwrite non-directory '{target}' with directory '{source_path}'"));
246 } else if !is_directory(&source) && existing_directory {
247 output.error(format!("mv: cannot overwrite directory '{target}' with non-directory '{source_path}'"));
248 } else if existing_directory && !children(existing.as_ref().unwrap()).is_empty() {
249 output.error(format!("mv: cannot move '{source_path}' to '{target}': Directory not empty"));
250 } else if existing.as_ref().is_some_and(|node| Rc::ptr_eq(node, &self.current)) {
251 output.error(format!("mv: cannot move '{source_path}' to '{target}': Device or resource busy"));
252 } else {
253 if let Some(node) = existing { self.detach(&node); }
254 self.detach(&source);
255 self.add(&directory, &name, source);
256 }
257 }
258 fn copy_one(&mut self, source_path: &str, destination: &str, recursive: bool, output: &mut Output<'_>) {
259 if invalid_source(source_path) {
260 output.error(format!("cp: cannot copy '{source_path}' to '{destination}': Invalid argument"));
261 return;
262 }
263 let source = match self.resolve(source_path, true, false) {
264 Ok(source) => source,
265 Err(error) => { output.error(format!("cp: cannot stat '{source_path}': {}", error.reason())); return; }
266 };
267 if is_directory(&source) && !recursive {
268 output.error(format!("cp: -r not specified; omitting directory '{source_path}'"));
269 return;
270 }
271 let target = match self.resolve(destination, true, false) {
272 Ok(node) if is_directory(&node) => join(destination, last(source_path)),
273 Ok(_) => destination.to_owned(),
274 Err(error) if error.creation.is_some() => destination.to_owned(),
275 Err(error) => { output.error(format!("cp: cannot copy '{source_path}' to '{destination}': {}", error.reason())); return; }
276 };
277 match self.resolve(&target, !is_directory(&source), false) {
278 Ok(_) if is_directory(&source) => output.error(format!("cp: cannot copy '{source_path}' to '{target}': File exists")),
279 Ok(node) if Rc::ptr_eq(&node, &source) => output.error(format!("cp: '{source_path}' and '{target}' are the same file")),
280 Ok(node) if is_directory(&node) => output.error(format!("cp: cannot overwrite directory '{target}' with non-directory '{source_path}'")),
281 Ok(node) => {
282 let Data::File(lines) = &source.borrow().data else { unreachable!() };
283 self.write(&node, lines.clone(), false);
284 }
285 Err(error) => {
286 if let Some((directory, name)) = error.creation.as_ref() {
287 if is_directory(&source) && self.descendant(directory, &source) {
288 output.error(format!("cp: cannot copy a directory, '{source_path}', into itself, '{target}'"));
289 } else { self.clone_node(&source, directory, name); }
290 } else { output.error(format!("cp: cannot copy '{source_path}' to '{target}': {}", error.reason())); }
291 }
292 }
293 }
294 fn listing(&mut self, opts: &Options<'_>, output: &mut Output<'_>) {
295 let paths = if opts.paths.is_empty() { &["."][..] } else { opts.paths };
296 for &path in paths {
297 let node = match self.resolve(path, !opts.has(b'l'), false) {
298 Ok(node) => node,
299 Err(error) => { output.error(format!("ls: cannot access '{path}': {}", error.reason())); continue; }
300 };
301 let mut entries = Vec::new();
302 if is_directory(&node) {
303 if paths.len() > 1 { output.emit(format!("{path}:")); }
304 if opts.has(b'a') { entries.push((".".to_owned(), node.clone())); entries.push(("..".to_owned(), parent(&node))); }
305 entries.extend(children(&node).into_iter().filter(|(name, _)| opts.has(b'a') || !name.starts_with('.')));
306 if opts.has(b't') { entries.sort_by_key(|(_, node)| Reverse(node.borrow().modified)); }
307 if opts.has(b'r') { entries.reverse(); }
308 } else { entries.push((path.to_owned(), node)); }
309 let mut widths = [1_usize; 3];
310 for (_, node) in &entries {
311 let node = node.borrow();
312 for (i, value) in [node.links(), node.size(), node.modified as u64].iter().enumerate() {
313 widths[i] = widths[i].max(value.to_string().len());
314 }
315 }
316 for (name, node) in entries {
317 if !opts.has(b'l') { output.emit(name); continue; }
318 let node = node.borrow();
319 let kind = if node.tag() == 'f' { '-' } else { node.tag() };
320 let suffix = match &node.data { Data::Link(target) => format!(" -> {target}"), _ => String::new() };
321 output.emit(format!("{kind} {:>lw$} {:>sw$} {:>tw$} {name}{suffix}", node.links(), node.size(), node.modified,
322 lw = widths[0], sw = widths[1], tw = widths[2]));
323 }
324 }

326 fn find(&mut self, arguments: &[&str], output: &mut Output<'_>) {
327 let mut position = usize::from(arguments.first().is_some_and(|path| !path.starts_with('-')));
328 let path = if position == 1 { arguments[0] } else { "." };
329 let mut pattern = None;
330 let mut kind = None;
331 let mut limit = usize::MAX;
332 while position < arguments.len() {
333 match arguments[position] {
334 "-name" => pattern = Some(arguments[position + 1]),
335 "-type" => kind = arguments[position + 1].chars().next(),
336 _ => limit = arguments[position + 1].parse().unwrap(),
337 }
338 position += 2;
339 }
340 match self.resolve(path, true, false) {
341 Err(error) => output.error(format!("find: '{path}': {}", error.reason())),
342 Ok(node) => walk_nodes(&node, path, 0, limit, &mut |node, display, depth| {
343 let node = node.borrow();
344 let name = if depth == 0 { if path == "/" { "/" } else { last(path) } } else { &node.name };
345 if pattern.is_none_or(|pattern| glob_matches(pattern, name)) && kind.is_none_or(|kind| kind == node.tag()) {
346 output.emit(display);
347 }
348 }),
349 }

394 fn read_inputs(&mut self, command: &str, paths: &[&str], input: &[Rc<str>], output: &mut Output<'_>,
395 mut consume: impl FnMut(&[Rc<str>], Option<&str>, &mut Output<'_>)) {
396 if paths.is_empty() { consume(input, None, output); return; }
397 for &path in paths {
398 match self.resolve(path, true, false) {
399 Err(error) => output.error(format!("{command}: {path}: {}", error.reason())),
400 Ok(node) => {
401 let node = node.borrow();
402 match &node.data {
403 Data::File(lines) => consume(lines, Some(path), output),
404 Data::Directory(_) => output.error(format!("{command}: {path}: Is a directory")),
405 Data::Link(_) => unreachable!(),
406 }
407 }
408 }
409 }

411 fn text_command(&mut self, command: &str, arguments: &[&str], input: &[Rc<str>], output: &mut Output<'_>) {
412 let opts = options(arguments);
413 if command == "grep" {
414 let pattern = opts.paths[0];
415 let paths = &opts.paths[1..];
416 let named = paths.len() > 1 || opts.has(b'r');
417 let consume = |lines: &[Rc<str>], name: Option<&str>, output: &mut Output<'_>| {
418 let prefix = if named { format!("{}:", name.unwrap()) } else { String::new() };
419 let mut selected = 0_u64;
420 for (i, line) in lines.iter().enumerate() {
421 let matches = if opts.has(b'i') {
422 line.as_bytes().windows(pattern.len()).any(|part| part.eq_ignore_ascii_case(pattern.as_bytes()))
423 } else { line.contains(pattern) };
424 if matches == opts.has(b'v') { continue; }
425 selected += 1;
426 if !opts.has(b'c') {
427 if opts.has(b'n') { output.emit(format!("{prefix}{}:{line}", i + 1)); }
428 else if named { output.emit(format!("{prefix}{line}")); }
429 else { output.emit(line.clone()); }
430 }
431 }
432 if opts.has(b'c') { output.emit(format!("{prefix}{selected}")); }
433 };
434 if !opts.has(b'r') { self.read_inputs(command, paths, input, output, consume); }
435 else {
436 let paths = if paths.is_empty() { &["."][..] } else { paths };
437 for &path in paths {
438 match self.resolve(path, true, false) {
439 Err(error) => output.error(format!("grep: {path}: {}", error.reason())),
440 Ok(node) => walk_nodes(&node, path, 0, usize::MAX, &mut |node, display, _| {
441 if let Data::File(lines) = &node.borrow().data { consume(lines, Some(display), output); }
442 }),
443 }
444 }
445 }
446 } else if command == "sort" {
447 let mut lines = Vec::new();
448 self.read_inputs(command, opts.paths, input, output, |part, _, _| lines.extend_from_slice(part));
449 lines.sort_by(|a, b| {
450 if opts.has(b'n') { numeric_prefix(a).cmp(&numeric_prefix(b)).then_with(|| a.cmp(b)) }
451 else { a.cmp(b) }
452 });
453 if opts.has(b'r') { lines.reverse(); }
454 for line in lines { output.emit(line); }
455 } else if command == "wc" {
456 let mut totals = [0_u64; 3];
457 let report = |counts: &[u64; 3], name: Option<&str>, output: &mut Output<'_>| {
458 let all = !opts.has(b'l') && !opts.has(b'w') && !opts.has(b'c');
459 let values: Vec<_> = counts.iter().zip(b"lwc").filter(|(_, flag)| all || opts.has(**flag)).map(|(n, _)| n.to_string()).collect();
460 output.emit(format!("{}{}", values.join(" "), name.map_or(String::new(), |name| format!(" {name}"))));
461 };
462 self.read_inputs(command, opts.paths, input, output, |lines, name, output| {
463 let mut counts = [lines.len() as u64, 0, 0];
464 for line in lines {
465 counts[2] += line.len() as u64 + 1;
466 let mut inside = false;
467 for character in line.bytes() {
468 if character != b' ' && !inside { counts[1] += 1; }
469 inside = character != b' ';
470 }
471 }
472 for i in 0..3 { totals[i] += counts[i]; }
473 report(&counts, name, output);
474 });
475 if opts.paths.len() > 1 { report(&totals, Some("total"), output); }
476 } else if command == "head" || command == "tail" {
477 let offset = if arguments.first() == Some(&"-n") { 2 } else { 0 };
478 let count_text = if offset == 0 { "10" } else { arguments[1] };
479 let special = count_text.starts_with('-') || count_text.starts_with('+');
480 let count: usize = count_text[usize::from(special)..].parse().unwrap();
481 self.read_inputs(command, &arguments[offset..], input, output, |lines, _, output| {
482 let mut first = 0;
483 let mut end = lines.len();
484 if command == "head" { end = if special { end.saturating_sub(count) } else { end.min(count) }; }
485 else { first = if special { count.saturating_sub(1).min(end) } else { end.saturating_sub(count) }; }
486 for line in &lines[first..end] { output.emit(line.clone()); }
487 });
488 } else {
489 self.read_inputs(command, opts.paths, input, output, |lines, _, output| {
490 if command == "cat" { for line in lines { output.emit(line.clone()); } }
491 else {
492 let mut first = 0;
493 while first < lines.len() {
494 let mut end = first + 1;
495 while end < lines.len() && lines[first] == lines[end] { end += 1; }
496 if opts.has(b'c') { output.emit(format!("{} {}", end - first, lines[first])); }
497 else { output.emit(lines[first].clone()); }
498 first = end;
499 }
500 }
501 });
502 }
503 }
504 fn execute(&mut self, text: &str, input: &[Rc<str>], output: &mut Output<'_>) {
505 let words: Vec<_> = text.split_whitespace().collect();
506 let command = words[0];
507 let arguments = &words[1..];
508 let opts = options(arguments);
509 match command {
510 "pwd" => output.emit(self.absolute(&self.current)),
511 "echo" => output.emit(&text[5..]),
512 "cd" => {
513 let back = arguments.first() == Some(&"-");
514 let path = if back {
515 let Some(previous) = &self.previous else { output.error("cd: OLDPWD not set"); return; };
516 previous.clone()
517 } else { arguments.first().copied().unwrap_or("/").to_owned() };
518 match self.resolve(&path, true, false) {
519 Err(error) => output.error(format!("cd: {path}: {}", error.reason())),
520 Ok(node) if !is_directory(&node) => output.error(format!("cd: {path}: Not a directory")),
521 Ok(node) => {
522 self.previous = Some(self.absolute(&self.current));
523 self.current = node;
524 if back { output.emit(self.absolute(&self.current)); }
525 }
526 }
527 }
528 "mkdir" => for &path in opts.paths {
529 match self.resolve(path, false, opts.has(b'p')) {
530 Err(error) => {
531 if let Some((directory, name)) = error.creation.as_ref() {
532 self.create(directory, name, Data::Directory(BTreeMap::new()));
533 } else { output.error(format!("mkdir: cannot create directory '{path}': {}", error.reason())); }
534 }
535 Ok(_) => {
536 if !opts.has(b'p') || !self.resolve(path, true, false).is_ok_and(|node| is_directory(&node)) {
537 output.error(format!("mkdir: cannot create directory '{path}': File exists"));
538 }
539 }
540 }
541 },
542 "touch" => for &path in arguments {
543 match self.resolve(path, true, false) {
544 Ok(node) => node.borrow_mut().modified = self.clock,
545 Err(error) => {
546 if let Some((directory, name)) = error.creation.as_ref() { self.create(directory, name, Data::File(Vec::new())); }
547 else { output.error(format!("touch: cannot touch '{path}': {}", error.reason())); }
548 }
549 }
550 },
551 "ls" => self.listing(&opts, output),
552 "stat" => for &path in arguments {
553 match self.resolve(path, false, false) {
554 Err(error) => output.error(format!("stat: cannot stat '{path}': {}", error.reason())),
555 Ok(node) => {
556 let node = node.borrow();
557 let kind = match &node.data {
558 Data::Directory(_) => "directory", Data::Link(_) => "symbolic link",
559 Data::File(lines) => if lines.is_empty() { "regular empty file" } else { "regular file" },
560 };
561 output.emit(format!("{path}: {kind}, size {}, inode {}, links {}, modified {}", node.size(), node.inode, node.links(), node.modified));
562 }
563 }
564 },
565 "rm" | "rmdir" => for &path in opts.paths {
566 let directory_only = command == "rmdir";
567 let dot = matches!(last(path), "." | "..");
568 if dot && !directory_only {
569 output.error(format!("rm: refusing to remove '.' or '..' directory: skipping '{path}'"));
570 continue;
571 }
572 let problem = if dot { Some("Invalid argument") }
573 else {
574 match self.resolve(path, false, false) {
575 Err(error) if !directory_only && opts.has(b'f') && matches!(error.failure, Failure::Missing) => None,
576 Err(error) => Some(error.reason()),
577 Ok(node) if directory_only && !is_directory(&node) => Some("Not a directory"),
578 Ok(node) if !directory_only && is_directory(&node) && !opts.has(b'r') => Some("Is a directory"),
579 Ok(node) if self.descendant(&self.current, &node) => Some("Device or resource busy"),
580 Ok(node) if directory_only && !children(&node).is_empty() => Some("Directory not empty"),
581 Ok(node) => { self.detach(&node); None }
582 }
583 };
584 if let Some(reason) = problem {
585 if directory_only { output.error(format!("rmdir: failed to remove '{path}': {reason}")); }
586 else { output.error(format!("rm: cannot remove '{path}': {reason}")); }
587 }
588 },
589 "mv" | "cp" => {
590 let destination = opts.paths.last().unwrap();
591 if opts.paths.len() > 2 && !self.resolve(destination, true, false).is_ok_and(|node| is_directory(&node)) {
592 output.error(format!("{command}: target '{destination}' is not a directory"));
593 return;
594 }
595 for &source in &opts.paths[..opts.paths.len() - 1] {
596 if command == "mv" { self.move_one(source, destination, output); }
597 else { self.copy_one(source, destination, opts.has(b'r'), output); }
598 }
599 }
600 "ln" => {
601 let path = arguments[2];
602 match self.resolve(path, false, false) {
603 Ok(_) => output.error(format!("ln: failed to create symbolic link '{path}': File exists")),
604 Err(error) => {
605 if let Some((directory, name)) = error.creation.as_ref() { self.create(directory, name, Data::Link(arguments[1].to_owned())); }
606 else { output.error(format!("ln: failed to create symbolic link '{path}': {}", error.reason())); }
607 }
608 }
609 }
610 "readlink" | "realpath" => {
611 let path = arguments[0];
612 match self.resolve(path, command == "realpath", false) {
613 Err(error) => output.error(format!("{command}: {path}: {}", error.reason())),
614 Ok(node) if command == "realpath" => output.emit(self.absolute(&node)),
615 Ok(node) => {
616 let node = node.borrow();
617 if let Data::Link(target) = &node.data { output.emit(target.as_str()); }
618 else { output.error(format!("readlink: {path}: Invalid argument")); }
619 }
620 }
621 }
622 "find" => self.find(arguments, output),
623 "du" => self.disk_usage(&opts, output),
624 "tree" => self.tree(&opts, output),
625 "cat" | "wc" | "head" | "tail" | "grep" | "sort" | "uniq" => self.text_command(command, arguments, input, output),
626 _ => output.error(format!("{command}: command not found")),
627 }
628 }
629 fn process(&mut self, line: &str, terminal: &mut dyn Write) {
630 self.clock += 1;
631 writeln!(terminal, "{}$ {line}", self.absolute(&self.current)).unwrap();
632 let marker = line.find(" >> ").map(|index| (index, true, 4)).or_else(|| line.find(" > ").map(|index| (index, false, 3)));
633 let mut pipeline = line;
634 let mut redirect = None;
635 if let Some((index, append, length)) = marker {
636 pipeline = &line[..index];
637 let path = &line[index + length..];
638 match self.resolve(path, true, false) {
639 Ok(node) if is_directory(&node) => { writeln!(terminal, "sh: {path}: Is a directory").unwrap(); return; }
640 Ok(node) => {
641 if !append { self.write(&node, Vec::new(), false); }
642 redirect = Some(node);
643 }
644 Err(error) => {
645 if let Some((directory, name)) = error.creation.as_ref() { redirect = Some(self.create(directory, name, Data::File(Vec::new()))); }
646 else { writeln!(terminal, "sh: {path}: {}", error.reason()).unwrap(); return; }
647 }
648 }
649 }
650 let commands: Vec<_> = pipeline.split(" | ").collect();
651 let mut input = Vec::new();
652 for (i, command) in commands.iter().enumerate() {
653 let mut output = Output { terminal: &mut *terminal, direct: i + 1 == commands.len() && redirect.is_none(), lines: Vec::new() };
654 self.execute(command, &input, &mut output);
655 input = output.lines;
656 }
657 if let Some(file) = redirect { if !input.is_empty() { self.write(&file, input, true); } }
658 }
659}

661fn walk_nodes(node: &NodeRef, path: &str, depth: usize, limit: usize, visit: &mut impl FnMut(&NodeRef, &str, usize)) {
662 visit(node, path, depth);
663 if depth < limit {
664 for (name, child) in children(node) { walk_nodes(&child, &join(path, &name), depth + 1, limit, visit); }
665 }
666}
667fn glob_matches(pattern: &str, text: &str) -> bool {
668 let mut previous = vec![false; text.len() + 1];
669 previous[0] = true;
670 for character in pattern.bytes() {
671 let mut next = vec![false; previous.len()];
672 next[0] = character == b'*' && previous[0];
673 for (i, byte) in text.bytes().enumerate() {
674 next[i + 1] = if character == b'*' { previous[i + 1] || next[i] }
675 else { previous[i] && (character == b'?' || character == byte) };
676 }
677 previous = next;
678 }
679 previous[text.len()]
680}
681fn numeric_prefix(text: &str) -> i64 {
682 let negative = text.starts_with('-');
683 let mut value = 0;
684 for digit in text.bytes().skip(usize::from(negative)).take_while(u8::is_ascii_digit) { value = value * 10 + i64::from(digit - b'0'); }
685 if negative { -value } else { value }
686}
