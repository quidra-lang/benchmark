15final class DirectoryNode extends FsNode {
16 final entries = SplayTreeMap<String, FsNode>();
17 DirectoryNode(super.inode, super.modified);
18 @override
19 int get size => entries.length;
20 @override
21 int get links => 2 + entries.values.whereType<DirectoryNode>().length;
22 @override
23 String get tag => 'd';
24}

45enum Failure { none, missing, notDirectory, loop }

47final class Resolution {
48 final FsNode? node;
49 final Failure error;
50 final DirectoryNode? creationParent;
51 final String creationName;
52 const Resolution.found(this.node)
53 : error = Failure.none, creationParent = null, creationName = '';
54 const Resolution.failed(this.error)
55 : node = null, creationParent = null, creationName = '';
56 const Resolution.absent(this.creationParent, this.creationName)
57 : node = null, error = Failure.missing;
58 bool get absent => creationParent != null;
59 String get reason => switch (error) {
60 Failure.missing => 'No such file or directory',
61 Failure.notDirectory => 'Not a directory',
62 Failure.loop => 'Too many levels of symbolic links',
63 Failure.none => '',
64 };
65}

81String joinPath(String path, String name) => path == '/' ? '/$name' : '$path/$name';

99 String absolute(FsNode node) {
100 final parts = <String>[];
101 while (!identical(node, root)) {
102 parts.add(node.name);
103 node = node.parent;
104 }
105 return '/' + parts.reversed.join('/');
106 }

108 bool descendant(FsNode node, FsNode ancestor) {
109 while (true) {
110 if (identical(node, ancestor)) return true;
111 if (identical(node, root)) return false;
112 node = node.parent;
113 }

142 Resolution resolve(String path, {bool follow = true, bool parents = false}) {
143 var expansions = 0;
144 Resolution walk(String path, DirectoryNode base, bool follow, bool original) {
145 FsNode node = path.startsWith('/') ? root : base;
146 final parts = path.split('/').where((part) => part.isNotEmpty).toList();
147 for (var i = 0; i < parts.length; i++) {
148 if (node is! DirectoryNode) return const Resolution.failed(Failure.notDirectory);
149 final directory = node;
150 final part = parts[i], last = i + 1 == parts.length;
151 if (part == '.') continue;
152 if (part == '..') {
153 node = directory.parent;
154 continue;
155 }
156 var child = directory.entries[part];
157 if (child == null) {
158 if (original && last) return Resolution.absent(directory, part);
159 if (original && parents) {
160 child = makeDirectory(directory, part);
161 } else {
162 return const Resolution.failed(Failure.missing);
163 }
164 }
165 if (child is LinkNode && (!last || follow)) {
166 if (++expansions > 8) return const Resolution.failed(Failure.loop);
167 final result = walk(child.target, directory, true, false);
168 if (result.node == null) return result;
169 child = result.node!;
170 }
171 node = child;
172 }
173 return Resolution.found(node);
174 }
175 return walk(path, current, follow, true);
176 }

192 void move(String sourcePath, String destination) {
193 void problem(String path, String reason) =>
194 terminal("mv: cannot move '$sourcePath' to '$path': $reason");
195 if (invalidSource(sourcePath)) {
196 problem(destination, 'Invalid argument');
197 return;
198 }
199 final sourceResult = resolve(sourcePath, follow: false);
200 final source = sourceResult.node;
201 if (source == null) {
202 terminal("mv: cannot stat '$sourcePath': " + sourceResult.reason);
203 return;
204 }
205 DirectoryNode parent;
206 String name, display;
207 final destinationNode = resolve(destination).node;
208 if (destinationNode is DirectoryNode) {
209 parent = destinationNode;
210 name = lastComponent(sourcePath);
211 display = joinPath(destination, name);
212 } else {
213 final result = resolve(destination, follow: false);
214 if (result.absent) {
215 parent = result.creationParent!;
216 name = result.creationName;
217 } else if (result.node case final node?) {
218 parent = node.parent;
219 name = node.name;
220 } else {
221 problem(destination, result.reason);
222 return;
223 }
224 display = destination;
225 }
226 final existing = parent.entries[name];
227 if (source is DirectoryNode && descendant(parent, source)) {
228 terminal("mv: cannot move '$sourcePath' to a subdirectory of itself, '$display'");
229 } else if (identical(existing, source)) {
230 terminal("mv: '$sourcePath' and '$display' are the same file");
231 } else if (existing != null && source is DirectoryNode && existing is! DirectoryNode) {
232 terminal("mv: cannot overwrite non-directory '$display' with directory '$sourcePath'");
233 } else if (source is! DirectoryNode && existing is DirectoryNode) {
234 terminal("mv: cannot overwrite directory '$display' with non-directory '$sourcePath'");
235 } else if (existing is DirectoryNode && existing.entries.isNotEmpty) {
236 problem(display, 'Directory not empty');
237 } else if (identical(existing, current)) {
238 problem(display, 'Device or resource busy');
239 } else {
240 if (existing != null) detach(existing);
241 detach(source);
242 add(parent, name, source);
243 }
244 }

246 void copy(String sourcePath, String destination, bool recursive) {
247 void problem(String path, String reason) =>
248 terminal("cp: cannot copy '$sourcePath' to '$path': $reason");
249 if (invalidSource(sourcePath)) {
250 problem(destination, 'Invalid argument');
251 return;
252 }
253 final sourceResult = resolve(sourcePath);
254 final source = sourceResult.node;
255 if (source == null) {
256 terminal("cp: cannot stat '$sourcePath': " + sourceResult.reason);
257 return;
258 }
259 if (source is DirectoryNode && !recursive) {
260 terminal("cp: -r not specified; omitting directory '$sourcePath'");
261 return;
262 }
263 final result = resolve(destination);
264 if (result.node == null && !result.absent) {
265 problem(destination, result.reason);
266 return;
267 }
268 final display = result.node is DirectoryNode
269 ? joinPath(destination, lastComponent(sourcePath)) : destination;
270 final target = resolve(display, follow: source is! DirectoryNode);
271 if (source is DirectoryNode) {
272 if (target.node != null) {
273 problem(display, 'File exists');
274 } else if (descendant(target.creationParent!, source)) {
275 terminal("cp: cannot copy a directory, '$sourcePath', into itself, '$display'");
276 } else {
277 clone(source, target.creationParent!, target.creationName);
278 }
279 } else {
280 final file = source as FileNode;
281 if (target.absent) {
282 makeFile(target.creationParent!, target.creationName, file.lines);
283 } else if (identical(target.node, source)) {
284 terminal("cp: '$sourcePath' and '$display' are the same file");
285 } else if (target.node is DirectoryNode) {
286 terminal("cp: cannot overwrite directory '$display' with non-directory '$sourcePath'");
287 } else if (target.node case final FileNode other) {
288 write(other, file.lines);
289 } else {
290 problem(display, target.reason);
291 }
292 }
293 }

295 void readInputs(String command, List<String> paths, List<String> input, InputReader consume) {
296 if (paths.isEmpty) {
297 consume(input, null);
298 return;
299 }
300 for (final path in paths) {
301 final result = resolve(path);
302 switch (result.node) {
303 case FileNode file:
304 consume(file.lines, path);
305 case DirectoryNode():
306 terminal('$command: $path: Is a directory');
307 default:
308 terminal('$command: $path: ' + result.reason);
309 }
310 }
311 }

313 void walk(FsNode node, String path, void Function(FsNode, String, int) visit,
314 {bool hidden = true, int? maxDepth, int depth = 0}) {
315 visit(node, path, depth);
316 if (node is DirectoryNode && (maxDepth == null || depth < maxDepth)) {
317 for (final entry in node.entries.entries) {
318 if (hidden || !entry.key.startsWith('.')) {
319 walk(entry.value, joinPath(path, entry.key), visit,
320 hidden: hidden, maxDepth: maxDepth, depth: depth + 1);
321 }
322 }
323 }
324 }

326 void list(List<String> args, Output output) {
327 final (flags, given) = options(args);
328 final paths = given.isEmpty ? ['.'] : given;
329 for (final path in paths) {
330 final result = resolve(path, follow: !flags.contains('l'));
331 final node = result.node;
332 if (node == null) {
333 terminal("ls: cannot access '$path': " + result.reason);
334 continue;
335 }
336 var entries = <(String, FsNode)>[];
337 if (node is DirectoryNode) {
338 if (paths.length >= 2) output('$path:');
339 if (flags.contains('a')) entries.addAll([('.', node), ('..', node.parent)]);
340 for (final entry in node.entries.entries) {
341 if (flags.contains('a') || !entry.key.startsWith('.')) {
342 entries.add((entry.key, entry.value));
343 }
344 }
345 } else {
346 entries.add((path, node));
347 }
348 if (flags.contains('t')) {
349 final indexed = entries.indexed.toList();
350 indexed.sort((a, b) {
351 final comparison = b.$2.$2.modified.compareTo(a.$2.$2.modified);
352 return comparison != 0 ? comparison : a.$1.compareTo(b.$1);
353 });
354 entries = [for (final entry in indexed) entry.$2];
355 }
356 if (flags.contains('r')) entries = entries.reversed.toList();
357 var linkWidth = 0, sizeWidth = 0, timeWidth = 0;
358 for (final (_, node) in entries) {
359 linkWidth = larger(linkWidth, node.links.toString().length);
360 sizeWidth = larger(sizeWidth, node.size.toString().length);
361 timeWidth = larger(timeWidth, node.modified.toString().length);
362 }
363 for (final (name, node) in entries) {
364 if (!flags.contains('l')) {
365 output(name);
366 } else {
367 var line = node.tag + ' ' + node.links.toString().padLeft(linkWidth) + ' ' +
368 node.size.toString().padLeft(sizeWidth) + ' ' +
369 node.modified.toString().padLeft(timeWidth) + ' ' + name;
370 if (node is LinkNode) line += ' -> ' + node.target;
371 output(line);
372 }
373 }
374 }

377 void find(List<String> args, Output output) {
378 var at = 0, path = '.';
379 if (args.isNotEmpty && !args[0].startsWith('-')) path = args[at++];
380 String? pattern, type;
381 int? maxDepth;
382 while (at < args.length) {
383 final test = args[at++], value = args[at++];
384 switch (test) {
385 case '-name': pattern = value;
386 case '-type': type = value;
387 case '-maxdepth': maxDepth = int.parse(value);
388 }
389 }
390 final result = resolve(path), node = result.node;
391 if (node == null) {
392 terminal("find: '$path': " + result.reason);
393 return;
394 }
395 walk(node, path, (node, display, depth) {
396 final name = depth == 0 ? (path == '/' ? '/' : lastComponent(path)) : node.name;
397 final kind = node is FileNode ? 'f' : node.tag;
398 if ((pattern == null || globMatch(pattern!, name)) && (type == null || type == kind)) {
399 output(display);
400 }
401 }, maxDepth: maxDepth);
402 }

463 void wordCount(List<String> args, List<String> input, Output output) {
464 final (flags, paths) = options(args);
465 if (flags.isEmpty) flags.addAll(['l', 'w', 'c']);
466 var totalLines = 0, totalWords = 0, totalBytes = 0;
467 void report(int lines, int words, int bytes, String? name) {
468 final values = [
469 if (flags.contains('l')) lines,
470 if (flags.contains('w')) words,
471 if (flags.contains('c')) bytes,
472 ];
473 output(values.join(' ') + (name == null ? '' : ' $name'));
474 }
475 readInputs('wc', paths, input, (lines, name) {
476 var words = 0, bytes = 0;
477 for (final line in lines) {
478 var inside = false;
479 for (final character in line.codeUnits) {
480 if (character != 32 && !inside) words++;
481 inside = character != 32;
482 }
483 bytes += line.length + 1;
484 }
485 totalLines += lines.length;
486 totalWords += words;
487 totalBytes += bytes;
488 report(lines.length, words, bytes, name);
489 });
490 if (paths.length >= 2) report(totalLines, totalWords, totalBytes, 'total');
491 }

512 void grep(List<String> args, List<String> input, Output output) {
513 final (flags, rest) = options(args);
514 final pattern = flags.contains('i') ? rest[0].toLowerCase() : rest[0];
515 final paths = rest.sublist(1);
516 void consume(List<String> lines, String? name) {
517 final prefix = paths.length >= 2 || flags.contains('r') ? '$name:' : '';
518 var selected = 0;
519 for (var i = 0; i < lines.length; i++) {
520 final line = lines[i], compared = flags.contains('i') ? line.toLowerCase() : line;
521 if (compared.contains(pattern) == flags.contains('v')) continue;
522 selected++;
523 if (!flags.contains('c')) {
524 final number = flags.contains('n') ? (i + 1).toString() + ':' : '';
525 output(prefix + number + line);
526 }
527 }
528 if (flags.contains('c')) output('$prefix$selected');
529 }
530 if (!flags.contains('r')) {
531 readInputs('grep', paths, input, consume);
532 } else {
533 for (final path in paths.isEmpty ? ['.'] : paths) {
534 final result = resolve(path), node = result.node;
535 if (node == null) {
536 terminal('grep: $path: ' + result.reason);
537 } else {
538 walk(node, path, (child, display, _) {
539 if (child is FileNode) consume(child.lines, display);
540 });
541 }
542 }
543 }

572 void execute(String text, List<String> input, Output output) {
573 final words = text.split(' '), command = words[0], args = words.sublist(1);
574 switch (command) {
575 case 'pwd': output(absolute(current));
576 case 'echo': output(text.substring(5));
577 case 'cd':
578 final back = args.isNotEmpty && args[0] == '-';
579 if (back && previous == null) {
580 terminal('cd: OLDPWD not set');
581 return;
582 }
583 final path = back ? previous! : (args.isEmpty ? '/' : args[0]);
584 final result = resolve(path), node = result.node;
585 if (node == null) {
586 terminal('cd: $path: ' + result.reason);
587 } else if (node is! DirectoryNode) {
588 terminal('cd: $path: Not a directory');
589 } else {
590 previous = absolute(current);
591 current = node;
592 if (back) output(absolute(current));
593 }
594 case 'mkdir':
595 final (flags, paths) = options(args);
596 for (final path in paths) {
597 final result = resolve(path, follow: false, parents: flags.contains('p'));
598 if (result.absent) {
599 makeDirectory(result.creationParent!, result.creationName);
600 } else if (result.node == null) {
601 terminal("mkdir: cannot create directory '$path': " + result.reason);
602 } else if (!flags.contains('p') || resolve(path).node is! DirectoryNode) {
603 terminal("mkdir: cannot create directory '$path': File exists");
604 }
605 }
606 case 'touch':
607 for (final path in args) {
608 final result = resolve(path), node = result.node;
609 if (result.absent) {
610 makeFile(result.creationParent!, result.creationName);
611 } else if (node != null) {
612 node.modified = clock;
613 } else {
614 terminal("touch: cannot touch '$path': " + result.reason);
615 }
616 }
617 case 'cat': readInputs(command, args, input, (lines, _) { lines.forEach(output); });
618 case 'ls': list(args, output);
619 case 'stat':
620 for (final path in args) {
621 final result = resolve(path, follow: false), node = result.node;
622 if (node == null) {
623 terminal("stat: cannot stat '$path': " + result.reason);
624 } else {
625 final type = switch (node) {
626 DirectoryNode() => 'directory',
627 LinkNode() => 'symbolic link',
628 FileNode(:final lines) => lines.isEmpty ? 'regular empty file' : 'regular file',
629 };
630 output('$path: $type, size ' + node.size.toString() + ', inode ' + node.inode.toString() +
631 ', links ' + node.links.toString() + ', modified ' + node.modified.toString());
632 }
633 }
634 case 'rm':
635 final (flags, paths) = options(args);
636 for (final path in paths) {
637 void problem(String reason) => terminal("rm: cannot remove '$path': $reason");
638 if (const ['.', '..'].contains(lastComponent(path))) {
639 terminal("rm: refusing to remove '.' or '..' directory: skipping '$path'");
640 continue;
641 }
642 final result = resolve(path, follow: false), node = result.node;
643 if (node == null) {
644 if (!(flags.contains('f') && result.error == Failure.missing)) problem(result.reason);
645 } else if (node is DirectoryNode && !flags.contains('r')) {
646 problem('Is a directory');
647 } else if (descendant(current, node)) {
648 problem('Device or resource busy');
649 } else {
650 detach(node);
651 }
652 }
653 case 'rmdir':
654 for (final path in args) {
655 void problem(String reason) => terminal("rmdir: failed to remove '$path': $reason");
656 if (const ['.', '..'].contains(lastComponent(path))) {
657 problem('Invalid argument');
658 continue;
659 }
660 final result = resolve(path, follow: false), node = result.node;
661 if (node == null) {
662 problem(result.reason);
663 } else if (node is! DirectoryNode) {
664 problem('Not a directory');
665 } else if (descendant(current, node)) {
666 problem('Device or resource busy');
667 } else if (node.entries.isNotEmpty) {
668 problem('Directory not empty');
669 } else {
670 detach(node);
671 }
672 }
673 case 'mv':
674 case 'cp':
675 final (flags, paths) = command == 'cp' ? options(args) : (<String>{}, args);
676 final destination = paths.last, sources = paths.sublist(0, paths.length - 1);
677 if (sources.length >= 2 && resolve(destination).node is! DirectoryNode) {
678 terminal("$command: target '$destination' is not a directory");
679 return;
680 }
681 for (final source in sources) {
682 if (command == 'mv') {
683 move(source, destination);
684 } else {
685 copy(source, destination, flags.contains('r'));
686 }
687 }
688 case 'ln':
689 final target = args[1], path = args[2], result = resolve(path, follow: false);
690 if (result.absent) {
691 add(result.creationParent!, result.creationName, LinkNode(nextInode++, clock, target));
692 } else {
693 final reason = result.node == null ? result.reason : 'File exists';
694 terminal("ln: failed to create symbolic link '$path': $reason");
695 }
696 case 'readlink':
697 final path = args[0], result = resolve(path, follow: false);
698 if (result.node case final LinkNode link) {
699 output(link.target);
700 } else {
701 final reason = result.node == null ? result.reason : 'Invalid argument';
702 terminal('readlink: $path: $reason');
703 }
704 case 'realpath':
705 final path = args[0], result = resolve(path), node = result.node;
706 if (node == null) {
707 terminal('realpath: $path: ' + result.reason);
708 } else {
709 output(absolute(node));
710 }
711 case 'find': find(args, output);
712 case 'du': diskUsage(args, output);
713 case 'tree': tree(args, output);
714 case 'wc': wordCount(args, input, output);
715 case 'head':
716 case 'tail': ends(command, args, input, output);
717 case 'grep': grep(args, input, output);
718 case 'sort': sortLines(args, input, output);
719 case 'uniq': unique(args, input, output);
720 default: terminal('$command: command not found');
721 }
722 }

724 void process(String line) {
725 clock++;
726 terminal(absolute(current) + r'$ ' + line);
727 var pipeline = line;
728 FileNode? redirect;
729 final marker = line.indexOf(' >');
730 if (marker >= 0) {
731 pipeline = line.substring(0, marker);
732 final append = line.startsWith(' >> ', marker);
733 final path = line.substring(marker + (append ? 4 : 3));
734 final result = resolve(path), node = result.node;
735 if (result.absent) {
736 redirect = makeFile(result.creationParent!, result.creationName);
737 } else if (node is FileNode) {
738 redirect = node;
739 if (!append) write(node, const []);
740 } else {
741 final reason = node is DirectoryNode ? 'Is a directory' : result.reason;
742 terminal('sh: $path: $reason');
743 return;
744 }
745 }
746 var input = <String>[];
747 final commands = pipeline.split(' | ');
748 for (var i = 0; i < commands.length; i++) {
749 final output = <String>[];
750 final direct = i + 1 == commands.length && redirect == null;
751 execute(commands[i], input, direct ? terminal : output.add);
752 input = output;
753 }
754 if (redirect != null && input.isNotEmpty) write(redirect, input, append: true);
755 }
756}

758bool globMatch(String pattern, String text) {
759 var previous = List<bool>.filled(text.length + 1, false);
760 previous[0] = true;
761 for (final character in pattern.split('')) {
762 final next = List<bool>.filled(text.length + 1, false);
763 next[0] = character == '*' && previous[0];
764 for (var i = 1; i <= text.length; i++) {
765 next[i] = character == '*' ? previous[i] || next[i - 1] :
766 previous[i - 1] && (character == '?' || character == text[i - 1]);
767 }
768 previous = next;
769 }
770 return previous[text.length];
771}

773int numericPrefix(String line) {
774 var at = line.startsWith('-') ? 1 : 0, value = 0;
775 while (at < line.length) {
776 final digit = line.codeUnitAt(at) - 48;
777 if (digit < 0 || digit > 9) break;
778 value = value * 10 + digit;
779 at++;
780 }
781 return line.startsWith('-') ? -value : value;
782}
