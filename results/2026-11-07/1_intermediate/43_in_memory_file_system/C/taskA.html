38static void *allocate(size_t size) {
39 void *memory = malloc(size);
40 if (!memory) abort();
41 return memory;
42}
43static void *resize(void *memory, size_t size) {
44 void *result = realloc(memory, size);
45 if (!result) abort();
46 return result;
47}
48static char *copy_text(const char *text) {
49 size_t size = strlen(text) + 1;
50 char *result = allocate(size);
51 memcpy(result, text, size);
52 return result;
53}
54static char *concatenate(const char *first, const char *last) {
55 size_t a = strlen(first), b = strlen(last);
56 char *result = allocate(a + b + 1);
57 memcpy(result, first, a);
58 memcpy(result + a, last, b + 1);
59 return result;
60}
61static void release_line(Line *line) {
62 if (--line->references == 0) free(line);
63}
64static void push_line(Lines *lines, Line *line) {
65 if (lines->count == lines->capacity) {
66 lines->capacity = lines->capacity ? lines->capacity * 2 : 8;
67 lines->items = resize(lines->items, lines->capacity * sizeof(*lines->items));
68 }
69 line->references++;
70 lines->items[lines->count++] = line;
71}
72static void clear_lines(Lines *lines) {
73 for (size_t i = 0; i < lines->count; i++) release_line(lines->items[i]);
74 free(lines->items);
75 *lines = (Lines){0};
76}
77static void append_lines(Lines *destination, const Lines *source) {
78 for (size_t i = 0; i < source->count; i++) push_line(destination, source->items[i]);
79}
80static void emit_line(Output *output, Line *line) {
81 if (output->terminal) puts(line->text);
82 else push_line(&output->lines, line);
83}
84static void emit(Output *output, const char *format, ...) {
85 va_list args, measured;
86 va_start(args, format);
87 va_copy(measured, args);
88 int length = vsnprintf(NULL, 0, format, measured);
89 va_end(measured);
90 if (length < 0) abort();
91 Line *line = allocate(sizeof(*line) + (size_t)length + 1);
92 line->references = 1;
93 line->length = (size_t)length;
94 vsnprintf(line->text, line->length + 1, format, args);
95 va_end(args);
96 emit_line(output, line);
97 release_line(line);
98}
99static void error(const char *format, ...) {
100 va_list args;
101 va_start(args, format);
102 vprintf(format, args);
103 va_end(args);
104 putchar('\n');
105}
106static void push_node(Nodes *nodes, Node *node) {
107 if (nodes->count == nodes->capacity) {
108 nodes->capacity = nodes->capacity ? nodes->capacity * 2 : 8;
109 nodes->items = resize(nodes->items, nodes->capacity * sizeof(*nodes->items));
110 }
111 nodes->items[nodes->count++] = node;
112}
113static Node *child(Node *directory, const char *name) {
114 for (size_t i = 0; i < directory->data.children.count; i++) {
115 Node *node = directory->data.children.items[i];
116 if (strcmp(node->name, name) == 0) return node;
117 }
118 return NULL;
119}
120static void add_node(Shell *shell, Node *parent, const char *name, Node *node) {
121 char *owned_name = copy_text(name);
122 free(node->name);
123 node->name = owned_name;
124 node->parent = parent;
125 Nodes *children = &parent->data.children;
126 push_node(children, node);
127 size_t index = children->count - 1;
128 while (index && strcmp(children->items[index - 1]->name, owned_name) > 0) {
129 children->items[index] = children->items[index - 1];
130 index--;
131 }
132 children->items[index] = node;
133 parent->modified = shell->clock;
134}
135static Node *create_node(Shell *shell, Node *parent, const char *name, Kind kind, const char *target) {
136 Node *node = allocate(sizeof(*node));
137 *node = (Node){.kind = kind, .inode = shell->next_inode++, .modified = shell->clock};
138 if (kind == LINK) node->data.target = copy_text(target);
139 else if (kind == FILE_NODE) node->data.lines = (Lines){0};
140 add_node(shell, parent, name, node);
141 return node;
142}
143static void detach_node(Shell *shell, Node *node) {
144 Nodes *children = &node->parent->data.children;
145 size_t index = 0;
146 while (children->items[index] != node) index++;
147 memmove(children->items + index, children->items + index + 1,
148 (children->count - index - 1) * sizeof(*children->items));
149 children->count--;
150 node->parent->modified = shell->clock;
151}
152static void remove_node(Shell *shell, Node *node) {
153 detach_node(shell, node);
154 push_node(&shell->retired, node);
155}
156static void free_node(Node *node) {
157 if (node->kind == DIRECTORY) {
158 for (size_t i = 0; i < node->data.children.count; i++) free_node(node->data.children.items[i]);
159 free(node->data.children.items);
160 } else if (node->kind == FILE_NODE) clear_lines(&node->data.lines);
161 else free(node->data.target);
162 free(node->name);
163 free(node);
164}
165static void write_file(Shell *shell, Node *file, const Lines *lines, bool append) {
166 if (append) append_lines(&file->data.lines, lines);
167 else {
168 Lines replacement = {0};
169 append_lines(&replacement, lines);
170 clear_lines(&file->data.lines);
171 file->data.lines = replacement;
172 }
173 file->modified = shell->clock;
174}
175static bool descendant(const Shell *shell, Node *node, Node *ancestor) {
176 for (;;) {
177 if (node == ancestor) return true;
178 if (node == shell->root) return false;
179 node = node->parent;
180 }
181}
182static char *absolute(const Shell *shell, Node *node) {
183 size_t length = 0;
184 for (Node *part = node; part != shell->root; part = part->parent) length += strlen(part->name) + 1;
185 if (!length) return copy_text("/");
186 char *result = allocate(length + 1);
187 result[length] = '\0';
188 while (node != shell->root) {
189 size_t size = strlen(node->name);
190 length -= size;
191 memcpy(result + length, node->name, size);
192 result[--length] = '/';
193 node = node->parent;
194 }
195 return result;
196}
197static const char *last_component(const char *path) {
198 const char *slash = strrchr(path, '/');
199 return slash ? slash + 1 : path;
200}
201static char *join(const char *path, const char *name) {
202 char *prefix = strcmp(path, "/") == 0 ? copy_text(path) : concatenate(path, "/");
203 char *result = concatenate(prefix, name);
204 free(prefix);
205 return result;
206}
207static const char *reason(Resolution result) {
208 switch (result.failure) {
209 case MISSING: return "No such file or directory";
210 case NOT_DIRECTORY: return "Not a directory";
211 case TOO_MANY_LINKS: return "Too many levels of symbolic links";
212 default: return "";
213 }
214}
215static Resolution walk(Shell *shell, const char *path, Node *base, bool follow,
216 bool original, bool make_parents, int *expansions) {
217 Node *node = path[0] == '/' ? shell->root : base;
218 const char *position = path;
219 while (*position) {
220 if (*position == '/') { position++; continue; }
221 if (node->kind != DIRECTORY) return (Resolution){.failure = NOT_DIRECTORY};
222 const char *end = strchr(position, '/');
223 if (!end) end = position + strlen(position);
224 char name[17];
225 size_t length = (size_t)(end - position);
226 memcpy(name, position, length);
227 name[length] = '\0';
228 bool last = !*end;
229 position = end;
230 if (strcmp(name, ".") == 0) continue;
231 if (strcmp(name, "..") == 0) { node = node->parent; continue; }
232 Node *next = child(node, name);
233 if (!next) {
234 if (original && make_parents && !last) next = create_node(shell, node, name, DIRECTORY, NULL);
235 else {
236 Resolution missing = {.failure = MISSING};
237 if (original && last) {
238 missing.creation_parent = node;
239 strcpy(missing.creation_name, name);
240 }
241 return missing;
242 }
243 }
244 if (next->kind == LINK && (!last || follow)) {
245 if (++*expansions > 8) return (Resolution){.failure = TOO_MANY_LINKS};
246 Resolution expanded = walk(shell, next->data.target, node, true, false, false, expansions);
247 if (!expanded.node) return expanded;
248 next = expanded.node;
249 }
250 node = next;
251 }
252 return (Resolution){.node = node};
253}
254static Resolution resolve(Shell *shell, const char *path, bool follow, bool parents) {
255 int expansions = 0;
256 return walk(shell, path, shell->current, follow, true, parents, &expansions);
257}
258static int64_t node_size(const Node *node) {
259 if (node->kind == DIRECTORY) return (int64_t)node->data.children.count;
260 if (node->kind == LINK) return (int64_t)strlen(node->data.target);
261 int64_t size = 0;
262 for (size_t i = 0; i < node->data.lines.count; i++) size += (int64_t)node->data.lines.items[i]->length + 1;
263 return size;
264}
265static int link_count(const Node *node) {
266 if (node->kind != DIRECTORY) return 1;
267 int count = 2;
268 for (size_t i = 0; i < node->data.children.count; i++) count += node->data.children.items[i]->kind == DIRECTORY;
269 return count;
270}
271static char tag(const Node *node) { return node->kind == DIRECTORY ? 'd' : node->kind == FILE_NODE ? 'f' : 'l'; }
272static Options options(char **arguments, int count) {
273 Options result = {0};
274 int position = 0;
275 while (position < count && arguments[position][0] == '-') {
276 for (const char *c = arguments[position++] + 1; *c; c++) result.flags[(unsigned char)*c] = true;
277 }
278 result.paths = arguments + position;
279 result.count = count - position;
280 return result;

282static bool invalid_source(const char *name) {
283 return !*name || strcmp(name, ".") == 0 || strcmp(name, "..") == 0;
284}
285static void clone_node(Shell *shell, Node *source, Node *parent, const char *name) {
286 Node *copy = create_node(shell, parent, name, source->kind, source->kind == LINK ? source->data.target : NULL);
287 if (source->kind == FILE_NODE) append_lines(&copy->data.lines, &source->data.lines);
288 else if (source->kind == DIRECTORY)
289 for (size_t i = 0; i < source->data.children.count; i++) {
290 Node *entry = source->data.children.items[i];
291 clone_node(shell, entry, copy, entry->name);
292 }

294static void move_one(Shell *shell, const char *source_path, const char *destination) {
295 const char *basename = last_component(source_path);
296 if (invalid_source(basename)) { error("mv: cannot move '%s' to '%s': Invalid argument", source_path, destination); return; }
297 Resolution source_result = resolve(shell, source_path, false, false);
298 Node *source = source_result.node;
299 if (!source) { error("mv: cannot stat '%s': %s", source_path, reason(source_result)); return; }
300 Resolution destination_result = resolve(shell, destination, true, false);
301 Node *parent;
302 char name[17];
303 char *target;
304 if (destination_result.node && destination_result.node->kind == DIRECTORY) {
305 parent = destination_result.node;
306 strcpy(name, basename);
307 target = join(destination, basename);
308 } else {
309 Resolution result = resolve(shell, destination, false, false);
310 if (result.node) { parent = result.node->parent; strcpy(name, result.node->name); }
311 else if (result.creation_parent) { parent = result.creation_parent; strcpy(name, result.creation_name); }
312 else { error("mv: cannot move '%s' to '%s': %s", source_path, destination, reason(result)); return; }
313 target = copy_text(destination);
314 }
315 Node *existing = child(parent, name);
316 if (source->kind == DIRECTORY && descendant(shell, parent, source))
317 error("mv: cannot move '%s' to a subdirectory of itself, '%s'", source_path, target);
318 else if (existing == source) error("mv: '%s' and '%s' are the same file", source_path, target);
319 else if (existing && source->kind == DIRECTORY && existing->kind != DIRECTORY)
320 error("mv: cannot overwrite non-directory '%s' with directory '%s'", target, source_path);
321 else if (existing && source->kind != DIRECTORY && existing->kind == DIRECTORY)
322 error("mv: cannot overwrite directory '%s' with non-directory '%s'", target, source_path);
323 else if (existing && existing->kind == DIRECTORY && existing->data.children.count)
324 error("mv: cannot move '%s' to '%s': Directory not empty", source_path, target);
325 else if (existing == shell->current) error("mv: cannot move '%s' to '%s': Device or resource busy", source_path, target);
326 else {
327 if (existing) remove_node(shell, existing);
328 detach_node(shell, source);
329 add_node(shell, parent, name, source);
330 }
331 free(target);
332}
333static void copy_one(Shell *shell, const char *source_path, const char *destination, bool recursive) {
334 const char *basename = last_component(source_path);
335 if (invalid_source(basename)) { error("cp: cannot copy '%s' to '%s': Invalid argument", source_path, destination); return; }
336 Resolution source_result = resolve(shell, source_path, true, false);
337 Node *source = source_result.node;
338 if (!source) { error("cp: cannot stat '%s': %s", source_path, reason(source_result)); return; }
339 if (source->kind == DIRECTORY && !recursive) { error("cp: -r not specified; omitting directory '%s'", source_path); return; }
340 Resolution destination_result = resolve(shell, destination, true, false);
341 if (!destination_result.node && !destination_result.creation_parent) {
342 error("cp: cannot copy '%s' to '%s': %s", source_path, destination, reason(destination_result));
343 return;
344 }
345 char *target = destination_result.node && destination_result.node->kind == DIRECTORY ? join(destination, basename) : copy_text(destination);
346 Resolution result = resolve(shell, target, source->kind != DIRECTORY, false);
347 if (source->kind == DIRECTORY) {
348 if (result.node) error("cp: cannot copy '%s' to '%s': File exists", source_path, target);
349 else if (!result.creation_parent) error("cp: cannot copy '%s' to '%s': %s", source_path, target, reason(result));
350 else if (descendant(shell, result.creation_parent, source)) error("cp: cannot copy a directory, '%s', into itself, '%s'", source_path, target);
351 else clone_node(shell, source, result.creation_parent, result.creation_name);
352 } else if (result.creation_parent) {
353 Node *copy = create_node(shell, result.creation_parent, result.creation_name, FILE_NODE, NULL);
354 append_lines(&copy->data.lines, &source->data.lines);
355 } else if (!result.node) error("cp: cannot copy '%s' to '%s': %s", source_path, target, reason(result));
356 else if (result.node == source) error("cp: '%s' and '%s' are the same file", source_path, target);
357 else if (result.node->kind == DIRECTORY) error("cp: cannot overwrite directory '%s' with non-directory '%s'", target, source_path);
358 else write_file(shell, result.node, &source->data.lines, false);
359 free(target);
360}

362typedef struct { const char *name; Node *node; size_t order; } ListingEntry;
363static int compare_modified(const void *left, const void *right) {
364 const ListingEntry *a = left, *b = right;
365 if (a->node->modified != b->node->modified) return a->node->modified > b->node->modified ? -1 : 1;
366 return (a->order > b->order) - (a->order < b->order);
367}
368static int digits(int64_t number) {
369 int width = 1;
370 while (number >= 10) { width++; number /= 10; }
371 return width;
372}
373static void listing(Shell *shell, Options opts, Output *output) {
374 char *default_path = ".";
375 if (!opts.count) { opts.paths = &default_path; opts.count = 1; }
376 for (int path_index = 0; path_index < opts.count; path_index++) {
377 const char *path = opts.paths[path_index];
378 Resolution result = resolve(shell, path, !opts.flags['l'], false);
379 if (!result.node) { error("ls: cannot access '%s': %s", path, reason(result)); continue; }
380 Node *node = result.node;
381 size_t capacity = node->kind == DIRECTORY ? node->data.children.count + 2 : 1;
382 ListingEntry *entries = allocate(capacity * sizeof(*entries));
383 size_t count = 0;
384 if (node->kind != DIRECTORY) entries[count++] = (ListingEntry){path, node, 0};
385 else {
386 if (opts.count > 1) emit(output, "%s:", path);
387 if (opts.flags['a']) {
388 entries[count++] = (ListingEntry){".", node, 0};
389 entries[count++] = (ListingEntry){"..", node->parent, 1};
390 }
391 for (size_t i = 0; i < node->data.children.count; i++) {
392 Node *entry = node->data.children.items[i];
393 if (!opts.flags['a'] && entry->name[0] == '.') continue;
394 entries[count] = (ListingEntry){entry->name, entry, count};
395 count++;
396 }
397 if (opts.flags['t']) qsort(entries, count, sizeof(*entries), compare_modified);
398 }
399 int link_width = 1, size_width = 1, time_width = 1;
400 for (size_t i = 0; i < count; i++) {
401 int a = digits(link_count(entries[i].node)), b = digits(node_size(entries[i].node)), c = digits(entries[i].node->modified);
402 if (a > link_width) link_width = a;
403 if (b > size_width) size_width = b;
404 if (c > time_width) time_width = c;
405 }
406 for (size_t i = 0; i < count; i++) {
407 ListingEntry entry = entries[opts.flags['r'] ? count - i - 1 : i];
408 Node *item = entry.node;
409 if (!opts.flags['l']) emit(output, "%s", entry.name);
410 else emit(output, "%c %*d %*" PRId64 " %*d %s%s%s", item->kind == FILE_NODE ? '-' : tag(item),
411 link_width, link_count(item), size_width, node_size(item), time_width, item->modified,
412 entry.name, item->kind == LINK ? " -> " : "", item->kind == LINK ? item->data.target : "");
413 }
414 free(entries);
415 }

417typedef void (*Visitor)(Node *, const char *, int, void *);
418static void visit_tree(Node *node, const char *path, int depth, int limit, Visitor visit, void *context) {
419 visit(node, path, depth, context);
420 if (node->kind == DIRECTORY && depth < limit)
421 for (size_t i = 0; i < node->data.children.count; i++) {
422 Node *entry = node->data.children.items[i];
423 char *display = join(path, entry->name);
424 visit_tree(entry, display, depth + 1, limit, visit, context);
425 free(display);
426 }
427}
428static bool glob_matches(const char *pattern, const char *text) {
429 size_t length = strlen(text);
430 bool previous[18] = {true};
431 for (const char *c = pattern; *c; c++) {
432 bool next[18] = {false};
433 if (*c == '*') {
434 next[0] = previous[0];
435 for (size_t i = 0; i < length; i++) next[i + 1] = previous[i + 1] || next[i];
436 } else
437 for (size_t i = 0; i < length; i++) next[i + 1] = previous[i] && (*c == '?' || *c == text[i]);
438 memcpy(previous, next, sizeof(previous));
439 }
440 return previous[length];
441}
442typedef struct { Output *output; const char *pattern, *root_name; char type; } FindContext;
443static void find_visit(Node *node, const char *path, int depth, void *context) {
444 FindContext *find = context;
445 const char *name = depth ? node->name : find->root_name;
446 if ((!find->pattern || glob_matches(find->pattern, name)) && (!find->type || tag(node) == find->type)) emit(find->output, "%s", path);
447}
448static void find_nodes(Shell *shell, char **arguments, int count, Output *output) {
449 int position = count && arguments[0][0] != '-' ? 1 : 0;
450 const char *path = position ? arguments[0] : ".";
451 FindContext find = {.output = output, .root_name = strcmp(path, "/") == 0 ? "/" : last_component(path)};
452 int limit = 1000;
453 for (; position < count; position += 2) {
454 const char *test = arguments[position], *value = arguments[position + 1];
455 if (strcmp(test, "-name") == 0) find.pattern = value;
456 else if (strcmp(test, "-type") == 0) find.type = value[0];
457 else limit = atoi(value);
458 }
459 Resolution result = resolve(shell, path, true, false);
460 if (!result.node) error("find: '%s': %s", path, reason(result));
461 else visit_tree(result.node, path, 0, limit, find_visit, &find);
462}
463static int64_t usage(Node *node, const char *path, bool show, Options opts, Output *output) {
464 int64_t total = node->kind == DIRECTORY ? 0 : node_size(node);
465 if (node->kind == DIRECTORY)
466 for (size_t i = 0; i < node->data.children.count; i++) {
467 Node *entry = node->data.children.items[i];
468 char *display = join(path, entry->name);
469 total += usage(entry, display, show && !opts.flags['s'] && (entry->kind == DIRECTORY || opts.flags['a']), opts, output);
470 free(display);
471 }
472 if (show) emit(output, "%" PRId64 " %s", total, path);
473 return total;
474}
475static void disk_usage(Shell *shell, Options opts, Output *output) {
476 char *default_path = ".";
477 if (!opts.count) { opts.paths = &default_path; opts.count = 1; }
478 for (int i = 0; i < opts.count; i++) {
479 Resolution result = resolve(shell, opts.paths[i], true, false);
480 if (!result.node) error("du: cannot access '%s': %s", opts.paths[i], reason(result));
481 else usage(result.node, opts.paths[i], true, opts, output);

484static void draw_tree(Node *node, const char *prefix, bool all, int *directories, int *files, Output *output) {
485 size_t visible = 0;
486 for (size_t i = 0; i < node->data.children.count; i++) visible += all || node->data.children.items[i]->name[0] != '.';
487 for (size_t i = 0; i < node->data.children.count; i++) {
488 Node *entry = node->data.children.items[i];
489 if (!all && entry->name[0] == '.') continue;
490 bool last = --visible == 0;
491 emit(output, "%s%s%s%s%s", prefix, last ? "`-- " : "|-- ", entry->name,
492 entry->kind == LINK ? " -> " : "", entry->kind == LINK ? entry->data.target : "");
493 if (entry->kind == DIRECTORY) {
494 (*directories)++;
495 char *next = concatenate(prefix, last ? " " : "| ");
496 draw_tree(entry, next, all, directories, files, output);
497 free(next);
498 } else (*files)++;
499 }
500}
501static void tree(Shell *shell, Options opts, Output *output) {
502 const char *path = opts.count ? opts.paths[0] : ".";
503 Resolution result = resolve(shell, path, true, false);
504 if (!result.node) { error("tree: %s: %s", path, reason(result)); return; }
505 if (result.node->kind != DIRECTORY) { error("tree: %s: Not a directory", path); return; }
506 int directories = 0, files = 0;
507 emit(output, "%s", path);
508 draw_tree(result.node, "", opts.flags['a'], &directories, &files, output);
509 emit(output, "%d %s, %d %s", directories, directories == 1 ? "directory" : "directories", files, files == 1 ? "file" : "files");
510}

520static char lower_ascii(char character) { return character >= 'A' && character <= 'Z' ? character + ('a' - 'A') : character; }
521static bool contains(const char *text, const char *pattern, bool insensitive) {
522 if (!insensitive) return strstr(text, pattern) != NULL;
523 for (; *text; text++) {
524 size_t i = 0;
525 while (text[i] && pattern[i] && lower_ascii(text[i]) == lower_ascii(pattern[i])) i++;
526 if (!pattern[i]) return true;
527 }
528 return false;
529}
530static void report_counts(TextJob *job, const int64_t counts[3], const char *name) {
531 char numbers[80];
532 size_t used = 0;
533 bool all = !job->options.flags['l'] && !job->options.flags['w'] && !job->options.flags['c'];
534 for (int i = 0; i < 3; i++)
535 if (all || job->options.flags[(unsigned char)"lwc"[i]])
536 used += (size_t)snprintf(numbers + used, sizeof(numbers) - used, "%s%" PRId64, used ? " " : "", counts[i]);
537 emit(job->output, "%s%s%s", numbers, name ? " " : "", name ? name : "");
538}
539static void consume_input(TextJob *job, const Lines *lines, const char *name) {
540 const char *command = job->command;
541 Output *output = job->output;
542 if (strcmp(command, "sort") == 0) { append_lines(&job->sorted, lines); return; }
543 if (strcmp(command, "cat") == 0) {
544 for (size_t i = 0; i < lines->count; i++) emit_line(output, lines->items[i]);
545 } else if (strcmp(command, "wc") == 0) {
546 int64_t counts[3] = {(int64_t)lines->count, 0, 0};
547 for (size_t i = 0; i < lines->count; i++) {
548 Line *line = lines->items[i];
549 counts[2] += (int64_t)line->length + 1;
550 bool inside = false;
551 for (const char *c = line->text; *c; c++) {
552 if (*c != ' ' && !inside) counts[1]++;
553 inside = *c != ' ';
554 }
555 }
556 for (int i = 0; i < 3; i++) job->totals[i] += counts[i];
557 report_counts(job, counts, name);
558 } else if (strcmp(command, "head") == 0 || strcmp(command, "tail") == 0) {
559 const char *count_text = job->count_text;
560 size_t count = (size_t)atoi(count_text + (count_text[0] == '-' || count_text[0] == '+'));
561 size_t first = 0, last = lines->count;
562 if (strcmp(command, "head") == 0) last = count_text[0] == '-' ? (last > count ? last - count : 0) : (last < count ? last : count);
563 else if (count_text[0] == '+') { first = count ? count - 1 : 0; if (first > last) first = last; }
564 else first = last > count ? last - count : 0;
565 for (size_t i = first; i < last; i++) emit_line(output, lines->items[i]);
566 } else if (strcmp(command, "grep") == 0) {
567 size_t selected = 0;
568 for (size_t i = 0; i < lines->count; i++) {
569 Line *line = lines->items[i];
570 if (contains(line->text, job->pattern, job->options.flags['i']) == job->options.flags['v']) continue;
571 selected++;
572 if (!job->options.flags['c']) {
573 if (job->options.flags['n']) emit(output, "%s%s%zu:%s", job->named ? name : "", job->named ? ":" : "", i + 1, line->text);
574 else if (job->named) emit(output, "%s:%s", name, line->text);
575 else emit_line(output, line);
576 }
577 }
578 if (job->options.flags['c']) emit(output, "%s%s%zu", job->named ? name : "", job->named ? ":" : "", selected);
579 } else {
580 for (size_t first = 0; first < lines->count;) {
581 size_t last = first + 1;
582 while (last < lines->count && strcmp(lines->items[first]->text, lines->items[last]->text) == 0) last++;
583 if (job->options.flags['c']) emit(output, "%zu %s", last - first, lines->items[first]->text);
584 else emit_line(output, lines->items[first]);
585 first = last;
586 }
587 }
588}
589static void grep_visit(Node *node, const char *path, int depth, void *context) {
590 (void)depth;
591 if (node->kind == FILE_NODE) consume_input(context, &node->data.lines, path);
592}
593static int numeric_value(const char *text) {
594 bool negative = *text == '-';
595 if (negative) text++;
596 int value = 0;
597 while (*text >= '0' && *text <= '9') value = value * 10 + *text++ - '0';
598 return negative ? -value : value;
599}
600static int compare_lines(const void *left, const void *right) {
601 Line *const *a = left, *const *b = right;
602 return strcmp((*a)->text, (*b)->text);
603}
604static int compare_numbers(const void *left, const void *right) {
605 Line *const *a = left, *const *b = right;
606 int x = numeric_value((*a)->text), y = numeric_value((*b)->text);
607 return x == y ? compare_lines(left, right) : (x > y) - (x < y);
608}
609static void text_command(Shell *shell, const char *command, char **arguments, int count, const Lines *input, Output *output) {
610 Options opts = options(arguments, count);
611 TextJob job = {.command = command, .options = opts, .output = output, .count_text = "10"};
612 char **paths = opts.paths;
613 int path_count = opts.count;
614 bool recursive = false;
615 if (strcmp(command, "grep") == 0) {
616 job.pattern = *paths++;
617 path_count--;
618 recursive = opts.flags['r'];
619 job.named = path_count > 1 || recursive;
620 } else if (strcmp(command, "head") == 0 || strcmp(command, "tail") == 0) {
621 int offset = count && strcmp(arguments[0], "-n") == 0 ? 2 : 0;
622 if (offset) job.count_text = arguments[1];
623 paths = arguments + offset;
624 path_count = count - offset;
625 }
626 char *default_path = ".";
627 if (recursive && !path_count) { paths = &default_path; path_count = 1; }
628 if (!path_count) consume_input(&job, input, NULL);
629 for (int i = 0; i < path_count; i++) {
630 Resolution result = resolve(shell, paths[i], true, false);
631 if (!result.node) error("%s: %s: %s", command, paths[i], reason(result));
632 else if (recursive) visit_tree(result.node, paths[i], 0, 1000, grep_visit, &job);
633 else if (result.node->kind == DIRECTORY) error("%s: %s: Is a directory", command, paths[i]);
634 else consume_input(&job, &result.node->data.lines, paths[i]);
635 }
636 if (strcmp(command, "wc") == 0 && path_count > 1) report_counts(&job, job.totals, "total");
637 if (strcmp(command, "sort") == 0) {
638 if (job.sorted.count > 1) qsort(job.sorted.items, job.sorted.count, sizeof(*job.sorted.items), opts.flags['n'] ? compare_numbers : compare_lines);
639 for (size_t i = 0; i < job.sorted.count; i++) emit_line(output, job.sorted.items[opts.flags['r'] ? job.sorted.count - i - 1 : i]);
640 clear_lines(&job.sorted);
641 }
642}

644static void execute(Shell *shell, char *text, const Lines *input, Output *output) {
645 if (strncmp(text, "echo ", 5) == 0) { emit(output, "%s", text + 5); return; }
646 char *words[101];
647 int length = 0;
648 for (char *word = strtok(text, " "); word; word = strtok(NULL, " ")) words[length++] = word;
649 const char *command = words[0];
650 char **arguments = words + 1;
651 int count = length - 1;
652 Options opts = options(arguments, count);
653 if (strcmp(command, "pwd") == 0) {
654 char *path = absolute(shell, shell->current);
655 emit(output, "%s", path);
656 free(path);
657 } else if (strcmp(command, "cd") == 0) {
658 const char *path = count ? arguments[0] : "/";
659 bool previous = strcmp(path, "-") == 0;
660 if (previous) {
661 if (!shell->previous) { error("cd: OLDPWD not set"); return; }
662 path = shell->previous;
663 }
664 Resolution result = resolve(shell, path, true, false);
665 if (!result.node) error("cd: %s: %s", path, reason(result));
666 else if (result.node->kind != DIRECTORY) error("cd: %s: Not a directory", path);
667 else {
668 char *old = absolute(shell, shell->current);
669 free(shell->previous);
670 shell->previous = old;
671 shell->current = result.node;
672 if (previous) {
673 char *current_path = absolute(shell, shell->current);
674 emit(output, "%s", current_path);
675 free(current_path);
676 }
677 }
678 } else if (strcmp(command, "mkdir") == 0) {
679 for (int i = 0; i < opts.count; i++) {
680 const char *path = opts.paths[i];
681 Resolution result = resolve(shell, path, false, opts.flags['p']);
682 if (result.creation_parent) create_node(shell, result.creation_parent, result.creation_name, DIRECTORY, NULL);
683 else if (!result.node) error("mkdir: cannot create directory '%s': %s", path, reason(result));
684 else {
685 Resolution followed = opts.flags['p'] ? resolve(shell, path, true, false) : (Resolution){0};
686 if (!followed.node || followed.node->kind != DIRECTORY) error("mkdir: cannot create directory '%s': File exists", path);
687 }
688 }
689 } else if (strcmp(command, "touch") == 0) {
690 for (int i = 0; i < count; i++) {
691 Resolution result = resolve(shell, arguments[i], true, false);
692 if (result.creation_parent) create_node(shell, result.creation_parent, result.creation_name, FILE_NODE, NULL);
693 else if (result.node) result.node->modified = shell->clock;
694 else error("touch: cannot touch '%s': %s", arguments[i], reason(result));
695 }
696 } else if (strcmp(command, "ls") == 0) listing(shell, opts, output);
697 else if (strcmp(command, "stat") == 0) {
698 for (int i = 0; i < count; i++) {
699 Resolution result = resolve(shell, arguments[i], false, false);
700 Node *node = result.node;
701 if (!node) { error("stat: cannot stat '%s': %s", arguments[i], reason(result)); continue; }
702 const char *type = node->kind == DIRECTORY ? "directory" : node->kind == LINK ? "symbolic link" :
703 node->data.lines.count ? "regular file" : "regular empty file";
704 emit(output, "%s: %s, size %" PRId64 ", inode %d, links %d, modified %d", arguments[i], type,
705 node_size(node), node->inode, link_count(node), node->modified);
706 }
707 } else if (strcmp(command, "rm") == 0 || strcmp(command, "rmdir") == 0) {
708 bool directory_only = strcmp(command, "rmdir") == 0;
709 for (int i = 0; i < opts.count; i++) {
710 const char *path = opts.paths[i], *last = last_component(path), *problem = NULL;
711 bool dot = strcmp(last, ".") == 0 || strcmp(last, "..") == 0;
712 if (dot && !directory_only) {
713 error("rm: refusing to remove '.' or '..' directory: skipping '%s'", path);
714 continue;
715 }
716 if (dot) problem = "Invalid argument";
717 else {
718 Resolution result = resolve(shell, path, false, false);
719 Node *node = result.node;
720 if (!node) {
721 if (!directory_only && opts.flags['f'] && result.failure == MISSING) continue;
722 problem = reason(result);
723 } else if (directory_only && node->kind != DIRECTORY) problem = "Not a directory";
724 else if (!directory_only && node->kind == DIRECTORY && !opts.flags['r']) problem = "Is a directory";
725 else if (descendant(shell, shell->current, node)) problem = "Device or resource busy";
726 else if (directory_only && node->data.children.count) problem = "Directory not empty";
727 else remove_node(shell, node);
728 }
729 if (problem) {
730 if (directory_only) error("rmdir: failed to remove '%s': %s", path, problem);
731 else error("rm: cannot remove '%s': %s", path, problem);
732 }
733 }
734 } else if (strcmp(command, "mv") == 0 || strcmp(command, "cp") == 0) {
735 const char *destination = opts.paths[opts.count - 1];
736 if (opts.count > 2) {
737 Resolution result = resolve(shell, destination, true, false);
738 if (!result.node || result.node->kind != DIRECTORY) {
739 error("%s: target '%s' is not a directory", command, destination);
740 return;
741 }
742 }
743 for (int i = 0; i + 1 < opts.count; i++) {
744 if (strcmp(command, "mv") == 0) move_one(shell, opts.paths[i], destination);
745 else copy_one(shell, opts.paths[i], destination, opts.flags['r']);
746 }
747 } else if (strcmp(command, "ln") == 0) {
748 const char *path = arguments[2];
749 Resolution result = resolve(shell, path, false, false);
750 if (result.creation_parent) create_node(shell, result.creation_parent, result.creation_name, LINK, arguments[1]);
751 else error("ln: failed to create symbolic link '%s': %s", path, result.node ? "File exists" : reason(result));
752 } else if (strcmp(command, "readlink") == 0 || strcmp(command, "realpath") == 0) {
753 bool canonical = strcmp(command, "realpath") == 0;
754 Resolution result = resolve(shell, arguments[0], canonical, false);
755 if (!result.node) error("%s: %s: %s", command, arguments[0], reason(result));
756 else if (canonical) {
757 char *path = absolute(shell, result.node);
758 emit(output, "%s", path);
759 free(path);
760 } else if (result.node->kind == LINK) emit(output, "%s", result.node->data.target);
761 else error("readlink: %s: Invalid argument", arguments[0]);
762 } else if (strcmp(command, "find") == 0) find_nodes(shell, arguments, count, output);
763 else if (strcmp(command, "du") == 0) disk_usage(shell, opts, output);
764 else if (strcmp(command, "tree") == 0) tree(shell, opts, output);
765 else if (strcmp(command, "cat") == 0 || strcmp(command, "wc") == 0 || strcmp(command, "head") == 0 ||
766 strcmp(command, "tail") == 0 || strcmp(command, "grep") == 0 || strcmp(command, "sort") == 0 || strcmp(command, "uniq") == 0)
767 text_command(shell, command, arguments, count, input, output);
768 else error("%s: command not found", command);
769}
770static void process(Shell *shell, char *line) {
771 shell->clock++;
772 char *path = absolute(shell, shell->current);
773 printf("%s$ %s\n", path, line);
774 free(path);
775 char *redirect = strstr(line, " >> ");
776 bool append = redirect != NULL;
777 if (!redirect) redirect = strstr(line, " > ");
778 Node *redirect_file = NULL;
779 if (redirect) {
780 const char *destination = redirect + (append ? 4 : 3);
781 *redirect = '\0';
782 Resolution result = resolve(shell, destination, true, false);
783 if (result.creation_parent) redirect_file = create_node(shell, result.creation_parent, result.creation_name, FILE_NODE, NULL);
784 else if (!result.node) { error("sh: %s: %s", destination, reason(result)); return; }
785 else if (result.node->kind != FILE_NODE) { error("sh: %s: Is a directory", destination); return; }
786 else {
787 redirect_file = result.node;
788 if (!append) { Lines empty = {0}; write_file(shell, redirect_file, &empty, false); }
789 }
790 }
791 char *commands[5];
792 int count = 0;
793 char *command = line;
794 for (;;) {
795 commands[count++] = command;
796 char *separator = strstr(command, " | ");
797 if (!separator) break;
798 *separator = '\0';
799 command = separator + 3;
800 }
801 Lines input = {0};
802 for (int i = 0; i < count; i++) {
803 Output output = {.terminal = i + 1 == count && !redirect_file};
804 execute(shell, commands[i], &input, &output);
805 clear_lines(&input);
806 input = output.lines;
807 }
808 if (redirect_file && input.count) write_file(shell, redirect_file, &input, true);
809 clear_lines(&input);
810 for (size_t i = 0; i < shell->retired.count; i++) free_node(shell->retired.items[i]);
811 shell->retired.count = 0;
812}
813int main(void) {
814 Node *root = allocate(sizeof(*root));
815 *root = (Node){.kind = DIRECTORY, .inode = 1, .parent = root, .name = copy_text("")};
816 Shell shell = {.root = root, .current = root, .next_inode = 2};
817 char line[202];
818 while (fgets(line, sizeof(line), stdin)) {
819 line[strcspn(line, "\r\n")] = '\0';
820 if (strcmp(line, "exit") == 0) break;
821 process(&shell, line);
822 }
823 free_node(root);
824 free(shell.previous);
825 free(shell.retired.items);
826 return 0;
827}
