432 previous[0] = true

799void process_line(FileSystem &source, string line)
800 source.clock += 1
801 terminal(absolute(source, source.current) + "$ " + line)
802 marker := -1
803 append := false
804 match line.find(" >> ")
805 nat position
806 marker = int(position)
807 append = true
808 none
809 match line.find(" > ")
810 nat position
811 marker = int(position)
812 none
813 marker = -1
814 pipeline := line
815 redirect := -1
816 if marker != -1
817 pipeline = line[0:<marker]
818 path := line[marker + (if append then 4 else 3):]
819 result := resolve(&source, path)
820 if result.parent != -1
821 redirect = create_node(&source, result.parent, result.name, NodeData.File([]))
822 elif result.node == -1
823 terminal("sh: {path}: {result.problem}")
824 return
825 elif get_node(source, result.node).directory()
826 terminal("sh: {path}: Is a directory")
827 return
828 else
829 redirect = result.node
830 if not append
831 write_file(&source, redirect, [], false)
832 commands := pipeline.split(" | ")
833 string[] input := []
834 for position := commands.indices()[]
835 output := Output(position + 1 == #commands and redirect == -1)
836 execute(&source, commands[position], input, &output)
837 input = output.lines
838 if redirect != -1 and #input > 0
839 write_file(&source, redirect, input, true)
840 reclaim(&source)
