4abstract class Node
5{
6 public DirectoryNode $parent;
7 public string $name = '';
8 public function __construct(public readonly int $inode, public int $modified) {}
9 abstract public function size(): int;
10 abstract public function tag(): string;
11 public function links(): int { return 1; }

14final class DirectoryNode extends Node
15{
16 public array $entries = [];
17 public function size(): int { return count($this->entries); }
18 public function tag(): string { return 'd'; }
19 public function links(): int
20 {
21 return 2 + count(array_filter($this->entries, fn(Node $node): bool => $node instanceof self));
22 }
23 public function children(bool $hidden = true): array
24 {
25 $children = array_values(array_filter($this->entries,
26 fn(Node $node): bool => $hidden || $node->name[0] !== '.'));
27 usort($children, fn(Node $a, Node $b): int => strcmp($a->name, $b->name));
28 return $children;
29 }
30}

76function options(array $args): array
77{
78 $flags = [];
79 $at = 0;
80 while ($at < count($args) && str_starts_with($args[$at], '-')) {
81 foreach (str_split(substr($args[$at], 1)) as $flag) $flags[$flag] = true;
82 $at++;
83 }
84 return [$flags, array_slice($args, $at)];

86function lastComponent(string $path): string
87{
88 $at = strrpos($path, '/');
89 return $at === false ? $path : substr($path, $at + 1);

159 private function resolve(string $path, bool $follow = true, bool $parents = false): Resolution
160 {
161 $expansions = 0;
162 $walk = function(string $path, DirectoryNode $base, bool $follow, bool $original)
163 use (&$walk, &$expansions, $parents): Resolution {
164 $node = str_starts_with($path, '/') ? $this->root : $base;
165 $parts = array_values(array_filter(explode('/', $path), fn(string $part): bool => $part !== ''));
166 foreach ($parts as $i => $part) {
167 if (!$node instanceof DirectoryNode) return new Resolution(error: Failure::NotDirectory);
168 if ($part === '.') continue;
169 if ($part === '..') {
170 $node = $node->parent;
171 continue;
172 }
173 $last = $i + 1 === count($parts);
174 $child = $node->entries[':' . $part] ?? null;
175 if ($child === null) {
176 if ($original && $last) {
177 return new Resolution(error: Failure::Missing, creationParent: $node, creationName: $part);
178 }
179 if ($original && $parents) $child = $this->makeDirectory($node, $part);
180 else return new Resolution(error: Failure::Missing);
181 }
182 if ($child instanceof LinkNode && (!$last || $follow)) {
183 if (++$expansions > 8) return new Resolution(error: Failure::Loop);
184 $result = $walk($child->target, $node, true, false);
185 if ($result->node === null) return $result;
186 $child = $result->node;
187 }
188 $node = $child;
189 }
190 return new Resolution($node);
191 };
192 return $walk($path, $this->current, $follow, true);
193 }
194 private function cloneNode(Node $source, DirectoryNode $parent, string $name): void
195 {
196 if ($source instanceof DirectoryNode) {
197 $directory = $this->makeDirectory($parent, $name);
198 foreach ($source->children() as $child) $this->cloneNode($child, $directory, $child->name);
199 } elseif ($source instanceof FileNode) {
200 $this->makeFile($parent, $name, $source->lines);
201 } else {
202 $this->add($parent, $name, new LinkNode($this->nextInode++, $this->clock, $source->target));
203 }

205 private function move(string $sourcePath, string $destination): void
206 {
207 $problem = fn(string $path, string $reason) => $this->error("mv: cannot move '$sourcePath' to '$path': $reason");
208 if (invalidSource($sourcePath)) { $problem($destination, 'Invalid argument'); return; }
209 $sourceResult = $this->resolve($sourcePath, false);
210 $source = $sourceResult->node;
211 if ($source === null) {
212 $this->error("mv: cannot stat '$sourcePath': " . $sourceResult->error->value);
213 return;
214 }
215 $destinationNode = $this->resolve($destination)->node;
216 if ($destinationNode instanceof DirectoryNode) {
217 $parent = $destinationNode;
218 $name = lastComponent($sourcePath);
219 $display = joinPath($destination, $name);
220 } else {
221 $result = $this->resolve($destination, false);
222 if ($result->absent()) {
223 $parent = $result->creationParent;
224 $name = $result->creationName;
225 } elseif ($result->node !== null) {
226 $parent = $result->node->parent;
227 $name = $result->node->name;
228 } else { $problem($destination, $result->error->value); return; }
229 $display = $destination;
230 }
231 $existing = $parent->entries[':' . $name] ?? null;
232 if ($source instanceof DirectoryNode && $this->descendant($parent, $source)) {
233 $this->error("mv: cannot move '$sourcePath' to a subdirectory of itself, '$display'");
234 } elseif ($existing === $source) {
235 $this->error("mv: '$sourcePath' and '$display' are the same file");
236 } elseif ($existing !== null && $source instanceof DirectoryNode && !$existing instanceof DirectoryNode) {
237 $this->error("mv: cannot overwrite non-directory '$display' with directory '$sourcePath'");
238 } elseif (!$source instanceof DirectoryNode && $existing instanceof DirectoryNode) {
239 $this->error("mv: cannot overwrite directory '$display' with non-directory '$sourcePath'");
240 } elseif ($existing instanceof DirectoryNode && $existing->entries !== []) {
241 $problem($display, 'Directory not empty');
242 } elseif ($existing === $this->current) {
243 $problem($display, 'Device or resource busy');
244 } else {
245 if ($existing !== null) $this->detach($existing);
246 $this->detach($source);
247 $this->add($parent, $name, $source);
248 }

250 private function copyNode(string $sourcePath, string $destination, bool $recursive): void
251 {
252 $problem = fn(string $path, string $reason) => $this->error("cp: cannot copy '$sourcePath' to '$path': $reason");
253 if (invalidSource($sourcePath)) { $problem($destination, 'Invalid argument'); return; }
254 $sourceResult = $this->resolve($sourcePath);
255 $source = $sourceResult->node;
256 if ($source === null) {
257 $this->error("cp: cannot stat '$sourcePath': " . $sourceResult->error->value);
258 return;
259 }
260 if ($source instanceof DirectoryNode && !$recursive) {
261 $this->error("cp: -r not specified; omitting directory '$sourcePath'");
262 return;
263 }
264 $result = $this->resolve($destination);
265 if ($result->node === null && !$result->absent()) {
266 $problem($destination, $result->error->value);
267 return;
268 }
269 $display = $result->node instanceof DirectoryNode ? joinPath($destination, lastComponent($sourcePath)) : $destination;
270 $target = $this->resolve($display, !$source instanceof DirectoryNode);
271 if ($source instanceof DirectoryNode) {
272 if ($target->node !== null) $problem($display, 'File exists');
273 elseif ($this->descendant($target->creationParent, $source)) {
274 $this->error("cp: cannot copy a directory, '$sourcePath', into itself, '$display'");
275 } else $this->cloneNode($source, $target->creationParent, $target->creationName);
276 } elseif ($target->absent()) {
277 $this->makeFile($target->creationParent, $target->creationName, $source->lines);
278 } elseif ($target->node === $source) {
279 $this->error("cp: '$sourcePath' and '$display' are the same file");
280 } elseif ($target->node instanceof DirectoryNode) {
281 $this->error("cp: cannot overwrite directory '$display' with non-directory '$sourcePath'");
282 } elseif ($target->node instanceof FileNode) {
283 $this->write($target->node, $source->lines);
284 } else $problem($display, $target->error->value);
285 }
286 private function readInputs(string $command, array $paths, array $input, Closure $consume): void

288 if ($paths === []) { $consume($input, null); return; }

292 elseif ($result->node instanceof DirectoryNode) $this->error("$command: $path: Is a directory");
293 else $consume($result->node->lines, $path);
294 }

296 private function walk(Node $node, string $path, Closure $visit, ?int $maxDepth = null, int $depth = 0): void
297 {
298 $visit($node, $path, $depth);
299 if ($node instanceof DirectoryNode && ($maxDepth === null || $depth < $maxDepth)) {
300 foreach ($node->children() as $child) $this->walk($child, joinPath($path, $child->name), $visit, $maxDepth, $depth + 1);
301 }

303 private function listNodes(array $args, Closure $output): void
304 {
305 [$flags, $paths] = options($args);
306 if ($paths === []) $paths = ['.'];
307 foreach ($paths as $path) {
308 $result = $this->resolve($path, !isset($flags['l']));
309 $node = $result->node;
310 if ($node === null) {
311 $this->error("ls: cannot access '$path': " . $result->error->value);
312 continue;
313 }
314 $entries = [];
315 if ($node instanceof DirectoryNode) {
316 if (count($paths) >= 2) $output("$path:");
317 if (isset($flags['a'])) $entries = [['.', $node], ['..', $node->parent]];
318 foreach ($node->children(isset($flags['a'])) as $child) $entries[] = [$child->name, $child];
319 } else $entries[] = [$path, $node];
320 if (isset($flags['t'])) usort($entries, fn(array $a, array $b): int => $b[1]->modified <=> $a[1]->modified);
321 if (isset($flags['r'])) $entries = array_reverse($entries);
322 $widths = [0, 0, 0];
323 foreach ($entries as [, $node]) {
324 foreach ([$node->links(), $node->size(), $node->modified] as $i => $number) {
325 $widths[$i] = max($widths[$i], strlen((string)$number));
326 }
327 }
328 foreach ($entries as [$name, $node]) {
329 if (!isset($flags['l'])) { $output($name); continue; }
330 $line = $node->tag();
331 foreach ([$node->links(), $node->size(), $node->modified] as $i => $number) {
332 $line .= ' ' . str_pad((string)$number, $widths[$i], ' ', STR_PAD_LEFT);
333 }
334 $line .= ' ' . $name;
335 if ($node instanceof LinkNode) $line .= ' -> ' . $node->target;
336 $output($line);
337 }
338 }

342 $at = 0;
343 $path = $args !== [] && !str_starts_with($args[0], '-') ? $args[$at++] : '.';

346 $tests[$args[$at]] = $args[$at + 1];

351 $this->error("find: '$path': " . $result->error->value);

354 $this->walk($result->node, $path, function(Node $node, string $display, int $depth) use ($path, $tests, $output): void {
355 $name = $depth === 0 ? ($path === '/' ? '/' : lastComponent($path)) : $node->name;
356 $kind = $node instanceof FileNode ? 'f' : $node->tag();
357 if ((!isset($tests['-name']) || globMatch($tests['-name'], $name)) &&
358 (!isset($tests['-type']) || $tests['-type'] === $kind)) $output($display);

360 }

382 private function tree(array $args, Closure $output): void
383 {
384 [$flags, $paths] = options($args);
385 $path = $paths[0] ?? '.';
386 $result = $this->resolve($path);
387 if ($result->node === null) { $this->error("tree: $path: " . $result->error->value); return; }
388 if (!$result->node instanceof DirectoryNode) { $this->error("tree: $path: Not a directory"); return; }
389 $directories = 0;
390 $files = 0;
391 $draw = function(DirectoryNode $directory, string $prefix) use (&$draw, &$directories, &$files, $flags, $output): void {
392 $children = $directory->children(isset($flags['a']));
393 foreach ($children as $i => $node) {
394 $last = $i + 1 === count($children);
395 $line = $prefix . ($last ? '`-- ' : '|-- ') . $node->name;
396 if ($node instanceof LinkNode) $line .= ' -> ' . $node->target;
397 $output($line);
398 if ($node instanceof DirectoryNode) {
399 $directories++;
400 $draw($node, $prefix . ($last ? ' ' : '| '));
401 } else $files++;
402 }
403 };
404 $output($path);
405 $draw($result->node, '');
406 $output("$directories " . ($directories === 1 ? 'directory' : 'directories') .
407 ", $files " . ($files === 1 ? 'file' : 'files'));
408 }
409 private function wordCount(array $args, array $input, Closure $output): void
410 {
411 [$flags, $paths] = options($args);
412 if ($flags === []) $flags = ['l' => true, 'w' => true, 'c' => true];
413 $totals = [0, 0, 0];
414 $report = function(array $counts, ?string $name) use ($flags, $output): void {
415 $selected = [];
416 foreach (['l', 'w', 'c'] as $i => $flag) if (isset($flags[$flag])) $selected[] = $counts[$i];
417 $output(implode(' ', $selected) . ($name === null ? '' : ' ' . $name));
418 };
419 $this->readInputs('wc', $paths, $input, function(array $lines, ?string $name) use (&$totals, $report): void {
420 $counts = [count($lines), 0, 0];
421 foreach ($lines as $line) {
422 $inside = false;
423 $length = strlen($line);
424 for ($i = 0; $i < $length; $i++) {
425 if ($line[$i] !== ' ' && !$inside) $counts[1]++;
426 $inside = $line[$i] !== ' ';
427 }
428 $counts[2] += $length + 1;
429 }
430 foreach ($counts as $i => $count) $totals[$i] += $count;
431 $report($counts, $name);
432 });
433 if (count($paths) >= 2) $report($totals, 'total');
434 }

450 private function grep(array $args, array $input, Closure $output): void
451 {
452 [$flags, $rest] = options($args);
453 $pattern = isset($flags['i']) ? strtolower($rest[0]) : $rest[0];
454 $paths = array_slice($rest, 1);
455 $consume = function(array $lines, ?string $name) use ($flags, $pattern, $paths, $output): void {
456 $prefix = count($paths) >= 2 || isset($flags['r']) ? "$name:" : '';
457 $selected = 0;
458 foreach ($lines as $i => $line) {
459 $compared = isset($flags['i']) ? strtolower($line) : $line;
460 if (str_contains($compared, $pattern) === isset($flags['v'])) continue;
461 $selected++;
462 if (!isset($flags['c'])) $output($prefix . (isset($flags['n']) ? ($i + 1) . ':' : '') . $line);
463 }
464 if (isset($flags['c'])) $output($prefix . $selected);
465 };
466 if (!isset($flags['r'])) { $this->readInputs('grep', $paths, $input, $consume); return; }
467 foreach ($paths === [] ? ['.'] : $paths as $path) {
468 $result = $this->resolve($path);
469 if ($result->node === null) $this->error("grep: $path: " . $result->error->value);
470 else $this->walk($result->node, $path, function(Node $node, string $display) use ($consume): void {
471 if ($node instanceof FileNode) $consume($node->lines, $display);
472 });
473 }

500 private function execute(string $text, array $input, Closure $output): void
501 {
502 $words = explode(' ', $text);
503 $command = $words[0];
504 $args = array_slice($words, 1);
505 switch ($command) {
506 case 'pwd': $output($this->absolute($this->current)); break;
507 case 'echo': $output(substr($text, 5)); break;
508 case 'cd':
509 $back = ($args[0] ?? '') === '-';
510 if ($back && $this->previous === null) { $this->error('cd: OLDPWD not set'); break; }
511 $path = $back ? $this->previous : ($args[0] ?? '/');
512 $result = $this->resolve($path);
513 if ($result->node === null) $this->error("cd: $path: " . $result->error->value);
514 elseif (!$result->node instanceof DirectoryNode) $this->error("cd: $path: Not a directory");
515 else {
516 $this->previous = $this->absolute($this->current);
517 $this->current = $result->node;
518 if ($back) $output($this->absolute($this->current));
519 }
520 break;
521 case 'mkdir':
522 [$flags, $paths] = options($args);
523 foreach ($paths as $path) {
524 $result = $this->resolve($path, false, isset($flags['p']));
525 if ($result->absent()) $this->makeDirectory($result->creationParent, $result->creationName);
526 elseif ($result->node === null) $this->error("mkdir: cannot create directory '$path': " . $result->error->value);
527 elseif (!isset($flags['p']) || !$this->resolve($path)->node instanceof DirectoryNode) {
528 $this->error("mkdir: cannot create directory '$path': File exists");
529 }
530 }
531 break;
532 case 'touch':
533 foreach ($args as $path) {
534 $result = $this->resolve($path);
535 if ($result->absent()) $this->makeFile($result->creationParent, $result->creationName);
536 elseif ($result->node !== null) $result->node->modified = $this->clock;
537 else $this->error("touch: cannot touch '$path': " . $result->error->value);
538 }
539 break;
540 case 'cat':
541 $this->readInputs($command, $args, $input, function(array $lines) use ($output): void {
542 foreach ($lines as $line) $output($line);
543 });
544 break;
545 case 'ls': $this->listNodes($args, $output); break;
546 case 'stat':
547 foreach ($args as $path) {
548 $result = $this->resolve($path, false);
549 $node = $result->node;
550 if ($node === null) { $this->error("stat: cannot stat '$path': " . $result->error->value); continue; }
551 $kind = match (true) {
552 $node instanceof DirectoryNode => 'directory',
553 $node instanceof LinkNode => 'symbolic link',
554 $node->lines === [] => 'regular empty file',
555 default => 'regular file',
556 };
557 $output("$path: $kind, size " . $node->size() . ", inode $node->inode, links " .
558 $node->links() . ", modified $node->modified");
559 }
560 break;
561 case 'rm':
562 case 'rmdir':
563 [$flags, $paths] = $command === 'rm' ? options($args) : [[], $args];
564 foreach ($paths as $path) {
565 $problem = fn(string $reason) => $this->error($command === 'rm'
566 ? "rm: cannot remove '$path': $reason" : "rmdir: failed to remove '$path': $reason");
567 if (in_array(lastComponent($path), ['.', '..'], true)) {
568 if ($command === 'rm') $this->error("rm: refusing to remove '.' or '..' directory: skipping '$path'");
569 else $problem('Invalid argument');
570 continue;
571 }
572 $result = $this->resolve($path, false);
573 $node = $result->node;
574 if ($node === null) {
575 if (!($command === 'rm' && isset($flags['f']) && $result->error === Failure::Missing)) {
576 $problem($result->error->value);
577 }
578 } elseif ($command === 'rm' && $node instanceof DirectoryNode && !isset($flags['r'])) {
579 $problem('Is a directory');
580 } elseif ($command === 'rmdir' && !$node instanceof DirectoryNode) {
581 $problem('Not a directory');
582 } elseif ($this->descendant($this->current, $node)) {
583 $problem('Device or resource busy');
584 } elseif ($command === 'rmdir' && $node->entries !== []) {
585 $problem('Directory not empty');
586 } else $this->detach($node);
587 }
588 break;
589 case 'mv':
590 case 'cp':
591 [$flags, $paths] = $command === 'cp' ? options($args) : [[], $args];
592 $destination = array_pop($paths);
593 if (count($paths) >= 2 && !$this->resolve($destination)->node instanceof DirectoryNode) {
594 $this->error("$command: target '$destination' is not a directory");
595 break;
596 }
597 foreach ($paths as $source) {
598 if ($command === 'mv') $this->move($source, $destination);
599 else $this->copyNode($source, $destination, isset($flags['r']));
600 }
601 break;
602 case 'ln':
603 $target = $args[1];
604 $path = $args[2];
605 $result = $this->resolve($path, false);
606 if ($result->absent()) {
607 $this->add($result->creationParent, $result->creationName, new LinkNode($this->nextInode++, $this->clock, $target));
608 } else {
609 $reason = $result->node === null ? $result->error->value : 'File exists';
610 $this->error("ln: failed to create symbolic link '$path': $reason");
611 }
612 break;
613 case 'readlink':
614 case 'realpath':
615 $path = $args[0];
616 $result = $this->resolve($path, $command === 'realpath');
617 if ($result->node === null) $this->error("$command: $path: " . $result->error->value);
618 elseif ($command === 'realpath') $output($this->absolute($result->node));
619 elseif ($result->node instanceof LinkNode) $output($result->node->target);
620 else $this->error("readlink: $path: Invalid argument");
621 break;
622 case 'find': $this->find($args, $output); break;
623 case 'du': $this->diskUsage($args, $output); break;
624 case 'tree': $this->tree($args, $output); break;
625 case 'wc': $this->wordCount($args, $input, $output); break;
626 case 'head':
627 case 'tail': $this->ends($command, $args, $input, $output); break;
628 case 'grep': $this->grep($args, $input, $output); break;
629 case 'sort': $this->sortLines($args, $input, $output); break;
630 case 'uniq': $this->unique($args, $input, $output); break;
631 default: $this->error("$command: command not found");
632 }
633 }
634 public function process(string $line): void
635 {
636 $this->clock++;
637 $this->error($this->absolute($this->current) . '$ ' . $line);
638 $pipeline = $line;
639 $redirect = null;
640 $marker = strpos($line, ' >');
641 if ($marker !== false) {
642 $pipeline = substr($line, 0, $marker);
643 $append = $line[$marker + 2] === '>';
644 $path = substr($line, $marker + ($append ? 4 : 3));
645 $result = $this->resolve($path);
646 if ($result->absent()) $redirect = $this->makeFile($result->creationParent, $result->creationName);
647 elseif ($result->node instanceof FileNode) {
648 $redirect = $result->node;
649 if (!$append) $this->write($redirect, []);
650 } else {
651 $reason = $result->node instanceof DirectoryNode ? 'Is a directory' : $result->error->value;
652 $this->error("sh: $path: $reason");
653 return;
654 }
655 }
656 $input = [];
657 $commands = explode(' | ', $pipeline);
658 foreach ($commands as $i => $command) {
659 $output = [];
660 $emit = $i + 1 === count($commands) && $redirect === null ? $this->terminal :
661 static function(string $text) use (&$output): void { $output[] = $text; };
662 $this->execute($command, $input, $emit);
663 $input = $output;
664 }
665 if ($redirect !== null && $input !== []) $this->write($redirect, $input, true);
666 }
667}

669function globMatch(string $pattern, string $text): bool
670{
671 $previous = array_fill(0, strlen($text) + 1, false);
672 $previous[0] = true;
673 foreach (str_split($pattern) as $character) {
674 $next = array_fill(0, strlen($text) + 1, false);
675 $next[0] = $character === '*' && $previous[0];
676 for ($i = 1; $i <= strlen($text); $i++) {
677 $next[$i] = $character === '*' ? $previous[$i] || $next[$i - 1] :
678 $previous[$i - 1] && ($character === '?' || $character === $text[$i - 1]);
679 }
680 $previous = $next;
681 }
682 return $previous[strlen($text)];
683}
684function numericPrefix(string $line): int
685{
686 $negative = str_starts_with($line, '-');
687 $at = $negative ? 1 : 0;
688 $value = 0;
689 while ($at < strlen($line) && $line[$at] >= '0' && $line[$at] <= '9') {
690 $value = $value * 10 + ord($line[$at]) - 48;
691 $at++;
692 }
693 return $negative ? -$value : $value;
694}
