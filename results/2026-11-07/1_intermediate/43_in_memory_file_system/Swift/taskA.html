3enum Kind: String { case directory = "d", file = "f", link = "l" }
4enum Content { case directory([String: Node]), file([String]), link(String) }
5final class Node {
6 let inode: Int
7 var modified: Int
8 weak var parent: Node?
9 var name = ""
10 var content: Content
11 init(_ inode: Int, _ modified: Int, _ content: Content) {
12 self.inode = inode
13 self.modified = modified
14 self.content = content
15 }
16 var kind: Kind {
17 switch content { case .directory:
18 return .directory
19 case .file:
20 return .file
21 case .link:
22 return .link }
23 }
24 var entries: [String: Node] {
25 get { guard case .directory(let entries) = content else {
26 preconditionFailure("Expected directory")
27 }
28 return entries
29 }
30 set { content = .directory(newValue) }
31 }
32 var lines: [String] {
33 get { guard case .file(let lines) = content else {
34 preconditionFailure("Expected file")
35 }
36 return lines
37 }
38 set { content = .file(newValue) }
39 }
40 var target: String {
41 guard case .link(let target) = content else {
42 preconditionFailure("Expected link")
43 }
44 return target
45 }
46 var size: Int {
47 switch content {
48 case .directory(let entries):
49 return entries.count
50 case .file(let lines):
51 return lines.reduce(0) { $0 + $1.utf8.count + 1 }
52 case .link(let target):
53 return target.utf8.count
54 }
55 }
56 var linkCount: Int {
57 kind == .directory ? 2 + entries.values.filter { $0.kind == .directory }.count : 1
58 }
59}
60enum Reason: String {
61 case missing = "No such file or directory", notDirectory = "Not a directory", tooManyLinks = "Too many levels of symbolic links"
62}
63enum Resolution {
64 case found(Node), absent(Node, String), failure(Reason)
65 var node: Node? {
66 if case .found(let node) = self {
67 return node
68 }
69 return nil
70 }
71 var creation: (parent: Node, name: String)? {
72 if case .absent(let parent, let name) = self {
73 return (parent, name)
74 }
75 return nil
76 }
77 var reason: String {
78 switch self { case .found:
79 return ""
80 case .absent:
81 return Reason.missing.rawValue
82 case .failure(let reason):
83 return reason.rawValue }
84 }
85}
86struct Options { let flags: Set<Character>
87 let paths: [String]
88}
89func options(_ arguments: [String]) -> Options {
90 var flags = Set<Character>(), position = 0
91 while position < arguments.count && arguments[position].hasPrefix("-") {
92 flags.formUnion(arguments[position].dropFirst())
93 position += 1
94 }
95 return Options(flags: flags, paths: Array(arguments.dropFirst(position)))
96}
97func join(_ path: String, _ name: String) -> String {
98 path == "/" ? "/" + name : path + "/" + name
99}
100func lastComponent(_ path: String)
101 -> String
102{
103 String(path.split(separator: "/", omittingEmptySubsequences: false).last!)
104}
105func invalidSource(_ name: String) -> Bool {
106 name.isEmpty || name == "." || name == ".."
107}
108func lower(_ text: String) -> String {
109 String(
110 decoding: text.utf8.map { 65 <= $0 && $0 <= 90 ? $0 + 32 : $0 },
111 as: UTF8.self
112 )
113}
114func globMatches(_ pattern: String, _ name: String) -> Bool {
115 let bytes = Array(name.utf8)
116 var previous = [Bool](repeating: false, count: bytes.count + 1)
117 previous[0] = true
118 for character in pattern.utf8 {
119 var current = [Bool](repeating: false, count: previous.count)
120 if character == 42 {
121 current[0] = previous[0]
122 for index in bytes.indices {
123 current[index + 1] = previous[index + 1] || current[index]
124 }
125 } else {
126 for index in bytes
127 .indices
128 {
129 current[index + 1] = previous[index] && (character == 63 || character == bytes[index])
130 }
131 }
132 previous = current
133 }
134 return previous.last!
135}

150 private func absolute(_ start: Node) -> String {
151 var node = start, names = [String]()
152 while node !== root {
153 names.append(node.name)
154 node = node.parent!
155 }
156 return "/" + names.reversed().joined(separator: "/")
157 }
158 private func descendant(_ start: Node, of ancestor: Node) -> Bool {
159 var node = start
160 while true {
161 if node === ancestor {
162 return true
163 }
164 if node === root {
165 return false
166 }
167 node = node.parent!
168 }

170 private func add(_ node: Node, to parent: Node, as name: String) {
171 parent.entries[name] = node
172 parent.modified = clock
173 node.parent = parent
174 node.name = name
175 }
176 @discardableResult private func create(_ content: Content, in parent: Node, as name: String) -> Node {
177 let node = Node(nextInode, clock, content)
178 nextInode += 1
179 add(node, to: parent, as: name)
180 return node
181 }
182 private func detach(_ node: Node) {
183 let parent = node.parent!
184 parent.entries.removeValue(forKey: node.name)
185 parent.modified = clock
186 }
187 private func write(_ node: Node, _ lines: [String], append: Bool = false) {
188 node.lines = append ? node.lines + lines : lines
189 node.modified = clock

191 private func resolve(_ path: String, follow: Bool = true, makeParents: Bool = false) -> Resolution {
192 var expansions = 0
193 func walk(_ text: String, from base: Node, finalLink: Bool, original: Bool) -> Resolution {
194 var node = text.hasPrefix("/") ? root : base
195 let components = text.split(separator: "/").map(String.init)
196 for (index, name) in components.enumerated() {
197 guard node.kind == .directory else {
198 return .failure(.notDirectory)
199 }
200 if name == "." {
201 continue
202 }
203 if name == ".." {
204 node = node.parent!
205 continue
206 }
207 var child: Node
208 if let found = node.entries[name] {
209 child = found
210 } else if original && makeParents && index + 1 < components.count {
211 child = create(
212 .directory([:]),
213 in: node,
214 as: name
215 )
216 } else {
217 return original && index + 1 == components.count ? .absent(node, name) : .failure(.missing)
218 }
219 if child.kind == .link && (index + 1 < components.count || finalLink) {
220 expansions += 1
221 if expansions > 8 {
222 return .failure(.tooManyLinks)
223 }
224 let expanded = walk(child.target, from: node, finalLink: true, original: false)
225 guard let target = expanded.node else {
226 return expanded
227 }
228 child = target
229 }
230 node = child
231 }
232 return .found(node)
233 }
234 return walk(path, from: current, finalLink: follow, original: true)
235 }
236 private func clone(_ source: Node, into parent: Node, as name: String) {
237 let content: Content = source.kind == .directory ? .directory([:]) : source.content
238 let copy = create(content, in: parent, as: name)
239 if source.kind == .directory {
240 for name in source.entries.keys.sorted() {
241 clone(source.entries[name]!, into: copy, as: name)
242 }
243 }
244 }
245 private func walkNodes(
246 _ node: Node,
247 _ path: String,
248 depth: Int = 0,
249 limit: Int = 1000,
250 _ visit: (Node, String, Int) -> Void
251 ) {
252 visit(node, path, depth)
253 if node.kind == .directory && depth < limit {
254 for name in node.entries.keys.sorted() {
255 walkNodes(
256 node.entries[name]!,
257 join(path, name),
258 depth: depth + 1,
259 limit: limit,
260 visit
261 )
262 }
263 }

265 private func inputs(
266 _ command: String,
267 _ paths: [String],
268 _ input: [String],
269 _ consume: ([String], String?) -> Void
270 ) {
271 if paths.isEmpty {
272 consume(input, nil)
273 return
274 }
275 for path in paths {
276 let result = resolve(path)
277 guard let node = result.node else {
278 terminal(command + ": " + path + ": " + result.reason)
279 continue
280 }
281 if node.kind == .directory {
282 terminal(command + ": " + path + ": Is a directory")
283 } else {
284 consume(node.lines, path)
285 }

288 private func move(_ paths: [String]) {
289 let destination = paths.last!
290 if paths.count > 2 {
291 guard resolve(destination).node?.kind == .directory else {
292 terminal("mv: target '\(destination)' is not a directory")
293 return
294 }
295 }
296 for sourcePath in paths.dropLast() {
297 let basename = lastComponent(sourcePath)
298 func problem(_ target: String, _ reason: String) {
299 terminal("mv: cannot move '\(sourcePath)' to '\(target)': \(reason)")
300 }
301 if invalidSource(basename) {
302 problem(destination, "Invalid argument")
303 continue
304 }
305 let sourceResult = resolve(sourcePath, follow: false)
306 guard let source = sourceResult.node
307 else {
308 terminal("mv: cannot stat '\(sourcePath)': \(sourceResult.reason)")
309 continue
310 }
311 let parent: Node, name: String, target: String
312 if let directory = resolve(destination).node, directory.kind == .directory {
313 parent = directory
314 name = basename
315 target = join(destination, basename)
316 } else {
317 let result = resolve(destination, follow: false)
318 if let node = result.node {
319 parent = node.parent!
320 name = node.name
321 } else if let creation = result.creation {
322 parent = creation.parent
323 name = creation.name
324 } else {
325 problem(destination, result.reason)
326 continue
327 }
328 target = destination
329 }
330 let existing = parent.entries[name]
331 if source.kind == .directory && descendant(parent, of: source) {
332 terminal("mv: cannot move '\(sourcePath)' to a subdirectory of itself, '\(target)'")
333 } else if existing === source {
334 terminal("mv: '\(sourcePath)' and '\(target)' are the same file")
335 } else if let existing, source.kind == .directory && existing.kind != .directory {
336 terminal("mv: cannot overwrite non-directory '\(target)' with directory '\(sourcePath)'")
337 } else if existing?.kind == .directory && source.kind != .directory {
338 terminal("mv: cannot overwrite directory '\(target)' with non-directory '\(sourcePath)'")
339 } else if let existing, existing.kind == .directory && !existing.entries.isEmpty {
340 problem(target, "Directory not empty")
341 } else if existing === current {
342 problem(target, "Device or resource busy")
343 } else {
344 if let existing {
345 detach(existing)
346 }
347 detach(source)
348 add(source, to: parent, as: name)
349 }
350 }
351 }
352 private func copy(_ parsed: Options) {
353 let destination = parsed.paths.last!
354 if parsed.paths.count > 2 {
355 guard resolve(destination).node?.kind == .directory else {
356 terminal("cp: target '\(destination)' is not a directory")
357 return
358 }
359 }
360 for sourcePath in parsed.paths.dropLast() {
361 let basename = lastComponent(sourcePath)
362 func problem(_ target: String, _ reason: String) {
363 terminal("cp: cannot copy '\(sourcePath)' to '\(target)': \(reason)")
364 }
365 if invalidSource(basename) {
366 problem(destination, "Invalid argument")
367 continue
368 }
369 let sourceResult = resolve(sourcePath)
370 guard let source = sourceResult.node
371 else {
372 terminal("cp: cannot stat '\(sourcePath)': \(sourceResult.reason)")
373 continue
374 }
375 if source.kind == .directory && !parsed.flags.contains("r") {
376 terminal("cp: -r not specified; omitting directory '\(sourcePath)'")
377 continue
378 }
379 let destinationResult = resolve(destination)
380 if destinationResult.node == nil && destinationResult.creation == nil {
381 problem(
382 destination,
383 destinationResult.reason
384 )
385 continue
386 }
387 let target = destinationResult.node?.kind == .directory ? join(destination, basename) : destination
388 let result = resolve(target, follow: source.kind != .directory)
389 if source.kind == .directory {
390 if result.node != nil {
391 problem(target, "File exists")
392 } else if let creation = result.creation {
393 if descendant(creation.parent, of: source) {
394 terminal("cp: cannot copy a directory, '\(sourcePath)', into itself, '\(target)'")
395 } else {
396 clone(source, into: creation.parent, as: creation.name)
397 }
398 } else {
399 problem(target, result.reason)
400 }
401 } else if let creation = result.creation {
402 create(.file(source.lines), in: creation.parent, as: creation.name)
403 } else if let node = result.node {
404 if node === source {
405 terminal("cp: '\(sourcePath)' and '\(target)' are the same file")
406 } else if node
407 .kind ==
408 .directory
409 {
410 terminal("cp: cannot overwrite directory '\(target)' with non-directory '\(sourcePath)'")
411 } else {
412 write(node, source.lines)
413 }
414 } else {
415 problem(target, result.reason)
416 }
417 }

419 private func listing(_ parsed: Options, _ output: (String) -> Void) {
420 let paths = parsed.paths.isEmpty ? ["."] : parsed.paths
421 for path in paths {
422 let result = resolve(path, follow: !parsed.flags.contains("l"))
423 guard let node = result.node else {
424 terminal("ls: cannot access '\(path)': \(result.reason)")
425 continue
426 }
427 var list: [(name: String, node: Node)] = []
428 if node.kind != .directory {
429 list = [(path, node)]
430 } else {
431 if paths.count > 1 {
432 output(path + ":")
433 }
434 if parsed.flags.contains("a") {
435 list = [(".", node), ("..", node.parent!)]
436 }
437 for name in node.entries.keys.sorted()
438 where parsed.flags.contains("a") || !name.hasPrefix(".")
439 {
440 list.append((
441 name,
442 node.entries[name]!
443 ))
444 }
445 if parsed.flags.contains("t") {
446 list = list.enumerated().sorted {
447 $0.element.node.modified != $1.element.node.modified ? $0.element.node.modified > $1.element
448 .node.modified : $0.offset < $1.offset
449 }.map { $0.element }
450 }
451 if parsed.flags.contains("r") {
452 list.reverse()
453 }
454 }
455 func numbers(_ item: Node) -> [String] {
456 [item.linkCount, item.size, item.modified].map { String($0) }
457 }
458 var widths = [0, 0, 0]
459 for entry in list {
460 for (index, value) in numbers(entry.node)
461 .enumerated()
462 {
463 widths[index] = max(widths[index], value.count)
464 }
465 }
466 for (name, item) in list {
467 if !parsed.flags.contains("l") {
468 output(name)
469 continue
470 }
471 var row = item.kind == .file ? "-" : item.kind.rawValue
472 for (index, value) in numbers(item).enumerated() {
473 row += " " + String(
474 repeating: " ",
475 count: widths[index] - value.count
476 ) + value
477 }
478 row += " " + name
479 if item.kind == .link {
480 row += " -> " + item.target
481 }
482 output(row)
483 }
484 }

486 private func find(_ arguments: [String], _ output: (String) -> Void) {
487 var position = arguments.first.map { !$0.hasPrefix("-") } == true ? 1 : 0
488 let path = position == 1 ? arguments[0] : "."
489 var pattern: String?, type: String?, limit = 1000
490 while position < arguments.count {
491 switch arguments[position] {
492 case "-name":
493 pattern = arguments[position + 1]
494 case "-type":
495 type = arguments[position + 1]
496 default:
497 limit = Int(arguments[position + 1])!
498 }
499 position += 2
500 }
501 let result = resolve(path)
502 guard let node = result.node else {
503 terminal("find: '\(path)': \(result.reason)")
504 return
505 }
506 walkNodes(node, path, limit: limit) { item, display, depth in
507 let name = depth == 0 ? (path == "/" ? "/" : lastComponent(path)) : item.name
508 if let pattern, !globMatches(pattern, name) {
509 return
510 }
511 if let type, type != item.kind.rawValue {
512 return
513 }
514 output(display)
515 }
516 }
517 private func diskUsage(_ parsed: Options, _ output: (String) -> Void) {
518 let paths = parsed.paths.isEmpty ? ["."] : parsed.paths
519 func report(_ node: Node, _ path: String, show: Bool) -> Int {
520 var usage = node.kind == .directory ? 0 : node.size
521 if node.kind == .directory {
522 for name in node.entries.keys.sorted() {
523 let child = node.entries[name]!
524 usage += report(
525 child,
526 join(path, name),
527 show: show && !parsed.flags
528 .contains("s") && (child.kind == .directory || parsed.flags.contains("a"))
529 )
530 }
531 }
532 if show {
533 output("\(usage) \(path)")
534 }
535 return usage
536 }
537 for path in paths {
538 let result = resolve(path)
539 if let node = result.node {
540 _ = report(node, path, show: true)
541 } else {
542 terminal("du: cannot access '\(path)': \(result.reason)")
543 }
544 }

546 private func tree(_ parsed: Options, _ output: (String) -> Void) {
547 let path = parsed.paths.first ?? "."
548 let result = resolve(path)
549 guard let node = result.node else {
550 terminal("tree: \(path): \(result.reason)")
551 return
552 }
553 guard node.kind == .directory else {
554 terminal("tree: \(path): Not a directory")
555 return
556 }
557 var directories = 0, files = 0
558 func draw(_ node: Node, _ prefix: String) {
559 let names = node.entries.keys.filter { parsed.flags.contains("a") || !$0.hasPrefix(".") }.sorted()
560 for (index, name) in names.enumerated() {
561 let child = node.entries[name]!, last = index + 1 == names.count
562 output(prefix + (last ? "`-- " : "|-- ") + name + (child.kind == .link ? " -> " + child.target : ""))
563 if child.kind == .directory {
564 directories += 1
565 draw(child, prefix + (last ? " " : "| "))
566 } else {
567 files += 1
568 }
569 }
570 }
571 output(path)
572 draw(node, "")
573 output("\(directories) " + (directories == 1 ? "directory" : "directories") + ", \(files) " +
574 (files == 1 ? "file" : "files"))
575 }
576 private func execute(_ text: String, _ input: [String], _ output: (String) -> Void) {
577 let words = text.split(separator: " ").map(String.init)
578 let command = words[0], arguments = Array(words.dropFirst()), parsed = options(arguments)
579 switch command {
580 case "pwd":
581 output(absolute(current))
582 case "echo":
583 output(String(text.dropFirst(5)))
584 case "cd":
585 var path = arguments.first ?? "/"
586 let old = path == "-"
587 if old {
588 guard let previous else {
589 terminal("cd: OLDPWD not set")
590 return
591 }
592 path = previous
593 }
594 let result = resolve(path)
595 if let node = result.node {
596 if node.kind != .directory {
597 terminal("cd: \(path): Not a directory")
598 } else {
599 previous = absolute(current)
600 current = node
601 if old {
602 output(absolute(current))
603 }
604 }
605 } else {
606 terminal("cd: \(path): \(result.reason)")
607 }
608 case "mkdir":
609 for path in parsed.paths {
610 let result = resolve(path, follow: false, makeParents: parsed.flags.contains("p"))
611 if let creation = result.creation {
612 create(.directory([:]), in: creation.parent, as: creation.name)
613 } else if result.node == nil {
614 terminal("mkdir: cannot create directory '\(path)': \(result.reason)")
615 } else if !parsed.flags.contains("p") || resolve(path).node?
616 .kind != .directory
617 {
618 terminal("mkdir: cannot create directory '\(path)': File exists")
619 }
620 }
621 case "touch":
622 for path in arguments {
623 let result = resolve(path)
624 if let creation = result.creation {
625 create(.file([]), in: creation.parent, as: creation.name)
626 } else if let node = result.node {
627 node.modified = clock
628 } else {
629 terminal("touch: cannot touch '\(path)': \(result.reason)")
630 }
631 }
632 case "ls":
633 listing(parsed, output)
634 case "stat":
635 for path in arguments {
636 let result = resolve(path, follow: false)
637 guard let node = result.node else {
638 terminal("stat: cannot stat '\(path)': \(result.reason)")
639 continue
640 }
641 let type = node.kind == .directory ? "directory" : node.kind == .link ? "symbolic link" : node.lines
642 .isEmpty ? "regular empty file" : "regular file"
643 output(
644 "\(path): \(type), size \(node.size), inode \(node.inode), links \(node.linkCount), modified \(node.modified)"
645 )
646 }
647 case "rm":
648 for path in parsed.paths {
649 func problem(_ reason: String) {
650 terminal("rm: cannot remove '\(path)': \(reason)")
651 }
652 let name = lastComponent(path)
653 if name == "." || name ==
654 ".."
655 {
656 terminal("rm: refusing to remove '.' or '..' directory: skipping '\(path)'")
657 continue
658 }
659 let result = resolve(path, follow: false)
660 guard let node = result.node else {
661 if !(parsed.flags.contains("f") && result.reason == Reason.missing.rawValue) {
662 problem(result.reason)
663 }
664 continue
665 }
666 if node.kind == .directory && !parsed.flags.contains("r") {
667 problem("Is a directory")
668 } else if descendant(current, of: node) {
669 problem("Device or resource busy")
670 } else {
671 detach(node)
672 }
673 }
674 case "rmdir":
675 for path in arguments {
676 func problem(_ reason: String) {
677 terminal("rmdir: failed to remove '\(path)': \(reason)")
678 }
679 let name = lastComponent(path)
680 if name == "." || name == ".." {
681 problem("Invalid argument")
682 continue
683 }
684 let result = resolve(path, follow: false)
685 guard let node = result.node else {
686 problem(result.reason)
687 continue
688 }
689 if node.kind != .directory {
690 problem("Not a directory")
691 } else if descendant(current, of: node) {
692 problem("Device or resource busy")
693 } else if !node.entries.isEmpty {
694 problem("Directory not empty")
695 } else {
696 detach(node)
697 }
698 }
699 case "mv":
700 move(arguments)
701 case "cp":
702 copy(parsed)
703 case "ln":
704 let path = arguments[2], result = resolve(path, follow: false)
705 if let creation = result.creation {
706 create(.link(arguments[1]), in: creation.parent, as: creation.name)
707 } else {
708 terminal("ln: failed to create symbolic link '\(path)': " +
709 (result.node == nil ? result.reason : "File exists"))
710 }
711 case "readlink", "realpath":
712 let path = arguments[0], result = resolve(path, follow: command == "realpath")
713 guard let node = result.node else {
714 terminal("\(command): \(path): \(result.reason)")
715 return
716 }
717 if command == "realpath" {
718 output(absolute(node))
719 } else if node.kind == .link {
720 output(node.target)
721 } else {
722 terminal("readlink: \(path): Invalid argument")
723 }
724 case "find":
725 find(arguments, output)
726 case "du":
727 diskUsage(parsed, output)
728 case "tree":
729 tree(parsed, output)
730 case "cat", "wc", "head", "tail", "grep", "sort", "uniq":
731 textCommand(command, arguments, input, output)
732 default:
733 terminal(command + ": command not found")
734 }
735 }
736 private func numericValue(_ text: String) -> Int {
737 var iterator = text.utf8.makeIterator(), byte = iterator.next()
738 let negative = byte == 45
739 if negative {
740 byte = iterator.next()
741 }
742 var value = 0
743 while let digit = byte, 48 <= digit && digit <= 57 {
744 value = value * 10 + Int(digit - 48)
745 byte = iterator.next()
746 }
747 return negative ? -value : value
748 }
749 private func textCommand(_ command: String, _ arguments: [String], _ input: [String], _ output: (String) -> Void) {
750 let parsed = options(arguments), flags = parsed.flags
751 var paths = parsed.paths
752 if command == "grep" {
753 var pattern = paths.removeFirst()
754 if flags.contains("i") {
755 pattern = lower(pattern)
756 }
757 let named = paths.count > 1 || flags.contains("r")
758 func consume(_ lines: [String], _ name: String?) {
759 let prefix = named ? name! + ":" : ""
760 var count = 0
761 for (index, line) in lines.enumerated() {
762 let comparison = flags.contains("i") ? lower(line) : line
763 if comparison.contains(pattern) != flags.contains("v") {
764 count += 1
765 if !flags.contains("c") {
766 output(prefix + (flags.contains("n") ? "\(index + 1):" : "") + line)
767 }
768 }
769 }
770 if flags.contains("c") {
771 output(prefix + String(count))
772 }
773 }
774 if !flags.contains("r") {
775 inputs(command, paths, input, consume)
776 } else {
777 if paths.isEmpty {
778 paths = ["."]
779 }
780 for path in paths {
781 let result = resolve(path)
782 guard let node = result.node else {
783 terminal("grep: \(path): \(result.reason)")
784 continue
785 }
786 walkNodes(node, path) { item, display, _ in
787 if item.kind == .file {
788 consume(item.lines, display)
789 }
790 }
791 }
