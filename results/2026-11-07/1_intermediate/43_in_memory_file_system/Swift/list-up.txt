enum Resolution {
    case found(Node), absent(Node, String), failure(Reason)
    var node: Node? {
        if case .found(let node) = self {
            return node
        }
        return nil
    }
    var creation: (parent: Node, name: String)? {
        if case .absent(let parent, let name) = self {
            return (parent, name)
        }
        return nil
    var reason: String {
        switch self { case .found:
            return ""
        case .absent:
            return Reason.missing.rawValue
        case .failure(let reason):
            return reason.rawValue }

        func walk(_ text: String, from base: Node, finalLink: Bool, original: Bool) -> Resolution {
            var node = text.hasPrefix("/") ? root : base
            let components = text.split(separator: "/").map(String.init)
            for (index, name) in components.enumerated() {
                guard node.kind == .directory else {
                    return .failure(.notDirectory)
                }
                if name == "." {
                    continue
                }
                if name == ".." {
                    node = node.parent!
                    continue
                }
                var child: Node
                if let found = node.entries[name] {
                    child = found
                } else if original && makeParents && index + 1 < components.count {
                    child = create(
                        .directory([:]),

                } else {

                    let expanded = walk(child.target, from: node, finalLink: true, original: false)
                    guard let target = expanded.node else {
                        return expanded
                    }
                    child = target
                node = child
