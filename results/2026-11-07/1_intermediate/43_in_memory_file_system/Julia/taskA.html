82function absolute(shell::Shell, node::Node)
83 names = String[]
84 while node !== shell.root
85 push!(names, node.name)
86 node = node.parent
87 end
88 "/" * join(reverse!(names), '/')

91 while true
92 node === ancestor && return true
93 node.parent === node && return false
94 node = node.parent

132 expansions = 0
133 function walk(text::AbstractString, base::Directory, follow_final::Bool, original::Bool)
134 node = startswith(text, "/") ? shell.root : base
135 parts = split(text, '/'; keepempty=false)
136 for (index, part) in enumerate(parts)
137 node isa Directory || return failure("Not a directory")
138 part == "." && continue
139 if part == ".."
140 node = node.parent
141 continue
142 end
143 final = index == length(parts)
144 child = get(node.entries, part, nothing)
145 if child === nothing
146 original && final && return absent(node, part)
147 original && parents || return failure("No such file or directory")
148 child = create_directory!(shell, node, part)
149 end
150 if child isa SymbolicLink && (!final || follow_final)
151 expansions += 1
152 expansions > 8 && return failure("Too many levels of symbolic links")
153 result = walk(child.target, node, true, false)
154 result.node === nothing && return result
155 child = result.node
156 end
157 node = child
158 end
159 found(node)
160 end
161 walk(path, shell.current, follow, true)
162end
163function clone_node!(shell::Shell, source::Node, parent::Directory, name::AbstractString)
164 if source isa Directory
165 copied = create_directory!(shell, parent, name)
166 for child in children(source)
167 clone_node!(shell, child, copied, child.name)
168 end
169 elseif source isa RegularFile
170 create_file!(shell, parent, name, source.lines)
171 else
172 create_link!(shell, parent, name, source.target)
173 end

175function move_one!(shell::Shell, path::String, destination::String)
176 problem(target, reason) = println("mv: cannot move '$path' to '$target': $reason")
177 invalid_source(path) && return problem(destination, "Invalid argument")
178 result = resolve(shell, path; follow=false)
179 source = result.node
180 source === nothing && return println("mv: cannot stat '$path': $(result.reason)")
181 destination_node = resolve(shell, destination).node
182 if destination_node isa Directory
183 parent, name = destination_node, last_component(path)
184 target = join_path(destination, name)
185 else
186 result = resolve(shell, destination; follow=false)
187 if result.parent !== nothing
188 parent, name = result.parent, result.name
189 elseif result.node !== nothing
190 parent, name = result.node.parent, result.node.name
191 else
192 return problem(destination, result.reason)
193 end
194 target = destination
195 end
196 existing = get(parent.entries, name, nothing)
197 if source isa Directory && descendant(parent, source)
198 println("mv: cannot move '$path' to a subdirectory of itself, '$target'")
199 elseif existing === source
200 println("mv: '$path' and '$target' are the same file")
201 elseif existing !== nothing && source isa Directory && !(existing isa Directory)
202 println("mv: cannot overwrite non-directory '$target' with directory '$path'")
203 elseif !(source isa Directory) && existing isa Directory
204 println("mv: cannot overwrite directory '$target' with non-directory '$path'")
205 elseif existing isa Directory && !isempty(existing.entries)
206 problem(target, "Directory not empty")
207 elseif existing === shell.current
208 problem(target, "Device or resource busy")
209 else
210 existing !== nothing && detach!(shell, existing)
211 detach!(shell, source)
212 attach!(shell, parent, name, source)
213 end
214end
215function copy_one!(shell::Shell, path::String, destination::String, recursive::Bool)
216 problem(target, reason) = println("cp: cannot copy '$path' to '$target': $reason")
217 invalid_source(path) && return problem(destination, "Invalid argument")
218 result = resolve(shell, path)
219 source = result.node
220 source === nothing && return println("cp: cannot stat '$path': $(result.reason)")
221 source isa Directory && !recursive && return println("cp: -r not specified; omitting directory '$path'")
222 result = resolve(shell, destination)
223 result.node === nothing && result.parent === nothing && return problem(destination, result.reason)
224 target = result.node isa Directory ? join_path(destination, last_component(path)) : destination
225 result = resolve(shell, target; follow=!(source isa Directory))
226 if source isa Directory
227 if result.node !== nothing
228 problem(target, "File exists")
229 elseif descendant(result.parent, source)
230 println("cp: cannot copy a directory, '$path', into itself, '$target'")
231 else
232 clone_node!(shell, source, result.parent, result.name)
233 end
234 elseif result.parent !== nothing
235 create_file!(shell, result.parent, result.name, source.lines)
236 elseif result.node === source
237 println("cp: '$path' and '$target' are the same file")
238 elseif result.node isa Directory
239 println("cp: cannot overwrite directory '$target' with non-directory '$path'")
240 elseif result.node isa RegularFile
241 write_file!(shell, result.node, source.lines)
242 else
243 problem(target, result.reason)
244 end
245end

247function listing(shell::Shell, opts::Options, output::Output)
248 paths = isempty(opts.paths) ? ["."] : opts.paths
249 for path in paths
250 result = resolve(shell, path; follow=!('l' in opts.flags))
251 node = result.node
252 if node === nothing
253 println("ls: cannot access '$path': $(result.reason)")
254 continue
255 end
256 entries = Tuple{String,Node}[]
257 if node isa Directory
258 length(paths) > 1 && emit!(output, "$path:")
259 if 'a' in opts.flags
260 push!(entries, (".", node), ("..", node.parent))
261 end
262 for child in children(node)
263 ('a' in opts.flags || !startswith(child.name, ".")) && push!(entries, (child.name, child))
264 end
265 else
266 push!(entries, (path, node))
267 end
268 't' in opts.flags && sort!(entries; by=entry -> -entry[2].modified)
269 'r' in opts.flags && reverse!(entries)
270 widths = [1, 1, 1]
271 for (_, child) in entries
272 widths = max.(widths, ndigits.([link_count(child), node_size(child), child.modified]))
273 end
274 for (name, child) in entries
275 if 'l' in opts.flags
276 numbers = [link_count(child), node_size(child), child.modified]
277 columns = join([lpad(string(numbers[i]), widths[i]) for i in 1:3], ' ')
278 kind = child isa RegularFile ? "-" : tag(child)
279 suffix = child isa SymbolicLink ? " -> $(child.target)" : ""
280 emit!(output, "$kind $columns $name$suffix")
281 else
282 emit!(output, name)
283 end
284 end
285 end

287function walk_nodes(visit::Function, node::Node, path::AbstractString, depth::Int=0, limit::Int=1000)
288 visit(node, path, depth)
289 if node isa Directory && depth < limit
290 for child in children(node)
291 walk_nodes(visit, child, join_path(path, child.name), depth + 1, limit)
292 end
293 end

295function glob_match(pattern::String, text::String)
296 previous = falses(ncodeunits(text) + 1)
297 previous[1] = true
298 for character in codeunits(pattern)
299 following = falses(length(previous))
300 following[1] = character == UInt8('*') && previous[1]
301 for (i, byte) in enumerate(codeunits(text))
302 following[i + 1] = character == UInt8('*') ? previous[i + 1] || following[i] :
303 previous[i] && (character == UInt8('?') || character == byte)
304 end
305 previous = following
306 end
307 previous[end]
308end
309function find_nodes(shell::Shell, arguments::Vector{String}, output::Output)
310 explicit = !isempty(arguments) && !startswith(arguments[1], "-")
311 path = explicit ? arguments[1] : "."
312 start = explicit ? 2 : 1
313 tests = Dict{String,String}(arguments[i] => arguments[i + 1] for i in start:2:length(arguments))
314 result = resolve(shell, path)
315 result.node === nothing && return println("find: '$path': $(result.reason)")
316 limit = parse(Int, get(tests, "-maxdepth", "1000"))
317 walk_nodes(result.node, path, 0, limit) do node, display, depth
318 name = depth == 0 ? (path == "/" ? "/" : last_component(path)) : node.name
319 if (!haskey(tests, "-name") || glob_match(tests["-name"], name)) &&
320 (!haskey(tests, "-type") || tests["-type"] == tag(node))
321 emit!(output, display)
322 end
323 end

346function draw_tree(node::Directory, prefix::String, all::Bool, output::Output)
347 selected = filter(child -> all || !startswith(child.name, "."), children(node))
348 directories, files = 0, 0
349 for (i, child) in enumerate(selected)
350 last = i == length(selected)
351 suffix = child isa SymbolicLink ? " -> $(child.target)" : ""
352 emit!(output, prefix * (last ? "`-- " : "|-- ") * child.name * suffix)
353 if child isa Directory
354 d, f = draw_tree(child, prefix * (last ? " " : "| "), all, output)
355 directories += 1 + d
356 files += f
357 else
358 files += 1
359 end
360 end
361 directories, files
362end
363function tree(shell::Shell, opts::Options, output::Output)
364 path = isempty(opts.paths) ? "." : opts.paths[1]
365 result = resolve(shell, path)
366 result.node === nothing && return println("tree: $path: $(result.reason)")
367 result.node isa Directory || return println("tree: $path: Not a directory")
368 emit!(output, path)
369 directories, files = draw_tree(result.node, "", 'a' in opts.flags, output)
370 emit!(output, "$directories $(directories == 1 ? "directory" : "directories"), $files $(files == 1 ? "file" : "files")")
371end

389function numeric_prefix(text::String)
390 matched = match(r"^-?[0-9]+", text)
391 matched === nothing ? 0 : parse(Int, matched.match)

393function word_count(text::String)
394 total, inside = 0, false
395 for byte in codeunits(text)
396 byte != 0x20 && !inside && (total += 1)
397 inside = byte != 0x20
398 end
399 total

401function text_command(shell::Shell, command::String, arguments::Vector{String}, input::Vector{String}, output::Output)
402 opts = options(arguments)
403 if command == "grep"
404 insensitive = 'i' in opts.flags
405 pattern = insensitive ? lowercase(opts.paths[1]) : opts.paths[1]
406 paths = opts.paths[2:end]
407 named = 'r' in opts.flags || length(paths) > 1
408 function consume(lines, name)
409 prefix = named ? "$name:" : ""
410 selected = 0
411 for (i, line) in enumerate(lines)
412 compared = insensitive ? lowercase(line) : line
413 occursin(pattern, compared) == ('v' in opts.flags) && continue
414 selected += 1
415 if !('c' in opts.flags)
416 emit!(output, prefix * ('n' in opts.flags ? "$i:" : "") * line)
417 end
418 end
419 'c' in opts.flags && emit!(output, prefix * string(selected))
420 end
421 if !('r' in opts.flags)
422 read_inputs(consume, shell, command, paths, input)
423 else
424 for path in (isempty(paths) ? ["."] : paths)
425 result = resolve(shell, path)
426 if result.node === nothing
427 println("grep: $path: $(result.reason)")
428 else
429 walk_nodes(result.node, path) do node, display, _
430 node isa RegularFile && consume(node.lines, display)
431 end
432 end
433 end
434 end
435 elseif command == "sort"
436 lines = String[]
437 read_inputs(shell, command, opts.paths, input) do part, _
438 append!(lines, part)
439 end
440 if 'n' in opts.flags
441 sort!(lines; by=line -> (numeric_prefix(line), line))
442 else
443 sort!(lines)
444 end
445 'r' in opts.flags && reverse!(lines)
446 foreach(line -> emit!(output, line), lines)
447 elseif command == "wc"
448 all = !any(flag -> flag in opts.flags, "lwc")
449 function report(counts, name)
450 fields = [string(counts[i]) for i in 1:3 if all || "lwc"[i] in opts.flags]
451 emit!(output, join(fields, ' ') * (name === nothing ? "" : " $name"))
452 end
453 totals = zeros(Int, 3)
454 read_inputs(shell, command, opts.paths, input) do lines, name
455 counts = [length(lines), sum(word_count, lines; init=0), sum(line -> ncodeunits(line) + 1, lines; init=0)]
456 totals .+= counts
457 report(counts, name)
458 end
459 length(opts.paths) > 1 && report(totals, "total")
460 elseif command in ("head", "tail")
461 explicit = !isempty(arguments) && arguments[1] == "-n"
462 raw = explicit ? arguments[2] : "10"
463 special = startswith(raw, "-") || startswith(raw, "+")
464 amount = parse(Int, special ? raw[2:end] : raw)
465 paths = arguments[(explicit ? 3 : 1):end]
466 read_inputs(shell, command, paths, input) do lines, _
467 start, stop = 1, length(lines)
468 if command == "head"
469 stop = special ? max(0, stop - amount) : min(stop, amount)
470 else
471 start = special ? min(stop + 1, max(1, amount)) : max(1, stop - amount + 1)
472 end
473 for i in start:stop
474 emit!(output, lines[i])
475 end
476 end
477 else
478 read_inputs(shell, command, opts.paths, input) do lines, _
479 if command == "cat"
480 foreach(line -> emit!(output, line), lines)
481 return
482 end
483 start = 1
484 while start <= length(lines)
485 stop = start + 1
486 while stop <= length(lines) && lines[stop] == lines[start]
487 stop += 1
488 end
489 emit!(output, ('c' in opts.flags ? "$(stop - start) " : "") * lines[start])
490 start = stop
491 end
492 end
493 end
494end

496function execute!(shell::Shell, text::AbstractString, input::Vector{String}, output::Output)
497 words = String.(split(text))
498 command, arguments = words[1], words[2:end]
499 opts = options(arguments)
500 if command == "pwd"
501 emit!(output, absolute(shell, shell.current))
502 elseif command == "echo"
503 emit!(output, text[6:end])
504 elseif command == "cd"
505 path = isempty(arguments) ? "/" : arguments[1]
506 back = path == "-"
507 if back
508 shell.previous === nothing && return println("cd: OLDPWD not set")
509 path = shell.previous
510 end
511 result = resolve(shell, path)
512 if result.node === nothing
513 println("cd: $path: $(result.reason)")
514 elseif !(result.node isa Directory)
515 println("cd: $path: Not a directory")
516 else
517 shell.previous = absolute(shell, shell.current)
518 shell.current = result.node
519 back && emit!(output, absolute(shell, shell.current))
520 end
521 elseif command == "mkdir"
522 for path in opts.paths
523 result = resolve(shell, path; follow=false, parents='p' in opts.flags)
524 if result.parent !== nothing
525 create_directory!(shell, result.parent, result.name)
526 elseif result.node === nothing
527 println("mkdir: cannot create directory '$path': $(result.reason)")
528 elseif !('p' in opts.flags) || !(resolve(shell, path).node isa Directory)
529 println("mkdir: cannot create directory '$path': File exists")
530 end
531 end
532 elseif command == "touch"
533 for path in arguments
534 result = resolve(shell, path)
535 if result.parent !== nothing
536 create_file!(shell, result.parent, result.name)
537 elseif result.node !== nothing
538 result.node.modified = shell.clock
539 else
540 println("touch: cannot touch '$path': $(result.reason)")
541 end
542 end
543 elseif command == "ls"
544 listing(shell, opts, output)
545 elseif command == "stat"
546 for path in arguments
547 result = resolve(shell, path; follow=false)
548 node = result.node
549 if node === nothing
550 println("stat: cannot stat '$path': $(result.reason)")
551 continue
552 end
553 kind = node isa Directory ? "directory" : node isa SymbolicLink ? "symbolic link" :
554 isempty(node.lines) ? "regular empty file" : "regular file"
555 emit!(output, "$path: $kind, size $(node_size(node)), inode $(node.inode), links $(link_count(node)), modified $(node.modified)")
556 end
557 elseif command in ("rm", "rmdir")
558 directory_only = command == "rmdir"
559 for path in opts.paths
560 dot = last_component(path) in (".", "..")
561 if dot && !directory_only
562 println("rm: refusing to remove '.' or '..' directory: skipping '$path'")
563 continue
564 end
565 reason = ""
566 if dot
567 reason = "Invalid argument"
568 else
569 result = resolve(shell, path; follow=false)
570 node = result.node
571 if node === nothing
572 !directory_only && 'f' in opts.flags && result.reason == "No such file or directory" && continue
573 reason = result.reason
574 elseif directory_only && !(node isa Directory)
575 reason = "Not a directory"
576 elseif !directory_only && node isa Directory && !('r' in opts.flags)
577 reason = "Is a directory"
578 elseif descendant(shell.current, node)
579 reason = "Device or resource busy"
580 elseif directory_only && !isempty(node.entries)
581 reason = "Directory not empty"
582 else
583 detach!(shell, node)
584 end
585 end
586 if !isempty(reason)
587 println(directory_only ? "rmdir: failed to remove '$path': $reason" : "rm: cannot remove '$path': $reason")
588 end
589 end
590 elseif command in ("mv", "cp")
591 destination = opts.paths[end]
592 if length(opts.paths) > 2 && !(resolve(shell, destination).node isa Directory)
593 return println("$command: target '$destination' is not a directory")
594 end
595 for path in opts.paths[1:end-1]
596 if command == "mv"
597 move_one!(shell, path, destination)
598 else
599 copy_one!(shell, path, destination, 'r' in opts.flags)
600 end
601 end
602 elseif command == "ln"
603 path = arguments[3]
604 result = resolve(shell, path; follow=false)
605 if result.parent !== nothing
606 create_link!(shell, result.parent, result.name, arguments[2])
607 else
608 reason = result.node === nothing ? result.reason : "File exists"
609 println("ln: failed to create symbolic link '$path': $reason")
610 end
611 elseif command in ("readlink", "realpath")
612 path = arguments[1]
613 result = resolve(shell, path; follow=command == "realpath")
614 if result.node === nothing
615 println("$command: $path: $(result.reason)")
616 elseif command == "realpath"
617 emit!(output, absolute(shell, result.node))
618 elseif result.node isa SymbolicLink
619 emit!(output, result.node.target)
620 else
621 println("readlink: $path: Invalid argument")
622 end
623 elseif command == "find"
624 find_nodes(shell, arguments, output)
625 elseif command == "du"
626 disk_usage(shell, opts, output)
627 elseif command == "tree"
628 tree(shell, opts, output)
629 elseif command in ("cat", "wc", "head", "tail", "grep", "sort", "uniq")
630 text_command(shell, command, arguments, input, output)
631 else
632 println("$command: command not found")
633 end
634end
635function process_line!(shell::Shell, line::String)
636 shell.clock += 1
637 println(absolute(shell, shell.current), "\$ ", line)
638 pipeline = line
639 redirect = nothing
640 marker = findfirst(" >", line)
641 if marker !== nothing
642 position = first(marker)
643 pipeline = line[1:position-1]
644 append = startswith(line[position:end], " >> ")
645 path = line[position + (append ? 4 : 3):end]
646 result = resolve(shell, path)
647 if result.parent !== nothing
648 redirect = create_file!(shell, result.parent, result.name)
649 elseif result.node isa RegularFile
650 redirect = result.node
651 !append && write_file!(shell, redirect, String[])
652 else
653 reason = result.node === nothing ? result.reason : "Is a directory"
654 return println("sh: $path: $reason")
655 end
656 end
657 commands = split(pipeline, " | ")
658 input = String[]
659 for (index, text) in enumerate(commands)
660 output = Output(redirect === nothing && index == length(commands))
661 execute!(shell, text, input, output)
662 input = output.lines
663 end
664 redirect !== nothing && !isempty(input) && write_file!(shell, redirect, input; append=true)
665end
666function main()
