4const compare = (left, right) => left < right ? -1 : left > right ? 1 : 0;
5const join = (path, name) => path === '/' ? `/${name}` : `${path}/${name}`;

10class Node {
11 constructor(kind, inode, modified) {
12 this.kind = kind;
13 this.inode = inode;
14 this.modified = modified;
15 this.parent = null;
16 this.name = '';
17 }
18 get links() { return 1; }
19}
20class Directory extends Node {
21 constructor(inode, modified) { super('d', inode, modified); this.entries = new Map(); }
22 get size() { return this.entries.size; }
23 get links() { return 2 + [...this.entries.values()].filter(node => node.kind === 'd').length; }
24}
25class File extends Node {
26 constructor(inode, modified, lines = []) { super('f', inode, modified); this.lines = lines.slice(); }
27 get size() { return this.lines.reduce((sum, line) => sum + line.length + 1, 0); }
28}
29class Link extends Node {
30 constructor(inode, modified, target) { super('l', inode, modified); this.target = target; }
31 get size() { return this.target.length; }

33class ResolutionError extends Error {
34 constructor(reason, parent = null, creationName = null) { super(reason); this.parent = parent; this.creationName = creationName; }
35}
36function options(arguments_) {
37 const flags = new Set();
38 let count = 0;
39 while (count < arguments_.length && arguments_[count].startsWith('-')) {
40 for (const flag of arguments_[count].slice(1)) flags.add(flag);
41 count++;
42 }
43 return { flags, paths: arguments_.slice(count) };
44}
45function entries(node, hidden = true) {
46 return [...node.entries].filter(([name]) => hidden || !name.startsWith('.')).sort(([a], [b]) => compare(a, b));
47}
48function globMatches(pattern, name) {
49 let previous = Array(name.length + 1).fill(false);
50 previous[0] = true;
51 for (const character of pattern) {
52 const current = Array(name.length + 1).fill(false);
53 if (character === '*') {
54 current[0] = previous[0];
55 for (let index = 1; index <= name.length; index++) current[index] = previous[index] || current[index - 1];
56 } else {
57 for (let index = 0; index < name.length; index++) current[index + 1] = previous[index] && (character === '?' || character === name[index]);
58 }
59 previous = current;
60 }
61 return previous[name.length];
62}

64class Shell {
65 constructor(terminal) {
66 this.root = new Directory(1, 0);
67 this.root.parent = this.root;
68 this.cwd = this.root;
69 this.previous = null;
70 this.clock = 0;
71 this.nextInode = 2;
72 this.terminal = terminal;
73 }
74 absolute(node) {
75 const names = [];
76 while (node !== this.root) { names.push(node.name); node = node.parent; }
77 return '/' + names.reverse().join('/');
78 }
79 descendant(node, ancestor) {
80 while (true) {
81 if (node === ancestor) return true;
82 if (node === this.root) return false;
83 node = node.parent;
84 }
85 }
86 add(parent, name, node) {
87 parent.entries.set(name, node);
88 parent.modified = this.clock;
89 node.parent = parent; node.name = name;
90 }
91 create(kind, parent, name, content = null) {
92 const inode = this.nextInode++;
93 const node = kind === 'd' ? new Directory(inode, this.clock)
94 : kind === 'f' ? new File(inode, this.clock, content ?? []) : new Link(inode, this.clock, content);
95 this.add(parent, name, node);
96 return node;
97 }
98 detach(node) { node.parent.entries.delete(node.name); node.parent.modified = this.clock; }
99 write(node, lines, append = false) {
100 node.lines = append ? node.lines.concat(lines) : lines.slice();
101 node.modified = this.clock;
102 }
103 resolve(path, follow = true, makeParents = false) {
104 let expansions = 0;
105 const walk = (text, base, finalLink, original) => {
106 let node = text.startsWith('/') ? this.root : base;
107 const components = text.split('/').filter(Boolean);
108 for (let index = 0; index < components.length; index++) {
109 if (node.kind !== 'd') throw new ResolutionError(notDirectory);
110 const name = components[index];
111 if (name === '.') continue;
112 if (name === '..') { node = node.parent; continue; }
113 let child = node.entries.get(name);
114 if (!child) {
115 if (original && makeParents && index + 1 < components.length) child = this.create('d', node, name);
116 else throw new ResolutionError(missing, original && index + 1 === components.length ? node : null, name);
117 }
118 if (child.kind === 'l' && (index + 1 < components.length || finalLink)) {
119 if (++expansions > 8) throw new ResolutionError('Too many levels of symbolic links');
120 child = walk(child.target, node, true, false);
121 }
122 node = child;
123 }
124 return node;
125 };
126 try { return { node: walk(path, this.cwd, follow, true) }; }
127 catch (error) {
128 if (!(error instanceof ResolutionError)) throw error;
129 return { reason: error.message, parent: error.parent, name: error.creationName };
130 }
131 }
132 clone(source, parent, name) {
133 const copy = this.create(source.kind, parent, name, source.kind === 'f' ? source.lines : source.kind === 'l' ? source.target : null);
134 if (source.kind === 'd') for (const [childName, child] of entries(source)) this.clone(child, copy, childName);
135 return copy;
136 }
137 *walk(node, path, depth = 0, limit = Infinity) {
138 yield { node, path, depth };
139 if (node.kind === 'd' && depth < limit) {
140 for (const [name, child] of entries(node)) yield* this.walk(child, join(path, name), depth + 1, limit);
141 }
142 }
143 *inputs(command, paths, input) {
144 if (paths.length === 0) { yield { lines: input, name: null }; return; }
145 for (const path of paths) {
146 const found = this.resolve(path);
147 if (!found.node) this.terminal(`${command}: ${path}: ${found.reason}`);
148 else if (found.node.kind === 'd') this.terminal(`${command}: ${path}: Is a directory`);
149 else yield { lines: found.node.lines, name: path };
150 }

152 move(paths) {
153 const destination = paths.at(-1), sources = paths.slice(0, -1);
154 if (sources.length > 1 && this.resolve(destination).node?.kind !== 'd') {
155 this.terminal(`mv: target '${destination}' is not a directory`); return;
156 }
157 for (const sourcePath of sources) {
158 const basename = lastComponent(sourcePath);
159 const problem = (target, reason) => this.terminal(`mv: cannot move '${sourcePath}' to '${target}': ${reason}`);
160 if (['', '.', '..'].includes(basename)) { problem(destination, 'Invalid argument'); continue; }
161 const sourceResult = this.resolve(sourcePath, false);
162 if (!sourceResult.node) { this.terminal(`mv: cannot stat '${sourcePath}': ${sourceResult.reason}`); continue; }
163 const source = sourceResult.node, targetResult = this.resolve(destination);
164 let parent, name, target;
165 if (targetResult.node?.kind === 'd') {
166 parent = targetResult.node; name = basename; target = join(destination, name);
167 } else {
168 const result = this.resolve(destination, false);
169 if (result.node) { parent = result.node.parent; name = result.node.name; }
170 else if (result.parent) { parent = result.parent; name = result.name; }
171 else { problem(destination, result.reason); continue; }
172 target = destination;
173 }
174 const existing = parent.entries.get(name);
175 if (source.kind === 'd' && this.descendant(parent, source)) {
176 this.terminal(`mv: cannot move '${sourcePath}' to a subdirectory of itself, '${target}'`);
177 } else if (existing === source) {
178 this.terminal(`mv: '${sourcePath}' and '${target}' are the same file`);
179 } else if (existing && source.kind === 'd' && existing.kind !== 'd') {
180 this.terminal(`mv: cannot overwrite non-directory '${target}' with directory '${sourcePath}'`);
181 } else if (existing?.kind === 'd' && source.kind !== 'd') {
182 this.terminal(`mv: cannot overwrite directory '${target}' with non-directory '${sourcePath}'`);
183 } else if (existing?.kind === 'd' && existing.entries.size > 0) {
184 problem(target, 'Directory not empty');
185 } else if (existing === this.cwd) {
186 problem(target, 'Device or resource busy');
187 } else {
188 if (existing) this.detach(existing);
189 this.detach(source); this.add(parent, name, source);
190 }
191 }
192 }
193 copy(flags, paths) {
194 const destination = paths.at(-1), sources = paths.slice(0, -1);
195 if (sources.length > 1 && this.resolve(destination).node?.kind !== 'd') {
196 this.terminal(`cp: target '${destination}' is not a directory`); return;
197 }
198 for (const sourcePath of sources) {
199 const basename = lastComponent(sourcePath);
200 const problem = (target, reason) => this.terminal(`cp: cannot copy '${sourcePath}' to '${target}': ${reason}`);
201 if (['', '.', '..'].includes(basename)) { problem(destination, 'Invalid argument'); continue; }
202 const sourceResult = this.resolve(sourcePath);
203 if (!sourceResult.node) { this.terminal(`cp: cannot stat '${sourcePath}': ${sourceResult.reason}`); continue; }
204 const source = sourceResult.node;
205 if (source.kind === 'd' && !flags.has('r')) { this.terminal(`cp: -r not specified; omitting directory '${sourcePath}'`); continue; }
206 const destinationResult = this.resolve(destination);
207 if (!destinationResult.node && !destinationResult.parent) { problem(destination, destinationResult.reason); continue; }
208 const target = destinationResult.node?.kind === 'd' ? join(destination, basename) : destination;
209 const result = this.resolve(target, source.kind !== 'd');
210 if (source.kind === 'd') {
211 if (result.node) problem(target, 'File exists');
212 else if (!result.parent) problem(target, result.reason);
213 else if (this.descendant(result.parent, source)) this.terminal(`cp: cannot copy a directory, '${sourcePath}', into itself, '${target}'`);
214 else this.clone(source, result.parent, result.name);
215 } else {
216 if (result.parent) this.create('f', result.parent, result.name, source.lines);
217 else if (!result.node) problem(target, result.reason);
218 else if (result.node === source) this.terminal(`cp: '${sourcePath}' and '${target}' are the same file`);
219 else if (result.node.kind === 'd') this.terminal(`cp: cannot overwrite directory '${target}' with non-directory '${sourcePath}'`);
220 else this.write(result.node, source.lines);
221 }
222 }

224 listing(flags, paths, output) {
225 if (paths.length === 0) paths = ['.'];
226 for (const path of paths) {
227 const result = this.resolve(path, !flags.has('l'));
228 if (!result.node) { this.terminal(`ls: cannot access '${path}': ${result.reason}`); continue; }
229 const node = result.node;
230 let list;
231 if (node.kind !== 'd') list = [[path, node]];
232 else {
233 if (paths.length > 1) output(`${path}:`);
234 list = entries(node, flags.has('a'));
235 if (flags.has('a')) list.unshift(['.', node], ['..', node.parent]);
236 if (flags.has('t')) list.sort((left, right) => right[1].modified - left[1].modified);
237 if (flags.has('r')) list.reverse();
238 }
239 const widths = ['links', 'size', 'modified'].map(key => Math.max(0, ...list.map(([, item]) => String(item[key]).length)));
240 for (const [name, item] of list) {
241 if (!flags.has('l')) output(name);
242 else {
243 const values = ['links', 'size', 'modified'].map((key, index) => String(item[key]).padStart(widths[index]));
244 output(`${item.kind === 'f' ? '-' : item.kind} ${values.join(' ')} ${name}` + (item.kind === 'l' ? ` -> ${item.target}` : ''));
245 }
246 }
247 }

249 execute(text, input, output) {
250 const space = text.indexOf(' ');
251 const command = space < 0 ? text : text.slice(0, space);
252 const arguments_ = space < 0 ? [] : text.slice(space + 1).split(' ');
253 const { flags, paths } = options(arguments_);
254 switch (command) {
255 case 'pwd': output(this.absolute(this.cwd)); break;
256 case 'echo': output(text.slice(space + 1)); break;
257 case 'cd': {
258 const previous = arguments_[0] === '-';
259 if (previous && this.previous === null) { this.terminal('cd: OLDPWD not set'); break; }
260 const path = previous ? this.previous : arguments_[0] ?? '/';
261 const result = this.resolve(path);
262 if (!result.node) this.terminal(`cd: ${path}: ${result.reason}`);
263 else if (result.node.kind !== 'd') this.terminal(`cd: ${path}: Not a directory`);
264 else {
265 this.previous = this.absolute(this.cwd); this.cwd = result.node;
266 if (previous) output(this.absolute(this.cwd));
267 }
268 break;
269 }
270 case 'mkdir':
271 for (const path of paths) {
272 const result = this.resolve(path, false, flags.has('p'));
273 if (result.parent) this.create('d', result.parent, result.name);
274 else if (!result.node) this.terminal(`mkdir: cannot create directory '${path}': ${result.reason}`);
275 else if (!flags.has('p') || this.resolve(path).node?.kind !== 'd') this.terminal(`mkdir: cannot create directory '${path}': File exists`);
276 }
277 break;
278 case 'touch':
279 for (const path of arguments_) {
280 const result = this.resolve(path);
281 if (result.parent) this.create('f', result.parent, result.name);
282 else if (result.node) result.node.modified = this.clock;
283 else this.terminal(`touch: cannot touch '${path}': ${result.reason}`);
284 }
285 break;
286 case 'ls': this.listing(flags, paths, output); break;
287 case 'stat':
288 for (const path of arguments_) {
289 const result = this.resolve(path, false), node = result.node;
290 if (!node) this.terminal(`stat: cannot stat '${path}': ${result.reason}`);
291 else {
292 const kind = node.kind === 'd' ? 'directory' : node.kind === 'l' ? 'symbolic link'
293 : node.lines.length === 0 ? 'regular empty file' : 'regular file';
294 output(`${path}: ${kind}, size ${node.size}, inode ${node.inode}, links ${node.links}, modified ${node.modified}`);
295 }
296 }
297 break;
298 case 'rm':
299 for (const path of paths) {
300 const problem = reason => this.terminal(`rm: cannot remove '${path}': ${reason}`);
301 if (['.', '..'].includes(lastComponent(path))) {
302 this.terminal(`rm: refusing to remove '.' or '..' directory: skipping '${path}'`); continue;
303 }
304 const result = this.resolve(path, false), node = result.node;
305 if (!node) { if (!(flags.has('f') && result.reason === missing)) problem(result.reason); }
306 else if (node.kind === 'd' && !flags.has('r')) problem('Is a directory');
307 else if (this.descendant(this.cwd, node)) problem('Device or resource busy');
308 else this.detach(node);
309 }
310 break;
311 case 'rmdir':
312 for (const path of arguments_) {
313 const problem = reason => this.terminal(`rmdir: failed to remove '${path}': ${reason}`);
314 if (['.', '..'].includes(lastComponent(path))) { problem('Invalid argument'); continue; }
315 const result = this.resolve(path, false), node = result.node;
316 if (!node) problem(result.reason);
317 else if (node.kind !== 'd') problem(notDirectory);
318 else if (this.descendant(this.cwd, node)) problem('Device or resource busy');
319 else if (node.entries.size > 0) problem('Directory not empty');
320 else this.detach(node);
321 }
322 break;
323 case 'mv': this.move(arguments_); break;
324 case 'cp': this.copy(flags, paths); break;
325 case 'ln': {
326 const [target, path] = arguments_.slice(1), result = this.resolve(path, false);
327 if (result.parent) this.create('l', result.parent, result.name, target);
328 else this.terminal(`ln: failed to create symbolic link '${path}': ${result.node ? 'File exists' : result.reason}`);
329 break;
330 }
331 case 'readlink':
332 case 'realpath': {
333 const path = arguments_[0], result = this.resolve(path, command === 'realpath');
334 if (!result.node) this.terminal(`${command}: ${path}: ${result.reason}`);
335 else if (command === 'realpath') output(this.absolute(result.node));
336 else if (result.node.kind === 'l') output(result.node.target);
337 else this.terminal(`readlink: ${path}: Invalid argument`);
338 break;
339 }
340 case 'find': this.find(arguments_, output); break;
341 case 'du': this.diskUsage(flags, paths, output); break;
342 case 'tree': this.tree(flags, paths[0] ?? '.', output); break;
343 case 'cat': case 'wc': case 'head': case 'tail': case 'grep': case 'sort': case 'uniq':
344 this.textCommand(command, arguments_, input, output); break;
345 default: this.terminal(`${command}: command not found`);
346 }
347 }
348 find(arguments_, output) {
349 const path = arguments_.length && !arguments_[0].startsWith('-') ? arguments_[0] : '.';
350 const tests = new Map();
351 for (let index = arguments_[0] === path ? 1 : 0; index < arguments_.length; index += 2) tests.set(arguments_[index], arguments_[index + 1]);
352 const result = this.resolve(path);
353 if (!result.node) { this.terminal(`find: '${path}': ${result.reason}`); return; }
354 for (const item of this.walk(result.node, path, 0, tests.has('-maxdepth') ? Number(tests.get('-maxdepth')) : Infinity)) {
355 const name = item.depth === 0 ? (path === '/' ? '/' : lastComponent(path)) : item.node.name;
356 if (tests.has('-name') && !globMatches(tests.get('-name'), name)) continue;
357 if (tests.has('-type') && tests.get('-type') !== item.node.kind) continue;
358 output(item.path);
359 }
360 }
361 diskUsage(flags, paths, output) {
362 if (paths.length === 0) paths = ['.'];
363 const report = (node, path, show) => {
364 let usage = node.kind === 'd' ? 0 : node.size;
365 if (node.kind === 'd') {
366 for (const [name, child] of entries(node)) usage += report(child, join(path, name), show && !flags.has('s') && (child.kind === 'd' || flags.has('a')));
367 }
368 if (show) output(`${usage} ${path}`);
369 return usage;
370 };
371 for (const path of paths) {
372 const result = this.resolve(path);
373 if (!result.node) this.terminal(`du: cannot access '${path}': ${result.reason}`);
374 else report(result.node, path, true);
375 }

377 tree(flags, path, output) {
378 const result = this.resolve(path);
379 if (!result.node) { this.terminal(`tree: ${path}: ${result.reason}`); return; }
380 if (result.node.kind !== 'd') { this.terminal(`tree: ${path}: Not a directory`); return; }
381 let directories = 0, files = 0;
382 const draw = (node, prefix) => {
383 const list = entries(node, flags.has('a'));
384 list.forEach(([name, child], index) => {
385 const last = index + 1 === list.length;
386 output(prefix + (last ? '`-- ' : '|-- ') + name + (child.kind === 'l' ? ` -> ${child.target}` : ''));
387 if (child.kind === 'd') { directories++; draw(child, prefix + (last ? ' ' : '| ')); }
388 else files++;
389 });
390 };
391 output(path); draw(result.node, '');
392 output(`${directories} ${directories === 1 ? 'directory' : 'directories'}, ${files} ${files === 1 ? 'file' : 'files'}`);
393 }
394 *grepInputs(paths, input, recursive) {
395 if (!recursive) { yield* this.inputs('grep', paths, input); return; }
396 if (paths.length === 0) paths = ['.'];
397 for (const path of paths) {
398 const result = this.resolve(path);
399 if (!result.node) { this.terminal(`grep: ${path}: ${result.reason}`); continue; }
400 for (const item of this.walk(result.node, path)) if (item.node.kind === 'f') yield { lines: item.node.lines, name: item.path };

403 textCommand(command, arguments_, input, output) {
404 const parsed = options(arguments_);
405 const flags = parsed.flags;
406 let paths = parsed.paths;
407 if (command === 'grep') {
408 const pattern = flags.has('i') ? paths[0].toLowerCase() : paths[0];
409 paths = paths.slice(1);
410 for (const item of this.grepInputs(paths, input, flags.has('r'))) {
411 const prefix = paths.length > 1 || flags.has('r') ? `${item.name}:` : '';
412 let count = 0;
413 item.lines.forEach((line, index) => {
414 const match = (flags.has('i') ? line.toLowerCase() : line).includes(pattern);
415 if (match !== flags.has('v')) {
416 count++;
417 if (!flags.has('c')) output(prefix + (flags.has('n') ? `${index + 1}:` : '') + line);
418 }
419 });
420 if (flags.has('c')) output(prefix + count);
421 }
422 return;
423 }
424 let countText = '10';
425 if (command === 'head' || command === 'tail') {
426 countText = arguments_[0] === '-n' ? arguments_[1] : '10';
427 paths = arguments_[0] === '-n' ? arguments_.slice(2) : arguments_;
428 }
429 if (command === 'sort') {
430 const lines = [];
431 for (const item of this.inputs(command, paths, input)) {
432 for (const line of item.lines) lines.push(line);
433 }
434 const numeric = line => Number(line.match(/^-?[0-9]+/)?.[0] ?? 0);
435 lines.sort((left, right) => (flags.has('n') ? numeric(left) - numeric(right) : 0) || compare(left, right));
436 if (flags.has('r')) lines.reverse();
437 lines.forEach(output);
438 return;
439 }
440 const totals = [0, 0, 0];
441 const selected = ['l', 'w', 'c'].map(letter => flags.size === 0 || flags.has(letter));
442 const countsLine = (counts, name) => counts.filter((_, index) => selected[index]).join(' ') + (name === null ? '' : ` ${name}`);
443 for (const { lines, name } of this.inputs(command, paths, input)) {
444 if (command === 'cat') lines.forEach(output);
445 else if (command === 'wc') {
446 const counts = [lines.length, lines.reduce((sum, line) => sum + (line.match(/[^ ]+/g)?.length ?? 0), 0), lines.reduce((sum, line) => sum + line.length + 1, 0)];
447 counts.forEach((count, index) => { totals[index] += count; });
448 output(countsLine(counts, name));
449 } else if (command === 'head' || command === 'tail') {
450 const count = Math.abs(Number(countText));
451 let part;
452 if (command === 'head') part = lines.slice(0, countText.startsWith('-') ? Math.max(0, lines.length - count) : count);
453 else part = lines.slice(countText.startsWith('+') ? Math.max(0, count - 1) : Math.max(0, lines.length - count));
454 part.forEach(output);
455 } else {
456 for (let first = 0; first < lines.length;) {
457 let end = first + 1;
458 while (end < lines.length && lines[end] === lines[first]) end++;
459 output((flags.has('c') ? `${end - first} ` : '') + lines[first]);
460 first = end;
461 }
462 }
463 }
464 if (command === 'wc' && paths.length > 1) output(countsLine(totals, 'total'));
465 }

468 this.terminal(`${this.absolute(this.cwd)}$ ${line}`);
469 const redirect = line.match(/ (>>|>) ([^ ]+)$/);

476 else if (result.node.kind === 'd') { this.terminal(`sh: ${path}: Is a directory`); return; }

483 pipeline.forEach((command, index) => {

491 if (redirectFile && input.length) this.write(redirectFile, input, true);
