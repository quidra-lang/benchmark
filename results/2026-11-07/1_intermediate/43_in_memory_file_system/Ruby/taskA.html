4class Node
5 attr_reader :inode
6 attr_accessor :parent, :name, :modified
7
8 def initialize(inode, modified)
9 @inode, @modified = inode, modified
10 @name = ''
11 end

13 def links = 1
14end

16class Directory < Node
17 attr_reader :entries
18
19 def initialize(inode, modified)
20 super
21 @entries = {}
22 end
23
24 def size = @entries.size
25 def tag = 'd'
26 def links = 2 + @entries.values.count { |node| node.is_a?(Directory) }
27 def children = @entries.keys.sort.map { |name| @entries.fetch(name) }
28end

87def options(args)
88 flags = Set.new
89 at = 0
90 while at < args.length && args[at].start_with?('-')
91 args[at][1..].each_char { |flag| flags.add(flag) }
92 at += 1
93 end
94 Options.new(flags.freeze, args.drop(at).freeze)

97def last_component(path) = path[(path.rindex('/') || -1) + 1..]
98def invalid_source?(path) = ['', '.', '..'].include?(last_component(path))
99def join_path(path, name) = path == '/' ? "/#{name}" : "#{path}/#{name}"

101class Shell
102 MISSING = 'No such file or directory'
103 NOT_DIRECTORY = 'Not a directory'
104 LOOP = 'Too many levels of symbolic links'

106 def initialize
107 @root = Directory.new(1, 0)
108 @root.parent = @root
109 @current = @root
110 @previous = nil
111 @clock = 0
112 @next_inode = 2
113 end

115 def absolute(node)
116 parts = []
117 until node.equal?(@root)
118 parts << node.name
119 node = node.parent
120 end
121 '/' + parts.reverse.join('/')
122 end

124 def descendant?(node, ancestor)
125 loop do
126 return true if node.equal?(ancestor)
127 return false if node.equal?(@root)
128 node = node.parent
129 end
130 end

132 def add(parent, name, node)
133 parent.entries[name] = node
134 parent.modified = @clock
135 node.parent = parent
136 node.name = name.freeze
137 end

139 def create(parent, name, type, payload = nil)
140 node = if type == Directory
141 Directory.new(@next_inode, @clock)
142 elsif type == FileNode
143 FileNode.new(@next_inode, @clock, payload || [])
144 else
145 LinkNode.new(@next_inode, @clock, payload)
146 end
147 @next_inode += 1
148 add(parent, name, node)
149 node
150 end
151
152 def detach(node)
153 node.parent.entries.delete(node.name)
154 node.parent.modified = @clock
155 end

157 def write(file, lines, append = false)
158 file.lines = (append ? file.lines + lines : lines.dup).freeze
159 file.modified = @clock
160 end

162 def resolve(path, follow = true, parents = false)
163 expansions = 0
164 walk = lambda do |text, base, follow_final, original|
165 node = text.start_with?('/') ? @root : base
166 parts = text.split('/').reject(&:empty?)
167 parts.each_with_index do |part, i|
168 return Resolution.new(reason: NOT_DIRECTORY) unless node.is_a?(Directory)
169 next if part == '.'
170 if part == '..'
171 node = node.parent
172 next
173 end
174 last = i + 1 == parts.length
175 child = node.entries[part]
176 unless child
177 if original && last
178 return Resolution.new(reason: MISSING, creation_parent: node, creation_name: part)
179 end
180 return Resolution.new(reason: MISSING) unless original && parents
181 child = create(node, part, Directory)
182 end
183 if child.is_a?(LinkNode) && (!last || follow_final)
184 expansions += 1
185 return Resolution.new(reason: LOOP) if expansions > 8
186 result = walk.call(child.target, node, true, false)
187 return result unless result.node
188 child = result.node
189 end
190 node = child
191 end
192 Resolution.new(node:)
193 end
194 walk.call(path, @current, follow, true)
195 end

197 def clone_node(source, parent, name)
198 payload = source.is_a?(FileNode) ? source.lines : (source.target if source.is_a?(LinkNode))
199 copy = create(parent, name, source.class, payload)
200 source.children.each { |child| clone_node(child, copy, child.name) } if source.is_a?(Directory)
201 end

203 def move(source_path, destination, out)
204 problem = ->(path, reason) { out.error("mv: cannot move '%s' to '%s': %s", source_path, path, reason) }
205 if invalid_source?(source_path)
206 problem.call(destination, 'Invalid argument')
207 return
208 end
209 result = resolve(source_path, false)
210 source = result.node
211 unless source
212 out.error("mv: cannot stat '%s': %s", source_path, result.reason)
213 return
214 end
215 destination_node = resolve(destination).node
216 if destination_node.is_a?(Directory)
217 parent, name = destination_node, last_component(source_path)
218 target = join_path(destination, name)
219 else
220 result = resolve(destination, false)
221 if result.absent?
222 parent, name = result.creation_parent, result.creation_name
223 elsif result.node
224 parent, name = result.node.parent, result.node.name
225 else
226 problem.call(destination, result.reason)
227 return
228 end
229 target = destination
230 end
231 existing = parent.entries[name]
232 if source.is_a?(Directory) && descendant?(parent, source)
233 out.error("mv: cannot move '%s' to a subdirectory of itself, '%s'", source_path, target)
234 elsif existing.equal?(source)
235 out.error("mv: '%s' and '%s' are the same file", source_path, target)
236 elsif existing && source.is_a?(Directory) && !existing.is_a?(Directory)
237 out.error("mv: cannot overwrite non-directory '%s' with directory '%s'", target, source_path)
238 elsif !source.is_a?(Directory) && existing.is_a?(Directory)
239 out.error("mv: cannot overwrite directory '%s' with non-directory '%s'", target, source_path)
240 elsif existing.is_a?(Directory) && !existing.entries.empty?
241 problem.call(target, 'Directory not empty')
242 elsif existing.equal?(@current)
243 problem.call(target, 'Device or resource busy')
244 else
245 detach(existing) if existing
246 detach(source)
247 add(parent, name, source)
248 end
249 end

251 def copy(source_path, destination, recursive, out)
252 problem = ->(path, reason) { out.error("cp: cannot copy '%s' to '%s': %s", source_path, path, reason) }
253 if invalid_source?(source_path)
254 problem.call(destination, 'Invalid argument')
255 return
256 end
257 result = resolve(source_path)
258 source = result.node
259 unless source
260 out.error("cp: cannot stat '%s': %s", source_path, result.reason)
261 return
262 end
263 if source.is_a?(Directory) && !recursive
264 out.error("cp: -r not specified; omitting directory '%s'", source_path)
265 return
266 end
267 result = resolve(destination)
268 if !result.node && !result.absent?
269 problem.call(destination, result.reason)
270 return
271 end
272 target = result.node.is_a?(Directory) ? join_path(destination, last_component(source_path)) : destination
273 result = resolve(target, !source.is_a?(Directory))
274 if source.is_a?(Directory)
275 if result.node
276 problem.call(target, 'File exists')
277 elsif descendant?(result.creation_parent, source)
278 out.error("cp: cannot copy a directory, '%s', into itself, '%s'", source_path, target)
279 else
280 clone_node(source, result.creation_parent, result.creation_name)
281 end
282 elsif result.absent?
283 create(result.creation_parent, result.creation_name, FileNode, source.lines)
284 elsif result.node.equal?(source)
285 out.error("cp: '%s' and '%s' are the same file", source_path, target)
286 elsif result.node.is_a?(Directory)
287 out.error("cp: cannot overwrite directory '%s' with non-directory '%s'", target, source_path)
288 elsif result.node.is_a?(FileNode)
289 write(result.node, source.lines)
290 else
291 problem.call(target, result.reason)
292 end
293 end

295 def listing(opts, out)
296 paths = opts.paths.empty? ? ['.'] : opts.paths
297 paths.each do |path|
298 result = resolve(path, !opts.has?('l'))
299 node = result.node
300 unless node
301 out.error("ls: cannot access '%s': %s", path, result.reason)
302 next
303 end
304 if node.is_a?(Directory)
305 out.emit("#{path}:") if paths.size > 1
306 entries = opts.has?('a') ? [Listed.new('.', node), Listed.new('..', node.parent)] : []
307 node.children.each do |child|
308 entries << Listed.new(child.name, child) if opts.has?('a') || !child.name.start_with?('.')
309 end
310 else
311 entries = [Listed.new(path, node)]
312 end
313 entries = entries.each_with_index.sort_by { |entry, i| [-entry.node.modified, i] }.map(&:first) if opts.has?('t')
314 entries.reverse! if opts.has?('r')
315 widths = [1, 1, 1]
316 entries.each do |entry|
317 [entry.node.links, entry.node.size, entry.node.modified].each_with_index do |number, i|
318 widths[i] = [widths[i], number.to_s.length].max
319 end
320 end
321 entries.each do |entry|
322 unless opts.has?('l')
323 out.emit(entry.name)
324 next
325 end
326 item = entry.node
327 numbers = [item.links, item.size, item.modified].each_with_index.map { |number, i| number.to_s.rjust(widths[i]) }.join(' ')
328 suffix = item.is_a?(LinkNode) ? " -> #{item.target}" : ''
329 out.emit("#{item.is_a?(FileNode) ? '-' : item.tag} #{numbers} #{entry.name}#{suffix}")
330 end
331 end
332 end

334 def walk(node, path, depth, limit, &visit)
335 visit.call(node, path, depth)
336 if node.is_a?(Directory) && depth < limit
337 node.children.each { |child| walk(child, join_path(path, child.name), depth + 1, limit, &visit) }
338 end
339 end

341 def find(args, out)
342 explicit = args.first && !args.first.start_with?('-')
343 path = explicit ? args.first : '.'
344 tests = args.drop(explicit ? 1 : 0).each_slice(2).to_h
345 result = resolve(path)
346 unless result.node
347 out.error("find: '%s': %s", path, result.reason)
348 return
349 end
350 walk(result.node, path, 0, tests.fetch('-maxdepth', '1000').to_i) do |node, display, depth|
351 name = depth.zero? ? (path == '/' ? '/' : last_component(path)) : node.name
352 if (!tests.key?('-name') || glob_match?(tests['-name'], name)) && (!tests.key?('-type') || tests['-type'] == node.tag)
353 out.emit(display)
354 end
355 end
356 end

381 def draw_tree(directory, prefix, all, counts, out)
382 children = directory.children.select { |node| all || !node.name.start_with?('.') }
383 children.each_with_index do |child, i|
384 last = i + 1 == children.length
385 suffix = child.is_a?(LinkNode) ? " -> #{child.target}" : ''
386 out.emit(prefix + (last ? '`-- ' : '|-- ') + child.name + suffix)
387 if child.is_a?(Directory)
388 counts[0] += 1
389 draw_tree(child, prefix + (last ? ' ' : '| '), all, counts, out)
390 else
391 counts[1] += 1
392 end
393 end

430 def text_command(command, args, input, out)
431 opts = options(args)
432 case command
433 when 'grep'
434 pattern = opts.has?('i') ? opts.paths.first.downcase : opts.paths.first
435 paths = opts.paths.drop(1)
436 named = opts.has?('r') || paths.size > 1
437 consume = lambda do |part|
438 prefix = named ? "#{part.name}:" : ''
439 selected = 0
440 part.lines.each_with_index do |line, i|
441 comparison = opts.has?('i') ? line.downcase : line
442 next if comparison.include?(pattern) == opts.has?('v')
443 selected += 1
444 out.emit(prefix + (opts.has?('n') ? "#{i + 1}:" : '') + line) unless opts.has?('c')
445 end
446 out.emit(prefix + selected.to_s) if opts.has?('c')
447 end
448 if !opts.has?('r')
449 read_inputs(command, paths, input, out, &consume)
450 else
451 (paths.empty? ? ['.'] : paths).each do |path|
452 result = resolve(path)
453 if result.node
454 walk(result.node, path, 0, 1000) do |node, display, _depth|
455 consume.call(Input.new(node.lines, display)) if node.is_a?(FileNode)
456 end
457 else
458 out.error('grep: %s: %s', path, result.reason)
459 end
460 end
461 end
462 when 'sort'
463 lines = []
464 read_inputs(command, opts.paths, input, out) { |part| lines.concat(part.lines) }
465 lines.sort! do |a, b|
466 comparison = opts.has?('n') ? numeric_prefix(a) <=> numeric_prefix(b) : 0
467 comparison.zero? ? a <=> b : comparison
468 end
469 lines.reverse! if opts.has?('r')
470 lines.each { |line| out.emit(line) }
471 when 'wc'
472 all = %w[l w c].none? { |flag| opts.has?(flag) }
473 report = lambda do |counts, name|
474 fields = (0..2).select { |i| all || opts.has?('lwc'[i]) }.map { |i| counts[i].to_s }
475 out.emit(fields.join(' ') + (name.nil? ? '' : " #{name}"))
476 end
477 totals = [0, 0, 0]
478 read_inputs(command, opts.paths, input, out) do |part|
479 counts = [part.lines.size, 0, 0]
480 part.lines.each do |line|
481 counts[2] += line.bytesize + 1
482 inside = false
483 line.each_byte do |character|
484 counts[1] += 1 if character != 32 && !inside
485 inside = character != 32
486 end
487 end
488 3.times { |i| totals[i] += counts[i] }
489 report.call(counts, part.name)
490 end
491 report.call(totals, 'total') if opts.paths.size > 1
492 when 'head', 'tail'
493 explicit = args.first == '-n'
494 raw = explicit ? args[1] : '10'
495 special = raw.start_with?('-', '+')
496 count = (special ? raw[1..] : raw).to_i
497 read_inputs(command, args.drop(explicit ? 2 : 0), input, out) do |part|
498 first, ending = 0, part.lines.length
499 if command == 'head'
500 ending = special ? [0, ending - count].max : [ending, count].min
501 else
502 first = special ? [ending, [0, count - 1].max].min : [0, ending - count].max
503 end
504 (first...ending).each { |i| out.emit(part.lines[i]) }
505 end
506 else
507 read_inputs(command, opts.paths, input, out) do |part|
508 if command == 'cat'
509 part.lines.each { |line| out.emit(line) }
510 next
511 end
512 first = 0
513 while first < part.lines.length
514 ending = first + 1
515 ending += 1 while ending < part.lines.length && part.lines[first] == part.lines[ending]
516 out.emit((opts.has?('c') ? "#{ending - first} " : '') + part.lines[first])
517 first = ending
518 end
519 end
520 end
521 end

523 def execute(text, input, out)
524 words = text.split(' ')
525 command, args = words.first, words.drop(1)
526 opts = options(args)
527 case command
528 when 'pwd' then out.emit(absolute(@current))
529 when 'echo' then out.emit(text[5..])
530 when 'cd'
531 path = args.first || '/'
532 back = path == '-'
533 if back
534 unless @previous
535 out.error('cd: OLDPWD not set')
536 return
537 end
538 path = @previous
539 end
540 result = resolve(path)
541 if !result.node
542 out.error('cd: %s: %s', path, result.reason)
543 elsif !result.node.is_a?(Directory)
544 out.error('cd: %s: Not a directory', path)
545 else
546 @previous = absolute(@current).freeze
547 @current = result.node
548 out.emit(absolute(@current)) if back
549 end
550 when 'mkdir'
551 opts.paths.each do |path|
552 result = resolve(path, false, opts.has?('p'))
553 if result.absent?
554 create(result.creation_parent, result.creation_name, Directory)
555 elsif !result.node
556 out.error("mkdir: cannot create directory '%s': %s", path, result.reason)
557 elsif !opts.has?('p') || !resolve(path).node.is_a?(Directory)
558 out.error("mkdir: cannot create directory '%s': File exists", path)
559 end
560 end
561 when 'touch'
562 args.each do |path|
563 result = resolve(path)
564 if result.absent?
565 create(result.creation_parent, result.creation_name, FileNode)
566 elsif result.node
567 result.node.modified = @clock
568 else
569 out.error("touch: cannot touch '%s': %s", path, result.reason)
570 end
571 end
572 when 'ls' then listing(opts, out)
573 when 'stat'
574 args.each do |path|
575 result = resolve(path, false)
576 node = result.node
577 unless node
578 out.error("stat: cannot stat '%s': %s", path, result.reason)
579 next
580 end
581 kind = case node
582 when Directory then 'directory'
583 when LinkNode then 'symbolic link'
584 else node.lines.empty? ? 'regular empty file' : 'regular file'
585 end
586 out.print('%s: %s, size %d, inode %d, links %d, modified %d', path, kind, node.size, node.inode, node.links, node.modified)
587 end
588 when 'rm', 'rmdir'
589 directory_only = command == 'rmdir'
590 opts.paths.each do |path|
591 dot = ['.', '..'].include?(last_component(path))
592 if dot && !directory_only
593 out.error("rm: refusing to remove '.' or '..' directory: skipping '%s'", path)
594 next
595 end
596 problem = nil
597 if dot
598 problem = 'Invalid argument'
599 else
600 result = resolve(path, false)
601 node = result.node
602 if !node
603 next if !directory_only && opts.has?('f') && result.reason == MISSING
604 problem = result.reason
605 elsif directory_only && !node.is_a?(Directory)
606 problem = 'Not a directory'
607 elsif !directory_only && node.is_a?(Directory) && !opts.has?('r')
608 problem = 'Is a directory'
609 elsif descendant?(@current, node)
610 problem = 'Device or resource busy'
611 elsif directory_only && !node.entries.empty?
612 problem = 'Directory not empty'
613 else
614 detach(node)
615 end
616 end
617 out.error(directory_only ? "rmdir: failed to remove '%s': %s" : "rm: cannot remove '%s': %s", path, problem) if problem
618 end
619 when 'mv', 'cp'
620 destination = opts.paths.last
621 if opts.paths.size > 2 && !resolve(destination).node.is_a?(Directory)
622 out.error("%s: target '%s' is not a directory", command, destination)
623 return
624 end
625 opts.paths[0...-1].each do |source|
626 command == 'mv' ? move(source, destination, out) : copy(source, destination, opts.has?('r'), out)
627 end
628 when 'ln'
629 path = args[2]
630 result = resolve(path, false)
631 if result.absent?
632 create(result.creation_parent, result.creation_name, LinkNode, args[1])
633 else
634 out.error("ln: failed to create symbolic link '%s': %s", path, result.node ? 'File exists' : result.reason)
635 end
636 when 'readlink', 'realpath'
637 path = args.first
638 result = resolve(path, command == 'realpath')
639 if !result.node
640 out.error('%s: %s: %s', command, path, result.reason)
641 elsif command == 'realpath'
642 out.emit(absolute(result.node))
643 elsif result.node.is_a?(LinkNode)
644 out.emit(result.node.target)
645 else
646 out.error('readlink: %s: Invalid argument', path)
647 end
648 when 'find' then find(args, out)
649 when 'du' then disk_usage(opts, out)
650 when 'tree' then tree(opts, out)
651 when 'cat', 'wc', 'head', 'tail', 'grep', 'sort', 'uniq' then text_command(command, args, input, out)
652 else out.error('%s: command not found', command)
653 end
654 end

656 def process(line, terminal)
657 @clock += 1
658 terminal.puts("#{absolute(@current)}$ #{line}")
659 pipeline = line
660 redirect = nil
661 marker = line.index(' >')
662 if marker
663 pipeline = line[0...marker]
664 append = line[marker..].start_with?(' >> ')
665 path = line[marker + (append ? 4 : 3)..]
666 result = resolve(path)
667 if result.absent?
668 redirect = create(result.creation_parent, result.creation_name, FileNode)
669 elsif result.node.is_a?(FileNode)
670 redirect = result.node
671 write(redirect, []) unless append
672 else
673 terminal.puts("sh: #{path}: #{result.node ? 'Is a directory' : result.reason}")
674 return
675 end
676 end
677 commands = pipeline.split(' | ')
678 input = [].freeze
679 commands.each_with_index do |command, i|
680 out = Output.new(terminal, i + 1 == commands.size && redirect.nil?)
681 execute(command, input, out)
682 input = out.lines.freeze
683 end
684 write(redirect, input, true) if redirect && !input.empty?
685 end
686end

688def glob_match?(pattern, text)
689 previous = Array.new(text.length + 1, false)
690 previous[0] = true
691 pattern.each_char do |character|
692 following = Array.new(previous.length, false)
693 following[0] = character == '*' && previous[0]
694 text.length.times do |i|
