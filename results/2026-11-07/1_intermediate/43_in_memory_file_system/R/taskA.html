106resolve <- function(shell, path, follow = TRUE, parents = FALSE) {
107 expansions <- 0L
108 walk <- function(path, base, follow, original) {
109 node <- if (startsWith(path, "/")) shell$root else base
110 parts <- strsplit(path, "/", fixed = TRUE)[[1L]]
111 parts <- parts[nzchar(parts)]
112 for (index in seq_along(parts)) {
113 if (node$kind != "d") return(resolution(problem = "Not a directory"))
114 name <- parts[index]
115 if (name == ".") next
116 if (name == "..") { node <- node$parent; next }
117 last <- index == length(parts)
118 candidate <- node$entries[[name]]
119 if (is.null(candidate)) {
120 if (original && parents && !last) candidate <- create_node(shell, node, name, "d")
121 else return(resolution(parent = if (original && last) node else NULL,
122 name = name, problem = "No such file or directory"))
123 }
124 if (candidate$kind == "l" && (!last || follow)) {
125 expansions <<- expansions + 1L
126 if (expansions > 8L) return(resolution(problem = "Too many levels of symbolic links"))
127 target <- walk(candidate$target, node, TRUE, FALSE)
128 if (is.null(target$node)) return(target)
129 candidate <- target$node
130 }
131 node <- candidate
132 }
133 resolution(node = node)
134 }
135 walk(path, shell$current, follow, TRUE)
136}
137clone_tree <- function(shell, source, parent, name) {

139 while (length(pending)) {

142 original <- step$source

144 if (original$kind == "d") for (child in rev(children(original))) {
145 pending[[length(pending) + 1L]] <- list(source = child, parent = copy, name = child$name)
146 }
147 }

149move_one <- function(shell, source_path, destination) {
150 problem <- function(path, reason) terminal(sprintf("mv: cannot move '%s' to '%s': %s", source_path, path, reason))
151 if (invalid_source(source_path)) { problem(destination, "Invalid argument"); return(invisible(NULL)) }
152 result <- resolve(shell, source_path, FALSE)
153 source <- result$node
154 if (is.null(source)) { terminal(sprintf("mv: cannot stat '%s': %s", source_path, result$problem)); return(invisible(NULL)) }
155 directory <- resolve(shell, destination)$node
156 if (!is.null(directory) && directory$kind == "d") {
157 parent <- directory
158 name <- last_component(source_path)
159 target <- joined(destination, name)
160 } else {
161 result <- resolve(shell, destination, FALSE)
162 if (!is.null(result$parent)) { parent <- result$parent; name <- result$name }
163 else if (!is.null(result$node)) { parent <- result$node$parent; name <- result$node$name }
164 else { problem(destination, result$problem); return(invisible(NULL)) }
165 target <- destination
166 }
167 existing <- parent$entries[[name]]
168 if (source$kind == "d" && descendant(shell, parent, source)) {
169 terminal(sprintf("mv: cannot move '%s' to a subdirectory of itself, '%s'", source_path, target))
170 } else if (identical(existing, source)) terminal(sprintf("mv: '%s' and '%s' are the same file", source_path, target))
171 else if (!is.null(existing) && source$kind == "d" && existing$kind != "d") {
172 terminal(sprintf("mv: cannot overwrite non-directory '%s' with directory '%s'", target, source_path))
173 } else if (!is.null(existing) && source$kind != "d" && existing$kind == "d") {
174 terminal(sprintf("mv: cannot overwrite directory '%s' with non-directory '%s'", target, source_path))
175 } else if (!is.null(existing) && existing$kind == "d" && length(existing$entries)) problem(target, "Directory not empty")
176 else if (identical(existing, shell$current)) problem(target, "Device or resource busy")
177 else {
178 if (!is.null(existing)) detach(shell, existing)
179 detach(shell, source)
180 attach(shell, parent, name, source)
181 }

183copy_one <- function(shell, source_path, destination, recursive) {
184 problem <- function(path, reason) terminal(sprintf("cp: cannot copy '%s' to '%s': %s", source_path, path, reason))
185 if (invalid_source(source_path)) { problem(destination, "Invalid argument"); return(invisible(NULL)) }
186 result <- resolve(shell, source_path)
187 source <- result$node
188 if (is.null(source)) { terminal(sprintf("cp: cannot stat '%s': %s", source_path, result$problem)); return(invisible(NULL)) }
189 if (source$kind == "d" && !recursive) {
190 terminal(sprintf("cp: -r not specified; omitting directory '%s'", source_path)); return(invisible(NULL))
191 }
192 result <- resolve(shell, destination)
193 if (is.null(result$node) && is.null(result$parent)) { problem(destination, result$problem); return(invisible(NULL)) }
194 target <- if (!is.null(result$node) && result$node$kind == "d") joined(destination, last_component(source_path)) else destination
195 result <- resolve(shell, target, source$kind != "d")
196 if (source$kind == "d") {
197 if (!is.null(result$node)) problem(target, "File exists")
198 else if (descendant(shell, result$parent, source)) {
199 terminal(sprintf("cp: cannot copy a directory, '%s', into itself, '%s'", source_path, target))
200 } else clone_tree(shell, source, result$parent, result$name)
201 } else if (!is.null(result$parent)) create_node(shell, result$parent, result$name, "f", source$lines)
202 else if (is.null(result$node)) problem(target, result$problem)
203 else if (identical(result$node, source)) terminal(sprintf("cp: '%s' and '%s' are the same file", source_path, target))
204 else if (result$node$kind == "d") terminal(sprintf("cp: cannot overwrite directory '%s' with non-directory '%s'", target, source_path))
205 else write_file(shell, result$node, source$lines)
206}

208listing <- function(shell, options, output) {
209 flags <- options$flags
210 paths <- if (length(options$paths)) options$paths else "."
211 for (path in paths) {
212 result <- resolve(shell, path, !("l" %in% flags))
213 node <- result$node
214 if (is.null(node)) { terminal(sprintf("ls: cannot access '%s': %s", path, result$problem)); next }
215 if (node$kind != "d") { entries <- list(node); labels <- path }
216 else {
217 if (length(paths) > 1L) emit(output, paste0(path, ":"))
218 entries <- children(node)
219 if (length(entries) && !("a" %in% flags)) entries <- entries[!startsWith(names(entries), ".")]
220 if ("a" %in% flags) entries <- c(list("." = node, ".." = node$parent), entries)
221 labels <- names(entries)
222 if ("t" %in% flags) {
223 order <- order(-vapply(entries, function(entry) entry$modified, integer(1)), method = "radix")
224 entries <- entries[order]
225 labels <- labels[order]
226 }
227 }
228 if ("r" %in% flags) { entries <- rev(entries); labels <- rev(labels) }
229 if (!length(entries)) next
230 if (!("l" %in% flags)) { emit(output, labels); next }
231 values <- cbind(vapply(entries, link_count, integer(1)), vapply(entries, node_size, numeric(1)),
232 vapply(entries, function(entry) entry$modified, integer(1)))
233 widths <- apply(values, 2L, function(column) max(nchar(number(column), type = "bytes")))
234 lines <- vapply(seq_along(entries), function(index) {
235 entry <- entries[[index]]
236 fields <- vapply(seq_len(3L), function(column) sprintf("%*s", widths[column], number(values[index, column])), character(1))
237 paste0(if (entry$kind == "f") "-" else entry$kind, " ", paste(fields, collapse = " "), " ", labels[index],
238 if (entry$kind == "l") paste0(" -> ", entry$target) else "")
239 }, character(1))
240 emit(output, lines)
241 }

258glob_matches <- function(pattern, text) {
259 previous <- c(TRUE, rep(FALSE, nchar(text, type = "bytes")))
260 characters <- strsplit(text, "", fixed = TRUE)[[1L]]
261 for (character in strsplit(pattern, "", fixed = TRUE)[[1L]]) {
262 following <- rep(FALSE, length(previous))
263 following[1L] <- character == "*" && previous[1L]
264 for (position in seq_along(characters)) following[position + 1L] <- if (character == "*") {
265 previous[position + 1L] || following[position]
266 } else previous[position] && (character == "?" || character == characters[position])
267 previous <- following
268 }
269 tail(previous, 1L)

360text_command <- function(shell, command, arguments, input, output) {
361 options <- options_of(arguments)
362 flags <- options$flags
363 paths <- options$paths
364 if (command == "grep") {
365 pattern <- paths[1L]
366 paths <- paths[-1L]
367 if ("i" %in% flags) pattern <- chartr("ABCDEFGHIJKLMNOPQRSTUVWXYZ", "abcdefghijklmnopqrstuvwxyz", pattern)
368 named <- "r" %in% flags || length(paths) > 1L
369 consume <- function(lines, name) {
370 comparison <- if ("i" %in% flags) chartr("ABCDEFGHIJKLMNOPQRSTUVWXYZ", "abcdefghijklmnopqrstuvwxyz", lines) else lines
371 selected <- grepl(pattern, comparison, fixed = TRUE, useBytes = TRUE)
372 if ("v" %in% flags) selected <- !selected
373 prefix <- if (named) paste0(name, ":") else ""
374 if ("c" %in% flags) emit(output, paste0(prefix, number(sum(selected))))
375 else if (any(selected)) emit(output, paste0(prefix, if ("n" %in% flags) paste0(which(selected), ":") else "", lines[selected]))
376 }
377 if (!("r" %in% flags)) read_inputs(shell, command, paths, input, consume)
378 else {
379 if (!length(paths)) paths <- "."
380 for (path in paths) {
381 result <- resolve(shell, path)
382 if (is.null(result$node)) { terminal(sprintf("grep: %s: %s", path, result$problem)); next }
383 for (row in visits(result$node, path)) if (row$node$kind == "f") consume(row$node$lines, row$path)
384 }
385 }
386 } else if (command == "sort") {
387 gathered <- new_output(FALSE)
388 read_inputs(shell, command, paths, input, function(lines, name) emit(gathered, lines))
389 lines <- output_lines(gathered)
390 ordering <- if ("n" %in% flags) order(vapply(lines, numeric_prefix, integer(1)), lines, method = "radix") else order(lines, method = "radix")
391 if ("r" %in% flags) ordering <- rev(ordering)
392 emit(output, lines[ordering])
393 } else if (command == "wc") {
394 selected <- c("l", "w", "c") %in% flags
395 if (!any(selected)) selected[] <- TRUE
396 report <- function(counts, name) emit(output, paste0(paste(number(counts[selected]), collapse = " "), if (is.null(name)) "" else paste0(" ", name)))
397 totals <- numeric(3L)
398 read_inputs(shell, command, paths, input, function(lines, name) {
399 counts <- c(as.double(length(lines)), sum(vapply(lines, word_count, integer(1))), sum(as.double(nchar(lines, type = "bytes")) + 1))
400 totals <<- totals + counts
401 report(counts, name)
402 })
403 if (length(paths) > 1L) report(totals, "total")
404 } else if (command %in% c("head", "tail")) {
405 explicit <- length(arguments) > 0L && arguments[1L] == "-n"
406 raw <- if (explicit) arguments[2L] else "10"
407 paths <- if (explicit) arguments[-seq_len(2L)] else arguments
408 special <- startsWith(raw, "-") || startsWith(raw, "+")
409 count <- as.integer(if (special) substring(raw, 2L) else raw)
410 read_inputs(shell, command, paths, input, function(lines, name) {
411 start <- 0L
412 stop <- length(lines)
413 if (command == "head") stop <- if (special) max(0L, stop - count) else min(stop, count)
414 else start <- if (special) min(stop, max(0L, count - 1L)) else max(0L, stop - count)
415 if (start < stop) emit(output, lines[seq.int(start + 1L, stop)])
416 })
417 } else read_inputs(shell, command, paths, input, function(lines, name) {
418 if (command == "cat") emit(output, lines)
419 else if (length(lines)) {
420 groups <- rle(lines)
421 emit(output, if ("c" %in% flags) paste(number(groups$lengths), groups$values) else groups$values)
422 }
423 })

426execute <- function(shell, text, input, output) {
427 words <- strsplit(text, " ", fixed = TRUE)[[1L]]
428 command <- words[1L]
429 arguments <- words[-1L]
430 options <- options_of(arguments)
431 flags <- options$flags
432 paths <- options$paths
433 if (command == "pwd") emit(output, absolute(shell, shell$current))
434 else if (command == "echo") emit(output, substring(text, 6L))
435 else if (command == "cd") {
436 path <- if (length(arguments)) arguments[1L] else "/"
437 previous <- path == "-"
438 if (previous) {
439 if (is.null(shell$previous)) { terminal("cd: OLDPWD not set"); return(invisible(NULL)) }
440 path <- shell$previous
441 }
442 result <- resolve(shell, path)
443 if (is.null(result$node)) terminal(sprintf("cd: %s: %s", path, result$problem))
444 else if (result$node$kind != "d") terminal(sprintf("cd: %s: Not a directory", path))
445 else {
446 shell$previous <- absolute(shell, shell$current)
447 shell$current <- result$node
448 if (previous) emit(output, absolute(shell, shell$current))
449 }
450 } else if (command == "mkdir") for (path in paths) {
451 result <- resolve(shell, path, FALSE, "p" %in% flags)
452 if (!is.null(result$parent)) create_node(shell, result$parent, result$name, "d")
453 else if (is.null(result$node)) terminal(sprintf("mkdir: cannot create directory '%s': %s", path, result$problem))
454 else {
455 directory <- NULL
456 if ("p" %in% flags) directory <- resolve(shell, path)$node
457 if (is.null(directory) || directory$kind != "d") terminal(sprintf("mkdir: cannot create directory '%s': File exists", path))
458 }
459 } else if (command == "touch") for (path in arguments) {
460 result <- resolve(shell, path)
461 if (!is.null(result$parent)) create_node(shell, result$parent, result$name, "f")
462 else if (is.null(result$node)) terminal(sprintf("touch: cannot touch '%s': %s", path, result$problem))
463 else result$node$modified <- shell$clock
464 } else if (command == "ls") listing(shell, options, output)
465 else if (command == "stat") for (path in arguments) {
466 result <- resolve(shell, path, FALSE)
467 node <- result$node
468 if (is.null(node)) { terminal(sprintf("stat: cannot stat '%s': %s", path, result$problem)); next }
469 kind <- switch(node$kind, d = "directory", l = "symbolic link", f = if (length(node$lines)) "regular file" else "regular empty file")
470 emit(output, sprintf("%s: %s, size %s, inode %s, links %s, modified %s", path, kind, number(node_size(node)),
471 number(node$inode), number(link_count(node)), number(node$modified)))
472 } else if (command %in% c("rm", "rmdir")) for (path in paths) {
473 directory_only <- command == "rmdir"
474 dot <- last_component(path) %in% c(".", "..")
475 if (dot && !directory_only) { terminal(sprintf("rm: refusing to remove '.' or '..' directory: skipping '%s'", path)); next }
476 problem <- ""
477 if (dot) problem <- "Invalid argument" else {
478 result <- resolve(shell, path, FALSE)
479 node <- result$node
480 if (is.null(node)) {
481 if (!directory_only && "f" %in% flags && result$problem == "No such file or directory") next
482 problem <- result$problem
483 } else if (directory_only && node$kind != "d") problem <- "Not a directory"
484 else if (!directory_only && node$kind == "d" && !("r" %in% flags)) problem <- "Is a directory"
485 else if (descendant(shell, shell$current, node)) problem <- "Device or resource busy"
486 else if (directory_only && length(node$entries)) problem <- "Directory not empty"
487 else detach(shell, node)
488 }
489 if (nzchar(problem)) terminal(sprintf(if (directory_only) "rmdir: failed to remove '%s': %s" else "rm: cannot remove '%s': %s", path, problem))
490 } else if (command %in% c("mv", "cp")) {
491 destination <- tail(paths, 1L)
492 if (length(paths) > 2L) {
493 node <- resolve(shell, destination)$node
494 if (is.null(node) || node$kind != "d") { terminal(sprintf("%s: target '%s' is not a directory", command, destination)); return(invisible(NULL)) }
495 }
496 for (path in head(paths, -1L)) {
497 if (command == "mv") move_one(shell, path, destination) else copy_one(shell, path, destination, "r" %in% flags)
498 }
499 } else if (command == "ln") {
500 path <- arguments[3L]
501 result <- resolve(shell, path, FALSE)
502 if (!is.null(result$parent)) create_node(shell, result$parent, result$name, "l", target = arguments[2L])
503 else terminal(sprintf("ln: failed to create symbolic link '%s': %s", path, if (is.null(result$node)) result$problem else "File exists"))
504 } else if (command %in% c("readlink", "realpath")) {
505 path <- arguments[1L]
506 result <- resolve(shell, path, command == "realpath")
507 if (is.null(result$node)) terminal(sprintf("%s: %s: %s", command, path, result$problem))
508 else if (command == "realpath") emit(output, absolute(shell, result$node))
509 else if (result$node$kind == "l") emit(output, result$node$target)
510 else terminal(sprintf("readlink: %s: Invalid argument", path))
511 } else if (command == "find") find_nodes(shell, arguments, output)
512 else if (command == "du") disk_usage(shell, options, output)
513 else if (command == "tree") draw_tree(shell, options, output)
514 else if (command %in% c("cat", "wc", "head", "tail", "grep", "sort", "uniq")) text_command(shell, command, arguments, input, output)
515 else terminal(paste0(command, ": command not found"))
516 invisible(NULL)
517}
518process_line <- function(shell, line) {
519 shell$clock <- shell$clock + 1L
520 terminal(paste0(absolute(shell, shell$current), "$ ", line))
521 marker <- regexpr(" >", line, fixed = TRUE)[1L]
522 pipeline <- line
523 redirect <- NULL
524 if (marker != -1L) {
525 pipeline <- substr(line, 1L, marker - 1L)
526 append <- startsWith(substring(line, marker), " >> ")
527 path <- substring(line, marker + if (append) 4L else 3L)
528 result <- resolve(shell, path)
529 if (!is.null(result$parent)) redirect <- create_node(shell, result$parent, result$name, "f")
530 else if (!is.null(result$node) && result$node$kind == "f") {
531 redirect <- result$node
532 if (!append) write_file(shell, redirect, character())
533 } else {
534 terminal(sprintf("sh: %s: %s", path, if (is.null(result$node)) result$problem else "Is a directory"))
535 return(invisible(NULL))
536 }
537 }
538 commands <- strsplit(pipeline, " | ", fixed = TRUE)[[1L]]
539 input <- character()
540 for (index in seq_along(commands)) {
541 output <- new_output(index == length(commands) && is.null(redirect))
542 execute(shell, commands[index], input, output)
543 input <- output_lines(output)
544 }
545 if (!is.null(redirect) && length(input)) write_file(shell, redirect, input, TRUE)
546 invisible(NULL)
547}
548main <- function() {
549 shell <- new_shell()
550 input <- file("stdin", open = "r")
551 on.exit(close(input))
552 repeat {
553 line <- readLines(input, n = 1L, warn = FALSE)
554 if (!length(line) || line == "exit") break
555 process_line(shell, line)
556 }
557}
