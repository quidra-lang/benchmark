func ladder(_ b: [Character], _ n: Int, _ p: Int) -> (String, Int) {
    let defender = b[p], attacker = other(defender)
    func read(_ current: [Character], _ depth: Int, _ defending: Bool) -> Reading {
        if depth >= 40 {
            return Reading(success: defending, move: -1)
        }
        let g = group(current, n, p)
        var candidates = g.liberties, seen: Set<Int> = []
        if defending {
            for q in g.stones {
                for t in adjacent(q, n) {
                    if current[t] == attacker && !seen.contains(t) {
                        let e = group(current, n, t)
                        seen.formUnion(e.stones)
                        if e.liberties.count == 1 {
                            candidates.formUnion(e.liberties)
                        }
                    }
                }
            }
        }
        for q in candidates.sorted() {
            let next = placement(current, n, defending ? defender : attacker, q)
            if !next.reason.isEmpty {
                continue
            }
            let success: Bool
            if defending {
                let count = group(next.board, n, p).liberties.count
                success = count >= 3 || (count == 2 && !read(next.board, depth + 1, false).success)
            } else {
                success = !read(next.board, depth + 1, true).success
            }
            if success {
                let r = Reading(success: true, move: q)
                return r
            }
        }
        let r = Reading(success: false, move: -1)
        return r
