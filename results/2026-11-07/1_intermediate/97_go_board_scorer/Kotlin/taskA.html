8fun komi(s:String):Int? {
9 if(!Regex("-?(0|[1-9][0-9]*)(?:\\.[05])?").matches(s))return null;
10 val negative=s.startsWith('-');
11 val a=s.removePrefix("-").split('.');
12 val whole=a[0].toIntOrNull()?:return null;
13 if(whole>100)return null;
14 val x=2*whole+if(a.size==2&&a[1]=="5")1 else 0;
15 return if(negative&&x==0)null else if(negative)-x else x
16}
17fun adjacent(p:Int,n:Int,diagonal:Boolean=false):List<Int> {
18 val r=p/n;
19 val c=p%n;
20 val steps=if(diagonal)listOf(-1 to -1,-1 to 1,1 to -1,1 to 1) else listOf(-1 to 0,1 to 0,0 to -1,0 to 1);
21 return steps.mapNotNull{
22 (dr,dc)->if(r+dr in 0 until n&&c+dc in 0 until n)(r+dr)*n+c+dc else null
23 }
24}
25fun flood(b:String,n:Int,p:Int,allowed:String):Set<Int> {
26 val seen=mutableSetOf(p);
27 val todo=ArrayDeque<Int>();
28 todo.add(p);
29 while(todo.isNotEmpty())for(q in adjacent(todo.removeLast(),n))if(q !in seen&&b[q] in allowed){
30 seen.add(q);
31 todo.add(q)
32 };
33 return seen
34}

39 q->adjacent(q,n).filter{

44}
45fun groups(b:String,n:Int,color:Char?=null):List<Group> {
46 val out=mutableListOf<Group>();
47 val seen=mutableSetOf<Int>();
48 for(p in b.indices)if(b[p]!='.'&&p !in seen&&(color==null||b[p]==color)){
49 val g=group(b,n,p);
50 seen.addAll(g.stones);
51 out.add(g)
52 };
53 return out

55data class Placement(val board:String,val captured:Int,val reason:String)
56fun placement(b:String,n:Int,c:Char,p:Int):Placement {
57 if(b[p]!='.')return Placement(b,0,"occupied");
58 val next=b.toCharArray();
59 next[p]=c;
60 var captured=0;
61 val seen=mutableSetOf<Int>();
62 for(q in adjacent(p,n))if(next[q]==other(c)&&q !in seen){
63 val g=group(next.concatToString(),n,q);
64 seen.addAll(g.stones);
65 if(g.liberties.isEmpty()){
66 for(t in g.stones)next[t]='.';
67 captured+=g.stones.size
68 }
69 };
70 val value=next.concatToString();
71 return Placement(value,captured,if(group(value,n,p).liberties.isEmpty())"suicide" else "")
72}
73data class Region(val first:Int,val points:Set<Int>,val owner:String)
74fun regions(b:String,n:Int):List<Region> {
75 val out=mutableListOf<Region>();
76 val seen=mutableSetOf<Int>();
77 for(p in b.indices)if(b[p]=='.'&&p !in seen){
78 val points=flood(b,n,p,".");
79 seen.addAll(points);
80 val border=points.flatMap{
81 q->adjacent(q,n).map{
82 b[it]
83 }
84 }.filter{
85 it!='.'
86 }.toSet();
87 out.add(Region(p,points,if(border.size==1)border.first().toString() else "neutral"))
88 };
89 return out
90}
91fun safeGroups(b:String,n:Int,c:Char):List<Group> {
92 val own=groups(b,n,c);
93 val touch=mutableListOf<Set<Int>>();
94 val vital=mutableListOf<Set<Int>>();
95 val seen=mutableSetOf<Int>()
96 for(p in b.indices)if(b[p]!=c&&p !in seen){
97 val area=flood(b,n,p,"."+other(c));
98 seen.addAll(area);
99 val empty=area.filter{
100 b[it]=='.'
101 };
102 touch.add(own.indices.filter{
103 i->area.any{
104 q->adjacent(q,n).any{
105 it in own[i].stones
106 }
107 }
108 }.toSet());
109 vital.add(own.indices.filter{
110 i->own[i].liberties.containsAll(empty)
111 }.toSet())
112 }
113 val s=own.indices.toMutableSet();
114 val t=touch.indices.toMutableSet();
115 while(true){
116 val before=s.size+t.size;
117 s.removeAll{
118 s0->t.count{
119 j->s0 in vital[j]
120 }<2
121 };
122 t.removeAll{
123 j->!s.containsAll(touch[j])
124 };
125 if(before==s.size+t.size)break
126 };
127 return s.sorted().map{
128 own[it]
129 }
130}
131data class Reading(val success:Boolean,val move:Int=-1)
132data class ReadKey(val board:String,val depth:Int,val defending:Boolean)
133data class LadderResult(val status:String,val move:Int=-1)
134fun ladder(b:String,n:Int,p:Int):LadderResult {
135 val defender=b[p];
136 val attacker=other(defender);
137 val memo=mutableMapOf<ReadKey,Reading>()
138 fun read(current:String,depth:Int,defending:Boolean):Reading {
139 if(depth>=40)return Reading(defending);
140 val key=ReadKey(current,depth,defending);
141 memo[key]?.let{
142 return it
143 };
144 val g=group(current,n,p);
145 val candidates=g.liberties.toMutableSet();
146 val seen=mutableSetOf<Int>()
147 if(defending)for(q in g.stones)for(t in adjacent(q,n))if(current[t]==attacker&&t !in seen){
148 val e=group(current,n,t);
149 seen.addAll(e.stones);
150 if(e.liberties.size==1)candidates.addAll(e.liberties)
151 }
152 for(q in candidates.sorted()){
153 val next=placement(current,n,if(defending)defender else attacker,q);
154 if(next.reason.isNotEmpty())continue;
155 val success=if(defending){
156 val count=group(next.board,n,p).liberties.size;
157 count>=3||(count==2&&!read(next.board,depth+1,false).success)
158 }else !read(next.board,depth+1,true).success;
159 if(success){
160 val r=Reading(true,q);
161 memo[key]=r;
162 return r
163 }
164 }
165 val r=Reading(false);
166 memo[key]=r;
167 return r
168 }
169 val count=group(b,n,p).liberties.size;
170 if(count>=3)return LadderResult("safe");
171 val r=read(b,0,count==1);
172 return LadderResult(if(if(count==1)r.success else !r.success)"escapes" else "captured",r.move)
173}

189 fun reset(size:Int,k:Int){
190 n=size;
191 komi2=k;
192 board=".".repeat(n*n);
193 turn='B';
194 captures=mutableMapOf('B' to 0,'W' to 0);
195 passes=0;

202 fun point(s:String):Int {
203 if(!Regex("[A-HJ-T](?:[1-9]|1[0-9])").matches(s))return -1;
204 val c=columns.indexOf(s[0]);
205 val r=s.substring(1).toInt();
206 return if(c<n&&r<=n)(r-1)*n+c else -1
207 }
208 fun name(p:Int)=columns[p%n].toString()+(p/n+1)
209 fun sgf(p:Int)=""+(97+p%n).toChar()+(97+n-1-p/n).toChar()
210 fun check(c:Char,p:Int):Placement {
211 val r=placement(board,n,c,p);
212 var reason=r.reason;
213 if(reason.isEmpty()&&positions.size>=2&&r.board==positions[positions.size-2])reason="ko";
214 if(reason.isEmpty()&&superko&&r.board in positions)reason="superko";
215 return r.copy(reason=reason)
216 }
217 fun move(c:Char,p:Int,checked:Placement?=null):Int {
218 val before=Snapshot(board,turn,captures.getValue('B'),captures.getValue('W'),passes,over);
219 var captured=0;
220 if(p<0){
221 passes++;
222 if(passes==2)over=true
223 }else{
224 val r=checked?:check(c,p);
225 board=r.board;
226 captured=r.captured;
227 captures[c]=captures.getValue(c)+captured;
228 passes=0
229 };
230 turn=other(c);
231 history.add(Move(c,p,captured,before));
232 positions.add(board);
233 dead.clear();
234 return captured
235 }
236 fun clean()=board.mapIndexed{
237 p,c->if(p in dead)'.' else c
238 }.joinToString("")
239 fun show(map:Boolean=false){
240 val b=if(map)clean() else board;
241 val chars=b.toCharArray();
242 val stars=handicapPoints.getValue(n).filterIndexed{
243 i,_->n==19||i<4||i==8
244 }.map(::point).toSet();
245 if(map)for(r in regions(b,n))for(p in r.points)chars[p]=when(r.owner){
246 "B"->'x';
247 "W"->'o';
248 else->'.'
249 };
250 for(p in b.indices)if(b[p]!='.'){
251 val c=if(b[p]=='B')'X' else 'O';
252 chars[p]=if(!map&&p in dead)c.lowercaseChar() else c
253 }else if(!map)chars[p]=if(p in stars)'+' else '.';
254 val header=" "+columns.take(n).toList().joinToString(" ");
255 emit(header);
256 for(r in n downTo 1)emit(r.toString().padStart(2),chars.slice((r-1)*n until r*n).joinToString(" "),r);
257 emit(header)
258 }
259 fun score(method:String){
260 val b=clean();
261 val territory=mutableMapOf('B' to 0,'W' to 0);
262 val total=mutableMapOf<Char,Int>();
263 for(r in regions(b,n))if(r.owner!="neutral")territory[r.owner[0]]=territory.getValue(r.owner[0])+r.points.size;
264 for(c in "BW"){
265 val fields=mutableListOf<Any>(c);
266 if(method=="AREA"){
267 val stones=b.count{
268 it==c
269 };
270 total[c]=2*(stones+territory.getValue(c))+(if(c=='W')komi2 else 0);
271 fields.addAll(listOf("stones",stones,"territory",territory.getValue(c)))
272 }else{
273 val prisoners=captures.getValue(c)+dead.count{
274 board[it]==other(c)
275 };
276 total[c]=2*(territory.getValue(c)+prisoners)+(if(c=='W')komi2 else 0);
277 fields.addAll(listOf("territory",territory.getValue(c),"prisoners",prisoners))
278 };
279 if(c=='W')fields.addAll(listOf("komi",half(komi2)));
280 fields.addAll(listOf("total",half(total.getValue(c))));
281 emit(*fields.toTypedArray())
282 };
283 val d=total.getValue('B')-total.getValue('W');
284 emit("result",if(d>0)"B+"+half(d) else if(d<0)"W+"+half(-d) else "Draw")
285 }
286 fun execute(line:String){
287 val words=line.split(' ');
288 val c=words[0];
289 val a=words.drop(1);
290 if(c !in arity){
291 emit("ERROR unknown command");
292 return
293 };
294 if(a.size!=arity[c]){
295 emit("ERROR bad arguments");
296 return
297 }
298 val valid=when(c){
299 "BOARD"->a[0] in setOf("9","13","19")&&komi(a[1])!=null;
300 "KOMI"->komi(a[0])!=null;
301 "SUPERKO"->a[0] in setOf("ON","OFF");
302 "HANDICAP"->Regex("[2-9]").matches(a[0]);
303 "PLAY","PASS","LEGAL","EYES","SAFE"->a[0] in setOf("B","W");
304 "SCORE"->a[0] in setOf("AREA","TERRITORY");
305 else->true
306 };
307 if(!valid){
308 emit("ERROR bad arguments");
309 return
310 }
311 var p=-1;
312 if(c in setOf("PLAY","GROUP","LADDER","DEAD")){
313 p=point(a[if(c=="PLAY")1 else 0]);
314 if(p<0){
315 emit("ERROR bad point");
316 return
317 }
318 }
319 val records=mutableListOf<Pair<Char,Int>>();
320 if(c=="REPLAY"){
321 if(!Regex("(?:;[BW]\\[[a-z]{0,2}\\])+").matches(a[0])){
322 emit("ERROR bad sequence");
323 return
324 };
325 for(m in Regex(";([BW])\\[([a-z]*)\\]").findAll(a[0])){
326 val text=m.groupValues[2];
327 if(text.isNotEmpty()&&(text.length!=2||text.any{
328 it-'a'>=n
329 })){
330 emit("ERROR bad sequence");
331 return
332 };
333 records.add(m.groupValues[1][0] to if(text.isEmpty())-1 else (n-1-(text[1]-'a'))*n+(text[0]-'a'))
334 }
335 }
336 if(c in setOf("PLAY","PASS","REPLAY")&&over){
337 emit("ERROR game over");
338 return
339 }
340 when(c){
341 "BOARD"->{
342 reset(a[0].toInt(),komi(a[1])!!);
343 emit("board",n,"komi",half(komi2))
344 }
345 "KOMI"->{
346 komi2=komi(a[0])!!;
347 emit("komi",half(komi2))
348 }
349 "SUPERKO"->{
350 superko=a[0]=="ON";
351 emit("superko",if(superko)"on" else "off")
352 }
353 "HANDICAP"->{
354 if(history.isNotEmpty()||board.any{
355 it!='.'
356 }){
357 emit("ERROR handicap not allowed");
358 return
359 };
360 val k=a[0].toInt();
361 val ix=if(k==5)listOf(0,1,2,3,8) else if(k==7)listOf(0,1,2,3,4,5,8) else (0 until k).toList();
362 handicap=ix.map{
363 point(handicapPoints.getValue(n)[it])
364 };
365 val b=board.toCharArray();
366 for(q in handicap)b[q]='B';
367 board=b.concatToString();
368 turn='W';
369 positions=mutableListOf(board);
370 dead.clear();
371 emit("handicap",k,*handicap.map(::name).toTypedArray())
372 }
373 "PLAY"->{
374 val color=a[0][0];
375 val r=check(color,p);
376 val reason=if(color!=turn)"turn" else r.reason;
377 if(reason.isNotEmpty())emit("ILLEGAL",color,a[1],reason) else {
378 val count=move(color,p,r);
379 if(count==0)emit(color,a[1]) else emit(color,a[1],"captures",count)
380 }
381 }
382 "PASS"->{
383 if(a[0][0]!=turn)emit("ILLEGAL",a[0],"pass turn") else {
384 move(a[0][0],-1);
385 emit(a[0],"pass");
386 if(over)emit("game over")
387 }
388 }
389 "REPLAY"->{
390 var played=0;
391 var stop=listOf<Any>();
392 for((color,q) in records){
393 val next=if(q<0)null else check(color,q);
394 val reason=if(over)"game over" else if(color!=turn)"turn" else next?.reason?:"";
395 if(reason.isNotEmpty()){
396 stop=listOf("stopped",color,if(q<0)"pass" else name(q),reason);
397 break
398 };
399 move(color,q,next);
400 played++
401 };
402 emit("replay",played,"moves",*stop.toTypedArray());
403 if(over)emit("game over")
404 }
405 "UNDO"->{
406 if(history.isEmpty()){
407 emit("ERROR nothing to undo");
408 return
409 };
410 val m=history.removeLast();
411 val s=m.before;
412 board=s.board;
413 turn=s.turn;
414 captures=mutableMapOf('B' to s.black,'W' to s.white);
415 passes=s.passes;
416 over=s.over;
417 positions.removeLast();
418 dead.clear();
419 emit("undo",m.color,if(m.point<0)"pass" else name(m.point))
420 }
421 "SHOW"->show();
422 "MAP"->show(true)
423 "STATUS"->{
424 emit("size",n,"komi",half(komi2),"handicap",handicap.size,"superko",if(superko)"on" else "off");
425 if(over)emit("moves",history.size,"game over") else emit("moves",history.size,"next",turn,"passes",passes)
426 }
427 "CAPTURES"->emit("captures B",captures.getValue('B'),"W",captures.getValue('W'))
428 "HISTORY"->{
429 emit("history",history.size);
430 for((i,m) in history.withIndex()){
431 val row=mutableListOf<Any>(i+1,m.color,if(m.point<0)"pass" else name(m.point));
432 if(m.captured>0)row.addAll(listOf("captures",m.captured));
433 emit(*row.toTypedArray())
434 }
435 }
436 "MOVES"->{
437 val s=StringBuilder("(;GM[1]FF[4]SZ[$n]KM[${half(komi2)}]");
438 if(handicap.isNotEmpty()){
439 s.append("HA[${handicap.size}]AB");
440 for(q in handicap)s.append('[').append(sgf(q)).append(']')
441 };
442 for(m in history)s.append(';').append(m.color).append('[').append(if(m.point<0)"" else sgf(m.point)).append(']');
443 emit(s.append(')'))
444 }
445 "GROUP","DEAD","LADDER"->{
446 if(board[p]=='.'){
447 emit("ERROR no stone");
448 return
449 };
450 val g=group(board,n,p);
451 when(c){
452 "GROUP"->{
453 emit("group",g.color,name(g.first),"size",g.stones.size,"liberties",g.liberties.size);
454 emit("stones",*g.stones.sorted().map(::name).toTypedArray());
455 emit("liberties",*g.liberties.sorted().map(::name).toTypedArray())
456 };
457 "DEAD"->{
458 val alive=p in dead;
459 if(alive)dead.removeAll(g.stones) else dead.addAll(g.stones);
460 emit(if(alive)"alive" else "dead",g.color,name(g.first),"size",g.stones.size)
461 };
462 else->{
463 val r=ladder(board,n,p);
464 if(r.move<0)emit("ladder",a[0],r.status) else emit("ladder",a[0],r.status,name(r.move))
465 }
466 }
467 }
468 "GROUPS"->{
469 val gs=groups(board,n);
470 emit("groups",gs.size);
471 for(g in gs){
472 val row=mutableListOf<Any>(g.color,name(g.first),"size",g.stones.size,"liberties",g.liberties.size);
473 if(g.first in dead)row.add("dead");
474 emit(*row.toTypedArray())
475 }
476 }
477 "ATARI"->{
478 val gs=groups(board,n).filter{
479 it.liberties.size==1
480 };
481 if(gs.isEmpty())emit("atari none");
482 for(g in gs)emit("atari",g.color,name(g.first),"size",g.stones.size,"at",name(g.liberties.first()))
483 }
484 "LEGAL"->{
485 val rejected=mutableListOf<Pair<String,String>>();
486 var count=0;
487 for(q in board.indices)if(board[q]=='.'){
488 val reason=check(a[0][0],q).reason;
489 if(reason.isEmpty())count++ else rejected.add(name(q) to reason)
490 };
491 emit("legal",a[0],count);
492 for((q,reason) in rejected)emit(q,reason)
493 }
494 "EYES"->{
495 val found=mutableListOf<Pair<String,String>>();
496 for(q in board.indices)if(board[q]=='.'){
497 val ns=adjacent(q,n);
498 if(ns.all{
499 board[it]==a[0][0]
500 }){
501 val enemies=adjacent(q,n,true).count{
502 board[it]==other(a[0][0])
503 };
504 found.add(name(q) to if(enemies<=if(ns.size==4)1 else 0)"real" else "false")
505 }
506 };
507 emit("eyes",a[0],found.size);
508 for((q,kind) in found)emit(q,kind)
509 }
510 "SAFE"->{
511 val gs=safeGroups(board,n,a[0][0]);
512 emit("safe",a[0],gs.size);
513 for(g in gs)emit(name(g.first),"size",g.stones.size)
514 }
515 "REGIONS"->{
516 val rs=regions(clean(),n);
517 emit("regions",rs.size);
518 for(r in rs)emit(name(r.first),"size",r.points.size,r.owner)
519 }
520 "SCORE"->score(a[0])
521 }
522 }
523}
