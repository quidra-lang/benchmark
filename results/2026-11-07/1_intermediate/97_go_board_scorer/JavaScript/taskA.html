43function adjacent(p, n, diagonal = false) {
44 const r = Math.floor(p / n),
45 c = p % n,
46 a = [];
47 const steps = diagonal
48 ? [
49 [-1, -1],
50 [-1, 1],
51 [1, -1],
52 [1, 1],
53 ]
54 : [
55 [-1, 0],
56 [1, 0],
57 [0, -1],
58 [0, 1],
59 ];
60 for (const [dr, dc] of steps)
61 if (r + dr >= 0 && r + dr < n && c + dc >= 0 && c + dc < n) a.push((r + dr) * n + c + dc);
62 return a;
63}
64function flood(b, n, p, allowed) {
65 const s = new Set([p]),
66 todo = [p];
67 while (todo.length)
68 for (const q of adjacent(todo.pop(), n))
69 if (!s.has(q) && allowed.includes(b[q])) {
70 s.add(q);
71 todo.push(q);
72 }
73 return s;
74}
75function group(b, n, p) {
76 const stones = flood(b, n, p, b[p]),
77 liberties = new Set();

79 return {
80 color: b[p],
81 first: Math.min(...stones),
82 stones,
83 liberties,
84 };

97function placement(b, n, c, p) {
98 if (b[p] !== ".")
99 return {
100 board: b,
101 captured: 0,

104 const a = b.split("");

107 const seen = new Set();

112 if (!g.liberties.size)

114 a[t] = ".";
115 captured++;
116 }
117 }

119 return {
120 board,
121 captured,

125function regions(b, n) {
126 const seen = new Set(),
127 r = [];
128 for (let p = 0; p < b.length; p++)
129 if (b[p] === "." && !seen.has(p)) {
130 const points = flood(b, n, p, "."),

132 for (const q of points) {

135 }
136 r.push({
137 first: p,
138 points,
139 owner: border.size === 1 ? [...border][0] : "neutral",
140 });
141 }

143}
144function safe(b, n, c) {
145 const own = groups(b, n, c),
146 areas = [],
147 seen = new Set();
148 for (let p = 0; p < b.length; p++)
149 if (b[p] !== c && !seen.has(p)) {
150 const points = flood(b, n, p, "." + other(c));
151 for (const q of points) seen.add(q);
152 const empty = [...points].filter((q) => b[q] === ".");
153 areas.push({
154 touch: new Set(
155 own.flatMap((g, i) =>
156 [...points].some((q) => adjacent(q, n).some((t) => g.stones.has(t)))
157 ? [i]
158 : [],
159 ),
160 ),
161 vital: new Set(
162 own.flatMap((g, i) => (empty.every((q) => g.liberties.has(q)) ? [i] : [])),
163 ),
164 });
165 }
166 let s = new Set(own.map((_, i) => i)),
167 t = new Set(areas.map((_, i) => i));
168 while (true) {
169 const count = s.size + t.size;
170 s = new Set([...s].filter((i) => [...t].filter((j) => areas[j].vital.has(i)).length >= 2));
171 t = new Set([...t].filter((j) => [...areas[j].touch].every((i) => s.has(i))));
172 if (count === s.size + t.size) break;
173 }
174 return sorted(s).map((i) => own[i]);
175}
176function ladder(b, n, p) {
177 const defender = b[p],
178 attacker = other(defender),
179 memo = new Map();
180 function read(current, depth, defending) {
181 if (depth >= 40)
182 return {
183 success: defending,
184 move: -1,
185 };
186 const key = current + ":" + depth + ":" + defending;
187 if (memo.has(key)) return memo.get(key);
188 const g = group(current, n, p),
189 candidates = new Set(g.liberties),
190 seen = new Set();
191 if (defending)
192 for (const q of g.stones)
193 for (const t of adjacent(q, n))
194 if (current[t] === attacker && !seen.has(t)) {
195 const e = group(current, n, t);
196 for (const s of e.stones) seen.add(s);
197 if (e.liberties.size === 1) for (const s of e.liberties) candidates.add(s);
198 }
199 for (const q of sorted(candidates)) {
200 const next = placement(current, n, defending ? defender : attacker, q);
201 if (next.reason) continue;
202 let success;
203 if (defending) {
204 const m = group(next.board, n, p).liberties.size;
205 success = m >= 3 || (m === 2 && !read(next.board, depth + 1, false).success);
206 } else success = !read(next.board, depth + 1, true).success;
207 if (success) {
208 const result = {
209 success: true,
210 move: q,
211 };
212 memo.set(key, result);
213 return result;
214 }
215 }
216 const result = {
217 success: false,
218 move: -1,
219 };
220 memo.set(key, result);
221 return result;
222 }
223 const m = group(b, n, p).liberties.size;
224 if (m >= 3)
225 return {
226 status: "safe",
227 move: -1,
228 };
229 const r = read(b, 0, m === 1);
230 return {
231 status: (m === 1 ? r.success : !r.success) ? "escapes" : "captured",
232 move: r.move,
233 };
234}

392 execute(line) {

398 if (c === "BOARD") valid = ["9", "13", "19"].includes(a[0]) && komi(a[1]) !== null;
399 else if (c === "KOMI") valid = komi(a[0]) !== null;
400 else if (c === "SUPERKO") valid = ["ON", "OFF"].includes(a[0]);
401 else if (c === "HANDICAP") valid = /^[2-9]$/.test(a[0]);
402 else if (["PLAY", "PASS", "LEGAL", "EYES", "SAFE"].includes(c))

405 if (!valid) return error("bad arguments");
406 const p = ["PLAY", "GROUP", "LADDER", "DEAD"].includes(c)

409 if (["PLAY", "GROUP", "LADDER", "DEAD"].includes(c) && p < 0) return error("bad point");
410 let records = [];
411 if (c === "REPLAY") {
412 if (!/^(?:;[BW]\[[a-z]{0,2}\])+$/.test(a[0])) return error("bad sequence");
413 records = [...a[0].matchAll(/;([BW])\[([a-z]*)\]/g)];

415 records.some(
416 (r) =>
417 r[2] &&
418 (r[2].length !== 2 ||
419 [...r[2]].some((ch) => ch.charCodeAt(0) - 97 >= this.n)),
420 )

423 }

425 switch (c) {

428 this.emit("board", this.n, "komi", half(this.komi));
429 break;
430 case "KOMI":
431 this.komi = komi(a[0]);

434 case "SUPERKO":

442 ix =

458 this.turn = "W";

462 break;
463 }

465 const r = this.check(a[0], p),
466 reason = a[0] !== this.turn ? "turn" : r.reason;
467 if (reason) this.emit("ILLEGAL", a[0], a[1], reason);
468 else {
469 const n = this.move(a[0], p, r);
470 this.emit(a[0], a[1], ...(n ? ["captures", n] : []));
471 }
472 break;
473 }
474 case "PASS":

476 else {
477 this.move(a[0], -1);

484 stop = [];
485 for (const r of records) {
486 const color = r[1],

490 97
