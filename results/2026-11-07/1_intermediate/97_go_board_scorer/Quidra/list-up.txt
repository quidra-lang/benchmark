bool has(int[] items, int target)
    for item := items[]
        if item == target
            return true
    return false
int[] unique(int[] items)
    int[] result := []
    for item := items.sorted()[]
        if not has(result, item)
            result = result.append(item)
    return result
class Reading(bool success, int point)
Reading defend(string[] board, int size, int point, int depth)
    if depth >= 40
        return Reading(true, -1)
    defender := board[point]
    attacker := other(defender)
    g := group(board, size, point)
    choices := g.liberties
    seen := array(#board, fill = false)
    for stone := g.stones[]
        for neighbor := adjacent(stone, size)[]
            if board[neighbor] == attacker and not seen[neighbor]
                enemy := group(board, size, neighbor)
                for member := enemy.stones[]
                    seen[member] = true
                if #enemy.liberties == 1
                    choices = choices.append(enemy.liberties[0])
    for q := unique(choices)[]
        candidate := placement(board, size, defender, q)
        if candidate.reason != ""
            continue
        m := int(#group(candidate.board, size, point).liberties)
        if m >= 3 or (m == 2 and not attack(candidate.board, size, point, depth + 1).success)
            return Reading(true, q)
    return Reading(false, -1)
Reading attack(string[] board, int size, int point, int depth)
    if depth >= 40
        return Reading(false, -1)
