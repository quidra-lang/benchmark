165 Ladder(String b,int n,int p){
166 this.n=n;
167 this.p=p;

171 Reading read(String b,int depth,boolean defending){
172 if(depth>=40)return new Reading(defending,-1);
173 ReadKey key=new ReadKey(b,depth,defending);
174 if(memo.containsKey(key))return memo.get(key);
175 Group g=group(b,n,p);
176 SortedSet<Integer>candidates=new TreeSet<>(g.liberties);
177 Set<Integer>seen=new HashSet<>();
178 if(defending)for(int q:g.stones)for(int t:adjacent(q,n))if(b.charAt(t)==attacker&&!seen.contains(t)){
179 Group e=group(b,n,t);
180 seen.addAll(e.stones);
181 if(e.liberties.size()==1)candidates.addAll(e.liberties);
182 }
183 for(int q:candidates){
184 Placement next=placement(b,n,defending?defender:attacker,q);
185 if(!next.reason.isEmpty())continue;
186 boolean success;
187 if(defending){
188 int count=group(next.board,n,p).liberties.size();
189 success=count>=3||(count==2&&!read(next.board,depth+1,false).success);
190 }else success=!read(next.board,depth+1,true).success;
191 if(success){
192 Reading r=new Reading(true,q);
193 memo.put(key,r);
194 return r;
195 }
196 }
197 Reading r=new Reading(false,-1);
198 memo.put(key,r);
199 return r;
200 }
201 LadderResult result(String b){
202 int count=group(b,n,p).liberties.size();
203 if(count>=3)return new LadderResult("safe",-1);
204 Reading r=read(b,0,count==1);
205 return new LadderResult((count==1?r.success:!r.success)?"escapes":"captured",r.move);
206 }

568 case "SCORE"->score(a[0]);
569 }
