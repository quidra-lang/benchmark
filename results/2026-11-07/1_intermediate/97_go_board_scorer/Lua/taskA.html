81local function komi(s)
82 local unsigned = s:sub(1, 1) == "-" and s:sub(2) or s
83 local whole, fraction = unsigned:match("^(%d+)%.([05])$")
84 if not whole then
85 whole = unsigned:match("^%d+$")
86 fraction = "0"
87 end
88 if not whole or (#whole > 1 and whole:sub(1, 1) == "0") or tonumber(whole) > 100 then
89 return nil
90 end
91 local x = 2 * tonumber(whole) + (fraction == "5" and 1 or 0)
92 if s:sub(1, 1) == "-" then
93 if x == 0 then
94 return nil
95 end
96 return -x
97 end
98 return x
99end
100local function adjacent(p, n, diagonal)
101 local r, c = (p - 1) // n, (p - 1) % n
102 local a = {}
103 local steps =
104 diagonal and {{-1, -1}, {-1, 1}, {1, -1}, {1, 1}} or {{-1, 0}, {1, 0}, {0, -1}, {0, 1}}
105 for _, step in ipairs(steps) do
106 local dr, dc = step[1], step[2]
107 if r + dr >= 0 and r + dr < n and c + dc >= 0 and c + dc < n then
108 a[#a + 1] = (r + dr) * n + c + dc + 1
109 end
110 end
111 return a
112end
113local function flood(b, n, p, allowed)
114 local seen, todo = {[p] = true}, {p}
115 while #todo > 0 do
116 local s = table.remove(todo)
117 for _, q in ipairs(adjacent(s, n)) do
118 if not seen[q] and allowed:find(b[q], 1, true) then
119 seen[q] = true
120 todo[#todo + 1] = q
121 end
122 end
123 end
124 return seen
125end
126local function group(b, n, p)
127 local stones, liberties = flood(b, n, p, b[p]), {}
128 for q in pairs(stones) do
129 for _, t in ipairs(adjacent(q, n)) do
130 if b[t] == "." then
131 liberties[t] = true
132 end
133 end
134 end
135 return {color = b[p], first = ordered(stones)[1], stones = stones, liberties = liberties}
136end
137local function groups(b, n, color)
138 local out, seen = {}, {}
139 for p = 1, #b do
140 if b[p] ~= "." and not seen[p] and (not color or b[p] == color) then
141 local g = group(b, n, p)
142 merge(seen, g.stones)
143 out[#out + 1] = g
144 end
145 end
146 return out
147end
148local function placement(b, n, c, p)
149 if b[p] ~= "." then
150 return b, 0, "occupied"
151 end
152 local updated = copy(b)
153 updated[p] = c
154 local captured, seen = 0, {}
155 for _, q in ipairs(adjacent(p, n)) do
156 if updated[q] == other(c) and not seen[q] then
157 local g = group(updated, n, q)
158 merge(seen, g.stones)
159 if size(g.liberties) == 0 then
160 for t in pairs(g.stones) do
161 updated[t] = "."
162 captured = captured + 1
163 end
164 end
165 end
166 end
167 return updated, captured, size(group(updated, n, p).liberties) == 0 and "suicide" or ""
168end
169local function regions(b, n)
170 local out, seen = {}, {}
171 for p = 1, #b do
172 if b[p] == "." and not seen[p] then
173 local points = flood(b, n, p, ".")
174 merge(seen, points)
175 local border = {}
176 for q in pairs(points) do
177 for _, t in ipairs(adjacent(q, n)) do
178 if b[t] ~= "." then
179 border[b[t]] = true
180 end
181 end
182 end
183 local owner = "neutral"
184 if size(border) == 1 then
185 for c in pairs(border) do
186 owner = c
187 end
188 end
189 out[#out + 1] = {first = p, points = points, owner = owner}
190 end
191 end
192 return out
193end
194local function safe_groups(b, n, c)
195 local own, areas, seen = groups(b, n, c), {}, {}
196 for p = 1, #b do
197 if b[p] ~= c and not seen[p] then
198 local area = flood(b, n, p, "." .. other(c))
199 merge(seen, area)
200 local empty = {}
201 for q in pairs(area) do
202 if b[q] == "." then
203 empty[q] = true
204 end
205 end
206 local touch, vital = {}, {}
207 for i, g in ipairs(own) do
208 local touching = false
209 for q in pairs(area) do
210 for _, t in ipairs(adjacent(q, n)) do
211 if g.stones[t] then
212 touching = true
213 end
214 end
215 end
216 if touching then
217 touch[i] = true
218 end
219 if subset(empty, g.liberties) then
220 vital[i] = true
221 end
222 end
223 areas[#areas + 1] = {touch = touch, vital = vital}
224 end
225 end
226 local s, t = {}, {}
227 for i in ipairs(own) do
228 s[i] = true
229 end
230 for j in ipairs(areas) do
231 t[j] = true
232 end
233 while true do
234 local before = size(s) + size(t)
235 local ns, nt = {}, {}
236 for i in pairs(s) do
237 local count = 0
238 for j in pairs(t) do
239 if areas[j].vital[i] then
240 count = count + 1
241 end
242 end
243 if count >= 2 then
244 ns[i] = true
245 end
246 end
247 s = ns
248 for j in pairs(t) do
249 if subset(areas[j].touch, s) then
250 nt[j] = true
251 end
252 end
253 t = nt
254 if before == size(s) + size(t) then
255 break
256 end
257 end
258 local out = {}
259 for _, i in ipairs(ordered(s)) do
260 out[#out + 1] = own[i]
261 end
262 return out
263end
264local function ladder(b, n, p)
265 local defender, attacker = b[p], other(b[p])
266 local memo = {}
267 local read
268 read = function(current, depth, defending)
269 if depth >= 40 then
270 return defending, -1
271 end
272 local key = table.concat(current) .. ":" .. depth .. ":" .. tostring(defending)
273 if memo[key] then
274 return table.unpack(memo[key])
275 end
276 local g = group(current, n, p)
277 local candidates, seen = copy(g.liberties), {}
278 if defending then
279 for q in pairs(g.stones) do
280 for _, t in ipairs(adjacent(q, n)) do
281 if current[t] == attacker and not seen[t] then
282 local e = group(current, n, t)
283 merge(seen, e.stones)
284 if size(e.liberties) == 1 then
285 merge(candidates, e.liberties)
286 end
287 end
288 end
289 end
290 end
291 for _, q in ipairs(ordered(candidates)) do
292 local updated, _, reason = placement(current, n, defending and defender or attacker, q)
293 if reason == "" then
294 local success
295 if defending then
296 local count = size(group(updated, n, p).liberties)
297 success = count >= 3 or count == 2 and not read(updated, depth + 1, false)
298 else
299 success = not read(updated, depth + 1, true)
300 end
301 if success then
302 memo[key] = {true, q}
303 return true, q
304 end
305 end
306 end
307 memo[key] = {false, -1}
308 return false, -1
309 end
310 local count = size(group(b, n, p).liberties)
311 if count >= 3 then
312 return "safe", -1
313 end
314 local success, move = read(b, 0, count == 1)
315 local escapes = count == 1 and success or count ~= 1 and not success
316 return escapes and "escapes" or "captured", move
317end

320function Game:reset(n, k)
321 self.n, self.komi, self.board, self.turn = n, k, {}, "B"
322 for p = 1, n * n do
323 self.board[p] = "."
324 end
325 self.captures, self.passes, self.over = {B = 0, W = 0}, 0, false
326 self.handicap, self.history, self.positions, self.dead = {}, {}, {self.board}, {}
327end
328function Game:point(s)
329 local letter, row = s:match("^([A-HJ-T])([1-9]%d?)$")
330 if not letter then
331 return -1
332 end
333 local c = columns:find(letter, 1, true)
334 local r = tonumber(row)
335 if c > self.n or r > self.n then
336 return -1
337 end
338 return (r - 1) * self.n + c
339end
340function Game:name(p)
341 local c = (p - 1) % self.n + 1
342 return columns:sub(c, c) .. ((p - 1) // self.n + 1)
343end
344function Game:sgf(p)
345 return string.char(97 + (p - 1) % self.n, 97 + self.n - 1 - (p - 1) // self.n)
346end
347function Game:check(c, p)
348 local b, captured, reason = placement(self.board, self.n, c, p)
349 local text = table.concat(b)
350 if
351 reason == "" and #self.positions >= 2 and
352 text == table.concat(self.positions[#self.positions - 1])
353 then
354 reason = "ko"
355 end
356 if reason == "" and self.superko then
357 for _, old in ipairs(self.positions) do
358 if text == table.concat(old) then
359 reason = "superko"
360 break
361 end
362 end
363 end
364 return b, captured, reason
365end
366function Game:move(c, p, checked)
367 local before = {
368 board = self.board,
369 turn = self.turn,
370 captures = copy(self.captures),
371 passes = self.passes,
372 over = self.over
373 }
374 local captured = 0
375 if p < 0 then
376 self.passes = self.passes + 1
377 if self.passes == 2 then
378 self.over = true
379 end
380 else
381 local b, count
382 if checked then
383 b, count = checked[1], checked[2]
384 else
385 b, count = self:check(c, p)
386 end
387 self.board = b
388 captured = count
389 self.captures[c] = self.captures[c] + captured
390 self.passes = 0
391 end
392 self.turn = other(c)
393 self.history[#self.history + 1] = {color = c, point = p, captured = captured, before = before}
394 self.positions[#self.positions + 1] = self.board
395 self.dead = {}
396 return captured
397end
398function Game:clean()

400 for p in pairs(self.dead) do

402 end

409 if self.n == 19 or i <= 4 or i == 9 then

412 end

414 for _, r in ipairs(regions(b, self.n)) do

417 end
418 end
419 end

421 if c ~= "." then
422 local letter = c == "B" and "X" or "O"
423 chars[p] = not map and self.dead[p] and letter:lower() or letter
424 elseif not map then
425 chars[p] = stars[p] and "+" or "."
426 end
427 end

429 for c = 1, self.n do
430 letters[c] = columns:sub(c, c)

434 for r = self.n, 1, -1 do

437 a[#a + 1] = chars[(r - 1) * self.n + c]
438 end

442end
443function Game:score(method)
444 local b, territory, total = self:clean(), {B = 0, W = 0}, {}
445 for _, r in ipairs(regions(b, self.n)) do
446 if territory[r.owner] then
447 territory[r.owner] = territory[r.owner] + size(r.points)
448 end
449 end
450 for _, c in ipairs({"B", "W"}) do
451 local fields = {c}
452 if method == "AREA" then
453 local stones = 0
454 for _, v in ipairs(b) do
455 if v == c then
456 stones = stones + 1
457 end
458 end
459 total[c] = 2 * (stones + territory[c]) + (c == "W" and self.komi or 0)
460 fields = {c, "stones", stones, "territory", territory[c]}
461 else
462 local prisoners = self.captures[c]
463 for p in pairs(self.dead) do
464 if self.board[p] == other(c) then
465 prisoners = prisoners + 1
466 end
467 end
468 total[c] = 2 * (territory[c] + prisoners) + (c == "W" and self.komi or 0)
469 fields = {c, "territory", territory[c], "prisoners", prisoners}
470 end
471 if c == "W" then
472 fields[#fields + 1] = "komi"
473 fields[#fields + 1] = half(self.komi)
474 end
475 fields[#fields + 1] = "total"
476 fields[#fields + 1] = half(total[c])
477 emit(table.unpack(fields))
478 end
479 local d = total.B - total.W
480 emit("result", d > 0 and "B+" .. half(d) or d < 0 and "W+" .. half(-d) or "Draw")
481end
482function Game:execute(line)
483 local words = {}
484 for word in line:gmatch("[^ ]+") do
485 words[#words + 1] = word
486 end
487 local c = table.remove(words, 1)
488 local a = words
489 if arity[c] == nil then
490 emit("ERROR unknown command")
491 return
492 end
493 if #a ~= arity[c] then
494 emit("ERROR bad arguments")
495 return
496 end
497 local valid = true
498 if c == "BOARD" then
499 valid = (a[1] == "9" or a[1] == "13" or a[1] == "19") and komi(a[2]) ~= nil
500 elseif c == "KOMI" then
501 valid = komi(a[1]) ~= nil
502 elseif c == "SUPERKO" then
503 valid = a[1] == "ON" or a[1] == "OFF"
504 elseif c == "HANDICAP" then
505 valid = a[1]:match("^[2-9]$") ~= nil
506 elseif c == "PLAY" or c == "PASS" or c == "LEGAL" or c == "EYES" or c == "SAFE" then
507 valid = a[1] == "B" or a[1] == "W"
508 elseif c == "SCORE" then
509 valid = a[1] == "AREA" or a[1] == "TERRITORY"
510 end
511 if not valid then
512 emit("ERROR bad arguments")
513 return
514 end
515 local p = -1
516 if c == "PLAY" or c == "GROUP" or c == "LADDER" or c == "DEAD" then
517 p = self:point(a[c == "PLAY" and 2 or 1])
518 if p < 0 then
519 emit("ERROR bad point")
520 return
521 end
522 end
523 local records = {}
524 if c == "REPLAY" then
525 local at = 1
526 while at <= #a[1] do
527 local s, e, color, text = a[1]:find(";([BW])%[([a-z]*)%]", at)
528 if
529 s ~= at or
530 (#text ~= 0 and
531 (#text ~= 2 or text:byte(1) - 97 >= self.n or text:byte(2) - 97 >= self.n))
532 then
533 emit("ERROR bad sequence")
534 return
535 end
536 records[#records + 1] = {
537 color = color,
538 point = #text == 0 and -1 or
539 (self.n - 1 - (text:byte(2) - 97)) * self.n + text:byte(1) - 96
540 }
541 at = e + 1
542 end
543 if #records == 0 then
544 emit("ERROR bad sequence")
545 return
546 end
547 end
548 if (c == "PLAY" or c == "PASS" or c == "REPLAY") and self.over then
549 emit("ERROR game over")
550 return
551 end
552 if c == "BOARD" then
553 self:reset(tonumber(a[1]), komi(a[2]))
554 emit("board", self.n, "komi", half(self.komi))
555 elseif c == "KOMI" then
556 self.komi = komi(a[1])
557 emit("komi", half(self.komi))
558 elseif c == "SUPERKO" then
559 self.superko = a[1] == "ON"
560 emit("superko", self.superko and "on" or "off")
561 elseif c == "HANDICAP" then
562 if #self.history > 0 then
563 emit("ERROR handicap not allowed")
564 return
565 end
566 for _, v in ipairs(self.board) do
567 if v ~= "." then
568 emit("ERROR handicap not allowed")
569 return
570 end
571 end
572 local k = tonumber(a[1])
573 local ix = k == 5 and {1, 2, 3, 4, 9} or k == 7 and {1, 2, 3, 4, 5, 6, 9} or {}
574 if #ix == 0 then
575 for i = 1, k do
576 ix[i] = i
577 end
578 end
579 self.board = copy(self.board)
580 self.handicap = {}
581 local fields = {"handicap", k}
582 for _, i in ipairs(ix) do
583 local q = self:point(handicap_points[self.n][i])
584 self.handicap[#self.handicap + 1] = q
585 self.board[q] = "B"
586 fields[#fields + 1] = self:name(q)
587 end
588 self.turn = "W"
589 self.positions = {self.board}
590 self.dead = {}
591 emit(table.unpack(fields))
592 elseif c == "PLAY" then
593 local checked = {self:check(a[1], p)}
594 local reason = a[1] ~= self.turn and "turn" or checked[3]
595 if reason ~= "" then
596 emit("ILLEGAL", a[1], a[2], reason)
597 else
598 local captured = self:move(a[1], p, checked)
599 if captured == 0 then
600 emit(a[1], a[2])
601 else
602 emit(a[1], a[2], "captures", captured)
603 end
604 end
605 elseif c == "PASS" then
606 if a[1] ~= self.turn then
607 emit("ILLEGAL", a[1], "pass turn")
608 else
609 self:move(a[1], -1)
610 emit(a[1], "pass")
611 if self.over then
612 emit("game over")
613 end
614 end
615 elseif c == "REPLAY" then
616 local count, stop = 0, {}
617 for _, r in ipairs(records) do
618 local checked
619 if r.point >= 0 then
620 checked = {self:check(r.color, r.point)}
621 end
622 local reason =
623 self.over and "game over" or r.color ~= self.turn and "turn" or
624 checked and checked[3] or
625 ""
626 if reason ~= "" then
627 stop = {"stopped", r.color, r.point < 0 and "pass" or self:name(r.point), reason}
628 break
629 end
630 self:move(r.color, r.point, checked)
631 count = count + 1
632 end
633 emit("replay", count, "moves", table.unpack(stop))
634 if self.over then
635 emit("game over")
636 end
637 elseif c == "UNDO" then
638 if #self.history == 0 then
639 emit("ERROR nothing to undo")
640 return
641 end
642 local m = table.remove(self.history)
643 for k, v in pairs(m.before) do
644 self[k] = v
645 end
646 table.remove(self.positions)
647 self.dead = {}
648 emit("undo", m.color, m.point < 0 and "pass" or self:name(m.point))
649 elseif c == "SHOW" or c == "MAP" then
650 self:show(c == "MAP")
651 elseif c == "STATUS" then
652 emit(
653 "size",
654 self.n,
655 "komi",
656 half(self.komi),
657 "handicap",
658 #self.handicap,
659 "superko",
660 self.superko and "on" or "off"
661 )
662 if self.over then
663 emit("moves", #self.history, "game over")
664 else
665 emit("moves", #self.history, "next", self.turn, "passes", self.passes)
666 end
667 elseif c == "CAPTURES" then
668 emit("captures B", self.captures.B, "W", self.captures.W)
669 elseif c == "HISTORY" then
670 emit("history", #self.history)
671 for i, m in ipairs(self.history) do
672 local fields = {i, m.color, m.point < 0 and "pass" or self:name(m.point)}
673 if m.captured > 0 then
674 fields[#fields + 1] = "captures"
675 fields[#fields + 1] = m.captured
676 end
677 emit(table.unpack(fields))
678 end
679 elseif c == "MOVES" then
680 local s = "(;GM[1]FF[4]SZ[" .. self.n .. "]KM[" .. half(self.komi) .. "]"
681 if #self.handicap > 0 then
682 s = s .. "HA[" .. #self.handicap .. "]AB"
683 for _, q in ipairs(self.handicap) do
684 s = s .. "[" .. self:sgf(q) .. "]"
685 end
686 end
687 for _, m in ipairs(self.history) do
688 s = s .. ";" .. m.color .. "[" .. (m.point < 0 and "" or self:sgf(m.point)) .. "]"
689 end
690 emit(s .. ")")
691 elseif c == "GROUP" or c == "DEAD" or c == "LADDER" then
692 if self.board[p] == "." then
693 emit("ERROR no stone")
694 return
695 end
696 local g = group(self.board, self.n, p)
697 if c == "GROUP" then
698 emit(
699 "group",
700 g.color,
701 self:name(g.first),
702 "size",
703 size(g.stones),
704 "liberties",
705 size(g.liberties)
706 )
707 local stones, liberties = {"stones"}, {"liberties"}
708 for _, q in ipairs(ordered(g.stones)) do
709 stones[#stones + 1] = self:name(q)
710 end
711 for _, q in ipairs(ordered(g.liberties)) do
712 liberties[#liberties + 1] = self:name(q)
713 end
714 emit(table.unpack(stones))
715 emit(table.unpack(liberties))
716 elseif c == "DEAD" then
717 local alive = self.dead[p]
718 for q in pairs(g.stones) do
719 self.dead[q] = not alive and true or nil
720 end
721 emit(alive and "alive" or "dead", g.color, self:name(g.first), "size", size(g.stones))
722 else
723 local status, q = ladder(self.board, self.n, p)
724 if q < 0 then
725 emit("ladder", a[1], status)
726 else
727 emit("ladder", a[1], status, self:name(q))
728 end
729 end
730 elseif c == "GROUPS" then
731 local gs = groups(self.board, self.n)
732 emit("groups", #gs)
733 for _, g in ipairs(gs) do
734 local fields = {
735 g.color,
736 self:name(g.first),
737 "size",
738 size(g.stones),
739 "liberties",
740 size(g.liberties)
741 }
742 if self.dead[g.first] then
743 fields[#fields + 1] = "dead"
744 end
