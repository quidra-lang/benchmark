32def parse_komi(text):
33 if not re.fullmatch(r"-?(0|[1-9][0-9]*)(?:\.[05])?", text):
34 return None
35 negative = text.startswith("-")
36 unsigned = text[1:] if negative else text
37 components = unsigned.split(".")
38 integer = int(components[0])
39 if integer > 100:
40 return None
41 fraction = int(components[1]) if len(components) > 1 else 0
42 result = 2 * integer + (fraction // 5)
43 if negative and result == 0:
44 return None
45 return -result if negative else result

48def point_index(text, n):
49 if not re.fullmatch(r"[A-HJ-T](?:[1-9]|1[0-9])", text):
50 return None
51 column = COLUMNS.find(text[0])
52 row = int(text[1:])
53 if not (0 <= column < n and 1 <= row <= n):
54 return None
55 return (row - 1) * n + column

96def flood(board, n, root, allowed):
97 visited = {root}
98 pending = [root]
99 while pending:
100 position = pending.pop()
101 for neighbor in adjacent(position, n):
102 if neighbor not in visited and board[neighbor] in allowed:
103 visited.add(neighbor)
104 pending.append(neighbor)
105 return visited

131def placement(board, n, color, position):
132 if board[position] != ".":
133 return None, 0, "occupied"
134 enemy = other(color)
135 next_board = list(board)
136 next_board[position] = color
137 visited = set()
138 captured = 0
139 for neighbor in adjacent(position, n):
140 if next_board[neighbor] != enemy or neighbor in visited:
141 continue
142 stones, liberties = group(next_board, n, neighbor)
143 visited.update(stones)
144 if not liberties:
145 for stone in stones:
146 next_board[stone] = "."
147 captured += len(stones)
148 next_board = "".join(next_board)
149 if not group(next_board, n, position)[1]:
150 return None, 0, "suicide"
151 return next_board, captured, None

154def regions(board, n):

158 if value != "." or position in visited:

170def eyes(board, n, color):
171 found = []
172 for position, value in enumerate(board):
173 if value != ".":
174 continue
175 neighbors = adjacent(position, n)
176 if all(board[neighbor] == color for neighbor in neighbors):
177 enemy = other(color)
178 enemy_diagonals = sum(board[d] == enemy for d in diagonal(position, n))
179 real = (enemy_diagonals <= 1 if len(neighbors) == 4
180 else enemy_diagonals == 0)
181 found.append((position, real))
182 return found

185def unconditionally_alive(board, n, color):
186 own = groups(board, n, color)
187 remaining_groups = set(range(len(own)))
188 opponent_areas = []
189 visited = set()
190 for position, disc in enumerate(board):
191 if disc == color or position in visited:
192 continue
193 area = flood(board, n, position, set(".BW") - {color})
194 visited.update(area)
195 opponent_areas.append(area)
196
197 touch = []
198 vital = []
199 for area in opponent_areas:
200 touching = set()
201 vital_for = set()
202 empty = {position for position in area if board[position] == "."}
203 for i, (_, _, stones, liberties) in enumerate(own):
204 if any(any(neighbor in stones for neighbor in adjacent(position, n))
205 for position in area):
206 touching.add(i)
207 if empty <= liberties:
208 vital_for.add(i)
209 touch.append(touching)
210 vital.append(vital_for)
211
212 remaining_areas = set(range(len(opponent_areas)))
213 while True:
214 previous = (len(remaining_groups), len(remaining_areas))
215 remaining_groups = {i for i in remaining_groups
216 if sum(i in vital[j] for j in remaining_areas) >= 2}
217 remaining_areas = {j for j in remaining_areas
218 if touch[j] <= remaining_groups}
219 if previous == (len(remaining_groups), len(remaining_areas)):
220 break
221 return [own[i] for i in sorted(remaining_groups)]

244 if depth >= 40:
245 return True, None

247 candidates = set(liberties)
248 checked = set()
249 for stone in stones:
250 for neighbor in adjacent(stone, n):

259 if reason:

263 return True, move
264 if len(next_liberties) == 2 and not attack(updated, depth + 1)[0]:
265 return True, move

403 def show_score(self, method):
404 board = self.no_dead_board()
405 territory = {"B": 0, "W": 0}
406 for _, points, owner in regions(board, self.n):
407 if owner in territory:
408 territory[owner] += len(points)

410 if method == "AREA":

412 w_total = 2 * (stones["W"] + territory["W"]) + self.komi2

414 "total", half_text(b_total))

417 else:
418 prisoners = {
419 "B": self.prisoners["B"] + sum(self.board[p] == "W" for p in self.dead),
420 "W": self.prisoners["W"] + sum(self.board[p] == "B" for p in self.dead),

422 b_total = 2 * (territory["B"] + prisoners["B"])
423 w_total = 2 * (territory["W"] + prisoners["W"]) + self.komi2

434 records = re.findall(r";([BW])\[([a-z]*)\]", text)
435 played = 0
436 stopped = None
437 for color, coordinate in records:
438 point = sgf_position(coordinate, self.n) if coordinate else None
439 label = self.point_text(point) if point is not None else "pass"
440 reason = ("game over" if self.over else
441 "turn" if color != self.turn else None)
442 checked = None
443 if reason is None and point is not None:
444 checked = self.check_placement(color, point)
445 reason = checked[2]
446 if reason:
447 stopped = (color, label, reason)

449 self.move(color, point, checked)
450 played += 1
451 if stopped:
452 self.emit("replay", played, "moves", "stopped", *stopped)
453 else:
454 self.emit("replay", played, "moves")

472 elif name == "KOMI":

637 def execute(self, line):
638 name, *a = line.split(" ")
639 if name not in ARITY:
640 self.emit("ERROR", "unknown command")
641 return
642 if len(a) != ARITY[name]:
643 self.emit("ERROR", "bad arguments")
644 return
645 valid = True
646 if name == "BOARD":
647 valid = a[0] in ("9", "13", "19") and parse_komi(a[1]) is not None
648 elif name == "KOMI":
649 valid = parse_komi(a[0]) is not None
650 elif name == "SUPERKO":
651 valid = a[0] in ("ON", "OFF")
652 elif name == "HANDICAP":
653 valid = a[0] in tuple(str(i) for i in range(2, 10))
654 elif name in ("PLAY", "PASS", "LEGAL", "EYES", "SAFE"):
655 valid = a[0] in ("B", "W")
656 elif name == "SCORE":
657 valid = a[0] in ("AREA", "TERRITORY")
658 if not valid:
659 self.emit("ERROR", "bad arguments")
660 return
661 if name in POINT_ARGUMENTS and point_index(a[POINT_ARGUMENTS[name]], self.n) is None:
662 self.emit("ERROR", "bad point")
663 return
664 if name == "REPLAY":
665 valid = re.fullmatch(r"(?:;[BW]\[[a-z]{0,2}\])+", a[0])
