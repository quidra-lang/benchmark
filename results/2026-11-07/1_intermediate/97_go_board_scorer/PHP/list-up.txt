function ladder(string $b, int $n, int $p): array
    $defender = $b[$p];
    $attacker = other($defender);
    $read = function (string $current, int $depth, bool $defending) use (
        &$read,
        &$memo,
        $defender,
        $attacker,
        $n,
        $p,
    ): array {
        if ($depth >= 40) {
            return [$defending, -1];
        }
        $g = group($current, $n, $p);
        $candidates = $g["liberties"];
        $seen = [];
        if ($defending) {
            foreach ($g["stones"] as $q => $_) {
                foreach (adjacent($q, $n) as $t) {
                    if ($current[$t] === $attacker && !isset($seen[$t])) {
                        $e = group($current, $n, $t);
                        $seen += $e["stones"];
                        if (count($e["liberties"]) === 1) {
                            $candidates += $e["liberties"];
                        }
                    }
                };
            };
        }
        foreach (ordered($candidates) as $q) {
            [$next, $_, $reason] = placement($current, $n, $defending ? $defender : $attacker, $q);
            if ($reason !== "") {
                continue;
            }
            if ($defending) {
                $m = count(group($next, $n, $p)["liberties"]);
                $success = $m >= 3 || ($m === 2 && !$read($next, $depth + 1, false)[0]);
            } else {
                $success = !$read($next, $depth + 1, true)[0];
            }
            if ($success) {
                return $memo[$key] = [true, $q];
            }
        }
        return $memo[$key] = [false, -1];
