44parse_komi <- function(s) {
45 if (!grepl('^-?(0|[1-9][0-9]*)(\\.[05])?$', s)) {
46 return(NA_real_)
47 }
48 parts <- strsplit(sub('^-', '', s), '.', fixed = TRUE)[[1]]
49 whole <- as.numeric(parts[1])
50 if (whole > 100) {
51 return(NA_real_)
52 }
53 value <- 2 * whole + if (length(parts) == 2 && parts[2] == '5') 1 else 0
54 if (startsWith(s, '-')) {
55 if (value == 0) {
56 return(NA_real_)
57 }
58 -value
59 } else {
60 value
61 }

63adjacent <- function(p, n, diagonal = FALSE) {
64 r <- (p - 1) %/% n
65 c <- (p - 1) %% n
66 steps <- if (diagonal) {
67 matrix(c(-1, -1, -1, 1, 1, -1, 1, 1), ncol = 2, byrow = TRUE)
68 } else {
69 matrix(c(-1, 0, 1, 0, 0, -1, 0, 1), ncol = 2, byrow = TRUE)
70 }
71 rr <- r + steps[, 1]
72 cc <- c + steps[, 2]
73 ok <- rr >= 0 & rr < n & cc >= 0 & cc < n
74 rr[ok] * n + cc[ok] + 1L
75}
76flood <- function(b, n, p, allowed) {
77 seen <- rep(FALSE, length(b))
78 seen[p] <- TRUE
79 todo <- p
80 while (length(todo)) {
81 q <- todo[length(todo)]
82 todo <- todo[-length(todo)]
83 for (t in adjacent(q, n)) {
84 if (!seen[t] && b[t] %in% allowed) {
85 seen[t] <- TRUE
86 todo <- c(todo, t)
87 }
88 }
89 }
90 which(seen)
91}

120placement <- function(b, n, c, p) {
121 if (b[p] != '.') {
122 return(list(board = b, captured = 0, reason = 'occupied'))
123 }
124 next_board <- b
125 next_board[p] <- c
126 captured <- 0
127 seen <- rep(FALSE, length(b))
128 for (q in adjacent(p, n)) {
129 if (next_board[q] == other(c) && !seen[q]) {
130 g <- group(next_board, n, q)
131 seen[g$stones] <- TRUE
132 if (!length(g$liberties)) {
133 next_board[g$stones] <- '.'
134 captured <- captured + length(g$stones)
135 }
136 }
137 }
138 list(
139 board = next_board,
140 captured = captured,
141 reason = if (!length(group(next_board, n, p)$liberties)) 'suicide' else ''
142 )

144regions <- function(b, n) {
145 out <- list()
146 seen <- rep(FALSE, length(b))
147 for (p in seq_along(b)) {
148 if (b[p] == '.' && !seen[p]) {
149 points <- flood(b, n, p, '.')
150 seen[points] <- TRUE
151 border <- unique(unlist(
152 lapply(points, function(q) b[adjacent(q, n)]),
153 use.names = FALSE
154 ))
155 border <- setdiff(border, '.')
156 out[[length(out) + 1]] <- list(
157 first = p,
158 points = points,
159 owner = if (length(border) == 1) border else 'neutral'
160 )
161 }
162 }
163 out

165safe_groups <- function(b, n, c) {
166 own <- groups(b, n, c)
167 areas <- list()
168 seen <- rep(FALSE, length(b))
169 for (p in seq_along(b)) {
170 if (b[p] != c && !seen[p]) {
171 area <- flood(b, n, p, c('.', other(c)))
172 seen[area] <- TRUE
173 empty <- area[b[area] == '.']
174 touch <- vital <- integer()
175 for (i in seq_along(own)) {
176 if (
177 any(vapply(
178 area,
179 function(q) any(adjacent(q, n) %in% own[[i]]$stones),
180 logical(1)
181 ))
182 ) {
183 touch <- c(touch, i)
184 }
185 if (all(empty %in% own[[i]]$liberties)) vital <- c(vital, i)
186 }
187 areas[[length(areas) + 1]] <- list(touch = touch, vital = vital)
188 }
189 }
190 s <- seq_along(own)
191 t <- seq_along(areas)
192 repeat {
193 before <- length(s) + length(t)
194 s <- s[vapply(
195 s,
196 function(i) {
197 sum(vapply(t, function(j) i %in% areas[[j]]$vital, logical(1))) >= 2
198 },
199 logical(1)
200 )]
201 t <- t[vapply(t, function(j) all(areas[[j]]$touch %in% s), logical(1))]
202 if (before == length(s) + length(t)) break
203 }
204 own[s]
205}
206ladder <- function(b, n, p) {
207 defender <- b[p]
208 attacker <- other(defender)
209 memo <- new.env(parent = emptyenv())
210 read_position <- function(current, depth, defending) {
211 if (depth >= 40) {
212 return(list(success = defending, move = 0L))
213 }
214 key <- paste0(paste(current, collapse = ''), ':', depth, ':', defending)
215 if (exists(key, envir = memo, inherits = FALSE)) {
216 return(memo[[key]])
217 }
218 g <- group(current, n, p)
219 candidates <- g$liberties
220 seen <- rep(FALSE, length(current))
221 if (defending) {
222 for (q in g$stones) {
223 for (t in adjacent(q, n)) {
224 if (current[t] == attacker && !seen[t]) {
225 e <- group(current, n, t)
226 seen[e$stones] <- TRUE
227 if (length(e$liberties) == 1) {
228 candidates <- union(candidates, e$liberties)
229 }
230 }
231 }
232 }
233 }
234 for (q in sort(candidates)) {
235 next_board <- placement(
236 current,
237 n,
238 if (defending) defender else attacker,
239 q
240 )
241 if (next_board$reason != '') {
242 next
243 }
244 if (defending) {
245 count <- length(group(next_board$board, n, p)$liberties)
246 success <- count >= 3 ||
247 (count == 2 &&
248 !read_position(next_board$board, depth + 1, FALSE)$success)
249 } else {
250 success <- !read_position(next_board$board, depth + 1, TRUE)$success
251 }
252 if (success) {
253 memo[[key]] <- list(success = TRUE, move = q)
254 return(memo[[key]])
255 }
256 }
257 memo[[key]] <- list(success = FALSE, move = 0L)
258 memo[[key]]
259 }
260 count <- length(group(b, n, p)$liberties)
261 if (count >= 3) {
262 return(list(status = 'safe', move = 0L))
263 }
264 result <- read_position(b, 0, count == 1)
265 list(
266 status = if (if (count == 1) result$success else !result$success) {
267 'escapes'
268 } else {
269 'captured'
270 },
271 move = result$move
272 )

397score <- function(g, method) {
398 b <- clean(g)
399 territory <- c(B = 0, W = 0)
400 total <- c(B = 0, W = 0)
401 for (r in regions(b, g$n)) {
402 if (r$owner != 'neutral') {
403 territory[r$owner] <- territory[r$owner] + length(r$points)
404 }
405 }
406 for (c in c('B', 'W')) {
407 if (method == 'AREA') {
408 stones <- sum(b == c)
409 total[c] <- 2 * (stones + territory[c]) + if (c == 'W') g$komi else 0
410 fields <- c(c, 'stones', stones, 'territory', territory[c])
411 } else {
412 prisoners <- g$captures[c] + sum(g$board[g$dead] == other(c))
413 total[c] <- 2 * (territory[c] + prisoners) + if (c == 'W') g$komi else 0
414 fields <- c(c, 'territory', territory[c], 'prisoners', prisoners)
415 }
416 if (c == 'W') {
417 fields <- c(fields, 'komi', half(g$komi))
418 }
419 emit(c(fields, 'total', half(total[c])))
420 }
421 d <- total['B'] - total['W']
422 emit(
423 'result',
424 if (d > 0) {
425 paste0('B+', half(d))
426 } else if (d < 0) {
427 paste0('W+', half(-d))
428 } else {
429 'Draw'
430 }
431 )

433execute <- function(g, line) {
434 words <- strsplit(line, ' ', fixed = TRUE)[[1]]
435 c <- words[1]
436 a <- words[-1]
437 if (!c %in% names(arity)) {
438 emit('ERROR unknown command')
439 return()
440 }
441 if (length(a) != arity[[c]]) {
442 emit('ERROR bad arguments')
443 return()
444 }
445 valid <- switch(
446 c,
447 BOARD = a[1] %in% c('9', '13', '19') && !is.na(parse_komi(a[2])),
448 KOMI = !is.na(parse_komi(a[1])),
449 SUPERKO = a[1] %in% c('ON', 'OFF'),
450 HANDICAP = grepl('^[2-9]$', a[1]),
451 PLAY = ,
452 PASS = ,
453 LEGAL = ,
454 EYES = ,
455 SAFE = a[1] %in% c('B', 'W'),
456 SCORE = a[1] %in% c('AREA', 'TERRITORY'),
457 TRUE
458 )
459 if (!valid) {
460 emit('ERROR bad arguments')
461 return()
462 }
463 p <- 0L
464 if (c %in% c('PLAY', 'GROUP', 'LADDER', 'DEAD')) {
465 p <- point(g, a[if (c == 'PLAY') 2 else 1])
466 if (p == 0) {
467 emit('ERROR bad point')
468 return()
469 }
470 }
471 records <- list()
472 if (c == 'REPLAY') {
473 if (!grepl('^(;[BW]\\[[a-z]{0,2}\\])+$', a[1])) {
474 emit('ERROR bad sequence')
475 return()
476 }
477 matches <- regmatches(a[1], gregexpr(';[BW]\\[[a-z]*\\]', a[1]))[[1]]
478 for (record in matches) {
479 text <- substr(record, 4, nchar(record) - 1)
480 if (
481 nchar(text) > 0 &&
482 (nchar(text) != 2 || any(utf8ToInt(text) - 97 >= g$n))
483 ) {
484 emit('ERROR bad sequence')
485 return()
486 }
487 q <- if (text == '') {
488 0
489 } else {
490 codes <- utf8ToInt(text)
491 (g$n - 1 - (codes[2] - 97)) * g$n + codes[1] - 96
492 }
493 records[[length(records) + 1]] <- list(
494 color = substr(record, 2, 2),
495 point = q
496 )
497 }
498 }
499 if (c %in% c('PLAY', 'PASS', 'REPLAY') && g$over) {
500 emit('ERROR game over')
501 return()
502 }
503 if (c == 'BOARD') {
504 reset(g, as.integer(a[1]), parse_komi(a[2]))
505 emit('board', g$n, 'komi', half(g$komi))
506 } else if (c == 'KOMI') {
507 g$komi <- parse_komi(a[1])
508 emit('komi', half(g$komi))
509 } else if (c == 'SUPERKO') {
510 g$superko <- a[1] == 'ON'
511 emit('superko', if (g$superko) 'on' else 'off')
512 } else if (c == 'HANDICAP') {
513 if (length(g$history) || any(g$board != '.')) {
514 emit('ERROR handicap not allowed')
515 return()
516 }
517 k <- as.integer(a[1])
518 indices <- if (k == 5) {
519 c(1, 2, 3, 4, 9)
520 } else if (k == 7) {
521 c(1, 2, 3, 4, 5, 6, 9)
522 } else {
523 seq_len(k)
524 }
525 g$handicap <- vapply(
526 handicap_points[[as.character(g$n)]][indices],
527 function(s) point(g, s),
528 numeric(1)
529 )
530 g$board[g$handicap] <- 'B'
531 g$turn <- 'W'
532 g$positions <- list(g$board)
533 g$dead <- integer()
534 emit('handicap', k, point_name(g, g$handicap))
535 } else if (c == 'PLAY') {
536 next_board <- check(g, a[1], p)
537 reason <- if (a[1] != g$turn) 'turn' else next_board$reason
538 if (reason != '') {
539 emit('ILLEGAL', a[1], a[2], reason)
540 } else {
541 captured <- play(g, a[1], p, next_board)
542 emit(
543 a[1],
544 a[2],
545 if (captured > 0) c('captures', captured) else character()
546 )
547 }
548 } else if (c == 'PASS') {
549 if (a[1] != g$turn) {
550 emit('ILLEGAL', a[1], 'pass turn')
551 } else {
552 play(g, a[1], 0)
553 emit(a[1], 'pass')
554 if (g$over) emit('game over')
555 }
556 } else if (c == 'REPLAY') {
557 played <- 0
558 stopped <- character()
559 for (record in records) {
560 col <- record$color
561 q <- record$point
562 checked <- if (q == 0) NULL else check(g, col, q)
563 reason <- if (g$over) {
564 'game over'
565 } else if (col != g$turn) {
566 'turn'
567 } else if (is.null(checked)) {
568 ''
569 } else {
570 checked$reason
571 }
572 if (reason != '') {
573 stopped <- c(
574 'stopped',
575 col,
576 if (q == 0) 'pass' else point_name(g, q),
577 reason
578 )
579 break
580 }
581 play(g, col, q, checked)
582 played <- played + 1
583 }
584 emit('replay', played, 'moves', stopped)
585 if (g$over) emit('game over')
586 } else if (c == 'UNDO') {
587 if (!length(g$history)) {
588 emit('ERROR nothing to undo')
589 return()
590 }
591 m <- g$history[[length(g$history)]]
592 g$history <- g$history[-length(g$history)]
593 for (key in names(m$before)) {
594 g[[key]] <- m$before[[key]]
595 }
596 g$positions <- g$positions[-length(g$positions)]
597 g$dead <- integer()
598 emit('undo', m$color, if (m$point == 0) 'pass' else point_name(g, m$point))
599 } else if (c %in% c('SHOW', 'MAP')) {
600 show_board(g, c == 'MAP')
601 } else if (c == 'STATUS') {
602 emit(
603 'size',
604 g$n,
605 'komi',
606 half(g$komi),
607 'handicap',
608 length(g$handicap),
609 'superko',
610 if (g$superko) 'on' else 'off'
611 )
612 emit(
613 'moves',
614 length(g$history),
615 if (g$over) 'game over' else c('next', g$turn, 'passes', g$passes)
616 )
617 } else if (c == 'CAPTURES') {
618 emit('captures B', g$captures['B'], 'W', g$captures['W'])
619 } else if (c == 'HISTORY') {
620 emit('history', length(g$history))
621 for (i in seq_along(g$history)) {
622 m <- g$history[[i]]
623 emit(
624 i,
625 m$color,
626 if (m$point == 0) 'pass' else point_name(g, m$point),
627 if (m$captured > 0) c('captures', m$captured) else character()
628 )
629 }
630 } else if (c == 'MOVES') {
631 s <- paste0('(;GM[1]FF[4]SZ[', g$n, ']KM[', half(g$komi), ']')
632 if (length(g$handicap)) {
633 s <- paste0(s, 'HA[', length(g$handicap), ']AB')
634 for (q in g$handicap) {
635 s <- paste0(s, '[', sgf(g, q), ']')
636 }
637 }
638 for (m in g$history) {
639 s <- paste0(
640 s,
641 ';',
642 m$color,
643 '[',
644 if (m$point == 0) '' else sgf(g, m$point),
645 ']'
646 )
647 }
648 emit(paste0(s, ')'))
649 } else if (c %in% c('GROUP', 'DEAD', 'LADDER')) {
650 if (g$board[p] == '.') {
651 emit('ERROR no stone')
652 return()
653 }
654 gr <- group(g$board, g$n, p)
655 if (c == 'GROUP') {
656 emit(
657 'group',
658 gr$color,
659 point_name(g, gr$first),
660 'size',
661 length(gr$stones),
662 'liberties',
663 length(gr$liberties)
664 )
665 emit('stones', point_name(g, gr$stones))
666 emit('liberties', point_name(g, gr$liberties))
667 } else if (c == 'DEAD') {
668 alive <- p %in% g$dead
669 g$dead <- if (alive) {
670 setdiff(g$dead, gr$stones)
671 } else {
672 union(g$dead, gr$stones)
673 }
674 emit(
675 if (alive) 'alive' else 'dead',
676 gr$color,
677 point_name(g, gr$first),
678 'size',
679 length(gr$stones)
680 )
681 } else {
682 result <- ladder(g$board, g$n, p)
683 emit(
684 'ladder',
685 a[1],
686 result$status,
687 if (result$move == 0) character() else point_name(g, result$move)
688 )
689 }
690 } else if (c == 'GROUPS') {
691 gs <- groups(g$board, g$n)
692 emit('groups', length(gs))
693 for (gr in gs) {
694 emit(
695 gr$color,
696 point_name(g, gr$first),
697 'size',
698 length(gr$stones),
699 'liberties',
700 length(gr$liberties),
701 if (gr$first %in% g$dead) 'dead' else character()
702 )
703 }
704 } else if (c == 'ATARI') {
705 gs <- Filter(function(gr) length(gr$liberties) == 1, groups(g$board, g$n))
706 if (!length(gs)) {
707 emit('atari none')
708 }
709 for (gr in gs) {
710 emit(
711 'atari',
712 gr$color,
713 point_name(g, gr$first),
714 'size',
715 length(gr$stones),
716 'at',
717 point_name(g, gr$liberties)
718 )
719 }
720 } else if (c == 'LEGAL') {
721 rejected <- list()
722 count <- 0
723 for (q in seq_along(g$board)) {
724 if (g$board[q] == '.') {
725 why <- check(g, a[1], q)$reason
726 if (why == '') {
727 count <- count + 1
728 } else {
729 rejected[[length(rejected) + 1]] <- c(point_name(g, q), why)
730 }
731 }
732 }
733 emit('legal', a[1], count)
734 for (row in rejected) {
735 emit(row)
736 }
737 } else if (c == 'EYES') {
738 found <- list()

740 if (g$board[q] == '.') {
741 ns <- adjacent(q, g$n)
742 if (all(g$board[ns] == a[1])) {
743 enemies <- sum(g$board[adjacent(q, g$n, TRUE)] == other(a[1]))
744 found[[length(found) + 1]] <- c(

746 if (enemies <= if (length(ns) == 4) 1 else 0) 'real' else 'false'
747 )
748 }
749 }
750 }

752 for (row in found) {

754 }
755 } else if (c == 'SAFE') {
756 gs <- safe_groups(g$board, g$n, a[1])

761 } else if (c == 'REGIONS') {

763 emit('regions', length(rs))
764 for (r in rs) {
765 emit(point_name(g, r$first), 'size', length(r$points), r$owner)
766 }
767 } else if (c == 'SCORE') {

769 }
