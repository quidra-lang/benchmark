11private fun stem(pattern: String, word: String): String? {
12 val i = pattern.indexOf('%')
13 if (i < 0) return if (pattern == word) "" else null
14 val a = pattern.take(i)
15 val b = pattern.drop(i+1)
16 return if (word.length >= a.length+b.length && word.startsWith(a) && word.endsWith(b)) word.substring(a.length,word.length-b.length) else null
17}

19private fun substitute(s: String,m: String): String {
20 val i = s.indexOf('%')

28private fun wildcard(pattern: String, word: String): Boolean {
29 var previous = BooleanArray(word.length+1)
30 previous[0] = true
31 for (c in pattern) {
32 val next = BooleanArray(word.length+1)
33 if (c == '*') next[0] = previous[0]
34 for (j in 1..word.length) next[j] = if (c == '*') previous[j] || (word[j-1] != '/' && next[j-1]) else previous[j-1] && (if (c == '?') word[j-1] != '/' else c == word[j-1])
35 previous = next
36 }
37 return previous.last()
38}

93 fun closing(s: String,start: Int): Int {
94 val opening = s[start]
95 val close = if (opening == '(') ')' else '}'
96 var depth = 0
97 for (i in start until s.length) {
98 if (s[i] == opening) depth++
99 if (s[i] == close) {
100 depth--
101 if (depth == 0) return i
102 }
103 }
104 error("unterminated variable reference")
105 }
106 fun outside(s: String): Map<Char,Int> {
107 val result = mutableMapOf<Char,Int>()
108 var i = 0
109 while (i < s.length) {
110 if (s[i] == '$') {
111 if (i+1 < s.length && (s[i+1] == '(' || s[i+1] == '{')) i = closing(s,i+1)+1 else i = minOf(s.length,i+2)
112 } else {
113 if (s[i] in "=:;" && s[i] !in result) result[s[i]] = i
114 i++
115 }
116 }
117 return result

119 fun value(name: String): String {
120 for (scope in scopes.asReversed()) when(scope) {
121 is ForeachScope -> if (scope.name == name) return scope.value
122 is CallScope -> if (name.isNotEmpty() && name.all {
123 it in '0'..'9'
124 }) return scope.values.getOrNull(name.toInt()) ?: ""
125 }
126 automatic[name]?.let {
127 return it
128 }
129 val v = variables[name] ?: return ""
130 if (v.simple) return v.value
131 if (!expanding.add(name)) error("Recursive variable '$name' references itself (eventually)")
132 try {
133 return expand(v.value)
134 } finally {
135 expanding.remove(name)
136 }

138 fun expand(s: String): String {
139 val out = StringBuilder()
140 var i = 0
141 while (i < s.length) {
142 if (s[i] != '$') {
143 out.append(s[i++])
144 continue
145 }
146 if (i+1 == s.length) break
147 val c = s[i+1]
148 when(c) {
149 '$' -> {
150 out.append('$')
151 i += 2
152 }
153 '(','{' -> {
154 val end = closing(s,i+1)
155 out.append(reference(s.substring(i+2,end),c))
156 i = end+1
157 } else -> {
158 out.append(value(c.toString()))
159 i += 2
160 }
161 }
162 }
163 return out.toString()
164 }
165 val functions = mapOf("subst" to (3 to 3),"patsubst" to (3 to 3),"strip" to (1 to 1),"findstring" to (2 to 2),"filter" to (2 to 2),"filter-out" to (2 to 2),"sort" to (1 to 1),"word" to (2 to 2),"wordlist" to (3 to 3),"words" to (1 to 1),"firstword" to (1 to 1),"lastword" to (1 to 1),"dir" to (1 to 1),"notdir" to (1 to 1),"suffix" to (1 to 1),"basename" to (1 to 1),"addsuffix" to (2 to 2),"addprefix" to (2 to 2),"wildcard" to (1 to 1),"foreach" to (3 to 3),"if" to (2 to 3),"call" to (1 to Int.MAX_VALUE),"value" to (1 to 1),"info" to (1 to 1),"warning" to (1 to 1),"error" to (1 to 1))
166 fun reference(content: String,opening: Char): String {
167 val space = content.indexOf(' ')
168 if (space >= 0 && content.take(space) in functions) return function(content.take(space),content.drop(space+1).trimStart(' '),opening)
169 val n = expand(content)
170 val colon = n.indexOf(':')
171 val equal = if (colon >= 0) n.indexOf('=',colon+1) else -1
172 if (equal < 0) return value(n)
173 var p = n.substring(colon+1,equal)
174 var r = n.drop(equal+1)
175 if ('%' !in p) {
176 p = "%$p"
177 r = "%$r"
178 }
179 return words(value(n.take(colon))).joinToString(" ") {
180 replaceWord(it,p,r)
181 }

183 fun arguments(text: String,opening: Char,maximum: Int): List<String> {
184 val close = if (opening == '(') ')' else '}'
185 var depth = 0
186 var start = 0
187 val result = mutableListOf<String>()
188 for ((i,c) in text.withIndex()) {
189 if (c == opening) depth++
190 if (c == close) depth--
191 if (c == ',' && depth == 0 && result.size < maximum-1) {
192 result.add(text.substring(start,i))
193 start = i+1
194 }
195 }
196 result.add(text.drop(start))
197 return result

199 fun function(name: String,text: String,opening: Char): String {
200 val limits = functions.getValue(name)
201 val raw = arguments(text,opening,limits.second)
202 if (raw.size < limits.first) error("insufficient number of arguments (${raw.size}) to function '$name'")
203 if (name == "foreach") {
204 val variable = expand(raw[0]).trim(' ')
205 val list = words(expand(raw[1]))
206 return list.joinToString(" ") {
207 word -> scopes.add(ForeachScope(variable,word))
208 try {
209 expand(raw[2])
210 } finally {
211 scopes.removeAt(scopes.lastIndex)
212 }
213 }
214 }
215 if (name == "if") {
216 val condition = expand(raw[0]).trim(' ')
217 return if (condition.isNotEmpty()) expand(raw[1]) else if (raw.size == 3) expand(raw[2]) else ""
218 }
219 val args = raw.map {
220 val expanded = expand(it)
221 if (name == "subst" || name == "call") expanded else expanded.trim(' ')
222 }
223 fun list(i: Int) = words(args[i])
224 fun joined(xs: List<String>) = xs.joinToString(" ")
225 when(name) {
226 "call" -> {
227 val n = args[0].trim(' ')
228 val v = variables[n] ?: return ""
229 if (v.simple) return v.value
230 scopes.add(CallScope(listOf(n)+args.drop(1)))
231 try {
232 return expand(v.value)
233 } finally {
234 scopes.removeAt(scopes.lastIndex)
235 }
236 }
237 "value" -> return variables[args[0]]?.value ?: ""
238 "info" -> {
239 output.add(args[0])
240 return ""
241 }
242 "warning" -> {
243 output.add("$location: ${args[0]}")
244 return ""
245 }
246 "error" -> error(args[0])
247 "subst" -> return if (args[0].isEmpty()) args[2] else args[2].replace(args[0],args[1])
248 "patsubst" -> return joined(list(2).map {
249 replaceWord(it,args[0],args[1])
250 })
251 "strip" -> return joined(list(0))
252 "findstring" -> return if (args[1].contains(args[0])) args[0] else ""
253 "filter","filter-out" -> return joined(list(1).filter {
254 w -> list(0).any {
255 stem(it,w) != null
256 } == (name == "filter")
257 })
258 "sort" -> return joined(list(0).toSortedSet().toList())
259 "word" -> {
260 val n = args[0]
261 if (n.isEmpty() || !n.all {
262 it in '0'..'9'
263 }) error("non-numeric first argument to 'word' function: '$n'")
264 if (n.toInt() == 0) error("first argument to 'word' function must be greater than 0")
265 return list(1).getOrNull(n.toInt()-1) ?: ""
266 }
267 "wordlist" -> {
268 val a = args[0]
269 val b = args[1]
270 if (a.isEmpty() || !a.all {
271 it in '0'..'9'
272 }) error("non-numeric first argument to 'wordlist' function: '$a'")
273 if (b.isEmpty() || !b.all {
274 it in '0'..'9'
275 }) error("non-numeric second argument to 'wordlist' function: '$b'")
276 if (a.toInt() == 0) error("invalid first argument to 'wordlist' function: '$a'")
277 return joined(list(2).filterIndexed {
278 i,_ -> i+1 >= a.toInt() && i+1 <= b.toInt()
279 })
280 }
281 "words" -> return list(0).size.toString()
282 "firstword" -> return list(0).firstOrNull() ?: ""
283 "lastword" -> return list(0).lastOrNull() ?: ""
284 "dir" -> return joined(list(0).map {
285 it.take(it.lastIndexOf('/')+1).ifEmpty {
286 "./"
287 }
288 })
289 "notdir" -> return joined(list(0).map {
290 it.substring(it.lastIndexOf('/')+1)
291 })
292 "suffix" -> return joined(list(0).filter {
293 it.lastIndexOf('.') > it.lastIndexOf('/')
294 }.map {
295 it.substring(it.lastIndexOf('.'))
296 })
297 "basename" -> return joined(list(0).map {
298 if (it.lastIndexOf('.') > it.lastIndexOf('/')) it.take(it.lastIndexOf('.')) else it
299 })
300 "addsuffix" -> return joined(list(1).map {
301 it+args[0]
302 })
303 "addprefix" -> return joined(list(1).map {
304 args[0]+it
305 })
306 "wildcard" -> return joined(list(0).flatMap {
307 glob(it)
308 })
309 }
310 return ""

339 fun load(body: List<String>) {
340 clear()
341 var current: Current? = null
342 val conditions = mutableListOf<Condition>()
343 fun active() = conditions.lastOrNull()?.let {
344 it.outer && it.value
345 }?: true
346 var i = 0
347 try {
348 while (i < body.size) {
349 val lineNumber = i+1
350 location = "Makefile:$lineNumber"
351 var text = body[i++]
352 if (text.startsWith('>')) {
353 if (!active()) continue
354 val rule = current ?: error("recipe commences before first target")
355 addRecipe(rule,RecipeLine(lineNumber,text.drop(1).trimStart(' ')))
356 continue
357 }
358 while (text.endsWith('\\') && i < body.size) text = text.dropLast(1).trimEnd(' ')+" "+body[i++].trimStart(' ')
359 text = text.substringBefore('#').trim(' ')
360 val first = text.substringBefore(' ')
361 val rest = text.substringAfter(' ',"").trim(' ')
362 when(first) {
363 "ifdef","ifndef","ifeq","ifneq" -> {
364 val outer = active()
365 var truth = false
366 if (outer) {
367 if (first == "ifdef" || first == "ifndef") {
368 val name = expand(rest).trim(' ')
369 truth = variables[name]?.value?.isNotEmpty() == true
370 if (first == "ifndef") truth = !truth
371 } else {
372 if (rest.length < 2 || !rest.startsWith('(') || !rest.endsWith(')')) error("invalid syntax in conditional")
373 val inner = rest.substring(1,rest.length-1)
374 var depth = 0
375 var comma = -1
376 for ((j,c) in inner.withIndex()) {
377 if (c == '(') depth++
378 if (c == ')') depth--
379 if (c == ',' && depth == 0) {
380 comma = j
381 break
382 }
383 }
384 if (comma < 0) error("invalid syntax in conditional")
385 val left = expand(inner.take(comma)).trim(' ')
386 val right = expand(inner.drop(comma+1)).trim(' ')
387 truth = (left == right) == (first == "ifeq")
388 }
389 }
390 conditions.add(Condition(outer,truth,false,lineNumber))
391 continue
392 }
393 "else" -> {
394 val condition = conditions.lastOrNull() ?: error("extraneous 'else'")
395 if (condition.hadElse) error("only one 'else' per conditional")
396 condition.value = !condition.value
397 condition.hadElse = true
398 continue
399 }
400 "endif" -> {
401 if (conditions.isEmpty()) error("extraneous 'endif'")
402 conditions.removeAt(conditions.lastIndex)
403 continue
404 }
405 }
406 if (text.isEmpty() || !active()) continue
407 val positions = outside(text)
408 val equal = positions['=']
409 val colon = positions[':']
410 if (equal != null && (colon == null || colon >= equal-1)) {
411 val special = equal > 0 && text[equal-1] in ":+?"
412 val op = if (special) text.substring(equal-1,equal+1) else "="
413 val n = text.take(equal - (if (special) 1 else 0)).trim(' ')
414 val rhs = text.drop(equal+1).trim(' ')
415 if (n.isEmpty()) error("empty variable name")
416 val old = variables[n]
417 when(op) {
418 "=" -> variables[n] = Variable(false,rhs)
419 ":=" -> variables[n] = Variable(true,expand(rhs))
420 "?=" -> if (old == null) variables[n] = Variable(false,rhs)
421 "+=" -> if (old == null) variables[n] = Variable(false,rhs) else {
422 val append = if (old.simple) expand(rhs) else rhs
423 old.value = if (old.value.isEmpty()) append else old.value+" "+append
424 }
425 }
426 current = null
427 } else if (colon != null) {
428 val targets = unique(words(expand(text.take(colon))))
429 var remainder = text.drop(colon+1)
430 val semi = outside(remainder)[';']
431 val inline = semi?.let {
432 remainder.drop(it+1).trimStart(' ')
433 }
434 if (semi != null) remainder = remainder.take(semi)
435 val secondColon = outside(remainder)[':']
436 val targetPattern = secondColon?.let {
437 expand(remainder.take(it)).trim(' ')
438 }
439 if (secondColon != null) remainder = remainder.drop(secondColon+1)
440 val deps = words(expand(remainder))
441 val registered = mutableListOf<String>()
442 val implicit = mutableListOf<Rule>()
443 if (targets == listOf(".PHONY")) phony.addAll(deps) else if (targetPattern != null) {
444 if ('%' !in targetPattern) error("target pattern contains no '%'")
445 for (t in targets) {
446 val m = stem(targetPattern,t)
447 if (m == null || m.isEmpty()) output.add("Makefile:$lineNumber: target '$t' doesn't match the target pattern") else {
448 register(t,deps.map {
449 substitute(it,m)
450 })
451 rules.getValue(t).stem = m
452 registered.add(t)
453 }
454 }
455 } else if (targets.any {
456 '%' in it
457 }) {
458 if (targets.any {
459 '%' !in it
460 }) error("mixed implicit and normal rules")
461 for (t in targets) {
462 val rule = Rule(t,deps.toMutableList())
463 patterns.add(rule)
464 implicit.add(rule)
465 }
466 } else for (t in targets) {
467 register(t,deps)
468 registered.add(t)
469 }
470 current = Current(registered,implicit)
471 if (inline != null) addRecipe(current,RecipeLine(lineNumber,inline))
472 } else if (expand(text).trim(' ').isNotEmpty()) error("missing separator")
473 }
474 if (conditions.isNotEmpty()) {
475 location = "Makefile:${conditions.last().line}"
476 error("missing 'endif'")
477 }
478 loaded = true
479 output.add("Makefile loaded: lines=${body.size} variables=${variables.size} targets=${rules.size} patterns=${patterns.count { it.recipe != null }} goal=${goal ?: "(none)"}")
480 } catch (e: Expansion) {
481 diagnostic(e)
482 clear()
483 } finally {
484 location = "make"
485 scopes.clear()
486 expanding.clear()
487 automatic = emptyMap()
488 }

490 data class Determined(val deps: List<String>,val recipe: Recipe?,val stem: String,val source: Boolean = false,val absent: Boolean = false)
491 fun determine(t: String): Determined {
492 val explicit = rules[t]
493 if (explicit?.recipe != null) return Determined(explicit.deps.toList(),explicit.recipe,explicit.stem)
494 if (t !in phony) {
495 val candidates = patterns.mapIndexedNotNull {
496 i,r -> val m = if (r.recipe != null) stem(r.target,t) else null
497 if (m.isNullOrEmpty()) null else Triple(r,m,i)
498 }.sortedWith(compareBy<Triple<Rule,String,Int>> {
499 it.second.length
500 }.thenBy {
501 it.third
502 })
503 for ((r,m,_) in candidates) {
504 val deps = r.deps.map {
505 substitute(it,m)
506 }
507 if (deps.all {
508 it in files || it in rules
509 }) return Determined(unique(deps+(explicit?.deps ?: emptyList())),r.recipe,m)
510 }
511 }
512 if (explicit != null || t in phony) return Determined(explicit?.deps?.toList() ?: emptyList(),null,explicit?.stem ?: "")
513 return Determined(emptyList(),null,"",source=t in files,absent=t !in files)
514 }
515 fun make(args: List<String>) {
516 val options = mutableSetOf<Char>()
517 val goals = mutableListOf<String>()
518 for (arg in args) if (arg.startsWith('-')) {
519 for (c in arg.drop(1)) {

532 if (goal == null) {
533 output.add("make: *** No targets. Stop.")
534 return

538 val dry = 'n' in options
539 val keep = 'k' in options
540 val original = if (dry) files.toSortedMap() else null

546 fun targetTime(t: String) = if (t in phony) null else files[t]
547 fun execute(command: String): Int {
548 val w = words(command)
549 when(w[0]) {
550 "echo" -> output.add(w.drop(1).joinToString(" "))
551 "touch" -> for (n in w.drop(1)) tick(n)
552 "rm" -> {
553 val force = w.getOrNull(1) == "-f"
554 var status = 0
555 for (p in w.drop(if (force) 2 else 1)) {
556 val names = glob(p).ifEmpty {
557 listOf(p)
558 }
559 for (n in names) if (files.remove(n) == null && !force) {
560 output.add("rm: cannot remove '$n': No such file or directory")
561 status = 1
562 }
563 }
564 return status
565 }
566 "false" -> return 1
567 "exit" -> return w.getOrNull(1)?.toInt() ?: 0
568 }
569 return 0
570 }
571 fun run(t: String,d: Determined,deps: List<String>,fresh: List<String>): Int {
572 automatic = mapOf("@" to t,"<" to (deps.firstOrNull() ?: ""),"^" to deps.joinToString(" "),"?" to fresh.joinToString(" "),"*" to d.stem)
573 val expanded = mutableListOf<Pair<RecipeLine,String>>()
574 try {
575 for (line in d.recipe!!.lines) {
576 location = "Makefile:${line.number}"
577 expanded.add(line to expand(line.text))
578 }
579 } catch (e: Expansion) {
580 diagnostic(e)
581 throw Abort()
582 } finally {
583 automatic = emptyMap()
584 location = "make"
585 }
586 for ((line,text) in expanded) {
587 val prefix = text.takeWhile {
588 it == ' ' || it == '@' || it == '-'
589 }
590 val command = text.drop(prefix.length).trimEnd(' ')
591 if (command.isEmpty()) continue
592 if (dry || ('@' !in prefix && 's' !in options)) output.add(command)
593 if (dry) continue
594 val status = execute(command)
595 if (status != 0) {
596 if ('-' in prefix || 'i' in options) output.add("make: [Makefile:${line.number}: $t] Error $status (ignored)") else {
597 output.add("make: *** [Makefile:${line.number}: $t] Error $status")
598 if (!keep) throw Abort()
599 return 2
600 }
601 }
602 }
603 return 1

605 fun build(t: String,parent: String?): Int {
606 results[t]?.let {
607 return it
608 }
609 val d = determine(t)
610 usedRules[t] = d
611 if (d.absent) {
612 results[t] = 2
613 val reason = "make: *** No rule to make target '$t'"+(parent?.let {
614 ", needed by '$it'"
615 }?: "")+"."
616 output.add(reason+if (keep) "" else " Stop.")
617 if (!keep) throw Abort()
618 return 2
619 }
620 if (d.source) {
621 results[t] = 1
622 return 1
623 }
624 being.add(t)
625 val deps = mutableListOf<String>()
626 var failed = false
627 for (q in d.deps) if (q in being) output.add("make: Circular $t <- $q dependency dropped.") else {
628 deps.add(q)
629 if (build(q,t) != 1) failed = true
630 }
631 being.remove(t)
632 var result = 3
633 if (!failed) {
634 val time = targetTime(t)
635 val fresh = deps.filter {
636 'B' in options || time == null || targetTime(it) == null || targetTime(it)!! > time
637 }
638 result = if (d.recipe == null || ('B' !in options && time != null && fresh.isEmpty())) 1 else {
639 runs++
640 val status = run(t,d,deps,fresh)
641 if (status == 1 && t !in phony) tick(t)
642 status
643 }
644 }
645 results[t] = result
646 return result
647 }
648 try {
649 for (g in goals) {
650 val before = runs
651 val result = build(g,null)
652 if (result == 1 && runs == before) output.add(if (g in phony || usedRules.getValue(g).recipe == null) "make: Nothing to be done for '$g'." else "make: '$g' is up to date.") else if (result == 3) output.add("make: Target '$g' not remade because of errors.")
653 }
654 } catch (_: Abort) {
655 } finally {
656 if (dry) {
657 files = original!!
658 clock = oldClock
659 }
660 automatic = emptyMap()
661 scopes.clear()
662 expanding.clear()
663 location = "make"
664 }
665 }
666 fun command(line: String) {
667 output.add("$ $line")
668 val w = words(line)
669 when(w[0]) {
670 "MAKE" -> make(w.drop(1))
671 "TOUCH" -> for (n in w.drop(1)) tick(n)
672 "RM" -> for (n in w.drop(1)) if (files.remove(n) == null) output.add("rm: cannot remove '$n': No such file or directory")
673 "FILES" -> {
674 output.add("clock $clock")
675 output.addAll(files.map {
676 (n,t) -> "$n $t"
677 })
678 }
679 "RULES" -> if (loaded) {
680 for (r in rules.values+patterns.filter {
681 it.recipe != null
682 }) {
683 output.add(r.target+":"+r.deps.joinToString("") {
684 " $it"
685 })
686 r.recipe?.lines?.forEach {
687 output.add(">"+it.text)
688 }
689 }
690 if (phony.isNotEmpty()) output.add(".PHONY:"+phony.joinToString("") {
691 " $it"
692 })
693 }
694 "VARS" -> if (loaded) {
695 try {
696 for ((n,v) in variables) {
697 val s = value(n).trim(' ')
698 output.add(n+" "+if (v.simple) ":="+(if (s.isEmpty()) "" else " $s") else "="+(if (s.isEmpty()) "" else " $s"))
699 }
700 } catch (e: Expansion) {
701 diagnostic(e)
702 }
703 }
704 "EXPAND" -> try {
705 output.add("["+expand(line.drop(7))+"]")
706 } catch (e: Expansion) {
707 diagnostic(e)
708 }
709 }
710 }
711}

713fun main() {
714 val input = generateSequence(::readLine).toList()
715 val simulator = Simulator()
716 var i = 1
717 while (input[i] != "END FILES") {
718 val w = words(input[i++])
719 val time = w[1].toInt()
720 simulator.files[w[0]] = time
721 simulator.clock = maxOf(simulator.clock,time)
722 }
723 i++
724 while (input[i] != "QUIT") {
725 val line = input[i++]
726 if (line == "MAKEFILE") {
727 simulator.output.add("$ MAKEFILE")
728 val body = mutableListOf<String>()
729 while (input[i] != "END MAKEFILE") body.add(input[i++])
730 i++
731 simulator.load(body)
732 } else simulator.command(line)
733 }
734 for (line in simulator.output) println(line)
735}
