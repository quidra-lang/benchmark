20static void *memory(Arena *arena, size_t size) {
21 void *p = calloc(1, size ? size : 1);
22 Allocation *a = malloc(sizeof *a);
23 if (!p || !a) exit(EXIT_FAILURE);
24 *a = (Allocation){p, arena->head};
25 arena->head = a;
26 return p;
27}

29static void clear_arena(Arena *arena) {
30 while (arena->head) {
31 Allocation *next = arena->head->next;
32 free(arena->head->pointer);
33 free(arena->head);
34 arena->head = next;
35 }
36}

50static char *format(const char *text, ...) {
51 va_list args, other;

53 va_copy(other, args);

88typedef struct {

93static void add_string(Strings *s, char *text, Arena *arena) {
94 if (s->count == s->capacity) {
95 int capacity = s->capacity ? 2 * s->capacity : 4;
96 char **items = memory(arena, (size_t)capacity * sizeof *items);
97 if (s->count) memcpy(items, s->items, (size_t)s->count * sizeof *items);
98 s->items = items;
99 s->capacity = capacity;
100 }
101 s->items[s->count++] = text;
102}

104static bool contains(const Strings *s, const char *text) {
105 for (int i = 0; i < s->count; i++) if (!strcmp(s->items[i], text)) return true;
106 return false;
107}

109static void add_unique(Strings *s, char *text, Arena *arena) {
110 if (!contains(s, text)) add_string(s, text, arena);
111}

113static Strings words(const char *text) {
114 Strings result = {0};
115 while (*text) {
116 while (*text == ' ') text++;
117 const char *end = text;
118 while (*end && *end != ' ') end++;
119 if (end != text) add_string(&result, part(text, (size_t)(end - text)), &temporary);
120 text = end;
121 }
122 return result;
123}

129static Strings sorted(const Strings *s) {
130 Strings result = {0};
131 for (int i = 0; i < s->count; i++) add_string(&result, s->items[i], &temporary);
132 if (result.count > 1) qsort(result.items, (size_t)result.count, sizeof *result.items, string_compare);
133 return result;
134}

136typedef struct {
137 char *text;
138 size_t length, capacity;
139} Buffer;

141static void append_n(Buffer *b, const char *text, size_t length) {
142 if (b->length + length + 1 > b->capacity) {
143 size_t capacity = b->capacity ? 2 * b->capacity : 32;
144 while (capacity < b->length + length + 1) capacity *= 2;
145 char *p = memory(&temporary, capacity);
146 if (b->length) memcpy(p, b->text, b->length);
147 b->text = p;
148 b->capacity = capacity;
149 }
150 memcpy(b->text + b->length, text, length);
151 b->length += length;
152 b->text[b->length] = '\0';
153}

155static void append(Buffer *b, const char *text) {
156 append_n(b, text, strlen(text));
157}

168 }
169 return finish(&b);

172static char *stem(const char *pattern, const char *word) {
173 const char *percent = strchr(pattern, '%');
174 if (!percent) return !strcmp(pattern, word) ? "" : NULL;
175 size_t prefix = (size_t)(percent - pattern), suffix = strlen(percent + 1), length = strlen(word);
176 if (length < prefix + suffix || strncmp(pattern, word, prefix) || strcmp(percent + 1, word + length - suffix)) return NULL;
177 return part(word + prefix, length - prefix - suffix);
178}

180static char *substitute(const char *pattern, const char *value) {
181 const char *p = strchr(pattern, '%');
182 if (!p) return (char *)pattern;
183 return format("%s%s%s", part(pattern, (size_t)(p - pattern)), value, p + 1);
184}

186static char *replacement(const char *word, const char *pattern, const char *value) {
187 char *s = stem(pattern, word);
188 return !s ? (char *)word : strchr(pattern, '%') ? substitute(value, s) : (char *)value;
189}

191static bool wildcard(const char *pattern, const char *name) {
192 size_t n = strlen(name);
193 bool *previous = calloc(n + 1, sizeof *previous), *next = calloc(n + 1, sizeof *next);
194 if (!previous || !next) exit(EXIT_FAILURE);
195 previous[0] = true;
196 for (const char *p = pattern; *p; p++) {
197 memset(next, 0, (n + 1) * sizeof *next);
198 for (size_t j = 0; j <= n; j++) {
199 if (*p == '*') next[j] = previous[j] || (j && name[j - 1] != '/' && next[j - 1]);
200 else if (j) next[j] = previous[j - 1] && (*p == '?' ? name[j - 1] != '/' : *p == name[j - 1]);
201 }
202 bool *swap = previous;
203 previous = next;
204 next = swap;
205 }
206 bool result = previous[n];
207 free(previous);
208 free(next);
209 return result;
210}

220static int file_index(const char *name) {
221 int lo = 0, hi = file_count;
222 while (lo < hi) {
223 int middle = lo + (hi - lo) / 2;
224 if (strcmp(files[middle].name, name) < 0) lo = middle + 1;
225 else hi = middle;
226 }
227 return lo < file_count && !strcmp(files[lo].name, name) ? lo : -1;
228}

230static void put_file(const char *name, int time) {
231 int i = file_index(name);
232 if (i >= 0) {
233 files[i].time = time;
234 return;
235 }
236 if (file_count == file_capacity) {
237 file_capacity = file_capacity ? 2 * file_capacity : 8;
238 File *p = realloc(files, (size_t)file_capacity * sizeof *p);
239 if (!p) exit(EXIT_FAILURE);
240 files = p;
241 }
242 i = 0;
243 while (i < file_count && strcmp(files[i].name, name) < 0) i++;
244 memmove(files + i + 1, files + i, (size_t)(file_count - i) * sizeof *files);
245 files[i] = (File){copy(&permanent, name), time};
246 file_count++;
247}

262 Strings result = {0};
263 if (!strchr(pattern, '*') && !strchr(pattern, '?')) {
264 int i = file_index(pattern);
265 if (i >= 0) add_string(&result, files[i].name, &temporary);

371static int bracket_end(const char *text, int opening) {
372 char open = text[opening], close = open == '(' ? ')' : '}';
373 int depth = 0;
374 for (int i = opening; text[i]; i++) {
375 if (text[i] == open) depth++;
376 else if (text[i] == close && --depth == 0) return i;
377 }
378 expansion_error("unterminated variable reference");
379 return -1;
380}

382static char *value(const char *name) {
383 for (Scope *s = scopes; s; s = s->previous) {
384 if (!s->arguments) {
385 if (!strcmp(s->name, name)) return s->value;
386 } else if (numeric(name)) {
387 int i = atoi(name);
388 return i < s->arguments->count ? s->arguments->items[i] : "";
389 }
390 }
391 if (has_automatic) for (int i = 0; i < 5; i++) if (!strcmp(name, auto_names[i])) return automatic[i];
392 Variable *v = find_variable(name);
393 if (!v) return "";
394 if (v->simple) return v->value;
395 for (Active *a = expanding; a; a = a->previous) if (!strcmp(a->name, name)) expansion_error(format("Recursive variable '%s' references itself (eventually)", name));
396 Active active = {name, expanding};
397 expanding = &active;
398 char *result = expand(v->value);
399 expanding = active.previous;
400 return result;
401}

403static char *expand(const char *text) {
404 Buffer result = {0};
405 for (int i = 0; text[i];) {
406 char c = text[i++];
407 if (c != '$') {
408 append_n(&result, &c, 1);
409 continue;
410 }
411 if (!text[i]) break;
412 c = text[i++];
413 if (c == '$') append(&result, "$");
414 else if (c == '(' || c == '{') {
415 int end = bracket_end(text, i - 1);
416 append(&result, reference(part(text + i, (size_t)(end - i)), c));
417 i = end + 1;
418 } else {
419 char name[2] = {c, '\0'};
420 append(&result, value(name));
421 }
422 }
423 return finish(&result);
424}

465static Strings arguments(const char *text, int maximum, char opening) {
466 char closing = opening == '(' ? ')' : '}';
467 int depth = 0, start = 0;
468 Strings result = {0};
469 int i;
470 for (i = 0; text[i]; i++) {
471 char c = text[i];
472 if (c == opening) depth++;
473 else if (c == closing) depth--;
474 else if (c == ',' && depth == 0 && result.count < maximum - 1) {
475 add_string(&result, part(text + start, (size_t)(i - start)), &temporary);
476 start = i + 1;
477 }
478 }
479 add_string(&result, part(text + start, (size_t)(i - start)), &temporary);
480 return result;
481}

483static char *function(int f, const char *text, char opening) {
484 const Function *fn = &functions[f];
485 Strings raw = arguments(text, fn->maximum, opening);
486 if (raw.count < fn->minimum) expansion_error(format("insufficient number of arguments (%d) to function '%s'", raw.count, fn->name));
487 if (f == FOREACH) {
488 char *name = trim(expand(raw.items[0]));
489 Strings list = words(expand(raw.items[1])), result = {0};
490 for (int i = 0; i < list.count; i++) {
491 Scope scope = {name, list.items[i], NULL, scopes};
492 scopes = &scope;
493 add_string(&result, expand(raw.items[2]), &temporary);
494 scopes = scope.previous;
495 }
496 return join(&result);
497 }
498 if (f == IF) {
499 bool condition = *trim(expand(raw.items[0])) != '\0';
500 return condition ? expand(raw.items[1]) : raw.count == 3 ? expand(raw.items[2]) : "";
501 }
502 Strings a = {0};
503 for (int i = 0; i < raw.count; i++) {
504 char *v = expand(raw.items[i]);
505 add_string(&a, f == SUBST || f == CALL ? v : trim(v), &temporary);
506 }
507 if (f == CALL) {
508 char *name = trim(a.items[0]);
509 Variable *v = find_variable(name);
510 if (!v) return "";
511 if (v->simple) return v->value;
512 a.items[0] = name;
513 Scope scope = {NULL, NULL, &a, scopes};
514 scopes = &scope;
515 char *result = expand(v->value);
516 scopes = scope.previous;
517 return result;
518 }
519 if (f == VALUE) {
520 Variable *v = find_variable(a.items[0]);
521 return v ? v->value : "";
522 }
523 if (f == INFO) {
524 puts(a.items[0]);
525 return "";
526 }
527 if (f == WARNING) {
528 printf("%s: %s\n", location, a.items[0]);
529 return "";
530 }
531 if (f == ERROR_FUNCTION) {
532 expansion_error(a.items[0]);
533 return "";
534 }
535 if (f == SUBST) {
536 if (!*a.items[0]) return a.items[2];
537 Buffer b = {0};
538 const char *p = a.items[2], *next;
539 while ((next = strstr(p, a.items[0])) != NULL) {
540 append_n(&b, p, (size_t)(next - p));
541 append(&b, a.items[1]);
542 p = next + strlen(a.items[0]);
543 }
544 append(&b, p);
545 return finish(&b);
546 }
547 if (f == FINDSTRING) return strstr(a.items[1], a.items[0]) ? a.items[0] : "";
548 if (f == WORD) {
549 if (!numeric(a.items[0])) expansion_error(format("non-numeric first argument to 'word' function: '%s'", a.items[0]));
550 int n = atoi(a.items[0]);
551 if (!n) expansion_error("first argument to 'word' function must be greater than 0");
552 Strings list = words(a.items[1]);
553 return n <= list.count ? list.items[n - 1] : "";
554 }
555 if (f == WORDLIST) {
556 if (!numeric(a.items[0])) expansion_error(format("non-numeric first argument to 'wordlist' function: '%s'", a.items[0]));
557 if (!numeric(a.items[1])) expansion_error(format("non-numeric second argument to 'wordlist' function: '%s'", a.items[1]));
558 int start = atoi(a.items[0]), end = atoi(a.items[1]);
559 if (!start) expansion_error(format("invalid first argument to 'wordlist' function: '%s'", a.items[0]));
560 Strings list = words(a.items[2]), result = {0};
561 for (int i = start - 1; i < end && i < list.count; i++) add_string(&result, list.items[i], &temporary);
562 return join(&result);
563 }
564 Strings list = words(a.items[a.count - 1]), result = {0};
565 if (f == WORDS) return format("%d", list.count);
566 if (f == FIRSTWORD) return list.count ? list.items[0] : "";
567 if (f == LASTWORD) return list.count ? list.items[list.count - 1] : "";
568 Strings filters = {0};
569 if (f == FILTER || f == FILTER_OUT) filters = words(a.items[0]);
570 for (int i = 0; i < list.count; i++) {
571 char *word = list.items[i], *v = NULL;
572 if (f == PATSUBST) v = replacement(word, a.items[0], a.items[1]);
573 else if (f == FILTER || f == FILTER_OUT) {
574 bool found = false;
575 for (int j = 0; j < filters.count; j++) if (stem(filters.items[j], word)) {
576 found = true;
577 break;
578 }
579 if (found == (f == FILTER)) v = word;
580 } else if (f == STRIP) v = word;
581 else if (f == SORT) {
582 add_unique(&result, word, &temporary);
583 continue;
584 } else if (f == ADDSUFFIX) v = format("%s%s", word, a.items[0]);
585 else if (f == ADDPREFIX) v = format("%s%s", a.items[0], word);
586 else if (f == WILDCARD) {
587 Strings matches = wildcard_expansion(word);
588 for (int j = 0; j < matches.count; j++) add_string(&result, matches.items[j], &temporary);
589 continue;
590 } else {
591 char *slash = strrchr(word, '/'), *dot = strrchr(word, '.');
592 bool extension = dot && (!slash || dot > slash);
593 if (f == DIR) v = slash ? part(word, (size_t)(slash - word + 1)) : "./";
594 else if (f == NOTDIR) v = slash ? slash + 1 : word;
595 else if (f == SUFFIX) {
596 if (extension) v = dot;
597 } else if (f == BASENAME) v = extension ? part(word, (size_t)(dot - word)) : word;
598 }
599 if (v) add_string(&result, v, &temporary);
600 }
601 if (f == SORT) result = sorted(&result);
602 return join(&result);
603}

605static char *reference(const char *content, char opening) {
606 const char *space = strchr(content, ' ');
607 if (space) {
608 int f = function_index(part(content, (size_t)(space - content)));
609 if (f >= 0) return function(f, left_trim(space + 1), opening);
610 }
611 char *name = expand(content), *colon = strchr(name, ':'), *equal = colon ? strchr(colon + 1, '=') : NULL;
612 if (equal) {
613 char *pattern = part(colon + 1, (size_t)(equal - colon - 1)), *repl = equal + 1;
614 if (!strchr(pattern, '%')) {
615 pattern = format("%%%s", pattern);
616 repl = format("%%%s", repl);
617 }
618 Strings list = words(value(part(name, (size_t)(colon - name)))), result = {0};
619 for (int i = 0; i < list.count; i++) add_string(&result, replacement(list.items[i], pattern, repl), &temporary);
620 return join(&result);
621 }
622 return value(name);
623}

629static Positions outside(const char *text) {
630 Positions p = {-1, -1, -1};
631 for (int i = 0; text[i];) {
632 char c = text[i];
633 if (c == '$') {
634 if (text[i + 1] == '(' || text[i + 1] == '{') i = bracket_end(text, i + 1) + 1;
635 else i += text[i + 1] ? 2 : 1;
636 } else {
637 if (c == '=' && p.equal < 0) p.equal = i;
638 if (c == ':' && p.colon < 0) p.colon = i;
639 if (c == ';' && p.semicolon < 0) p.semicolon = i;
640 i++;
641 }
642 }
643 return p;
644}

663static void group_add(Group *g, Rule *r) {
664 if (g->count == g->capacity) {
665 int capacity = g->capacity ? 2 * g->capacity : 4;
666 Rule **p = memory(&temporary, (size_t)capacity * sizeof *p);
667 if (g->count) memcpy(p, g->rules, (size_t)g->count * sizeof *p);
668 g->rules = p;
669 g->capacity = capacity;
670 }
671 g->rules[g->count++] = r;
672}

674static void add_recipe(Group *g, int number, const char *text) {
675 if (!g->count) return;
676 if (!g->recipe) {
677 g->recipe = memory(&makefile_memory, sizeof *g->recipe);
678 for (int i = 0; i < g->count; i++) {
679 Rule *r = g->rules[i];
680 if (!r->pattern && r->recipe) {
681 printf("Makefile:%d: warning: overriding recipe for target '%s'\n", number, r->name);
682 printf("Makefile:%d: warning: ignoring old recipe for target '%s'\n", r->recipe->first->number, r->name);
683 }
684 r->recipe = g->recipe;
685 }
686 }
687 RecipeLine *line = memory(&makefile_memory, sizeof *line);
688 line->number = number;
689 line->text = copy(&makefile_memory, text);
690 if (g->recipe->last) g->recipe->last->next = line;
691 else g->recipe->first = line;
692 g->recipe->last = line;
693}

695static bool active(Condition *conditions, int count) {
696 return !count || (conditions[count - 1].outer && conditions[count - 1].value);
697}

699static void conditional(const char *name, const char *rest, int line, Condition *conditions, int *count) {
700 if (!strcmp(name, "else")) {
701 if (!*count) expansion_error("extraneous 'else'");
702 Condition *top = &conditions[*count - 1];
703 if (top->alternative) expansion_error("only one 'else' per conditional");
704 top->value = !top->value;
705 top->alternative = true;
706 return;
707 }
708 if (!strcmp(name, "endif")) {
709 if (!*count) expansion_error("extraneous 'endif'");
710 (*count)--;
711 return;
712 }
713 bool outer = active(conditions, *count), condition = false;
714 if (outer) {
715 if (!strcmp(name, "ifdef") || !strcmp(name, "ifndef")) {
716 Variable *v = find_variable(trim(expand(rest)));
717 condition = v && *v->value;
718 if (!strcmp(name, "ifndef")) condition = !condition;
719 } else {
720 size_t n = strlen(rest);
721 if (n < 2 || rest[0] != '(' || rest[n - 1] != ')') expansion_error("invalid syntax in conditional");
722 char *inside = part(rest + 1, n - 2);
723 int depth = 0, comma = -1;
724 for (int i = 0; inside[i]; i++) {
725 if (inside[i] == '(') depth++;
726 else if (inside[i] == ')') depth--;
727 else if (inside[i] == ',' && !depth) {
728 comma = i;
729 break;
730 }
731 }
732 if (comma < 0) expansion_error("invalid syntax in conditional");
733 char *a = trim(expand(part(inside, (size_t)comma)));
734 char *b = trim(expand(inside + comma + 1));
735 condition = !strcmp(a, b);
736 if (!strcmp(name, "ifneq")) condition = !condition;
737 }
738 }
739 conditions[(*count)++] = (Condition){outer, condition, false, line};
740}

742static void load(const Strings *body) {
743 discard();
744 Group *current = NULL;
745 Condition *conditions = memory(&temporary, (size_t)(body->count + 1) * sizeof *conditions);
746 int condition_count = 0;
747 for (int i = 0; i < body->count; i++) {
748 int number = i + 1;
749 location = format("Makefile:%d", number);
750 char *line = body->items[i];
751 if (*line == '>') {
752 if (!active(conditions, condition_count)) continue;
753 if (!current) expansion_error("recipe commences before first target");
754 add_recipe(current, number, left_trim(line + 1));
755 continue;
756 }
757 while (*line && line[strlen(line) - 1] == '\\' && i + 1 < body->count) line = format("%s %s", right_trim(part(line, strlen(line) - 1)), left_trim(body->items[++i]));
758 char *comment = strchr(line, '#'), *t = trim(comment ? part(line, (size_t)(comment - line)) : line), *space = strchr(t, ' ');
759 char *first = space ? part(t, (size_t)(space - t)) : t, *rest = space ? trim(space + 1) : "";
760 if (!strcmp(first, "ifeq") || !strcmp(first, "ifneq") || !strcmp(first, "ifdef") || !strcmp(first, "ifndef") || !strcmp(first, "else") || !strcmp(first, "endif")) {
761 conditional(first, rest, number, conditions, &condition_count);
762 continue;
763 }
764 if (!*t || !active(conditions, condition_count)) continue;
765 Positions positions = outside(t);
766 int equal = positions.equal, colon = positions.colon;
767 if (equal >= 0 && (colon < 0 || colon >= equal - 1)) {
768 char op = equal > 0 && strchr(":+?", t[equal - 1]) ? t[equal - 1] : '=';
769 char *name = trim(part(t, (size_t)(equal - (op == '=' ? 0 : 1)))), *rhs = trim(t + equal + 1);
770 if (!*name) expansion_error("empty variable name");
771 Variable *old = find_variable(name);
772 if (op == ':') set_variable(name, true, expand(rhs));
773 else if (op == '+' && old) {
774 char *add = old->simple ? expand(rhs) : rhs;
775 set_variable(name, old->simple, *old->value ? format("%s %s", old->value, add) : add);
776 } else if (op != '?' || !old) set_variable(name, false, rhs);
777 current = NULL;
778 } else if (colon >= 0) {
779 Strings expanded = words(expand(part(t, (size_t)colon))), targets = {0};
780 for (int j = 0; j < expanded.count; j++) add_unique(&targets, expanded.items[j], &temporary);
781 char *r = t + colon + 1, *inline_recipe = NULL, *target_pattern = NULL;
782 int semicolon = outside(r).semicolon;
783 if (semicolon >= 0) {
784 inline_recipe = left_trim(r + semicolon + 1);
785 r = part(r, (size_t)semicolon);
786 }
787 int static_colon = outside(r).colon;
788 if (static_colon >= 0) {
789 target_pattern = trim(expand(part(r, (size_t)static_colon)));
790 r += static_colon + 1;
791 }
792 Strings prerequisites = words(expand(r));
793 current = memory(&temporary, sizeof *current);
794 if (targets.count == 1 && !strcmp(targets.items[0], ".PHONY")) {
795 for (int j = 0; j < prerequisites.count; j++) if (!contains(&phony, prerequisites.items[j])) add_string(&phony, copy(&makefile_memory, prerequisites.items[j]), &makefile_memory);
796 } else if (target_pattern) {
797 if (!strchr(target_pattern, '%')) expansion_error("target pattern contains no '%'");
798 for (int j = 0; j < targets.count; j++) {
799 char *s = stem(target_pattern, targets.items[j]);
800 if (!s || !*s) printf("%s: target '%s' doesn't match the target pattern\n", location, targets.items[j]);
801 else {
802 Strings p = {0};
803 for (int k = 0; k < prerequisites.count; k++) add_string(&p, substitute(prerequisites.items[k], s), &temporary);
804 Rule *rule = register_rule(targets.items[j], &p);
805 rule->stem = copy(&makefile_memory, s);
806 group_add(current, rule);
807 }
808 }
809 } else {
810 bool has_pattern = false, has_normal = false;
811 for (int j = 0; j < targets.count; j++) {
812 if (strchr(targets.items[j], '%')) has_pattern = true;
813 else has_normal = true;
814 }
815 if (has_pattern && has_normal) expansion_error("mixed implicit and normal rules");
816 for (int j = 0; j < targets.count; j++) {
817 Rule *rule;
818 if (has_pattern) {
819 rule = memory(&makefile_memory, sizeof *rule);
820 rule->name = copy(&makefile_memory, targets.items[j]);
821 rule->pattern = true;
822 rule->stem = "";
823 for (int k = 0; k < prerequisites.count; k++) add_string(&rule->prerequisites, copy(&makefile_memory, prerequisites.items[k]), &makefile_memory);
824 if (pattern_tail) pattern_tail->next = rule;
825 else patterns = rule;
826 pattern_tail = rule;
827 } else rule = register_rule(targets.items[j], &prerequisites);
828 group_add(current, rule);
829 }
830 }
831 if (inline_recipe) add_recipe(current, number, inline_recipe);
832 } else if (*trim(expand(t))) expansion_error("missing separator");
833 }
834 if (condition_count) {
835 location = format("Makefile:%d", conditions[condition_count - 1].line);
836 expansion_error("missing 'endif'");
837 }
838 int pattern_count = 0;
839 for (Rule *r = patterns; r; r = r->next) pattern_count += r->recipe != NULL;
840 loaded = true;
841 printf("Makefile loaded: lines=%d variables=%d targets=%d patterns=%d goal=%s\n", body->count, variable_count, target_count, pattern_count, default_goal ? default_goal : "(none)");
842}

851static Choice determine(const char *target) {
852 Rule *explicit = find_rule(target);
853 Choice choice = {0};
854 choice.stem = "";
855 if (explicit && explicit->recipe) {
856 choice.prerequisites = explicit->prerequisites;
857 choice.recipe = explicit->recipe;
858 choice.stem = explicit->stem;
859 return choice;
860 }
861 if (!contains(&phony, target)) {
862 Rule *best = NULL;
863 char *best_stem = NULL;
864 for (Rule *r = patterns; r; r = r->next) if (r->recipe) {
865 char *s = stem(r->name, target);
866 if (!s || !*s || (best_stem && strlen(s) >= strlen(best_stem))) continue;
867 bool usable = true;
868 for (int i = 0; i < r->prerequisites.count; i++) {
869 char *name = substitute(r->prerequisites.items[i], s);
870 if (file_index(name) < 0 && !find_rule(name)) {
871 usable = false;
872 break;
873 }
874 }
875 if (usable) {
876 best = r;
877 best_stem = s;
878 }
879 }
880 if (best) {
881 for (int i = 0; i < best->prerequisites.count; i++) add_unique(&choice.prerequisites, substitute(best->prerequisites.items[i], best_stem), &temporary);
882 if (explicit) for (int i = 0; i < explicit->prerequisites.count; i++) add_unique(&choice.prerequisites, explicit->prerequisites.items[i], &temporary);
883 choice.recipe = best->recipe;
884 choice.stem = best_stem;
885 return choice;
886 }
887 }
888 if (explicit || contains(&phony, target)) {
889 if (explicit) {
890 choice.prerequisites = explicit->prerequisites;
891 choice.stem = explicit->stem;
892 }
893 return choice;
894 }
895 if (file_index(target) >= 0) choice.source = true;
896 else choice.missing = true;
897 return choice;
898}

905static int execute(const char *command) {
906 Strings w = words(command);
907 char *c = w.items[0];
908 if (!strcmp(c, "echo")) {
909 Strings rest = {w.items + 1, w.count - 1, w.count - 1};
910 puts(join(&rest));
911 return 0;
912 }
913 if (!strcmp(c, "touch")) {
914 for (int i = 1; i < w.count; i++) new_time(w.items[i]);
915 return 0;
916 }
917 if (!strcmp(c, "rm")) {
918 bool forced = w.count > 1 && !strcmp(w.items[1], "-f");
919 int status = 0;
920 for (int i = forced ? 2 : 1; i < w.count; i++) {
921 Strings names = wildcard_expansion(w.items[i]);
922 if (!names.count) add_string(&names, w.items[i], &temporary);
923 for (int j = 0; j < names.count; j++) if (!remove_file(names.items[j]) && !forced) {
924 printf("rm: cannot remove '%s': No such file or directory\n", names.items[j]);
925 status = 1;
926 }
927 }
928 return status;
929 }
930 if (!strcmp(c, "false")) return 1;
931 if (!strcmp(c, "exit")) return w.count > 1 ? atoi(w.items[1]) : 0;
932 return 0;
933}

964static int update_target(Build *build, const char *target, Choice *choice) {
965 int old = target_time(target);
966 Strings fresh = {0};
967 for (int i = 0; i < choice->prerequisites.count; i++) {
968 char *p = choice->prerequisites.items[i];
969 int time = target_time(p);
970 if (build->options['B'] || old < 0 || time < 0 || time > old) add_string(&fresh, p, &temporary);
971 }
972 if (!choice->recipe || (!build->options['B'] && old >= 0 && !fresh.count)) return 0;
973 build->runs++;
974 RecipeLine *first = NULL, *last = NULL;
975 has_automatic = true;
976 automatic[0] = (char *)target;
977 automatic[1] = choice->prerequisites.count ? choice->prerequisites.items[0] : "";
978 automatic[2] = join(&choice->prerequisites);
979 automatic[3] = join(&fresh);
980 automatic[4] = choice->stem;
981 for (RecipeLine *line = choice->recipe->first; line; line = line->next) {
982 location = format("Makefile:%d", line->number);
983 RecipeLine *expanded = memory(&temporary, sizeof *expanded);
984 expanded->number = line->number;
985 expanded->text = expand(line->text);
986 if (last) last->next = expanded;
987 else first = expanded;
988 last = expanded;
989 }
990 has_automatic = false;
991 for (RecipeLine *line = first; line; line = line->next) {
992 int i = 0;
993 bool silent = false, ignore = false;
994 while (line->text[i] && strchr(" @-", line->text[i])) {
995 silent |= line->text[i] == '@';
996 ignore |= line->text[i] == '-';
997 i++;
998 }
999 char *command = right_trim(line->text + i);
1000 if (!*command) continue;
1001 if (build->options['n'] || (!silent && !build->options['s'])) puts(command);
1002 if (build->options['n']) continue;
1003 int status = execute(command);
1004 if (status) {
1005 if (ignore || build->options['i']) printf("make: [Makefile:%d: %s] Error %d (ignored)\n", line->number, target, status);
1006 else {
1007 printf("make: *** [Makefile:%d: %s] Error %d\n", line->number, target, status);
1008 if (!build->options['k']) longjmp(command_error, 2);
1009 return 1;
1010 }
1011 }
1012 }
1013 if (!contains(&phony, target)) new_time(target);
1014 return 0;
1015}

1017static int build_target(Build *build, const char *target, const char *parent) {
1018 Result *result = find_result(build, target);
1019 if (result && result->status >= 0) return result->status;
1020 if (!result) {
1021 result = memory(&temporary, sizeof *result);
1022 result->name = (char *)target;
1023 result->status = -1;
1024 result->next = build->results;
1025 build->results = result;
1026 }
1027 result->choice = determine(target);
1028 Choice *choice = &result->choice;
1029 if (choice->missing) {
1030 result->status = 1;
1031 printf("make: *** No rule to make target '%s'", target);
1032 if (parent) printf(", needed by '%s'.", parent);
1033 else putchar('.');
1034 puts(build->options['k'] ? "" : " Stop.");
1035 if (!build->options['k']) longjmp(command_error, 2);
1036 return 1;
1037 }
1038 if (choice->source) return result->status = 0;
1039 Building building = {target, build->building};
1040 build->building = &building;
1041 bool failed = false;
1042 Strings kept = {0};
1043 for (int i = 0; i < choice->prerequisites.count; i++) {
1044 char *p = choice->prerequisites.items[i];
1045 if (being_built(build, p)) printf("make: Circular %s <- %s dependency dropped.\n", target, p);
1046 else {
1047 add_string(&kept, p, &temporary);
1048 if (build_target(build, p, target) != 0) failed = true;
1049 }
1050 }
1051 choice->prerequisites = kept;
1052 build->building = building.previous;
1053 return result->status = failed ? 2 : update_target(build, target, choice);
1054}

1056static void make(const char *args) {
1057 Strings w = words(args), goals = {0};
1058 Build build = {0};
1059 for (int i = 0; i < w.count; i++) {
1060 char *word = w.items[i];
1061 if (*word == '-') for (int j = 1; word[j]; j++) {
1062 if (!strchr("nBkis", word[j])) {
1063 printf("make: invalid option -- '%c'\n", word[j]);
1064 return;
1065 }
1066 build.options[(unsigned char)word[j]] = true;
1067 } else add_string(&goals, word, &temporary);
1068 }
1069 if (!goals.count) {
1070 if (!loaded) {
1071 puts("make: *** No targets specified and no makefile found. Stop.");
1072 return;
1073 }
1074 if (!default_goal) {
1075 puts("make: *** No targets. Stop.");
1076 return;
1077 }
1078 add_string(&goals, default_goal, &temporary);
1079 }
1080 if (build.options['n']) {
1081 saved_files = malloc((size_t)(file_count ? file_count : 1) * sizeof *saved_files);
1082 if (!saved_files) exit(EXIT_FAILURE);
1083 if (file_count) memcpy(saved_files, files, (size_t)file_count * sizeof *files);
1084 saved_count = file_count;
1085 saved_clock = clock_value;
1086 dry_saved = true;
1087 }
1088 for (int i = 0; i < goals.count; i++) {
1089 char *name = goals.items[i];
1090 int previous = build.runs, result = build_target(&build, name, NULL);
1091 if (result == 0 && previous == build.runs) {
1092 if (contains(&phony, name) || !find_result(&build, name)->choice.recipe) printf("make: Nothing to be done for '%s'.\n", name);
1093 else printf("make: '%s' is up to date.\n", name);
1094 } else if (result == 2) printf("make: Target '%s' not remade because of errors.\n", name);
1095 }
1096}

1105static void command(const char *line, const Strings *body) {
1106 printf("$ %s\n", line);
1107 const char *space = strchr(line, ' ');
1108 char *c = space ? part(line, (size_t)(space - line)) : (char *)line;
1109 const char *rest = space ? space + 1 : "";
1110 int status = setjmp(command_error);
1111 if (status == 0) {
1112 if (!strcmp(c, "MAKEFILE")) load(body);
1113 else if (!strcmp(c, "MAKE")) make(rest);
1114 else if (!strcmp(c, "TOUCH")) {
1115 Strings names = words(rest);
1116 for (int i = 0; i < names.count; i++) new_time(names.items[i]);
1117 } else if (!strcmp(c, "RM")) {
1118 Strings names = words(rest);
1119 for (int i = 0; i < names.count; i++) if (!remove_file(names.items[i])) printf("rm: cannot remove '%s': No such file or directory\n", names.items[i]);
1120 } else if (!strcmp(c, "FILES")) {
1121 printf("clock %d\n", clock_value);
1122 for (int i = 0; i < file_count; i++) printf("%s %d\n", files[i].name, files[i].time);
1123 } else if (!strcmp(c, "VARS")) {
1124 if (loaded) for (Variable *v = variables; v; v = v->next) {
1125 char *s = trim(value(v->name));
1126 printf("%s %s%s%s\n", v->name, v->simple ? ":=" : "=", *s ? " " : "", s);
1127 }
1128 } else if (!strcmp(c, "RULES")) {
1129 if (loaded) {
1130 for (Rule *r = rules; r; r = r->next) show_rule(r);
1131 for (Rule *r = patterns; r; r = r->next) if (r->recipe) show_rule(r);
1132 if (phony.count) {
1133 Strings names = sorted(&phony);
1134 printf(".PHONY: %s\n", join(&names));
1135 }
1136 }
1137 } else if (!strcmp(c, "EXPAND")) {
1138 char *text = expand(rest);
1139 printf("[%s]\n", text);
1140 }
1141 } else if (!strcmp(c, "MAKEFILE")) discard();
1142 if (dry_saved) {
1143 free(files);
1144 files = saved_files;
1145 file_count = saved_count;
1146 file_capacity = saved_count;
1147 clock_value = saved_clock;
1148 saved_files = NULL;
1149 dry_saved = false;
1150 }
1151 clean_expansion();
1152}

1154static char *read_line(void) {
1155 size_t length = 0, capacity = 128;
1156 char *line = malloc(capacity);
1157 int ch;
1158 if (!line) exit(EXIT_FAILURE);
1159 while ((ch = getchar()) != EOF && ch != '\n') {
1160 if (length + 1 == capacity) {
1161 capacity *= 2;
1162 char *p = realloc(line, capacity);
1163 if (!p) exit(EXIT_FAILURE);
1164 line = p;
1165 }
1166 line[length++] = (char)ch;
1167 }
1168 line[length] = '\0';
1169 return line;
1170}

1174 free(line);

1176 line = read_line();
1177 if (!strcmp(line, "END FILES")) {
1178 free(line);
