92 static string? Stem(string pattern, string word)
93 {
94 int p = pattern.IndexOf('%');
95 if (p < 0) return pattern == word ? "" : null;
96 string prefix = pattern[..p], suffix = pattern[(p + 1)..];
97 return word.Length >= prefix.Length + suffix.Length && word.StartsWith(prefix, StringComparison.Ordinal) && word.EndsWith(suffix, StringComparison.Ordinal) ? word[prefix.Length..(word.Length - suffix.Length)] : null;
98 }

100 static string Substitute(string pattern, string value)
101 {
102 int p = pattern.IndexOf('%');
103 return p < 0 ? pattern : pattern[..p] + value + pattern[(p + 1)..];
104 }

106 static string Replacement(string word, string pattern, string replacement)
107 {
108 string? stem = Stem(pattern, word);
109 return stem == null ? word : pattern.Contains('%') ? Substitute(replacement, stem) : replacement;
110 }

112 static bool Wildcard(string pattern, string name)
113 {
114 bool[,] match = new bool[pattern.Length + 1, name.Length + 1];
115 match[0, 0] = true;
116 for (int i = 0; i < pattern.Length; i++) for (int j = 0; j <= name.Length; j++) if (match[i, j])
117 {
118 char c = pattern[i];
119 if (c == '*')
120 {
121 match[i + 1, j] = true;
122 if (j < name.Length && name[j] != '/') match[i, j + 1] = true;
123 }
124 else if (j < name.Length && (c == '?' ? name[j] != '/' : c == name[j])) match[i + 1, j + 1] = true;
125 }
126 return match[pattern.Length, name.Length];
127 }

155 static int BracketEnd(string text, int opening)
156 {
157 char open = text[opening], close = open == '(' ? ')' : '}';
158 int depth = 0;
159 for (int i = opening; i < text.Length; i++)
160 {
161 if (text[i] == open) depth++;
162 else if (text[i] == close)
163 {
164 depth--;
165 if (depth == 0) return i;
166 }
167 }
168 Error("unterminated variable reference");
169 return -1;
170 }

172 string Expand(string text)
173 {
174 StringBuilder result = new();
175 for (int i = 0; i < text.Length;)
176 {
177 char c = text[i++];
178 if (c != '$')
179 {
180 result.Append(c);
181 continue;
182 }
183 if (i == text.Length) break;
184 c = text[i++];
185 if (c == '$') result.Append('$');
186 else if (c is '(' or '{')
187 {
188 int end = BracketEnd(text, i - 1);
189 result.Append(Reference(text[i..end], c));
190 i = end + 1;
191 }
192 else result.Append(Value(c.ToString()));
193 }
194 return result.ToString();
195 }

197 string Value(string name)
198 {
199 for (int i = scopes.Count - 1; i >= 0; i--)
200 {
201 Scope scope = scopes[i];
202 if (scope.Arguments == null)
203 {
204 if (scope.Name == name) return scope.Value!;
205 }
206 else if (Numeric(name))
207 {
208 int index = int.Parse(name);
209 return index < scope.Arguments.Count ? scope.Arguments[index] : "";
210 }
211 }
212 if (automatic != null && automatic.TryGetValue(name, out string? auto)) return auto;
213 if (!variables.TryGetValue(name, out Variable? variable)) return "";
214 if (variable.Simple) return variable.Value;
215 if (!expanding.Add(name)) Error($"Recursive variable '{name}' references itself (eventually)");
216 try
217 {
218 return Expand(variable.Value);
219 }
220 finally
221 {
222 expanding.Remove(name);
223 }
224 }

226 string Reference(string content, char opening)
227 {
228 int first = content.IndexOf(' ');
229 if (first >= 0 && Functions.ContainsKey(content[..first])) return Function(content[..first], LeftTrim(content[(first + 1)..]), opening);
230 string name = Expand(content);
231 int colon = name.IndexOf(':'), equal = colon < 0 ? -1 : name.IndexOf('=', colon + 1);
232 if (equal >= 0)
233 {
234 string pattern = name[(colon + 1)..equal], replacement = name[(equal + 1)..];
235 if (!pattern.Contains('%'))
236 {
237 pattern = "%" + pattern;
238 replacement = "%" + replacement;
239 }
240 return string.Join(" ", Words(Value(name[..colon])).Select(word => Replacement(word, pattern, replacement)));
241 }
242 return Value(name);
243 }

245 static List<string> Arguments(string text, int maximum, char opening)
246 {
247 char closing = opening == '(' ? ')' : '}';
248 List<string> result = [];
249 int depth = 0, start = 0;
250 for (int i = 0; i < text.Length; i++)
251 {
252 char c = text[i];
253 if (c == opening) depth++;
254 else if (c == closing) depth--;
255 else if (c == ',' && depth == 0 && result.Count < maximum - 1)
256 {
257 result.Add(text[start..i]);
258 start = i + 1;
259 }
260 }
261 result.Add(text[start..]);
262 return result;
263 }

265 string Function(string name, string text, char opening)
266 {
267 var limits = Functions[name];
268 List<string> raw = Arguments(text, limits.Max, opening);
269 if (raw.Count < limits.Min) Error($"insufficient number of arguments ({raw.Count}) to function '{name}'");
270 if (name == "foreach")
271 {
272 string variable = Trim(Expand(raw[0]));
273 List<string> list = Words(Expand(raw[1])), result = [];
274 foreach (string word in list)
275 {
276 scopes.Add(new(variable, word, null));
277 try
278 {
279 result.Add(Expand(raw[2]));
280 }
281 finally
282 {
283 scopes.RemoveAt(scopes.Count - 1);
284 }
285 }
286 return string.Join(" ", result);
287 }
288 if (name == "if")
289 {
290 bool condition = Trim(Expand(raw[0])).Length != 0;
291 return condition ? Expand(raw[1]) : raw.Count == 3 ? Expand(raw[2]) : "";
292 }
293 List<string> a = [];
294 foreach (string arg in raw)
295 {
296 string v = Expand(arg);
297 a.Add(name is "subst" or "call" ? v : Trim(v));
298 }
299 if (name == "call")
300 {
301 string variable = Trim(a[0]);
302 if (!variables.TryGetValue(variable, out Variable? v)) return "";
303 if (v.Simple) return v.Value;
304 a[0] = variable;
305 scopes.Add(new(null, null, a));
306 try
307 {
308 return Expand(v.Value);
309 }
310 finally
311 {
312 scopes.RemoveAt(scopes.Count - 1);
313 }
314 }
315 if (name == "value") return variables.GetValueOrDefault(a[0])?.Value ?? "";
316 if (name == "info")
317 {
318 output.Add(a[0]);
319 return "";
320 }
321 if (name == "warning")
322 {
323 output.Add(location + ": " + a[0]);
324 return "";
325 }
326 if (name == "error")
327 {
328 Error(a[0]);
329 return "";
330 }
331 if (name == "subst") return a[0].Length == 0 ? a[2] : a[2].Replace(a[0], a[1], StringComparison.Ordinal);
332 if (name == "findstring") return a[1].Contains(a[0], StringComparison.Ordinal) ? a[0] : "";
333 if (name == "word")
334 {
335 string n = a[0];
336 if (!Numeric(n)) Error($"non-numeric first argument to 'word' function: '{n}'");
337 int index = int.Parse(n);
338 if (index == 0) Error("first argument to 'word' function must be greater than 0");
339 List<string> list = Words(a[1]);
340 return index <= list.Count ? list[index - 1] : "";
341 }
342 if (name == "wordlist")
343 {
344 if (!Numeric(a[0])) Error($"non-numeric first argument to 'wordlist' function: '{a[0]}'");
345 if (!Numeric(a[1])) Error($"non-numeric second argument to 'wordlist' function: '{a[1]}'");
346 int start = int.Parse(a[0]), end = int.Parse(a[1]);
347 if (start == 0) Error($"invalid first argument to 'wordlist' function: '{a[0]}'");
348 List<string> list = Words(a[2]);
349 return start > end || start > list.Count ? "" : string.Join(" ", list.Skip(start - 1).Take(end - start + 1));
350 }
351 List<string> values = Words(a[^1]), resultWords = [];
352 switch (name)
353 {
354 case "patsubst":
355 resultWords.AddRange(values.Select(word => Replacement(word, a[0], a[1])));
356 break;
357 case "filter":
358 case "filter-out":
359 {
360 List<string> filters = Words(a[0]);
361 resultWords.AddRange(values.Where(word => filters.Any(p => Stem(p, word) != null) == (name == "filter")));
362 break;
363 }
364 case "strip":
365 resultWords.AddRange(values);
366 break;
367 case "sort":
368 resultWords.AddRange(values.Distinct(StringComparer.Ordinal).OrderBy(v => v, StringComparer.Ordinal));
369 break;
370 case "words":
371 return values.Count.ToString();
372 case "firstword":
373 return values.FirstOrDefault() ?? "";
374 case "lastword":
375 return values.LastOrDefault() ?? "";
376 case "dir":
377 case "notdir":
378 case "suffix":
379 case "basename":
380 foreach (string word in values)
381 {
382 int slash = word.LastIndexOf('/'), dot = word.LastIndexOf('.');
383 bool extension = dot > slash;
384 switch (name)
385 {
386 case "dir":
387 resultWords.Add(slash < 0 ? "./" : word[..(slash + 1)]);
388 break;
389 case "notdir":
390 resultWords.Add(word[(slash + 1)..]);
391 break;
392 case "suffix":
393 if (extension) resultWords.Add(word[dot..]);
394 break;
395 default:
396 resultWords.Add(extension ? word[..dot] : word);
397 break;
398 }
399 }
400 break;
401 case "addsuffix":
402 resultWords.AddRange(values.Select(word => word + a[0]));
403 break;
404 case "addprefix":
405 resultWords.AddRange(values.Select(word => a[0] + word));
406 break;
407 case "wildcard":
408 foreach (string word in values) resultWords.AddRange(WildcardExpansion(word));
409 break;
410 }
411 return string.Join(" ", resultWords);
412 }

414 static (int Equal, int Colon, int Semicolon) Outside(string text)
415 {
416 int equal = -1, colon = -1, semicolon = -1;
417 for (int i = 0; i < text.Length;)
418 {
419 char c = text[i];
420 if (c == '$')
421 {
422 if (i + 1 < text.Length && text[i + 1] is '(' or '{') i = BracketEnd(text, i + 1) + 1;
423 else i += Math.Min(2, text.Length - i);
424 }
425 else
426 {
427 if (c == '=' && equal < 0) equal = i;
428 if (c == ':' && colon < 0) colon = i;
429 if (c == ';' && semicolon < 0) semicolon = i;
430 i++;
431 }
432 }
433 return (equal, colon, semicolon);
434 }

463 static bool Active(List<Condition> conditions) => conditions.Count == 0 || conditions[^1].Outer && conditions[^1].Value;

465 void Conditional(string name, string rest, int line, List<Condition> conditions)
466 {
467 if (name == "else")
468 {
469 if (conditions.Count == 0) Error("extraneous 'else'");
470 Condition top = conditions[^1];
471 if (top.Alternative) Error("only one 'else' per conditional");
472 top.Value = !top.Value;
473 top.Alternative = true;
474 return;
475 }
476 if (name == "endif")
477 {
478 if (conditions.Count == 0) Error("extraneous 'endif'");
479 conditions.RemoveAt(conditions.Count - 1);
480 return;
481 }
482 bool outer = Active(conditions), condition = false;
483 if (outer)
484 {
485 if (name is "ifdef" or "ifndef")
486 {
487 Variable? v = variables.GetValueOrDefault(Trim(Expand(rest)));
488 condition = v != null && v.Value.Length != 0;
489 if (name == "ifndef") condition = !condition;
490 }
491 else
492 {
493 if (rest.Length < 2 || !rest.StartsWith('(') || !rest.EndsWith(')')) Error("invalid syntax in conditional");
494 string inside = rest[1..^1];
495 int depth = 0, comma = -1;
496 for (int i = 0; i < inside.Length; i++)
497 {
498 char c = inside[i];
499 if (c == '(') depth++;
500 else if (c == ')') depth--;
501 else if (c == ',' && depth == 0)
502 {
503 comma = i;
504 break;
505 }
506 }
507 if (comma < 0) Error("invalid syntax in conditional");
508 string a = Trim(Expand(inside[..comma])), b = Trim(Expand(inside[(comma + 1)..]));
509 condition = a == b;
510 if (name == "ifneq") condition = !condition;
511 }
512 }
513 conditions.Add(new(outer, condition, line));
514 }

516 void Load(List<string> body)
517 {
518 Discard();
519 Group? current = null;
520 List<Condition> conditions = [];
521 try
522 {
523 for (int i = 0; i < body.Count; i++)
524 {
525 int number = i + 1;
526 location = "Makefile:" + number;
527 string line = body[i];
528 if (line.StartsWith('>'))
529 {
530 if (!Active(conditions)) continue;
531 if (current == null) Error("recipe commences before first target");
532 AddRecipe(current!, number, LeftTrim(line[1..]));
533 continue;
534 }
535 while (line.EndsWith('\\') && i + 1 < body.Count) line = RightTrim(line[..^1]) + " " + LeftTrim(body[++i]);
536 int comment = line.IndexOf('#');
537 string t = Trim(comment < 0 ? line : line[..comment]);
538 int space = t.IndexOf(' ');
539 string first = space < 0 ? t : t[..space], rest = space < 0 ? "" : Trim(t[(space + 1)..]);
540 if (new[] { "ifeq", "ifneq", "ifdef", "ifndef", "else", "endif" }.Contains(first))
541 {
542 Conditional(first, rest, number, conditions);
543 continue;
544 }
545 if (t.Length == 0 || !Active(conditions)) continue;
546 var positions = Outside(t);
547 int equal = positions.Equal, colon = positions.Colon;
548 if (equal >= 0 && (colon < 0 || colon >= equal - 1))
549 {
550 char op = equal > 0 && ":+?".Contains(t[equal - 1]) ? t[equal - 1] : '=';
551 string name = Trim(t[..(equal - (op == '=' ? 0 : 1))]), rhs = Trim(t[(equal + 1)..]);
552 if (name.Length == 0) Error("empty variable name");
553 Variable? old = variables.GetValueOrDefault(name);
554 if (op == ':') variables[name] = new(true, Expand(rhs));
555 else if (op == '+' && old != null)
556 {
557 string add = old.Simple ? Expand(rhs) : rhs;
558 variables[name] = new(old.Simple, old.Value.Length == 0 ? add : old.Value + " " + add);
559 }
560 else if (op != '?' || old == null) variables[name] = new(false, rhs);
561 current = null;
562 }
563 else if (colon >= 0)
564 {
565 List<string> targets = Unique(Words(Expand(t[..colon])));
566 string r = t[(colon + 1)..];
567 string? inline = null;
568 int semicolon = Outside(r).Semicolon;
569 if (semicolon >= 0)
570 {
571 inline = LeftTrim(r[(semicolon + 1)..]);
572 r = r[..semicolon];
573 }
574 int staticColon = Outside(r).Colon;
575 string? targetPattern = null;
576 if (staticColon >= 0)
577 {
578 targetPattern = Trim(Expand(r[..staticColon]));
579 r = r[(staticColon + 1)..];
580 }
581 List<string> prerequisites = Words(Expand(r));
582 current = new();
583 if (targets.Count == 1 && targets[0] == ".PHONY") phony.UnionWith(prerequisites);
584 else if (targetPattern != null)
585 {
586 if (!targetPattern.Contains('%')) Error("target pattern contains no '%'");
587 foreach (string target in targets)
588 {
589 string? stem = Stem(targetPattern, target);
590 if (stem == null || stem.Length == 0) output.Add($"{location}: target '{target}' doesn't match the target pattern");
591 else
592 {
593 Rule rule = Register(target, prerequisites.Select(p => Substitute(p, stem)).ToList());
594 rule.Stem = stem;
595 current.Rules.Add(rule);
596 }
597 }
598 }
599 else if (targets.Any(p => p.Contains('%')))
600 {
601 if (targets.Any(p => !p.Contains('%'))) Error("mixed implicit and normal rules");
602 foreach (string target in targets)
603 {
604 Rule rule = new(target, true);
605 rule.Prerequisites.AddRange(prerequisites);
606 patterns.Add(rule);
607 current.Rules.Add(rule);
608 }
609 }
610 else foreach (string target in targets) current.Rules.Add(Register(target, prerequisites));
611 if (inline != null) AddRecipe(current, number, inline);
612 }
613 else if (Trim(Expand(t)).Length != 0) Error("missing separator");
614 }
615 if (conditions.Count != 0)
616 {
617 location = "Makefile:" + conditions[^1].Line;
618 Error("missing 'endif'");
619 }
620 loaded = true;
621 output.Add($"Makefile loaded: lines={body.Count} variables={variables.Count} targets={rules.Count} patterns={patterns.Count(p => p.Recipe != null)} goal={goal ?? "(none)"}");
622 }
623 catch (ExpansionError e)
624 {
625 ExpansionMessage(e);
626 Discard();
627 }
628 finally
629 {
630 CleanExpansion();
631 }
632 }

634 Choice Determine(string target)
635 {
636 Rule? explicitRule = rules.GetValueOrDefault(target);
637 Choice choice = new();
638 if (explicitRule?.Recipe != null)
639 {
640 choice.Prerequisites.AddRange(explicitRule.Prerequisites);
641 choice.Recipe = explicitRule.Recipe;
642 choice.Stem = explicitRule.Stem;
643 return choice;
644 }
645 if (!phony.Contains(target))
646 {
647 Rule? best = null;
648 string? bestStem = null;
649 foreach (Rule pattern in patterns) if (pattern.Recipe != null)
650 {
651 string? stem = Stem(pattern.Name, target);
652 if (stem == null || stem.Length == 0 || bestStem != null && stem.Length >= bestStem.Length) continue;
653 bool usable = pattern.Prerequisites.All(p => files.ContainsKey(Substitute(p, stem)) || rules.ContainsKey(Substitute(p, stem)));
654 if (usable)
655 {
656 best = pattern;
657 bestStem = stem;
658 }
659 }
660 if (best != null)
661 {
662 choice.Prerequisites.AddRange(best.Prerequisites.Select(p => Substitute(p, bestStem!)));
663 if (explicitRule != null) choice.Prerequisites.AddRange(explicitRule.Prerequisites);
664 choice.Prerequisites = Unique(choice.Prerequisites);
665 choice.Recipe = best.Recipe;
666 choice.Stem = bestStem!;
667 return choice;
668 }
669 }
670 if (explicitRule != null || phony.Contains(target))
671 {
672 if (explicitRule != null)
673 {
674 choice.Prerequisites.AddRange(explicitRule.Prerequisites);
675 choice.Stem = explicitRule.Stem;
676 }
677 return choice;
678 }
679 if (files.ContainsKey(target)) choice.Source = true;
680 else choice.Missing = true;
681 return choice;
682 }

685 int Execute(string command)
686 {
687 List<string> w = Words(command);
688 switch (w[0])
689 {
690 case "echo":
691 output.Add(string.Join(" ", w.Skip(1)));
692 return 0;
693 case "touch":
694 foreach (string name in w.Skip(1)) NewTime(name);
695 return 0;
696 case "rm":
697 {
698 bool force = w.Count > 1 && w[1] == "-f";
699 int status = 0;
700 foreach (string pattern in w.Skip(force ? 2 : 1))
701 {
702 List<string> names = WildcardExpansion(pattern);
703 if (names.Count == 0) names = [pattern];
704 foreach (string name in names) if (!files.Remove(name) && !force)
705 {
706 output.Add($"rm: cannot remove '{name}': No such file or directory");
707 status = 1;
708 }
709 }
710 return status;
711 }
712 case "false":
713 return 1;
714 case "exit":
715 return w.Count > 1 ? int.Parse(w[1]) : 0;
716 default:
717 return 0;
718 }
719 }

723 readonly Dictionary<string, int> results = new(StringComparer.Ordinal);
724 readonly HashSet<string> building = new(StringComparer.Ordinal);
725 public Dictionary<string, Choice> Choices = new(StringComparer.Ordinal);
726 public int Runs;
727 bool Has(char option) => options.Contains(option);
728 public int Target(string name, string? parent)
729 {
730 if (results.TryGetValue(name, out int recorded)) return recorded;
731 Choice choice = runner.Determine(name);
732 Choices[name] = choice;
733 if (choice.Missing)
734 {
735 results[name] = 1;
736 runner.output.Add($"make: *** No rule to make target '{name}'" + (parent == null ? "." : $", needed by '{parent}'.") + (Has('k') ? "" : " Stop."));
737 if (!Has('k')) throw new AbortBuild();
738 return 1;
739 }
740 if (choice.Source)
741 {
742 results[name] = 0;
743 return 0;
744 }
745 building.Add(name);
746 bool failed = false;
747 List<string> kept = [];
748 foreach (string p in choice.Prerequisites)
749 {
750 if (building.Contains(p)) runner.output.Add($"make: Circular {name} <- {p} dependency dropped.");
751 else
752 {
753 kept.Add(p);
754 if (Target(p, name) != 0) failed = true;
755 }
756 }
757 choice.Prerequisites = kept;
758 building.Remove(name);
759 int result = failed ? 2 : Update(name, choice);
760 results[name] = result;
761 return result;
762 }
763 int Update(string target, Choice choice)
764 {
765 int? old = runner.TargetTime(target);
766 List<string> fresh = [];
767 foreach (string p in choice.Prerequisites)
768 {
769 int? time = runner.TargetTime(p);
770 if (Has('B') || !old.HasValue || !time.HasValue || time.Value > old.Value) fresh.Add(p);
771 }
772 if (choice.Recipe == null || !Has('B') && old.HasValue && fresh.Count == 0) return 0;
773 Runs++;
774 List<RecipeLine> expanded = [];
775 runner.automatic = new(StringComparer.Ordinal) { ["@"] = target, ["<"] = choice.Prerequisites.FirstOrDefault() ?? "", ["^"] = string.Join(" ", choice.Prerequisites), ["?"] = string.Join(" ", fresh), ["*"] = choice.Stem };
776 try
777 {
778 foreach (RecipeLine line in choice.Recipe.Lines)
779 {
780 runner.location = "Makefile:" + line.Number;
781 expanded.Add(new(line.Number, runner.Expand(line.Text)));
782 }
783 }
784 finally
785 {
786 runner.automatic = null;
787 }
788 foreach (RecipeLine line in expanded)
789 {
790 int i = 0;
791 bool silent = false, ignore = false;
792 while (i < line.Text.Length && " @-".Contains(line.Text[i]))
793 {
794 silent |= line.Text[i] == '@';
795 ignore |= line.Text[i] == '-';
796 i++;
797 }
798 string command = RightTrim(line.Text[i..]);
799 if (command.Length == 0) continue;
800 if (Has('n') || !silent && !Has('s')) runner.output.Add(command);
801 if (Has('n')) continue;
802 int status = runner.Execute(command);
803 if (status != 0)
804 {
805 if (ignore || Has('i')) runner.output.Add($"make: [Makefile:{line.Number}: {target}] Error {status} (ignored)");
806 else
807 {
808 runner.output.Add($"make: *** [Makefile:{line.Number}: {target}] Error {status}");
809 if (!Has('k')) throw new AbortBuild();
810 return 1;
811 }
812 }
813 }
814 if (!runner.phony.Contains(target)) runner.NewTime(target);
815 return 0;
816 }
817 }

819 void Make(string args)
820 {
821 List<string> goals = [];
822 HashSet<char> options = [];
823 foreach (string word in Words(args))
824 {
825 if (word.StartsWith('-'))
826 {
827 foreach (char option in word.Skip(1))
828 {
829 if (!"nBkis".Contains(option))
830 {
831 output.Add($"make: invalid option -- '{option}'");
832 return;
833 }
834 options.Add(option);
835 }
836 }
837 else goals.Add(word);
838 }
839 if (goals.Count == 0)
840 {
841 if (!loaded)
842 {
843 output.Add("make: *** No targets specified and no makefile found. Stop.");
844 return;
845 }
846 if (goal == null)
847 {
848 output.Add("make: *** No targets. Stop.");
849 return;
850 }
851 goals.Add(goal);
852 }
853 SortedDictionary<string, int>? saved = options.Contains('n') ? new(files, StringComparer.Ordinal) : null;
854 int oldClock = clock;
855 Build build = new(this, options);
856 try
857 {
858 foreach (string name in goals)
859 {
860 int previous = build.Runs, result = build.Target(name, null);
861 if (result == 0 && previous == build.Runs) output.Add(phony.Contains(name) || build.Choices[name].Recipe == null ? $"make: Nothing to be done for '{name}'." : $"make: '{name}' is up to date.");
862 else if (result == 2) output.Add($"make: Target '{name}' not remade because of errors.");
863 }
864 }
865 catch (AbortBuild) { } catch (ExpansionError e)
866 {
867 ExpansionMessage(e);
868 }
869 finally
870 {
871 if (saved != null)
872 {
873 files = saved;
874 clock = oldClock;
875 }
876 CleanExpansion();
877 }
878 }

880 void Command(string line, List<string> body)
881 {
882 output = ["$ " + line];
883 int space = line.IndexOf(' ');
884 string c = space < 0 ? line : line[..space], rest = space < 0 ? "" : line[(space + 1)..];
885 try
886 {
887 switch (c)
888 {
889 case "MAKEFILE":
890 Load(body);
891 break;
892 case "MAKE":
893 Make(rest);
894 break;
895 case "TOUCH":
896 foreach (string name in Words(rest)) NewTime(name);
897 break;
898 case "RM":
899 foreach (string name in Words(rest)) if (!files.Remove(name)) output.Add($"rm: cannot remove '{name}': No such file or directory");
900 break;
901 case "FILES":
902 output.Add("clock " + clock);
903 foreach ((string name, int value) in files) output.Add($"{name} {value}");
904 break;
905 case "VARS":
906 if (loaded) foreach ((string name, Variable variable) in variables)
907 {
908 string value = Trim(Value(name));
909 output.Add(name + (variable.Simple ? " :=" : " =") + (value.Length == 0 ? "" : " " + value));
910 }
911 break;
912 case "RULES":
913 if (loaded)
914 {
915 List<Rule> all = [.. rules.Values, .. patterns.Where(p => p.Recipe != null)];
916 foreach (Rule rule in all)
917 {
918 output.Add(rule.Name + ":" + (rule.Prerequisites.Count == 0 ? "" : " " + string.Join(" ", rule.Prerequisites)));
919 if (rule.Recipe != null) foreach (RecipeLine recipe in rule.Recipe.Lines) output.Add(">" + recipe.Text);
920 }
921 if (phony.Count != 0) output.Add(".PHONY: " + string.Join(" ", phony));
922 }
923 break;
924 case "EXPAND":
925 output.Add("[" + Expand(rest) + "]");
926 break;
927 }
928 }
929 catch (ExpansionError e)
930 {
931 ExpansionMessage(e);
932 }
933 finally
934 {
935 CleanExpansion();
936 }
937 foreach (string row in output) Console.WriteLine(row);
938 }

940 static void Main()
941 {
942 CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
943 Program runner = new();
944 Console.ReadLine();
945 string line;
946 while ((line = Console.ReadLine()!) != "END FILES")

955 if (line == "MAKEFILE")

959 }
960 runner.Command(line, body);
961 }
