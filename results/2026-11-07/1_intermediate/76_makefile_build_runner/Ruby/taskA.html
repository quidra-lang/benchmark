61 def closing(text, opening)
62 left = text[opening]
63 right = left == '(' ? ')' : '}'
64 depth = 0
65 (opening...text.length).each do |i|
66 depth += 1 if text[i] == left
67 depth -= 1 if text[i] == right
68 return i if depth.zero?
69 end
70 error('unterminated variable reference')
71 end

73 def outside(text)
74 result = {}
75 i = 0
76 while i < text.length
77 if text[i] == '$'
78 i = ['(', '{'].include?(text[i + 1]) ? closing(text, i + 1) + 1 : i + 2
79 else
80 result[text[i]] ||= i
81 i += 1
82 end
83 end
84 result
85 end

87 def stem(pattern, text)
88 percent = pattern.index('%')
89 return pattern == text ? '' : nil unless percent
90 left = pattern[0...percent]
91 right = pattern[(percent + 1)..]
92 return nil unless text.length >= left.length + right.length && text.start_with?(left) && text.end_with?(right)
93 text[left.length, text.length - left.length - right.length]
94 end

96 def substitute(text, value)

100 def replace(word, pattern, replacement)
101 value = stem(pattern, word)
102 return word unless value
103 pattern.include?('%') ? substitute(replacement, value) : replacement
104 end

106 def wildcard(pattern)
107 unless pattern.include?('*') || pattern.include?('?')
108 return @files.key?(pattern) ? [pattern] : []
109 end
110 regex = Regexp.new('\\A' + Regexp.escape(pattern).gsub('\\*', '[^/]*').gsub('\\?', '[^/]') + '\\z')
111 @files.keys.select { |name| regex.match?(name) }.sort

114 def reference_value(name)
115 @scopes.reverse_each do |scope|
116 if scope[0] == :foreach
117 return scope[2] if scope[1] == name
118 elsif /\A[0-9]+\z/.match?(name)
119 return scope[1][name.to_i] || ''
120 end
121 end
122 return @automatic[name] if @automatic && @automatic.key?(name)
123 variable_value(name)
124 end

126 def variable_value(name)
127 variable = @variables[name]
128 return '' unless variable
129 return variable[1] if variable[0] == ':='
130 error("Recursive variable '#{name}' references itself (eventually)") if @active_variables[name]
131 @active_variables[name] = true

133 expand(variable[1])
134 ensure
135 @active_variables.delete(name)
136 end

139 def expand(text)
140 result = +''
141 i = 0
142 while i < text.length
143 if text[i] != '$'
144 result << text[i]
145 i += 1
146 elsif text[i + 1] == '$'
147 result << '$'
148 i += 2
149 elsif ['(', '{'].include?(text[i + 1])
150 finish = closing(text, i + 1)
151 result << reference(text[(i + 2)...finish], text[i + 1])
152 i = finish + 1
153 else
154 result << reference_value(text[i + 1]) if i + 1 < text.length
155 i += 2
156 end
157 end
158 result
159 end

161 def reference(content, bracket)
162 space = content.index(' ')
163 if space && FUNCTIONS.key?(content[0...space])
164 return function(content[0...space], content[(space + 1)..].sub(/\A +/, ''), bracket)
165 end
166 name = expand(content)
167 colon = name.index(':')
168 equals = colon ? name.index('=', colon + 1) : nil
169 return reference_value(name) unless equals
170 variable = name[0...colon]
171 pattern = name[(colon + 1)...equals]
172 replacement = name[(equals + 1)..]
173 unless pattern.include?('%')
174 pattern = '%' + pattern
175 replacement = '%' + replacement
176 end
177 words(reference_value(variable)).map { |word| replace(word, pattern, replacement) }.join(' ')
178 end

180 def function(name, text, bracket)
181 minimum, maximum = FUNCTIONS[name]
182 args = []
183 depth = 0
184 start = 0
185 close = bracket == '(' ? ')' : '}'
186 text.each_char.with_index do |char, i|
187 depth += 1 if char == bracket
188 depth -= 1 if char == close
189 if char == ',' && depth.zero? && (!maximum || args.length < maximum - 1)
190 args << text[start...i]
191 start = i + 1
192 end
193 end
194 args << text[start..]
195 error("insufficient number of arguments (#{args.length}) to function '#{name}'") if args.length < minimum
196 if name == 'foreach'
197 variable = trim(expand(args[0]))
198 list = words(expand(args[1]))
199 return list.map do |word|
200 @scopes << [:foreach, variable, word]
201 begin
202 expand(args[2])
203 ensure
204 @scopes.pop
205 end
206 end.join(' ')
207 end
208 if name == 'if'
209 condition = trim(expand(args[0]))
210 return expand(condition.empty? ? (args[2] || '') : args[1])
211 end
212 if name == 'call'
213 values = args.map { |arg| expand(arg) }
214 variable = trim(values.shift)
215 entry = @variables[variable]
216 return '' unless entry
217 return entry[1] if entry[0] == ':='
218 @scopes << [:call, [variable] + values]
219 begin
220 return expand(entry[1])
221 ensure
222 @scopes.pop
223 end
224 end
225 values = args.map { |arg| expand(arg) }
226 values.map! { |arg| trim(arg) } unless name == 'subst'
227 a, b, c = values
228 case name
229 when 'subst'
230 a.empty? ? c : c.gsub(a) { b }
231 when 'patsubst'
232 words(c).map { |word| replace(word, a, b) }.join(' ')
233 when 'strip'
234 words(a).join(' ')
235 when 'findstring'
236 b.include?(a) ? a : ''
237 when 'filter', 'filter-out'
238 patterns = words(a)
239 words(b).select { |word| patterns.any? { |pattern| !stem(pattern, word).nil? } == (name == 'filter') }.join(' ')
240 when 'sort'
241 words(a).uniq.sort.join(' ')
242 when 'word'
243 error("non-numeric first argument to 'word' function: '#{a}'") unless /\A[0-9]+\z/.match?(a)
244 error("first argument to 'word' function must be greater than 0") if a.to_i.zero?
245 words(b)[a.to_i - 1] || ''
246 when 'wordlist'
247 error("non-numeric first argument to 'wordlist' function: '#{a}'") unless /\A[0-9]+\z/.match?(a)
248 error("non-numeric second argument to 'wordlist' function: '#{b}'") unless /\A[0-9]+\z/.match?(b)
249 error("invalid first argument to 'wordlist' function: '#{a}'") if a.to_i.zero?
250 b.to_i < a.to_i ? '' : words(c).drop(a.to_i - 1).take(b.to_i - a.to_i + 1).join(' ')
251 when 'words'
252 words(a).length.to_s
253 when 'firstword'
254 words(a).first || ''
255 when 'lastword'
256 words(a).last || ''
257 when 'dir'
258 words(a).map { |word| word.include?('/') ? word[0..word.rindex('/')] : './' }.join(' ')
259 when 'notdir'
260 words(a).map { |word| word[(word.rindex('/') || -1) + 1..] }.join(' ')
261 when 'suffix', 'basename'
262 words(a).filter_map do |word|
263 dot = word.rindex('.')
264 slash = word.rindex('/') || -1
265 extension = dot && dot > slash
266 name == 'suffix' ? (extension ? word[dot..] : nil) : (extension ? word[0...dot] : word)
267 end.join(' ')
268 when 'addsuffix'
269 words(b).map { |word| word + a }.join(' ')
270 when 'addprefix'
271 words(b).map { |word| a + word }.join(' ')
272 when 'wildcard'
273 words(a).flat_map { |pattern| wildcard(pattern) }.join(' ')
274 when 'value'
275 @variables[a]&.[](1) || ''
276 when 'info'
277 puts a
278 ''
279 when 'warning'
280 puts "#{@location}: #{a}"
281 ''
282 when 'error'
283 error(a)
284 end
285 end

295 def add_recipe(current, number, text)
296 error('recipe commences before first target') unless current
297 return if current[:entries].empty?
298 unless current[:recipe]
299 recipe = []
300 current[:entries].each do |rule|
301 if current[:explicit] && rule.recipe
302 puts "Makefile:#{number}: warning: overriding recipe for target '#{rule.target}'"
303 puts "Makefile:#{rule.recipe.first[0]}: warning: ignoring old recipe for target '#{rule.target}'"
304 end
305 rule.recipe = recipe
306 end
307 current[:recipe] = recipe
308 end
309 current[:recipe] << [number, text]
310 end

312 def parse_rule(text, colon, number)
313 targets = words(expand(text[0...colon])).uniq
314 rest = text[(colon + 1)..]
315 semicolon = outside(rest)[';']
316 inline = semicolon ? rest[(semicolon + 1)..].sub(/\A +/, '') : nil
317 rest = rest[0...semicolon] if semicolon
318 second = outside(rest)[':']
319 pattern = second ? trim(expand(rest[0...second])) : nil
320 rest = rest[(second + 1)..] if second
321 prerequisites = words(expand(rest))
322 entries = []
323 explicit = true
324 if targets == ['.PHONY']
325 prerequisites.each { |target| @phony[target] = true }
326 elsif pattern
327 error("target pattern contains no '%'") unless pattern.include?('%')
328 targets.each do |target|
329 value = stem(pattern, target)
330 if value && !value.empty?
331 entry = register(target, prerequisites.map { |p| substitute(p, value) })
332 entry.stem = value
333 entries << entry
334 else
335 puts "Makefile:#{number}: target '#{target}' doesn't match the target pattern"
336 end
337 end
338 elsif targets.any? { |target| target.include?('%') }
339 error('mixed implicit and normal rules') unless targets.all? { |target| target.include?('%') }
340 explicit = false
341 targets.each do |target|
342 entry = Rule.new(target, prerequisites.dup, nil, '')
343 @patterns << entry
344 entries << entry
345 end
346 else
347 entries = targets.map { |target| register(target, prerequisites) }
348 end
349 current = { entries: entries, explicit: explicit, recipe: nil }
350 add_recipe(current, number, inline) unless inline.nil?
351 current
352 end

354 def load_makefile(body)
355 clear_makefile
356 current = nil
357 conditions = []
358 i = 0
359 while i < body.length
360 number = i + 1
361 @location = "Makefile:#{number}"
362 active = conditions.empty? || (conditions.last[:outer] && conditions.last[:value])
363 text = body[i]
364 i += 1
365 if text.start_with?('>')
366 add_recipe(current, number, text[1..].sub(/\A +/, '')) if active
367 next
368 end
369 while text.end_with?('\\') && i < body.length
370 text = text[0...-1].sub(/ +\z/, '') + ' ' + body[i].sub(/\A +/, '')
371 i += 1
372 end
373 text = trim(text.split('#', 2).first || '')
374 first, rest = text.split(' ', 2)
375 rest = trim(rest || '')
376 if %w[ifdef ifndef ifeq ifneq].include?(first)
377 value = false
378 if active
379 if %w[ifdef ifndef].include?(first)
380 variable = @variables[trim(expand(rest))]
381 value = !variable.nil? && !variable[1].empty?
382 value = !value if first == 'ifndef'
383 else
384 error('invalid syntax in conditional') unless rest.length >= 2 && rest.start_with?('(') && rest.end_with?(')')
385 inner = rest[1...-1]
386 depth = 0
387 comma = nil
388 inner.each_char.with_index do |char, p|
389 depth += 1 if char == '('
390 depth -= 1 if char == ')'
391 if char == ',' && depth.zero?
392 comma = p
393 break
394 end
395 end
396 error('invalid syntax in conditional') unless comma
397 left = trim(expand(inner[0...comma]))
398 right = trim(expand(inner[(comma + 1)..]))
399 value = left == right
400 value = !value if first == 'ifneq'
401 end
402 end
403 conditions << { outer: active, value: value, seen_else: false, number: number }
404 elsif first == 'else'
405 error("extraneous 'else'") if conditions.empty?
406 error("only one 'else' per conditional") if conditions.last[:seen_else]
407 conditions.last[:value] = !conditions.last[:value]
408 conditions.last[:seen_else] = true
409 elsif first == 'endif'
410 error("extraneous 'endif'") if conditions.empty?
411 conditions.pop
412 elsif active && !text.empty?
413 positions = outside(text)
414 equals = positions['=']
415 colon = positions[':']
416 if equals && (!colon || colon >= equals - 1)
417 operator_start = equals.positive? && [':', '+', '?'].include?(text[equals - 1]) ? equals - 1 : equals
418 operator = text[operator_start..equals]
419 variable = trim(text[0...operator_start])
420 error('empty variable name') if variable.empty?
421 rhs = trim(text[(equals + 1)..])
422 old = @variables[variable]
423 if operator == ':='
424 @variables[variable] = [':=', expand(rhs)]
425 elsif operator == '+=' && old
426 addition = old[0] == ':=' ? expand(rhs) : rhs
427 @variables[variable] = [old[0], old[1].empty? ? addition : old[1] + ' ' + addition]
428 elsif operator != '?=' || !old
429 @variables[variable] = ['=', rhs]
430 end
431 current = nil
432 elsif colon
433 current = parse_rule(text, colon, number)
434 else
435 error('missing separator') unless trim(expand(text)).empty?
436 end
437 end
438 end
439 unless conditions.empty?
440 @location = "Makefile:#{conditions.last[:number]}"
441 error("missing 'endif'")
442 end
443 @loaded = true
444 puts "Makefile loaded: lines=#{body.length} variables=#{@variables.length} targets=#{@rules.length} patterns=#{@patterns.count(&:recipe)} goal=#{@default || '(none)'}"
445 rescue MakeError => e
446 report(e)
447 clear_makefile
448 ensure
449 @location = 'make'
450 end

469 def determine(target)
470 explicit = @rules[target]
471 return [explicit.prerequisites.dup, explicit.recipe, explicit.stem, :rule] if explicit&.recipe
472 unless @phony[target]
473 candidates = @patterns.each_with_index.filter_map do |rule, i|
474 value = rule.recipe ? stem(rule.target, target) : nil
475 [value.length, i, rule, value] if value && !value.empty?
476 end.sort_by { |length, i, _, _| [length, i] }
477 candidates.each do |_, _, rule, value|
478 prerequisites = rule.prerequisites.map { |p| substitute(p, value) }
479 if prerequisites.all? { |p| @files.key?(p) || @rules.key?(p) }
480 return [(prerequisites + (explicit&.prerequisites || [])).uniq, rule.recipe, value, :rule]
481 end
482 end
483 end
484 return [explicit&.prerequisites&.dup || [], nil, explicit&.stem || '', :rule] if explicit || @phony[target]
485 [[], nil, '', @files.key?(target) ? :source : :missing]
486 end

492 def simulate(command)
493 parts = words(command)
494 first = parts.shift
495 case first
496 when 'echo'
497 puts parts.join(' ')
498 0
499 when 'touch'
500 parts.each { |name| new_time(name) }
501 0
502 when 'rm'
503 forced = parts.first == '-f'
504 parts.shift if forced
505 result = 0
506 parts.each do |pattern|
507 matches = wildcard(pattern)
508 matches = [pattern] if matches.empty?
509 matches.each { |name| result = [result, remove_file(name, forced)].max }
510 end
511 result
512 when 'false'
513 1
514 when 'exit'
515 parts.first.to_i
516 else
517 0
518 end
519 end

521 def run_recipe(target, prerequisites, recipe, value, newer, options)
522 @automatic = { '@' => target, '<' => prerequisites.first || '', '^' => prerequisites.join(' '), '?' => newer.join(' '), '*' => value }
523 expanded = recipe.map do |number, text|
524 @location = "Makefile:#{number}"
525 [number, expand(text)]
526 end
527 expanded.each do |number, text|
528 prefix = text[/\A[ @-]*/]
529 command = text[prefix.length..].sub(/ +\z/, '')
530 next if command.empty?
531 puts command if options['n'] || (!prefix.include?('@') && !options['s'])
532 next if options['n']
533 code = simulate(command)
534 next if code.zero?
535 if prefix.include?('-') || options['i']
536 puts "make: [Makefile:#{number}: #{target}] Error #{code} (ignored)"
537 else
538 puts "make: *** [Makefile:#{number}: #{target}] Error #{code}"
539 raise BuildAbort unless options['k']
540 return :failure
541 end
542 end
543 :success
544 ensure
545 @automatic = nil
546 @location = 'make'
547 end

549 def make(args)
550 options = {}
551 goals = []
552 words(args).each do |word|
553 if word.start_with?('-')
554 word[1..].each_char do |char|
555 unless 'nBkis'.include?(char)
556 puts "make: invalid option -- '#{char}'"
557 return
558 end
559 options[char] = true
560 end
561 else
562 goals << word
563 end
564 end
565 if goals.empty?
566 unless @loaded
567 puts 'make: *** No targets specified and no makefile found. Stop.'
568 return
569 end
570 unless @default
571 puts 'make: *** No targets. Stop.'
572 return
573 end
574 goals = [@default]
575 end
576 saved_files = @files.dup if options['n']
577 saved_clock = @clock if options['n']
578 results = {}
579 used = {}
580 building = {}
581 count = 0
582 build = lambda do |target, parent|
583 return results[target] if results.key?(target)
584 prerequisites, recipe, value, kind = determine(target)
585 used[target] = [recipe, kind]
586 if kind == :missing
587 results[target] = :failure
588 message = "make: *** No rule to make target '#{target}'"
589 message += ", needed by '#{parent}'" if parent
590 puts message + '.' + (options['k'] ? '' : ' Stop.')
591 raise BuildAbort unless options['k']
592 return :failure
593 end
594 if kind == :source
595 results[target] = :success
596 return :success
597 end
598 building[target] = true
599 failed = false
600 kept = []
601 prerequisites.each do |prerequisite|
602 if building[prerequisite]
603 puts "make: Circular #{target} <- #{prerequisite} dependency dropped."
604 else
605 kept << prerequisite
606 failed = true unless build.call(prerequisite, target) == :success
607 end
608 end
609 building.delete(target)
610 if failed
611 result = :dependency_failure
612 else
613 time = target_time(target)
614 newer = kept.select { |p| options['B'] || time.nil? || target_time(p).nil? || target_time(p) > time }
615 if recipe.nil? || (!options['B'] && !time.nil? && newer.empty?)
616 result = :success
617 else
618 count += 1
619 result = run_recipe(target, kept, recipe, value, newer, options)
620 new_time(target) if result == :success && !@phony[target]
621 end
622 end
623 results[target] = result
624 end
625 goals.each do |goal|
626 before = count
627 result = build.call(goal, nil)
628 if result == :success && count == before
629 recipe, = used[goal]
630 puts(@phony[goal] || recipe.nil? ? "make: Nothing to be done for '#{goal}'." : "make: '#{goal}' is up to date.")
631 elsif result == :dependency_failure
632 puts "make: Target '#{goal}' not remade because of errors."
633 end
634 end
635 rescue BuildAbort
636 nil
637 rescue MakeError => e
638 report(e)
639 ensure
640 if saved_files
641 @files = saved_files
642 @clock = saved_clock
643 end
644 @location = 'make'
645 @automatic = nil
646 end

648 def execute(line, body = nil)
649 puts '$ ' + line
650 command, args = line.split(' ', 2)
651 args ||= ''
652 case command
653 when 'MAKEFILE'
654 load_makefile(body)
655 when 'MAKE'
656 make(args)
657 when 'TOUCH'
658 words(args).each { |name| new_time(name) }
659 when 'RM'
660 words(args).each { |name| remove_file(name) }
661 when 'FILES'
662 puts "clock #{@clock}"
663 @files.keys.sort.each { |name| puts "#{name} #{@files[name]}" }
664 when 'VARS'
665 return unless @loaded
666 @variables.keys.sort.each do |name|
667 value = trim(variable_value(name))
668 puts "#{name} #{@variables[name][0]}#{value.empty? ? '' : ' ' + value}"
669 end
670 when 'RULES'
671 return unless @loaded
672 (@rules.keys.sort.map { |name| @rules[name] } + @patterns.select(&:recipe)).each do |rule|
673 puts rule.target + ':' + rule.prerequisites.map { |p| ' ' + p }.join
674 rule.recipe&.each { |_, text| puts '>' + text }
675 end
676 puts '.PHONY: ' + @phony.keys.sort.join(' ') unless @phony.empty?
677 when 'EXPAND'
678 puts '[' + expand(args) + ']'
679 end
680 rescue MakeError => e
681 report(e)
682 end
683end
