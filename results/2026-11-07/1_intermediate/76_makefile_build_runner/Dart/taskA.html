71final output = <String>[];
72String leftTrim(String text) {
73 var i = 0;
74 while (i < text.length && text[i] == ' ') { i++; }
75 return text.substring(i);
76}
77String rightTrim(String text) {
78 var i = text.length;
79 while (i > 0 && text[i - 1] == ' ') { i--; }
80 return text.substring(0, i);
81}
82String trim(String text) => rightTrim(leftTrim(text));
83List<String> words(String text) => text.split(' ').where((word) => word.isNotEmpty).toList();
84List<String> unique(List<String> items) => items.toSet().toList();
85bool numeric(String text) => text.isNotEmpty && !RegExp(r'[^0-9]').hasMatch(text);
86String substitute(String text, String stem) {
87 final i = text.indexOf('%');
88 return i < 0 ? text : text.substring(0, i) + stem + text.substring(i + 1);
89}
90String? patternStem(String pattern, String word) {
91 final i = pattern.indexOf('%');
92 if (i < 0) return pattern == word ? '' : null;
93 final before = pattern.substring(0, i), after = pattern.substring(i + 1);
94 return word.length >= before.length + after.length && word.startsWith(before) && word.endsWith(after)
95 ? word.substring(before.length, word.length - after.length) : null;
96}
97String replaceWord(String word, String pattern, String replacement) {
98 final stem = patternStem(pattern, word);
99 return stem == null ? word : pattern.contains('%') ? substitute(replacement, stem) : replacement;
100}
101bool wildcardMatch(String pattern, String name) {
102 var previous = List<bool>.filled(name.length + 1, false);
103 previous[0] = true;
104 for (final c in pattern.split('')) {
105 final next = List<bool>.filled(name.length + 1, false);
106 for (var j = 0; j <= name.length; j++) {
107 next[j] = c == '*' ? previous[j] || j > 0 && name[j - 1] != '/' && next[j - 1]
108 : j > 0 && previous[j - 1] && (c == '?' ? name[j - 1] != '/' : c == name[j - 1]);
109 }
110 previous = next;
111 }
112 return previous[name.length];
113}

137 List<String> wildcard(String pattern) => (files.keys.toList()..sort())

139 int closing(String text, int dollar) {
140 final open = text[dollar + 1], close = open == '(' ? ')' : '}';
141 var depth = 0;
142 for (var i = dollar + 1; i < text.length; i++) {
143 if (text[i] == open) depth++;
144 if (text[i] == close && --depth == 0) return i;
145 }
146 fail('unterminated variable reference');
147 }
148 String expand(String text) {
149 final result = StringBuffer();
150 for (var i = 0; i < text.length;) {
151 if (text[i] != r'$') { result.write(text[i++]); continue; }
152 if (i + 1 == text.length) break;
153 final next = text[i + 1];
154 if (next == r'$') { result.write(r'$'); i += 2; }
155 else if (next == '(' || next == '{') {
156 final end = closing(text, i);
157 result.write(reference(text.substring(i + 2, end), next));
158 i = end + 1;
159 } else { result.write(value(next)); i += 2; }
160 }
161 return result.toString();
162 }
163 String reference(String content, String open) {
164 final space = content.indexOf(' ');
165 final name = space < 0 ? '' : content.substring(0, space);
166 if (space >= 0 && limits.containsKey(name)) return callFunction(name, leftTrim(content.substring(space + 1)), open);
167 final expanded = expand(content);
168 final colon = expanded.indexOf(':');
169 final equal = expanded.indexOf('=', colon + 1);
170 if (colon >= 0 && equal >= 0) {
171 var pattern = expanded.substring(colon + 1, equal), replacement = expanded.substring(equal + 1);
172 if (!pattern.contains('%')) { pattern = '%$pattern'; replacement = '%$replacement'; }
173 return words(value(expanded.substring(0, colon))).map((word) => replaceWord(word, pattern, replacement)).join(' ');
174 }
175 return value(expanded);
176 }
177 String value(String name) {
178 for (final scope in scopes.reversed) {
179 if (scope is ForeachScope && scope.name == name) return scope.word;
180 if (scope is CallScope && numeric(name)) {
181 final i = int.parse(name);
182 return i < scope.values.length ? scope.values[i] : '';
183 }
184 }
185 if (automatic != null && automatic!.containsKey(name)) return automatic![name]!;
186 final variable = variables[name];
187 if (variable == null) return '';
188 if (variable.kind == ':=') return variable.value;
189 if (expanding.contains(name)) fail("Recursive variable '$name' references itself (eventually)");
190 expanding.add(name);
191 try { return expand(variable.value); } finally { expanding.remove(name); }
192 }
193 String callFunction(String name, String text, String open) {
194 final (minimum, maximum) = limits[name]!;
195 final args = <String>[];
196 final close = open == '(' ? ')' : '}';
197 var depth = 0, start = 0;
198 for (var i = 0; i < text.length; i++) {
199 if (text[i] == open) { depth++; }
200 else if (text[i] == close) { depth--; }
201 else if (text[i] == ',' && depth == 0 && (maximum == null || args.length < maximum - 1)) {
202 args.add(text.substring(start, i));
203 start = i + 1;
204 }
205 }
206 args.add(text.substring(start));
207 if (args.length < minimum) fail("insufficient number of arguments (${args.length}) to function '$name'");
208 if (name == 'foreach') {
209 final variable = trim(expand(args[0])), list = words(expand(args[1])), result = <String>[];
210 for (final word in list) {
211 scopes.add(ForeachScope(variable, word));
212 try { result.add(expand(args[2])); } finally { scopes.removeLast(); }
213 }
214 return result.join(' ');
215 }
216 if (name == 'if') {
217 final index = trim(expand(args[0])).isNotEmpty ? 1 : 2;
218 return expand(index < args.length ? args[index] : '');
219 }
220 final expanded = args.map(expand).toList();
221 if (name == 'subst') return expanded[0].isEmpty ? expanded[2] : expanded[2].replaceAll(expanded[0], expanded[1]);
222 if (name == 'call') {
223 final variableName = trim(expanded[0]), variable = variables[trim(expanded[0])];
224 if (variable == null) return '';
225 if (variable.kind == ':=') return variable.value;
226 scopes.add(CallScope([variableName, ...expanded.skip(1)]));
227 try { return expand(variable.value); } finally { scopes.removeLast(); }
228 }
229 final values = expanded.map(trim).toList(), a = trim(expanded[0]);
230 final b = values.length > 1 ? values[1] : '', c = values.length > 2 ? values[2] : '';
231 switch (name) {
232 case 'patsubst': return words(c).map((word) => replaceWord(word, a, b)).join(' ');
233 case 'strip': return words(a).join(' ');
234 case 'findstring': return b.contains(a) ? a : '';
235 case 'filter':
236 case 'filter-out': return words(b).where((word) => words(a).any((pattern) => patternStem(pattern, word) != null) == (name == 'filter')).join(' ');
237 case 'sort': return (unique(words(a))..sort()).join(' ');
238 case 'word':
239 if (!numeric(a)) fail("non-numeric first argument to 'word' function: '$a'");
240 final index = int.parse(a), list = words(b);
241 if (index == 0) fail("first argument to 'word' function must be greater than 0");
242 return index <= list.length ? list[index - 1] : '';
243 case 'wordlist':
244 if (!numeric(a)) fail("non-numeric first argument to 'wordlist' function: '$a'");
245 if (!numeric(b)) fail("non-numeric second argument to 'wordlist' function: '$b'");
246 if (int.parse(a) == 0) fail("invalid first argument to 'wordlist' function: '$a'");
247 final start = int.parse(a) - 1, end = int.parse(b), list = words(c);
248 return start >= list.length || end <= start ? '' : list.sublist(start, math.min(end, list.length)).join(' ');
249 case 'words': return '${words(a).length}';
250 case 'firstword': return words(a).isEmpty ? '' : words(a).first;
251 case 'lastword': return words(a).isEmpty ? '' : words(a).last;
252 case 'dir': return words(a).map((word) => word.contains('/') ? word.substring(0, word.lastIndexOf('/') + 1) : './').join(' ');
253 case 'notdir': return words(a).map((word) => word.substring(word.lastIndexOf('/') + 1)).join(' ');
254 case 'suffix': return words(a).where((word) => word.lastIndexOf('.') > word.lastIndexOf('/')).map((word) => word.substring(word.lastIndexOf('.'))).join(' ');
255 case 'basename': return words(a).map((word) => word.lastIndexOf('.') > word.lastIndexOf('/') ? word.substring(0, word.lastIndexOf('.')) : word).join(' ');
256 case 'addsuffix': return words(b).map((word) => word + a).join(' ');
257 case 'addprefix': return words(b).map((word) => a + word).join(' ');
258 case 'wildcard': return words(a).expand(wildcard).join(' ');
259 case 'value': return variables[a]?.value ?? '';
260 case 'info': output.add(a); return '';
261 case 'warning': output.add('$location: $a'); return '';
262 case 'error': fail(a);
263 }
264 return '';
265 }
266 Map<String, int> punctuation(String text) {
267 final result = <String, int>{};
268 for (var i = 0; i < text.length; i++) {
269 if (text[i] == r'$') {
270 if (i + 1 < text.length && (text[i + 1] == '(' || text[i + 1] == '{')) { i = closing(text, i); }
271 else if (i + 1 < text.length) { i++; }
272 } else if ('=:;'.contains(text[i])) { result.putIfAbsent(text[i], () => i); }
273 }
274 return result;
275 }
276 Rule register(String name, List<String> prerequisites) {
277 final rule = explicit.putIfAbsent(name, () => Rule(name, []));
278 rule.prerequisites = unique([...rule.prerequisites, ...prerequisites]);
279 if (goal == null && !name.startsWith('.')) goal = name;
280 return rule;
281 }
282 void addRecipe(CurrentRule current, int line, String text) {
283 if (current.entries.isEmpty) return;
284 if (current.recipe == null) {
285 current.recipe = [];
286 for (final entry in current.entries) {
287 if (current.explicit && entry.recipe != null) {
288 output.add("Makefile:$line: warning: overriding recipe for target '${entry.name}'");
289 output.add("Makefile:${entry.recipe!.first.line}: warning: ignoring old recipe for target '${entry.name}'");
290 }
291 entry.recipe = current.recipe;
292 }
293 }
294 current.recipe!.add(RecipeLine(line, text));
295 }
296 void load(List<String> body) {
297 resetMakefile();
298 final conditions = <Condition>[];
299 CurrentRule? current;
300 bool active() => conditions.isEmpty || conditions.last.outer && conditions.last.value;
301 try {
302 for (var i = 0; i < body.length; i++) {
303 final line = i + 1;
304 location = 'Makefile:$line';
305 if (body[i].startsWith('>')) {
306 if (!active()) continue;
307 if (current == null) fail('recipe commences before first target');
308 addRecipe(current, line, leftTrim(body[i].substring(1)));
309 continue;
310 }
311 var logical = body[i];
312 while (logical.endsWith(r'\') && i + 1 < body.length) {
313 logical = '${rightTrim(logical.substring(0, logical.length - 1))} ${leftTrim(body[++i])}';
314 }
315 final text = trim(logical.split('#').first), space = trim(logical.split('#').first).indexOf(' ');
316 final first = space < 0 ? text : text.substring(0, space), rest = space < 0 ? '' : trim(text.substring(space + 1));
317 if (['ifeq', 'ifneq', 'ifdef', 'ifndef'].contains(first)) {
318 final outer = active();
319 var condition = false;
320 if (outer) {
321 if (first == 'ifdef' || first == 'ifndef') {
322 final variable = variables[trim(expand(rest))];
323 condition = variable != null && variable.value.isNotEmpty;
324 if (first == 'ifndef') condition = !condition;
325 } else {
326 if (rest.length < 2 || !rest.startsWith('(') || !rest.endsWith(')')) fail('invalid syntax in conditional');
327 final inside = rest.substring(1, rest.length - 1);
328 var depth = 0, comma = -1;
329 for (var j = 0; j < inside.length; j++) {
330 if (inside[j] == '(') { depth++; }
331 else if (inside[j] == ')') { depth--; }
332 else if (inside[j] == ',' && depth == 0) { comma = j; break; }
333 }
334 if (comma < 0) fail('invalid syntax in conditional');
335 final left = trim(expand(inside.substring(0, comma))), right = trim(expand(inside.substring(comma + 1)));
336 condition = left == right;
337 if (first == 'ifneq') condition = !condition;
338 }
339 }
340 conditions.add(Condition(outer, condition, line));
341 continue;
342 }
343 if (first == 'else') {
344 if (conditions.isEmpty) fail("extraneous 'else'");
345 final condition = conditions.last;
346 if (condition.alternate) fail("only one 'else' per conditional");
347 condition.value = !condition.value;
348 condition.alternate = true;
349 continue;
350 }
351 if (first == 'endif') {
352 if (conditions.isEmpty) fail("extraneous 'endif'");
353 conditions.removeLast();
354 continue;
355 }
356 if (text.isEmpty || !active()) continue;
357 final positions = punctuation(text);
358 final equal = positions['='], colon = positions[':'];
359 if (equal != null && (colon == null || colon >= equal - 1)) {
360 final special = equal > 0 && ':+?'.contains(text[equal - 1]);
361 final op = special ? '${text[equal - 1]}=' : '=';
362 final name = trim(text.substring(0, equal - (special ? 1 : 0))), rhs = trim(text.substring(equal + 1));
363 if (name.isEmpty) fail('empty variable name');
364 final previous = variables[name];
365 if (op == '=' || op == '?=' && previous == null || op == '+=' && previous == null) { variables[name] = Variable('=', rhs); }
366 else if (op == ':=') { variables[name] = Variable(':=', expand(rhs)); }
367 else if (op == '+=') {
368 final addition = previous!.kind == ':=' ? expand(rhs) : rhs;
369 previous.value = previous.value.isEmpty ? addition : '${previous.value} $addition';
370 }
371 current = null;
372 } else if (colon != null) {
373 final targets = unique(words(expand(text.substring(0, colon))));
374 var rhs = text.substring(colon + 1);
375 String? inline, targetPattern;
376 final semicolon = punctuation(rhs)[';'];
377 if (semicolon != null) { inline = leftTrim(rhs.substring(semicolon + 1)); rhs = rhs.substring(0, semicolon); }
378 final secondColon = punctuation(rhs)[':'];
379 if (secondColon != null) { targetPattern = trim(expand(rhs.substring(0, secondColon))); rhs = rhs.substring(secondColon + 1); }
380 final prerequisites = words(expand(rhs));
381 current = CurrentRule();
382 if (targets.length == 1 && targets.first == '.PHONY') { phony.addAll(prerequisites); }
383 else if (targetPattern != null) {
384 if (!targetPattern.contains('%')) fail("target pattern contains no '%'");
385 for (final name in targets) {
386 final stem = patternStem(targetPattern, name);
387 if (stem == null || stem.isEmpty) { output.add("Makefile:$line: target '$name' doesn't match the target pattern"); }
388 else {
389 final entry = register(name, prerequisites.map((value) => substitute(value, stem)).toList());
390 entry.stem = stem;
391 current.entries.add(entry);
392 }
393 }
394 } else if (targets.any((name) => name.contains('%'))) {
395 if (targets.any((name) => !name.contains('%'))) fail('mixed implicit and normal rules');
396 current.explicit = false;
397 for (final name in targets) {
398 final entry = Rule(name, List.of(prerequisites));
399 patterns.add(entry);
400 current.entries.add(entry);
401 }
402 } else {
403 for (final name in targets) { current.entries.add(register(name, prerequisites)); }
404 }
405 if (inline != null) addRecipe(current, line, inline);
406 } else if (trim(expand(text)).isNotEmpty) { fail('missing separator'); }
407 }
408 if (conditions.isNotEmpty) { location = 'Makefile:${conditions.last.line}'; fail("missing 'endif'"); }
409 loaded = true;
410 output.add('Makefile loaded: lines=${body.length} variables=${variables.length} targets=${explicit.length} patterns=${patterns.where((rule) => rule.recipe != null).length} goal=${goal ?? '(none)'}');
411 } on MakeError catch (error) {
412 output.add(error.message);
413 resetMakefile();
414 } finally { location = 'make'; }
415 }
416 Determined determine(String name) {
417 final entry = explicit[name];
418 if (entry != null && entry.recipe != null) return Determined(name, List.of(entry.prerequisites), entry.recipe, entry.stem, 'rule');
419 if (!phony.contains(name)) {
420 final candidates = <PatternCandidate>[];
421 for (var i = 0; i < patterns.length; i++) {
422 final rule = patterns[i], stem = patternStem(patterns[i].name, name);
423 if (rule.recipe != null && stem != null && stem.isNotEmpty) candidates.add(PatternCandidate(rule, i, stem));
424 }
425 candidates.sort((a, b) {
426 final order = a.stem.length.compareTo(b.stem.length);
427 return order == 0 ? a.index.compareTo(b.index) : order;
428 });
429 for (final candidate in candidates) {
430 final prerequisites = candidate.rule.prerequisites.map((value) => substitute(value, candidate.stem)).toList();
431 if (prerequisites.every((value) => files.containsKey(value) || explicit.containsKey(value))) {
432 return Determined(name, unique([...prerequisites, ...?entry?.prerequisites]), candidate.rule.recipe, candidate.stem, 'rule');
433 }
434 }
435 }
436 if (entry != null || phony.contains(name)) return Determined(name, List.of(entry?.prerequisites ?? []), null, entry?.stem ?? '', 'rule');
437 return Determined(name, [], null, '', files.containsKey(name) ? 'source' : 'missing');
438 }
439 int executeRecipeCommand(String command) {
440 final list = words(command), args = words(command).skip(1).toList();
441 if (list.first == 'echo') { output.add(args.join(' ')); }
442 else if (list.first == 'touch') { for (final name in args) { touch(name); } }
443 else if (list.first == 'rm') {
444 final forced = args.isNotEmpty && args.first == '-f';
445 var status = 0;
446 for (final pattern in forced ? args.skip(1) : args) {
447 final matches = wildcard(pattern);
448 for (final name in matches.isEmpty ? [pattern] : matches) {
449 if (files.remove(name) == null && !forced) { output.add("rm: cannot remove '$name': No such file or directory"); status = 1; }
450 }
451 }
452 return status;
453 } else if (list.first == 'false') { return 1; }
454 else if (list.first == 'exit') { return args.isEmpty ? 0 : int.parse(args.first); }
455 return 0;
456 }
457 void make(List<String> args) {
458 final flags = <String>{}, goals = <String>[];
459 for (final argument in args) {
460 if (argument.startsWith('-')) {
461 for (final c in argument.substring(1).split('')) {
462 if (!'nBkis'.contains(c)) { output.add("make: invalid option -- '$c'"); return; }
463 flags.add(c);
464 }
465 } else { goals.add(argument); }
466 }
467 if (goals.isEmpty) {
468 if (!loaded) { output.add('make: *** No targets specified and no makefile found. Stop.'); return; }
469 if (goal == null) { output.add('make: *** No targets. Stop.'); return; }
470 goals.add(goal!);
471 }
472 final savedFiles = Map<String, int>.of(files), savedClock = clock;
473 final results = <String, BuildResult>{}, beingBuilt = <String>{};
474 var runs = 0;
475 int? targetTime(String name) => phony.contains(name) ? null : files[name];
476 BuildResult build(String name, String? parent) {
477 final cached = results[name];
478 if (cached != null) return cached;
479 final rule = determine(name);
480 if (rule.kind == 'missing') {
481 output.add("make: *** No rule to make target '$name'${parent == null ? '' : ", needed by '$parent'"}.${flags.contains('k') ? '' : ' Stop.'}");
482 if (!flags.contains('k')) throw BuildAbort();
483 return results[name] = BuildResult('failure', rule);
484 }
485 if (rule.kind == 'source') return results[name] = BuildResult('success', rule);
486 beingBuilt.add(name);
487 var failed = false;
488 final kept = <String>[];
489 for (final prerequisite in rule.prerequisites) {
490 if (beingBuilt.contains(prerequisite)) { output.add('make: Circular $name <- $prerequisite dependency dropped.'); continue; }
491 kept.add(prerequisite);
492 if (build(prerequisite, name).status != 'success') failed = true;
493 }
494 rule.prerequisites = kept;
495 beingBuilt.remove(name);
496 var status = 'dependency failure';
497 if (!failed) {
498 status = 'success';
499 final time = targetTime(name);
500 final newer = kept.where((value) => flags.contains('B') || time == null || targetTime(value) == null || targetTime(value)! > time).toList();
501 if (rule.recipe != null && (flags.contains('B') || time == null || newer.isNotEmpty)) {
502 runs++;
503 automatic = {'@': name, '<': kept.isEmpty ? '' : kept.first, '^': kept.join(' '), '?': newer.join(' '), '*': rule.stem};
504 final commands = <RecipeLine>[];
505 try {
506 for (final line in rule.recipe!) { location = 'Makefile:${line.line}'; commands.add(RecipeLine(line.line, expand(line.text))); }
507 } finally { automatic = null; location = 'make'; }
508 for (final line in commands) {
509 var i = 0;
510 while (i < line.text.length && ' @-'.contains(line.text[i])) { i++; }
511 final prefix = line.text.substring(0, i), command = rightTrim(line.text.substring(i));
512 if (command.isEmpty) continue;
513 if (flags.contains('n') || !prefix.contains('@') && !flags.contains('s')) output.add(command);
514 if (flags.contains('n')) continue;
515 final code = executeRecipeCommand(command);
516 if (code != 0) {
517 if (prefix.contains('-') || flags.contains('i')) { output.add('make: [Makefile:${line.line}: $name] Error $code (ignored)'); }
518 else {
519 output.add('make: *** [Makefile:${line.line}: $name] Error $code');
520 if (!flags.contains('k')) throw BuildAbort();
521 status = 'failure';
522 break;
523 }
524 }
525 }
526 if (status == 'success' && !phony.contains(name)) touch(name);
527 }
528 }
529 return results[name] = BuildResult(status, rule);
530 }
531 try {
532 for (final goal in goals) {
533 final previous = runs, result = build(goal, null);
534 if (result.status == 'dependency failure') { output.add("make: Target '$goal' not remade because of errors."); }
535 else if (result.status == 'success' && previous == runs) {
536 output.add(phony.contains(goal) || result.rule.recipe == null ? "make: Nothing to be done for '$goal'." : "make: '$goal' is up to date.");
537 }
538 }
539 } on MakeError catch (error) { output.add(error.message); }
540 on BuildAbort {}
541 finally { if (flags.contains('n')) { files = savedFiles; clock = savedClock; } }
542 }
543 void other(String commandLine) {
544 location = 'make';
545 final list = commandLine.split(' '), command = commandLine.split(' ').first;
546 final args = list.skip(1).toList();
547 if (command == 'MAKE') { make(args); }
548 else if (command == 'TOUCH') { for (final name in args) { touch(name); } }
549 else if (command == 'RM') { for (final name in args) { if (files.remove(name) == null) output.add("rm: cannot remove '$name': No such file or directory"); } }
550 else if (command == 'FILES') {
551 output.add('clock $clock');
552 for (final name in files.keys.toList()..sort()) { output.add('$name ${files[name]}'); }
553 } else if (command == 'EXPAND') { output.add('[${expand(commandLine.substring(7))}]'); }
554 else if (command == 'VARS' && loaded) {
555 for (final name in variables.keys.toList()..sort()) {
556 final expanded = trim(value(name));
557 output.add('$name ${variables[name]!.kind}${expanded.isEmpty ? '' : ' $expanded'}');
558 }
559 } else if (command == 'RULES' && loaded) {
560 final rules = [...(explicit.keys.toList()..sort()).map((name) => explicit[name]!), ...patterns.where((rule) => rule.recipe != null)];
561 for (final rule in rules) {
562 output.add('${rule.name}:${rule.prerequisites.map((value) => ' $value').join()}');
563 for (final line in rule.recipe ?? <RecipeLine>[]) { output.add('>${line.text}'); }
564 }
565 if (phony.isNotEmpty) output.add('.PHONY: ${(phony.toList()..sort()).join(' ')}');
566 }
567 }
568}
569void main() {
570 final simulator = Simulator();
571 stdin.readLineSync();
572 String line;
573 while ((line = stdin.readLineSync()!) != 'END FILES') {
574 final parts = line.split(' '), time = int.parse(line.split(' ')[1]);
575 simulator.files[parts[0]] = time;
576 simulator.clock = math.max(simulator.clock, time);
577 }
578 while ((line = stdin.readLineSync()!) != 'QUIT') {
579 output.add('\$ $line');
580 if (line == 'MAKEFILE') {
581 final body = <String>[];
582 while ((line = stdin.readLineSync()!) != 'END MAKEFILE') { body.add(line); }
583 simulator.load(body);
584 } else {
585 try { simulator.other(line); }
586 on MakeError catch (error) { output.add(error.message); }
587 }
588 }
589 if (output.isNotEmpty) stdout.write('${output.join('\n')}\n');
590}
