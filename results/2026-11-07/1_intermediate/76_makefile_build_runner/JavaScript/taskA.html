38function leftTrim(text) {
39 let i = 0;
40 while (text[i] === " ") i++;
41 return text.slice(i);
42}
43function rightTrim(text) {
44 let i = text.length;
45 while (i > 0 && text[i - 1] === " ") i--;
46 return text.slice(0, i);
47}
48function trim(text) {
49 return rightTrim(leftTrim(text));
50}
51function words(text) {
52 return text.split(" ").filter((word) => word.length > 0);
53}
54function unique(items) {
55 return [...new Set(items)];
56}
57function numeric(text) {
58 return text.length > 0 && !/[^0-9]/.test(text);

60function substitute(text, stem) {
61 const i = text.indexOf("%");
62 return i < 0 ? text : text.slice(0, i) + stem + text.slice(i + 1);
63}
64function patternStem(pattern, word) {
65 const i = pattern.indexOf("%");
66 if (i < 0) return pattern === word ? "" : null;
67 const before = pattern.slice(0, i),
68 after = pattern.slice(i + 1);
69 return word.length >= before.length + after.length &&
70 word.startsWith(before) &&
71 word.endsWith(after)
72 ? word.slice(before.length, word.length - after.length)
73 : null;
74}
75function replaceWord(word, pattern, replacement) {
76 const stem = patternStem(pattern, word);
77 return stem === null
78 ? word
79 : pattern.includes("%")
80 ? substitute(replacement, stem)
81 : replacement;
82}
83function wildcardMatch(pattern, name) {
84 let previous = Array(name.length + 1).fill(false);
85 previous[0] = true;
86 for (const c of pattern) {
87 const next = Array(name.length + 1).fill(false);
88 for (let j = 0; j <= name.length; j++)
89 next[j] =
90 c === "*"
91 ? previous[j] || (j > 0 && name[j - 1] !== "/" && next[j - 1])
92 : j > 0 &&
93 previous[j - 1] &&
94 (c === "?" ? name[j - 1] !== "/" : c === name[j - 1]);
95 previous = next;
96 }
97 return previous[name.length];
98}

139 expand(text) {
140 let result = "";

142 if (text[i] !== "$") {
143 result += text[i++];

145 }

309 .map((word) => word.slice(word.lastIndexOf(".")))

311 case "basename":
312 return words(a)
313 .map((word) =>
314 word.lastIndexOf(".") > word.lastIndexOf("/")

323 case "addprefix":
324 return words(b)
325 .map((word) => a + word)

327 case "wildcard":
328 return words(a)
329 .flatMap((pattern) => this.wildcard(pattern))

432 }

479 rhs = rhs.slice(0, semicolon);

484 rhs = rhs.slice(secondColon + 1);
485 }

487 current = { entries: [], explicit: true, recipe: null };

506 }
507 }
508 } else if (targets.some((name) => name.includes("%"))) {

514 name,
515 prerequisites: [...prerequisites],
516 recipe: null,

521 }

616 goals = [];
