33func stem(_ pattern: String,_ word: String) -> String? {
34 guard let i = position(pattern,37) else {
35 return pattern == word ? "" : nil
36 }
37 let a = part(pattern,0,i),b = part(pattern,i+1)
38 return word.count >= a.count+b.count && word.hasPrefix(a) && word.hasSuffix(b) ? part(word,a.count,word.count-b.count) : nil
39}

41func substitute(_ s: String,_ m: String) -> String {
42 guard let i = position(s,37) else {
43 return s
44 }
45 return part(s,0,i)+m+part(s,i+1)
46}

48func replaceWord(_ w: String,_ p: String,_ r: String) -> String {
49 guard let m = stem(p,w) else {
50 return w
51 }
52 return p.contains("%") ? substitute(r,m) : r
53}

55func wildcard(_ pattern: String,_ word: String) -> Bool {
56 let bytes = Array(word.utf8)
57 var previous = Array(repeating: false,count: bytes.count+1)
58 previous[0] = true
59 for c in pattern.utf8 {
60 var next = Array(repeating: false,count: bytes.count+1)
61 if c == 42 {
62 next[0] = previous[0]
63 }
64 for (i,b) in bytes.enumerated() {
65 let j = i+1
66 next[j] = c == 42 ? previous[j] || (b != 47 && next[j-1]) : previous[j-1] && (c == 63 ? b != 47 : c == b)
67 }
68 previous = next
69 }
70 return previous.last!
71}

162 func closing(_ s: String,_ start: Int) throws -> Int {
163 let b = Array(s.utf8)
164 let opening = b[start]
165 let close: UInt8 = opening == 40 ? 41 : 125
166 var depth = 0
167 for i in start..<b.count {
168 if b[i] == opening {
169 depth += 1
170 }
171 if b[i] == close {
172 depth -= 1
173 if depth == 0 {
174 return i
175 }
176 }
177 }
178 try error("unterminated variable reference")
179 }
180 func outside(_ s: String) throws -> [UInt8:Int] {
181 let b = Array(s.utf8)
182 var result = [UInt8:Int]()
183 var i = 0
184 while i < b.count {
185 if b[i] == 36 {
186 if i+1 < b.count && (b[i+1] == 40 || b[i+1] == 123) {
187 i = try closing(s,i+1)+1
188 } else {
189 i = min(b.count,i+2)
190 }
191 } else {
192 if [UInt8(61),58,59].contains(b[i]) && result[b[i]] == nil {
193 result[b[i]] = i
194 }
195 i += 1
196 }
197 }
198 return result
199 }
200 func value(_ name: String) throws -> String {
201 for scope in scopes.reversed() {
202 switch scope {
203 case let .foreach(n,v):
204 if n == name {
205 return v
206 }
207 case let .call(values):
208 if !name.isEmpty && name.utf8.allSatisfy({
209 (48...57).contains($0)
210 }) {
211 let i = Int(name)!
212 return i < values.count ? values[i] : ""
213 }
214 }
215 }
216 if let v = automatic[name] {
217 return v
218 }
219 guard let v = variables[name] else {
220 return ""
221 }
222 if v.simple {
223 return v.value
224 }
225 if !expanding.insert(name).inserted {
226 try error("Recursive variable '\(name)' references itself (eventually)")
227 }
228 defer {
229 expanding.remove(name)
230 }
231 return try expand(v.value)
232 }
233 func expand(_ s: String) throws -> String {
234 let b = Array(s.utf8)
235 var out = ""
236 var i = 0
237 while i < b.count {
238 if b[i] != 36 {
239 out.append(Character(UnicodeScalar(b[i])))
240 i += 1
241 continue
242 }
243 if i+1 == b.count {
244 break
245 }
246 let c = b[i+1]
247 if c == 36 {
248 out += "$"
249 i += 2
250 } else if c == 40 || c == 123 {
251 let end = try closing(s,i+1)
252 out += try reference(part(s,i+2,end),c)
253 i = end+1
254 } else {
255 out += try value(part(s,i+1,i+2))
256 i += 2
257 }
258 }
259 return out
260 }
261 func limits(_ n: String) -> (Int,Int)? {
262 switch n {
263 case "subst","patsubst","wordlist","foreach":
264 return (3,3)
265 case "findstring","filter","filter-out","word","addsuffix","addprefix":
266 return (2,2)
267 case "if":
268 return (2,3)
269 case "call":
270 return (1,Int.max)
271 case "strip","sort","words","firstword","lastword","dir","notdir","suffix","basename","wildcard","value","info","warning","error":
272 return (1,1)
273 default:
274 return nil
275 }
276 }
277 func reference(_ content: String,_ opening: UInt8) throws -> String {
278 if let space = position(content,32),limits(part(content,0,space)) != nil {
279 let raw = part(content,space+1).drop(while: {
280 $0 == " "
281 })
282 return try function(part(content,0,space),String(raw),opening)
283 }
284 let n = try expand(content)
285 if let colon = position(n,58),let after = position(part(n,colon+1),61) {
286 let equal = colon+1+after
287 var p = part(n,colon+1,equal),r = part(n,equal+1)
288 if !p.contains("%") {
289 p = "%"+p
290 r = "%"+r
291 }
292 return try words(value(part(n,0,colon))).map {
293 replaceWord($0,p,r)
294 }.joined(separator: " ")
295 }
296 return try value(n)
297 }
298 func arguments(_ text: String,_ opening: UInt8,_ maximum: Int) -> [String] {
299 let close: UInt8 = opening == 40 ? 41 : 125
300 var depth = 0
301 var start = 0
302 var result = [String]()
303 for (i,c) in text.utf8.enumerated() {
304 if c == opening {
305 depth += 1
306 }
307 if c == close {
308 depth -= 1
309 }
310 if c == 44 && depth == 0 && result.count < maximum-1 {
311 result.append(part(text,start,i))
312 start = i+1
313 }
314 }
315 result.append(part(text,start))
316 return result
317 }
318 func function(_ name: String,_ text: String,_ opening: UInt8) throws -> String {
319 let (minimum,maximum) = limits(name)!
320 let raw = arguments(text,opening,maximum)
321 if raw.count < minimum {
322 try error("insufficient number of arguments (\(raw.count)) to function '\(name)'")
323 }
324 if name == "foreach" {
325 let n = try trim(expand(raw[0]))
326 let list = try words(expand(raw[1]))
327 var result = [String]()
328 for word in list {
329 scopes.append(.foreach(n,word))
330 do {
331 let s = try expand(raw[2])
332 scopes.removeLast()
333 result.append(s)
334 } catch {
335 scopes.removeLast()
336 throw error
337 }
338 }
339 return result.joined(separator: " ")
340 }
341 if name == "if" {
342 let condition = try trim(expand(raw[0]))
343 if !condition.isEmpty {
344 return try expand(raw[1])
345 }
346 if raw.count == 3 {
347 return try expand(raw[2])
348 }
349 return ""
350 }
351 var args = [String]()
352 for r in raw {
353 let s = try expand(r)
354 args.append(name == "subst" || name == "call" ? s : trim(s))
355 }
356 func list(_ i: Int) -> [String] {
357 words(args[i])
358 }
359 func joined(_ xs: [String]) -> String {
360 xs.joined(separator: " ")
361 }
362 switch name {
363 case "call":
364 let n = trim(args[0])
365 guard let v = variables[n] else {
366 return ""
367 }
368 if v.simple {
369 return v.value
370 }
371 scopes.append(.call([n]+Array(args.dropFirst())))
372 defer {
373 scopes.removeLast()
374 }
375 return try expand(v.value)
376 case "value":
377 return variables[args[0]]?.value ?? ""
378 case "info":
379 output.append(args[0])
380 return ""
381 case "warning":
382 output.append("\(location): \(args[0])")
383 return ""
384 case "error":
385 try error(args[0])
386 case "subst":
387 return args[0].isEmpty ? args[2] : args[2].replacingOccurrences(of: args[0],with: args[1])
388 case "patsubst":
389 return joined(list(2).map {
390 replaceWord($0,args[0],args[1])
391 })
392 case "strip":
393 return joined(list(0))
394 case "findstring":
395 return args[1].contains(args[0]) ? args[0] : ""
396 case "filter","filter-out":
397 return joined(list(1).filter {
398 w in list(0).contains {
399 stem($0,w) != nil
400 } == (name == "filter")
401 })
402 case "sort":
403 return joined(Set(list(0)).sorted())
404 case "word":
405 let n = args[0]
406 if n.isEmpty || !n.utf8.allSatisfy({
407 (48...57).contains($0)
408 }) {
409 try error("non-numeric first argument to 'word' function: '\(n)'")
410 }
411 let i = Int(n)!
412 if i == 0 {
413 try error("first argument to 'word' function must be greater than 0")
414 }
415 let values = list(1)
416 return i <= values.count ? values[i-1] : ""
417 case "wordlist":
418 let a = args[0],b = args[1]
419 if a.isEmpty || !a.utf8.allSatisfy({
420 (48...57).contains($0)
421 }) {
422 try error("non-numeric first argument to 'wordlist' function: '\(a)'")
423 }
424 if b.isEmpty || !b.utf8.allSatisfy({
425 (48...57).contains($0)
426 }) {
427 try error("non-numeric second argument to 'wordlist' function: '\(b)'")
428 }
429 let start = Int(a)!,end = Int(b)!
430 if start == 0 {
431 try error("invalid first argument to 'wordlist' function: '\(a)'")
432 }
433 return joined(list(2).enumerated().filter {
434 $0.offset+1 >= start && $0.offset+1 <= end
435 }.map(\.element))
436 case "words":
437 return String(list(0).count)
438 case "firstword":
439 return list(0).first ?? ""
440 case "lastword":
441 return list(0).last ?? ""
442 case "dir":
443 return joined(list(0).map {
444 w in if let i = Array(w.utf8).lastIndex(of: 47) {
445 return part(w,0,i+1)
446 }
447 return "./"
448 })
449 case "notdir":
450 return joined(list(0).map {
451 w in part(w,(Array(w.utf8).lastIndex(of: 47) ?? -1)+1)
452 })
453 case "suffix":
454 return joined(list(0).compactMap {
455 w in let b = Array(w.utf8)
456 if let i = b.lastIndex(of: 46),i > (b.lastIndex(of: 47) ?? -1) {
457 return part(w,i)
458 }
459 return nil
460 })
461 case "basename":
462 return joined(list(0).map {
463 w in let b = Array(w.utf8)
464 if let i = b.lastIndex(of: 46),i > (b.lastIndex(of: 47) ?? -1) {
465 return part(w,0,i)
466 }
467 return w
468 })
469 case "addsuffix":
470 return joined(list(1).map {
471 $0+args[0]
472 })
473 case "addprefix":
474 return joined(list(1).map {
475 args[0]+$0
476 })
477 case "wildcard":
478 return joined(list(0).flatMap {
479 glob($0)
480 })
481 default:
482 return ""
483 }
484 }

526 do {
527 try loadBody(body)

538 func loadBody(_ body: [String]) throws {
539 var current: Current?
540 var conditions = [Condition]()
541 func active() -> Bool {
542 conditions.last.map {
543 $0.outer && $0.value
544 } ?? true
545 }
546 var i = 0
547 while i < body.count {
548 let lineNumber = i+1
549 location = "Makefile:\(lineNumber)"
550 var text = body[i]
551 i += 1
552 if text.hasPrefix(">") {
553 if !active() {
554 continue
555 }
556 guard let current = current else {
557 try error("recipe commences before first target")
558 }
559 addRecipe(current,RecipeLine(number: lineNumber,text: String(text.dropFirst()).drop(while: {
560 $0 == " "
561 }).description))
562 continue
563 }
564 while text.hasSuffix("\\") && i < body.count {
565 text.removeLast()
566 while text.hasSuffix(" ") {
567 text.removeLast()
568 }
569 text += " "+String(body[i].drop(while: {
570 $0 == " "
571 }))
572 i += 1
573 }
574 text = trim(text.components(separatedBy: "#")[0])
575 let space = position(text,32)
576 let first = space.map {
577 part(text,0,$0)
578 } ?? text
579 let rest = space.map {
580 trim(part(text,$0+1))
581 } ?? ""
582 if ["ifdef","ifndef","ifeq","ifneq"].contains(first) {
583 let outer = active()
584 var truth = false
585 if outer {
586 if first == "ifdef" || first == "ifndef" {
587 let n = try trim(expand(rest))
588 truth = variables[n]?.value.isEmpty == false
589 if first == "ifndef" {
590 truth = !truth
591 }
592 } else {
593 if rest.count < 2 || !rest.hasPrefix("(") || !rest.hasSuffix(")") {
594 try error("invalid syntax in conditional")
595 }
596 let inner = part(rest,1,rest.count-1)
597 var depth = 0
598 var comma: Int?
599 for (j,c) in inner.utf8.enumerated() {
600 if c == 40 {
601 depth += 1
602 }
603 if c == 41 {
604 depth -= 1
605 }
606 if c == 44 && depth == 0 {
607 comma = j
608 break
609 }
610 }
611 guard let comma = comma else {
612 try error("invalid syntax in conditional")
613 }
614 let left = try trim(expand(part(inner,0,comma)))
615 let right = try trim(expand(part(inner,comma+1)))
616 truth = (left == right) == (first == "ifeq")
617 }
618 }
619 conditions.append(Condition(outer: outer,value: truth,hadElse: false,line: lineNumber))
620 continue
621 }
622 if first == "else" {
623 if conditions.isEmpty {
624 try error("extraneous 'else'")
625 }
626 let j = conditions.count-1
627 if conditions[j].hadElse {
628 try error("only one 'else' per conditional")
629 }
630 conditions[j].value.toggle()
631 conditions[j].hadElse = true
632 continue
633 }
634 if first == "endif" {
635 if conditions.isEmpty {
636 try error("extraneous 'endif'")
637 }
638 conditions.removeLast()
639 continue
640 }
641 if text.isEmpty || !active() {
642 continue
643 }
644 let positions = try outside(text)
645 let equal = positions[61],colon = positions[58]
646 if let equal = equal,colon == nil || colon! >= equal-1 {
647 let bytes = Array(text.utf8)
648 let special = equal > 0 && [UInt8(58),43,63].contains(bytes[equal-1])
649 let op = special ? part(text,equal-1,equal+1) : "="
650 let n = trim(part(text,0,equal-(special ? 1 : 0)))
651 let rhs = trim(part(text,equal+1))
652 if n.isEmpty {
653 try error("empty variable name")
654 }
655 let old = variables[n]
656 switch op {
657 case "=":
658 variables[n] = Variable(simple: false,value: rhs)
659 case ":=":
660 variables[n] = Variable(simple: true,value: try expand(rhs))
661 case "?=":
662 if old == nil {
663 variables[n] = Variable(simple: false,value: rhs)
664 }
665 case "+=":
666 if let old = old {
667 let addition: String
668 if old.simple {
669 addition = try expand(rhs)
670 } else {
671 addition = rhs
672 }
673 variables[n] = Variable(simple: old.simple,value: old.value.isEmpty ? addition : old.value+" "+addition)
674 } else {
675 variables[n] = Variable(simple: false,value: rhs)
676 }
677 default:
678 break
679 }
680 current = nil
681 } else if let colon = colon {
682 let targets = try unique(words(expand(part(text,0,colon))))
683 var remainder = part(text,colon+1)
684 let semi = try outside(remainder)[59]
685 let inline = semi.map {
686 String(part(remainder,$0+1).drop(while: {
687 $0 == " "
688 }))
689 }
690 if let semi = semi {
691 remainder = part(remainder,0,semi)
692 }
693 let second = try outside(remainder)[58]
694 var targetPattern: String?
695 if let second = second {
696 targetPattern = try trim(expand(part(remainder,0,second)))
697 remainder = part(remainder,second+1)
698 }
699 let deps = try words(expand(remainder))
700 var registered = [String]()
701 var implicit = [Rule]()
702 if targets == [".PHONY"] {
703 phony.formUnion(deps)
704 } else if let pattern = targetPattern {
705 if !pattern.contains("%") {
706 try error("target pattern contains no '%'")
707 }
708 for t in targets {
709 if let m = stem(pattern,t),!m.isEmpty {
710 register(t,deps.map {
711 substitute($0,m)
712 })
713 rules[t]!.stem = m
714 registered.append(t)
715 } else {
716 output.append("Makefile:\(lineNumber): target '\(t)' doesn't match the target pattern")
717 }
718 }
719 } else if targets.contains(where: {
720 $0.contains("%")
721 }) {
722 if targets.contains(where: {
723 !$0.contains("%")
724 }) {
725 try error("mixed implicit and normal rules")
726 }
727 for t in targets {
728 let r = Rule(t,deps)
729 patterns.append(r)
730 implicit.append(r)
731 }
732 } else {
733 for t in targets {
734 register(t,deps)
735 registered.append(t)
736 }
737 }
738 let rule = Current(registered,implicit)
739 if let inline = inline {
740 addRecipe(rule,RecipeLine(number: lineNumber,text: inline))
741 }
742 current = rule
743 } else if try !trim(expand(text)).isEmpty {
744 try error("missing separator")
745 }
746 }
747 if let top = conditions.last {
748 location = "Makefile:\(top.line)"
749 try error("missing 'endif'")
750 }
751 loaded = true
752 output.append("Makefile loaded: lines=\(body.count) variables=\(variables.count) targets=\(rules.count) patterns=\(patterns.filter { $0.recipe != nil }.count) goal=\(goal ?? "(none)")")
753 }
754 func determine(_ t: String) -> Determined {
755 let explicit = rules[t]
756 if let r = explicit,r.recipe != nil {
757 return Determined(deps: r.deps,recipe: r.recipe,stem: r.stem)
758 }
759 if !phony.contains(t) {
760 let candidates = patterns.enumerated().compactMap {
761 i,r -> (Rule,String,Int)? in guard r.recipe != nil,let m = stem(r.target,t),!m.isEmpty else {
762 return nil
763 }
764 return (r,m,i)
765 }.sorted {
766 $0.1.count == $1.1.count ? $0.2 < $1.2 : $0.1.count < $1.1.count
767 }
768 for (r,m,_) in candidates {
769 let deps = r.deps.map {
770 substitute($0,m)
771 }
772 if deps.allSatisfy({
773 files[$0] != nil || rules[$0] != nil
774 }) {
775 return Determined(deps: unique(deps+(explicit?.deps ?? [])),recipe: r.recipe,stem: m)
776 }
777 }
778 }
779 if explicit != nil || phony.contains(t) {
780 return Determined(deps: explicit?.deps ?? [],recipe: nil,stem: explicit?.stem ?? "")
781 }
782 return Determined(deps: [],recipe: nil,stem: "",source: files[t] != nil,absent: files[t] == nil)
783 }
784 func make(_ args: [String]) {
785 var options = Set<Character>()
786 var goals = [String]()
787 for arg in args {
788 if arg.hasPrefix("-") {
789 for c in arg.dropFirst() {
790 if !"nBkis".contains(c) {
791 output.append("make: invalid option -- '\(c)'")
792 return
793 }
794 options.insert(c)
795 }
796 } else {
797 goals.append(arg)
798 }
799 }
800 if goals.isEmpty {
801 if !loaded {
802 output.append("make: *** No targets specified and no makefile found. Stop.")
803 return
804 }
805 guard let goal = goal else {
806 output.append("make: *** No targets. Stop.")
807 return
808 }
809 goals = [goal]
810 }

813 do {
814 for g in goals {
815 let before = build.runs

817 if result == .success && build.runs == before {
818 output.append(phony.contains(g) || build.used[g]!.recipe == nil ? "make: Nothing to be done for '\(g)'." : "make: '\(g)' is up to date.")
819 } else if result == .dependencyFailure {
820 output.append("make: Target '\(g)' not remade because of errors.")
821 }
822 }
823 } catch {
824 }
825 if options.contains("n") {
826 files = original
827 clock = oldClock
828 }

830 scopes = []
831 expanding = []
832 location = "make"

834 func command(_ line: String) {
835 output.append("$ "+line)
836 let w = words(line)
837 switch w[0] {
838 case "MAKE":
839 make(Array(w.dropFirst()))
840 case "TOUCH":
841 for n in w.dropFirst() {
842 tick(n)
843 }
844 case "RM":
845 for n in w.dropFirst() {
846 if files.removeValue(forKey: n) == nil {
847 output.append("rm: cannot remove '\(n)': No such file or directory")
848 }
849 }
850 case "FILES":
851 output.append("clock \(clock)")
852 output += files.keys.sorted().map {
853 "\($0) \(files[$0]!)"
854 }
855 case "RULES":
856 if loaded {
857 for r in rules.keys.sorted().map({
858 rules[$0]!
859 })+patterns.filter({
860 $0.recipe != nil
861 }) {
862 output.append(r.target+":"+r.deps.map {
863 " "+$0
864 }.joined())
865 if let recipe = r.recipe {
866 output += recipe.lines.map {
867 ">"+$0.text
868 }
869 }
870 }
871 if !phony.isEmpty {
872 output.append(".PHONY:"+phony.sorted().map {
873 " "+$0
874 }.joined())
875 }
876 }
877 case "VARS":
878 if loaded {
879 do {
880 for n in variables.keys.sorted() {
881 let s = try trim(value(n))
882 output.append(n+" "+(variables[n]!.simple ? ":=" : "=")+(s.isEmpty ? "" : " "+s))
883 }
884 } catch let e as Expansion {
885 diagnostic(e)
886 } catch {
887 }
888 }
889 case "EXPAND":
890 do {
891 output.append("["+(try expand(part(line,7)))+"]")
892 } catch let e as Expansion {
893 diagnostic(e)
894 } catch {
895 }
896 default:
897 break
898 }
899 }
900}

917 func time(_ t: String) -> Int? {
918 sim.phony.contains(t) ? nil : sim.files[t]
919 }
920 func execute(_ command: String) -> Int {
921 let w = words(command)
922 switch w[0] {
923 case "echo":
924 sim.output.append(w.dropFirst().joined(separator: " "))
925 case "touch":
926 for n in w.dropFirst() {
927 sim.tick(n)
928 }
929 case "rm":
930 let force = w.count > 1 && w[1] == "-f"
931 var status = 0
932 for p in w.dropFirst(force ? 2 : 1) {
933 var names = sim.glob(p)
934 if names.isEmpty {
935 names = [p]
936 }
937 for n in names {
938 if sim.files.removeValue(forKey: n) == nil && !force {
939 sim.output.append("rm: cannot remove '\(n)': No such file or directory")
940 status = 1
941 }
942 }
943 }
944 return status
945 case "false":
946 return 1
947 case "exit":
948 return w.count > 1 ? Int(w[1])! : 0
949 default:
950 break
951 }
952 return 0
953 }
954 func run(_ t: String,_ d: Determined,_ deps: [String],_ fresh: [String]) throws -> Status {
955 sim.automatic = ["@":t,"<":deps.first ?? "","^":deps.joined(separator: " "),"?":fresh.joined(separator: " "),"*":d.stem]
956 var expanded = [(RecipeLine,String)]()
957 do {
958 for line in d.recipe!.lines {
959 sim.location = "Makefile:\(line.number)"
960 expanded.append((line,try sim.expand(line.text)))
961 }
962 } catch let e as Expansion {
963 sim.diagnostic(e)
964 sim.automatic = [:]
965 sim.location = "make"
966 throw Abort()
967 }
968 sim.automatic = [:]
969 sim.location = "make"
970 let dry = options.contains("n")
971 for (line,text) in expanded {
972 let prefix = String(text.prefix {
973 $0 == " " || $0 == "@" || $0 == "-"
974 })
975 var command = String(text.dropFirst(prefix.count))
976 while command.hasSuffix(" ") {
977 command.removeLast()
978 }
979 if command.isEmpty {
980 continue
981 }
982 if dry || (!prefix.contains("@") && !options.contains("s")) {
983 sim.output.append(command)
984 }
985 if dry {
986 continue
987 }
988 let status = execute(command)
989 if status != 0 {
990 if prefix.contains("-") || options.contains("i") {
991 sim.output.append("make: [Makefile:\(line.number): \(t)] Error \(status) (ignored)")
992 } else {
993 sim.output.append("make: *** [Makefile:\(line.number): \(t)] Error \(status)")
994 if !options.contains("k") {
995 throw Abort()
996 }
997 return .failure
998 }
999 }
1000 }
1001 return .success
1002 }
1003 func target(_ t: String,_ parent: String?) throws -> Status {
1004 if let result = results[t] {
1005 return result
1006 }
1007 let d = sim.determine(t)
1008 used[t] = d
1009 if d.absent {
1010 results[t] = .failure
1011 let keep = options.contains("k")
1012 sim.output.append("make: *** No rule to make target '\(t)'"+(parent.map {
1013 ", needed by '\($0)'"
1014 } ?? "")+"."+(keep ? "" : " Stop."))
1015 if !keep {
1016 throw Abort()
1017 }
1018 return .failure
1019 }
1020 if d.source {
1021 results[t] = .success
1022 return .success
1023 }
1024 being.insert(t)
1025 var deps = [String]()
1026 var failed = false
1027 for q in d.deps {
1028 if being.contains(q) {
1029 sim.output.append("make: Circular \(t) <- \(q) dependency dropped.")
1030 } else {
1031 deps.append(q)
1032 if try target(q,t) != .success {
1033 failed = true
1034 }
1035 }
1036 }
1037 being.remove(t)
1038 let result: Status
1039 if failed {
1040 result = .dependencyFailure
1041 } else {
1042 let oldTime = time(t)
1043 let fresh = deps.filter {
1044 n in options.contains("B") || oldTime == nil || time(n) == nil || time(n)! > oldTime!
1045 }
1046 if d.recipe == nil || (!options.contains("B") && oldTime != nil && fresh.isEmpty) {
1047 result = .success
1048 } else {
1049 runs += 1
1050 result = try run(t,d,deps,fresh)
1051 if result == .success && !sim.phony.contains(t) {
1052 sim.tick(t)
1053 }
1054 }
1055 }
1056 results[t] = result
1057 return result
1058 }
1059}
