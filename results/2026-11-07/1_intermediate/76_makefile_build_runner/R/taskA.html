48 closing <- function(text, opening) {
49 left <- at(text, opening)
50 right <- if (left == "(") ")" else "}"
51 depth <- 0
52 for (i in seq.int(opening, nchar(text) - 1L)) {
53 char <- at(text, i)
54 if (char == left) depth <- depth + 1
55 if (char == right) depth <- depth - 1
56 if (depth == 0) return(i)
57 }
58 error("unterminated variable reference")
59 }
60 outside <- function(text) {
61 result <- list()
62 i <- 0
63 while (i < nchar(text)) {
64 char <- at(text, i)
65 if (char == "$") {
66 i <- if (at(text, i + 1L) %in% c("(", "{")) closing(text, i + 1L) + 1L else i + 2L
67 } else {
68 if (is.null(result[[char]])) result[[char]] <- i
69 i <- i + 1L
70 }
71 }
72 result
73 }
74 stem <- function(pattern, text) {
75 position <- regexpr("%", pattern, fixed = TRUE)[1]
76 if (position < 0) return(if (pattern == text) "" else NULL)
77 left <- part(pattern, 0, position - 1L)
78 right <- part(pattern, position)
79 if (nchar(text) < nchar(left) + nchar(right) || !startsWith(text, left) || !endsWith(text, right)) return(NULL)
80 part(text, nchar(left), nchar(text) - nchar(right))
81 }
82 substitute <- function(text, value) {
83 position <- regexpr("%", text, fixed = TRUE)[1]
84 if (position < 0) text else paste0(part(text, 0, position - 1L), value, part(text, position))
85 }
86 replace <- function(word, pattern, replacement) {
87 value <- stem(pattern, word)
88 if (is.null(value)) return(word)
89 if (grepl("%", pattern, fixed = TRUE)) substitute(replacement, value) else replacement

91 wildcard <- function(pattern) {
92 if (!grepl("*", pattern, fixed = TRUE) && !grepl("?", pattern, fixed = TRUE)) {

96 quoted <- vapply(chars, function(char) {

98 }, character(1))
99 regex <- paste0("\\A", paste0(quoted, collapse = ""), "\\z")

108 variable_value <- function(name) {
109 variable <- state$variables[[name]]
110 if (is.null(variable)) return("")

117 reference_value <- function(name) {
118 for (i in rev(seq_along(state$scopes))) {
119 scope <- state$scopes[[i]]
120 if (scope$type == "foreach") {
121 if (scope$name == name) return(scope$value)
122 } else if (grepl("\\A[0-9]+\\z", name, perl = TRUE)) {
123 index <- as.integer(name) + 1L
124 return(if (index <= length(scope$values)) scope$values[index] else "")
125 }
126 }
127 if (!is.null(state$automatic) && name %in% names(state$automatic)) return(state$automatic[[name]])
128 variable_value(name)

130 expand <- function(text) {
131 result <- character()
132 i <- 0
133 while (i < nchar(text)) {
134 next_char <- at(text, i + 1L)
135 if (at(text, i) != "$") {
136 result <- c(result, at(text, i))
137 i <- i + 1L
138 } else if (next_char == "$") {
139 result <- c(result, "$")
140 i <- i + 2L
141 } else if (next_char %in% c("(", "{")) {
142 finish <- closing(text, i + 1L)
143 result <- c(result, reference(part(text, i + 2L, finish), next_char))
144 i <- finish + 1L
145 } else {
146 if (nzchar(next_char)) result <- c(result, reference_value(next_char))
147 i <- i + 2L
148 }
149 }
150 paste0(result, collapse = "")
151 }
152 reference <- function(content, bracket) {
153 space <- regexpr(" ", content, fixed = TRUE)[1]
154 name <- if (space > 0) part(content, 0, space - 1L) else ""
155 if (space > 0 && name %in% names(functions)) return(apply_function(name, sub("\\A +", "", part(content, space), perl = TRUE), bracket))
156 name <- expand(content)
157 colon <- regexpr(":", name, fixed = TRUE)[1]
158 equals <- if (colon > 0) regexpr("=", part(name, colon), fixed = TRUE)[1] else -1L
159 if (equals < 0) return(reference_value(name))
160 equals <- equals + colon
161 variable <- part(name, 0, colon - 1L)
162 pattern <- part(name, colon, equals - 1L)
163 replacement <- part(name, equals)
164 if (!grepl("%", pattern, fixed = TRUE)) {
165 pattern <- paste0("%", pattern)
166 replacement <- paste0("%", replacement)
167 }
168 paste(vapply(words(reference_value(variable)), function(word) replace(word, pattern, replacement), character(1)), collapse = " ")
169 }
170 literal_replace <- function(text, from, to) {
171 if (!nzchar(from)) return(text)
172 matches <- gregexpr(from, text, fixed = TRUE)[[1]]
173 if (matches[1] < 0) return(text)
174 result <- character()
175 start <- 0
176 for (position in matches) {
177 result <- c(result, part(text, start, position - 1L), to)
178 start <- position - 1L + nchar(from)
179 }
180 paste0(c(result, part(text, start)), collapse = "")
181 }
182 apply_function <- function(name, text, bracket) {
183 limits <- functions[[name]]
184 args <- character()
185 depth <- start <- 0
186 close <- if (bracket == "(") ")" else "}"
187 for (i in seq_len(nchar(text)) - 1L) {
188 char <- at(text, i)
189 if (char == bracket) depth <- depth + 1
190 if (char == close) depth <- depth - 1
191 if (char == "," && depth == 0 && length(args) < limits[2] - 1) {
192 args <- c(args, part(text, start, i))
193 start <- i + 1L
194 }
195 }
196 args <- c(args, part(text, start))
197 if (length(args) < limits[1]) error(paste0("insufficient number of arguments (", length(args), ") to function '", name, "'"))
198 if (name == "foreach") {
199 variable <- trim(expand(args[1]))
200 list <- words(expand(args[2]))
201 result <- vapply(list, function(word) with_scope(list(type = "foreach", name = variable, value = word), function() expand(args[3])), character(1))
202 return(paste(result, collapse = " "))
203 }
204 if (name == "if") {
205 condition <- trim(expand(args[1]))
206 return(expand(if (nzchar(condition)) args[2] else if (length(args) >= 3) args[3] else ""))
207 }
208 if (name == "call") {
209 values <- vapply(args, expand, character(1))
210 variable <- trim(values[1])
211 entry <- state$variables[[variable]]
212 if (is.null(entry)) return("")
213 if (entry$kind == ":=") return(entry$value)
214 return(with_scope(list(type = "call", values = c(variable, values[-1])), function() expand(entry$value)))
215 }
216 values <- vapply(args, expand, character(1))
217 if (name != "subst") values <- vapply(values, trim, character(1))
218 a <- values[1]
219 b <- if (length(values) >= 2) values[2] else ""
220 c <- if (length(values) >= 3) values[3] else ""
221 if (name == "subst") return(literal_replace(c, a, b))
222 if (name == "patsubst") return(paste(vapply(words(c), function(word) replace(word, a, b), character(1)), collapse = " "))
223 if (name == "strip") return(paste(words(a), collapse = " "))
224 if (name == "findstring") return(if (grepl(a, b, fixed = TRUE)) a else "")
225 if (name %in% c("filter", "filter-out")) {
226 patterns <- words(a)
227 list <- words(b)
228 matches <- vapply(list, function(word) any(vapply(patterns, function(pattern) !is.null(stem(pattern, word)), logical(1))), logical(1))
229 return(paste(list[matches == (name == "filter")], collapse = " "))
230 }
231 if (name == "sort") return(paste(ascii_sort(unique_words(words(a))), collapse = " "))
232 if (name == "word") {
233 if (!grepl("\\A[0-9]+\\z", a, perl = TRUE)) error(paste0("non-numeric first argument to 'word' function: '", a, "'"))
234 index <- as.integer(a)
235 if (index == 0) error("first argument to 'word' function must be greater than 0")
236 list <- words(b)
237 return(if (index <= length(list)) list[index] else "")
238 }
239 if (name == "wordlist") {
240 if (!grepl("\\A[0-9]+\\z", a, perl = TRUE)) error(paste0("non-numeric first argument to 'wordlist' function: '", a, "'"))
241 if (!grepl("\\A[0-9]+\\z", b, perl = TRUE)) error(paste0("non-numeric second argument to 'wordlist' function: '", b, "'"))
242 start <- as.integer(a)
243 end <- as.integer(b)
244 if (start == 0) error(paste0("invalid first argument to 'wordlist' function: '", a, "'"))
245 list <- words(c)
246 end <- min(end, length(list))
247 return(if (start > end) "" else paste(list[seq.int(start, end)], collapse = " "))
248 }
249 if (name == "words") return(as.character(length(words(a))))
250 if (name == "firstword") return(if (length(words(a)) > 0) words(a)[1] else "")
251 if (name == "lastword") return(if (length(words(a)) > 0) tail(words(a), 1) else "")
252 if (name %in% c("dir", "notdir", "suffix", "basename")) {
253 result <- character()
254 for (word in words(a)) {
255 slashes <- gregexpr("/", word, fixed = TRUE)[[1]]
256 dots <- gregexpr(".", word, fixed = TRUE)[[1]]
257 slash <- max(slashes)
258 dot <- max(dots)
259 extension <- dot > 0 && dot > slash
260 if (name == "dir") {
261 result <- c(result, if (slash < 0) "./" else part(word, 0, slash))
262 } else if (name == "notdir") {
263 result <- c(result, if (slash < 0) word else part(word, slash))
264 } else if (name == "basename") {
265 result <- c(result, if (extension) part(word, 0, dot - 1L) else word)
266 } else if (extension) {
267 result <- c(result, part(word, dot - 1L))
268 }
269 }
270 return(paste(result, collapse = " "))
271 }
272 if (name == "addsuffix") return(paste(vapply(words(b), function(word) paste0(word, a), character(1)), collapse = " "))
273 if (name == "addprefix") return(paste(vapply(words(b), function(word) paste0(a, word), character(1)), collapse = " "))
274 if (name == "wildcard") return(paste(unlist(lapply(words(a), wildcard), use.names = FALSE), collapse = " "))
275 if (name == "value") return(if (is.null(state$variables[[a]])) "" else state$variables[[a]]$value)
276 if (name == "info") emit(a)
277 if (name == "warning") emit(state$location, ": ", a)
278 if (name == "error") error(a)
279 ""
280 }

296 add_recipe <- function(current, number, text) {
297 if (is.null(current)) error("recipe commences before first target")
298 if (length(current$entries) == 0) return(invisible(NULL))
299 if (is.null(current$recipe)) {
300 recipe <- new.env(parent = emptyenv())
301 recipe$lines <- list()
302 for (rule in current$entries) {
303 if (current$explicit && !is.null(rule$recipe)) {
304 emit("Makefile:", number, ": warning: overriding recipe for target '", rule$target, "'")
305 emit("Makefile:", rule$recipe$lines[[1]]$number, ": warning: ignoring old recipe for target '", rule$target, "'")
306 }
307 rule$recipe <- recipe
308 }
309 current$recipe <- recipe
310 }
311 current$recipe$lines[[length(current$recipe$lines) + 1L]] <- list(number = number, text = text)
312 }
313 parse_rule <- function(text, colon, number) {
314 targets <- unique_words(words(expand(part(text, 0, colon))))
315 rest <- part(text, colon + 1L)
316 semicolon <- outside(rest)[[";"]]
317 inline <- if (is.null(semicolon)) NULL else sub("\\A +", "", part(rest, semicolon + 1L), perl = TRUE)
318 if (!is.null(semicolon)) rest <- part(rest, 0, semicolon)
319 second <- outside(rest)[[":"]]
320 pattern <- if (is.null(second)) NULL else trim(expand(part(rest, 0, second)))
321 if (!is.null(second)) rest <- part(rest, second + 1L)
322 prerequisites <- words(expand(rest))
323 entries <- list()
324 explicit <- TRUE
325 if (identical(targets, ".PHONY")) {
326 state$phony <- unique_words(c(state$phony, prerequisites))
327 } else if (!is.null(pattern)) {
328 if (!grepl("%", pattern, fixed = TRUE)) error("target pattern contains no '%'")
329 for (target in targets) {
330 value <- stem(pattern, target)
331 if (!is.null(value) && nzchar(value)) {
332 entry <- register(target, vapply(prerequisites, function(p) substitute(p, value), character(1)))
333 entry$stem <- value
334 entries[[length(entries) + 1L]] <- entry
335 } else {
336 emit("Makefile:", number, ": target '", target, "' doesn't match the target pattern")
337 }
338 }
339 } else if (any(grepl("%", targets, fixed = TRUE))) {
340 if (!all(grepl("%", targets, fixed = TRUE))) error("mixed implicit and normal rules")
341 explicit <- FALSE
342 for (target in targets) {
343 entry <- new_rule(target, prerequisites)
344 state$patterns[[length(state$patterns) + 1L]] <- entry
345 entries[[length(entries) + 1L]] <- entry
346 }
347 } else {
348 entries <- lapply(targets, function(target) register(target, prerequisites))
349 }
350 current <- new.env(parent = emptyenv())
351 current$entries <- entries
352 current$explicit <- explicit
353 current$recipe <- NULL
354 if (!is.null(inline)) add_recipe(current, number, inline)
355 current
356 }
357 load_makefile <- function(body) {
358 clear_makefile()
359 on.exit(state$location <- "make")
360 tryCatch({
361 current <- NULL
362 conditions <- list()
363 i <- 1L
364 while (i <= length(body)) {
365 number <- i
366 state$location <- paste0("Makefile:", number)
367 top <- if (length(conditions) == 0) NULL else tail(conditions, 1)[[1]]
368 active <- is.null(top) || (top$outer && top$value)
369 text <- body[i]
370 i <- i + 1L
371 if (startsWith(text, ">")) {
372 if (active) add_recipe(current, number, sub("\\A +", "", part(text, 1), perl = TRUE))
373 next
374 }
375 while (endsWith(text, "\\") && i <= length(body)) {
376 text <- paste0(sub(" +\\z", "", part(text, 0, nchar(text) - 1L), perl = TRUE), " ", sub("\\A +", "", body[i], perl = TRUE))
377 i <- i + 1L
378 }
379 hash <- regexpr("#", text, fixed = TRUE)[1]
380 if (hash > 0) text <- part(text, 0, hash - 1L)
381 text <- trim(text)
382 space <- regexpr(" ", text, fixed = TRUE)[1]
383 first <- if (space < 0) text else part(text, 0, space - 1L)
384 rest <- if (space < 0) "" else trim(part(text, space))
385 if (first %in% c("ifdef", "ifndef", "ifeq", "ifneq")) {
386 value <- FALSE
387 if (active) {
388 if (first %in% c("ifdef", "ifndef")) {
389 variable <- state$variables[[trim(expand(rest))]]
390 value <- !is.null(variable) && nzchar(variable$value)
391 if (first == "ifndef") value <- !value
392 } else {
393 if (nchar(rest) < 2 || !startsWith(rest, "(") || !endsWith(rest, ")")) error("invalid syntax in conditional")
394 inner <- part(rest, 1, nchar(rest) - 1L)
395 depth <- 0
396 comma <- NULL
397 for (p in seq_len(nchar(inner)) - 1L) {
398 char <- at(inner, p)
399 if (char == "(") depth <- depth + 1
400 if (char == ")") depth <- depth - 1
401 if (char == "," && depth == 0) {
402 comma <- p
403 break
404 }
405 }
406 if (is.null(comma)) error("invalid syntax in conditional")
407 left <- trim(expand(part(inner, 0, comma)))
408 right <- trim(expand(part(inner, comma + 1L)))
409 value <- left == right
410 if (first == "ifneq") value <- !value
411 }
412 }
413 conditions[[length(conditions) + 1L]] <- list(outer = active, value = value, seen_else = FALSE, number = number)
414 } else if (first == "else") {
415 if (length(conditions) == 0) error("extraneous 'else'")
416 p <- length(conditions)
417 if (conditions[[p]]$seen_else) error("only one 'else' per conditional")
418 conditions[[p]]$value <- !conditions[[p]]$value
419 conditions[[p]]$seen_else <- TRUE
420 } else if (first == "endif") {
421 if (length(conditions) == 0) error("extraneous 'endif'")
422 conditions <- head(conditions, -1L)
423 } else if (active && nzchar(text)) {
424 positions <- outside(text)
425 equals <- positions[["="]]
426 colon <- positions[[":"]]
427 if (!is.null(equals) && (is.null(colon) || colon >= equals - 1L)) {
428 operator_start <- if (equals > 0 && at(text, equals - 1L) %in% c(":", "+", "?")) equals - 1L else equals
429 operator <- part(text, operator_start, equals + 1L)
430 variable <- trim(part(text, 0, operator_start))
431 if (!nzchar(variable)) error("empty variable name")
432 rhs <- trim(part(text, equals + 1L))
433 old <- state$variables[[variable]]
434 if (operator == ":=") {
435 state$variables[[variable]] <- list(kind = ":=", value = expand(rhs))
436 } else if (operator == "+=" && !is.null(old)) {
437 addition <- if (old$kind == ":=") expand(rhs) else rhs
438 state$variables[[variable]] <- list(kind = old$kind, value = if (!nzchar(old$value)) addition else paste0(old$value, " ", addition))
439 } else if (operator != "?=" || is.null(old)) {
440 state$variables[[variable]] <- list(kind = "=", value = rhs)
441 }
442 current <- NULL
443 } else if (!is.null(colon)) {
444 current <- parse_rule(text, colon, number)
445 } else if (nzchar(trim(expand(text)))) {
446 error("missing separator")
447 }
448 }
449 }
450 if (length(conditions) > 0) {
451 state$location <- paste0("Makefile:", tail(conditions, 1)[[1]]$number)
452 error("missing 'endif'")
453 }
454 state$loaded <- TRUE
455 patterns <- sum(vapply(state$patterns, function(rule) !is.null(rule$recipe), logical(1)))
456 emit("Makefile loaded: lines=", length(body), " variables=", length(state$variables), " targets=", length(state$rules), " patterns=", patterns, " goal=", if (is.null(state$default)) "(none)" else state$default)
457 }, make_error = function(error) {
458 report(error)
459 clear_makefile()
460 })
461 }

501 }

503 simulate <- function(command) {
504 parts <- words(command)
505 first <- parts[1]
506 parts <- parts[-1]
507 if (first == "echo") {
508 emit(paste(parts, collapse = " "))
509 } else if (first == "touch") {
510 for (name in parts) new_time(name)
511 } else if (first == "rm") {
512 forced <- length(parts) > 0 && parts[1] == "-f"
513 if (forced) parts <- parts[-1]
514 result <- 0
515 for (pattern in parts) {
516 matches <- wildcard(pattern)
517 if (length(matches) == 0) matches <- pattern
518 for (name in matches) result <- max(result, remove_file(name, forced))
519 }
520 return(result)
521 } else if (first == "false") {
522 return(1)
523 } else if (first == "exit") {
524 return(if (length(parts) == 0) 0 else as.integer(parts[1]))
525 }
526 0

528 run_recipe <- function(target, prerequisites, recipe, value, newer, options) {
529 state$automatic <- list("@" = target, "<" = if (length(prerequisites) == 0) "" else prerequisites[1], "^" = paste(prerequisites, collapse = " "), "?" = paste(newer, collapse = " "), "*" = value)
530 on.exit({
531 state$automatic <- NULL
532 state$location <- "make"
533 })
534 expanded <- lapply(recipe$lines, function(line) {
535 state$location <- paste0("Makefile:", line$number)
536 list(number = line$number, text = expand(line$text))
537 })
538 for (line in expanded) {
539 text <- line$text
540 prefix_length <- 0L
541 while (prefix_length < nchar(text) && at(text, prefix_length) %in% c(" ", "@", "-")) prefix_length <- prefix_length + 1L
542 prefix <- part(text, 0, prefix_length)
543 command <- sub(" +\\z", "", part(text, prefix_length), perl = TRUE)
544 if (!nzchar(command)) next
545 if ("n" %in% options || (!grepl("@", prefix, fixed = TRUE) && !("s" %in% options))) emit(command)
546 if ("n" %in% options) next
547 code <- simulate(command)
548 if (code == 0) next
549 if (grepl("-", prefix, fixed = TRUE) || "i" %in% options) {
550 emit("make: [Makefile:", line$number, ": ", target, "] Error ", code, " (ignored)")
551 } else {
552 emit("make: *** [Makefile:", line$number, ": ", target, "] Error ", code)
553 if (!("k" %in% options)) abort_build()
554 return("failure")
555 }
556 }
557 "success"
558 }
559 make <- function(args) {
560 options <- goals <- character()
561 for (word in words(args)) {
562 if (startsWith(word, "-")) {
563 for (char in strsplit(part(word, 1), "", fixed = TRUE)[[1]]) {
564 if (!(char %in% c("n", "B", "k", "i", "s"))) {
565 emit("make: invalid option -- '", char, "'")
566 return(invisible(NULL))
567 }
568 options <- c(options, char)
569 }
570 } else {
571 goals <- c(goals, word)
572 }
573 }
574 if (length(goals) == 0) {
575 if (!state$loaded) {
576 emit("make: *** No targets specified and no makefile found. Stop.")
577 return(invisible(NULL))
578 }
579 if (is.null(state$default)) {
580 emit("make: *** No targets. Stop.")
581 return(invisible(NULL))
582 }
583 goals <- state$default
584 }

586 saved_clock <- state$clock
587 on.exit({

595 results <- used <- building <- list()
596 count <- 0
597 build <- function(target, parent = NULL) {
598 if (!is.null(results[[target]])) return(results[[target]])
599 rule <- determine(target)
600 used[[target]] <<- rule
601 if (rule$kind == "missing") {
602 results[[target]] <<- "failure"
603 emit("make: *** No rule to make target '", target, "'", if (is.null(parent)) "" else paste0(", needed by '", parent, "'"), ".", if ("k" %in% options) "" else " Stop.")
604 if (!("k" %in% options)) abort_build()
605 return("failure")
606 }
607 if (rule$kind == "source") {
608 results[[target]] <<- "success"
609 return("success")
610 }
611 building[[target]] <<- TRUE
612 failed <- FALSE
613 kept <- character()
614 for (p in rule$prerequisites) {
615 if (!is.null(building[[p]])) {
616 emit("make: Circular ", target, " <- ", p, " dependency dropped.")
617 } else {
618 kept <- c(kept, p)
619 if (build(p, target) != "success") failed <- TRUE
620 }
621 }
622 building[[target]] <<- NULL
623 if (failed) {
624 result <- "dependency_failure"
625 } else {
626 time <- target_time(target)
627 newer <- kept[vapply(kept, function(p) "B" %in% options || is.null(time) || is.null(target_time(p)) || target_time(p) > time, logical(1))]
628 if (is.null(rule$recipe) || (!("B" %in% options) && !is.null(time) && length(newer) == 0)) {
629 result <- "success"
630 } else {
631 count <<- count + 1
632 result <- run_recipe(target, kept, rule$recipe, rule$stem, newer, options)
633 if (result == "success" && !(target %in% state$phony)) new_time(target)
634 }
635 }
636 results[[target]] <<- result
637 result
638 }
639 tryCatch({
640 for (goal in goals) {
641 before <- count
642 result <- build(goal)
643 if (result == "success" && count == before) {
644 emit(if (goal %in% state$phony || is.null(used[[goal]]$recipe)) paste0("make: Nothing to be done for '", goal, "'.") else paste0("make: '", goal, "' is up to date."))
645 } else if (result == "dependency_failure") {
646 emit("make: Target '", goal, "' not remade because of errors.")
647 }
648 }
649 }, build_abort = function(error) NULL, make_error = report)
650 invisible(NULL)
651 }
652 execute <- function(line, body) {
653 emit("$ ", line)
654 space <- regexpr(" ", line, fixed = TRUE)[1]
655 command <- if (space < 0) line else part(line, 0, space - 1L)
656 args <- if (space < 0) "" else part(line, space)
657 if (command == "MAKEFILE") {
658 load_makefile(body)
659 } else if (command == "MAKE") {
660 make(args)
661 } else if (command == "TOUCH") {
662 for (name in words(args)) new_time(name)
663 } else if (command == "RM") {
664 for (name in words(args)) remove_file(name)
665 } else if (command == "FILES") {
666 emit("clock ", state$clock)
667 for (name in ascii_sort(names(state$files))) emit(name, " ", state$files[[name]])
668 } else if (command == "VARS" && state$loaded) {
669 for (name in ascii_sort(names(state$variables))) {
670 value <- trim(variable_value(name))
671 emit(name, " ", state$variables[[name]]$kind, if (nzchar(value)) paste0(" ", value) else "")
672 }
673 } else if (command == "RULES" && state$loaded) {
674 rules <- c(state$rules[ascii_sort(names(state$rules))], Filter(function(rule) !is.null(rule$recipe), state$patterns))
675 for (rule in rules) {
676 emit(rule$target, ":", if (length(rule$prerequisites) == 0) "" else paste0(" ", paste(rule$prerequisites, collapse = " ")))
677 if (!is.null(rule$recipe)) for (line in rule$recipe$lines) emit(">", line$text)
678 }
679 if (length(state$phony) > 0) emit(".PHONY: ", paste(ascii_sort(state$phony), collapse = " "))
680 } else if (command == "EXPAND") {
681 result <- expand(args)
682 emit("[", result, "]")
683 }
684 invisible(NULL)
685 }
686 function(line, body = NULL) tryCatch(execute(line, body), make_error = report)
687}
