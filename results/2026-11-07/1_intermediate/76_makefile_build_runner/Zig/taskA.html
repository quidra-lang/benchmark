7fn equal(a: Text, b: Text) bool {

13fn number(s: Text) usize {
14 var result: usize = 0;
15 for (s) |c| result = result * 10 + c - '0';
16 return result;
17}
18fn numeric(s: Text) bool {
19 if (s.len == 0) return false;
20 for (s) |c| if (!std.ascii.isDigit(c)) return false;
21 return true;
22}

36fn words(gpa: Allocator, s: Text) Allocator.Error![]Text {
37 var list: std.ArrayList(Text) = .empty;
38 var iterator = std.mem.tokenizeScalar(u8, s, ' ');
39 while (iterator.next()) |word| try list.append(gpa, word);
40 return list.items;
41}
42fn unique(gpa: Allocator, list: []const Text) Allocator.Error![]Text {
43 var result: std.ArrayList(Text) = .empty;
44 var seen: std.StringHashMapUnmanaged(void) = .empty;
45 for (list) |item| if (!seen.contains(item)) {
46 try seen.put(gpa, item, {});
47 try result.append(gpa, item);
48 };
49 return result.items;
50}
51fn textLess(_: void, a: Text, b: Text) bool {
52 return std.mem.order(u8, a, b) == .lt;
53}
54fn keys(gpa: Allocator, map: anytype) Allocator.Error![]Text {
55 var list: std.ArrayList(Text) = .empty;
56 var iterator = map.keyIterator();
57 while (iterator.next()) |key| try list.append(gpa, key.*);
58 std.mem.sort(Text, list.items, {}, textLess);
59 return list.items;
60}
61fn join(gpa: Allocator, list: []const Text, separator: Text) Allocator.Error!Text {
62 var out: Builder = .{ .gpa = gpa };
63 for (list, 0..) |item, i| {
64 if (i > 0) try out.add(separator);
65 try out.add(item);
66 }
67 return out.bytes.items;
68}
69fn lastScalar(s: Text, c: u8) ?usize {
70 var i = s.len;
71 while (i > 0) {
72 i -= 1;
73 if (s[i] == c) return i;
74 }
75 return null;
76}
77fn match(pattern: Text, word: Text) ?Text {
78 const at = std.mem.findScalar(u8, pattern, '%') orelse return if (equal(pattern, word)) "" else null;
79 const a = pattern[0..at];
80 const b = pattern[at + 1 ..];
81 if (word.len < a.len + b.len or !std.mem.startsWith(u8, word, a) or !std.mem.endsWith(u8, word, b)) return null;
82 return word[a.len .. word.len - b.len];
83}
84fn substitute(gpa: Allocator, pattern: Text, stem: Text) Allocator.Error!Text {
85 const at = std.mem.findScalar(u8, pattern, '%') orelse return pattern;
86 return std.fmt.allocPrint(gpa, "{s}{s}{s}", .{ pattern[0..at], stem, pattern[at + 1 ..] });
87}
88fn replacement(gpa: Allocator, word: Text, pattern: Text, value: Text) Allocator.Error!Text {
89 const stem = match(pattern, word) orelse return word;
90 return if (std.mem.findScalar(u8, pattern, '%') != null) try substitute(gpa, value, stem) else value;
91}
92fn wildcard(gpa: Allocator, pattern: Text, name: Text) Allocator.Error!bool {
93 var previous = try gpa.alloc(bool, name.len + 1);
94 var next = try gpa.alloc(bool, name.len + 1);
95 @memset(previous, false);
96 previous[0] = true;
97 for (pattern) |c| {
98 @memset(next, false);
99 next[0] = c == '*' and previous[0];
100 for (name, 0..) |n, i| {
101 next[i + 1] = if (c == '*') previous[i + 1] or (n != '/' and next[i]) else previous[i] and (c == n or c == '?' and n != '/');
102 }
103 std.mem.swap([]bool, &previous, &next);
104 }
105 return previous[name.len];
106}

170 fn glob(self: *Simulator, pattern: Text) Error![]Text {
171 var list: std.ArrayList(Text) = .empty;
172 for (try keys(self.gpa, self.files)) |name| if (try wildcard(self.gpa, pattern, name)) try list.append(self.gpa, name);
173 return list.items;
174 }
175 fn reference(self: *Simulator, name: Text) Error!Text {
176 var i = self.scopes.items.len;
177 while (i > 0) {
178 i -= 1;
179 switch (self.scopes.items[i]) {
180 .foreach => |scope| if (equal(scope.name, name)) return scope.word,
181 .call => |values| if (numeric(name)) {
182 const n = number(name);
183 return if (n < values.len) values[n] else "";
184 },
185 }
186 }
187 if (self.automatic.get(name)) |value| return value;
188 const v = self.variables.get(name) orelse return "";
189 if (v.simple) return v.value;
190 if (self.expanding.contains(name)) return self.fail("Recursive variable '{s}' references itself (eventually)", .{name});
191 try self.expanding.put(self.gpa, name, {});
192 defer _ = self.expanding.remove(name);
193 return self.expand(v.value);
194 }
195 fn closing(self: *Simulator, s: Text, open: usize) Error!usize {
196 const close: u8 = if (s[open] == '(') ')' else '}';
197 var depth: isize = 0;
198 for (open..s.len) |i| {
199 if (s[i] == s[open]) {
200 depth += 1;
201 } else if (s[i] == close) {
202 depth -= 1;
203 if (depth == 0) return i;
204 }
205 }
206 return self.fail("unterminated variable reference", .{});
207 }
208 fn outside(self: *Simulator, s: Text, wanted: u8) Error!?usize {
209 var found: ?usize = null;
210 var i: usize = 0;
211 while (i < s.len) : (i += 1) {
212 if (s[i] == '$') {
213 if (i + 1 < s.len and (s[i + 1] == '(' or s[i + 1] == '{')) {
214 i = try self.closing(s, i + 1);
215 } else if (i + 1 < s.len) {
216 i += 1;
217 }
218 } else if (s[i] == wanted and found == null) {
219 found = i;
220 }
221 }
222 return found;
223 }
224 fn expand(self: *Simulator, s: Text) Error!Text {
225 var out: Builder = .{ .gpa = self.gpa };
226 var i: usize = 0;
227 while (i < s.len) : (i += 1) {
228 if (s[i] != '$') {
229 try out.bytes.append(self.gpa, s[i]);
230 continue;
231 }
232 i += 1;
233 if (i == s.len) break;
234 if (s[i] == '$') {
235 try out.add("$");
236 } else if (s[i] == '(' or s[i] == '{') {
237 const finish = try self.closing(s, i);
238 const content = s[i + 1 .. finish];
239 var called = false;
240 if (std.mem.findScalar(u8, content, ' ')) |at| {
241 if (findFunction(content[0..at])) |function| {
242 try out.add(try self.invoke(function, std.mem.trimStart(u8, content[at + 1 ..], " "), s[i]));
243 called = true;
244 }
245 }
246 if (!called) {
247 const name = try self.expand(content);
248 var substituted = false;
249 if (std.mem.findScalar(u8, name, ':')) |colon| {
250 if (std.mem.findScalar(u8, name[colon + 1 ..], '=')) |offset| {
251 const eq = colon + 1 + offset;
252 var from = name[colon + 1 .. eq];
253 var to = name[eq + 1 ..];
254 if (std.mem.findScalar(u8, from, '%') == null) {
255 from = try std.fmt.allocPrint(self.gpa, "%{s}", .{from});
256 to = try std.fmt.allocPrint(self.gpa, "%{s}", .{to});
257 }
258 const list = try words(self.gpa, try self.reference(name[0..colon]));
259 for (list) |*word| word.* = try replacement(self.gpa, word.*, from, to);
260 try out.add(try join(self.gpa, list, " "));
261 substituted = true;
262 }
263 }
264 if (!substituted) try out.add(try self.reference(name));
265 }
266 i = finish;
267 } else {
268 try out.add(try self.reference(s[i .. i + 1]));
269 }
270 }
271 return out.bytes.items;
272 }
273 fn invoke(self: *Simulator, function: FunctionInfo, text: Text, open: u8) Error!Text {
274 var raw: std.ArrayList(Text) = .empty;
275 var depth: isize = 0;
276 var start: usize = 0;
277 const close: u8 = if (open == '(') ')' else '}';
278 for (text, 0..) |c, i| {
279 if (c == open) {
280 depth += 1;
281 } else if (c == close) {
282 depth -= 1;
283 }
284 if (c == ',' and depth == 0 and (function.maximum == 0 or raw.items.len + 1 < function.maximum)) {
285 try raw.append(self.gpa, text[start..i]);
286 start = i + 1;
287 }
288 }
289 try raw.append(self.gpa, text[start..]);
290 if (raw.items.len < function.minimum) return self.fail("insufficient number of arguments ({d}) to function '{s}'", .{ raw.items.len, function.name });
291 const args = raw.items;
292 const kind = function.kind;
293 if (kind == .conditional) {
294 if (trim(try self.expand(args[0])).len > 0) return self.expand(args[1]);
295 return if (args.len > 2) try self.expand(args[2]) else "";
296 }
297 if (kind == .foreach) {
298 const name = trim(try self.expand(args[0]));
299 const list = try words(self.gpa, try self.expand(args[1]));
300 var result: std.ArrayList(Text) = .empty;
301 for (list) |word| {
302 try self.scopes.append(self.gpa, .{ .foreach = .{ .name = name, .word = word } });
303 const value = self.expand(args[2]) catch |err| {
304 _ = self.scopes.pop();
305 return err;
306 };
307 _ = self.scopes.pop();
308 try result.append(self.gpa, value);
309 }
310 return join(self.gpa, result.items, " ");
311 }
312 for (args) |*arg| {
313 arg.* = try self.expand(arg.*);
314 if (kind != .subst and kind != .call) arg.* = trim(arg.*);
315 }
316 if (kind == .call) {
317 args[0] = trim(args[0]);
318 const v = self.variables.get(args[0]) orelse return "";
319 if (v.simple) return v.value;
320 try self.scopes.append(self.gpa, .{ .call = args });
321 defer _ = self.scopes.pop();
322 return self.expand(v.value);
323 }
324 var result: std.ArrayList(Text) = .empty;
325 switch (kind) {
326 .subst => {
327 if (args[0].len == 0) return args[2];
328 var out: Builder = .{ .gpa = self.gpa };
329 var cursor: usize = 0;
330 while (std.mem.find(u8, args[2][cursor..], args[0])) |offset| {
331 const at = cursor + offset;
332 try out.add(args[2][cursor..at]);
333 try out.add(args[1]);
334 cursor = at + args[0].len;
335 }
336 try out.add(args[2][cursor..]);
337 return out.bytes.items;
338 },
339 .patsubst => {
340 const list = try words(self.gpa, args[2]);
341 for (list) |*word| word.* = try replacement(self.gpa, word.*, args[0], args[1]);
342 return join(self.gpa, list, " ");
343 },
344 .strip => return join(self.gpa, try words(self.gpa, args[0]), " "),
345 .findstring => return if (std.mem.find(u8, args[1], args[0]) != null) args[0] else "",
346 .filter, .filter_out => {
347 const patterns = try words(self.gpa, args[0]);
348 for (try words(self.gpa, args[1])) |word| {
349 var matched = false;
350 for (patterns) |pattern| if (match(pattern, word) != null) {
351 matched = true;
352 break;
353 };
354 if (matched == (kind == .filter)) try result.append(self.gpa, word);
355 }
356 },
357 .sort => {
358 const list = try unique(self.gpa, try words(self.gpa, args[0]));
359 std.mem.sort(Text, list, {}, textLess);
360 return join(self.gpa, list, " ");
361 },
362 .word => {
363 if (!numeric(args[0])) return self.fail("non-numeric first argument to 'word' function: '{s}'", .{args[0]});
364 const n = number(args[0]);
365 if (n == 0) return self.fail("first argument to 'word' function must be greater than 0", .{});
366 const list = try words(self.gpa, args[1]);
367 return if (n <= list.len) list[n - 1] else "";
368 },
369 .wordlist => {
370 if (!numeric(args[0])) return self.fail("non-numeric first argument to 'wordlist' function: '{s}'", .{args[0]});
371 if (!numeric(args[1])) return self.fail("non-numeric second argument to 'wordlist' function: '{s}'", .{args[1]});
372 const a = number(args[0]);
373 if (a == 0) return self.fail("invalid first argument to 'wordlist' function: '{s}'", .{args[0]});
374 const list = try words(self.gpa, args[2]);
375 const b = @min(number(args[1]), list.len);
376 return if (a > b) "" else try join(self.gpa, list[a - 1 .. b], " ");
377 },
378 .words => return std.fmt.allocPrint(self.gpa, "{d}", .{(try words(self.gpa, args[0])).len}),
379 .firstword, .lastword => {
380 const list = try words(self.gpa, args[0]);
381 return if (list.len == 0) "" else list[if (kind == .firstword) 0 else list.len - 1];
382 },
383 .dir, .notdir, .suffix, .basename => {
384 for (try words(self.gpa, args[0])) |word| {
385 const slash = lastScalar(word, '/');
386 const dot = lastScalar(word, '.');
387 const extension = dot != null and (slash == null or dot.? > slash.?);
388 switch (kind) {
389 .dir => try result.append(self.gpa, if (slash) |at| word[0 .. at + 1] else "./"),
390 .notdir => try result.append(self.gpa, word[if (slash) |at| at + 1 else 0..]),
391 .suffix => if (extension) {
392 try result.append(self.gpa, word[dot.?..]);
393 },
394 .basename => try result.append(self.gpa, if (extension) word[0..dot.?] else word),
395 else => unreachable,
396 }
397 }
398 },
399 .addprefix, .addsuffix => for (try words(self.gpa, args[1])) |word| {
400 try result.append(self.gpa, try std.fmt.allocPrint(self.gpa, "{s}{s}", .{ if (kind == .addprefix) args[0] else word, if (kind == .addprefix) word else args[0] }));
401 },
402 .wildcard => for (try words(self.gpa, args[0])) |pattern| {
403 try result.appendSlice(self.gpa, try self.glob(pattern));
404 },
405 .value => return if (self.variables.get(args[0])) |v| v.value else "",
406 .info => try self.out.line("{s}", .{args[0]}),
407 .warning => try self.out.line("{s}: {s}", .{ self.location, args[0] }),
408 .failure => return self.fail("{s}", .{args[0]}),
409 else => unreachable,
410 }
411 return join(self.gpa, result.items, " ");
412 }
413 fn register(self: *Simulator, target: Text, prerequisites: []const Text) Error!*Rule {
414 const r = self.rules.get(target) orelse blk: {
415 const created = try self.gpa.create(Rule);
416 created.* = .{ .target = target };
417 try self.rules.put(self.gpa, target, created);
418 break :blk created;
419 };
420 var combined: std.ArrayList(Text) = .empty;
421 try combined.appendSlice(self.gpa, r.prerequisites);
422 try combined.appendSlice(self.gpa, prerequisites);
423 r.prerequisites = try unique(self.gpa, combined.items);
424 if (self.goal == null and !std.mem.startsWith(u8, target, ".")) self.goal = target;
425 return r;
426 }
427 fn addRecipe(self: *Simulator, group: *Group, line: RecipeLine) Error!void {
428 if (group.targets.items.len == 0) return;
429 if (group.recipe == null) {
430 const created = try self.gpa.create(Recipe);
431 created.* = .empty;
432 group.recipe = created;
433 for (group.targets.items) |r| {
434 if (group.explicit) {
435 if (r.recipe) |old| {
436 try self.out.line("Makefile:{d}: warning: overriding recipe for target '{s}'", .{ line.number, r.target });
437 try self.out.line("Makefile:{d}: warning: ignoring old recipe for target '{s}'", .{ old.items[0].number, r.target });
438 }
439 }
440 r.recipe = created;
441 }
442 }
443 try group.recipe.?.append(self.gpa, line);
444 }
445 fn active(stack: []const Condition) bool {
446 return stack.len == 0 or stack[stack.len - 1].outer and stack[stack.len - 1].value;
447 }
448 fn load(self: *Simulator, body: []const Text) Error!void {
449 self.clear();
450 self.parseBody(body) catch |err| {
451 if (err != error.Invalid) return err;
452 try self.out.line("{s}: *** {s}. Stop.", .{ failureLocation, failure });
453 self.clear();
454 return;
455 };
456 self.loaded = true;
457 var count: usize = 0;
458 for (self.patterns.items) |r| {
459 if (r.recipe != null) count += 1;
460 }
461 try self.out.line("Makefile loaded: lines={d} variables={d} targets={d} patterns={d} goal={s}", .{ body.len, self.variables.count(), self.rules.count(), count, self.goal orelse "(none)" });
462 }
463 fn parseBody(self: *Simulator, body: []const Text) Error!void {
464 var current: ?*Group = null;
465 var stack: std.ArrayList(Condition) = .empty;
466 var index: usize = 0;
467 while (index < body.len) : (index += 1) {
468 const lineNumber = index + 1;
469 var text = body[index];
470 self.location = try std.fmt.allocPrint(self.gpa, "Makefile:{d}", .{lineNumber});
471 if (std.mem.startsWith(u8, text, ">")) {
472 if (!active(stack.items)) continue;
473 if (current == null) return self.fail("recipe commences before first target", .{});
474 try self.addRecipe(current.?, .{ .number = lineNumber, .text = std.mem.trimStart(u8, text[1..], " ") });
475 continue;
476 }
477 while (std.mem.endsWith(u8, text, "\\") and index + 1 < body.len) {
478 index += 1;
479 text = try std.fmt.allocPrint(self.gpa, "{s} {s}", .{ std.mem.trimEnd(u8, text[0 .. text.len - 1], " "), std.mem.trimStart(u8, body[index], " ") });
480 }
481 if (std.mem.findScalar(u8, text, '#')) |at| text = text[0..at];
482 text = trim(text);
483 const space = std.mem.findScalar(u8, text, ' ') orelse text.len;
484 const first = text[0..space];
485 const rest = if (space < text.len) trim(text[space + 1 ..]) else "";
486 if (equal(first, "else")) {
487 if (stack.items.len == 0) return self.fail("extraneous 'else'", .{});
488 const top = &stack.items[stack.items.len - 1];
489 if (top.otherwise) return self.fail("only one 'else' per conditional", .{});
490 top.value = !top.value;
491 top.otherwise = true;
492 continue;
493 }
494 if (equal(first, "endif")) {
495 if (stack.items.len == 0) return self.fail("extraneous 'endif'", .{});
496 _ = stack.pop();
497 continue;
498 }
499 if (equal(first, "ifdef") or equal(first, "ifndef") or equal(first, "ifeq") or equal(first, "ifneq")) {
500 const outer = active(stack.items);
501 var value = false;
502 if (outer) {
503 if (equal(first, "ifdef") or equal(first, "ifndef")) {
504 const v = self.variables.get(trim(try self.expand(rest)));
505 value = v != null and v.?.value.len > 0;
506 if (equal(first, "ifndef")) value = !value;
507 } else {
508 if (rest.len < 2 or rest[0] != '(' or rest[rest.len - 1] != ')') return self.fail("invalid syntax in conditional", .{});
509 const inside = rest[1 .. rest.len - 1];
510 var depth: isize = 0;
511 var comma: ?usize = null;
512 for (inside, 0..) |c, i| {
513 if (c == '(') {
514 depth += 1;
515 } else if (c == ')') {
516 depth -= 1;
517 } else if (c == ',' and depth == 0) {
518 comma = i;
519 break;
520 }
521 }
522 const at = comma orelse return self.fail("invalid syntax in conditional", .{});
523 const left = trim(try self.expand(inside[0..at]));
524 const right = trim(try self.expand(inside[at + 1 ..]));
525 value = equal(left, right);
526 if (equal(first, "ifneq")) value = !value;
527 }
528 }
529 try stack.append(self.gpa, .{ .outer = outer, .value = value, .number = lineNumber });
530 continue;
531 }
532 if (text.len == 0 or !active(stack.items)) continue;
533 const eq = try self.outside(text, '=');
534 const colon = try self.outside(text, ':');
535 if (eq != null and (colon == null or colon.? + 1 >= eq.?)) {
536 const at = eq.?;
537 var begin = at;
538 var op: u8 = '=';
539 if (at > 0 and std.mem.findScalar(u8, ":+?", text[at - 1]) != null) {
540 begin = at - 1;
541 op = text[at - 1];
542 }
543 const name = trim(text[0..begin]);
544 var value = trim(text[at + 1 ..]);
545 if (name.len == 0) return self.fail("empty variable name", .{});
546 const old = self.variables.get(name);
547 switch (op) {
548 '=' => try self.variables.put(self.gpa, name, .{ .value = value }),
549 ':' => try self.variables.put(self.gpa, name, .{ .simple = true, .value = try self.expand(value) }),
550 '?' => if (old == null) {
551 try self.variables.put(self.gpa, name, .{ .value = value });
552 },
553 '+' => {
554 if (old) |v| {
555 if (v.simple) value = try self.expand(value);
556 if (v.value.len > 0) value = try std.fmt.allocPrint(self.gpa, "{s} {s}", .{ v.value, value });
557 try self.variables.put(self.gpa, name, .{ .simple = v.simple, .value = value });
558 } else {
559 try self.variables.put(self.gpa, name, .{ .value = value });
560 }
561 },
562 else => unreachable,
563 }
564 current = null;
565 continue;
566 }
567 const at = colon orelse {
568 if (trim(try self.expand(text)).len > 0) return self.fail("missing separator", .{});
569 continue;
570 };
571 const targets = try unique(self.gpa, try words(self.gpa, try self.expand(text[0..at])));
572 var remainder = text[at + 1 ..];
573 var inlineRecipe: ?Text = null;
574 var targetPattern: ?Text = null;
575 if (try self.outside(remainder, ';')) |semi| {
576 inlineRecipe = std.mem.trimStart(u8, remainder[semi + 1 ..], " ");
577 remainder = remainder[0..semi];
578 }
579 if (try self.outside(remainder, ':')) |splitAt| {
580 targetPattern = trim(try self.expand(remainder[0..splitAt]));
581 remainder = remainder[splitAt + 1 ..];
582 }
583 const prerequisites = try words(self.gpa, try self.expand(remainder));
584 const group = try self.gpa.create(Group);
585 group.* = .{};
586 current = group;
587 if (targets.len == 1 and equal(targets[0], ".PHONY")) {
588 for (prerequisites) |p| try self.phony.put(self.gpa, p, {});
589 } else if (targetPattern) |pattern| {
590 if (std.mem.findScalar(u8, pattern, '%') == null) return self.fail("target pattern contains no '%'", .{});
591 for (targets) |target| {
592 const stem = match(pattern, target);
593 if (stem == null or stem.?.len == 0) {
594 try self.out.line("Makefile:{d}: target '{s}' doesn't match the target pattern", .{ lineNumber, target });
595 continue;
596 }
597 const list = try self.gpa.alloc(Text, prerequisites.len);
598 for (prerequisites, 0..) |p, i| list[i] = try substitute(self.gpa, p, stem.?);
599 const r = try self.register(target, list);
600 r.stem = stem.?;
601 try group.targets.append(self.gpa, r);
602 }
603 } else {
604 var count: usize = 0;
605 for (targets) |target| {
606 if (std.mem.findScalar(u8, target, '%') != null) count += 1;
607 }
608 if (count > 0 and count != targets.len) return self.fail("mixed implicit and normal rules", .{});
609 group.explicit = count == 0;
610 for (targets) |target| {
611 const r = if (count > 0) blk: {
612 const created = try self.gpa.create(Rule);
613 created.* = .{ .target = target, .prerequisites = prerequisites };
614 try self.patterns.append(self.gpa, created);
615 break :blk created;
616 } else try self.register(target, prerequisites);
617 try group.targets.append(self.gpa, r);
618 }
619 }
620 if (inlineRecipe) |command| try self.addRecipe(group, .{ .number = lineNumber, .text = command });
621 }
622 if (stack.items.len > 0) {
623 self.location = try std.fmt.allocPrint(self.gpa, "Makefile:{d}", .{stack.items[stack.items.len - 1].number});
624 return self.fail("missing 'endif'", .{});
625 }
626 }
627 fn determine(self: *Simulator, target: Text) Error!Chosen {
628 const explicit = self.rules.get(target);
629 if (explicit) |r| {
630 if (r.recipe != null) return .{ .prerequisites = r.prerequisites, .recipe = r.recipe, .stem = r.stem };
631 }
632 if (!self.phony.contains(target)) {
633 var best: ?*Rule = null;
634 var bestStem: Text = "";
635 for (self.patterns.items) |r| {
636 if (r.recipe == null) continue;
637 const stem = match(r.target, target) orelse continue;
638 if (stem.len == 0) continue;
639 var usable = true;
640 for (r.prerequisites) |p| {
641 const name = try substitute(self.gpa, p, stem);
642 if (!self.files.contains(name) and !self.rules.contains(name)) {
643 usable = false;
644 break;
645 }
646 }
647 if (usable and (best == null or stem.len < bestStem.len)) {
648 best = r;
649 bestStem = stem;
650 }
651 }
652 if (best) |r| {
653 var list: std.ArrayList(Text) = .empty;
654 for (r.prerequisites) |p| try list.append(self.gpa, try substitute(self.gpa, p, bestStem));
655 if (explicit) |e| try list.appendSlice(self.gpa, e.prerequisites);
656 return .{ .prerequisites = try unique(self.gpa, list.items), .recipe = r.recipe, .stem = bestStem };
657 }
658 }
659 if (explicit) |r| return .{ .prerequisites = r.prerequisites, .stem = r.stem };
660 if (self.phony.contains(target)) return .{};
661 return if (self.files.contains(target)) .{ .source = true } else .{ .missing = true };
662 }
663 fn touch(self: *Simulator, name: Text) Error!void {
664 self.clock += 1;
665 try self.files.put(self.gpa, name, self.clock);
666 }
667 fn execute(self: *Simulator, text: Text) Error!usize {
668 const list = try words(self.gpa, text);
669 if (equal(list[0], "echo")) {
670 try self.out.line("{s}", .{try join(self.gpa, list[1..], " ")});
671 } else if (equal(list[0], "touch")) {
672 for (list[1..]) |name| try self.touch(name);
673 } else if (equal(list[0], "rm")) {
674 const forced = list.len > 1 and equal(list[1], "-f");
675 var code: usize = 0;
676 for (list[if (forced) 2 else 1..]) |pattern| {
677 var names = try self.glob(pattern);
678 if (names.len == 0) {
679 names = try self.gpa.alloc(Text, 1);
680 names[0] = pattern;
681 }
682 for (names) |name| if (!self.files.remove(name) and !forced) {
683 try self.out.line("rm: cannot remove '{s}': No such file or directory", .{name});
684 code = 1;
685 };
686 }
687 return code;
688 } else if (equal(list[0], "false")) {
689 return 1;
690 } else if (equal(list[0], "exit") and list.len > 1) {
691 return number(list[1]);
692 }
693 return 0;
694 }
695 fn display(self: *Simulator, r: *Rule) Error!void {
696 try self.out.print("{s}:", .{r.target});
697 for (r.prerequisites) |p| try self.out.print(" {s}", .{p});
698 try self.out.add("\n");
699 if (r.recipe) |recipe| for (recipe.items) |line| {
700 try self.out.line(">{s}", .{line.text});
701 };
702 }
703 fn make(self: *Simulator, args: []const Text) Error!void {
704 var options: [256]bool = @splat(false);
705 var goals: std.ArrayList(Text) = .empty;
706 for (args) |arg| {
707 if (arg.len > 0 and arg[0] == '-') {
708 for (arg[1..]) |c| {
709 if (std.mem.findScalar(u8, "nBkis", c) == null) {
710 try self.out.line("make: invalid option -- '{c}'", .{c});
711 return;
712 }
713 options[c] = true;
714 }
715 } else try goals.append(self.gpa, arg);
716 }
717 if (goals.items.len == 0) {
718 if (!self.loaded) {
719 try self.out.line("make: *** No targets specified and no makefile found. Stop.", .{});
720 return;
721 }
722 const goal = self.goal orelse {
723 try self.out.line("make: *** No targets. Stop.", .{});
724 return;
725 };
726 try goals.append(self.gpa, goal);
727 }
728 var saved: std.StringHashMapUnmanaged(usize) = .empty;
729 const clock = self.clock;
730 if (options['n']) {
731 var iterator = self.files.iterator();
732 while (iterator.next()) |entry| try saved.put(self.gpa, entry.key_ptr.*, entry.value_ptr.*);
733 }
734 defer {
735 if (options['n']) {
736 self.files = saved;
737 self.clock = clock;
738 }
739 }
740 var build: Build = .{ .sim = self, .options = options };
741 for (goals.items) |goal| {
742 const before = build.runs;
743 const result = build.target(goal, null) catch |err| {
744 if (err == error.Abort) return;
745 return err;
746 };
747 if (result == 0 and build.runs == before) {
748 if (self.phony.contains(goal) or build.chosen.get(goal).?.recipe == null) {
749 try self.out.line("make: Nothing to be done for '{s}'.", .{goal});
750 } else {
751 try self.out.line("make: '{s}' is up to date.", .{goal});
752 }
753 } else if (result == 2) {
754 try self.out.line("make: Target '{s}' not remade because of errors.", .{goal});
755 }
756 }
757 }
758 fn command(self: *Simulator, line: Text, body: []const Text) Error!void {
759 self.resetExpansion();
760 try self.out.line("$ {s}", .{line});
761 const at = std.mem.findScalar(u8, line, ' ') orelse line.len;
762 const name = line[0..at];
763 const text = if (at < line.len) line[at + 1 ..] else "";
764 if (equal(name, "MAKEFILE")) {
765 try self.load(body);
766 } else if (equal(name, "MAKE")) {
767 try self.make(try words(self.gpa, text));
768 } else if (equal(name, "TOUCH")) {
769 for (try words(self.gpa, text)) |n| try self.touch(n);
770 } else if (equal(name, "RM")) {
771 for (try words(self.gpa, text)) |n| if (!self.files.remove(n)) {
772 try self.out.line("rm: cannot remove '{s}': No such file or directory", .{n});
773 };
774 } else if (equal(name, "FILES")) {
775 try self.out.line("clock {d}", .{self.clock});
776 for (try keys(self.gpa, self.files)) |n| try self.out.line("{s} {d}", .{ n, self.files.get(n).? });
777 } else if (equal(name, "VARS")) {
778 for (try keys(self.gpa, self.variables)) |n| {
779 const value = trim(try self.reference(n));
780 try self.out.print("{s} {s}", .{ n, if (self.variables.get(n).?.simple) ":=" else "=" });
781 if (value.len > 0) try self.out.print(" {s}", .{value});
782 try self.out.add("\n");
783 }
784 } else if (equal(name, "RULES")) {
785 for (try keys(self.gpa, self.rules)) |n| try self.display(self.rules.get(n).?);
786 for (self.patterns.items) |r| if (r.recipe != null) {
787 try self.display(r);
788 };
789 if (self.phony.count() > 0) try self.out.line(".PHONY: {s}", .{try join(self.gpa, try keys(self.gpa, self.phony), " ")});
790 } else if (equal(name, "EXPAND")) {
791 const value = try self.expand(text);
792 try self.out.line("[{s}]", .{value});
793 }
794 }
795};
796const Build = struct {
797 sim: *Simulator,
798 options: [256]bool,
799 results: std.StringHashMapUnmanaged(u8) = .empty,
800 chosen: std.StringHashMapUnmanaged(Chosen) = .empty,
801 building: std.StringHashMapUnmanaged(void) = .empty,
802 runs: usize = 0,
803 fn target(self: *Build, name: Text, parent: ?Text) Error!u8 {
804 const m = self.sim;
805 const gpa = m.gpa;
806 if (self.results.get(name)) |result| return result;
807 const r = try m.determine(name);
808 try self.chosen.put(gpa, name, r);
809 if (r.missing) {
810 try m.out.print("make: *** No rule to make target '{s}'", .{name});
811 if (parent) |p| try m.out.print(", needed by '{s}'", .{p});
812 try m.out.add(if (self.options['k']) ".\n" else ". Stop.\n");
813 try self.results.put(gpa, name, 1);
814 if (!self.options['k']) return error.Abort;
815 return 1;
816 }
817 if (r.source) {
818 try self.results.put(gpa, name, 0);
819 return 0;
820 }
821 try self.building.put(gpa, name, {});
822 var failed = false;
823 var prerequisites: std.ArrayList(Text) = .empty;
824 for (r.prerequisites) |p| {
825 if (self.building.contains(p)) {
826 try m.out.line("make: Circular {s} <- {s} dependency dropped.", .{ name, p });
827 continue;
828 }
829 try prerequisites.append(gpa, p);
830 if (try self.target(p, name) != 0) failed = true;
831 }
832 _ = self.building.remove(name);
833 if (failed) {
834 try self.results.put(gpa, name, 2);
835 return 2;
836 }
837 const time = if (m.phony.contains(name)) null else m.files.get(name);
838 var newer: std.ArrayList(Text) = .empty;
839 for (prerequisites.items) |p| {
840 const pt = if (m.phony.contains(p)) null else m.files.get(p);
841 if (self.options['B'] or time == null or pt == null or pt.? > time.?) try newer.append(gpa, p);
842 }
843 if (r.recipe == null or !self.options['B'] and time != null and newer.items.len == 0) {
844 try self.results.put(gpa, name, 0);
845 return 0;
846 }
847 self.runs += 1;
848 try m.automatic.put(gpa, "@", name);
849 try m.automatic.put(gpa, "<", if (prerequisites.items.len > 0) prerequisites.items[0] else "");
850 try m.automatic.put(gpa, "^", try join(gpa, prerequisites.items, " "));
851 try m.automatic.put(gpa, "?", try join(gpa, newer.items, " "));
852 try m.automatic.put(gpa, "*", r.stem);
853 const recipe = r.recipe.?;
854 const expanded = try gpa.alloc(Text, recipe.items.len);
855 for (recipe.items, 0..) |line, i| {
856 m.location = try std.fmt.allocPrint(gpa, "Makefile:{d}", .{line.number});
857 expanded[i] = try m.expand(line.text);
858 }
859 m.automatic = .empty;
860 for (expanded, 0..) |text, i| {
861 var pos: usize = 0;
862 while (pos < text.len and std.mem.findScalar(u8, " @-", text[pos]) != null) : (pos += 1) {}
863 const prefix = text[0..pos];
864 const command = std.mem.trimEnd(u8, text[pos..], " ");
865 if (command.len == 0) continue;
866 if (self.options['n'] or std.mem.findScalar(u8, prefix, '@') == null and !self.options['s']) try m.out.line("{s}", .{command});
867 if (self.options['n']) continue;
868 const code = try m.execute(command);
869 if (code == 0) continue;
870 if (std.mem.findScalar(u8, prefix, '-') != null or self.options['i']) {
871 try m.out.line("make: [Makefile:{d}: {s}] Error {d} (ignored)", .{ recipe.items[i].number, name, code });
872 } else {
873 try m.out.line("make: *** [Makefile:{d}: {s}] Error {d}", .{ recipe.items[i].number, name, code });
874 try self.results.put(gpa, name, 1);
875 if (!self.options['k']) return error.Abort;
876 return 1;
877 }
878 }
879 if (!m.phony.contains(name)) try m.touch(name);
880 try self.results.put(gpa, name, 0);
881 return 0;
882 }
883};
884pub fn main(init: std.process.Init) !void {
885 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
886 defer arena.deinit();
887 const gpa = arena.allocator();
888 var inputBuffer: [8192]u8 = undefined;
889 var input = std.Io.File.stdin().readerStreaming(init.io, &inputBuffer);
890 const text = try input.interface.allocRemaining(gpa, .unlimited);
891 var lines = std.mem.splitScalar(u8, text, '\n');
892 _ = lines.next();
893 var out: Builder = .{ .gpa = gpa };
894 var sim: Simulator = .{ .gpa = gpa, .out = &out };
895 while (lines.next()) |raw| {
896 const line = std.mem.trimEnd(u8, raw, "\r");
897 if (equal(line, "END FILES")) break;
898 const at = std.mem.findScalar(u8, line, ' ').?;
899 const time = number(line[at + 1 ..]);
900 try sim.files.put(gpa, line[0..at], time);
901 sim.clock = @max(sim.clock, time);
902 }
903 while (lines.next()) |raw| {
904 const line = std.mem.trimEnd(u8, raw, "\r");
905 if (equal(line, "QUIT")) break;
906 var body: std.ArrayList(Text) = .empty;
907 if (equal(line, "MAKEFILE")) {
908 while (lines.next()) |next| {
909 const item = std.mem.trimEnd(u8, next, "\r");
910 if (equal(item, "END MAKEFILE")) break;
911 try body.append(gpa, item);
912 }
913 }
914 sim.command(line, body.items) catch |err| {
915 if (err != error.Invalid) return err;
916 try out.line("{s}: *** {s}. Stop.", .{ failureLocation, failure });
917 };
918 }
919 var outputBuffer: [8192]u8 = undefined;
920 var output = std.Io.File.stdout().writerStreaming(init.io, &outputBuffer);
921 try output.interface.writeAll(out.bytes.items);
922 try output.interface.flush();
923}
