78function leftTrim(text: string): string {
79 let i = 0;
80 while (text[i] === " ") i++;
81 return text.slice(i);
82}
83function rightTrim(text: string): string {
84 let i = text.length;
85 while (i > 0 && text[i - 1] === " ") i--;
86 return text.slice(0, i);

88function trim(text: string): string {

100function substitute(text: string, stem: string): string {
101 const i = text.indexOf("%");
102 return i < 0 ? text : text.slice(0, i) + stem + text.slice(i + 1);
103}
104function patternStem(pattern: string, word: string): string | null {
105 const i = pattern.indexOf("%");
106 if (i < 0) return pattern === word ? "" : null;
107 const before = pattern.slice(0, i),
108 after = pattern.slice(i + 1);
109 return word.length >= before.length + after.length &&
110 word.startsWith(before) &&
111 word.endsWith(after)
112 ? word.slice(before.length, word.length - after.length)
113 : null;
114}
115function replaceWord(word: string, pattern: string, replacement: string): string {
116 const stem = patternStem(pattern, word);
117 return stem === null
118 ? word
119 : pattern.includes("%")
120 ? substitute(replacement, stem)
121 : replacement;
122}
123function wildcardMatch(pattern: string, name: string): boolean {
124 let previous = Array<boolean>(name.length + 1).fill(false);
125 previous[0] = true;
126 for (const c of pattern) {
127 const next = Array<boolean>(name.length + 1).fill(false);
128 for (let j = 0; j <= name.length; j++)
129 next[j] =
130 c === "*"
131 ? previous[j] || (j > 0 && name[j - 1] !== "/" && next[j - 1])
132 : j > 0 &&
133 previous[j - 1] &&
134 (c === "?" ? name[j - 1] !== "/" : c === name[j - 1]);
135 previous = next;
136 }
137 return previous[name.length];
138}

169 closing(text: string, dollar: number): number {
170 const open = text[dollar + 1],
171 close = open === "(" ? ")" : "}";
172 let depth = 0;
173 for (let i = dollar + 1; i < text.length; i++) {
174 if (text[i] === open) depth++;
175 if (text[i] === close && --depth === 0) return i;
176 }
177 this.fail("unterminated variable reference");

179 expand(text: string): string {
180 let result = "";
181 for (let i = 0; i < text.length;) {
182 if (text[i] !== "$") {
183 result += text[i++];
184 continue;
185 }
186 if (i + 1 === text.length) break;
187 const next = text[i + 1];
188 if (next === "$") {
189 result += "$";
190 i += 2;
191 } else if (next === "(" || next === "{") {
192 const end = this.closing(text, i);
193 result += this.reference(text.slice(i + 2, end), next);
194 i = end + 1;
195 } else {
196 result += this.value(next);
197 i += 2;
198 }
199 }
200 return result;

202 reference(content: string, open: string): string {
203 const space = content.indexOf(" "),
204 name = content.slice(0, space);
205 if (space >= 0 && limits.has(name))
206 return this.callFunction(name, leftTrim(content.slice(space + 1)), open);
207 const expanded = this.expand(content),
208 colon = expanded.indexOf(":"),
209 equal = expanded.indexOf("=", colon + 1);
210 if (colon >= 0 && equal >= 0) {
211 let pattern = expanded.slice(colon + 1, equal),
212 replacement = expanded.slice(equal + 1);
213 if (!pattern.includes("%")) {
214 pattern = "%" + pattern;
215 replacement = "%" + replacement;
216 }

218 .map((word) => replaceWord(word, pattern, replacement))

223 value(name: string): string {

242 callFunction(name: string, text: string, open: string): string {
243 const [minimum, maximum] = limits.get(name)!,
244 args: string[] = [];
245 const close = open === "(" ? ")" : "}";
246 let depth = 0,
247 start = 0;
248 for (let i = 0; i < text.length; i++) {
249 if (text[i] === open) depth++;
250 else if (text[i] === close) depth--;
251 else if (text[i] === "," && depth === 0 && args.length < maximum - 1) {
252 args.push(text.slice(start, i));
253 start = i + 1;
254 }
255 }
256 args.push(text.slice(start));
257 if (args.length < minimum)
258 this.fail(`insufficient number of arguments (${args.length}) to function '${name}'`);
259 if (name === "foreach") {
260 const variable = trim(this.expand(args[0])),
261 list = words(this.expand(args[1])),
262 result: string[] = [];
263 for (const word of list) {
264 this.scopes.push({ kind: "foreach", name: variable, word });
265 try {
266 result.push(this.expand(args[2]));
267 } finally {
268 this.scopes.pop();
269 }
270 }
271 return result.join(" ");
272 }
273 if (name === "if") return this.expand(args[trim(this.expand(args[0])) ? 1 : 2] ?? "");

275 if (name === "subst")
276 return expanded[0] === ""
277 ? expanded[2]
278 : expanded[2].split(expanded[0]).join(expanded[1]);
279 if (name === "call") {
280 const variableName = trim(expanded[0]),
281 variable = this.variables.get(variableName);
282 if (variable === undefined) return "";

288 this.scopes.pop();

292 a = values[0],
293 b = values[1],
294 c = values[2];
295 switch (name) {
296 case "patsubst":

298 .map((word) => replaceWord(word, a, b))
299 .join(" ");
300 case "strip":
301 return words(a).join(" ");
302 case "findstring":
303 return b.includes(a) ? a : "";
304 case "filter":
305 case "filter-out":
306 return words(b)
307 .filter(
308 (word) =>
309 words(a).some((pattern) => patternStem(pattern, word) !== null) ===
310 (name === "filter"),
311 )
312 .join(" ");
313 case "sort":
314 return unique(words(a)).sort().join(" ");
315 case "word":
316 if (!numeric(a)) this.fail(`non-numeric first argument to 'word' function: '${a}'`);
317 if (Number(a) === 0)
318 this.fail("first argument to 'word' function must be greater than 0");
319 return words(b)[Number(a) - 1] ?? "";
320 case "wordlist":
321 if (!numeric(a))
322 this.fail(`non-numeric first argument to 'wordlist' function: '${a}'`);
323 if (!numeric(b))
324 this.fail(`non-numeric second argument to 'wordlist' function: '${b}'`);
325 if (Number(a) === 0)
326 this.fail(`invalid first argument to 'wordlist' function: '${a}'`);
327 return words(c)
328 .slice(Number(a) - 1, Number(b))
329 .join(" ");
330 case "words":
331 return String(words(a).length);
332 case "firstword":
333 return words(a)[0] ?? "";
334 case "lastword":
335 return words(a).at(-1) ?? "";
336 case "dir":
337 return words(a)
338 .map((word) =>
339 word.includes("/") ? word.slice(0, word.lastIndexOf("/") + 1) : "./",
340 )
341 .join(" ");
342 case "notdir":
343 return words(a)
344 .map((word) => word.slice(word.lastIndexOf("/") + 1))
345 .join(" ");
346 case "suffix":
347 return words(a)
348 .filter((word) => word.lastIndexOf(".") > word.lastIndexOf("/"))
349 .map((word) => word.slice(word.lastIndexOf(".")))
350 .join(" ");
351 case "basename":
352 return words(a)
353 .map((word) =>
354 word.lastIndexOf(".") > word.lastIndexOf("/")
355 ? word.slice(0, word.lastIndexOf("."))
356 : word,
357 )
358 .join(" ");
359 case "addsuffix":
360 return words(b)
361 .map((word) => word + a)
362 .join(" ");
363 case "addprefix":
364 return words(b)
365 .map((word) => a + word)
366 .join(" ");
367 case "wildcard":
368 return words(a)
369 .flatMap((pattern) => this.wildcard(pattern))
370 .join(" ");
371 case "value":
372 return this.variables.get(a)?.value ?? "";
373 case "info":
374 output.push(a);
375 return "";
376 case "warning":
377 output.push(this.location + ": " + a);
378 return "";
379 case "error":
380 this.fail(a);
381 }
382 return "";

384 punctuation(text: string): Map<string, number> {
385 const result = new Map<string, number>();
386 for (let i = 0; i < text.length; i++) {
387 if (text[i] === "$") {
388 if (text[i + 1] === "(" || text[i + 1] === "{") i = this.closing(text, i);
389 else if (i + 1 < text.length) i++;
390 } else if ("=:;".includes(text[i]) && !result.has(text[i])) result.set(text[i], i);
391 }
392 return result;

418 load(body: string[]): void {
419 this.resetMakefile();
420 const conditions: Condition[] = [];
421 let current: CurrentRule | null = null;
422 const active = (): boolean =>
423 !conditions.length || (conditions.at(-1)!.outer && conditions.at(-1)!.value);
424 try {
425 for (let i = 0; i < body.length; i++) {
426 const line = i + 1;
427 this.location = "Makefile:" + line;
428 if (body[i].startsWith(">")) {
429 if (!active()) continue;
430 if (current === null) this.fail("recipe commences before first target");
431 this.addRecipe(current, line, leftTrim(body[i].slice(1)));
432 continue;
433 }
434 let logical = body[i];
435 while (logical.endsWith("\\") && i + 1 < body.length)
436 logical = rightTrim(logical.slice(0, -1)) + " " + leftTrim(body[++i]);
437 const text = trim(logical.split("#")[0]),
438 space = text.indexOf(" "),
439 first = space < 0 ? text : text.slice(0, space),
440 rest = space < 0 ? "" : trim(text.slice(space + 1));
441 if (["ifeq", "ifneq", "ifdef", "ifndef"].includes(first)) {
442 const outer = active();
443 let value = false;
444 if (outer) {
445 if (first === "ifdef" || first === "ifndef") {
446 const variable = this.variables.get(trim(this.expand(rest)));
447 value = variable !== undefined && variable.value !== "";
448 if (first === "ifndef") value = !value;
449 } else {
450 if (rest.length < 2 || !rest.startsWith("(") || !rest.endsWith(")"))
451 this.fail("invalid syntax in conditional");
452 const inside = rest.slice(1, -1);
453 let depth = 0,
454 comma = -1;
455 for (let j = 0; j < inside.length; j++) {
456 if (inside[j] === "(") depth++;
457 else if (inside[j] === ")") depth--;
458 else if (inside[j] === "," && depth === 0) {
459 comma = j;
460 break;
461 }
462 }
463 if (comma < 0) this.fail("invalid syntax in conditional");
464 const left = trim(this.expand(inside.slice(0, comma))),
465 right = trim(this.expand(inside.slice(comma + 1)));
466 value = left === right;
467 if (first === "ifneq") value = !value;
468 }
469 }
470 conditions.push({ outer, value, alternate: false, line });
471 continue;
472 }
473 if (first === "else") {
474 if (!conditions.length) this.fail("extraneous 'else'");
475 const condition = conditions.at(-1)!;
476 if (condition.alternate) this.fail("only one 'else' per conditional");
477 condition.value = !condition.value;
478 condition.alternate = true;
479 continue;
480 }
481 if (first === "endif") {
482 if (!conditions.length) this.fail("extraneous 'endif'");
483 conditions.pop();
484 continue;
485 }
486 if (text === "" || !active()) continue;
487 const positions = this.punctuation(text),
488 equal = positions.get("="),
489 colon = positions.get(":");
490 if (equal !== undefined && (colon === undefined || colon >= equal - 1)) {
491 const special = equal > 0 && ":+?".includes(text[equal - 1]),
492 op = special ? text[equal - 1] + "=" : "=";
493 const name = trim(text.slice(0, equal - (special ? 1 : 0))),
494 rhs = trim(text.slice(equal + 1));
495 if (!name) this.fail("empty variable name");
496 const previous = this.variables.get(name);
497 if (
498 op === "=" ||
499 (op === "?=" && previous === undefined) ||
500 (op === "+=" && previous === undefined)
501 )
502 this.variables.set(name, { kind: "=", value: rhs });
503 else if (op === ":=")
504 this.variables.set(name, { kind: ":=", value: this.expand(rhs) });
505 else if (op === "+=") {
506 const value = previous!.kind === ":=" ? this.expand(rhs) : rhs;
507 previous!.value =
508 previous!.value === "" ? value : previous!.value + " " + value;
509 }
510 current = null;
511 } else if (colon !== undefined) {
512 const targets = unique(words(this.expand(text.slice(0, colon))));

514 inline: string | null = null,
515 targetPattern: string | null = null;
516 const semicolon = this.punctuation(rhs).get(";");
517 if (semicolon !== undefined) {
518 inline = leftTrim(rhs.slice(semicolon + 1));
519 rhs = rhs.slice(0, semicolon);
520 }
521 const secondColon = this.punctuation(rhs).get(":");
522 if (secondColon !== undefined) {
523 targetPattern = trim(this.expand(rhs.slice(0, secondColon)));
524 rhs = rhs.slice(secondColon + 1);
525 }
526 const prerequisites = words(this.expand(rhs));
527 current = { entries: [], explicit: true, recipe: null };
528 if (targets.length === 1 && targets[0] === ".PHONY")
529 for (const name of prerequisites) this.phony.add(name);
530 else if (targetPattern !== null) {
531 if (!targetPattern.includes("%"))
532 this.fail("target pattern contains no '%'");
533 for (const name of targets) {
534 const stem = patternStem(targetPattern, name);
535 if (stem === null || stem === "")
536 output.push(
537 `Makefile:${line}: target '${name}' doesn't match the target pattern`,
538 );
539 else {
540 const entry = this.register(
541 name,
542 prerequisites.map((value) => substitute(value, stem)),
543 );
544 entry.stem = stem;
545 current.entries.push(entry);
546 }
547 }
548 } else if (targets.some((name) => name.includes("%"))) {
549 if (targets.some((name) => !name.includes("%")))
550 this.fail("mixed implicit and normal rules");
551 current.explicit = false;
552 for (const name of targets) {
553 const entry: Rule = {
554 name,
555 prerequisites: [...prerequisites],
556 recipe: null,
557 stem: "",
558 };
559 this.patterns.push(entry);
560 current.entries.push(entry);
561 }
562 } else
563 for (const name of targets)
564 current.entries.push(this.register(name, prerequisites));
565 if (inline !== null) this.addRecipe(current, line, inline);
566 } else if (trim(this.expand(text)) !== "") this.fail("missing separator");
567 }
568 if (conditions.length) {
569 this.location = "Makefile:" + conditions.at(-1)!.line;
570 this.fail("missing 'endif'");
571 }
572 this.loaded = true;
573 output.push(
574 `Makefile loaded: lines=${body.length} variables=${this.variables.size} targets=${this.explicit.size} patterns=${this.patterns.filter((rule) => rule.recipe !== null).length} goal=${this.goal ?? "(none)"}`,
575 );
576 } catch (error) {
577 if (!(error instanceof MakeError)) throw error;
578 output.push(error.message);
579 this.resetMakefile();
580 } finally {
581 this.location = "make";
582 }

584 determine(name: string): Determined {
585 const explicit = this.explicit.get(name);
586 if (explicit !== undefined && explicit.recipe !== null)
587 return { ...explicit, prerequisites: [...explicit.prerequisites], kind: "rule" };
588 if (!this.phony.has(name)) {
589 const candidates = this.patterns
590 .map((rule, index) => ({ rule, index, stem: patternStem(rule.name, name) }))
591 .filter(
592 (item): item is PatternCandidate =>
593 item.rule.recipe !== null && item.stem !== null && item.stem !== "",
594 )
595 .sort((a, b) => a.stem.length - b.stem.length || a.index - b.index);
596 for (const item of candidates) {
597 const prerequisites = item.rule.prerequisites.map((value) =>
598 substitute(value, item.stem),
599 );
600 if (
601 prerequisites.every(
602 (value) => this.files.has(value) || this.explicit.has(value),
603 )
604 )
605 return {
606 name,
607 prerequisites: unique([
608 ...prerequisites,
609 ...(explicit?.prerequisites ?? []),
610 ]),
611 recipe: item.rule.recipe,
612 stem: item.stem,
613 kind: "rule",
614 };
615 }
616 }
617 if (explicit !== undefined || this.phony.has(name))
618 return {
619 name,
620 prerequisites: [...(explicit?.prerequisites ?? [])],
621 recipe: null,
622 stem: explicit?.stem ?? "",
623 kind: "rule",
624 };
625 return {
626 name,
627 prerequisites: [],
628 recipe: null,
629 stem: "",
630 kind: this.files.has(name) ? "source" : "missing",
631 };

633 executeRecipeCommand(command: string): number {
634 const list = words(command),
635 args = list.slice(1);
636 if (list[0] === "echo") output.push(args.join(" "));
637 else if (list[0] === "touch") for (const name of args) this.touch(name);
638 else if (list[0] === "rm") {
639 const forced = args[0] === "-f";
640 let status = 0;
641 for (const pattern of forced ? args.slice(1) : args) {
642 const matches = this.wildcard(pattern);
643 for (const name of matches.length ? matches : [pattern]) {
644 if (!this.files.delete(name) && !forced) {
645 output.push(`rm: cannot remove '${name}': No such file or directory`);
646 status = 1;
647 }
648 }
649 }

651 } else if (list[0] === "false") return 1;
652 else if (list[0] === "exit") return Number(args[0] ?? 0);
653 return 0;
