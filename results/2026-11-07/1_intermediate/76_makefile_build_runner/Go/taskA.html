67func unique(list []string) []string {
68 seen := map[string]bool{}
69 result := []string{}
70 for _, item := range list {
71 if !seen[item] {
72 seen[item] = true
73 result = append(result, item)
74 }
75 }
76 return result

86func integer(s string) (int, bool) {
87 if s == "" {
88 return 0, false
89 }
90 for _, c := range s {
91 if c < '0' || c > '9' {
92 return 0, false
93 }
94 }
95 n, _ := strconv.Atoi(s)
96 return n, true
97}
98func match(pattern, word string) (string, bool) {
99 at := strings.IndexByte(pattern, '%')
100 if at < 0 {
101 return "", pattern == word
102 }
103 a, b := pattern[:at], pattern[at+1:]
104 if len(word) < len(a)+len(b) || !strings.HasPrefix(word, a) || !strings.HasSuffix(word, b) {
105 return "", false
106 }
107 return word[len(a) : len(word)-len(b)], true
108}
109func substitute(pattern, stem string) string { return strings.Replace(pattern, "%", stem, 1) }
110func replacement(word, pattern, value string) string {
111 stem, ok := match(pattern, word)
112 if !ok {
113 return word
114 }
115 if strings.Contains(pattern, "%") {
116 return substitute(value, stem)
117 }
118 return value
119}
120func wildcard(pattern, name string) bool {
121 type position struct{ p, n int }
122 memo := map[position]bool{}
123 known := map[position]bool{}
124 var visit func(int, int) bool
125 visit = func(p, n int) bool {
126 key := position{p, n}
127 if known[key] {
128 return memo[key]
129 }
130 result := false
131 if p == len(pattern) {
132 result = n == len(name)
133 } else if pattern[p] == '*' {
134 result = visit(p+1, n) || n < len(name) && name[n] != '/' && visit(p, n+1)
135 } else if n < len(name) && (pattern[p] == name[n] || pattern[p] == '?' && name[n] != '/') {
136 result = visit(p+1, n+1)
137 }
138 known[key], memo[key] = true, result
139 return result
140 }
141 return visit(0, 0)
142}

170func (m *simulator) reference(name string) string {
171 for i := len(m.scopes) - 1; i >= 0; i-- {
172 s := m.scopes[i]
173 if s.call != nil {
174 if n, ok := integer(name); ok {
175 if n < len(s.call) {
176 return s.call[n]
177 }
178 return ""
179 }
180 } else if s.name == name {
181 return s.word
182 }
183 }
184 if value, ok := m.automatic[name]; ok {
185 return value
186 }
187 v, ok := m.variables[name]
188 if !ok {
189 return ""
190 }
191 if v.simple {
192 return v.value
193 }
194 if m.expanding[name] {
195 m.fail("Recursive variable '" + name + "' references itself (eventually)")
196 }
197 m.expanding[name] = true
198 defer delete(m.expanding, name)
199 return m.expand(v.value)
200}
201func (m *simulator) closing(s string, open int) int {
202 close := byte(')')
203 if s[open] == '{' {
204 close = '}'
205 }
206 depth := 0
207 for i := open; i < len(s); i++ {
208 if s[i] == s[open] {
209 depth++
210 } else if s[i] == close {
211 depth--
212 if depth == 0 {
213 return i
214 }
215 }
216 }
217 m.fail("unterminated variable reference")
218 return 0
219}
220func (m *simulator) outside(s string, wanted byte) int {
221 found := -1
222 for i := 0; i < len(s); i++ {
223 if s[i] == '$' {
224 if i+1 < len(s) && (s[i+1] == '(' || s[i+1] == '{') {
225 i = m.closing(s, i+1)
226 } else if i+1 < len(s) {
227 i++
228 }
229 } else if s[i] == wanted && found < 0 {
230 found = i
231 }
232 }
233 return found
234}

236var limits = map[string][2]int{
237 "subst": {3, 3}, "patsubst": {3, 3}, "strip": {1, 1}, "findstring": {2, 2},
238 "filter": {2, 2}, "filter-out": {2, 2}, "sort": {1, 1}, "word": {2, 2},
239 "wordlist": {3, 3}, "words": {1, 1}, "firstword": {1, 1}, "lastword": {1, 1},
240 "dir": {1, 1}, "notdir": {1, 1}, "suffix": {1, 1}, "basename": {1, 1},
241 "addsuffix": {2, 2}, "addprefix": {2, 2}, "wildcard": {1, 1}, "foreach": {3, 3},
242 "if": {2, 3}, "call": {1, 0}, "value": {1, 1}, "info": {1, 1}, "warning": {1, 1}, "error": {1, 1},
243}

245func (m *simulator) expand(s string) string {
246 var result strings.Builder
247 for i := 0; i < len(s); i++ {
248 if s[i] != '$' {
249 result.WriteByte(s[i])
250 continue
251 }
252 i++
253 if i == len(s) {
254 break
255 }
256 if s[i] == '$' {
257 result.WriteByte('$')
258 } else if s[i] == '(' || s[i] == '{' {
259 end := m.closing(s, i)
260 content := s[i+1 : end]
261 at := strings.IndexByte(content, ' ')
262 if at >= 0 {
263 if _, ok := limits[content[:at]]; ok {
264 result.WriteString(m.function(content[:at], strings.TrimLeft(content[at+1:], " "), s[i]))
265 i = end
266 continue
267 }
268 }
269 name := m.expand(content)
270 colon := strings.IndexByte(name, ':')
271 if colon >= 0 && strings.Contains(name[colon+1:], "=") {
272 eq := colon + 1 + strings.IndexByte(name[colon+1:], '=')
273 from, to := name[colon+1:eq], name[eq+1:]
274 if !strings.Contains(from, "%") {
275 from, to = "%"+from, "%"+to
276 }
277 list := words(m.reference(name[:colon]))
278 for n, word := range list {
279 list[n] = replacement(word, from, to)
280 }
281 result.WriteString(strings.Join(list, " "))
282 } else {
283 result.WriteString(m.reference(name))
284 }
285 i = end
286 } else {
287 result.WriteString(m.reference(s[i : i+1]))
288 }
289 }
290 return result.String()
291}
292func (m *simulator) function(name, text string, open byte) string {
293 bound := limits[name]
294 args := []string{}
295 depth, start := 0, 0
296 close := byte(')')
297 if open == '{' {
298 close = '}'
299 }
300 for i := 0; i < len(text); i++ {
301 if text[i] == open {
302 depth++
303 } else if text[i] == close {
304 depth--
305 }
306 if text[i] == ',' && depth == 0 && (bound[1] == 0 || len(args) < bound[1]-1) {
307 args = append(args, text[start:i])
308 start = i + 1
309 }
310 }
311 args = append(args, text[start:])
312 if len(args) < bound[0] {
313 m.fail(fmt.Sprintf("insufficient number of arguments (%d) to function '%s'", len(args), name))
314 }
315 if name == "if" {
316 if trim(m.expand(args[0])) != "" {
317 return m.expand(args[1])
318 }
319 if len(args) > 2 {
320 return m.expand(args[2])
321 }
322 return ""
323 }
324 if name == "foreach" {
325 variableName := trim(m.expand(args[0]))
326 list := words(m.expand(args[1]))
327 result := []string{}
328 for _, word := range list {
329 m.scopes = append(m.scopes, scope{name: variableName, word: word})
330 value := func() string { defer func() { m.scopes = m.scopes[:len(m.scopes)-1] }(); return m.expand(args[2]) }()
331 result = append(result, value)
332 }
333 return strings.Join(result, " ")
334 }
335 for i := range args {
336 args[i] = m.expand(args[i])
337 if name != "subst" && name != "call" {
338 args[i] = trim(args[i])
339 }
340 }
341 if name == "call" {
342 called := trim(args[0])
343 v, ok := m.variables[called]
344 if !ok {
345 return ""
346 }
347 if v.simple {
348 return v.value
349 }
350 args[0] = called
351 m.scopes = append(m.scopes, scope{call: args})
352 defer func() { m.scopes = m.scopes[:len(m.scopes)-1] }()
353 return m.expand(v.value)
354 }
355 switch name {
356 case "subst":
357 if args[0] == "" {
358 return args[2]
359 }
360 return strings.ReplaceAll(args[2], args[0], args[1])
361 case "patsubst":
362 list := words(args[2])
363 for i, word := range list {
364 list[i] = replacement(word, args[0], args[1])
365 }
366 return strings.Join(list, " ")
367 case "strip":
368 return strings.Join(words(args[0]), " ")
369 case "findstring":
370 if strings.Contains(args[1], args[0]) {
371 return args[0]
372 }
373 return ""
374 case "filter", "filter-out":
375 result := []string{}
376 for _, word := range words(args[1]) {
377 matched := false
378 for _, pattern := range words(args[0]) {
379 if _, ok := match(pattern, word); ok {
380 matched = true
381 break
382 }
383 }
384 if matched == (name == "filter") {
385 result = append(result, word)
386 }
387 }
388 return strings.Join(result, " ")
389 case "sort":
390 list := unique(words(args[0]))
391 sort.Strings(list)
392 return strings.Join(list, " ")
393 case "word":
394 n, ok := integer(args[0])
395 if !ok {
396 m.fail("non-numeric first argument to 'word' function: '" + args[0] + "'")
397 }
398 if n == 0 {
399 m.fail("first argument to 'word' function must be greater than 0")
400 }
401 list := words(args[1])
402 if n <= len(list) {
403 return list[n-1]
404 }
405 return ""
406 case "wordlist":
407 a, ok := integer(args[0])
408 if !ok {
409 m.fail("non-numeric first argument to 'wordlist' function: '" + args[0] + "'")
410 }
411 b, ok := integer(args[1])
412 if !ok {
413 m.fail("non-numeric second argument to 'wordlist' function: '" + args[1] + "'")
414 }
415 if a == 0 {
416 m.fail("invalid first argument to 'wordlist' function: '" + args[0] + "'")
417 }
418 list := words(args[2])
419 if b > len(list) {
420 b = len(list)
421 }
422 if a > b {
423 return ""
424 }
425 return strings.Join(list[a-1:b], " ")
426 case "words":
427 return strconv.Itoa(len(words(args[0])))
428 case "firstword", "lastword":
429 list := words(args[0])
430 if len(list) == 0 {
431 return ""
432 }
433 if name == "firstword" {
434 return list[0]
435 }
436 return list[len(list)-1]
437 case "dir", "notdir", "suffix", "basename":
438 result := []string{}
439 for _, word := range words(args[0]) {
440 slash, dot := strings.LastIndexByte(word, '/'), strings.LastIndexByte(word, '.')
441 switch name {
442 case "dir":
443 if slash < 0 {
444 result = append(result, "./")
445 } else {
446 result = append(result, word[:slash+1])
447 }
448 case "notdir":
449 result = append(result, word[slash+1:])
450 case "suffix":
451 if dot > slash {
452 result = append(result, word[dot:])
453 }
454 case "basename":
455 if dot > slash {
456 word = word[:dot]
457 }
458 result = append(result, word)
459 }
460 }
461 return strings.Join(result, " ")
462 case "addprefix", "addsuffix":
463 list := words(args[1])
464 for i, word := range list {
465 if name == "addprefix" {
466 list[i] = args[0] + word
467 } else {
468 list[i] = word + args[0]
469 }
470 }
471 return strings.Join(list, " ")
472 case "wildcard":
473 result := []string{}
474 for _, pattern := range words(args[0]) {
475 result = append(result, m.glob(pattern)...)
476 }
477 return strings.Join(result, " ")
478 case "value":
479 return m.variables[args[0]].value
480 case "info":
481 m.emit("%s", args[0])
482 case "warning":
483 m.emit("%s: %s", m.location, args[0])
484 case "error":
485 m.fail(args[0])
486 }
487 return ""
488}
489func (m *simulator) register(target string, prerequisites []string) *rule {
490 r := m.rules[target]
491 if r == nil {
492 r = &rule{target: target}
493 m.rules[target] = r
494 }
495 r.prerequisites = unique(append(r.prerequisites, prerequisites...))
496 if m.goal == "" && !strings.HasPrefix(target, ".") {
497 m.goal = target
498 }
499 return r
500}
501func (m *simulator) addRecipe(g *group, line recipeLine) {
502 if len(g.targets) == 0 {
503 return
504 }
505 if g.recipe == nil {
506 g.recipe = &recipe{}
507 for _, r := range g.targets {
508 if g.explicit && r.recipe != nil {
509 m.emit("Makefile:%d: warning: overriding recipe for target '%s'", line.number, r.target)
510 m.emit("Makefile:%d: warning: ignoring old recipe for target '%s'", r.recipe.lines[0].number, r.target)
511 }
512 r.recipe = g.recipe
513 }
514 }
515 g.recipe.lines = append(g.recipe.lines, line)
516}
517func (m *simulator) load(body []string) {
518 m.clear()
519 defer func() {
520 if caught := recover(); caught != nil {
521 if e, ok := caught.(expansionError); ok {
522 m.emit("%s: *** %s. Stop.", e.location, e.reason)
523 m.clear()
524 } else {
525 panic(caught)
526 }
527 }
528 }()
529 var current *group
530 stack := []condition{}
531 active := func() bool { return len(stack) == 0 || stack[len(stack)-1].outer && stack[len(stack)-1].value }
532 for index := 0; index < len(body); index++ {
533 number, text := index+1, body[index]
534 m.location = fmt.Sprintf("Makefile:%d", number)
535 if strings.HasPrefix(text, ">") {
536 if !active() {
537 continue
538 }
539 if current == nil {
540 m.fail("recipe commences before first target")
541 }
542 m.addRecipe(current, recipeLine{number, strings.TrimLeft(text[1:], " ")})
543 continue
544 }
545 for strings.HasSuffix(text, "\\") && index+1 < len(body) {
546 index++
547 text = strings.TrimRight(text[:len(text)-1], " ") + " " + strings.TrimLeft(body[index], " ")
548 }
549 if hash := strings.IndexByte(text, '#'); hash >= 0 {
550 text = text[:hash]
551 }
552 text = trim(text)
553 first, rest, _ := strings.Cut(text, " ")
554 rest = trim(rest)
555 if first == "else" {
556 if len(stack) == 0 {
557 m.fail("extraneous 'else'")
558 }
559 top := &stack[len(stack)-1]
560 if top.otherwise {
561 m.fail("only one 'else' per conditional")
562 }
563 top.value, top.otherwise = !top.value, true
564 continue
565 }
566 if first == "endif" {
567 if len(stack) == 0 {
568 m.fail("extraneous 'endif'")
569 }
570 stack = stack[:len(stack)-1]
571 continue
572 }
573 if first == "ifdef" || first == "ifndef" || first == "ifeq" || first == "ifneq" {
574 outer, value := active(), false
575 if outer {
576 if first == "ifdef" || first == "ifndef" {
577 v, exists := m.variables[trim(m.expand(rest))]
578 value = exists && v.value != ""
579 if first == "ifndef" {
580 value = !value
581 }
582 } else {
583 if len(rest) < 2 || rest[0] != '(' || rest[len(rest)-1] != ')' {
584 m.fail("invalid syntax in conditional")
585 }
586 inside, depth, comma := rest[1:len(rest)-1], 0, -1
587 for i := 0; i < len(inside); i++ {
588 if inside[i] == '(' {
589 depth++
590 } else if inside[i] == ')' {
591 depth--
592 } else if inside[i] == ',' && depth == 0 {
593 comma = i
594 break
595 }
596 }
597 if comma < 0 {
598 m.fail("invalid syntax in conditional")
599 }
600 left := trim(m.expand(inside[:comma]))
601 right := trim(m.expand(inside[comma+1:]))
602 value = left == right
603 if first == "ifneq" {
604 value = !value
605 }
606 }
607 }
608 stack = append(stack, condition{outer: outer, value: value, number: number})
609 continue
610 }
611 if text == "" || !active() {
612 continue
613 }
614 eq, colon := m.outside(text, '='), m.outside(text, ':')
615 if eq >= 0 && (colon < 0 || colon >= eq-1) {
616 start, op := eq, byte('=')
617 if eq > 0 && strings.ContainsRune(":+?", rune(text[eq-1])) {
618 start, op = eq-1, text[eq-1]
619 }
620 name, value := trim(text[:start]), trim(text[eq+1:])
621 if name == "" {
622 m.fail("empty variable name")
623 }
624 old, exists := m.variables[name]
625 switch op {
626 case '=':
627 m.variables[name] = variable{value: value}
628 case ':':
629 m.variables[name] = variable{simple: true, value: m.expand(value)}
630 case '?':
631 if !exists {
632 m.variables[name] = variable{value: value}
633 }
634 case '+':
635 if exists {
636 if old.simple {
637 value = m.expand(value)
638 }
639 if old.value != "" {
640 value = old.value + " " + value
641 }
642 old.value = value
643 m.variables[name] = old
644 } else {
645 m.variables[name] = variable{value: value}
646 }
647 }
648 current = nil
649 continue
650 }
651 if colon < 0 {
652 if trim(m.expand(text)) != "" {
653 m.fail("missing separator")
654 }
655 continue
656 }
657 targets := unique(words(m.expand(text[:colon])))
658 remainder := text[colon+1:]
659 inline := ""
660 hasInline := false
661 if semi := m.outside(remainder, ';'); semi >= 0 {
662 inline, remainder, hasInline = strings.TrimLeft(remainder[semi+1:], " "), remainder[:semi], true
663 }
664 pattern := ""
665 static := false
666 if at := m.outside(remainder, ':'); at >= 0 {
667 pattern, remainder, static = trim(m.expand(remainder[:at])), remainder[at+1:], true
668 }
669 prerequisites := words(m.expand(remainder))
670 current = &group{explicit: true}
671 if len(targets) == 1 && targets[0] == ".PHONY" {
672 for _, p := range prerequisites {
673 m.phony[p] = true
674 }
675 } else if static {
676 if !strings.Contains(pattern, "%") {
677 m.fail("target pattern contains no '%'")
678 }
679 for _, target := range targets {
680 stem, ok := match(pattern, target)
681 if !ok || stem == "" {
682 m.emit("Makefile:%d: target '%s' doesn't match the target pattern", number, target)
683 continue
684 }
685 list := make([]string, len(prerequisites))
686 for i, p := range prerequisites {
687 list[i] = substitute(p, stem)
688 }
689 r := m.register(target, list)
690 r.stem = stem
691 current.targets = append(current.targets, r)
692 }
693 } else {
694 patterns := 0
695 for _, target := range targets {
696 if strings.Contains(target, "%") {
697 patterns++
698 }
699 }
700 if patterns > 0 && patterns != len(targets) {
701 m.fail("mixed implicit and normal rules")
702 }
703 if patterns > 0 {
704 current.explicit = false
705 for _, target := range targets {
706 r := &rule{target: target, prerequisites: append([]string(nil), prerequisites...)}
707 m.patterns = append(m.patterns, r)
708 current.targets = append(current.targets, r)
709 }
710 } else {
711 for _, target := range targets {
712 current.targets = append(current.targets, m.register(target, prerequisites))
713 }
714 }
715 }
716 if hasInline {
717 m.addRecipe(current, recipeLine{number, inline})
718 }
719 }
720 if len(stack) > 0 {
721 m.location = fmt.Sprintf("Makefile:%d", stack[len(stack)-1].number)
722 m.fail("missing 'endif'")
723 }
724 m.loaded = true
725 count := 0
726 for _, r := range m.patterns {
727 if r.recipe != nil {
728 count++
729 }
730 }
731 goal := m.goal
732 if goal == "" {
733 goal = "(none)"
734 }
735 m.emit("Makefile loaded: lines=%d variables=%d targets=%d patterns=%d goal=%s", len(body), len(m.variables), len(m.rules), count, goal)
736}

740 recipe *recipe

742 source, missing bool

745func (m *simulator) determine(target string) chosenRule {
746 explicit := m.rules[target]
747 if explicit != nil && explicit.recipe != nil {
748 return chosenRule{prerequisites: append([]string(nil), explicit.prerequisites...), recipe: explicit.recipe, stem: explicit.stem}
749 }
750 if !m.phony[target] {
751 type candidate struct {
752 rule *rule
753 stem string
754 }
755 candidates := []candidate{}
756 for _, r := range m.patterns {
757 if r.recipe != nil {
758 if stem, ok := match(r.target, target); ok && stem != "" {
759 candidates = append(candidates, candidate{r, stem})
760 }
761 }
762 }
763 sort.SliceStable(candidates, func(i, j int) bool { return len(candidates[i].stem) < len(candidates[j].stem) })
764 for _, c := range candidates {
765 list, usable := []string{}, true
766 for _, p := range c.rule.prerequisites {
767 p = substitute(p, c.stem)
768 list = append(list, p)
769 if _, exists := m.files[p]; !exists && m.rules[p] == nil {
770 usable = false
771 }
772 }
773 if usable {
774 if explicit != nil {
775 list = append(list, explicit.prerequisites...)
776 }
777 return chosenRule{prerequisites: unique(list), recipe: c.rule.recipe, stem: c.stem}
778 }
779 }
780 }
781 if explicit != nil {
782 return chosenRule{prerequisites: append([]string(nil), explicit.prerequisites...), stem: explicit.stem}
783 }
784 if m.phony[target] {
785 return chosenRule{}
786 }
787 if _, exists := m.files[target]; exists {
788 return chosenRule{source: true}
789 }
790 return chosenRule{missing: true}
791}
792func (m *simulator) touch(name string) { m.clock++; m.files[name] = m.clock }
793func (m *simulator) executeCommand(text string) int {
794 list := words(text)
795 switch list[0] {
796 case "echo":
797 m.emit("%s", strings.Join(list[1:], " "))
798 case "touch":
799 for _, name := range list[1:] {
800 m.touch(name)
801 }
802 case "rm":
803 forced, start, result := false, 1, 0
804 if len(list) > 1 && list[1] == "-f" {
805 forced, start = true, 2
806 }
807 for _, pattern := range list[start:] {
808 names := m.glob(pattern)
809 if len(names) == 0 {
810 names = []string{pattern}
811 }
812 for _, name := range names {
813 if _, exists := m.files[name]; exists {
814 delete(m.files, name)
815 } else if !forced {
816 m.emit("rm: cannot remove '%s': No such file or directory", name)
817 result = 1
818 }
819 }
820 }
821 return result
822 case "false":
823 return 1
824 case "exit":
825 if len(list) > 1 {
826 n, _ := integer(list[1])
827 return n
828 }
829 }
830 return 0
831}
832func (m *simulator) make(args []string) {
833 options := map[byte]bool{}
834 goals := []string{}
835 for _, arg := range args {
836 if strings.HasPrefix(arg, "-") {
837 for i := 1; i < len(arg); i++ {
838 if !strings.ContainsRune("nBkis", rune(arg[i])) {
839 m.emit("make: invalid option -- '%c'", arg[i])
840 return
841 }
842 options[arg[i]] = true
843 }
844 } else {
845 goals = append(goals, arg)
846 }
847 }
848 if len(goals) == 0 {
849 if !m.loaded {
850 m.emit("make: *** No targets specified and no makefile found. Stop.")
851 return
852 }
853 if m.goal == "" {
854 m.emit("make: *** No targets. Stop.")
855 return
856 }
857 goals = []string{m.goal}
858 }
859 if options['n'] {
860 saved := map[string]int{}
861 for name, t := range m.files {
862 saved[name] = t
863 }
864 clock := m.clock
865 defer func() { m.files, m.clock = saved, clock }()
866 }
867 defer func() {
868 if caught := recover(); caught != nil {
869 if _, ok := caught.(abortBuild); !ok {
870 panic(caught)
871 }
872 }
873 }()
874 results := map[string]int{}
875 selected := map[string]chosenRule{}
876 building := map[string]bool{}
877 runs := 0
878 var build func(string, string) int
879 build = func(target, parent string) int {
880 if result, ok := results[target]; ok {
881 return result
882 }
883 r := m.determine(target)
884 selected[target] = r
885 if r.missing {
886 message := fmt.Sprintf("make: *** No rule to make target '%s'", target)
887 if parent != "" {
888 message += fmt.Sprintf(", needed by '%s'", parent)
889 }
890 message += "."
891 if !options['k'] {
892 message += " Stop."
893 }
894 m.emit("%s", message)
895 results[target] = 1
896 if !options['k'] {
897 panic(abortBuild{})
898 }
899 return 1
900 }
901 if r.source {
902 results[target] = 0
903 return 0
904 }
905 building[target] = true
906 failed := false
907 prerequisites := []string{}
908 for _, p := range r.prerequisites {
909 if building[p] {
910 m.emit("make: Circular %s <- %s dependency dropped.", target, p)
911 continue
912 }
913 prerequisites = append(prerequisites, p)
914 if build(p, target) != 0 {
915 failed = true
916 }
917 }
918 delete(building, target)
919 if failed {
920 results[target] = 2
921 return 2
922 }
923 time, exists := m.files[target]
924 exists = exists && !m.phony[target]
925 newer := []string{}
926 for _, p := range prerequisites {
927 pt, present := m.files[p]
928 if options['B'] || !exists || !present || m.phony[p] || pt > time {
929 newer = append(newer, p)
930 }
931 }
932 if r.recipe == nil || !options['B'] && exists && len(newer) == 0 {
933 results[target] = 0
934 return 0
935 }
936 runs++
937 first := ""
938 if len(prerequisites) > 0 {
939 first = prerequisites[0]
940 }
941 m.automatic = map[string]string{"@": target, "<": first, "^": strings.Join(prerequisites, " "), "?": strings.Join(newer, " "), "*": r.stem}
942 expanded := make([]string, len(r.recipe.lines))
943 for i, line := range r.recipe.lines {
944 m.location = fmt.Sprintf("Makefile:%d", line.number)
945 expanded[i] = m.expand(line.text)
946 }
947 m.automatic = nil
948 for i, text := range expanded {
949 pos := 0
950 for pos < len(text) && (text[pos] == ' ' || text[pos] == '@' || text[pos] == '-') {
951 pos++
952 }
953 prefix, command := text[:pos], strings.TrimRight(text[pos:], " ")
954 if command == "" {
955 continue
956 }
957 if options['n'] || !strings.Contains(prefix, "@") && !options['s'] {
958 m.emit("%s", command)
959 }
960 if options['n'] {
961 continue
962 }
963 code := m.executeCommand(command)
964 if code == 0 {
965 continue
966 }
967 number := r.recipe.lines[i].number
968 if strings.Contains(prefix, "-") || options['i'] {
969 m.emit("make: [Makefile:%d: %s] Error %d (ignored)", number, target, code)
970 } else {
971 m.emit("make: *** [Makefile:%d: %s] Error %d", number, target, code)
972 results[target] = 1
973 if !options['k'] {
974 panic(abortBuild{})
975 }
976 return 1
977 }
978 }
979 if !m.phony[target] {
980 m.touch(target)
981 }
982 results[target] = 0
983 return 0
984 }
985 for _, goal := range goals {
986 before := runs
987 result := build(goal, "")
988 if result == 0 && runs == before {
989 if m.phony[goal] || selected[goal].recipe == nil {
990 m.emit("make: Nothing to be done for '%s'.", goal)
991 } else {
992 m.emit("make: '%s' is up to date.", goal)
993 }
994 } else if result == 2 {
995 m.emit("make: Target '%s' not remade because of errors.", goal)
996 }
997 }
998}
999func (m *simulator) command(line string, body []string) {
1000 m.resetExpansion()
1001 defer func() {
1002 if caught := recover(); caught != nil {
1003 if e, ok := caught.(expansionError); ok {
1004 m.emit("%s: *** %s. Stop.", e.location, e.reason)
1005 } else {
1006 panic(caught)
1007 }
1008 }
1009 }()
1010 m.emit("$ %s", line)
1011 command, text, _ := strings.Cut(line, " ")
1012 switch command {
1013 case "MAKEFILE":
1014 m.load(body)
1015 case "MAKE":
1016 m.make(words(text))
1017 case "TOUCH":
1018 for _, name := range words(text) {
1019 m.touch(name)
1020 }
1021 case "RM":
1022 for _, name := range words(text) {
1023 if _, exists := m.files[name]; exists {
1024 delete(m.files, name)
1025 } else {
1026 m.emit("rm: cannot remove '%s': No such file or directory", name)
1027 }
1028 }
1029 case "FILES":
1030 m.emit("clock %d", m.clock)
1031 for _, name := range keys(m.files) {
1032 m.emit("%s %d", name, m.files[name])
1033 }
1034 case "VARS":
1035 for _, name := range keys(m.variables) {
1036 value := trim(m.reference(name))
1037 symbol := "="
1038 if m.variables[name].simple {
1039 symbol = ":="
1040 }
1041 if value == "" {
1042 m.emit("%s %s", name, symbol)
1043 } else {
1044 m.emit("%s %s %s", name, symbol, value)
1045 }
1046 }
1047 case "RULES":
1048 display := func(r *rule) {
1049 m.emit("%s:", r.target)
1050 if len(r.prerequisites) > 0 {
1051 m.output[len(m.output)-1] += " " + strings.Join(r.prerequisites, " ")
1052 }
1053 if r.recipe != nil {
1054 for _, line := range r.recipe.lines {
1055 m.emit(">%s", line.text)
1056 }
1057 }
1058 }
1059 for _, name := range keys(m.rules) {
1060 display(m.rules[name])
1061 }
1062 for _, r := range m.patterns {
1063 if r.recipe != nil {
1064 display(r)
1065 }
1066 }
1067 if len(m.phony) > 0 {
1068 m.emit(".PHONY: %s", strings.Join(keys(m.phony), " "))
1069 }
1070 case "EXPAND":
1071 m.emit("[%s]", m.expand(text))
1072 }
1073}
1074func main() {
1075 scanner := bufio.NewScanner(os.Stdin)
1076 m := &simulator{files: map[string]int{}}
1077 m.clear()
1078 scanner.Scan()
1079 for scanner.Scan() {
1080 line := scanner.Text()
1081 if line == "END FILES" {
1082 break
1083 }
1084 name, text, _ := strings.Cut(line, " ")
1085 n, _ := strconv.Atoi(text)
1086 m.files[name] = n
1087 if n > m.clock {
1088 m.clock = n
1089 }
1090 }
1091 for scanner.Scan() {
1092 line := scanner.Text()
1093 if line == "QUIT" {
1094 break
1095 }
1096 body := []string{}
1097 if line == "MAKEFILE" {
1098 for scanner.Scan() {
1099 next := scanner.Text()
1100 if next == "END MAKEFILE" {
1101 break
1102 }
1103 body = append(body, next)
1104 }
1105 }
1106 m.command(line, body)
1107 }
1108 writer := bufio.NewWriter(os.Stdout)
1109 defer writer.Flush()
1110 for _, line := range m.output {
1111 fmt.Fprintln(writer, line)
1112 }
1113}
