23function closingindex(text, start)
24 opening=text[start]
25 closing=opening=='(' ? ')' : '}'
26 depth=0
27 for i in start:length(text)
28 if text[i]==opening
29 depth+=1
30 elseif text[i]==closing
31 depth-=1
32 depth==0 && return i
33 end
34 end
35 0
36end
37function outsideindex(text, wanted)
38 found=0
39 i=1
40 while i<=length(text)
41 if text[i]!='$'
42 text[i]==wanted && found==0 && (found=i)
43 i+=1
44 elseif i<length(text) && text[i+1] in ('(', '{')
45 last=closingindex(text, i+1)
46 last==0 && fail("unterminated variable reference")
47 i=last+1
48 else
49 i+=2
50 end
51 end
52 found
53end
54function patternstem(pattern, word)
55 at=findfirst(==('%'), pattern)
56 at===nothing && return pattern==word ? "" : nothing
57 first=pattern[1:(at-1)]
58 last=pattern[(at+1):end]
59 length(word)>=length(first)+length(last) && startswith(word, first) && endswith(word, last) ?
60 String(word[(length(first)+1):(end-length(last))]) : nothing
61end
62substitute(text, stem) = replace(text, "%"=>stem; count = 1)
63function replaceword(word, pattern, replacement)
64 stem=patternstem(pattern, word)
65 stem===nothing ? word : occursin('%', pattern) ? substitute(replacement, stem) : replacement
66end
67function splitarguments(text, opening, limit)
68 closing=opening=='(' ? ')' : '}'
69 depth=0
70 start=1
71 result=String[]
72 for i in eachindex(text)
73 if text[i]==opening
74 depth+=1
75 elseif text[i]==closing
76 depth-=1
77 elseif text[i]==',' && depth==0 && (limit==0 || length(result)<limit-1)
78 push!(result, String(text[start:(i-1)]))
79 start=i+1
80 end
81 end
82 push!(result, String(text[start:end]))
83 result
84end
85function globmatches(pattern, name)
86 previous=falses(length(name)+1)
87 previous[1]=true
88 for token in pattern
89 next=falses(length(name)+1)
90 if token=='*'
91 next[1]=previous[1]
92 for i in eachindex(name)
93 next[i+1]=previous[i+1] || (name[i]!='/' && next[i])
94 end
95 else
96 for i in eachindex(name)
97 next[i+1]=previous[i] && (token==name[i] || (token=='?' && name[i]!='/'))
98 end
99 end
100 previous=next
101 end
102 last(previous)
103end

199function wildcard(s, pattern)
200 if occursin('*', pattern) || occursin('?', pattern)
201 return [name for name in sort!(collect(keys(s.files))) if globmatches(pattern, name)]
202 end

205function expand(s, text)
206 result=IOBuffer()
207 i=1
208 while i<=length(text)
209 if text[i]!='$'
210 print(result, text[i])
211 i+=1
212 elseif i==length(text)
213 break
214 elseif text[i+1]=='$'
215 print(result, '$')
216 i+=2
217 elseif text[i+1] in ('(', '{')
218 last=closingindex(text, i+1)
219 last==0 && fail("unterminated variable reference")
220 print(result, reference(s, String(text[(i+2):(last-1)]), text[i+1]))
221 i=last+1
222 else
223 print(result, referencevalue(s, string(text[i+1])))
224 i+=2
225 end
226 end
227 String(take!(result))
228end
229function reference(s, content, opening)
230 space=findfirst(==(' '), content)
231 if space!==nothing && haskey(FUNCTIONS, content[1:(space-1)])
232 return callfunction(s, content[1:(space-1)], trimleft(content[(space+1):end]), opening)
233 end
234 name=expand(s, content)
235 colon=findfirst(==(':'), name)
236 if colon!==nothing
237 equals=findnext(==('='), name, colon+1)
238 if equals!==nothing
239 pattern=name[(colon+1):(equals-1)]
240 replacement=name[(equals+1):end]
241 if !occursin('%', pattern)
242 pattern="%"*pattern
243 replacement="%"*replacement
244 end
245 return join(
246 [
247 replaceword(word, pattern, replacement) for
248 word in words(referencevalue(s, name[1:(colon-1)]))
249 ],
250 ' ',
251 )
252 end
253 end
254 referencevalue(s, name)
255end
256function referencevalue(s, name)
257 for scope in reverse(s.scopes)
258 if scope.name!==nothing
259 scope.name==name && return scope.values[1]
260 elseif numerictext(name)
261 index=parse(Int, name)
262 return index<length(scope.values) ? scope.values[index+1] : ""
263 end
264 end
265 haskey(s.automatic, name) ? s.automatic[name] : variablevalue(s, name)
266end

275 finally

279function callfunction(s, name, argument, opening)
280 minimum, maximum=FUNCTIONS[name]
281 args=splitarguments(argument, opening, maximum)
282 length(args)>=minimum ||
283 fail("insufficient number of arguments ($(length(args))) to function '$name'")
284 if name=="foreach"
285 variable=trimspaces(expand(s, args[1]))
286 items=words(expand(s, args[2]))
287 result=String[]
288 for word in items
289 push!(s.scopes, ExpansionScope(variable, [word]))
290 try
291 push!(result, expand(s, args[3]))
292 finally
293 pop!(s.scopes)
294 end
295 end
296 return join(result, ' ')
297 elseif name=="if"
298 if !isempty(trimspaces(expand(s, args[1])))
299 return expand(s, args[2])
300 end
301 return length(args)==3 ? expand(s, args[3]) : ""
302 end
303 vals=[expand(s, arg) for arg in args]
304 !(name in ("subst", "call")) && (vals=trimspaces.(vals))
305 if name=="subst"
306 return isempty(vals[1]) ? vals[3] : replace(vals[3], vals[1]=>vals[2])
307 elseif name=="patsubst"
308 return join([replaceword(word, vals[1], vals[2]) for word in words(vals[3])], ' ')
309 elseif name=="strip"
310 return join(words(vals[1]), ' ')
311 elseif name=="findstring"
312 return occursin(vals[1], vals[2]) ? vals[1] : ""
313 elseif name in ("filter", "filter-out")
314 return join(
315 [
316 word for word in words(vals[2]) if
317 any(p->patternstem(p, word)!==nothing, words(vals[1]))==(name=="filter")
318 ],
319 ' ',
320 )
321 elseif name=="sort"
322 return join(sort!(unique(words(vals[1]))), ' ')
323 elseif name=="word"
324 numerictext(vals[1]) || fail("non-numeric first argument to 'word' function: '$(vals[1])'")
325 index=parse(Int, vals[1])
326 index>0 || fail("first argument to 'word' function must be greater than 0")
327 items=words(vals[2])
328 return index<=length(items) ? items[index] : ""
329 elseif name=="wordlist"
330 numerictext(vals[1]) ||
331 fail("non-numeric first argument to 'wordlist' function: '$(vals[1])'")
332 numerictext(vals[2]) ||
333 fail("non-numeric second argument to 'wordlist' function: '$(vals[2])'")
334 startindex=parse(Int, vals[1])
335 endindex=parse(Int, vals[2])
336 startindex>0 || fail("invalid first argument to 'wordlist' function: '$(vals[1])'")
337 items=words(vals[3])
338 return startindex>min(endindex, length(items)) ? "" :
339 join(items[startindex:min(endindex, length(items))], ' ')
340 elseif name=="words"
341 return string(length(words(vals[1])))
342 elseif name in ("firstword", "lastword")
343 items=words(vals[1])
344 return isempty(items) ? "" : name=="firstword" ? first(items) : last(items)
345 elseif name in ("dir", "notdir", "suffix", "basename")
346 result=String[]
347 for word in words(vals[1])
348 slash=something(findlast(==('/'), word), 0)
349 dot=something(findlast(==('.'), word), 0)
350 if name=="dir"
351 push!(result, slash==0 ? "./" : word[1:slash])
352 elseif name=="notdir"
353 push!(result, word[(slash+1):end])
354 elseif name=="suffix"
355 dot>slash && push!(result, word[dot:end])
356 else
357 push!(result, dot>slash ? word[1:(dot-1)] : word)
358 end
359 end
360 return join(result, ' ')
361 elseif name=="addsuffix"
362 return join([word*vals[1] for word in words(vals[2])], ' ')
363 elseif name=="addprefix"
364 return join([vals[1]*word for word in words(vals[2])], ' ')
365 elseif name=="wildcard"
366 return join([found for pattern in words(vals[1]) for found in wildcard(s, pattern)], ' ')
367 elseif name=="call"
368 variable=trimspaces(vals[1])
369 haskey(s.makefile.variables, variable) || return ""
370 value=s.makefile.variables[variable]
371 value.recursive || return value.value
372 push!(s.scopes, ExpansionScope(nothing, [variable; vals[2:end]]))
373 try
374 return expand(s, value.value)
375 finally
376 pop!(s.scopes)
377 end
378 elseif name=="value"
379 return haskey(s.makefile.variables, vals[1]) ? s.makefile.variables[vals[1]].value : ""
380 elseif name=="info"
381 println(vals[1])
382 elseif name=="warning"
383 println("$(s.location): $(vals[1])")
384 elseif name=="error"
385 fail(vals[1])
386 end
387 ""
388end
389active(conditions) = isempty(conditions) || (last(conditions).outer && last(conditions).value)
390function evaluatecondition(s, keyword, rest)
391 if keyword in ("ifdef", "ifndef")
392 name=trimspaces(expand(s, rest))
393 defined=haskey(s.makefile.variables, name) && !isempty(s.makefile.variables[name].value)
394 return defined==(keyword=="ifdef")
395 end
396 length(rest)>=2 && first(rest)=='(' && last(rest)==')' || fail("invalid syntax in conditional")
397 inner=rest[2:(end-1)]
398 depth=0
399 for i in eachindex(inner)
400 if inner[i]=='('
401 depth+=1
402 elseif inner[i]==')'
403 depth-=1
404 elseif inner[i]==',' && depth==0
405 left=trimspaces(expand(s, inner[1:(i-1)]))
406 right=trimspaces(expand(s, inner[(i+1):end]))
407 return (left==right)==(keyword=="ifeq")
408 end
409 end
410 fail("invalid syntax in conditional")
411end
412function assign!(s, text, equals)
413 op=equals>1 && text[equals-1] in (':', '+', '?') ? string(text[equals-1]) : ""
414 name=trimspaces(text[1:(equals-1-length(op))])
415 value=trimspaces(text[(equals+1):end])
416 isempty(name) && fail("empty variable name")
417 variables=s.makefile.variables
418 if op==":"
419 variables[name]=Variable(false, expand(s, value))
420 elseif op=="+" && haskey(variables, name)
421 old=variables[name]
422 addition=old.recursive ? value : expand(s, value)
423 old.value=isempty(old.value) ? addition : old.value*" "*addition
424 elseif op!="?" || !haskey(variables, name)
425 variables[name]=Variable(true, value)
426 end
427end

452function definerule!(s, targettext, rest, number)
453 targets=unique(words(expand(s, targettext)))
454 inline=nothing
455 semicolon=outsideindex(rest, ';')
456 if semicolon!=0
457 inline=trimleft(rest[(semicolon+1):end])
458 rest=rest[1:(semicolon-1)]
459 end
460 colon=outsideindex(rest, ':')
461 pattern=nothing
462 if colon!=0
463 pattern=trimspaces(expand(s, rest[1:(colon-1)]))
464 rest=rest[(colon+1):end]
465 end
466 prerequisites=words(expand(s, rest))
467 group=RuleGroup()
468 if targets==[".PHONY"]
469 union!(s.makefile.phony, prerequisites)
470 elseif pattern!==nothing
471 occursin('%', pattern) || fail("target pattern contains no '%'")
472 for target in targets
473 stem=patternstem(pattern, target)
474 if stem!==nothing && !isempty(stem)
475 register!(s, target, [substitute(p, stem) for p in prerequisites])
476 s.makefile.rules[target].stem=stem
477 push!(group.targets, target)
478 else
479 println("Makefile:$number: target '$target' doesn't match the target pattern")
480 end
481 end
482 elseif any(t->occursin('%', t), targets)
483 all(t->occursin('%', t), targets) || fail("mixed implicit and normal rules")
484 for target in targets
485 rule=PatternRule(target, prerequisites, nothing)
486 push!(s.makefile.patterns, rule)
487 push!(group.patterns, rule)
488 end
489 else
490 for target in targets
491 register!(s, target, prerequisites)
492 end
493 group.targets=targets
494 end
495 inline!==nothing && addrecipe!(s, group, RecipeLine(number, inline))
496 group
497end
498function loadmakefile!(s, lines)
499 s.makefile=Makefile()
500 conditions=MakeCondition[]
501 group=nothing
502 i=1
503 try
504 while i<=length(lines)
505 number=i
506 line=lines[i]
507 i+=1
508 s.location="Makefile:$number"
509 if startswith(line, ">")
510 if active(conditions)
511 group===nothing && fail("recipe commences before first target")
512 addrecipe!(s, group, RecipeLine(number, trimleft(line[2:end])))
513 end
514 continue
515 end
516 while endswith(line, "\\") && i<=length(lines)
517 line=trimright(line[1:(end-1)])*" "*trimleft(lines[i])
518 i+=1
519 end
520 hash=findfirst(==('#'), line)
521 text=trimspaces(hash===nothing ? line : line[1:(hash-1)])
522 parts=split(text, ' '; limit = 2)
523 keyword=isempty(parts) ? "" : parts[1]
524 rest=length(parts)==2 ? trimspaces(parts[2]) : ""
525 if keyword in ("ifeq", "ifneq", "ifdef", "ifndef", "else", "endif")
526 if keyword=="else"
527 isempty(conditions) && fail("extraneous 'else'")
528 last(conditions).seenelse && fail("only one 'else' per conditional")
529 last(conditions).value=!last(conditions).value
530 last(conditions).seenelse=true
531 elseif keyword=="endif"
532 isempty(conditions) && fail("extraneous 'endif'")
533 pop!(conditions)
534 else
535 outer=active(conditions)
536 value=outer && evaluatecondition(s, keyword, rest)
537 push!(conditions, MakeCondition(outer, value, false, number))
538 end
539 elseif !isempty(text) && active(conditions)
540 equals=outsideindex(text, '=')
541 colon=outsideindex(text, ':')
542 if equals!=0 && (colon==0 || colon>=equals-1)
543 assign!(s, text, equals)
544 group=nothing
545 elseif colon!=0
546 group=definerule!(s, text[1:(colon-1)], text[(colon+1):end], number)
547 else
548 isempty(trimspaces(expand(s, text))) || fail("missing separator")
549 end
550 end
551 end
552 if !isempty(conditions)
553 s.location="Makefile:$(last(conditions).number)"
554 fail("missing 'endif'")
555 end
556 filter!(r->r.recipe!==nothing, s.makefile.patterns)
557 s.makefile.loaded=true
558 println(
559 "Makefile loaded: lines=$(length(lines)) variables=$(length(s.makefile.variables)) targets=$(length(s.makefile.rules)) patterns=$(length(s.makefile.patterns)) goal=$(isempty(s.makefile.goal) ? "(none)" : s.makefile.goal)",
560 )
561 catch e
562 e isa MakeError || rethrow()
563 println("$(s.location): *** $(e.message). Stop.")
564 s.makefile=Makefile()
565 finally
566 s.location="make"
567 end
568end

575function resolverule(s, target)
576 m=s.makefile
577 explicit=get(m.rules, target, nothing)
578 explicit!==nothing &&
579 explicit.recipe!==nothing &&
580 return ResolvedRule(explicit.prerequisites, explicit.recipe, explicit.stem, false)
581 if !(target in m.phony)
582 candidates=Tuple{String,Int}[]
583 for (i, rule) in enumerate(m.patterns)
584 stem=patternstem(rule.target, target)
585 stem!==nothing && !isempty(stem) && push!(candidates, (stem, i))
586 end
587 sort!(candidates; by = x->(length(x[1]), x[2]))
588 for (stem, i) in candidates
589 rule=m.patterns[i]
590 prerequisites=[substitute(p, stem) for p in rule.prerequisites]
591 if all(p->haskey(s.files, p) || haskey(m.rules, p), prerequisites)
592 explicit!==nothing && append!(prerequisites, explicit.prerequisites)
593 return ResolvedRule(unique(prerequisites), rule.recipe, stem, false)
594 end
595 end
596 end
597 explicit!==nothing && return ResolvedRule(explicit.prerequisites, nothing, explicit.stem, false)
598 target in m.phony && return ResolvedRule(String[], nothing, "", false)
599 haskey(s.files, target) ? ResolvedRule(String[], nothing, "", true) : nothing
600end
601targettime(s, target) = target in s.makefile.phony ? nothing : get(s.files, target, nothing)
602function executerecipe!(s, command)
603 parts=words(command)
604 name=parts[1]
605 args=parts[2:end]
606 if name=="echo"
607 println(join(args, ' '))
608 elseif name=="touch"
609 for arg in args
610 touch!(s, arg)
611 end
612 elseif name=="false"
613 return 1
614 elseif name=="exit"
615 return isempty(args) ? 0 : parse(Int, args[1])
616 elseif name=="rm"
617 forced=!isempty(args) && args[1]=="-f"
618 forced && (args=args[2:end])
619 status=0
620 for arg in args
621 matches=wildcard(s, arg)
622 isempty(matches) && (matches=[arg])
623 for name in matches
624 if haskey(s.files, name)
625 delete!(s.files, name)
626 elseif !forced
627 println("rm: cannot remove '$name': No such file or directory")
628 status=1
629 end
630 end
631 end
632 return status
633 end
634 0
635end
636function make!(s, arguments)
637 options=Set{Char}()
638 goals=String[]
639 for arg in arguments
640 if !startswith(arg, "-")
641 push!(goals, arg)
642 continue
643 end
644 for option in arg[2:end]
645 if !(option in "nBkis")
646 println("make: invalid option -- '$option'")
647 return
648 end
649 push!(options, option)
650 end
651 end
652 if isempty(goals)
653 if !s.makefile.loaded
654 println("make: *** No targets specified and no makefile found. Stop.")
655 return
656 elseif isempty(s.makefile.goal)
657 println("make: *** No targets. Stop.")
658 return
659 end
660 goals=[s.makefile.goal]
661 end
662 savedfiles=copy(s.files)
663 savedclock=s.clock
664 results=Dict{String,Symbol}()
665 resolved=Dict{String,ResolvedRule}()
666 building=Set{String}()
667 runs=0
668 function runrecipe(target, rule, prerequisites, changed)
669 s.automatic=Dict(
670 "@"=>target,
671 "<"=>(isempty(prerequisites) ? "" : first(prerequisites)),
672 "^"=>join(prerequisites, ' '),
673 "?"=>join(changed, ' '),
674 "*"=>rule.stem,
675 )
676 expanded=String[]
677 try
678 for line in rule.recipe
679 s.location="Makefile:$(line.number)"
680 push!(expanded, expand(s, line.command))
681 end
682 catch e
683 e isa MakeError || rethrow()
684 println("$(s.location): *** $(e.message). Stop.")
685 throw(BuildAborted())
686 finally
687 empty!(s.automatic)
688 s.location="make"
689 end
690 for (line, text) in zip(rule.recipe, expanded)
691 firstcommand=1
692 while firstcommand<=length(text) && text[firstcommand] in (' ', '@', '-')
693 firstcommand+=1
694 end
695 prefix=text[1:(firstcommand-1)]
696 command=trimright(text[firstcommand:end])
697 isempty(command) && continue
698 ('n' in options || (!occursin('@', prefix) && !('s' in options))) && println(command)
699 'n' in options && continue
700 status=executerecipe!(s, command)
701 status==0 && continue
702 if occursin('-', prefix) || 'i' in options
703 println("make: [Makefile:$(line.number): $target] Error $status (ignored)")
704 else
705 println("make: *** [Makefile:$(line.number): $target] Error $status")
706 'k' in options || throw(BuildAborted())
707 return false
708 end
709 end
710 true
711 end
712 function build(target, parent)
713 haskey(results, target) && return results[target]
714 rule=resolverule(s, target)
715 if rule===nothing
716 results[target]=:failure
717 message="make: *** No rule to make target '$target'"*(
718 parent===nothing ? "." : ", needed by '$parent'."
719 )
720 if !('k' in options)
721 println(message*" Stop.")
722 throw(BuildAborted())
723 end
724 println(message)
725 return :failure
726 end
727 resolved[target]=rule
728 if rule.source
729 results[target]=:success
730 return :success
731 end
732 push!(building, target)
733 prerequisites=String[]
734 failed=false
735 for dep in rule.prerequisites
736 if dep in building
737 println("make: Circular $target <- $dep dependency dropped.")
738 continue
739 end
740 push!(prerequisites, dep)
741 build(dep, target)!=:success && (failed=true)
742 end
743 delete!(building, target)
744 result=:success
745 if failed
746 result=:dependency
747 else
748 timestamp=targettime(s, target)
749 changed=String[]
750 for dep in prerequisites
751 value=targettime(s, dep)
752 ('B' in options || timestamp===nothing || value===nothing || value>timestamp) &&
753 push!(changed, dep)
754 end
755 if rule.recipe!==nothing && ('B' in options || timestamp===nothing || !isempty(changed))
756 runs+=1
757 if runrecipe(target, rule, prerequisites, changed)
758 !(target in s.makefile.phony) && touch!(s, target)
759 else
760 result=:failure
761 end
762 end
763 end
764 results[target]=result
765 result
766 end
767 try
768 for goal in goals
769 before=runs
770 result=build(goal, nothing)
771 if result==:success && runs==before
772 if goal in s.makefile.phony || resolved[goal].recipe===nothing
773 println("make: Nothing to be done for '$goal'.")
774 else
775 println("make: '$goal' is up to date.")
776 end
777 elseif result==:dependency
778 println("make: Target '$goal' not remade because of errors.")
779 end
780 end
781 catch e
782 e isa BuildAborted || rethrow()
783 finally
784 if 'n' in options
785 s.files=savedfiles
786 s.clock=savedclock
787 end
788 end
789end
790function printrule(target, prerequisites, recipe)
791 println(join([target*":"; prerequisites], ' '))
792 recipe===nothing || foreach(line->println(">"*line.command), recipe)
793end
794function main()
795 lines=readlines()
796 files=Dict{String,Int}()
797 i=2
798 while lines[i]!="END FILES"
799 name, timestamp=split(lines[i], ' ')
800 files[name]=parse(Int, timestamp)
801 i+=1
802 end
803 i+=1
804 simulator=Simulator(files)
805 while lines[i]!="QUIT"
806 line=lines[i]
807 i+=1
808 println("\$ ", line)
809 parts=split(line, ' '; limit = 2)
810 command=parts[1]
811 argument=length(parts)==2 ? String(parts[2]) : ""
812 if command=="MAKEFILE"
813 body=String[]
814 while lines[i]!="END MAKEFILE"
815 push!(body, lines[i])
816 i+=1
817 end
818 i+=1
819 loadmakefile!(simulator, body)
820 elseif command=="MAKE"
821 make!(simulator, words(argument))
822 elseif command=="TOUCH"
823 for name in words(argument)
824 touch!(simulator, name)
825 end
826 elseif command=="RM"
827 for name in words(argument)
828 if haskey(simulator.files, name)
829 delete!(simulator.files, name)
830 else
831 println("rm: cannot remove '$name': No such file or directory")
832 end
833 end
834 elseif command=="FILES"
835 println("clock $(simulator.clock)")
836 for name in sort!(collect(keys(simulator.files)))
837 println("$name $(simulator.files[name])")
838 end
839 elseif command=="VARS"
840 try
841 for name in sort!(collect(keys(simulator.makefile.variables)))
842 symbol=simulator.makefile.variables[name].recursive ? "=" : ":="
843 value=trimspaces(variablevalue(simulator, name))
844 println("$name $symbol"*(isempty(value) ? "" : " "*value))
845 end
846 catch e
847 e isa MakeError || rethrow()
848 println("make: *** $(e.message). Stop.")
849 end
850 elseif command=="RULES"
851 m=simulator.makefile
852 for target in sort!(collect(keys(m.rules)))
853 rule=m.rules[target]
854 printrule(target, rule.prerequisites, rule.recipe)
855 end
856 for rule in m.patterns
857 printrule(rule.target, rule.prerequisites, rule.recipe)
858 end
859 !isempty(m.phony) && println(join([".PHONY:"; sort!(collect(m.phony))], ' '))
860 elseif command=="EXPAND"
861 try
862 println("[", expand(simulator, argument), "]")
863 catch e
864 e isa MakeError || rethrow()
865 println("make: *** $(e.message). Stop.")
866 end
867 end
868 end
869end
