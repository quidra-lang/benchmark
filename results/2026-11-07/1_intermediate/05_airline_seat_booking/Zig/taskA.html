1const std = @import("std");
2const Allocator = std.mem.Allocator;
3const Error = error{BadCommand} || Allocator.Error || std.Io.Writer.Error;
4const class_names = [_][]const u8{ "F", "J", "Y" };
5const tier_names = [_][]const u8{ "PLATINUM", "GOLD", "SILVER", "NONE" };
6fn integer(text: []const u8) usize {
7 return std.fmt.parseInt(usize, text, 10) catch unreachable;
8}
9fn classIndex(text: []const u8) usize {
10 return std.mem.findScalar(u8, "FJY", text[0]).?;
11}
12fn split(gpa: Allocator, text: []const u8, delimiter: u8) Allocator.Error![][]const u8 {
13 var result: std.ArrayList([]const u8) = .empty;
14 var it = std.mem.splitScalar(u8, text, delimiter);
15 while (it.next()) |part| try result.append(gpa, part);
16 return result.items;
17}
18const Seat = struct { row: usize, letter: u8, cabin: usize, block: usize, kind: u8 };
19const Cabin = struct { first: usize, last: usize, blocks: [][]const u8 };
20const Aircraft = struct {
21 name: []const u8,
22 base: usize,
23 limit: i64,
24 cabins: [3]?Cabin = .{ null, null, null },
25 seats: std.ArrayList(Seat) = .empty,
26 grid: [100][26]?usize = @splat(@splat(null)),
27 exits: [100]bool = @splat(false),
28 blocked: [2600]bool = @splat(false),
29 fn seatIndex(self: *const Aircraft, code: []const u8) ?usize {
30 const row = integer(code[0 .. code.len - 1]);
31 return self.grid[row][code[code.len - 1] - 'A'];
32 }
33 fn addCabin(self: *Aircraft, gpa: Allocator, c: usize, first: usize, last: usize, blocks: [][]const u8) Allocator.Error!void {
34 self.cabins[c] = .{ .first = first, .last = last, .blocks = blocks };
35 for (first..last + 1) |row| {
36 for (blocks, 0..) |block, b| {
37 for (block, 0..) |letter, x| {
38 const kind: u8 = if ((b == 0 and x == 0) or (b == blocks.len - 1 and x == block.len - 1)) 'W' else if (x == 0 or x == block.len - 1) 'A' else 'M';
39 self.grid[row][letter - 'A'] = self.seats.items.len;
40 try self.seats.append(gpa, .{ .row = row, .letter = letter, .cabin = c, .block = b, .kind = kind });
41 }
42 }
43 }
44 }
45 fn capacity(self: *const Aircraft, c: usize) usize {
46 var count: usize = 0;
47 for (self.seats.items, 0..) |s, i| {
48 if (s.cabin == c and !self.blocked[i]) count += 1;
49 }
50 return count;
51 }

53const PassengerType = enum { ADULT, CHILD, INFANT };
54const Passenger = struct {
55 name: []const u8,
56 kind: PassengerType,
57 preference: u8,
58 cabin: usize,
59 booking: *Booking = undefined,
60 seat: ?usize = null,
61 active: bool = true,
62 infant: ?*Passenger = null,
63 guardian: ?*Passenger = null,
64 fn needsSeat(self: *const Passenger) bool {
65 return self.kind != .INFANT;
66 }
67 fn carrying(self: *const Passenger) bool {
68 return if (self.infant) |p| p.active else false;
69 }

71const Booking = struct {
72 pnr: []const u8,
73 flight: *Flight,
74 listed: []*Passenger,
75 checked: bool = false,
76 fn count(self: *const Booking) usize {
77 var result: usize = 0;
78 for (self.listed) |p| {
79 if (p.active) result += 1;
80 }
81 return result;
82 }
83 fn hasInfants(self: *const Booking) bool {
84 for (self.listed) |p| {
85 if (p.active and !p.needsSeat()) return true;
86 }
87 return false;
88 }

90const Flight = struct {
91 name: []const u8,
92 aircraft: *Aircraft,
93 limits: [3]usize,
94 blocked: [2600]bool,
95 occupied: [2600]?*Passenger = @splat(null),
96 bookings: std.ArrayList(*Booking) = .empty,
97 volunteers: std.ArrayList(*Passenger) = .empty,
98 closed: bool = false,
99 fn allPassengers(self: *const Flight, gpa: Allocator) Allocator.Error![]*Passenger {
100 var result: std.ArrayList(*Passenger) = .empty;
101 for (self.bookings.items) |b| for (b.listed) |p| {
102 if (p.active) try result.append(gpa, p);
103 };
104 return result.items;
105 }
106 fn booked(self: *const Flight, c: usize) usize {
107 var result: usize = 0;
108 for (self.bookings.items) |b| for (b.listed) |p| {
109 if (p.active and p.needsSeat() and p.cabin == c) result += 1;
110 };
111 return result;
112 }
113 fn place(self: *Flight, p: *Passenger, seat: ?usize) void {
114 if (p.seat) |old| self.occupied[old] = null;
115 p.seat = seat;
116 if (seat) |index| self.occupied[index] = p;
117 }
118 fn fits(self: *const Flight, ps: []const *Passenger, ss: []const usize, c: usize, vacated: ?*Passenger) bool {
119 for (ps, ss, 0..) |p, index, i| {
120 const s = self.aircraft.seats.items[index];
121 if (s.cabin != c or self.blocked[index]) return false;
122 if (std.mem.findScalar(usize, ss[0..i], index) != null) return false;
123 if (self.occupied[index]) |occupant| {
124 if (std.mem.findScalar(*Passenger, ps, occupant) == null and occupant != vacated) return false;
125 }
126 if ((p.kind == .CHILD or p.carrying()) and self.aircraft.exits[s.row]) return false;
127 if (p.carrying()) {
128 const block = self.aircraft.cabins[c].?.blocks[s.block];
129 for (block) |letter| {
130 const other_index = self.aircraft.grid[s.row][letter - 'A'].?;
131 if (self.occupied[other_index]) |other| {
132 if (other != p and std.mem.findScalar(*Passenger, ps, other) == null and other != vacated and other.carrying()) return false;
133 }
134 }
135 for (ps, ss, 0..) |other_p, other_index, j| {
136 const other_s = self.aircraft.seats.items[other_index];
137 if (j != i and other_p.carrying() and s.row == other_s.row and s.block == other_s.block) return false;
138 }
139 }
140 }
141 return true;
142 }
143 fn choose(self: *const Flight, p: *Passenger, c: usize) ?usize {
144 var first: ?usize = null;
145 for (self.aircraft.seats.items, 0..) |s, i| {
146 if (!self.fits(&.{p}, &.{i}, c, null)) continue;
147 if (first == null) first = i;
148 if (p.preference == s.kind) return i;
149 }
150 return first;
151 }
152 fn adopt(self: *Flight, ps: []const *Passenger, ss: []const usize, c: usize) bool {
153 if (!self.fits(ps, ss, c, null)) return false;
154 for (ps, ss) |p, seat| self.place(p, seat);
155 return true;
156 }
157 fn assign(self: *Flight, b: *Booking, c: usize) []const u8 {
158 var passengers: [9]*Passenger = undefined;
159 var k: usize = 0;
160 for (b.listed) |p| {
161 if (p.needsSeat()) {
162 passengers[k] = p;
163 k += 1;
164 }
165 }
166 const ps = passengers[0..k];
167 if (k == 1) {
168 self.place(ps[0], self.choose(ps[0], c));
169 return "SINGLE";
170 }
171 const cabin = self.aircraft.cabins[c].?;
172 var candidate: [9]usize = undefined;
173 for (cabin.first..cabin.last + 1) |row| {
174 for (cabin.blocks) |block| {
175 var x: usize = 0;
176 while (x + k <= block.len) : (x += 1) {
177 for (0..k) |j| candidate[j] = self.aircraft.grid[row][block[x + j] - 'A'].?;
178 if (self.adopt(ps, candidate[0..k], c)) return "BLOCK";
179 }
180 }
181 }
182 for (cabin.first..cabin.last + 1) |row| {
183 var line: [10]usize = undefined;
184 var len: usize = 0;
185 for (cabin.blocks) |block| for (block) |letter| {
186 line[len] = self.aircraft.grid[row][letter - 'A'].?;
187 len += 1;
188 };
189 var x: usize = 0;
190 while (x + k <= len) : (x += 1) {
191 if (self.adopt(ps, line[x .. x + k], c)) return "ROW";
192 }
193 }
194 for (2..k + 1) |h| {
195 const width = (k + h - 1) / h;
196 const height = (k + width - 1) / width;
197 var row = cabin.first;
198 while (row + height - 1 <= cabin.last) : (row += 1) {
199 for (cabin.blocks) |block| {
200 var x: usize = 0;
201 while (x + width <= block.len) : (x += 1) {
202 for (0..k) |j| candidate[j] = self.aircraft.grid[row + j / width][block[x + j % width] - 'A'].?;
203 if (self.adopt(ps, candidate[0..k], c)) return "STACK";
204 }
205 }
206 }
207 }
208 for (ps) |p| self.place(p, self.choose(p, c));
209 return "SPLIT";
210 }
211 fn displaySeat(self: *const Flight, gpa: Allocator, p: *const Passenger) Allocator.Error![]const u8 {
212 if (!p.needsSeat()) return "LAP";
213 const seat = p.seat orelse return "UNASSIGNED";
214 const s = self.aircraft.seats.items[seat];
215 return std.fmt.allocPrint(gpa, "{d}{c}", .{ s.row, s.letter });
216 }
217 fn reassign(self: *Flight, gpa: Allocator, out: *std.Io.Writer) Error!void {
218 for (self.bookings.items) |b| for (b.listed) |p| {
219 if (!p.active or !p.needsSeat() or p.seat != null) continue;
220 if (self.choose(p, p.cabin)) |seat| {
221 self.place(p, seat);
222 try out.print("REASSIGNED {s} {s} {s}\n", .{ b.pnr, p.name, try self.displaySeat(gpa, p) });
223 }
224 };
225 }
226 fn remove(self: *Flight, p: *Passenger) void {
227 if (!p.active) return;
228 if (p.carrying()) self.remove(p.infant.?);
229 if (p.guardian) |guardian| {
230 guardian.infant = null;
231 p.guardian = null;
232 }
233 self.place(p, null);
234 p.active = false;
235 if (std.mem.findScalar(*Passenger, self.volunteers.items, p)) |i|
236 _ = self.volunteers.orderedRemove(i);
237 }
238};
239const Airline = struct {
240 gpa: Allocator,
241 models: std.StringHashMapUnmanaged(*Aircraft) = .empty,
242 flights: std.StringArrayHashMapUnmanaged(*Flight) = .empty,
243 bookings: std.StringHashMapUnmanaged(*Booking) = .empty,
244 statuses: std.StringHashMapUnmanaged(usize) = .empty,
245 defining: ?*Aircraft = null,
246 error_buffer: [1024]u8 = undefined,
247 error_len: usize = 0,
248 fn fail(self: *Airline, comptime fmt: []const u8, args: anytype) error{BadCommand} {
249 const message = std.fmt.bufPrint(&self.error_buffer, fmt, args) catch unreachable;
250 self.error_len = message.len;
251 return error.BadCommand;
252 }
253 fn tier(self: *const Airline, name: []const u8) usize {
254 return self.statuses.get(name) orelse 3;
255 }
256 fn findFlight(self: *Airline, name: []const u8, open: bool) Error!*Flight {
257 const f = self.flights.get(name) orelse return self.fail("unknown flight {s}", .{name});
258 if (open and f.closed) return self.fail("flight {s} is closed", .{name});
259 return f;
260 }
261 fn findBooking(self: *Airline, pnr: []const u8, open: bool) Error!*Booking {
262 const b = self.bookings.get(pnr) orelse return self.fail("unknown PNR {s}", .{pnr});
263 if (b.count() == 0) return self.fail("unknown PNR {s}", .{pnr});
264 if (open and b.flight.closed) return self.fail("flight {s} is closed", .{b.flight.name});
265 return b;
266 }
267 fn findPassenger(self: *Airline, b: *Booking, name: []const u8) Error!*Passenger {
268 for (b.listed) |p| {
269 if (p.active and std.mem.eql(u8, p.name, name)) return p;
270 }
271 return self.fail("no passenger {s} in {s}", .{ name, b.pnr });
272 }
273 fn hasCabin(self: *Airline, f: *Flight, c: usize) Error!void {
274 if (f.aircraft.cabins[c] == null) return self.fail("no cabin {s} on {s}", .{ class_names[c], f.name });
275 }
276 fn deny(self: *Airline, f: *Flight, p: *Passenger, voluntary: bool, out: *std.Io.Writer) Error!i64 {
277 var value: i64 = ([_]i64{ 500, 300, 150 })[p.cabin] * @as(i64, if (voluntary) 1 else 2);
278 if (p.kind == .CHILD) value = @divFloor(value, 2);
279 try out.print("DENIED {s} {s} {s} {s} {d}\n", .{ @as([]const u8, if (voluntary) "VOLUNTARY" else "INVOLUNTARY"), p.booking.pnr, p.name, class_names[p.cabin], value });
280 f.remove(p);
281 _ = self;
282 return value;
283 }
284 fn execute(self: *Airline, line: []const u8, out: *std.Io.Writer) Error!void {
285 const a = try split(self.gpa, line, ' ');
286 const op = a[0];
287 if (std.mem.eql(u8, op, "AIRCRAFT")) {
288 const model = try self.gpa.create(Aircraft);
289 model.* = .{ .name = a[1], .base = integer(a[2]), .limit = @intCast(integer(a[3])) };
290 self.defining = model;
291 } else if (std.mem.eql(u8, op, "CABIN")) {
292 try self.defining.?.addCabin(self.gpa, classIndex(a[1]), integer(a[2]), integer(a[3]), a[4..]);
293 } else if (std.mem.eql(u8, op, "EXIT")) {
294 for (a[1..]) |row| self.defining.?.exits[integer(row)] = true;
295 } else if (std.mem.eql(u8, op, "BLOCKED")) {
296 for (a[1..]) |seat| self.defining.?.blocked[self.defining.?.seatIndex(seat).?] = true;
297 } else if (std.mem.eql(u8, op, "END")) {
298 const model = self.defining.?;
299 try self.models.put(self.gpa, model.name, model);
300 self.defining = null;
301 try out.print("AIRCRAFT {s}", .{model.name});
302 for (model.cabins, 0..) |cabin, c| {
303 if (cabin != null) try out.print(" {s} {d}", .{ class_names[c], model.capacity(c) });
304 }
305 try out.writeByte('\n');
306 } else if (std.mem.eql(u8, op, "FLIGHT")) {
307 if (self.flights.contains(a[1])) return self.fail("duplicate flight {s}", .{a[1]});
308 const model = self.models.get(a[2]) orelse return self.fail("unknown aircraft {s}", .{a[2]});
309 const f = try self.gpa.create(Flight);
310 f.* = .{ .name = a[1], .aircraft = model, .blocked = model.blocked, .limits = .{ model.capacity(0) * (100 + integer(a[3])) / 100, model.capacity(1) * (100 + integer(a[3])) / 100, model.capacity(2) * (100 + integer(a[3])) / 100 } };
311 try self.flights.put(self.gpa, f.name, f);
312 try out.print("FLIGHT {s} {s}", .{ f.name, model.name });
313 for (model.cabins, 0..) |cabin, c| {
314 if (cabin != null) try out.print(" {s} {d}/{d}", .{ class_names[c], model.capacity(c), f.limits[c] });
315 }
316 try out.writeByte('\n');
317 } else if (std.mem.eql(u8, op, "STATUS")) {
318 var value: usize = 3;
319 for (tier_names, 0..) |name, i| {
320 if (std.mem.eql(u8, name, a[2])) value = i;
321 }
322 try self.statuses.put(self.gpa, a[1], value);
323 try out.print("STATUS {s} {s}\n", .{ a[1], a[2] });
324 } else if (std.mem.eql(u8, op, "BOOK")) {
325 if (self.bookings.contains(a[1])) return self.fail("duplicate PNR {s}", .{a[1]});
326 const f = try self.findFlight(a[2], true);
327 const c = classIndex(a[3]);
328 try self.hasCabin(f, c);
329 const ps = try self.gpa.alloc(*Passenger, a.len - 4);
330 var adults: [9]*Passenger = undefined;
331 var adult_count: usize = 0;
332 var infants: [9]*Passenger = undefined;
333 var infant_count: usize = 0;
334 for (a[4..], 0..) |token, i| {
335 const parts = try split(self.gpa, token, ':');
336 const p = try self.gpa.create(Passenger);
337 p.* = .{ .name = parts[0], .kind = std.meta.stringToEnum(PassengerType, parts[1]).?, .preference = if (parts.len == 3) parts[2][0] else 0, .cabin = c };
338 ps[i] = p;
339 if (p.kind == .ADULT) {
340 adults[adult_count] = p;
341 adult_count += 1;
342 }
343 if (p.kind == .INFANT) {
344 infants[infant_count] = p;
345 infant_count += 1;
346 }
347 for (ps[0..i]) |previous| {
348 if (std.mem.eql(u8, previous.name, p.name)) return self.fail("duplicate name {s} in {s}", .{ p.name, a[1] });
349 }
350 }
351 for (ps) |p| for (f.bookings.items) |b| for (b.listed) |existing| {
352 if (existing.active and std.mem.eql(u8, existing.name, p.name)) return self.fail("{s} already on {s}", .{ p.name, f.name });
353 };
354 if (adult_count == 0) return self.fail("no adult in {s}", .{a[1]});
355 if (infant_count > adult_count) return self.fail("too many infants in {s}", .{a[1]});
356 if (f.booked(c) + ps.len - infant_count > f.limits[c]) return self.fail("limit exceeded for {s} {s}", .{ f.name, class_names[c] });
357 for (0..infant_count) |i| {
358 adults[i].infant = infants[i];
359 infants[i].guardian = adults[i];
360 }
361 const b = try self.gpa.create(Booking);
362 b.* = .{ .pnr = a[1], .flight = f, .listed = ps };
363 for (ps) |p| p.booking = b;
364 try self.bookings.put(self.gpa, b.pnr, b);
365 try f.bookings.append(self.gpa, b);
366 const method = f.assign(b, c);
367 try out.print("BOOKED {s} {s} {s} {s}", .{ b.pnr, f.name, class_names[c], method });
368 for (ps) |p| try out.print(" {s}:{s}", .{ p.name, try f.displaySeat(self.gpa, p) });
369 try out.writeByte('\n');
370 } else if (std.mem.eql(u8, op, "SEAT")) {
371 const b = try self.findBooking(a[1], true);
372 const p = try self.findPassenger(b, a[2]);
373 const f = b.flight;
374 if (!p.needsSeat()) return self.fail("infant {s} has no seat", .{p.name});
375 const found = f.aircraft.seatIndex(a[3]);
376 if (found == null or f.aircraft.seats.items[found.?].cabin != p.cabin)
377 return self.fail("no seat {s} in cabin {s}", .{ a[3], class_names[p.cabin] });
378 const seat = found.?;
379 if (f.blocked[seat]) return self.fail("seat {s} is blocked", .{a[3]});
380 if (f.occupied[seat] != null) return self.fail("seat {s} is occupied", .{a[3]});
381 const s = f.aircraft.seats.items[seat];
382 if ((p.kind == .CHILD or p.carrying()) and f.aircraft.exits[s.row]) return self.fail("exit row {d} not allowed for {s}", .{ s.row, p.name });
383 if (!f.fits(&.{p}, &.{seat}, p.cabin, null)) return self.fail("infant limit reached at {s}", .{a[3]});
384 const old = try f.displaySeat(self.gpa, p);
385 f.place(p, seat);
386 try out.print("MOVED {s} {s} {s} -> {s}\n", .{ b.pnr, p.name, old, a[3] });
387 try f.reassign(self.gpa, out);
388 } else if (std.mem.eql(u8, op, "SWAP")) {
389 const b = try self.findBooking(a[1], true);
390 const p = try self.findPassenger(b, a[2]);
391 const d = try self.findBooking(a[3], true);
392 const q = try self.findPassenger(d, a[4]);
393 if (p == q) return self.fail("cannot swap {s} with itself", .{p.name});
394 for ([_]*Passenger{ p, q }) |person| {
395 if (!person.needsSeat()) return self.fail("infant {s} has no seat", .{person.name});
396 }
397 if (b.flight != d.flight) return self.fail("{s} and {s} are on different flights", .{ b.pnr, d.pnr });
398 if (p.cabin != q.cabin) return self.fail("{s} and {s} are in different cabins", .{ p.name, q.name });
399 for ([_]*Passenger{ p, q }) |person| {
400 if (person.seat == null) return self.fail("{s} has no seat", .{person.name});
401 }
402 const f = b.flight;
403 const old = p.seat.?;
404 const other = q.seat.?;
405 if (!f.fits(&.{ p, q }, &.{ other, old }, p.cabin, null)) return self.fail("swap of {s} and {s} breaks seating rules", .{ p.name, q.name });
406 f.place(p, null);
407 f.place(q, null);
408 f.place(p, other);
409 f.place(q, old);
410 try out.print("SWAPPED {s} {s} {s} {s} {s} {s}\n", .{ b.pnr, p.name, try f.displaySeat(self.gpa, p), d.pnr, q.name, try f.displaySeat(self.gpa, q) });
411 try f.reassign(self.gpa, out);
412 } else if (std.mem.eql(u8, op, "CANCEL")) {
413 const b = try self.findBooking(a[1], true);
414 const f = b.flight;
415 const target: ?*Passenger = if (a.len == 2) null else try self.findPassenger(b, a[2]);
416 var removed: std.ArrayList(*Passenger) = .empty;
417 var remaining: usize = 0;
418 var adults: usize = 0;
419 for (b.listed) |p| {
420 if (!p.active) continue;
421 if (target == null or p == target or target.?.infant == p) try removed.append(self.gpa, p) else {
422 remaining += 1;
423 if (p.kind == .ADULT) adults += 1;
424 }
425 }
426 if (target != null and remaining > 0 and adults == 0) return self.fail("{s} would have no adult", .{b.pnr});
427 try out.print("CANCELLED {s}", .{b.pnr});
428 for (removed.items) |p| try out.print(" {s}", .{p.name});
429 try out.writeByte('\n');
430 for (removed.items) |p| f.remove(p);
431 try f.reassign(self.gpa, out);
432 } else if (std.mem.eql(u8, op, "CHECKIN") or std.mem.eql(u8, op, "VOLUNTEER")) {
433 const b = try self.findBooking(a[1], true);
434 const f = b.flight;
435 if (std.mem.eql(u8, op, "CHECKIN")) {
436 if (b.checked) return self.fail("{s} already checked in", .{b.pnr});
437 b.checked = true;
438 try out.print("CHECKIN {s} {d}\n", .{ b.pnr, b.count() });
439 } else {
440 const p = try self.findPassenger(b, a[2]);
441 if (p.kind != .ADULT or p.carrying()) return self.fail("{s} cannot volunteer", .{p.name});
442 if (p.seat == null) return self.fail("{s} has no seat", .{p.name});
443 if (std.mem.findScalar(*Passenger, f.volunteers.items, p) != null) return self.fail("{s} already volunteered", .{p.name});
444 try f.volunteers.append(self.gpa, p);
445 try out.print("VOLUNTEER {s} {s} {s}\n", .{ b.pnr, p.name, try f.displaySeat(self.gpa, p) });
446 }
447 } else if (std.mem.eql(u8, op, "UPGRADE")) {
448 const f = try self.findFlight(a[1], true);
449 var cs: [3]usize = undefined;
450 var len: usize = 0;
451 for (f.aircraft.cabins, 0..) |cabin, c| {
452 if (cabin != null) {
453 cs[len] = c;
454 len += 1;
455 }
456 }
457 var count: usize = 0;
458 for (0..len - 1) |i| {
459 const higher = cs[i];
460 const lower = cs[i + 1];
461 var targets: std.ArrayList(*Passenger) = .empty;
462 for (try f.allPassengers(self.gpa)) |p| {
463 if (p.cabin == lower and p.kind == .ADULT and !p.carrying() and self.tier(p.name) < 3) try targets.append(self.gpa, p);
464 }
465 std.mem.sort(*Passenger, targets.items, @as(*const Airline, self), tierLess);
466 for (targets.items) |p| {
467 if (f.booked(higher) >= f.limits[higher]) break;
468 const seat = f.choose(p, higher) orelse break;
469 const old = try f.displaySeat(self.gpa, p);
470 f.place(p, seat);
471 p.cabin = higher;
472 count += 1;
473 try out.print("UPGRADED {s} {s} {s} {s} -> {s} {s}\n", .{ p.booking.pnr, p.name, class_names[lower], old, class_names[higher], try f.displaySeat(self.gpa, p) });
474 }
475 }
476 try f.reassign(self.gpa, out);
477 try out.print("UPGRADE {s} {d}\n", .{ f.name, count });
478 } else if (std.mem.eql(u8, op, "CLOSE")) {
479 const f = try self.findFlight(a[1], true);
480 var denied: usize = 0;
481 var compensation: i64 = 0;
482 for (f.bookings.items) |b| {
483 if (b.count() == 0 or b.checked) continue;
484 try out.print("NOSHOW {s} {d}\n", .{ b.pnr, b.count() });
485 for (b.listed) |p| {
486 if (p.active) f.remove(p);
487 }
488 }
489 try f.reassign(self.gpa, out);
490 for (0..3) |c| {
491 var targets: std.ArrayList(*Passenger) = .empty;
492 for (f.volunteers.items) |p| {
493 if (p.active and p.seat != null and p.cabin == c) try targets.append(self.gpa, p);
494 }
495 for (targets.items) |p| {
496 var useful = false;
497 for (try f.allPassengers(self.gpa)) |q| {
498 if (q.needsSeat() and q.seat == null and q.cabin == c and f.fits(&.{q}, &.{p.seat.?}, c, p)) {
499 useful = true;
500 break;
501 }
502 }
503 if (useful) {
504 compensation += try self.deny(f, p, true, out);
505 denied += 1;
506 try f.reassign(self.gpa, out);
507 }
508 }
509 }
510 for (try f.allPassengers(self.gpa)) |p| {
511 if (p.active and p.needsSeat() and p.seat == null) {
512 compensation += try self.deny(f, p, false, out);
513 denied += 1;
514 }
515 }
516 f.closed = true;
517 try out.print("CLOSED {s} denied {d} compensation {d}\n", .{ f.name, denied, compensation });
518 } else if (std.mem.eql(u8, op, "BLOCK") or std.mem.eql(u8, op, "UNBLOCK")) {
519 const f = try self.findFlight(a[1], true);
520 const seat = f.aircraft.seatIndex(a[2]) orelse return self.fail("no seat {s} on {s}", .{ a[2], f.name });
521 const blocking = std.mem.eql(u8, op, "BLOCK");
522 if (blocking and f.blocked[seat]) return self.fail("seat {s} is blocked", .{a[2]});
523 if (!blocking and !f.blocked[seat]) return self.fail("seat {s} is not blocked", .{a[2]});
524 try out.print("{s} {s} {s}\n", .{ op, f.name, a[2] });
525 f.blocked[seat] = blocking;
526 if (blocking) {
527 if (f.occupied[seat]) |p| {
528 f.place(p, null);
529 f.place(p, f.choose(p, p.cabin));
530 try out.print("DISPLACED {s} {s} {s} -> {s}\n", .{ p.booking.pnr, p.name, a[2], try f.displaySeat(self.gpa, p) });
531 }
532 }
533 try f.reassign(self.gpa, out);
534 } else if (std.mem.eql(u8, op, "INFO")) {
535 const b = try self.findBooking(a[1], false);
536 const f = b.flight;
537 try out.print("INFO {s} {s} {d} {s}\n", .{ b.pnr, f.name, b.count(), @as([]const u8, if (b.checked) "CHECKED" else "NOT-CHECKED") });
538 for (b.listed) |p| {
539 if (p.active) try out.print("{s} {s} {s} {s}\n", .{ p.name, @tagName(p.kind), class_names[p.cabin], try f.displaySeat(self.gpa, p) });
540 }
541 } else if (std.mem.eql(u8, op, "FIND")) {
542 var found: std.ArrayList(*Passenger) = .empty;
543 for (self.flights.values()) |f| for (try f.allPassengers(self.gpa)) |p| {
544 if (std.mem.eql(u8, p.name, a[1])) try found.append(self.gpa, p);
545 };
546 try out.print("FIND {s} {s} {d}\n", .{ a[1], tier_names[self.tier(a[1])], found.items.len });
547 for (found.items) |p| {
548 const f = p.booking.flight;
549 try out.print("{s} {s} {s} {s} {s}\n", .{ f.name, p.booking.pnr, class_names[p.cabin], try f.displaySeat(self.gpa, p), @as([]const u8, if (f.closed) "CLOSED" else "OPEN") });
550 }
551 } else {
552 const f = try self.findFlight(a[1], false);
553 const model = f.aircraft;
554 if (std.mem.eql(u8, op, "MAP")) {
555 try out.print("MAP {s}\n", .{f.name});
556 for (model.cabins, 0..) |optional, c| {
557 const cabin = optional orelse continue;
558 try out.print("CABIN {s}\n ", .{class_names[c]});
559 for (cabin.blocks, 0..) |block, b| {
560 if (b > 0) try out.writeByte(' ');
561 try out.writeAll(block);
562 }
563 try out.writeByte('\n');
564 for (cabin.first..cabin.last + 1) |row| {
565 try out.print("{d:>2} ", .{row});
566 for (cabin.blocks, 0..) |block, b| {
567 if (b > 0) try out.writeByte(' ');
568 for (block) |letter| {
569 const seat = model.grid[row][letter - 'A'].?;
570 try out.writeByte(if (f.blocked[seat]) '#' else if (f.occupied[seat] != null) 'X' else if (model.exits[row]) 'E' else '.');
571 }
572 }
573 try out.writeByte('\n');
574 }
575 }
576 } else if (std.mem.eql(u8, op, "MANIFEST")) {
577 var seated: std.ArrayList(*Passenger) = .empty;
578 var waiting: std.ArrayList(*Passenger) = .empty;
579 var infants: usize = 0;
580 for (try f.allPassengers(self.gpa)) |p| {
581 if (!p.needsSeat()) infants += 1 else if (p.seat == null) try waiting.append(self.gpa, p) else try seated.append(self.gpa, p);
582 }
583 std.mem.sort(*Passenger, seated.items, {}, seatLess);
584 try out.print("MANIFEST {s} seated {d} unassigned {d} infants {d}\n", .{ f.name, seated.items.len, waiting.items.len, infants });
585 try seated.appendSlice(self.gpa, waiting.items);
586 for (seated.items) |p| {
587 try out.print("{s} {s} {s} {s} {s}", .{ if (p.seat == null) "---" else try f.displaySeat(self.gpa, p), class_names[p.cabin], p.booking.pnr, p.name, @tagName(p.kind) });
588 if (p.carrying()) try out.print(" +{s}", .{p.infant.?.name});
589 if (self.tier(p.name) < 3) try out.print(" {s}", .{tier_names[self.tier(p.name)]});
590 if (p.booking.checked) try out.writeAll(" CHECKED");
591 if (std.mem.findScalar(*Passenger, f.volunteers.items, p) != null) try out.writeAll(" VOL");
592 try out.writeByte('\n');
593 }
594 } else if (std.mem.eql(u8, op, "LOADFACTOR")) {
595 try out.print("LOADFACTOR {s}\n", .{f.name});
596 var total_seated: usize = 0;
597 var total_seats: usize = 0;
598 for (model.cabins, 0..) |cabin, c| {
599 if (cabin == null) continue;
600 var seats: usize = 0;
601 var seated: usize = 0;
602 for (model.seats.items, 0..) |s, i| {
603 if (s.cabin == c and !f.blocked[i]) {
604 seats += 1;
605 if (f.occupied[i] != null) seated += 1;
606 }
607 }
608 total_seated += seated;
609 total_seats += seats;
610 try out.print("{s} {d}/{d} {d}.{d}% booked {d}/{d}\n", .{ class_names[c], seated, seats, (seated * 1000 / seats) / 10, (seated * 1000 / seats) % 10, f.booked(c), f.limits[c] });
611 }
612 try out.print("ALL {d}/{d} {d}.{d}%\n", .{ total_seated, total_seats, (total_seated * 1000 / total_seats) / 10, (total_seated * 1000 / total_seats) % 10 });
613 } else if (std.mem.eql(u8, op, "BALANCE")) {
614 var weight: i64 = 0;
615 var moment: i64 = 0;
616 for (try f.allPassengers(self.gpa)) |p| {
617 const seat = p.seat orelse continue;
618 const w: i64 = @as(i64, if (p.kind == .ADULT) 80 else 40) + @as(i64, if (p.carrying()) 10 else 0);
619 weight += w;
620 moment += w * (@as(i64, @intCast(model.seats.items[seat].row)) - @as(i64, @intCast(model.base)));
621 }
622 try out.print("BALANCE {s} weight {d} moment {d} limit {d} {s}\n", .{ f.name, weight, moment, model.limit, @as([]const u8, if (moment < -model.limit) "NOSE-HEAVY" else if (moment > model.limit) "TAIL-HEAVY" else "OK") });
623 } else if (std.mem.eql(u8, op, "AVAIL")) {
624 const c = classIndex(a[2]);
625 try self.hasCabin(f, c);
626 var counts = [_]usize{ 0, 0, 0, 0, 0 };
627 for (model.seats.items, 0..) |s, i| {
628 if (s.cabin != c or f.blocked[i] or f.occupied[i] != null) continue;
629 counts[0] += 1;
630 counts[1 + std.mem.findScalar(u8, "WMA", s.kind).?] += 1;
631 if (model.exits[s.row]) counts[4] += 1;
632 }
633 try out.print("AVAIL {s} {s} free {d} W {d} M {d} A {d} EXIT {d}\n", .{ f.name, class_names[c], counts[0], counts[1], counts[2], counts[3], counts[4] });
634 } else if (std.mem.eql(u8, op, "BOARDING")) {
635 const Entry = struct {
636 passenger: *Passenger,
637 key: [5]i64,
638
639 fn less(_: void, a: @This(), b: @This()) bool {
640 return std.mem.order(i64, &a.key, &b.key) == .lt;
641 }
642 };
643 var ordered: std.ArrayList(Entry) = .empty;
644 for (try f.allPassengers(self.gpa)) |p| {
645 if (p.seat != null)
646 try ordered.append(self.gpa, .{ .passenger = p, .key = self.boardingKey(p) });
647 }
648 std.mem.sort(Entry, ordered.items, {}, Entry.less);
649 try out.print("BOARDING {s} {d}\n", .{ f.name, ordered.items.len });
650 for (ordered.items, 0..) |entry, i| {
651 const p = entry.passenger;
652 const key = entry.key;
653 const group = if (key[0] == 3) try std.fmt.allocPrint(self.gpa, "ZONE{d}", .{-key[1]}) else ([_][]const u8{ "PRE", "CABIN", "TIER" })[@intCast(key[0])];
654 try out.print("{d} {s} {s} {s} {s}\n", .{ i + 1, group, try f.displaySeat(self.gpa, p), p.booking.pnr, p.name });
655 }
656 }
657 }
658 }
659 fn boardingKey(self: *const Airline, p: *Passenger) [5]i64 {
660 const model = p.booking.flight.aircraft;
661 const s = model.seats.items[p.seat.?];
662 const row: i64 = @intCast(s.row);
663 if (p.booking.hasInfants()) return .{ 0, 0, row, 0, s.letter };
664 if (p.cabin < 2) return .{ 1, 0, row, 0, s.letter };
665 if (self.tier(p.name) < 2) return .{ 2, 0, row, 0, s.letter };
666 const zone: i64 = @intCast((s.row - model.cabins[2].?.first) / 5 + 1);
667 return .{ 3, -zone, @intCast(std.mem.findScalar(u8, "WMA", s.kind).?), -row, s.letter };
668 }
669};
670fn tierLess(self: *const Airline, a: *Passenger, b: *Passenger) bool {
671 return self.tier(a.name) < self.tier(b.name);
672}
673fn seatLess(_: void, a: *Passenger, b: *Passenger) bool {
674 return a.seat.? < b.seat.?;
675}
676pub fn main(init: std.process.Init) !void {
677 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
678 defer arena.deinit();
679 const gpa = arena.allocator();
680 var input_buffer: [4096]u8 = undefined;
681 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
682 var output_buffer: [8192]u8 = undefined;
683 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
684 var airline = Airline{ .gpa = gpa };
685 const count = integer((try input.interface.takeDelimiter('\n')).?);
686 for (0..count) |_| {
687 const line = try gpa.dupe(u8, (try input.interface.takeDelimiter('\n')).?);
688 var draft: std.Io.Writer.Allocating = .init(gpa);
689 airline.execute(line, &draft.writer) catch |err| {
690 if (err == error.BadCommand) {
691 try output.interface.print("ERROR {s}\n", .{airline.error_buffer[0..airline.error_len]});
692 continue;
693 }
694 return err;
695 };
696 try output.interface.writeAll(draft.writer.buffered());
697 }
698 try output.interface.flush();
699}
