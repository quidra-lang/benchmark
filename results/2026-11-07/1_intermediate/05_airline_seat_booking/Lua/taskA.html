1local CLASSES = {'F', 'J', 'Y'}
2local TIERS = {PLATINUM = 0, GOLD = 1, SILVER = 2, NONE = 3}
3local aircraft, flights, bookings, used_pnr, people, statuses = {}, {}, {}, {}, {}, {}
4local flight_order, definition = {}, nil
5local function fail(message) error(message, 0) end
6local function words(text, pattern)
7 local values = {}
8 for value in text:gmatch(pattern or '%S+') do values[#values + 1] = value end
9 return values
10end
11local function append(target, items)
12 table.move(items, 1, #items, #target + 1, target)
13end
14local function copy_set(values)
15 local result = {}
16 for key, value in pairs(values) do result[key] = value end
17 return result
18end
19local function tier(p) return statuses[p.name] or 'NONE' end
20local function active_ids(ids)
21 local result = {}
22 for _, id in ipairs(ids) do
23 if people[id].active then result[#result + 1] = id end
24 end
25 return result
26end
27local function members(b) return active_ids(b.people) end
28local function passengers(f) return active_ids(f.people) end
29local function passenger_result(p)
30 if p.type == 'INFANT' then return 'LAP' end
31 return p.seat == '' and 'UNASSIGNED' or p.seat
32end
33local function get_flight(id, open)
34 local f = flights[id]
35 if not f then fail('unknown flight ' .. id) end
36 if open and f.closed then fail('flight ' .. id .. ' is closed') end
37 return f
38end
39local function get_booking(pnr, open)
40 local b = bookings[pnr]
41 if not b or #members(b) == 0 then fail('unknown PNR ' .. pnr) end
42 get_flight(b.flight, open)
43 return b
44end
45local function get_person(b, name)
46 for _, id in ipairs(members(b)) do
47 if people[id].name == name then return people[id] end
48 end
49 fail('no passenger ' .. name .. ' in ' .. b.pnr)
50end
51local function cabin(f, class)
52 local c = f.model.cabins[class]
53 if not c then fail('no cabin ' .. class .. ' on ' .. f.id) end
54 return c
55end
56local function booked(f, class)
57 local count = 0
58 for _, id in ipairs(passengers(f)) do
59 local p = people[id]
60 if p.type ~= 'INFANT' and p.class == class then count = count + 1 end
61 end
62 return count
63end
64local function allowed(f, p, seat, class, vacated, assignments)
65 vacated, assignments = vacated or {}, assignments or {}
66 local s = f.model.seats[seat]
67 if not s or s.class ~= class or f.blocked[seat] then return false end
68 if f.occupied[seat] and not vacated[f.occupied[seat]] then return false end
69 if (p.type == 'CHILD' or p.infant ~= 0) and f.model.exits[s.row] then return false end
70 if p.infant ~= 0 then
71 for other_seat, id in pairs(f.occupied) do
72 if id ~= p.id and not vacated[id] and people[id].infant ~= 0 then
73 local t = f.model.seats[other_seat]
74 if t.row == s.row and t.block == s.block then return false end
75 end
76 end
77 for id, other_seat in pairs(assignments) do
78 if id ~= p.id and people[id].infant ~= 0 then
79 local t = f.model.seats[other_seat]
80 if t.row == s.row and t.block == s.block then return false end
81 end
82 end
83 end
84 return true
85end
86local function select_seat(f, p, class)
87 class = class or p.class
88 local first = ''
89 for _, seat in ipairs(f.model.cabins[class].seats) do
90 if allowed(f, p, seat, class) then
91 if first == '' then first = seat end
92 if p.pref ~= '' and f.model.seats[seat].kind == p.pref then return seat end
93 end
94 end
95 return first
96end
97local function move_to(f, p, seat)
98 if p.seat ~= '' then f.occupied[p.seat] = nil end
99 p.seat = seat
100 if seat ~= '' then f.occupied[seat] = p.id end
101end
102local function reassign(f, output)
103 for _, id in ipairs(passengers(f)) do
104 local p = people[id]
105 if p.type ~= 'INFANT' and p.seat == '' then
106 local seat = select_seat(f, p)
107 if seat ~= '' then
108 move_to(f, p, seat)
109 output[#output + 1] = 'REASSIGNED ' .. p.pnr .. ' ' .. p.name .. ' ' .. seat
110 end
111 end
112 end
113end
114local function fits(f, ids, seats, class)
115 local assignments = {}
116 for index, id in ipairs(ids) do assignments[id] = seats[index] end
117 for index, id in ipairs(ids) do
118 if not allowed(f, people[id], seats[index], class, {}, assignments) then return false end
119 end
120 return true
121end
122local function group_seats(f, ids, class)
123 local c, total = f.model.cabins[class], #ids
124 for _, method in ipairs({'BLOCK', 'ROW'}) do
125 for row = c.from, c.to do
126 local blocks = c.blocks
127 if method == 'ROW' then blocks = {table.concat(c.blocks)} end
128 for _, block in ipairs(blocks) do
129 for x = 1, #block - total + 1 do
130 local seats = {}
131 for j = 0, total - 1 do seats[#seats + 1] = row .. block:sub(x + j, x + j) end
132 if fits(f, ids, seats, class) then return method, seats end
133 end
134 end
135 end
136 end
137 for height = 2, total do
138 local width = (total + height - 1) // height
139 local rows = (total + width - 1) // width
140 for row = c.from, c.to - rows + 1 do
141 for _, block in ipairs(c.blocks) do
142 for x = 1, #block - width + 1 do
143 local seats = {}
144 for j = 0, total - 1 do
145 local position = x + j % width
146 seats[#seats + 1] = (row + j // width) .. block:sub(position, position)
147 end
148 if fits(f, ids, seats, class) then return 'STACK', seats end
149 end
150 end
151 end
152 end
153 return 'SPLIT', {}
154end
155local function removal_targets(p)
156 if p.infant == 0 then return {p.id} end
157 return {p.id, p.infant}
158end
159local function remove_people(f, ids)
160 for _, id in ipairs(ids) do
161 local p = people[id]
162 if p.active then
163 move_to(f, p, '')
164 p.active, p.volunteer = false, false
165 if p.type == 'INFANT' then people[p.adult].infant = 0 end
166 end
167 end
168end
169local function load_text(seated, seats)
170 local q = (seated * 1000) // seats
171 return (q // 10) .. '.' .. (q % 10) .. '%'
172end
173local function seat_less(f, a, b)
174 local x, y = f.model.seats[people[a].seat], f.model.seats[people[b].seat]
175 if x.row ~= y.row then return x.row < y.row end
176 return x.letter < y.letter
177end
178local function tuple_less(a, b)
179 for index = 1, math.min(#a, #b) do
180 if a[index] ~= b[index] then return a[index] < b[index] end
181 end
182 return #a < #b
183end
184local function execute(line)
185 local t, output = words(line), {}
186 local command = t[1]
187 local function emit(text) output[#output + 1] = text end
188 if command == 'AIRCRAFT' then
189 definition = {id = t[2], base = tonumber(t[3]), limit = tonumber(t[4]), cabins = {}, seats = {}, exits = {}, blocked = {}}
190 elseif command == 'CABIN' then
191 local class, blocks = t[2], {table.unpack(t, 5)}
192 local c = {from = tonumber(t[3]), to = tonumber(t[4]), blocks = blocks, seats = {}}
193 for row = c.from, c.to do
194 for b, block in ipairs(blocks) do
195 for x = 1, #block do
196 local kind
197 if (b == 1 and x == 1) or (b == #blocks and x == #block) then
198 kind = 'W'
199 elseif x == 1 or x == #block then
200 kind = 'A'
201 else
202 kind = 'M'
203 end
204 local letter = block:sub(x, x)
205 local seat = row .. letter
206 definition.seats[seat] = {row = row, letter = letter, class = class, block = b, kind = kind}
207 c.seats[#c.seats + 1] = seat
208 end
209 end
210 end
211 definition.cabins[class] = c
212 elseif command == 'EXIT' then
213 for index = 2, #t do definition.exits[tonumber(t[index])] = true end
214 elseif command == 'BLOCKED' then
215 for index = 2, #t do definition.blocked[t[index]] = true end
216 elseif command == 'END' then
217 local parts = {'AIRCRAFT', definition.id}
218 for _, class in ipairs(CLASSES) do
219 local c = definition.cabins[class]
220 if c then
221 c.capacity = 0
222 for _, seat in ipairs(c.seats) do
223 if not definition.blocked[seat] then c.capacity = c.capacity + 1 end
224 end
225 append(parts, {class, c.capacity})
226 end
227 end
228 aircraft[definition.id] = definition
229 emit(table.concat(parts, ' '))
230 definition = nil
231 elseif command == 'FLIGHT' then
232 if flights[t[2]] then fail('duplicate flight ' .. t[2]) end
233 local model = aircraft[t[3]]
234 if not model then fail('unknown aircraft ' .. t[3]) end
235 local f = {id = t[2], model = model, blocked = copy_set(model.blocked), occupied = {}, limits = {}, closed = false, bookings = {}, people = {}, volunteers = {}}
236 local parts = {'FLIGHT', f.id, model.id}
237 for _, class in ipairs(CLASSES) do
238 local c = model.cabins[class]
239 if c then
240 f.limits[class] = (c.capacity * (100 + tonumber(t[4]))) // 100
241 append(parts, {class, c.capacity .. '/' .. f.limits[class]})
242 end
243 end
244 flights[f.id] = f
245 flight_order[#flight_order + 1] = f.id
246 emit(table.concat(parts, ' '))
247 elseif command == 'STATUS' then
248 statuses[t[2]] = t[3]
249 emit('STATUS ' .. t[2] .. ' ' .. t[3])
250 elseif command == 'BOOK' then
251 local pnr, flight, class = t[2], t[3], t[4]
252 if used_pnr[pnr] then fail('duplicate PNR ' .. pnr) end
253 local f = get_flight(flight, true)
254 cabin(f, class)
255 local parsed, names = {}, {}
256 for index = 5, #t do
257 local parts = words(t[index], '[^:]+')
258 parsed[#parsed + 1] = parts
259 if names[parts[1]] then fail('duplicate name ' .. parts[1] .. ' in ' .. pnr) end
260 names[parts[1]] = true
261 end
262 local aboard = {}
263 for _, id in ipairs(passengers(f)) do aboard[people[id].name] = true end
264 local adults, infants = 0, 0
265 for _, parts in ipairs(parsed) do
266 if aboard[parts[1]] then fail(parts[1] .. ' already on ' .. flight) end
267 if parts[2] == 'ADULT' then adults = adults + 1 end
268 if parts[2] == 'INFANT' then infants = infants + 1 end
269 end
270 if adults == 0 then fail('no adult in ' .. pnr) end
271 if infants > adults then fail('too many infants in ' .. pnr) end
272 if booked(f, class) + #parsed - infants > f.limits[class] then fail('limit exceeded for ' .. flight .. ' ' .. class) end
273 local b = {pnr = pnr, flight = flight, people = {}, checked = false}
274 local adult_ids, infant_ids, seat_ids = {}, {}, {}
275 for _, parts in ipairs(parsed) do
276 local id = #people + 1
277 local p = {id = id, pnr = pnr, name = parts[1], type = parts[2], pref = parts[3] or '', class = class,
278 seat = '', active = true, infant = 0, adult = 0, volunteer = false}
279 people[id] = p
280 b.people[#b.people + 1] = id
281 f.people[#f.people + 1] = id
282 if p.type == 'ADULT' then adult_ids[#adult_ids + 1] = id end
283 if p.type == 'INFANT' then infant_ids[#infant_ids + 1] = id else seat_ids[#seat_ids + 1] = id end
284 end
285 for index, id in ipairs(infant_ids) do
286 people[id].adult = adult_ids[index]
287 people[adult_ids[index]].infant = id
288 end
289 bookings[pnr], used_pnr[pnr] = b, true
290 f.bookings[#f.bookings + 1] = pnr
291 local method, seats
292 if #seat_ids == 1 then
293 method = 'SINGLE'
294 local p = people[seat_ids[1]]
295 move_to(f, p, select_seat(f, p))
296 else
297 method, seats = group_seats(f, seat_ids, class)
298 for index, id in ipairs(seat_ids) do
299 local seat = seats[index]
300 if method == 'SPLIT' then seat = select_seat(f, people[id]) end
301 move_to(f, people[id], seat)
302 end
303 end
304 local parts = {'BOOKED', pnr, flight, class, method}
305 for _, id in ipairs(b.people) do parts[#parts + 1] = people[id].name .. ':' .. passenger_result(people[id]) end
306 emit(table.concat(parts, ' '))
307 elseif command == 'FIND' then
308 local lines = {}
309 for _, flight in ipairs(flight_order) do
310 local f = flights[flight]
311 for _, id in ipairs(passengers(f)) do
312 local p = people[id]
313 if p.name == t[2] then
314 lines[#lines + 1] = flight .. ' ' .. p.pnr .. ' ' .. p.class .. ' ' .. passenger_result(p) .. ' ' .. (f.closed and 'CLOSED' or 'OPEN')
315 end
316 end
317 end
318 emit('FIND ' .. t[2] .. ' ' .. (statuses[t[2]] or 'NONE') .. ' ' .. #lines)
319 append(output, lines)
320 elseif command == 'SEAT' or command == 'SWAP' or command == 'CANCEL' or command == 'CHECKIN' or command == 'VOLUNTEER' or command == 'INFO' then
321 local b = get_booking(t[2], command ~= 'INFO')
322 local f = flights[b.flight]
323 if command == 'INFO' then
324 emit('INFO ' .. b.pnr .. ' ' .. b.flight .. ' ' .. #members(b) .. ' ' .. (b.checked and 'CHECKED' or 'NOT-CHECKED'))
325 for _, id in ipairs(members(b)) do
326 local p = people[id]
327 emit(p.name .. ' ' .. p.type .. ' ' .. p.class .. ' ' .. passenger_result(p))
328 end
329 elseif command == 'CHECKIN' then
330 if b.checked then fail(b.pnr .. ' already checked in') end
331 b.checked = true
332 emit('CHECKIN ' .. b.pnr .. ' ' .. #members(b))
333 elseif command == 'CANCEL' then
334 local targets = members(b)
335 if t[3] then
336 local p = get_person(b, t[3])
337 local target_set = {}
338 for _, id in ipairs(removal_targets(p)) do target_set[id] = true end
339 local remaining, adult = 0, false
340 targets = {}
341 for _, id in ipairs(members(b)) do
342 if target_set[id] then
343 targets[#targets + 1] = id
344 else
345 remaining = remaining + 1
346 if people[id].type == 'ADULT' then adult = true end
347 end
348 end
349 if remaining > 0 and not adult then fail(b.pnr .. ' would have no adult') end
350 end
351 local names = {}
352 for _, id in ipairs(targets) do names[#names + 1] = people[id].name end
353 remove_people(f, targets)
354 emit('CANCELLED ' .. b.pnr .. ' ' .. table.concat(names, ' '))
355 reassign(f, output)
356 else
357 local p = get_person(b, t[3])
358 if command == 'SWAP' then
359 local b2 = get_booking(t[4], true)
360 local q = get_person(b2, t[5])
361 if p.id == q.id then fail('cannot swap ' .. p.name .. ' with itself') end
362 for _, passenger in ipairs({p, q}) do
363 if passenger.type == 'INFANT' then fail('infant ' .. passenger.name .. ' has no seat') end
364 end
365 if b.flight ~= b2.flight then fail(b.pnr .. ' and ' .. b2.pnr .. ' are on different flights') end
366 if p.class ~= q.class then fail(p.name .. ' and ' .. q.name .. ' are in different cabins') end
367 for _, passenger in ipairs({p, q}) do
368 if passenger.seat == '' then fail(passenger.name .. ' has no seat') end
369 end
370 local vacated = {[p.id] = true, [q.id] = true}
371 local assignments = {[p.id] = q.seat, [q.id] = p.seat}
372 if not allowed(f, p, q.seat, p.class, vacated, assignments) or not allowed(f, q, p.seat, q.class, vacated, assignments) then
373 fail('swap of ' .. p.name .. ' and ' .. q.name .. ' breaks seating rules')
374 end
375 local old_p, old_q = p.seat, q.seat
376 move_to(f, p, '')
377 move_to(f, q, '')
378 move_to(f, p, old_q)
379 move_to(f, q, old_p)
380 emit('SWAPPED ' .. b.pnr .. ' ' .. p.name .. ' ' .. p.seat .. ' ' .. b2.pnr .. ' ' .. q.name .. ' ' .. q.seat)
381 reassign(f, output)
382 elseif command == 'VOLUNTEER' then
383 if p.type ~= 'ADULT' or p.infant ~= 0 then fail(p.name .. ' cannot volunteer') end
384 if p.seat == '' then fail(p.name .. ' has no seat') end
385 if p.volunteer then fail(p.name .. ' already volunteered') end
386 p.volunteer = true
387 f.volunteers[#f.volunteers + 1] = p.id
388 emit('VOLUNTEER ' .. p.pnr .. ' ' .. p.name .. ' ' .. p.seat)
389 else
390 local seat = t[4]
391 if p.type == 'INFANT' then fail('infant ' .. p.name .. ' has no seat') end
392 local s = f.model.seats[seat]
393 if not s or s.class ~= p.class then fail('no seat ' .. seat .. ' in cabin ' .. p.class) end
394 if f.blocked[seat] then fail('seat ' .. seat .. ' is blocked') end
395 if f.occupied[seat] then fail('seat ' .. seat .. ' is occupied') end
396 if (p.type == 'CHILD' or p.infant ~= 0) and f.model.exits[s.row] then fail('exit row ' .. s.row .. ' not allowed for ' .. p.name) end
397 if not allowed(f, p, seat, p.class) then fail('infant limit reached at ' .. seat) end
398 local old = p.seat
399 move_to(f, p, seat)
400 emit('MOVED ' .. p.pnr .. ' ' .. p.name .. ' ' .. old .. ' -> ' .. seat)
401 reassign(f, output)
402 end
403 end
404 else
405 local mutating = command == 'UPGRADE' or command == 'CLOSE' or command == 'BLOCK' or command == 'UNBLOCK'
406 local f = get_flight(t[2], mutating)
407 local classes = {}
408 for _, class in ipairs(CLASSES) do
409 if f.model.cabins[class] then classes[#classes + 1] = class end
410 end
411 if command == 'UPGRADE' then
412 local total = 0
413 for index = 1, #classes - 1 do
414 local higher, lower = classes[index], classes[index + 1]
415 local targets = {}
416 for _, id in ipairs(passengers(f)) do
417 local p = people[id]
418 if p.class == lower and p.type == 'ADULT' and p.infant == 0 and tier(p) ~= 'NONE' then targets[#targets + 1] = id end
419 end
420 table.sort(targets, function(a, b)
421 local rank_a, rank_b = TIERS[tier(people[a])], TIERS[tier(people[b])]
422 if rank_a ~= rank_b then return rank_a < rank_b end
423 return a < b
424 end)
425 for _, id in ipairs(targets) do
426 if booked(f, higher) >= f.limits[higher] then break end
427 local p = people[id]
428 local seat = select_seat(f, p, higher)
429 if seat == '' then break end
430 local old = passenger_result(p)
431 move_to(f, p, seat)
432 p.class = higher
433 total = total + 1
434 emit('UPGRADED ' .. p.pnr .. ' ' .. p.name .. ' ' .. lower .. ' ' .. old .. ' -> ' .. higher .. ' ' .. seat)
435 end
436 end
437 reassign(f, output)
438 emit('UPGRADE ' .. f.id .. ' ' .. total)
439 elseif command == 'CLOSE' then
440 for _, pnr in ipairs(f.bookings) do
441 local b = bookings[pnr]
442 local ids = members(b)
443 if #ids > 0 and not b.checked then
444 emit('NOSHOW ' .. pnr .. ' ' .. #ids)
445 remove_people(f, ids)
446 end
447 end
448 reassign(f, output)
449 local denied, total = 0, 0
450 local rates = {F = 500, J = 300, Y = 150}
451 for _, class in ipairs(classes) do
452 local targets = {}
453 for _, id in ipairs(f.volunteers) do
454 local p = people[id]
455 if p.active and p.volunteer and p.seat ~= '' and p.class == class then targets[#targets + 1] = id end
456 end
457 for _, id in ipairs(targets) do
458 local p, useful = people[id], false
459 for _, waiting_id in ipairs(passengers(f)) do
460 local waiting = people[waiting_id]
461 if waiting.type ~= 'INFANT' and waiting.class == class and waiting.seat == '' and allowed(f, waiting, p.seat, class, {[id] = true}) then
462 useful = true
463 break
464 end
465 end
466 if useful then
467 local amount = rates[class]
468 emit('DENIED VOLUNTARY ' .. p.pnr .. ' ' .. p.name .. ' ' .. class .. ' ' .. amount)
469 remove_people(f, {id})
470 denied, total = denied + 1, total + amount
471 reassign(f, output)
472 end
473 end
474 end
475 for _, id in ipairs(passengers(f)) do
476 local p = people[id]
477 if p.active and p.type ~= 'INFANT' and p.seat == '' then
478 local amount = rates[p.class] * 2
479 if p.type == 'CHILD' then amount = amount // 2 end
480 emit('DENIED INVOLUNTARY ' .. p.pnr .. ' ' .. p.name .. ' ' .. p.class .. ' ' .. amount)
481 remove_people(f, removal_targets(p))
482 denied, total = denied + 1, total + amount
483 end
484 end
485 f.closed = true
486 emit('CLOSED ' .. f.id .. ' denied ' .. denied .. ' compensation ' .. total)
487 elseif command == 'BLOCK' or command == 'UNBLOCK' then
488 local seat = t[3]
489 if not f.model.seats[seat] then fail('no seat ' .. seat .. ' on ' .. f.id) end
490 if command == 'BLOCK' and f.blocked[seat] then fail('seat ' .. seat .. ' is blocked') end
491 if command == 'UNBLOCK' and not f.blocked[seat] then fail('seat ' .. seat .. ' is not blocked') end
492 emit(command .. ' ' .. f.id .. ' ' .. seat)
493 if command == 'BLOCK' then
494 f.blocked[seat] = true
495 if f.occupied[seat] then
496 local p = people[f.occupied[seat]]
497 move_to(f, p, '')
498 move_to(f, p, select_seat(f, p))
499 emit('DISPLACED ' .. p.pnr .. ' ' .. p.name .. ' ' .. seat .. ' -> ' .. passenger_result(p))
500 end
501 else
502 f.blocked[seat] = nil
503 end
504 reassign(f, output)
505 elseif command == 'MAP' then
506 emit('MAP ' .. f.id)
507 for _, class in ipairs(classes) do
508 local c = f.model.cabins[class]
509 emit('CABIN ' .. class)
510 emit(' ' .. table.concat(c.blocks, ' '))
511 for row = c.from, c.to do
512 local blocks = {}
513 for _, block in ipairs(c.blocks) do
514 local marks = {}
515 for x = 1, #block do
516 local seat, mark = row .. block:sub(x, x), nil
517 if f.blocked[seat] then
518 mark = '#'
519 elseif f.occupied[seat] then
520 mark = 'X'
521 else
522 mark = f.model.exits[row] and 'E' or '.'
523 end
524 marks[#marks + 1] = mark
525 end
526 blocks[#blocks + 1] = table.concat(marks)
527 end
528 emit(string.format('%2d ', row) .. table.concat(blocks, ' '))
529 end
530 end
531 elseif command == 'MANIFEST' or command == 'BOARDING' then
532 local seated, unassigned, infants = {}, {}, 0
533 for _, id in ipairs(passengers(f)) do
534 local p = people[id]
535 if p.type == 'INFANT' then
536 infants = infants + 1
537 elseif p.seat == '' then
538 unassigned[#unassigned + 1] = id
539 else
540 seated[#seated + 1] = id
541 end
542 end
543 if command == 'MANIFEST' then
544 table.sort(seated, function(a, b) return seat_less(f, a, b) end)
545 emit('MANIFEST ' .. f.id .. ' seated ' .. #seated .. ' unassigned ' .. #unassigned .. ' infants ' .. infants)
546 local ids = {table.unpack(seated)}
547 append(ids, unassigned)
548 for _, id in ipairs(ids) do
549 local p = people[id]
550 local parts = {p.seat == '' and '---' or p.seat, p.class, p.pnr, p.name, p.type}
551 if p.infant ~= 0 then parts[#parts + 1] = '+' .. people[p.infant].name end
552 if tier(p) ~= 'NONE' then parts[#parts + 1] = tier(p) end
553 if bookings[p.pnr].checked then parts[#parts + 1] = 'CHECKED' end
554 if p.volunteer then parts[#parts + 1] = 'VOL' end
555 emit(table.concat(parts, ' '))
556 end
557 else
558 local groups, keys = {}, {}
559 for _, id in ipairs(seated) do
560 local p = people[id]
561 local s = f.model.seats[p.seat]
562 local has_infant = false
563 for _, member in ipairs(members(bookings[p.pnr])) do
564 if people[member].type == 'INFANT' then has_infant = true end
565 end
566 local group, key
567 if has_infant then
568 group, key = 'PRE', {0, s.row, s.letter}
569 elseif p.class ~= 'Y' then
570 group, key = 'CABIN', {1, s.row, s.letter}
571 elseif TIERS[tier(p)] <= 1 then
572 group, key = 'TIER', {2, s.row, s.letter}
573 else
574 local zone = (s.row - f.model.cabins.Y.from) // 5 + 1
575 group = 'ZONE' .. zone
576 key = {3, -zone, ({W = 0, M = 1, A = 2})[s.kind], -s.row, s.letter}
577 end
578 groups[id], keys[id] = group, key
579 end
580 table.sort(seated, function(a, b) return tuple_less(keys[a], keys[b]) end)
581 emit('BOARDING ' .. f.id .. ' ' .. #seated)
582 for number, id in ipairs(seated) do
583 local p = people[id]
584 emit(number .. ' ' .. groups[id] .. ' ' .. p.seat .. ' ' .. p.pnr .. ' ' .. p.name)
585 end
586 end
587 elseif command == 'LOADFACTOR' then
588 emit('LOADFACTOR ' .. f.id)
589 local total_seated, total_seats = 0, 0
590 for _, class in ipairs(classes) do
591 local seats, seated = 0, 0
592 for _, seat in ipairs(f.model.cabins[class].seats) do
593 if not f.blocked[seat] then seats = seats + 1 end
594 end
595 for _, id in ipairs(passengers(f)) do
596 if people[id].class == class and people[id].seat ~= '' then seated = seated + 1 end
597 end
598 emit(class .. ' ' .. seated .. '/' .. seats .. ' ' .. load_text(seated, seats) .. ' booked ' .. booked(f, class) .. '/' .. f.limits[class])
599 total_seated, total_seats = total_seated + seated, total_seats + seats
600 end
601 emit('ALL ' .. total_seated .. '/' .. total_seats .. ' ' .. load_text(total_seated, total_seats))
602 elseif command == 'BALANCE' then
603 local weight, moment = 0, 0
604 for _, id in ipairs(passengers(f)) do
605 local p = people[id]
606 if p.seat ~= '' then
607 local w = p.type == 'ADULT' and 80 or 40
608 if p.infant ~= 0 then w = w + 10 end
609 weight = weight + w
610 moment = moment + w * (f.model.seats[p.seat].row - f.model.base)
611 end
612 end
613 local limit, verdict = f.model.limit, nil
614 if moment < -limit then
615 verdict = 'NOSE-HEAVY'
616 elseif moment > limit then
617 verdict = 'TAIL-HEAVY'
618 else
619 verdict = 'OK'
620 end
621 emit('BALANCE ' .. f.id .. ' weight ' .. weight .. ' moment ' .. moment .. ' limit ' .. limit .. ' ' .. verdict)
622 elseif command == 'AVAIL' then
623 local class = t[3]
624 local c = cabin(f, class)
625 local counts, free = {W = 0, M = 0, A = 0, EXIT = 0}, 0
626 for _, seat in ipairs(c.seats) do
627 if not f.blocked[seat] and not f.occupied[seat] then
628 local s = f.model.seats[seat]
629 free = free + 1
630 counts[s.kind] = counts[s.kind] + 1
631 if f.model.exits[s.row] then counts.EXIT = counts.EXIT + 1 end
632 end
633 end
634 emit('AVAIL ' .. f.id .. ' ' .. class .. ' free ' .. free .. ' W ' .. counts.W .. ' M ' .. counts.M .. ' A ' .. counts.A .. ' EXIT ' .. counts.EXIT)
635 end
636 end
637 return output
638end
