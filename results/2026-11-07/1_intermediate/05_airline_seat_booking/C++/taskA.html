51[[noreturn]] void fail(const string &message) { throw std::runtime_error(message); }
52vector<string> split(const string &s, char delimiter = ' ') {
53 vector<string> result;
54 std::size_t start = 0;
55 for (std::size_t i = 0; i <= s.size(); ++i) {
56 if (i == s.size() || s[i] == delimiter) {
57 result.push_back(s.substr(start, i - start));
58 start = i + 1;
59 }
60 }
61 return result;
62}
63string join(const vector<string> &items, const string &separator) {
64 string result;
65 for (std::size_t i = 0; i < items.size(); ++i) {
66 if (i) result += separator;
67 result += items[i];
68 }
69 return result;
70}

115bool contains(const vector<int> &items, int value) { return std::find(items.begin(), items.end(), value) != items.end(); }
116bool exitAllowed(int id, const Aircraft &a, int seat) { return !a.exits[a.seats[seat].row] || people[id].type == 0 && !hasInfant(id); }
117bool infantAllowed(int flight, int id, int seat, const vector<int> &moving, const vector<std::pair<int, int>> &proposed) {
118 if (!hasInfant(id)) return true;
119 const Flight &f = flights[flight];
120 const Aircraft &a = aircraft[f.model];
121 const Seat &s = a.seats[seat];
122 const string &letters = a.cabins[s.cabin].blocks[s.block];
123 for (char letter : letters) {
124 int peer = a.lookup[s.row * 26 + letter - 'A'], owner = f.occupants[peer];
125 if (owner >= 0 && owner != id && !contains(moving, owner) && hasInfant(owner)) return false;
126 }
127 for (const auto &[other, seat] : proposed) {
128 const Seat &target = a.seats[seat];
129 if (other != id && hasInfant(other) && target.row == s.row && target.block == s.block) return false;
130 }
131 return true;
132}
133bool fits(int flight, const vector<int> &ids, const vector<int> &seats, const vector<int> &vacated = {}) {
134 const Flight &f = flights[flight];
135 const Aircraft &a = aircraft[f.model];
136 vector<int> moving = ids;
137 moving.insert(moving.end(), vacated.begin(), vacated.end());
138 vector<std::pair<int, int>> proposed;
139 for (int i = 0; i < int(ids.size()); ++i) proposed.push_back({ids[i], seats[i]});
140 for (int i = 0; i < int(ids.size()); ++i) {
141 int seat = seats[i], owner = f.occupants[seat];
142 if (f.blocked[seat] || owner >= 0 && !contains(moving, owner) || !exitAllowed(ids[i], a, seat) || !infantAllowed(flight, ids[i], seat, moving, proposed)) return false;
143 }
144 return true;
145}
146int chooseSeat(int flight, int id, int cabin) {
147 const Flight &f = flights[flight];
148 const Aircraft &a = aircraft[f.model];
149 int first = -1;
150 for (int seat = 0; seat < int(a.seats.size()); ++seat) {
151 if (a.seats[seat].cabin != cabin || f.blocked[seat] || f.occupants[seat] >= 0 || !exitAllowed(id, a, seat) || !infantAllowed(flight, id, seat, {}, {})) continue;
152 if (first < 0) first = seat;
153 if (people[id].preference < 0 || a.seats[seat].kind == people[id].preference) return seat;
154 }
155 return first;
156}
157void seatPerson(int flight, int id, int seat) {
158 Flight &f = flights[flight];
159 if (people[id].seat >= 0) f.occupants[people[id].seat] = -1;
160 people[id].seat = seat;
161 if (seat >= 0) f.occupants[seat] = id;
162}
163void reassign(int flight, vector<string> &out) {
164 const Aircraft &a = aircraft[flights[flight].model];
165 for (int id : flights[flight].passengers) {
166 if (!people[id].active || people[id].type == 2 || people[id].seat >= 0) continue;
167 int seat = chooseSeat(flight, id, people[id].cabin);
168 if (seat < 0) continue;
169 seatPerson(flight, id, seat);
170 out.push_back("REASSIGNED " + bookings[people[id].booking].pnr + " " + people[id].name + " " + seatName(a, seat));
171 }
172}
173void removePerson(int id) {
174 if (!people[id].active) return;
175 int lap = people[id].lap;
176 if (people[id].type == 0 && lap >= 0) removePerson(lap);
177 if (people[id].type == 2 && lap >= 0) people[lap].lap = -1;
178 seatPerson(bookings[people[id].booking].flight, id, -1);
179 people[id].active = false;
180 people[id].volunteer = false;
181 people[id].lap = -1;
182}
183string passengerResult(int id) {
184 return people[id].type == 2 ? "LAP" : seatName(aircraft[flights[bookings[people[id].booking].flight].model], people[id].seat);
185}
186string assignBooking(int flight, int cabin, const vector<int> &ids) {
187 if (ids.size() == 1) {
188 seatPerson(flight, ids[0], chooseSeat(flight, ids[0], cabin));
189 return "SINGLE";
190 }
191 const Aircraft &a = aircraft[flights[flight].model];
192 const Cabin &c = a.cabins[cabin];
193 int count = int(ids.size());
194 for (int stage = 0; stage < 3; ++stage) {
195 vector<string> blocks = stage == 1 ? vector<string>{join(c.blocks, "")} : c.blocks;
196 int firstHeight = stage == 2 ? 2 : 1, lastHeight = stage == 2 ? count : 1;
197 for (int height = firstHeight; height <= lastHeight; ++height) {
198 int width = stage == 2 ? (count + height - 1) / height : count;
199 int usedRows = (count + width - 1) / width;
200 for (int row = c.from; row + usedRows - 1 <= c.to; ++row) {
201 for (const string &block : blocks) {
202 for (int x = 0; x + width <= int(block.size()); ++x) {
203 vector<int> candidate;
204 for (int j = 0; j < count; ++j) candidate.push_back(a.lookup[(row + j / width) * 26 + block[x + j % width] - 'A']);
205 if (!fits(flight, ids, candidate)) continue;
206 for (int j = 0; j < count; ++j) seatPerson(flight, ids[j], candidate[j]);
207 return std::array<const char *, 3>{"BLOCK", "ROW", "STACK"}[stage];
208 }
209 }
210 }
211 }
212 }
213 for (int id : ids) seatPerson(flight, id, chooseSeat(flight, id, cabin));
214 return "SPLIT";
215}
216vector<string> execute(const vector<string> &v) {
217 const string &command = v[0];
218 vector<string> out;
219 if (command == "AIRCRAFT") {
220 building = Aircraft{};
221 building.model = v[1];
222 building.base = std::stoi(v[2]);
223 building.limit = std::stoi(v[3]);
224 } else if (command == "CABIN") {
225 int cabin = cabinId(v[1]);
226 Cabin &c = building.cabins[cabin];
227 c.from = std::stoi(v[2]);
228 c.to = std::stoi(v[3]);
229 c.blocks.assign(v.begin() + 4, v.end());
230 for (int row = c.from; row <= c.to; ++row) {
231 for (int block = 0; block < int(c.blocks.size()); ++block) {
232 const string &letters = c.blocks[block];
233 for (int position = 0; position < int(letters.size()); ++position) {
234 char letter = letters[position];
235 int kind = block == 0 && position == 0 || block + 1 == int(c.blocks.size()) && position + 1 == int(letters.size()) ? 0 : position == 0 || position + 1 == int(letters.size()) ? 2 : 1;
236 building.lookup[row * 26 + letter - 'A'] = int(building.seats.size());
237 building.seats.push_back({row, cabin, block, kind, letter});
238 building.blocked.push_back(false);
239 }
240 }
241 }
242 } else if (command == "EXIT") {
243 for (int i = 1; i < int(v.size()); ++i) building.exits[std::stoi(v[i])] = true;
244 } else if (command == "BLOCKED") {
245 for (int i = 1; i < int(v.size()); ++i) building.blocked[building.lookup[seatKey(v[i])]] = true;
246 } else if (command == "END") {
247 for (int i = 0; i < int(building.seats.size()); ++i) if (!building.blocked[i]) ++building.capacity[building.seats[i].cabin];
248 string line = "AIRCRAFT " + building.model;
249 for (int cabin = 0; cabin < 3; ++cabin) if (building.cabins[cabin].from) line += text(" ", classes[cabin], " ", building.capacity[cabin]);
250 modelIds[building.model] = int(aircraft.size());
251 aircraft.push_back(std::move(building));
252 out.push_back(line);
253 } else if (command == "FLIGHT") {
254 if (flightIds.count(v[1])) fail("duplicate flight " + v[1]);
255 if (!modelIds.count(v[2])) fail("unknown aircraft " + v[2]);
256 Flight f;
257 f.name = v[1];
258 f.model = modelIds[v[2]];
259 const Aircraft &a = aircraft[f.model];
260 f.blocked = a.blocked;
261 f.occupants.assign(a.seats.size(), -1);
262 string line = "FLIGHT " + f.name + " " + a.model;
263 for (int cabin = 0; cabin < 3; ++cabin) {
264 f.limit[cabin] = a.capacity[cabin] * (100 + std::stoi(v[3])) / 100;
265 if (a.cabins[cabin].from) line += text(" ", classes[cabin], " ", a.capacity[cabin], "/", f.limit[cabin]);
266 }
267 flightIds[f.name] = int(flights.size());
268 flights.push_back(std::move(f));
269 out.push_back(line);
270 } else if (command == "BOOK") {
271 if (bookingIds.count(v[1])) fail("duplicate PNR " + v[1]);
272 int flight = flightId(v[2], true), cabin = cabinId(v[3]);
273 const Aircraft &a = aircraft[flights[flight].model];
274 if (!a.cabins[cabin].from) fail("no cabin " + v[3] + " on " + v[2]);
275 vector<Person> draft;
276 std::set<string> names;
277 vector<int> adults, infants;
278 for (int i = 4; i < int(v.size()); ++i) {
279 vector<string> fields = split(v[i], ':');
280 if (!names.insert(fields[0]).second) fail("duplicate name " + fields[0] + " in " + v[1]);
281 int type = int(std::find(typeNames.begin(), typeNames.end(), fields[1]) - typeNames.begin());
282 Person p;
283 p.name = fields[0];
284 p.type = type;
285 p.cabin = cabin;
286 p.booking = int(bookings.size());
287 if (fields.size() == 3) p.preference = int(string("WMA").find(fields[2]));
288 if (type == 0) adults.push_back(int(draft.size()));
289 if (type == 2) infants.push_back(int(draft.size()));
290 draft.push_back(p);
291 }
292 for (const Person &p : draft) {
293 for (int id : flights[flight].passengers) if (people[id].active && people[id].name == p.name) fail(p.name + " already on " + v[2]);
294 }
295 if (adults.empty()) fail("no adult in " + v[1]);
296 if (infants.size() > adults.size()) fail("too many infants in " + v[1]);
297 if (bookedCount(flight, cabin) + int(draft.size() - infants.size()) > flights[flight].limit[cabin]) fail("limit exceeded for " + v[2] + " " + v[3]);
298 int base = int(people.size());
299 for (int i = 0; i < int(infants.size()); ++i) {
300 draft[adults[i]].lap = base + infants[i];
301 draft[infants[i]].lap = base + adults[i];
302 }
303 Booking b{v[1], flight, false, {}};
304 vector<int> needs;
305 for (const Person &p : draft) {
306 int id = int(people.size());
307 people.push_back(p);
308 b.people.push_back(id);
309 flights[flight].passengers.push_back(id);
310 if (p.type != 2) needs.push_back(id);
311 }
312 bookingIds[b.pnr] = int(bookings.size());
313 bookings.push_back(b);
314 string method = assignBooking(flight, cabin, needs);
315 string line = "BOOKED " + b.pnr + " " + v[2] + " " + v[3] + " " + method;
316 for (int id : b.people) line += " " + people[id].name + ":" + passengerResult(id);
317 out.push_back(line);
318 } else if (command == "STATUS") {
319 tiers[v[1]] = int(std::find(tierNames.begin(), tierNames.end(), v[2]) - tierNames.begin());
320 out.push_back("STATUS " + v[1] + " " + v[2]);
321 } else if (command == "FIND") {
322 int membership = tiers.count(v[1]) ? tiers[v[1]] : 3;
323 vector<string> found;
324 for (const Flight &f : flights) {
325 for (int id : f.passengers) {
326 const Person &p = people[id];
327 if (p.active && p.name == v[1]) found.push_back(text(f.name, " ", bookings[p.booking].pnr, " ", classes[p.cabin], " ", passengerResult(id), f.closed ? " CLOSED" : " OPEN"));
328 }
329 }
330 out.push_back(text("FIND ", v[1], " ", tierNames[membership], " ", found.size()));
331 out.insert(out.end(), found.begin(), found.end());
332 } else if (command == "SEAT") {
333 int id = resolve(v[1], v[2]);
334 Person &p = people[id];
335 int flight = bookings[p.booking].flight;
336 Flight &f = flights[flight];
337 const Aircraft &a = aircraft[f.model];
338 if (p.type == 2) fail("infant " + p.name + " has no seat");
339 int seat = a.lookup[seatKey(v[3])];
340 if (seat < 0 || a.seats[seat].cabin != p.cabin) fail("no seat " + v[3] + " in cabin " + classes[p.cabin]);
341 if (f.blocked[seat]) fail("seat " + v[3] + " is blocked");
342 if (f.occupants[seat] >= 0) fail("seat " + v[3] + " is occupied");
343 if (!exitAllowed(id, a, seat)) fail(text("exit row ", a.seats[seat].row, " not allowed for ", p.name));
344 if (!infantAllowed(flight, id, seat, {}, {})) fail("infant limit reached at " + v[3]);
345 string old = seatName(a, p.seat);
346 seatPerson(flight, id, seat);
347 out.push_back("MOVED " + v[1] + " " + v[2] + " " + old + " -> " + v[3]);
348 reassign(flight, out);
349 } else if (command == "SWAP") {
350 int first = resolve(v[1], v[2]);
351 int second = resolve(v[3], v[4]);
352 if (first == second) fail("cannot swap " + v[2] + " with itself");
353 for (int id : {first, second}) if (people[id].type == 2) fail("infant " + people[id].name + " has no seat");
354 int flight = bookings[people[first].booking].flight;
355 if (flight != bookings[people[second].booking].flight) fail(v[1] + " and " + v[3] + " are on different flights");
356 if (people[first].cabin != people[second].cabin) fail(v[2] + " and " + v[4] + " are in different cabins");
357 for (int id : {first, second}) if (people[id].seat < 0) fail(people[id].name + " has no seat");
358 int oldFirst = people[first].seat, oldSecond = people[second].seat;
359 if (!fits(flight, {first, second}, {oldSecond, oldFirst})) fail("swap of " + v[2] + " and " + v[4] + " breaks seating rules");
360 flights[flight].occupants[oldFirst] = flights[flight].occupants[oldSecond] = -1;
361 people[first].seat = people[second].seat = -1;
362 seatPerson(flight, first, oldSecond);
363 seatPerson(flight, second, oldFirst);
364 const Aircraft &a = aircraft[flights[flight].model];
365 out.push_back("SWAPPED " + v[1] + " " + v[2] + " " + seatName(a, oldSecond) + " " + v[3] + " " + v[4] + " " + seatName(a, oldFirst));
366 reassign(flight, out);
367 } else if (command == "CANCEL") {
368 int booking = bookingId(v[1], true), target = v.size() == 3 ? passengerId(booking, v[2]) : -1;
369 vector<int> removed;
370 for (int id : bookings[booking].people) if (people[id].active && (target < 0 || id == target || people[target].type == 0 && people[target].lap == id)) removed.push_back(id);
371 int remaining = 0, adults = 0;
372 for (int id : bookings[booking].people) {
373 if (people[id].active && !contains(removed, id)) {
374 ++remaining;
375 adults += people[id].type == 0;
376 }
377 }
378 if (target >= 0 && remaining && !adults) fail(v[1] + " would have no adult");
379 string line = "CANCELLED " + v[1];
380 for (int id : removed) line += " " + people[id].name;
381 for (int id : removed) removePerson(id);
382 out.push_back(line);
383 reassign(bookings[booking].flight, out);
384 } else if (command == "CHECKIN" || command == "VOLUNTEER" || command == "INFO") {
385 int booking = bookingId(v[1], command != "INFO");
386 Booking &b = bookings[booking];
387 if (command == "CHECKIN") {
388 if (b.checked) fail(v[1] + " already checked in");
389 b.checked = true;
390 out.push_back(text("CHECKIN ", b.pnr, " ", activeCount(booking)));
391 } else if (command == "INFO") {
392 out.push_back(text("INFO ", b.pnr, " ", flights[b.flight].name, " ", activeCount(booking), b.checked ? " CHECKED" : " NOT-CHECKED"));
393 for (int id : b.people) if (people[id].active) out.push_back(text(people[id].name, " ", typeNames[people[id].type], " ", classes[people[id].cabin], " ", passengerResult(id)));
394 } else {
395 int id = passengerId(booking, v[2]);
396 Person &p = people[id];
397 if (p.type != 0 || hasInfant(id)) fail(p.name + " cannot volunteer");
398 if (p.seat < 0) fail(p.name + " has no seat");
399 if (p.volunteer) fail(p.name + " already volunteered");
400 p.volunteer = true;
401 flights[b.flight].volunteers.push_back(id);
402 out.push_back("VOLUNTEER " + b.pnr + " " + p.name + " " + passengerResult(id));
403 }
404 } else {
405 bool modifying = command == "UPGRADE" || command == "CLOSE" || command == "BLOCK" || command == "UNBLOCK";
406 int flight = flightId(v[1], modifying);
407 Flight &f = flights[flight];
408 const Aircraft &a = aircraft[f.model];
409 if (command == "BLOCK" || command == "UNBLOCK") {
410 int seat = a.lookup[seatKey(v[2])];
411 if (seat < 0) fail("no seat " + v[2] + " on " + f.name);
412 if (command == "BLOCK" && f.blocked[seat]) fail("seat " + v[2] + " is blocked");
413 if (command == "UNBLOCK" && !f.blocked[seat]) fail("seat " + v[2] + " is not blocked");
414 f.blocked[seat] = command == "BLOCK";
415 out.push_back(command + " " + f.name + " " + v[2]);
416 int id = f.occupants[seat];
417 if (command == "BLOCK" && id >= 0) {
418 seatPerson(flight, id, -1);
419 seatPerson(flight, id, chooseSeat(flight, id, people[id].cabin));
420 out.push_back("DISPLACED " + bookings[people[id].booking].pnr + " " + people[id].name + " " + v[2] + " -> " + passengerResult(id));
421 }
422 reassign(flight, out);
423 } else if (command == "UPGRADE") {
424 vector<int> cabins;
425 for (int c = 0; c < 3; ++c) if (a.cabins[c].from) cabins.push_back(c);
426 int upgraded = 0;
427 for (int pair = 0; pair + 1 < int(cabins.size()); ++pair) {
428 int higher = cabins[pair], lower = cabins[pair + 1], booked = bookedCount(flight, higher);
429 vector<std::pair<int, int>> targets;
430 for (int id : f.passengers) if (people[id].active && people[id].cabin == lower && people[id].type == 0 && !hasInfant(id) && tier(id) < 3) targets.push_back({tier(id), id});
431 std::stable_sort(targets.begin(), targets.end(), [](const std::pair<int, int> &x, const std::pair<int, int> &y) { return x.first < y.first; });
432 for (const std::pair<int, int> &target : targets) {
433 if (booked >= f.limit[higher]) break;
434 int id = target.second, seat = chooseSeat(flight, id, higher);
435 if (seat < 0) break;
436 string old = passengerResult(id);
437 people[id].cabin = higher;
438 seatPerson(flight, id, seat);
439 ++booked;
440 ++upgraded;
441 out.push_back(text("UPGRADED ", bookings[people[id].booking].pnr, " ", people[id].name, " ", classes[lower], " ", old, " -> ", classes[higher], " ", seatName(a, seat)));
442 }
443 }
444 reassign(flight, out);
445 out.push_back(text("UPGRADE ", f.name, " ", upgraded));
446 } else if (command == "CLOSE") {
447 for (int b = 0; b < int(bookings.size()); ++b) {
448 if (bookings[b].flight != flight || bookings[b].checked || !activeCount(b)) continue;
449 out.push_back(text("NOSHOW ", bookings[b].pnr, " ", activeCount(b)));
450 for (int id : bookings[b].people) removePerson(id);
451 }
452 reassign(flight, out);
453 int denied = 0, compensation = 0;
454 const int voluntary[] = {500, 300, 150}, involuntary[] = {1000, 600, 300};
455 for (int cabin = 0; cabin < 3; ++cabin) {
456 vector<int> targets;
457 for (int id : f.volunteers) if (people[id].active && people[id].volunteer && people[id].seat >= 0 && people[id].cabin == cabin) targets.push_back(id);
458 for (int id : targets) {
459 int seat = people[id].seat;
460 bool useful = std::any_of(f.passengers.begin(), f.passengers.end(), [&](int other) {
461 const Person &p = people[other];
462 return p.active && p.type != 2 && p.cabin == cabin && p.seat < 0 && fits(flight, {other}, {seat}, {id});
463 });
464 if (!useful) continue;
465 out.push_back(text("DENIED VOLUNTARY ", bookings[people[id].booking].pnr, " ", people[id].name, " ", classes[cabin], " ", voluntary[cabin]));
466 ++denied;
467 compensation += voluntary[cabin];
468 removePerson(id);
469 reassign(flight, out);
470 }
471 }
472 for (int id : f.passengers) {
473 if (!people[id].active || people[id].type == 2 || people[id].seat >= 0) continue;
474 int cabin = people[id].cabin, value = involuntary[cabin] / (people[id].type == 1 ? 2 : 1);
475 out.push_back(text("DENIED INVOLUNTARY ", bookings[people[id].booking].pnr, " ", people[id].name, " ", classes[cabin], " ", value));
476 ++denied;
477 compensation += value;
478 removePerson(id);
479 }
480 f.closed = true;
481 out.push_back(text("CLOSED ", f.name, " denied ", denied, " compensation ", compensation));
482 } else if (command == "MAP") {
483 out.push_back("MAP " + f.name);
484 for (int cabin = 0; cabin < 3; ++cabin) {
485 const Cabin &c = a.cabins[cabin];
486 if (!c.from) continue;
487 out.push_back(text("CABIN ", classes[cabin]));
488 out.push_back(" " + join(c.blocks, " "));
489 for (int row = c.from; row <= c.to; ++row) {
490 string line = text(row < 10 ? " " : "", row, " ");
491 for (int block = 0; block < int(c.blocks.size()); ++block) {
492 if (block) line += " ";
493 for (char letter : c.blocks[block]) {
494 int seat = a.lookup[row * 26 + letter - 'A'];
495 line += f.blocked[seat] ? '#' : f.occupants[seat] >= 0 ? 'X' : a.exits[row] ? 'E' : '.';
496 }
497 }
498 out.push_back(line);
499 }
500 }
501 } else if (command == "MANIFEST") {
502 int seated = 0, unassigned = 0, infants = 0;
503 vector<int> order;
504 for (int owner : f.occupants) if (owner >= 0) order.push_back(owner);
505 seated = int(order.size());
506 for (int id : f.passengers) {
507 if (!people[id].active) continue;
508 if (people[id].type == 2) ++infants;
509 else if (people[id].seat < 0) {
510 ++unassigned;
511 order.push_back(id);
512 }
513 }
514 out.push_back(text("MANIFEST ", f.name, " seated ", seated, " unassigned ", unassigned, " infants ", infants));
515 for (int id : order) {
516 const Person &p = people[id];
517 const Booking &b = bookings[p.booking];
518 string line = text(p.seat < 0 ? "---" : seatName(a, p.seat), " ", classes[p.cabin], " ", b.pnr, " ", p.name, " ", typeNames[p.type]);
519 if (hasInfant(id)) line += " +" + people[p.lap].name;
520 if (tier(id) < 3) line += " " + tierNames[tier(id)];
521 if (b.checked) line += " CHECKED";
522 if (p.volunteer) line += " VOL";
523 out.push_back(line);
524 }
525 } else if (command == "LOADFACTOR") {
526 out.push_back("LOADFACTOR " + f.name);
527 int allSeated = 0, allSeats = 0;
528 for (int cabin = 0; cabin < 3; ++cabin) {
529 if (!a.cabins[cabin].from) continue;
530 int seats = 0, seated = 0;
531 for (int i = 0; i < int(a.seats.size()); ++i) {
532 if (a.seats[i].cabin != cabin) continue;
533 seats += !f.blocked[i];
534 seated += f.occupants[i] >= 0;
535 }
536 int factor = seated * 1000 / seats;
537 out.push_back(text(classes[cabin], " ", seated, "/", seats, " ", factor / 10, ".", factor % 10, "% booked ", bookedCount(flight, cabin), "/", f.limit[cabin]));
538 allSeated += seated;
539 allSeats += seats;
540 }
541 int factor = allSeated * 1000 / allSeats;
542 out.push_back(text("ALL ", allSeated, "/", allSeats, " ", factor / 10, ".", factor % 10, "%"));
543 } else if (command == "BALANCE") {
544 int weight = 0, moment = 0;
545 for (int id : f.occupants) {
546 if (id < 0) continue;
547 int value = people[id].type == 0 ? 80 : 40;
548 if (hasInfant(id)) value += 10;
549 weight += value;
550 moment += value * (a.seats[people[id].seat].row - a.base);
551 }
552 string verdict = moment < -a.limit ? "NOSE-HEAVY" : moment > a.limit ? "TAIL-HEAVY" : "OK";
553 out.push_back(text("BALANCE ", f.name, " weight ", weight, " moment ", moment, " limit ", a.limit, " ", verdict));
554 } else if (command == "BOARDING") {
555 struct Item { int id, group, zone, kind, row; char letter; };
556 vector<Item> order;
557 for (int id : f.occupants) {
558 if (id < 0) continue;
559 const Person &p = people[id];
560 const Seat &s = a.seats[p.seat];
561 int group = bookingInfants(p.booking) ? 0 : p.cabin < 2 ? 1 : tier(id) < 2 ? 2 : 3;
562 int zone = group == 3 ? (s.row - a.cabins[2].from) / 5 + 1 : 0;
563 order.push_back({id, group, zone, s.kind, s.row, s.letter});
564 }
565 std::sort(order.begin(), order.end(), [](const Item &x, const Item &y) {
566 if (x.group != y.group) return x.group < y.group;
567 if (x.group == 3) {
568 if (x.zone != y.zone) return x.zone > y.zone;
569 if (x.kind != y.kind) return x.kind < y.kind;
570 if (x.row != y.row) return x.row > y.row;
571 } else if (x.row != y.row) return x.row < y.row;
572 return x.letter < y.letter;
573 });
574 out.push_back(text("BOARDING ", f.name, " ", order.size()));
575 for (int i = 0; i < int(order.size()); ++i) {
576 const Item &item = order[i];
577 const Person &p = people[item.id];
578 string group = item.group == 3 ? text("ZONE", item.zone) : std::array<const char *, 3>{"PRE", "CABIN", "TIER"}[item.group];
579 out.push_back(text(i + 1, " ", group, " ", seatName(a, p.seat), " ", bookings[p.booking].pnr, " ", p.name));
580 }
581 } else if (command == "AVAIL") {
582 int cabin = cabinId(v[2]);
583 if (!a.cabins[cabin].from) fail("no cabin " + v[2] + " on " + f.name);
584 int free = 0, exits = 0;
585 std::array<int, 3> kinds{};
586 for (int i = 0; i < int(a.seats.size()); ++i) {
587 const Seat &s = a.seats[i];
588 if (s.cabin != cabin || f.blocked[i] || f.occupants[i] >= 0) continue;
589 ++free;
590 ++kinds[s.kind];
591 exits += a.exits[s.row];
592 }
593 out.push_back(text("AVAIL ", f.name, " ", classes[cabin], " free ", free, " W ", kinds[0], " M ", kinds[1], " A ", kinds[2], " EXIT ", exits));
594 }
595 }
596 return out;
597}
