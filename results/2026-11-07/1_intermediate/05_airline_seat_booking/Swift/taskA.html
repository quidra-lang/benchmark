88final class Airline {
89 var aircraft: [String:Aircraft] = [:]
90 var flights: [String:Flight] = [:]
91 var flightOrder: [Flight] = []
92 var bookings: [String:Booking] = [:]
93 var used: Set<String> = []
94 var tiers: [String:String] = [:]
95 var defining: Aircraft!
96 var nextOrder = 0
97 func fail(_ message: String) -> Problem { Problem(message: message) }
98 func tier(_ p: Passenger) -> String { tiers[p.name] ?? "NONE" }
99 func tierRank(_ p: Passenger) -> Int { ["PLATINUM","GOLD","SILVER","NONE"].firstIndex(of: tier(p))! }
100 func flight(_ name: String, _ open: Bool) throws -> Flight {
101 guard let f = flights[name] else { throw fail("unknown flight "+name) }
102 if open && f.closed { throw fail("flight "+name+" is closed") }
103 return f
104 }
105 func booking(_ pnr: String, _ open: Bool) throws -> Booking {
106 guard let b = bookings[pnr], !b.passengers.isEmpty else { throw fail("unknown PNR "+pnr) }
107 if open && b.flight.closed { throw fail("flight "+b.flight.name+" is closed") }
108 return b
109 }
110 func passenger(_ b: Booking, _ name: String) throws -> Passenger {
111 guard let p = b.passengers.first(where: { $0.name == name }) else { throw fail("no passenger "+name+" in "+b.pnr) }
112 return p
113 }
114 func passengers(_ f: Flight) -> [Passenger] { f.bookings.flatMap { $0.passengers } }
115 func booked(_ f: Flight, _ cabin: String) -> Int { passengers(f).filter { $0.needsSeat && $0.cabin == cabin }.count }
116 func seats(_ a: Aircraft, _ cabin: String, _ blocked: Set<String>) -> Int { a.cabins[cabin]!.seats.filter { !blocked.contains($0.id) }.count }
117 func requireCabin(_ f: Flight, _ cabin: String) throws {
118 guard f.aircraft.cabins[cabin] != nil else { throw fail("no cabin "+cabin+" on "+f.name) }
119 }
120 func sameBlock(_ a: Seat, _ b: Seat) -> Bool { a.row == b.row && a.block == b.block }
121 func exitAllowed(_ p: Passenger, _ s: Seat) -> Bool {
122 !p.booking.flight.aircraft.exits.contains(s.row) || (p.type != "CHILD" && p.lap == nil)
123 }
124 func infantAllowed(_ p: Passenger, _ s: Seat, _ vacated: Set<Passenger>, _ proposed: [Passenger:Seat]) -> Bool {
125 if p.lap == nil { return true }
126 let f = p.booking.flight
127 for letter in f.aircraft.cabins[s.cabin]!.blocks[s.block] {
128 if let q = f.occupied[String(s.row)+String(letter)], q !== p && q.lap != nil && !vacated.contains(q) { return false }
129 }
130 for (q,target) in proposed {
131 if q !== p && q.lap != nil && sameBlock(target,s) { return false }
132 }
133 return true
134 }
135 func fits(_ p: Passenger, _ s: Seat, _ vacated: Set<Passenger> = [], _ proposed: [Passenger:Seat] = [:]) -> Bool {
136 let f = p.booking.flight
137 let occupant = f.occupied[s.id]
138 return !f.blocked.contains(s.id) && (occupant == nil || vacated.contains(occupant!)) &&
139 exitAllowed(p,s) && infantAllowed(p,s,vacated,proposed)
140 }
141 func seat(_ p: Passenger, _ target: Seat?) {
142 let f = p.booking.flight
143 if let old = p.seat { f.occupied.removeValue(forKey: old.id) }
144 p.seat = target
145 if let target { f.occupied[target.id] = p }
146 }
147 func select(_ p: Passenger, _ cabin: String) -> Seat? {
148 let candidates = p.booking.flight.aircraft.cabins[cabin]!.seats.filter { fits(p,$0) }
149 return candidates.first { $0.kind == p.pref } ?? candidates.first
150 }
151 func reassign(_ f: Flight, _ output: inout [String]) {
152 for p in passengers(f) where p.needsSeat && p.seat == nil {
153 if let target = select(p,p.cabin) {
154 seat(p,target)
155 output.append("REASSIGNED \(p.booking.pnr) \(p.name) \(target.id)")
156 }
157 }
158 }
159 func candidate(_ people: [Passenger], _ positions: [Seat]) -> Bool {
160 let proposed = Dictionary(uniqueKeysWithValues: zip(people,positions))
161 guard people.allSatisfy({ fits($0,proposed[$0]!,[],proposed) }) else { return false }
162 for p in people { seat(p,proposed[p]!) }
163 return true
164 }
165 func assign(_ b: Booking, _ cabinName: String) -> String {
166 let people = b.passengers.filter { $0.needsSeat }
167 if people.count == 1 {
168 seat(people[0],select(people[0],cabinName))
169 return "SINGLE"
170 }
171 let cabin = b.flight.aircraft.cabins[cabinName]!
172 let a = b.flight.aircraft
173 let k = people.count
174 for stage in 1...3 {
175 let lastH = stage == 3 ? k : 2
176 for h in 2...lastH {
177 let width = (k+h-1)/h
178 for row in cabin.first...cabin.last {
179 let blocks = stage == 2 ? [cabin.blocks.joined()] : cabin.blocks
180 for block in blocks {
181 let letters = Array(block)
182 let span = stage == 3 ? width : k
183 if span > letters.count { continue }
184 for x in 0...(letters.count-span) {
185 var positions: [Seat] = []
186 for j in 0..<k {
187 let r = row+(stage == 3 ? j/width : 0)
188 if r > cabin.last { break }
189 let letter = letters[x+(stage == 3 ? j%width : j)]
190 positions.append(a.seats[String(r)+String(letter)]!)
191 }
192 if positions.count == k && candidate(people,positions) { return stage == 1 ? "BLOCK" : (stage == 2 ? "ROW" : "STACK") }
193 }
194 }
195 }
196 }
197 }
198 for p in people { seat(p,select(p,cabinName)) }
199 return "SPLIT"
200 }
201 func targets(_ p: Passenger) -> [Passenger] { p.lap.map { [p,$0] } ?? [p] }
202 func remove(_ targets: [Passenger]) {
203 for p in targets {
204 seat(p,nil)
205 p.booking.flight.volunteers.removeAll { $0 === p }
206 p.booking.passengers.removeAll { $0 === p }
207 p.parent?.lap = nil
208 }
209 }
210 func seatBefore(_ a: Passenger, _ b: Passenger) -> Bool {
211 a.seat!.row == b.seat!.row ? a.seat!.letter < b.seat!.letter : a.seat!.row < b.seat!.row
212 }
213 func load(_ seated: Int, _ count: Int) -> String {
214 let q = seated*1000/count
215 return "\(q/10).\(q%10)%"
216 }
217 func boardingRank(_ p: Passenger) -> Int {
218 if p.booking.passengers.contains(where: { !$0.needsSeat }) { return 0 }
219 if p.cabin != "Y" { return 1 }
220 return tierRank(p) <= 1 ? 2 : 3
221 }
222 func zone(_ p: Passenger) -> Int { (p.seat!.row-p.booking.flight.aircraft.cabins["Y"]!.first)/5+1 }
223 func execute(_ line: String) throws -> [String] {
224 let t = line.split(separator: " ").map(String.init)
225 let command = t[0]
226 var output: [String] = []
227 switch command {
228 case "AIRCRAFT": defining = Aircraft(t[1],Int(t[2])!,Int(t[3])!)
229 case "CABIN":
230 let c = Cabin(t[1],Int(t[2])!,Int(t[3])!,Array(t.dropFirst(4)))
231 for row in c.first...c.last {
232 for (block,string) in c.blocks.enumerated() {
233 let letters = Array(string)
234 for (i,letter) in letters.enumerated() {
235 let window = (block == 0 && i == 0) || (block == c.blocks.count-1 && i == letters.count-1)
236 let kind = window ? "W" : ((i == 0 || i == letters.count-1) ? "A" : "M")
237 let s = Seat(id: String(row)+String(letter),row: row,letter: letter,block: block,cabin: c.name,kind: kind)
238 c.seats.append(s)
239 defining.seats[s.id] = s
240 }
241 }
242 }
243 defining.cabins[c.name] = c
244 case "EXIT":
245 for value in t.dropFirst() { defining.exits.insert(Int(value)!) }
246 case "BLOCKED": defining.blocked.formUnion(t.dropFirst())
247 case "END":
248 aircraft[defining.model] = defining
249 let counts = classes.filter { defining.cabins[$0] != nil }.map { " \($0) \(seats(defining,$0,defining.blocked))" }.joined()
250 output.append("AIRCRAFT "+defining.model+counts)
251 case "FLIGHT":
252 guard flights[t[1]] == nil else { throw fail("duplicate flight "+t[1]) }
253 guard let a = aircraft[t[2]] else { throw fail("unknown aircraft "+t[2]) }
254 let f = Flight(t[1],a)
255 var result = "FLIGHT "+f.name+" "+a.model
256 for c in classes where a.cabins[c] != nil {
257 let count = seats(a,c,a.blocked)
258 let limit = count*(100+Int(t[3])!)/100
259 f.limits[c] = limit
260 result += " \(c) \(count)/\(limit)"
261 }
262 flights[f.name] = f
263 flightOrder.append(f)
264 output.append(result)
265 case "BOOK":
266 guard !used.contains(t[1]) else { throw fail("duplicate PNR "+t[1]) }
267 let f = try flight(t[2],true)
268 try requireCabin(f,t[3])
269 let b = Booking(t[1],f)
270 var seen: Set<String> = []
271 var adults: [Passenger] = []
272 var infants: [Passenger] = []
273 for token in t.dropFirst(4) {
274 let parts = token.split(separator: ":").map(String.init)
275 guard seen.insert(parts[0]).inserted else { throw fail("duplicate name "+parts[0]+" in "+b.pnr) }
276 let p = Passenger(parts[0],parts[1],parts.count == 3 ? parts[2] : nil,t[3],b)
277 b.passengers.append(p)
278 if p.type == "ADULT" { adults.append(p) }
279 if p.type == "INFANT" { infants.append(p) }
280 }
281 let present = Set(passengers(f).map { $0.name })
282 for p in b.passengers {
283 if present.contains(p.name) { throw fail(p.name+" already on "+f.name) }
284 }
285 guard !adults.isEmpty else { throw fail("no adult in "+b.pnr) }
286 guard infants.count <= adults.count else { throw fail("too many infants in "+b.pnr) }
287 guard booked(f,t[3])+b.passengers.filter({ $0.needsSeat }).count <= f.limits[t[3]]! else {
288 throw fail("limit exceeded for "+f.name+" "+t[3])
289 }
290 for i in infants.indices {
291 adults[i].lap = infants[i]
292 infants[i].parent = adults[i]
293 }
294 for p in b.passengers {
295 p.order = nextOrder
296 nextOrder += 1
297 }
298 let method = assign(b,t[3])
299 f.bookings.append(b)
300 bookings[b.pnr] = b
301 used.insert(b.pnr)
302 output.append("BOOKED \(b.pnr) \(f.name) \(t[3]) \(method) "+b.passengers.map { $0.name+":"+$0.result }.joined(separator: " "))
303 case "SEAT":
304 let b = try booking(t[1],true)
305 let p = try passenger(b,t[2])
306 guard p.needsSeat else { throw fail("infant "+p.name+" has no seat") }
307 guard let target = b.flight.aircraft.seats[t[3]], target.cabin == p.cabin else { throw fail("no seat "+t[3]+" in cabin "+p.cabin) }
308 guard !b.flight.blocked.contains(t[3]) else { throw fail("seat "+t[3]+" is blocked") }
309 guard b.flight.occupied[t[3]] == nil else { throw fail("seat "+t[3]+" is occupied") }
310 guard exitAllowed(p,target) else { throw fail("exit row \(target.row) not allowed for "+p.name) }
311 guard infantAllowed(p,target,[],[:]) else { throw fail("infant limit reached at "+t[3]) }
312 let old = p.result
313 seat(p,target)
314 output.append("MOVED \(b.pnr) \(p.name) \(old) -> \(t[3])")
315 reassign(b.flight,&output)
316 case "SWAP":
317 let a = try booking(t[1],true)
318 let p = try passenger(a,t[2])
319 let b = try booking(t[3],true)
320 let q = try passenger(b,t[4])
321 guard p !== q else { throw fail("cannot swap "+p.name+" with itself") }
322 for x in [p,q] {
323 guard x.needsSeat else { throw fail("infant "+x.name+" has no seat") }
324 }
325 guard a.flight === b.flight else { throw fail(a.pnr+" and "+b.pnr+" are on different flights") }
326 guard p.cabin == q.cabin else { throw fail(p.name+" and "+q.name+" are in different cabins") }
327 for x in [p,q] {
328 guard x.seat != nil else { throw fail(x.name+" has no seat") }
329 }
330 let ps = p.seat!
331 let qs = q.seat!
332 let vacated: Set<Passenger> = [p,q]
333 let proposed = [p:qs,q:ps]
334 guard fits(p,qs,vacated,proposed) && fits(q,ps,vacated,proposed) else {
335 throw fail("swap of "+p.name+" and "+q.name+" breaks seating rules")
336 }
337 seat(p,nil)
338 seat(q,nil)
339 seat(p,qs)
340 seat(q,ps)
341 output.append("SWAPPED \(a.pnr) \(p.name) \(p.result) \(b.pnr) \(q.name) \(q.result)")
342 reassign(a.flight,&output)
343 case "CANCEL":
344 let b = try booking(t[1],true)
345 let selected: [Passenger]
346 if t.count == 2 { selected = b.passengers }
347 else {
348 let p = try passenger(b,t[2])
349 let removing = targets(p)
350 let remaining = b.passengers.filter { !removing.contains($0) }
351 guard remaining.isEmpty || remaining.contains(where: { $0.type == "ADULT" }) else { throw fail(b.pnr+" would have no adult") }
352 selected = b.passengers.filter { removing.contains($0) }
353 }
354 let names = selected.map { $0.name }.joined(separator: " ")
355 remove(selected)
356 output.append("CANCELLED "+b.pnr+" "+names)
357 reassign(b.flight,&output)
358 case "STATUS":
359 tiers[t[1]] = t[2]
360 output.append("STATUS "+t[1]+" "+t[2])
361 case "CHECKIN":
362 let b = try booking(t[1],true)
363 guard !b.checked else { throw fail(b.pnr+" already checked in") }
364 b.checked = true
365 output.append("CHECKIN \(b.pnr) \(b.passengers.count)")
366 case "VOLUNTEER":
367 let b = try booking(t[1],true)
368 let p = try passenger(b,t[2])
369 guard p.type == "ADULT" && p.lap == nil else { throw fail(p.name+" cannot volunteer") }
370 guard p.seat != nil else { throw fail(p.name+" has no seat") }
371 guard !b.flight.volunteers.contains(p) else { throw fail(p.name+" already volunteered") }
372 b.flight.volunteers.append(p)
373 output.append("VOLUNTEER \(b.pnr) \(p.name) \(p.result)")
374 case "UPGRADE":
375 let f = try flight(t[1],true)
376 let cabins = classes.filter { f.aircraft.cabins[$0] != nil }
377 var count = 0
378 if cabins.count > 1 {
379 for i in 0..<(cabins.count-1) {
380 let higher = cabins[i]
381 let lower = cabins[i+1]
382 let targets = passengers(f).filter { $0.cabin == lower && $0.type == "ADULT" && $0.lap == nil && tier($0) != "NONE" }.sorted {
383 tierRank($0) == tierRank($1) ? $0.order < $1.order : tierRank($0) < tierRank($1)
384 }
385 for p in targets {
386 if booked(f,higher) >= f.limits[higher]! { break }
387 guard let target = select(p,higher) else { break }
388 let old = p.result
389 p.cabin = higher
390 seat(p,target)
391 output.append("UPGRADED \(p.booking.pnr) \(p.name) \(lower) \(old) -> \(higher) \(target.id)")
392 count += 1
393 }
394 }
395 }
396 reassign(f,&output)
397 output.append("UPGRADE \(f.name) \(count)")
398 case "CLOSE":
399 let f = try flight(t[1],true)
400 for b in f.bookings where !b.passengers.isEmpty && !b.checked {
401 output.append("NOSHOW \(b.pnr) \(b.passengers.count)")
402 remove(b.passengers)
403 }
404 reassign(f,&output)
405 var count = 0
406 var compensation = 0
407 for c in classes where f.aircraft.cabins[c] != nil {
408 let volunteers = f.volunteers.filter { $0.seat != nil && $0.cabin == c }
409 for p in volunteers {
410 let useful = passengers(f).contains { $0.needsSeat && $0.seat == nil && $0.cabin == c && fits($0,p.seat!,[p]) }
411 if useful {
412 let value = c == "F" ? 500 : (c == "J" ? 300 : 150)
413 output.append("DENIED VOLUNTARY \(p.booking.pnr) \(p.name) \(c) \(value)")
414 count += 1
415 compensation += value
416 remove(targets(p))
417 reassign(f,&output)
418 }
419 }
420 }
421 for p in passengers(f) where p.needsSeat && p.seat == nil {
422 var value = p.cabin == "F" ? 1000 : (p.cabin == "J" ? 600 : 300)
423 if p.type == "CHILD" { value /= 2 }
424 output.append("DENIED INVOLUNTARY \(p.booking.pnr) \(p.name) \(p.cabin) \(value)")
425 count += 1
426 compensation += value
427 remove(targets(p))
428 }
429 f.closed = true
430 output.append("CLOSED \(f.name) denied \(count) compensation \(compensation)")
431 case "BLOCK", "UNBLOCK":
432 let f = try flight(t[1],true)
433 guard f.aircraft.seats[t[2]] != nil else { throw fail("no seat "+t[2]+" on "+f.name) }
434 let blocked = f.blocked.contains(t[2])
435 if command == "BLOCK" && blocked { throw fail("seat "+t[2]+" is blocked") }
436 if command == "UNBLOCK" && !blocked { throw fail("seat "+t[2]+" is not blocked") }
437 output.append(command+" "+f.name+" "+t[2])
438 if command == "BLOCK" {
439 f.blocked.insert(t[2])
440 if let p = f.occupied[t[2]] {
441 seat(p,nil)
442 seat(p,select(p,p.cabin))
443 output.append("DISPLACED \(p.booking.pnr) \(p.name) \(t[2]) -> \(p.result)")
444 }
445 } else { f.blocked.remove(t[2]) }
446 reassign(f,&output)
447 case "MAP":
448 let f = try flight(t[1],false)
449 output.append("MAP "+f.name)
450 for name in classes {
451 guard let c = f.aircraft.cabins[name] else { continue }
452 output.append("CABIN "+name)
453 output.append(" "+c.blocks.joined(separator: " "))
454 for row in c.first...c.last {
455 let blocks = c.blocks.map { block in String(block.map { letter in
456 let id = String(row)+String(letter)
457 return f.blocked.contains(id) ? Character("#") : (f.occupied[id] != nil ? Character("X") : (f.aircraft.exits.contains(row) ? Character("E") : Character(".")))
458 }) }
459 output.append(String(format: "%2ld %@",row,blocks.joined(separator: " ")))
460 }
461 }
462 case "MANIFEST":
463 let f = try flight(t[1],false)
464 let all = passengers(f)
465 let seated = all.filter { $0.seat != nil }.sorted(by: seatBefore)
466 let unassigned = all.filter { $0.needsSeat && $0.seat == nil }
467 output.append("MANIFEST \(f.name) seated \(seated.count) unassigned \(unassigned.count) infants \(all.filter { !$0.needsSeat }.count)")
468 for p in seated+unassigned {
469 var result = "\(p.seat?.id ?? "---") \(p.cabin) \(p.booking.pnr) \(p.name) \(p.type)"
470 if let infant = p.lap { result += " +"+infant.name }
471 if tier(p) != "NONE" { result += " "+tier(p) }
472 if p.booking.checked { result += " CHECKED" }
473 if f.volunteers.contains(p) { result += " VOL" }
474 output.append(result)
475 }
476 case "LOADFACTOR":
477 let f = try flight(t[1],false)
478 output.append("LOADFACTOR "+f.name)
479 var totalSeated = 0
480 var totalSeats = 0
481 for c in classes where f.aircraft.cabins[c] != nil {
482 let count = seats(f.aircraft,c,f.blocked)
483 let seated = passengers(f).filter { $0.cabin == c && $0.seat != nil }.count
484 output.append("\(c) \(seated)/\(count) \(load(seated,count)) booked \(booked(f,c))/\(f.limits[c]!)")
485 totalSeated += seated
486 totalSeats += count
487 }
488 output.append("ALL \(totalSeated)/\(totalSeats) \(load(totalSeated,totalSeats))")
489 case "BALANCE":
490 let f = try flight(t[1],false)
491 var weight = 0
492 var moment = 0
493 for p in passengers(f) {
494 guard let s = p.seat else { continue }
495 let w = (p.type == "ADULT" ? 80 : 40)+(p.lap == nil ? 0 : 10)
496 weight += w
497 moment += w*(s.row-f.aircraft.base)
498 }
499 let limit = f.aircraft.limit
500 let verdict = moment < -limit ? "NOSE-HEAVY" : (moment > limit ? "TAIL-HEAVY" : "OK")
501 output.append("BALANCE \(f.name) weight \(weight) moment \(moment) limit \(limit) \(verdict)")
502 case "BOARDING":
503 let f = try flight(t[1],false)
504 let ordered = passengers(f).filter { $0.seat != nil }.sorted { a,b in
505 let ar = boardingRank(a)
506 let br = boardingRank(b)
507 if ar != br { return ar < br }
508 if ar < 3 { return seatBefore(a,b) }
509 if zone(a) != zone(b) { return zone(a) > zone(b) }
510 let ak = ["W","M","A"].firstIndex(of: a.seat!.kind)!
511 let bk = ["W","M","A"].firstIndex(of: b.seat!.kind)!
512 if ak != bk { return ak < bk }
513 if a.seat!.row != b.seat!.row { return a.seat!.row > b.seat!.row }
514 return a.seat!.letter < b.seat!.letter
515 }
516 output.append("BOARDING \(f.name) \(ordered.count)")
517 for (i,p) in ordered.enumerated() {
518 let rank = boardingRank(p)
519 let group = rank == 0 ? "PRE" : (rank == 1 ? "CABIN" : (rank == 2 ? "TIER" : "ZONE\(zone(p))"))
520 output.append("\(i+1) \(group) \(p.result) \(p.booking.pnr) \(p.name)")
521 }
522 case "AVAIL":
523 let f = try flight(t[1],false)
524 try requireCabin(f,t[2])
525 let free = f.aircraft.cabins[t[2]]!.seats.filter { !f.blocked.contains($0.id) && f.occupied[$0.id] == nil }
526 output.append("AVAIL \(f.name) \(t[2]) free \(free.count) W \(free.filter { $0.kind == "W" }.count) M \(free.filter { $0.kind == "M" }.count) A \(free.filter { $0.kind == "A" }.count) EXIT \(free.filter { f.aircraft.exits.contains($0.row) }.count)")
527 case "INFO":
528 let b = try booking(t[1],false)
529 output.append("INFO \(b.pnr) \(b.flight.name) \(b.passengers.count) \(b.checked ? "CHECKED" : "NOT-CHECKED")")
530 for p in b.passengers { output.append("\(p.name) \(p.type) \(p.cabin) \(p.result)") }
531 case "FIND":
532 var found: [String] = []
533 for f in flightOrder {
534 for p in passengers(f) where p.name == t[1] {
535 found.append("\(f.name) \(p.booking.pnr) \(p.cabin) \(p.result) \(f.closed ? "CLOSED" : "OPEN")")
536 }
537 }
538 output.append("FIND \(t[1]) \(tiers[t[1]] ?? "NONE") \(found.count)")
539 output += found
540 default: preconditionFailure("Unreachable command")
541 }
542 return output
543 }
544}
