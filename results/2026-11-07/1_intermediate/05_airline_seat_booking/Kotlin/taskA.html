1private val classes = listOf("F","J","Y")
2data class Seat(val id: String, val row: Int, val letter: Char, val block: Int, val cabin: String, val kind: String)
3class Cabin(val name: String, val first: Int, val last: Int, val blocks: List<String>) {
4 val seats = mutableListOf<Seat>()
5}
6class Aircraft(val model: String, val base: Int, val limit: Int) {
7 val cabins = linkedMapOf<String,Cabin>()
8 val seats = linkedMapOf<String,Seat>()
9 val exits = mutableSetOf<Int>()
10 val blocked = mutableSetOf<String>()
11}
12class Flight(val name: String, val aircraft: Aircraft) {
13 var closed = false
14 val limits = mutableMapOf<String,Int>()
15 val blocked = aircraft.blocked.toMutableSet()
16 val occupied = mutableMapOf<String,Passenger>()
17 val bookings = mutableListOf<Booking>()
18 val volunteers = mutableListOf<Passenger>()
19}
20class Booking(val pnr: String, val flight: Flight) {
21 var checked = false
22 val passengers = mutableListOf<Passenger>()
23}
24class Passenger(val name: String, val type: String, val pref: String?, var cabin: String, val booking: Booking) {
25 var order = 0
26 var lap: Passenger? = null
27 var parent: Passenger? = null
28 var seat: Seat? = null
29 val needsSeat get() = type != "INFANT"
30 val result get() = if (!needsSeat) "LAP" else seat?.id ?: "UNASSIGNED"
31}
32class Problem(message: String) : RuntimeException(message)
33
34class Airline {
35 private val aircraft = mutableMapOf<String,Aircraft>()
36 private val flights = linkedMapOf<String,Flight>()
37 private val bookings = mutableMapOf<String,Booking>()
38 private val used = mutableSetOf<String>()
39 private val tiers = mutableMapOf<String,String>()
40 private lateinit var defining: Aircraft
41 private var nextOrder = 0
42 private fun fail(message: String): Nothing = throw Problem(message)
43 private fun tier(p: Passenger) = tiers.getOrDefault(p.name,"NONE")
44 private fun tierRank(p: Passenger) = listOf("PLATINUM","GOLD","SILVER","NONE").indexOf(tier(p))
45 private fun flight(name: String, open: Boolean): Flight {
46 val f = flights[name] ?: fail("unknown flight $name")
47 if (open && f.closed) fail("flight $name is closed")
48 return f
49 }
50 private fun booking(pnr: String, open: Boolean): Booking {
51 val b = bookings[pnr] ?: fail("unknown PNR $pnr")
52 if (b.passengers.isEmpty()) fail("unknown PNR $pnr")
53 if (open && b.flight.closed) fail("flight ${b.flight.name} is closed")
54 return b
55 }
56 private fun passenger(b: Booking, name: String) = b.passengers.firstOrNull { it.name == name }
57 ?: fail("no passenger $name in ${b.pnr}")
58 private fun passengers(f: Flight) = f.bookings.flatMap { it.passengers }
59 private fun booked(f: Flight, cabin: String) = passengers(f).count { it.needsSeat && it.cabin == cabin }
60 private fun seats(a: Aircraft, cabin: String, blocked: Set<String>) = a.cabins.getValue(cabin).seats.count { it.id !in blocked }
61 private fun requireCabin(f: Flight, cabin: String) {
62 if (cabin !in f.aircraft.cabins) fail("no cabin $cabin on ${f.name}")
63 }
64 private fun sameBlock(a: Seat, b: Seat) = a.row == b.row && a.block == b.block
65 private fun exitAllowed(p: Passenger, s: Seat) = s.row !in p.booking.flight.aircraft.exits || p.type != "CHILD" && p.lap == null
66 private fun infantAllowed(p: Passenger, s: Seat, vacated: Set<Passenger>, proposed: Map<Passenger,Seat>): Boolean {
67 if (p.lap == null) return true
68 val f = p.booking.flight
69 for (letter in f.aircraft.cabins.getValue(s.cabin).blocks[s.block]) {
70 val q = f.occupied["${s.row}$letter"]
71 if (q != null && q !== p && q.lap != null && q !in vacated) return false
72 }
73 for ((q,target) in proposed) {
74 if (q !== p && q.lap != null && sameBlock(target,s)) return false
75 }
76 return true
77 }
78 private fun fits(p: Passenger, s: Seat, vacated: Set<Passenger> = emptySet(), proposed: Map<Passenger,Seat> = emptyMap()): Boolean {
79 val f = p.booking.flight
80 val occupant = f.occupied[s.id]
81 return s.id !in f.blocked && (occupant == null || occupant in vacated) && exitAllowed(p,s) && infantAllowed(p,s,vacated,proposed)
82 }
83 private fun seat(p: Passenger, target: Seat?) {
84 val f = p.booking.flight
85 p.seat?.let { f.occupied.remove(it.id) }
86 p.seat = target
87 if (target != null) f.occupied[target.id] = p
88 }
89 private fun select(p: Passenger, cabin: String): Seat? {
90 val candidates = p.booking.flight.aircraft.cabins.getValue(cabin).seats.filter { fits(p,it) }
91 return candidates.firstOrNull { it.kind == p.pref } ?: candidates.firstOrNull()
92 }
93 private fun reassign(f: Flight, output: MutableList<String>) {
94 for (p in passengers(f)) {
95 if (p.needsSeat && p.seat == null) {
96 val target = select(p,p.cabin)
97 if (target != null) {
98 seat(p,target)
99 output.add("REASSIGNED ${p.booking.pnr} ${p.name} ${target.id}")
100 }
101 }
102 }
103 }
104 private fun candidate(people: List<Passenger>, positions: List<Seat>): Boolean {
105 val proposed = people.zip(positions).toMap()
106 if (!people.all { fits(it,proposed.getValue(it),proposed = proposed) }) return false
107 for (p in people) seat(p,proposed.getValue(p))
108 return true
109 }
110 private fun assign(b: Booking, cabinName: String): String {
111 val people = b.passengers.filter { it.needsSeat }
112 if (people.size == 1) {
113 seat(people[0],select(people[0],cabinName))
114 return "SINGLE"
115 }
116 val cabin = b.flight.aircraft.cabins.getValue(cabinName)
117 val a = b.flight.aircraft
118 val k = people.size
119 for (stage in 1..3) {
120 val lastH = if (stage == 3) k else 2
121 for (h in 2..lastH) {
122 val width = (k+h-1)/h
123 for (row in cabin.first..cabin.last) {
124 val blocks = if (stage == 2) listOf(cabin.blocks.joinToString("")) else cabin.blocks
125 for (block in blocks) {
126 val span = if (stage == 3) width else k
127 for (x in 0..block.length-span) {
128 val positions = mutableListOf<Seat>()
129 for (j in 0 until k) {
130 val r = row+if (stage == 3) j/width else 0
131 if (r > cabin.last) break
132 val letter = block[x+if (stage == 3) j%width else j]
133 positions.add(a.seats.getValue("$r$letter"))
134 }
135 if (positions.size == k && candidate(people,positions)) return when (stage) {
136 1 -> "BLOCK"
137 2 -> "ROW"
138 else -> "STACK"
139 }
140 }
141 }
142 }
143 }
144 }
145 for (p in people) seat(p,select(p,cabinName))
146 return "SPLIT"
147 }
148 private fun targets(p: Passenger) = listOfNotNull(p,p.lap)
149 private fun remove(targets: List<Passenger>) {
150 for (p in targets) {
151 seat(p,null)
152 p.booking.flight.volunteers.remove(p)
153 p.booking.passengers.remove(p)
154 p.parent?.lap = null
155 }
156 }
157 private val seatOrder = compareBy<Passenger> { it.seat!!.row }.thenBy { it.seat!!.letter }
158 private fun load(seated: Int, count: Int): String {
159 val q = seated*1000/count
160 return "${q/10}.${q%10}%"
161 }
162 private fun boardingRank(p: Passenger) = when {
163 p.booking.passengers.any { !it.needsSeat } -> 0
164 p.cabin != "Y" -> 1
165 tierRank(p) <= 1 -> 2
166 else -> 3
167 }
168 private fun zone(p: Passenger) = (p.seat!!.row-p.booking.flight.aircraft.cabins.getValue("Y").first)/5+1
169 fun execute(line: String): List<String> {
170 val t = line.split(' ')
171 val command = t[0]
172 val output = mutableListOf<String>()
173 when (command) {
174 "AIRCRAFT" -> defining = Aircraft(t[1],t[2].toInt(),t[3].toInt())
175 "CABIN" -> {
176 val c = Cabin(t[1],t[2].toInt(),t[3].toInt(),t.drop(4))
177 for (row in c.first..c.last) {
178 for ((block,letters) in c.blocks.withIndex()) {
179 for ((i,letter) in letters.withIndex()) {
180 val window = block == 0 && i == 0 || block == c.blocks.lastIndex && i == letters.lastIndex
181 val kind = if (window) "W" else if (i == 0 || i == letters.lastIndex) "A" else "M"
182 val s = Seat("$row$letter",row,letter,block,c.name,kind)
183 c.seats.add(s)
184 defining.seats[s.id] = s
185 }
186 }
187 }
188 defining.cabins[c.name] = c
189 }
190 "EXIT" -> defining.exits.addAll(t.drop(1).map { it.toInt() })
191 "BLOCKED" -> defining.blocked.addAll(t.drop(1))
192 "END" -> {
193 aircraft[defining.model] = defining
194 output.add("AIRCRAFT ${defining.model}"+classes.filter { it in defining.cabins }.joinToString("") { " $it ${seats(defining,it,defining.blocked)}" })
195 }
196 "FLIGHT" -> {
197 if (t[1] in flights) fail("duplicate flight ${t[1]}")
198 val a = aircraft[t[2]] ?: fail("unknown aircraft ${t[2]}")
199 val f = Flight(t[1],a)
200 var result = "FLIGHT ${f.name} ${a.model}"
201 for (c in classes.filter { it in a.cabins }) {
202 val count = seats(a,c,a.blocked)
203 val limit = count*(100+t[3].toInt())/100
204 f.limits[c] = limit
205 result += " $c $count/$limit"
206 }
207 flights[f.name] = f
208 output.add(result)
209 }
210 "BOOK" -> {
211 if (t[1] in used) fail("duplicate PNR ${t[1]}")
212 val f = flight(t[2],true)
213 requireCabin(f,t[3])
214 val b = Booking(t[1],f)
215 val seen = mutableSetOf<String>()
216 val adults = mutableListOf<Passenger>()
217 val infants = mutableListOf<Passenger>()
218 for (token in t.drop(4)) {
219 val parts = token.split(':')
220 if (!seen.add(parts[0])) fail("duplicate name ${parts[0]} in ${b.pnr}")
221 val p = Passenger(parts[0],parts[1],parts.getOrNull(2),t[3],b)
222 b.passengers.add(p)
223 if (p.type == "ADULT") adults.add(p)
224 if (p.type == "INFANT") infants.add(p)
225 }
226 val present = passengers(f).map { it.name }.toSet()
227 for (p in b.passengers) {
228 if (p.name in present) fail("${p.name} already on ${f.name}")
229 }
230 if (adults.isEmpty()) fail("no adult in ${b.pnr}")
231 if (infants.size > adults.size) fail("too many infants in ${b.pnr}")
232 if (booked(f,t[3])+b.passengers.count { it.needsSeat } > f.limits.getValue(t[3])) fail("limit exceeded for ${f.name} ${t[3]}")
233 for (i in infants.indices) {
234 adults[i].lap = infants[i]
235 infants[i].parent = adults[i]
236 }
237 for (p in b.passengers) p.order = nextOrder++
238 val method = assign(b,t[3])
239 f.bookings.add(b)
240 bookings[b.pnr] = b
241 used.add(b.pnr)
242 output.add("BOOKED ${b.pnr} ${f.name} ${t[3]} $method "+b.passengers.joinToString(" ") { "${it.name}:${it.result}" })
243 }
244 "SEAT" -> {
245 val b = booking(t[1],true)
246 val p = passenger(b,t[2])
247 if (!p.needsSeat) fail("infant ${p.name} has no seat")
248 val target = b.flight.aircraft.seats[t[3]]
249 if (target == null || target.cabin != p.cabin) fail("no seat ${t[3]} in cabin ${p.cabin}")
250 if (t[3] in b.flight.blocked) fail("seat ${t[3]} is blocked")
251 if (t[3] in b.flight.occupied) fail("seat ${t[3]} is occupied")
252 if (!exitAllowed(p,target)) fail("exit row ${target.row} not allowed for ${p.name}")
253 if (!infantAllowed(p,target,emptySet(),emptyMap())) fail("infant limit reached at ${t[3]}")
254 val old = p.result
255 seat(p,target)
256 output.add("MOVED ${b.pnr} ${p.name} $old -> ${t[3]}")
257 reassign(b.flight,output)
258 }
259 "SWAP" -> {
260 val a = booking(t[1],true)
261 val p = passenger(a,t[2])
262 val b = booking(t[3],true)
263 val q = passenger(b,t[4])
264 if (p === q) fail("cannot swap ${p.name} with itself")
265 for (x in listOf(p,q)) {
266 if (!x.needsSeat) fail("infant ${x.name} has no seat")
267 }
268 if (a.flight !== b.flight) fail("${a.pnr} and ${b.pnr} are on different flights")
269 if (p.cabin != q.cabin) fail("${p.name} and ${q.name} are in different cabins")
270 for (x in listOf(p,q)) {
271 if (x.seat == null) fail("${x.name} has no seat")
272 }
273 val ps = p.seat!!
274 val qs = q.seat!!
275 val vacated = setOf(p,q)
276 val proposed = mapOf(p to qs,q to ps)
277 if (!fits(p,qs,vacated,proposed) || !fits(q,ps,vacated,proposed)) fail("swap of ${p.name} and ${q.name} breaks seating rules")
278 seat(p,null)
279 seat(q,null)
280 seat(p,qs)
281 seat(q,ps)
282 output.add("SWAPPED ${a.pnr} ${p.name} ${p.result} ${b.pnr} ${q.name} ${q.result}")
283 reassign(a.flight,output)
284 }
285 "CANCEL" -> {
286 val b = booking(t[1],true)
287 val targets = if (t.size == 2) b.passengers.toList() else {
288 val selected = targets(passenger(b,t[2]))
289 val remaining = b.passengers.filter { it !in selected }
290 if (remaining.isNotEmpty() && remaining.none { it.type == "ADULT" }) fail("${b.pnr} would have no adult")
291 b.passengers.filter { it in selected }
292 }
293 val names = targets.joinToString(" ") { it.name }
294 remove(targets)
295 output.add("CANCELLED ${b.pnr} $names")
296 reassign(b.flight,output)
297 }
298 "STATUS" -> {
299 tiers[t[1]] = t[2]
300 output.add("STATUS ${t[1]} ${t[2]}")
301 }
302 "CHECKIN" -> {
303 val b = booking(t[1],true)
304 if (b.checked) fail("${b.pnr} already checked in")
305 b.checked = true
306 output.add("CHECKIN ${b.pnr} ${b.passengers.size}")
307 }
308 "VOLUNTEER" -> {
309 val b = booking(t[1],true)
310 val p = passenger(b,t[2])
311 if (p.type != "ADULT" || p.lap != null) fail("${p.name} cannot volunteer")
312 if (p.seat == null) fail("${p.name} has no seat")
313 if (p in b.flight.volunteers) fail("${p.name} already volunteered")
314 b.flight.volunteers.add(p)
315 output.add("VOLUNTEER ${b.pnr} ${p.name} ${p.result}")
316 }
317 "UPGRADE" -> {
318 val f = flight(t[1],true)
319 val cabins = classes.filter { it in f.aircraft.cabins }
320 var count = 0
321 for (i in 0 until cabins.size-1) {
322 val higher = cabins[i]
323 val lower = cabins[i+1]
324 val targets = passengers(f).filter { it.cabin == lower && it.type == "ADULT" && it.lap == null && tier(it) != "NONE" }
325 .sortedWith(compareBy<Passenger> { tierRank(it) }.thenBy { it.order })
326 for (p in targets) {
327 if (booked(f,higher) >= f.limits.getValue(higher)) break
328 val target = select(p,higher) ?: break
329 val old = p.result
330 p.cabin = higher
331 seat(p,target)
332 output.add("UPGRADED ${p.booking.pnr} ${p.name} $lower $old -> $higher ${target.id}")
333 count++
334 }
335 }
336 reassign(f,output)
337 output.add("UPGRADE ${f.name} $count")
338 }
339 "CLOSE" -> {
340 val f = flight(t[1],true)
341 for (b in f.bookings) {
342 if (b.passengers.isNotEmpty() && !b.checked) {
343 output.add("NOSHOW ${b.pnr} ${b.passengers.size}")
344 remove(b.passengers.toList())
345 }
346 }
347 reassign(f,output)
348 var count = 0
349 var compensation = 0
350 for (c in classes.filter { it in f.aircraft.cabins }) {
351 val volunteers = f.volunteers.filter { it.seat != null && it.cabin == c }
352 for (p in volunteers) {
353 val useful = passengers(f).any { it.needsSeat && it.seat == null && it.cabin == c && fits(it,p.seat!!,setOf(p)) }
354 if (useful) {
355 val value = if (c == "F") 500 else if (c == "J") 300 else 150
356 output.add("DENIED VOLUNTARY ${p.booking.pnr} ${p.name} $c $value")
357 count++
358 compensation += value
359 remove(targets(p))
360 reassign(f,output)
361 }
362 }
363 }
364 for (p in passengers(f)) {
365 if (p.needsSeat && p.seat == null) {
366 var value = if (p.cabin == "F") 1000 else if (p.cabin == "J") 600 else 300
367 if (p.type == "CHILD") value /= 2
368 output.add("DENIED INVOLUNTARY ${p.booking.pnr} ${p.name} ${p.cabin} $value")
369 count++
370 compensation += value
371 remove(targets(p))
372 }
373 }
374 f.closed = true
375 output.add("CLOSED ${f.name} denied $count compensation $compensation")
376 }
377 "BLOCK", "UNBLOCK" -> {
378 val f = flight(t[1],true)
379 if (t[2] !in f.aircraft.seats) fail("no seat ${t[2]} on ${f.name}")
380 val blocked = t[2] in f.blocked
381 if (command == "BLOCK" && blocked) fail("seat ${t[2]} is blocked")
382 if (command == "UNBLOCK" && !blocked) fail("seat ${t[2]} is not blocked")
383 output.add("$command ${f.name} ${t[2]}")
384 if (command == "BLOCK") {
385 f.blocked.add(t[2])
386 val p = f.occupied[t[2]]
387 if (p != null) {
388 seat(p,null)
389 seat(p,select(p,p.cabin))
390 output.add("DISPLACED ${p.booking.pnr} ${p.name} ${t[2]} -> ${p.result}")
391 }
392 } else f.blocked.remove(t[2])
393 reassign(f,output)
394 }
395 "MAP" -> {
396 val f = flight(t[1],false)
397 output.add("MAP ${f.name}")
398 for (name in classes) {
399 val c = f.aircraft.cabins[name] ?: continue
400 output.add("CABIN $name")
401 output.add(" "+c.blocks.joinToString(" "))
402 for (row in c.first..c.last) {
403 val blocks = c.blocks.map { block -> block.map { letter ->
404 val id = "$row$letter"
405 when {
406 id in f.blocked -> '#'
407 id in f.occupied -> 'X'
408 row in f.aircraft.exits -> 'E'
409 else -> '.'
410 }
411 }.joinToString("") }
412 output.add(row.toString().padStart(2)+" "+blocks.joinToString(" "))
413 }
414 }
415 }
416 "MANIFEST" -> {
417 val f = flight(t[1],false)
418 val all = passengers(f)
419 val seated = all.filter { it.seat != null }.sortedWith(seatOrder)
420 val unassigned = all.filter { it.needsSeat && it.seat == null }
421 output.add("MANIFEST ${f.name} seated ${seated.size} unassigned ${unassigned.size} infants ${all.count { !it.needsSeat }}")
422 for (p in seated+unassigned) {
423 var result = "${p.seat?.id ?: "---"} ${p.cabin} ${p.booking.pnr} ${p.name} ${p.type}"
424 p.lap?.let { result += " +${it.name}" }
425 if (tier(p) != "NONE") result += " ${tier(p)}"
426 if (p.booking.checked) result += " CHECKED"
427 if (p in f.volunteers) result += " VOL"
428 output.add(result)
429 }
430 }
431 "LOADFACTOR" -> {
432 val f = flight(t[1],false)
433 output.add("LOADFACTOR ${f.name}")
434 var totalSeated = 0
435 var totalSeats = 0
436 for (c in classes.filter { it in f.aircraft.cabins }) {
437 val count = seats(f.aircraft,c,f.blocked)
438 val seated = passengers(f).count { it.cabin == c && it.seat != null }
439 output.add("$c $seated/$count ${load(seated,count)} booked ${booked(f,c)}/${f.limits.getValue(c)}")
440 totalSeated += seated
441 totalSeats += count
442 }
443 output.add("ALL $totalSeated/$totalSeats ${load(totalSeated,totalSeats)}")
444 }
445 "BALANCE" -> {
446 val f = flight(t[1],false)
447 var weight = 0
448 var moment = 0
449 for (p in passengers(f)) {
450 val s = p.seat ?: continue
451 val w = (if (p.type == "ADULT") 80 else 40)+(if (p.lap == null) 0 else 10)
452 weight += w
453 moment += w*(s.row-f.aircraft.base)
454 }
455 val limit = f.aircraft.limit
456 val verdict = if (moment < -limit) "NOSE-HEAVY" else if (moment > limit) "TAIL-HEAVY" else "OK"
457 output.add("BALANCE ${f.name} weight $weight moment $moment limit $limit $verdict")
458 }
459 "BOARDING" -> {
460 val f = flight(t[1],false)
461 val ordered = passengers(f).filter { it.seat != null }.sortedWith { a,b ->
462 val ar = boardingRank(a)
463 val br = boardingRank(b)
464 when {
465 ar != br -> ar.compareTo(br)
466 ar < 3 -> seatOrder.compare(a,b)
467 zone(a) != zone(b) -> zone(b).compareTo(zone(a))
468 a.seat!!.kind != b.seat!!.kind -> "WMA".indexOf(a.seat!!.kind).compareTo("WMA".indexOf(b.seat!!.kind))
469 a.seat!!.row != b.seat!!.row -> b.seat!!.row.compareTo(a.seat!!.row)
470 else -> a.seat!!.letter.compareTo(b.seat!!.letter)
471 }
472 }
473 output.add("BOARDING ${f.name} ${ordered.size}")
474 for ((i,p) in ordered.withIndex()) {
475 val group = when (boardingRank(p)) {
476 0 -> "PRE"
477 1 -> "CABIN"
478 2 -> "TIER"
479 else -> "ZONE${zone(p)}"
480 }
481 output.add("${i+1} $group ${p.result} ${p.booking.pnr} ${p.name}")
482 }
483 }
484 "AVAIL" -> {
485 val f = flight(t[1],false)
486 requireCabin(f,t[2])
487 val free = f.aircraft.cabins.getValue(t[2]).seats.filter { it.id !in f.blocked && it.id !in f.occupied }
488 output.add("AVAIL ${f.name} ${t[2]} free ${free.size} W ${free.count { it.kind == "W" }} M ${free.count { it.kind == "M" }} A ${free.count { it.kind == "A" }} EXIT ${free.count { it.row in f.aircraft.exits }}")
489 }
490 "INFO" -> {
491 val b = booking(t[1],false)
492 output.add("INFO ${b.pnr} ${b.flight.name} ${b.passengers.size} ${if (b.checked) "CHECKED" else "NOT-CHECKED"}")
493 for (p in b.passengers) output.add("${p.name} ${p.type} ${p.cabin} ${p.result}")
494 }
495 "FIND" -> {
496 val found = mutableListOf<String>()
497 for (f in flights.values) {
498 for (p in passengers(f)) {
499 if (p.name == t[1]) found.add("${f.name} ${p.booking.pnr} ${p.cabin} ${p.result} ${if (f.closed) "CLOSED" else "OPEN"}")
500 }
501 }
502 output.add("FIND ${t[1]} ${tiers.getOrDefault(t[1],"NONE")} ${found.size}")
503 output.addAll(found)
504 }
505 }
506 return output
507 }
508}

510fun main() {
511 val input = System.`in`.bufferedReader()
512 val output = System.out.bufferedWriter()
513 val app = Airline()
514 repeat(input.readLine().toInt()) {
515 val lines = try {
516 app.execute(input.readLine())
517 } catch (problem: Problem) {
518 listOf("ERROR ${problem.message}")
519 }
520 for (line in lines) {
521 output.write(line)
522 output.newLine()
523 }
524 }
525 output.flush()
526}
