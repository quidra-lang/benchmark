90fn number(s: &str) -> usize {

99impl World {
100 fn tier(&self, id: usize) -> usize {
101 self.tiers.get(&self.people[id].name).copied().unwrap_or(3)
102 }
103 fn flight_id(&self, name: &str, open: bool) -> Result<usize, String> {
104 let &id = self
105 .flight_ids
106 .get(name)
107 .ok_or_else(|| format!("unknown flight {name}"))?;
108 if open && self.flights[id].closed {
109 return Err(format!("flight {name} is closed"));
110 }
111 Ok(id)
112 }
113 fn active_count(&self, b: usize) -> usize {
114 self.bookings[b]
115 .people
116 .iter()
117 .filter(|&&i| self.people[i].active)
118 .count()
119 }
120 fn booking_id(&self, pnr: &str, open: bool) -> Result<usize, String> {
121 let id = self
122 .booking_ids
123 .get(pnr)
124 .copied()
125 .filter(|&i| self.active_count(i) > 0)
126 .ok_or_else(|| format!("unknown PNR {pnr}"))?;
127 if open && self.flights[self.bookings[id].flight].closed {
128 return Err(format!(
129 "flight {} is closed",
130 self.flights[self.bookings[id].flight].name
131 ));
132 }
133 Ok(id)
134 }
135 fn passenger_id(&self, b: usize, name: &str) -> Result<usize, String> {
136 self.bookings[b]
137 .people
138 .iter()
139 .copied()
140 .find(|&i| self.people[i].active && self.people[i].name == name)
141 .ok_or_else(|| format!("no passenger {name} in {}", self.bookings[b].pnr))
142 }
143 fn resolve(&self, pnr: &str, name: &str) -> Result<usize, String> {
144 self.passenger_id(self.booking_id(pnr, true)?, name)
145 }
146 fn has_infant(&self, id: usize) -> bool {
147 self.people[id].kind == 0 && self.people[id].lap.is_some_and(|i| self.people[i].active)
148 }
149 fn booking_infants(&self, b: usize) -> bool {
150 self.bookings[b]
151 .people
152 .iter()
153 .any(|&i| self.people[i].active && self.people[i].kind == 2)
154 }
155 fn booked_count(&self, f: usize, c: usize) -> usize {
156 self.flights[f]
157 .passengers
158 .iter()
159 .filter(|&&i| {
160 self.people[i].active && self.people[i].kind != 2 && self.people[i].cabin == c
161 })
162 .count()
163 }
164 fn exit_allowed(&self, id: usize, plane: &Aircraft, target: usize) -> bool {
165 !plane.exits[plane.seats[target].row] || self.people[id].kind == 0 && !self.has_infant(id)
166 }
167 fn infant_allowed(
168 &self,
169 f: usize,
170 id: usize,
171 target: usize,
172 moving: &[usize],
173 ids: &[usize],
174 assigned: &[usize],
175 ) -> bool {
176 if !self.has_infant(id) {
177 return true;
178 }
179 let plane = &self.aircraft[self.flights[f].model];
180 let s = &plane.seats[target];
181 for letter in plane.cabins[s.cabin].blocks[s.block].bytes() {
182 let peer = plane.lookup[s.row * 26 + (letter - b'A') as usize].unwrap();
183 if let Some(owner) = self.flights[f].occupants[peer] {
184 if owner != id && !moving.contains(&owner) && self.has_infant(owner) {
185 return false;
186 }
187 }
188 }
189 for (&other, &seat) in ids.iter().zip(assigned) {
190 let t = &plane.seats[seat];
191 if other != id && self.has_infant(other) && t.row == s.row && t.block == s.block {
192 return false;
193 }
194 }
195 true
196 }
197 fn fits(&self, f: usize, ids: &[usize], assigned: &[usize], vacated: &[usize]) -> bool {
198 let plane = &self.aircraft[self.flights[f].model];
199 let mut moving = ids.to_vec();
200 moving.extend_from_slice(vacated);
201 for (&id, &target) in ids.iter().zip(assigned) {
202 if self.flights[f].blocked[target]
203 || self.flights[f].occupants[target].is_some_and(|owner| !moving.contains(&owner))
204 || !self.exit_allowed(id, plane, target)
205 || !self.infant_allowed(f, id, target, &moving, ids, assigned)
206 {
207 return false;
208 }
209 }
210 true
211 }
212 fn choose_seat(&self, f: usize, id: usize, c: usize) -> Option<usize> {
213 let plane = &self.aircraft[self.flights[f].model];
214 let mut first = None;
215 for (target, s) in plane.seats.iter().enumerate() {
216 if s.cabin != c
217 || self.flights[f].blocked[target]
218 || self.flights[f].occupants[target].is_some()
219 || !self.exit_allowed(id, plane, target)
220 || !self.infant_allowed(f, id, target, &[], &[], &[])
221 {
222 continue;
223 }
224 if first.is_none() {
225 first = Some(target);
226 }
227 if self.people[id].preference.is_none_or(|kind| kind == s.kind) {
228 return Some(target);
229 }
230 }
231 first
232 }
233 fn seat_person(&mut self, f: usize, id: usize, target: Option<usize>) {
234 if let Some(old) = self.people[id].seat {
235 self.flights[f].occupants[old] = None;
236 }
237 self.people[id].seat = target;
238 if let Some(target) = target {
239 self.flights[f].occupants[target] = Some(id);
240 }
241 }
242 fn reassign(&mut self, f: usize) -> Vec<String> {
243 let mut out = Vec::new();
244 for position in 0..self.flights[f].passengers.len() {
245 let id = self.flights[f].passengers[position];
246 let p = &self.people[id];
247 if !p.active || p.kind == 2 || p.seat.is_some() {
248 continue;
249 }
250 if let Some(target) = self.choose_seat(f, id, p.cabin) {
251 self.seat_person(f, id, Some(target));
252 let p = &self.people[id];
253 out.push(format!(
254 "REASSIGNED {} {} {}",
255 self.bookings[p.booking].pnr,
256 p.name,
257 self.aircraft[self.flights[f].model].seat_name(Some(target))
258 ));
259 }
260 }
261 out
262 }
263 fn remove_person(&mut self, id: usize) {
264 if !self.people[id].active {
265 return;
266 }
267 let (kind, lap) = (self.people[id].kind, self.people[id].lap);
268 if let Some(lap) = lap {
269 if kind == 0 {
270 self.remove_person(lap);
271 }
272 if kind == 2 {
273 self.people[lap].lap = None;
274 }
275 }
276 self.seat_person(self.bookings[self.people[id].booking].flight, id, None);
277 self.people[id].active = false;
278 self.people[id].volunteer = false;
279 self.people[id].lap = None;
280 }
281 fn passenger_result(&self, id: usize) -> String {
282 let p = &self.people[id];
283 if p.kind == 2 {
284 "LAP".to_owned()
285 } else {
286 self.aircraft[self.flights[self.bookings[p.booking].flight].model].seat_name(p.seat)
287 }
288 }
289 fn assign_booking(&mut self, f: usize, c: usize, ids: &[usize]) -> &'static str {
290 if ids.len() == 1 {
291 self.seat_person(f, ids[0], self.choose_seat(f, ids[0], c));
292 return "SINGLE";
293 }
294 let model = self.flights[f].model;
295 let cabin = self.aircraft[model].cabins[c].clone();
296 let count = ids.len();
297 for stage in 0..3 {
298 let blocks = if stage == 1 {
299 vec![cabin.blocks.concat()]
300 } else {
301 cabin.blocks.clone()
302 };
303 let heights = if stage == 2 { 2..=count } else { 1..=1 };
304 for height in heights {
305 let width = if stage == 2 {
306 count.div_ceil(height)
307 } else {
308 count
309 };
310 let used_rows = count.div_ceil(width);
311 for row in cabin.from..=cabin.to {
312 if row + used_rows - 1 > cabin.to {
313 break;
314 }
315 for block in &blocks {
316 if block.len() < width {
317 continue;
318 }
319 for x in 0..=block.len() - width {
320 let candidate: Vec<usize> = (0..count)
321 .map(|j| {
322 self.aircraft[model].lookup[(row + j / width) * 26
323 + (block.as_bytes()[x + j % width] - b'A') as usize]
324 .unwrap()
325 })
326 .collect();
327 if !self.fits(f, ids, &candidate, &[]) {
328 continue;
329 }
330 for (&id, &seat) in ids.iter().zip(&candidate) {
331 self.seat_person(f, id, Some(seat));
332 }
333 return ["BLOCK", "ROW", "STACK"][stage];
334 }
335 }
336 }
337 }
338 }
339 for &id in ids {
340 self.seat_person(f, id, self.choose_seat(f, id, c));
341 }
342 "SPLIT"
343 }
344 fn execute(&mut self, v: &[&str]) -> Result<Vec<String>, String> {
345 let mut out = Vec::new();
346 match v[0] {
347 "AIRCRAFT" => {
348 self.building = Aircraft::new();
349 self.building.model = v[1].to_owned();
350 self.building.base = number(v[2]) as i32;
351 self.building.limit = number(v[3]) as i32;
352 }
353 "CABIN" => {
354 let c = cabin_id(v[1]);
355 let cabin = Cabin {
356 from: number(v[2]),
357 to: number(v[3]),
358 blocks: v[4..].iter().map(|s| (*s).to_owned()).collect(),
359 };
360 for row in cabin.from..=cabin.to {
361 for (b, block) in cabin.blocks.iter().enumerate() {
362 for (position, letter) in block.bytes().enumerate() {
363 let kind = if b == 0 && position == 0
364 || b + 1 == cabin.blocks.len() && position + 1 == block.len()
365 {
366 0
367 } else if position == 0 || position + 1 == block.len() {
368 2
369 } else {
370 1
371 };
372 self.building.lookup[row * 26 + (letter - b'A') as usize] =
373 Some(self.building.seats.len());
374 self.building.seats.push(Seat {
375 row,
376 cabin: c,
377 block: b,
378 kind,
379 letter,
380 });
381 self.building.blocked.push(false);
382 }
383 }
384 }
385 self.building.cabins[c] = cabin;
386 }
387 "EXIT" => {
388 for value in &v[1..] {
389 self.building.exits[number(value)] = true;
390 }
391 }
392 "BLOCKED" => {
393 for value in &v[1..] {
394 self.building.blocked[self.building.lookup[seat_key(value)].unwrap()] = true;
395 }
396 }
397 "END" => {
398 for (i, seat) in self.building.seats.iter().enumerate() {
399 if !self.building.blocked[i] {
400 self.building.capacity[seat.cabin] += 1;
401 }
402 }
403 let mut line = format!("AIRCRAFT {}", self.building.model);
404 for c in 0..3 {
405 if self.building.cabins[c].from > 0 {
406 line.push_str(&format!(" {} {}", CLASSES[c], self.building.capacity[c]));
407 }
408 }
409 let plane = std::mem::replace(&mut self.building, Aircraft::new());
410 self.model_ids
411 .insert(plane.model.clone(), self.aircraft.len());
412 self.aircraft.push(plane);
413 out.push(line);
414 }
415 "FLIGHT" => {
416 if self.flight_ids.contains_key(v[1]) {
417 return Err(format!("duplicate flight {}", v[1]));
418 }
419 let &model = self
420 .model_ids
421 .get(v[2])
422 .ok_or_else(|| format!("unknown aircraft {}", v[2]))?;
423 let plane = &self.aircraft[model];
424 let mut flight = Flight {
425 name: v[1].to_owned(),
426 model,
427 closed: false,
428 limit: [0; 3],
429 blocked: plane.blocked.clone(),
430 occupants: vec![None; plane.seats.len()],
431 passengers: Vec::new(),
432 volunteers: Vec::new(),
433 };
434 let mut line = format!("FLIGHT {} {}", flight.name, plane.model);
435 for c in 0..3 {
436 flight.limit[c] = plane.capacity[c] * (100 + number(v[3])) / 100;
437 if plane.cabins[c].from > 0 {
438 line.push_str(&format!(
439 " {} {}/{}",
440 CLASSES[c], plane.capacity[c], flight.limit[c]
441 ));
442 }
443 }
444 self.flight_ids
445 .insert(flight.name.clone(), self.flights.len());
446 self.flights.push(flight);
447 out.push(line);
448 }
449 "BOOK" => {
450 if self.booking_ids.contains_key(v[1]) {
451 return Err(format!("duplicate PNR {}", v[1]));
452 }
453 let (f, c) = (self.flight_id(v[2], true)?, cabin_id(v[3]));
454 if self.aircraft[self.flights[f].model].cabins[c].from == 0 {
455 return Err(format!("no cabin {} on {}", v[3], v[2]));
456 }
457 let mut draft = Vec::new();
458 let mut names = HashSet::new();
459 let (mut adults, mut infants) = (Vec::new(), Vec::new());
460 for item in &v[4..] {
461 let fields: Vec<&str> = item.split(':').collect();
462 if !names.insert(fields[0]) {
463 return Err(format!("duplicate name {} in {}", fields[0], v[1]));
464 }
465 let kind = TYPES.iter().position(|&s| s == fields[1]).unwrap();
466 let preference = fields.get(2).map(|s| "WMA".find(*s).unwrap());
467 if kind == 0 {
468 adults.push(draft.len());
469 }
470 if kind == 2 {
471 infants.push(draft.len());
472 }
473 draft.push(Person {
474 name: fields[0].to_owned(),
475 booking: self.bookings.len(),
476 kind,
477 preference,
478 cabin: c,
479 seat: None,
480 lap: None,
481 active: true,
482 volunteer: false,
483 });
484 }
485 for p in &draft {
486 if self.flights[f]
487 .passengers
488 .iter()
489 .any(|&id| self.people[id].active && self.people[id].name == p.name)
490 {
491 return Err(format!("{} already on {}", p.name, v[2]));
492 }
493 }
494 if adults.is_empty() {
495 return Err(format!("no adult in {}", v[1]));
496 }
497 if infants.len() > adults.len() {
498 return Err(format!("too many infants in {}", v[1]));
499 }
500 if self.booked_count(f, c) + draft.len() - infants.len() > self.flights[f].limit[c]
501 {
502 return Err(format!("limit exceeded for {} {}", v[2], v[3]));
503 }
504 let base = self.people.len();
505 for (i, &infant) in infants.iter().enumerate() {
506 draft[adults[i]].lap = Some(base + infant);
507 draft[infant].lap = Some(base + adults[i]);
508 }
509 let mut booking = Booking {
510 pnr: v[1].to_owned(),
511 flight: f,
512 checked: false,
513 people: Vec::new(),
514 };
515 let mut needs = Vec::new();
516 for p in draft {
517 let id = self.people.len();
518 if p.kind != 2 {
519 needs.push(id);
520 }
521 self.people.push(p);
522 booking.people.push(id);
523 self.flights[f].passengers.push(id);
524 }
525 let b = self.bookings.len();
526 self.booking_ids.insert(booking.pnr.clone(), b);
527 self.bookings.push(booking);
528 let method = self.assign_booking(f, c, &needs);
529 let mut line = format!("BOOKED {} {} {} {method}", v[1], v[2], v[3]);
530 for &id in &self.bookings[b].people {
531 line.push_str(&format!(
532 " {}:{}",
533 self.people[id].name,
534 self.passenger_result(id)
535 ));
536 }
537 out.push(line);
538 }
539 "STATUS" => {
540 self.tiers.insert(
541 v[1].to_owned(),
542 TIERS.iter().position(|&s| s == v[2]).unwrap(),
543 );
544 out.push(format!("STATUS {} {}", v[1], v[2]));
545 }
546 "FIND" => {
547 let membership = self.tiers.get(v[1]).copied().unwrap_or(3);
548 let mut found = Vec::new();
549 for f in &self.flights {
550 for &id in &f.passengers {
551 let p = &self.people[id];
552 if p.active && p.name == v[1] {
553 found.push(format!(
554 "{} {} {} {} {}",
555 f.name,
556 self.bookings[p.booking].pnr,
557 CLASSES[p.cabin],
558 self.passenger_result(id),
559 if f.closed { "CLOSED" } else { "OPEN" }
560 ));
561 }
562 }
563 }
564 out.push(format!(
565 "FIND {} {} {}",
566 v[1],
567 TIERS[membership],
568 found.len()
569 ));
570 out.extend(found);
571 }
572 "SEAT" => {
573 let id = self.resolve(v[1], v[2])?;
574 let p = &self.people[id];
575 let f = self.bookings[p.booking].flight;
576 let plane = &self.aircraft[self.flights[f].model];
577 if p.kind == 2 {
578 return Err(format!("infant {} has no seat", p.name));
579 }
580 let target = plane.lookup[seat_key(v[3])]
581 .filter(|&seat| plane.seats[seat].cabin == p.cabin)
582 .ok_or_else(|| format!("no seat {} in cabin {}", v[3], CLASSES[p.cabin]))?;
583 if self.flights[f].blocked[target] {
584 return Err(format!("seat {} is blocked", v[3]));
585 }
586 if self.flights[f].occupants[target].is_some() {
587 return Err(format!("seat {} is occupied", v[3]));
588 }
589 if !self.exit_allowed(id, plane, target) {
590 return Err(format!(
591 "exit row {} not allowed for {}",
592 plane.seats[target].row, p.name
593 ));
594 }
595 if !self.infant_allowed(f, id, target, &[], &[], &[]) {
596 return Err(format!("infant limit reached at {}", v[3]));
597 }
598 let old = plane.seat_name(p.seat);
599 self.seat_person(f, id, Some(target));
600 out.push(format!("MOVED {} {} {old} -> {}", v[1], v[2], v[3]));
601 out.extend(self.reassign(f));
602 }
603 "SWAP" => {
604 let first = self.resolve(v[1], v[2])?;
605 let second = self.resolve(v[3], v[4])?;
606 if first == second {
607 return Err(format!("cannot swap {} with itself", v[2]));
608 }
609 for id in [first, second] {
610 if self.people[id].kind == 2 {
611 return Err(format!("infant {} has no seat", self.people[id].name));
612 }
613 }
614 let f = self.bookings[self.people[first].booking].flight;
615 if f != self.bookings[self.people[second].booking].flight {
616 return Err(format!("{} and {} are on different flights", v[1], v[3]));
617 }
618 if self.people[first].cabin != self.people[second].cabin {
619 return Err(format!("{} and {} are in different cabins", v[2], v[4]));
620 }
621 for id in [first, second] {
622 if self.people[id].seat.is_none() {
623 return Err(format!("{} has no seat", self.people[id].name));
624 }
625 }
626 let (old_first, old_second) = (
627 self.people[first].seat.unwrap(),
628 self.people[second].seat.unwrap(),
629 );
630 if !self.fits(f, &[first, second], &[old_second, old_first], &[]) {
631 return Err(format!(
632 "swap of {} and {} breaks seating rules",
633 v[2], v[4]
634 ));
635 }
636 self.flights[f].occupants[old_first] = None;
637 self.flights[f].occupants[old_second] = None;
638 self.people[first].seat = None;
639 self.people[second].seat = None;
640 self.seat_person(f, first, Some(old_second));
641 self.seat_person(f, second, Some(old_first));
642 let plane = &self.aircraft[self.flights[f].model];
643 out.push(format!(
644 "SWAPPED {} {} {} {} {} {}",
645 v[1],
646 v[2],
647 plane.seat_name(Some(old_second)),
648 v[3],
649 v[4],
650 plane.seat_name(Some(old_first))
651 ));
652 out.extend(self.reassign(f));
653 }
654 "CANCEL" => {
655 let b = self.booking_id(v[1], true)?;
656 let target = if v.len() == 3 {
657 Some(self.passenger_id(b, v[2])?)
658 } else {
659 None
660 };
661 let removed: Vec<usize> = self.bookings[b]
662 .people
663 .iter()
664 .copied()
665 .filter(|&id| {
666 self.people[id].active
667 && target.is_none_or(|t| {
668 id == t
669 || self.people[t].kind == 0 && self.people[t].lap == Some(id)
670 })
671 })
672 .collect();
673 let remaining: Vec<usize> = self.bookings[b]
674 .people
675 .iter()
676 .copied()
677 .filter(|&id| self.people[id].active && !removed.contains(&id))
678 .collect();
679 if target.is_some()
680 && !remaining.is_empty()
681 && !remaining.iter().any(|&id| self.people[id].kind == 0)
682 {
683 return Err(format!("{} would have no adult", v[1]));
684 }
685 let mut line = format!("CANCELLED {}", v[1]);
686 for &id in &removed {
687 line.push_str(&format!(" {}", self.people[id].name));
688 }
689 for id in removed {
690 self.remove_person(id);
691 }
692 out.push(line);
693 out.extend(self.reassign(self.bookings[b].flight));
694 }
695 "CHECKIN" | "VOLUNTEER" | "INFO" => {
696 let b = self.booking_id(v[1], v[0] != "INFO")?;
697 if v[0] == "CHECKIN" {
698 if self.bookings[b].checked {
699 return Err(format!("{} already checked in", v[1]));
700 }
701 self.bookings[b].checked = true;
702 out.push(format!("CHECKIN {} {}", v[1], self.active_count(b)));
703 } else if v[0] == "INFO" {
704 let book = &self.bookings[b];
705 out.push(format!(
706 "INFO {} {} {} {}",
707 book.pnr,
708 self.flights[book.flight].name,
709 self.active_count(b),
710 if book.checked {
711 "CHECKED"
712 } else {
713 "NOT-CHECKED"
714 }
715 ));
716 for &id in &book.people {
717 let p = &self.people[id];
718 if p.active {
719 out.push(format!(
720 "{} {} {} {}",
721 p.name,
722 TYPES[p.kind],
723 CLASSES[p.cabin],
724 self.passenger_result(id)
725 ));
726 }
727 }
728 } else {
729 let id = self.passenger_id(b, v[2])?;
730 let p = &self.people[id];
731 if p.kind != 0 || self.has_infant(id) {
732 return Err(format!("{} cannot volunteer", p.name));
733 }
734 if p.seat.is_none() {
735 return Err(format!("{} has no seat", p.name));
736 }
737 if p.volunteer {
738 return Err(format!("{} already volunteered", p.name));
739 }
740 self.people[id].volunteer = true;
741 self.flights[self.bookings[b].flight].volunteers.push(id);
742 out.push(format!(
743 "VOLUNTEER {} {} {}",
744 v[1],
745 v[2],
746 self.passenger_result(id)
747 ));
748 }
749 }
750 _ => {
751 let modifying = ["UPGRADE", "CLOSE", "BLOCK", "UNBLOCK"].contains(&v[0]);
752 let f = self.flight_id(v[1], modifying)?;
753 out.extend(self.flight_command(f, v)?);
754 }
755 }
756 Ok(out)
757 }
758 fn flight_command(&mut self, f: usize, v: &[&str]) -> Result<Vec<String>, String> {
759 let mut out = Vec::new();
760 let model = self.flights[f].model;
761 let name = self.flights[f].name.clone();
762 match v[0] {
763 "BLOCK" | "UNBLOCK" => {
764 let target = self.aircraft[model].lookup[seat_key(v[2])]
765 .ok_or_else(|| format!("no seat {} on {name}", v[2]))?;
766 if v[0] == "BLOCK" && self.flights[f].blocked[target] {
767 return Err(format!("seat {} is blocked", v[2]));
768 }
769 if v[0] == "UNBLOCK" && !self.flights[f].blocked[target] {
770 return Err(format!("seat {} is not blocked", v[2]));
771 }
772 self.flights[f].blocked[target] = v[0] == "BLOCK";
773 out.push(format!("{} {name} {}", v[0], v[2]));
774 if v[0] == "BLOCK" {
775 if let Some(id) = self.flights[f].occupants[target] {
776 self.seat_person(f, id, None);
777 let seat = self.choose_seat(f, id, self.people[id].cabin);
778 self.seat_person(f, id, seat);
779 out.push(format!(
780 "DISPLACED {} {} {} -> {}",
781 self.bookings[self.people[id].booking].pnr,
782 self.people[id].name,
783 v[2],
784 self.passenger_result(id)
785 ));
786 }
787 }
788 out.extend(self.reassign(f));
789 }
790 "UPGRADE" => {
791 let cabins: Vec<usize> = (0..3)
792 .filter(|&c| self.aircraft[model].cabins[c].from > 0)
793 .collect();
794 let mut upgraded = 0;
795 for pair in cabins.windows(2) {
796 let (higher, lower) = (pair[0], pair[1]);
797 let mut booked = self.booked_count(f, higher);
798 let mut targets: Vec<(usize, usize)> = self.flights[f]
799 .passengers
800 .iter()
801 .copied()
802 .filter(|&id| {
803 let p = &self.people[id];
804 p.active
805 && p.cabin == lower
806 && p.kind == 0
807 && !self.has_infant(id)
808 && self.tier(id) < 3
809 })
810 .map(|id| (self.tier(id), id))
811 .collect();
812 targets.sort_by_key(|t| t.0);
813 for (_, id) in targets {
814 if booked >= self.flights[f].limit[higher] {
815 break;
816 }
817 let Some(target) = self.choose_seat(f, id, higher) else {
818 break;
819 };
820 let old = self.passenger_result(id);
821 self.people[id].cabin = higher;
822 self.seat_person(f, id, Some(target));
823 booked += 1;
824 upgraded += 1;
825 out.push(format!(
826 "UPGRADED {} {} {} {old} -> {} {}",
827 self.bookings[self.people[id].booking].pnr,
828 self.people[id].name,
829 CLASSES[lower],
830 CLASSES[higher],
831 self.aircraft[model].seat_name(Some(target))
832 ));
833 }
834 }
835 out.extend(self.reassign(f));
836 out.push(format!("UPGRADE {name} {upgraded}"));
837 }
838 "CLOSE" => {
839 for b in 0..self.bookings.len() {
840 if self.bookings[b].flight != f
841 || self.bookings[b].checked
842 || self.active_count(b) == 0
843 {
844 continue;
845 }
846 out.push(format!(
847 "NOSHOW {} {}",
848 self.bookings[b].pnr,
849 self.active_count(b)
850 ));
851 for position in 0..self.bookings[b].people.len() {
852 let id = self.bookings[b].people[position];
853 self.remove_person(id);
854 }
855 }
856 out.extend(self.reassign(f));
857 let (mut denied, mut compensation) = (0, 0);
858 let voluntary = [500, 300, 150];
859 let involuntary = [1000, 600, 300];
860 for c in 0..3 {
861 let targets: Vec<usize> = self.flights[f]
862 .volunteers
863 .iter()
864 .copied()
865 .filter(|&id| {
866 let p = &self.people[id];
867 p.active && p.volunteer && p.seat.is_some() && p.cabin == c
868 })
869 .collect();
870 for id in targets {
871 let target = self.people[id].seat.unwrap();
872 let useful = self.flights[f].passengers.iter().copied().any(|other| {
873 let p = &self.people[other];
874 p.active
875 && p.kind != 2
876 && p.cabin == c
877 && p.seat.is_none()
878 && self.fits(f, &[other], &[target], &[id])
879 });
880 if !useful {
881 continue;
882 }
883 out.push(format!(
884 "DENIED VOLUNTARY {} {} {} {}",
885 self.bookings[self.people[id].booking].pnr,
886 self.people[id].name,
887 CLASSES[c],
888 voluntary[c]
889 ));
890 denied += 1;
891 compensation += voluntary[c];
892 self.remove_person(id);
893 out.extend(self.reassign(f));
894 }
895 }
896 for position in 0..self.flights[f].passengers.len() {
897 let id = self.flights[f].passengers[position];
898 let p = &self.people[id];
899 if !p.active || p.kind == 2 || p.seat.is_some() {
900 continue;
901 }
902 let value = involuntary[p.cabin] / if p.kind == 1 { 2 } else { 1 };
903 out.push(format!(
904 "DENIED INVOLUNTARY {} {} {} {value}",
905 self.bookings[p.booking].pnr, p.name, CLASSES[p.cabin]
906 ));
907 denied += 1;
908 compensation += value;
909 self.remove_person(id);
910 }
911 self.flights[f].closed = true;
912 out.push(format!(
913 "CLOSED {name} denied {denied} compensation {compensation}"
914 ));
915 }
916 "MAP" => {
917 out.push(format!("MAP {name}"));
918 let plane = &self.aircraft[model];
919 for c in 0..3 {
920 let cabin = &plane.cabins[c];
921 if cabin.from == 0 {
922 continue;
923 }
924 out.push(format!("CABIN {}", CLASSES[c]));
925 out.push(format!(" {}", cabin.blocks.join(" ")));
926 for row in cabin.from..=cabin.to {
927 let mut line = format!("{row:2} ");
928 for (b, block) in cabin.blocks.iter().enumerate() {
929 if b > 0 {
930 line.push(' ');
931 }
932 for letter in block.bytes() {
933 let target =
934 plane.lookup[row * 26 + (letter - b'A') as usize].unwrap();
935 line.push(if self.flights[f].blocked[target] {
936 '#'
937 } else if self.flights[f].occupants[target].is_some() {
938 'X'
939 } else if plane.exits[row] {
940 'E'
941 } else {
942 '.'
943 });
944 }
945 }
946 out.push(line);
947 }
948 }
949 }
950 "MANIFEST" => {
951 let mut order: Vec<usize> = self.flights[f]
952 .occupants
953 .iter()
954 .flatten()
955 .copied()
956 .collect();
957 let seated = order.len();
958 let (mut unassigned, mut infants) = (0, 0);
959 for &id in &self.flights[f].passengers {
960 let p = &self.people[id];
961 if !p.active {
962 continue;
963 }
964 if p.kind == 2 {
965 infants += 1;
966 } else if p.seat.is_none() {
967 unassigned += 1;
968 order.push(id);
969 }
970 }
971 out.push(format!(
972 "MANIFEST {name} seated {seated} unassigned {unassigned} infants {infants}"
973 ));
974 for id in order {
975 let p = &self.people[id];
976 let b = &self.bookings[p.booking];
977 let position = if p.seat.is_some() {
978 self.aircraft[model].seat_name(p.seat)
979 } else {
980 "---".to_owned()
981 };
982 let mut line = format!(
983 "{position} {} {} {} {}",
984 CLASSES[p.cabin], b.pnr, p.name, TYPES[p.kind]
985 );
986 if self.has_infant(id) {
987 line.push_str(&format!(" +{}", self.people[p.lap.unwrap()].name));
988 }
989 if self.tier(id) < 3 {
990 line.push_str(&format!(" {}", TIERS[self.tier(id)]));
991 }
992 if b.checked {
993 line.push_str(" CHECKED");
994 }
995 if p.volunteer {
996 line.push_str(" VOL");
997 }
998 out.push(line);
999 }
1000 }
1001 "LOADFACTOR" => {
1002 out.push(format!("LOADFACTOR {name}"));
1003 let (mut all_seated, mut all_seats) = (0, 0);
1004 for c in 0..3 {
1005 if self.aircraft[model].cabins[c].from == 0 {
1006 continue;
1007 }
1008 let (mut seats, mut seated) = (0, 0);
1009 for (i, s) in self.aircraft[model].seats.iter().enumerate() {
1010 if s.cabin != c {
1011 continue;
1012 }
1013 seats += usize::from(!self.flights[f].blocked[i]);
1014 seated += usize::from(self.flights[f].occupants[i].is_some());
1015 }
1016 let factor = seated * 1000 / seats;
1017 out.push(format!(
1018 "{} {seated}/{seats} {}.{}% booked {}/{}",
1019 CLASSES[c],
1020 factor / 10,
1021 factor % 10,
1022 self.booked_count(f, c),
1023 self.flights[f].limit[c]
1024 ));
1025 all_seated += seated;
1026 all_seats += seats;
1027 }
1028 let factor = all_seated * 1000 / all_seats;
1029 out.push(format!(
1030 "ALL {all_seated}/{all_seats} {}.{}%",
1031 factor / 10,
1032 factor % 10
1033 ));
1034 }
1035 "BALANCE" => {
1036 let (mut weight, mut moment) = (0i32, 0i32);
1037 for &id in self.flights[f].occupants.iter().flatten() {
1038 let p = &self.people[id];
1039 let value = (if p.kind == 0 { 80 } else { 40 })
1040 + (if self.has_infant(id) { 10 } else { 0 });
1041 weight += value;
1042 moment += value
1043 * (self.aircraft[model].seats[p.seat.unwrap()].row as i32
1044 - self.aircraft[model].base);
1045 }
1046 let limit = self.aircraft[model].limit;
1047 let verdict = if moment < -limit {
1048 "NOSE-HEAVY"
1049 } else if moment > limit {
1050 "TAIL-HEAVY"
1051 } else {
1052 "OK"
1053 };
1054 out.push(format!(
1055 "BALANCE {name} weight {weight} moment {moment} limit {limit} {verdict}"
1056 ));
1057 }
1058 "BOARDING" => {
1059 struct Item {
1060 id: usize,
1061 group: usize,
1062 zone: usize,
1063 kind: usize,
1064 row: usize,
1065 letter: u8,
1066 }
1067 let mut order = Vec::new();
1068 for &id in self.flights[f].occupants.iter().flatten() {
1069 let p = &self.people[id];
1070 let s = &self.aircraft[model].seats[p.seat.unwrap()];
1071 let group = if self.booking_infants(p.booking) {
1072 0
1073 } else if p.cabin < 2 {
1074 1
1075 } else if self.tier(id) < 2 {
1076 2
1077 } else {
1078 3
1079 };
1080 let zone = if group == 3 {
1081 (s.row - self.aircraft[model].cabins[2].from) / 5 + 1
1082 } else {
1083 0
1084 };
1085 order.push(Item {
1086 id,
1087 group,
1088 zone,
1089 kind: s.kind,
1090 row: s.row,
1091 letter: s.letter,
1092 });
1093 }
1094 order.sort_by(|x, y| {
1095 x.group
1096 .cmp(&y.group)
1097 .then_with(|| {
1098 if x.group == 3 {
1099 y.zone
1100 .cmp(&x.zone)
1101 .then(x.kind.cmp(&y.kind))
1102 .then(y.row.cmp(&x.row))
1103 } else {
1104 x.row.cmp(&y.row)
1105 }
1106 })
1107 .then(x.letter.cmp(&y.letter))
1108 });
1109 out.push(format!("BOARDING {name} {}", order.len()));
1110 for (i, item) in order.iter().enumerate() {
1111 let p = &self.people[item.id];
1112 let group = if item.group == 3 {
1113 format!("ZONE{}", item.zone)
1114 } else {
1115 ["PRE", "CABIN", "TIER"][item.group].to_owned()
1116 };
1117 out.push(format!(
1118 "{} {group} {} {} {}",
1119 i + 1,
1120 self.aircraft[model].seat_name(p.seat),
1121 self.bookings[p.booking].pnr,
1122 p.name
1123 ));
1124 }
1125 }
1126 "AVAIL" => {
1127 let c = cabin_id(v[2]);
1128 if self.aircraft[model].cabins[c].from == 0 {
1129 return Err(format!("no cabin {} on {name}", v[2]));
1130 }
1131 let (mut free, mut exits) = (0, 0);
1132 let mut kinds = [0; 3];
1133 for (i, s) in self.aircraft[model].seats.iter().enumerate() {
1134 if s.cabin != c
1135 || self.flights[f].blocked[i]
1136 || self.flights[f].occupants[i].is_some()
1137 {
1138 continue;
1139 }
1140 free += 1;
1141 kinds[s.kind] += 1;
1142 exits += usize::from(self.aircraft[model].exits[s.row]);
1143 }
1144 out.push(format!(
1145 "AVAIL {name} {} free {free} W {} M {} A {} EXIT {exits}",
1146 CLASSES[c], kinds[0], kinds[1], kinds[2]
1147 ));
1148 }
1149 _ => unreachable!(),
1150 }
1151 Ok(out)
1152 }
1153}
