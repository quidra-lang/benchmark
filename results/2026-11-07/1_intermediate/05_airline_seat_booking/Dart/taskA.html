29 void addCabin(int cabin, int first, int last, List<String> blocks) {
30 cabins[cabin] = Cabin(first, last, blocks);
31 for (var row = first; row <= last; row++) {
32 final groups = <List<int>>[];
33 for (var b = 0; b < blocks.length; b++) {
34 final group = <int>[];
35 for (var x = 0; x < blocks[b].length; x++) {
36 final kind = b == 0 && x == 0 || b == blocks.length - 1 && x == blocks[b].length - 1 ? 'W' :
37 x == 0 || x == blocks[b].length - 1 ? 'A' : 'M';
38 final seat = Seat(row, blocks[b][x], cabin, b, kind);
39 byCode[seat.code] = seats.length;
40 group.add(seats.length);
41 seats.add(seat);
42 }
43 groups.add(group);
44 }
45 rows[row] = groups;
46 }

70class Flight {
71 final String name;
72 final Aircraft aircraft;
73 final List<int> limits;
74 final Set<int> blocked;
75 final List<Passenger?> occupied;
76 final bookings = <Booking>[], volunteers = <Passenger>[];

78 Flight(this.name, this.aircraft, int overbook) :
79 limits = [for (var c = 0; c < 3; c++) aircraft.capacity(c) * (100 + overbook) ~/ 100],
80 blocked = Set.of(aircraft.blocked), occupied = List.filled(aircraft.seats.length, null);
81 List<Passenger> get passengers => bookings.expand((b) => b.passengers).toList();
82 int booked(int cabin) => passengers.where((p) => p.needsSeat && p.cabin == cabin).length;
83 void place(Passenger p, int seat) {
84 if (p.seat >= 0) occupied[p.seat] = null;
85 p.seat = seat;
86 if (seat >= 0) occupied[seat] = p;
87 }
88 bool fits(List<Passenger> ps, List<int> ss, int cabin, [Set<Passenger> vacated = const {}]) {
89 final moving = ps.toSet(), targets = <int>{};
90 for (var i = 0; i < ps.length; i++) {
91 final p = ps[i], index = ss[i], s = aircraft.seats[index];
92 final occupant = occupied[index];
93 if (!targets.add(index) || s.cabin != cabin || blocked.contains(index) ||
94 occupant != null && !moving.contains(occupant) && !vacated.contains(occupant)) return false;
95 if ((p.type == 'CHILD' || p.carrying) && aircraft.exits.contains(s.row)) return false;
96 if (p.carrying) {
97 for (final otherSeat in aircraft.rows[s.row]![s.block]) {
98 final other = occupied[otherSeat];
99 if (other != null && other != p && !moving.contains(other) && !vacated.contains(other) && other.carrying) return false;
100 }
101 for (var j = 0; j < ps.length; j++) {
102 final other = aircraft.seats[ss[j]];
103 if (j != i && ps[j].carrying && s.row == other.row && s.block == other.block) return false;
104 }
105 }
106 }
107 return true;
108 }
109 int choose(Passenger p, int cabin) {
110 var first = -1;
111 for (var s = 0; s < aircraft.seats.length; s++) {
112 if (!fits([p], [s], cabin)) continue;
113 if (first < 0) first = s;
114 if (p.preference == aircraft.seats[s].kind) return s;
115 }
116 return first;
117 }
118 String assign(Booking b, int cabin) {
119 final ps = b.passengers.where((p) => p.needsSeat).toList();
120 final k = ps.length;
121 if (k == 1) {
122 place(ps[0], choose(ps[0], cabin));
123 return 'SINGLE';
124 }
125 final layout = aircraft.cabins[cabin]!;
126 bool adopt(List<int> ss) {
127 if (!fits(ps, ss, cabin)) return false;
128 for (var i = 0; i < k; i++) place(ps[i], ss[i]);
129 return true;
130 }
131 for (var row = layout.first; row <= layout.last; row++) {
132 for (final block in aircraft.rows[row]!) {
133 for (var x = 0; x + k <= block.length; x++) {
134 if (adopt(block.sublist(x, x + k))) return 'BLOCK';
135 }
136 }
137 }
138 for (var row = layout.first; row <= layout.last; row++) {
139 final line = aircraft.rows[row]!.expand((b) => b).toList();
140 for (var x = 0; x + k <= line.length; x++) {
141 if (adopt(line.sublist(x, x + k))) return 'ROW';
142 }
143 }
144 for (var h = 2; h <= k; h++) {
145 final width = (k + h - 1) ~/ h, height = (k + width - 1) ~/ width;
146 for (var row = layout.first; row + height - 1 <= layout.last; row++) {
147 for (var b = 0; b < layout.blocks.length; b++) {
148 for (var x = 0; x + width <= layout.blocks[b].length; x++) {
149 final ss = [for (var j = 0; j < k; j++) aircraft.rows[row + j ~/ width]![b][x + j % width]];
150 if (adopt(ss)) return 'STACK';
151 }
152 }
153 }
154 }
155 for (final p in ps) place(p, choose(p, cabin));
156 return 'SPLIT';
157 }
158 String result(Passenger p) => !p.needsSeat ? 'LAP' : p.seat < 0 ? 'UNASSIGNED' : aircraft.seats[p.seat].code;
159 void reassign(StringBuffer out) {
160 for (final p in passengers) {
161 if (!p.needsSeat || p.seat >= 0) continue;
162 final seat = choose(p, p.cabin);
163 if (seat >= 0) {
164 place(p, seat);
165 out.writeln('REASSIGNED ${p.booking.pnr} ${p.name} ${result(p)}');
166 }
167 }
168 }
169 void remove(Passenger p) {
170 if (!p.active) return;
171 if (p.carrying) remove(p.infant!);
172 if (p.guardian != null) {
173 p.guardian!.infant = null;
174 p.guardian = null;
175 }
176 place(p, -1);
177 p.active = false;
178 volunteers.remove(p);
179 }

448 final x = a.key, y = b.key, xs = [x.$1, x.$2, x.$3, x.$4], ys = [y.$1, y.$2, y.$3, y.$4];
449 for (var i = 0; i < 4; i++)
450 if (xs[i] != ys[i]) return xs[i].compareTo(ys[i]);
451 return x.$5.compareTo(y.$5);
452 });
453 out.writeln('BOARDING ${f.name} ${ordered.length}');
454 for (var i = 0; i < ordered.length; i++) {
455 final p = ordered[i].passenger, k = ordered[i].key;
456 final group = k.$1 == 3 ? 'ZONE${-k.$2}' : ['PRE', 'CABIN', 'TIER'][k.$1];
457 out.writeln('${i + 1} $group ${f.result(p)} ${p.booking.pnr} ${p.name}');
458 }
459 }
460 }
461 return out.toString();
462 }
