27class Cabin:
28 def __init__(self, name, first, last, blocks):
29 self.name = name
30 self.first = first
31 self.last = last
32 self.blocks = blocks
33 self.rows = {}
34 self.seats = []
35 for row in range(first, last + 1):
36 row_blocks = []
37 for block_number, letters in enumerate(blocks):
38 seats = []
39 for position, letter in enumerate(letters):
40 at_left_window = block_number == 0 and position == 0
41 at_right_window = (
42 block_number == len(blocks) - 1
43 and position == len(letters) - 1
44 )
45 if at_left_window or at_right_window:
46 kind = "W"
47 elif position == 0 or position == len(letters) - 1:
48 kind = "A"
49 else:
50 kind = "M"
51 seat = Seat(name, row, block_number, position, letter, kind)
52 seats.append(seat)
53 self.seats.append(seat)
54 row_blocks.append(seats)
55 self.rows[row] = row_blocks
56
57 def candidates(self, count):
58 for blocks in self.rows.values():
59 for block in blocks:
60 for start in range(len(block) - count + 1):
61 yield "BLOCK", block[start:start + count]
62 for blocks in self.rows.values():
63 row_seats = [seat for block in blocks for seat in block]
64 for start in range(len(row_seats) - count + 1):
65 yield "ROW", row_seats[start:start + count]
66 for height in range(2, count + 1):
67 width = (count + height - 1) // height
68 used_rows = (count + width - 1) // width
69 for first_row, blocks in self.rows.items():
70 if first_row + used_rows - 1 > self.last:
71 continue
72 for block_number, block in enumerate(blocks):
73 for start in range(len(block) - width + 1):
74 yield "STACK", [
75 self.rows[first_row + i // width][block_number][start + i % width]
76 for i in range(count)
77 ]

186 def unassigned(self):

189 def infant_conflict(self, passenger, seat, ignored=(), proposed=()):
190 if passenger.infant is None:
191 return False
192 cabin = self.aircraft.cabins[seat.cabin]
193 for neighbor in cabin.rows[seat.row][seat.block]:
194 other = self.occupants.get(neighbor.code)
195 if other is None:
196 continue
197 if other is passenger or other in ignored or other.infant is None:
198 continue
199 return True
200 for other, destination in proposed:
201 if other is passenger or other.infant is None:
202 continue
203 if destination.row == seat.row and destination.block == seat.block:
204 return True
205 return False

207 def can_sit(self, passenger, seat, cabin=None, ignored=(), proposed=()):
208 if seat.cabin != (passenger.cabin if cabin is None else cabin):
209 return False
210 if seat.code in self.blocked:
211 return False
212 occupant = self.occupants.get(seat.code)
213 if occupant is not None and occupant not in ignored:
214 return False
215 if passenger.kind == "CHILD" or passenger.infant is not None:
216 if seat.row in self.aircraft.exit_rows:
217 return False
218 return not self.infant_conflict(passenger, seat, ignored, proposed)

220 def choose_seat(self, passenger, cabin=None):
221 name = passenger.cabin if cabin is None else cabin
222 candidates = [
223 seat for seat in self.aircraft.cabins[name].seats
224 if self.can_sit(passenger, seat, name)
225 ]
226 return next(
227 (seat for seat in candidates if seat.kind == passenger.preference),
228 candidates[0] if candidates else None,
229 )
230
231 def vacate(self, passenger):
232 if passenger.seat is not None:
233 del self.occupants[passenger.seat.code]
234 passenger.seat = None
235
236 def place(self, passenger, seat):
237 self.vacate(passenger)
238 passenger.seat = seat
239 if seat is not None:
240 self.occupants[seat.code] = passenger
241
242 def assign_booking(self, booking, cabin):
243 people = [p for p in booking.passengers if p.needs_seat()]
244 if len(people) == 1:
245 self.place(people[0], self.choose_seat(people[0]))
246 return "SINGLE"
247 for method, seats in self.aircraft.cabins[cabin].candidates(len(people)):
248 proposed = list(zip(people, seats))
249 if all(self.can_sit(p, s, proposed=proposed) for p, s in proposed):
250 for passenger, seat in proposed:
251 self.place(passenger, seat)
252 return method
253 for passenger in people:
254 self.place(passenger, self.choose_seat(passenger))
255 return "SPLIT"

257 def reassign(self, emit):
258 for passenger in self.unassigned():
259 seat = self.choose_seat(passenger)
260 if seat is not None:
261 self.place(passenger, seat)
262 emit("REASSIGNED", passenger.booking.pnr, passenger.name, seat.code)

264 def removal_targets(self, booking, passenger=None):
265 if passenger is None:
266 return list(booking.passengers)
267 targets = {passenger}
268 if passenger.infant is not None:
269 targets.add(passenger.infant)
270 return [p for p in booking.passengers if p in targets]

272 def remove(self, booking, targets):
273 removed = set(targets)
274 for passenger in targets:
275 self.vacate(passenger)
276 if passenger.adult is not None:
277 passenger.adult.infant = None
278 passenger.adult = None
279 if passenger.infant is not None:
280 passenger.infant.adult = None
281 passenger.infant = None
282 booking.passengers = [p for p in booking.passengers if p not in removed]
283 self.volunteers = [p for p in self.volunteers if p not in removed]

341 def book(self, arguments):
342 pnr, flight_name, cabin, *descriptors = arguments
343 if pnr in self.bookings:
344 raise CommandError(f"duplicate PNR {pnr}")
345 flight = self.flight(flight_name, True)
346 flight.require_cabin(cabin)
347 names = [descriptor.split(":")[0] for descriptor in descriptors]
348 seen = set()
349 for name in names:
350 if name in seen:
351 raise CommandError(f"duplicate name {name} in {pnr}")
352 seen.add(name)
353 existing_names = {p.name for p in flight.passengers()}
354 for name in names:
355 if name in existing_names:
356 raise CommandError(f"{name} already on {flight_name}")
357 kinds = [descriptor.split(":")[1] for descriptor in descriptors]
358 adults = kinds.count("ADULT")
359 infants = kinds.count("INFANT")
360 if adults == 0:
361 raise CommandError(f"no adult in {pnr}")
362 if infants > adults:
363 raise CommandError(f"too many infants in {pnr}")
364 if flight.booked_count(cabin) + len(kinds) - infants > flight.limits[cabin]:
365 raise CommandError(f"limit exceeded for {flight_name} {cabin}")
366 booking = Booking(pnr, flight, cabin, descriptors)
367 self.bookings[pnr] = booking
368 flight.bookings.append(booking)
369 method = flight.assign_booking(booking, cabin)
370 results = [p.name + ":" + p.result() for p in booking.passengers]
371 self.emit("BOOKED", pnr, flight_name, cabin, method, *results)

373 def change_seat(self, arguments):
374 pnr, name, code = arguments
375 booking = self.booking(pnr, True)
376 passenger = booking.passenger(name)
377 flight = booking.flight
378 if not passenger.needs_seat():
379 raise CommandError(f"infant {name} has no seat")
380 seat = flight.aircraft.seats.get(code)
381 if seat is None or seat.cabin != passenger.cabin:
382 raise CommandError(f"no seat {code} in cabin {passenger.cabin}")
383 if code in flight.blocked:
384 raise CommandError(f"seat {code} is blocked")
385 if code in flight.occupants:
386 raise CommandError(f"seat {code} is occupied")
387 if passenger.kind == "CHILD" or passenger.infant is not None:
388 if seat.row in flight.aircraft.exit_rows:
389 raise CommandError(f"exit row {seat.row} not allowed for {name}")
390 if flight.infant_conflict(passenger, seat):
391 raise CommandError(f"infant limit reached at {code}")
392 old = passenger.seat.code
393 flight.place(passenger, seat)
394 self.emit("MOVED", pnr, name, old, "->", code)
395 flight.reassign(self.emit)

397 def swap(self, arguments):
398 pnr1, name1, pnr2, name2 = arguments
399 booking1 = self.booking(pnr1, True)
400 first = booking1.passenger(name1)
401 booking2 = self.booking(pnr2, True)
402 second = booking2.passenger(name2)
403 if first is second:
404 raise CommandError(f"cannot swap {name1} with itself")
405 for passenger in (first, second):
406 if not passenger.needs_seat():
407 raise CommandError(f"infant {passenger.name} has no seat")
408 if booking1.flight is not booking2.flight:
409 raise CommandError(f"{pnr1} and {pnr2} are on different flights")
410 if first.cabin != second.cabin:
411 raise CommandError(f"{name1} and {name2} are in different cabins")
412 for passenger in (first, second):
413 if passenger.seat is None:
414 raise CommandError(f"{passenger.name} has no seat")
415 flight = booking1.flight
416 first_seat, second_seat = first.seat, second.seat
417 proposed = ((first, second_seat), (second, first_seat))
418 ignored = (first, second)
419 if not all(
420 flight.can_sit(p, s, ignored=ignored, proposed=proposed)
421 for p, s in proposed
422 ):
423 raise CommandError(f"swap of {name1} and {name2} breaks seating rules")
424 flight.vacate(first)
425 flight.vacate(second)
426 flight.place(first, second_seat)
427 flight.place(second, first_seat)
428 self.emit("SWAPPED", pnr1, name1, second_seat.code, pnr2, name2, first_seat.code)
429 flight.reassign(self.emit)

473 def upgrade(self, arguments):
474 flight = self.flight(arguments[0], True)
475 cabins = flight.aircraft.class_order()
476 count = 0
477 for higher, lower in zip(cabins, cabins[1:]):
478 targets = [
479 p for p in flight.passengers()
480 if p.cabin == lower and p.kind == "ADULT"
481 and p.infant is None and self.tier(p) != "NONE"
482 ]
483 targets.sort(key=lambda p: TIER_ORDER[self.tier(p)])
484 for passenger in targets:
485 if flight.booked_count(higher) >= flight.limits[higher]:
486 break
487 seat = flight.choose_seat(passenger, higher)
488 if seat is None:
489 break
490 old = passenger.result()
491 passenger.cabin = higher
492 flight.place(passenger, seat)
493 self.emit(
494 "UPGRADED", passenger.booking.pnr, passenger.name,
495 lower, old, "->", higher, seat.code,
496 )
497 count += 1
498 flight.reassign(self.emit)
499 self.emit("UPGRADE", flight.name, count)

502 compensation = {"F": 500, "J": 300, "Y": 150}[passenger.cabin]

507 mode = "VOLUNTARY" if voluntary else "INVOLUNTARY"
508 self.emit(
509 "DENIED", mode, passenger.booking.pnr, passenger.name,

516 def close(self, arguments):
517 flight = self.flight(arguments[0], True)
518 for booking in flight.bookings:
519 if booking.passengers and not booking.checked:
520 self.emit("NOSHOW", booking.pnr, len(booking.passengers))
521 flight.remove(booking, flight.removal_targets(booking))
522 flight.reassign(self.emit)
523 denied = 0
524 total = 0
525 for cabin in flight.aircraft.class_order():
526 volunteers = [
527 p for p in flight.volunteers
528 if p.seat is not None and p.cabin == cabin
529 ]
530 for passenger in volunteers:
531 seat = passenger.seat
532 can_replace = any(
533 p.cabin == cabin
534 and flight.can_sit(p, seat, ignored=(passenger,))
535 for p in flight.unassigned()
536 )
537 if can_replace:
538 total += self.deny(flight, passenger, True)
539 denied += 1
540 flight.reassign(self.emit)
541 for passenger in flight.unassigned():
542 total += self.deny(flight, passenger, False)
543 denied += 1
544 flight.closed = True
545 self.emit("CLOSED", flight.name, "denied", denied, "compensation", total)

667 def balance(self, arguments):
668 flight = self.flight(arguments[0])
669 aircraft = flight.aircraft
670 weight = 0
671 moment = 0
672 for passenger in flight.passengers():
673 if passenger.seat is None:
674 continue
675 current = 80 if passenger.kind == "ADULT" else 40
676 if passenger.infant is not None:
677 current += 10
678 weight += current
679 moment += current * (passenger.seat.row - aircraft.base_row)
680 if moment < -aircraft.limit:
681 verdict = "NOSE-HEAVY"
682 elif moment > aircraft.limit:
683 verdict = "TAIL-HEAVY"
684 else:
685 verdict = "OK"
686 self.emit(
687 "BALANCE", flight.name, "weight", weight, "moment", moment,
688 "limit", aircraft.limit, verdict,
689 )
690
691 def boarding_group(self, flight, passenger):
692 seat = passenger.seat
693 if passenger.booking.has_infant():
694 return "PRE", (0, seat.row, seat.letter)
695 if passenger.cabin in ("F", "J"):
696 return "CABIN", (1, seat.row, seat.letter)
697 if self.tier(passenger) in ("PLATINUM", "GOLD"):
698 return "TIER", (2, seat.row, seat.letter)
699 start = flight.aircraft.cabins["Y"].first
700 zone = (seat.row - start) // 5 + 1
701 order = (3, -zone, KIND_ORDER[seat.kind], -seat.row, seat.letter)
702 return "ZONE" + str(zone), order

704 def boarding(self, arguments):
705 flight = self.flight(arguments[0])
706 entries = []
707 for passenger in flight.passengers():
708 if passenger.seat is not None:
709 group, order = self.boarding_group(flight, passenger)
710 entries.append((order, group, passenger))
711 entries.sort(key=lambda item: item[0])
712 self.emit("BOARDING", flight.name, len(entries))
713 for number, (_, group, passenger) in enumerate(entries, 1):
714 self.emit(
715 number, group, passenger.seat.code,
716 passenger.booking.pnr, passenger.name,
717 )
718
719 def avail(self, arguments):
720 flight = self.flight(arguments[0])
721 name = arguments[1]
722 cabin = flight.require_cabin(name)
723 free = [
724 seat for seat in cabin.seats
725 if seat.code not in flight.blocked and seat.code not in flight.occupants
726 ]
727 counts = {kind: sum(s.kind == kind for s in free) for kind in KIND_ORDER}
728 exits = sum(s.row in flight.aircraft.exit_rows for s in free)
729 self.emit(
730 "AVAIL", flight.name, name, "free", len(free),
731 "W", counts["W"], "M", counts["M"], "A", counts["A"], "EXIT", exits,
732 )

734 def info(self, arguments):
735 booking = self.booking(arguments[0])
736 self.emit(
737 "INFO", booking.pnr, booking.flight.name,
738 len(booking.passengers), booking.state(),
739 )
740 for passenger in booking.passengers:
741 self.emit(passenger.name, passenger.kind, passenger.cabin, passenger.result())

743 def find(self, arguments):
744 name = arguments[0]
745 matches = []
746 for flight in self.flights.values():
747 for passenger in flight.passengers():
748 if passenger.name == name:
749 state = "CLOSED" if flight.closed else "OPEN"
750 matches.append((
751 flight.name, passenger.booking.pnr, passenger.cabin,
752 passenger.result(), state,
753 ))
754 break
755 self.emit("FIND", name, self.tiers.get(name, "NONE"), len(matches))
756 for match in matches:
757 self.emit(*match)

759 def execute(self, line):
760 command, *arguments = line.split()
761 handlers = {
762 "AIRCRAFT": self.define_aircraft,
763 "CABIN": self.define_cabin,
764 "EXIT": self.define_exits,
765 "BLOCKED": self.define_blocked,
766 "END": self.end_aircraft,
767 "FLIGHT": self.create_flight,
768 "BOOK": self.book,
769 "SEAT": self.change_seat,
770 "SWAP": self.swap,
771 "CANCEL": self.cancel,
772 "STATUS": self.status,
773 "CHECKIN": self.checkin,
774 "VOLUNTEER": self.volunteer,
775 "UPGRADE": self.upgrade,
776 "CLOSE": self.close,
777 "MAP": self.show_map,
778 "MANIFEST": self.manifest,
779 "LOADFACTOR": self.loadfactor,
780 "BALANCE": self.balance,
781 "BOARDING": self.boarding,
782 "AVAIL": self.avail,
783 "INFO": self.info,
784 "FIND": self.find,
785 }
786 try:
787 if command in ("BLOCK", "UNBLOCK"):
788 self.change_block(arguments, command == "BLOCK")
789 else:
790 handlers[command](arguments)
791 except CommandError as error:
792 self.emit("ERROR", error)
