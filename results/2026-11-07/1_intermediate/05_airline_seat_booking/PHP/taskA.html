81function allowed(object $f, object $p, string $seat, string $class, array $vacated = [], array $assignments = []): bool
82{
83 global $people;
84 $s = $f->model->seats[$seat] ?? null;
85 if ($s === null || $s->class !== $class || isset($f->blocked[$seat])) {
86 return false;
87 }
88 if (isset($f->occupied[$seat]) && !isset($vacated[$f->occupied[$seat]])) {
89 return false;
90 }
91 if (($p->type === 'CHILD' || $p->infant !== 0) && isset($f->model->exits[$s->row])) {
92 return false;
93 }
94 if ($p->infant !== 0) {
95 foreach ($f->occupied as $otherSeat => $id) {
96 if ($id === $p->id || isset($vacated[$id]) || $people[$id]->infant === 0) {
97 continue;
98 }
99 $t = $f->model->seats[$otherSeat];
100 if ($t->row === $s->row && $t->block === $s->block) {
101 return false;
102 }
103 }
104 foreach ($assignments as $id => $otherSeat) {
105 if ($id !== $p->id && $people[$id]->infant !== 0) {
106 $t = $f->model->seats[$otherSeat];
107 if ($t->row === $s->row && $t->block === $s->block) {
108 return false;
109 }
110 }
111 }
112 }
113 return true;

115function select_seat(object $f, object $p, ?string $class = null): string
116{
117 $class ??= $p->class;
118 $first = '';
119 foreach ($f->model->cabins[$class]->seats as $seat) {
120 if (allowed($f, $p, $seat, $class)) {
121 if ($first === '') {
122 $first = $seat;
123 }
124 if ($p->pref !== '' && $f->model->seats[$seat]->kind === $p->pref) {
125 return $seat;
126 }
127 }
128 }
129 return $first;

136 $p->seat = $seat;

141function reassign(object $f, array &$output): void
142{
143 global $people;
144 foreach (passengers($f) as $id) {
145 $p = $people[$id];
146 if ($p->type !== 'INFANT' && $p->seat === '') {
147 $seat = select_seat($f, $p);
148 if ($seat !== '') {
149 move_to($f, $p, $seat);
150 $output[] = "REASSIGNED $p->pnr $p->name $seat";
151 }
152 }

155function fits(object $f, array $ids, array $seats, string $class): bool
156{
157 global $people;
158 $assignments = array_combine($ids, $seats);
159 foreach ($ids as $index => $id) {
160 if (!allowed($f, $people[$id], $seats[$index], $class, [], $assignments)) {
161 return false;
162 }
163 }
164 return true;

166function group_seats(object $f, array $ids, string $class): array
167{
168 $c = $f->model->cabins[$class];
169 $count = count($ids);
170 foreach (['BLOCK', 'ROW'] as $method) {
171 for ($row = $c->from; $row <= $c->to; ++$row) {
172 $blocks = $method === 'BLOCK' ? $c->blocks : [implode('', $c->blocks)];
173 foreach ($blocks as $block) {
174 for ($x = 0; $x + $count <= strlen($block); ++$x) {
175 $seats = [];
176 for ($j = 0; $j < $count; ++$j) {
177 $seats[] = $row . $block[$x + $j];
178 }
179 if (fits($f, $ids, $seats, $class)) {
180 return [$method, $seats];
181 }
182 }
183 }
184 }
185 }
186 for ($height = 2; $height <= $count; ++$height) {
187 $width = intdiv($count + $height - 1, $height);
188 $rows = intdiv($count + $width - 1, $width);
189 for ($row = $c->from; $row + $rows - 1 <= $c->to; ++$row) {
190 foreach ($c->blocks as $block) {
191 for ($x = 0; $x + $width <= strlen($block); ++$x) {
192 $seats = [];
193 for ($j = 0; $j < $count; ++$j) {
194 $seats[] = ($row + intdiv($j, $width)) . $block[$x + $j % $width];
195 }
196 if (fits($f, $ids, $seats, $class)) {
197 return ['STACK', $seats];
198 }
199 }
200 }
201 }
202 }
203 return ['SPLIT', []];

243 } elseif ($command === 'CABIN') {
244 $class = $t[1];

261 }
262 }

282 $definition = null;
283 } elseif ($command === 'FLIGHT') {

288 fail("unknown aircraft $t[2]");

291 $f = (object) ['id' => $t[1], 'model' => $model, 'blocked' => $model->blocked, 'occupied' => [], 'limits' => [], 'closed' => false, 'bookings' => [], 'people' => [], 'volunteers' => []];

293 foreach (CLASSES as $class) {
294 if (isset($model->cabins[$class])) {
295 $capacity = $model->cabins[$class]->capacity;

297 $parts = [...$parts, $class, "$capacity/" . $f->limits[$class]];
298 }
299 }
300 $flights[$f->id] = $f;
301 $flightOrder[] = $f->id;
302 $output[] = implode(' ', $parts);
303 } elseif ($command === 'STATUS') {
304 $statuses[$t[1]] = $t[2];
305 $output[] = "STATUS $t[1] $t[2]";
306 } elseif ($command === 'BOOK') {
307 [$pnr, $flight, $class] = array_slice($t, 1, 3);
308 if (isset($usedPNR[$pnr])) {
309 fail("duplicate PNR $pnr");
310 }
311 $f = get_flight($flight, true);
312 cabin($f, $class);
313 $parsed = array_map(fn($token) => explode(':', $token), array_slice($t, 4));
314 $names = [];
315 foreach ($parsed as $parts) {
316 if (isset($names[$parts[0]])) {
317 fail("duplicate name $parts[0] in $pnr");
318 }
319 $names[$parts[0]] = true;
320 }
321 $aboard = [];
322 foreach (passengers($f) as $id) {
323 $aboard[$people[$id]->name] = true;
324 }
325 foreach ($parsed as $parts) {
326 if (isset($aboard[$parts[0]])) {
327 fail("$parts[0] already on $flight");
328 }
329 }
330 $types = array_column($parsed, 1);
331 $adults = count(array_filter($types, fn($type) => $type === 'ADULT'));
332 $infants = count(array_filter($types, fn($type) => $type === 'INFANT'));
333 if ($adults === 0) {

335 }
336 if ($infants > $adults) {

338 }

341 }
342 $b = (object) ['pnr' => $pnr, 'flight' => $flight, 'people' => [], 'checked' => false];
343 $adultIds = [];
344 $infantIds = [];
345 $seatIds = [];
346 foreach ($parsed as $parts) {
347 $id = count($people) + 1;
348 $p = (object) ['id' => $id, 'pnr' => $pnr, 'name' => $parts[0], 'type' => $parts[1], 'pref' => $parts[2] ?? '', 'class' => $class, 'seat' => '', 'active' => true, 'infant' => 0, 'adult' => 0, 'volunteer' => false];

350 $b->people[] = $id;
351 $f->people[] = $id;
352 if ($p->type === 'ADULT') {
353 $adultIds[] = $id;
354 }
355 if ($p->type === 'INFANT') {
356 $infantIds[] = $id;
357 } else {

373 foreach ($seatIds as $index => $id) {

376 }

378 $parts = ['BOOKED', $pnr, $flight, $class, $method];
379 foreach ($b->people as $id) {
380 $parts[] = $people[$id]->name . ':' . result($people[$id]);
381 }
382 $output[] = implode(' ', $parts);
383 } elseif ($command === 'FIND') {
384 $lines = [];
385 foreach ($flightOrder as $flight) {
386 $f = $flights[$flight];
387 foreach (passengers($f) as $id) {
388 $p = $people[$id];
389 if ($p->name === $t[1]) {
390 $lines[] = "$flight $p->pnr $p->class " . result($p) . ' ' . ($f->closed ? 'CLOSED' : 'OPEN');
391 }
392 }
393 }
394 $output[] = "FIND $t[1] " . ($statuses[$t[1]] ?? 'NONE') . ' ' . count($lines);
395 $output = [...$output, ...$lines];
396 } elseif (in_array($command, ['SEAT', 'SWAP', 'CANCEL', 'CHECKIN', 'VOLUNTEER', 'INFO'], true)) {
397 $b = get_booking($t[1], $command !== 'INFO');
398 $f = $flights[$b->flight];
399 if ($command === 'INFO') {
400 $output[] = "INFO $b->pnr $b->flight " . count(members($b)) . ' ' . ($b->checked ? 'CHECKED' : 'NOT-CHECKED');
401 foreach (members($b) as $id) {
402 $p = $people[$id];
403 $output[] = "$p->name $p->type $p->class " . result($p);

405 } elseif ($command === 'CHECKIN') {
406 if ($b->checked) {
407 fail("$b->pnr already checked in");
408 }
409 $b->checked = true;
410 $output[] = "CHECKIN $b->pnr " . count(members($b));
411 } elseif ($command === 'CANCEL') {

413 if (isset($t[2])) {

415 $targets = removal_targets($p);
416 $remaining = array_values(array_diff(members($b), $targets));
417 if ($remaining !== [] && !array_any($remaining, fn($id) => $people[$id]->type === 'ADULT')) {
418 fail("$b->pnr would have no adult");
419 }
420 $targets = array_values(array_intersect(members($b), $targets));
421 }
422 $names = array_map(fn($id) => $people[$id]->name, $targets);
423 remove_people($f, $targets);
424 $output[] = "CANCELLED $b->pnr " . implode(' ', $names);
425 reassign($f, $output);
426 } else {
427 $p = get_person($b, $t[2]);
428 if ($command === 'SWAP') {
429 $b2 = get_booking($t[3], true);
430 $q = get_person($b2, $t[4]);
431 if ($p->id === $q->id) {
432 fail("cannot swap $p->name with itself");
433 }
434 foreach ([$p, $q] as $person) {
435 if ($person->type === 'INFANT') {
436 fail("infant $person->name has no seat");
437 }
438 }
439 if ($b->flight !== $b2->flight) {
440 fail("$b->pnr and $b2->pnr are on different flights");
441 }
442 if ($p->class !== $q->class) {
443 fail("$p->name and $q->name are in different cabins");
444 }
445 foreach ([$p, $q] as $person) {
446 if ($person->seat === '') {
447 fail("$person->name has no seat");
448 }
449 }
450 $vacated = [$p->id => true, $q->id => true];
451 $assignments = [$p->id => $q->seat, $q->id => $p->seat];

453 fail("swap of $p->name and $q->name breaks seating rules");

456 move_to($f, $p, '');
457 move_to($f, $q, '');
458 move_to($f, $p, $oldQ);
459 move_to($f, $q, $oldP);
460 $output[] = "SWAPPED $b->pnr $p->name $p->seat $b2->pnr $q->name $q->seat";
461 reassign($f, $output);
462 } elseif ($command === 'VOLUNTEER') {
463 if ($p->type !== 'ADULT' || $p->infant !== 0) {
464 fail("$p->name cannot volunteer");

467 fail("$p->name has no seat");
468 }

470 fail("$p->name already volunteered");

473 $f->volunteers[] = $p->id;
474 $output[] = "VOLUNTEER $p->pnr $p->name $p->seat";

477 if ($p->type === 'INFANT') {
478 fail("infant $p->name has no seat");

481 if ($s === null || $s->class !== $p->class) {

485 fail("seat $seat is blocked");
486 }
487 if (isset($f->occupied[$seat])) {
488 fail("seat $seat is occupied");
489 }

491 fail("exit row $s->row not allowed for $p->name");
492 }

494 fail("infant limit reached at $seat");
495 }
496 $old = $p->seat;

498 $output[] = "MOVED $p->pnr $p->name $old -> $seat";
499 reassign($f, $output);
500 }
501 }

514 break;
515 }
516 $p = $people[$id];

520 }

522 move_to($f, $p, $seat);

525 $output[] = "UPGRADED $p->pnr $p->name $lower $old -> $higher $seat";
526 }
527 }
528 reassign($f, $output);
529 $output[] = "UPGRADE $f->id $count";
530 } elseif ($command === 'CLOSE') {

532 $b = $bookings[$pnr];
533 $ids = members($b);
534 if ($ids !== [] && !$b->checked) {
535 $output[] = "NOSHOW $pnr " . count($ids);
536 remove_people($f, $ids);
537 }
538 }
539 reassign($f, $output);
540 $denied = 0;
541 $total = 0;
542 $rates = ['F' => 500, 'J' => 300, 'Y' => 150];
543 foreach ($classes as $class) {
544 $targets = array_values(array_filter($f->volunteers, fn($id) => $people[$id]->active && $people[$id]->volunteer && $people[$id]->seat !== '' && $people[$id]->class === $class));
545 foreach ($targets as $id) {
546 $p = $people[$id];
547 $useful = array_any(passengers($f), function($waitingId) use ($people, $f, $p, $class, $id) {
548 $waiting = $people[$waitingId];
549 return $waiting->type !== 'INFANT' && $waiting->class === $class && $waiting->seat === ''
550 && allowed($f, $waiting, $p->seat, $class, [$id => true]);
551 });
552 if ($useful) {
553 $amount = $rates[$class];
554 $output[] = "DENIED VOLUNTARY $p->pnr $p->name $class $amount";
555 remove_people($f, [$id]);
556 ++$denied;
557 $total += $amount;
558 reassign($f, $output);
559 }
560 }
561 }
562 foreach (passengers($f) as $id) {
563 $p = $people[$id];
564 if ($p->active && $p->type !== 'INFANT' && $p->seat === '') {
565 $amount = $rates[$p->class] * 2;
566 if ($p->type === 'CHILD') {
567 $amount = intdiv($amount, 2);
568 }
569 $output[] = "DENIED INVOLUNTARY $p->pnr $p->name $p->class $amount";
570 remove_people($f, removal_targets($p));
571 ++$denied;
572 $total += $amount;
573 }
574 }

576 $output[] = "CLOSED $f->id denied $denied compensation $total";
577 } elseif ($command === 'BLOCK' || $command === 'UNBLOCK') {
578 $seat = $t[2];
579 if (!isset($f->model->seats[$seat])) {
580 fail("no seat $seat on $f->id");
581 }
582 $blocked = isset($f->blocked[$seat]);
583 if ($command === 'BLOCK' && $blocked) {
584 fail("seat $seat is blocked");
585 }
586 if ($command === 'UNBLOCK' && !$blocked) {
587 fail("seat $seat is not blocked");
588 }
589 $output[] = "$command $f->id $seat";
590 if ($command === 'BLOCK') {
591 $f->blocked[$seat] = true;
592 if (isset($f->occupied[$seat])) {
593 $p = $people[$f->occupied[$seat]];
594 move_to($f, $p, '');
595 move_to($f, $p, select_seat($f, $p));
596 $output[] = "DISPLACED $p->pnr $p->name $seat -> " . result($p);
597 }
598 } else {
599 unset($f->blocked[$seat]);

601 reassign($f, $output);
602 } elseif ($command === 'MAP') {
603 $output[] = "MAP $f->id";
