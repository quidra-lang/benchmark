50static char *slice(const char *s, size_t length) {
51 char *result = allocate(length + 1);
52 memcpy(result, s, length);
53 result[length] = '\0';
54 return result;
55}
56static char *format(const char *pattern, ...) {
57 va_list args, copy;
58 va_start(args, pattern);
59 va_copy(copy, args);
60 int length = vsnprintf(NULL, 0, pattern, copy);
61 va_end(copy);
62 char *result = allocate((size_t)length + 1);
63 vsnprintf(result, (size_t)length + 1, pattern, args);
64 va_end(args);
65 return result;
66}
67static void appendf(char **s, const char *pattern, ...) {
68 va_list args, copy;
69 va_start(args, pattern);
70 va_copy(copy, args);
71 int length = vsnprintf(NULL, 0, pattern, copy);
72 va_end(copy);
73 size_t old = strlen(*s);
74 *s = realloc(*s, old + (size_t)length + 1);
75 if (!*s) exit(EXIT_FAILURE);
76 vsnprintf(*s + old, (size_t)length + 1, pattern, args);
77 va_end(args);
78}
79static void push_owned(Strings *s, char *item) {
80 if (s->count == s->capacity) {
81 s->capacity = s->capacity ? s->capacity * 2 : 8;
82 s->items = realloc(s->items, (size_t)s->capacity * sizeof *s->items);
83 if (!s->items) exit(EXIT_FAILURE);
84 }
85 s->items[s->count++] = item;
86}
87static void free_strings(Strings *s) {
88 for (int i = 0; i < s->count; ++i) free(s->items[i]);
89 free(s->items);
90 *s = (Strings){0};
91}
92static Strings split(const char *s, char delimiter) {
93 Strings result = {0};
94 const char *start = s;
95 for (const char *p = s;; ++p) {
96 if (!*p || *p == delimiter) {
97 push_owned(&result, slice(start, (size_t)(p - start)));
98 start = p + 1;
99 if (!*p) break;
100 }
101 }
102 return result;
103}
104static const char *keep(char *s) {
105 push_owned(&temporary, s);
106 return s;

121 values->items[values->count++] = value;

151 if (!strcmp(bookings[i].pnr, pnr) && active_count(i)) {

153 return i;

161 int id = bookings[b].people[i];

168static bool has_infant(int id) { return people[id].type == 0 && people[id].lap >= 0 && people[people[id].lap].active; }
169static bool booking_infants(int b) {
170 for (int i = 0; i < bookings[b].count; ++i) {
171 int id = bookings[b].people[i];
172 if (people[id].active && people[id].type == 2) return true;
173 }
174 return false;

176static int booked_count(int f, int c) {
177 int count = 0;
178 for (int i = 0; i < flights[f]->passengers.count; ++i) {
179 const Person *p = &people[flights[f]->passengers.items[i]];
180 if (p->active && p->type != 2 && p->cabin == c) ++count;
181 }
182 return count;

184static bool exit_allowed(int id, const Aircraft *a, int target) { return !a->exits[a->seats[target].row] || (people[id].type == 0 && !has_infant(id)); }
185static bool infant_allowed(int f, int id, int target, const int *moving, int count, const int *assigned, int extra) {
186 if (!has_infant(id)) return true;
187 const Flight *flight = flights[f];
188 const Aircraft *a = aircraft[flight->model];
189 Seat s = a->seats[target];
190 const char *letters = a->cabins[s.cabin].blocks[s.block];
191 for (int i = 0; letters[i]; ++i) {
192 int peer = a->lookup[s.row * 26 + letters[i] - 'A'], owner = flight->occupants[peer];
193 if (owner >= 0 && owner != id && owner != extra && !contains(moving, count, owner) && has_infant(owner)) return false;
194 }
195 for (int i = 0; i < count; ++i) {
196 int other = moving[i];
197 Seat t = a->seats[assigned[i]];
198 if (other != id && has_infant(other) && t.row == s.row && t.block == s.block) return false;
199 }
200 return true;
201}
202static bool fits(int f, const int *ids, const int *assigned, int count, int extra) {
203 const Flight *flight = flights[f];
204 const Aircraft *a = aircraft[flight->model];
205 for (int i = 0; i < count; ++i) {
206 int target = assigned[i], owner = flight->occupants[target];
207 if (flight->blocked[target] || (owner >= 0 && owner != extra && !contains(ids, count, owner)) || !exit_allowed(ids[i], a, target) || !infant_allowed(f, ids[i], target, ids, count, assigned, extra)) return false;
208 }
209 return true;
210}
211static int choose_seat(int f, int id, int c) {
212 const Flight *flight = flights[f];
213 const Aircraft *a = aircraft[flight->model];
214 int first = -1;
215 for (int target = 0; target < a->seat_count; ++target) {
216 Seat s = a->seats[target];
217 if (s.cabin != c || flight->blocked[target] || flight->occupants[target] >= 0 || !exit_allowed(id, a, target) || !infant_allowed(f, id, target, NULL, 0, NULL, -1)) continue;
218 if (first < 0) first = target;
219 if (people[id].preference < 0 || people[id].preference == s.kind) return target;
220 }
221 return first;
222}
223static void seat_person(int f, int id, int target) {
224 if (people[id].seat >= 0) flights[f]->occupants[people[id].seat] = -1;
225 people[id].seat = target;
226 if (target >= 0) flights[f]->occupants[target] = id;
227}
228static void reassign(int f, Strings *out) {
229 for (int i = 0; i < flights[f]->passengers.count; ++i) {
230 int id = flights[f]->passengers.items[i];
231 Person *p = &people[id];
232 if (!p->active || p->type == 2 || p->seat >= 0) continue;
233 int target = choose_seat(f, id, p->cabin);
234 if (target < 0) continue;
235 seat_person(f, id, target);
236 push_owned(out, format("REASSIGNED %s %s %s", bookings[p->booking].pnr, p->name, seat_name(aircraft[flights[f]->model], target)));
237 }
238}
239static void remove_person(int id) {
240 Person *p = &people[id];
241 if (!p->active) return;
242 int lap = p->lap;
243 if (p->type == 0 && lap >= 0) remove_person(lap);
244 if (p->type == 2 && lap >= 0) people[lap].lap = -1;
245 seat_person(bookings[p->booking].flight, id, -1);
246 p->active = p->volunteer = false;
247 p->lap = -1;
248}
249static const char *passenger_result(int id) {
250 const Person *p = &people[id];
251 return p->type == 2 ? "LAP" : seat_name(aircraft[flights[bookings[p->booking].flight]->model], p->seat);
252}
253static const char *assign_booking(int f, int c, const int *ids, int count) {
254 if (count == 1) {
255 seat_person(f, ids[0], choose_seat(f, ids[0], c));
256 return "SINGLE";
257 }
258 const Aircraft *a = aircraft[flights[f]->model];
259 const Cabin *cabin = &a->cabins[c];
260 for (int stage = 0; stage < 3; ++stage) {
261 int first_height = stage == 2 ? 2 : 1, last_height = stage == 2 ? count : 1;
262 for (int height = first_height; height <= last_height; ++height) {
263 int width = stage == 2 ? (count + height - 1) / height : count;
264 int rows = (count + width - 1) / width;
265 for (int row = cabin->from; row + rows - 1 <= cabin->to; ++row) {
266 int blocks = stage == 1 ? 1 : cabin->block_count;
267 for (int b = 0; b < blocks; ++b) {
268 char letters[11] = "";
269 if (stage == 1) {
270 for (int j = 0; j < cabin->block_count; ++j) strcat(letters, cabin->blocks[j]);
271 } else strcpy(letters, cabin->blocks[b]);
272 for (int x = 0; x + width <= (int)strlen(letters); ++x) {
273 int candidate[9];
274 for (int j = 0; j < count; ++j) candidate[j] = a->lookup[(row + j / width) * 26 + letters[x + j % width] - 'A'];
275 if (!fits(f, ids, candidate, count, -1)) continue;
276 for (int j = 0; j < count; ++j) seat_person(f, ids[j], candidate[j]);
277 const char *methods[] = {"BLOCK", "ROW", "STACK"};
278 return methods[stage];
279 }
280 }
281 }
282 }
283 }
284 for (int i = 0; i < count; ++i) seat_person(f, ids[i], choose_seat(f, ids[i], c));
285 return "SPLIT";
286}
287typedef struct { int tier, id; } Target;
288static int compare_targets(const void *left, const void *right) {
289 const Target *a = left, *b = right;
290 if (a->tier != b->tier) return a->tier < b->tier ? -1 : 1;
291 return a->id < b->id ? -1 : a->id > b->id;
292}
293typedef struct { int id, group, zone, kind, row; char letter; } BoardingItem;
294static int compare_boarding(const void *left, const void *right) {
295 const BoardingItem *a = left, *b = right;
296 if (a->group != b->group) return a->group < b->group ? -1 : 1;
297 if (a->group == 3) {
298 if (a->zone != b->zone) return a->zone > b->zone ? -1 : 1;
299 if (a->kind != b->kind) return a->kind < b->kind ? -1 : 1;
300 if (a->row != b->row) return a->row > b->row ? -1 : 1;
301 } else if (a->row != b->row) return a->row < b->row ? -1 : 1;
302 return a->letter < b->letter ? -1 : a->letter > b->letter;
303}
304static void flight_command(int f, const Strings *v, Strings *out) {
305 const char *command = v->items[0];
306 Flight *flight = flights[f];
307 const Aircraft *a = aircraft[flight->model];
308 if (!strcmp(command, "BLOCK") || !strcmp(command, "UNBLOCK")) {
309 int target = a->lookup[seat_key(v->items[2])];
310 if (target < 0) failf("no seat %s on %s", v->items[2], flight->name);
311 bool block = !strcmp(command, "BLOCK");
312 if (block && flight->blocked[target]) failf("seat %s is blocked", v->items[2]);
313 if (!block && !flight->blocked[target]) failf("seat %s is not blocked", v->items[2]);
314 flight->blocked[target] = block;
315 push_owned(out, format("%s %s %s", command, flight->name, v->items[2]));
316 int id = flight->occupants[target];
317 if (block && id >= 0) {
318 seat_person(f, id, -1);
319 seat_person(f, id, choose_seat(f, id, people[id].cabin));
320 push_owned(out, format("DISPLACED %s %s %s -> %s", bookings[people[id].booking].pnr, people[id].name, v->items[2], passenger_result(id)));
321 }
322 reassign(f, out);
323 } else if (!strcmp(command, "UPGRADE")) {
324 int cabins[3], cabin_count = 0, upgraded = 0;
325 for (int c = 0; c < 3; ++c) if (a->cabins[c].from) cabins[cabin_count++] = c;
326 for (int pair = 0; pair + 1 < cabin_count; ++pair) {
327 int higher = cabins[pair], lower = cabins[pair + 1], booked = booked_count(f, higher);
328 Target targets[9000];
329 int count = 0;
330 for (int i = 0; i < flight->passengers.count; ++i) {
331 int id = flight->passengers.items[i];
332 const Person *p = &people[id];
333 if (p->active && p->cabin == lower && p->type == 0 && !has_infant(id) && tier(id) < 3) targets[count++] = (Target){tier(id), id};
334 }
335 if (count > 1) qsort(targets, count, sizeof *targets, compare_targets);
336 for (int i = 0; i < count; ++i) {
337 if (booked >= flight->limit[higher]) break;
338 int id = targets[i].id, target = choose_seat(f, id, higher);
339 if (target < 0) break;
340 const char *old = passenger_result(id);
341 people[id].cabin = higher;
342 seat_person(f, id, target);
343 ++booked;
344 ++upgraded;
345 push_owned(out, format("UPGRADED %s %s %c %s -> %c %s", bookings[people[id].booking].pnr, people[id].name, classes[lower], old, classes[higher], seat_name(a, target)));
346 }
347 }
348 reassign(f, out);
349 push_owned(out, format("UPGRADE %s %d", flight->name, upgraded));
350 } else if (!strcmp(command, "CLOSE")) {
351 for (int b = 0; b < booking_count; ++b) {
352 if (bookings[b].flight != f || bookings[b].checked || !active_count(b)) continue;
353 push_owned(out, format("NOSHOW %s %d", bookings[b].pnr, active_count(b)));
354 for (int i = 0; i < bookings[b].count; ++i) remove_person(bookings[b].people[i]);
355 }
356 reassign(f, out);
357 int denied = 0, compensation = 0;
358 const int voluntary[] = {500, 300, 150}, involuntary[] = {1000, 600, 300};
359 for (int c = 0; c < 3; ++c) {
360 int targets[9000], count = 0;
361 for (int i = 0; i < flight->volunteers.count; ++i) {
362 int id = flight->volunteers.items[i];
363 const Person *p = &people[id];
364 if (p->active && p->volunteer && p->seat >= 0 && p->cabin == c) targets[count++] = id;
365 }
366 for (int i = 0; i < count; ++i) {
367 int id = targets[i], target = people[id].seat;
368 bool useful = false;
369 for (int j = 0; j < flight->passengers.count; ++j) {
370 int other = flight->passengers.items[j];
371 const Person *p = &people[other];
372 if (p->active && p->type != 2 && p->cabin == c && p->seat < 0 && fits(f, &other, &target, 1, id)) {
373 useful = true;
374 break;
375 }
376 }
377 if (!useful) continue;
378 push_owned(out, format("DENIED VOLUNTARY %s %s %c %d", bookings[people[id].booking].pnr, people[id].name, classes[c], voluntary[c]));
379 ++denied;
380 compensation += voluntary[c];
381 remove_person(id);
382 reassign(f, out);
383 }
384 }
385 for (int i = 0; i < flight->passengers.count; ++i) {
386 int id = flight->passengers.items[i];
387 const Person *p = &people[id];
388 if (!p->active || p->type == 2 || p->seat >= 0) continue;
389 int value = involuntary[p->cabin] / (p->type == 1 ? 2 : 1);
390 push_owned(out, format("DENIED INVOLUNTARY %s %s %c %d", bookings[p->booking].pnr, p->name, classes[p->cabin], value));
391 ++denied;
392 compensation += value;
393 remove_person(id);
394 }
395 flight->closed = true;
396 push_owned(out, format("CLOSED %s denied %d compensation %d", flight->name, denied, compensation));
397 } else if (!strcmp(command, "MAP")) {
398 push_owned(out, format("MAP %s", flight->name));
399 for (int c = 0; c < 3; ++c) {
400 const Cabin *cabin = &a->cabins[c];
401 if (!cabin->from) continue;
402 push_owned(out, format("CABIN %c", classes[c]));
403 char *header = format(" ");
404 for (int b = 0; b < cabin->block_count; ++b) appendf(&header, "%s%s", b ? " " : "", cabin->blocks[b]);
405 push_owned(out, header);
406 for (int row = cabin->from; row <= cabin->to; ++row) {
407 char *line = format("%2d ", row);
408 for (int b = 0; b < cabin->block_count; ++b) {
409 if (b) appendf(&line, " ");
410 for (int i = 0; cabin->blocks[b][i]; ++i) {
411 int target = a->lookup[row * 26 + cabin->blocks[b][i] - 'A'];
412 char mark = flight->blocked[target] ? '#' : flight->occupants[target] >= 0 ? 'X' : a->exits[row] ? 'E' : '.';
413 appendf(&line, "%c", mark);
414 }
415 }
416 push_owned(out, line);
417 }
418 }
419 } else if (!strcmp(command, "MANIFEST")) {
420 int order[9000], count = 0, seated, unassigned = 0, infants = 0;
421 for (int i = 0; i < a->seat_count; ++i) if (flight->occupants[i] >= 0) order[count++] = flight->occupants[i];
422 seated = count;
423 for (int i = 0; i < flight->passengers.count; ++i) {
424 int id = flight->passengers.items[i];
425 const Person *p = &people[id];
426 if (!p->active) continue;
427 if (p->type == 2) ++infants;
428 else if (p->seat < 0) {
429 ++unassigned;
430 order[count++] = id;
431 }
432 }
433 push_owned(out, format("MANIFEST %s seated %d unassigned %d infants %d", flight->name, seated, unassigned, infants));
434 for (int i = 0; i < count; ++i) {
435 int id = order[i];
436 const Person *p = &people[id];
437 const Booking *b = &bookings[p->booking];
438 char *line = format("%s %c %s %s %s", p->seat < 0 ? "---" : seat_name(a, p->seat), classes[p->cabin], b->pnr, p->name, type_names[p->type]);
439 if (has_infant(id)) appendf(&line, " +%s", people[p->lap].name);
440 if (tier(id) < 3) appendf(&line, " %s", tier_names[tier(id)]);
441 if (b->checked) appendf(&line, " CHECKED");
442 if (p->volunteer) appendf(&line, " VOL");
443 push_owned(out, line);
444 }
445 } else if (!strcmp(command, "LOADFACTOR")) {
446 push_owned(out, format("LOADFACTOR %s", flight->name));
447 int all_seated = 0, all_seats = 0;
448 for (int c = 0; c < 3; ++c) {
449 if (!a->cabins[c].from) continue;
450 int seats = 0, seated = 0;
451 for (int i = 0; i < a->seat_count; ++i) {
452 if (a->seats[i].cabin != c) continue;
453 seats += !flight->blocked[i];
454 seated += flight->occupants[i] >= 0;
455 }
456 int factor = seated * 1000 / seats;
457 push_owned(out, format("%c %d/%d %d.%d%% booked %d/%d", classes[c], seated, seats, factor / 10, factor % 10, booked_count(f, c), flight->limit[c]));
458 all_seated += seated;
459 all_seats += seats;
460 }
461 int factor = all_seated * 1000 / all_seats;
462 push_owned(out, format("ALL %d/%d %d.%d%%", all_seated, all_seats, factor / 10, factor % 10));
463 } else if (!strcmp(command, "BALANCE")) {
464 int weight = 0, moment = 0;
465 for (int i = 0; i < a->seat_count; ++i) {
466 int id = flight->occupants[i];
467 if (id < 0) continue;
468 int value = (people[id].type == 0 ? 80 : 40) + (has_infant(id) ? 10 : 0);
469 weight += value;
470 moment += value * (a->seats[people[id].seat].row - a->base);
471 }
472 const char *verdict = moment < -a->limit ? "NOSE-HEAVY" : moment > a->limit ? "TAIL-HEAVY" : "OK";
473 push_owned(out, format("BALANCE %s weight %d moment %d limit %d %s", flight->name, weight, moment, a->limit, verdict));
474 } else if (!strcmp(command, "BOARDING")) {
475 BoardingItem order[990];
476 int count = 0;
477 for (int i = 0; i < a->seat_count; ++i) {
478 int id = flight->occupants[i];
479 if (id < 0) continue;
480 const Person *p = &people[id];
481 Seat s = a->seats[p->seat];
482 int group = booking_infants(p->booking) ? 0 : p->cabin < 2 ? 1 : tier(id) < 2 ? 2 : 3;
483 int zone = group == 3 ? (s.row - a->cabins[2].from) / 5 + 1 : 0;
484 order[count++] = (BoardingItem){id, group, zone, s.kind, s.row, s.letter};
485 }
486 if (count > 1) qsort(order, count, sizeof *order, compare_boarding);
487 push_owned(out, format("BOARDING %s %d", flight->name, count));
488 for (int i = 0; i < count; ++i) {
489 const BoardingItem *item = &order[i];
490 const Person *p = &people[item->id];
491 const char *groups[] = {"PRE", "CABIN", "TIER"};
492 const char *group = item->group == 3 ? keep(format("ZONE%d", item->zone)) : groups[item->group];
493 push_owned(out, format("%d %s %s %s %s", i + 1, group, seat_name(a, p->seat), bookings[p->booking].pnr, p->name));
494 }
495 } else if (!strcmp(command, "AVAIL")) {
496 int c = cabin_id(v->items[2]);
497 if (!a->cabins[c].from) failf("no cabin %s on %s", v->items[2], flight->name);
498 int free_count = 0, exits = 0, kinds[3] = {0};
499 for (int i = 0; i < a->seat_count; ++i) {
500 Seat s = a->seats[i];
501 if (s.cabin != c || flight->blocked[i] || flight->occupants[i] >= 0) continue;
502 ++free_count;
503 ++kinds[s.kind];
504 exits += a->exits[s.row];
505 }
506 push_owned(out, format("AVAIL %s %c free %d W %d M %d A %d EXIT %d", flight->name, classes[c], free_count, kinds[0], kinds[1], kinds[2], exits));
507 }
508}
509static void execute(const Strings *v, Strings *out) {
510 const char *command = v->items[0];
511 if (!strcmp(command, "AIRCRAFT")) {
512 building = allocate(sizeof *building);
513 memset(building, 0, sizeof *building);
514 for (int i = 0; i < 2600; ++i) building->lookup[i] = -1;
515 strcpy(building->model, v->items[1]);
516 building->base = atoi(v->items[2]);
517 building->limit = atoi(v->items[3]);
518 } else if (!strcmp(command, "CABIN")) {
519 int c = cabin_id(v->items[1]);
520 Cabin *cabin = &building->cabins[c];
521 cabin->from = atoi(v->items[2]);
522 cabin->to = atoi(v->items[3]);
523 cabin->block_count = v->count - 4;
524 for (int b = 0; b < cabin->block_count; ++b) strcpy(cabin->blocks[b], v->items[b + 4]);
525 for (int row = cabin->from; row <= cabin->to; ++row) {
526 for (int b = 0; b < cabin->block_count; ++b) {
527 const char *letters = cabin->blocks[b];
528 for (int position = 0; letters[position]; ++position) {
529 char letter = letters[position];
530 int kind = (b == 0 && position == 0) || (b + 1 == cabin->block_count && position + 1 == (int)strlen(letters)) ? 0 : position == 0 || position + 1 == (int)strlen(letters) ? 2 : 1;
531 building->lookup[row * 26 + letter - 'A'] = building->seat_count;
532 building->seats[building->seat_count++] = (Seat){row, c, b, kind, letter};
533 }
534 }
535 }
536 } else if (!strcmp(command, "EXIT")) {
537 for (int i = 1; i < v->count; ++i) building->exits[atoi(v->items[i])] = true;
538 } else if (!strcmp(command, "BLOCKED")) {
539 for (int i = 1; i < v->count; ++i) building->blocked[building->lookup[seat_key(v->items[i])]] = true;
540 } else if (!strcmp(command, "END")) {
541 for (int i = 0; i < building->seat_count; ++i) if (!building->blocked[i]) ++building->capacity[building->seats[i].cabin];
542 char *line = format("AIRCRAFT %s", building->model);
543 for (int c = 0; c < 3; ++c) if (building->cabins[c].from) appendf(&line, " %c %d", classes[c], building->capacity[c]);
544 aircraft[aircraft_count++] = building;
545 building = NULL;
546 push_owned(out, line);
547 } else if (!strcmp(command, "FLIGHT")) {
548 for (int i = 0; i < flight_count; ++i) if (!strcmp(flights[i]->name, v->items[1])) failf("duplicate flight %s", v->items[1]);
549 int model = -1;
550 for (int i = 0; i < aircraft_count; ++i) if (!strcmp(aircraft[i]->model, v->items[2])) model = i;
551 if (model < 0) failf("unknown aircraft %s", v->items[2]);
552 Flight *f = allocate(sizeof *f);
553 memset(f, 0, sizeof *f);
554 strcpy(f->name, v->items[1]);
555 f->model = model;
556 const Aircraft *a = aircraft[model];
557 memcpy(f->blocked, a->blocked, sizeof f->blocked);
558 for (int i = 0; i < 990; ++i) f->occupants[i] = -1;
559 char *line = format("FLIGHT %s %s", f->name, a->model);
560 for (int c = 0; c < 3; ++c) {
561 f->limit[c] = a->capacity[c] * (100 + atoi(v->items[3])) / 100;
562 if (a->cabins[c].from) appendf(&line, " %c %d/%d", classes[c], a->capacity[c], f->limit[c]);
563 }
564 flights[flight_count++] = f;
565 push_owned(out, line);
566 } else if (!strcmp(command, "BOOK")) {
567 for (int i = 0; i < booking_count; ++i) if (!strcmp(bookings[i].pnr, v->items[1])) failf("duplicate PNR %s", v->items[1]);
568 int f = find_flight(v->items[2], true), c = cabin_id(v->items[3]);
569 if (!aircraft[flights[f]->model]->cabins[c].from) failf("no cabin %s on %s", v->items[3], v->items[2]);
570 Person draft[9] = {0};
571 int adults[9], infants[9], adult_count = 0, infant_count = 0, count = v->count - 4;
572 for (int i = 0; i < count; ++i) {
573 Strings fields = split(v->items[i + 4], ':');
574 Person *p = &draft[i];
575 strcpy(p->name, fields.items[0]);
576 for (int j = 0; j < 3; ++j) if (!strcmp(type_names[j], fields.items[1])) p->type = j;
577 p->booking = booking_count;
578 p->cabin = c;
579 p->preference = fields.count == 3 ? (fields.items[2][0] == 'W' ? 0 : fields.items[2][0] == 'M' ? 1 : 2) : -1;
580 p->seat = p->lap = -1;
581 p->active = true;
582 if (p->type == 0) adults[adult_count++] = i;
583 if (p->type == 2) infants[infant_count++] = i;
584 free_strings(&fields);
585 }
586 for (int i = 0; i < count; ++i) {
587 for (int j = 0; j < i; ++j) if (!strcmp(draft[i].name, draft[j].name)) failf("duplicate name %s in %s", draft[i].name, v->items[1]);
588 }
589 for (int i = 0; i < count; ++i) {
590 for (int j = 0; j < flights[f]->passengers.count; ++j) {
591 const Person *p = &people[flights[f]->passengers.items[j]];
592 if (p->active && !strcmp(p->name, draft[i].name)) failf("%s already on %s", draft[i].name, v->items[2]);
593 }
594 }
595 if (!adult_count) failf("no adult in %s", v->items[1]);
596 if (infant_count > adult_count) failf("too many infants in %s", v->items[1]);
597 if (booked_count(f, c) + count - infant_count > flights[f]->limit[c]) failf("limit exceeded for %s %s", v->items[2], v->items[3]);
598 for (int i = 0; i < infant_count; ++i) {
599 draft[adults[i]].lap = people_count + infants[i];
600 draft[infants[i]].lap = people_count + adults[i];
601 }
602 Booking *b = &bookings[booking_count++];
603 strcpy(b->pnr, v->items[1]);
604 b->flight = f;
605 int needs[9], need_count = 0;
606 for (int i = 0; i < count; ++i) {
607 int id = people_count++;
608 people[id] = draft[i];
609 b->people[b->count++] = id;
610 add_index(&flights[f]->passengers, id);
611 if (people[id].type != 2) needs[need_count++] = id;
612 }
613 const char *method = assign_booking(f, c, needs, need_count);
614 char *line = format("BOOKED %s %s %s %s", b->pnr, v->items[2], v->items[3], method);
615 for (int i = 0; i < b->count; ++i) appendf(&line, " %s:%s", people[b->people[i]].name, passenger_result(b->people[i]));
616 push_owned(out, line);
617 } else if (!strcmp(command, "STATUS")) {
618 int id = -1, value = 3;
619 for (int i = 0; i < tier_count; ++i) if (!strcmp(tiers[i].name, v->items[1])) id = i;
620 for (int i = 0; i < 4; ++i) if (!strcmp(tier_names[i], v->items[2])) value = i;
621 if (id < 0) {
622 id = tier_count++;
623 strcpy(tiers[id].name, v->items[1]);
624 }
625 tiers[id].value = value;
626 push_owned(out, format("STATUS %s %s", v->items[1], v->items[2]));
627 } else if (!strcmp(command, "FIND")) {
628 int membership = 3;
629 for (int i = 0; i < tier_count; ++i) if (!strcmp(tiers[i].name, v->items[1])) membership = tiers[i].value;
630 Strings found = {0};
631 for (int f = 0; f < flight_count; ++f) {
632 for (int i = 0; i < flights[f]->passengers.count; ++i) {
633 int id = flights[f]->passengers.items[i];
634 const Person *p = &people[id];
635 if (p->active && !strcmp(p->name, v->items[1])) push_owned(&found, format("%s %s %c %s %s", flights[f]->name, bookings[p->booking].pnr, classes[p->cabin], passenger_result(id), flights[f]->closed ? "CLOSED" : "OPEN"));
636 }
637 }
638 push_owned(out, format("FIND %s %s %d", v->items[1], tier_names[membership], found.count));
639 for (int i = 0; i < found.count; ++i) push_owned(out, found.items[i]);
640 free(found.items);
641 } else if (!strcmp(command, "SEAT")) {
642 int id = resolve(v->items[1], v->items[2]);
643 Person *p = &people[id];
644 int f = bookings[p->booking].flight;
645 const Aircraft *a = aircraft[flights[f]->model];
646 if (p->type == 2) failf("infant %s has no seat", p->name);
647 int target = a->lookup[seat_key(v->items[3])];
648 if (target < 0 || a->seats[target].cabin != p->cabin) failf("no seat %s in cabin %c", v->items[3], classes[p->cabin]);
649 if (flights[f]->blocked[target]) failf("seat %s is blocked", v->items[3]);
650 if (flights[f]->occupants[target] >= 0) failf("seat %s is occupied", v->items[3]);
651 if (!exit_allowed(id, a, target)) failf("exit row %d not allowed for %s", a->seats[target].row, p->name);
652 if (!infant_allowed(f, id, target, NULL, 0, NULL, -1)) failf("infant limit reached at %s", v->items[3]);
653 const char *old = seat_name(a, p->seat);
654 seat_person(f, id, target);
655 push_owned(out, format("MOVED %s %s %s -> %s", v->items[1], v->items[2], old, v->items[3]));
656 reassign(f, out);
657 } else if (!strcmp(command, "SWAP")) {
658 int ids[2];
659 ids[0] = resolve(v->items[1], v->items[2]);
660 ids[1] = resolve(v->items[3], v->items[4]);
661 if (ids[0] == ids[1]) failf("cannot swap %s with itself", v->items[2]);
662 for (int i = 0; i < 2; ++i) if (people[ids[i]].type == 2) failf("infant %s has no seat", people[ids[i]].name);
663 int f = bookings[people[ids[0]].booking].flight;
664 if (f != bookings[people[ids[1]].booking].flight) failf("%s and %s are on different flights", v->items[1], v->items[3]);
665 if (people[ids[0]].cabin != people[ids[1]].cabin) failf("%s and %s are in different cabins", v->items[2], v->items[4]);
666 for (int i = 0; i < 2; ++i) if (people[ids[i]].seat < 0) failf("%s has no seat", people[ids[i]].name);
667 int assigned[] = {people[ids[1]].seat, people[ids[0]].seat};
668 if (!fits(f, ids, assigned, 2, -1)) failf("swap of %s and %s breaks seating rules", v->items[2], v->items[4]);
669 flights[f]->occupants[assigned[0]] = flights[f]->occupants[assigned[1]] = -1;
670 people[ids[0]].seat = people[ids[1]].seat = -1;
671 seat_person(f, ids[0], assigned[0]);
672 seat_person(f, ids[1], assigned[1]);
673 const Aircraft *a = aircraft[flights[f]->model];
674 push_owned(out, format("SWAPPED %s %s %s %s %s %s", v->items[1], v->items[2], seat_name(a, assigned[0]), v->items[3], v->items[4], seat_name(a, assigned[1])));
675 reassign(f, out);
676 } else if (!strcmp(command, "CANCEL")) {
677 int b = find_booking(v->items[1], true), target = v->count == 3 ? find_passenger(b, v->items[2]) : -1;
678 int removed[9], count = 0, remaining = 0, adults = 0;
679 for (int i = 0; i < bookings[b].count; ++i) {
680 int id = bookings[b].people[i];
681 if (people[id].active && (target < 0 || id == target || (people[target].type == 0 && people[target].lap == id))) removed[count++] = id;
682 }
683 for (int i = 0; i < bookings[b].count; ++i) {
684 int id = bookings[b].people[i];
685 if (people[id].active && !contains(removed, count, id)) {
686 ++remaining;
687 adults += people[id].type == 0;
688 }
689 }
690 if (target >= 0 && remaining && !adults) failf("%s would have no adult", v->items[1]);
691 char *line = format("CANCELLED %s", v->items[1]);
692 for (int i = 0; i < count; ++i) appendf(&line, " %s", people[removed[i]].name);
693 for (int i = 0; i < count; ++i) remove_person(removed[i]);
694 push_owned(out, line);
695 reassign(bookings[b].flight, out);
696 } else if (!strcmp(command, "CHECKIN") || !strcmp(command, "VOLUNTEER") || !strcmp(command, "INFO")) {
697 int b = find_booking(v->items[1], strcmp(command, "INFO") != 0);
698 Booking *book = &bookings[b];
699 if (!strcmp(command, "CHECKIN")) {
700 if (book->checked) failf("%s already checked in", v->items[1]);
701 book->checked = true;
702 push_owned(out, format("CHECKIN %s %d", book->pnr, active_count(b)));
703 } else if (!strcmp(command, "INFO")) {
704 push_owned(out, format("INFO %s %s %d %s", book->pnr, flights[book->flight]->name, active_count(b), book->checked ? "CHECKED" : "NOT-CHECKED"));
705 for (int i = 0; i < book->count; ++i) {
706 int id = book->people[i];
707 const Person *p = &people[id];
708 if (p->active) push_owned(out, format("%s %s %c %s", p->name, type_names[p->type], classes[p->cabin], passenger_result(id)));
709 }
710 } else {
711 int id = find_passenger(b, v->items[2]);
712 Person *p = &people[id];
713 if (p->type != 0 || has_infant(id)) failf("%s cannot volunteer", p->name);
714 if (p->seat < 0) failf("%s has no seat", p->name);
715 if (p->volunteer) failf("%s already volunteered", p->name);
716 p->volunteer = true;
717 add_index(&flights[book->flight]->volunteers, id);
718 push_owned(out, format("VOLUNTEER %s %s %s", book->pnr, p->name, passenger_result(id)));
719 }
720 } else {
721 bool modifying = !strcmp(command, "UPGRADE") || !strcmp(command, "CLOSE") || !strcmp(command, "BLOCK") || !strcmp(command, "UNBLOCK");
722 int f = find_flight(v->items[1], modifying);
723 flight_command(f, v, out);
724 }
725}
726int main(void) {
727 char line[8192];
728 if (!fgets(line, sizeof line, stdin)) return 0;
729 int count = atoi(line);
730 for (int i = 0; i < count; ++i) {
731 if (!fgets(line, sizeof line, stdin)) break;
732 line[strcspn(line, "\n")] = '\0';
733 Strings values = split(line, ' ');
734 if (setjmp(recovery) == 0) {
735 Strings out = {0};
736 execute(&values, &out);
737 for (int j = 0; j < out.count; ++j) puts(out.items[j]);
738 free_strings(&out);
739 } else printf("ERROR %s\n", error_message);
740 free_strings(&values);
741 free_strings(&temporary);
742 }
743 for (int i = 0; i < aircraft_count; ++i) free(aircraft[i]);
744 for (int i = 0; i < flight_count; ++i) {
745 free(flights[i]->passengers.items);
746 free(flights[i]->volunteers.items);
747 free(flights[i]);
748 }
749 free(building);
750 return 0;
