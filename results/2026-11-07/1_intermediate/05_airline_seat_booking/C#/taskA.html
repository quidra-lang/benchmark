81 static int Booked(Flight f, string cabin) {
82 return Passengers(f).Count(p => p.NeedsSeat() && p.cabin == cabin);

84 static int Seats(Aircraft a, string cabin, HashSet<string> blocked) {

87 static void RequireCabin(Flight f, string cabin) {
88 if (!f.aircraft.cabins.ContainsKey(cabin)) Fail("no cabin "+cabin+" on "+f.name);
89 }
90 static bool SameBlock(Seat a, Seat b) { return a.Row == b.Row && a.Block == b.Block; }
91 static bool ExitAllowed(Passenger p, Seat s) {
92 return !p.booking.flight.aircraft.exits.Contains(s.Row) || !p.type.Equals("CHILD") && p.lap == null;
93 }
94 static bool InfantAllowed(Passenger p, Seat s, HashSet<Passenger> vacated, Dictionary<Passenger,Seat> proposed) {
95 if (p.lap == null) return true;
96 Flight f = p.booking.flight;
97 string letters = f.aircraft.cabins[s.Cabin].blocks[s.Block];
98 foreach (char letter in letters.ToCharArray()) {
99 Passenger other = f.occupied.GetValueOrDefault(s.Row+Convert.ToString(letter));
100 if (other != null && other != p && other.lap != null && !vacated.Contains(other)) return false;
101 }
102 foreach (var entry in proposed) {
103 if (entry.Key != p && entry.Key.lap != null && SameBlock(entry.Value,s)) return false;
104 }
105 return true;
106 }
107 static bool Fits(Passenger p, Seat s, HashSet<Passenger> vacated, Dictionary<Passenger,Seat> proposed) {
108 Flight f = p.booking.flight;
109 Passenger occupant = f.occupied.GetValueOrDefault(s.Id);
110 return !f.blocked.Contains(s.Id) && (occupant == null || vacated.Contains(occupant)) &&
111 ExitAllowed(p,s) && InfantAllowed(p,s,vacated,proposed);
112 }
113 static void PlaceSeat(Passenger p, Seat target) {
114 Flight f = p.booking.flight;
115 if (p.seat != null) f.occupied.Remove(p.seat.Id);
116 p.seat = target;
117 if (target != null) f.occupied[target.Id] = p;
118 }
119 static Seat Select(Passenger p, string cabin) {
120 var candidates = p.booking.flight.aircraft.cabins[cabin].seats
121 .Where(s => Fits(p,s,new HashSet<Passenger>(),new Dictionary<Passenger,Seat>())).ToList();
122 return candidates.FirstOrDefault(s => s.Kind == p.pref) ?? candidates.FirstOrDefault();
123 }
124 static void Reassign(Flight f, List<string> output) {
125 foreach (Passenger p in Passengers(f)) {
126 if (p.NeedsSeat() && p.seat == null) {
127 Seat target = Select(p,p.cabin);
128 if (target != null) {
129 PlaceSeat(p,target);
130 output.Add("REASSIGNED "+p.booking.pnr+" "+p.name+" "+target.Id);
131 }
132 }
133 }

135 static bool Candidate(List<Passenger> people, List<Seat> positions) {
136 var proposed = people.Zip(positions).ToDictionary(pair => pair.First,pair => pair.Second);
137 if (!people.All(p => Fits(p,proposed[p],new HashSet<Passenger>(),proposed))) return false;
138 foreach (Passenger p in people) PlaceSeat(p,proposed[p]);
139 return true;
140 }
141 static string Assign(Booking b, string cabinName) {
142 List<Passenger> people = b.passengers.Where(p => p.NeedsSeat()).ToList();
143 if (people.Count == 1) {
144 PlaceSeat(people.First(),Select(people.First(),cabinName));
145 return "SINGLE";
146 }
147 Cabin cabin = b.flight.aircraft.cabins[cabinName];
148 Aircraft a = b.flight.aircraft;
149 int k = people.Count;
150 for (int stage = 1; stage <= 3; stage++) {
151 int lastH = stage == 3 ? k : 2;
152 for (int h = 2; h <= lastH; h++) {
153 int width = (k+h-1)/h;
154 for (int row = cabin.first; row <= cabin.last; row++) {
155 List<string> blocks = stage == 2 ? new List<string> { string.Join("",cabin.blocks) } : cabin.blocks;
156 foreach (string block in blocks) {
157 int span = stage == 3 ? width : k;
158 for (int x = 0; x+span <= block.Length; x++) {
159 List<Seat> positions = new();
160 for (int j = 0; j < k; j++) {
161 int r = row+(stage == 3 ? j/width : 0);
162 if (r > cabin.last) break;
163 positions.Add(a.seats[r+Convert.ToString(block[x+(stage == 3 ? j%width : j)])]);
164 }
165 if (positions.Count == k && Candidate(people,positions)) return stage == 1 ? "BLOCK" : stage == 2 ? "ROW" : "STACK";
166 }
167 }
168 }
169 }
170 }
171 foreach (Passenger p in people) PlaceSeat(p,Select(p,cabinName));
172 return "SPLIT";
173 }
174 static List<Passenger> RemovalTargets(Passenger p) {
175 return p.lap == null ? new List<Passenger> { p } : new List<Passenger> { p,p.lap };
176 }
177 static void Remove(List<Passenger> targets) {
178 foreach (Passenger p in targets) {
179 PlaceSeat(p,null);
180 p.booking.flight.volunteers.Remove(p);
181 p.booking.passengers.Remove(p);
182 if (p.parent != null) p.parent.lap = null;
183 }

185 static Comparison<Passenger> SeatOrder() {
186 return (a,b) => a.seat.Row != b.seat.Row ? a.seat.Row.CompareTo(b.seat.Row) : a.seat.Letter.CompareTo(b.seat.Letter);

188 static string Load(int seated, int count) {
189 int q = seated*1000/count;
190 return q/10+"."+q%10+"%";
191 }
192 int BoardingRank(Passenger p) {
193 if (p.booking.passengers.Any(q => !q.NeedsSeat())) return 0;
194 if (!p.cabin.Equals("Y")) return 1;
195 if (TierRank(p) <= 1) return 2;
196 return 3;
197 }
198 static int Zone(Passenger p) { return (p.seat.Row-p.booking.flight.aircraft.cabins["Y"].first)/5+1; }
199 List<string> Execute(string line) {
200 string[] t = line.Split(' ');
201 string command = t[0];
202 List<string> output = new();
203 switch (command) {
204 case "AIRCRAFT": {
205 defining = new Aircraft();
206 defining.model = t[1];
207 defining.BaseRow = int.Parse(t[2]);
208 defining.limit = int.Parse(t[3]);
209 break;
210 }
211 case "CABIN": {
212 Cabin c = new Cabin();
213 c.name = t[1];
214 c.first = int.Parse(t[2]);
215 c.last = int.Parse(t[3]);
216 c.blocks = t.Skip(4).ToList();
217 for (int row = c.first; row <= c.last; row++) {
218 for (int block = 0; block < c.blocks.Count; block++) {
219 string letters = c.blocks[block];
220 for (int i = 0; i < letters.Length; i++) {
221 bool window = block == 0 && i == 0 || block == c.blocks.Count-1 && i == letters.Length-1;
222 string kind = window ? "W" : i == 0 || i == letters.Length-1 ? "A" : "M";
223 Seat s = new Seat(row+Convert.ToString(letters[i]),row,letters[i],block,c.name,kind);
224 c.seats.Add(s);
225 defining.seats[s.Id] = s;
226 }
227 }
228 }
229 defining.cabins[c.name] = c;
230 break;
231 }
232 case "EXIT": {
233 for (int i = 1; i < t.Length; i++) defining.exits.Add(int.Parse(t[i]));
234 break;
235 }
236 case "BLOCKED": {
237 defining.blocked.UnionWith(t.Skip(1));
238 break;
239 }
240 case "END": {
241 aircraft[defining.model] = defining;
242 string result = "AIRCRAFT "+defining.model;
243 foreach (string c in Classes) {
244 if (defining.cabins.ContainsKey(c)) result += " "+c+" "+Seats(defining,c,defining.blocked);
245 }
246 output.Add(result);
247 defining = null;
248 break;
249 }
250 case "FLIGHT": {
251 if (flights.ContainsKey(t[1])) Fail("duplicate flight "+t[1]);
252 Aircraft a = aircraft.GetValueOrDefault(t[2]);
253 if (a == null) Fail("unknown aircraft "+t[2]);
254 Flight f = new Flight();
255 f.name = t[1];
256 f.aircraft = a;
257 f.blocked = new HashSet<string>(a.blocked,StringComparer.Ordinal);
258 string result = "FLIGHT "+f.name+" "+a.model;
259 foreach (string c in Classes) {
260 if (a.cabins.ContainsKey(c)) {
261 int count = Seats(a,c,a.blocked), limit = count*(100+int.Parse(t[3]))/100;
262 f.limits[c] = limit;
263 result += " "+c+" "+count+"/"+limit;
264 }
265 }
266 flights[f.name] = f;
267 flightOrder.Add(f);
268 output.Add(result);
269 break;
270 }
271 case "BOOK": {
272 if (used.Contains(t[1])) Fail("duplicate PNR "+t[1]);
273 Flight f = GetFlight(t[2],true);
274 RequireCabin(f,t[3]);
275 Booking b = new Booking();
276 b.pnr = t[1];
277 b.flight = f;
278 HashSet<string> seen = new(StringComparer.Ordinal);
279 List<Passenger> adults = new(), infants = new();
280 for (int i = 4; i < t.Length; i++) {
281 string[] parts = t[i].Split(':');
282 if (!seen.Add(parts[0])) Fail("duplicate name "+parts[0]+" in "+b.pnr);
283 Passenger p = new Passenger();
284 p.name = parts[0];
285 p.type = parts[1];
286 p.pref = parts.Length == 3 ? parts[2] : null;
287 p.cabin = t[3];
288 p.booking = b;
289 b.passengers.Add(p);
290 if (p.type.Equals("ADULT")) adults.Add(p);
291 if (p.type.Equals("INFANT")) infants.Add(p);
292 }
293 var present = Passengers(f).Select(p => p.name).ToHashSet(StringComparer.Ordinal);
294 foreach (Passenger p in b.passengers) {
295 if (present.Contains(p.name)) Fail(p.name+" already on "+f.name);
296 }
297 if (adults.Count == 0) Fail("no adult in "+b.pnr);
298 if (infants.Count > adults.Count) Fail("too many infants in "+b.pnr);
299 int count = b.passengers.Count(p => p.NeedsSeat());
300 if (Booked(f,t[3])+count > f.limits[t[3]]) Fail("limit exceeded for "+f.name+" "+t[3]);
301 for (int i = 0; i < infants.Count; i++) {
302 adults[i].lap = infants[i];
303 infants[i].parent = adults[i];
304 }
305 foreach (Passenger p in b.passengers) p.order = nextOrder++;
306 string method = Assign(b,t[3]);
307 f.bookings.Add(b);
308 bookings[b.pnr] = b;
309 used.Add(b.pnr);
310 List<string> results = b.passengers.Select(p => p.name+":"+p.Result()).ToList();
311 output.Add("BOOKED "+b.pnr+" "+f.name+" "+t[3]+" "+method+" "+string.Join(" ",results));
312 break;
313 }
314 case "SEAT": {
315 Booking b = GetBooking(t[1],true);
316 Passenger p = GetPassenger(b,t[2]);
317 if (!p.NeedsSeat()) Fail("infant "+p.name+" has no seat");
318 Seat target = b.flight.aircraft.seats.GetValueOrDefault(t[3]);
319 if (target == null || !target.Cabin.Equals(p.cabin)) Fail("no seat "+t[3]+" in cabin "+p.cabin);
320 if (b.flight.blocked.Contains(t[3])) Fail("seat "+t[3]+" is blocked");
321 if (b.flight.occupied.ContainsKey(t[3])) Fail("seat "+t[3]+" is occupied");
322 if (!ExitAllowed(p,target)) Fail("exit row "+target.Row+" not allowed for "+p.name);
323 if (!InfantAllowed(p,target,new HashSet<Passenger> { },new Dictionary<Passenger,Seat>())) Fail("infant limit reached at "+t[3]);
324 string old = p.Result();
325 PlaceSeat(p,target);
326 output.Add("MOVED "+b.pnr+" "+p.name+" "+old+" -> "+t[3]);
327 Reassign(b.flight,output);
328 break;
329 }
330 case "SWAP": {
331 Booking a = GetBooking(t[1],true);
332 Passenger p = GetPassenger(a,t[2]);
333 Booking b = GetBooking(t[3],true);
334 Passenger q = GetPassenger(b,t[4]);
335 if (p == q) Fail("cannot swap "+p.name+" with itself");
336 foreach (Passenger x in new List<Passenger> { p,q }) {
337 if (!x.NeedsSeat()) Fail("infant "+x.name+" has no seat");
338 }
339 if (a.flight != b.flight) Fail(a.pnr+" and "+b.pnr+" are on different flights");
340 if (!p.cabin.Equals(q.cabin)) Fail(p.name+" and "+q.name+" are in different cabins");
341 foreach (Passenger x in new List<Passenger> { p,q }) {
342 if (x.seat == null) Fail(x.name+" has no seat");
343 }
344 HashSet<Passenger> vacated = new() { p,q };
345 Dictionary<Passenger,Seat> proposed = new Dictionary<Passenger,Seat> { [p] = q.seat, [q] = p.seat };
346 if (!Fits(p,q.seat,vacated,proposed) || !Fits(q,p.seat,vacated,proposed)) Fail("swap of "+p.name+" and "+q.name+" breaks seating rules");
347 Seat ps = p.seat, qs = q.seat;
348 PlaceSeat(p,null);
349 PlaceSeat(q,null);
350 PlaceSeat(p,qs);
351 PlaceSeat(q,ps);
352 output.Add("SWAPPED "+a.pnr+" "+p.name+" "+p.Result()+" "+b.pnr+" "+q.name+" "+q.Result());
353 Reassign(a.flight,output);
354 break;
355 }
356 case "CANCEL": {
357 Booking b = GetBooking(t[1],true);
358 List<Passenger> targets;
359 if (t.Length == 2) targets = new List<Passenger>(b.passengers);
360 else {
361 Passenger p = GetPassenger(b,t[2]);
362 List<Passenger> selected = RemovalTargets(p);
363 targets = b.passengers.Where(selected.Contains).ToList();
364 List<Passenger> remaining = b.passengers.Where(q => !selected.Contains(q)).ToList();
365 if (remaining.Count != 0 && !remaining.Any(q => q.type == "ADULT")) Fail(b.pnr+" would have no adult");
366 }
367 string names = string.Join(" ",targets.Select(p => p.name).ToList());
368 Remove(targets);
369 output.Add("CANCELLED "+b.pnr+" "+names);
370 Reassign(b.flight,output);
371 break;
372 }
373 case "STATUS": {
374 tiers[t[1]] = t[2];
375 output.Add("STATUS "+t[1]+" "+t[2]);
376 break;
377 }
378 case "CHECKIN": {
379 Booking b = GetBooking(t[1],true);
380 if (b.Checked) Fail(b.pnr+" already checked in");
381 b.Checked = true;
382 output.Add("CHECKIN "+b.pnr+" "+b.passengers.Count);
383 break;
384 }
385 case "VOLUNTEER": {
386 Booking b = GetBooking(t[1],true);
387 Passenger p = GetPassenger(b,t[2]);
388 if (!p.type.Equals("ADULT") || p.lap != null) Fail(p.name+" cannot volunteer");
389 if (p.seat == null) Fail(p.name+" has no seat");
390 if (b.flight.volunteers.Contains(p)) Fail(p.name+" already volunteered");
391 b.flight.volunteers.Add(p);
392 output.Add("VOLUNTEER "+b.pnr+" "+p.name+" "+p.Result());
393 break;
394 }
395 case "UPGRADE": {
396 Flight f = GetFlight(t[1],true);
397 List<string> classes = Classes.Where(f.aircraft.cabins.ContainsKey).ToList();
398 int count = 0;
399 for (int i = 0; i+1 < classes.Count; i++) {
400 string higher = classes[i], lower = classes[i+1];
401 List<Passenger> targets = Passengers(f).Where(p => p.cabin == lower && p.type == "ADULT" && p.lap == null && Tier(p) != "NONE")
402 .OrderBy(TierRank).ThenBy(p => p.order).ToList();
403 foreach (Passenger p in targets) {
404 if (Booked(f,higher) >= f.limits[higher]) break;
405 Seat target = Select(p,higher);
406 if (target == null) break;
407 string old = p.Result();
408 p.cabin = higher;
409 PlaceSeat(p,target);
410 output.Add("UPGRADED "+p.booking.pnr+" "+p.name+" "+lower+" "+old+" -> "+higher+" "+target.Id);
411 count++;
412 }
413 }
414 Reassign(f,output);
415 output.Add("UPGRADE "+f.name+" "+count);
416 break;
417 }
418 case "CLOSE": {
419 Flight f = GetFlight(t[1],true);
420 foreach (Booking b in f.bookings) {
421 if (b.passengers.Count != 0 && !b.Checked) {
422 output.Add("NOSHOW "+b.pnr+" "+b.passengers.Count);
423 Remove(new List<Passenger>(b.passengers));
424 }
425 }
426 Reassign(f,output);
427 int count = 0, compensation = 0;
428 foreach (string c in Classes) {
429 if (!f.aircraft.cabins.ContainsKey(c)) continue;
430 List<Passenger> volunteers = f.volunteers.Where(p => p.seat != null && p.cabin == c).ToList();
431 foreach (Passenger p in volunteers) {
432 bool useful = Passengers(f).Any(q => q.NeedsSeat() && q.seat == null && q.cabin == c && Fits(q,p.seat,new HashSet<Passenger> { p },new Dictionary<Passenger,Seat>()));
433 if (useful) {
434 int value = c.Equals("F") ? 500 : c.Equals("J") ? 300 : 150;
435 output.Add("DENIED VOLUNTARY "+p.booking.pnr+" "+p.name+" "+c+" "+value);
436 count++;
437 compensation += value;
438 Remove(RemovalTargets(p));
439 Reassign(f,output);
440 }
441 }
442 }
443 foreach (Passenger p in Passengers(f)) {
444 if (p.NeedsSeat() && p.seat == null) {
445 int value = p.cabin.Equals("F") ? 1000 : p.cabin.Equals("J") ? 600 : 300;
446 if (p.type.Equals("CHILD")) value /= 2;
447 output.Add("DENIED INVOLUNTARY "+p.booking.pnr+" "+p.name+" "+p.cabin+" "+value);
448 count++;
449 compensation += value;
450 Remove(RemovalTargets(p));
451 }
452 }
453 f.closed = true;
454 output.Add("CLOSED "+f.name+" denied "+count+" compensation "+compensation);
455 break;
456 }
457 case "BLOCK":
458 case "UNBLOCK": {
459 Flight f = GetFlight(t[1],true);
460 Seat target = f.aircraft.seats.GetValueOrDefault(t[2]);
461 if (target == null) Fail("no seat "+t[2]+" on "+f.name);
462 bool blocked = f.blocked.Contains(t[2]);
463 if (command.Equals("BLOCK") && blocked) Fail("seat "+t[2]+" is blocked");
464 if (command.Equals("UNBLOCK") && !blocked) Fail("seat "+t[2]+" is not blocked");
465 output.Add(command+" "+f.name+" "+t[2]);
466 if (command.Equals("BLOCK")) {
467 f.blocked.Add(t[2]);
468 Passenger p = f.occupied.GetValueOrDefault(t[2]);
469 if (p != null) {
470 PlaceSeat(p,null);
471 PlaceSeat(p,Select(p,p.cabin));
472 output.Add("DISPLACED "+p.booking.pnr+" "+p.name+" "+t[2]+" -> "+p.Result());
473 }
474 } else f.blocked.Remove(t[2]);
475 Reassign(f,output);
476 break;
477 }
478 case "MAP": {
479 Flight f = GetFlight(t[1],false);
480 output.Add("MAP "+f.name);
481 foreach (string name in Classes) {
482 Cabin c = f.aircraft.cabins.GetValueOrDefault(name);
483 if (c == null) continue;
484 output.Add("CABIN "+name);
485 output.Add(" "+string.Join(" ",c.blocks));
486 for (int row = c.first; row <= c.last; row++) {
487 List<string> blocks = new();
488 foreach (string block in c.blocks) {
489 StringBuilder result = new StringBuilder();
490 foreach (char letter in block.ToCharArray()) {
491 string id = row+Convert.ToString(letter);
492 result.Append(f.blocked.Contains(id) ? '#' : f.occupied.ContainsKey(id) ? 'X' : f.aircraft.exits.Contains(row) ? 'E' : '.');
493 }
494 blocks.Add(result.ToString());
495 }
496 output.Add($"{row,2} "+string.Join(" ",blocks));
497 }
498 }
499 break;
500 }
501 case "MANIFEST": {
502 Flight f = GetFlight(t[1],false);
503 List<Passenger> all = Passengers(f);
504 List<Passenger> seated = all.Where(p => p.seat != null).OrderBy(p => p.seat.Row).ThenBy(p => p.seat.Letter).ToList();
505 List<Passenger> unassigned = all.Where(p => p.NeedsSeat() && p.seat == null).ToList();
506 output.Add("MANIFEST "+f.name+" seated "+seated.Count+" unassigned "+unassigned.Count+" infants "+all.Count(p => !p.NeedsSeat()));
507 foreach (Passenger p in seated.Concat(unassigned)) {
508 string result = (p.seat == null ? "---" : p.seat.Id)+" "+p.cabin+" "+p.booking.pnr+" "+p.name+" "+p.type;
509 if (p.lap != null) result += " +"+p.lap.name;
510 if (!Tier(p).Equals("NONE")) result += " "+Tier(p);
511 if (p.booking.Checked) result += " CHECKED";
512 if (f.volunteers.Contains(p)) result += " VOL";
513 output.Add(result);
514 }
515 break;
516 }
517 case "LOADFACTOR": {
518 Flight f = GetFlight(t[1],false);
519 output.Add("LOADFACTOR "+f.name);
520 int totalSeated = 0, totalSeats = 0;
521 foreach (string c in Classes) {
522 if (!f.aircraft.cabins.ContainsKey(c)) continue;
523 int count = Seats(f.aircraft,c,f.blocked);
524 int seated = Passengers(f).Count(p => p.cabin == c && p.seat != null);
525 output.Add(c+" "+seated+"/"+count+" "+Load(seated,count)+" booked "+Booked(f,c)+"/"+f.limits[c]);
526 totalSeated += seated;
527 totalSeats += count;
528 }
529 output.Add("ALL "+totalSeated+"/"+totalSeats+" "+Load(totalSeated,totalSeats));
530 break;
531 }
532 case "BALANCE": {
533 Flight f = GetFlight(t[1],false);
534 int weight = 0, moment = 0;
535 foreach (Passenger p in Passengers(f)) {
536 if (p.seat == null) continue;
537 int w = (p.type.Equals("ADULT") ? 80 : 40)+(p.lap == null ? 0 : 10);
538 weight += w;
539 moment += w*(p.seat.Row-f.aircraft.BaseRow);
540 }
541 int limit = f.aircraft.limit;
542 output.Add("BALANCE "+f.name+" weight "+weight+" moment "+moment+" limit "+limit+" "+(moment < -limit ? "NOSE-HEAVY" : moment > limit ? "TAIL-HEAVY" : "OK"));
543 break;
544 }
545 case "BOARDING": {
546 Flight f = GetFlight(t[1],false);
547 List<Passenger> ordered = Passengers(f).Where(p => p.seat != null).ToList();
548 ordered.Sort((a,b) => {
549 int ar = BoardingRank(a), br = BoardingRank(b);
550 if (ar != br) return ar.CompareTo(br);
551 if (ar < 3) return SeatOrder()(a,b);
552 if (Zone(a) != Zone(b)) return Zone(b).CompareTo(Zone(a));
553 int ak = "WMA".IndexOf(a.seat.Kind), bk = "WMA".IndexOf(b.seat.Kind);
554 if (ak != bk) return ak.CompareTo(bk);
555 if (a.seat.Row != b.seat.Row) return b.seat.Row.CompareTo(a.seat.Row);
556 return a.seat.Letter.CompareTo(b.seat.Letter);
557 });
558 output.Add("BOARDING "+f.name+" "+ordered.Count);
559 for (int i = 0; i < ordered.Count; i++) {
560 Passenger p = ordered[i];
561 int rank = BoardingRank(p);
562 string group = rank == 0 ? "PRE" : rank == 1 ? "CABIN" : rank == 2 ? "TIER" : "ZONE"+Zone(p);
563 output.Add((i+1)+" "+group+" "+p.Result()+" "+p.booking.pnr+" "+p.name);
564 }
565 break;
566 }
567 case "AVAIL": {
568 Flight f = GetFlight(t[1],false);
569 RequireCabin(f,t[2]);
570 var free = f.aircraft.cabins[t[2]].seats.Where(s => !f.blocked.Contains(s.Id) && !f.occupied.ContainsKey(s.Id)).ToList();
571 output.Add("AVAIL "+f.name+" "+t[2]+" free "+free.Count+
572 " W "+free.Count(s => s.Kind == "W")+" M "+free.Count(s => s.Kind == "M")+
573 " A "+free.Count(s => s.Kind == "A")+" EXIT "+free.Count(s => f.aircraft.exits.Contains(s.Row)));
574 break;
575 }
576 case "INFO": {
577 Booking b = GetBooking(t[1],false);
578 output.Add("INFO "+b.pnr+" "+b.flight.name+" "+b.passengers.Count+" "+(b.Checked ? "CHECKED" : "NOT-CHECKED"));
579 foreach (Passenger p in b.passengers) output.Add(p.name+" "+p.type+" "+p.cabin+" "+p.Result());
580 break;
581 }
582 case "FIND": {
583 List<string> found = new();
584 foreach (Flight f in flightOrder) {
585 foreach (Passenger p in Passengers(f)) {
586 if (p.name.Equals(t[1])) found.Add(f.name+" "+p.booking.pnr+" "+p.cabin+" "+p.Result()+" "+(f.closed ? "CLOSED" : "OPEN"));
587 }
588 }
589 output.Add("FIND "+t[1]+" "+tiers.GetValueOrDefault(t[1],"NONE")+" "+found.Count);
590 output.AddRange(found);
591 break;
592 }
593 }
594 return output;
595 }
