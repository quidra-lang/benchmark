34 closed bool
35 limit [3]int

47 booking, kind, preference, cabin, seat, lap int

54var modelIDs = map[string]int{}
55var flightIDs = map[string]int{}
56var bookingIDs = map[string]int{}
57var tiers = map[string]int{}
58var building *aircraft
59const classes = "FJY"
60var typeNames = []string{"ADULT", "CHILD", "INFANT"}

63func fail(message string) { panic(message) }
64func number(s string) int {
65 value, _ := strconv.Atoi(s)
66 return value
67}
68func cabinID(s string) int { return strings.IndexByte(classes, s[0]) }
69func tier(id int) int {
70 if value, exists := tiers[people[id].name]; exists { return value }
71 return 3
72}
73func seatKey(s string) int { return number(s[:len(s)-1])*26+int(s[len(s)-1]-'A') }
74func seatName(a *aircraft, id int) string {
75 if id < 0 { return "UNASSIGNED" }
76 return fmt.Sprintf("%d%c", a.seats[id].row, a.seats[id].letter)
77}
78func flightID(name string, open bool) int {
79 id, exists := flightIDs[name]
80 if !exists { fail("unknown flight "+name) }
81 if open && flights[id].closed { fail("flight "+name+" is closed") }
82 return id
83}
84func activeCount(b int) int {
85 count := 0
86 for _, id := range bookings[b].people { if people[id].active { count++ } }
87 return count
88}
89func bookingID(pnr string, open bool) int {
90 id, exists := bookingIDs[pnr]
91 if !exists || activeCount(id) == 0 { fail("unknown PNR "+pnr) }
92 if open && flights[bookings[id].flight].closed { fail("flight "+flights[bookings[id].flight].name+" is closed") }
93 return id
94}
95func passengerID(b int, name string) int {
96 for _, id := range bookings[b].people {
97 if people[id].active && people[id].name == name { return id }
98 }
99 fail("no passenger "+name+" in "+bookings[b].pnr)
100 return -1
101}
102func resolve(pnr, name string) int { return passengerID(bookingID(pnr, true), name) }
103func hasInfant(id int) bool { return people[id].kind == 0 && people[id].lap >= 0 && people[people[id].lap].active }
104func bookingInfants(b int) bool {
105 for _, id := range bookings[b].people {
106 if people[id].active && people[id].kind == 2 { return true }
107 }
108 return false
109}
110func bookedCount(f, c int) int {
111 count := 0
112 for _, id := range flights[f].passengers {
113 p := people[id]
114 if p.active && p.kind != 2 && p.cabin == c { count++ }
115 }
116 return count
117}
118func exitAllowed(id int, a *aircraft, target int) bool {
119 return !a.exits[a.seats[target].row] || people[id].kind == 0 && !hasInfant(id)
120}
121func infantAllowed(f, id, target int, moving, ids, assigned []int) bool {
122 if !hasInfant(id) { return true }
123 plane := aircrafts[flights[f].model]
124 s := plane.seats[target]
125 for _, letter := range plane.cabins[s.cabin].blocks[s.block] {
126 peer := plane.lookup[s.row*26+int(letter-'A')]
127 owner := flights[f].occupants[peer]
128 if owner >= 0 && owner != id && !slices.Contains(moving, owner) && hasInfant(owner) { return false }
129 }
130 for i, other := range ids {
131 t := plane.seats[assigned[i]]
132 if other != id && hasInfant(other) && t.row == s.row && t.block == s.block { return false }
133 }
134 return true
135}
136func fits(f int, ids, assigned, vacated []int) bool {
137 plane := aircrafts[flights[f].model]
138 moving := slices.Concat(ids, vacated)
139 for i, id := range ids {
140 target := assigned[i]
141 owner := flights[f].occupants[target]
142 if flights[f].blocked[target] || owner >= 0 && !slices.Contains(moving, owner) || !exitAllowed(id, plane, target) || !infantAllowed(f, id, target, moving, ids, assigned) { return false }
143 }
144 return true
145}
146func chooseSeat(f, id, c int) int {
147 plane := aircrafts[flights[f].model]
148 first := -1
149 for target, s := range plane.seats {
150 if s.cabin != c || flights[f].blocked[target] || flights[f].occupants[target] >= 0 || !exitAllowed(id, plane, target) || !infantAllowed(f, id, target, nil, nil, nil) { continue }
151 if first < 0 { first = target }
152 if people[id].preference < 0 || people[id].preference == s.kind { return target }
153 }
154 return first
155}
156func seatPerson(f, id, target int) {
157 if people[id].seat >= 0 { flights[f].occupants[people[id].seat] = -1 }
158 people[id].seat = target
159 if target >= 0 { flights[f].occupants[target] = id }
160}
161func reassign(f int) []string {
162 var out []string
163 plane := aircrafts[flights[f].model]
164 for _, id := range flights[f].passengers {
165 p := people[id]
166 if !p.active || p.kind == 2 || p.seat >= 0 { continue }
167 target := chooseSeat(f, id, p.cabin)
168 if target < 0 { continue }
169 seatPerson(f, id, target)
170 out = append(out, "REASSIGNED "+bookings[p.booking].pnr+" "+p.name+" "+seatName(plane, target))
171 }
172 return out
173}
174func removePerson(id int) {
175 p := people[id]
176 if !p.active { return }
177 lap := p.lap
178 if p.kind == 0 && lap >= 0 { removePerson(lap) }
179 if p.kind == 2 && lap >= 0 { people[lap].lap = -1 }
180 seatPerson(bookings[p.booking].flight, id, -1)
181 p.active, p.volunteer, p.lap = false, false, -1
182}
183func passengerResult(id int) string {
184 if people[id].kind == 2 { return "LAP" }
185 f := flights[bookings[people[id].booking].flight]
186 return seatName(aircrafts[f.model], people[id].seat)
187}
188func assignBooking(f, c int, ids []int) string {
189 if len(ids) == 1 {
190 seatPerson(f, ids[0], chooseSeat(f, ids[0], c))
191 return "SINGLE"
192 }
193 plane := aircrafts[flights[f].model]
194 cabin := plane.cabins[c]
195 count := len(ids)
196 for stage := 0; stage < 3; stage++ {
197 blocks := cabin.blocks
198 if stage == 1 { blocks = []string{strings.Join(cabin.blocks, "")} }
199 firstHeight, lastHeight := 1, 1
200 if stage == 2 { firstHeight, lastHeight = 2, count }
201 for height := firstHeight; height <= lastHeight; height++ {
202 width := count
203 if stage == 2 { width = (count+height-1)/height }
204 rows := (count+width-1)/width
205 for row := cabin.from; row+rows-1 <= cabin.to; row++ {
206 for _, block := range blocks {
207 for x := 0; x+width <= len(block); x++ {
208 candidate := make([]int, count)
209 for j := range candidate { candidate[j] = plane.lookup[(row+j/width)*26+int(block[x+j%width]-'A')] }
210 if !fits(f, ids, candidate, nil) { continue }
211 for j, id := range ids { seatPerson(f, id, candidate[j]) }
212 return []string{"BLOCK", "ROW", "STACK"}[stage]
213 }
214 }
215 }
216 }
217 }
218 for _, id := range ids { seatPerson(f, id, chooseSeat(f, id, c)) }
219 return "SPLIT"
220}
221func execute(v []string) []string {
222 command := v[0]
223 var out []string
224 switch command {
225 case "AIRCRAFT":
226 building = &aircraft{model: v[1], base: number(v[2]), limit: number(v[3])}
227 for i := range building.lookup { building.lookup[i] = -1 }
228 case "CABIN":
229 c := cabinID(v[1])
230 building.cabins[c] = cabin{number(v[2]), number(v[3]), slices.Clone(v[4:])}
231 definition := building.cabins[c]
232 for row := definition.from; row <= definition.to; row++ {
233 for b, block := range definition.blocks {
234 for position := range block {
235 kind := 1
236 if b == 0 && position == 0 || b == len(definition.blocks)-1 && position == len(block)-1 { kind = 0 } else if position == 0 || position == len(block)-1 { kind = 2 }
237 letter := block[position]
238 building.lookup[row*26+int(letter-'A')] = len(building.seats)
239 building.seats = append(building.seats, seat{row, c, b, kind, letter})
240 building.blocked = append(building.blocked, false)
241 }
242 }
243 }
244 case "EXIT":
245 for _, value := range v[1:] { building.exits[number(value)] = true }
246 case "BLOCKED":
247 for _, value := range v[1:] { building.blocked[building.lookup[seatKey(value)]] = true }
248 case "END":
249 for i, s := range building.seats {
250 if !building.blocked[i] { building.capacity[s.cabin]++ }
251 }
252 line := "AIRCRAFT "+building.model
253 for c := 0; c < 3; c++ {
254 if building.cabins[c].from > 0 { line += fmt.Sprintf(" %c %d", classes[c], building.capacity[c]) }
255 }
256 modelIDs[building.model] = len(aircrafts)
257 aircrafts = append(aircrafts, building)
258 out = append(out, line)
259 case "FLIGHT":
260 if _, exists := flightIDs[v[1]]; exists { fail("duplicate flight "+v[1]) }
261 model, exists := modelIDs[v[2]]
262 if !exists { fail("unknown aircraft "+v[2]) }
263 plane := aircrafts[model]
264 f := &flight{name: v[1], model: model, blocked: slices.Clone(plane.blocked), occupants: make([]int, len(plane.seats))}
265 for i := range f.occupants { f.occupants[i] = -1 }
266 line := "FLIGHT "+f.name+" "+plane.model
267 for c := 0; c < 3; c++ {
268 f.limit[c] = plane.capacity[c]*(100+number(v[3]))/100
269 if plane.cabins[c].from > 0 { line += fmt.Sprintf(" %c %d/%d", classes[c], plane.capacity[c], f.limit[c]) }
270 }
271 flightIDs[f.name] = len(flights)
272 flights = append(flights, f)
273 out = append(out, line)
274 case "BOOK":
275 if _, exists := bookingIDs[v[1]]; exists { fail("duplicate PNR "+v[1]) }
276 f, c := flightID(v[2], true), cabinID(v[3])
277 plane := aircrafts[flights[f].model]
278 if plane.cabins[c].from == 0 { fail("no cabin "+v[3]+" on "+v[2]) }
279 var draft []*person
280 seen := map[string]bool{}
281 var adults, infants []int
282 for _, item := range v[4:] {
283 fields := strings.Split(item, ":")
284 if seen[fields[0]] { fail("duplicate name "+fields[0]+" in "+v[1]) }
285 seen[fields[0]] = true
286 p := &person{name: fields[0], booking: len(bookings), kind: slices.Index(typeNames, fields[1]), cabin: c, preference: -1, seat: -1, lap: -1, active: true}
287 if len(fields) == 3 { p.preference = strings.IndexByte("WMA", fields[2][0]) }
288 if p.kind == 0 { adults = append(adults, len(draft)) }
289 if p.kind == 2 { infants = append(infants, len(draft)) }
290 draft = append(draft, p)
291 }
292 for _, p := range draft {
293 for _, id := range flights[f].passengers {
294 if people[id].active && people[id].name == p.name { fail(p.name+" already on "+v[2]) }
295 }
296 }
297 if len(adults) == 0 { fail("no adult in "+v[1]) }
298 if len(infants) > len(adults) { fail("too many infants in "+v[1]) }
299 if bookedCount(f, c)+len(draft)-len(infants) > flights[f].limit[c] { fail("limit exceeded for "+v[2]+" "+v[3]) }
300 base := len(people)
301 for i, infant := range infants {
302 draft[adults[i]].lap = base+infant
303 draft[infant].lap = base+adults[i]
304 }
305 b := &booking{pnr: v[1], flight: f}
306 var needs []int
307 for _, p := range draft {
308 id := len(people)
309 people = append(people, p)
310 b.people = append(b.people, id)
311 flights[f].passengers = append(flights[f].passengers, id)
312 if p.kind != 2 { needs = append(needs, id) }
313 }
314 bookingIDs[b.pnr] = len(bookings)
315 bookings = append(bookings, b)
316 method := assignBooking(f, c, needs)
317 line := "BOOKED "+b.pnr+" "+v[2]+" "+v[3]+" "+method
318 for _, id := range b.people { line += " "+people[id].name+":"+passengerResult(id) }
319 out = append(out, line)
320 case "STATUS":
321 tiers[v[1]] = slices.Index(tierNames, v[2])
322 out = append(out, "STATUS "+v[1]+" "+v[2])
323 case "FIND":
324 membership := 3
325 if value, exists := tiers[v[1]]; exists { membership = value }
326 var found []string
327 for _, f := range flights {
328 for _, id := range f.passengers {
329 p := people[id]
330 if !p.active || p.name != v[1] { continue }
331 state := "OPEN"
332 if f.closed { state = "CLOSED" }
333 found = append(found, fmt.Sprintf("%s %s %c %s %s", f.name, bookings[p.booking].pnr, classes[p.cabin], passengerResult(id), state))
334 }
335 }
336 out = append(out, fmt.Sprintf("FIND %s %s %d", v[1], tierNames[membership], len(found)))
337 out = append(out, found...)
338 case "SEAT":
339 id := resolve(v[1], v[2])
340 p := people[id]
341 f := bookings[p.booking].flight
342 plane := aircrafts[flights[f].model]
343 if p.kind == 2 { fail("infant "+p.name+" has no seat") }
344 target := plane.lookup[seatKey(v[3])]
345 if target < 0 || plane.seats[target].cabin != p.cabin { fail(fmt.Sprintf("no seat %s in cabin %c", v[3], classes[p.cabin])) }
346 if flights[f].blocked[target] { fail("seat "+v[3]+" is blocked") }
347 if flights[f].occupants[target] >= 0 { fail("seat "+v[3]+" is occupied") }
348 if !exitAllowed(id, plane, target) { fail(fmt.Sprintf("exit row %d not allowed for %s", plane.seats[target].row, p.name)) }
349 if !infantAllowed(f, id, target, nil, nil, nil) { fail("infant limit reached at "+v[3]) }
350 old := seatName(plane, p.seat)
351 seatPerson(f, id, target)
352 out = append(out, "MOVED "+v[1]+" "+v[2]+" "+old+" -> "+v[3])
353 out = append(out, reassign(f)...)
354 case "SWAP":
355 first, second := resolve(v[1], v[2]), resolve(v[3], v[4])
356 if first == second { fail("cannot swap "+v[2]+" with itself") }
357 for _, id := range []int{first, second} {
358 if people[id].kind == 2 { fail("infant "+people[id].name+" has no seat") }
359 }
360 f := bookings[people[first].booking].flight
361 if f != bookings[people[second].booking].flight { fail(v[1]+" and "+v[3]+" are on different flights") }
362 if people[first].cabin != people[second].cabin { fail(v[2]+" and "+v[4]+" are in different cabins") }
363 for _, id := range []int{first, second} {
364 if people[id].seat < 0 { fail(people[id].name+" has no seat") }
365 }
366 oldFirst, oldSecond := people[first].seat, people[second].seat
367 if !fits(f, []int{first, second}, []int{oldSecond, oldFirst}, nil) { fail("swap of "+v[2]+" and "+v[4]+" breaks seating rules") }
368 flights[f].occupants[oldFirst], flights[f].occupants[oldSecond] = -1, -1
369 people[first].seat, people[second].seat = -1, -1
370 seatPerson(f, first, oldSecond)
371 seatPerson(f, second, oldFirst)
372 plane := aircrafts[flights[f].model]
373 out = append(out, "SWAPPED "+v[1]+" "+v[2]+" "+seatName(plane, oldSecond)+" "+v[3]+" "+v[4]+" "+seatName(plane, oldFirst))
374 out = append(out, reassign(f)...)
375 case "CANCEL":
376 b := bookingID(v[1], true)
377 target := -1
378 if len(v) == 3 { target = passengerID(b, v[2]) }
379 var removed []int
380 for _, id := range bookings[b].people {
381 if people[id].active && (target < 0 || id == target || people[target].kind == 0 && people[target].lap == id) { removed = append(removed, id) }
382 }
383 remaining, adults := 0, 0
384 for _, id := range bookings[b].people {
385 if people[id].active && !slices.Contains(removed, id) {
386 remaining++
387 if people[id].kind == 0 { adults++ }
388 }
389 }
390 if target >= 0 && remaining > 0 && adults == 0 { fail(v[1]+" would have no adult") }
391 line := "CANCELLED "+v[1]
392 for _, id := range removed { line += " "+people[id].name }
393 for _, id := range removed { removePerson(id) }
394 out = append(out, line)
395 out = append(out, reassign(bookings[b].flight)...)
396 case "CHECKIN", "VOLUNTEER", "INFO":
397 b := bookingID(v[1], command != "INFO")
398 book := bookings[b]
399 if command == "CHECKIN" {
400 if book.checked { fail(v[1]+" already checked in") }
401 book.checked = true
402 out = append(out, fmt.Sprintf("CHECKIN %s %d", book.pnr, activeCount(b)))
403 } else if command == "INFO" {
404 state := "NOT-CHECKED"
405 if book.checked { state = "CHECKED" }
406 out = append(out, fmt.Sprintf("INFO %s %s %d %s", book.pnr, flights[book.flight].name, activeCount(b), state))
407 for _, id := range book.people {
408 p := people[id]
409 if p.active { out = append(out, fmt.Sprintf("%s %s %c %s", p.name, typeNames[p.kind], classes[p.cabin], passengerResult(id))) }
410 }
411 } else {
412 id := passengerID(b, v[2])
413 p := people[id]
414 if p.kind != 0 || hasInfant(id) { fail(p.name+" cannot volunteer") }
415 if p.seat < 0 { fail(p.name+" has no seat") }
416 if p.volunteer { fail(p.name+" already volunteered") }
417 p.volunteer = true
418 flights[book.flight].volunteers = append(flights[book.flight].volunteers, id)
419 out = append(out, "VOLUNTEER "+book.pnr+" "+p.name+" "+passengerResult(id))
420 }
421 default:
422 modifying := command == "UPGRADE" || command == "CLOSE" || command == "BLOCK" || command == "UNBLOCK"
423 f := flightID(v[1], modifying)
424 flight := flights[f]
425 plane := aircrafts[flight.model]
426 switch command {
427 case "BLOCK", "UNBLOCK":
428 target := plane.lookup[seatKey(v[2])]
429 if target < 0 { fail("no seat "+v[2]+" on "+flight.name) }
430 if command == "BLOCK" && flight.blocked[target] { fail("seat "+v[2]+" is blocked") }
431 if command == "UNBLOCK" && !flight.blocked[target] { fail("seat "+v[2]+" is not blocked") }
432 flight.blocked[target] = command == "BLOCK"
433 out = append(out, command+" "+flight.name+" "+v[2])
434 id := flight.occupants[target]
435 if command == "BLOCK" && id >= 0 {
436 seatPerson(f, id, -1)
437 seatPerson(f, id, chooseSeat(f, id, people[id].cabin))
438 out = append(out, "DISPLACED "+bookings[people[id].booking].pnr+" "+people[id].name+" "+v[2]+" -> "+passengerResult(id))
439 }
440 out = append(out, reassign(f)...)
441 case "UPGRADE":
442 var cabins []int
443 for c := 0; c < 3; c++ { if plane.cabins[c].from > 0 { cabins = append(cabins, c) } }
444 upgraded := 0
445 for pair := 0; pair+1 < len(cabins); pair++ {
446 higher, lower := cabins[pair], cabins[pair+1]
447 booked := bookedCount(f, higher)
448 type target struct { tier, id int }
449 var targets []target
450 for _, id := range flight.passengers {
451 p := people[id]
452 if p.active && p.cabin == lower && p.kind == 0 && !hasInfant(id) && tier(id) < 3 { targets = append(targets, target{tier(id), id}) }
453 }
454 sort.SliceStable(targets, func(i, j int) bool { return targets[i].tier < targets[j].tier })
455 for _, item := range targets {
456 if booked >= flight.limit[higher] { break }
457 id := item.id
458 target := chooseSeat(f, id, higher)
459 if target < 0 { break }
460 old := passengerResult(id)
461 people[id].cabin = higher
462 seatPerson(f, id, target)
463 booked++
464 upgraded++
465 out = append(out, fmt.Sprintf("UPGRADED %s %s %c %s -> %c %s", bookings[people[id].booking].pnr, people[id].name, classes[lower], old, classes[higher], seatName(plane, target)))
466 }
467 }
468 out = append(out, reassign(f)...)
469 out = append(out, fmt.Sprintf("UPGRADE %s %d", flight.name, upgraded))
470 case "CLOSE":
471 for b, book := range bookings {
472 if book.flight != f || book.checked || activeCount(b) == 0 { continue }
473 out = append(out, fmt.Sprintf("NOSHOW %s %d", book.pnr, activeCount(b)))
474 for _, id := range book.people { removePerson(id) }
475 }
476 out = append(out, reassign(f)...)
477 denied, compensation := 0, 0
478 voluntary, involuntary := []int{500, 300, 150}, []int{1000, 600, 300}
479 for c := 0; c < 3; c++ {
480 var targets []int
481 for _, id := range flight.volunteers {
482 p := people[id]
483 if p.active && p.volunteer && p.seat >= 0 && p.cabin == c { targets = append(targets, id) }
484 }
485 for _, id := range targets {
486 target := people[id].seat
487 useful := false
488 for _, other := range flight.passengers {
489 p := people[other]
490 if p.active && p.kind != 2 && p.cabin == c && p.seat < 0 && fits(f, []int{other}, []int{target}, []int{id}) {
491 useful = true
492 break
493 }
494 }
495 if !useful { continue }
496 out = append(out, fmt.Sprintf("DENIED VOLUNTARY %s %s %c %d", bookings[people[id].booking].pnr, people[id].name, classes[c], voluntary[c]))
497 denied++
498 compensation += voluntary[c]
499 removePerson(id)
500 out = append(out, reassign(f)...)
501 }
502 }
503 for _, id := range flight.passengers {
504 p := people[id]
505 if !p.active || p.kind == 2 || p.seat >= 0 { continue }
506 value := involuntary[p.cabin]
507 if p.kind == 1 { value /= 2 }
508 out = append(out, fmt.Sprintf("DENIED INVOLUNTARY %s %s %c %d", bookings[p.booking].pnr, p.name, classes[p.cabin], value))
509 denied++
510 compensation += value
511 removePerson(id)
512 }
513 flight.closed = true
514 out = append(out, fmt.Sprintf("CLOSED %s denied %d compensation %d", flight.name, denied, compensation))
515 case "MAP":
516 out = append(out, "MAP "+flight.name)
517 for c := 0; c < 3; c++ {
518 cabin := plane.cabins[c]
519 if cabin.from == 0 { continue }
520 out = append(out, fmt.Sprintf("CABIN %c", classes[c]), " "+strings.Join(cabin.blocks, " "))
521 for row := cabin.from; row <= cabin.to; row++ {
522 line := fmt.Sprintf("%2d ", row)
523 for b, block := range cabin.blocks {
524 if b > 0 { line += " " }
525 for i := range block {
526 target := plane.lookup[row*26+int(block[i]-'A')]
527 mark := "."
528 if plane.exits[row] { mark = "E" }
529 if flight.occupants[target] >= 0 { mark = "X" }
530 if flight.blocked[target] { mark = "#" }
531 line += mark
532 }
533 }
534 out = append(out, line)
535 }
536 }
537 case "MANIFEST":
538 var order []int
539 for _, owner := range flight.occupants { if owner >= 0 { order = append(order, owner) } }
540 seated, unassigned, infants := len(order), 0, 0
541 for _, id := range flight.passengers {
542 p := people[id]
543 if !p.active { continue }
544 if p.kind == 2 { infants++ } else if p.seat < 0 {
545 unassigned++
546 order = append(order, id)
547 }
548 }
549 out = append(out, fmt.Sprintf("MANIFEST %s seated %d unassigned %d infants %d", flight.name, seated, unassigned, infants))
550 for _, id := range order {
551 p := people[id]
552 b := bookings[p.booking]
553 position := "---"
554 if p.seat >= 0 { position = seatName(plane, p.seat) }
555 line := fmt.Sprintf("%s %c %s %s %s", position, classes[p.cabin], b.pnr, p.name, typeNames[p.kind])
556 if hasInfant(id) { line += " +"+people[p.lap].name }
557 if tier(id) < 3 { line += " "+tierNames[tier(id)] }
558 if b.checked { line += " CHECKED" }
559 if p.volunteer { line += " VOL" }
560 out = append(out, line)
561 }
562 case "LOADFACTOR":
563 out = append(out, "LOADFACTOR "+flight.name)
564 allSeated, allSeats := 0, 0
565 for c := 0; c < 3; c++ {
566 if plane.cabins[c].from == 0 { continue }
567 seats, seated := 0, 0
568 for i, s := range plane.seats {
569 if s.cabin != c { continue }
570 if !flight.blocked[i] { seats++ }
571 if flight.occupants[i] >= 0 { seated++ }
572 }
573 factor := seated*1000/seats
574 out = append(out, fmt.Sprintf("%c %d/%d %d.%d%% booked %d/%d", classes[c], seated, seats, factor/10, factor%10, bookedCount(f, c), flight.limit[c]))
575 allSeated += seated
576 allSeats += seats
577 }
578 factor := allSeated*1000/allSeats
579 out = append(out, fmt.Sprintf("ALL %d/%d %d.%d%%", allSeated, allSeats, factor/10, factor%10))
580 case "BALANCE":
581 weight, moment := 0, 0
582 for _, id := range flight.occupants {
583 if id < 0 { continue }
584 value := 80
585 if people[id].kind == 1 { value = 40 }
586 if hasInfant(id) { value += 10 }
587 weight += value
588 moment += value*(plane.seats[people[id].seat].row-plane.base)
589 }
590 verdict := "OK"
591 if moment < -plane.limit { verdict = "NOSE-HEAVY" }
592 if moment > plane.limit { verdict = "TAIL-HEAVY" }
593 out = append(out, fmt.Sprintf("BALANCE %s weight %d moment %d limit %d %s", flight.name, weight, moment, plane.limit, verdict))
594 case "BOARDING":
595 type item struct {
596 id, group, zone, kind, row int
597 letter byte
598 }
599 var order []item
600 for _, id := range flight.occupants {
601 if id < 0 { continue }
602 p := people[id]
603 s := plane.seats[p.seat]
604 group := 3
605 if bookingInfants(p.booking) { group = 0 } else if p.cabin < 2 { group = 1 } else if tier(id) < 2 { group = 2 }
606 zone := 0
607 if group == 3 { zone = (s.row-plane.cabins[2].from)/5+1 }
608 order = append(order, item{id, group, zone, s.kind, s.row, s.letter})
609 }
610 sort.Slice(order, func(i, j int) bool {
611 x, y := order[i], order[j]
612 if x.group != y.group { return x.group < y.group }
613 if x.group == 3 {
614 if x.zone != y.zone { return x.zone > y.zone }
615 if x.kind != y.kind { return x.kind < y.kind }
616 if x.row != y.row { return x.row > y.row }
617 } else if x.row != y.row { return x.row < y.row }
618 return x.letter < y.letter
619 })
620 out = append(out, fmt.Sprintf("BOARDING %s %d", flight.name, len(order)))
621 for i, item := range order {
622 p := people[item.id]
623 group := ""
624 if item.group == 3 { group = fmt.Sprintf("ZONE%d", item.zone) } else { group = []string{"PRE", "CABIN", "TIER"}[item.group] }
625 out = append(out, fmt.Sprintf("%d %s %s %s %s", i+1, group, seatName(plane, p.seat), bookings[p.booking].pnr, p.name))
626 }
627 case "AVAIL":
628 c := cabinID(v[2])
629 if plane.cabins[c].from == 0 { fail("no cabin "+v[2]+" on "+flight.name) }
630 free, exits := 0, 0
631 kinds := [3]int{}
632 for i, s := range plane.seats {
633 if s.cabin != c || flight.blocked[i] || flight.occupants[i] >= 0 { continue }
634 free++
635 kinds[s.kind]++
636 if plane.exits[s.row] { exits++ }
637 }
638 out = append(out, fmt.Sprintf("AVAIL %s %c free %d W %d M %d A %d EXIT %d", flight.name, classes[c], free, kinds[0], kinds[1], kinds[2], exits))
639 }
640 }
641 return out
642}
643func commandResult(v []string) (out []string) {
644 defer func() {
645 if message := recover(); message != nil { out = []string{"ERROR "+message.(string)} }
646 }()
647 return execute(v)
648}
649func main() {
650 scanner := bufio.NewScanner(os.Stdin)
651 scanner.Scan()
652 count := number(scanner.Text())
653 output := bufio.NewWriter(os.Stdout)
654 defer output.Flush()
655 for i := 0; i < count && scanner.Scan(); i++ {
656 for _, line := range commandResult(strings.Split(scanner.Text(), " ")) { fmt.Fprintln(output, line) }
657 }
658}
