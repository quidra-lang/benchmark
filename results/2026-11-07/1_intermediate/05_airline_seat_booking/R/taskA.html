51allowed <- function(f, p, seat, class, vacated = integer(), assignments = character()) {
52 s <- f$model$seats[[seat]]
53 if (is.null(s) || s$class != class || seat %in% f$blocked) return(FALSE)
54 occupied <- f$occupied[[seat]]
55 if (!is.null(occupied) && !(occupied %in% vacated)) return(FALSE)
56 if ((p$type == "CHILD" || p$infant != 0L) && s$row %in% f$model$exits) return(FALSE)
57 if (p$infant != 0L) {
58 for (other_seat in ls(f$occupied, all.names = TRUE)) {
59 id <- f$occupied[[other_seat]]
60 if (id == p$id || id %in% vacated || person(id)$infant == 0L) next
61 t <- f$model$seats[[other_seat]]
62 if (t$row == s$row && t$block == s$block) return(FALSE)
63 }
64 for (key in names(assignments)) {
65 id <- as.integer(key)
66 if (id != p$id && person(id)$infant != 0L) {
67 t <- f$model$seats[[assignments[[key]]]]
68 if (t$row == s$row && t$block == s$block) return(FALSE)
69 }
70 }
71 }
72 TRUE

103fits <- function(f, ids, seats, class) {

107group_seats <- function(f, ids, class) {
108 c <- f$model$cabins[[class]]
109 total <- length(ids)
110 for (method in c("BLOCK", "ROW")) {
111 for (row in ints(c$from, c$to)) {
112 blocks <- if (method == "BLOCK") c$blocks else paste(c$blocks, collapse = "")
113 for (block in blocks) {
114 letters <- strsplit(block, "", fixed = TRUE)[[1L]]
115 for (x in ints(1L, length(letters) - total + 1L)) {
116 seats <- paste0(row, letters[x + seq_len(total) - 1L])
117 if (fits(f, ids, seats, class)) return(list(method = method, seats = seats))
118 }
119 }
120 }
121 }
122 for (height in ints(2L, total)) {
123 width <- (total + height - 1L) %/% height
124 rows <- (total + width - 1L) %/% width
125 for (row in ints(c$from, c$to - rows + 1L)) {
126 for (block in c$blocks) {
127 letters <- strsplit(block, "", fixed = TRUE)[[1L]]
128 for (x in ints(1L, length(letters) - width + 1L)) {
129 j <- seq_len(total) - 1L
130 seats <- paste0(row + j %/% width, letters[x + j %% width])
131 if (fits(f, ids, seats, class)) return(list(method = "STACK", seats = seats))
132 }
133 }
134 }
135 }
136 list(method = "SPLIT", seats = character())

531 } else if (p$class != "Y") {

534 } else if (tier_order[[tier(p)]] <= 1L) {
535 group <- "TIER"

538 zone <- (s$row - f$model$cabins[["Y"]]$from) %/% 5L + 1L

540 key <- c(3L, -zone, match(s$kind, c("W", "M", "A")), -s$row, letter)
541 }

543 keys[index, ] <- key
544 }

552 }
553 } else if (command == "LOADFACTOR") {

560 total_seated <- total_seated + seated
561 total_seats <- total_seats + seats
562 }
563 emit(paste("ALL", paste0(total_seated, "/", total_seats), load_text(total_seated, total_seats)))
564 } else if (command == "BALANCE") {
565 weight <- moment <- 0L
566 for (id in passengers(f)) {
567 p <- person(id)
568 if (p$seat != "") {
569 w <- if (p$type == "ADULT") 80L else 40L
570 if (p$infant != 0L) w <- w + 10L
571 weight <- weight + w
572 moment <- moment + w * (f$model$seats[[p$seat]]$row - f$model$base)
573 }
574 }
575 limit <- f$model$limit
576 if (moment < -limit) {
577 verdict <- "NOSE-HEAVY"
578 } else if (moment > limit) {
579 verdict <- "TAIL-HEAVY"
580 } else {
581 verdict <- "OK"
582 }
583 emit(paste("BALANCE", f$id, "weight", weight, "moment", moment, "limit", limit, verdict))
584 } else if (command == "AVAIL") {
585 class <- t[3L]

588 free <- 0L

592 free <- free + 1L
