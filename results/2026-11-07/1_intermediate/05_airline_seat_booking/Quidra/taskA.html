175string assign_group(State &state, int flight, int cabin, int[] people)
176 count := int(#people)
177 if count == 1
178 place(&state, people[0], choose(state, flight, people[0], cabin))
179 return "SINGLE"
180 plane := state.aircraft[state.flights[flight].aircraft]
181 section := plane.cabins[cabin_id(plane, cabin)]
182 for: 0 <= stage < 3
183 first_height := if stage == 2 then 2 else 1
184 last_height := if stage == 2 then count else 1
185 for: first_height <= height <= last_height
186 width := if stage == 2 then (count + height - 1) / height else count
187 for: section.first <= row <= section.last
188 block_count := if stage == 1 then 1 else int(#section.blocks)
189 for: 0 <= block < block_count
190 seats := row_seats(plane, row, if stage == 1 then -1 else block)
191 for: 0 <= start <= int(#seats) - width
192 int[] candidate := []
193 if stage < 2
194 candidate = seats[start:<start + count]
195 else
196 for: 0 <= k < count
197 next_row := row + k / width
198 next_seats := row_seats(plane, next_row, block)
199 if next_row > section.last or int(#next_seats) < start + width
200 candidate = []
201 break
202 candidate = candidate.append(next_seats[start + k % width])
203 if #candidate == #people and fits(state, flight, people, candidate)
204 for: 0 <= k < count
205 place(&state, people[k], candidate[k])
206 return ["BLOCK", "ROW", "STACK"][stage]
207 for person := people[]
208 place(&state, person, choose(state, flight, person, cabin))
209 return "SPLIT"
