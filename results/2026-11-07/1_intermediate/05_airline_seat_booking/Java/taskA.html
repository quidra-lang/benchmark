88 static boolean sameBlock(Seat a, Seat b) { return a.row() == b.row() && a.block() == b.block(); }
89 static boolean exitAllowed(Passenger p, Seat s) {

92 static boolean infantAllowed(Passenger p, Seat s, Set<Passenger> vacated, Map<Passenger,Seat> proposed) {
93 if (p.lap == null) return true;
94 Flight f = p.booking.flight;
95 String letters = f.aircraft.cabins.get(s.cabin()).blocks.get(s.block());
96 for (char letter : letters.toCharArray()) {
97 Passenger other = f.occupied.get(s.row()+String.valueOf(letter));
98 if (other != null && other != p && other.lap != null && !vacated.contains(other)) return false;
99 }
100 for (var entry : proposed.entrySet()) {
101 if (entry.getKey() != p && entry.getKey().lap != null && sameBlock(entry.getValue(),s)) return false;
102 }
103 return true;
104 }
105 static boolean fits(Passenger p, Seat s, Set<Passenger> vacated, Map<Passenger,Seat> proposed) {
106 Flight f = p.booking.flight;
107 Passenger occupant = f.occupied.get(s.id());
108 return !f.blocked.contains(s.id()) && (occupant == null || vacated.contains(occupant)) &&
109 exitAllowed(p,s) && infantAllowed(p,s,vacated,proposed);
110 }
111 static void seat(Passenger p, Seat target) {
112 Flight f = p.booking.flight;
113 if (p.seat != null) f.occupied.remove(p.seat.id());
114 p.seat = target;
115 if (target != null) f.occupied.put(target.id(),p);
116 }
117 static Seat select(Passenger p, String cabin) {
118 List<Seat> candidates = p.booking.flight.aircraft.cabins.get(cabin).seats.stream()
119 .filter(s -> fits(p,s,Set.of(),Map.of())).toList();
120 return candidates.stream().filter(s -> s.kind().equals(p.pref)).findFirst().orElse(candidates.isEmpty() ? null : candidates.getFirst());
121 }

123 for (Passenger p : passengers(f)) {

126 if (target != null) {
127 seat(p,target);
128 out.add("REASSIGNED "+p.booking.pnr+" "+p.name+" "+target.id());
129 }
130 }

133 static boolean candidate(List<Passenger> people, List<Seat> positions) {
134 Map<Passenger,Seat> proposed = new HashMap<>();
135 for (int i = 0; i < people.size(); i++) proposed.put(people.get(i),positions.get(i));
136 if (!people.stream().allMatch(p -> fits(p,proposed.get(p),Set.of(),proposed))) return false;
137 for (Passenger p : people) seat(p,proposed.get(p));
138 return true;
139 }
140 static String assign(Booking b, String cabinName) {
141 List<Passenger> people = b.passengers.stream().filter(Passenger::needsSeat).toList();
142 if (people.size() == 1) {
143 seat(people.getFirst(),select(people.getFirst(),cabinName));
144 return "SINGLE";
145 }
146 Cabin cabin = b.flight.aircraft.cabins.get(cabinName);
147 Aircraft a = b.flight.aircraft;
148 int k = people.size();
149 for (int stage = 1; stage <= 3; stage++) {
150 int lastH = stage == 3 ? k : 2;
151 for (int h = 2; h <= lastH; h++) {
152 int width = (k+h-1)/h;
153 for (int row = cabin.first; row <= cabin.last; row++) {
154 List<String> blocks = stage == 2 ? List.of(String.join("",cabin.blocks)) : cabin.blocks;
155 for (String block : blocks) {
156 int span = stage == 3 ? width : k;
157 for (int x = 0; x+span <= block.length(); x++) {
158 List<Seat> positions = new ArrayList<>();
159 for (int j = 0; j < k; j++) {
160 int r = row+(stage == 3 ? j/width : 0);
161 if (r > cabin.last) break;
162 positions.add(a.seats.get(r+String.valueOf(block.charAt(x+(stage == 3 ? j%width : j)))));
163 }
164 if (positions.size() == k && candidate(people,positions)) return stage == 1 ? "BLOCK" : stage == 2 ? "ROW" : "STACK";
165 }
166 }
167 }
168 }
169 }
170 for (Passenger p : people) seat(p,select(p,cabinName));
171 return "SPLIT";
172 }

182 }

198 List<String> execute(String line) {
199 String[] t = line.split(" ");
200 String command = t[0];
201 List<String> out = new ArrayList<>();
202 switch (command) {
203 case "AIRCRAFT": {
204 defining = new Aircraft();
205 defining.model = t[1];
206 defining.base = Integer.parseInt(t[2]);
207 defining.limit = Integer.parseInt(t[3]);
208 break;
209 }
210 case "CABIN": {
211 Cabin c = new Cabin();
212 c.name = t[1];
213 c.first = Integer.parseInt(t[2]);
214 c.last = Integer.parseInt(t[3]);
215 c.blocks = Arrays.asList(t).subList(4,t.length);
216 for (int row = c.first; row <= c.last; row++) {
217 for (int block = 0; block < c.blocks.size(); block++) {
218 String letters = c.blocks.get(block);
219 for (int i = 0; i < letters.length(); i++) {
220 boolean window = block == 0 && i == 0 || block == c.blocks.size()-1 && i == letters.length()-1;
221 String kind = window ? "W" : i == 0 || i == letters.length()-1 ? "A" : "M";
222 Seat s = new Seat(row+String.valueOf(letters.charAt(i)),row,letters.charAt(i),block,c.name,kind);
223 c.seats.add(s);
224 defining.seats.put(s.id(),s);
225 }
226 }
227 }
228 defining.cabins.put(c.name,c);
229 break;
230 }
231 case "EXIT": {
232 for (int i = 1; i < t.length; i++) defining.exits.add(Integer.parseInt(t[i]));
233 break;
234 }
235 case "BLOCKED": {
236 defining.blocked.addAll(Arrays.asList(t).subList(1,t.length));
237 break;
238 }
239 case "END": {
240 aircraft.put(defining.model,defining);
241 String result = "AIRCRAFT "+defining.model;
242 for (String c : CLASSES) {
243 if (defining.cabins.containsKey(c)) result += " "+c+" "+seats(defining,c,defining.blocked);
244 }
245 out.add(result);
246 defining = null;
247 break;
248 }
249 case "FLIGHT": {
250 if (flights.containsKey(t[1])) fail("duplicate flight "+t[1]);
251 Aircraft a = aircraft.get(t[2]);
252 if (a == null) fail("unknown aircraft "+t[2]);
253 Flight f = new Flight();
254 f.name = t[1];
255 f.aircraft = a;
256 f.blocked = new HashSet<>(a.blocked);
257 String result = "FLIGHT "+f.name+" "+a.model;
258 for (String c : CLASSES) {
259 if (a.cabins.containsKey(c)) {
260 int count = seats(a,c,a.blocked), limit = count*(100+Integer.parseInt(t[3]))/100;
261 f.limits.put(c,limit);
262 result += " "+c+" "+count+"/"+limit;
263 }
264 }
265 flights.put(f.name,f);
266 out.add(result);
267 break;
268 }
269 case "BOOK": {
270 if (used.contains(t[1])) fail("duplicate PNR "+t[1]);
271 Flight f = flight(t[2],true);
272 requireCabin(f,t[3]);
273 Booking b = new Booking();
274 b.pnr = t[1];
275 b.flight = f;
276 Set<String> seen = new HashSet<>();
277 List<Passenger> adults = new ArrayList<>(), infants = new ArrayList<>();
278 for (int i = 4; i < t.length; i++) {
279 String[] parts = t[i].split(":");
280 if (!seen.add(parts[0])) fail("duplicate name "+parts[0]+" in "+b.pnr);
281 Passenger p = new Passenger();
282 p.name = parts[0];
283 p.type = parts[1];
284 p.pref = parts.length == 3 ? parts[2] : null;
285 p.cabin = t[3];
286 p.booking = b;
287 b.passengers.add(p);
288 if (p.type.equals("ADULT")) adults.add(p);
289 if (p.type.equals("INFANT")) infants.add(p);
290 }
291 Set<String> present = passengers(f).stream().map(p -> p.name).collect(Collectors.toSet());
292 for (Passenger p : b.passengers) {
293 if (present.contains(p.name)) fail(p.name+" already on "+f.name);
294 }
295 if (adults.isEmpty()) fail("no adult in "+b.pnr);
296 if (infants.size() > adults.size()) fail("too many infants in "+b.pnr);
297 int count = (int)b.passengers.stream().filter(Passenger::needsSeat).count();
298 if (booked(f,t[3])+count > f.limits.get(t[3])) fail("limit exceeded for "+f.name+" "+t[3]);
299 for (int i = 0; i < infants.size(); i++) {
300 adults.get(i).lap = infants.get(i);
301 infants.get(i).parent = adults.get(i);
302 }
303 for (Passenger p : b.passengers) p.order = nextOrder++;
304 String method = assign(b,t[3]);
305 f.bookings.add(b);
306 bookings.put(b.pnr,b);
307 used.add(b.pnr);
308 List<String> results = b.passengers.stream().map(p -> p.name+":"+p.result()).toList();
309 out.add("BOOKED "+b.pnr+" "+f.name+" "+t[3]+" "+method+" "+String.join(" ",results));
310 break;
311 }
312 case "SEAT": {
313 Booking b = booking(t[1],true);
314 Passenger p = passenger(b,t[2]);
315 if (!p.needsSeat()) fail("infant "+p.name+" has no seat");
316 Seat target = b.flight.aircraft.seats.get(t[3]);
317 if (target == null || !target.cabin().equals(p.cabin)) fail("no seat "+t[3]+" in cabin "+p.cabin);
318 if (b.flight.blocked.contains(t[3])) fail("seat "+t[3]+" is blocked");
319 if (b.flight.occupied.containsKey(t[3])) fail("seat "+t[3]+" is occupied");
320 if (!exitAllowed(p,target)) fail("exit row "+target.row()+" not allowed for "+p.name);
321 if (!infantAllowed(p,target,Set.of(),Map.of())) fail("infant limit reached at "+t[3]);
322 String old = p.result();
323 seat(p,target);
324 out.add("MOVED "+b.pnr+" "+p.name+" "+old+" -> "+t[3]);
325 reassign(b.flight,out);
326 break;
327 }
328 case "SWAP": {
329 Booking a = booking(t[1],true);
330 Passenger p = passenger(a,t[2]);
331 Booking b = booking(t[3],true);
332 Passenger q = passenger(b,t[4]);
333 if (p == q) fail("cannot swap "+p.name+" with itself");
334 for (Passenger x : List.of(p,q)) {
335 if (!x.needsSeat()) fail("infant "+x.name+" has no seat");
336 }
337 if (a.flight != b.flight) fail(a.pnr+" and "+b.pnr+" are on different flights");
338 if (!p.cabin.equals(q.cabin)) fail(p.name+" and "+q.name+" are in different cabins");
339 for (Passenger x : List.of(p,q)) {
340 if (x.seat == null) fail(x.name+" has no seat");
341 }
342 Set<Passenger> vacated = Set.of(p,q);
343 Map<Passenger,Seat> proposed = Map.of(p,q.seat,q,p.seat);
344 if (!fits(p,q.seat,vacated,proposed) || !fits(q,p.seat,vacated,proposed)) fail("swap of "+p.name+" and "+q.name+" breaks seating rules");
345 Seat ps = p.seat, qs = q.seat;
346 seat(p,null);
347 seat(q,null);
348 seat(p,qs);
349 seat(q,ps);
350 out.add("SWAPPED "+a.pnr+" "+p.name+" "+p.result()+" "+b.pnr+" "+q.name+" "+q.result());
351 reassign(a.flight,out);
352 break;
353 }
354 case "CANCEL": {
355 Booking b = booking(t[1],true);
356 List<Passenger> targets;
357 if (t.length == 2) targets = new ArrayList<>(b.passengers);
358 else {
359 Passenger p = passenger(b,t[2]);
360 List<Passenger> selected = removalTargets(p);
361 targets = b.passengers.stream().filter(selected::contains).toList();
362 List<Passenger> remaining = b.passengers.stream().filter(q -> !selected.contains(q)).toList();
363 if (!remaining.isEmpty() && remaining.stream().noneMatch(q -> q.type.equals("ADULT"))) fail(b.pnr+" would have no adult");
364 }
365 String names = String.join(" ",targets.stream().map(p -> p.name).toList());
366 remove(targets);
367 out.add("CANCELLED "+b.pnr+" "+names);
368 reassign(b.flight,out);
369 break;
370 }
371 case "STATUS": {
372 tiers.put(t[1],t[2]);
373 out.add("STATUS "+t[1]+" "+t[2]);
374 break;
375 }
376 case "CHECKIN": {
377 Booking b = booking(t[1],true);
378 if (b.checked) fail(b.pnr+" already checked in");
379 b.checked = true;
380 out.add("CHECKIN "+b.pnr+" "+b.passengers.size());
381 break;
382 }
383 case "VOLUNTEER": {
384 Booking b = booking(t[1],true);
385 Passenger p = passenger(b,t[2]);
386 if (!p.type.equals("ADULT") || p.lap != null) fail(p.name+" cannot volunteer");
387 if (p.seat == null) fail(p.name+" has no seat");
388 if (b.flight.volunteers.contains(p)) fail(p.name+" already volunteered");
389 b.flight.volunteers.add(p);
390 out.add("VOLUNTEER "+b.pnr+" "+p.name+" "+p.result());
391 break;
392 }
393 case "UPGRADE": {
394 Flight f = flight(t[1],true);
395 List<String> classes = CLASSES.stream().filter(f.aircraft.cabins::containsKey).toList();
396 int count = 0;
397 for (int i = 0; i+1 < classes.size(); i++) {
398 String higher = classes.get(i), lower = classes.get(i+1);
399 List<Passenger> targets = passengers(f).stream().filter(p -> p.cabin.equals(lower) && p.type.equals("ADULT") && p.lap == null && !tier(p).equals("NONE"))
400 .sorted(Comparator.comparingInt(this::tierRank).thenComparingInt(p -> p.order)).toList();
401 for (Passenger p : targets) {
402 if (booked(f,higher) >= f.limits.get(higher)) break;
403 Seat target = select(p,higher);
404 if (target == null) break;
405 String old = p.result();
406 p.cabin = higher;
407 seat(p,target);
408 out.add("UPGRADED "+p.booking.pnr+" "+p.name+" "+lower+" "+old+" -> "+higher+" "+target.id());
409 count++;
410 }
411 }
412 reassign(f,out);
413 out.add("UPGRADE "+f.name+" "+count);
414 break;
415 }
416 case "CLOSE": {
417 Flight f = flight(t[1],true);
418 for (Booking b : f.bookings) {
419 if (!b.passengers.isEmpty() && !b.checked) {
420 out.add("NOSHOW "+b.pnr+" "+b.passengers.size());
421 remove(new ArrayList<>(b.passengers));
422 }
423 }
424 reassign(f,out);
425 int count = 0, compensation = 0;
426 for (String c : CLASSES) {
427 if (!f.aircraft.cabins.containsKey(c)) continue;
428 List<Passenger> volunteers = f.volunteers.stream().filter(p -> p.seat != null && p.cabin.equals(c)).toList();
429 for (Passenger p : volunteers) {
430 boolean useful = passengers(f).stream().anyMatch(q -> q.needsSeat() && q.seat == null && q.cabin.equals(c) && fits(q,p.seat,Set.of(p),Map.of()));
431 if (useful) {
432 int value = c.equals("F") ? 500 : c.equals("J") ? 300 : 150;
433 out.add("DENIED VOLUNTARY "+p.booking.pnr+" "+p.name+" "+c+" "+value);
434 count++;
435 compensation += value;
436 remove(removalTargets(p));
437 reassign(f,out);
438 }
439 }
440 }
441 for (Passenger p : passengers(f)) {
442 if (p.needsSeat() && p.seat == null) {
443 int value = p.cabin.equals("F") ? 1000 : p.cabin.equals("J") ? 600 : 300;
444 if (p.type.equals("CHILD")) value /= 2;
445 out.add("DENIED INVOLUNTARY "+p.booking.pnr+" "+p.name+" "+p.cabin+" "+value);
446 count++;
447 compensation += value;
448 remove(removalTargets(p));
449 }
450 }
451 f.closed = true;
452 out.add("CLOSED "+f.name+" denied "+count+" compensation "+compensation);
453 break;
454 }

456 Flight f = flight(t[1],true);
457 Seat target = f.aircraft.seats.get(t[2]);
458 if (target == null) fail("no seat "+t[2]+" on "+f.name);

471 } else f.blocked.remove(t[2]);

473 break;
474 }

481 out.add("CABIN "+name);

484 List<String> blocks = new ArrayList<>();
485 for (String block : c.blocks) {
486 StringBuilder result = new StringBuilder();

492 }
493 out.add(String.format(Locale.ROOT,"%2d %s",row,String.join(" ",blocks)));

495 }
496 break;
497 }

499 Flight f = flight(t[1],false);
500 List<Passenger> all = passengers(f);
501 List<Passenger> seated = all.stream().filter(p -> p.seat != null).sorted(seatOrder()).toList();
502 List<Passenger> unassigned = all.stream().filter(p -> p.needsSeat() && p.seat == null).toList();
503 out.add("MANIFEST "+f.name+" seated "+seated.size()+" unassigned "+unassigned.size()+" infants "+all.stream().filter(p -> !p.needsSeat()).count());
504 List<Passenger> ordered = new ArrayList<>(seated);
505 ordered.addAll(unassigned);
506 for (Passenger p : ordered) {
507 String result = (p.seat == null ? "---" : p.seat.id())+" "+p.cabin+" "+p.booking.pnr+" "+p.name+" "+p.type;
508 if (p.lap != null) result += " +"+p.lap.name;
509 if (!tier(p).equals("NONE")) result += " "+tier(p);
510 if (p.booking.checked) result += " CHECKED";
511 if (f.volunteers.contains(p)) result += " VOL";
512 out.add(result);
513 }
514 break;
515 }
516 case "LOADFACTOR": {
517 Flight f = flight(t[1],false);
518 out.add("LOADFACTOR "+f.name);
519 int totalSeated = 0, totalSeats = 0;
520 for (String c : CLASSES) {
521 if (!f.aircraft.cabins.containsKey(c)) continue;
522 int count = seats(f.aircraft,c,f.blocked);
523 int seated = (int)passengers(f).stream().filter(p -> p.cabin.equals(c) && p.seat != null).count();
524 out.add(c+" "+seated+"/"+count+" "+load(seated,count)+" booked "+booked(f,c)+"/"+f.limits.get(c));

526 totalSeats += count;
527 }
528 out.add("ALL "+totalSeated+"/"+totalSeats+" "+load(totalSeated,totalSeats));

531 case "BALANCE": {
532 Flight f = flight(t[1],false);
533 int weight = 0, moment = 0;
534 for (Passenger p : passengers(f)) {
535 if (p.seat == null) continue;
536 int w = (p.type.equals("ADULT") ? 80 : 40)+(p.lap == null ? 0 : 10);
537 weight += w;
538 moment += w*(p.seat.row()-f.aircraft.base);
539 }
540 int limit = f.aircraft.limit;
541 out.add("BALANCE "+f.name+" weight "+weight+" moment "+moment+" limit "+limit+" "+(moment < -limit ? "NOSE-HEAVY" : moment > limit ? "TAIL-HEAVY" : "OK"));
542 break;
543 }
544 case "BOARDING": {
545 Flight f = flight(t[1],false);
546 List<Passenger> ordered = new ArrayList<>(passengers(f).stream().filter(p -> p.seat != null).toList());
547 ordered.sort((a,b) -> {
548 int ar = boardingRank(a), br = boardingRank(b);
549 if (ar != br) return Integer.compare(ar,br);
550 if (ar < 3) return seatOrder().compare(a,b);
551 if (zone(a) != zone(b)) return Integer.compare(zone(b),zone(a));
552 int ak = "WMA".indexOf(a.seat.kind()), bk = "WMA".indexOf(b.seat.kind());
553 if (ak != bk) return Integer.compare(ak,bk);
554 if (a.seat.row() != b.seat.row()) return Integer.compare(b.seat.row(),a.seat.row());
555 return Character.compare(a.seat.letter(),b.seat.letter());
556 });
557 out.add("BOARDING "+f.name+" "+ordered.size());
558 for (int i = 0; i < ordered.size(); i++) {
559 Passenger p = ordered.get(i);
560 int rank = boardingRank(p);
561 String group = rank == 0 ? "PRE" : rank == 1 ? "CABIN" : rank == 2 ? "TIER" : "ZONE"+zone(p);
562 out.add((i+1)+" "+group+" "+p.result()+" "+p.booking.pnr+" "+p.name);
563 }
564 break;
565 }
566 case "AVAIL": {
567 Flight f = flight(t[1],false);
568 requireCabin(f,t[2]);
569 List<Seat> free = f.aircraft.cabins.get(t[2]).seats.stream()
570 .filter(s -> !f.blocked.contains(s.id()) && !f.occupied.containsKey(s.id())).toList();
571 out.add("AVAIL "+f.name+" "+t[2]+" free "+free.size()+
572 " W "+free.stream().filter(s -> s.kind().equals("W")).count()+
573 " M "+free.stream().filter(s -> s.kind().equals("M")).count()+
574 " A "+free.stream().filter(s -> s.kind().equals("A")).count()+
575 " EXIT "+free.stream().filter(s -> f.aircraft.exits.contains(s.row())).count());
576 break;
577 }
578 case "INFO": {
579 Booking b = booking(t[1],false);
580 out.add("INFO "+b.pnr+" "+b.flight.name+" "+b.passengers.size()+" "+(b.checked ? "CHECKED" : "NOT-CHECKED"));
581 for (Passenger p : b.passengers) out.add(p.name+" "+p.type+" "+p.cabin+" "+p.result());
582 break;
583 }
584 case "FIND": {
585 List<String> found = new ArrayList<>();
586 for (Flight f : flights.values()) {
587 for (Passenger p : passengers(f)) {
588 if (p.name.equals(t[1])) found.add(f.name+" "+p.booking.pnr+" "+p.cabin+" "+p.result()+" "+(f.closed ? "CLOSED" : "OPEN"));
589 }
590 }
591 out.add("FIND "+t[1]+" "+tiers.getOrDefault(t[1],"NONE")+" "+found.size());
592 out.addAll(found);
593 break;
594 }
595 }
596 return out;
597 }
