71 def fits?(flight, people, seats, desired = nil, vacated = [])
72 ignored = people + vacated
73 existing = passengers(flight).select { |person| person[:seat] && !ignored.include?(person) }
74 occupied = existing.map { |person| person[:seat] }.to_set
75 infants = Set.new
76 existing.select { |person| person[:infant] }.each do |person|
77 seat = flight[:model][:seats][person[:seat]]
78 infants << [seat[:row], seat[:block]]
79 end
80 people.each_with_index do |person, i|
81 seat = flight[:model][:seats][seats[i]]
82 return false if !seat || seat[:cabin] != (desired || person[:cabin])
83 return false if flight[:blocked].include?(seat[:id]) || occupied.include?(seat[:id])
84 if (person[:type] == "CHILD" || person[:infant]) && flight[:model][:exits].include?(seat[:row])
85 return false
86 end
87 key = [seat[:row], seat[:block]]
88 return false if person[:infant] && infants.include?(key)
89 infants << key if person[:infant]
90 occupied << seat[:id]
91 end
92 true
93 end

95 def choose(flight, person, desired = person[:cabin])
96 candidates = cabin(flight, desired)[:seats].select { |seat| fits?(flight, [person], [seat], desired) }
97 candidates.find { |seat| flight[:model][:seats][seat][:kind] == person[:pref] } || candidates[0]
98 end

100 def reassign(flight)
101 output = []
102 passengers(flight).each do |person|
103 next if person[:type] == "INFANT" || person[:seat]
104 seat = choose(flight, person)
105 if seat
106 person[:seat] = seat
107 output << "REASSIGNED #{person[:pnr]} #{person[:name]} #{seat}"
108 end
109 end
110 output
111 end

113 def assign(flight, people, cabin_name)
114 if people.length == 1
115 people[0][:seat] = choose(flight, people[0])
116 return "SINGLE"
117 end
118 cabin = cabin(flight, cabin_name)
119 k = people.length
120 adopt = lambda do |seats|
121 next false unless fits?(flight, people, seats)
122 people.each_with_index { |person, i| person[:seat] = seats[i] }
123 true
124 end
125 %w[BLOCK ROW].each do |stage|
126 (cabin[:start]..cabin[:end]).each do |row|
127 blocks = stage == "BLOCK" ? cabin[:blocks] : [cabin[:blocks].join]
128 blocks.each do |block|
129 (0..block.length - k).each do |start|
130 seats = block[start, k].chars.map { |letter| "#{row}#{letter}" }
131 return stage if adopt.call(seats)
132 end
133 end
134 end
135 end
136 (2..k).each do |h|
137 width = (k + h - 1).div(h)
138 height = (k + width - 1).div(width)
139 (cabin[:start]..cabin[:end] - height + 1).each do |row|
140 cabin[:blocks].each do |block|
141 (0..block.length - width).each do |start|
142 seats = Array.new(k) { |j| "#{row + j.div(width)}#{block[start + j % width]}" }
143 return "STACK" if adopt.call(seats)
144 end
145 end
146 end
147 end
148 people.each { |person| person[:seat] = choose(flight, person) }
149 "SPLIT"
150 end

168 end

170 def define(name, args)
171 if name == "AIRCRAFT"
172 @defining = {
173 name: args[0], base: args[1].to_i, limit: args[2].to_i,
174 cabins: {}, seats: {}, exits: Set.new, blocked: Set.new
175 }
176 return []
177 end
178 model = @defining
179 case name
180 when "CABIN"
181 blocks = args.drop(3)
182 start = args[1].to_i
183 ending = args[2].to_i
184 seats = []
185 (start..ending).each do |row|
186 blocks.each_with_index do |block, block_index|
187 block.chars.each_with_index do |letter, i|
188 window = (block_index.zero? && i.zero?) || (block_index == blocks.length - 1 && i == block.length - 1)
189 kind = window ? "W" : (i.zero? || i == block.length - 1 ? "A" : "M")
190 id = "#{row}#{letter}"
191 seats << id
192 model[:seats][id] = { id: id, row: row, letter: letter, block: block_index, kind: kind, cabin: args[0] }
193 end
194 end
195 end
196 model[:cabins][args[0]] = { start: start, end: ending, blocks: blocks, seats: seats }
197 when "EXIT"
198 model[:exits] = args.map(&:to_i).to_set
199 when "BLOCKED"
200 model[:blocked] = args.to_set
201 when "END"
202 @models[model[:name]] = model
203 @defining = nil
204 counts = CLASSES.select { |item| model[:cabins].key?(item) }.map do |item|
205 count = model[:cabins][item][:seats].count { |seat| !model[:blocked].include?(seat) }
206 "#{item} #{count}"
207 end
208 return ["AIRCRAFT #{model[:name]} " + counts.join(" ")]
209 end
210 []
211 end

213 def book(args)
214 pnr, flight_name, cabin_name = args[0, 3]
215 fail_with("duplicate PNR #{pnr}") if @bookings.key?(pnr)
216 flight = get_flight(flight_name, true)
217 cabin(flight, cabin_name)
218 people = args.drop(3).each_with_index.map do |token, i|
219 name, type, pref = token.split(":")
220 {
221 name: name, type: type, pref: pref, cabin: cabin_name, seat: nil,
222 infant: nil, adult: nil, pnr: pnr, order: @serial * 10 + i
223 }
224 end
225 seen = Set.new
226 people.each do |person|
227 fail_with("duplicate name #{person[:name]} in #{pnr}") if seen.include?(person[:name])
228 seen << person[:name]
229 end
230 existing = passengers(flight).map { |person| person[:name] }.to_set
231 people.each do |person|
232 fail_with("#{person[:name]} already on #{flight_name}") if existing.include?(person[:name])
233 end
234 adults = people.select { |person| person[:type] == "ADULT" }
235 infants = people.select { |person| person[:type] == "INFANT" }
236 fail_with("no adult in #{pnr}") if adults.empty?
237 fail_with("too many infants in #{pnr}") if infants.length > adults.length
238 seated = people.reject { |person| person[:type] == "INFANT" }
239 if booked(flight, cabin_name) + seated.length > flight[:limits][cabin_name]
240 fail_with("limit exceeded for #{flight_name} #{cabin_name}")
241 end
242 infants.each_with_index do |infant, i|
243 infant[:adult] = adults[i][:name]
244 adults[i][:infant] = infant[:name]
245 end
246 booking = { pnr: pnr, flight: flight_name, passengers: people, checked: false }
247 @serial += 1
248 @bookings[pnr] = booking
249 flight[:bookings] << booking
250 method = assign(flight, seated, cabin_name)
251 items = people.map { |person| "#{person[:name]}:#{display(person)}" }.join(" ")
252 ["BOOKED #{pnr} #{flight_name} #{cabin_name} #{method} #{items}"]
253 end

255 def close(flight)
256 output = []
257 flight[:bookings].each do |booking|
258 next if booking[:passengers].empty? || booking[:checked]
259 output << "NOSHOW #{booking[:pnr]} #{booking[:passengers].length}"
260 remove(flight, booking[:passengers].dup)
261 end
262 output.concat(reassign(flight))
263 count = 0
264 total = 0
265 compensation = { "F" => 500, "J" => 300, "Y" => 150 }
266 deny = lambda do |person, voluntary|
267 value = compensation[person[:cabin]] * (voluntary ? 1 : 2)
268 value = value.div(2) if person[:type] == "CHILD"
269 output << "DENIED #{voluntary ? 'VOLUNTARY' : 'INVOLUNTARY'} #{person[:pnr]} #{person[:name]} #{person[:cabin]} #{value}"
270 count += 1
271 total += value
272 remove(flight, targets(person))
273 end
274 CLASSES.select { |item| flight[:model][:cabins].key?(item) }.each do |item|
275 volunteers = flight[:volunteers].select { |person| person[:seat] && person[:cabin] == item }
276 volunteers.each do |volunteer|
277 candidates = passengers(flight).select do |person|
278 person[:type] != "INFANT" && !person[:seat] && person[:cabin] == item
279 end
280 if candidates.any? { |person| fits?(flight, [person], [volunteer[:seat]], nil, [volunteer]) }
281 deny.call(volunteer, true)
282 output.concat(reassign(flight))
283 end
284 end
285 end
286 unassigned = passengers(flight).select { |person| person[:type] != "INFANT" && !person[:seat] }
287 unassigned.each { |person| deny.call(person, false) }
288 flight[:closed] = true
289 output << "CLOSED #{flight[:name]} denied #{count} compensation #{total}"
290 output
291 end

293 def command(line)
294 name, *args = line.split(" ")
295 return define(name, args) if %w[AIRCRAFT CABIN EXIT BLOCKED END].include?(name)
296 if name == "FLIGHT"
297 fail_with("duplicate flight #{args[0]}") if @flights.key?(args[0])
298 model = @models[args[1]]
299 fail_with("unknown aircraft #{args[1]}") unless model
300 limits = {}
301 items = []
302 CLASSES.select { |item| model[:cabins].key?(item) }.each do |item|
303 count = model[:cabins][item][:seats].count { |seat| !model[:blocked].include?(seat) }
304 limit = (count * (100 + args[2].to_i)).div(100)
305 limits[item] = limit
306 items << "#{item} #{count}/#{limit}"
307 end
308 @flights[args[0]] = {
309 name: args[0], model: model, blocked: model[:blocked].dup, limits: limits,
310 closed: false, bookings: [], volunteers: []
311 }
312 return ["FLIGHT #{args[0]} #{model[:name]} " + items.join(" ")]
313 end
314 return book(args) if name == "BOOK"
315 if name == "STATUS"
316 @tiers[args[0]] = args[1]
317 return ["STATUS " + args.join(" ")]
318 end
319 if name == "FIND"
320 items = []
321 @flights.each_value do |flight|
322 person = passengers(flight).find { |passenger| passenger[:name] == args[0] }
323 if person
324 items << "#{flight[:name]} #{person[:pnr]} #{person[:cabin]} #{display(person)} #{flight[:closed] ? 'CLOSED' : 'OPEN'}"
325 end
326 end
327 return ["FIND #{args[0]} #{tier(args[0])} #{items.length}", *items]
328 end
329 if name == "INFO"
330 booking = get_booking(args[0])
331 items = booking[:passengers].map { |person| "#{person[:name]} #{person[:type]} #{person[:cabin]} #{display(person)}" }
332 return ["INFO #{booking[:pnr]} #{booking[:flight]} #{items.length} #{booking[:checked] ? 'CHECKED' : 'NOT-CHECKED'}", *items]
333 end
334 if %w[SEAT SWAP CANCEL CHECKIN VOLUNTEER].include?(name)
335 booking = get_booking(args[0], true)
336 flight = get_flight(booking[:flight])
337 if name == "CHECKIN"
338 fail_with("#{booking[:pnr]} already checked in") if booking[:checked]
339 booking[:checked] = true
340 return ["CHECKIN #{booking[:pnr]} #{booking[:passengers].length}"]
341 end
342 if name == "CANCEL"
343 removed = args.length == 1 ? booking[:passengers].dup : targets(get_passenger(args[0], args[1]))
344 remaining = booking[:passengers].reject { |person| removed.include?(person) }
345 if args.length == 2 && !remaining.empty? && remaining.none? { |person| person[:type] == "ADULT" }
346 fail_with("#{booking[:pnr]} would have no adult")
347 end
348 output = ["CANCELLED #{booking[:pnr]} " + removed.map { |person| person[:name] }.join(" ")]
349 remove(flight, removed)
350 return output + reassign(flight)
351 end
352 person = get_passenger(args[0], args[1])
353 if name == "SWAP"
354 other = get_passenger(args[2], args[3])
355 fail_with("cannot swap #{person[:name]} with itself") if person.equal?(other)
356 [person, other].each do |passenger|
357 fail_with("infant #{passenger[:name]} has no seat") if passenger[:type] == "INFANT"
358 end
359 if booking[:flight] != @bookings[other[:pnr]][:flight]
360 fail_with("#{args[0]} and #{args[2]} are on different flights")
361 end
362 if person[:cabin] != other[:cabin]
363 fail_with("#{person[:name]} and #{other[:name]} are in different cabins")
364 end
365 [person, other].each { |passenger| fail_with("#{passenger[:name]} has no seat") unless passenger[:seat] }
366 unless fits?(flight, [person, other], [other[:seat], person[:seat]])
367 fail_with("swap of #{person[:name]} and #{other[:name]} breaks seating rules")
368 end
369 person[:seat], other[:seat] = other[:seat], person[:seat]
370 return ["SWAPPED #{person[:pnr]} #{person[:name]} #{person[:seat]} #{other[:pnr]} #{other[:name]} #{other[:seat]}", *reassign(flight)]
371 end
372 if name == "VOLUNTEER"
373 fail_with("#{person[:name]} cannot volunteer") if person[:type] != "ADULT" || person[:infant]
374 fail_with("#{person[:name]} has no seat") unless person[:seat]
375 fail_with("#{person[:name]} already volunteered") if flight[:volunteers].include?(person)
376 flight[:volunteers] << person
377 return ["VOLUNTEER #{person[:pnr]} #{person[:name]} #{person[:seat]}"]
378 end
379 fail_with("infant #{person[:name]} has no seat") if person[:type] == "INFANT"
380 seat = flight[:model][:seats][args[2]]
381 fail_with("no seat #{args[2]} in cabin #{person[:cabin]}") if !seat || seat[:cabin] != person[:cabin]
382 fail_with("seat #{seat[:id]} is blocked") if flight[:blocked].include?(seat[:id])
383 fail_with("seat #{seat[:id]} is occupied") if passengers(flight).any? { |other| other[:seat] == seat[:id] }
384 if (person[:type] == "CHILD" || person[:infant]) && flight[:model][:exits].include?(seat[:row])
385 fail_with("exit row #{seat[:row]} not allowed for #{person[:name]}")
386 end
387 fail_with("infant limit reached at #{seat[:id]}") unless fits?(flight, [person], [seat[:id]])
388 old = person[:seat]
389 person[:seat] = seat[:id]
390 return ["MOVED #{person[:pnr]} #{person[:name]} #{old} -> #{seat[:id]}", *reassign(flight)]
391 end
392 modifying = %w[UPGRADE CLOSE BLOCK UNBLOCK].include?(name)
393 flight = get_flight(args[0], modifying)
394 return close(flight) if name == "CLOSE"
395 if name == "UPGRADE"
396 cabins = CLASSES.select { |item| flight[:model][:cabins].key?(item) }
397 output = []
398 count = 0
399 cabins.each_cons(2) do |higher, lower|
400 targets = passengers(flight).select do |person|
401 person[:cabin] == lower && person[:type] == "ADULT" && !person[:infant] && tier(person[:name]) != "NONE"
402 end.sort_by { |person| [TIERS.index(tier(person[:name])), person[:order]] }
403 targets.each do |person|
404 break if booked(flight, higher) >= flight[:limits][higher]
405 seat = choose(flight, person, higher)
406 break unless seat
407 old = person[:seat] || "UNASSIGNED"
408 person[:cabin] = higher
409 person[:seat] = seat
410 output << "UPGRADED #{person[:pnr]} #{person[:name]} #{lower} #{old} -> #{higher} #{seat}"
411 count += 1
412 end
413 end
414 return output + reassign(flight) + ["UPGRADE #{flight[:name]} #{count}"]
415 end
416 if name == "BLOCK" || name == "UNBLOCK"
417 seat = args[1]
418 fail_with("no seat #{seat} on #{flight[:name]}") unless flight[:model][:seats].key?(seat)
419 blocked = flight[:blocked].include?(seat)
420 fail_with("seat #{seat} is blocked") if name == "BLOCK" && blocked
421 fail_with("seat #{seat} is not blocked") if name == "UNBLOCK" && !blocked
422 output = ["#{name} #{flight[:name]} #{seat}"]
423 if name == "BLOCK"
424 flight[:blocked] << seat
425 person = passengers(flight).find { |passenger| passenger[:seat] == seat }
426 if person
427 person[:seat] = nil
428 person[:seat] = choose(flight, person)
429 output << "DISPLACED #{person[:pnr]} #{person[:name]} #{seat} -> #{person[:seat] || 'UNASSIGNED'}"
430 end
431 else
432 flight[:blocked].delete(seat)
433 end
434 return output + reassign(flight)
435 end
436 inspect_flight(name, flight, args)
437 end

439 def inspect_flight(name, flight, args)
440 people = passengers(flight)
441 seated = people.select { |person| person[:seat] }.sort_by do |person|
442 seat = flight[:model][:seats][person[:seat]]
443 [seat[:row], seat[:letter]]
444 end
445 unassigned = people.select { |person| person[:type] != "INFANT" && !person[:seat] }
446 cabins = CLASSES.select { |item| flight[:model][:cabins].key?(item) }
447 if name == "MAP"
448 occupied = seated.map { |person| person[:seat] }.to_set
449 output = ["MAP #{flight[:name]}"]
450 cabins.each do |item|
451 cabin = flight[:model][:cabins][item]
452 output.concat(["CABIN #{item}", " " + cabin[:blocks].join(" ")])
453 (cabin[:start]..cabin[:end]).each do |row|
454 blocks = cabin[:blocks].map do |block|
455 block.chars.map do |letter|
456 id = "#{row}#{letter}"
457 if flight[:blocked].include?(id)
458 "#"
459 elsif occupied.include?(id)
460 "X"
461 elsif flight[:model][:exits].include?(row)
462 "E"
463 else
464 "."
465 end
466 end.join
467 end
468 output << row.to_s.rjust(2) + " " + blocks.join(" ")
469 end
470 end
471 return output
472 end
473 if name == "AVAIL"
474 cabin = cabin(flight, args[1])
475 occupied = seated.map { |person| person[:seat] }.to_set
476 free = cabin[:seats].reject { |seat| flight[:blocked].include?(seat) || occupied.include?(seat) }
477 .map { |seat| flight[:model][:seats][seat] }
478 return ["AVAIL #{flight[:name]} #{args[1]} free #{free.length} W #{free.count { |seat| seat[:kind] == 'W' }} " +
479 "M #{free.count { |seat| seat[:kind] == 'M' }} A #{free.count { |seat| seat[:kind] == 'A' }} " +
480 "EXIT #{free.count { |seat| flight[:model][:exits].include?(seat[:row]) }}"]
481 end
482 if name == "MANIFEST"
483 infants = people.count { |person| person[:type] == "INFANT" }
484 output = ["MANIFEST #{flight[:name]} seated #{seated.length} unassigned #{unassigned.length} infants #{infants}"]
485 (seated + unassigned).each do |person|
486 line = "#{person[:seat] || '---'} #{person[:cabin]} #{person[:pnr]} #{person[:name]} #{person[:type]}"
487 line += " +#{person[:infant]}" if person[:infant]
488 line += " #{tier(person[:name])}" unless tier(person[:name]) == "NONE"
489 line += " CHECKED" if @bookings[person[:pnr]][:checked]
490 line += " VOL" if flight[:volunteers].include?(person)
491 output << line
492 end
493 return output
494 end
495 if name == "LOADFACTOR"
496 output = ["LOADFACTOR #{flight[:name]}"]
497 total_seated = 0
498 total_seats = 0
499 cabins.each do |item|
500 count = seated.count { |person| person[:cabin] == item }
501 seats = flight[:model][:cabins][item][:seats].count { |seat| !flight[:blocked].include?(seat) }
502 total_seated += count
503 total_seats += seats
504 output << "#{item} #{count}/#{seats} #{load_factor(count, seats)} booked #{booked(flight, item)}/#{flight[:limits][item]}"
505 end
506 output << "ALL #{total_seated}/#{total_seats} #{load_factor(total_seated, total_seats)}"
507 return output
508 end
509 if name == "BALANCE"
510 weight = 0
511 moment = 0
512 seated.each do |person|
513 value = (person[:type] == "ADULT" ? 80 : 40) + (person[:infant] ? 10 : 0)
514 weight += value
515 moment += value * (flight[:model][:seats][person[:seat]][:row] - flight[:model][:base])
516 end
517 limit = flight[:model][:limit]
518 verdict = moment < -limit ? "NOSE-HEAVY" : moment > limit ? "TAIL-HEAVY" : "OK"
519 return ["BALANCE #{flight[:name]} weight #{weight} moment #{moment} limit #{limit} #{verdict}"]
520 end
521 entries = seated.map do |person|
522 booking = @bookings[person[:pnr]]
523 seat = flight[:model][:seats][person[:seat]]
524 if booking[:passengers].any? { |passenger| passenger[:type] == "INFANT" }
525 group = "PRE"
526 key = [0, seat[:row], seat[:letter]]
527 elsif person[:cabin] != "Y"
528 group = "CABIN"
529 key = [1, seat[:row], seat[:letter]]
530 elsif %w[PLATINUM GOLD].include?(tier(person[:name]))
531 group = "TIER"
532 key = [2, seat[:row], seat[:letter]]
533 else
534 zone = (seat[:row] - flight[:model][:cabins]["Y"][:start]).div(5) + 1
535 group = "ZONE#{zone}"
536 key = [3, -zone, %w[W M A].index(seat[:kind]), -seat[:row], seat[:letter]]
537 end
538 { person: person, group: group, key: key }
539 end.sort_by { |entry| entry[:key] }
540 output = ["BOARDING #{flight[:name]} #{entries.length}"]
541 entries.each_with_index do |entry, i|
542 person = entry[:person]
543 output << "#{i + 1} #{entry[:group]} #{person[:seat]} #{person[:pnr]} #{person[:name]}"
544 end
545 output
546 end
547end
