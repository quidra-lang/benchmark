70function get_flight(id,open=false)
71 haskey(FLIGHTS,id) || fail("unknown flight $id")
72 f = FLIGHTS[id]
73 open && f.closed && fail("flight $id is closed")
74 return f
75end
76function get_booking(pnr,open=false)
77 haskey(BOOKINGS,pnr) && !isempty(members(BOOKINGS[pnr])) || fail("unknown PNR $pnr")
78 b = BOOKINGS[pnr]
79 get_flight(b.flight,open)
80 return b
81end
82function get_person(b,name)
83 for id in members(b)
84 PEOPLE[id].name == name && return PEOPLE[id]
85 end
86 fail("no passenger $name in $(b.pnr)")
87end
88function cabin(f,class)
89 haskey(f.model.cabins,class) || fail("no cabin $class on $(f.id)")
90 return f.model.cabins[class]
91end
92booked(f,class) = count(id -> PEOPLE[id].kind != "INFANT" && PEOPLE[id].cabin == class,passengers(f))
93function allowed(f,p,seat,class,vacated=Set{Int}(),assignments=Dict{Int,String}())
94 s = get(f.model.seats,seat,nothing)
95 (s === nothing || s.cabin != class || seat in f.blocked) && return false
96 haskey(f.occupied,seat) && !(f.occupied[seat] in vacated) && return false
97 (p.kind == "CHILD" || p.infant != 0) && s.row in f.model.exits && return false
98 if p.infant != 0
99 for (otherseat,id) in f.occupied
100 (id == p.id || id in vacated || PEOPLE[id].infant == 0) && continue
101 t = f.model.seats[otherseat]
102 t.row == s.row && t.block == s.block && return false
103 end
104 for (id,otherseat) in assignments
105 if id != p.id && PEOPLE[id].infant != 0
106 t = f.model.seats[otherseat]
107 t.row == s.row && t.block == s.block && return false
108 end
109 end
110 end
111 return true
112end
113function select_seat(f,p,class=p.cabin)
114 firstseat = ""
115 for seat in f.model.cabins[class].seats
116 if allowed(f,p,seat,class)
117 isempty(firstseat) && (firstseat = seat)
118 !isempty(p.preference) && f.model.seats[seat].kind == p.preference && return seat
119 end
120 end
121 return firstseat
122end
123function move_to!(f,p,seat)
124 !isempty(p.seat) && delete!(f.occupied,p.seat)
125 p.seat = seat
126 !isempty(seat) && (f.occupied[seat] = p.id)
127end
128function reassign!(f,output)
129 for id in passengers(f)
130 p = PEOPLE[id]
131 if p.kind != "INFANT" && isempty(p.seat)
132 seat = select_seat(f,p)
133 if !isempty(seat)
134 move_to!(f,p,seat)
135 push!(output,"REASSIGNED $(p.pnr) $(p.name) $seat")
136 end
137 end
138 end
139end
140function fits(f,ids,seats,class)
141 assignments = Dict(zip(ids,seats))
142 return all(allowed(f,PEOPLE[id],seats[index],class,Set{Int}(),assignments) for (index,id) in enumerate(ids))
143end
144function group_seats(f,ids,class)
145 c = f.model.cabins[class]
146 total = length(ids)
147 for method in ("BLOCK","ROW")
148 for row in c.firstrow:c.lastrow
149 blocks = method == "BLOCK" ? c.blocks : [join(c.blocks)]
150 for block in blocks
151 for x in 1:length(block)-total+1
152 seats = [string(row,block[x+j]) for j in 0:total-1]
153 fits(f,ids,seats,class) && return method,seats
154 end
155 end
156 end
157 end
158 for height in 2:total
159 width = cld(total,height)
160 rows = cld(total,width)
161 for row in c.firstrow:c.lastrow-rows+1
162 for block in c.blocks
163 for x in 1:length(block)-width+1
164 seats = [string(row+div(j,width),block[x+mod(j,width)]) for j in 0:total-1]
165 fits(f,ids,seats,class) && return "STACK",seats
166 end
167 end
168 end
169 end
170 return "SPLIT",String[]
171end
172removal_targets(p) = p.infant == 0 ? [p.id] : [p.id,p.infant]
173function remove_people!(f,ids)
174 for id in ids
175 p = PEOPLE[id]
176 p.active || continue
177 move_to!(f,p,"")
178 p.active = false
179 p.volunteer = false
180 p.kind == "INFANT" && (PEOPLE[p.adult].infant = 0)
181 end
182end
183function load_text(seated,seats)
184 q = div(seated*1000,seats)
185 return string(div(q,10),'.',mod(q,10),'%')
186end
187seat_key(f,p) = (f.model.seats[p.seat].row,f.model.seats[p.seat].letter)
188function execute(line)
189 t = String.(split(line,' '))
190 command = t[1]
191 output = String[]
192 if command == "AIRCRAFT"
193 DEFINITION[] = Aircraft(id=t[2],base=parse(Int,t[3]),limit=parse(Int,t[4]))
194 elseif command == "CABIN"
195 model = DEFINITION[]
196 class = t[2]
197 blocks = t[5:end]
198 c = Cabin(firstrow=parse(Int,t[3]),lastrow=parse(Int,t[4]),blocks=blocks)
199 for row in c.firstrow:c.lastrow
200 for (b,block) in enumerate(blocks)
201 for x in eachindex(block)
202 letter = string(block[x])
203 if (b == 1 && x == 1) || (b == length(blocks) && x == length(block))
204 kind = "W"
205 elseif x == 1 || x == length(block)
206 kind = "A"
207 else
208 kind = "M"
209 end
210 seat = string(row,letter)
211 model.seats[seat] = Seat(row,letter,class,b,kind)
212 push!(c.seats,seat)
213 end
214 end
215 end
216 model.cabins[class] = c
217 elseif command == "EXIT"
218 union!(DEFINITION[].exits,parse.(Int,t[2:end]))
219 elseif command == "BLOCKED"
220 union!(DEFINITION[].blocked,t[2:end])
221 elseif command == "END"
222 model = DEFINITION[]
223 parts = ["AIRCRAFT",model.id]
224 for class in CLASSES
225 if haskey(model.cabins,class)
226 c = model.cabins[class]
227 c.capacity = count(seat -> !(seat in model.blocked),c.seats)
228 append!(parts,[class,string(c.capacity)])
229 end
230 end
231 AIRCRAFT[model.id] = model
232 push!(output,join(parts," "))
233 DEFINITION[] = nothing
234 elseif command == "FLIGHT"
235 haskey(FLIGHTS,t[2]) && fail("duplicate flight $(t[2])")
236 haskey(AIRCRAFT,t[3]) || fail("unknown aircraft $(t[3])")
237 model = AIRCRAFT[t[3]]
238 f = Flight(id=t[2],model=model,blocked=copy(model.blocked))
239 parts = ["FLIGHT",f.id,model.id]
240 for class in CLASSES
241 if haskey(model.cabins,class)
242 capacity = model.cabins[class].capacity
243 f.limits[class] = div(capacity*(100+parse(Int,t[4])),100)
244 append!(parts,[class,"$capacity/$(f.limits[class])"])
245 end
246 end
247 FLIGHTS[f.id] = f
248 push!(FLIGHT_ORDER,f.id)
249 push!(output,join(parts," "))
250 elseif command == "STATUS"
251 STATUSES[t[2]] = t[3]
252 push!(output,"STATUS $(t[2]) $(t[3])")
253 elseif command == "BOOK"
254 pnr,flight,class = t[2:4]
255 pnr in USED_PNR && fail("duplicate PNR $pnr")
256 f = get_flight(flight,true)
257 cabin(f,class)
258 parsed = [String.(split(token,':')) for token in t[5:end]]
259 names = Set{String}()
260 for parts in parsed
261 parts[1] in names && fail("duplicate name $(parts[1]) in $pnr")
262 push!(names,parts[1])
263 end
264 aboard = Set(PEOPLE[id].name for id in passengers(f))
265 for parts in parsed
266 parts[1] in aboard && fail("$(parts[1]) already on $flight")
267 end
268 adults = count(parts -> parts[2] == "ADULT",parsed)
269 infants = count(parts -> parts[2] == "INFANT",parsed)
270 adults > 0 || fail("no adult in $pnr")
271 infants <= adults || fail("too many infants in $pnr")
272 booked(f,class)+length(parsed)-infants <= f.limits[class] || fail("limit exceeded for $flight $class")
273 b = Booking(pnr=pnr,flight=flight)
274 adultids,infantids,seatids = Int[],Int[],Int[]
275 for parts in parsed
276 id = length(PEOPLE)+1
277 preference = length(parts) == 3 ? parts[3] : ""
278 p = Passenger(id=id,pnr=pnr,name=parts[1],kind=parts[2],preference=preference,cabin=class)
279 PEOPLE[id] = p
280 push!(b.people,id)
281 push!(f.people,id)
282 p.kind == "ADULT" && push!(adultids,id)
283 if p.kind == "INFANT"
284 push!(infantids,id)
285 else
286 push!(seatids,id)
287 end
288 end
289 for (index,id) in enumerate(infantids)
290 PEOPLE[id].adult = adultids[index]
291 PEOPLE[adultids[index]].infant = id
292 end
293 BOOKINGS[pnr] = b
294 push!(USED_PNR,pnr)
295 push!(f.bookings,pnr)
296 if length(seatids) == 1
297 method = "SINGLE"
298 p = PEOPLE[seatids[1]]
299 move_to!(f,p,select_seat(f,p))
300 else
301 method,seats = group_seats(f,seatids,class)
302 for (index,id) in enumerate(seatids)
303 seat = method == "SPLIT" ? select_seat(f,PEOPLE[id]) : seats[index]
304 move_to!(f,PEOPLE[id],seat)
305 end
306 end
307 parts = ["BOOKED",pnr,flight,class,method]
308 append!(parts,[PEOPLE[id].name*":"*passenger_result(PEOPLE[id]) for id in b.people])
309 push!(output,join(parts," "))
310 elseif command == "FIND"
311 lines = String[]
312 for flight in FLIGHT_ORDER
313 f = FLIGHTS[flight]
314 for id in passengers(f)
315 p = PEOPLE[id]
316 if p.name == t[2]
317 state = f.closed ? "CLOSED" : "OPEN"
318 push!(lines,"$flight $(p.pnr) $(p.cabin) $(passenger_result(p)) $state")
319 end
320 end
321 end
322 push!(output,"FIND $(t[2]) $(get(STATUSES,t[2],"NONE")) $(length(lines))")
323 append!(output,lines)
324 elseif command in ("SEAT","SWAP","CANCEL","CHECKIN","VOLUNTEER","INFO")
325 b = get_booking(t[2],command != "INFO")
326 f = FLIGHTS[b.flight]
327 if command == "INFO"
328 state = b.checked ? "CHECKED" : "NOT-CHECKED"
329 push!(output,"INFO $(b.pnr) $(b.flight) $(length(members(b))) $state")
330 for id in members(b)
331 p = PEOPLE[id]
332 push!(output,"$(p.name) $(p.kind) $(p.cabin) $(passenger_result(p))")
333 end
334 elseif command == "CHECKIN"
335 b.checked && fail("$(b.pnr) already checked in")
336 b.checked = true
337 push!(output,"CHECKIN $(b.pnr) $(length(members(b)))")
338 elseif command == "CANCEL"
339 targets = members(b)
340 if length(t) == 3
341 p = get_person(b,t[3])
342 targetset = Set(removal_targets(p))
343 remaining = filter(id -> !(id in targetset),members(b))
344 !isempty(remaining) && !any(PEOPLE[id].kind == "ADULT" for id in remaining) && fail("$(b.pnr) would have no adult")
345 targets = filter(id -> id in targetset,members(b))
346 end
347 names = [PEOPLE[id].name for id in targets]
348 remove_people!(f,targets)
349 push!(output,"CANCELLED $(b.pnr) "*join(names," "))
350 reassign!(f,output)
351 else
352 p = get_person(b,t[3])
353 if command == "SWAP"
354 b2 = get_booking(t[4],true)
355 q = get_person(b2,t[5])
356 p.id == q.id && fail("cannot swap $(p.name) with itself")
357 for person in (p,q)
358 person.kind == "INFANT" && fail("infant $(person.name) has no seat")
359 end
360 b.flight == b2.flight || fail("$(b.pnr) and $(b2.pnr) are on different flights")
361 p.cabin == q.cabin || fail("$(p.name) and $(q.name) are in different cabins")
362 for person in (p,q)
363 isempty(person.seat) && fail("$(person.name) has no seat")
364 end
365 vacated = Set([p.id,q.id])
366 assignments = Dict(p.id=>q.seat,q.id=>p.seat)
367 if !allowed(f,p,q.seat,p.cabin,vacated,assignments) || !allowed(f,q,p.seat,q.cabin,vacated,assignments)
368 fail("swap of $(p.name) and $(q.name) breaks seating rules")
369 end
370 oldp,oldq = p.seat,q.seat
371 move_to!(f,p,"")
372 move_to!(f,q,"")
373 move_to!(f,p,oldq)
374 move_to!(f,q,oldp)
375 push!(output,"SWAPPED $(b.pnr) $(p.name) $(p.seat) $(b2.pnr) $(q.name) $(q.seat)")
376 reassign!(f,output)
377 elseif command == "VOLUNTEER"
378 p.kind == "ADULT" && p.infant == 0 || fail("$(p.name) cannot volunteer")
379 isempty(p.seat) && fail("$(p.name) has no seat")
380 p.volunteer && fail("$(p.name) already volunteered")
381 p.volunteer = true
382 push!(f.volunteers,p.id)
383 push!(output,"VOLUNTEER $(p.pnr) $(p.name) $(p.seat)")
384 else
385 seat = t[4]
386 p.kind == "INFANT" && fail("infant $(p.name) has no seat")
387 s = get(f.model.seats,seat,nothing)
388 s !== nothing && s.cabin == p.cabin || fail("no seat $seat in cabin $(p.cabin)")
389 seat in f.blocked && fail("seat $seat is blocked")
390 haskey(f.occupied,seat) && fail("seat $seat is occupied")
391 (p.kind == "CHILD" || p.infant != 0) && s.row in f.model.exits && fail("exit row $(s.row) not allowed for $(p.name)")
392 allowed(f,p,seat,p.cabin) || fail("infant limit reached at $seat")
393 old = p.seat
394 move_to!(f,p,seat)
395 push!(output,"MOVED $(p.pnr) $(p.name) $old -> $seat")
396 reassign!(f,output)
397 end
398 end
399 else
400 f = get_flight(t[2],command in ("UPGRADE","CLOSE","BLOCK","UNBLOCK"))
401 classes = filter(class -> haskey(f.model.cabins,class),collect(CLASSES))
402 if command == "UPGRADE"
403 total = 0
404 for index in 1:length(classes)-1
405 higher,lower = classes[index:index+1]
406 targets = filter(id -> PEOPLE[id].cabin == lower && PEOPLE[id].kind == "ADULT" && PEOPLE[id].infant == 0 && tier(PEOPLE[id]) != "NONE",passengers(f))
407 sort!(targets;by=id -> (TIERS[tier(PEOPLE[id])],id))
408 for id in targets
409 booked(f,higher) >= f.limits[higher] && break
410 p = PEOPLE[id]
411 seat = select_seat(f,p,higher)
412 isempty(seat) && break
413 old = passenger_result(p)
414 move_to!(f,p,seat)
415 p.cabin = higher
416 total += 1
417 push!(output,"UPGRADED $(p.pnr) $(p.name) $lower $old -> $higher $seat")
418 end
419 end
420 reassign!(f,output)
421 push!(output,"UPGRADE $(f.id) $total")
422 elseif command == "CLOSE"
423 for pnr in f.bookings
424 b = BOOKINGS[pnr]
425 ids = members(b)
426 if !isempty(ids) && !b.checked
427 push!(output,"NOSHOW $pnr $(length(ids))")
428 remove_people!(f,ids)
429 end
430 end
431 reassign!(f,output)
432 denied,total = 0,0
433 rates = Dict("F"=>500,"J"=>300,"Y"=>150)
434 for class in classes
435 targets = filter(id -> PEOPLE[id].active && PEOPLE[id].volunteer && !isempty(PEOPLE[id].seat) && PEOPLE[id].cabin == class,f.volunteers)
436 for id in targets
437 p = PEOPLE[id]
438 useful = any(passengers(f)) do waitingid
439 waiting = PEOPLE[waitingid]
440 waiting.kind != "INFANT" && waiting.cabin == class && isempty(waiting.seat) && allowed(f,waiting,p.seat,class,Set([id]))
441 end
442 if useful
443 amount = rates[class]
444 push!(output,"DENIED VOLUNTARY $(p.pnr) $(p.name) $class $amount")
445 remove_people!(f,[id])
446 denied += 1
447 total += amount
448 reassign!(f,output)
449 end
450 end
451 end
452 for id in passengers(f)
453 p = PEOPLE[id]
454 if p.active && p.kind != "INFANT" && isempty(p.seat)
455 amount = rates[p.cabin]*2
456 p.kind == "CHILD" && (amount = div(amount,2))
457 push!(output,"DENIED INVOLUNTARY $(p.pnr) $(p.name) $(p.cabin) $amount")
458 remove_people!(f,removal_targets(p))
459 denied += 1
460 total += amount
461 end
462 end
463 f.closed = true
464 push!(output,"CLOSED $(f.id) denied $denied compensation $total")
465 elseif command in ("BLOCK","UNBLOCK")
466 seat = t[3]
467 haskey(f.model.seats,seat) || fail("no seat $seat on $(f.id)")
468 blocked = seat in f.blocked
469 command == "BLOCK" && blocked && fail("seat $seat is blocked")
470 command == "UNBLOCK" && !blocked && fail("seat $seat is not blocked")
471 push!(output,"$command $(f.id) $seat")
472 if command == "BLOCK"
473 push!(f.blocked,seat)
474 if haskey(f.occupied,seat)
475 p = PEOPLE[f.occupied[seat]]
476 move_to!(f,p,"")
477 move_to!(f,p,select_seat(f,p))
478 push!(output,"DISPLACED $(p.pnr) $(p.name) $seat -> $(passenger_result(p))")
479 end
480 else
481 delete!(f.blocked,seat)
482 end
483 reassign!(f,output)
484 elseif command == "MAP"
485 push!(output,"MAP $(f.id)")
486 for class in classes
487 c = f.model.cabins[class]
488 push!(output,"CABIN $class")
489 push!(output," "*join(c.blocks," "))
490 for row in c.firstrow:c.lastrow
491 blocks = String[]
492 for block in c.blocks
493 text = ""
494 for letter in block
495 seat = string(row,letter)
496 if seat in f.blocked
497 text *= "#"
498 elseif haskey(f.occupied,seat)
499 text *= "X"
500 else
501 text *= row in f.model.exits ? "E" : "."
502 end
503 end
504 push!(blocks,text)
505 end
506 push!(output,@sprintf("%2d ",row)*join(blocks," "))
507 end
508 end
509 elseif command in ("MANIFEST","BOARDING")
510 seated,unassigned = Int[],Int[]
511 infants = 0
512 for id in passengers(f)
513 p = PEOPLE[id]
514 if p.kind == "INFANT"
515 infants += 1
516 elseif isempty(p.seat)
517 push!(unassigned,id)
518 else
519 push!(seated,id)
520 end
521 end
522 if command == "MANIFEST"
523 sort!(seated;by=id -> seat_key(f,PEOPLE[id]))
524 push!(output,"MANIFEST $(f.id) seated $(length(seated)) unassigned $(length(unassigned)) infants $infants")
525 for id in vcat(seated,unassigned)
526 p = PEOPLE[id]
527 parts = [isempty(p.seat) ? "---" : p.seat,p.cabin,p.pnr,p.name,p.kind]
528 p.infant != 0 && push!(parts,"+"*PEOPLE[p.infant].name)
529 tier(p) != "NONE" && push!(parts,tier(p))
530 BOOKINGS[p.pnr].checked && push!(parts,"CHECKED")
531 p.volunteer && push!(parts,"VOL")
532 push!(output,join(parts," "))
533 end
534 else
535 groups = Dict{Int,String}()
536 sortkeys = Dict{Int,Tuple}()
537 for id in seated
538 p = PEOPLE[id]
539 s = f.model.seats[p.seat]
540 hasinfant = any(PEOPLE[member].kind == "INFANT" for member in members(BOOKINGS[p.pnr]))
541 if hasinfant
542 group,key = "PRE",(0,s.row,s.letter)
543 elseif p.cabin != "Y"
544 group,key = "CABIN",(1,s.row,s.letter)
545 elseif TIERS[tier(p)] <= 1
546 group,key = "TIER",(2,s.row,s.letter)
547 else
548 zone = div(s.row-f.model.cabins["Y"].firstrow,5)+1
549 group = "ZONE$zone"
550 key = (3,-zone,findfirst(==(s.kind),["W","M","A"]),-s.row,s.letter)
551 end
552 groups[id],sortkeys[id] = group,key
553 end
554 sort!(seated;by=id -> sortkeys[id])
555 push!(output,"BOARDING $(f.id) $(length(seated))")
556 for (index,id) in enumerate(seated)
557 p = PEOPLE[id]
558 push!(output,"$index $(groups[id]) $(p.seat) $(p.pnr) $(p.name)")
559 end
560 end
561 elseif command == "LOADFACTOR"
562 push!(output,"LOADFACTOR $(f.id)")
563 totalseated,totalseats = 0,0
564 for class in classes
565 seats = count(seat -> !(seat in f.blocked),f.model.cabins[class].seats)
566 seated = count(id -> PEOPLE[id].cabin == class && !isempty(PEOPLE[id].seat),passengers(f))
567 push!(output,"$class $seated/$seats $(load_text(seated,seats)) booked $(booked(f,class))/$(f.limits[class])")
568 totalseated += seated
569 totalseats += seats
570 end
571 push!(output,"ALL $totalseated/$totalseats $(load_text(totalseated,totalseats))")
572 elseif command == "BALANCE"
573 weight,moment = 0,0
574 for id in passengers(f)
575 p = PEOPLE[id]
576 if !isempty(p.seat)
577 w = p.kind == "ADULT" ? 80 : 40
578 p.infant != 0 && (w += 10)
579 weight += w
580 moment += w*(f.model.seats[p.seat].row-f.model.base)
581 end
582 end
583 limit = f.model.limit
584 if moment < -limit
585 verdict = "NOSE-HEAVY"
586 elseif moment > limit
587 verdict = "TAIL-HEAVY"
588 else
589 verdict = "OK"
590 end
591 push!(output,"BALANCE $(f.id) weight $weight moment $moment limit $limit $verdict")
592 elseif command == "AVAIL"

602 s.row in f.model.exits && (counts["EXIT"] += 1)

604 end
605 push!(output,"AVAIL $(f.id) $class free $free W $(counts["W"]) M $(counts["M"]) A $(counts["A"]) EXIT $(counts["EXIT"])")
606 end
607 end
608 return output
