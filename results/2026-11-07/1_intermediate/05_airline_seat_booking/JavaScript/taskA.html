13function seatOrder(a, b) {
14 return a.row - b.row || a.letter.charCodeAt(0) - b.letter.charCodeAt(0);

17function display(passenger) {

21function loadFactor(seated, seats) {
22 const value = Math.floor(seated * 1000 / seats);
23 return Math.floor(value / 10) + "." + value % 10 + "%";

82 fits(flight, people, seats, cabin = "", vacated = []) {
83 const ignored = new Set([...people, ...vacated]);
84 const existing = this.passengers(flight).filter(person => person.seat && !ignored.has(person));
85 const occupied = new Set(existing.map(person => person.seat));
86 const infants = new Set();
87 for (const person of existing.filter(person => person.infant)) {
88 const seat = flight.model.seats.get(person.seat);
89 infants.add(seat.row + ":" + seat.block);
90 }
91 for (let i = 0; i < people.length; i++) {
92 const person = people[i];
93 const seat = flight.model.seats.get(seats[i]);
94 if (!seat || seat.cabin !== (cabin || person.cabin) || flight.blocked.has(seat.id) || occupied.has(seat.id)) {
95 return false;
96 }
97 if ((person.type === "CHILD" || person.infant) && flight.model.exits.has(seat.row)) {
98 return false;
99 }
100 const key = seat.row + ":" + seat.block;
101 if (person.infant && infants.has(key)) {
102 return false;
103 }
104 if (person.infant) {
105 infants.add(key);
106 }
107 occupied.add(seat.id);
108 }
109 return true;
110 }

112 choose(flight, person, cabin = person.cabin) {
113 const candidates = this.cabin(flight, cabin).seats.filter(seat => this.fits(flight, [person], [seat], cabin));
114 const preferred = candidates.find(seat => flight.model.seats.get(seat).kind === person.pref);
115 return preferred ?? candidates[0] ?? "";
116 }

118 reassign(flight) {
119 const output = [];
120 for (const person of this.passengers(flight)) {
121 if (person.type !== "INFANT" && !person.seat) {
122 const seat = this.choose(flight, person);
123 if (seat) {
124 person.seat = seat;
125 output.push("REASSIGNED " + person.pnr + " " + person.name + " " + seat);
126 }
127 }
128 }
129 return output;
130 }

132 assign(flight, people, cabinName) {
133 if (people.length === 1) {
134 people[0].seat = this.choose(flight, people[0]);
135 return "SINGLE";
136 }
137 const cabin = this.cabin(flight, cabinName);
138 const k = people.length;
139 const adopt = (seats) => {
140 if (!this.fits(flight, people, seats)) {
141 return false;
142 }
143 people.forEach((person, i) => person.seat = seats[i]);
144 return true;
145 };
146 for (const stage of ["BLOCK", "ROW"]) {
147 for (let row = cabin.start; row <= cabin.end; row++) {
148 const blocks = stage === "BLOCK" ? cabin.blocks : [cabin.blocks.join("")];
149 for (const block of blocks) {
150 for (let start = 0; start + k <= block.length; start++) {
151 const seats = block.slice(start, start + k).split("").map(letter => row + letter);
152 if (adopt(seats)) {
153 return stage;
154 }
155 }
156 }
157 }
158 }
159 for (let h = 2; h <= k; h++) {
160 const width = Math.floor((k + h - 1) / h);
161 const height = Math.floor((k + width - 1) / width);
162 for (let row = cabin.start; row + height - 1 <= cabin.end; row++) {
163 for (const block of cabin.blocks) {
164 for (let start = 0; start + width <= block.length; start++) {
165 const seats = Array.from({ length: k }, (_, j) => (row + Math.floor(j / width)) + block[start + j % width]);
166 if (adopt(seats)) {
167 return "STACK";
168 }
169 }
170 }
171 }
172 }
173 for (const person of people) {
174 person.seat = this.choose(flight, person);
175 }
176 return "SPLIT";
177 }

179 targets(person) {
180 const booking = this.bookings.get(person.pnr);
181 return booking.passengers.filter(other => other === person || person.infant === other.name);

184 remove(flight, targets) {
185 const removed = new Set(targets);
186 for (const person of targets) {
187 if (person.type === "INFANT") {
188 const adult = this.bookings.get(person.pnr).passengers.find(other => other.name === person.adult);
189 if (adult) {
190 adult.infant = "";
191 }
192 }
193 }
194 for (const booking of flight.bookings) {
195 booking.passengers = booking.passengers.filter(person => !removed.has(person));
196 }
197 flight.volunteers = flight.volunteers.filter(person => !removed.has(person));
198 }

200 define(name, args) {
201 if (name === "AIRCRAFT") {
202 this.defining = {
203 name: args[0], base: Number(args[1]), limit: Number(args[2]),
204 cabins: new Map(), seats: new Map(),
205 exits: new Set(), blocked: new Set()
206 };
207 return [];
208 }
209 const model = this.defining;
210 if (name === "CABIN") {
211 const blocks = args.slice(3);
212 const start = Number(args[1]);
213 const end = Number(args[2]);
214 const seats = [];
215 for (let row = start; row <= end; row++) {
216 blocks.forEach((block, blockIndex) => {
217 block.split("").forEach((letter, i) => {
218 const window = blockIndex === 0 && i === 0 || blockIndex === blocks.length - 1 && i === block.length - 1;
219 const kind = window ? "W" : i === 0 || i === block.length - 1 ? "A" : "M";
220 const id = row + letter;
221 seats.push(id);
222 model.seats.set(id, { id, row, letter, block: blockIndex, kind, cabin: args[0] });
223 });
224 });
225 }
226 model.cabins.set(args[0], { start, end, blocks, seats });
227 } else if (name === "EXIT") {
228 model.exits = new Set(args.map(Number));
229 } else if (name === "BLOCKED") {
230 model.blocked = new Set(args);
231 } else {
232 this.models.set(model.name, model);
233 this.defining = null;
234 const counts = classes.filter(cabin => model.cabins.has(cabin)).map(cabin =>
235 cabin + " " + model.cabins.get(cabin).seats.filter(seat => !model.blocked.has(seat)).length);
236 return ["AIRCRAFT " + model.name + " " + counts.join(" ")];
237 }
238 return [];
239 }

241 book(args) {
242 const [pnr, flightName, cabinName] = args;
243 if (this.bookings.has(pnr)) {
244 fail("duplicate PNR " + pnr);
245 }
246 const flight = this.getFlight(flightName, true);
247 this.cabin(flight, cabinName);
248 const people = args.slice(3).map(token => {
249 const [name, type, pref = ""] = token.split(":");
250 return { name, type, pref, cabin: cabinName, seat: "", infant: "", adult: "", pnr };
251 });
252 const seen = new Set();
253 for (const person of people) {
254 if (seen.has(person.name)) {
255 fail("duplicate name " + person.name + " in " + pnr);
256 }
257 seen.add(person.name);
258 }
259 const existing = new Set(this.passengers(flight).map(person => person.name));
260 for (const person of people) {
261 if (existing.has(person.name)) {
262 fail(person.name + " already on " + flight.name);
263 }
264 }
265 const adults = people.filter(person => person.type === "ADULT");
266 const infants = people.filter(person => person.type === "INFANT");
267 if (!adults.length) {
268 fail("no adult in " + pnr);
269 }
270 if (infants.length > adults.length) {
271 fail("too many infants in " + pnr);
272 }
273 const seated = people.filter(person => person.type !== "INFANT");
274 if (this.booked(flight, cabinName) + seated.length > flight.limits.get(cabinName)) {
275 fail("limit exceeded for " + flight.name + " " + cabinName);
276 }
277 infants.forEach((infant, i) => {
278 infant.adult = adults[i].name;
279 adults[i].infant = infant.name;
280 });
281 const booking = { pnr, flight: flight.name, passengers: people, checked: false };
282 this.bookings.set(pnr, booking);
283 flight.bookings.push(booking);
284 const method = this.assign(flight, seated, cabinName);
285 return ["BOOKED " + pnr + " " + flight.name + " " + cabinName + " " + method + " " +
286 people.map(person => person.name + ":" + display(person)).join(" ")];
287 }

289 close(flight) {
290 const output = [];
291 for (const booking of flight.bookings) {
292 if (booking.passengers.length && !booking.checked) {
293 output.push("NOSHOW " + booking.pnr + " " + booking.passengers.length);
294 this.remove(flight, [...booking.passengers]);
295 }
296 }
297 output.push(...this.reassign(flight));
298 let count = 0;
299 let total = 0;
300 const compensation = { F: 500, J: 300, Y: 150 };
301 const deny = (person, voluntary) => {
302 let value = compensation[person.cabin] * (voluntary ? 1 : 2);
303 if (person.type === "CHILD") {
304 value /= 2;
305 }
306 output.push("DENIED " + (voluntary ? "VOLUNTARY" : "INVOLUNTARY") + " " +
307 person.pnr + " " + person.name + " " + person.cabin + " " + value);
308 count++;
309 total += value;
310 this.remove(flight, this.targets(person));
311 };
312 for (const cabin of classes.filter(cabin => flight.model.cabins.has(cabin))) {
313 const volunteers = flight.volunteers.filter(person => person.seat && person.cabin === cabin);
314 for (const volunteer of volunteers) {
315 const candidates = this.passengers(flight).filter(person => person.type !== "INFANT" && !person.seat && person.cabin === cabin);
316 if (candidates.some(person => this.fits(flight, [person], [volunteer.seat], "", [volunteer]))) {
317 deny(volunteer, true);
318 output.push(...this.reassign(flight));
319 }
320 }
321 }
322 const unassigned = this.passengers(flight).filter(person => person.type !== "INFANT" && !person.seat);
323 for (const person of unassigned) {
324 deny(person, false);
325 }
326 flight.closed = true;
327 output.push("CLOSED " + flight.name + " denied " + count + " compensation " + total);
328 return output;
329 }

331 command(line) {
332 const [name, ...args] = line.split(" ");
333 if (["AIRCRAFT", "CABIN", "EXIT", "BLOCKED", "END"].includes(name)) {
334 return this.define(name, args);
335 }
336 if (name === "FLIGHT") {
337 if (this.flights.has(args[0])) {
338 fail("duplicate flight " + args[0]);
339 }
340 const model = this.models.get(args[1]);
341 if (!model) {
342 fail("unknown aircraft " + args[1]);
343 }
344 const limits = new Map();
345 const items = [];
346 for (const cabin of classes.filter(cabin => model.cabins.has(cabin))) {
347 const count = model.cabins.get(cabin).seats.filter(seat => !model.blocked.has(seat)).length;
348 const limit = Math.floor(count * (100 + Number(args[2])) / 100);
349 limits.set(cabin, limit);
350 items.push(cabin, count + "/" + limit);
351 }
352 this.flights.set(args[0], {
353 name: args[0], model, blocked: new Set(model.blocked), limits,
354 closed: false, bookings: [], volunteers: []
355 });
356 return ["FLIGHT " + args[0] + " " + model.name + " " + items.join(" ")];
357 }
358 if (name === "BOOK") {
359 return this.book(args);
360 }
361 if (name === "STATUS") {
362 this.tiers.set(args[0], args[1]);
363 return ["STATUS " + args.join(" ")];
364 }
365 if (name === "FIND") {
366 const items = [];
367 for (const flight of this.flights.values()) {
368 const person = this.passengers(flight).find(person => person.name === args[0]);
369 if (person) {
370 items.push(flight.name + " " + person.pnr + " " + person.cabin + " " + display(person) + (flight.closed ? " CLOSED" : " OPEN"));
371 }
372 }
373 return ["FIND " + args[0] + " " + this.tier(args[0]) + " " + items.length, ...items];
374 }
375 if (name === "INFO") {
376 const booking = this.getBooking(args[0]);
377 return ["INFO " + booking.pnr + " " + booking.flight + " " + booking.passengers.length +
378 (booking.checked ? " CHECKED" : " NOT-CHECKED"),
379 ...booking.passengers.map(person => person.name + " " + person.type + " " + person.cabin + " " + display(person))];
380 }
381 if (["SEAT", "SWAP", "CANCEL", "CHECKIN", "VOLUNTEER"].includes(name)) {
382 const booking = this.getBooking(args[0], true);
383 const flight = this.getFlight(booking.flight);
384 if (name === "CHECKIN") {
385 if (booking.checked) {
386 fail(booking.pnr + " already checked in");
387 }
388 booking.checked = true;
389 return ["CHECKIN " + booking.pnr + " " + booking.passengers.length];
390 }
391 if (name === "CANCEL") {
392 const targets = args.length === 1 ? [...booking.passengers] : this.targets(this.getPassenger(args[0], args[1]));
393 const remaining = booking.passengers.filter(person => !targets.includes(person));
394 if (args.length === 2 && remaining.length && !remaining.some(person => person.type === "ADULT")) {
395 fail(booking.pnr + " would have no adult");
396 }
397 const output = ["CANCELLED " + booking.pnr + " " + targets.map(person => person.name).join(" ")];
398 this.remove(flight, targets);
399 return [...output, ...this.reassign(flight)];
400 }
401 const person = this.getPassenger(args[0], args[1]);
402 if (name === "SWAP") {
403 const other = this.getPassenger(args[2], args[3]);
404 if (person === other) {
405 fail("cannot swap " + person.name + " with itself");
406 }
407 for (const p of [person, other]) {
408 if (p.type === "INFANT") {
409 fail("infant " + p.name + " has no seat");
410 }
411 }
412 if (booking.flight !== this.bookings.get(other.pnr).flight) {
413 fail(args[0] + " and " + args[2] + " are on different flights");
414 }
415 if (person.cabin !== other.cabin) {
416 fail(person.name + " and " + other.name + " are in different cabins");
417 }
418 for (const p of [person, other]) {
419 if (!p.seat) {
420 fail(p.name + " has no seat");
421 }
422 }
423 if (!this.fits(flight, [person, other], [other.seat, person.seat])) {
424 fail("swap of " + person.name + " and " + other.name + " breaks seating rules");
425 }
426 [person.seat, other.seat] = [other.seat, person.seat];
427 return ["SWAPPED " + person.pnr + " " + person.name + " " + person.seat + " " + other.pnr +
428 " " + other.name + " " + other.seat, ...this.reassign(flight)];
429 }
430 if (name === "VOLUNTEER") {
431 if (person.type !== "ADULT" || person.infant) {
432 fail(person.name + " cannot volunteer");
433 }
434 if (!person.seat) {
435 fail(person.name + " has no seat");
436 }
437 if (flight.volunteers.includes(person)) {
438 fail(person.name + " already volunteered");
439 }
440 flight.volunteers.push(person);
441 return ["VOLUNTEER " + person.pnr + " " + person.name + " " + person.seat];
442 }
443 if (person.type === "INFANT") {
444 fail("infant " + person.name + " has no seat");
445 }
446 const seat = flight.model.seats.get(args[2]);
447 if (!seat || seat.cabin !== person.cabin) {
448 fail("no seat " + args[2] + " in cabin " + person.cabin);
449 }
450 if (flight.blocked.has(seat.id)) {
451 fail("seat " + seat.id + " is blocked");
452 }
453 if (this.passengers(flight).some(other => other.seat === seat.id)) {
454 fail("seat " + seat.id + " is occupied");
455 }
456 if ((person.type === "CHILD" || person.infant) && flight.model.exits.has(seat.row)) {
457 fail("exit row " + seat.row + " not allowed for " + person.name);
458 }
459 if (!this.fits(flight, [person], [seat.id])) {
460 fail("infant limit reached at " + seat.id);
461 }
462 const old = person.seat;
463 person.seat = seat.id;
464 return ["MOVED " + person.pnr + " " + person.name + " " + old + " -> " + seat.id, ...this.reassign(flight)];
465 }
466 const modifying = ["UPGRADE", "CLOSE", "BLOCK", "UNBLOCK"].includes(name);
467 const flight = this.getFlight(args[0], modifying);
468 if (name === "CLOSE") {
469 return this.close(flight);
470 }
471 if (name === "UPGRADE") {
472 const cabins = classes.filter(cabin => flight.model.cabins.has(cabin));
473 const output = [];
474 let count = 0;
475 for (let i = 0; i + 1 < cabins.length; i++) {
476 const [higher, lower] = cabins.slice(i, i + 2);
477 const targets = this.passengers(flight).filter(person => person.cabin === lower && person.type === "ADULT" &&
478 !person.infant && this.tier(person.name) !== "NONE");
479 targets.sort((a, b) => tierOrder.indexOf(this.tier(a.name)) - tierOrder.indexOf(this.tier(b.name)));
480 for (const person of targets) {
481 if (this.booked(flight, higher) >= flight.limits.get(higher)) {
482 break;
483 }
484 const seat = this.choose(flight, person, higher);
485 if (!seat) {
486 break;
487 }
488 const old = person.seat || "UNASSIGNED";
489 person.cabin = higher;
490 person.seat = seat;
491 output.push("UPGRADED " + person.pnr + " " + person.name + " " + lower + " " + old + " -> " + higher + " " + seat);
492 count++;
493 }
494 }
495 return [...output, ...this.reassign(flight), "UPGRADE " + flight.name + " " + count];
496 }
497 if (name === "BLOCK" || name === "UNBLOCK") {
498 const seat = args[1];
499 if (!flight.model.seats.has(seat)) {
500 fail("no seat " + seat + " on " + flight.name);
501 }
502 const blocked = flight.blocked.has(seat);
503 if (name === "BLOCK" && blocked) {
504 fail("seat " + seat + " is blocked");
505 }
506 if (name === "UNBLOCK" && !blocked) {
507 fail("seat " + seat + " is not blocked");
508 }
509 const output = [name + " " + flight.name + " " + seat];
510 if (name === "BLOCK") {
511 flight.blocked.add(seat);
512 const person = this.passengers(flight).find(person => person.seat === seat);
513 if (person) {
514 person.seat = "";
515 person.seat = this.choose(flight, person);
516 output.push("DISPLACED " + person.pnr + " " + person.name + " " + seat + " -> " + (person.seat || "UNASSIGNED"));
517 }
518 } else {
519 flight.blocked.delete(seat);
520 }
521 return [...output, ...this.reassign(flight)];
522 }
523 return this.inspect(name, flight, args);
524 }

526 inspect(name, flight, args) {
527 const people = this.passengers(flight);
528 const seated = people.filter(person => person.seat).sort((a, b) =>
529 seatOrder(flight.model.seats.get(a.seat), flight.model.seats.get(b.seat)));
530 const unassigned = people.filter(person => person.type !== "INFANT" && !person.seat);
531 const cabins = classes.filter(cabin => flight.model.cabins.has(cabin));
532 if (name === "MAP") {
533 const occupied = new Set(seated.map(person => person.seat));
534 const output = ["MAP " + flight.name];
535 for (const cabinName of cabins) {
536 const cabin = flight.model.cabins.get(cabinName);
537 output.push("CABIN " + cabinName, " " + cabin.blocks.join(" "));
538 for (let row = cabin.start; row <= cabin.end; row++) {
539 const blocks = cabin.blocks.map(block => block.split("").map(letter => {
540 const id = row + letter;
541 return flight.blocked.has(id) ? "#" : occupied.has(id) ? "X" : flight.model.exits.has(row) ? "E" : ".";
542 }).join(""));
543 output.push(String(row).padStart(2, " ") + " " + blocks.join(" "));
544 }
545 }
546 return output;
547 }
548 if (name === "AVAIL") {
549 const cabin = this.cabin(flight, args[1]);
550 const occupied = new Set(seated.map(person => person.seat));
551 const free = cabin.seats.filter(seat => !flight.blocked.has(seat) && !occupied.has(seat)).map(seat => flight.model.seats.get(seat));
552 return ["AVAIL " + flight.name + " " + args[1] + " free " + free.length + " W " + free.filter(seat => seat.kind === "W").length +
553 " M " + free.filter(seat => seat.kind === "M").length + " A " + free.filter(seat => seat.kind === "A").length +
554 " EXIT " + free.filter(seat => flight.model.exits.has(seat.row)).length];
555 }
556 if (name === "MANIFEST") {
557 const infants = people.filter(person => person.type === "INFANT").length;
558 const output = ["MANIFEST " + flight.name + " seated " + seated.length + " unassigned " + unassigned.length + " infants " + infants];
559 for (const person of [...seated, ...unassigned]) {
560 let line = (person.seat || "---") + " " + person.cabin + " " + person.pnr + " " + person.name + " " + person.type;
561 if (person.infant) {
562 line += " +" + person.infant;
563 }
564 if (this.tier(person.name) !== "NONE") {
565 line += " " + this.tier(person.name);
566 }
567 if (this.bookings.get(person.pnr).checked) {
568 line += " CHECKED";
569 }
570 if (flight.volunteers.includes(person)) {
571 line += " VOL";
572 }
573 output.push(line);
574 }
575 return output;
576 }
577 if (name === "LOADFACTOR") {
578 const output = ["LOADFACTOR " + flight.name];
579 let totalSeated = 0;
580 let totalSeats = 0;
581 for (const cabin of cabins) {
582 const count = seated.filter(person => person.cabin === cabin).length;
583 const seats = flight.model.cabins.get(cabin).seats.filter(seat => !flight.blocked.has(seat)).length;
584 totalSeated += count;
585 totalSeats += seats;
586 output.push(cabin + " " + count + "/" + seats + " " + loadFactor(count, seats) + " booked " +
587 this.booked(flight, cabin) + "/" + flight.limits.get(cabin));
588 }
589 output.push("ALL " + totalSeated + "/" + totalSeats + " " + loadFactor(totalSeated, totalSeats));
590 return output;
591 }
592 if (name === "BALANCE") {
593 let weight = 0;
594 let moment = 0;
595 for (const person of seated) {
596 const value = (person.type === "ADULT" ? 80 : 40) + (person.infant ? 10 : 0);
597 weight += value;
598 moment += value * (flight.model.seats.get(person.seat).row - flight.model.base);
599 }
600 const limit = flight.model.limit;
601 const verdict = moment < -limit ? "NOSE-HEAVY" : moment > limit ? "TAIL-HEAVY" : "OK";
602 return ["BALANCE " + flight.name + " weight " + weight + " moment " + moment + " limit " + limit + " " + verdict];
603 }
604 const entries = seated.map(person => {
605 const booking = this.bookings.get(person.pnr);
606 const seat = flight.model.seats.get(person.seat);
607 let group = "";
608 let key;
609 if (booking.passengers.some(p => p.type === "INFANT")) {
610 group = "PRE";
611 key = [0, seat.row, seat.letter.charCodeAt(0)];
612 } else if (person.cabin !== "Y") {
613 group = "CABIN";
614 key = [1, seat.row, seat.letter.charCodeAt(0)];
615 } else if (["PLATINUM", "GOLD"].includes(this.tier(person.name))) {
616 group = "TIER";
617 key = [2, seat.row, seat.letter.charCodeAt(0)];
618 } else {
619 const zone = Math.floor((seat.row - flight.model.cabins.get("Y").start) / 5) + 1;
620 group = "ZONE" + zone;
621 key = [3, -zone, ["W", "M", "A"].indexOf(seat.kind), -seat.row, seat.letter.charCodeAt(0)];
622 }
623 return { person, group, key };
624 });
625 entries.sort((a, b) => {
626 for (let i = 0; i < Math.min(a.key.length, b.key.length); i++) {
627 if (a.key[i] !== b.key[i]) {
628 return a.key[i] - b.key[i];
629 }
630 }
631 return 0;
632 });
633 return ["BOARDING " + flight.name + " " + entries.length, ...entries.map((entry, i) =>
634 (i + 1) + " " + entry.group + " " + entry.person.seat + " " + entry.person.pnr + " " + entry.person.name)];
635 }
636}
