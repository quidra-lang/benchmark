8func bit(_ word: Word, _ length: Int, _ column: Int) -> Bool {
9 ((word >> (length - column - 1)) & 1) != 0

11func bits(_ word: Word, _ length: Int) -> String {

13 return String(repeating: "0", count: length - text.count) + text

15func degree(_ word: Word) -> Int {
16 Word.bitWidth - 1 - word.leadingZeroBitCount

21func gcd(_ first: Int, _ second: Int) -> Int {
22 var a = abs(first), b = abs(second)
23 while b != 0 {
24 (a, b) = (b, a % b)
25 }
26 return a
27}
28func divide(_ original: Word, _ divisor: Word) -> (quotient: Word, remainder: Word) {
29 var dividend = original, quotient: Word = 0
30 let d = degree(divisor)
31 while dividend != 0 && degree(dividend) >= d {
32 let shift = degree(dividend) - d
33 quotient ^= 1 << shift
34 dividend ^= divisor << shift
35 }
36 return (quotient, dividend)
37}
38func basis(_ rows: Matrix, _ length: Int) -> Matrix {
39 var pivots = Matrix(repeating: 0, count: length), kept: Matrix = []
40 for original in rows {
41 var row = original
42 for column in 0..<length where bit(row, length, column) {
43 if pivots[column] != 0 {
44 row ^= pivots[column]
45 } else {
46 pivots[column] = row
47 kept.append(original)
48 break
49 }
50 }
51 }
52 return kept
53}
54func rref(_ original: Matrix, _ length: Int) -> (rows: Matrix, columns: [Int]) {
55 var rows = original, columns: [Int] = []
56 for column in 0..<length {
57 let pivot = columns.count
58 guard let found = (pivot..<rows.count).first(where: { bit(rows[$0], length, column) }) else {
59 continue
60 }
61 rows.swapAt(pivot, found)
62 let pivotRow = rows[pivot]
63 for row in rows.indices where row != pivot && bit(rows[row], length, column) {
64 rows[row] ^= pivotRow
65 }
66 columns.append(column)
67 }
68 return (Array(rows.prefix(columns.count)), columns)
69}
70func nullspace(_ original: Matrix, _ length: Int) -> Matrix {
71 let reduced = rref(original, length)
72 var result: Matrix = []
73 for column in 0..<length where !reduced.columns.contains(column) {
74 var word: Word = 1 << (length - column - 1)
75 for row in reduced.rows.indices
76 where bit(reduced.rows[row], length, column)
77 {
78 word |= 1 << (length - reduced.columns[row] - 1)
79 }
80 result.append(word)
81 }
82 return result
83}
84func span(_ rows: Matrix) -> Matrix {
85 var words: Matrix = [0]
86 for row in rows {
87 let count = words.count
88 for index in 0..<count {
89 let word = words[index] ^ row
90 words.append(word)
91 }
92 }
93 return words
94}
95func distribution(_ words: Matrix, _ length: Int) -> [Int] {
96 var counts = [Int](repeating: 0, count: length + 1)
97 for word in words {
98 counts[word.nonzeroBitCount] += 1
99 }
100 return counts
101}
102func weightList(_ counts: [Int]) -> String {
103 counts.indices.filter { counts[$0] != 0 }.map { "\($0):\(counts[$0])" }.joined(separator: " ")
104}
105func dimension(_ length: Int, _ rank: Int) throws {
106 if rank < 1 || rank > 12 || length - rank < 1 || length - rank > 12 {
107 try fail("bad dimension \(length) \(rank)")

110func removeBit(_ word: Word, _ length: Int, _ column: Int) -> Word {
111 let low = length - column - 1
112 return ((word >> (low + 1)) << low) | (word & ((1 << low) - 1))
113}
114func reedMuller(_ order: Int, _ variables: Int) -> Matrix {
115 let length = 1 << variables
116 if order == 0 {
117 return [(1 << length) - 1]
118 }
119 let upper = reedMuller(min(order, variables - 1), variables - 1).map { ($0 << (length / 2)) | $0 }
120 return upper + reedMuller(order - 1, variables - 1)
121}

123struct Decoded { let error: Word
124 let word: Word
125 let message: Int

127final class Code {
128 let n: Int
129 let k: Int
130 let generator: Matrix
131 let check: Matrix
132 let polynomial: Word?
133 private(set) var words: Matrix = []
134 private(set) var messages: [Word: Int] = [:]
135 private var leaderTable: Matrix?
136 var encoder: String {
137 polynomial == nil ? "matrix" : "polynomial"
138 }
139
140 init(_ length: Int, _ generator: Matrix, _ check: Matrix? = nil, _ polynomial: Word? = nil) throws {
141 try dimension(length, generator.count)
142 n = length
143 k = generator.count
144 self.generator = generator
145 self.check = check ?? nullspace(generator, length)
146 self.polynomial = polynomial
147 for message in 0..<(1 << k) {
148 let word = encode(message)
149 messages[word] = message
150 words.append(word)
151 }
152 words.sort()
153 }
154 func encode(_ message: Int) -> Word {
155 if let polynomial {
156 let shifted = Word(message) << (n - k)
157 return shifted | divide(shifted, polynomial).remainder
158 }
159 var word: Word = 0
160 for row in 0..<k where bit(Word(message), k, row) {
161 word ^= generator[row]
162 }
163 return word
164 }
165 func syndrome(_ word: Word) -> Int {
166 check.reduce(0) { ($0 << 1) | (($1 & word).nonzeroBitCount % 2) }
167 }
168 func leaders() -> Matrix {
169 if let leaderTable {
170 return leaderTable
171 }
172 var result = Matrix(repeating: .max, count: 1 << (n - k))
173 result[0] = 0
174 var remaining = result.count - 1, count = 1
175 while remaining > 0 {
176 var word: Word = (1 << count) - 1
177 while word < (1 << n) {
178 let s = syndrome(word)
179 if result[s] == .max {
180 result[s] = word
181 remaining -= 1
182 if remaining == 0 {
183 break
184 }
185 }
186 let lowest = word & (Word(0) &- word), next = word + lowest
187 word = next | (((next ^ word) >> 2) / lowest)
188 }
189 count += 1
190 }
191 leaderTable = result
192 return result
193 }
194 func decode(_ received: Word) -> Decoded {
195 let error = leaders()[syndrome(received)], word = received ^ error
196 return Decoded(error: error, word: word, message: messages[word]!)
197 }
198 func decodedText(_ value: Decoded) -> String {
199 "codeword \(bits(value.word, n)) message \(bits(Word(value.message), k)) error \(bits(value.error, n)) weight \(value.error.nonzeroBitCount)"
200 }
201}

257func valid(_ text: String, _ kind: Argument) -> Bool {
258 let bytes = Array(text.utf8)
259 func letter(_ c: UInt8) -> Bool {
260 (65...90).contains(c) || (97...122).contains(c)
261 }
262 func digit(_ c: UInt8) -> Bool {
263 (48...57).contains(c)
264 }
265 guard let first = bytes.first else {
266 return false
267 }
268 switch kind {
269 case .name, .code:
270 return bytes.count <= 8 && letter(first) && bytes.allSatisfy { letter($0) || digit($0) }
271 case .number:
272 return bytes.count <= 4 && (bytes.count == 1 || first != 48) && bytes.allSatisfy(digit)
273 case .bits,
274 .erasure:
275 return bytes.count <= 30 && bytes
276 .allSatisfy { $0 == 48 || $0 == 49 || (kind == .erasure && $0 == 63) }
277 }
278}

302func independent(_ verb: Verb, _ args: [String]) throws -> [String] {
303 if verb == .crc || verb == .crccheck {
304 let generator = args[0], text = args[1], d = generator.count - 1
305 if generator.first == "0" || generator.count < 2 || generator.count > 17 {
306 try fail("bad parameter")
307 }
308 if verb == .crc && text.count + d > 30 {
309 try fail("bad length")
310 }
311 let word = parseWord(text), divisor = parseWord(generator)
312 let remainder = divide(verb == .crc ? word << d : word, divisor).remainder, suffix = bits(remainder, d)
313 if verb == .crc {
314 return ["crc remainder \(suffix) frame \(text)\(suffix)"]
315 }
316 return ["crccheck \(remainder == 0 ? "ok" : "error") remainder \(suffix)"]
317 }
318 let n = Int(args[0])!
319 if verb == .bounds {
320 let d = Int(args[1])!
321 if d < 1 || d > n || n > 24 {
322 try fail("bad parameter")
323 }
324 let volume = choose[n].prefix((d - 1) / 2 + 1).reduce(0, +)
325 var hamming = 0, griesmer = 0, sum = 0
326 for k in 0...n where (1 << k) * volume <= (1 << n) {
327 hamming = k
328 }
329 for k in 1...n {
330 let denominator = 1 << (k - 1)
331 sum += (d + denominator - 1) / denominator
332 if sum <= n {
333 griesmer = k
334 }
335 }
336 return ["bounds n \(n) d \(d) singleton \(n - d + 1) hamming \(hamming) griesmer \(griesmer)"]
337 }
338 if n < 1 || n > 24 {
339 try fail("bad parameter")
340 }
341 if verb == .factor {
342 var remaining: Word = (1 << n) | 1, candidate: Word = 2
343 var divisors = 1, factors: [String] = []
344 while 2 * degree(candidate) <= degree(remaining) {
345 var exponent = 0
346 while divide(remaining, candidate).remainder == 0 {
347 remaining = divide(remaining, candidate).quotient
348 exponent += 1
349 }
350 if exponent > 0 {
351 factors.append("\(String(candidate, radix: 2))^\(exponent)")
352 divisors *= exponent + 1
353 }
354 candidate += 1
355 }
356 if remaining > 1 {
357 factors.append("\(String(remaining, radix: 2))^1")
358 divisors *= 2
359 }
360 return ["factor \(n) \(factors.joined(separator: " ")) divisors \(divisors)"]
361 }
362 let counts = args.dropFirst().map { Int($0)! }
363 if counts.count != n + 1 {
364 try fail("bad length")
365 }
366 if counts[0] != 1 {
367 try fail("bad parameter")
368 }
369 let size = counts.reduce(0, +)
370 var feasible = (size & (size - 1)) == 0, transformed: [String] = []
371 for j in 0...n {
372 var numerator = 0
373 for x in 0...n {
374 var krawtchouk = 0
375 for y in 0...j {
376 krawtchouk += (y % 2 == 0 ? 1 : -1) * binomial(x, y) * binomial(n - x, j - y)
377 }
378 numerator += counts[x] * krawtchouk
379 }
380 let divisor = gcd(numerator, size), p = numerator / divisor, q = size / divisor
381 transformed.append(q == 1 ? "\(p)" : "\(p)/\(q)")
382 if q != 1 || p < 0 {
383 feasible = false
384 }
385 }
386 return [
387 "macwilliams n \(n) size \(size)",
388 "dual \(transformed.joined(separator: " "))",
389 "feasible \(yes(feasible))"
390 ]
391}

396 func parse(_ line: String) throws -> Command {
397 let words = line.split(separator: " ").map(String.init)
398 guard let verb = Verb(rawValue: words[0]) else {
399 try fail("unknown command \(words[0])")
400 }
401 let spec = formats[verb]!, args = Array(words.dropFirst()), extra = args.count - spec.types.count
402 if spec.repeated == nil ? extra != 0 : extra < 1 || extra > spec.maximum {
403 try fail("bad arguments")
404 }
405 for index in args.indices {
406 let kind = index < spec.types.count ? spec.types[index] : spec.repeated!
407 if !valid(args[index], kind) {
408 try fail("bad arguments")
409 }
410 }
411 for index in spec.types.indices
412 where spec.types[index] == .code && codes[args[index]] == nil
413 {
414 try fail("no code \(args[index])")
415 }
416 return Command(verb: verb, args: args)
417 }
418 private func define(_ verb: Verb, _ args: [String]) throws -> [String] {
419 let name = args[0], source = codes[args[1]], other = args.count > 2 ? codes[args[2]] : nil
420 var n = 0, generator: Matrix = []
421 var check: Matrix? = nil
422 var polynomial: Word? = nil
423 switch verb {
424 case .gen, .pcheck:
425 n = args[1].count
426 let values = args.dropFirst()
427 if values.contains(where: { $0.count != n }) {
428 try fail("bad length")
429 }
430 let rows = basis(values.map(parseWord), n)
431 if verb == .gen {
432 generator = rows
433 } else {
434 check = rows
435 generator = nullspace(rows, n)
436 }
437 case .hamming, .simplex:
438 let r = Int(args[1])!
439 if r < 2 || r > 4 {
440 try fail("bad parameter")
441 }
442 n = (1 << r) - 1
443 var rows = Matrix(repeating: 0, count: r)
444 for column in 1...n {
445 for row in 0..<r {
446 rows[row] = (rows[row] << 1) | (bit(Word(column), r, row) ? 1 : 0)
447 }
448 }
449 if verb == .hamming {
450 check = rows
451 generator = nullspace(rows, n)
452 } else {
453 generator = rows
454 }
455 case .repetition, .parity:
456 n = Int(args[1])!
457 if n < 2 || n > 24 {
458 try fail("bad parameter")
459 }
460 generator = verb == .repetition ? [(1 << n) - 1] : (0..<(n - 1)).map { (1 << (n - $0 - 1)) | 1 }
461 case .rm:
462 let r = Int(args[1])!, m = Int(args[2])!
463 if m < 1 || m > 4 || r > m {
464 try fail("bad parameter")
465 }
466 n = 1 << m
467 generator = reedMuller(r, m)
468 case .cyclic:
469 n = Int(args[1])!
470 let text = args[2], d = text.count - 1
471 if n < 2 || n > 24 || text.first == "0" || text.count < 2 || text.count > n {
472 try fail("bad parameter")
473 }
474 let g = parseWord(text)
475 if divide((1 << n) | 1, g).remainder != 0 {
476 try fail("not a divisor")
477 }
478 let k = n - d
479 try dimension(n, k)
480 polynomial = g
481 generator = (0..<k).map { g << (k - $0 - 1) }
482 var rows = Matrix(repeating: 0, count: d)
483 for column in 0..<n {
484 let remainder = divide(1 << (n - column - 1), g).remainder
485 for row in 0..<d {
486 rows[row] = (rows[row] << 1) | (bit(remainder, d, row) ? 1 : 0)
487 }
488 }
489 check = rows
490 case .dual:
491 let a = source!
492 n = a.n
493 generator = a.check
494 check = a.generator
495 case .extend:
496 let a = source!
497 n = a.n + 1
498 generator = a.generator.map { ($0 << 1) | Word($0.nonzeroBitCount % 2) }
499 case .puncture, .shorten:
500 let a = source!, column = Int(args[2])! - 1
501 if column < 0 || column >= a.n {
502 try fail("bad position")
503 }
504 var rows = a.generator
505 if verb == .shorten, let pivot = rows.firstIndex(where: { bit($0, a.n, column) }) {
506 let z = rows.remove(at: pivot)
507 rows = rows.map { bit($0, a.n, column) ? $0 ^ z : $0 }
508 }
509 n = a.n - 1
510 generator = rows.map { removeBit($0, a.n, column) }
511 if verb == .puncture {
512 generator = basis(generator, n)
513 }
514 case .augment:
515 let a = source!
516 n = a.n
517 generator = basis(a.generator + [(1 << n) - 1], n)
518 case .expurgate:
519 let a = source!
520 n = a.n
521 generator = a.generator
522 if let pivot = generator.firstIndex(where: { $0.nonzeroBitCount % 2 != 0 }) {
523 let z = generator.remove(at: pivot)
524 generator = generator.map { $0.nonzeroBitCount % 2 != 0 ? $0 ^ z : $0 }
525 }
526 case .permute:
527 let a = source!, order = args.dropFirst(2).map { Int($0)! }
528 n = a.n
529 if order.count != n || Set(order) != Set(1...n) {
530 try fail("bad permutation")
531 }
532 generator = a.generator.map { row in order.reduce(Word(0)) { ($0 << 1) | (bit(row, n, $1 - 1) ? 1 : 0) } }
533 case .directsum, .uuv, .product:
534 let a = source!, b = other!
535 if verb == .uuv && a.n != b.n {
536 try fail("bad length")
537 }
538 n = verb == .product ? a.n * b.n : a.n + b.n
539 try dimension(n, verb == .product ? a.k * b.k : a.k + b.k)
540 if verb == .product {
541 generator = a.generator.flatMap { x in b.generator.map { y in
542 (0..<a.n).reduce(Word(0)) { ($0 << b.n) | (bit(x, a.n, $1) ? y : 0) }
543 } }
544 } else {
545 generator = a.generator.map { ($0 << b.n) | (verb == .uuv ? $0 : 0) } + b.generator
546 }
547 default:
548 preconditionFailure("Not a defining command")
549 }
550 let code = try Code(n, generator, check, polynomial), replaced = codes[name] != nil
551 codes[name] = code
552 return ["\(replaced ? "redefined" : "defined") \(name) n \(n) k \(code.k)"]
553 }
554 private func systematic(_ name: String, _ code: Code) -> [String] {
555 let n = code.n, k = code.k
556 var rows = code.generator, order = Array(1...n)
557 for pivot in 0..<k {
558 var column = pivot
559 while !rows[pivot...].contains(where: { bit($0, n, column) }) {
560 column += 1
561 }
562 if column != pivot {
563 let mask: Word = (1 << (n - pivot - 1)) | (1 << (n - column - 1))
564 for row in rows.indices
565 where bit(rows[row], n, pivot) != bit(rows[row], n, column)
566 {
567 rows[row] ^= mask
568 }
569 order.swapAt(pivot, column)
570 }
571 let found = (pivot..<k).first { bit(rows[$0], n, pivot) }!
572 rows.swapAt(pivot, found)
573 let pivotRow = rows[pivot]
574 for row in rows.indices where row != pivot && bit(rows[row], n, pivot) {
575 rows[row] ^= pivotRow
576 }
577 }
578 var output = ["systematic \(name) perm \(order.map(String.init).joined(separator: " "))"]
579 output += rows.map { "G \(bits($0, n))" }
580 for column in k..<n {
581 var row = rows.reduce(Word(0)) { ($0 << 1) | (bit($1, n, column) ? 1 : 0) }
582 row = (row << (n - k)) | (1 << (n - column - 1))
583 output.append("H \(bits(row, n))")
584 }
585 return output
586 }
587 private func burst(_ name: String, _ code: Code) -> [String] {
588 let detect = code.words.dropFirst().map { degree($0) - $0.trailingZeroBitCount }.min()!
589 var seen: Set<Int> = [0]
590 for length in 1...code.n {
591 for start in 0...(code.n - length) {
592 let limit: Word = length == 1 ? 1 : 1 << (length - 2)
593 for middle in 0..<limit {
594 let pattern: Word = length == 1 ? 1 : (1 << (length - 1)) | (middle << 1) | 1
595 let syndrome = code.syndrome(pattern << (code.n - start - length))
596 if !seen.insert(syndrome)
597 .inserted
598 {
599 return ["burst \(name) detect \(detect) correct \(length - 1)"]
600 }
601 }
602 }
603 }
604 preconditionFailure("A nonzero codeword must be an undetectable burst")
605 }
606 private func interleave(_ name: String, _ code: Code, _ args: [String]) throws -> [String] {
607 let messages = Array(args.dropFirst(3)), start = Int(args[1])!, length = Int(args[2])!, depth = messages.count
608 if messages.contains(where: { $0.count != code.k }) {
609 try fail("bad length")
610 }
611 if start < 1 || length < 1 || start + length - 1 > depth * code.n {
612 try fail("bad burst")
613 }
614 let sentWords = messages.map { code.encode(Int($0, radix: 2)!) }
615 var receivedWords = Matrix(repeating: 0, count: depth), sent = "", received = ""
616 for column in 0..<code.n {
617 for row in 0..<depth {
618 let position = column * depth + row + 1, original = bit(sentWords[row], code.n, column)
619 let changed = original != (start <= position && position < start + length)
620 sent += original ? "1" : "0"
621 received += changed ? "1" : "0"
622 receivedWords[row] = (receivedWords[row] << 1) | (changed ? 1 : 0)
623 }
624 }
625 var output = ["interleave \(name) depth \(depth) burst \(start) \(length)", "sent \(sent)", "recv \(received)"]
626 var correct = 0
627 for row in 0..<depth {
628 let decoded = code.decode(receivedWords[row]), message = bits(Word(decoded.message), code.k),
629 ok = message == messages[row]
630 if ok {
631 correct += 1
632 }
633 output
634 .append(
635 "row \(row + 1) errors \((receivedWords[row] ^ sentWords[row]).nonzeroBitCount) corrected \(decoded.error.nonzeroBitCount) message \(message) \(ok ? "ok" : "wrong")"
636 )
637 }
638 output.append("result ok \(correct) wrong \(depth - correct)")
639 return output
640 }
641 func execute(_ command: Command) throws -> [String] {
642 let verb = command.verb, args = command.args
643 if verb.defining {
644 return try define(verb, args)
645 }
646 if [Verb.crc, .crccheck, .factor, .macwilliams, .bounds].contains(verb) {
647 return try independent(verb, args)
648 }
649 if verb == .list {
650 return ["codes \(codes.count)"] + codes.keys.sorted().map { name in
651 let code = codes[name]!
652 return "\(name) n \(code.n) k \(code.k) encoder \(code.encoder)"
653 }
654 }
655 let name = args[0]
656 if verb == .delete {
657 codes.removeValue(forKey: name)
658 return ["deleted \(name)"]
659 }
660 let code = codes[name]!, n = code.n, k = code.k
661 switch verb {
662 case .show:
663 var output = ["code \(name) n \(n) k \(k) encoder \(code.encoder)"]
664 if let polynomial = code.polynomial {
665 output
666 .append(
667 "poly g \(bits(polynomial, n - k + 1)) h \(bits(divide((1 << n) | 1, polynomial).quotient, k + 1))"
668 )
669 }
670 return output + code.generator.map { "G \(bits($0, n))" } + code.check.map { "H \(bits($0, n))" }
671 case .info:
672 let distance = code.words.dropFirst().map(\.nonzeroBitCount).min()!, t = (distance - 1) / 2, divisor = gcd(
673 n,
674 k
675 )
676 let volume = choose[n].prefix(t + 1).reduce(0, +)
677 let cyclic = code.generator
678 .allSatisfy { code.messages[(($0 << 1) & ((1 << n) - 1)) | ($0 >> (n - 1))] != nil }
679 return [
680 "info \(name) n \(n) k \(k) rate \(k / divisor)/\(n / divisor) d \(distance) t \(t) perfect \(yes(volume == (1 << (n - k)))) mds \(yes(distance == n - k + 1)) cyclic \(cyclic ? String(code.words[1], radix: 2) : "no")"
681 ]
682 case .systematic:
683 return systematic(name, code)
684 case .weights:
685 let a = distribution(code.words, n), b = distribution(span(code.check), n)
686 let orthogonal = code.generator
687 .allSatisfy { x in code.generator.allSatisfy { (x & $0).nonzeroBitCount % 2 == 0 } }
688 return ["weights \(name) \(weightList(a))", "dual \(name) \(weightList(b))",
689 "duality \(name) self-orthogonal \(yes(orthogonal)) self-dual \(yes(orthogonal && n == 2 * k)) formally-self-dual \(yes(a == b))"]
690 case .covering:
691 let counts = distribution(code.leaders(), n), radius = counts.lastIndex(where: { $0 != 0 })!
692 let items = (0...radius).map { "\($0):\(counts[$0])" }.joined(separator: " ")
693 return ["covering \(name) radius \(radius) leaders \(items)"]
694 case .burst:
695 return burst(name, code)
696 case .trellis:
697 let profile = (0...n).map { cut in
698 let right = n - cut, prefix = code.generator.map { $0 >> right },
699 suffix = code.generator.map { $0 & ((1 << right) - 1) }
700 return basis(prefix, cut).count + basis(suffix, right).count - k
701 }
702 return ["trellis \(name) profile \(profile.map(String.init).joined(separator: " ")) max \(profile.max()!)"]
703 case .stdarray:
704 if n - k > 6 || k > 4 {
705 try fail("too large")
706 }
707 let leaders = code.leaders()
708 let syndromes = leaders.indices.sorted {
709 let a = leaders[$0], b = leaders[$1]
710 return a.nonzeroBitCount != b.nonzeroBitCount ? a.nonzeroBitCount < b.nonzeroBitCount : a < b
711 }
712 return ["stdarray \(name) rows \(leaders.count) columns \(code.words.count)"] + syndromes.map { s in
713 "\(bits(Word(s), n - k)): \(code.words.map { bits($0 ^ leaders[s], n) }.joined(separator: " "))"
714 }
715 case .equal:
716 let other = codes[args[1]]!
717 if n != other.n {
718 return ["equal \(name) \(args[1]) no lengths \(n) \(other.n)"]
719 }
720 let common = k + other.k - basis(code.generator + other.generator, n).count
721 return ["equal \(name) \(args[1]) \(yes(k == common && other.k == common)) intersection \(common)"]
722 case .encode:
723 if args[1].count != k {
724 try fail("bad length")
725 }
726 return ["encode \(name) message \(args[1]) codeword \(bits(code.encode(Int(args[1], radix: 2)!), n))"]
727 case .syndrome, .decode, .bounded, .coset:
728 let text = args[verb == .bounded ? 2 : 1]
729 if text.count != n {
730 try fail("bad length")
731 }
732 let word = parseWord(text), syndrome = code.syndrome(word)
733 if verb ==
734 .syndrome
735 {
736 return ["syndrome \(name) \(bits(Word(syndrome), n - k)) codeword \(yes(syndrome == 0))"]
737 }
738 let decoded = code.decode(word)
739 if verb == .coset {
740 let counts = distribution(code.words.map { $0 ^ word }, n)
741 return [
742 "coset \(name) syndrome \(bits(Word(syndrome), n - k)) leader \(bits(decoded.error, n)) weights \(weightList(counts))"
743 ]
744 }
745 if verb == .bounded && decoded.error
746 .nonzeroBitCount >
747 Int(args[1])!
748 {
749 return ["bounded \(name) fail weight \(decoded.error.nonzeroBitCount)"]
750 }
751 return ["\(verb == .bounded ? "bounded" : "decode") \(name) \(code.decodedText(decoded))"]
752 case .erasure:
753 let text = args[1]
754 if text.count != n {
755 try fail("bad length")
756 }
757 var mask: Word = 0, known: Word = 0, erased = 0
758 for character in text {
759 mask = (mask << 1) | (character != "?" ? 1 : 0)
760 known = (known << 1) | (character == "1" ? 1 : 0)
761 if character == "?" {
762 erased += 1
763 }
764 }
765 let candidates = code.words.filter { ($0 & mask) == known }
766 var output = "erasure \(name) erased \(erased) candidates \(candidates.count)"
767 if candidates
768 .count ==
769 1
770 {
771 output += " codeword \(bits(candidates[0], n)) message \(bits(Word(code.messages[candidates[0]]!), k))"
772 }
773 return [output]
774 case .interleave:
775 return try interleave(name, code, args)
776 default:
777 preconditionFailure("Unhandled command")
778 }
779 }
780}
