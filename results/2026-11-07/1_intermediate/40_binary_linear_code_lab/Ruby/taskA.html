34def valid_word?(text, kind)
35 case kind
36 when 'N', 'A', 'B' then text.match?(/\A[A-Za-z][A-Za-z0-9]{0,7}\z/)
37 when 'I' then text.match?(/\A(?:0|[1-9][0-9]{0,3})\z/)
38 when 'X' then text.match?(/\A[01?]{1,30}\z/)
39 else text.match?(/\A[01]{1,30}\z/)
40 end
41end

43def mask(n) = (1 << n) - 1
44def bit(n, column) = 1 << (n - column - 1)
45def weight(value) = value.to_s(2).count('1')
46def bits(value, n) = value.to_s(2).rjust(n, '0')
47def yes(value) = value ? 'yes' : 'no'

49def divide(value, divisor)
50 quotient = 0
51 while value != 0 && value.bit_length >= divisor.bit_length
52 shift = value.bit_length - divisor.bit_length
53 quotient ^= 1 << shift
54 value ^= divisor << shift
55 end
56 Division.new(quotient, value)
57end

59def basis(rows, n)
60 pivots = Array.new(n, 0)
61 result = []
62 rows.each do |original|
63 row = original
64 (n - 1).downto(0) do |column|
65 next if (row & (1 << column)).zero?
66 if pivots[column] != 0
67 row ^= pivots[column]
68 else
69 pivots[column] = row
70 result << original
71 break
72 end
73 end
74 end
75 result
76end

78def rref(source, n)
79 rows = source.dup
80 columns = []
81 n.times do |column|
82 rank = columns.size
83 found = (rank...rows.size).find { |i| (rows[i] & bit(n, column)) != 0 }
84 next unless found
85 rows[rank], rows[found] = rows[found], rows[rank]
86 rows.each_index do |i|
87 rows[i] ^= rows[rank] if i != rank && (rows[i] & bit(n, column)) != 0
88 end
89 columns << column
90 end
91 Echelon.new(rows.take(columns.size).freeze, columns.freeze)
92end

94def null_space(source, n)
95 reduced = rref(source, n)
96 (0...n).reject { |column| reduced.columns.include?(column) }.map do |column|
97 row = bit(n, column)
98 reduced.rows.each_with_index do |value, i|
99 row |= bit(n, reduced.columns[i]) if (value & bit(n, column)) != 0
100 end
101 row
102 end
103end

105def span(rows)
106 words = [0]
107 rows.each { |row| words.concat(words.map { |word| word ^ row }) }
108 words
109end

111def reed_muller(r, m)
112 return [mask(1 << m)] if r.zero?
113 half = 1 << (m - 1)
114 reed_muller([r, m - 1].min, m - 1).map { |row| (row << half) | row } + reed_muller(r - 1, m - 1)
115end

117def check_dimension(n, k)
118 check((1..12).cover?(k) && (1..12).cover?(n - k), "bad dimension #{n} #{k}")

121class Code
122 attr_reader :n, :g, :h, :polynomial
123
124 def initialize(n, generator, checks = nil, polynomial = nil)
125 check_dimension(n, generator.size)
126 @n = n
127 @g = generator.dup.freeze
128 @h = (checks || null_space(@g, n)).dup.freeze
129 @polynomial = polynomial
130 end
131
132 def k = g.size
133 def encoder = polynomial ? 'polynomial' : 'matrix'
134 def words = @words ||= span(g).freeze
135
136 def encode(message)
137 if polynomial
138 shifted = message << h.size
139 return shifted | divide(shifted, polynomial).remainder
140 end
141 g.each_with_index.reduce(0) do |word, (row, i)|
142 (message & bit(k, i)).zero? ? word : word ^ row
143 end
144 end
145
146 def syndrome(word)
147 h.reduce(0) { |result, row| (result << 1) | (weight(row & word) & 1) }
148 end
149
150 def message_for(word)
151 @messages ||= (0...(1 << k)).to_h { |message| [encode(message), message] }.freeze
152 @messages.fetch(word)
153 end
154
155 def leaders
156 return @leaders if @leaders
157 result = Array.new(1 << h.size)
158 result[0] = 0
159 found = 1
160 (1..n).each do |w|
161 word = mask(w)
162 while word < (1 << n)
163 syndrome = syndrome(word)
164 if result[syndrome].nil?
165 result[syndrome] = word
166 found += 1
167 return @leaders = result.freeze if found == result.size
168 end
169 lowest = word & -word
170 following = word + lowest
171 word = following | (((following ^ word) >> 2) / lowest)
172 end
173 end
174 @leaders = result.freeze
175 end
176
177 def decode(received)
178 error = leaders[syndrome(received)]
179 word = received ^ error
180 Decoded.new(word, message_for(word), error)
181 end
182
183 def decoded_text(decoded)
184 "codeword #{bits(decoded.word, n)} message #{bits(decoded.message, k)} " \
185 "error #{bits(decoded.error, n)} weight #{weight(decoded.error)}"
186 end

188 def burst_correction
189 seen = Array.new(1 << h.size, false)
190 seen[0] = true
191 (1..n).each do |size|
192 variants = size == 1 ? 1 : 1 << (size - 2)
193 (0..n - size).each do |offset|
194 variants.times do |middle|
195 word = size == 1 ? 1 : (1 << (size - 1)) | (middle << 1) | 1
196 syndrome = syndrome(word << offset)
197 return size - 1 if seen[syndrome]
198 seen[syndrome] = true
199 end
200 end
201 end
202 raise 'No colliding burst'
203 end
204end

206def construct(command, args, a, b)
207 case command
208 when 'GEN', 'PCHECK'
209 n = args[1].size
210 check(args.drop(1).all? { |row| row.size == n }, 'bad length')
211 rows = basis(args.drop(1).map { |row| row.to_i(2) }, n)
212 command == 'GEN' ? Code.new(n, rows) : Code.new(n, null_space(rows, n), rows)
213 when 'HAMMING', 'SIMPLEX'
214 r = args[1].to_i
215 check((2..4).cover?(r), 'bad parameter')
216 n = (1 << r) - 1
217 rows = r.times.map do |i|
218 (1..n).reduce(0) { |row, j| (row << 1) | ((j >> (r - i - 1)) & 1) }
219 end
220 command == 'SIMPLEX' ? Code.new(n, rows) : Code.new(n, null_space(rows, n), rows)
221 when 'REPETITION', 'PARITY'
222 n = args[1].to_i
223 check((2..24).cover?(n), 'bad parameter')
224 Code.new(n, command == 'REPETITION' ? [mask(n)] : (0...n - 1).map { |i| bit(n, i) | 1 })
225 when 'RM'
226 r, m = args.drop(1).map(&:to_i)
227 check((1..4).cover?(m) && r <= m, 'bad parameter')
228 Code.new(1 << m, reed_muller(r, m))
229 when 'CYCLIC'
230 n = args[1].to_i
231 d = args[2].size - 1
232 check((2..24).cover?(n) && args[2][0] == '1' && (1...n).cover?(d), 'bad parameter')
233 divisor = args[2].to_i(2)
234 check(divide((1 << n) | 1, divisor).remainder.zero?, 'not a divisor')
235 k = n - d
236 rows = k.times.map { |i| divisor << (k - i - 1) }
237 checks = Array.new(d, 0)
238 n.times do |j|
239 remainder = divide(bit(n, j), divisor).remainder
240 d.times { |i| checks[i] = (checks[i] << 1) | ((remainder >> (d - i - 1)) & 1) }
241 end
242 Code.new(n, rows, checks, divisor)
243 when 'DUAL'
244 Code.new(a.n, a.h, a.g)
245 when 'EXTEND'
246 Code.new(a.n + 1, a.g.map { |row| (row << 1) | (weight(row) & 1) })
247 when 'PUNCTURE', 'SHORTEN', 'EXPURGATE'
248 expurgating = command == 'EXPURGATE'
249 position = expurgating ? 0 : args[2].to_i
250 check((1..a.n).cover?(position), 'bad position') unless expurgating
251 n = a.n - (expurgating ? 0 : 1)
252 selected = ->(row) { expurgating ? weight(row).odd? : (row & bit(a.n, position - 1)) != 0 }
253 removed = command == 'PUNCTURE' ? nil : a.g.index(&selected)
254 rows = []
255 a.g.each_with_index do |original, i|
256 next if i == removed
257 row = original
258 row ^= a.g[removed] if removed && selected.call(row)
259 unless expurgating
260 low = a.n - position
261 row = ((row >> (low + 1)) << low) | (row & mask(low))
262 end
263 rows << row
264 end
265 Code.new(n, command == 'PUNCTURE' ? basis(rows, n) : rows)
266 when 'AUGMENT'
267 Code.new(a.n, basis(a.g + [mask(a.n)], a.n))
268 when 'PERMUTE'
269 positions = args.drop(2).map(&:to_i)
270 check(positions.size == a.n && positions.sort == (1..a.n).to_a, 'bad permutation')
271 rows = a.g.map do |original|
272 positions.reduce(0) { |row, position| (row << 1) | ((original & bit(a.n, position - 1)).zero? ? 0 : 1) }
273 end
274 Code.new(a.n, rows)
275 when 'DIRECTSUM', 'UUV', 'PRODUCT'
276 check(a.n == b.n, 'bad length') if command == 'UUV'
277 product = command == 'PRODUCT'
278 n = product ? a.n * b.n : a.n + b.n
279 check_dimension(n, product ? a.k * b.k : a.k + b.k)
280 rows = if product
281 a.g.flat_map do |x|
282 b.g.map do |y|
283 a.n.times.reduce(0) { |row, i| (row << b.n) | ((x & bit(a.n, i)).zero? ? 0 : y) }
284 end
285 end
286 else
287 a.g.map { |row| (row << b.n) | (command == 'UUV' ? row : 0) } + b.g
288 end
289 Code.new(n, rows)
290 else
291 raise 'Not a defining command'
292 end
293end

295def distribution(words, offset, n)
296 counts = Array.new(n + 1, 0)
297 words.each { |word| counts[weight(word ^ offset)] += 1 }
298 counts
299end

301def weight_list(counts)
302 counts.each_index.filter_map { |i| "#{i}:#{counts[i]}" unless counts[i].zero? }.join(' ')
303end

305COMBINATIONS = Array.new(25) { Array.new(25, 0) }
306(0..24).each do |n|
307 COMBINATIONS[n][0] = COMBINATIONS[n][n] = 1
308 (1...n).each { |k| COMBINATIONS[n][k] = COMBINATIONS[n - 1][k - 1] + COMBINATIONS[n - 1][k] }
309 COMBINATIONS[n].freeze
310end

313def choose(n, k) = k < 0 || k > n ? 0 : COMBINATIONS[n][k]
314def volume(n, radius) = (0..radius).sum { |i| choose(n, i) }
315def krawtchouk(n, j, x)
316 (0..j).sum do |y|
317 term = choose(x, y) * choose(n - x, j - y)
318 y.even? ? term : -term
319 end
320end

322def matrix_text(label, rows, n) = rows.map { |row| "#{label} #{bits(row, n)}" }

324def systematic(name, code)
325 rows = code.g.dup
326 permutation = (1..code.n).to_a
327 code.k.times do |r|
328 column = r
329 found = nil
330 loop do
331 found = (r...rows.size).find { |i| (rows[i] & bit(code.n, column)) != 0 }
332 break if found
333 column += 1
334 end
335 if column != r
336 left = bit(code.n, r)
337 right = bit(code.n, column)
338 rows.map! { |row| ((row & left) != 0) != ((row & right) != 0) ? row ^ left ^ right : row }
339 permutation[r], permutation[column] = permutation[column], permutation[r]
340 end
341 rows[r], rows[found] = rows[found], rows[r]
342 rows.each_index { |i| rows[i] ^= rows[r] if i != r && (rows[i] & bit(code.n, r)) != 0 }
343 end
344 out = ["systematic #{name} perm #{permutation.join(' ')}"] + matrix_text('G', rows, code.n)
345 code.h.size.times do |j|
346 row = rows.reduce(0) { |value, x| (value << 1) | ((x & bit(code.n, code.k + j)).zero? ? 0 : 1) }
347 out << "H #{bits((row << code.h.size) | bit(code.h.size, j), code.n)}"
348 end
349 out
350end

352def independent(command, args)
353 if command == 'CRC' || command == 'CRCCHECK'
354 d = args[0].size - 1
355 check(args[0][0] == '1' && (1..16).cover?(d), 'bad parameter')
356 check(args[1].size + d <= 30, 'bad length') if command == 'CRC'
357 value = args[1].to_i(2) << (command == 'CRC' ? d : 0)
358 remainder = divide(value, args[0].to_i(2)).remainder
359 suffix = bits(remainder, d)
360 return [command == 'CRC' ? "crc remainder #{suffix} frame #{args[1]}#{suffix}" :
361 "crccheck #{remainder.zero? ? 'ok' : 'error'} remainder #{suffix}"]
362 end
363 n = args[0].to_i
364 if command == 'BOUNDS'
365 d = args[1].to_i
366 check(d >= 1 && d <= n && n <= 24, 'bad parameter')
367 ball = volume(n, (d - 1) / 2)
368 hamming = (0..n).select { |k| (1 << k) * ball <= (1 << n) }.last
369 sum = 0
370 griesmer = 0
371 (1..n).each do |k|
372 denominator = 1 << (k - 1)
373 sum += (d + denominator - 1) / denominator
374 griesmer = k if sum <= n
375 end
376 return ["bounds n #{n} d #{d} singleton #{n - d + 1} hamming #{hamming} griesmer #{griesmer}"]
377 end
378 check((1..24).cover?(n), 'bad parameter')
379 if command == 'FACTOR'
380 remaining = (1 << n) | 1
381 divisors = 1
382 factor = 3
383 terms = []
384 while 2 * (factor.bit_length - 1) <= remaining.bit_length - 1
385 exponent = 0
386 loop do
387 result = divide(remaining, factor)
388 break unless result.remainder.zero?
389 remaining = result.quotient
390 exponent += 1
391 end
392 if exponent != 0
393 terms << "#{factor.to_s(2)}^#{exponent}"
394 divisors *= exponent + 1
395 end
396 factor += 2
397 end
398 if remaining != 1
399 terms << "#{remaining.to_s(2)}^1"
400 divisors *= 2
401 end
402 return ["factor #{n} #{terms.join(' ')} divisors #{divisors}"]
403 end
404 check(args.size == n + 2, 'bad length')
405 check(args[1].to_i == 1, 'bad parameter')
406 coefficients = args.drop(1).map(&:to_i)
407 size = coefficients.sum
408 feasible = (size & (size - 1)).zero?
409 dual = (0..n).map do |j|
410 numerator = (0..n).sum { |x| coefficients[x] * krawtchouk(n, j, x) }
411 feasible = false if numerator.negative? || numerator % size != 0
412 divisor = numerator.gcd(size)
413 denominator = size / divisor
414 denominator == 1 ? (numerator / divisor).to_s : "#{numerator / divisor}/#{denominator}"
415 end
416 ["macwilliams n #{n} size #{size}", "dual #{dual.join(' ')}", "feasible #{yes(feasible)}"]
417end

419class Laboratory
420 def initialize
421 @codes = {}
422 end
423
424 def process(input)
425 words = input.split(' ')
426 command = words[0]
427 format = FORMATS[command]
428 raise CommandError, "unknown command #{command}" unless format
429 args = words.drop(1)
430 check((format.minimum..format.maximum).cover?(args.size), 'bad arguments')
431 args.each_with_index do |word, i|
432 check(valid_word?(word, format.types[[i, format.types.size - 1].min]), 'bad arguments')
433 end
434 a = b = nil
435 format.types.each_char.with_index do |kind, i|
436 next unless kind == 'A' || kind == 'B'
437 code = @codes[args[i]]
438 raise CommandError, "no code #{args[i]}" unless code
439 kind == 'A' ? a = code : b = code
440 end
441 if DEFINITIONS.include?(command)
442 code = construct(command, args, a, b)
443 status = @codes.key?(args[0]) ? 'redefined' : 'defined'
444 @codes[args[0]] = code
445 return "#{status} #{args[0]} n #{code.n} k #{code.k}\n"
446 end
447 return independent(command, args).join("\n") + "\n" if INDEPENDENT.include?(command)
448 if command == 'LIST'
449 out = ["codes #{@codes.size}"]
450 @codes.keys.sort.each do |name|
451 code = @codes[name]
452 out << "#{name} n #{code.n} k #{code.k} encoder #{code.encoder}"
453 end
454 return out.join("\n") + "\n"
455 end
456 name = args[0]
457 if command == 'DELETE'
458 @codes.delete(name)
459 return "deleted #{name}\n"
460 end
461 n = a.n
462 k = a.k
463 out = []
464 case command
465 when 'SHOW'
466 out << "code #{name} n #{n} k #{k} encoder #{a.encoder}"
467 if a.polynomial
468 quotient = divide((1 << n) | 1, a.polynomial).quotient
469 out << "poly g #{bits(a.polynomial, n - k + 1)} h #{bits(quotient, k + 1)}"
470 end
471 out.concat(matrix_text('G', a.g, n)).concat(matrix_text('H', a.h, n))
472 when 'INFO'
473 nonzero = a.words.drop(1)
474 distance = nonzero.map { |word| weight(word) }.min
475 generator = nonzero.min
476 t = (distance - 1) / 2
477 divisor = k.gcd(n)
478 cyclic = a.g.all? { |row| a.syndrome(((row << 1) & mask(n)) | (row >> (n - 1))).zero? }
479 out << "info #{name} n #{n} k #{k} rate #{k / divisor}/#{n / divisor} d #{distance} t #{t} " \
480 "perfect #{yes(volume(n, t) == (1 << (n - k)))} mds #{yes(distance == n - k + 1)} " \
481 "cyclic #{cyclic ? generator.to_s(2) : 'no'}"
482 when 'SYSTEMATIC'
483 out = systematic(name, a)
484 when 'WEIGHTS'
485 primary = distribution(a.words, 0, n)
486 dual = distribution(span(a.h), 0, n)
487 orthogonal = a.g.all? { |x| a.g.all? { |y| weight(x & y).even? } }
488 out << "weights #{name} #{weight_list(primary)}" << "dual #{name} #{weight_list(dual)}"
489 out << "duality #{name} self-orthogonal #{yes(orthogonal)} self-dual #{yes(orthogonal && n == 2 * k)} " \
490 "formally-self-dual #{yes(primary == dual)}"
491 when 'COVERING'
492 counts = distribution(a.leaders, 0, n)
493 radius = counts.rindex { |count| count != 0 }
494 out << "covering #{name} radius #{radius} leaders #{(0..radius).map { |i| "#{i}:#{counts[i]}" }.join(' ')}"
495 when 'BURST'
496 detect = a.words.drop(1).map { |word| word.bit_length - (word & -word).bit_length }.min
497 out << "burst #{name} detect #{detect} correct #{a.burst_correction}"
498 when 'TRELLIS'
499 profile = (0..n).map do |cut|
500 basis(a.g.map { |row| row >> (n - cut) }, cut).size +
501 basis(a.g.map { |row| row & mask(n - cut) }, n - cut).size - k
502 end
503 out << "trellis #{name} profile #{profile.join(' ')} max #{profile.max}"
504 when 'STDARRAY'
505 check(n - k <= 6 && k <= 4, 'too large')
506 leaders = a.leaders
507 syndromes = leaders.each_index.sort_by { |s| [weight(leaders[s]), leaders[s]] }
508 codewords = a.words.sort
509 out << "stdarray #{name} rows #{leaders.size} columns #{codewords.size}"
510 syndromes.each do |s|
511 out << "#{bits(s, n - k)}: #{codewords.map { |word| bits(leaders[s] ^ word, n) }.join(' ')}"
512 end
513 when 'EQUAL'
514 if n != b.n
515 out << "equal #{name} #{args[1]} no lengths #{n} #{b.n}"
516 else
517 common = k + b.k - basis(a.g + b.g, n).size
518 out << "equal #{name} #{args[1]} #{yes(k == common && b.k == common)} intersection #{common}"
519 end
520 when 'ENCODE'
521 check(args[1].size == k, 'bad length')
522 out << "encode #{name} message #{args[1]} codeword #{bits(a.encode(args[1].to_i(2)), n)}"
523 when 'SYNDROME', 'DECODE', 'BOUNDED', 'COSET'
524 text = args[command == 'BOUNDED' ? 2 : 1]
525 check(text.size == n, 'bad length')
526 word = text.to_i(2)
527 syndrome = a.syndrome(word)
528 if command == 'SYNDROME'
529 out << "syndrome #{name} #{bits(syndrome, n - k)} codeword #{yes(syndrome.zero?)}"
530 else
531 decoded = a.decode(word)
532 if command == 'COSET'
533 out << "coset #{name} syndrome #{bits(syndrome, n - k)} leader #{bits(decoded.error, n)} " \
534 "weights #{weight_list(distribution(a.words, word, n))}"
535 elsif command == 'BOUNDED' && weight(decoded.error) > args[1].to_i
536 out << "bounded #{name} fail weight #{weight(decoded.error)}"
537 else
538 out << "#{command == 'BOUNDED' ? 'bounded' : 'decode'} #{name} #{a.decoded_text(decoded)}"
539 end
540 end
541 when 'ERASURE'
542 check(args[1].size == n, 'bad length')
543 known = value = erased = 0
544 args[1].each_char.with_index do |c, i|
545 if c == '?'
546 erased += 1
547 else
548 known |= bit(n, i)
549 value |= bit(n, i) if c == '1'
550 end
551 end
552 candidates = a.words.select { |word| (word & known) == value }
553 text = "erasure #{name} erased #{erased} candidates #{candidates.size}"
554 if candidates.size == 1
555 text += " codeword #{bits(candidates[0], n)} message #{bits(a.message_for(candidates[0]), k)}"
556 end
557 out << text
558 when 'INTERLEAVE'
559 depth = args.size - 3
560 start = args[1].to_i
561 size = args[2].to_i
562 check(args.drop(3).all? { |message| message.size == k }, 'bad length')
563 check(start >= 1 && size >= 1 && start + size - 1 <= depth * n, 'bad burst')
564 sent_rows = args.drop(3).map { |message| a.encode(message.to_i(2)) }
565 received_rows = Array.new(depth, 0)
566 sent = +''
567 received = +''
568 n.times do |column|
569 depth.times do |i|
570 position = column * depth + i + 1
571 value = (sent_rows[i] & bit(n, column)) != 0
572 sent << (value ? '1' : '0')
573 value = !value if position >= start && position < start + size
574 received << (value ? '1' : '0')
575 received_rows[i] = (received_rows[i] << 1) | (value ? 1 : 0)
576 end
577 end
578 out << "interleave #{name} depth #{depth} burst #{start} #{size}" << "sent #{sent}" << "recv #{received}"
579 correct = 0
580 depth.times do |i|
581 decoded = a.decode(received_rows[i])
582 ok = decoded.message == args[i + 3].to_i(2)
583 correct += 1 if ok
584 out << "row #{i + 1} errors #{weight(received_rows[i] ^ sent_rows[i])} corrected #{weight(decoded.error)} " \
585 "message #{bits(decoded.message, k)} #{ok ? 'ok' : 'wrong'}"
586 end
587 out << "result ok #{correct} wrong #{depth - correct}"
588 end
589 out.join("\n") + "\n"
590 end
