48 use Command::*;

50 | Dual | Extend | Puncture | Shorten | Augment | Expurgate | Permute | Directsum | Uuv | Product)

54fn require(condition: bool, message: &str) -> Answer<()> {
55 if condition { Ok(()) } else { Err(message.to_owned()) }

58fn valid_word(text: &str, kind: u8) -> bool {
59 if text.is_empty() { return false; }
60 match kind {
61 b'N' | b'A' | b'B' => text.len() <= 8 && text.as_bytes()[0].is_ascii_alphabetic()
62 && text.bytes().all(|c| c.is_ascii_alphanumeric()),
63 b'I' => text.len() <= 4 && (text.len() == 1 || !text.starts_with('0'))
64 && text.bytes().all(|c| c.is_ascii_digit()),
65 b'V' => text.len() <= 30 && text.bytes().all(|c| matches!(c, b'0' | b'1')),
66 b'X' => text.len() <= 30 && text.bytes().all(|c| matches!(c, b'0' | b'1' | b'?')),
67 _ => false,
68 }
69}

71fn number(text: &str) -> usize { text.parse().unwrap() }
72fn parse_bits(text: &str) -> Word { Word::from_str_radix(text, 2).unwrap() }
73fn mask(n: usize) -> Word { (1 << n) - 1 }
74fn bit(n: usize, column: usize) -> Word { 1 << (n - column - 1) }
75fn bits(value: Word, n: usize) -> String { format!("{value:0width$b}", width = n) }
76fn length(value: Word) -> usize { (Word::BITS - value.leading_zeros()) as usize }
77fn yes(value: bool) -> &'static str { if value { "yes" } else { "no" } }

79fn divide(mut value: Word, divisor: Word) -> (Word, Word) {
80 let mut quotient = 0;
81 while value != 0 && length(value) >= length(divisor) {
82 let shift = length(value) - length(divisor);
83 quotient ^= 1 << shift;
84 value ^= divisor << shift;
85 }
86 (quotient, value)
87}

89fn basis(rows: &[Word], n: usize) -> Vec<Word> {
90 let mut pivots = [0; 32];
91 let mut result = Vec::new();
92 for &original in rows {
93 let mut row = original;
94 for column in (0..n).rev() {
95 if row & (1 << column) == 0 { continue; }
96 if pivots[column] != 0 { row ^= pivots[column]; }
97 else { pivots[column] = row; result.push(original); break; }
98 }
99 }
100 result
101}

103fn rref(source: &[Word], n: usize) -> (Vec<Word>, Vec<usize>) {
104 let mut rows = source.to_vec();
105 let mut pivots = Vec::new();
106 for column in 0..n {
107 let rank = pivots.len();
108 let Some(found) = (rank..rows.len()).find(|&i| rows[i] & bit(n, column) != 0) else { continue };
109 rows.swap(rank, found);
110 let pivot = rows[rank];
111 for (i, row) in rows.iter_mut().enumerate() {
112 if i != rank && *row & bit(n, column) != 0 { *row ^= pivot; }
113 }
114 pivots.push(column);
115 }
116 rows.truncate(pivots.len());
117 (rows, pivots)
118}
119
120fn null_space(source: &[Word], n: usize) -> Vec<Word> {
121 let (rows, pivots) = rref(source, n);
122 (0..n).filter(|column| !pivots.contains(column)).map(|column| {
123 let mut row = bit(n, column);
124 for (i, &pivot) in pivots.iter().enumerate() {
125 if rows[i] & bit(n, column) != 0 { row |= bit(n, pivot); }
126 }
127 row
128 }).collect()
129}

131fn span(rows: &[Word]) -> Vec<Word> {
132 let mut words = vec![0];
133 for &row in rows {
134 for i in 0..words.len() { words.push(words[i] ^ row); }
135 }
136 words
137}

139fn reed_muller(r: usize, m: usize) -> Vec<Word> {
140 if r == 0 { return vec![mask(1 << m)]; }
141 let half = 1 << (m - 1);
142 let mut rows: Vec<_> = reed_muller(r.min(m - 1), m - 1).into_iter()
143 .map(|row| (row << half) | row).collect();
144 rows.extend(reed_muller(r - 1, m - 1));
145 rows
146}

148fn check_dimension(n: usize, k: usize) -> Answer<()> {
149 if (1..=12).contains(&k) && n > k && n - k <= 12 { Ok(()) }
150 else { Err(format!("bad dimension {n} {k}")) }
151}

153struct Code {
154 n: usize,
155 g: Vec<Word>,
156 h: Vec<Word>,
157 polynomial: Option<Word>,
158}

160struct Decoded { word: Word, message: Word, error: Word }
161
162impl Code {
163 fn new(n: usize, g: Vec<Word>, h: Option<Vec<Word>>, polynomial: Option<Word>) -> Answer<Self> {
164 check_dimension(n, g.len())?;
165 let h = h.unwrap_or_else(|| null_space(&g, n));
166 Ok(Self { n, g, h, polynomial })
167 }
168
169 fn k(&self) -> usize { self.g.len() }
170 fn encoder(&self) -> &'static str { if self.polynomial.is_some() { "polynomial" } else { "matrix" } }
171
172 fn encode(&self, message: Word) -> Word {
173 if let Some(polynomial) = self.polynomial {
174 let shifted = message << self.h.len();
175 shifted | divide(shifted, polynomial).1
176 } else {
177 self.g.iter().enumerate().filter(|(i, _)| message & bit(self.k(), *i) != 0)
178 .fold(0, |value, (_, row)| value ^ row)
179 }
180 }
181
182 fn syndrome(&self, word: Word) -> Word {
183 self.h.iter().fold(0, |value, row| (value << 1) | ((word & row).count_ones() & 1))
184 }
185
186 fn message(&self, word: Word) -> Word {
187 (0..1 << self.k()).find(|&message| self.encode(message) == word).unwrap()
188 }
189
190 fn leaders(&self) -> Vec<Word> {
191 let mut result = vec![Word::MAX; 1 << self.h.len()];
192 result[0] = 0;
193 let mut found = 1;
194 for weight in 1..=self.n {
195 let mut word = mask(weight);
196 while word < 1 << self.n {
197 let syndrome = self.syndrome(word) as usize;
198 if result[syndrome] == Word::MAX {
199 result[syndrome] = word;
200 found += 1;
201 if found == result.len() { return result; }
202 }
203 let lowest = word & word.wrapping_neg();
204 let next = word + lowest;
205 word = next | (((next ^ word) >> 2) / lowest);
206 }
207 }
208 result
209 }
210
211 fn decode(&self, word: Word, leaders: &[Word]) -> Decoded {
212 let error = leaders[self.syndrome(word) as usize];
213 Decoded { word: word ^ error, message: self.message(word ^ error), error }
214 }
215
216 fn decoded_text(&self, decoded: &Decoded) -> String {
217 format!("codeword {} message {} error {} weight {}", bits(decoded.word, self.n),
218 bits(decoded.message, self.k()), bits(decoded.error, self.n), decoded.error.count_ones())
219 }
220
221 fn burst_correction(&self) -> usize {
222 let mut seen = vec![false; 1 << self.h.len()];
223 seen[0] = true;
224 for size in 1..=self.n {
225 let variants = if size == 1 { 1 } else { 1 << (size - 2) };
226 for offset in 0..=self.n - size {
227 for middle in 0..variants {
228 let word = if size == 1 { 1 } else { (1 << (size - 1)) | (middle << 1) | 1 };
229 let syndrome = self.syndrome(word << offset) as usize;
230 if seen[syndrome] { return size - 1; }
231 seen[syndrome] = true;
232 }
233 }
234 }
235 unreachable!()
236 }
237}

239fn construct(command: Command, args: &[&str], a: Option<&Code>, b: Option<&Code>) -> Answer<Code> {
240 use Command::*;
241 let n;
242 let mut g;
243 let mut h = None;
244 let mut polynomial = None;
245 match command {
246 Gen | Pcheck => {
247 n = args[1].len();
248 require(args[1..].iter().all(|word| word.len() == n), "bad length")?;
249 let rows = basis(&args[1..].iter().map(|word| parse_bits(word)).collect::<Vec<_>>(), n);
250 if command == Gen { g = rows; }
251 else { g = null_space(&rows, n); h = Some(rows); }
252 }
253 Hamming | Simplex => {
254 let r = number(args[1]);
255 require((2..=4).contains(&r), "bad parameter")?;
256 n = (1 << r) - 1;
257 let rows: Vec<_> = (0..r).map(|i| (1..=n).fold(0, |row, j|
258 (row << 1) | (((j >> (r - i - 1)) & 1) as Word))).collect();
259 if command == Simplex { g = rows; }
260 else { g = null_space(&rows, n); h = Some(rows); }
261 }
262 Repetition | Parity => {
263 n = number(args[1]);
264 require((2..=24).contains(&n), "bad parameter")?;
265 g = if command == Repetition { vec![mask(n)] }
266 else { (0..n - 1).map(|i| bit(n, i) | 1).collect() };
267 }
268 Rm => {
269 let r = number(args[1]);
270 let m = number(args[2]);
271 require((1..=4).contains(&m) && r <= m, "bad parameter")?;
272 n = 1 << m;
273 g = reed_muller(r, m);
274 }
275 Cyclic => {
276 n = number(args[1]);
277 let d = args[2].len() - 1;
278 require((2..=24).contains(&n) && args[2].starts_with('1') && d >= 1 && d < n, "bad parameter")?;
279 let divisor = parse_bits(args[2]);
280 require(divide((1 << n) | 1, divisor).1 == 0, "not a divisor")?;
281 let k = n - d;
282 g = (0..k).map(|i| divisor << (k - i - 1)).collect();
283 let mut check = vec![0; d];
284 for j in 0..n {
285 let remainder = divide(bit(n, j), divisor).1;
286 for (i, row) in check.iter_mut().enumerate() { *row = (*row << 1) | ((remainder >> (d - i - 1)) & 1); }
287 }
288 h = Some(check);
289 polynomial = Some(divisor);
290 }
291 Dual => { let a = a.unwrap(); n = a.n; g = a.h.clone(); h = Some(a.g.clone()); }
292 Extend => {
293 let a = a.unwrap(); n = a.n + 1;
294 g = a.g.iter().map(|row| (row << 1) | (row.count_ones() & 1)).collect();
295 }
296 Puncture | Shorten | Expurgate => {
297 let a = a.unwrap();
298 let position = if command == Expurgate { 0 } else { number(args[2]) };
299 if command != Expurgate { require((1..=a.n).contains(&position), "bad position")?; }
300 n = a.n - usize::from(command != Expurgate);
301 let selected = |row: Word| if command == Expurgate { row.count_ones() & 1 != 0 }
302 else { row & bit(a.n, position - 1) != 0 };
303 let removed = if command == Puncture { None } else { a.g.iter().position(|&row| selected(row)) };
304 g = Vec::new();
305 for (i, &original) in a.g.iter().enumerate() {
306 if Some(i) == removed { continue; }
307 let mut row = original;
308 if let Some(pivot) = removed { if selected(row) { row ^= a.g[pivot]; } }
309 if command != Expurgate {
310 let low = a.n - position;
311 row = ((row >> (low + 1)) << low) | (row & mask(low));
312 }
313 g.push(row);
314 }
315 if command == Puncture { g = basis(&g, n); }
316 }
317 Augment => {
318 let a = a.unwrap(); n = a.n;
319 g = a.g.clone(); g.push(mask(n)); g = basis(&g, n);
320 }
321 Permute => {
322 let a = a.unwrap(); n = a.n;
323 let permutation: Vec<_> = args[2..].iter().map(|word| number(word)).collect();
324 let mut sorted = permutation.clone(); sorted.sort_unstable();
325 require(sorted == (1..=n).collect::<Vec<_>>(), "bad permutation")?;
326 g = a.g.iter().map(|row| permutation.iter().fold(0, |result, &column|
327 (result << 1) | Word::from(row & bit(n, column - 1) != 0))).collect();
328 }
329 Directsum | Uuv | Product => {
330 let a = a.unwrap(); let b = b.unwrap();
331 if command == Uuv { require(a.n == b.n, "bad length")?; }
332 n = if command == Product { a.n * b.n } else { a.n + b.n };
333 check_dimension(n, if command == Product { a.k() * b.k() } else { a.k() + b.k() })?;
334 g = Vec::new();
335 if command == Product {
336 for &x in &a.g { for &y in &b.g {
337 g.push((0..a.n).fold(0, |row, column| (row << b.n) | if x & bit(a.n, column) != 0 { y } else { 0 }));
338 } }
339 } else {
340 g.extend(a.g.iter().map(|row| (row << b.n) | if command == Uuv { *row } else { 0 }));
341 g.extend_from_slice(&b.g);
342 }
343 }
344 _ => unreachable!(),
345 }
346 Code::new(n, g, h, polynomial)
347}

349fn distribution(words: impl IntoIterator<Item = Word>, n: usize) -> Vec<i64> {
350 let mut counts = vec![0; n + 1];
351 for word in words { counts[word.count_ones() as usize] += 1; }
352 counts
353}

360fn gcd(mut a: i64, mut b: i64) -> i64 {
361 a = a.abs();
362 while b != 0 { (a, b) = (b, a % b); }
363 a

366const fn binomials() -> [[i64; 25]; 25] {
367 let mut table = [[0; 25]; 25];
368 let mut n = 0;
369 while n <= 24 {
370 table[n][0] = 1; table[n][n] = 1;
371 let mut k = 1;
372 while k < n { table[n][k] = table[n - 1][k - 1] + table[n - 1][k]; k += 1; }
373 n += 1;
374 }
375 table
376}

378const COMBINATIONS: [[i64; 25]; 25] = binomials();
379fn choose(n: usize, k: isize) -> i64 { if k < 0 || k as usize > n { 0 } else { COMBINATIONS[n][k as usize] } }
380fn volume(n: usize, radius: usize) -> i64 { (0..=radius).map(|i| choose(n, i as isize)).sum() }
381
382fn krawtchouk(n: usize, j: usize, x: usize) -> i64 {
383 (0..=j).map(|y| {
384 let term = choose(x, y as isize) * choose(n - x, j as isize - y as isize);
385 if y % 2 == 0 { term } else { -term }
386 }).sum()
387}

393fn systematic(name: &str, code: &Code) -> String {
394 let mut rows = code.g.clone();
395 let mut permutation: Vec<_> = (1..=code.n).collect();
396 for r in 0..code.k() {
397 let column = (r..code.n).find(|&column| rows[r..].iter().any(|row| row & bit(code.n, column) != 0)).unwrap();
398 if column != r {
399 let left = bit(code.n, r); let right = bit(code.n, column);
400 for row in &mut rows { if (*row & left != 0) != (*row & right != 0) { *row ^= left | right; } }
401 permutation.swap(r, column);
402 }
403 let found = (r..code.k()).find(|&i| rows[i] & bit(code.n, r) != 0).unwrap();
404 rows.swap(r, found);
405 let pivot = rows[r];
406 for (i, row) in rows.iter_mut().enumerate() { if i != r && *row & bit(code.n, r) != 0 { *row ^= pivot; } }
407 }
408 let mut out = format!("systematic {name} perm");
409 for column in permutation { write!(out, " {column}").unwrap(); }
410 out.push('\n');
411 matrix_text(&mut out, 'G', &rows, code.n);
412 for j in 0..code.h.len() {
413 let row = rows.iter().fold(0, |value, row| (value << 1) | Word::from(row & bit(code.n, code.k() + j) != 0));
414 writeln!(out, "H {}", bits((row << code.h.len()) | bit(code.h.len(), j), code.n)).unwrap();
415 }
416 out
417}

419fn independent(command: Command, args: &[&str]) -> Answer<String> {
420 use Command::*;
421 if matches!(command, Crc | Crccheck) {
422 let d = args[0].len() - 1;
423 require(args[0].starts_with('1') && (1..=16).contains(&d), "bad parameter")?;
424 if command == Crc { require(args[1].len() + d <= 30, "bad length")?; }
425 let word = parse_bits(args[1]); let divisor = parse_bits(args[0]);
426 let remainder = divide(if command == Crc { word << d } else { word }, divisor).1;
427 let suffix = bits(remainder, d);
428 return Ok(if command == Crc { format!("crc remainder {suffix} frame {}{suffix}\n", args[1]) }
429 else { format!("crccheck {} remainder {suffix}\n", if remainder == 0 { "ok" } else { "error" }) });
430 }
431 let n = number(args[0]);
432 if command == Bounds {
433 let d = number(args[1]);
434 require(d >= 1 && d <= n && n <= 24, "bad parameter")?;
435 let v = volume(n, (d - 1) / 2);
436 let hamming = (0..=n).filter(|&k| (1i64 << k) * v <= 1i64 << n).last().unwrap();
437 let mut sum = 0; let mut griesmer = 0;
438 for k in 1..=n { sum += d.div_ceil(1 << (k - 1)); if sum <= n { griesmer = k; } }
439 return Ok(format!("bounds n {n} d {d} singleton {} hamming {hamming} griesmer {griesmer}\n", n - d + 1));
440 }
441 require((1..=24).contains(&n), "bad parameter")?;
442 if command == Factor {
443 let mut remaining = (1 << n) | 1;
444 let mut factor = 3; let mut divisors = 1;
445 let mut out = format!("factor {n}");
446 while 2 * (length(factor) - 1) <= length(remaining) - 1 {
447 let mut exponent = 0;
448 loop {
449 let (quotient, remainder) = divide(remaining, factor);
450 if remainder != 0 { break; }
451 exponent += 1; remaining = quotient;
452 }
453 if exponent != 0 { write!(out, " {}^{exponent}", bits(factor, length(factor))).unwrap(); divisors *= exponent + 1; }
454 factor += 2;
455 }
456 if remaining != 1 { write!(out, " {}^1", bits(remaining, length(remaining))).unwrap(); divisors *= 2; }
457 writeln!(out, " divisors {divisors}").unwrap();
458 return Ok(out);
459 }
460 require(args.len() == n + 2, "bad length")?;
461 require(number(args[1]) == 1, "bad parameter")?;
462 let counts: Vec<_> = args[1..].iter().map(|word| number(word) as i64).collect();
463 let size: i64 = counts.iter().sum();
464 let mut feasible = size & (size - 1) == 0;
465 let mut out = format!("macwilliams n {n} size {size}\ndual");
466 for j in 0..=n {
467 let numerator: i64 = (0..=n).map(|x| counts[x] * krawtchouk(n, j, x)).sum();
468 if numerator < 0 || numerator % size != 0 { feasible = false; }
469 let divisor = gcd(numerator, size); let denominator = size / divisor;
470 write!(out, " {}", numerator / divisor).unwrap();
471 if denominator != 1 { write!(out, "/{denominator}").unwrap(); }
472 }
473 writeln!(out, "\nfeasible {}", yes(feasible)).unwrap();
474 Ok(out)
475}

477#[derive(Default)]
478struct Laboratory { codes: BTreeMap<String, Code> }
479
480impl Laboratory {
481 fn process(&mut self, line: &str) -> Answer<String> {
482 use Command::*;
483 let words: Vec<_> = line.split(' ').collect();
484 let (command, types, minimum, maximum) = Command::format(words[0])
485 .ok_or_else(|| format!("unknown command {}", words[0]))?;
486 let args = &words[1..];
487 require((minimum..=maximum).contains(&args.len()), "bad arguments")?;
488 for (i, text) in args.iter().enumerate() {
489 require(valid_word(text, types[i.min(types.len() - 1)]), "bad arguments")?;
490 }
491 let mut a = None; let mut b = None;
492 for (i, &kind) in types.iter().enumerate() {
493 if matches!(kind, b'A' | b'B') {
494 let code = self.codes.get(args[i]).ok_or_else(|| format!("no code {}", args[i]))?;
495 if kind == b'A' { a = Some(code); } else { b = Some(code); }
496 }
497 }
498 if command.defining() {
499 let code = construct(command, args, a, b)?;
500 let status = if self.codes.contains_key(args[0]) { "redefined" } else { "defined" };
501 let out = format!("{status} {} n {} k {}\n", args[0], code.n, code.k());
502 self.codes.insert(args[0].to_owned(), code);
503 return Ok(out);
504 }
505 if matches!(command, Crc | Crccheck | Factor | Macwilliams | Bounds) { return independent(command, args); }
506 if command == List {
507 let mut out = format!("codes {}\n", self.codes.len());
508 for (name, code) in &self.codes { writeln!(out, "{name} n {} k {} encoder {}", code.n, code.k(), code.encoder()).unwrap(); }
509 return Ok(out);
510 }
511 let name = args[0];
512 if command == Delete { self.codes.remove(name); return Ok(format!("deleted {name}\n")); }
513 let code = a.unwrap(); let n = code.n; let k = code.k();
514 let mut out = String::new();
515 match command {
516 Show => {
517 writeln!(out, "code {name} n {n} k {k} encoder {}", code.encoder()).unwrap();
518 if let Some(polynomial) = code.polynomial { writeln!(out, "poly g {} h {}", bits(polynomial, n - k + 1), bits(divide((1 << n) | 1, polynomial).0, k + 1)).unwrap(); }
519 matrix_text(&mut out, 'G', &code.g, n); matrix_text(&mut out, 'H', &code.h, n);
520 }
521 Info => {
522 let words = span(&code.g);
523 let distance = words[1..].iter().map(|word| word.count_ones() as usize).min().unwrap();
524 let t = (distance - 1) / 2; let divisor = gcd(k as i64, n as i64) as usize;
525 let cyclic = code.g.iter().all(|&row| code.syndrome(((row << 1) & mask(n)) | (row >> (n - 1))) == 0);
526 let generator = *words[1..].iter().min().unwrap();
527 let generator = if cyclic { bits(generator, length(generator)) } else { "no".into() };
528 writeln!(out, "info {name} n {n} k {k} rate {}/{} d {distance} t {t} perfect {} mds {} cyclic {generator}",
529 k / divisor, n / divisor, yes(volume(n, t) == 1 << (n - k)), yes(distance == n - k + 1)).unwrap();
530 }
531 Systematic => return Ok(systematic(name, code)),
532 Weights => {
533 let primary = distribution(span(&code.g), n); let dual = distribution(span(&code.h), n);
534 let orthogonal = code.g.iter().all(|x| code.g.iter().all(|y| (x & y).count_ones() & 1 == 0));
535 writeln!(out, "weights {name} {}\ndual {name} {}", weight_list(&primary), weight_list(&dual)).unwrap();
536 writeln!(out, "duality {name} self-orthogonal {} self-dual {} formally-self-dual {}",
537 yes(orthogonal), yes(orthogonal && n == 2 * k), yes(primary == dual)).unwrap();
538 }
539 Covering => {
540 let counts = distribution(code.leaders(), n);
541 let radius = counts.iter().rposition(|&count| count != 0).unwrap();
542 write!(out, "covering {name} radius {radius} leaders").unwrap();
543 for (i, count) in counts.iter().enumerate().take(radius + 1) { write!(out, " {i}:{count}").unwrap(); }
544 out.push('\n');
545 }
546 Burst => {
547 let detect = span(&code.g).into_iter().filter(|&word| word != 0)
548 .map(|word| length(word) - word.trailing_zeros() as usize - 1).min().unwrap();
549 writeln!(out, "burst {name} detect {detect} correct {}", code.burst_correction()).unwrap();
550 }
551 Trellis => {
552 write!(out, "trellis {name} profile").unwrap();
553 let mut maximum = 0;
554 for cut in 0..=n {
555 let right = n - cut;
556 let prefix: Vec<_> = code.g.iter().map(|row| row >> right).collect();
557 let suffix: Vec<_> = code.g.iter().map(|row| row & mask(right)).collect();
558 let states = basis(&prefix, cut).len() + basis(&suffix, right).len() - k;
559 maximum = maximum.max(states); write!(out, " {states}").unwrap();
560 }
561 writeln!(out, " max {maximum}").unwrap();
562 }
563 Stdarray => {
564 require(n - k <= 6 && k <= 4, "too large")?;
565 let leaders = code.leaders();
566 let mut syndromes: Vec<_> = (0..leaders.len()).collect();
567 syndromes.sort_unstable_by_key(|&s| (leaders[s].count_ones(), leaders[s]));
568 let mut words = span(&code.g); words.sort_unstable();
569 writeln!(out, "stdarray {name} rows {} columns {}", leaders.len(), words.len()).unwrap();
570 for s in syndromes {
571 write!(out, "{}:", bits(s as Word, n - k)).unwrap();
572 for &word in &words { write!(out, " {}", bits(leaders[s] ^ word, n)).unwrap(); }
573 out.push('\n');
574 }
575 }
576 Equal => {
577 let other = b.unwrap();
578 if n != other.n { writeln!(out, "equal {name} {} no lengths {n} {}", args[1], other.n).unwrap(); }
579 else {
580 let mut rows = code.g.clone(); rows.extend_from_slice(&other.g);
581 let common = k + other.k() - basis(&rows, n).len();
582 writeln!(out, "equal {name} {} {} intersection {common}", args[1], yes(k == common && other.k() == common)).unwrap();
583 }
584 }
585 Encode => {
586 require(args[1].len() == k, "bad length")?;
587 writeln!(out, "encode {name} message {} codeword {}", args[1], bits(code.encode(parse_bits(args[1])), n)).unwrap();
588 }
589 Syndrome | Decode | Bounded | Coset => {
590 let text = args[if command == Bounded { 2 } else { 1 }];
591 require(text.len() == n, "bad length")?;
592 let word = parse_bits(text); let s = code.syndrome(word);
593 if command == Syndrome { writeln!(out, "syndrome {name} {} codeword {}", bits(s, n - k), yes(s == 0)).unwrap(); }
594 else {
595 let leaders = code.leaders(); let decoded = code.decode(word, &leaders);
596 if command == Coset {
597 let counts = distribution(span(&code.g).into_iter().map(|c| c ^ word), n);
598 writeln!(out, "coset {name} syndrome {} leader {} weights {}", bits(s, n - k), bits(decoded.error, n), weight_list(&counts)).unwrap();
599 } else if command == Bounded && decoded.error.count_ones() as usize > number(args[1]) {
600 writeln!(out, "bounded {name} fail weight {}", decoded.error.count_ones()).unwrap();
601 } else {
602 writeln!(out, "{} {name} {}", if command == Bounded { "bounded" } else { "decode" }, code.decoded_text(&decoded)).unwrap();
603 }
604 }
605 }
606 Erasure => {
607 require(args[1].len() == n, "bad length")?;
608 let mut known = 0; let mut value = 0; let mut erased = 0;
609 for (i, digit) in args[1].bytes().enumerate() {
610 if digit == b'?' { erased += 1; }
611 else { known |= bit(n, i); if digit == b'1' { value |= bit(n, i); } }
612 }
613 let candidates: Vec<_> = span(&code.g).into_iter().filter(|word| word & known == value).collect();
614 write!(out, "erasure {name} erased {erased} candidates {}", candidates.len()).unwrap();
615 if candidates.len() == 1 { write!(out, " codeword {} message {}", bits(candidates[0], n), bits(code.message(candidates[0]), k)).unwrap(); }
616 out.push('\n');
617 }
618 Interleave => {
619 let depth = args.len() - 3; let start = number(args[1]); let size = number(args[2]);
620 require(args[3..].iter().all(|word| word.len() == k), "bad length")?;
621 require(start >= 1 && size >= 1 && start + size - 1 <= depth * n, "bad burst")?;
622 let sent_rows: Vec<_> = args[3..].iter().map(|word| code.encode(parse_bits(word))).collect();
623 let mut received_rows = vec![0; depth];
624 let mut sent = String::new(); let mut received = String::new();
625 for column in 0..n { for i in 0..depth {
626 let position = column * depth + i + 1;
627 let value = sent_rows[i] & bit(n, column) != 0;
628 sent.push(if value { '1' } else { '0' });
629 let changed = value ^ (position >= start && position < start + size);
630 received.push(if changed { '1' } else { '0' });
631 received_rows[i] = (received_rows[i] << 1) | Word::from(changed);
632 } }
633 writeln!(out, "interleave {name} depth {depth} burst {start} {size}\nsent {sent}\nrecv {received}").unwrap();
634 let leaders = code.leaders(); let mut correct = 0;
635 for i in 0..depth {
636 let decoded = code.decode(received_rows[i], &leaders);
637 let ok = decoded.message == parse_bits(args[i + 3]); correct += usize::from(ok);
638 writeln!(out, "row {} errors {} corrected {} message {} {}", i + 1,
639 (sent_rows[i] ^ received_rows[i]).count_ones(), decoded.error.count_ones(), bits(decoded.message, k), if ok { "ok" } else { "wrong" }).unwrap();
640 }
641 writeln!(out, "result ok {correct} wrong {}", depth - correct).unwrap();
642 }
643 _ => unreachable!(),
644 }
645 Ok(out)
646 }
647}

649fn main() -> io::Result<()> {
650 let stdin = io::stdin();
651 let mut lines = stdin.lock().lines();
652 let count: usize = lines.next().unwrap()?.parse().unwrap();
653 let mut laboratory = Laboratory::default();
654 let stdout = io::stdout();
655 let mut output = io::BufWriter::new(stdout.lock());
656 for line in lines.take(count) {
657 match laboratory.process(&line?) {
658 Ok(text) => output.write_all(text.as_bytes())?,
659 Err(error) => writeln!(output, "error: {error}")?,
660 }
661 }
662 output.flush()
663}
