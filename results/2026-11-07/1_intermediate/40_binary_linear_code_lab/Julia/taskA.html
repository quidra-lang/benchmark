85function valid_word(text::String, kind::Char)
86 pattern = if kind in ('N', 'A', 'B')
87 r"^[A-Za-z][A-Za-z0-9]{0,7}$"
88 elseif kind == 'I'
89 r"^(0|[1-9][0-9]{0,3})$"
90 elseif kind == 'X'
91 r"^[01?]{1,30}$"
92 else
93 r"^[01]{1,30}$"
94 end
95 occursin(pattern, text)
96end

98power2(exponent::Int) = 1 << exponent
99word_mask(size::Int) = (one(Word) << size) - one(Word)
100bit_at(size::Int, column::Int) = one(Word) << (size - column)
101parse_bits(text::String) = parse(Word, text; base = 2)
102weight(word::Word) = count_ones(word)
103word_length(word::Word) = 32 - leading_zeros(word)
104bits(value::Integer, size::Int) = string(value; base = 2, pad = size)
105yes(value::Bool) = value ? "yes" : "no"
106joined(values) = join(values, " ")
107matrix_text(label::String, rows::Vector{Word}, size::Int) = ["$label $(bits(row, size))" for row in rows]

109function poly_divide(value::Word, divisor::Word)
110 quotient = zero(Word)
111 while value != 0 && word_length(value) >= word_length(divisor)
112 shift = word_length(value) - word_length(divisor)
113 quotient ⊻= one(Word) << shift
114 value ⊻= divisor << shift
115 end
116 (; quotient, remainder = value)
117end

119function basis(source::Vector{Word}, size::Int)
120 pivots = zeros(Word, size)
121 result = Word[]
122 for original in source
123 row = original
124 for column in 1:size
125 row & bit_at(size, column) == 0 && continue
126 if pivots[column] != 0
127 row ⊻= pivots[column]
128 else
129 pivots[column] = row
130 push!(result, original)
131 break
132 end
133 end
134 end
135 result
136end

138function rref(source::Vector{Word}, size::Int)
139 rows = copy(source)
140 columns = Int[]
141 for column in 1:size
142 pivot = length(columns) + 1
143 found = findfirst(i -> i >= pivot && rows[i] & bit_at(size, column) != 0, eachindex(rows))
144 isnothing(found) && continue
145 rows[pivot], rows[found] = rows[found], rows[pivot]
146 for i in eachindex(rows)
147 i != pivot && rows[i] & bit_at(size, column) != 0 && (rows[i] ⊻= rows[pivot])
148 end
149 push!(columns, column)
150 end
151 (; rows = rows[1:length(columns)], columns)
152end

154function null_space(source::Vector{Word}, size::Int)
155 reduced = rref(source, size)
156 result = Word[]
157 for column in setdiff(1:size, reduced.columns)
158 row = bit_at(size, column)
159 for (index, pivot) in enumerate(reduced.columns)
160 reduced.rows[index] & bit_at(size, column) != 0 && (row |= bit_at(size, pivot))
161 end
162 push!(result, row)
163 end
164 result
165end

175function reed_muller(order::Int, variables::Int)
176 order == 0 && return Word[word_mask(power2(variables))]
177 half = power2(variables - 1)
178 first = reed_muller(min(order, variables - 1), variables - 1)
179 vcat((first .<< half) .| first, reed_muller(order - 1, variables - 1))
180end

192function make_code(size::Int, generator::Vector{Word}; parity::Union{Nothing, Vector{Word}} = nothing,

200function encode(code::Code, message::Word)
201 if !isnothing(code.polynomial)
202 shifted = message << (code.n - dimension(code))
203 return shifted | poly_divide(shifted, code.polynomial).remainder
204 end
205 result = zero(Word)
206 for (column, row) in enumerate(code.g)
207 message & bit_at(dimension(code), column) != 0 && (result ⊻= row)
208 end
209 result
210end
211encoded_words(code::Code) = isnothing(code.polynomial) ? span(reverse(code.g)) :
212 [encode(code, Word(message)) for message in 0:power2(dimension(code))-1]

214function syndrome(code::Code, word::Word)
215 result = zero(Word)
216 for row in code.h
217 result = (result << 1) | Word(isodd(weight(row & word)))
218 end
219 result
220end

222function leaders(code::Code)
223 result = fill(typemax(Word), power2(code.n - dimension(code)))
224 result[1] = 0
225 found = 1
226 for size in 1:code.n
227 word = word_mask(size)
228 while word < one(Word) << code.n
229 state = Int(syndrome(code, word)) + 1
230 if result[state] == typemax(Word)
231 result[state] = word
232 found += 1
233 found == length(result) && return result
234 end
235 lowest = word & -word
236 following = word + lowest
237 word = following | ((following ⊻ word) >> 2) ÷ lowest
238 end
239 end
240 result
241end

248function decode(code::Code, received::Word, errors::Vector{Word}, encoding::Vector{Word})
249 pattern = errors[Int(syndrome(code, received)) + 1]
250 word = received ⊻ pattern
251 Decoded(word, something(findfirst(==(word), encoding)) - 1, pattern)
252end
253decoded_text(code::Code, result::Decoded) =
254 "codeword $(bits(result.word, code.n)) message $(bits(result.message, dimension(code))) error $(bits(result.error_pattern, code.n)) weight $(weight(result.error_pattern))"

256function burst_correction(code::Code)
257 seen = falses(power2(code.n - dimension(code)))
258 seen[1] = true
259 for size in 1:code.n
260 variants = size == 1 ? 1 : power2(size - 2)
261 for offset in 0:code.n-size, middle in 0:variants-1
262 word = size == 1 ? one(Word) : (one(Word) << (size - 1)) | (Word(middle) << 1) | one(Word)
263 state = Int(syndrome(code, word << offset)) + 1
264 seen[state] && return size - 1
265 seen[state] = true
266 end
267 end
268 0
269end

271function construct(command::CommandKind, args::Vector{String}, sources::Vector{Code})
272 if command in (GEN, PCHECK)
273 size = length(args[2])
274 ensure(all(text -> length(text) == size, args[2:end]), "bad length")
275 rows = basis(parse_bits.(args[2:end]), size)
276 return command == GEN ? make_code(size, rows) : make_code(size, null_space(rows, size); parity = rows)
277 elseif command in (HAMMING, SIMPLEX)
278 order = parse(Int, args[2])
279 ensure(2 <= order <= 4, "bad parameter")
280 size = power2(order) - 1
281 rows = zeros(Word, order)
282 for column in 1:size, row in 1:order
283 rows[row] = (rows[row] << 1) | Word((column >> (order - row)) & 1)
284 end
285 return command == SIMPLEX ? make_code(size, rows) : make_code(size, null_space(rows, size); parity = rows)
286 elseif command in (REPETITION, PARITY)
287 size = parse(Int, args[2])
288 ensure(2 <= size <= 24, "bad parameter")
289 rows = command == REPETITION ? Word[word_mask(size)] : [bit_at(size, i) | one(Word) for i in 1:size-1]
290 return make_code(size, rows)
291 elseif command == RM
292 order, variables = parse.(Int, args[2:3])
293 ensure(1 <= variables <= 4 && order <= variables, "bad parameter")
294 return make_code(power2(variables), reed_muller(order, variables))
295 elseif command == CYCLIC
296 size = parse(Int, args[2])
297 degree = length(args[3]) - 1
298 ensure(2 <= size <= 24 && first(args[3]) == '1' && 1 <= degree < size, "bad parameter")
299 polynomial = parse_bits(args[3])
300 ensure(poly_divide((one(Word) << size) | one(Word), polynomial).remainder == 0, "not a divisor")
301 count = size - degree
302 rows = [polynomial << (count - i) for i in 1:count]
303 checks = zeros(Word, degree)
304 for column in 1:size
305 residue = poly_divide(bit_at(size, column), polynomial).remainder
306 for row in 1:degree
307 checks[row] = (checks[row] << 1) | Word(residue & bit_at(degree, row) != 0)
308 end
309 end
310 return make_code(size, rows; parity = checks, polynomial = polynomial)
311 end
312
313 source = first(sources)
314 size = source.n
315 command == DUAL && return make_code(size, source.h; parity = source.g)
316 command == EXTEND && return make_code(size + 1, [(row << 1) | Word(isodd(weight(row))) for row in source.g])
317 if command in (PUNCTURE, SHORTEN, EXPURGATE)
318 expurgating = command == EXPURGATE
319 position = expurgating ? 0 : parse(Int, args[3])
320 ensure(expurgating || 1 <= position <= size, "bad position")
321 selected(row) = expurgating ? isodd(weight(row)) : row & bit_at(size, position) != 0
322 removed = command == PUNCTURE ? nothing : findfirst(selected, source.g)
323 rows = Word[]
324 for (index, original) in enumerate(source.g)
325 index == removed && continue
326 row = !isnothing(removed) && selected(original) ? original ⊻ source.g[removed] : original
327 if !expurgating
328 low = size - position
329 row = ((row >> (low + 1)) << low) | (row & word_mask(low))
330 end
331 push!(rows, row)
332 end
333 expurgating || (size -= 1)
334 return make_code(size, command == PUNCTURE ? basis(rows, size) : rows)
335 elseif command == AUGMENT
336 return make_code(size, basis(vcat(source.g, word_mask(size)), size))
337 elseif command == PERMUTE
338 permutation = parse.(Int, args[3:end])
339 ensure(sort(permutation) == collect(1:size), "bad permutation")
340 rows = Word[]
341 for row in source.g
342 result = zero(Word)
343 for column in permutation
344 result = (result << 1) | Word(row & bit_at(size, column) != 0)
345 end
346 push!(rows, result)
347 end
348 return make_code(size, rows)
349 end
350
351 other = sources[2]
352 ensure(command != UUV || size == other.n, "bad length")
353 product = command == PRODUCT
354 combined = product ? size * other.n : size + other.n
355 count = product ? dimension(source) * dimension(other) : dimension(source) + dimension(other)
356 check_dimension(combined, count)
357 rows = Word[]
358 if product
359 for left in source.g, right in other.g
360 row = zero(Word)
361 for column in 1:size
362 row = (row << other.n) | (left & bit_at(size, column) != 0 ? right : zero(Word))
363 end
364 push!(rows, row)
365 end
366 else
367 append!(rows, [(row << other.n) | (command == UUV ? row : zero(Word)) for row in source.g])
368 append!(rows, other.g)
369 end
370 make_code(combined, rows)
371end

373const COMBINATIONS = let table = zeros(Int64, 25, 25)
374 for n in 0:24
375 table[n + 1, 1] = 1
376 for k in 1:n
377 table[n + 1, k + 1] = table[n, k] + table[n, k + 1]
378 end
379 end
380 table

382choose_count(size::Int, selected::Int) = 0 <= selected <= size ? COMBINATIONS[size + 1, selected + 1] : Int64(0)
383volume(size::Int, radius::Int) = sum(@view COMBINATIONS[size + 1, 1:radius+1])
384krawtchouk(size::Int, index::Int, argument::Int) =
385 sum((iseven(selected) ? 1 : -1) * choose_count(argument, selected) * choose_count(size - argument, index - selected)
386 for selected in 0:index)
387function fraction_text(numerator::Integer, denominator::Integer)
388 common = gcd(numerator, denominator)
389 top, bottom = numerator ÷ common, denominator ÷ common
390 bottom == 1 ? string(top) : "$top/$bottom"
391end
392function distribution(words::Vector{Word}, size::Int, offset::Word = zero(Word))
393 counts = zeros(Int, size + 1)
394 for word in words
395 counts[weight(word ⊻ offset) + 1] += 1
396 end
397 counts
398end
399weight_items(counts::Vector{Int}) = joined(["$(index - 1):$count" for (index, count) in enumerate(counts) if count != 0])

401function systematic(name::String, code::Code)
402 rows = copy(code.g)
403 n, k = code.n, dimension(code)
404 permutation = collect(1:n)
405 for pivot in 1:k
406 column = pivot
407 found = nothing
408 while isnothing(found)
409 found = findfirst(i -> i >= pivot && rows[i] & bit_at(n, column) != 0, eachindex(rows))
410 isnothing(found) && (column += 1)
411 end
412 if column != pivot
413 left, right = bit_at(n, pivot), bit_at(n, column)
414 for i in eachindex(rows)
415 (rows[i] & left != 0) != (rows[i] & right != 0) && (rows[i] ⊻= left | right)
416 end
417 permutation[pivot], permutation[column] = permutation[column], permutation[pivot]
418 end
419 rows[pivot], rows[found] = rows[found], rows[pivot]
420 for i in eachindex(rows)
421 i != pivot && rows[i] & bit_at(n, pivot) != 0 && (rows[i] ⊻= rows[pivot])
422 end
423 end
424 checks = Word[]
425 for column in 1:n-k
426 row = zero(Word)
427 for original in rows
428 row = (row << 1) | Word(original & bit_at(n, k + column) != 0)
429 end
430 push!(checks, (row << (n - k)) | bit_at(n - k, column))
431 end
432 vcat(["systematic $name perm $(joined(permutation))"], matrix_text("G", rows, n), matrix_text("H", checks, n))
433end

435function inspect_code(command::CommandKind, args::Vector{String}, sources::Vector{Code})
436 code = first(sources)
437 name, n, k = args[1], code.n, dimension(code)
438 if command == SHOW
439 output = ["code $name n $n k $k encoder $(encoder(code))"]
440 if !isnothing(code.polynomial)
441 quotient = poly_divide((one(Word) << n) | one(Word), code.polynomial).quotient
442 push!(output, "poly g $(bits(code.polynomial, n - k + 1)) h $(bits(quotient, k + 1))")
443 end
444 return vcat(output, matrix_text("G", code.g, n), matrix_text("H", code.h, n))
445 elseif command == SYSTEMATIC
446 return systematic(name, code)
447 elseif command == EQUAL
448 other = sources[2]
449 n != other.n && return ["equal $name $(args[2]) no lengths $n $(other.n)"]
450 common = k + dimension(other) - length(basis(vcat(code.g, other.g), n))
451 return ["equal $name $(args[2]) $(yes(k == common == dimension(other))) intersection $common"]
452 elseif command == TRELLIS
453 profile = [length(basis(code.g .>> (n - column), column)) +
454 length(basis(code.g .& word_mask(n - column), n - column)) - k for column in 0:n]
455 return ["trellis $name profile $(joined(profile)) max $(maximum(profile))"]
456 elseif command == COVERING
457 counts = distribution(leaders(code), n)
458 radius = something(findlast(!=(0), counts)) - 1
459 items = ["$i:$(counts[i + 1])" for i in 0:radius]
460 return ["covering $name radius $radius leaders $(joined(items))"]
461 end
462 ensure(command != STDARRAY || (n - k <= 6 && k <= 4), "too large")
463 words = span(code.g)
464 if command == INFO
465 distance = minimum(weight, @view words[2:end])
466 correction = (distance - 1) ÷ 2
467 rotated = ((code.g .<< 1) .& word_mask(n)) .| (code.g .>> (n - 1))
468 first_word = minimum(@view words[2:end])
469 cyclic = all(row -> syndrome(code, row) == 0, rotated) ? bits(first_word, word_length(first_word)) : "no"
470 return ["info $name n $n k $k rate $(fraction_text(k, n)) d $distance t $correction perfect $(yes(volume(n, correction) == power2(n - k))) mds $(yes(distance == n - k + 1)) cyclic $cyclic"]
471 elseif command == WEIGHTS
472 primal, dual = distribution(words, n), distribution(span(code.h), n)
473 orthogonal = all(iseven(weight(left & right)) for left in code.g, right in code.g)
474 return ["weights $name $(weight_items(primal))", "dual $name $(weight_items(dual))",
475 "duality $name self-orthogonal $(yes(orthogonal)) self-dual $(yes(orthogonal && n == 2k)) formally-self-dual $(yes(primal == dual))"]
476 elseif command == BURST
477 detection = minimum(word -> word_length(word) - word_length(word & -word), @view words[2:end])
478 return ["burst $name detect $detection correct $(burst_correction(code))"]
479 end
480 errors = leaders(code)
481 ordering = sortperm(errors; by = word -> (weight(word), word))
482 sort!(words)
483 output = ["stdarray $name rows $(length(errors)) columns $(length(words))"]
484 for state in ordering
485 row = [bits(errors[state] ⊻ word, n) for word in words]
486 push!(output, "$(bits(state - 1, n - k)): $(joined(row))")
487 end
488 output
489end

491function encoding_report(command::CommandKind, args::Vector{String}, code::Code)
492 name, n, k = args[1], code.n, dimension(code)
493 if command == INTERLEAVE
494 messages = args[4:end]
495 ensure(all(message -> length(message) == k, messages), "bad length")
496 start, burst_length = parse.(Int, args[2:3])
497 depth = length(messages)
498 ensure(start >= 1 && burst_length >= 1 && start + burst_length - 1 <= depth * n, "bad burst")
499 encoded = [encode(code, parse_bits(message)) for message in messages]
500 sent_matrix = [word & bit_at(n, column) != 0 for word in encoded, column in 1:n]
501 sent = vec(sent_matrix)
502 received = copy(sent)
503 selected = start:start+burst_length-1
504 received[selected] .= .!received[selected]
505 received_matrix = reshape(received, depth, n)
506 received_words = zeros(Word, depth)
507 for column in 1:n, row in 1:depth
508 received_words[row] = (received_words[row] << 1) | Word(received_matrix[row, column])
509 end
510 output = ["interleave $name depth $depth burst $start $burst_length",
511 "sent $(join(Int.(sent)))", "recv $(join(Int.(received)))"]
512 errors, encoding = leaders(code), encoded_words(code)
513 correct = 0
514 for row in 1:depth
515 result = decode(code, received_words[row], errors, encoding)
516 success = result.message == parse_bits(messages[row])
517 correct += success
518 push!(output, "row $row errors $(weight(received_words[row] ⊻ encoded[row])) corrected $(weight(result.error_pattern)) message $(bits(result.message, k)) $(success ? "ok" : "wrong")")
519 end
520 push!(output, "result ok $correct wrong $(depth - correct)")
521 return output
522 end
523
524 text = args[command == BOUNDED ? 3 : 2]
525 expected = command == ENCODE ? k : n
526 ensure(length(text) == expected, "bad length")
527 if command == ENCODE
528 return ["encode $name message $text codeword $(bits(encode(code, parse_bits(text)), n))"]
529 elseif command == ERASURE
530 known, value = zero(Word), zero(Word)
531 for (column, unit) in enumerate(text)
532 unit != '?' && (known |= bit_at(n, column))
533 unit == '1' && (value |= bit_at(n, column))
534 end
535 candidates = filter(word -> word & known == value, span(code.g))
536 output = "erasure $name erased $(count(==('?'), text)) candidates $(length(candidates))"
537 if length(candidates) == 1
538 word = only(candidates)
539 message = something(findfirst(==(word), encoded_words(code))) - 1
540 output *= " codeword $(bits(word, n)) message $(bits(message, k))"
541 end
542 return [output]
543 end
544 word = parse_bits(text)
545 state = syndrome(code, word)
546 command == SYNDROME && return ["syndrome $name $(bits(state, n - k)) codeword $(yes(state == 0))"]
547 errors = leaders(code)
548 if command == COSET
549 return ["coset $name syndrome $(bits(state, n - k)) leader $(bits(errors[Int(state) + 1], n)) weights $(weight_items(distribution(span(code.g), n, word)))"]
550 end
551 result = decode(code, word, errors, encoded_words(code))
552 if command == BOUNDED && weight(result.error_pattern) > parse(Int, args[2])
553 return ["bounded $name fail weight $(weight(result.error_pattern))"]
554 end
555 ["$(lowercase(string(command))) $name $(decoded_text(code, result))"]
556end

558function independent_report(command::CommandKind, args::Vector{String})
559 if command in (CRC, CRCCHECK)
560 degree = length(args[1]) - 1
561 ensure(first(args[1]) == '1' && 1 <= degree <= 16, "bad parameter")
562 ensure(command != CRC || length(args[2]) + degree <= 30, "bad length")
563 word = parse_bits(args[2]) << (command == CRC ? degree : 0)
564 residue = poly_divide(word, parse_bits(args[1])).remainder
565 suffix = bits(residue, degree)
566 return command == CRC ? ["crc remainder $suffix frame $(args[2])$suffix"] :
567 ["crccheck $(residue == 0 ? "ok" : "error") remainder $suffix"]
568 end
569 size = parse(Int, args[1])
570 if command == BOUNDS
571 distance = parse(Int, args[2])
572 ensure(1 <= distance <= size <= 24, "bad parameter")
573 ball = volume(size, (distance - 1) ÷ 2)
574 hamming = maximum(k for k in 0:size if Int64(power2(k)) * ball <= power2(size))
575 griesmer = maximum(k for k in 1:size if sum(cld(distance, power2(i)) for i in 0:k-1) <= size)
576 return ["bounds n $size d $distance singleton $(size - distance + 1) hamming $hamming griesmer $griesmer"]
577 end
578 ensure(1 <= size <= 24, "bad parameter")
579 if command == FACTOR
580 remaining = (one(Word) << size) | one(Word)
581 factor = Word(3)
582 divisors = 1
583 factors = String[]
584 while 2 * (word_length(factor) - 1) <= word_length(remaining) - 1
585 exponent = 0
586 while true
587 divided = poly_divide(remaining, factor)
588 divided.remainder != 0 && break
589 remaining = divided.quotient
590 exponent += 1
591 end
592 if exponent != 0
593 push!(factors, "$(bits(factor, word_length(factor)))^$exponent")
594 divisors *= exponent + 1
595 end
596 factor += Word(2)
597 end
598 if remaining != 1
599 push!(factors, "$(bits(remaining, word_length(remaining)))^1")
600 divisors *= 2
601 end
602 return ["factor $size $(joined(factors)) divisors $divisors"]
603 end
604 ensure(length(args) == size + 2, "bad length")
605 counts = parse.(Int, args[2:end])
606 ensure(counts[1] == 1, "bad parameter")
607 total = sum(counts)
608 numerators = [sum(counts[argument + 1] * krawtchouk(size, index, argument) for argument in 0:size) for index in 0:size]
609 feasible = ispow2(total) && all(numerator -> numerator >= 0 && numerator % total == 0, numerators)
610 dual = [fraction_text(numerator, total) for numerator in numerators]
611 ["macwilliams n $size size $total", "dual $(joined(dual))", "feasible $(yes(feasible))"]
612end

614function execute(line::String, registry::Dict{String, Code})
615 words = String.(split(line, ' '))
616 name = first(words)
617 form = get(SPECS, name, nothing)
618 ensure(!isnothing(form), "unknown command $name")
619 args = words[2:end]
620 ensure(form.minimum <= length(args) <= form.maximum, "bad arguments")
621 for (index, argument) in enumerate(args)
622 ensure(valid_word(argument, form.types[min(index, length(form.types))]), "bad arguments")
623 end
624 sources = Code[]
625 for (index, kind) in enumerate(form.types)
626 if kind in ('A', 'B')
627 ensure(haskey(registry, args[index]), "no code $(args[index])")
628 push!(sources, registry[args[index]])
629 end
630 end
631 command = form.kind
632 if Int(command) <= Int(PRODUCT)
633 code = construct(command, args, sources)
634 action = haskey(registry, args[1]) ? "redefined" : "defined"
635 registry[args[1]] = code
636 return ["$action $(args[1]) n $(code.n) k $(dimension(code))"]
637 elseif command in INDEPENDENT
638 return independent_report(command, args)
639 elseif command == LIST
640 names = sort!(collect(keys(registry)))
641 rows = ["codes $(length(names))"]
642 for code_name in names
643 code = registry[code_name]
644 push!(rows, "$code_name n $(code.n) k $(dimension(code)) encoder $(encoder(code))")
645 end
646 return rows
647 elseif command == DELETE
648 delete!(registry, args[1])
649 return ["deleted $(args[1])"]
650 elseif command in (ENCODE, SYNDROME, DECODE, BOUNDED, COSET, ERASURE, INTERLEAVE)
651 return encoding_report(command, args, only(sources))
652 end
653 inspect_code(command, args, sources)
654end

656function main()
657 registry = Dict{String, Code}()
658 command_count = parse(Int, readline())
659 for _ in 1:command_count
660 try
661 foreach(println, execute(readline(), registry))
662 catch problem
663 problem isa CommandError || rethrow()
664 println("error: ", problem.message)
665 end
666 end
667end
