5const bit = (word, length, column) => (word >>> (length - column - 1)) & 1;

9function weight(word) {

11 for (; word; word &= word - 1) ++count;

14function gcd(a, b) {
15 a = Math.abs(a); b = Math.abs(b);
16 while (b) [a, b] = [b, a % b];
17 return a;

19function divide(dividend, divisor) {
20 let quotient = 0;
21 const d = degree(divisor);
22 while (dividend && degree(dividend) >= d) {
23 const shift = degree(dividend) - d;
24 quotient ^= 1 << shift;
25 dividend ^= divisor << shift;
26 }
27 return { quotient, remainder: dividend };
28}
29function basis(rows, length) {
30 const pivots = Array(length).fill(0), kept = [];
31 for (const original of rows) {
32 let row = original;
33 for (let column = 0; column < length; ++column) if (bit(row, length, column)) {
34 if (pivots[column]) row ^= pivots[column];
35 else { pivots[column] = row; kept.push(original); break; }
36 }
37 }
38 return kept;
39}
40function rref(original, length) {
41 const rows = [...original], columns = [];
42 for (let column = 0; column < length; ++column) {
43 const pivot = columns.length;
44 const found = rows.findIndex((row, index) => index >= pivot && bit(row, length, column));
45 if (found < 0) continue;
46 [rows[pivot], rows[found]] = [rows[found], rows[pivot]];
47 for (let row = 0; row < rows.length; ++row) if (row !== pivot && bit(rows[row], length, column)) rows[row] ^= rows[pivot];
48 columns.push(column);
49 }
50 return { rows: rows.slice(0, columns.length), columns };
51}
52function nullspace(original, length) {
53 const { rows, columns } = rref(original, length), free = [];
54 for (let column = 0; column < length; ++column) if (!columns.includes(column)) {
55 let word = 1 << (length - column - 1);
56 for (let row = 0; row < rows.length; ++row) if (bit(rows[row], length, column)) word |= 1 << (length - columns[row] - 1);
57 free.push(word);
58 }
59 return free;
60}
61function span(rows) {
62 const words = [0];
63 for (const row of rows) {
64 const count = words.length;
65 for (let index = 0; index < count; ++index) words.push(words[index] ^ row);
66 }
67 return words;
68}
69function distribution(words, length) {
70 const counts = Array(length + 1).fill(0);
71 for (const word of words) ++counts[weight(word)];
72 return counts;

75function* fixedWeight(length, count) {
76 if (count === 0) { yield 0; return; }
77 for (let word = 2 ** count - 1; word < 2 ** length;) {
78 yield word;
79 const lowest = word & -word, next = word + lowest;
80 word = next | (((next ^ word) >>> 2) / lowest);
81 }
82}
83function dimension(length, rank) {
84 if (rank < 1 || rank > 12 || length - rank < 1 || length - rank > 12) fail(`bad dimension ${length} ${rank}`);
85}
86function removeBit(word, length, column) {
87 const low = length - column - 1;
88 return ((word >>> (low + 1)) << low) | (word & ((1 << low) - 1));
89}
90function reedMuller(order, variables) {
91 const length = 1 << variables;
92 if (order === 0) return [(1 << length) - 1];
93 const half = length / 2;
94 return [...reedMuller(Math.min(order, variables - 1), variables - 1).map(row => (row << half) | row),
95 ...reedMuller(order - 1, variables - 1)];
96}

98class Code {
99 constructor(length, generator, check = null, polynomial = null) {
100 dimension(length, generator.length);
101 this.n = length; this.k = generator.length;
102 this.G = generator; this.H = check ?? nullspace(generator, length);
103 this.polynomial = polynomial;
104 this.messages = new Map();
105 this.encoded = Array.from({ length: 1 << this.k }, (_, message) => {
106 const word = this.encode(message);
107 this.messages.set(word, message);
108 return word;
109 });
110 this.words = [...this.encoded].sort((a, b) => a - b);
111 this.leaderTable = null;
112 }
113 get encoder() { return this.polynomial === null ? 'matrix' : 'polynomial'; }
114 encode(message) {
115 if (this.polynomial !== null) {
116 const shifted = message << (this.n - this.k);
117 return shifted | divide(shifted, this.polynomial).remainder;
118 }
119 let word = 0;
120 for (let row = 0; row < this.k; ++row) if (bit(message, this.k, row)) word ^= this.G[row];
121 return word;
122 }
123 syndrome(word) {
124 let result = 0;
125 for (const row of this.H) result = (result << 1) | (weight(row & word) % 2);
126 return result;
127 }
128 leaders() {
129 if (this.leaderTable !== null) return this.leaderTable;
130 const result = Array(1 << (this.n - this.k)).fill(-1);
131 let remaining = result.length;
132 for (let count = 0; remaining; ++count) for (const word of fixedWeight(this.n, count)) {
133 const syndrome = this.syndrome(word);
134 if (result[syndrome] === -1) {
135 result[syndrome] = word;
136 if (--remaining === 0) break;
137 }
138 }
139 this.leaderTable = result;
140 return result;
141 }
142 decode(word) {
143 const error = this.leaders()[this.syndrome(word)], corrected = word ^ error;
144 return { error, word: corrected, message: this.messages.get(corrected) };
145 }
146 decodedText(decoded) {
147 return `codeword ${bits(decoded.word, this.n)} message ${bits(decoded.message, this.k)} error ${bits(decoded.error, this.n)} weight ${weight(decoded.error)}`;
148 }

188 parse(line) {

190 if (!spec) fail(`unknown command ${command}`);
191 const extra = args.length - spec.types.length;
192 if (spec.repeat === null ? extra !== 0 : extra < 1 || extra > spec.maximum) fail('bad arguments');
193 for (let index = 0; index < args.length; ++index) if (!patterns[spec.types[index] ?? spec.repeat].test(args[index])) fail('bad arguments');

197 define(command, args) {
198 const name = args[0], a = this.codes.get(args[1]), b = this.codes.get(args[2]);
199 let n, G, H = null, polynomial = null;
200 switch (command) {
201 case 'GEN': case 'PCHECK': {
202 const values = args.slice(1);
203 n = values[0].length;
204 if (values.some(value => value.length !== n)) fail('bad length');
205 const rows = basis(values.map(value => parseInt(value, 2)), n);
206 if (command === 'GEN') G = rows;
207 else { H = rows; G = nullspace(H, n); }
208 break;
209 }
210 case 'HAMMING': case 'SIMPLEX': {
211 const r = Number(args[1]);
212 if (r < 2 || r > 4) fail('bad parameter');

214 const rows = Array(r).fill(0);
215 for (let column = 1; column <= n; ++column) for (let row = 0; row < r; ++row) rows[row] = (rows[row] << 1) | bit(column, r, row);
216 if (command === 'HAMMING') { H = rows; G = nullspace(H, n); }
217 else G = rows;
218 break;
219 }
220 case 'REPETITION': case 'PARITY': {
221 n = Number(args[1]);
222 if (n < 2 || n > 24) fail('bad parameter');
223 G = command === 'REPETITION' ? [(1 << n) - 1] : Array.from({ length: n - 1 }, (_, row) => (1 << (n - row - 1)) | 1);
224 break;
225 }
226 case 'RM': {
227 const r = Number(args[1]), m = Number(args[2]);
228 if (m < 1 || m > 4 || r > m) fail('bad parameter');
229 n = 1 << m; G = reedMuller(r, m);
230 break;
231 }
232 case 'CYCLIC': {
233 n = Number(args[1]);
234 const text = args[2], d = text.length - 1;

240 G = Array.from({ length: k }, (_, row) => polynomial << (k - row - 1));
241 H = Array(d).fill(0);
242 for (let column = 0; column < n; ++column) {
243 const remainder = divide(1 << (n - column - 1), polynomial).remainder;
244 for (let row = 0; row < d; ++row) H[row] = (H[row] << 1) | bit(remainder, d, row);
245 }
246 break;
247 }
248 case 'DUAL': n = a.n; G = [...a.H]; H = [...a.G]; break;
249 case 'EXTEND': n = a.n + 1; G = a.G.map(row => (row << 1) | (weight(row) % 2)); break;
250 case 'PUNCTURE': case 'SHORTEN': {
251 const column = Number(args[2]) - 1;
252 if (column < 0 || column >= a.n) fail('bad position');
253 let rows = [...a.G];
254 if (command === 'SHORTEN') {
255 const pivot = rows.findIndex(row => bit(row, a.n, column));
256 if (pivot >= 0) {
257 const z = rows[pivot];

259 }
260 }
261 n = a.n - 1; G = rows.map(row => removeBit(row, a.n, column));
262 if (command === 'PUNCTURE') G = basis(G, n);
263 break;
264 }
265 case 'AUGMENT': n = a.n; G = basis([...a.G, (1 << n) - 1], n); break;
266 case 'EXPURGATE': {
267 n = a.n;
268 const pivot = a.G.findIndex(row => weight(row) % 2);
269 G = pivot < 0 ? [...a.G] : a.G.filter((_, index) => index !== pivot).map(row => weight(row) % 2 ? row ^ a.G[pivot] : row);
270 break;
271 }
272 case 'PERMUTE': {
273 const order = args.slice(2).map(Number);
274 if (order.length !== a.n || new Set(order).size !== a.n || order.some(column => column < 1 || column > a.n)) fail('bad permutation');
275 n = a.n; G = a.G.map(row => order.reduce((word, column) => (word << 1) | bit(row, n, column - 1), 0));
276 break;
277 }
278 case 'DIRECTSUM': case 'UUV': case 'PRODUCT': {
279 if (command === 'UUV' && a.n !== b.n) fail('bad length');

282 if (command === 'PRODUCT') G = a.G.flatMap(x => b.G.map(y => {
283 let word = 0;
284 for (let column = 0; column < a.n; ++column) word = (word << b.n) | (bit(x, a.n, column) ? y : 0);
285 return word;
286 }));
287 else G = [...a.G.map(row => (row << b.n) | (command === 'UUV' ? row : 0)), ...b.G];

292 this.codes.set(name, code);
293 return [`${replaced ? 'redefined' : 'defined'} ${name} n ${n} k ${code.k}`];

295 systematic(name, code) {
296 const { n, k } = code, rows = [...code.G], order = Array.from({ length: n }, (_, index) => index + 1);
297 for (let pivot = 0; pivot < k; ++pivot) {
298 let column = pivot;
299 while (!rows.some((row, index) => index >= pivot && bit(row, n, column))) ++column;
300 if (column !== pivot) {
301 const mask = (1 << (n - pivot - 1)) | (1 << (n - column - 1));
302 for (let row = 0; row < k; ++row) if (bit(rows[row], n, pivot) !== bit(rows[row], n, column)) rows[row] ^= mask;
303 [order[pivot], order[column]] = [order[column], order[pivot]];
304 }
305 const found = rows.findIndex((row, index) => index >= pivot && bit(row, n, pivot));
306 [rows[pivot], rows[found]] = [rows[found], rows[pivot]];
307 for (let row = 0; row < k; ++row) if (row !== pivot && bit(rows[row], n, pivot)) rows[row] ^= rows[pivot];
308 }
309 const output = [`systematic ${name} perm ${order.join(' ')}`, ...rows.map(row => `G ${bits(row, n)}`)];
310 for (let column = k; column < n; ++column) {
311 let row = 0;
312 for (const value of rows) row = (row << 1) | bit(value, n, column);
313 row = (row << (n - k)) | (1 << (n - column - 1));
314 output.push(`H ${bits(row, n)}`);
315 }
316 return output;

318 burst(name, code) {
319 const detect = Math.min(...code.words.slice(1).map(word => degree(word) - degree(word & -word)));
320 const seen = new Set([0]);
321 for (let length = 1; length <= code.n; ++length) {
322 for (let start = 0; start + length <= code.n; ++start) {

324 const pattern = length === 1 ? 1 : (1 << (length - 1)) | (middle << 1) | 1;
325 const syndrome = code.syndrome(pattern << (code.n - start - length));
326 if (seen.has(syndrome)) return [`burst ${name} detect ${detect} correct ${length - 1}`];
327 seen.add(syndrome);
328 }

332 interleave(name, code, args) {
333 const messages = args.slice(3), start = Number(args[1]), length = Number(args[2]), depth = messages.length;
334 if (messages.some(message => message.length !== code.k)) fail('bad length');
335 if (start < 1 || length < 1 || start + length - 1 > depth * code.n) fail('bad burst');
336 const sentWords = messages.map(message => code.encode(parseInt(message, 2)));
337 const receivedWords = Array(depth).fill(0), sent = [], received = [];
338 for (let column = 0; column < code.n; ++column) for (let row = 0; row < depth; ++row) {
339 const position = column * depth + row + 1, original = bit(sentWords[row], code.n, column);
340 const changed = original ^ Number(start <= position && position < start + length);
341 sent.push(original); received.push(changed);
342 receivedWords[row] = (receivedWords[row] << 1) | changed;
343 }
344 const output = [`interleave ${name} depth ${depth} burst ${start} ${length}`, `sent ${sent.join('')}`, `recv ${received.join('')}`];
345 let correct = 0;
346 for (let row = 0; row < depth; ++row) {
347 const decoded = code.decode(receivedWords[row]), message = bits(decoded.message, code.k), ok = message === messages[row];
348 if (ok) ++correct;
349 output.push(`row ${row + 1} errors ${weight(receivedWords[row] ^ sentWords[row])} corrected ${weight(decoded.error)} message ${message} ${ok ? 'ok' : 'wrong'}`);
350 }
351 output.push(`result ok ${correct} wrong ${depth - correct}`);
352 return output;
353 }
354 execute({ command, args }) {
355 if (defining.has(command)) return this.define(command, args);
356 const name = args[0], code = this.codes.get(name);
357 switch (command) {
358 case 'LIST': return [`codes ${this.codes.size}`, ...[...this.codes].sort(([a], [b]) => a < b ? -1 : a > b ? 1 : 0)

361 case 'SHOW': {

365 }
366 case 'INFO': {

368 const volume = choose[n].slice(0, t + 1).reduce((a, b) => a + b, 0);
369 const cyclic = code.G.every(row => code.messages.has(((row << 1) & ((1 << n) - 1)) | (row >>> (n - 1))));
370 return [`info ${name} n ${n} k ${k} rate ${k / divisor}/${n / divisor} d ${distance} t ${t} perfect ${yes(volume === 2 ** (n - k))} mds ${yes(distance === n - k + 1)} cyclic ${cyclic ? code.words[1].toString(2) : 'no'}`];
371 }
372 case 'SYSTEMATIC': return this.systematic(name, code);
373 case 'WEIGHTS': {
374 const a = distribution(code.words, code.n), b = distribution(span(code.H), code.n);
375 const orthogonal = code.G.every(x => code.G.every(y => weight(x & y) % 2 === 0));
376 return [`weights ${name} ${weightList(a)}`, `dual ${name} ${weightList(b)}`,
377 `duality ${name} self-orthogonal ${yes(orthogonal)} self-dual ${yes(orthogonal && code.n === 2 * code.k)} formally-self-dual ${yes(a.every((value, index) => value === b[index]))}`];
378 }
379 case 'COVERING': {
380 const counts = distribution(code.leaders(), code.n);
381 let radius = code.n;
382 while (counts[radius] === 0) --radius;
383 return [`covering ${name} radius ${radius} leaders ${counts.slice(0, radius + 1).map((count, index) => `${index}:${count}`).join(' ')}`];
384 }
385 case 'BURST': return this.burst(name, code);
386 case 'TRELLIS': {
387 const profile = [];
388 for (let cut = 0; cut <= code.n; ++cut) {
389 const right = code.n - cut, mask = (1 << right) - 1;
390 profile.push(basis(code.G.map(row => row >>> right), cut).length + basis(code.G.map(row => row & mask), right).length - code.k);
391 }
392 return [`trellis ${name} profile ${profile.join(' ')} max ${Math.max(...profile)}`];
393 }
394 case 'STDARRAY': {
395 if (code.n - code.k > 6 || code.k > 4) fail('too large');
396 const leaders = [...code.leaders().entries()].sort(([, a], [, b]) => weight(a) - weight(b) || a - b);
397 return [`stdarray ${name} rows ${leaders.length} columns ${code.words.length}`, ...leaders.map(([syndrome, leader]) =>
398 `${bits(syndrome, code.n - code.k)}: ${code.words.map(word => bits(word ^ leader, code.n)).join(' ')}`)];
399 }
400 case 'EQUAL': {
401 const other = this.codes.get(args[1]);
402 if (code.n !== other.n) return [`equal ${name} ${args[1]} no lengths ${code.n} ${other.n}`];
403 const common = code.k + other.k - basis([...code.G, ...other.G], code.n).length;
404 return [`equal ${name} ${args[1]} ${yes(code.k === common && other.k === common)} intersection ${common}`];
405 }
406 case 'ENCODE': {
407 if (args[1].length !== code.k) fail('bad length');
408 return [`encode ${name} message ${args[1]} codeword ${bits(code.encode(parseInt(args[1], 2)), code.n)}`];
409 }
410 case 'SYNDROME': case 'DECODE': case 'BOUNDED': case 'COSET': {
411 const text = args[command === 'BOUNDED' ? 2 : 1];
412 if (text.length !== code.n) fail('bad length');
413 const word = parseInt(text, 2), syndrome = code.syndrome(word);
414 if (command === 'SYNDROME') return [`syndrome ${name} ${bits(syndrome, code.n - code.k)} codeword ${yes(syndrome === 0)}`];
415 const decoded = code.decode(word);
416 if (command === 'COSET') return [`coset ${name} syndrome ${bits(syndrome, code.n - code.k)} leader ${bits(decoded.error, code.n)} weights ${weightList(distribution(code.words.map(value => value ^ word), code.n))}`];
417 if (command === 'BOUNDED' && weight(decoded.error) > Number(args[1])) return [`bounded ${name} fail weight ${weight(decoded.error)}`];
418 return [`${command.toLowerCase()} ${name} ${code.decodedText(decoded)}`];
419 }
420 case 'ERASURE': {
421 const text = args[1];
422 if (text.length !== code.n) fail('bad length');
423 let mask = 0, known = 0, erased = 0;
424 for (const character of text) {
425 mask = (mask << 1) | Number(character !== '?');
426 known = (known << 1) | Number(character === '1');
427 if (character === '?') ++erased;
428 }
429 const candidates = code.words.filter(word => (word & mask) === known);
430 const detail = candidates.length === 1 ? ` codeword ${bits(candidates[0], code.n)} message ${bits(code.messages.get(candidates[0]), code.k)}` : '';
431 return [`erasure ${name} erased ${erased} candidates ${candidates.length}${detail}`];
432 }
433 case 'INTERLEAVE': return this.interleave(name, code, args);
434 default: return independent(command, args);
435 }

439function independent(command, args) {
440 if (command === 'CRC' || command === 'CRCCHECK') {
441 const [generator, text] = args, d = generator.length - 1;
442 if (generator[0] === '0' || generator.length < 2 || generator.length > 17) fail('bad parameter');
443 if (command === 'CRC' && text.length + d > 30) fail('bad length');
444 const word = parseInt(text, 2), divisor = parseInt(generator, 2);
445 const remainder = divide(command === 'CRC' ? word << d : word, divisor).remainder, suffix = bits(remainder, d);
446 return [command === 'CRC' ? `crc remainder ${suffix} frame ${text}${suffix}` : `crccheck ${remainder === 0 ? 'ok' : 'error'} remainder ${suffix}`];
447 }
448 const n = Number(args[0]);
449 if (command === 'BOUNDS') {
450 const d = Number(args[1]);
451 if (d < 1 || d > n || n > 24) fail('bad parameter');
452 const volume = choose[n].slice(0, Math.floor((d - 1) / 2) + 1).reduce((a, b) => a + b, 0);
453 let hamming = 0, griesmer = 0, sum = 0;
454 for (let k = 0; k <= n; ++k) if (2 ** k * volume <= 2 ** n) hamming = k;
