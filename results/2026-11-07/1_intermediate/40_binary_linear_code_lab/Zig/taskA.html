123fn divide(original: Word, divisor: Word) Division {
124 var dividend = original;
125 var quotient: Word = 0;
126 const d = degree(divisor);
127 while (dividend != 0 and degree(dividend) >= d) {
128 const shift = degree(dividend) - d;
129 quotient ^= one(shift);
130 dividend ^= divisor << @intCast(shift);
131 }
132 return .{ .quotient = quotient, .remainder = dividend };
133}
134fn basis(rows: []const Word, length: usize) Matrix {
135 var pivots: [30]Word = @splat(0);
136 var result: Matrix = .{};
137 for (rows) |original| {
138 var row = original;
139 for (0..length) |column| {
140 if (!bit(row, length, column)) continue;
141 if (pivots[column] != 0) {
142 row ^= pivots[column];
143 } else {
144 pivots[column] = row;
145 result.add(original);
146 break;
147 }
148 }
149 }
150 return result;
151}
152fn rref(original: Matrix, length: usize) Echelon {
153 var rows = original;
154 var columns: [30]usize = undefined;
155 var count: usize = 0;
156 for (0..length) |column| {
157 var found = count;
158 while (found < rows.len and !bit(rows.rows[found], length, column)) : (found += 1) {}
159 if (found == rows.len) continue;
160 const old = rows.rows[count];
161 rows.rows[count] = rows.rows[found];
162 rows.rows[found] = old;
163 for (0..rows.len) |row| if (row != count and bit(rows.rows[row], length, column)) {
164 rows.rows[row] ^= rows.rows[count];
165 };
166 columns[count] = column;
167 count += 1;
168 }
169 rows.len = count;
170 return .{ .rows = rows, .columns = columns };
171}
172fn nullspace(original: Matrix, length: usize) Matrix {
173 const reduced = rref(original, length);
174 var result: Matrix = .{};
175 for (0..length) |column| {
176 var is_pivot = false;
177 for (reduced.columns[0..reduced.rows.len]) |pivot| if (pivot == column) {
178 is_pivot = true;
179 break;
180 };
181 if (is_pivot) continue;
182 var word = one(length - column - 1);
183 for (reduced.rows.slice(), 0..) |row, index| {
184 if (bit(row, length, column)) word |= one(length - reduced.columns[index] - 1);
185 }
186 result.add(word);
187 }
188 return result;
189}
190fn spanWord(rows: Matrix, message: usize) Word {
191 var result: Word = 0;
192 for (rows.slice(), 0..) |row, index| if (message & (@as(usize, 1) << @intCast(index)) != 0) {
193 result ^= row;
194 };
195 return result;
196}
197fn distribution(words: []const Word) [25]usize {
198 var result: [25]usize = @splat(0);
199 for (words) |word| result[weight(word)] += 1;
200 return result;
201}
202fn weightList(out: *Writer, counts: []const usize) Writer.Error!void {
203 var first = true;
204 for (counts, 0..) |count, index| {
205 if (count == 0) continue;
206 if (!first) try out.writeByte(' ');

209 }

211fn removeBit(word: Word, length: usize, column: usize) Word {
212 const low = length - column - 1;
213 return ((word >> @intCast(low + 1)) << @intCast(low)) | (word & (one(low) - 1));
214}
215fn reedMuller(order: usize, variables: usize) Matrix {
216 const length = @as(usize, 1) << @intCast(variables);
217 var result: Matrix = .{};
218 if (order == 0) {
219 result.add(one(length) - 1);
220 return result;
221 }
222 const first = reedMuller(@min(order, variables - 1), variables - 1);
223 for (first.slice()) |row| result.add((row << @intCast(length / 2)) | row);
224 const second = reedMuller(order - 1, variables - 1);
225 for (second.slice()) |row| result.add(row);
226 return result;
227}

229const binomials = blk: {
230 @setEvalBranchQuota(5000);
231 var table: [25][25]i64 = @splat(@splat(0));
232 for (0..25) |n| {
233 table[n][0] = 1;
234 table[n][n] = 1;
235 if (n >= 2) for (1..n) |k| {
236 table[n][k] = table[n - 1][k - 1] + table[n - 1][k];
237 };
238 }
239 break :blk table;
240};
241fn binomial(n: usize, k: usize) i64 {
242 return if (k > n) 0 else binomials[n][k];
243}
244fn volume(n: usize, t: usize) i64 {
245 var sum: i64 = 0;
246 for (binomials[n][0 .. t + 1]) |value| sum += value;
247 return sum;
248}
249fn gcd(first: i64, second: i64) i64 {
250 var a: i64 = @intCast(@abs(first));
251 var b: i64 = @intCast(@abs(second));
252 while (b != 0) {
253 const remainder = @mod(a, b);
254 a = b;
255 b = remainder;
256 }
257 return a;
258}

261const Code = struct {

270 fn k(code: *const Code) usize {
271 return code.g.len;

273 fn encoder(code: *const Code) []const u8 {
274 return if (code.polynomial == null) "matrix" else "polynomial";
275 }
276 fn create(gpa: Allocator, n: usize, g: Matrix, h: ?Matrix, polynomial: ?Word) Allocator.Error!*Code {
277 const code = try gpa.create(Code);
278 code.* = .{ .n = n, .g = g, .h = h orelse nullspace(g, n), .polynomial = polynomial };
279 const encoded = try gpa.alloc(Word, one(g.len));
280 for (encoded, 0..) |*word, message| word.* = code.encode(@intCast(message));
281 code.encoded = encoded;
282 const words = try gpa.dupe(Word, encoded);
283 std.sort.heap(Word, words, {}, lessWord);
284 code.words = words;
285 return code;
286 }
287 fn encode(code: *const Code, message: Word) Word {
288 if (code.polynomial) |polynomial| {
289 const shifted = message << @intCast(code.n - code.k());
290 return shifted | divide(shifted, polynomial).remainder;
291 }
292 var word: Word = 0;
293 for (code.g.slice(), 0..) |row, index| if (bit(message, code.k(), index)) {
294 word ^= row;
295 };
296 return word;
297 }
298 fn syndrome(code: *const Code, word: Word) usize {
299 var result: usize = 0;
300 for (code.h.slice()) |row| result = (result << 1) | (weight(row & word) % 2);
301 return result;
302 }
303 fn messageFor(code: *const Code, word: Word) Word {
304 for (code.encoded, 0..) |value, index| if (value == word) return @intCast(index);
305 unreachable;
306 }
307 fn leaders(code: *Code, gpa: Allocator) Allocator.Error![]const Word {
308 if (code.cached_leaders) |values| return values;
309 const missing = ~@as(Word, 0);
310 const result = try gpa.alloc(Word, one(code.n - code.k()));
311 @memset(result, missing);
312 result[0] = 0;
313 var remaining = result.len - 1;
314 var count: usize = 1;
315 while (remaining > 0) : (count += 1) {
316 var word = one(count) - 1;
317 while (word < one(code.n)) {
318 const s = code.syndrome(word);
319 if (result[s] == missing) {
320 result[s] = word;
321 remaining -= 1;
322 if (remaining == 0) break;
323 }
324 const lowest = word & (0 -% word);
325 const next = word + lowest;
326 word = next | (((next ^ word) >> 2) / lowest);
327 }
328 }
329 code.cached_leaders = result;
330 return result;
331 }
332 fn decode(code: *Code, gpa: Allocator, received: Word) Allocator.Error!Decoded {
333 const table = try code.leaders(gpa);
334 const err = table[code.syndrome(received)];
335 const word = received ^ err;
336 return .{ .word = word, .message = code.messageFor(word), .err = err };
337 }

352fn formatOf(verb: Verb) Format {
353 return switch (verb) {
354 .GEN, .PCHECK => .{ .types = &.{.name}, .repeated = .bits, .maximum = 24 },
355 .HAMMING, .SIMPLEX, .REPETITION, .PARITY => .{ .types = &.{ .name, .number } },
356 .RM => .{ .types = &.{ .name, .number, .number } },
357 .CYCLIC => .{ .types = &.{ .name, .number, .bits } },
358 .DUAL, .EXTEND, .AUGMENT, .EXPURGATE => .{ .types = &.{ .name, .code } },
359 .PUNCTURE, .SHORTEN => .{ .types = &.{ .name, .code, .number } },
360 .PERMUTE => .{ .types = &.{ .name, .code }, .repeated = .number, .maximum = 24 },
361 .DIRECTSUM, .UUV, .PRODUCT => .{ .types = &.{ .name, .code, .code } },
362 .LIST => .{ .types = &.{} },
363 .DELETE, .SHOW, .INFO, .SYSTEMATIC, .WEIGHTS, .COVERING, .BURST, .TRELLIS, .STDARRAY => .{ .types = &.{.code} },
364 .EQUAL => .{ .types = &.{ .code, .code } },
365 .ENCODE, .SYNDROME, .DECODE, .COSET => .{ .types = &.{ .code, .bits } },
366 .BOUNDED => .{ .types = &.{ .code, .number, .bits } },
367 .ERASURE => .{ .types = &.{ .code, .erasure } },
368 .INTERLEAVE => .{ .types = &.{ .code, .number, .number }, .repeated = .bits, .maximum = 8 },
369 .CRC, .CRCCHECK => .{ .types = &.{ .bits, .bits } },
370 .FACTOR => .{ .types = &.{.number} },
371 .MACWILLIAMS => .{ .types = &.{.number}, .repeated = .number, .maximum = 25 },
372 .BOUNDS => .{ .types = &.{ .number, .number } },
373 };

375fn letter(character: u8) bool {
376 return (character >= 'A' and character <= 'Z') or (character >= 'a' and character <= 'z');
377}
378fn valid(text: []const u8, kind: Argument) bool {
379 if (text.len == 0) return false;
380 switch (kind) {
381 .name, .code => {
382 if (text.len > 8 or !letter(text[0])) return false;
383 for (text) |character| if (!letter(character) and (character < '0' or character > '9')) return false;
384 },
385 .number => {
386 if (text.len > 4 or (text.len > 1 and text[0] == '0')) return false;
387 for (text) |character| if (character < '0' or character > '9') return false;
388 },
389 .bits, .erasure => {
390 if (text.len > 30) return false;
391 for (text) |character| if (character != '0' and character != '1' and !(kind == .erasure and character == '?')) return false;
392 },
393 }
394 return true;
395}

398 lab: *Laboratory,
399 gpa: Allocator,
400 out: *Writer,
401 problem: []const u8 = "",
402 fn fail(session: *Session, comptime fmt: []const u8, args: anytype) Error {
403 session.problem = try std.fmt.allocPrint(session.gpa, fmt, args);
404 return error.Invalid;
405 }
406 fn dimension(session: *Session, n: usize, k: usize) Error!void {
407 if (k < 1 or k > 12 or n <= k or n - k > 12) return session.fail("bad dimension {d} {d}", .{ n, k });
408 }
409 fn parse(session: *Session, words: []const []const u8) Error!Verb {
410 const verb = std.meta.stringToEnum(Verb, words[0]) orelse return session.fail("unknown command {s}", .{words[0]});
411 const spec = formatOf(verb);
412 const args = words[1..];
413 if (args.len < spec.types.len) return session.fail("bad arguments", .{});
414 const extra = args.len - spec.types.len;
415 if (if (spec.repeated == null) extra != 0 else extra < 1 or extra > spec.maximum) return session.fail("bad arguments", .{});
416 for (args, 0..) |word, index| {
417 const kind = if (index < spec.types.len) spec.types[index] else spec.repeated.?;
418 if (!valid(word, kind)) return session.fail("bad arguments", .{});
419 }
420 for (spec.types, 0..) |kind, index| {
421 if (kind == .code and session.lab.find(args[index]) == null) return session.fail("no code {s}", .{args[index]});
422 }
423 return verb;
424 }
425 fn define(session: *Session, verb: Verb, args: []const []const u8) Error!void {
426 const lab = session.lab;
427 const name = args[0];
428 var n: usize = 0;
429 var g: Matrix = .{};
430 var h: ?Matrix = null;
431 var polynomial: ?Word = null;
432 switch (verb) {
433 .GEN, .PCHECK => {
434 n = args[1].len;
435 var input: Matrix = .{};
436 for (args[1..]) |text| {
437 if (text.len != n) return session.fail("bad length", .{});
438 input.add(binary(text));
439 }
440 const rows = basis(input.slice(), n);
441 if (verb == .GEN) g = rows else {
442 h = rows;
443 g = nullspace(rows, n);
444 }
445 },
446 .HAMMING, .SIMPLEX => {
447 const r = integer(args[1]);
448 if (r < 2 or r > 4) return session.fail("bad parameter", .{});
449 n = one(r) - 1;
450 var rows: Matrix = .{};
451 for (0..r) |_| rows.add(0);
452 for (1..n + 1) |column| {
453 for (0..r) |row| rows.rows[row] = (rows.rows[row] << 1) | @intFromBool(bit(@intCast(column), r, row));
454 }
455 if (verb == .HAMMING) {
456 h = rows;
457 g = nullspace(rows, n);
458 } else g = rows;
459 },
460 .REPETITION, .PARITY => {
461 n = integer(args[1]);
462 if (n < 2 or n > 24) return session.fail("bad parameter", .{});
463 if (verb == .REPETITION) g.add(one(n) - 1) else {
464 for (0..n - 1) |row| g.add(one(n - row - 1) | 1);
465 }
466 },
467 .RM => {
468 const order = integer(args[1]);
469 const variables = integer(args[2]);
470 if (variables < 1 or variables > 4 or order > variables) return session.fail("bad parameter", .{});
471 n = one(variables);
472 g = reedMuller(order, variables);
473 },
474 .CYCLIC => {
475 n = integer(args[1]);
476 const text = args[2];
477 if (n < 2 or n > 24 or text[0] == '0' or text.len < 2 or text.len > n) return session.fail("bad parameter", .{});
478 const p = binary(text);
479 if (divide(one(n) | 1, p).remainder != 0) return session.fail("not a divisor", .{});
480 const d = text.len - 1;
481 const k = n - d;
482 try session.dimension(n, k);
483 polynomial = p;
484 for (0..k) |row| g.add(p << @intCast(k - row - 1));
485 var rows: Matrix = .{};
486 for (0..d) |_| rows.add(0);
487 for (0..n) |column| {
488 const remainder = divide(one(n - column - 1), p).remainder;
489 for (0..d) |row| rows.rows[row] = (rows.rows[row] << 1) | @intFromBool(bit(remainder, d, row));
490 }
491 h = rows;
492 },
493 .DUAL => {
494 const a = lab.get(args[1]);
495 n = a.n;
496 g = a.h;
497 h = a.g;
498 },
499 .EXTEND => {
500 const a = lab.get(args[1]);
501 n = a.n + 1;
502 for (a.g.slice()) |row| g.add((row << 1) | @as(Word, @intCast(weight(row) % 2)));
503 },
504 .PUNCTURE, .SHORTEN => {
505 const a = lab.get(args[1]);
506 const position = integer(args[2]);
507 if (position < 1 or position > a.n) return session.fail("bad position", .{});
508 const column = position - 1;
509 var rows = a.g;
510 if (verb == .SHORTEN) {
511 var pivot: ?usize = null;
512 for (rows.slice(), 0..) |row, index| if (bit(row, a.n, column)) {
513 pivot = index;
514 break;
515 };
516 if (pivot) |index| {
517 var kept: Matrix = .{};
518 const z = rows.rows[index];
519 for (rows.slice(), 0..) |row, i| if (i != index) {
520 kept.add(if (bit(row, a.n, column)) row ^ z else row);
521 };
522 rows = kept;
523 }
524 }
525 n = a.n - 1;
526 for (rows.slice()) |row| g.add(removeBit(row, a.n, column));
527 if (verb == .PUNCTURE) g = basis(g.slice(), n);
528 },
529 .AUGMENT => {
530 const a = lab.get(args[1]);
531 n = a.n;
532 var rows = a.g;
533 rows.add(one(n) - 1);
534 g = basis(rows.slice(), n);
535 },
536 .EXPURGATE => {
537 const a = lab.get(args[1]);
538 n = a.n;
539 var pivot: ?usize = null;
540 for (a.g.slice(), 0..) |row, index| if (weight(row) % 2 == 1) {
541 pivot = index;
542 break;
543 };
544 if (pivot) |index| {
545 const z = a.g.rows[index];
546 for (a.g.slice(), 0..) |row, i| if (i != index) {
547 g.add(if (weight(row) % 2 == 1) row ^ z else row);
548 };
549 } else g = a.g;
550 },
551 .PERMUTE => {
552 const a = lab.get(args[1]);
553 n = a.n;
554 if (args.len - 2 != n) return session.fail("bad permutation", .{});
555 var seen: [24]bool = @splat(false);
556 var order: [24]usize = undefined;
557 for (args[2..], 0..) |text, index| {
558 const position = integer(text);
559 if (position < 1 or position > n or seen[position - 1]) return session.fail("bad permutation", .{});
560 seen[position - 1] = true;
561 order[index] = position - 1;
562 }
563 for (a.g.slice()) |row| {
564 var word: Word = 0;
565 for (order[0..n]) |column| word = (word << 1) | @intFromBool(bit(row, n, column));
566 g.add(word);
567 }
568 },
569 .DIRECTSUM, .UUV, .PRODUCT => {
570 const a = lab.get(args[1]);
571 const b = lab.get(args[2]);
572 if (verb == .UUV and a.n != b.n) return session.fail("bad length", .{});
573 n = if (verb == .PRODUCT) a.n * b.n else a.n + b.n;
574 try session.dimension(n, if (verb == .PRODUCT) a.k() * b.k() else a.k() + b.k());
575 if (verb == .PRODUCT) {
576 for (a.g.slice()) |x| for (b.g.slice()) |y| {
577 var row: Word = 0;
578 for (0..a.n) |column| row = (row << @intCast(b.n)) | (if (bit(x, a.n, column)) y else 0);
579 g.add(row);
580 };
581 } else {
582 for (a.g.slice()) |row| g.add((row << @intCast(b.n)) | (if (verb == .UUV) row else 0));
583 for (b.g.slice()) |row| g.add(row);
584 }
585 },
586 else => unreachable,
587 }
588 try session.dimension(n, g.len);
589 const code = try Code.create(lab.gpa, n, g, h, polynomial);
590 const previous = lab.find(name);
591 if (previous) |index| lab.codes.items[index].code = code else {
592 try lab.codes.append(lab.gpa, .{ .name = Name.init(name), .code = code });
593 }
594 try line(session.out, "{s} {s} n {d} k {d}", .{ if (previous != null) "redefined" else "defined", name, n, code.k() });
595 }
596 fn systematic(session: *Session, name: []const u8, code: *const Code) Error!void {
597 const n = code.n;
598 const k = code.k();
599 var rows = code.g;
600 var order: [24]usize = undefined;
601 for (0..n) |column| order[column] = column + 1;
602 for (0..k) |pivot| {
603 var column = pivot;
604 var found = pivot;
605 while (true) {
606 found = pivot;
607 while (found < k and !bit(rows.rows[found], n, column)) : (found += 1) {}
608 if (found < k) break;
609 column += 1;
610 }
611 if (column != pivot) {
612 const mask = one(n - column - 1) | one(n - pivot - 1);
613 for (rows.rows[0..k]) |*row| if (bit(row.*, n, column) != bit(row.*, n, pivot)) {
614 row.* ^= mask;
615 };
616 const old = order[pivot];
617 order[pivot] = order[column];
618 order[column] = old;
619 }
620 const old = rows.rows[pivot];
621 rows.rows[pivot] = rows.rows[found];
622 rows.rows[found] = old;
623 for (0..k) |row| if (row != pivot and bit(rows.rows[row], n, pivot)) {
624 rows.rows[row] ^= rows.rows[pivot];
625 };
626 }
627 const out = session.out;
628 try out.print("systematic {s} perm", .{name});
629 for (order[0..n]) |column| try out.print(" {d}", .{column});
630 try out.writeByte('\n');
631 for (rows.slice()) |row| try line(out, "G {f}", .{bits(row, n)});
632 for (k..n) |column| {
633 var row: Word = 0;
634 for (rows.slice()) |value| row = (row << 1) | @intFromBool(bit(value, n, column));
635 row = (row << @intCast(n - k)) | one(n - column - 1);
636 try line(out, "H {f}", .{bits(row, n)});
637 }
638 }
639 fn burst(session: *Session, name: []const u8, code: *const Code) Error!void {
640 var detect = code.n;
641 for (code.words[1..]) |word| detect = @min(detect, degree(word) - @as(usize, @ctz(word)));
642 var seen: [4096]bool = @splat(false);
643 seen[0] = true;
644 for (1..code.n + 1) |length| {
645 for (0..code.n - length + 1) |start| {
646 const limit: usize = if (length == 1) 1 else one(length - 2);
647 for (0..limit) |middle| {
648 const pattern: Word = if (length == 1) 1 else one(length - 1) | (@as(Word, @intCast(middle)) << 1) | 1;
649 const s = code.syndrome(pattern << @intCast(code.n - start - length));
650 if (seen[s]) return line(session.out, "burst {s} detect {d} correct {d}", .{ name, detect, length - 1 });
651 seen[s] = true;
652 }
653 }
654 }
655 unreachable;
656 }
657 fn decodedText(session: *Session, code: *const Code, value: Decoded) Error!void {
658 try session.out.print("codeword {f} message {f} error {f} weight {d}", .{
659 bits(value.word, code.n), bits(value.message, code.k()), bits(value.err, code.n), weight(value.err),
660 });
661 }
662 fn interleave(session: *Session, name: []const u8, code: *Code, args: []const []const u8) Error!void {
663 const messages = args[3..];
664 const depth = messages.len;
665 const start = integer(args[1]);
666 const length = integer(args[2]);
667 for (messages) |message| if (message.len != code.k()) return session.fail("bad length", .{});
668 if (start < 1 or length < 1 or start + length - 1 > depth * code.n) return session.fail("bad burst", .{});
669 var sent_words: [8]Word = undefined;
670 var received_words: [8]Word = @splat(0);
671 for (messages, 0..) |message, row| sent_words[row] = code.encode(binary(message));
672 var sent: [192]u8 = undefined;
673 var received: [192]u8 = undefined;
674 for (0..code.n) |column| for (0..depth) |row| {
675 const index = column * depth + row;
676 const original = bit(sent_words[row], code.n, column);
677 const changed = original != (start <= index + 1 and index + 1 < start + length);
678 sent[index] = if (original) '1' else '0';
679 received[index] = if (changed) '1' else '0';
680 received_words[row] = (received_words[row] << 1) | @intFromBool(changed);
681 };
682 const out = session.out;
683 try line(out, "interleave {s} depth {d} burst {d} {d}", .{ name, depth, start, length });
684 try line(out, "sent {s}", .{sent[0 .. depth * code.n]});
685 try line(out, "recv {s}", .{received[0 .. depth * code.n]});
686 var correct: usize = 0;
687 for (messages, 0..) |message, row| {
688 const decoded = try code.decode(session.lab.gpa, received_words[row]);
689 const ok = decoded.message == binary(message);
690 correct += @intFromBool(ok);
691 try line(out, "row {d} errors {d} corrected {d} message {f} {s}", .{
692 row + 1, weight(received_words[row] ^ sent_words[row]), weight(decoded.err),
693 bits(decoded.message, code.k()), if (ok) "ok" else "wrong",
694 });
695 }
696 try line(out, "result ok {d} wrong {d}", .{ correct, depth - correct });
697 }
698 fn independent(session: *Session, verb: Verb, args: []const []const u8) Error!void {
699 const out = session.out;
700 if (verb == .CRC or verb == .CRCCHECK) {
701 const generator = args[0];
702 const text = args[1];
703 if (generator[0] == '0' or generator.len < 2 or generator.len > 17) return session.fail("bad parameter", .{});
704 const d = generator.len - 1;
705 if (verb == .CRC and text.len + d > 30) return session.fail("bad length", .{});
706 const word = binary(text);
707 const remainder = divide(if (verb == .CRC) word << @intCast(d) else word, binary(generator)).remainder;
708 if (verb == .CRC) return line(out, "crc remainder {f} frame {s}{f}", .{ bits(remainder, d), text, bits(remainder, d) });
709 return line(out, "crccheck {s} remainder {f}", .{ if (remainder == 0) "ok" else "error", bits(remainder, d) });
710 }
711 const n = integer(args[0]);
712 if (verb == .BOUNDS) {
713 const d = integer(args[1]);
714 if (d < 1 or d > n or n > 24) return session.fail("bad parameter", .{});
715 const v = volume(n, (d - 1) / 2);
716 var hamming: usize = 0;
717 var griesmer: usize = 0;
718 var sum: usize = 0;
719 for (0..n + 1) |k| if (@as(i64, one(k)) * v <= @as(i64, one(n))) {
720 hamming = k;
721 };
722 for (1..n + 1) |k| {
723 const denominator: usize = one(k - 1);
724 sum += (d + denominator - 1) / denominator;
725 if (sum <= n) griesmer = k;
726 }
727 return line(out, "bounds n {d} d {d} singleton {d} hamming {d} griesmer {d}", .{ n, d, n - d + 1, hamming, griesmer });
728 }
729 if (n < 1 or n > 24) return session.fail("bad parameter", .{});
730 if (verb == .FACTOR) {
731 var remaining = one(n) | 1;
732 var candidate: Word = 2;
733 var divisors: usize = 1;
734 try out.print("factor {d}", .{n});
735 while (2 * degree(candidate) <= degree(remaining)) : (candidate += 1) {
736 var exponent: usize = 0;
737 while (divide(remaining, candidate).remainder == 0) {
738 remaining = divide(remaining, candidate).quotient;
739 exponent += 1;
740 }
741 if (exponent > 0) {
742 try out.print(" {f}^{d}", .{ bits(candidate, degree(candidate) + 1), exponent });
743 divisors *= exponent + 1;
744 }
745 }
746 if (remaining > 1) {
747 try out.print(" {f}^1", .{bits(remaining, degree(remaining) + 1)});
748 divisors *= 2;
749 }
750 return line(out, " divisors {d}", .{divisors});
751 }
752 if (args.len - 1 != n + 1) return session.fail("bad length", .{});
753 if (integer(args[1]) != 1) return session.fail("bad parameter", .{});
754 var counts: [25]i64 = undefined;
755 var size: i64 = 0;
756 for (args[1..], 0..) |text, index| {
757 counts[index] = @intCast(integer(text));
758 size += counts[index];
759 }
760 try line(out, "macwilliams n {d} size {d}", .{ n, size });
761 try out.writeAll("dual");
762 var feasible = size & (size - 1) == 0;
763 for (0..n + 1) |j| {
764 var numerator: i64 = 0;
765 for (0..n + 1) |x| {
766 var krawtchouk: i64 = 0;
767 for (0..j + 1) |y| {
768 const sign: i64 = if (y % 2 == 0) 1 else -1;
769 krawtchouk += sign * binomial(x, y) * binomial(n - x, j - y);
770 }
771 numerator += counts[x] * krawtchouk;
772 }
773 const divisor = gcd(numerator, size);
774 const p = @divTrunc(numerator, divisor);
775 const q = @divTrunc(size, divisor);
776 if (q == 1) try out.print(" {d}", .{p}) else try out.print(" {d}/{d}", .{ p, q });
777 if (q != 1 or p < 0) feasible = false;
778 }
779 try out.writeByte('\n');
780 try line(out, "feasible {s}", .{yes(feasible)});
781 }
782 fn execute(session: *Session, words: []const []const u8) Error!void {
783 const verb = try session.parse(words);
784 const args = words[1..];
785 const lab = session.lab;
786 const out = session.out;
787 if (@backingInt(verb) <= @backingInt(Verb.PRODUCT)) return session.define(verb, args);
788 switch (verb) {
789 .CRC, .CRCCHECK, .FACTOR, .MACWILLIAMS, .BOUNDS => return session.independent(verb, args),
790 .LIST => {
791 const entries = try session.gpa.dupe(Entry, lab.codes.items);
792 std.sort.heap(Entry, entries, {}, struct {
793 fn less(_: void, a: Entry, b: Entry) bool {
794 return std.mem.order(u8, a.name.slice(), b.name.slice()) == .lt;
795 }
796 }.less);
797 try line(out, "codes {d}", .{entries.len});
798 for (entries) |entry| try line(out, "{f} n {d} k {d} encoder {s}", .{ entry.name, entry.code.n, entry.code.k(), entry.code.encoder() });
799 return;
800 },
801 .DELETE => {
802 _ = lab.codes.orderedRemove(lab.find(args[0]).?);
803 return line(out, "deleted {s}", .{args[0]});
804 },
805 else => {},
806 }
807 const name = args[0];
808 const code = lab.get(name);
809 const n = code.n;
810 const k = code.k();
811 switch (verb) {
812 .SHOW => {
813 try line(out, "code {s} n {d} k {d} encoder {s}", .{ name, n, k, code.encoder() });
814 if (code.polynomial) |p| try line(out, "poly g {f} h {f}", .{ bits(p, n - k + 1), bits(divide(one(n) | 1, p).quotient, k + 1) });
815 for (code.g.slice()) |row| try line(out, "G {f}", .{bits(row, n)});
816 for (code.h.slice()) |row| try line(out, "H {f}", .{bits(row, n)});
817 },
818 .INFO => {
819 var distance = n;
820 for (code.words[1..]) |word| distance = @min(distance, weight(word));
821 const t = (distance - 1) / 2;
822 const divisor: usize = @intCast(gcd(@intCast(n), @intCast(k)));
823 var cyclic = true;
824 for (code.g.slice()) |row| {
825 const rotated = ((row << 1) & (one(n) - 1)) | (row >> @intCast(n - 1));
826 if (code.syndrome(rotated) != 0) cyclic = false;
827 }
828 try out.print("info {s} n {d} k {d} rate {d}/{d} d {d} t {d} perfect {s} mds {s} cyclic ", .{
829 name, n, k, k / divisor, n / divisor, distance, t,
830 yes(volume(n, t) == @as(i64, one(n - k))), yes(distance == n - k + 1),
831 });
832 if (cyclic) try line(out, "{f}", .{bits(code.words[1], degree(code.words[1]) + 1)}) else try line(out, "no", .{});
833 },
834 .SYSTEMATIC => try session.systematic(name, code),
835 .WEIGHTS => {
836 const a = distribution(code.words);
837 var b: [25]usize = @splat(0);
838 for (0..one(code.h.len)) |message| b[weight(spanWord(code.h, message))] += 1;
839 var orthogonal = true;
840 for (code.g.slice()) |x| for (code.g.slice()) |y| {
841 if (weight(x & y) % 2 != 0) orthogonal = false;
842 };
843 try out.print("weights {s} ", .{name});
844 try weightList(out, a[0 .. n + 1]);
845 try out.writeByte('\n');
846 try out.print("dual {s} ", .{name});
847 try weightList(out, b[0 .. n + 1]);
848 try out.writeByte('\n');
849 try line(out, "duality {s} self-orthogonal {s} self-dual {s} formally-self-dual {s}", .{
850 name, yes(orthogonal), yes(orthogonal and n == 2 * k), yes(std.mem.eql(usize, &a, &b)),
851 });
852 },
853 .COVERING => {
854 const counts = distribution(try code.leaders(lab.gpa));
855 var radius = n;
856 while (counts[radius] == 0) : (radius -= 1) {}
857 try out.print("covering {s} radius {d} leaders", .{ name, radius });
858 for (counts[0 .. radius + 1], 0..) |count, index| try out.print(" {d}:{d}", .{ index, count });
859 try out.writeByte('\n');
860 },
861 .BURST => try session.burst(name, code),
862 .TRELLIS => {
863 var profile: [25]usize = undefined;
864 var maximum: usize = 0;
865 for (0..n + 1) |cut| {
866 const right = n - cut;
867 var prefix: Matrix = .{};
868 var suffix: Matrix = .{};
869 for (code.g.slice()) |row| {
870 prefix.add(row >> @intCast(right));
871 suffix.add(row & (one(right) - 1));
872 }
873 profile[cut] = basis(prefix.slice(), cut).len + basis(suffix.slice(), right).len - k;
874 maximum = @max(maximum, profile[cut]);
875 }
876 try out.print("trellis {s} profile", .{name});
877 for (profile[0 .. n + 1]) |value| try out.print(" {d}", .{value});
878 try line(out, " max {d}", .{maximum});
879 },
880 .STDARRAY => {
881 if (n - k > 6 or k > 4) return session.fail("too large", .{});
882 const leaders = try code.leaders(lab.gpa);
883 const syndromes = try session.gpa.alloc(usize, leaders.len);
884 for (syndromes, 0..) |*s, index| s.* = index;
885 std.sort.heap(usize, syndromes, leaders, struct {
886 fn less(values: []const Word, a: usize, b: usize) bool {
887 const x = values[a];
888 const y = values[b];
889 return if (weight(x) != weight(y)) weight(x) < weight(y) else x < y;
890 }
891 }.less);
892 try line(out, "stdarray {s} rows {d} columns {d}", .{ name, leaders.len, code.words.len });
893 for (syndromes) |s| {
894 try out.print("{f}:", .{bits(@intCast(s), n - k)});
895 for (code.words) |word| try out.print(" {f}", .{bits(leaders[s] ^ word, n)});
896 try out.writeByte('\n');
897 }
898 },
899 .EQUAL => {
900 const other = lab.get(args[1]);
901 if (n != other.n) return line(out, "equal {s} {s} no lengths {d} {d}", .{ name, args[1], n, other.n });
902 var combined = code.g;
903 for (other.g.slice()) |row| combined.add(row);
904 const common = k + other.k() - basis(combined.slice(), n).len;
905 try line(out, "equal {s} {s} {s} intersection {d}", .{ name, args[1], yes(k == common and other.k() == common), common });
906 },
907 .ENCODE => {
908 if (args[1].len != k) return session.fail("bad length", .{});
909 try line(out, "encode {s} message {s} codeword {f}", .{ name, args[1], bits(code.encode(binary(args[1])), n) });
910 },
911 .SYNDROME, .DECODE, .BOUNDED, .COSET => {
912 const text = args[if (verb == .BOUNDED) @as(usize, 2) else 1];
913 if (text.len != n) return session.fail("bad length", .{});
914 const word = binary(text);
915 const s = code.syndrome(word);
916 if (verb == .SYNDROME) return line(out, "syndrome {s} {f} codeword {s}", .{ name, bits(@intCast(s), n - k), yes(s == 0) });
917 const decoded = try code.decode(lab.gpa, word);
918 if (verb == .COSET) {
919 var counts: [25]usize = @splat(0);
920 for (code.words) |c| counts[weight(c ^ word)] += 1;
921 try out.print("coset {s} syndrome {f} leader {f} weights ", .{ name, bits(@intCast(s), n - k), bits(decoded.err, n) });
922 try weightList(out, counts[0 .. n + 1]);
923 return out.writeByte('\n');
924 }
925 if (verb == .BOUNDED and weight(decoded.err) > integer(args[1])) return line(out, "bounded {s} fail weight {d}", .{ name, weight(decoded.err) });
926 try out.print("{s} {s} ", .{ if (verb == .BOUNDED) "bounded" else "decode", name });
927 try session.decodedText(code, decoded);
928 try out.writeByte('\n');
929 },
930 .ERASURE => {
931 const text = args[1];
932 if (text.len != n) return session.fail("bad length", .{});
933 var mask: Word = 0;
934 var known: Word = 0;
935 var erased: usize = 0;
936 for (text) |character| {
937 mask = (mask << 1) | @intFromBool(character != '?');
938 known = (known << 1) | @intFromBool(character == '1');
939 erased += @intFromBool(character == '?');
940 }
941 var count: usize = 0;
942 var candidate: Word = 0;
943 for (code.words) |word| if (word & mask == known) {
944 count += 1;
945 candidate = word;
946 };
947 try out.print("erasure {s} erased {d} candidates {d}", .{ name, erased, count });
948 if (count == 1) try out.print(" codeword {f} message {f}", .{ bits(candidate, n), bits(code.messageFor(candidate), k) });
949 try out.writeByte('\n');
950 },
951 .INTERLEAVE => try session.interleave(name, code, args),
952 else => unreachable,
953 }
954 }
955};

957fn readLine(gpa: Allocator, input: *std.Io.Reader) ![]const u8 {
958 var buffer: Writer.Allocating = .init(gpa);
959 defer buffer.deinit();
960 _ = try input.streamDelimiterEnding(&buffer.writer, '\n');
961 if (input.seek < input.end) input.toss(1) else if (buffer.written().len == 0) return error.EndOfStream;
962 const text = try buffer.toOwnedSlice();
963 return if (text.len > 0 and text[text.len - 1] == '\r') text[0 .. text.len - 1] else text;
964}
965fn splitWords(gpa: Allocator, text: []const u8) Allocator.Error![]const []const u8 {
966 var result: std.ArrayList([]const u8) = .empty;
967 var tokens = std.mem.tokenizeScalar(u8, text, ' ');
968 while (tokens.next()) |word| try result.append(gpa, word);
969 return result.toOwnedSlice(gpa);
970}
971pub fn main(init: std.process.Init) !void {
972 var storage = std.heap.ArenaAllocator.init(std.heap.page_allocator);
973 defer storage.deinit();
974 var lab: Laboratory = .{ .gpa = storage.allocator() };
975 var input_buffer: [4096]u8 = undefined;
976 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
977 var output_buffer: [4096]u8 = undefined;
978 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
979 const count = integer(try readLine(lab.gpa, &input.interface));
980 for (0..count) |_| {
981 var scratch = std.heap.ArenaAllocator.init(std.heap.page_allocator);
982 defer scratch.deinit();
983 const gpa = scratch.allocator();
984 const words = try splitWords(gpa, try readLine(gpa, &input.interface));
985 var response: Writer.Allocating = .init(gpa);
986 defer response.deinit();
987 var session: Session = .{ .lab = &lab, .gpa = gpa, .out = &response.writer };
988 session.execute(words) catch |err| {
989 if (err != error.Invalid) return err;
990 try line(&output.interface, "error: {s}", .{session.problem});
991 try output.interface.flush();
992 continue;
993 };
994 try output.interface.writeAll(response.written());
995 try output.interface.flush();
996 }
