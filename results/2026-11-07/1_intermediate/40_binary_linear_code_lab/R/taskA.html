60poly_div <- function(value, divisor) {
61 quotient <- 0L
62 while (value != 0L && bit_length(value) >= bit_length(divisor)) {
63 shift <- bit_length(value) - bit_length(divisor)
64 quotient <- bitwXor(quotient, power2(shift))
65 value <- bitwXor(value, bitwShiftL(divisor, shift))
66 }
67 list(quotient = quotient, remainder = value)

69basis <- function(source, size) {
70 pivots <- integer(size)
71 result <- integer()
72 for (original in source) {
73 row <- original
74 for (column in seq_len(size)) {
75 if (bitwAnd(row, bit_at(size, column)) == 0L) next
76 if (pivots[column] != 0L) row <- bitwXor(row, pivots[column]) else {
77 pivots[column] <- row
78 result <- c(result, original)
79 break
80 }
81 }
82 }
83 result

85rref <- function(source, size) {
86 rows <- source
87 columns <- integer()
88 for (column in seq_len(size)) {
89 pivot <- length(columns) + 1L
90 found <- which(seq_along(rows) >= pivot & bitwAnd(rows, bit_at(size, column)) != 0L)
91 if (!length(found)) next
92 rows[c(pivot, found[1L])] <- rows[c(found[1L], pivot)]
93 others <- which(seq_along(rows) != pivot & bitwAnd(rows, bit_at(size, column)) != 0L)
94 rows[others] <- bitwXor(rows[others], rows[pivot])
95 columns <- c(columns, column)
96 }
97 list(rows = rows[seq_along(columns)], columns = columns)
98}

111reed_muller <- function(order, variables) {
112 if (order == 0L) return(mask(power2(variables)))
113 half <- power2(variables - 1L)
114 first <- reed_muller(min(order, variables - 1L), variables - 1L)
115 c(bitwOr(bitwShiftL(first, half), first), reed_muller(order - 1L, variables - 1L))

144leaders <- function(code) {
145 result <- rep(-1L, power2(code$n - code$k))
146 result[1L] <- 0L
147 found <- 1L
148 for (size in seq_len(code$n)) {
149 word <- mask(size)
150 while (word < power2(code$n)) {
151 state <- syndrome(code, word) + 1L
152 if (result[state] == -1L) {
153 result[state] <- word
154 found <- found + 1L
155 if (found == length(result)) return(result)
156 }
157 lowest <- bitwAnd(word, -word)
158 following <- word + lowest
159 word <- bitwOr(following, bitwShiftR(bitwXor(following, word), 2L) %/% lowest)
160 }
161 }
162 result
163}
164decode <- function(code, received, errors, encoding) {
165 pattern <- errors[syndrome(code, received) + 1L]
166 word <- bitwXor(received, pattern)
167 list(word = word, message = match(word, encoding) - 1L, error = pattern)

171burst_correction <- function(code) {
172 seen <- logical(power2(code$n - code$k))
173 seen[1L] <- TRUE
174 for (size in seq_len(code$n)) {
175 variants <- if (size == 1L) 1L else power2(size - 2L)
176 for (offset in seq_len(code$n - size + 1L) - 1L) {
177 for (middle in seq_len(variants) - 1L) {
178 word <- if (size == 1L) 1L else bitwOr(bitwOr(power2(size - 1L), bitwShiftL(middle, 1L)), 1L)
179 state <- syndrome(code, bitwShiftL(word, offset)) + 1L
180 if (seen[state]) return(size - 1L)
181 seen[state] <- TRUE
182 }
183 }
184 }
185 0L
186}

188construct <- function(command, args, sources) {
189 if (command %in% c("GEN", "PCHECK")) {
190 size <- nchar(args[2L])
191 if (any(nchar(args[-1L]) != size)) fail("bad length")
192 rows <- basis(parse_bits(args[-1L]), size)
193 return(if (command == "GEN") make_code(size, rows) else make_code(size, null_space(rows, size), rows))
194 }
195 if (command %in% c("HAMMING", "SIMPLEX")) {
196 order <- as.integer(args[2L])
197 if (!order %in% 2L:4L) fail("bad parameter")
198 size <- mask(order)
199 rows <- vapply(seq_len(order), function(i) {
200 Reduce(function(row, column) bitwOr(bitwShiftL(row, 1L), bitwAnd(bitwShiftR(column, order - i), 1L)),
201 seq_len(size), init = 0L)
202 }, integer(1L))
203 return(if (command == "SIMPLEX") make_code(size, rows) else make_code(size, null_space(rows, size), rows))
204 }
205 if (command %in% c("REPETITION", "PARITY")) {
206 size <- as.integer(args[2L])
207 if (!size %in% 2L:24L) fail("bad parameter")
208 rows <- if (command == "REPETITION") mask(size) else bitwOr(bit_at(size, seq_len(size - 1L)), 1L)
209 return(make_code(size, rows))
210 }
211 if (command == "RM") {
212 order <- as.integer(args[2L]); variables <- as.integer(args[3L])
213 if (!variables %in% 1L:4L || order > variables) fail("bad parameter")
214 return(make_code(power2(variables), reed_muller(order, variables)))
215 }
216 if (command == "CYCLIC") {
217 size <- as.integer(args[2L]); degree <- nchar(args[3L]) - 1L
218 if (!size %in% 2L:24L || substr(args[3L], 1L, 1L) != "1" || degree < 1L || degree >= size) fail("bad parameter")
219 polynomial <- parse_bits(args[3L])
220 if (poly_div(bitwOr(power2(size), 1L), polynomial)$remainder != 0L) fail("not a divisor")
221 dimension <- size - degree
222 rows <- bitwShiftL(polynomial, dimension - seq_len(dimension))
223 parity <- integer(degree)
224 for (column in seq_len(size)) {
225 residue <- poly_div(bit_at(size, column), polynomial)$remainder
226 parity <- bitwOr(bitwShiftL(parity, 1L), as.integer(bitwAnd(residue, bit_at(degree, seq_len(degree))) != 0L))
227 }
228 return(make_code(size, rows, parity, polynomial))
229 }
230 source <- sources[[1L]]
231 size <- source$n
232 if (command == "DUAL") return(make_code(size, source$h, source$g))
233 if (command == "EXTEND") return(make_code(size + 1L, bitwOr(bitwShiftL(source$g, 1L), weight(source$g) %% 2L)))
234 if (command %in% c("PUNCTURE", "SHORTEN", "EXPURGATE")) {
235 expurgating <- command == "EXPURGATE"
236 position <- if (expurgating) 0L else as.integer(args[3L])
237 if (!expurgating && !position %in% seq_len(size)) fail("bad position")
238 selected <- if (expurgating) weight(source$g) %% 2L != 0L else bitwAnd(source$g, bit_at(size, position)) != 0L
239 removed <- if (command == "PUNCTURE" || !any(selected)) 0L else which(selected)[1L]
240 rows <- source$g
241 if (removed != 0L) {
242 rows[selected] <- bitwXor(rows[selected], source$g[removed])
243 rows <- rows[-removed]
244 }
245 if (!expurgating) {
246 low <- size - position
247 rows <- bitwOr(bitwShiftL(bitwShiftR(rows, low + 1L), low), bitwAnd(rows, mask(low)))
248 size <- size - 1L
249 }
250 if (command == "PUNCTURE") rows <- basis(rows, size)
251 return(make_code(size, rows))
252 }
253 if (command == "AUGMENT") return(make_code(size, basis(c(source$g, mask(size)), size)))
254 if (command == "PERMUTE") {
255 permutation <- as.integer(args[-c(1L, 2L)])
256 if (!identical(sort(permutation), seq_len(size))) fail("bad permutation")
257 rows <- vapply(source$g, function(row) {
258 Reduce(function(value, column) bitwOr(bitwShiftL(value, 1L), as.integer(bitwAnd(row, bit_at(size, column)) != 0L)),
259 permutation, init = 0L)
260 }, integer(1L))
261 return(make_code(size, rows))
262 }
263 other <- sources[[2L]]
264 if (command == "UUV" && size != other$n) fail("bad length")
265 product <- command == "PRODUCT"
266 combined <- if (product) size * other$n else size + other$n
267 dimension <- if (product) source$k * other$k else source$k + other$k
268 check_dimension(combined, dimension)
269 if (product) {
270 rows <- unlist(lapply(source$g, function(left) vapply(other$g, function(right) {
271 Reduce(function(value, column) bitwOr(bitwShiftL(value, other$n), if (bitwAnd(left, bit_at(size, column)) != 0L) right else 0L),
272 seq_len(size), init = 0L)
273 }, integer(1L))), use.names = FALSE)
274 } else rows <- c(bitwOr(bitwShiftL(source$g, other$n), if (command == "UUV") source$g else 0L), other$g)
275 make_code(combined, rows)
276}

301systematic <- function(name, code) {
302 rows <- code$g
303 permutation <- seq_len(code$n)
304 for (pivot in seq_len(code$k)) {
305 column <- pivot
306 repeat {
307 found <- which(seq_along(rows) >= pivot & bitwAnd(rows, bit_at(code$n, column)) != 0L)
308 if (length(found)) break
309 column <- column + 1L
310 }
311 if (column != pivot) {
312 left <- bit_at(code$n, pivot); right <- bit_at(code$n, column)
313 different <- (bitwAnd(rows, left) != 0L) != (bitwAnd(rows, right) != 0L)
314 rows[different] <- bitwXor(rows[different], bitwOr(left, right))
315 permutation[c(pivot, column)] <- permutation[c(column, pivot)]
316 }
317 rows[c(pivot, found[1L])] <- rows[c(found[1L], pivot)]
318 others <- which(seq_along(rows) != pivot & bitwAnd(rows, bit_at(code$n, pivot)) != 0L)
319 rows[others] <- bitwXor(rows[others], rows[pivot])
320 }
321 checks <- vapply(seq_len(code$n - code$k), function(column) {
322 first <- Reduce(function(value, row) bitwOr(bitwShiftL(value, 1L), as.integer(bitwAnd(row, bit_at(code$n, code$k + column)) != 0L)),
323 rows, init = 0L)
324 bitwOr(bitwShiftL(first, code$n - code$k), bit_at(code$n - code$k, column))
325 }, integer(1L))
326 c(paste("systematic", name, "perm", joined(permutation)), matrix_text("G", rows, code$n), matrix_text("H", checks, code$n))
327}
328inspect_code <- function(command, args, sources) {
329 code <- sources[[1L]]
330 name <- args[1L]; n <- code$n; k <- code$k
331 if (command == "SHOW") {
332 output <- paste("code", name, "n", n, "k", k, "encoder", encoder(code))
333 if (!is.null(code$polynomial)) {
334 quotient <- poly_div(bitwOr(power2(n), 1L), code$polynomial)$quotient
335 output <- c(output, paste("poly g", bits(code$polynomial, n - k + 1L), "h", bits(quotient, k + 1L)))
336 }
337 return(c(output, matrix_text("G", code$g, n), matrix_text("H", code$h, n)))
338 }
339 if (command == "SYSTEMATIC") return(systematic(name, code))
340 if (command == "EQUAL") {
341 other <- sources[[2L]]
342 if (n != other$n) return(paste("equal", name, args[2L], "no lengths", n, other$n))
343 common <- k + other$k - length(basis(c(code$g, other$g), n))
344 return(paste("equal", name, args[2L], yes(k == common && other$k == common), "intersection", common))
345 }
346 if (command == "TRELLIS") {
347 profile <- vapply(0L:n, function(column) {
348 length(basis(bitwShiftR(code$g, n - column), column)) +
349 length(basis(bitwAnd(code$g, mask(n - column)), n - column)) - k
350 }, integer(1L))
351 return(paste("trellis", name, "profile", joined(profile), "max", max(profile)))
352 }
353 if (command == "COVERING") {
354 counts <- distribution(leaders(code), n)
355 radius <- max(which(counts != 0L)) - 1L
356 items <- paste0(0L:radius, ":", counts[seq_len(radius + 1L)])
357 return(paste("covering", name, "radius", radius, "leaders", joined(items)))
358 }
359 if (command == "STDARRAY" && (n - k > 6L || k > 4L)) fail("too large")
360 words <- span(code$g)
361 if (command == "INFO") {
362 distance <- min(weight(words[-1L]))
363 correction <- (distance - 1L) %/% 2L
364 rotated <- bitwOr(bitwAnd(bitwShiftL(code$g, 1L), mask(n)), bitwShiftR(code$g, n - 1L))
365 first <- min(words[-1L])
366 cyclic <- if (all(syndrome(code, rotated) == 0L)) bits(first, bit_length(first)) else "no"
367 return(paste("info", name, "n", n, "k", k, "rate", fraction_text(k, n), "d", distance, "t", correction,
368 "perfect", yes(volume(n, correction) == power2(n - k)), "mds", yes(distance == n - k + 1L), "cyclic", cyclic))
369 }
370 if (command == "WEIGHTS") {
371 primal <- distribution(words, n); dual <- distribution(span(code$h), n)
372 orthogonal <- all(weight(base::outer(code$g, code$g, bitwAnd)) %% 2L == 0L)
373 return(c(paste("weights", name, weight_items(primal)), paste("dual", name, weight_items(dual)),
374 paste("duality", name, "self-orthogonal", yes(orthogonal), "self-dual", yes(orthogonal && n == 2L * k),
375 "formally-self-dual", yes(identical(primal, dual)))))
376 }
377 if (command == "BURST") {
378 nonzero <- words[-1L]
379 detection <- min(bit_length(nonzero) - bit_length(bitwAnd(nonzero, -nonzero)))
380 return(paste("burst", name, "detect", detection, "correct", burst_correction(code)))
381 }
382 errors <- leaders(code)
383 ordering <- order(weight(errors), errors, method = "radix")
384 words <- sort(words)
385 output <- paste("stdarray", name, "rows", length(errors), "columns", length(words))
386 for (state in ordering) {
387 row <- vapply(bitwXor(errors[state], words), bits, character(1L), size = n)
388 output <- c(output, paste0(bits(state - 1L, n - k), ": ", joined(row)))
389 }
390 output
391}
392encoding_report <- function(command, args, code) {
393 name <- args[1L]; n <- code$n; k <- code$k
394 if (command == "INTERLEAVE") {
395 messages <- args[-c(1L, 2L, 3L)]
396 if (any(nchar(messages) != k)) fail("bad length")
397 start <- as.integer(args[2L]); burst_length <- as.integer(args[3L]); depth <- length(messages)
398 if (start < 1L || burst_length < 1L || start + burst_length - 1L > depth * n) fail("bad burst")
399 encoded <- vapply(parse_bits(messages), function(message) encode(code, message), integer(1L))
400 sent_matrix <- matrix(FALSE, depth, n)
401 for (column in seq_len(n)) sent_matrix[, column] <- bitwAnd(encoded, bit_at(n, column)) != 0L
402 sent <- as.vector(sent_matrix)
403 received <- sent
404 selected <- seq_len(burst_length) + start - 1L
405 received[selected] <- !received[selected]
406 received_matrix <- matrix(received, nrow = depth)
407 received_words <- integer(depth)
408 for (column in seq_len(n)) received_words <- bitwOr(bitwShiftL(received_words, 1L), as.integer(received_matrix[, column]))
409 output <- c(paste("interleave", name, "depth", depth, "burst", start, burst_length),
410 paste("sent", paste(as.integer(sent), collapse = "")), paste("recv", paste(as.integer(received), collapse = "")))
411 errors <- leaders(code); encoding <- encoded_words(code); correct <- 0L
412 for (row in seq_len(depth)) {
413 result <- decode(code, received_words[row], errors, encoding)
414 success <- result$message == parse_bits(messages[row])
415 if (success) correct <- correct + 1L
416 output <- c(output, paste("row", row, "errors", weight(bitwXor(received_words[row], encoded[row])),
417 "corrected", weight(result$error), "message", bits(result$message, k), if (success) "ok" else "wrong"))
418 }
419 return(c(output, paste("result ok", correct, "wrong", depth - correct)))
420 }
421 text <- args[if (command == "BOUNDED") 3L else 2L]
422 expected <- if (command == "ENCODE") k else n
423 if (nchar(text) != expected) fail("bad length")
424 if (command == "ENCODE") return(paste("encode", name, "message", text, "codeword", bits(encode(code, parse_bits(text)), n)))
425 if (command == "ERASURE") {
426 units <- strsplit(text, "", fixed = TRUE)[[1L]]
427 erased <- sum(units == "?")
428 known <- Reduce(bitwOr, bit_at(n, which(units != "?")), init = 0L)
429 value <- Reduce(bitwOr, bit_at(n, which(units == "1")), init = 0L)
430 words <- span(code$g)
431 candidates <- words[bitwAnd(words, known) == value]
432 output <- paste("erasure", name, "erased", erased, "candidates", length(candidates))
433 if (length(candidates) == 1L) output <- paste(output, "codeword", bits(candidates[1L], n),
434 "message", bits(match(candidates[1L], encoded_words(code)) - 1L, k))
435 return(output)
436 }
437 word <- parse_bits(text)
438 state <- syndrome(code, word)
439 if (command == "SYNDROME") return(paste("syndrome", name, bits(state, n - k), "codeword", yes(state == 0L)))
440 errors <- leaders(code)
441 if (command == "COSET") return(paste("coset", name, "syndrome", bits(state, n - k), "leader", bits(errors[state + 1L], n),
442 "weights", weight_items(distribution(span(code$g), n, word))))
443 result <- decode(code, word, errors, encoded_words(code))
444 if (command == "BOUNDED" && weight(result$error) > as.integer(args[2L]))
445 return(paste("bounded", name, "fail weight", weight(result$error)))
446 paste(tolower(command), name, decoded_text(code, result))
447}

449independent_report <- function(command, args) {
450 if (command %in% c("CRC", "CRCCHECK")) {
451 degree <- nchar(args[1L]) - 1L
452 if (substr(args[1L], 1L, 1L) != "1" || !degree %in% 1L:16L) fail("bad parameter")
453 if (command == "CRC" && nchar(args[2L]) + degree > 30L) fail("bad length")
454 word <- bitwShiftL(parse_bits(args[2L]), if (command == "CRC") degree else 0L)
455 residue <- poly_div(word, parse_bits(args[1L]))$remainder
456 suffix <- bits(residue, degree)
457 return(if (command == "CRC") paste("crc remainder", suffix, "frame", paste0(args[2L], suffix)) else
458 paste("crccheck", if (residue == 0L) "ok" else "error", "remainder", suffix))
459 }
460 size <- as.integer(args[1L])
461 if (command == "BOUNDS") {
462 distance <- as.integer(args[2L])
463 if (distance < 1L || distance > size || size > 24L) fail("bad parameter")
464 ball <- as.bigz(volume(size, (distance - 1L) %/% 2L))
465 hamming <- max(which(as.bigz(power2(0L:size)) * ball <= power2(size))) - 1L
466 requirements <- (distance + power2(0L:(size - 1L)) - 1L) %/% power2(0L:(size - 1L))
467 griesmer <- max(which(cumsum(requirements) <= size))
468 return(paste("bounds n", size, "d", distance, "singleton", size - distance + 1L, "hamming", hamming, "griesmer", griesmer))
469 }
470 if (!size %in% 1L:24L) fail("bad parameter")
471 if (command == "FACTOR") {
472 remaining <- bitwOr(power2(size), 1L)
473 factor <- 3L; divisors <- 1L; factors <- character()
474 while (2L * (bit_length(factor) - 1L) <= bit_length(remaining) - 1L) {
475 exponent <- 0L
476 repeat {
477 divided <- poly_div(remaining, factor)
478 if (divided$remainder != 0L) break
479 remaining <- divided$quotient
480 exponent <- exponent + 1L
481 }
482 if (exponent != 0L) {
483 factors <- c(factors, paste0(bits(factor, bit_length(factor)), "^", exponent))
484 divisors <- divisors * (exponent + 1L)
485 }
486 factor <- factor + 2L
487 }
488 if (remaining != 1L) {
489 factors <- c(factors, paste0(bits(remaining, bit_length(remaining)), "^1"))
490 divisors <- divisors * 2L
491 }
492 return(paste("factor", size, joined(factors), "divisors", divisors))
493 }
494 if (length(args) != size + 2L) fail("bad length")
495 counts <- as.integer(args[-1L])
496 if (counts[1L] != 1L) fail("bad parameter")
497 total <- sum(counts)
498 numerators <- c_bigz(lapply(0L:size, function(index) {
499 coefficients <- vapply(0L:size, function(argument) krawtchouk(size, index, argument), integer(1L))
500 sum(as.bigz(counts) * coefficients)
501 }))
502 feasible <- bitwAnd(total, total - 1L) == 0L && all(numerators >= 0L & numerators %% total == 0L)
503 dual <- vapply(seq_len(size + 1L), function(i) fraction_text(numerators[i], total), character(1L))
504 c(paste("macwilliams n", size, "size", total), paste("dual", joined(dual)), paste("feasible", yes(feasible)))
505}

507execute <- function(line, registry) {
508 words <- strsplit(line, " ", fixed = TRUE)[[1L]]
509 command <- words[1L]
510 form <- FORMATS[[command]]
511 if (is.null(form)) fail(paste("unknown command", command))
512 args <- words[-1L]
513 if (length(args) < form$minimum || length(args) > form$maximum) fail("bad arguments")
514 types <- strsplit(form$types, "", fixed = TRUE)[[1L]]
515 for (i in seq_along(args)) if (!valid_word(args[i], types[min(i, length(types))])) fail("bad arguments")
516 sources <- list()
517 for (i in seq_along(types)) if (types[i] %in% c("A", "B")) {
518 if (!exists(args[i], registry, inherits = FALSE)) fail(paste("no code", args[i]))
519 sources[[length(sources) + 1L]] <- get(args[i], registry, inherits = FALSE)
520 }
521 if (command %in% DEFINITIONS) {
522 code <- construct(command, args, sources)
523 action <- if (exists(args[1L], registry, inherits = FALSE)) "redefined" else "defined"
524 assign(args[1L], code, registry)
525 return(paste(action, args[1L], "n", code$n, "k", code$k))
526 }
527 if (command %in% INDEPENDENT) return(independent_report(command, args))
528 if (command == "LIST") {
529 names <- sort(ls(registry, all.names = TRUE), method = "radix")
530 rows <- vapply(names, function(name) {
531 code <- get(name, registry, inherits = FALSE)
532 paste(name, "n", code$n, "k", code$k, "encoder", encoder(code))
533 }, character(1L))
534 return(c(paste("codes", length(names)), rows))
535 }
536 if (command == "DELETE") {
537 rm(list = args[1L], envir = registry)
538 return(paste("deleted", args[1L]))
539 }
540 if (command %in% c("ENCODE", "SYNDROME", "DECODE", "BOUNDED", "COSET", "ERASURE", "INTERLEAVE"))
541 encoding_report(command, args, sources[[1L]]) else inspect_code(command, args, sources)
542}
543
544registry <- new.env(hash = TRUE, parent = emptyenv())
545input <- file("stdin", open = "r")
546command_count <- as.integer(readLines(input, n = 1L, warn = FALSE))
547for (command_index in seq_len(command_count)) {
548 line <- readLines(input, n = 1L, warn = FALSE)
549 output <- tryCatch(execute(line, registry), command_error = function(problem) paste("error:", problem$message))
550 cat(paste0(output, "\n"), sep = "")
551}
552close(input)
