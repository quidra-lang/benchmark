46struct Division { Word quotient, remainder; };
47Division divide(Word dividend, Word divisor) {
48 Word quotient = 0;
49 const int d = degree(divisor);
50 while (dividend && degree(dividend) >= d) {
51 const int shift = degree(dividend) - d;
52 quotient ^= Word{1} << shift;
53 dividend ^= divisor << shift;
54 }
55 return {quotient, dividend};
56}
57Matrix basis(const Matrix& rows, int length) {
58 Matrix pivots(length), kept;
59 for (Word original : rows) {
60 Word row = original;
61 for (int column = 0; column < length; ++column) if (bit(row, length, column)) {
62 if (pivots[column]) row ^= pivots[column];
63 else { pivots[column] = row; kept.push_back(original); break; }
64 }
65 }
66 return kept;
67}
68struct Echelon { Matrix rows; std::vector<int> columns; };
69Echelon rref(Matrix rows, int length) {
70 std::vector<int> columns;
71 for (int column = 0; column < length; ++column) {
72 const int pivot = static_cast<int>(columns.size());
73 int found = pivot;
74 while (found < static_cast<int>(rows.size()) && !bit(rows[found], length, column)) ++found;
75 if (found == static_cast<int>(rows.size())) continue;
76 std::swap(rows[pivot], rows[found]);
77 for (int row = 0; row < static_cast<int>(rows.size()); ++row) if (row != pivot && bit(rows[row], length, column)) rows[row] ^= rows[pivot];
78 columns.push_back(column);
79 }
80 rows.resize(columns.size());
81 return {std::move(rows), std::move(columns)};
82}
83Matrix nullspace(const Matrix& original, int length) {
84 const auto reduced = rref(original, length);
85 Matrix result;
86 for (int column = 0; column < length; ++column) {
87 if (std::find(reduced.columns.begin(), reduced.columns.end(), column) != reduced.columns.end()) continue;
88 Word word = Word{1} << (length - column - 1);
89 for (std::size_t row = 0; row < reduced.rows.size(); ++row) if (bit(reduced.rows[row], length, column)) word |= Word{1} << (length - reduced.columns[row] - 1);
90 result.push_back(word);
91 }
92 return result;
93}
94Matrix span(const Matrix& rows) {
95 Matrix words{0};
96 for (Word row : rows) {
97 const auto count = words.size();
98 for (std::size_t index = 0; index < count; ++index) words.push_back(words[index] ^ row);
99 }
100 return words;
101}
102std::vector<int> distribution(const Matrix& words, int length) {
103 std::vector<int> counts(length + 1);
104 for (Word word : words) ++counts[weight(word)];
105 return counts;
106}
107std::string weightList(const std::vector<int>& counts) {
108 Lines result;
109 for (std::size_t index = 0; index < counts.size(); ++index) if (counts[index]) result.push_back(std::to_string(index) + ":" + std::to_string(counts[index]));
110 return join(result);
111}
112void dimension(int length, int rank) {
113 if (rank < 1 || rank > 12 || length - rank < 1 || length - rank > 12) fail("bad dimension " + std::to_string(length) + " " + std::to_string(rank));
114}
115Word removeBit(Word word, int length, int column) {
116 const int low = length - column - 1;
117 return ((word >> (low + 1)) << low) | (word & ((Word{1} << low) - 1));
118}
119Matrix reedMuller(int order, int variables) {
120 const int length = 1 << variables;
121 if (order == 0) return {(Word{1} << length) - 1};
122 Matrix result;
123 for (Word row : reedMuller(std::min(order, variables - 1), variables - 1)) result.push_back((row << (length / 2)) | row);
124 const auto lower = reedMuller(order - 1, variables - 1);
125 result.insert(result.end(), lower.begin(), lower.end());
126 return result;
127}

130class Code {
131 std::optional<Matrix> leaderTable;
132public:
133 int n, k;
134 Matrix G, H, words;
135 std::optional<Word> polynomial;
136 std::map<Word, Word> messages;
137
138 Code(int length, Matrix generator, std::optional<Matrix> check = {}, std::optional<Word> poly = {})
139 : n(length), k(static_cast<int>(generator.size())), G(std::move(generator)), polynomial(poly) {
140 dimension(n, k);
141 H = check ? std::move(*check) : nullspace(G, n);
142 for (Word message = 0; message < (Word{1} << k); ++message) {
143 const Word word = encode(message);
144 messages.emplace(word, message); words.push_back(word);
145 }
146 std::sort(words.begin(), words.end());
147 }
148 std::string encoder() const { return polynomial ? "polynomial" : "matrix"; }
149 Word encode(Word message) const {
150 if (polynomial) {
151 const Word shifted = message << (n - k);
152 return shifted | divide(shifted, *polynomial).remainder;
153 }
154 Word word = 0;
155 for (int row = 0; row < k; ++row) if (bit(message, k, row)) word ^= G[row];
156 return word;
157 }
158 Word syndrome(Word word) const {
159 Word result = 0;
160 for (Word row : H) result = (result << 1) | (weight(row & word) % 2);
161 return result;
162 }
163 const Matrix& leaders() {
164 if (leaderTable) return *leaderTable;
165 const Word absent = std::numeric_limits<Word>::max();
166 Matrix leaders(Word{1} << (n - k), absent);
167 leaders[0] = 0;
168 std::size_t remaining = leaders.size() - 1;
169 for (int count = 1; remaining; ++count) {
170 for (Word word = (Word{1} << count) - 1; word < (Word{1} << n);) {
171 const Word s = syndrome(word);
172 if (leaders[s] == absent) {
173 leaders[s] = word;
174 if (--remaining == 0) break;
175 }
176 const Word lowest = word & (Word{0} - word), next = word + lowest;
177 word = next | (((next ^ word) >> 2) / lowest);
178 }
179 }
180 leaderTable = std::move(leaders);
181 return *leaderTable;
182 }
183 Decoded decode(Word received) {
184 const Word error = leaders()[syndrome(received)], word = received ^ error;
185 return {error, word, messages.at(word)};
186 }
187 std::string decodedText(const Decoded& value) const {
188 return "codeword " + bits(value.word, n) + " message " + bits(value.message, k) + " error " + bits(value.error, n) + " weight " + std::to_string(weight(value.error));
189 }
190};

230bool valid(const std::string& text, Argument type) {
231 const auto letter = [](char c) { return (c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z'); };
232 const auto digit = [](char c) { return c >= '0' && c <= '9'; };
233 if (text.empty()) return false;
234 if (type == Argument::Name || type == Argument::Code) return text.size() <= 8 && letter(text[0]) &&
235 std::all_of(text.begin(), text.end(), [&](char c) { return letter(c) || digit(c); });
236 if (type == Argument::Number) return text.size() <= 4 && (text.size() == 1 || text[0] != '0') && std::all_of(text.begin(), text.end(), digit);
237 return text.size() <= 30 && std::all_of(text.begin(), text.end(), [&](char c) { return c == '0' || c == '1' || (type == Argument::Erasure && c == '?'); });
238}

259Lines independent(Verb verb, const Lines& args) {
260 if (verb == Verb::Crc || verb == Verb::Crccheck) {
261 const auto& generator = args[0]; const auto& text = args[1];
262 const int d = static_cast<int>(generator.size()) - 1;
263 if (generator[0] == '0' || generator.size() < 2 || generator.size() > 17) fail("bad parameter");
264 if (verb == Verb::Crc && text.size() + d > 30) fail("bad length");
265 const Word word = parseWord(text), divisor = parseWord(generator);
266 const Word remainder = divide(verb == Verb::Crc ? word << d : word, divisor).remainder;
267 const auto suffix = bits(remainder, d);
268 return {verb == Verb::Crc ? "crc remainder " + suffix + " frame " + text + suffix :
269 std::string("crccheck ") + (remainder == 0 ? "ok" : "error") + " remainder " + suffix};
270 }
271 const int n = std::stoi(args[0]);
272 if (verb == Verb::Bounds) {
273 const int d = std::stoi(args[1]);
274 if (d < 1 || d > n || n > 24) fail("bad parameter");
275 const Wide volume = std::accumulate(choose[n].begin(), choose[n].begin() + (d - 1) / 2 + 1, Wide{0});
276 int hamming = 0, griesmer = 0, sum = 0;
277 for (int k = 0; k <= n; ++k) if ((Wide{1} << k) * volume <= (Wide{1} << n)) hamming = k;
278 for (int k = 1; k <= n; ++k) {
279 const int denominator = 1 << (k - 1);
280 sum += (d + denominator - 1) / denominator;
281 if (sum <= n) griesmer = k;
282 }
283 return {"bounds n " + std::to_string(n) + " d " + std::to_string(d) + " singleton " + std::to_string(n - d + 1) +
284 " hamming " + std::to_string(hamming) + " griesmer " + std::to_string(griesmer)};
285 }
286 if (n < 1 || n > 24) fail("bad parameter");
287 if (verb == Verb::Factor) {
288 Word remaining = (Word{1} << n) | 1;
289 int divisors = 1;
290 Lines factors;
291 for (Word candidate = 2; 2 * degree(candidate) <= degree(remaining); ++candidate) {
292 int exponent = 0;
293 while (divide(remaining, candidate).remainder == 0) { remaining = divide(remaining, candidate).quotient; ++exponent; }
294 if (exponent) { factors.push_back(bits(candidate, degree(candidate) + 1) + "^" + std::to_string(exponent)); divisors *= exponent + 1; }
295 }
296 if (remaining > 1) { factors.push_back(bits(remaining, degree(remaining) + 1) + "^1"); divisors *= 2; }
297 return {"factor " + std::to_string(n) + " " + join(factors) + " divisors " + std::to_string(divisors)};
298 }
299 if (args.size() != static_cast<std::size_t>(n + 2)) fail("bad length");
300 if (std::stoi(args[1]) != 1) fail("bad parameter");
301 std::vector<Wide> counts;
302 for (std::size_t index = 1; index < args.size(); ++index) counts.push_back(std::stoi(args[index]));
303 const Wide size = std::accumulate(counts.begin(), counts.end(), Wide{0});
304 bool feasible = (size & (size - 1)) == 0;
305 Lines transformed;
306 for (int j = 0; j <= n; ++j) {
307 Wide numerator = 0;
308 for (int x = 0; x <= n; ++x) {
309 Wide krawtchouk = 0;
310 for (int y = 0; y <= j; ++y) krawtchouk += (y % 2 ? -1 : 1) * binomial(x, y) * binomial(n - x, j - y);
311 numerator += counts[x] * krawtchouk;
312 }
313 const Wide divisor = std::gcd(numerator, size), p = numerator / divisor, q = size / divisor;
314 transformed.push_back(std::to_string(p) + (q == 1 ? "" : "/" + std::to_string(q)));
315 if (q != 1 || p < 0) feasible = false;
316 }
317 return {"macwilliams n " + std::to_string(n) + " size " + std::to_string(size), "dual " + join(transformed), "feasible " + yes(feasible)};
318}

323 Lines define(Verb verb, const Lines& args) {
324 const auto& name = args[0];
325 const auto first = codes.find(args[1]);
326 const auto second = args.size() > 2 ? codes.find(args[2]) : codes.end();
327 const Code* a = first == codes.end() ? nullptr : &first->second;
328 const Code* b = second == codes.end() ? nullptr : &second->second;
329 int n = 0;
330 Matrix G;
331 std::optional<Matrix> H;
332 std::optional<Word> polynomial;
333 switch (verb) {
334 case Verb::Gen: case Verb::Pcheck: {
335 n = static_cast<int>(args[1].size());
336 Matrix rows;
337 for (std::size_t index = 1; index < args.size(); ++index) {
338 if (static_cast<int>(args[index].size()) != n) fail("bad length");
339 rows.push_back(parseWord(args[index]));
340 }
341 rows = basis(rows, n);
342 if (verb == Verb::Gen) G = std::move(rows);
343 else { H = std::move(rows); G = nullspace(*H, n); }
344 break;
345 }
346 case Verb::Hamming: case Verb::Simplex: {
347 const int r = std::stoi(args[1]);
348 if (r < 2 || r > 4) fail("bad parameter");
349 n = (1 << r) - 1;
350 Matrix rows(r);
351 for (int column = 1; column <= n; ++column) for (int row = 0; row < r; ++row) rows[row] = (rows[row] << 1) | bit(column, r, row);
352 if (verb == Verb::Hamming) { H = std::move(rows); G = nullspace(*H, n); }
353 else G = std::move(rows);
354 break;
355 }
356 case Verb::Repetition: case Verb::Parity: {
357 n = std::stoi(args[1]);
358 if (n < 2 || n > 24) fail("bad parameter");
359 if (verb == Verb::Repetition) G.push_back((Word{1} << n) - 1);
360 else for (int row = 0; row < n - 1; ++row) G.push_back((Word{1} << (n - row - 1)) | 1);
361 break;
362 }
363 case Verb::Rm: {
364 const int r = std::stoi(args[1]), m = std::stoi(args[2]);
365 if (m < 1 || m > 4 || r > m) fail("bad parameter");
366 n = 1 << m; G = reedMuller(r, m);
367 break;
368 }
369 case Verb::Cyclic: {
370 n = std::stoi(args[1]);
371 const auto& text = args[2];
372 const int d = static_cast<int>(text.size()) - 1;
373 if (n < 2 || n > 24 || text[0] == '0' || text.size() < 2 || static_cast<int>(text.size()) > n) fail("bad parameter");
374 polynomial = parseWord(text);
375 if (divide((Word{1} << n) | 1, *polynomial).remainder) fail("not a divisor");
376 const int k = n - d;
377 dimension(n, k);
378 for (int row = 0; row < k; ++row) G.push_back(*polynomial << (k - row - 1));
379 H = Matrix(d);
380 for (int column = 0; column < n; ++column) {
381 const Word remainder = divide(Word{1} << (n - column - 1), *polynomial).remainder;
382 for (int row = 0; row < d; ++row) (*H)[row] = ((*H)[row] << 1) | bit(remainder, d, row);
383 }
384 break;
385 }
386 case Verb::Dual: n = a->n; G = a->H; H = a->G; break;
387 case Verb::Extend:
388 n = a->n + 1;
389 for (Word row : a->G) G.push_back((row << 1) | (weight(row) % 2));
390 break;
391 case Verb::Puncture: case Verb::Shorten: {
392 const int column = std::stoi(args[2]) - 1;
393 if (column < 0 || column >= a->n) fail("bad position");
394 Matrix rows = a->G;
395 if (verb == Verb::Shorten) {
396 const auto pivot = std::find_if(rows.begin(), rows.end(), [&](Word row) { return bit(row, a->n, column); });
397 if (pivot != rows.end()) {
398 const Word z = *pivot;
399 rows.erase(pivot);
400 for (Word& row : rows) if (bit(row, a->n, column)) row ^= z;
401 }
402 }
403 n = a->n - 1;
404 for (Word row : rows) G.push_back(removeBit(row, a->n, column));
405 if (verb == Verb::Puncture) G = basis(G, n);
406 break;
407 }
408 case Verb::Augment:
409 n = a->n; G = a->G; G.push_back((Word{1} << n) - 1); G = basis(G, n);
410 break;
411 case Verb::Expurgate: {
412 n = a->n; G = a->G;
413 const auto pivot = std::find_if(G.begin(), G.end(), [](Word row) { return weight(row) % 2 != 0; });
414 if (pivot != G.end()) {
415 const Word z = *pivot;
416 G.erase(pivot);
417 for (Word& row : G) if (weight(row) % 2) row ^= z;
418 }
419 break;
420 }
421 case Verb::Permute: {
422 n = a->n;
423 std::vector<int> order;
424 for (std::size_t index = 2; index < args.size(); ++index) order.push_back(std::stoi(args[index]));
425 const std::set<int> distinct(order.begin(), order.end());
426 if (static_cast<int>(order.size()) != n || static_cast<int>(distinct.size()) != n || *distinct.begin() != 1 || *distinct.rbegin() != n) fail("bad permutation");
427 for (Word row : a->G) {
428 Word word = 0;
429 for (int column : order) word = (word << 1) | bit(row, n, column - 1);
430 G.push_back(word);
431 }
432 break;
433 }
434 case Verb::Directsum: case Verb::Uuv: case Verb::Product: {
435 if (verb == Verb::Uuv && a->n != b->n) fail("bad length");
436 n = verb == Verb::Product ? a->n * b->n : a->n + b->n;
437 dimension(n, verb == Verb::Product ? a->k * b->k : a->k + b->k);
438 if (verb == Verb::Product) {
439 for (Word x : a->G) for (Word y : b->G) {
440 Word word = 0;
441 for (int column = 0; column < a->n; ++column) word = (word << b->n) | (bit(x, a->n, column) ? y : 0);
442 G.push_back(word);
443 }
444 } else {
445 for (Word row : a->G) G.push_back((row << b->n) | (verb == Verb::Uuv ? row : 0));
446 G.insert(G.end(), b->G.begin(), b->G.end());
447 }
448 break;
449 }
450 default: throw std::logic_error("not a defining command");
451 }
452 Code code(n, std::move(G), std::move(H), polynomial);
453 const bool replaced = codes.count(name) != 0;
454 const int k = code.k;
455 codes.insert_or_assign(name, std::move(code));
456 return {std::string(replaced ? "redefined " : "defined ") + name + " n " + std::to_string(n) + " k " + std::to_string(k)};
457 }
458 Lines systematic(const std::string& name, const Code& code) const {
459 const int n = code.n, k = code.k;
460 Matrix rows = code.G;
461 std::vector<int> order(n);
462 std::iota(order.begin(), order.end(), 1);
463 for (int pivot = 0; pivot < k; ++pivot) {
464 int column = pivot;
465 while (std::none_of(rows.begin() + pivot, rows.end(), [&](Word row) { return bit(row, n, column); })) ++column;
466 if (column != pivot) {
467 const Word mask = (Word{1} << (n - pivot - 1)) | (Word{1} << (n - column - 1));
468 for (Word& row : rows) if (bit(row, n, pivot) != bit(row, n, column)) row ^= mask;
469 std::swap(order[pivot], order[column]);
470 }
471 int found = pivot;
472 while (!bit(rows[found], n, pivot)) ++found;
473 std::swap(rows[pivot], rows[found]);
474 for (int row = 0; row < k; ++row) if (row != pivot && bit(rows[row], n, pivot)) rows[row] ^= rows[pivot];
475 }
476 Lines permutation;
477 for (int column : order) permutation.push_back(std::to_string(column));
478 Lines output{"systematic " + name + " perm " + join(permutation)};
479 for (Word row : rows) output.push_back("G " + bits(row, n));
480 for (int column = k; column < n; ++column) {
481 Word row = 0;
482 for (Word value : rows) row = (row << 1) | bit(value, n, column);
483 row = (row << (n - k)) | (Word{1} << (n - column - 1));
484 output.push_back("H " + bits(row, n));
485 }
486 return output;
487 }
488 Lines burst(const std::string& name, const Code& code) const {
489 int detect = code.n;
490 for (Word word : code.words) if (word) detect = std::min(detect, degree(word) - degree(word & (Word{0} - word)));
491 std::set<Word> seen{0};
492 for (int length = 1; length <= code.n; ++length) {
493 for (int start = 0; start + length <= code.n; ++start) {
494 const Word limit = length == 1 ? 1 : Word{1} << (length - 2);
495 for (Word middle = 0; middle < limit; ++middle) {
496 const Word pattern = length == 1 ? 1 : (Word{1} << (length - 1)) | (middle << 1) | 1;
497 const Word syndrome = code.syndrome(pattern << (code.n - start - length));
498 if (!seen.insert(syndrome).second) return {"burst " + name + " detect " + std::to_string(detect) + " correct " + std::to_string(length - 1)};
499 }
500 }
501 }
502 throw std::logic_error("a nonzero codeword must be an undetectable burst");
503 }
504 Lines interleave(const std::string& name, Code& code, const Lines& args) const {
505 const int start = std::stoi(args[1]), length = std::stoi(args[2]), depth = static_cast<int>(args.size()) - 3;
506 for (std::size_t index = 3; index < args.size(); ++index) if (static_cast<int>(args[index].size()) != code.k) fail("bad length");
507 if (start < 1 || length < 1 || start + length - 1 > depth * code.n) fail("bad burst");
508 Matrix sentWords, receivedWords(depth);
509 for (std::size_t index = 3; index < args.size(); ++index) sentWords.push_back(code.encode(parseWord(args[index])));
510 std::string sent, received;
511 for (int column = 0; column < code.n; ++column) for (int row = 0; row < depth; ++row) {
512 const int position = column * depth + row + 1;
513 const bool original = bit(sentWords[row], code.n, column), changed = original != (start <= position && position < start + length);
514 sent += original ? '1' : '0'; received += changed ? '1' : '0';
515 receivedWords[row] = (receivedWords[row] << 1) | changed;
516 }
517 Lines output{"interleave " + name + " depth " + std::to_string(depth) + " burst " + std::to_string(start) + " " + std::to_string(length), "sent " + sent, "recv " + received};
518 int correct = 0;
519 for (int row = 0; row < depth; ++row) {
520 const auto decoded = code.decode(receivedWords[row]);
521 const auto message = bits(decoded.message, code.k);
522 const bool ok = message == args[row + 3];
523 if (ok) ++correct;
524 output.push_back("row " + std::to_string(row + 1) + " errors " + std::to_string(weight(receivedWords[row] ^ sentWords[row])) + " corrected " +
525 std::to_string(weight(decoded.error)) + " message " + message + (ok ? " ok" : " wrong"));
526 }
527 output.push_back("result ok " + std::to_string(correct) + " wrong " + std::to_string(depth - correct));
528 return output;
529 }
530public:
531 Command parse(const std::string& line) const {
532 std::istringstream input(line);
533 std::string name, word;
534 input >> name;
535 const auto found = formats.find(name);
536 if (found == formats.end()) fail("unknown command " + name);
537 const auto& spec = found->second;
538 Lines args;
539 while (input >> word) args.push_back(word);
540 const int extra = static_cast<int>(args.size()) - static_cast<int>(spec.types.size());
541 if (spec.repeat ? extra < 1 || extra > spec.maximum : extra != 0) fail("bad arguments");
542 for (std::size_t index = 0; index < args.size(); ++index) if (!valid(args[index], index < spec.types.size() ? spec.types[index] : *spec.repeat)) fail("bad arguments");
543 for (std::size_t index = 0; index < spec.types.size(); ++index) if (spec.types[index] == Argument::Code && !codes.count(args[index])) fail("no code " + args[index]);
544 return {spec.verb, std::move(args)};
545 }
546 Lines execute(const Command& command) {
547 const Verb verb = command.verb;
548 const auto& args = command.args;
549 if (defining(verb)) return define(verb, args);
550 if (verb == Verb::Crc || verb == Verb::Crccheck || verb == Verb::Factor || verb == Verb::Macwilliams || verb == Verb::Bounds) return independent(verb, args);
551 if (verb == Verb::List) {
552 Lines output{"codes " + std::to_string(codes.size())};
553 for (const auto& [name, code] : codes) output.push_back(name + " n " + std::to_string(code.n) + " k " + std::to_string(code.k) + " encoder " + code.encoder());
554 return output;
555 }
556 const auto& name = args[0];
557 if (verb == Verb::Delete) { codes.erase(name); return {"deleted " + name}; }
558 Code& code = codes.at(name);
559 const int n = code.n, k = code.k;
560 switch (verb) {
561 case Verb::Show: {
562 Lines output{"code " + name + " n " + std::to_string(n) + " k " + std::to_string(k) + " encoder " + code.encoder()};
563 if (code.polynomial) output.push_back("poly g " + bits(*code.polynomial, n - k + 1) + " h " + bits(divide((Word{1} << n) | 1, *code.polynomial).quotient, k + 1));
564 for (Word row : code.G) output.push_back("G " + bits(row, n));
565 for (Word row : code.H) output.push_back("H " + bits(row, n));
566 return output;
567 }
568 case Verb::Info: {
569 int distance = n;
570 for (Word word : code.words) if (word) distance = std::min(distance, weight(word));
571 const int t = (distance - 1) / 2, divisor = std::gcd(n, k);
572 const Wide volume = std::accumulate(choose[n].begin(), choose[n].begin() + t + 1, Wide{0});
573 const bool cyclic = std::all_of(code.G.begin(), code.G.end(), [&](Word row) {
574 return code.messages.count(((row << 1) & ((Word{1} << n) - 1)) | (row >> (n - 1))) != 0;
575 });
576 return {"info " + name + " n " + std::to_string(n) + " k " + std::to_string(k) + " rate " + std::to_string(k / divisor) + "/" + std::to_string(n / divisor) +
577 " d " + std::to_string(distance) + " t " + std::to_string(t) + " perfect " + yes(volume == (Wide{1} << (n - k))) +
578 " mds " + yes(distance == n - k + 1) + " cyclic " + (cyclic ? bits(code.words[1], degree(code.words[1]) + 1) : "no")};
579 }
580 case Verb::Systematic: return systematic(name, code);
581 case Verb::Weights: {
582 const auto a = distribution(code.words, n), b = distribution(span(code.H), n);
583 bool orthogonal = true;
584 for (Word x : code.G) for (Word y : code.G) if (weight(x & y) % 2) orthogonal = false;
585 return {"weights " + name + " " + weightList(a), "dual " + name + " " + weightList(b), "duality " + name +
586 " self-orthogonal " + yes(orthogonal) + " self-dual " + yes(orthogonal && n == 2 * k) + " formally-self-dual " + yes(a == b)};
587 }
588 case Verb::Covering: {
589 const auto counts = distribution(code.leaders(), n);
590 int radius = n;
591 while (counts[radius] == 0) --radius;
592 Lines items;
593 for (int value = 0; value <= radius; ++value) items.push_back(std::to_string(value) + ":" + std::to_string(counts[value]));
594 return {"covering " + name + " radius " + std::to_string(radius) + " leaders " + join(items)};
595 }
596 case Verb::Burst: return burst(name, code);
597 case Verb::Trellis: {
598 Lines profile;
599 int maximum = 0;
600 for (int cut = 0; cut <= n; ++cut) {
601 const int right = n - cut;
602 Matrix prefix, suffix;
603 for (Word row : code.G) { prefix.push_back(row >> right); suffix.push_back(row & ((Word{1} << right) - 1)); }
604 const int value = static_cast<int>(basis(prefix, cut).size() + basis(suffix, right).size()) - k;
605 profile.push_back(std::to_string(value)); maximum = std::max(maximum, value);
606 }
607 return {"trellis " + name + " profile " + join(profile) + " max " + std::to_string(maximum)};
608 }
609 case Verb::Stdarray: {
610 if (n - k > 6 || k > 4) fail("too large");
611 const auto& leaders = code.leaders();
612 Matrix syndromes(leaders.size());
613 std::iota(syndromes.begin(), syndromes.end(), Word{0});
614 std::sort(syndromes.begin(), syndromes.end(), [&](Word a, Word b) {
615 return weight(leaders[a]) != weight(leaders[b]) ? weight(leaders[a]) < weight(leaders[b]) : leaders[a] < leaders[b];
616 });
617 Lines output{"stdarray " + name + " rows " + std::to_string(leaders.size()) + " columns " + std::to_string(code.words.size())};
618 for (Word s : syndromes) {
619 std::string row = bits(s, n - k) + ":";
620 for (Word word : code.words) row += " " + bits(word ^ leaders[s], n);
621 output.push_back(std::move(row));
622 }
623 return output;
624 }
625 case Verb::Equal: {
626 const auto& other = codes.at(args[1]);
627 if (n != other.n) return {"equal " + name + " " + args[1] + " no lengths " + std::to_string(n) + " " + std::to_string(other.n)};
628 Matrix combined = code.G;
629 combined.insert(combined.end(), other.G.begin(), other.G.end());
630 const int common = k + other.k - static_cast<int>(basis(combined, n).size());
631 return {"equal " + name + " " + args[1] + " " + yes(k == common && other.k == common) + " intersection " + std::to_string(common)};
632 }
633 case Verb::Encode:
634 if (static_cast<int>(args[1].size()) != k) fail("bad length");
635 return {"encode " + name + " message " + args[1] + " codeword " + bits(code.encode(parseWord(args[1])), n)};
636 case Verb::Syndrome: case Verb::Decode: case Verb::Bounded: case Verb::Coset: {
637 const auto& text = args[verb == Verb::Bounded ? 2 : 1];
638 if (static_cast<int>(text.size()) != n) fail("bad length");
639 const Word word = parseWord(text), syndrome = code.syndrome(word);
640 if (verb == Verb::Syndrome) return {"syndrome " + name + " " + bits(syndrome, n - k) + " codeword " + yes(syndrome == 0)};
641 const auto decoded = code.decode(word);
642 if (verb == Verb::Coset) {
643 Matrix shifted;
644 for (Word value : code.words) shifted.push_back(value ^ word);
645 return {"coset " + name + " syndrome " + bits(syndrome, n - k) + " leader " + bits(decoded.error, n) + " weights " + weightList(distribution(shifted, n))};
646 }
647 if (verb == Verb::Bounded && weight(decoded.error) > std::stoi(args[1])) return {"bounded " + name + " fail weight " + std::to_string(weight(decoded.error))};
648 return {std::string(verb == Verb::Bounded ? "bounded " : "decode ") + name + " " + code.decodedText(decoded)};
649 }
650 case Verb::Erasure: {
651 const auto& text = args[1];
652 if (static_cast<int>(text.size()) != n) fail("bad length");
653 Word mask = 0, known = 0;
654 int erased = 0;
655 for (char character : text) {
656 mask = (mask << 1) | (character != '?'); known = (known << 1) | (character == '1');
657 if (character == '?') ++erased;
658 }
659 Matrix candidates;
660 for (Word word : code.words) if ((word & mask) == known) candidates.push_back(word);
661 std::string output = "erasure " + name + " erased " + std::to_string(erased) + " candidates " + std::to_string(candidates.size());
662 if (candidates.size() == 1) output += " codeword " + bits(candidates[0], n) + " message " + bits(code.messages.at(candidates[0]), k);
663 return {output};
664 }
665 case Verb::Interleave: return interleave(name, code, args);
666 default: throw std::logic_error("unhandled command");
667 }
668 }
669};
