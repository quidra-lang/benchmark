23({int quotient, int remainder}) divide(int dividend, int divisor) {
24 var quotient = 0;
25 while (dividend != 0 && dividend.bitLength >= divisor.bitLength) {
26 final shift = dividend.bitLength - divisor.bitLength;
27 quotient ^= 1 << shift;
28 dividend ^= divisor << shift;
29 }
30 return (quotient: quotient, remainder: dividend);
31}

33List<int> basis(Iterable<int> rows, int length) {
34 final pivots = List<int>.filled(length, 0), kept = <int>[];
35 for (final original in rows) {
36 var row = original;
37 for (var column = 0; column < length; column++) {
38 if (!bit(row, length, column)) continue;
39 if (pivots[column] != 0) {
40 row ^= pivots[column];
41 } else {
42 pivots[column] = row;
43 kept.add(original);
44 break;
45 }
46 }
47 }
48 return kept;
49}

51({List<int> rows, List<int> columns}) rref(Iterable<int> original, int length) {
52 final rows = original.toList(), columns = <int>[];
53 for (var column = 0; column < length; column++) {
54 final pivot = columns.length;
55 var found = pivot;
56 while (found < rows.length && !bit(rows[found], length, column)) {
57 found++;
58 }
59 if (found == rows.length) continue;
60 final old = rows[pivot];
61 rows[pivot] = rows[found];
62 rows[found] = old;
63 for (var row = 0; row < rows.length; row++) {
64 if (row != pivot && bit(rows[row], length, column)) {
65 rows[row] ^= rows[pivot];
66 }
67 }
68 columns.add(column);
69 }
70 return (rows: rows.take(columns.length).toList(), columns: columns);
71}

73List<int> nullspace(Iterable<int> original, int length) {
74 final reduced = rref(original, length), result = <int>[];
75 for (var column = 0; column < length; column++) {
76 if (reduced.columns.contains(column)) continue;
77 var word = 1 << (length - column - 1);
78 for (var row = 0; row < reduced.rows.length; row++) {
79 if (bit(reduced.rows[row], length, column)) {
80 word |= 1 << (length - reduced.columns[row] - 1);
81 }
82 }
83 result.add(word);
84 }
85 return result;
86}

123List<int> reedMuller(int order, int variables) {
124 final length = 1 << variables;
125 if (order == 0) return [(1 << length) - 1];
126 final firstOrder = order < variables ? order : variables - 1;
127 return [
128 for (final row in reedMuller(firstOrder, variables - 1))
129 (row << (length ~/ 2)) | row,
130 ...reedMuller(order - 1, variables - 1),
131 ];

136final class Code {
137 final int n;
138 final List<int> g, h;
139 final int? polynomial;
140 late final List<int> words;
141 late final Map<int, int> _messages;
142 List<int>? _leaders;
143 int get k => g.length;
144 String get encoder => polynomial == null ? 'matrix' : 'polynomial';
145
146 factory Code(int n, List<int> g, {List<int>? h, int? polynomial}) {
147 checkDimension(n, g.length);
148 return Code._(n, List.unmodifiable(g),
149 List.unmodifiable(h ?? nullspace(g, n)), polynomial);
150 }
151
152 Code._(this.n, this.g, this.h, this.polynomial) {
153 _messages = {
154 for (var message = 0; message < (1 << k); message++) encode(message): message,
155 };
156 words = List.unmodifiable(_messages.keys.toList()..sort());
157 }
158
159 int encode(int message) {
160 final p = polynomial;
161 if (p != null) {
162 final shifted = message << (n - k);
163 return shifted | divide(shifted, p).remainder;
164 }
165 var word = 0;
166 for (var row = 0; row < k; row++) {
167 if (bit(message, k, row)) word ^= g[row];
168 }
169 return word;
170 }
171
172 int syndrome(int word) =>
173 h.fold(0, (result, row) => (result << 1) | (weight(row & word) % 2));
174 int message(int word) => _messages[word]!;
175 bool contains(int word) => _messages.containsKey(word);
176
177 List<int> get leaders {
178 if (_leaders != null) return _leaders!;
179 final result = List<int>.filled(1 << (n - k), -1);
180 result[0] = 0;
181 var remaining = result.length - 1;
182 for (var count = 1; remaining > 0; count++) {
183 var word = (1 << count) - 1;
184 while (word < (1 << n)) {
185 final s = syndrome(word);
186 if (result[s] == -1) {
187 result[s] = word;
188 if (--remaining == 0) break;
189 }
190 final lowest = word & -word, next = word + lowest;
191 word = next | (((next ^ word) >> 2) ~/ lowest);
192 }
193 }
194 return _leaders = List.unmodifiable(result);
195 }
196
197 Decoded decode(int received) {
198 final error = leaders[syndrome(received)], word = received ^ error;
199 return (error: error, word: word, message: message(word));
200 }
201
202 String decodedText(Decoded value) =>
203 'codeword ${bits(value.word, n)} message ${bits(value.message, k)} '
204 'error ${bits(value.error, n)} weight ${weight(value.error)}';
205}

267bool valid(String text, Argument kind) {
268 final bytes = text.codeUnits;
269 bool letter(int c) => c >= 65 && c <= 90 || c >= 97 && c <= 122;
270 bool digit(int c) => c >= 48 && c <= 57;
271 if (bytes.isEmpty) return false;
272 return switch (kind) {
273 Argument.name || Argument.code => bytes.length <= 8 &&
274 letter(bytes.first) && bytes.every((c) => letter(c) || digit(c)),
275 Argument.number => bytes.length <= 4 &&
276 (bytes.length == 1 || bytes.first != 48) && bytes.every(digit),
277 Argument.bits || Argument.erasure => bytes.length <= 30 &&
278 bytes.every((c) => c == 48 || c == 49 || kind == Argument.erasure && c == 63),
279 };
280}

296 Command parse(String line) {
297 final words = line.split(' '), verb = verbs[words[0]];
298 if (verb == null) fail('unknown command ${words[0]}');
299 final spec = formats[verb]!, args = words.sublist(1);
300 final extra = args.length - spec.types.length;
301 if (spec.repeated == null ? extra != 0 : extra < 1 || extra > spec.maximum) {
302 fail('bad arguments');
303 }
304 for (var i = 0; i < args.length; i++) {
305 final kind = i < spec.types.length ? spec.types[i] : spec.repeated!;
306 if (!valid(args[i], kind)) fail('bad arguments');
307 }
308 for (var i = 0; i < spec.types.length; i++) {
309 if (spec.types[i] == Argument.code && !_codes.containsKey(args[i])) {
310 fail('no code ${args[i]}');
311 }
312 }
313 return Command(verb, args);
314 }

316 List<String> define(Verb verb, List<String> args) {
317 final name = args[0], source = _codes[args[1]];
318 final other = args.length > 2 ? _codes[args[2]] : null;
319 var n = 0, g = <int>[];
320 List<int>? h;
321 int? polynomial;
322 switch (verb) {
323 case Verb.gen || Verb.pcheck:
324 n = args[1].length;
325 if (args.skip(1).any((word) => word.length != n)) fail('bad length');
326 final rows = basis(args.skip(1).map((word) => int.parse(word, radix: 2)), n);
327 if (verb == Verb.gen) {
328 g = rows;
329 } else {
330 h = rows;
331 g = nullspace(rows, n);
332 }
333 case Verb.hamming || Verb.simplex:
334 final r = int.parse(args[1]);
335 if (r < 2 || r > 4) fail('bad parameter');
336 n = (1 << r) - 1;
337 final rows = List<int>.filled(r, 0);
338 for (var column = 1; column <= n; column++) {
339 for (var row = 0; row < r; row++) {
340 rows[row] = (rows[row] << 1) | (bit(column, r, row) ? 1 : 0);
341 }
342 }
343 if (verb == Verb.hamming) {
344 h = rows;
345 g = nullspace(rows, n);
346 } else {
347 g = rows;
348 }
349 case Verb.repetition || Verb.parity:
350 n = int.parse(args[1]);
351 if (n < 2 || n > 24) fail('bad parameter');
352 g = verb == Verb.repetition ? [(1 << n) - 1] : [
353 for (var row = 0; row < n - 1; row++) (1 << (n - row - 1)) | 1,
354 ];
355 case Verb.rm:
356 final r = int.parse(args[1]), m = int.parse(args[2]);
357 if (m < 1 || m > 4 || r > m) fail('bad parameter');
358 n = 1 << m;
359 g = reedMuller(r, m);
360 case Verb.cyclic:
361 n = int.parse(args[1]);
362 final text = args[2], d = text.length - 1;
363 if (n < 2 || n > 24 || text.startsWith('0') ||
364 text.length < 2 || text.length > n) fail('bad parameter');
365 final p = int.parse(text, radix: 2);
366 if (divide((1 << n) | 1, p).remainder != 0) fail('not a divisor');
367 final k = n - d;
368 checkDimension(n, k);
369 polynomial = p;
370 g = [for (var row = 0; row < k; row++) p << (k - row - 1)];
371 h = List<int>.filled(d, 0);
372 for (var column = 0; column < n; column++) {
373 final remainder = divide(1 << (n - column - 1), p).remainder;
374 for (var row = 0; row < d; row++) {
375 h[row] = (h[row] << 1) | (bit(remainder, d, row) ? 1 : 0);
376 }
377 }
378 case Verb.dual:
379 n = source!.n;
380 g = source.h;
381 h = source.g;
382 case Verb.extend:
383 n = source!.n + 1;
384 g = [for (final row in source.g) (row << 1) | (weight(row) % 2)];
385 case Verb.puncture || Verb.shorten:
386 final a = source!, column = int.parse(args[2]) - 1;
387 if (column < 0 || column >= a.n) fail('bad position');
388 var rows = a.g.toList();
389 if (verb == Verb.shorten) {
390 final pivot = rows.indexWhere((row) => bit(row, a.n, column));
391 if (pivot != -1) {
392 final z = rows.removeAt(pivot);
393 rows = [for (final row in rows) bit(row, a.n, column) ? row ^ z : row];
394 }
395 }
396 n = a.n - 1;
397 g = [for (final row in rows) removeBit(row, a.n, column)];
398 if (verb == Verb.puncture) g = basis(g, n);
399 case Verb.augment:
400 n = source!.n;
401 g = basis([...source.g, (1 << n) - 1], n);
402 case Verb.expurgate:
403 n = source!.n;
404 g = source.g.toList();
405 final pivot = g.indexWhere((row) => weight(row).isOdd);
406 if (pivot != -1) {
407 final z = g.removeAt(pivot);
408 g = [for (final row in g) weight(row).isOdd ? row ^ z : row];
409 }
410 case Verb.permute:
411 final a = source!, order = args.skip(2).map(int.parse).toList();
412 n = a.n;
413 if (order.length != n || order.toSet().length != n ||
414 order.any((column) => column < 1 || column > n)) fail('bad permutation');
415 g = [
416 for (final row in a.g)
417 order.fold(0, (result, column) =>
418 (result << 1) | (bit(row, n, column - 1) ? 1 : 0)),
419 ];
420 case Verb.directsum || Verb.uuv || Verb.product:
421 final a = source!, b = other!;
422 if (verb == Verb.uuv && a.n != b.n) fail('bad length');
423 n = verb == Verb.product ? a.n * b.n : a.n + b.n;
424 checkDimension(n, verb == Verb.product ? a.k * b.k : a.k + b.k);
425 if (verb == Verb.product) {
426 for (final x in a.g) {
427 for (final y in b.g) {
428 var row = 0;
429 for (var i = 0; i < a.n; i++) {
430 row = (row << b.n) | (bit(x, a.n, i) ? y : 0);
431 }
432 g.add(row);
433 }
434 }
435 } else {
436 g = [
437 for (final row in a.g) (row << b.n) | (verb == Verb.uuv ? row : 0),
438 ...b.g,
439 ];
440 }
441 default:
442 throw StateError('Not a defining command');
443 }
444 final code = Code(n, g, h: h, polynomial: polynomial);
445 final prefix = _codes.containsKey(name) ? 'redefined' : 'defined';
446 _codes[name] = code;
447 return ['$prefix $name n $n k ${code.k}'];
448 }

450 List<String> systematic(String name, Code code) {
451 final n = code.n, k = code.k, rows = code.g.toList();
452 final order = List.generate(n, (i) => i + 1);
453 for (var pivot = 0; pivot < k; pivot++) {
454 var column = pivot;
455 while (!rows.skip(pivot).any((row) => bit(row, n, column))) {
456 column++;
457 }
458 if (column != pivot) {
459 final mask = (1 << (n - pivot - 1)) | (1 << (n - column - 1));
460 for (var row = 0; row < k; row++) {
461 if (bit(rows[row], n, pivot) != bit(rows[row], n, column)) rows[row] ^= mask;
462 }
463 final old = order[pivot];
464 order[pivot] = order[column];
465 order[column] = old;
466 }
467 var found = pivot;
468 while (!bit(rows[found], n, pivot)) {
469 found++;
470 }
471 final old = rows[pivot];
472 rows[pivot] = rows[found];
473 rows[found] = old;
474 for (var row = 0; row < k; row++) {
475 if (row != pivot && bit(rows[row], n, pivot)) rows[row] ^= rows[pivot];
476 }
477 }
478 final output = ['systematic $name perm ${order.join(' ')}',
479 for (final row in rows) 'G ${bits(row, n)}'];
480 for (var column = k; column < n; column++) {
481 var row = rows.fold(0, (result, word) =>
482 (result << 1) | (bit(word, n, column) ? 1 : 0));
483 row = (row << (n - k)) | (1 << (n - column - 1));
484 output.add('H ${bits(row, n)}');
485 }
486 return output;
487 }

489 List<String> burst(String name, Code code) {
490 var detect = code.n;
491 for (final word in code.words.skip(1)) {
492 final span = word.bitLength - (word & -word).bitLength;
493 if (span < detect) detect = span;
494 }
495 final seen = <int>{0};
496 for (var length = 1; length <= code.n; length++) {
497 for (var start = 0; start <= code.n - length; start++) {
498 final limit = length == 1 ? 1 : 1 << (length - 2);
499 for (var middle = 0; middle < limit; middle++) {
500 final pattern = length == 1 ? 1 : (1 << (length - 1)) | (middle << 1) | 1;
501 final s = code.syndrome(pattern << (code.n - start - length));
502 if (!seen.add(s)) return ['burst $name detect $detect correct ${length - 1}'];
503 }
504 }
505 }
506 throw StateError('A nonzero codeword is an undetectable burst');
507 }

509 List<String> interleave(String name, Code code, List<String> args) {
510 final messages = args.sublist(3), start = int.parse(args[1]);
511 final length = int.parse(args[2]), depth = messages.length;
512 if (messages.any((word) => word.length != code.k)) fail('bad length');
513 if (start < 1 || length < 1 || start + length - 1 > depth * code.n) {
514 fail('bad burst');
515 }
516 final sentWords = [for (final word in messages) code.encode(int.parse(word, radix: 2))];
517 final receivedWords = List<int>.filled(depth, 0);
518 final sent = StringBuffer(), received = StringBuffer();
519 for (var column = 0; column < code.n; column++) {
520 for (var row = 0; row < depth; row++) {
521 final position = column * depth + row + 1;
522 final original = bit(sentWords[row], code.n, column);
523 final changed = original != (start <= position && position < start + length);
524 sent.write(original ? '1' : '0');
525 received.write(changed ? '1' : '0');
526 receivedWords[row] = (receivedWords[row] << 1) | (changed ? 1 : 0);
527 }
528 }
529 final output = ['interleave $name depth $depth burst $start $length',
530 'sent $sent', 'recv $received'];
531 var correct = 0;
532 for (var row = 0; row < depth; row++) {
533 final decoded = code.decode(receivedWords[row]);
534 final message = bits(decoded.message, code.k), ok = message == messages[row];
535 if (ok) correct++;
536 output.add('row ${row + 1} errors ${weight(receivedWords[row] ^ sentWords[row])} '
537 'corrected ${weight(decoded.error)} message $message ${ok ? 'ok' : 'wrong'}');
538 }
539 output.add('result ok $correct wrong ${depth - correct}');
540 return output;
541 }

543 List<String> execute(Command command) {
544 final verb = command.verb, args = command.args;
545 if (verb.defining) return define(verb, args);
546 if ([Verb.crc, Verb.crccheck, Verb.factor, Verb.macwilliams, Verb.bounds].contains(verb)) {
547 return independent(verb, args);
548 }
549 if (verb == Verb.list) {
550 return ['codes ${_codes.length}', for (final entry in _codes.entries)
551 '${entry.key} n ${entry.value.n} k ${entry.value.k} encoder ${entry.value.encoder}'];
552 }
553 final name = args[0];
554 if (verb == Verb.delete) {
555 _codes.remove(name);
556 return ['deleted $name'];
557 }
558 final code = _codes[name]!, n = code.n, k = code.k;
559 switch (verb) {
560 case Verb.show:
561 final p = code.polynomial;
562 return ['code $name n $n k $k encoder ${code.encoder}',
563 if (p != null) 'poly g ${bits(p, n - k + 1)} h ${bits(divide((1 << n) | 1, p).quotient, k + 1)}',
564 for (final row in code.g) 'G ${bits(row, n)}',
565 for (final row in code.h) 'H ${bits(row, n)}'];
566 case Verb.info:
567 final distance = code.words.skip(1).map(weight).reduce((a, b) => a < b ? a : b);
568 final t = (distance - 1) ~/ 2, divisor = n.gcd(k);
569 final cyclic = code.g.every((row) =>
570 code.contains(((row << 1) & ((1 << n) - 1)) | (row >> (n - 1))));
571 return ['info $name n $n k $k rate ${k ~/ divisor}/${n ~/ divisor} '
572 'd $distance t $t perfect ${yes(volume(n, t) == (1 << (n - k)))} '
573 'mds ${yes(distance == n - k + 1)} cyclic ${cyclic ? code.words[1].toRadixString(2) : 'no'}'];
574 case Verb.systematic:
575 return systematic(name, code);
576 case Verb.weights:
577 final a = distribution(code.words, n), b = distribution(span(code.h), n);
578 final orthogonal = code.g.every((x) => code.g.every((y) => weight(x & y).isEven));
579 final formal = List.generate(n + 1, (i) => i).every((i) => a[i] == b[i]);
580 return ['weights $name ${weightList(a)}', 'dual $name ${weightList(b)}',
581 'duality $name self-orthogonal ${yes(orthogonal)} self-dual ${yes(orthogonal && n == 2 * k)} '
582 'formally-self-dual ${yes(formal)}'];
583 case Verb.covering:
584 final counts = distribution(code.leaders, n);
585 final radius = counts.lastIndexWhere((count) => count != 0);
586 return ['covering $name radius $radius leaders ${[
587 for (var i = 0; i <= radius; i++) '$i:${counts[i]}',
588 ].join(' ')}'];
589 case Verb.burst:
590 return burst(name, code);
591 case Verb.trellis:
592 final profile = <int>[];
593 for (var cut = 0; cut <= n; cut++) {
594 final right = n - cut;
595 profile.add(basis(code.g.map((row) => row >> right), cut).length +
596 basis(code.g.map((row) => row & ((1 << right) - 1)), right).length - k);
597 }
598 return ['trellis $name profile ${profile.join(' ')} max ${profile.reduce((a, b) => a > b ? a : b)}'];
599 case Verb.stdarray:
600 if (n - k > 6 || k > 4) fail('too large');
601 final leaders = code.leaders;
602 final syndromes = List.generate(leaders.length, (i) => i)..sort((a, b) {
603 final byWeight = weight(leaders[a]).compareTo(weight(leaders[b]));
604 return byWeight != 0 ? byWeight : leaders[a].compareTo(leaders[b]);
605 });
606 return ['stdarray $name rows ${leaders.length} columns ${code.words.length}',
607 for (final s in syndromes)
608 '${bits(s, n - k)}: ${code.words.map((word) => bits(word ^ leaders[s], n)).join(' ')}'];
609 case Verb.equal:
610 final other = _codes[args[1]]!;
611 if (n != other.n) return ['equal $name ${args[1]} no lengths $n ${other.n}'];
612 final common = k + other.k - basis([...code.g, ...other.g], n).length;
613 return ['equal $name ${args[1]} ${yes(k == common && other.k == common)} intersection $common'];
614 case Verb.encode:
615 if (args[1].length != k) fail('bad length');
616 return ['encode $name message ${args[1]} codeword ${bits(code.encode(int.parse(args[1], radix: 2)), n)}'];
617 case Verb.syndrome || Verb.decode || Verb.bounded || Verb.coset:
618 final text = args[verb == Verb.bounded ? 2 : 1];
619 if (text.length != n) fail('bad length');
620 final word = int.parse(text, radix: 2), s = code.syndrome(word);
621 if (verb == Verb.syndrome) {
622 return ['syndrome $name ${bits(s, n - k)} codeword ${yes(s == 0)}'];
623 }
624 final decoded = code.decode(word);
625 if (verb == Verb.coset) {
626 final counts = distribution(code.words.map((c) => c ^ word), n);
627 return ['coset $name syndrome ${bits(s, n - k)} leader ${bits(decoded.error, n)} weights ${weightList(counts)}'];
628 }
629 if (verb == Verb.bounded && weight(decoded.error) > int.parse(args[1])) {
630 return ['bounded $name fail weight ${weight(decoded.error)}'];
631 }
632 return ['${verb == Verb.bounded ? 'bounded' : 'decode'} $name ${code.decodedText(decoded)}'];
633 case Verb.erasure:
634 final text = args[1];
635 if (text.length != n) fail('bad length');
636 var mask = 0, known = 0, erased = 0;
637 for (final character in text.codeUnits) {
638 mask = (mask << 1) | (character != 63 ? 1 : 0);
639 known = (known << 1) | (character == 49 ? 1 : 0);
640 if (character == 63) erased++;
641 }
642 final candidates = code.words.where((word) => (word & mask) == known).toList();
643 var output = 'erasure $name erased $erased candidates ${candidates.length}';
644 if (candidates.length == 1) {
645 output += ' codeword ${bits(candidates.single, n)} message ${bits(code.message(candidates.single), k)}';
646 }
647 return [output];
648 case Verb.interleave:
649 return interleave(name, code, args);
650 default:
651 throw StateError('Unhandled command');
652 }
653 }
654}

656List<String> independent(Verb verb, List<String> args) {
657 if (verb == Verb.crc || verb == Verb.crccheck) {
658 final generator = args[0], text = args[1], d = generator.length - 1;
659 if (generator.startsWith('0') || generator.length < 2 || generator.length > 17) {
660 fail('bad parameter');
661 }
662 if (verb == Verb.crc && text.length + d > 30) fail('bad length');
663 final word = int.parse(text, radix: 2), divisor = int.parse(generator, radix: 2);
664 final remainder = divide(verb == Verb.crc ? word << d : word, divisor).remainder;
665 final suffix = bits(remainder, d);
666 if (verb == Verb.crc) return ['crc remainder $suffix frame $text$suffix'];
667 return ['crccheck ${remainder == 0 ? 'ok' : 'error'} remainder $suffix'];
668 }
669 final n = int.parse(args[0]);
670 if (verb == Verb.bounds) {
671 final d = int.parse(args[1]);
672 if (d < 1 || d > n || n > 24) fail('bad parameter');
673 final v = volume(n, (d - 1) ~/ 2);
674 var hamming = 0, griesmer = 0, sum = 0;
675 for (var k = 0; k <= n; k++) {
676 if ((1 << k) * v <= (1 << n)) hamming = k;
677 }
678 for (var k = 1; k <= n; k++) {
679 final denominator = 1 << (k - 1);
680 sum += (d + denominator - 1) ~/ denominator;
681 if (sum <= n) griesmer = k;
682 }
683 return ['bounds n $n d $d singleton ${n - d + 1} hamming $hamming griesmer $griesmer'];
684 }
685 if (n < 1 || n > 24) fail('bad parameter');
686 if (verb == Verb.factor) {
687 var remaining = (1 << n) | 1, candidate = 2, divisors = 1;
688 final factors = <String>[];
689 while (2 * (candidate.bitLength - 1) <= remaining.bitLength - 1) {
690 var exponent = 0;
691 while (divide(remaining, candidate).remainder == 0) {
692 remaining = divide(remaining, candidate).quotient;
693 exponent++;
694 }
695 if (exponent > 0) {
696 factors.add('${candidate.toRadixString(2)}^$exponent');
697 divisors *= exponent + 1;
698 }
699 candidate++;
700 }
701 if (remaining > 1) {
702 factors.add('${remaining.toRadixString(2)}^1');
703 divisors *= 2;
704 }
705 return ['factor $n ${factors.join(' ')} divisors $divisors'];
706 }
707 final counts = args.skip(1).map(int.parse).toList();
708 if (counts.length != n + 1) fail('bad length');
709 if (counts[0] != 1) fail('bad parameter');
710 final size = counts.fold(0, (a, b) => a + b);
711 var feasible = (size & (size - 1)) == 0;
712 final transformed = <String>[];
713 for (var j = 0; j <= n; j++) {
714 var numerator = 0;
715 for (var x = 0; x <= n; x++) {
716 var krawtchouk = 0;
717 for (var y = 0; y <= j; y++) {
718 krawtchouk += (y.isEven ? 1 : -1) * binomial(x, y) * binomial(n - x, j - y);
719 }
720 numerator += counts[x] * krawtchouk;
721 }
722 final divisor = numerator.gcd(size), p = numerator ~/ divisor, q = size ~/ divisor;
723 transformed.add(q == 1 ? '$p' : '$p/$q');
724 if (q != 1 || p < 0) feasible = false;
725 }
726 return ['macwilliams n $n size $size', 'dual ${transformed.join(' ')}', 'feasible ${yes(feasible)}'];
727}
