46 static boolean letter(char c) { return c >= 'A' && c <= 'Z' || c >= 'a' && c <= 'z'; }

49 static boolean validWord(String text, char kind) {
50 if (text.isEmpty()) return false;
51 if (kind == 'N' || kind == 'A' || kind == 'B') {
52 if (text.length() > 8 || !letter(text.charAt(0))) return false;
53 for (char c : text.toCharArray()) if (!letter(c) && !digit(c)) return false;
54 } else if (kind == 'I') {
55 if (text.length() > 4 || text.length() > 1 && text.charAt(0) == '0') return false;
56 for (char c : text.toCharArray()) if (!digit(c)) return false;
57 } else {
58 if (text.length() > 30) return false;
59 for (char c : text.toCharArray()) if (c != '0' && c != '1' && !(kind == 'X' && c == '?')) return false;
60 }
61 return true;
62 }

79 static Division divide(int value, int divisor) {
80 int quotient = 0;
81 while (value != 0 && length(value) >= length(divisor)) {
82 int shift = length(value) - length(divisor);
83 quotient ^= 1 << shift;
84 value ^= divisor << shift;
85 }
86 return new Division(quotient, value);
87 }

89 static List<Integer> basis(List<Integer> rows, int n) {
90 int[] pivots = new int[32];
91 List<Integer> result = new ArrayList<>();
92 for (int original : rows) {
93 int row = original;
94 for (int column = n - 1; column >= 0; --column) {
95 if ((row & (1 << column)) == 0) continue;
96 if (pivots[column] != 0) row ^= pivots[column];
97 else { pivots[column] = row; result.add(original); break; }
98 }
99 }
100 return result;
101 }

103 static Echelon rref(List<Integer> source, int n) {
104 List<Integer> rows = new ArrayList<>(source), columns = new ArrayList<>();
105 for (int column = 0; column < n; ++column) {
106 int rank = columns.size(), found = rank;
107 while (found < rows.size() && (rows.get(found) & bit(n, column)) == 0) ++found;
108 if (found == rows.size()) continue;
109 int temporary = rows.get(rank);
110 rows.set(rank, rows.get(found)); rows.set(found, temporary);
111 for (int i = 0; i < rows.size(); ++i)
112 if (i != rank && (rows.get(i) & bit(n, column)) != 0) rows.set(i, rows.get(i) ^ rows.get(rank));
113 columns.add(column);
114 }
115 return new Echelon(List.copyOf(rows.subList(0, columns.size())), List.copyOf(columns));
116 }

118 static List<Integer> nullSpace(List<Integer> source, int n) {
119 Echelon reduced = rref(source, n);
120 List<Integer> result = new ArrayList<>();
121 for (int column = 0; column < n; ++column) {
122 if (reduced.columns.contains(column)) continue;
123 int row = bit(n, column);
124 for (int i = 0; i < reduced.rows.size(); ++i)
125 if ((reduced.rows.get(i) & bit(n, column)) != 0) row |= bit(n, reduced.columns.get(i));
126 result.add(row);
127 }
128 return result;
129 }

140 static List<Integer> reedMuller(int r, int m) {
141 if (r == 0) return List.of(mask(1 << m));
142 int half = 1 << (m - 1);
143 List<Integer> result = new ArrayList<>();
144 for (int row : reedMuller(Math.min(r, m - 1), m - 1)) result.add((row << half) | row);
145 result.addAll(reedMuller(r - 1, m - 1));
146 return result;
147 }

153 record Code(int n, List<Integer> g, List<Integer> h, Integer polynomial) {
154 Code {
155 checkDimension(n, g.size());
156 g = List.copyOf(g);
157 h = List.copyOf(h == null ? nullSpace(g, n) : h);
158 }
159 int k() { return g.size(); }
160 String encoder() { return polynomial == null ? "matrix" : "polynomial"; }
161 int encode(int message) {
162 if (polynomial != null) {
163 int shifted = message << h.size();
164 return shifted | divide(shifted, polynomial).remainder;
165 }
166 int word = 0;
167 for (int i = 0; i < k(); ++i) if ((message & bit(k(), i)) != 0) word ^= g.get(i);
168 return word;
169 }
170 int syndrome(int word) {
171 int result = 0;
172 for (int row : h) result = (result << 1) | (weight(row & word) & 1);
173 return result;
174 }
175 int message(int word) {
176 for (int message = 0; message < (1 << k()); ++message) if (encode(message) == word) return message;
177 throw new IllegalStateException("Word outside code");
178 }
179 int[] leaders() {
180 int[] result = new int[1 << h.size()];
181 Arrays.fill(result, -1); result[0] = 0;
182 int found = 1;
183 for (int w = 1; w <= n; ++w) {
184 for (int word = mask(w); word < (1 << n);) {
185 int syndrome = syndrome(word);
186 if (result[syndrome] == -1) {
187 result[syndrome] = word;
188 if (++found == result.length) return result;
189 }
190 int lowest = word & -word, next = word + lowest;
191 word = next | (((next ^ word) >> 2) / lowest);
192 }
193 }
194 return result;
195 }
196 Decoded decode(int received, int[] leaders) {
197 int error = leaders[syndrome(received)], word = received ^ error;
198 return new Decoded(word, message(word), error);
199 }
200 String decodedText(Decoded decoded) {
201 return format("codeword %s message %s error %s weight %d", bits(decoded.word, n),
202 bits(decoded.message, k()), bits(decoded.error, n), weight(decoded.error));
203 }
204 int burstCorrection() {
205 boolean[] seen = new boolean[1 << h.size()]; seen[0] = true;
206 for (int size = 1; size <= n; ++size) {
207 int variants = size == 1 ? 1 : 1 << (size - 2);
208 for (int offset = 0; offset <= n - size; ++offset) for (int middle = 0; middle < variants; ++middle) {
209 int word = size == 1 ? 1 : (1 << (size - 1)) | (middle << 1) | 1;
210 int syndrome = syndrome(word << offset);
211 if (seen[syndrome]) return size - 1;
212 seen[syndrome] = true;
213 }
214 }
215 throw new IllegalStateException("No colliding burst");
216 }
217 }

219 static Code construct(Command command, String[] args, Code a, Code b) {
220 int n;
221 List<Integer> g = new ArrayList<>(), h = null;
222 Integer polynomial = null;
223 switch (command) {
224 case GEN, PCHECK -> {
225 n = args[1].length();
226 for (int i = 1; i < args.length; ++i) {
227 require(args[i].length() == n, "bad length");
228 g.add(parseBits(args[i]));
229 }
230 g = basis(g, n);
231 if (command == Command.PCHECK) { h = g; g = nullSpace(h, n); }
232 }
233 case HAMMING, SIMPLEX -> {
234 int r = number(args[1]); require(r >= 2 && r <= 4, "bad parameter");
235 n = (1 << r) - 1;
236 for (int i = 0; i < r; ++i) {
237 int row = 0;
238 for (int j = 1; j <= n; ++j) row = (row << 1) | ((j >> (r - i - 1)) & 1);
239 g.add(row);
240 }
241 if (command == Command.HAMMING) { h = g; g = nullSpace(h, n); }
242 }
243 case REPETITION, PARITY -> {
244 n = number(args[1]); require(n >= 2 && n <= 24, "bad parameter");
245 if (command == Command.REPETITION) g.add(mask(n));
246 else for (int i = 0; i < n - 1; ++i) g.add(bit(n, i) | 1);
247 }
248 case RM -> {
249 int r = number(args[1]), m = number(args[2]);
250 require(m >= 1 && m <= 4 && r <= m, "bad parameter");
251 n = 1 << m; g = reedMuller(r, m);
252 }
253 case CYCLIC -> {
254 n = number(args[1]); int d = args[2].length() - 1;
255 require(n >= 2 && n <= 24 && args[2].charAt(0) == '1' && d >= 1 && d < n, "bad parameter");
256 int divisor = parseBits(args[2]);
257 require(divide((1 << n) | 1, divisor).remainder == 0, "not a divisor");
258 int k = n - d;
259 for (int i = 0; i < k; ++i) g.add(divisor << (k - i - 1));
260 h = new ArrayList<>();
261 for (int i = 0; i < d; ++i) h.add(0);
262 for (int j = 0; j < n; ++j) {
263 int remainder = divide(bit(n, j), divisor).remainder;
264 for (int i = 0; i < d; ++i) h.set(i, (h.get(i) << 1) | ((remainder >> (d - i - 1)) & 1));
265 }
266 polynomial = divisor;
267 }
268 case DUAL -> { n = a.n; g = a.h; h = a.g; }
269 case EXTEND -> {
270 n = a.n + 1;
271 for (int row : a.g) g.add((row << 1) | (weight(row) & 1));
272 }
273 case PUNCTURE, SHORTEN, EXPURGATE -> {
274 boolean expurgating = command == Command.EXPURGATE;
275 int position = expurgating ? 0 : number(args[2]);
276 if (!expurgating) require(position >= 1 && position <= a.n, "bad position");
277 n = a.n - (expurgating ? 0 : 1);
278 int removed = -1;
279 if (command != Command.PUNCTURE) for (int i = 0; i < a.k(); ++i) {
280 boolean selected = expurgating ? (weight(a.g.get(i)) & 1) != 0 : (a.g.get(i) & bit(a.n, position - 1)) != 0;
281 if (selected) { removed = i; break; }
282 }
283 for (int i = 0; i < a.k(); ++i) {
284 if (i == removed) continue;
285 int row = a.g.get(i);
286 if (removed >= 0 && (expurgating ? (weight(row) & 1) != 0 : (row & bit(a.n, position - 1)) != 0)) row ^= a.g.get(removed);
287 if (!expurgating) { int low = a.n - position; row = ((row >> (low + 1)) << low) | (row & mask(low)); }
288 g.add(row);
289 }
290 if (command == Command.PUNCTURE) g = basis(g, n);
291 }
292 case AUGMENT -> {
293 n = a.n; g.addAll(a.g); g.add(mask(n)); g = basis(g, n);
294 }
295 case PERMUTE -> {
296 n = a.n; require(args.length - 2 == n, "bad permutation");
297 boolean[] used = new boolean[n + 1];
298 for (int i = 2; i < args.length; ++i) {
299 int position = number(args[i]);
300 require(position >= 1 && position <= n && !used[position], "bad permutation");
301 used[position] = true;
302 }
303 for (int original : a.g) {
304 int row = 0;
305 for (int i = 2; i < args.length; ++i) row = (row << 1) | ((original & bit(n, number(args[i]) - 1)) != 0 ? 1 : 0);
306 g.add(row);
307 }
308 }
309 case DIRECTSUM, UUV, PRODUCT -> {
310 if (command == Command.UUV) require(a.n == b.n, "bad length");
311 n = command == Command.PRODUCT ? a.n * b.n : a.n + b.n;
312 checkDimension(n, command == Command.PRODUCT ? a.k() * b.k() : a.k() + b.k());
313 if (command == Command.PRODUCT) {
314 for (int x : a.g) for (int y : b.g) {
315 int row = 0;
316 for (int i = 0; i < a.n; ++i) row = (row << b.n) | ((x & bit(a.n, i)) != 0 ? y : 0);
317 g.add(row);
318 }
319 } else {
320 for (int row : a.g) g.add((row << b.n) | (command == Command.UUV ? row : 0));
321 g.addAll(b.g);
322 }
323 }
324 default -> throw new IllegalStateException("Not a defining command");
325 }
326 return new Code(n, g, h, polynomial);
327 }

347 static final long[][] COMBINATIONS = new long[25][25];
348 static {
349 for (int n = 0; n <= 24; ++n) {
350 COMBINATIONS[n][0] = COMBINATIONS[n][n] = 1;
351 for (int k = 1; k < n; ++k) COMBINATIONS[n][k] = COMBINATIONS[n - 1][k - 1] + COMBINATIONS[n - 1][k];

354 static long choose(int n, int k) { return k < 0 || k > n ? 0 : COMBINATIONS[n][k]; }
355 static long volume(int n, int radius) {
356 long sum = 0;
357 for (int i = 0; i <= radius; ++i) sum += choose(n, i);
358 return sum;
359 }
360 static long krawtchouk(int n, int j, int x) {
361 long sum = 0;
362 for (int y = 0; y <= j; ++y) {
363 long term = choose(x, y) * choose(n - x, j - y);
364 sum += y % 2 == 0 ? term : -term;
365 }
366 return sum;
367 }

373 static String systematic(String name, Code code) {
374 List<Integer> rows = new ArrayList<>(code.g);
375 int[] permutation = new int[code.n];
376 for (int i = 0; i < code.n; ++i) permutation[i] = i + 1;
377 for (int r = 0; r < code.k(); ++r) {
378 int column = r, found;
379 while (true) {
380 found = r;
381 while (found < rows.size() && (rows.get(found) & bit(code.n, column)) == 0) ++found;
382 if (found < rows.size()) break;
383 ++column;
384 }
385 if (column != r) {
386 int left = bit(code.n, r), right = bit(code.n, column);
387 for (int i = 0; i < rows.size(); ++i)
388 if (((rows.get(i) & left) != 0) != ((rows.get(i) & right) != 0)) rows.set(i, rows.get(i) ^ left ^ right);
389 int temporary = permutation[r]; permutation[r] = permutation[column]; permutation[column] = temporary;
390 }
391 int temporary = rows.get(r); rows.set(r, rows.get(found)); rows.set(found, temporary);
392 for (int i = 0; i < rows.size(); ++i)
393 if (i != r && (rows.get(i) & bit(code.n, r)) != 0) rows.set(i, rows.get(i) ^ rows.get(r));
394 }
395 StringBuilder out = new StringBuilder("systematic " + name + " perm");
396 for (int column : permutation) out.append(' ').append(column);
397 out.append('\n'); matrixText(out, "G", rows, code.n);
398 for (int j = 0; j < code.h.size(); ++j) {
399 int row = 0;
400 for (int value : rows) row = (row << 1) | ((value & bit(code.n, code.k() + j)) != 0 ? 1 : 0);
401 line(out, "H %s", bits((row << code.h.size()) | bit(code.h.size(), j), code.n));
402 }
403 return out.toString();
404 }

406 static String independent(Command command, String[] args) {
407 if (command == Command.CRC || command == Command.CRCCHECK) {
408 int d = args[0].length() - 1;
409 require(args[0].charAt(0) == '1' && d >= 1 && d <= 16, "bad parameter");
410 if (command == Command.CRC) require(args[1].length() + d <= 30, "bad length");
411 int value = parseBits(args[1]);
412 if (command == Command.CRC) value <<= d;
413 int remainder = divide(value, parseBits(args[0])).remainder;
414 String suffix = bits(remainder, d);
415 return command == Command.CRC ? line("crc remainder %s frame %s%s", suffix, args[1], suffix)
416 : line("crccheck %s remainder %s", remainder == 0 ? "ok" : "error", suffix);
417 }
418 int n = number(args[0]);
419 if (command == Command.BOUNDS) {
420 int d = number(args[1]); require(d >= 1 && d <= n && n <= 24, "bad parameter");
421 long volume = volume(n, (d - 1) / 2), sum = 0;
422 int hamming = 0, griesmer = 0;
423 for (int k = 0; k <= n; ++k) if ((1L << k) * volume <= (1L << n)) hamming = k;
424 for (int k = 1; k <= n; ++k) {
425 long denominator = 1L << (k - 1);
426 sum += (d + denominator - 1) / denominator;
427 if (sum <= n) griesmer = k;
428 }
429 return line("bounds n %d d %d singleton %d hamming %d griesmer %d", n, d, n - d + 1, hamming, griesmer);
430 }
431 require(n >= 1 && n <= 24, "bad parameter");
432 StringBuilder out = new StringBuilder();
433 if (command == Command.FACTOR) {
434 int remaining = (1 << n) | 1, divisors = 1;
435 out.append("factor ").append(n);
436 for (int factor = 3; 2 * (length(factor) - 1) <= length(remaining) - 1; factor += 2) {
437 int exponent = 0;
438 while (true) {
439 Division result = divide(remaining, factor);
440 if (result.remainder != 0) break;
441 remaining = result.quotient; ++exponent;
442 }
443 if (exponent != 0) { out.append(' ').append(bits(factor, length(factor))).append('^').append(exponent); divisors *= exponent + 1; }
444 }
445 if (remaining != 1) { out.append(' ').append(bits(remaining, length(remaining))).append("^1"); divisors *= 2; }
446 line(out, " divisors %d", divisors);
447 return out.toString();
448 }
449 require(args.length == n + 2, "bad length"); require(number(args[1]) == 1, "bad parameter");
450 long size = 0;
451 for (int i = 1; i < args.length; ++i) size += number(args[i]);
452 boolean feasible = (size & (size - 1)) == 0;
453 line(out, "macwilliams n %d size %d", n, size); out.append("dual");
454 for (int j = 0; j <= n; ++j) {
455 long numerator = 0;
456 for (int x = 0; x <= n; ++x) numerator += number(args[x + 1]) * krawtchouk(n, j, x);
457 if (numerator < 0 || numerator % size != 0) feasible = false;
458 long divisor = gcd(numerator, size), denominator = size / divisor;
459 out.append(' ').append(numerator / divisor);
460 if (denominator != 1) out.append('/').append(denominator);
461 }
462 out.append('\n'); line(out, "feasible %s", yes(feasible));
463 return out.toString();
464 }

466 static final class Laboratory {
467 final TreeMap<String, Code> codes = new TreeMap<>();
468
469 String process(String input) {
470 String[] words = input.split(" "), args = Arrays.copyOfRange(words, 1, words.length);
471 Command command;
472 try { command = Command.valueOf(words[0]); }
473 catch (IllegalArgumentException error) { throw new CommandError("unknown command " + words[0]); }
474 require(args.length >= command.minimum && args.length <= command.maximum, "bad arguments");
475 for (int i = 0; i < args.length; ++i)
476 require(validWord(args[i], command.types.charAt(Math.min(i, command.types.length() - 1))), "bad arguments");
477 Code a = null, b = null;
478 for (int i = 0; i < command.types.length(); ++i) {
479 char kind = command.types.charAt(i);
480 if (kind != 'A' && kind != 'B') continue;
481 Code code = codes.get(args[i]); require(code != null, "no code " + args[i]);
482 if (kind == 'A') a = code; else b = code;
483 }
484 if (command.defining()) {
485 Code code = construct(command, args, a, b);
486 String status = codes.containsKey(args[0]) ? "redefined" : "defined";
487 codes.put(args[0], code);
488 return line("%s %s n %d k %d", status, args[0], code.n, code.k());
489 }
490 switch (command) {
491 case CRC, CRCCHECK, FACTOR, MACWILLIAMS, BOUNDS -> { return independent(command, args); }
492 default -> { }
493 }
494 StringBuilder out = new StringBuilder();
495 if (command == Command.LIST) {
496 line(out, "codes %d", codes.size());
497 for (var entry : codes.entrySet()) {
498 Code code = entry.getValue();
499 line(out, "%s n %d k %d encoder %s", entry.getKey(), code.n, code.k(), code.encoder());
500 }
501 return out.toString();
502 }
503 String name = args[0];
504 if (command == Command.DELETE) { codes.remove(name); return line("deleted %s", name); }
505 int n = a.n, k = a.k();
506 switch (command) {
507 case SHOW -> {
508 line(out, "code %s n %d k %d encoder %s", name, n, k, a.encoder());
509 if (a.polynomial != null) line(out, "poly g %s h %s", bits(a.polynomial, n - k + 1), bits(divide((1 << n) | 1, a.polynomial).quotient, k + 1));
510 matrixText(out, "G", a.g, n); matrixText(out, "H", a.h, n);
511 }
512 case INFO -> {
513 List<Integer> codewords = span(a.g);
514 int distance = n, generator = Integer.MAX_VALUE;
515 for (int word : codewords.subList(1, codewords.size())) { distance = Math.min(distance, weight(word)); generator = Math.min(generator, word); }
516 int t = (distance - 1) / 2; long divisor = gcd(k, n);
517 boolean cyclic = true;
518 for (int row : a.g) if (a.syndrome(((row << 1) & mask(n)) | (row >> (n - 1))) != 0) cyclic = false;
519 line(out, "info %s n %d k %d rate %d/%d d %d t %d perfect %s mds %s cyclic %s", name, n, k,
520 k / divisor, n / divisor, distance, t, yes(volume(n, t) == (1L << (n - k))), yes(distance == n - k + 1),
521 cyclic ? bits(generator, length(generator)) : "no");
522 }
523 case SYSTEMATIC -> { return systematic(name, a); }
524 case WEIGHTS -> {
525 int[] primary = distribution(span(a.g), 0, n), dual = distribution(span(a.h), 0, n);
526 boolean orthogonal = true;
527 for (int x : a.g) for (int y : a.g) if ((weight(x & y) & 1) != 0) orthogonal = false;
528 line(out, "weights %s %s", name, weightList(primary)); line(out, "dual %s %s", name, weightList(dual));
529 line(out, "duality %s self-orthogonal %s self-dual %s formally-self-dual %s", name,
530 yes(orthogonal), yes(orthogonal && n == 2 * k), yes(Arrays.equals(primary, dual)));
531 }
532 case COVERING -> {
533 int[] counts = new int[n + 1];
534 for (int leader : a.leaders()) ++counts[weight(leader)];
535 int radius = n; while (counts[radius] == 0) --radius;
536 out.append(format("covering %s radius %d leaders", name, radius));
537 for (int i = 0; i <= radius; ++i) out.append(' ').append(i).append(':').append(counts[i]);
538 out.append('\n');
539 }
540 case BURST -> {
541 int detect = n;
542 for (int word : span(a.g)) if (word != 0) detect = Math.min(detect, length(word) - Integer.numberOfTrailingZeros(word) - 1);
543 line(out, "burst %s detect %d correct %d", name, detect, a.burstCorrection());
544 }
545 case TRELLIS -> {
546 out.append("trellis ").append(name).append(" profile"); int maximum = 0;
547 for (int cut = 0; cut <= n; ++cut) {
548 List<Integer> prefix = new ArrayList<>(), suffix = new ArrayList<>();
549 for (int row : a.g) { prefix.add(row >> (n - cut)); suffix.add(row & mask(n - cut)); }
550 int states = basis(prefix, cut).size() + basis(suffix, n - cut).size() - k;
551 maximum = Math.max(maximum, states); out.append(' ').append(states);
552 }
553 line(out, " max %d", maximum);
554 }
555 case STDARRAY -> {
556 require(n - k <= 6 && k <= 4, "too large");
557 int[] leaders = a.leaders();
558 List<Integer> syndromes = new ArrayList<>(), codewords = span(a.g);
559 for (int i = 0; i < leaders.length; ++i) syndromes.add(i);
560 syndromes.sort(Comparator.comparingInt((Integer s) -> weight(leaders[s])).thenComparingInt(s -> leaders[s]));
561 codewords.sort(Comparator.naturalOrder());
562 line(out, "stdarray %s rows %d columns %d", name, leaders.length, codewords.size());
563 for (int s : syndromes) {
564 out.append(bits(s, n - k)).append(':');
565 for (int word : codewords) out.append(' ').append(bits(leaders[s] ^ word, n));
566 out.append('\n');
567 }
568 }
569 case EQUAL -> {
570 if (n != b.n) line(out, "equal %s %s no lengths %d %d", name, args[1], n, b.n);
571 else {
572 List<Integer> rows = new ArrayList<>(a.g); rows.addAll(b.g);
573 int common = k + b.k() - basis(rows, n).size();
574 line(out, "equal %s %s %s intersection %d", name, args[1], yes(k == common && b.k() == common), common);
575 }
576 }
577 case ENCODE -> {
578 require(args[1].length() == k, "bad length");
579 line(out, "encode %s message %s codeword %s", name, args[1], bits(a.encode(parseBits(args[1])), n));
580 }
581 case SYNDROME, DECODE, BOUNDED, COSET -> {
582 String text = args[command == Command.BOUNDED ? 2 : 1]; require(text.length() == n, "bad length");
583 int word = parseBits(text), syndrome = a.syndrome(word);
584 if (command == Command.SYNDROME) line(out, "syndrome %s %s codeword %s", name, bits(syndrome, n - k), yes(syndrome == 0));
585 else {
586 Decoded decoded = a.decode(word, a.leaders());
587 if (command == Command.COSET) line(out, "coset %s syndrome %s leader %s weights %s", name,
588 bits(syndrome, n - k), bits(decoded.error, n), weightList(distribution(span(a.g), word, n)));
589 else if (command == Command.BOUNDED && weight(decoded.error) > number(args[1])) line(out, "bounded %s fail weight %d", name, weight(decoded.error));
590 else line(out, "%s %s %s", command == Command.BOUNDED ? "bounded" : "decode", name, a.decodedText(decoded));
591 }
592 }
593 case ERASURE -> {
594 require(args[1].length() == n, "bad length");
595 int known = 0, value = 0, erased = 0, candidates = 0, candidate = 0;
596 for (int i = 0; i < n; ++i) {
597 char c = args[1].charAt(i);
598 if (c == '?') ++erased; else { known |= bit(n, i); if (c == '1') value |= bit(n, i); }
599 }
600 for (int word : span(a.g)) if ((word & known) == value) { candidate = word; ++candidates; }
601 out.append(format("erasure %s erased %d candidates %d", name, erased, candidates));
602 if (candidates == 1) out.append(format(" codeword %s message %s", bits(candidate, n), bits(a.message(candidate), k)));
603 out.append('\n');
604 }
605 case INTERLEAVE -> {
606 int depth = args.length - 3, start = number(args[1]), size = number(args[2]);
607 for (int i = 3; i < args.length; ++i) require(args[i].length() == k, "bad length");
608 require(start >= 1 && size >= 1 && start + size - 1 <= depth * n, "bad burst");
609 int[] sentRows = new int[depth], receivedRows = new int[depth];
610 for (int i = 0; i < depth; ++i) sentRows[i] = a.encode(parseBits(args[i + 3]));
611 StringBuilder sent = new StringBuilder(), received = new StringBuilder();
612 for (int column = 0; column < n; ++column) for (int i = 0; i < depth; ++i) {
613 int position = column * depth + i + 1;
614 boolean value = (sentRows[i] & bit(n, column)) != 0;
615 sent.append(value ? '1' : '0');
616 if (position >= start && position < start + size) value = !value;
617 received.append(value ? '1' : '0'); receivedRows[i] = (receivedRows[i] << 1) | (value ? 1 : 0);
618 }
619 line(out, "interleave %s depth %d burst %d %d", name, depth, start, size);
620 line(out, "sent %s", sent); line(out, "recv %s", received);
621 int[] leaders = a.leaders(); int correct = 0;
622 for (int i = 0; i < depth; ++i) {
623 Decoded decoded = a.decode(receivedRows[i], leaders);
624 boolean ok = decoded.message == parseBits(args[i + 3]); if (ok) ++correct;
625 line(out, "row %d errors %d corrected %d message %s %s", i + 1, weight(receivedRows[i] ^ sentRows[i]),
626 weight(decoded.error), bits(decoded.message, k), ok ? "ok" : "wrong");
627 }
628 line(out, "result ok %d wrong %d", correct, depth - correct);
629 }
630 default -> throw new IllegalStateException("Unhandled command");
631 }
632 return out.toString();
633 }
634 }
