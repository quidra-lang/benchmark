77static bool valid_word(const char *s, char type) {
78 size_t length = strlen(s);
79 if (type == 'N' || type == 'A' || type == 'B') {
80 if (length > 8 || !letter(s[0])) return false;
81 for (size_t i = 1; i < length; ++i)
82 if (!letter(s[i]) && (s[i] < '0' || s[i] > '9')) return false;
83 } else if (type == 'I') {
84 if (length > 4 || (length > 1 && s[0] == '0')) return false;
85 for (size_t i = 0; i < length; ++i)
86 if (s[i] < '0' || s[i] > '9') return false;
87 } else {
88 if (length > 30) return false;
89 for (size_t i = 0; i < length; ++i)
90 if (s[i] != '0' && s[i] != '1' && !(type == 'X' && s[i] == '?'))
91 return false;
92 }
93 return length != 0;
94}

120 Word result = 0;
121 for (; *s; ++s) result = (result << 1) | (Word)(*s - '0');

125static Bits bits(Word value, int n) {
126 Bits result;
127 for (int i = 0; i < n; ++i) result.text[i] = (value & bit(n, i)) ? '1' : '0';
128 result.text[n] = '\0';
129 return result;
130}

132static Division divide(Word value, Word divisor) {
133 Division result = {0, value};
134 int degree = length(divisor) - 1;
135 while (result.remainder && length(result.remainder) - 1 >= degree) {
136 int shift = length(result.remainder) - 1 - degree;
137 result.quotient ^= UINT32_C(1) << shift;
138 result.remainder ^= divisor << shift;
139 }
140 return result;
141}

143static Matrix basis(Matrix source, int n) {
144 Matrix result = {0};
145 Word pivots[32] = {0};
146 for (int i = 0; i < source.count; ++i) {
147 Word row = source.rows[i];
148 for (int j = n - 1; j >= 0; --j) {
149 if (!(row & (UINT32_C(1) << j))) continue;
150 if (pivots[j]) row ^= pivots[j];
151 else {
152 pivots[j] = row;
153 result.rows[result.count++] = source.rows[i];
154 break;
155 }
156 }
157 }
158 return result;
159}

161static int reduce(Matrix *matrix, int n, int pivots[32]) {
162 int rank = 0;
163 for (int column = 0; column < n; ++column) {
164 Word flag = bit(n, column);
165 int row = rank;
166 while (row < matrix->count && !(matrix->rows[row] & flag)) ++row;
167 if (row == matrix->count) continue;
168 Word temporary = matrix->rows[rank];
169 matrix->rows[rank] = matrix->rows[row];
170 matrix->rows[row] = temporary;
171 for (int i = 0; i < matrix->count; ++i)
172 if (i != rank && (matrix->rows[i] & flag)) matrix->rows[i] ^= matrix->rows[rank];
173 pivots[rank++] = column;
174 }
175 matrix->count = rank;
176 return rank;
177}

179static Matrix null_space(Matrix source, int n) {
180 int pivots[32];
181 int rank = reduce(&source, n, pivots);
182 Matrix result = {0};
183 for (int column = 0, next = 0; column < n; ++column) {
184 if (next < rank && pivots[next] == column) { ++next; continue; }
185 Word row = bit(n, column);
186 for (int i = 0; i < rank; ++i)
187 if (source.rows[i] & bit(n, column)) row |= bit(n, pivots[i]);
188 result.rows[result.count++] = row;
189 }
190 return result;
191}

193static Matrix reed_muller(int r, int m) {
194 Matrix result = {0};
195 if (r == 0) {
196 result.rows[result.count++] = mask(1 << m);
197 } else {
198 int half = 1 << (m - 1);
199 Matrix first = reed_muller(r < m ? r : m - 1, m - 1);
200 Matrix second = reed_muller(r - 1, m - 1);
201 for (int i = 0; i < first.count; ++i)
202 result.rows[result.count++] = (first.rows[i] << half) | first.rows[i];
203 for (int i = 0; i < second.count; ++i) result.rows[result.count++] = second.rows[i];
204 }
205 return result;
206}

208static Word remove_bit(Word row, int n, int position) {
209 int low = n - position;
210 return ((row >> (low + 1)) << low) | (row & mask(low));
211}

213static Code construct(Command command, char **args, int count, const Code *a, const Code *b) {
214 Code code = {0};
215 bool explicit_h = false;
216 if (command == GEN || command == PCHECK) {
217 code.n = (int)strlen(args[1]);
218 Matrix source = {0};
219 for (int i = 1; i < count; ++i) {
220 require((int)strlen(args[i]) == code.n, "bad length");
221 source.rows[source.count++] = parse_bits(args[i]);
222 }
223 if (command == GEN) code.g = basis(source, code.n);
224 else {
225 code.h = basis(source, code.n);
226 code.g = null_space(code.h, code.n);
227 explicit_h = true;
228 }
229 } else if (command == HAMMING || command == SIMPLEX) {
230 int r = atoi(args[1]);
231 require(r >= 2 && r <= 4, "bad parameter");
232 code.n = (1 << r) - 1;
233 Matrix h = {0};
234 for (int i = 0; i < r; ++i) {
235 Word row = 0;
236 for (int j = 1; j <= code.n; ++j) row = (row << 1) | ((j >> (r - i - 1)) & 1);
237 h.rows[h.count++] = row;
238 }
239 if (command == SIMPLEX) code.g = h;
240 else { code.h = h; code.g = null_space(h, code.n); explicit_h = true; }
241 } else if (command == REPETITION || command == PARITY) {
242 code.n = atoi(args[1]);
243 require(code.n >= 2 && code.n <= 24, "bad parameter");
244 if (command == REPETITION) code.g.rows[code.g.count++] = mask(code.n);
245 else for (int i = 0; i < code.n - 1; ++i) code.g.rows[code.g.count++] = bit(code.n, i) | 1;
246 } else if (command == RM) {
247 int r = atoi(args[1]), m = atoi(args[2]);
248 require(m >= 1 && m <= 4 && r <= m, "bad parameter");
249 code.n = 1 << m;
250 code.g = reed_muller(r, m);
251 } else if (command == CYCLIC) {
252 code.n = atoi(args[1]);
253 int d = (int)strlen(args[2]) - 1;
254 require(code.n >= 2 && code.n <= 24 && args[2][0] == '1' && d >= 1 && d < code.n,
255 "bad parameter");
256 code.polynomial = parse_bits(args[2]);
257 require(divide((UINT32_C(1) << code.n) | 1, code.polynomial).remainder == 0, "not a divisor");
258 int k = code.n - d;
259 for (int i = 0; i < k; ++i) code.g.rows[code.g.count++] = code.polynomial << (k - i - 1);
260 code.h.count = d;
261 for (int j = 0; j < code.n; ++j) {
262 Word remainder = divide(bit(code.n, j), code.polynomial).remainder;
263 for (int i = 0; i < d; ++i)
264 code.h.rows[i] = (code.h.rows[i] << 1) | ((remainder >> (d - i - 1)) & 1);
265 }
266 explicit_h = true;
267 } else if (command == DUAL) {
268 code.n = a->n; code.g = a->h; code.h = a->g; explicit_h = true;
269 } else if (command == EXTEND) {
270 code.n = a->n + 1;
271 for (int i = 0; i < a->g.count; ++i)
272 code.g.rows[code.g.count++] = (a->g.rows[i] << 1) | (weight(a->g.rows[i]) & 1);
273 } else if (command == PUNCTURE || command == SHORTEN || command == EXPURGATE) {
274 int position = command == EXPURGATE ? 0 : atoi(args[2]);
275 if (command != EXPURGATE) require(position >= 1 && position <= a->n, "bad position");
276 code.n = a->n - (command != EXPURGATE);
277 int removed = -1;
278 if (command != PUNCTURE) {
279 for (int i = 0; i < a->g.count; ++i) {
280 bool selected = command == EXPURGATE ? (weight(a->g.rows[i]) & 1) != 0
281 : (a->g.rows[i] & bit(a->n, position - 1)) != 0;
282 if (selected) { removed = i; break; }
283 }
284 }
285 for (int i = 0; i < a->g.count; ++i) {
286 if (i == removed) continue;
287 Word row = a->g.rows[i];
288 if (removed >= 0 && (command == EXPURGATE ? (weight(row) & 1) != 0
289 : (row & bit(a->n, position - 1)) != 0)) row ^= a->g.rows[removed];
290 if (command != EXPURGATE) row = remove_bit(row, a->n, position);
291 code.g.rows[code.g.count++] = row;
292 }
293 if (command == PUNCTURE) code.g = basis(code.g, code.n);
294 } else if (command == AUGMENT) {
295 code.n = a->n; code.g = a->g;
296 code.g.rows[code.g.count++] = mask(code.n);
297 code.g = basis(code.g, code.n);
298 } else if (command == PERMUTE) {
299 require(count - 2 == a->n, "bad permutation");
300 bool used[25] = {false};
301 for (int i = 2; i < count; ++i) {
302 int position = atoi(args[i]);
303 require(position >= 1 && position <= a->n && !used[position], "bad permutation");
304 used[position] = true;
305 }
306 code.n = a->n;
307 for (int i = 0; i < a->g.count; ++i) {
308 Word row = 0;
309 for (int j = 2; j < count; ++j)
310 row = (row << 1) | ((a->g.rows[i] & bit(a->n, atoi(args[j]) - 1)) != 0);
311 code.g.rows[code.g.count++] = row;
312 }
313 } else {
314 if (command == UUV) require(a->n == b->n, "bad length");
315 code.n = command == PRODUCT ? a->n * b->n : a->n + b->n;
316 int k = command == PRODUCT ? a->g.count * b->g.count : a->g.count + b->g.count;
317 check_dimension(code.n, k);
318 if (command == PRODUCT) {
319 for (int i = 0; i < a->g.count; ++i) for (int j = 0; j < b->g.count; ++j) {
320 Word row = 0;
321 for (int column = 0; column < a->n; ++column)
322 row = (row << b->n) | ((a->g.rows[i] & bit(a->n, column)) ? b->g.rows[j] : 0);
323 code.g.rows[code.g.count++] = row;
324 }
325 } else {
326 for (int i = 0; i < a->g.count; ++i)
327 code.g.rows[code.g.count++] = (a->g.rows[i] << b->n) | (command == UUV ? a->g.rows[i] : 0);
328 for (int i = 0; i < b->g.count; ++i) code.g.rows[code.g.count++] = b->g.rows[i];
329 }
330 }
331 check_dimension(code.n, code.g.count);
332 if (!explicit_h) code.h = null_space(code.g, code.n);
333 return code;
334}

336static Word syndrome(const Code *code, Word row) {
337 Word result = 0;
338 for (int i = 0; i < code->h.count; ++i) result = (result << 1) | (weight(row & code->h.rows[i]) & 1);
339 return result;
340}

342static Word encode(const Code *code, Word message) {
343 if (code->polynomial) {
344 Word shifted = message << code->h.count;
345 return shifted | divide(shifted, code->polynomial).remainder;
346 }
347 Word result = 0;
348 for (int i = 0; i < code->g.count; ++i)
349 if (message & bit(code->g.count, i)) result ^= code->g.rows[i];
350 return result;
351}

353static Word message_of(const Code *code, Word word) {
354 for (Word message = 0; message < (UINT32_C(1) << code->g.count); ++message)
355 if (encode(code, message) == word) return message;
356 return 0;
357}

359static int span(Matrix matrix, Word words[4096]) {
360 int count = 1;
361 words[0] = 0;
362 for (int i = 0; i < matrix.count; ++i) {
363 for (int j = 0; j < count; ++j) words[count + j] = words[j] ^ matrix.rows[i];
364 count *= 2;
365 }
366 return count;
367}

369static Leaders leaders(const Code *code) {
370 Leaders result;
371 result.count = 1 << code->h.count;
372 for (int i = 0; i < result.count; ++i) result.words[i] = UINT32_MAX;
373 result.words[0] = 0;
374 int found = 1;
375 Word limit = UINT32_C(1) << code->n;
376 for (int w = 1; w <= code->n; ++w) {
377 for (Word row = mask(w); row < limit;) {
378 Word s = syndrome(code, row);
379 if (result.words[s] == UINT32_MAX) {
380 result.words[s] = row;
381 if (++found == result.count) return result;
382 }
383 Word lowest = row & (0u - row), next = row + lowest;
384 row = next | (((next ^ row) >> 2) / lowest);
385 }
386 }
387 return result;
388}

390static void distribution(Matrix matrix, Word offset, int64_t counts[25]) {
391 Word words[4096];
392 int count = span(matrix, words);
393 memset(counts, 0, 25 * sizeof(*counts));
394 for (int i = 0; i < count; ++i) ++counts[weight(words[i] ^ offset)];
395}

402static int64_t gcd(int64_t a, int64_t b) {
403 if (a < 0) a = -a;
404 while (b) { int64_t remainder = a % b; a = b; b = remainder; }
405 return a;
406}

410static int64_t krawtchouk(int n, int j, int x) {
411 int64_t sum = 0;
412 for (int y = 0; y <= j; ++y) {
413 int64_t term = choose(x, y) * choose(n - x, j - y);
414 sum += y % 2 ? -term : term;
415 }
416 return sum;
417}

423static void systematic(const char *name, const Code *code) {
424 Matrix matrix = code->g;
425 int permutation[24];
426 for (int i = 0; i < code->n; ++i) permutation[i] = i + 1;
427 for (int r = 0; r < matrix.count; ++r) {
428 int column = r, row;
429 for (;;) {
430 row = r;
431 while (row < matrix.count && !(matrix.rows[row] & bit(code->n, column))) ++row;
432 if (row < matrix.count) break;
433 ++column;
434 }
435 if (column != r) {
436 Word left = bit(code->n, r), right = bit(code->n, column);
437 for (int i = 0; i < matrix.count; ++i)
438 if (!!(matrix.rows[i] & left) != !!(matrix.rows[i] & right)) matrix.rows[i] ^= left | right;
439 int temporary = permutation[r]; permutation[r] = permutation[column]; permutation[column] = temporary;
440 }
441 Word temporary = matrix.rows[r]; matrix.rows[r] = matrix.rows[row]; matrix.rows[row] = temporary;
442 for (int i = 0; i < matrix.count; ++i)
443 if (i != r && (matrix.rows[i] & bit(code->n, r))) matrix.rows[i] ^= matrix.rows[r];
444 }
445 printf("systematic %s perm", name);
446 for (int i = 0; i < code->n; ++i) printf(" %d", permutation[i]);
447 putchar('\n');
448 print_matrix('G', matrix, code->n);
449 for (int j = 0; j < code->h.count; ++j) {
450 Word row = 0;
451 for (int i = 0; i < matrix.count; ++i)
452 row = (row << 1) | ((matrix.rows[i] & bit(code->n, matrix.count + j)) != 0);
453 row = (row << code->h.count) | bit(code->h.count, j);
454 printf("H %s\n", bits(row, code->n).text);
455 }
456}

463static int burst_correction(const Code *code) {
464 bool seen[4096] = {true};
465 for (int size = 1; size <= code->n; ++size) {
466 Word variants = size == 1 ? 1 : UINT32_C(1) << (size - 2);
467 for (int offset = 0; offset <= code->n - size; ++offset) for (Word middle = 0; middle < variants; ++middle) {
468 Word row = size == 1 ? 1 : (UINT32_C(1) << (size - 1)) | (middle << 1) | 1;
469 Word s = syndrome(code, row << offset);
470 if (seen[s]) return size - 1;
471 seen[s] = true;
472 }
473 }
474 return code->n;
475}

500static void execute(Command command, char **args, int count) {
501 const Format *format = &formats[command];
502 require(count >= format->minimum && count <= format->maximum, "bad arguments");
503 int type_count = (int)strlen(format->types);
504 for (int i = 0; i < count; ++i)
505 require(valid_word(args[i], format->types[i < type_count ? i : type_count - 1]), "bad arguments");
506 const Code *a = NULL, *b = NULL;
507 for (int i = 0; i < type_count; ++i) {
508 char type = format->types[i];
509 if (type != 'A' && type != 'B') continue;
510 int index = find_code(args[i]);
511 if (index < 0) {
512 printf("error: no code %s\n", args[i]);
513 longjmp(command_error, 1);
514 }
515 if (type == 'A') a = &registry[index].code; else b = &registry[index].code;
516 }
517 if (command <= PRODUCT) {
518 Code code = construct(command, args, count, a, b);
519 int index = find_code(args[0]);
520 bool replacing = index >= 0;
521 if (!replacing) { index = registry_count++; strcpy(registry[index].name, args[0]); }
522 registry[index].code = code;
523 printf("%s %s n %d k %d\n", replacing ? "redefined" : "defined", args[0], code.n, code.g.count);
524 return;
525 }
526 switch (command) {
527 case LIST: {
528 const Entry *sorted[500];
529 for (int i = 0; i < registry_count; ++i) sorted[i] = &registry[i];
530 qsort(sorted, registry_count, sizeof(*sorted), compare_entries);
531 printf("codes %d\n", registry_count);
532 for (int i = 0; i < registry_count; ++i)
533 printf("%s n %d k %d encoder %s\n", sorted[i]->name, sorted[i]->code.n,
534 sorted[i]->code.g.count, encoder(&sorted[i]->code));
535 break;
536 }
537 case DELETE: {
538 int index = find_code(args[0]);
539 memmove(registry + index, registry + index + 1, (registry_count - index - 1) * sizeof(*registry));
540 --registry_count;
541 printf("deleted %s\n", args[0]);
542 break;
543 }
544 case SHOW:
545 printf("code %s n %d k %d encoder %s\n", args[0], a->n, a->g.count, encoder(a));
546 if (a->polynomial) printf("poly g %s h %s\n", bits(a->polynomial, a->h.count + 1).text,
547 bits(divide((UINT32_C(1) << a->n) | 1, a->polynomial).quotient, a->g.count + 1).text);
548 print_matrix('G', a->g, a->n); print_matrix('H', a->h, a->n);
549 break;
550 case INFO: {
551 Word words[4096], generator = UINT32_MAX;
552 int total = span(a->g, words), distance = a->n;
553 for (int i = 1; i < total; ++i) {
554 int w = weight(words[i]);
555 if (w < distance) distance = w;
556 if (words[i] < generator) generator = words[i];
557 }
558 int t = (distance - 1) / 2;
559 int64_t volume = 0, divisor = gcd(a->g.count, a->n);
560 for (int i = 0; i <= t; ++i) volume += choose(a->n, i);
561 bool cyclic = true;
562 for (int i = 0; i < a->g.count; ++i) {
563 Word row = a->g.rows[i];
564 if (syndrome(a, ((row << 1) & mask(a->n)) | (row >> (a->n - 1)))) cyclic = false;
565 }
566 printf("info %s n %d k %d rate %" PRId64 "/%" PRId64 " d %d t %d perfect %s mds %s cyclic %s\n",
567 args[0], a->n, a->g.count, a->g.count / divisor, a->n / divisor, distance, t,
568 yes(volume == (INT64_C(1) << a->h.count)), yes(distance == a->h.count + 1),
569 cyclic ? bits(generator, length(generator)).text : "no");
570 break;
571 }
572 case SYSTEMATIC: systematic(args[0], a); break;
573 case WEIGHTS: {
574 int64_t primary[25], dual[25];
575 distribution(a->g, 0, primary); distribution(a->h, 0, dual);
576 printf("weights %s", args[0]); print_distribution(primary, a->n);
577 printf("dual %s", args[0]); print_distribution(dual, a->n);
578 bool orthogonal = true, formal = true;
579 for (int i = 0; i < a->g.count; ++i) for (int j = 0; j < a->g.count; ++j)
580 if (weight(a->g.rows[i] & a->g.rows[j]) & 1) orthogonal = false;
581 for (int i = 0; i <= a->n; ++i) if (primary[i] != dual[i]) formal = false;
582 printf("duality %s self-orthogonal %s self-dual %s formally-self-dual %s\n",
583 args[0], yes(orthogonal), yes(orthogonal && a->n == 2 * a->g.count), yes(formal));
584 break;
585 }
586 case COVERING: {
587 Leaders table = leaders(a);
588 int counts[25] = {0}, radius = 0;
589 for (int i = 0; i < table.count; ++i) {
590 int w = weight(table.words[i]); ++counts[w];
591 if (w > radius) radius = w;
592 }
593 printf("covering %s radius %d leaders", args[0], radius);
594 for (int i = 0; i <= radius; ++i) printf(" %d:%d", i, counts[i]);
595 putchar('\n');
596 break;
597 }
598 case BURST: {
599 Word words[4096];
600 int total = span(a->g, words), minimum = a->n;
601 for (int i = 1; i < total; ++i) {
602 int size = burst_length(words[i]);
603 if (size < minimum) minimum = size;
604 }
605 printf("burst %s detect %d correct %d\n", args[0], minimum - 1, burst_correction(a));
606 break;
607 }
608 case TRELLIS: {
609 printf("trellis %s profile", args[0]);
610 int maximum = 0;
611 for (int i = 0; i <= a->n; ++i) {
612 Matrix prefix = a->g, suffix = a->g;
613 for (int j = 0; j < a->g.count; ++j) {
614 prefix.rows[j] >>= a->n - i;
615 suffix.rows[j] &= mask(a->n - i);
616 }
617 int states = basis(prefix, i).count + basis(suffix, a->n - i).count - a->g.count;
618 if (states > maximum) maximum = states;
619 printf(" %d", states);
620 }
621 printf(" max %d\n", maximum);
622 break;
623 }
624 case STDARRAY: {
625 require(a->h.count <= 6 && a->g.count <= 4, "too large");
626 Leaders table = leaders(a);
627 Coset ordered[64];
628 for (int i = 0; i < table.count; ++i) ordered[i] = (Coset){(Word)i, table.words[i]};
629 qsort(ordered, table.count, sizeof(*ordered), compare_cosets);
630 Word words[4096];
631 int total = span(a->g, words);
632 qsort(words, total, sizeof(*words), compare_words);
633 printf("stdarray %s rows %d columns %d\n", args[0], table.count, total);
634 for (int i = 0; i < table.count; ++i) {
635 printf("%s:", bits(ordered[i].syndrome, a->h.count).text);
636 for (int j = 0; j < total; ++j) printf(" %s", bits(ordered[i].leader ^ words[j], a->n).text);
637 putchar('\n');
638 }
639 break;
640 }
641 case EQUAL: {
642 if (a->n != b->n) printf("equal %s %s no lengths %d %d\n", args[0], args[1], a->n, b->n);
643 else {
644 Matrix combined = a->g;
645 for (int i = 0; i < b->g.count; ++i) combined.rows[combined.count++] = b->g.rows[i];
646 int common = a->g.count + b->g.count - basis(combined, a->n).count;
647 printf("equal %s %s %s intersection %d\n", args[0], args[1],
648 yes(a->g.count == common && b->g.count == common), common);
649 }
650 break;
651 }
652 case ENCODE:
653 require((int)strlen(args[1]) == a->g.count, "bad length");
654 printf("encode %s message %s codeword %s\n", args[0], args[1], bits(encode(a, parse_bits(args[1])), a->n).text);
655 break;
656 case SYNDROME: {
657 require((int)strlen(args[1]) == a->n, "bad length");
658 Word s = syndrome(a, parse_bits(args[1]));
659 printf("syndrome %s %s codeword %s\n", args[0], bits(s, a->h.count).text, yes(s == 0));
660 break;
661 }
662 case DECODE: case BOUNDED: case COSET: {
663 const char *source = args[command == BOUNDED ? 2 : 1];
664 require((int)strlen(source) == a->n, "bad length");
665 Word word = parse_bits(source), s = syndrome(a, word);
666 Leaders table = leaders(a);
667 Word error = table.words[s];
668 if (command == COSET) {
669 int64_t counts[25]; distribution(a->g, word, counts);
670 printf("coset %s syndrome %s leader %s weights", args[0], bits(s, a->h.count).text, bits(error, a->n).text);
671 print_distribution(counts, a->n);
672 } else if (command == BOUNDED && weight(error) > atoi(args[1]))
673 printf("bounded %s fail weight %d\n", args[0], weight(error));
674 else decoded(command == BOUNDED ? "bounded" : "decode", args[0], a, word, error);
675 break;
676 }
677 case ERASURE: {
678 require((int)strlen(args[1]) == a->n, "bad length");
679 Word known = 0, value = 0, candidate = 0, message = 0;
680 int erased = 0, candidates = 0;
681 for (int i = 0; i < a->n; ++i) {
682 if (args[1][i] == '?') ++erased;
683 else { known |= bit(a->n, i); if (args[1][i] == '1') value |= bit(a->n, i); }
684 }
685 for (Word u = 0; u < (UINT32_C(1) << a->g.count); ++u) {
686 Word word = encode(a, u);
687 if ((word & known) == value) { ++candidates; candidate = word; message = u; }
688 }
689 printf("erasure %s erased %d candidates %d", args[0], erased, candidates);
690 if (candidates == 1) printf(" codeword %s message %s", bits(candidate, a->n).text, bits(message, a->g.count).text);
691 putchar('\n');
692 break;
693 }
694 case INTERLEAVE: {
695 int depth = count - 3, start = atoi(args[1]), size = atoi(args[2]);
696 for (int i = 3; i < count; ++i) require((int)strlen(args[i]) == a->g.count, "bad length");
697 require(start >= 1 && size >= 1 && start + size - 1 <= depth * a->n, "bad burst");
698 Word sent_rows[8], received_rows[8] = {0};
699 char sent[193], received[193];
700 for (int i = 0; i < depth; ++i) sent_rows[i] = encode(a, parse_bits(args[i + 3]));
701 for (int column = 0; column < a->n; ++column) for (int i = 0; i < depth; ++i) {
702 int position = column * depth + i;
703 bool value = (sent_rows[i] & bit(a->n, column)) != 0;
704 sent[position] = value ? '1' : '0';
705 if (position >= start - 1 && position < start - 1 + size) value = !value;
706 received[position] = value ? '1' : '0';
707 received_rows[i] = (received_rows[i] << 1) | value;
708 }
709 sent[depth * a->n] = received[depth * a->n] = '\0';
710 printf("interleave %s depth %d burst %d %d\nsent %s\nrecv %s\n", args[0], depth, start, size, sent, received);
711 Leaders table = leaders(a);
712 int correct = 0;
713 for (int i = 0; i < depth; ++i) {
714 Word error = table.words[syndrome(a, received_rows[i])];
715 Word message = message_of(a, received_rows[i] ^ error);
716 bool ok = message == parse_bits(args[i + 3]);
717 correct += ok;
718 printf("row %d errors %d corrected %d message %s %s\n", i + 1,
719 weight(received_rows[i] ^ sent_rows[i]), weight(error), bits(message, a->g.count).text, ok ? "ok" : "wrong");
720 }
721 printf("result ok %d wrong %d\n", correct, depth - correct);
722 break;
723 }
724 case CRC: case CRCCHECK: {
725 int d = (int)strlen(args[0]) - 1;
726 require(args[0][0] == '1' && d >= 1 && d <= 16, "bad parameter");
727 Word polynomial = parse_bits(args[0]), value = parse_bits(args[1]);
728 if (command == CRC) {
729 require((int)strlen(args[1]) + d <= 30, "bad length");
730 Word remainder = divide(value << d, polynomial).remainder;
731 printf("crc remainder %s frame %s%s\n", bits(remainder, d).text, args[1], bits(remainder, d).text);
732 } else {
733 Word remainder = divide(value, polynomial).remainder;
734 printf("crccheck %s remainder %s\n", remainder ? "error" : "ok", bits(remainder, d).text);
735 }
736 break;
737 }
738 case FACTOR: {
739 int n = atoi(args[0]);
740 require(n >= 1 && n <= 24, "bad parameter");
741 Word remaining = (UINT32_C(1) << n) | 1;
742 int64_t divisors = 1;
743 printf("factor %d", n);
744 for (Word factor = 3; 2 * (length(factor) - 1) <= length(remaining) - 1; factor += 2) {
745 int exponent = 0;
746 Division result = divide(remaining, factor);
747 while (!result.remainder) {
748 ++exponent; remaining = result.quotient;
749 result = divide(remaining, factor);
750 }
751 if (exponent) {
752 printf(" %s^%d", bits(factor, length(factor)).text, exponent);
753 divisors *= exponent + 1;
754 }
755 }
756 if (remaining != 1) { printf(" %s^1", bits(remaining, length(remaining)).text); divisors *= 2; }
757 printf(" divisors %" PRId64 "\n", divisors);
758 break;
759 }
760 case MACWILLIAMS: {
761 int n = atoi(args[0]);
762 require(n >= 1 && n <= 24, "bad parameter");
763 require(count - 1 == n + 1, "bad length");
764 require(atoi(args[1]) == 1, "bad parameter");
765 int64_t size = 0;
766 for (int i = 1; i < count; ++i) size += atoi(args[i]);
767 bool feasible = (size & (size - 1)) == 0;
768 printf("macwilliams n %d size %" PRId64 "\ndual", n, size);
769 for (int j = 0; j <= n; ++j) {
770 int64_t numerator = 0;
771 for (int x = 0; x <= n; ++x) numerator += atoi(args[x + 1]) * krawtchouk(n, j, x);
772 if (numerator < 0 || numerator % size) feasible = false;
773 int64_t divisor = gcd(numerator, size), denominator = size / divisor;
774 printf(" %" PRId64, numerator / divisor);
775 if (denominator != 1) printf("/%" PRId64, denominator);
776 }
777 printf("\nfeasible %s\n", yes(feasible));
778 break;
779 }
780 case BOUNDS: {
781 int n = atoi(args[0]), d = atoi(args[1]);
782 require(d >= 1 && d <= n && n <= 24, "bad parameter");
783 int64_t volume = 0, sum = 0;
784 for (int i = 0; i <= (d - 1) / 2; ++i) volume += choose(n, i);
785 int hamming = 0, griesmer = 0;
786 for (int k = 0; k <= n; ++k) if ((INT64_C(1) << k) * volume <= (INT64_C(1) << n)) hamming = k;
787 for (int k = 1; k <= n; ++k) {
788 int64_t denominator = INT64_C(1) << (k - 1);
789 sum += (d + denominator - 1) / denominator;
790 if (sum <= n) griesmer = k;
791 }
792 printf("bounds n %d d %d singleton %d hamming %d griesmer %d\n", n, d, n - d + 1, hamming, griesmer);
793 break;
794 }
795 default: break;
796 }
