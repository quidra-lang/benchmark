114def bursts(length, size):
115 if size == 1:
116 patterns = [1]
117 else:
118 patterns = [(1 << (size - 1)) | (inner << 1) | 1 for inner in range(1 << (size - 2))]
119 for shift in range(length - size + 1):

124def degree(polynomial):
125 return polynomial.bit_length() - 1

128def poly_divmod(dividend, divisor):
129 quotient = 0
130 while degree(dividend) >= degree(divisor):
131 shift = degree(dividend) - degree(divisor)
132 quotient |= 1 << shift
133 dividend ^= divisor << shift
134 return quotient, dividend

137def poly_mod(dividend, divisor):
138 return poly_divmod(dividend, divisor)[1]

141def factorize(polynomial):
142 factors = []
143 candidate = 2
144 while 2 * degree(candidate) <= degree(polynomial):
145 exponent = 0
146 quotient, remainder = poly_divmod(polynomial, candidate)
147 while remainder == 0:
148 polynomial = quotient
149 exponent += 1
150 quotient, remainder = poly_divmod(polynomial, candidate)
151 if exponent > 0:
152 factors.append((candidate, exponent))
153 candidate += 1
154 if polynomial != 1:
155 factors.append((polynomial, 1))
156 return factors

159def reduced_echelon(vectors, length):
160 rows = list(vectors)
161 pivots = []
162 for position in range(1, length + 1):
163 top = len(pivots)
164 found = next((i for i in range(top, len(rows)) if bit(rows[i], length, position)), None)
165 if found is None:
166 continue
167 rows[top], rows[found] = rows[found], rows[top]
168 for i in range(len(rows)):
169 if i != top and bit(rows[i], length, position):
170 rows[i] ^= rows[top]
171 pivots.append(position)
172 return rows[:len(pivots)], pivots

175def null_space(vectors, length):
176 rows, pivots = reduced_echelon(vectors, length)
177 result = []
178 for free in range(1, length + 1):
179 if free in pivots:
180 continue
181 vector = 1 << (length - free)
182 for row, pivot in zip(rows, pivots):
183 if bit(row, length, free):
184 vector |= 1 << (length - pivot)
185 result.append(vector)
186 return result

189class Echelon:
190 def __init__(self):
191 self.rows = []
192
193 def reduce(self, vector):
194 combination = 0
195 for row, leading, row_combination in self.rows:
196 if vector & leading:
197 vector ^= row
198 combination ^= row_combination
199 return vector, combination
200
201 def insert(self, vector, combination=0):
202 residue, used = self.reduce(vector)
203 if residue == 0:
204 return False
205 self.rows.append((residue, 1 << degree(residue), combination ^ used))
206 return True

258def reed_muller(order, m):
259 if order == 0:
260 return [all_ones(1 << m)]
261 half = 1 << (m - 1)
262 doubled = [(row << half) | row for row in reed_muller(min(order, m - 1), m - 1)]
263 return doubled + reed_muller(order - 1, m - 1)

267 if True not in marked:
268 return list(rows)
269 top = marked.index(True)

282class Code:
283 def __init__(self, length, generator, check, polynomial):
284 self.n = length
285 self.k = len(generator)
286 self.generator = generator
287 self.check = check
288 self.polynomial = polynomial
289
290 def encoder(self):
291 return "matrix" if self.polynomial is None else "polynomial"
292
293 def syndrome(self, word):
294 value = 0
295 for row in self.check:
296 value = (value << 1) | parity(row & word)
297 return value
298
299 def encode(self, message):
300 if self.polynomial is not None:
301 shifted = message << (self.n - self.k)
302 return shifted | poly_mod(shifted, self.polynomial)
303 codeword = 0
304 for position, row in enumerate(self.generator, 1):
305 if bit(message, self.k, position):
306 codeword ^= row
307 return codeword
308
309 def message_of(self, codeword):
310 if self.polynomial is not None:
311 return codeword >> (self.n - self.k)
312 echelon = Echelon()
313 for position, row in enumerate(self.generator, 1):
314 echelon.insert(row, 1 << (self.k - position))
315 return echelon.reduce(codeword)[1]
316
317 def codewords(self):
318 return span(self.generator)
319
320 def coset_leaders(self):
321 leaders = [-1] * (1 << (self.n - self.k))
322 leaders[0] = 0
323 remaining = len(leaders) - 1
324 for size in range(1, self.n + 1):
325 vector = all_ones(size)
326 while vector < 1 << self.n:
327 syndrome = self.syndrome(vector)
328 if leaders[syndrome] < 0:
329 leaders[syndrome] = vector
330 remaining -= 1
331 if remaining == 0:
332 return leaders
333 vector = next_combination(vector)
334 return leaders
335
336 def decode(self, word, leaders):
337 error = leaders[self.syndrome(word)]
338 return word ^ error, error
339
340 def burst_correction(self):
341 seen = [False] * (1 << (self.n - self.k))
342 seen[0] = True
343 for size in range(1, self.n + 1):
344 for burst in bursts(self.n, size):
345 syndrome = self.syndrome(burst)
346 if seen[syndrome]:
347 return size - 1
348 seen[syndrome] = True
349 return self.n
350
351 def received(self, text):
352 require(len(text) == self.n, "bad length")
353 return int(text, 2)
354
355 def decoding_fields(self, codeword, error):
356 return [
357 "codeword", bit_text(codeword, self.n),
358 "message", bit_text(self.message_of(codeword), self.k),
359 "error", bit_text(error, self.n),
360 "weight", weight(error),
361 ]

415 def check_arguments(self, command, kinds, arguments):

474 def cyclic_code(self, arguments):
475 length, text = int(arguments[1]), arguments[2]
476 require(
477 2 <= length <= MAX_LENGTH and text[0] == "1" and 2 <= len(text) <= length,
478 "bad parameter",
479 )
480 generator = int(text, 2)
481 redundancy = len(text) - 1
482 require(poly_mod((1 << length) | 1, generator) == 0, "not a divisor")
483 dimension = length - redundancy
484 rows = [generator << (dimension - i) for i in range(1, dimension + 1)]
485 columns = [poly_mod(1 << (length - j), generator) for j in range(1, length + 1)]
486 self.define(arguments[0], length, rows, transpose(columns, redundancy), generator)

594 def systematic(self, arguments):
595 name = arguments[0]
596 code = self.codes[name]
597 n, k = code.n, code.k
598 matrix = list(code.generator)
599 order = list(range(1, n + 1))
600 for r in range(1, k + 1):
601 if not any(bit(row, n, r) for row in matrix[r - 1:]):
602 target = next(
603 c for c in range(r + 1, n + 1)
604 if any(bit(row, n, c) for row in matrix[r - 1:])
605 )
606 matrix = [swap_bits(row, n, r, target) for row in matrix]
607 order[r - 1], order[target - 1] = order[target - 1], order[r - 1]
608 found = next(i for i in range(r - 1, k) if bit(matrix[i], n, r))
609 matrix[r - 1], matrix[found] = matrix[found], matrix[r - 1]
610 for i in range(k):
611 if i != r - 1 and bit(matrix[i], n, r):
612 matrix[i] ^= matrix[r - 1]
613 self.emit("systematic", name, "perm", *order)
614 for row in matrix:
615 self.emit("G", bit_text(row, n))
616 for j in range(1, n - k + 1):
617 identity = bit_text(1 << (n - k - j), n - k)
618 self.emit("H", bit_text(column(matrix, n, k + j), k) + identity)

661 def standard_array(self, arguments):
662 name = arguments[0]
663 code = self.codes[name]
664 require(code.n - code.k <= 6 and code.k <= 4, "too large")
665 leaders = code.coset_leaders()
666 words = sorted(code.codewords())
667 self.emit("stdarray", name, "rows", len(leaders), "columns", len(words))
668 syndromes = sorted(range(len(leaders)), key=lambda s: (weight(leaders[s]), leaders[s]))
669 for syndrome in syndromes:
670 row = [bit_text(leaders[syndrome] ^ word, code.n) for word in words]
671 self.emit(bit_text(syndrome, code.n - code.k) + ":", *row)

673 def equal(self, arguments):
674 first_name, second_name = arguments
675 first, second = self.codes[first_name], self.codes[second_name]
676 if first.n != second.n:
677 self.emit("equal", first_name, second_name, "no", "lengths", first.n, second.n)
678 return

680 same = first.k == second.k == common

728 def erasure(self, arguments):
729 name, text = arguments
730 code = self.codes[name]
731 require(len(text) == code.n, "bad length")
732 known = int("".join("0" if symbol == "?" else "1" for symbol in text), 2)
733 value = int(text.replace("?", "0"), 2)
734 candidates = [word for word in code.codewords() if (word & known) == value]
735 erased = text.count("?")
736 if len(candidates) == 1:
737 candidate = candidates[0]
738 self.emit(
739 "erasure", name, "erased", erased, "candidates", 1,
740 "codeword", bit_text(candidate, code.n),
741 "message", bit_text(code.message_of(candidate), code.k),
742 )
743 else:
744 self.emit("erasure", name, "erased", erased, "candidates", len(candidates))

746 def interleave(self, arguments):
747 name, start, size = arguments[0], int(arguments[1]), int(arguments[2])
748 code = self.codes[name]
749 n = code.n
750 messages = arguments[3:]
751 depth = len(messages)
752 require(all(len(text) == code.k for text in messages), "bad length")
753 require(start >= 1 and size >= 1 and start + size - 1 <= depth * n, "bad burst")
754 codewords = [code.encode(int(text, 2)) for text in messages]
755 sent = [0] * (depth * n)
756 for i, codeword in enumerate(codewords):
757 for j in range(n):
758 sent[j * depth + i] = bit(codeword, n, j + 1)
759 received = list(sent)
760 for position in range(start - 1, start + size - 1):
761 received[position] ^= 1
762 self.emit("interleave", name, "depth", depth, "burst", start, size)
763 self.emit("sent", "".join(str(value) for value in sent))
764 self.emit("recv", "".join(str(value) for value in received))
765 leaders = code.coset_leaders()
766 correct_rows = 0
767 for i, codeword in enumerate(codewords):
768 word = 0
769 for j in range(n):
770 word = (word << 1) | received[j * depth + i]
771 decoded, error = code.decode(word, leaders)
772 message = code.message_of(decoded)
773 correct = message == int(messages[i], 2)
774 if correct:
775 correct_rows += 1
776 self.emit(
777 "row", i + 1, "errors", weight(word ^ codeword), "corrected", weight(error),
778 "message", bit_text(message, code.k), "ok" if correct else "wrong",
779 )
780 self.emit("result", "ok", correct_rows, "wrong", depth - correct_rows)

806 def macwilliams(self, arguments):
807 n = int(arguments[0])
808 counts = [int(text) for text in arguments[1:]]
809 require(1 <= n <= MAX_LENGTH, "bad parameter")
810 require(len(counts) == n + 1, "bad length")
811 require(counts[0] == 1, "bad parameter")
812 size = sum(counts)
813 numerators = [
814 sum(count * krawtchouk(n, j, x) for x, count in enumerate(counts))
815 for j in range(n + 1)
816 ]
817 power_of_two = (size & (size - 1)) == 0
818 integral = all(value >= 0 and value % size == 0 for value in numerators)
819 self.emit("macwilliams", "n", n, "size", size)
820 self.emit("dual", *(fraction_text(value, size) for value in numerators))
821 self.emit("feasible", yes_no(power_of_two and integral))

823 def bounds(self, arguments):
824 n, d = int(arguments[0]), int(arguments[1])
825 require(1 <= d <= n <= MAX_LENGTH, "bad parameter")
826 volume = sum(comb(n, i) for i in range((d - 1) // 2 + 1))
827 hamming = max(k for k in range(n + 1) if (volume << k) <= (1 << n))
828 griesmer = max(
829 k for k in range(1, n + 1)
830 if sum((d + (1 << i) - 1) >> i for i in range(k)) <= n
831 )
832 self.emit(
833 "bounds", "n", n, "d", d, "singleton", n - d + 1,
834 "hamming", hamming, "griesmer", griesmer,
835 )
