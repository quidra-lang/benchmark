28fun letter(c: Char) = c in 'A'..'Z' || c in 'a'..'z'
29fun digit(c: Char) = c in '0'..'9'
30fun validWord(text: String, kind: Char): Boolean {
31 if (text.isEmpty()) return false
32 return when (kind) {
33 'N', 'A', 'B' -> text.length <= 8 && letter(text[0]) && text.all { letter(it) || digit(it) }
34 'I' -> text.length <= 4 && (text.length == 1 || text[0] != '0') && text.all(::digit)
35 else -> text.length <= 30 && text.all { it == '0' || it == '1' || kind == 'X' && it == '?' }
36 }
37}

39fun mask(n: Int) = (1 shl n) - 1
40fun bit(n: Int, column: Int) = 1 shl (n - column - 1)
41fun weight(value: Int) = Integer.bitCount(value)
42fun length(value: Int) = 32 - Integer.numberOfLeadingZeros(value)
43fun bits(value: Int, n: Int) = value.toString(2).padStart(n, '0')

47fun divide(original: Int, divisor: Int): Division {
48 var value = original
49 var quotient = 0
50 while (value != 0 && length(value) >= length(divisor)) {
51 val shift = length(value) - length(divisor)
52 quotient = quotient xor (1 shl shift)
53 value = value xor (divisor shl shift)
54 }
55 return Division(quotient, value)
56}

58fun basis(rows: List<Int>, n: Int): List<Int> {
59 val pivots = IntArray(32)
60 val result = mutableListOf<Int>()
61 for (original in rows) {
62 var row = original
63 for (column in n - 1 downTo 0) {
64 if ((row and (1 shl column)) == 0) continue
65 if (pivots[column] != 0) row = row xor pivots[column]
66 else { pivots[column] = row; result.add(original); break }
67 }
68 }
69 return result
70}

72fun rref(source: List<Int>, n: Int): Echelon {
73 val rows = source.toMutableList()
74 val columns = mutableListOf<Int>()
75 for (column in 0 until n) {
76 val rank = columns.size
77 val found = (rank until rows.size).firstOrNull { (rows[it] and bit(n, column)) != 0 } ?: continue
78 val temporary = rows[rank]; rows[rank] = rows[found]; rows[found] = temporary
79 for (i in rows.indices)
80 if (i != rank && (rows[i] and bit(n, column)) != 0) rows[i] = rows[i] xor rows[rank]
81 columns.add(column)
82 }
83 return Echelon(rows.take(columns.size), columns)
84}

86fun nullSpace(source: List<Int>, n: Int): List<Int> {
87 val reduced = rref(source, n)
88 return (0 until n).filter { it !in reduced.columns }.map { column ->
89 var row = bit(n, column)
90 for (i in reduced.rows.indices)
91 if ((reduced.rows[i] and bit(n, column)) != 0) row = row or bit(n, reduced.columns[i])
92 row
93 }
94}

105fun reedMuller(r: Int, m: Int): List<Int> {
106 if (r == 0) return listOf(mask(1 shl m))
107 val half = 1 shl (m - 1)
108 return reedMuller(minOf(r, m - 1), m - 1).map { (it shl half) or it } + reedMuller(r - 1, m - 1)
109}

115class Code(val n: Int, generator: List<Int>, check: List<Int>? = null, val polynomial: Int? = null) {
116 val g: List<Int>
117 val h: List<Int>
118 val k get() = g.size
119 val encoder get() = if (polynomial == null) "matrix" else "polynomial"
120
121 init {
122 checkDimension(n, generator.size)
123 g = generator.toList()
124 h = (check ?: nullSpace(g, n)).toList()
125 }

127 fun encode(message: Int): Int {
128 if (polynomial != null) {
129 val shifted = message shl h.size
130 return shifted or divide(shifted, polynomial).remainder
131 }
132 var word = 0
133 for (i in g.indices) if ((message and bit(k, i)) != 0) word = word xor g[i]
134 return word
135 }
136
137 fun syndrome(word: Int): Int {
138 var result = 0
139 for (row in h) result = (result shl 1) or (weight(row and word) and 1)
140 return result
141 }
142
143 fun messageFor(word: Int): Int = (0 until (1 shl k)).first { encode(it) == word }
144
145 fun leaders(): IntArray {
146 val result = IntArray(1 shl h.size) { -1 }
147 result[0] = 0
148 var found = 1
149 for (w in 1..n) {
150 var word = mask(w)
151 while (word < (1 shl n)) {
152 val syndrome = syndrome(word)
153 if (result[syndrome] == -1) {
154 result[syndrome] = word
155 if (++found == result.size) return result
156 }
157 val lowest = word and -word
158 val next = word + lowest
159 word = next or (((next xor word) shr 2) / lowest)
160 }
161 }
162 return result
163 }
164
165 fun decode(received: Int, leaders: IntArray): Decoded {
166 val error = leaders[syndrome(received)]
167 val word = received xor error
168 return Decoded(word, messageFor(word), error)
169 }

171 fun decodedText(decoded: Decoded) = "codeword ${bits(decoded.word, n)} message ${bits(decoded.message, k)} " +
172 "error ${bits(decoded.error, n)} weight ${weight(decoded.error)}"

174 fun burstCorrection(): Int {
175 val seen = BooleanArray(1 shl h.size)
176 seen[0] = true
177 for (size in 1..n) {
178 val variants = if (size == 1) 1 else 1 shl (size - 2)
179 for (offset in 0..n - size) for (middle in 0 until variants) {
180 val word = if (size == 1) 1 else (1 shl (size - 1)) or (middle shl 1) or 1
181 val syndrome = syndrome(word shl offset)
182 if (seen[syndrome]) return size - 1
183 seen[syndrome] = true
184 }
185 }
186 error("No colliding burst")
187 }
188}

190fun construct(command: Command, args: List<String>, a: Code?, b: Code?): Code = when (command) {
191 Command.GEN, Command.PCHECK -> {
192 val n = args[1].length
193 ensure(args.drop(1).all { it.length == n }, "bad length")
194 val rows = basis(args.drop(1).map { it.toInt(2) }, n)
195 if (command == Command.GEN) Code(n, rows) else Code(n, nullSpace(rows, n), rows)
196 }
197 Command.HAMMING, Command.SIMPLEX -> {
198 val r = args[1].toInt()
199 ensure(r in 2..4, "bad parameter")
200 val n = (1 shl r) - 1
201 val rows = (0 until r).map { i ->
202 var row = 0
203 for (j in 1..n) row = (row shl 1) or ((j shr (r - i - 1)) and 1)
204 row
205 }
206 if (command == Command.SIMPLEX) Code(n, rows) else Code(n, nullSpace(rows, n), rows)
207 }
208 Command.REPETITION, Command.PARITY -> {
209 val n = args[1].toInt()
210 ensure(n in 2..24, "bad parameter")
211 Code(n, if (command == Command.REPETITION) listOf(mask(n)) else (0 until n - 1).map { bit(n, it) or 1 })
212 }
213 Command.RM -> {
214 val r = args[1].toInt(); val m = args[2].toInt()
215 ensure(m in 1..4 && r <= m, "bad parameter")
216 Code(1 shl m, reedMuller(r, m))
217 }
218 Command.CYCLIC -> {
219 val n = args[1].toInt(); val d = args[2].length - 1
220 ensure(n in 2..24 && args[2][0] == '1' && d in 1 until n, "bad parameter")
221 val divisor = args[2].toInt(2)
222 ensure(divide((1 shl n) or 1, divisor).remainder == 0, "not a divisor")
223 val k = n - d
224 val rows = (0 until k).map { divisor shl (k - it - 1) }
225 val checks = IntArray(d)
226 for (j in 0 until n) {
227 val remainder = divide(bit(n, j), divisor).remainder
228 for (i in 0 until d) checks[i] = (checks[i] shl 1) or ((remainder shr (d - i - 1)) and 1)
229 }
230 Code(n, rows, checks.toList(), divisor)
231 }
232 Command.DUAL -> Code(a!!.n, a.h, a.g)
233 Command.EXTEND -> Code(a!!.n + 1, a.g.map { (it shl 1) or (weight(it) and 1) })
234 Command.PUNCTURE, Command.SHORTEN, Command.EXPURGATE -> {
235 val source = a!!
236 val expurgating = command == Command.EXPURGATE
237 val position = if (expurgating) 0 else args[2].toInt()
238 if (!expurgating) ensure(position in 1..source.n, "bad position")
239 val n = source.n - if (expurgating) 0 else 1
240 fun selected(row: Int) = if (expurgating) (weight(row) and 1) != 0 else (row and bit(source.n, position - 1)) != 0
241 val removed = if (command == Command.PUNCTURE) -1 else source.g.indexOfFirst(::selected)
242 val rows = mutableListOf<Int>()
243 for (i in source.g.indices) {
244 if (i == removed) continue
245 var row = source.g[i]
246 if (removed >= 0 && selected(row)) row = row xor source.g[removed]
247 if (!expurgating) {
248 val low = source.n - position
249 row = ((row shr (low + 1)) shl low) or (row and mask(low))
250 }
251 rows.add(row)
252 }
253 Code(n, if (command == Command.PUNCTURE) basis(rows, n) else rows)
254 }
255 Command.AUGMENT -> Code(a!!.n, basis(a.g + mask(a.n), a.n))
256 Command.PERMUTE -> {
257 val source = a!!
258 val positions = args.drop(2).map(String::toInt)
259 ensure(positions.size == source.n && positions.sorted() == (1..source.n).toList(), "bad permutation")
260 Code(source.n, source.g.map { original ->
261 var row = 0
262 for (position in positions) row = (row shl 1) or if ((original and bit(source.n, position - 1)) != 0) 1 else 0
263 row
264 })
265 }
266 Command.DIRECTSUM, Command.UUV, Command.PRODUCT -> {
267 val left = a!!; val right = b!!
268 if (command == Command.UUV) ensure(left.n == right.n, "bad length")
269 val product = command == Command.PRODUCT
270 val n = if (product) left.n * right.n else left.n + right.n
271 checkDimension(n, if (product) left.k * right.k else left.k + right.k)
272 val rows = if (product) left.g.flatMap { x -> right.g.map { y ->
273 var row = 0
274 for (i in 0 until left.n) row = (row shl right.n) or if ((x and bit(left.n, i)) != 0) y else 0
275 row
276 } } else left.g.map { (it shl right.n) or if (command == Command.UUV) it else 0 } + right.g
277 Code(n, rows)
278 }
279 else -> error("Not a defining command")
280}

288fun weightList(counts: IntArray) = counts.indices.filter { counts[it] != 0 }.joinToString(" ") { "$it:${counts[it]}" }
289fun gcd(first: Long, second: Long): Long {
290 var a = abs(first); var b = second
291 while (b != 0L) { val remainder = a % b; a = b; b = remainder }
292 return a
293}

295val combinations = Array(25) { LongArray(25) }.also { table ->
296 for (n in 0..24) {
297 table[n][0] = 1; table[n][n] = 1
298 for (k in 1 until n) table[n][k] = table[n - 1][k - 1] + table[n - 1][k]
299 }

301fun choose(n: Int, k: Int) = if (k < 0 || k > n) 0L else combinations[n][k]
302fun volume(n: Int, radius: Int) = (0..radius).sumOf { choose(n, it) }
303fun krawtchouk(n: Int, j: Int, x: Int): Long = (0..j).sumOf { y ->
304 val term = choose(x, y) * choose(n - x, j - y)
305 if (y % 2 == 0) term else -term
306}

311fun systematic(name: String, code: Code): String {
312 val rows = code.g.toMutableList()
313 val permutation = IntArray(code.n) { it + 1 }
314 for (r in 0 until code.k) {
315 var column = r
316 var found: Int
317 while (true) {
318 found = (r until rows.size).firstOrNull { (rows[it] and bit(code.n, column)) != 0 } ?: rows.size
319 if (found < rows.size) break
320 ++column
321 }
322 if (column != r) {
323 val left = bit(code.n, r); val right = bit(code.n, column)
324 for (i in rows.indices)
325 if (((rows[i] and left) != 0) != ((rows[i] and right) != 0)) rows[i] = rows[i] xor left xor right
326 val temporary = permutation[r]; permutation[r] = permutation[column]; permutation[column] = temporary
327 }
328 val temporary = rows[r]; rows[r] = rows[found]; rows[found] = temporary
329 for (i in rows.indices)
330 if (i != r && (rows[i] and bit(code.n, r)) != 0) rows[i] = rows[i] xor rows[r]
331 }
332 return buildString {
333 line("systematic $name perm ${permutation.joinToString(" ")}")
334 matrixText("G", rows, code.n)
335 for (j in code.h.indices) {
336 var row = 0
337 for (value in rows) row = (row shl 1) or if ((value and bit(code.n, code.k + j)) != 0) 1 else 0
338 line("H ${bits((row shl code.h.size) or bit(code.h.size, j), code.n)}")
339 }
340 }
341}

343fun independent(command: Command, args: List<String>): String {
344 if (command == Command.CRC || command == Command.CRCCHECK) {
345 val d = args[0].length - 1
346 ensure(args[0][0] == '1' && d in 1..16, "bad parameter")
347 if (command == Command.CRC) ensure(args[1].length + d <= 30, "bad length")
348 val value = args[1].toInt(2) shl if (command == Command.CRC) d else 0
349 val remainder = divide(value, args[0].toInt(2)).remainder
350 val suffix = bits(remainder, d)
351 return if (command == Command.CRC) "crc remainder $suffix frame ${args[1]}$suffix\n"
352 else "crccheck ${if (remainder == 0) "ok" else "error"} remainder $suffix\n"
353 }
354 val n = args[0].toInt()
355 if (command == Command.BOUNDS) {
356 val d = args[1].toInt()
357 ensure(d >= 1 && d <= n && n <= 24, "bad parameter")
358 val ball = volume(n, (d - 1) / 2)
359 val hamming = (0..n).last { (1L shl it) * ball <= (1L shl n) }
360 var sum = 0L; var griesmer = 0
361 for (k in 1..n) {
362 val denominator = 1L shl (k - 1)
363 sum += (d + denominator - 1) / denominator
364 if (sum <= n) griesmer = k
365 }
366 return "bounds n $n d $d singleton ${n - d + 1} hamming $hamming griesmer $griesmer\n"
367 }
368 ensure(n in 1..24, "bad parameter")
369 if (command == Command.FACTOR) return buildString {
370 var remaining = (1 shl n) or 1
371 var divisors = 1; var factor = 3
372 append("factor $n")
373 while (2 * (length(factor) - 1) <= length(remaining) - 1) {
374 var exponent = 0
375 while (true) {
376 val result = divide(remaining, factor)
377 if (result.remainder != 0) break
378 remaining = result.quotient; ++exponent
379 }
380 if (exponent != 0) { append(" ${bits(factor, length(factor))}^$exponent"); divisors *= exponent + 1 }
381 factor += 2
382 }
383 if (remaining != 1) { append(" ${bits(remaining, length(remaining))}^1"); divisors *= 2 }
384 line(" divisors $divisors")
385 }
386 ensure(args.size == n + 2, "bad length")
387 ensure(args[1].toInt() == 1, "bad parameter")
388 val coefficients = args.drop(1).map(String::toLong)
389 val size = coefficients.sum()
390 var feasible = (size and (size - 1)) == 0L
391 return buildString {
392 line("macwilliams n $n size $size")
393 append("dual")
394 for (j in 0..n) {
395 var numerator = 0L
396 for (x in 0..n) numerator += coefficients[x] * krawtchouk(n, j, x)
397 if (numerator < 0 || numerator % size != 0L) feasible = false
398 val divisor = gcd(numerator, size)
399 val denominator = size / divisor
400 append(' ').append(numerator / divisor)
401 if (denominator != 1L) append('/').append(denominator)
402 }
403 append('\n'); line("feasible ${yes(feasible)}")
404 }
405}

407class Laboratory {
408 private val codes = TreeMap<String, Code>()
409 private val commands = Command.entries.associateBy { it.name }
410
411 fun process(input: String): String {
412 val words = input.split(' ')
413 val command = commands[words[0]] ?: throw CommandError("unknown command ${words[0]}")
414 val args = words.drop(1)
415 ensure(args.size in command.minimum..command.maximum, "bad arguments")
416 for (i in args.indices) ensure(validWord(args[i], command.types[minOf(i, command.types.lastIndex)]), "bad arguments")
417 var a: Code? = null; var b: Code? = null
418 for (i in command.types.indices) {
419 val kind = command.types[i]
420 if (kind != 'A' && kind != 'B') continue
421 val code = codes[args[i]] ?: throw CommandError("no code ${args[i]}")
422 if (kind == 'A') a = code else b = code
423 }
424 if (command.defining) {
425 val code = construct(command, args, a, b)
426 val status = if (codes.containsKey(args[0])) "redefined" else "defined"
427 codes[args[0]] = code
428 return "$status ${args[0]} n ${code.n} k ${code.k}\n"
429 }
430 if (command in listOf(Command.CRC, Command.CRCCHECK, Command.FACTOR, Command.MACWILLIAMS, Command.BOUNDS))
431 return independent(command, args)
432 if (command == Command.LIST) return buildString {
433 line("codes ${codes.size}")
434 for ((name, code) in codes) line("$name n ${code.n} k ${code.k} encoder ${code.encoder}")
435 }
436 val name = args[0]
437 if (command == Command.DELETE) { codes.remove(name); return "deleted $name\n" }
438 val code = a!!
439 val n = code.n; val k = code.k
440 return buildString {
441 when (command) {
442 Command.SHOW -> {
443 line("code $name n $n k $k encoder ${code.encoder}")
444 if (code.polynomial != null) line("poly g ${bits(code.polynomial, n - k + 1)} h ${bits(divide((1 shl n) or 1, code.polynomial).quotient, k + 1)}")
445 matrixText("G", code.g, n); matrixText("H", code.h, n)
446 }
447 Command.INFO -> {
448 val nonzero = span(code.g).drop(1)
449 val distance = nonzero.minOf(::weight)
450 val generator = nonzero.min()
451 val t = (distance - 1) / 2; val divisor = gcd(k.toLong(), n.toLong())
452 val cyclic = code.g.all { code.syndrome(((it shl 1) and mask(n)) or (it shr (n - 1))) == 0 }
453 line("info $name n $n k $k rate ${k / divisor}/${n / divisor} d $distance t $t " +
454 "perfect ${yes(volume(n, t) == (1L shl (n - k)))} mds ${yes(distance == n - k + 1)} " +
455 "cyclic ${if (cyclic) bits(generator, length(generator)) else "no"}")
456 }
457 Command.SYSTEMATIC -> append(systematic(name, code))
458 Command.WEIGHTS -> {
459 val primary = distribution(span(code.g), 0, n); val dual = distribution(span(code.h), 0, n)
460 val orthogonal = code.g.all { x -> code.g.all { y -> (weight(x and y) and 1) == 0 } }
461 line("weights $name ${weightList(primary)}"); line("dual $name ${weightList(dual)}")
462 line("duality $name self-orthogonal ${yes(orthogonal)} self-dual ${yes(orthogonal && n == 2 * k)} " +
463 "formally-self-dual ${yes(primary.contentEquals(dual))}")
464 }
465 Command.COVERING -> {
466 val counts = distribution(code.leaders().toList(), 0, n)
467 val radius = counts.indexOfLast { it != 0 }
468 append("covering $name radius $radius leaders")
469 for (i in 0..radius) append(" $i:${counts[i]}")
470 append('\n')
471 }
472 Command.BURST -> {
473 val detect = span(code.g).drop(1).minOf { length(it) - Integer.numberOfTrailingZeros(it) - 1 }
474 line("burst $name detect $detect correct ${code.burstCorrection()}")
475 }
476 Command.TRELLIS -> {
477 val profile = (0..n).map { cut ->
478 basis(code.g.map { it shr (n - cut) }, cut).size +
479 basis(code.g.map { it and mask(n - cut) }, n - cut).size - k
480 }
481 line("trellis $name profile ${profile.joinToString(" ")} max ${profile.max()}")
482 }
483 Command.STDARRAY -> {
484 ensure(n - k <= 6 && k <= 4, "too large")
485 val leaders = code.leaders()
486 val syndromes = leaders.indices.sortedWith(compareBy<Int> { weight(leaders[it]) }.thenBy { leaders[it] })
487 val codewords = span(code.g).sorted()
488 line("stdarray $name rows ${leaders.size} columns ${codewords.size}")
489 for (s in syndromes) {
490 append(bits(s, n - k)).append(':')
491 for (word in codewords) append(' ').append(bits(leaders[s] xor word, n))
492 append('\n')
493 }
494 }
495 Command.EQUAL -> {
496 val other = b!!
497 if (n != other.n) line("equal $name ${args[1]} no lengths $n ${other.n}")
498 else {
499 val common = k + other.k - basis(code.g + other.g, n).size
500 line("equal $name ${args[1]} ${yes(k == common && other.k == common)} intersection $common")
501 }
502 }
503 Command.ENCODE -> {
504 ensure(args[1].length == k, "bad length")
505 line("encode $name message ${args[1]} codeword ${bits(code.encode(args[1].toInt(2)), n)}")
506 }
507 Command.SYNDROME, Command.DECODE, Command.BOUNDED, Command.COSET -> {
508 val text = args[if (command == Command.BOUNDED) 2 else 1]
509 ensure(text.length == n, "bad length")
510 val word = text.toInt(2); val syndrome = code.syndrome(word)
511 if (command == Command.SYNDROME) line("syndrome $name ${bits(syndrome, n - k)} codeword ${yes(syndrome == 0)}")
512 else {
513 val decoded = code.decode(word, code.leaders())
514 if (command == Command.COSET) line("coset $name syndrome ${bits(syndrome, n - k)} " +
515 "leader ${bits(decoded.error, n)} weights ${weightList(distribution(span(code.g), word, n))}")
516 else if (command == Command.BOUNDED && weight(decoded.error) > args[1].toInt())
517 line("bounded $name fail weight ${weight(decoded.error)}")
518 else line("${if (command == Command.BOUNDED) "bounded" else "decode"} $name ${code.decodedText(decoded)}")
519 }
520 }
521 Command.ERASURE -> {
522 ensure(args[1].length == n, "bad length")
523 var known = 0; var value = 0; var erased = 0
524 for (i in 0 until n) {
525 val c = args[1][i]
526 if (c == '?') ++erased else { known = known or bit(n, i); if (c == '1') value = value or bit(n, i) }
527 }
528 val candidates = span(code.g).filter { (it and known) == value }
529 append("erasure $name erased $erased candidates ${candidates.size}")
530 if (candidates.size == 1) append(" codeword ${bits(candidates[0], n)} message ${bits(code.messageFor(candidates[0]), k)}")
531 append('\n')
532 }
533 Command.INTERLEAVE -> {
534 val depth = args.size - 3; val start = args[1].toInt(); val size = args[2].toInt()
535 ensure(args.drop(3).all { it.length == k }, "bad length")
536 ensure(start >= 1 && size >= 1 && start + size - 1 <= depth * n, "bad burst")
537 val sentRows = IntArray(depth) { code.encode(args[it + 3].toInt(2)) }
538 val receivedRows = IntArray(depth)
539 val sent = StringBuilder(); val received = StringBuilder()
540 for (column in 0 until n) for (i in 0 until depth) {
541 val position = column * depth + i + 1
542 var value = (sentRows[i] and bit(n, column)) != 0
543 sent.append(if (value) '1' else '0')
544 if (position >= start && position < start + size) value = !value
545 received.append(if (value) '1' else '0')
546 receivedRows[i] = (receivedRows[i] shl 1) or if (value) 1 else 0
547 }
548 line("interleave $name depth $depth burst $start $size"); line("sent $sent"); line("recv $received")
549 val leaders = code.leaders()
550 var correct = 0
551 for (i in 0 until depth) {
552 val decoded = code.decode(receivedRows[i], leaders)
553 val ok = decoded.message == args[i + 3].toInt(2)
554 if (ok) ++correct
555 line("row ${i + 1} errors ${weight(receivedRows[i] xor sentRows[i])} corrected ${weight(decoded.error)} " +
556 "message ${bits(decoded.message, k)} ${if (ok) "ok" else "wrong"}")
557 }
558 line("result ok $correct wrong ${depth - correct}")
559 }
560 else -> error("Unhandled command")
561 }
562 }
563 }
564}

568 val output = System.out.bufferedWriter()
569 val laboratory = Laboratory()
570 repeat(input.readLine().toInt()) {
571 try { output.write(laboratory.process(input.readLine())) }
572 catch (error: CommandError) { output.write("error: ${error.message}\n") }
573 }
574 output.flush()
575}
