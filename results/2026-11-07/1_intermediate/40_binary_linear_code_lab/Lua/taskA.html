47local function divide(dividend, divisor)
48 local quotient, d = 0, degree(divisor)
49 while dividend ~= 0 and degree(dividend) >= d do
50 local shift = degree(dividend) - d
51 quotient = quotient ~ (1 << shift)
52 dividend = dividend ~ (divisor << shift)
53 end
54 return quotient, dividend
55end
56local function remainder(dividend, divisor)
57 local _, result = divide(dividend, divisor)

60local function basis(rows, length)
61 local pivots, kept = zeros(length), {}
62 for _, original in ipairs(rows) do
63 local row = original
64 for column = 1, length do
65 if bit(row, length, column) ~= 0 then
66 if pivots[column] ~= 0 then
67 row = row ~ pivots[column]
68 else
69 pivots[column] = row
70 kept[#kept + 1] = original
71 break
72 end
73 end
74 end
75 end
76 return kept
77end
78local function rref(original, length)
79 local rows, columns = copy(original), {}
80 for column = 1, length do
81 local pivot = #columns + 1
82 local found = pivot
83 while found <= #rows and bit(rows[found], length, column) == 0 do
84 found = found + 1
85 end
86 if found <= #rows then
87 rows[pivot], rows[found] = rows[found], rows[pivot]
88 for i, row in ipairs(rows) do
89 if i ~= pivot and bit(row, length, column) ~= 0 then
90 rows[i] = row ~ rows[pivot]
91 end
92 end
93 columns[#columns + 1] = column
94 end
95 end
96 local reduced = {}
97 for i = 1, #columns do
98 reduced[i] = rows[i]
99 end
100 return reduced, columns
101end
102local function nullspace(original, length)
103 local rows, columns = rref(original, length)
104 local pivot, result = {}, {}
105 for _, column in ipairs(columns) do
106 pivot[column] = true
107 end
108 for column = 1, length do
109 if not pivot[column] then
110 local word = 1 << (length - column)
111 for i, row in ipairs(rows) do
112 if bit(row, length, column) ~= 0 then
113 word = word | (1 << (length - columns[i]))
114 end
115 end
116 result[#result + 1] = word
117 end
118 end
119 return result
120end

157local function reedMuller(order, variables)
158 local length = 1 << variables
159 if order == 0 then
160 return {(1 << length) - 1}
161 end
162 local result = {}
163 for _, row in ipairs(reedMuller(math.min(order, variables - 1), variables - 1)) do
164 result[#result + 1] = (row << (length // 2)) | row
165 end
166 for _, row in ipairs(reedMuller(order - 1, variables - 1)) do
167 result[#result + 1] = row
168 end
169 return result
170end

196 local result = 0

205function Code.new(n, g, h, polynomial)
206 checkDimension(n, #g)
207 local self =
208 setmetatable(
209 {
210 n = n,
211 k = #g,
212 g = g,
213 h = h or nullspace(g, n),
214 polynomial = polynomial,
215 words = {},
216 messages = {}
217 },
218 Code
219 )
220 for message = 0, (1 << self.k) - 1 do
221 local word = self:encode(message)
222 self.words[#self.words + 1] = word
223 self.messages[word] = message
224 end
225 table.sort(self.words)
226 return self
227end
228function Code:encoder()
229 return self.polynomial and "polynomial" or "matrix"
230end
231function Code:encode(message)
232 if self.polynomial then
233 local shifted = message << (self.n - self.k)
234 return shifted | remainder(shifted, self.polynomial)
235 end
236 local result = 0
237 for row, word in ipairs(self.g) do
238 if bit(message, self.k, row) ~= 0 then
239 result = result ~ word
240 end
241 end
242 return result
243end
244function Code:syndrome(word)
245 local result = 0
246 for _, row in ipairs(self.h) do
247 result = (result << 1) | (weight(word & row) % 2)
248 end
249 return result
250end
251function Code:leaders()
252 if self.cachedLeaders then
253 return self.cachedLeaders
254 end
255 local result = zeros(1 << (self.n - self.k))
256 for i = 2, #result do
257 result[i] = -1
258 end
259 local remaining, count = #result - 1, 1
260 while remaining > 0 do
261 local word = (1 << count) - 1
262 while word < (1 << self.n) do
263 local index = self:syndrome(word) + 1
264 if result[index] == -1 then
265 result[index] = word
266 remaining = remaining - 1
267 if remaining == 0 then
268 break
269 end
270 end
271 local lowest = word & -word
272 local nextWord = word + lowest
273 word = nextWord | (((nextWord ~ word) >> 2) // lowest)
274 end
275 count = count + 1
276 end
277 self.cachedLeaders = result
278 return result
279end
280function Code:decode(received)
281 local errorWord = self:leaders()[self:syndrome(received) + 1]
282 local word = received ~ errorWord
283 return {error = errorWord, word = word, message = self.messages[word]}
284end
285function Code:decodedText(result)
286 return "codeword " ..
287 bits(result.word, self.n) ..
288 " message " ..
289 bits(result.message, self.k) ..
290 " error " .. bits(result.error, self.n) .. " weight " .. weight(result.error)
291end

340local function valid(word, kind)
341 if kind == Name or kind == Existing then
342 return #word <= 8 and word:match("^[A-Za-z][A-Za-z0-9]*$") ~= nil
343 end
344 if kind == Number then
345 return #word <= 4 and word:match("^[0-9]+$") ~= nil and
346 (#word == 1 or word:sub(1, 1) ~= "0")
347 end
348 return #word <= 30 and word:match(kind == Bits and "^[01]+$" or "^[01?]+$") ~= nil
349end
350local codes = {}
351local function parse(line)
352 local words = {}
353 for word in line:gmatch("[^ ]+") do
354 words[#words + 1] = word
355 end
356 local verb = table.remove(words, 1)
357 local form = formats[verb]
358 if not form then
359 fail("unknown command " .. verb)
360 end
361 local extra = #words - #form.fixed
362 if form.repeated then
363 if extra < 1 or extra > form.maximum then
364 fail("bad arguments")
365 end
366 elseif extra ~= 0 then
367 fail("bad arguments")
368 end
369 for i, word in ipairs(words) do
370 if not valid(word, form.fixed[i] or form.repeated) then
371 fail("bad arguments")
372 end
373 end
374 for i, kind in ipairs(form.fixed) do
375 if kind == Existing and not codes[words[i]] then
376 fail("no code " .. words[i])
377 end
378 end
379 return verb, words, form
380end
381local function define(verb, args)
382 local name, n, g, h, polynomial = args[1], 0, {}, nil, nil
383 if verb == "GEN" or verb == "PCHECK" then
384 n = #args[2]
385 for i = 2, #args do
386 if #args[i] ~= n then
387 fail("bad length")
388 end
389 g[#g + 1] = tonumber(args[i], 2)
390 end
391 g = basis(g, n)
392 if verb == "PCHECK" then
393 h = g
394 g = nullspace(h, n)
395 end
396 elseif verb == "HAMMING" or verb == "SIMPLEX" then
397 local r = tonumber(args[2], 10)
398 if r < 2 or r > 4 then
399 fail("bad parameter")
400 end
401 n = (1 << r) - 1
402 g = zeros(r)
403 for column = 1, n do
404 for row = 1, r do
405 g[row] = (g[row] << 1) | bit(column, r, row)
406 end
407 end
408 if verb == "HAMMING" then
409 h = g
410 g = nullspace(h, n)
411 end
412 elseif verb == "REPETITION" or verb == "PARITY" then
413 n = tonumber(args[2], 10)
414 if n < 2 or n > 24 then
415 fail("bad parameter")
416 end
417 if verb == "REPETITION" then
418 g = {(1 << n) - 1}
419 else
420 for row = 1, n - 1 do
421 g[row] = (1 << (n - row)) | 1
422 end
423 end
424 elseif verb == "RM" then
425 local order, variables = tonumber(args[2], 10), tonumber(args[3], 10)
426 if variables < 1 or variables > 4 or order > variables then
427 fail("bad parameter")
428 end
429 n = 1 << variables
430 g = reedMuller(order, variables)
431 elseif verb == "CYCLIC" then
432 local text = args[3]
433 n = tonumber(args[2], 10)
434 if n < 2 or n > 24 or text:sub(1, 1) == "0" or #text < 2 or #text > n then
435 fail("bad parameter")
436 end
437 polynomial = tonumber(text, 2)
438 if remainder((1 << n) | 1, polynomial) ~= 0 then
439 fail("not a divisor")
440 end
441 local d = #text - 1
442 local k = n - d
443 for row = 1, k do
444 g[row] = polynomial << (k - row)
445 end
446 h = zeros(d)
447 for column = 1, n do
448 local rem = remainder(1 << (n - column), polynomial)
449 for row = 1, d do
450 h[row] = (h[row] << 1) | bit(rem, d, row)
451 end
452 end
453 elseif verb == "DIRECTSUM" or verb == "UUV" or verb == "PRODUCT" then
454 local a, b = codes[args[2]], codes[args[3]]
455 if verb == "UUV" and a.n ~= b.n then
456 fail("bad length")
457 end
458 n = verb == "PRODUCT" and a.n * b.n or a.n + b.n
459 checkDimension(n, verb == "PRODUCT" and a.k * b.k or a.k + b.k)
460 if verb == "PRODUCT" then
461 for _, x in ipairs(a.g) do
462 for _, y in ipairs(b.g) do
463 local word = 0
464 for column = 1, a.n do
465 word = (word << b.n) | (bit(x, a.n, column) ~= 0 and y or 0)
466 end
467 g[#g + 1] = word
468 end
469 end
470 else
471 for _, row in ipairs(a.g) do
472 g[#g + 1] = (row << b.n) | (verb == "UUV" and row or 0)
473 end
474 for _, row in ipairs(b.g) do
475 g[#g + 1] = row
476 end
477 end
478 else
479 local source = codes[args[2]]
480 n = source.n
481 if verb == "DUAL" then
482 g, h = source.h, source.g
483 elseif verb == "EXTEND" then
484 n = n + 1
485 for i, row in ipairs(source.g) do
486 g[i] = (row << 1) | (weight(row) % 2)
487 end
488 elseif verb == "PUNCTURE" or verb == "SHORTEN" then
489 local column = tonumber(args[3], 10)
490 if column < 1 or column > n then
491 fail("bad position")
492 end
493 local rows = copy(source.g)
494 if verb == "SHORTEN" then
495 for pivot, z in ipairs(rows) do
496 if bit(z, n, column) ~= 0 then
497 table.remove(rows, pivot)
498 for i, row in ipairs(rows) do
499 if bit(row, n, column) ~= 0 then
500 rows[i] = row ~ z
501 end
502 end
503 break
504 end
505 end
506 end
507 for i, row in ipairs(rows) do
508 g[i] = removeBit(row, n, column)
509 end
510 n = n - 1
511 if verb == "PUNCTURE" then
512 g = basis(g, n)
513 end
514 elseif verb == "AUGMENT" then
515 g = copy(source.g)
516 g[#g + 1] = (1 << n) - 1
517 g = basis(g, n)
518 elseif verb == "EXPURGATE" then
519 g = copy(source.g)
520 for pivot, z in ipairs(g) do
521 if weight(z) % 2 ~= 0 then
522 table.remove(g, pivot)
523 for i, row in ipairs(g) do
524 if weight(row) % 2 ~= 0 then
525 g[i] = row ~ z
526 end
527 end
528 break
529 end
530 end
531 elseif verb == "PERMUTE" then
532 if #args - 2 ~= n then
533 fail("bad permutation")
534 end
535 local seen, permutation = {}, {}
536 for i = 3, #args do
537 local column = tonumber(args[i], 10)
538 if column < 1 or column > n or seen[column] then
539 fail("bad permutation")
540 end
541 seen[column] = true
542 permutation[#permutation + 1] = column
543 end
544 for i, row in ipairs(source.g) do
545 local word = 0
546 for _, column in ipairs(permutation) do
547 word = (word << 1) | bit(row, n, column)
548 end
549 g[i] = word
550 end
551 end
552 end
553 local code = Code.new(n, g, h, polynomial)
554 local action = codes[name] and "redefined" or "defined"
555 codes[name] = code
556 return {action .. " " .. name .. " n " .. n .. " k " .. code.k}
557end
558local function systematic(name, code)
559 local n, k, rows, permutation = code.n, code.k, copy(code.g), {}
560 for i = 1, n do
561 permutation[i] = i
562 end
563 for pivot = 1, k do
564 local column, found = pivot, pivot
565 while true do
566 found = pivot
567 while found <= k and bit(rows[found], n, column) == 0 do
568 found = found + 1
569 end
570 if found <= k then
571 break
572 end
573 column = column + 1
574 end
575 if column ~= pivot then
576 local mask = (1 << (n - pivot)) | (1 << (n - column))
577 for i, row in ipairs(rows) do
578 if bit(row, n, pivot) ~= bit(row, n, column) then
579 rows[i] = row ~ mask
580 end
581 end
582 permutation[pivot], permutation[column] = permutation[column], permutation[pivot]
583 end
584 rows[pivot], rows[found] = rows[found], rows[pivot]
585 for i, row in ipairs(rows) do
586 if i ~= pivot and bit(row, n, pivot) ~= 0 then
587 rows[i] = row ~ rows[pivot]
588 end
589 end
590 end
591 local output = {"systematic " .. name .. " perm " .. table.concat(permutation, " ")}
592 for _, row in ipairs(rows) do
593 output[#output + 1] = "G " .. bits(row, n)
594 end
595 for column = k + 1, n do
596 local word = 0
597 for _, row in ipairs(rows) do
598 word = (word << 1) | bit(row, n, column)
599 end
600 word = (word << (n - k)) | (1 << (n - column))
601 output[#output + 1] = "H " .. bits(word, n)
602 end
603 return output
604end
605local function burst(name, code)
606 local detect = code.n
607 for _, word in ipairs(code.words) do
608 if word ~= 0 then
609 detect = math.min(detect, degree(word) - degree(word & -word))
610 end
611 end
612 local seen = {[0] = true}
613 for length = 1, code.n do
614 for start = 0, code.n - length do
615 local limit = length == 1 and 1 or (1 << (length - 2))
616 for middle = 0, limit - 1 do
617 local pattern = length == 1 and 1 or ((1 << (length - 1)) | (middle << 1) | 1)
618 local syndrome = code:syndrome(pattern << (code.n - start - length))
619 if seen[syndrome] then
620 return {"burst " .. name .. " detect " .. detect .. " correct " .. (length - 1)}
621 end
622 seen[syndrome] = true
623 end
624 end
625 end
626 error("A nonzero codeword has zero syndrome")
627end
628local function interleave(name, code, args)
629 local start, length, depth = tonumber(args[2], 10), tonumber(args[3], 10), #args - 3
630 for i = 4, #args do
631 if #args[i] ~= code.k then
632 fail("bad length")
633 end
634 end
635 if start < 1 or length < 1 or start + length - 1 > depth * code.n then
636 fail("bad burst")
637 end
638 local sentWords, receivedWords, sent, received = {}, zeros(depth), {}, {}
639 for i = 1, depth do
640 sentWords[i] = code:encode(tonumber(args[i + 3], 2))
641 end
642 for column = 1, code.n do
643 for row = 1, depth do
644 local position = (column - 1) * depth + row
645 local original = bit(sentWords[row], code.n, column)
646 local changed =
647 original ~ ((position >= start and position < start + length) and 1 or 0)
648 sent[position], received[position] = original, changed
649 receivedWords[row] = (receivedWords[row] << 1) | changed
650 end
651 end
652 local output = {
653 "interleave " .. name .. " depth " .. depth .. " burst " .. start .. " " .. length,
654 "sent " .. table.concat(sent),
655 "recv " .. table.concat(received)
656 }
657 local correct = 0
658 for row, receivedWord in ipairs(receivedWords) do
659 local decoded = code:decode(receivedWord)
660 local message = bits(decoded.message, code.k)
661 local ok = message == args[row + 3]
662 if ok then
663 correct = correct + 1
664 end
665 output[#output + 1] =
666 "row " ..
667 row ..
668 " errors " ..
669 weight(receivedWord ~ sentWords[row]) ..
670 " corrected " ..
671 weight(decoded.error) ..
672 " message " .. message .. " " .. (ok and "ok" or "wrong")
673 end
674 output[#output + 1] = "result ok " .. correct .. " wrong " .. (depth - correct)
675 return output
676end
677local function independent(verb, args)
678 if verb == "CRC" or verb == "CRCCHECK" then
679 local generator, text = args[1], args[2]
680 if generator:sub(1, 1) == "0" or #generator < 2 or #generator > 17 then
681 fail("bad parameter")
682 end
683 local d = #generator - 1
684 if verb == "CRC" and #text + d > 30 then
685 fail("bad length")
686 end
687 local word = tonumber(text, 2)
688 local rem = remainder(verb == "CRC" and (word << d) or word, tonumber(generator, 2))
689 local suffix = bits(rem, d)
690 if verb == "CRC" then
691 return {"crc remainder " .. suffix .. " frame " .. text .. suffix}
692 end
693 return {"crccheck " .. (rem == 0 and "ok" or "error") .. " remainder " .. suffix}
694 end
695 local n = tonumber(args[1], 10)
696 if verb == "BOUNDS" then
697 local distance = tonumber(args[2], 10)
698 if distance < 1 or distance > n or n > 24 then
699 fail("bad parameter")
700 end
701 local v = volume(n, (distance - 1) // 2)
702 local hamming, griesmer, sum = 0, 0, 0
703 for k = 0, n do
704 if (1 << k) * v <= (1 << n) then
705 hamming = k
706 end
707 end
708 for k = 1, n do
709 local denominator = 1 << (k - 1)
710 sum = sum + (distance + denominator - 1) // denominator
711 if sum <= n then
712 griesmer = k
713 end
714 end
715 return {
716 "bounds n " ..
717 n ..
718 " d " ..
719 distance ..
720 " singleton " ..
721 (n - distance + 1) ..
722 " hamming " .. hamming .. " griesmer " .. griesmer
723 }
724 end
725 if n < 1 or n > 24 then
726 fail("bad parameter")
727 end
728 if verb == "FACTOR" then
729 local remaining, candidate, divisors, factors = (1 << n) | 1, 2, 1, {}
730 while 2 * degree(candidate) <= degree(remaining) do
731 local exponent = 0
732 while true do
733 local quotient, rem = divide(remaining, candidate)
734 if rem ~= 0 then
735 break
736 end
737 remaining = quotient
738 exponent = exponent + 1
739 end
740 if exponent > 0 then
741 factors[#factors + 1] = bits(candidate, degree(candidate) + 1) .. "^" .. exponent
742 divisors = divisors * (exponent + 1)
743 end
744 candidate = candidate + 1
745 end
746 if remaining > 1 then
747 factors[#factors + 1] = bits(remaining, degree(remaining) + 1) .. "^1"
748 divisors = divisors * 2
749 end
750 return {"factor " .. n .. " " .. table.concat(factors, " ") .. " divisors " .. divisors}
751 end
752 if #args ~= n + 2 then
753 fail("bad length")
754 end
755 if tonumber(args[2], 10) ~= 1 then
756 fail("bad parameter")
757 end
758 local counts, size = {}, 0
759 for i = 2, #args do
760 counts[#counts + 1] = tonumber(args[i], 10)
761 size = size + counts[#counts]
762 end
763 local feasible, transformed = (size & (size - 1)) == 0, {}
764 for j = 0, n do
765 local numerator = 0
766 for x = 0, n do
767 numerator = numerator + counts[x + 1] * krawtchouk(n, j, x)
768 end
769 local divisor = gcd(numerator, size)
770 local p, q = numerator // divisor, size // divisor
771 transformed[#transformed + 1] = q == 1 and tostring(p) or (p .. "/" .. q)
772 if q ~= 1 or p < 0 then
773 feasible = false
774 end
775 end
776 return {
777 "macwilliams n " .. n .. " size " .. size,
778 "dual " .. table.concat(transformed, " "),
779 "feasible " .. yes(feasible)
780 }
781end
782local function process(line)
783 local verb, args, form = parse(line)
784 if form.defining then
785 return define(verb, args)
786 end
787 if
788 verb == "CRC" or verb == "CRCCHECK" or verb == "FACTOR" or verb == "MACWILLIAMS" or
789 verb == "BOUNDS"
790 then
791 return independent(verb, args)
792 end
793 if verb == "LIST" then
794 local names = {}
795 for name in pairs(codes) do
796 names[#names + 1] = name
797 end
798 table.sort(names)
799 local output = {"codes " .. #names}
800 for _, name in ipairs(names) do
801 local code = codes[name]
802 output[#output + 1] =
803 name .. " n " .. code.n .. " k " .. code.k .. " encoder " .. code:encoder()
804 end
805 return output
806 end
807 local name = args[1]
808 if verb == "DELETE" then
809 codes[name] = nil
810 return {"deleted " .. name}
811 end
812 local code = codes[name]
813 local n, k = code.n, code.k
814 if verb == "SHOW" then
815 local output = {
816 "code " .. name .. " n " .. n .. " k " .. k .. " encoder " .. code:encoder()
817 }
818 if code.polynomial then
819 local quotient = divide((1 << n) | 1, code.polynomial)
820 output[#output + 1] =
821 "poly g " .. bits(code.polynomial, n - k + 1) .. " h " .. bits(quotient, k + 1)
822 end
823 for _, row in ipairs(code.g) do
824 output[#output + 1] = "G " .. bits(row, n)
825 end
826 for _, row in ipairs(code.h) do
827 output[#output + 1] = "H " .. bits(row, n)
828 end
829 return output
830 elseif verb == "INFO" then
831 local distance, cyclic = n, true
832 for _, word in ipairs(code.words) do
833 if word ~= 0 then
834 distance = math.min(distance, weight(word))
835 end
836 end
837 for _, row in ipairs(code.g) do
838 local rotated = ((row << 1) & ((1 << n) - 1)) | (row >> (n - 1))
839 if code.messages[rotated] == nil then
840 cyclic = false
841 end
842 end
843 local t, divisor = (distance - 1) // 2, gcd(n, k)
844 local generator = cyclic and bits(code.words[2], degree(code.words[2]) + 1) or "no"
845 return {
846 "info " ..
847 name ..
848 " n " ..
849 n ..
850 " k " ..
851 k ..
852 " rate " ..
853 (k // divisor) ..
854 "/" ..
855 (n // divisor) ..
856 " d " ..
857 distance ..
858 " t " ..
859 t ..
860 " perfect " ..
861 yes(
862 volume(n, t) ==
863 (1 << (n - k))
864 ) ..
865 " mds " ..
866 yes(
867 distance ==
868 n - k + 1
869 ) ..
870 " cyclic " ..
871 generator
872 }
873 elseif verb == "SYSTEMATIC" then
874 return systematic(name, code)
875 elseif verb == "WEIGHTS" then
876 local a, b, orthogonal, equal =
877 distribution(code.words, n),
878 distribution(span(code.h), n),
879 true,
880 true
881 for _, x in ipairs(code.g) do
882 for _, y in ipairs(code.g) do
883 if weight(x & y) % 2 ~= 0 then
884 orthogonal = false
885 end
886 end
887 end
888 for i = 1, n + 1 do
889 if a[i] ~= b[i] then
890 equal = false
891 end
892 end
893 return {
894 "weights " .. name .. " " .. weightList(a),
895 "dual " .. name .. " " .. weightList(b),
896 "duality " ..
897 name ..
898 " self-orthogonal " ..
899 yes(orthogonal) ..
900 " self-dual " ..
901 yes(orthogonal and n == 2 * k) ..
902 " formally-self-dual " .. yes(equal)
903 }
904 elseif verb == "COVERING" then
905 local counts, radius, items = distribution(code:leaders(), n), n, {}
906 while counts[radius + 1] == 0 do
907 radius = radius - 1
908 end
909 for i = 0, radius do
910 items[#items + 1] = i .. ":" .. counts[i + 1]
911 end
912 return {
913 "covering " .. name .. " radius " .. radius .. " leaders " .. table.concat(items, " ")
914 }
915 elseif verb == "BURST" then
916 return burst(name, code)
917 elseif verb == "TRELLIS" then
918 local profile, maximum = {}, 0
919 for cut = 0, n do
920 local right, prefix, suffix = n - cut, {}, {}
921 for i, row in ipairs(code.g) do
922 prefix[i] = row >> right
923 suffix[i] = row & ((1 << right) - 1)
924 end
925 local value = #basis(prefix, cut) + #basis(suffix, right) - k
926 profile[#profile + 1] = value
927 maximum = math.max(maximum, value)
928 end
929 return {
930 "trellis " .. name .. " profile " .. table.concat(profile, " ") .. " max " .. maximum
931 }
932 elseif verb == "STDARRAY" then
933 if n - k > 6 or k > 4 then
934 fail("too large")
935 end
936 local leaders, syndromes = code:leaders(), {}
937 for i = 1, #leaders do
938 syndromes[i] = i
939 end
940 table.sort(
941 syndromes,
942 function(a, b)
943 local wa, wb = weight(leaders[a]), weight(leaders[b])
944 if wa ~= wb then
945 return wa < wb
946 end
947 return leaders[a] < leaders[b]
948 end
949 )
950 local output = {"stdarray " .. name .. " rows " .. #leaders .. " columns " .. #code.words}
951 for _, index in ipairs(syndromes) do
952 local row = {}
953 for _, word in ipairs(code.words) do
954 row[#row + 1] = bits(word ~ leaders[index], n)
955 end
956 output[#output + 1] = bits(index - 1, n - k) .. ": " .. table.concat(row, " ")
957 end
958 return output
959 elseif verb == "EQUAL" then
960 local other = codes[args[2]]
961 if n ~= other.n then
962 return {"equal " .. name .. " " .. args[2] .. " no lengths " .. n .. " " .. other.n}
963 end
964 local rows = copy(code.g)
965 for _, row in ipairs(other.g) do
966 rows[#rows + 1] = row
967 end
968 local common = k + other.k - #basis(rows, n)
969 return {
970 "equal " ..
971 name ..
972 " " ..
973 args[2] ..
974 " " ..
975 yes(k == common and other.k == common) .. " intersection " .. common
976 }
977 elseif verb == "ENCODE" then
978 if #args[2] ~= k then
979 fail("bad length")
980 end
981 return {
982 "encode " ..
983 name ..
984 " message " ..
985 args[2] .. " codeword " .. bits(code:encode(tonumber(args[2], 2)), n)
986 }
987 elseif verb == "SYNDROME" or verb == "DECODE" or verb == "BOUNDED" or verb == "COSET" then
988 local text = args[verb == "BOUNDED" and 3 or 2]
989 if #text ~= n then
990 fail("bad length")
991 end
992 local word = tonumber(text, 2)
993 local syndrome = code:syndrome(word)
994 if verb == "SYNDROME" then
995 return {
996 "syndrome " ..
997 name .. " " .. bits(syndrome, n - k) .. " codeword " .. yes(syndrome == 0)
998 }
999 end
1000 local decoded = code:decode(word)
1001 if verb == "COSET" then
1002 local coset = {}
1003 for i, value in ipairs(code.words) do
1004 coset[i] = value ~ word
1005 end
1006 return {
1007 "coset " ..
1008 name ..
1009 " syndrome " ..
1010 bits(syndrome, n - k) ..
1011 " leader " ..
1012 bits(decoded.error, n) ..
1013 " weights " .. weightList(distribution(coset, n))
1014 }
1015 end
1016 if verb == "BOUNDED" and weight(decoded.error) > tonumber(args[2], 10) then
1017 return {"bounded " .. name .. " fail weight " .. weight(decoded.error)}
1018 end
1019 return {verb:lower() .. " " .. name .. " " .. code:decodedText(decoded)}
1020 elseif verb == "ERASURE" then
1021 local text = args[2]
1022 if #text ~= n then
1023 fail("bad length")
1024 end
1025 local mask, known, erased = 0, 0, 0
1026 for i = 1, n do
1027 local digit = text:sub(i, i)
1028 mask = (mask << 1) | (digit ~= "?" and 1 or 0)
1029 known = (known << 1) | (digit == "1" and 1 or 0)
1030 if digit == "?" then
1031 erased = erased + 1
1032 end
1033 end
1034 local candidates = {}
1035 for _, word in ipairs(code.words) do
1036 if (word & mask) == known then
1037 candidates[#candidates + 1] = word
1038 end
1039 end
1040 local output = "erasure " .. name .. " erased " .. erased .. " candidates " .. #candidates
1041 if #candidates == 1 then
1042 output =
1043 output ..
1044 " codeword " ..
1045 bits(candidates[1], n) .. " message " .. bits(code.messages[candidates[1]], k)
1046 end
1047 return {output}
1048 elseif verb == "INTERLEAVE" then
1049 return interleave(name, code, args)
1050 end
1051 error("Unreachable command")
1052end
1053
1054local count = tonumber(io.read("l"), 10)
1055for _ = 1, count do
1056 local ok, result = pcall(process, io.read("l"))
1057 if ok then
1058 for _, line in ipairs(result) do
1059 io.write(line, "\n")
1060 end
1061 elseif type(result) == "table" and result.reason then
1062 io.write("error: ", result.reason, "\n")
1063 else
1064 error(result, 0)
1065 end
1066end
