157Division divide(int64 value, int64 divisor)
158 int64 quotient := 0
159 while value != 0 and bit_length(value) >= bit_length(divisor)
160 shift := bit_length(value) - bit_length(divisor)
161 quotient = quotient XOR power2(shift)
162 value = value XOR (divisor << int64(shift))

181class Echelon(int64[] rows, int[] columns)
182
183Echelon rref(const int64[] source, int size)
184 rows := source
185 int[] columns := []
186 for: column < size
187 rank := int(#columns)
188 found := rank
189 while found < int(#rows) and (rows[found] AND bit(size, column)) == 0
190 found += 1
191 if found == int(#rows)
192 continue
193 temporary := rows[rank]
194 rows[rank] = rows[found]
195 rows[found] = temporary
196 for index := rows.indices()[]
197 if int(index) != rank and (rows[index] AND bit(size, column)) != 0
198 rows[index] = rows[index] XOR rows[rank]
199 columns = columns.append(column)
200 return Echelon(rows[0:<#columns], columns)

202int64[] null_space(const int64[] source, int size)
203 reduced := rref(source, size)
204 int64[] result := []
205 for: column < size
206 if index_of(reduced.columns, column) >= 0
207 continue
208 row := bit(size, column)
209 for index := reduced.rows.indices()[]
210 if (reduced.rows[index] AND bit(size, column)) != 0
211 row = row OR bit(size, reduced.columns[index])
212 result = result.append(row)
213 return result

221int64[] reed_muller(int order, int variables)
222 if order == 0
223 return [mask(2 ^ variables)]
224 half := 2 ^ (variables - 1)
225 first := reed_muller(math.min(order, variables - 1), variables - 1)
226 rows := [for row := first[] then (row << int64(half)) OR row]
227 return rows.concat(reed_muller(order - 1, variables - 1))

276int64[] leaders(const Code code)
277 int64[] result := array(nat(power2(int(#code.parity))), fill = -1)
278 result[0] = 0
279 found := 1
280 for: 1 <= size <= code.size
281 word := mask(size)
282 while word < power2(code.size)
283 state := syndrome(code, word)
284 if result[state] == -1
285 result[state] = word
286 found += 1
287 if found == int(#result)
288 return result
289 lowest := word AND -word
290 next := word + lowest
291 word = next OR (((next XOR word) >> 2) / lowest)
292 return result

304int burst_correction(const Code code)
305 seen := array(nat(power2(int(#code.parity))), fill = false)
306 seen[0] = true
307 for: 1 <= size <= code.size
308 variants := if size == 1 then int64(1) else power2(size - 2)
309 for: 0 <= offset <= code.size - size
310 for: middle < variants
311 word := if size == 1 then int64(1) else power2(size - 1) OR (middle << 1) OR 1
312 state := syndrome(code, word << int64(offset))
313 if seen[state]
314 return size - 1
315 seen[state] = true
316 return 0

318Code | error construct(Command command, const string[] args, const Code[] sources)
319 if command == Command.Gen or command == Command.Pcheck
320 size := int(#args[1])
321 for word := args[1:][]
322 if int(#word) != size
323 return error("bad length")
324 rows := basis([for word := args[1:][] then parse_bits(word)], size)
325 if command == Command.Pcheck
326 return make_code(size, null_space(rows, size), rows)
327 return make_code(size, rows)
328 if command == Command.Hamming or command == Command.Simplex
329 order := int(args[1])
330 if not (2 <= order <= 4)
331 return error("bad parameter")
332 size := int(mask(order))
333 int64[] rows := array(nat(order), fill = 0)
334 for: index < order
335 for: 1 <= column <= size
336 rows[index] = (rows[index] << 1) OR ((int64(column) >> int64(order - index - 1)) AND 1)
337 if command == Command.Hamming
338 return make_code(size, null_space(rows, size), rows)
339 return make_code(size, rows)
340 if command == Command.Repetition or command == Command.Parity
341 size := int(args[1])
342 if not (2 <= size <= 24)
343 return error("bad parameter")
344 int64[] rows := if command == Command.Repetition then [mask(size)] else [for column := (0:<int(size - 1))[] then bit(size, column) OR 1]
345 return make_code(size, rows)
346 if command == Command.Rm
347 order := int(args[1])
348 variables := int(args[2])
349 if not (1 <= variables <= 4) or order > variables
350 return error("bad parameter")
351 return make_code(2 ^ variables, reed_muller(order, variables))
352 if command == Command.Cyclic
353 size := int(args[1])
354 degree := int(#args[2]) - 1
355 if not (2 <= size <= 24) or args[2][0] != "1" or not (1 <= degree < size)
356 return error("bad parameter")
357 polynomial := parse_bits(args[2])
358 if divide(power2(size) OR 1, polynomial).remainder != 0
359 return error("not a divisor")
360 dimension := size - degree
361 rows := [for index := (0:<int(dimension))[] then polynomial << int64(dimension - index - 1)]
362 int64[] parity := array(nat(degree), fill = 0)
363 for: column < size
364 residue := divide(bit(size, column), polynomial).remainder
365 for: index < degree
366 parity[index] = (parity[index] << 1) OR ((residue >> int64(degree - index - 1)) AND 1)
367 return make_code(size, rows, parity, polynomial)
368 source := sources[0]
369 size := source.size
370 int64[] rows := []
371 if command == Command.Dual
372 return make_code(size, source.parity, source.generator)
373 if command == Command.Extend
374 rows = [for row := source.generator[] then (row << 1) OR int64(weight(row) % 2)]
375 return make_code(size + 1, rows)
376 if command == Command.Puncture or command == Command.Shorten or command == Command.Expurgate
377 expurgating := command == Command.Expurgate
378 int position := if expurgating then 0 else int(args[2])
379 if not expurgating and not (1 <= position <= size)
380 return error("bad position")
381 removed := -1
382 if command != Command.Puncture
383 for index := source.generator.indices()[]
384 row := source.generator[index]
385 selected := if expurgating then weight(row) % 2 != 0 else (row AND bit(size, position - 1)) != 0
386 if selected
387 removed = int(index)
388 break
389 for index := source.generator.indices()[]
390 if int(index) == removed
391 continue
392 row := source.generator[index]
393 selected := if expurgating then weight(row) % 2 != 0 else (row AND bit(size, position - 1)) != 0
394 if removed >= 0 and selected
395 row = row XOR source.generator[removed]
396 if not expurgating
397 low := size - position
398 row = ((row >> int64(low + 1)) << int64(low)) OR (row AND mask(low))
399 rows = rows.append(row)
400 if not expurgating
401 size -= 1
402 if command == Command.Puncture
403 rows = basis(rows, size)
404 return make_code(size, rows)
405 if command == Command.Augment
406 return make_code(size, basis(source.generator.append(mask(size)), size))
407 if command == Command.Permute
408 permutation := parsed(args[2:])
409 if permutation.sorted() != [for column := (1:<(int(size) + 1))[] then column]
410 return error("bad permutation")
411 for original := source.generator[]
412 int64 row := 0
413 for column := permutation[]
414 row = (row << 1) OR (if (original AND bit(size, column - 1)) != 0 then 1 else 0)
415 rows = rows.append(row)
416 return make_code(size, rows)
417 other := sources[1]
418 if command == Command.Uuv and size != other.size
419 return error("bad length")
420 dimension := if command == Command.Product then source.dimension() * other.dimension() else source.dimension() + other.dimension()
421 combined_size := if command == Command.Product then size * other.size else size + other.size
422 try check_dimension(combined_size, dimension)
423 if command == Command.Product
424 for left := source.generator[]
425 for right := other.generator[]
426 int64 row := 0
427 for: column < size
428 row = (row << int64(other.size)) OR (if (left AND bit(size, column)) != 0 then right else int64(0))
429 rows = rows.append(row)
430 else
431 rows = [for row := source.generator[] then (row << int64(other.size)) OR (if command == Command.Uuv then row else int64(0))]
432 rows = rows.concat(other.generator)
433 return make_code(combined_size, rows)

474 size := code.size
475 dimension := code.dimension()
476 rows := code.generator
477 permutation := [for column := (1:<(int(size) + 1))[] then column]
478 for: pivot < dimension
479 column := pivot
480 while true
481 present := false
482 for: pivot <= row < dimension
483 if (rows[row] AND bit(size, column)) != 0
484 present = true
485 break
486 if present
487 break
488 column += 1
489 if column != pivot
490 for index := rows.indices()[]

492 rows[index] = rows[index] XOR bit(size, pivot) XOR bit(size, column)

495 permutation[column] = temporary
496 found := pivot
497 while (rows[found] AND bit(size, pivot)) == 0

511 row = (row << int64(size - dimension)) OR bit(size - dimension, column)
512 output = output.append("H " + bit_text(row, size))
513 return output

515string[] | error inspect_code(Command command, const string[] args, const Code[] sources, const int[][] table)
516 code := sources[0]
517 name := args[0]
518 size := code.size
519 dimension := code.dimension()
520 if command == Command.Show
521 output := ["code {name} n {size} k {dimension} encoder {code.encoder()}"]
522 match code.polynomial
523 int64 polynomial
524 quotient := divide(power2(size) OR 1, polynomial).quotient
525 output = output.append("poly g " + bit_text(polynomial, size - dimension + 1) + " h " + bit_text(quotient, dimension + 1))
526 none
527 void
528 output = output.concat([for row := code.generator[] then "G " + bit_text(row, size)])
529 return output.concat([for row := code.parity[] then "H " + bit_text(row, size)])
530 if command == Command.Systematic
531 return systematic(name, code)
532 if command == Command.Equal
533 other := sources[1]
534 if size != other.size
535 return ["equal {name} {args[1]} no lengths {size} {other.size}"]
536 common := dimension + other.dimension() - int(#basis(code.generator.concat(other.generator), size))
537 return ["equal {name} {args[1]} " + yes(dimension == other.dimension() == common) + " intersection {common}"]
538 if command == Command.Trellis
539 int[] profile := []
540 for: 0 <= column <= size
541 left := [for row := code.generator[] then row >> int64(size - column)]
542 right := [for row := code.generator[] then row AND mask(size - column)]
543 profile = profile.append(int(#basis(left, column)) + int(#basis(right, size - column)) - dimension)
544 return ["trellis {name} profile " + numbers(profile) + " max {profile.max()}"]
545 if command == Command.Covering
546 counts := distribution(leaders(code), size)
547 radius := size
548 while counts[radius] == 0
549 radius -= 1
550 items := [for index := (0:<(int(radius) + 1))[] then "{index}:{counts[index]}"]
551 return ["covering {name} radius {radius} leaders " + items.join(" ")]
552 if command == Command.Stdarray and (size - dimension > 6 or dimension > 4)
553 return error("too large")
554 words := span(code.generator)
555 if command == Command.Info
556 distance := [for word := words[1:][] then weight(word)].min()
557 correction := (distance - 1) / 2
558 cyclic := true
559 for row := code.generator[]
560 rotated := ((row << 1) AND mask(size)) OR (row >> int64(size - 1))
561 if syndrome(code, rotated) != 0
562 cyclic = false
563 break
564 polynomial := "no"
565 if cyclic
566 first := words[1:].min()
567 polynomial = bit_text(first, bit_length(first))
568 perfect := volume(table, size, correction) == 2 ^ (size - dimension)
569 return ["info {name} n {size} k {dimension} rate " + fraction_text(dimension, size) + " d {distance} t {correction} perfect " + yes(perfect) + " mds " + yes(distance == size - dimension + 1) + " cyclic {polynomial}"]
570 if command == Command.Weights
571 primal := distribution(words, size)
572 dual := distribution(span(code.parity), size)
573 orthogonal := true
574 for left := code.generator[]
575 for right := code.generator[]
576 if weight(left AND right) % 2 != 0
577 orthogonal = false
578 return [
579 "weights {name} " + weight_items(primal),
580 "dual {name} " + weight_items(dual),
581 "duality {name} self-orthogonal " + yes(orthogonal) + " self-dual " + yes(orthogonal and size == 2 * dimension) + " formally-self-dual " + yes(primal == dual),
582 ]
583 if command == Command.Burst
584 detection := [for word := words[1:][] then bit_length(word) - bit_length(word AND -word)].min()
585 return ["burst {name} detect {detection} correct {burst_correction(code)}"]
586 errors := leaders(code)
587 keys := [for pattern := errors[] then int64(weight(pattern)) * power2(size) + pattern]
588 output := ["stdarray {name} rows {#errors} columns {#words}"]
589 words = words.sorted()
590 for key := keys.sorted()[]
591 pattern := key % power2(size)
592 state := index_of(errors, pattern)
593 row := [for word := words[] then bit_text(pattern XOR word, size)]
594 output = output.append(bit_text(int64(state), size - dimension) + ": " + row.join(" "))
595 return output

597string[] | error encode_report(Command command, const string[] args, const Code code)
598 name := args[0]
599 size := code.size
600 dimension := code.dimension()
601 if command == Command.Interleave
602 messages := args[3:]
603 for message := messages[]
604 if int(#message) != dimension
605 return error("bad length")
606 start := int(args[1])
607 length := int(args[2])
608 depth := int(#messages)
609 if start < 1 or length < 1 or start + length - 1 > depth * size
610 return error("bad burst")
611 encoded := [for message := messages[] then encode(code, parse_bits(message))]
612 received := encoded
613 sent_text := ""
614 received_text := ""
615 for: column < size
616 for: row < depth
617 sent := (encoded[row] AND bit(size, column)) != 0
618 position := column * depth + row + 1
619 flipped := start <= position < start + length
620 sent_text += if sent then "1" else "0"
621 received_text += if sent != flipped then "1" else "0"
622 if flipped
623 received[row] = received[row] XOR bit(size, column)
624 output := ["interleave {name} depth {depth} burst {start} {length}", "sent {sent_text}", "recv {received_text}"]
625 errors := leaders(code)
626 correct := 0
627 for: row < depth
628 result := decode(code, received[row], errors)
629 success := result.message == parse_bits(messages[row])
630 if success
631 correct += 1

700 int64 factor := 3

704 exponent := 0

706 remaining = divide(remaining, factor).quotient
707 exponent += 1
708 if exponent != 0
709 factors = factors.append(bit_text(factor, bit_length(factor)) + "^{exponent}")

712 if remaining != 1
713 factors = factors.append(bit_text(remaining, bit_length(remaining)) + "^1")
714 divisors *= 2
