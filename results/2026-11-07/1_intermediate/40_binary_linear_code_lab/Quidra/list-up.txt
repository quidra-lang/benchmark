int64 power2(int exponent)
    return int64(1) << int64(exponent)
int64 bit(int size, int column)
    return power2(size - column - 1)
int64 parse_bits(string text)
    int64 result := 0
    for: character_index < #text
        unit := text[character_index]
        result = result * 2 + (if unit == "1" then 1 else 0)
    return result
int weight(int64 value)
    result := 0
    while value != 0
        value = value AND (value - 1)
        result += 1
    return result
int bit_length(int64 value)
    result := 0
    while value != 0
        value = value >> 1
        result += 1
    return result
string bit_text(int64 value, int size)
    result := ""
    for: column < size
        result += if (value AND bit(size, column)) == 0 then "0" else "1"
    return result
class Division(int64 quotient, int64 remainder)
Division divide(int64 value, int64 divisor)
    while value != 0 and bit_length(value) >= bit_length(divisor)
        shift := bit_length(value) - bit_length(divisor)
        value = value XOR (divisor << int64(shift))
    return Division(quotient, value)
class Code(int size, int64[] generator, int64[] parity, int64 | none polynomial := none)
    int dimension()
        return int(#generator)
Code | error make_code(int size, const int64[] generator, int64[] | none parity := none, int64 | none polynomial := none)
    match parity
        int64[] rows
            return Code(size, generator, rows, polynomial)
        none
            return Code(size, generator, null_space(generator, size), polynomial)
int64 encode(const Code code, int64 message)
    match code.polynomial
        int64 divisor
            shifted := message << int64(#code.parity)
            return shifted OR divide(shifted, divisor).remainder
        none
            int64 result := 0
            for index := code.generator.indices()[]
                if (message AND bit(code.dimension(), int(index))) != 0
                    result = result XOR code.generator[index]
            return result

    for row := code.parity[]
        result = (result << 1) OR int64(weight(row AND word) % 2)
