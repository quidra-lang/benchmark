88interface Division { readonly quotient: Word; readonly remainder: Word }
89function divide(dividend: Word, divisor: Word): Division {
90 let quotient = 0;
91 const d = degree(divisor);
92 while (dividend && degree(dividend) >= d) {
93 const shift = degree(dividend) - d;
94 quotient ^= 1 << shift;
95 dividend ^= divisor << shift;
96 }
97 return { quotient, remainder: dividend };
98}
99function basis(rows: readonly Word[], length: number): Word[] {
100 const pivots = Array<Word>(length).fill(0), kept: Word[] = [];
101 for (const original of rows) {
102 let row = original;
103 for (let column = 0; column < length; ++column) if (bit(row, length, column)) {
104 if (pivots[column]) row ^= pivots[column];
105 else { pivots[column] = row; kept.push(original); break; }
106 }
107 }
108 return kept;
109}
110interface Echelon { readonly rows: readonly Word[]; readonly columns: readonly number[] }
111function rref(original: readonly Word[], length: number): Echelon {
112 const rows = original.slice(), columns: number[] = [];
113 for (let column = 0; column < length; ++column) {
114 const pivot = columns.length;
115 const found = rows.findIndex((row, index) => index >= pivot && bit(row, length, column));
116 if (found < 0) continue;
117 [rows[pivot], rows[found]] = [rows[found], rows[pivot]];
118 for (let row = 0; row < rows.length; ++row) if (row !== pivot && bit(rows[row], length, column)) rows[row] ^= rows[pivot];
119 columns.push(column);
120 }
121 return { rows: rows.slice(0, columns.length), columns };
122}
123function nullspace(original: readonly Word[], length: number): Word[] {
124 const { rows, columns } = rref(original, length), free: Word[] = [];
125 for (let column = 0; column < length; ++column) if (!columns.includes(column)) {
126 let word = 1 << (length - column - 1);
127 for (let row = 0; row < rows.length; ++row) if (bit(rows[row], length, column)) word |= 1 << (length - columns[row] - 1);
128 free.push(word);
129 }
130 return free;
131}
132function span(rows: readonly Word[]): Word[] {
133 const words = [0];
134 for (const row of rows) {
135 const count = words.length;
136 for (let index = 0; index < count; ++index) words.push(words[index] ^ row);
137 }
138 return words;
139}
140function distribution(words: readonly Word[], length: number): number[] {
141 const counts = Array<number>(length + 1).fill(0);
142 for (const word of words) ++counts[weight(word)];
143 return counts;

145const weightList = (counts: readonly number[]): string => counts.flatMap((count, index) => count ? [`${index}:${count}`] : []).join(' ');
146function* fixedWeight(length: number, count: number): Generator<Word, void, unknown> {
147 if (count === 0) { yield 0; return; }
148 for (let word = 2 ** count - 1; word < 2 ** length;) {
149 yield word;
150 const lowest = word & -word, next = word + lowest;
151 word = next | (((next ^ word) >>> 2) / lowest);
152 }
153}
154function dimension(length: number, rank: number): void {
155 if (rank < 1 || rank > 12 || length - rank < 1 || length - rank > 12) fail(`bad dimension ${length} ${rank}`);
156}
157function removeBit(word: Word, length: number, column: number): Word {
158 const low = length - column - 1;
159 return ((word >>> (low + 1)) << low) | (word & ((1 << low) - 1));
160}
161function reedMuller(order: number, variables: number): Word[] {
162 const length = 1 << variables;
163 if (order === 0) return [(1 << length) - 1];
164 const half = length / 2;
165 return [...reedMuller(Math.min(order, variables - 1), variables - 1).map(row => (row << half) | row),
166 ...reedMuller(order - 1, variables - 1)];
167}

169interface Decoded { readonly error: Word; readonly word: Word; readonly message: Word }
170class Code {
171 readonly n: number;
172 readonly k: number;
173 readonly G: readonly Word[];
174 readonly H: readonly Word[];
175 readonly polynomial: Word | null;
176 readonly words: readonly Word[];
177 private readonly messages = new Map<Word, Word>();
178 private leaderTable: readonly Word[] | null = null;
179 constructor(length: number, generator: readonly Word[], check: readonly Word[] | null = null, polynomial: Word | null = null) {
180 dimension(length, generator.length);
181 this.n = length; this.k = generator.length;
182 this.G = generator.slice(); this.H = check?.slice() ?? nullspace(generator, length);
183 this.polynomial = polynomial;
184 this.words = Array.from({ length: 1 << this.k }, (_, message) => {
185 const word = this.encode(message);
186 this.messages.set(word, message);
187 return word;
188 }).sort((a, b) => a - b);
189 }
190 get encoder(): 'matrix' | 'polynomial' { return this.polynomial === null ? 'matrix' : 'polynomial'; }
191 contains(word: Word): boolean { return this.messages.has(word); }
192 messageOf(word: Word): Word { return this.messages.get(word)!; }
193 encode(message: Word): Word {
194 if (this.polynomial !== null) {
195 const shifted = message << (this.n - this.k);
196 return shifted | divide(shifted, this.polynomial).remainder;
197 }
198 let word = 0;
199 for (let row = 0; row < this.k; ++row) if (bit(message, this.k, row)) word ^= this.G[row];
200 return word;
201 }
202 syndrome(word: Word): Word {
203 let result = 0;
204 for (const row of this.H) result = (result << 1) | (weight(row & word) % 2);
205 return result;
206 }
207 leaders(): readonly Word[] {
208 if (this.leaderTable !== null) return this.leaderTable;
209 const result = Array<Word>(1 << (this.n - this.k)).fill(-1);
210 let remaining = result.length;
211 for (let count = 0; remaining; ++count) for (const word of fixedWeight(this.n, count)) {
212 const syndrome = this.syndrome(word);
213 if (result[syndrome] === -1) {
214 result[syndrome] = word;
215 if (--remaining === 0) break;
216 }
217 }
218 this.leaderTable = result;
219 return result;
220 }
221 decode(word: Word): Decoded {
222 const error = this.leaders()[this.syndrome(word)], corrected = word ^ error;
223 return { error, word: corrected, message: this.messageOf(corrected) };
224 }
225 decodedText(decoded: Decoded): string {
226 return `codeword ${bits(decoded.word, this.n)} message ${bits(decoded.message, this.k)} error ${bits(decoded.error, this.n)} weight ${weight(decoded.error)}`;
227 }
228}

230function construct(definition: Definition): Code {
231 switch (definition.command) {
232 case 'GEN': case 'PCHECK': {
233 const n = definition.rows[0].length;
234 if (definition.rows.some(row => row.length !== n)) fail('bad length');
235 const rows = basis(definition.rows.map(row => parseInt(row, 2)), n);
236 return definition.command === 'GEN' ? new Code(n, rows) : new Code(n, nullspace(rows, n), rows);
237 }
238 case 'HAMMING': case 'SIMPLEX': {
239 const r = definition.parameter;
240 if (r < 2 || r > 4) fail('bad parameter');
241 const n = (1 << r) - 1, rows = Array<Word>(r).fill(0);
242 for (let column = 1; column <= n; ++column) for (let row = 0; row < r; ++row) rows[row] = (rows[row] << 1) | bit(column, r, row);
243 return definition.command === 'HAMMING' ? new Code(n, nullspace(rows, n), rows) : new Code(n, rows);
244 }
245 case 'REPETITION': case 'PARITY': {
246 const n = definition.parameter;
247 if (n < 2 || n > 24) fail('bad parameter');
248 const rows = definition.command === 'REPETITION' ? [(1 << n) - 1] : Array.from({ length: n - 1 }, (_, row) => (1 << (n - row - 1)) | 1);
249 return new Code(n, rows);
250 }
251 case 'RM': {
252 const { order, variables } = definition;
253 if (variables < 1 || variables > 4 || order > variables) fail('bad parameter');
254 return new Code(1 << variables, reedMuller(order, variables));
255 }
256 case 'CYCLIC': {
257 const { length: n, polynomial: text } = definition, d = text.length - 1;
258 if (n < 2 || n > 24 || text[0] === '0' || text.length < 2 || text.length > n) fail('bad parameter');
259 const polynomial = parseInt(text, 2);
260 if (divide((1 << n) | 1, polynomial).remainder) fail('not a divisor');
261 const k = n - d;
262 dimension(n, k);
263 const rows = Array.from({ length: k }, (_, row) => polynomial << (k - row - 1)), checks = Array<Word>(d).fill(0);
264 for (let column = 0; column < n; ++column) {
265 const residue = divide(1 << (n - column - 1), polynomial).remainder;
266 for (let row = 0; row < d; ++row) checks[row] = (checks[row] << 1) | bit(residue, d, row);
267 }
268 return new Code(n, rows, checks, polynomial);
269 }
270 case 'DUAL': return new Code(definition.source.n, definition.source.H, definition.source.G);
271 case 'EXTEND': return new Code(definition.source.n + 1, definition.source.G.map(row => (row << 1) | (weight(row) % 2)));
272 case 'PUNCTURE': case 'SHORTEN': {
273 const a = definition.source, column = definition.position - 1;
274 if (column < 0 || column >= a.n) fail('bad position');
275 let rows = a.G.slice();
276 if (definition.command === 'SHORTEN') {
277 const pivot = rows.findIndex(row => bit(row, a.n, column));
278 if (pivot >= 0) {
279 const z = rows[pivot];
280 rows = rows.filter((_, index) => index !== pivot).map(row => bit(row, a.n, column) ? row ^ z : row);
281 }
282 }
283 const n = a.n - 1;
284 rows = rows.map(row => removeBit(row, a.n, column));
285 return new Code(n, definition.command === 'PUNCTURE' ? basis(rows, n) : rows);
286 }
287 case 'AUGMENT': {
288 const a = definition.source;
289 return new Code(a.n, basis([...a.G, (1 << a.n) - 1], a.n));
290 }
291 case 'EXPURGATE': {
292 const a = definition.source, pivot = a.G.findIndex(row => weight(row) % 2);
293 const rows = pivot < 0 ? a.G : a.G.filter((_, index) => index !== pivot).map(row => weight(row) % 2 ? row ^ a.G[pivot] : row);
294 return new Code(a.n, rows);
295 }
296 case 'PERMUTE': {
297 const { source: a, order } = definition;
298 if (order.length !== a.n || new Set(order).size !== a.n || order.some(column => column < 1 || column > a.n)) fail('bad permutation');
299 return new Code(a.n, a.G.map(row => order.reduce((word, column) => (word << 1) | bit(row, a.n, column - 1), 0)));
300 }
301 case 'DIRECTSUM': case 'UUV': case 'PRODUCT': {
302 const { source: a, other: b, command } = definition;
303 if (command === 'UUV' && a.n !== b.n) fail('bad length');
304 const n = command === 'PRODUCT' ? a.n * b.n : a.n + b.n;
305 dimension(n, command === 'PRODUCT' ? a.k * b.k : a.k + b.k);
306 const rows = command === 'PRODUCT' ? a.G.flatMap(x => b.G.map(y => {
307 let word = 0;
308 for (let column = 0; column < a.n; ++column) word = (word << b.n) | (bit(x, a.n, column) ? y : 0);
309 return word;
310 })) : [...a.G.map(row => (row << b.n) | (command === 'UUV' ? row : 0)), ...b.G];
311 return new Code(n, rows);
312 }
313 default: return unreachable(definition);
314 }
315}

324function systematic(name: string, code: Code): string[] {
325 const { n, k } = code, rows = code.G.slice(), order = Array.from({ length: n }, (_, index) => index + 1);
326 for (let pivot = 0; pivot < k; ++pivot) {
327 let column = pivot;
328 while (!rows.some((row, index) => index >= pivot && bit(row, n, column))) ++column;
329 if (column !== pivot) {
330 const mask = (1 << (n - pivot - 1)) | (1 << (n - column - 1));
331 for (let row = 0; row < k; ++row) if (bit(rows[row], n, pivot) !== bit(rows[row], n, column)) rows[row] ^= mask;
332 [order[pivot], order[column]] = [order[column], order[pivot]];
333 }
334 const found = rows.findIndex((row, index) => index >= pivot && bit(row, n, pivot));
335 [rows[pivot], rows[found]] = [rows[found], rows[pivot]];
336 for (let row = 0; row < k; ++row) if (row !== pivot && bit(rows[row], n, pivot)) rows[row] ^= rows[pivot];
337 }
338 const output = [`systematic ${name} perm ${order.join(' ')}`, ...rows.map(row => `G ${bits(row, n)}`)];
339 for (let column = k; column < n; ++column) {
340 let row = 0;
341 for (const value of rows) row = (row << 1) | bit(value, n, column);
342 row = (row << (n - k)) | (1 << (n - column - 1));
343 output.push(`H ${bits(row, n)}`);
344 }
345 return output;
346}
347function burst(name: string, code: Code): string[] {
348 const detect = Math.min(...code.words.slice(1).map(word => degree(word) - degree(word & -word)));
349 const seen = new Set<Word>([0]);
350 for (let length = 1; length <= code.n; ++length) {
351 for (let start = 0; start + length <= code.n; ++start) {
352 for (let middle = 0; middle < (length === 1 ? 1 : 1 << (length - 2)); ++middle) {
353 const pattern = length === 1 ? 1 : (1 << (length - 1)) | (middle << 1) | 1;
354 const syndrome = code.syndrome(pattern << (code.n - start - length));
355 if (seen.has(syndrome)) return [`burst ${name} detect ${detect} correct ${length - 1}`];
356 seen.add(syndrome);
357 }
358 }
359 }
360 throw new Error('Missing burst collision');

362function inspect(command: Inspection): string[] {
363 const { name, code } = command;
364 switch (command.command) {
365 case 'SHOW': {
366 const output = [`code ${name} n ${code.n} k ${code.k} encoder ${code.encoder}`];
367 if (code.polynomial !== null) output.push(`poly g ${bits(code.polynomial, code.n - code.k + 1)} h ${bits(divide((1 << code.n) | 1, code.polynomial).quotient, code.k + 1)}`);
368 return [...output, ...code.G.map(row => `G ${bits(row, code.n)}`), ...code.H.map(row => `H ${bits(row, code.n)}`)];
369 }
370 case 'INFO': {
371 const { n, k } = code, divisor = gcd(n, k), distance = Math.min(...code.words.slice(1).map(weight)), t = Math.floor((distance - 1) / 2);
372 const volume = choose[n].slice(0, t + 1).reduce((a, b) => a + b, 0);
373 const cyclic = code.G.every(row => code.contains(((row << 1) & ((1 << n) - 1)) | (row >>> (n - 1))));
374 return [`info ${name} n ${n} k ${k} rate ${k / divisor}/${n / divisor} d ${distance} t ${t} perfect ${yes(volume === 2 ** (n - k))} mds ${yes(distance === n - k + 1)} cyclic ${cyclic ? code.words[1].toString(2) : 'no'}`];
375 }
376 case 'SYSTEMATIC': return systematic(name, code);
377 case 'WEIGHTS': {
378 const a = distribution(code.words, code.n), b = distribution(span(code.H), code.n);
379 const orthogonal = code.G.every(x => code.G.every(y => weight(x & y) % 2 === 0));
380 return [`weights ${name} ${weightList(a)}`, `dual ${name} ${weightList(b)}`,
381 `duality ${name} self-orthogonal ${yes(orthogonal)} self-dual ${yes(orthogonal && code.n === 2 * code.k)} formally-self-dual ${yes(a.every((value, index) => value === b[index]))}`];
382 }
383 case 'COVERING': {
384 const counts = distribution(code.leaders(), code.n);
385 let radius = code.n;
386 while (counts[radius] === 0) --radius;
387 return [`covering ${name} radius ${radius} leaders ${counts.slice(0, radius + 1).map((count, index) => `${index}:${count}`).join(' ')}`];
388 }
389 case 'BURST': return burst(name, code);
390 case 'TRELLIS': {
391 const profile: number[] = [];
392 for (let cut = 0; cut <= code.n; ++cut) {
393 const right = code.n - cut, mask = (1 << right) - 1;
394 profile.push(basis(code.G.map(row => row >>> right), cut).length + basis(code.G.map(row => row & mask), right).length - code.k);
395 }
396 return [`trellis ${name} profile ${profile.join(' ')} max ${Math.max(...profile)}`];
397 }
398 case 'STDARRAY': {
399 if (code.n - code.k > 6 || code.k > 4) fail('too large');
400 const leaders = [...code.leaders().entries()].sort(([, a], [, b]) => weight(a) - weight(b) || a - b);
401 return [`stdarray ${name} rows ${leaders.length} columns ${code.words.length}`, ...leaders.map(([syndrome, leader]) =>
402 `${bits(syndrome, code.n - code.k)}: ${code.words.map(word => bits(word ^ leader, code.n)).join(' ')}`)];
403 }
404 case 'EQUAL': {
405 const { other, otherName } = command;
406 if (code.n !== other.n) return [`equal ${name} ${otherName} no lengths ${code.n} ${other.n}`];
407 const common = code.k + other.k - basis([...code.G, ...other.G], code.n).length;
408 return [`equal ${name} ${otherName} ${yes(code.k === common && other.k === common)} intersection ${common}`];
409 }
410 default: return unreachable(command);
411 }
412}
413function interleave(command: Extract<Encoding, { command: 'INTERLEAVE' }>): string[] {
414 const { name, code, messages, start, length } = command, depth = messages.length;
415 if (messages.some(message => message.length !== code.k)) fail('bad length');
416 if (start < 1 || length < 1 || start + length - 1 > depth * code.n) fail('bad burst');
417 const sentWords = messages.map(message => code.encode(parseInt(message, 2)));
418 const receivedWords = Array<Word>(depth).fill(0), sent: number[] = [], received: number[] = [];
419 for (let column = 0; column < code.n; ++column) for (let row = 0; row < depth; ++row) {
420 const position = column * depth + row + 1, original = bit(sentWords[row], code.n, column);
421 const changed = original ^ Number(start <= position && position < start + length);
422 sent.push(original); received.push(changed);
423 receivedWords[row] = (receivedWords[row] << 1) | changed;
424 }
425 const output = [`interleave ${name} depth ${depth} burst ${start} ${length}`, `sent ${sent.join('')}`, `recv ${received.join('')}`];
426 let correct = 0;
427 for (let row = 0; row < depth; ++row) {
428 const decoded = code.decode(receivedWords[row]), message = bits(decoded.message, code.k), ok = message === messages[row];
429 if (ok) ++correct;
430 output.push(`row ${row + 1} errors ${weight(receivedWords[row] ^ sentWords[row])} corrected ${weight(decoded.error)} message ${message} ${ok ? 'ok' : 'wrong'}`);
431 }
432 output.push(`result ok ${correct} wrong ${depth - correct}`);
433 return output;
434}
435function encodingReport(command: Encoding): string[] {
436 if (command.command === 'INTERLEAVE') return interleave(command);
437 const { name, code, text } = command;
438 if (text.length !== (command.command === 'ENCODE' ? code.k : code.n)) fail('bad length');
439 if (command.command === 'ENCODE') return [`encode ${name} message ${text} codeword ${bits(code.encode(parseInt(text, 2)), code.n)}`];
440 if (command.command === 'ERASURE') {
441 let mask = 0, known = 0, erased = 0;
442 for (const character of text) {
443 mask = (mask << 1) | Number(character !== '?');
444 known = (known << 1) | Number(character === '1');
445 if (character === '?') ++erased;
446 }
447 const candidates = code.words.filter(word => (word & mask) === known);
448 const detail = candidates.length === 1 ? ` codeword ${bits(candidates[0], code.n)} message ${bits(code.messageOf(candidates[0]), code.k)}` : '';
449 return [`erasure ${name} erased ${erased} candidates ${candidates.length}${detail}`];
450 }
451 const word = parseInt(text, 2), syndrome = code.syndrome(word);
452 if (command.command === 'SYNDROME') return [`syndrome ${name} ${bits(syndrome, code.n - code.k)} codeword ${yes(syndrome === 0)}`];
453 const decoded = code.decode(word);
454 if (command.command === 'COSET') return [`coset ${name} syndrome ${bits(syndrome, code.n - code.k)} leader ${bits(decoded.error, code.n)} weights ${weightList(distribution(code.words.map(value => value ^ word), code.n))}`];
455 if (command.command === 'BOUNDED' && weight(decoded.error) > command.limit) return [`bounded ${name} fail weight ${weight(decoded.error)}`];
456 return [`${command.command.toLowerCase()} ${name} ${code.decodedText(decoded)}`];
457}
458function independent(command: Independent): string[] {
459 if ('generator' in command) {
460 const { generator, text } = command, d = generator.length - 1;
461 if (generator[0] === '0' || generator.length < 2 || generator.length > 17) fail('bad parameter');
462 if (command.command === 'CRC' && text.length + d > 30) fail('bad length');
463 const word = parseInt(text, 2), divisor = parseInt(generator, 2);
464 const remainder = divide(command.command === 'CRC' ? word << d : word, divisor).remainder, suffix = bits(remainder, d);
465 return [command.command === 'CRC' ? `crc remainder ${suffix} frame ${text}${suffix}` : `crccheck ${remainder === 0 ? 'ok' : 'error'} remainder ${suffix}`];
466 }
467 const n = command.length;
468 if (command.command === 'BOUNDS') {
469 const d = command.distance;
470 if (d < 1 || d > n || n > 24) fail('bad parameter');
471 const volume = choose[n].slice(0, Math.floor((d - 1) / 2) + 1).reduce((a, b) => a + b, 0);
472 let hamming = 0, griesmer = 0, sum = 0;
473 for (let k = 0; k <= n; ++k) if (2 ** k * volume <= 2 ** n) hamming = k;
474 for (let k = 1; k <= n; ++k) {
475 sum += Math.ceil(d / 2 ** (k - 1));
476 if (sum <= n) griesmer = k;
477 }
478 return [`bounds n ${n} d ${d} singleton ${n - d + 1} hamming ${hamming} griesmer ${griesmer}`];
479 }
480 if (n < 1 || n > 24) fail('bad parameter');
481 if (command.command === 'FACTOR') {
482 let remaining = (1 << n) | 1, divisors = 1;
483 const factors: string[] = [];
484 for (let candidate = 2; 2 * degree(candidate) <= degree(remaining); ++candidate) {
485 let exponent = 0;
486 while (divide(remaining, candidate).remainder === 0) {
487 remaining = divide(remaining, candidate).quotient;
488 ++exponent;
489 }
490 if (exponent) { factors.push(`${candidate.toString(2)}^${exponent}`); divisors *= exponent + 1; }
491 }
492 if (remaining > 1) { factors.push(`${remaining.toString(2)}^1`); divisors *= 2; }
493 return [`factor ${n} ${factors.join(' ')} divisors ${divisors}`];
494 }
495 const counts = command.counts;
496 if (counts.length !== n + 1) fail('bad length');
497 if (counts[0] !== 1) fail('bad parameter');
498 const size = counts.reduce((a, b) => a + b, 0), transformed: string[] = [];
499 let feasible = (size & (size - 1)) === 0;
500 for (let j = 0; j <= n; ++j) {
501 let numerator = 0;
502 for (let x = 0; x <= n; ++x) {
503 let krawtchouk = 0;
504 for (let y = 0; y <= j; ++y) krawtchouk += (y % 2 ? -1 : 1) * binomial(x, y) * binomial(n - x, j - y);
505 numerator += counts[x] * krawtchouk;
506 }
507 const divisor = gcd(numerator, size), p = numerator / divisor, q = size / divisor;
508 transformed.push(q === 1 ? String(p) : `${p}/${q}`);
509 if (q !== 1 || p < 0) feasible = false;
510 }
511 return [`macwilliams n ${n} size ${size}`, `dual ${transformed.join(' ')}`, `feasible ${yes(feasible)}`];
512}

514class Laboratory {
515 private readonly codes = new Map<string, Code>();
516 private code(name: string): Code {
517 const code = this.codes.get(name);
518 if (!code) fail(`no code ${name}`);
519 return code;
520 }
521 parse(line: string): Command {
522 const [command, ...args] = line.split(' ');
523 if (!isCommandName(command)) fail(`unknown command ${command}`);
524 const spec = formats[command], extra = args.length - spec.types.length;
525 if (spec.repeat === null ? extra !== 0 : extra < 1 || extra > spec.maximum) fail('bad arguments');
526 for (let index = 0; index < args.length; ++index) {
527 const kind = spec.types[index] ?? spec.repeat;
528 if (kind === null || !patterns[kind].test(args[index])) fail('bad arguments');
529 }
530 for (let index = 0; index < spec.types.length; ++index) if (spec.types[index] === 'code') this.code(args[index]);
531 const name = args[0];
532 switch (command) {
533 case 'GEN': case 'PCHECK': return { command, name, rows: args.slice(1) };
534 case 'HAMMING': case 'SIMPLEX': case 'REPETITION': case 'PARITY': return { command, name, parameter: Number(args[1]) };
535 case 'RM': return { command, name, order: Number(args[1]), variables: Number(args[2]) };
536 case 'CYCLIC': return { command, name, length: Number(args[1]), polynomial: args[2] };
537 case 'DUAL': case 'EXTEND': case 'AUGMENT': case 'EXPURGATE': return { command, name, source: this.code(args[1]) };
538 case 'PUNCTURE': case 'SHORTEN': return { command, name, source: this.code(args[1]), position: Number(args[2]) };
539 case 'PERMUTE': return { command, name, source: this.code(args[1]), order: args.slice(2).map(Number) };
540 case 'DIRECTSUM': case 'UUV': case 'PRODUCT': return { command, name, source: this.code(args[1]), other: this.code(args[2]) };
541 case 'LIST': return { command };
542 case 'DELETE': case 'SHOW': case 'INFO': case 'SYSTEMATIC': case 'WEIGHTS': case 'COVERING': case 'BURST': case 'TRELLIS': case 'STDARRAY':
543 return { command, name, code: this.code(name) };
544 case 'EQUAL': return { command, name, code: this.code(name), otherName: args[1], other: this.code(args[1]) };
545 case 'ENCODE': case 'SYNDROME': case 'DECODE': case 'COSET': case 'ERASURE': return { command, name, code: this.code(name), text: args[1] };
546 case 'BOUNDED': return { command, name, code: this.code(name), limit: Number(args[1]), text: args[2] };
547 case 'INTERLEAVE': return { command, name, code: this.code(name), start: Number(args[1]), length: Number(args[2]), messages: args.slice(3) };
548 case 'CRC': case 'CRCCHECK': return { command, generator: args[0], text: args[1] };
549 case 'FACTOR': return { command, length: Number(args[0]) };
550 case 'MACWILLIAMS': return { command, length: Number(args[0]), counts: args.slice(1).map(Number) };
551 case 'BOUNDS': return { command, length: Number(args[0]), distance: Number(args[1]) };
552 default: return unreachable(command);
553 }
554 }
555 execute(command: Command): string[] {
556 if (isDefinition(command)) {
557 const code = construct(command), replaced = this.codes.has(command.name);
558 this.codes.set(command.name, code);
559 return [`${replaced ? 'redefined' : 'defined'} ${command.name} n ${code.n} k ${code.k}`];
560 }
561 switch (command.command) {
562 case 'LIST': return [`codes ${this.codes.size}`, ...[...this.codes].sort(([a], [b]) => a < b ? -1 : a > b ? 1 : 0)
563 .map(([name, code]) => `${name} n ${code.n} k ${code.k} encoder ${code.encoder}`)];
564 case 'DELETE': this.codes.delete(command.name); return [`deleted ${command.name}`];
565 case 'CRC': case 'CRCCHECK': case 'FACTOR': case 'MACWILLIAMS': case 'BOUNDS': return independent(command);
566 case 'ENCODE': case 'SYNDROME': case 'DECODE': case 'BOUNDED': case 'COSET': case 'ERASURE': case 'INTERLEAVE': return encodingReport(command);
567 default: return inspect(command);
568 }
569 }
570}

572const input = readFileSync(0, 'utf8').split('\n'), laboratory = new Laboratory();
573for (let index = 1; index <= Number(input[0]); ++index) {
574 try { process.stdout.write(laboratory.execute(laboratory.parse(input[index])).join('\n') + '\n'); }
575 catch (error) {
576 if (!(error instanceof CommandError)) throw error;
577 process.stdout.write(`error: ${error.message}\n`);
578 }
