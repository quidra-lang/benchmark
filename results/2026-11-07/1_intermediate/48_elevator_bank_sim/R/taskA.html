44parse_config <- function(lines) {
45 entries <- list()
46 for (words in lines) {
47 key <- words[1]
48 if (!key %in% config_keys) {
49 config_error(paste("unknown key", key))
50 }
51 if (key %in% names(entries)) {
52 config_error(paste("duplicate key", key))
53 }
54 entries[key] <- list(words[-1])
55 }
56 for (key in config_keys[1:5]) {
57 if (!key %in% names(entries)) config_error(paste("missing key", key))
58 }
59 cfg <- list(
60 floors = 0,
61 cars = 0,
62 capacity = 0,
63 door = 0,
64 starts = numeric(),
65 zones = list(),
66 homes = numeric(),

68 park = -1,

75 bounded <- function(values, low, high) all(values >= low & values <= high)
76 for (key in config_keys) {
77 if (!key %in% names(entries)) {
78 next
79 }
80 words <- entries[[key]]
81 expected <- if (key %in% c("START", "ZONES")) {
82 cfg$cars
83 } else if (key == "PARK") {
84 cfg$cars + 1
85 } else {
86 1
87 }
88 good_count <- if (key == "COMPARE") {
89 length(words) >= 1 && length(words) <= 4
90 } else {
91 length(words) == expected
92 }
93 if (!good_count) {
94 config_error(paste("wrong number of values for", key))
95 }
96 values <- if (!key %in% c("POLICY", "COMPARE", "ZONES")) {
97 as.numeric(words)
98 } else {
99 numeric()
100 }
101 good <- TRUE
102 if (key == "FLOORS") {
103 cfg$floors <- values[1]
104 good <- bounded(values, 2, 50)
105 } else if (key == "CARS") {
106 cfg$cars <- values[1]
107 good <- bounded(values, 1, 8)
108 } else if (key == "CAPACITY") {
109 cfg$capacity <- values[1]
110 good <- bounded(values, 1, 20)
111 } else if (key == "DOOR") {
112 cfg$door <- values[1]
113 good <- bounded(values, 1, 9)
114 } else if (key == "START") {
115 cfg$starts <- values
116 good <- bounded(values, 1, cfg$floors)
117 } else if (key == "POLICY") {
118 cfg$policy <- words[1]
119 good <- words[1] %in% policies
120 } else if (key == "ZONES") {
121 cfg$zones <- lapply(strsplit(words, ":", fixed = TRUE), as.numeric)
122 good <- all(vapply(
123 cfg$zones,
124 function(z) 1 <= z[1] && z[1] < z[2] && z[2] <= cfg$floors,
125 logical(1)
126 ))
127 } else if (key == "PARK") {
128 cfg$park <- values[1]
129 cfg$homes <- values[-1]
130 good <- bounded(cfg$park, 0, 100) && bounded(cfg$homes, 1, cfg$floors)
131 } else if (key == "PATIENCE") {
132 cfg$patience <- values[1]
133 good <- bounded(values, 1, 1000)
134 } else if (key == "BYPASS") {
135 cfg$bypass <- values[1]
136 good <- bounded(values, 50, 100)
137 } else if (key == "PERIOD") {
138 cfg$period <- values[1]
139 good <- bounded(values, 1, 1000)
140 } else if (key == "LIMIT") {
141 cfg$limit <- values[1]
142 good <- bounded(values, 1, 5000)
143 } else {
144 cfg$compare <- words
145 good <- all(words %in% policies) && !anyDuplicated(words)
146 }
147 if (!good) config_error(paste("bad value for", key))
148 }
149 if (!length(cfg$zones)) {
150 cfg$zones <- rep(list(c(1, cfg$floors)), cfg$cars)
151 }
152 for (i in seq_len(cfg$cars)) {
153 zone <- cfg$zones[[i]]
154 if (!bounded(cfg$starts[i], zone[1], zone[2])) {
155 config_error(paste("car", i, "starts outside its zone"))
156 }
157 if (cfg$park >= 0 && !bounded(cfg$homes[i], zone[1], zone[2])) {
158 config_error(paste("car", i, "home outside its zone"))
159 }
160 }
161 cfg

195validate_events <- function(cfg, lines) {
196 last <- groups <- 0
197 fire <- FALSE
198 out <- rep(FALSE, cfg$cars)
199 accepted <- list()
200 errors <- character()
201 serves <- function(a, b) {
202 any(vapply(
203 cfg$zones,
204 function(z) z[1] <= a && a <= z[2] && z[1] <= b && b <= z[2],
205 logical(1)
206 ))
207 }
208 for (line in lines) {
209 w <- line$words
210 time <- as.numeric(w[1])
211 command <- w[2]
212 args <- w[-c(1, 2)]
213 message <- ""
214 values <- numeric()
215 if (!command %in% names(argument_counts)) {
216 message <- paste("unknown command", command)
217 } else if (length(args) != argument_counts[[command]]) {
218 message <- paste("wrong number of arguments for", command)
219 } else if (time < last) {
220 message <- "time goes backwards"
221 } else {
222 values <- as.numeric(args)
223 if (command == "CALL") {
224 a <- values[1]
225 b <- values[2]
226 n <- values[3]
227 if (a < 1 || a > cfg$floors || b < 1 || b > cfg$floors) {
228 message <- "bad floor"
229 } else if (a == b) {
230 message <- "same floor"
231 } else if (n < 1 || n > 99) {
232 message <- "bad count"
233 } else if (!serves(a, b)) {
234 message <- paste0("no car serves ", a, "->", b)
235 } else if (fire) {
236 message <- "fire service active"
237 }
238 } else if (command %in% c("CANCEL", "WHERE")) {
239 if (values[1] < 1 || values[1] > groups) message <- "unknown group"
240 } else if (command %in% c("OUT", "IN")) {
241 e <- values[1]
242 if (e < 1 || e > cfg$cars) {
243 message <- "bad car"
244 } else if (command == "OUT" && out[e]) {
245 message <- paste("car", e, "already out")
246 } else if (command == "IN" && !out[e]) {
247 message <- paste("car", e, "already in")
248 }
249 } else if (command == "HOLD") {
250 if (values[1] < 1 || values[1] > cfg$cars) {
251 message <- "bad car"
252 } else if (values[2] < 1 || values[2] > 9) {
253 message <- "bad ticks"
254 }
255 } else if (command == "FIRE") {
256 if (values[1] < 1 || values[1] > cfg$floors) {
257 message <- "bad floor"
258 } else if (fire) {
259 message <- "fire already active"
260 }
261 } else if (command == "RESET" && !fire) {
262 message <- "fire not active"
263 }
264 }
265 if (nzchar(message)) {
266 errors <- c(errors, paste0("line ", line$number, ": ", message))
267 next
268 }
269 last <- time
270 if (command == "CALL") {
271 groups <- groups + 1
272 } else if (command == "OUT") {
273 out[values[1]] <- TRUE
274 } else if (command == "IN") {
275 out[values[1]] <- FALSE
276 } else if (command == "FIRE") {
277 fire <- TRUE
278 } else if (command == "RESET") {
279 fire <- FALSE
280 }
281 accepted[[length(accepted) + 1]] <- list(
282 time = time,
283 command = command,
284 values = values
285 )
286 }
287 list(events = accepted, errors = errors)
288}

290simulate <- function(cfg, events, policy, logging = TRUE) {
291 s <- new.env(parent = emptyenv())
292 s$groups <- list()
293 s$pending <- numeric()
294 s$fire <- 0
295 s$round <- 0
296 s$time <- 0
297 s$log <- character()
298 s$status <- ""
299 s$abandoned <- s$evacuated <- s$delivered <- 0
300 s$wait <- s$travel <- s$max_wait <- 0
301 s$cars <- lapply(seq_len(cfg$cars), function(i) {
302 car <- new.env(parent = emptyenv())
303 car$number <- i
304 car$floor <- cfg$starts[i]
305 car$zone <- cfg$zones[[i]]
306 car$home <- if (cfg$park < 0) 0 else cfg$homes[i]
307 car$direction <- car$door <- car$idle <- 0

314 emit <- function(...) {
315 if (logging) s$log <- c(s$log, paste(s$time, paste(..., collapse = " ")))
316 }
317 load <- function(car) {
318 sum(vapply(car$passengers, function(p) p$count, numeric(1)))
319 }
320 targets <- function(car) {
321 unique(c(
322 vapply(
323 car$passengers,
324 function(p) s$groups[[p$group]]$destination,
325 numeric(1)
326 ),
327 vapply(car$calls, function(id) s$groups[[id]]$landing, numeric(1))
328 ))
329 }
330 bypassing <- function(car) load(car) * 100 >= cfg$capacity * cfg$bypass
331 cost <- function(car, group) {
332 f <- car$floor
333 d <- car$direction
334 x <- group$landing
335 if (policy == "NEAREST" || d == 0) {
336 return(abs(f - x))
337 }
338 stops <- targets(car)
339 if (group$direction == d && (x - f) * d >= 0) {
340 distance <- abs(x - f)
341 via <- sum((stops - f) * d > 0 & (x - stops) * d > 0)
342 } else {
343 turn <- if (d > 0) max(c(stops, f)) else min(c(stops, f))
344 distance <- abs(f - turn) + abs(turn - x)
345 via <- sum((stops - f) * d > 0)
346 }
347 if (policy == "COST") distance else distance + cfg$door * via + car$door
348 }
349 assign_call <- function(group, excluded = 0) {
350 candidates <- Filter(
351 function(car) {
352 car$service &&
353 car$number != excluded &&
354 car$zone[1] <= group$landing &&
355 group$landing <= car$zone[2] &&
356 car$zone[1] <= group$destination &&
357 group$destination <= car$zone[2]
358 },
359 s$cars
360 )
361 if (!length(candidates)) {
362 return(FALSE)
363 }
364 if (policy == "ROUND") {
365 after <- which(vapply(
366 candidates,
367 function(car) car$number > s$round,
368 logical(1)
369 ))
370 chosen <- candidates[[if (length(after)) after[1] else 1]]
371 s$round <- chosen$number
372 emit(paste0("E", chosen$number), "ASSIGN", paste0("G", group$number))
373 } else {
374 costs <- vapply(candidates, function(car) cost(car, group), numeric(1))
375 i <- which.min(costs)
376 chosen <- candidates[[i]]
377 emit(
378 paste0("E", chosen$number),
379 "ASSIGN",
380 paste0("G", group$number),
381 paste0("cost=", costs[i])
382 )
383 }
384 if (group$car > 0) {
385 old <- s$cars[[group$car]]
386 old$calls <- old$calls[old$calls != group$number]
387 }
388 chosen$calls <- c(chosen$calls, group$number)
389 group$car <- chosen$number
390 TRUE
391 }
392 dispatch <- function(group) {
393 if (!assign_call(group)) {
394 s$pending <- c(s$pending, group$number)
395 emit("PENDING", paste0("G", group$number))
396 }

398 walk_away <- function(group, word) {
399 if (group$car > 0) {
400 car <- s$cars[[group$car]]
401 car$calls <- car$calls[car$calls != group$number]
402 group$car <- 0
403 } else {
404 s$pending <- s$pending[s$pending != group$number]
405 }
406 emit(word, paste0("G", group$number), paste0("x", group$waiting))
407 group$lost <- group$lost + group$waiting
408 s$abandoned <- s$abandoned + group$waiting
409 group$waiting <- 0
410 }
411 execute <- function(event) {
412 v <- event$values
413 command <- event$command
414 if (command == "CALL") {
415 g <- new.env(parent = emptyenv())
416 g$number <- length(s$groups) + 1
417 g$landing <- v[1]
418 g$destination <- v[2]
419 g$count <- g$waiting <- v[3]
420 g$time <- s$time
421 g$direction <- sign(v[2] - v[1])
422 g$car <- g$boarded <- g$delivered <- g$lost <- 0
423 g$last_board <- g$last_alight <- g$wait <- g$travel <- 0
424 g$cars <- numeric()
425 s$groups[[g$number]] <- g
426 emit(
427 "CALL",
428 paste0("G", g$number),
429 paste0(g$landing, "->", g$destination),
430 paste0("x", g$count)
431 )
432 dispatch(g)
433 } else if (command == "CANCEL") {
434 g <- s$groups[[v[1]]]
435 if (g$waiting > 0) {
436 walk_away(g, "CANCEL")
437 } else {
438 emit("CANCEL", paste0("G", g$number), "ignored")
439 }
440 } else if (command == "OUT") {
441 car <- s$cars[[v[1]]]
442 car$service <- FALSE
443 emit(paste0("E", car$number), "OUT")
444 calls <- car$calls
445 car$calls <- numeric()
446 for (id in calls) {
447 g <- s$groups[[id]]
448 g$car <- 0
449 dispatch(g)
450 }
451 } else if (command == "IN") {
452 car <- s$cars[[v[1]]]
453 car$service <- TRUE
454 car$idle <- s$time
455 emit(paste0("E", car$number), "IN")
456 pending <- s$pending
457 s$pending <- numeric()
458 for (id in pending) {
459 if (!assign_call(s$groups[[id]])) s$pending <- c(s$pending, id)
460 }
461 } else if (command == "HOLD") {
462 car <- s$cars[[v[1]]]
463 if (car$door > 0) {
464 car$door <- car$door + v[2]
465 emit(paste0("E", car$number), "HOLD", car$door)
466 } else {
467 emit(paste0("E", car$number), "HOLD", "ignored")
468 }
469 } else if (command == "FIRE") {
470 s$fire <- v[1]
471 emit("FIRE", s$fire)
472 for (g in s$groups) {
473 if (g$waiting > 0) walk_away(g, "ABANDON")
474 }
475 } else if (command == "RESET") {
476 s$fire <- 0
477 emit("RESET")
478 for (car in s$cars) {
479 car$idle <- s$time
480 }
481 } else if (command == "WHERE") {
482 g <- s$groups[[v[1]]]
483 place <- if (!g$waiting) {
484 "-"
485 } else if (g$car) {
486 paste0("E", g$car)
487 } else {
488 "pending"
489 }
490 riding <- character()
491 for (car in s$cars) {
492 count <- sum(vapply(
493 car$passengers,
494 function(p) if (p$group == g$number) p$count else 0,
495 numeric(1)
496 ))
497 if (count > 0) riding <- c(riding, paste0("E", car$number, "x", count))
498 }
499 emit(
500 "WHERE",
501 paste0("G", g$number),
502 paste0("waiting=", g$waiting),
503 paste0("car=", place),
504 paste0("riding=", joined(riding)),
505 paste0("delivered=", g$delivered),
506 paste0("lost=", g$lost)
507 )
508 } else {
509 for (car in s$cars) {
510 stops <- sort(unique(vapply(
511 car$passengers,
512 function(p) s$groups[[p$group]]$destination,
513 numeric(1)
514 )))
515 direction <- if (car$direction > 0) {
516 "up"
517 } else if (car$direction < 0) {
518 "down"
519 } else {
520 "idle"
521 }
522 emit(
523 "STATUS",
524 paste0("E", car$number),
525 paste0("floor=", car$floor),
526 paste0("dir=", direction),
527 paste0("door=", car$door),
528 paste0("load=", load(car)),
529 paste0("stops=", joined(stops)),
530 paste0(
531 "calls=",
532 joined(
533 if (length(car$calls)) paste0("G", car$calls) else character()
534 )
535 ),
536 paste0("service=", if (car$service) "in" else "out")
537 )
538 }
539 emit(
540 "STATUS",
541 paste0(
542 "pending=",
543 joined(if (length(s$pending)) paste0("G", s$pending) else character())
544 )
545 )
546 }
547 }
548 choose_direction <- function(car) {
549 f <- car$floor
550 d <- car$direction
551 stops <- targets(car)
552 here <- if (bypassing(car)) {
553 numeric()
554 } else {
555 Filter(function(id) s$groups[[id]]$landing == f, car$calls)
556 }
557 if (
558 d != 0 &&
559 (any((stops - f) * d > 0) ||
560 any(vapply(
561 here,
562 function(id) s$groups[[id]]$direction == d,
563 logical(1)
564 )))
565 ) {
566 return(d)
567 }
568 if (length(here)) {
569 return(s$groups[[here[1]]]$direction)
570 }
571 if (d != 0 && any((stops - f) * d < 0)) {
572 return(-d)
573 }
574 if (d == 0 && length(car$calls)) {
575 return(sign(s$groups[[car$calls[1]]]$landing - f))
576 }
577 0
578 }
579 deliver <- function(p) {
580 g <- s$groups[[p$group]]
581 wait <- p$boarded - g$time
582 travel <- s$time - g$time
583 g$delivered <- g$delivered + p$count
584 g$last_alight <- s$time
585 g$wait <- g$wait + wait * p$count
586 g$travel <- g$travel + travel * p$count
587 s$delivered <- s$delivered + p$count
588 s$wait <- s$wait + wait * p$count
589 s$travel <- s$travel + travel * p$count
590 s$max_wait <- max(s$max_wait, wait)
591 }
592 open_doors <- function(car) {
593 emit(paste0("E", car$number), "OPEN", car$floor)
594 car$door <- cfg$door
595 car$stops <- car$stops + 1
596 car$energy <- car$energy + 2
597 riding <- list()
598 for (p in car$passengers) {
599 g <- s$groups[[p$group]]
600 if (g$destination == car$floor) {
601 emit(
602 paste0("E", car$number),
603 "ALIGHT",
604 paste0("G", g$number),
605 paste0("x", p$count)
606 )
607 deliver(p)
608 } else if (s$fire > 0) {
609 emit(
610 paste0("E", car$number),
611 "EVACUATE",
612 paste0("G", g$number),
613 paste0("x", p$count)
614 )
615 g$lost <- g$lost + p$count
616 s$evacuated <- s$evacuated + p$count
617 } else {
618 riding[[length(riding) + 1]] <- p
619 }
620 }
621 car$passengers <- riding
622 if (s$fire > 0) {
623 return(invisible(NULL))
624 }
625 heading <- choose_direction(car)
626 if (heading != 0) {
627 car$direction <- heading
628 }
629 boarding <- Filter(
630 function(id) {
631 g <- s$groups[[id]]
632 g$landing == car$floor && g$direction == heading
633 },
634 car$calls
635 )
636 for (id in boarding) {
637 g <- s$groups[[id]]
638 count <- min(cfg$capacity - load(car), g$waiting)
639 if (count > 0) {
640 car$passengers[[length(car$passengers) + 1]] <- list(
641 group = id,
642 count = count,
643 boarded = s$time
644 )
645 car$boarded <- car$boarded + count
646 g$waiting <- g$waiting - count
647 g$boarded <- g$boarded + count
648 g$last_board <- s$time
649 g$cars <- unique(c(g$cars, car$number))
650 emit(
651 paste0("E", car$number),
652 "BOARD",
653 paste0("G", id),
654 paste0("x", count)
655 )
656 }
657 if (!g$waiting) {
658 car$calls <- car$calls[car$calls != id]
659 g$car <- 0
660 } else {
661 emit(
662 paste0("E", car$number),
663 "LEFT",
664 paste0("G", id),
665 paste0("x", g$waiting)
666 )
667 assign_call(g, car$number)
668 }
669 }

671 move <- function(car, step) {
672 car$energy <- car$energy + if (step > 0) 3 + load(car) else 1
673 car$distance <- car$distance + 1
674 car$floor <- car$floor + step
675 }
676 advance <- function(car) {
677 previous_floor <- car$floor
678 if (car$door > 0) {
679 car$door <- car$door - 1
680 }
681 if (!car$door) {
682 if (s$fire > 0) {
683 heading <- sign(s$fire - car$floor)
684 stopping <- heading == 0 && length(car$passengers) > 0
685 } else {
686 heading <- choose_direction(car)
687 stopping <- any(vapply(
688 car$passengers,
689 function(p) s$groups[[p$group]]$destination == car$floor,
690 logical(1)
691 )) ||
692 (!bypassing(car) &&
693 any(vapply(
694 car$calls,
695 function(id) {
696 g <- s$groups[[id]]
697 g$landing == car$floor && g$direction == heading
698 },
699 logical(1)
700 )))
701 }
702 if (stopping) {
703 open_doors(car)
704 } else if (heading != 0) {
705 car$direction <- heading
706 move(car, heading)
707 } else {
708 if (car$direction != 0) {
709 emit(paste0("E", car$number), "IDLE", car$floor)
710 car$direction <- 0
711 car$idle <- s$time
712 }
713 if (
714 !s$fire &&
715 car$service &&
716 car$home > 0 &&
717 car$floor != car$home &&
718 s$time - car$idle >= cfg$park
719 ) {
720 move(car, sign(car$home - car$floor))
721 if (car$floor == car$home) {
722 emit(paste0("E", car$number), "PARK", car$home)
723 }
724 }
725 }
726 }
727 if (car$door > 0 || car$floor != previous_floor) car$busy <- car$busy + 1
728 }
729 settled <- function(car) {
730 if (car$door > 0 || length(car$passengers) || length(car$calls)) {
731 return(FALSE)
732 }
733 if (s$fire > 0) {
734 car$floor == s$fire
735 } else {
736 !car$service || !car$home || car$floor == car$home
737 }
738 }
739 index <- 1
740 repeat {
741 if (cfg$patience >= 0) {
742 for (g in s$groups) {
743 if (g$waiting > 0 && s$time == g$time + cfg$patience) {
744 walk_away(g, "ABANDON")
745 }
746 }
747 }
748 while (index <= length(events) && events[[index]]$time == s$time) {
749 execute(events[[index]])
750 index <- index + 1
751 }
752 for (car in s$cars) {
753 advance(car)
754 }
755 if (index > length(events) && all(vapply(s$cars, settled, logical(1)))) {
756 s$status <- "end"
757 emit("END")
758 break
759 }
760 if (s$time == cfg$limit) {
761 s$status <- "abort"
762 emit("ABORT")
763 break
764 }
765 s$time <- s$time + 1
766 }
767 s
768}

884outcome <- function(scenario) {
885 s <- scenario$simulation
886 if (is.null(s)) {
887 "invalid"
888 } else if (s$status == "abort") {
889 "aborted"
890 } else if (!s$delivered) {
891 "empty"
892 } else {
893 "ranked"
894 }
895}
896lines <- readLines(file("stdin"), warn = FALSE)
897index <- 2
898output <- character()
899scenarios <- list()
900for (q in seq_len(as.integer(lines[1]))) {
901 name <- strsplit(lines[index], " ", fixed = TRUE)[[1]][2]
902 index <- index + 1
903 settings <- event_lines <- list()
904 while (lines[index] != "END") {
905 words <- strsplit(lines[index], " ", fixed = TRUE)[[1]]
906 if (substr(words[1], 1, 1) %in% as.character(0:9)) {
907 event_lines[[length(event_lines) + 1]] <- list(
908 number = index,
909 words = words
910 )
911 } else {
912 settings[[length(settings) + 1]] <- words
913 }
914 index <- index + 1
915 }
916 index <- index + 1
917 scenario <- list(name = name, config = NULL, simulation = NULL)
918 output <- c(output, paste0("== ", name, " =="))
919 parsed <- tryCatch(
920 list(config = parse_config(settings)),
921 config_error = function(e) list(error = conditionMessage(e))
922 )
923 if (!is.null(parsed$error)) {
924 output <- c(output, paste0("error: ", parsed$error))
925 } else {
926 cfg <- parsed$config
927 scenario$config <- cfg
928 checked <- validate_events(cfg, event_lines)
929 output <- c(output, config_header(cfg), checked$errors)
930 s <- simulate(cfg, checked$events, cfg$policy)
931 scenario$simulation <- s
932 output <- c(output, s$log, report(s, cfg))
933 for (policy in cfg$compare) {
934 rerun <- simulate(cfg, checked$events, policy, FALSE)
935 output <- c(
936 output,
937 sprintf(
938 "compare %s: delivered=%.0f %s end=%.0f status=%s",
939 policy,
940 rerun$delivered,
941 averages(rerun),
942 rerun$time,
943 rerun$status
944 )
945 )
946 }
947 }
948 scenarios[[q]] <- scenario
949}
950output <- c(output, "== ranking ==")
951ranked <- Filter(function(x) outcome(x) == "ranked", scenarios)
952if (length(ranked) > 1) {
953 for (i in 2:length(ranked)) {
954 j <- i
955 while (j > 1) {
956 a <- ranked[[j]]$simulation
957 b <- ranked[[j - 1]]$simulation
958 left <- a$wait * b$delivered
959 right <- b$wait * a$delivered
960 before <- left < right ||
961 (left == right && a$travel * b$delivered < b$travel * a$delivered)
962 if (!before) {
963 break
964 }
965 temp <- ranked[[j - 1]]
966 ranked[[j - 1]] <- ranked[[j]]
967 ranked[[j]] <- temp
968 j <- j - 1
969 }
970 }
971}
972for (i in seq_along(ranked)) {
973 x <- ranked[[i]]
974 output <- c(
975 output,
976 paste0(
977 i,
978 ". ",
979 x$name,
980 " policy=",
981 x$config$policy,
982 " ",
983 averages(x$simulation)
984 )
985 )
986}
987for (x in scenarios) {
988 if (outcome(x) != "ranked") {
989 output <- c(output, paste("-", x$name, outcome(x)))
990 }
991}
992output <- c(output, "== policies ==")
993for (policy in policies) {
994 runs <- Filter(function(x) x$config$policy == policy, ranked)
995 if (!length(runs)) {
996 next
997 }
998 count <- sum(vapply(runs, function(x) x$simulation$delivered, numeric(1)))
999 wait <- sum(vapply(runs, function(x) x$simulation$wait, numeric(1)))
1000 travel <- sum(vapply(runs, function(x) x$simulation$travel, numeric(1)))
1001 output <- c(
1002 output,
1003 sprintf(
1004 "%s: runs=%d avg_wait=%s avg_travel=%s",
1005 policy,
1006 length(runs),
1007 average(wait, count),
1008 average(travel, count)
1009 )
1010 )
1011}
1012cat(paste(output, collapse = "\n"), "\n", sep = "")
