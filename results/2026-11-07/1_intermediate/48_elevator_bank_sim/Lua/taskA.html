53local function parse_config(lines)
54 local entries={}
55 for _,words in ipairs(lines) do
56 local k=words[1]
57 check(not contains(CONFIG_KEYS,k),'unknown key '..k)
58 check(entries[k]~=nil,'duplicate key '..k)
59 local vs={}
60 for i=2,#words do
61 vs[#vs+1]=words[i]
62 end
63 entries[k]=vs
64 end
65 for i=1,5 do
66 local k=CONFIG_KEYS[i]
67 check(not entries[k],'missing key '..k)
68 end
69 local c={POLICY='COST',BYPASS=100,LIMIT=1000,COMPARE={}}
70 local ranges={FLOORS={2,50},CARS={1,8},CAPACITY={1,20},DOOR={1,9},PATIENCE={1,1000},BYPASS={50,100},PERIOD={1,1000},LIMIT={1,5000}}
71 for _,k in ipairs(CONFIG_KEYS) do
72 if entries[k] then
73 local vs=entries[k]
74 local count=#vs
75 local valid_count
76 if k=='START' or k=='ZONES' then
77 valid_count=count==c.CARS
78 elseif k=='PARK' then
79 valid_count=count==c.CARS+1
80 elseif k=='COMPARE' then
81 valid_count=count>=1 and count<=4
82 else
83 valid_count=count==1
84 end
85 check(not valid_count,'wrong number of values for '..k)
86 local valid=true
87 if ranges[k] then
88 local v=tonumber(vs[1])
89 valid=v>=ranges[k][1] and v<=ranges[k][2]
90 c[k]=v
91 elseif k=='POLICY' then
92 valid=contains(POLICIES,vs[1])
93 c[k]=vs[1]
94 elseif k=='COMPARE' then
95 local seen={}
96 for _,v in ipairs(vs) do
97 if not contains(POLICIES,v) or seen[v] then
98 valid=false
99 end
100 seen[v]=true
101 end
102 c[k]=vs
103 elseif k=='ZONES' then
104 local zs={}
105 for _,v in ipairs(vs) do
106 local a,b=v:match('^([^:]+):([^:]+)$')
107 a,b=tonumber(a),tonumber(b)
108 valid=valid and 1<=a and a<b and b<=c.FLOORS
109 zs[#zs+1]={a,b}
110 end
111 c[k]=zs
112 else
113 local ns={}
114 for i,text in ipairs(vs) do
115 local n=tonumber(text)
116 ns[#ns+1]=n
117 if k=='PARK' and i==1 then
118 valid=valid and n>=0 and n<=100
119 else
120 valid=valid and n>=1 and n<=c.FLOORS
121 end
122 end
123 c[k]=ns
124 end
125 check(not valid,'bad value for '..k)
126 end
127 end
128 if not c.ZONES then
129 c.ZONES={}
130 for _=1,c.CARS do
131 c.ZONES[#c.ZONES+1]={1,c.FLOORS}
132 end
133 end
134 for i=1,c.CARS do
135 local a,b=c.ZONES[i][1],c.ZONES[i][2]
136 check(c.START[i]<a or c.START[i]>b,'car '..i..' starts outside its zone')
137 if c.PARK then
138 check(c.PARK[i+1]<a or c.PARK[i+1]>b,'car '..i..' home outside its zone')
139 end
140 end
141 return c
142end
143local function validate_events(c,lines)
144 local accepted,last,groups,out,fire={},0,0,{},false
145 for _,entry in ipairs(lines) do
146 local line,words=entry[1],entry[2]
147 local time,cmd=tonumber(words[1]),words[2]
148 local v={}
149 for i=3,#words do
150 v[#v+1]=tonumber(words[i]) or words[i]
151 end
152 local message=nil
153 if not ARG_COUNTS[cmd] then
154 message='unknown command '..cmd
155 elseif #v~=ARG_COUNTS[cmd] then
156 message='wrong number of arguments for '..cmd
157 elseif time<last then
158 message='time goes backwards'
159 elseif cmd=='CALL' then
160 local a,b,n=v[1],v[2],v[3]
161 local serves=false
162 for _,z in ipairs(c.ZONES) do
163 if z[1]<=a and a<=z[2] and z[1]<=b and b<=z[2] then
164 serves=true
165 end
166 end
167 if a<1 or a>c.FLOORS or b<1 or b>c.FLOORS then
168 message='bad floor'
169 elseif a==b then
170 message='same floor'
171 elseif n<1 or n>99 then
172 message='bad count'
173 elseif not serves then
174 message='no car serves '..a..'->'..b
175 elseif fire then
176 message='fire service active'
177 end
178 elseif cmd=='CANCEL' or cmd=='WHERE' then
179 if v[1]<1 or v[1]>groups then
180 message='unknown group'
181 end
182 elseif cmd=='OUT' or cmd=='IN' then
183 local e=v[1]
184 if e<1 or e>c.CARS then
185 message='bad car'
186 elseif cmd=='OUT' and out[e] then
187 message='car '..e..' already out'
188 elseif cmd=='IN' and not out[e] then
189 message='car '..e..' already in'
190 end
191 elseif cmd=='HOLD' then
192 if v[1]<1 or v[1]>c.CARS then
193 message='bad car'
194 elseif v[2]<1 or v[2]>9 then
195 message='bad ticks'
196 end
197 elseif cmd=='FIRE' then
198 if v[1]<1 or v[1]>c.FLOORS then
199 message='bad floor'
200 elseif fire then
201 message='fire already active'
202 end
203 elseif cmd=='RESET' and not fire then
204 message='fire not active'
205 end
206 if message then
207 emit('line '..line..': '..message)
208 else
209 accepted[#accepted+1]={time,cmd,v}
210 last=time
211 if cmd=='CALL' then
212 groups=groups+1
213 elseif cmd=='OUT' then
214 out[v[1]]=true
215 elseif cmd=='IN' then
216 out[v[1]]=nil
217 elseif cmd=='FIRE' then
218 fire=true
219 elseif cmd=='RESET' then
220 fire=false
221 end
222 end
223 end
224 return accepted
225end

235local function load(car)

242local function targets(car)
243 local set={}
244 for _,p in ipairs(car.passengers) do
245 set[p.group.b]=true
246 end
247 for _,g in ipairs(car.calls) do
248 set[g.a]=true
249 end
250 return keys(set)
251end
252local function serves(car,g)
253 return car.zone[1]<=g.a and g.a<=car.zone[2] and car.zone[1]<=g.b and g.b<=car.zone[2]
254end

262function Sim:bypass(car)
263 return load(car)*100>=self.c.CAPACITY*self.c.BYPASS

265function Sim:cost(car,g)
266 local f,d,x=car.floor,car.direction,g.a
267 if self.policy=='NEAREST' or d==0 then
268 return math.abs(f-x)
269 end
270 local ts=targets(car)
271 local distance,via=0,0
272 if direction(g)==d and (x-f)*d>=0 then
273 distance=math.abs(x-f)
274 for _,y in ipairs(ts) do
275 if (y-f)*d>0 and (x-y)*d>0 then
276 via=via+1
277 end
278 end
279 else
280 local m=f
281 for _,y in ipairs(ts) do
282 if d>0 then
283 m=math.max(m,y)
284 else
285 m=math.min(m,y)
286 end
287 if (y-f)*d>0 then
288 via=via+1
289 end
290 end
291 distance=math.abs(f-m)+math.abs(m-x)
292 end
293 return distance+(self.policy=='ETA' and self.c.DOOR*via+car.door or 0)
294end
295function Sim:assign(g,excluded)
296 local candidates={}
297 for _,car in ipairs(self.cars) do
298 if car.service and car~=excluded and serves(car,g) then
299 candidates[#candidates+1]=car
300 end
301 end
302 if #candidates==0 then
303 return false
304 end
305 local chosen=candidates[1]
306 if self.policy=='ROUND' then
307 for _,car in ipairs(candidates) do
308 if car.number>self.round then
309 chosen=car
310 break
311 end
312 end
313 self.round=chosen.number
314 self:log(car_label(chosen),'ASSIGN',group_label(g))
315 else
316 local cost=self:cost(chosen,g)
317 for i=2,#candidates do
318 local car=candidates[i]
319 local v=self:cost(car,g)
320 if v<cost then
321 chosen,cost=car,v
322 end
323 end
324 self:log(car_label(chosen),'ASSIGN',group_label(g),'cost='..cost)
325 end
326 if g.car then
327 remove(g.car.calls,g)
328 end
329 chosen.calls[#chosen.calls+1]=g
330 g.car=chosen
331 return true
332end
333function Sim:dispatch(g)
334 if not self:assign(g) then
335 self.pending[#self.pending+1]=g
336 self:log('PENDING',group_label(g))
337 end
338end
339function Sim:walk(g,word)
340 if g.car then
341 remove(g.car.calls,g)
342 else
343 remove(self.pending,g)
344 end
345 g.car=nil
346 self:log(word,group_label(g),'x'..g.waiting)
347 g.lost=g.lost+g.waiting
348 self.abandoned=self.abandoned+g.waiting
349 g.waiting=0
350end
351function Sim:event(cmd,v)
352 if cmd=='CALL' then
353 local a,b,n=v[1],v[2],v[3]
354 local g={number=#self.groups+1,a=a,b=b,count=n,time=self.time,waiting=n,boarded=0,delivered=0,lost=0,last_board=0,last_alight=0,cars={},wait_total=0,travel_total=0}
355 self.groups[#self.groups+1]=g
356 self:log(cmd,group_label(g),a..'->'..b,'x'..n)
357 self:dispatch(g)
358 elseif cmd=='CANCEL' then
359 local g=self.groups[v[1]]
360 if g.waiting>0 then
361 self:walk(g,cmd)
362 else
363 self:log(cmd,group_label(g),'ignored')
364 end
365 elseif cmd=='OUT' then
366 local car=self.cars[v[1]]
367 car.service=false
368 self:log(car_label(car),cmd)
369 local calls=car.calls
370 car.calls={}
371 for _,g in ipairs(calls) do
372 g.car=nil
373 self:dispatch(g)
374 end
375 elseif cmd=='IN' then
376 local car=self.cars[v[1]]
377 car.service=true
378 car.idle=self.time
379 self:log(car_label(car),cmd)
380 local pending=self.pending
381 self.pending={}
382 for _,g in ipairs(pending) do
383 if not self:assign(g) then
384 self.pending[#self.pending+1]=g
385 end
386 end
387 elseif cmd=='HOLD' then
388 local car=self.cars[v[1]]
389 if car.door>0 then
390 car.door=car.door+v[2]
391 self:log(car_label(car),cmd,car.door)
392 else
393 self:log(car_label(car),cmd,'ignored')
394 end
395 elseif cmd=='FIRE' then
396 self.fire=v[1]
397 self:log(cmd,self.fire)
398 for _,g in ipairs(self.groups) do
399 if g.waiting>0 then
400 self:walk(g,'ABANDON')
401 end
402 end
403 elseif cmd=='RESET' then
404 self.fire=nil
405 self:log(cmd)
406 for _,car in ipairs(self.cars) do
407 car.idle=self.time
408 end
409 elseif cmd=='WHERE' then
410 local g=self.groups[v[1]]
411 local place=g.waiting==0 and '-' or g.car and car_label(g.car) or 'pending'
412 local riding={}
413 for _,car in ipairs(self.cars) do
414 local n=0
415 for _,p in ipairs(car.passengers) do
416 if p.group==g then
417 n=n+p.count
418 end
419 end
420 if n>0 then
421 riding[#riding+1]=car_label(car)..'x'..n
422 end
423 end
424 self:log(cmd,group_label(g),'waiting='..g.waiting,'car='..place,'riding='..joined(riding),'delivered='..g.delivered,'lost='..g.lost)
425 elseif cmd=='STATUS' then
426 for _,car in ipairs(self.cars) do
427 local set,calls={},{}
428 for _,p in ipairs(car.passengers) do
429 set[p.group.b]=true
430 end
431 for _,g in ipairs(car.calls) do
432 calls[#calls+1]=group_label(g)
433 end
434 local dir=({[-1]='down',[0]='idle',[1]='up'})[car.direction]
435 self:log(cmd,car_label(car),'floor='..car.floor,'dir='..dir,'door='..car.door,'load='..load(car),'stops='..joined(keys(set)),'calls='..joined(calls),'service='..(car.service and 'in' or 'out'))
436 end
437 local pending={}
438 for _,g in ipairs(self.pending) do
439 pending[#pending+1]=group_label(g)
440 end
441 self:log(cmd,'pending='..joined(pending))
442 end
443end
444function Sim:heading(car)
445 local f,d=car.floor,car.direction
446 local here={}
447 if not self:bypass(car) then
448 for _,g in ipairs(car.calls) do
449 if g.a==f then
450 here[#here+1]=g
451 end
452 end
453 end
454 local ts=targets(car)
455 if d~=0 then
456 for _,y in ipairs(ts) do
457 if (y-f)*d>0 then
458 return d
459 end
460 end
461 for _,g in ipairs(here) do
462 if direction(g)==d then
463 return d
464 end
465 end
466 end
467 if #here>0 then
468 return direction(here[1])
469 end
470 if d~=0 then
471 for _,y in ipairs(ts) do
472 if (y-f)*d<0 then
473 return -d
474 end
475 end
476 end
477 if d==0 and #car.calls>0 then
478 return sign(car.calls[1].a-f)
479 end
480 return 0
481end
482function Sim:move(car,n)
483 car.energy=car.energy+(n>0 and 3+load(car) or 1)
484 car.floor=car.floor+n
485 car.distance=car.distance+1
486end
487function Sim:deliver(p)
488 local g=p.group
489 local w,r=p.board_time-g.time,self.time-g.time
490 g.delivered=g.delivered+p.count
491 g.last_alight=self.time
492 g.wait_total=g.wait_total+w*p.count
493 g.travel_total=g.travel_total+r*p.count
494 self.delivered=self.delivered+p.count
495 self.wait_total=self.wait_total+w*p.count
496 self.travel_total=self.travel_total+r*p.count
497 self.max_wait=math.max(self.max_wait,w)
498end
499function Sim:open(car)
500 self:log(car_label(car),'OPEN',car.floor)
501 car.door=self.c.DOOR
502 car.stops=car.stops+1
503 car.energy=car.energy+2
504 local riding={}
505 for _,p in ipairs(car.passengers) do
506 if p.group.b==car.floor then
507 self:log(car_label(car),'ALIGHT',group_label(p.group),'x'..p.count)
508 self:deliver(p)
509 elseif self.fire then
510 self:log(car_label(car),'EVACUATE',group_label(p.group),'x'..p.count)
511 p.group.lost=p.group.lost+p.count
512 self.evacuated=self.evacuated+p.count
513 else
514 riding[#riding+1]=p
515 end
516 end
517 car.passengers=riding
518 if self.fire then
519 return
520 end
521 local n=self:heading(car)
522 if n~=0 then
523 car.direction=n
524 end
525 local boarding={}
526 for _,g in ipairs(car.calls) do
527 if g.a==car.floor and direction(g)==n then
528 boarding[#boarding+1]=g
529 end
530 end
531 for _,g in ipairs(boarding) do
532 local k=math.min(self.c.CAPACITY-load(car),g.waiting)
533 if k>0 then
534 car.passengers[#car.passengers+1]={group=g,count=k,board_time=self.time}
535 car.boarded=car.boarded+k
536 g.waiting=g.waiting-k
537 g.boarded=g.boarded+k
538 g.last_board=self.time
539 g.cars[car.number]=true
540 self:log(car_label(car),'BOARD',group_label(g),'x'..k)
541 end
542 if g.waiting==0 then
543 remove(car.calls,g)
544 g.car=nil
545 else
546 self:log(car_label(car),'LEFT',group_label(g),'x'..g.waiting)
547 self:assign(g,car)
548 end
549 end
550end
551function Sim:advance(car)
552 local old=car.floor
553 if car.door>0 then
554 car.door=car.door-1
555 end
556 if car.door==0 then
557 local n,stop
558 if self.fire then
559 n=sign(self.fire-car.floor)
560 stop=n==0 and #car.passengers>0
561 else
562 n=self:heading(car)
563 stop=false
564 for _,p in ipairs(car.passengers) do
565 if p.group.b==car.floor then
566 stop=true
567 end
568 end
569 if not self:bypass(car) then
570 for _,g in ipairs(car.calls) do
571 if g.a==car.floor and direction(g)==n then
572 stop=true
573 end
574 end
575 end
576 end
577 if stop then
578 self:open(car)
579 elseif n~=0 then
580 car.direction=n
581 self:move(car,n)
582 else
583 if car.direction~=0 then
584 self:log(car_label(car),'IDLE',car.floor)
585 car.direction=0
586 car.idle=self.time
587 end
588 if not self.fire and car.service and car.home and car.floor~=car.home and self.time-car.idle>=self.c.PARK[1] then
589 self:move(car,sign(car.home-car.floor))
590 if car.floor==car.home then
591 self:log(car_label(car),'PARK',car.home)
592 end
593 end
594 end
595 end
596 if car.door>0 or old~=car.floor then
597 car.busy=car.busy+1
598 end
599end
600function Sim:settled(car)
601 if car.door>0 or #car.passengers>0 or #car.calls>0 then
602 return false
603 end
604 if self.fire then
605 return car.floor==self.fire
606 end
607 return not car.service or not car.home or car.floor==car.home
608end
609function Sim:run()
610 local index=1
611 while true do
612 if self.c.PATIENCE then
613 for _,g in ipairs(self.groups) do
614 if g.waiting>0 and self.time==g.time+self.c.PATIENCE then
615 self:walk(g,'ABANDON')
616 end
617 end
618 end
619 while index<=#self.events and self.events[index][1]==self.time do
620 local e=self.events[index]
621 self:event(e[2],e[3])
622 index=index+1
623 end
624 for _,car in ipairs(self.cars) do
625 self:advance(car)
626 end
627 local settled=true
628 for _,car in ipairs(self.cars) do
629 if not self:settled(car) then
630 settled=false
631 end
632 end
633 if index>#self.events and settled then
634 self.status='end'
635 self:log('END')
636 return
637 end
638 if self.time==self.c.LIMIT then
639 self.status='abort'
640 self:log('ABORT')
641 return
642 end
643 self.time=self.time+1
644 end
645end
646function Sim:averages()
647 return 'avg_wait='..average(self.wait_total,self.delivered)..' avg_travel='..average(self.travel_total,self.delivered)..' max_wait='..(self.delivered>0 and self.max_wait or '-')
648end
649local function breakdown(label,gs)
650 local n,d,w=0,0,0
651 for _,g in ipairs(gs) do
652 n=n+g.count
653 d=d+g.delivered
654 w=w+g.wait_total
655 end
656 emit(label..':','groups='..#gs,'passengers='..n,'delivered='..d,'avg_wait='..average(w,d))
657end
658function Sim:report()
659 local passengers,floors,periods=0,{},{}
660 for _,g in ipairs(self.groups) do
661 passengers=passengers+g.count
662 floors[g.a]=floors[g.a] or {}
663 floors[g.a][#floors[g.a]+1]=g
664 if self.c.PERIOD then
665 local k=g.time//self.c.PERIOD
666 periods[k]=periods[k] or {}
667 periods[k][#periods[k]+1]=g
668 end
669 emit(group_label(g),'t='..g.time,g.a..'->'..g.b,'n='..g.count,'delivered='..g.delivered,'lost='..g.lost,'wait='..(g.boarded==g.count and g.last_board-g.time or '-'),'travel='..(g.delivered==g.count and g.last_alight-g.time or '-'),'cars='..joined(keys(g.cars)))
670 end
671 for _,car in ipairs(self.cars) do
672 local n=car.busy*1000//(self.time+1)
673 emit(car_label(car),'distance='..car.distance,'stops='..car.stops,'boarded='..car.boarded,'busy='..car.busy,'util='..(n//10)..'.'..(n%10)..'%','energy='..car.energy)
674 end
675 for _,f in ipairs(keys(floors)) do
676 breakdown('F'..f,floors[f])
677 end
678 for _,k in ipairs(keys(periods)) do
679 local w=self.c.PERIOD
680 breakdown('P'..k..' '..(k*w)..'-'..(k*w+w-1),periods[k])
681 end
682 emit('summary:','groups='..#self.groups,'passengers='..passengers,'delivered='..self.delivered,'abandoned='..self.abandoned,'evacuated='..self.evacuated,self:averages(),'end='..self.time)
683end
684local function simulate(c,events,policy,logging)
685 local s=setmetatable({c=c,events=events,policy=policy,logging=logging,cars={},groups={},pending={},round=0,time=0,delivered=0,wait_total=0,travel_total=0,max_wait=0,abandoned=0,evacuated=0},Sim)
686 for i=1,c.CARS do
687 s.cars[#s.cars+1]={number=i,floor=c.START[i],zone=c.ZONES[i],home=c.PARK and c.PARK[i+1] or nil,direction=0,door=0,passengers={},calls={},service=true,idle=0,distance=0,stops=0,boarded=0,busy=0,energy=0}
688 end
689 s:run()
690 return s
691end

699local lines={}
700for line in io.lines() do
701 lines[#lines+1]=line
702end
703local index,scenarios=2,{}
704for ordinal=1,tonumber(lines[1]) do
705 local name=words(lines[index])[2]
706 index=index+1
707 local settings,events={},{}
708 while lines[index]~='END' do
709 local ws=words(lines[index])
710 if ws[1]:match('^%d') then
711 events[#events+1]={index,ws}
712 else
713 settings[#settings+1]=ws
714 end
715 index=index+1
716 end
717 index=index+1
718 emit('== '..name..' ==')
719 local ok,c=pcall(parse_config,settings)
720 local scenario={name=name,ordinal=ordinal}
721 scenarios[#scenarios+1]=scenario
722 if not ok then
723 emit('error:',c)
724 else
725 scenario.config=c
726 emit('config:','floors='..c.FLOORS,'cars='..c.CARS,'capacity='..c.CAPACITY,'door='..c.DOOR,'policy='..c.POLICY,'park='..(c.PARK and c.PARK[1] or 'off'),'patience='..(c.PATIENCE or 'off'),'bypass='..c.BYPASS,'period='..(c.PERIOD or 'off'),'limit='..c.LIMIT,'compare='..joined(c.COMPARE))
727 for i=1,c.CARS do
728 emit('E'..i..':','start='..c.START[i],'zone='..c.ZONES[i][1]..'-'..c.ZONES[i][2],'home='..(c.PARK and c.PARK[i+1] or '-'))
729 end
730 local accepted=validate_events(c,events)
731 local s=simulate(c,accepted,c.POLICY,true)
732 s:report()
733 scenario.sim=s
734 for _,p in ipairs(c.COMPARE) do
735 local o=simulate(c,accepted,p,false)
736 emit('compare '..p..':','delivered='..o.delivered,o:averages(),'end='..o.time,'status='..o.status)
737 end
738 end
739end
740emit('== ranking ==')
741local ranked={}
742for _,s in ipairs(scenarios) do
743 if s.sim and s.sim.status=='end' and s.sim.delivered>0 then
744 ranked[#ranked+1]=s
745 end
746end
747table.sort(ranked,function(x,y)
748 local a,b=x.sim,y.sim
749 local w1,w2=a.wait_total*b.delivered,b.wait_total*a.delivered
750 if w1~=w2 then
751 return w1<w2
752 end
753 local t1,t2=a.travel_total*b.delivered,b.travel_total*a.delivered
754 if t1~=t2 then
755 return t1<t2
756 end
757 return x.ordinal<y.ordinal
758end)
759for i,r in ipairs(ranked) do
760 emit(i..'.',r.name,'policy='..r.config.POLICY,r.sim:averages())
761end
762for _,r in ipairs(scenarios) do
763 local s=r.sim
764 if not s or s.status~='end' or s.delivered==0 then
765 emit('-',r.name,not s and 'invalid' or s.status=='abort' and 'aborted' or 'empty')
766 end
767end
768emit('== policies ==')
769for _,policy in ipairs(POLICIES) do
770 local n,d,w,t=0,0,0,0
771 for _,r in ipairs(ranked) do
772 if r.config.POLICY==policy then
773 local s=r.sim
774 n=n+1
775 d=d+s.delivered
776 w=w+s.wait_total
777 t=t+s.travel_total
778 end
779 end
780 if n>0 then
781 emit(policy..':','runs='..n,'avg_wait='..average(w,d),'avg_travel='..average(t,d))
782 end
783end
