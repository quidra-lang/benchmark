25sealed class Config
26{
27 public int Floors, Cars, Capacity, Door, Bypass = 100, Limit = 1000;

33 public static Config Parse(List<string[]> lines)
34 {
35 var entries = new Dictionary<string, string[]>();
36 foreach (string[] w in lines) {
37 Rules.Check(Rules.Keys.Contains(w[0]), $"unknown key {w[0]}");
38 Rules.Check(!entries.ContainsKey(w[0]), $"duplicate key {w[0]}");
39 entries[w[0]] = w.Skip(1).ToArray();
40 }
41 foreach (string k in Rules.Keys.Take(5)) Rules.Check(entries.ContainsKey(k), $"missing key {k}");
42 var c = new Config();
43 foreach (string k in Rules.Keys)
44 {
45 if (!entries.TryGetValue(k, out string[]? a)) continue;
46 bool count = k is "START" or "ZONES" ? a.Length == c.Cars : k == "PARK" ? a.Length == c.Cars + 1 : k == "COMPARE" ? a.Length >= 1 && a.Length <= 4 : a.Length == 1;
47 Rules.Check(count, $"wrong number of values for {k}");
48 int Bound(string s, int lo, int hi) {
49 int n = int.Parse(s);
50 Rules.Check(n >= lo && n <= hi, $"bad value for {k}");
51 return n;
52 }
53 switch (k)
54 {
55 case "FLOORS": c.Floors = Bound(a[0], 2, 50);
56 break;
57 case "CARS": c.Cars = Bound(a[0], 1, 8);
58 break;
59 case "CAPACITY": c.Capacity = Bound(a[0], 1, 20);
60 break;
61 case "DOOR": c.Door = Bound(a[0], 1, 9);
62 break;
63 case "START": c.Starts = a.Select(s => Bound(s, 1, c.Floors)).ToArray();
64 break;
65 case "POLICY": Rules.Check(Rules.Policies.Contains(a[0]), $"bad value for {k}");
66 c.Policy = a[0];
67 break;
68 case "ZONES": c.Zones = a.Select(s => {
69 string[] z = s.Split(':');
70 int lo = int.Parse(z[0]), hi = int.Parse(z[1]);
71 Rules.Check(1 <= lo && lo < hi && hi <= c.Floors, "bad value for ZONES");
72 return (lo, hi);
73 }
74 ).ToArray();
75 break;
76 case "PARK": c.Park = Bound(a[0], 0, 100);
77 c.Homes = a.Skip(1).Select(s => Bound(s, 1, c.Floors)).ToArray();
78 break;
79 case "PATIENCE": c.Patience = Bound(a[0], 1, 1000);
80 break;
81 case "BYPASS": c.Bypass = Bound(a[0], 50, 100);
82 break;
83 case "PERIOD": c.Period = Bound(a[0], 1, 1000);
84 break;
85 case "LIMIT": c.Limit = Bound(a[0], 1, 5000);
86 break;
87 case "COMPARE": Rules.Check(a.All(Rules.Policies.Contains) && a.Distinct().Count() == a.Length, $"bad value for {k}");
88 c.Compare = a;
89 break;
90 }
91 }
92 if (c.Zones.Length == 0) c.Zones = Enumerable.Repeat((1, c.Floors), c.Cars).ToArray();
93 for (int i = 0;
94 i < c.Cars;
95 i++) {
96 Rules.Check(c.Contains(i, c.Starts[i]), $"car {i + 1} starts outside its zone");
97 Rules.Check(c.Park == null || c.Contains(i, c.Homes[i]), $"car {i + 1} home outside its zone");
98 }
99 return c;
100 }
101 public bool Contains(int e, int f) => Zones[e].Low <= f && f <= Zones[e].High;
102 public List<string> Header()
103 {
104 string Opt(int? n) => n?.ToString() ?? "off";
105 var lines = new List<string> {
106 $"config: floors={Floors} cars={Cars} capacity={Capacity} door={Door} policy={Policy} park={Opt(Park)} patience={Opt(Patience)} bypass={Bypass} period={Opt(Period)} limit={Limit} compare={Rules.Join(Compare)}"
107 };
108 for (int i = 0;
109 i < Cars;
110 i++) lines.Add($"E{i + 1}: start={Starts[i]} zone={Zones[i].Low}-{Zones[i].High} home={(Park == null ? "-" : Homes[i].ToString())}");
111 return lines;
112 }

147sealed class Simulation
148{
149 readonly Config config;
150 readonly List<Event> events;
151 readonly string policy;
152 readonly List<Car> cars;
153 readonly List<Group> groups = new();
154 List<Group> pending = new();
155 int? fire;
156 int lastRound, abandoned, evacuated;
157 public int Time, Delivered, MaxWait;
158 public long WaitTotal, TravelTotal;
159 public string Status = "";
160 public readonly List<string> Log = new();
161 public Simulation(Config c, List<Event> e, string p) {
162 config = c;
163 events = e;
164 policy = p;
165 cars = c.Starts.Select((f, i) => new Car(i + 1, f)).ToList();
166 }
167 void Emit(string s) => Log.Add($"{Time} {s}");
168 bool Bypass(Car c) => c.Load * 100 >= config.Capacity * config.Bypass;
169 int Cost(Car c, Group g)
170 {
171 int f = c.Floor, d = c.Direction, x = g.A;
172 if (policy == "NEAREST" || d == 0) return Math.Abs(f - x);
173 HashSet<int> targets = c.Targets;
174 int distance, via;
175 if (g.Direction == d && (x - f) * d >= 0) {
176 distance = Math.Abs(x - f);
177 via = targets.Count(y => (y - f) * d > 0 && (x - y) * d > 0);
178 } else {
179 int m = d > 0 ? targets.Append(f).Max() : targets.Append(f).Min();
180 distance = Math.Abs(f - m) + Math.Abs(m - x);
181 via = targets.Count(y => (y - f) * d > 0);
182 }
183 return policy == "COST" ? distance : distance + config.Door * via + c.Door;
184 }
185 bool Assign(Group g, Car? excluded = null)
186 {
187 var candidates = cars.Where(c => c.Service && c != excluded && config.Contains(c.Number - 1, g.A) && config.Contains(c.Number - 1, g.B)).ToList();
188 if (candidates.Count == 0) return false;
189 Car chosen;
190 if (policy == "ROUND") {
191 chosen = candidates.FirstOrDefault(c => c.Number > lastRound) ?? candidates[0];
192 lastRound = chosen.Number;
193 Emit($"{chosen.Label} ASSIGN {g.Label}");
194 } else {
195 chosen = candidates.OrderBy(c => Cost(c, g)).ThenBy(c => c.Number).First();
196 Emit($"{chosen.Label} ASSIGN {g.Label} cost={Cost(chosen, g)}");
197 }
198 if (g.Car is int old) cars[old].Calls.Remove(g);
199 chosen.Calls.Add(g);
200 g.Car = chosen.Number - 1;
201 return true;
202 }
203 void Dispatch(Group g) {
204 if (!Assign(g)) {
205 pending.Add(g);
206 Emit($"PENDING {g.Label}");
207 }
208 }
209 void Walk(Group g, string word) {
210 if (g.Car is int e) cars[e].Calls.Remove(g);
211 else pending.Remove(g);
212 g.Car = null;
213 Emit($"{word} {g.Label} x{g.Waiting}");
214 g.Lost += g.Waiting;
215 abandoned += g.Waiting;
216 g.Waiting = 0;
217 }
218 void Execute(Event ev)
219 {
220 int[] a = ev.Values;
221 switch (ev.Command)
222 {
223 case "CALL": {
224 var g = new Group(groups.Count + 1, a[0], a[1], a[2], Time);
225 groups.Add(g);
226 Emit($"CALL {g.Label} {g.A}->{g.B} x{g.Count}");
227 Dispatch(g);
228 break;
229 }
230 case "CANCEL": {
231 Group g = groups[a[0] - 1];
232 if (g.Waiting > 0) Walk(g, "CANCEL");
233 else Emit($"CANCEL {g.Label} ignored");
234 break;
235 }
236 case "OUT": {
237 Car c = cars[a[0] - 1];
238 c.Service = false;
239 Emit($"{c.Label} OUT");
240 List<Group> calls = c.Calls;
241 c.Calls = new();
242 foreach (Group g in calls) {
243 g.Car = null;
244 Dispatch(g);
245 }
246 break;
247 }
248 case "IN": {
249 Car c = cars[a[0] - 1];
250 c.Service = true;
251 c.IdleStart = Time;
252 Emit($"{c.Label} IN");
253 List<Group> waiting = pending;
254 pending = new();
255 foreach (Group g in waiting) if (!Assign(g)) pending.Add(g);
256 break;
257 }
258 case "HOLD": {
259 Car c = cars[a[0] - 1];
260 if (c.Door > 0) {
261 c.Door += a[1];
262 Emit($"{c.Label} HOLD {c.Door}");
263 } else Emit($"{c.Label} HOLD ignored");
264 break;
265 }
266 case "FIRE": fire = a[0];
267 Emit($"FIRE {fire}");
268 foreach (Group g in groups) if (g.Waiting > 0) Walk(g, "ABANDON");
269 break;
270 case "RESET": fire = null;
271 Emit("RESET");
272 foreach (Car c in cars) c.IdleStart = Time;
273 break;
274 case "WHERE": {
275 Group g = groups[a[0] - 1];
276 string place = g.Waiting == 0 ? "-" : g.Car is int e ? cars[e].Label : "pending";
277 var riding = cars.Select(c => (c, n: c.Passengers.Where(p => p.Group == g).Sum(p => p.Count))).Where(x => x.n > 0).Select(x => $"{x.c.Label}x{x.n}");
278 Emit($"WHERE {g.Label} waiting={g.Waiting} car={place} riding={Rules.Join(riding)} delivered={g.Delivered} lost={g.Lost}");
279 break;
280 }
281 case "STATUS": foreach (Car c in cars) Emit($"STATUS {c.Label} floor={c.Floor} dir={(c.Direction > 0 ? "up" : c.Direction < 0 ? "down" : "idle")} door={c.Door} load={c.Load} stops={Rules.Join(c.Passengers.Select(p => p.Group.B).Distinct().OrderBy(f => f))} calls={Rules.Join(c.Calls.Select(g => g.Label))} service={(c.Service ? "in" : "out")}");
282 Emit($"STATUS pending={Rules.Join(pending.Select(g => g.Label))}");
283 break;
284 }
285 }
286 int Direction(Car c)
287 {
288 int f = c.Floor, d = c.Direction;
289 List<Group> here = Bypass(c) ? new() : c.Calls.Where(g => g.A == f).ToList();
290 HashSet<int> targets = c.Targets;
291 if (d != 0 && (targets.Any(y => (y - f) * d > 0) || here.Any(g => g.Direction == d))) return d;
292 if (here.Count > 0) return here[0].Direction;
293 if (d != 0 && targets.Any(y => (y - f) * d < 0)) return -d;
294 if (d == 0 && c.Calls.Count > 0) return Math.Sign(c.Calls[0].A - f);
295 return 0;
296 }
297 void Move(Car c, int d) {
298 c.Energy += d > 0 ? 3 + c.Load : 1;
299 c.Distance++;
300 c.Floor += d;
301 }
302 void Open(Car c)
303 {
304 Emit($"{c.Label} OPEN {c.Floor}");
305 c.Door = config.Door;
306 c.Stops++;
307 c.Energy += 2;
308 var riding = new List<Passenger>();
309 foreach (Passenger p in c.Passengers)
310 {
311 Group g = p.Group;
312 if (g.B == c.Floor)
313 {
314 Emit($"{c.Label} ALIGHT {g.Label} x{p.Count}");
315 int w = p.Time - g.Time, r = Time - g.Time;
316 g.Delivered += p.Count;
317 g.LastAlight = Time;
318 g.WaitTotal += (long)w * p.Count;
319 g.TravelTotal += (long)r * p.Count;
320 Delivered += p.Count;
321 WaitTotal += (long)w * p.Count;
322 TravelTotal += (long)r * p.Count;
323 MaxWait = Math.Max(MaxWait, w);
324 } else if (fire != null) {
325 Emit($"{c.Label} EVACUATE {g.Label} x{p.Count}");
326 g.Lost += p.Count;
327 evacuated += p.Count;
328 } else riding.Add(p);
329 }
330 c.Passengers = riding;
331 if (fire != null) return;
332 int n = Direction(c);
333 if (n != 0) c.Direction = n;
334 foreach (Group g in c.Calls.Where(g => g.A == c.Floor && g.Direction == n).ToList())
335 {
336 int k = Math.Min(config.Capacity - c.Load, g.Waiting);
337 if (k > 0) {
338 c.Passengers.Add(new(g, k, Time));
339 c.Boarded += k;
340 g.Waiting -= k;
341 g.Boarded += k;
342 g.LastBoard = Time;
343 g.Cars.Add(c.Number);
344 Emit($"{c.Label} BOARD {g.Label} x{k}");
345 }
346 if (g.Waiting == 0) {
347 c.Calls.Remove(g);
348 g.Car = null;
349 } else {
350 Emit($"{c.Label} LEFT {g.Label} x{g.Waiting}");
351 Assign(g, c);
352 }
353 }
354 }
355 void Advance(Car c)
356 {
357 int old = c.Floor;
358 if (c.Door > 0) c.Door--;
359 if (c.Door == 0)
360 {
361 int n = fire is int f ? Math.Sign(f - c.Floor) : Direction(c);
362 bool stop = fire != null ? n == 0 && c.Passengers.Count > 0 : c.Passengers.Any(p => p.Group.B == c.Floor) || !Bypass(c) && c.Calls.Any(g => g.A == c.Floor && g.Direction == n);
363 if (stop) Open(c);
364 else if (n != 0) {
365 c.Direction = n;
366 Move(c, n);
367 } else
368 {
369 if (c.Direction != 0) {
370 Emit($"{c.Label} IDLE {c.Floor}");
371 c.Direction = 0;
372 c.IdleStart = Time;
373 }
374 if (fire == null && c.Service && config.Park is int p && c.Floor != config.Homes[c.Number - 1] && Time - c.IdleStart >= p)
375 {
376 int h = config.Homes[c.Number - 1];
377 Move(c, Math.Sign(h - c.Floor));
378 if (c.Floor == h) Emit($"{c.Label} PARK {h}");
379 }
380 }
381 }
382 if (c.Door > 0 || c.Floor != old) c.Busy++;
383 }
384 public void Run()
385 {
386 int index = 0;
387 for (Time = 0;
388 ;
389 Time++)
390 {
391 if (config.Patience is int p) foreach (Group g in groups) if (g.Waiting > 0 && Time == g.Time + p) Walk(g, "ABANDON");
392 while (index < events.Count && events[index].Time == Time) Execute(events[index++]);
393 foreach (Car c in cars) Advance(c);
394 bool settled = cars.All(c => c.Door == 0 && c.Passengers.Count == 0 && c.Calls.Count == 0 && (fire is int f ? c.Floor == f : !c.Service || config.Park == null || c.Floor == config.Homes[c.Number - 1]));
395 if (index == events.Count && settled) {
396 Status = "end";
397 Emit("END");
398 break;
399 }
400 if (Time == config.Limit) {
401 Status = "abort";
402 Emit("ABORT");
403 break;
404 }
405 }
406 }
407 public string Averages => $"avg_wait={Rules.Average(WaitTotal, Delivered)} avg_travel={Rules.Average(TravelTotal, Delivered)} max_wait={(Delivered == 0 ? "-" : MaxWait.ToString())}";
408 public List<string> Report()
409 {
410 var lines = new List<string>();
411 foreach (Group g in groups) lines.Add($"{g.Label} t={g.Time} {g.A}->{g.B} n={g.Count} delivered={g.Delivered} lost={g.Lost} wait={(g.Boarded == g.Count ? (g.LastBoard - g.Time).ToString() : "-")} travel={(g.Delivered == g.Count ? (g.LastAlight - g.Time).ToString() : "-")} cars={Rules.Join(g.Cars)}");
412 foreach (Car c in cars) {
413 long x = c.Busy * 1000L / (Time + 1);
414 lines.Add($"{c.Label} distance={c.Distance} stops={c.Stops} boarded={c.Boarded} busy={c.Busy} util={x / 10}.{x % 10}% energy={c.Energy}");
415 }
416 string Breakdown(string label, IEnumerable<Group> gs) {
417 var a = gs.ToList();
418 return $"{label}: groups={a.Count} passengers={a.Sum(g => g.Count)} delivered={a.Sum(g => g.Delivered)} avg_wait={Rules.Average(a.Sum(g => g.WaitTotal), a.Sum(g => g.Delivered))}";
419 }
420 foreach (int f in groups.Select(g => g.A).Distinct().OrderBy(f => f)) lines.Add(Breakdown($"F{f}", groups.Where(g => g.A == f)));
421 if (config.Period is int w) foreach (int k in groups.Select(g => g.Time / w).Distinct().OrderBy(k => k)) lines.Add(Breakdown($"P{k} {k * w}-{k * w + w - 1}", groups.Where(g => g.Time / w == k)));
422 lines.Add($"summary: groups={groups.Count} passengers={groups.Sum(g => g.Count)} delivered={Delivered} abandoned={abandoned} evacuated={evacuated} {Averages} end={Time}");
423 return lines;
424 }

436static class Program
437{
438 static List<Event> Validate(Config c, List<(int Line, string[] Words)> raw, List<string> output)
439 {
440 var accepted = new List<Event>();
441 int last = 0, groups = 0;
442 bool fire = false;
443 var outs = new HashSet<int>();
444 foreach (var (line, w) in raw)
445 {
446 int time = int.Parse(w[0]);
447 string cmd = w[1];
448 string? error = null;
449 int[] a = Array.Empty<int>();
450 if (!Rules.Arguments.TryGetValue(cmd, out int count)) error = $"unknown command {cmd}";
451 else if (w.Length - 2 != count) error = $"wrong number of arguments for {cmd}";
452 else if (time < last) error = "time goes backwards";
453 else
454 {
455 a = w.Skip(2).Select(int.Parse).ToArray();
456 bool Floor(int f) => f >= 1 && f <= c.Floors;
457 bool Car(int e) => e >= 1 && e <= c.Cars;
458 switch (cmd)
459 {
460 case "CALL": error = !Floor(a[0]) || !Floor(a[1]) ? "bad floor" : a[0] == a[1] ? "same floor" : a[2] < 1 || a[2] > 99 ? "bad count" : !Enumerable.Range(0, c.Cars).Any(e => c.Contains(e, a[0]) && c.Contains(e, a[1])) ? $"no car serves {a[0]}->{a[1]}" : fire ? "fire service active" : null;
461 break;
462 case "CANCEL": case "WHERE": if (a[0] < 1 || a[0] > groups) error = "unknown group";
463 break;
464 case "OUT": case "IN": error = !Car(a[0]) ? "bad car" : cmd == "OUT" && outs.Contains(a[0]) ? $"car {a[0]} already out" : cmd == "IN" && !outs.Contains(a[0]) ? $"car {a[0]} already in" : null;
465 break;
466 case "HOLD": error = !Car(a[0]) ? "bad car" : a[1] < 1 || a[1] > 9 ? "bad ticks" : null;
467 break;
468 case "FIRE": error = !Floor(a[0]) ? "bad floor" : fire ? "fire already active" : null;
469 break;
470 case "RESET": if (!fire) error = "fire not active";
471 break;
472 }
473 }
474 if (error != null) {
475 output.Add($"line {line}: {error}");
476 continue;
477 }
478 last = time;
479 if (cmd == "CALL") groups++;
480 if (cmd == "OUT") outs.Add(a[0]);
481 if (cmd == "IN") outs.Remove(a[0]);
482 if (cmd == "FIRE") fire = true;
483 if (cmd == "RESET") fire = false;
484 accepted.Add(new(time, cmd, a));
485 }
486 return accepted;
487 }
488 static void Main()
489 {
490 var all = new List<string>();
491 string? line;
492 while ((line = Console.ReadLine()) != null) all.Add(line);
493 int q = int.Parse(all[0]), index = 1;
494 var scenarios = new List<Scenario>();
495 var output = new List<string>();
496 for (int i = 0;
497 i < q;
498 i++)
499 {
500 var s = new Scenario(all[index++].Split(' ')[1]);
501 scenarios.Add(s);
502 var settings = new List<string[]>();
503 var raw = new List<(int, string[])>();
504 while (all[index] != "END") {
505 string[] w = all[index].Split(' ');
506 if (char.IsDigit(w[0][0])) raw.Add((index + 1, w));
507 else settings.Add(w);
508 index++;
509 }
510 index++;
511 output.Add($"== {s.Name} ==");
512 try {
513 s.Config = Config.Parse(settings);
514 } catch (ArgumentException e) {
515 output.Add("error: " + e.Message);
516 continue;
517 }
518 Config c = s.Config;
519 output.AddRange(c.Header());
520 List<Event> events = Validate(c, raw, output);
521 var sim = new Simulation(c, events, c.Policy);
522 s.Run = sim;
523 sim.Run();
524 output.AddRange(sim.Log);
525 output.AddRange(sim.Report());
526 foreach (string p in c.Compare) {
527 var r = new Simulation(c, events, p);
528 r.Run();
529 output.Add($"compare {p}: delivered={r.Delivered} {r.Averages} end={r.Time} status={r.Status}");
530 }
531 }
532 output.Add("== ranking ==");
533 var ranked = scenarios.Select((s, i) => (s, i)).Where(x => x.s.Outcome == "ranked").ToList();
534 ranked.Sort((a, b) => {
535 Simulation x = a.s.Run!, y = b.s.Run!;
536 int n = (x.WaitTotal * y.Delivered).CompareTo(y.WaitTotal * x.Delivered);
537 if (n == 0) n = (x.TravelTotal * y.Delivered).CompareTo(y.TravelTotal * x.Delivered);
538 return n == 0 ? a.i.CompareTo(b.i) : n;
539 });
540 for (int i = 0;
541 i < ranked.Count;
542 i++) {
543 Scenario s = ranked[i].s;
544 output.Add($"{i + 1}. {s.Name} policy={s.Config!.Policy} {s.Run!.Averages}");
545 }
546 foreach (Scenario s in scenarios) if (s.Outcome != "ranked") output.Add($"- {s.Name} {s.Outcome}");
