1const POLICIES = ("COST", "NEAREST", "ROUND", "ETA")
2const CONFIG_KEYS = ("FLOORS", "CARS", "CAPACITY", "DOOR", "START", "POLICY", "ZONES", "PARK", "PATIENCE", "BYPASS", "PERIOD", "LIMIT", "COMPARE")
3const ARITY = Dict("CALL" => 3, "CANCEL" => 1, "OUT" => 1, "IN" => 1, "HOLD" => 2, "FIRE" => 1, "RESET" => 0, "WHERE" => 1, "STATUS" => 0)
4struct ConfigError <: Exception
5 reason::String
6end
7fail(s) = throw(ConfigError(s))
8joined(xs) = isempty(xs) ? "-" : join(xs, ',')
9function average(total, count)
10 count == 0 && return "-"
11 x = 100total ÷ count
12 string(x ÷ 100, ".", lpad(x % 100, 2, '0'))
13end

30function parse_config(settings)
31 entries = Dict{String,Vector{String}}()
32 for w in settings
33 k = w[1]
34 k in CONFIG_KEYS || fail("unknown key $k")
35 haskey(entries, k) && fail("duplicate key $k")
36 entries[k] = w[2:end]
37 end
38 for k in CONFIG_KEYS[1:5]
39 haskey(entries, k) || fail("missing key $k")
40 end
41 c = Config(0, 0, 0, 0, Int[], "COST", Tuple{Int,Int}[], -1, Int[], 0, 100, 0, 1000, String[])
42 for k in CONFIG_KEYS
43 haskey(entries, k) || continue
44 v = entries[k]
45 expected = k in ("START", "ZONES") ? c.cars : k == "PARK" ? c.cars + 1 : 1
46 correct = k == "COMPARE" ? 1 <= length(v) <= 4 : length(v) == expected
47 correct || fail("wrong number of values for $k")
48 bounded(s, lo, hi) = let x = parse(Int, s)
49 lo <= x <= hi || fail("bad value for $k")
50 x
51 end
52 if k == "FLOORS"
53 c.floors = bounded(v[1], 2, 50)
54 elseif k == "CARS"
55 c.cars = bounded(v[1], 1, 8)
56 elseif k == "CAPACITY"
57 c.capacity = bounded(v[1], 1, 20)
58 elseif k == "DOOR"
59 c.door = bounded(v[1], 1, 9)
60 elseif k == "START"
61 c.starts = [bounded(s, 1, c.floors) for s in v]
62 elseif k == "POLICY"
63 v[1] in POLICIES || fail("bad value for $k")
64 c.policy = v[1]
65 elseif k == "ZONES"
66 for s in v
67 a, b = parse.(Int, split(s, ':'))
68 1 <= a < b <= c.floors || fail("bad value for $k")
69 push!(c.zones, (a, b))
70 end
71 elseif k == "PARK"
72 c.park = bounded(v[1], 0, 100)
73 c.homes = [bounded(s, 1, c.floors) for s in v[2:end]]
74 elseif k == "PATIENCE"
75 c.patience = bounded(v[1], 1, 1000)
76 elseif k == "BYPASS"
77 c.bypass = bounded(v[1], 50, 100)
78 elseif k == "PERIOD"
79 c.period = bounded(v[1], 1, 1000)
80 elseif k == "LIMIT"
81 c.limit = bounded(v[1], 1, 5000)
82 else
83 (all(s -> s in POLICIES, v) && allunique(v)) || fail("bad value for $k")
84 c.compare = v
85 end
86 end
87 isempty(c.zones) && (c.zones = fill((1, c.floors), c.cars))
88 for i in 1:c.cars
89 lo, hi = c.zones[i]
90 lo <= c.starts[i] <= hi || fail("car $i starts outside its zone")
91 c.park >= 0 && !(lo <= c.homes[i] <= hi) && fail("car $i home outside its zone")
92 end
93 c
94end

100function validate_events(c, lines, output)
101 accepted = ElevatorEvent[]
102 lasttime = groups = 0
103 out = Set{Int}()
104 fire = false
105 for (line, w) in lines
106 t = parse(Int, w[1])
107 cmd = w[2]
108 reason = ""
109 v = Int[]
110 if !haskey(ARITY, cmd)
111 reason = "unknown command $cmd"
112 elseif length(w) - 2 != ARITY[cmd]
113 reason = "wrong number of arguments for $cmd"
114 elseif t < lasttime
115 reason = "time goes backwards"
116 else
117 v = parse.(Int, w[3:end])
118 if cmd == "CALL"
119 a, b, n = v
120 if !(1 <= a <= c.floors && 1 <= b <= c.floors)
121 reason = "bad floor"
122 elseif a == b
123 reason = "same floor"
124 elseif !(1 <= n <= 99)
125 reason = "bad count"
126 elseif !any(lo <= a <= hi && lo <= b <= hi for (lo, hi) in c.zones)
127 reason = "no car serves $a->$b"
128 elseif fire
129 reason = "fire service active"
130 end
131 elseif cmd in ("CANCEL", "WHERE")
132 !(1 <= v[1] <= groups) && (reason = "unknown group")
133 elseif cmd in ("OUT", "IN", "HOLD")
134 e = v[1]
135 if !(1 <= e <= c.cars)
136 reason = "bad car"
137 elseif cmd == "OUT" && e in out
138 reason = "car $e already out"
139 elseif cmd == "IN" && !(e in out)
140 reason = "car $e already in"
141 elseif cmd == "HOLD" && !(1 <= v[2] <= 9)
142 reason = "bad ticks"
143 end
144 elseif cmd == "FIRE"
145 if !(1 <= v[1] <= c.floors)
146 reason = "bad floor"
147 elseif fire
148 reason = "fire already active"
149 end
150 elseif cmd == "RESET" && !fire
151 reason = "fire not active"
152 end
153 end
154 if !isempty(reason)
155 push!(output, "line $line: $reason")
156 continue
157 end
158 push!(accepted, ElevatorEvent(t, cmd, v))
159 lasttime = t
160 if cmd == "CALL"
161 groups += 1
162 elseif cmd == "OUT"
163 push!(out, v[1])
164 elseif cmd == "IN"
165 delete!(out, v[1])
166 elseif cmd == "FIRE"
167 fire = true
168 elseif cmd == "RESET"
169 fire = false
170 end
171 end
172 accepted
173end

234function Simulation(c, policy, logging)
235 cars = [Car(i, c.starts[i], c.zones[i]..., c.park >= 0 ? c.homes[i] : 0,
236 0, 0, Passenger[], Group[], true, 0, 0, 0, 0, 0, 0) for i in 1:c.cars]
237 Simulation(c, policy, cars, Group[], Group[], 0, 0, 0, "", String[], logging, 0, 0, 0, 0, 0, 0)
238end
239function log!(s, parts...)
240 s.logging && push!(s.log, string(s.time, " ", join(parts, " ")))
241end
242load(c) = sum((p.count for p in c.passengers); init=0)
243targets(c) = union(Set(p.group.destination for p in c.passengers), Set(g.landing for g in c.calls))
244bypassing(s, c) = load(c) * 100 >= s.config.capacity * s.config.bypass
245function cost(s, c, g)
246 f, d, x = c.floor, c.direction, g.landing
247 (s.policy == "NEAREST" || d == 0) && return abs(f - x)
248 floors = targets(c)
249 if g.direction == d && (x - f) * d >= 0
250 distance = abs(x - f)
251 via = count(y -> (y - f) * d > 0 && (x - y) * d > 0, floors)
252 else
253 m = d > 0 ? maximum(union(floors, Set([f]))) : minimum(union(floors, Set([f])))
254 distance = abs(f - m) + abs(m - x)
255 via = count(y -> (y - f) * d > 0, floors)
256 end
257 s.policy == "COST" ? distance : distance + s.config.door * via + c.door
258end
259function assign!(s, g, excluded=0)
260 candidates = [c for c in s.cars if c.service && c.number != excluded &&
261 c.low <= g.landing <= c.high && c.low <= g.destination <= c.high]
262 isempty(candidates) && return false
263 if s.policy == "ROUND"
264 next = findfirst(c -> c.number > s.lastround, candidates)
265 chosen = candidates[isnothing(next) ? 1 : next]
266 s.lastround = chosen.number
267 log!(s, "E$(chosen.number)", "ASSIGN", "G$(g.number)")
268 else
269 costs = [cost(s, c, g) for c in candidates]
270 i = argmin(costs)
271 chosen = candidates[i]
272 log!(s, "E$(chosen.number)", "ASSIGN", "G$(g.number)", "cost=$(costs[i])")
273 end
274 g.car != 0 && filter!(x -> x !== g, s.cars[g.car].calls)
275 push!(chosen.calls, g)
276 g.car = chosen.number
277 true
278end
279function dispatch!(s, g)
280 if !assign!(s, g)
281 push!(s.pending, g)
282 log!(s, "PENDING", "G$(g.number)")
283 end
284end
285function walk!(s, g, word)
286 if g.car == 0
287 filter!(x -> x !== g, s.pending)
288 else
289 filter!(x -> x !== g, s.cars[g.car].calls)
290 g.car = 0
291 end
292 log!(s, word, "G$(g.number)", "x$(g.waiting)")
293 g.lost += g.waiting
294 s.abandoned += g.waiting
295 g.waiting = 0
296end
297function event!(s, event)
298 cmd, v = event.command, event.args
299 if cmd == "CALL"
300 a, b, n = v
301 g = Group(length(s.groups) + 1, a, b, n, s.time, sign(b - a), n, 0,
302 0, 0, 0, 0, 0, Set{Int}(), 0, 0)
303 push!(s.groups, g)
304 log!(s, "CALL", "G$(g.number)", "$a->$b", "x$n")
305 dispatch!(s, g)
306 elseif cmd == "CANCEL"
307 g = s.groups[v[1]]
308 g.waiting > 0 ? walk!(s, g, "CANCEL") : log!(s, "CANCEL", "G$(g.number)", "ignored")
309 elseif cmd == "OUT"
310 c = s.cars[v[1]]
311 c.service = false
312 log!(s, "E$(c.number)", "OUT")
313 calls = copy(c.calls)
314 empty!(c.calls)
315 for g in calls
316 g.car = 0
317 dispatch!(s, g)
318 end
319 elseif cmd == "IN"
320 c = s.cars[v[1]]
321 c.service = true
322 c.idlestart = s.time
323 log!(s, "E$(c.number)", "IN")
324 pending = copy(s.pending)
325 empty!(s.pending)
326 for g in pending
327 assign!(s, g) || push!(s.pending, g)
328 end
329 elseif cmd == "HOLD"
330 c = s.cars[v[1]]
331 if c.door > 0
332 c.door += v[2]
333 log!(s, "E$(c.number)", "HOLD", c.door)
334 else
335 log!(s, "E$(c.number)", "HOLD", "ignored")
336 end
337 elseif cmd == "FIRE"
338 s.fire = v[1]
339 log!(s, "FIRE", s.fire)
340 for g in s.groups
341 g.waiting > 0 && walk!(s, g, "ABANDON")
342 end
343 elseif cmd == "RESET"
344 s.fire = 0
345 log!(s, "RESET")
346 for c in s.cars
347 c.idlestart = s.time
348 end
349 elseif cmd == "WHERE"
350 g = s.groups[v[1]]
351 place = g.waiting == 0 ? "-" : g.car == 0 ? "pending" : "E$(g.car)"
352 riding = String[]
353 for c in s.cars
354 n = sum((p.count for p in c.passengers if p.group === g); init=0)
355 n > 0 && push!(riding, "E$(c.number)x$n")
356 end
357 log!(s, "WHERE", "G$(g.number)", "waiting=$(g.waiting)", "car=$place",
358 "riding=$(joined(riding))", "delivered=$(g.delivered)", "lost=$(g.lost)")
359 elseif cmd == "STATUS"
360 for c in s.cars
361 stops = sort(unique([p.group.destination for p in c.passengers]))
362 calls = ["G$(g.number)" for g in c.calls]
363 direction = c.direction == 0 ? "idle" : c.direction > 0 ? "up" : "down"
364 service = c.service ? "in" : "out"
365 log!(s, "STATUS", "E$(c.number)", "floor=$(c.floor)", "dir=$direction",
366 "door=$(c.door)", "load=$(load(c))", "stops=$(joined(stops))",
367 "calls=$(joined(calls))", "service=$service")
368 end
369 log!(s, "STATUS", "pending=$(joined(["G$(g.number)" for g in s.pending]))")
370 end
371end
372function direction(s, c)
373 f, d = c.floor, c.direction
374 here = bypassing(s, c) ? Group[] : [g for g in c.calls if g.landing == f]
375 floors = targets(c)
376 if d != 0 && (any(y -> (y - f) * d > 0, floors) || any(g -> g.direction == d, here))
377 return d
378 elseif !isempty(here)
379 return here[1].direction
380 elseif d != 0 && any(y -> (y - f) * d < 0, floors)
381 return -d
382 elseif d == 0 && !isempty(c.calls)
383 return sign(c.calls[1].landing - f)
384 end
385 0
386end
387function move!(c, step)
388 c.energy += step > 0 ? 3 + load(c) : 1
389 c.distance += 1
390 c.floor += step
391end
392function open!(s, c)
393 log!(s, "E$(c.number)", "OPEN", c.floor)
394 c.door = s.config.door
395 c.stops += 1
396 c.energy += 2
397 riding = Passenger[]
398 for p in c.passengers
399 g = p.group
400 if g.destination == c.floor
401 log!(s, "E$(c.number)", "ALIGHT", "G$(g.number)", "x$(p.count)")
402 wait = p.boarded - g.time
403 travel = s.time - g.time
404 g.delivered += p.count
405 g.lastalight = s.time
406 g.waittotal += wait * p.count
407 g.traveltotal += travel * p.count
408 s.delivered += p.count
409 s.waittotal += wait * p.count
410 s.traveltotal += travel * p.count
411 s.maxwait = max(s.maxwait, wait)
412 elseif s.fire != 0
413 log!(s, "E$(c.number)", "EVACUATE", "G$(g.number)", "x$(p.count)")
414 g.lost += p.count
415 s.evacuated += p.count
416 else
417 push!(riding, p)
418 end
419 end
420 c.passengers = riding
421 s.fire != 0 && return
422 heading = direction(s, c)
423 heading != 0 && (c.direction = heading)
424 boarding = [g for g in c.calls if g.landing == c.floor && g.direction == heading]
425 for g in boarding
426 n = min(s.config.capacity - load(c), g.waiting)
427 if n > 0
428 push!(c.passengers, Passenger(g, n, s.time))
429 c.boarded += n
430 g.waiting -= n
431 g.boarded += n
432 g.lastboard = s.time
433 push!(g.cars, c.number)
434 log!(s, "E$(c.number)", "BOARD", "G$(g.number)", "x$n")
435 end
436 if g.waiting == 0
437 filter!(x -> x !== g, c.calls)
438 g.car = 0
439 else
440 log!(s, "E$(c.number)", "LEFT", "G$(g.number)", "x$(g.waiting)")
441 assign!(s, g, c.number)
442 end
443 end
444end
445function advance!(s, c)
446 oldfloor = c.floor
447 c.door > 0 && (c.door -= 1)
448 if c.door == 0
449 if s.fire != 0
450 heading = sign(s.fire - c.floor)
451 stopping = heading == 0 && !isempty(c.passengers)
452 else
453 heading = direction(s, c)
454 stopping = any(p -> p.group.destination == c.floor, c.passengers) ||
455 (!bypassing(s, c) && any(g -> g.landing == c.floor && g.direction == heading, c.calls))
456 end
457 if stopping
458 open!(s, c)
459 elseif heading != 0
460 c.direction = heading
461 move!(c, heading)
462 else
463 if c.direction != 0
464 log!(s, "E$(c.number)", "IDLE", c.floor)
465 c.direction = 0
466 c.idlestart = s.time
467 end
468 if s.fire == 0 && c.service && c.home != 0 && c.floor != c.home && s.time - c.idlestart >= s.config.park
469 move!(c, sign(c.home - c.floor))
470 c.floor == c.home && log!(s, "E$(c.number)", "PARK", c.home)
471 end
472 end
473 end
474 (c.door > 0 || c.floor != oldfloor) && (c.busy += 1)
475end
476function settled(s, c)
477 (c.door > 0 || !isempty(c.passengers) || !isempty(c.calls)) && return false
478 s.fire != 0 ? c.floor == s.fire : !c.service || c.home == 0 || c.floor == c.home
479end
480function run!(s, events)
481 index = 1
482 while true
483 if s.config.patience > 0
484 for g in s.groups
485 g.waiting > 0 && s.time == g.time + s.config.patience && walk!(s, g, "ABANDON")
486 end
487 end
488 while index <= length(events) && events[index].time == s.time
489 event!(s, events[index])
490 index += 1
491 end
492 for c in s.cars
493 advance!(s, c)
494 end
495 if index > length(events) && all(c -> settled(s, c), s.cars)
496 s.status = "end"
497 log!(s, "END")
498 return
499 elseif s.time == s.config.limit
500 s.status = "abort"
501 log!(s, "ABORT")
502 return
503 end
504 s.time += 1
505 end
506end
507function averages(s)
508 "avg_wait=$(average(s.waittotal, s.delivered)) avg_travel=$(average(s.traveltotal, s.delivered)) max_wait=$(s.delivered == 0 ? "-" : s.maxwait)"
509end
510function breakdown(label, groups)
511 passengers = sum((g.count for g in groups); init=0)
512 delivered = sum((g.delivered for g in groups); init=0)
513 total = sum((g.waittotal for g in groups); init=0)
514 "$label: groups=$(length(groups)) passengers=$passengers delivered=$delivered avg_wait=$(average(total, delivered))"
515end
516function report!(output, s)
517 for g in s.groups
518 wait = g.boarded == g.count ? string(g.lastboard - g.time) : "-"
519 travel = g.delivered == g.count ? string(g.lastalight - g.time) : "-"
520 cars = joined(sort(collect(g.cars)))
521 push!(output, "G$(g.number) t=$(g.time) $(g.landing)->$(g.destination) n=$(g.count) delivered=$(g.delivered) lost=$(g.lost) wait=$wait travel=$travel cars=$cars")
522 end
523 for c in s.cars
524 tenths = c.busy * 1000 ÷ (s.time + 1)
525 push!(output, "E$(c.number) distance=$(c.distance) stops=$(c.stops) boarded=$(c.boarded) busy=$(c.busy) util=$(tenths ÷ 10).$(tenths % 10)% energy=$(c.energy)")
526 end
527 for floor in sort(unique([g.landing for g in s.groups]))
528 push!(output, breakdown("F$floor", [g for g in s.groups if g.landing == floor]))
529 end
530 width = s.config.period
531 if width > 0
532 for period in sort(unique([g.time ÷ width for g in s.groups]))
533 groups = [g for g in s.groups if g.time ÷ width == period]
534 push!(output, breakdown("P$period $(period * width)-$(period * width + width - 1)", groups))
535 end
536 end
537 passengers = sum((g.count for g in s.groups); init=0)
538 push!(output, "summary: groups=$(length(s.groups)) passengers=$passengers delivered=$(s.delivered) abandoned=$(s.abandoned) evacuated=$(s.evacuated) $(averages(s)) end=$(s.time)")
539end
540function main()
541 lines = readlines()
542 output = String[]
543 names = String[]
544 configs = Union{Nothing,Config}[]
545 results = Union{Nothing,Simulation}[]
546 index = 2
547 for _ in 1:parse(Int, lines[1])
548 name = String(split(lines[index])[2])
549 index += 1
550 settings = Vector{String}[]
551 eventlines = Tuple{Int,Vector{String}}[]
552 while lines[index] != "END"
553 w = String.(split(lines[index]))
554 if isdigit(w[1][1])
555 push!(eventlines, (index, w))
556 else
557 push!(settings, w)
558 end
559 index += 1
560 end
561 index += 1
562 push!(names, name)
563 push!(output, "== $name ==")
564 config = nothing
565 try
566 config = parse_config(settings)
567 catch err
568 err isa ConfigError || rethrow()
569 push!(output, "error: $(err.reason)")
570 end
571 push!(configs, config)
572 if isnothing(config)
573 push!(results, nothing)
574 continue
575 end
576 c = config
577 park = c.park < 0 ? "off" : string(c.park)
578 patience = c.patience == 0 ? "off" : string(c.patience)
579 period = c.period == 0 ? "off" : string(c.period)
580 push!(output, "config: floors=$(c.floors) cars=$(c.cars) capacity=$(c.capacity) door=$(c.door) policy=$(c.policy) park=$park patience=$patience bypass=$(c.bypass) period=$period limit=$(c.limit) compare=$(joined(c.compare))")
581 for i in 1:c.cars
582 lo, hi = c.zones[i]
583 home = c.park < 0 ? "-" : string(c.homes[i])
584 push!(output, "E$i: start=$(c.starts[i]) zone=$lo-$hi home=$home")
585 end
586 events = validate_events(c, eventlines, output)
587 s = Simulation(c, c.policy, true)
588 run!(s, events)
589 append!(output, s.log)
590 report!(output, s)
591 push!(results, s)
592 for policy in c.compare
593 rerun = Simulation(c, policy, false)
594 run!(rerun, events)
595 push!(output, "compare $policy: delivered=$(rerun.delivered) $(averages(rerun)) end=$(rerun.time) status=$(rerun.status)")
596 end
597 end
598 push!(output, "== ranking ==")
599 ranked = [i for i in eachindex(results) if !isnothing(results[i]) && results[i].status == "end" && results[i].delivered > 0]
600 function precedes(i, j)
601 a, b = results[i], results[j]
602 wa, wb = a.waittotal * b.delivered, b.waittotal * a.delivered
603 wa != wb && return wa < wb
604 ta, tb = a.traveltotal * b.delivered, b.traveltotal * a.delivered
605 ta != tb ? ta < tb : i < j
606 end
607 sort!(ranked; lt=precedes)
608 for (rank, i) in enumerate(ranked)
609 push!(output, "$rank. $(names[i]) policy=$(configs[i].policy) $(averages(results[i]))")
610 end
611 for i in eachindex(results)
612 i in ranked && continue
613 s = results[i]
614 outcome = isnothing(s) ? "invalid" : s.status == "abort" ? "aborted" : "empty"
615 push!(output, "- $(names[i]) $outcome")
616 end
617 push!(output, "== policies ==")
618 for policy in POLICIES
619 runs = [results[i] for i in ranked if configs[i].policy == policy]
620 isempty(runs) && continue
621 count = sum(s.delivered for s in runs)
622 wait = sum(s.waittotal for s in runs)
623 travel = sum(s.traveltotal for s in runs)
624 push!(output, "$policy: runs=$(length(runs)) avg_wait=$(average(wait, count)) avg_travel=$(average(travel, count))")
625 end
626 foreach(println, output)
627end
