32 Config(List<String[]> lines) {
33 Map<String,List<String>> entries=new HashMap<>();
34 for(String[] a:lines) {
35 String k=a[0];
36 require(KEYS.contains(k),"unknown key "+k);
37 require(!entries.containsKey(k),"duplicate key "+k);
38 entries.put(k,Arrays.asList(a).subList(1,a.length));
39 }
40 for(String k:KEYS.subList(0,5)) require(entries.containsKey(k),"missing key "+k);
41 for(String k:KEYS) {
42 if(!entries.containsKey(k)) continue;
43 List<String> v=entries.get(k);
44 int n=v.size();
45 require(k.equals("START")||k.equals("ZONES")?n==cars:k.equals("PARK")?n==cars+1:k.equals("COMPARE")?n>=1&&n<=4:n==1,"wrong number of values for "+k);
46 switch(k) {
47 case "FLOORS" -> floors=bound(k,v.getFirst(),2,50);
48 case "CARS" -> {
49 cars=bound(k,v.getFirst(),1,8);
50 starts=new int[cars];
51 homes=new int[cars];
52 low=new int[cars];
53 high=new int[cars];
54 }
55 case "CAPACITY" -> capacity=bound(k,v.getFirst(),1,20);
56 case "DOOR" -> door=bound(k,v.getFirst(),1,9);
57 case "START" -> {
58 for(int i=0;i<cars;i++) starts[i]=bound(k,v.get(i),1,floors);
59 }
60 case "POLICY" -> {
61 require(POLICIES.contains(v.getFirst()),"bad value for "+k);
62 policy=v.getFirst();
63 }
64 case "ZONES" -> {
65 for(int i=0;i<cars;i++) {
66 String[] z=v.get(i).split(":");

68 high[i]=Integer.parseInt(z[1]);

70 }
71 }
72 case "PARK" -> {
73 park=bound(k,v.getFirst(),0,100);

75 }
76 case "PATIENCE" -> patience=bound(k,v.getFirst(),1,1000);
77 case "BYPASS" -> bypass=bound(k,v.getFirst(),50,100);
78 case "PERIOD" -> period=bound(k,v.getFirst(),1,1000);
79 case "LIMIT" -> limit=bound(k,v.getFirst(),1,5000);
80 case "COMPARE" -> {

83 }
84 }
85 }
86 if(!entries.containsKey("ZONES")) {

88 Arrays.fill(high,floors);
89 }

91 require(starts[i]>=low[i]&&starts[i]<=high[i],"car "+(i+1)+" starts outside its zone");
92 require(park<0||homes[i]>=low[i]&&homes[i]<=high[i],"car "+(i+1)+" home outside its zone");
93 }

107 static List<Event> validate(Config c,List<Raw> raw,List<String> out) {
108 List<Event> events=new ArrayList<>();
109 int last=0,groups=0;
110 boolean fire=false;
111 Set<Integer> gone=new HashSet<>();
112 for(Raw r:raw) {
113 String[] a=r.words;
114 int t=Integer.parseInt(a[0]);
115 String cmd=a[1],problem=null;
116 int[] v=new int[0];
117 if(!ARGS.containsKey(cmd)) problem="unknown command "+cmd;
118 else if(a.length-2!=ARGS.get(cmd)) problem="wrong number of arguments for "+cmd;
119 else if(t<last) problem="time goes backwards";
120 else {
121 v=Arrays.stream(a).skip(2).mapToInt(Integer::parseInt).toArray();
122 int x=v.length==0?0:v[0];
123 switch(cmd) {
124 case "CALL" -> {
125 boolean served=false;
126 for(int i=0;i<c.cars;i++) served|=c.serves(i,x,v[1]);
127 if(x<1||x>c.floors||v[1]<1||v[1]>c.floors) problem="bad floor";
128 else if(x==v[1]) problem="same floor";
129 else if(v[2]<1||v[2]>99) problem="bad count";
130 else if(!served) problem="no car serves "+x+"->"+v[1];
131 else if(fire) problem="fire service active";
132 }
133 case "CANCEL","WHERE" -> {
134 if(x<1||x>groups) problem="unknown group";
135 }
136 case "OUT","IN" -> {
137 if(x<1||x>c.cars) problem="bad car";
138 else if(cmd.equals("OUT")&&gone.contains(x)) problem="car "+x+" already out";
139 else if(cmd.equals("IN")&&!gone.contains(x)) problem="car "+x+" already in";
140 }
141 case "HOLD" -> {
142 if(x<1||x>c.cars) problem="bad car";
143 else if(v[1]<1||v[1]>9) problem="bad ticks";
144 }
145 case "FIRE" -> {
146 if(x<1||x>c.floors) problem="bad floor";
147 else if(fire) problem="fire already active";
148 }
149 case "RESET" -> {
150 if(!fire) problem="fire not active";
151 }
152 }
153 }
154 if(problem!=null) {
155 out.add("line "+r.line+": "+problem);
156 continue;
157 }
158 last=t;
159 switch(cmd) {
160 case "CALL" -> groups++;
161 case "OUT" -> gone.add(v[0]);
162 case "IN" -> gone.remove(v[0]);
163 case "FIRE" -> fire=true;
164 case "RESET" -> fire=false;
165 }
166 events.add(new Event(t,cmd,v));
167 }
168 return events;

233 int cost(Car e,Group g) {
234 int f=e.floor,d=e.direction,x=g.landing;
235 if(policy.equals("NEAREST")||d==0) return Math.abs(f-x);
236 Set<Integer> targets=e.targets();
237 int distance,via=0;
238 if(g.direction==d&&(x-f)*d>=0) {
239 distance=Math.abs(x-f);
240 for(int y:targets) if((y-f)*d>0&&(x-y)*d>0) via++;
241 }
242 else {
243 int m=f;
244 for(int y:targets) m=d>0?Math.max(m,y):Math.min(m,y);
245 distance=Math.abs(f-m)+Math.abs(m-x);
246 for(int y:targets) if((y-f)*d>0) via++;
247 }
248 return policy.equals("COST")?distance:distance+c.door*via+e.door;
249 }
250 boolean assign(Group g,Car excluded) {
251 List<Car> candidates=cars.stream().filter(e -> e.service&&e!=excluded&&c.serves(e.number-1,g.landing,g.destination)).toList();
252 if(candidates.isEmpty()) return false;
253 Car chosen=candidates.getFirst();
254 int best=0;
255 if(policy.equals("ROUND")) {
256 for(Car e:candidates) if(e.number>lastRound) {
257 chosen=e;
258 break;
259 }
260 lastRound=chosen.number;
261 }
262 else {
263 best=cost(chosen,g);
264 for(Car e:candidates) {
265 int value=cost(e,g);
266 if(value<best) {
267 chosen=e;
268 best=value;
269 }
270 }
271 }
272 emit(chosen.label()+" ASSIGN "+g.label()+(policy.equals("ROUND")?"":" cost="+best));
273 if(g.car!=null) g.car.calls.remove(g);
274 chosen.calls.add(g);
275 g.car=chosen;
276 return true;
277 }

366 int direction(Car e) {
367 int f=e.floor,d=e.direction;
368 List<Group> here=bypass(e)?List.of():e.calls.stream().filter(g -> g.landing==f).toList();
369 Set<Integer> targets=e.targets();
370 if(d!=0&&(targets.stream().anyMatch(y -> (y-f)*d>0)||here.stream().anyMatch(g -> g.direction==d))) return d;
371 if(!here.isEmpty()) return here.getFirst().direction;
372 if(d!=0&&targets.stream().anyMatch(y -> (y-f)*d<0)) return -d;
373 if(d==0&&!e.calls.isEmpty()) return Integer.signum(e.calls.getFirst().landing-f);
374 return 0;

527 List<Raw> raw=new ArrayList<>();
528 while(true) {

530 lineNo++;
531 if(line.equals("END")) break;

534 else settings.add(a);
535 }
536 out.add("== "+s.name+" ==");

552 out.add("compare "+p+": delivered="+rerun.delivered+" "+rerun.averages()+" end="+rerun.time+" status="+rerun.status);
553 }
554 }
555 out.add("== ranking ==");
556 List<Scenario> ranked=new ArrayList<>(scenarios.stream().filter(s -> s.outcome().equals("ranked")).toList());
557 ranked.sort((a,b) -> {
558 long aw=a.sim.waitTotal*b.sim.delivered,bw=b.sim.waitTotal*a.sim.delivered;
559 int r=Long.compare(aw,bw);
560 return r!=0?r:Long.compare(a.sim.travelTotal*b.sim.delivered,b.sim.travelTotal*a.sim.delivered);
561 });
