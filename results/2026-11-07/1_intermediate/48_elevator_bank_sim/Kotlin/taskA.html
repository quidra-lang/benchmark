1import kotlin.math.abs
2import kotlin.math.min
3private val POLICIES = listOf("COST", "NEAREST", "ROUND", "ETA")
4private val KEYS = listOf("FLOORS", "CARS", "CAPACITY", "DOOR", "START", "POLICY", "ZONES", "PARK", "PATIENCE", "BYPASS", "PERIOD", "LIMIT", "COMPARE")
5private val ARGS = mapOf("CALL" to 3, "CANCEL" to 1, "OUT" to 1, "IN" to 1, "HOLD" to 2, "FIRE" to 1, "RESET" to 0, "WHERE" to 1, "STATUS" to 0)
6private class Invalid(message: String) : RuntimeException(message)
7private fun requireThat(ok: Boolean, message: String) {
8 if (!ok) throw Invalid(message)
9}
10private fun joined(items: Collection<*>) = items.joinToString(",").ifEmpty {
11 "-"
12}
13private fun avg(total: Long, count: Int): String {
14 if (count == 0) return "-"
15 val n = total * 100 / count
16 return "${n / 100}.${(n % 100).toString().padStart(2, '0')}"
17}
18private fun bound(key: String, text: String, low: Int, high: Int): Int {
19 val n = text.toInt()
20 requireThat(n in low..high, "bad value for $key")
21 return n
22}
23private class Config(lines: List<List<String>>) {
24 var floors = 0
25 var cars = 0
26 var capacity = 0
27 var door = 0
28 var park = -1
29 var patience = -1
30 var bypass = 100
31 var period = -1
32 var limit = 1000
33 var starts = emptyList<Int>()
34 var homes = emptyList<Int>()
35 var zones = emptyList<IntRange>()
36 var policy = "COST"
37 var compare = emptyList<String>()
38 init {
39 val entries = mutableMapOf<String, List<String>>()
40 for (a in lines) {
41 val k = a[0]
42 requireThat(k in KEYS, "unknown key $k")
43 requireThat(k !in entries, "duplicate key $k")
44 entries[k] = a.drop(1)
45 }
46 for (k in KEYS.take(5)) requireThat(k in entries, "missing key $k")
47 for (k in KEYS) {
48 val v = entries[k] ?: continue
49 val n = v.size
50 val countOK = when (k) {
51 "START", "ZONES" -> n == cars
52 "PARK" -> n == cars + 1
53 "COMPARE" -> n in 1..4
54 else -> n == 1
55 }
56 requireThat(countOK, "wrong number of values for $k")
57 when (k) {
58 "FLOORS" -> floors = bound(k, v[0], 2, 50)
59 "CARS" -> cars = bound(k, v[0], 1, 8)
60 "CAPACITY" -> capacity = bound(k, v[0], 1, 20)
61 "DOOR" -> door = bound(k, v[0], 1, 9)
62 "START" -> starts = v.map {
63 bound(k, it, 1, floors)
64 }
65 "POLICY" -> {
66 requireThat(v[0] in POLICIES, "bad value for $k")
67 policy = v[0]
68 }
69 "ZONES" -> zones = v.map {
70 val z = it.split(':').map(String::toInt)
71 requireThat(z[0] >= 1 && z[0] < z[1] && z[1] <= floors, "bad value for $k")
72 z[0]..z[1]
73 }
74 "PARK" -> {
75 park = bound(k, v[0], 0, 100)
76 homes = v.drop(1).map {
77 bound(k, it, 1, floors)
78 }
79 }
80 "PATIENCE" -> patience = bound(k, v[0], 1, 1000)
81 "BYPASS" -> bypass = bound(k, v[0], 50, 100)
82 "PERIOD" -> period = bound(k, v[0], 1, 1000)
83 "LIMIT" -> limit = bound(k, v[0], 1, 5000)
84 "COMPARE" -> {
85 requireThat(v.all {
86 it in POLICIES
87 }
88 && v.toSet().size == v.size, "bad value for $k")
89 compare = v
90 }
91 }
92 }
93 if (zones.isEmpty()) zones = List(cars) {
94 1..floors
95 }
96 for (i in 0 until cars) {
97 requireThat(starts[i] in zones[i], "car ${i + 1} starts outside its zone")
98 requireThat(park < 0 || homes[i] in zones[i], "car ${i + 1} home outside its zone")
99 }
100 }
101 fun serves(i: Int, a: Int, b: Int) = a in zones[i] && b in zones[i]
102 fun header(out: MutableList<String>) {
103 out.add("config: floors=$floors cars=$cars capacity=$capacity door=$door policy=$policy park=${if (park < 0) "off" else park} patience=${if (patience < 0) "off" else patience} bypass=$bypass period=${if (period < 0) "off" else period} limit=$limit compare=${joined(compare)}")
104 for (i in 0 until cars) out.add("E${i + 1}: start=${starts[i]} zone=${zones[i].first}-${zones[i].last} home=${if (park < 0) "-" else homes[i]}")
105 }
106}
107private data class Raw(val line: Int, val words: List<String>)
108private data class Event(val time: Int, val command: String, val values: List<Int>)
109private fun validate(c: Config, raw: List<Raw>, out: MutableList<String>): List<Event> {
110 val events = mutableListOf<Event>()
111 var last = 0
112 var groups = 0
113 var fire = false
114 val gone = mutableSetOf<Int>()
115 for (r in raw) {
116 val a = r.words
117 val t = a[0].toInt()
118 val cmd = a[1]
119 var problem: String? = null
120 var v = emptyList<Int>()
121 if (cmd !in ARGS) problem = "unknown command $cmd"
122 else if (a.size - 2 != ARGS[cmd]) problem = "wrong number of arguments for $cmd"
123 else if (t < last) problem = "time goes backwards"
124 else {
125 v = a.drop(2).map(String::toInt)
126 val x = v.firstOrNull() ?: 0
127 problem = when (cmd) {
128 "CALL" -> when {
129 x !in 1..c.floors || v[1] !in 1..c.floors -> "bad floor"
130 x == v[1] -> "same floor"
131 v[2] !in 1..99 -> "bad count"
132 (0 until c.cars).none {
133 c.serves(it, x, v[1])
134 } -> "no car serves $x->${v[1]}"
135 fire -> "fire service active"
136 else -> null
137 }
138 "CANCEL", "WHERE" -> if (x !in 1..groups) "unknown group" else null
139 "OUT", "IN" -> when {
140 x !in 1..c.cars -> "bad car"
141 cmd == "OUT" && x in gone -> "car $x already out"
142 cmd == "IN" && x !in gone -> "car $x already in"
143 else -> null
144 }
145 "HOLD" -> when {
146 x !in 1..c.cars -> "bad car"
147 v[1] !in 1..9 -> "bad ticks"
148 else -> null
149 }
150 "FIRE" -> when {
151 x !in 1..c.floors -> "bad floor"
152 fire -> "fire already active"
153 else -> null
154 }
155 "RESET" -> if (!fire) "fire not active" else null
156 else -> null
157 }
158 }
159 if (problem != null) {
160 out.add("line ${r.line}: $problem")
161 continue
162 }
163 last = t
164 when (cmd) {
165 "CALL" -> groups++
166 "OUT" -> gone.add(v[0])
167 "IN" -> gone.remove(v[0])
168 "FIRE" -> fire = true
169 "RESET" -> fire = false
170 }
171 events.add(Event(t, cmd, v))
172 }
173 return events
174}

191 val label get() = "E$number"
192 var direction = 0

194 var idleStart = 0

212private class Simulation(val c: Config, val events: List<Event>, val policy: String) {
213 val cars = List(c.cars) {
214 Car(it + 1, c.starts[it])
215 }
216 val groups = mutableListOf<Group>()
217 val pending = mutableListOf<Group>()
218 val log = mutableListOf<String>()
219 var time = 0
220 var fireFloor = 0
221 var lastRound = 0
222 var abandoned = 0
223 var evacuated = 0
224 var delivered = 0
225 var maxWait = 0
226 var waitTotal = 0L
227 var travelTotal = 0L
228 var status = ""
229 fun emit(s: String) {
230 log.add("$time $s")
231 }
232 fun bypass(e: Car) = e.load() * 100 >= c.capacity * c.bypass
233 fun cost(e: Car, g: Group): Int {
234 val f = e.floor
235 val d = e.direction
236 val x = g.landing
237 if (policy == "NEAREST" || d == 0) return abs(f - x)
238 val targets = e.targets()
239 val distance: Int
240 val via: Int
241 if (g.direction == d && (x - f) * d >= 0) {
242 distance = abs(x - f)
243 via = targets.count {
244 (it - f) * d > 0 && (x - it) * d > 0
245 }
246 }
247 else {
248 val m = if (d > 0) (targets + f).max() else (targets + f).min()
249 distance = abs(f - m) + abs(m - x)
250 via = targets.count {
251 (it - f) * d > 0
252 }
253 }
254 return if (policy == "COST") distance else distance + c.door * via + e.door
255 }
256 fun assign(g: Group, excluded: Car? = null): Boolean {
257 val candidates = cars.filter {
258 it.service && it !== excluded && c.serves(it.number - 1, g.landing, g.destination)
259 }
260 if (candidates.isEmpty()) return false
261 val chosen = if (policy == "ROUND") candidates.firstOrNull {
262 it.number > lastRound
263 } ?: candidates.first() else candidates.minBy {
264 cost(it, g)
265 }
266 if (policy == "ROUND") lastRound = chosen.number
267 emit("${chosen.label} ASSIGN ${g.label}" + if (policy == "ROUND") "" else " cost=${cost(chosen, g)}")
268 g.car?.calls?.remove(g)
269 chosen.calls.add(g)
270 g.car = chosen
271 return true
272 }
273 fun dispatch(g: Group) {
274 if (!assign(g)) {
275 pending.add(g)
276 emit("PENDING ${g.label}")
277 }
278 }
279 fun walk(g: Group, word: String) {
280 if (g.car == null) pending.remove(g) else {
281 g.car!!.calls.remove(g)
282 g.car = null
283 }
284 emit("$word ${g.label} x${g.waiting}")
285 g.lost += g.waiting
286 abandoned += g.waiting
287 g.waiting = 0
288 }
289 fun event(event: Event) {
290 val v = event.values
291 val x = v.firstOrNull() ?: 0
292 when (event.command) {
293 "CALL" -> {
294 val g = Group(groups.size + 1, x, v[1], v[2], time)
295 groups.add(g)
296 emit("CALL ${g.label} $x->${v[1]} x${v[2]}")
297 dispatch(g)
298 }
299 "CANCEL" -> {
300 val g = groups[x - 1]
301 if (g.waiting > 0) walk(g, "CANCEL") else emit("CANCEL ${g.label} ignored")
302 }
303 "OUT" -> {
304 val e = cars[x - 1]
305 e.service = false
306 emit("${e.label} OUT")
307 val old = e.calls.toList()
308 e.calls.clear()
309 for (g in old) {
310 g.car = null
311 dispatch(g)
312 }
313 }
314 "IN" -> {
315 val e = cars[x - 1]
316 e.service = true
317 e.idleStart = time
318 emit("${e.label} IN")
319 val old = pending.toList()
320 pending.clear()
321 for (g in old) if (!assign(g)) pending.add(g)
322 }
323 "HOLD" -> {
324 val e = cars[x - 1]
325 if (e.door > 0) {
326 e.door += v[1]
327 emit("${e.label} HOLD ${e.door}")
328 } else emit("${e.label} HOLD ignored")
329 }
330 "FIRE" -> {
331 fireFloor = x
332 emit("FIRE $x")
333 for (g in groups) if (g.waiting > 0) walk(g, "ABANDON")
334 }
335 "RESET" -> {
336 fireFloor = 0
337 emit("RESET")
338 cars.forEach {
339 it.idleStart = time
340 }
341 }
342 "WHERE" -> {
343 val g = groups[x - 1]
344 val riding = cars.mapNotNull {
345 e -> val n = e.passengers.filter {
346 it.group === g
347 }.sumOf {
348 it.count
349 }
350 if (n == 0) null else "${e.label}x$n"
351 }
352 emit("WHERE ${g.label} waiting=${g.waiting} car=${if (g.waiting == 0) "-" else g.car?.label ?: "pending"} riding=${joined(riding)} delivered=${g.delivered} lost=${g.lost}")
353 }
354 "STATUS" -> {
355 for (e in cars) emit("STATUS ${e.label} floor=${e.floor} dir=${if (e.direction == 0) "idle" else if (e.direction > 0) "up" else "down"} door=${e.door} load=${e.load()} stops=${joined(e.passengers.map { it.group.destination }.toSortedSet())} calls=${joined(e.calls.map { it.label })} service=${if (e.service) "in" else "out"}")
356 emit("STATUS pending=${joined(pending.map { it.label })}")
357 }
358 }
359 }
360 fun direction(e: Car): Int {
361 val f = e.floor
362 val d = e.direction
363 val here = if (bypass(e)) emptyList() else e.calls.filter {
364 it.landing == f
365 }
366 val targets = e.targets()
367 if (d != 0 && (targets.any {
368 (it - f) * d > 0
369 }
370 || here.any {
371 it.direction == d
372 })) return d
373 if (here.isNotEmpty()) return here.first().direction
374 if (d != 0 && targets.any {
375 (it - f) * d < 0
376 }) return -d
377 if (d == 0 && e.calls.isNotEmpty()) return (e.calls.first().landing - f).compareTo(0)
378 return 0
379 }
380 fun deliver(p: Passenger) {
381 val g = p.group
382 val wait = p.boarded - g.time
383 val travel = time - g.time
384 g.delivered += p.count
385 g.lastAlight = time
386 g.waitTotal += wait.toLong() * p.count
387 g.travelTotal += travel.toLong() * p.count
388 delivered += p.count
389 waitTotal += wait.toLong() * p.count
390 travelTotal += travel.toLong() * p.count
391 maxWait = maxOf(maxWait, wait)
392 }
393 fun open(e: Car) {
394 emit("${e.label} OPEN ${e.floor}")
395 e.door = c.door
396 e.stops++
397 e.energy += 2
398 val keep = mutableListOf<Passenger>()
399 for (p in e.passengers) {
400 if (p.group.destination == e.floor) {
401 emit("${e.label} ALIGHT ${p.group.label} x${p.count}")
402 deliver(p)
403 }
404 else if (fireFloor != 0) {
405 emit("${e.label} EVACUATE ${p.group.label} x${p.count}")
406 p.group.lost += p.count
407 evacuated += p.count
408 }
409 else keep.add(p)
410 }
411 e.passengers = keep
412 if (fireFloor != 0) return
413 val n = direction(e)
414 if (n != 0) e.direction = n
415 val boarding = e.calls.filter {
416 it.landing == e.floor && it.direction == n
417 }
418 for (g in boarding) {
419 val k = min(c.capacity - e.load(), g.waiting)
420 if (k > 0) {
421 e.passengers.add(Passenger(g, k, time))
422 e.boarded += k
423 g.waiting -= k
424 g.boarded += k
425 g.lastBoard = time
426 g.cars.add(e.number)
427 emit("${e.label} BOARD ${g.label} x$k")
428 }
429 if (g.waiting == 0) {
430 e.calls.remove(g)
431 g.car = null
432 } else {
433 emit("${e.label} LEFT ${g.label} x${g.waiting}")
434 assign(g, e)
435 }
436 }
437 }
438 fun move(e: Car, d: Int) {
439 e.energy += if (d > 0) 3L + e.load() else 1L
440 e.distance++
441 e.floor += d
442 }
443 fun advance(e: Car) {
444 val old = e.floor
445 if (e.door > 0) e.door--
446 if (e.door == 0) {
447 val n = if (fireFloor == 0) direction(e) else (fireFloor - e.floor).compareTo(0)
448 val stop = if (fireFloor != 0) n == 0 && e.passengers.isNotEmpty() else e.passengers.any {
449 it.group.destination == e.floor
450 }
451 || !bypass(e) && e.calls.any {
452 it.landing == e.floor && it.direction == n
453 }
454 if (stop) open(e)
455 else if (n != 0) {
456 e.direction = n
457 move(e, n)
458 }
459 else {
460 if (e.direction != 0) {
461 emit("${e.label} IDLE ${e.floor}")
462 e.direction = 0
463 e.idleStart = time
464 }
465 if (fireFloor == 0 && e.service && c.park >= 0 && e.floor != c.homes[e.number - 1] && time - e.idleStart >= c.park) {
466 move(e, (c.homes[e.number - 1] - e.floor).compareTo(0))
467 if (e.floor == c.homes[e.number - 1]) emit("${e.label} PARK ${e.floor}")
468 }
469 }
470 }
471 if (e.door > 0 || e.floor != old) e.busy++
472 }
473 fun settled(e: Car) = e.door == 0 && e.passengers.isEmpty() && e.calls.isEmpty() && (if (fireFloor != 0) e.floor == fireFloor else !e.service || c.park < 0 || e.floor == c.homes[e.number - 1])
474 fun run() {
475 var index = 0
476 while (true) {
477 if (c.patience >= 0) for (g in groups) if (g.waiting > 0 && time == g.time + c.patience) walk(g, "ABANDON")
478 while (index < events.size && events[index].time == time) event(events[index++])
479 cars.forEach {
480 advance(it)
481 }
482 if (index == events.size && cars.all {
483 settled(it)
484 }) {
485 status = "end"
486 emit("END")
487 return
488 }
489 if (time == c.limit) {
490 status = "abort"
491 emit("ABORT")
492 return
493 }
494 time++
495 }
496 }
497 fun averages() = "avg_wait=${avg(waitTotal, delivered)} avg_travel=${avg(travelTotal, delivered)} max_wait=${if (delivered == 0) "-" else maxWait}"
498 fun breakdown(label: String, gs: List<Group>) = "$label: groups=${gs.size} passengers=${gs.sumOf { it.count }} delivered=${gs.sumOf { it.delivered }} avg_wait=${avg(gs.sumOf { it.waitTotal }, gs.sumOf { it.delivered })}"
499 fun report(out: MutableList<String>) {
500 for (g in groups) out.add("${g.label} t=${g.time} ${g.landing}->${g.destination} n=${g.count} delivered=${g.delivered} lost=${g.lost} wait=${if (g.boarded == g.count) g.lastBoard - g.time else "-"} travel=${if (g.delivered == g.count) g.lastAlight - g.time else "-"} cars=${joined(g.cars)}")
501 for (e in cars) {
502 val u = e.busy.toLong() * 1000 / (time + 1)
503 out.add("${e.label} distance=${e.distance} stops=${e.stops} boarded=${e.boarded} busy=${e.busy} util=${u / 10}.${u % 10}% energy=${e.energy}")
504 }
505 for (f in groups.map {
506 it.landing
507 }.toSortedSet()) out.add(breakdown("F$f", groups.filter {
508 it.landing == f
509 }))
510 if (c.period > 0) for (p in groups.map {
511 it.time / c.period
512 }.toSortedSet()) out.add(breakdown("P$p ${p * c.period}-${p * c.period + c.period - 1}", groups.filter {
513 it.time / c.period == p
514 }))
515 out.add("summary: groups=${groups.size} passengers=${groups.sumOf { it.count }} delivered=$delivered abandoned=$abandoned evacuated=$evacuated ${averages()} end=$time")
516 }
517}
518private class Scenario(val name: String) {
519 var config: Config? = null
520 var sim: Simulation? = null
521 fun outcome() = sim?.let {
522 if (it.status == "abort") "aborted" else if (it.delivered == 0) "empty" else "ranked"
523 } ?: "invalid"
524}
525fun main() {
526 val reader = System.`in`.bufferedReader()
527 val q = reader.readLine().toInt()
528 var lineNo = 1
529 val scenarios = mutableListOf<Scenario>()
530 val out = mutableListOf<String>()
531 repeat(q) {
532 val s = Scenario(reader.readLine().split(' ')[1])
533 lineNo++
534 scenarios.add(s)
535 val settings = mutableListOf<List<String>>()
536 val raw = mutableListOf<Raw>()
537 while (true) {
538 val line = reader.readLine()
539 lineNo++
540 if (line == "END") break
541 val a = line.split(' ')
542 if (a[0][0].isDigit()) raw.add(Raw(lineNo, a)) else settings.add(a)
543 }
544 out.add("== ${s.name} ==")
545 try {
546 val c = Config(settings)
547 s.config = c
548 c.header(out)
549 val events = validate(c, raw, out)
550 val sim = Simulation(c, events, c.policy)
551 s.sim = sim
552 sim.run()
553 out.addAll(sim.log)
554 sim.report(out)
555 for (p in c.compare) {
556 val rerun = Simulation(c, events, p)
557 rerun.run()
558 out.add("compare $p: delivered=${rerun.delivered} ${rerun.averages()} end=${rerun.time} status=${rerun.status}")
559 }
560 } catch (e: Invalid) {
561 out.add("error: ${e.message}")
562 }
563 }
564 out.add("== ranking ==")
565 val ranked = scenarios.filter {
566 it.outcome() == "ranked"
567 }.sortedWith {
568 a, b -> val x = a.sim!!
569 val y = b.sim!!
570 val r = (x.waitTotal * y.delivered).compareTo(y.waitTotal * x.delivered)
571 if (r != 0) r else (x.travelTotal * y.delivered).compareTo(y.travelTotal * x.delivered)
572 }
573 ranked.forEachIndexed {
574 i, s -> out.add("${i + 1}. ${s.name} policy=${s.config!!.policy} ${s.sim!!.averages()}")
575 }
576 for (s in scenarios) if (s.outcome() != "ranked") out.add("- ${s.name} ${s.outcome()}")
577 out.add("== policies ==")
578 for (p in POLICIES) {
579 val runs = ranked.filter {
580 it.config!!.policy == p
581 }.map {
582 it.sim!!
583 }
584 if (runs.isNotEmpty()) {
585 val n = runs.sumOf {
586 it.delivered
587 }
588 out.add("$p: runs=${runs.size} avg_wait=${avg(runs.sumOf { it.waitTotal }, n)} avg_travel=${avg(runs.sumOf { it.travelTotal }, n)}")
589 }
590 }
591 println(out.joinToString("\n"))
592}
