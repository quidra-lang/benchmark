65 def count_matches(self, key, count):
66 if key in ("START", "ZONES"):
67 return count == self.cars
68 if key == "PARK":
69 return count == self.cars + 1
70 if key == "COMPARE":
71 return 1 <= count <= 4
72 return count == 1

74 def zone(self, text):
75 low_text, high_text = text.split(":")
76 low, high = int(low_text), int(high_text)
77 if not 1 <= low < high <= self.floors:
78 raise ConfigError("bad value for ZONES")
79 return low, high

81 def apply(self, key, values):
82 if not self.count_matches(key, len(values)):
83 raise ConfigError(f"wrong number of values for {key}")
84 if key == "FLOORS":
85 self.floors = bounded(key, values[0], 2, 50)
86 elif key == "CARS":
87 self.cars = bounded(key, values[0], 1, 8)
88 elif key == "CAPACITY":
89 self.capacity = bounded(key, values[0], 1, 20)
90 elif key == "DOOR":
91 self.door = bounded(key, values[0], 1, 9)
92 elif key == "START":
93 self.starts = [bounded(key, value, 1, self.floors) for value in values]
94 elif key == "POLICY":
95 if values[0] not in POLICIES:
96 raise ConfigError(f"bad value for {key}")
97 self.policy = values[0]
98 elif key == "ZONES":
99 self.zones = [self.zone(value) for value in values]
100 elif key == "PARK":
101 self.park_wait = bounded(key, values[0], 0, 100)
102 self.homes = [bounded(key, value, 1, self.floors) for value in values[1:]]
103 elif key == "PATIENCE":
104 self.patience = bounded(key, values[0], 1, 1000)
105 elif key == "BYPASS":
106 self.bypass = bounded(key, values[0], 50, 100)
107 elif key == "PERIOD":
108 self.period = bounded(key, values[0], 1, 1000)
109 elif key == "LIMIT":
110 self.limit = bounded(key, values[0], 1, 5000)
111 else:
112 invalid = any(value not in POLICIES for value in values)
113 if invalid or len(set(values)) != len(values):
114 raise ConfigError(f"bad value for {key}")
115 self.compare = values
116
117 def check_cars(self):
118 for index in range(self.cars):
119 low, high = self.zones[index]
120 if not low <= self.starts[index] <= high:
121 raise ConfigError(f"car {index + 1} starts outside its zone")
122 if self.park_wait is not None and not low <= self.homes[index] <= high:
123 raise ConfigError(f"car {index + 1} home outside its zone")
124
125 def home(self, index):
126 return self.homes[index] if self.park_wait is not None else None
127
128 def served(self, landing, destination):
129 return any(
130 low <= landing <= high and low <= destination <= high
131 for low, high in self.zones
132 )
133
134 def header_lines(self):
135 park = "off" if self.park_wait is None else self.park_wait
136 patience = "off" if self.patience is None else self.patience
137 period = "off" if self.period is None else self.period
138 lines = [
139 f"config: floors={self.floors} cars={self.cars} capacity={self.capacity} "
140 f"door={self.door} policy={self.policy} park={park} patience={patience} "
141 f"bypass={self.bypass} period={period} limit={self.limit} "
142 f"compare={joined(self.compare)}"
143 ]
144 for index in range(self.cars):
145 low, high = self.zones[index]
146 home = self.home(index)
147 lines.append(
148 f"E{index + 1}: start={self.starts[index]} zone={low}-{high} "
149 f"home={'-' if home is None else home}"
150 )
151 return lines

154def parse_config(settings):
155 entries = {}
156 for key, *values in settings:
157 if key not in CONFIG_KEYS:
158 raise ConfigError(f"unknown key {key}")
159 if key in entries:
160 raise ConfigError(f"duplicate key {key}")
161 entries[key] = values
162 for key in REQUIRED_KEYS:
163 if key not in entries:
164 raise ConfigError(f"missing key {key}")
165 config = Config()
166 for key in CONFIG_KEYS:
167 if key in entries:
168 config.apply(key, entries[key])
169 if not config.zones:
170 config.zones = [(1, config.floors)] * config.cars
171 config.check_cars()
172 return config

182class EventValidator:
183 def __init__(self, config):
184 self.config = config
185 self.last_time = 0
186 self.groups = 0
187 self.out_cars = set()
188 self.fire = False
189
190 def problem(self, time, command, arguments):
191 if command not in ARGUMENT_COUNTS:
192 return f"unknown command {command}"
193 if len(arguments) != ARGUMENT_COUNTS[command]:
194 return f"wrong number of arguments for {command}"
195 if time < self.last_time:
196 return "time goes backwards"
197 values = [int(argument) for argument in arguments]
198 floors = self.config.floors
199 cars = self.config.cars
200 if command == "CALL":
201 landing, destination, count = values
202 if not (1 <= landing <= floors and 1 <= destination <= floors):
203 return "bad floor"
204 if landing == destination:
205 return "same floor"
206 if not 1 <= count <= 99:
207 return "bad count"
208 if not self.config.served(landing, destination):
209 return f"no car serves {landing}->{destination}"
210 if self.fire:
211 return "fire service active"
212 elif command in ("CANCEL", "WHERE"):
213 if not 1 <= values[0] <= self.groups:
214 return "unknown group"
215 elif command in ("OUT", "IN"):
216 car = values[0]
217 if not 1 <= car <= cars:
218 return "bad car"
219 if command == "OUT" and car in self.out_cars:
220 return f"car {car} already out"
221 if command == "IN" and car not in self.out_cars:
222 return f"car {car} already in"
223 elif command == "HOLD":
224 car, ticks = values
225 if not 1 <= car <= cars:
226 return "bad car"
227 if not 1 <= ticks <= 9:
228 return "bad ticks"
229 elif command == "FIRE":
230 if not 1 <= values[0] <= floors:
231 return "bad floor"
232 if self.fire:
233 return "fire already active"
234 elif command == "RESET" and not self.fire:
235 return "fire not active"
236 return None
237
238 def accept(self, event):
239 self.last_time = event.time
240 if event.command == "CALL":
241 self.groups += 1
242 elif event.command == "OUT":
243 self.out_cars.add(event.values[0])
244 elif event.command == "IN":
245 self.out_cars.discard(event.values[0])
246 elif event.command == "FIRE":
247 self.fire = True
248 elif event.command == "RESET":
249 self.fire = False
250
251
252def validate_events(config, event_lines, output):
253 validator = EventValidator(config)
254 accepted = []
255 for line, words in event_lines:
256 time, command, arguments = int(words[0]), words[1], words[2:]
257 message = validator.problem(time, command, arguments)
258 if message is not None:
259 output.append(f"line {line}: {message}")
260 continue
261 event = Event(time, command, [int(argument) for argument in arguments])
262 validator.accept(event)
263 accepted.append(event)
264 return accepted

340 def __init__(self, config, events, policy):
341 self.config = config
342 self.events = events
343 self.policy = policy
344 self.cars = [

365 def run(self):
366 handlers = {
367 "CALL": self.call,
368 "CANCEL": self.cancel,
369 "OUT": self.take_out,
370 "IN": self.put_in,
371 "HOLD": self.hold,
372 "FIRE": self.fire,
373 "RESET": self.reset,
374 "WHERE": self.where,
375 "STATUS": self.show_status,
376 }
377 index = 0
378 while True:
379 if self.config.patience is not None:
380 for group in self.groups:
381 if group.waiting > 0 and self.time == group.time + self.config.patience:
382 self.walk_away(group, "ABANDON")
383 while index < len(self.events) and self.events[index].time == self.time:
384 event = self.events[index]
385 handlers[event.command](event.values)
386 index += 1
387 for car in self.cars:
388 self.advance(car)
389 if index == len(self.events) and all(self.settled(car) for car in self.cars):
390 self.status = "end"
391 self.emit("END")
392 return
393 if self.time == self.config.limit:
394 self.status = "abort"
395 self.emit("ABORT")
396 return
397 self.time += 1

399 def settled(self, car):
400 if car.door > 0 or car.passengers or car.calls:
401 return False
402 if self.fire_floor is not None:
403 return car.floor == self.fire_floor
404 return not car.in_service or car.home is None or car.floor == car.home

406 def bypassing(self, car):
407 return car.load() * 100 >= self.config.capacity * self.config.bypass

409 def cost(self, car, group):
410 floor = car.floor
411 direction = car.direction
412 landing = group.landing
413 if self.policy == "NEAREST" or direction == 0:
414 return abs(floor - landing)
415 targets = car.target_floors()
416 if group.direction == direction and (landing - floor) * direction >= 0:
417 distance = abs(landing - floor)
418 via = sum(
419 1 for target in targets
420 if (target - floor) * direction > 0 and (landing - target) * direction > 0
421 )
422 else:
423 targets_and_floor = targets | {floor}
424 turn = max(targets_and_floor) if direction > 0 else min(targets_and_floor)
425 distance = abs(floor - turn) + abs(turn - landing)
426 via = sum(1 for target in targets if (target - floor) * direction > 0)
427 if self.policy == "COST":
428 return distance
429 return distance + self.config.door * via + car.door

431 def assign(self, group, excluded=None):
432 candidates = [
433 car for car in self.cars
434 if car.in_service and car is not excluded and car.serves(group)
435 ]
436 if not candidates:
437 return False
438 if self.policy == "ROUND":
439 chosen = next(
440 (car for car in candidates if car.number > self.last_round),
441 candidates[0],
442 )
443 self.last_round = chosen.number
444 self.emit(chosen.label, "ASSIGN", group.label)
445 else:
446 costs = [self.cost(car, group) for car in candidates]
447 best = min(costs)
448 chosen = candidates[costs.index(best)]
449 self.emit(chosen.label, "ASSIGN", group.label, f"cost={best}")
450 if group.car is not None:
451 group.car.calls.remove(group)
452 chosen.calls.append(group)
453 group.car = chosen
454 return True

559 def choose_direction(self, car):
560 floor = car.floor
561 direction = car.direction
562 here = []
563 if not self.bypassing(car):
564 here = [group for group in car.calls if group.landing == floor]
565 targets = car.target_floors()
566 if direction != 0:
567 ahead = any((target - floor) * direction > 0 for target in targets)
568 if ahead or any(group.direction == direction for group in here):
569 return direction
570 if here:
571 return here[0].direction
572 if direction != 0 and any((target - floor) * direction < 0 for target in targets):
573 return -direction
574 if direction == 0 and car.calls:
575 return sign(car.calls[0].landing - floor)
576 return 0

578 def should_stop(self, car, heading):
579 if any(entry.group.destination == car.floor for entry in car.passengers):
580 return True
581 if self.bypassing(car):
582 return False
583 return any(
584 group.landing == car.floor and group.direction == heading
585 for group in car.calls

588 def advance(self, car):
589 floor = car.floor
590 if car.door > 0:
591 car.door -= 1
592 if car.door == 0:
593 self.decide(car)
594 if car.door > 0 or car.floor != floor:
595 car.busy += 1

597 def decide(self, car):
598 if self.fire_floor is not None:
599 heading = sign(self.fire_floor - car.floor)
600 stopping = heading == 0 and len(car.passengers) > 0
601 else:
602 heading = self.choose_direction(car)
603 stopping = self.should_stop(car, heading)
604 if stopping:
605 self.open_doors(car)
606 elif heading != 0:
607 car.direction = heading
608 self.move(car, heading)
609 else:
610 if car.direction != 0:
611 self.emit(car.label, "IDLE", car.floor)
612 car.direction = 0
613 car.idle_start = self.time
614 self.park(car)

616 def move(self, car, step):
617 if step > 0:
618 car.energy += 3 + car.load()
619 else:
620 car.energy += 1
621 car.distance += 1
622 car.floor += step

624 def park(self, car):
625 if self.fire_floor is not None or not car.in_service or car.home is None:
626 return
627 if car.floor == car.home or self.time - car.idle_start < self.config.park_wait:
628 return
629 self.move(car, sign(car.home - car.floor))
630 if car.floor == car.home:
631 self.emit(car.label, "PARK", car.home)
632
633 def open_doors(self, car):
634 self.emit(car.label, "OPEN", car.floor)
635 car.door = self.config.door
636 car.stops += 1
637 car.energy += 2
638 riding = []
639 for entry in car.passengers:
640 if entry.group.destination == car.floor:
641 self.emit(car.label, "ALIGHT", entry.group.label, f"x{entry.count}")
642 self.deliver(entry)
643 elif self.fire_floor is not None:
644 self.emit(car.label, "EVACUATE", entry.group.label, f"x{entry.count}")
645 entry.group.lost += entry.count
646 self.evacuated += entry.count
647 else:
648 riding.append(entry)
649 car.passengers = riding
650 if self.fire_floor is not None:
651 return
652 heading = self.choose_direction(car)
653 if heading != 0:
654 car.direction = heading
655 boarding = [
656 group for group in car.calls
657 if group.landing == car.floor and group.direction == heading
658 ]
659 for group in boarding:
660 self.board(car, group)

662 def board(self, car, group):
663 count = min(self.config.capacity - car.load(), group.waiting)
664 if count >= 1:
665 car.passengers.append(PassengerEntry(group, count, self.time))
666 car.boarded += count
667 group.waiting -= count
668 group.boarded += count
669 group.last_board = self.time
670 group.cars.add(car.number)
671 self.emit(car.label, "BOARD", group.label, f"x{count}")
672 if group.waiting == 0:
673 car.calls.remove(group)
674 group.car = None
675 else:
676 self.emit(car.label, "LEFT", group.label, f"x{group.waiting}")
677 self.assign(group, car)

679 def deliver(self, entry):
680 group = entry.group
681 wait = entry.board_time - group.time
682 travel = self.time - group.time
683 group.delivered += entry.count
684 group.last_alight = self.time
685 group.wait_total += wait * entry.count
686 group.travel_total += travel * entry.count
687 self.delivered += entry.count
688 self.wait_total += wait * entry.count
689 self.travel_total += travel * entry.count
690 self.max_wait = max(self.max_wait, wait)

692 def averages(self):
693 max_wait = self.max_wait if self.delivered > 0 else "-"
694 return (
695 f"avg_wait={average(self.wait_total, self.delivered)} "
696 f"avg_travel={average(self.travel_total, self.delivered)} max_wait={max_wait}"

699 def report(self, output):
700 for group in self.groups:
701 wait = group.last_board - group.time if group.boarded == group.count else "-"
702 travel = group.last_alight - group.time if group.delivered == group.count else "-"
703 cars = joined([str(number) for number in sorted(group.cars)])
704 output.append(
705 f"{group.label} t={group.time} {group.landing}->{group.destination} "
706 f"n={group.count} delivered={group.delivered} lost={group.lost} "
707 f"wait={wait} travel={travel} cars={cars}"
708 )
709 for car in self.cars:
710 tenths = car.busy * 1000 // (self.time + 1)
711 output.append(
712 f"{car.label} distance={car.distance} stops={car.stops} "
713 f"boarded={car.boarded} busy={car.busy} "
714 f"util={tenths // 10}.{tenths % 10}% energy={car.energy}"
715 )
716 for floor in sorted({group.landing for group in self.groups}):
717 floor_groups = [group for group in self.groups if group.landing == floor]
718 output.append(breakdown(f"F{floor}", floor_groups))
719 width = self.config.period
720 if width is not None:
721 for period in sorted({group.time // width for group in self.groups}):
722 period_groups = [group for group in self.groups if group.time // width == period]
723 label = f"P{period} {period * width}-{period * width + width - 1}"
724 output.append(breakdown(label, period_groups))
725 passengers = sum(group.count for group in self.groups)
726 output.append(
727 f"summary: groups={len(self.groups)} passengers={passengers} "
728 f"delivered={self.delivered} abandoned={self.abandoned} "
729 f"evacuated={self.evacuated} {self.averages()} end={self.time}"
730 )

733class Scenario:
734 def __init__(self, name, settings, event_lines):
735 self.name = name
736 self.settings = settings
737 self.event_lines = event_lines
738 self.config = None
739 self.simulation = None
740
741 def outcome(self):
742 if self.simulation is None:
743 return "invalid"
744 if self.simulation.status == "abort":
745 return "aborted"
746 if self.simulation.delivered == 0:
747 return "empty"
748 return "ranked"
749
750 def process(self, output):
751 output.append(f"== {self.name} ==")
752 try:
753 self.config = parse_config(self.settings)
754 except ConfigError as error:
755 output.append(f"error: {error}")
756 return
757 output.extend(self.config.header_lines())
758 events = validate_events(self.config, self.event_lines, output)
759 self.simulation = Simulation(self.config, events, self.config.policy)
760 self.simulation.run()
761 output.extend(self.simulation.log)
762 self.simulation.report(output)
763 for policy in self.config.compare:
764 rerun = Simulation(self.config, events, policy)
765 rerun.run()
766 output.append(
767 f"compare {policy}: delivered={rerun.delivered} {rerun.averages()} "
768 f"end={rerun.time} status={rerun.status}"
769 )

772def read_scenarios(lines):
773 scenarios = []
774 index = 1
775 for _ in range(int(lines[0])):
776 name = lines[index].split(" ")[1]
777 index += 1
778 settings = []
779 event_lines = []
780 while lines[index] != "END":
781 words = lines[index].split(" ")
782 if words[0][0].isdigit():
783 event_lines.append((index + 1, words))
784 else:
785 settings.append(words)
786 index += 1
787 index += 1
788 scenarios.append(Scenario(name, settings, event_lines))
789 return scenarios

792def ranking_lines(scenarios, output):
793 output.append("== ranking ==")
794 ranked = [scenario for scenario in scenarios if scenario.outcome() == "ranked"]
795 ranked.sort(key=lambda scenario: (
796 Fraction(scenario.simulation.wait_total, scenario.simulation.delivered),
797 Fraction(scenario.simulation.travel_total, scenario.simulation.delivered),
798 ))
799 for rank, scenario in enumerate(ranked, 1):
800 output.append(
801 f"{rank}. {scenario.name} policy={scenario.config.policy} "
802 f"{scenario.simulation.averages()}"
803 )
804 for scenario in scenarios:
805 outcome = scenario.outcome()
806 if outcome != "ranked":
807 output.append(f"- {scenario.name} {outcome}")
808 output.append("== policies ==")
809 for policy in POLICIES:
810 runs = [scenario.simulation for scenario in ranked if scenario.config.policy == policy]
811 if not runs:
812 continue
813 delivered = sum(run.delivered for run in runs)
814 wait_total = sum(run.wait_total for run in runs)
815 travel_total = sum(run.travel_total for run in runs)
816 output.append(
817 f"{policy}: runs={len(runs)} avg_wait={average(wait_total, delivered)} "
818 f"avg_travel={average(travel_total, delivered)}"
819 )
