38function configuration(lines) {
39 const entries = new Map();
40 for (const [key, ...values] of lines) {
41 ensure(!keys.includes(key), `unknown key ${key}`);
42 ensure(entries.has(key), `duplicate key ${key}`);
43 entries.set(key, values);
44 }
45 for (const key of keys.slice(0, 5)) ensure(!entries.has(key), `missing key ${key}`);
46 const c = {
47 floors: 0,
48 cars: 0,
49 capacity: 0,
50 door: 0,
51 starts: [],
52 policy: "COST",
53 zones: [],
54 park: -1,
55 homes: [],
56 patience: -1,
57 bypass: 100,
58 period: -1,
59 limit: 1000,
60 compare: [],
61 };
62 for (const key of keys) {
63 const a = entries.get(key);
64 if (!a) continue;
65 const n = key === "START" || key === "ZONES" ? c.cars : key === "PARK" ? c.cars + 1 : 1;
66 ensure(
67 key === "COMPARE" ? a.length < 1 || a.length > 4 : a.length !== n,
68 `wrong number of values for ${key}`,
69 );
70 const bounded = (text, lo, hi) => {
71 const n = Number(text);
72 ensure(n < lo || n > hi, `bad value for ${key}`);
73 return n;
74 };
75 switch (key) {
76 case "FLOORS":
77 c.floors = bounded(a[0], 2, 50);
78 break;
79 case "CARS":
80 c.cars = bounded(a[0], 1, 8);
81 break;
82 case "CAPACITY":
83 c.capacity = bounded(a[0], 1, 20);
84 break;
85 case "DOOR":
86 c.door = bounded(a[0], 1, 9);
87 break;
88 case "START":
89 c.starts = a.map((x) => bounded(x, 1, c.floors));
90 break;
91 case "POLICY":
92 ensure(!policies.includes(a[0]), "bad value for POLICY");
93 c.policy = a[0];
94 break;
95 case "ZONES":
96 c.zones = a.map((x) => {
97 const [lo, hi] = x.split(":").map(Number);
98 ensure(lo < 1 || lo >= hi || hi > c.floors, "bad value for ZONES");
99 return [lo, hi];
100 });
101 break;
102 case "PARK":
103 c.park = bounded(a[0], 0, 100);
104 c.homes = a.slice(1).map((x) => bounded(x, 1, c.floors));
105 break;
106 case "PATIENCE":
107 c.patience = bounded(a[0], 1, 1000);
108 break;
109 case "BYPASS":
110 c.bypass = bounded(a[0], 50, 100);
111 break;
112 case "PERIOD":
113 c.period = bounded(a[0], 1, 1000);
114 break;
115 case "LIMIT":
116 c.limit = bounded(a[0], 1, 5000);
117 break;
118 case "COMPARE":
119 ensure(
120 a.some((x) => !policies.includes(x)) || new Set(a).size !== a.length,
121 "bad value for COMPARE",
122 );
123 c.compare = a;
124 break;
125 }
126 }
127 if (!c.zones.length) c.zones = c.starts.map(() => [1, c.floors]);
128 for (let i = 0; i < c.cars; ++i) {
129 const [lo, hi] = c.zones[i];
130 ensure(c.starts[i] < lo || c.starts[i] > hi, `car ${i + 1} starts outside its zone`);
131 ensure(
132 c.park >= 0 && (c.homes[i] < lo || c.homes[i] > hi),
133 `car ${i + 1} home outside its zone`,
134 );
135 }
136 return c;

138const serves = (c, i, a, b) => c.zones[i][0] <= Math.min(a, b) && Math.max(a, b) <= c.zones[i][1];
139function events(c, lines, output) {
140 const accepted = [];
141 let last = 0,
142 groups = 0,
143 fire = false;
144 const out = new Set();
145 for (const line of lines) {
146 const [time, command, ...args] = line.words;
147 const t = Number(time),
148 v = args.map(Number),
149 [a, b, n] = v;
150 let error = "";
151 if (!(command in counts)) error = `unknown command ${command}`;
152 else if (args.length !== counts[command])
153 error = `wrong number of arguments for ${command}`;
154 else if (t < last) error = "time goes backwards";
155 else if (command === "CALL") {
156 if (a < 1 || a > c.floors || b < 1 || b > c.floors) error = "bad floor";
157 else if (a === b) error = "same floor";
158 else if (n < 1 || n > 99) error = "bad count";
159 else if (!c.starts.some((_, i) => serves(c, i, a, b)))
160 error = `no car serves ${a}->${b}`;
161 else if (fire) error = "fire service active";
162 } else if (command === "CANCEL" || command === "WHERE") {
163 if (a < 1 || a > groups) error = "unknown group";
164 } else if (["OUT", "IN", "HOLD"].includes(command)) {
165 if (a < 1 || a > c.cars) error = "bad car";
166 else if (command === "OUT" && out.has(a)) error = `car ${a} already out`;
167 else if (command === "IN" && !out.has(a)) error = `car ${a} already in`;
168 else if (command === "HOLD" && (b < 1 || b > 9)) error = "bad ticks";
169 } else if (command === "FIRE") {
170 if (a < 1 || a > c.floors) error = "bad floor";
171 else if (fire) error = "fire already active";
172 } else if (command === "RESET" && !fire) error = "fire not active";
173 if (error) {
174 output.push(`line ${line.number}: ${error}`);
175 continue;
176 }
177 accepted.push({
178 time: t,
179 command,
180 values: v,
181 });
182 last = t;
183 if (command === "CALL") ++groups;
184 if (command === "OUT") out.add(a);
185 if (command === "IN") out.delete(a);
186 if (command === "FIRE") fire = true;
187 if (command === "RESET") fire = false;
188 }
189 return accepted;
190}

193 groups: [],

198 inService: true,

244 function assign(g, exclude = -1) {
245 const group = r.groups[g],
246 candidates = r.cars
247 .map((car, i) => ({
248 car,
249 i,

251 .filter(

255 const cost = (car) => {
256 const f = car.floor,
257 d = car.direction,
258 x = group.a,
259 u = direction(group),
260 s = targets(car);
261 if (policy === "NEAREST" || !d) return Math.abs(f - x);
262 let distance, via;
263 if (u === d && (x - f) * d >= 0) {
264 distance = Math.abs(x - f);
265 via = s.filter((y) => (y - f) * d > 0 && (x - y) * d > 0).length;
266 } else {
267 const m = d > 0 ? Math.max(f, ...s) : Math.min(f, ...s);
268 distance = Math.abs(f - m) + Math.abs(m - x);
269 via = s.filter((y) => (y - f) * d > 0).length;
270 }
271 return distance + (policy === "ETA" ? c.door * via + car.door : 0);
272 };
273 let chosen = candidates[0];

275 else for (const x of candidates) if (cost(x.car) < cost(chosen.car)) chosen = x;
276 const chosenCost = cost(chosen.car);
277 remove(g);
278 group.car = chosen.i;
279 chosen.car.calls.push(g);
280 if (policy === "ROUND") round = chosen.i;
281 emit(
282 `E${chosen.i + 1} ASSIGN G${g + 1}` + (policy === "ROUND" ? "" : ` cost=${chosenCost}`),

286 const makePending = (g) => {

288 pending.push(g);

291 const abandon = (g, word) => {

293 remove(g);

301 d = car.direction,

313 if (d && s.some((y) => (y - f) * d < 0)) return -d;
314 if (!d && car.calls.length) return Math.sign(r.groups[car.calls[0]].a - f);
315 return 0;

403 );

418 d = car.direction,

423 (load(car) * 100 < c.capacity * c.bypass &&
424 car.calls.some(

448 g.lost += p.count;

450 } else riding.push(p);
451 }

466 time,

469 g.boarded += k;

486 } else {

522 }
523 if (time === c.limit) {
524 emit("ABORT");

526 }
527 }
