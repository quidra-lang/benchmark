113function configuration(lines: string[][]): Config {
114 const entries = new Map<string, string[]>();
115 for (const [key, ...values] of lines) {
116 ensure(!keys.includes(key), `unknown key ${key}`);
117 ensure(entries.has(key), `duplicate key ${key}`);
118 entries.set(key, values);
119 }
120 for (const key of keys.slice(0, 5)) ensure(!entries.has(key), `missing key ${key}`);
121 const c: Config = {
122 floors: 0,
123 cars: 0,
124 capacity: 0,
125 door: 0,
126 starts: [],
127 policy: "COST",
128 zones: [],
129 park: -1,
130 homes: [],
131 patience: -1,
132 bypass: 100,
133 period: -1,
134 limit: 1000,
135 compare: [],
136 };
137 for (const key of keys) {
138 const a = entries.get(key);
139 if (!a) continue;
140 const n = key === "START" || key === "ZONES" ? c.cars : key === "PARK" ? c.cars + 1 : 1;
141 ensure(
142 key === "COMPARE" ? a.length < 1 || a.length > 4 : a.length !== n,
143 `wrong number of values for ${key}`,
144 );
145 const bounded = (text: string, lo: number, hi: number) => {
146 const n = Number(text);
147 ensure(n < lo || n > hi, `bad value for ${key}`);
148 return n;
149 };
150 switch (key) {
151 case "FLOORS":
152 c.floors = bounded(a[0], 2, 50);
153 break;
154 case "CARS":
155 c.cars = bounded(a[0], 1, 8);
156 break;
157 case "CAPACITY":
158 c.capacity = bounded(a[0], 1, 20);
159 break;
160 case "DOOR":
161 c.door = bounded(a[0], 1, 9);
162 break;
163 case "START":
164 c.starts = a.map((x) => bounded(x, 1, c.floors));
165 break;
166 case "POLICY":
167 ensure(!policies.includes(a[0]), "bad value for POLICY");
168 c.policy = a[0];
169 break;
170 case "ZONES":
171 c.zones = a.map((x) => {
172 const [lo, hi] = x.split(":").map(Number);
173 ensure(lo < 1 || lo >= hi || hi > c.floors, "bad value for ZONES");
174 return [lo, hi];
175 });
176 break;
177 case "PARK":
178 c.park = bounded(a[0], 0, 100);
179 c.homes = a.slice(1).map((x) => bounded(x, 1, c.floors));
180 break;
181 case "PATIENCE":
182 c.patience = bounded(a[0], 1, 1000);
183 break;
184 case "BYPASS":
185 c.bypass = bounded(a[0], 50, 100);
186 break;
187 case "PERIOD":
188 c.period = bounded(a[0], 1, 1000);
189 break;
190 case "LIMIT":
191 c.limit = bounded(a[0], 1, 5000);
192 break;
193 case "COMPARE":
194 ensure(
195 a.some((x) => !policies.includes(x)) || new Set(a).size !== a.length,
196 "bad value for COMPARE",
197 );
198 c.compare = a;
199 break;
200 }
201 }
202 if (!c.zones.length) c.zones = c.starts.map(() => [1, c.floors]);
203 for (let i = 0; i < c.cars; ++i) {
204 const [lo, hi] = c.zones[i];
205 ensure(c.starts[i] < lo || c.starts[i] > hi, `car ${i + 1} starts outside its zone`);
206 ensure(
207 c.park >= 0 && (c.homes[i] < lo || c.homes[i] > hi),
208 `car ${i + 1} home outside its zone`,
209 );
210 }
211 return c;

213const serves = (c: Config, i: number, a: number, b: number) =>

215function events(
216 c: Config,
217 lines: {
218 number: number;
219 words: string[];
220 }[],
221 output: string[],
222) {
223 const accepted: Event[] = [];
224 let last = 0,
225 groups = 0,
226 fire = false;
227 const out = new Set<number>();
228 for (const line of lines) {
229 const [time, command, ...args] = line.words;
230 const t = Number(time),
231 v = args.map(Number),
232 [a, b, n] = v;
233 let error = "";
234 if (!(command in counts)) error = `unknown command ${command}`;
235 else if (args.length !== counts[command])
236 error = `wrong number of arguments for ${command}`;
237 else if (t < last) error = "time goes backwards";
238 else if (command === "CALL") {
239 if (a < 1 || a > c.floors || b < 1 || b > c.floors) error = "bad floor";
240 else if (a === b) error = "same floor";
241 else if (n < 1 || n > 99) error = "bad count";
242 else if (!c.starts.some((_, i) => serves(c, i, a, b)))
243 error = `no car serves ${a}->${b}`;
244 else if (fire) error = "fire service active";
245 } else if (command === "CANCEL" || command === "WHERE") {
246 if (a < 1 || a > groups) error = "unknown group";
247 } else if (["OUT", "IN", "HOLD"].includes(command)) {
248 if (a < 1 || a > c.cars) error = "bad car";
249 else if (command === "OUT" && out.has(a)) error = `car ${a} already out`;
250 else if (command === "IN" && !out.has(a)) error = `car ${a} already in`;
251 else if (command === "HOLD" && (b < 1 || b > 9)) error = "bad ticks";
252 } else if (command === "FIRE") {
253 if (a < 1 || a > c.floors) error = "bad floor";
254 else if (fire) error = "fire already active";
255 } else if (command === "RESET" && !fire) error = "fire not active";
256 if (error) {
257 output.push(`line ${line.number}: ${error}`);
258 continue;
259 }
260 accepted.push({
261 time: t,
262 command,
263 values: v,
264 });
265 last = t;
266 if (command === "CALL") ++groups;
267 if (command === "OUT") out.add(a);
268 if (command === "IN") out.delete(a);
269 if (command === "FIRE") fire = true;
270 if (command === "RESET") fire = false;
271 }
272 return accepted;
273}

687 `config: floors=${c.floors} cars=${c.cars} capacity=${c.capacity} door=${c.door} policy=${c.policy} park=${c.park < 0 ? "off" : c.park} patience=${c.patience < 0 ? "off" : c.patience} bypass=${c.bypass} period=${c.period < 0 ? "off" : c.period} limit=${c.limit} compare=${join(c.compare)}`,
688 );
689 c.starts.forEach((floor, i) =>
690 output.push(
691 `E${i + 1}: start=${floor} zone=${c.zones[i][0]}-${c.zones[i][1]} home=${c.park < 0 ? "-" : c.homes[i]}`,
692 ),
693 );
694 const input = events(c, raw, output),
695 r = simulate(c, input, c.policy);
696 for (const line of r.log) output.push(line);
697 statistics(c, r, output);

700 config: c,
701 result: r,

703 for (const policy of c.compare) {

706 `compare ${policy}: delivered=${x.delivered} avg_wait=${avg(x.wait, x.delivered)} avg_travel=${avg(x.travel, x.delivered)} max_wait=${x.delivered ? x.maxWait : "-"} end=${x.end} status=${x.aborted ? "abort" : "end"}`,
707 );

710output.push("== ranking ==");
711const ranked = scenarios.filter(
712 (
713 s,
714 ): s is Scenario & {
715 config: Config;
716 result: Result;
717 } => s.config !== null && s.result !== null && !s.result.aborted && s.result.delivered > 0,
718);
719ranked.sort(
720 (a, b) =>
721 a.result.wait * b.result.delivered - b.result.wait * a.result.delivered ||
722 a.result.travel * b.result.delivered - b.result.travel * a.result.delivered,
723);
724ranked.forEach((s, i) =>
725 output.push(
726 `${i + 1}. ${s.name} policy=${s.config.policy} avg_wait=${avg(s.result.wait, s.result.delivered)} avg_travel=${avg(s.result.travel, s.result.delivered)} max_wait=${s.result.maxWait}`,
727 ),
728);
729for (const s of scenarios)
730 if (
731 !ranked.includes(
732 s as Scenario & {
733 config: Config;
734 result: Result;
735 },
736 )
737 )
738 output.push(
739 `- ${s.name} ${!s.result ? "invalid" : s.result.aborted ? "aborted" : "empty"}`,
740 );
741output.push("== policies ==");
742for (const p of policies) {
743 const ss = ranked.filter((s) => s.config.policy === p);
744 if (ss.length) {
745 const n = ss.reduce((n, s) => n + s.result.delivered, 0);
746 output.push(
747 `${p}: runs=${ss.length} avg_wait=${avg(
748 ss.reduce((n, s) => n + s.result.wait, 0),
749 n,
750 )} avg_travel=${avg(
751 ss.reduce((n, s) => n + s.result.travel, 0),
752 n,
753 )}`,
754 );
755 }
756}
757process.stdout.write(output.join("\n") + "\n");
