13def config_error(bad, message) = (raise ConfigError, message if bad)
14def parse_config(lines)
15 entries = {}
16 lines.each do |key, *vs|
17 config_error(!CONFIG_KEYS.include?(key), "unknown key #{key}")
18 config_error(entries.key?(key), "duplicate key #{key}")
19 entries[key] = vs
20 end
21 CONFIG_KEYS.take(5).each { |k| config_error(!entries.key?(k), "missing key #{k}") }
22 c = {'POLICY'=>'COST','BYPASS'=>100,'LIMIT'=>1000,'COMPARE'=>[]}
23 CONFIG_KEYS.each do |k|
24 next unless entries.key?(k)
25 vs = entries[k]
26 e = c['CARS']
27 f = c['FLOORS']
28 valid_count = case k
29 when 'START', 'ZONES'
30 vs.length == e
31 when 'PARK'
32 vs.length == e + 1
33 when 'COMPARE'
34 (1..4).include?(vs.length)
35 else
36 vs.length == 1
37 end
38 config_error(!valid_count, "wrong number of values for #{k}")
39 ranges = {'FLOORS'=>2..50,'CARS'=>1..8,'CAPACITY'=>1..20,'DOOR'=>1..9,'PATIENCE'=>1..1000,'BYPASS'=>50..100,'PERIOD'=>1..1000,'LIMIT'=>1..5000}
40 if ranges.key?(k)
41 n = vs[0].to_i
42 config_error(!ranges[k].include?(n), "bad value for #{k}")
43 c[k] = n
44 elsif k == 'POLICY'
45 config_error(!POLICIES.include?(vs[0]), "bad value for #{k}")
46 c[k] = vs[0]
47 elsif k == 'COMPARE'
48 config_error(vs.uniq.length != vs.length || vs.any? { |v| !POLICIES.include?(v) }, "bad value for #{k}")
49 c[k] = vs
50 elsif k == 'ZONES'
51 zs = vs.map { |v| v.split(':').map(&:to_i) }
52 config_error(zs.any? { |a, b| !(1 <= a && a < b && b <= f) }, "bad value for #{k}")
53 c[k] = zs
54 else
55 ns = vs.map(&:to_i)
56 valid = k == 'PARK' ? (0..100).include?(ns[0]) && ns.drop(1).all? { |n| (1..f).include?(n) } : ns.all? { |n| (1..f).include?(n) }
57 config_error(!valid, "bad value for #{k}")
58 c[k] = ns
59 end
60 end
61 c['ZONES'] ||= Array.new(c['CARS']) { [1, c['FLOORS']] }
62 c['CARS'].times do |i|
63 a, b = c['ZONES'][i]
64 config_error(!(a..b).include?(c['START'][i]), "car #{i + 1} starts outside its zone")
65 config_error(c['PARK'] && !(a..b).include?(c['PARK'][i + 1]), "car #{i + 1} home outside its zone")
66 end
67 c
68end
69def validate_events(c, lines)
70 accepted = []
71 last = groups = 0
72 out = []
73 fire = false
74 lines.each do |line_number, words|
75 time = words[0].to_i
76 cmd = words[1]
77 args = words.drop(2)
78 v = args.map(&:to_i)
79 error = nil
80 if !ARG_COUNTS.key?(cmd)
81 error = "unknown command #{cmd}"
82 elsif args.length != ARG_COUNTS[cmd]
83 error = "wrong number of arguments for #{cmd}"
84 elsif time < last
85 error = 'time goes backwards'
86 else
87 case cmd
88 when 'CALL'
89 a, b, n = v
90 error = if !((1..c['FLOORS']).include?(a) && (1..c['FLOORS']).include?(b))
91 'bad floor'
92 elsif a == b
93 'same floor'
94 elsif !(1..99).include?(n)
95 'bad count'
96 elsif !c['ZONES'].any? { |lo, hi| (lo..hi).include?(a) && (lo..hi).include?(b) }
97 "no car serves #{a}->#{b}"
98 elsif fire
99 'fire service active'
100 end
101 when 'CANCEL', 'WHERE'
102 error = 'unknown group' unless (1..groups).include?(v[0])
103 when 'OUT', 'IN'
104 error = if !(1..c['CARS']).include?(v[0])
105 'bad car'
106 elsif cmd == 'OUT' && out.include?(v[0])
107 "car #{v[0]} already out"
108 elsif cmd == 'IN' && !out.include?(v[0])
109 "car #{v[0]} already in"
110 end
111 when 'HOLD'
112 error = !(1..c['CARS']).include?(v[0]) ? 'bad car' : !(1..9).include?(v[1]) ? 'bad ticks' : nil
113 when 'FIRE'
114 error = !(1..c['FLOORS']).include?(v[0]) ? 'bad floor' : fire ? 'fire already active' : nil
115 when 'RESET'
116 error = 'fire not active' unless fire
117 end
118 end
119 if error
120 puts "line #{line_number}: #{error}"
121 next
122 end
123 accepted << [time, cmd, v]
124 last = time
125 case cmd
126 when 'CALL'
127 groups += 1
128 when 'OUT'
129 out << v[0]
130 when 'IN'
131 out.delete(v[0])
132 when 'FIRE'
133 fire = true
134 when 'RESET'
135 fire = false
136 end
137 end
138 accepted
139end
140Group = Struct.new(:number, :a, :b, :count, :time, :waiting, :car, :boarded, :delivered, :lost, :last_board, :last_alight, :cars, :wait_total, :travel_total) do
141 def label = "G#{number}"
142 def direction = sign(b - a)
143end
144Passenger = Struct.new(:group, :count, :board_time)
145Car = Struct.new(:number, :floor, :zone, :home, :direction, :door, :passengers, :calls, :service, :idle, :distance, :stops, :boarded, :busy, :energy) do
146 def label = "E#{number}"
147 def load = passengers.sum(&:count)
148 def targets = (passengers.map { |p| p.group.b } + calls.map(&:a)).uniq
149 def serves?(g) = (zone[0]..zone[1]).include?(g.a) && (zone[0]..zone[1]).include?(g.b)
150end
151class Simulation
152 attr_reader :groups, :cars, :time, :status, :delivered, :wait_total, :travel_total, :policy
153 def initialize(c, events, policy, logging)
154 @c, @events, @policy, @logging = c, events, policy, logging
155 @cars = c['CARS'].times.map { |i| Car.new(i + 1, c['START'][i], c['ZONES'][i], c['PARK']&.[](i + 1), 0, 0, [], [], true, 0, 0, 0, 0, 0, 0) }
156 @groups, @pending = [], []
157 @fire = nil
158 @round = @time = @delivered = @wait_total = @travel_total = @max_wait = @abandoned = @evacuated = 0
159 run
160 end
161 def emit(*parts)
162 puts([@time, *parts].join(' ')) if @logging
163 end
164 def bypass?(car) = car.load * 100 >= @c['CAPACITY'] * @c['BYPASS']
165 def cost(car, g)
166 f, d, x = car.floor, car.direction, g.a
167 return (f - x).abs if @policy == 'NEAREST' || d.zero?
168 ts = car.targets
169 if g.direction == d && (x - f) * d >= 0
170 distance = (x - f).abs
171 via = ts.count { |y| (y - f) * d > 0 && (x - y) * d > 0 }
172 else
173 m = d > 0 ? (ts + [f]).max : (ts + [f]).min
174 distance = (f - m).abs + (m - x).abs
175 via = ts.count { |y| (y - f) * d > 0 }
176 end
177 distance + (@policy == 'ETA' ? @c['DOOR'] * via + car.door : 0)
178 end
179 def assign(g, excluded = nil)
180 candidates = @cars.select { |car| car.service && car != excluded && car.serves?(g) }
181 return false if candidates.empty?
182 if @policy == 'ROUND'
183 chosen = candidates.find { |car| car.number > @round } || candidates.first
184 @round = chosen.number
185 emit(chosen.label, 'ASSIGN', g.label)
186 else
187 chosen = candidates.min_by { |car| [cost(car, g), car.number] }
188 emit(chosen.label, 'ASSIGN', g.label, "cost=#{cost(chosen, g)}")
189 end
190 g.car.calls.delete(g) if g.car
191 chosen.calls << g
192 g.car = chosen
193 true
194 end
195 def dispatch(g)
196 return if assign(g)
197 @pending << g
198 emit('PENDING', g.label)
199 end
200 def walk(g, word)
201 g.car ? g.car.calls.delete(g) : @pending.delete(g)
202 g.car = nil
203 emit(word, g.label, "x#{g.waiting}")
204 g.lost += g.waiting
205 @abandoned += g.waiting
206 g.waiting = 0
207 end
208 def event(cmd, v)
209 case cmd
210 when 'CALL'
211 a, b, n = v
212 g = Group.new(@groups.length + 1, a, b, n, @time, n, nil, 0, 0, 0, 0, 0, [], 0, 0)
213 @groups << g
214 emit(cmd, g.label, "#{a}->#{b}", "x#{n}")
215 dispatch(g)
216 when 'CANCEL'
217 g = @groups[v[0] - 1]
218 g.waiting > 0 ? walk(g, cmd) : emit(cmd, g.label, 'ignored')
219 when 'OUT'
220 car = @cars[v[0] - 1]
221 car.service = false
222 emit(car.label, cmd)
223 calls = car.calls
224 car.calls = []
225 calls.each { |g| g.car = nil
226 dispatch(g) }
227 when 'IN'
228 car = @cars[v[0] - 1]
229 car.service = true
230 car.idle = @time
231 emit(car.label, cmd)
232 pending = @pending
233 @pending = []
234 pending.each { |g| @pending << g unless assign(g) }
235 when 'HOLD'
236 car = @cars[v[0] - 1]
237 car.door += v[1] if car.door > 0
238 emit(car.label, cmd, car.door > 0 ? car.door : 'ignored')
239 when 'FIRE'
240 @fire = v[0]
241 emit(cmd, @fire)
242 @groups.each { |g| walk(g, 'ABANDON') if g.waiting > 0 }
243 when 'RESET'
244 @fire = nil
245 emit(cmd)
246 @cars.each { |car| car.idle = @time }
247 when 'WHERE'
248 g = @groups[v[0] - 1]
249 place = g.waiting.zero? ? '-' : g.car ? g.car.label : 'pending'
250 riding = @cars.filter_map { |car| n = car.passengers.sum { |p| p.group == g ? p.count : 0 }
251 "#{car.label}x#{n}" if n > 0 }
252 emit(cmd, g.label, "waiting=#{g.waiting}", "car=#{place}", "riding=#{joined(riding)}", "delivered=#{g.delivered}", "lost=#{g.lost}")
253 when 'STATUS'
254 @cars.each do |car|
255 emit(cmd, car.label, "floor=#{car.floor}", "dir=#{ {1=>'up', -1=>'down', 0=>'idle'}[car.direction] }", "door=#{car.door}", "load=#{car.load}", "stops=#{joined(car.passengers.map { |p| p.group.b }.uniq.sort)}", "calls=#{joined(car.calls.map(&:label))}", "service=#{car.service ? 'in' : 'out'}")
256 end
257 emit(cmd, "pending=#{joined(@pending.map(&:label))}")
258 end
259 end
260 def heading(car)
261 f, d = car.floor, car.direction
262 here = bypass?(car) ? [] : car.calls.select { |g| g.a == f }
263 ts = car.targets
264 return d if d != 0 && (ts.any? { |y| (y - f) * d > 0 } || here.any? { |g| g.direction == d })
265 return here.first.direction unless here.empty?
266 return -d if d != 0 && ts.any? { |y| (y - f) * d < 0 }
267 return sign(car.calls.first.a - f) if d.zero? && !car.calls.empty?
268 0
269 end
270 def move(car, n)
271 car.energy += n > 0 ? 3 + car.load : 1
272 car.floor += n
273 car.distance += 1
274 end
275 def deliver(p)
276 g = p.group
277 w, r = p.board_time - g.time, @time - g.time
278 g.delivered += p.count
279 g.last_alight = @time
280 g.wait_total += w * p.count
281 g.travel_total += r * p.count
282 @delivered += p.count
283 @wait_total += w * p.count
284 @travel_total += r * p.count
285 @max_wait = [@max_wait, w].max
286 end
287 def open(car)
288 emit(car.label, 'OPEN', car.floor)
289 car.door = @c['DOOR']
290 car.stops += 1
291 car.energy += 2
292 riding = []
293 car.passengers.each do |p|
294 if p.group.b == car.floor
295 emit(car.label, 'ALIGHT', p.group.label, "x#{p.count}")
296 deliver(p)
297 elsif @fire
298 emit(car.label, 'EVACUATE', p.group.label, "x#{p.count}")
299 p.group.lost += p.count
300 @evacuated += p.count
301 else
302 riding << p
303 end
304 end
305 car.passengers = riding
306 return if @fire
307 n = heading(car)
308 car.direction = n unless n.zero?
309 car.calls.select { |g| g.a == car.floor && g.direction == n }.each do |g|
310 k = [@c['CAPACITY'] - car.load, g.waiting].min
311 if k > 0
312 car.passengers << Passenger.new(g, k, @time)
313 car.boarded += k
314 g.waiting -= k
315 g.boarded += k
316 g.last_board = @time
317 g.cars |= [car.number]
318 emit(car.label, 'BOARD', g.label, "x#{k}")
319 end
320 if g.waiting.zero?
321 car.calls.delete(g)
322 g.car = nil
323 else
324 emit(car.label, 'LEFT', g.label, "x#{g.waiting}")
325 assign(g, car)
326 end
327 end
328 end
329 def advance(car)
330 old = car.floor
331 car.door -= 1 if car.door > 0
332 if car.door.zero?
333 n = @fire ? sign(@fire - car.floor) : heading(car)
334 stop = @fire ? n.zero? && !car.passengers.empty? : car.passengers.any? { |p| p.group.b == car.floor } || (!bypass?(car) && car.calls.any? { |g| g.a == car.floor && g.direction == n })
335 if stop
336 open(car)
337 elsif n != 0
338 car.direction = n
339 move(car, n)
340 else
341 if car.direction != 0
342 emit(car.label, 'IDLE', car.floor)
343 car.direction = 0
344 car.idle = @time
345 end
346 if !@fire && car.service && car.home && car.floor != car.home && @time - car.idle >= @c['PARK'][0]
347 move(car, sign(car.home - car.floor))
348 emit(car.label, 'PARK', car.home) if car.floor == car.home
349 end
350 end
351 end
352 car.busy += 1 if car.door > 0 || old != car.floor
353 end
354 def settled?(car)
355 car.door.zero? && car.passengers.empty? && car.calls.empty? && (@fire ? car.floor == @fire : !car.service || !car.home || car.floor == car.home)
356 end
357 def run
358 index = 0
359 loop do
360 @groups.each { |g| walk(g, 'ABANDON') if g.waiting > 0 && @c['PATIENCE'] && @time == g.time + @c['PATIENCE'] }
361 while index < @events.length && @events[index][0] == @time
362 _, cmd, v = @events[index]
363 event(cmd, v)
364 index += 1
365 end
366 @cars.each { |car| advance(car) }
367 if index == @events.length && @cars.all? { |car| settled?(car) }
368 @status = 'end'
369 emit('END')
370 break
371 end
372 if @time == @c['LIMIT']
373 @status = 'abort'
374 emit('ABORT')
375 break
376 end
377 @time += 1
378 end
379 end
380 def averages = "avg_wait=#{average(@wait_total, @delivered)} avg_travel=#{average(@travel_total, @delivered)} max_wait=#{@delivered.zero? ? '-' : @max_wait}"
381 def breakdown(label, gs)
382 "#{label}: groups=#{gs.length} passengers=#{gs.sum(&:count)} delivered=#{gs.sum(&:delivered)} avg_wait=#{average(gs.sum(&:wait_total), gs.sum(&:delivered))}"
383 end
384 def report
385 @groups.each do |g|
386 puts "#{g.label} t=#{g.time} #{g.a}->#{g.b} n=#{g.count} delivered=#{g.delivered} lost=#{g.lost} wait=#{g.boarded == g.count ? g.last_board - g.time : '-'} travel=#{g.delivered == g.count ? g.last_alight - g.time : '-'} cars=#{joined(g.cars.sort)}"
387 end
388 @cars.each do |car|
389 n = car.busy * 1000 / (@time + 1)
390 puts "#{car.label} distance=#{car.distance} stops=#{car.stops} boarded=#{car.boarded} busy=#{car.busy} util=#{n / 10}.#{n % 10}% energy=#{car.energy}"
391 end
392 @groups.group_by(&:a).sort.each { |f, gs| puts breakdown("F#{f}", gs) }
393 if (w = @c['PERIOD'])
394 @groups.group_by { |g| g.time / w }.sort.each { |k, gs| puts breakdown("P#{k} #{k * w}-#{k * w + w - 1}", gs) }
395 end
396 puts "summary: groups=#{@groups.length} passengers=#{@groups.sum(&:count)} delivered=#{@delivered} abandoned=#{@abandoned} evacuated=#{@evacuated} #{averages} end=#{@time}"
397 end
