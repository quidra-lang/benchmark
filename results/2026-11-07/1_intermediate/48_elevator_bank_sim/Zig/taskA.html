139fn before(a: Result, b: Result) bool {
140 const left = a.wait_total * b.delivered;
141 const right = b.wait_total * a.delivered;
142 return if (left != right) left < right else a.travel_total * b.delivered < b.travel_total * a.delivered;

177 fn configuration(h: *Host, lines: []const Line, c: *Config) !?Text {
178 var found: [13]?Line = @splat(null);
179 for (lines) |source| {
180 const key = lookup(source.words[0], &keys) orelse return try h.text("unknown key {s}", .{source.words[0]});
181 if (found[key] != null) return try h.text("duplicate key {s}", .{keys[key]});
182 found[key] = source;
183 }
184 for (0..5) |key| if (found[key] == null) return try h.text("missing key {s}", .{keys[key]});
185 for (found, 0..) |entry, key| {
186 const source = entry orelse continue;
187 const words = source.words[1..];
188 const n = words.len;
189 const valid_count = if (key == 4 or key == 6) n == c.count else if (key == 7) n == c.count + 1 else if (key == 12) n >= 1 and n <= 4 else n == 1;
190 if (!valid_count) return try h.text("wrong number of values for {s}", .{keys[key]});
191 var valid = true;
192 switch (key) {
193 0 => {
194 c.floors = number(words[0]);
195 valid = bounded(c.floors, 2, 50);
196 },
197 1 => {
198 const value = number(words[0]);
199 valid = bounded(value, 1, 8);
200 if (valid) c.count = index(value);
201 },
202 2 => {
203 c.capacity = number(words[0]);
204 valid = bounded(c.capacity, 1, 20);
205 },
206 3 => {
207 c.door = number(words[0]);
208 valid = bounded(c.door, 1, 9);
209 },
210 4 => for (words, 0..) |word, i| {
211 c.starts[i] = number(word);
212 valid = valid and bounded(c.starts[i], 1, c.floors);
213 },
214 5 => {
215 if (lookup(words[0], &policies)) |policy| c.policy = policy else valid = false;
216 },
217 6 => for (words, 0..) |word, i| {
218 const split = std.mem.findScalar(u8, word, ':').?;
219 c.lows[i] = number(word[0..split]);
220 c.highs[i] = number(word[split + 1 ..]);
221 valid = valid and c.lows[i] >= 1 and c.lows[i] < c.highs[i] and c.highs[i] <= c.floors;
222 },
223 7 => {
224 c.park = number(words[0]);
225 valid = bounded(c.park, 0, 100);
226 for (words[1..], 0..) |word, i| {
227 c.homes[i] = number(word);
228 valid = valid and bounded(c.homes[i], 1, c.floors);
229 }
230 },
231 8 => {
232 c.patience = number(words[0]);
233 valid = bounded(c.patience, 1, 1000);
234 },
235 9 => {
236 c.bypass = number(words[0]);
237 valid = bounded(c.bypass, 50, 100);
238 },
239 10 => {
240 c.period = number(words[0]);
241 valid = bounded(c.period, 1, 1000);
242 },
243 11 => {
244 c.limit = number(words[0]);
245 valid = bounded(c.limit, 1, 5000);
246 },
247 12 => {
248 c.comparisons = n;
249 var seen: [4]bool = @splat(false);
250 for (words, 0..) |word, i| {
251 if (lookup(word, &policies)) |policy| {
252 if (seen[policy]) valid = false;
253 seen[policy] = true;
254 c.compare[i] = policy;
255 } else valid = false;
256 }
257 },
258 else => unreachable,
259 }
260 if (!valid) return try h.text("bad value for {s}", .{keys[key]});
261 }
262 if (found[6] == null) for (0..c.count) |i| {
263 c.lows[i] = 1;
264 c.highs[i] = c.floors;
265 };
266 for (0..c.count) |i| {
267 if (!bounded(c.starts[i], c.lows[i], c.highs[i])) return try h.text("car {d} starts outside its zone", .{i + 1});
268 if (c.park >= 0 and !bounded(c.homes[i], c.lows[i], c.highs[i])) return try h.text("car {d} home outside its zone", .{i + 1});
269 }
270 return null;
271 }

273 fn eventProblem(h: *Host, c: Config, source: Line, last: I, groups: I, fire: bool, out: [8]bool) !?Text {
274 const words = source.words;
275 const command = lookup(words[1], &commands) orelse return try h.text("unknown command {s}", .{words[1]});
276 const counts = [_]usize{ 3, 1, 1, 1, 2, 1, 0, 1, 0 };
277 if (words.len - 2 != counts[command]) return try h.text("wrong number of arguments for {s}", .{words[1]});
278 if (number(words[0]) < last) return "time goes backwards";
279 const a = if (words.len > 2) number(words[2]) else 0;
280 const b = if (words.len > 3) number(words[3]) else 0;
281 switch (command) {
282 0 => {
283 if (!bounded(a, 1, c.floors) or !bounded(b, 1, c.floors)) return "bad floor";
284 if (a == b) return "same floor";
285 if (!bounded(number(words[4]), 1, 99)) return "bad count";
286 var served = false;
287 for (0..c.count) |car| served = served or c.serves(car, a, b);
288 if (!served) return try h.text("no car serves {d}->{d}", .{ a, b });
289 if (fire) return "fire service active";
290 },
291 1, 7 => if (!bounded(a, 1, groups)) {
292 return "unknown group";
293 },
294 2, 3 => {
295 if (!bounded(a, 1, @intCast(c.count))) return "bad car";
296 if (out[index(a - 1)] == (command == 2)) return try h.text("car {d} already {s}", .{ a, if (command == 2) "out" else "in" });
297 },
298 4 => {
299 if (!bounded(a, 1, @intCast(c.count))) return "bad car";
300 if (!bounded(b, 1, 9)) return "bad ticks";
301 },
302 5 => {
303 if (!bounded(a, 1, c.floors)) return "bad floor";
304 if (fire) return "fire already active";
305 },
306 6 => if (!fire) {
307 return "fire not active";
308 },
309 else => {},
310 }
311 return null;
312 }

314 fn validateEvents(h: *Host, c: Config, lines: []const Line) ![]Event {
315 var result: std.ArrayList(Event) = .empty;
316 var last: I = 0;
317 var groups: I = 0;
318 var fire = false;
319 var out: [8]bool = @splat(false);
320 for (lines) |source| {
321 if (try h.eventProblem(c, source, last, groups, fire, out)) |problem| {
322 try h.line("line {d}: {s}", .{ source.number, problem });
323 continue;
324 }
325 const w = source.words;
326 const command = lookup(w[1], &commands).?;
327 const a = if (w.len > 2) number(w[2]) else 0;
328 const b = if (w.len > 3) number(w[3]) else 0;
329 const value = if (w.len > 4) number(w[4]) else 0;
330 last = number(w[0]);
331 if (command == 0) groups += 1;
332 if (command == 2 or command == 3) out[index(a - 1)] = command == 2;
333 if (command == 5 or command == 6) fire = command == 5;
334 try result.append(h.gpa, .{ .time = last, .command = command, .a = a, .b = b, .c = value });
335 }
336 return result.items;
337 }

381 fn targets(s: *Simulation, i: usize) [51]bool {
382 var floors: [51]bool = @splat(false);
383 const car = &s.cars[i];
384 for (car.passengers[0..car.riding]) |entry| floors[index(s.groups[entry.group].destination)] = true;
385 for (car.calls.ids[0..car.calls.count]) |group| floors[index(s.groups[group].landing)] = true;
386 return floors;
387 }

390 return s.cars[i].load() * 100 >= s.config.capacity * s.config.bypass;
391 }

393 fn cost(s: *Simulation, i: usize, group_index: usize) I {
394 const car = &s.cars[i];
395 const group = s.groups[group_index];
396 const f = car.floor;
397 const d = car.direction;
398 const x = group.landing;
399 if (s.policy == 1 or d == 0) return absolute(f - x);
400 const floors = s.targets(i);
401 var distance: I = 0;
402 var via: I = 0;
403 if (sign(group.destination - x) == d and (x - f) * d >= 0) {
404 distance = absolute(x - f);
405 for (1..index(s.config.floors) + 1) |target| {
406 const y: I = @intCast(target);
407 if (floors[target] and (y - f) * d > 0 and (x - y) * d > 0) via += 1;
408 }
409 } else {
410 var turn = f;
411 for (1..index(s.config.floors) + 1) |target| {
412 const y: I = @intCast(target);
413 if (floors[target]) {
414 if ((y - turn) * d > 0) turn = y;
415 if ((y - f) * d > 0) via += 1;
416 }
417 }
418 distance = absolute(f - turn) + absolute(turn - x);
419 }
420 return if (s.policy == 3) distance + s.config.door * via + car.door else distance;
421 }

423 fn assign(s: *Simulation, group_index: usize, exclude: ?usize) !bool {
424 const group = &s.groups[group_index];
425 var selected: ?usize = null;
426 var best_cost: I = 0;
427 for (0..s.config.count) |i| {
428 if ((exclude != null and exclude.? == i) or !s.cars[i].service or !s.config.serves(i, group.landing, group.destination)) continue;
429 if (s.policy == 2) {
430 if (selected == null) selected = i;
431 if (i + 1 > s.last_round) {
432 selected = i;
433 break;
434 }
435 } else {
436 const value = s.cost(i, group_index);
437 if (selected == null or value < best_cost) {
438 selected = i;
439 best_cost = value;
440 }
441 }
442 }
443 const chosen = selected orelse return false;
444 if (s.policy == 2) {
445 s.last_round = chosen + 1;
446 try s.emit("E{d} ASSIGN G{d}", .{ chosen + 1, group_index + 1 });
447 } else try s.emit("E{d} ASSIGN G{d} cost={d}", .{ chosen + 1, group_index + 1, best_cost });
448 if (group.car) |old| s.cars[old].calls.remove(group_index);
449 s.cars[chosen].calls.add(group_index);
450 group.car = chosen;
451 return true;
452 }

461 fn walk(s: *Simulation, i: usize, word: Text) !void {
462 const group = &s.groups[i];
463 if (group.car) |car| s.cars[car].calls.remove(i) else s.pending.remove(i);
464 group.car = null;
465 try s.emit("{s} G{d} x{d}", .{ word, i + 1, group.waiting });
466 s.abandoned += group.waiting;
467 group.lost += group.waiting;
468 group.waiting = 0;
469 }

471 fn perform(s: *Simulation, event: Event) !void {
472 const a = event.a;
473 const b = event.b;
474 switch (event.command) {
475 0 => {
476 const i = s.ng;
477 s.ng += 1;
478 s.groups[i] = .{ .time = s.time, .landing = a, .destination = b, .count = event.c, .waiting = event.c };
479 try s.emit("CALL G{d} {d}->{d} x{d}", .{ i + 1, a, b, event.c });
480 try s.dispatch(i);
481 },
482 1 => {
483 if (s.groups[index(a - 1)].waiting > 0) try s.walk(index(a - 1), "CANCEL") else try s.emit("CANCEL G{d} ignored", .{a});
484 },
485 2 => {
486 const car = &s.cars[index(a - 1)];
487 car.service = false;
488 try s.emit("E{d} OUT", .{a});
489 const calls = car.calls;
490 car.calls.count = 0;
491 for (calls.ids[0..calls.count]) |i| {
492 s.groups[i].car = null;
493 try s.dispatch(i);
494 }
495 },
496 3 => {
497 const car = &s.cars[index(a - 1)];
498 car.service = true;
499 car.idle_start = s.time;
500 try s.emit("E{d} IN", .{a});
501 const pending = s.pending;
502 s.pending.count = 0;
503 for (pending.ids[0..pending.count]) |i| if (!try s.assign(i, null)) s.pending.add(i);
504 },
505 4 => {
506 const car = &s.cars[index(a - 1)];
507 if (car.door > 0) {
508 car.door += b;
509 try s.emit("E{d} HOLD {d}", .{ a, car.door });
510 } else try s.emit("E{d} HOLD ignored", .{a});
511 },
512 5 => {
513 s.fire = a;
514 try s.emit("FIRE {d}", .{a});
515 for (0..s.ng) |i| if (s.groups[i].waiting > 0) {
516 try s.walk(i, "ABANDON");
517 };
518 },
519 6 => {
520 s.fire = 0;
521 try s.emit("RESET", .{});
522 for (s.cars[0..s.config.count]) |*car| car.idle_start = s.time;
523 },
524 7 => {
525 const group = s.groups[index(a - 1)];
526 const place = if (group.waiting == 0) "-" else if (group.car) |car| try s.host.text("E{d}", .{car + 1}) else "pending";
527 var riding: std.Io.Writer.Allocating = .init(s.host.gpa);
528 var first = true;
529 for (s.cars[0..s.config.count], 0..) |car, i| {
530 var people: I = 0;
531 for (car.passengers[0..car.riding]) |entry| if (entry.group == index(a - 1)) {
532 people += entry.count;
533 };
534 if (people > 0) {
535 try riding.writer.print("{s}E{d}x{d}", .{ if (first) "" else ",", i + 1, people });
536 first = false;
537 }
538 }
539 try s.emit("WHERE G{d} waiting={d} car={s} riding={s} delivered={d} lost={d}", .{ a, group.waiting, place, if (first) "-" else riding.written(), group.delivered, group.lost });
540 },
541 8 => {
542 for (s.cars[0..s.config.count], 0..) |car, i| {
543 var floors: [51]bool = @splat(false);
544 for (car.passengers[0..car.riding]) |entry| floors[index(s.groups[entry.group].destination)] = true;
545 var stops: std.Io.Writer.Allocating = .init(s.host.gpa);
546 var first = true;
547 for (1..index(s.config.floors) + 1) |floor| if (floors[floor]) {
548 try stops.writer.print("{s}{d}", .{ if (first) "" else ",", floor });
549 first = false;
550 };
551 try s.emit("STATUS E{d} floor={d} dir={s} door={d} load={d} stops={s} calls={s} service={s}", .{ i + 1, car.floor, if (car.direction > 0) "up" else if (car.direction < 0) "down" else "idle", car.door, car.load(), if (first) "-" else stops.written(), try s.host.groupList(car.calls), if (car.service) "in" else "out" });
552 }
553 try s.emit("STATUS pending={s}", .{try s.host.groupList(s.pending)});
554 },
555 else => unreachable,
556 }
557 }

559 fn direction(s: *Simulation, i: usize) I {
560 const car = &s.cars[i];
561 const f = car.floor;
562 const d = car.direction;
563 const floors = s.targets(i);
564 var ahead = false;
565 var behind = false;
566 var first: ?usize = null;
567 var same = false;
568 for (1..index(s.config.floors) + 1) |target| {
569 const y: I = @intCast(target);
570 if (floors[target]) {
571 ahead = ahead or (y - f) * d > 0;
572 behind = behind or (y - f) * d < 0;
573 }
574 }
575 if (!s.bypassing(i)) for (car.calls.ids[0..car.calls.count]) |g| {
576 const group = s.groups[g];
577 if (group.landing == f) {
578 if (first == null) first = g;
579 same = same or sign(group.destination - group.landing) == d;
580 }
581 };
582 if (d != 0 and (ahead or same)) return d;
583 if (first) |g| return sign(s.groups[g].destination - f);
584 if (d != 0 and behind) return -d;
585 if (d == 0 and car.calls.count > 0) return sign(s.groups[car.calls.ids[0]].landing - f);
586 return 0;
587 }

589 fn move(s: *Simulation, i: usize, step: I) void {
590 const car = &s.cars[i];
591 car.energy += if (step > 0) 3 + car.load() else 1;
592 car.distance += 1;
593 car.floor += step;
594 }

596 fn openDoors(s: *Simulation, i: usize) !void {
597 const car = &s.cars[i];
598 const f = car.floor;
599 try s.emit("E{d} OPEN {d}", .{ i + 1, f });
600 car.door = s.config.door;
601 car.stops += 1;
602 car.energy += 2;
603 var kept: usize = 0;
604 for (car.passengers[0..car.riding]) |entry| {
605 const group = &s.groups[entry.group];
606 if (group.destination == f) {
607 try s.emit("E{d} ALIGHT G{d} x{d}", .{ i + 1, entry.group + 1, entry.count });
608 const wait = entry.time - group.time;
609 const travel = s.time - group.time;
610 group.delivered += entry.count;
611 group.last_alight = s.time;
612 group.wait_total += wait * entry.count;
613 group.travel_total += travel * entry.count;
614 s.delivered += entry.count;
615 s.wait_total += wait * entry.count;
616 s.travel_total += travel * entry.count;
617 s.max_wait = @max(s.max_wait, wait);
618 } else if (s.fire > 0) {
619 try s.emit("E{d} EVACUATE G{d} x{d}", .{ i + 1, entry.group + 1, entry.count });
620 group.lost += entry.count;
621 s.evacuated += entry.count;
622 } else {
623 car.passengers[kept] = entry;
624 kept += 1;
625 }
626 }
627 car.riding = kept;
628 if (s.fire > 0) return;
629 const heading = s.direction(i);
630 if (heading != 0) car.direction = heading;
631 var boarding: List = .{};
632 for (car.calls.ids[0..car.calls.count]) |g| {
633 const group = s.groups[g];
634 if (group.landing == f and sign(group.destination - group.landing) == heading) boarding.add(g);
635 }
636 for (boarding.ids[0..boarding.count]) |g| {
637 const group = &s.groups[g];
638 const people = @min(s.config.capacity - car.load(), group.waiting);
639 if (people > 0) {
640 car.passengers[car.riding] = .{ .group = g, .count = people, .time = s.time };
641 car.riding += 1;
642 car.boarded += people;
643 group.waiting -= people;
644 group.boarded += people;
645 group.last_board = s.time;
646 group.cars[i] = true;
647 try s.emit("E{d} BOARD G{d} x{d}", .{ i + 1, g + 1, people });
648 }
649 if (group.waiting == 0) {
650 car.calls.remove(g);
651 group.car = null;
652 } else {
653 try s.emit("E{d} LEFT G{d} x{d}", .{ i + 1, g + 1, group.waiting });
654 _ = try s.assign(g, i);
655 }
656 }
657 }
658
659 fn advance(s: *Simulation, i: usize) !void {
660 const car = &s.cars[i];
661 const old = car.floor;
662 if (car.door > 0) car.door -= 1;
663 if (car.door == 0) {
664 const heading = if (s.fire > 0) sign(s.fire - car.floor) else s.direction(i);
665 var stop = s.fire > 0 and heading == 0 and car.riding > 0;
666 if (s.fire == 0) {
667 for (car.passengers[0..car.riding]) |entry| stop = stop or s.groups[entry.group].destination == car.floor;
668 if (!s.bypassing(i)) for (car.calls.ids[0..car.calls.count]) |g| {
669 const group = s.groups[g];
670 stop = stop or (group.landing == car.floor and sign(group.destination - group.landing) == heading);
671 };
672 }
673 if (stop) try s.openDoors(i) else if (heading != 0) {
674 car.direction = heading;
675 s.move(i, heading);
676 } else {
677 if (car.direction != 0) {
678 try s.emit("E{d} IDLE {d}", .{ i + 1, car.floor });
679 car.direction = 0;
680 car.idle_start = s.time;
681 }
682 if (s.fire == 0 and car.service and s.config.park >= 0 and car.floor != s.config.homes[i] and s.time - car.idle_start >= s.config.park) {
683 s.move(i, sign(s.config.homes[i] - car.floor));
684 if (car.floor == s.config.homes[i]) try s.emit("E{d} PARK {d}", .{ i + 1, car.floor });
685 }
686 }
687 }
688 if (car.door > 0 or car.floor != old) car.busy += 1;
689 }
690
691 fn settled(s: *Simulation, i: usize) bool {
692 const car = &s.cars[i];
693 if (car.door > 0 or car.riding > 0 or car.calls.count > 0) return false;
694 if (s.fire > 0) return car.floor == s.fire;
695 return !car.service or s.config.park < 0 or car.floor == s.config.homes[i];
696 }

698 fn run(s: *Simulation, events: []const Event) !void {
699 var next: usize = 0;
700 while (true) : (s.time += 1) {
701 if (s.config.patience >= 0) for (0..s.ng) |i| if (s.groups[i].waiting > 0 and s.time == s.groups[i].time + s.config.patience) {
702 try s.walk(i, "ABANDON");
703 };
704 while (next < events.len and events[next].time == s.time) : (next += 1) try s.perform(events[next]);
705 for (0..s.config.count) |i| try s.advance(i);
706 var all = true;
707 for (0..s.config.count) |i| all = all and s.settled(i);
708 if (next == events.len and all) {
709 try s.emit("END", .{});
710 break;
711 }
712 if (s.time == s.config.limit) {
713 s.aborted = true;
714 try s.emit("ABORT", .{});
715 break;
716 }
717 }
718 }

720 fn breakdown(s: *Simulation, key: I, period: bool) !void {
721 var groups: usize = 0;
722 var passengers: I = 0;
723 var delivered: I = 0;
724 var wait: I = 0;
725 for (s.groups[0..s.ng]) |group| if ((if (period) @divTrunc(group.time, s.config.period) else group.landing) == key) {
726 groups += 1;
727 passengers += group.count;
728 delivered += group.delivered;
729 wait += group.wait_total;
730 };
731 if (groups == 0) return;
732 if (period) try s.host.out.print("P{d} {d}-{d}:", .{ key, key * s.config.period, key * s.config.period + s.config.period - 1 }) else try s.host.out.print("F{d}:", .{key});
733 try s.host.line(" groups={d} passengers={d} delivered={d} avg_wait={s}", .{ groups, passengers, delivered, try s.host.average(wait, delivered) });
734 }
735
736 fn report(s: *Simulation) !void {
737 var passengers: I = 0;
738 for (s.groups[0..s.ng], 0..) |group, g| {
739 var car_list: std.Io.Writer.Allocating = .init(s.host.gpa);
740 var first = true;
741 for (group.cars[0..s.config.count], 0..) |used, i| if (used) {
742 try car_list.writer.print("{s}{d}", .{ if (first) "" else ",", i + 1 });
743 first = false;
744 };
745 const wait = if (group.boarded == group.count) try s.host.text("{d}", .{group.last_board - group.time}) else "-";
746 const travel = if (group.delivered == group.count) try s.host.text("{d}", .{group.last_alight - group.time}) else "-";
747 try s.host.line("G{d} t={d} {d}->{d} n={d} delivered={d} lost={d} wait={s} travel={s} cars={s}", .{ g + 1, group.time, group.landing, group.destination, group.count, group.delivered, group.lost, wait, travel, if (first) "-" else car_list.written() });
748 passengers += group.count;
749 }
750 for (s.cars[0..s.config.count], 0..) |car, i| {
751 const util = @divTrunc(car.busy * 1000, s.time + 1);
752 try s.host.line("E{d} distance={d} stops={d} boarded={d} busy={d} util={d}.{d}% energy={d}", .{ i + 1, car.distance, car.stops, car.boarded, car.busy, @divTrunc(util, 10), @mod(util, 10), car.energy });
753 }
754 var f: I = 1;
755 while (f <= s.config.floors) : (f += 1) try s.breakdown(f, false);
756 if (s.config.period >= 0) {
757 var p: I = 0;
758 while (p <= @divTrunc(10000, s.config.period)) : (p += 1) try s.breakdown(p, true);
759 }
760 try s.host.line("summary: groups={d} passengers={d} delivered={d} abandoned={d} evacuated={d} {s} end={d}", .{ s.ng, passengers, s.delivered, s.abandoned, s.evacuated, try s.host.averages(s.delivered, s.wait_total, s.travel_total, s.max_wait), s.time });
761 }
762};

764fn readLine(reader: *std.Io.Reader, gpa: std.mem.Allocator) !?Text {
765 var buffer: std.Io.Writer.Allocating = .init(gpa);
766 defer buffer.deinit();
767 _ = try reader.streamDelimiterEnding(&buffer.writer, '\n');
768 if (reader.bufferedLen() == 0 and buffer.written().len == 0) return null;
769 if (reader.bufferedLen() > 0) reader.toss(1);
770 return buffer.toOwnedSlice();
771}

773fn words(gpa: std.mem.Allocator, line: Text) ![]const Text {
774 var result: std.ArrayList(Text) = .empty;
775 var it = std.mem.splitScalar(u8, line, ' ');
776 while (it.next()) |word| try result.append(gpa, word);
777 return result.items;
778}

780pub fn main(init: std.process.Init) !void {
781 var arena: std.heap.ArenaAllocator = .init(std.heap.page_allocator);
782 defer arena.deinit();
783 const gpa = arena.allocator();
784 var ib: [4096]u8 = undefined;
785 var ob: [4096]u8 = undefined;
786 var input = std.Io.File.stdin().readerStreaming(init.io, &ib);
787 var output = std.Io.File.stdout().writerStreaming(init.io, &ob);
788 var host: Host = .{ .gpa = gpa, .out = &output.interface };
789 const q = index(number(std.mem.trim(u8, (try readLine(&input.interface, gpa)).?, "\r")));
790 var line_number: usize = 1;
791 var results: [50]Result = @splat(.{});
792 for (0..q) |scenario| {
793 const heading = try words(gpa, std.mem.trim(u8, (try readLine(&input.interface, gpa)).?, "\r"));
794 line_number += 1;
795 results[scenario].name = heading[1];
796 var settings: std.ArrayList(Line) = .empty;
797 var event_lines: std.ArrayList(Line) = .empty;
798 while (true) {
799 const source = std.mem.trim(u8, (try readLine(&input.interface, gpa)).?, "\r");
800 line_number += 1;
801 if (eq(source, "END")) break;
802 const line: Line = .{ .number = line_number, .words = try words(gpa, source) };
803 if (source[0] >= '0' and source[0] <= '9') try event_lines.append(gpa, line) else try settings.append(gpa, line);
804 }
805 try host.line("== {s} ==", .{heading[1]});
806 var config: Config = .{};
807 if (try host.configuration(settings.items, &config)) |problem| {
808 try host.line("error: {s}", .{problem});
809 continue;
810 }
811 try host.configLines(config);
812 const events = try host.validateEvents(config, event_lines.items);
813 const simulation = try host.simulate(config, events, config.policy, true);
814 try simulation.report();
815 results[scenario] = .{ .name = heading[1], .valid = true, .aborted = simulation.aborted, .policy = config.policy, .delivered = simulation.delivered, .wait_total = simulation.wait_total, .travel_total = simulation.travel_total, .max_wait = simulation.max_wait };
816 std.heap.page_allocator.destroy(simulation);
817 for (config.compare[0..config.comparisons]) |policy| {
818 const other = try host.simulate(config, events, policy, false);
819 try host.line("compare {s}: delivered={d} {s} end={d} status={s}", .{ policies[policy], other.delivered, try host.averages(other.delivered, other.wait_total, other.travel_total, other.max_wait), other.time, if (other.aborted) "abort" else "end" });
820 std.heap.page_allocator.destroy(other);
821 }
822 }
823 try host.line("== ranking ==", .{});
824 var ranked: [50]usize = undefined;
825 var nr: usize = 0;
826 for (results[0..q], 0..) |result, i| if (result.valid and !result.aborted and result.delivered > 0) {
827 var position = nr;
828 while (position > 0 and before(result, results[ranked[position - 1]])) : (position -= 1) ranked[position] = ranked[position - 1];
829 ranked[position] = i;
830 nr += 1;
831 };
832 for (ranked[0..nr], 0..) |i, rank| {
833 const result = results[i];
834 try host.line("{d}. {s} policy={s} {s}", .{ rank + 1, result.name, policies[result.policy], try host.averages(result.delivered, result.wait_total, result.travel_total, result.max_wait) });
835 }
836 for (results[0..q]) |result| if (!result.valid or result.aborted or result.delivered == 0) {
837 try host.line("- {s} {s}", .{ result.name, if (!result.valid) "invalid" else if (result.aborted) "aborted" else "empty" });
838 };
839 try host.line("== policies ==", .{});
840 for (policies, 0..) |policy, p| {
841 var runs: usize = 0;
842 var delivered: I = 0;
843 var wait: I = 0;
844 var travel: I = 0;
845 for (ranked[0..nr]) |i| {
846 const result = results[i];
847 if (result.policy == p) {
848 runs += 1;
849 delivered += result.delivered;
850 wait += result.wait_total;
851 travel += result.travel_total;
852 }
853 }
854 if (runs > 0) try host.line("{s}: runs={d} avg_wait={s} avg_travel={s}", .{ policy, runs, try host.average(wait, delivered), try host.average(travel, delivered) });
855 }
856 try output.interface.flush();
857}
