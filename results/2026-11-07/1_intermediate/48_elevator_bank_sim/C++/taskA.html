68Config configuration(const vector<vector<string>> &lines) {
69 map<string, vector<string>> entries;
70 for (const vector<string> &line : lines) {
71 const string &k = line[0];
72 require(!contains(keys, k), "unknown key " + k);
73 require(entries.count(k), "duplicate key " + k);
74 entries[k] = vector<string>(line.begin() + 1, line.end());
75 }
76 for (int i = 0; i < 5; ++i)
77 require(!entries.count(keys[i]), "missing key " + keys[i]);
78 Config c;
79 for (const string &k : keys) {
80 if (!entries.count(k))
81 continue;
82 const vector<string> &a = entries[k];
83 int n = k == "START" || k == "ZONES" ? c.cars : k == "PARK" ? c.cars + 1 : 1;
84 require(k == "COMPARE" ? a.empty() || a.size() > 4 : (int)a.size() != n,
85 "wrong number of values for " + k);
86 auto bounded = [&](const string &text, int lo, int hi) {
87 int x = stoi(text);
88 require(x < lo || x > hi, "bad value for " + k);
89 return x;
90 };
91 if (k == "FLOORS")
92 c.floors = bounded(a[0], 2, 50);
93 else if (k == "CARS")
94 c.cars = bounded(a[0], 1, 8);
95 else if (k == "CAPACITY")
96 c.capacity = bounded(a[0], 1, 20);
97 else if (k == "DOOR")
98 c.door = bounded(a[0], 1, 9);
99 else if (k == "START")
100 for (const string &x : a)
101 c.starts.push_back(bounded(x, 1, c.floors));
102 else if (k == "POLICY") {
103 require(!contains(policies, a[0]), "bad value for POLICY");
104 c.policy = a[0];
105 } else if (k == "ZONES")
106 for (const string &x : a) {
107 size_t p = x.find(':');
108 int lo = stoi(x.substr(0, p)), hi = stoi(x.substr(p + 1));
109 require(lo < 1 || lo >= hi || hi > c.floors, "bad value for ZONES");
110 c.zones.emplace_back(lo, hi);
111 }
112 else if (k == "PARK") {
113 c.park = bounded(a[0], 0, 100);
114 for (int i = 1; i < (int)a.size(); ++i)
115 c.homes.push_back(bounded(a[i], 1, c.floors));
116 } else if (k == "PATIENCE")
117 c.patience = bounded(a[0], 1, 1000);
118 else if (k == "BYPASS")
119 c.bypass = bounded(a[0], 50, 100);
120 else if (k == "PERIOD")
121 c.period = bounded(a[0], 1, 1000);
122 else if (k == "LIMIT")
123 c.limit = bounded(a[0], 1, 5000);
124 else {
125 set<string> seen;
126 for (const string &p : a) {
127 require(!contains(policies, p) || seen.count(p), "bad value for COMPARE");
128 seen.insert(p);
129 }
130 c.compare = a;
131 }
132 }
133 if (c.zones.empty())
134 c.zones.assign(c.cars, {1, c.floors});
135 for (int i = 0; i < c.cars; ++i) {
136 auto [lo, hi] = c.zones[i];
137 require(c.starts[i] < lo || c.starts[i] > hi,
138 "car " + num(i + 1) + " starts outside its zone");
139 require(c.park >= 0 && (c.homes[i] < lo || c.homes[i] > hi),
140 "car " + num(i + 1) + " home outside its zone");
141 }
142 return c;
143}

149vector<Event> validate(const Config &c, const vector<pair<int, vector<string>>> &raw) {
150 vector<Event> accepted;
151 int last = 0, groups = 0;
152 bool fire = false;
153 set<int> out;
154 for (const auto &line : raw) {
155 int t = stoi(line.second[0]);
156 string cmd = line.second[1], error;
157 vector<int> v;
158 if (!argCounts.count(cmd))
159 error = "unknown command " + cmd;
160 else if ((int)line.second.size() - 2 != argCounts.at(cmd))
161 error = "wrong number of arguments for " + cmd;
162 else if (t < last)
163 error = "time goes backwards";
164 else {
165 for (int i = 2; i < (int)line.second.size(); ++i)
166 v.push_back(stoi(line.second[i]));
167 int a = v.empty() ? 0 : v[0], b = v.size() > 1 ? v[1] : 0;
168 if (cmd == "CALL") {
169 bool served = false;
170 for (int i = 0; i < c.cars; ++i)
171 served |= c.serves(i, a, b);
172 if (a < 1 || a > c.floors || b < 1 || b > c.floors)
173 error = "bad floor";
174 else if (a == b)
175 error = "same floor";
176 else if (v[2] < 1 || v[2] > 99)
177 error = "bad count";
178 else if (!served)
179 error = "no car serves " + num(a) + "->" + num(b);
180 else if (fire)
181 error = "fire service active";
182 } else if (cmd == "CANCEL" || cmd == "WHERE") {
183 if (a < 1 || a > groups)
184 error = "unknown group";
185 } else if (cmd == "OUT" || cmd == "IN" || cmd == "HOLD") {
186 if (a < 1 || a > c.cars)
187 error = "bad car";
188 else if (cmd == "OUT" && out.count(a))
189 error = "car " + num(a) + " already out";
190 else if (cmd == "IN" && !out.count(a))
191 error = "car " + num(a) + " already in";
192 else if (cmd == "HOLD" && (b < 1 || b > 9))
193 error = "bad ticks";
194 } else if (cmd == "FIRE") {
195 if (a < 1 || a > c.floors)
196 error = "bad floor";
197 else if (fire)
198 error = "fire already active";
199 } else if (cmd == "RESET" && !fire)
200 error = "fire not active";
201 }
202 if (!error.empty()) {
203 cout << "line " << line.first << ": " << error << '\n';
204 continue;
205 }
206 accepted.push_back({t, cmd, v});
207 last = t;
208 if (cmd == "CALL")
209 ++groups;
210 if (cmd == "OUT")
211 out.insert(v[0]);
212 if (cmd == "IN")
213 out.erase(v[0]);
214 if (cmd == "FIRE")
215 fire = true;
216 if (cmd == "RESET")
217 fire = false;
218 }
219 return accepted;
220}

253class Simulation {
254 const Config &c;
255 const vector<Event> &input;
256 string policy;
257 bool logging;
258 Result r;
259 vector<int> pending;
260 int time = 0, evacuation = 0, round = -1;
261 bool fire = false;
262 void emit(const string &s) const {
263 if (logging)
264 cout << time << ' ' << s << '\n';
265 }
266 set<int> targets(const Car &car) const {
267 set<int> s;
268 for (const Passenger &p : car.passengers)
269 s.insert(r.groups[p.group].b);
270 for (int g : car.calls)
271 s.insert(r.groups[g].a);
272 return s;
273 }
274 void remove(int gi) {
275 Group &g = r.groups[gi];
276 vector<int> &list = g.car < 0 ? pending : r.cars[g.car].calls;
277 list.erase(std::remove(list.begin(), list.end(), gi), list.end());
278 g.car = -1;
279 }
280 int cost(const Car &car, const Group &g) const {
281 int f = car.floor, d = car.direction, x = g.a;
282 if (policy == "NEAREST" || !d)
283 return abs(f - x);
284 set<int> s = targets(car);
285 int distance, via = 0;
286 if (g.direction() == d && (x - f) * d >= 0) {
287 distance = abs(x - f);
288 for (int y : s)
289 if ((y - f) * d > 0 && (x - y) * d > 0)
290 ++via;
291 } else {
292 int m = f;
293 for (int y : s)
294 m = d > 0 ? max(m, y) : min(m, y);
295 distance = abs(f - m) + abs(m - x);
296 for (int y : s)
297 if ((y - f) * d > 0)
298 ++via;
299 }
300 return distance + (policy == "ETA" ? c.door * via + car.door : 0);
301 }
302 bool assign(int gi, int exclude = -1) {
303 const Group &g = r.groups[gi];
304 vector<int> candidates;
305 for (int i = 0; i < c.cars; ++i)
306 if (i != exclude && r.cars[i].inService && c.serves(i, g.a, g.b))
307 candidates.push_back(i);
308 if (candidates.empty())
309 return false;
310 int chosen = candidates.front();
311 if (policy == "ROUND") {
312 for (int i : candidates)
313 if (i > round) {
314 chosen = i;
315 break;
316 }
317 } else
318 for (int i : candidates)
319 if (cost(r.cars[i], g) < cost(r.cars[chosen], g))
320 chosen = i;
321 int value = cost(r.cars[chosen], g);
322 remove(gi);
323 r.groups[gi].car = chosen;
324 r.cars[chosen].calls.push_back(gi);
325 if (policy == "ROUND")
326 round = chosen;
327 emit("E" + num(chosen + 1) + " ASSIGN G" + num(gi + 1) +
328 (policy == "ROUND" ? "" : " cost=" + num(value)));
329 return true;
330 }
331 void makePending(int g) {
332 r.groups[g].car = -1;
333 pending.push_back(g);
334 emit("PENDING G" + num(g + 1));
335 }
336 void abandon(int gi, const string &word) {
337 Group &g = r.groups[gi];
338 remove(gi);
339 emit(word + " G" + num(gi + 1) + " x" + num(g.waiting));
340 r.abandoned += g.waiting;
341 g.lost += g.waiting;
342 g.waiting = 0;
343 }
344 int choose(const Car &car) const {
345 int f = car.floor, d = car.direction;
346 set<int> s = targets(car);
347 vector<int> calls;
348 if (car.load() * 100 < c.capacity * c.bypass)
349 for (int g : car.calls)
350 if (r.groups[g].a == f)
351 calls.push_back(g);
352 if (d) {
353 for (int y : s)
354 if ((y - f) * d > 0)
355 return d;
356 for (int g : calls)
357 if (r.groups[g].direction() == d)
358 return d;
359 }
360 if (!calls.empty())
361 return r.groups[calls.front()].direction();
362 if (d)
363 for (int y : s)
364 if ((y - f) * d < 0)
365 return -d;
366 if (!d && !car.calls.empty())
367 return sign(r.groups[car.calls.front()].a - f);
368 return 0;
369 }
370 void move(Car &car, int n) {
371 car.energy += n > 0 ? 3 + car.load() : 1;
372 car.floor += n;
373 ++car.distance;
374 }
375 void event(const Event &e) {
376 int a = e.values.empty() ? 0 : e.values[0], b = e.values.size() > 1 ? e.values[1] : 0;
377 const string &cmd = e.command;
378 if (cmd == "CALL") {
379 int n = e.values[2], gi = (int)r.groups.size();
380 Group g;
381 g.a = a;
382 g.b = b;
383 g.n = n;
384 g.time = time;
385 g.waiting = n;
386 r.groups.push_back(g);
387 emit("CALL G" + num(gi + 1) + " " + num(a) + "->" + num(b) + " x" + num(n));
388 if (!assign(gi))
389 makePending(gi);
390 } else if (cmd == "CANCEL") {
391 if (r.groups[a - 1].waiting)
392 abandon(a - 1, "CANCEL");
393 else
394 emit("CANCEL G" + num(a) + " ignored");
395 } else if (cmd == "OUT") {
396 Car &car = r.cars[a - 1];
397 car.inService = false;
398 emit("E" + num(a) + " OUT");
399 vector<int> calls = car.calls;
400 car.calls.clear();
401 for (int g : calls) {
402 r.groups[g].car = -1;
403 if (!assign(g))
404 makePending(g);
405 }
406 } else if (cmd == "IN") {
407 Car &car = r.cars[a - 1];
408 car.inService = true;
409 car.idle = time;
410 emit("E" + num(a) + " IN");
411 vector<int> calls = pending;
412 pending.clear();
413 for (int g : calls)
414 if (!assign(g))
415 pending.push_back(g);
416 } else if (cmd == "HOLD") {
417 Car &car = r.cars[a - 1];
418 if (car.door) {
419 car.door += b;
420 emit("E" + num(a) + " HOLD " + num(car.door));
421 } else
422 emit("E" + num(a) + " HOLD ignored");
423 } else if (cmd == "FIRE") {
424 fire = true;
425 evacuation = a;
426 emit("FIRE " + num(a));
427 for (int i = 0; i < (int)r.groups.size(); ++i)
428 if (r.groups[i].waiting)
429 abandon(i, "ABANDON");
430 } else if (cmd == "RESET") {
431 fire = false;
432 emit("RESET");
433 for (Car &car : r.cars)
434 car.idle = time;
435 } else if (cmd == "WHERE") {
436 const Group &g = r.groups[a - 1];
437 vector<string> riding;
438 for (int i = 0; i < c.cars; ++i) {
439 int n = 0;
440 for (const Passenger &p : r.cars[i].passengers)
441 if (p.group == a - 1)
442 n += p.count;
443 if (n)
444 riding.push_back("E" + num(i + 1) + "x" + num(n));
445 }
446 emit("WHERE G" + num(a) + " waiting=" + num(g.waiting) + " car=" +
447 (!g.waiting ? "-"
448 : g.car < 0 ? "pending"
449 : "E" + num(g.car + 1)) +
450 " riding=" + join(riding) + " delivered=" + num(g.delivered) +
451 " lost=" + num(g.lost));
452 } else {
453 for (int i = 0; i < c.cars; ++i) {
454 const Car &car = r.cars[i];
455 set<int> stops;
456 for (const Passenger &p : car.passengers)
457 stops.insert(r.groups[p.group].b);
458 vector<string> ss, gs;
459 for (int f : stops)
460 ss.push_back(num(f));
461 for (int g : car.calls)
462 gs.push_back("G" + num(g + 1));
463 emit("STATUS E" + num(i + 1) + " floor=" + num(car.floor) + " dir=" +
464 (car.direction > 0 ? "up"
465 : car.direction < 0 ? "down"
466 : "idle") +
467 " door=" + num(car.door) + " load=" + num(car.load()) + " stops=" + join(ss) +
468 " calls=" + join(gs) + " service=" + (car.inService ? "in" : "out"));
469 }
470 vector<string> ps;
471 for (int g : pending)
472 ps.push_back("G" + num(g + 1));
473 emit("STATUS pending=" + join(ps));
474 }
475 }
476 void action(int i) {
477 Car &car = r.cars[i];
478 bool moved = false;
479 if (car.door)
480 --car.door;
481 if (!car.door) {
482 int f = car.floor, d = car.direction, n = fire ? sign(evacuation - f) : choose(car);
483 bool stop = fire ? (n == 0 && !car.passengers.empty()) : false;
484 if (!fire) {
485 for (const Passenger &p : car.passengers)
486 if (r.groups[p.group].b == f)
487 stop = true;
488 if (car.load() * 100 < c.capacity * c.bypass)
489 for (int g : car.calls)
490 if (r.groups[g].a == f && r.groups[g].direction() == n)
491 stop = true;
492 }
493 if (stop) {
494 emit("E" + num(i + 1) + " OPEN " + num(f));
495 car.door = c.door;
496 ++car.stops;
497 car.energy += 2;
498 vector<Passenger> riding;
499 for (const Passenger &p : car.passengers) {
500 Group &g = r.groups[p.group];
501 if (g.b == f) {
502 emit("E" + num(i + 1) + " ALIGHT G" + num(p.group + 1) + " x" +
503 num(p.count));
504 g.delivered += p.count;
505 g.lastAlight = time;
506 g.wait += (p.time - g.time) * p.count;
507 g.travel += (time - g.time) * p.count;
508 g.maxWait = max(g.maxWait, p.time - g.time);
509 r.delivered += p.count;
510 r.wait += (p.time - g.time) * p.count;
511 r.travel += (time - g.time) * p.count;
512 r.maxWait = max(r.maxWait, p.time - g.time);
513 } else if (fire) {
514 emit("E" + num(i + 1) + " EVACUATE G" + num(p.group + 1) + " x" +
515 num(p.count));
516 g.lost += p.count;
517 r.evacuated += p.count;
518 } else
519 riding.push_back(p);
520 }
521 car.passengers = riding;
522 if (!fire) {
523 n = choose(car);
524 if (n)
525 car.direction = n;
526 vector<int> calls;
527 for (int g : car.calls)
528 if (r.groups[g].a == f && r.groups[g].direction() == n)
529 calls.push_back(g);
530 for (int gi : calls) {
531 Group &g = r.groups[gi];
532 int k = min(c.capacity - car.load(), g.waiting);
533 if (k) {
534 car.passengers.push_back({gi, k, time});
535 g.waiting -= k;
536 g.boarded += k;
537 g.lastBoard = time;
538 g.cars.insert(i + 1);
539 car.boarded += k;
540 emit("E" + num(i + 1) + " BOARD G" + num(gi + 1) + " x" + num(k));
541 }
542 if (!g.waiting)
543 remove(gi);
544 else {
545 emit("E" + num(i + 1) + " LEFT G" + num(gi + 1) + " x" +
546 num(g.waiting));
547 assign(gi, i);
548 }
549 }
550 }
551 } else if (n) {
552 car.direction = n;
553 move(car, n);
554 moved = true;
555 } else {
556 if (d) {
557 emit("E" + num(i + 1) + " IDLE " + num(f));
558 car.direction = 0;
559 car.idle = time;
560 }
561 if (!fire && car.inService && c.park >= 0 && f != c.homes[i] &&
562 time - car.idle >= c.park) {
563 move(car, sign(c.homes[i] - f));
564 moved = true;
565 if (car.floor == c.homes[i])
566 emit("E" + num(i + 1) + " PARK " + num(car.floor));
567 }
568 }
569 }
570 if (car.door || moved)
571 ++car.busy;
572 }

574 public:
575 Simulation(const Config &config, const vector<Event> &events, string p, bool log)
576 : c(config), input(events), policy(std::move(p)), logging(log) {
577 for (int f : c.starts) {
578 Car car;
579 car.floor = f;
580 r.cars.push_back(car);
581 }
582 }
583 Result run() {
584 size_t cursor = 0;
585 for (time = 0; time <= c.limit; ++time) {
586 if (c.patience >= 0)
587 for (int i = 0; i < (int)r.groups.size(); ++i)
588 if (r.groups[i].waiting && time == r.groups[i].time + c.patience)
589 abandon(i, "ABANDON");
590 while (cursor < input.size() && input[cursor].time == time)
591 event(input[cursor++]);
592 for (int i = 0; i < c.cars; ++i)
593 action(i);
594 r.end = time;
595 bool done = cursor == input.size();
596 for (int i = 0; i < c.cars; ++i) {
597 const Car &car = r.cars[i];
598 done = done && !car.door && car.passengers.empty() && car.calls.empty() &&
599 (fire ? car.floor == evacuation
600 : !car.inService || c.park < 0 || car.floor == c.homes[i]);
601 }
602 if (done) {
603 emit("END");
604 break;
605 }
606 if (time == c.limit) {
607 emit("ABORT");
608 r.aborted = true;
609 }
610 }
611 return r;
612 }

614void aggregate(const string &label, const vector<Group> &gs) {
615 Integer passengers = 0, delivered = 0, wait = 0;
616 for (const Group &g : gs) {
617 passengers += g.n;
618 delivered += g.delivered;
619 wait += g.wait;
620 }
621 cout << label << ": groups=" << gs.size() << " passengers=" << passengers
622 << " delivered=" << delivered << " avg_wait=" << avg(wait, delivered) << '\n';
623}
624void statistics(const Config &c, const Result &r) {
625 Integer passengers = 0;
626 map<int, vector<Group>> floors, periods;
627 for (int i = 0; i < (int)r.groups.size(); ++i) {
628 const Group &g = r.groups[i];
629 passengers += g.n;
630 vector<string> cs;
631 for (int car : g.cars)
632 cs.push_back(num(car));
633 cout << "G" << i + 1 << " t=" << g.time << ' ' << g.a << "->" << g.b << " n=" << g.n
634 << " delivered=" << g.delivered << " lost=" << g.lost
635 << " wait=" << (g.boarded == g.n ? num(g.lastBoard - g.time) : "-")
636 << " travel=" << (g.delivered == g.n ? num(g.lastAlight - g.time) : "-")
637 << " cars=" << join(cs) << '\n';
638 floors[g.a].push_back(g);
639 if (c.period >= 0)
640 periods[g.time / c.period].push_back(g);
641 }
642 for (int i = 0; i < c.cars; ++i) {
643 const Car &car = r.cars[i];
644 Integer u = car.busy * 1000 / (r.end + 1);
645 cout << "E" << i + 1 << " distance=" << car.distance << " stops=" << car.stops
646 << " boarded=" << car.boarded << " busy=" << car.busy << " util=" << u / 10 << '.'
647 << u % 10 << "% energy=" << car.energy << '\n';
648 }
649 for (const auto &entry : floors)
650 aggregate("F" + num(entry.first), entry.second);
651 for (const auto &entry : periods) {
652 int k = entry.first;
653 aggregate("P" + num(k) + " " + num(k * c.period) + "-" + num(k * c.period + c.period - 1),
654 entry.second);
655 }
656 cout << "summary: groups=" << r.groups.size() << " passengers=" << passengers
657 << " delivered=" << r.delivered << " abandoned=" << r.abandoned
658 << " evacuated=" << r.evacuated << " avg_wait=" << avg(r.wait, r.delivered)
659 << " avg_travel=" << avg(r.travel, r.delivered)
660 << " max_wait=" << (r.delivered ? num(r.maxWait) : "-") << " end=" << r.end << '\n';
661}

713 vector<Event> input = validate(c, raw);

734 if (d)

736 return x.travel * y.delivered < y.travel * x.delivered;
737 });

762 travel += scenarios[i].result.travel;
763 }

766 << " avg_travel=" << avg(travel, delivered) << '\n';
767 }
