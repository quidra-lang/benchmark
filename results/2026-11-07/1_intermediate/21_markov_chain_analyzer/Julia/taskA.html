15delta(i::Int, j::Int) = i == j ? ONE : ZERO
16answer(flag::Bool) = flag ? "yes" : "no"
17fraction_text(value::Rational) = "$(numerator(value))/$(denominator(value))"
18function value_text(value::Rational)
19 whole, remainder = fldmod(numerator(value), denominator(value))
20 digits = div(remainder * 1000000, denominator(value))
21 return "$(fraction_text(value))($whole.$(lpad(string(digits), 6, '0')))"
22end
23time_text(value) = value === nothing ? "inf" : value_text(value)

25function parse_probability(text::AbstractString)
26 parts = split(text, '/')
27 numerator_ = parse(BigInt, parts[1])
28 denominator_ = length(parts) == 2 ? parse(BigInt, parts[2]) : big(1)
29 if denominator_ == 0 || numerator_ > denominator_
30 fail("BAD PROBABILITY")
31 end
32 return numerator_ // denominator_
33end

43unit_matrix(n::Int) = [delta(i, j) for i in 1:n, j in 1:n]

45function eliminate!(matrix::Vector{Vector{Rat}}, n::Int)
46 for column in 1:n
47 pivot_row = column
48 while matrix[pivot_row][column] == ZERO
49 pivot_row += 1
50 end
51 matrix[column], matrix[pivot_row] = matrix[pivot_row], matrix[column]
52 pivot = matrix[column][column]
53 matrix[column] = [value / pivot for value in matrix[column]]
54 for row in 1:n
55 row == column && continue
56 factor = matrix[row][column]
57 if factor != ZERO
58 matrix[row] = [value - factor * base
59 for (value, base) in zip(matrix[row], matrix[column])]
60 end
61 end
62 end
63 return [row[n+1:end] for row in matrix]
64end

92labels(c::Chain, indices) = [c.names[i] for i in indices]
93row_sum(c::Chain, i::Int) = sum(c.matrix[i, :]; init=ZERO)

95edge(c::Chain, i::Int, j::Int) = c.matrix[i, j] > ZERO

98function advance(c::Chain, distribution::Vector{Rat})
99 n = nstates(c)
100 return Rat[sum((distribution[i] * c.matrix[i, j] for i in 1:n); init=ZERO)
101 for j in 1:n]
102end

104function reachable(c::Chain, start::Int)
105 seen = fill(false, nstates(c))
106 seen[start] = true
107 stack = [start]
108 while !isempty(stack)
109 i = pop!(stack)
110 for j in 1:nstates(c)
111 if edge(c, i, j) && !seen[j]
112 seen[j] = true
113 push!(stack, j)
114 end
115 end
116 end
117 return seen
118end

120struct CommunicatingClass
121 number::Int
122 members::Vector{Int}
123 closed::Bool
124 levels::Vector{Int}
125 period::Int
126end
127
128function CommunicatingClass(c::Chain, number::Int, members::Vector{Int})
129 inside = membership(nstates(c), members)
130 closed = !any(edge(c, i, j) && !inside[j]
131 for i in members for j in 1:nstates(c))
132 levels = fill(-1, nstates(c))
133 levels[first(members)] = 0
134 queue = [first(members)]
135 while !isempty(queue)
136 u = popfirst!(queue)
137 for v in members
138 if edge(c, u, v) && levels[v] < 0
139 levels[v] = levels[u] + 1
140 push!(queue, v)
141 end
142 end
143 end
144 period = 0
145 for u in members, v in members
146 if edge(c, u, v)
147 period = gcd(period, levels[u] + 1 - levels[v])
148 end
149 end
150 return CommunicatingClass(number, members, closed, levels, period)
151end
152
153struct Decomposition
154 reach::Vector{Vector{Bool}}
155 classes::Vector{CommunicatingClass}
156 class_of::Vector{Int}
157end
158
159function Decomposition(c::Chain)
160 n = nstates(c)
161 reach = [reachable(c, i) for i in 1:n]
162 classes = CommunicatingClass[]
163 class_of = fill(0, n)
164 for i in 1:n
165 class_of[i] != 0 && continue
166 members = [j for j in 1:n if reach[i][j] && reach[j][i]]
167 for j in members
168 class_of[j] = length(classes) + 1
169 end
170 push!(classes, CommunicatingClass(c, length(classes) + 1, members))
171 end
172 return Decomposition(reach, classes, class_of)
173end
174
175closed_classes(d::Decomposition) = [item for item in d.classes if item.closed]
176containing(d::Decomposition, i::Int) = d.classes[d.class_of[i]]
177reaches(d::Decomposition, i::Int, targets::Vector{Bool}) =
178 any(d.reach[i][j] && targets[j] for j in eachindex(targets))

180struct Absorption
181 transient::Vector{Int}
182 absorbing::Vector{Int}
183 fundamental::Vector{Vector{Rat}}
184 steps::Vector{Rat}
185 probabilities::Vector{Vector{Rat}}
186end
187
188function Absorption(c::Chain)
189 transient = [i for i in 1:nstates(c) if !absorbing(c, i)]
190 absorbing_states = [i for i in 1:nstates(c) if absorbing(c, i)]
191 count = length(transient)
192 matrix = Vector{Rat}[]
193 for (row, i) in enumerate(transient)
194 left = Rat[delta(i, j) - c.matrix[i, j] for j in transient]
195 right = Rat[delta(row, col) for col in 1:count]
196 push!(matrix, vcat(left, right))
197 end
198 fundamental = eliminate!(matrix, count)
199 steps = Rat[sum(row; init=ZERO) for row in fundamental]
200 probabilities = Vector{Rat}[]
201 for row in fundamental
202 push!(probabilities, Rat[
203 sum((row[k] * c.matrix[transient[k], a] for k in 1:count); init=ZERO)
204 for a in absorbing_states
205 ])
206 end
207 return Absorption(transient, absorbing_states, fundamental, steps, probabilities)
208end
209
210function variance(a::Absorption, i::Int)
211 count = length(a.transient)
212 total = sum(((2 * a.fundamental[i][j] - delta(i, j)) * a.steps[j]
213 for j in 1:count); init=ZERO)
214 return total - a.steps[i]^2
215end

217function stationary_distribution(c::Chain, members::Vector{Int})
218 n = length(members)
219 matrix = Vector{Rat}[]
220 for r in 1:(n-1)
221 row = Rat[c.matrix[members[k], members[r]] - delta(k, r) for k in 1:n]
222 push!(matrix, vcat(row, Rat[ZERO]))
223 end
224 push!(matrix, fill(ONE, n + 1))
225 return Rat[row[1] for row in eliminate!(matrix, n)]
226end
227
228function irreducible_distribution(c::Chain, d::Decomposition)
229 length(d.classes) == 1 || fail("NOT IRREDUCIBLE")
230 return stationary_distribution(c, first(d.classes).members)
231end

233function solve_restricted(c::Chain, states::Vector{Int}, constants::Vector{Rat})
234 matrix = Vector{Rat}[]
235 for (row, i) in enumerate(states)
236 coefficients = Rat[delta(i, j) - c.matrix[i, j] for j in states]
237 push!(matrix, vcat(coefficients, Rat[constants[row]]))
238 end
239 return Rat[row[1] for row in eliminate!(matrix, length(states))]
240end

242function hitting_probabilities(c::Chain, d::Decomposition, targets::Vector{Bool})
243 n = nstates(c)
244 probabilities = Rat[targets[i] ? ONE : ZERO for i in 1:n]
245 remaining = [i for i in 1:n if !targets[i] && reaches(d, i, targets)]
246 if !isempty(remaining)
247 constants = Rat[
248 sum((c.matrix[i, j] for j in 1:n if targets[j]); init=ZERO)
249 for i in remaining
250 ]
251 for (i, value) in zip(remaining, solve_restricted(c, remaining, constants))
252 probabilities[i] = value
253 end
254 end
255 return probabilities
256end
257
258function hitting_times(c::Chain, targets::Vector{Bool}, probabilities::Vector{Rat})
259 n = nstates(c)
260 times = Union{Nothing,Rat}[targets[i] ? ZERO : nothing for i in 1:n]
261 certain = [i for i in 1:n if !targets[i] && probabilities[i] == ONE]
262 if !isempty(certain)
263 constants = fill(ONE, length(certain))
264 for (i, value) in zip(certain, solve_restricted(c, certain, constants))
265 times[i] = value
266 end
267 end
268 return times
269end

271function regular_exponent(c::Chain)
272 n = nstates(c)
273 edges = [edge(c, i, j) for i in 1:n, j in 1:n]
274 power = edges
275 for exponent in 1:((n-1)^2+1)
276 all(power) && return exponent
277 power = [any(power[i, k] && edges[k, j] for k in 1:n)
278 for i in 1:n, j in 1:n]
279 end
280 return nothing
281end

359function build_chain!(m::MarkovAnalyzer, a::Vector{String})
360 name = a[1]
361 order = parse(Int, a[2])
362 order in (1, 2) || fail("BAD ORDER")
363 tokens = a[3:end]
364 length(tokens) > order || fail("TOO FEW TOKENS")
365 observations = if order == 1
366 tokens
367 else
368 ["$(tokens[i]).$(tokens[i+1])" for i in 1:length(tokens)-1]
369 end
370 names = unique(observations)
371 length(names) <= MAX_STATES || fail("TOO MANY STATES")
372 c = Chain(names)
373 counts = zeros(Int, nstates(c), nstates(c))
374 for i in 1:length(observations)-1
375 counts[c.positions[observations[i]], c.positions[observations[i+1]]] += 1
376 end
377 emit!(m, "BUILD", name, "order=$order", "states=$(nstates(c))",
378 "transitions=$(length(observations)-1)")
379 for i in 1:nstates(c)
380 total = sum(counts[i, :])
381 if total == 0
382 c.matrix[i, i] = ONE
383 emit!(m, "FROM", names[i], "total=0", "absorbing")
384 continue
385 end
386 parts = String[]
387 for j in 1:nstates(c)
388 if counts[i, j] > 0
389 c.matrix[i, j] = big(counts[i, j]) // big(total)
390 push!(parts, "$(names[j]):$(counts[i,j])")
391 end
392 end
393 emit!(m, "FROM", names[i], "total=$total", join(parts, " "))
394 end
395 m.chains[name] = c
396end

398function gambler!(m::MarkovAnalyzer, a::Vector{String})
399 name = a[1]
400 goal, initial = parse(Int, a[2]), parse(Int, a[3])
401 (goal > 0 && goal <= MAX_GAMBLER_SIZE) || fail("BAD SIZE")
402 initial <= goal || fail("BAD START")
403 p = parse_probability(a[4])
404 p != ZERO && p != ONE || fail("BAD PROBABILITY")
405 c = Chain(["s$i" for i in 0:goal])
406 c.matrix[1, 1] = ONE
407 c.matrix[goal+1, goal+1] = ONE
408 for i in 1:goal-1
409 c.matrix[i+1, i+2] = p
410 c.matrix[i+1, i] = ONE - p
411 end
412 m.chains[name] = c
413 absorption = Absorption(c)
414 ruin = vcat(Rat[ONE], Rat[row[1] for row in absorption.probabilities], Rat[ZERO])
415 duration = vcat(Rat[ZERO], absorption.steps, Rat[ZERO])
416 emit!(m, "GAMBLER", name, "g=$goal", "p=$(fraction_text(p))",
417 "start=s$initial", "ruin=$(value_text(ruin[initial+1]))",
418 "duration=$(value_text(duration[initial+1]))")
419 for i in 0:goal
420 emit!(m, "CAPITAL", "s$i",
421 "ruin=$(value_text(ruin[i+1]))",
422 "win=$(value_text(ONE - ruin[i+1]))",
423 "duration=$(value_text(duration[i+1]))")
424 end
425end

519function absorb!(m::MarkovAnalyzer, a::Vector{String})
520 c = stochastic(m, a[1])
521 absorbing_states = Bool[absorbing(c, i) for i in 1:nstates(c)]
522 any(absorbing_states) || fail("NO ABSORBING STATE")
523 d = Decomposition(c)
524 for i in 1:nstates(c)
525 if !absorbing_states[i] && !reaches(d, i, absorbing_states)
526 fail("NOT ABSORBING $(c.names[i])")
527 end
528 end
529 result = Absorption(c)
530 emit!(m, "ABSORB", a[1],
531 "transient=$(length(result.transient))",
532 "absorbing=$(length(result.absorbing))")
533 isempty(result.transient) && return
534 names = labels(c, result.transient)
535 emit!(m, "TRANSIENT", join(names, " "))
536 emit!(m, "ABSORBING", join(labels(c, result.absorbing), " "))
537 for (name, row) in zip(names, result.fundamental)
538 emit!(m, "N", name, join((fraction_text(v) for v in row), " "))
539 end
540 for (i, name) in enumerate(names)
541 emit!(m, "STEPS", name, "mean=$(value_text(result.steps[i]))",
542 "var=$(value_text(variance(result, i)))")
543 end
544 for (name, row) in zip(names, result.probabilities)
545 emit!(m, "B", name, join((fraction_text(v) for v in row), " "))
546 end

564 c = stochastic(m, a[1])
565 d = Decomposition(c)
566 pi = irreducible_distribution(c, d)
567 times = fill(ZERO, nstates(c), nstates(c))

577 emit!(m, "PASSAGE", a[1], "kemeny=$(value_text(kemeny))")

581function first_passage!(m::MarkovAnalyzer, a::Vector{String})
582 c = stochastic(m, a[1])
583 x, y = state_index(c, a[2]), state_index(c, a[3])
584 steps = parse(Int, a[4])
585 steps != 0 || fail("BAD STEPS")
586 current = Rat[c.matrix[i, y] for i in 1:nstates(c)]
587 firsts = Rat[current[x]]
588 for _ in 2:steps
589 current = Rat[
590 sum((c.matrix[i, j] * current[j] for j in 1:nstates(c) if j != y); init=ZERO)
591 for i in 1:nstates(c)
592 ]
593 push!(firsts, current[x])
594 end
595 emit!(m, "FIRSTPASSAGE", a[1], a[2], a[3], "k=$steps",
596 "total=$(value_text(sum(firsts; init=ZERO)))")
597 cumulative = ZERO
598 for (t, value) in enumerate(firsts)
599 cumulative += value
600 emit!(m, "FP", t, value_text(value), "cum=$(value_text(cumulative))")
601 end
602end

622 start = state_index(c, a[2])
623 steps = parse(Int, a[3])
624 steps > 0 || fail("BAD STEPS")
625 best = fill(ONE, steps + 1, nstates(c))
626 for t in steps:-1:1
627 for i in 1:nstates(c)
628 best[t, i] = maximum(c.matrix[i, j] * best[t+1, j]
629 for j in 1:nstates(c))
630 end
631 end
632 path = [start]
633 for t in 1:steps
634 current = last(path)
635 j = findfirst(j -> c.matrix[current, j] * best[t+1, j] == best[t, current],
636 1:nstates(c))
637 push!(path, j)
638 end
639 emit!(m, "MOSTLIKELY", a[1], a[2], "k=$steps",
640 "prob=$(value_text(best[1, start]))")
641 emit!(m, "PATH", join(labels(c, path), " "))
642end

644function reversible!(m::MarkovAnalyzer, a::Vector{String})
645 c = stochastic(m, a[1])
646 pi = irreducible_distribution(c, Decomposition(c))
647 violations = Tuple{Int, Int, Rat, Rat}[]
648 for i in 1:nstates(c)
649 for j in i+1:nstates(c)
650 forward = pi[i] * c.matrix[i, j]
651 backward = pi[j] * c.matrix[j, i]
652 forward != backward && push!(violations, (i, j, forward, backward))
653 end
654 end
655 if isempty(violations)
656 emit!(m, "REVERSIBLE", a[1], "yes")
657 else
658 emit!(m, "REVERSIBLE", a[1], "no", "violations=$(length(violations))")
659 for (i, j, forward, backward) in violations
660 emit!(m, "VIOLATION", c.names[i], c.names[j],
661 value_text(forward), value_text(backward))
662 end
663 end
664end

666function reverse_chain!(m::MarkovAnalyzer, a::Vector{String})
667 c = stochastic(m, a[1])
668 pi = irreducible_distribution(c, Decomposition(c))
669 reversed = Chain(copy(c.names))
670 for i in 1:nstates(c), j in 1:nstates(c)
671 reversed.matrix[i, j] = pi[j] * c.matrix[j, i] / pi[i]
672 end
673 m.chains[a[2]] = reversed
674 emit!(m, "REVERSE", a[1], a[2],
675 "reversible=$(answer(reversed.matrix == c.matrix))")
676 display!(m, reversed.names, reversed.matrix)
677end

679function mixing!(m::MarkovAnalyzer, a::Vector{String})
680 c = stochastic(m, a[1])
681 steps = parse(Int, a[2])
682 steps != 0 || fail("BAD STEPS")
683 epsilon = parse_probability(a[3])
684 pi = irreducible_distribution(c, Decomposition(c))
685 matrix = unit_matrix(nstates(c))
686 mixing_time = nothing
687 lines = Tuple{Int, Rat, String}[]
688 for t in 1:steps
689 updated = fill(ZERO, nstates(c), nstates(c))
690 for i in 1:nstates(c)
691 updated[i, :] = advance(c, collect(matrix[i, :]))
692 end
693 matrix = updated
694 distances = Rat[
695 sum((abs(matrix[i, j] - pi[j]) for j in 1:nstates(c)); init=ZERO) / 2
696 for i in 1:nstates(c)
697 ]
698 largest = maximum(distances)
699 worst = findfirst(==(largest), distances)
700 if mixing_time === nothing && largest <= epsilon
701 mixing_time = t
702 end
703 push!(lines, (t, largest, c.names[worst]))
704 end
705 emit!(m, "MIXING", a[1], "k=$steps", "eps=$(fraction_text(epsilon))",
706 "time=$(mixing_time === nothing ? "none" : mixing_time)")
707 for (t, largest, worst) in lines
708 emit!(m, "TV", t, value_text(largest), "from=$worst")
709 end
710end

712function lump!(m::MarkovAnalyzer, a::Vector{String})
713 c = stochastic(m, a[1])
714 seen = fill(false, nstates(c))
715 blocks = Vector{Int}[]
716 for group in split(a[3], '|')
717 block = Int[]
718 for text in split(group, ',')
719 index = state_index(c, String(text))
720 seen[index] && fail("BAD PARTITION")
721 seen[index] = true
722 push!(block, index)
723 end
724 push!(blocks, sort(block))
725 end
726 all(seen) || fail("BAD PARTITION")
727 sort!(blocks; by=first)
728 for block in blocks
729 representative = first(block)
730 for target in blocks
731 expected = block_sum(c, representative, target)
732 for i in block
733 actual = block_sum(c, i, target)
734 if actual != expected
735 emit!(m, "LUMP", a[1], "not", "lumpable",
736 "block=$(c.names[representative])",
737 "target=$(c.names[first(target)])",
738 "$(c.names[representative])=$(value_text(expected))",
739 "$(c.names[i])=$(value_text(actual))")
740 return
741 end
742 end
743 end
744 end
745 lumped = Chain([c.names[first(block)] for block in blocks])
746 for (i, block) in enumerate(blocks), (j, target) in enumerate(blocks)
747 lumped.matrix[i, j] = block_sum(c, first(block), target)
748 end
749 m.chains[a[2]] = lumped
750 emit!(m, "LUMP", a[1], a[2], "blocks=$(length(blocks))")
751 display!(m, lumped.names, lumped.matrix)
752end

754function reward!(m::MarkovAnalyzer, a::Vector{String})
755 c = stochastic(m, a[1])
756 length(a) - 2 == nstates(c) || fail("BAD LENGTH")
757 gamma = parse_probability(a[2])
758 gamma != ONE || fail("BAD DISCOUNT")
759 rewards = Rat[parse(BigInt, text) // big(1) for text in a[3:end]]
760 matrix = Vector{Rat}[]
761 for i in 1:nstates(c)
762 coefficients = Rat[delta(i, j) - gamma * c.matrix[i, j] for j in 1:nstates(c)]
763 push!(matrix, vcat(coefficients, Rat[rewards[i]]))
764 end
765 values = Rat[row[1] for row in eliminate!(matrix, nstates(c))]
766 d = Decomposition(c)
767 gains = fill(ZERO, nstates(c))
768 for item in closed_classes(d)
769 distribution = stationary_distribution(c, item.members)
770 gain = sum((rho * rewards[i] for (i, rho) in zip(item.members, distribution));
771 init=ZERO)
772 targets = membership(nstates(c), item.members)
773 probabilities = hitting_probabilities(c, d, targets)
774 for i in 1:nstates(c)
775 gains[i] += probabilities[i] * gain
776 end
777 end
778 emit!(m, "REWARD", a[1], "gamma=$(fraction_text(gamma))")
779 for i in 1:nstates(c)
780 emit!(m, "VALUE", c.names[i],
781 "discounted=$(value_text(values[i]))",
782 "gain=$(value_text(gains[i]))")
783 end
784end

786function restrict_chain!(m::MarkovAnalyzer, a::Vector{String})
787 c = stochastic(m, a[1])
788 inside = membership(nstates(c), distinct_states(c, a[3:end]))
789 for i in 1:nstates(c)
790 if inside[i] && any(edge(c, i, j) && !inside[j] for j in 1:nstates(c))
791 fail("NOT CLOSED $(c.names[i])")
792 end
793 end
794 members = [i for i in 1:nstates(c) if inside[i]]
795 restricted = Chain(labels(c, members))
796 for (i, original_i) in enumerate(members), (j, original_j) in enumerate(members)
797 restricted.matrix[i, j] = c.matrix[original_i, original_j]
798 end
799 m.chains[a[2]] = restricted
800 emit!(m, "RESTRICT", a[1], a[2], "states=$(length(members))")
801 display!(m, restricted.names, restricted.matrix)
802end

804function simulate!(m::MarkovAnalyzer, a::Vector{String})
805 c = stochastic(m, a[1])
806 start = state_index(c, a[2])
807 steps, seed = parse(Int, a[3]), parse(Int, a[4])
808 steps > 0 || fail("BAD STEPS")
809 seed > 0 && seed < MODULUS || fail("BAD SEED")
810 random_value = seed
811 path = [start]
812 for _ in 1:steps
813 row = collect(c.matrix[last(path), :])
814 common = foldl(lcm, (denominator(v) for v in row); init=big(1))
815 random_value = mod(MULTIPLIER * random_value, MODULUS)
816 threshold = mod(random_value, common)
817 cumulative = ZERO
818 for (j, value) in enumerate(row)
819 cumulative += value
820 if cumulative * common > threshold
821 push!(path, j)
822 break
823 end
824 end
825 end
826 emit!(m, "SIMULATE", a[1], a[2], "steps=$steps", "seed=$seed",
827 "end=$(c.names[last(path)])")
828 emit!(m, "PATH", join(labels(c, path), " "))
829 for i in 1:nstates(c)
830 emit!(m, "VISIT", c.names[i], count(==(i), path))
831 end
832end

834function execute!(m::MarkovAnalyzer, line::String)
835 parts = String.(split(line, ' '))
836 command, arguments = parts[1], parts[2:end]
837 try
838 if command == "CHAIN"
839 create_chain!(m, arguments)
840 elseif command == "P"
841 set_probability!(m, arguments)
842 elseif command == "ROW"
843 set_row!(m, arguments)
844 elseif command == "SHOW"
845 show_chain!(m, arguments)
846 elseif command == "CHECK"
847 check_chain!(m, arguments)
848 elseif command == "BUILD"
849 build_chain!(m, arguments)
850 elseif command == "GAMBLER"
851 gambler!(m, arguments)
852 elseif command == "CLASSES"
853 classes!(m, arguments)
854 elseif command == "CLASSIFY"
855 classify!(m, arguments)
856 elseif command == "STEP"
857 step!(m, arguments)
858 elseif command == "POWER"
859 power!(m, arguments)
860 elseif command == "STATIONARY"
861 stationary!(m, arguments)
862 elseif command == "RECURRENCE"
863 recurrence!(m, arguments)
864 elseif command == "ABSORB"
865 absorb!(m, arguments)
866 elseif command == "HITTING"
867 hitting!(m, arguments)
868 elseif command == "PASSAGE"
869 passage!(m, arguments)
870 elseif command == "FIRSTPASSAGE"
871 first_passage!(m, arguments)
872 elseif command == "PATHPROB"
873 path_probability!(m, arguments)
874 elseif command == "MOSTLIKELY"
875 most_likely!(m, arguments)
876 elseif command == "REVERSIBLE"
877 reversible!(m, arguments)
878 elseif command == "REVERSE"
879 reverse_chain!(m, arguments)
880 elseif command == "MIXING"
881 mixing!(m, arguments)
882 elseif command == "LUMP"
883 lump!(m, arguments)
884 elseif command == "REWARD"
885 reward!(m, arguments)
886 elseif command == "RESTRICT"
887 restrict_chain!(m, arguments)
888 elseif command == "SIMULATE"
889 simulate!(m, arguments)
890 end
891 catch error
892 if error isa CommandError
893 emit!(m, "ERROR", error.reason)
894 else
895 rethrow()
896 end
897 end
898end
