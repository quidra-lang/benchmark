8#[derive(Clone, Copy, PartialEq, Eq)]
9struct Fraction {
10 numerator: i64,
11 denominator: i64,
12}
13
14fn gcd(mut a: i64, mut b: i64) -> i64 {
15 a = a.abs();
16 b = b.abs();
17 while b != 0 {
18 (a, b) = (b, a % b);
19 }
20 a
21}
22
23impl Fraction {
24 const ZERO: Self = Self { numerator: 0, denominator: 1 };
25 const ONE: Self = Self { numerator: 1, denominator: 1 };
26
27 fn new(mut numerator: i64, mut denominator: i64) -> Self {
28 if denominator < 0 {
29 numerator = -numerator;
30 denominator = -denominator;
31 }
32 let divisor = gcd(numerator, denominator);
33 Self { numerator: numerator / divisor, denominator: denominator / divisor }
34 }
35
36 fn integer(value: i64) -> Self {
37 Self { numerator: value, denominator: 1 }
38 }
39
40 fn abs(self) -> Self {
41 Self { numerator: self.numerator.abs(), ..self }
42 }
43
44 fn value(self) -> String {
45 let whole = self.numerator / self.denominator;
46 let decimal = self.numerator % self.denominator * 1_000_000 / self.denominator;
47 format!("{self}({whole}.{decimal:06})")
48 }
49}
50
51impl Add for Fraction {
52 type Output = Self;
53 fn add(self, other: Self) -> Self {
54 Self::new(self.numerator * other.denominator + other.numerator * self.denominator,
55 self.denominator * other.denominator)
56 }
57}
58
59impl Sub for Fraction {
60 type Output = Self;
61 fn sub(self, other: Self) -> Self {
62 Self::new(self.numerator * other.denominator - other.numerator * self.denominator,
63 self.denominator * other.denominator)
64 }
65}
66
67impl Mul for Fraction {
68 type Output = Self;
69 fn mul(self, other: Self) -> Self {
70 Self::new(self.numerator * other.numerator, self.denominator * other.denominator)
71 }
72}
73
74impl Div for Fraction {
75 type Output = Self;
76 fn div(self, other: Self) -> Self {
77 Self::new(self.numerator * other.denominator, self.denominator * other.numerator)
78 }
79}
80
81impl Ord for Fraction {
82 fn cmp(&self, other: &Self) -> Ordering {
83 (self.numerator * other.denominator).cmp(&(other.numerator * self.denominator))
84 }
85}
86
87impl PartialOrd for Fraction {
88 fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
89 Some(self.cmp(other))
90 }
91}
92
93impl Sum for Fraction {
94 fn sum<I: Iterator<Item = Self>>(values: I) -> Self {
95 values.fold(Self::ZERO, |total, value| total + value)
96 }
97}
98
99impl fmt::Display for Fraction {
100 fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
101 write!(formatter, "{}/{}", self.numerator, self.denominator)
102 }
103}

109fn probability(text: &str) -> Result<Fraction, String> {
110 let mut parts = text.split('/');
111 let numerator: i64 = parts.next().unwrap().parse().unwrap();
112 let denominator: i64 = parts.next().unwrap_or("1").parse().unwrap();
113 if denominator == 0 || numerator > denominator {
114 return Err("BAD PROBABILITY".into());
115 }
116 Ok(Fraction::new(numerator, denominator))
117}

119fn number(text: &str) -> usize {
120 text.parse().unwrap()
121}

123fn matrix(rows: usize, columns: usize) -> Matrix {
124 vec![vec![ZERO; columns]; rows]
125}
126
127fn identity(size: usize) -> Matrix {
128 let mut result = matrix(size, size);
129 for (i, row) in result.iter_mut().enumerate() {
130 row[i] = ONE;
131 }
132 result
133}
134
135fn matrix_product(left: &Matrix, right: &Matrix) -> Matrix {
136 let mut result = matrix(left.len(), right[0].len());
137 for (i, row) in result.iter_mut().enumerate() {
138 for (j, value) in row.iter_mut().enumerate() {
139 *value = (0..right.len()).map(|k| left[i][k] * right[k][j]).sum();
140 }
141 }
142 result
143}
144
145fn eliminate(mut equations: Matrix) -> Matrix {
146 let size = equations.len();
147 for column in 0..size {
148 let pivot_row = (column..size).find(|&row| equations[row][column] != ZERO).unwrap();
149 equations.swap(column, pivot_row);
150 let pivot = equations[column][column];
151 for value in &mut equations[column] {
152 *value = *value / pivot;
153 }
154 let normalized = equations[column].clone();
155 for (row_number, row) in equations.iter_mut().enumerate() {
156 if row_number == column || row[column] == ZERO {
157 continue;
158 }
159 let factor = row[column];
160 for (j, value) in row.iter_mut().enumerate() {
161 *value = *value - factor * normalized[j];
162 }
163 }
164 }
165 equations
166}

168#[derive(Clone)]
169struct Chain {
170 states: Vec<String>,
171 transition: Matrix,
172}
173
174impl Chain {
175 fn new(states: Vec<String>) -> Self {
176 let transition = matrix(states.len(), states.len());
177 Self { states, transition }
178 }
179
180 fn state(&self, name: &str) -> Result<usize, String> {
181 self.states.iter().position(|state| state == name)
182 .ok_or_else(|| format!("UNKNOWN STATE {name}"))
183 }
184
185 fn row_sum(&self, row: usize, selected: &[usize]) -> Fraction {
186 selected.iter().map(|&j| self.transition[row][j]).sum()
187 }
188
189 fn targets(&self, names: &[&str]) -> Result<Vec<usize>, String> {
190 let mut selected = vec![false; self.states.len()];
191 for &name in names {
192 let index = self.state(name)?;
193 if selected[index] {
194 return Err(format!("DUPLICATE STATE {name}"));
195 }
196 selected[index] = true;
197 }
198 Ok((0..selected.len()).filter(|&i| selected[i]).collect())
199 }
200
201 fn display(&self, values: &Matrix, label: &str) -> Vec<String> {
202 let mut lines = vec![format!("STATES {}", self.states.join(" "))];
203 for (i, row) in values.iter().enumerate() {
204 let text: Vec<String> = row.iter().map(ToString::to_string).collect();
205 lines.push(format!("{label} {} {}", self.states[i], text.join(" ")));
206 }
207 lines
208 }

210 fn absorbing(&self) -> Vec<usize> {
211 (0..self.states.len()).filter(|&i| self.transition[i][i] == ONE).collect()
212 }

215struct Class {
216 states: Vec<usize>,
217 closed: bool,
218 period: usize,
219}
220
221struct Graph {
222 reachable: Vec<Vec<bool>>,
223 classes: Vec<Class>,
224 class_of: Vec<usize>,
225 levels: Vec<usize>,
226}
227
228impl Graph {
229 fn new(chain: &Chain) -> Self {
230 let size = chain.states.len();
231 let mut reachable = vec![vec![false; size]; size];
232 for (i, row) in reachable.iter_mut().enumerate() {
233 for (j, value) in row.iter_mut().enumerate() {
234 *value = i == j || chain.transition[i][j] > ZERO;
235 }
236 }
237 for k in 0..size {
238 for i in 0..size {
239 for j in 0..size {
240 if reachable[i][k] && reachable[k][j] {
241 reachable[i][j] = true;
242 }
243 }
244 }
245 }
246 let mut classes = Vec::new();
247 let mut class_of = vec![usize::MAX; size];
248 let mut levels = vec![0; size];
249 for root in 0..size {
250 if class_of[root] != usize::MAX {
251 continue;
252 }
253 let states: Vec<usize> = (0..size).filter(|&i| reachable[root][i] && reachable[i][root]).collect();
254 for &state in &states {
255 class_of[state] = classes.len();
256 }
257 let mut closed = true;
258 let mut visited = vec![false; size];
259 let mut queue = VecDeque::from([root]);
260 visited[root] = true;
261 while let Some(current) = queue.pop_front() {
262 for next in 0..size {
263 if chain.transition[current][next] == ZERO {
264 continue;
265 }
266 if !states.contains(&next) {
267 closed = false;
268 } else if !visited[next] {
269 visited[next] = true;
270 levels[next] = levels[current] + 1;
271 queue.push_back(next);
272 }
273 }
274 }
275 let mut period = 0;
276 for &i in &states {
277 for &j in &states {
278 if chain.transition[i][j] > ZERO {
279 period = gcd(period, levels[i] as i64 + 1 - levels[j] as i64);
280 }
281 }
282 }
283 classes.push(Class { states, closed, period: period as usize });
284 }
285 Self { reachable, classes, class_of, levels }
286 }
287
288 fn reaches(&self, source: usize, targets: &[usize]) -> bool {
289 targets.iter().any(|&target| self.reachable[source][target])
290 }
291}

293fn stationary(chain: &Chain, states: &[usize]) -> Vec<Fraction> {
294 let count = states.len();
295 let mut equations = matrix(count, count + 1);
296 for i in 0..count - 1 {
297 for j in 0..count {
298 equations[i][j] = chain.transition[states[j]][states[i]] - if i == j { ONE } else { ZERO };
299 }
300 }
301 equations[count - 1].fill(ONE);
302 let solved = eliminate(equations);
303 let mut distribution = vec![ZERO; chain.states.len()];
304 for (i, &state) in states.iter().enumerate() {
305 distribution[state] = solved[i][count];
306 }
307 distribution
308}

310struct Hitting {
311 probabilities: Vec<Fraction>,
312 times: Vec<Option<Fraction>>,
313}
314
315fn hitting(chain: &Chain, graph: &Graph, targets: &[usize], with_times: bool) -> Hitting {
316 let size = chain.states.len();
317 let mut probabilities = vec![ZERO; size];
318 let mut times = vec![None; size];
319 for &i in targets {
320 probabilities[i] = ONE;
321 times[i] = Some(ZERO);
322 }
323 let unknown: Vec<usize> = (0..size).filter(|i| !targets.contains(i) && graph.reaches(*i, targets)).collect();
324 if !unknown.is_empty() {
325 let count = unknown.len();
326 let mut equations = matrix(count, count + 1);
327 for i in 0..count {
328 for j in 0..count {
329 equations[i][j] = (if i == j { ONE } else { ZERO }) - chain.transition[unknown[i]][unknown[j]];
330 }
331 equations[i][count] = chain.row_sum(unknown[i], targets);
332 }
333 let solved = eliminate(equations);
334 for (i, &state) in unknown.iter().enumerate() {
335 probabilities[state] = solved[i][count];
336 }
337 }
338 if with_times {
339 let certain: Vec<usize> = unknown.into_iter().filter(|&i| probabilities[i] == ONE).collect();
340 if !certain.is_empty() {
341 let count = certain.len();
342 let mut equations = matrix(count, count + 1);
343 for i in 0..count {
344 for j in 0..count {
345 equations[i][j] = (if i == j { ONE } else { ZERO }) - chain.transition[certain[i]][certain[j]];
346 }
347 equations[i][count] = ONE;
348 }
349 let solved = eliminate(equations);
350 for (i, &state) in certain.iter().enumerate() {
351 times[state] = Some(solved[i][count]);
352 }
353 }
354 }
355 Hitting { probabilities, times }
356}

358struct Absorption {
359 transient: Vec<usize>,
360 absorbing: Vec<usize>,
361 fundamental: Matrix,
362 means: Vec<Fraction>,
363 variances: Vec<Fraction>,
364 probabilities: Matrix,
365}
366
367fn absorption(chain: &Chain) -> Absorption {
368 let absorbing = chain.absorbing();
369 let transient: Vec<usize> = (0..chain.states.len()).filter(|i| !absorbing.contains(i)).collect();
370 let count = transient.len();
371 let mut equations = matrix(count, 2 * count);
372 for i in 0..count {
373 for j in 0..count {
374 equations[i][j] = (if i == j { ONE } else { ZERO }) - chain.transition[transient[i]][transient[j]];
375 }
376 equations[i][count + i] = ONE;
377 }
378 let solved = eliminate(equations);
379 let fundamental: Matrix = solved.iter().map(|row| row[count..].to_vec()).collect();
380 let means: Vec<Fraction> = fundamental.iter().map(|row| row.iter().copied().sum()).collect();
381 let mut variances = vec![ZERO; count];
382 let mut probabilities = matrix(count, absorbing.len());
383 for i in 0..count {
384 let second: Fraction = (0..count).map(|j| {
385 (Fraction::integer(2) * fundamental[i][j] - if i == j { ONE } else { ZERO }) * means[j]
386 }).sum();
387 variances[i] = second - means[i] * means[i];
388 for (j, &target) in absorbing.iter().enumerate() {
389 probabilities[i][j] = (0..count).map(|k| fundamental[i][k] * chain.transition[transient[k]][target]).sum();
390 }
391 }
392 Absorption { transient, absorbing, fundamental, means, variances, probabilities }
393}

403fn build(words: &[&str]) -> Result<(Chain, Vec<String>), String> {
404 let order = number(words[2]);
405 let tokens = &words[3..];
406 if order != 1 && order != 2 {
407 return Err("BAD ORDER".into());
408 }
409 if tokens.len() <= order {
410 return Err("TOO FEW TOKENS".into());
411 }
412 let mut states = Vec::new();
413 let mut observations = Vec::new();
414 for i in 0..tokens.len() - order + 1 {
415 let name = if order == 1 { tokens[i].to_owned() } else { format!("{}.{}", tokens[i], tokens[i + 1]) };
416 let index = if let Some(index) = states.iter().position(|state| state == &name) {
417 index
418 } else {
419 if states.len() == 10 {
420 return Err("TOO MANY STATES".into());
421 }
422 states.push(name);
423 states.len() - 1
424 };
425 observations.push(index);
426 }
427 let mut chain = Chain::new(states);
428 let size = chain.states.len();
429 let mut counts = vec![vec![0_i64; size]; size];
430 let mut totals = vec![0_i64; size];
431 for pair in observations.windows(2) {
432 counts[pair[0]][pair[1]] += 1;
433 totals[pair[0]] += 1;
434 }
435 let mut lines = vec![format!("BUILD {} order={order} states={size} transitions={}", words[1], observations.len() - 1)];
436 for i in 0..size {
437 if totals[i] == 0 {
438 chain.transition[i][i] = ONE;
439 lines.push(format!("FROM {} total=0 absorbing", chain.states[i]));
440 } else {
441 let mut line = format!("FROM {} total={}", chain.states[i], totals[i]);
442 for j in 0..size {
443 chain.transition[i][j] = Fraction::new(counts[i][j], totals[i]);
444 if counts[i][j] > 0 {
445 line.push_str(&format!(" {}:{}", chain.states[j], counts[i][j]));
446 }
447 }
448 lines.push(line);
449 }
450 }
451 Ok((chain, lines))
452}

454fn gambler(words: &[&str]) -> Result<(Chain, Vec<String>), String> {
455 let goal = number(words[2]);
456 let start = number(words[3]);
457 if goal == 0 || goal > 9 {
458 return Err("BAD SIZE".into());
459 }
460 if start > goal {
461 return Err("BAD START".into());
462 }
463 let chance = probability(words[4])?;
464 if chance == ZERO || chance == ONE {
465 return Err("BAD PROBABILITY".into());
466 }
467 let mut chain = Chain::new((0..=goal).map(|i| format!("s{i}")).collect());
468 chain.transition[0][0] = ONE;
469 chain.transition[goal][goal] = ONE;
470 for i in 1..goal {
471 chain.transition[i][i + 1] = chance;
472 chain.transition[i][i - 1] = ONE - chance;
473 }
474 let result = absorption(&chain);
475 let mut ruin = vec![ZERO; goal + 1];
476 let mut duration = vec![ZERO; goal + 1];
477 ruin[0] = ONE;
478 for (i, &state) in result.transient.iter().enumerate() {
479 ruin[state] = result.probabilities[i][0];
480 duration[state] = result.means[i];
481 }
482 let mut lines = vec![format!("GAMBLER {} g={goal} p={chance} start=s{start} ruin={} duration={}", words[1], ruin[start].value(), duration[start].value())];
483 for i in 0..=goal {
484 lines.push(format!("CAPITAL s{i} ruin={} win={} duration={}", ruin[i].value(), (ONE - ruin[i]).value(), duration[i].value()));
485 }
486 Ok((chain, lines))
487}

489fn classify(name: &str, chain: &Chain, graph: &Graph) -> String {
490 let size = chain.states.len();
491 let absorbing = chain.absorbing();
492 let is_absorbing = !absorbing.is_empty() && (0..size).all(|i| graph.reaches(i, &absorbing));
493 let edges: Vec<Vec<bool>> = chain.transition.iter().map(|row| row.iter().map(|&p| p > ZERO).collect()).collect();
494 let mut current = edges.clone();
495 let mut exponent = None;
496 for step in 1..=(size - 1) * (size - 1) + 1 {
497 if current.iter().all(|row| row.iter().all(|&value| value)) {
498 exponent = Some(step);
499 break;
500 }
501 current = (0..size).map(|i| (0..size).map(|j| (0..size).any(|k| current[i][k] && edges[k][j])).collect()).collect();
502 }
503 let yes_no = |value| if value { "yes" } else { "no" };
504 format!("CLASSIFY {name} absorbing_states={} absorbing={} ergodic={} regular={} exponent={}",
505 absorbing.len(), yes_no(is_absorbing), yes_no(graph.classes.len() == 1), yes_no(exponent.is_some()),
506 exponent.map_or_else(|| "-".into(), |value| value.to_string()))
507}

509fn execute(chains: &mut HashMap<String, Chain>, words: &[&str]) -> Result<Vec<String>, String> {
510 let verb = words[0];
511 let name = words[1];
512 match verb {
513 "CHAIN" => {
514 let states = &words[2..];
515 if states.len() > 10 {
516 return Err("TOO MANY STATES".into());
517 }
518 for (i, state) in states.iter().enumerate() {
519 if states[..i].contains(state) {
520 return Err(format!("DUPLICATE STATE {state}"));
521 }
522 }
523 let chain = Chain::new(states.iter().map(|state| (*state).to_owned()).collect());
524 chains.insert(name.to_owned(), chain);
525 return Ok(vec![format!("CHAIN {name} states={}", states.len())]);
526 }
527 "BUILD" | "GAMBLER" => {
528 let (chain, lines) = if verb == "BUILD" { build(words)? } else { gambler(words)? };
529 chains.insert(name.to_owned(), chain);
530 return Ok(lines);
531 }
532 _ => {}
533 }
534 let mut chain = chains.get(name).cloned().ok_or_else(|| format!("UNKNOWN CHAIN {name}"))?;
535 let size = chain.states.len();
536 let all: Vec<usize> = (0..size).collect();
537 match verb {
538 "P" | "ROW" => {
539 let row = chain.state(words[2])?;
540 let line = if verb == "P" {
541 let column = chain.state(words[3])?;
542 chain.transition[row][column] = probability(words[4])?;
543 format!("P {name} {} {} {} rowsum={}", words[2], words[3], chain.transition[row][column], chain.row_sum(row, &all))
544 } else {
545 if words.len() - 3 != size {
546 return Err("BAD LENGTH".into());
547 }
548 chain.transition[row] = words[3..].iter().map(|text| probability(text)).collect::<Result<_, _>>()?;
549 format!("ROW {name} {} rowsum={}", words[2], chain.row_sum(row, &all))
550 };
551 chains.insert(name.to_owned(), chain);
552 return Ok(vec![line]);
553 }
554 "SHOW" => {
555 let mut lines = vec![format!("MATRIX {name} states={size}")];
556 lines.extend(chain.display(&chain.transition, "ROW"));
557 return Ok(lines);
558 }
559 "CHECK" => {
560 let mut bad = Vec::new();
561 for i in 0..size {
562 let sum = chain.row_sum(i, &all);
563 if sum != ONE {
564 bad.push(format!("BADROW {} sum={sum}", chain.states[i]));
565 }
566 }
567 let mut lines = vec![if bad.is_empty() { format!("CHECK {name} ok") } else { format!("CHECK {name} bad={}", bad.len()) }];
568 lines.extend(bad);
569 return Ok(lines);
570 }
571 _ => {}
572 }
573 for i in 0..size {
574 if chain.row_sum(i, &all) != ONE {
575 return Err(format!("NOT STOCHASTIC {}", chain.states[i]));
576 }
577 }
578 let graph = Graph::new(&chain);
579 let mut lines = Vec::new();
580 match verb {
581 "CLASSES" => {
582 lines.push(format!("CLASSES {name} count={}", graph.classes.len()));
583 for (i, class) in graph.classes.iter().enumerate() {
584 let kind = if class.closed { "RECURRENT" } else { "TRANSIENT" };
585 lines.push(format!("CLASS {} {kind} period={} size={} {}", i + 1, class.period, class.states.len(), names(&chain, &class.states, " ")));
586 if class.period >= 2 {
587 for remainder in 0..class.period {
588 let states: Vec<usize> = class.states.iter().copied().filter(|&state| graph.levels[state] % class.period == remainder).collect();
589 lines.push(format!("CYCLIC {remainder} {}", names(&chain, &states, " ")));
590 }
591 }
592 }
593 }
594 "CLASSIFY" => lines.push(classify(name, &chain, &graph)),
595 "STEP" => {
596 let mut distribution = vec![ZERO; size];
597 if words[2].as_bytes()[0].is_ascii_digit() {
598 let values: Vec<&str> = words[2].split(',').collect();
599 if values.len() != size {
600 return Err("BAD LENGTH".into());
601 }
602 distribution = values.iter().map(|text| probability(text)).collect::<Result<_, _>>()?;
603 if distribution.iter().copied().sum::<Fraction>() != ONE {
604 return Err("BAD DISTRIBUTION".into());
605 }
606 } else {
607 distribution[chain.state(words[2])?] = ONE;
608 }
609 let steps = number(words[3]);
610 for _ in 0..steps {
611 distribution = (0..size).map(|j| (0..size).map(|i| distribution[i] * chain.transition[i][j]).sum()).collect();
612 }
613 lines.push(format!("STEP {name} k={steps}"));
614 for (i, value) in distribution.iter().enumerate() {
615 lines.push(format!("PROB {} {}", chain.states[i], value.value()));
616 }
617 }
618 "POWER" => {
619 let steps = number(words[2]);
620 let mut power = identity(size);
621 for _ in 0..steps {
622 power = matrix_product(&power, &chain.transition);
623 }
624 lines.push(format!("POWER {name} k={steps}"));
625 lines.extend(chain.display(&power, "ROW"));
626 }
627 "STATIONARY" => {
628 lines.push(format!("STATIONARY {name} count={}", graph.classes.iter().filter(|class| class.closed).count()));
629 let mut number = 0;
630 for (i, class) in graph.classes.iter().enumerate() {
631 if class.closed {
632 number += 1;
633 let distribution = stationary(&chain, &class.states);
634 lines.push(format!("DIST {number} class={} size={}", i + 1, class.states.len()));
635 for &state in &class.states {
636 lines.push(format!("PI {} {}", chain.states[state], distribution[state].value()));
637 }
638 }
639 }
640 }
641 "RECURRENCE" => {
642 let state = chain.state(words[2])?;
643 let class = &graph.classes[graph.class_of[state]];
644 if !class.closed {
645 return Err(format!("TRANSIENT STATE {}", words[2]));
646 }
647 let chance = stationary(&chain, &class.states)[state];
648 lines.push(format!("RECURRENCE {name} {} pi={} mean={} period={}", words[2], chance.value(), (ONE / chance).value(), class.period));
649 }
650 "ABSORB" => {
651 let absorbing = chain.absorbing();
652 if absorbing.is_empty() {
653 return Err("NO ABSORBING STATE".into());
654 }
655 for i in 0..size {
656 if !graph.reaches(i, &absorbing) {
657 return Err(format!("NOT ABSORBING {}", chain.states[i]));
658 }
659 }
660 let result = absorption(&chain);
661 lines.push(format!("ABSORB {name} transient={} absorbing={}", result.transient.len(), result.absorbing.len()));
662 if !result.transient.is_empty() {
663 lines.push(format!("TRANSIENT {}", names(&chain, &result.transient, " ")));
664 lines.push(format!("ABSORBING {}", names(&chain, &result.absorbing, " ")));
665 for (i, &state) in result.transient.iter().enumerate() {
666 lines.push(format!("N {} {}", chain.states[state], fraction_list(&result.fundamental[i])));
667 }
668 for (i, &state) in result.transient.iter().enumerate() {
669 lines.push(format!("STEPS {} mean={} var={}", chain.states[state], result.means[i].value(), result.variances[i].value()));
670 }
671 for (i, &state) in result.transient.iter().enumerate() {
672 lines.push(format!("B {} {}", chain.states[state], fraction_list(&result.probabilities[i])));
673 }
674 }
675 }
676 "HITTING" | "RESTRICT" => {
677 let first = if verb == "HITTING" { 2 } else { 3 };
678 let targets = chain.targets(&words[first..])?;
679 if verb == "HITTING" {
680 let result = hitting(&chain, &graph, &targets, true);
681 lines.push(format!("HITTING {name} targets={}", names(&chain, &targets, ",")));
682 for i in 0..size {
683 let time = result.times[i].map_or_else(|| "inf".into(), Fraction::value);
684 lines.push(format!("HIT {} prob={} time={time}", chain.states[i], result.probabilities[i].value()));
685 }
686 } else {
687 for &i in &targets {
688 if (0..size).any(|j| !targets.contains(&j) && chain.transition[i][j] > ZERO) {
689 return Err(format!("NOT CLOSED {}", chain.states[i]));
690 }
691 }
692 let mut result = Chain::new(targets.iter().map(|&i| chain.states[i].clone()).collect());
693 for (i, &source) in targets.iter().enumerate() {
694 for (j, &target) in targets.iter().enumerate() {
695 result.transition[i][j] = chain.transition[source][target];
696 }
697 }
698 lines.push(format!("RESTRICT {name} {} states={}", words[2], targets.len()));
699 lines.extend(result.display(&result.transition, "ROW"));
700 chains.insert(words[2].to_owned(), result);
701 }
702 }
703 "FIRSTPASSAGE" => {
704 let source = chain.state(words[2])?;
705 let target = chain.state(words[3])?;
706 let steps = number(words[4]);
707 if steps == 0 {
708 return Err("BAD STEPS".into());
709 }
710 let mut previous: Vec<Fraction> = (0..size).map(|i| chain.transition[i][target]).collect();
711 let mut cumulative = ZERO;
712 let mut records = Vec::new();
713 for t in 1..=steps {
714 if t > 1 {
715 previous = (0..size).map(|i| (0..size).filter(|&j| j != target).map(|j| chain.transition[i][j] * previous[j]).sum()).collect();
716 }
717 cumulative = cumulative + previous[source];
718 records.push(format!("FP {t} {} cum={}", previous[source].value(), cumulative.value()));
719 }
720 lines.push(format!("FIRSTPASSAGE {name} {} {} k={steps} total={}", words[2], words[3], cumulative.value()));
721 lines.extend(records);
722 }
723 "PATHPROB" => {
724 let path: Vec<usize> = words[2..].iter().map(|name| chain.state(name)).collect::<Result<_, _>>()?;
725 let mut chance = ONE;
726 let mut zero_at = None;
727 for (i, pair) in path.windows(2).enumerate() {
728 chance = chance * chain.transition[pair[0]][pair[1]];
729 if chance == ZERO && zero_at.is_none() {
730 zero_at = Some(i + 1);
731 }
732 }
733 let suffix = zero_at.map_or_else(String::new, |step| format!(" zero_at={step}"));
734 lines.push(format!("PATHPROB {name} steps={} prob={}{suffix}", path.len() - 1, chance.value()));
735 }
736 "MOSTLIKELY" => {
737 let source = chain.state(words[2])?;
738 let steps = number(words[3]);
739 if steps == 0 {
740 return Err("BAD STEPS".into());
741 }
742 let mut best = matrix(steps + 1, size);
743 let mut choice = vec![vec![0; size]; steps];
744 best[steps].fill(ONE);
745 for t in (0..steps).rev() {
746 for i in 0..size {
747 for j in 0..size {
748 let candidate = chain.transition[i][j] * best[t + 1][j];
749 if candidate > best[t][i] {
750 best[t][i] = candidate;
751 choice[t][i] = j;
752 }
753 }
754 }
755 }
756 let mut path = vec![source];
757 let mut current = source;
758 for row in choice {
759 current = row[current];
760 path.push(current);
761 }
762 lines.push(format!("MOSTLIKELY {name} {} k={steps} prob={}", words[2], best[0][source].value()));
763 lines.push(format!("PATH {}", names(&chain, &path, " ")));
764 }
765 "LUMP" => {
766 let mut blocks = Vec::new();
767 let mut seen = vec![false; size];
768 for text in words[3].split('|') {
769 let mut block = Vec::new();
770 for state in text.split(',') {
771 let index = chain.state(state)?;
772 if seen[index] {
773 return Err("BAD PARTITION".into());
774 }
775 seen[index] = true;
776 block.push(index);
777 }
778 block.sort_unstable();
779 blocks.push(block);
780 }
781 if seen.iter().any(|&value| !value) {
782 return Err("BAD PARTITION".into());
783 }
784 blocks.sort_unstable_by_key(|block| block[0]);
785 let mut result = Chain::new(blocks.iter().map(|block| chain.states[block[0]].clone()).collect());
786 for (i, source) in blocks.iter().enumerate() {
787 for (j, target) in blocks.iter().enumerate() {
788 let expected = chain.row_sum(source[0], target);
789 result.transition[i][j] = expected;
790 for &state in source {
791 let actual = chain.row_sum(state, target);
792 if actual != expected {
793 return Ok(vec![format!("LUMP {name} not lumpable block={} target={} {}={} {}={}",
794 chain.states[source[0]], chain.states[target[0]], chain.states[source[0]], expected.value(), chain.states[state], actual.value())]);
795 }
796 }
797 }
798 }
799 lines.push(format!("LUMP {name} {} blocks={}", words[2], blocks.len()));
800 lines.extend(result.display(&result.transition, "ROW"));
801 chains.insert(words[2].to_owned(), result);
802 }
803 "REWARD" => {
804 if words.len() - 3 != size {
805 return Err("BAD LENGTH".into());
806 }
807 let gamma = probability(words[2])?;
808 if gamma == ONE {
809 return Err("BAD DISCOUNT".into());
810 }
811 let rewards: Vec<Fraction> = words[3..].iter().map(|value| Fraction::integer(value.parse().unwrap())).collect();
812 let mut equations = matrix(size, size + 1);
813 for i in 0..size {
814 for j in 0..size {
815 equations[i][j] = (if i == j { ONE } else { ZERO }) - gamma * chain.transition[i][j];
816 }
817 equations[i][size] = rewards[i];
818 }
819 let solved = eliminate(equations);
820 let mut gain = vec![ZERO; size];
821 for class in &graph.classes {
822 if !class.closed {
823 continue;
824 }
825 let distribution = stationary(&chain, &class.states);
826 let average: Fraction = class.states.iter().map(|&i| distribution[i] * rewards[i]).sum();
827 let hit = hitting(&chain, &graph, &class.states, false);
828 for i in 0..size {
829 gain[i] = gain[i] + hit.probabilities[i] * average;
830 }
831 }
832 lines.push(format!("REWARD {name} gamma={gamma}"));
833 for i in 0..size {
834 lines.push(format!("VALUE {} discounted={} gain={}", chain.states[i], solved[i][size].value(), gain[i].value()));
835 }
836 }
837 "SIMULATE" => {
838 let mut current = chain.state(words[2])?;
839 let steps = number(words[3]);
840 if steps == 0 {
841 return Err("BAD STEPS".into());
842 }
843 let seed: i64 = words[4].parse().unwrap();
844 if seed == 0 || seed >= 2_147_483_647 {
845 return Err("BAD SEED".into());
846 }
847 let mut random = seed;
848 let mut path = vec![current];
849 let mut visits = vec![0; size];
850 visits[current] += 1;
851 for _ in 0..steps {
852 let common = chain.transition[current].iter().fold(1, |value, p| value / gcd(value, p.denominator) * p.denominator);
853 random = 48_271 * random % 2_147_483_647;
854 let draw = random % common;
855 let mut cumulative = ZERO;
856 for j in 0..size {
857 cumulative = cumulative + chain.transition[current][j];
858 if cumulative.numerator * (common / cumulative.denominator) > draw {
859 current = j;
860 break;
861 }
862 }
863 path.push(current);
864 visits[current] += 1;
865 }
866 lines.push(format!("SIMULATE {name} {} steps={steps} seed={seed} end={}", words[2], chain.states[current]));
867 lines.push(format!("PATH {}", names(&chain, &path, " ")));
868 for (i, count) in visits.iter().enumerate() {
869 lines.push(format!("VISIT {} {count}", chain.states[i]));
870 }
871 }
872 "PASSAGE" | "REVERSIBLE" | "REVERSE" | "MIXING" => {
873 let mut steps = 0;
874 let mut epsilon = ZERO;
875 if verb == "MIXING" {
876 steps = number(words[2]);
877 if steps == 0 {
878 return Err("BAD STEPS".into());
879 }
880 epsilon = probability(words[3])?;
881 }
882 if graph.classes.len() != 1 {
883 return Err("NOT IRREDUCIBLE".into());
884 }
885 let distribution = stationary(&chain, &all);
886 match verb {
887 "PASSAGE" => {
888 let mut passage = matrix(size, size);
889 for j in 0..size {
890 let hit = hitting(&chain, &graph, &[j], true);
891 for i in 0..size {
892 passage[i][j] = if i == j { ONE / distribution[j] } else { hit.times[i].unwrap() };
893 }
894 }
895 let kemeny: Fraction = (1..size).map(|j| distribution[j] * passage[0][j]).sum();
896 lines.push(format!("PASSAGE {name} kemeny={}", kemeny.value()));
897 lines.extend(chain.display(&passage, "M"));
898 }
899 "REVERSIBLE" => {
900 let mut violations = Vec::new();
901 for i in 0..size {
902 for j in i + 1..size {
903 let forward = distribution[i] * chain.transition[i][j];
904 let backward = distribution[j] * chain.transition[j][i];
905 if forward != backward {
906 violations.push(format!("VIOLATION {} {} {} {}", chain.states[i], chain.states[j], forward.value(), backward.value()));
907 }
908 }
909 }
910 lines.push(if violations.is_empty() { format!("REVERSIBLE {name} yes") } else { format!("REVERSIBLE {name} no violations={}", violations.len()) });
911 lines.extend(violations);
912 }
913 "REVERSE" => {
914 let mut result = Chain::new(chain.states.clone());
915 for i in 0..size {
916 for j in 0..size {
917 result.transition[i][j] = distribution[j] * chain.transition[j][i] / distribution[i];
918 }
919 }
920 let reversible = if result.transition == chain.transition { "yes" } else { "no" };
921 lines.push(format!("REVERSE {name} {} reversible={reversible}", words[2]));
922 lines.extend(result.display(&result.transition, "ROW"));
923 chains.insert(words[2].to_owned(), result);
924 }
925 _ => {
926 let mut power = identity(size);
927 let mut records = Vec::new();
928 let mut mixing = None;
929 for t in 1..=steps {
930 power = matrix_product(&power, &chain.transition);
931 let mut distance = ZERO;
932 let mut worst = 0;
933 for i in 0..size {
934 let total: Fraction = (0..size).map(|j| (power[i][j] - distribution[j]).abs()).sum();
935 let candidate = total / Fraction::integer(2);
936 if candidate > distance {
937 distance = candidate;
938 worst = i;
939 }
940 }
941 if mixing.is_none() && distance <= epsilon {
942 mixing = Some(t);
943 }
944 records.push(format!("TV {t} {} from={}", distance.value(), chain.states[worst]));
945 }
946 let time = mixing.map_or_else(|| "none".into(), |value| value.to_string());
947 lines.push(format!("MIXING {name} k={steps} eps={epsilon} time={time}"));
948 lines.extend(records);
949 }
950 }
951 }
952 _ => unreachable!(),
953 }
954 Ok(lines)
955}
956
957fn main() {
958 let mut input = String::new();
959 io::stdin().read_to_string(&mut input).unwrap();
960 let mut lines = input.lines();
961 let count = number(lines.next().unwrap());
962 let mut chains = HashMap::new();
963 let mut output = io::BufWriter::new(io::stdout().lock());
964 for line in lines.take(count) {
965 let words: Vec<&str> = line.split_whitespace().collect();
966 match execute(&mut chains, &words) {
967 Ok(result) => {
968 for text in result {
969 writeln!(output, "{text}").unwrap();
970 }
971 }
972 Err(problem) => writeln!(output, "ERROR {problem}").unwrap(),
973 }
974 }
975}
