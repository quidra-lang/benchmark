21Ratio ratio_parts(real quantity)
22 text := string(quantity)
23 fields := text.split("/")
24 if #fields == 2
25 return Ratio(parse_integer(fields[0]), parse_integer(fields[1]))
26 fields = text.split(".")
27 if #fields == 1
28 return Ratio(parse_integer(text), 1)
29 denominator := 10 ^ int(#fields[1])
30 numerator := parse_integer(fields[0]) * denominator + (if text.starts_with("-") then -1 else 1) * parse_integer(fields[1])
31 divisor := gcd(numerator, denominator)
32 return Ratio(numerator / divisor, denominator / divisor)

85real[][] solve(real[][] rows)
86 size := int(#rows)
87 for: 0 <= column < size
88 pivot_row := column
89 while rows[pivot_row][column] == 0.0
90 pivot_row += 1
91 saved := rows[column]
92 rows[column] = rows[pivot_row]
93 rows[pivot_row] = saved
94 pivot := rows[column][column]
95 rows[column] = rows[column] / pivot
96 for: 0 <= row < size
97 if row == column
98 continue
99 factor := rows[row][column]
100 if factor != 0.0
101 rows[row] = rows[row] - rows[column] * factor
102 return [for entries := rows[] then entries[size:]]

136class Component(int[] members, bool closed, int[] levels, int period)
137class Graph(bool[][] reachable, Component[] classes, int[] membership)
138
139Graph decompose(const Chain chain)
140 size := int(#chain.names)
141 states := state_numbers(chain)
142 bool[][] reachable := chain.transitions > 0.0
143 for index := states[]
144 reachable[index][index] = true
145 for middle := states[]
146 for source := states[]
147 for target := states[]
148 reachable[source][target] = reachable[source][target] or (reachable[source][middle] and reachable[middle][target])
149 Component[] classes := []
150 membership := array(nat(size), fill = -1)
151 for root := states[]
152 if membership[root] >= 0
153 continue
154 members := [for index := states[] if reachable[root][index] and reachable[index][root] then index]
155 inside := array(nat(size), fill = false)
156 for member := members[]
157 membership[member] = int(#classes)
158 inside[member] = true
159 closed := true
160 for source := members[]
161 for target := states[]
162 if not inside[target] and chain.transitions[source][target] > 0.0
163 closed = false
164 levels := array(nat(size), fill = -1)
165 levels[root] = 0
166 queue := [root]
167 head := 0
168 while head < int(#queue)
169 source := queue[head]
170 head += 1
171 for target := members[]
172 if levels[target] < 0 and chain.transitions[source][target] > 0.0
173 levels[target] = levels[source] + 1
174 queue = queue.append(target)
175 period := 0
176 for source := members[]
177 for target := members[]
178 if chain.transitions[source][target] > 0.0
179 period = gcd(period, levels[source] + 1 - levels[target])
180 classes = classes.append(Component(members, closed, levels, period))
181 return Graph(reachable, classes, membership)

183real[] stationary(const Chain chain, const int[] members)
184 size := int(#members)
185 rows := matrix(size, size + 1)
186 for: 0 <= row < size - 1
187 for: 0 <= column < size
188 rows[row][column] = chain.transitions[members[column]][members[row]] - unit_value(column, row)
189 rows[size - 1] = array(nat(size + 1), fill = real(1))
190 return [for entries := solve(rows)[] then entries[0]]

192real[] | error irreducible(const Chain chain, const Graph graph)
193 if #graph.classes != 1
194 return error("NOT IRREDUCIBLE")
195 return stationary(chain, graph.classes[0].members)

197class HittingResult(real[] chances, real[] durations)
198
199HittingResult hitting(const Chain chain, const int[] targets, const bool[][] reachable, with_times := true)
200 size := int(#chain.names)
201 chances := array(nat(size), fill = real(0))
202 durations := array(nat(size), fill = real(0))
203 inside := array(nat(size), fill = false)
204 for target := targets[]
205 chances[target] = 1.0
206 inside[target] = true
207 int[] unknown := []
208 for index := state_numbers(chain)[]
209 reaches := false
210 for target := targets[]
211 reaches = reaches or reachable[index][target]
212 if not inside[index] and reaches
213 unknown = unknown.append(index)
214 count := int(#unknown)
215 if count > 0
216 rows := matrix(count, count + 1)
217 for: 0 <= row < count
218 for: 0 <= column < count
219 rows[row][column] = unit_value(row, column) - chain.transitions[unknown[row]][unknown[column]]
220 rows[row][count] = block_sum(chain, unknown[row], targets)
221 solution := solve(rows)
222 for: 0 <= index < count
223 chances[unknown[index]] = solution[index][0]
224 if with_times
225 certain := [for index := unknown[] if chances[index] == 1.0 then index]
226 count_certain := int(#certain)
227 if count_certain > 0
228 rows := matrix(count_certain, count_certain + 1)
229 for: 0 <= row < count_certain
230 for: 0 <= column < count_certain
231 rows[row][column] = unit_value(row, column) - chain.transitions[certain[row]][certain[column]]
232 rows[row][count_certain] = 1.0
233 solution := solve(rows)
234 for: 0 <= index < count_certain
235 durations[certain[index]] = solution[index][0]
236 return HittingResult(chances, durations)

241class AbsorptionResult(int[] transient, int[] absorbing, real[][] fundamental, real[] means, real[] variances, real[][] chances)
242
243AbsorptionResult absorption(const Chain chain)
244 absorbing := absorbing_states(chain)
245 transient := [for index := state_numbers(chain)[] if chain.transitions[index][index] != 1.0 then index]
246 count := int(#transient)
247 rows := matrix(count, 2 * count)
248 for: 0 <= row < count
249 for: 0 <= column < count
250 rows[row][column] = unit_value(row, column) - chain.transitions[transient[row]][transient[column]]
251 rows[row][count + row] = 1.0
252 fundamental := solve(rows)
253 means := [for entries := fundamental[] then ordered_sum(entries)]
254 real[] variances := []
255 chances := matrix(count, int(#absorbing))
256 for: 0 <= row < count
257 real[] terms := []
258 for: 0 <= column < count
259 terms = terms.append((2.0 * fundamental[row][column] - unit_value(row, column)) * means[column])
260 variances = variances.append(ordered_sum(terms) - means[row] * means[row])
261 for column := absorbing.indices()[]
262 arrivals := [for index := transient[] then chain.transitions[index][absorbing[column]]]
263 chances[row][column] = ordered_sum(fundamental[row] * arrivals)
264 return AbsorptionResult(transient, absorbing, fundamental, means, variances, chances)

266int regular_exponent(const Chain chain)
267 size := int(#chain.names)
268 states := state_numbers(chain)
269 bool[][] edges := chain.transitions > 0.0
270 power := edges
271 bound := (size - 1) * (size - 1) + 1
272 for: 1 <= exponent <= bound
273 positive := true
274 for entries := power[]
275 for edge := entries[]
276 positive = positive and edge
277 if positive
278 return exponent
279 if exponent == bound
280 break
281 bool[][] next := array(nat(size), fill = array(nat(size), fill = false))
282 for source := states[]
283 for target := states[]
284 for middle := states[]
285 next[source][target] = next[source][target] or (power[source][middle] and edges[middle][target])
286 power = next
287 return 0

289void | error create(map<string, Chain> &chains, const string[] words)
290 op := words[0]
291 name := words[1]
292 if op == "CHAIN"
293 names := words[2:]
294 if #names > 10
295 return error("TOO MANY STATES")
296 distinct<string> seen
297 for label := names[]
298 if seen.has(label)
299 return error("DUPLICATE STATE " + label)

302 emit("CHAIN {name} states={#names}")
303 elif op == "BUILD"
304 order := parse_integer(words[2])
305 if order != 1 and order != 2
306 return error("BAD ORDER")
307 tokens := words[3:]
308 if int(#tokens) <= order
309 return error("TOO FEW TOKENS")
310 observations := tokens
311 if order == 2
312 observations = []
313 for: 0 <= index < int(#tokens) - 1
314 observations = observations.append(tokens[index] + "." + tokens[index + 1])
315 string[] names := []
316 map<string, int> positions
317 int[] sequence := []
318 for observation := observations[]
319 match positions.get(observation)
320 int index
321 sequence = sequence.append(index)
322 none
323 position := int(#names)
324 .positions.set(observation, position)
325 sequence = sequence.append(position)
326 names = names.append(observation)
327 size := int(#names)
328 if size > 10
329 return error("TOO MANY STATES")
330 int[][] counts := array(nat(size), fill = array(nat(size), fill = 0))
331 for: 1 <= index < int(#sequence)
332 counts[sequence[index - 1]][sequence[index]] += 1
333 transitions := matrix(size, size)
334 totals := [for entries := counts[] then entries.sum()]
335 for: 0 <= row < size
336 if totals[row] == 0
337 transitions[row][row] = 1.0
338 else
339 transitions[row] = real(counts[row]) / real(totals[row])
340 .chains.set(name, Chain(names, transitions))
341 emit("BUILD {name} order={order} states={size} transitions={int(#observations) - 1}")
342 for: 0 <= row < size
343 detail := "absorbing"
344 if totals[row] > 0
345 string[] entries := []
346 for: 0 <= column < size
347 if counts[row][column] > 0
348 entries = entries.append("{names[column]}:{counts[row][column]}")
349 detail = entries.join(" ")
350 emit("FROM {names[row]} total={totals[row]} {detail}")
351 else
352 goal := parse_integer(words[2])
353 start := parse_integer(words[3])
354 if goal == 0 or goal > 9
355 return error("BAD SIZE")
356 if start > goal
357 return error("BAD START")
358 chance := try probability(words[4])
359 if chance == 0.0 or chance == 1.0
360 return error("BAD PROBABILITY")
361 string[] names := []
362 for: 0 <= capital <= goal
363 names = names.append("s{capital}")
364 transitions := matrix(goal + 1, goal + 1)
365 transitions[0][0] = 1.0
366 transitions[goal][goal] = 1.0
367 for: 1 <= capital < goal
368 transitions[capital][capital + 1] = chance
369 transitions[capital][capital - 1] = 1.0 - chance
370 chain := Chain(names, transitions)
371 result := absorption(chain)
372 ruin := array(nat(goal + 1), fill = real(0))
373 duration := array(nat(goal + 1), fill = real(0))
374 ruin[0] = 1.0
375 for: 1 <= capital < goal
376 ruin[capital] = result.chances[capital - 1][0]
377 duration[capital] = result.means[capital - 1]
378 .chains.set(name, chain)
379 emit("GAMBLER {name} g={goal} p={fraction_text(chance)} start=s{start} ruin={value_text(ruin[start])} duration={value_text(duration[start])}")
380 for: 0 <= capital <= goal
381 emit("CAPITAL s{capital} ruin={value_text(ruin[capital])} win={value_text(1.0 - ruin[capital])} duration={value_text(duration[capital])}")
382 return void

386 name := words[1]

389 Chain chain
390 match chains.get(name)
391 Chain found
392 chain = found
393 none
394 return error("UNKNOWN CHAIN " + name)

399 target := try state_index(chain, words[3])

425 for index := states[]

429 if op == "CLASSES"
430 emit("CLASSES {name} count={#graph.classes}")
431 for index := graph.classes.indices()[]
432 group := graph.classes[index]
433 kind := if group.closed then "RECURRENT" else "TRANSIENT"
434 emit("CLASS {index + 1} {kind} period={group.period} size={#group.members} " + labels(chain, group.members))
435 if group.period >= 2
436 for: 0 <= residue < group.period
437 selected := [for member := group.members[] if group.levels[member] % group.period == residue then member]
438 emit("CYCLIC {residue} " + labels(chain, selected))
439 elif op == "CLASSIFY"
440 absorbing := absorbing_states(chain)
441 absorbs := #absorbing > 0
442 for source := states[]
443 reaches := false
444 for target := absorbing[]
445 reaches = reaches or graph.reachable[source][target]
446 absorbs = absorbs and reaches

448 exponent_text := if exponent == 0 then "-" else string(exponent)
449 emit("CLASSIFY {name} absorbing_states={#absorbing} absorbing={yes(absorbs)} ergodic={yes(#graph.classes == 1)} regular={yes(exponent > 0)} exponent={exponent_text}")
450 elif op == "STEP"
451 initial := words[2]
452 steps := parse_integer(words[3])
453 distribution := array(nat(size), fill = real(0))
454 if "0123456789".contains(initial[0])
455 entries := initial.split(",")
456 if int(#entries) != size
457 return error("BAD LENGTH")
458 distribution = try probability_row(entries)
459 if ordered_sum(distribution) != 1.0
460 return error("BAD DISTRIBUTION")
461 else
462 source := try state_index(chain, initial)
463 distribution[source] = 1.0
464 for: 0 <= iteration < steps
465 distribution = multiply([distribution], chain.transitions)[0]

467 for index := states[]
468 emit("PROB {chain.names[index]} {value_text(distribution[index])}")
469 elif op == "POWER"
470 steps := parse_integer(words[2])

472 for: 0 <= iteration < steps
473 power = multiply(power, chain.transitions)

475 display(chain.names, power)
476 elif op == "STATIONARY"

487 elif op == "RECURRENCE"

491 return error("TRANSIENT STATE " + words[2])

496 emit("RECURRENCE {name} {words[2]} pi={value_text(rho)} mean={value_text(1.0 / rho)} period={group.period}")

503 for target := absorbing[]

528 elif op == "PASSAGE"

538 elif op == "FIRSTPASSAGE"

540 target := try state_index(chain, words[3])

587 path = path.append(target)

590 emit("PATH " + labels(chain, path))
591 elif op == "REVERSIBLE" or op == "REVERSE"
592 pi := try irreducible(chain, graph)
593 if op == "REVERSIBLE"
594 string[] violations := []

596 for: source + 1 <= target < size
597 forward := pi[source] * chain.transitions[source][target]
598 backward := pi[target] * chain.transitions[target][source]

604 else

606 for source := states[]
607 for target := states[]
608 reversed[source][target] = pi[target] * chain.transitions[target][source] / pi[source]
609 .chains.set(words[2], Chain(chain.names, reversed))

611 display(chain.names, reversed)
612 elif op == "MIXING"
613 steps := parse_integer(words[2])
614 if steps == 0
615 return error("BAD STEPS")

617 pi := try irreducible(chain, graph)
618 power := identity(size)
619 real[] distances := []
620 int[] worst := []
