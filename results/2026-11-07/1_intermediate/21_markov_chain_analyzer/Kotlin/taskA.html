8private fun gcd(left: Long, right: Long): Long {
9 var a = abs(left)
10 var b = abs(right)
11 while (b != 0L) {
12 val remainder = a % b
13 a = b
14 b = remainder
15 }
16 return a
17}

19private data class Fraction private constructor(val numerator: Long, val denominator: Long) : Comparable<Fraction> {
20 companion object {
21 fun of(numerator: Long, denominator: Long = 1): Fraction {
22 val sign = if (denominator < 0) -1 else 1
23 val divisor = gcd(numerator, denominator)
24 return Fraction(sign * numerator / divisor, sign * denominator / divisor)
25 }
26 }
27
28 operator fun plus(other: Fraction): Fraction = of(
29 numerator * other.denominator + other.numerator * denominator,
30 denominator * other.denominator
31 )
32 operator fun minus(other: Fraction): Fraction = of(
33 numerator * other.denominator - other.numerator * denominator,
34 denominator * other.denominator
35 )
36 operator fun times(other: Fraction): Fraction = of(numerator * other.numerator, denominator * other.denominator)
37 operator fun div(other: Fraction): Fraction = of(numerator * other.denominator, denominator * other.numerator)
38 override fun compareTo(other: Fraction): Int =
39 (numerator * other.denominator).compareTo(other.numerator * denominator)
40 fun abs(): Fraction = of(abs(numerator), denominator)
41 override fun toString(): String = "$numerator/$denominator"
42
43 fun value(): String {
44 val decimal = numerator % denominator * 1_000_000 / denominator
45 return "$this(${numerator / denominator}.${decimal.toString().padStart(6, '0')})"
46 }
47}

49private val ZERO = Fraction.of(0)
50private val ONE = Fraction.of(1)
51private val TWO = Fraction.of(2)
52private typealias Matrix = Array<Array<Fraction>>

54private fun probability(text: String): Fraction {
55 val parts = text.split('/')
56 val numerator = parts[0].toLong()
57 val denominator = parts.getOrNull(1)?.toLong() ?: 1
58 if (denominator == 0L || numerator > denominator) fail("BAD PROBABILITY")
59 return Fraction.of(numerator, denominator)
60}
61
62private fun vector(size: Int): Array<Fraction> = Array(size) { ZERO }
63private fun matrix(rows: Int, columns: Int): Matrix = Array(rows) { vector(columns) }
64private fun Array<Fraction>.total(): Fraction = fold(ZERO) { sum, value -> sum + value }
65private fun Array<Fraction>.text(): String = joinToString(" ")
66
67private fun identity(size: Int): Matrix = matrix(size, size).also { result ->
68 for (i in 0 until size) result[i][i] = ONE
69}
70
71private fun product(left: Matrix, right: Matrix): Matrix = Array(left.size) { i ->
72 Array(right[0].size) { j -> right.indices.fold(ZERO) { sum, k -> sum + left[i][k] * right[k][j] } }
73}
74
75private fun eliminate(equations: Matrix): Matrix {
76 for (column in equations.indices) {
77 var pivotRow = column
78 while (equations[pivotRow][column] == ZERO) pivotRow++
79 val swapped = equations[column]
80 equations[column] = equations[pivotRow]
81 equations[pivotRow] = swapped
82 val pivot = equations[column][column]
83 for (j in equations[column].indices) equations[column][j] = equations[column][j] / pivot
84 for (row in equations.indices) {
85 if (row == column || equations[row][column] == ZERO) continue
86 val factor = equations[row][column]
87 for (j in equations[row].indices) {
88 equations[row][j] = equations[row][j] - factor * equations[column][j]
89 }
90 }
91 }
92 return equations
93}

95private class Chain(states: List<String>) {
96 val states = states.toList()
97 val size: Int get() = states.size
98 val transition = matrix(size, size)
99
100 fun state(name: String): Int {
101 val index = states.indexOf(name)
102 if (index < 0) fail("UNKNOWN STATE $name")
103 return index
104 }
105
106 fun rowSum(row: Int, selected: List<Int>): Fraction =
107 selected.fold(ZERO) { sum, column -> sum + transition[row][column] }
108
109 fun targets(names: List<String>): List<Int> {
110 val selected = mutableSetOf<Int>()
111 for (name in names) {
112 val index = state(name)
113 if (!selected.add(index)) fail("DUPLICATE STATE $name")
114 }
115 return selected.sorted()
116 }
117
118 fun absorbing(): List<Int> = states.indices.filter { transition[it][it] == ONE }
119 fun names(indices: List<Int>, separator: String = " "): String = indices.joinToString(separator) { states[it] }
120
121 fun display(lines: MutableList<String>, values: Matrix = transition, label: String = "ROW") {
122 lines.add("STATES ${states.joinToString(" ")}")
123 for (i in states.indices) lines.add("$label ${states[i]} ${values[i].text()}")
124 }
125}

127private data class CommunicatingClass(val states: List<Int>, val closed: Boolean, val period: Int)
128
129private class Graph(chain: Chain) {
130 val reachable = Array(chain.size) { i -> BooleanArray(chain.size) { j -> i == j || chain.transition[i][j] != ZERO } }
131 val classes = mutableListOf<CommunicatingClass>()
132 val classOf = IntArray(chain.size) { -1 }
133 val levels = IntArray(chain.size)
134
135 init {
136 val size = chain.size
137 for (k in 0 until size) {
138 for (i in 0 until size) {
139 for (j in 0 until size) reachable[i][j] = reachable[i][j] || reachable[i][k] && reachable[k][j]
140 }
141 }
142 for (root in 0 until size) {
143 if (classOf[root] >= 0) continue
144 val states = (0 until size).filter { reachable[root][it] && reachable[it][root] }
145 for (state in states) classOf[state] = classes.size
146 var closed = true
147 val visited = BooleanArray(size)
148 val queue = ArrayDeque<Int>()
149 queue.addLast(root)
150 visited[root] = true
151 while (queue.isNotEmpty()) {
152 val current = queue.removeFirst()
153 for (next in 0 until size) {
154 if (chain.transition[current][next] == ZERO) continue
155 if (next !in states) {
156 closed = false
157 } else if (!visited[next]) {
158 visited[next] = true
159 levels[next] = levels[current] + 1
160 queue.addLast(next)
161 }
162 }
163 }
164 var period = 0L
165 for (i in states) {
166 for (j in states) {
167 if (chain.transition[i][j] != ZERO) period = gcd(period, levels[i] + 1L - levels[j])
168 }
169 }
170 classes.add(CommunicatingClass(states, closed, period.toInt()))
171 }
172 }
173
174 fun reaches(source: Int, targets: List<Int>): Boolean = targets.any { reachable[source][it] }
175 fun requireIrreducible() {
176 if (classes.size != 1) fail("NOT IRREDUCIBLE")
177 }
178}

180private fun stationary(chain: Chain, states: List<Int>): Array<Fraction> {
181 val count = states.size
182 val equations = matrix(count, count + 1)
183 for (i in 0 until count - 1) {
184 for (j in 0 until count) equations[i][j] = chain.transition[states[j]][states[i]] - if (i == j) ONE else ZERO
185 }
186 equations[count - 1].fill(ONE)
187 eliminate(equations)
188 return vector(chain.size).also { distribution ->
189 for (i in states.indices) distribution[states[i]] = equations[i][count]
190 }
191}

193private data class Hitting(val probabilities: Array<Fraction>, val times: Array<Fraction?>)
194
195private fun hitting(chain: Chain, graph: Graph, targets: List<Int>, withTimes: Boolean): Hitting {
196 val probabilities = vector(chain.size)
197 val times = arrayOfNulls<Fraction>(chain.size)
198 for (target in targets) {
199 probabilities[target] = ONE
200 times[target] = ZERO
201 }
202 val unknown = chain.states.indices.filter { it !in targets && graph.reaches(it, targets) }
203 if (unknown.isNotEmpty()) {
204 val count = unknown.size
205 val equations = matrix(count, count + 1)
206 for (i in unknown.indices) {
207 for (j in unknown.indices) equations[i][j] = (if (i == j) ONE else ZERO) - chain.transition[unknown[i]][unknown[j]]
208 equations[i][count] = chain.rowSum(unknown[i], targets)
209 }
210 eliminate(equations)
211 for (i in unknown.indices) probabilities[unknown[i]] = equations[i][count]
212 }
213 if (withTimes) {
214 val certain = unknown.filter { probabilities[it] == ONE }
215 if (certain.isNotEmpty()) {
216 val count = certain.size
217 val equations = matrix(count, count + 1)
218 for (i in certain.indices) {
219 for (j in certain.indices) equations[i][j] = (if (i == j) ONE else ZERO) - chain.transition[certain[i]][certain[j]]
220 equations[i][count] = ONE
221 }
222 eliminate(equations)
223 for (i in certain.indices) times[certain[i]] = equations[i][count]
224 }
225 }
226 return Hitting(probabilities, times)
227}

229private data class Absorption(
230 val transient: List<Int>, val absorbing: List<Int>, val fundamental: Matrix,
231 val means: Array<Fraction>, val variances: Array<Fraction>, val probabilities: Matrix
232)
233
234private fun absorption(chain: Chain): Absorption {
235 val absorbing = chain.absorbing()
236 val transient = chain.states.indices.filter { it !in absorbing }
237 val count = transient.size
238 val equations = matrix(count, 2 * count)
239 for (i in transient.indices) {
240 for (j in transient.indices) equations[i][j] = (if (i == j) ONE else ZERO) - chain.transition[transient[i]][transient[j]]
241 equations[i][count + i] = ONE
242 }
243 eliminate(equations)
244 val fundamental = Array(count) { equations[it].copyOfRange(count, 2 * count) }
245 val means = Array(count) { fundamental[it].total() }
246 val variances = vector(count)
247 val probabilities = matrix(count, absorbing.size)
248 for (i in transient.indices) {
249 var second = ZERO
250 for (j in transient.indices) {
251 second += (TWO * fundamental[i][j] - if (i == j) ONE else ZERO) * means[j]
252 }
253 variances[i] = second - means[i] * means[i]
254 for (j in absorbing.indices) {
255 for (k in transient.indices) probabilities[i][j] = probabilities[i][j] + fundamental[i][k] * chain.transition[transient[k]][absorbing[j]]
256 }
257 }
258 return Absorption(transient, absorbing, fundamental, means, variances, probabilities)
259}

261private fun build(words: List<String>, lines: MutableList<String>): Chain {
262 val order = words[2].toInt()
263 val tokens = words.drop(3)
264 if (order != 1 && order != 2) fail("BAD ORDER")
265 if (tokens.size <= order) fail("TOO FEW TOKENS")
266 val states = mutableListOf<String>()
267 val observations = mutableListOf<Int>()
268 for (i in 0 until tokens.size - order + 1) {
269 val name = if (order == 1) tokens[i] else "${tokens[i]}.${tokens[i + 1]}"
270 var index = states.indexOf(name)
271 if (index < 0) {
272 if (states.size == 10) fail("TOO MANY STATES")
273 index = states.size
274 states.add(name)
275 }
276 observations.add(index)
277 }
278 val chain = Chain(states)
279 val counts = Array(chain.size) { IntArray(chain.size) }
280 val totals = IntArray(chain.size)
281 for (i in 0 until observations.size - 1) {
282 counts[observations[i]][observations[i + 1]]++
283 totals[observations[i]]++
284 }
285 lines.add("BUILD ${words[1]} order=$order states=${chain.size} transitions=${observations.size - 1}")
286 for (i in states.indices) {
287 val line = StringBuilder("FROM ${states[i]} total=${totals[i]}")
288 if (totals[i] == 0) {
289 chain.transition[i][i] = ONE
290 line.append(" absorbing")
291 } else {
292 for (j in states.indices) {
293 chain.transition[i][j] = Fraction.of(counts[i][j].toLong(), totals[i].toLong())
294 if (counts[i][j] > 0) line.append(" ${states[j]}:${counts[i][j]}")
295 }
296 }
297 lines.add(line.toString())
298 }
299 return chain
300}

302private fun gambler(words: List<String>, lines: MutableList<String>): Chain {
303 val goal = words[2].toInt()
304 val start = words[3].toInt()
305 if (goal == 0 || goal > 9) fail("BAD SIZE")
306 if (start > goal) fail("BAD START")
307 val chance = probability(words[4])
308 if (chance == ZERO || chance == ONE) fail("BAD PROBABILITY")
309 val chain = Chain((0..goal).map { "s$it" })
310 chain.transition[0][0] = ONE
311 chain.transition[goal][goal] = ONE
312 for (i in 1 until goal) {
313 chain.transition[i][i + 1] = chance
314 chain.transition[i][i - 1] = ONE - chance
315 }
316 val result = absorption(chain)
317 val ruin = vector(goal + 1)
318 val duration = vector(goal + 1)
319 ruin[0] = ONE
320 for (i in result.transient.indices) {
321 ruin[result.transient[i]] = result.probabilities[i][0]
322 duration[result.transient[i]] = result.means[i]
323 }
324 lines.add("GAMBLER ${words[1]} g=$goal p=$chance start=s$start ruin=${ruin[start].value()} duration=${duration[start].value()}")
325 for (i in 0..goal) {
326 lines.add("CAPITAL s$i ruin=${ruin[i].value()} win=${(ONE - ruin[i]).value()} duration=${duration[i].value()}")
327 }
328 return chain
329}

331private fun execute(chains: MutableMap<String, Chain>, words: List<String>): List<String> {
332 val command = words[0]
333 val name = words[1]
334 val lines = mutableListOf<String>()
335 when (command) {
336 "CHAIN" -> {
337 val states = words.drop(2)
338 if (states.size > 10) fail("TOO MANY STATES")
339 val seen = mutableSetOf<String>()
340 for (state in states) if (!seen.add(state)) fail("DUPLICATE STATE $state")
341 chains[name] = Chain(states)
342 lines.add("CHAIN $name states=${states.size}")
343 return lines
344 }
345 "BUILD" -> {
346 chains[name] = build(words, lines)
347 return lines
348 }
349 "GAMBLER" -> {
350 chains[name] = gambler(words, lines)
351 return lines
352 }
353 }
354 val chain = chains[name] ?: fail("UNKNOWN CHAIN $name")
355 val size = chain.size
356 val transition = chain.transition
357 when (command) {
358 "P" -> {
359 val from = chain.state(words[2])
360 val to = chain.state(words[3])
361 val value = probability(words[4])
362 transition[from][to] = value
363 lines.add("P $name ${words[2]} ${words[3]} $value rowsum=${transition[from].total()}")
364 return lines
365 }
366 "ROW" -> {
367 val from = chain.state(words[2])
368 if (words.size - 3 != size) fail("BAD LENGTH")
369 val row = Array(size) { probability(words[3 + it]) }
370 transition[from] = row
371 lines.add("ROW $name ${words[2]} rowsum=${row.total()}")
372 return lines
373 }
374 "SHOW" -> {
375 lines.add("MATRIX $name states=$size")
376 chain.display(lines)
377 return lines
378 }
379 "CHECK" -> {
380 val badRows = mutableListOf<String>()
381 for (i in 0 until size) {
382 val total = transition[i].total()
383 if (total != ONE) badRows.add("BADROW ${chain.states[i]} sum=$total")
384 }
385 lines.add("CHECK $name ${if (badRows.isEmpty()) "ok" else "bad=${badRows.size}"}")
386 lines.addAll(badRows)
387 return lines
388 }
389 }
390 for (i in 0 until size) if (transition[i].total() != ONE) fail("NOT STOCHASTIC ${chain.states[i]}")
391 val graph = Graph(chain)
392 when (command) {
393 "CLASSES" -> {
394 lines.add("CLASSES $name count=${graph.classes.size}")
395 for ((index, group) in graph.classes.withIndex()) {
396 val kind = if (group.closed) "RECURRENT" else "TRANSIENT"
397 lines.add("CLASS ${index + 1} $kind period=${group.period} size=${group.states.size} ${chain.names(group.states)}")
398 if (group.period >= 2) {
399 for (remainder in 0 until group.period) {
400 val cyclic = group.states.filter { graph.levels[it] % group.period == remainder }
401 lines.add("CYCLIC $remainder ${chain.names(cyclic)}")
402 }
403 }
404 }
405 }
406 "CLASSIFY" -> {
407 val absorbing = chain.absorbing()
408 val isAbsorbing = absorbing.isNotEmpty() && (0 until size).all { graph.reaches(it, absorbing) }
409 val edges = Array(size) { i -> BooleanArray(size) { j -> transition[i][j] != ZERO } }
410 var power = edges
411 var exponent = 0
412 val limit = (size - 1) * (size - 1) + 1
413 for (step in 1..limit) {
414 if (power.all { row -> row.all { it } }) {
415 exponent = step
416 break
417 }
418 if (step == limit) break
419 power = Array(size) { i -> BooleanArray(size) { j -> (0 until size).any { k -> power[i][k] && edges[k][j] } } }
420 }
421 val exponentText = if (exponent == 0) "-" else exponent.toString()
422 lines.add("CLASSIFY $name absorbing_states=${absorbing.size} absorbing=${yes(isAbsorbing)} " +
423 "ergodic=${yes(graph.classes.size == 1)} regular=${yes(exponent != 0)} exponent=$exponentText")
424 }
425 "STEP" -> {
426 var distribution = vector(size)
427 if (words[2][0].isDigit()) {
428 val values = words[2].split(',')
429 if (values.size != size) fail("BAD LENGTH")
430 for (i in 0 until size) distribution[i] = probability(values[i])
431 if (distribution.total() != ONE) fail("BAD DISTRIBUTION")
432 } else {
433 distribution[chain.state(words[2])] = ONE
434 }
435 val steps = words[3].toInt()
436 repeat(steps) {
437 val next = vector(size)
438 for (j in 0 until size) {
439 for (i in 0 until size) next[j] = next[j] + distribution[i] * transition[i][j]
440 }
441 distribution = next
442 }
443 lines.add("STEP $name k=$steps")
444 for (i in 0 until size) lines.add("PROB ${chain.states[i]} ${distribution[i].value()}")
445 }
446 "POWER" -> {
447 val steps = words[2].toInt()
448 var power = identity(size)
449 repeat(steps) { power = product(power, transition) }
450 lines.add("POWER $name k=$steps")
451 chain.display(lines, power)
452 }
453 "STATIONARY" -> {
454 lines.add("STATIONARY $name count=${graph.classes.count { it.closed }}")
455 var distribution = 0
456 for ((index, group) in graph.classes.withIndex()) {
457 if (!group.closed) continue
458 val pi = stationary(chain, group.states)
459 lines.add("DIST ${++distribution} class=${index + 1} size=${group.states.size}")
460 for (state in group.states) lines.add("PI ${chain.states[state]} ${pi[state].value()}")
461 }
462 }
463 "RECURRENCE" -> {
464 val state = chain.state(words[2])
465 val group = graph.classes[graph.classOf[state]]
466 if (!group.closed) fail("TRANSIENT STATE ${words[2]}")
467 val rho = stationary(chain, group.states)[state]
468 lines.add("RECURRENCE $name ${words[2]} pi=${rho.value()} mean=${(ONE / rho).value()} period=${group.period}")
469 }
470 "ABSORB" -> {
471 val absorbing = chain.absorbing()
472 if (absorbing.isEmpty()) fail("NO ABSORBING STATE")
473 for (i in 0 until size) if (!graph.reaches(i, absorbing)) fail("NOT ABSORBING ${chain.states[i]}")
474 val result = absorption(chain)
475 val count = result.transient.size
476 lines.add("ABSORB $name transient=$count absorbing=${absorbing.size}")
477 if (count == 0) return lines
478 lines.add("TRANSIENT ${chain.names(result.transient)}")
479 lines.add("ABSORBING ${chain.names(absorbing)}")
480 for (i in 0 until count) lines.add("N ${chain.states[result.transient[i]]} ${result.fundamental[i].text()}")
481 for (i in 0 until count) {
482 lines.add("STEPS ${chain.states[result.transient[i]]} mean=${result.means[i].value()} var=${result.variances[i].value()}")
483 }
484 for (i in 0 until count) lines.add("B ${chain.states[result.transient[i]]} ${result.probabilities[i].text()}")
485 }
486 "HITTING" -> {
487 val targets = chain.targets(words.drop(2))
488 val result = hitting(chain, graph, targets, true)
489 lines.add("HITTING $name targets=${chain.names(targets, ",")}")
490 for (i in 0 until size) {
491 lines.add("HIT ${chain.states[i]} prob=${result.probabilities[i].value()} time=${result.times[i]?.value() ?: "inf"}")
492 }
493 }
494 "PASSAGE" -> {
495 graph.requireIrreducible()
496 val pi = stationary(chain, graph.classes.first().states)
497 val passage = matrix(size, size)
498 for (j in 0 until size) {
499 val times = hitting(chain, graph, listOf(j), true).times
500 for (i in 0 until size) passage[i][j] = if (i == j) ONE / pi[j] else times[i]!!
501 }
502 val kemeny = (1 until size).fold(ZERO) { sum, j -> sum + pi[j] * passage[0][j] }
503 lines.add("PASSAGE $name kemeny=${kemeny.value()}")
504 chain.display(lines, passage, "M")
505 }
506 "FIRSTPASSAGE" -> {
507 val from = chain.state(words[2])
508 val target = chain.state(words[3])
509 val steps = words[4].toInt()
510 if (steps == 0) fail("BAD STEPS")
511 var current = Array(size) { transition[it][target] }
512 var cumulative = ZERO
513 val records = mutableListOf<String>()
514 for (step in 1..steps) {
515 cumulative += current[from]
516 records.add("FP $step ${current[from].value()} cum=${cumulative.value()}")
517 if (step == steps) break
518 val next = vector(size)
519 for (i in 0 until size) {
520 for (j in 0 until size) if (j != target) next[i] = next[i] + transition[i][j] * current[j]
521 }
522 current = next
523 }
524 lines.add("FIRSTPASSAGE $name ${words[2]} ${words[3]} k=$steps total=${cumulative.value()}")
525 lines.addAll(records)
526 }
527 "PATHPROB" -> {
528 val path = words.drop(2).map(chain::state)
529 var chance = ONE
530 var zeroAt = 0
531 for (i in 1 until path.size) {
532 chance *= transition[path[i - 1]][path[i]]
533 if (zeroAt == 0 && chance == ZERO) zeroAt = i
534 }
535 val suffix = if (zeroAt == 0) "" else " zero_at=$zeroAt"
536 lines.add("PATHPROB $name steps=${path.size - 1} prob=${chance.value()}$suffix")
537 }
538 "MOSTLIKELY" -> {
539 val start = chain.state(words[2])
540 val steps = words[3].toInt()
541 if (steps == 0) fail("BAD STEPS")
542 val best = matrix(steps + 1, size)
543 best[steps].fill(ONE)
544 val choices = Array(steps) { IntArray(size) }
545 for (step in steps - 1 downTo 0) {
546 for (i in 0 until size) {
547 for (j in 0 until size) {
548 val candidate = transition[i][j] * best[step + 1][j]
549 if (candidate > best[step][i]) {
550 best[step][i] = candidate
551 choices[step][i] = j
552 }
553 }
554 }
555 }
556 var current = start
557 val path = mutableListOf(current)
558 for (step in 0 until steps) {
559 current = choices[step][current]
560 path.add(current)
561 }
562 lines.add("MOSTLIKELY $name ${words[2]} k=$steps prob=${best[0][start].value()}")
563 lines.add("PATH ${chain.names(path)}")
564 }
565 "REVERSIBLE" -> {
566 graph.requireIrreducible()
567 val pi = stationary(chain, graph.classes.first().states)
568 val violations = mutableListOf<String>()
569 for (i in 0 until size) {
570 for (j in i + 1 until size) {
571 val forward = pi[i] * transition[i][j]
572 val backward = pi[j] * transition[j][i]
573 if (forward != backward) {
574 violations.add("VIOLATION ${chain.states[i]} ${chain.states[j]} ${forward.value()} ${backward.value()}")
575 }
576 }
577 }
578 lines.add("REVERSIBLE $name ${if (violations.isEmpty()) "yes" else "no violations=${violations.size}"}")
579 lines.addAll(violations)
580 }
581 "REVERSE" -> {
582 graph.requireIrreducible()
583 val pi = stationary(chain, graph.classes.first().states)
584 val reversed = Chain(chain.states)
585 var reversible = true
586 for (i in 0 until size) {
587 for (j in 0 until size) {
588 reversed.transition[i][j] = pi[j] * transition[j][i] / pi[i]
589 if (reversed.transition[i][j] != transition[i][j]) reversible = false
590 }
591 }
592 chains[words[2]] = reversed
593 lines.add("REVERSE $name ${words[2]} reversible=${yes(reversible)}")
594 reversed.display(lines)
595 }
596 "MIXING" -> {
597 val steps = words[2].toInt()
598 if (steps == 0) fail("BAD STEPS")
599 val epsilon = probability(words[3])
600 graph.requireIrreducible()
601 val pi = stationary(chain, graph.classes.first().states)
602 var power = identity(size)
603 var mixingTime = 0
604 val records = mutableListOf<String>()
605 for (step in 1..steps) {
606 power = product(power, transition)
607 var maximum = ZERO
608 var worst = 0
609 for (i in 0 until size) {
610 val distance = (0 until size).fold(ZERO) { sum, j -> sum + (power[i][j] - pi[j]).abs() } / TWO
611 if (distance > maximum) {
612 maximum = distance
613 worst = i
614 }
615 }
616 if (mixingTime == 0 && maximum <= epsilon) mixingTime = step
617 records.add("TV $step ${maximum.value()} from=${chain.states[worst]}")
618 }
619 val timeText = if (mixingTime == 0) "none" else mixingTime.toString()
620 lines.add("MIXING $name k=$steps eps=$epsilon time=$timeText")
621 lines.addAll(records)
622 }
623 "LUMP" -> {
624 val blocks = mutableListOf<List<Int>>()
625 val seen = BooleanArray(size)
626 for (group in words[3].split('|')) {
627 val block = mutableListOf<Int>()
628 for (stateName in group.split(',')) {
629 val state = chain.state(stateName)
630 if (seen[state]) fail("BAD PARTITION")
631 seen[state] = true
632 block.add(state)
633 }
634 blocks.add(block.sorted())
635 }
636 if (seen.any { !it }) fail("BAD PARTITION")
637 blocks.sortBy { it.first() }
638 val lumped = Chain(blocks.map { chain.states[it.first()] })
639 for (k in blocks.indices) {
640 val representative = blocks[k].first()
641 for (l in blocks.indices) {
642 val target = blocks[l]
643 val expected = chain.rowSum(representative, target)
644 lumped.transition[k][l] = expected
645 for (state in blocks[k]) {
646 val actual = if (state == representative) expected else chain.rowSum(state, target)
647 if (actual != expected) {
648 lines.add("LUMP $name not lumpable block=${chain.states[representative]} target=${chain.states[target.first()]} " +
649 "${chain.states[representative]}=${expected.value()} ${chain.states[state]}=${actual.value()}")
650 return lines
651 }
652 }
653 }
654 }
655 chains[words[2]] = lumped
656 lines.add("LUMP $name ${words[2]} blocks=${blocks.size}")
657 lumped.display(lines)
658 }
659 "REWARD" -> {
660 if (words.size - 3 != size) fail("BAD LENGTH")
661 val discount = probability(words[2])
662 if (discount == ONE) fail("BAD DISCOUNT")
663 val rewards = Array(size) { Fraction.of(words[3 + it].toLong()) }
664 val equations = matrix(size, size + 1)
665 for (i in 0 until size) {
666 for (j in 0 until size) equations[i][j] = (if (i == j) ONE else ZERO) - discount * transition[i][j]
667 equations[i][size] = rewards[i]
668 }
669 eliminate(equations)
670 val gains = vector(size)
671 for (group in graph.classes) {
672 if (!group.closed) continue
673 val rho = stationary(chain, group.states)
674 val gain = group.states.fold(ZERO) { sum, state -> sum + rho[state] * rewards[state] }
675 val chances = hitting(chain, graph, group.states, false).probabilities
676 for (i in 0 until size) gains[i] = gains[i] + chances[i] * gain
677 }
678 lines.add("REWARD $name gamma=$discount")
679 for (i in 0 until size) lines.add("VALUE ${chain.states[i]} discounted=${equations[i][size].value()} gain=${gains[i].value()}")
680 }
681 "RESTRICT" -> {
682 val selected = chain.targets(words.drop(3))
683 for (state in selected) {
684 for (j in 0 until size) if (j !in selected && transition[state][j] != ZERO) fail("NOT CLOSED ${chain.states[state]}")
685 }
686 val restricted = Chain(selected.map { chain.states[it] })
687 for (i in selected.indices) {
688 for (j in selected.indices) restricted.transition[i][j] = transition[selected[i]][selected[j]]
689 }
690 chains[words[2]] = restricted
691 lines.add("RESTRICT $name ${words[2]} states=${selected.size}")
692 restricted.display(lines)
693 }
694 "SIMULATE" -> {
695 var current = chain.state(words[2])
696 val steps = words[3].toInt()
697 val seed = words[4].toLong()
698 if (steps == 0) fail("BAD STEPS")
699 if (seed == 0L || seed >= 2_147_483_647L) fail("BAD SEED")
700 var random = seed
701 val path = mutableListOf(current)
702 val visits = IntArray(size)
703 visits[current]++
704 repeat(steps) {
705 val row = transition[current]
706 var denominator = 1L
707 for (value in row) denominator = denominator / gcd(denominator, value.denominator) * value.denominator
708 random = 48_271 * random % 2_147_483_647L
709 val pick = random % denominator
710 var cumulative = ZERO
711 for (j in 0 until size) {
712 cumulative += row[j]
713 val boundary = cumulative.numerator * (denominator / cumulative.denominator)
714 if (boundary > pick) {
715 current = j
716 break
717 }
718 }
719 path.add(current)
720 visits[current]++
721 }
722 lines.add("SIMULATE $name ${words[2]} steps=$steps seed=$seed end=${chain.states[current]}")
723 lines.add("PATH ${chain.names(path)}")
724 for (i in 0 until size) lines.add("VISIT ${chain.states[i]} ${visits[i]}")
725 }
726 else -> error(command)
727 }
728 return lines
729}

731fun main() {

733 val count = input.readLine().toInt()
734 val chains = mutableMapOf<String, Chain>()

736 repeat(count) {

738 try {
739 for (line in execute(chains, words)) output.appendLine(line)
740 } catch (error: CommandError) {
741 output.appendLine("ERROR ${error.message}")
742 }
743 }
