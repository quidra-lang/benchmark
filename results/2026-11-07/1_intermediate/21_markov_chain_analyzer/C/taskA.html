42static int64_t gcd(int64_t a, int64_t b) {
43 if (a < 0) a = -a;
44 if (b < 0) b = -b;
45 while (b) {
46 int64_t remainder = a % b;
47 a = b;
48 b = remainder;
49 }
50 return a;
51}

53static Fraction fraction(int64_t numerator, int64_t denominator) {
54 if (!numerator) return ZERO;
55 if (denominator < 0) {
56 numerator = -numerator;
57 denominator = -denominator;
58 }
59 int64_t divisor = gcd(numerator, denominator);
60 return (Fraction){numerator / divisor, denominator / divisor};
61}

63static Fraction add(Fraction a, Fraction b) {
64 return fraction(a.numerator * b.denominator + b.numerator * a.denominator,
65 a.denominator * b.denominator);
66}

68static Fraction subtract(Fraction a, Fraction b) {
69 return fraction(a.numerator * b.denominator - b.numerator * a.denominator,
70 a.denominator * b.denominator);
71}

73static Fraction multiply(Fraction a, Fraction b) {
74 return fraction(a.numerator * b.numerator, a.denominator * b.denominator);

77static Fraction divide(Fraction a, Fraction b) {
78 return fraction(a.numerator * b.denominator, a.denominator * b.numerator);

82 return a.numerator == b.numerator && a.denominator == b.denominator;

86 return a.numerator * b.denominator > b.numerator * a.denominator;

114static bool probability(const char *text, Fraction *result) {
115 const char *slash = strchr(text, '/');
116 int64_t numerator = strtoll(text, NULL, 10);
117 int64_t denominator = slash ? strtoll(slash + 1, NULL, 10) : 1;
118 if (!denominator || numerator > denominator) {
119 error("BAD PROBABILITY");
120 return false;
121 }
122 *result = fraction(numerator, denominator);
123 return true;
124}

145static Matrix matrix_product(const Matrix *left, const Matrix *right) {
146 Matrix result = zero_matrix(left->rows, right->columns);
147 for (int i = 0; i < result.rows; ++i)
148 for (int j = 0; j < result.columns; ++j)
149 for (int k = 0; k < left->columns; ++k)
150 result.value[i][j] = add(result.value[i][j], multiply(left->value[i][k], right->value[k][j]));
151 return result;
152}

154static void eliminate(Matrix *matrix) {
155 for (int column = 0; column < matrix->rows; ++column) {
156 int pivot_row = column;
157 while (!matrix->value[pivot_row][column].numerator) ++pivot_row;
158 for (int j = 0; j < matrix->columns; ++j) {
159 Fraction temporary = matrix->value[column][j];
160 matrix->value[column][j] = matrix->value[pivot_row][j];
161 matrix->value[pivot_row][j] = temporary;
162 }
163 Fraction pivot = matrix->value[column][column];
164 for (int j = 0; j < matrix->columns; ++j)
165 matrix->value[column][j] = divide(matrix->value[column][j], pivot);
166 for (int row = 0; row < matrix->rows; ++row) {
167 if (row == column) continue;
168 Fraction factor = matrix->value[row][column];
169 if (!factor.numerator) continue;
170 for (int j = 0; j < matrix->columns; ++j)
171 matrix->value[row][j] = subtract(matrix->value[row][j], multiply(factor, matrix->value[column][j]));
172 }
173 }
174}

236static Graph analyze(const Chain *chain) {
237 Graph graph = {0};
238 int size = chain->size;
239 for (int i = 0; i < size; ++i) {
240 graph.class_of[i] = -1;
241 graph.level[i] = -1;
242 for (int j = 0; j < size; ++j)
243 graph.reachable[i][j] = i == j || chain->transition.value[i][j].numerator > 0;
244 }
245 for (int k = 0; k < size; ++k)
246 for (int i = 0; i < size; ++i)
247 for (int j = 0; j < size; ++j)
248 graph.reachable[i][j] |= graph.reachable[i][k] && graph.reachable[k][j];
249 for (int root = 0; root < size; ++root) {
250 if (graph.class_of[root] >= 0) continue;
251 Class *group = &graph.classes[graph.count];
252 group->closed = true;
253 for (int i = 0; i < size; ++i) {
254 if (graph.reachable[root][i] && graph.reachable[i][root]) {
255 group->states |= 1u << i;
256 graph.class_of[i] = graph.count;
257 }
258 }
259 int queue[MAX_STATES], head = 0, tail = 0;
260 queue[tail++] = root;
261 graph.level[root] = 0;
262 while (head < tail) {
263 int current = queue[head++];
264 for (int next = 0; next < size; ++next) {
265 if (!chain->transition.value[current][next].numerator) continue;
266 if (!(group->states & (1u << next))) {
267 group->closed = false;
268 } else if (graph.level[next] < 0) {
269 graph.level[next] = graph.level[current] + 1;
270 queue[tail++] = next;
271 }
272 }
273 }
274 for (int i = 0; i < size; ++i)
275 for (int j = 0; j < size; ++j)
276 if ((group->states & (1u << i)) && (group->states & (1u << j)) && chain->transition.value[i][j].numerator)
277 group->period = (int)gcd(group->period, graph.level[i] + 1 - graph.level[j]);
278 ++graph.count;
279 }
280 return graph;
281}

296static Vector stationary(const Chain *chain, unsigned selected) {
297 int states[MAX_STATES];
298 int count = members(selected, chain->size, states);
299 Matrix equations = zero_matrix(count, count + 1);
300 for (int i = 0; i < count - 1; ++i)
301 for (int j = 0; j < count; ++j)
302 equations.value[i][j] = subtract(chain->transition.value[states[j]][states[i]], i == j ? ONE : ZERO);
303 for (int j = 0; j <= count; ++j) equations.value[count - 1][j] = ONE;
304 eliminate(&equations);
305 Vector result = zero_vector();
306 for (int i = 0; i < count; ++i) result.value[states[i]] = equations.value[i][count];
307 return result;
308}

310static Hitting hitting(const Chain *chain, const Graph *graph, unsigned target, bool with_times) {
311 Hitting result = {.probability = zero_vector(), .time = zero_vector()};
312 int unknown[MAX_STATES], count = 0;
313 for (int i = 0; i < chain->size; ++i) {
314 if (target & (1u << i)) result.probability.value[i] = ONE;
315 else if (reaches(graph, i, target, chain->size)) unknown[count++] = i;
316 }
317 if (count) {
318 Matrix equations = zero_matrix(count, count + 1);
319 for (int i = 0; i < count; ++i) {
320 for (int j = 0; j < count; ++j)
321 equations.value[i][j] = subtract(i == j ? ONE : ZERO, chain->transition.value[unknown[i]][unknown[j]]);
322 equations.value[i][count] = row_sum(chain, unknown[i], target);
323 }
324 eliminate(&equations);
325 for (int i = 0; i < count; ++i) result.probability.value[unknown[i]] = equations.value[i][count];
326 }
327 count = 0;
328 for (int i = 0; i < chain->size; ++i) {
329 result.finite[i] = equal(result.probability.value[i], ONE);
330 if (result.finite[i] && !(target & (1u << i))) unknown[count++] = i;
331 }
332 if (with_times && count) {
333 Matrix equations = zero_matrix(count, count + 1);
334 for (int i = 0; i < count; ++i) {
335 for (int j = 0; j < count; ++j)
336 equations.value[i][j] = subtract(i == j ? ONE : ZERO, chain->transition.value[unknown[i]][unknown[j]]);
337 equations.value[i][count] = ONE;
338 }
339 eliminate(&equations);
340 for (int i = 0; i < count; ++i) result.time.value[unknown[i]] = equations.value[i][count];
341 }
342 return result;
343}

345static Absorption absorption(const Chain *chain) {
346 Absorption result = {.mean = zero_vector(), .variance = zero_vector()};
347 for (int i = 0; i < chain->size; ++i) {
348 if (equal(chain->transition.value[i][i], ONE)) result.absorbing[result.a++] = i;
349 else result.transient[result.t++] = i;
350 }
351 Matrix equations = zero_matrix(result.t, 2 * result.t);
352 for (int i = 0; i < result.t; ++i) {
353 for (int j = 0; j < result.t; ++j)
354 equations.value[i][j] = subtract(i == j ? ONE : ZERO, chain->transition.value[result.transient[i]][result.transient[j]]);
355 equations.value[i][result.t + i] = ONE;
356 }
357 eliminate(&equations);
358 result.fundamental = zero_matrix(result.t, result.t);
359 result.probability = zero_matrix(result.t, result.a);
360 for (int i = 0; i < result.t; ++i)
361 for (int j = 0; j < result.t; ++j) {
362 result.fundamental.value[i][j] = equations.value[i][result.t + j];
363 result.mean.value[i] = add(result.mean.value[i], result.fundamental.value[i][j]);
364 }
365 for (int i = 0; i < result.t; ++i) {
366 Fraction second = ZERO;
367 for (int j = 0; j < result.t; ++j) {
368 Fraction coefficient = subtract(multiply(fraction(2, 1), result.fundamental.value[i][j]), i == j ? ONE : ZERO);
369 second = add(second, multiply(coefficient, result.mean.value[j]));
370 }
371 result.variance.value[i] = subtract(second, multiply(result.mean.value[i], result.mean.value[i]));
372 for (int j = 0; j < result.a; ++j)
373 for (int k = 0; k < result.t; ++k)
374 result.probability.value[i][j] = add(result.probability.value[i][j], multiply(result.fundamental.value[i][k], chain->transition.value[result.transient[k]][result.absorbing[j]]));
375 }
376 return result;
377}

400 int size = count - 2;
401 if (size > MAX_STATES) { error("TOO MANY STATES"); return; }
402 Chain chain = new_chain(words[1], size);
403 for (int i = 0; i < size; ++i) {
404 for (int j = 0; j < i; ++j)

409 printf("CHAIN %s states=%d\n", chain.name, size);

412static void build_chain(Store *store, char **words, int count) {
413 int order = atoi(words[2]), tokens = count - 3;
414 if (order != 1 && order != 2) { error("BAD ORDER"); return; }
415 if (tokens <= order) { error("TOO FEW TOKENS"); return; }
416 Chain chain = new_chain(words[1], 0);
417 int observations[60], length = tokens - order + 1;
418 for (int i = 0; i < length; ++i) {
419 char name[18];
420 if (order == 1) strcpy(name, words[i + 3]);
421 else snprintf(name, sizeof name, "%s.%s", words[i + 3], words[i + 4]);
422 int index = state_index(&chain, name);
423 if (index < 0) {
424 if (chain.size == MAX_STATES) { error("TOO MANY STATES"); return; }
425 index = chain.size++;
426 strcpy(chain.states[index], name);
427 }
428 observations[i] = index;
429 }
430 int transitions[MAX_STATES][MAX_STATES] = {0}, totals[MAX_STATES] = {0};
431 for (int i = 1; i < length; ++i) {
432 ++transitions[observations[i - 1]][observations[i]];
433 ++totals[observations[i - 1]];
434 }
435 chain.transition = zero_matrix(chain.size, chain.size);
436 for (int i = 0; i < chain.size; ++i) {
437 if (!totals[i]) chain.transition.value[i][i] = ONE;
438 else for (int j = 0; j < chain.size; ++j) chain.transition.value[i][j] = fraction(transitions[i][j], totals[i]);
439 }
440 save_chain(store, &chain);
441 printf("BUILD %s order=%d states=%d transitions=%d\n", chain.name, order, chain.size, length - 1);
442 for (int i = 0; i < chain.size; ++i) {
443 printf("FROM %s total=%d", chain.states[i], totals[i]);
444 if (!totals[i]) fputs(" absorbing", stdout);
445 else for (int j = 0; j < chain.size; ++j)
446 if (transitions[i][j]) printf(" %s:%d", chain.states[j], transitions[i][j]);
447 putchar('\n');
448 }
449}

451static void gambler(Store *store, char **words) {
452 int goal = atoi(words[2]), start = atoi(words[3]);
453 if (!goal || goal > 9) { error("BAD SIZE"); return; }
454 if (start > goal) { error("BAD START"); return; }
455 Fraction chance;
456 if (!probability(words[4], &chance)) return;
457 if (equal(chance, ZERO) || equal(chance, ONE)) { error("BAD PROBABILITY"); return; }
458 Chain chain = new_chain(words[1], goal + 1);
459 for (int i = 0; i <= goal; ++i) snprintf(chain.states[i], sizeof chain.states[i], "s%d", i);
460 chain.transition.value[0][0] = chain.transition.value[goal][goal] = ONE;
461 for (int i = 1; i < goal; ++i) {
462 chain.transition.value[i][i + 1] = chance;
463 chain.transition.value[i][i - 1] = subtract(ONE, chance);
464 }
465 Absorption result = absorption(&chain);
466 Vector ruin = zero_vector(), duration = zero_vector();
467 ruin.value[0] = ONE;
468 for (int i = 0; i < result.t; ++i) {
469 ruin.value[result.transient[i]] = result.probability.value[i][0];
470 duration.value[result.transient[i]] = result.mean.value[i];
471 }
472 save_chain(store, &chain);
473 printf("GAMBLER %s g=%d p=%s start=s%d ruin=%s duration=%s\n", chain.name, goal,
474 fraction_text(chance).text, start, value_text(ruin.value[start]).text, value_text(duration.value[start]).text);
475 for (int i = 0; i <= goal; ++i)
476 printf("CAPITAL s%d ruin=%s win=%s duration=%s\n", i, value_text(ruin.value[i]).text,
477 value_text(subtract(ONE, ruin.value[i])).text, value_text(duration.value[i]).text);
478}

480static void classify(const Chain *chain, const Graph *graph) {
481 unsigned targets = absorbing_states(chain);
482 int absorbing_count = 0;
483 bool absorbing = targets != 0;
484 bool edges[MAX_STATES][MAX_STATES] = {0}, current[MAX_STATES][MAX_STATES] = {0};
485 for (int i = 0; i < chain->size; ++i) {
486 if (targets & (1u << i)) ++absorbing_count;
487 absorbing &= reaches(graph, i, targets, chain->size);
488 for (int j = 0; j < chain->size; ++j)
489 edges[i][j] = current[i][j] = chain->transition.value[i][j].numerator > 0;
490 }
491 int exponent = 0, bound = (chain->size - 1) * (chain->size - 1) + 1;
492 for (int step = 1; step <= bound; ++step) {
493 bool positive = true;
494 for (int i = 0; i < chain->size; ++i)
495 for (int j = 0; j < chain->size; ++j) positive &= current[i][j];
496 if (positive) { exponent = step; break; }
497 bool next[MAX_STATES][MAX_STATES] = {0};
498 for (int i = 0; i < chain->size; ++i)
499 for (int j = 0; j < chain->size; ++j)
500 for (int k = 0; k < chain->size; ++k) next[i][j] |= current[i][k] && edges[k][j];
501 memcpy(current, next, sizeof current);
502 }
503 printf("CLASSIFY %s absorbing_states=%d absorbing=%s ergodic=%s regular=%s exponent=",
504 chain->name, absorbing_count, absorbing ? "yes" : "no", graph->count == 1 ? "yes" : "no", exponent ? "yes" : "no");
505 if (exponent) printf("%d\n", exponent); else puts("-");
506}

508static void show_absorption(const Chain *chain, const Graph *graph) {
509 unsigned targets = absorbing_states(chain);
510 if (!targets) { error("NO ABSORBING STATE"); return; }
511 for (int i = 0; i < chain->size; ++i)
512 if (!reaches(graph, i, targets, chain->size)) { error("NOT ABSORBING %s", chain->states[i]); return; }
513 Absorption result = absorption(chain);
514 printf("ABSORB %s transient=%d absorbing=%d\n", chain->name, result.t, result.a);
515 if (!result.t) return;
516 fputs("TRANSIENT", stdout);
517 for (int i = 0; i < result.t; ++i) printf(" %s", chain->states[result.transient[i]]);

528 for (int i = 0; i < result.t; ++i) {
529 printf("B %s", chain->states[result.transient[i]]);
530 for (int j = 0; j < result.a; ++j) printf(" %s", fraction_text(result.probability.value[i][j]).text);
531 putchar('\n');
532 }

535static void lump(Store *store, const Chain *chain, const char *destination, char *partition) {
536 char *groups[12];
537 int count = split(partition, "|", groups);
538 unsigned blocks[12] = {0}, seen = 0;
539 int representatives[12];
540 for (int block = 0; block < count; ++block) {
541 char *names[12];
542 int length = split(groups[block], ",", names);
543 for (int j = 0; j < length; ++j) {
544 int index = require_state(chain, names[j]);
545 if (index < 0) return;
546 if (seen & (1u << index)) { error("BAD PARTITION"); return; }
547 seen |= 1u << index;
548 blocks[block] |= 1u << index;
549 }
550 }
551 if (seen != (1u << chain->size) - 1) { error("BAD PARTITION"); return; }
552 for (int i = 0; i < count; ++i) {
553 representatives[i] = 0;
554 while (!(blocks[i] & (1u << representatives[i]))) ++representatives[i];
555 }
556 for (int i = 1; i < count; ++i) {
557 int representative = representatives[i], position = i;
558 unsigned block = blocks[i];
559 while (position > 0 && representatives[position - 1] > representative) {
560 representatives[position] = representatives[position - 1];
561 blocks[position] = blocks[position - 1];
562 --position;
563 }
564 representatives[position] = representative;
565 blocks[position] = block;
566 }
567 Chain result = new_chain(destination, count);
568 for (int i = 0; i < count; ++i) {
569 strcpy(result.states[i], chain->states[representatives[i]]);
570 for (int j = 0; j < count; ++j) {
571 Fraction expected = row_sum(chain, representatives[i], blocks[j]);
572 result.transition.value[i][j] = expected;
573 for (int state = 0; state < chain->size; ++state) {
574 if (!(blocks[i] & (1u << state))) continue;
575 Fraction actual = row_sum(chain, state, blocks[j]);
576 if (!equal(expected, actual)) {
577 printf("LUMP %s not lumpable block=%s target=%s %s=%s %s=%s\n", chain->name,
578 chain->states[representatives[i]], chain->states[representatives[j]],
579 chain->states[representatives[i]], value_text(expected).text, chain->states[state], value_text(actual).text);
580 return;
581 }
582 }
583 }
584 }
585 save_chain(store, &result);
586 printf("LUMP %s %s blocks=%d\n", chain->name, destination, count);
587 display_matrix(&result, &result.transition, "ROW");
588}

590static void command(Store *store, char **words, int count) {
591 const char *verb = words[0];
592 if (!strcmp(verb, "CHAIN")) { create_chain(store, words, count); return; }
593 if (!strcmp(verb, "BUILD")) { build_chain(store, words, count); return; }
594 if (!strcmp(verb, "GAMBLER")) { gambler(store, words); return; }
595 Chain *stored = find_chain(store, words[1]);
596 if (!stored) { error("UNKNOWN CHAIN %s", words[1]); return; }
597 Chain chain = *stored;
598 int size = chain.size;
599 unsigned all = (1u << size) - 1;
600 if (!strcmp(verb, "P") || !strcmp(verb, "ROW")) {
601 int row = require_state(&chain, words[2]);
602 if (row < 0) return;
603 if (!strcmp(verb, "P")) {
604 int column = require_state(&chain, words[3]);
605 if (column < 0 || !probability(words[4], &chain.transition.value[row][column])) return;
606 printf("P %s %s %s %s rowsum=%s\n", chain.name, words[2], words[3], fraction_text(chain.transition.value[row][column]).text, fraction_text(row_sum(&chain, row, all)).text);
607 } else {
608 if (count - 3 != size) { error("BAD LENGTH"); return; }
609 for (int j = 0; j < size; ++j)
610 if (!probability(words[j + 3], &chain.transition.value[row][j])) return;
611 printf("ROW %s %s rowsum=%s\n", chain.name, words[2], fraction_text(row_sum(&chain, row, all)).text);
612 }
613 save_chain(store, &chain);
614 return;
615 }
616 if (!strcmp(verb, "SHOW")) {
617 printf("MATRIX %s states=%d\n", chain.name, size);
618 display_matrix(&chain, &chain.transition, "ROW");
619 return;
620 }
621 Vector sums = zero_vector();
622 int bad = 0;
623 for (int i = 0; i < size; ++i) {
624 sums.value[i] = row_sum(&chain, i, all);
625 if (!equal(sums.value[i], ONE)) {
626 if (strcmp(verb, "CHECK")) { error("NOT STOCHASTIC %s", chain.states[i]); return; }
627 ++bad;
628 }
629 }
630 if (!strcmp(verb, "CHECK")) {
631 if (!bad) printf("CHECK %s ok\n", chain.name);
632 else {
633 printf("CHECK %s bad=%d\n", chain.name, bad);
634 for (int i = 0; i < size; ++i)
635 if (!equal(sums.value[i], ONE)) printf("BADROW %s sum=%s\n", chain.states[i], fraction_text(sums.value[i]).text);
636 }
637 return;
638 }
639 Graph graph = analyze(&chain);
640 if (!strcmp(verb, "CLASSES")) {
641 printf("CLASSES %s count=%d\n", chain.name, graph.count);
642 for (int c = 0; c < graph.count; ++c) {
643 Class group = graph.classes[c];
644 int indices[MAX_STATES], length = members(group.states, size, indices);
645 printf("CLASS %d %s period=%d size=%d", c + 1, group.closed ? "RECURRENT" : "TRANSIENT", group.period, length);
646 for (int i = 0; i < length; ++i) printf(" %s", chain.states[indices[i]]);
647 putchar('\n');
648 if (group.period >= 2)
649 for (int remainder = 0; remainder < group.period; ++remainder) {
650 printf("CYCLIC %d", remainder);
651 for (int i = 0; i < length; ++i)
652 if (graph.level[indices[i]] % group.period == remainder) printf(" %s", chain.states[indices[i]]);
653 putchar('\n');
654 }
655 }
656 } else if (!strcmp(verb, "CLASSIFY")) {
657 classify(&chain, &graph);
658 } else if (!strcmp(verb, "STEP")) {
659 Vector distribution = zero_vector();
660 if (isdigit((unsigned char)words[2][0])) {
661 char *values[12];
662 int length = split(words[2], ",", values);
663 if (length != size) { error("BAD LENGTH"); return; }
664 for (int i = 0; i < size; ++i)
665 if (!probability(values[i], &distribution.value[i])) return;
666 Fraction sum = ZERO;
667 for (int i = 0; i < size; ++i) sum = add(sum, distribution.value[i]);
668 if (!equal(sum, ONE)) { error("BAD DISTRIBUTION"); return; }
669 } else {
670 int state = require_state(&chain, words[2]);
671 if (state < 0) return;
672 distribution.value[state] = ONE;
673 }
674 int steps = atoi(words[3]);
675 for (int t = 0; t < steps; ++t) {
676 Vector next = zero_vector();
677 for (int j = 0; j < size; ++j)
678 for (int i = 0; i < size; ++i) next.value[j] = add(next.value[j], multiply(distribution.value[i], chain.transition.value[i][j]));
679 distribution = next;
680 }
681 printf("STEP %s k=%d\n", chain.name, steps);
682 for (int i = 0; i < size; ++i) printf("PROB %s %s\n", chain.states[i], value_text(distribution.value[i]).text);
683 } else if (!strcmp(verb, "POWER")) {
684 int steps = atoi(words[2]);
685 Matrix power = identity(size);
686 for (int t = 0; t < steps; ++t) power = matrix_product(&power, &chain.transition);
687 printf("POWER %s k=%d\n", chain.name, steps);
688 display_matrix(&chain, &power, "ROW");
689 } else if (!strcmp(verb, "STATIONARY")) {
690 int closed = 0;
691 for (int c = 0; c < graph.count; ++c) if (graph.classes[c].closed) ++closed;
692 printf("STATIONARY %s count=%d\n", chain.name, closed);
693 int number = 0;
694 for (int c = 0; c < graph.count; ++c) {
695 if (!graph.classes[c].closed) continue;
696 int indices[MAX_STATES], length = members(graph.classes[c].states, size, indices);
697 Vector distribution = stationary(&chain, graph.classes[c].states);
698 printf("DIST %d class=%d size=%d\n", ++number, c + 1, length);
699 for (int i = 0; i < length; ++i) printf("PI %s %s\n", chain.states[indices[i]], value_text(distribution.value[indices[i]]).text);
700 }
701 } else if (!strcmp(verb, "RECURRENCE")) {
702 int state = require_state(&chain, words[2]);
703 if (state < 0) return;
704 Class group = graph.classes[graph.class_of[state]];
705 if (!group.closed) { error("TRANSIENT STATE %s", words[2]); return; }
706 Vector distribution = stationary(&chain, group.states);
707 printf("RECURRENCE %s %s pi=%s mean=%s period=%d\n", chain.name, words[2], value_text(distribution.value[state]).text, value_text(divide(ONE, distribution.value[state])).text, group.period);
708 } else if (!strcmp(verb, "ABSORB")) {
709 show_absorption(&chain, &graph);
710 } else if (!strcmp(verb, "HITTING") || !strcmp(verb, "RESTRICT")) {
711 bool restrict_chain = !strcmp(verb, "RESTRICT");
712 int first = restrict_chain ? 3 : 2;
713 unsigned target;
714 if (!target_set(&chain, words + first, count - first, &target)) return;
715 if (restrict_chain) {
716 for (int i = 0; i < size; ++i)
717 for (int j = 0; j < size; ++j)
718 if ((target & (1u << i)) && !(target & (1u << j)) && chain.transition.value[i][j].numerator) { error("NOT CLOSED %s", chain.states[i]); return; }
719 int indices[MAX_STATES], length = members(target, size, indices);
720 Chain result = new_chain(words[2], length);
721 for (int i = 0; i < length; ++i) {
722 strcpy(result.states[i], chain.states[indices[i]]);
723 for (int j = 0; j < length; ++j) result.transition.value[i][j] = chain.transition.value[indices[i]][indices[j]];
724 }
725 save_chain(store, &result);
726 printf("RESTRICT %s %s states=%d\n", chain.name, words[2], length);
727 display_matrix(&result, &result.transition, "ROW");
728 } else {
729 Hitting result = hitting(&chain, &graph, target, true);
730 printf("HITTING %s targets=", chain.name);
731 bool first_name = true;
732 for (int i = 0; i < size; ++i)
733 if (target & (1u << i)) { printf("%s%s", first_name ? "" : ",", chain.states[i]); first_name = false; }
734 putchar('\n');
735 for (int i = 0; i < size; ++i)
736 printf("HIT %s prob=%s time=%s\n", chain.states[i], value_text(result.probability.value[i]).text, result.finite[i] ? value_text(result.time.value[i]).text : "inf");
737 }
738 } else if (!strcmp(verb, "FIRSTPASSAGE")) {
739 int source = require_state(&chain, words[2]);
740 if (source < 0) return;
741 int target = require_state(&chain, words[3]);
742 if (target < 0) return;
743 int steps = atoi(words[4]);
744 if (!steps) { error("BAD STEPS"); return; }
745 Fraction first[30], cumulative[30], total = ZERO;
746 Vector previous = zero_vector();
747 for (int i = 0; i < size; ++i) previous.value[i] = chain.transition.value[i][target];
748 for (int t = 0; t < steps; ++t) {
749 if (t) {
750 Vector next = zero_vector();
751 for (int i = 0; i < size; ++i)
752 for (int j = 0; j < size; ++j)
753 if (j != target) next.value[i] = add(next.value[i], multiply(chain.transition.value[i][j], previous.value[j]));
754 previous = next;
755 }
756 first[t] = previous.value[source];
757 total = cumulative[t] = add(total, first[t]);
758 }
759 printf("FIRSTPASSAGE %s %s %s k=%d total=%s\n", chain.name, words[2], words[3], steps, value_text(total).text);
760 for (int t = 0; t < steps; ++t) printf("FP %d %s cum=%s\n", t + 1, value_text(first[t]).text, value_text(cumulative[t]).text);
761 } else if (!strcmp(verb, "PATHPROB")) {
762 int path[30];
763 for (int i = 2; i < count; ++i) {
764 path[i - 2] = require_state(&chain, words[i]);
765 if (path[i - 2] < 0) return;
766 }
767 Fraction chance = ONE;
768 int zero_at = 0;
769 for (int i = 1; i < count - 2; ++i) {
770 chance = multiply(chance, chain.transition.value[path[i - 1]][path[i]]);
771 if (!chance.numerator && !zero_at) zero_at = i;
772 }
773 printf("PATHPROB %s steps=%d prob=%s", chain.name, count - 3, value_text(chance).text);
774 if (zero_at) printf(" zero_at=%d", zero_at);
775 putchar('\n');
776 } else if (!strcmp(verb, "MOSTLIKELY")) {
777 int source = require_state(&chain, words[2]);
778 if (source < 0) return;
779 int steps = atoi(words[3]);
780 if (!steps) { error("BAD STEPS"); return; }
781 Fraction best[21][MAX_STATES];
782 int choice[20][MAX_STATES];
783 for (int i = 0; i < size; ++i) best[steps][i] = ONE;
784 for (int t = steps - 1; t >= 0; --t)
785 for (int i = 0; i < size; ++i) {
786 best[t][i] = ZERO;
787 choice[t][i] = 0;
788 for (int j = 0; j < size; ++j) {
789 Fraction candidate = multiply(chain.transition.value[i][j], best[t + 1][j]);
790 if (greater(candidate, best[t][i])) { best[t][i] = candidate; choice[t][i] = j; }
791 }
792 }
793 printf("MOSTLIKELY %s %s k=%d prob=%s\nPATH %s", chain.name, words[2], steps, value_text(best[0][source]).text, chain.states[source]);
794 for (int t = 0; t < steps; ++t) { source = choice[t][source]; printf(" %s", chain.states[source]); }
795 putchar('\n');
796 } else if (!strcmp(verb, "LUMP")) {
797 lump(store, &chain, words[2], words[3]);
798 } else if (!strcmp(verb, "REWARD")) {
799 if (count - 3 != size) { error("BAD LENGTH"); return; }
800 Fraction gamma;
801 if (!probability(words[2], &gamma)) return;
802 if (equal(gamma, ONE)) { error("BAD DISCOUNT"); return; }
803 Vector rewards = zero_vector(), gain = zero_vector();
804 Matrix equations = zero_matrix(size, size + 1);
805 for (int i = 0; i < size; ++i) {
806 rewards.value[i] = fraction(atoi(words[i + 3]), 1);
807 for (int j = 0; j < size; ++j) equations.value[i][j] = subtract(i == j ? ONE : ZERO, multiply(gamma, chain.transition.value[i][j]));
808 equations.value[i][size] = rewards.value[i];
809 }
810 eliminate(&equations);
811 for (int c = 0; c < graph.count; ++c) {
812 if (!graph.classes[c].closed) continue;
813 unsigned group = graph.classes[c].states;
814 Vector distribution = stationary(&chain, group);
815 Fraction average = ZERO;
816 for (int i = 0; i < size; ++i)
817 if (group & (1u << i)) average = add(average, multiply(distribution.value[i], rewards.value[i]));
818 Hitting hit = hitting(&chain, &graph, group, false);
819 for (int i = 0; i < size; ++i) gain.value[i] = add(gain.value[i], multiply(hit.probability.value[i], average));
820 }
821 printf("REWARD %s gamma=%s\n", chain.name, fraction_text(gamma).text);
822 for (int i = 0; i < size; ++i) printf("VALUE %s discounted=%s gain=%s\n", chain.states[i], value_text(equations.value[i][size]).text, value_text(gain.value[i]).text);
823 } else if (!strcmp(verb, "SIMULATE")) {
824 int current = require_state(&chain, words[2]);
825 if (current < 0) return;
826 int steps = atoi(words[3]);
827 if (!steps) { error("BAD STEPS"); return; }
828 int64_t seed = strtoll(words[4], NULL, 10), random_value = seed;
829 if (!seed || seed >= 2147483647) { error("BAD SEED"); return; }
830 int path[61], visits[MAX_STATES] = {0};
831 path[0] = current;
832 ++visits[current];
833 for (int t = 1; t <= steps; ++t) {
834 int64_t common = 1;
835 for (int j = 0; j < size; ++j) {
836 int64_t denominator = chain.transition.value[current][j].denominator;
837 common = common / gcd(common, denominator) * denominator;
838 }
839 random_value = 48271 * random_value % 2147483647;
840 int64_t draw = random_value % common;
841 Fraction cumulative = ZERO;
842 for (int j = 0; j < size; ++j) {
843 cumulative = add(cumulative, chain.transition.value[current][j]);
844 if (cumulative.numerator * (common / cumulative.denominator) > draw) { current = j; break; }
845 }
846 path[t] = current;
847 ++visits[current];
848 }
849 printf("SIMULATE %s %s steps=%d seed=%" PRId64 " end=%s\nPATH", chain.name, words[2], steps, seed, chain.states[current]);
850 for (int t = 0; t <= steps; ++t) printf(" %s", chain.states[path[t]]);
851 putchar('\n');
852 for (int i = 0; i < size; ++i) printf("VISIT %s %d\n", chain.states[i], visits[i]);
853 } else {
854 int steps = 0;
855 Fraction epsilon = ZERO;
856 if (!strcmp(verb, "MIXING")) {
857 steps = atoi(words[2]);
858 if (!steps) { error("BAD STEPS"); return; }
859 if (!probability(words[3], &epsilon)) return;
860 }
861 if (graph.count != 1) { error("NOT IRREDUCIBLE"); return; }
862 Vector distribution = stationary(&chain, all);
863 if (!strcmp(verb, "PASSAGE")) {
864 Matrix passage = zero_matrix(size, size);
865 for (int j = 0; j < size; ++j) {
866 Hitting hit = hitting(&chain, &graph, 1u << j, true);
867 for (int i = 0; i < size; ++i) passage.value[i][j] = i == j ? divide(ONE, distribution.value[j]) : hit.time.value[i];
868 }
869 Fraction kemeny = ZERO;
870 for (int j = 1; j < size; ++j) kemeny = add(kemeny, multiply(distribution.value[j], passage.value[0][j]));
871 printf("PASSAGE %s kemeny=%s\n", chain.name, value_text(kemeny).text);
872 display_matrix(&chain, &passage, "M");
873 } else if (!strcmp(verb, "REVERSE")) {
874 Chain result = chain;
875 strcpy(result.name, words[2]);
876 bool reversible = true;
877 for (int i = 0; i < size; ++i)
878 for (int j = 0; j < size; ++j) {
879 result.transition.value[i][j] = divide(multiply(distribution.value[j], chain.transition.value[j][i]), distribution.value[i]);
880 reversible &= equal(result.transition.value[i][j], chain.transition.value[i][j]);
881 }
882 save_chain(store, &result);
883 printf("REVERSE %s %s reversible=%s\n", chain.name, words[2], reversible ? "yes" : "no");
884 display_matrix(&result, &result.transition, "ROW");
885 } else if (!strcmp(verb, "REVERSIBLE")) {
886 Fraction forward[MAX_STATES][MAX_STATES], backward[MAX_STATES][MAX_STATES];
887 int violations = 0;
888 for (int i = 0; i < size; ++i)
889 for (int j = i + 1; j < size; ++j) {
890 forward[i][j] = multiply(distribution.value[i], chain.transition.value[i][j]);
891 backward[i][j] = multiply(distribution.value[j], chain.transition.value[j][i]);
892 if (!equal(forward[i][j], backward[i][j])) ++violations;
893 }
894 if (!violations) printf("REVERSIBLE %s yes\n", chain.name);
895 else {
896 printf("REVERSIBLE %s no violations=%d\n", chain.name, violations);
897 for (int i = 0; i < size; ++i)
898 for (int j = i + 1; j < size; ++j)
899 if (!equal(forward[i][j], backward[i][j])) printf("VIOLATION %s %s %s %s\n", chain.states[i], chain.states[j], value_text(forward[i][j]).text, value_text(backward[i][j]).text);
900 }
901 } else {
902 Fraction distances[20];
903 int worst[20], mixing_time = 0;
904 Matrix power = identity(size);
905 for (int t = 0; t < steps; ++t) {
906 power = matrix_product(&power, &chain.transition);
907 distances[t] = ZERO;
908 worst[t] = 0;
909 for (int i = 0; i < size; ++i) {
910 Fraction distance = ZERO;
911 for (int j = 0; j < size; ++j) {
912 Fraction difference = subtract(power.value[i][j], distribution.value[j]);
913 if (difference.numerator < 0) difference.numerator = -difference.numerator;
914 distance = add(distance, difference);
915 }
916 distance = divide(distance, fraction(2, 1));
917 if (greater(distance, distances[t])) { distances[t] = distance; worst[t] = i; }
918 }
919 if (!mixing_time && !greater(distances[t], epsilon)) mixing_time = t + 1;
920 }
921 printf("MIXING %s k=%d eps=%s time=", chain.name, steps, fraction_text(epsilon).text);
922 if (mixing_time) printf("%d\n", mixing_time); else puts("none");
923 for (int t = 0; t < steps; ++t) printf("TV %d %s from=%s\n", t + 1, value_text(distances[t]).text, chain.states[worst[t]]);
924 }
925 }
