11 static long Gcd(long a, long b)
12 {
13 a = Math.Abs(a);
14 b = Math.Abs(b);
15 while (b != 0) (a, b) = (b, a % b);
16 return a;
17 }
18
19 readonly record struct Fraction : IComparable<Fraction>
20 {
21 public long Numerator { get; }
22 public long Denominator { get; }
23 public static readonly Fraction Zero = new(0);
24 public static readonly Fraction One = new(1);
25 public static readonly Fraction Two = new(2);
26
27 public Fraction(long numerator, long denominator = 1)
28 {
29 if (denominator < 0) (numerator, denominator) = (-numerator, -denominator);
30 long divisor = Gcd(numerator, denominator);
31 Numerator = numerator / divisor;
32 Denominator = denominator / divisor;
33 }
34
35 public static Fraction operator +(Fraction a, Fraction b) =>
36 new(a.Numerator * b.Denominator + b.Numerator * a.Denominator, a.Denominator * b.Denominator);
37 public static Fraction operator -(Fraction a, Fraction b) =>
38 new(a.Numerator * b.Denominator - b.Numerator * a.Denominator, a.Denominator * b.Denominator);
39 public static Fraction operator *(Fraction a, Fraction b) =>
40 new(a.Numerator * b.Numerator, a.Denominator * b.Denominator);
41 public static Fraction operator /(Fraction a, Fraction b) =>
42 new(a.Numerator * b.Denominator, a.Denominator * b.Numerator);
43 public int CompareTo(Fraction other) =>
44 (Numerator * other.Denominator).CompareTo(other.Numerator * Denominator);
45 public static bool operator <(Fraction a, Fraction b) => a.CompareTo(b) < 0;
46 public static bool operator >(Fraction a, Fraction b) => a.CompareTo(b) > 0;
47 public static bool operator <=(Fraction a, Fraction b) => a.CompareTo(b) <= 0;
48 public static bool operator >=(Fraction a, Fraction b) => a.CompareTo(b) >= 0;
49 public bool IsZero => Numerator == 0;
50 public Fraction Absolute => new(Math.Abs(Numerator), Denominator);
51 public override string ToString() => $"{Numerator}/{Denominator}";
52 public string Value => $"{this}({Numerator / Denominator}.{Numerator % Denominator * 1000000 / Denominator:D6})";
53 }

55 static Fraction Probability(string text)
56 {
57 var parts = text.Split('/');
58 long numerator = long.Parse(parts[0]);
59 long denominator = parts.Length == 1 ? 1 : long.Parse(parts[1]);
60 if (denominator == 0 || numerator > denominator) throw new CommandError("BAD PROBABILITY");
61 return new Fraction(numerator, denominator);
62 }

64 static Fraction[] Vector(int size) => Enumerable.Repeat(Fraction.Zero, size).ToArray();
65 static Fraction[][] Matrix(int rows, int columns) =>
66 Enumerable.Range(0, rows).Select(_ => Vector(columns)).ToArray();
67 static Fraction Delta(int i, int j) => i == j ? Fraction.One : Fraction.Zero;
68 static Fraction Sum(IEnumerable<Fraction> values)
69 {
70 var total = Fraction.Zero;
71 foreach (var value in values) total += value;
72 return total;
73 }
74
75 static Fraction[][] Identity(int size)
76 {
77 var result = Matrix(size, size);
78 for (int i = 0; i < size; i++) result[i][i] = Fraction.One;
79 return result;
80 }
81
82 static Fraction[] Advance(Fraction[] distribution, Fraction[][] transitions)
83 {
84 var result = Vector(distribution.Length);
85 for (int j = 0; j < result.Length; j++)
86 for (int i = 0; i < distribution.Length; i++)
87 result[j] += distribution[i] * transitions[i][j];
88 return result;
89 }
90
91 static Fraction[][] Multiply(Fraction[][] left, Fraction[][] right) =>
92 left.Select(row => Advance(row, right)).ToArray();
93
94 static Fraction[][] Solve(Fraction[][] rows)
95 {
96 int size = rows.Length;
97 if (size == 0) return [];
98 int width = rows[0].Length;
99 for (int column = 0; column < size; column++)
100 {
101 int pivotRow = column;
102 while (rows[pivotRow][column].IsZero) pivotRow++;
103 (rows[column], rows[pivotRow]) = (rows[pivotRow], rows[column]);
104 var pivot = rows[column][column];
105 for (int j = 0; j < width; j++) rows[column][j] /= pivot;
106 for (int i = 0; i < size; i++)
107 {
108 if (i == column) continue;
109 var factor = rows[i][column];
110 if (factor.IsZero) continue;
111 for (int j = 0; j < width; j++) rows[i][j] -= factor * rows[column][j];
112 }
113 }
114 return rows.Select(row => row[size..]).ToArray();
115 }

123 sealed class Chain(string[] names, Fraction[][] transitions)
124 {
125 public string[] Names { get; } = names;
126 public Fraction[][] P { get; } = transitions;
127 public int Size => Names.Length;
128 public IEnumerable<int> Indices => Enumerable.Range(0, Size);
129 public string Labels(IEnumerable<int> states, string separator = " ") =>
130 string.Join(separator, states.Select(i => Names[i]));
131
132 public int State(string name)
133 {
134 int index = Array.IndexOf(Names, name);
135 if (index < 0) throw new CommandError($"UNKNOWN STATE {name}");
136 return index;
137 }
138
139 public int[] StateSet(IEnumerable<string> names)
140 {
141 var states = new HashSet<int>();
142 foreach (string name in names)
143 if (!states.Add(State(name))) throw new CommandError($"DUPLICATE STATE {name}");
144 return states.Order().ToArray();
145 }
146
147 public void CheckStochastic()
148 {
149 foreach (int i in Indices)
150 if (Sum(P[i]) != Fraction.One) throw new CommandError($"NOT STOCHASTIC {Names[i]}");
151 }
152
153 public Graph AnalyzeGraph()
154 {
155 var reach = Indices.Select(i => Indices.Select(j => i == j || !P[i][j].IsZero).ToArray()).ToArray();
156 for (int k = 0; k < Size; k++)
157 for (int i = 0; i < Size; i++)
158 for (int j = 0; j < Size; j++)
159 reach[i][j] |= reach[i][k] && reach[k][j];
160 var assigned = new bool[Size];
161 var classes = new List<Component>();
162 for (int first = 0; first < Size; first++)
163 {
164 if (assigned[first]) continue;
165 int[] states = Indices.Where(j => reach[first][j] && reach[j][first]).ToArray();
166 var members = states.ToHashSet();
167 foreach (int i in states) assigned[i] = true;
168 bool closed = states.All(i => Indices.All(j => members.Contains(j) || P[i][j].IsZero));
169 var levels = Enumerable.Repeat(-1, Size).ToArray();
170 levels[first] = 0;
171 var queue = new Queue<int>();
172 queue.Enqueue(first);
173 while (queue.TryDequeue(out int current))
174 foreach (int j in states)
175 if (levels[j] < 0 && !P[current][j].IsZero)
176 {
177 levels[j] = levels[current] + 1;
178 queue.Enqueue(j);
179 }
180 int period = 0;
181 foreach (int i in states)
182 foreach (int j in states)
183 if (!P[i][j].IsZero) period = (int)Gcd(period, levels[i] + 1 - levels[j]);
184 classes.Add(new Component(states, closed, period, levels));
185 }
186 return new Graph(reach, classes.ToArray());
187 }
188
189 public Fraction[] Stationary(int[] states)
190 {
191 int count = states.Length;
192 var equations = Matrix(count, count + 1);
193 for (int row = 0; row < count - 1; row++)
194 for (int col = 0; col < count; col++)
195 equations[row][col] = P[states[col]][states[row]] - Delta(col, row);
196 Array.Fill(equations[count - 1], Fraction.One);
197 var solution = Solve(equations);
198 var distribution = Vector(Size);
199 for (int i = 0; i < count; i++) distribution[states[i]] = solution[i][0];
200 return distribution;
201 }
202
203 public Fraction[] UniqueStationary()
204 {
205 var classes = AnalyzeGraph().Classes;
206 if (classes.Length != 1) throw new CommandError("NOT IRREDUCIBLE");
207 return Stationary(classes[0].States);
208 }

210 public HitResult Hitting(int[] targets, bool withTimes = true)
211 {
212 var reach = AnalyzeGraph().Reach;
213 var targetSet = targets.ToHashSet();
214 int[] unknown = Indices.Where(i => !targetSet.Contains(i) && targets.Any(j => reach[i][j])).ToArray();
215 var probability = Vector(Size);
216 var time = new Fraction?[Size];
217 foreach (int target in targets)
218 {
219 probability[target] = Fraction.One;
220 time[target] = Fraction.Zero;
221 }
222 if (unknown.Length > 0)
223 {
224 var equations = Matrix(unknown.Length, unknown.Length + 1);
225 for (int i = 0; i < unknown.Length; i++)
226 {
227 for (int j = 0; j < unknown.Length; j++)
228 equations[i][j] = Delta(i, j) - P[unknown[i]][unknown[j]];
229 equations[i][unknown.Length] = Sum(targets.Select(j => P[unknown[i]][j]));
230 }
231 var solution = Solve(equations);
232 for (int i = 0; i < unknown.Length; i++) probability[unknown[i]] = solution[i][0];
233 }
234 int[] certain = unknown.Where(i => probability[i] == Fraction.One).ToArray();
235 if (withTimes && certain.Length > 0)
236 {
237 var equations = Matrix(certain.Length, certain.Length + 1);
238 for (int i = 0; i < certain.Length; i++)
239 {
240 for (int j = 0; j < certain.Length; j++)
241 equations[i][j] = Delta(i, j) - P[certain[i]][certain[j]];
242 equations[i][certain.Length] = Fraction.One;
243 }
244 var solution = Solve(equations);
245 for (int i = 0; i < certain.Length; i++) time[certain[i]] = solution[i][0];
246 }
247 return new HitResult(probability, time);
248 }
249
250 public Absorption Absorb()
251 {
252 int[] absorbing = Indices.Where(i => P[i][i] == Fraction.One).ToArray();
253 if (absorbing.Length == 0) throw new CommandError("NO ABSORBING STATE");
254 int[] transient = Indices.Where(i => P[i][i] != Fraction.One).ToArray();
255 var reach = AnalyzeGraph().Reach;
256 foreach (int i in transient)
257 if (!absorbing.Any(j => reach[i][j])) throw new CommandError($"NOT ABSORBING {Names[i]}");
258 int size = transient.Length;
259 var equations = Matrix(size, 2 * size);
260 for (int i = 0; i < size; i++)
261 for (int j = 0; j < size; j++)
262 {
263 equations[i][j] = Delta(i, j) - P[transient[i]][transient[j]];
264 equations[i][size + j] = Delta(i, j);
265 }
266 var fundamental = Solve(equations);
267 var mean = fundamental.Select(Sum).ToArray();
268 var variance = Vector(size);
269 var probability = Matrix(size, absorbing.Length);
270 for (int i = 0; i < size; i++)
271 {
272 for (int j = 0; j < size; j++)
273 variance[i] += (Fraction.Two * fundamental[i][j] - Delta(i, j)) * mean[j];
274 variance[i] -= mean[i] * mean[i];
275 for (int j = 0; j < absorbing.Length; j++)
276 for (int k = 0; k < size; k++)
277 probability[i][j] += fundamental[i][k] * P[transient[k]][absorbing[j]];
278 }
279 return new Absorption(transient, absorbing, fundamental, probability, mean, variance);
280 }

299 void Create(string[] words)
300 {
301 string name = words[1];
302 string[] names = words[2..];
303 if (names.Length > 10) throw new CommandError("TOO MANY STATES");
304 var seen = new HashSet<string>(StringComparer.Ordinal);
305 foreach (string state in names)
306 if (!seen.Add(state)) throw new CommandError($"DUPLICATE STATE {state}");
307 chains[name] = new Chain(names, Matrix(names.Length, names.Length));
308 Emit($"CHAIN {name} states={names.Length}");
309 }

311 void Build(string[] words)
312 {
313 string name = words[1];
314 int order = int.Parse(words[2]);
315 if (order is not (1 or 2)) throw new CommandError("BAD ORDER");
316 string[] tokens = words[3..];
317 if (tokens.Length <= order) throw new CommandError("TOO FEW TOKENS");
318 string[] observations = order == 1 ? tokens :
319 Enumerable.Range(0, tokens.Length - 1).Select(i => $"{tokens[i]}.{tokens[i + 1]}").ToArray();
320 var names = new List<string>();
321 var seen = new HashSet<string>(StringComparer.Ordinal);
322 foreach (string state in observations) if (seen.Add(state)) names.Add(state);
323 if (names.Count > 10) throw new CommandError("TOO MANY STATES");
324 var chain = new Chain(names.ToArray(), Matrix(names.Count, names.Count));
325 var counts = new int[chain.Size, chain.Size];
326 var totals = new int[chain.Size];
327 for (int i = 0; i < observations.Length - 1; i++)
328 {
329 int from = chain.State(observations[i]), to = chain.State(observations[i + 1]);
330 counts[from, to]++;
331 totals[from]++;
332 }
333 foreach (int i in chain.Indices)
334 if (totals[i] == 0) chain.P[i][i] = Fraction.One;
335 else foreach (int j in chain.Indices) chain.P[i][j] = new Fraction(counts[i, j], totals[i]);
336 chains[name] = chain;
337 Emit($"BUILD {name} order={order} states={chain.Size} transitions={observations.Length - 1}");
338 foreach (int i in chain.Indices)
339 {
340 string suffix = totals[i] == 0 ? "absorbing" : string.Join(" ", chain.Indices
341 .Where(j => counts[i, j] > 0).Select(j => $"{chain.Names[j]}:{counts[i, j]}"));
342 Emit($"FROM {chain.Names[i]} total={totals[i]} {suffix}");
343 }
344 }

346 void Gambler(string[] words)
347 {
348 string name = words[1];
349 int goal = int.Parse(words[2]), start = int.Parse(words[3]);
350 if (goal is 0 or > 9) throw new CommandError("BAD SIZE");
351 if (start > goal) throw new CommandError("BAD START");
352 var up = Probability(words[4]);
353 if (up.IsZero || up == Fraction.One) throw new CommandError("BAD PROBABILITY");
354 var names = Enumerable.Range(0, goal + 1).Select(i => $"s{i}").ToArray();
355 var chain = new Chain(names, Matrix(goal + 1, goal + 1));
356 chain.P[0][0] = chain.P[goal][goal] = Fraction.One;
357 for (int i = 1; i < goal; i++)
358 {
359 chain.P[i][i - 1] = Fraction.One - up;
360 chain.P[i][i + 1] = up;
361 }
362 var result = chain.Absorb();
363 var ruin = Vector(chain.Size);
364 var duration = Vector(chain.Size);
365 ruin[0] = Fraction.One;
366 for (int i = 0; i < result.Transient.Length; i++)
367 {
368 ruin[result.Transient[i]] = result.Probability[i][0];
369 duration[result.Transient[i]] = result.Mean[i];
370 }
371 chains[name] = chain;
372 Emit($"GAMBLER {name} g={goal} p={up} start=s{start} ruin={ruin[start].Value} duration={duration[start].Value}");
373 foreach (int i in chain.Indices)
374 Emit($"CAPITAL s{i} ruin={ruin[i].Value} win={(Fraction.One - ruin[i]).Value} duration={duration[i].Value}");
375 }

392 void Classify(string name, Chain chain)
393 {
394 var graph = chain.AnalyzeGraph();
395 int[] absorbing = chain.Indices.Where(i => chain.P[i][i] == Fraction.One).ToArray();
396 bool allAbsorbed = absorbing.Length > 0 && chain.Indices.All(i => absorbing.Any(j => graph.Reach[i][j]));
397 var edges = chain.P.Select(row => row.Select(p => !p.IsZero).ToArray()).ToArray();
398 var power = edges;
399 int? exponent = null;
400 int bound = (chain.Size - 1) * (chain.Size - 1) + 1;
401 for (int t = 1; t <= bound; t++)
402 {
403 if (power.All(row => row.All(value => value)))
404 {
405 exponent = t;
406 break;
407 }
408 power = chain.Indices.Select(i => chain.Indices.Select(j =>
409 chain.Indices.Any(k => power[i][k] && edges[k][j])).ToArray()).ToArray();
410 }
411 Emit($"CLASSIFY {name} absorbing_states={absorbing.Length} absorbing={Yes(allAbsorbed)} " +
412 $"ergodic={Yes(graph.Classes.Length == 1)} regular={Yes(exponent.HasValue)} exponent={exponent?.ToString() ?? "-"}");
413 }

415 void Step(string[] words, Chain chain)
416 {
417 Fraction[] distribution;
418 string initial = words[2];
419 if (char.IsAsciiDigit(initial[0]))
420 {
421 string[] values = initial.Split(',');
422 if (values.Length != chain.Size) throw new CommandError("BAD LENGTH");
423 distribution = values.Select(Probability).ToArray();
424 if (Sum(distribution) != Fraction.One) throw new CommandError("BAD DISTRIBUTION");
425 }
426 else
427 {
428 distribution = Vector(chain.Size);
429 distribution[chain.State(initial)] = Fraction.One;
430 }
431 int steps = int.Parse(words[3]);
432 for (int t = 0; t < steps; t++) distribution = Advance(distribution, chain.P);
433 Emit($"STEP {words[1]} k={steps}");
434 foreach (int i in chain.Indices) Emit($"PROB {chain.Names[i]} {distribution[i].Value}");
435 }

500 void FirstPassage(string[] words, Chain chain)
501 {
502 int from = chain.State(words[2]), target = chain.State(words[3]);
503 int steps = int.Parse(words[4]);
504 if (steps == 0) throw new CommandError("BAD STEPS");
505 var previous = chain.Indices.Select(i => chain.P[i][target]).ToArray();
506 var entries = new List<(Fraction Probability, Fraction Cumulative)>();
507 var total = Fraction.Zero;
508 for (int t = 1; t <= steps; t++)
509 {
510 total += previous[from];
511 entries.Add((previous[from], total));
512 if (t < steps)
513 previous = chain.Indices.Select(i => Sum(chain.Indices.Where(j => j != target)
514 .Select(j => chain.P[i][j] * previous[j]))).ToArray();
515 }
516 Emit($"FIRSTPASSAGE {words[1]} {words[2]} {words[3]} k={steps} total={total.Value}");
517 for (int i = 0; i < entries.Count; i++)
518 Emit($"FP {i + 1} {entries[i].Probability.Value} cum={entries[i].Cumulative.Value}");
519 }

521 void PathProbability(string[] words, Chain chain)
522 {
523 int[] states = words[2..].Select(chain.State).ToArray();
524 var product = Fraction.One;
525 int? zeroAt = null;
526 for (int i = 1; i < states.Length; i++)
527 {
528 product *= chain.P[states[i - 1]][states[i]];
529 if (product.IsZero) zeroAt ??= i;
530 }

532 (zeroAt.HasValue ? $" zero_at={zeroAt}" : ""));

535 void MostLikely(string[] words, Chain chain)
536 {
537 int start = chain.State(words[2]), steps = int.Parse(words[3]);
538 if (steps == 0) throw new CommandError("BAD STEPS");
539 var best = Matrix(steps + 1, chain.Size);
540 Array.Fill(best[steps], Fraction.One);
541 for (int t = steps - 1; t >= 0; t--)
542 foreach (int i in chain.Indices)
543 foreach (int j in chain.Indices)
544 {
545 var candidate = chain.P[i][j] * best[t + 1][j];
546 if (candidate > best[t][i]) best[t][i] = candidate;
547 }
548 var path = new List<int> { start };
549 int state = start;
550 for (int t = 0; t < steps; t++)
551 foreach (int j in chain.Indices)
552 if (chain.P[state][j] * best[t + 1][j] == best[t][state])
553 {
554 state = j;
555 path.Add(j);
556 break;
557 }
558 Emit($"MOSTLIKELY {words[1]} {words[2]} k={steps} prob={best[0][start].Value}");
559 Emit($"PATH {chain.Labels(path)}");
560 }

562 void Reversible(string name, Chain chain)
563 {
564 var pi = chain.UniqueStationary();
565 var violations = new List<(int From, int To, Fraction Forward, Fraction Backward)>();
566 foreach (int i in chain.Indices)
567 for (int j = i + 1; j < chain.Size; j++)
568 {
569 var forward = pi[i] * chain.P[i][j];
570 var backward = pi[j] * chain.P[j][i];
571 if (forward != backward) violations.Add((i, j, forward, backward));
572 }
573 Emit($"REVERSIBLE {name} {(violations.Count == 0 ? "yes" : $"no violations={violations.Count}")}");
574 foreach (var pair in violations)
575 Emit($"VIOLATION {chain.Names[pair.From]} {chain.Names[pair.To]} {pair.Forward.Value} {pair.Backward.Value}");
576 }

578 void Reverse(string[] words, Chain chain)
579 {
580 var pi = chain.UniqueStationary();
581 var reversed = new Chain([.. chain.Names], Matrix(chain.Size, chain.Size));
582 bool same = true;
583 foreach (int i in chain.Indices)
584 foreach (int j in chain.Indices)
585 {
586 reversed.P[i][j] = pi[j] * chain.P[j][i] / pi[i];
587 if (reversed.P[i][j] != chain.P[i][j]) same = false;
588 }
589 chains[words[2]] = reversed;
590 Emit($"REVERSE {words[1]} {words[2]} reversible={Yes(same)}");
591 Display(reversed);
592 }

594 void Mixing(string[] words, Chain chain)
595 {
596 int steps = int.Parse(words[2]);
597 if (steps == 0) throw new CommandError("BAD STEPS");
598 var epsilon = Probability(words[3]);
599 var pi = chain.UniqueStationary();
600 var power = Identity(chain.Size);
601 var entries = new List<(Fraction Distance, int From)>();
602 int? mixing = null;
603 for (int t = 1; t <= steps; t++)
604 {
605 power = Multiply(power, chain.P);
606 var distance = Fraction.Zero;
607 int from = 0;
608 foreach (int i in chain.Indices)
609 {
610 var current = Sum(chain.Indices.Select(j => (power[i][j] - pi[j]).Absolute)) / Fraction.Two;
611 if (current > distance) (distance, from) = (current, i);
612 }
613 if (distance <= epsilon) mixing ??= t;
614 entries.Add((distance, from));
615 }
616 Emit($"MIXING {words[1]} k={steps} eps={epsilon} time={mixing?.ToString() ?? "none"}");
617 for (int t = 0; t < entries.Count; t++)
618 Emit($"TV {t + 1} {entries[t].Distance.Value} from={chain.Names[entries[t].From]}");
619 }

621 void Lump(string[] words, Chain chain)
622 {
623 var blocks = new List<int[]>();
624 var seen = new HashSet<int>();
625 foreach (string text in words[3].Split('|'))
626 {
627 var block = new List<int>();
628 foreach (string name in text.Split(','))
629 {
630 int state = chain.State(name);
631 if (!seen.Add(state)) throw new CommandError("BAD PARTITION");
632 block.Add(state);
633 }
634 block.Sort();
635 blocks.Add(block.ToArray());
636 }
637 if (seen.Count != chain.Size) throw new CommandError("BAD PARTITION");
638 blocks.Sort((a, b) => a[0].CompareTo(b[0]));
639 var lumped = new Chain(blocks.Select(block => chain.Names[block[0]]).ToArray(), Matrix(blocks.Count, blocks.Count));
640 for (int i = 0; i < blocks.Count; i++)
641 for (int j = 0; j < blocks.Count; j++)
642 {
643 int representative = blocks[i][0];
644 var baseline = Sum(blocks[j].Select(state => chain.P[representative][state]));
645 foreach (int state in blocks[i])
646 {
647 var value = Sum(blocks[j].Select(target => chain.P[state][target]));
648 if (value != baseline)
649 {
650 Emit($"LUMP {words[1]} not lumpable block={chain.Names[representative]} target={chain.Names[blocks[j][0]]} " +
651 $"{chain.Names[representative]}={baseline.Value} {chain.Names[state]}={value.Value}");
652 return;
653 }
654 }
655 lumped.P[i][j] = baseline;
656 }
657 chains[words[2]] = lumped;
658 Emit($"LUMP {words[1]} {words[2]} blocks={blocks.Count}");
659 Display(lumped);
660 }

662 void Reward(string[] words, Chain chain)
663 {
664 if (words.Length - 3 != chain.Size) throw new CommandError("BAD LENGTH");
665 var discount = Probability(words[2]);
666 if (discount == Fraction.One) throw new CommandError("BAD DISCOUNT");
667 var rewards = words[3..].Select(text => new Fraction(long.Parse(text))).ToArray();
668 var equations = Matrix(chain.Size, chain.Size + 1);
669 foreach (int i in chain.Indices)
670 {
671 foreach (int j in chain.Indices) equations[i][j] = Delta(i, j) - discount * chain.P[i][j];
672 equations[i][chain.Size] = rewards[i];
673 }
674 var discounted = Solve(equations);
675 var gain = Vector(chain.Size);
676 foreach (var component in chain.AnalyzeGraph().Classes)
677 {
678 if (!component.Closed) continue;
679 var rho = chain.Stationary(component.States);
680 var average = Sum(component.States.Select(i => rho[i] * rewards[i]));
681 var hitting = chain.Hitting(component.States, withTimes: false).Probability;
682 foreach (int i in chain.Indices) gain[i] += hitting[i] * average;
683 }
684 Emit($"REWARD {words[1]} gamma={discount}");
685 foreach (int i in chain.Indices)
686 Emit($"VALUE {chain.Names[i]} discounted={discounted[i][0].Value} gain={gain[i].Value}");
687 }

689 void Restrict(string[] words, Chain chain)
690 {
691 int[] states = chain.StateSet(words[3..]);
692 var inside = states.ToHashSet();
693 foreach (int i in states)
694 if (chain.Indices.Any(j => !inside.Contains(j) && !chain.P[i][j].IsZero))
695 throw new CommandError($"NOT CLOSED {chain.Names[i]}");
696 var restricted = new Chain(states.Select(i => chain.Names[i]).ToArray(),
697 states.Select(i => states.Select(j => chain.P[i][j]).ToArray()).ToArray());
698 chains[words[2]] = restricted;
699 Emit($"RESTRICT {words[1]} {words[2]} states={states.Length}");
700 Display(restricted);
701 }

703 void Simulate(string[] words, Chain chain)
704 {
705 int state = chain.State(words[2]), steps = int.Parse(words[3]);
706 if (steps == 0) throw new CommandError("BAD STEPS");
707 long seed = long.Parse(words[4]);
708 if (seed == 0 || seed >= 2147483647) throw new CommandError("BAD SEED");
709 long random = seed;
710 var visits = new int[chain.Size];
711 visits[state] = 1;
712 var path = new List<int> { state };
713 for (int t = 0; t < steps; t++)
714 {
715 long denominator = 1;
716 foreach (var value in chain.P[state])
717 denominator = denominator / Gcd(denominator, value.Denominator) * value.Denominator;
718 random = 48271 * random % 2147483647;
719 long sample = random % denominator;
720 var cumulative = Fraction.Zero;
721 foreach (int j in chain.Indices)
722 {
723 cumulative += chain.P[state][j];
724 if (cumulative.Numerator * (denominator / cumulative.Denominator) > sample)
725 {
726 state = j;
727 break;
728 }
729 }
730 path.Add(state);
731 visits[state]++;
732 }
733 Emit($"SIMULATE {words[1]} {words[2]} steps={steps} seed={seed} end={chain.Names[state]}");
734 Emit($"PATH {chain.Labels(path)}");
735 foreach (int i in chain.Indices) Emit($"VISIT {chain.Names[i]} {visits[i]}");
736 }

738 public void Execute(string line)
739 {
740 var words = line.Split(' ');
741 try
742 {
743 string command = words[0], name = words[1];
744 if (command == "CHAIN") { Create(words); return; }
745 if (command == "BUILD") { Build(words); return; }
746 if (command == "GAMBLER") { Gambler(words); return; }
747 if (!chains.TryGetValue(name, out var chain)) throw new CommandError($"UNKNOWN CHAIN {name}");
748 if (command is not ("P" or "ROW" or "SHOW" or "CHECK")) chain.CheckStochastic();
749 switch (command)
750 {
751 case "P":
752 {
753 int from = chain.State(words[2]), to = chain.State(words[3]);
754 var value = Probability(words[4]);
755 chain.P[from][to] = value;
756 Emit($"P {name} {words[2]} {words[3]} {value} rowsum={Sum(chain.P[from])}");
757 break;
758 }
759 case "ROW":
760 {
761 int state = chain.State(words[2]);
762 if (words.Length - 3 != chain.Size) throw new CommandError("BAD LENGTH");
763 var row = words[3..].Select(Probability).ToArray();
764 chain.P[state] = row;
765 Emit($"ROW {name} {words[2]} rowsum={Sum(row)}");
766 break;
767 }
768 case "SHOW":
769 Emit($"MATRIX {name} states={chain.Size}");
770 Display(chain);
771 break;
772 case "CHECK":
773 {
774 var sums = chain.P.Select(Sum).ToArray();
775 int[] bad = chain.Indices.Where(i => sums[i] != Fraction.One).ToArray();
776 Emit($"CHECK {name} {(bad.Length == 0 ? "ok" : $"bad={bad.Length}")}");
777 foreach (int i in bad) Emit($"BADROW {chain.Names[i]} sum={sums[i]}");
778 break;
779 }
780 case "CLASSES": Classes(name, chain); break;
781 case "CLASSIFY": Classify(name, chain); break;
782 case "STEP": Step(words, chain); break;
783 case "POWER":
784 {
785 int steps = int.Parse(words[2]);
786 var power = Identity(chain.Size);
787 for (int t = 0; t < steps; t++) power = Multiply(power, chain.P);
788 Emit($"POWER {name} k={steps}");
789 Display(chain, power);
790 break;
791 }
792 case "STATIONARY": Stationary(name, chain); break;
793 case "RECURRENCE": Recurrence(words, chain); break;
794 case "ABSORB": Absorb(name, chain); break;
795 case "HITTING": Hitting(words, chain); break;
796 case "PASSAGE": Passage(name, chain); break;
797 case "FIRSTPASSAGE": FirstPassage(words, chain); break;
798 case "PATHPROB": PathProbability(words, chain); break;
799 case "MOSTLIKELY": MostLikely(words, chain); break;
800 case "REVERSIBLE": Reversible(name, chain); break;
801 case "REVERSE": Reverse(words, chain); break;
802 case "MIXING": Mixing(words, chain); break;
803 case "LUMP": Lump(words, chain); break;
804 case "REWARD": Reward(words, chain); break;
805 case "RESTRICT": Restrict(words, chain); break;
806 case "SIMULATE": Simulate(words, chain); break;
807 }
808 }
809 catch (CommandError error)
810 {
811 Emit($"ERROR {error.Message}");
812 }
813 }
814 }
