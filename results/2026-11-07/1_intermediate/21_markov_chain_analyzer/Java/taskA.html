19 private static long gcd(long a, long b) {
20 a = Math.abs(a);
21 b = Math.abs(b);
22 while (b != 0) {
23 long remainder = a % b;
24 a = b;
25 b = remainder;
26 }
27 return a;
28 }

30 private record Fraction(long numerator, long denominator) implements Comparable<Fraction> {
31 Fraction {
32 if (denominator < 0) {
33 numerator = -numerator;
34 denominator = -denominator;
35 }
36 long divisor = gcd(numerator, denominator);
37 numerator /= divisor;
38 denominator /= divisor;
39 }
40
41 Fraction add(Fraction other) {
42 return new Fraction(numerator * other.denominator + other.numerator * denominator,
43 denominator * other.denominator);
44 }
45
46 Fraction subtract(Fraction other) {
47 return new Fraction(numerator * other.denominator - other.numerator * denominator,
48 denominator * other.denominator);
49 }
50
51 Fraction multiply(Fraction other) {
52 return new Fraction(numerator * other.numerator, denominator * other.denominator);
53 }
54
55 Fraction divide(Fraction other) {
56 return new Fraction(numerator * other.denominator, denominator * other.numerator);
57 }
58
59 Fraction abs() { return new Fraction(Math.abs(numerator), denominator); }
60
61 @Override
62 public int compareTo(Fraction other) {
63 return Long.compare(numerator * other.denominator, other.numerator * denominator);
64 }
65
66 @Override
67 public String toString() { return numerator + "/" + denominator; }
68
69 String value() {
70 long decimal = numerator % denominator * 1_000_000 / denominator;
71 return this + "(" + numerator / denominator + "." +
72 String.format(Locale.ROOT, "%06d", decimal) + ")";
73 }
74 }

83 private static Fraction probability(String text) {
84 String[] parts = text.split("/");
85 long numerator = Long.parseLong(parts[0]);
86 long denominator = parts.length == 1 ? 1 : Long.parseLong(parts[1]);
87 if (denominator == 0 || numerator > denominator) throw new CommandError("BAD PROBABILITY");
88 return new Fraction(numerator, denominator);

115 private static Fraction[][] product(Fraction[][] left, Fraction[][] right) {
116 Fraction[][] result = matrix(left.length, right[0].length);
117 for (int i = 0; i < left.length; i++) {
118 for (int j = 0; j < right[0].length; j++) {
119 for (int k = 0; k < right.length; k++) {
120 result[i][j] = result[i][j].add(left[i][k].multiply(right[k][j]));
121 }
122 }
123 }
124 return result;
125 }

127 private static Fraction[][] eliminate(Fraction[][] equations) {
128 int size = equations.length;
129 for (int column = 0; column < size; column++) {
130 int pivotRow = column;
131 while (equations[pivotRow][column].equals(ZERO)) pivotRow++;
132 Fraction[] swapped = equations[column];
133 equations[column] = equations[pivotRow];
134 equations[pivotRow] = swapped;
135 Fraction pivot = equations[column][column];
136 for (int j = 0; j < equations[column].length; j++) {
137 equations[column][j] = equations[column][j].divide(pivot);
138 }
139 for (int row = 0; row < size; row++) {
140 if (row == column || equations[row][column].equals(ZERO)) continue;
141 Fraction factor = equations[row][column];
142 for (int j = 0; j < equations[row].length; j++) {
143 equations[row][j] = equations[row][j].subtract(factor.multiply(equations[column][j]));
144 }
145 }
146 }
147 return equations;
148 }

158 Chain(List<String> states) {
159 this.states = List.copyOf(states);
160 transition = matrix(states.size(), states.size());

162
163 int size() { return states.size(); }
164
165 int state(String name) {
166 int index = states.indexOf(name);
167 if (index < 0) throw new CommandError("UNKNOWN STATE " + name);
168 return index;
169 }
170
171 Fraction rowSum(int row, List<Integer> selected) {
172 Fraction result = ZERO;
173 for (int column : selected) result = result.add(transition[row][column]);
174 return result;
175 }

179 for (int i = start; i < words.length; i++) {
180 int index = state(words[i]);

182 selected[index] = true;
183 }
184 List<Integer> result = new ArrayList<>();
185 for (int i = 0; i < size(); i++) if (selected[i]) result.add(i);
186 return result;
187 }

190 List<Integer> result = new ArrayList<>();
191 for (int i = 0; i < size(); i++) if (transition[i][i].equals(ONE)) result.add(i);
192 return result;
193 }

205 private record CommunicatingClass(List<Integer> states, boolean closed, int period) {}
206
207 private static final class Graph {
208 final boolean[][] reachable;
209 final List<CommunicatingClass> classes = new ArrayList<>();
210 final int[] classOf;
211 final int[] levels;
212
213 Graph(Chain chain) {
214 int size = chain.size();
215 reachable = new boolean[size][size];
216 for (int i = 0; i < size; i++) {
217 for (int j = 0; j < size; j++) {
218 reachable[i][j] = i == j || !chain.transition[i][j].equals(ZERO);
219 }
220 }
221 for (int k = 0; k < size; k++) {
222 for (int i = 0; i < size; i++) {
223 for (int j = 0; j < size; j++) reachable[i][j] |= reachable[i][k] && reachable[k][j];
224 }
225 }
226 classOf = new int[size];
227 Arrays.fill(classOf, -1);
228 levels = new int[size];
229 for (int root = 0; root < size; root++) {
230 if (classOf[root] >= 0) continue;
231 List<Integer> states = new ArrayList<>();
232 for (int i = 0; i < size; i++) {
233 if (reachable[root][i] && reachable[i][root]) {
234 states.add(i);
235 classOf[i] = classes.size();
236 }
237 }
238 boolean closed = true;
239 boolean[] visited = new boolean[size];
240 ArrayDeque<Integer> queue = new ArrayDeque<>();
241 queue.add(root);
242 visited[root] = true;
243 while (!queue.isEmpty()) {
244 int current = queue.remove();
245 for (int next = 0; next < size; next++) {
246 if (chain.transition[current][next].equals(ZERO)) continue;
247 if (!states.contains(next)) {
248 closed = false;
249 } else if (!visited[next]) {
250 visited[next] = true;
251 levels[next] = levels[current] + 1;
252 queue.add(next);
253 }
254 }
255 }
256 long period = 0;
257 for (int i : states) {
258 for (int j : states) {
259 if (!chain.transition[i][j].equals(ZERO)) period = gcd(period, levels[i] + 1L - levels[j]);
260 }
261 }
262 classes.add(new CommunicatingClass(states, closed, (int) period));
263 }
264 }
265
266 boolean reaches(int source, List<Integer> targets) {
267 for (int target : targets) if (reachable[source][target]) return true;
268 return false;
269 }
270
271 void requireIrreducible() {
272 if (classes.size() != 1) throw new CommandError("NOT IRREDUCIBLE");
273 }
274 }

276 private static Fraction[] stationary(Chain chain, List<Integer> states) {
277 int count = states.size();
278 Fraction[][] equations = matrix(count, count + 1);
279 for (int i = 0; i < count - 1; i++) {
280 for (int j = 0; j < count; j++) {
281 equations[i][j] = chain.transition[states.get(j)][states.get(i)].subtract(i == j ? ONE : ZERO);
282 }
283 }
284 Arrays.fill(equations[count - 1], ONE);
285 eliminate(equations);
286 Fraction[] distribution = vector(chain.size());
287 for (int i = 0; i < count; i++) distribution[states.get(i)] = equations[i][count];
288 return distribution;
289 }

291 private record Hitting(Fraction[] probabilities, Fraction[] times) {}
292
293 private static Hitting hitting(Chain chain, Graph graph, List<Integer> targets, boolean withTimes) {
294 int size = chain.size();
295 Fraction[] probabilities = vector(size);
296 Fraction[] times = new Fraction[size];
297 for (int target : targets) {
298 probabilities[target] = ONE;
299 times[target] = ZERO;
300 }
301 List<Integer> unknown = new ArrayList<>();
302 for (int i = 0; i < size; i++) {
303 if (!targets.contains(i) && graph.reaches(i, targets)) unknown.add(i);
304 }
305 int count = unknown.size();
306 if (count > 0) {
307 Fraction[][] equations = matrix(count, count + 1);
308 for (int i = 0; i < count; i++) {
309 for (int j = 0; j < count; j++) {
310 equations[i][j] = (i == j ? ONE : ZERO).subtract(chain.transition[unknown.get(i)][unknown.get(j)]);
311 }
312 equations[i][count] = chain.rowSum(unknown.get(i), targets);
313 }
314 eliminate(equations);
315 for (int i = 0; i < count; i++) probabilities[unknown.get(i)] = equations[i][count];
316 }
317 if (withTimes) {
318 List<Integer> certain = unknown.stream().filter(i -> probabilities[i].equals(ONE)).toList();
319 count = certain.size();
320 if (count > 0) {
321 Fraction[][] equations = matrix(count, count + 1);
322 for (int i = 0; i < count; i++) {
323 for (int j = 0; j < count; j++) {
324 equations[i][j] = (i == j ? ONE : ZERO).subtract(chain.transition[certain.get(i)][certain.get(j)]);
325 }
326 equations[i][count] = ONE;
327 }
328 eliminate(equations);
329 for (int i = 0; i < count; i++) times[certain.get(i)] = equations[i][count];
330 }
331 }
332 return new Hitting(probabilities, times);
333 }

335 private record Absorption(List<Integer> transientStates, List<Integer> absorbingStates,
336 Fraction[][] fundamental, Fraction[] means, Fraction[] variances,
337 Fraction[][] probabilities) {}
338
339 private static Absorption absorption(Chain chain) {
340 List<Integer> absorbing = chain.absorbing();
341 List<Integer> transientStates = new ArrayList<>();
342 for (int i = 0; i < chain.size(); i++) if (!absorbing.contains(i)) transientStates.add(i);
343 int count = transientStates.size();
344 Fraction[][] equations = matrix(count, 2 * count);
345 for (int i = 0; i < count; i++) {
346 for (int j = 0; j < count; j++) {
347 equations[i][j] = (i == j ? ONE : ZERO).subtract(chain.transition[transientStates.get(i)][transientStates.get(j)]);
348 }
349 equations[i][count + i] = ONE;
350 }
351 eliminate(equations);
352 Fraction[][] fundamental = matrix(count, count);
353 Fraction[] means = vector(count);
354 for (int i = 0; i < count; i++) {
355 fundamental[i] = Arrays.copyOfRange(equations[i], count, 2 * count);
356 means[i] = sum(fundamental[i]);
357 }
358 Fraction[] variances = vector(count);
359 Fraction[][] probabilities = matrix(count, absorbing.size());
360 for (int i = 0; i < count; i++) {
361 Fraction second = ZERO;
362 for (int j = 0; j < count; j++) {
363 Fraction coefficient = TWO.multiply(fundamental[i][j]).subtract(i == j ? ONE : ZERO);
364 second = second.add(coefficient.multiply(means[j]));
365 }
366 variances[i] = second.subtract(means[i].multiply(means[i]));
367 for (int j = 0; j < absorbing.size(); j++) {
368 for (int k = 0; k < count; k++) {
369 probabilities[i][j] = probabilities[i][j].add(
370 fundamental[i][k].multiply(chain.transition[transientStates.get(k)][absorbing.get(j)]));
371 }
372 }
373 }
374 return new Absorption(transientStates, absorbing, fundamental, means, variances, probabilities);
375 }

377 private static Chain build(String[] words, List<String> lines) {
378 int order = number(words[2]);
379 int tokens = words.length - 3;
380 if (order != 1 && order != 2) throw new CommandError("BAD ORDER");
381 if (tokens <= order) throw new CommandError("TOO FEW TOKENS");
382 List<String> states = new ArrayList<>();
383 List<Integer> observations = new ArrayList<>();
384 for (int i = 0; i < tokens - order + 1; i++) {
385 String name = order == 1 ? words[3 + i] : words[3 + i] + "." + words[4 + i];
386 int index = states.indexOf(name);
387 if (index < 0) {
388 if (states.size() == 10) throw new CommandError("TOO MANY STATES");
389 index = states.size();
390 states.add(name);
391 }
392 observations.add(index);
393 }
394 Chain chain = new Chain(states);
395 int size = chain.size();
396 int[][] counts = new int[size][size];
397 int[] totals = new int[size];
398 for (int i = 0; i < observations.size() - 1; i++) {
399 int from = observations.get(i);
400 counts[from][observations.get(i + 1)]++;
401 totals[from]++;
402 }
403 lines.add("BUILD " + words[1] + " order=" + order + " states=" + size + " transitions=" + (observations.size() - 1));
404 for (int i = 0; i < size; i++) {
405 StringBuilder line = new StringBuilder("FROM " + states.get(i) + " total=" + totals[i]);
406 if (totals[i] == 0) {
407 chain.transition[i][i] = ONE;
408 line.append(" absorbing");
409 } else {
410 for (int j = 0; j < size; j++) {
411 chain.transition[i][j] = new Fraction(counts[i][j], totals[i]);
412 if (counts[i][j] > 0) line.append(' ').append(states.get(j)).append(':').append(counts[i][j]);
413 }
414 }
415 lines.add(line.toString());
416 }
417 return chain;
418 }

420 private static Chain gambler(String[] words, List<String> lines) {
421 int goal = number(words[2]);
422 int start = number(words[3]);
423 if (goal == 0 || goal > 9) throw new CommandError("BAD SIZE");
424 if (start > goal) throw new CommandError("BAD START");
425 Fraction chance = probability(words[4]);
426 if (chance.equals(ZERO) || chance.equals(ONE)) throw new CommandError("BAD PROBABILITY");
427 List<String> states = new ArrayList<>();
428 for (int i = 0; i <= goal; i++) states.add("s" + i);
429 Chain chain = new Chain(states);
430 chain.transition[0][0] = ONE;
431 chain.transition[goal][goal] = ONE;
432 for (int i = 1; i < goal; i++) {
433 chain.transition[i][i + 1] = chance;
434 chain.transition[i][i - 1] = ONE.subtract(chance);
435 }
436 Absorption result = absorption(chain);
437 Fraction[] ruin = vector(goal + 1);
438 Fraction[] duration = vector(goal + 1);
439 ruin[0] = ONE;
440 for (int i = 0; i < result.transientStates().size(); i++) {
441 int state = result.transientStates().get(i);
442 ruin[state] = result.probabilities()[i][0];
443 duration[state] = result.means()[i];
444 }
445 lines.add("GAMBLER " + words[1] + " g=" + goal + " p=" + chance + " start=s" + start +
446 " ruin=" + ruin[start].value() + " duration=" + duration[start].value());
447 for (int i = 0; i <= goal; i++) {
448 lines.add("CAPITAL s" + i + " ruin=" + ruin[i].value() + " win=" + ONE.subtract(ruin[i]).value() +
449 " duration=" + duration[i].value());
450 }
451 return chain;
452 }

454 private static List<String> execute(Map<String, Chain> chains, String[] words) {
455 String command = words[0];
456 String name = words[1];
457 List<String> lines = new ArrayList<>();
458 switch (command) {
459 case "CHAIN" -> {
460 List<String> states = Arrays.asList(words).subList(2, words.length);
461 if (states.size() > 10) throw new CommandError("TOO MANY STATES");
462 Set<String> seen = new HashSet<>();
463 for (String state : states) if (!seen.add(state)) throw new CommandError("DUPLICATE STATE " + state);
464 chains.put(name, new Chain(states));
465 lines.add("CHAIN " + name + " states=" + states.size());
466 return lines;
467 }
468 case "BUILD" -> {
469 chains.put(name, build(words, lines));
470 return lines;
471 }
472 case "GAMBLER" -> {
473 chains.put(name, gambler(words, lines));
474 return lines;
475 }
476 }
477 Chain chain = chains.get(name);
478 if (chain == null) throw new CommandError("UNKNOWN CHAIN " + name);
479 int size = chain.size();
480 Fraction[][] transition = chain.transition;
481 switch (command) {
482 case "P" -> {
483 int from = chain.state(words[2]);
484 int to = chain.state(words[3]);
485 Fraction value = probability(words[4]);
486 transition[from][to] = value;
487 lines.add("P " + name + " " + words[2] + " " + words[3] + " " + value + " rowsum=" + sum(transition[from]));
488 return lines;
489 }
490 case "ROW" -> {
491 int from = chain.state(words[2]);
492 if (words.length - 3 != size) throw new CommandError("BAD LENGTH");
493 Fraction[] row = vector(size);
494 for (int i = 0; i < size; i++) row[i] = probability(words[3 + i]);
495 transition[from] = row;
496 lines.add("ROW " + name + " " + words[2] + " rowsum=" + sum(row));
497 return lines;
498 }
499 case "SHOW" -> {
500 lines.add("MATRIX " + name + " states=" + size);
501 chain.display(lines, transition, "ROW");
502 return lines;
503 }
504 case "CHECK" -> {
505 List<String> badRows = new ArrayList<>();
506 for (int i = 0; i < size; i++) {
507 Fraction total = sum(transition[i]);
508 if (!total.equals(ONE)) badRows.add("BADROW " + chain.states.get(i) + " sum=" + total);
509 }
510 lines.add("CHECK " + name + (badRows.isEmpty() ? " ok" : " bad=" + badRows.size()));
511 lines.addAll(badRows);
512 return lines;
513 }
514 }
515 for (int i = 0; i < size; i++) {
516 if (!sum(transition[i]).equals(ONE)) throw new CommandError("NOT STOCHASTIC " + chain.states.get(i));
517 }
518 Graph graph = new Graph(chain);
519 switch (command) {
520 case "CLASSES" -> {
521 lines.add("CLASSES " + name + " count=" + graph.classes.size());
522 for (int i = 0; i < graph.classes.size(); i++) {
523 CommunicatingClass group = graph.classes.get(i);
524 lines.add("CLASS " + (i + 1) + " " + (group.closed() ? "RECURRENT" : "TRANSIENT") +
525 " period=" + group.period() + " size=" + group.states().size() + " " + chain.names(group.states(), " "));
526 if (group.period() >= 2) {
527 for (int remainder = 0; remainder < group.period(); remainder++) {
528 List<Integer> cyclic = new ArrayList<>();
529 for (int state : group.states()) if (graph.levels[state] % group.period() == remainder) cyclic.add(state);
530 lines.add("CYCLIC " + remainder + " " + chain.names(cyclic, " "));
531 }
532 }
533 }
534 }
535 case "CLASSIFY" -> {
536 List<Integer> absorbing = chain.absorbing();
537 boolean isAbsorbing = !absorbing.isEmpty();
538 for (int i = 0; i < size; i++) isAbsorbing &= graph.reaches(i, absorbing);
539 boolean[][] edges = new boolean[size][size];
540 for (int i = 0; i < size; i++) {
541 for (int j = 0; j < size; j++) edges[i][j] = !transition[i][j].equals(ZERO);
542 }
543 boolean[][] power = edges;
544 int exponent = 0;
545 int limit = (size - 1) * (size - 1) + 1;
546 for (int step = 1; step <= limit; step++) {
547 boolean positive = true;
548 for (boolean[] row : power) for (boolean value : row) positive &= value;
549 if (positive) {
550 exponent = step;
551 break;
552 }
553 if (step == limit) break;
554 boolean[][] next = new boolean[size][size];
555 for (int i = 0; i < size; i++) {
556 for (int j = 0; j < size; j++) {
557 for (int k = 0; k < size; k++) next[i][j] |= power[i][k] && edges[k][j];
558 }
559 }
560 power = next;
561 }
562 lines.add("CLASSIFY " + name + " absorbing_states=" + absorbing.size() + " absorbing=" + yes(isAbsorbing) +
563 " ergodic=" + yes(graph.classes.size() == 1) + " regular=" + yes(exponent != 0) +
564 " exponent=" + (exponent == 0 ? "-" : exponent));
565 }
566 case "STEP" -> {
567 Fraction[] distribution = vector(size);
568 if (Character.isDigit(words[2].charAt(0))) {
569 String[] values = words[2].split(",");
570 if (values.length != size) throw new CommandError("BAD LENGTH");
571 for (int i = 0; i < size; i++) distribution[i] = probability(values[i]);
572 if (!sum(distribution).equals(ONE)) throw new CommandError("BAD DISTRIBUTION");
573 } else {
574 distribution[chain.state(words[2])] = ONE;
575 }
576 int steps = number(words[3]);
577 for (int step = 0; step < steps; step++) {
578 Fraction[] next = vector(size);
579 for (int j = 0; j < size; j++) {
580 for (int i = 0; i < size; i++) next[j] = next[j].add(distribution[i].multiply(transition[i][j]));
581 }
582 distribution = next;
583 }
584 lines.add("STEP " + name + " k=" + steps);
585 for (int i = 0; i < size; i++) lines.add("PROB " + chain.states.get(i) + " " + distribution[i].value());
586 }
587 case "POWER" -> {
588 int steps = number(words[2]);
589 Fraction[][] power = identity(size);
590 for (int step = 0; step < steps; step++) power = product(power, transition);
591 lines.add("POWER " + name + " k=" + steps);
592 chain.display(lines, power, "ROW");
593 }
594 case "STATIONARY" -> {
595 long count = graph.classes.stream().filter(CommunicatingClass::closed).count();
596 lines.add("STATIONARY " + name + " count=" + count);
597 int distribution = 0;
598 for (int i = 0; i < graph.classes.size(); i++) {
599 CommunicatingClass group = graph.classes.get(i);
600 if (!group.closed()) continue;
601 Fraction[] pi = stationary(chain, group.states());
602 lines.add("DIST " + (++distribution) + " class=" + (i + 1) + " size=" + group.states().size());
603 for (int state : group.states()) lines.add("PI " + chain.states.get(state) + " " + pi[state].value());
604 }
605 }
606 case "RECURRENCE" -> {
607 int state = chain.state(words[2]);
608 CommunicatingClass group = graph.classes.get(graph.classOf[state]);
609 if (!group.closed()) throw new CommandError("TRANSIENT STATE " + words[2]);
610 Fraction rho = stationary(chain, group.states())[state];
611 lines.add("RECURRENCE " + name + " " + words[2] + " pi=" + rho.value() +
612 " mean=" + ONE.divide(rho).value() + " period=" + group.period());
613 }
614 case "ABSORB" -> {
615 List<Integer> absorbing = chain.absorbing();
616 if (absorbing.isEmpty()) throw new CommandError("NO ABSORBING STATE");
617 for (int i = 0; i < size; i++) {
618 if (!graph.reaches(i, absorbing)) throw new CommandError("NOT ABSORBING " + chain.states.get(i));
619 }
620 Absorption result = absorption(chain);
621 int count = result.transientStates().size();
622 lines.add("ABSORB " + name + " transient=" + count + " absorbing=" + absorbing.size());
623 if (count == 0) break;
624 lines.add("TRANSIENT " + chain.names(result.transientStates(), " "));
625 lines.add("ABSORBING " + chain.names(absorbing, " "));
626 for (int i = 0; i < count; i++) {
627 lines.add("N " + chain.states.get(result.transientStates().get(i)) + " " + fractions(result.fundamental()[i]));
628 }
629 for (int i = 0; i < count; i++) {
630 lines.add("STEPS " + chain.states.get(result.transientStates().get(i)) + " mean=" + result.means()[i].value() +
631 " var=" + result.variances()[i].value());
632 }
633 for (int i = 0; i < count; i++) {
634 lines.add("B " + chain.states.get(result.transientStates().get(i)) + " " + fractions(result.probabilities()[i]));
635 }
636 }
637 case "HITTING" -> {
638 List<Integer> targets = chain.targets(words, 2);
639 Hitting result = hitting(chain, graph, targets, true);
640 lines.add("HITTING " + name + " targets=" + chain.names(targets, ","));
641 for (int i = 0; i < size; i++) {
642 lines.add("HIT " + chain.states.get(i) + " prob=" + result.probabilities()[i].value() +
643 " time=" + (result.times()[i] == null ? "inf" : result.times()[i].value()));
644 }
645 }
646 case "PASSAGE" -> {
647 graph.requireIrreducible();
648 Fraction[] pi = stationary(chain, graph.classes.getFirst().states());
649 Fraction[][] passage = matrix(size, size);
650 for (int j = 0; j < size; j++) {
651 Fraction[] times = hitting(chain, graph, List.of(j), true).times();
652 for (int i = 0; i < size; i++) passage[i][j] = i == j ? ONE.divide(pi[j]) : times[i];
653 }
654 Fraction kemeny = ZERO;
655 for (int j = 1; j < size; j++) kemeny = kemeny.add(pi[j].multiply(passage[0][j]));
656 lines.add("PASSAGE " + name + " kemeny=" + kemeny.value());
657 chain.display(lines, passage, "M");
658 }
659 case "FIRSTPASSAGE" -> {
660 int from = chain.state(words[2]);
661 int target = chain.state(words[3]);
662 int steps = number(words[4]);
663 if (steps == 0) throw new CommandError("BAD STEPS");
664 Fraction[] current = vector(size);
665 for (int i = 0; i < size; i++) current[i] = transition[i][target];
666 Fraction cumulative = ZERO;
667 List<String> records = new ArrayList<>();
668 for (int step = 1; step <= steps; step++) {
669 cumulative = cumulative.add(current[from]);
670 records.add("FP " + step + " " + current[from].value() + " cum=" + cumulative.value());
671 if (step == steps) break;
672 Fraction[] next = vector(size);
673 for (int i = 0; i < size; i++) {
674 for (int j = 0; j < size; j++) {
675 if (j != target) next[i] = next[i].add(transition[i][j].multiply(current[j]));
676 }
677 }
678 current = next;
679 }
680 lines.add("FIRSTPASSAGE " + name + " " + words[2] + " " + words[3] + " k=" + steps + " total=" + cumulative.value());
681 lines.addAll(records);
682 }
683 case "PATHPROB" -> {
684 int[] path = new int[words.length - 2];
685 for (int i = 0; i < path.length; i++) path[i] = chain.state(words[2 + i]);
686 Fraction chance = ONE;
687 int zeroAt = 0;
688 for (int i = 1; i < path.length; i++) {
689 chance = chance.multiply(transition[path[i - 1]][path[i]]);
690 if (zeroAt == 0 && chance.equals(ZERO)) zeroAt = i;
691 }
692 lines.add("PATHPROB " + name + " steps=" + (path.length - 1) + " prob=" + chance.value() +
693 (zeroAt == 0 ? "" : " zero_at=" + zeroAt));
694 }
695 case "MOSTLIKELY" -> {
696 int start = chain.state(words[2]);
697 int steps = number(words[3]);
698 if (steps == 0) throw new CommandError("BAD STEPS");
699 Fraction[][] best = matrix(steps + 1, size);
700 Arrays.fill(best[steps], ONE);
701 int[][] choice = new int[steps][size];
702 for (int step = steps - 1; step >= 0; step--) {
703 for (int i = 0; i < size; i++) {
704 for (int j = 0; j < size; j++) {
705 Fraction candidate = transition[i][j].multiply(best[step + 1][j]);
706 if (candidate.compareTo(best[step][i]) > 0) {
707 best[step][i] = candidate;
708 choice[step][i] = j;
709 }
710 }
711 }
712 }
713 List<Integer> path = new ArrayList<>();
714 int current = start;
715 path.add(current);
716 for (int step = 0; step < steps; step++) {
717 current = choice[step][current];
718 path.add(current);
719 }
720 lines.add("MOSTLIKELY " + name + " " + words[2] + " k=" + steps + " prob=" + best[0][start].value());
721 lines.add("PATH " + chain.names(path, " "));
722 }
723 case "REVERSIBLE" -> {
724 graph.requireIrreducible();
725 Fraction[] pi = stationary(chain, graph.classes.getFirst().states());
726 List<String> violations = new ArrayList<>();
727 for (int i = 0; i < size; i++) {
728 for (int j = i + 1; j < size; j++) {
729 Fraction forward = pi[i].multiply(transition[i][j]);
730 Fraction backward = pi[j].multiply(transition[j][i]);
731 if (!forward.equals(backward)) {
732 violations.add("VIOLATION " + chain.states.get(i) + " " + chain.states.get(j) +
733 " " + forward.value() + " " + backward.value());
734 }
735 }
736 }
737 lines.add("REVERSIBLE " + name + (violations.isEmpty() ? " yes" : " no violations=" + violations.size()));
738 lines.addAll(violations);
739 }
740 case "REVERSE" -> {
741 graph.requireIrreducible();
742 Fraction[] pi = stationary(chain, graph.classes.getFirst().states());
743 Chain reversed = new Chain(chain.states);
744 boolean reversible = true;
745 for (int i = 0; i < size; i++) {
746 for (int j = 0; j < size; j++) {
747 reversed.transition[i][j] = pi[j].multiply(transition[j][i]).divide(pi[i]);
748 reversible &= reversed.transition[i][j].equals(transition[i][j]);
749 }
750 }
751 chains.put(words[2], reversed);
752 lines.add("REVERSE " + name + " " + words[2] + " reversible=" + yes(reversible));
753 reversed.display(lines, reversed.transition, "ROW");
754 }
755 case "MIXING" -> {
756 int steps = number(words[2]);
757 if (steps == 0) throw new CommandError("BAD STEPS");
758 Fraction epsilon = probability(words[3]);
759 graph.requireIrreducible();
760 Fraction[] pi = stationary(chain, graph.classes.getFirst().states());
761 Fraction[][] power = identity(size);
762 int mixingTime = 0;
763 List<String> records = new ArrayList<>();
764 for (int step = 1; step <= steps; step++) {
765 power = product(power, transition);
766 Fraction maximum = ZERO;
767 int worst = 0;
768 for (int i = 0; i < size; i++) {
769 Fraction distance = ZERO;
770 for (int j = 0; j < size; j++) distance = distance.add(power[i][j].subtract(pi[j]).abs());
771 distance = distance.divide(TWO);
772 if (distance.compareTo(maximum) > 0) {
773 maximum = distance;
774 worst = i;
775 }
776 }
777 if (mixingTime == 0 && maximum.compareTo(epsilon) <= 0) mixingTime = step;
778 records.add("TV " + step + " " + maximum.value() + " from=" + chain.states.get(worst));
779 }
780 lines.add("MIXING " + name + " k=" + steps + " eps=" + epsilon + " time=" + (mixingTime == 0 ? "none" : mixingTime));
781 lines.addAll(records);
782 }
783 case "LUMP" -> {
784 List<List<Integer>> blocks = new ArrayList<>();
785 boolean[] seen = new boolean[size];
786 for (String group : words[3].split("\\|")) {
787 List<Integer> block = new ArrayList<>();
788 for (String stateName : group.split(",")) {
789 int state = chain.state(stateName);
790 if (seen[state]) throw new CommandError("BAD PARTITION");
791 seen[state] = true;
792 block.add(state);
793 }
794 block.sort(Comparator.naturalOrder());
795 blocks.add(block);
796 }
797 for (boolean included : seen) if (!included) throw new CommandError("BAD PARTITION");
798 blocks.sort(Comparator.comparingInt(block -> block.getFirst()));
799 Chain lumped = new Chain(blocks.stream().map(block -> chain.states.get(block.getFirst())).toList());
800 for (int k = 0; k < blocks.size(); k++) {
801 int representative = blocks.get(k).getFirst();
802 for (int l = 0; l < blocks.size(); l++) {
803 List<Integer> target = blocks.get(l);
804 Fraction expected = chain.rowSum(representative, target);
805 lumped.transition[k][l] = expected;
806 for (int state : blocks.get(k)) {
807 Fraction actual = state == representative ? expected : chain.rowSum(state, target);
808 if (!actual.equals(expected)) {
809 lines.add("LUMP " + name + " not lumpable block=" + chain.states.get(representative) +
810 " target=" + chain.states.get(target.getFirst()) + " " + chain.states.get(representative) +
811 "=" + expected.value() + " " + chain.states.get(state) + "=" + actual.value());
812 return lines;
813 }
814 }
815 }
816 }
817 chains.put(words[2], lumped);
818 lines.add("LUMP " + name + " " + words[2] + " blocks=" + blocks.size());
819 lumped.display(lines, lumped.transition, "ROW");
820 }
821 case "REWARD" -> {
822 if (words.length - 3 != size) throw new CommandError("BAD LENGTH");
823 Fraction discount = probability(words[2]);
824 if (discount.equals(ONE)) throw new CommandError("BAD DISCOUNT");
825 Fraction[] rewards = vector(size);
826 for (int i = 0; i < size; i++) rewards[i] = new Fraction(number(words[3 + i]), 1);
827 Fraction[][] equations = matrix(size, size + 1);
828 for (int i = 0; i < size; i++) {
829 for (int j = 0; j < size; j++) equations[i][j] = (i == j ? ONE : ZERO).subtract(discount.multiply(transition[i][j]));
830 equations[i][size] = rewards[i];
831 }
832 eliminate(equations);
833 Fraction[] gains = vector(size);
834 for (CommunicatingClass group : graph.classes) {
835 if (!group.closed()) continue;
836 Fraction[] rho = stationary(chain, group.states());
837 Fraction gain = ZERO;
838 for (int state : group.states()) gain = gain.add(rho[state].multiply(rewards[state]));
839 Fraction[] chances = hitting(chain, graph, group.states(), false).probabilities();
840 for (int i = 0; i < size; i++) gains[i] = gains[i].add(chances[i].multiply(gain));
841 }
842 lines.add("REWARD " + name + " gamma=" + discount);
843 for (int i = 0; i < size; i++) {
844 lines.add("VALUE " + chain.states.get(i) + " discounted=" + equations[i][size].value() + " gain=" + gains[i].value());
845 }
846 }
847 case "RESTRICT" -> {
848 List<Integer> selected = chain.targets(words, 3);
849 for (int state : selected) {
850 for (int j = 0; j < size; j++) {
851 if (!selected.contains(j) && !transition[state][j].equals(ZERO)) {
852 throw new CommandError("NOT CLOSED " + chain.states.get(state));
853 }
854 }
855 }
856 Chain restricted = new Chain(selected.stream().map(chain.states::get).toList());
857 for (int i = 0; i < selected.size(); i++) {
858 for (int j = 0; j < selected.size(); j++) restricted.transition[i][j] = transition[selected.get(i)][selected.get(j)];
859 }
860 chains.put(words[2], restricted);
861 lines.add("RESTRICT " + name + " " + words[2] + " states=" + selected.size());
862 restricted.display(lines, restricted.transition, "ROW");
863 }
864 case "SIMULATE" -> {
865 int current = chain.state(words[2]);
866 int steps = number(words[3]);
867 long seed = Long.parseLong(words[4]);
868 if (steps == 0) throw new CommandError("BAD STEPS");
869 if (seed == 0 || seed >= 2_147_483_647L) throw new CommandError("BAD SEED");
870 long random = seed;
871 List<Integer> path = new ArrayList<>();
872 int[] visits = new int[size];
873 path.add(current);
874 visits[current]++;
875 for (int step = 0; step < steps; step++) {
876 Fraction[] row = transition[current];
877 long denominator = 1;
878 for (Fraction value : row) denominator = denominator / gcd(denominator, value.denominator()) * value.denominator();
879 random = 48_271 * random % 2_147_483_647L;
880 long pick = random % denominator;
881 Fraction cumulative = ZERO;
882 for (int j = 0; j < size; j++) {
883 cumulative = cumulative.add(row[j]);
884 long boundary = cumulative.numerator() * (denominator / cumulative.denominator());
885 if (boundary > pick) {
886 current = j;
887 break;
888 }
889 }
890 path.add(current);
891 visits[current]++;
892 }
893 lines.add("SIMULATE " + name + " " + words[2] + " steps=" + steps + " seed=" + seed + " end=" + chain.states.get(current));
894 lines.add("PATH " + chain.names(path, " "));
895 for (int i = 0; i < size; i++) lines.add("VISIT " + chain.states.get(i) + " " + visits[i]);
896 }
897 default -> throw new IllegalStateException(command);
898 }
899 return lines;
900 }

902 public static void main(String[] args) throws Exception {
903 BufferedReader input = new BufferedReader(new InputStreamReader(System.in));
904 int count = number(input.readLine());
905 Map<String, Chain> chains = new HashMap<>();
906 StringBuilder output = new StringBuilder();
907 for (int i = 0; i < count; i++) {
908 String[] words = input.readLine().split(" ");
909 try {
910 for (String line : execute(chains, words)) output.append(line).append('\n');
911 } catch (CommandError error) {
912 output.append("ERROR ").append(error.getMessage()).append('\n');
913 }
914 }
915 System.out.print(output);
916 }
