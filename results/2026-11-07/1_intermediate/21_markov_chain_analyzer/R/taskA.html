1gcd_exact <- function(a, b) {
2 a <- abs(a)
3 b <- abs(b)
4 while (b != 0) {
5 remainder <- a %% b
6 a <- b
7 b <- remainder
8 }
9 a
10}

12fraction <- function(n, d = 1) {
13 if (d < 0) {
14 n <- -n
15 d <- -d
16 }
17 divisor <- gcd_exact(n, d)
18 c(n / divisor, d / divisor)
19}

46probability <- function(text) {
47 parts <- strsplit(text, "/", fixed = TRUE)[[1]]
48 numerator <- as.numeric(parts[1])
49 denominator <- if (length(parts) == 2) as.numeric(parts[2]) else 1

58eliminate <- function(rows, size) {
59 if (size == 0) return(list())
60 for (column in seq_len(size)) {
61 pivot_row <- column
62 while (equal(rows[[pivot_row]][[column]], zero))
63 pivot_row <- pivot_row + 1
64 if (pivot_row != column) {
65 temporary <- rows[[column]]
66 rows[[column]] <- rows[[pivot_row]]
67 rows[[pivot_row]] <- temporary
68 }
69 pivot <- rows[[column]][[column]]
70 rows[[column]] <- lapply(rows[[column]], divide, b = pivot)
71 for (row in seq_len(size)) {
72 factor <- rows[[row]][[column]]
73 if (row != column && !equal(factor, zero)) {
74 rows[[row]] <- Map(function(value, base)
75 subtract(value, multiply(factor, base)), rows[[row]], rows[[column]])
76 }
77 }
78 }
79 lapply(rows, function(row) tail(row, length(row) - size))
80}

122reachable <- function(chain, start) {
123 n <- chain_size(chain)
124 seen <- rep(FALSE, n)
125 seen[start] <- TRUE
126 pending <- c(start)
127 while (length(pending)) {
128 i <- tail(pending, 1)
129 pending <- head(pending, -1)
130 for (j in seq_len(n)) {
131 if (edge(chain, i, j) && !seen[j]) {
132 seen[j] <- TRUE
133 pending <- c(pending, j)
134 }
135 }
136 }
137 seen
138}

140communicating_class <- function(chain, number, members) {
141 n <- chain_size(chain)
142 inside <- member_mask(n, members)
143 closed <- !any(vapply(members, function(i)
144 any(vapply(seq_len(n), function(j)
145 edge(chain, i, j) && !inside[j], logical(1))), logical(1)))
146 levels <- rep(-1, n)
147 levels[members[1]] <- 0
148 queue <- c(members[1])
149 index <- 1
150 while (index <= length(queue)) {
151 u <- queue[index]
152 index <- index + 1
153 for (v in members) {
154 if (edge(chain, u, v) && levels[v] < 0) {
155 levels[v] <- levels[u] + 1
156 queue <- c(queue, v)
157 }
158 }
159 }
160 period <- 0
161 for (u in members) {
162 for (v in members) {
163 if (edge(chain, u, v))
164 period <- gcd_exact(period, levels[u] + 1 - levels[v])
165 }
166 }
167 list(number = number, members = members, closed = closed,
168 levels = levels, period = period)
169}

171decompose <- function(chain) {
172 n <- chain_size(chain)
173 reach <- lapply(seq_len(n), function(i) reachable(chain, i))
174 classes <- list()
175 class_of <- integer(n)
176 for (i in seq_len(n)) {
177 if (class_of[i] == 0) {
178 members <- which(reach[[i]] & vapply(seq_len(n),
179 function(j) reach[[j]][i], logical(1)))
180 number <- length(classes) + 1
181 class_of[members] <- number
182 classes[[number]] <- communicating_class(chain, number, members)
183 }
184 }
185 list(reach = reach, classes = classes, class_of = class_of)
186}

190absorption <- function(chain) {
191 n <- chain_size(chain)
192 absorbing <- which(vapply(seq_len(n), function(i)
193 absorbing_state(chain, i), logical(1)))
194 transient <- setdiff(seq_len(n), absorbing)
195 count <- length(transient)
196 if (count == 0) {
197 return(list(transient = transient, absorbing = absorbing,
198 fundamental = list(), steps = list(), probabilities = list()))
199 }
200 equations <- lapply(seq_len(count), function(row) {
201 i <- transient[row]
202 left <- lapply(transient, function(j)
203 subtract(delta(i, j), chain$matrix[[i]][[j]]))
204 c(left, lapply(seq_len(count), function(j) delta(row, j)))
205 })
206 fundamental <- eliminate(equations, count)
207 steps <- lapply(fundamental, sum_fractions)
208 probabilities <- lapply(fundamental, function(row)
209 lapply(absorbing, function(a) sum_fractions(
210 lapply(seq_len(count), function(k)
211 multiply(row[[k]], chain$matrix[[transient[k]]][[a]])))))
212 list(transient = transient, absorbing = absorbing,
213 fundamental = fundamental, steps = steps, probabilities = probabilities)
214}

216absorption_variance <- function(result, i) {
217 n <- length(result$transient)
218 total <- sum_fractions(lapply(seq_len(n), function(j)
219 multiply(subtract(multiply(fraction(2), result$fundamental[[i]][[j]]),
220 delta(i, j)), result$steps[[j]])))
221 subtract(total, multiply(result$steps[[i]], result$steps[[i]]))
222}

224stationary_distribution <- function(chain, members) {
225 n <- length(members)
226 rows <- vector("list", n)
227 if (n > 1) {
228 for (r in seq_len(n - 1)) {
229 coefficients <- lapply(seq_len(n), function(k)
230 subtract(chain$matrix[[members[k]]][[members[r]]], delta(k, r)))
231 rows[[r]] <- c(coefficients, list(zero))
232 }
233 }
234 rows[[n]] <- c(rep(list(one), n), list(one))
235 lapply(eliminate(rows, n), function(row) row[[1]])
236}

280regular_exponent <- function(chain) {
281 n <- chain_size(chain)
282 adjacent <- matrix(FALSE, n, n)
283 for (i in seq_len(n)) for (j in seq_len(n)) adjacent[i, j] <- edge(chain, i, j)
284 power <- adjacent
285 for (exponent in seq_len((n - 1)^2 + 1)) {
286 if (all(power)) return(exponent)
287 power <- (power %*% adjacent) > 0
288 }
289 NULL
290}

321execute_command <- function(command, a) {
322 if (command == "CHAIN") {
323 name <- a[1]
324 states <- a[-1]
325 if (length(states) > 10) command_error("TOO MANY STATES")
326 seen <- character()
327 for (state in states) {
328 if (state %in% seen) command_error(paste("DUPLICATE STATE", state))
329 seen <- c(seen, state)
330 }
331 put_chain(name, create_chain(states))
332 emit("CHAIN", name, paste0("states=", length(states)))
333 return(invisible(NULL))
334 }
335 if (command == "BUILD") {
336 name <- a[1]
337 order <- as.numeric(a[2])
338 tokens <- a[-c(1, 2)]
339 if (!(order %in% c(1, 2))) command_error("BAD ORDER")
340 if (length(tokens) <= order) command_error("TOO FEW TOKENS")
341 observations <- if (order == 1) tokens else paste(head(tokens, -1), tail(tokens, -1), sep = ".")
342 states <- unique(observations)
343 if (length(states) > 10) command_error("TOO MANY STATES")
344 chain <- create_chain(states)
345 n <- chain_size(chain)
346 counts <- matrix(0, n, n)
347 for (t in seq_len(length(observations) - 1)) {
348 i <- match(observations[t], states)
349 j <- match(observations[t + 1], states)
350 counts[i, j] <- counts[i, j] + 1
351 }
352 emit("BUILD", name, paste0("order=", order), paste0("states=", n),
353 paste0("transitions=", length(observations) - 1))
354 for (i in seq_len(n)) {
355 total <- sum(counts[i, ])
356 if (total == 0) {
357 chain$matrix[[i]][[i]] <- one
358 emit("FROM", states[i], "total=0", "absorbing")
359 } else {
360 parts <- character()
361 for (j in seq_len(n)) {
362 if (counts[i, j] > 0) {
363 chain$matrix[[i]][[j]] <- fraction(counts[i, j], total)
364 parts <- c(parts, paste0(states[j], ":", counts[i, j]))
365 }
366 }
367 emit("FROM", states[i], paste0("total=", total), parts)
368 }
369 }
370 put_chain(name, chain)
371 return(invisible(NULL))
372 }
373 if (command == "GAMBLER") {
374 name <- a[1]
375 goal <- as.numeric(a[2])
376 start <- as.numeric(a[3])
377 if (goal == 0 || goal > 9) command_error("BAD SIZE")
378 if (start > goal) command_error("BAD START")
379 p <- probability(a[4])
380 if (equal(p, zero) || equal(p, one)) command_error("BAD PROBABILITY")
381 chain <- create_chain(paste0("s", 0:goal))
382 chain$matrix[[1]][[1]] <- one
383 chain$matrix[[goal + 1]][[goal + 1]] <- one
384 if (goal > 1) {
385 for (i in 2:goal) {
386 chain$matrix[[i]][[i + 1]] <- p
387 chain$matrix[[i]][[i - 1]] <- subtract(one, p)
388 }
389 }
390 put_chain(name, chain)
391 result <- absorption(chain)
392 ruin <- c(list(one), lapply(result$probabilities, function(x) x[[1]]), list(zero))
393 duration <- c(list(zero), result$steps, list(zero))
394 emit("GAMBLER", name, paste0("g=", goal), paste0("p=", fraction_text(p)),
395 paste0("start=s", start), paste0("ruin=", value_text(ruin[[start + 1]])),
396 paste0("duration=", value_text(duration[[start + 1]])))
397 for (i in 0:goal) {
398 emit("CAPITAL", paste0("s", i), paste0("ruin=", value_text(ruin[[i + 1]])),
399 paste0("win=", value_text(subtract(one, ruin[[i + 1]]))),
400 paste0("duration=", value_text(duration[[i + 1]])))
401 }
402 return(invisible(NULL))
403 }
404
405 chain <- if (command %in% c("P", "ROW", "SHOW", "CHECK"))
406 require_chain(a[1]) else require_stochastic(a[1])
407 n <- chain_size(chain)
408 name <- a[1]
409
410 if (command == "P") {
411 i <- state_index(chain, a[2])
412 j <- state_index(chain, a[3])
413 value <- probability(a[4])
414 chain$matrix[[i]][[j]] <- value
415 emit("P", name, a[2], a[3], fraction_text(value),
416 paste0("rowsum=", fraction_text(row_sum(chain, i))))
417 } else if (command == "ROW") {
418 i <- state_index(chain, a[2])
419 if (length(a) - 2 != n) command_error("BAD LENGTH")
420 values <- lapply(a[-c(1, 2)], probability)
421 chain$matrix[[i]] <- values
422 emit("ROW", name, a[2], paste0("rowsum=", fraction_text(row_sum(chain, i))))
423 } else if (command == "SHOW") {
424 emit("MATRIX", name, paste0("states=", n))
425 display_matrix(chain$names, chain$matrix)
426 } else if (command == "CHECK") {
427 bad <- which(!vapply(seq_len(n), function(i) equal(row_sum(chain, i), one), logical(1)))
428 if (!length(bad)) emit("CHECK", name, "ok") else {
429 emit("CHECK", name, paste0("bad=", length(bad)))
430 for (i in bad) emit("BADROW", chain$names[i], paste0("sum=", fraction_text(row_sum(chain, i))))
431 }
432 } else if (command == "CLASSES") {
433 decomp <- decompose(chain)
434 emit("CLASSES", name, paste0("count=", length(decomp$classes)))
435 for (item in decomp$classes) {
436 emit("CLASS", item$number, if (item$closed) "RECURRENT" else "TRANSIENT",
437 paste0("period=", item$period), paste0("size=", length(item$members)),
438 labels(chain, item$members))
439 if (item$period >= 2) {
440 for (remainder in 0:(item$period - 1)) {
441 members <- item$members[item$levels[item$members] %% item$period == remainder]
442 emit("CYCLIC", remainder, labels(chain, members))
443 }
444 }
445 }
446 } else if (command == "CLASSIFY") {
447 decomp <- decompose(chain)
448 absorbers <- vapply(seq_len(n), function(i) absorbing_state(chain, i), logical(1))
449 count <- sum(absorbers)
450 is_absorbing <- count > 0 && all(vapply(seq_len(n), function(i)
451 reaches_targets(decomp, i, absorbers), logical(1)))
452 exponent <- regular_exponent(chain)
453 emit("CLASSIFY", name, paste0("absorbing_states=", count),
454 paste0("absorbing=", yesno(is_absorbing)),
455 paste0("ergodic=", yesno(length(decomp$classes) == 1)),
456 paste0("regular=", yesno(!is.null(exponent))),
457 paste0("exponent=", if (is.null(exponent)) "-" else exponent))
458 } else if (command == "STEP") {
459 initial <- a[2]
460 steps <- as.numeric(a[3])
461 if (grepl("^[0-9]", initial)) {
462 texts <- strsplit(initial, ",", fixed = TRUE)[[1]]
463 if (length(texts) != n) command_error("BAD LENGTH")
464 distribution <- lapply(texts, probability)
465 if (!equal(sum_fractions(distribution), one)) command_error("BAD DISTRIBUTION")
466 } else {
467 state <- state_index(chain, initial)
468 distribution <- lapply(seq_len(n), function(i) delta(state, i))
469 }
470 if (steps > 0) for (t in seq_len(steps)) distribution <- advance(chain, distribution)
471 emit("STEP", name, paste0("k=", steps))
472 for (i in seq_len(n))
473 emit("PROB", chain$names[i], value_text(distribution[[i]]))
474 } else if (command == "POWER") {
475 steps <- as.numeric(a[2])
476 mat <- identity_matrix(n)
477 if (steps > 0) for (t in seq_len(steps))
478 mat <- lapply(mat, function(row) advance(chain, row))
479 emit("POWER", name, paste0("k=", steps))
480 display_matrix(chain$names, mat)
481 } else if (command == "STATIONARY") {
482 closed <- closed_classes(decompose(chain))
483 emit("STATIONARY", name, paste0("count=", length(closed)))
484 for (i in seq_along(closed)) {
485 item <- closed[[i]]
486 emit("DIST", i, paste0("class=", item$number),
487 paste0("size=", length(item$members)))
488 values <- stationary_distribution(chain, item$members)
489 for (j in seq_along(item$members))
490 emit("PI", chain$names[item$members[j]], value_text(values[[j]]))
491 }
492 } else if (command == "RECURRENCE") {
493 i <- state_index(chain, a[2])
494 decomp <- decompose(chain)
495 item <- decomp$classes[[decomp$class_of[i]]]
496 if (!item$closed) command_error(paste("TRANSIENT STATE", a[2]))
497 pi <- stationary_distribution(chain, item$members)
498 rho <- pi[[match(i, item$members)]]
499 emit("RECURRENCE", name, a[2], paste0("pi=", value_text(rho)),
500 paste0("mean=", value_text(divide(one, rho))),
501 paste0("period=", item$period))
502 } else if (command == "ABSORB") {
503 absorbers <- vapply(seq_len(n), function(i) absorbing_state(chain, i), logical(1))
504 if (!any(absorbers)) command_error("NO ABSORBING STATE")
505 decomp <- decompose(chain)
506 for (i in seq_len(n)) if (!absorbers[i] && !reaches_targets(decomp, i, absorbers))
507 command_error(paste("NOT ABSORBING", chain$names[i]))
508 result <- absorption(chain)
509 emit("ABSORB", name, paste0("transient=", length(result$transient)),
510 paste0("absorbing=", length(result$absorbing)))
511 if (length(result$transient)) {
512 emit("TRANSIENT", labels(chain, result$transient))
513 emit("ABSORBING", labels(chain, result$absorbing))
514 for (i in seq_along(result$transient))
515 emit("N", chain$names[result$transient[i]],
516 vapply(result$fundamental[[i]], fraction_text, character(1)))
517 for (i in seq_along(result$transient))
518 emit("STEPS", chain$names[result$transient[i]],
519 paste0("mean=", value_text(result$steps[[i]])),
520 paste0("var=", value_text(absorption_variance(result, i))))
521 for (i in seq_along(result$transient))
522 emit("B", chain$names[result$transient[i]],
523 vapply(result$probabilities[[i]], fraction_text, character(1)))
524 }
525 } else if (command == "HITTING") {
526 targets <- member_mask(n, distinct_states(chain, a[-1]))
527 decomp <- decompose(chain)
528 probabilities <- hitting_probabilities(chain, decomp, targets)
529 times <- hitting_times(chain, targets, probabilities)
530 emit("HITTING", name, paste0("targets=", paste(chain$names[targets], collapse = ",")))
531 for (i in seq_len(n))
532 emit("HIT", chain$names[i], paste0("prob=", value_text(probabilities[[i]])),
533 paste0("time=", time_text(times[[i]])))
534 } else if (command == "PASSAGE") {
535 decomp <- decompose(chain)
536 pi <- irreducible_distribution(chain, decomp)
537 times <- lapply(seq_len(n), function(i) replicate(n, zero, simplify = FALSE))
538 for (j in seq_len(n)) {
539 targets <- member_mask(n, j)
540 probabilities <- hitting_probabilities(chain, decomp, targets)
541 column <- hitting_times(chain, targets, probabilities)
542 for (i in seq_len(n))
543 times[[i]][[j]] <- if (i == j) divide(one, pi[[j]]) else column[[i]]
544 }
545 kemeny <- zero
546 if (n > 1) for (j in 2:n)
547 kemeny <- add(kemeny, multiply(pi[[j]], times[[1]][[j]]))
548 emit("PASSAGE", name, paste0("kemeny=", value_text(kemeny)))
549 display_matrix(chain$names, times, "M")
550 } else if (command == "FIRSTPASSAGE") {
551 x <- state_index(chain, a[2])
552 y <- state_index(chain, a[3])
553 steps <- as.numeric(a[4])
554 if (steps == 0) command_error("BAD STEPS")
555 current <- lapply(seq_len(n), function(i) chain$matrix[[i]][[y]])
556 first <- list(current[[x]])
557 if (steps > 1) for (t in 2:steps) {
558 current <- lapply(seq_len(n), function(i) sum_fractions(lapply(
559 setdiff(seq_len(n), y), function(j)
560 multiply(chain$matrix[[i]][[j]], current[[j]]))))
561 first[[t]] <- current[[x]]
562 }
563 emit("FIRSTPASSAGE", name, a[2], a[3], paste0("k=", steps),
564 paste0("total=", value_text(sum_fractions(first))))
565 cumulative <- zero
566 for (t in seq_along(first)) {
567 cumulative <- add(cumulative, first[[t]])
568 emit("FP", t, value_text(first[[t]]), paste0("cum=", value_text(cumulative)))
569 }
570 } else if (command == "PATHPROB") {
571 indices <- vapply(a[-1], function(text) state_index(chain, text), integer(1))
572 prob <- one
573 zero_at <- NULL
574 if (length(indices) > 1) for (t in seq_len(length(indices) - 1)) {
575 prob <- multiply(prob, chain$matrix[[indices[t]]][[indices[t + 1]]])
576 if (is.null(zero_at) && equal(prob, zero)) zero_at <- t
577 }
578 parts <- c("PATHPROB", name, paste0("steps=", length(indices) - 1),
579 paste0("prob=", value_text(prob)))
580 if (!is.null(zero_at)) parts <- c(parts, paste0("zero_at=", zero_at))
581 emit(parts)
582 } else if (command == "MOSTLIKELY") {
583 x <- state_index(chain, a[2])
584 steps <- as.numeric(a[3])
585 if (steps == 0) command_error("BAD STEPS")
586 best <- lapply(seq_len(steps + 1), function(i) rep(list(one), n))
587 for (t in steps:1) {
588 best[[t]] <- lapply(seq_len(n), function(i) {
589 choices <- lapply(seq_len(n), function(j)
590 multiply(chain$matrix[[i]][[j]], best[[t + 1]][[j]]))
591 maximum <- choices[[1]]
592 for (value in choices) if (greater(value, maximum)) maximum <- value
593 maximum
594 })
595 }
596 path <- c(x)
597 for (t in seq_len(steps)) {
598 current <- tail(path, 1)
599 for (j in seq_len(n)) {
600 candidate <- multiply(chain$matrix[[current]][[j]], best[[t + 1]][[j]])
601 if (equal(candidate, best[[t]][[current]])) {
602 path <- c(path, j)
603 break
604 }
605 }
606 }
607 emit("MOSTLIKELY", name, a[2], paste0("k=", steps),
608 paste0("prob=", value_text(best[[1]][[x]])))
609 emit("PATH", labels(chain, path))
610 } else if (command == "REVERSIBLE" || command == "REVERSE") {
611 decomp <- decompose(chain)
612 pi <- irreducible_distribution(chain, decomp)
613 if (command == "REVERSIBLE") {
614 violations <- list()
615 if (n > 1) for (i in seq_len(n - 1)) for (j in (i + 1):n) {
616 forward <- multiply(pi[[i]], chain$matrix[[i]][[j]])
617 backward <- multiply(pi[[j]], chain$matrix[[j]][[i]])
618 if (!equal(forward, backward))
619 violations[[length(violations) + 1]] <- list(i, j, forward, backward)
620 }
621 if (!length(violations)) emit("REVERSIBLE", name, "yes") else {
622 emit("REVERSIBLE", name, "no", paste0("violations=", length(violations)))
623 for (v in violations)
624 emit("VIOLATION", chain$names[v[[1]]], chain$names[v[[2]]],
625 value_text(v[[3]]), value_text(v[[4]]))
626 }
627 } else {
628 reversed_name <- a[2]
629 reversed <- create_chain(chain$names)
630 same <- TRUE
631 for (i in seq_len(n)) for (j in seq_len(n)) {
632 val <- divide(multiply(pi[[j]], chain$matrix[[j]][[i]]), pi[[i]])
633 reversed$matrix[[i]][[j]] <- val
634 if (!equal(val, chain$matrix[[i]][[j]])) same <- FALSE
635 }
636 put_chain(reversed_name, reversed)
637 emit("REVERSE", name, reversed_name, paste0("reversible=", yesno(same)))
638 display_matrix(reversed$names, reversed$matrix)
639 }
640 } else if (command == "MIXING") {
641 steps <- as.numeric(a[2])
642 if (steps == 0) command_error("BAD STEPS")
643 epsilon <- probability(a[3])
644 pi <- irreducible_distribution(chain, decompose(chain))
645 mat <- identity_matrix(n)
646 records <- list()
647 mixing_time <- NULL
648 for (t in seq_len(steps)) {
649 mat <- lapply(mat, function(row) advance(chain, row))
650 distances <- lapply(mat, function(row)
651 divide(sum_fractions(Map(function(x, y) absolute(subtract(x, y)), row, pi)),
652 fraction(2)))
653 largest <- distances[[1]]
654 worst <- 1
655 for (i in seq_len(n)) if (greater(distances[[i]], largest)) {
656 largest <- distances[[i]]
657 worst <- i
658 }
659 if (is.null(mixing_time) && !greater(largest, epsilon)) mixing_time <- t
660 records[[t]] <- list(largest, worst)
661 }
662 emit("MIXING", name, paste0("k=", steps),
663 paste0("eps=", fraction_text(epsilon)),
664 paste0("time=", if (is.null(mixing_time)) "none" else mixing_time))
665 for (t in seq_len(steps))
666 emit("TV", t, value_text(records[[t]][[1]]),
667 paste0("from=", chain$names[records[[t]][[2]]]))
668 } else if (command == "LUMP") {
669 new_name <- a[2]
670 groups <- strsplit(a[3], "|", fixed = TRUE)[[1]]
671 seen <- rep(FALSE, n)
672 blocks <- list()
673 for (group in groups) {
674 block <- integer()
675 for (text in strsplit(group, ",", fixed = TRUE)[[1]]) {
676 position <- state_index(chain, text)
677 if (seen[position]) command_error("BAD PARTITION")
678 seen[position] <- TRUE
679 block <- c(block, position)
680 }
681 blocks[[length(blocks) + 1]] <- sort(block)
682 }
683 if (!all(seen)) command_error("BAD PARTITION")
684 blocks <- blocks[order(vapply(blocks, function(block) block[1], numeric(1)))]
685 for (block in blocks) {
686 representative <- block[1]
687 for (target in blocks) {
688 expected <- block_sum(chain, representative, target)
689 for (i in block) {
690 actual <- block_sum(chain, i, target)
691 if (!equal(actual, expected)) {
692 emit("LUMP", name, "not", "lumpable",
693 paste0("block=", chain$names[representative]),
694 paste0("target=", chain$names[target[1]]),
695 paste0(chain$names[representative], "=", value_text(expected)),
696 paste0(chain$names[i], "=", value_text(actual)))
697 return(invisible(NULL))
698 }
699 }
700 }
701 }
702 lumped <- create_chain(vapply(blocks, function(block)
703 chain$names[block[1]], character(1)))
704 lumped$matrix <- lapply(blocks, function(block)
705 lapply(blocks, function(target) block_sum(chain, block[1], target)))
706 put_chain(new_name, lumped)
707 emit("LUMP", name, new_name, paste0("blocks=", length(blocks)))
708 display_matrix(lumped$names, lumped$matrix)
709 } else if (command == "REWARD") {
710 texts <- a[-c(1, 2)]
711 if (length(texts) != n) command_error("BAD LENGTH")
712 gamma <- probability(a[2])
713 if (equal(gamma, one)) command_error("BAD DISCOUNT")
714 rewards <- lapply(texts, function(text) fraction(as.numeric(text)))
715 equations <- lapply(seq_len(n), function(i) {
716 left <- lapply(seq_len(n), function(j)
717 subtract(delta(i, j), multiply(gamma, chain$matrix[[i]][[j]])))
718 c(left, list(rewards[[i]]))
719 })
720 discounted <- lapply(eliminate(equations, n), function(row) row[[1]])
721 decomp <- decompose(chain)
722 gains <- rep(list(zero), n)
723 for (item in closed_classes(decomp)) {
724 pi <- stationary_distribution(chain, item$members)
725 gain <- sum_fractions(Map(function(rho, j)
726 multiply(rho, rewards[[j]]), pi, item$members))
727 targets <- member_mask(n, item$members)
728 hit <- hitting_probabilities(chain, decomp, targets)
729 for (i in seq_len(n)) gains[[i]] <- add(gains[[i]], multiply(hit[[i]], gain))
730 }
731 emit("REWARD", name, paste0("gamma=", fraction_text(gamma)))
732 for (i in seq_len(n)) {
733 emit("VALUE", chain$names[i],
734 paste0("discounted=", value_text(discounted[[i]])),
735 paste0("gain=", value_text(gains[[i]])))
736 }
737 } else if (command == "RESTRICT") {
738 new_name <- a[2]
739 inside <- member_mask(n, distinct_states(chain, a[-c(1, 2)]))
740 for (i in seq_len(n)) {
741 if (inside[i] && any(vapply(seq_len(n), function(j)
742 edge(chain, i, j) && !inside[j], logical(1))))
743 command_error(paste("NOT CLOSED", chain$names[i]))
744 }
745 members <- which(inside)
746 restricted <- create_chain(chain$names[members])
747 restricted$matrix <- lapply(members, function(i)
748 lapply(members, function(j) chain$matrix[[i]][[j]]))
749 put_chain(new_name, restricted)
750 emit("RESTRICT", name, new_name, paste0("states=", length(members)))
751 display_matrix(restricted$names, restricted$matrix)
752 } else if (command == "SIMULATE") {
753 start <- state_index(chain, a[2])
754 steps <- as.numeric(a[3])
755 seed <- as.numeric(a[4])
756 if (steps == 0) command_error("BAD STEPS")
757 if (seed == 0 || seed >= 2147483647) command_error("BAD SEED")
758 random_value <- seed
759 path <- c(start)
760 for (t in seq_len(steps)) {
761 row <- chain$matrix[[tail(path, 1)]]
762 common <- 1
763 for (p in row) common <- common / gcd_exact(common, p[2]) * p[2]
764 random_value <- (48271 * random_value) %% 2147483647
765 threshold <- random_value %% common
766 cumulative <- zero
767 for (j in seq_len(n)) {
768 cumulative <- add(cumulative, row[[j]])
769 if (greater(multiply(cumulative, fraction(common)), fraction(threshold))) {
770 path <- c(path, j)
771 break
772 }
773 }
774 }
775 emit("SIMULATE", name, a[2], paste0("steps=", steps),
776 paste0("seed=", seed), paste0("end=", chain$names[tail(path, 1)]))
777 emit("PATH", labels(chain, path))
778 for (i in seq_len(n)) emit("VISIT", chain$names[i], sum(path == i))
779 } else {
780 command_error(paste("UNKNOWN COMMAND", command))
781 }
782 invisible(NULL)
783}
