10class Fraction implements Comparable<Fraction> {
11 final int numerator;
12 final int denominator;
13 const Fraction._(this.numerator, this.denominator);
14 static const zero = Fraction._(0, 1);
15 static const one = Fraction._(1, 1);
16 static const two = Fraction._(2, 1);
17
18 factory Fraction(int numerator, [int denominator = 1]) {
19 if (denominator < 0) {
20 numerator = -numerator;
21 denominator = -denominator;
22 }
23 final divisor = numerator.abs().gcd(denominator);
24 return Fraction._(numerator ~/ divisor, denominator ~/ divisor);
25 }
26
27 Fraction operator +(Fraction other) => Fraction(
28 numerator * other.denominator + other.numerator * denominator,
29 denominator * other.denominator);
30 Fraction operator -(Fraction other) => Fraction(
31 numerator * other.denominator - other.numerator * denominator,
32 denominator * other.denominator);
33 Fraction operator *(Fraction other) =>
34 Fraction(numerator * other.numerator, denominator * other.denominator);
35 Fraction operator /(Fraction other) =>
36 Fraction(numerator * other.denominator, denominator * other.numerator);
37 Fraction get absolute => Fraction(numerator.abs(), denominator);
38 bool get isZero => numerator == 0;
39
40 @override
41 int compareTo(Fraction other) =>
42 (numerator * other.denominator).compareTo(other.numerator * denominator);
43 @override
44 bool operator ==(Object other) => other is Fraction &&
45 numerator == other.numerator && denominator == other.denominator;
46 @override
47 int get hashCode => Object.hash(numerator, denominator);
48 @override
49 String toString() => '$numerator/$denominator';
50
51 String get value => '$this(${numerator ~/ denominator}.'
52 '${(numerator % denominator * 1000000 ~/ denominator).toString().padLeft(6, '0')})';
53}
54
55Fraction probability(String text) {
56 final parts = text.split('/');
57 final numerator = int.parse(parts[0]);
58 final denominator = parts.length == 1 ? 1 : int.parse(parts[1]);
59 if (denominator == 0 || numerator > denominator) fail('BAD PROBABILITY');
60 return Fraction(numerator, denominator);
61}

63typedef Matrix = List<List<Fraction>>;
64
65Fraction total(Iterable<Fraction> values) =>
66 values.fold(Fraction.zero, (sum, value) => sum + value);
67
68Matrix matrix(int rows, int columns) =>
69 List.generate(rows, (_) => List.filled(columns, Fraction.zero));
70
71Fraction delta(int i, int j) => i == j ? Fraction.one : Fraction.zero;
72
73Matrix identity(int size) => List.generate(size,
74 (i) => List.generate(size, (j) => delta(i, j)));
75
76List<Fraction> advance(List<Fraction> distribution, Matrix transitions) =>
77 List.generate(distribution.length, (j) => total(List.generate(
78 distribution.length, (i) => distribution[i] * transitions[i][j])));
79
80Matrix multiply(Matrix a, Matrix b) => a.map((row) => advance(row, b)).toList();

82Matrix solve(Matrix augmented) {
83 final size = augmented.length;
84 if (size == 0) return [];
85 final width = augmented[0].length;
86 for (var column = 0; column < size; column++) {
87 var pivotRow = column;
88 while (augmented[pivotRow][column].isZero) {
89 pivotRow++;
90 }
91 final swap = augmented[column];
92 augmented[column] = augmented[pivotRow];
93 augmented[pivotRow] = swap;
94 final pivot = augmented[column][column];
95 for (var j = 0; j < width; j++) {
96 augmented[column][j] = augmented[column][j] / pivot;
97 }
98 for (var row = 0; row < size; row++) {
99 if (row == column) continue;
100 final factor = augmented[row][column];
101 if (factor.isZero) continue;
102 for (var j = 0; j < width; j++) {
103 augmented[row][j] = augmented[row][j] - factor * augmented[column][j];
104 }
105 }
106 }
107 return augmented.map((row) => row.sublist(size)).toList();
108}

171 Graph graph() {
172 final reach = List.generate(size,
173 (i) => List.generate(size, (j) => i == j || !p[i][j].isZero));
174 for (var k = 0; k < size; k++) {
175 for (var i = 0; i < size; i++) {
176 for (var j = 0; j < size; j++) {
177 reach[i][j] = reach[i][j] || reach[i][k] && reach[k][j];
178 }
179 }
180 }
181 final assigned = <int>{};
182 final classes = <Component>[];
183 for (var first = 0; first < size; first++) {
184 if (assigned.contains(first)) continue;
185 final states = indices.where((j) => reach[first][j] && reach[j][first]).toList();
186 final members = states.toSet();
187 assigned.addAll(states);
188 final closed = states.every((i) => indices.every(
189 (j) => members.contains(j) || p[i][j].isZero));
190 final levels = List<int>.filled(size, -1);
191 levels[first] = 0;
192 final queue = [first];
193 for (var head = 0; head < queue.length; head++) {
194 final i = queue[head];
195 for (final j in states) {
196 if (levels[j] < 0 && !p[i][j].isZero) {
197 levels[j] = levels[i] + 1;
198 queue.add(j);
199 }
200 }
201 }
202 var period = 0;
203 for (final i in states) {
204 for (final j in states) {
205 if (!p[i][j].isZero) period = period.gcd((levels[i] + 1 - levels[j]).abs());
206 }
207 }
208 classes.add(Component(states, closed, period, levels));
209 }
210 return Graph(reach, classes);
211 }

213 List<Fraction> stationary(List<int> states) {
214 final count = states.length;
215 final equations = matrix(count, count + 1);
216 for (var row = 0; row < count - 1; row++) {
217 for (var col = 0; col < count; col++) {
218 equations[row][col] = p[states[col]][states[row]] - delta(col, row);
219 }
220 }
221 equations[count - 1] = List.filled(count + 1, Fraction.one);
222 final solution = solve(equations);
223 final distribution = List.filled(size, Fraction.zero);
224 for (var i = 0; i < count; i++) {
225 distribution[states[i]] = solution[i][0];
226 }
227 return distribution;
228 }

232 if (classes.length != 1) fail('NOT IRREDUCIBLE');
233 return stationary(classes.single.states);
234 }

236 Hitting hitting(List<int> targets, {bool withTimes = true}) {
237 final reach = graph().reach;
238 final targetSet = targets.toSet();
239 final unknown = indices.where((i) => !targetSet.contains(i) &&
240 targets.any((j) => reach[i][j])).toList();
241 final probability = List.filled(size, Fraction.zero);
242 final time = List<Fraction?>.filled(size, null);
243 for (final target in targets) {
244 probability[target] = Fraction.one;
245 time[target] = Fraction.zero;
246 }
247 if (unknown.isNotEmpty) {
248 final equations = List.generate(unknown.length, (row) => [
249 for (var col = 0; col < unknown.length; col++)
250 delta(row, col) - p[unknown[row]][unknown[col]],
251 total(targets.map((j) => p[unknown[row]][j])),
252 ]);
253 final solution = solve(equations);
254 for (var i = 0; i < unknown.length; i++) {
255 probability[unknown[i]] = solution[i][0];
256 }
257 }
258 final certain = unknown.where((i) => probability[i] == Fraction.one).toList();
259 if (withTimes && certain.isNotEmpty) {
260 final equations = List.generate(certain.length, (row) => [
261 for (var col = 0; col < certain.length; col++)
262 delta(row, col) - p[certain[row]][certain[col]],
263 Fraction.one,
264 ]);
265 final solution = solve(equations);
266 for (var i = 0; i < certain.length; i++) {
267 time[certain[i]] = solution[i][0];
268 }
269 }
270 return Hitting(probability, time);
271 }
272
273 Absorption absorption() {
274 final absorbing = indices.where((i) => p[i][i] == Fraction.one).toList();
275 if (absorbing.isEmpty) fail('NO ABSORBING STATE');
276 final transient = indices.where((i) => !absorbing.contains(i)).toList();
277 final reach = graph().reach;
278 for (final i in transient) {
279 if (!absorbing.any((j) => reach[i][j])) fail('NOT ABSORBING ${names[i]}');
280 }
281 final count = transient.length;
282 final equations = List.generate(count, (row) => [
283 for (var col = 0; col < count; col++) delta(row, col) - p[transient[row]][transient[col]],
284 for (var col = 0; col < count; col++) delta(row, col),
285 ]);
286 final fundamental = solve(equations);
287 final mean = fundamental.map(total).toList();
288 final variance = List.generate(count, (i) => total(List.generate(count,
289 (j) => (Fraction.two * fundamental[i][j] - delta(i, j)) * mean[j])) - mean[i] * mean[i]);
290 final chances = List.generate(count, (i) => List.generate(absorbing.length,
291 (j) => total(List.generate(count, (k) => fundamental[i][k] * p[transient[k]][absorbing[j]]))));
292 return Absorption(transient, absorbing, fundamental, chances, mean, variance);
293 }

320 void build(List<String> words) {
321 final name = words[1];
322 final order = int.parse(words[2]);
323 if (order != 1 && order != 2) fail('BAD ORDER');
324 final tokens = words.sublist(3);
325 if (tokens.length <= order) fail('TOO FEW TOKENS');
326 final observations = order == 1 ? tokens : List.generate(tokens.length - 1,
327 (i) => '${tokens[i]}.${tokens[i + 1]}');
328 final names = <String>[];
329 final seen = <String>{};
330 for (final observation in observations) {
331 if (seen.add(observation)) names.add(observation);
332 }
333 if (names.length > 10) fail('TOO MANY STATES');
334 final count = names.length;
335 final counts = List.generate(count, (_) => List<int>.filled(count, 0));
336 for (var i = 0; i + 1 < observations.length; i++) {
337 counts[names.indexOf(observations[i])][names.indexOf(observations[i + 1])]++;
338 }
339 final totals = counts.map((row) => row.fold(0, (sum, value) => sum + value)).toList();
340 final p = List.generate(count, (i) => List.generate(count,
341 (j) => totals[i] == 0 ? delta(i, j) : Fraction(counts[i][j], totals[i])));
342 chains[name] = Chain(names, p);
343 emit('BUILD $name order=$order states=$count transitions=${observations.length - 1}');
344 for (var i = 0; i < count; i++) {
345 final transitions = [for (var j = 0; j < count; j++)
346 if (counts[i][j] > 0) '${names[j]}:${counts[i][j]}'];
347 emit('FROM ${names[i]} total=${totals[i]} ${totals[i] == 0 ? 'absorbing' : transitions.join(' ')}');
348 }

351 void gambler(List<String> words) {
352 final name = words[1];
353 final goal = int.parse(words[2]);
354 final start = int.parse(words[3]);
355 if (goal == 0 || goal > 9) fail('BAD SIZE');
356 if (start > goal) fail('BAD START');
357 final chance = probability(words[4]);
358 if (chance == Fraction.zero || chance == Fraction.one) fail('BAD PROBABILITY');
359 final p = matrix(goal + 1, goal + 1);
360 p[0][0] = p[goal][goal] = Fraction.one;
361 for (var i = 1; i < goal; i++) {
362 p[i][i + 1] = chance;
363 p[i][i - 1] = Fraction.one - chance;
364 }
365 final chain = Chain(List.generate(goal + 1, (i) => 's$i'), p);
366 final absorption = chain.absorption();
367 final ruin = List.filled(goal + 1, Fraction.zero);
368 final duration = List.filled(goal + 1, Fraction.zero);
369 ruin[0] = Fraction.one;
370 for (var i = 0; i < absorption.transient.length; i++) {
371 ruin[absorption.transient[i]] = absorption.probability[i][0];
372 duration[absorption.transient[i]] = absorption.mean[i];
373 }
374 chains[name] = chain;
375 emit('GAMBLER $name g=$goal p=$chance start=s$start ruin=${ruin[start].value} duration=${duration[start].value}');
376 for (var i = 0; i <= goal; i++) {
377 emit('CAPITAL s$i ruin=${ruin[i].value} win=${(Fraction.one - ruin[i]).value} duration=${duration[i].value}');
378 }
379 }

397 final graph = chain.graph();
398 final absorbing = chain.indices.where((i) => chain.p[i][i] == Fraction.one).toList();
399 final isAbsorbing = absorbing.isNotEmpty && chain.indices.every(
400 (i) => absorbing.any((j) => graph.reach[i][j]));
401 final edges = List.generate(chain.size,
402 (i) => List.generate(chain.size, (j) => !chain.p[i][j].isZero));

409 break;

412 (j) => chain.indices.any((k) => power[i][k] && edges[k][j])));
413 }
414 emit('CLASSIFY $name absorbing_states=${absorbing.length} absorbing=${isAbsorbing ? 'yes' : 'no'} '
415 'ergodic=${graph.classes.length == 1 ? 'yes' : 'no'} regular=${exponent == null ? 'no' : 'yes'} exponent=${exponent ?? '-'}');
416 }

485 void firstPassage(String name, Chain chain, String from, String to, int steps) {
486 final x = chain.state(from);
487 final y = chain.state(to);
488 if (steps == 0) fail('BAD STEPS');
489 var previous = chain.indices.map((i) => chain.p[i][y]).toList();
490 final values = <Fraction>[];
491 final cumulative = <Fraction>[];
492 var sum = Fraction.zero;
493 for (var t = 1; t <= steps; t++) {
494 if (t > 1) {
495 previous = List.generate(chain.size, (i) => total(chain.indices
496 .where((j) => j != y).map((j) => chain.p[i][j] * previous[j])));
497 }
498 values.add(previous[x]);
499 sum = sum + previous[x];
500 cumulative.add(sum);
501 }
502 emit('FIRSTPASSAGE $name $from $to k=$steps total=${sum.value}');
503 for (var t = 0; t < steps; t++) {
504 emit('FP ${t + 1} ${values[t].value} cum=${cumulative[t].value}');
505 }

508 void mostLikely(String name, Chain chain, String from, int steps) {
509 final start = chain.state(from);
510 if (steps == 0) fail('BAD STEPS');
511 final best = matrix(steps + 1, chain.size);
512 best[steps] = List.filled(chain.size, Fraction.one);
513 for (var t = steps - 1; t >= 0; t--) {
514 for (var i = 0; i < chain.size; i++) {
515 for (var j = 0; j < chain.size; j++) {
516 final candidate = chain.p[i][j] * best[t + 1][j];
517 if (candidate.compareTo(best[t][i]) > 0) best[t][i] = candidate;
518 }
519 }
520 }
521 final path = [start];
522 for (var t = 0; t < steps; t++) {
523 final current = path.last;
524 path.add(chain.indices.firstWhere(
525 (j) => chain.p[current][j] * best[t + 1][j] == best[t][current]));
526 }
527 emit('MOSTLIKELY $name $from k=$steps prob=${best[0][start].value}');
528 emit('PATH ${chain.labels(path)}');
529 }

557 void mixing(String name, Chain chain, int steps, String epsilon) {
558 if (steps == 0) fail('BAD STEPS');
559 final eps = probability(epsilon);
560 final pi = chain.uniqueStationary();
561 var power = identity(chain.size);
562 final rows = <({Fraction distance, int state})>[];
563 int? time;
564 for (var t = 1; t <= steps; t++) {
565 power = multiply(power, chain.p);
566 var distance = Fraction.zero;
567 var worst = 0;
568 for (var i = 0; i < chain.size; i++) {
569 final value = total(chain.indices.map((j) => (power[i][j] - pi[j]).absolute)) / Fraction.two;
570 if (value.compareTo(distance) > 0) {
571 distance = value;
572 worst = i;
573 }
574 }
575 if (time == null && distance.compareTo(eps) <= 0) time = t;
576 rows.add((distance: distance, state: worst));
577 }
578 emit('MIXING $name k=$steps eps=$eps time=${time ?? 'none'}');
579 for (var t = 0; t < rows.length; t++) {
580 emit('TV ${t + 1} ${rows[t].distance.value} from=${chain.names[rows[t].state]}');
581 }

584 void lump(String name, Chain chain, String destination, String partition) {
585 final seen = <int>{};
586 final blocks = <List<int>>[];
587 for (final block in partition.split('|')) {
588 final states = <int>[];
589 for (final text in block.split(',')) {
590 final state = chain.state(text);
591 if (!seen.add(state)) fail('BAD PARTITION');
592 states.add(state);
593 }
594 states.sort();
595 blocks.add(states);
596 }
597 if (seen.length != chain.size) fail('BAD PARTITION');
598 blocks.sort((a, b) => a.first.compareTo(b.first));
599 final p = matrix(blocks.length, blocks.length);
600 for (var i = 0; i < blocks.length; i++) {
601 final block = blocks[i];
602 final representative = block.first;
603 for (var j = 0; j < blocks.length; j++) {
604 final baseline = total(blocks[j].map((target) => chain.p[representative][target]));
605 for (final state in block) {
606 final value = total(blocks[j].map((target) => chain.p[state][target]));
607 if (value == baseline) continue;
608 emit('LUMP $name not lumpable block=${chain.names[representative]} '
609 'target=${chain.names[blocks[j].first]} ${chain.names[representative]}=${baseline.value} '
610 '${chain.names[state]}=${value.value}');
611 return;
612 }
613 p[i][j] = baseline;
614 }
615 }
616 final names = blocks.map((block) => chain.names[block.first]).toList();
617 chains[destination] = Chain(names, p);
618 emit('LUMP $name $destination blocks=${blocks.length}');
619 showMatrix(names, p);
620 }
621
622 void reward(String name, Chain chain, String discount, List<String> words) {
623 if (words.length != chain.size) fail('BAD LENGTH');
624 final gamma = probability(discount);
625 if (gamma == Fraction.one) fail('BAD DISCOUNT');
626 final rewards = words.map((word) => Fraction(int.parse(word))).toList();
627 final equations = List.generate(chain.size, (i) => [
628 for (var j = 0; j < chain.size; j++) delta(i, j) - gamma * chain.p[i][j],
629 rewards[i],
630 ]);
631 final discounted = solve(equations);
632 final gain = List.filled(chain.size, Fraction.zero);
633 for (final component in chain.graph().classes) {
634 if (!component.closed) continue;
635 final pi = chain.stationary(component.states);
636 final average = total(component.states.map((i) => pi[i] * rewards[i]));
637 final h = chain.hitting(component.states, withTimes: false).probability;
638 for (var i = 0; i < chain.size; i++) {
639 gain[i] = gain[i] + h[i] * average;
640 }
641 }
642 emit('REWARD $name gamma=$gamma');
643 for (var i = 0; i < chain.size; i++) {
644 emit('VALUE ${chain.names[i]} discounted=${discounted[i][0].value} gain=${gain[i].value}');
645 }
646 }
647
648 void restrict(String name, Chain chain, String destination, List<String> words) {
649 final states = chain.stateSet(words);
650 for (final i in states) {
651 if (chain.indices.any((j) => !states.contains(j) && !chain.p[i][j].isZero)) {
652 fail('NOT CLOSED ${chain.names[i]}');
653 }
654 }
655 final names = states.map((i) => chain.names[i]).toList();
656 final p = states.map((i) => states.map((j) => chain.p[i][j]).toList()).toList();
657 chains[destination] = Chain(names, p);
658 emit('RESTRICT $name $destination states=${states.length}');
659 showMatrix(names, p);
660 }

662 void simulate(String name, Chain chain, String from, int steps, int seed) {
663 final start = chain.state(from);
664 if (steps == 0) fail('BAD STEPS');
665 if (seed == 0 || seed >= 2147483647) fail('BAD SEED');
666 var random = seed;
667 final path = [start];
668 final visits = List<int>.filled(chain.size, 0);
669 visits[start]++;
670 for (var t = 0; t < steps; t++) {
671 final row = chain.p[path.last];
672 var denominator = 1;
673 for (final value in row) {
674 denominator = denominator ~/ denominator.gcd(value.denominator) * value.denominator;
675 }
676 random = 48271 * random % 2147483647;
677 final target = random % denominator;
678 var cumulative = Fraction.zero;
679 for (var j = 0; j < chain.size; j++) {
680 cumulative = cumulative + row[j];
681 if (cumulative.numerator * denominator ~/ cumulative.denominator > target) {
682 path.add(j);
683 visits[j]++;
684 break;
685 }
686 }
687 }
688 emit('SIMULATE $name $from steps=$steps seed=$seed end=${chain.names[path.last]}');
689 emit('PATH ${chain.labels(path)}');
690 for (var i = 0; i < chain.size; i++) {
691 emit('VISIT ${chain.names[i]} ${visits[i]}');
692 }
693 }

695 void command(List<String> words) {
696 final op = words[0];
697 final name = words[1];
698 switch (op) {
699 case 'CHAIN': create(words); return;
700 case 'BUILD': build(words); return;
701 case 'GAMBLER': gambler(words); return;
702 }
703 final chain = chains[name] ?? (fail('UNKNOWN CHAIN $name'));
704 if (!const {'P', 'ROW', 'SHOW', 'CHECK'}.contains(op)) chain.stochastic();
705 switch (op) {
706 case 'P': {
707 final x = chain.state(words[2]);
708 final y = chain.state(words[3]);
709 final value = probability(words[4]);
710 chain.p[x][y] = value;
711 emit('P $name ${words[2]} ${words[3]} $value rowsum=${total(chain.p[x])}');
712 break;
713 }
714 case 'ROW': {
715 final x = chain.state(words[2]);
716 if (words.length - 3 != chain.size) fail('BAD LENGTH');
717 final values = words.skip(3).map(probability).toList();
718 chain.p[x] = values;
719 emit('ROW $name ${words[2]} rowsum=${total(values)}');
720 break;
721 }
722 case 'SHOW':
723 emit('MATRIX $name states=${chain.size}');
724 showMatrix(chain.names, chain.p);
725 break;
726 case 'CHECK': {
727 final sums = chain.p.map(total).toList();
728 final bad = chain.indices.where((i) => sums[i] != Fraction.one).toList();
729 emit('CHECK $name ${bad.isEmpty ? 'ok' : 'bad=${bad.length}'}');
730 for (final i in bad) {
731 emit('BADROW ${chain.names[i]} sum=${sums[i]}');
732 }
733 break;
734 }
735 case 'CLASSES': classes(name, chain); break;
736 case 'CLASSIFY': classify(name, chain); break;
737 case 'STEP': step(name, chain, words[2], int.parse(words[3])); break;
738 case 'POWER': {
739 final steps = int.parse(words[2]);
740 var power = identity(chain.size);
741 for (var t = 0; t < steps; t++) {
742 power = multiply(power, chain.p);
743 }
744 emit('POWER $name k=$steps');
745 showMatrix(chain.names, power);
746 break;
747 }
748 case 'STATIONARY': stationary(name, chain); break;
749 case 'RECURRENCE': {
750 final x = chain.state(words[2]);
751 final component = chain.graph().classes.firstWhere((component) => component.states.contains(x));
752 if (!component.closed) fail('TRANSIENT STATE ${words[2]}');
753 final pi = chain.stationary(component.states)[x];
754 emit('RECURRENCE $name ${words[2]} pi=${pi.value} mean=${(Fraction.one / pi).value} period=${component.period}');
755 break;
756 }
757 case 'ABSORB': absorb(name, chain); break;
758 case 'HITTING': {
759 final targets = chain.stateSet(words.skip(2));
760 final result = chain.hitting(targets);
761 emit('HITTING $name targets=${chain.labels(targets, ',')}');
762 for (var i = 0; i < chain.size; i++) {
763 emit('HIT ${chain.names[i]} prob=${result.probability[i].value} time=${result.time[i]?.value ?? 'inf'}');
764 }
765 break;
766 }
767 case 'PASSAGE': passage(name, chain); break;
768 case 'FIRSTPASSAGE': firstPassage(name, chain, words[2], words[3], int.parse(words[4])); break;
769 case 'PATHPROB': {
770 final path = words.skip(2).map(chain.state).toList();
771 var chance = Fraction.one;
772 int? zeroAt;
773 for (var i = 1; i < path.length; i++) {
774 chance = chance * chain.p[path[i - 1]][path[i]];
775 if (chance.isZero) zeroAt ??= i;
776 }
777 emit('PATHPROB $name steps=${path.length - 1} prob=${chance.value}${zeroAt == null ? '' : ' zero_at=$zeroAt'}');
778 break;
779 }
780 case 'MOSTLIKELY': mostLikely(name, chain, words[2], int.parse(words[3])); break;
781 case 'REVERSIBLE': reversible(name, chain); break;
782 case 'REVERSE': reverse(name, chain, words[2]); break;
783 case 'MIXING': mixing(name, chain, int.parse(words[2]), words[3]); break;
784 case 'LUMP': lump(name, chain, words[2], words[3]); break;
785 case 'REWARD': reward(name, chain, words[2], words.sublist(3)); break;
786 case 'RESTRICT': restrict(name, chain, words[2], words.sublist(3)); break;
787 case 'SIMULATE': simulate(name, chain, words[2], int.parse(words[3]), int.parse(words[4])); break;
788 }
789 }
790
791 void execute(String line) {
792 try {
793 command(line.split(' '));
794 } on CommandError catch (error) {
795 emit('ERROR ${error.message}');
796 }
797 }
798}

803 for (var i = 0; i < count; i++) {
804 analyzer.execute(stdin.readLineSync()!);
805 }
806 stdout.write(analyzer.output);
