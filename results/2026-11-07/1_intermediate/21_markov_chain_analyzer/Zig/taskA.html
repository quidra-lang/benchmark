18fn gcd(a: i64, b: i64) i64 {
19 var x = if (a < 0) -a else a;
20 var y = if (b < 0) -b else b;
21 while (y != 0) {
22 const remainder = @mod(x, y);
23 x = y;
24 y = remainder;
25 }
26 return x;
27}

29const Fraction = struct {
30 numerator: i64 = 0,
31 denominator: i64 = 1,
32 const zero: Fraction = .{};
33 const one: Fraction = .{ .numerator = 1 };
34 const two: Fraction = .{ .numerator = 2 };
35
36 fn init(numerator: i64, denominator: i64) Fraction {
37 const divisor = gcd(numerator, denominator) * @as(i64, if (denominator < 0) -1 else 1);
38 return .{ .numerator = @divTrunc(numerator, divisor), .denominator = @divTrunc(denominator, divisor) };
39 }
40 fn add(a: Fraction, b: Fraction) Fraction {
41 return init(a.numerator * b.denominator + b.numerator * a.denominator, a.denominator * b.denominator);
42 }
43 fn sub(a: Fraction, b: Fraction) Fraction {
44 return init(a.numerator * b.denominator - b.numerator * a.denominator, a.denominator * b.denominator);
45 }
46 fn mul(a: Fraction, b: Fraction) Fraction {
47 return init(a.numerator * b.numerator, a.denominator * b.denominator);
48 }
49 fn div(a: Fraction, b: Fraction) Fraction {
50 return init(a.numerator * b.denominator, a.denominator * b.numerator);
51 }
52 fn absolute(a: Fraction) Fraction {
53 return .{ .numerator = if (a.numerator < 0) -a.numerator else a.numerator, .denominator = a.denominator };
54 }
55 fn eq(a: Fraction, b: Fraction) bool {
56 return a.numerator == b.numerator and a.denominator == b.denominator;
57 }
58 fn less(a: Fraction, b: Fraction) bool {
59 return a.numerator * b.denominator < b.numerator * a.denominator;
60 }
61 fn isZero(a: Fraction) bool {
62 return a.numerator == 0;
63 }
64 fn value(a: Fraction) Value {
65 return .{ .fraction = a };
66 }
67 pub fn format(a: Fraction, out: *std.Io.Writer) std.Io.Writer.Error!void {
68 try out.print("{d}/{d}", .{ a.numerator, a.denominator });
69 }
70};
71const Value = struct {
72 fraction: Fraction,
73 pub fn format(value: Value, out: *std.Io.Writer) std.Io.Writer.Error!void {
74 const f = value.fraction;
75 try out.print("{f}({d}.{d:0>6})", .{ f, @divTrunc(f.numerator, f.denominator), @divTrunc(@mod(f.numerator, f.denominator) * 1000000, f.denominator) });
76 }

79const Matrix = [max_states]Vector;
80const Equations = [max_states][2 * max_states]Fraction;
81const zero_vector: Vector = @splat(Fraction.zero);
82const zero_matrix: Matrix = @splat(zero_vector);
83const zero_equations: Equations = @splat(@splat(Fraction.zero));

85fn delta(i: usize, j: usize) Fraction {
86 return if (i == j) Fraction.one else Fraction.zero;
87}
88fn sum(values: []const Fraction) Fraction {
89 var total = Fraction.zero;
90 for (values) |value| total = total.add(value);
91 return total;
92}
93fn identity(size: usize) Matrix {
94 var result = zero_matrix;
95 for (0..size) |i| result[i][i] = Fraction.one;
96 return result;
97}
98fn advance(distribution: Vector, transitions: *const Matrix, size: usize) Vector {
99 var result = zero_vector;
100 for (0..size) |j| for (0..size) |i| {
101 result[j] = result[j].add(distribution[i].mul(transitions[i][j]));
102 };
103 return result;
104}
105fn multiply(left: *const Matrix, right: *const Matrix, size: usize) Matrix {
106 var result = zero_matrix;
107 for (0..size) |i| result[i] = advance(left[i], right, size);
108 return result;
109}
110fn solve(rows: *Equations, size: usize, width: usize) Matrix {
111 for (0..size) |column| {
112 var pivot_row = column;
113 while (rows[pivot_row][column].isZero()) pivot_row += 1;
114 const swap = rows[column];
115 rows[column] = rows[pivot_row];
116 rows[pivot_row] = swap;
117 const pivot = rows[column][column];
118 for (0..width) |j| rows[column][j] = rows[column][j].div(pivot);
119 for (0..size) |i| {
120 if (i == column) continue;
121 const factor = rows[i][column];
122 if (factor.isZero()) continue;
123 for (0..width) |j| rows[i][j] = rows[i][j].sub(factor.mul(rows[column][j]));
124 }
125 }
126 var result = zero_matrix;
127 for (0..size) |i| for (0..width - size) |j| {
128 result[i][j] = rows[i][size + j];
129 };
130 return result;
131}

180const Chain = struct {
181 size: usize = 0,
182 names: [max_states][]const u8 = @splat(""),
183 p: Matrix = zero_matrix,
184
185 fn find(chain: *const Chain, name: []const u8) ?usize {
186 for (chain.names[0..chain.size], 0..) |candidate, i| if (equal(candidate, name)) return i;
187 return null;
188 }
189 fn graph(chain: *const Chain) Graph {
190 var result: Graph = .{};
191 for (0..chain.size) |i| for (0..chain.size) |j| {
192 result.reach[i][j] = i == j or !chain.p[i][j].isZero();
193 };
194 for (0..chain.size) |k| for (0..chain.size) |i| for (0..chain.size) |j| {
195 result.reach[i][j] = result.reach[i][j] or (result.reach[i][k] and result.reach[k][j]);
196 };
197 var assigned: [max_states]bool = @splat(false);
198 for (0..chain.size) |first| {
199 if (assigned[first]) continue;
200 var component: Component = .{};
201 for (0..chain.size) |j| if (result.reach[first][j] and result.reach[j][first]) {
202 component.states.append(j);
203 assigned[j] = true;
204 result.class_of[j] = result.count;
205 };
206 for (component.states.slice()) |i| for (0..chain.size) |j| {
207 if (!component.states.contains(j) and !chain.p[i][j].isZero()) component.closed = false;
208 };
209 component.levels[first] = 0;
210 var queue: StateList = .{};
211 queue.append(first);
212 var head: usize = 0;
213 while (head < queue.len) : (head += 1) {
214 const i = queue.items[head];
215 for (component.states.slice()) |j| if (component.levels[j] < 0 and !chain.p[i][j].isZero()) {
216 component.levels[j] = component.levels[i] + 1;
217 queue.append(j);
218 };
219 }
220 for (component.states.slice()) |i| for (component.states.slice()) |j| {
221 if (!chain.p[i][j].isZero()) component.period = gcd(component.period, component.levels[i] + 1 - component.levels[j]);
222 };
223 result.classes[result.count] = component;
224 result.count += 1;
225 }
226 return result;
227 }
228 fn stationary(chain: *const Chain, states: *const StateList) Vector {
229 var equations = zero_equations;
230 for (0..states.len - 1) |row| for (0..states.len) |col| {
231 equations[row][col] = chain.p[states.items[col]][states.items[row]].sub(delta(col, row));
232 };
233 @memset(equations[states.len - 1][0 .. states.len + 1], Fraction.one);
234 const solution = solve(&equations, states.len, states.len + 1);
235 var distribution = zero_vector;
236 for (states.slice(), 0..) |state, i| distribution[state] = solution[i][0];
237 return distribution;
238 }
239 fn blockSum(chain: *const Chain, from: usize, targets: *const StateList) Fraction {
240 var total = Fraction.zero;
241 for (targets.slice()) |j| total = total.add(chain.p[from][j]);
242 return total;
243 }
244 fn hitting(chain: *const Chain, targets: *const StateList, with_times: bool) Hitting {
245 const topology = chain.graph();
246 var unknown: StateList = .{};
247 var result: Hitting = .{};
248 for (targets.slice()) |i| {
249 result.probability[i] = Fraction.one;
250 result.time[i] = Fraction.zero;
251 }
252 for (0..chain.size) |i| if (!targets.contains(i) and topology.reaches(i, targets)) {
253 unknown.append(i);
254 };
255 if (unknown.len > 0) {
256 var equations = zero_equations;
257 for (unknown.slice(), 0..) |from, i| {
258 for (unknown.slice(), 0..) |to, j| equations[i][j] = delta(i, j).sub(chain.p[from][to]);
259 equations[i][unknown.len] = chain.blockSum(from, targets);
260 }
261 const solution = solve(&equations, unknown.len, unknown.len + 1);
262 for (unknown.slice(), 0..) |state, i| result.probability[state] = solution[i][0];
263 }
264 if (with_times) {
265 var certain: StateList = .{};
266 for (unknown.slice()) |i| if (result.probability[i].eq(Fraction.one)) {
267 certain.append(i);
268 };
269 if (certain.len > 0) {
270 var equations = zero_equations;
271 for (certain.slice(), 0..) |from, i| {
272 for (certain.slice(), 0..) |to, j| equations[i][j] = delta(i, j).sub(chain.p[from][to]);
273 equations[i][certain.len] = Fraction.one;
274 }
275 const solution = solve(&equations, certain.len, certain.len + 1);
276 for (certain.slice(), 0..) |state, i| result.time[state] = solution[i][0];
277 }
278 }
279 return result;
280 }
281 fn absorption(chain: *const Chain) Absorption {
282 var result: Absorption = .{};
283 for (0..chain.size) |i| {
284 if (chain.p[i][i].eq(Fraction.one)) result.absorbing.append(i) else result.transient.append(i);
285 }
286 const size = result.transient.len;
287 if (size == 0) return result;
288 var equations = zero_equations;
289 for (result.transient.slice(), 0..) |from, i| for (result.transient.slice(), 0..) |to, j| {
290 equations[i][j] = delta(i, j).sub(chain.p[from][to]);
291 equations[i][size + j] = delta(i, j);
292 };
293 result.fundamental = solve(&equations, size, 2 * size);
294 for (0..size) |i| result.mean[i] = sum(result.fundamental[i][0..size]);
295 for (0..size) |i| {
296 for (0..size) |j| result.variance[i] = result.variance[i].add(Fraction.two.mul(result.fundamental[i][j]).sub(delta(i, j)).mul(result.mean[j]));
297 result.variance[i] = result.variance[i].sub(result.mean[i].mul(result.mean[i]));
298 for (result.absorbing.slice(), 0..) |absorbing, j| for (result.transient.slice(), 0..) |transient, k| {
299 result.probability[i][j] = result.probability[i][j].add(result.fundamental[i][k].mul(chain.p[transient][absorbing]));
300 };
301 }
302 return result;
303 }

334const Analyzer = struct {
335 gpa: Allocator,
336 out: *std.Io.Writer,
337 chains: std.StringHashMapUnmanaged(Chain) = .empty,
338 error_message: []const u8 = "",

340 fn fail(self: *Analyzer, comptime fmt: []const u8, args: anytype) Error {
341 self.error_message = std.fmt.allocPrint(self.gpa, fmt, args) catch return error.OutOfMemory;
342 return error.BadCommand;

344 fn state(self: *Analyzer, chain: *const Chain, name: []const u8) Error!usize {
345 return chain.find(name) orelse return self.fail("UNKNOWN STATE {s}", .{name});
346 }
347 fn stateSet(self: *Analyzer, chain: *const Chain, words: []const []const u8) Error!StateList {
348 var states: StateList = .{};
349 for (words) |word| {
350 const i = try self.state(chain, word);
351 if (states.contains(i)) return self.fail("DUPLICATE STATE {s}", .{word});
352 states.append(i);
353 }
354 states.sort();
355 return states;
356 }
357 fn probability(self: *Analyzer, text: []const u8) Error!Fraction {
358 var parts = std.mem.splitScalar(u8, text, '/');
359 const numerator = integer(parts.next().?);
360 const denominator = if (parts.next()) |part| integer(part) else 1;
361 if (denominator == 0 or numerator > denominator) return self.fail("BAD PROBABILITY", .{});
362 return Fraction.init(numerator, denominator);
363 }
364 fn stochastic(self: *Analyzer, chain: *const Chain) Error!void {
365 for (0..chain.size) |i| if (!sum(chain.p[i][0..chain.size]).eq(Fraction.one)) {
366 return self.fail("NOT STOCHASTIC {s}", .{chain.names[i]});
367 };
368 }
369 fn uniqueStationary(self: *Analyzer, chain: *const Chain) Error!Vector {
370 const topology = chain.graph();
371 if (topology.count != 1) return self.fail("NOT IRREDUCIBLE", .{});
372 return chain.stationary(&topology.classes[0].states);
373 }

390 fn create(self: *Analyzer, words: []const []const u8) Error!void {
391 if (words.len - 2 > max_states) return self.fail("TOO MANY STATES", .{});
392 var chain: Chain = .{};
393 for (words[2..]) |name| {
394 if (chain.find(name) != null) return self.fail("DUPLICATE STATE {s}", .{name});
395 chain.names[chain.size] = name;
396 chain.size += 1;
397 }
398 try self.chains.put(self.gpa, words[1], chain);
399 try self.out.print("CHAIN {s} states={d}\n", .{ words[1], chain.size });
400 }
401 fn build(self: *Analyzer, words: []const []const u8) Error!void {
402 const order = index(words[2]);
403 if (order != 1 and order != 2) return self.fail("BAD ORDER", .{});
404 const tokens = words[3..];
405 if (tokens.len <= order) return self.fail("TOO FEW TOKENS", .{});
406 const count = tokens.len - order + 1;
407 var observations: [60]usize = undefined;
408 var chain: Chain = .{};
409 for (0..count) |i| {
410 const name = if (order == 1) tokens[i] else try std.fmt.allocPrint(self.gpa, "{s}.{s}", .{ tokens[i], tokens[i + 1] });
411 if (chain.find(name)) |state_index| {
412 observations[i] = state_index;
413 } else {
414 if (chain.size == max_states) return self.fail("TOO MANY STATES", .{});
415 observations[i] = chain.size;
416 chain.names[chain.size] = name;
417 chain.size += 1;
418 }
419 }
420 var counts: [max_states][max_states]i64 = @splat(@splat(0));
421 var totals: [max_states]i64 = @splat(0);
422 for (0..count - 1) |i| {
423 counts[observations[i]][observations[i + 1]] += 1;
424 totals[observations[i]] += 1;
425 }
426 for (0..chain.size) |i| {
427 if (totals[i] == 0) {
428 chain.p[i][i] = Fraction.one;
429 } else {
430 for (0..chain.size) |j| chain.p[i][j] = Fraction.init(counts[i][j], totals[i]);
431 }
432 }
433 try self.chains.put(self.gpa, words[1], chain);
434 try self.out.print("BUILD {s} order={d} states={d} transitions={d}\n", .{ words[1], order, chain.size, count - 1 });
435 for (0..chain.size) |i| {
436 try self.out.print("FROM {s} total={d}", .{ chain.names[i], totals[i] });
437 if (totals[i] == 0) {
438 try self.out.writeAll(" absorbing");
439 } else {
440 for (0..chain.size) |j| if (counts[i][j] > 0) {
441 try self.out.print(" {s}:{d}", .{ chain.names[j], counts[i][j] });
442 };
443 }
444 try self.out.writeAll("\n");
445 }
446 }
447 fn gambler(self: *Analyzer, words: []const []const u8) Error!void {
448 const goal = index(words[2]);
449 const start = index(words[3]);
450 if (goal == 0 or goal > 9) return self.fail("BAD SIZE", .{});
451 if (start > goal) return self.fail("BAD START", .{});
452 const up = try self.probability(words[4]);
453 if (up.isZero() or up.eq(Fraction.one)) return self.fail("BAD PROBABILITY", .{});
454 var chain: Chain = .{ .size = goal + 1 };
455 for (0..chain.size) |i| chain.names[i] = try std.fmt.allocPrint(self.gpa, "s{d}", .{i});
456 chain.p[0][0] = Fraction.one;
457 chain.p[goal][goal] = Fraction.one;
458 for (1..goal) |i| {
459 chain.p[i][i - 1] = Fraction.one.sub(up);
460 chain.p[i][i + 1] = up;
461 }
462 const result = chain.absorption();
463 var ruin = zero_vector;
464 var duration = zero_vector;
465 ruin[0] = Fraction.one;
466 for (result.transient.slice(), 0..) |state_index, i| {
467 ruin[state_index] = result.probability[i][0];
468 duration[state_index] = result.mean[i];
469 }
470 try self.chains.put(self.gpa, words[1], chain);
471 try self.out.print("GAMBLER {s} g={d} p={f} start=s{d} ruin={f} duration={f}\n", .{ words[1], goal, up, start, ruin[start].value(), duration[start].value() });
472 for (0..chain.size) |i| try self.out.print("CAPITAL s{d} ruin={f} win={f} duration={f}\n", .{ i, ruin[i].value(), Fraction.one.sub(ruin[i]).value(), duration[i].value() });
473 }
474 fn classes(self: *Analyzer, name: []const u8, chain: *const Chain) Error!void {
475 const topology = chain.graph();
476 try self.out.print("CLASSES {s} count={d}\n", .{ name, topology.count });
477 for (topology.classes[0..topology.count], 0..) |component, i| {
478 try self.out.print("CLASS {d} {s} period={d} size={d} ", .{ i + 1, if (component.closed) "RECURRENT" else "TRANSIENT", component.period, component.states.len });
479 try self.labels(chain, component.states.slice(), " ");
480 try self.out.writeAll("\n");
481 if (component.period >= 2) {
482 var residue: i64 = 0;
483 while (residue < component.period) : (residue += 1) {
484 try self.out.print("CYCLIC {d}", .{residue});
485 for (component.states.slice()) |state_index| if (@mod(component.levels[state_index], component.period) == residue) {
486 try self.out.print(" {s}", .{chain.names[state_index]});
487 };
488 try self.out.writeAll("\n");
489 }
490 }
491 }

493 fn classify(self: *Analyzer, name: []const u8, chain: *const Chain) Error!void {
494 const topology = chain.graph();
495 var absorbing: StateList = .{};
496 for (0..chain.size) |i| if (chain.p[i][i].eq(Fraction.one)) {
497 absorbing.append(i);
498 };
499 var all_absorbed = absorbing.len > 0;
500 for (0..chain.size) |i| if (!topology.reaches(i, &absorbing)) {
501 all_absorbed = false;
502 };
503 var edges: [max_states][max_states]bool = @splat(@splat(false));
504 for (0..chain.size) |i| for (0..chain.size) |j| {
505 edges[i][j] = !chain.p[i][j].isZero();
506 };
507 var power = edges;
508 var exponent: ?usize = null;
509 const bound = (chain.size - 1) * (chain.size - 1) + 1;
510 for (1..bound + 1) |t| {
511 var all = true;
512 for (0..chain.size) |i| for (0..chain.size) |j| {
513 if (!power[i][j]) all = false;
514 };
515 if (all) {
516 exponent = t;
517 break;
518 }
519 var next: [max_states][max_states]bool = @splat(@splat(false));
520 for (0..chain.size) |i| for (0..chain.size) |j| for (0..chain.size) |k| {
521 next[i][j] = next[i][j] or (power[i][k] and edges[k][j]);
522 };
523 power = next;
524 }
525 try self.out.print("CLASSIFY {s} absorbing_states={d} absorbing={s} ergodic={s} regular={s} exponent=", .{ name, absorbing.len, yes(all_absorbed), yes(topology.count == 1), yes(exponent != null) });
526 if (exponent) |value| try self.out.print("{d}\n", .{value}) else try self.out.writeAll("-\n");
527 }
528 fn step(self: *Analyzer, words: []const []const u8, chain: *const Chain) Error!void {
529 var distribution = zero_vector;
530 if (std.ascii.isDigit(words[2][0])) {
531 var parts = std.mem.splitScalar(u8, words[2], ',');
532 var count: usize = 0;
533 while (parts.next() != null) count += 1;
534 if (count != chain.size) return self.fail("BAD LENGTH", .{});
535 parts = std.mem.splitScalar(u8, words[2], ',');
536 for (0..chain.size) |i| distribution[i] = try self.probability(parts.next().?);
537 if (!sum(distribution[0..chain.size]).eq(Fraction.one)) return self.fail("BAD DISTRIBUTION", .{});
538 } else {
539 distribution[try self.state(chain, words[2])] = Fraction.one;
540 }
541 const steps = index(words[3]);
542 for (0..steps) |_| {
543 const next = advance(distribution, &chain.p, chain.size);
544 distribution = next;
545 }
546 try self.out.print("STEP {s} k={d}\n", .{ words[1], steps });
547 for (0..chain.size) |i| try self.out.print("PROB {s} {f}\n", .{ chain.names[i], distribution[i].value() });
548 }
549 fn stationary(self: *Analyzer, name: []const u8, chain: *const Chain) Error!void {
550 const topology = chain.graph();
551 var closed: usize = 0;
552 for (topology.classes[0..topology.count]) |component| if (component.closed) {
553 closed += 1;
554 };
555 try self.out.print("STATIONARY {s} count={d}\n", .{ name, closed });
556 var number: usize = 0;
557 for (topology.classes[0..topology.count], 0..) |component, i| {
558 if (!component.closed) continue;
559 const distribution = chain.stationary(&component.states);
560 number += 1;
561 try self.out.print("DIST {d} class={d} size={d}\n", .{ number, i + 1, component.states.len });
562 for (component.states.slice()) |state_index| try self.out.print("PI {s} {f}\n", .{ chain.names[state_index], distribution[state_index].value() });
563 }
564 }
565 fn recurrence(self: *Analyzer, words: []const []const u8, chain: *const Chain) Error!void {
566 const state_index = try self.state(chain, words[2]);
567 const topology = chain.graph();
568 const component = topology.classes[topology.class_of[state_index]];
569 if (!component.closed) return self.fail("TRANSIENT STATE {s}", .{words[2]});
570 const rho = chain.stationary(&component.states)[state_index];
571 try self.out.print("RECURRENCE {s} {s} pi={f} mean={f} period={d}\n", .{ words[1], words[2], rho.value(), Fraction.one.div(rho).value(), component.period });
572 }
573 fn absorb(self: *Analyzer, name: []const u8, chain: *const Chain) Error!void {
574 var absorbing: StateList = .{};
575 for (0..chain.size) |i| if (chain.p[i][i].eq(Fraction.one)) {
576 absorbing.append(i);
577 };
578 if (absorbing.len == 0) return self.fail("NO ABSORBING STATE", .{});
579 const topology = chain.graph();
580 for (0..chain.size) |i| if (!absorbing.contains(i) and !topology.reaches(i, &absorbing)) {
581 return self.fail("NOT ABSORBING {s}", .{chain.names[i]});
582 };
583 const result = chain.absorption();
584 try self.out.print("ABSORB {s} transient={d} absorbing={d}\n", .{ name, result.transient.len, result.absorbing.len });
585 if (result.transient.len == 0) return;
586 try self.out.writeAll("TRANSIENT ");
587 try self.labels(chain, result.transient.slice(), " ");
588 try self.out.writeAll("\nABSORBING ");
589 try self.labels(chain, result.absorbing.slice(), " ");
590 try self.out.writeAll("\n");
591 for (result.transient.slice(), 0..) |state_index, i| {
592 try self.out.print("N {s}", .{chain.names[state_index]});
593 for (result.fundamental[i][0..result.transient.len]) |value| try self.out.print(" {f}", .{value});
594 try self.out.writeAll("\n");
595 }
596 for (result.transient.slice(), 0..) |state_index, i| try self.out.print("STEPS {s} mean={f} var={f}\n", .{ chain.names[state_index], result.mean[i].value(), result.variance[i].value() });
597 for (result.transient.slice(), 0..) |state_index, i| {
598 try self.out.print("B {s}", .{chain.names[state_index]});
599 for (result.probability[i][0..result.absorbing.len]) |value| try self.out.print(" {f}", .{value});
600 try self.out.writeAll("\n");
601 }
602 }
603 fn hitting(self: *Analyzer, words: []const []const u8, chain: *const Chain) Error!void {
604 const targets = try self.stateSet(chain, words[2..]);
605 const result = chain.hitting(&targets, true);
606 try self.out.print("HITTING {s} targets=", .{words[1]});
607 try self.labels(chain, targets.slice(), ",");
608 try self.out.writeAll("\n");
609 for (0..chain.size) |i| {
610 try self.out.print("HIT {s} prob={f} time=", .{ chain.names[i], result.probability[i].value() });
611 if (result.time[i]) |time| try self.out.print("{f}\n", .{time.value()}) else try self.out.writeAll("inf\n");
612 }

614 fn passage(self: *Analyzer, name: []const u8, chain: *const Chain) Error!void {
615 const pi = try self.uniqueStationary(chain);
616 var values = zero_matrix;
617 for (0..chain.size) |j| {
618 var targets: StateList = .{};
619 targets.append(j);
620 const result = chain.hitting(&targets, true);
621 for (0..chain.size) |i| values[i][j] = if (i == j) Fraction.one.div(pi[j]) else result.time[i].?;
622 }
623 var kemeny = Fraction.zero;
624 for (1..chain.size) |j| kemeny = kemeny.add(pi[j].mul(values[0][j]));
625 try self.out.print("PASSAGE {s} kemeny={f}\n", .{ name, kemeny.value() });
626 try self.display(chain, &values, "M");
627 }
628 fn firstPassage(self: *Analyzer, words: []const []const u8, chain: *const Chain) Error!void {
629 const from = try self.state(chain, words[2]);
630 const target = try self.state(chain, words[3]);
631 const steps = index(words[4]);
632 if (steps == 0) return self.fail("BAD STEPS", .{});
633 var current = zero_vector;
634 for (0..chain.size) |i| current[i] = chain.p[i][target];
635 var probabilities: [30]Fraction = undefined;
636 var cumulative: [30]Fraction = undefined;
637 var total = Fraction.zero;
638 for (0..steps) |t| {
639 probabilities[t] = current[from];
640 total = total.add(current[from]);
641 cumulative[t] = total;
642 if (t + 1 < steps) {
643 var next = zero_vector;
644 for (0..chain.size) |i| for (0..chain.size) |j| {
645 if (j != target) next[i] = next[i].add(chain.p[i][j].mul(current[j]));
646 };
647 current = next;
648 }
649 }
650 try self.out.print("FIRSTPASSAGE {s} {s} {s} k={d} total={f}\n", .{ words[1], words[2], words[3], steps, total.value() });
651 for (0..steps) |t| try self.out.print("FP {d} {f} cum={f}\n", .{ t + 1, probabilities[t].value(), cumulative[t].value() });
652 }
653 fn pathProbability(self: *Analyzer, words: []const []const u8, chain: *const Chain) Error!void {
654 var states: [30]usize = undefined;
655 const count = words.len - 2;
656 for (words[2..], 0..) |word, i| states[i] = try self.state(chain, word);
657 var product = Fraction.one;
658 var zero_at: ?usize = null;
659 for (1..count) |i| {
660 product = product.mul(chain.p[states[i - 1]][states[i]]);
661 if (product.isZero() and zero_at == null) zero_at = i;
662 }
663 try self.out.print("PATHPROB {s} steps={d} prob={f}", .{ words[1], count - 1, product.value() });
664 if (zero_at) |value| try self.out.print(" zero_at={d}", .{value});
665 try self.out.writeAll("\n");
666 }
667 fn mostLikely(self: *Analyzer, words: []const []const u8, chain: *const Chain) Error!void {
668 const start = try self.state(chain, words[2]);
669 const steps = index(words[3]);
670 if (steps == 0) return self.fail("BAD STEPS", .{});
671 var best: [21]Vector = @splat(zero_vector);
672 best[steps] = @splat(Fraction.one);
673 var remaining = steps;
674 while (remaining > 0) {
675 remaining -= 1;
676 for (0..chain.size) |i| for (0..chain.size) |j| {
677 const candidate = chain.p[i][j].mul(best[remaining + 1][j]);
678 if (best[remaining][i].less(candidate)) best[remaining][i] = candidate;
679 };
680 }
681 var path: [21]usize = undefined;
682 path[0] = start;
683 for (0..steps) |t| {
684 const from = path[t];
685 for (0..chain.size) |j| if (chain.p[from][j].mul(best[t + 1][j]).eq(best[t][from])) {
686 path[t + 1] = j;
687 break;
688 };
689 }
690 try self.out.print("MOSTLIKELY {s} {s} k={d} prob={f}\nPATH ", .{ words[1], words[2], steps, best[0][start].value() });
691 try self.labels(chain, path[0 .. steps + 1], " ");
692 try self.out.writeAll("\n");
693 }
694 fn reversible(self: *Analyzer, name: []const u8, chain: *const Chain) Error!void {
695 const pi = try self.uniqueStationary(chain);
696 const Violation = struct { from: usize, to: usize, forward: Fraction, backward: Fraction };
697 var violations: [45]Violation = undefined;
698 var count: usize = 0;
699 for (0..chain.size) |i| for (i + 1..chain.size) |j| {
700 const forward = pi[i].mul(chain.p[i][j]);
701 const backward = pi[j].mul(chain.p[j][i]);
702 if (!forward.eq(backward)) {
703 violations[count] = .{ .from = i, .to = j, .forward = forward, .backward = backward };
704 count += 1;
705 }
706 };
707 if (count == 0) {
708 try self.out.print("REVERSIBLE {s} yes\n", .{name});
709 } else {
710 try self.out.print("REVERSIBLE {s} no violations={d}\n", .{ name, count });
711 for (violations[0..count]) |pair| try self.out.print("VIOLATION {s} {s} {f} {f}\n", .{ chain.names[pair.from], chain.names[pair.to], pair.forward.value(), pair.backward.value() });
712 }
713 }
714 fn reverse(self: *Analyzer, words: []const []const u8, chain: *const Chain) Error!void {
715 const pi = try self.uniqueStationary(chain);
716 var reversed = chain.*;
717 var same = true;
718 for (0..chain.size) |i| for (0..chain.size) |j| {
719 reversed.p[i][j] = pi[j].mul(chain.p[j][i]).div(pi[i]);
720 if (!reversed.p[i][j].eq(chain.p[i][j])) same = false;
721 };
722 try self.chains.put(self.gpa, words[2], reversed);
723 try self.out.print("REVERSE {s} {s} reversible={s}\n", .{ words[1], words[2], yes(same) });
724 try self.display(&reversed, &reversed.p, "ROW");
725 }
726 fn mixing(self: *Analyzer, words: []const []const u8, chain: *const Chain) Error!void {
727 const steps = index(words[2]);
728 if (steps == 0) return self.fail("BAD STEPS", .{});
729 const epsilon = try self.probability(words[3]);
730 const pi = try self.uniqueStationary(chain);
731 var power = identity(chain.size);
732 var distances: [20]Fraction = undefined;
733 var worst: [20]usize = undefined;
734 var mixing_time: ?usize = null;
735 for (0..steps) |t| {
736 const next = multiply(&power, &chain.p, chain.size);
737 power = next;
738 var distance = Fraction.zero;
739 var from: usize = 0;
740 for (0..chain.size) |i| {
741 var current = Fraction.zero;
742 for (0..chain.size) |j| current = current.add(power[i][j].sub(pi[j]).absolute());
743 current = current.div(Fraction.two);
744 if (distance.less(current)) {
745 distance = current;
746 from = i;
747 }
748 }
749 if (!epsilon.less(distance) and mixing_time == null) mixing_time = t + 1;
750 distances[t] = distance;
751 worst[t] = from;
752 }
753 try self.out.print("MIXING {s} k={d} eps={f} time=", .{ words[1], steps, epsilon });
754 if (mixing_time) |time| try self.out.print("{d}\n", .{time}) else try self.out.writeAll("none\n");
755 for (0..steps) |t| try self.out.print("TV {d} {f} from={s}\n", .{ t + 1, distances[t].value(), chain.names[worst[t]] });
756 }
757 fn lessBlock(_: void, a: StateList, b: StateList) bool {
758 return a.items[0] < b.items[0];
759 }
760 fn lump(self: *Analyzer, words: []const []const u8, chain: *const Chain) Error!void {
761 var blocks: [max_states]StateList = undefined;
762 var count: usize = 0;
763 var seen: StateList = .{};
764 var groups = std.mem.splitScalar(u8, words[3], '|');
765 while (groups.next()) |group| {
766 var block: StateList = .{};
767 var names = std.mem.splitScalar(u8, group, ',');
768 while (names.next()) |name| {
769 const state_index = try self.state(chain, name);
770 if (seen.contains(state_index)) return self.fail("BAD PARTITION", .{});
771 seen.append(state_index);
772 block.append(state_index);
773 }
774 block.sort();
775 blocks[count] = block;
776 count += 1;
777 }
778 if (seen.len != chain.size) return self.fail("BAD PARTITION", .{});
779 std.mem.sort(StateList, blocks[0..count], {}, lessBlock);
780 var lumped: Chain = .{ .size = count };
781 for (blocks[0..count], 0..) |block, i| lumped.names[i] = chain.names[block.items[0]];
782 for (blocks[0..count], 0..) |block, i| for (blocks[0..count], 0..) |target, j| {
783 const representative = block.items[0];
784 const baseline = chain.blockSum(representative, &target);
785 for (block.slice()) |state_index| {
786 const value = chain.blockSum(state_index, &target);
787 if (!value.eq(baseline)) {
788 try self.out.print("LUMP {s} not lumpable block={s} target={s} {s}={f} {s}={f}\n", .{
789 words[1], chain.names[representative], chain.names[target.items[0]],
790 chain.names[representative], baseline.value(), chain.names[state_index],
791 value.value(),
792 });
793 return;
794 }
795 }
796 lumped.p[i][j] = baseline;
797 };
798 try self.chains.put(self.gpa, words[2], lumped);
799 try self.out.print("LUMP {s} {s} blocks={d}\n", .{ words[1], words[2], count });
800 try self.display(&lumped, &lumped.p, "ROW");
801 }
802 fn reward(self: *Analyzer, words: []const []const u8, chain: *const Chain) Error!void {
803 if (words.len - 3 != chain.size) return self.fail("BAD LENGTH", .{});
804 const discount = try self.probability(words[2]);
805 if (discount.eq(Fraction.one)) return self.fail("BAD DISCOUNT", .{});
806 var rewards = zero_vector;
807 var equations = zero_equations;
808 for (0..chain.size) |i| {
809 rewards[i] = Fraction.init(integer(words[i + 3]), 1);
810 for (0..chain.size) |j| equations[i][j] = delta(i, j).sub(discount.mul(chain.p[i][j]));
811 equations[i][chain.size] = rewards[i];
812 }
813 const discounted = solve(&equations, chain.size, chain.size + 1);
814 var gain = zero_vector;
815 const topology = chain.graph();
816 for (topology.classes[0..topology.count]) |component| {
817 if (!component.closed) continue;
818 const rho = chain.stationary(&component.states);
819 var average = Fraction.zero;
820 for (component.states.slice()) |i| average = average.add(rho[i].mul(rewards[i]));
821 const result = chain.hitting(&component.states, false);
822 for (0..chain.size) |i| gain[i] = gain[i].add(result.probability[i].mul(average));
823 }
824 try self.out.print("REWARD {s} gamma={f}\n", .{ words[1], discount });
825 for (0..chain.size) |i| try self.out.print("VALUE {s} discounted={f} gain={f}\n", .{ chain.names[i], discounted[i][0].value(), gain[i].value() });
826 }
827 fn restrict(self: *Analyzer, words: []const []const u8, chain: *const Chain) Error!void {
828 const states = try self.stateSet(chain, words[3..]);
829 for (states.slice()) |i| for (0..chain.size) |j| {
830 if (!states.contains(j) and !chain.p[i][j].isZero()) return self.fail("NOT CLOSED {s}", .{chain.names[i]});
831 };
832 var restricted: Chain = .{ .size = states.len };
833 for (states.slice(), 0..) |from, i| {
834 restricted.names[i] = chain.names[from];
835 for (states.slice(), 0..) |to, j| restricted.p[i][j] = chain.p[from][to];
836 }
837 try self.chains.put(self.gpa, words[2], restricted);
838 try self.out.print("RESTRICT {s} {s} states={d}\n", .{ words[1], words[2], states.len });
839 try self.display(&restricted, &restricted.p, "ROW");
840 }
841 fn simulate(self: *Analyzer, words: []const []const u8, chain: *const Chain) Error!void {
842 var current = try self.state(chain, words[2]);
843 const steps = index(words[3]);
844 if (steps == 0) return self.fail("BAD STEPS", .{});
845 const seed = integer(words[4]);
846 if (seed == 0 or seed >= 2147483647) return self.fail("BAD SEED", .{});
847 var random = seed;
848 var path: [61]usize = undefined;
849 var visits: [max_states]usize = @splat(0);
850 path[0] = current;
851 visits[current] = 1;
852 for (0..steps) |t| {
853 var denominator: i64 = 1;
854 for (chain.p[current][0..chain.size]) |value| denominator = @divTrunc(denominator, gcd(denominator, value.denominator)) * value.denominator;
855 random = @mod(48271 * random, 2147483647);
856 const sample = @mod(random, denominator);
857 var cumulative = Fraction.zero;
858 for (0..chain.size) |j| {
859 cumulative = cumulative.add(chain.p[current][j]);
860 if (cumulative.numerator * @divTrunc(denominator, cumulative.denominator) > sample) {
861 current = j;
862 break;
863 }
864 }
865 path[t + 1] = current;
866 visits[current] += 1;
867 }
868 try self.out.print("SIMULATE {s} {s} steps={d} seed={d} end={s}\nPATH ", .{ words[1], words[2], steps, seed, chain.names[current] });
869 try self.labels(chain, path[0 .. steps + 1], " ");
870 try self.out.writeAll("\n");
871 for (0..chain.size) |i| try self.out.print("VISIT {s} {d}\n", .{ chain.names[i], visits[i] });
872 }
873 fn execute(self: *Analyzer, words: []const []const u8) Error!void {
874 const command = std.meta.stringToEnum(Command, words[0]).?;
875 const name = words[1];
876 switch (command) {
877 .CHAIN => return self.create(words),
878 .BUILD => return self.build(words),
879 .GAMBLER => return self.gambler(words),
880 else => {},
881 }
882 var chain = self.chains.get(name) orelse return self.fail("UNKNOWN CHAIN {s}", .{name});
883 switch (command) {
884 .P, .ROW, .SHOW, .CHECK => {},
885 else => try self.stochastic(&chain),
886 }
887 switch (command) {
888 .P => {
889 const from = try self.state(&chain, words[2]);
890 const to = try self.state(&chain, words[3]);
891 const value = try self.probability(words[4]);
892 chain.p[from][to] = value;
893 try self.chains.put(self.gpa, name, chain);
894 try self.out.print("P {s} {s} {s} {f} rowsum={f}\n", .{ name, words[2], words[3], value, sum(chain.p[from][0..chain.size]) });
895 },
896 .ROW => {
897 const from = try self.state(&chain, words[2]);
898 if (words.len - 3 != chain.size) return self.fail("BAD LENGTH", .{});
899 for (words[3..], 0..) |word, j| chain.p[from][j] = try self.probability(word);
900 try self.chains.put(self.gpa, name, chain);
901 try self.out.print("ROW {s} {s} rowsum={f}\n", .{ name, words[2], sum(chain.p[from][0..chain.size]) });
902 },
903 .SHOW => {
904 try self.out.print("MATRIX {s} states={d}\n", .{ name, chain.size });
905 try self.display(&chain, &chain.p, "ROW");
906 },
907 .CHECK => {
908 var sums = zero_vector;
909 var bad: StateList = .{};
910 for (0..chain.size) |i| {
911 sums[i] = sum(chain.p[i][0..chain.size]);
912 if (!sums[i].eq(Fraction.one)) bad.append(i);
913 }
914 if (bad.len == 0) {
915 try self.out.print("CHECK {s} ok\n", .{name});
916 } else {
917 try self.out.print("CHECK {s} bad={d}\n", .{ name, bad.len });
918 for (bad.slice()) |i| try self.out.print("BADROW {s} sum={f}\n", .{ chain.names[i], sums[i] });
919 }
920 },
921 .CLASSES => try self.classes(name, &chain),
922 .CLASSIFY => try self.classify(name, &chain),
923 .STEP => try self.step(words, &chain),
924 .POWER => {
925 const steps = index(words[2]);
926 var power = identity(chain.size);
927 for (0..steps) |_| {
928 const next = multiply(&power, &chain.p, chain.size);
929 power = next;
930 }
931 try self.out.print("POWER {s} k={d}\n", .{ name, steps });
932 try self.display(&chain, &power, "ROW");
933 },
934 .STATIONARY => try self.stationary(name, &chain),
935 .RECURRENCE => try self.recurrence(words, &chain),
936 .ABSORB => try self.absorb(name, &chain),
937 .HITTING => try self.hitting(words, &chain),
938 .PASSAGE => try self.passage(name, &chain),
939 .FIRSTPASSAGE => try self.firstPassage(words, &chain),
940 .PATHPROB => try self.pathProbability(words, &chain),
941 .MOSTLIKELY => try self.mostLikely(words, &chain),
942 .REVERSIBLE => try self.reversible(name, &chain),
943 .REVERSE => try self.reverse(words, &chain),
944 .MIXING => try self.mixing(words, &chain),
945 .LUMP => try self.lump(words, &chain),
946 .REWARD => try self.reward(words, &chain),
947 .RESTRICT => try self.restrict(words, &chain),
948 .SIMULATE => try self.simulate(words, &chain),
949 .CHAIN, .BUILD, .GAMBLER => unreachable,
950 }
951 }
952};

954fn readWords(gpa: Allocator, input: *std.Io.Reader) ![]const []const u8 {
955 var buffer: std.Io.Writer.Allocating = .init(gpa);
956 defer buffer.deinit();
957 _ = try input.streamDelimiterEnding(&buffer.writer, '\n');
958 if (input.seek < input.end) input.toss(1) else if (buffer.written().len == 0) return error.EndOfStream;
959 const line = try buffer.toOwnedSlice();
960 var words: std.ArrayList([]const u8) = .empty;
961 var parts = std.mem.tokenizeAny(u8, line, " \r");
962 while (parts.next()) |part| try words.append(gpa, part);
963 return try words.toOwnedSlice(gpa);
964}
