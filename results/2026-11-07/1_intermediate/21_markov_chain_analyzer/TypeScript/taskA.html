5function absInt(n: bigint): bigint { return n < 0n ? -n : n; }
6function gcd(a: bigint, b: bigint): bigint {
7 a = absInt(a);
8 b = absInt(b);
9 while (b !== 0n) [a, b] = [b, a % b];
10 return a;
11}
12function lcm(a: bigint, b: bigint): bigint { return a / gcd(a, b) * b; }

14class Fraction {
15 readonly n: bigint;
16 readonly d: bigint;
17
18 constructor(n: bigint | number | string, d: bigint | number | string = 1n) {
19 n = BigInt(n);
20 d = BigInt(d);
21 if (d === 0n) throw Error("Zero denominator");
22 if (d < 0n) [n, d] = [-n, -d];
23 const factor = gcd(n, d);
24 this.n = n / factor;
25 this.d = d / factor;
26 }
27 static of(value: Scalar): Fraction { return value instanceof Fraction ? value : new Fraction(value); }
28 add(other: Scalar): Fraction {
29 other = Fraction.of(other);
30 return new Fraction(this.n * other.d + other.n * this.d, this.d * other.d);
31 }
32 sub(other: Scalar): Fraction {
33 other = Fraction.of(other);
34 return new Fraction(this.n * other.d - other.n * this.d, this.d * other.d);
35 }
36 mul(other: Scalar): Fraction {
37 other = Fraction.of(other);
38 return new Fraction(this.n * other.n, this.d * other.d);
39 }
40 div(other: Scalar): Fraction {
41 other = Fraction.of(other);
42 return new Fraction(this.n * other.d, this.d * other.n);
43 }
44 cmp(other: Scalar): number {
45 other = Fraction.of(other);
46 const difference = this.n * other.d - other.n * this.d;
47 return difference < 0n ? -1 : difference > 0n ? 1 : 0;
48 }
49 eq(other: Scalar): boolean { return this.cmp(other) === 0; }
50 abs(): Fraction { return new Fraction(absInt(this.n), this.d); }
51 neg(): Fraction { return new Fraction(-this.n, this.d); }
52 toString(): string { return this.n + "/" + this.d; }
53}
54const ZERO = new Fraction(0n);
55const ONE = new Fraction(1n);
56const fractionText = (value: Scalar): string => Fraction.of(value).toString();
57const answer = (flag: boolean): string => flag ? "yes" : "no";
58
59function valueText(value: Scalar): string {
60 value = Fraction.of(value);
61 let whole = value.n / value.d;
62 let remainder = value.n % value.d;
63 if (remainder < 0n) {
64 --whole;
65 remainder += value.d;
66 }
67 const fraction = (remainder * 1000000n / value.d).toString().padStart(6, "0");
68 return fractionText(value) + "(" + whole + "." + fraction + ")";
69}
70function timeText(value: Scalar | null): string { return value === null ? "inf" : valueText(value); }
71function parseProbability(text: string): Fraction {
72 const parts = text.split("/");
73 const n = BigInt(parts[0]);
74 const d = parts.length === 2 ? BigInt(parts[1]) : 1n;
75 if (d === 0n || n > d) throw new CommandError("BAD PROBABILITY");
76 return new Fraction(n, d);
77}
78function sum(values: Fraction[]): Fraction { return values.reduce((total, value) => total.add(value), ZERO); }
79function delta(i: number, j: number): Fraction { return i === j ? ONE : ZERO; }
80function identity(size: number): Fraction[][] {
81 return Array.from({length: size}, (_, i) =>
82 Array.from({length: size}, (_, j) => delta(i, j)));
83}
84function membership(size: number, states: number[]): boolean[] {
85 const inside = Array(size).fill(false);
86 for (const state of states) inside[state] = true;
87 return inside;
88}
89function eliminate(rows: Fraction[][], size: number): Fraction[][] {
90 for (let column = 0; column < size; ++column) {
91 let pivotRow = column;
92 while (rows[pivotRow][column].eq(ZERO)) ++pivotRow;
93 [rows[column], rows[pivotRow]] = [rows[pivotRow], rows[column]];
94 const pivot = rows[column][column];
95 rows[column] = rows[column].map(value => value.div(pivot));
96 for (let i = 0; i < size; ++i) {
97 const factor = rows[i][column];
98 if (i !== column && !factor.eq(ZERO)) {
99 rows[i] = rows[i].map((value, j) => value.sub(factor.mul(rows[column][j])));
100 }
101 }
102 }
103 return rows.map(row => row.slice(size));
104}
105function igcd(a: number, b: number): number {
106 a = Math.abs(a);
107 b = Math.abs(b);
108 while (b !== 0) [a, b] = [b, a % b];
109 return a;
110}

114class Chain {
115 readonly names: string[];
116 readonly positions: Map<string, number>;
117 matrix: Fraction[][];
118
119 constructor(names: string[]) {
120 this.names = [...names];
121 this.positions = new Map<string, number>(names.map((name, i) => [name, i]));
122 this.matrix = names.map(() => names.map(() => ZERO));
123 }
124 get size(): number { return this.names.length; }
125 state(text: string): number {
126 if (!this.positions.has(text)) throw new CommandError("UNKNOWN STATE " + text);
127 return this.positions.get(text);
128 }
129 distinctStates(texts: string[]): number[] {
130 const found = new Set();
131 const states = [];
132 for (const text of texts) {
133 const state = this.state(text);
134 if (found.has(state)) throw new CommandError("DUPLICATE STATE " + text);
135 states.push(state);
136 found.add(state);
137 }
138 return states;
139 }
140 labels(states: number[]): string[] { return states.map(i => this.names[i]); }
141 rowSum(i: number): Fraction { return sum(this.matrix[i]); }
142 blockSum(i: number, block: number[]): Fraction { return sum(block.map(j => this.matrix[i][j])); }
143 edge(i: number, j: number): boolean { return this.matrix[i][j].cmp(ZERO) > 0; }
144 absorbing(i: number): boolean { return this.matrix[i][i].eq(ONE); }
145 advance(distribution: Fraction[]): Fraction[] {
146 return this.names.map((_, j) => sum(distribution.map((prob, i) =>
147 prob.mul(this.matrix[i][j]))));
148 }
149 reachable(start: number): boolean[] {
150 const seen = Array(this.size).fill(false);
151 const stack = [start];
152 seen[start] = true;
153 while (stack.length) {
154 const i = stack.pop();
155 for (let j = 0; j < this.size; ++j) {
156 if (this.edge(i, j) && !seen[j]) {
157 seen[j] = true;
158 stack.push(j);
159 }
160 }
161 }
162 return seen;
163 }
164}

166class CommunicatingClass {
167 readonly number: number;
168 readonly members: number[];
169 readonly closed: boolean;
170 readonly levels: number[];
171 readonly period: number;
172
173 constructor(chain: Chain, number: number, members: number[]) {
174 this.number = number;
175 this.members = members;
176 const inside = membership(chain.size, members);
177 this.closed = !members.some(i => chain.names.some((_, j) => chain.edge(i, j) && !inside[j]));
178 this.levels = Array(chain.size).fill(-1);
179 const queue = [members[0]];
180 this.levels[members[0]] = 0;
181 for (let head = 0; head < queue.length; ++head) {
182 const u = queue[head];
183 for (const v of members) {
184 if (chain.edge(u, v) && this.levels[v] < 0) {
185 this.levels[v] = this.levels[u] + 1;
186 queue.push(v);
187 }
188 }
189 }
190 this.period = 0;
191 for (const u of members) {
192 for (const v of members) {
193 if (chain.edge(u, v)) {
194 this.period = igcd(this.period, this.levels[u] + 1 - this.levels[v]);
195 }
196 }
197 }
198 }
199}

201class Decomposition {
202 readonly reach: boolean[][];
203 readonly classes: CommunicatingClass[];
204 readonly classOf: number[];
205
206 constructor(chain: Chain) {
207 this.reach = chain.names.map((_, i) => chain.reachable(i));
208 this.classes = [];
209 this.classOf = Array(chain.size).fill(-1);
210 for (let i = 0; i < chain.size; ++i) {
211 if (this.classOf[i] >= 0) continue;
212 const members = chain.names.map((_, j) => j)
213 .filter(j => this.reach[i][j] && this.reach[j][i]);
214 for (const j of members) this.classOf[j] = this.classes.length;
215 this.classes.push(new CommunicatingClass(chain, this.classes.length + 1, members));
216 }
217 }
218 closedClasses(): CommunicatingClass[] { return this.classes.filter(item => item.closed); }
219 containing(state: number): CommunicatingClass { return this.classes[this.classOf[state]]; }
220 reaches(state: number, targets: boolean[]): boolean { return this.reach[state].some((found, i) => found && targets[i]); }
221}

223class Absorption {
224 readonly transient: number[];
225 readonly absorbing: number[];
226 readonly fundamental: Fraction[][];
227 readonly steps: Fraction[];
228 readonly probabilities: Fraction[][];
229
230 constructor(chain: Chain) {
231 this.transient = chain.names.map((_, i) => i).filter(i => !chain.absorbing(i));
232 this.absorbing = chain.names.map((_, i) => i).filter(i => chain.absorbing(i));
233 const count = this.transient.length;
234 const rows = this.transient.map((i, r) =>
235 this.transient.map(j => delta(i, j).sub(chain.matrix[i][j]))
236 .concat(Array.from({length: count}, (_, c) => delta(r, c))));
237 this.fundamental = eliminate(rows, count);
238 this.steps = this.fundamental.map(row => sum(row));
239 this.probabilities = this.fundamental.map(row =>
240 this.absorbing.map(a => sum(row.map((rho, k) =>
241 rho.mul(chain.matrix[this.transient[k]][a])))));
242 }
243 variance(i: number): Fraction {
244 const total = sum(this.fundamental[i].map((value, j) =>
245 value.mul(2).sub(delta(i, j)).mul(this.steps[j])));
246 return total.sub(this.steps[i].mul(this.steps[i]));
247 }
248}

250function stationaryDistribution(chain: Chain, members: number[]): Fraction[] {
251 const n = members.length;
252 const rows = [];
253 for (let r = 0; r < n - 1; ++r) {
254 const coefficients = members.map((member, k) =>
255 chain.matrix[member][members[r]].sub(delta(k, r)));
256 rows.push(coefficients.concat([ZERO]));
257 }
258 rows.push(Array(n + 1).fill(ONE));
259 return eliminate(rows, n).map(row => row[0]);
260}
261function irreducibleDistribution(chain: Chain, decomposition: Decomposition): Fraction[] {
262 if (decomposition.classes.length > 1) throw new CommandError("NOT IRREDUCIBLE");
263 return stationaryDistribution(chain, decomposition.classes[0].members);
264}
265function solveRestricted(chain: Chain, states: number[], constants: Fraction[]): Fraction[] {
266 const rows = states.map((i, k) =>
267 states.map(j => delta(i, j).sub(chain.matrix[i][j])).concat([constants[k]]));
268 return eliminate(rows, states.length).map(row => row[0]);
269}
270function hittingProbabilities(chain: Chain, decomposition: Decomposition, targets: boolean[]): Fraction[] {
271 const probabilities = chain.names.map((_, i) => targets[i] ? ONE : ZERO);
272 const remaining = chain.names.map((_, i) => i)
273 .filter(i => !targets[i] && decomposition.reaches(i, targets));
274 if (remaining.length) {
275 const constants = remaining.map(i => sum(chain.names.map((_, j) =>
276 targets[j] ? chain.matrix[i][j] : ZERO)));
277 const values = solveRestricted(chain, remaining, constants);
278 remaining.forEach((i, k) => { probabilities[i] = values[k]; });
279 }
280 return probabilities;
281}
282function hittingTimes(chain: Chain, targets: boolean[], probabilities: Fraction[]): (Fraction | null)[] {
283 const times = chain.names.map((_, i) => targets[i] ? ZERO : null);
284 const certain = chain.names.map((_, i) => i)
285 .filter(i => !targets[i] && probabilities[i].eq(ONE));
286 if (certain.length) {
287 const values = solveRestricted(chain, certain, certain.map(() => ONE));
288 certain.forEach((i, k) => { times[i] = values[k]; });
289 }
290 return times;
291}
292function regularExponent(chain: Chain): number | null {
293 const n = chain.size;
294 const edges = chain.names.map((_, i) => chain.names.map((_, j) => chain.edge(i, j)));
295 let power = edges;
296 for (let exponent = 1; exponent < (n - 1) * (n - 1) + 2; ++exponent) {
297 if (power.every(row => row.every(Boolean))) return exponent;
298 power = power.map((row, i) => row.map((_, j) =>
299 chain.names.some((__, k) => power[i][k] && edges[k][j])));
300 }
301 return null;
302}

377 build(a: string[]) {
378 const name = a[0], order = Number(a[1]), tokens = a.slice(2);
379 if (order !== 1 && order !== 2) throw new CommandError("BAD ORDER");
380 if (tokens.length <= order) throw new CommandError("TOO FEW TOKENS");
381 const observations = order === 1 ? tokens :
382 tokens.slice(0, -1).map((token, i) => token + "." + tokens[i + 1]);
383 const names = [...new Set(observations)];
384 if (names.length > 10) throw new CommandError("TOO MANY STATES");
385 const chain = new Chain(names);
386 const counts = names.map(() => names.map(() => 0));
387 for (let i = 0; i < observations.length - 1; ++i) {
388 counts[chain.positions.get(observations[i])][chain.positions.get(observations[i + 1])]++;
389 }
390 this.emit("BUILD", name, "order=" + order, "states=" + names.length,
391 "transitions=" + (observations.length - 1));
392 for (let i = 0; i < chain.size; ++i) {
393 const total = counts[i].reduce((x, y) => x + y, 0);
394 if (total === 0) {
395 chain.matrix[i][i] = ONE;
396 this.emit("FROM", names[i], "total=0", "absorbing");
397 continue;
398 }
399 const parts = [];
400 for (let j = 0; j < chain.size; ++j) {
401 if (counts[i][j] > 0) {
402 chain.matrix[i][j] = new Fraction(counts[i][j], total);
403 parts.push(names[j] + ":" + counts[i][j]);
404 }
405 }
406 this.emit("FROM", names[i], "total=" + total, ...parts);
407 }
408 this.chains.set(name, chain);
409 }
410 gambler(a: string[]) {
411 const name = a[0], goal = Number(a[1]), start = Number(a[2]);
412 if (goal === 0 || goal > 9) throw new CommandError("BAD SIZE");
413 if (start > goal) throw new CommandError("BAD START");
414 const probability = parseProbability(a[3]);
415 if (probability.eq(ZERO) || probability.eq(ONE))
416 throw new CommandError("BAD PROBABILITY");
417 const chain = new Chain(Array.from({length: goal + 1}, (_, i) => "s" + i));
418 chain.matrix[0][0] = ONE;
419 chain.matrix[goal][goal] = ONE;
420 for (let i = 1; i < goal; ++i) {
421 chain.matrix[i][i + 1] = probability;
422 chain.matrix[i][i - 1] = ONE.sub(probability);
423 }
424 this.chains.set(name, chain);
425 const absorption = new Absorption(chain);
426 const ruin = [ONE, ...absorption.probabilities.map(row => row[0]), ZERO];
427 const duration = [ZERO, ...absorption.steps, ZERO];
428 this.emit("GAMBLER", name, "g=" + goal, "p=" + fractionText(probability),
429 "start=s" + start, "ruin=" + valueText(ruin[start]),
430 "duration=" + valueText(duration[start]));
431 for (let i = 0; i <= goal; ++i) {
432 this.emit("CAPITAL", "s" + i, "ruin=" + valueText(ruin[i]),
433 "win=" + valueText(ONE.sub(ruin[i])), "duration=" + valueText(duration[i]));
434 }
435 }
436 classes(a: string[]) {
437 const chain = this.requireStochastic(a[0]);
438 const decomposition = new Decomposition(chain);
439 this.emit("CLASSES", a[0], "count=" + decomposition.classes.length);
440 for (const item of decomposition.classes) {
441 this.emit("CLASS", item.number, item.closed ? "RECURRENT" : "TRANSIENT",
442 "period=" + item.period, "size=" + item.members.length,
443 ...chain.labels(item.members));
444 if (item.period < 2) continue;
445 for (let remainder = 0; remainder < item.period; ++remainder) {
446 const states = item.members.filter(i => item.levels[i] % item.period === remainder);
447 this.emit("CYCLIC", remainder, ...chain.labels(states));
448 }
449 }
450 }
451 classify(a: string[]) {
452 const chain = this.requireStochastic(a[0]);
453 const decomposition = new Decomposition(chain);
454 const absorbing = chain.names.map((_, i) => chain.absorbing(i));
455 const count = absorbing.filter(Boolean).length;
456 const absorbingChain = count > 0 &&
457 chain.names.every((_, i) => decomposition.reaches(i, absorbing));
458 const exponent = regularExponent(chain);
459 this.emit("CLASSIFY", a[0], "absorbing_states=" + count,
460 "absorbing=" + answer(absorbingChain),
461 "ergodic=" + answer(decomposition.classes.length === 1),
462 "regular=" + answer(exponent !== null),
463 "exponent=" + (exponent === null ? "-" : exponent));
464 }
465 step(a: string[]) {
466 const chain = this.requireStochastic(a[0]);
467 const initial = a[1], steps = Number(a[2]);
468 let distribution;
469 if (/^[0-9]/.test(initial)) {
470 const texts = initial.split(",");
471 if (texts.length !== chain.size) throw new CommandError("BAD LENGTH");
472 distribution = texts.map(parseProbability);
473 if (!sum(distribution).eq(ONE)) throw new CommandError("BAD DISTRIBUTION");
474 } else {
475 const state = chain.state(initial);
476 distribution = chain.names.map((_, i) => delta(state, i));
477 }
478 for (let k = 0; k < steps; ++k) distribution = chain.advance(distribution);
479 this.emit("STEP", a[0], "k=" + steps);
480 chain.names.forEach((name, i) => this.emit("PROB", name, valueText(distribution[i])));
481 }

483 const chain = this.requireStochastic(a[0]);
484 const steps = Number(a[1]);

486 for (let k = 0; k < steps; ++k) matrix = matrix.map(row => chain.advance(row));
487 this.emit("POWER", a[0], "k=" + steps);
488 this.display(chain.names, matrix);
489 }

500 recurrence(a: string[]) {
501 const chain = this.requireStochastic(a[0]);
502 const state = chain.state(a[1]);
503 const item = new Decomposition(chain).containing(state);
504 if (!item.closed) throw new CommandError("TRANSIENT STATE " + a[1]);

506 const rho = distribution[item.members.indexOf(state)];
507 this.emit("RECURRENCE", a[0], a[1], "pi=" + valueText(rho),

509 }
510 absorb(a: string[]) {
511 const chain = this.requireStochastic(a[0]);
512 const flags = chain.names.map((_, i) => chain.absorbing(i));
513 if (!flags.some(Boolean)) throw new CommandError("NO ABSORBING STATE");
514 const decomposition = new Decomposition(chain);
515 for (let i = 0; i < chain.size; ++i) {
516 if (!flags[i] && !decomposition.reaches(i, flags))
517 throw new CommandError("NOT ABSORBING " + chain.names[i]);
518 }
519 const absorption = new Absorption(chain);
520 const transientNames = chain.labels(absorption.transient);
521 this.emit("ABSORB", a[0], "transient=" + absorption.transient.length,
522 "absorbing=" + absorption.absorbing.length);
523 if (!absorption.transient.length) return;
524 this.emit("TRANSIENT", ...transientNames);
525 this.emit("ABSORBING", ...chain.labels(absorption.absorbing));
526 transientNames.forEach((name, i) => this.emit(
527 "N", name, ...absorption.fundamental[i].map(fractionText)));
528 transientNames.forEach((name, i) => this.emit(
529 "STEPS", name, "mean=" + valueText(absorption.steps[i]),
530 "var=" + valueText(absorption.variance(i))));
531 transientNames.forEach((name, i) => this.emit(
532 "B", name, ...absorption.probabilities[i].map(fractionText)));
533 }
534 hitting(a: string[]) {
535 const chain = this.requireStochastic(a[0]);
536 const targets = membership(chain.size, chain.distinctStates(a.slice(1)));
537 const probabilities = hittingProbabilities(chain, new Decomposition(chain), targets);
538 const times = hittingTimes(chain, targets, probabilities);
539 const names = chain.names.filter((_, i) => targets[i]);
540 this.emit("HITTING", a[0], "targets=" + names.join(","));
541 chain.names.forEach((name, i) => this.emit(
542 "HIT", name, "prob=" + valueText(probabilities[i]),
543 "time=" + timeText(times[i])));
544 }
545 passage(a: string[]) {
546 const chain = this.requireStochastic(a[0]);
547 const decomposition = new Decomposition(chain);
548 const pi = irreducibleDistribution(chain, decomposition);
549 const times = chain.names.map(() => chain.names.map(() => ZERO));
550 for (let j = 0; j < chain.size; ++j) {
551 const targets = membership(chain.size, [j]);
552 const probabilities = hittingProbabilities(chain, decomposition, targets);
553 const column = hittingTimes(chain, targets, probabilities);
554 for (let i = 0; i < chain.size; ++i) {
555 times[i][j] = i === j ? ONE.div(pi[j]) : column[i];
556 }
557 }
558 const kemeny = sum(pi.slice(1).map((rho, j) => rho.mul(times[0][j + 1])));
559 this.emit("PASSAGE", a[0], "kemeny=" + valueText(kemeny));
560 this.display(chain.names, times, "M");
561 }

569 const first = [current[x]];

573 chain.matrix[i][j].mul(current[j]) : ZERO)));
574 first.push(current[x]);
575 }

577 "total=" + valueText(sum(first)));
578 let cumulative = ZERO;
579 first.forEach((value, i) => {

581 this.emit("FP", i + 1, valueText(value), "cum=" + valueText(cumulative));
582 });

597 mostLikely(a: string[]) {

599 const start = chain.state(a[1]), steps = Number(a[2]);
600 if (steps === 0) throw new CommandError("BAD STEPS");
601 const best = Array.from({length: steps + 1}, () => Array(chain.size).fill(ONE));
602 for (let t = steps - 1; t >= 0; --t) {
603 best[t] = chain.names.map((_, i) =>
604 chain.names.map((__, j) => chain.matrix[i][j].mul(best[t + 1][j]))
605 .reduce((a, b) => a.cmp(b) >= 0 ? a : b));
606 }

609 const current = path[path.length - 1];

611 chain.matrix[current][j].mul(best[t + 1][j]).eq(best[t][current]));
612 path.push(next);
613 }
614 this.emit("MOSTLIKELY", a[0], a[1], "k=" + steps,
615 "prob=" + valueText(best[0][start]));
616 this.emit("PATH", ...chain.labels(path));

618 reversible(a: string[]) {
619 const chain = this.requireStochastic(a[0]);
620 const pi = irreducibleDistribution(chain, new Decomposition(chain));
621 const violations = [];
622 for (let i = 0; i < chain.size; ++i) {
623 for (let j = i + 1; j < chain.size; ++j) {
624 const forward = pi[i].mul(chain.matrix[i][j]);
625 const backward = pi[j].mul(chain.matrix[j][i]);
626 if (!forward.eq(backward)) violations.push([i, j, forward, backward]);
627 }
628 }
629 if (!violations.length) {
630 this.emit("REVERSIBLE", a[0], "yes");
631 return;
632 }
633 this.emit("REVERSIBLE", a[0], "no", "violations=" + violations.length);
634 for (const [i, j, forward, backward] of violations) {
635 this.emit("VIOLATION", chain.names[i], chain.names[j],
636 valueText(forward), valueText(backward));
637 }

639 reverse(a: string[]) {

641 const chain = this.requireStochastic(name);
642 const pi = irreducibleDistribution(chain, new Decomposition(chain));
643 const reversed = new Chain(chain.names);
644 for (let i = 0; i < chain.size; ++i) {
645 for (let j = 0; j < chain.size; ++j) {
646 reversed.matrix[i][j] = pi[j].mul(chain.matrix[j][i]).div(pi[i]);
647 }
648 }
649 this.chains.set(reversedName, reversed);
650 const same = reversed.matrix.every((row, i) =>
651 row.every((value, j) => value.eq(chain.matrix[i][j])));
652 this.emit("REVERSE", name, reversedName, "reversible=" + answer(same));
653 this.display(reversed.names, reversed.matrix);
654 }
655 mixing(a: string[]) {
656 const chain = this.requireStochastic(a[0]);
657 const steps = Number(a[1]);
658 if (steps === 0) throw new CommandError("BAD STEPS");
659 const epsilon = parseProbability(a[2]);
660 const pi = irreducibleDistribution(chain, new Decomposition(chain));
661 let matrix = identity(chain.size), mixingTime = null;
662 const lines = [];
663 for (let t = 1; t <= steps; ++t) {
664 matrix = matrix.map(row => chain.advance(row));
665 const distances = matrix.map(row => sum(row.map((value, j) =>
666 value.sub(pi[j]).abs())).div(2));
667 const largest = distances.reduce((a, b) => a.cmp(b) >= 0 ? a : b);
668 const worst = distances.findIndex(distance => distance.eq(largest));
669 if (mixingTime === null && largest.cmp(epsilon) <= 0) mixingTime = t;
670 lines.push([t, largest, chain.names[worst]]);
671 }
672 this.emit("MIXING", a[0], "k=" + steps, "eps=" + fractionText(epsilon),
673 "time=" + (mixingTime === null ? "none" : mixingTime));
674 for (const [t, value, name] of lines) {
675 this.emit("TV", t, valueText(value), "from=" + name);
676 }
677 }
678 lump(a: string[]) {
679 const [name, lumpedName, partition] = a;
680 const chain = this.requireStochastic(name);
681 const seen = Array(chain.size).fill(false);
682 const blocks = [];
683 for (const group of partition.split("|")) {
684 const block = [];
685 for (const text of group.split(",")) {
686 const state = chain.state(text);
687 if (seen[state]) throw new CommandError("BAD PARTITION");
688 seen[state] = true;
689 block.push(state);
690 }
691 block.sort((a, b) => a - b);
692 blocks.push(block);
693 }
694 if (!seen.every(Boolean)) throw new CommandError("BAD PARTITION");
695 blocks.sort((a, b) => a[0] - b[0]);
696 for (const block of blocks) {
697 const representative = block[0];
698 for (const target of blocks) {
699 const expected = chain.blockSum(representative, target);
700 for (const i of block) {
701 const actual = chain.blockSum(i, target);
702 if (!actual.eq(expected)) {
703 this.emit("LUMP", name, "not", "lumpable",
704 "block=" + chain.names[representative],
705 "target=" + chain.names[target[0]],
706 chain.names[representative] + "=" + valueText(expected),
707 chain.names[i] + "=" + valueText(actual));
708 return;
709 }
710 }
711 }
712 }
713 const lumped = new Chain(blocks.map(block => chain.names[block[0]]));
714 lumped.matrix = blocks.map(block => blocks.map(target =>
715 chain.blockSum(block[0], target)));
716 this.chains.set(lumpedName, lumped);
717 this.emit("LUMP", name, lumpedName, "blocks=" + blocks.length);
718 this.display(lumped.names, lumped.matrix);
719 }
720 reward(a: string[]) {
721 const chain = this.requireStochastic(a[0]);
722 const texts = a.slice(2);
723 if (texts.length !== chain.size) throw new CommandError("BAD LENGTH");
724 const gamma = parseProbability(a[1]);
725 if (gamma.eq(ONE)) throw new CommandError("BAD DISCOUNT");
726 const rewards = texts.map(text => new Fraction(BigInt(text)));
727 const matrix = chain.names.map((_, i) =>
728 chain.names.map((__, j) => delta(i, j).sub(gamma.mul(chain.matrix[i][j])))
729 .concat([rewards[i]]));
730 const values = eliminate(matrix, chain.size).map(row => row[0]);
731 const decomposition = new Decomposition(chain);
732 const gains = chain.names.map(() => ZERO);
733 for (const item of decomposition.closedClasses()) {
734 const distribution = stationaryDistribution(chain, item.members);
735 const gain = sum(distribution.map((rho, j) => rho.mul(rewards[item.members[j]])));
736 const targets = membership(chain.size, item.members);
737 const probabilities = hittingProbabilities(chain, decomposition, targets);
738 for (let i = 0; i < chain.size; ++i) {
739 gains[i] = gains[i].add(probabilities[i].mul(gain));
740 }
741 }
742 this.emit("REWARD", a[0], "gamma=" + fractionText(gamma));
743 chain.names.forEach((name, i) => this.emit(
744 "VALUE", name, "discounted=" + valueText(values[i]),
745 "gain=" + valueText(gains[i])));
746 }
747 restrict(a: string[]) {
748 const [name, restrictedName, ...texts] = a;
749 const chain = this.requireStochastic(name);
750 const inside = membership(chain.size, chain.distinctStates(texts));
751 for (let i = 0; i < chain.size; ++i) {
752 if (inside[i] && chain.names.some((_, j) => chain.edge(i, j) && !inside[j]))
753 throw new CommandError("NOT CLOSED " + chain.names[i]);
754 }
755 const members = chain.names.map((_, i) => i).filter(i => inside[i]);
756 const restricted = new Chain(chain.labels(members));
757 restricted.matrix = members.map(i => members.map(j => chain.matrix[i][j]));
758 this.chains.set(restrictedName, restricted);
759 this.emit("RESTRICT", name, restrictedName, "states=" + members.length);
760 this.display(restricted.names, restricted.matrix);
761 }
762 simulate(a: string[]) {
763 const [name, source] = a;
764 const chain = this.requireStochastic(name);
765 const start = chain.state(source);
766 const steps = Number(a[2]), seed = Number(a[3]);
767 if (steps === 0) throw new CommandError("BAD STEPS");
768 if (seed === 0 || seed >= 2147483647) throw new CommandError("BAD SEED");
769 let random = seed;
770 const path = [start];
771 for (let t = 0; t < steps; ++t) {
772 const row = chain.matrix[path[path.length - 1]];
773 const common = row.map(value => value.d).reduce(lcm);
774 random = 48271 * random % 2147483647;
775 const threshold = BigInt(random) % common;
776 let cumulative = ZERO;
777 for (let j = 0; j < chain.size; ++j) {
778 cumulative = cumulative.add(row[j]);
779 if (cumulative.n * common > threshold * cumulative.d) {
780 path.push(j);
781 break;
782 }
783 }
784 }
785 this.emit("SIMULATE", name, source, "steps=" + steps, "seed=" + seed,
786 "end=" + chain.names[path[path.length - 1]]);
787 this.emit("PATH", ...chain.labels(path));
788 chain.names.forEach((state, i) => this.emit(
789 "VISIT", state, path.filter(value => value === i).length));
790 }
