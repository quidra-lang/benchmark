7function gcd(int $a, int $b): int
8{
9 $a = abs($a);
10 $b = abs($b);
11 while ($b !== 0) [$a, $b] = [$b, $a % $b];
12 return $a;
13}

15final readonly class Fraction
16{
17 public int $numerator;
18 public int $denominator;
19
20 public function __construct(int $numerator, int $denominator = 1)
21 {
22 if ($denominator < 0) [$numerator, $denominator] = [-$numerator, -$denominator];
23 $divisor = gcd($numerator, $denominator);
24 $this->numerator = intdiv($numerator, $divisor);
25 $this->denominator = intdiv($denominator, $divisor);
26 }
27
28 public static function zero(): self { static $value; return $value ??= new self(0); }
29 public static function one(): self { static $value; return $value ??= new self(1); }
30 public static function two(): self { static $value; return $value ??= new self(2); }
31 public function add(self $other): self
32 {
33 return new self($this->numerator * $other->denominator + $other->numerator * $this->denominator,
34 $this->denominator * $other->denominator);
35 }
36 public function subtract(self $other): self
37 {
38 return new self($this->numerator * $other->denominator - $other->numerator * $this->denominator,
39 $this->denominator * $other->denominator);
40 }
41 public function multiply(self $other): self
42 {
43 return new self($this->numerator * $other->numerator, $this->denominator * $other->denominator);
44 }
45 public function divide(self $other): self
46 {
47 return new self($this->numerator * $other->denominator, $this->denominator * $other->numerator);
48 }
49 public function absolute(): self { return new self(abs($this->numerator), $this->denominator); }
50 public function isZero(): bool { return $this->numerator === 0; }
51 public function equals(self $other): bool
52 {
53 return $this->numerator === $other->numerator && $this->denominator === $other->denominator;
54 }
55 public function compareTo(self $other): int
56 {
57 return $this->numerator * $other->denominator <=> $other->numerator * $this->denominator;
58 }
59 public function __toString(): string { return "$this->numerator/$this->denominator"; }
60 public function value(): string
61 {
62 $whole = intdiv($this->numerator, $this->denominator);
63 $decimal = intdiv(($this->numerator % $this->denominator) * 1000000, $this->denominator);
64 return $this . '(' . $whole . '.' . str_pad((string) $decimal, 6, '0', STR_PAD_LEFT) . ')';
65 }
66}

68function probability(string $text): Fraction
69{
70 $parts = explode('/', $text);
71 $numerator = (int) $parts[0];
72 $denominator = isset($parts[1]) ? (int) $parts[1] : 1;
73 if ($denominator === 0 || $numerator > $denominator) throw new CommandError('BAD PROBABILITY');
74 return new Fraction($numerator, $denominator);
75}

101function advance(array $distribution, array $transitions): array
102{
103 $size = count($distribution);
104 $result = vector($size);
105 for ($j = 0; $j < $size; ++$j)
106 for ($i = 0; $i < $size; ++$i)
107 $result[$j] = $result[$j]->add($distribution[$i]->multiply($transitions[$i][$j]));
108 return $result;
109}

118function solve(array $rows): array
119{
120 $size = count($rows);
121 if ($size === 0) return [];
122 $width = count($rows[0]);
123 for ($column = 0; $column < $size; ++$column) {
124 $pivotRow = $column;
125 while ($rows[$pivotRow][$column]->isZero()) ++$pivotRow;
126 [$rows[$column], $rows[$pivotRow]] = [$rows[$pivotRow], $rows[$column]];
127 $pivot = $rows[$column][$column];
128 for ($j = 0; $j < $width; ++$j) $rows[$column][$j] = $rows[$column][$j]->divide($pivot);
129 for ($i = 0; $i < $size; ++$i) {
130 if ($i === $column) continue;
131 $factor = $rows[$i][$column];
132 if ($factor->isZero()) continue;
133 for ($j = 0; $j < $width; ++$j)
134 $rows[$i][$j] = $rows[$i][$j]->subtract($factor->multiply($rows[$column][$j]));
135 }
136 }
137 return array_map(static fn(array $row): array => array_slice($row, $size), $rows);
138}

158final class Chain
159{
160 private readonly array $positions;
161 public function __construct(public readonly array $names, public array $p)
162 {
163 $this->positions = array_flip($names);
164 }
165 public function size(): int { return count($this->names); }
166 public function labels(array $states, string $separator = ' '): string
167 {
168 return implode($separator, array_map(fn(int $i): string => $this->names[$i], $states));
169 }
170 public function state(string $name): int
171 {
172 return $this->positions[$name] ?? throw new CommandError("UNKNOWN STATE $name");
173 }
174 public function stateSet(array $names): array
175 {
176 $seen = [];
177 foreach ($names as $name) {
178 $state = $this->state($name);
179 if (isset($seen[$state])) throw new CommandError("DUPLICATE STATE $name");
180 $seen[$state] = true;
181 }
182 $states = array_keys($seen);
183 sort($states, SORT_NUMERIC);
184 return $states;
185 }
186 public function stochastic(): void
187 {
188 foreach ($this->p as $i => $row)
189 if (!sum($row)->equals(Fraction::one())) throw new CommandError('NOT STOCHASTIC ' . $this->names[$i]);
190 }
191 public function graph(): Graph
192 {
193 $size = $this->size();
194 $reach = [];
195 for ($i = 0; $i < $size; ++$i)
196 for ($j = 0; $j < $size; ++$j) $reach[$i][$j] = $i === $j || !$this->p[$i][$j]->isZero();
197 for ($k = 0; $k < $size; ++$k)
198 for ($i = 0; $i < $size; ++$i)
199 for ($j = 0; $j < $size; ++$j) $reach[$i][$j] = $reach[$i][$j] || $reach[$i][$k] && $reach[$k][$j];
200 $assigned = [];
201 $classes = [];
202 for ($first = 0; $first < $size; ++$first) {
203 if (isset($assigned[$first])) continue;
204 $states = [];
205 for ($j = 0; $j < $size; ++$j)
206 if ($reach[$first][$j] && $reach[$j][$first]) $states[] = $j;
207 $members = array_fill_keys($states, true);
208 $closed = true;
209 foreach ($states as $i) {
210 $assigned[$i] = true;
211 for ($j = 0; $j < $size; ++$j)
212 if (!isset($members[$j]) && !$this->p[$i][$j]->isZero()) $closed = false;
213 }
214 $levels = array_fill(0, $size, -1);
215 $levels[$first] = 0;
216 $queue = [$first];
217 for ($head = 0; $head < count($queue); ++$head) {
218 $i = $queue[$head];
219 foreach ($states as $j)
220 if ($levels[$j] < 0 && !$this->p[$i][$j]->isZero()) {
221 $levels[$j] = $levels[$i] + 1;
222 $queue[] = $j;
223 }
224 }
225 $period = 0;
226 foreach ($states as $i)
227 foreach ($states as $j)
228 if (!$this->p[$i][$j]->isZero()) $period = gcd($period, $levels[$i] + 1 - $levels[$j]);
229 $classes[] = new Component($states, $closed, $period, $levels);
230 }
231 return new Graph($reach, $classes);
232 }
233 public function stationary(array $states): array
234 {
235 $count = count($states);
236 $equations = matrix($count, $count + 1);
237 for ($row = 0; $row < $count - 1; ++$row)
238 for ($col = 0; $col < $count; ++$col)
239 $equations[$row][$col] = $this->p[$states[$col]][$states[$row]]->subtract(delta($col, $row));
240 $equations[$count - 1] = array_fill(0, $count + 1, Fraction::one());
241 $solution = solve($equations);
242 $distribution = vector($this->size());
243 foreach ($states as $i => $state) $distribution[$state] = $solution[$i][0];
244 return $distribution;
245 }
246 public function uniqueStationary(): array
247 {
248 $classes = $this->graph()->classes;
249 if (count($classes) !== 1) throw new CommandError('NOT IRREDUCIBLE');
250 return $this->stationary($classes[0]->states);
251 }
252 public function blockSum(int $from, array $targets): Fraction
253 {
254 $total = Fraction::zero();
255 foreach ($targets as $target) $total = $total->add($this->p[$from][$target]);
256 return $total;
257 }
258 public function hitting(array $targets, bool $withTimes = true): Hitting
259 {
260 $reach = $this->graph()->reach;
261 $inside = array_fill_keys($targets, true);
262 $probability = vector($this->size());
263 $time = array_fill(0, $this->size(), null);
264 $unknown = [];
265 for ($i = 0; $i < $this->size(); ++$i) {
266 if (isset($inside[$i])) {
267 $probability[$i] = Fraction::one();
268 $time[$i] = Fraction::zero();
269 } else {
270 foreach ($targets as $j)
271 if ($reach[$i][$j]) { $unknown[] = $i; break; }
272 }
273 }
274 if ($unknown !== []) {
275 $count = count($unknown);
276 $equations = matrix($count, $count + 1);
277 foreach ($unknown as $i => $from) {
278 foreach ($unknown as $j => $to) $equations[$i][$j] = delta($i, $j)->subtract($this->p[$from][$to]);
279 $equations[$i][$count] = $this->blockSum($from, $targets);
280 }
281 $solution = solve($equations);
282 foreach ($unknown as $i => $state) $probability[$state] = $solution[$i][0];
283 }
284 if ($withTimes) {
285 $certain = [];
286 foreach ($unknown as $i) if ($probability[$i]->equals(Fraction::one())) $certain[] = $i;
287 if ($certain !== []) {
288 $count = count($certain);
289 $equations = matrix($count, $count + 1);
290 foreach ($certain as $i => $from) {
291 foreach ($certain as $j => $to) $equations[$i][$j] = delta($i, $j)->subtract($this->p[$from][$to]);
292 $equations[$i][$count] = Fraction::one();
293 }
294 $solution = solve($equations);
295 foreach ($certain as $i => $state) $time[$state] = $solution[$i][0];
296 }
297 }
298 return new Hitting($probability, $time);
299 }
300 public function absorption(): Absorption
301 {
302 $absorbing = $transient = [];
303 for ($i = 0; $i < $this->size(); ++$i) {
304 if ($this->p[$i][$i]->equals(Fraction::one())) $absorbing[] = $i;
305 else $transient[] = $i;
306 }
307 if ($absorbing === []) throw new CommandError('NO ABSORBING STATE');
308 $reach = $this->graph()->reach;
309 foreach ($transient as $i) {
310 $found = false;
311 foreach ($absorbing as $j) if ($reach[$i][$j]) { $found = true; break; }
312 if (!$found) throw new CommandError('NOT ABSORBING ' . $this->names[$i]);
313 }
314 $count = count($transient);
315 $equations = matrix($count, 2 * $count);
316 foreach ($transient as $i => $from)
317 foreach ($transient as $j => $to) {
318 $equations[$i][$j] = delta($i, $j)->subtract($this->p[$from][$to]);
319 $equations[$i][$count + $j] = delta($i, $j);
320 }
321 $fundamental = solve($equations);
322 $mean = array_map('sum', $fundamental);
323 $variance = vector($count);
324 $probability = matrix($count, count($absorbing));
325 for ($i = 0; $i < $count; ++$i) {
326 for ($j = 0; $j < $count; ++$j)
327 $variance[$i] = $variance[$i]->add(Fraction::two()->multiply($fundamental[$i][$j])
328 ->subtract(delta($i, $j))->multiply($mean[$j]));
329 $variance[$i] = $variance[$i]->subtract($mean[$i]->multiply($mean[$i]));
330 foreach ($absorbing as $j => $target)
331 foreach ($transient as $k => $state)
332 $probability[$i][$j] = $probability[$i][$j]->add($fundamental[$i][$k]->multiply($this->p[$state][$target]));
333 }
334 return new Absorption($transient, $absorbing, $fundamental, $probability, $mean, $variance);
335 }

353 private function create(array $words): void
354 {
355 $names = array_slice($words, 2);
356 if (count($names) > 10) throw new CommandError('TOO MANY STATES');
357 $seen = [];
358 foreach ($names as $name) {
359 if (isset($seen[$name])) throw new CommandError("DUPLICATE STATE $name");
360 $seen[$name] = true;
361 }
362 $this->chains[$words[1]] = new Chain($names, matrix(count($names), count($names)));
363 $this->emit('CHAIN', $words[1], 'states=' . count($names));

365 private function build(array $words): void
366 {
367 $order = (int) $words[2];
368 if ($order !== 1 && $order !== 2) throw new CommandError('BAD ORDER');
369 $tokens = array_slice($words, 3);
370 if (count($tokens) <= $order) throw new CommandError('TOO FEW TOKENS');
371 $observations = [];
372 for ($i = 0; $i < count($tokens) - $order + 1; ++$i)
373 $observations[] = $order === 1 ? $tokens[$i] : $tokens[$i] . '.' . $tokens[$i + 1];
374 $names = $positions = [];
375 foreach ($observations as $state)
376 if (!isset($positions[$state])) {
377 $positions[$state] = count($names);
378 $names[] = $state;
379 }
380 $size = count($names);
381 if ($size > 10) throw new CommandError('TOO MANY STATES');
382 $chain = new Chain($names, matrix($size, $size));
383 $counts = array_fill(0, $size, array_fill(0, $size, 0));
384 $totals = array_fill(0, $size, 0);
385 for ($i = 1; $i < count($observations); ++$i) {
386 $from = $positions[$observations[$i - 1]];
387 $to = $positions[$observations[$i]];
388 ++$counts[$from][$to];
389 ++$totals[$from];
390 }
391 for ($i = 0; $i < $size; ++$i) {
392 if ($totals[$i] === 0) $chain->p[$i][$i] = Fraction::one();
393 else for ($j = 0; $j < $size; ++$j) $chain->p[$i][$j] = new Fraction($counts[$i][$j], $totals[$i]);
394 }
395 $this->chains[$words[1]] = $chain;
396 $this->emit('BUILD', $words[1], 'order=' . $order, 'states=' . $size, 'transitions=' . (count($observations) - 1));
397 for ($i = 0; $i < $size; ++$i) {
398 $targets = [];
399 for ($j = 0; $j < $size; ++$j) if ($counts[$i][$j] > 0) $targets[] = $names[$j] . ':' . $counts[$i][$j];
400 $this->emit('FROM', $names[$i], 'total=' . $totals[$i], $totals[$i] === 0 ? 'absorbing' : implode(' ', $targets));
401 }
402 }
403 private function gambler(array $words): void
404 {
405 $goal = (int) $words[2];
406 $start = (int) $words[3];
407 if ($goal === 0 || $goal > 9) throw new CommandError('BAD SIZE');
408 if ($start > $goal) throw new CommandError('BAD START');
409 $up = probability($words[4]);
410 if ($up->isZero() || $up->equals(Fraction::one())) throw new CommandError('BAD PROBABILITY');
411 $names = [];
412 for ($i = 0; $i <= $goal; ++$i) $names[] = 's' . $i;
413 $chain = new Chain($names, matrix($goal + 1, $goal + 1));
414 $chain->p[0][0] = $chain->p[$goal][$goal] = Fraction::one();
415 for ($i = 1; $i < $goal; ++$i) {
416 $chain->p[$i][$i - 1] = Fraction::one()->subtract($up);
417 $chain->p[$i][$i + 1] = $up;
418 }
419 $result = $chain->absorption();
420 $ruin = $duration = vector($goal + 1);
421 $ruin[0] = Fraction::one();
422 foreach ($result->transient as $i => $state) {
423 $ruin[$state] = $result->probability[$i][0];
424 $duration[$state] = $result->mean[$i];
425 }
426 $this->chains[$words[1]] = $chain;
427 $this->emit('GAMBLER', $words[1], 'g=' . $goal, 'p=' . $up, 'start=s' . $start,
428 'ruin=' . $ruin[$start]->value(), 'duration=' . $duration[$start]->value());
429 for ($i = 0; $i <= $goal; ++$i)
430 $this->emit('CAPITAL', 's' . $i, 'ruin=' . $ruin[$i]->value(),
431 'win=' . Fraction::one()->subtract($ruin[$i])->value(), 'duration=' . $duration[$i]->value());
432 }
433 private function classes(string $name, Chain $chain): void
434 {
435 $classes = $chain->graph()->classes;
436 $this->emit('CLASSES', $name, 'count=' . count($classes));
437 foreach ($classes as $i => $component) {
438 $this->emit('CLASS', $i + 1, $component->closed ? 'RECURRENT' : 'TRANSIENT',
439 'period=' . $component->period, 'size=' . count($component->states), $chain->labels($component->states));
440 if ($component->period >= 2)
441 for ($r = 0; $r < $component->period; ++$r) {
442 $states = [];
443 foreach ($component->states as $state)
444 if ($component->levels[$state] % $component->period === $r) $states[] = $state;
445 $this->emit('CYCLIC', $r, $chain->labels($states));
446 }
447 }

449 private function classify(string $name, Chain $chain): void
450 {
451 $graph = $chain->graph();
452 $size = $chain->size();
453 $absorbing = [];
454 for ($i = 0; $i < $size; ++$i) if ($chain->p[$i][$i]->equals(Fraction::one())) $absorbing[] = $i;
455 $allAbsorbed = $absorbing !== [];
456 for ($i = 0; $i < $size; ++$i) {
457 $found = false;
458 foreach ($absorbing as $j) if ($graph->reach[$i][$j]) { $found = true; break; }
459 if (!$found) $allAbsorbed = false;
460 }
461 $edges = [];
462 for ($i = 0; $i < $size; ++$i)
463 for ($j = 0; $j < $size; ++$j) $edges[$i][$j] = !$chain->p[$i][$j]->isZero();
464 $power = $edges;
465 $exponent = null;
466 $bound = ($size - 1) * ($size - 1) + 1;
467 for ($t = 1; $t <= $bound; ++$t) {
468 $all = true;
469 foreach ($power as $row) foreach ($row as $positive) if (!$positive) $all = false;
470 if ($all) { $exponent = $t; break; }
471 $next = array_fill(0, $size, array_fill(0, $size, false));
472 for ($i = 0; $i < $size; ++$i)
473 for ($j = 0; $j < $size; ++$j)
474 for ($k = 0; $k < $size; ++$k) $next[$i][$j] = $next[$i][$j] || $power[$i][$k] && $edges[$k][$j];
475 $power = $next;
476 }
477 $this->emit('CLASSIFY', $name, 'absorbing_states=' . count($absorbing), 'absorbing=' . self::yes($allAbsorbed),
478 'ergodic=' . self::yes(count($graph->classes) === 1), 'regular=' . self::yes($exponent !== null),
479 'exponent=' . ($exponent ?? '-'));
480 }
481 private function step(array $words, Chain $chain): void
482 {
483 if (ctype_digit($words[2][0])) {
484 $values = explode(',', $words[2]);
485 if (count($values) !== $chain->size()) throw new CommandError('BAD LENGTH');
486 $distribution = array_map('probability', $values);
487 if (!sum($distribution)->equals(Fraction::one())) throw new CommandError('BAD DISTRIBUTION');
488 } else {
489 $distribution = vector($chain->size());
490 $distribution[$chain->state($words[2])] = Fraction::one();
491 }
492 $steps = (int) $words[3];
493 for ($t = 0; $t < $steps; ++$t) $distribution = advance($distribution, $chain->p);
494 $this->emit('STEP', $words[1], 'k=' . $steps);
495 foreach ($chain->names as $i => $name) $this->emit('PROB', $name, $distribution[$i]->value());
496 }
497 private function stationary(string $name, Chain $chain): void
498 {
499 $classes = $chain->graph()->classes;
500 $count = count(array_filter($classes, static fn(Component $component): bool => $component->closed));
501 $this->emit('STATIONARY', $name, 'count=' . $count);
502 $number = 0;
503 foreach ($classes as $i => $component) {
504 if (!$component->closed) continue;
505 $distribution = $chain->stationary($component->states);
506 $this->emit('DIST', ++$number, 'class=' . ($i + 1), 'size=' . count($component->states));
507 foreach ($component->states as $state) $this->emit('PI', $chain->names[$state], $distribution[$state]->value());
508 }
509 }
510 private function recurrence(array $words, Chain $chain): void
511 {
512 $state = $chain->state($words[2]);
513 foreach ($chain->graph()->classes as $component) {
514 if (!in_array($state, $component->states, true)) continue;
515 if (!$component->closed) throw new CommandError('TRANSIENT STATE ' . $words[2]);
516 $rho = $chain->stationary($component->states)[$state];
517 $this->emit('RECURRENCE', $words[1], $words[2], 'pi=' . $rho->value(),
518 'mean=' . Fraction::one()->divide($rho)->value(), 'period=' . $component->period);
519 return;
520 }

526 if ($result->transient === []) return;
527 $this->emit('TRANSIENT', $chain->labels($result->transient));
528 $this->emit('ABSORBING', $chain->labels($result->absorbing));
529 foreach ($result->transient as $i => $state)

531 foreach ($result->transient as $i => $state)
532 $this->emit('STEPS', $chain->names[$state], 'mean=' . $result->mean[$i]->value(), 'var=' . $result->variance[$i]->value());
533 foreach ($result->transient as $i => $state)
534 $this->emit('B', $chain->names[$state], implode(' ', array_map('strval', $result->probability[$i])));
535 }

544 private function passage(string $name, Chain $chain): void
545 {
546 $pi = $chain->uniqueStationary();
547 $size = $chain->size();
548 $values = matrix($size, $size);
549 for ($j = 0; $j < $size; ++$j) {
550 $times = $chain->hitting([$j])->time;
551 for ($i = 0; $i < $size; ++$i) $values[$i][$j] = $i === $j ? Fraction::one()->divide($pi[$j]) : $times[$i];
552 }
553 $kemeny = Fraction::zero();
554 for ($j = 1; $j < $size; ++$j) $kemeny = $kemeny->add($pi[$j]->multiply($values[0][$j]));
555 $this->emit('PASSAGE', $name, 'kemeny=' . $kemeny->value());
556 $this->display($chain, $values, 'M');
557 }
558 private function firstPassage(array $words, Chain $chain): void
559 {
560 $from = $chain->state($words[2]);
561 $target = $chain->state($words[3]);
562 $steps = (int) $words[4];
563 if ($steps === 0) throw new CommandError('BAD STEPS');
564 $current = [];
565 foreach ($chain->p as $row) $current[] = $row[$target];
566 $entries = [];
567 $total = Fraction::zero();
568 for ($t = 1; $t <= $steps; ++$t) {
569 $total = $total->add($current[$from]);
570 $entries[] = [$current[$from], $total];
571 if ($t < $steps) {
572 $next = vector($chain->size());
573 for ($i = 0; $i < $chain->size(); ++$i)
574 for ($j = 0; $j < $chain->size(); ++$j)
575 if ($j !== $target) $next[$i] = $next[$i]->add($chain->p[$i][$j]->multiply($current[$j]));
576 $current = $next;
577 }
578 }
579 $this->emit('FIRSTPASSAGE', $words[1], $words[2], $words[3], 'k=' . $steps, 'total=' . $total->value());
580 foreach ($entries as $i => [$value, $cumulative]) $this->emit('FP', $i + 1, $value->value(), 'cum=' . $cumulative->value());
581 }
582 private function pathProbability(array $words, Chain $chain): void
583 {
584 $states = array_map([$chain, 'state'], array_slice($words, 2));
585 $product = Fraction::one();
586 $zeroAt = null;
587 for ($i = 1; $i < count($states); ++$i) {
588 $product = $product->multiply($chain->p[$states[$i - 1]][$states[$i]]);
589 if ($product->isZero()) $zeroAt ??= $i;
590 }
591 $line = 'PATHPROB ' . $words[1] . ' steps=' . (count($states) - 1) . ' prob=' . $product->value();
592 if ($zeroAt !== null) $line .= ' zero_at=' . $zeroAt;
593 $this->emit($line);
594 }
595 private function mostLikely(array $words, Chain $chain): void
596 {
597 $start = $chain->state($words[2]);
598 $steps = (int) $words[3];
599 if ($steps === 0) throw new CommandError('BAD STEPS');
600 $best = matrix($steps + 1, $chain->size());
601 $best[$steps] = array_fill(0, $chain->size(), Fraction::one());
602 for ($t = $steps - 1; $t >= 0; --$t)
603 for ($i = 0; $i < $chain->size(); ++$i)
604 for ($j = 0; $j < $chain->size(); ++$j) {
605 $candidate = $chain->p[$i][$j]->multiply($best[$t + 1][$j]);
606 if ($candidate->compareTo($best[$t][$i]) > 0) $best[$t][$i] = $candidate;
607 }
608 $path = [$start];
609 $state = $start;
610 for ($t = 0; $t < $steps; ++$t)
611 for ($j = 0; $j < $chain->size(); ++$j)
612 if ($chain->p[$state][$j]->multiply($best[$t + 1][$j])->equals($best[$t][$state])) {
613 $state = $j;
614 $path[] = $j;
615 break;
616 }
617 $this->emit('MOSTLIKELY', $words[1], $words[2], 'k=' . $steps, 'prob=' . $best[0][$start]->value());
618 $this->emit('PATH', $chain->labels($path));
619 }
620 private function reversible(string $name, Chain $chain): void
621 {
622 $pi = $chain->uniqueStationary();
623 $violations = [];
624 for ($i = 0; $i < $chain->size(); ++$i)
625 for ($j = $i + 1; $j < $chain->size(); ++$j) {
626 $forward = $pi[$i]->multiply($chain->p[$i][$j]);
627 $backward = $pi[$j]->multiply($chain->p[$j][$i]);
628 if (!$forward->equals($backward)) $violations[] = [$i, $j, $forward, $backward];
629 }
630 $this->emit('REVERSIBLE', $name, $violations === [] ? 'yes' : 'no violations=' . count($violations));
631 foreach ($violations as [$i, $j, $forward, $backward])
632 $this->emit('VIOLATION', $chain->names[$i], $chain->names[$j], $forward->value(), $backward->value());
633 }
634 private function reverse(array $words, Chain $chain): void
635 {
636 $pi = $chain->uniqueStationary();
637 $reversed = new Chain($chain->names, matrix($chain->size(), $chain->size()));
638 $same = true;
639 for ($i = 0; $i < $chain->size(); ++$i)
640 for ($j = 0; $j < $chain->size(); ++$j) {
641 $reversed->p[$i][$j] = $pi[$j]->multiply($chain->p[$j][$i])->divide($pi[$i]);
642 if (!$reversed->p[$i][$j]->equals($chain->p[$i][$j])) $same = false;
643 }
644 $this->chains[$words[2]] = $reversed;
645 $this->emit('REVERSE', $words[1], $words[2], 'reversible=' . self::yes($same));
646 $this->display($reversed);
647 }
648 private function mixing(array $words, Chain $chain): void
649 {
650 $steps = (int) $words[2];
651 if ($steps === 0) throw new CommandError('BAD STEPS');
652 $epsilon = probability($words[3]);
653 $pi = $chain->uniqueStationary();
654 $power = identity($chain->size());
655 $entries = [];
656 $mixingTime = null;
657 for ($t = 1; $t <= $steps; ++$t) {
658 $power = multiply($power, $chain->p);
659 $distance = Fraction::zero();
660 $from = 0;
661 for ($i = 0; $i < $chain->size(); ++$i) {
662 $current = Fraction::zero();
663 for ($j = 0; $j < $chain->size(); ++$j)
664 $current = $current->add($power[$i][$j]->subtract($pi[$j])->absolute());
665 $current = $current->divide(Fraction::two());
666 if ($current->compareTo($distance) > 0) [$distance, $from] = [$current, $i];
667 }
668 if ($distance->compareTo($epsilon) <= 0) $mixingTime ??= $t;
669 $entries[] = [$distance, $from];
670 }
671 $this->emit('MIXING', $words[1], 'k=' . $steps, 'eps=' . $epsilon, 'time=' . ($mixingTime ?? 'none'));
672 foreach ($entries as $i => [$distance, $from]) $this->emit('TV', $i + 1, $distance->value(), 'from=' . $chain->names[$from]);
673 }
674 private function lump(array $words, Chain $chain): void
675 {
676 $blocks = $seen = [];
677 foreach (explode('|', $words[3]) as $group) {
678 $block = [];
679 foreach (explode(',', $group) as $name) {
680 $state = $chain->state($name);
681 if (isset($seen[$state])) throw new CommandError('BAD PARTITION');
682 $seen[$state] = true;
683 $block[] = $state;
684 }
685 sort($block, SORT_NUMERIC);
686 $blocks[] = $block;
687 }
688 if (count($seen) !== $chain->size()) throw new CommandError('BAD PARTITION');
689 usort($blocks, static fn(array $a, array $b): int => $a[0] <=> $b[0]);
690 $names = array_map(fn(array $block): string => $chain->names[$block[0]], $blocks);
691 $lumped = new Chain($names, matrix(count($blocks), count($blocks)));
692 foreach ($blocks as $i => $block)
693 foreach ($blocks as $j => $target) {
694 $representative = $block[0];
695 $baseline = $chain->blockSum($representative, $target);
696 foreach ($block as $state) {
697 $value = $chain->blockSum($state, $target);
698 if (!$value->equals($baseline)) {
699 $this->emit('LUMP', $words[1], 'not lumpable', 'block=' . $chain->names[$representative],
700 'target=' . $chain->names[$target[0]], $chain->names[$representative] . '=' . $baseline->value(),
701 $chain->names[$state] . '=' . $value->value());
702 return;
703 }
704 }
705 $lumped->p[$i][$j] = $baseline;
706 }
707 $this->chains[$words[2]] = $lumped;
708 $this->emit('LUMP', $words[1], $words[2], 'blocks=' . count($blocks));
709 $this->display($lumped);
710 }
711 private function reward(array $words, Chain $chain): void
712 {
713 if (count($words) - 3 !== $chain->size()) throw new CommandError('BAD LENGTH');
714 $discount = probability($words[2]);
715 if ($discount->equals(Fraction::one())) throw new CommandError('BAD DISCOUNT');
716 $rewards = array_map(static fn(string $text): Fraction => new Fraction((int) $text), array_slice($words, 3));
717 $equations = matrix($chain->size(), $chain->size() + 1);
718 for ($i = 0; $i < $chain->size(); ++$i) {
719 for ($j = 0; $j < $chain->size(); ++$j)
720 $equations[$i][$j] = delta($i, $j)->subtract($discount->multiply($chain->p[$i][$j]));
721 $equations[$i][$chain->size()] = $rewards[$i];
722 }
723 $discounted = solve($equations);
724 $gain = vector($chain->size());
725 foreach ($chain->graph()->classes as $component) {
726 if (!$component->closed) continue;
727 $rho = $chain->stationary($component->states);
728 $average = Fraction::zero();
729 foreach ($component->states as $i) $average = $average->add($rho[$i]->multiply($rewards[$i]));
730 $hitting = $chain->hitting($component->states, false)->probability;
731 for ($i = 0; $i < $chain->size(); ++$i) $gain[$i] = $gain[$i]->add($hitting[$i]->multiply($average));
732 }
733 $this->emit('REWARD', $words[1], 'gamma=' . $discount);
734 foreach ($chain->names as $i => $name)
735 $this->emit('VALUE', $name, 'discounted=' . $discounted[$i][0]->value(), 'gain=' . $gain[$i]->value());
736 }
737 private function restrict(array $words, Chain $chain): void
738 {
739 $states = $chain->stateSet(array_slice($words, 3));
740 $inside = array_fill_keys($states, true);
741 foreach ($states as $i)
742 for ($j = 0; $j < $chain->size(); ++$j)
743 if (!isset($inside[$j]) && !$chain->p[$i][$j]->isZero()) throw new CommandError('NOT CLOSED ' . $chain->names[$i]);
744 $names = $p = [];
745 foreach ($states as $i) {
746 $names[] = $chain->names[$i];
747 $row = [];
748 foreach ($states as $j) $row[] = $chain->p[$i][$j];
749 $p[] = $row;
750 }
751 $restricted = new Chain($names, $p);
752 $this->chains[$words[2]] = $restricted;
753 $this->emit('RESTRICT', $words[1], $words[2], 'states=' . count($states));
754 $this->display($restricted);
755 }
756 private function simulate(array $words, Chain $chain): void
757 {
758 $state = $chain->state($words[2]);
759 $steps = (int) $words[3];
760 if ($steps === 0) throw new CommandError('BAD STEPS');
761 $seed = (int) $words[4];
762 if ($seed === 0 || $seed >= 2147483647) throw new CommandError('BAD SEED');
763 $random = $seed;
764 $path = [$state];
765 $visits = array_fill(0, $chain->size(), 0);
766 ++$visits[$state];
767 for ($t = 0; $t < $steps; ++$t) {
768 $denominator = 1;
769 foreach ($chain->p[$state] as $value)
770 $denominator = intdiv($denominator, gcd($denominator, $value->denominator)) * $value->denominator;
771 $random = 48271 * $random % 2147483647;
772 $sample = $random % $denominator;
773 $cumulative = Fraction::zero();
774 for ($j = 0; $j < $chain->size(); ++$j) {
775 $cumulative = $cumulative->add($chain->p[$state][$j]);
776 if ($cumulative->numerator * intdiv($denominator, $cumulative->denominator) > $sample) {
777 $state = $j;
778 break;
779 }
780 }
781 $path[] = $state;
782 ++$visits[$state];
783 }
784 $this->emit('SIMULATE', $words[1], $words[2], 'steps=' . $steps, 'seed=' . $seed, 'end=' . $chain->names[$state]);
785 $this->emit('PATH', $chain->labels($path));
786 foreach ($chain->names as $i => $name) $this->emit('VISIT', $name, $visits[$i]);
787 }
788 public function execute(string $line): void
789 {
790 $words = explode(' ', $line);
791 [$command, $name] = $words;
792 try {
793 if ($command === 'CHAIN') { $this->create($words); return; }
794 if ($command === 'BUILD') { $this->build($words); return; }
795 if ($command === 'GAMBLER') { $this->gambler($words); return; }
796 $chain = $this->chains[$name] ?? throw new CommandError("UNKNOWN CHAIN $name");
797 if (!in_array($command, ['P', 'ROW', 'SHOW', 'CHECK'], true)) $chain->stochastic();
798 switch ($command) {
799 case 'P':
800 $from = $chain->state($words[2]);
801 $to = $chain->state($words[3]);
802 $value = probability($words[4]);
803 $chain->p[$from][$to] = $value;
804 $this->emit('P', $name, $words[2], $words[3], (string) $value, 'rowsum=' . sum($chain->p[$from]));
805 break;
806 case 'ROW':
807 $from = $chain->state($words[2]);
808 if (count($words) - 3 !== $chain->size()) throw new CommandError('BAD LENGTH');
809 $row = array_map('probability', array_slice($words, 3));
810 $chain->p[$from] = $row;
811 $this->emit('ROW', $name, $words[2], 'rowsum=' . sum($row));
812 break;
813 case 'SHOW':
814 $this->emit('MATRIX', $name, 'states=' . $chain->size());
815 $this->display($chain);
816 break;
817 case 'CHECK':
818 $sums = array_map('sum', $chain->p);
819 $bad = [];
820 foreach ($sums as $i => $value) if (!$value->equals(Fraction::one())) $bad[] = $i;
821 $this->emit('CHECK', $name, $bad === [] ? 'ok' : 'bad=' . count($bad));
822 foreach ($bad as $i) $this->emit('BADROW', $chain->names[$i], 'sum=' . $sums[$i]);
823 break;
824 case 'CLASSES': $this->classes($name, $chain); break;
825 case 'CLASSIFY': $this->classify($name, $chain); break;
826 case 'STEP': $this->step($words, $chain); break;
827 case 'POWER':
828 $steps = (int) $words[2];
829 $power = identity($chain->size());
830 for ($t = 0; $t < $steps; ++$t) $power = multiply($power, $chain->p);
831 $this->emit('POWER', $name, 'k=' . $steps);
832 $this->display($chain, $power);
833 break;
834 case 'STATIONARY': $this->stationary($name, $chain); break;
835 case 'RECURRENCE': $this->recurrence($words, $chain); break;
836 case 'ABSORB': $this->absorb($name, $chain); break;
837 case 'HITTING': $this->hitting($words, $chain); break;
838 case 'PASSAGE': $this->passage($name, $chain); break;
839 case 'FIRSTPASSAGE': $this->firstPassage($words, $chain); break;
840 case 'PATHPROB': $this->pathProbability($words, $chain); break;
841 case 'MOSTLIKELY': $this->mostLikely($words, $chain); break;
842 case 'REVERSIBLE': $this->reversible($name, $chain); break;
843 case 'REVERSE': $this->reverse($words, $chain); break;
844 case 'MIXING': $this->mixing($words, $chain); break;
845 case 'LUMP': $this->lump($words, $chain); break;
846 case 'REWARD': $this->reward($words, $chain); break;
847 case 'RESTRICT': $this->restrict($words, $chain); break;
848 case 'SIMULATE': $this->simulate($words, $chain); break;
849 }
850 } catch (CommandError $error) {
851 $this->emit('ERROR', $error->getMessage());
852 }
853 }
