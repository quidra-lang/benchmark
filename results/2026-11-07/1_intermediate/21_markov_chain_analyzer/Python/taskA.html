61def eliminate(matrix, size):
62 for column in range(size):
63 pivot_row = column
64 while matrix[pivot_row][column] == 0:
65 pivot_row += 1
66 matrix[column], matrix[pivot_row] = matrix[pivot_row], matrix[column]
67 pivot = matrix[column][column]
68 matrix[column] = [value / pivot for value in matrix[column]]
69 for row in range(size):
70 factor = matrix[row][column]
71 if row != column and factor != 0:
72 matrix[row] = [
73 value - factor * base
74 for value, base in zip(matrix[row], matrix[column])
75 ]
76 return [row[size:] for row in matrix]

79class Chain:
80 def __init__(self, names):
81 self.names = names

94 states = []

117 def advance(self, distribution):
118 size = self.size()
119 return [
120 sum(distribution[i] * self.matrix[i][j] for i in range(size))
121 for j in range(size)

124 def reachable(self, start):
125 seen = [False] * self.size()
126 seen[start] = True
127 stack = [start]
128 while stack:
129 i = stack.pop()
130 for j in range(self.size()):
131 if self.edge(i, j) and not seen[j]:
132 seen[j] = True
133 stack.append(j)
134 return seen

137class CommunicatingClass:
138 def __init__(self, chain, number, members):
139 size = chain.size()
140 self.number = number
141 self.members = members
142 inside = membership(size, members)
143 self.closed = not any(
144 chain.edge(i, j) and not inside[j] for i in members for j in range(size)
145 )
146 self.levels = [-1] * size
147 self.levels[members[0]] = 0
148 queue = deque([members[0]])
149 while queue:
150 u = queue.popleft()
151 for v in members:
152 if chain.edge(u, v) and self.levels[v] < 0:
153 self.levels[v] = self.levels[u] + 1
154 queue.append(v)
155 self.period = 0
156 for u in members:
157 for v in members:
158 if chain.edge(u, v):
159 self.period = gcd(self.period, self.levels[u] + 1 - self.levels[v])

162class Decomposition:
163 def __init__(self, chain):
164 size = chain.size()
165 self.reach = [chain.reachable(i) for i in range(size)]
166 self.classes = []
167 self.class_of = [-1] * size
168 for i in range(size):
169 if self.class_of[i] < 0:
170 members = [j for j in range(size) if self.reach[i][j] and self.reach[j][i]]
171 for j in members:
172 self.class_of[j] = len(self.classes)
173 self.classes.append(CommunicatingClass(chain, len(self.classes) + 1, members))
174
175 def closed_classes(self):
176 return [item for item in self.classes if item.closed]
177
178 def class_containing(self, state):
179 return self.classes[self.class_of[state]]
180
181 def reaches(self, state, targets):
182 return any(reached and target for reached, target in zip(self.reach[state], targets))

185class Absorption:
186 def __init__(self, chain):
187 size = chain.size()
188 self.transient = [i for i in range(size) if not chain.absorbing(i)]
189 self.absorbing = [i for i in range(size) if chain.absorbing(i)]
190 count = len(self.transient)
191 matrix = []
192 for row, i in enumerate(self.transient):
193 left = [delta(i, j) - chain.matrix[i][j] for j in self.transient]
194 right = [delta(row, column) for column in range(count)]
195 matrix.append(left + right)
196 self.fundamental = eliminate(matrix, count)
197 self.steps = [sum(row) for row in self.fundamental]
198 self.probabilities = [
199 [
200 sum(row[k] * chain.matrix[self.transient[k]][a] for k in range(count))
201 for a in self.absorbing
202 ]
203 for row in self.fundamental
204 ]
205
206 def variance(self, i):
207 total = sum(
208 (2 * self.fundamental[i][j] - delta(i, j)) * self.steps[j]
209 for j in range(len(self.transient))
210 )
211 return total - self.steps[i] * self.steps[i]

214def stationary_distribution(chain, members):
215 size = len(members)
216 matrix = []
217 for r in range(size - 1):
218 row = [chain.matrix[members[k]][members[r]] - delta(k, r) for k in range(size)]
219 matrix.append(row + [ZERO])
220 matrix.append([ONE] * (size + 1))
221 return [row[0] for row in eliminate(matrix, size)]

224def irreducible_distribution(chain, decomposition):
225 if len(decomposition.classes) > 1:
226 raise CommandError("NOT IRREDUCIBLE")
227 return stationary_distribution(chain, decomposition.classes[0].members)

230def solve_restricted(chain, states, constants):
231 matrix = []
232 for row, i in enumerate(states):
233 coefficients = [delta(i, j) - chain.matrix[i][j] for j in states]
234 matrix.append(coefficients + [constants[row]])
235 return [values[0] for values in eliminate(matrix, len(states))]

238def hitting_probabilities(chain, decomposition, targets):
239 size = chain.size()
240 probabilities = [ONE if targets[i] else ZERO for i in range(size)]
241 remaining = [
242 i for i in range(size) if not targets[i] and decomposition.reaches(i, targets)
243 ]
244 if remaining:
245 constants = [
246 sum(chain.matrix[i][j] for j in range(size) if targets[j]) for i in remaining
247 ]
248 for i, value in zip(remaining, solve_restricted(chain, remaining, constants)):
249 probabilities[i] = value
250 return probabilities

253def hitting_times(chain, targets, probabilities):
254 size = chain.size()
255 times = [ZERO if targets[i] else None for i in range(size)]
256 certain = [i for i in range(size) if not targets[i] and probabilities[i] == 1]
257 if certain:
258 constants = [ONE] * len(certain)
259 for i, value in zip(certain, solve_restricted(chain, certain, constants)):
260 times[i] = value
261 return times

264def regular_exponent(chain):
265 size = chain.size()
266 edges = [[chain.edge(i, j) for j in range(size)] for i in range(size)]
267 power = edges
268 for exponent in range(1, (size - 1) * (size - 1) + 2):
269 if all(all(row) for row in power):
270 return exponent
271 power = [
272 [any(power[i][k] and edges[k][j] for k in range(size)) for j in range(size)]
273 for i in range(size)
274 ]
275 return None

377 def build(self, arguments):
378 name, order, tokens = arguments[0], int(arguments[1]), arguments[2:]
379 if order not in (1, 2):
380 raise CommandError("BAD ORDER")
381 if len(tokens) <= order:
382 raise CommandError("TOO FEW TOKENS")
383 if order == 1:
384 observations = tokens
385 else:
386 observations = [f"{first}.{second}" for first, second in zip(tokens, tokens[1:])]
387 names = []
388 for observation in observations:
389 if observation not in names:
390 names.append(observation)
391 if len(names) > MAX_STATES:
392 raise CommandError("TOO MANY STATES")
393 chain = Chain(names)
394 size = chain.size()
395 counts = [[0] * size for _ in range(size)]
396 for source, target in zip(observations, observations[1:]):
397 counts[chain.positions[source]][chain.positions[target]] += 1
398 self.emit(
399 "BUILD", name, f"order={order}", f"states={size}",
400 f"transitions={len(observations) - 1}",
401 )
402 for a in range(size):
403 total = sum(counts[a])
404 if total == 0:
405 chain.matrix[a][a] = ONE
406 self.emit("FROM", names[a], "total=0", "absorbing")
407 continue
408 parts = []
409 for b in range(size):
410 if counts[a][b] > 0:
411 chain.matrix[a][b] = Fraction(counts[a][b], total)
412 parts.append(f"{names[b]}:{counts[a][b]}")
413 self.emit("FROM", names[a], f"total={total}", *parts)
414 self.chains[name] = chain

416 def gambler(self, arguments):
417 name, goal, start = arguments[0], int(arguments[1]), int(arguments[2])
418 if goal == 0 or goal > MAX_GAMBLER_SIZE:
419 raise CommandError("BAD SIZE")
420 if start > goal:
421 raise CommandError("BAD START")
422 p = parse_probability(arguments[3])
423 if p == 0 or p == 1:
424 raise CommandError("BAD PROBABILITY")
425 chain = Chain([f"s{i}" for i in range(goal + 1)])
426 chain.matrix[0][0] = ONE
427 chain.matrix[goal][goal] = ONE
428 for i in range(1, goal):
429 chain.matrix[i][i + 1] = p
430 chain.matrix[i][i - 1] = 1 - p
431 self.chains[name] = chain
432 absorption = Absorption(chain)
433 ruin = [ONE] + [row[0] for row in absorption.probabilities] + [ZERO]
434 duration = [ZERO] + absorption.steps + [ZERO]
435 self.emit(
436 "GAMBLER", name, f"g={goal}", f"p={fraction_text(p)}", f"start=s{start}",
437 f"ruin={value_text(ruin[start])}", f"duration={value_text(duration[start])}",
438 )
439 for i in range(goal + 1):
440 self.emit(
441 "CAPITAL", f"s{i}", f"ruin={value_text(ruin[i])}",
442 f"win={value_text(1 - ruin[i])}", f"duration={value_text(duration[i])}",
443 )

575 def passage(self, arguments):
576 chain = self.require_stochastic(arguments[0])
577 size = chain.size()
578 decomposition = Decomposition(chain)
579 pi = irreducible_distribution(chain, decomposition)
580 times = [[ZERO] * size for _ in range(size)]
581 for j in range(size):
582 targets = membership(size, [j])
583 probabilities = hitting_probabilities(chain, decomposition, targets)
584 column = hitting_times(chain, targets, probabilities)
585 for i in range(size):
586 times[i][j] = 1 / pi[j] if i == j else column[i]
587 kemeny = sum(pi[j] * times[0][j] for j in range(1, size))
588 self.emit("PASSAGE", arguments[0], f"kemeny={value_text(kemeny)}")
589 self.display(chain.names, times, "M")

591 def first_passage(self, arguments):
592 name, source, target, text = arguments

610 f"total={value_text(sum(first))}",
611 )
612 cumulative = ZERO
613 for t, value in enumerate(first, 1):
614 cumulative += value

634 def most_likely(self, arguments):
635 chain = self.require_stochastic(arguments[0])
636 start = chain.state(arguments[1])
637 steps = int(arguments[2])
638 if steps == 0:
639 raise CommandError("BAD STEPS")
640 size = chain.size()
641 best = [[ONE] * size for _ in range(steps + 1)]
642 for t in range(steps - 1, -1, -1):
643 best[t] = [
644 max(chain.matrix[i][j] * best[t + 1][j] for j in range(size))
645 for i in range(size)
646 ]
647 path = [start]
648 for t in range(steps):
649 current = path[-1]
650 path.append(next(
651 j for j in range(size)
652 if chain.matrix[current][j] * best[t + 1][j] == best[t][current]
653 ))
654 self.emit(
655 "MOSTLIKELY", arguments[0], arguments[1], f"k={steps}",
656 f"prob={value_text(best[0][start])}",
657 )
658 self.emit("PATH", *chain.labels(path))

660 def reversible(self, arguments):
661 chain = self.require_stochastic(arguments[0])
662 pi = irreducible_distribution(chain, Decomposition(chain))
663 size = chain.size()
664 violations = []
665 for i in range(size):
666 for j in range(i + 1, size):
667 forward = pi[i] * chain.matrix[i][j]
668 backward = pi[j] * chain.matrix[j][i]
669 if forward != backward:
670 violations.append((i, j, forward, backward))
671 if not violations:
672 self.emit("REVERSIBLE", arguments[0], "yes")
673 return
674 self.emit("REVERSIBLE", arguments[0], "no", f"violations={len(violations)}")
675 for i, j, forward, backward in violations:
676 self.emit(
677 "VIOLATION", chain.names[i], chain.names[j],
678 value_text(forward), value_text(backward),
679 )

681 def reverse(self, arguments):
682 name, reversed_name = arguments
683 chain = self.require_stochastic(name)
684 pi = irreducible_distribution(chain, Decomposition(chain))
685 size = chain.size()
686 reversed_chain = Chain(list(chain.names))
687 for i in range(size):
688 for j in range(size):
689 reversed_chain.matrix[i][j] = pi[j] * chain.matrix[j][i] / pi[i]
690 self.chains[reversed_name] = reversed_chain
691 same = reversed_chain.matrix == chain.matrix
692 self.emit("REVERSE", name, reversed_name, f"reversible={answer(same)}")
693 self.display(reversed_chain.names, reversed_chain.matrix)

695 def mixing(self, arguments):
696 chain = self.require_stochastic(arguments[0])
697 steps = int(arguments[1])
698 if steps == 0:
699 raise CommandError("BAD STEPS")
700 epsilon = parse_probability(arguments[2])
701 pi = irreducible_distribution(chain, Decomposition(chain))
702 size = chain.size()
703 matrix = identity(size)
704 mixing_time = None
705 lines = []
706 for t in range(1, steps + 1):
707 matrix = [chain.advance(row) for row in matrix]
708 distances = [
709 sum(abs(row[j] - pi[j]) for j in range(size)) / 2 for row in matrix
710 ]
711 largest = max(distances)
712 worst = distances.index(largest)
713 if mixing_time is None and largest <= epsilon:
714 mixing_time = t
715 lines.append((t, largest, chain.names[worst]))
716 self.emit(
717 "MIXING", arguments[0], f"k={steps}", f"eps={fraction_text(epsilon)}",
718 f"time={'none' if mixing_time is None else mixing_time}",
719 )
720 for t, largest, worst in lines:
721 self.emit("TV", t, value_text(largest), f"from={worst}")

723 def lump(self, arguments):
724 name, lumped_name, partition = arguments
725 chain = self.require_stochastic(name)
726 size = chain.size()
727 seen = [False] * size
728 blocks = []
729 for group in partition.split("|"):
730 block = []
731 for text in group.split(","):
732 state = chain.state(text)
733 if seen[state]:
734 raise CommandError("BAD PARTITION")
735 seen[state] = True
736 block.append(state)
737 blocks.append(sorted(block))
738 if not all(seen):
739 raise CommandError("BAD PARTITION")
740 blocks.sort(key=lambda block: block[0])
741 for block in blocks:
742 representative = block[0]
743 for target in blocks:
744 expected = chain.block_sum(representative, target)
745 for i in block:
746 actual = chain.block_sum(i, target)
747 if actual != expected:
748 self.emit(
749 "LUMP", name, "not", "lumpable",
750 f"block={chain.names[representative]}",
751 f"target={chain.names[target[0]]}",
752 f"{chain.names[representative]}={value_text(expected)}",
753 f"{chain.names[i]}={value_text(actual)}",
754 )
755 return
756 lumped = Chain([chain.names[block[0]] for block in blocks])
757 lumped.matrix = [
758 [chain.block_sum(block[0], target) for target in blocks] for block in blocks
759 ]
760 self.chains[lumped_name] = lumped
761 self.emit("LUMP", name, lumped_name, f"blocks={len(blocks)}")
762 self.display(lumped.names, lumped.matrix)

764 def reward(self, arguments):
765 chain = self.require_stochastic(arguments[0])
766 size = chain.size()
767 texts = arguments[2:]
768 if len(texts) != size:
769 raise CommandError("BAD LENGTH")
770 gamma = parse_probability(arguments[1])
771 if gamma == 1:
772 raise CommandError("BAD DISCOUNT")
773 rewards = [Fraction(int(text)) for text in texts]
774 matrix = []
775 for i in range(size):
776 coefficients = [delta(i, j) - gamma * chain.matrix[i][j] for j in range(size)]
777 matrix.append(coefficients + [rewards[i]])
778 values = [row[0] for row in eliminate(matrix, size)]
779 decomposition = Decomposition(chain)
780 gains = [ZERO] * size
781 for item in decomposition.closed_classes():
782 distribution = stationary_distribution(chain, item.members)
783 gain = sum(rho * rewards[i] for i, rho in zip(item.members, distribution))
784 targets = membership(size, item.members)
785 probabilities = hitting_probabilities(chain, decomposition, targets)
786 for i in range(size):
787 gains[i] += probabilities[i] * gain
788 self.emit("REWARD", arguments[0], f"gamma={fraction_text(gamma)}")
789 for i in range(size):
790 self.emit(
791 "VALUE", chain.names[i], f"discounted={value_text(values[i])}",
792 f"gain={value_text(gains[i])}",
793 )

795 def restrict(self, arguments):
796 name, restricted_name = arguments[0], arguments[1]
797 chain = self.require_stochastic(name)
798 size = chain.size()
799 inside = membership(size, chain.distinct_states(arguments[2:]))
800 for i in range(size):
801 if inside[i] and any(chain.edge(i, j) and not inside[j] for j in range(size)):
802 raise CommandError(f"NOT CLOSED {chain.names[i]}")
803 members = [i for i in range(size) if inside[i]]
804 restricted = Chain(chain.labels(members))
805 restricted.matrix = [[chain.matrix[i][j] for j in members] for i in members]
806 self.chains[restricted_name] = restricted
807 self.emit("RESTRICT", name, restricted_name, f"states={len(members)}")
808 self.display(restricted.names, restricted.matrix)

810 def simulate(self, arguments):
811 name, source = arguments[0], arguments[1]
812 chain = self.require_stochastic(name)
813 start = chain.state(source)
814 steps, seed = int(arguments[2]), int(arguments[3])
815 if steps == 0:
816 raise CommandError("BAD STEPS")
817 if seed == 0 or seed >= MODULUS:
818 raise CommandError("BAD SEED")
819 random_value = seed
820 path = [start]
821 for _ in range(steps):
822 row = chain.matrix[path[-1]]
823 common = lcm(*(value.denominator for value in row))
824 random_value = MULTIPLIER * random_value % MODULUS
825 threshold = random_value % common
826 cumulative = ZERO
827 for j, value in enumerate(row):
828 cumulative += value
829 if cumulative * common > threshold:
830 path.append(j)
831 break
832 self.emit(
833 "SIMULATE", name, source, f"steps={steps}", f"seed={seed}",
834 f"end={chain.names[path[-1]]}",
835 )
836 self.emit("PATH", *chain.labels(path))
837 for i in range(chain.size()):
838 self.emit("VISIT", chain.names[i], path.count(i))

841 command, *arguments = line.split(" ")
842 try:
843 self.handlers[command](arguments)
844 except CommandError as error:
845 self.emit("ERROR", error)
