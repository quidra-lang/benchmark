18struct Fraction {
19 Integer numerator;
20 Integer denominator;
21
22 Fraction(Integer n = 0, Integer d = 1) {
23 if (d < 0) { n = -n; d = -d; }
24 const Integer divisor = std::gcd(n, d);
25 numerator = n / divisor;
26 denominator = d / divisor;
27 }
28
29 friend Fraction operator+(Fraction a, Fraction b) {
30 return {a.numerator * b.denominator + b.numerator * a.denominator,
31 a.denominator * b.denominator};
32 }
33 friend Fraction operator-(Fraction a, Fraction b) {
34 return {a.numerator * b.denominator - b.numerator * a.denominator,
35 a.denominator * b.denominator};
36 }
37 friend Fraction operator*(Fraction a, Fraction b) {
38 return {a.numerator * b.numerator, a.denominator * b.denominator};
39 }
40 friend Fraction operator/(Fraction a, Fraction b) {
41 return {a.numerator * b.denominator, a.denominator * b.numerator};
42 }
43 friend bool operator==(Fraction a, Fraction b) {
44 return a.numerator == b.numerator && a.denominator == b.denominator;
45 }
46 friend bool operator!=(Fraction a, Fraction b) { return !(a == b); }
47 friend bool operator<(Fraction a, Fraction b) {
48 return a.numerator * b.denominator < b.numerator * a.denominator;
49 }
50 friend bool operator<=(Fraction a, Fraction b) { return !(b < a); }
51 friend std::ostream& operator<<(std::ostream& out, Fraction value) {
52 return out << value.numerator << '/' << value.denominator;
53 }
54
55 Fraction absolute() const {
56 return {numerator < 0 ? -numerator : numerator, denominator};
57 }
58
59 std::string value() const {
60 std::ostringstream out;
61 out << *this << '(' << numerator / denominator << '.'
62 << std::setfill('0') << std::setw(6)
63 << (numerator % denominator) * 1000000 / denominator << ')';
64 return out.str();
65 }
66};

99Fraction probability(const std::string& text) {
100 const Names parts = split(text, '/');
101 const Integer numerator = std::stoll(parts[0]);
102 const Integer denominator = parts.size() == 2 ? std::stoll(parts[1]) : 1;
103 if (denominator == 0 || numerator > denominator) fail("BAD PROBABILITY");
104 return {numerator, denominator};
105}

115Matrix identity(int size) {
116 Matrix result = matrix(size, size);
117 for (int i = 0; i < size; ++i) result[i][i] = 1;
118 return result;
119}

121Matrix multiply(const Matrix& left, const Matrix& right) {
122 Matrix result = matrix(static_cast<int>(left.size()), static_cast<int>(right[0].size()));
123 for (std::size_t i = 0; i < left.size(); ++i)
124 for (std::size_t j = 0; j < right[0].size(); ++j)
125 for (std::size_t k = 0; k < right.size(); ++k)
126 result[i][j] = result[i][j] + left[i][k] * right[k][j];
127 return result;
128}

130Matrix eliminate(Matrix rows) {
131 const int size = static_cast<int>(rows.size());
132 const int width = static_cast<int>(rows[0].size());
133 for (int column = 0; column < size; ++column) {
134 int pivotRow = column;
135 while (rows[pivotRow][column].numerator == 0) ++pivotRow;
136 std::swap(rows[column], rows[pivotRow]);
137 const Fraction pivot = rows[column][column];
138 for (int j = 0; j < width; ++j) rows[column][j] = rows[column][j] / pivot;
139 for (int i = 0; i < size; ++i) {
140 const Fraction factor = rows[i][column];
141 if (i == column || factor.numerator == 0) continue;
142 for (int j = 0; j < width; ++j)
143 rows[i][j] = rows[i][j] - factor * rows[column][j];
144 }
145 }
146 Matrix result;
147 for (const Row& row : rows) result.emplace_back(row.begin() + size, row.end());
148 return result;
149}

151struct Chain {
152 Names names;
153 Matrix p;
154
155 explicit Chain(Names stateNames = {})
156 : names(std::move(stateNames)), p(matrix(size(), size())) {}
157
158 int size() const { return static_cast<int>(names.size()); }
159
160 int state(const std::string& name) const {
161 for (int i = 0; i < size(); ++i) if (names[i] == name) return i;
162 fail("UNKNOWN STATE " + name);
163 }
164
165 States states(const Names& requested) const {
166 States result;
167 std::set<int> seen;
168 for (const std::string& name : requested) {
169 const int index = state(name);
170 if (!seen.insert(index).second) fail("DUPLICATE STATE " + name);
171 result.push_back(index);
172 }
173 return result;
174 }
175
176 Names labels(const States& indices) const {
177 Names result;
178 for (int index : indices) result.push_back(names[index]);
179 return result;
180 }
181
182 Fraction blockSum(int state, const States& block) const {
183 Fraction result;
184 for (int target : block) result = result + p[state][target];
185 return result;
186 }

188 void display(const Matrix& values, const std::string& prefix = "ROW") const {
189 std::cout << "STATES " << join(names) << '\n';
190 for (int i = 0; i < size(); ++i)
191 std::cout << prefix << ' ' << names[i] << ' ' << join(values[i]) << '\n';
192 }
193 void display() const { display(p); }

196struct CommunicatingClass {
197 States members;
198 bool closed = true;
199 std::vector<int> levels;
200 int period = 0;
201};

203struct Decomposition {
204 BooleanMatrix reach;
205 std::vector<CommunicatingClass> classes;
206 std::vector<int> classOf;
207
208 explicit Decomposition(const Chain& chain)
209 : reach(chain.size(), std::vector<bool>(chain.size())), classOf(chain.size(), -1) {
210 const int n = chain.size();
211 for (int i = 0; i < n; ++i)
212 for (int j = 0; j < n; ++j)
213 reach[i][j] = i == j || chain.p[i][j].numerator > 0;
214 for (int k = 0; k < n; ++k)
215 for (int i = 0; i < n; ++i)
216 for (int j = 0; j < n; ++j)
217 reach[i][j] = reach[i][j] || (reach[i][k] && reach[k][j]);
218 for (int root = 0; root < n; ++root) {
219 if (classOf[root] != -1) continue;
220 CommunicatingClass group;
221 group.levels.assign(n, -1);
222 std::vector<bool> inside(n);
223 for (int j = 0; j < n; ++j) {
224 if (!reach[root][j] || !reach[j][root]) continue;
225 group.members.push_back(j);
226 inside[j] = true;
227 classOf[j] = static_cast<int>(classes.size());
228 }
229 for (int u : group.members)
230 for (int v = 0; v < n; ++v)
231 if (chain.p[u][v].numerator > 0 && !inside[v]) group.closed = false;
232 States queue{root};
233 group.levels[root] = 0;
234 for (std::size_t head = 0; head < queue.size(); ++head) {
235 const int u = queue[head];
236 for (int v : group.members) {
237 if (chain.p[u][v].numerator == 0 || group.levels[v] != -1) continue;
238 group.levels[v] = group.levels[u] + 1;
239 queue.push_back(v);
240 }
241 }
242 for (int u : group.members)
243 for (int v : group.members)
244 if (chain.p[u][v].numerator > 0)
245 group.period = std::gcd(group.period, group.levels[u] + 1 - group.levels[v]);
246 classes.push_back(std::move(group));
247 }
248 }
249};

251Row stationary(const Chain& chain, const States& members) {
252 const int size = static_cast<int>(members.size());
253 Matrix rows = matrix(size, size + 1);
254 for (int r = 0; r < size - 1; ++r)
255 for (int k = 0; k < size; ++k)
256 rows[r][k] = chain.p[members[k]][members[r]] - Fraction(k == r);
257 std::fill(rows.back().begin(), rows.back().end(), Fraction(1));
258 const Matrix solution = eliminate(std::move(rows));
259 Row result;
260 for (const Row& row : solution) result.push_back(row[0]);
261 return result;
262}
263
264Row irreducible(const Chain& chain, const Decomposition& graph) {
265 if (graph.classes.size() != 1) fail("NOT IRREDUCIBLE");
266 return stationary(chain, graph.classes[0].members);
267}

269struct Hitting {
270 Row probabilities;
271 std::vector<std::optional<Fraction>> times;
272};
273
274Hitting hitting(const Chain& chain, const States& targets, const BooleanMatrix& reach,
275 bool probabilityOnly = false) {
276 const int n = chain.size();
277 Hitting result{Row(n), std::vector<std::optional<Fraction>>(n)};
278 std::vector<bool> inside(n);
279 for (int target : targets) {
280 inside[target] = true;
281 result.probabilities[target] = 1;
282 result.times[target] = Fraction(0);
283 }
284 States unknown;
285 for (int i = 0; i < n; ++i) {
286 if (inside[i]) continue;
287 for (int target : targets) {
288 if (!reach[i][target]) continue;
289 unknown.push_back(i);
290 break;
291 }
292 }
293 if (!unknown.empty()) {
294 const int size = static_cast<int>(unknown.size());
295 Matrix rows = matrix(size, size + 1);
296 for (int r = 0; r < size; ++r) {
297 for (int k = 0; k < size; ++k)
298 rows[r][k] = Fraction(r == k) - chain.p[unknown[r]][unknown[k]];
299 rows[r][size] = chain.blockSum(unknown[r], targets);
300 }
301 const Matrix solution = eliminate(std::move(rows));
302 for (int r = 0; r < size; ++r) result.probabilities[unknown[r]] = solution[r][0];
303 }
304 if (probabilityOnly) return result;
305 States certain;
306 for (int i : unknown) if (result.probabilities[i] == Fraction(1)) certain.push_back(i);
307 if (!certain.empty()) {
308 const int size = static_cast<int>(certain.size());
309 Matrix rows = matrix(size, size + 1);
310 for (int r = 0; r < size; ++r) {
311 for (int k = 0; k < size; ++k)
312 rows[r][k] = Fraction(r == k) - chain.p[certain[r]][certain[k]];
313 rows[r][size] = 1;
314 }
315 const Matrix solution = eliminate(std::move(rows));
316 for (int r = 0; r < size; ++r) result.times[certain[r]] = solution[r][0];
317 }
318 return result;
319}

321struct Absorption {
322 States transient;
323 States absorbing;
324 Matrix fundamental;
325 Row means;
326 Row variances;
327 Matrix probabilities;
328
329 explicit Absorption(const Chain& chain) {
330 for (int i = 0; i < chain.size(); ++i)
331 (chain.p[i][i] == Fraction(1) ? absorbing : transient).push_back(i);
332 const int t = static_cast<int>(transient.size());
333 const int a = static_cast<int>(absorbing.size());
334 if (t == 0) return;
335 Matrix rows = matrix(t, 2 * t);
336 for (int i = 0; i < t; ++i) {
337 for (int j = 0; j < t; ++j)
338 rows[i][j] = Fraction(i == j) - chain.p[transient[i]][transient[j]];
339 rows[i][t + i] = 1;
340 }
341 fundamental = eliminate(std::move(rows));
342 for (const Row& row : fundamental) means.push_back(sum(row));
343 variances.resize(t);
344 probabilities = matrix(t, a);
345 for (int i = 0; i < t; ++i) {
346 Fraction secondMoment;
347 for (int j = 0; j < t; ++j)
348 secondMoment = secondMoment + (Fraction(2) * fundamental[i][j] - Fraction(i == j)) * means[j];
349 variances[i] = secondMoment - means[i] * means[i];
350 for (int j = 0; j < a; ++j)
351 for (int l = 0; l < t; ++l)
352 probabilities[i][j] = probabilities[i][j] + fundamental[i][l] * chain.p[transient[l]][absorbing[j]];
353 }
354 }
355};

371 void build(const Names& words) {
372 const std::string& name = words[1];
373 const int order = std::stoi(words[2]);
374 if (order != 1 && order != 2) fail("BAD ORDER");
375 const int tokenCount = static_cast<int>(words.size()) - 3;
376 if (tokenCount <= order) fail("TOO FEW TOKENS");
377 Names names;
378 States observations;
379 for (int i = 0; i < tokenCount - order + 1; ++i) {
380 const std::string observation = order == 1 ? words[i + 3] : words[i + 3] + '.' + words[i + 4];
381 const Names::const_iterator position = std::find(names.cbegin(), names.cend(), observation);
382 const int index = static_cast<int>(position - names.cbegin());
383 if (position == names.cend()) names.push_back(observation);
384 observations.push_back(index);
385 }
386 if (names.size() > 10) fail("TOO MANY STATES");
387 Chain chain(names);
388 const int n = chain.size();
389 std::vector<std::vector<int>> counts(n, std::vector<int>(n));
390 std::vector<int> totals(n);
391 for (std::size_t i = 1; i < observations.size(); ++i) {
392 ++counts[observations[i - 1]][observations[i]];
393 ++totals[observations[i - 1]];
394 }
395 for (int i = 0; i < n; ++i) {
396 if (totals[i] == 0) chain.p[i][i] = 1;
397 else for (int j = 0; j < n; ++j) chain.p[i][j] = Fraction(counts[i][j], totals[i]);
398 }
399 chains[name] = chain;
400 std::cout << "BUILD " << name << " order=" << order << " states=" << n
401 << " transitions=" << observations.size() - 1 << '\n';
402 for (int i = 0; i < n; ++i) {
403 std::cout << "FROM " << names[i] << " total=" << totals[i];
404 if (totals[i] == 0) std::cout << " absorbing";
405 else for (int j = 0; j < n; ++j)
406 if (counts[i][j] > 0) std::cout << ' ' << names[j] << ':' << counts[i][j];
407 std::cout << '\n';
408 }

411 void gambler(const Names& words) {
412 const std::string& name = words[1];
413 const int goal = std::stoi(words[2]);
414 const int start = std::stoi(words[3]);
415 if (goal == 0 || goal > 9) fail("BAD SIZE");
416 if (start > goal) fail("BAD START");
417 const Fraction chance = probability(words[4]);
418 if (chance == Fraction(0) || chance == Fraction(1)) fail("BAD PROBABILITY");
419 Names names;
420 for (int i = 0; i <= goal; ++i) names.push_back("s" + std::to_string(i));
421 Chain chain(names);
422 chain.p[0][0] = chain.p[goal][goal] = 1;
423 for (int i = 1; i < goal; ++i) {
424 chain.p[i][i + 1] = chance;
425 chain.p[i][i - 1] = Fraction(1) - chance;
426 }
427 const Absorption data(chain);
428 Row ruin(goal + 1), duration(goal + 1);
429 ruin[0] = 1;
430 for (int i = 1; i < goal; ++i) {
431 ruin[i] = data.probabilities[i - 1][0];
432 duration[i] = data.means[i - 1];
433 }
434 chains[name] = chain;
435 std::cout << "GAMBLER " << name << " g=" << goal << " p=" << chance
436 << " start=" << names[start] << " ruin=" << ruin[start].value()
437 << " duration=" << duration[start].value() << '\n';
438 for (int i = 0; i <= goal; ++i)
439 std::cout << "CAPITAL " << names[i] << " ruin=" << ruin[i].value()
440 << " win=" << (Fraction(1) - ruin[i]).value()
441 << " duration=" << duration[i].value() << '\n';
442 }

481 void analyze(const Names& words, const Chain& chain) {
482 const std::string& command = words[0];
483 const std::string& name = words[1];
484 const int n = chain.size();
485 for (int i = 0; i < n; ++i)
486 if (sum(chain.p[i]) != Fraction(1)) fail("NOT STOCHASTIC " + chain.names[i]);
487 const Decomposition graph(chain);
488 if (command == "CLASSES") {
489 std::cout << "CLASSES " << name << " count=" << graph.classes.size() << '\n';
490 for (std::size_t index = 0; index < graph.classes.size(); ++index) {
491 const CommunicatingClass& group = graph.classes[index];
492 std::cout << "CLASS " << index + 1 << ' ' << (group.closed ? "RECURRENT" : "TRANSIENT")
493 << " period=" << group.period << " size=" << group.members.size()
494 << ' ' << join(chain.labels(group.members)) << '\n';
495 if (group.period >= 2) {
496 for (int residue = 0; residue < group.period; ++residue) {
497 States cyclic;
498 for (int i : group.members)
499 if (group.levels[i] % group.period == residue) cyclic.push_back(i);
500 std::cout << "CYCLIC " << residue << ' ' << join(chain.labels(cyclic)) << '\n';
501 }
502 }
503 }
504 } else if (command == "CLASSIFY") {
505 States absorbing;
506 for (int i = 0; i < n; ++i) if (chain.p[i][i] == Fraction(1)) absorbing.push_back(i);
507 bool absorbs = !absorbing.empty();
508 for (int i = 0; i < n; ++i) {
509 bool reaches = false;
510 for (int target : absorbing) reaches = reaches || graph.reach[i][target];
511 absorbs = absorbs && reaches;
512 }
513 BooleanMatrix edges(n, std::vector<bool>(n));
514 for (int i = 0; i < n; ++i)
515 for (int j = 0; j < n; ++j) edges[i][j] = chain.p[i][j].numerator > 0;
516 BooleanMatrix power = edges;
517 int exponent = 0;
518 const int bound = (n - 1) * (n - 1) + 1;
519 for (int t = 1; t <= bound; ++t) {
520 bool positive = true;
521 for (const std::vector<bool>& row : power)
522 for (bool entry : row) positive = positive && entry;
523 if (positive) { exponent = t; break; }
524 if (t == bound) break;
525 BooleanMatrix next(n, std::vector<bool>(n));
526 for (int i = 0; i < n; ++i)
527 for (int j = 0; j < n; ++j)
528 for (int l = 0; l < n; ++l)
529 next[i][j] = next[i][j] || (power[i][l] && edges[l][j]);
530 power = std::move(next);
531 }
532 std::cout << "CLASSIFY " << name << " absorbing_states=" << absorbing.size()
533 << " absorbing=" << yes(absorbs) << " ergodic=" << yes(graph.classes.size() == 1)
534 << " regular=" << yes(exponent != 0)
535 << " exponent=" << (exponent == 0 ? "-" : std::to_string(exponent)) << '\n';
536 } else if (command == "STEP") {
537 const std::string& initial = words[2];
538 Row distribution(n);
539 if (std::isdigit(static_cast<unsigned char>(initial[0]))) {
540 const Names parts = split(initial, ',');
541 if (static_cast<int>(parts.size()) != n) fail("BAD LENGTH");
542 for (int i = 0; i < n; ++i) distribution[i] = probability(parts[i]);
543 if (sum(distribution) != Fraction(1)) fail("BAD DISTRIBUTION");
544 } else distribution[chain.state(initial)] = 1;
545 const int steps = std::stoi(words[3]);
546 for (int t = 0; t < steps; ++t) distribution = multiply(Matrix{distribution}, chain.p)[0];
547 std::cout << "STEP " << name << " k=" << steps << '\n';
548 for (int i = 0; i < n; ++i)
549 std::cout << "PROB " << chain.names[i] << ' ' << distribution[i].value() << '\n';
550 } else if (command == "POWER") {
551 const int steps = std::stoi(words[2]);
552 Matrix power = identity(n);
553 for (int t = 0; t < steps; ++t) power = multiply(power, chain.p);
554 std::cout << "POWER " << name << " k=" << steps << '\n';
555 chain.display(power);
556 } else if (command == "STATIONARY") {
557 int count = 0;
558 for (const CommunicatingClass& group : graph.classes) if (group.closed) ++count;
559 std::cout << "STATIONARY " << name << " count=" << count << '\n';
560 int number = 0;
561 for (std::size_t index = 0; index < graph.classes.size(); ++index) {
562 const CommunicatingClass& group = graph.classes[index];
563 if (!group.closed) continue;
564 const Row pi = stationary(chain, group.members);
565 std::cout << "DIST " << ++number << " class=" << index + 1
566 << " size=" << group.members.size() << '\n';
567 for (std::size_t i = 0; i < group.members.size(); ++i)
568 std::cout << "PI " << chain.names[group.members[i]] << ' ' << pi[i].value() << '\n';
569 }
570 } else if (command == "RECURRENCE") {
571 const int state = chain.state(words[2]);
572 const CommunicatingClass& group = graph.classes[graph.classOf[state]];
573 if (!group.closed) fail("TRANSIENT STATE " + words[2]);
574 const Row pi = stationary(chain, group.members);
575 const std::size_t position = std::find(group.members.begin(), group.members.end(), state) - group.members.begin();
576 const Fraction rho = pi[position];
577 std::cout << "RECURRENCE " << name << ' ' << words[2] << " pi=" << rho.value()
578 << " mean=" << (Fraction(1) / rho).value() << " period=" << group.period << '\n';
579 } else if (command == "ABSORB") {
580 States absorbing;
581 for (int i = 0; i < n; ++i) if (chain.p[i][i] == Fraction(1)) absorbing.push_back(i);
582 if (absorbing.empty()) fail("NO ABSORBING STATE");
583 for (int i = 0; i < n; ++i) {
584 bool reaches = false;
585 for (int target : absorbing) reaches = reaches || graph.reach[i][target];
586 if (!reaches) fail("NOT ABSORBING " + chain.names[i]);
587 }
588 const Absorption data(chain);
589 const int count = static_cast<int>(data.transient.size());
590 std::cout << "ABSORB " << name << " transient=" << count << " absorbing=" << absorbing.size() << '\n';
591 if (count == 0) return;
592 std::cout << "TRANSIENT " << join(chain.labels(data.transient)) << '\n';
593 std::cout << "ABSORBING " << join(chain.labels(absorbing)) << '\n';
594 for (int i = 0; i < count; ++i)
595 std::cout << "N " << chain.names[data.transient[i]] << ' ' << join(data.fundamental[i]) << '\n';
596 for (int i = 0; i < count; ++i)
597 std::cout << "STEPS " << chain.names[data.transient[i]] << " mean=" << data.means[i].value()
598 << " var=" << data.variances[i].value() << '\n';
599 for (int i = 0; i < count; ++i)
600 std::cout << "B " << chain.names[data.transient[i]] << ' ' << join(data.probabilities[i]) << '\n';
601 } else if (command == "HITTING") {
602 States targets = chain.states(Names(words.begin() + 2, words.end()));
603 std::sort(targets.begin(), targets.end());
604 const Hitting result = hitting(chain, targets, graph.reach);
605 std::cout << "HITTING " << name << " targets=" << join(chain.labels(targets), ",") << '\n';
606 for (int i = 0; i < n; ++i)
607 std::cout << "HIT " << chain.names[i] << " prob=" << result.probabilities[i].value()
608 << " time=" << (result.times[i] ? result.times[i]->value() : "inf") << '\n';
609 } else if (command == "PASSAGE") {
610 const Row pi = irreducible(chain, graph);
611 Matrix passage = matrix(n, n);
612 for (int j = 0; j < n; ++j) {
613 const Hitting result = hitting(chain, States{j}, graph.reach);
614 for (int i = 0; i < n; ++i)
615 passage[i][j] = i == j ? Fraction(1) / pi[j] : *result.times[i];
616 }
617 Fraction kemeny;
618 for (int j = 1; j < n; ++j) kemeny = kemeny + pi[j] * passage[0][j];
619 std::cout << "PASSAGE " << name << " kemeny=" << kemeny.value() << '\n';
620 chain.display(passage, "M");
621 } else if (command == "FIRSTPASSAGE") {
622 const int from = chain.state(words[2]);
623 const int target = chain.state(words[3]);
624 const int steps = std::stoi(words[4]);
625 if (steps == 0) fail("BAD STEPS");
626 Row probabilities(n), first, cumulative;
627 for (int i = 0; i < n; ++i) probabilities[i] = chain.p[i][target];
628 Fraction total;
629 for (int t = 1; t <= steps; ++t) {
630 first.push_back(probabilities[from]);
631 total = total + probabilities[from];
632 cumulative.push_back(total);
633 if (t == steps) break;
634 Row next(n);
635 for (int i = 0; i < n; ++i)
636 for (int u = 0; u < n; ++u)
637 if (u != target) next[i] = next[i] + chain.p[i][u] * probabilities[u];
638 probabilities = std::move(next);
639 }
640 std::cout << "FIRSTPASSAGE " << name << ' ' << words[2] << ' ' << words[3]
641 << " k=" << steps << " total=" << total.value() << '\n';
642 for (int t = 0; t < steps; ++t)
643 std::cout << "FP " << t + 1 << ' ' << first[t].value() << " cum=" << cumulative[t].value() << '\n';
644 } else if (command == "PATHPROB") {
645 States path;
646 for (std::size_t i = 2; i < words.size(); ++i) path.push_back(chain.state(words[i]));
647 Fraction chance(1);
648 std::size_t zeroAt = 0;
649 for (std::size_t i = 1; i < path.size(); ++i) {
650 chance = chance * chain.p[path[i - 1]][path[i]];
651 if (chance.numerator == 0 && zeroAt == 0) zeroAt = i;
652 }
653 std::cout << "PATHPROB " << name << " steps=" << path.size() - 1 << " prob=" << chance.value();
654 if (zeroAt != 0) std::cout << " zero_at=" << zeroAt;
655 std::cout << '\n';
656 } else if (command == "MOSTLIKELY") {
657 const int start = chain.state(words[2]);
658 const int steps = std::stoi(words[3]);
659 if (steps == 0) fail("BAD STEPS");
660 Matrix best = matrix(steps + 1, n);
661 std::fill(best.back().begin(), best.back().end(), Fraction(1));
662 for (int t = steps - 1; t >= 0; --t)
663 for (int i = 0; i < n; ++i)
664 for (int j = 0; j < n; ++j) {
665 const Fraction candidate = chain.p[i][j] * best[t + 1][j];
666 if (best[t][i] < candidate) best[t][i] = candidate;
667 }
668 States path{start};
669 for (int t = 0; t < steps; ++t) {
670 const int current = path.back();
671 for (int j = 0; j < n; ++j) {
672 if (chain.p[current][j] * best[t + 1][j] != best[t][current]) continue;
673 path.push_back(j);
674 break;
675 }
676 }
677 std::cout << "MOSTLIKELY " << name << ' ' << words[2] << " k=" << steps
678 << " prob=" << best[0][start].value() << '\n';
679 std::cout << "PATH " << join(chain.labels(path)) << '\n';
680 } else if (command == "REVERSIBLE") {
681 const Row pi = irreducible(chain, graph);
682 struct Violation { int from; int to; Fraction forward; Fraction backward; };
683 std::vector<Violation> violations;
684 for (int i = 0; i < n; ++i)
685 for (int j = i + 1; j < n; ++j) {
686 const Fraction forward = pi[i] * chain.p[i][j];
687 const Fraction backward = pi[j] * chain.p[j][i];
688 if (forward != backward) violations.push_back({i, j, forward, backward});
689 }
690 if (violations.empty()) std::cout << "REVERSIBLE " << name << " yes\n";
691 else {
692 std::cout << "REVERSIBLE " << name << " no violations=" << violations.size() << '\n';
693 for (const Violation& violation : violations)
694 std::cout << "VIOLATION " << chain.names[violation.from] << ' ' << chain.names[violation.to]
695 << ' ' << violation.forward.value() << ' ' << violation.backward.value() << '\n';
696 }
697 } else if (command == "REVERSE") {
698 const Row pi = irreducible(chain, graph);
699 Chain reversed(chain.names);
700 bool reversible = true;
701 for (int i = 0; i < n; ++i)
702 for (int j = 0; j < n; ++j) {
703 reversed.p[i][j] = pi[j] * chain.p[j][i] / pi[i];
704 if (reversed.p[i][j] != chain.p[i][j]) reversible = false;
705 }
706 chains[words[2]] = reversed;
707 std::cout << "REVERSE " << name << ' ' << words[2] << " reversible=" << yes(reversible) << '\n';
708 reversed.display();
709 } else if (command == "MIXING") {
710 const int steps = std::stoi(words[2]);
711 if (steps == 0) fail("BAD STEPS");
712 const Fraction epsilon = probability(words[3]);
713 const Row pi = irreducible(chain, graph);
714 Matrix power = identity(n);
715 struct Distance { Fraction value; int state; };
716 std::vector<Distance> distances;
717 int mixingTime = 0;
718 for (int t = 1; t <= steps; ++t) {
719 power = multiply(power, chain.p);
720 Distance maximum{Fraction(0), 0};
721 for (int i = 0; i < n; ++i) {
722 Fraction distance;
723 for (int j = 0; j < n; ++j) distance = distance + (power[i][j] - pi[j]).absolute();
724 distance = distance / Fraction(2);
725 if (maximum.value < distance) maximum = {distance, i};
726 }
727 distances.push_back(maximum);
728 if (mixingTime == 0 && maximum.value <= epsilon) mixingTime = t;
729 }
730 std::cout << "MIXING " << name << " k=" << steps << " eps=" << epsilon
731 << " time=" << (mixingTime == 0 ? "none" : std::to_string(mixingTime)) << '\n';
732 for (int t = 0; t < steps; ++t)
733 std::cout << "TV " << t + 1 << ' ' << distances[t].value.value()
734 << " from=" << chain.names[distances[t].state] << '\n';
735 } else if (command == "LUMP") {
736 std::vector<States> blocks;
737 std::set<int> seen;
738 for (const std::string& block : split(words[3], '|')) {
739 States members;
740 for (const std::string& state : split(block, ',')) {
741 const int index = chain.state(state);
742 if (!seen.insert(index).second) fail("BAD PARTITION");
743 members.push_back(index);
744 }
745 std::sort(members.begin(), members.end());
746 blocks.push_back(std::move(members));
747 }
748 if (static_cast<int>(seen.size()) != n) fail("BAD PARTITION");
749 std::sort(blocks.begin(), blocks.end(), [](const States& a, const States& b) { return a[0] < b[0]; });
750 States representatives;
751 for (const States& block : blocks) representatives.push_back(block[0]);
752 Chain lumped(chain.labels(representatives));
753 for (std::size_t k = 0; k < blocks.size(); ++k)
754 for (std::size_t l = 0; l < blocks.size(); ++l) {
755 const int representative = blocks[k][0];
756 const Fraction expected = chain.blockSum(representative, blocks[l]);
757 for (int state : blocks[k]) {
758 const Fraction actual = chain.blockSum(state, blocks[l]);
759 if (actual == expected) continue;
760 std::cout << "LUMP " << name << " not lumpable block=" << chain.names[representative]
761 << " target=" << chain.names[blocks[l][0]]
762 << ' ' << chain.names[representative] << '=' << expected.value()
763 << ' ' << chain.names[state] << '=' << actual.value() << '\n';
764 return;
765 }
766 lumped.p[k][l] = expected;
767 }
768 chains[words[2]] = lumped;
769 std::cout << "LUMP " << name << ' ' << words[2] << " blocks=" << blocks.size() << '\n';
770 lumped.display();
771 } else if (command == "REWARD") {
772 if (static_cast<int>(words.size()) - 3 != n) fail("BAD LENGTH");
773 const Fraction gamma = probability(words[2]);
774 if (gamma == Fraction(1)) fail("BAD DISCOUNT");
775 Row rewards;
776 for (std::size_t i = 3; i < words.size(); ++i) rewards.emplace_back(std::stoll(words[i]));
777 Matrix rows = matrix(n, n + 1);
778 for (int i = 0; i < n; ++i) {
779 for (int j = 0; j < n; ++j) rows[i][j] = Fraction(i == j) - gamma * chain.p[i][j];
780 rows[i][n] = rewards[i];
781 }
782 const Matrix values = eliminate(std::move(rows));
783 Row gains(n);
784 for (const CommunicatingClass& group : graph.classes) {
785 if (!group.closed) continue;
786 const Row pi = stationary(chain, group.members);
787 Fraction average;
788 for (std::size_t j = 0; j < group.members.size(); ++j)
789 average = average + pi[j] * rewards[group.members[j]];
790 const Hitting result = hitting(chain, group.members, graph.reach, true);
791 for (int i = 0; i < n; ++i) gains[i] = gains[i] + result.probabilities[i] * average;
792 }
793 std::cout << "REWARD " << name << " gamma=" << gamma << '\n';
794 for (int i = 0; i < n; ++i)
795 std::cout << "VALUE " << chain.names[i] << " discounted=" << values[i][0].value()
796 << " gain=" << gains[i].value() << '\n';
797 } else if (command == "RESTRICT") {
798 States selected = chain.states(Names(words.begin() + 3, words.end()));
799 std::sort(selected.begin(), selected.end());
800 std::vector<bool> inside(n);
801 for (int state : selected) inside[state] = true;
802 for (int state : selected)
803 for (int j = 0; j < n; ++j)
804 if (!inside[j] && chain.p[state][j].numerator > 0) fail("NOT CLOSED " + chain.names[state]);
805 Chain restricted(chain.labels(selected));
806 for (std::size_t i = 0; i < selected.size(); ++i)
807 for (std::size_t j = 0; j < selected.size(); ++j)
808 restricted.p[i][j] = chain.p[selected[i]][selected[j]];
809 chains[words[2]] = restricted;
810 std::cout << "RESTRICT " << name << ' ' << words[2] << " states=" << selected.size() << '\n';
811 restricted.display();
812 } else if (command == "SIMULATE") {
813 const int start = chain.state(words[2]);
814 const int steps = std::stoi(words[3]);
815 if (steps == 0) fail("BAD STEPS");
816 const Integer seed = std::stoll(words[4]);
817 if (seed == 0 || seed >= 2147483647) fail("BAD SEED");
818 Integer random = seed;
819 int current = start;
820 States path{start};
821 std::vector<int> visits(n);
822 ++visits[start];
823 for (int t = 0; t < steps; ++t) {
824 Integer common = 1;
825 for (Fraction chance : chain.p[current]) common = std::lcm(common, chance.denominator);
826 random = 48271 * random % 2147483647;
827 const Fraction threshold(random % common);
828 Fraction cumulative;
829 for (int j = 0; j < n; ++j) {
830 cumulative = cumulative + chain.p[current][j];
831 if (!(threshold < cumulative * Fraction(common))) continue;
832 current = j;
833 break;
834 }
835 path.push_back(current);
836 ++visits[current];
837 }
838 std::cout << "SIMULATE " << name << ' ' << words[2] << " steps=" << steps << " seed=" << seed
839 << " end=" << chain.names[current] << '\n';
840 std::cout << "PATH " << join(chain.labels(path)) << '\n';
841 for (int i = 0; i < n; ++i) std::cout << "VISIT " << chain.names[i] << ' ' << visits[i] << '\n';
842 }
843 }

845public:
846 void run(const std::string& line) {
847 const Names words = split(line);
848 if (words[0] == "CHAIN") { create(words); return; }
849 if (words[0] == "BUILD") { build(words); return; }
850 if (words[0] == "GAMBLER") { gambler(words); return; }
851 const auto entry = chains.find(words[1]);
852 if (entry == chains.end()) fail("UNKNOWN CHAIN " + words[1]);
853 if (!edit(words, entry->second)) analyze(words, entry->second);
854 }
