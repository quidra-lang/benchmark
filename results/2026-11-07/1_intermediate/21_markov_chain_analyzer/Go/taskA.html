43func valueText(value Rat) string {
44 whole := new(big.Int)
45 remainder := new(big.Int)
46 whole.DivMod(value.Num(), value.Denom(), remainder)
47 digits := new(big.Int).Mul(remainder, big.NewInt(1000000))
48 digits.Quo(digits, value.Denom())
49 return fmt.Sprintf("%s(%s.%06d)", fractionText(value), whole.String(), digits.Int64())

59func parseProbability(text string) (Rat, error) {
60 parts := strings.Split(text, "/")
61 numerator := new(big.Int)
62 numerator.SetString(parts[0], 10)
63 denominator := big.NewInt(1)
64 if len(parts) == 2 {
65 denominator = new(big.Int)
66 denominator.SetString(parts[1], 10)
67 }
68 if denominator.Sign() == 0 || numerator.Cmp(denominator) > 0 {
69 return nil, fmt.Errorf("BAD PROBABILITY")
70 }
71 return new(big.Rat).SetFrac(numerator, denominator), nil
72}

112func eliminate(matrix [][]Rat, size int) [][]Rat {
113 for col := 0; col < size; col++ {
114 pivotRow := col
115 for matrix[pivotRow][col].Sign() == 0 {
116 pivotRow++
117 }
118 matrix[col], matrix[pivotRow] = matrix[pivotRow], matrix[col]
119 pivot := matrix[col][col]
120 for k := range matrix[col] {
121 matrix[col][k] = div(matrix[col][k], pivot)
122 }
123 for row := 0; row < size; row++ {
124 if row == col {
125 continue
126 }
127 factor := matrix[row][col]
128 if factor.Sign() != 0 {
129 for k := range matrix[row] {
130 matrix[row][k] = sub(matrix[row][k], mul(factor, matrix[col][k]))
131 }
132 }
133 }
134 }
135 result := make([][]Rat, size)
136 for i := range result {
137 result[i] = matrix[i][size:]
138 }
139 return result
140}

222func (chain *Chain) reachable(start int) []bool {
223 seen := make([]bool, chain.size())
224 seen[start] = true
225 stack := []int{start}
226 for len(stack) > 0 {
227 i := stack[len(stack)-1]
228 stack = stack[:len(stack)-1]
229 for j := 0; j < chain.size(); j++ {
230 if chain.edge(i, j) && !seen[j] {
231 seen[j] = true
232 stack = append(stack, j)
233 }
234 }
235 }
236 return seen
237}

239func gcdInt(a, b int) int {
240 if a < 0 {
241 a = -a
242 }
243 if b < 0 {
244 b = -b
245 }
246 for b != 0 {
247 a, b = b, a%b
248 }
249 return a
250}

252type CommunicatingClass struct {
253 number int
254 members []int
255 closed bool
256 levels []int
257 period int
258}
259
260func newCommunicatingClass(chain *Chain, number int, members []int) *CommunicatingClass {
261 inside := membership(chain.size(), members)
262 closed := true
263 for _, i := range members {
264 for j := 0; j < chain.size(); j++ {
265 if chain.edge(i, j) && !inside[j] {
266 closed = false
267 }
268 }
269 }
270 levels := make([]int, chain.size())
271 for i := range levels {
272 levels[i] = -1
273 }
274 levels[members[0]] = 0
275 queue := []int{members[0]}
276 for len(queue) > 0 {
277 u := queue[0]
278 queue = queue[1:]
279 for _, v := range members {
280 if chain.edge(u, v) && levels[v] < 0 {
281 levels[v] = levels[u] + 1
282 queue = append(queue, v)
283 }
284 }
285 }
286 period := 0
287 for _, u := range members {
288 for _, v := range members {
289 if chain.edge(u, v) {
290 period = gcdInt(period, levels[u]+1-levels[v])
291 }
292 }
293 }
294 return &CommunicatingClass{number, members, closed, levels, period}
295}

297type Decomposition struct {
298 reach [][]bool
299 classes []*CommunicatingClass
300 classOf []int
301}
302
303func decompose(chain *Chain) *Decomposition {
304 n := chain.size()
305 d := &Decomposition{make([][]bool, n), nil, make([]int, n)}
306 for i := 0; i < n; i++ {
307 d.reach[i] = chain.reachable(i)
308 d.classOf[i] = -1
309 }
310 for i := 0; i < n; i++ {
311 if d.classOf[i] >= 0 {
312 continue
313 }
314 var members []int
315 for j := 0; j < n; j++ {
316 if d.reach[i][j] && d.reach[j][i] {
317 members = append(members, j)
318 d.classOf[j] = len(d.classes)
319 }
320 }
321 d.classes = append(d.classes, newCommunicatingClass(chain, len(d.classes)+1, members))
322 }
323 return d
324}
325
326func (d *Decomposition) closedClasses() []*CommunicatingClass {
327 var closed []*CommunicatingClass
328 for _, item := range d.classes {
329 if item.closed {
330 closed = append(closed, item)
331 }
332 }
333 return closed
334}
335
336func (d *Decomposition) containing(i int) *CommunicatingClass { return d.classes[d.classOf[i]] }

338func (d *Decomposition) reaches(i int, targets []bool) bool {
339 for j := range targets {
340 if d.reach[i][j] && targets[j] {
341 return true
342 }
343 }
344 return false
345}

347type Absorption struct {
348 transient []int
349 absorbing []int
350 fundamental [][]Rat
351 steps []Rat
352 probabilities [][]Rat
353}
354
355func newAbsorption(chain *Chain) *Absorption {
356 a := &Absorption{}
357 for i := 0; i < chain.size(); i++ {
358 if chain.absorbing(i) {
359 a.absorbing = append(a.absorbing, i)
360 } else {
361 a.transient = append(a.transient, i)
362 }
363 }
364 n := len(a.transient)
365 matrix := make([][]Rat, n)
366 for row, i := range a.transient {
367 values := make([]Rat, 0, 2*n)
368 for _, j := range a.transient {
369 values = append(values, sub(delta(i, j), chain.matrix[i][j]))
370 }
371 for k := 0; k < n; k++ {
372 values = append(values, delta(row, k))
373 }
374 matrix[row] = values
375 }
376 a.fundamental = eliminate(matrix, n)
377 a.steps = make([]Rat, n)
378 a.probabilities = make([][]Rat, n)
379 for i, row := range a.fundamental {
380 a.steps[i] = sum(row)
381 for _, j := range a.absorbing {
382 p := zero()
383 for k := 0; k < n; k++ {
384 p = add(p, mul(row[k], chain.matrix[a.transient[k]][j]))
385 }
386 a.probabilities[i] = append(a.probabilities[i], p)
387 }
388 }
389 return a
390}
391
392func (a *Absorption) variance(i int) Rat {
393 total := zero()
394 for j := range a.transient {
395 value := sub(mul(rat(2), a.fundamental[i][j]), delta(i, j))
396 total = add(total, mul(value, a.steps[j]))
397 }
398 return sub(total, mul(a.steps[i], a.steps[i]))
399}

401func stationaryDistribution(chain *Chain, members []int) []Rat {
402 n := len(members)
403 matrix := make([][]Rat, n)
404 for r := 0; r < n-1; r++ {
405 for k := 0; k < n; k++ {
406 matrix[r] = append(matrix[r], sub(chain.matrix[members[k]][members[r]], delta(k, r)))
407 }
408 matrix[r] = append(matrix[r], zero())
409 }
410 matrix[n-1] = make([]Rat, n+1)
411 for i := range matrix[n-1] {
412 matrix[n-1][i] = one()
413 }
414 solution := eliminate(matrix, n)
415 distribution := make([]Rat, n)
416 for i, row := range solution {
417 distribution[i] = row[0]
418 }
419 return distribution
420}

429func solveRestricted(chain *Chain, indices []int, constants []Rat) []Rat {
430 n := len(indices)
431 matrix := make([][]Rat, n)
432 for r, i := range indices {
433 for _, j := range indices {
434 matrix[r] = append(matrix[r], sub(delta(i, j), chain.matrix[i][j]))
435 }
436 matrix[r] = append(matrix[r], constants[r])
437 }
438 values := eliminate(matrix, n)
439 answer := make([]Rat, n)
440 for r := range values {
441 answer[r] = values[r][0]
442 }
443 return answer
444}

446func hittingProbabilities(chain *Chain, d *Decomposition, targets []bool) []Rat {
447 n := chain.size()
448 probabilities := make([]Rat, n)
449 remaining := []int{}
450 constants := []Rat{}
451 for i := 0; i < n; i++ {
452 if targets[i] {
453 probabilities[i] = one()
454 } else {
455 probabilities[i] = zero()
456 if d.reaches(i, targets) {
457 remaining = append(remaining, i)
458 value := zero()
459 for j := 0; j < n; j++ {
460 if targets[j] {
461 value = add(value, chain.matrix[i][j])
462 }
463 }
464 constants = append(constants, value)
465 }
466 }
467 }
468 if len(remaining) != 0 {
469 values := solveRestricted(chain, remaining, constants)
470 for k, i := range remaining {
471 probabilities[i] = values[k]
472 }
473 }
474 return probabilities
475}

499func regularExponent(chain *Chain) (int, bool) {
500 n := chain.size()
501 edges := make([][]bool, n)
502 for i := 0; i < n; i++ {
503 edges[i] = make([]bool, n)
504 for j := 0; j < n; j++ {
505 edges[i][j] = chain.edge(i, j)
506 }
507 }
508 power := edges
509 for exponent := 1; exponent < (n-1)*(n-1)+2; exponent++ {
510 allPositive := true
511 for _, row := range power {
512 for _, value := range row {
513 if !value {
514 allPositive = false
515 }
516 }
517 }
518 if allPositive {
519 return exponent, true
520 }
521 next := make([][]bool, n)
522 for i := 0; i < n; i++ {
523 next[i] = make([]bool, n)
524 for j := 0; j < n; j++ {
525 for k := 0; k < n; k++ {
526 if power[i][k] && edges[k][j] {
527 next[i][j] = true
528 break
529 }
530 }
531 }
532 }
533 power = next
534 }
535 return 0, false
536}

685func (a *Analyzer) cmdBuild(args []string) error {
686 name, order := args[0], parseInt(args[1])
687 if order != 1 && order != 2 {
688 return fmt.Errorf("BAD ORDER")
689 }
690 tokens := args[2:]
691 if len(tokens) <= order {
692 return fmt.Errorf("TOO FEW TOKENS")
693 }
694 observations := []string{}
695 if order == 1 {
696 observations = tokens
697 } else {
698 for i := 1; i < len(tokens); i++ {
699 observations = append(observations, tokens[i-1]+"."+tokens[i])
700 }
701 }
702 names := []string{}
703 positions := make(map[string]bool)
704 for _, observation := range observations {
705 if !positions[observation] {
706 positions[observation] = true
707 names = append(names, observation)
708 }
709 }
710 if len(names) > maxStates {
711 return fmt.Errorf("TOO MANY STATES")
712 }
713 c := newChain(names)
714 n := c.size()
715 counts := make([][]int, n)
716 for i := range counts {
717 counts[i] = make([]int, n)
718 }
719 for i := 1; i < len(observations); i++ {
720 counts[c.positions[observations[i-1]]][c.positions[observations[i]]]++
721 }
722 a.emit("BUILD", name, fmt.Sprintf("order=%d", order), fmt.Sprintf("states=%d", n),
723 fmt.Sprintf("transitions=%d", len(observations)-1))
724 for i := 0; i < n; i++ {
725 total := 0
726 for _, count := range counts[i] {
727 total += count
728 }
729 if total == 0 {
730 c.matrix[i][i] = one()
731 a.emit("FROM", names[i], "total=0", "absorbing")
732 continue
733 }
734 parts := []any{"FROM", names[i], fmt.Sprintf("total=%d", total)}
735 for j := 0; j < n; j++ {
736 if counts[i][j] > 0 {
737 c.matrix[i][j] = big.NewRat(int64(counts[i][j]), int64(total))
738 parts = append(parts, fmt.Sprintf("%s:%d", names[j], counts[i][j]))
739 }
740 }
741 a.emit(parts...)
742 }
743 a.chains[name] = c
744 return nil
745}

747func (a *Analyzer) cmdGambler(args []string) error {
748 name, goal, start := args[0], parseInt(args[1]), parseInt(args[2])
749 if goal < 1 || goal > maxGamblerSize {
750 return fmt.Errorf("BAD SIZE")
751 }
752 if start > goal {
753 return fmt.Errorf("BAD START")
754 }
755 p, err := parseProbability(args[3])
756 if err != nil {
757 return err
758 }
759 if equal(p, zero()) || equal(p, one()) {
760 return fmt.Errorf("BAD PROBABILITY")
761 }
762 names := make([]string, goal+1)
763 for i := range names {
764 names[i] = fmt.Sprintf("s%d", i)
765 }
766 c := newChain(names)
767 c.matrix[0][0], c.matrix[goal][goal] = one(), one()
768 for i := 1; i < goal; i++ {
769 c.matrix[i][i+1] = p
770 c.matrix[i][i-1] = sub(one(), p)
771 }
772 a.chains[name] = c
773 absorption := newAbsorption(c)
774 ruin := []Rat{one()}
775 duration := []Rat{zero()}
776 for i := range absorption.transient {
777 ruin = append(ruin, absorption.probabilities[i][0])
778 duration = append(duration, absorption.steps[i])
779 }
780 ruin = append(ruin, zero())
781 duration = append(duration, zero())
782 a.emit("GAMBLER", name, fmt.Sprintf("g=%d", goal),
783 "p="+fractionText(p), fmt.Sprintf("start=s%d", start),
784 "ruin="+valueText(ruin[start]), "duration="+valueText(duration[start]))
785 for i := 0; i <= goal; i++ {
786 a.emit("CAPITAL", fmt.Sprintf("s%d", i), "ruin="+valueText(ruin[i]),
787 "win="+valueText(sub(one(), ruin[i])), "duration="+valueText(duration[i]))
788 }
789 return nil
790}

792func (a *Analyzer) cmdClasses(args []string) error {
793 c, err := a.stochastic(args[0])
794 if err != nil {
795 return err
796 }
797 d := decompose(c)

1523 threshold := new(big.Int).Mod(big.NewInt(randomValue), common)
1524 thresholdRat := new(big.Rat).SetInt(threshold)
1525 commonRat := new(big.Rat).SetInt(common)
1526 cumulative := zero()
1527 for j, value := range row {

1529 if mul(cumulative, commonRat).Cmp(thresholdRat) > 0 {

1532 }
1533 }
1534 }
1535 a.emit("SIMULATE", args[0], args[1], fmt.Sprintf("steps=%d", steps),
1536 fmt.Sprintf("seed=%d", seed), "end="+c.names[path[len(path)-1]])
1537 parts := []any{"PATH"}
1538 for _, name := range c.labels(path) {
1539 parts = append(parts, name)
1540 }
1541 a.emit(parts...)
1542 for i, name := range c.names {
1543 count := 0
1544 for _, state := range path {
1545 if state == i {
1546 count++
1547 }
1548 }
1549 a.emit("VISIT", name, count)
1550 }

1552}
