  def row_sum(i)
    @matrix[i].sum
  def set_probability(args)
    name, source, target, text = args
    c = chain(name)
    i, j = c.state(source), c.state(target)
    value = parse_probability(text)
    c.matrix[i][j] = value
    emit("P", name, source, target, fraction_text(value), "rowsum=#{fraction_text(c.row_sum(i))}")
