4function gcd(a, b) {
5 a = absInt(a);
6 b = absInt(b);
7 while (b !== 0n) [a, b] = [b, a % b];
8 return a;

12class Fraction {
13 constructor(n, d = 1n) {
14 n = BigInt(n);
15 d = BigInt(d);
16 if (d === 0n) throw Error("Zero denominator");
17 if (d < 0n) [n, d] = [-n, -d];
18 const factor = gcd(n, d);
19 this.n = n / factor;
20 this.d = d / factor;
21 }
22 static of(value) { return value instanceof Fraction ? value : new Fraction(value); }
23 add(other) {
24 other = Fraction.of(other);
25 return new Fraction(this.n * other.d + other.n * this.d, this.d * other.d);
26 }
27 sub(other) {
28 other = Fraction.of(other);
29 return new Fraction(this.n * other.d - other.n * this.d, this.d * other.d);
30 }
31 mul(other) {
32 other = Fraction.of(other);
33 return new Fraction(this.n * other.n, this.d * other.d);
34 }
35 div(other) {
36 other = Fraction.of(other);
37 return new Fraction(this.n * other.d, this.d * other.n);
38 }
39 cmp(other) {
40 other = Fraction.of(other);
41 const difference = this.n * other.d - other.n * this.d;
42 return difference < 0n ? -1 : difference > 0n ? 1 : 0;
43 }
44 eq(other) { return this.cmp(other) === 0; }
45 abs() { return new Fraction(absInt(this.n), this.d); }
46 neg() { return new Fraction(-this.n, this.d); }
47 toString() { return this.n + "/" + this.d; }
48}
49const ZERO = new Fraction(0n);
50const ONE = new Fraction(1n);
51const fractionText = value => Fraction.of(value).toString();
52const answer = flag => flag ? "yes" : "no";

54function valueText(value) {
55 value = Fraction.of(value);
56 let whole = value.n / value.d;
57 let remainder = value.n % value.d;
58 if (remainder < 0n) {
59 --whole;
60 remainder += value.d;
61 }
62 const fraction = (remainder * 1000000n / value.d).toString().padStart(6, "0");
63 return fractionText(value) + "(" + whole + "." + fraction + ")";
64}
65function timeText(value) { return value === null ? "inf" : valueText(value); }
66function parseProbability(text) {
67 const parts = text.split("/");
68 const n = BigInt(parts[0]);
69 const d = parts.length === 2 ? BigInt(parts[1]) : 1n;
70 if (d === 0n || n > d) throw new CommandError("BAD PROBABILITY");
71 return new Fraction(n, d);
72}

75function identity(size) {
76 return Array.from({length: size}, (_, i) =>
77 Array.from({length: size}, (_, j) => delta(i, j)));

79function membership(size, states) {
80 const inside = Array(size).fill(false);
81 for (const state of states) inside[state] = true;
82 return inside;

84function eliminate(rows, size) {
85 for (let column = 0; column < size; ++column) {
86 let pivotRow = column;
87 while (rows[pivotRow][column].eq(ZERO)) ++pivotRow;
88 [rows[column], rows[pivotRow]] = [rows[pivotRow], rows[column]];
89 const pivot = rows[column][column];
90 rows[column] = rows[column].map(value => value.div(pivot));
91 for (let i = 0; i < size; ++i) {
92 const factor = rows[i][column];
93 if (i !== column && !factor.eq(ZERO)) {
94 rows[i] = rows[i].map((value, j) => value.sub(factor.mul(rows[column][j])));
95 }
96 }
97 }
98 return rows.map(row => row.slice(size));
99}
100function igcd(a, b) {
101 a = Math.abs(a);
102 b = Math.abs(b);
103 while (b !== 0) [a, b] = [b, a % b];
104 return a;

109class Chain {
110 constructor(names) {
111 this.names = [...names];
112 this.positions = new Map(names.map((name, i) => [name, i]));
113 this.matrix = names.map(() => names.map(() => ZERO));
114 }
115 get size() { return this.names.length; }
116 state(text) {
117 if (!this.positions.has(text)) throw new CommandError("UNKNOWN STATE " + text);
118 return this.positions.get(text);
119 }
120 distinctStates(texts) {
121 const found = new Set();
122 const states = [];
123 for (const text of texts) {
124 const state = this.state(text);
125 if (found.has(state)) throw new CommandError("DUPLICATE STATE " + text);
126 states.push(state);
127 found.add(state);
128 }
129 return states;
130 }
131 labels(states) { return states.map(i => this.names[i]); }
132 rowSum(i) { return sum(this.matrix[i]); }
133 blockSum(i, block) { return sum(block.map(j => this.matrix[i][j])); }
134 edge(i, j) { return this.matrix[i][j].cmp(ZERO) > 0; }
135 absorbing(i) { return this.matrix[i][i].eq(ONE); }
136 advance(distribution) {
137 return this.names.map((_, j) => sum(distribution.map((prob, i) =>
138 prob.mul(this.matrix[i][j]))));
139 }
140 reachable(start) {
141 const seen = Array(this.size).fill(false);
142 const stack = [start];
143 seen[start] = true;
144 while (stack.length) {
145 const i = stack.pop();
146 for (let j = 0; j < this.size; ++j) {
147 if (this.edge(i, j) && !seen[j]) {
148 seen[j] = true;
149 stack.push(j);
150 }
151 }
152 }
153 return seen;
154 }
155}

157class CommunicatingClass {
158 constructor(chain, number, members) {
159 this.number = number;
160 this.members = members;
161 const inside = membership(chain.size, members);
162 this.closed = !members.some(i => chain.names.some((_, j) => chain.edge(i, j) && !inside[j]));
163 this.levels = Array(chain.size).fill(-1);
164 const queue = [members[0]];
165 this.levels[members[0]] = 0;
166 for (let head = 0; head < queue.length; ++head) {
167 const u = queue[head];
168 for (const v of members) {
169 if (chain.edge(u, v) && this.levels[v] < 0) {
170 this.levels[v] = this.levels[u] + 1;
171 queue.push(v);
172 }
173 }
174 }
175 this.period = 0;
176 for (const u of members) {
177 for (const v of members) {
178 if (chain.edge(u, v)) {
179 this.period = igcd(this.period, this.levels[u] + 1 - this.levels[v]);
180 }
181 }
182 }
183 }

186class Decomposition {
187 constructor(chain) {
188 this.reach = chain.names.map((_, i) => chain.reachable(i));
189 this.classes = [];
190 this.classOf = Array(chain.size).fill(-1);
191 for (let i = 0; i < chain.size; ++i) {
192 if (this.classOf[i] >= 0) continue;
193 const members = chain.names.map((_, j) => j)
194 .filter(j => this.reach[i][j] && this.reach[j][i]);
195 for (const j of members) this.classOf[j] = this.classes.length;
196 this.classes.push(new CommunicatingClass(chain, this.classes.length + 1, members));
197 }
198 }
199 closedClasses() { return this.classes.filter(item => item.closed); }
200 containing(state) { return this.classes[this.classOf[state]]; }
201 reaches(state, targets) { return this.reach[state].some((found, i) => found && targets[i]); }
202}

204class Absorption {
205 constructor(chain) {
206 this.transient = chain.names.map((_, i) => i).filter(i => !chain.absorbing(i));
207 this.absorbing = chain.names.map((_, i) => i).filter(i => chain.absorbing(i));
208 const count = this.transient.length;
209 const rows = this.transient.map((i, r) =>
210 this.transient.map(j => delta(i, j).sub(chain.matrix[i][j]))
211 .concat(Array.from({length: count}, (_, c) => delta(r, c))));
212 this.fundamental = eliminate(rows, count);
213 this.steps = this.fundamental.map(row => sum(row));
214 this.probabilities = this.fundamental.map(row =>
215 this.absorbing.map(a => sum(row.map((rho, k) =>
216 rho.mul(chain.matrix[this.transient[k]][a])))));
217 }
218 variance(i) {
219 const total = sum(this.fundamental[i].map((value, j) =>
220 value.mul(2).sub(delta(i, j)).mul(this.steps[j])));
221 return total.sub(this.steps[i].mul(this.steps[i]));
222 }
223}

225function stationaryDistribution(chain, members) {
226 const n = members.length;
227 const rows = [];
228 for (let r = 0; r < n - 1; ++r) {
229 const coefficients = members.map((member, k) =>
230 chain.matrix[member][members[r]].sub(delta(k, r)));
231 rows.push(coefficients.concat([ZERO]));
232 }
233 rows.push(Array(n + 1).fill(ONE));
234 return eliminate(rows, n).map(row => row[0]);
235}
236function irreducibleDistribution(chain, decomposition) {
237 if (decomposition.classes.length > 1) throw new CommandError("NOT IRREDUCIBLE");
238 return stationaryDistribution(chain, decomposition.classes[0].members);
239}
240function solveRestricted(chain, states, constants) {
241 const rows = states.map((i, k) =>
242 states.map(j => delta(i, j).sub(chain.matrix[i][j])).concat([constants[k]]));
243 return eliminate(rows, states.length).map(row => row[0]);
244}
245function hittingProbabilities(chain, decomposition, targets) {
246 const probabilities = chain.names.map((_, i) => targets[i] ? ONE : ZERO);
247 const remaining = chain.names.map((_, i) => i)
248 .filter(i => !targets[i] && decomposition.reaches(i, targets));
249 if (remaining.length) {
250 const constants = remaining.map(i => sum(chain.names.map((_, j) =>
251 targets[j] ? chain.matrix[i][j] : ZERO)));
252 const values = solveRestricted(chain, remaining, constants);
253 remaining.forEach((i, k) => { probabilities[i] = values[k]; });
254 }
255 return probabilities;
256}
257function hittingTimes(chain, targets, probabilities) {
258 const times = chain.names.map((_, i) => targets[i] ? ZERO : null);
259 const certain = chain.names.map((_, i) => i)
260 .filter(i => !targets[i] && probabilities[i].eq(ONE));
261 if (certain.length) {
262 const values = solveRestricted(chain, certain, certain.map(() => ONE));
263 certain.forEach((i, k) => { times[i] = values[k]; });
264 }
265 return times;
266}
267function regularExponent(chain) {
268 const n = chain.size;
269 const edges = chain.names.map((_, i) => chain.names.map((_, j) => chain.edge(i, j)));
270 let power = edges;
271 for (let exponent = 1; exponent < (n - 1) * (n - 1) + 2; ++exponent) {
272 if (power.every(row => row.every(Boolean))) return exponent;
273 power = power.map((row, i) => row.map((_, j) =>
274 chain.names.some((__, k) => power[i][k] && edges[k][j])));
275 }
276 return null;
277}

382 gambler(a) {
383 const name = a[0], goal = Number(a[1]), start = Number(a[2]);
384 if (goal === 0 || goal > 9) throw new CommandError("BAD SIZE");
385 if (start > goal) throw new CommandError("BAD START");
386 const probability = parseProbability(a[3]);
387 if (probability.eq(ZERO) || probability.eq(ONE))
388 throw new CommandError("BAD PROBABILITY");
389 const chain = new Chain(Array.from({length: goal + 1}, (_, i) => "s" + i));
390 chain.matrix[0][0] = ONE;
391 chain.matrix[goal][goal] = ONE;
392 for (let i = 1; i < goal; ++i) {
393 chain.matrix[i][i + 1] = probability;
394 chain.matrix[i][i - 1] = ONE.sub(probability);
395 }
396 this.chains.set(name, chain);
397 const absorption = new Absorption(chain);
398 const ruin = [ONE, ...absorption.probabilities.map(row => row[0]), ZERO];
399 const duration = [ZERO, ...absorption.steps, ZERO];
400 this.emit("GAMBLER", name, "g=" + goal, "p=" + fractionText(probability),
401 "start=s" + start, "ruin=" + valueText(ruin[start]),
402 "duration=" + valueText(duration[start]));
403 for (let i = 0; i <= goal; ++i) {
404 this.emit("CAPITAL", "s" + i, "ruin=" + valueText(ruin[i]),
405 "win=" + valueText(ONE.sub(ruin[i])), "duration=" + valueText(duration[i]));

556 pathProbability(a) {
557 const chain = this.requireStochastic(a[0]);
558 const path = a.slice(1).map(name => chain.state(name));
559 let probability = ONE, zeroAt = null;
560 for (let t = 1; t < path.length; ++t) {

563 }
564 const parts = ["PATHPROB", a[0], "steps=" + (path.length - 1),
565 "prob=" + valueText(probability)];
566 if (zeroAt !== null) parts.push("zero_at=" + zeroAt);
567 this.emit(...parts);
568 }
569 mostLikely(a) {
570 const chain = this.requireStochastic(a[0]);
571 const start = chain.state(a[1]), steps = Number(a[2]);
572 if (steps === 0) throw new CommandError("BAD STEPS");
573 const best = Array.from({length: steps + 1}, () => Array(chain.size).fill(ONE));
574 for (let t = steps - 1; t >= 0; --t) {
575 best[t] = chain.names.map((_, i) =>
576 chain.names.map((__, j) => chain.matrix[i][j].mul(best[t + 1][j]))
577 .reduce((a, b) => a.cmp(b) >= 0 ? a : b));
578 }
579 const path = [start];
580 for (let t = 0; t < steps; ++t) {
581 const current = path[path.length - 1];
582 const next = chain.names.findIndex((_, j) =>
583 chain.matrix[current][j].mul(best[t + 1][j]).eq(best[t][current]));
584 path.push(next);
585 }
586 this.emit("MOSTLIKELY", a[0], a[1], "k=" + steps,
587 "prob=" + valueText(best[0][start]));
588 this.emit("PATH", ...chain.labels(path));
589 }
590 reversible(a) {
591 const chain = this.requireStochastic(a[0]);
592 const pi = irreducibleDistribution(chain, new Decomposition(chain));
593 const violations = [];
594 for (let i = 0; i < chain.size; ++i) {
595 for (let j = i + 1; j < chain.size; ++j) {
596 const forward = pi[i].mul(chain.matrix[i][j]);
597 const backward = pi[j].mul(chain.matrix[j][i]);
598 if (!forward.eq(backward)) violations.push([i, j, forward, backward]);
599 }
600 }
601 if (!violations.length) {
602 this.emit("REVERSIBLE", a[0], "yes");
603 return;
604 }
605 this.emit("REVERSIBLE", a[0], "no", "violations=" + violations.length);
606 for (const [i, j, forward, backward] of violations) {
607 this.emit("VIOLATION", chain.names[i], chain.names[j],
608 valueText(forward), valueText(backward));
609 }
610 }
611 reverse(a) {
612 const [name, reversedName] = a;
613 const chain = this.requireStochastic(name);
614 const pi = irreducibleDistribution(chain, new Decomposition(chain));
615 const reversed = new Chain(chain.names);
616 for (let i = 0; i < chain.size; ++i) {
617 for (let j = 0; j < chain.size; ++j) {
618 reversed.matrix[i][j] = pi[j].mul(chain.matrix[j][i]).div(pi[i]);
619 }
620 }
621 this.chains.set(reversedName, reversed);
622 const same = reversed.matrix.every((row, i) =>
623 row.every((value, j) => value.eq(chain.matrix[i][j])));
624 this.emit("REVERSE", name, reversedName, "reversible=" + answer(same));
625 this.display(reversed.names, reversed.matrix);
626 }
627 mixing(a) {
628 const chain = this.requireStochastic(a[0]);
629 const steps = Number(a[1]);
630 if (steps === 0) throw new CommandError("BAD STEPS");
631 const epsilon = parseProbability(a[2]);
632 const pi = irreducibleDistribution(chain, new Decomposition(chain));
633 let matrix = identity(chain.size), mixingTime = null;
634 const lines = [];
635 for (let t = 1; t <= steps; ++t) {
636 matrix = matrix.map(row => chain.advance(row));
637 const distances = matrix.map(row => sum(row.map((value, j) =>
638 value.sub(pi[j]).abs())).div(2));
639 const largest = distances.reduce((a, b) => a.cmp(b) >= 0 ? a : b);
640 const worst = distances.findIndex(distance => distance.eq(largest));
641 if (mixingTime === null && largest.cmp(epsilon) <= 0) mixingTime = t;
642 lines.push([t, largest, chain.names[worst]]);
643 }
644 this.emit("MIXING", a[0], "k=" + steps, "eps=" + fractionText(epsilon),
645 "time=" + (mixingTime === null ? "none" : mixingTime));
646 for (const [t, value, name] of lines) {
647 this.emit("TV", t, valueText(value), "from=" + name);
648 }
649 }
650 lump(a) {
651 const [name, lumpedName, partition] = a;
652 const chain = this.requireStochastic(name);
653 const seen = Array(chain.size).fill(false);
654 const blocks = [];
655 for (const group of partition.split("|")) {
656 const block = [];
657 for (const text of group.split(",")) {
658 const state = chain.state(text);
659 if (seen[state]) throw new CommandError("BAD PARTITION");
660 seen[state] = true;
661 block.push(state);
662 }
663 block.sort((a, b) => a - b);
664 blocks.push(block);
665 }
666 if (!seen.every(Boolean)) throw new CommandError("BAD PARTITION");
667 blocks.sort((a, b) => a[0] - b[0]);
668 for (const block of blocks) {
669 const representative = block[0];
670 for (const target of blocks) {
671 const expected = chain.blockSum(representative, target);
672 for (const i of block) {
673 const actual = chain.blockSum(i, target);
674 if (!actual.eq(expected)) {
675 this.emit("LUMP", name, "not", "lumpable",
676 "block=" + chain.names[representative],
677 "target=" + chain.names[target[0]],
678 chain.names[representative] + "=" + valueText(expected),
679 chain.names[i] + "=" + valueText(actual));
680 return;
681 }
682 }
683 }
684 }
685 const lumped = new Chain(blocks.map(block => chain.names[block[0]]));
686 lumped.matrix = blocks.map(block => blocks.map(target =>
687 chain.blockSum(block[0], target)));
688 this.chains.set(lumpedName, lumped);
689 this.emit("LUMP", name, lumpedName, "blocks=" + blocks.length);
690 this.display(lumped.names, lumped.matrix);
691 }
692 reward(a) {
693 const chain = this.requireStochastic(a[0]);
694 const texts = a.slice(2);
695 if (texts.length !== chain.size) throw new CommandError("BAD LENGTH");
696 const gamma = parseProbability(a[1]);
697 if (gamma.eq(ONE)) throw new CommandError("BAD DISCOUNT");
698 const rewards = texts.map(text => new Fraction(BigInt(text)));
699 const matrix = chain.names.map((_, i) =>
700 chain.names.map((__, j) => delta(i, j).sub(gamma.mul(chain.matrix[i][j])))
701 .concat([rewards[i]]));
702 const values = eliminate(matrix, chain.size).map(row => row[0]);
703 const decomposition = new Decomposition(chain);
704 const gains = chain.names.map(() => ZERO);
705 for (const item of decomposition.closedClasses()) {
706 const distribution = stationaryDistribution(chain, item.members);
707 const gain = sum(distribution.map((rho, j) => rho.mul(rewards[item.members[j]])));
708 const targets = membership(chain.size, item.members);
709 const probabilities = hittingProbabilities(chain, decomposition, targets);
710 for (let i = 0; i < chain.size; ++i) {
711 gains[i] = gains[i].add(probabilities[i].mul(gain));
712 }
713 }
714 this.emit("REWARD", a[0], "gamma=" + fractionText(gamma));
715 chain.names.forEach((name, i) => this.emit(
716 "VALUE", name, "discounted=" + valueText(values[i]),
717 "gain=" + valueText(gains[i])));
718 }
719 restrict(a) {
720 const [name, restrictedName, ...texts] = a;
721 const chain = this.requireStochastic(name);
722 const inside = membership(chain.size, chain.distinctStates(texts));
723 for (let i = 0; i < chain.size; ++i) {
724 if (inside[i] && chain.names.some((_, j) => chain.edge(i, j) && !inside[j]))
725 throw new CommandError("NOT CLOSED " + chain.names[i]);
726 }
727 const members = chain.names.map((_, i) => i).filter(i => inside[i]);
728 const restricted = new Chain(chain.labels(members));
729 restricted.matrix = members.map(i => members.map(j => chain.matrix[i][j]));
730 this.chains.set(restrictedName, restricted);
731 this.emit("RESTRICT", name, restrictedName, "states=" + members.length);
732 this.display(restricted.names, restricted.matrix);
733 }
734 simulate(a) {
735 const [name, source] = a;
736 const chain = this.requireStochastic(name);
737 const start = chain.state(source);
738 const steps = Number(a[2]), seed = Number(a[3]);
739 if (steps === 0) throw new CommandError("BAD STEPS");
740 if (seed === 0 || seed >= 2147483647) throw new CommandError("BAD SEED");
741 let random = seed;
742 const path = [start];
743 for (let t = 0; t < steps; ++t) {
744 const row = chain.matrix[path[path.length - 1]];
745 const common = row.map(value => value.d).reduce(lcm);
746 random = 48271 * random % 2147483647;
747 const threshold = BigInt(random) % common;
748 let cumulative = ZERO;
749 for (let j = 0; j < chain.size; ++j) {
750 cumulative = cumulative.add(row[j]);
751 if (cumulative.n * common > threshold * cumulative.d) {
752 path.push(j);
753 break;
754 }
755 }
756 }
757 this.emit("SIMULATE", name, source, "steps=" + steps, "seed=" + seed,
758 "end=" + chain.names[path[path.length - 1]]);
759 this.emit("PATH", ...chain.labels(path));
760 chain.names.forEach((state, i) => this.emit(
761 "VISIT", state, path.filter(value => value === i).length));
762 }
