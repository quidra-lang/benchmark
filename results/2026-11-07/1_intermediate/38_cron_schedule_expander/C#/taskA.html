8readonly record struct CivilDate(int Year, int Month, int Day)
9{
10 public int Number
11 {
12 get
13 {
14 int result = Day - 1;
15 for (int year = 2000; year < Year; year++) result += CalendarMath.Leap(year) ? 366 : 365;
16 for (int month = 1; month < Month; month++) result += CalendarMath.Days(Year, month);
17 return result;
18 }
19 }
20 public int Weekday => (Number + 6) % 7;
21 public override string ToString() => $"{Year:D4}-{Month:D2}-{Day:D2} {CalendarMath.Weekdays[Weekday][..3]}";
22}

24static class CalendarMath
25{
26 public static readonly string[] Weekdays = ["Sunday", "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday"];
27 public static readonly string[] Months = ["January", "February", "March", "April", "May", "June", "July", "August", "September", "October", "November", "December"];
28 static readonly int[] MonthDays = [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31];
29 public static bool Leap(int year) => year % 4 == 0 && (year % 100 != 0 || year % 400 == 0);
30 public static int Days(int year, int month) => month == 2 && Leap(year) ? 29 : MonthDays[month - 1];
31 public static bool Digits(string text, int minimum, int maximum) => text.Length >= minimum && text.Length <= maximum && text.All(c => c is >= '0' and <= '9');
32 public static int Integer(string text) => int.Parse(text, CultureInfo.InvariantCulture);
33 public static CivilDate FromNumber(int number)
34 {
35 int year = 2000, month = 1;
36 while (number >= (Leap(year) ? 366 : 365)) { number -= Leap(year) ? 366 : 365; year++; }
37 while (number >= Days(year, month)) { number -= Days(year, month); month++; }
38 return new CivilDate(year, month, number + 1);
39 }
40 static bool TryDate(string text, out CivilDate date)
41 {
42 date = default;
43 if (text.Length != 10 || text[4] != '-' || text[7] != '-' || !Digits(text[..4], 4, 4) || !Digits(text[5..7], 2, 2) || !Digits(text[8..], 2, 2)) return false;
44 int year = Integer(text[..4]), month = Integer(text[5..7]), day = Integer(text[8..]);
45 if (year is < 2001 or > 2098 || month is < 1 or > 12 || day < 1 || day > Days(year, month)) return false;
46 date = new CivilDate(year, month, day);
47 return true;
48 }
49 public static CivilDate ReadDate(string text) => TryDate(text, out CivilDate date) ? date : throw new CommandError("invalid date");
50 public static int ReadMinute(string dateText, string time)
51 {
52 if (!TryDate(dateText, out CivilDate date) || time.Length != 5 || time[2] != ':' || !Digits(time[..2], 2, 2) || !Digits(time[3..], 2, 2)) throw new CommandError("invalid datetime");
53 int hour = Integer(time[..2]), minute = Integer(time[3..]);
54 if (hour > 23 || minute > 59) throw new CommandError("invalid datetime");
55 return date.Number * 1440 + hour * 60 + minute;
56 }
57 public static (int Year, int Month) ReadMonth(string text)
58 {
59 if (text.Length != 7 || text[4] != '-' || !Digits(text[..4], 4, 4) || !Digits(text[5..], 2, 2)) throw new CommandError("invalid month");
60 int year = Integer(text[..4]), month = Integer(text[5..]);
61 if (year is < 2001 or > 2098 || month is < 1 or > 12) throw new CommandError("invalid month");
62 return (year, month);
63 }
64 public static string MinuteText(int minute)
65 {
66 CivilDate date = FromNumber(minute / 1440);
67 return $"{date.Year:D4}-{date.Month:D2}-{date.Day:D2} {minute % 1440 / 60:D2}:{minute % 60:D2} {Weekdays[date.Weekday][..3]}";
68 }
69 public static string Span(int minutes) => $"{minutes / 1440}d{minutes % 1440 / 60:D2}h{minutes % 60:D2}m";
70 public static void Window(int low, int high)
71 {
72 if (low > high) throw new CommandError("invalid window");
73 if (high - low > 44640) throw new CommandError("window too long");
74 }

96sealed class Cron
97{
98 public static readonly string[] FieldNames = ["minute", "hour", "day-of-month", "month", "day-of-week"];
99 static readonly int[] Minimum = [0, 0, 1, 1, 0], Maximum = [59, 23, 31, 12, 7];
100 static readonly string[] MonthNames = ["JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV", "DEC"];
101 static readonly string[] DayNames = ["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"];
102 static readonly Dictionary<string, string> Macros = new(StringComparer.Ordinal)
103 {
104 ["@yearly"] = "0 0 1 1 *", ["@annually"] = "0 0 1 1 *", ["@monthly"] = "0 0 1 * *",
105 ["@weekly"] = "0 0 * * 0", ["@daily"] = "0 0 * * *", ["@midnight"] = "0 0 * * *", ["@hourly"] = "0 * * * *"
106 };
107 public CronField[] Fields { get; }
108 public string Expanded { get; }
109 public int[] Daily { get; }
110 public bool OrRule => Fields[2].Restricted && Fields[4].Restricted;
111 public string Rule => OrRule ? "OR" : "AND";
112 Cron(string[] words, CronField[] fields)
113 {
114 Fields = fields;
115 Expanded = string.Join(' ', words);
116 Daily = fields[1].Values.SelectMany(hour => fields[0].Values.Select(minute => hour * 60 + minute)).ToArray();
117 }
118 static CommandError FieldError(int index, string reason) => new($"{FieldNames[index]} {reason}");
119 static int ReadValue(string text, int index)
120 {
121 int named = index == 3 ? Array.IndexOf(MonthNames, text) : index == 4 ? Array.IndexOf(DayNames, text) : -1;
122 if (named >= 0) return named + (index == 3 ? 1 : 0);
123 if (!CalendarMath.Digits(text, 1, 2)) throw FieldError(index, "invalid value");
124 int value = CalendarMath.Integer(text);
125 if (value < Minimum[index] || value > Maximum[index]) throw FieldError(index, "out of range");
126 return value;
127 }
128 static int SpecialNumber(string text, int index, int maximum, int digits = 2)
129 {
130 if (!CalendarMath.Digits(text, 1, digits)) throw FieldError(index, "invalid value");
131 int value = CalendarMath.Integer(text);
132 if (value < 1 || value > maximum) throw FieldError(index, "out of range");
133 return value;
134 }
135 static CronField ReadField(string text, int index)
136 {
137 SortedSet<int> values = [];
138 List<Special> specials = [];
139 foreach (string item in text.Split(','))
140 {
141 if (item.Length == 0) throw FieldError(index, "empty item");
142 Special? special = null;
143 int hash = item.IndexOf('#');
144 if (index == 2 && (item.StartsWith('L') || item.EndsWith('W')))
145 {
146 if (item == "L") special = new Special(SpecialKind.Last);
147 else if (item == "LW") special = new Special(SpecialKind.LastWeekday);
148 else if (item.StartsWith("L-", StringComparison.Ordinal)) special = new Special(SpecialKind.BeforeLast, SpecialNumber(item[2..], index, 30));
149 else if (item.EndsWith('W')) special = new Special(SpecialKind.NearestWeekday, SpecialNumber(item[..^1], index, 31));
150 else throw FieldError(index, "invalid value");
151 }
152 else if (index == 4 && hash >= 0)
153 {
154 int weekday = ReadValue(item[..hash], index);
155 int occurrence = SpecialNumber(item[(hash + 1)..], index, 5, 1);
156 special = new Special(SpecialKind.NthWeekday, weekday % 7, occurrence);
157 }
158 else if (index == 4 && item.Length >= 2 && item.EndsWith('L'))
159 special = new Special(SpecialKind.LastOccurrence, ReadValue(item[..^1], index) % 7);
160 if (special is Special entry)
161 {
162 if (!specials.Contains(entry)) specials.Add(entry);
163 continue;
164 }
165 int slash = item.IndexOf('/');
166 string basis = slash < 0 ? item : item[..slash];
167 int low, high;
168 if (basis == "*") { low = Minimum[index]; high = Maximum[index]; }
169 else
170 {
171 int dash = basis.IndexOf('-');
172 if (dash >= 0)
173 {
174 low = ReadValue(basis[..dash], index);
175 high = ReadValue(basis[(dash + 1)..], index);
176 if (low > high) throw FieldError(index, "invalid range");
177 }
178 else { low = ReadValue(basis, index); high = slash < 0 ? low : Maximum[index]; }
179 }
180 int step = 1;
181 if (slash >= 0)
182 {
183 string part = item[(slash + 1)..];
184 if (!CalendarMath.Digits(part, 1, 2) || CalendarMath.Integer(part) == 0) throw FieldError(index, "invalid step");
185 step = CalendarMath.Integer(part);
186 }
187 for (int value = low; value <= high; value += step) values.Add(index == 4 ? value % 7 : value);
188 }
189 return new CronField(index, values.ToArray(), specials.ToArray(), text[0] != '*');
190 }
191 public static Cron Parse(string[] words)
192 {
193 if (words.Length == 1 && words[0].StartsWith('@'))
194 {
195 if (!Macros.TryGetValue(words[0], out string? replacement)) throw new CommandError("unknown macro");
196 words = replacement.Split(' ');
197 }
198 if (words.Length != 5) throw new CommandError("field count");
199 CronField[] fields = new CronField[5];
200 for (int i = 0; i < 5; i++) fields[i] = ReadField(words[i], i);
201 return new Cron(words, fields);
202 }
203 static int Nearest(CivilDate date, int day)
204 {
205 int weekday = (date.Number - date.Day + day + 6) % 7;
206 return weekday == 6 ? day == 1 ? 3 : day - 1 : weekday == 0 ? day == CalendarMath.Days(date.Year, date.Month) ? day - 2 : day + 1 : day;
207 }
208 public (bool MonthDay, bool WeekDay, bool Condition) MatchDay(CivilDate date)
209 {
210 int last = CalendarMath.Days(date.Year, date.Month), day = date.Day, weekday = date.Weekday;
211 bool monthDay = Fields[2].Contains(day), weekDay = Fields[4].Contains(weekday);
212 foreach (Special special in Fields[2].Specials)
213 monthDay |= special.Kind switch
214 {
215 SpecialKind.Last => day == last, SpecialKind.LastWeekday => day == Nearest(date, last),
216 SpecialKind.BeforeLast => day == last - special.Value,
217 _ => special.Value <= last && day == Nearest(date, special.Value)
218 };
219 foreach (Special special in Fields[4].Specials)
220 weekDay |= weekday == special.Value && (special.Kind == SpecialKind.NthWeekday ? (day - 1) / 7 + 1 == special.Occurrence : day + 7 > last);
221 return (monthDay, weekDay, OrRule ? monthDay || weekDay : monthDay && weekDay);
222 }
223 public bool Scheduled(CivilDate date) => Fields[3].Contains(date.Month) && MatchDay(date).Condition;
224 public IEnumerable<int> FireTimes(int low, int high, HashSet<int> holidays, bool reverse = false)
225 {
226 int first = low / 1440, last = high / 1440;
227 for (int day = reverse ? last : first; reverse ? day >= first : day <= last; day += reverse ? -1 : 1)
228 {
229 if (holidays.Contains(day) || !Scheduled(CalendarMath.FromNumber(day))) continue;
230 for (int i = reverse ? Daily.Length - 1 : 0; reverse ? i >= 0 : i < Daily.Length; i += reverse ? -1 : 1)
231 {
232 int time = day * 1440 + Daily[i];
233 if (low <= time && time <= high) yield return time;
234 }
235 }
236 }
237 public string[] Reasons(int minute, HashSet<int> holidays)
238 {
239 CivilDate date = CalendarMath.FromNumber(minute / 1440);
240 var (monthDay, weekDay, condition) = MatchDay(date);
241 List<string> reasons = [];
242 if (!Fields[0].Contains(minute % 60)) reasons.Add("minute");
243 if (!Fields[1].Contains(minute % 1440 / 60)) reasons.Add("hour");
244 if (!condition && !monthDay) reasons.Add("day-of-month");
245 if (!Fields[3].Contains(date.Month)) reasons.Add("month");
246 if (!condition && !weekDay) reasons.Add("day-of-week");
247 if (holidays.Contains(date.Number)) reasons.Add("excluded");
248 return reasons.ToArray();
249 }
250 static string Join(IEnumerable<string> values, bool either = false)
251 {
252 string[] parts = values.ToArray();
253 if (parts.Length <= 1) return parts.Length == 0 ? "" : parts[0];
254 return string.Join(", ", parts[..^1]) + (either ? " or " : " and ") + parts[^1];
255 }
256 static string ValueName(int index, int value) => index == 3 ? CalendarMath.Months[value - 1] : index == 4 ? CalendarMath.Weekdays[value] : value.ToString(CultureInfo.InvariantCulture);
257 static string ValueList(CronField field, bool either = false)
258 {
259 List<string> parts = [];
260 for (int start = 0; start < field.Values.Length;)
261 {
262 int end = start;
263 while (end + 1 < field.Values.Length && field.Values[end + 1] == field.Values[end] + 1) end++;
264 if (end - start >= 2) parts.Add($"{ValueName(field.Index, field.Values[start])} through {ValueName(field.Index, field.Values[end])}");
265 else for (int i = start; i <= end; i++) parts.Add(ValueName(field.Index, field.Values[i]));
266 start = end + 1;
267 }
268 return Join(parts, either);
269 }
270 static int Step(CronField field)
271 {
272 int[] values = field.Values;
273 if (values.Length < 2 || values[0] != 0 || values[1] < 2) return 0;
274 int step = values[1];
275 return Enumerable.Range(0, Maximum[field.Index] / step + 1).Select(i => i * step).SequenceEqual(values) ? step : 0;
276 }
277 string DayClause(bool weekday, bool either = false)
278 {
279 CronField field = Fields[weekday ? 4 : 2];
280 List<string> parts = [];
281 if (field.Values.Length != 0)
282 parts.Add(weekday ? ValueList(field, either) : field.Values.Length == 1 ? $"day {field.Values[0]}" : "days " + ValueList(field));
283 string[] ordinals = ["1st", "2nd", "3rd", "4th", "5th"];
284 foreach (Special special in field.Specials)
285 parts.Add(special.Kind switch
286 {
287 SpecialKind.Last => "the last day", SpecialKind.LastWeekday => "the last weekday",
288 SpecialKind.BeforeLast => special.Value == 1 ? "1 day before the last day" : $"{special.Value} days before the last day",
289 SpecialKind.NearestWeekday => $"the weekday nearest day {special.Value}",
290 SpecialKind.NthWeekday => $"the {ordinals[special.Occurrence - 1]} {CalendarMath.Weekdays[special.Value]}",
291 _ => $"the last {CalendarMath.Weekdays[special.Value]}"
292 });
293 return Join(parts, either) + (!weekday || field.Specials.Length != 0 ? " of the month" : "");
294 }
295 public string Describe()
296 {
297 CronField minute = Fields[0], hour = Fields[1];
298 string text;
299 if (Daily.Length <= 4) text = "At " + Join(Daily.Select(t => $"{t / 60:D2}:{t % 60:D2}"));
300 else
301 {
302 int minuteStep = Step(minute), hourStep = Step(hour);
303 text = minute.Values.Length == 60 ? "Every minute" : minuteStep != 0 ? $"Every {minuteStep} minutes"
304 : minute.Values.Length == 1 ? $"At minute {minute.Values[0]}" : "At minutes " + ValueList(minute);
305 if (hour.Values.Length == 24)
306 {
307 if (text.StartsWith("At minute", StringComparison.Ordinal)) text += " of every hour";
308 }
309 else if (hourStep != 0) text += $", every {hourStep} hours";
310 else if (hour.Values[^1] - hour.Values[0] + 1 == hour.Values.Length) text += $", between {hour.Values[0]:D2}:00 and {hour.Values[^1]:D2}:59";
311 else text += ", during hours " + ValueList(hour);
312 }
313 bool monthFull = Fields[2].Values.Length == 31, weekFull = Fields[4].Values.Length == 7;
314 if (OrRule)
315 {
316 if (!monthFull && !weekFull) text += $", on {DayClause(false)} or on {DayClause(true)}";
317 }
318 else if (!monthFull && !weekFull) text += $", on {DayClause(false)} if it is {DayClause(true, true)}";
319 else if (!monthFull) text += ", on " + DayClause(false);
320 else if (!weekFull) text += ", on " + DayClause(true);
321 if (Fields[3].Values.Length != 12) text += ", in " + ValueList(Fields[3]);
322 return text;
323 }

342sealed class Scheduler
343{
344 readonly Dictionary<string, Job> jobs = new(StringComparer.Ordinal);
345 readonly HashSet<int> holidays = [];
346 static readonly Dictionary<string, int> Arity = new(StringComparer.Ordinal)
347 {
348 ["JOB"] = -2, ["VALIDATE"] = -1, ["REMOVE"] = 1, ["RENAME"] = 2, ["ENABLE"] = 1, ["DISABLE"] = 1,
349 ["RUNTIME"] = 3, ["LIST"] = 0, ["EXPAND"] = 2, ["DESCRIBE"] = 1, ["MATCH"] = 3, ["NEXT"] = 4,
350 ["PREV"] = 4, ["COUNT"] = 5, ["GAP"] = 3, ["OVERLAP"] = 6, ["TIMELINE"] = 4, ["CONFLICTS"] = 4,
351 ["WEEK"] = 1, ["CALENDAR"] = 2, ["SIMULATE"] = 4, ["EXCLUDE"] = 1, ["INCLUDE"] = 1, ["HOLIDAYS"] = 0
352 };
353 Job GetJob(string id) => jobs.TryGetValue(id, out Job? job) ? job : throw new CommandError("unknown job");
354 void NewId(string id)
355 {
356 if (id.Length is < 1 or > 16 || id[0] is < 'a' or > 'z' || id.Any(c => !(c is >= 'a' and <= 'z' or >= '0' and <= '9' or '_'))) throw new CommandError("invalid id");
357 if (jobs.ContainsKey(id)) throw new CommandError("duplicate job");
358 }
359 static int Positive(string text, int digits, int maximum, string reason)
360 {
361 if (!CalendarMath.Digits(text, 1, digits) || text[0] == '0' || CalendarMath.Integer(text) > maximum) throw new CommandError(reason);
362 return CalendarMath.Integer(text);
363 }
364 Job[] Ordered(bool enabledOnly = false) => jobs.Values.Where(job => !enabledOnly || job.Enabled).OrderBy(job => job.Id, StringComparer.Ordinal).ToArray();
365 SortedDictionary<int, List<string>> Schedule(int low, int high)
366 {
367 SortedDictionary<int, List<string>> result = new();
368 foreach (Job job in Ordered(true))
369 foreach (int time in job.Cron.FireTimes(low, high, holidays))
370 {
371 if (!result.TryGetValue(time, out List<string>? ids)) result[time] = ids = [];
372 ids.Add(job.Id);
373 }
374 return result;
375 }
376 List<string> Timeline(int low, int high, bool conflicts)
377 {
378 SortedDictionary<int, List<string>> schedule = Schedule(low, high);
379 List<string> rows = [];
380 if (conflicts)
381 {
382 var entries = schedule.Where(entry => entry.Value.Count >= 2).ToArray();
383 int peak = entries.Length == 0 ? 0 : entries.Max(entry => entry.Value.Count);
384 rows.Add($"CONFLICTS {entries.Length} peak {peak}");
385 foreach (var entry in entries.Take(100)) rows.Add($"{CalendarMath.MinuteText(entry.Key)} {entry.Value.Count} {string.Join(' ', entry.Value)}");
386 if (entries.Length > 100) rows.Add($"... {entries.Length - 100} more");
387 }
388 else
389 {
390 int count = schedule.Values.Sum(ids => ids.Count), written = 0;
391 rows.Add($"TIMELINE {count}");
392 foreach (var entry in schedule)
393 foreach (string id in entry.Value)
394 {
395 if (written == 100) break;
396 rows.Add($"{CalendarMath.MinuteText(entry.Key)} {id}");
397 written++;
398 }
399 if (count > 100) rows.Add($"... {count - 100} more");
400 }
401 return rows;
402 }
403 List<string> Week(string input, CivilDate start)
404 {
405 int[,] counts = new int[7, 24];
406 int low = start.Number * 1440, sum = 0;
407 foreach (Job job in Ordered(true))
408 foreach (int time in job.Cron.FireTimes(low, low + 7 * 1440 - 1, holidays)) { counts[(time - low) / 1440, time % 1440 / 60]++; sum++; }
409 List<string> rows = [$"WEEK {input} {sum}"];
410 for (int day = 0; day < 7; day++)
411 {
412 char[] marks = new char[24];
413 int daySum = 0;
414 for (int hour = 0; hour < 24; hour++)
415 {
416 int count = counts[day, hour]; daySum += count;
417 marks[hour] = count == 0 ? '.' : count >= 10 ? '+' : (char)('0' + count);
418 }
419 rows.Add($"{CalendarMath.FromNumber(start.Number + day)} {new string(marks)} {daySum}");
420 }
421 return rows;
422 }
423 List<string> Month(Job job, string input, int year, int month)
424 {
425 CivilDate first = new(year, month, 1);
426 List<string> cells = Enumerable.Repeat("...", first.Weekday).ToList();
427 int days = 0;
428 for (int day = 1; day <= CalendarMath.Days(year, month); day++)
429 {
430 CivilDate date = new(year, month, day);
431 char mark = job.Cron.Scheduled(date) ? holidays.Contains(date.Number) ? 'x' : '*' : '-';
432 if (mark == '*') days++;
433 cells.Add($"{day:D2}{mark}");
434 }
435 while (cells.Count % 7 != 0) cells.Add("...");
436 List<string> rows = [$"CALENDAR {job.Id} {input} days {days} runs {days * job.Cron.Daily.Length}", "Sun Mon Tue Wed Thu Fri Sat"];
437 for (int i = 0; i < cells.Count; i += 7) rows.Add(string.Join(' ', cells.Skip(i).Take(7)));
438 return rows;
439 }
440 List<string> Simulate(int low, int high)
441 {
442 Job[] enabled = Ordered(true);
443 SortedDictionary<int, int> events = new();
444 List<string> details = [];
445 foreach (Job job in enabled)
446 {
447 List<Execution> executions = [];
448 int skipped = 0, replaced = 0;
449 foreach (int time in job.Cron.FireTimes(low, high, holidays))
450 {
451 Execution? previous = executions.Count == 0 ? null : executions[^1];
452 if (previous is not null && previous.End > time)
453 {
454 if (job.Policy == Policy.FORBID) { skipped++; continue; }
455 if (job.Policy == Policy.REPLACE) { previous.End = time; replaced++; }
456 }
457 executions.Add(new Execution(time, time + job.Duration));
458 }
459 int busy = 0;
460 foreach (Execution execution in executions)
461 {
462 busy += execution.End - execution.Start;
463 events[execution.Start] = events.GetValueOrDefault(execution.Start) + 1;
464 events[execution.End] = events.GetValueOrDefault(execution.End) - 1;
465 }
466 details.Add($"{job.Id} {job.Policy} started {executions.Count} skipped {skipped} replaced {replaced} busy {busy}");
467 }
468 int running = 0, peak = 0, at = 0;
469 foreach (var entry in events)
470 {
471 running += entry.Value;
472 if (running > peak) { peak = running; at = entry.Key; }
473 }
474 List<string> rows = [$"SIMULATE {enabled.Length} peak {peak}" + (peak == 0 ? "" : " at " + CalendarMath.MinuteText(at))];
475 rows.AddRange(details);
476 return rows;
477 }
478 public List<string> Execute(string[] words)
479 {
480 string command = words[0];
481 if (!Arity.TryGetValue(command, out int expected)) throw new CommandError("unknown command");
482 int count = words.Length - 1;
483 if (expected < 0 ? count < -expected : count != expected) throw new CommandError("wrong arguments");
484 switch (command)
485 {
486 case "JOB":
487 {
488 string id = words[1]; NewId(id);
489 string[] expression = words[2..];
490 Cron cron = Cron.Parse(expression);
491 jobs[id] = new Job(id, cron, string.Join(' ', expression));
492 return [$"JOB {id} added"];
493 }
494 case "VALIDATE":
495 {
496 Cron cron = Cron.Parse(words[1..]);
497 return [$"VALID {cron.Expanded} {cron.Rule} {cron.Daily.Length}"];
498 }
499 case "REMOVE":
500 {
501 Job job = GetJob(words[1]); jobs.Remove(job.Id);
502 return [$"REMOVE {job.Id}"];
503 }
504 case "RENAME":
505 {
506 Job job = GetJob(words[1]); NewId(words[2]);
507 jobs.Remove(job.Id); job.Id = words[2]; jobs.Add(job.Id, job);
508 return [$"RENAME {words[1]} {job.Id}"];
509 }
510 case "ENABLE": case "DISABLE":
511 {
512 Job job = GetJob(words[1]);
513 bool enable = command == "ENABLE";
514 if (job.Enabled == enable) throw new CommandError(enable ? "already enabled" : "already disabled");
515 job.Enabled = enable;
516 return [$"{command} {job.Id}"];
517 }
518 case "RUNTIME":
519 {
520 Job job = GetJob(words[1]);
521 int duration = Positive(words[2], 4, 1440, "invalid duration");
522 Policy policy = words[3] switch { "ALLOW" => Policy.ALLOW, "FORBID" => Policy.FORBID, "REPLACE" => Policy.REPLACE, _ => throw new CommandError("invalid policy") };
523 job.Duration = duration; job.Policy = policy;
524 return [$"RUNTIME {job.Id} {duration} {policy}"];
525 }
526 case "LIST":
527 {
528 List<string> rows = [$"LIST {jobs.Count}"];
529 foreach (Job job in Ordered()) rows.Add($"{job.Id} {(job.Enabled ? "on" : "off")} {job.Duration} {job.Policy} {job.Text}");
530 return rows;
531 }
532 case "EXPAND":
533 {
534 Job job = GetJob(words[1]);
535 int index = Array.IndexOf(Cron.FieldNames, words[2]);
536 if (index < 0) throw new CommandError("invalid field");
537 CronField field = job.Cron.Fields[index];
538 string[] items = field.Values.Select(value => value.ToString(CultureInfo.InvariantCulture)).Concat(field.Specials.Select(item => item.ToString())).ToArray();
539 return [$"EXPAND {job.Id} {words[2]} {field.Values.Length}" + (items.Length == 0 ? "" : " " + string.Join(' ', items))];
540 }
541 case "DESCRIBE":
542 {
543 Job job = GetJob(words[1]); return [$"DESCRIBE {job.Id} {job.Cron.Describe()}"];
544 }
545 case "MATCH":
546 {
547 Job job = GetJob(words[1]);
548 int time = CalendarMath.ReadMinute(words[2], words[3]);
549 string[] reasons = job.Cron.Reasons(time, holidays);
550 return [$"MATCH {job.Id} {CalendarMath.MinuteText(time)} " + (reasons.Length == 0 ? "yes" : "no " + string.Join(' ', reasons))];
551 }
552 case "NEXT": case "PREV":
553 {
554 Job job = GetJob(words[1]);
555 int start = CalendarMath.ReadMinute(words[2], words[3]);
556 int limit = Positive(words[4], 2, 50, "invalid count");
557 bool reverse = command == "PREV";
558 int[] times = job.Cron.FireTimes(reverse ? start - 527040 : start + 1, reverse ? start - 1 : start + 527040, holidays, reverse).Take(limit).ToArray();
559 List<string> rows = [$"{command} {job.Id} {times.Length}"];
560 foreach (int time in times) rows.Add($"{CalendarMath.MinuteText(time)} {(reverse ? '-' : '+')}{CalendarMath.Span(Math.Abs(time - start))}");
561 if (times.Length < limit) rows.Add("NONE");
562 return rows;
563 }
564 case "COUNT":
565 {
566 Job job = GetJob(words[1]);
567 int low = CalendarMath.ReadMinute(words[2], words[3]);
568 int high = CalendarMath.ReadMinute(words[4], words[5]); CalendarMath.Window(low, high);
569 int[] times = job.Cron.FireTimes(low, high, holidays).ToArray();
570 return [$"COUNT {job.Id} runs {times.Length} days {times.Select(time => time / 1440).Distinct().Count()}"];
571 }
572 case "GAP":
573 {
574 Job job = GetJob(words[1]);
575 int start = CalendarMath.ReadMinute(words[2], words[3]);
576 int[] times = job.Cron.FireTimes(start, start + 44640, holidays).ToArray();
577 if (times.Length < 2) return [$"GAP {job.Id} runs {times.Length} none"];
578 SortedDictionary<int, int> frequencies = new();
579 int shortest = int.MaxValue, longest = 0, from = 0;
580 for (int i = 1; i < times.Length; i++)
581 {
582 int gap = times[i] - times[i - 1];
583 shortest = Math.Min(shortest, gap);
584 if (gap > longest) { longest = gap; from = i - 1; }
585 frequencies[gap] = frequencies.GetValueOrDefault(gap) + 1;
586 }
587 List<string> rows = [$"GAP {job.Id} runs {times.Length} shortest {CalendarMath.Span(shortest)} longest {CalendarMath.Span(longest)} from {CalendarMath.MinuteText(times[from])} to {CalendarMath.MinuteText(times[from + 1])}"];
588 foreach (var entry in frequencies) rows.Add($"{CalendarMath.Span(entry.Key)} {entry.Value}");
589 return rows;
590 }
591 case "OVERLAP":
592 {
593 Job a = GetJob(words[1]); Job b = GetJob(words[2]);
594 int low = CalendarMath.ReadMinute(words[3], words[4]);
595 int high = CalendarMath.ReadMinute(words[5], words[6]); CalendarMath.Window(low, high);
596 HashSet<int> other = b.Cron.FireTimes(low, high, holidays).ToHashSet();
597 int[] times = a.Cron.FireTimes(low, high, holidays).Where(other.Contains).ToArray();
598 return [$"OVERLAP {a.Id} {b.Id} {times.Length}" + (times.Length == 0 ? "" : $" first {CalendarMath.MinuteText(times[0])} last {CalendarMath.MinuteText(times[^1])}")];
599 }
600 case "TIMELINE": case "CONFLICTS": case "SIMULATE":
601 {
602 int low = CalendarMath.ReadMinute(words[1], words[2]);
603 int high = CalendarMath.ReadMinute(words[3], words[4]); CalendarMath.Window(low, high);
604 return command == "SIMULATE" ? Simulate(low, high) : Timeline(low, high, command == "CONFLICTS");
605 }
606 case "WEEK":
607 {
608 CivilDate date = CalendarMath.ReadDate(words[1]); return Week(words[1], date);
609 }
610 case "CALENDAR":
611 {
612 Job job = GetJob(words[1]);
613 var (year, month) = CalendarMath.ReadMonth(words[2]);
614 return Month(job, words[2], year, month);
615 }
616 case "EXCLUDE": case "INCLUDE":
617 {
618 CivilDate date = CalendarMath.ReadDate(words[1]);
619 int day = date.Number;
620 if (command == "EXCLUDE")
621 {
622 if (!holidays.Add(day)) throw new CommandError("already excluded");
623 }
624 else if (!holidays.Remove(day)) throw new CommandError("not excluded");
625 return [$"{command} {date} holidays {holidays.Count}"];
626 }
627 case "HOLIDAYS":
628 {
629 List<string> rows = [$"HOLIDAYS {holidays.Count}"];
630 rows.AddRange(holidays.Order().Select(day => CalendarMath.FromNumber(day).ToString()));
631 return rows;
632 }
633 }
634 throw new InvalidOperationException();
635 }
636}

638static class Program
639{
640 static void Main()
641 {
642 CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
643 string? first = Console.ReadLine();
644 if (first is null) return;
645 int count = CalendarMath.Integer(first);
646 Scheduler scheduler = new();
647 for (int i = 0; i < count; i++)
648 {
649 string[] words = Console.ReadLine()!.Split(' ', StringSplitOptions.RemoveEmptyEntries);
650 try { foreach (string row in scheduler.Execute(words)) Console.WriteLine(row); }
651 catch (CommandError error) { Console.WriteLine($"ERROR {error.Message}"); }
652 }
653 }
