20func digits(text string, least, most int) bool {
21 if len(text) < least || len(text) > most {
22 return false
23 }
24 for _, c := range text {
25 if c < '0' || c > '9' {
26 return false
27 }
28 }
29 return true
30}
31func integer(text string) int { value, _ := strconv.Atoi(text); return value }
32func leap(year int) bool { return year%4 == 0 && (year%100 != 0 || year%400 == 0) }
33func monthLength(year, month int) int {
34 if month == 2 && leap(year) {
35 return 29
36 }
37 return [...]int{31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31}[month-1]
38}

40type Date struct{ year, month, day int }
41
42func (d Date) number() int {
43 result := d.day - 1
44 for year := 2000; year < d.year; year++ {
45 if leap(year) { result += 366 } else { result += 365 }
46 }
47 for month := 1; month < d.month; month++ {
48 result += monthLength(d.year, month)
49 }
50 return result
51}
52func dateFromNumber(number int) Date {
53 year, month := 2000, 1
54 for {
55 days := 365
56 if leap(year) { days++ }
57 if number < days { break }
58 number -= days
59 year++
60 }
61 for number >= monthLength(year, month) {
62 number -= monthLength(year, month)
63 month++
64 }
65 return Date{year, month, number + 1}
66}
67func (d Date) weekday() int { return (d.number() + 6) % 7 }
68func (d Date) String() string {
69 return fmt.Sprintf("%04d-%02d-%02d %s", d.year, d.month, d.day, weekNames[d.weekday()][:3])
70}
71func pattern(text, form string) bool {
72 if len(text) != len(form) { return false }
73 for i := range len(form) {
74 if form[i] == '#' {
75 if text[i] < '0' || text[i] > '9' { return false }
76 } else if text[i] != form[i] { return false }
77 }
78 return true
79}
80func readDate(text, reason string) (Date, error) {
81 if !pattern(text, "####-##-##") { return Date{}, errors.New(reason) }
82 date := Date{integer(text[:4]), integer(text[5:7]), integer(text[8:])}
83 if date.year < 2001 || date.year > 2098 || date.month < 1 || date.month > 12 || date.day < 1 || date.day > monthLength(date.year, date.month) {
84 return Date{}, errors.New(reason)
85 }
86 return date, nil
87}
88func readDatetime(words []string, at int) (int, error) {
89 date, err := readDate(words[at], "invalid datetime")
90 if err != nil { return 0, err }
91 time := words[at+1]
92 if !pattern(time, "##:##") { return 0, errors.New("invalid datetime") }
93 hour, minute := integer(time[:2]), integer(time[3:])
94 if hour > 23 || minute > 59 { return 0, errors.New("invalid datetime") }
95 return date.number()*1440 + hour*60 + minute, nil
96}
97func readMonth(text string) (Date, error) {
98 if !pattern(text, "####-##") { return Date{}, errors.New("invalid month") }
99 date := Date{integer(text[:4]), integer(text[5:]), 1}
100 if date.year < 2001 || date.year > 2098 || date.month < 1 || date.month > 12 {
101 return Date{}, errors.New("invalid month")
102 }
103 return date, nil
104}
105func datetimeText(time int) string {
106 date := dateFromNumber(time / 1440)
107 return fmt.Sprintf("%04d-%02d-%02d %02d:%02d %s", date.year, date.month, date.day, time%1440/60, time%60, weekNames[date.weekday()][:3])
108}
109func span(minutes int) string { return fmt.Sprintf("%dd%02dh%02dm", minutes/1440, minutes%1440/60, minutes%60) }
110func positive(text string, width, maximum int, reason string) (int, error) {
111 if !digits(text, 1, width) || text[0] == '0' || integer(text) > maximum { return 0, errors.New(reason) }
112 return integer(text), nil
113}
114func readWindow(words []string, at int) (int, int, error) {
115 low, err := readDatetime(words, at)
116 if err != nil { return 0, 0, err }
117 high, err := readDatetime(words, at+2)
118 if err != nil { return 0, 0, err }
119 if low > high { return 0, 0, errors.New("invalid window") }
120 if high-low > 44640 { return 0, 0, errors.New("window too long") }
121 return low, high, nil
122}

135type Special struct {

140func (s Special) String() string {
141 switch s.kind {
142 case LastDay: return "L"
143 case LastWorkday: return "LW"
144 case BeforeLast: return fmt.Sprintf("L-%d", s.value)
145 case NearestWorkday: return fmt.Sprintf("%dW", s.value)
146 case NthWeekday: return fmt.Sprintf("%d#%d", s.value, s.occurrence)
147 default: return fmt.Sprintf("%dL", s.value)
148 }

157func readValue(text string, index int) (int, error) {
158 names := []string{}
159 if index == 3 { names = []string{"JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV", "DEC"} }
160 if index == 4 { names = []string{"SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"} }
161 if at := slices.Index(names, text); at >= 0 {
162 if index == 3 { at++ }
163 return at, nil
164 }
165 if !digits(text, 1, 2) { return 0, errors.New("invalid value") }
166 value := integer(text)
167 if value < minimum[index] || value > maximum[index] { return 0, errors.New("out of range") }
168 return value, nil
169}
170func specialNumber(text string, upper, width int) (int, error) {
171 if !digits(text, 1, width) { return 0, errors.New("invalid value") }
172 value := integer(text)
173 if value < 1 || value > upper { return 0, errors.New("out of range") }
174 return value, nil
175}
176func (f *Field) addItem(item string, index int, values *[60]bool) error {
177 if item == "" { return errors.New("empty item") }
178 var special Special
179 isSpecial := false
180 if index == 2 && (strings.HasPrefix(item, "L") || strings.HasSuffix(item, "W")) {
181 isSpecial = true
182 switch {
183 case item == "L": special.kind = LastDay
184 case item == "LW": special.kind = LastWorkday
185 case strings.HasPrefix(item, "L-"):
186 n, err := specialNumber(item[2:], 30, 2)
187 if err != nil { return err }
188 special = Special{kind: BeforeLast, value: n}
189 case strings.HasSuffix(item, "W"):
190 n, err := specialNumber(item[:len(item)-1], 31, 2)
191 if err != nil { return err }
192 special = Special{kind: NearestWorkday, value: n}
193 default: return errors.New("invalid value")
194 }
195 } else if index == 4 && strings.Contains(item, "#") {
196 v, n, _ := strings.Cut(item, "#")
197 weekday, err := readValue(v, index)
198 if err != nil { return err }
199 occurrence, err := specialNumber(n, 5, 1)
200 if err != nil { return err }
201 special, isSpecial = Special{NthWeekday, weekday % 7, occurrence}, true
202 } else if index == 4 && len(item) >= 2 && strings.HasSuffix(item, "L") {
203 weekday, err := readValue(item[:len(item)-1], index)
204 if err != nil { return err }
205 special, isSpecial = Special{kind: LastWeekday, value: weekday % 7}, true
206 }
207 if isSpecial {
208 if !slices.Contains(f.specials, special) { f.specials = append(f.specials, special) }
209 return nil
210 }
211 base, stepPart, hasStep := strings.Cut(item, "/")
212 var first, last int
213 if base == "*" { first, last = minimum[index], maximum[index] } else {
214 a, b, hasRange := strings.Cut(base, "-")
215 var err error
216 first, err = readValue(a, index)
217 if err != nil { return err }
218 if hasRange {
219 last, err = readValue(b, index)
220 if err != nil { return err }
221 if first > last { return errors.New("invalid range") }
222 } else if hasStep { last = maximum[index] } else { last = first }
223 }
224 step := 1
225 if hasStep {
226 if !digits(stepPart, 1, 2) || integer(stepPart) == 0 { return errors.New("invalid step") }
227 step = integer(stepPart)
228 }
229 for value := first; value <= last; value += step {
230 normalized := value
231 if index == 4 { normalized %= 7 }
232 values[normalized] = true
233 }
234 return nil
235}
236func readField(text string, index int) (Field, error) {
237 field := Field{restricted: text[0] != '*'}
238 var values [60]bool
239 for _, item := range strings.Split(text, ",") {
240 if err := field.addItem(item, index, &values); err != nil { return Field{}, fmt.Errorf("%s %s", fieldNames[index], err) }
241 }
242 for value, present := range values {
243 if present { field.values = append(field.values, value) }
244 }
245 return field, nil
246}
247func (f Field) contains(value int) bool { _, found := slices.BinarySearch(f.values, value); return found }
248func wordList(parts []string, either bool) string {
249 if len(parts) < 2 { return strings.Join(parts, "") }
250 conjunction := " and "
251 if either { conjunction = " or " }
252 return strings.Join(parts[:len(parts)-1], ", ") + conjunction + parts[len(parts)-1]
253}
254func valueName(index, value int) string {
255 if index == 3 { return monthNames[value-1] }
256 if index == 4 { return weekNames[value] }
257 return strconv.Itoa(value)
258}
259func (f Field) valueList(index int, either bool) string {
260 var parts []string
261 for first := 0; first < len(f.values); {
262 last := first
263 for last+1 < len(f.values) && f.values[last+1] == f.values[last]+1 { last++ }
264 if last-first >= 2 { parts = append(parts, valueName(index, f.values[first])+" through "+valueName(index, f.values[last])) } else {
265 for _, value := range f.values[first:last+1] { parts = append(parts, valueName(index, value)) }
266 }
267 first = last + 1
268 }
269 return wordList(parts, either)
270}
271func (f Field) stepped(upper int) int {
272 if len(f.values) < 2 || f.values[0] != 0 || f.values[1] < 2 { return 0 }
273 step := f.values[1]
274 if len(f.values) != upper/step+1 { return 0 }
275 for i, value := range f.values { if value != i*step { return 0 } }
276 return step
277}
278func (f Field) dayPhrase(weekday, either bool) string {
279 var parts []string
280 if len(f.values) != 0 {
281 if weekday { parts = append(parts, f.valueList(4, either)) } else {
282 prefix := "days "
283 if len(f.values) == 1 { prefix = "day " }
284 parts = append(parts, prefix+f.valueList(2, false))
285 }
286 }
287 for _, special := range f.specials {
288 var text string
289 switch special.kind {
290 case LastDay: text = "the last day"
291 case LastWorkday: text = "the last weekday"
292 case BeforeLast:
293 noun := "days"
294 if special.value == 1 { noun = "day" }
295 text = fmt.Sprintf("%d %s before the last day", special.value, noun)
296 case NearestWorkday: text = fmt.Sprintf("the weekday nearest day %d", special.value)
297 case NthWeekday: text = fmt.Sprintf("the %s %s", [...]string{"1st", "2nd", "3rd", "4th", "5th"}[special.occurrence-1], weekNames[special.value])
298 case LastWeekday: text = "the last " + weekNames[special.value]
299 }
300 parts = append(parts, text)
301 }
302 text := wordList(parts, either)
303 if !weekday || len(f.specials) != 0 { text += " of the month" }
304 return text
305}

313func parseCron(words []string) (*Cron, error) {
314 cron := &Cron{text: strings.Join(words, " ")}
315 if len(words) == 1 && strings.HasPrefix(words[0], "@") {
316 var expansion string
317 switch words[0] {
318 case "@yearly", "@annually": expansion = "0 0 1 1 *"
319 case "@monthly": expansion = "0 0 1 * *"
320 case "@weekly": expansion = "0 0 * * 0"
321 case "@daily", "@midnight": expansion = "0 0 * * *"
322 case "@hourly": expansion = "0 * * * *"
323 default: return nil, errors.New("unknown macro")
324 }
325 words = strings.Split(expansion, " ")
326 }
327 if len(words) != 5 { return nil, errors.New("field count") }
328 cron.expanded = strings.Join(words, " ")
329 for i, word := range words {
330 field, err := readField(word, i)
331 if err != nil { return nil, err }
332 cron.fields[i] = field
333 }
334 for _, hour := range cron.fields[1].values {
335 for _, minute := range cron.fields[0].values { cron.daily = append(cron.daily, hour*60+minute) }
336 }
337 return cron, nil
338}
339func (c *Cron) orRule() bool { return c.fields[2].restricted && c.fields[4].restricted }
340func (c *Cron) matchDay(date Date) (bool, bool, bool) {
341 last, weekday := monthLength(date.year, date.month), date.weekday()
342 nearest := func(day int) int {
343 w := (date.number() - date.day + day + 6) % 7
344 if w == 6 { if day == 1 { return 3 }; return day - 1 }
345 if w == 0 { if day == last { return day - 2 }; return day + 1 }
346 return day
347 }
348 dom, dow := c.fields[2].contains(date.day), c.fields[4].contains(weekday)
349 for _, special := range c.fields[2].specials {
350 target := 0
351 switch special.kind {
352 case LastDay: target = last
353 case LastWorkday: target = nearest(last)
354 case BeforeLast: target = last - special.value
355 case NearestWorkday: if special.value <= last { target = nearest(special.value) }
356 }
357 dom = dom || target == date.day
358 }
359 for _, special := range c.fields[4].specials {
360 matches := date.day+7 > last
361 if special.kind == NthWeekday { matches = (date.day-1)/7+1 == special.occurrence }
362 dow = dow || weekday == special.value && matches
363 }
364 if c.orRule() { return dom, dow, dom || dow }
365 return dom, dow, dom && dow
366}
367func (c *Cron) scheduled(date Date) bool {
368 _, _, condition := c.matchDay(date)
369 return c.fields[3].contains(date.month) && condition
370}
371func (c *Cron) fires(low, high int, holidays map[int]bool, reverse bool, limit int) []int {
372 first, last, step := low/1440, high/1440, 1
373 day := first
374 if reverse { day, step = last, -1 }
375 var result []int
376 for ; day >= first && day <= last; day += step {
377 if holidays[day] || !c.scheduled(dateFromNumber(day)) { continue }
378 i := 0
379 if reverse { i = len(c.daily) - 1 }
380 for ; i >= 0 && i < len(c.daily); i += step {
381 time := day*1440 + c.daily[i]
382 if time < low || time > high { continue }
383 result = append(result, time)
384 if limit > 0 && len(result) == limit { return result }
385 }
386 }
387 return result
388}
389func (c *Cron) description() string {
390 minute, hour, dom, month, dow := c.fields[0], c.fields[1], c.fields[2], c.fields[3], c.fields[4]
391 var text string
392 if len(c.daily) <= 4 {
393 var times []string
394 for _, t := range c.daily { times = append(times, fmt.Sprintf("%02d:%02d", t/60, t%60)) }
395 text = "At " + wordList(times, false)
396 } else {
397 ms, hs := minute.stepped(59), hour.stepped(23)
398 switch {
399 case len(minute.values) == 60: text = "Every minute"
400 case ms > 0: text = fmt.Sprintf("Every %d minutes", ms)
401 case len(minute.values) == 1: text = fmt.Sprintf("At minute %d", minute.values[0])
402 default: text = "At minutes " + minute.valueList(0, false)
403 }
404 first, last := hour.values[0], hour.values[len(hour.values)-1]
405 switch {
406 case len(hour.values) == 24: if strings.HasPrefix(text, "At minute") { text += " of every hour" }
407 case hs > 0: text += fmt.Sprintf(", every %d hours", hs)
408 case last-first+1 == len(hour.values): text += fmt.Sprintf(", between %02d:00 and %02d:59", first, last)
409 default: text += ", during hours " + hour.valueList(1, false)
410 }
411 }
412 dFull, wFull := len(dom.values) == 31, len(dow.values) == 7
413 if c.orRule() {
414 if !dFull && !wFull { text += ", on " + dom.dayPhrase(false, false) + " or on " + dow.dayPhrase(true, false) }
415 } else if !dFull && !wFull { text += ", on " + dom.dayPhrase(false, false) + " if it is " + dow.dayPhrase(true, true) } else if !dFull {
416 text += ", on " + dom.dayPhrase(false, false)
417 } else if !wFull { text += ", on " + dow.dayPhrase(true, false) }
418 if len(month.values) != 12 { text += ", in " + month.valueList(3, false) }
419 return text
420}

444func (s *Scheduler) job(id string) (*Job, error) {

449func (s *Scheduler) newID(id string) error {
450 if len(id) < 1 || len(id) > 16 || id[0] < 'a' || id[0] > 'z' { return errors.New("invalid id") }
451 for _, c := range id {
452 if !(c >= 'a' && c <= 'z') && !(c >= '0' && c <= '9') && c != '_' { return errors.New("invalid id") }
453 }
454 if s.jobs[id] != nil { return errors.New("duplicate job") }
455 return nil
456}
457func (s *Scheduler) ordered(enabledOnly bool) []*Job {
458 var result []*Job
459 for _, job := range s.jobs { if !enabledOnly || job.enabled { result = append(result, job) } }
460 slices.SortFunc(result, func(a, b *Job) int { return strings.Compare(a.id, b.id) })
461 return result
462}
463func (s *Scheduler) timeline(low, high int, conflicts bool) []string {
464 jobs := s.ordered(true)
465 masks := make([]uint64, high-low+1)
466 for i, job := range jobs {
467 for _, time := range job.cron.fires(low, high, s.holidays, false, 0) { masks[time-low] |= uint64(1) << i }
468 }
469 total, peak := 0, 0
470 for _, mask := range masks {
471 count := bits.OnesCount64(mask)
472 if !conflicts { total += count } else if count >= 2 { total++; peak = max(peak, count) }
473 }
474 rows := []string{fmt.Sprintf("TIMELINE %d", total)}
475 if conflicts { rows[0] = fmt.Sprintf("CONFLICTS %d peak %d", total, peak) }
476 for offset, mask := range masks {
477 if len(rows) == 101 { break }
478 if conflicts {
479 if bits.OnesCount64(mask) < 2 { continue }
480 var ids []string
481 for i, job := range jobs { if mask&(uint64(1)<<i) != 0 { ids = append(ids, job.id) } }
482 rows = append(rows, fmt.Sprintf("%s %d %s", datetimeText(low+offset), len(ids), strings.Join(ids, " ")))
483 } else {
484 for i, job := range jobs {
485 if len(rows) == 101 { break }
486 if mask&(uint64(1)<<i) != 0 { rows = append(rows, datetimeText(low+offset)+" "+job.id) }
487 }
488 }
489 }
490 if total > 100 { rows = append(rows, fmt.Sprintf("... %d more", total-100)) }
491 return rows
492}
493func (s *Scheduler) simulate(low, high int) []string {
494 jobs := s.ordered(true)
495 events := make([]int, high-low+1441)
496 var rows []string
497 for _, job := range jobs {
498 started, skipped, replaced, busy, lastEnd := 0, 0, 0, 0, -1
499 for _, time := range job.cron.fires(low, high, s.holidays, false, 0) {
500 if lastEnd > time && job.policy == Forbid { skipped++; continue }
501 if lastEnd > time && job.policy == Replace {
502 events[lastEnd-low]++; events[time-low]--
503 busy -= lastEnd - time; replaced++
504 }
505 lastEnd = time + job.duration
506 events[time-low]++; events[lastEnd-low]--
507 started++; busy += job.duration
508 }
509 rows = append(rows, fmt.Sprintf("%s %s started %d skipped %d replaced %d busy %d", job.id, job.policy, started, skipped, replaced, busy))
510 }
511 running, peak, at := 0, 0, 0
512 for i, change := range events {
513 running += change
514 if running > peak { peak, at = running, low+i }
515 }
516 header := fmt.Sprintf("SIMULATE %d peak %d", len(jobs), peak)
517 if peak > 0 { header += " at " + datetimeText(at) }
518 return append([]string{header}, rows...)
519}

521var arity = map[string]int{
522 "JOB": -3, "VALIDATE": -2, "REMOVE": 2, "RENAME": 3, "ENABLE": 2, "DISABLE": 2,
523 "RUNTIME": 4, "LIST": 1, "EXPAND": 3, "DESCRIBE": 2, "MATCH": 4, "NEXT": 5,
524 "PREV": 5, "COUNT": 6, "GAP": 4, "OVERLAP": 7, "TIMELINE": 5, "CONFLICTS": 5,

528func (s *Scheduler) execute(words []string) ([]string, error) {
529 command := words[0]
530 expected, known := arity[command]
531 if !known { return nil, errors.New("unknown command") }
532 if expected < 0 && len(words) < -expected || expected >= 0 && len(words) != expected {
533 return nil, errors.New("wrong arguments")
534 }
535 var job *Job
536 var err error
537 switch command {
538 case "REMOVE", "RENAME", "ENABLE", "DISABLE", "RUNTIME", "EXPAND", "DESCRIBE", "MATCH", "NEXT", "PREV", "COUNT", "GAP", "OVERLAP", "CALENDAR":
539 job, err = s.job(words[1])
540 if err != nil { return nil, err }
541 }
542 switch command {
543 case "JOB":
544 if err := s.newID(words[1]); err != nil { return nil, err }
545 cron, err := parseCron(words[2:])
546 if err != nil { return nil, err }
547 s.jobs[words[1]] = &Job{id: words[1], cron: cron, enabled: true, duration: 1, policy: Allow}
548 return []string{"JOB " + words[1] + " added"}, nil
549 case "VALIDATE":
550 cron, err := parseCron(words[1:])
551 if err != nil { return nil, err }
552 rule := "AND"
553 if cron.orRule() { rule = "OR" }
554 return []string{fmt.Sprintf("VALID %s %s %d", cron.expanded, rule, len(cron.daily))}, nil
555 case "REMOVE":
556 delete(s.jobs, job.id)
557 return []string{"REMOVE " + job.id}, nil
558 case "RENAME":
559 if err := s.newID(words[2]); err != nil { return nil, err }
560 delete(s.jobs, job.id)
561 job.id = words[2]
562 s.jobs[job.id] = job
563 return []string{"RENAME " + words[1] + " " + words[2]}, nil
564 case "ENABLE", "DISABLE":
565 enabled := command == "ENABLE"
566 if job.enabled == enabled {
567 if enabled { return nil, errors.New("already enabled") }
568 return nil, errors.New("already disabled")
569 }
570 job.enabled = enabled
571 return []string{command + " " + job.id}, nil
572 case "RUNTIME":
573 duration, err := positive(words[2], 4, 1440, "invalid duration")
574 if err != nil { return nil, err }
575 var policy Policy
576 switch words[3] {
577 case "ALLOW": policy = Allow
578 case "FORBID": policy = Forbid
579 case "REPLACE": policy = Replace
580 default: return nil, errors.New("invalid policy")
581 }
582 job.duration, job.policy = duration, policy
583 return []string{fmt.Sprintf("RUNTIME %s %d %s", job.id, duration, policy)}, nil
584 case "LIST":
585 rows := []string{fmt.Sprintf("LIST %d", len(s.jobs))}
586 for _, job := range s.ordered(false) {
587 status := "off"
588 if job.enabled { status = "on" }
589 rows = append(rows, fmt.Sprintf("%s %s %d %s %s", job.id, status, job.duration, job.policy, job.cron.text))
590 }
591 return rows, nil
592 case "EXPAND":
593 index := slices.Index(fieldNames[:], words[2])
594 if index < 0 { return nil, errors.New("invalid field") }
595 field := job.cron.fields[index]
596 parts := []string{fmt.Sprintf("EXPAND %s %s %d", job.id, words[2], len(field.values))}
597 for _, value := range field.values { parts = append(parts, strconv.Itoa(value)) }
598 for _, special := range field.specials { parts = append(parts, special.String()) }
599 return []string{strings.Join(parts, " ")}, nil
600 case "DESCRIBE":
601 return []string{"DESCRIBE " + job.id + " " + job.cron.description()}, nil
602 case "MATCH":
603 time, err := readDatetime(words, 2)
604 if err != nil { return nil, err }
605 date := dateFromNumber(time / 1440)
606 dom, dow, condition := job.cron.matchDay(date)
607 var reasons []string
608 if !job.cron.fields[0].contains(time % 60) { reasons = append(reasons, "minute") }
609 if !job.cron.fields[1].contains(time % 1440 / 60) { reasons = append(reasons, "hour") }
610 if !condition && !dom { reasons = append(reasons, "day-of-month") }
611 if !job.cron.fields[3].contains(date.month) { reasons = append(reasons, "month") }
612 if !condition && !dow { reasons = append(reasons, "day-of-week") }
613 if s.holidays[time/1440] { reasons = append(reasons, "excluded") }
614 result := "yes"
615 if len(reasons) != 0 { result = "no " + strings.Join(reasons, " ") }
616 return []string{fmt.Sprintf("MATCH %s %s %s", job.id, datetimeText(time), result)}, nil
617 case "NEXT", "PREV":
618 base, err := readDatetime(words, 2)
619 if err != nil { return nil, err }
620 limit, err := positive(words[4], 2, 50, "invalid count")
621 if err != nil { return nil, err }
622 reverse := command == "PREV"
623 low, high, sign := base+1, base+527040, "+"
624 if reverse { low, high, sign = base-527040, base-1, "-" }
625 times := job.cron.fires(low, high, s.holidays, reverse, limit)
626 rows := []string{fmt.Sprintf("%s %s %d", command, job.id, len(times))}
627 for _, time := range times {
628 distance := time - base
629 if reverse { distance = -distance }
630 rows = append(rows, datetimeText(time)+" "+sign+span(distance))
631 }
632 if len(times) < limit { rows = append(rows, "NONE") }
633 return rows, nil
634 case "COUNT":
635 low, high, err := readWindow(words, 2)
636 if err != nil { return nil, err }
637 times := job.cron.fires(low, high, s.holidays, false, 0)
638 days, previous := 0, -1
639 for _, time := range times {
640 if time/1440 != previous { days++; previous = time / 1440 }
641 }
642 return []string{fmt.Sprintf("COUNT %s runs %d days %d", job.id, len(times), days)}, nil
643 case "GAP":
644 base, err := readDatetime(words, 2)
645 if err != nil { return nil, err }
646 times := job.cron.fires(base, base+44640, s.holidays, false, 0)
647 if len(times) < 2 { return []string{fmt.Sprintf("GAP %s runs %d none", job.id, len(times))}, nil }
648 histogram := make(map[int]int)
649 shortest, longest, first := 44641, 0, 0
650 for i := 0; i+1 < len(times); i++ {
651 gap := times[i+1] - times[i]
652 histogram[gap]++
653 shortest = min(shortest, gap)
654 if gap > longest { longest, first = gap, i }
655 }
656 rows := []string{fmt.Sprintf("GAP %s runs %d shortest %s longest %s from %s to %s", job.id, len(times), span(shortest), span(longest), datetimeText(times[first]), datetimeText(times[first+1]))}
657 var gaps []int
658 for gap := range histogram { gaps = append(gaps, gap) }
659 slices.Sort(gaps)
660 for _, gap := range gaps { rows = append(rows, fmt.Sprintf("%s %d", span(gap), histogram[gap])) }
661 return rows, nil
662 case "OVERLAP":
663 second, err := s.job(words[2])
664 if err != nil { return nil, err }
665 low, high, err := readWindow(words, 3)
666 if err != nil { return nil, err }
667 a, b := job.cron.fires(low, high, s.holidays, false, 0), second.cron.fires(low, high, s.holidays, false, 0)
668 count, first, last := 0, 0, 0
669 for i, j := 0, 0; i < len(a) && j < len(b); {
670 if a[i] < b[j] { i++ } else if a[i] > b[j] { j++ } else {
671 if count == 0 { first = a[i] }
672 last = a[i]; count++; i++; j++
673 }
674 }
675 row := fmt.Sprintf("OVERLAP %s %s %d", job.id, second.id, count)
676 if count > 0 { row += " first " + datetimeText(first) + " last " + datetimeText(last) }
677 return []string{row}, nil
678 case "TIMELINE", "CONFLICTS", "SIMULATE":
679 low, high, err := readWindow(words, 1)
680 if err != nil { return nil, err }
681 if command == "SIMULATE" { return s.simulate(low, high), nil }
682 return s.timeline(low, high, command == "CONFLICTS"), nil
683 case "WEEK":
684 date, err := readDate(words[1], "invalid date")
685 if err != nil { return nil, err }
686 var counts [7][24]int
687 first := date.number()
688 total := 0
689 for _, job := range s.ordered(true) {
690 for _, time := range job.cron.fires(first*1440, (first+7)*1440-1, s.holidays, false, 0) {
691 counts[time/1440-first][time%1440/60]++
692 total++
693 }
694 }
695 rows := []string{fmt.Sprintf("WEEK %s %d", words[1], total)}
696 for d, hours := range counts {
697 var heat [24]byte
698 dayTotal := 0
699 for hour, count := range hours {
700 dayTotal += count
701 if count == 0 { heat[hour] = '.' } else if count >= 10 { heat[hour] = '+' } else { heat[hour] = byte('0' + count) }
702 }
703 rows = append(rows, fmt.Sprintf("%s %s %d", dateFromNumber(first+d), string(heat[:]), dayTotal))
704 }
705 return rows, nil
706 case "CALENDAR":
707 first, err := readMonth(words[2])
708 if err != nil { return nil, err }
709 var cells []string
710 for range first.weekday() { cells = append(cells, "...") }
711 days := 0
712 for d := 1; d <= monthLength(first.year, first.month); d++ {
713 date := Date{first.year, first.month, d}
714 mark := '-'
715 if job.cron.scheduled(date) {
716 if s.holidays[date.number()] { mark = 'x' } else { mark = '*'; days++ }
717 }
718 cells = append(cells, fmt.Sprintf("%02d%c", d, mark))
719 }
720 for len(cells)%7 != 0 { cells = append(cells, "...") }
721 rows := []string{fmt.Sprintf("CALENDAR %s %s days %d runs %d", job.id, words[2], days, days*len(job.cron.daily)), "Sun Mon Tue Wed Thu Fri Sat"}
722 for i := 0; i < len(cells); i += 7 { rows = append(rows, strings.Join(cells[i:i+7], " ")) }
723 return rows, nil
724 case "EXCLUDE", "INCLUDE":
725 date, err := readDate(words[1], "invalid date")
726 if err != nil { return nil, err }
727 day := date.number()
728 if command == "EXCLUDE" {
729 if s.holidays[day] { return nil, errors.New("already excluded") }
730 s.holidays[day] = true
731 } else {
732 if !s.holidays[day] { return nil, errors.New("not excluded") }
733 delete(s.holidays, day)
734 }
735 return []string{fmt.Sprintf("%s %s holidays %d", command, date, len(s.holidays))}, nil
736 case "HOLIDAYS":
737 var dates []int
738 for day := range s.holidays { dates = append(dates, day) }
739 slices.Sort(dates)
740 rows := []string{fmt.Sprintf("HOLIDAYS %d", len(dates))}
741 for _, day := range dates { rows = append(rows, dateFromNumber(day).String()) }
742 return rows, nil
743 }
744 panic("unreachable command")
745}

747func main() {
