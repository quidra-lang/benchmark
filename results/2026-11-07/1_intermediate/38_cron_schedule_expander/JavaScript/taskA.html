12class CommandError extends Error {}
13const fail = reason => { throw new CommandError(reason); };
14const pad = value => String(value).padStart(2, '0');
15const ascii = (a, b) => a < b ? -1 : a > b ? 1 : 0;
16const leap = year => year % 4 === 0 && (year % 100 !== 0 || year % 400 === 0);
17const monthLength = (year, month) => month === 2 && leap(year) ? 29 : [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31][month - 1];
18const leapCount = year => Math.floor(year / 4) - Math.floor(year / 100) + Math.floor(year / 400);
19const yearStart = year => 365 * (year - 2000) + leapCount(year - 1) - leapCount(1999);
20function dayNumber(year, month, day) {
21 let value = yearStart(year) + day - 1;
22 for (let index = 1; index < month; ++index) value += monthLength(year, index);
23 return value;
24}
25function civil(day) {
26 let year = 2000 + Math.floor(day / 366);
27 while (yearStart(year + 1) <= day) ++year;
28 let remaining = day - yearStart(year), month = 1;
29 while (remaining >= monthLength(year, month)) remaining -= monthLength(year, month++);
30 return { number: day, year, month, day: remaining + 1, weekday: (day + 6) % 7, length: monthLength(year, month) };
31}
32const dateText = date => `${date.year}-${pad(date.month)}-${pad(date.day)}`;
33const timeText = minute => `${pad(Math.floor(minute / 60))}:${pad(minute % 60)}`;
34function dateNotation(day) { const date = civil(day); return `${dateText(date)} ${weekdays[date.weekday].slice(0, 3)}`; }
35function datetimeNotation(minute) {
36 const date = civil(Math.floor(minute / 1440));
37 return `${dateText(date)} ${timeText(minute % 1440)} ${weekdays[date.weekday].slice(0, 3)}`;
38}
39const spanText = minutes => `${Math.floor(minutes / 1440)}d${pad(Math.floor(minutes % 1440 / 60))}h${pad(minutes % 60)}m`;
40function parseDate(text, reason = 'invalid date') {
41 if (!/^\d{4}-\d{2}-\d{2}$/.test(text)) fail(reason);
42 const [year, month, day] = text.split('-').map(Number);
43 if (year < 2001 || year > 2098 || month < 1 || month > 12 || day < 1 || day > monthLength(year, month)) fail(reason);
44 return dayNumber(year, month, day);
45}
46function parseDatetime(date, time) {
47 const day = parseDate(date, 'invalid datetime');
48 if (!/^\d{2}:\d{2}$/.test(time)) fail('invalid datetime');
49 const [hour, minute] = time.split(':').map(Number);
50 if (hour > 23 || minute > 59) fail('invalid datetime');
51 return 1440 * day + 60 * hour + minute;
52}
53function parseMonth(text) {
54 if (!/^\d{4}-\d{2}$/.test(text)) fail('invalid month');
55 const [year, month] = text.split('-').map(Number);
56 if (year < 2001 || year > 2098 || month < 1 || month > 12) fail('invalid month');
57 return { year, month };
58}
59function boundedNumber(text, digits, maximum, reason) {
60 if (!/^[1-9]\d*$/.test(text) || text.length > digits || Number(text) > maximum) fail(reason);
61 return Number(text);
62}
63function natural(text, minimum, maximum, error) {
64 if (!/^\d{1,2}$/.test(text)) error('invalid value');
65 const value = Number(text);
66 if (value < minimum || value > maximum) error('out of range');
67 return value;

69class Field {
70 constructor(text, index) {
71 this.index = index;
72 this.unrestricted = text.startsWith('*');
73 this.set = new Set();
74 this.specials = [];
75 const [minimum, maximum] = limits[index];
76 const error = reason => fail(`${fieldNames[index]} ${reason}`);
77 const value = text => {
78 const names = index === 3 ? months : index === 4 ? weekdays : [];
79 const named = names.findIndex(name => name.slice(0, 3).toUpperCase() === text);
80 return named >= 0 ? named + (index === 3 ? 1 : 0) : natural(text, minimum, maximum, error);
81 };
82 const special = (kind, canonical, number = 0, weekday = 0) => {
83 if (!this.specials.some(item => item.canonical === canonical)) this.specials.push({ kind, canonical, number, weekday });
84 };
85 for (const item of text.split(',')) {
86 if (item === '') error('empty item');
87 if (index === 2 && (item.startsWith('L') || item.endsWith('W'))) {
88 if (item === 'L') special('last', 'L');
89 else if (item === 'LW') special('lastWeekday', 'LW');
90 else if (item.startsWith('L-')) {
91 const offset = natural(item.slice(2), 1, 30, error);
92 special('before', `L-${offset}`, offset);
93 } else if (item.endsWith('W')) {
94 const day = natural(item.slice(0, -1), 1, 31, error);
95 special('nearest', `${day}W`, day);
96 } else error('invalid value');
97 continue;
98 }
99 if (index === 4 && item.includes('#')) {
100 const split = item.indexOf('#'), weekday = value(item.slice(0, split)) % 7, part = item.slice(split + 1);
101 if (!/^\d$/.test(part)) error('invalid value');
102 const number = Number(part);
103 if (number < 1 || number > 5) error('out of range');
104 special('nth', `${weekday}#${number}`, number, weekday);
105 continue;
106 }
107 if (index === 4 && item.length >= 2 && item.endsWith('L')) {
108 const weekday = value(item.slice(0, -1)) % 7;
109 special('lastDow', `${weekday}L`, 0, weekday);
110 continue;
111 }
112 const slash = item.indexOf('/'), base = slash < 0 ? item : item.slice(0, slash);
113 let first, last, step = 1;
114 if (base === '*') { first = minimum; last = maximum; }
115 else if (base.includes('-')) {
116 const dash = base.indexOf('-');
117 first = value(base.slice(0, dash)); last = value(base.slice(dash + 1));
118 if (first > last) error('invalid range');
119 } else { first = value(base); last = slash < 0 ? first : maximum; }
120 if (slash >= 0) {
121 const part = item.slice(slash + 1);
122 if (!/^\d{1,2}$/.test(part) || Number(part) < 1) error('invalid step');
123 step = Number(part);
124 }
125 for (let number = first; number <= last; number += step) this.set.add(index === 4 ? number % 7 : number);
126 }
127 this.values = [...this.set].sort((a, b) => a - b);
128 }
129}
130class Cron {
131 constructor(tokens) {
132 this.text = tokens.join(' ');
133 if (tokens.length === 1 && tokens[0].startsWith('@')) {
134 const expanded = macros.get(tokens[0]);
135 if (expanded === undefined) fail('unknown macro');
136 tokens = expanded.split(' ');
137 }
138 if (tokens.length !== 5) fail('field count');
139 this.expanded = tokens.join(' ');
140 this.fields = tokens.map((text, index) => new Field(text, index));
141 this.or = !this.fields[2].unrestricted && !this.fields[4].unrestricted;
142 this.times = this.fields[1].values.flatMap(hour => this.fields[0].values.map(minute => 60 * hour + minute));
143 }
144 days(date) {
145 const dom = this.fields[2], dow = this.fields[4];
146 const nearest = day => {
147 const weekday = (date.number - date.day + day + 6) % 7;
148 if (weekday === 6) return day === 1 ? 3 : day - 1;
149 if (weekday === 0) return day === date.length ? day - 2 : day + 1;
150 return day;
151 };
152 const monthDay = dom.set.has(date.day) || dom.specials.some(item => {
153 switch (item.kind) {
154 case 'last': return date.day === date.length;
155 case 'lastWeekday': return date.day === nearest(date.length);
156 case 'before': return date.day === date.length - item.number;
157 case 'nearest': return item.number <= date.length && date.day === nearest(item.number);
158 }
159 });
160 const weekDay = dow.set.has(date.weekday) || dow.specials.some(item => item.weekday === date.weekday &&
161 (item.kind === 'nth' ? Math.floor((date.day - 1) / 7) + 1 === item.number : date.day + 7 > date.length));
162 return { monthDay, weekDay, matches: this.or ? monthDay || weekDay : monthDay && weekDay };
163 }
164 scheduled(date) { return this.fields[3].set.has(date.month) && this.days(date).matches; }
165 reasons(minute, holidays) {
166 const date = civil(Math.floor(minute / 1440)), days = this.days(date), reasons = [];
167 if (!this.fields[0].set.has(minute % 60)) reasons.push('minute');
168 if (!this.fields[1].set.has(Math.floor(minute % 1440 / 60))) reasons.push('hour');
169 if (!days.matches && !days.monthDay) reasons.push('day-of-month');
170 if (!this.fields[3].set.has(date.month)) reasons.push('month');
171 if (!days.matches && !days.weekDay) reasons.push('day-of-week');
172 if (holidays.has(date.number)) reasons.push('excluded');
173 return reasons;
174 }
175 *fires(low, high, holidays, reverse = false) {
176 const first = Math.floor(low / 1440), last = Math.floor(high / 1440), step = reverse ? -1 : 1;
177 const times = reverse ? this.times.slice().reverse() : this.times;
178 for (let day = reverse ? last : first; reverse ? day >= first : day <= last; day += step) {
179 if (holidays.has(day) || !this.scheduled(civil(day))) continue;
180 for (const time of times) {
181 const minute = 1440 * day + time;
182 if (low <= minute && minute <= high) yield minute;
183 }
184 }
185 }
186}

188function list(parts, conjunction = 'and') {
189 return parts.length === 1 ? parts[0] : parts.slice(0, -1).join(', ') + ` ${conjunction} ` + parts.at(-1);
190}
191function valueList(values, index, conjunction = 'and') {
192 const label = value => index === 3 ? months[value - 1] : index === 4 ? weekdays[value] : String(value);
193 const parts = [];
194 for (let first = 0; first < values.length;) {
195 let end = first + 1;
196 while (end < values.length && values[end] === values[end - 1] + 1) ++end;
197 if (end - first >= 3) parts.push(`${label(values[first])} through ${label(values[end - 1])}`);
198 else for (let position = first; position < end; ++position) parts.push(label(values[position]));
199 first = end;
200 }
201 return list(parts, conjunction);
202}
203function stepped(values, maximum) {
204 if (values.length < 2 || values[0] !== 0 || values[1] < 2) return null;
205 const step = values[1];
206 return values.length === Math.floor(maximum / step) + 1 && values.every((value, index) => value === index * step) ? step : null;
207}
208function description(cron) {
209 const [minutes, hours, dom, month, dow] = cron.fields;
210 let time;
211 if (cron.times.length <= 4) time = 'At ' + list(cron.times.map(timeText));
212 else {
213 const minuteStep = stepped(minutes.values, 59), hourStep = stepped(hours.values, 23);
214 if (minutes.values.length === 60) time = 'Every minute';
215 else if (minuteStep !== null) time = `Every ${minuteStep} minutes`;
216 else if (minutes.values.length === 1) time = `At minute ${minutes.values[0]}`;
217 else time = 'At minutes ' + valueList(minutes.values, 0);
218 if (hours.values.length === 24) { if (time.startsWith('At minute')) time += ' of every hour'; }
219 else if (hourStep !== null) time += `, every ${hourStep} hours`;
220 else if (hours.values.at(-1) - hours.values[0] + 1 === hours.values.length) time += `, between ${pad(hours.values[0])}:00 and ${pad(hours.values.at(-1))}:59`;
221 else time += ', during hours ' + valueList(hours.values, 1);
222 }
223 const domParts = [];
224 if (dom.values.length) domParts.push(dom.values.length === 1 ? `day ${dom.values[0]}` : 'days ' + valueList(dom.values, 2));
225 for (const item of dom.specials) {
226 switch (item.kind) {
227 case 'last': domParts.push('the last day'); break;
228 case 'lastWeekday': domParts.push('the last weekday'); break;
229 case 'before': domParts.push(`${item.number} ${item.number === 1 ? 'day' : 'days'} before the last day`); break;
230 case 'nearest': domParts.push(`the weekday nearest day ${item.number}`); break;
231 }
232 }
233 const domClause = list(domParts) + ' of the month';
234 const dowClause = (conjunction = 'and') => {
235 const parts = dow.values.length ? [valueList(dow.values, 4, conjunction)] : [];
236 for (const item of dow.specials) {
237 const ordinal = ['1st', '2nd', '3rd', '4th', '5th'][item.number - 1];
238 parts.push(item.kind === 'nth' ? `the ${ordinal} ${weekdays[item.weekday]}` : `the last ${weekdays[item.weekday]}`);
239 }
240 return list(parts, conjunction) + (dow.specials.length ? ' of the month' : '');
241 };
242 const domFull = dom.values.length === 31, dowFull = dow.values.length === 7;
243 let day = '';
244 if (cron.or) {
245 if (!domFull && !dowFull) day = `, on ${domClause} or on ${dowClause()}`;
246 } else if (!domFull && !dowFull) day = `, on ${domClause} if it is ${dowClause('or')}`;
247 else if (!domFull) day = `, on ${domClause}`;
248 else if (!dowFull) day = `, on ${dowClause()}`;
249 return time + day + (month.values.length === 12 ? '' : ', in ' + valueList(month.values, 3));
250}

252 constructor(id, cron) { this.id = id; this.cron = cron; this.enabled = true; this.duration = 1; this.policy = 'ALLOW'; }

254const signatures = new Map([
255 ['REMOVE', ['job']], ['RENAME', ['job', 'newId']], ['ENABLE', ['job']], ['DISABLE', ['job']],

262class Scheduler {
263 constructor() { this.jobs = new Map(); this.holidays = new Set(); }
264 newId(id) {
265 if (!/^[a-z][a-z0-9_]{0,15}$/.test(id)) fail('invalid id');
266 if (this.jobs.has(id)) fail('duplicate job');
267 return id;
268 }
269 ordered(enabled = false) { return [...this.jobs.values()].filter(job => !enabled || job.enabled).sort((a, b) => ascii(a.id, b.id)); }
270 parseArguments(types, words) {
271 let position = 0;
272 const arguments_ = [], datetimes = [];
273 for (const type of types) {
274 const text = words[position++];
275 let value;
276 switch (type) {
277 case 'job': value = this.jobs.get(text); if (value === undefined) fail('unknown job'); break;
278 case 'newId': value = this.newId(text); break;
279 case 'datetime': value = parseDatetime(text, words[position++]); datetimes.push(value); break;
280 case 'date': value = parseDate(text); break;
281 case 'month': value = parseMonth(text); break;
282 case 'count': value = boundedNumber(text, 2, 50, 'invalid count'); break;
283 case 'duration': value = boundedNumber(text, 4, 1440, 'invalid duration'); break;
284 case 'policy':
285 if (!['ALLOW', 'FORBID', 'REPLACE'].includes(text)) fail('invalid policy');
286 value = text; break;
287 case 'field':
288 value = fieldNames.indexOf(text);
289 if (value < 0) fail('invalid field');
290 break;
291 }
292 arguments_.push(value);
293 }
294 if (datetimes.length === 2) {
295 if (datetimes[0] > datetimes[1]) fail('invalid window');
296 if (datetimes[1] - datetimes[0] > 44640) fail('window too long');
297 }
298 return arguments_;
299 }
300 fires(job, low, high, reverse = false) { return job.cron.fires(low, high, this.holidays, reverse); }
301 grouped(low, high) {
302 const groups = new Map();
303 for (const job of this.ordered(true)) for (const minute of this.fires(job, low, high)) {
304 if (!groups.has(minute)) groups.set(minute, []);
305 groups.get(minute).push(job.id);
306 }
307 return [...groups].sort((a, b) => a[0] - b[0]);
308 }
309 simulate(low, high) {
310 const jobs = this.ordered(true), events = new Map(), rows = [];
311 const event = (minute, difference) => events.set(minute, (events.get(minute) ?? 0) + difference);
312 for (const job of jobs) {
313 let lastEnd = null, started = 0, skipped = 0, replaced = 0, busy = 0;
314 for (const minute of this.fires(job, low, high)) {
315 if (lastEnd !== null && lastEnd > minute) {
316 if (job.policy === 'FORBID') { ++skipped; continue; }
317 if (job.policy === 'REPLACE') {
318 event(lastEnd, 1); event(minute, -1);
319 busy -= lastEnd - minute;
320 ++replaced;
321 }
322 }
323 lastEnd = minute + job.duration;
324 event(minute, 1); event(lastEnd, -1);
325 busy += job.duration;
326 ++started;
327 }
328 rows.push(`${job.id} ${job.policy} started ${started} skipped ${skipped} replaced ${replaced} busy ${busy}`);
329 }
330 let running = 0, peak = 0, earliest = null;
331 for (const [minute, difference] of [...events].sort((a, b) => a[0] - b[0])) {
332 running += difference;
333 if (running > peak) { peak = running; earliest = minute; }
334 }
335 return [`SIMULATE ${jobs.length} peak ${peak}${peak ? ` at ${datetimeNotation(earliest)}` : ''}`, ...rows];
336 }
337 week(first, text) {
338 const jobs = this.ordered(true), rows = [];
339 let total = 0;
340 for (let day = first; day < first + 7; ++day) {
341 const hours = Array(24).fill(0), date = civil(day);
342 if (!this.holidays.has(day)) for (const job of jobs) if (job.cron.scheduled(date)) {
343 for (const hour of job.cron.fields[1].values) hours[hour] += job.cron.fields[0].values.length;
344 }
345 const count = hours.reduce((a, b) => a + b, 0);
346 total += count;
347 rows.push(`${dateNotation(day)} ${hours.map(value => value === 0 ? '.' : value < 10 ? String(value) : '+').join('')} ${count}`);
348 }
349 return [`WEEK ${text} ${total}`, ...rows];
350 }
351 calendar(job, date, text) {
352 const first = dayNumber(date.year, date.month, 1), length = monthLength(date.year, date.month);
353 const cells = Array((first + 6) % 7).fill('...');
354 let active = 0;
355 for (let day = 1; day <= length; ++day) {
356 const number = first + day - 1;
357 const mark = !job.cron.scheduled(civil(number)) ? '-' : this.holidays.has(number) ? 'x' : '*';
358 if (mark === '*') ++active;
359 cells.push(`${pad(day)}${mark}`);
360 }
361 while (cells.length % 7) cells.push('...');
362 const rows = [`CALENDAR ${job.id} ${text} days ${active} runs ${active * job.cron.times.length}`, 'Sun Mon Tue Wed Thu Fri Sat'];
363 for (let index = 0; index < cells.length; index += 7) rows.push(cells.slice(index, index + 7).join(' '));
364 return rows;
365 }
366 gap(job, base) {
367 const times = [...this.fires(job, base, base + 44640)];
368 if (times.length < 2) return [`GAP ${job.id} runs ${times.length} none`];
369 const counts = new Map();
370 let shortest = times[1] - times[0], longest = shortest, from = 0;
371 for (let index = 1; index < times.length; ++index) {
372 const gap = times[index] - times[index - 1];
373 counts.set(gap, (counts.get(gap) ?? 0) + 1);
374 shortest = Math.min(shortest, gap);
375 if (gap > longest) { longest = gap; from = index - 1; }
376 }
377 return [`GAP ${job.id} runs ${times.length} shortest ${spanText(shortest)} longest ${spanText(longest)} from ${datetimeNotation(times[from])} to ${datetimeNotation(times[from + 1])}`,
378 ...[...counts].sort((a, b) => a[0] - b[0]).map(([gap, count]) => `${spanText(gap)} ${count}`)];
379 }
380 execute(line) {
381 const [command, ...words] = line.trim().split(/ +/);
382 if (command !== 'JOB' && command !== 'VALIDATE' && !signatures.has(command)) fail('unknown command');
383 if (command === 'JOB' || command === 'VALIDATE') {
384 if (words.length < (command === 'JOB' ? 2 : 1)) fail('wrong arguments');
385 if (command === 'VALIDATE') {
386 const cron = new Cron(words);
387 return [`VALID ${cron.expanded} ${cron.or ? 'OR' : 'AND'} ${cron.times.length}`];
388 }
389 const id = this.newId(words[0]), cron = new Cron(words.slice(1));
390 this.jobs.set(id, new Job(id, cron));
391 return [`JOB ${id} added`];
392 }
393 const signature = signatures.get(command), count = signature.reduce((size, type) => size + (type === 'datetime' ? 2 : 1), 0);
394 if (words.length !== count) fail('wrong arguments');
395 const args = this.parseArguments(signature, words);
396 switch (command) {
397 case 'REMOVE': {
398 const job = args[0]; this.jobs.delete(job.id);
399 return [`REMOVE ${job.id}`];
400 }
401 case 'RENAME': {
402 const [job, id] = args, previous = job.id;
403 this.jobs.delete(previous); job.id = id; this.jobs.set(id, job);
404 return [`RENAME ${previous} ${id}`];
405 }
406 case 'ENABLE': case 'DISABLE': {
407 const job = args[0], enabled = command === 'ENABLE';
408 if (job.enabled === enabled) fail(enabled ? 'already enabled' : 'already disabled');
409 job.enabled = enabled;
410 return [`${command} ${job.id}`];
411 }
412 case 'RUNTIME': {
413 const [job, duration, policy] = args;
414 job.duration = duration; job.policy = policy;
415 return [`RUNTIME ${job.id} ${duration} ${policy}`];
416 }
417 case 'LIST': return [`LIST ${this.jobs.size}`, ...this.ordered().map(job => `${job.id} ${job.enabled ? 'on' : 'off'} ${job.duration} ${job.policy} ${job.cron.text}`)];
418 case 'EXPAND': {
419 const [job, index] = args, field = job.cron.fields[index];
420 return [[`EXPAND ${job.id} ${fieldNames[index]} ${field.values.length}`, ...field.values, ...field.specials.map(item => item.canonical)].join(' ')];
421 }
422 case 'DESCRIBE': return [`DESCRIBE ${args[0].id} ${description(args[0].cron)}`];
423 case 'MATCH': {
424 const [job, minute] = args, reasons = job.cron.reasons(minute, this.holidays);
425 return [`MATCH ${job.id} ${datetimeNotation(minute)} ${reasons.length ? 'no ' + reasons.join(' ') : 'yes'}`];
426 }
427 case 'NEXT': case 'PREV': {
428 const [job, base, count] = args, reverse = command === 'PREV', times = [];
429 const low = reverse ? base - 527040 : base + 1, high = reverse ? base - 1 : base + 527040;
430 for (const minute of this.fires(job, low, high, reverse)) {
431 times.push(minute);
432 if (times.length === count) break;
433 }
434 const rows = [`${command} ${job.id} ${times.length}`, ...times.map(minute => `${datetimeNotation(minute)} ${reverse ? '-' : '+'}${spanText(Math.abs(minute - base))}`)];
435 if (times.length < count) rows.push('NONE');
436 return rows;
437 }
438 case 'COUNT': {
439 const [job, low, high] = args, days = new Set();
440 let runs = 0;
441 for (const minute of this.fires(job, low, high)) { ++runs; days.add(Math.floor(minute / 1440)); }
442 return [`COUNT ${job.id} runs ${runs} days ${days.size}`];
443 }
444 case 'GAP': return this.gap(...args);
445 case 'OVERLAP': {
446 const [first, second, low, high] = args, other = new Set(this.fires(second, low, high));
447 const shared = [...this.fires(first, low, high)].filter(minute => other.has(minute));
448 const detail = shared.length ? ` first ${datetimeNotation(shared[0])} last ${datetimeNotation(shared.at(-1))}` : '';
449 return [`OVERLAP ${first.id} ${second.id} ${shared.length}${detail}`];
450 }
451 case 'TIMELINE': {
452 const groups = this.grouped(...args), count = groups.reduce((value, [, ids]) => value + ids.length, 0), rows = [`TIMELINE ${count}`];
453 let shown = 0;
454 for (const [minute, ids] of groups) {
455 if (shown === 100) break;
456 for (const id of ids) {
457 rows.push(`${datetimeNotation(minute)} ${id}`);
458 if (++shown === 100) break;
459 }
460 }
461 if (count > 100) rows.push(`... ${count - 100} more`);
462 return rows;
463 }
464 case 'CONFLICTS': {
465 const groups = this.grouped(...args).filter(([, ids]) => ids.length >= 2), peak = groups.reduce((value, [, ids]) => Math.max(value, ids.length), 0);
466 const rows = [`CONFLICTS ${groups.length} peak ${peak}`, ...groups.slice(0, 100).map(([minute, ids]) => `${datetimeNotation(minute)} ${ids.length} ${ids.join(' ')}`)];
467 if (groups.length > 100) rows.push(`... ${groups.length - 100} more`);
468 return rows;
469 }
470 case 'WEEK': return this.week(args[0], words[0]);
471 case 'CALENDAR': return this.calendar(args[0], args[1], words[1]);
472 case 'SIMULATE': return this.simulate(...args);
473 case 'EXCLUDE': case 'INCLUDE': {
474 const day = args[0], exclude = command === 'EXCLUDE';
475 if (this.holidays.has(day) === exclude) fail(exclude ? 'already excluded' : 'not excluded');
476 if (exclude) this.holidays.add(day); else this.holidays.delete(day);
477 return [`${command} ${dateNotation(day)} holidays ${this.holidays.size}`];
478 }
479 case 'HOLIDAYS': return [`HOLIDAYS ${this.holidays.size}`, ...[...this.holidays].sort((a, b) => a - b).map(dateNotation)];
480 }
481 }
482}

485for (let index = 1; index <= Number(input[0]); ++index) {
486 try { process.stdout.write(scheduler.execute(input[index]).join('\n') + '\n'); }
487 catch (error) {
488 if (!(error instanceof CommandError)) throw error;
