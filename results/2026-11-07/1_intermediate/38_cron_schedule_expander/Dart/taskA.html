12String two(int value) => value.toString().padLeft(2, '0');
13bool leap(int year) => year % 4 == 0 && (year % 100 != 0 || year % 400 == 0);
14int daysInMonth(int year, int month) => month == 2 && leap(year) ? 29 : monthDays[month - 1];
15bool digits(String text, int minimum, int maximum) => text.length >= minimum && text.length <= maximum && text.codeUnits.every((c) => c >= 48 && c <= 57);

17final class CivilDate {
18 final int year, month, day;
19 const CivilDate(this.year, this.month, this.day);
20 int get number {
21 var result = day - 1;
22 for (var y = 2000; y < year; y++) result += leap(y) ? 366 : 365;
23 for (var m = 1; m < month; m++) result += daysInMonth(year, m);
24 return result;
25 }
26 int get weekday => (number + 6) % 7;
27 String get dateText => '${year.toString().padLeft(4, '0')}-${two(month)}-${two(day)}';
28 @override
29 String toString() => '$dateText ${weekdays[weekday].substring(0, 3)}';
30 static CivilDate fromNumber(int number) {
31 var year = 2000, month = 1;
32 while (number >= (leap(year) ? 366 : 365)) { number -= leap(year) ? 366 : 365; year++; }
33 while (number >= daysInMonth(year, month)) { number -= daysInMonth(year, month); month++; }
34 return CivilDate(year, month, number + 1);
35 }
36}

38CivilDate? tryDate(String text) {
39 if (text.length != 10 || text[4] != '-' || text[7] != '-' || !digits(text.substring(0, 4), 4, 4) || !digits(text.substring(5, 7), 2, 2) || !digits(text.substring(8), 2, 2)) return null;
40 final year = int.parse(text.substring(0, 4)), month = int.parse(text.substring(5, 7)), day = int.parse(text.substring(8));
41 if (year < 2001 || year > 2098 || month < 1 || month > 12 || day < 1 || day > daysInMonth(year, month)) return null;
42 return CivilDate(year, month, day);

44CivilDate readDate(String text) => tryDate(text) ?? (throw const CommandError('invalid date'));
45int readMinute(String dateText, String time) {
46 final date = tryDate(dateText);

48 final hour = int.parse(time.substring(0, 2)), minute = int.parse(time.substring(3));
49 if (hour > 23 || minute > 59) throw const CommandError('invalid datetime');

68enum SpecialKind { last, lastWeekday, beforeLast, nearestWeekday, nthWeekday, lastOccurrence }
69final class Special {
70 final SpecialKind kind;
71 final int value, occurrence;
72 const Special(this.kind, [this.value = 0, this.occurrence = 0]);
73 @override
74 bool operator ==(Object other) => other is Special && kind == other.kind && value == other.value && occurrence == other.occurrence;
75 @override
76 int get hashCode => Object.hash(kind, value, occurrence);
77 @override
78 String toString() => switch (kind) {
79 SpecialKind.last => 'L', SpecialKind.lastWeekday => 'LW', SpecialKind.beforeLast => 'L-$value',
80 SpecialKind.nearestWeekday => '${value}W', SpecialKind.nthWeekday => '$value#$occurrence', SpecialKind.lastOccurrence => '${value}L',
81 };
82}

93final class Cron {
94 static const fieldNames = ['minute', 'hour', 'day-of-month', 'month', 'day-of-week'];
95 static const minimum = [0, 0, 1, 1, 0], maximum = [59, 23, 31, 12, 7];
96 static const monthNames = ['JAN', 'FEB', 'MAR', 'APR', 'MAY', 'JUN', 'JUL', 'AUG', 'SEP', 'OCT', 'NOV', 'DEC'];
97 static const dayNames = ['SUN', 'MON', 'TUE', 'WED', 'THU', 'FRI', 'SAT'];
98 static const macros = {
99 '@yearly': '0 0 1 1 *', '@annually': '0 0 1 1 *', '@monthly': '0 0 1 * *', '@weekly': '0 0 * * 0',
100 '@daily': '0 0 * * *', '@midnight': '0 0 * * *', '@hourly': '0 * * * *',
101 };
102 final List<CronField> fields;
103 final String expanded;
104 final List<int> daily;
105 Cron._(List<String> words, List<CronField> fields)
106 : fields = List.unmodifiable(fields), expanded = words.join(' '),
107 daily = [for (final hour in fields[1].values) for (final minute in fields[0].values) hour * 60 + minute];
108 bool get orRule => fields[2].restricted && fields[4].restricted;
109 String get rule => orRule ? 'OR' : 'AND';
110 static CommandError fieldError(int index, String reason) => CommandError('${fieldNames[index]} $reason');
111 static int readValue(String text, int index) {
112 final named = index == 3 ? monthNames.indexOf(text) : index == 4 ? dayNames.indexOf(text) : -1;
113 if (named >= 0) return named + (index == 3 ? 1 : 0);
114 if (!digits(text, 1, 2)) throw fieldError(index, 'invalid value');
115 final value = int.parse(text);
116 if (value < minimum[index] || value > maximum[index]) throw fieldError(index, 'out of range');
117 return value;
118 }
119 static int specialNumber(String text, int index, int maximum, [int size = 2]) {
120 if (!digits(text, 1, size)) throw fieldError(index, 'invalid value');
121 final value = int.parse(text);
122 if (value < 1 || value > maximum) throw fieldError(index, 'out of range');
123 return value;
124 }
125 static CronField readField(String text, int index) {
126 final values = SplayTreeSet<int>(), specials = <Special>[];
127 for (final item in text.split(',')) {
128 if (item.isEmpty) throw fieldError(index, 'empty item');
129 Special? special;
130 final hash = item.indexOf('#');
131 if (index == 2 && (item.startsWith('L') || item.endsWith('W'))) {
132 if (item == 'L') special = const Special(SpecialKind.last);
133 else if (item == 'LW') special = const Special(SpecialKind.lastWeekday);
134 else if (item.startsWith('L-')) special = Special(SpecialKind.beforeLast, specialNumber(item.substring(2), index, 30));
135 else if (item.endsWith('W')) special = Special(SpecialKind.nearestWeekday, specialNumber(item.substring(0, item.length - 1), index, 31));
136 else throw fieldError(index, 'invalid value');
137 } else if (index == 4 && hash >= 0) {
138 final weekday = readValue(item.substring(0, hash), index);
139 final occurrence = specialNumber(item.substring(hash + 1), index, 5, 1);
140 special = Special(SpecialKind.nthWeekday, weekday % 7, occurrence);
141 } else if (index == 4 && item.length >= 2 && item.endsWith('L')) {
142 special = Special(SpecialKind.lastOccurrence, readValue(item.substring(0, item.length - 1), index) % 7);
143 }
144 if (special != null) {
145 if (!specials.contains(special)) specials.add(special);
146 continue;
147 }
148 final slash = item.indexOf('/'), basis = item.split('/').first;
149 int low, high;
150 if (basis == '*') { low = minimum[index]; high = maximum[index]; }
151 else {
152 final dash = basis.indexOf('-');
153 if (dash >= 0) {
154 low = readValue(basis.substring(0, dash), index);
155 high = readValue(basis.substring(dash + 1), index);
156 if (low > high) throw fieldError(index, 'invalid range');
157 } else { low = readValue(basis, index); high = slash < 0 ? low : maximum[index]; }
158 }
159 var step = 1;
160 if (slash >= 0) {
161 final part = item.substring(slash + 1);
162 if (!digits(part, 1, 2) || int.parse(part) == 0) throw fieldError(index, 'invalid step');
163 step = int.parse(part);
164 }
165 for (var value = low; value <= high; value += step) values.add(index == 4 ? value % 7 : value);
166 }
167 return CronField(index, values, specials, !text.startsWith('*'));
168 }
169 factory Cron.parse(List<String> words) {
170 if (words.length == 1 && words.first.startsWith('@')) {
171 final replacement = macros[words.first];
172 if (replacement == null) throw const CommandError('unknown macro');
173 words = replacement.split(' ');
174 }
175 if (words.length != 5) throw const CommandError('field count');
176 final fields = <CronField>[];
177 for (var index = 0; index < 5; index++) fields.add(readField(words[index], index));
178 return Cron._(words, fields);
179 }
180 static int nearest(CivilDate date, int day) {
181 final weekday = (date.number - date.day + day + 6) % 7;
182 if (weekday == 6) return day == 1 ? 3 : day - 1;
183 if (weekday == 0) return day == daysInMonth(date.year, date.month) ? day - 2 : day + 1;
184 return day;
185 }
186 ({bool monthDay, bool weekDay, bool condition}) matchDay(CivilDate date) {
187 final last = daysInMonth(date.year, date.month), day = date.day, weekday = date.weekday;
188 var monthDay = fields[2].contains(day), weekDay = fields[4].contains(weekday);
189 for (final special in fields[2].specials) {
190 final matches = switch (special.kind) {
191 SpecialKind.last => day == last, SpecialKind.lastWeekday => day == nearest(date, last),
192 SpecialKind.beforeLast => day == last - special.value,
193 _ => special.value <= last && day == nearest(date, special.value),
194 };
195 monthDay = monthDay || matches;
196 }
197 for (final special in fields[4].specials) {
198 final matches = weekday == special.value && (special.kind == SpecialKind.nthWeekday ? (day - 1) ~/ 7 + 1 == special.occurrence : day + 7 > last);
199 weekDay = weekDay || matches;
200 }
201 return (monthDay: monthDay, weekDay: weekDay, condition: orRule ? monthDay || weekDay : monthDay && weekDay);
202 }
203 bool scheduled(CivilDate date) => fields[3].contains(date.month) && matchDay(date).condition;
204 Iterable<int> fireTimes(int low, int high, Set<int> holidays, {bool reverse = false}) sync* {
205 final first = low ~/ 1440, last = high ~/ 1440;
206 for (var day = reverse ? last : first; reverse ? day >= first : day <= last; day += reverse ? -1 : 1) {
207 if (holidays.contains(day) || !scheduled(CivilDate.fromNumber(day))) continue;
208 for (var i = reverse ? daily.length - 1 : 0; reverse ? i >= 0 : i < daily.length; i += reverse ? -1 : 1) {
209 final time = day * 1440 + daily[i];
210 if (low <= time && time <= high) yield time;
211 }
212 }
213 }
214 List<String> reasons(int minute, Set<int> holidays) {
215 final date = CivilDate.fromNumber(minute ~/ 1440), result = <String>[];
216 final matches = matchDay(date);
217 if (!fields[0].contains(minute % 60)) result.add('minute');
218 if (!fields[1].contains(minute % 1440 ~/ 60)) result.add('hour');
219 if (!matches.condition && !matches.monthDay) result.add('day-of-month');
220 if (!fields[3].contains(date.month)) result.add('month');
221 if (!matches.condition && !matches.weekDay) result.add('day-of-week');
222 if (holidays.contains(date.number)) result.add('excluded');
223 return result;
224 }
225 static String listJoin(Iterable<String> values, {bool either = false}) {
226 final parts = values.toList();
227 if (parts.length <= 1) return parts.isEmpty ? '' : parts.first;
228 return '${parts.take(parts.length - 1).join(', ')}${either ? ' or ' : ' and '}${parts.last}';
229 }
230 static String valueName(int index, int value) => index == 3 ? months[value - 1] : index == 4 ? weekdays[value] : '$value';
231 static String valueList(CronField field, {bool either = false}) {
232 final parts = <String>[];
233 for (var start = 0; start < field.values.length;) {
234 var end = start;
235 while (end + 1 < field.values.length && field.values[end + 1] == field.values[end] + 1) end++;
236 if (end - start >= 2) parts.add('${valueName(field.index, field.values[start])} through ${valueName(field.index, field.values[end])}');
237 else for (var i = start; i <= end; i++) parts.add(valueName(field.index, field.values[i]));
238 start = end + 1;
239 }
240 return listJoin(parts, either: either);
241 }
242 static int step(CronField field) {
243 final values = field.values;
244 if (values.length < 2 || values[0] != 0 || values[1] < 2) return 0;
245 final step = values[1];
246 if (values.length != maximum[field.index] ~/ step + 1) return 0;
247 for (var i = 0; i < values.length; i++) if (values[i] != i * step) return 0;
248 return step;
249 }
250 String dayClause(bool weekday, {bool either = false}) {
251 final field = fields[weekday ? 4 : 2], parts = <String>[];
252 if (field.values.isNotEmpty) parts.add(weekday ? valueList(field, either: either) : field.values.length == 1 ? 'day ${field.values[0]}' : 'days ${valueList(field)}');
253 const ordinals = ['1st', '2nd', '3rd', '4th', '5th'];
254 for (final special in field.specials) {
255 parts.add(switch (special.kind) {
256 SpecialKind.last => 'the last day', SpecialKind.lastWeekday => 'the last weekday',
257 SpecialKind.beforeLast => special.value == 1 ? '1 day before the last day' : '${special.value} days before the last day',
258 SpecialKind.nearestWeekday => 'the weekday nearest day ${special.value}',
259 SpecialKind.nthWeekday => 'the ${ordinals[special.occurrence - 1]} ${weekdays[special.value]}',
260 SpecialKind.lastOccurrence => 'the last ${weekdays[special.value]}',
261 });
262 }
263 return listJoin(parts, either: either) + (!weekday || field.specials.isNotEmpty ? ' of the month' : '');
264 }
265 String describe() {
266 final minute = fields[0], hour = fields[1];
267 String text;
268 if (daily.length <= 4) text = 'At ${listJoin(daily.map((t) => '${two(t ~/ 60)}:${two(t % 60)}'))}';
269 else {
270 final minuteStep = step(minute), hourStep = step(hour);
271 text = minute.values.length == 60 ? 'Every minute' : minuteStep != 0 ? 'Every $minuteStep minutes'
272 : minute.values.length == 1 ? 'At minute ${minute.values[0]}' : 'At minutes ${valueList(minute)}';
273 if (hour.values.length == 24) {
274 if (text.startsWith('At minute')) text += ' of every hour';
275 } else if (hourStep != 0) text += ', every $hourStep hours';
276 else if (hour.values.last - hour.values.first + 1 == hour.values.length) text += ', between ${two(hour.values.first)}:00 and ${two(hour.values.last)}:59';
277 else text += ', during hours ${valueList(hour)}';
278 }
279 final monthFull = fields[2].values.length == 31, weekFull = fields[4].values.length == 7;
280 if (orRule) {
281 if (!monthFull && !weekFull) text += ', on ${dayClause(false)} or on ${dayClause(true)}';
282 } else if (!monthFull && !weekFull) text += ', on ${dayClause(false)} if it is ${dayClause(true, either: true)}';
283 else if (!monthFull) text += ', on ${dayClause(false)}';
284 else if (!weekFull) text += ', on ${dayClause(true)}';
285 if (fields[3].values.length != 12) text += ', in ${valueList(fields[3])}';
286 return text;
287 }

308final class Scheduler {
309 final jobs = <String, Job>{};
310 final holidays = <int>{};
311 static const arity = {
312 'JOB': -2, 'VALIDATE': -1, 'REMOVE': 1, 'RENAME': 2, 'ENABLE': 1, 'DISABLE': 1, 'RUNTIME': 3, 'LIST': 0,
313 'EXPAND': 2, 'DESCRIBE': 1, 'MATCH': 3, 'NEXT': 4, 'PREV': 4, 'COUNT': 5, 'GAP': 3, 'OVERLAP': 6,
314 'TIMELINE': 4, 'CONFLICTS': 4, 'WEEK': 1, 'CALENDAR': 2, 'SIMULATE': 4, 'EXCLUDE': 1, 'INCLUDE': 1, 'HOLIDAYS': 0,
315 };
316 Job getJob(String id) => jobs[id] ?? (throw const CommandError('unknown job'));
317 void newId(String id) {
318 if (id.isEmpty || id.length > 16 || id.codeUnitAt(0) < 97 || id.codeUnitAt(0) > 122 || id.codeUnits.any((c) => !(c >= 97 && c <= 122 || c >= 48 && c <= 57 || c == 95))) throw const CommandError('invalid id');
319 if (jobs.containsKey(id)) throw const CommandError('duplicate job');
320 }
321 static int positive(String text, int size, int maximum, String reason) {
322 if (!digits(text, 1, size) || text[0] == '0' || int.parse(text) > maximum) throw CommandError(reason);
323 return int.parse(text);
324 }
325 List<Job> ordered({bool enabledOnly = false}) => jobs.values.where((job) => !enabledOnly || job.enabled).toList()..sort((a, b) => a.id.compareTo(b.id));
326 SplayTreeMap<int, List<String>> schedule(int low, int high) {
327 final result = SplayTreeMap<int, List<String>>();
328 for (final job in ordered(enabledOnly: true)) {
329 for (final time in job.cron.fireTimes(low, high, holidays)) result.putIfAbsent(time, () => []).add(job.id);
330 }
331 return result;
332 }
333 List<String> timeline(int low, int high, bool conflicts) {
334 final times = schedule(low, high), rows = <String>[];
335 if (conflicts) {
336 final entries = times.entries.where((entry) => entry.value.length >= 2).toList();
337 var peak = 0;
338 for (final entry in entries) if (entry.value.length > peak) peak = entry.value.length;
339 rows.add('CONFLICTS ${entries.length} peak $peak');
340 for (final entry in entries.take(100)) rows.add('${minuteText(entry.key)} ${entry.value.length} ${entry.value.join(' ')}');
341 if (entries.length > 100) rows.add('... ${entries.length - 100} more');
342 } else {
343 final count = times.values.fold(0, (sum, ids) => sum + ids.length);
344 var written = 0;
345 rows.add('TIMELINE $count');
346 for (final entry in times.entries) {
347 for (final id in entry.value) {
348 if (written == 100) break;
349 rows.add('${minuteText(entry.key)} $id');
350 written++;
351 }
352 }
353 if (count > 100) rows.add('... ${count - 100} more');
354 }
355 return rows;
356 }
357 List<String> week(String input, CivilDate start) {
358 final counts = List.generate(7, (_) => List.filled(24, 0));
359 final low = start.number * 1440;
360 var total = 0;
361 for (final job in ordered(enabledOnly: true)) {
362 for (final time in job.cron.fireTimes(low, low + 7 * 1440 - 1, holidays)) { counts[(time - low) ~/ 1440][time % 1440 ~/ 60]++; total++; }
363 }
364 final rows = ['WEEK $input $total'];
365 for (var day = 0; day < 7; day++) {
366 final marks = counts[day].map((count) => count == 0 ? '.' : count >= 10 ? '+' : '$count').join();
367 rows.add('${CivilDate.fromNumber(start.number + day)} $marks ${counts[day].fold(0, (sum, count) => sum + count)}');
368 }
369 return rows;
370 }
371 List<String> month(Job job, String input, int year, int month) {
372 final first = CivilDate(year, month, 1);
373 final cells = List.filled(first.weekday, '...', growable: true);
374 var count = 0;
375 for (var day = 1; day <= daysInMonth(year, month); day++) {
376 final date = CivilDate(year, month, day);
377 final mark = job.cron.scheduled(date) ? holidays.contains(date.number) ? 'x' : '*' : '-';
378 if (mark == '*') count++;
379 cells.add('${two(day)}$mark');
380 }
381 while (cells.length % 7 != 0) cells.add('...');
382 final rows = ['CALENDAR ${job.id} $input days $count runs ${count * job.cron.daily.length}', 'Sun Mon Tue Wed Thu Fri Sat'];
383 for (var i = 0; i < cells.length; i += 7) rows.add(cells.skip(i).take(7).join(' '));
384 return rows;
385 }
386 List<String> simulate(int low, int high) {
387 final enabled = ordered(enabledOnly: true), events = SplayTreeMap<int, int>(), details = <String>[];
388 for (final job in enabled) {
389 final executions = <Execution>[];
390 var skipped = 0, replaced = 0;
391 for (final time in job.cron.fireTimes(low, high, holidays)) {
392 final previous = executions.isEmpty ? null : executions.last;
393 if (previous != null && previous.end > time) {
394 if (job.policy == Policy.forbid) { skipped++; continue; }
395 if (job.policy == Policy.replace) { previous.end = time; replaced++; }
396 }
397 executions.add(Execution(time, time + job.duration));
398 }
399 var busy = 0;
400 for (final execution in executions) {
401 busy += execution.end - execution.start;
402 events[execution.start] = (events[execution.start] ?? 0) + 1;
403 events[execution.end] = (events[execution.end] ?? 0) - 1;
404 }
405 details.add('${job.id} ${job.policy.text} started ${executions.length} skipped $skipped replaced $replaced busy $busy');
406 }
407 var running = 0, peak = 0, at = 0;
408 for (final entry in events.entries) {
409 running += entry.value;
410 if (running > peak) { peak = running; at = entry.key; }
411 }
412 return ['SIMULATE ${enabled.length} peak $peak${peak == 0 ? '' : ' at ${minuteText(at)}'}', ...details];
413 }
414 List<String> execute(List<String> words) {
415 final command = words[0], expected = arity[words[0]];
416 if (expected == null) throw const CommandError('unknown command');
417 final count = words.length - 1;
418 if (expected < 0 ? count < -expected : count != expected) throw const CommandError('wrong arguments');
419 switch (command) {
420 case 'JOB':
421 final id = words[1]; newId(id);
422 final expression = words.sublist(2);
423 final cron = Cron.parse(expression);
424 jobs[id] = Job(id, cron, expression.join(' '));
425 return ['JOB $id added'];
426 case 'VALIDATE':
427 final cron = Cron.parse(words.sublist(1));
428 return ['VALID ${cron.expanded} ${cron.rule} ${cron.daily.length}'];
429 case 'REMOVE':
430 final job = getJob(words[1]); jobs.remove(job.id);
431 return ['REMOVE ${job.id}'];
432 case 'RENAME':
433 final job = getJob(words[1]); newId(words[2]);
434 jobs.remove(job.id); job.id = words[2]; jobs[job.id] = job;
435 return ['RENAME ${words[1]} ${job.id}'];
436 case 'ENABLE': case 'DISABLE':
437 final job = getJob(words[1]), enable = command == 'ENABLE';
438 if (job.enabled == enable) throw CommandError(enable ? 'already enabled' : 'already disabled');
439 job.enabled = enable;
440 return ['$command ${job.id}'];
441 case 'RUNTIME':
442 final job = getJob(words[1]);
443 final duration = positive(words[2], 4, 1440, 'invalid duration');
444 final policy = switch (words[3]) { 'ALLOW' => Policy.allow, 'FORBID' => Policy.forbid, 'REPLACE' => Policy.replace, _ => throw const CommandError('invalid policy') };
445 job.duration = duration; job.policy = policy;
446 return ['RUNTIME ${job.id} $duration ${policy.text}'];
447 case 'LIST':
448 return ['LIST ${jobs.length}', for (final job in ordered()) '${job.id} ${job.enabled ? 'on' : 'off'} ${job.duration} ${job.policy.text} ${job.text}'];
449 case 'EXPAND':
450 final job = getJob(words[1]), index = Cron.fieldNames.indexOf(words[2]);
451 if (index < 0) throw const CommandError('invalid field');
452 final field = job.cron.fields[index];
453 final items = [...field.values.map((v) => '$v'), ...field.specials.map((item) => '$item')];
454 return ['EXPAND ${job.id} ${words[2]} ${field.values.length}${items.isEmpty ? '' : ' ${items.join(' ')}'}'];
455 case 'DESCRIBE':
456 final job = getJob(words[1]); return ['DESCRIBE ${job.id} ${job.cron.describe()}'];
457 case 'MATCH':
458 final job = getJob(words[1]);
459 final time = readMinute(words[2], words[3]);
460 final reasons = job.cron.reasons(time, holidays);
461 return ['MATCH ${job.id} ${minuteText(time)} ${reasons.isEmpty ? 'yes' : 'no ${reasons.join(' ')}'}'];
462 case 'NEXT': case 'PREV':
463 final job = getJob(words[1]);
464 final start = readMinute(words[2], words[3]);
465 final limit = positive(words[4], 2, 50, 'invalid count'), reverse = command == 'PREV';
466 final times = job.cron.fireTimes(reverse ? start - 527040 : start + 1, reverse ? start - 1 : start + 527040, holidays, reverse: reverse).take(limit).toList();
467 return ['$command ${job.id} ${times.length}', for (final time in times) '${minuteText(time)} ${reverse ? '-' : '+'}${span((time - start).abs())}', if (times.length < limit) 'NONE'];
468 case 'COUNT':
469 final job = getJob(words[1]);
470 final low = readMinute(words[2], words[3]);
471 final high = readMinute(words[4], words[5]); checkWindow(low, high);
472 final times = job.cron.fireTimes(low, high, holidays).toList();
473 return ['COUNT ${job.id} runs ${times.length} days ${times.map((time) => time ~/ 1440).toSet().length}'];
474 case 'GAP':
475 final job = getJob(words[1]);
476 final start = readMinute(words[2], words[3]);
477 final times = job.cron.fireTimes(start, start + 44640, holidays).toList();
478 if (times.length < 2) return ['GAP ${job.id} runs ${times.length} none'];
479 final frequencies = SplayTreeMap<int, int>();
480 var shortest = 44641, longest = 0, from = 0;
481 for (var i = 1; i < times.length; i++) {
482 final gap = times[i] - times[i - 1];
483 if (gap < shortest) shortest = gap;
484 if (gap > longest) { longest = gap; from = i - 1; }
485 frequencies[gap] = (frequencies[gap] ?? 0) + 1;
486 }
487 return ['GAP ${job.id} runs ${times.length} shortest ${span(shortest)} longest ${span(longest)} from ${minuteText(times[from])} to ${minuteText(times[from + 1])}',
488 for (final entry in frequencies.entries) '${span(entry.key)} ${entry.value}'];
489 case 'OVERLAP':
490 final a = getJob(words[1]); final b = getJob(words[2]);
491 final low = readMinute(words[3], words[4]);
492 final high = readMinute(words[5], words[6]); checkWindow(low, high);
493 final other = b.cron.fireTimes(low, high, holidays).toSet();
494 final times = a.cron.fireTimes(low, high, holidays).where(other.contains).toList();
495 return ['OVERLAP ${a.id} ${b.id} ${times.length}${times.isEmpty ? '' : ' first ${minuteText(times.first)} last ${minuteText(times.last)}'}'];
496 case 'TIMELINE': case 'CONFLICTS': case 'SIMULATE':
497 final low = readMinute(words[1], words[2]);
498 final high = readMinute(words[3], words[4]); checkWindow(low, high);
499 return command == 'SIMULATE' ? simulate(low, high) : timeline(low, high, command == 'CONFLICTS');
500 case 'WEEK':
501 final date = readDate(words[1]); return week(words[1], date);
502 case 'CALENDAR':
503 final job = getJob(words[1]);
504 final (year, monthValue) = readMonth(words[2]);
505 return month(job, words[2], year, monthValue);
506 case 'EXCLUDE': case 'INCLUDE':
507 final date = readDate(words[1]);
508 final day = date.number;
509 if (command == 'EXCLUDE') {
510 if (!holidays.add(day)) throw const CommandError('already excluded');
511 } else if (!holidays.remove(day)) throw const CommandError('not excluded');
512 return ['$command $date holidays ${holidays.length}'];
513 case 'HOLIDAYS':
514 return ['HOLIDAYS ${holidays.length}', for (final day in holidays.toList()..sort()) '${CivilDate.fromNumber(day)}'];
515 }
516 throw StateError('Unknown command');
517 }
518}
