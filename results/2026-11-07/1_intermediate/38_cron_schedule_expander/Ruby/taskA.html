13class CommandError < StandardError; end

15def fail_command(reason) = raise(CommandError, reason)
16def digits?(text, minimum, maximum) = text.length.between?(minimum, maximum) && text.each_byte.all? { |c| c.between?(48, 57) }
17def leap?(year) = year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
18def month_length(year, month) = month == 2 && leap?(year) ? 29 : MONTH_DAYS[month - 1]

20CivilDate = Data.define(:year, :month, :day) do
21 def number
22 result = day - 1
23 (2000...year).each { |y| result += leap?(y) ? 366 : 365 }
24 (1...month).each { |m| result += month_length(year, m) }
25 result
26 end
27
28 def weekday = (number + 6) % 7
29 def to_s = format('%04d-%02d-%02d %s', year, month, day, WEEKDAY_NAMES[weekday][0, 3])

31 def self.from_number(number)
32 year = 2000
33 month = 1
34 while number >= (leap?(year) ? 366 : 365)
35 number -= leap?(year) ? 366 : 365
36 year += 1
37 end
38 while number >= month_length(year, month)
39 number -= month_length(year, month)
40 month += 1
41 end
42 new(year, month, number + 1)
43 end
44end

46def read_date(text, reason = 'invalid date')
47 fail_command(reason) unless text.match?(/\A[0-9]{4}-[0-9]{2}-[0-9]{2}\z/)
48 date = CivilDate.new(text[0, 4].to_i, text[5, 2].to_i, text[8, 2].to_i)
49 unless date.year.between?(2001, 2098) && date.month.between?(1, 12) && date.day.between?(1, month_length(date.year, date.month))
50 fail_command(reason)
51 end
52 date
53end

55def read_datetime(words, at)
56 date = read_date(words[at], 'invalid datetime')
57 time = words[at + 1]
58 fail_command('invalid datetime') unless time.match?(/\A[0-9]{2}:[0-9]{2}\z/)
59 hour = time[0, 2].to_i
60 minute = time[3, 2].to_i
61 fail_command('invalid datetime') if hour > 23 || minute > 59
62 date.number * 1440 + hour * 60 + minute
63end

65def read_month(text)
66 fail_command('invalid month') unless text.match?(/\A[0-9]{4}-[0-9]{2}\z/)
67 date = CivilDate.new(text[0, 4].to_i, text[5, 2].to_i, 1)
68 fail_command('invalid month') unless date.year.between?(2001, 2098) && date.month.between?(1, 12)
69 date
70end

94Special = Data.define(:kind, :value, :occurrence) do
95 def to_s
96 case kind
97 when :last then 'L'
98 when :last_workday then 'LW'
99 when :before_last then "L-#{value}"
100 when :nearest then "#{value}W"
101 when :nth_weekday then "#{value}##{occurrence}"
102 when :last_weekday then "#{value}L"
103 end
104 end

106 def phrase
107 case kind
108 when :last then 'the last day'
109 when :last_workday then 'the last weekday'
110 when :before_last then "#{value} #{value == 1 ? 'day' : 'days'} before the last day"
111 when :nearest then "the weekday nearest day #{value}"
112 when :nth_weekday then "the #{%w[1st 2nd 3rd 4th 5th][occurrence - 1]} #{WEEKDAY_NAMES[value]}"
113 when :last_weekday then "the last #{WEEKDAY_NAMES[value]}"
114 end
115 end
116end

118def read_value(text, index)
119 names = index == 3 ? MONTH_CODES : index == 4 ? WEEKDAY_CODES : []
120 named = names.index(text)
121 return named + (index == 3 ? 1 : 0) if named
122 fail_command('invalid value') unless digits?(text, 1, 2)
123 value = text.to_i
124 fail_command('out of range') unless value.between?(MINIMUM[index], MAXIMUM[index])
125 value
126end

128def special_number(text, maximum, width = 2)
129 fail_command('invalid value') unless digits?(text, 1, width)
130 value = text.to_i
131 fail_command('out of range') unless value.between?(1, maximum)
132 value
133end

135def word_list(parts, either = false)
136 return parts.join if parts.length < 2
137 parts[0...-1].join(', ') + (either ? ' or ' : ' and ') + parts[-1]

140CronField = Data.define(:index, :values, :specials, :restricted) do
141 def include?(value) = values.include?(value)
142
143 def stepped
144 return 0 if values.length < 2 || values[0] != 0 || values[1] < 2
145 step = values[1]
146 return 0 unless values.length == MAXIMUM[index] / step + 1 && values.each_with_index.all? { |value, i| value == i * step }
147 step
148 end
149
150 def value_name(value)
151 case index
152 when 3 then MONTH_NAMES[value - 1]
153 when 4 then WEEKDAY_NAMES[value]
154 else value.to_s
155 end
156 end
157
158 def value_list(either = false)
159 parts = []
160 first = 0
161 while first < values.length
162 last = first
163 last += 1 while last + 1 < values.length && values[last + 1] == values[last] + 1
164 if last - first >= 2
165 parts << "#{value_name(values[first])} through #{value_name(values[last])}"
166 else
167 (first..last).each { |i| parts << value_name(values[i]) }
168 end
169 first = last + 1
170 end
171 word_list(parts, either)
172 end
173
174 def day_phrase(either = false)
175 parts = []
176 unless values.empty?
177 parts << (index == 4 ? value_list(either) : "#{values.length == 1 ? 'day' : 'days'} #{value_list}")
178 end
179 parts.concat(specials.map(&:phrase))
180 word_list(parts, either) + (index == 2 || !specials.empty? ? ' of the month' : '')
181 end

183 def self.parse(text, index)
184 values = Set.new
185 specials = []
186 text.split(',', -1).each do |item|
187 fail_command('empty item') if item.empty?
188 special = if index == 2 && (item.start_with?('L') || item.end_with?('W'))
189 if item == 'L'
190 Special.new(:last, 0, 0)
191 elsif item == 'LW'
192 Special.new(:last_workday, 0, 0)
193 elsif item.start_with?('L-')
194 Special.new(:before_last, special_number(item[2..], 30), 0)
195 elsif item.end_with?('W')
196 Special.new(:nearest, special_number(item[0...-1], 31), 0)
197 else
198 fail_command('invalid value')
199 end
200 elsif index == 4 && item.include?('#')
201 at = item.index('#')
202 weekday = read_value(item[0...at], index)
203 occurrence = special_number(item[(at + 1)..], 5, 1)
204 Special.new(:nth_weekday, weekday % 7, occurrence)
205 elsif index == 4 && item.length >= 2 && item.end_with?('L')
206 Special.new(:last_weekday, read_value(item[0...-1], index) % 7, 0)
207 end
208 if special
209 specials << special unless specials.include?(special)
210 next
211 end
212 slash = item.index('/')
213 base = slash ? item[0...slash] : item
214 if base == '*'
215 first = MINIMUM[index]
216 last = MAXIMUM[index]
217 elsif (dash = base.index('-'))
218 first = read_value(base[0...dash], index)
219 last = read_value(base[(dash + 1)..], index)
220 fail_command('invalid range') if first > last
221 else
222 first = read_value(base, index)
223 last = slash ? MAXIMUM[index] : first
224 end
225 step = 1
226 if slash
227 part = item[(slash + 1)..]
228 fail_command('invalid step') unless digits?(part, 1, 2) && part.to_i >= 1
229 step = part.to_i
230 end
231 first.step(last, step) { |value| values.add(index == 4 ? value % 7 : value) }
232 end
233 new(index, values.to_a.sort.freeze, specials.freeze, !text.start_with?('*'))
234 rescue CommandError => error
235 fail_command("#{FIELD_NAMES[index]} #{error.message}")
236 end
237end

239DayMatch = Data.define(:dom, :dow, :condition)
240Cron = Data.define(:fields, :daily, :text, :expanded) do
241 def or_rule? = fields[2].restricted && fields[4].restricted
242
243 def match_day(date)
244 last = month_length(date.year, date.month)
245 weekday = date.weekday
246 nearest = lambda do |day|
247 case (date.number - date.day + day + 6) % 7
248 when 6 then day == 1 ? 3 : day - 1
249 when 0 then day == last ? day - 2 : day + 1
250 else day
251 end
252 end
253 dom = fields[2].include?(date.day) || fields[2].specials.any? { |special|
254 case special.kind
255 when :last then date.day == last
256 when :last_workday then date.day == nearest.call(last)
257 when :before_last then date.day == last - special.value
258 when :nearest then special.value <= last && date.day == nearest.call(special.value)
259 end
260 }
261 dow = fields[4].include?(weekday) || fields[4].specials.any? { |special|
262 weekday == special.value && (special.kind == :nth_weekday ? (date.day - 1) / 7 + 1 == special.occurrence : date.day + 7 > last)
263 }
264 DayMatch.new(dom, dow, or_rule? ? dom || dow : dom && dow)
265 end
266
267 def scheduled?(date) = fields[3].include?(date.month) && match_day(date).condition
268
269 def fires(low, high, holidays, reverse: false)
270 return enum_for(__method__, low, high, holidays, reverse: reverse) unless block_given?
271 first = low / 1440
272 last = high / 1440
273 days = reverse ? last.downto(first) : first.upto(last)
274 times = reverse ? daily.reverse : daily
275 days.each do |day|
276 next if holidays.include?(day) || !scheduled?(CivilDate.from_number(day))
277 times.each do |offset|
278 time = day * 1440 + offset
279 yield time if time.between?(low, high)
280 end
281 end
282 end

284 def description
285 minute, hour, dom, month, dow = fields
286 text = if daily.length <= 4
287 'At ' + word_list(daily.map { |time| format('%02d:%02d', time / 60, time % 60) })
288 else
289 ms = minute.stepped
290 hs = hour.stepped
291 time = if minute.values.length == 60
292 'Every minute'
293 elsif ms > 0
294 "Every #{ms} minutes"
295 elsif minute.values.length == 1
296 "At minute #{minute.values[0]}"
297 else
298 "At minutes #{minute.value_list}"
299 end
300 if hour.values.length == 24
301 time += ' of every hour' if time.start_with?('At minute')
302 elsif hs > 0
303 time += ", every #{hs} hours"
304 elsif hour.values[-1] - hour.values[0] + 1 == hour.values.length
305 time += format(', between %02d:00 and %02d:59', hour.values[0], hour.values[-1])
306 else
307 time += ", during hours #{hour.value_list}"
308 end
309 time
310 end
311 d_full = dom.values.length == 31
312 w_full = dow.values.length == 7
313 if or_rule?
314 text += ", on #{dom.day_phrase} or on #{dow.day_phrase}" unless d_full || w_full
315 elsif !d_full && !w_full
316 text += ", on #{dom.day_phrase} if it is #{dow.day_phrase(true)}"
317 elsif !d_full
318 text += ", on #{dom.day_phrase}"
319 elsif !w_full
320 text += ", on #{dow.day_phrase}"
321 end
322 text += ", in #{month.value_list}" unless month.values.length == 12
323 text
324 end

326 def self.parse(input)
327 text = input.join(' ')
328 words = input
329 if input.length == 1 && input[0].start_with?('@')
330 expanded = case input[0]
331 when '@yearly', '@annually' then '0 0 1 1 *'
332 when '@monthly' then '0 0 1 * *'
333 when '@weekly' then '0 0 * * 0'
334 when '@daily', '@midnight' then '0 0 * * *'
335 when '@hourly' then '0 * * * *'
336 else fail_command('unknown macro')
337 end
338 words = expanded.split(' ')
339 end
340 fail_command('field count') unless words.length == 5
341 fields = words.each_with_index.map { |word, index| CronField.parse(word, index) }
342 daily = fields[1].values.flat_map { |hour| fields[0].values.map { |minute| hour * 60 + minute } }
343 new(fields.freeze, daily.freeze, text.freeze, words.join(' ').freeze)
344 end
345end

349def bit_count(value)
350 count = 0
351 until value == 0
352 value &= value - 1
353 count += 1
354 end
355 count
356end

358class Scheduler
359 def initialize
360 @jobs = {}
361 @holidays = Set.new
362 end
363
364 def job(id) = @jobs.fetch(id) { fail_command('unknown job') }
365
366 def new_id(id)
367 fail_command('invalid id') unless id.match?(/\A[a-z][a-z0-9_]{0,15}\z/)
368 fail_command('duplicate job') if @jobs.key?(id)
369 end
370
371 def enabled = @jobs.sort.select { |_, job| job.enabled }
372
373 def timeline(window, conflicts)
374 enabled = self.enabled
375 masks = Array.new(window.high - window.low + 1, 0)
376 enabled.each_with_index do |(_, job), i|
377 job.cron.fires(window.low, window.high, @holidays).each { |time| masks[time - window.low] |= 1 << i }
378 end
379 counts = masks.map { |mask| bit_count(mask) }
380 total = conflicts ? counts.count { |count| count >= 2 } : counts.sum
381 peak = counts.max >= 2 ? counts.max : 0
382 rows = [conflicts ? "CONFLICTS #{total} peak #{peak}" : "TIMELINE #{total}"]
383 masks.each_with_index do |mask, i|
384 break if rows.length == 101
385 if conflicts
386 next if counts[i] < 2
387 ids = enabled.each_with_index.filter_map { |(id, _), index| id if mask & (1 << index) != 0 }
388 rows << "#{datetime_text(window.low + i)} #{ids.length} #{ids.join(' ')}"
389 else
390 enabled.each_with_index do |(id, _), index|
391 break if rows.length == 101
392 rows << "#{datetime_text(window.low + i)} #{id}" if mask & (1 << index) != 0
393 end
394 end
395 end
396 rows << "... #{total - 100} more" if total > 100
397 rows
398 end

400 def simulate(window)
401 enabled = self.enabled
402 events = Array.new(window.high - window.low + 1441, 0)
403 rows = enabled.map do |id, job|
404 started = skipped = replaced = busy = 0
405 last_end = nil
406 job.cron.fires(window.low, window.high, @holidays).each do |time|
407 if last_end && last_end > time && job.policy == :FORBID
408 skipped += 1
409 next
410 end
411 if last_end && last_end > time && job.policy == :REPLACE
412 events[last_end - window.low] += 1
413 events[time - window.low] -= 1
414 busy -= last_end - time
415 replaced += 1
416 end
417 last_end = time + job.duration
418 events[time - window.low] += 1
419 events[last_end - window.low] -= 1
420 started += 1
421 busy += job.duration
422 end
423 "#{id} #{job.policy} started #{started} skipped #{skipped} replaced #{replaced} busy #{busy}"
424 end
425 running = peak = at = 0
426 events.each_with_index do |change, i|
427 running += change
428 if running > peak
429 peak = running
430 at = window.low + i
431 end
432 end
433 rows.unshift("SIMULATE #{enabled.length} peak #{peak}" + (peak == 0 ? '' : " at #{datetime_text(at)}"))
434 end

436 def execute(words)
437 command = words[0]
438 expected = case command
439 when 'JOB' then -3
440 when 'VALIDATE' then -2
441 when 'LIST', 'HOLIDAYS' then 1
442 when 'REMOVE', 'ENABLE', 'DISABLE', 'DESCRIBE', 'WEEK', 'EXCLUDE', 'INCLUDE' then 2
443 when 'RENAME', 'EXPAND', 'CALENDAR' then 3
444 when 'RUNTIME', 'MATCH', 'GAP' then 4
445 when 'NEXT', 'PREV', 'TIMELINE', 'CONFLICTS', 'SIMULATE' then 5
446 when 'COUNT' then 6
447 when 'OVERLAP' then 7
448 else fail_command('unknown command')
449 end
450 wrong_count = expected < 0 ? words.length < -expected : words.length != expected
451 fail_command('wrong arguments') if wrong_count
452 case command
453 when 'JOB'
454 new_id(words[1])
455 cron = Cron.parse(words[2..])
456 @jobs[words[1]] = Job.new(cron, true, 1, :ALLOW)
457 ["JOB #{words[1]} added"]
458 when 'VALIDATE'
459 cron = Cron.parse(words[1..])
460 ["VALID #{cron.expanded} #{cron.or_rule? ? 'OR' : 'AND'} #{cron.daily.length}"]
461 when 'REMOVE'
462 job(words[1])
463 @jobs.delete(words[1])
464 ["REMOVE #{words[1]}"]
465 when 'RENAME'
466 old = job(words[1])
467 new_id(words[2])
468 @jobs.delete(words[1])
469 @jobs[words[2]] = old
470 ["RENAME #{words[1]} #{words[2]}"]
471 when 'ENABLE', 'DISABLE'
472 current = job(words[1])
473 desired = command == 'ENABLE'
474 fail_command(desired ? 'already enabled' : 'already disabled') if current.enabled == desired
475 current.enabled = desired
476 ["#{command} #{words[1]}"]
477 when 'RUNTIME'
478 current = job(words[1])
479 duration = positive(words[2], 4, 1440, 'invalid duration')
480 fail_command('invalid policy') unless %w[ALLOW FORBID REPLACE].include?(words[3])
481 current.duration = duration
482 current.policy = words[3].to_sym
483 ["RUNTIME #{words[1]} #{duration} #{current.policy}"]
484 when 'LIST'
485 ["LIST #{@jobs.length}"] + @jobs.sort.map { |id, current| "#{id} #{current.enabled ? 'on' : 'off'} #{current.duration} #{current.policy} #{current.cron.text}" }
486 when 'EXPAND'
487 current = job(words[1])
488 index = FIELD_NAMES.index(words[2])
489 fail_command('invalid field') unless index
490 field = current.cron.fields[index]
491 [["EXPAND #{words[1]} #{words[2]} #{field.values.length}", *field.values, *field.specials].join(' ')]
492 when 'DESCRIBE'
493 current = job(words[1])
494 ["DESCRIBE #{words[1]} #{current.cron.description}"]
495 when 'MATCH'
496 current = job(words[1])
497 time = read_datetime(words, 2)
498 date = CivilDate.from_number(time / 1440)
499 match = current.cron.match_day(date)
500 reasons = []
501 reasons << 'minute' unless current.cron.fields[0].include?(time % 60)
502 reasons << 'hour' unless current.cron.fields[1].include?(time % 1440 / 60)
503 reasons << 'day-of-month' if !match.condition && !match.dom
504 reasons << 'month' unless current.cron.fields[3].include?(date.month)
505 reasons << 'day-of-week' if !match.condition && !match.dow
506 reasons << 'excluded' if @holidays.include?(time / 1440)
507 ["MATCH #{words[1]} #{datetime_text(time)} " + (reasons.empty? ? 'yes' : "no #{reasons.join(' ')}")]
508 when 'NEXT', 'PREV'
509 current = job(words[1])
510 base = read_datetime(words, 2)
511 limit = positive(words[4], 2, 50, 'invalid count')
512 reverse = command == 'PREV'
513 times = current.cron.fires(reverse ? base - 527_040 : base + 1, reverse ? base - 1 : base + 527_040, @holidays, reverse: reverse).take(limit)
514 rows = ["#{command} #{words[1]} #{times.length}"] + times.map { |time| "#{datetime_text(time)} #{reverse ? '-' : '+'}#{span((time - base).abs)}" }
515 rows << 'NONE' if times.length < limit
516 rows
517 when 'COUNT'
518 current = job(words[1])
519 window = read_window(words, 2)
520 times = current.cron.fires(window.low, window.high, @holidays).to_a
521 ["COUNT #{words[1]} runs #{times.length} days #{times.map { |time| time / 1440 }.uniq.length}"]
522 when 'GAP'
523 current = job(words[1])
524 base = read_datetime(words, 2)
525 times = current.cron.fires(base, base + 44_640, @holidays).to_a
526 return ["GAP #{words[1]} runs #{times.length} none"] if times.length < 2
527 histogram = Hash.new(0)
528 longest = first = 0
529 times.each_cons(2).with_index do |(a, b), i|
530 gap = b - a
531 histogram[gap] += 1
532 if gap > longest
533 longest = gap
534 first = i
535 end
536 end
537 ["GAP #{words[1]} runs #{times.length} shortest #{span(histogram.keys.min)} longest #{span(longest)} from #{datetime_text(times[first])} to #{datetime_text(times[first + 1])}"] + histogram.sort.map { |gap, count| "#{span(gap)} #{count}" }
538 when 'OVERLAP'
539 a = job(words[1])
540 b = job(words[2])
541 window = read_window(words, 3)
542 other = b.cron.fires(window.low, window.high, @holidays).to_set
543 times = a.cron.fires(window.low, window.high, @holidays).select { |time| other.include?(time) }
544 ["OVERLAP #{words[1]} #{words[2]} #{times.length}" + (times.empty? ? '' : " first #{datetime_text(times[0])} last #{datetime_text(times[-1])}")]
545 when 'TIMELINE', 'CONFLICTS', 'SIMULATE'
546 window = read_window(words, 1)
547 command == 'SIMULATE' ? simulate(window) : timeline(window, command == 'CONFLICTS')
548 when 'WEEK'
549 first = read_date(words[1]).number
550 counts = Array.new(7) { Array.new(24, 0) }
551 enabled.each do |_, current|
552 current.cron.fires(first * 1440, (first + 7) * 1440 - 1, @holidays).each { |time| counts[time / 1440 - first][time % 1440 / 60] += 1 }
553 end
554 rows = counts.each_with_index.map do |hours, day|
555 heat = hours.map { |count| count == 0 ? '.' : count >= 10 ? '+' : count.to_s }.join
556 "#{CivilDate.from_number(first + day)} #{heat} #{hours.sum}"
557 end
558 ["WEEK #{words[1]} #{counts.sum(&:sum)}"] + rows
559 when 'CALENDAR'
560 current = job(words[1])
561 first = read_month(words[2])
562 cells = Array.new(first.weekday, '...')
563 days = 0
564 1.upto(month_length(first.year, first.month)) do |day|
565 date = first.with(day: day)
566 mark = !current.cron.scheduled?(date) ? '-' : @holidays.include?(date.number) ? 'x' : '*'
567 days += 1 if mark == '*'
568 cells << format('%02d%s', day, mark)
569 end
570 cells << '...' until cells.length % 7 == 0
571 ["CALENDAR #{words[1]} #{words[2]} days #{days} runs #{days * current.cron.daily.length}", 'Sun Mon Tue Wed Thu Fri Sat'] + cells.each_slice(7).map { |week| week.join(' ') }
572 when 'EXCLUDE', 'INCLUDE'
573 date = read_date(words[1])
574 if command == 'EXCLUDE'
575 fail_command('already excluded') unless @holidays.add?(date.number)
576 else
577 fail_command('not excluded') unless @holidays.delete?(date.number)
578 end
