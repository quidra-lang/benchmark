DayStatus day_status(const Scheduler scheduler, const Cron cron, int day)
    date := scheduler.calendar[day]
    dom := cron.fields[2]
    dow := cron.fields[4]
    matches_dom := position_of(dom.values, date.day) >= 0
    matches_dow := position_of(dow.values, date.weekday) >= 0
    first := day - date.day + 1
    for special := dom.specials[]
        target := 0
        match special.kind
            SpecialKind.LastDay
                target = date.length
            SpecialKind.LastWorkday
                target = nearest_workday(first, date.length, date.length)
            SpecialKind.OffsetDay
                target = date.length - special.value
            SpecialKind.NearestWorkday
                if special.value <= date.length
                    target = nearest_workday(first, special.value, date.length)
            SpecialKind.NthWeekday
                continue
            SpecialKind.LastWeekday
                continue
        matches_dom = matches_dom or target == date.day
    for special := dow.specials[]
        if special.value == date.weekday
            matches_dow = matches_dow or (if special.kind == SpecialKind.NthWeekday then (date.day - 1) / 7 + 1 == special.nth else date.day + 7 > date.length)
    condition := if cron.disjunction then matches_dom or matches_dow else matches_dom and matches_dow
    return DayStatus(position_of(cron.fields[3].values, date.month) >= 0, matches_dom, matches_dow, condition)

bool scheduled_day(const Scheduler scheduler, const Cron cron, int day)
    status := day_status(scheduler, cron, day)
    return status.month and status.condition
