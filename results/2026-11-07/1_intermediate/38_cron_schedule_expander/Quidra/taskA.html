220Field | error parse_field(int kind, string text)
221 result := Field(kind, text[0] != "*")
222 distinct<int> values
223 for item := text.split(",")[]
224 length := int(#item)
225 if length == 0
226 return field_error(kind, "empty item")
227 if kind == 2 and (item[0] == "L" or item[length - 1] == "W")
228 Special special
229 if item == "L"
230 special = Special(SpecialKind.LastDay)
231 elif item == "LW"
232 special = Special(SpecialKind.LastWorkday)
233 elif item.starts_with("L-")
234 value := try special_number(kind, item[2:], 2, 30)
235 special = Special(SpecialKind.OffsetDay, value)
236 elif item[length - 1] == "W"
237 value := try special_number(kind, item[0:<length - 1], 2, 31)
238 special = Special(SpecialKind.NearestWorkday, value)
239 else
240 return field_error(kind, "invalid value")
241 add_special(&result, special)
242 elif kind == 4 and item.contains("#")
243 hash := find(item, "#")
244 value := try field_value(kind, item[0:<hash])
245 nth := try special_number(kind, item[hash + 1:], 1, 5)
246 add_special(&result, Special(SpecialKind.NthWeekday, value % 7, nth))
247 elif kind == 4 and length >= 2 and item[length - 1] == "L"
248 value := try field_value(kind, item[0:<length - 1])
249 add_special(&result, Special(SpecialKind.LastWeekday, value % 7))
250 else
251 slash := find(item, "/")
252 base := if slash < 0 then item else item[0:<slash]
253 int first
254 int last
255 if base == "*"
256 first = field_low(kind)
257 last = field_high(kind)
258 elif base.contains("-")
259 dash := find(base, "-")
260 first = try field_value(kind, base[0:<dash])
261 last = try field_value(kind, base[dash + 1:])
262 if first > last
263 return field_error(kind, "invalid range")
264 else
265 first = try field_value(kind, base)
266 last = if slash < 0 then first else field_high(kind)
267 step := 1
268 if slash >= 0
269 part := item[slash + 1:]
270 if not digits(part, 1, 2)
271 return field_error(kind, "invalid step")
272 step = int(part)
273 if step == 0
274 return field_error(kind, "invalid step")
275 for: first <= value <= last step step
276 .values.add(if kind == 4 then value % 7 else value)
277 result.values = values.sorted()
278 return result

653string[] | error execute(Scheduler &scheduler, string line)
654 words := command_words(line)
655 command := words[0]
656 commands := "JOB VALIDATE REMOVE RENAME ENABLE DISABLE RUNTIME LIST EXPAND DESCRIBE MATCH NEXT PREV COUNT GAP OVERLAP TIMELINE CONFLICTS WEEK CALENDAR SIMULATE EXCLUDE INCLUDE HOLIDAYS".split(" ")
657 arities := [-2, -1, 1, 2, 1, 1, 3, 0, 2, 1, 3, 4, 4, 5, 3, 6, 4, 4, 1, 2, 4, 1, 1, 0]
658 code := position_of(commands, command)
659 if code < 0
660 return error("unknown command")
661 count := int(#words) - 1
662 required := arities[code]
663 if (required < 0 and count < -required) or (required >= 0 and count != required)
664 return error("wrong arguments")
665 if command == "JOB"
666 try new_id(scheduler, words[1])
667 cron := try parse_cron(words[2:])
668 .scheduler.jobs.set(words[1], Job(words[1], cron))
669 return ["JOB {words[1]} added"]
670 if command == "VALIDATE"
671 cron := try parse_cron(words[1:])
672 rule := if cron.disjunction then "OR" else "AND"
673 return ["VALID {cron.expanded} {rule} {#cron.times}"]
674 if command == "LIST"
675 rows := ["LIST {scheduler.jobs.size()}"]
