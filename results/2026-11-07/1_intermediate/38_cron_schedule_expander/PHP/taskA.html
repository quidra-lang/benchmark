21function isDigits(string $text, int $minimum, int $maximum): bool
22{
23 $length = strlen($text);
24 return $length >= $minimum && $length <= $maximum && strspn($text, "0123456789") === $length;

26function leapYear(int $year): bool
27{
28 return $year % 4 === 0 && ($year % 100 !== 0 || $year % 400 === 0);
29}
30function monthLength(int $year, int $month): int
31{
32 return $month === 2 && leapYear($year) ? 29 : MONTH_DAYS[$month - 1];

38 public function number(): int
39 {
40 $result = $this->day - 1;
41 for ($year = 2000; $year < $this->year; $year++) {
42 $result += leapYear($year) ? 366 : 365;
43 }
44 for ($month = 1; $month < $this->month; $month++) {
45 $result += monthLength($this->year, $month);
46 }
47 return $result;
48 }

51 return ($this->number() + 6) % 7;

56 }
57 public function __toString(): string

61 public static function fromNumber(int $number): self
62 {
63 $year = 2000;
64 $month = 1;
65 while ($number >= (leapYear($year) ? 366 : 365)) {
66 $number -= leapYear($year) ? 366 : 365;
67 $year++;
68 }
69 while ($number >= monthLength($year, $month)) {
70 $number -= monthLength($year, $month);
71 $month++;
72 }
73 return new self($year, $month, $number + 1);
74 }
75 public static function parse(string $text): ?self
76 {
77 if (
78 strlen($text) !== 10 ||
79 $text[4] !== "-" ||
80 $text[7] !== "-" ||
81 !isDigits(substr($text, 0, 4), 4, 4) ||
82 !isDigits(substr($text, 5, 2), 2, 2) ||
83 !isDigits(substr($text, 8), 2, 2)
84 ) {
85 return null;
86 }
87 $year = (int) substr($text, 0, 4);
88 $month = (int) substr($text, 5, 2);
89 $day = (int) substr($text, 8);
90 if (
91 $year < 2001 ||
92 $year > 2098 ||
93 $month < 1 ||
94 $month > 12 ||
95 $day < 1 ||
96 $day > monthLength($year, $month)
97 ) {
98 return null;
99 }
100 return new self($year, $month, $day);

299 private static function field(string $text, int $index): CronField
300 {
301 $values = [];
302 $specials = [];
303 $seen = [];
304 foreach (explode(",", $text) as $item) {
305 if ($item === "") {
306 throw self::fieldError($index, "empty item");
307 }
308 $special = null;
309 $hash = strpos($item, "#");
310 if ($index === 2 && (str_starts_with($item, "L") || str_ends_with($item, "W"))) {
311 if ($item === "L") {
312 $special = new Special(SpecialKind::Last);
313 } elseif ($item === "LW") {
314 $special = new Special(SpecialKind::LastWeekday);
315 } elseif (str_starts_with($item, "L-")) {
316 $special = new Special(
317 SpecialKind::BeforeLast,
318 self::specialNumber(substr($item, 2), $index, 30),
319 );
320 } elseif (str_ends_with($item, "W")) {
321 $special = new Special(
322 SpecialKind::NearestWeekday,
323 self::specialNumber(substr($item, 0, -1), $index, 31),
324 );
325 } else {
326 throw self::fieldError($index, "invalid value");
327 }
328 } elseif ($index === 4 && $hash !== false) {
329 $weekday = self::value(substr($item, 0, $hash), $index);
330 $occurrence = self::specialNumber(substr($item, $hash + 1), $index, 5, 1);
331 $special = new Special(SpecialKind::NthWeekday, $weekday % 7, $occurrence);
332 } elseif ($index === 4 && strlen($item) >= 2 && str_ends_with($item, "L")) {
333 $special = new Special(
334 SpecialKind::LastOccurrence,
335 self::value(substr($item, 0, -1), $index) % 7,
336 );
337 }
338 if ($special !== null) {
339 $canonical = (string) $special;
340 if (!isset($seen[$canonical])) {
341 $seen[$canonical] = true;
342 $specials[] = $special;
343 }
344 continue;
345 }
346 $slash = strpos($item, "/");
347 $base = $slash === false ? $item : substr($item, 0, $slash);
348 $dash = strpos($base, "-");
349 if ($base === "*") {
350 $low = self::MINIMUM[$index];
351 $high = self::MAXIMUM[$index];
352 } elseif ($dash !== false) {
353 $low = self::value(substr($base, 0, $dash), $index);
354 $high = self::value(substr($base, $dash + 1), $index);
355 if ($low > $high) {
356 throw self::fieldError($index, "invalid range");
357 }
358 } else {
359 $low = self::value($base, $index);
360 $high = $slash === false ? $low : self::MAXIMUM[$index];
361 }
362 $step = 1;
363 if ($slash !== false) {
364 $part = substr($item, $slash + 1);
365 if (!isDigits($part, 1, 2) || (int) $part === 0) {
366 throw self::fieldError($index, "invalid step");
367 }
368 $step = (int) $part;
369 }
370 for ($value = $low; $value <= $high; $value += $step) {
371 $values[$index === 4 ? $value % 7 : $value] = true;
372 }
373 }
374 $numbers = array_keys($values);
375 sort($numbers, SORT_NUMERIC);
376 return new CronField($index, $numbers, $specials, $text[0] !== "*");
377 }
378 public static function parse(array $words): self

385 throw new CommandError("field count");

393 private static function nearest(CivilDate $date, int $day): int
394 {
395 $weekday = ($date->number() - $date->day + $day + 6) % 7;
396 if ($weekday === 6) {
397 return $day === 1 ? 3 : $day - 1;
398 }
399 if ($weekday === 0) {
400 return $day === monthLength($date->year, $date->month) ? $day - 2 : $day + 1;
401 }
402 return $day;
403 }
404 public function matchDay(CivilDate $date): array
405 {
406 $last = monthLength($date->year, $date->month);
407 $day = $date->day;
408 $weekday = $date->weekday();
409 $monthDay = $this->fields[2]->contains($day);
410 $weekDay = $this->fields[4]->contains($weekday);
411 foreach ($this->fields[2]->specials as $special) {
412 $matches = match ($special->kind) {
413 SpecialKind::Last => $day === $last,
414 SpecialKind::LastWeekday => $day === self::nearest($date, $last),
415 SpecialKind::BeforeLast => $day === $last - $special->value,
416 default => $special->value <= $last &&
417 $day === self::nearest($date, $special->value),
418 };
419 $monthDay = $monthDay || $matches;
420 }
421 foreach ($this->fields[4]->specials as $special) {
422 $matches =
423 $weekday === $special->value &&
424 ($special->kind === SpecialKind::NthWeekday
425 ? intdiv($day - 1, 7) + 1 === $special->occurrence
426 : $day + 7 > $last);
427 $weekDay = $weekDay || $matches;
428 }
429 return [
430 $monthDay,
431 $weekDay,
432 $this->orRule() ? $monthDay || $weekDay : $monthDay && $weekDay,
433 ];
434 }
435 public function scheduled(CivilDate $date): bool
436 {
437 return $this->fields[3]->contains($date->month) && $this->matchDay($date)[2];

439 public function fireTimes(
440 int $low,
441 int $high,
442 array $holidays,
443 bool $reverse = false,
444 ): Generator {
445 $first = intdiv($low, 1440);
446 $last = intdiv($high, 1440);
447 $count = count($this->daily);
448 for (
449 $day = $reverse ? $last : $first;
450 $reverse ? $day >= $first : $day <= $last;
451 $day += $reverse ? -1 : 1
452 ) {
453 if (isset($holidays[$day]) || !$this->scheduled(CivilDate::fromNumber($day))) {
454 continue;
455 }
456 for (
457 $i = $reverse ? $count - 1 : 0;
458 $reverse ? $i >= 0 : $i < $count;
459 $i += $reverse ? -1 : 1
460 ) {
461 $time = $day * 1440 + $this->daily[$i];
462 if ($low <= $time && $time <= $high) {
463 yield $time;
464 }
465 }
466 }

470 $date = CivilDate::fromNumber(intdiv($time, 1440));

472 $reasons = [];
473 if (!$this->fields[0]->contains($time % 60)) {
474 $reasons[] = "minute";

476 if (!$this->fields[1]->contains(intdiv($time % 1440, 60))) {
477 $reasons[] = "hour";
478 }
479 if (!$condition && !$monthDay) {
480 $reasons[] = "day-of-month";
481 }
482 if (!$this->fields[3]->contains($date->month)) {
483 $reasons[] = "month";
484 }
485 if (!$condition && !$weekDay) {
486 $reasons[] = "day-of-week";
487 }
488 if (isset($holidays[$date->number()])) {
489 $reasons[] = "excluded";
490 }

510 private static function valueList(CronField $field, bool $either = false): string
511 {
512 $parts = [];
513 $values = $field->values;
514 $count = count($values);
515 for ($start = 0; $start < $count; ) {
516 $end = $start;
517 while ($end + 1 < $count && $values[$end + 1] === $values[$end] + 1) {
518 $end++;
519 }
520 if ($end - $start >= 2) {
521 $parts[] =
522 self::valueName($field->index, $values[$start]) .
523 " through " .
524 self::valueName($field->index, $values[$end]);
525 } else {
526 for ($i = $start; $i <= $end; $i++) {
527 $parts[] = self::valueName($field->index, $values[$i]);
528 };
529 }
530 $start = $end + 1;
531 }
532 return self::listJoin($parts, $either);
533 }
534 private static function stepped(CronField $field): int
535 {
536 $values = $field->values;
537 if (count($values) < 2 || $values[0] !== 0 || $values[1] < 2) {
538 return 0;
539 }
540 $step = $values[1];
541 if (count($values) !== intdiv(self::MAXIMUM[$field->index], $step) + 1) {
542 return 0;
543 }
544 foreach ($values as $i => $value) {
545 if ($value !== $i * $step) {
546 return 0;
547 }
548 }
549 return $step;
550 }
551 private function dayClause(bool $weekday, bool $either = false): string
552 {
553 $field = $this->fields[$weekday ? 4 : 2];
554 $parts = [];
555 if ($field->values !== []) {
556 $parts[] = $weekday
557 ? self::valueList($field, $either)
558 : (count($field->values) === 1 ? "day " : "days ") . self::valueList($field);
559 }
560 $ordinals = ["1st", "2nd", "3rd", "4th", "5th"];
561 foreach ($field->specials as $special) {
562 $parts[] = match ($special->kind) {
563 SpecialKind::Last => "the last day",
564 SpecialKind::LastWeekday => "the last weekday",
565 SpecialKind::BeforeLast => $special->value .
566 ($special->value === 1 ? " day" : " days") .
567 " before the last day",
568 SpecialKind::NearestWeekday => "the weekday nearest day " . $special->value,
569 SpecialKind::NthWeekday => "the " .
570 $ordinals[$special->occurrence - 1] .
571 " " .
572 WEEKDAYS[$special->value],
573 SpecialKind::LastOccurrence => "the last " . WEEKDAYS[$special->value],
574 };
575 }
576 return self::listJoin($parts, $either) .
577 (!$weekday || $field->specials !== [] ? " of the month" : "");
578 }
579 public function describe(): string
580 {
581 $minute = $this->fields[0];
582 $hour = $this->fields[1];
583 if (count($this->daily) <= 4) {
584 $text =
585 "At " .
586 self::listJoin(
587 array_map(
588 static fn(int $time): string => sprintf(
589 "%02d:%02d",
590 intdiv($time, 60),
591 $time % 60,
592 ),
593 $this->daily,
594 ),
595 );
596 } else {
597 $minuteStep = self::stepped($minute);
598 $hourStep = self::stepped($hour);
599 $text = match (true) {
600 count($minute->values) === 60 => "Every minute",
601 $minuteStep !== 0 => "Every $minuteStep minutes",
602 count($minute->values) === 1 => "At minute " . $minute->values[0],
603 default => "At minutes " . self::valueList($minute),
604 };
605 if (count($hour->values) === 24) {
606 if (str_starts_with($text, "At minute")) {
607 $text .= " of every hour";
608 }
609 } elseif ($hourStep !== 0) {
610 $text .= ", every $hourStep hours";
611 } elseif (
612 $hour->values[count($hour->values) - 1] - $hour->values[0] + 1 ===
613 count($hour->values)
614 ) {
615 $text .= sprintf(
616 ", between %02d:00 and %02d:59",
617 $hour->values[0],
618 $hour->values[count($hour->values) - 1],
619 );
620 } else {
621 $text .= ", during hours " . self::valueList($hour);
622 }
623 }
624 $monthFull = count($this->fields[2]->values) === 31;
625 $weekFull = count($this->fields[4]->values) === 7;
626 if ($this->orRule()) {
627 if (!$monthFull && !$weekFull) {
628 $text .= ", on " . $this->dayClause(false) . " or on " . $this->dayClause(true);
629 }
630 } elseif (!$monthFull && !$weekFull) {
631 $text .=
632 ", on " . $this->dayClause(false) . " if it is " . $this->dayClause(true, true);
633 } elseif (!$monthFull) {
634 $text .= ", on " . $this->dayClause(false);
635 } elseif (!$weekFull) {
636 $text .= ", on " . $this->dayClause(true);
637 }
638 if (count($this->fields[3]->values) !== 12) {
639 $text .= ", in " . self::valueList($this->fields[3]);
640 }
641 return $text;
642 }

830 $events = [];

861 $peak = 0;
862 $at = 0;

866 $peak = $running;
867 $at = $time;
868 }
869 }
870 return array_merge(
871 [
872 "SIMULATE " .
873 count($jobs) .
874 " peak $peak" .
