5type Checked<T> = Result<T, String>;
6const FIELD_NAMES: [&str; 5] = ["minute", "hour", "day-of-month", "month", "day-of-week"];
7const MINIMUM: [i32; 5] = [0, 0, 1, 1, 0];
8const MAXIMUM: [i32; 5] = [59, 23, 31, 12, 7];
9const MONTHS: [&str; 12] = ["January", "February", "March", "April", "May", "June", "July", "August", "September", "October", "November", "December"];
10const WEEKDAYS: [&str; 7] = ["Sunday", "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday"];

12fn digits(text: &str, minimum: usize, maximum: usize) -> bool {
13 (minimum..=maximum).contains(&text.len()) && text.bytes().all(|c| c.is_ascii_digit())
14}
15fn integer(text: &str) -> i32 { text.parse().unwrap() }
16fn leap(year: i32) -> bool { year % 4 == 0 && (year % 100 != 0 || year % 400 == 0) }
17fn month_length(year: i32, month: i32) -> i32 {
18 [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31][month as usize - 1] + (month == 2 && leap(year)) as i32
19}
20#[derive(Clone, Copy)]
21struct Date { year: i32, month: i32, day: i32 }
22impl Date {
23 fn number(self) -> i32 {
24 let years: i32 = (2000..self.year).map(|y| if leap(y) { 366 } else { 365 }).sum();
25 years + (1..self.month).map(|m| month_length(self.year, m)).sum::<i32>() + self.day - 1
26 }
27 fn from_number(mut number: i32) -> Self {
28 let mut year = 2000;
29 loop {
30 let days = if leap(year) { 366 } else { 365 };
31 if number < days { break; }
32 number -= days; year += 1;
33 }
34 let mut month = 1;
35 while number >= month_length(year, month) { number -= month_length(year, month); month += 1; }
36 Self { year, month, day: number + 1 }
37 }
38 fn weekday(self) -> i32 { (self.number() + 6) % 7 }
39}
40impl fmt::Display for Date {
41 fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
42 write!(f, "{:04}-{:02}-{:02} {}", self.year, self.month, self.day, &WEEKDAYS[self.weekday() as usize][..3])
43 }
44}
45fn pattern(text: &str, pattern: &str) -> bool {
46 text.len() == pattern.len() && text.bytes().zip(pattern.bytes()).all(|(a, b)| if b == b'#' { a.is_ascii_digit() } else { a == b })
47}
48fn read_date(text: &str, reason: &str) -> Checked<Date> {
49 if !pattern(text, "####-##-##") { return Err(reason.into()); }
50 let date = Date { year: integer(&text[..4]), month: integer(&text[5..7]), day: integer(&text[8..]) };
51 if !(2001..=2098).contains(&date.year) || !(1..=12).contains(&date.month) || date.day < 1 || date.day > month_length(date.year, date.month) {
52 return Err(reason.into());
53 }
54 Ok(date)
55}
56fn read_datetime(words: &[&str], at: usize) -> Checked<i32> {
57 let date = read_date(words[at], "invalid datetime")?;
58 let time = words[at + 1];
59 if !pattern(time, "##:##") { return Err("invalid datetime".into()); }
60 let hour = integer(&time[..2]); let minute = integer(&time[3..]);
61 if hour > 23 || minute > 59 { return Err("invalid datetime".into()); }
62 Ok(date.number() * 1440 + hour * 60 + minute)
63}
64fn read_month(text: &str) -> Checked<Date> {
65 if !pattern(text, "####-##") { return Err("invalid month".into()); }
66 let date = Date { year: integer(&text[..4]), month: integer(&text[5..]), day: 1 };
67 if !(2001..=2098).contains(&date.year) || !(1..=12).contains(&date.month) { return Err("invalid month".into()); }
68 Ok(date)
69}
70fn datetime_text(time: i32) -> String {
71 let date = Date::from_number(time / 1440);
72 format!("{:04}-{:02}-{:02} {:02}:{:02} {}", date.year, date.month, date.day, time % 1440 / 60, time % 60, &WEEKDAYS[date.weekday() as usize][..3])
73}
74fn span(minutes: i32) -> String { format!("{}d{:02}h{:02}m", minutes / 1440, minutes % 1440 / 60, minutes % 60) }
75fn positive(text: &str, width: usize, maximum: i32, reason: &str) -> Checked<i32> {
76 if !digits(text, 1, width) || text.starts_with('0') || integer(text) > maximum { Err(reason.into()) } else { Ok(integer(text)) }
77}
78fn window(words: &[&str], at: usize) -> Checked<(i32, i32)> {
79 let low = read_datetime(words, at)?;
80 let high = read_datetime(words, at + 2)?;
81 if low > high { return Err("invalid window".into()); }
82 if high - low > 44640 { return Err("window too long".into()); }
83 Ok((low, high))
84}

86#[derive(Clone, Copy, PartialEq, Eq)]
87enum Special { Last, LastWorkday, BeforeLast(i32), Nearest(i32), Nth(i32, i32), LastWeekday(i32) }
88impl fmt::Display for Special {
89 fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
90 match self {
91 Self::Last => write!(f, "L"), Self::LastWorkday => write!(f, "LW"), Self::BeforeLast(n) => write!(f, "L-{n}"),
92 Self::Nearest(n) => write!(f, "{n}W"), Self::Nth(v, n) => write!(f, "{v}#{n}"), Self::LastWeekday(v) => write!(f, "{v}L"),
93 }
94 }
95}
96struct Field { values: BTreeSet<i32>, specials: Vec<Special>, restricted: bool }
97fn read_value(text: &str, index: usize) -> Checked<i32> {
98 let names: &[&str] = match index {
99 3 => &["JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV", "DEC"],
100 4 => &["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"], _ => &[],
101 };
102 if let Some(position) = names.iter().position(|&name| name == text) { return Ok(position as i32 + (index == 3) as i32); }
103 if !digits(text, 1, 2) { return Err("invalid value".into()); }
104 let value = integer(text);
105 if !(MINIMUM[index]..=MAXIMUM[index]).contains(&value) { return Err("out of range".into()); }
106 Ok(value)
107}
108fn special_number(text: &str, maximum: i32, width: usize) -> Checked<i32> {
109 if !digits(text, 1, width) { return Err("invalid value".into()); }
110 let value = integer(text);
111 if !(1..=maximum).contains(&value) { return Err("out of range".into()); }
112 Ok(value)
113}
114impl Field {
115 fn parse(text: &str, index: usize) -> Checked<Self> {
116 let mut field = Self { values: BTreeSet::new(), specials: Vec::new(), restricted: !text.starts_with('*') };
117 for item in text.split(',') { field.add_item(item, index).map_err(|reason| format!("{} {reason}", FIELD_NAMES[index]))?; }
118 Ok(field)
119 }
120 fn add_item(&mut self, item: &str, index: usize) -> Checked<()> {
121 if item.is_empty() { return Err("empty item".into()); }
122 let special = if index == 2 && (item.starts_with('L') || item.ends_with('W')) {
123 Some(match item {
124 "L" => Special::Last, "LW" => Special::LastWorkday,
125 _ if item.starts_with("L-") => Special::BeforeLast(special_number(&item[2..], 30, 2)?),
126 _ if item.ends_with('W') => Special::Nearest(special_number(&item[..item.len() - 1], 31, 2)?),
127 _ => return Err("invalid value".into()),
128 })
129 } else if index == 4 && item.contains('#') {
130 let (value, n) = item.split_once('#').unwrap();
131 let weekday = read_value(value, index)?;
132 Some(Special::Nth(weekday % 7, special_number(n, 5, 1)?))
133 } else if index == 4 && item.len() >= 2 && item.ends_with('L') {
134 Some(Special::LastWeekday(read_value(&item[..item.len() - 1], index)? % 7))
135 } else { None };
136 if let Some(special) = special {
137 if !self.specials.contains(&special) { self.specials.push(special); }
138 return Ok(());
139 }
140 let (base, step) = item.split_once('/').map_or((item, None), |(base, step)| (base, Some(step)));
141 let (first, last) = if base == "*" { (MINIMUM[index], MAXIMUM[index]) }
142 else if let Some((a, b)) = base.split_once('-') {
143 let a = read_value(a, index)?; let b = read_value(b, index)?;
144 if a > b { return Err("invalid range".into()); }
145 (a, b)
146 } else {
147 let a = read_value(base, index)?;
148 (a, if step.is_some() { MAXIMUM[index] } else { a })
149 };
150 let step = match step {
151 Some(text) if !digits(text, 1, 2) || integer(text) == 0 => return Err("invalid step".into()),
152 Some(text) => integer(text), None => 1,
153 };
154 for value in (first..=last).step_by(step as usize) { self.values.insert(if index == 4 { value % 7 } else { value }); }
155 Ok(())
156 }
157 fn stepped(&self, maximum: i32) -> i32 {
158 let values: Vec<i32> = self.values.iter().copied().collect();
159 if values.len() < 2 || values[0] != 0 || values[1] < 2 { return 0; }
160 let step = values[1];
161 if (0..=maximum).step_by(step as usize).eq(values) { step } else { 0 }
162 }
163 fn value_list(&self, index: usize, either: bool) -> String {
164 let name = |value: i32| match index { 3 => MONTHS[value as usize - 1].to_string(), 4 => WEEKDAYS[value as usize].to_string(), _ => value.to_string() };
165 let values: Vec<i32> = self.values.iter().copied().collect();
166 let mut parts = Vec::new(); let mut first = 0;
167 while first < values.len() {
168 let mut last = first;
169 while last + 1 < values.len() && values[last + 1] == values[last] + 1 { last += 1; }
170 if last - first >= 2 { parts.push(format!("{} through {}", name(values[first]), name(values[last]))); }
171 else { parts.extend(values[first..=last].iter().map(|&v| name(v))); }
172 first = last + 1;
173 }
174 word_list(&parts, either)
175 }
176 fn day_phrase(&self, weekday: bool, either: bool) -> String {
177 let mut parts = Vec::new();
178 if !self.values.is_empty() {
179 parts.push(if weekday { self.value_list(4, either) }
180 else { format!("{} {}", if self.values.len() == 1 { "day" } else { "days" }, self.value_list(2, false)) });
181 }
182 for &item in &self.specials {
183 parts.push(match item {
184 Special::Last => "the last day".into(), Special::LastWorkday => "the last weekday".into(),
185 Special::BeforeLast(n) => format!("{n} {} before the last day", if n == 1 { "day" } else { "days" }),
186 Special::Nearest(n) => format!("the weekday nearest day {n}"),
187 Special::Nth(v, n) => format!("the {} {}", ["1st", "2nd", "3rd", "4th", "5th"][n as usize - 1], WEEKDAYS[v as usize]),
188 Special::LastWeekday(v) => format!("the last {}", WEEKDAYS[v as usize]),
189 });
190 }
191 format!("{}{}", word_list(&parts, either), if !weekday || !self.specials.is_empty() { " of the month" } else { "" })
192 }
193}
194fn word_list(parts: &[String], either: bool) -> String {
195 if parts.len() < 2 { return parts.join(""); }
196 format!("{} {} {}", parts[..parts.len() - 1].join(", "), if either { "or" } else { "and" }, parts.last().unwrap())
197}
198struct Cron { fields: [Field; 5], daily: Vec<i32>, text: String, expanded: String }
199impl Cron {
200 fn parse(words: &[&str]) -> Checked<Self> {
201 let text = words.join(" ");
202 let macro_text = if words.len() == 1 && words[0].starts_with('@') {
203 Some(match words[0] {
204 "@yearly" | "@annually" => "0 0 1 1 *", "@monthly" => "0 0 1 * *", "@weekly" => "0 0 * * 0",
205 "@daily" | "@midnight" => "0 0 * * *", "@hourly" => "0 * * * *", _ => return Err("unknown macro".into()),
206 })
207 } else { None };
208 let expanded_words: Vec<&str> = macro_text.map_or_else(|| words.to_vec(), |m| m.split(' ').collect());
209 if expanded_words.len() != 5 { return Err("field count".into()); }
210 let mut parsed = Vec::new();
211 for (i, word) in expanded_words.iter().enumerate() { parsed.push(Field::parse(word, i)?); }
212 let fields: [Field; 5] = parsed.try_into().ok().unwrap();
213 let daily = fields[1].values.iter().flat_map(|&h| fields[0].values.iter().map(move |&m| h * 60 + m)).collect();
214 Ok(Self { fields, daily, text, expanded: expanded_words.join(" ") })
215 }
216 fn or_rule(&self) -> bool { self.fields[2].restricted && self.fields[4].restricted }
217 fn match_day(&self, date: Date) -> (bool, bool, bool) {
218 let last = month_length(date.year, date.month); let weekday = date.weekday();
219 let nearest = |day: i32| match (date.number() - date.day + day + 6) % 7 {
220 6 => if day == 1 { 3 } else { day - 1 }, 0 => if day == last { day - 2 } else { day + 1 }, _ => day,
221 };
222 let dom = self.fields[2].values.contains(&date.day) || self.fields[2].specials.iter().any(|&item| match item {
223 Special::Last => date.day == last, Special::LastWorkday => date.day == nearest(last),
224 Special::BeforeLast(n) => date.day == last - n, Special::Nearest(n) => n <= last && date.day == nearest(n), _ => false,
225 });
226 let dow = self.fields[4].values.contains(&weekday) || self.fields[4].specials.iter().any(|&item| match item {
227 Special::Nth(v, n) => weekday == v && (date.day - 1) / 7 + 1 == n,
228 Special::LastWeekday(v) => weekday == v && date.day + 7 > last, _ => false,
229 });
230 (dom, dow, if self.or_rule() { dom || dow } else { dom && dow })
231 }
232 fn scheduled(&self, date: Date) -> bool { self.fields[3].values.contains(&date.month) && self.match_day(date).2 }
233 fn fires(&self, low: i32, high: i32, holidays: &BTreeSet<i32>, reverse: bool, limit: Option<usize>) -> Vec<i32> {
234 let first = low / 1440; let last = high / 1440; let step = if reverse { -1 } else { 1 };
235 let mut result = Vec::new(); let mut day = if reverse { last } else { first };
236 while day >= first && day <= last {
237 if !holidays.contains(&day) && self.scheduled(Date::from_number(day)) {
238 let mut i = if reverse { self.daily.len() as i32 - 1 } else { 0 };
239 while i >= 0 && i < self.daily.len() as i32 {
240 let time = day * 1440 + self.daily[i as usize];
241 if low <= time && time <= high {
242 result.push(time);
243 if limit == Some(result.len()) { return result; }
244 }
245 i += step;
246 }
247 }
248 day += step;
249 }
250 result
251 }
252 fn description(&self) -> String {
253 let [minute, hour, dom, month, dow] = &self.fields;
254 let mut text = if self.daily.len() <= 4 {
255 let times: Vec<String> = self.daily.iter().map(|t| format!("{:02}:{:02}", t / 60, t % 60)).collect();
256 format!("At {}", word_list(&times, false))
257 } else {
258 let ms = minute.stepped(59); let hs = hour.stepped(23);
259 let mut text = if minute.values.len() == 60 { "Every minute".into() }
260 else if ms > 0 { format!("Every {ms} minutes") }
261 else { format!("At {} {}", if minute.values.len() == 1 { "minute" } else { "minutes" }, minute.value_list(0, false)) };
262 if hour.values.len() == 24 { if text.starts_with("At minute") { text.push_str(" of every hour"); } }
263 else if hs > 0 { text.push_str(&format!(", every {hs} hours")); }
264 else if hour.values.last().unwrap() - hour.values.first().unwrap() + 1 == hour.values.len() as i32 {
265 text.push_str(&format!(", between {:02}:00 and {:02}:59", hour.values.first().unwrap(), hour.values.last().unwrap()));
266 } else { text.push_str(&format!(", during hours {}", hour.value_list(1, false))); }
267 text
268 };
269 let d_full = dom.values.len() == 31; let w_full = dow.values.len() == 7;
270 if self.or_rule() {
271 if !d_full && !w_full { text.push_str(&format!(", on {} or on {}", dom.day_phrase(false, false), dow.day_phrase(true, false))); }
272 } else if !d_full && !w_full { text.push_str(&format!(", on {} if it is {}", dom.day_phrase(false, false), dow.day_phrase(true, true))); }
273 else if !d_full { text.push_str(&format!(", on {}", dom.day_phrase(false, false))); }
274 else if !w_full { text.push_str(&format!(", on {}", dow.day_phrase(true, false))); }
275 if month.values.len() != 12 { text.push_str(&format!(", in {}", month.value_list(3, false))); }
276 text
277 }
278}

280#[derive(Clone, Copy, PartialEq, Eq)]
281enum Policy { Allow, Forbid, Replace }
282impl fmt::Display for Policy {
283 fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result { write!(f, "{}", match self { Self::Allow => "ALLOW", Self::Forbid => "FORBID", Self::Replace => "REPLACE" }) }
284}
285struct Job { cron: Cron, enabled: bool, duration: i32, policy: Policy }
286#[derive(Default)]
287struct Scheduler { jobs: BTreeMap<String, Job>, holidays: BTreeSet<i32> }
288impl Scheduler {
289 fn job(&self, id: &str) -> Checked<&Job> { self.jobs.get(id).ok_or_else(|| "unknown job".into()) }
290 fn new_id(&self, id: &str) -> Checked<()> {
291 if !(1..=16).contains(&id.len()) || !id.as_bytes()[0].is_ascii_lowercase() || !id.bytes().all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || c == b'_') {
292 return Err("invalid id".into());
293 }
294 if self.jobs.contains_key(id) { return Err("duplicate job".into()); }
295 Ok(())
296 }
297 fn enabled(&self) -> Vec<(&str, &Job)> { self.jobs.iter().filter(|(_, j)| j.enabled).map(|(id, j)| (id.as_str(), j)).collect() }
298 fn timeline(&self, low: i32, high: i32, conflicts: bool) -> Vec<String> {
299 let enabled = self.enabled();
300 let mut masks = vec![0_u64; (high - low + 1) as usize];
301 for (i, (_, job)) in enabled.iter().enumerate() {
302 for time in job.cron.fires(low, high, &self.holidays, false, None) { masks[(time - low) as usize] |= 1 << i; }
303 }
304 let total: usize = masks.iter().map(|m| if conflicts { (m.count_ones() >= 2) as usize } else { m.count_ones() as usize }).sum();
305 let peak = masks.iter().map(|m| m.count_ones()).filter(|&n| n >= 2).max().unwrap_or(0);
306 let mut rows = vec![if conflicts { format!("CONFLICTS {total} peak {peak}") } else { format!("TIMELINE {total}") }];
307 for (offset, &mask) in masks.iter().enumerate() {
308 if rows.len() == 101 { break; }
309 if conflicts {
310 if mask.count_ones() < 2 { continue; }
311 let ids: Vec<&str> = enabled.iter().enumerate().filter(|(i, _)| mask & (1_u64 << *i) != 0).map(|(_, (id, _))| *id).collect();
312 rows.push(format!("{} {} {}", datetime_text(low + offset as i32), ids.len(), ids.join(" ")));
313 } else {
314 for (i, (id, _)) in enabled.iter().enumerate() {
315 if rows.len() == 101 { break; }
316 if mask & (1 << i) != 0 { rows.push(format!("{} {id}", datetime_text(low + offset as i32))); }
317 }
318 }
319 }
320 if total > 100 { rows.push(format!("... {} more", total - 100)); }
321 rows
322 }
323 fn simulate(&self, low: i32, high: i32) -> Vec<String> {
324 let enabled = self.enabled();
325 let mut events = vec![0; (high - low + 1441) as usize];
326 let mut rows = Vec::new();
327 for &(id, job) in &enabled {
328 let (mut started, mut skipped, mut replaced, mut busy) = (0, 0, 0, 0);
329 let mut last_end = -1;
330 for time in job.cron.fires(low, high, &self.holidays, false, None) {
331 if last_end > time && job.policy == Policy::Forbid { skipped += 1; continue; }
332 if last_end > time && job.policy == Policy::Replace {
333 events[(last_end - low) as usize] += 1; events[(time - low) as usize] -= 1;
334 busy -= last_end - time; replaced += 1;
335 }
336 last_end = time + job.duration;
337 events[(time - low) as usize] += 1; events[(last_end - low) as usize] -= 1;
338 started += 1; busy += job.duration;
339 }
340 rows.push(format!("{id} {} started {started} skipped {skipped} replaced {replaced} busy {busy}", job.policy));
341 }
342 let (mut running, mut peak, mut at) = (0, 0, 0);
343 for (i, change) in events.into_iter().enumerate() { running += change; if running > peak { peak = running; at = low + i as i32; } }
344 rows.insert(0, format!("SIMULATE {} peak {peak}{}", enabled.len(), if peak == 0 { String::new() } else { format!(" at {}", datetime_text(at)) }));
345 rows
346 }
347 fn execute(&mut self, words: &[&str]) -> Checked<Vec<String>> {
348 let command = words[0];
349 let expected: i32 = match command {
350 "JOB" => -3, "VALIDATE" => -2, "REMOVE" | "ENABLE" | "DISABLE" | "DESCRIBE" | "WEEK" | "EXCLUDE" | "INCLUDE" => 2,
351 "RENAME" | "EXPAND" | "CALENDAR" => 3, "RUNTIME" | "MATCH" | "GAP" => 4,
352 "NEXT" | "PREV" | "TIMELINE" | "CONFLICTS" | "SIMULATE" => 5, "COUNT" => 6, "OVERLAP" => 7,
353 "LIST" | "HOLIDAYS" => 1, _ => return Err("unknown command".into()),
354 };
355 let wrong_count = if expected < 0 { words.len() < (-expected) as usize } else { words.len() != expected as usize };
356 if wrong_count { return Err("wrong arguments".into()); }
357 match command {
358 "JOB" => {
359 self.new_id(words[1])?;
360 let cron = Cron::parse(&words[2..])?;
361 self.jobs.insert(words[1].into(), Job { cron, enabled: true, duration: 1, policy: Policy::Allow });
362 Ok(vec![format!("JOB {} added", words[1])])
363 }
364 "VALIDATE" => { let cron = Cron::parse(&words[1..])?; Ok(vec![format!("VALID {} {} {}", cron.expanded, if cron.or_rule() { "OR" } else { "AND" }, cron.daily.len())]) }
365 "REMOVE" => { self.job(words[1])?; self.jobs.remove(words[1]); Ok(vec![format!("REMOVE {}", words[1])]) }
366 "RENAME" => {
367 self.job(words[1])?; self.new_id(words[2])?;
368 let job = self.jobs.remove(words[1]).unwrap(); self.jobs.insert(words[2].into(), job);
369 Ok(vec![format!("RENAME {} {}", words[1], words[2])])
370 }
371 "ENABLE" | "DISABLE" => {
372 let job = self.jobs.get_mut(words[1]).ok_or("unknown job")?;
373 let enabled = command == "ENABLE";
374 if job.enabled == enabled { return Err(format!("already {}", if enabled { "enabled" } else { "disabled" })); }
375 job.enabled = enabled;
376 Ok(vec![format!("{command} {}", words[1])])
377 }
378 "RUNTIME" => {
379 self.job(words[1])?;
380 let duration = positive(words[2], 4, 1440, "invalid duration")?;
381 let policy = match words[3] { "ALLOW" => Policy::Allow, "FORBID" => Policy::Forbid, "REPLACE" => Policy::Replace, _ => return Err("invalid policy".into()) };
382 let job = self.jobs.get_mut(words[1]).unwrap(); job.duration = duration; job.policy = policy;
383 Ok(vec![format!("RUNTIME {} {duration} {policy}", words[1])])
384 }
385 "LIST" => {
386 let mut rows = vec![format!("LIST {}", self.jobs.len())];
387 rows.extend(self.jobs.iter().map(|(id, job)| format!("{id} {} {} {} {}", if job.enabled { "on" } else { "off" }, job.duration, job.policy, job.cron.text)));
388 Ok(rows)
389 }
390 "EXPAND" => {
391 let job = self.job(words[1])?;
392 let index = FIELD_NAMES.iter().position(|&name| name == words[2]).ok_or("invalid field")?;
393 let field = &job.cron.fields[index];
394 let mut items: Vec<String> = field.values.iter().map(ToString::to_string).collect();
395 items.extend(field.specials.iter().map(ToString::to_string));
396 Ok(vec![format!("EXPAND {} {} {}{}", words[1], words[2], field.values.len(), if items.is_empty() { String::new() } else { format!(" {}", items.join(" ")) })])
397 }
398 "DESCRIBE" => Ok(vec![format!("DESCRIBE {} {}", words[1], self.job(words[1])?.cron.description())]),
399 "MATCH" => {
400 let job = self.job(words[1])?; let time = read_datetime(words, 2)?;
401 let date = Date::from_number(time / 1440); let (dom, dow, condition) = job.cron.match_day(date);
402 let mut reasons = Vec::new();
403 if !job.cron.fields[0].values.contains(&(time % 60)) { reasons.push("minute"); }
404 if !job.cron.fields[1].values.contains(&(time % 1440 / 60)) { reasons.push("hour"); }
405 if !condition && !dom { reasons.push("day-of-month"); }
406 if !job.cron.fields[3].values.contains(&date.month) { reasons.push("month"); }
407 if !condition && !dow { reasons.push("day-of-week"); }
408 if self.holidays.contains(&(time / 1440)) { reasons.push("excluded"); }
409 Ok(vec![format!("MATCH {} {} {}", words[1], datetime_text(time), if reasons.is_empty() { "yes".into() } else { format!("no {}", reasons.join(" ")) })])
410 }
411 "NEXT" | "PREV" => {
412 let job = self.job(words[1])?; let base = read_datetime(words, 2)?;
413 let limit = positive(words[4], 2, 50, "invalid count")? as usize; let reverse = command == "PREV";
414 let times = job.cron.fires(if reverse { base - 527040 } else { base + 1 }, if reverse { base - 1 } else { base + 527040 }, &self.holidays, reverse, Some(limit));
415 let mut rows = vec![format!("{command} {} {}", words[1], times.len())];
416 rows.extend(times.iter().map(|&t| format!("{} {}{}", datetime_text(t), if reverse { '-' } else { '+' }, span((t - base).abs()))));
417 if times.len() < limit { rows.push("NONE".into()); }
418 Ok(rows)
419 }
420 "COUNT" => {
421 let job = self.job(words[1])?; let (low, high) = window(words, 2)?;
422 let times = job.cron.fires(low, high, &self.holidays, false, None);
423 let days: BTreeSet<i32> = times.iter().map(|t| t / 1440).collect();
424 Ok(vec![format!("COUNT {} runs {} days {}", words[1], times.len(), days.len())])
425 }
426 "GAP" => {
427 let job = self.job(words[1])?; let base = read_datetime(words, 2)?;
428 let times = job.cron.fires(base, base + 44640, &self.holidays, false, None);
429 if times.len() < 2 { return Ok(vec![format!("GAP {} runs {} none", words[1], times.len())]); }
430 let mut histogram = BTreeMap::new(); let mut longest = 0; let mut first = 0;
431 for (i, pair) in times.windows(2).enumerate() {
432 let gap = pair[1] - pair[0]; *histogram.entry(gap).or_insert(0) += 1;
433 if gap > longest { longest = gap; first = i; }
434 }
435 let shortest = *histogram.keys().next().unwrap();
436 let mut rows = vec![format!("GAP {} runs {} shortest {} longest {} from {} to {}", words[1], times.len(), span(shortest), span(longest), datetime_text(times[first]), datetime_text(times[first + 1]))];
437 rows.extend(histogram.iter().map(|(&gap, count)| format!("{} {count}", span(gap))));
438 Ok(rows)
439 }
440 "OVERLAP" => {
441 let a = self.job(words[1])?; let b = self.job(words[2])?; let (low, high) = window(words, 3)?;
442 let other: BTreeSet<i32> = b.cron.fires(low, high, &self.holidays, false, None).into_iter().collect();
443 let times: Vec<i32> = a.cron.fires(low, high, &self.holidays, false, None).into_iter().filter(|t| other.contains(t)).collect();
444 Ok(vec![format!("OVERLAP {} {} {}{}", words[1], words[2], times.len(), if times.is_empty() { String::new() } else { format!(" first {} last {}", datetime_text(times[0]), datetime_text(*times.last().unwrap())) })])
445 }
446 "TIMELINE" | "CONFLICTS" | "SIMULATE" => {
447 let (low, high) = window(words, 1)?;
448 Ok(if command == "SIMULATE" { self.simulate(low, high) } else { self.timeline(low, high, command == "CONFLICTS") })
449 }
450 "WEEK" => {
451 let first = read_date(words[1], "invalid date")?.number(); let mut counts = [[0; 24]; 7];
452 for (_, job) in self.enabled() {
453 for time in job.cron.fires(first * 1440, (first + 7) * 1440 - 1, &self.holidays, false, None) { counts[(time / 1440 - first) as usize][(time % 1440 / 60) as usize] += 1; }
454 }
455 let total: i32 = counts.iter().flatten().sum(); let mut rows = vec![format!("WEEK {} {total}", words[1])];
456 for (d, hours) in counts.iter().enumerate() {
457 let heat: String = hours.iter().map(|&n| if n == 0 { '.' } else if n >= 10 { '+' } else { (b'0' + n as u8) as char }).collect();
458 rows.push(format!("{} {heat} {}", Date::from_number(first + d as i32), hours.iter().sum::<i32>()));
459 }
460 Ok(rows)
461 }
462 "CALENDAR" => {
463 let job = self.job(words[1])?; let first = read_month(words[2])?;
464 let mut cells = vec!["...".to_string(); first.weekday() as usize]; let mut days = 0;
465 for d in 1..=month_length(first.year, first.month) {
466 let date = Date { day: d, ..first };
467 let mark = if !job.cron.scheduled(date) { '-' } else if self.holidays.contains(&date.number()) { 'x' } else { '*' };
468 if mark == '*' { days += 1; }
469 cells.push(format!("{d:02}{mark}"));
470 }
471 while cells.len() % 7 != 0 { cells.push("...".into()); }
472 let mut rows = vec![format!("CALENDAR {} {} days {days} runs {}", words[1], words[2], days * job.cron.daily.len()), "Sun Mon Tue Wed Thu Fri Sat".into()];
473 rows.extend(cells.chunks(7).map(|week| week.join(" "))); Ok(rows)
474 }
475 "EXCLUDE" | "INCLUDE" => {
476 let date = read_date(words[1], "invalid date")?;
477 if command == "EXCLUDE" { if !self.holidays.insert(date.number()) { return Err("already excluded".into()); } }
478 else if !self.holidays.remove(&date.number()) { return Err("not excluded".into()); }
479 Ok(vec![format!("{command} {date} holidays {}", self.holidays.len())])
480 }
481 "HOLIDAYS" => {
482 let mut rows = vec![format!("HOLIDAYS {}", self.holidays.len())];
483 rows.extend(self.holidays.iter().map(|&day| Date::from_number(day).to_string())); Ok(rows)
484 }
485 _ => unreachable!(),
486 }
487 }
488}

490fn main() {
491 let stdin = io::stdin(); let mut lines = stdin.lock().lines();
492 let Some(Ok(first)) = lines.next() else { return; };
493 let count: usize = first.parse().unwrap();
494 let mut scheduler = Scheduler::default(); let mut output = io::BufWriter::new(io::stdout().lock());
495 for _ in 0..count {
496 let line = lines.next().unwrap().unwrap();
497 let words: Vec<&str> = line.split(' ').filter(|word| !word.is_empty()).collect();
498 match scheduler.execute(&words) {
499 Ok(rows) => for row in rows { writeln!(output, "{row}").unwrap(); },
500 Err(reason) => writeln!(output, "ERROR {reason}").unwrap(),
501 }
502 }
503}
