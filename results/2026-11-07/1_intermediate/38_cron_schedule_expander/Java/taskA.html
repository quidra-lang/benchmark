15 static boolean digits(String text, int minimum, int maximum) {

18 static int integer(String text) { return Integer.parseInt(text); }
19 static boolean leap(int year) { return year % 4 == 0 && (year % 100 != 0 || year % 400 == 0); }
20 static int monthLength(int year, int month) { return month == 2 && leap(year) ? 29 : MONTH_DAYS[month - 1]; }
21 record Date(int year, int month, int day) {
22 int number() {
23 int result = day - 1;
24 for (int y = 2000; y < year; y++) result += leap(y) ? 366 : 365;
25 for (int m = 1; m < month; m++) result += monthLength(year, m);
26 return result;
27 }
28 int weekday() { return (number() + 6) % 7; }
29 static Date fromNumber(int number) {
30 int year = 2000, month = 1;
31 while (number >= (leap(year) ? 366 : 365)) { number -= leap(year) ? 366 : 365; year++; }
32 while (number >= monthLength(year, month)) { number -= monthLength(year, month); month++; }
33 return new Date(year, month, number + 1);
34 }
35 @Override public String toString() { return format("%04d-%02d-%02d %s", year, month, day, WEEKDAYS[weekday()].substring(0, 3)); }

37 static boolean pattern(String text, String pattern) {
38 if (text.length() != pattern.length()) return false;
39 for (int i = 0; i < pattern.length(); i++) {
40 char value = text.charAt(i), required = pattern.charAt(i);
41 if (required == '#' ? value < '0' || value > '9' : value != required) return false;
42 }
43 return true;
44 }
45 static Date readDate(String text, String reason) {
46 if (!pattern(text, "####-##-##")) throw new CommandError(reason);
47 Date date = new Date(integer(text.substring(0, 4)), integer(text.substring(5, 7)), integer(text.substring(8)));
48 if (date.year() < 2001 || date.year() > 2098 || date.month() < 1 || date.month() > 12 || date.day() < 1 || date.day() > monthLength(date.year(), date.month())) throw new CommandError(reason);
49 return date;
50 }
51 static int readDatetime(String[] words, int at) {
52 Date date = readDate(words[at], "invalid datetime");
53 String time = words[at + 1];
54 if (!pattern(time, "##:##")) throw new CommandError("invalid datetime");
55 int hour = integer(time.substring(0, 2)), minute = integer(time.substring(3));
56 if (hour > 23 || minute > 59) throw new CommandError("invalid datetime");
57 return date.number() * 1440 + hour * 60 + minute;
58 }
59 static Date readMonth(String text) {
60 if (!pattern(text, "####-##")) throw new CommandError("invalid month");
61 Date date = new Date(integer(text.substring(0, 4)), integer(text.substring(5)), 1);
62 if (date.year() < 2001 || date.year() > 2098 || date.month() < 1 || date.month() > 12) throw new CommandError("invalid month");
63 return date;
64 }

83 enum SpecialKind { LAST, LAST_WORKDAY, BEFORE_LAST, NEAREST, NTH_WEEKDAY, LAST_WEEKDAY }
84 record Special(SpecialKind kind, int value, int occurrence) {
85 @Override public String toString() {
86 return switch (kind) {
87 case LAST -> "L"; case LAST_WORKDAY -> "LW"; case BEFORE_LAST -> "L-" + value;
88 case NEAREST -> value + "W"; case NTH_WEEKDAY -> value + "#" + occurrence; case LAST_WEEKDAY -> value + "L";
89 };
90 }
91 String phrase() {
92 return switch (kind) {
93 case LAST -> "the last day"; case LAST_WORKDAY -> "the last weekday";
94 case BEFORE_LAST -> value + (value == 1 ? " day" : " days") + " before the last day";
95 case NEAREST -> "the weekday nearest day " + value;
96 case NTH_WEEKDAY -> "the " + List.of("1st", "2nd", "3rd", "4th", "5th").get(occurrence - 1) + " " + WEEKDAYS[value];
97 case LAST_WEEKDAY -> "the last " + WEEKDAYS[value];
98 };
99 }
100 }
101 static int readValue(String text, int index) {
102 List<String> names = switch (index) {
103 case 3 -> List.of("JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV", "DEC");
104 case 4 -> List.of("SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"); default -> List.of();
105 };
106 int named = names.indexOf(text);
107 if (named >= 0) return named + (index == 3 ? 1 : 0);
108 if (!digits(text, 1, 2)) throw new CommandError("invalid value");
109 int value = integer(text);
110 if (value < MINIMUM[index] || value > MAXIMUM[index]) throw new CommandError("out of range");
111 return value;
112 }
113 static int specialNumber(String text, int maximum, int width) {
114 if (!digits(text, 1, width)) throw new CommandError("invalid value");
115 int value = integer(text);
116 if (value < 1 || value > maximum) throw new CommandError("out of range");
117 return value;
118 }
119 static String wordList(List<String> parts, boolean either) {
120 if (parts.size() < 2) return String.join("", parts);
121 return String.join(", ", parts.subList(0, parts.size() - 1)) + (either ? " or " : " and ") + parts.getLast();
122 }
123 record Field(int index, List<Integer> values, List<Special> specials, boolean restricted) {
124 Field { values = List.copyOf(values); specials = List.copyOf(specials); }
125 boolean contains(int value) { return Collections.binarySearch(values, value) >= 0; }
126 static Field parse(String text, int index) {
127 TreeSet<Integer> values = new TreeSet<>();
128 List<Special> specials = new ArrayList<>();
129 try {
130 for (String item : text.split(",", -1)) {
131 if (item.isEmpty()) throw new CommandError("empty item");
132 Special special = null;
133 if (index == 2 && (item.startsWith("L") || item.endsWith("W"))) {
134 if (item.equals("L")) special = new Special(SpecialKind.LAST, 0, 0);
135 else if (item.equals("LW")) special = new Special(SpecialKind.LAST_WORKDAY, 0, 0);
136 else if (item.startsWith("L-")) special = new Special(SpecialKind.BEFORE_LAST, specialNumber(item.substring(2), 30, 2), 0);
137 else if (item.endsWith("W")) special = new Special(SpecialKind.NEAREST, specialNumber(item.substring(0, item.length() - 1), 31, 2), 0);
138 else throw new CommandError("invalid value");
139 } else if (index == 4 && item.contains("#")) {
140 int at = item.indexOf('#');
141 int weekday = readValue(item.substring(0, at), index);
142 int occurrence = specialNumber(item.substring(at + 1), 5, 1);
143 special = new Special(SpecialKind.NTH_WEEKDAY, weekday % 7, occurrence);
144 } else if (index == 4 && item.length() >= 2 && item.endsWith("L")) {
145 special = new Special(SpecialKind.LAST_WEEKDAY, readValue(item.substring(0, item.length() - 1), index) % 7, 0);
146 }
147 if (special != null) {
148 if (!specials.contains(special)) specials.add(special);
149 continue;
150 }
151 int slash = item.indexOf('/');
152 String base = slash < 0 ? item : item.substring(0, slash);
153 int first, last;
154 if (base.equals("*")) { first = MINIMUM[index]; last = MAXIMUM[index]; }
155 else {
156 int dash = base.indexOf('-');
157 if (dash >= 0) {
158 first = readValue(base.substring(0, dash), index);
159 last = readValue(base.substring(dash + 1), index);
160 if (first > last) throw new CommandError("invalid range");
161 } else { first = readValue(base, index); last = slash < 0 ? first : MAXIMUM[index]; }
162 }
163 int step = 1;
164 if (slash >= 0) {
165 String part = item.substring(slash + 1);
166 if (!digits(part, 1, 2) || integer(part) == 0) throw new CommandError("invalid step");
167 step = integer(part);
168 }
169 for (int value = first; value <= last; value += step) values.add(index == 4 ? value % 7 : value);
170 }
171 } catch (CommandError error) { throw new CommandError(FIELD_NAMES[index] + " " + error.getMessage()); }
172 return new Field(index, new ArrayList<>(values), specials, text.charAt(0) != '*');
173 }
174 int stepped() {
175 if (values.size() < 2 || values.getFirst() != 0 || values.get(1) < 2) return 0;
176 int step = values.get(1);
177 if (values.size() != MAXIMUM[index] / step + 1) return 0;
178 for (int i = 0; i < values.size(); i++) if (values.get(i) != i * step) return 0;
179 return step;
180 }
181 String valueName(int value) { return index == 3 ? MONTHS[value - 1] : index == 4 ? WEEKDAYS[value] : Integer.toString(value); }
182 String valueList(boolean either) {
183 List<String> parts = new ArrayList<>();
184 for (int first = 0; first < values.size();) {
185 int last = first;
186 while (last + 1 < values.size() && values.get(last + 1) == values.get(last) + 1) last++;
187 if (last - first >= 2) parts.add(valueName(values.get(first)) + " through " + valueName(values.get(last)));
188 else for (int i = first; i <= last; i++) parts.add(valueName(values.get(i)));
189 first = last + 1;
190 }
191 return wordList(parts, either);
192 }
193 String dayPhrase(boolean either) {
194 List<String> parts = new ArrayList<>();
195 if (!values.isEmpty()) parts.add(index == 4 ? valueList(either) : (values.size() == 1 ? "day " : "days ") + valueList(false));
196 for (Special special : specials) parts.add(special.phrase());
197 return wordList(parts, either) + (index == 2 || !specials.isEmpty() ? " of the month" : "");
198 }
199 }
200 record DayMatch(boolean dom, boolean dow, boolean condition) {}
201 record Cron(List<Field> fields, List<Integer> daily, String text, String expanded) {
202 Cron { fields = List.copyOf(fields); daily = List.copyOf(daily); }
203 static Cron parse(String[] input) {
204 String text = String.join(" ", input);
205 String[] words = input;
206 if (words.length == 1 && words[0].startsWith("@")) {
207 String expansion = switch (words[0]) {
208 case "@yearly", "@annually" -> "0 0 1 1 *"; case "@monthly" -> "0 0 1 * *"; case "@weekly" -> "0 0 * * 0";
209 case "@daily", "@midnight" -> "0 0 * * *"; case "@hourly" -> "0 * * * *"; default -> throw new CommandError("unknown macro");
210 };
211 words = expansion.split(" ");
212 }
213 if (words.length != 5) throw new CommandError("field count");
214 List<Field> fields = new ArrayList<>();
215 for (int i = 0; i < 5; i++) fields.add(Field.parse(words[i], i));
216 List<Integer> daily = new ArrayList<>();
217 for (int hour : fields.get(1).values()) for (int minute : fields.get(0).values()) daily.add(hour * 60 + minute);
218 return new Cron(fields, daily, text, String.join(" ", words));
219 }
220 boolean orRule() { return fields.get(2).restricted() && fields.get(4).restricted(); }
221 static int nearest(Date date, int day) {
222 int weekday = (date.number() - date.day() + day + 6) % 7;
223 return weekday == 6 ? day == 1 ? 3 : day - 1 : weekday == 0 ? day == monthLength(date.year(), date.month()) ? day - 2 : day + 1 : day;
224 }
225 DayMatch matchDay(Date date) {
226 int last = monthLength(date.year(), date.month()), day = date.day(), weekday = date.weekday();
227 boolean dom = fields.get(2).contains(day), dow = fields.get(4).contains(weekday);
228 for (Special special : fields.get(2).specials()) {
229 dom |= switch (special.kind()) {
230 case LAST -> day == last; case LAST_WORKDAY -> day == nearest(date, last); case BEFORE_LAST -> day == last - special.value();
231 case NEAREST -> special.value() <= last && day == nearest(date, special.value()); default -> false;
232 };
233 }
234 for (Special special : fields.get(4).specials()) {
235 dow |= weekday == special.value() && (special.kind() == SpecialKind.NTH_WEEKDAY ? (day - 1) / 7 + 1 == special.occurrence() : day + 7 > last);
236 }
237 return new DayMatch(dom, dow, orRule() ? dom || dow : dom && dow);
238 }
239 boolean scheduled(Date date) { return fields.get(3).contains(date.month()) && matchDay(date).condition(); }
240 List<Integer> fires(int low, int high, Set<Integer> holidays, boolean reverse, int limit) {
241 List<Integer> result = new ArrayList<>();
242 int first = low / 1440, last = high / 1440, step = reverse ? -1 : 1;
243 for (int day = reverse ? last : first; day >= first && day <= last; day += step) {
244 if (holidays.contains(day) || !scheduled(Date.fromNumber(day))) continue;
245 for (int i = reverse ? daily.size() - 1 : 0; i >= 0 && i < daily.size(); i += step) {
246 int time = day * 1440 + daily.get(i);
247 if (time < low || time > high) continue;
248 result.add(time);
249 if (limit > 0 && result.size() == limit) return result;
250 }
251 }
252 return result;
253 }
254 String description() {
255 Field minute = fields.get(0), hour = fields.get(1), dom = fields.get(2), month = fields.get(3), dow = fields.get(4);
256 String text;
257 if (daily.size() <= 4) text = "At " + wordList(daily.stream().map(t -> format("%02d:%02d", t / 60, t % 60)).toList(), false);
258 else {
259 int ms = minute.stepped(), hs = hour.stepped();
260 text = minute.values().size() == 60 ? "Every minute" : ms > 0 ? "Every " + ms + " minutes"
261 : "At " + (minute.values().size() == 1 ? "minute " : "minutes ") + minute.valueList(false);
262 if (hour.values().size() == 24) { if (text.startsWith("At minute")) text += " of every hour"; }
263 else if (hs > 0) text += ", every " + hs + " hours";
264 else if (hour.values().getLast() - hour.values().getFirst() + 1 == hour.values().size()) text += format(", between %02d:00 and %02d:59", hour.values().getFirst(), hour.values().getLast());
265 else text += ", during hours " + hour.valueList(false);
266 }
267 boolean dFull = dom.values().size() == 31, wFull = dow.values().size() == 7;
268 if (orRule()) { if (!dFull && !wFull) text += ", on " + dom.dayPhrase(false) + " or on " + dow.dayPhrase(false); }
269 else if (!dFull && !wFull) text += ", on " + dom.dayPhrase(false) + " if it is " + dow.dayPhrase(true);
270 else if (!dFull) text += ", on " + dom.dayPhrase(false);
271 else if (!wFull) text += ", on " + dow.dayPhrase(false);
272 if (month.values().size() != 12) text += ", in " + month.valueList(false);
273 return text;
274 }

286 static final class Scheduler {
287 final TreeMap<String, Job> jobs = new TreeMap<>();
288 final TreeSet<Integer> holidays = new TreeSet<>();
289 Job job(String id) {
290 Job result = jobs.get(id);
291 if (result == null) throw new CommandError("unknown job");
292 return result;
293 }
294 void newId(String id) {
295 if (id.isEmpty() || id.length() > 16 || id.charAt(0) < 'a' || id.charAt(0) > 'z' || !id.chars().allMatch(c -> c >= 'a' && c <= 'z' || c >= '0' && c <= '9' || c == '_')) throw new CommandError("invalid id");
296 if (jobs.containsKey(id)) throw new CommandError("duplicate job");
297 }
298 List<NamedJob> enabled() { return jobs.entrySet().stream().filter(e -> e.getValue().enabled).map(e -> new NamedJob(e.getKey(), e.getValue())).toList(); }
299 List<String> timeline(Window window, boolean conflicts) {
300 List<NamedJob> enabled = enabled();
301 long[] masks = new long[window.high() - window.low() + 1];
302 for (int i = 0; i < enabled.size(); i++) {
303 for (int time : enabled.get(i).job().cron.fires(window.low(), window.high(), holidays, false, 0)) masks[time - window.low()] |= 1L << i;
304 }
305 int total = 0, peak = 0;
306 for (long mask : masks) {
307 int count = Long.bitCount(mask);
308 if (!conflicts) total += count;
309 else if (count >= 2) { total++; peak = Math.max(peak, count); }
310 }
311 List<String> rows = new ArrayList<>();
312 rows.add(conflicts ? "CONFLICTS " + total + " peak " + peak : "TIMELINE " + total);
313 for (int i = 0; i < masks.length && rows.size() < 101; i++) {
314 long mask = masks[i];
315 if (conflicts) {
316 if (Long.bitCount(mask) < 2) continue;
317 List<String> ids = new ArrayList<>();
318 for (int j = 0; j < enabled.size(); j++) if ((mask & (1L << j)) != 0) ids.add(enabled.get(j).id());
319 rows.add(datetimeText(window.low() + i) + " " + ids.size() + " " + String.join(" ", ids));
320 } else {
321 for (int j = 0; j < enabled.size() && rows.size() < 101; j++) if ((mask & (1L << j)) != 0) rows.add(datetimeText(window.low() + i) + " " + enabled.get(j).id());
322 }
323 }
324 if (total > 100) rows.add("... " + (total - 100) + " more");
325 return rows;
326 }
327 List<String> simulate(Window window) {
328 List<NamedJob> enabled = enabled();
329 int[] events = new int[window.high() - window.low() + 1441];
330 List<String> rows = new ArrayList<>();
331 for (NamedJob entry : enabled) {
332 Job job = entry.job();
333 int started = 0, skipped = 0, replaced = 0, busy = 0, lastEnd = -1;
334 for (int time : job.cron.fires(window.low(), window.high(), holidays, false, 0)) {
335 if (lastEnd > time && job.policy == Policy.FORBID) { skipped++; continue; }
336 if (lastEnd > time && job.policy == Policy.REPLACE) {
337 events[lastEnd - window.low()]++; events[time - window.low()]--;
338 busy -= lastEnd - time; replaced++;
339 }
340 lastEnd = time + job.duration;
341 events[time - window.low()]++; events[lastEnd - window.low()]--;
342 started++; busy += job.duration;
343 }
344 rows.add(format("%s %s started %d skipped %d replaced %d busy %d", entry.id(), job.policy, started, skipped, replaced, busy));
345 }
346 int running = 0, peak = 0, at = 0;
347 for (int i = 0; i < events.length; i++) { running += events[i]; if (running > peak) { peak = running; at = window.low() + i; } }
348 rows.addFirst("SIMULATE " + enabled.size() + " peak " + peak + (peak == 0 ? "" : " at " + datetimeText(at)));
349 return rows;
350 }
351 List<String> execute(String[] words) {
352 String command = words[0];
353 int expected = switch (command) {
354 case "JOB" -> -3; case "VALIDATE" -> -2; case "LIST", "HOLIDAYS" -> 1;
355 case "REMOVE", "ENABLE", "DISABLE", "DESCRIBE", "WEEK", "EXCLUDE", "INCLUDE" -> 2;
356 case "RENAME", "EXPAND", "CALENDAR" -> 3; case "RUNTIME", "MATCH", "GAP" -> 4;
357 case "NEXT", "PREV", "TIMELINE", "CONFLICTS", "SIMULATE" -> 5; case "COUNT" -> 6; case "OVERLAP" -> 7;
358 default -> throw new CommandError("unknown command");
359 };
360 if (expected < 0 ? words.length < -expected : words.length != expected) throw new CommandError("wrong arguments");
361 Job job = switch (command) {
362 case "REMOVE", "RENAME", "ENABLE", "DISABLE", "RUNTIME", "EXPAND", "DESCRIBE", "MATCH", "NEXT", "PREV", "COUNT", "GAP", "OVERLAP", "CALENDAR" -> job(words[1]);
363 default -> null;
364 };
365 switch (command) {
366 case "JOB" -> {
367 newId(words[1]); Cron cron = Cron.parse(Arrays.copyOfRange(words, 2, words.length));
368 jobs.put(words[1], new Job(cron)); return List.of("JOB " + words[1] + " added");
369 }
370 case "VALIDATE" -> {
371 Cron cron = Cron.parse(Arrays.copyOfRange(words, 1, words.length));
372 return List.of("VALID " + cron.expanded() + " " + (cron.orRule() ? "OR" : "AND") + " " + cron.daily().size());
373 }
374 case "REMOVE" -> { jobs.remove(words[1]); return List.of("REMOVE " + words[1]); }
375 case "RENAME" -> { newId(words[2]); jobs.remove(words[1]); jobs.put(words[2], job); return List.of("RENAME " + words[1] + " " + words[2]); }
376 case "ENABLE", "DISABLE" -> {
377 boolean desired = command.equals("ENABLE");
378 if (job.enabled == desired) throw new CommandError(desired ? "already enabled" : "already disabled");
379 job.enabled = desired; return List.of(command + " " + words[1]);
380 }
381 case "RUNTIME" -> {
382 int duration = positive(words[2], 4, 1440, "invalid duration");
383 Policy policy = switch (words[3]) { case "ALLOW" -> Policy.ALLOW; case "FORBID" -> Policy.FORBID; case "REPLACE" -> Policy.REPLACE; default -> throw new CommandError("invalid policy"); };
384 job.duration = duration; job.policy = policy;
385 return List.of("RUNTIME " + words[1] + " " + duration + " " + policy);
386 }
387 case "LIST" -> {
388 List<String> rows = new ArrayList<>(); rows.add("LIST " + jobs.size());
389 jobs.forEach((id, item) -> rows.add(id + " " + (item.enabled ? "on" : "off") + " " + item.duration + " " + item.policy + " " + item.cron.text()));
390 return rows;
391 }
392 case "EXPAND" -> {
393 int index = Arrays.asList(FIELD_NAMES).indexOf(words[2]);
394 if (index < 0) throw new CommandError("invalid field");
395 Field field = job.cron.fields().get(index);
396 List<String> parts = new ArrayList<>(); parts.add("EXPAND " + words[1] + " " + words[2] + " " + field.values().size());
397 for (int value : field.values()) parts.add(Integer.toString(value));
398 for (Special item : field.specials()) parts.add(item.toString());
399 return List.of(String.join(" ", parts));
400 }
401 case "DESCRIBE" -> { return List.of("DESCRIBE " + words[1] + " " + job.cron.description()); }
402 case "MATCH" -> {
403 int time = readDatetime(words, 2); Date date = Date.fromNumber(time / 1440); DayMatch match = job.cron.matchDay(date);
404 List<String> reasons = new ArrayList<>();
405 if (!job.cron.fields().get(0).contains(time % 60)) reasons.add("minute");
406 if (!job.cron.fields().get(1).contains(time % 1440 / 60)) reasons.add("hour");
407 if (!match.condition() && !match.dom()) reasons.add("day-of-month");
408 if (!job.cron.fields().get(3).contains(date.month())) reasons.add("month");
409 if (!match.condition() && !match.dow()) reasons.add("day-of-week");
410 if (holidays.contains(time / 1440)) reasons.add("excluded");
411 return List.of("MATCH " + words[1] + " " + datetimeText(time) + " " + (reasons.isEmpty() ? "yes" : "no " + String.join(" ", reasons)));
412 }
413 case "NEXT", "PREV" -> {
414 int base = readDatetime(words, 2), limit = positive(words[4], 2, 50, "invalid count"); boolean reverse = command.equals("PREV");
415 List<Integer> times = job.cron.fires(reverse ? base - 527040 : base + 1, reverse ? base - 1 : base + 527040, holidays, reverse, limit);
416 List<String> rows = new ArrayList<>(); rows.add(command + " " + words[1] + " " + times.size());
417 for (int time : times) rows.add(datetimeText(time) + (reverse ? " -" : " +") + span(Math.abs(time - base)));
418 if (times.size() < limit) rows.add("NONE"); return rows;
419 }
420 case "COUNT" -> {
421 Window window = Window.read(words, 2); List<Integer> times = job.cron.fires(window.low(), window.high(), holidays, false, 0);
422 int days = 0, previous = -1;
423 for (int time : times) if (time / 1440 != previous) { previous = time / 1440; days++; }
424 return List.of("COUNT " + words[1] + " runs " + times.size() + " days " + days);
425 }
426 case "GAP" -> {
427 int base = readDatetime(words, 2); List<Integer> times = job.cron.fires(base, base + 44640, holidays, false, 0);
428 if (times.size() < 2) return List.of("GAP " + words[1] + " runs " + times.size() + " none");
429 TreeMap<Integer, Integer> histogram = new TreeMap<>(); int longest = 0, first = 0;
430 for (int i = 0; i + 1 < times.size(); i++) {
431 int gap = times.get(i + 1) - times.get(i); histogram.merge(gap, 1, Integer::sum);
432 if (gap > longest) { longest = gap; first = i; }
433 }
434 List<String> rows = new ArrayList<>();
435 rows.add(format("GAP %s runs %d shortest %s longest %s from %s to %s", words[1], times.size(), span(histogram.firstKey()), span(longest), datetimeText(times.get(first)), datetimeText(times.get(first + 1))));
436 histogram.forEach((gap, count) -> rows.add(span(gap) + " " + count)); return rows;
437 }
438 case "OVERLAP" -> {
439 Job otherJob = job(words[2]); Window window = Window.read(words, 3);
440 Set<Integer> other = new HashSet<>(otherJob.cron.fires(window.low(), window.high(), holidays, false, 0));
441 List<Integer> times = job.cron.fires(window.low(), window.high(), holidays, false, 0).stream().filter(other::contains).toList();
442 return List.of("OVERLAP " + words[1] + " " + words[2] + " " + times.size() + (times.isEmpty() ? "" : " first " + datetimeText(times.getFirst()) + " last " + datetimeText(times.getLast())));
443 }
444 case "TIMELINE", "CONFLICTS", "SIMULATE" -> {
445 Window window = Window.read(words, 1); return command.equals("SIMULATE") ? simulate(window) : timeline(window, command.equals("CONFLICTS"));
446 }
447 case "WEEK" -> {
448 int first = readDate(words[1], "invalid date").number(); int[][] counts = new int[7][24]; int total = 0;
449 for (NamedJob entry : enabled()) for (int time : entry.job().cron.fires(first * 1440, (first + 7) * 1440 - 1, holidays, false, 0)) { counts[time / 1440 - first][time % 1440 / 60]++; total++; }
450 List<String> rows = new ArrayList<>(); rows.add("WEEK " + words[1] + " " + total);
451 for (int d = 0; d < 7; d++) {
452 StringBuilder heat = new StringBuilder(); int dayTotal = 0;
453 for (int count : counts[d]) { heat.append(count == 0 ? '.' : count >= 10 ? '+' : (char) ('0' + count)); dayTotal += count; }
454 rows.add(Date.fromNumber(first + d) + " " + heat + " " + dayTotal);
455 }
456 return rows;
457 }
458 case "CALENDAR" -> {
459 Date first = readMonth(words[2]); List<String> cells = new ArrayList<>(Collections.nCopies(first.weekday(), "...")); int days = 0;
460 for (int day = 1; day <= monthLength(first.year(), first.month()); day++) {
461 Date date = new Date(first.year(), first.month(), day);
462 char mark = !job.cron.scheduled(date) ? '-' : holidays.contains(date.number()) ? 'x' : '*';
463 if (mark == '*') days++;
464 cells.add(format("%02d%c", day, mark));
465 }
466 while (cells.size() % 7 != 0) cells.add("...");
467 List<String> rows = new ArrayList<>(List.of("CALENDAR " + words[1] + " " + words[2] + " days " + days + " runs " + days * job.cron.daily().size(), "Sun Mon Tue Wed Thu Fri Sat"));
468 for (int i = 0; i < cells.size(); i += 7) rows.add(String.join(" ", cells.subList(i, i + 7))); return rows;
469 }
470 case "EXCLUDE", "INCLUDE" -> {
471 Date date = readDate(words[1], "invalid date");
472 if (command.equals("EXCLUDE")) { if (!holidays.add(date.number())) throw new CommandError("already excluded"); }
473 else if (!holidays.remove(date.number())) throw new CommandError("not excluded");
474 return List.of(command + " " + date + " holidays " + holidays.size());
475 }
476 case "HOLIDAYS" -> {
477 List<String> rows = new ArrayList<>(); rows.add("HOLIDAYS " + holidays.size());
478 for (int day : holidays) rows.add(Date.fromNumber(day).toString()); return rows;
479 }
480 }
481 throw new IllegalStateException();
482 }
483 }
