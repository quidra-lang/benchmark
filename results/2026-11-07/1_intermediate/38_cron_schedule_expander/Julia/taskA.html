27ascii_digits(text::AbstractString, low::Int, high::Int) = low <= length(text) <= high && all(c -> '0' <= c <= '9', text)
28leap(year::Int) = year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
29month_length(year::Int, month::Int) = (31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31)[month] + (month == 2 && leap(year))
30leaps(year::Int) = year ÷ 4 - year ÷ 100 + year ÷ 400

32struct CivilDate
33 year::Int
34 month::Int
35 day::Int
36end
37function day_number(date::CivilDate)
38 result = 365 * (date.year - 2000) + leaps(date.year - 1) - leaps(1999) + date.day - 1
39 for month in 1:date.month-1
40 result += month_length(date.year, month)
41 end
42 result
43end
44const CALENDAR = vcat([CivilDate(year, month, day) for year in 2000:2099 for month in 1:12
45 for day in 1:month_length(year, month)], [CivilDate(2100, 1, 1)])
46weekday(day::Int) = (day + 6) % 7
47function plain_date(day::Int)
48 date = CALENDAR[day + 1]
49 @sprintf("%04d-%02d-%02d", date.year, date.month, date.day)
50end
51date_text(day::Int) = "$(plain_date(day)) $(WEEKDAY_NAMES[weekday(day) + 1][1:3])"
52clock_text(minute::Int) = @sprintf("%02d:%02d", minute ÷ 60, minute % 60)
53datetime_text(minute::Int) = "$(plain_date(minute ÷ 1440)) $(clock_text(minute % 1440)) $(WEEKDAY_NAMES[weekday(minute ÷ 1440) + 1][1:3])"
54span_text(minutes::Int) = @sprintf("%dd%02dh%02dm", minutes ÷ 1440, minutes % 1440 ÷ 60, minutes % 60)

56function read_date(text::AbstractString, reason::String = "invalid date")
57 occursin(r"^[0-9]{4}-[0-9]{2}-[0-9]{2}$", text) || fail(reason)
58 year, month, day = parse(Int, text[1:4]), parse(Int, text[6:7]), parse(Int, text[9:10])
59 2001 <= year <= 2098 && 1 <= month <= 12 && 1 <= day <= month_length(year, month) || fail(reason)
60 day_number(CivilDate(year, month, day))
61end
62function read_datetime(words, position::Int)
63 day = read_date(words[position], "invalid datetime")
64 text = words[position + 1]
65 occursin(r"^[0-9]{2}:[0-9]{2}$", text) || fail("invalid datetime")
66 hour, minute = parse(Int, text[1:2]), parse(Int, text[4:5])
67 hour <= 23 && minute <= 59 || fail("invalid datetime")
68 day * 1440 + hour * 60 + minute
69end
70function read_month(text::AbstractString)
71 occursin(r"^[0-9]{4}-[0-9]{2}$", text) || fail("invalid month")
72 year, month = parse(Int, text[1:4]), parse(Int, text[6:7])
73 2001 <= year <= 2098 && 1 <= month <= 12 || fail("invalid month")
74 day_number(CivilDate(year, month, 1))
75end
76function positive(text::AbstractString, width::Int, maximum::Int, reason::String)
77 ascii_digits(text, 1, width) && first(text) != '0' || fail(reason)
78 value = parse(Int, text)
79 value <= maximum || fail(reason)
80 value
81end

96 value::Int
97 occurrence::Int
98end
99Special(kind::SpecialKind, value::Int = 0) = Special(kind, value, 0)
100function special_text(special::Special)
101 special.kind == LAST && return "L"
102 special.kind == LAST_WORKDAY && return "LW"
103 special.kind == BEFORE_LAST && return "L-$(special.value)"
104 special.kind == NEAREST && return "$(special.value)W"
105 special.kind == NTH_WEEKDAY && return "$(special.value)#$(special.occurrence)"
106 "$(special.value)L"
107end
108function special_phrase(special::Special)
109 special.kind == LAST && return "the last day"
110 special.kind == LAST_WORKDAY && return "the last weekday"
111 special.kind == BEFORE_LAST && return "$(special.value) $(special.value == 1 ? "day" : "days") before the last day"
112 special.kind == NEAREST && return "the weekday nearest day $(special.value)"
113 name = WEEKDAY_NAMES[special.value + 1]
114 special.kind == LAST_WEEKDAY && return "the last $name"
115 ordinal = ("1st", "2nd", "3rd", "4th", "5th")[special.occurrence]
116 "the $ordinal $name"
117end
118function field_value(text::AbstractString, kind::FieldKind)
119 names = kind == MONTH ? MONTH_CODES : kind == DAY_OF_WEEK ? WEEKDAY_CODES : ()
120 index = findfirst(==(text), names)
121 isnothing(index) || return index - (kind == DAY_OF_WEEK)
122 ascii_digits(text, 1, 2) || field_error(kind, "invalid value")
123 value = parse(Int, text)
124 MINIMUM[Int(kind)] <= value <= MAXIMUM[Int(kind)] || field_error(kind, "out of range")
125 value
126end
127function special_number(text::AbstractString, kind::FieldKind, width::Int, maximum::Int)
128 ascii_digits(text, 1, width) || field_error(kind, "invalid value")
129 value = parse(Int, text)
130 1 <= value <= maximum || field_error(kind, "out of range")
131 value
132end
133struct CronField
134 kind::FieldKind
135 values::Vector{Int}
136 specials::Vector{Special}
137 restricted::Bool
138end
139function parse_field(text::AbstractString, kind::FieldKind)
140 values, specials = Set{Int}(), Special[]
141 for item in split(text, ','; keepempty = true)
142 isempty(item) && field_error(kind, "empty item")
143 special = nothing
144 if kind == DAY_OF_MONTH && (startswith(item, 'L') || endswith(item, 'W'))
145 if item == "L"
146 special = Special(LAST)
147 elseif item == "LW"
148 special = Special(LAST_WORKDAY)
149 elseif startswith(item, "L-")
150 special = Special(BEFORE_LAST, special_number(item[3:end], kind, 2, 30))
151 elseif endswith(item, 'W')
152 special = Special(NEAREST, special_number(item[1:end-1], kind, 2, 31))
153 else
154 field_error(kind, "invalid value")
155 end
156 elseif kind == DAY_OF_WEEK && occursin('#', item)
157 at = findfirst(==('#'), item)
158 value = field_value(item[1:at-1], kind)
159 occurrence = special_number(item[at+1:end], kind, 1, 5)
160 special = Special(NTH_WEEKDAY, value % 7, occurrence)
161 elseif kind == DAY_OF_WEEK && length(item) >= 2 && endswith(item, 'L')
162 special = Special(LAST_WEEKDAY, field_value(item[1:end-1], kind) % 7)
163 end
164 if !isnothing(special)
165 special in specials || push!(specials, special)
166 continue
167 end
168 slash = findfirst(==('/'), item)
169 base = isnothing(slash) ? item : item[1:slash-1]
170 dash = findfirst(==('-'), base)
171 if base == "*"
172 low, high = MINIMUM[Int(kind)], MAXIMUM[Int(kind)]
173 elseif !isnothing(dash)
174 low = field_value(base[1:dash-1], kind)
175 high = field_value(base[dash+1:end], kind)
176 low <= high || field_error(kind, "invalid range")
177 else
178 low = field_value(base, kind)
179 high = isnothing(slash) ? low : MAXIMUM[Int(kind)]
180 end
181 step = 1
182 if !isnothing(slash)
183 part = item[slash+1:end]
184 ascii_digits(part, 1, 2) || field_error(kind, "invalid step")
185 step = parse(Int, part)
186 step >= 1 || field_error(kind, "invalid step")
187 end
188 for value in low:step:high
189 push!(values, kind == DAY_OF_WEEK ? value % 7 : value)
190 end
191 end
192 CronField(kind, sort!(collect(values)), specials, !startswith(text, '*'))
193end
194struct Cron
195 fields::NTuple{5,CronField}
196 times::Vector{Int}
197 text::String
198 expanded::String
199end
200function parse_cron(input)
201 text = join(input, " ")
202 words = input
203 if length(words) == 1 && startswith(words[1], '@')
204 haskey(MACROS, words[1]) || fail("unknown macro")
205 words = split(MACROS[words[1]], ' ')
206 end
207 length(words) == 5 || fail("field count")
208 fields = CronField[]
209 for (index, token) in enumerate(words)
210 push!(fields, parse_field(token, FieldKind(index)))
211 end
212 times = [60hour + minute for hour in fields[2].values for minute in fields[1].values]
213 Cron(Tuple(fields), times, text, join(words, " "))
214end
215or_rule(cron::Cron) = cron.fields[3].restricted && cron.fields[5].restricted
216function nearest_workday(first::Int, day::Int, last::Int)
217 w = weekday(first + day - 1)
218 w == 6 && return day == 1 ? 3 : day - 1
219 w == 0 && return day == last ? day - 2 : day + 1
220 day
221end
222struct DayStatus
223 month::Bool
224 dom::Bool
225 dow::Bool
226 condition::Bool
227end
228function day_status(cron::Cron, day::Int)
229 date = CALENDAR[day + 1]
230 d, last, w = date.day, month_length(date.year, date.month), weekday(day)
231 first = day - d + 1
232 dom, dow = d in cron.fields[3].values, w in cron.fields[5].values
233 for special in cron.fields[3].specials
234 target = special.kind == LAST ? last : special.kind == LAST_WORKDAY ? nearest_workday(first, last, last) :
235 special.kind == BEFORE_LAST ? last - special.value : special.value <= last ? nearest_workday(first, special.value, last) : 0
236 dom |= d == target
237 end
238 for special in cron.fields[5].specials
239 dow |= w == special.value && (special.kind == NTH_WEEKDAY ? (d - 1) ÷ 7 + 1 == special.occurrence : d + 7 > last)
240 end
241 DayStatus(date.month in cron.fields[4].values, dom, dow, or_rule(cron) ? dom || dow : dom && dow)
242end
243function scheduled_day(cron::Cron, day::Int)
244 status = day_status(cron, day)
245 status.month && status.condition
246end
247function fire_times(cron::Cron, low::Int, high::Int, holidays::Set{Int}; backward::Bool = false, limit::Int = 0)
248 first_day, last_day = low ÷ 1440, high ÷ 1440
249 days = backward ? (last_day:-1:first_day) : (first_day:last_day)
250 indices = backward ? (length(cron.times):-1:1) : eachindex(cron.times)
251 result = Int[]
252 for day in days
253 (day in holidays || !scheduled_day(cron, day)) && continue
254 for index in indices
255 time = 1440day + cron.times[index]
256 low <= time <= high || continue
257 push!(result, time)
258 limit > 0 && length(result) == limit && return result
259 end
260 end
261 result
262end

264function word_list(parts::Vector{String}, either::Bool = false)
265 length(parts) < 2 && return join(parts)
266 join(parts[1:end-1], ", ") * (either ? " or " : " and ") * parts[end]
267end
268function value_name(kind::FieldKind, value::Int)
269 kind == MONTH ? MONTH_NAMES[value] : kind == DAY_OF_WEEK ? WEEKDAY_NAMES[value + 1] : string(value)
270end
271function value_list(field::CronField, either::Bool = false)
272 values, parts = field.values, String[]
273 first = 1
274 while first <= length(values)
275 last = first
276 while last < length(values) && values[last + 1] == values[last] + 1
277 last += 1
278 end
279 if last - first >= 2
280 push!(parts, "$(value_name(field.kind, values[first])) through $(value_name(field.kind, values[last]))")
281 else
282 append!(parts, [value_name(field.kind, values[index]) for index in first:last])
283 end
284 first = last + 1
285 end
286 word_list(parts, either)
287end
288function stepped(field::CronField)
289 values = field.values
290 (length(values) < 2 || values[1] != 0 || values[2] < 2) && return 0
291 step = values[2]
292 values == collect(0:step:MAXIMUM[Int(field.kind)]) ? step : 0
293end
294function day_phrase(field::CronField, either::Bool = false)
295 parts = String[]
296 if !isempty(field.values)
297 push!(parts, field.kind == DAY_OF_WEEK ? value_list(field, either) :
298 "$(length(field.values) == 1 ? "day" : "days") $(value_list(field))")
299 end
300 append!(parts, special_phrase.(field.specials))
301 word_list(parts, either) * (field.kind == DAY_OF_MONTH || !isempty(field.specials) ? " of the month" : "")
302end
303function description(cron::Cron)
304 minute, hour, dom, month, dow = cron.fields
305 if length(cron.times) <= 4
306 text = "At " * word_list(clock_text.(cron.times))
307 else
308 ms, hs = stepped(minute), stepped(hour)
309 text = length(minute.values) == 60 ? "Every minute" : ms > 0 ? "Every $ms minutes" :
310 length(minute.values) == 1 ? "At minute $(minute.values[1])" : "At minutes $(value_list(minute))"
311 if length(hour.values) == 24
312 startswith(text, "At minute") && (text *= " of every hour")
313 elseif hs > 0
314 text *= ", every $hs hours"
315 elseif hour.values[end] - hour.values[1] + 1 == length(hour.values)
316 text *= @sprintf(", between %02d:00 and %02d:59", hour.values[1], hour.values[end])
317 else
318 text *= ", during hours $(value_list(hour))"
319 end
320 end
321 dfull, ufull = length(dom.values) == 31, length(dow.values) == 7
322 if or_rule(cron)
323 !dfull && !ufull && (text *= ", on $(day_phrase(dom)) or on $(day_phrase(dow))")
324 elseif !dfull && !ufull
325 text *= ", on $(day_phrase(dom)) if it is $(day_phrase(dow, true))"
326 elseif !dfull
327 text *= ", on $(day_phrase(dom))"
328 elseif !ufull
329 text *= ", on $(day_phrase(dow))"
330 end
331 length(month.values) != 12 && (text *= ", in $(value_list(month))")
332 text
333end

350function require_new_id(state::Scheduler, id::AbstractString)
351 occursin(r"^[a-z][a-z0-9_]{0,15}$", id) || fail("invalid id")
352 haskey(state.jobs, id) && fail("duplicate job")
353end
354ordered_ids(state::Scheduler) = sort!(collect(keys(state.jobs)))
355enabled_jobs(state::Scheduler) = [id => state.jobs[id] for id in ordered_ids(state) if state.jobs[id].enabled]
356function combined_report(state::Scheduler, window::Window, conflicts::Bool)
357 jobs = enabled_jobs(state)
358 incidence = falses(length(jobs), window.high - window.low + 1)
359 for (index, (_, job)) in enumerate(jobs)
360 for time in fire_times(job.cron, window.low, window.high, state.holidays)
361 incidence[index, time - window.low + 1] = true
362 end
363 end
364 if conflicts
365 counts = vec(sum(incidence; dims = 1))
366 positions = findall(>=(2), counts)
367 total = length(positions)
368 peak = isempty(positions) ? 0 : maximum(counts[positions])
369 rows = ["CONFLICTS $total peak $peak"]
370 for position in Iterators.take(positions, 100)
371 ids = [jobs[index].first for index in eachindex(jobs) if incidence[index, position]]
372 push!(rows, "$(datetime_text(window.low + position - 1)) $(counts[position]) $(join(ids, " "))")
373 end
374 else
375 pairs = findall(incidence)
376 total = length(pairs)
377 rows = ["TIMELINE $total"]
378 for pair in Iterators.take(pairs, 100)
379 push!(rows, "$(datetime_text(window.low + pair[2] - 1)) $(jobs[pair[1]].first)")
380 end
381 end
382 total > 100 && push!(rows, "... $(total - 100) more")
383 rows
384end
385function week_report(state::Scheduler, first::Int, original::AbstractString)
386 counts = zeros(Int, 24, 7)
387 for offset in 0:6
388 day = first + offset
389 day in state.holidays && continue
390 for (_, job) in enabled_jobs(state)
391 scheduled_day(job.cron, day) || continue
392 for hour in job.cron.fields[2].values
393 counts[hour + 1, offset + 1] += length(job.cron.fields[1].values)
394 end
395 end
396 end
397 rows = ["WEEK $original $(sum(counts))"]
398 for offset in 0:6
399 hours = @view counts[:, offset + 1]
400 heat = join([count == 0 ? "." : count < 10 ? string(count) : "+" for count in hours])
401 push!(rows, "$(date_text(first + offset)) $heat $(sum(hours))")
402 end
403 rows
404end
405function calendar_report(state::Scheduler, job::SchedulerJob, first::Int, original::AbstractString, id::AbstractString)
406 date = CALENDAR[first + 1]
407 marks = [!scheduled_day(job.cron, first + day - 1) ? '-' : first + day - 1 in state.holidays ? 'x' : '*'
408 for day in 1:month_length(date.year, date.month)]
409 days = count(==('*'), marks)
410 cells = fill("...", weekday(first))
411 append!(cells, [@sprintf("%02d%c", day, mark) for (day, mark) in enumerate(marks)])
412 append!(cells, fill("...", mod(-length(cells), 7)))
413 rows = ["CALENDAR $id $original days $days runs $(days * length(job.cron.times))", "Sun Mon Tue Wed Thu Fri Sat"]
414 append!(rows, [join(week, " ") for week in Iterators.partition(cells, 7)])
415 rows
416end
417function simulation_report(state::Scheduler, window::Window)
418 jobs = enabled_jobs(state)
419 events = zeros(Int, window.high - window.low + 1441)
420 details = String[]
421 for (id, job) in jobs
422 started = skipped = replaced = busy = 0
423 last_end = nothing
424 for time in fire_times(job.cron, window.low, window.high, state.holidays)
425 if !isnothing(last_end) && last_end > time
426 if job.policy == FORBID
427 skipped += 1
428 continue
429 elseif job.policy == REPLACE
430 events[last_end - window.low + 1] += 1
431 events[time - window.low + 1] -= 1
432 busy -= last_end - time
433 replaced += 1
434 end
435 end
436 last_end = time + job.duration
437 events[time - window.low + 1] += 1
438 events[last_end - window.low + 1] -= 1
439 started += 1
440 busy += job.duration
441 end
442 push!(details, "$id $(job.policy) started $started skipped $skipped replaced $replaced busy $busy")
443 end
444 running = cumsum(events)
445 peak = maximum(running)
446 header = "SIMULATE $(length(jobs)) peak $peak"
447 peak > 0 && (header *= " at $(datetime_text(window.low + argmax(running) - 1))")
448 [header; details]
449end

451function execute!(state::Scheduler, line::AbstractString)::Vector{String}
452 words = split(line, ' '; keepempty = false)
453 command = words[1]
454 expected = get(ARITIES, command, nothing)
455 isnothing(expected) && fail("unknown command")
456 (expected < 0 ? length(words) >= -expected : length(words) == expected) || fail("wrong arguments")
457 job = command in ("REMOVE", "RENAME", "ENABLE", "DISABLE", "RUNTIME", "EXPAND", "DESCRIBE",
458 "MATCH", "NEXT", "PREV", "COUNT", "GAP", "OVERLAP", "CALENDAR") ? require_job(state, words[2]) : nothing
459 if command == "JOB"
460 require_new_id(state, words[2])
461 cron = parse_cron(words[3:end])
462 state.jobs[words[2]] = SchedulerJob(cron = cron)
463 return ["JOB $(words[2]) added"]
464 elseif command == "VALIDATE"
465 cron = parse_cron(words[2:end])
466 return ["VALID $(cron.expanded) $(or_rule(cron) ? "OR" : "AND") $(length(cron.times))"]
467 elseif command == "REMOVE"
468 delete!(state.jobs, words[2])
469 return ["REMOVE $(words[2])"]
470 elseif command == "RENAME"
471 require_new_id(state, words[3])
472 delete!(state.jobs, words[2])
473 state.jobs[words[3]] = job
474 return ["RENAME $(words[2]) $(words[3])"]
475 elseif command == "ENABLE" || command == "DISABLE"
476 desired = command == "ENABLE"
477 job.enabled == desired && fail(desired ? "already enabled" : "already disabled")
478 job.enabled = desired
479 return ["$command $(words[2])"]
480 elseif command == "RUNTIME"
481 duration = positive(words[3], 4, 1440, "invalid duration")
482 policy = findfirst(p -> string(p) == words[4], instances(Policy))
483 isnothing(policy) && fail("invalid policy")
484 job.duration, job.policy = duration, instances(Policy)[policy]
485 return ["RUNTIME $(words[2]) $duration $(job.policy)"]
486 elseif command == "LIST"
487 rows = ["LIST $(length(state.jobs))"]
488 for id in ordered_ids(state)
489 current = state.jobs[id]
490 push!(rows, "$id $(current.enabled ? "on" : "off") $(current.duration) $(current.policy) $(current.cron.text)")
491 end
492 return rows
493 elseif command == "EXPAND"
494 index = findfirst(==(words[3]), FIELD_NAMES)
495 isnothing(index) && fail("invalid field")
496 field = job.cron.fields[index]
497 return [join(["EXPAND $(words[2]) $(words[3]) $(length(field.values))"; string.(field.values); special_text.(field.specials)], " ")]
498 elseif command == "DESCRIBE"
499 return ["DESCRIBE $(words[2]) $(description(job.cron))"]
500 elseif command == "MATCH"
501 time = read_datetime(words, 3)
502 day = time ÷ 1440
503 status = day_status(job.cron, day)
504 reasons = String[]
505 time % 60 in job.cron.fields[1].values || push!(reasons, "minute")
506 time ÷ 60 % 24 in job.cron.fields[2].values || push!(reasons, "hour")
507 !status.condition && !status.dom && push!(reasons, "day-of-month")
508 status.month || push!(reasons, "month")
509 !status.condition && !status.dow && push!(reasons, "day-of-week")
510 day in state.holidays && push!(reasons, "excluded")
511 return ["MATCH $(words[2]) $(datetime_text(time)) " * (isempty(reasons) ? "yes" : "no $(join(reasons, " "))")]
512 elseif command == "NEXT" || command == "PREV"
513 base = read_datetime(words, 3)
514 count = positive(words[5], 2, 50, "invalid count")
515 backward = command == "PREV"
516 times = fire_times(job.cron, backward ? base - 527040 : base + 1, backward ? base - 1 : base + 527040,
517 state.holidays; backward, limit = count)
518 rows = ["$command $(words[2]) $(length(times))"]
519 append!(rows, ["$(datetime_text(time)) $(backward ? "-" : "+")$(span_text(abs(time - base)))" for time in times])
520 length(times) < count && push!(rows, "NONE")
521 return rows
522 elseif command == "COUNT"
523 window = read_window(words, 3)
524 times = fire_times(job.cron, window.low, window.high, state.holidays)
525 return ["COUNT $(words[2]) runs $(length(times)) days $(length(unique(times .÷ 1440)))"]
526 elseif command == "GAP"
527 base = read_datetime(words, 3)
528 times = fire_times(job.cron, base, base + 44640, state.holidays)
529 length(times) < 2 && return ["GAP $(words[2]) runs $(length(times)) none"]
530 gaps = diff(times)
531 longest = argmax(gaps)
532 histogram = Dict{Int,Int}()
533 for gap in gaps
534 histogram[gap] = get(histogram, gap, 0) + 1
535 end
536 rows = ["GAP $(words[2]) runs $(length(times)) shortest $(span_text(minimum(gaps))) longest $(span_text(maximum(gaps))) from $(datetime_text(times[longest])) to $(datetime_text(times[longest + 1]))"]
537 append!(rows, ["$(span_text(gap)) $(histogram[gap])" for gap in sort!(collect(keys(histogram)))])
538 return rows
539 elseif command == "OVERLAP"
540 other = require_job(state, words[3])
541 window = read_window(words, 4)
542 first = fire_times(job.cron, window.low, window.high, state.holidays)
543 other_times = Set(fire_times(other.cron, window.low, window.high, state.holidays))
544 times = filter(in(other_times), first)
545 suffix = isempty(times) ? "" : " first $(datetime_text(times[1])) last $(datetime_text(times[end]))"
546 return ["OVERLAP $(words[2]) $(words[3]) $(length(times))$suffix"]
547 elseif command in ("TIMELINE", "CONFLICTS", "SIMULATE")
548 window = read_window(words, 2)
549 return command == "SIMULATE" ? simulation_report(state, window) : combined_report(state, window, command == "CONFLICTS")
550 elseif command == "WEEK"
551 first = read_date(words[2])
552 return week_report(state, first, words[2])
553 elseif command == "CALENDAR"
554 first = read_month(words[3])
555 return calendar_report(state, job, first, words[3], words[2])
556 elseif command == "EXCLUDE" || command == "INCLUDE"
557 day = read_date(words[2])
558 if command == "EXCLUDE"
559 day in state.holidays && fail("already excluded")
560 push!(state.holidays, day)
561 else
562 day in state.holidays || fail("not excluded")
563 delete!(state.holidays, day)
564 end
565 return ["$command $(date_text(day)) holidays $(length(state.holidays))"]
566 end
567 ["HOLIDAYS $(length(state.holidays))"; date_text.(sort!(collect(state.holidays)))]
568end
