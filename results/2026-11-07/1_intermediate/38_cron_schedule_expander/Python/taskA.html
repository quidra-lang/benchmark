40def is_leap(year):
41 return year % 4 == 0 and year % 100 != 0 or year % 400 == 0
42
43
44def month_length(year, month):
45 if month == 2 and is_leap(year):
46 return 29
47 return MONTH_LENGTHS[month - 1]
48
49
50def leap_years_through(year):
51 return year // 4 - year // 100 + year // 400
52
53
54def year_start(year):
55 return 365 * (year - 2000) + leap_years_through(year - 1) - leap_years_through(1999)
56
57
58def day_number(year, month, day):
59 number = year_start(year) + day - 1
60 for earlier in range(1, month):
61 number += month_length(year, earlier)
62 return number

65def civil_date(number):
66 year = 2000 + number // 366
67 while year_start(year + 1) <= number:
68 year += 1
69 day = number - year_start(year) + 1
70 month = 1
71 while day > month_length(year, month):
72 day -= month_length(year, month)
73 month += 1
74 return year, month, day

77def weekday_of(number):
78 return (number + 6) % 7

81def clock_text(offset):
82 return f"{offset // 60:02d}:{offset % 60:02d}"

85def date_text(number):
86 year, month, day = civil_date(number)
87 return f"{year:04d}-{month:02d}-{day:02d}"

90def date_notation(number):
91 return date_text(number) + " " + WEEKDAY_ABBREVIATIONS[weekday_of(number)]

94def datetime_notation(minute):
95 number, offset = divmod(minute, MINUTES_PER_DAY)
96 weekday = WEEKDAY_ABBREVIATIONS[weekday_of(number)]
97 return f"{date_text(number)} {clock_text(offset)} {weekday}"

100def span_notation(minutes):
101 days, rest = divmod(minutes, MINUTES_PER_DAY)
102 return f"{days}d{rest // 60:02d}h{rest % 60:02d}m"

105class CalendarDate:
106 def __init__(self, number):
107 self.number = number
108 self.year, self.month, self.day = civil_date(number)
109 self.length = month_length(self.year, self.month)
110 self.weekday = weekday_of(number)
111
112 def nearest_weekday(self, target):
113 weekday = weekday_of(self.number - self.day + target)
114 if weekday == 6:
115 return 3 if target == 1 else target - 1
116 if weekday == 0:
117 return target - 2 if target == self.length else target + 1
118 return target

121class FieldSpec:
122 def __init__(self, name, low, high, codes=(), labels=()):
123 self.name = name
124 self.low = low
125 self.high = high
126 self.codes = codes
127 self.labels = labels
128
129 def error(self, reason):
130 return CommandError(f"{self.name} {reason}")
131
132 def number(self, text, low, high, width=2):
133 if not (1 <= len(text) <= width and text.isdigit()):
134 raise self.error("invalid value")
135 value = int(text)
136 if not low <= value <= high:
137 raise self.error("out of range")
138 return value
139
140 def value(self, text):
141 if text in self.codes:
142 return self.low + self.codes.index(text)
143 return self.number(text, self.low, self.high)
144
145 def label(self, value):
146 if self.labels:
147 return self.labels[value - self.low]
148 return str(value)

160class Special:
161 def __init__(self, kind, number=0, weekday=0):
162 self.kind = kind
163 self.number = number
164 self.weekday = weekday
165
166 def canonical(self):
167 if self.kind == "L-n":
168 return f"L-{self.number}"
169 if self.kind == "nW":
170 return f"{self.number}W"
171 if self.kind == "v#n":
172 return f"{self.weekday}#{self.number}"
173 if self.kind == "vL":
174 return f"{self.weekday}L"
175 return self.kind
176
177 def matches(self, date):
178 if self.kind == "L":
179 return date.day == date.length
180 if self.kind == "LW":
181 return date.day == date.nearest_weekday(date.length)
182 if self.kind == "L-n":
183 return date.day == date.length - self.number
184 if self.kind == "nW":
185 return self.number <= date.length and date.day == date.nearest_weekday(self.number)
186 if self.kind == "v#n":
187 return date.weekday == self.weekday and (date.day - 1) // 7 + 1 == self.number
188 return date.weekday == self.weekday and date.day + 7 > date.length
189
190 def phrase(self):
191 if self.kind == "L":
192 return "the last day"
193 if self.kind == "LW":
194 return "the last weekday"
195 if self.kind == "L-n":
196 if self.number == 1:
197 return "1 day before the last day"
198 return f"{self.number} days before the last day"
199 if self.kind == "nW":
200 return f"the weekday nearest day {self.number}"
201 if self.kind == "v#n":
202 return f"the {ORDINALS[self.number - 1]} {WEEKDAY_NAMES[self.weekday]}"
203 return f"the last {WEEKDAY_NAMES[self.weekday]}"

206class Field:
207 def __init__(self, spec, text):
208 self.spec = spec
209 self.values = set()
210 self.specials = []
211 self.restricted = text[0] != "*"
212 for item in text.split(","):
213 self.add_item(item)
214
215 def add_item(self, item):
216 spec = self.spec
217 if item == "":
218 raise spec.error("empty item")
219 if spec.name == "day-of-month" and (item[0] == "L" or item[-1] == "W"):
220 self.add_special(self.day_special(item))
221 elif spec.name == "day-of-week" and "#" in item:
222 weekday_text, _, number_text = item.partition("#")
223 weekday = spec.value(weekday_text) % 7
224 self.add_special(Special("v#n", spec.number(number_text, 1, 5, 1), weekday))
225 elif spec.name == "day-of-week" and len(item) >= 2 and item[-1] == "L":
226 self.add_special(Special("vL", 0, spec.value(item[:-1]) % 7))
227 else:
228 self.add_range(item)
229
230 def day_special(self, item):
231 if item in ("L", "LW"):
232 return Special(item)
233 if item.startswith("L-"):
234 return Special("L-n", self.spec.number(item[2:], 1, 30))
235 if item[-1] == "W":
236 return Special("nW", self.spec.number(item[:-1], 1, 31))
237 raise self.spec.error("invalid value")
238
239 def add_range(self, item):
240 spec = self.spec
241 base, slash, step_text = item.partition("/")
242 if base == "*":
243 first, last = spec.low, spec.high
244 elif "-" in base:
245 first_text, _, last_text = base.partition("-")
246 first = spec.value(first_text)
247 last = spec.value(last_text)
248 if first > last:
249 raise spec.error("invalid range")
250 else:
251 first = spec.value(base)
252 last = spec.high if slash else first
253 step = 1
254 if slash:
255 if not (1 <= len(step_text) <= 2 and step_text.isdigit() and int(step_text) >= 1):
256 raise spec.error("invalid step")
257 step = int(step_text)
258 for value in range(first, last + 1, step):
259 self.values.add(value % 7 if spec.name == "day-of-week" else value)
260
261 def add_special(self, special):
262 if all(other.canonical() != special.canonical() for other in self.specials):
263 self.specials.append(special)
264
265 def matches(self, value, date):
266 return value in self.values or any(special.matches(date) for special in self.specials)

269class Cron:
270 def __init__(self, tokens):
271 if len(tokens) == 1 and tokens[0][0] == "@":
272 if tokens[0] not in MACROS:
273 raise CommandError("unknown macro")
274 tokens = MACROS[tokens[0]].split()
275 if len(tokens) != 5:
276 raise CommandError("field count")
277 self.expanded = " ".join(tokens)
278 self.fields = [Field(spec, text) for spec, text in zip(FIELD_SPECS, tokens)]
279 self.minute, self.hour, self.day, self.month, self.weekday = self.fields
280 self.times = [
281 hour * 60 + minute
282 for hour in sorted(self.hour.values)
283 for minute in sorted(self.minute.values)
284 ]
285
286 def field(self, name):
287 for field in self.fields:
288 if field.spec.name == name:
289 return field
290 return None

292 def both_restricted(self):
293 return self.day.restricted and self.weekday.restricted

295 def day_condition(self, date):
296 day_matches = self.day.matches(date.day, date)
297 weekday_matches = self.weekday.matches(date.weekday, date)
298 if self.both_restricted():
299 return day_matches or weekday_matches
300 return day_matches and weekday_matches

302 def scheduled(self, date):
303 return date.month in self.month.values and self.day_condition(date)

306def join_list(items, word="and"):
307 if len(items) == 1:
308 return items[0]
309 return ", ".join(items[:-1]) + f" {word} " + items[-1]

312def value_list(field, word="and"):
313 values = sorted(field.values)
314 label = field.spec.label
315 items = []
316 start = 0
317 while start < len(values):
318 end = start
319 while end + 1 < len(values) and values[end + 1] == values[end] + 1:
320 end += 1
321 if end - start >= 2:
322 items.append(f"{label(values[start])} through {label(values[end])}")
323 else:
324 items.extend(label(value) for value in values[start:end + 1])
325 start = end + 1
326 return join_list(items, word)

329def step_of(field):
330 values = sorted(field.values)
331 if len(values) < 2 or values[0] != 0 or values[1] < 2:
332 return None
333 if values != list(range(0, field.spec.high + 1, values[1])):
334 return None
335 return values[1]

338def time_phrase(cron):
339 minutes = sorted(cron.minute.values)
340 hours = sorted(cron.hour.values)
341 if len(cron.times) <= 4:
342 return "At " + join_list([clock_text(offset) for offset in cron.times])
343 minute_step = step_of(cron.minute)
344 if len(minutes) == 60:
345 phrase = "Every minute"
346 elif minute_step is not None:
347 phrase = f"Every {minute_step} minutes"
348 elif len(minutes) == 1:
349 phrase = f"At minute {minutes[0]}"
350 else:
351 phrase = "At minutes " + value_list(cron.minute)
352 if phrase.startswith("At minute") and len(hours) == 24:
353 return phrase + " of every hour"
354 hour_step = step_of(cron.hour)
355 if len(hours) == 24:
356 return phrase
357 if hour_step is not None:
358 return phrase + f", every {hour_step} hours"
359 if hours[-1] - hours[0] == len(hours) - 1:
360 return phrase + f", between {hours[0]:02d}:00 and {hours[-1]:02d}:59"
361 return phrase + ", during hours " + value_list(cron.hour)

364def day_of_month_clause(field):
365 parts = []
366 days = sorted(field.values)
367 if len(days) == 1:
368 parts.append(f"day {days[0]}")
369 elif days:
370 parts.append("days " + value_list(field))
371 parts.extend(special.phrase() for special in field.specials)
372 return join_list(parts) + " of the month"
373
374
375def day_of_week_clause(field, word):
376 parts = []
377 if field.values:
378 parts.append(value_list(field, word))
379 parts.extend(special.phrase() for special in field.specials)
380 clause = join_list(parts, word)
381 if field.specials:
382 clause += " of the month"
383 return clause

386def day_phrase(cron):
387 day_full = len(cron.day.values) == 31
388 weekday_full = len(cron.weekday.values) == 7
389 if cron.both_restricted():
390 if day_full or weekday_full:
391 return ""
392 day_clause = day_of_month_clause(cron.day)
393 return f", on {day_clause} or on {day_of_week_clause(cron.weekday, 'and')}"
394 if not day_full and not weekday_full:
395 day_clause = day_of_month_clause(cron.day)
396 return f", on {day_clause} if it is {day_of_week_clause(cron.weekday, 'or')}"
397 if not day_full:
398 return ", on " + day_of_month_clause(cron.day)
399 if not weekday_full:
400 return ", on " + day_of_week_clause(cron.weekday, "and")
401 return ""

404def month_phrase(cron):
405 if len(cron.month.values) == 12:
406 return ""
407 return ", in " + value_list(cron.month)

420def date_value(text):
421 if len(text) != 10 or text[4] != "-" or text[7] != "-":
422 return None
423 if not (text[:4] + text[5:7] + text[8:]).isdigit():
424 return None
425 year, month, day = int(text[:4]), int(text[5:7]), int(text[8:])
426 if not (2001 <= year <= 2098 and 1 <= month <= 12):
427 return None
428 if not 1 <= day <= month_length(year, month):
429 return None
430 return day_number(year, month, day)

433def time_value(text):
434 if len(text) != 5 or text[2] != ":" or not (text[:2] + text[3:]).isdigit():
435 return None
436 hour, minute = int(text[:2]), int(text[3:])
437 if hour > 23 or minute > 59:
438 return None
439 return hour * 60 + minute

546 lo = parse_datetime(arguments[0], arguments[1])

560 def runs_on(self, job, date):
561 return job.cron.scheduled(date) and date.number not in self.holidays

563 def fire_times(self, job, lo, hi, limit=None):
564 times = []
565 for number in range(lo // MINUTES_PER_DAY, hi // MINUTES_PER_DAY + 1):
566 if not self.runs_on(job, CalendarDate(number)):
567 continue
568 for offset in job.cron.times:
569 minute = number * MINUTES_PER_DAY + offset
570 if lo <= minute <= hi:
571 times.append(minute)
572 if len(times) == limit:
573 return times
574 return times
575
576 def fire_times_backward(self, job, lo, hi, limit):
577 times = []
578 for number in range(hi // MINUTES_PER_DAY, lo // MINUTES_PER_DAY - 1, -1):
579 if not self.runs_on(job, CalendarDate(number)):
580 continue
581 for offset in reversed(job.cron.times):
582 minute = number * MINUTES_PER_DAY + offset
583 if lo <= minute <= hi:
584 times.append(minute)
585 if len(times) == limit:
586 return times
587 return times

589 def firing_slots(self, lo, hi):
590 slots = [[] for _ in range(hi - lo + 1)]
591 for job in self.enabled_jobs():
592 for minute in self.fire_times(job, lo, hi):
593 slots[minute - lo].append(job.id)
594 return slots

596 def mismatches(self, job, minute):
597 cron = job.cron
598 date = CalendarDate(minute // MINUTES_PER_DAY)
599 offset = minute % MINUTES_PER_DAY
600 satisfied = cron.day_condition(date)
601 reasons = []
602 if offset % 60 not in cron.minute.values:
603 reasons.append("minute")
604 if offset // 60 not in cron.hour.values:
605 reasons.append("hour")
606 if not satisfied and not cron.day.matches(date.day, date):
607 reasons.append("day-of-month")
608 if date.month not in cron.month.values:
609 reasons.append("month")
610 if not satisfied and not cron.weekday.matches(date.weekday, date):
611 reasons.append("day-of-week")
612 if date.number in self.holidays:
613 reasons.append("excluded")
614 return reasons

683 def match(self, arguments):

721 def gap(self, arguments):
722 job = self.job(arguments[0])
723 base = parse_datetime(arguments[1], arguments[2])
724 times = self.fire_times(job, base, base + WINDOW_LIMIT)
725 if len(times) < 2:
726 self.emit("GAP", job.id, "runs", len(times), "none")
727 return
728 gaps = [later - earlier for earlier, later in zip(times, times[1:])]
729 longest = max(gaps)
730 index = gaps.index(longest)
731 self.emit(
732 "GAP", job.id, "runs", len(times),
733 "shortest", span_notation(min(gaps)), "longest", span_notation(longest),
734 "from", datetime_notation(times[index]),
735 "to", datetime_notation(times[index + 1]),
736 )
737 tally = Counter(gaps)
738 for value in sorted(tally):
739 self.emit(span_notation(value), tally[value])

781 def week(self, arguments):
782 start = parse_date(arguments[0])
783 lo = start * MINUTES_PER_DAY
784 hi = lo + 7 * MINUTES_PER_DAY - 1
785 counts = [0] * (7 * 24)
786 for job in self.enabled_jobs():
787 for minute in self.fire_times(job, lo, hi):
788 counts[(minute - lo) // 60] += 1
789 self.emit("WEEK", arguments[0], sum(counts))
790 for index in range(7):
791 hours = counts[index * 24:(index + 1) * 24]
792 symbols = "".join(hour_symbol(count) for count in hours)
793 self.emit(date_notation(start + index), symbols, sum(hours))

795 def calendar(self, arguments):
796 job = self.job(arguments[0])
797 year, month = parse_year_month(arguments[1])
798 first = day_number(year, month, 1)
799 cells = ["..."] * weekday_of(first)
800 days = 0
801 for day in range(1, month_length(year, month) + 1):
802 date = CalendarDate(first + day - 1)
803 if not job.cron.scheduled(date):
804 mark = "-"
805 elif date.number in self.holidays:
806 mark = "x"
807 else:
808 mark = "*"
809 days += 1
810 cells.append(f"{day:02d}{mark}")
811 while len(cells) % 7 != 0:
812 cells.append("...")
813 runs = days * len(job.cron.times)
814 self.emit("CALENDAR", job.id, arguments[1], "days", days, "runs", runs)
815 self.emit(*WEEKDAY_ABBREVIATIONS)
816 for start in range(0, len(cells), 7):
817 self.emit(*cells[start:start + 7])

819 def simulate(self, arguments):
820 lo, hi = self.window(arguments)
821 jobs = self.enabled_jobs()
822 changes = [0] * (hi - lo + MINUTES_PER_DAY + 1)
823 summaries = []
824 for job in jobs:
825 executions = []
826 skipped = 0
827 replaced = 0
828 for minute in self.fire_times(job, lo, hi):
829 previous = executions[-1] if executions else None
830 running = previous is not None and previous.end > minute
831 if running and job.policy == "FORBID":
832 skipped += 1
833 continue
834 if running and job.policy == "REPLACE":
835 previous.end = minute
836 replaced += 1
837 executions.append(Execution(minute, minute + job.duration))
838 for execution in executions:
839 changes[execution.start - lo] += 1
840 changes[execution.end - lo] -= 1
841 busy = sum(execution.end - execution.start for execution in executions)
842 summaries.append((job, len(executions), skipped, replaced, busy))
843 peak = 0
844 peak_minute = lo
845 running = 0
846 for offset, change in enumerate(changes):
847 running += change
848 if running > peak:
849 peak = running
850 peak_minute = lo + offset
851 if peak > 0:
852 self.emit("SIMULATE", len(jobs), "peak", peak, "at", datetime_notation(peak_minute))
853 else:
854 self.emit("SIMULATE", len(jobs), "peak", 0)
855 for job, started, skipped, replaced, busy in summaries:
856 self.emit(
857 job.id, job.policy, "started", started, "skipped", skipped,
858 "replaced", replaced, "busy", busy,
859 )

880 def execute(self, tokens):
881 name, arguments = tokens[0], tokens[1:]
882 try:
883 if name not in self.handlers:
884 raise CommandError("unknown command")
885 minimum, maximum, handler = self.handlers[name]
886 if len(arguments) < minimum or (maximum is not None and len(arguments) > maximum):
887 raise CommandError("wrong arguments")
888 handler(arguments)
889 except CommandError as error:
890 self.emit("ERROR", error)
