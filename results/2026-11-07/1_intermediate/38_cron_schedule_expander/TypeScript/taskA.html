31function civil(day: number): CivilDate {
32 let year = 2000 + Math.floor(day / 366);
33 while (yearStart(year + 1) <= day) ++year;
34 let remaining = day - yearStart(year), month = 1;
35 while (remaining >= monthLength(year, month)) remaining -= monthLength(year, month++);
36 return { number: day, year, month, day: remaining + 1, weekday: (day + 6) % 7, length: monthLength(year, month) };

88function canonical(item: Special): string {
89 switch (item.kind) {
90 case 'last': return 'L';
91 case 'lastWeekday': return 'LW';
92 case 'before': return `L-${item.number}`;
93 case 'nearest': return `${item.number}W`;
94 case 'nth': return `${item.weekday}#${item.number}`;
95 case 'lastDow': return `${item.weekday}L`;

98class Field {
99 readonly unrestricted: boolean;
100 readonly set: ReadonlySet<number>;
101 readonly values: readonly number[];
102 readonly specials: readonly Special[];
103 constructor(text: string, readonly index: FieldIndex) {
104 this.unrestricted = text.startsWith('*');
105 const values = new Set<number>(), specials: Special[] = [];
106 const [minimum, maximum] = limits[index];
107 const error = (reason: string): never => fail(`${fieldNames[index]} ${reason}`);
108 const value = (text: string): number => {
109 const names: readonly string[] = index === 3 ? months : index === 4 ? weekdays : [];
110 const named = names.findIndex(name => name.slice(0, 3).toUpperCase() === text);
111 return named >= 0 ? named + (index === 3 ? 1 : 0) : natural(text, minimum, maximum, error);
112 };
113 const addSpecial = (item: Special): void => {
114 if (!specials.some(existing => canonical(existing) === canonical(item))) specials.push(item);
115 };
116 for (const item of text.split(',')) {
117 if (item === '') error('empty item');
118 if (index === 2 && (item.startsWith('L') || item.endsWith('W'))) {
119 if (item === 'L') addSpecial({ kind: 'last' });
120 else if (item === 'LW') addSpecial({ kind: 'lastWeekday' });
121 else if (item.startsWith('L-')) addSpecial({ kind: 'before', number: natural(item.slice(2), 1, 30, error) });
122 else if (item.endsWith('W')) addSpecial({ kind: 'nearest', number: natural(item.slice(0, -1), 1, 31, error) });
123 else error('invalid value');
124 continue;
125 }
126 if (index === 4 && item.includes('#')) {
127 const split = item.indexOf('#'), weekday = value(item.slice(0, split)) % 7, part = item.slice(split + 1);
128 if (!/^[0-9]$/.test(part)) error('invalid value');
129 const number = Number(part);
130 if (number < 1 || number > 5) error('out of range');
131 addSpecial({ kind: 'nth', weekday, number });
132 continue;
133 }
134 if (index === 4 && item.length >= 2 && item.endsWith('L')) {
135 addSpecial({ kind: 'lastDow', weekday: value(item.slice(0, -1)) % 7 });
136 continue;
137 }
138 const slash = item.indexOf('/'), base = slash < 0 ? item : item.slice(0, slash);
139 let first: number, last: number, step = 1;
140 if (base === '*') { first = minimum; last = maximum; }
141 else if (base.includes('-')) {
142 const dash = base.indexOf('-');
143 first = value(base.slice(0, dash)); last = value(base.slice(dash + 1));
144 if (first > last) error('invalid range');
145 } else { first = value(base); last = slash < 0 ? first : maximum; }
146 if (slash >= 0) {
147 const part = item.slice(slash + 1);
148 if (!/^[0-9]{1,2}$/.test(part) || Number(part) < 1) error('invalid step');
149 step = Number(part);
150 }
151 for (let number = first; number <= last; number += step) values.add(index === 4 ? number % 7 : number);
152 }
153 this.set = values;
154 this.values = [...values].sort((a, b) => a - b);
155 this.specials = specials;
156 }

159class Cron {
160 readonly text: string;
161 readonly expanded: string;
162 readonly fields: readonly [Field, Field, Field, Field, Field];

164 readonly times: readonly number[];
165 constructor(tokens: readonly string[]) {
166 this.text = tokens.join(' ');
167 if (tokens.length === 1 && tokens[0].startsWith('@')) {
168 const expanded = macros.get(tokens[0]);
169 if (expanded === undefined) fail('unknown macro');
170 tokens = expanded.split(' ');
171 }
172 if (tokens.length !== 5) fail('field count');
173 this.expanded = tokens.join(' ');
174 this.fields = [new Field(tokens[0], 0), new Field(tokens[1], 1), new Field(tokens[2], 2), new Field(tokens[3], 3), new Field(tokens[4], 4)];
175 this.or = !this.fields[2].unrestricted && !this.fields[4].unrestricted;
176 this.times = this.fields[1].values.flatMap(hour => this.fields[0].values.map(minute => 60 * hour + minute));
177 }
178 days(date: CivilDate): DayStatus {
179 const dom = this.fields[2], dow = this.fields[4];
180 const nearest = (day: number): number => {
181 const weekday = (date.number - date.day + day + 6) % 7;
182 if (weekday === 6) return day === 1 ? 3 : day - 1;
183 if (weekday === 0) return day === date.length ? day - 2 : day + 1;
184 return day;
185 };
186 const monthDay = dom.set.has(date.day) || dom.specials.some(item => {
187 switch (item.kind) {
188 case 'last': return date.day === date.length;
189 case 'lastWeekday': return date.day === nearest(date.length);
190 case 'before': return date.day === date.length - item.number;
191 case 'nearest': return item.number <= date.length && date.day === nearest(item.number);
192 default: return false;
193 }
194 });
195 const weekDay = dow.set.has(date.weekday) || dow.specials.some(item => {
196 switch (item.kind) {
197 case 'nth': return item.weekday === date.weekday && Math.floor((date.day - 1) / 7) + 1 === item.number;
198 case 'lastDow': return item.weekday === date.weekday && date.day + 7 > date.length;
199 default: return false;
200 }
201 });
202 return { monthDay, weekDay, matches: this.or ? monthDay || weekDay : monthDay && weekDay };
203 }
204 scheduled(date: CivilDate): boolean { return this.fields[3].set.has(date.month) && this.days(date).matches; }
205 reasons(minute: number, holidays: ReadonlySet<number>): string[] {
206 const date = civil(Math.floor(minute / 1440)), days = this.days(date), reasons: string[] = [];
207 if (!this.fields[0].set.has(minute % 60)) reasons.push('minute');
208 if (!this.fields[1].set.has(Math.floor(minute % 1440 / 60))) reasons.push('hour');
209 if (!days.matches && !days.monthDay) reasons.push('day-of-month');
210 if (!this.fields[3].set.has(date.month)) reasons.push('month');
211 if (!days.matches && !days.weekDay) reasons.push('day-of-week');
212 if (holidays.has(date.number)) reasons.push('excluded');
213 return reasons;
214 }
215 *fires(low: number, high: number, holidays: ReadonlySet<number>, reverse = false): Generator<number> {
216 const first = Math.floor(low / 1440), last = Math.floor(high / 1440), step = reverse ? -1 : 1;
217 const times = reverse ? this.times.slice().reverse() : this.times;
218 for (let day = reverse ? last : first; reverse ? day >= first : day <= last; day += step) {
219 if (holidays.has(day) || !this.scheduled(civil(day))) continue;
220 for (const time of times) {
221 const minute = 1440 * day + time;
222 if (low <= minute && minute <= high) yield minute;
223 }
224 }
225 }

227function list(parts: readonly string[], conjunction: Conjunction = 'and'): string {
228 return parts.length < 2 ? parts.join('') : parts.slice(0, -1).join(', ') + ` ${conjunction} ` + parts[parts.length - 1];

230function valueList(values: readonly number[], index: FieldIndex, conjunction: Conjunction = 'and'): string {
231 const label = (value: number): string => index === 3 ? months[value - 1] : index === 4 ? weekdays[value] : String(value);
232 const parts: string[] = [];
233 for (let first = 0; first < values.length;) {
234 let end = first + 1;
235 while (end < values.length && values[end] === values[end - 1] + 1) ++end;
236 if (end - first >= 3) parts.push(`${label(values[first])} through ${label(values[end - 1])}`);
237 else for (let position = first; position < end; ++position) parts.push(label(values[position]));
238 first = end;
239 }
240 return list(parts, conjunction);
241}
242function stepped(values: readonly number[], maximum: number): number | null {
243 if (values.length < 2 || values[0] !== 0 || values[1] < 2) return null;
244 const step = values[1];
245 return values.length === Math.floor(maximum / step) + 1 && values.every((value, index) => value === index * step) ? step : null;
246}
247function description(cron: Cron): string {
248 const [minutes, hours, dom, month, dow] = cron.fields;
249 let time: string;
250 if (cron.times.length <= 4) time = 'At ' + list(cron.times.map(timeText));
251 else {
252 const minuteStep = stepped(minutes.values, 59), hourStep = stepped(hours.values, 23);
253 if (minutes.values.length === 60) time = 'Every minute';
254 else if (minuteStep !== null) time = `Every ${minuteStep} minutes`;
255 else if (minutes.values.length === 1) time = `At minute ${minutes.values[0]}`;
256 else time = 'At minutes ' + valueList(minutes.values, 0);
257 if (hours.values.length === 24) { if (time.startsWith('At minute')) time += ' of every hour'; }
258 else if (hourStep !== null) time += `, every ${hourStep} hours`;
259 else if (hours.values[hours.values.length - 1] - hours.values[0] + 1 === hours.values.length) {
260 time += `, between ${pad(hours.values[0])}:00 and ${pad(hours.values[hours.values.length - 1])}:59`;
261 } else time += ', during hours ' + valueList(hours.values, 1);
262 }
263 const domParts: string[] = [];
264 if (dom.values.length) domParts.push(dom.values.length === 1 ? `day ${dom.values[0]}` : 'days ' + valueList(dom.values, 2));
265 for (const item of dom.specials) {
266 switch (item.kind) {
267 case 'last': domParts.push('the last day'); break;
268 case 'lastWeekday': domParts.push('the last weekday'); break;
269 case 'before': domParts.push(`${item.number} ${item.number === 1 ? 'day' : 'days'} before the last day`); break;
270 case 'nearest': domParts.push(`the weekday nearest day ${item.number}`); break;
271 }
272 }
273 const domClause = list(domParts) + ' of the month';
274 const dowClause = (conjunction: Conjunction = 'and'): string => {
275 const parts = dow.values.length ? [valueList(dow.values, 4, conjunction)] : [];
276 for (const item of dow.specials) {
277 if (item.kind === 'nth') parts.push(`the ${['1st', '2nd', '3rd', '4th', '5th'][item.number - 1]} ${weekdays[item.weekday]}`);
278 else if (item.kind === 'lastDow') parts.push(`the last ${weekdays[item.weekday]}`);
279 }
280 return list(parts, conjunction) + (dow.specials.length ? ' of the month' : '');
281 };
282 const domFull = dom.values.length === 31, dowFull = dow.values.length === 7;
283 let day = '';
284 if (cron.or) { if (!domFull && !dowFull) day = `, on ${domClause} or on ${dowClause()}`; }
285 else if (!domFull && !dowFull) day = `, on ${domClause} if it is ${dowClause('or')}`;
286 else if (!domFull) day = `, on ${domClause}`;
287 else if (!dowFull) day = `, on ${dowClause()}`;
288 return time + day + (month.values.length === 12 ? '' : ', in ' + valueList(month.values, 3));
289}

332 parse(line: string): Command {
333 const words = line.trim().split(/ +/), kind = words[0];
334 if (!isCommandName(kind)) fail('unknown command');
335 const expected = arities[kind];
336 if (expected < 0 ? words.length < -expected : words.length !== expected) fail('wrong arguments');
337 switch (kind) {
338 case 'JOB': {
339 const id = this.newId(words[1]), cron = new Cron(words.slice(2));
340 return { kind, id, cron };
341 }
342 case 'VALIDATE': return { kind, cron: new Cron(words.slice(1)) };
343 case 'REMOVE': case 'ENABLE': case 'DISABLE': case 'DESCRIBE': return { kind, job: this.job(words[1]) };
344 case 'RENAME': {
345 const job = this.job(words[1]), id = this.newId(words[2]);
346 return { kind, job, id };
347 }
348 case 'RUNTIME': {
349 const job = this.job(words[1]), duration = positive(words[2], 4, 1440, 'invalid duration'), policy = words[3];
350 if (policy !== 'ALLOW' && policy !== 'FORBID' && policy !== 'REPLACE') fail('invalid policy');
351 return { kind, job, duration, policy };
352 }
353 case 'LIST': case 'HOLIDAYS': return { kind };
354 case 'EXPAND': {
355 const job = this.job(words[1]), field = fieldIndices.find(index => fieldNames[index] === words[2]);
356 if (field === undefined) fail('invalid field');
357 return { kind, job, field };
358 }
359 case 'MATCH': case 'GAP': {
360 const job = this.job(words[1]), base = parseDatetime(words, 2);
361 return { kind, job, base };
362 }
363 case 'NEXT': case 'PREV': {
364 const job = this.job(words[1]), base = parseDatetime(words, 2), count = positive(words[4], 2, 50, 'invalid count');
365 return { kind, job, base, count };
366 }
367 case 'COUNT': {
368 const job = this.job(words[1]), window = parseWindow(words, 2);
369 return { kind, job, window };
370 }
371 case 'OVERLAP': {
372 const first = this.job(words[1]), second = this.job(words[2]), window = parseWindow(words, 3);
373 return { kind, first, second, window };
374 }
375 case 'TIMELINE': case 'CONFLICTS': case 'SIMULATE': return { kind, window: parseWindow(words, 1) };
376 case 'WEEK': return { kind, day: parseDate(words[1]), original: words[1] };
377 case 'CALENDAR': {
378 const job = this.job(words[1]), month = parseMonth(words[2]);
379 return { kind, job, month, original: words[2] };
380 }
381 case 'EXCLUDE': case 'INCLUDE': return { kind, day: parseDate(words[1]) };

395 private simulate(window: Window): string[] {
396 const jobs = this.ordered(true), events = new Map<number, number>(), rows: string[] = [];
397 const event = (minute: number, difference: number): void => { events.set(minute, (events.get(minute) ?? 0) + difference); };
398 for (const job of jobs) {
399 let lastEnd: number | null = null, started = 0, skipped = 0, replaced = 0, busy = 0;
400 for (const minute of this.fires(job, window.low, window.high)) {
401 if (lastEnd !== null && lastEnd > minute) {
402 if (job.policy === 'FORBID') { ++skipped; continue; }
403 if (job.policy === 'REPLACE') {
404 event(lastEnd, 1); event(minute, -1);
405 busy -= lastEnd - minute;
406 ++replaced;
407 }
408 }
409 lastEnd = minute + job.duration;
410 event(minute, 1); event(lastEnd, -1);
411 busy += job.duration;
412 ++started;
413 }
414 rows.push(`${job.id} ${job.policy} started ${started} skipped ${skipped} replaced ${replaced} busy ${busy}`);
415 }
416 let running = 0, peak = 0, earliest: number | null = null;
417 for (const [minute, difference] of [...events].sort((a, b) => a[0] - b[0])) {
418 running += difference;
419 if (running > peak) { peak = running; earliest = minute; }
420 }
421 return [`SIMULATE ${jobs.length} peak ${peak}${earliest === null ? '' : ` at ${datetimeNotation(earliest)}`}`, ...rows];
422 }
423 private week(first: number, text: string): string[] {
424 const jobs = this.ordered(true), rows: string[] = [];
425 let total = 0;
426 for (let day = first; day < first + 7; ++day) {
427 const hours = Array<number>(24).fill(0), date = civil(day);
428 if (!this.holidays.has(day)) for (const job of jobs) if (job.cron.scheduled(date)) {
429 for (const hour of job.cron.fields[1].values) hours[hour] += job.cron.fields[0].values.length;
430 }
431 const count = hours.reduce((a, b) => a + b, 0);
432 total += count;
433 rows.push(`${dateNotation(day)} ${hours.map(value => value === 0 ? '.' : value < 10 ? String(value) : '+').join('')} ${count}`);
434 }
435 return [`WEEK ${text} ${total}`, ...rows];
436 }

439 const cells = Array<string>((first + 6) % 7).fill('...');

445 cells.push(`${pad(day)}${mark}`);
446 }
447 while (cells.length % 7) cells.push('...');
448 const rows = [`CALENDAR ${job.id} ${text} days ${active} runs ${active * job.cron.times.length}`, 'Sun Mon Tue Wed Thu Fri Sat'];
449 for (let index = 0; index < cells.length; index += 7) rows.push(cells.slice(index, index + 7).join(' '));
450 return rows;
451 }
452 private gap(job: Job, base: number): string[] {
453 const times = [...this.fires(job, base, base + 44640)];
454 if (times.length < 2) return [`GAP ${job.id} runs ${times.length} none`];
455 const counts = new Map<number, number>();
456 let shortest = times[1] - times[0], longest = shortest, from = 0;
457 for (let index = 1; index < times.length; ++index) {
458 const gap = times[index] - times[index - 1];
459 counts.set(gap, (counts.get(gap) ?? 0) + 1);
460 shortest = Math.min(shortest, gap);
461 if (gap > longest) { longest = gap; from = index - 1; }
462 }
463 return [`GAP ${job.id} runs ${times.length} shortest ${spanText(shortest)} longest ${spanText(longest)} from ${datetimeNotation(times[from])} to ${datetimeNotation(times[from + 1])}`,
464 ...[...counts].sort((a, b) => a[0] - b[0]).map(([gap, count]) => `${spanText(gap)} ${count}`)];
465 }
466 execute(command: Command): string[] {
467 switch (command.kind) {
468 case 'JOB': {
469 this.jobs.set(command.id, new Job(command.id, command.cron));
470 return [`JOB ${command.id} added`];
471 }
472 case 'VALIDATE': return [`VALID ${command.cron.expanded} ${command.cron.or ? 'OR' : 'AND'} ${command.cron.times.length}`];
473 case 'REMOVE': {
474 this.jobs.delete(command.job.id);
475 return [`REMOVE ${command.job.id}`];
476 }
477 case 'RENAME': {
478 const { job, id } = command, previous = job.id;
479 this.jobs.delete(previous); job.id = id; this.jobs.set(id, job);
480 return [`RENAME ${previous} ${id}`];
481 }
482 case 'ENABLE': case 'DISABLE': {
483 const job = command.job, enabled = command.kind === 'ENABLE';
484 if (job.enabled === enabled) fail(enabled ? 'already enabled' : 'already disabled');
485 job.enabled = enabled;
486 return [`${command.kind} ${job.id}`];
487 }
488 case 'RUNTIME': {
489 const { job, duration, policy } = command;
490 job.duration = duration; job.policy = policy;
491 return [`RUNTIME ${job.id} ${duration} ${policy}`];
492 }
493 case 'LIST': return [`LIST ${this.jobs.size}`, ...this.ordered().map(job => `${job.id} ${job.enabled ? 'on' : 'off'} ${job.duration} ${job.policy} ${job.cron.text}`)];
494 case 'EXPAND': {
495 const { job, field: index } = command, field = job.cron.fields[index];
496 return [[`EXPAND ${job.id} ${fieldNames[index]} ${field.values.length}`, ...field.values, ...field.specials.map(canonical)].join(' ')];
497 }
498 case 'DESCRIBE': return [`DESCRIBE ${command.job.id} ${description(command.job.cron)}`];
499 case 'MATCH': {
500 const { job, base } = command, reasons = job.cron.reasons(base, this.holidays);
501 return [`MATCH ${job.id} ${datetimeNotation(base)} ${reasons.length ? 'no ' + reasons.join(' ') : 'yes'}`];
502 }
503 case 'NEXT': case 'PREV': {
504 const { job, base, count } = command, reverse = command.kind === 'PREV', times: number[] = [];
505 const low = reverse ? base - 527040 : base + 1, high = reverse ? base - 1 : base + 527040;
506 for (const minute of this.fires(job, low, high, reverse)) {
507 times.push(minute);
508 if (times.length === count) break;
509 }
510 const rows = [`${command.kind} ${job.id} ${times.length}`, ...times.map(minute => `${datetimeNotation(minute)} ${reverse ? '-' : '+'}${spanText(Math.abs(minute - base))}`)];
511 if (times.length < count) rows.push('NONE');
512 return rows;
513 }
514 case 'COUNT': {
515 const { job, window } = command, days = new Set<number>();
516 let runs = 0;
517 for (const minute of this.fires(job, window.low, window.high)) { ++runs; days.add(Math.floor(minute / 1440)); }
518 return [`COUNT ${job.id} runs ${runs} days ${days.size}`];
519 }
520 case 'GAP': return this.gap(command.job, command.base);
521 case 'OVERLAP': {
522 const { first, second, window } = command, other = new Set(this.fires(second, window.low, window.high));
523 const shared = [...this.fires(first, window.low, window.high)].filter(minute => other.has(minute));
524 const detail = shared.length ? ` first ${datetimeNotation(shared[0])} last ${datetimeNotation(shared[shared.length - 1])}` : '';
525 return [`OVERLAP ${first.id} ${second.id} ${shared.length}${detail}`];
526 }
527 case 'TIMELINE': {
528 const groups = this.grouped(command.window), count = groups.reduce((value, [, ids]) => value + ids.length, 0), rows = [`TIMELINE ${count}`];
529 let shown = 0;
530 for (const [minute, ids] of groups) {
531 if (shown === 100) break;
532 for (const id of ids) {
533 rows.push(`${datetimeNotation(minute)} ${id}`);
534 if (++shown === 100) break;
535 }
536 }
537 if (count > 100) rows.push(`... ${count - 100} more`);
538 return rows;
539 }
540 case 'CONFLICTS': {
541 const groups = this.grouped(command.window).filter(([, ids]) => ids.length >= 2);
542 const peak = groups.reduce((value, [, ids]) => Math.max(value, ids.length), 0);
543 const rows = [`CONFLICTS ${groups.length} peak ${peak}`, ...groups.slice(0, 100).map(([minute, ids]) => `${datetimeNotation(minute)} ${ids.length} ${ids.join(' ')}`)];
544 if (groups.length > 100) rows.push(`... ${groups.length - 100} more`);
545 return rows;
546 }
547 case 'WEEK': return this.week(command.day, command.original);
548 case 'CALENDAR': return this.calendar(command.job, command.month, command.original);
549 case 'SIMULATE': return this.simulate(command.window);
550 case 'EXCLUDE': case 'INCLUDE': {
551 const day = command.day, exclude = command.kind === 'EXCLUDE';
552 if (this.holidays.has(day) === exclude) fail(exclude ? 'already excluded' : 'not excluded');
553 if (exclude) this.holidays.add(day); else this.holidays.delete(day);
554 return [`${command.kind} ${dateNotation(day)} holidays ${this.holidays.size}`];
555 }
556 case 'HOLIDAYS': return [`HOLIDAYS ${this.holidays.size}`, ...[...this.holidays].sort((a, b) => a - b).map(dateNotation)];
557 }
558 }
559}
