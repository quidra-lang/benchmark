37vector<string> split(const string& text, char separator) {
38 vector<string> result;
39 size_t start = 0;
40 for (;;) {
41 const size_t end = text.find(separator, start);
42 result.push_back(text.substr(start, end == string::npos ? end : end - start));
43 if (end == string::npos) return result;
44 start = end + 1;
45 }
46}

48 string result;
49 for (const auto& part : parts) { if (!result.empty()) result += separator; result += part; }

52bool leap(int year) { return year % 4 == 0 && (year % 100 != 0 || year % 400 == 0); }
53int monthLength(int year, int month) {
54 static const array<int, 12> lengths = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
55 return lengths[month - 1] + (month == 2 && leap(year));
56}
57int leapCount(int year) { return year / 4 - year / 100 + year / 400; }
58int yearStart(int year) { return 365 * (year - 2000) + leapCount(year - 1) - leapCount(1999); }
59int dayNumber(int year, int month, int day) {
60 int result = yearStart(year) + day - 1;
61 for (int index = 1; index < month; ++index) result += monthLength(year, index);
62 return result;
63}
64struct CivilDate { int number, year, month, day, weekday, length; };
65CivilDate civil(int number) {
66 int year = 2000 + number / 366;
67 while (yearStart(year + 1) <= number) ++year;
68 int remaining = number - yearStart(year), month = 1;
69 while (remaining >= monthLength(year, month)) remaining -= monthLength(year, month++);
70 return {number, year, month, remaining + 1, (number + 6) % 7, monthLength(year, month)};
71}
72string dateText(const CivilDate& date) { return to_string(date.year) + "-" + pad(date.month) + "-" + pad(date.day); }
73string timeText(int minute) { return pad(minute / 60) + ":" + pad(minute % 60); }
74string dateNotation(int day) { const auto date = civil(day); return dateText(date) + " " + weekdays[date.weekday].substr(0, 3); }
75string datetimeNotation(int minute) {
76 const auto date = civil(minute / 1440);
77 return dateText(date) + " " + timeText(minute % 1440) + " " + weekdays[date.weekday].substr(0, 3);
78}
79string spanText(int minutes) { return to_string(minutes / 1440) + "d" + pad(minutes % 1440 / 60) + "h" + pad(minutes % 60) + "m"; }
80int parseDate(const string& text, const string& reason = "invalid date") {
81 if (text.size() != 10 || text[4] != '-' || text[7] != '-' || !digits(text.substr(0, 4), 4, 4) ||
82 !digits(text.substr(5, 2), 2, 2) || !digits(text.substr(8), 2, 2)) fail(reason);
83 const int year = stoi(text.substr(0, 4)), month = stoi(text.substr(5, 2)), day = stoi(text.substr(8));
84 if (year < 2001 || year > 2098 || month < 1 || month > 12 || day < 1 || day > monthLength(year, month)) fail(reason);
85 return dayNumber(year, month, day);
86}
87int parseDatetime(const string& date, const string& time) {
88 const int day = parseDate(date, "invalid datetime");
89 if (time.size() != 5 || time[2] != ':' || !digits(time.substr(0, 2), 2, 2) || !digits(time.substr(3), 2, 2)) fail("invalid datetime");
90 const int hour = stoi(time.substr(0, 2)), minute = stoi(time.substr(3));
91 if (hour > 23 || minute > 59) fail("invalid datetime");
92 return day * 1440 + hour * 60 + minute;
93}
94pair<int, int> parseMonth(const string& text) {
95 if (text.size() != 7 || text[4] != '-' || !digits(text.substr(0, 4), 4, 4) || !digits(text.substr(5), 2, 2)) fail("invalid month");
96 const int year = stoi(text.substr(0, 4)), month = stoi(text.substr(5));
97 if (year < 2001 || year > 2098 || month < 1 || month > 12) fail("invalid month");
98 return {year, month};
99}
100int boundedNumber(const string& text, size_t length, int maximum, const string& reason) {
101 if (!digits(text, 1, length) || text.front() == '0' || stoi(text) > maximum) fail(reason);
102 return stoi(text);
103}

105enum class SpecialKind { LastDay, LastWeekday, BeforeLast, NearestWeekday, NthWeekday, LastOccurrence };
106struct Special { SpecialKind kind; string text; int number = 0, weekday = 0; };
107struct Field {
108 set<int> values;
109 vector<Special> specials;
110 bool unrestricted = false;
111
112 Field() = default;
113 Field(const string& text, int index) : unrestricted(text.front() == '*') {
114 const auto [minimum, maximum] = limits[index];
115 const auto error = [&](const string& reason) { fail(fieldNames[index] + " " + reason); };
116 const auto natural = [&](const string& text, int low, int high) {
117 if (!digits(text, 1, 2)) error("invalid value");
118 const int number = stoi(text);
119 if (number < low || number > high) error("out of range");
120 return number;
121 };
122 const auto value = [&](const string& text) {
123 if (index == 3 || index == 4) {
124 const int count = index == 3 ? 12 : 7;
125 for (int i = 0; i < count; ++i) {
126 string name = (index == 3 ? months[i] : weekdays[i]).substr(0, 3);
127 for (char& ch : name) if (ch >= 'a' && ch <= 'z') ch -= 'a' - 'A';
128 if (text == name) return i + (index == 3);
129 }
130 }
131 return natural(text, minimum, maximum);
132 };
133 const auto special = [&](SpecialKind kind, string canonical, int number = 0, int weekday = 0) {
134 if (none_of(specials.begin(), specials.end(), [&](const Special& item) { return item.text == canonical; }))
135 specials.push_back({kind, move(canonical), number, weekday});
136 };
137 for (const string& item : split(text, ',')) {
138 if (item.empty()) error("empty item");
139 if (index == 2 && (item.front() == 'L' || item.back() == 'W')) {
140 if (item == "L") special(SpecialKind::LastDay, "L");
141 else if (item == "LW") special(SpecialKind::LastWeekday, "LW");
142 else if (item.rfind("L-", 0) == 0) {
143 const int number = natural(item.substr(2), 1, 30);
144 special(SpecialKind::BeforeLast, "L-" + to_string(number), number);
145 } else if (item.back() == 'W') {
146 const int number = natural(item.substr(0, item.size() - 1), 1, 31);
147 special(SpecialKind::NearestWeekday, to_string(number) + "W", number);
148 } else error("invalid value");
149 continue;
150 }
151 const size_t hash = item.find('#');
152 if (index == 4 && hash != string::npos) {
153 const int weekday = value(item.substr(0, hash)) % 7;
154 const string part = item.substr(hash + 1);
155 if (!digits(part, 1, 1)) error("invalid value");
156 const int number = stoi(part);
157 if (number < 1 || number > 5) error("out of range");
158 special(SpecialKind::NthWeekday, to_string(weekday) + "#" + to_string(number), number, weekday);
159 continue;
160 }
161 if (index == 4 && item.size() >= 2 && item.back() == 'L') {
162 const int weekday = value(item.substr(0, item.size() - 1)) % 7;
163 special(SpecialKind::LastOccurrence, to_string(weekday) + "L", 0, weekday);
164 continue;
165 }
166 const size_t slash = item.find('/');
167 const string base = item.substr(0, slash);
168 int first, last, step = 1;
169 if (base == "*") { first = minimum; last = maximum; }
170 else if (const size_t dash = base.find('-'); dash != string::npos) {
171 first = value(base.substr(0, dash)); last = value(base.substr(dash + 1));
172 if (first > last) error("invalid range");
173 } else { first = value(base); last = slash == string::npos ? first : maximum; }
174 if (slash != string::npos) {
175 const string part = item.substr(slash + 1);
176 if (!digits(part, 1, 2) || stoi(part) < 1) error("invalid step");
177 step = stoi(part);
178 }
179 for (int number = first; number <= last; number += step) values.insert(index == 4 ? number % 7 : number);
180 }
181 }

184struct Cron {
185 string original, expanded;
186 array<Field, 5> fields;
187 vector<int> times;
188 bool useOr;
189
190 explicit Cron(vector<string> tokens) : original(join(tokens)) {
191 if (tokens.size() == 1 && tokens[0].front() == '@') {
192 const auto found = macros.find(tokens[0]);
193 if (found == macros.end()) fail("unknown macro");
194 tokens = words(found->second);
195 }
196 if (tokens.size() != 5) fail("field count");
197 expanded = join(tokens);
198 for (int index = 0; index < 5; ++index) fields[index] = Field(tokens[index], index);
199 useOr = !fields[2].unrestricted && !fields[4].unrestricted;
200 for (int hour : fields[1].values) for (int minute : fields[0].values) times.push_back(hour * 60 + minute);
201 }
202 pair<bool, bool> days(const CivilDate& date) const {
203 const auto nearest = [&](int day) {
204 const int weekday = (date.number - date.day + day + 6) % 7;
205 if (weekday == 6) return day == 1 ? 3 : day - 1;
206 if (weekday == 0) return day == date.length ? day - 2 : day + 1;
207 return day;
208 };
209 bool dom = fields[2].values.count(date.day), dow = fields[4].values.count(date.weekday);
210 for (const auto& item : fields[2].specials) {
211 switch (item.kind) {
212 case SpecialKind::LastDay: dom |= date.day == date.length; break;
213 case SpecialKind::LastWeekday: dom |= date.day == nearest(date.length); break;
214 case SpecialKind::BeforeLast: dom |= date.day == date.length - item.number; break;
215 case SpecialKind::NearestWeekday: dom |= item.number <= date.length && date.day == nearest(item.number); break;
216 default: break;
217 }
218 }
219 for (const auto& item : fields[4].specials) {
220 dow |= date.weekday == item.weekday && (item.kind == SpecialKind::NthWeekday ?
221 (date.day - 1) / 7 + 1 == item.number : date.day + 7 > date.length);
222 }
223 return {dom, dow};
224 }
225 bool dayMatches(bool dom, bool dow) const { return useOr ? dom || dow : dom && dow; }
226 bool scheduled(const CivilDate& date) const {
227 const auto [dom, dow] = days(date);
228 return fields[3].values.count(date.month) && dayMatches(dom, dow);
229 }
230 vector<string> reasons(int minute, const set<int>& holidays) const {
231 const auto date = civil(minute / 1440);
232 const auto [dom, dow] = days(date);
233 const bool day = dayMatches(dom, dow);
234 vector<string> result;
235 if (!fields[0].values.count(minute % 60)) result.push_back("minute");
236 if (!fields[1].values.count(minute % 1440 / 60)) result.push_back("hour");
237 if (!day && !dom) result.push_back("day-of-month");
238 if (!fields[3].values.count(date.month)) result.push_back("month");
239 if (!day && !dow) result.push_back("day-of-week");
240 if (holidays.count(date.number)) result.push_back("excluded");
241 return result;
242 }
243 template<class Visitor>
244 void eachFire(int low, int high, const set<int>& holidays, Visitor visit, bool reverse = false) const {
245 const int first = low / 1440, last = high / 1440, step = reverse ? -1 : 1;
246 for (int day = reverse ? last : first; reverse ? day >= first : day <= last; day += step) {
247 if (holidays.count(day) || !scheduled(civil(day))) continue;
248 for (size_t index = 0; index < times.size(); ++index) {
249 const int minute = 1440 * day + times[reverse ? times.size() - 1 - index : index];
250 if (low <= minute && minute <= high && !visit(minute)) return;
251 }
252 }
253 }

256string naturalList(vector<string> parts, const string& conjunction = "and") {
257 if (parts.empty()) return "";
258 if (parts.size() == 1) return parts[0];
259 const string last = parts.back();
260 parts.pop_back();
261 return join(parts, ", ") + " " + conjunction + " " + last;
262}
263string valueList(const set<int>& values, int index, const string& conjunction = "and") {
264 const auto label = [&](int value) { return index == 3 ? months[value - 1] : index == 4 ? weekdays[value] : to_string(value); };
265 const vector<int> ordered(values.begin(), values.end());
266 vector<string> parts;
267 for (size_t first = 0; first < ordered.size();) {
268 size_t end = first + 1;
269 while (end < ordered.size() && ordered[end] == ordered[end - 1] + 1) ++end;
270 if (end - first >= 3) parts.push_back(label(ordered[first]) + " through " + label(ordered[end - 1]));
271 else for (size_t i = first; i < end; ++i) parts.push_back(label(ordered[i]));
272 first = end;
273 }
274 return naturalList(move(parts), conjunction);
275}
276optional<int> stepped(const set<int>& values, int maximum) {
277 if (values.size() < 2 || *values.begin() != 0) return nullopt;
278 const int step = *next(values.begin());
279 if (step < 2 || values.size() != static_cast<size_t>(maximum / step + 1)) return nullopt;
280 int expected = 0;
281 for (int value : values) { if (value != expected) return nullopt; expected += step; }
282 return step;
283}
284string description(const Cron& cron) {
285 const auto& minutes = cron.fields[0].values;
286 const auto& hours = cron.fields[1].values;
287 const auto& dom = cron.fields[2];
288 const auto& month = cron.fields[3].values;
289 const auto& dow = cron.fields[4];
290 string time;
291 if (cron.times.size() <= 4) {
292 vector<string> parts;
293 for (int minute : cron.times) parts.push_back(timeText(minute));
294 time = "At " + naturalList(move(parts));
295 } else {
296 const auto minuteStep = stepped(minutes, 59), hourStep = stepped(hours, 23);
297 if (minutes.size() == 60) time = "Every minute";
298 else if (minuteStep) time = "Every " + to_string(*minuteStep) + " minutes";
299 else if (minutes.size() == 1) time = "At minute " + to_string(*minutes.begin());
300 else time = "At minutes " + valueList(minutes, 0);
301 if (hours.size() == 24) { if (time.rfind("At minute", 0) == 0) time += " of every hour"; }
302 else if (hourStep) time += ", every " + to_string(*hourStep) + " hours";
303 else if (*hours.rbegin() - *hours.begin() + 1 == static_cast<int>(hours.size()))
304 time += ", between " + pad(*hours.begin()) + ":00 and " + pad(*hours.rbegin()) + ":59";
305 else time += ", during hours " + valueList(hours, 1);
306 }
307 vector<string> domParts;
308 if (!dom.values.empty()) domParts.push_back(dom.values.size() == 1 ? "day " + to_string(*dom.values.begin()) : "days " + valueList(dom.values, 2));
309 for (const auto& item : dom.specials) {
310 switch (item.kind) {
311 case SpecialKind::LastDay: domParts.push_back("the last day"); break;
312 case SpecialKind::LastWeekday: domParts.push_back("the last weekday"); break;
313 case SpecialKind::BeforeLast:
314 domParts.push_back(to_string(item.number) + (item.number == 1 ? " day" : " days") + " before the last day"); break;
315 case SpecialKind::NearestWeekday: domParts.push_back("the weekday nearest day " + to_string(item.number)); break;
316 default: break;
317 }
318 }
319 const string domClause = naturalList(move(domParts)) + " of the month";
320 const auto dowClause = [&](const string& conjunction = "and") {
321 static const array<string, 5> ordinals = {"1st", "2nd", "3rd", "4th", "5th"};
322 vector<string> parts;
323 if (!dow.values.empty()) parts.push_back(valueList(dow.values, 4, conjunction));
324 for (const auto& item : dow.specials) {
325 const string occurrence = item.kind == SpecialKind::NthWeekday ? ordinals[item.number - 1] : "last";
326 parts.push_back("the " + occurrence + " " + weekdays[item.weekday]);
327 }
328 return naturalList(move(parts), conjunction) + (dow.specials.empty() ? "" : " of the month");
329 };
330 const bool domFull = dom.values.size() == 31, dowFull = dow.values.size() == 7;
331 string day;
332 if (cron.useOr) {
333 if (!domFull && !dowFull) day = ", on " + domClause + " or on " + dowClause();
334 } else if (!domFull && !dowFull) day = ", on " + domClause + " if it is " + dowClause("or");
335 else if (!domFull) day = ", on " + domClause;
336 else if (!dowFull) day = ", on " + dowClause();
337 return time + day + (month.size() == 12 ? "" : ", in " + valueList(month, 3));
338}

361 void newId(const string& id) const {
362 if (id.empty() || id.size() > 16 || id[0] < 'a' || id[0] > 'z' ||
363 !all_of(id.begin(), id.end(), [](char ch) { return (ch >= 'a' && ch <= 'z') || (ch >= '0' && ch <= '9') || ch == '_'; })) fail("invalid id");
364 if (jobs.count(id)) fail("duplicate job");

371 pair<int, int> window(const vector<string>& args, size_t start) const {
372 const int low = parseDatetime(args[start], args[start + 1]);
373 const int high = parseDatetime(args[start + 2], args[start + 3]);
374 if (low > high) fail("invalid window");
375 if (high - low > 44640) fail("window too long");
376 return {low, high};
377 }
378 vector<int> fireTimes(const Job& job, int low, int high, int limit = 0, bool reverse = false) const {
379 vector<int> result;
380 job.cron.eachFire(low, high, holidays, [&](int minute) {
381 result.push_back(minute);
382 return limit == 0 || result.size() < static_cast<size_t>(limit);
383 }, reverse);
384 return result;
385 }
386 map<int, vector<string>> grouped(int low, int high) const {
387 map<int, vector<string>> groups;
388 for (const auto& [id, job] : jobs) if (job.enabled) {
389 job.cron.eachFire(low, high, holidays, [&](int minute) { groups[minute].push_back(id); return true; });
390 }
391 return groups;
392 }
393 vector<string> simulate(int low, int high) const {
394 map<int, int> events;
395 vector<string> rows;
396 for (const auto& [id, job] : jobs) if (job.enabled) {
397 optional<int> lastEnd;
398 int started = 0, skipped = 0, replaced = 0, busy = 0;
399 job.cron.eachFire(low, high, holidays, [&](int minute) {
400 if (lastEnd && *lastEnd > minute) {
401 if (job.policy == Policy::Forbid) { ++skipped; return true; }
402 if (job.policy == Policy::Replace) {
403 ++events[*lastEnd]; --events[minute];
404 busy -= *lastEnd - minute;
405 ++replaced;
406 }
407 }
408 lastEnd = minute + job.duration;
409 ++events[minute]; --events[*lastEnd];
410 busy += job.duration;
411 ++started;
412 return true;
413 });
414 rows.push_back(id + " " + policyText(job.policy) + " started " + to_string(started) + " skipped " + to_string(skipped) +
415 " replaced " + to_string(replaced) + " busy " + to_string(busy));
416 }
417 int running = 0, peak = 0, earliest = 0;
418 for (const auto& [minute, difference] : events) {
419 running += difference;
420 if (running > peak) { peak = running; earliest = minute; }
421 }
422 rows.insert(rows.begin(), "SIMULATE " + to_string(rows.size()) + " peak " + to_string(peak) +
423 (peak ? " at " + datetimeNotation(earliest) : ""));
424 return rows;
425 }
426 vector<string> week(int first, const string& text) const {
427 vector<string> rows;
428 int total = 0;
429 for (int day = first; day < first + 7; ++day) {
430 array<int, 24> hours{};
431 const auto date = civil(day);
432 if (!holidays.count(day)) for (const auto& [id, job] : jobs) if (job.enabled && job.cron.scheduled(date)) {
433 for (int hour : job.cron.fields[1].values) hours[hour] += static_cast<int>(job.cron.fields[0].values.size());
434 }
435 int count = 0;
436 string cells;
437 for (int value : hours) { count += value; cells += value == 0 ? '.' : value < 10 ? static_cast<char>('0' + value) : '+'; }
438 total += count;
439 rows.push_back(dateNotation(day) + " " + cells + " " + to_string(count));
440 }
441 rows.insert(rows.begin(), "WEEK " + text + " " + to_string(total));
442 return rows;
443 }
444 vector<string> calendar(const Job& job, int year, int month, const string& text) const {
445 const int first = dayNumber(year, month, 1), length = monthLength(year, month);
446 vector<string> cells((first + 6) % 7, "...");
447 int active = 0;
448 for (int day = 1; day <= length; ++day) {
449 const int number = first + day - 1;
450 const char mark = !job.cron.scheduled(civil(number)) ? '-' : holidays.count(number) ? 'x' : '*';
451 if (mark == '*') ++active;
452 cells.push_back(pad(day) + mark);
453 }
454 while (cells.size() % 7) cells.push_back("...");
455 vector<string> rows = {"CALENDAR " + job.id + " " + text + " days " + to_string(active) + " runs " +
456 to_string(active * static_cast<int>(job.cron.times.size())), "Sun Mon Tue Wed Thu Fri Sat"};
457 for (size_t index = 0; index < cells.size(); index += 7)
458 rows.push_back(join(vector<string>(cells.begin() + index, cells.begin() + index + 7)));
459 return rows;
460 }
461 vector<string> gap(const Job& job, int base) const {
462 const auto times = fireTimes(job, base, base + 44640);
463 const string prefix = "GAP " + job.id + " runs " + to_string(times.size());
464 if (times.size() < 2) return {prefix + " none"};
465 map<int, int> counts;
466 int shortest = times[1] - times[0], longest = shortest;
467 size_t from = 0;
468 for (size_t index = 1; index < times.size(); ++index) {
469 const int difference = times[index] - times[index - 1];
470 ++counts[difference]; shortest = min(shortest, difference);
471 if (difference > longest) { longest = difference; from = index - 1; }
472 }
473 vector<string> rows = {prefix + " shortest " + spanText(shortest) + " longest " + spanText(longest) + " from " +
474 datetimeNotation(times[from]) + " to " + datetimeNotation(times[from + 1])};
475 for (const auto& [difference, count] : counts) rows.push_back(spanText(difference) + " " + to_string(count));
476 return rows;
477 }

481};

483vector<string> Scheduler::execute(const string& line) {
484 const auto tokens = words(line);
485 const string& command = tokens[0];
486 const vector<string> args(tokens.begin() + 1, tokens.end());
487 static const map<string, size_t> counts = {
488 {"REMOVE", 1}, {"RENAME", 2}, {"ENABLE", 1}, {"DISABLE", 1}, {"RUNTIME", 3}, {"LIST", 0},
489 {"EXPAND", 2}, {"DESCRIBE", 1}, {"MATCH", 3}, {"NEXT", 4}, {"PREV", 4}, {"COUNT", 5},
490 {"GAP", 3}, {"OVERLAP", 6}, {"TIMELINE", 4}, {"CONFLICTS", 4}, {"WEEK", 1},
491 {"CALENDAR", 2}, {"SIMULATE", 4}, {"EXCLUDE", 1}, {"INCLUDE", 1}, {"HOLIDAYS", 0}
492 };
493 if (command == "JOB" || command == "VALIDATE") {
494 if (args.size() < (command == "JOB" ? 2U : 1U)) fail("wrong arguments");
495 if (command == "VALIDATE") {
496 const Cron cron(args);
497 return {"VALID " + cron.expanded + (cron.useOr ? " OR " : " AND ") + to_string(cron.times.size())};
498 }
499 const string& id = args[0];
500 newId(id);
501 Cron cron(vector<string>(args.begin() + 1, args.end()));
502 jobs.emplace(id, Job{id, move(cron)});
503 return {"JOB " + id + " added"};
504 }
505 const auto expected = counts.find(command);
506 if (expected == counts.end()) fail("unknown command");
507 if (args.size() != expected->second) fail("wrong arguments");
508 if (command == "LIST") {
509 vector<string> rows = {"LIST " + to_string(jobs.size())};
510 for (const auto& [id, job] : jobs) rows.push_back(id + (job.enabled ? " on " : " off ") + to_string(job.duration) + " " + policyText(job.policy) + " " + job.cron.original);
511 return rows;
512 }
513 if (command == "HOLIDAYS") {
514 vector<string> rows = {"HOLIDAYS " + to_string(holidays.size())};
515 for (int day : holidays) rows.push_back(dateNotation(day));
516 return rows;
517 }
518 if (command == "EXCLUDE" || command == "INCLUDE") {
519 const int day = parseDate(args[0]);
520 const bool exclude = command == "EXCLUDE";
521 if (static_cast<bool>(holidays.count(day)) == exclude) fail(exclude ? "already excluded" : "not excluded");
522 if (exclude) holidays.insert(day); else holidays.erase(day);
523 return {command + " " + dateNotation(day) + " holidays " + to_string(holidays.size())};
524 }
525 if (command == "WEEK") return week(parseDate(args[0]), args[0]);
526 if (command == "TIMELINE" || command == "CONFLICTS" || command == "SIMULATE") {
527 const auto [low, high] = window(args, 0);
528 if (command == "SIMULATE") return simulate(low, high);
529 const auto groups = grouped(low, high);
530 vector<string> rows;
531 int total = 0, peak = 0, shown = 0;
532 for (const auto& [minute, ids] : groups) {
533 if (command == "TIMELINE") {
534 total += static_cast<int>(ids.size());
535 for (const auto& id : ids) if (shown < 100) { rows.push_back(datetimeNotation(minute) + " " + id); ++shown; }
536 } else if (ids.size() >= 2) {
537 ++total; peak = max(peak, static_cast<int>(ids.size()));
538 if (shown < 100) { rows.push_back(datetimeNotation(minute) + " " + to_string(ids.size()) + " " + join(ids)); ++shown; }
539 }
540 }
541 rows.insert(rows.begin(), command + " " + to_string(total) + (command == "CONFLICTS" ? " peak " + to_string(peak) : ""));
542 if (total > 100) rows.push_back("... " + to_string(total - 100) + " more");
543 return rows;
544 }
545 Job& job = existing(args[0]);
546 if (command == "REMOVE") {
547 const string id = job.id;
548 jobs.erase(id);
549 return {"REMOVE " + id};
550 }
551 if (command == "RENAME") {
552 newId(args[1]);
553 const string old = job.id;
554 auto node = jobs.extract(old);
555 node.key() = args[1]; node.mapped().id = args[1];
556 jobs.insert(move(node));
557 return {"RENAME " + old + " " + args[1]};
558 }
559 if (command == "ENABLE" || command == "DISABLE") {
560 const bool enabled = command == "ENABLE";
561 if (job.enabled == enabled) fail(enabled ? "already enabled" : "already disabled");
562 job.enabled = enabled;
563 return {command + " " + job.id};
564 }
565 if (command == "RUNTIME") {
566 const int duration = boundedNumber(args[1], 4, 1440, "invalid duration");
567 const Policy policy = parsePolicy(args[2]);
568 job.duration = duration; job.policy = policy;
569 return {"RUNTIME " + job.id + " " + to_string(duration) + " " + policyText(policy)};
570 }
571 if (command == "EXPAND") {
572 const auto found = find(fieldNames.begin(), fieldNames.end(), args[1]);
573 if (found == fieldNames.end()) fail("invalid field");
574 const auto& field = job.cron.fields[found - fieldNames.begin()];
575 vector<string> parts = {"EXPAND", job.id, args[1], to_string(field.values.size())};
576 for (int value : field.values) parts.push_back(to_string(value));
577 for (const auto& special : field.specials) parts.push_back(special.text);
578 return {join(parts)};
579 }
580 if (command == "DESCRIBE") return {"DESCRIBE " + job.id + " " + description(job.cron)};
581 if (command == "CALENDAR") {
582 const auto [year, month] = parseMonth(args[1]);
583 return calendar(job, year, month, args[1]);
584 }
585 if (command == "OVERLAP") {
586 const Job& second = existing(args[1]);
587 const auto [low, high] = window(args, 2);
588 const auto firstTimes = fireTimes(job, low, high), secondTimes = fireTimes(second, low, high);
589 vector<int> shared;
590 set_intersection(firstTimes.begin(), firstTimes.end(), secondTimes.begin(), secondTimes.end(), back_inserter(shared));
591 string row = "OVERLAP " + job.id + " " + second.id + " " + to_string(shared.size());
592 if (!shared.empty()) row += " first " + datetimeNotation(shared.front()) + " last " + datetimeNotation(shared.back());
593 return {row};
594 }
595 if (command == "COUNT") {
596 const auto [low, high] = window(args, 1);
597 set<int> days;
598 int runs = 0;
599 job.cron.eachFire(low, high, holidays, [&](int minute) { ++runs; days.insert(minute / 1440); return true; });
600 return {"COUNT " + job.id + " runs " + to_string(runs) + " days " + to_string(days.size())};
601 }
602 const int base = parseDatetime(args[1], args[2]);
603 if (command == "MATCH") {
604 const auto reasons = job.cron.reasons(base, holidays);
605 return {"MATCH " + job.id + " " + datetimeNotation(base) + (reasons.empty() ? " yes" : " no " + join(reasons))};
606 }
607 if (command == "GAP") return gap(job, base);
608 const int count = boundedNumber(args[3], 2, 50, "invalid count");
609 const bool reverse = command == "PREV";
610 const auto times = fireTimes(job, reverse ? base - 527040 : base + 1, reverse ? base - 1 : base + 527040, count, reverse);
611 vector<string> rows = {command + " " + job.id + " " + to_string(times.size())};
612 for (int minute : times) rows.push_back(datetimeNotation(minute) + (reverse ? " -" : " +") + spanText(reverse ? base - minute : minute - base));
613 if (times.size() < static_cast<size_t>(count)) rows.push_back("NONE");
614 return rows;
615}

617int main() {
618 ios::sync_with_stdio(false);
619 cin.tie(nullptr);
620 int count;
621 cin >> count;
622 string line;
623 getline(cin, line);
624 Scheduler scheduler;
625 for (int index = 0; index < count; ++index) {
626 getline(cin, line);
627 try { for (const auto& row : scheduler.execute(line)) cout << row << '\n'; }
628 catch (const CommandError& error) { cout << "ERROR " << error.what() << '\n'; }
629 }
