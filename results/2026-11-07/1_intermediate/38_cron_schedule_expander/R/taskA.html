97field_value <- function(field, text) {
98 names <- if (field == 4L) toupper(substr(months, 1L, 3L)) else if (field == 5L) toupper(substr(weekdays, 1L, 3L)) else character()
99 index <- match(text, names)
100 if (!is.na(index)) return(index - as.integer(field == 5L))
101 if (!small_digits(text)) field_error(field, "invalid value")
102 value <- as.integer(text)

117parse_field <- function(text, index) {
118 values <- integer()
119 specials <- list()
120 for (item in split_items(text)) {
121 if (item == "") field_error(index, "empty item")
122 special <- NULL
123 if (index == 3L && (startsWith(item, "L") || endsWith(item, "W"))) {
124 special <- if (item == "L") new_special("last") else if (item == "LW") new_special("workday") else if (startsWith(item, "L-")) {
125 new_special("before", special_number(index, substring(item, 3L), 2L, 30L))
126 } else if (endsWith(item, "W")) {
127 new_special("nearest", special_number(index, substr(item, 1L, nchar(item) - 1L), 2L, 31L))
128 } else field_error(index, "invalid value")
129 } else if (index == 5L && grepl("#", item, fixed = TRUE)) {
130 hash <- regexpr("#", item, fixed = TRUE)[[1L]]
131 value <- field_value(index, substr(item, 1L, hash - 1L))
132 nth <- special_number(index, substring(item, hash + 1L), 1L, 5L)
133 special <- new_special("nth", value %% 7L, nth)
134 } else if (index == 5L && nchar(item) >= 2L && endsWith(item, "L")) {
135 value <- field_value(index, substr(item, 1L, nchar(item) - 1L))
136 special <- new_special("weekday", value %% 7L)
137 } else {
138 slash <- regexpr("/", item, fixed = TRUE)[[1L]]
139 base <- if (slash < 0L) item else substr(item, 1L, slash - 1L)
140 dash <- regexpr("-", base, fixed = TRUE)[[1L]]
141 if (base == "*") {
142 first <- minimum[index]
143 last <- maximum[index]
144 } else if (dash > 0L) {
145 first <- field_value(index, substr(base, 1L, dash - 1L))
146 last <- field_value(index, substring(base, dash + 1L))
147 if (first > last) field_error(index, "invalid range")
148 } else {
149 first <- field_value(index, base)
150 last <- if (slash < 0L) first else maximum[index]
151 }
152 step <- 1L
153 if (slash > 0L) {
154 part <- substring(item, slash + 1L)
155 if (!small_digits(part)) field_error(index, "invalid step")
156 step <- as.integer(part)
157 if (step == 0L) field_error(index, "invalid step")
158 }
159 expanded <- seq.int(first, last, by = step)
160 if (index == 5L) expanded <- expanded %% 7L
161 values <- c(values, expanded)
162 }
163 if (!is.null(special) && !special$text %in% vapply(specials, function(s) s$text, character(1L))) specials <- c(specials, list(special))
164 }
165 list(index = index, values = sort(unique(values), method = "radix"), specials = specials, restricted = !startsWith(text, "*"))
166}
167parse_cron <- function(tokens) {
168 text <- joined(tokens)
169 if (length(tokens) == 1L && startsWith(tokens, "@")) {
170 if (!tokens %in% names(macros)) fail("unknown macro")
171 tokens <- words(macros[[tokens]])
172 }
173 if (length(tokens) != 5L) fail("field count")
174 fields <- lapply(seq_len(5L), function(index) parse_field(tokens[index], index))
175 minutes <- fields[[1L]]$values
176 hours <- fields[[2L]]$values
177 times <- rep(hours, each = length(minutes)) * 60L + rep(minutes, times = length(hours))
178 list(text = text, expanded = joined(tokens), fields = fields,
179 disjunction = fields[[3L]]$restricted && fields[[5L]]$restricted, times = times)
180}

188day_status <- function(cron, day) {
189 date <- calendar[day + 1L, ]
190 month <- date[["month"]]
191 d <- date[["day"]]
192 e <- date[["length"]]
193 w <- date[["weekday"]]
194 first <- day - d + 1L
195 dom <- d %in% cron$fields[[3L]]$values
196 dow <- w %in% cron$fields[[5L]]$values
197 for (special in cron$fields[[3L]]$specials) {
198 target <- switch(special$kind, last = e, workday = nearest_workday(first, e, e),
199 before = e - special$value, nearest = if (special$value <= e) nearest_workday(first, special$value, e) else 0L)
200 dom <- dom || d == target
201 }
202 for (special in cron$fields[[5L]]$specials) {
203 matches <- if (special$kind == "nth") (d - 1L) %/% 7L + 1L == special$nth else d + 7L > e
204 dow <- dow || (w == special$value && matches)
205 }
206 c(month = month %in% cron$fields[[4L]]$values, dom = dom, dow = dow,
207 condition = if (cron$disjunction) dom || dow else dom && dow)
208}
209scheduled_day <- function(cron, day) {
210 status <- day_status(cron, day)
211 status[["month"]] && status[["condition"]]

213fire_times <- function(job, lo, hi, holidays, reverse = FALSE, limit = 0L) {
214 days <- seq.int(lo %/% 1440L, hi %/% 1440L)
215 daily <- job$cron$times
216 if (reverse) {
217 days <- rev(days)
218 daily <- rev(daily)
219 }
220 result <- integer()
221 for (day in days) {
222 if (day %in% holidays || !scheduled_day(job$cron, day)) next
223 times <- day * 1440L + daily
224 result <- c(result, times[lo <= times & times <= hi])
225 if (limit > 0L && length(result) >= limit) return(head(result, limit))
226 }
227 result

238 values <- field$values
239 parts <- character()

275description <- function(cron) {
276 minute <- cron$fields[[1L]]
277 hour <- cron$fields[[2L]]
278 if (length(cron$times) <= 4L) {
279 result <- paste0("At ", word_list(clock_text(cron$times)))
280 } else {
281 ms <- stepped(minute)
282 hs <- stepped(hour)
283 result <- if (length(minute$values) == 60L) "Every minute" else if (ms > 0L) paste("Every", ms, "minutes") else if (length(minute$values) == 1L) {
284 paste("At minute", minute$values[1L])
285 } else paste0("At minutes ", value_list(minute))
286 if (length(hour$values) == 24L) {
287 if (startsWith(result, "At minute")) result <- paste0(result, " of every hour")
288 } else if (hs > 0L) result <- paste0(result, ", every ", hs, " hours") else {
289 first <- head(hour$values, 1L)
290 last <- tail(hour$values, 1L)
291 result <- paste0(result, if (last - first + 1L == length(hour$values)) sprintf(", between %02d:00 and %02d:59", first, last) else paste0(", during hours ", value_list(hour)))
292 }
293 }
294 dom <- cron$fields[[3L]]
295 dow <- cron$fields[[5L]]
296 dfull <- length(dom$values) == 31L
297 ufull <- length(dow$values) == 7L
298 if (cron$disjunction) {
299 if (!dfull && !ufull) result <- paste0(result, ", on ", dom_phrase(dom), " or on ", dow_phrase(dow))
300 } else if (!dfull && !ufull) result <- paste0(result, ", on ", dom_phrase(dom), " if it is ", dow_phrase(dow, TRUE)) else if (!dfull) {
301 result <- paste0(result, ", on ", dom_phrase(dom))
302 } else if (!ufull) result <- paste0(result, ", on ", dow_phrase(dow))
303 if (length(cron$fields[[4L]]$values) != 12L) result <- paste0(result, ", in ", value_list(cron$fields[[4L]]))
304 result
305}

416simulation_report <- function(state, lo, hi) {
417 jobs <- ordered_jobs(state, TRUE)
418 events <- integer(hi - lo + 1441L)
419 details <- character()
420 for (job in jobs) {
421 times <- fire_times(job, lo, hi, state$holidays)
422 starts <- times
423 ends <- times + job$duration
424 skipped <- 0L
425 replaced <- 0L
426 if (job$policy == "FORBID") {
427 accepted <- 0L
428 last_end <- lo
429 for (time in times) if (time >= last_end) {
430 accepted <- accepted + 1L
431 starts[accepted] <- time
432 last_end <- time + job$duration
433 ends[accepted] <- last_end
434 }
435 starts <- starts[seq_len(accepted)]
436 ends <- ends[seq_len(accepted)]
437 skipped <- length(times) - accepted
438 } else if (job$policy == "REPLACE" && length(times) > 1L) {
439 next_start <- tail(starts, -1L)
440 previous_end <- head(ends, -1L)
441 replaced <- sum(previous_end > next_start)
442 ends <- c(pmin(previous_end, next_start), tail(ends, 1L))
443 }
444 events <- events + tabulate(starts - lo + 1L, nbins = length(events)) - tabulate(ends - lo + 1L, nbins = length(events))
445 details <- c(details, paste(job$id, job$policy, "started", length(starts), "skipped", skipped,
446 "replaced", replaced, "busy", sum(ends - starts)))
447 }
448 running <- cumsum(events)
449 peak <- max(running)
450 header <- paste("SIMULATE", length(jobs), "peak", peak)
451 if (peak > 0L) header <- paste(header, "at", datetime_text(lo + which.max(running) - 1L))
452 c(header, details)
453}

455execute <- function(state, line) {
456 tokens <- words(line)
457 command <- tokens[1L]
458 tokens <- tokens[-1L]
459 if (!command %in% c("JOB", "VALIDATE", names(signatures))) fail("unknown command")
460 if (command %in% c("JOB", "VALIDATE")) {
461 if (length(tokens) < if (command == "JOB") 2L else 1L) fail("wrong arguments")
462 if (command == "JOB") {
463 id <- new_id(state, tokens[1L])
464 cron <- parse_cron(tokens[-1L])
465 state$jobs[[id]] <- list(id = id, cron = cron, enabled = TRUE, duration = 1L, policy = "ALLOW")
466 return(paste("JOB", id, "added"))
467 }
468 cron <- parse_cron(tokens)
469 return(paste("VALID", cron$expanded, if (cron$disjunction) "OR" else "AND", length(cron$times)))
470 }
471 types <- signatures[[command]]
472 if (length(tokens) != sum(ifelse(types == "datetime", 2L, 1L))) fail("wrong arguments")
473 args <- parse_arguments(state, types, tokens)
474 if (length(types) && types[1L] == "id") job <- args[[1L]]
475 switch(command,
476 REMOVE = {
477 state$jobs[[job$id]] <- NULL
478 paste("REMOVE", job$id)
479 },
480 RENAME = {
481 old_id <- job$id
482 job$id <- args[[2L]]
