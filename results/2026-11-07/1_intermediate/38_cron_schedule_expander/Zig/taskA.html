36 }

39 }

45fn leap(year: usize) bool {
46 return year % 4 == 0 and (year % 100 != 0 or year % 400 == 0);
47}
48fn monthLength(year: usize, month: usize) usize {
49 const lengths = [_]usize{ 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31 };
50 return if (month == 2 and leap(year)) 29 else lengths[month - 1];
51}
52const Date = struct {
53 year: usize,
54 month: usize,
55 day: usize,
56 fn number(date: Date) usize {
57 var result = date.day - 1;
58 for (2000..date.year) |year| result += if (leap(year)) 366 else 365;
59 for (1..date.month) |month| result += monthLength(date.year, month);
60 return result;
61 }
62 fn weekday(date: Date) usize {
63 return (date.number() + 6) % 7;
64 }
65 fn fromNumber(value: usize) Date {
66 var rest = value;
67 var year: usize = 2000;
68 while (rest >= @as(usize, if (leap(year)) 366 else 365)) : (year += 1) rest -= if (leap(year)) 366 else 365;
69 var month: usize = 1;
70 while (rest >= monthLength(year, month)) : (month += 1) rest -= monthLength(year, month);
71 return .{ .year = year, .month = month, .day = rest + 1 };
72 }
73 fn parse(text: []const u8) ?Date {
74 if (text.len != 10 or text[4] != '-' or text[7] != '-' or !digits(text[0..4], 4, 4) or !digits(text[5..7], 2, 2) or !digits(text[8..], 2, 2)) return null;
75 const result: Date = .{ .year = integer(text[0..4]), .month = integer(text[5..7]), .day = integer(text[8..]) };
76 if (result.year < 2001 or result.year > 2098 or result.month < 1 or result.month > 12 or result.day < 1 or result.day > monthLength(result.year, result.month)) return null;
77 return result;
78 }

102const Special = struct {
103 kind: enum { last, last_weekday, before_last, nearest_weekday, nth_weekday, last_occurrence },
104 value: usize = 0,
105 occurrence: usize = 0,
106 fn eql(a: Special, b: Special) bool {
107 return a.kind == b.kind and a.value == b.value and a.occurrence == b.occurrence;
108 }
109 pub fn format(value: Special, out: *Writer) Writer.Error!void {
110 switch (value.kind) {
111 .last => try out.writeAll("L"),
112 .last_weekday => try out.writeAll("LW"),
113 .before_last => try out.print("L-{d}", .{value.value}),
114 .nearest_weekday => try out.print("{d}W", .{value.value}),
115 .nth_weekday => try out.print("{d}#{d}", .{ value.value, value.occurrence }),
116 .last_occurrence => try out.print("{d}L", .{value.value}),
117 }
118 }

120const Field = struct {
121 values: [60]bool = @splat(false),
122 specials: [50]Special = undefined,
123 special_count: usize = 0,
124 restricted: bool = false,
125 fn count(field: *const Field) usize {
126 var result: usize = 0;
127 for (field.values) |present| result += @intFromBool(present);
128 return result;
129 }
130 fn addSpecial(field: *Field, special: Special) void {
131 for (field.specials[0..field.special_count]) |old| if (old.eql(special)) return;
132 field.specials[field.special_count] = special;
133 field.special_count += 1;
134 }
135 fn stepped(field: *const Field, index: usize) usize {
136 if (!field.values[0] or field.count() < 2) return 0;
137 var step: usize = 1;
138 while (!field.values[step]) : (step += 1) {}
139 if (step == 1) return 0;
140 for (0..maximum[index] + 1) |value| if (field.values[value] != (value % step == 0)) return 0;
141 return step;
142 }

144const DayMatch = struct { month_day: bool, week_day: bool, condition: bool };
145const Cron = struct {
146 fields: [5]Field,
147 expanded: Text(100),
148 daily: [1440]u16 = undefined,
149 daily_count: usize = 0,
150 fn orRule(cron: *const Cron) bool {
151 return cron.fields[2].restricted and cron.fields[4].restricted;
152 }
153 fn rule(cron: *const Cron) []const u8 {
154 return if (cron.orRule()) "OR" else "AND";
155 }
156 fn nearest(date: Date, day: usize) usize {
157 const weekday = (date.number() + 6 + day - date.day) % 7;
158 return switch (weekday) {
159 6 => if (day == 1) 3 else day - 1,
160 0 => if (day == monthLength(date.year, date.month)) day - 2 else day + 1,
161 else => day,
162 };
163 }
164 fn matchDay(cron: *const Cron, date: Date) DayMatch {
165 const last = monthLength(date.year, date.month);
166 var month_day = cron.fields[2].values[date.day];
167 var week_day = cron.fields[4].values[date.weekday()];
168 for (cron.fields[2].specials[0..cron.fields[2].special_count]) |special| {
169 const matches = switch (special.kind) {
170 .last => date.day == last,
171 .last_weekday => date.day == nearest(date, last),
172 .before_last => special.value < last and date.day == last - special.value,
173 .nearest_weekday => special.value <= last and date.day == nearest(date, special.value),
174 else => unreachable,
175 };
176 month_day = month_day or matches;
177 }
178 for (cron.fields[4].specials[0..cron.fields[4].special_count]) |special| {
179 const matches = date.weekday() == special.value and (if (special.kind == .nth_weekday) (date.day - 1) / 7 + 1 == special.occurrence else date.day + 7 > last);
180 week_day = week_day or matches;
181 }
182 return .{ .month_day = month_day, .week_day = week_day, .condition = if (cron.orRule()) month_day or week_day else month_day and week_day };
183 }
184 fn scheduled(cron: *const Cron, date: Date) bool {
185 return cron.fields[3].values[date.month] and cron.matchDay(date).condition;
186 }
187 fn fires(cron: *const Cron, minute: usize, holidays: *const [36526]bool) bool {
188 return cron.fields[0].values[minute % 60] and cron.fields[1].values[minute % 1440 / 60] and !holidays[minute / 1440] and cron.scheduled(Date.fromNumber(minute / 1440));
189 }
190 fn iterator(cron: *const Cron, holidays: *const [36526]bool, low: usize, high: usize, reverse: bool) FireIterator {
191 return .{ .cron = cron, .holidays = holidays, .low = low, .high = high, .first = @intCast(low / 1440), .last = @intCast(high / 1440), .day = @intCast(if (reverse) high / 1440 else low / 1440), .index = if (reverse) @as(i32, @intCast(cron.daily_count)) - 1 else 0, .reverse = reverse };
192 }
193};
194const FireIterator = struct {
195 cron: *const Cron,
196 holidays: *const [36526]bool,
197 low: usize,
198 high: usize,
199 first: i32,
200 last: i32,
201 day: i32,
202 index: i32,
203 reverse: bool,
204 ready: bool = false,
205 fn advance(iterator: *FireIterator) void {
206 iterator.day += if (iterator.reverse) @as(i32, -1) else 1;
207 iterator.index = if (iterator.reverse) @as(i32, @intCast(iterator.cron.daily_count)) - 1 else 0;
208 iterator.ready = false;
209 }
210 fn next(iterator: *FireIterator) ?usize {
211 while (iterator.day >= iterator.first and iterator.day <= iterator.last) {
212 const day: usize = @intCast(iterator.day);
213 if (!iterator.ready) {
214 if (iterator.holidays[day] or !iterator.cron.scheduled(Date.fromNumber(day))) {
215 iterator.advance();
216 continue;
217 }
218 iterator.ready = true;
219 }
220 while (iterator.index >= 0 and iterator.index < @as(i32, @intCast(iterator.cron.daily_count))) {
221 const time = day * 1440 + iterator.cron.daily[@intCast(iterator.index)];
222 iterator.index += if (iterator.reverse) @as(i32, -1) else 1;
223 if (time >= iterator.low and time <= iterator.high) return time;
224 }
225 iterator.advance();
226 }
227 return null;
228 }

271 fn readValue(session: *Session, text: []const u8, index: usize) Error!usize {
272 const month_names = [_][]const u8{ "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV", "DEC" };
273 const day_names = [_][]const u8{ "SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT" };
274 if (index == 3) {
275 for (month_names, 1..) |name, value| if (equal(text, name)) return value;
276 }
277 if (index == 4) {
278 for (day_names, 0..) |name, value| if (equal(text, name)) return value;
279 }
280 if (!digits(text, 1, 2)) return session.fieldError(index, "invalid value");
281 const value = integer(text);
282 if (value < minimum[index] or value > maximum[index]) return session.fieldError(index, "out of range");
283 return value;
284 }
285 fn specialNumber(session: *Session, text: []const u8, index: usize, upper: usize, size: usize) Error!usize {
286 if (!digits(text, 1, size)) return session.fieldError(index, "invalid value");
287 const value = integer(text);
288 if (value < 1 or value > upper) return session.fieldError(index, "out of range");
289 return value;
290 }
291 fn readField(session: *Session, text: []const u8, index: usize) Error!Field {
292 var result: Field = .{ .restricted = text[0] != '*' };
293 var items = std.mem.splitScalar(u8, text, ',');
294 while (items.next()) |item| {
295 if (item.len == 0) return session.fieldError(index, "empty item");
296 var special: ?Special = null;
297 if (index == 2 and (item[0] == 'L' or item[item.len - 1] == 'W')) {
298 if (equal(item, "L")) special = .{ .kind = .last } else if (equal(item, "LW")) special = .{ .kind = .last_weekday } else if (std.mem.startsWith(u8, item, "L-")) special = .{ .kind = .before_last, .value = try session.specialNumber(item[2..], index, 30, 2) } else if (item[item.len - 1] == 'W') special = .{ .kind = .nearest_weekday, .value = try session.specialNumber(item[0 .. item.len - 1], index, 31, 2) } else return session.fieldError(index, "invalid value");
299 } else if (index == 4 and std.mem.findScalar(u8, item, '#') != null) {
300 const hash = std.mem.findScalar(u8, item, '#').?;
301 const weekday = try session.readValue(item[0..hash], index);
302 const occurrence = try session.specialNumber(item[hash + 1 ..], index, 5, 1);
303 special = .{ .kind = .nth_weekday, .value = weekday % 7, .occurrence = occurrence };
304 } else if (index == 4 and item.len >= 2 and item[item.len - 1] == 'L') {
305 special = .{ .kind = .last_occurrence, .value = (try session.readValue(item[0 .. item.len - 1], index)) % 7 };
306 }
307 if (special) |value| {
308 result.addSpecial(value);
309 continue;
310 }
311 const slash = std.mem.findScalar(u8, item, '/');
312 const base = item[0 .. slash orelse item.len];
313 var low: usize = undefined;
314 var high: usize = undefined;
315 if (equal(base, "*")) {
316 low = minimum[index];
317 high = maximum[index];
318 } else if (std.mem.findScalar(u8, base, '-')) |dash| {
319 low = try session.readValue(base[0..dash], index);
320 high = try session.readValue(base[dash + 1 ..], index);
321 if (low > high) return session.fieldError(index, "invalid range");
322 } else {
323 low = try session.readValue(base, index);
324 high = if (slash == null) low else maximum[index];
325 }
326 var step: usize = 1;
327 if (slash) |position| {
328 const part = item[position + 1 ..];
329 if (!digits(part, 1, 2) or integer(part) == 0) return session.fieldError(index, "invalid step");
330 step = integer(part);
331 }
332 var value = low;
333 while (value <= high) : (value += step) result.values[if (index == 4) value % 7 else value] = true;
334 }
335 return result;
336 }
337 fn joinWords(session: *Session, words: []const []const u8) Error![]const u8 {
338 var result: Writer.Allocating = .init(session.gpa);
339 defer result.deinit();
340 for (words, 0..) |word, index| {
341 if (index != 0) try result.writer.writeByte(' ');
342 try result.writer.writeAll(word);
343 }
344 return result.toOwnedSlice();
345 }
346 fn parseCron(session: *Session, input: []const []const u8) Error!Cron {
347 var words = input;
348 if (words.len == 1 and words[0][0] == '@') {
349 const macro = words[0];
350 words = if (equal(macro, "@yearly") or equal(macro, "@annually")) &.{ "0", "0", "1", "1", "*" } else if (equal(macro, "@monthly")) &.{ "0", "0", "1", "*", "*" } else if (equal(macro, "@weekly")) &.{ "0", "0", "*", "*", "0" } else if (equal(macro, "@daily") or equal(macro, "@midnight")) &.{ "0", "0", "*", "*", "*" } else if (equal(macro, "@hourly")) &.{ "0", "*", "*", "*", "*" } else return session.fail("unknown macro", .{});
351 }
352 if (words.len != 5) return session.fail("field count", .{});
353 var result: Cron = .{ .fields = undefined, .expanded = Text(100).init(try session.joinWords(words)) };
354 for (words, 0..) |word, index| result.fields[index] = try session.readField(word, index);
355 for (0..24) |hour| {
356 if (!result.fields[1].values[hour]) continue;
357 for (0..60) |minute| {
358 if (!result.fields[0].values[minute]) continue;
359 result.daily[result.daily_count] = @intCast(hour * 60 + minute);
360 result.daily_count += 1;
361 }
362 }
363 return result;
364 }

417 fn valueList(session: *Session, field: *const Field, index: usize, either: bool) Error![]const u8 {
418 var parts: std.ArrayList([]const u8) = .empty;
419 var start: usize = 0;
420 while (start < 60) {
421 if (!field.values[start]) {
422 start += 1;
423 continue;
424 }
425 var end = start;
426 while (end + 1 < 60 and field.values[end + 1]) : (end += 1) {}
427 if (end - start >= 2) try parts.append(session.gpa, try std.fmt.allocPrint(session.gpa, "{s} through {s}", .{ try session.valueName(index, start), try session.valueName(index, end) })) else for (start..end + 1) |value| try parts.append(session.gpa, try session.valueName(index, value));
428 start = end + 1;
429 }
430 return session.listJoin(parts.items, either);
431 }
432 fn dayClause(session: *Session, cron: *const Cron, weekday: bool, either: bool) Error![]const u8 {
433 const index: usize = if (weekday) 4 else 2;
434 const field = &cron.fields[index];
435 var parts: std.ArrayList([]const u8) = .empty;
436 if (field.count() != 0) {
437 const values = try session.valueList(field, index, either);
438 try parts.append(session.gpa, if (weekday) values else try std.fmt.allocPrint(session.gpa, "{s} {s}", .{ if (field.count() == 1) "day" else "days", values }));
439 }
440 const ordinals = [_][]const u8{ "1st", "2nd", "3rd", "4th", "5th" };
441 for (field.specials[0..field.special_count]) |special| {
442 const part: []const u8 = switch (special.kind) {
443 .last => "the last day",
444 .last_weekday => "the last weekday",
445 .before_last => try std.fmt.allocPrint(session.gpa, "{d} {s} before the last day", .{ special.value, if (special.value == 1) "day" else "days" }),
446 .nearest_weekday => try std.fmt.allocPrint(session.gpa, "the weekday nearest day {d}", .{special.value}),
447 .nth_weekday => try std.fmt.allocPrint(session.gpa, "the {s} {s}", .{ ordinals[special.occurrence - 1], weekdays[special.value] }),
448 .last_occurrence => try std.fmt.allocPrint(session.gpa, "the last {s}", .{weekdays[special.value]}),
449 };
450 try parts.append(session.gpa, part);
451 }
452 return std.fmt.allocPrint(session.gpa, "{s}{s}", .{ try session.listJoin(parts.items, either), if (!weekday or field.special_count != 0) " of the month" else "" });
453 }
454 fn describe(session: *Session, cron: *const Cron) Error!void {
455 const out = session.out;
456 const minute = &cron.fields[0];
457 const hour = &cron.fields[1];
458 if (cron.daily_count <= 4) {
459 try out.writeAll("At ");
460 for (cron.daily[0..cron.daily_count], 0..) |time, index| {
461 if (index != 0) try out.writeAll(if (index + 1 == cron.daily_count) " and " else ", ");
462 try out.print("{d:0>2}:{d:0>2}", .{ time / 60, time % 60 });
463 }
464 } else {
465 const minute_step = minute.stepped(0);
466 if (minute.count() == 60) try out.writeAll("Every minute") else if (minute_step != 0) try out.print("Every {d} minutes", .{minute_step}) else try out.print("At {s} {s}", .{ if (minute.count() == 1) "minute" else "minutes", try session.valueList(minute, 0, false) });
467 if (hour.count() == 24) {
468 if (minute.count() != 60 and minute_step == 0) try out.writeAll(" of every hour");
469 } else if (hour.stepped(1) != 0) try out.print(", every {d} hours", .{hour.stepped(1)}) else {
470 var first: usize = 0;
471 while (!hour.values[first]) : (first += 1) {}
472 var last: usize = 23;
473 while (!hour.values[last]) : (last -= 1) {}
474 if (last - first + 1 == hour.count()) try out.print(", between {d:0>2}:00 and {d:0>2}:59", .{ first, last }) else try out.print(", during hours {s}", .{try session.valueList(hour, 1, false)});
475 }
476 }
477 const month_full = cron.fields[2].count() == 31;
478 const week_full = cron.fields[4].count() == 7;
479 if (cron.orRule()) {
480 if (!month_full and !week_full) try out.print(", on {s} or on {s}", .{ try session.dayClause(cron, false, false), try session.dayClause(cron, true, false) });
481 } else if (!month_full and !week_full) try out.print(", on {s} if it is {s}", .{ try session.dayClause(cron, false, false), try session.dayClause(cron, true, true) }) else if (!month_full) try out.print(", on {s}", .{try session.dayClause(cron, false, false)}) else if (!week_full) try out.print(", on {s}", .{try session.dayClause(cron, true, false)});
482 if (cron.fields[3].count() != 12) try out.print(", in {s}", .{try session.valueList(&cron.fields[3], 3, false)});
483 try out.writeByte('\n');
484 }
485 fn timeline(session: *Session, low: usize, high: usize, conflicts: bool) Error!void {
486 const jobs = session.state.ordered(true);
487 const masks = try session.gpa.alloc(u64, high - low + 1);
488 @memset(masks, 0);
489 for (jobs.values[0..jobs.len], 0..) |job, index| {
490 var times = job.cron.iterator(&session.state.holidays, low, high, false);
491 while (times.next()) |time| masks[time - low] |= @as(u64, 1) << @as(u6, @intCast(index));
492 }
493 var count: usize = 0;
494 var peak: usize = 0;
495 for (masks) |mask| {
496 const size = @popCount(mask);
497 if (conflicts) {
498 if (size < 2) continue;
499 count += 1;
500 peak = @max(peak, size);
501 } else count += size;
502 }
503 if (conflicts) try line(session.out, "CONFLICTS {d} peak {d}", .{ count, peak }) else try line(session.out, "TIMELINE {d}", .{count});
504 var written: usize = 0;
505 for (masks, 0..) |mask, offset| {
506 if (written == 100) break;
507 const size = @popCount(mask);
508 if (size == 0 or (conflicts and size < 2)) continue;
509 if (conflicts) try session.out.print("{f} {d}", .{ datetime(low + offset), size });
510 for (jobs.values[0..jobs.len], 0..) |job, index| {
511 if ((mask & (@as(u64, 1) << @as(u6, @intCast(index)))) == 0) continue;
512 if (conflicts) try session.out.print(" {f}", .{job.id}) else {
513 try line(session.out, "{f} {f}", .{ datetime(low + offset), job.id });
514 written += 1;
515 if (written == 100) break;
516 }
517 }
518 if (conflicts) {
519 try session.out.writeByte('\n');
520 written += 1;
521 }
522 }
523 if (count > 100) try line(session.out, "... {d} more", .{count - 100});
524 }
525 fn week(session: *Session, input: []const u8, date: Date) Error!void {
526 var counts: [7][24]usize = @splat(@splat(0));
527 const low = date.number() * 1440;
528 var total: usize = 0;
529 for (session.state.jobs[0..session.state.job_count]) |*job| {
530 if (!job.enabled) continue;
531 var times = job.cron.iterator(&session.state.holidays, low, low + 7 * 1440 - 1, false);
532 while (times.next()) |time| {
533 counts[(time - low) / 1440][time % 1440 / 60] += 1;
534 total += 1;
535 }
536 }
537 try line(session.out, "WEEK {s} {d}", .{ input, total });
538 for (counts, 0..) |hours, day| {
539 var marks: [24]u8 = undefined;
540 var subtotal: usize = 0;
541 for (hours, 0..) |count, hour| {
542 subtotal += count;
543 marks[hour] = if (count == 0) '.' else if (count >= 10) '+' else '0' + @as(u8, @intCast(count));
544 }
545 try line(session.out, "{f} {s} {d}", .{ Date.fromNumber(date.number() + day), &marks, subtotal });
546 }
547 }
548 fn calendar(session: *Session, job: *const Job, input: []const u8, first: Date) Error!void {
549 var cells: [42][3]u8 = @splat(.{ '.', '.', '.' });
550 const padding = first.weekday();
551 const length = monthLength(first.year, first.month);
552 var count: usize = 0;
553 for (1..length + 1) |day| {
554 const date: Date = .{ .year = first.year, .month = first.month, .day = day };
555 const mark: u8 = if (!job.cron.scheduled(date)) '-' else if (session.state.holidays[date.number()]) 'x' else '*';
556 count += @intFromBool(mark == '*');
557 cells[padding + day - 1] = .{ '0' + @as(u8, @intCast(day / 10)), '0' + @as(u8, @intCast(day % 10)), mark };
558 }
559 try line(session.out, "CALENDAR {f} {s} days {d} runs {d}", .{ job.id, input, count, count * job.cron.daily_count });
560 try line(session.out, "Sun Mon Tue Wed Thu Fri Sat", .{});
561 const end = (padding + length + 6) / 7 * 7;
562 for (cells[0..end], 0..) |cell, index| {
563 if (index % 7 != 0) try session.out.writeByte(' ');
564 try session.out.writeAll(&cell);
565 if (index % 7 == 6) try session.out.writeByte('\n');
566 }
567 }
568 fn simulate(session: *Session, low: usize, high: usize) Error!void {
569 const jobs = session.state.ordered(true);
570 const Stats = struct { started: usize = 0, skipped: usize = 0, replaced: usize = 0, busy: usize = 0 };
571 var stats: [40]Stats = @splat(.{});
572 const events = try session.gpa.alloc(i32, high - low + 1441);
573 @memset(events, 0);
574 for (jobs.values[0..jobs.len], 0..) |job, index| {
575 var times = job.cron.iterator(&session.state.holidays, low, high, false);
576 var last_end: usize = 0;
577 while (times.next()) |time| {
578 if (last_end > time) {
579 if (job.policy == .FORBID) {
580 stats[index].skipped += 1;
581 continue;
582 }
583 if (job.policy == .REPLACE) {
584 events[last_end - low] += 1;
585 events[time - low] -= 1;
586 stats[index].busy -= last_end - time;
587 stats[index].replaced += 1;
588 }
589 }
590 last_end = time + job.duration;
591 events[time - low] += 1;
592 events[last_end - low] -= 1;
593 stats[index].started += 1;
594 stats[index].busy += job.duration;
595 }
596 }
597 var running: i32 = 0;
598 var peak: i32 = 0;
599 var at: usize = 0;
600 for (events, 0..) |delta, offset| {
601 running += delta;
602 if (running > peak) {
603 peak = running;
604 at = low + offset;
605 }
606 }
607 try session.out.print("SIMULATE {d} peak {d}", .{ jobs.len, peak });
608 if (peak != 0) try session.out.print(" at {f}", .{datetime(at)});
609 try session.out.writeByte('\n');
610 for (jobs.values[0..jobs.len], stats[0..jobs.len]) |job, result| {
611 try line(session.out, "{f} {s} started {d} skipped {d} replaced {d} busy {d}", .{ job.id, @tagName(job.policy), result.started, result.skipped, result.replaced, result.busy });
612 }
613 }
614 fn execute(session: *Session, words: []const []const u8) Error!void {
615 const command = std.meta.stringToEnum(Command, words[0]) orelse return session.fail("unknown command", .{});
616 const arity: i8 = switch (command) {
617 .JOB => -2,
618 .VALIDATE => -1,
619 .LIST, .HOLIDAYS => 0,
620 .REMOVE, .ENABLE, .DISABLE, .DESCRIBE, .WEEK, .EXCLUDE, .INCLUDE => 1,
621 .RENAME, .EXPAND, .CALENDAR => 2,
622 .RUNTIME, .MATCH, .GAP => 3,
623 .NEXT, .PREV, .TIMELINE, .CONFLICTS, .SIMULATE => 4,
624 .COUNT => 5,
625 .OVERLAP => 6,
626 };
627 const count = words.len - 1;
628 if (if (arity < 0) count < @as(usize, @intCast(-arity)) else count != @as(usize, @intCast(arity))) return session.fail("wrong arguments", .{});
629 const out = session.out;
630 const state = session.state;
631 switch (command) {
632 .JOB => {
633 try session.newId(words[1]);
634 const cron = try session.parseCron(words[2..]);
635 const text = Text(100).init(try session.joinWords(words[2..]));
636 state.jobs[state.job_count] = .{ .id = Text(16).init(words[1]), .cron = cron, .text = text };
637 state.job_count += 1;
638 try line(out, "JOB {s} added", .{words[1]});
639 },
640 .VALIDATE => {
641 const cron = try session.parseCron(words[1..]);
642 try line(out, "VALID {f} {s} {d}", .{ cron.expanded, cron.rule(), cron.daily_count });
643 },
644 .REMOVE => {
645 const index = state.find(words[1]) orelse return session.fail("unknown job", .{});
646 for (index..state.job_count - 1) |position| state.jobs[position] = state.jobs[position + 1];
647 state.job_count -= 1;
648 try line(out, "REMOVE {s}", .{words[1]});
649 },
650 .RENAME => {
651 const job = try session.getJob(words[1]);
652 try session.newId(words[2]);
653 job.id = Text(16).init(words[2]);
654 try line(out, "RENAME {s} {s}", .{ words[1], words[2] });
655 },
656 .ENABLE, .DISABLE => {
657 const job = try session.getJob(words[1]);
658 const enabled = command == .ENABLE;
659 if (job.enabled == enabled) return session.fail("already {s}", .{if (enabled) "enabled" else "disabled"});
660 job.enabled = enabled;
661 try line(out, "{s} {f}", .{ @tagName(command), job.id });
662 },
663 .RUNTIME => {
664 const job = try session.getJob(words[1]);
665 const duration = try session.positive(words[2], 4, 1440, "invalid duration");
666 const policy = std.meta.stringToEnum(Policy, words[3]) orelse return session.fail("invalid policy", .{});
667 job.duration = duration;
668 job.policy = policy;
669 try line(out, "RUNTIME {f} {d} {s}", .{ job.id, duration, @tagName(policy) });
670 },
671 .LIST => {
672 const jobs = state.ordered(false);
673 try line(out, "LIST {d}", .{jobs.len});
674 for (jobs.values[0..jobs.len]) |job| try line(out, "{f} {s} {d} {s} {f}", .{ job.id, if (job.enabled) "on" else "off", job.duration, @tagName(job.policy), job.text });
675 },
676 .EXPAND => {
677 const job = try session.getJob(words[1]);
678 var selected: ?usize = null;
679 for (field_names, 0..) |name, index| if (equal(name, words[2])) {
680 selected = index;
681 break;
682 };
683 const index = selected orelse return session.fail("invalid field", .{});
684 const field = &job.cron.fields[index];
685 try out.print("EXPAND {f} {s} {d}", .{ job.id, words[2], field.count() });
686 for (field.values, 0..) |present, value| if (present) try out.print(" {d}", .{value});
687 for (field.specials[0..field.special_count]) |special| try out.print(" {f}", .{special});
688 try out.writeByte('\n');
689 },
690 .DESCRIBE => {
691 const job = try session.getJob(words[1]);
692 try out.print("DESCRIBE {f} ", .{job.id});
693 try session.describe(&job.cron);
694 },
695 .MATCH => {
696 const job = try session.getJob(words[1]);
697 const time = try session.readMinute(words[2], words[3]);
698 const date = Date.fromNumber(time / 1440);
699 const matched = job.cron.matchDay(date);
700 const reasons = [_]bool{ !job.cron.fields[0].values[time % 60], !job.cron.fields[1].values[time % 1440 / 60], !matched.condition and !matched.month_day, !job.cron.fields[3].values[date.month], !matched.condition and !matched.week_day, state.holidays[date.number()] };
701 const labels = field_names ++ [_][]const u8{"excluded"};
702 var yes = true;
703 for (reasons) |reason| if (reason) {
704 yes = false;
705 break;
706 };
707 try out.print("MATCH {f} {f} {s}", .{ job.id, datetime(time), if (yes) "yes" else "no" });
708 for (reasons, labels) |reason, label| if (reason) try out.print(" {s}", .{label});
709 try out.writeByte('\n');
710 },
711 .NEXT, .PREV => {
712 const job = try session.getJob(words[1]);
713 const start = try session.readMinute(words[2], words[3]);
714 const limit = try session.positive(words[4], 2, 50, "invalid count");
715 const reverse = command == .PREV;
716 var iterator = job.cron.iterator(&state.holidays, if (reverse) start - 527040 else start + 1, if (reverse) start - 1 else start + 527040, reverse);
717 var times: [50]usize = undefined;
718 var found: usize = 0;
719 while (found < limit) {
720 times[found] = iterator.next() orelse break;
721 found += 1;
722 }
723 try line(out, "{s} {f} {d}", .{ @tagName(command), job.id, found });
724 for (times[0..found]) |time| try line(out, "{f} {s}{f}", .{ datetime(time), if (reverse) "-" else "+", span(if (reverse) start - time else time - start) });
725 if (found < limit) try line(out, "NONE", .{});
726 },
727 .COUNT => {
728 const job = try session.getJob(words[1]);
729 const low = try session.readMinute(words[2], words[3]);
730 const high = try session.readMinute(words[4], words[5]);
731 try session.checkWindow(low, high);
732 var times = job.cron.iterator(&state.holidays, low, high, false);
733 var runs: usize = 0;
734 var days: usize = 0;
735 var previous: ?usize = null;
736 while (times.next()) |time| {
737 const day = time / 1440;
738 runs += 1;
739 if (previous == null or previous.? != day) days += 1;
740 previous = day;
741 }
742 try line(out, "COUNT {f} runs {d} days {d}", .{ job.id, runs, days });
743 },
744 .GAP => {
745 const job = try session.getJob(words[1]);
746 const start = try session.readMinute(words[2], words[3]);
747 var times = job.cron.iterator(&state.holidays, start, start + 44640, false);
748 const frequencies = try session.gpa.alloc(usize, 44641);
749 @memset(frequencies, 0);
750 var previous: ?usize = null;
751 var runs: usize = 0;
752 var shortest: usize = 44641;
753 var longest: usize = 0;
754 var from: usize = 0;
755 var to: usize = 0;
756 while (times.next()) |time| {
757 runs += 1;
758 if (previous) |old| {
759 const gap = time - old;
760 frequencies[gap] += 1;
761 shortest = @min(shortest, gap);
762 if (gap > longest) {
763 longest = gap;
764 from = old;
765 to = time;
766 }
767 }
768 previous = time;
769 }
770 if (runs < 2) return line(out, "GAP {f} runs {d} none", .{ job.id, runs });
771 try line(out, "GAP {f} runs {d} shortest {f} longest {f} from {f} to {f}", .{ job.id, runs, span(shortest), span(longest), datetime(from), datetime(to) });
772 for (frequencies, 0..) |frequency, gap| if (frequency != 0) try line(out, "{f} {d}", .{ span(gap), frequency });
773 },
774 .OVERLAP => {
775 const first = try session.getJob(words[1]);
776 const second = try session.getJob(words[2]);
777 const low = try session.readMinute(words[3], words[4]);
778 const high = try session.readMinute(words[5], words[6]);
779 try session.checkWindow(low, high);
780 var times = first.cron.iterator(&state.holidays, low, high, false);
781 var common: usize = 0;
782 var earliest: usize = 0;
783 var latest: usize = 0;
784 while (times.next()) |time| {
785 if (!second.cron.fires(time, &state.holidays)) continue;
786 if (common == 0) earliest = time;
787 latest = time;
788 common += 1;
789 }
790 try out.print("OVERLAP {f} {f} {d}", .{ first.id, second.id, common });
791 if (common != 0) try out.print(" first {f} last {f}", .{ datetime(earliest), datetime(latest) });
792 try out.writeByte('\n');
793 },
794 .TIMELINE, .CONFLICTS, .SIMULATE => {
795 const low = try session.readMinute(words[1], words[2]);
796 const high = try session.readMinute(words[3], words[4]);
797 try session.checkWindow(low, high);
798 if (command == .SIMULATE) try session.simulate(low, high) else try session.timeline(low, high, command == .CONFLICTS);
799 },
800 .WEEK => {
801 const date = try session.readDate(words[1]);
802 try session.week(words[1], date);
803 },
804 .CALENDAR => {
805 const job = try session.getJob(words[1]);
806 const first = try session.readMonth(words[2]);
807 try session.calendar(job, words[2], first);
808 },
809 .EXCLUDE, .INCLUDE => {
810 const date = try session.readDate(words[1]);
811 const day = date.number();
812 if (command == .EXCLUDE) {
813 if (state.holidays[day]) return session.fail("already excluded", .{});
814 state.holidays[day] = true;
815 state.holiday_count += 1;
816 } else {
817 if (!state.holidays[day]) return session.fail("not excluded", .{});
818 state.holidays[day] = false;
819 state.holiday_count -= 1;
820 }
821 try line(out, "{s} {f} holidays {d}", .{ @tagName(command), date, state.holiday_count });
822 },
823 .HOLIDAYS => {
824 try line(out, "HOLIDAYS {d}", .{state.holiday_count});
825 for (state.holidays, 0..) |present, day| if (present) try line(out, "{f}", .{Date.fromNumber(day)});
826 },
827 }
828 }
829};

831fn readLine(gpa: Allocator, input: *std.Io.Reader) ![]const u8 {
832 var buffer: Writer.Allocating = .init(gpa);
833 defer buffer.deinit();
834 _ = try input.streamDelimiterEnding(&buffer.writer, '\n');
835 if (input.seek < input.end) input.toss(1) else if (buffer.written().len == 0) return error.EndOfStream;
836 const text = try buffer.toOwnedSlice();
837 return if (text.len > 0 and text[text.len - 1] == '\r') text[0 .. text.len - 1] else text;
838}
839pub fn main(init: std.process.Init) !void {
840 var header = std.heap.ArenaAllocator.init(std.heap.page_allocator);
841 defer header.deinit();
842 var state: State = .{};
843 var input_buffer: [1024]u8 = undefined;
844 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
845 var output_buffer: [4096]u8 = undefined;
846 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
847 const count = integer(try readLine(header.allocator(), &input.interface));
848 for (0..count) |_| {
849 var scratch = std.heap.ArenaAllocator.init(std.heap.page_allocator);
850 defer scratch.deinit();
851 const gpa = scratch.allocator();
852 const text = try readLine(gpa, &input.interface);
853 var words: std.ArrayList([]const u8) = .empty;
854 var tokens = std.mem.tokenizeScalar(u8, text, ' ');
855 while (tokens.next()) |word| try words.append(gpa, word);
856 var response: Writer.Allocating = .init(gpa);
857 defer response.deinit();
858 var session: Session = .{ .state = &state, .gpa = gpa, .out = &response.writer };
859 session.execute(words.items) catch |err| {
860 if (err != error.Invalid) return err;
861 try line(&output.interface, "ERROR {s}", .{session.problem});
862 try output.interface.flush();
863 continue;
864 };
865 try output.interface.writeAll(response.written());
866 try output.interface.flush();
867 }
