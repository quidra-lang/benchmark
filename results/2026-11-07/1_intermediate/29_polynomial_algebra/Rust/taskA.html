7#[derive(Debug)]
8enum Error {
9 Syntax(Option<usize>), Unknown(String), DivisionByZero, NonconstantDivisor, DegreeLimit,
10 SameName, ZeroDenominator, ZeroPolynomial, BadInterval, DegreeTooLow, NotCoprime,
11 DuplicateX(Fraction), UnknownKind(String),
12}
13impl fmt::Display for Error {
14 fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
15 let text = match self {
16 Self::Syntax(Some(position)) => return write!(f, "ERROR SYNTAX AT {position}"),
17 Self::Syntax(None) => "SYNTAX AT END",
18 Self::Unknown(name) => return write!(f, "ERROR UNKNOWN {name}"),
19 Self::DuplicateX(value) => return write!(f, "ERROR DUPLICATE X {value}"),
20 Self::UnknownKind(kind) => return write!(f, "ERROR UNKNOWN KIND {kind}"),
21 Self::DivisionByZero => "DIVISION BY ZERO", Self::NonconstantDivisor => "NONCONSTANT DIVISOR",
22 Self::DegreeLimit => "DEGREE LIMIT", Self::SameName => "SAME NAME",
23 Self::ZeroDenominator => "ZERO DENOMINATOR", Self::ZeroPolynomial => "ZERO POLYNOMIAL",
24 Self::BadInterval => "BAD INTERVAL", Self::DegreeTooLow => "DEGREE TOO LOW",
25 Self::NotCoprime => "NOT COPRIME",
26 };
27 write!(f, "ERROR {text}")
28 }
29}
30type CalcResult<T> = Result<T, Error>;

32fn gcd(mut a: i64, mut b: i64) -> i64 {
33 a = a.abs();
34 b = b.abs();
35 while b != 0 { (a, b) = (b, a % b); }
36 a
37}

39#[derive(Clone, Copy, Debug, Eq, PartialEq)]
40struct Fraction { n: i64, d: i64 }
41impl Fraction {
42 const ZERO: Self = Self { n: 0, d: 1 };
43 const ONE: Self = Self { n: 1, d: 1 };
44 fn new(mut n: i64, mut d: i64) -> Self {
45 if d < 0 { n = -n; d = -d; }
46 let divisor = gcd(n, d);
47 Self { n: n / divisor, d: d / divisor }
48 }
49 fn integer(n: i64) -> Self { Self { n, d: 1 } }
50 fn parse(text: &str) -> CalcResult<Self> {
51 let (n, d) = match text.split_once('/') {
52 Some((n, d)) => (n.parse().unwrap(), d.parse().unwrap()),
53 None => (text.parse().unwrap(), 1),
54 };
55 if d == 0 { Err(Error::ZeroDenominator) } else { Ok(Self::new(n, d)) }
56 }
57 fn abs(self) -> Self { Self { n: self.n.abs(), d: self.d } }
58 fn pow(self, n: u32) -> Self {
59 let mut result = Self::ONE;
60 for _ in 0..n { result = result * self; }
61 result
62 }
63}
64impl Add for Fraction {
65 type Output = Self;
66 fn add(self, b: Self) -> Self { Self::new(self.n * b.d + b.n * self.d, self.d * b.d) }
67}
68impl Sub for Fraction {
69 type Output = Self;
70 fn sub(self, b: Self) -> Self { Self::new(self.n * b.d - b.n * self.d, self.d * b.d) }
71}
72impl Mul for Fraction {
73 type Output = Self;
74 fn mul(self, b: Self) -> Self { Self::new(self.n * b.n, self.d * b.d) }
75}
76impl Div for Fraction {
77 type Output = Self;
78 fn div(self, b: Self) -> Self { Self::new(self.n * b.d, self.d * b.n) }
79}
80impl Neg for Fraction {
81 type Output = Self;
82 fn neg(self) -> Self { Self { n: -self.n, d: self.d } }
83}
84impl Ord for Fraction {
85 fn cmp(&self, other: &Self) -> Ordering { (self.n * other.d).cmp(&(other.n * self.d)) }
86}
87impl PartialOrd for Fraction {
88 fn partial_cmp(&self, other: &Self) -> Option<Ordering> { Some(self.cmp(other)) }
89}
90impl fmt::Display for Fraction {
91 fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
92 if self.d == 1 { write!(f, "{}", self.n) } else { write!(f, "{}/{}", self.n, self.d) }
93 }
94}

96#[derive(Clone, Default)]
97struct Polynomial(Vec<Fraction>);
98impl Polynomial {
99 fn new(mut values: Vec<Fraction>) -> Self {
100 while values.last().is_some_and(|c| c.n == 0) { values.pop(); }
101 Self(values)
102 }
103 fn zero() -> Self { Self::default() }
104 fn constant(value: Fraction) -> Self { Self::new(vec![value]) }
105 fn one() -> Self { Self::constant(Fraction::ONE) }
106 fn linear(value: Fraction, slope: Fraction) -> Self { Self::new(vec![value, slope]) }
107 fn is_zero(&self) -> bool { self.0.is_empty() }
108 fn degree(&self) -> i32 { self.0.len() as i32 - 1 }
109 fn leading(&self) -> Fraction { *self.0.last().unwrap() }
110 fn coefficient(&self, i: usize) -> Fraction { self.0.get(i).copied().unwrap_or(Fraction::ZERO) }
111 fn add(&self, b: &Self) -> Self {
112 Self::new((0..self.0.len().max(b.0.len())).map(|i| self.coefficient(i) + b.coefficient(i)).collect())
113 }
114 fn sub(&self, b: &Self) -> Self {
115 Self::new((0..self.0.len().max(b.0.len())).map(|i| self.coefficient(i) - b.coefficient(i)).collect())
116 }
117 fn neg(&self) -> Self { Self(self.0.iter().map(|c| -*c).collect()) }
118 fn scale(&self, value: Fraction) -> Self { Self::new(self.0.iter().map(|c| value * *c).collect()) }
119 fn divide_scalar(&self, value: Fraction) -> Self { Self::new(self.0.iter().map(|c| *c / value).collect()) }
120 fn mul(&self, b: &Self) -> Self {
121 if self.is_zero() || b.is_zero() { return Self::zero(); }
122 let mut result = vec![Fraction::ZERO; self.0.len() + b.0.len() - 1];
123 for (i, &a) in self.0.iter().enumerate() {
124 for (j, &b) in b.0.iter().enumerate() { result[i + j] = result[i + j] + a * b; }
125 }
126 Self::new(result)
127 }
128 fn pow(&self, n: u32) -> Self {
129 if n != 0 && self.is_zero() { return Self::zero(); }
130 if self.0.len() == 1 && self.0[0].d == 1 && self.0[0].n.abs() == 1 {
131 return Self::constant(Fraction::integer(if self.0[0].n < 0 && n % 2 == 1 { -1 } else { 1 }));
132 }
133 let mut result = Self::one();
134 for _ in 0..n { result = result.mul(self); }
135 result
136 }
137 fn divmod(&self, b: &Self) -> (Self, Self) {
138 let count = if self.0.len() >= b.0.len() { self.0.len() - b.0.len() + 1 } else { 0 };
139 let mut q = vec![Fraction::ZERO; count];
140 let mut r = self.0.clone();
141 while !r.is_empty() && r.len() >= b.0.len() {
142 let factor = *r.last().unwrap() / b.leading();
143 let shift = r.len() - b.0.len();
144 q[shift] = factor;
145 for (j, &c) in b.0.iter().enumerate() { r[shift + j] = r[shift + j] - factor * c; }
146 while r.last().is_some_and(|c| c.n == 0) { r.pop(); }
147 }
148 (Self::new(q), Self(r))
149 }
150 fn monic(&self) -> Self { if self.is_zero() { Self::zero() } else { self.divide_scalar(self.leading()) } }
151 fn derivative(&self) -> Self {
152 Self(self.0.iter().enumerate().skip(1).map(|(i, &c)| Fraction::integer(i as i64) * c).collect())
153 }
154 fn differentiate(&self, n: u32) -> Self {
155 let mut result = self.clone();
156 for _ in 0..n {
157 if result.is_zero() { break; }
158 result = result.derivative();
159 }
160 result
161 }
162 fn integral(&self) -> Self {
163 let mut result = vec![Fraction::ZERO];
164 for (i, &c) in self.0.iter().enumerate() { result.push(c / Fraction::integer(i as i64 + 1)); }
165 Self::new(result)
166 }
167 fn evaluate(&self, point: Fraction) -> Fraction {
168 let mut result = Fraction::ZERO;
169 for &c in self.0.iter().rev() { result = result * point + c; }
170 result
171 }
172 fn compose(&self, inner: &Self) -> Self {
173 let mut result = Self::zero();
174 for &c in self.0.iter().rev() { result = result.mul(inner).add(&Self::constant(c)); }
175 result
176 }
177 fn gcd(&self, b: &Self) -> Self {
178 let (mut a, mut b) = (self.clone(), b.clone());
179 while !b.is_zero() {
180 let remainder = a.divmod(&b).1;
181 (a, b) = (b, remainder);
182 }
183 a.monic()
184 }
185 fn extended_gcd(&self, b: &Self) -> Bezout {
186 let (mut r0, mut r1) = (self.clone(), b.clone());
187 let (mut s0, mut s1) = (Self::one(), Self::zero());
188 let (mut t0, mut t1) = (Self::zero(), Self::one());
189 while !r1.is_zero() {
190 let (q, r) = r0.divmod(&r1);
191 let s = s0.sub(&q.mul(&s1));
192 let t = t0.sub(&q.mul(&t1));
193 (r0, r1) = (r1, r);
194 (s0, s1) = (s1, s);
195 (t0, t1) = (t1, t);
196 }
197 if r0.is_zero() { Bezout { g: r0, s: s0, t: t0 } }
198 else { Bezout { g: r0.monic(), s: s0.divide_scalar(r0.leading()), t: t0.divide_scalar(r0.leading()) } }
199 }
200 fn content(&self) -> Fraction {
201 let (mut lcm, mut divisor) = (1, 0);
202 for c in &self.0 {
203 lcm = lcm * c.d / gcd(lcm, c.d);
204 divisor = gcd(divisor, c.n);
205 }
206 Fraction::new(if self.leading().n < 0 { -divisor } else { divisor }, lcm)
207 }
208 fn primitive(&self) -> Self { self.divide_scalar(self.content()) }
209 fn square_free(&self) -> Self {
210 if self.degree() < 1 { self.clone() } else { self.divmod(&self.gcd(&self.derivative())).0 }
211 }
212 fn is_one(&self) -> bool { self.0.as_slice() == [Fraction::ONE] }
213 fn check_product(&self, b: &Self) -> CalcResult<()> {
214 if !self.is_zero() && !b.is_zero() && self.degree() + b.degree() > 30 { Err(Error::DegreeLimit) } else { Ok(()) }
215 }
216 fn check_power(&self, n: u32) -> CalcResult<()> {
217 if !self.is_zero() && self.degree() as i64 * n as i64 > 30 { Err(Error::DegreeLimit) } else { Ok(()) }
218 }
219 fn check_composition(&self, b: &Self) -> CalcResult<()> {
220 if self.degree() >= 1 && b.degree() >= 1 && self.degree() * b.degree() > 30 { Err(Error::DegreeLimit) } else { Ok(()) }
221 }
222 fn require_nonzero(&self) -> CalcResult<()> { if self.is_zero() { Err(Error::ZeroPolynomial) } else { Ok(()) } }
223 fn require_divisor(&self) -> CalcResult<()> { if self.is_zero() { Err(Error::DivisionByZero) } else { Ok(()) } }
224 fn require_degree(&self) -> CalcResult<()> { if self.degree() == 0 { Err(Error::DegreeTooLow) } else { Ok(()) } }
225 fn format(&self, variable: &str) -> String {
226 let mut result = String::new();
227 for (i, &c) in self.0.iter().enumerate().rev() {
228 if c.n == 0 { continue; }
229 if result.is_empty() { if c.n < 0 { result.push('-'); } }
230 else { result.push_str(if c.n < 0 { " - " } else { " + " }); }
231 if i == 0 || c.abs() != Fraction::ONE { result.push_str(&c.abs().to_string()); }
232 if i > 0 { result.push_str(variable); }
233 if i > 1 { result.push_str(&format!("^{i}")); }
234 }
235 if result.is_empty() { result.push('0'); }
236 result
237 }
238}
239struct Bezout { g: Polynomial, s: Polynomial, t: Polynomial }

241enum Expression {
242 Number(u32), X, Name(String, u32), Negative(Box<Expression>),
243 Binary(u8, Box<Expression>, Box<Expression>), Power(Box<Expression>, u32),
244 Compose(Box<Expression>, Box<Expression>),
245}
246impl Expression {
247 fn evaluate(&self, names: &BTreeMap<String, Polynomial>) -> CalcResult<Polynomial> {
248 match self {
249 Self::Number(n) => Ok(Polynomial::constant(Fraction::integer(*n as i64))),
250 Self::X => Ok(Polynomial::linear(Fraction::ZERO, Fraction::ONE)),
251 Self::Name(name, n) => names.get(name).map(|p| p.differentiate(*n)).ok_or_else(|| Error::Unknown(name.clone())),
252 Self::Negative(expression) => Ok(expression.evaluate(names)?.neg()),
253 Self::Power(expression, n) => {
254 let value = expression.evaluate(names)?;
255 value.check_power(*n)?;
256 Ok(value.pow(*n))
257 }
258 Self::Binary(op, left, right) => {
259 let a = left.evaluate(names)?;
260 let b = right.evaluate(names)?;
261 match op {
262 b'+' => Ok(a.add(&b)), b'-' => Ok(a.sub(&b)),
263 b'*' => { a.check_product(&b)?; Ok(a.mul(&b)) }
264 b'/' => {
265 b.require_divisor()?;
266 if b.degree() >= 1 { return Err(Error::NonconstantDivisor); }
267 Ok(a.divide_scalar(b.coefficient(0)))
268 }
269 _ => unreachable!(),
270 }
271 }
272 Self::Compose(function, argument) => {
273 let a = function.evaluate(names)?;
274 let b = argument.evaluate(names)?;
275 a.check_composition(&b)?;
276 Ok(a.compose(&b))
277 }
278 }
279 }
280}

282#[derive(Clone)]
283enum TokenKind { Number(u32), Name(String), Symbol(u8), End }
284struct Token { kind: TokenKind, position: usize }
285struct Parser { tokens: Vec<Token>, position: usize }
286impl Parser {
287 fn new(text: &str) -> Self {
288 let bytes = text.as_bytes();
289 let mut tokens = Vec::new();
290 let mut i = 0;
291 while i < bytes.len() {
292 if bytes[i] == b' ' { i += 1; continue; }
293 let start = i;
294 i += 1;
295 let kind = if bytes[start].is_ascii_digit() {
296 while i < bytes.len() && bytes[i].is_ascii_digit() { i += 1; }
297 TokenKind::Number(text[start..i].parse().unwrap())
298 } else if bytes[start].is_ascii_uppercase() {
299 while i < bytes.len() && (bytes[i].is_ascii_uppercase() || bytes[i].is_ascii_digit()) { i += 1; }
300 TokenKind::Name(text[start..i].to_owned())
301 } else { TokenKind::Symbol(bytes[start]) };
302 tokens.push(Token { kind, position: start + 1 });
303 }
304 tokens.push(Token { kind: TokenKind::End, position: text.len() + 1 });
305 Self { tokens, position: 0 }
306 }
307 fn current(&self) -> &TokenKind { &self.tokens[self.position].kind }
308 fn syntax(&self) -> Error {
309 Error::Syntax(if matches!(self.current(), TokenKind::End) { None } else { Some(self.tokens[self.position].position) })
310 }
311 fn accept(&mut self, symbol: u8) -> bool {
312 if matches!(self.current(), TokenKind::Symbol(c) if *c == symbol) { self.position += 1; true } else { false }
313 }
314 fn expect(&mut self, symbol: u8) -> CalcResult<()> { if self.accept(symbol) { Ok(()) } else { Err(self.syntax()) } }
315 fn parse(mut self) -> CalcResult<Expression> {
316 let expression = self.expression()?;
317 if !matches!(self.current(), TokenKind::End) { return Err(self.syntax()); }
318 Ok(expression)
319 }
320 fn expression(&mut self) -> CalcResult<Expression> {
321 let negative = self.accept(b'-');
322 let mut result = self.term()?;
323 if negative { result = Expression::Negative(Box::new(result)); }
324 loop {
325 let op = if self.accept(b'+') { b'+' } else if self.accept(b'-') { b'-' } else { break; };
326 result = Expression::Binary(op, Box::new(result), Box::new(self.term()?));
327 }
328 Ok(result)
329 }
330 fn term(&mut self) -> CalcResult<Expression> {
331 let mut result = self.factor()?;
332 loop {
333 let op = if self.accept(b'*') { b'*' }
334 else if self.accept(b'/') { b'/' }
335 else if matches!(self.current(), TokenKind::Name(_) | TokenKind::Symbol(b'x' | b'(')) { b'*' }
336 else { break; };
337 result = Expression::Binary(op, Box::new(result), Box::new(self.factor()?));
338 }
339 Ok(result)
340 }
341 fn factor(&mut self) -> CalcResult<Expression> {
342 let mut result = self.primary()?;
343 if self.accept(b'^') {
344 let TokenKind::Number(n) = self.current().clone() else { return Err(self.syntax()); };
345 self.position += 1;
346 result = Expression::Power(Box::new(result), n);
347 }
348 Ok(result)
349 }
350 fn primary(&mut self) -> CalcResult<Expression> {
351 match self.current().clone() {
352 TokenKind::Number(n) => { self.position += 1; Ok(Expression::Number(n)) }
353 TokenKind::Symbol(b'x') => { self.position += 1; Ok(Expression::X) }
354 TokenKind::Name(name) => {
355 self.position += 1;
356 let mut order = 0;
357 while self.accept(b'\'') { order += 1; }
358 let reference = Expression::Name(name, order);
359 if !self.accept(b'(') { return Ok(reference); }
360 let argument = self.expression()?;
361 self.expect(b')')?;
362 Ok(Expression::Compose(Box::new(reference), Box::new(argument)))
363 }
364 TokenKind::Symbol(b'(') => {
365 self.position += 1;
366 let expression = self.expression()?;
367 self.expect(b')')?;
368 Ok(expression)
369 }
370 _ => Err(self.syntax()),
371 }
372 }
373}

375struct RootResult { content: Fraction, remaining: Polynomial, roots: Vec<(Fraction, u32)> }
376fn rational_roots(a: &Polynomial) -> RootResult {
377 let content = a.content();
378 let primitive = a.divide_scalar(content);
379 let zeros = primitive.0.iter().take_while(|c| c.n == 0).count();
380 let mut p = Polynomial(primitive.0[zeros..].to_vec());
381 let first = p.0[0].n.abs();
382 let last = p.leading().n;
383 let numerators: Vec<_> = (1..=first).filter(|s| first % s == 0).collect();
384 let denominators: Vec<_> = (1..=last).filter(|t| last % t == 0).collect();
385 let mut candidates = Vec::new();
386 for s in numerators {
387 for &t in &denominators {
388 if gcd(s, t) == 1 { candidates.extend([Fraction::new(-s, t), Fraction::new(s, t)]); }
389 }
390 }
391 candidates.sort();
392 let mut roots = Vec::new();
393 for r in candidates {
394 let mut multiplicity = 0;
395 while p.degree() >= 1 && p.evaluate(r) == Fraction::ZERO {
396 p = p.divmod(&Polynomial::linear(Fraction::integer(-r.n), Fraction::integer(r.d))).0;
397 multiplicity += 1;
398 }
399 if multiplicity != 0 { roots.push((r, multiplicity)); }
400 }
401 if zeros != 0 { roots.push((Fraction::ZERO, zeros as u32)); }
402 roots.sort_by_key(|r| r.0);
403 RootResult { content, remaining: p, roots }
404}
405fn append_factor(text: &mut String, p: &Polynomial, multiplicity: u32, simple_x: bool) {
406 text.push_str(" * ");
407 if simple_x { text.push('x'); } else { text.push_str(&format!("({})", p.format("x"))); }
408 if multiplicity > 1 { text.push_str(&format!("^{multiplicity}")); }
409}
410fn square_free_factors(a: &Polynomial) -> Vec<(Polynomial, u32)> {
411 if a.degree() < 1 { return Vec::new(); }
412 let w0 = a.monic();
413 let mut c = w0.gcd(&w0.derivative());
414 let mut w = w0.divmod(&c).0;
415 let mut result = Vec::new();
416 let mut multiplicity = 1;
417 while w.degree() >= 1 {
418 let y = w.gcd(&c);
419 let f = w.divmod(&y).0;
420 if f.degree() >= 1 { result.push((f.primitive(), multiplicity)); }
421 c = c.divmod(&y).0;
422 w = y;
423 multiplicity += 1;
424 }
425 result
426}

428#[derive(Clone, Copy, Eq, PartialEq, Ord, PartialOrd)]
429enum Bound { NegativeInfinity, Finite(Fraction), PositiveInfinity }
430impl Bound {
431 fn parse(text: &str) -> CalcResult<Self> {
432 match text {
433 "-inf" => Ok(Self::NegativeInfinity), "inf" => Ok(Self::PositiveInfinity),
434 _ => Ok(Self::Finite(Fraction::parse(text)?)),
435 }
436 }
437}
438impl fmt::Display for Bound {
439 fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
440 match self {
441 Self::NegativeInfinity => write!(f, "-inf"), Self::PositiveInfinity => write!(f, "inf"),
442 Self::Finite(value) => write!(f, "{value}"),
443 }
444 }
445}
446fn sturm_sequence(a: &Polynomial) -> Vec<Polynomial> {
447 let mut p = a.square_free();
448 let mut q = p.derivative();
449 let mut result = vec![p.clone()];
450 while !q.is_zero() {
451 result.push(q.clone());
452 let remainder = p.divmod(&q).1.neg();
453 (p, q) = (q, remainder);
454 }
455 result
456}
457fn variations(sequence: &[Polynomial], point: Bound) -> i32 {
458 let (mut previous, mut count) = (0, 0);
459 for p in sequence {
460 let mut sign = match point {
461 Bound::Finite(value) => p.evaluate(value).n.signum(),
462 _ => p.leading().n.signum(),
463 };
464 if point == Bound::NegativeInfinity && p.degree() % 2 == 1 { sign = -sign; }
465 if sign != 0 { if previous != 0 && previous != sign { count += 1; } previous = sign; }
466 }
467 count
468}
469fn root_count(sequence: &[Polynomial], low: Bound, high: Bound) -> i32 {
470 let left = variations(sequence, low);
471 let right = variations(sequence, high);
472 left - right
473}
474fn examine(sequence: &[Polynomial], mut low: Fraction, mut high: Fraction, width: Fraction,
475 result: &mut Vec<(Fraction, Fraction)>) {
476 let count = root_count(sequence, Bound::Finite(low), Bound::Finite(high));
477 if count >= 2 {
478 let middle = (low + high) / Fraction::integer(2);
479 examine(sequence, low, middle, width, result);
480 examine(sequence, middle, high, width, result);
481 } else if count == 1 {
482 while high - low > width {
483 let middle = (low + high) / Fraction::integer(2);
484 if root_count(sequence, Bound::Finite(low), Bound::Finite(middle)) >= 1 { high = middle; }
485 else { low = middle; }
486 }
487 result.push((low, high));
488 }
489}
490fn isolate(a: &Polynomial, n: u32) -> Vec<(Fraction, Fraction)> {
491 let sequence = sturm_sequence(a);
492 let first = &sequence[0];
493 let mut maximum = Fraction::ZERO;
494 for &c in &first.0[..first.0.len() - 1] { maximum = maximum.max((c / first.leading()).abs()); }
495 let beta = Fraction::ONE + maximum;
496 let width = Fraction::new(1, 1_i64 << n);
497 let mut result = Vec::new();
498 examine(&sequence, -beta, beta, width, &mut result);
499 result
500}
501fn resultant(a: &Polynomial, b: &Polynomial) -> Fraction {
502 let (mut p, mut q) = (a.clone(), b.clone());
503 let mut r = Fraction::ONE;
504 while q.degree() >= 1 {
505 let t = p.divmod(&q).1;
506 if t.is_zero() { return Fraction::ZERO; }
507 r = r * q.leading().pow((p.degree() - t.degree()) as u32);
508 if p.degree() * q.degree() % 2 == 1 { r = -r; }
509 (p, q) = (q, t);
510 }
511 r * q.0[0].pow(p.degree() as u32)
512}
513fn interpolate(values: &[Fraction]) -> CalcResult<Polynomial> {
514 let x: Vec<_> = values.iter().step_by(2).copied().collect();
515 let mut d: Vec<_> = values.iter().skip(1).step_by(2).copied().collect();
516 for j in 1..x.len() {
517 for i in 0..j { if x[j] == x[i] { return Err(Error::DuplicateX(x[j])); } }
518 }
519 for j in 1..x.len() {
520 for i in (j..x.len()).rev() { d[i] = (d[i] - d[i - 1]) / (x[i] - x[i - j]); }
521 }
522 let mut result = Polynomial::zero();
523 for i in (0..x.len()).rev() {
524 result = result.mul(&Polynomial::linear(-x[i], Fraction::ONE)).add(&Polynomial::constant(d[i]));
525 }
526 Ok(result)
527}
528fn special(kind: &str, n: u32) -> CalcResult<Polynomial> {
529 if !matches!(kind, "chebyshev" | "chebyshev2" | "legendre" | "hermite" | "laguerre") {
530 return Err(Error::UnknownKind(kind.to_owned()));
531 }
532 if n > 30 { return Err(Error::DegreeLimit); }
533 let mut a = Polynomial::one();
534 let mut b = if kind == "laguerre" { Polynomial::linear(Fraction::ONE, Fraction::integer(-1)) }
535 else { Polynomial::linear(Fraction::ZERO, Fraction::integer(if matches!(kind, "chebyshev2" | "hermite") { 2 } else { 1 })) };
536 if n == 0 { return Ok(a); }
537 for j in 1..n as i64 {
538 let factor = if kind == "laguerre" { Polynomial::linear(Fraction::integer(2 * j + 1), Fraction::integer(-1)) }
539 else { Polynomial::linear(Fraction::ZERO, Fraction::integer(if kind == "legendre" { 2 * j + 1 } else { 2 })) };
540 let multiplier = Fraction::integer(match kind { "legendre" | "laguerre" => j, "hermite" => 2 * j, _ => 1 });
541 let mut next = factor.mul(&b).sub(&a.scale(multiplier));
542 if matches!(kind, "legendre" | "laguerre") { next = next.scale(Fraction::new(1, j + 1)); }
543 (a, b) = (b, next);
544 }
545 Ok(b)
546}

548#[derive(Default)]
549struct Response { stores: Vec<(String, Polynomial)>, output: Vec<String>, deletion: Option<String> }
550impl Response {
551 fn save(&mut self, name: &str, value: Polynomial) {
552 self.output.push(format!("{name} = {}", value.format("x")));
553 self.stores.push((name.to_owned(), value));
554 }
555}
556#[derive(Default)]
557struct Calculator { names: BTreeMap<String, Polynomial> }
558impl Calculator {
559 fn process(&mut self, line: &str) -> CalcResult<Vec<String>> {
560 let mut response = Response::default();
561 if line.starts_with("DEF ") {
562 let mut parts = line.splitn(3, ' ');
563 parts.next();
564 let name = parts.next().unwrap();
565 let expression = Parser::new(parts.next().unwrap()).parse()?;
566 response.save(name, expression.evaluate(&self.names)?);
567 } else {
568 let words: Vec<_> = line.split_ascii_whitespace().collect();
569 let command = words[0];
570 if (command == "DIV" && words[1] == words[2]) || (command == "XGCD" &&
571 (words[1] == words[2] || words[1] == words[3] || words[2] == words[3])) { return Err(Error::SameName); }
572 let positions: &[usize] = match command {
573 "ADD" | "SUB" | "MUL" | "GCD" | "LCM" | "INVMOD" | "COMPOSE" => &[2, 3],
574 "POW" | "DERIV" | "INTEG" | "TANGENT" => &[2], "MODPOW" => &[2, 4],
575 "DIV" => &[3, 4], "XGCD" => &[4, 5], "CRT" => &[2, 3, 4, 5], "RESULTANT" => &[1, 2],
576 "LIST" | "INTERP" | "SPECIAL" => &[], _ => &[1],
577 };
578 let p: Vec<_> = positions.iter().map(|&i| self.names.get(words[i]).cloned()
579 .ok_or_else(|| Error::Unknown(words[i].to_owned()))).collect::<CalcResult<_>>()?;
580 let start = match command { "AREA" | "TAYLOR" | "EVAL" | "INTERP" => 2, "TANGENT" => 3, _ => words.len() };
581 let values = words[start..].iter().map(|text| Fraction::parse(text)).collect::<CalcResult<Vec<_>>>()?;
582 let bounds = if command == "STURM" { Some((Bound::parse(words[2])?, Bound::parse(words[3])?)) } else { None };
583 let zero = Polynomial::zero();
584 let a = p.first().unwrap_or(&zero);
585 let b = p.get(1).unwrap_or(&zero);
586 match command {
587 "PRINT" => response.output.push(format!("{} = {}", words[1], a.format("x"))),
588 "DELETE" => { response.deletion = Some(words[1].to_owned()); response.output.push(format!("deleted {}", words[1])); }
589 "LIST" => {
590 for (name, value) in &self.names { response.output.push(format!("{name} = {}", value.format("x"))); }
591 if self.names.is_empty() { response.output.push("no polynomials".to_owned()); }
592 }
593 "DEGREE" => response.output.push(format!("deg({}) = {}", words[1], if a.is_zero() { "-inf".to_owned() } else { a.degree().to_string() })),
594 "ADD" => response.save(words[1], a.add(b)), "SUB" => response.save(words[1], a.sub(b)),
595 "MUL" => { a.check_product(b)?; response.save(words[1], a.mul(b)); }
596 "POW" => { let n = words[3].parse().unwrap(); a.check_power(n)?; response.save(words[1], a.pow(n)); }
597 "DIV" => {
598 b.require_divisor()?;
599 let (quotient, remainder) = a.divmod(b);
600 response.save(words[1], quotient); response.save(words[2], remainder);
601 }
602 "GCD" => response.save(words[1], a.gcd(b)),
603 "XGCD" => {
604 let e = a.extended_gcd(b);
605 response.save(words[1], e.g); response.save(words[2], e.s); response.save(words[3], e.t);
606 }
607 "LCM" => {
608 let value = if a.is_zero() || b.is_zero() { Polynomial::zero() } else {
609 let g = a.gcd(b);
610 if a.degree() + b.degree() - g.degree() > 30 { return Err(Error::DegreeLimit); }
611 a.divmod(&g).0.mul(b).monic()
612 };
613 response.save(words[1], value);
614 }
615 "INVMOD" => {
616 b.require_divisor()?; b.require_degree()?;
617 let e = a.extended_gcd(b);
618 if !e.g.is_one() { return Err(Error::NotCoprime); }
619 response.save(words[1], e.s.divmod(b).1);
620 }
621 "CRT" => {
622 let (m1, a2, m2) = (&p[1], &p[2], &p[3]);
623 m1.require_divisor()?; m2.require_divisor()?;
624 if m1.degree() + m2.degree() > 30 { return Err(Error::DegreeLimit); }
625 let e = m1.extended_gcd(m2);
626 if !e.g.is_one() { return Err(Error::NotCoprime); }
627 let r1 = a.divmod(m1).1;
628 let r2 = a2.divmod(m2).1;
629 let u = r2.sub(&r1).mul(&e.s).divmod(m2).1;
630 response.save(words[1], r1.add(&m1.mul(&u)));
631 }
632 "MODPOW" => {
633 b.require_divisor()?;
634 if 2 * b.degree() - 2 > 30 { return Err(Error::DegreeLimit); }
635 let mut x = Polynomial::one().divmod(b).1;
636 let mut y = a.divmod(b).1;
637 let mut exponent: u32 = words[3].parse().unwrap();
638 while exponent != 0 {
639 if exponent % 2 == 1 { x = x.mul(&y).divmod(b).1; }
640 exponent /= 2;
641 if exponent != 0 { y = y.mul(&y).divmod(b).1; }
642 }
643 response.save(words[1], x);
644 }
645 "DERIV" => response.save(words[1], a.differentiate(words[3].parse().unwrap())),
646 "INTEG" => { if !a.is_zero() && a.degree() + 1 > 30 { return Err(Error::DegreeLimit); } response.save(words[1], a.integral()); }
647 "AREA" => {
648 let integral = a.integral();
649 let high = integral.evaluate(values[1]);
650 let low = integral.evaluate(values[0]);
651 response.output.push(format!("integral({}, {}, {}) = {}", words[1], values[0], values[1], high - low));
652 }
653 "TANGENT" => {
654 let y = a.evaluate(values[0]);
655 let slope = a.derivative().evaluate(values[0]);
656 response.save(words[1], Polynomial::linear(y - slope * values[0], slope));
657 }
658 "TAYLOR" => {
659 let center = values[0];
660 let shifted = a.compose(&Polynomial::linear(center, Fraction::ONE));
661 let variable = if center.n == 0 { "x".to_owned() }
662 else { format!("(x {} {})", if center.n > 0 { "-" } else { "+" }, center.abs()) };
663 response.output.push(format!("taylor({}, {center}) = {}", words[1], shifted.format(&variable)));
664 }
665 "COMPOSE" => { a.check_composition(b)?; response.save(words[1], a.compose(b)); }
666 "EVAL" => for value in values { response.output.push(format!("{}({value}) = {}", words[1], a.evaluate(value))); },
667 "ROOTS" | "FACTOR" => {
668 a.require_nonzero()?;
669 let roots = rational_roots(a);
670 if command == "ROOTS" {
671 let entries: Vec<_> = roots.roots.iter().map(|(value, m)| format!("{value} ({m})")).collect();
672 response.output.push(format!("roots({}) = {}", words[1], if entries.is_empty() { "none".to_owned() } else { entries.join(", ") }));
673 } else {
674 let mut text = format!("factor({}) = {}", words[1], roots.content);
675 for (value, m) in roots.roots {
676 append_factor(&mut text, &Polynomial::linear(Fraction::integer(-value.n), Fraction::integer(value.d)), m, value.n == 0);
677 }
678 if roots.remaining.degree() >= 1 { append_factor(&mut text, &roots.remaining, 1, false); }
679 response.output.push(text);
680 }
681 }
682 "SQFREE" => {
683 a.require_nonzero()?;
684 let mut text = format!("sqfree({}) = {}", words[1], a.content());
685 for (h, m) in square_free_factors(a) {
686 let simple_x = h.degree() == 1 && h.0[0] == Fraction::ZERO && h.0[1] == Fraction::ONE;
687 append_factor(&mut text, &h, m, simple_x);
688 }
689 response.output.push(text);
690 }
691 "DESCARTES" => {
692 a.require_nonzero()?;
693 let mut counts = [0; 2];
694 for (negative, count) in counts.iter_mut().enumerate() {
695 let mut previous = 0;
696 for (i, c) in a.0.iter().enumerate().rev() {
697 let mut sign = c.n.signum();
698 if negative == 1 && i % 2 == 1 { sign = -sign; }
699 if sign != 0 { if previous != 0 && previous != sign { *count += 1; } previous = sign; }
700 }
701 }
702 response.output.push(format!("descartes({}) = positive {}, negative {}", words[1], counts[0], counts[1]));
703 }
704 "STURM" => {
705 a.require_nonzero()?;
706 let (low, high) = bounds.unwrap();
707 if low >= high { return Err(Error::BadInterval); }
708 let sequence = sturm_sequence(a);
709 response.output.push(format!("sturm({}, {low}, {high}) = {}", words[1], root_count(&sequence, low, high)));
710 }
711 "ISOLATE" => {
712 a.require_nonzero()?;
713 let n = words[2].parse().unwrap();
714 let entries: Vec<_> = isolate(a, n).into_iter().map(|(low, high)| format!("({low}, {high}]")).collect();
715 response.output.push(format!("isolate({}, {n}) = {}", words[1], if entries.is_empty() { "none".to_owned() } else { entries.join(", ") }));
716 }
717 "RESULTANT" => {
718 a.require_nonzero()?; b.require_nonzero()?;
719 response.output.push(format!("res({}, {}) = {}", words[1], words[2], resultant(a, b)));
720 }
721 "DISCRIM" | "POWERSUM" => {
722 a.require_nonzero()?; a.require_degree()?;
723 let degree = a.degree() as usize;
724 if command == "DISCRIM" {
725 let mut d = resultant(a, &a.derivative()) / a.leading();
726 if degree * (degree - 1) / 2 % 2 == 1 { d = -d; }
727 response.output.push(format!("disc({}) = {d}", words[1]));
728 } else {
729 let n = words[2].parse().unwrap();
730 let c = a.monic();
731 let mut sums = vec![Fraction::ZERO; n + 1];
732 for j in 1..=n {
733 let mut w = if j <= degree { Fraction::integer(j as i64) * c.0[degree - j] } else { Fraction::ZERO };
734 for i in 1..=degree.min(j - 1) { w = w + c.0[degree - i] * sums[j - i]; }
735 sums[j] = -w;
736 }
737 let text: Vec<_> = sums[1..].iter().map(ToString::to_string).collect();
738 response.output.push(format!("powersum({}) = {}", words[1], text.join(", ")));
739 }
740 }
741 "INTERP" => response.save(words[1], interpolate(&values)?),
742 "SPECIAL" => response.save(words[1], special(words[2], words[3].parse().unwrap())?),
743 _ => unreachable!(),
744 }
745 }
746 if let Some(name) = response.deletion { self.names.remove(&name); }
747 for (name, value) in response.stores { self.names.insert(name, value); }
748 Ok(response.output)
749 }
750}
751
752fn main() {
753 let stdin = io::stdin();
754 let mut lines = stdin.lock().lines();
755 let Some(Ok(first)) = lines.next() else { return; };
756 let count: usize = first.trim().parse().unwrap();
757 let mut calculator = Calculator::default();
758 let mut output = io::BufWriter::new(io::stdout().lock());
759 for line in lines.take(count) {
760 match calculator.process(&line.unwrap()) {
761 Ok(response) => for text in response { writeln!(output, "{text}").unwrap(); },
762 Err(error) => writeln!(output, "{error}").unwrap(),
763 }
764 }
765}
