20struct Fraction {
21 std::int64_t n, d;
22 Fraction(std::int64_t numerator = 0, std::int64_t denominator = 1) {
23 if (denominator < 0) { numerator = -numerator; denominator = -denominator; }
24 const auto divisor = std::gcd(numerator, denominator);
25 n = numerator / divisor;
26 d = denominator / divisor;
27 }
28 static Fraction parse(const std::string& word) {
29 const auto slash = word.find('/');
30 const auto denominator = slash == std::string::npos ? 1 : std::stoll(word.substr(slash + 1));
31 if (denominator == 0) fail("ZERO DENOMINATOR");
32 return {std::stoll(word.substr(0, slash)), denominator};
33 }
34 Fraction operator+(const Fraction& b) const { return {n * b.d + b.n * d, d * b.d}; }
35 Fraction operator-(const Fraction& b) const { return {n * b.d - b.n * d, d * b.d}; }
36 Fraction operator*(const Fraction& b) const { return {n * b.n, d * b.d}; }
37 Fraction operator/(const Fraction& b) const { return {n * b.d, d * b.n}; }
38 Fraction operator-() const { return {-n, d}; }
39 Fraction absolute() const { return n < 0 ? -*this : *this; }
40 int compare(const Fraction& b) const {
41 const auto left = n * b.d, right = b.n * d;
42 return (left > right) - (left < right);
43 }
44 bool operator<(const Fraction& b) const { return compare(b) < 0; }
45 bool operator==(const Fraction& b) const { return compare(b) == 0; }
46 Fraction power(int exponent) const {
47 Fraction result(1);
48 for (int i = 0; i < exponent; ++i) result = result * *this;
49 return result;
50 }
51 std::string notation() const {
52 return std::to_string(n) + (d == 1 ? "" : "/" + std::to_string(d));
53 }
54};

56struct Polynomial {
57 std::vector<Fraction> coefficients;
58 Polynomial(std::vector<Fraction> values = {}) : coefficients(std::move(values)) {
59 while (!coefficients.empty() && coefficients.back().n == 0) coefficients.pop_back();
60 }
61 static Polynomial constant(Fraction value) { return Polynomial({value}); }
62 int degree() const { return static_cast<int>(coefficients.size()) - 1; }
63 bool zero() const { return coefficients.empty(); }
64 Fraction leading() const { return coefficients.back(); }
65 Fraction coefficient(int index) const { return index < 0 || index > degree() ? Fraction() : coefficients[index]; }
66 Polynomial operator+(const Polynomial& b) const {
67 std::vector<Fraction> result(std::max(degree(), b.degree()) + 1);
68 for (int i = 0; i < static_cast<int>(result.size()); ++i) result[i] = coefficient(i) + b.coefficient(i);
69 return Polynomial(std::move(result));
70 }
71 Polynomial operator-(const Polynomial& b) const {
72 std::vector<Fraction> result(std::max(degree(), b.degree()) + 1);
73 for (int i = 0; i < static_cast<int>(result.size()); ++i) result[i] = coefficient(i) - b.coefficient(i);
74 return Polynomial(std::move(result));
75 }
76 Polynomial operator-() const {
77 auto result = coefficients;
78 for (auto& value : result) value = -value;
79 return Polynomial(std::move(result));
80 }
81 Polynomial scale(Fraction scalar) const {
82 auto result = coefficients;
83 for (auto& value : result) value = value * scalar;
84 return Polynomial(std::move(result));
85 }
86 Polynomial divideScalar(Fraction scalar) const {
87 auto result = coefficients;
88 for (auto& value : result) value = value / scalar;
89 return Polynomial(std::move(result));
90 }
91 Polynomial operator*(const Polynomial& b) const {
92 if (zero() || b.zero()) return {};
93 std::vector<Fraction> result(degree() + b.degree() + 1);
94 for (int i = 0; i <= degree(); ++i) {
95 for (int j = 0; j <= b.degree(); ++j) result[i + j] = result[i + j] + coefficients[i] * b.coefficients[j];
96 }
97 return Polynomial(std::move(result));
98 }
99 Polynomial power(int exponent) const {
100 if (exponent == 0) return constant(1);
101 if (zero()) return *this;
102 if (degree() == 0 && leading().d == 1 && std::abs(leading().n) == 1) {
103 return constant(leading().n < 0 && exponent % 2 ? -1 : 1);
104 }
105 Polynomial result = constant(1);
106 for (int i = 0; i < exponent; ++i) result = result * *this;
107 return result;
108 }
109 std::pair<Polynomial, Polynomial> divide(const Polynomial& b) const {
110 auto remainder = coefficients;
111 std::vector<Fraction> quotient(std::max(0, degree() - b.degree() + 1));
112 while (!remainder.empty() && remainder.size() >= b.coefficients.size()) {
113 const Fraction value = remainder.back() / b.leading();
114 const int shift = static_cast<int>(remainder.size() - b.coefficients.size());
115 quotient[shift] = value;
116 for (int j = 0; j <= b.degree(); ++j) remainder[shift + j] = remainder[shift + j] - value * b.coefficients[j];
117 while (!remainder.empty() && remainder.back().n == 0) remainder.pop_back();
118 }
119 return {Polynomial(std::move(quotient)), Polynomial(std::move(remainder))};
120 }
121 Polynomial mod(const Polynomial& b) const { return divide(b).second; }
122 Polynomial monic() const { return zero() ? *this : divideScalar(leading()); }
123 Polynomial derivative(int count = 1) const {
124 Polynomial result = *this;
125 for (int repetition = 0; repetition < count && !result.zero(); ++repetition) {
126 std::vector<Fraction> next(result.degree());
127 for (int i = 1; i <= result.degree(); ++i) next[i - 1] = Fraction(i) * result.coefficients[i];
128 result = Polynomial(std::move(next));
129 }
130 return result;
131 }
132 Polynomial integral() const {
133 std::vector<Fraction> result(coefficients.size() + 1);
134 for (int i = 0; i <= degree(); ++i) result[i + 1] = coefficients[i] / Fraction(i + 1);
135 return Polynomial(std::move(result));
136 }
137 Fraction evaluate(Fraction value) const {
138 Fraction result;
139 for (int i = degree(); i >= 0; --i) result = result * value + coefficients[i];
140 return result;
141 }
142 Polynomial compose(const Polynomial& b) const {
143 Polynomial result;
144 for (int i = degree(); i >= 0; --i) result = result * b + constant(coefficients[i]);
145 return result;
146 }
147 Fraction content() const {
148 std::int64_t multiple = 1, divisor = 0;
149 for (const auto& value : coefficients) {
150 multiple = multiple * value.d / std::gcd(multiple, value.d);
151 divisor = std::gcd(divisor, value.n);
152 }
153 return {leading().n < 0 ? -divisor : divisor, multiple};
154 }
155 Polynomial primitive() const { return divideScalar(content()); }
156 std::string notation(const std::string& variable = "x") const {
157 std::string result;
158 for (int i = degree(); i >= 0; --i) {
159 const Fraction value = coefficients[i];
160 if (value.n == 0) continue;
161 result += result.empty() ? (value.n < 0 ? "-" : "") : (value.n < 0 ? " - " : " + ");
162 if (i == 0 || !(value.absolute() == Fraction(1))) result += value.absolute().notation();
163 if (i > 0) result += variable + (i == 1 ? "" : "^" + std::to_string(i));
164 }
165 return result.empty() ? "0" : result;
166 }
167};

183 while (!b.zero()) {
184 Polynomial remainder = a.mod(b);
185 a = std::move(b);
186 b = std::move(remainder);
187 }
188 return a.monic();
189}

191std::tuple<Polynomial, Polynomial, Polynomial> extendedGcd(Polynomial a, Polynomial b) {
192 Polynomial s0 = unit, s1, t0, t1 = unit;
193 while (!b.zero()) {
194 auto [quotient, remainder] = a.divide(b);
195 a = std::move(b);
196 b = std::move(remainder);
197 Polynomial nextS = s0 - quotient * s1;
198 s0 = std::move(s1);
199 s1 = std::move(nextS);
200 Polynomial nextT = t0 - quotient * t1;
201 t0 = std::move(t1);
202 t1 = std::move(nextT);
203 }
204 if (a.zero()) return {a, s0, t0};
205 return {a.monic(), s0.divideScalar(a.leading()), t0.divideScalar(a.leading())};
206}

208std::vector<std::int64_t> divisors(std::int64_t number) {
209 std::vector<std::int64_t> result;
210 for (std::int64_t divisor = 1; divisor * divisor <= number; ++divisor) {
211 if (number % divisor != 0) continue;
212 result.push_back(divisor);
213 if (divisor * divisor != number) result.push_back(number / divisor);
214 }
215 return result;
216}

221RootFactorization rationalRoots(const Polynomial& a) {
222 Polynomial remaining = a.primitive();
223 int zeros = 0;
224 while (remaining.coefficient(0).n == 0) {
225 remaining = Polynomial(std::vector<Fraction>(remaining.coefficients.begin() + 1, remaining.coefficients.end()));
226 ++zeros;
227 }
228 std::vector<Fraction> candidates;
229 const auto numerators = divisors(std::abs(remaining.coefficient(0).n));
230 const auto denominators = divisors(remaining.leading().n);
231 for (auto numerator : numerators) {
232 for (auto denominator : denominators) {
233 if (std::gcd(numerator, denominator) == 1) {
234 candidates.emplace_back(numerator, denominator);
235 candidates.emplace_back(-numerator, denominator);
236 }
237 }
238 }
239 std::sort(candidates.begin(), candidates.end());
240 std::vector<Root> roots;
241 for (const auto& root : candidates) {
242 int multiplicity = 0;
243 const Polynomial factor = linear(-root.n, root.d);
244 while (remaining.degree() >= 1 && remaining.evaluate(root).n == 0) {
245 remaining = remaining.divide(factor).first;
246 ++multiplicity;
247 }
248 if (multiplicity > 0) roots.push_back({root, multiplicity});
249 }
250 if (zeros > 0) roots.push_back({Fraction(), zeros});
251 std::sort(roots.begin(), roots.end(), [](const Root& a, const Root& b) { return a.value < b.value; });
252 return {std::move(roots), std::move(remaining)};
253}

259std::vector<Polynomial> sturmSequence(const Polynomial& a) {
260 Polynomial previous = squareFreePart(a), current = previous.derivative();
261 std::vector<Polynomial> sequence{previous};
262 while (!current.zero()) {
263 sequence.push_back(current);
264 Polynomial next = -previous.mod(current);
265 previous = std::move(current);
266 current = std::move(next);
267 }
268 return sequence;
269}

301int variations(const std::vector<Polynomial>& sequence, const Endpoint& endpoint) {
302 std::vector<int> signs;
303 for (const auto& polynomial : sequence) {
304 int sign;
305 if (endpoint.kind == EndpointKind::Finite) sign = polynomial.evaluate(endpoint.value).compare(0);
306 else {
307 sign = polynomial.leading().compare(0);
308 if (endpoint.kind == EndpointKind::NegativeInfinity && polynomial.degree() % 2) sign = -sign;
309 }
310 signs.push_back(sign);
311 }
312 return signChanges(signs);

315std::vector<std::pair<Fraction, Fraction>> isolate(const Polynomial& a, int precision) {
316 const auto sequence = sturmSequence(a);
317 const Polynomial& first = sequence.front();
318 Fraction maximum;
319 for (int i = 0; i < first.degree(); ++i) maximum = std::max(maximum, (first.coefficients[i] / first.leading()).absolute());
320 const Fraction bound = Fraction(1) + maximum, width(1, std::int64_t(1) << precision);
321 std::vector<std::pair<Fraction, Fraction>> intervals;
322 const auto count = [&](Fraction left, Fraction right) { return variations(sequence, left) - variations(sequence, right); };
323 std::function<void(Fraction, Fraction)> examine = [&](Fraction left, Fraction right) {
324 const int roots = count(left, right);
325 if (roots >= 2) {
326 const Fraction middle = (left + right) / Fraction(2);
327 examine(left, middle);
328 examine(middle, right);
329 } else if (roots == 1) {
330 while (width < right - left) {
331 const Fraction middle = (left + right) / Fraction(2);
332 if (count(left, middle) >= 1) right = middle;
333 else left = middle;
334 }
335 intervals.emplace_back(left, right);
336 }
337 };
338 examine(-bound, bound);
339 return intervals;
340}

342Fraction resultant(Polynomial a, Polynomial b) {
343 Fraction result(1);
344 while (b.degree() >= 1) {
345 Polynomial remainder = a.mod(b);
346 if (remainder.zero()) return 0;
347 result = result * b.leading().power(a.degree() - remainder.degree());
348 if (a.degree() * b.degree() % 2) result = -result;
349 a = std::move(b);
350 b = std::move(remainder);
351 }
352 return result * b.coefficient(0).power(a.degree());
353}

355Polynomial interpolate(const std::vector<Endpoint>& values) {
356 std::vector<Fraction> xs, differences;
357 for (std::size_t i = 0; i < values.size(); i += 2) {
358 xs.push_back(values[i].value);
359 differences.push_back(values[i + 1].value);
360 }
361 const int count = static_cast<int>(xs.size());
362 for (int j = 1; j < count; ++j) {
363 for (int i = 0; i < j; ++i) if (xs[j] == xs[i]) fail("DUPLICATE X " + xs[j].notation());
364 }
365 for (int j = 1; j < count; ++j) {
366 for (int i = count - 1; i >= j; --i) differences[i] = (differences[i] - differences[i - 1]) / (xs[i] - xs[i - j]);
367 }
368 Polynomial result;
369 for (int i = count - 1; i >= 0; --i) result = result * linear(-xs[i]) + Polynomial::constant(differences[i]);
370 return result;
371}

373Polynomial special(const std::string& kind, int order) {
374 const std::set<std::string> kinds{"chebyshev", "chebyshev2", "legendre", "hermite", "laguerre"};
375 if (!kinds.count(kind)) fail("UNKNOWN KIND " + kind);
376 degreeLimit(order > 30);
377 if (order == 0) return unit;
378 Polynomial previous = unit;
379 Polynomial current = kind == "laguerre" ? linear(1, -1)
380 : kind == "chebyshev2" || kind == "hermite" ? variable.scale(2) : variable;
381 for (int j = 1; j < order; ++j) {
382 Polynomial next;
383 if (kind == "chebyshev" || kind == "chebyshev2") next = variable.scale(2) * current - previous;
384 else if (kind == "hermite") next = variable.scale(2) * current - previous.scale(2 * j);
385 else {
386 const Polynomial multiplier = kind == "legendre" ? variable.scale(2 * j + 1) : linear(2 * j + 1, -1);
387 next = (multiplier * current - previous.scale(j)).scale(Fraction(1, j + 1));
388 }
389 previous = std::move(current);
390 current = std::move(next);
391 }
392 return current;
393}

407class ExpressionParser {
408 std::vector<Token> tokens;
409 std::size_t position = 0;
410 const Token& token() const { return tokens[position]; }
411 bool take(char kind) {
412 if (token().kind != kind) return false;
413 ++position;
414 return true;
415 }
416 [[noreturn]] void syntaxError() const {
417 fail("SYNTAX AT " + (token().kind == '\0' ? std::string("END") : std::to_string(token().position)));
418 }
419 void expect(char kind) { if (!take(kind)) syntaxError(); }
420 Expression binary(NodeKind kind, Expression left, Expression right) {
421 auto result = std::make_unique<Node>(kind);
422 result->left = std::move(left);
423 result->right = std::move(right);
424 return result;
425 }
426 Expression expression() {
427 const bool negative = take('-');
428 Expression result = term();
429 if (negative) {
430 auto negation = std::make_unique<Node>(NodeKind::Negate);
431 negation->left = std::move(result);
432 result = std::move(negation);
433 }
434 while (token().kind == '+' || token().kind == '-') {
435 const NodeKind kind = token().kind == '+' ? NodeKind::Add : NodeKind::Subtract;
436 ++position;
437 Expression right = term();
438 result = binary(kind, std::move(result), std::move(right));
439 }
440 return result;
441 }
442 Expression term() {
443 Expression result = factor();
444 while (token().kind == '*' || token().kind == '/' || token().kind == 'x' || token().kind == 'N' || token().kind == '(') {
445 NodeKind kind = NodeKind::Multiply;
446 if (token().kind == '*' || token().kind == '/') {
447 if (token().kind == '/') kind = NodeKind::Divide;
448 ++position;
449 }
450 Expression right = factor();
451 result = binary(kind, std::move(result), std::move(right));
452 }
453 return result;
454 }
455 Expression factor() {
456 Expression result = primary();
457 if (take('^')) {
458 if (token().kind != '0') syntaxError();
459 auto power = std::make_unique<Node>(NodeKind::Power);
460 power->amount = std::stoi(token().word);
461 ++position;
462 power->left = std::move(result);
463 result = std::move(power);
464 }
465 return result;
466 }
467 Expression primary() {
468 const Token current = token();
469 if (take('0')) {
470 auto result = std::make_unique<Node>(NodeKind::Number);
471 result->value = Fraction(std::stoll(current.word));
472 return result;
473 }
474 if (take('x')) return std::make_unique<Node>(NodeKind::Variable);
475 if (take('N')) {
476 auto reference = std::make_unique<Node>(NodeKind::Name);
477 reference->name = current.word;
478 while (take('\'')) ++reference->amount;
479 if (!take('(')) return reference;
480 Expression argument = expression();
481 expect(')');
482 return binary(NodeKind::Call, std::move(reference), std::move(argument));
483 }
484 if (take('(')) {
485 Expression result = expression();
486 expect(')');
487 return result;
488 }
489 syntaxError();
490 }
491public:
492 explicit ExpressionParser(const std::string& source) {
493 const auto digit = [](char ch) { return '0' <= ch && ch <= '9'; };
494 const auto uppercase = [](char ch) { return 'A' <= ch && ch <= 'Z'; };
495 for (std::size_t i = 0; i < source.size();) {
496 if (source[i] == ' ') { ++i; continue; }
497 const std::size_t start = i++;
498 char kind = source[start];
499 if (digit(kind)) {
500 kind = '0';
501 while (i < source.size() && digit(source[i])) ++i;
502 } else if (uppercase(kind)) {
503 kind = 'N';
504 while (i < source.size() && (uppercase(source[i]) || digit(source[i]))) ++i;
505 }
506 tokens.push_back({kind, source.substr(start, i - start), static_cast<int>(start + 1)});
507 }
508 tokens.push_back({'\0', "", static_cast<int>(source.size() + 1)});
509 }
510 Expression parse() {
511 Expression result = expression();
512 if (token().kind != '\0') syntaxError();
513 return result;
514 }

535class Calculator {
536 std::map<std::string, Polynomial> polynomials;
537 const Polynomial& lookup(const std::string& name) const {
538 const auto found = polynomials.find(name);
539 if (found == polynomials.end()) fail("UNKNOWN " + name);
540 return found->second;
541 }
542 Polynomial evaluate(const Node& node) const {
543 switch (node.kind) {
544 case NodeKind::Number: return Polynomial::constant(node.value);
545 case NodeKind::Variable: return variable;
546 case NodeKind::Name: return lookup(node.name).derivative(node.amount);
547 case NodeKind::Negate: return -evaluate(*node.left);
548 case NodeKind::Power: {
549 const Polynomial value = evaluate(*node.left);
550 degreeLimit(!value.zero() && value.degree() * node.amount > 30);
551 return value.power(node.amount);
552 }
553 default: break;
554 }
555 const Polynomial left = evaluate(*node.left);
556 const Polynomial right = evaluate(*node.right);
557 if (node.kind == NodeKind::Add) return left + right;
558 if (node.kind == NodeKind::Subtract) return left - right;
559 if (node.kind == NodeKind::Multiply) { productLimit(left, right); return left * right; }
560 if (node.kind == NodeKind::Call) { compositionLimit(left, right); return left.compose(right); }
561 if (right.zero()) fail("DIVISION BY ZERO");
562 if (right.degree() > 0) fail("NONCONSTANT DIVISOR");
563 return left.divideScalar(right.leading());
564 }

566 std::vector<std::string> execute(const std::string& source) {
567 std::istringstream input(source);
568 std::vector<std::string> words;
569 for (std::string word; input >> word;) words.push_back(word);
570 const std::string command = words[0], name = words.size() > 1 ? words[1] : "";
571 std::vector<std::pair<std::string, Polynomial>> pending;
572 std::vector<std::string> output;
573 const auto store = [&](const std::string& target, const Polynomial& value) {
574 pending.emplace_back(target, value);
575 output.push_back(target + " = " + value.notation());
576 };
577 if (command == "DEF") {
578 Expression syntax = ExpressionParser(source.substr(name.size() + 5)).parse();
579 store(name, evaluate(*syntax));
580 } else {
581 if (command == "DIV" && words[1] == words[2]) fail("SAME NAME");
582 if (command == "XGCD" && std::set<std::string>(words.begin() + 1, words.begin() + 4).size() < 3) fail("SAME NAME");
583 std::vector<Polynomial> operands;
584 const auto positions = references.find(command);
585 if (positions != references.end()) for (int index : positions->second) operands.push_back(lookup(words[index]));
586 const Polynomial empty;
587 const Polynomial& a = operands.empty() ? empty : operands[0];
588 const Polynomial& b = operands.size() < 2 ? empty : operands[1];
589 std::size_t literalStart = words.size(), literalEnd = words.size();
590 if (command == "AREA" || command == "STURM") { literalStart = 2; literalEnd = 4; }
591 else if (command == "TANGENT") { literalStart = 3; literalEnd = 4; }
592 else if (command == "TAYLOR") { literalStart = 2; literalEnd = 3; }
593 else if (command == "EVAL" || command == "INTERP") literalStart = 2;
594 std::vector<Endpoint> values;
595 for (std::size_t i = literalStart; i < literalEnd; ++i) values.push_back(Endpoint::parse(words[i]));
596 if (command == "PRINT") output.push_back(name + " = " + a.notation());
597 else if (command == "DELETE") { polynomials.erase(name); output.push_back("deleted " + name); }
598 else if (command == "LIST") {
599 for (const auto& [key, value] : polynomials) output.push_back(key + " = " + value.notation());
600 if (output.empty()) output.push_back("no polynomials");
601 } else if (command == "DEGREE") output.push_back("deg(" + name + ") = " + (a.zero() ? "-inf" : std::to_string(a.degree())));
602 else if (command == "ADD") store(name, a + b);
603 else if (command == "SUB") store(name, a - b);
604 else if (command == "MUL") { productLimit(a, b); store(name, a * b); }
605 else if (command == "POW") {
606 const int exponent = std::stoi(words[3]);
607 degreeLimit(!a.zero() && a.degree() * exponent > 30);
608 store(name, a.power(exponent));
609 } else if (command == "DIV") {
610 if (b.zero()) fail("DIVISION BY ZERO");
611 const auto [quotient, remainder] = a.divide(b);
612 store(name, quotient);
613 store(words[2], remainder);
614 } else if (command == "GCD") store(name, polynomialGcd(a, b));
615 else if (command == "XGCD") {
616 const auto [common, s, t] = extendedGcd(a, b);
617 store(name, common); store(words[2], s); store(words[3], t);
618 } else if (command == "LCM") {
619 if (a.zero() || b.zero()) store(name, {});
620 else {
621 const Polynomial common = polynomialGcd(a, b);
622 degreeLimit(a.degree() + b.degree() - common.degree() > 30);
623 store(name, (a.divide(common).first * b).monic());
624 }
625 } else if (command == "INVMOD") {
626 if (b.zero()) fail("DIVISION BY ZERO");
627 if (b.degree() == 0) fail("DEGREE TOO LOW");
628 const auto [common, inverse, unused] = extendedGcd(a, b);
629 if (!isOne(common)) fail("NOT COPRIME");
630 store(name, inverse.mod(b));
631 } else if (command == "CRT") {
632 const Polynomial& c = operands[2];
633 const Polynomial& d = operands[3];
634 if (b.zero() || d.zero()) fail("DIVISION BY ZERO");
635 degreeLimit(b.degree() + d.degree() > 30);
636 const auto [common, inverse, unused] = extendedGcd(b, d);
637 if (!isOne(common)) fail("NOT COPRIME");
638 const Polynomial first = a.mod(b), second = c.mod(d);
639 const Polynomial correction = ((second - first) * inverse).mod(d);
640 store(name, first + b * correction);
641 } else if (command == "MODPOW") {
642 if (b.zero()) fail("DIVISION BY ZERO");
643 degreeLimit(2 * b.degree() - 2 > 30);
644 Polynomial result = unit.mod(b), base = a.mod(b);
645 int exponent = std::stoi(words[3]);
646 while (exponent > 0) {
647 if (exponent % 2) result = (result * base).mod(b);
648 exponent /= 2;
649 if (exponent > 0) base = (base * base).mod(b);
650 }
651 store(name, result);
652 } else if (command == "DERIV") store(name, a.derivative(std::stoi(words[3])));
653 else if (command == "INTEG") { degreeLimit(!a.zero() && a.degree() + 1 > 30); store(name, a.integral()); }
654 else if (command == "AREA") {
655 const Polynomial integral = a.integral();
656 const Fraction result = integral.evaluate(values[1].value) - integral.evaluate(values[0].value);
657 output.push_back("integral(" + name + ", " + values[0].notation() + ", " + values[1].notation() + ") = " + result.notation());
658 } else if (command == "TANGENT") {
659 const Fraction point = values[0].value, value = a.evaluate(point), slope = a.derivative().evaluate(point);
660 store(name, linear(value - slope * point, slope));
661 } else if (command == "TAYLOR") {
662 const Fraction point = values[0].value;
663 const std::string symbol = point.n == 0 ? "x" : point.n > 0 ? "(x - " + point.notation() + ")" : "(x + " + (-point).notation() + ")";
664 output.push_back("taylor(" + name + ", " + point.notation() + ") = " + a.compose(linear(point)).notation(symbol));
665 } else if (command == "COMPOSE") { compositionLimit(a, b); store(name, a.compose(b)); }
666 else if (command == "EVAL") {
667 for (const auto& point : values) output.push_back(name + "(" + point.notation() + ") = " + a.evaluate(point.value).notation());
668 } else if (command == "ROOTS" || command == "FACTOR") {
669 requireNonzero(a);
670 const auto factorization = rationalRoots(a);
671 if (command == "ROOTS") {
672 std::vector<std::string> roots;
673 for (const auto& root : factorization.roots) roots.push_back(root.value.notation() + " (" + std::to_string(root.multiplicity) + ")");
674 output.push_back("roots(" + name + ") = " + (roots.empty() ? "none" : join(roots, ", ")));
675 } else {
676 std::string result = "factor(" + name + ") = " + a.content().notation();
677 for (const auto& root : factorization.roots) {
678 result += " * " + (root.value.n == 0 ? std::string("x") : "(" + linear(-root.value.n, root.value.d).notation() + ")");
679 if (root.multiplicity > 1) result += "^" + std::to_string(root.multiplicity);
680 }
681 if (factorization.remaining.degree() >= 1) result += " * (" + factorization.remaining.notation() + ")";
682 output.push_back(result);
683 }
684 } else if (command == "SQFREE") {
685 requireNonzero(a);
686 std::string result = "sqfree(" + name + ") = " + a.content().notation();
687 if (a.degree() >= 1) {
688 const Polynomial monic = a.monic();
689 Polynomial common = polynomialGcd(monic, monic.derivative()), remaining = monic.divide(common).first;
690 for (int multiplicity = 1; remaining.degree() >= 1; ++multiplicity) {
691 const Polynomial shared = polynomialGcd(remaining, common), factor = remaining.divide(shared).first;
692 if (factor.degree() >= 1) {
693 const std::string notation = factor.primitive().notation();
694 result += " * " + (notation == "x" ? notation : "(" + notation + ")");
695 if (multiplicity > 1) result += "^" + std::to_string(multiplicity);
696 }
697 common = common.divide(shared).first;
698 remaining = shared;
699 }
700 }
701 output.push_back(result);
702 } else if (command == "DESCARTES") {
703 requireNonzero(a);
704 std::vector<int> positive, negative;
705 for (int i = a.degree(); i >= 0; --i) {
706 const int sign = a.coefficients[i].compare(0);
707 positive.push_back(sign);
708 negative.push_back(i % 2 ? -sign : sign);
709 }
710 output.push_back("descartes(" + name + ") = positive " + std::to_string(signChanges(positive)) + ", negative " + std::to_string(signChanges(negative)));
711 } else if (command == "STURM") {
712 requireNonzero(a);
713 if (values[0].compare(values[1]) >= 0) fail("BAD INTERVAL");
714 const auto sequence = sturmSequence(a);
715 const int count = variations(sequence, values[0]) - variations(sequence, values[1]);
716 output.push_back("sturm(" + name + ", " + values[0].notation() + ", " + values[1].notation() + ") = " + std::to_string(count));
717 } else if (command == "ISOLATE") {
718 requireNonzero(a);
719 const int precision = std::stoi(words[2]);
720 std::vector<std::string> intervals;
721 for (const auto& [left, right] : isolate(a, precision)) intervals.push_back("(" + left.notation() + ", " + right.notation() + "]");
722 output.push_back("isolate(" + name + ", " + std::to_string(precision) + ") = " + (intervals.empty() ? "none" : join(intervals, ", ")));
723 } else if (command == "RESULTANT") {
724 requireNonzero(a); requireNonzero(b);
725 output.push_back("res(" + name + ", " + words[2] + ") = " + resultant(a, b).notation());
726 } else if (command == "DISCRIM") {
727 requireNonzero(a);
728 if (a.degree() == 0) fail("DEGREE TOO LOW");
729 Fraction discriminant = resultant(a, a.derivative()) / a.leading();
730 if (a.degree() * (a.degree() - 1) / 2 % 2) discriminant = -discriminant;
731 output.push_back("disc(" + name + ") = " + discriminant.notation());
732 } else if (command == "POWERSUM") {
733 requireNonzero(a);
734 if (a.degree() == 0) fail("DEGREE TOO LOW");
735 const Polynomial monic = a.monic();
736 std::vector<Fraction> sums(1);
737 std::vector<std::string> parts;
738 const int count = std::stoi(words[2]);
739 for (int j = 1; j <= count; ++j) {
740 Fraction sum = j <= a.degree() ? Fraction(j) * monic.coefficient(a.degree() - j) : Fraction();
741 for (int i = 1; i <= std::min(j - 1, a.degree()); ++i) sum = sum + monic.coefficient(a.degree() - i) * sums[j - i];
742 sums.push_back(-sum);
743 parts.push_back((-sum).notation());
744 }
745 output.push_back("powersum(" + name + ") = " + join(parts, ", "));
746 } else if (command == "INTERP") store(name, interpolate(values));
747 else if (command == "SPECIAL") store(name, special(words[2], std::stoi(words[3])));
748 }
749 for (auto& [target, value] : pending) polynomials[target] = std::move(value);
750 return output;
751 }
752};

759 Calculator calculator;
760 for (int i = 0; i < count; ++i) {
761 std::getline(std::cin, source);
762 try {
763 for (const auto& line : calculator.execute(source)) std::cout << line << '\n';
764 } catch (const CalculatorError& error) {
765 std::cout << "ERROR " << error.what() << '\n';
766 }
767 }
