683 case. 'ROOTS';'FACTOR';'SQFREE';'DESCARTES';'STURM';'ISOLATE';'RESULTANT';'DISCRIM';'POWERSUM' do.
684 if. 0=#a do. fail 'ZERO POLYNOMIAL' end.
685 select. command
686 case. 'ROOTS' do.
687 'roots remaining' =. rational_roots a
688 terms =. 0$a:
689 for_row. roots do.
690 'r multiplicity' =. row
691 terms =. terms,<(fraction r),' (',(number multiplicity),')'
692 end.
693 text =. 'none'
694 if. #terms do. text =. ', ' join terms end.
695 emit 'roots(',name,') = ',text
696 case. 'FACTOR' do. emit 'factor(',name,') = ',factorization a
697 case. 'SQFREE' do. emit 'sqfree(',name,') = ',squarefree_factors a
698 case. 'DESCARTES' do.
699 positive =. sign_changes |.a
700 negative =. sign_changes |.a*_1x^i.#a
701 emit 'descartes(',name,') = positive ',(number positive),', negative ',number negative
702 case. 'STURM' do.
703 'u v' =. literals
704 if. -.u before v do. fail 'BAD INTERVAL' end.
705 count =. (sturm_sequence a) root_count u;v
706 emit 'sturm(',name,', ',(endpoint_text u),', ',(endpoint_text v),') = ',number count
707 case. 'ISOLATE' do.
708 n =. integer 2{::args
709 IsolationSequence =: sturm_sequence a
710 IsolationWidth =: 1x%2x^n
711 first =. 0{::IsolationSequence
712 bound =. 1x
713 if. 1<#first do. bound =. 1x+>./|(}:first)%{:first end.
714 intervals =. examine (-bound),bound
715 terms =. 0$a:
716 for_interval. intervals do.
717 'low high' =. >interval
718 terms =. terms,<'(',(fraction low),', ',(fraction high),']'
719 end.
720 text =. 'none'
721 if. #terms do. text =. ', ' join terms end.
722 emit 'isolate(',name,', ',(number n),') = ',text
723 case. 'RESULTANT' do.
724 if. 0=#b do. fail 'ZERO POLYNOMIAL' end.
725 emit 'res(',name,', ',(2{::args),') = ',fraction a resultant b
726 case. 'DISCRIM' do.
727 if. 1=#a do. fail 'DEGREE TOO LOW' end.
728 d =. degree a
729 result =. (a resultant derivative a)%{:a
730 if. 2|<.(d*(d-1))%2x do. result =. -result end.
731 emit 'disc(',name,') = ',fraction result
732 case. 'POWERSUM' do.
733 if. 1=#a do. fail 'DEGREE TOO LOW' end.
734 d =. degree a
735 c =. monic a
736 values =. ,0x
737 for_j. 1+i.integer 2{::args do.
738 total =. 0x
739 if. j<:d do. total =. j*(d-j){c end.
740 for_i. 1+i.(j-1)<.d do. total =. total+((d-i){c)*(j-i){values end.
741 values =. values,-total
742 end.
743 emit 'powersum(',name,') = ', ', ' join fraction each <"0 }.values
744 end.
745 case. 'INTERP' do. name store_poly interpolate _2]\>literals
746 case. 'SPECIAL' do. name store_poly (2{::args) special integer 3{::args
747 end.
748)
749main =: 3 : 0
750 count =. integer 1!:1 3
751 for_i. i.count do.
752 line =. 1!:1 3
753 try. execute line
754 catcht. emit 'ERROR ',Error
755 end.
756 end.
757)
758main ''
759exit 0
