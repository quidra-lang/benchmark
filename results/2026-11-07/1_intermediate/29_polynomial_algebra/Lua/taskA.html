5local function gcd(a, b)
6 a, b = math.abs(a), math.abs(b)
7 while b ~= 0 do a, b = b, a % b end
8 return a
9end

14local function fraction(n, d)
15 d = d or 1
16 if d < 0 then n, d = -n, -d end
17 local common = gcd(n, d)
18 return setmetatable({n = n // common, d = d // common}, Fraction)
19end

62local function parse_fraction(text)
63 local numerator, denominator = text:match("^(-?[0-9]+)/([0-9]+)$")
64 if numerator == nil then return fraction(tonumber(text, 10)) end

73local function polynomial(values)
74 local size = #values
75 while size > 0 and values[size].n == 0 do size = size - 1 end
76 local result = {}
77 for i = 1, size do result[i] = values[i] end
78 return setmetatable(result, Polynomial)

107function Polynomial.__mul(a, b)
108 if #a == 0 or #b == 0 then return PZERO end
109 local result = {}
110 for i = 1, #a + #b - 1 do result[i] = ZERO end
111 for i, left in ipairs(a) do

113 end
114 return polynomial(result)

129function Polynomial.__pow(a, exponent)
130 if exponent == 0 then return PONE end
131 if #a == 0 then return PZERO end
132 if #a == 1 and a[1].d == 1 and math.abs(a[1].n) == 1 then
133 return exponent % 2 == 0 and PONE or a
134 end

136 for _ = 1, exponent do result = result * a end
137 return result

140function Polynomial:divrem(divisor)
141 local quotient, remainder = {}, {}
142 for i = 1, math.max(0, #self - #divisor + 1) do quotient[i] = ZERO end
143 for i, value in ipairs(self) do remainder[i] = value end
144 while #remainder > 0 and #remainder >= #divisor do
145 local shift = #remainder - #divisor
146 local factor = remainder[#remainder] / divisor:leading()
147 quotient[shift + 1] = factor
148 for j, value in ipairs(divisor) do remainder[shift + j] = remainder[shift + j] - factor * value end
149 while #remainder > 0 and remainder[#remainder].n == 0 do table.remove(remainder) end
150 end
151 return polynomial(quotient), polynomial(remainder)
152end

202local function extended_gcd(a, b)
203 local s0, s1, t0, t1 = PONE, PZERO, PZERO, PONE
204 while #b > 0 do
205 local quotient, remainder = a:divrem(b)
206 a, b = b, remainder
207 s0, s1 = s1, s0 - quotient * s1
208 t0, t1 = t1, t0 - quotient * t1
209 end
210 if #a == 0 then return a, s0, t0 end
211 local divisor = a:leading()
212 return a:divide_scalar(divisor), s0:divide_scalar(divisor), t0:divide_scalar(divisor)
213end

215function Polynomial:content()
216 local multiple, common = 1, 0
217 for _, value in ipairs(self) do
218 multiple = multiple * value.d // gcd(multiple, value.d)
219 common = gcd(common, value.n)
220 end
221 return fraction(self:leading().n < 0 and -common or common, multiple)
222end

257local function rational_roots(a)
258 local remaining, zero_count = a:primitive(), 0
259 while remaining[1].n == 0 do
260 local coefficients = {}
261 for i = 2, #remaining do coefficients[i - 1] = remaining[i] end
262 remaining = polynomial(coefficients)
263 zero_count = zero_count + 1
264 end
265 local candidates = {}
266 for _, n in ipairs(divisors(math.abs(remaining[1].n))) do
267 for _, d in ipairs(divisors(remaining:leading().n)) do
268 if gcd(n, d) == 1 then
269 candidates[#candidates + 1] = fraction(-n, d)
270 candidates[#candidates + 1] = fraction(n, d)
271 end
272 end
273 end
274 table.sort(candidates)
275 local roots = {}
276 for _, value in ipairs(candidates) do
277 local factor = linear(fraction(-value.n), fraction(value.d))
278 local multiplicity = 0
279 while #remaining >= 2 and remaining:evaluate(value).n == 0 do
280 remaining = remaining:quotient(factor)
281 multiplicity = multiplicity + 1
282 end
283 if multiplicity > 0 then roots[#roots + 1] = {value = value, multiplicity = multiplicity} end
284 end
285 if zero_count > 0 then roots[#roots + 1] = {value = ZERO, multiplicity = zero_count} end
286 table.sort(roots, function(left, right) return left.value < right.value end)
287 return roots, remaining
288end

305local function squarefree_factors(a)
306 local result = tostring(a:content())
307 if #a == 1 then return result end
308 local normalized = a:monic()
309 local common = polynomial_gcd(normalized, normalized:derivative())
310 local remaining, multiplicity = normalized:quotient(common), 1
311 while #remaining >= 2 do
312 local shared = polynomial_gcd(remaining, common)
313 local factor = remaining:quotient(shared)
314 if #factor >= 2 then
315 local text = factor:primitive():format()
316 result = result .. factor_text(text == "x" and text or "(" .. text .. ")", multiplicity)
317 end
318 common, remaining = common:quotient(shared), shared
319 multiplicity = multiplicity + 1
320 end
321 return result
322end

376local function isolate(a, precision)
377 local sequence = sturm_sequence(a)
378 local first, maximum = sequence[1], ZERO
379 for i = 1, #first - 1 do
380 local value = (first[i] / first:leading()):absolute()
381 if maximum < value then maximum = value end
382 end
383 local beta, width = ONE + maximum, fraction(1, 1 << precision)
384 local intervals = {}
385 local function examine(low, high)
386 local count = root_count(sequence, endpoint(low), endpoint(high))
387 if count >= 2 then
388 local middle = (low + high) / 2
389 examine(low, middle)
390 examine(middle, high)
391 elseif count == 1 then
392 while width < high - low do
393 local middle = (low + high) / 2
394 if root_count(sequence, endpoint(low), endpoint(middle)) >= 1 then high = middle
395 else low = middle end
396 end
397 intervals[#intervals + 1] = "(" .. tostring(low) .. ", " .. tostring(high) .. "]"
398 end
399 end
400 examine(-beta, beta)
401 return #intervals == 0 and "none" or table.concat(intervals, ", ")
402end

404local function resultant(a, b)
405 local result = ONE
406 while #b >= 2 do
407 local remainder = a:modulo(b)
408 if #remainder == 0 then return ZERO end
409 result = result * scalar_power(b:leading(), a:degree() - remainder:degree())
410 if a:degree() * b:degree() % 2 == 1 then result = -result end
411 a, b = b, remainder
412 end
413 return result * scalar_power(b[1], a:degree())
414end

416local function power_sums(a, count)
417 local d, normalized, sums, output = a:degree(), a:monic(), {}, {}
418 for j = 1, count do
419 local value = j <= d and j * normalized:at(d - j) or ZERO
420 for i = 1, math.min(j - 1, d) do value = value + normalized:at(d - i) * sums[j - i] end
421 sums[j] = -value
422 output[j] = tostring(sums[j])
423 end
424 return table.concat(output, ", ")
425end

427local function interpolate(values)
428 local xs, differences = {}, {}
429 for i = 1, #values, 2 do
430 xs[#xs + 1] = values[i]
431 differences[#differences + 1] = values[i + 1]
432 end
433 for j = 2, #xs do
434 for i = 1, j - 1 do
435 if xs[i] == xs[j] then fail("DUPLICATE X " .. tostring(xs[j])) end
436 end
437 end
438 for j = 1, #xs - 1 do
439 for i = #xs, j + 1, -1 do
440 differences[i] = (differences[i] - differences[i - 1]) / (xs[i] - xs[i - j])
441 end
442 end
443 local result = PZERO
444 for i = #xs, 1, -1 do result = result * linear(-xs[i]) + constant(differences[i]) end
445 return result
446end

448local function special(kind, count)
449 if kind ~= "chebyshev" and kind ~= "chebyshev2" and kind ~= "legendre" and kind ~= "hermite" and kind ~= "laguerre" then
450 fail("UNKNOWN KIND " .. kind)
451 end
452 degree_limit(count > 30)
453 local previous = PONE
454 if count == 0 then return previous end
455 local current = (kind == "chebyshev2" or kind == "hermite") and X:scale(fraction(2))
456 or kind == "laguerre" and linear(ONE, -ONE) or X
457 for j = 1, count - 1 do
458 local factor = kind == "legendre" and X:scale(fraction(2 * j + 1))
459 or kind == "laguerre" and linear(fraction(2 * j + 1), -ONE) or X:scale(fraction(2))
460 local multiplier = (kind == "legendre" or kind == "laguerre") and j or kind == "hermite" and 2 * j or 1
461 local next_value = factor * current - previous:scale(fraction(multiplier))
462 if kind == "legendre" or kind == "laguerre" then next_value = next_value:scale(fraction(1, j + 1)) end
463 previous, current = current, next_value
464 end
465 return current
466end

468local Parser = {}

471local function digit(byte) return byte ~= nil and byte >= 48 and byte <= 57 end
472local function upper(byte) return byte ~= nil and byte >= 65 and byte <= 90 end
473
474function Parser.new(source)
475 local tokens, i = {}, 1
476 while i <= #source do
477 if source:sub(i, i) == " " then
478 i = i + 1
479 else
480 local start, kind = i, "symbol"
481 i = i + 1
482 if digit(source:byte(start)) then
483 kind = "number"
484 while digit(source:byte(i)) do i = i + 1 end
485 elseif upper(source:byte(start)) then
486 kind = "name"
487 while upper(source:byte(i)) or digit(source:byte(i)) do i = i + 1 end
488 end
489 tokens[#tokens + 1] = {kind = kind, text = source:sub(start, i - 1), position = start}
490 end
491 end
492 tokens[#tokens + 1] = {kind = "end", text = "", position = #source + 1}
493 return setmetatable({tokens = tokens, position = 1}, Parser)
494end
495
496function Parser:current() return self.tokens[self.position] end
497function Parser:syntax_error()
498 local token = self:current()
499 fail("SYNTAX AT " .. (token.kind == "end" and "END" or tostring(token.position)))
500end
501
502function Parser:accept(text)
503 local token = self:current()
504 if token.kind ~= "symbol" or token.text ~= text then return false end
505 self.position = self.position + 1
506 return true
507end
508
509function Parser:expect(text) if not self:accept(text) then self:syntax_error() end end
510
511function Parser:primary()
512 local token = self:current()
513 if token.kind == "number" then
514 self.position = self.position + 1
515 return {kind = "literal", value = constant(fraction(tonumber(token.text, 10)))}
516 end
517 if self:accept("x") then return {kind = "variable"} end
518 if token.kind == "name" then
519 self.position = self.position + 1
520 local order = 0
521 while self:accept("'") do order = order + 1 end
522 local reference = {kind = "reference", name = token.text, order = order}
523 if not self:accept("(") then return reference end
524 local argument = self:expression()
525 self:expect(")")
526 return {kind = "call", left = reference, right = argument}
527 end
528 if self:accept("(") then
529 local result = self:expression()
530 self:expect(")")
531 return result
532 end
533 self:syntax_error()
534end
535
536function Parser:factor()
537 local result = self:primary()
538 if self:accept("^") then
539 if self:current().kind ~= "number" then self:syntax_error() end
540 result = {kind = "power", left = result, exponent = tonumber(self:current().text, 10)}
541 self.position = self.position + 1
542 end
543 return result
544end
545
546function Parser:term()
547 local result = self:factor()
548 while true do
549 local operation
550 if self:accept("*") then operation = "*"
551 elseif self:accept("/") then operation = "/"
552 elseif self:current().kind == "name" or self:current().text == "x" or self:current().text == "(" then operation = "*"
553 else break end
554 result = {kind = operation, left = result, right = self:factor()}
555 end
556 return result
557end
558
559function Parser:expression()
560 local negative = self:accept("-")
561 local result = self:term()
562 if negative then result = {kind = "negative", left = result} end
563 while true do
564 local operation
565 if self:accept("+") then operation = "+"
566 elseif self:accept("-") then operation = "-"
567 else break end
568 result = {kind = operation, left = result, right = self:term()}
569 end
570 return result
571end
572
573function Parser:parse()
574 local result = self:expression()
575 if self:current().kind ~= "end" then self:syntax_error() end
576 return result
577end

592local function evaluate_tree(node, storage)
593 if node.kind == "literal" then return node.value end
594 if node.kind == "variable" then return X end
595 if node.kind == "reference" then return lookup(storage, node.name):derivative(node.order) end
596 local a = evaluate_tree(node.left, storage)
597 if node.kind == "negative" then return -a end
598 if node.kind == "power" then
599 degree_limit(#a > 0 and a:degree() * node.exponent > 30)
600 return a ^ node.exponent
601 end
602 local b = evaluate_tree(node.right, storage)
603 if node.kind == "+" then return a + b end
604 if node.kind == "-" then return a - b end
605 if node.kind == "*" then check_product(a, b); return a * b end
606 if node.kind == "call" then check_composition(a, b); return a:compose(b) end
607 divisor(b)
608 if #b >= 2 then fail("NONCONSTANT DIVISOR") end
609 return a:divide_scalar(b[1])
610end

620local function execute(storage, source)
621 local words = {}
622 for word in source:gmatch("[^ ]+") do words[#words + 1] = word end
623 local command, name = words[1], words[2]
624 local output, pending, deletion = {}, {}, nil
625 local function save(destination, value)
626 pending[#pending + 1] = {name = destination, value = value}
627 output[#output + 1] = destination .. " = " .. value:format()
628 end
629 if command == "DEF" then
630 local tree = Parser.new(source:sub(#name + 6)):parse()
631 save(name, evaluate_tree(tree, storage))
632 else
633 if command == "DIV" and words[2] == words[3]
634 or command == "XGCD" and (words[2] == words[3] or words[2] == words[4] or words[3] == words[4]) then
635 fail("SAME NAME")
636 end
637 local operands = {}
638 for _, i in ipairs(references[command] or {}) do operands[#operands + 1] = lookup(storage, words[i]) end
639 local a, b = operands[1] or PZERO, operands[2] or PZERO
640 local literal_words, values, bounds = {}, {}, {}
641 if command == "AREA" or command == "STURM" then literal_words = {words[3], words[4]}
642 elseif command == "TANGENT" then literal_words = {words[4]}
643 elseif command == "TAYLOR" then literal_words = {words[3]}
644 elseif command == "EVAL" or command == "INTERP" then
645 for i = 3, #words do literal_words[#literal_words + 1] = words[i] end
646 end
647 for i, word in ipairs(literal_words) do
648 if command == "STURM" then bounds[i] = parse_endpoint(word)
649 else values[i] = parse_fraction(word) end
650 end
651 if command == "PRINT" then
652 output[1] = name .. " = " .. a:format()
653 elseif command == "DELETE" then
654 deletion = name
655 output[1] = "deleted " .. name
656 elseif command == "LIST" then
657 local names = {}
658 for key in pairs(storage) do names[#names + 1] = key end
659 table.sort(names, ascii_less)
660 for _, key in ipairs(names) do output[#output + 1] = key .. " = " .. storage[key]:format() end
661 if #output == 0 then output[1] = "no polynomials" end
662 elseif command == "DEGREE" then
663 output[1] = "deg(" .. name .. ") = " .. (#a == 0 and "-inf" or tostring(a:degree()))
664 elseif command == "ADD" then save(name, a + b)
665 elseif command == "SUB" then save(name, a - b)
666 elseif command == "MUL" then check_product(a, b); save(name, a * b)
667 elseif command == "POW" then
668 local exponent = tonumber(words[4], 10)
669 degree_limit(#a > 0 and a:degree() * exponent > 30)
670 save(name, a ^ exponent)
671 elseif command == "DIV" then
672 divisor(b)
673 local quotient, remainder = a:divrem(b)
674 save(name, quotient)
675 save(words[3], remainder)
676 elseif command == "GCD" then save(name, polynomial_gcd(a, b))
677 elseif command == "XGCD" then
678 local common, s, t = extended_gcd(a, b)
679 save(name, common)
680 save(words[3], s)
681 save(words[4], t)
682 elseif command == "LCM" then
683 local result = PZERO
684 if #a > 0 and #b > 0 then
685 local common = polynomial_gcd(a, b)
686 degree_limit(a:degree() + b:degree() - common:degree() > 30)
687 result = (a:quotient(common) * b):monic()
688 end
689 save(name, result)
690 elseif command == "INVMOD" then
691 divisor(b)
692 if #b == 1 then fail("DEGREE TOO LOW") end
693 local common, inverse = extended_gcd(a, b)
694 if not common:is_one() then fail("NOT COPRIME") end
695 save(name, inverse:modulo(b))
696 elseif command == "CRT" then
697 local c, d = operands[3], operands[4]
698 divisor(b)
699 divisor(d)
700 degree_limit(b:degree() + d:degree() > 30)
701 local common, inverse = extended_gcd(b, d)
702 if not common:is_one() then fail("NOT COPRIME") end
703 local first, second = a:modulo(b), c:modulo(d)
704 local correction = ((second - first) * inverse):modulo(d)
705 save(name, first + b * correction)
706 elseif command == "MODPOW" then
707 divisor(b)
708 degree_limit(2 * b:degree() - 2 > 30)
709 local result, base = PONE:modulo(b), a:modulo(b)
710 local exponent = tonumber(words[4], 10)
711 while exponent > 0 do
712 if exponent % 2 == 1 then result = (result * base):modulo(b) end
713 exponent = exponent // 2
714 if exponent > 0 then base = (base * base):modulo(b) end
715 end
716 save(name, result)
717 elseif command == "DERIV" then save(name, a:derivative(tonumber(words[4], 10)))
718 elseif command == "INTEG" then
719 degree_limit(#a > 0 and a:degree() + 1 > 30)
720 save(name, a:integral())
721 elseif command == "AREA" then
722 local antiderivative = a:integral()
723 local high = antiderivative:evaluate(values[2])
724 local low = antiderivative:evaluate(values[1])
725 output[1] = "integral(" .. name .. ", " .. tostring(values[1]) .. ", " .. tostring(values[2]) .. ") = " .. tostring(high - low)
726 elseif command == "TANGENT" then
727 local point = values[1]
728 local value = a:evaluate(point)
729 local slope = a:derivative():evaluate(point)
730 save(name, linear(value - slope * point, slope))
731 elseif command == "TAYLOR" then
732 local point = values[1]
733 local variable = point.n == 0 and "x" or point.n > 0 and "(x - " .. tostring(point) .. ")"
734 or "(x + " .. tostring(-point) .. ")"
735 output[1] = "taylor(" .. name .. ", " .. tostring(point) .. ") = " .. a:compose(linear(point)):format(variable)
736 elseif command == "COMPOSE" then check_composition(a, b); save(name, a:compose(b))
737 elseif command == "EVAL" then
738 for _, point in ipairs(values) do output[#output + 1] = name .. "(" .. tostring(point) .. ") = " .. tostring(a:evaluate(point)) end
739 elseif command == "ROOTS" or command == "FACTOR" then
740 nonzero(a)
741 local roots, remaining = rational_roots(a)
742 if command == "ROOTS" then
743 local descriptions = {}
744 for _, root in ipairs(roots) do descriptions[#descriptions + 1] = tostring(root.value) .. " (" .. root.multiplicity .. ")" end
745 output[1] = "roots(" .. name .. ") = " .. (#descriptions == 0 and "none" or table.concat(descriptions, ", "))
746 else output[1] = "factor(" .. name .. ") = " .. factorization(a, roots, remaining) end
747 elseif command == "SQFREE" then
748 nonzero(a)
749 output[1] = "sqfree(" .. name .. ") = " .. squarefree_factors(a)
750 elseif command == "DESCARTES" then
751 nonzero(a)
752 local positive, negative = {}, {}
753 for i = a:degree(), 0, -1 do
754 local value = sign(a:at(i))
755 positive[#positive + 1] = value
756 negative[#negative + 1] = i % 2 == 0 and value or -value
757 end
758 output[1] = "descartes(" .. name .. ") = positive " .. sign_changes(positive) .. ", negative " .. sign_changes(negative)
759 elseif command == "STURM" then
760 nonzero(a)
761 if not endpoint_less(bounds[1], bounds[2]) then fail("BAD INTERVAL") end
762 local sequence = sturm_sequence(a)
763 output[1] = "sturm(" .. name .. ", " .. endpoint_text(bounds[1]) .. ", " .. endpoint_text(bounds[2]) .. ") = " .. root_count(sequence, bounds[1], bounds[2])
764 elseif command == "ISOLATE" then
765 nonzero(a)
766 local precision = tonumber(words[3], 10)
767 output[1] = "isolate(" .. name .. ", " .. precision .. ") = " .. isolate(a, precision)
768 elseif command == "RESULTANT" then
769 nonzero(a)
770 nonzero(b)
771 output[1] = "res(" .. name .. ", " .. words[3] .. ") = " .. tostring(resultant(a, b))
772 elseif command == "DISCRIM" or command == "POWERSUM" then
773 nonzero(a)
774 if #a == 1 then fail("DEGREE TOO LOW") end
775 if command == "DISCRIM" then
776 local value = resultant(a, a:derivative()) / a:leading()
777 if a:degree() * (a:degree() - 1) // 2 % 2 == 1 then value = -value end
778 output[1] = "disc(" .. name .. ") = " .. tostring(value)
779 else output[1] = "powersum(" .. name .. ") = " .. power_sums(a, tonumber(words[3], 10)) end
780 elseif command == "INTERP" then save(name, interpolate(values))
781 elseif command == "SPECIAL" then save(name, special(words[3], tonumber(words[4], 10))) end
782 end
783 if deletion ~= nil then storage[deletion] = nil end
784 for _, item in ipairs(pending) do storage[item.name] = item.value end
785 return output
786end

789local count = tonumber(io.read("l"), 10)

794 elseif type(result) == "table" and result.command_error then
795 io.write("ERROR ", result.message, "\n")
796 else
797 error(result, 0)
798 end
