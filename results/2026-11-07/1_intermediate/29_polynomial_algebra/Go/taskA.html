14func failure(message string) error { return errors.New("ERROR " + message) }
15func abs(n int64) int64 { if n < 0 { return -n }; return n }
16func gcd(a, b int64) int64 {
17 a, b = abs(a), abs(b)
18 for b != 0 { a, b = b, a%b }
19 return a
20}

23type fraction struct { n, d int64 }
24var zero = fraction{0, 1}
25var one = fraction{1, 1}
26func rational(n, d int64) fraction {
27 if d < 0 { n, d = -n, -d }
28 g := gcd(n, d)
29 return fraction{n/g, d/g}
30}
31func whole(n int) fraction { return fraction{int64(n), 1} }
32func parseFraction(text string) (fraction, error) {
33 a, b, found := strings.Cut(text, "/")
34 n, _ := strconv.ParseInt(a, 10, 64)
35 d := int64(1)
36 if found { d, _ = strconv.ParseInt(b, 10, 64) }
37 if d == 0 { return zero, failure("ZERO DENOMINATOR") }
38 return rational(n, d), nil
39}
40func (a fraction) add(b fraction) fraction { return rational(a.n*b.d+b.n*a.d, a.d*b.d) }
41func (a fraction) sub(b fraction) fraction { return rational(a.n*b.d-b.n*a.d, a.d*b.d) }
42func (a fraction) mul(b fraction) fraction { return rational(a.n*b.n, a.d*b.d) }
43func (a fraction) div(b fraction) fraction { return rational(a.n*b.d, a.d*b.n) }
44func (a fraction) neg() fraction { return fraction{-a.n, a.d} }
45func (a fraction) abs() fraction { return fraction{abs(a.n), a.d} }
46func (a fraction) sign() int { return cmp.Compare(a.n, 0) }
47func (a fraction) compare(b fraction) int { return cmp.Compare(a.n*b.d, b.n*a.d) }
48func (a fraction) power(n int) fraction {
49 result := one
50 for range n { result = result.mul(a) }
51 return result
52}
53func (a fraction) String() string {
54 if a.d == 1 { return strconv.FormatInt(a.n, 10) }
55 return fmt.Sprintf("%d/%d", a.n, a.d)
56}

58type polynomial []fraction
59func coefficients(n int) polynomial {
60 p := make(polynomial, n)
61 for i := range p { p[i] = zero }
62 return p
63}
64func normalize(p polynomial) polynomial {
65 for len(p) > 0 && p[len(p)-1].n == 0 { p = p[:len(p)-1] }
66 return p
67}
68func constant(value fraction) polynomial { return normalize(polynomial{value}) }
69func linear(value, slope fraction) polynomial { return normalize(polynomial{value, slope}) }
70func (a polynomial) degree() int { return len(a)-1 }
71func (a polynomial) leading() fraction { return a[len(a)-1] }
72func (a polynomial) at(i int) fraction { if i < len(a) { return a[i] }; return zero }
73func (a polynomial) add(b polynomial) polynomial {
74 p := coefficients(max(len(a), len(b)))
75 for i := range p { p[i] = a.at(i).add(b.at(i)) }
76 return normalize(p)
77}
78func (a polynomial) sub(b polynomial) polynomial {
79 p := coefficients(max(len(a), len(b)))
80 for i := range p { p[i] = a.at(i).sub(b.at(i)) }
81 return normalize(p)
82}
83func (a polynomial) neg() polynomial {
84 p := coefficients(len(a))
85 for i, c := range a { p[i] = c.neg() }
86 return p
87}
88func (a polynomial) scale(factor fraction) polynomial {
89 p := coefficients(len(a))
90 for i, c := range a { p[i] = c.mul(factor) }
91 return normalize(p)
92}
93func (a polynomial) divideScalar(divisor fraction) polynomial {
94 p := coefficients(len(a))
95 for i, c := range a { p[i] = c.div(divisor) }
96 return normalize(p)
97}
98func (a polynomial) mul(b polynomial) polynomial {
99 if len(a) == 0 || len(b) == 0 { return nil }
100 p := coefficients(len(a)+len(b)-1)
101 for i, x := range a {
102 for j, y := range b { p[i+j] = p[i+j].add(x.mul(y)) }
103 }
104 return normalize(p)
105}
106func (a polynomial) power(n int) polynomial {
107 if n != 0 && len(a) == 0 { return nil }
108 if len(a) == 1 && a[0].d == 1 && abs(a[0].n) == 1 {
109 if a[0].n < 0 && n%2 == 1 { return constant(whole(-1)) }
110 return constant(one)
111 }
112 p := constant(one)
113 for range n { p = p.mul(a) }
114 return p
115}
116func (a polynomial) divide(b polynomial) (polynomial, polynomial) {
117 q := coefficients(max(0, len(a)-len(b)+1))
118 r := slices.Clone(a)
119 for len(r) != 0 && len(r) >= len(b) {
120 factor, shift := r.leading().div(b.leading()), len(r)-len(b)
121 q[shift] = factor
122 for j, c := range b { r[shift+j] = r[shift+j].sub(factor.mul(c)) }
123 r = normalize(r)
124 }
125 return normalize(q), r
126}
127func (a polynomial) quotient(b polynomial) polynomial { q, _ := a.divide(b); return q }
128func (a polynomial) remainder(b polynomial) polynomial { _, r := a.divide(b); return r }
129func (a polynomial) monic() polynomial {
130 if len(a) == 0 { return nil }
131 return a.divideScalar(a.leading())
132}
133func (a polynomial) derivative() polynomial {
134 p := coefficients(max(0, len(a)-1))
135 for i := 1; i < len(a); i++ { p[i-1] = whole(i).mul(a[i]) }
136 return p
137}
138func (a polynomial) differentiate(n int) polynomial {
139 for i := 0; i < n && len(a) != 0; i++ { a = a.derivative() }
140 return a
141}
142func (a polynomial) integral() polynomial {
143 p := coefficients(len(a)+1)
144 for i, c := range a { p[i+1] = c.div(whole(i+1)) }
145 return normalize(p)
146}
147func (a polynomial) evaluate(x fraction) fraction {
148 result := zero
149 for i := len(a)-1; i >= 0; i-- { result = result.mul(x).add(a[i]) }
150 return result
151}
152func (a polynomial) compose(b polynomial) polynomial {
153 var p polynomial
154 for i := len(a)-1; i >= 0; i-- { p = p.mul(b).add(constant(a[i])) }
155 return p
156}
157func (a polynomial) gcd(b polynomial) polynomial {
158 for len(b) != 0 { a, b = b, a.remainder(b) }
159 return a.monic()
160}
161type bezout struct { g, s, t polynomial }
162func (a polynomial) extendedGCD(b polynomial) bezout {
163 r0, r1 := a, b
164 s0, s1, t0, t1 := constant(one), polynomial(nil), polynomial(nil), constant(one)
165 for len(r1) != 0 {
166 q, r := r0.divide(r1)
167 s, t := s0.sub(q.mul(s1)), t0.sub(q.mul(t1))
168 r0, r1 = r1, r
169 s0, s1 = s1, s
170 t0, t1 = t1, t
171 }
172 if len(r0) == 0 { return bezout{r0, s0, t0} }
173 return bezout{r0.monic(), s0.divideScalar(r0.leading()), t0.divideScalar(r0.leading())}
174}
175func (a polynomial) content() fraction {
176 denominator, numerator := int64(1), int64(0)
177 for _, c := range a {
178 denominator = denominator*c.d/gcd(denominator, c.d)
179 numerator = gcd(numerator, c.n)
180 }
181 if a.leading().n < 0 { numerator = -numerator }
182 return rational(numerator, denominator)
183}
184func (a polynomial) primitive() polynomial { return a.divideScalar(a.content()) }
185func (a polynomial) squareFree() polynomial {
186 if a.degree() < 1 { return a }
187 return a.quotient(a.gcd(a.derivative()))
188}
189func (a polynomial) isOne() bool { return len(a) == 1 && a[0] == one }
190func (a polynomial) checkProduct(b polynomial) error {
191 if len(a) != 0 && len(b) != 0 && a.degree()+b.degree() > 30 { return failure("DEGREE LIMIT") }
192 return nil
193}
194func (a polynomial) checkPower(n int) error {
195 if len(a) != 0 && a.degree()*n > 30 { return failure("DEGREE LIMIT") }
196 return nil
197}
198func (a polynomial) checkComposition(b polynomial) error {
199 if a.degree() >= 1 && b.degree() >= 1 && a.degree()*b.degree() > 30 { return failure("DEGREE LIMIT") }
200 return nil
201}
202func (a polynomial) nonzero() error { if len(a) == 0 { return failure("ZERO POLYNOMIAL") }; return nil }
203func (a polynomial) divisor() error { if len(a) == 0 { return failure("DIVISION BY ZERO") }; return nil }
204func (a polynomial) positiveDegree() error { if a.degree() == 0 { return failure("DEGREE TOO LOW") }; return nil }
205func (a polynomial) format(variable string) string {
206 var text strings.Builder
207 for i := len(a)-1; i >= 0; i-- {
208 c := a[i]
209 if c.n == 0 { continue }
210 if text.Len() == 0 { if c.n < 0 { text.WriteByte('-') } } else if c.n < 0 { text.WriteString(" - ") } else { text.WriteString(" + ") }
211 if i == 0 || c.abs() != one { text.WriteString(c.abs().String()) }
212 if i > 0 { text.WriteString(variable) }
213 if i > 1 { fmt.Fprintf(&text, "^%d", i) }
214 }
215 if text.Len() == 0 { return "0" }
216 return text.String()
217}

219type expressionKind uint8
220const (
221 literalExpression expressionKind = iota
222 variableExpression
223 referenceExpression
224 negativeExpression
225 binaryExpression
226 powerExpression
227 callExpression
228)
229type expression struct {
230 kind expressionKind
231 number int
232 name string
233 operation byte
234 left, right *expression
235}
236func (e *expression) evaluate(names map[string]polynomial) (polynomial, error) {
237 switch e.kind {
238 case literalExpression: return constant(whole(e.number)), nil
239 case variableExpression: return linear(zero, one), nil
240 case referenceExpression:
241 p, found := names[e.name]
242 if !found { return nil, fmt.Errorf("ERROR UNKNOWN %s", e.name) }
243 return p.differentiate(e.number), nil
244 }
245 a, err := e.left.evaluate(names)
246 if err != nil { return nil, err }
247 if e.kind == negativeExpression { return a.neg(), nil }
248 if e.kind == powerExpression {
249 if err := a.checkPower(e.number); err != nil { return nil, err }
250 return a.power(e.number), nil
251 }
252 b, err := e.right.evaluate(names)
253 if err != nil { return nil, err }
254 if e.kind == callExpression {
255 if err := a.checkComposition(b); err != nil { return nil, err }
256 return a.compose(b), nil
257 }
258 switch e.operation {
259 case '+': return a.add(b), nil
260 case '-': return a.sub(b), nil
261 case '*':
262 if err := a.checkProduct(b); err != nil { return nil, err }
263 return a.mul(b), nil
264 default:
265 if err := b.divisor(); err != nil { return nil, err }
266 if b.degree() >= 1 { return nil, failure("NONCONSTANT DIVISOR") }
267 return a.divideScalar(b[0]), nil
268 }
269}

271type tokenKind uint8
272const (numberToken tokenKind = iota; nameToken; symbolToken; endToken)
273type token struct { kind tokenKind; text string; position int }
274type parser struct { tokens []token; position int; err error }
275func digit(c byte) bool { return c >= '0' && c <= '9' }
276func upper(c byte) bool { return c >= 'A' && c <= 'Z' }
277func newParser(source string) *parser {
278 p := &parser{}
279 for i := 0; i < len(source); {
280 if source[i] == ' ' { i++; continue }
281 start, kind := i, symbolToken
282 i++
283 if digit(source[start]) {
284 kind = numberToken
285 for i < len(source) && digit(source[i]) { i++ }
286 } else if upper(source[start]) {
287 kind = nameToken
288 for i < len(source) && (upper(source[i]) || digit(source[i])) { i++ }
289 }
290 p.tokens = append(p.tokens, token{kind, source[start:i], start+1})
291 }
292 p.tokens = append(p.tokens, token{kind: endToken, position: len(source)+1})
293 return p
294}
295func (p *parser) current() token { return p.tokens[p.position] }
296func (p *parser) accept(symbol string) bool {
297 if p.current().kind == symbolToken && p.current().text == symbol { p.position++; return true }
298 return false
299}
300func (p *parser) syntax() *expression {
301 if p.err == nil {
302 if p.current().kind == endToken { p.err = failure("SYNTAX AT END") } else { p.err = fmt.Errorf("ERROR SYNTAX AT %d", p.current().position) }
303 }
304 return nil
305}
306func (p *parser) expect(symbol string) bool {
307 if p.err != nil { return false }
308 if !p.accept(symbol) { p.syntax(); return false }
309 return true
310}
311func (p *parser) parse() (*expression, error) {
312 result := p.expression()
313 if p.err == nil && p.current().kind != endToken { p.syntax() }
314 return result, p.err
315}
316func (p *parser) expression() *expression {
317 negative := p.accept("-")
318 result := p.term()
319 if p.err != nil { return nil }
320 if negative { result = &expression{kind: negativeExpression, left: result} }
321 for {
322 var operation byte
323 if p.accept("+") { operation = '+' } else if p.accept("-") { operation = '-' } else { break }
324 right := p.term()
325 if p.err != nil { return nil }
326 result = &expression{kind: binaryExpression, operation: operation, left: result, right: right}
327 }
328 return result
329}
330func (p *parser) term() *expression {
331 result := p.factor()
332 if p.err != nil { return nil }
333 for {
334 var operation byte
335 if p.accept("*") { operation = '*' } else if p.accept("/") { operation = '/' } else {
336 current := p.current()
337 if current.kind != nameToken && !(current.kind == symbolToken && (current.text == "x" || current.text == "(")) { break }
338 operation = '*'
339 }
340 right := p.factor()
341 if p.err != nil { return nil }
342 result = &expression{kind: binaryExpression, operation: operation, left: result, right: right}
343 }
344 return result
345}
346func (p *parser) factor() *expression {
347 result := p.primary()
348 if p.err != nil { return nil }
349 if p.accept("^") {
350 if p.current().kind != numberToken { return p.syntax() }
351 n := integer(p.current().text)
352 p.position++
353 result = &expression{kind: powerExpression, number: n, left: result}
354 }
355 return result
356}
357func (p *parser) primary() *expression {
358 current := p.current()
359 if current.kind == numberToken { p.position++; return &expression{kind: literalExpression, number: integer(current.text)} }
360 if p.accept("x") { return &expression{kind: variableExpression} }
361 if current.kind == nameToken {
362 p.position++
363 order := 0
364 for p.accept("'") { order++ }
365 reference := &expression{kind: referenceExpression, name: current.text, number: order}
366 if !p.accept("(") { return reference }
367 argument := p.expression()
368 if !p.expect(")") { return nil }
369 return &expression{kind: callExpression, left: reference, right: argument}
370 }
371 if p.accept("(") {
372 result := p.expression()
373 if !p.expect(")") { return nil }
374 return result
375 }
376 return p.syntax()
377}

379type root struct { value fraction; multiplicity int }
380type rootResult struct { content fraction; remaining polynomial; roots []root }
381func rationalRoots(a polynomial) rootResult {
382 content := a.content()
383 p := a.divideScalar(content)
384 zeros := 0
385 for p[0].n == 0 { p = p[1:]; zeros++ }
386 divisors := func(n int64) []int64 {
387 var result []int64
388 for i := int64(1); i <= n; i++ { if n%i == 0 { result = append(result, i) } }
389 return result
390 }
391 numerators, denominators := divisors(abs(p[0].n)), divisors(p.leading().n)
392 var candidates []fraction
393 for _, s := range numerators {
394 for _, t := range denominators {
395 if gcd(s, t) == 1 { candidates = append(candidates, fraction{-s, t}, fraction{s, t}) }
396 }
397 }
398 slices.SortFunc(candidates, fraction.compare)
399 var roots []root
400 for _, r := range candidates {
401 multiplicity := 0
402 for p.degree() >= 1 && p.evaluate(r).n == 0 {
403 p = p.quotient(linear(fraction{-r.n, 1}, fraction{r.d, 1}))
404 multiplicity++
405 }
406 if multiplicity != 0 { roots = append(roots, root{r, multiplicity}) }
407 }
408 if zeros != 0 { roots = append(roots, root{zero, zeros}) }
409 slices.SortFunc(roots, func(a, b root) int { return a.value.compare(b.value) })
410 return rootResult{content, p, roots}
411}
412type factor struct { value polynomial; multiplicity int }
413func squareFreeFactors(a polynomial) []factor {
414 if a.degree() < 1 { return nil }
415 w0 := a.monic()
416 c := w0.gcd(w0.derivative())
417 w := w0.quotient(c)
418 var factors []factor
419 for i := 1; w.degree() >= 1; i++ {
420 y := w.gcd(c)
421 f := w.quotient(y)
422 if f.degree() >= 1 { factors = append(factors, factor{f.primitive(), i}) }
423 c, w = c.quotient(y), y
424 }
425 return factors
426}

450func sturmSequence(a polynomial) []polynomial {
451 p := a.squareFree()
452 q := p.derivative()
453 sequence := []polynomial{p}
454 for len(q) != 0 {
455 sequence = append(sequence, q)
456 p, q = q, p.remainder(q).neg()
457 }
458 return sequence
459}
460func variations(sequence []polynomial, point bound) int {
461 previous, changes := 0, 0
462 for _, p := range sequence {
463 var sign int
464 if point.infinity == 0 { sign = p.evaluate(point.value).sign() } else { sign = p.leading().sign() }
465 if point.infinity < 0 && p.degree()%2 == 1 { sign = -sign }
466 if sign == 0 { continue }
467 if previous != 0 && previous != sign { changes++ }
468 previous = sign
469 }
470 return changes
471}
472func rootCount(sequence []polynomial, low, high bound) int {
473 left, right := variations(sequence, low), variations(sequence, high)
474 return left-right
475}
476type interval struct { low, high fraction }
477func examine(sequence []polynomial, low, high, width fraction, result *[]interval) {
478 count := rootCount(sequence, finite(low), finite(high))
479 if count >= 2 {
480 middle := low.add(high).div(whole(2))
481 examine(sequence, low, middle, width, result)
482 examine(sequence, middle, high, width, result)
483 } else if count == 1 {
484 for high.sub(low).compare(width) > 0 {
485 middle := low.add(high).div(whole(2))
486 if rootCount(sequence, finite(low), finite(middle)) >= 1 { high = middle } else { low = middle }
487 }
488 *result = append(*result, interval{low, high})
489 }
490}
491func isolate(a polynomial, precision int) []interval {
492 sequence := sturmSequence(a)
493 first := sequence[0]
494 maximum := zero
495 for _, c := range first[:len(first)-1] {
496 candidate := c.div(first.leading()).abs()
497 if candidate.compare(maximum) > 0 { maximum = candidate }
498 }
499 beta, width := one.add(maximum), rational(1, int64(1)<<precision)
500 var result []interval
501 examine(sequence, beta.neg(), beta, width, &result)
502 return result
503}
504func resultant(p, q polynomial) fraction {
505 r := one
506 for q.degree() >= 1 {
507 t := p.remainder(q)
508 if len(t) == 0 { return zero }
509 r = r.mul(q.leading().power(p.degree()-t.degree()))
510 if p.degree()*q.degree()%2 == 1 { r = r.neg() }
511 p, q = q, t
512 }
513 return r.mul(q[0].power(p.degree()))
514}
515func descartes(a polynomial, negative bool) int {
516 previous, count := 0, 0
517 for i := len(a)-1; i >= 0; i-- {
518 sign := a[i].sign()
519 if negative && i%2 == 1 { sign = -sign }
520 if sign == 0 { continue }
521 if previous != 0 && previous != sign { count++ }
522 previous = sign
523 }
524 return count
525}
526func interpolate(values []fraction) (polynomial, error) {
527 m := len(values)/2
528 x, d := make([]fraction, m), make([]fraction, m)
529 for i := range m { x[i], d[i] = values[2*i], values[2*i+1] }
530 for j := 1; j < m; j++ {
531 for i := range j { if x[j] == x[i] { return nil, fmt.Errorf("ERROR DUPLICATE X %s", x[j]) } }
532 }
533 for j := 1; j < m; j++ {
534 for i := m-1; i >= j; i-- { d[i] = d[i].sub(d[i-1]).div(x[i].sub(x[i-j])) }
535 }
536 var result polynomial
537 for i := m-1; i >= 0; i-- { result = result.mul(linear(x[i].neg(), one)).add(constant(d[i])) }
538 return result, nil
539}
540func special(kind string, n int) (polynomial, error) {
541 if !slices.Contains([]string{"chebyshev", "chebyshev2", "legendre", "hermite", "laguerre"}, kind) { return nil, fmt.Errorf("ERROR UNKNOWN KIND %s", kind) }
542 if n > 30 { return nil, failure("DEGREE LIMIT") }
543 a, b := constant(one), linear(zero, one)
544 if kind == "chebyshev2" || kind == "hermite" { b = linear(zero, whole(2)) }
545 if kind == "laguerre" { b = linear(one, whole(-1)) }
546 if n == 0 { return a, nil }
547 for j := 1; j < n; j++ {
548 factor, multiplier := linear(zero, whole(2)), 1
549 switch kind {
550 case "legendre": factor, multiplier = linear(zero, whole(2*j+1)), j
551 case "hermite": multiplier = 2*j
552 case "laguerre": factor, multiplier = linear(whole(2*j+1), whole(-1)), j
553 }
554 next := factor.mul(b).sub(a.scale(whole(multiplier)))
555 if kind == "legendre" || kind == "laguerre" { next = next.scale(rational(1, int64(j+1))) }
556 a, b = b, next
557 }
558 return b, nil
559}

561type named struct { name string; value polynomial }
562type calculator struct { names map[string]polynomial }
563func sourcePositions(command string) []int {
564 switch command {
565 case "ADD", "SUB", "MUL", "GCD", "LCM", "INVMOD", "COMPOSE": return []int{2, 3}
566 case "POW", "DERIV", "INTEG", "TANGENT": return []int{2}
567 case "MODPOW": return []int{2, 4}
568 case "DIV": return []int{3, 4}
569 case "XGCD": return []int{4, 5}
570 case "CRT": return []int{2, 3, 4, 5}
571 case "RESULTANT": return []int{1, 2}
572 case "LIST", "INTERP", "SPECIAL": return nil
573 default: return []int{1}
574 }
575}
576func (c *calculator) process(line string) ([]string, error) {
577 var output []string
578 var pending []named
579 deletion := ""
580 save := func(name string, value polynomial) {
581 pending = append(pending, named{name, value})
582 output = append(output, name+" = "+value.format("x"))
583 }
584 if strings.HasPrefix(line, "DEF ") {
585 parts := strings.SplitN(line, " ", 3)
586 tree, err := newParser(parts[2]).parse()
587 if err != nil { return nil, err }
588 value, err := tree.evaluate(c.names)
589 if err != nil { return nil, err }
590 save(parts[1], value)
591 } else {
592 w := strings.Fields(line)
593 command := w[0]
594 if command == "DIV" && w[1] == w[2] || command == "XGCD" && (w[1] == w[2] || w[1] == w[3] || w[2] == w[3]) { return nil, failure("SAME NAME") }
595 var sources []polynomial
596 for _, position := range sourcePositions(command) {
597 p, found := c.names[w[position]]
598 if !found { return nil, fmt.Errorf("ERROR UNKNOWN %s", w[position]) }
599 sources = append(sources, p)
600 }
601 start := len(w)
602 switch command { case "AREA", "TAYLOR", "EVAL", "INTERP": start = 2; case "TANGENT": start = 3 }
603 var values []fraction
604 for _, text := range w[start:] {
605 value, err := parseFraction(text)
606 if err != nil { return nil, err }
607 values = append(values, value)
608 }
609 var bounds [2]bound
610 if command == "STURM" {
611 for i := range 2 {
612 value, err := parseBound(w[2+i])
613 if err != nil { return nil, err }
614 bounds[i] = value
615 }
616 }
617 var a, b polynomial
618 if len(sources) > 0 { a = sources[0] }
619 if len(sources) > 1 { b = sources[1] }
620 switch command {
621 case "PRINT": output = append(output, w[1]+" = "+a.format("x"))
622 case "DELETE": deletion = w[1]; output = append(output, "deleted "+w[1])
623 case "LIST":
624 var names []string
625 for name := range c.names { names = append(names, name) }
626 slices.Sort(names)
627 for _, name := range names { output = append(output, name+" = "+c.names[name].format("x")) }
628 if len(names) == 0 { output = append(output, "no polynomials") }
629 case "DEGREE":
630 degree := "-inf"
631 if len(a) != 0 { degree = strconv.Itoa(a.degree()) }
632 output = append(output, fmt.Sprintf("deg(%s) = %s", w[1], degree))
633 case "ADD": save(w[1], a.add(b))
634 case "SUB": save(w[1], a.sub(b))
635 case "MUL":
636 if err := a.checkProduct(b); err != nil { return nil, err }
637 save(w[1], a.mul(b))
638 case "POW":
639 n := integer(w[3])
640 if err := a.checkPower(n); err != nil { return nil, err }
641 save(w[1], a.power(n))
642 case "DIV":
643 if err := b.divisor(); err != nil { return nil, err }
644 quotient, remainder := a.divide(b)
645 save(w[1], quotient); save(w[2], remainder)
646 case "GCD": save(w[1], a.gcd(b))
647 case "XGCD":
648 e := a.extendedGCD(b)
649 save(w[1], e.g); save(w[2], e.s); save(w[3], e.t)
650 case "LCM":
651 var value polynomial
652 if len(a) != 0 && len(b) != 0 {
653 g := a.gcd(b)
654 if a.degree()+b.degree()-g.degree() > 30 { return nil, failure("DEGREE LIMIT") }
655 value = a.quotient(g).mul(b).monic()
656 }
657 save(w[1], value)
658 case "INVMOD":
659 if err := b.divisor(); err != nil { return nil, err }
660 if err := b.positiveDegree(); err != nil { return nil, err }
661 e := a.extendedGCD(b)
662 if !e.g.isOne() { return nil, failure("NOT COPRIME") }
663 save(w[1], e.s.remainder(b))
664 case "CRT":
665 a2, m2 := sources[2], sources[3]
666 if err := b.divisor(); err != nil { return nil, err }
667 if err := m2.divisor(); err != nil { return nil, err }
668 if b.degree()+m2.degree() > 30 { return nil, failure("DEGREE LIMIT") }
669 e := b.extendedGCD(m2)
670 if !e.g.isOne() { return nil, failure("NOT COPRIME") }
671 r1, r2 := a.remainder(b), a2.remainder(m2)
672 u := r2.sub(r1).mul(e.s).remainder(m2)
673 save(w[1], r1.add(b.mul(u)))
674 case "MODPOW":
675 if err := b.divisor(); err != nil { return nil, err }
676 if 2*b.degree()-2 > 30 { return nil, failure("DEGREE LIMIT") }
677 x, y := constant(one).remainder(b), a.remainder(b)
678 for exponent := integer(w[3]); exponent != 0; {
679 if exponent%2 == 1 { x = x.mul(y).remainder(b) }
680 exponent /= 2
681 if exponent != 0 { y = y.mul(y).remainder(b) }
682 }
683 save(w[1], x)
684 case "DERIV": save(w[1], a.differentiate(integer(w[3])))
685 case "INTEG":
686 if len(a) != 0 && a.degree()+1 > 30 { return nil, failure("DEGREE LIMIT") }
687 save(w[1], a.integral())
688 case "AREA":
689 integral := a.integral()
690 high, low := integral.evaluate(values[1]), integral.evaluate(values[0])
691 output = append(output, fmt.Sprintf("integral(%s, %s, %s) = %s", w[1], values[0], values[1], high.sub(low)))
692 case "TANGENT":
693 point := values[0]
694 y, slope := a.evaluate(point), a.derivative().evaluate(point)
695 save(w[1], linear(y.sub(slope.mul(point)), slope))
696 case "TAYLOR":
697 point := values[0]
698 shifted := a.compose(linear(point, one))
699 variable := "x"
700 if point.n > 0 { variable = fmt.Sprintf("(x - %s)", point) }
701 if point.n < 0 { variable = fmt.Sprintf("(x + %s)", point.neg()) }
702 output = append(output, fmt.Sprintf("taylor(%s, %s) = %s", w[1], point, shifted.format(variable)))
703 case "COMPOSE":
704 if err := a.checkComposition(b); err != nil { return nil, err }
705 save(w[1], a.compose(b))
706 case "EVAL":
707 for _, value := range values { output = append(output, fmt.Sprintf("%s(%s) = %s", w[1], value, a.evaluate(value))) }
708 case "ROOTS", "FACTOR":
709 if err := a.nonzero(); err != nil { return nil, err }
710 roots := rationalRoots(a)
711 if command == "ROOTS" {
712 var entries []string
713 for _, root := range roots.roots { entries = append(entries, fmt.Sprintf("%s (%d)", root.value, root.multiplicity)) }
714 text := "none"
715 if len(entries) != 0 { text = strings.Join(entries, ", ") }
716 output = append(output, fmt.Sprintf("roots(%s) = %s", w[1], text))
717 } else {
718 var text strings.Builder
719 fmt.Fprintf(&text, "factor(%s) = %s", w[1], roots.content)
720 for _, root := range roots.roots {
721 r := root.value
722 writeFactor(&text, linear(fraction{-r.n, 1}, fraction{r.d, 1}), root.multiplicity, r.n == 0)
723 }
724 if roots.remaining.degree() >= 1 { writeFactor(&text, roots.remaining, 1, false) }
725 output = append(output, text.String())
726 }
727 case "SQFREE":
728 if err := a.nonzero(); err != nil { return nil, err }
729 var text strings.Builder
730 fmt.Fprintf(&text, "sqfree(%s) = %s", w[1], a.content())
731 for _, factor := range squareFreeFactors(a) {
732 p := factor.value
733 writeFactor(&text, p, factor.multiplicity, len(p) == 2 && p[0] == zero && p[1] == one)
734 }
735 output = append(output, text.String())
736 case "DESCARTES":
737 if err := a.nonzero(); err != nil { return nil, err }
738 output = append(output, fmt.Sprintf("descartes(%s) = positive %d, negative %d", w[1], descartes(a, false), descartes(a, true)))
739 case "STURM":
740 if err := a.nonzero(); err != nil { return nil, err }
741 if bounds[0].compare(bounds[1]) >= 0 { return nil, failure("BAD INTERVAL") }
742 sequence := sturmSequence(a)
743 output = append(output, fmt.Sprintf("sturm(%s, %s, %s) = %d", w[1], bounds[0], bounds[1], rootCount(sequence, bounds[0], bounds[1])))
744 case "ISOLATE":
745 if err := a.nonzero(); err != nil { return nil, err }
746 n := integer(w[2])
747 var entries []string
748 for _, interval := range isolate(a, n) { entries = append(entries, fmt.Sprintf("(%s, %s]", interval.low, interval.high)) }
749 text := "none"
750 if len(entries) != 0 { text = strings.Join(entries, ", ") }
751 output = append(output, fmt.Sprintf("isolate(%s, %d) = %s", w[1], n, text))
752 case "RESULTANT":
753 if err := a.nonzero(); err != nil { return nil, err }
754 if err := b.nonzero(); err != nil { return nil, err }
755 output = append(output, fmt.Sprintf("res(%s, %s) = %s", w[1], w[2], resultant(a, b)))
756 case "DISCRIM", "POWERSUM":
757 if err := a.nonzero(); err != nil { return nil, err }
758 if err := a.positiveDegree(); err != nil { return nil, err }
759 degree := a.degree()
760 if command == "DISCRIM" {
761 d := resultant(a, a.derivative()).div(a.leading())
762 if degree*(degree-1)/2%2 == 1 { d = d.neg() }
763 output = append(output, fmt.Sprintf("disc(%s) = %s", w[1], d))
764 } else {
765 n, monic := integer(w[2]), a.monic()
766 sums := coefficients(n+1)
767 var entries []string
768 for j := 1; j <= n; j++ {
769 value := zero
770 if j <= degree { value = whole(j).mul(monic[degree-j]) }
771 for i := 1; i <= min(j-1, degree); i++ { value = value.add(monic[degree-i].mul(sums[j-i])) }
772 sums[j] = value.neg()
773 entries = append(entries, sums[j].String())
774 }
775 output = append(output, fmt.Sprintf("powersum(%s) = %s", w[1], strings.Join(entries, ", ")))
776 }
777 case "INTERP":
778 value, err := interpolate(values)
779 if err != nil { return nil, err }
780 save(w[1], value)
781 case "SPECIAL":
782 value, err := special(w[2], integer(w[3]))
783 if err != nil { return nil, err }
784 save(w[1], value)
785 }
786 }
787 if deletion != "" { delete(c.names, deletion) }
788 for _, item := range pending { c.names[item.name] = item.value }
789 return output, nil
790}

792func main() {
793 input := bufio.NewScanner(os.Stdin)
794 if !input.Scan() { return }
795 count := integer(input.Text())
796 calculator := calculator{names: make(map[string]polynomial)}
797 output := bufio.NewWriter(os.Stdout)
798 defer output.Flush()
799 for i := 0; i < count && input.Scan(); i++ {
800 lines, err := calculator.process(input.Text())
801 if err != nil { fmt.Fprintln(output, err); continue }
802 for _, line := range lines { fmt.Fprintln(output, line) }
803 }
804}
