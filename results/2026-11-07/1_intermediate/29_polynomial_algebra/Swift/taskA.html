11func integerGcd(_ first: Int64, _ second: Int64) -> Int64 {
12 var a = abs(first), b = abs(second)
13 while b != 0 {
14 (a, b) = (b, a % b)
15 }
16 return a
17}

19struct Fraction: Comparable, CustomStringConvertible {
20 let n: Int64
21 let d: Int64
22
23 init(_ numerator: Int64 = 0, _ denominator: Int64 = 1) {
24 let sign: Int64 = denominator < 0 ? -1 : 1
25 let divisor = integerGcd(numerator, denominator)
26 n = sign * numerator / divisor
27 d = sign * denominator / divisor
28 }
29
30 static let zero = Fraction()
31 static let one = Fraction(1)
32 static let two = Fraction(2)
33
34 static func parse(_ word: String) throws -> Fraction {
35 let parts = word.split(separator: "/")
36 let denominator = parts.count == 1 ? 1 : Int64(parts[1])!
37 try check(denominator == 0, "ZERO DENOMINATOR")
38 return Fraction(Int64(parts[0])!, denominator)
39 }
40
41 static func + (a: Fraction, b: Fraction) -> Fraction {
42 Fraction(a.n * b.d + b.n * a.d, a.d * b.d)
43 }
44 static func - (a: Fraction, b: Fraction) -> Fraction {
45 Fraction(a.n * b.d - b.n * a.d, a.d * b.d)
46 }
47 static func * (a: Fraction, b: Fraction) -> Fraction {
48 Fraction(a.n * b.n, a.d * b.d)
49 }
50 static func / (a: Fraction, b: Fraction) -> Fraction {
51 Fraction(a.n * b.d, a.d * b.n)
52 }
53 static prefix func - (value: Fraction) -> Fraction {
54 Fraction(-value.n, value.d)
55 }
56 static func < (a: Fraction, b: Fraction) -> Bool {
57 a.n * b.d < b.n * a.d
58 }
59 static func == (a: Fraction, b: Fraction) -> Bool {
60 a.n * b.d == b.n * a.d
61 }
62 var magnitude: Fraction {
63 n < 0 ? -self : self
64 }
65 var sign: Int {
66 n < 0 ? -1 : n > 0 ? 1 : 0
67 }
68 var description: String {
69 d == 1 ? String(n) : "\(n)/\(d)"
70 }
71
72 func power(_ exponent: Int) -> Fraction {
73 var result = Fraction.one
74 for _ in 0..<exponent {
75 result = result * self
76 }
77 return result
78 }
79}

81struct Polynomial {
82 let coefficients: [Fraction]
83
84 init(_ values: [Fraction] = []) {
85 var trimmed = values
86 while trimmed.last?.n == 0 {
87 trimmed.removeLast()
88 }
89 coefficients = trimmed
90 }
91
92 static let unit = Polynomial([.one])
93 static let x = Polynomial([.zero, .one])
94 static func constant(_ value: Fraction) -> Polynomial {
95 Polynomial([value])
96 }
97 var degree: Int {
98 coefficients.count - 1
99 }
100 var isZero: Bool {
101 coefficients.isEmpty
102 }
103 var leading: Fraction {
104 coefficients.last!
105 }
106 var isOne: Bool {
107 degree == 0 && leading == .one
108 }
109
110 func coefficient(_ index: Int) -> Fraction {
111 index >= 0 && index < coefficients.count ? coefficients[index] : .zero
112 }
113
114 static func + (a: Polynomial, b: Polynomial) -> Polynomial {
115 Polynomial((0..<max(a.coefficients.count, b.coefficients.count)).map { a.coefficient($0) + b.coefficient($0) })
116 }
117
118 static func - (a: Polynomial, b: Polynomial) -> Polynomial {
119 Polynomial((0..<max(a.coefficients.count, b.coefficients.count)).map { a.coefficient($0) - b.coefficient($0) })
120 }
121
122 static prefix func - (value: Polynomial) -> Polynomial {
123 Polynomial(value.coefficients.map { -$0 })
124 }
125 func scale(_ scalar: Fraction) -> Polynomial {
126 Polynomial(coefficients.map { $0 * scalar })
127 }
128 func divideScalar(_ scalar: Fraction) -> Polynomial {
129 Polynomial(coefficients.map { $0 / scalar })
130 }
131
132 static func * (a: Polynomial, b: Polynomial) -> Polynomial {
133 if a.isZero || b.isZero {
134 return Polynomial()
135 }
136 var result = [Fraction](repeating: .zero, count: a.degree + b.degree + 1)
137 for i in a.coefficients.indices {
138 for j in b.coefficients.indices {
139 result[i + j] = result[i + j] + a.coefficients[i] * b.coefficients[j]
140 }
141 }
142 return Polynomial(result)
143 }
144
145 func power(_ exponent: Int) -> Polynomial {
146 if exponent == 0 {
147 return .unit
148 }
149 if isZero {
150 return self
151 }
152 if degree == 0 && leading.d == 1 && abs(leading.n) == 1 {
153 return .constant(leading.n < 0 && exponent % 2 == 1 ? Fraction(-1) : .one)
154 }
155 var result = Polynomial.unit
156 for _ in 0..<exponent {
157 result = result * self
158 }
159 return result
160 }
161
162 func divide(_ divisor: Polynomial) -> (quotient: Polynomial, remainder: Polynomial) {
163 var remainder = coefficients
164 var quotient = [Fraction](repeating: .zero, count: max(0, degree - divisor.degree + 1))
165 while !remainder.isEmpty && remainder.count >= divisor.coefficients.count {
166 let value = remainder.last! / divisor.leading
167 let shift = remainder.count - divisor.coefficients.count
168 quotient[shift] = value
169 for j in divisor.coefficients
170 .indices
171 {
172 remainder[shift + j] = remainder[shift + j] - value * divisor.coefficients[j]
173 }
174 while remainder.last?.n == 0 {
175 remainder.removeLast()
176 }
177 }
178 return (Polynomial(quotient), Polynomial(remainder))
179 }
180
181 func mod(_ divisor: Polynomial) -> Polynomial {
182 divide(divisor).remainder
183 }
184 func monic() -> Polynomial {
185 isZero ? self : divideScalar(leading)
186 }
187
188 func derivative(_ count: Int = 1) -> Polynomial {
189 var result = self
190 for _ in 0..<count {
191 if result.isZero {
192 break
193 }
194 result = Polynomial(result.coefficients.dropFirst().enumerated().map { index, value in
195 Fraction(Int64(index + 1)) * value
196 })
197 }
198 return result
199 }
200
201 func integral() -> Polynomial {
202 Polynomial([.zero] + coefficients.enumerated().map { index, value in value / Fraction(Int64(index + 1)) })
203 }
204
205 func evaluate(_ value: Fraction) -> Fraction {
206 var result = Fraction.zero
207 for coefficient in coefficients.reversed() {
208 result = result * value + coefficient
209 }
210 return result
211 }
212
213 func compose(_ other: Polynomial) -> Polynomial {
214 var result = Polynomial()
215 for coefficient in coefficients.reversed() {
216 result = result * other + .constant(coefficient)
217 }
218 return result
219 }
220
221 func content() -> Fraction {
222 var multiple: Int64 = 1, divisor: Int64 = 0
223 for value in coefficients {
224 multiple = multiple * value.d / integerGcd(multiple, value.d)
225 divisor = integerGcd(divisor, value.n)
226 }
227 return Fraction(leading.n < 0 ? -divisor : divisor, multiple)
228 }
229
230 func primitive() -> Polynomial {
231 divideScalar(content())
232 }
233
234 func notation(_ variable: String = "x") -> String {
235 var result = ""
236 for i in coefficients.indices.reversed() {
237 let value = coefficients[i]
238 if value.n == 0 {
239 continue
240 }
241 result += result.isEmpty ? (value.n < 0 ? "-" : "") : (value.n < 0 ? " - " : " + ")
242 if i == 0 || value.magnitude != .one {
243 result += value.magnitude.description
244 }
245 if i > 0 {
246 result += variable + (i == 1 ? "" : "^\(i)")
247 }
248 }
249 return result.isEmpty ? "0" : result
250 }
251}

269 (a, b) = (b, a.mod(b))

274func extendedGcd(_ first: Polynomial, _ second: Polynomial) -> (Polynomial, Polynomial, Polynomial) {
275 var a = first, b = second
276 var s0 = Polynomial.unit, s1 = Polynomial(), t0 = Polynomial(), t1 = Polynomial.unit
277 while !b.isZero {
278 let (quotient, remainder) = a.divide(b)
279 (a, b) = (b, remainder)
280 (s0, s1) = (s1, s0 - quotient * s1)
281 (t0, t1) = (t1, t0 - quotient * t1)
282 }
283 return a.isZero ? (a, s0, t0) : (a.monic(), s0.divideScalar(a.leading), t0.divideScalar(a.leading))
284}

286func divisors(_ number: Int64) -> [Int64] {
287 var result: [Int64] = [], divisor: Int64 = 1
288 while divisor * divisor <= number {
289 if number % divisor == 0 {
290 result.append(divisor)
291 if divisor * divisor != number {
292 result.append(number / divisor)
293 }
294 }
295 divisor += 1
296 }
297 return result
298}

301 let value: Fraction
302 let multiplicity: Int

305func rationalRoots(_ a: Polynomial) -> (roots: [Root], remaining: Polynomial) {
306 var remaining = a.primitive(), zeros = 0
307 while remaining.coefficient(0).n == 0 {
308 remaining = Polynomial(Array(remaining.coefficients.dropFirst()))
309 zeros += 1
310 }
311 var candidates: [Fraction] = []
312 let numerators = divisors(abs(remaining.coefficient(0).n)), denominators = divisors(remaining.leading.n)
313 for numerator in numerators {
314 for denominator in denominators where integerGcd(numerator, denominator) == 1 {
315 let root = Fraction(numerator, denominator)
316 candidates.append(root)
317 candidates.append(-root)
318 }
319 }
320 candidates.sort()
321 var roots: [Root] = []
322 for root in candidates {
323 var multiplicity = 0
324 let factor = linear(Fraction(-root.n), Fraction(root.d))
325 while remaining.degree >= 1 && remaining.evaluate(root).n == 0 {
326 remaining = remaining.divide(factor).quotient
327 multiplicity += 1
328 }
329 if multiplicity > 0 {
330 roots.append(Root(value: root, multiplicity: multiplicity))
331 }
332 }
333 if zeros > 0 {
334 roots.append(Root(value: .zero, multiplicity: zeros))
335 }
336 roots.sort { $0.value < $1.value }
337 return (roots, remaining)
338}

341 a.degree < 1 ? a : a.divide(polynomialGcd(a, a.derivative())).quotient

344func sturmSequence(_ a: Polynomial) -> [Polynomial] {
345 var previous = squareFreePart(a)
346 var current = previous.derivative()
347 var sequence = [previous]
348 while !current.isZero {
349 sequence.append(current)
350 (previous, current) = (current, -previous.mod(current))
351 }
352 return sequence
353}

355enum Endpoint: Comparable, CustomStringConvertible {
356 case negativeInfinity
357 case finite(Fraction)
358 case positiveInfinity
359
360 static func parse(_ word: String) throws -> Endpoint {
361 if word == "-inf" {
362 return .negativeInfinity
363 }
364 if word == "inf" {
365 return .positiveInfinity
366 }
367 return .finite(try Fraction.parse(word))
368 }
369
370 static func < (a: Endpoint, b: Endpoint) -> Bool {
371 switch (a, b) {
372 case (.negativeInfinity, .negativeInfinity), (.positiveInfinity, .positiveInfinity):
373 return false
374 case (.negativeInfinity, _), (_, .positiveInfinity):
375 return true
376 case (.positiveInfinity, _), (_, .negativeInfinity):
377 return false
378 case let (.finite(left), .finite(right)):
379 return left < right
380 }
381 }
382
383 var description: String {
384 switch self {
385 case .negativeInfinity:
386 return "-inf"
387 case .positiveInfinity:
388 return "inf"
389 case .finite(let value):
390 return value.description
391 }
392 }
393}

419func isolate(_ a: Polynomial, _ precision: Int) -> [(Fraction, Fraction)] {
420 let sequence = sturmSequence(a)
421 let first = sequence[0]
422 var maximum = Fraction.zero
423 for i in 0..<first.degree {
424 maximum = max(maximum, (first.coefficients[i] / first.leading).magnitude)
425 }
426 let bound = Fraction.one + maximum, width = Fraction(1, Int64(1) << precision)
427 var intervals: [(Fraction, Fraction)] = []
428 func count(_ left: Fraction, _ right: Fraction) -> Int {
429 variations(sequence, .finite(left)) - variations(sequence, .finite(right))
430 }
431 func examine(_ lower: Fraction, _ upper: Fraction) {
432 var left = lower, right = upper
433 let roots = count(left, right)
434 if roots >= 2 {
435 let middle = (left + right) / .two
436 examine(left, middle)
437 examine(middle, right)
438 } else if roots == 1 {
439 while right - left > width {
440 let middle = (left + right) / .two
441 if count(left, middle) >= 1 {
442 right = middle
443 } else {
444 left = middle
445 }
446 }
447 intervals.append((left, right))
448 }
449 }
450 examine(-bound, bound)
451 return intervals
452}

454func resultant(_ first: Polynomial, _ second: Polynomial) -> Fraction {
455 var a = first, b = second, result = Fraction.one
456 while b.degree >= 1 {
457 let remainder = a.mod(b)
458 if remainder.isZero {
459 return .zero
460 }
461 result = result * b.leading.power(a.degree - remainder.degree)
462 if a.degree * b.degree % 2 == 1 {
463 result = -result
464 }
465 (a, b) = (b, remainder)
466 }
467 return result * b.coefficient(0).power(a.degree)
468}

470func interpolate(_ values: [Fraction]) throws -> Polynomial {
471 let xs = stride(from: 0, to: values.count, by: 2).map { values[$0] }
472 var differences = stride(from: 1, to: values.count, by: 2).map { values[$0] }
473 for j in 1..<xs.count {
474 try check(xs[..<j].contains(xs[j]), "DUPLICATE X \(xs[j])")
475 }
476 for j in 1..<xs.count {
477 for i in stride(from: xs.count - 1, through: j, by: -1) {
478 differences[i] = (differences[i] - differences[i - 1]) / (xs[i] - xs[i - j])
479 }
480 }
481 var result = Polynomial()
482 for i in xs.indices.reversed() {
483 result = result * linear(-xs[i]) + .constant(differences[i])
484 }
485 return result
486}

488func special(_ kind: String, _ order: Int) throws -> Polynomial {
489 let kinds: Set<String> = ["chebyshev", "chebyshev2", "legendre", "hermite", "laguerre"]
490 try check(!kinds.contains(kind), "UNKNOWN KIND \(kind)")
491 try check(order > 30, "DEGREE LIMIT")
492 if order == 0 {
493 return .unit
494 }
495 var previous = Polynomial.unit
496 var current: Polynomial = kind == "laguerre" ? linear(.one, Fraction(-1))
497 : kind == "chebyshev2" || kind == "hermite" ? Polynomial.x.scale(.two) : .x
498 for j in 1..<order {
499 let next: Polynomial
500 if kind == "chebyshev" || kind == "chebyshev2" {
501 next = Polynomial.x.scale(.two) * current - previous
502 } else if kind == "hermite" {
503 next = Polynomial.x.scale(.two) * current - previous.scale(Fraction(Int64(2 * j)))
504 } else {
505 let multiplier = kind == "legendre" ? Polynomial.x.scale(Fraction(Int64(2 * j + 1))) : linear(
506 Fraction(Int64(2 * j + 1)),
507 Fraction(-1)
508 )
509 next = (multiplier * current - previous.scale(Fraction(Int64(j)))).scale(Fraction(1, Int64(j + 1)))
510 }
511 (previous, current) = (current, next)
512 }
513 return current
514}

526enum TokenKind: Equatable { case number, name, symbol(Character), end }
527struct Token {
528 let kind: TokenKind
529 let word: String
530 let position: Int

533final class ExpressionParser {
534 private let tokens: [Token]
535 private var position = 0
536 private var token: Token {
537 tokens[position]
538 }
539
540 init(_ source: String) {
541 let characters = Array(source)
542 func digit(_ ch: Character) -> Bool {
543 ch >= "0" && ch <= "9"
544 }
545 func uppercase(_ ch: Character) -> Bool {
546 ch >= "A" && ch <= "Z"
547 }
548 var result: [Token] = [], i = 0
549 while i < characters.count {
550 if characters[i] == " " {
551 i += 1
552 continue
553 }
554 let start = i, ch = characters[i]
555 i += 1
556 let kind: TokenKind
557 if digit(ch) {
558 kind = .number
559 while i < characters.count && digit(characters[i]) {
560 i += 1
561 }
562 } else if uppercase(ch) {
563 kind = .name
564 while i < characters.count && (uppercase(characters[i]) || digit(characters[i])) {
565 i += 1
566 }
567 } else {
568 kind = .symbol(ch)
569 }
570 result.append(Token(kind: kind, word: String(characters[start..<i]), position: start + 1))
571 }
572 result.append(Token(kind: .end, word: "", position: characters.count + 1))
573 tokens = result
574 }
575
576 private func take(_ kind: TokenKind) -> Bool {
577 if token.kind != kind {
578 return false
579 }
580 position += 1
581 return true
582 }
583 private func take(_ symbol: Character) -> Bool {
584 take(.symbol(symbol))
585 }
586 private func syntaxError() -> CalculatorError {
587 CalculatorError(message: "SYNTAX AT " + (token.kind == .end ? "END" : String(token.position)))
588 }
589 private func expect(_ symbol: Character) throws {
590 if !take(symbol) {
591 throw syntaxError()
592 }
593 }
594
595 func parse() throws -> Expression {
596 let result = try expression()
597 if token.kind != .end {
598 throw syntaxError()
599 }
600 return result
601 }
602
603 private func expression() throws -> Expression {
604 let negative = take("-")
605 var result = try term()
606 if negative {
607 result = .negate(result)
608 }
609 while token.kind == .symbol("+") || token.kind == .symbol("-") {
610 let operation: BinaryOperation = token.kind == .symbol("+") ? .add : .subtract
611 position += 1
612 result = .binary(operation, result, try term())
613 }
614 return result
615 }
616
617 private func term() throws -> Expression {
618 var result = try factor()
619 let starts: [TokenKind] = [.symbol("*"), .symbol("/"), .symbol("x"), .name, .symbol("(")]
620 while starts.contains(token.kind) {
621 var operation = BinaryOperation.multiply
622 if token.kind == .symbol("*") || token.kind == .symbol("/") {
623 if token.kind == .symbol("/") {
624 operation = .divide
625 }
626 position += 1
627 }
628 result = .binary(operation, result, try factor())
629 }
630 return result
631 }
632
633 private func factor() throws -> Expression {
634 var result = try primary()
635 if take("^") {
636 if token.kind != .number {
637 throw syntaxError()
638 }
639 let exponent = Int(token.word)!
640 position += 1
641 result = .power(result, exponent)
642 }
643 return result
644 }
645
646 private func primary() throws -> Expression {
647 let current = token
648 if take(.number) {
649 return .number(Fraction(Int64(current.word)!))
650 }
651 if take("x") {
652 return .variable
653 }
654 if take(.name) {
655 var derivatives = 0
656 while take("'") {
657 derivatives += 1
658 }
659 let reference = Expression.reference(current.word, derivatives)
660 if !take("(") {
661 return reference
662 }
663 let argument = try expression()
664 try expect(")")
665 return .binary(.compose, reference, argument)
666 }
667 if take("(") {
668 let result = try expression()
669 try expect(")")
670 return result
671 }
672 throw syntaxError()
673 }
674}

685final class Calculator {
686 private var polynomials: [String: Polynomial] = [:]
687
688 private func lookup(_ name: String) throws -> Polynomial {
689 guard let value = polynomials[name] else {
690 throw CalculatorError(message: "UNKNOWN \(name)")
691 }
692 return value
693 }
694
695 private func evaluate(_ expression: Expression) throws -> Polynomial {
696 switch expression {
697 case .number(let value):
698 return .constant(value)
699 case .variable:
700 return .x
701 case let .reference(name, count):
702 return try lookup(name).derivative(count)
703 case .negate(let operand):
704 return -(try evaluate(operand))
705 case let .power(operand, exponent):
706 let value = try evaluate(operand)
707 try check(!value.isZero && value.degree * exponent > 30, "DEGREE LIMIT")
708 return value.power(exponent)
709 case let .binary(operation, left, right):
710 let a = try evaluate(left)
711 let b = try evaluate(right)
712 switch operation {
713 case .add:
714 return a + b
715 case .subtract:
716 return a - b
717 case .multiply:
718 try productLimit(a, b)
719 return a * b
720 case .compose:
721 try compositionLimit(a, b)
722 return a.compose(b)
723 case .divide:
724 try check(b.isZero, "DIVISION BY ZERO")
725 try check(b.degree > 0, "NONCONSTANT DIVISOR")
726 return a.divideScalar(b.leading)
727 }
728 }
729 }

731 func execute(_ source: String) throws -> [String] {
732 let words = source.split(separator: " ").map(String.init)
733 let command = words[0], name = words.count > 1 ? words[1] : ""
734 var pending: [(String, Polynomial)] = [], output: [String] = []
735 func store(_ target: String, _ value: Polynomial) {
736 pending.append((target, value))
737 output.append("\(target) = \(value.notation())")
738 }
739 if command == "DEF" {
740 let syntax = try ExpressionParser(String(source.dropFirst(name.count + 5))).parse()
741 store(name, try evaluate(syntax))
742 } else {
743 if command == "DIV" {
744 try check(words[1] == words[2], "SAME NAME")
745 }
746 if command == "XGCD" {
747 try check(Set(words[1..<4]).count < 3, "SAME NAME")
748 }
749 let operands = try (references[command] ?? []).map { try lookup(words[$0]) }
750 let a = operands.first ?? Polynomial(), b = operands.count > 1 ? operands[1] : Polynomial()
751 var values: [Fraction] = [], endpoints: [Endpoint] = []
752 if command == "STURM" {
753 endpoints = try words[2..<4].map { try Endpoint.parse($0) }
754 } else {
755 var start = words.count, end = words.count
756 if command == "AREA" {
757 start = 2
758 end = 4
759 } else if command == "TANGENT" {
760 start = 3
761 end = 4
762 } else if command == "TAYLOR" {
763 start = 2
764 end = 3
765 } else if command == "EVAL" || command == "INTERP" {
766 start = 2
767 }
768 values = try words[start..<end].map { try Fraction.parse($0) }
769 }
770 switch command {
771 case "PRINT":
772 output.append("\(name) = \(a.notation())")
773 case "DELETE":
774 polynomials.removeValue(forKey: name)
775 output.append("deleted \(name)")
776 case "LIST":
777 for key in polynomials.keys.sorted() {
778 output.append("\(key) = \(polynomials[key]!.notation())")
779 }
780 if output.isEmpty {
781 output.append("no polynomials")
782 }
783 case "DEGREE":
784 output.append("deg(\(name)) = " + (a.isZero ? "-inf" : String(a.degree)))
785 case "ADD":
786 store(name, a + b)
787 case "SUB":
788 store(name, a - b)
789 case "MUL":
790 try productLimit(a, b)
791 store(name, a * b)
792 case "POW":
793 let exponent = Int(words[3])!
794 try check(!a.isZero && a.degree * exponent > 30, "DEGREE LIMIT")
795 store(name, a.power(exponent))
796 case "DIV":
797 try check(b.isZero, "DIVISION BY ZERO")
798 let (quotient, remainder) = a.divide(b)
799 store(name, quotient)
800 store(words[2], remainder)
801 case "GCD":
802 store(name, polynomialGcd(a, b))
803 case "XGCD":
804 let (common, s, t) = extendedGcd(a, b)
805 store(name, common)
806 store(words[2], s)
807 store(words[3], t)
808 case "LCM":
809 if a.isZero || b.isZero {
810 store(name, Polynomial())
811 } else {
812 let common = polynomialGcd(a, b)
813 try check(a.degree + b.degree - common.degree > 30, "DEGREE LIMIT")
814 store(name, (a.divide(common).quotient * b).monic())
815 }
816 case "INVMOD":
817 try check(b.isZero, "DIVISION BY ZERO")
818 try check(b.degree == 0, "DEGREE TOO LOW")
819 let (common, inverse, _) = extendedGcd(a, b)
820 try check(!common.isOne, "NOT COPRIME")
821 store(name, inverse.mod(b))
822 case "CRT":
823 let c = operands[2], d = operands[3]
824 try check(b.isZero || d.isZero, "DIVISION BY ZERO")
825 try check(b.degree + d.degree > 30, "DEGREE LIMIT")
826 let (common, inverse, _) = extendedGcd(b, d)
827 try check(!common.isOne, "NOT COPRIME")
828 let first = a.mod(b), second = c.mod(d)
829 let correction = ((second - first) * inverse).mod(d)
830 store(name, first + b * correction)
831 case "MODPOW":
832 try check(b.isZero, "DIVISION BY ZERO")
833 try check(2 * b.degree - 2 > 30, "DEGREE LIMIT")
834 var result = Polynomial.unit.mod(b), base = a.mod(b), exponent = Int(words[3])!
835 while exponent > 0 {
836 if exponent % 2 == 1 {
837 result = (result * base).mod(b)
838 }
839 exponent /= 2
840 if exponent > 0 {
841 base = (base * base).mod(b)
842 }
843 }
844 store(name, result)
845 case "DERIV":
846 store(name, a.derivative(Int(words[3])!))
847 case "INTEG":
848 try check(!a.isZero && a.degree + 1 > 30, "DEGREE LIMIT")
849 store(name, a.integral())
850 case "AREA":
851 let integral = a.integral(), u = values[0], v = values[1]
852 output.append("integral(\(name), \(u), \(v)) = \(integral.evaluate(v) - integral.evaluate(u))")
853 case "TANGENT":
854 let point = values[0]
855 let value = a.evaluate(point)
856 let slope = a.derivative().evaluate(point)
857 store(name, linear(value - slope * point, slope))
858 case "TAYLOR":
859 let point = values[0]
860 let symbol = point.n == 0 ? "x" : point.n > 0 ? "(x - \(point))" : "(x + \(-point))"
861 output.append("taylor(\(name), \(point)) = \(a.compose(linear(point)).notation(symbol))")
862 case "COMPOSE":
863 try compositionLimit(a, b)
864 store(name, a.compose(b))
865 case "EVAL":
866 for point in values {
867 output.append("\(name)(\(point)) = \(a.evaluate(point))")
868 }
869 case "ROOTS", "FACTOR":
870 try requireNonzero(a)
871 let (roots, remaining) = rationalRoots(a)
872 if command == "ROOTS" {
873 let descriptions = roots.map { "\($0.value) (\($0.multiplicity))" }
874 output
875 .append("roots(\(name)) = " +
876 (descriptions.isEmpty ? "none" : descriptions.joined(separator: ", ")))
877 } else {
878 var result = "factor(\(name)) = \(a.content())"
879 for root in roots {
880 let factor = root.value.n == 0 ? "x" : "(\(linear(Fraction(-root.value.n), Fraction(root.value.d)).notation()))"
881 result += " * " + factor + (root.multiplicity > 1 ? "^\(root.multiplicity)" : "")
882 }
883 if remaining.degree >= 1 {
884 result += " * (\(remaining.notation()))"
885 }
886 output.append(result)
887 }
888 case "SQFREE":
889 try requireNonzero(a)
890 var result = "sqfree(\(name)) = \(a.content())"
891 if a.degree >= 1 {
892 let monic = a.monic()
893 var common = polynomialGcd(monic, monic.derivative())
894 var remaining = monic.divide(common).quotient
895 var multiplicity = 1
896 while remaining.degree >= 1 {
897 let shared = polynomialGcd(remaining, common)
898 let factor = remaining.divide(shared).quotient
899 if factor.degree >= 1 {
900 let notation = factor.primitive().notation()
901 result += " * " + (notation == "x" ? "x" : "(\(notation))") +
902 (multiplicity > 1 ? "^\(multiplicity)" : "")
903 }
904 common = common.divide(shared).quotient
905 remaining = shared
906 multiplicity += 1
907 }
908 }
909 output.append(result)
910 case "DESCARTES":
911 try requireNonzero(a)
912 let positive = a.coefficients.reversed().map(\.sign)
913 let negative = a.coefficients.enumerated().map { i, value in i % 2 == 0 ? value.sign : -value.sign }
914 .reversed()
915 output
916 .append(
917 "descartes(\(name)) = positive \(signChanges(positive)), negative \(signChanges(Array(negative)))"
918 )
919 case "STURM":
920 try requireNonzero(a)
921 let u = endpoints[0], v = endpoints[1]
922 try check(u >= v, "BAD INTERVAL")
923 let sequence = sturmSequence(a)
924 output.append("sturm(\(name), \(u), \(v)) = \(variations(sequence, u) - variations(sequence, v))")
925 case "ISOLATE":
926 try requireNonzero(a)
927 let precision = Int(words[2])!
928 let intervals = isolate(a, precision).map { "(\($0.0), \($0.1)]" }
929 output
930 .append("isolate(\(name), \(precision)) = " +
931 (intervals.isEmpty ? "none" : intervals.joined(separator: ", ")))
932 case "RESULTANT":
933 try requireNonzero(a)
934 try requireNonzero(b)
935 output.append("res(\(name), \(words[2])) = \(resultant(a, b))")
936 case "DISCRIM":
937 try requireNonzero(a)
938 try check(a.degree == 0, "DEGREE TOO LOW")
939 var discriminant = resultant(a, a.derivative()) / a.leading
940 if a.degree * (a.degree - 1) / 2 % 2 == 1 {
941 discriminant = -discriminant
942 }
943 output.append("disc(\(name)) = \(discriminant)")
944 case "POWERSUM":
945 try requireNonzero(a)
946 try check(a.degree == 0, "DEGREE TOO LOW")
947 let monic = a.monic(), count = Int(words[2])!
948 var sums: [Fraction] = [.zero]
949 for j in 1...count {
950 var sum = j <= a.degree ? Fraction(Int64(j)) * monic.coefficient(a.degree - j) : .zero
951 for i in 1..<(min(j - 1, a.degree) + 1) {
952 sum = sum + monic.coefficient(a.degree - i) * sums[j - i]
953 }
954 sums.append(-sum)
955 }
956 output.append("powersum(\(name)) = \(sums.dropFirst().map(\.description).joined(separator: ", "))")
957 case "INTERP":
958 store(name, try interpolate(values))
959 case "SPECIAL":
960 store(name, try special(words[2], Int(words[3])!))
961 default:
962 preconditionFailure("Unknown command")
963 }
964 }
965 for (target, value) in pending {
966 polynomials[target] = value
967 }
968 return output
969 }
970}
