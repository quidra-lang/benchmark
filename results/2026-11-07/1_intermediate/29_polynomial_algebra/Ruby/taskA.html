5Fraction = Data.define(:n, :d) do
6 include(Comparable)
7
8 def self.make(n, d = 1)
9 n, d = -n, -d if d.negative?
10 divisor = n.gcd(d)
11 new(n / divisor, d / divisor)
12 end
13
14 def self.parse(text)
15 parts = text.split("/")
16 n = Integer(parts[0], 10)
17 d = parts.length == 1 ? 1 : Integer(parts[1], 10)
18 raise CommandError, "ZERO DENOMINATOR" if d.zero?
19
20 make(n, d)
21 end

23 def +(b) = Fraction.make(n * b.d + b.n * d, d * b.d)
24 def -(b) = Fraction.make(n * b.d - b.n * d, d * b.d)
25 def *(b) = Fraction.make(n * b.n, d * b.d)
26 def /(b) = Fraction.make(n * b.d, d * b.n)
27 def -@ = Fraction.new(-n, d)
28 def abs = Fraction.new(n.abs, d)
29 def sign = n <=> 0
30 def <=>(b) = (n * b.d) <=> (b.n * d)
31 def power(exponent)
32 result = Fraction::ONE
33 exponent.times { result *= self }
34 result
35 end
36
37 def to_s = d == 1 ? n.to_s : "#{n}/#{d}"
38end

44class Polynomial
45 attr_reader :coefficients
46
47 def initialize(values = [])
48 values = values.dup
49 values.pop while !values.empty? && values.last.n.zero?
50 @coefficients = values.freeze
51 freeze
52 end
53
54 ZERO = new
55 ONE = new([Fraction::ONE])
56 def self.constant(value) = new([value])
57 def self.linear(value, slope) = new([value, slope])
58 def degree = coefficients.length - 1
59 def zero? = coefficients.empty?
60 def one? = degree.zero? && self[0] == Fraction::ONE
61 def leading = coefficients.last
62 def [](i) = coefficients[i] || Fraction::ZERO
63 def +(b) = Polynomial.new(Array.new([coefficients.length, b.coefficients.length].max) { |i| self[i] + b[i] })
64 def -(b) = Polynomial.new(Array.new([coefficients.length, b.coefficients.length].max) { |i| self[i] - b[i] })
65 def -@ = Polynomial.new(coefficients.map { |c| -c })
66 def scale(factor) = Polynomial.new(coefficients.map { |c| c * factor })
67 def divide_scalar(divisor) = Polynomial.new(coefficients.map { |c| c / divisor })
68
69 def *(b)
70 return ZERO if zero? || b.zero?
71
72 result = Array.new(degree + b.degree + 1, Fraction::ZERO)
73 coefficients.each_with_index do |a, i|
74 b.coefficients.each_with_index { |c, j| result[i + j] += a * c }
75 end
76
77 Polynomial.new(result)
78 end
79
80 def power(exponent)
81 return ONE if exponent.zero?
82 return ZERO if zero?
83 if degree.zero? && self[0].d == 1 && self[0].n.abs == 1
84 return self[0].n == 1 || exponent.even? ? ONE : self
85 end
86
87 result = ONE
88 exponent.times { result *= self }
89 result
90 end
91
92 def divide(b)
93 quotient = Array.new([0, degree - b.degree + 1].max, Fraction::ZERO)
94 remainder = coefficients.dup
95 current = degree
96 while current >= 0 && current >= b.degree
97 factor = remainder[current] / b.leading
98 shift = current - b.degree
99 quotient[shift] = factor
100 b.coefficients.each_with_index { |c, j| remainder[shift + j] -= factor * c }
101 current -= 1 while current >= 0 && remainder[current].n.zero?
102 end
103
104 Division.new(Polynomial.new(quotient), Polynomial.new(remainder))
105 end
106
107 def quotient(b) = divide(b).quotient
108 def remainder(b) = divide(b).remainder
109 def monic = zero? ? ZERO : divide_scalar(leading)
110 def derivative = Polynomial.new((1..degree).map { |i| whole(i) * self[i] })
111 def differentiate(order)
112 result = self
113 order.times do
114 break if result.zero?
115 result = result.derivative
116 end
117
118 result
119 end
120
121 def integral = Polynomial.new([Fraction::ZERO] + coefficients.each_with_index.map { |c, i| c / whole(i + 1) })
122 def evaluate(point)
123 result = Fraction::ZERO
124 coefficients.reverse_each { |c| result = result * point + c }
125 result
126 end
127
128 def compose(inner)
129 result = ZERO
130 coefficients.reverse_each { |c| result = result * inner + Polynomial.constant(c) }
131 result
132 end
133
134 def gcd(other)
135 a, b = self, other
136 a, b = b, a.remainder(b) until b.zero?
137 a.monic
138 end
139
140 def extended_gcd(other)
141 r0, r1, s0, s1, t0, t1 = self, other, ONE, ZERO, ZERO, ONE
142 until r1.zero?
143 division = r0.divide(r1)
144 s = s0 - division.quotient * s1
145 t = t0 - division.quotient * t1
146 r0, r1 = r1, division.remainder
147 s0, s1 = s1, s
148 t0, t1 = t1, t
149 end
150
151 return Bezout.new(r0, s0, t0) if r0.zero?
152
153 Bezout.new(r0.monic, s0.divide_scalar(r0.leading), t0.divide_scalar(r0.leading))
154 end

156 def content
157 denominator, numerator = 1, 0
158 coefficients.each do |c|
159 denominator = denominator * c.d / denominator.gcd(c.d)
160 numerator = numerator.gcd(c.n)
161 end
162
163 Fraction.make(leading.n.negative? ? -numerator : numerator, denominator)
164 end
165
166 def primitive = divide_scalar(content)
167 def square_free_part = degree < 1 ? self : quotient(gcd(derivative))
168 def without_constant = Polynomial.new(coefficients.drop(1))
169 def check_product(b)
170 raise CommandError, "DEGREE LIMIT" if !zero? && !b.zero? && degree + b.degree > 30
171 end
172
173 def check_power(exponent)
174 raise CommandError, "DEGREE LIMIT" if !zero? && degree * exponent > 30
175 end
176
177 def check_composition(b)
178 raise CommandError, "DEGREE LIMIT" if degree >= 1 && b.degree >= 1 && degree * b.degree > 30
179 end
180
181 def require_nonzero = (raise CommandError, "ZERO POLYNOMIAL" if zero?)
182 def require_divisor = (raise CommandError, "DIVISION BY ZERO" if zero?)
183 def require_positive_degree = (raise CommandError, "DEGREE TOO LOW" if degree.zero?)

185 def format(variable = "x")
186 text = +""
187 degree.downto(0) do |i|
188 c = self[i]
189 next if c.n.zero?
190 if text.empty?
191 text << "-" if c.n.negative?
192 else
193 text << (c.n.negative? ? " - " : " + ")
194 end
195
196 text << c.abs.to_s if i.zero? || c.abs != Fraction::ONE
197 text << variable if i.positive?
198 text << "^#{i}" if i > 1
199 end
200
201 text.empty? ? "0" : text
202 end
203end

208Expression = Data.define(:kind, :value, :left, :right) do
209 def self.make(kind, value: nil, left: nil, right: nil) = new(kind, value, left, right)
210 def evaluate(storage)
211 case kind
212 when :literal
213 return Polynomial.constant(whole(value))
214 when :variable
215 return Polynomial.linear(Fraction::ZERO, Fraction::ONE)
216 when :reference
217 name, order = value
218 return storage.fetch(name) { raise CommandError, "UNKNOWN #{name}" }.differentiate(order)
219 end
220
221 a = left.evaluate(storage)
222 return -a if kind == :negative
223 if kind == :power
224 a.check_power(value)
225 return a.power(value)
226 end
227
228 b = right.evaluate(storage)
229 if kind == :call
230 a.check_composition(b)
231 return a.compose(b)
232 end
233
234 case value
235 when "+"
236 a + b
237 when "-"
238 a - b
239 when "*"
240 a.check_product(b)
241 a * b
242 else
243 b.require_divisor
244 raise CommandError, "NONCONSTANT DIVISOR" if b.degree >= 1
245 a.divide_scalar(b[0])
246 end
247 end

252class Parser
253 def initialize(source)
254 @tokens = []
255 @position = 0
256 i = 0
257 while i < source.length
258 if source[i] == " "
259 i += 1
260 next
261 end
262
263 start = i
264 i += 1
265 kind = if digit?(source[start])
266 :number
267 elsif upper?(source[start])
268 :name
269 else
270 :symbol
271 end
272
273 if kind == :number
274 i += 1 while i < source.length && digit?(source[i])
275 elsif kind == :name
276 i += 1 while i < source.length && (upper?(source[i]) || digit?(source[i]))
277 end
278
279 @tokens << Token.new(kind, source[start...i], start + 1)
280 end
281
282 @tokens << Token.new(:end, "", source.length + 1)
283 end
284
285 def digit?(c) = c >= "0" && c <= "9"
286 def upper?(c) = c >= "A" && c <= "Z"
287 def current = @tokens[@position]
288 def syntax = raise CommandError, "SYNTAX AT #{current.kind == :end ? "END" : current.position}"
289 def accept(symbol)
290 return false unless current.kind == :symbol && current.text == symbol
291 @position += 1
292 true
293 end
294
295 def expect(symbol) = (syntax unless accept(symbol))
296 def parse
297 result = expression
298 syntax unless current.kind == :end
299 result
300 end
301
302 def expression
303 negative = accept("-")
304 result = term
305 result = Expression.make(:negative, left: result) if negative
306 loop do
307 operation = if accept("+")
308 "+"
309 elsif accept("-")
310 "-"
311 else
312 break
313 end
314
315 result = Expression.make(:binary, value: operation, left: result, right: term)
316 end
317
318 result
319 end
320
321 def term
322 result = factor
323 loop do
324 operation = if accept("*")
325 "*"
326 elsif accept("/")
327 "/"
328 elsif current.kind == :name || current.text == "x" || current.text == "("
329 "*"
330 else
331 break
332 end
333
334 result = Expression.make(:binary, value: operation, left: result, right: factor)
335 end
336
337 result
338 end
339
340 def factor
341 result = primary
342 if accept("^")
343 syntax unless current.kind == :number
344 exponent = Integer(current.text, 10)
345 @position += 1
346 result = Expression.make(:power, value: exponent, left: result)
347 end
348
349 result
350 end
351
352 def primary
353 if current.kind == :number
354 value = Integer(current.text, 10)
355 @position += 1
356 return Expression.make(:literal, value: value)
357 end
358
359 return Expression.make(:variable) if accept("x")
360 if current.kind == :name
361 name = current.text
362 @position += 1
363 order = 0
364 order += 1 while accept("'")
365 reference = Expression.make(:reference, value: [name, order].freeze)
366 return reference unless accept("(")
367 argument = expression
368 expect(")")
369 return Expression.make(:call, left: reference, right: argument)
370 end
371
372 if accept("(")
373 result = expression
374 expect(")")
375 return result
376 end
377
378 syntax
379 end
380end

385def rational_roots(a)
386 content = a.content
387 remaining = a.divide_scalar(content)
388 zeros = 0
389 while remaining[0].n.zero?
390 remaining = remaining.without_constant
391 zeros += 1
392 end
393
394 first, last = remaining[0].n.abs, remaining.leading.n
395 numerators = (1..first).select { |n| (first % n).zero? }
396 denominators = (1..last).select { |d| (last % d).zero? }
397 candidates = []
398 numerators.each do |n|
399 denominators.each do |d|
400 candidates.concat([Fraction.make(-n, d), Fraction.make(n, d)]) if n.gcd(d) == 1
401 end
402 end
403
404 candidates.sort!
405 roots = []
406 candidates.each do |candidate|
407 multiplicity = 0
408 while remaining.degree >= 1 && remaining.evaluate(candidate).n.zero?
409 remaining = remaining.quotient(Polynomial.linear(whole(-candidate.n), whole(candidate.d)))
410 multiplicity += 1
411 end
412
413 roots << Root.new(candidate, multiplicity) if multiplicity.positive?
414 end
415
416 roots << Root.new(Fraction::ZERO, zeros) if zeros.positive?
417 RootResult.new(content, remaining, roots.sort_by(&:value))
418end

420def square_free_factors(a)
421 return [] if a.degree < 1
422
423 monic = a.monic
424 c = monic.gcd(monic.derivative)
425 w = monic.quotient(c)
426 factors = []
427 multiplicity = 1
428 while w.degree >= 1
429 y = w.gcd(c)
430 f = w.quotient(y)
431 factors << Factor.new(f.primitive, multiplicity) if f.degree >= 1
432 c, w = c.quotient(y), y
433 multiplicity += 1
434 end
435
436 factors
437end

466def sturm_sequence(a)
467 p = a.square_free_part
468 q = p.derivative
469 sequence = [p]

471 sequence << q

478def variations(sequence, point)
479 previous = count = 0
480 sequence.each do |p|
481 sign = point.infinity.zero? ? p.evaluate(point.value).sign : p.leading.sign
482 sign = -sign if point.infinity.negative? && p.degree.odd?
483 next if sign.zero?
484 count += 1 if !previous.zero? && previous != sign
485 previous = sign
486 end
487
488 count
489end

491def root_count(sequence, low, high)
492 left = variations(sequence, low)
493 right = variations(sequence, high)
494 left - right
495end
496
497def examine(sequence, low, high, width, result)
498 count = root_count(sequence, Bound.finite(low), Bound.finite(high))
499 if count >= 2
500 middle = (low + high) / whole(2)
501 examine(sequence, low, middle, width, result)
502 examine(sequence, middle, high, width, result)
503 elsif count == 1
504 while high - low > width
505 middle = (low + high) / whole(2)
506 if root_count(sequence, Bound.finite(low), Bound.finite(middle)) >= 1
507 high = middle
508 else
509 low = middle
510 end
511 end
512
513 result << Interval.new(low, high)
514 end
515end

517def isolate(a, precision)
518 sequence = sturm_sequence(a)
519 first = sequence.first
520 maximum = Fraction::ZERO
521 first.degree.times { |i| maximum = [maximum, (first[i] / first.leading).abs].max }
522 beta = Fraction::ONE + maximum
523 result = []
524 examine(sequence, -beta, beta, Fraction.make(1, 1 << precision), result)
525 result
526end

528def resultant(p, q)
529 result = Fraction::ONE
530 while q.degree >= 1
531 remainder = p.remainder(q)
532 return Fraction::ZERO if remainder.zero?
533 result *= q.leading.power(p.degree - remainder.degree)
534 result = -result if (p.degree * q.degree).odd?
535 p, q = q, remainder
536 end
537
538 result * q[0].power(p.degree)
539end

546 next if sign.zero?

554def interpolate(values)
555 points = values.each_slice(2).map(&:first)
556 differences = values.each_slice(2).map(&:last)
557 count = points.length
558 (1...count).each do |j|
559 j.times { |i| raise CommandError, "DUPLICATE X #{points[j]}" if points[j] == points[i] }
560 end
561
562 (1...count).each do |j|
563 (count - 1).downto(j) { |i| differences[i] = (differences[i] - differences[i - 1]) / (points[i] - points[i - j]) }
564 end
565
566 result = Polynomial::ZERO
567 (count - 1).downto(0) { |i|
568 result = result * Polynomial.linear(-points[i], Fraction::ONE) + Polynomial.constant(differences[i])
569 }
570 result
571end

573def special(kind, n)
574 raise CommandError, "UNKNOWN KIND #{kind}" unless %w[chebyshev chebyshev2 legendre hermite laguerre].include?(kind)
575 raise CommandError, "DEGREE LIMIT" if n > 30
576
577 previous = Polynomial::ONE
578 return previous if n.zero?
579 current = case kind
580 when "chebyshev2", "hermite"
581 Polynomial.linear(Fraction::ZERO, whole(2))
582 when "laguerre"
583 Polynomial.linear(Fraction::ONE, whole(-1))
584 else
585 Polynomial.linear(Fraction::ZERO, Fraction::ONE)
586 end
587
588 (1...n).each do |j|
589 factor = case kind
590 when "legendre"
591 Polynomial.linear(Fraction::ZERO, whole(2 * j + 1))
592 when "laguerre"
593 Polynomial.linear(whole(2 * j + 1), whole(-1))
594 else
595 Polynomial.linear(Fraction::ZERO, whole(2))
596 end
597
598 multiplier = case kind
599 when "legendre", "laguerre"
600 j
601 when "hermite"
602 2 * j
603 else
604 1
605 end
606
607 following = factor * current - previous.scale(whole(multiplier))
608 following = following.scale(Fraction.make(1, j + 1)) if kind == "legendre" || kind == "laguerre"
609 previous, current = current, following
610 end
611
612 current
613end

632class Calculator
633 def initialize = (@storage = {})
634 def lookup(name) = @storage.fetch(name) { raise CommandError, "UNKNOWN #{name}" }
635 def source_positions(command)
636 case command
637 when "ADD", "SUB", "MUL", "GCD", "LCM", "INVMOD", "COMPOSE"
638 [2, 3]
639 when "POW", "DERIV", "INTEG", "TANGENT"
640 [2]
641 when "MODPOW"
642 [2, 4]
643 when "DIV"
644 [3, 4]
645 when "XGCD"
646 [4, 5]
647 when "CRT"
648 [2, 3, 4, 5]
649 when "RESULTANT"
650 [1, 2]
651 when "LIST", "INTERP", "SPECIAL"
652 []
653 else
654 [1]
655 end
656 end
657
658 def execute(line)
659 response = Response.new
660 if line.start_with?("DEF ")
661 separator = line.index(" ", 4)
662 name = line[4...separator]
663 text = line[(separator + 1)..]
664 expression = Parser.new(text).parse
665 response.save(name, expression.evaluate(@storage))
666 else
667 words = line.split(" ")
668 command = words[0]
669 if command == "DIV" &&
670 words[1] == words[2] ||
671 command == "XGCD" &&
672 (words[1] == words[2] || words[1] == words[3] || words[2] == words[3])
673 raise CommandError, "SAME NAME"
674 end
675
676 operands = source_positions(command).map { |i| lookup(words[i]) }
677 start = case command
678 when "AREA", "TAYLOR", "EVAL", "INTERP"
679 2
680 when "TANGENT"
681 3
682 else
683 words.length
684 end
685
686 values = words.drop(start).map { |text| Fraction.parse(text) }
687 bounds = [Bound.parse(words[2]), Bound.parse(words[3])] if command == "STURM"
688 a, b = operands.fetch(0, Polynomial::ZERO), operands.fetch(1, Polynomial::ZERO)
689 case command
690 when "PRINT"
691 response.lines << "#{words[1]} = #{a.format}"
692 when "DELETE"
693 response.deletion = words[1]
694 response.lines << "deleted #{words[1]}"
695 when "LIST"
696 @storage.keys.sort.each { |name| response.lines << "#{name} = #{@storage[name].format}" }
697 response.lines << "no polynomials" if @storage.empty?
698 when "DEGREE"
699 response.lines << "deg(#{words[1]}) = #{a.zero? ? "-inf" : a.degree}"
700 when "ADD"
701 response.save(words[1], a + b)
702 when "SUB"
703 response.save(words[1], a - b)
704 when "MUL"
705 a.check_product(b)
706 response.save(words[1], a * b)
707 when "POW"
708 exponent = Integer(words[3], 10)
709 a.check_power(exponent)
710 response.save(words[1], a.power(exponent))
711 when "DIV"
712 b.require_divisor
713 division = a.divide(b)
714 response.save(words[1], division.quotient)
715 response.save(words[2], division.remainder)
716 when "GCD"
717 response.save(words[1], a.gcd(b))
718 when "XGCD"
719 result = a.extended_gcd(b)
720 response.save(words[1], result.g)
721 response.save(words[2], result.s)
722 response.save(words[3], result.t)
723 when "LCM"
724 value = Polynomial::ZERO
725 unless a.zero? || b.zero?
726 common = a.gcd(b)
727 raise CommandError, "DEGREE LIMIT" if a.degree + b.degree - common.degree > 30
728 value = (a.quotient(common) * b).monic
729 end
730
731 response.save(words[1], value)
732 when "INVMOD"
733 b.require_divisor
734 b.require_positive_degree
735 result = a.extended_gcd(b)
736 raise CommandError, "NOT COPRIME" unless result.g.one?
737 response.save(words[1], result.s.remainder(b))
738 when "CRT"
739 a2, m2 = operands[2], operands[3]
740 b.require_divisor
741 m2.require_divisor
742 raise CommandError, "DEGREE LIMIT" if b.degree + m2.degree > 30
743 result = b.extended_gcd(m2)
744 raise CommandError, "NOT COPRIME" unless result.g.one?
745 r1, r2 = a.remainder(b), a2.remainder(m2)
746 u = ((r2 - r1) * result.s).remainder(m2)
747 response.save(words[1], r1 + b * u)
748 when "MODPOW"
749 b.require_divisor
750 raise CommandError, "DEGREE LIMIT" if 2 * b.degree - 2 > 30
751 x, y = Polynomial::ONE.remainder(b), a.remainder(b)
752 exponent = Integer(words[3], 10)
753 until exponent.zero?
754 x = (x * y).remainder(b) if exponent.odd?
755 exponent /= 2
756 y = (y * y).remainder(b) unless exponent.zero?
757 end
758
759 response.save(words[1], x)
760 when "DERIV"
761 response.save(words[1], a.differentiate(Integer(words[3], 10)))
762 when "INTEG"
763 raise CommandError, "DEGREE LIMIT" if !a.zero? && a.degree + 1 > 30
764 response.save(words[1], a.integral)
765 when "AREA"
766 integral = a.integral
767 value = integral.evaluate(values[1]) - integral.evaluate(values[0])
768 response.lines << "integral(#{words[1]}, #{values[0]}, #{values[1]}) = #{value}"
769 when "TANGENT"
770 point = values[0]
771 height = a.evaluate(point)
772 slope = a.derivative.evaluate(point)
773 response.save(words[1], Polynomial.linear(height - slope * point, slope))
774 when "TAYLOR"
775 point = values[0]
776 shifted = a.compose(Polynomial.linear(point, Fraction::ONE))
777 variable = point.n.zero? ? "x" : point.n.positive? ? "(x - #{point})" : "(x + #{-point})"
778 response.lines << "taylor(#{words[1]}, #{point}) = #{shifted.format(variable)}"
779 when "COMPOSE"
780 a.check_composition(b)
781 response.save(words[1], a.compose(b))
782 when "EVAL"
783 values.each { |point| response.lines << "#{words[1]}(#{point}) = #{a.evaluate(point)}" }
784 when "ROOTS", "FACTOR"
785 a.require_nonzero
786 roots = rational_roots(a)
787 if command == "ROOTS"
788 text = roots.roots.empty? ? "none" : roots
789 .roots
790 .map { |root| "#{root.value} (#{root.multiplicity})" }
791 .join(", ")
792 response.lines << "roots(#{words[1]}) = #{text}"
793 else
794 text = +"factor(#{words[1]}) = #{roots.content}"
795 roots.roots.each do |root|
796 value = root.value
797 append_factor(text, Polynomial.linear(whole(-value.n), whole(value.d)), root.multiplicity, value.n.zero?)
798 end
799
800 append_factor(text, roots.remaining, 1, false) if roots.remaining.degree >= 1
801 response.lines << text
802 end
803
804 when "SQFREE"
805 a.require_nonzero
806 text = +"sqfree(#{words[1]}) = #{a.content}"
807 square_free_factors(a).each do |factor|
808 p = factor.value
809 append_factor(text, p, factor.multiplicity, p.degree == 1 && p[0].n.zero? && p[1] == Fraction::ONE)
810 end
811
812 response.lines << text
813 when "DESCARTES"
814 a.require_nonzero
815 response.lines << "descartes(#{words[1]}) = positive #{descartes(a, false)}, negative #{descartes(a, true)}"
816 when "STURM"
817 a.require_nonzero
818 low, high = bounds
819 raise CommandError, "BAD INTERVAL" unless low < high
820 sequence = sturm_sequence(a)
821 response.lines << "sturm(#{words[1]}, #{low}, #{high}) = #{root_count(sequence, low, high)}"
822 when "ISOLATE"
823 a.require_nonzero
824 precision = Integer(words[2], 10)
825 intervals = isolate(a, precision)
826 text = intervals.empty? ? "none" : intervals
827 .map { |interval| "(#{interval.low}, #{interval.high}]" }
828 .join(", ")
829 response.lines << "isolate(#{words[1]}, #{precision}) = #{text}"
830 when "RESULTANT"
831 a.require_nonzero
832 b.require_nonzero
833 response.lines << "res(#{words[1]}, #{words[2]}) = #{resultant(a, b)}"
834 when "DISCRIM", "POWERSUM"
835 a.require_nonzero
836 a.require_positive_degree
837 degree = a.degree
838 if command == "DISCRIM"
839 value = resultant(a, a.derivative) / a.leading
840 value = -value if (degree * (degree - 1) / 2).odd?
841 response.lines << "disc(#{words[1]}) = #{value}"
842 else
843 count = Integer(words[2], 10)
844 monic = a.monic
845 sums = Array.new(count + 1, Fraction::ZERO)
846 (1..count).each do |j|
847 value = j <= degree ? whole(j) * monic[degree - j] : Fraction::ZERO
848 (1..[j - 1, degree].min).each { |i| value += monic[degree - i] * sums[j - i] }
849 sums[j] = -value
850 end
851
852 response.lines << "powersum(#{words[1]}) = #{sums.drop(1).join(", ")}"
853 end
854
855 when "INTERP"
856 response.save(words[1], interpolate(values))
857 when "SPECIAL"
858 response.save(words[1], special(words[2], Integer(words[3], 10)))
859 end
860 end
861
862 @storage.delete(response.deletion) if response.deletion
863 response.stores.each { |entry| @storage[entry.name] = entry.value }
864 response.lines
865 end
