9final class Fraction implements Comparable<Fraction> {
10 final int numerator, denominator;
11 const Fraction._(this.numerator, this.denominator);
12 static const zero = Fraction._(0, 1), one = Fraction._(1, 1);
13
14 factory Fraction(int numerator, [int denominator = 1]) {
15 if (denominator < 0) {
16 numerator = -numerator;
17 denominator = -denominator;
18 }
19 final divisor = numerator.abs().gcd(denominator);
20 return Fraction._(numerator ~/ divisor, denominator ~/ divisor);
21 }
22
23 factory Fraction.parse(String text) {
24 final parts = text.split('/');
25 final numerator = int.parse(parts[0], radix: 10);
26 final denominator = parts.length == 1 ? 1 : int.parse(parts[1], radix: 10);
27 if (denominator == 0) throw CommandError('ERROR ZERO DENOMINATOR');
28 return Fraction(numerator, denominator);
29 }
30
31 bool get isZero => numerator == 0;
32 int get sign => numerator.sign;
33 Fraction operator +(Fraction other) => Fraction(
34 numerator * other.denominator + other.numerator * denominator,
35 denominator * other.denominator);
36 Fraction operator -(Fraction other) => Fraction(
37 numerator * other.denominator - other.numerator * denominator,
38 denominator * other.denominator);
39 Fraction operator *(Fraction other) => Fraction(
40 numerator * other.numerator, denominator * other.denominator);
41 Fraction operator /(Fraction other) => Fraction(
42 numerator * other.denominator, denominator * other.numerator);
43 Fraction operator -() => Fraction(-numerator, denominator);
44 Fraction abs() => Fraction(numerator.abs(), denominator);
45 Fraction power(int exponent) {
46 var result = one;
47 for (var i = 0; i < exponent; i++) result = result * this;
48 return result;
49 }
50
51 @override
52 int compareTo(Fraction other) =>
53 (numerator * other.denominator).compareTo(other.numerator * denominator);
54 @override
55 bool operator ==(Object other) =>
56 other is Fraction && numerator == other.numerator && denominator == other.denominator;
57 @override
58 int get hashCode => Object.hash(numerator, denominator);
59 @override
60 String toString() => denominator == 1 ? '$numerator' : '$numerator/$denominator';
61}

63final class Polynomial {
64 final List<Fraction> _coefficients;
65 Polynomial._(this._coefficients);
66 static final zero = Polynomial([]), one = Polynomial([Fraction.one]);
67 static final x = Polynomial([Fraction.zero, Fraction.one]);
68
69 factory Polynomial(List<Fraction> values) {
70 var length = values.length;
71 while (length > 0 && values[length - 1].isZero) length--;
72 return Polynomial._(List<Fraction>.unmodifiable(values.take(length)));
73 }
74
75 int get degree => _coefficients.length - 1;
76 bool get isZero => _coefficients.isEmpty;
77 Fraction get leading => _coefficients.last;
78 Fraction operator [](int index) => index < _coefficients.length ? _coefficients[index] : Fraction.zero;
79 Polynomial operator +(Polynomial other) => Polynomial([
80 for (var i = 0; i <= math.max(degree, other.degree); i++) this[i] + other[i],
81 ]);
82 Polynomial operator -(Polynomial other) => Polynomial([
83 for (var i = 0; i <= math.max(degree, other.degree); i++) this[i] - other[i],
84 ]);
85 Polynomial operator -() => Polynomial([for (final value in _coefficients) -value]);
86 Polynomial operator *(Polynomial other) {
87 if (isZero || other.isZero) return zero;
88 final result = List.filled(degree + other.degree + 1, Fraction.zero);
89 for (var i = 0; i <= degree; i++) {
90 for (var j = 0; j <= other.degree; j++) {
91 result[i + j] = result[i + j] + this[i] * other[j];
92 }
93 }
94 return Polynomial(result);
95 }
96
97 Polynomial scale(Fraction factor) => Polynomial([for (final value in _coefficients) value * factor]);
98 Polynomial divideScalar(Fraction divisor) => Polynomial([for (final value in _coefficients) value / divisor]);
99 Polynomial monic() => isZero ? zero : divideScalar(leading);
100 Polynomial power(int exponent) {
101 if (exponent == 0) return one;
102 if (isZero) return zero;
103 if (degree == 0 && leading.denominator == 1 && leading.numerator.abs() == 1) {
104 return leading.numerator == 1 || exponent.isEven ? one : this;
105 }
106 var result = one;
107 for (var i = 0; i < exponent; i++) result = result * this;
108 return result;
109 }
110
111 Fraction evaluate(Fraction point) {
112 var result = Fraction.zero;
113 for (var i = degree; i >= 0; i--) result = result * point + this[i];
114 return result;
115 }
116
117 Polynomial compose(Polynomial inner) {
118 var result = zero;
119 for (var i = degree; i >= 0; i--) result = result * inner + Polynomial([this[i]]);
120 return result;
121 }
122
123 Polynomial derivative([int order = 1]) {
124 var result = this;
125 for (var step = 0; step < order; step++) {
126 result = Polynomial([for (var i = 1; i <= result.degree; i++) Fraction(i) * result[i]]);
127 }
128 return result;
129 }
130
131 Polynomial integral() => isZero ? zero : Polynomial([
132 Fraction.zero,
133 for (var i = 0; i <= degree; i++) this[i] / Fraction(i + 1),
134 ]);
135
136 ({Polynomial quotient, Polynomial remainder}) divide(Polynomial divisor) {
137 final quotient = List.filled(math.max(0, degree - divisor.degree + 1), Fraction.zero);
138 final remainder = List<Fraction>.of(_coefficients);
139 var currentDegree = degree;
140 while (currentDegree >= divisor.degree && currentDegree >= 0) {
141 final factor = remainder[currentDegree] / divisor.leading;
142 final shift = currentDegree - divisor.degree;
143 quotient[shift] = factor;
144 for (var j = 0; j <= divisor.degree; j++) {
145 remainder[shift + j] = remainder[shift + j] - factor * divisor[j];
146 }
147 while (currentDegree >= 0 && remainder[currentDegree].isZero) currentDegree--;
148 }
149 return (quotient: Polynomial(quotient), remainder: Polynomial(remainder));
150 }
151
152 static Polynomial gcd(Polynomial a, Polynomial b) {
153 while (!b.isZero) (a, b) = (b, a.divide(b).remainder);
154 return a.monic();
155 }
156
157 static ({Polynomial g, Polynomial s, Polynomial t}) extendedGcd(Polynomial a, Polynomial b) {
158 var r0 = a, r1 = b, s0 = one, s1 = zero, t0 = zero, t1 = one;
159 while (!r1.isZero) {
160 final division = r0.divide(r1);
161 (r0, r1) = (r1, division.remainder);
162 (s0, s1) = (s1, s0 - division.quotient * s1);
163 (t0, t1) = (t1, t0 - division.quotient * t1);
164 }
165 return r0.isZero
166 ? (g: r0, s: s0, t: t0)
167 : (g: r0.monic(), s: s0.divideScalar(r0.leading), t: t0.divideScalar(r0.leading));
168 }
169
170 Fraction content() {
171 var denominator = 1, numerator = 0;
172 for (final coefficient in _coefficients) {
173 denominator = denominator * coefficient.denominator ~/ denominator.gcd(coefficient.denominator);
174 numerator = numerator.gcd(coefficient.numerator.abs());
175 }
176 return Fraction(leading.sign < 0 ? -numerator : numerator, denominator);
177 }
178
179 Polynomial primitive() => divideScalar(content());
180 Polynomial squareFreePart() => degree < 1 ? this : divide(gcd(this, derivative())).quotient;
181 Polynomial withoutConstant() => Polynomial(_coefficients.sublist(1));
182
183 String format([String variable = 'x']) {
184 if (isZero) return '0';
185 final result = StringBuffer();
186 for (var i = degree; i >= 0; i--) {
187 final coefficient = this[i];
188 if (coefficient.isZero) continue;
189 if (result.isEmpty) {
190 if (coefficient.sign < 0) result.write('-');
191 } else {
192 result.write(coefficient.sign < 0 ? ' - ' : ' + ');
193 }
194 final magnitude = coefficient.abs();
195 if (i == 0 || magnitude != Fraction.one) result.write(magnitude);
196 if (i > 0) result.write(variable);
197 if (i > 1) result.write('^$i');
198 }
199 return result.toString();
200 }
201}

248final class BinaryExpression extends Expression {
249 final String operation;
250 final Expression left, right;
251 BinaryExpression(this.operation, this.left, this.right);
252 @override
253 Polynomial evaluate(Map<String, Polynomial> storage) {
254 final a = left.evaluate(storage), b = right.evaluate(storage);
255 switch (operation) {
256 case '+': return a + b;
257 case '-': return a - b;
258 case '*': checkProduct(a, b); return a * b;
259 default:
260 checkDivisor(b);
261 if (b.degree >= 1) throw CommandError('ERROR NONCONSTANT DIVISOR');
262 return a.divideScalar(b[0]);
263 }
264 }
265}

267 final Expression operand;
268 final int exponent;
269 PowerExpression(this.operand, this.exponent);
270 @override
271 Polynomial evaluate(Map<String, Polynomial> storage) {
272 final value = operand.evaluate(storage);
273 checkPower(value, exponent);
274 return value.power(exponent);
275 }
276}

282 final outer = function.evaluate(storage), inner = argument.evaluate(storage);

296final class Parser {
297 final List<Token> tokens = [];
298 int position = 0;
299 Token get current => tokens[position];
300 bool at(String symbol) => current.kind == TokenKind.symbol && current.text == symbol;
301 static bool digit(int code) => code >= 48 && code <= 57;
302 static bool upper(int code) => code >= 65 && code <= 90;
303
304 Parser(String source) {
305 var i = 0;
306 while (i < source.length) {
307 if (source.codeUnitAt(i) == 32) { i++; continue; }
308 final start = i++;
309 TokenKind kind;
310 if (digit(source.codeUnitAt(start))) {
311 kind = TokenKind.number;
312 while (i < source.length && digit(source.codeUnitAt(i))) i++;
313 } else if (upper(source.codeUnitAt(start))) {
314 kind = TokenKind.name;
315 while (i < source.length && (upper(source.codeUnitAt(i)) || digit(source.codeUnitAt(i)))) i++;
316 } else {
317 kind = TokenKind.symbol;
318 }
319 tokens.add(Token(kind, source.substring(start, i), start + 1));
320 }
321 tokens.add(Token(TokenKind.end, '', source.length + 1));
322 }
323
324 Never syntax() => throw CommandError(current.kind == TokenKind.end
325 ? 'ERROR SYNTAX AT END' : 'ERROR SYNTAX AT ${current.position}');
326 void expect(String symbol) {
327 if (!at(symbol)) syntax();
328 position++;
329 }
330
331 Expression parse() {
332 final result = expression();
333 if (current.kind != TokenKind.end) syntax();
334 return result;
335 }
336
337 Expression expression() {
338 final negative = at('-');
339 if (negative) position++;
340 var result = term();
341 if (negative) result = NegativeExpression(result);
342 while (at('+') || at('-')) {
343 final operation = current.text;
344 position++;
345 result = BinaryExpression(operation, result, term());
346 }
347 return result;
348 }
349
350 Expression term() {
351 var result = factor();
352 while (true) {
353 String operation;
354 if (at('*') || at('/')) { operation = current.text; position++; }
355 else if (at('x') || at('(') || current.kind == TokenKind.name) { operation = '*'; }
356 else { break; }
357 result = BinaryExpression(operation, result, factor());
358 }
359 return result;
360 }
361
362 Expression factor() {
363 var result = primary();
364 if (at('^')) {
365 position++;
366 if (current.kind != TokenKind.number) syntax();
367 final exponent = int.parse(current.text, radix: 10);
368 position++;
369 result = PowerExpression(result, exponent);
370 }
371 return result;
372 }
373
374 Expression primary() {
375 if (current.kind == TokenKind.number) {
376 final value = Polynomial([Fraction(int.parse(current.text, radix: 10))]);
377 position++;
378 return LiteralExpression(value);
379 }
380 if (at('x')) { position++; return LiteralExpression(Polynomial.x); }
381 if (current.kind == TokenKind.name) {
382 final name = current.text;
383 position++;
384 var order = 0;
385 while (at("'")) { order++; position++; }
386 final reference = NameExpression(name, order);
387 if (!at('(')) return reference;
388 position++;
389 final argument = expression();
390 expect(')');
391 return CallExpression(reference, argument);
392 }
393 if (at('(')) {
394 position++;
395 final result = expression();
396 expect(')');
397 return result;
398 }
399 syntax();
400 }
401}

403final class Endpoint implements Comparable<Endpoint> {
404 final int infinity;
405 final Fraction value;
406 const Endpoint(this.infinity, this.value);
407 factory Endpoint.parse(String text) => switch (text) {
408 '-inf' => const Endpoint(-1, Fraction.zero),
409 'inf' => const Endpoint(1, Fraction.zero),
410 _ => Endpoint(0, Fraction.parse(text)),
411 };
412 @override
413 int compareTo(Endpoint other) => infinity == 0 && other.infinity == 0
414 ? value.compareTo(other.value) : infinity.compareTo(other.infinity);
415 @override
416 String toString() => infinity == 0 ? '$value' : infinity < 0 ? '-inf' : 'inf';
417}

419typedef Root = ({Fraction root, int multiplicity});
420typedef Factor = ({Polynomial polynomial, int multiplicity});
421typedef Interval = ({Fraction left, Fraction right});
422
423({Fraction content, Polynomial remaining, List<Root> roots}) rationalRoots(Polynomial value) {
424 final content = value.content();
425 var remaining = value.divideScalar(content);
426 var zeros = 0;
427 while (remaining[0].isZero) { remaining = remaining.withoutConstant(); zeros++; }
428 List<int> divisors(int number) => [for (var i = 1; i <= number; i++) if (number % i == 0) i];
429 final numerators = divisors(remaining[0].numerator.abs());
430 final denominators = divisors(remaining.leading.numerator);
431 final candidates = <Fraction>[];
432 for (final numerator in numerators) {
433 for (final denominator in denominators) {
434 if (numerator.gcd(denominator) == 1) {
435 candidates.add(Fraction(numerator, denominator));
436 candidates.add(Fraction(-numerator, denominator));
437 }
438 }
439 }
440 candidates.sort();
441 final roots = <Root>[];
442 for (final candidate in candidates) {
443 var count = 0;
444 final divisor = Polynomial([Fraction(-candidate.numerator), Fraction(candidate.denominator)]);
445 while (remaining.degree >= 1 && remaining.evaluate(candidate).isZero) {
446 remaining = remaining.divide(divisor).quotient;
447 count++;
448 }
449 if (count > 0) roots.add((root: candidate, multiplicity: count));
450 }
451 if (zeros > 0) roots.add((root: Fraction.zero, multiplicity: zeros));
452 roots.sort((a, b) => a.root.compareTo(b.root));
453 return (content: content, remaining: remaining, roots: roots);
454}

456List<Factor> squareFreeFactors(Polynomial value) {
457 final result = <Factor>[];
458 if (value.degree < 1) return result;
459 final monic = value.monic();
460 var c = Polynomial.gcd(monic, monic.derivative());
461 var w = monic.divide(c).quotient;
462 for (var i = 1; w.degree >= 1; i++) {
463 final y = Polynomial.gcd(w, c);
464 final factor = w.divide(y).quotient;
465 if (factor.degree >= 1) result.add((polynomial: factor.primitive(), multiplicity: i));
466 c = c.divide(y).quotient;
467 w = y;
468 }
469 return result;
470}

472List<Polynomial> sturmSequence(Polynomial value) {
473 var previous = value.squareFreePart();
474 var current = previous.derivative();
475 final result = [previous];
476 while (!current.isZero) {
477 result.add(current);
478 final next = -previous.divide(current).remainder;
479 previous = current;
480 current = next;
481 }
482 return result;
483}

485int variations(List<Polynomial> sequence, Endpoint point) {
486 var previous = 0, changes = 0;
487 for (final value in sequence) {
488 var sign = point.infinity == 0 ? value.evaluate(point.value).sign : value.leading.sign;
489 if (point.infinity < 0 && value.degree.isOdd) sign = -sign;
490 if (sign == 0) continue;
491 if (previous != 0 && previous != sign) changes++;
492 previous = sign;
493 }
494 return changes;
495}

500List<Interval> isolate(Polynomial value, int precision) {
501 final sequence = sturmSequence(value);
502 final first = sequence[0];
503 var maximum = Fraction.zero;
504 for (var i = 0; i < first.degree; i++) {
505 final candidate = (first[i] / first.leading).abs();
506 if (candidate.compareTo(maximum) > 0) maximum = candidate;
507 }
508 final beta = Fraction.one + maximum, width = Fraction(1, 1 << precision);
509 final result = <Interval>[];
510 int count(Fraction left, Fraction right) => rootCount(sequence, Endpoint(0, left), Endpoint(0, right));
511 void examine(Fraction left, Fraction right) {
512 final roots = count(left, right);
513 if (roots >= 2) {
514 final middle = (left + right) / Fraction(2);
515 examine(left, middle);
516 examine(middle, right);
517 } else if (roots == 1) {
518 while ((right - left).compareTo(width) > 0) {
519 final middle = (left + right) / Fraction(2);
520 if (count(left, middle) >= 1) { right = middle; } else { left = middle; }
521 }
522 result.add((left: left, right: right));
523 }
524 }
525 examine(-beta, beta);
526 return result;
527}

529Fraction resultant(Polynomial p, Polynomial q) {
530 var result = Fraction.one;
531 while (q.degree >= 1) {
532 final remainder = p.divide(q).remainder;
533 if (remainder.isZero) return Fraction.zero;
534 result = result * q.leading.power(p.degree - remainder.degree);
535 if ((p.degree * q.degree).isOdd) result = -result;
536 (p, q) = (q, remainder);
537 }
538 return result * q[0].power(p.degree);
539}

541int descartes(Polynomial value, bool negative) {
542 var previous = 0, count = 0;
543 for (var i = value.degree; i >= 0; i--) {
544 var sign = value[i].sign;
545 if (negative && i.isOdd) sign = -sign;
546 if (sign == 0) continue;
547 if (previous != 0 && previous != sign) count++;
548 previous = sign;
549 }
550 return count;
551}

553Polynomial interpolate(List<Fraction> values) {
554 final count = values.length ~/ 2;
555 final x = [for (var i = 0; i < count; i++) values[2 * i]];
556 final differences = [for (var i = 0; i < count; i++) values[2 * i + 1]];
557 for (var j = 1; j < count; j++) {
558 for (var i = 0; i < j; i++) {
559 if (x[j].compareTo(x[i]) == 0) throw CommandError('ERROR DUPLICATE X ${x[j]}');
560 }
561 }
562 for (var j = 1; j < count; j++) {
563 for (var i = count - 1; i >= j; i--) {
564 differences[i] = (differences[i] - differences[i - 1]) / (x[i] - x[i - j]);
565 }
566 }
567 var result = Polynomial.zero;
568 for (var i = count - 1; i >= 0; i--) {
569 result = result * Polynomial([-x[i], Fraction.one]) + Polynomial([differences[i]]);
570 }
571 return result;
572}

574Polynomial special(String kind, int degree) {
575 if (!['chebyshev', 'chebyshev2', 'legendre', 'hermite', 'laguerre'].contains(kind)) {
576 throw CommandError('ERROR UNKNOWN KIND $kind');
577 }
578 if (degree > 30) throw CommandError('ERROR DEGREE LIMIT');
579 var previous = Polynomial.one;
580 var current = switch (kind) {
581 'chebyshev2' || 'hermite' => Polynomial([Fraction.zero, Fraction(2)]),
582 'laguerre' => Polynomial([Fraction.one, Fraction(-1)]),
583 _ => Polynomial.x,
584 };
585 if (degree == 0) return previous;
586 for (var j = 1; j < degree; j++) {
587 final factor = switch (kind) {
588 'legendre' => Polynomial([Fraction.zero, Fraction(2 * j + 1)]),
589 'laguerre' => Polynomial([Fraction(2 * j + 1), Fraction(-1)]),
590 _ => Polynomial([Fraction.zero, Fraction(2)]),
591 };
592 final scale = kind == 'legendre' || kind == 'laguerre' ? j : kind == 'hermite' ? 2 * j : 1;
593 var next = factor * current - previous.scale(Fraction(scale));
594 if (kind == 'legendre' || kind == 'laguerre') next = next.scale(Fraction(1, j + 1));
595 previous = current;
596 current = next;
597 }
598 return current;
599}

614 List<String> execute(String line) {
615 final output = <String>[];
616 final pending = <(String, Polynomial)>[];
617 String? deletion;
618 void save(String name, Polynomial value) {
619 pending.add((name, value));
620 output.add('$name = ${value.format()}');
621 }
622
623 if (line.startsWith('DEF ')) {
624 final separator = line.indexOf(' ', 4);
625 final name = line.substring(4, separator);
626 final expression = Parser(line.substring(separator + 1)).parse();
627 save(name, expression.evaluate(storage));
628 } else {
629 final words = line.split(' ');
630 final command = words[0];
631 if (command == 'DIV' && words[1] == words[2] || command == 'XGCD' &&
632 (words[1] == words[2] || words[1] == words[3] || words[2] == words[3])) {
633 throw CommandError('ERROR SAME NAME');
634 }
635 final operands = [for (final index in nameArguments(command)) lookup(words[index])];
636 final literalArguments = switch (command) {
637 'AREA' => [2, 3], 'TANGENT' => [3], 'TAYLOR' => [2],
638 'EVAL' || 'INTERP' => [for (var i = 2; i < words.length; i++) i], _ => <int>[],
639 };
640 final literals = [for (final index in literalArguments) Fraction.parse(words[index])];
641 late final Endpoint left, right;
642 if (command == 'STURM') {
643 left = Endpoint.parse(words[2]);
644 right = Endpoint.parse(words[3]);
645 }
646 final a = operands.isEmpty ? Polynomial.zero : operands[0];
647 final b = operands.length < 2 ? Polynomial.zero : operands[1];
648 switch (command) {
649 case 'PRINT': output.add('${words[1]} = ${a.format()}');
650 case 'DELETE': deletion = words[1]; output.add('deleted ${words[1]}');
651 case 'LIST':
652 if (storage.isEmpty) {
653 output.add('no polynomials');
654 } else {
655 final names = storage.keys.toList()..sort();
656 for (final name in names) output.add('$name = ${storage[name]!.format()}');
657 }
658 case 'DEGREE': output.add('deg(${words[1]}) = ${a.isZero ? '-inf' : a.degree}');
659 case 'ADD': save(words[1], a + b);
660 case 'SUB': save(words[1], a - b);
661 case 'MUL': checkProduct(a, b); save(words[1], a * b);
662 case 'POW':
663 final exponent = int.parse(words[3]);
664 checkPower(a, exponent);
665 save(words[1], a.power(exponent));
666 case 'DIV':
667 checkDivisor(b);
668 final division = a.divide(b);
669 save(words[1], division.quotient);
670 save(words[2], division.remainder);
671 case 'GCD': save(words[1], Polynomial.gcd(a, b));
672 case 'XGCD':
673 final result = Polynomial.extendedGcd(a, b);
674 save(words[1], result.g);
675 save(words[2], result.s);
676 save(words[3], result.t);
677 case 'LCM':
678 if (a.isZero || b.isZero) {
679 save(words[1], Polynomial.zero);
680 } else {
681 final gcd = Polynomial.gcd(a, b);
682 if (a.degree + b.degree - gcd.degree > 30) throw CommandError('ERROR DEGREE LIMIT');
683 save(words[1], (a.divide(gcd).quotient * b).monic());
684 }
685 case 'INVMOD':
686 checkDivisor(b);
687 checkPositiveDegree(b);
688 final result = Polynomial.extendedGcd(a, b);
689 if (result.g.degree != 0 || result.g[0] != Fraction.one) throw CommandError('ERROR NOT COPRIME');
690 save(words[1], result.s.divide(b).remainder);
691 case 'CRT':
692 final a2 = operands[2], m2 = operands[3];
693 checkDivisor(b);
694 checkDivisor(m2);
695 if (b.degree + m2.degree > 30) throw CommandError('ERROR DEGREE LIMIT');
696 final result = Polynomial.extendedGcd(b, m2);
697 if (result.g.degree != 0 || result.g[0] != Fraction.one) throw CommandError('ERROR NOT COPRIME');
698 final r1 = a.divide(b).remainder, r2 = a2.divide(m2).remainder;
699 final u = ((r2 - r1) * result.s).divide(m2).remainder;
700 save(words[1], r1 + b * u);
701 case 'MODPOW':
702 checkDivisor(b);
703 if (2 * b.degree - 2 > 30) throw CommandError('ERROR DEGREE LIMIT');
704 var x = Polynomial.one.divide(b).remainder, y = a.divide(b).remainder;
705 var exponent = int.parse(words[3]);
706 while (exponent > 0) {
707 if (exponent.isOdd) x = (x * y).divide(b).remainder;
708 exponent ~/= 2;
709 if (exponent != 0) y = (y * y).divide(b).remainder;
710 }
711 save(words[1], x);
712 case 'DERIV': save(words[1], a.derivative(int.parse(words[3])));
713 case 'INTEG':
714 if (a.degree == 30) throw CommandError('ERROR DEGREE LIMIT');
715 save(words[1], a.integral());
716 case 'AREA':
717 final integral = a.integral();
718 final value = integral.evaluate(literals[1]) - integral.evaluate(literals[0]);
719 output.add('integral(${words[1]}, ${literals[0]}, ${literals[1]}) = $value');
720 case 'TANGENT':
721 final point = literals[0];
722 final value = a.evaluate(point), slope = a.derivative().evaluate(point);
723 save(words[1], Polynomial([value - slope * point, slope]));
724 case 'TAYLOR':
725 final point = literals[0];
726 final shifted = a.compose(Polynomial([point, Fraction.one]));
727 final variable = point.isZero ? 'x' : point.sign > 0 ? '(x - $point)' : '(x + ${-point})';
728 output.add('taylor(${words[1]}, $point) = ${shifted.format(variable)}');
729 case 'COMPOSE': checkComposition(a, b); save(words[1], a.compose(b));
730 case 'EVAL':
731 for (final point in literals) output.add('${words[1]}($point) = ${a.evaluate(point)}');
732 case 'ROOTS':
733 checkNonzero(a);
734 final roots = rationalRoots(a).roots;
735 final result = roots.isEmpty ? 'none' : roots.map((root) => '${root.root} (${root.multiplicity})').join(', ');
736 output.add('roots(${words[1]}) = $result');
737 case 'FACTOR':
738 checkNonzero(a);
739 final factorization = rationalRoots(a);
740 final result = StringBuffer('factor(${words[1]}) = ${factorization.content}');
741 for (final root in factorization.roots) {
742 final factor = root.root.isZero ? 'x'
743 : '(${Polynomial([Fraction(-root.root.numerator), Fraction(root.root.denominator)]).format()})';
744 result.write(' * $factor');
745 if (root.multiplicity >= 2) result.write('^${root.multiplicity}');
746 }
747 if (factorization.remaining.degree >= 1) result.write(' * (${factorization.remaining.format()})');
748 output.add(result.toString());
749 case 'SQFREE':
750 checkNonzero(a);
751 final factors = squareFreeFactors(a);
752 final result = StringBuffer('sqfree(${words[1]}) = ${a.content()}');
753 for (final factor in factors) {
754 final value = factor.polynomial;
755 final isX = value.degree == 1 && value[0].isZero && value[1] == Fraction.one;
756 result.write(' * ${isX ? 'x' : '(${value.format()})'}');
757 if (factor.multiplicity >= 2) result.write('^${factor.multiplicity}');
758 }
759 output.add(result.toString());
760 case 'DESCARTES':
761 checkNonzero(a);
762 output.add('descartes(${words[1]}) = positive ${descartes(a, false)}, negative ${descartes(a, true)}');
763 case 'STURM':
764 checkNonzero(a);
765 if (left.compareTo(right) >= 0) throw CommandError('ERROR BAD INTERVAL');
766 final sequence = sturmSequence(a);
767 output.add('sturm(${words[1]}, $left, $right) = ${rootCount(sequence, left, right)}');
768 case 'ISOLATE':
769 checkNonzero(a);
770 final precision = int.parse(words[2]);
771 final intervals = isolate(a, precision);
772 final result = intervals.isEmpty ? 'none' : intervals.map((interval) => '(${interval.left}, ${interval.right}]').join(', ');
773 output.add('isolate(${words[1]}, $precision) = $result');
774 case 'RESULTANT':
775 checkNonzero(a);
776 checkNonzero(b);
777 output.add('res(${words[1]}, ${words[2]}) = ${resultant(a, b)}');
778 case 'DISCRIM':
779 checkNonzero(a);
780 checkPositiveDegree(a);
781 var value = resultant(a, a.derivative()) / a.leading;
782 if ((a.degree * (a.degree - 1) ~/ 2).isOdd) value = -value;
783 output.add('disc(${words[1]}) = $value');
784 case 'POWERSUM':
785 checkNonzero(a);
786 checkPositiveDegree(a);
787 final count = int.parse(words[2]);
788 final monic = a.monic();
789 final sums = List.filled(count + 1, Fraction.zero);
790 for (var j = 1; j <= count; j++) {
791 var value = j <= a.degree ? Fraction(j) * monic[a.degree - j] : Fraction.zero;
792 for (var i = 1; i <= math.min(j - 1, a.degree); i++) value = value + monic[a.degree - i] * sums[j - i];
793 sums[j] = -value;
794 }
795 output.add('powersum(${words[1]}) = ${sums.skip(1).join(', ')}');
796 case 'INTERP': save(words[1], interpolate(literals));
797 case 'SPECIAL': save(words[1], special(words[2], int.parse(words[3])));
798 default: throw StateError('Unknown command');
799 }
800 }
801 if (deletion != null) storage.remove(deletion);
802 for (final (name, value) in pending) storage[name] = value;
803 return output;
804 }
805}
