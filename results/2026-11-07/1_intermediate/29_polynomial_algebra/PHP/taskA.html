6function integerGcd(int $a, int $b): int {
7 $a = abs($a); $b = abs($b);
8 while ($b !== 0) [$a, $b] = [$b, $a % $b];
9 return $a;
10}

12final readonly class Fraction {
13 public int $n;
14 public int $d;
15 public function __construct(int $n, int $d = 1) {
16 if ($d < 0) { $n = -$n; $d = -$d; }
17 $divisor = integerGcd($n, $d);
18 $this->n = intdiv($n, $divisor);
19 $this->d = intdiv($d, $divisor);
20 }
21 public static function zero(): self { static $value; return $value ??= new self(0); }
22 public static function one(): self { static $value; return $value ??= new self(1); }
23 public static function parse(string $text): self {
24 $parts = explode('/', $text);
25 $denominator = count($parts) === 1 ? 1 : (int)$parts[1];
26 if ($denominator === 0) throw new CommandError('ERROR ZERO DENOMINATOR');
27 return new self((int)$parts[0], $denominator);
28 }
29 public function isZero(): bool { return $this->n === 0; }
30 public function sign(): int { return $this->n <=> 0; }
31 public function add(self $b): self { return new self($this->n * $b->d + $b->n * $this->d, $this->d * $b->d); }
32 public function sub(self $b): self { return new self($this->n * $b->d - $b->n * $this->d, $this->d * $b->d); }
33 public function mul(self $b): self { return new self($this->n * $b->n, $this->d * $b->d); }
34 public function div(self $b): self { return new self($this->n * $b->d, $this->d * $b->n); }
35 public function neg(): self { return new self(-$this->n, $this->d); }
36 public function absolute(): self { return new self(abs($this->n), $this->d); }
37 public function equals(self $b): bool { return $this->n === $b->n && $this->d === $b->d; }
38 public function compare(self $b): int { return ($this->n * $b->d) <=> ($b->n * $this->d); }
39 public function power(int $exponent): self {
40 $result = self::one();
41 for ($i = 0; $i < $exponent; $i++) $result = $result->mul($this);
42 return $result;
43 }
44 public function __toString(): string { return $this->d === 1 ? (string)$this->n : $this->n . '/' . $this->d; }

47final readonly class Polynomial {
48 public array $coefficients;
49 public function __construct(array $values = []) {
50 while ($values !== [] && $values[count($values) - 1]->isZero()) array_pop($values);
51 $this->coefficients = $values;
52 }
53 public static function zero(): self { static $value; return $value ??= new self(); }
54 public static function one(): self { static $value; return $value ??= new self([Fraction::one()]); }
55 public static function variable(): self { static $value; return $value ??= new self([Fraction::zero(), Fraction::one()]); }
56 public function degree(): int { return count($this->coefficients) - 1; }
57 public function isZero(): bool { return $this->coefficients === []; }
58 public function leading(): Fraction { return $this->coefficients[$this->degree()]; }
59 public function at(int $index): Fraction { return $this->coefficients[$index] ?? Fraction::zero(); }
60 public function add(self $b): self {
61 $result = [];
62 for ($i = 0; $i <= max($this->degree(), $b->degree()); $i++) $result[] = $this->at($i)->add($b->at($i));
63 return new self($result);
64 }
65 public function sub(self $b): self {
66 $result = [];
67 for ($i = 0; $i <= max($this->degree(), $b->degree()); $i++) $result[] = $this->at($i)->sub($b->at($i));
68 return new self($result);
69 }
70 public function neg(): self { return new self(array_map(static fn(Fraction $c): Fraction => $c->neg(), $this->coefficients)); }
71 public function mul(self $b): self {
72 if ($this->isZero() || $b->isZero()) return self::zero();
73 $result = array_fill(0, $this->degree() + $b->degree() + 1, Fraction::zero());
74 foreach ($this->coefficients as $i => $left) {
75 foreach ($b->coefficients as $j => $right) $result[$i + $j] = $result[$i + $j]->add($left->mul($right));
76 }
77 return new self($result);
78 }
79 public function scale(Fraction $factor): self {
80 return new self(array_map(static fn(Fraction $c): Fraction => $c->mul($factor), $this->coefficients));
81 }
82 public function divideScalar(Fraction $divisor): self {
83 return new self(array_map(static fn(Fraction $c): Fraction => $c->div($divisor), $this->coefficients));
84 }
85 public function monic(): self { return $this->isZero() ? self::zero() : $this->divideScalar($this->leading()); }
86 public function power(int $exponent): self {
87 if ($exponent === 0) return self::one();
88 if ($this->isZero()) return self::zero();
89 if ($this->degree() === 0 && $this->leading()->d === 1 && abs($this->leading()->n) === 1) {
90 return $this->leading()->n === 1 || $exponent % 2 === 0 ? self::one() : $this;
91 }
92 $result = self::one();
93 for ($i = 0; $i < $exponent; $i++) $result = $result->mul($this);
94 return $result;
95 }
96 public function evaluate(Fraction $point): Fraction {
97 $result = Fraction::zero();
98 for ($i = $this->degree(); $i >= 0; $i--) $result = $result->mul($point)->add($this->at($i));
99 return $result;
100 }
101 public function compose(self $inner): self {
102 $result = self::zero();
103 for ($i = $this->degree(); $i >= 0; $i--) $result = $result->mul($inner)->add(new self([$this->at($i)]));
104 return $result;
105 }
106 public function derivative(int $order = 1): self {
107 $result = $this;
108 for ($step = 0; $step < $order; $step++) {
109 $values = [];
110 for ($i = 1; $i <= $result->degree(); $i++) $values[] = (new Fraction($i))->mul($result->at($i));
111 $result = new self($values);
112 }
113 return $result;
114 }
115 public function integral(): self {
116 $result = [Fraction::zero()];
117 foreach ($this->coefficients as $i => $value) $result[] = $value->div(new Fraction($i + 1));
118 return new self($result);
119 }
120 public function divide(self $divisor): array {
121 $quotient = array_fill(0, max(0, $this->degree() - $divisor->degree() + 1), Fraction::zero());
122 $remainder = $this->coefficients;
123 while ($remainder !== [] && count($remainder) >= count($divisor->coefficients)) {
124 $factor = $remainder[count($remainder) - 1]->div($divisor->leading());
125 $shift = count($remainder) - count($divisor->coefficients);
126 $quotient[$shift] = $factor;
127 foreach ($divisor->coefficients as $j => $value) $remainder[$shift + $j] = $remainder[$shift + $j]->sub($factor->mul($value));
128 while ($remainder !== [] && $remainder[count($remainder) - 1]->isZero()) array_pop($remainder);
129 }
130 return [new self($quotient), new self($remainder)];
131 }
132 public static function gcd(self $a, self $b): self {
133 while (!$b->isZero()) [$a, $b] = [$b, $a->divide($b)[1]];
134 return $a->monic();
135 }
136 public static function extendedGcd(self $a, self $b): array {
137 $r0 = $a; $r1 = $b; $s0 = self::one(); $s1 = self::zero(); $t0 = self::zero(); $t1 = self::one();
138 while (!$r1->isZero()) {
139 [$quotient, $remainder] = $r0->divide($r1);
140 [$r0, $r1] = [$r1, $remainder];
141 [$s0, $s1] = [$s1, $s0->sub($quotient->mul($s1))];
142 [$t0, $t1] = [$t1, $t0->sub($quotient->mul($t1))];
143 }
144 return $r0->isZero() ? [$r0, $s0, $t0]
145 : [$r0->monic(), $s0->divideScalar($r0->leading()), $t0->divideScalar($r0->leading())];
146 }
147 public function content(): Fraction {
148 $denominator = 1; $numerator = 0;
149 foreach ($this->coefficients as $value) {
150 $denominator = intdiv($denominator * $value->d, integerGcd($denominator, $value->d));
151 $numerator = integerGcd($numerator, $value->n);
152 }
153 return new Fraction($this->leading()->n < 0 ? -$numerator : $numerator, $denominator);
154 }
155 public function primitive(): self { return $this->divideScalar($this->content()); }
156 public function squareFreePart(): self {
157 return $this->degree() < 1 ? $this : $this->divide(self::gcd($this, $this->derivative()))[0];
158 }
159 public function format(string $variable = 'x'): string {
160 $result = '';
161 for ($i = $this->degree(); $i >= 0; $i--) {
162 $value = $this->at($i);
163 if ($value->isZero()) continue;
164 $result .= $result === '' ? ($value->n < 0 ? '-' : '') : ($value->n < 0 ? ' - ' : ' + ');
165 $magnitude = $value->absolute();
166 if ($i === 0 || !$magnitude->equals(Fraction::one())) $result .= $magnitude;
167 if ($i > 0) $result .= $variable;
168 if ($i > 1) $result .= '^' . $i;
169 }
170 return $result === '' ? '0' : $result;
171 }

187enum NodeKind { case Number; case Variable; case Name; case Negative; case Add; case Subtract; case Multiply; case Divide; case Power; case Call; }
188final readonly class Expression {
189 public function __construct(
190 public NodeKind $kind,
191 public ?Polynomial $literal = null,
192 public string $name = '',
193 public int $count = 0,
194 public ?self $left = null,
195 public ?self $right = null,
196 ) {}
197 public function evaluate(array $storage): Polynomial {
198 switch ($this->kind) {
199 case NodeKind::Number: return $this->literal;
200 case NodeKind::Variable: return Polynomial::variable();
201 case NodeKind::Name:
202 $value = $storage[$this->name] ?? throw new CommandError('ERROR UNKNOWN ' . $this->name);
203 return $value->derivative($this->count);
204 case NodeKind::Negative: return $this->left->evaluate($storage)->neg();
205 case NodeKind::Power:
206 $value = $this->left->evaluate($storage);
207 checkPower($value, $this->count);
208 return $value->power($this->count);
209 default: break;
210 }
211 $a = $this->left->evaluate($storage);
212 $b = $this->right->evaluate($storage);
213 switch ($this->kind) {
214 case NodeKind::Add: return $a->add($b);
215 case NodeKind::Subtract: return $a->sub($b);
216 case NodeKind::Multiply: checkProduct($a, $b); return $a->mul($b);
217 case NodeKind::Call: checkComposition($a, $b); return $a->compose($b);
218 default:
219 checkDivisor($b);
220 if ($b->degree() >= 1) throw new CommandError('ERROR NONCONSTANT DIVISOR');
221 return $a->divideScalar($b->at(0));
222 }
223 }

229final class Parser {
230 private array $tokens = [];
231 private int $position = 0;
232 private static function digit(string $ch): bool { return ord($ch) >= 48 && ord($ch) <= 57; }
233 private static function upper(string $ch): bool { return ord($ch) >= 65 && ord($ch) <= 90; }
234 public function __construct(string $source) {
235 $length = strlen($source);
236 for ($i = 0; $i < $length;) {
237 if ($source[$i] === ' ') { $i++; continue; }
238 $start = $i++;
239 if (self::digit($source[$start])) {
240 $kind = TokenKind::Number;
241 while ($i < $length && self::digit($source[$i])) $i++;
242 } elseif (self::upper($source[$start])) {
243 $kind = TokenKind::Name;
244 while ($i < $length && (self::upper($source[$i]) || self::digit($source[$i]))) $i++;
245 } else $kind = TokenKind::Symbol;
246 $this->tokens[] = new Token($kind, substr($source, $start, $i - $start), $start + 1);
247 }
248 $this->tokens[] = new Token(TokenKind::End, '', $length + 1);
249 }
250 private function current(): Token { return $this->tokens[$this->position]; }
251 private function at(string $symbol): bool { return $this->current()->kind === TokenKind::Symbol && $this->current()->text === $symbol; }
252 private function syntax(): never {
253 throw new CommandError('ERROR SYNTAX AT ' . ($this->current()->kind === TokenKind::End ? 'END' : $this->current()->position));
254 }
255 private function expect(string $symbol): void {
256 if (!$this->at($symbol)) $this->syntax();
257 $this->position++;
258 }
259 public function parse(): Expression {
260 $result = $this->expression();
261 if ($this->current()->kind !== TokenKind::End) $this->syntax();
262 return $result;
263 }
264 private function expression(): Expression {
265 $negative = $this->at('-');
266 if ($negative) $this->position++;
267 $result = $this->term();
268 if ($negative) $result = new Expression(NodeKind::Negative, left: $result);
269 while ($this->at('+') || $this->at('-')) {
270 $kind = $this->at('+') ? NodeKind::Add : NodeKind::Subtract;
271 $this->position++;
272 $right = $this->term();
273 $result = new Expression($kind, left: $result, right: $right);
274 }
275 return $result;
276 }
277 private function term(): Expression {
278 $result = $this->factor();
279 while (true) {
280 $kind = NodeKind::Multiply;
281 if ($this->at('*') || $this->at('/')) {
282 if ($this->at('/')) $kind = NodeKind::Divide;
283 $this->position++;
284 } elseif (!$this->at('x') && !$this->at('(') && $this->current()->kind !== TokenKind::Name) break;
285 $right = $this->factor();
286 $result = new Expression($kind, left: $result, right: $right);
287 }
288 return $result;
289 }
290 private function factor(): Expression {
291 $result = $this->primary();
292 if ($this->at('^')) {
293 $this->position++;
294 if ($this->current()->kind !== TokenKind::Number) $this->syntax();
295 $exponent = (int)$this->current()->text;
296 $this->position++;
297 $result = new Expression(NodeKind::Power, count: $exponent, left: $result);
298 }
299 return $result;
300 }
301 private function primary(): Expression {
302 $token = $this->current();
303 if ($token->kind === TokenKind::Number) {
304 $this->position++;
305 return new Expression(NodeKind::Number, literal: new Polynomial([new Fraction((int)$token->text)]));
306 }
307 if ($this->at('x')) { $this->position++; return new Expression(NodeKind::Variable); }
308 if ($token->kind === TokenKind::Name) {
309 $this->position++;
310 $order = 0;
311 while ($this->at("'")) { $order++; $this->position++; }
312 $reference = new Expression(NodeKind::Name, name: $token->text, count: $order);
313 if (!$this->at('(')) return $reference;
314 $this->position++;
315 $argument = $this->expression();
316 $this->expect(')');
317 return new Expression(NodeKind::Call, left: $reference, right: $argument);
318 }
319 if ($this->at('(')) {
320 $this->position++;
321 $result = $this->expression();
322 $this->expect(')');
323 return $result;
324 }
325 $this->syntax();
326 }
327}

350function rationalRoots(Polynomial $value): array {
351 $content = $value->content();
352 $remaining = $value->divideScalar($content);
353 $zeros = 0;
354 while ($remaining->at(0)->isZero()) {
355 $remaining = new Polynomial(array_slice($remaining->coefficients, 1));
356 $zeros++;
357 }
358 $divisors = static function(int $number): array {
359 $result = [];
360 for ($i = 1; $i <= $number; $i++) if ($number % $i === 0) $result[] = $i;
361 return $result;
362 };
363 $numerators = $divisors(abs($remaining->at(0)->n));
364 $denominators = $divisors($remaining->leading()->n);
365 $candidates = [];
366 foreach ($numerators as $numerator) {
367 foreach ($denominators as $denominator) {
368 if (integerGcd($numerator, $denominator) !== 1) continue;
369 $candidates[] = new Fraction($numerator, $denominator);
370 $candidates[] = new Fraction(-$numerator, $denominator);
371 }
372 }
373 usort($candidates, static fn(Fraction $a, Fraction $b): int => $a->compare($b));
374 $roots = [];
375 foreach ($candidates as $candidate) {
376 $count = 0;
377 $divisor = new Polynomial([new Fraction(-$candidate->n), new Fraction($candidate->d)]);
378 while ($remaining->degree() >= 1 && $remaining->evaluate($candidate)->isZero()) {
379 $remaining = $remaining->divide($divisor)[0];
380 $count++;
381 }
382 if ($count > 0) $roots[] = new Root($candidate, $count);
383 }
384 if ($zeros > 0) $roots[] = new Root(Fraction::zero(), $zeros);
385 usort($roots, static fn(Root $a, Root $b): int => $a->value->compare($b->value));
386 return [$content, $remaining, $roots];
387}

390 $result = [];
391 if ($value->degree() < 1) return $result;
392 $monic = $value->monic();

399 $c = $c->divide($y)[0];

405function sturmSequence(Polynomial $value): array {
406 $previous = $value->squareFreePart();
407 $current = $previous->derivative();
408 $result = [$previous];
409 while (!$current->isZero()) {
410 $result[] = $current;
411 $next = $previous->divide($current)[1]->neg();
412 $previous = $current;
413 $current = $next;
414 }
415 return $result;

417function variations(array $sequence, Endpoint $point): int {
418 $previous = 0; $changes = 0;
419 foreach ($sequence as $value) {
420 $sign = $point->infinity === 0 ? $value->evaluate($point->value)->sign() : $value->leading()->sign();
421 if ($point->infinity < 0 && $value->degree() % 2 !== 0) $sign = -$sign;
422 if ($sign === 0) continue;
423 if ($previous !== 0 && $previous !== $sign) $changes++;
424 $previous = $sign;
425 }
426 return $changes;
427}

433function isolate(Polynomial $value, int $precision): array {
434 $sequence = sturmSequence($value);
435 $first = $sequence[0];
436 $maximum = Fraction::zero();
437 for ($i = 0; $i < $first->degree(); $i++) {
438 $candidate = $first->at($i)->div($first->leading())->absolute();
439 if ($candidate->compare($maximum) > 0) $maximum = $candidate;
440 }
441 $bound = Fraction::one()->add($maximum);
442 $width = new Fraction(1, 1 << $precision);
443 $intervals = [];
444 $count = static fn(Fraction $left, Fraction $right): int => rootCount($sequence, new Endpoint(0, $left), new Endpoint(0, $right));
445 $examine = static function(Fraction $left, Fraction $right) use (&$examine, &$intervals, $count, $width): void {
446 $roots = $count($left, $right);
447 if ($roots >= 2) {
448 $middle = $left->add($right)->div(new Fraction(2));
449 $examine($left, $middle);
450 $examine($middle, $right);
451 } elseif ($roots === 1) {
452 while ($right->sub($left)->compare($width) > 0) {
453 $middle = $left->add($right)->div(new Fraction(2));
454 if ($count($left, $middle) >= 1) $right = $middle;
455 else $left = $middle;
456 }
457 $intervals[] = [$left, $right];
458 }
459 };
460 $examine($bound->neg(), $bound);
461 return $intervals;
462}
463function resultant(Polynomial $p, Polynomial $q): Fraction {
464 $result = Fraction::one();
465 while ($q->degree() >= 1) {
466 $remainder = $p->divide($q)[1];
467 if ($remainder->isZero()) return Fraction::zero();
468 $result = $result->mul($q->leading()->power($p->degree() - $remainder->degree()));
469 if ($p->degree() * $q->degree() % 2 !== 0) $result = $result->neg();
470 $p = $q;
471 $q = $remainder;
472 }
473 return $result->mul($q->at(0)->power($p->degree()));
474}
475function descartes(Polynomial $value, bool $negative): int {
476 $previous = 0; $count = 0;

480 if ($sign === 0) continue;
481 if ($previous !== 0 && $previous !== $sign) $count++;
482 $previous = $sign;
483 }

486function interpolate(array $values): Polynomial {
487 $count = intdiv(count($values), 2);
488 $points = []; $differences = [];
489 for ($i = 0; $i < $count; $i++) { $points[] = $values[2 * $i]; $differences[] = $values[2 * $i + 1]; }
490 for ($j = 1; $j < $count; $j++) {
491 for ($i = 0; $i < $j; $i++) {
492 if ($points[$j]->equals($points[$i])) throw new CommandError('ERROR DUPLICATE X ' . $points[$j]);
493 }
494 }
495 for ($j = 1; $j < $count; $j++) {
496 for ($i = $count - 1; $i >= $j; $i--) {
497 $differences[$i] = $differences[$i]->sub($differences[$i - 1])->div($points[$i]->sub($points[$i - $j]));
498 }
499 }
500 $result = Polynomial::zero();
501 for ($i = $count - 1; $i >= 0; $i--) {
502 $result = $result->mul(new Polynomial([$points[$i]->neg(), Fraction::one()]))->add(new Polynomial([$differences[$i]]));
503 }
504 return $result;
505}
506function special(string $kind, int $degree): Polynomial {
507 if (!in_array($kind, ['chebyshev', 'chebyshev2', 'legendre', 'hermite', 'laguerre'], true)) {
508 throw new CommandError('ERROR UNKNOWN KIND ' . $kind);
509 }
510 if ($degree > 30) throw new CommandError('ERROR DEGREE LIMIT');
511 $previous = Polynomial::one();
512 if ($degree === 0) return $previous;
513 $current = match ($kind) {
514 'chebyshev2', 'hermite' => new Polynomial([Fraction::zero(), new Fraction(2)]),
515 'laguerre' => new Polynomial([Fraction::one(), new Fraction(-1)]), default => Polynomial::variable(),
516 };
517 for ($j = 1; $j < $degree; $j++) {
518 $factor = match ($kind) {
519 'legendre' => new Polynomial([Fraction::zero(), new Fraction(2 * $j + 1)]),
520 'laguerre' => new Polynomial([new Fraction(2 * $j + 1), new Fraction(-1)]),
521 default => new Polynomial([Fraction::zero(), new Fraction(2)]),
522 };
523 $scale = $kind === 'legendre' || $kind === 'laguerre' ? $j : ($kind === 'hermite' ? 2 * $j : 1);
524 $next = $factor->mul($current)->sub($previous->scale(new Fraction($scale)));
525 if ($kind === 'legendre' || $kind === 'laguerre') $next = $next->scale(new Fraction(1, $j + 1));
526 $previous = $current;
527 $current = $next;
528 }
529 return $current;
530}
531
532final class Calculator {
533 private array $storage = [];
534 private function lookup(string $name): Polynomial {
535 return $this->storage[$name] ?? throw new CommandError('ERROR UNKNOWN ' . $name);
536 }
537 private static function nameArguments(string $command): array {
538 return match ($command) {
539 'PRINT', 'DELETE', 'DEGREE', 'AREA', 'TAYLOR', 'EVAL', 'ROOTS', 'FACTOR',
540 'SQFREE', 'DESCARTES', 'STURM', 'ISOLATE', 'DISCRIM', 'POWERSUM' => [1],
541 'ADD', 'SUB', 'MUL', 'GCD', 'LCM', 'INVMOD', 'COMPOSE' => [2, 3],
542 'POW', 'DERIV', 'INTEG', 'TANGENT' => [2], 'DIV' => [3, 4], 'XGCD' => [4, 5],
543 'CRT' => [2, 3, 4, 5], 'MODPOW' => [2, 4], 'RESULTANT' => [1, 2], default => [],
544 };
545 }
546 public function execute(string $line): array {
547 $output = []; $pending = []; $deletion = null;
548 $save = static function(string $name, Polynomial $value) use (&$output, &$pending): void {
549 $pending[] = [$name, $value];
550 $output[] = $name . ' = ' . $value->format();
551 };
552 if (str_starts_with($line, 'DEF ')) {
553 $definition = explode(' ', $line, 3);
554 $expression = (new Parser($definition[2]))->parse();
555 $save($definition[1], $expression->evaluate($this->storage));
556 } else {
557 $words = explode(' ', $line);
558 $command = $words[0];
559 if ($command === 'DIV' && $words[1] === $words[2] || $command === 'XGCD' &&
560 ($words[1] === $words[2] || $words[1] === $words[3] || $words[2] === $words[3])) {
561 throw new CommandError('ERROR SAME NAME');
562 }
563 $operands = [];
564 foreach (self::nameArguments($command) as $index) $operands[] = $this->lookup($words[$index]);
565 $literalArguments = match ($command) {
566 'AREA' => [2, 3], 'TANGENT' => [3], 'TAYLOR' => [2],
567 'EVAL', 'INTERP' => range(2, count($words) - 1), default => [],
568 };
569 $literals = [];
570 foreach ($literalArguments as $index) $literals[] = Fraction::parse($words[$index]);
571 $left = $right = null;
572 if ($command === 'STURM') {
573 $left = Endpoint::parse($words[2]);
574 $right = Endpoint::parse($words[3]);
575 }
576 $a = $operands[0] ?? Polynomial::zero();
577 $b = $operands[1] ?? Polynomial::zero();
578 switch ($command) {
579 case 'PRINT': $output[] = $words[1] . ' = ' . $a->format(); break;
580 case 'DELETE': $deletion = $words[1]; $output[] = 'deleted ' . $words[1]; break;
581 case 'LIST':
582 $sorted = $this->storage;
583 ksort($sorted, SORT_STRING);
584 foreach ($sorted as $name => $value) $output[] = $name . ' = ' . $value->format();
585 if ($sorted === []) $output[] = 'no polynomials';
586 break;
587 case 'DEGREE': $output[] = 'deg(' . $words[1] . ') = ' . ($a->isZero() ? '-inf' : $a->degree()); break;
588 case 'ADD': $save($words[1], $a->add($b)); break;
589 case 'SUB': $save($words[1], $a->sub($b)); break;
590 case 'MUL': checkProduct($a, $b); $save($words[1], $a->mul($b)); break;
591 case 'POW':
592 $exponent = (int)$words[3];
593 checkPower($a, $exponent);
594 $save($words[1], $a->power($exponent));
595 break;
596 case 'DIV':
597 checkDivisor($b);
598 [$quotient, $remainder] = $a->divide($b);
599 $save($words[1], $quotient);
600 $save($words[2], $remainder);
601 break;
602 case 'GCD': $save($words[1], Polynomial::gcd($a, $b)); break;
603 case 'XGCD':
604 [$g, $s, $t] = Polynomial::extendedGcd($a, $b);
605 $save($words[1], $g); $save($words[2], $s); $save($words[3], $t);
606 break;
607 case 'LCM':
608 if ($a->isZero() || $b->isZero()) $save($words[1], Polynomial::zero());
609 else {
610 $gcd = Polynomial::gcd($a, $b);
611 if ($a->degree() + $b->degree() - $gcd->degree() > 30) throw new CommandError('ERROR DEGREE LIMIT');
612 $save($words[1], $a->divide($gcd)[0]->mul($b)->monic());
613 }
614 break;
615 case 'INVMOD':
616 checkDivisor($b); checkPositiveDegree($b);
617 [$g, $s] = Polynomial::extendedGcd($a, $b);
618 if ($g->degree() !== 0 || !$g->at(0)->equals(Fraction::one())) throw new CommandError('ERROR NOT COPRIME');
619 $save($words[1], $s->divide($b)[1]);
620 break;
621 case 'CRT':
622 $a2 = $operands[2]; $m2 = $operands[3];
623 checkDivisor($b); checkDivisor($m2);
624 if ($b->degree() + $m2->degree() > 30) throw new CommandError('ERROR DEGREE LIMIT');
625 [$g, $s] = Polynomial::extendedGcd($b, $m2);
626 if ($g->degree() !== 0 || !$g->at(0)->equals(Fraction::one())) throw new CommandError('ERROR NOT COPRIME');
627 $r1 = $a->divide($b)[1]; $r2 = $a2->divide($m2)[1];
628 $u = $r2->sub($r1)->mul($s)->divide($m2)[1];
629 $save($words[1], $r1->add($b->mul($u)));
630 break;
631 case 'MODPOW':
632 checkDivisor($b);
633 if (2 * $b->degree() - 2 > 30) throw new CommandError('ERROR DEGREE LIMIT');
634 $x = Polynomial::one()->divide($b)[1]; $y = $a->divide($b)[1]; $exponent = (int)$words[3];
635 while ($exponent > 0) {
636 if ($exponent % 2 !== 0) $x = $x->mul($y)->divide($b)[1];
637 $exponent = intdiv($exponent, 2);
638 if ($exponent !== 0) $y = $y->mul($y)->divide($b)[1];
639 }
640 $save($words[1], $x);
641 break;
642 case 'DERIV': $save($words[1], $a->derivative((int)$words[3])); break;
643 case 'INTEG':
644 if ($a->degree() === 30) throw new CommandError('ERROR DEGREE LIMIT');
645 $save($words[1], $a->integral());
646 break;
647 case 'AREA':
648 $integral = $a->integral();
649 $upper = $integral->evaluate($literals[1]); $lower = $integral->evaluate($literals[0]);
650 $output[] = 'integral(' . $words[1] . ', ' . $literals[0] . ', ' . $literals[1] . ') = ' . $upper->sub($lower);
651 break;
652 case 'TANGENT':
653 $point = $literals[0]; $value = $a->evaluate($point); $slope = $a->derivative()->evaluate($point);
654 $save($words[1], new Polynomial([$value->sub($slope->mul($point)), $slope]));
655 break;
656 case 'TAYLOR':
657 $point = $literals[0];
658 $shifted = $a->compose(new Polynomial([$point, Fraction::one()]));
659 $variable = $point->isZero() ? 'x' : ($point->n > 0 ? '(x - ' . $point . ')' : '(x + ' . $point->neg() . ')');
660 $output[] = 'taylor(' . $words[1] . ', ' . $point . ') = ' . $shifted->format($variable);
661 break;
662 case 'COMPOSE': checkComposition($a, $b); $save($words[1], $a->compose($b)); break;
663 case 'EVAL':
664 foreach ($literals as $point) $output[] = $words[1] . '(' . $point . ') = ' . $a->evaluate($point);
665 break;
666 case 'ROOTS':
667 checkNonzero($a);
668 [, , $roots] = rationalRoots($a);
669 $parts = array_map(static fn(Root $root): string => $root->value . ' (' . $root->multiplicity . ')', $roots);
670 $output[] = 'roots(' . $words[1] . ') = ' . ($parts === [] ? 'none' : implode(', ', $parts));
671 break;
672 case 'FACTOR':
673 checkNonzero($a);
674 [$content, $remaining, $roots] = rationalRoots($a);
675 $result = 'factor(' . $words[1] . ') = ' . $content;
676 foreach ($roots as $root) {
677 $factor = $root->value->isZero() ? 'x'
678 : '(' . (new Polynomial([new Fraction(-$root->value->n), new Fraction($root->value->d)]))->format() . ')';
679 $result .= ' * ' . $factor;
680 if ($root->multiplicity >= 2) $result .= '^' . $root->multiplicity;
681 }
682 if ($remaining->degree() >= 1) $result .= ' * (' . $remaining->format() . ')';
683 $output[] = $result;
684 break;
685 case 'SQFREE':
686 checkNonzero($a);
687 $factors = squareFreeFactors($a);
688 $result = 'sqfree(' . $words[1] . ') = ' . $a->content();
689 foreach ($factors as $factor) {
690 $value = $factor->value;
691 $isX = $value->degree() === 1 && $value->at(0)->isZero() && $value->at(1)->equals(Fraction::one());
692 $result .= ' * ' . ($isX ? 'x' : '(' . $value->format() . ')');
693 if ($factor->multiplicity >= 2) $result .= '^' . $factor->multiplicity;
694 }
695 $output[] = $result;
696 break;
697 case 'DESCARTES':
698 checkNonzero($a);
699 $output[] = 'descartes(' . $words[1] . ') = positive ' . descartes($a, false) . ', negative ' . descartes($a, true);
700 break;
701 case 'STURM':
702 checkNonzero($a);
703 if ($left->compare($right) >= 0) throw new CommandError('ERROR BAD INTERVAL');
704 $sequence = sturmSequence($a);
705 $output[] = 'sturm(' . $words[1] . ', ' . $left . ', ' . $right . ') = ' . rootCount($sequence, $left, $right);
706 break;
707 case 'ISOLATE':
708 checkNonzero($a);
709 $precision = (int)$words[2]; $intervals = isolate($a, $precision);
710 $parts = array_map(static fn(array $interval): string => '(' . $interval[0] . ', ' . $interval[1] . ']', $intervals);
711 $output[] = 'isolate(' . $words[1] . ', ' . $precision . ') = ' . ($parts === [] ? 'none' : implode(', ', $parts));
712 break;
713 case 'RESULTANT':
714 checkNonzero($a); checkNonzero($b);
715 $output[] = 'res(' . $words[1] . ', ' . $words[2] . ') = ' . resultant($a, $b);
716 break;
717 case 'DISCRIM':
718 checkNonzero($a); checkPositiveDegree($a);
719 $value = resultant($a, $a->derivative())->div($a->leading());
720 if (intdiv($a->degree() * ($a->degree() - 1), 2) % 2 !== 0) $value = $value->neg();
721 $output[] = 'disc(' . $words[1] . ') = ' . $value;
722 break;
723 case 'POWERSUM':
724 checkNonzero($a); checkPositiveDegree($a);
725 $count = (int)$words[2]; $monic = $a->monic(); $sums = array_fill(0, $count + 1, Fraction::zero());
726 for ($j = 1; $j <= $count; $j++) {
727 $value = $j <= $a->degree() ? (new Fraction($j))->mul($monic->at($a->degree() - $j)) : Fraction::zero();
728 for ($i = 1; $i <= min($j - 1, $a->degree()); $i++) $value = $value->add($monic->at($a->degree() - $i)->mul($sums[$j - $i]));
729 $sums[$j] = $value->neg();
730 }
731 $output[] = 'powersum(' . $words[1] . ') = ' . implode(', ', array_slice($sums, 1));
732 break;
733 case 'INTERP': $save($words[1], interpolate($literals)); break;
734 case 'SPECIAL': $save($words[1], special($words[2], (int)$words[3])); break;
735 }
736 }
737 if ($deletion !== null) unset($this->storage[$deletion]);
738 foreach ($pending as [$name, $value]) $this->storage[$name] = $value;
739 return $output;
740 }
741}

743$count = (int)trim(fgets(STDIN));
744$calculator = new Calculator();
745for ($i = 0; $i < $count; $i++) {
746 $line = rtrim(fgets(STDIN), "\r\n");
747 try {
748 foreach ($calculator->execute($line) as $output) echo $output, "\n";
749 } catch (CommandError $error) {
750 echo $error->getMessage(), "\n";
751 }
