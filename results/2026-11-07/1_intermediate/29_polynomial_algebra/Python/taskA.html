65def multiply(first, second):
66 if not first or not second:
67 return []
68 product = [Fraction(0)] * (len(first) + len(second) - 1)
69 for i, left in enumerate(first):
70 for j, right in enumerate(second):
71 product[i + j] += left * right
72 return product

75def divide(dividend, divisor):
76 result = [Fraction(0)] * max(len(dividend) - len(divisor) + 1, 0)
77 remainder = list(dividend)
78 while remainder and len(remainder) >= len(divisor):
79 factor = remainder[-1] / divisor[-1]
80 shift = len(remainder) - len(divisor)
81 result[shift] = factor
82 for j, value in enumerate(divisor):
83 remainder[shift + j] -= factor * value
84 trim(remainder)
85 return result, remainder

138def extended_gcd(first, second):
139 r0, r1 = first, second
140 s0, s1 = constant(1), []
141 t0, t1 = [], constant(1)
142 while r1:
143 q, rest = divide(r0, r1)
144 r0, r1 = r1, rest
145 s0, s1 = s1, subtract(s0, multiply(q, s1))
146 t0, t1 = t1, subtract(t0, multiply(q, t1))
147 if not r0:
148 return r0, s0, t0
149 return monic(r0), divide_scalar(s0, r0[-1]), divide_scalar(t0, r0[-1])

152def content(polynomial):
153 denominator_lcm = 1
154 numerator_gcd = 0
155 for value in polynomial:
156 denominator_lcm = math.lcm(denominator_lcm, value.denominator)
157 numerator_gcd = math.gcd(numerator_gcd, abs(value.numerator))
158 result = Fraction(numerator_gcd, denominator_lcm)
159 return result if polynomial[-1] > 0 else -result

226def polynomial_text(polynomial, variable="x"):
227 if not polynomial:
228 return "0"
229 parts = []

231 value = polynomial[exponent]
232 if value == 0:

239 parts.append(fraction_text(abs(value)))

242 elif exponent >= 2:
243 parts.append(f"{variable}^{exponent}")

277def modular_power(base, exponent, modulus):
278 result = modulo(constant(1), modulus)
279 square = modulo(base, modulus)
280 while exponent > 0:
281 if exponent % 2 == 1:
282 result = modulo(multiply(result, square), modulus)
283 exponent //= 2
284 if exponent:
285 square = modulo(multiply(square, square), modulus)
286 return result

293def root_candidates(constant_term, leading):
294 candidates = []
295 for numerator in divisors(constant_term):
296 for denominator in divisors(leading):
297 if math.gcd(numerator, denominator) == 1:
298 candidates.append(Fraction(numerator, denominator))
299 candidates.append(Fraction(-numerator, denominator))

303def rational_roots(polynomial):
304 remaining = primitive(polynomial)
305 zeros = 0
306 while remaining[0] == 0:
307 remaining = remaining[1:]
308 zeros += 1
309 roots = []
310 for candidate in root_candidates(abs(remaining[0].numerator), remaining[-1].numerator):
311 factor = linear(candidate.denominator, -candidate.numerator)
312 multiplicity = 0
313 while degree(remaining) >= 1 and evaluate(remaining, candidate) == 0:
314 remaining = quotient(remaining, factor)
315 multiplicity += 1
316 if multiplicity:
317 roots.append((candidate, multiplicity))
318 if zeros:
319 roots.append((Fraction(0), zeros))
320 roots.sort()
321 return roots, remaining

324def square_free_factors(polynomial):
325 factors = []
326 if degree(polynomial) == 0:
327 return factors
328 base = monic(polynomial)
329 common = polynomial_gcd(base, derivative(base))
330 rest = quotient(base, common)
331 multiplicity = 1
332 while degree(rest) >= 1:
333 shared = polynomial_gcd(rest, common)
334 factor = quotient(rest, shared)
335 if degree(factor) >= 1:
336 factors.append((primitive(factor), multiplicity))
337 common = quotient(common, shared)
338 rest = shared
339 multiplicity += 1
340 return factors

343def descartes_counts(polynomial):
344 positive = []
345 negative = []
346 for exponent in range(degree(polynomial), -1, -1):
347 value = sign(polynomial[exponent])
348 positive.append(value)

353def sturm_sequence(polynomial):
354 sequence = [square_free_part(polynomial)]
355 following = derivative(sequence[0])
356 while following:
357 sequence.append(following)
358 following = negate(modulo(sequence[-2], sequence[-1]))

389def examine(sequence, low, high, width, intervals):
390 count = root_count(sequence, low, high)
391 if count >= 2:
392 middle = (low + high) / 2
393 examine(sequence, low, middle, width, intervals)
394 examine(sequence, middle, high, width, intervals)
395 elif count == 1:
396 while high - low > width:
397 middle = (low + high) / 2
398 if root_count(sequence, low, middle) >= 1:
399 high = middle
400 else:
401 low = middle
402 intervals.append((low, high))

405def isolating_intervals(polynomial, precision):
406 sequence = sturm_sequence(polynomial)
407 base = sequence[0]
408 largest = max((abs(value / base[-1]) for value in base[:-1]), default=Fraction(0))
409 beta = 1 + largest
410 intervals = []
411 examine(sequence, -beta, beta, Fraction(1, 2 ** precision), intervals)
412 return intervals

422def resultant(first, second):
423 result = Fraction(1)
424 p, q = first, second
425 while degree(q) >= 1:
426 rest = modulo(p, q)
427 if not rest:
428 return Fraction(0)
429 result *= fraction_power(q[-1], degree(p) - degree(rest))
430 if degree(p) * degree(q) % 2 == 1:
431 result = -result
432 p, q = q, rest
433 return result * fraction_power(q[0], degree(p))

436def power_sums(polynomial, count):
437 size = degree(polynomial)
438 normalized = monic(polynomial)
439 sums = []
440 for j in range(1, count + 1):
441 total = j * normalized[size - j] if j <= size else Fraction(0)
442 for i in range(1, min(j - 1, size) + 1):
443 total += normalized[size - i] * sums[j - i - 1]
444 sums.append(-total)
445 return sums

448def interpolate(xs, ys):
449 count = len(xs)
450 differences = list(ys)
451 for j in range(1, count):
452 for i in range(count - 1, j - 1, -1):
453 differences[i] = (differences[i] - differences[i - 1]) / (xs[i] - xs[i - j])
454 result = []
455 for i in range(count - 1, -1, -1):
456 result = add(multiply(result, linear(1, -xs[i])), constant(differences[i]))
457 return result

460def special_step(kind, j, previous, current):
461 if kind in ("chebyshev", "chebyshev2"):
462 return subtract(multiply(linear(2, 0), current), previous)
463 if kind == "legendre":
464 combined = subtract(multiply(linear(2 * j + 1, 0), current), scale(previous, j))
465 return scale(combined, Fraction(1, j + 1))
466 if kind == "hermite":
467 return subtract(multiply(linear(2, 0), current), scale(previous, 2 * j))
468 combined = subtract(multiply(linear(-1, 2 * j + 1), current), scale(previous, j))
469 return scale(combined, Fraction(1, j + 1))

472def special_polynomial(kind, order):
473 previous = constant(1)
474 if order == 0:
475 return previous
476 current = linear(*SPECIAL_KINDS[kind])
477 for j in range(1, order):
478 previous, current = current, special_step(kind, j, previous, current)
479 return current

490 tokens = []
491 index = 0
492 while index < len(expression):
493 start = index
494 character = expression[index]
495 index += 1
496 if character == " ":

499 while index < len(expression) and expression[index].isdigit():
500 index += 1
501 kind = "number"
502 elif character.isupper():
503 while index < len(expression) and (

506 index += 1
507 kind = "name"
508 else:
509 kind = character
510 tokens.append(Token(kind, expression[start:index], start + 1))

554class Term:
555 def __init__(self, first, rest):
556 self.first = first
557 self.rest = rest
558
559 def evaluate(self, polynomials):
560 value = self.first.evaluate(polynomials)
561 for operator, factor in self.rest:
562 right = factor.evaluate(polynomials)
563 if operator == "*":
564 value = checked_product(value, right)
565 elif not right:
566 raise CommandError("DIVISION BY ZERO")
567 elif degree(right) >= 1:
568 raise CommandError("NONCONSTANT DIVISOR")
569 else:
570 value = divide_scalar(value, right[0])
571 return value

593class Parser:
594 def __init__(self, tokens):
595 self.tokens = tokens
596 self.index = 0
597
598 def peek(self):
599 if self.index < len(self.tokens):
600 return self.tokens[self.index].kind
601 return None
602
603 def error(self):
604 if self.index < len(self.tokens):
605 return CommandError(f"SYNTAX AT {self.tokens[self.index].position}")
606 return CommandError("SYNTAX AT END")
607
608 def advance(self):
609 token = self.tokens[self.index]
610 self.index += 1
611 return token
612
613 def accept(self, kind):
614 if self.peek() == kind:
615 self.index += 1
616 return True
617 return False
618
619 def take(self, kind):
620 if self.peek() != kind:
621 raise self.error()
622 return self.advance()
623
624 def parse(self):
625 expression = self.expression()
626 if self.peek() is not None:
627 raise self.error()
628 return expression
629
630 def expression(self):
631 negated = self.accept("-")
632 first = self.term()
633 rest = []
634 while self.peek() in ("+", "-"):
635 operator = self.advance().kind
636 rest.append((operator, self.term()))
637 return Expression(negated, first, rest)
638
639 def term(self):
640 first = self.factor()
641 rest = []
642 while self.peek() in PRODUCT_STARTS:
643 operator = "*"
644 if self.peek() in ("*", "/"):
645 operator = self.advance().kind
646 rest.append((operator, self.factor()))
647 return Term(first, rest)
648
649 def factor(self):
650 base = self.primary()
651 exponent = None
652 if self.accept("^"):
653 exponent = int(self.take("number").text)
654 return Factor(base, exponent)
655
656 def primary(self):
657 kind = self.peek()
658 if kind == "number":
659 return Constant(constant(int(self.advance().text)))
660 if kind == "x":
661 self.advance()
662 return Constant(linear(1, 0))
663 if kind == "name":
664 name = self.advance().text
665 primes = 0
666 while self.accept("'"):
667 primes += 1
668 reference = Reference(name, primes)
669 if not self.accept("("):
670 return reference
671 argument = self.expression()
672 self.take(")")
673 return Call(reference, argument)
674 if kind == "(":
675 self.advance()
676 inner = self.expression()
677 self.take(")")
678 return inner
679 raise self.error()

769 def extended_euclid(self, arguments):
770 targets = arguments[:3]
771 if len(set(targets)) < 3:
772 raise CommandError("SAME NAME")
773 left, right = self.lookup(arguments[3]), self.lookup(arguments[4])
774 for target, result in zip(targets, extended_gcd(left, right)):
775 self.store(target, result)

777 def least_common_multiple(self, arguments):
778 target, first, second = arguments
779 left, right = self.lookup(first), self.lookup(second)
780 if not left or not right:
781 self.store(target, [])
782 return
783 common = polynomial_gcd(left, right)
784 if degree(left) + degree(right) - degree(common) > MAX_DEGREE:
785 raise CommandError("DEGREE LIMIT")
786 self.store(target, monic(multiply(quotient(left, common), right)))

788 def inverse_modulo(self, arguments):

791 require_divisor(modulus)

793 raise CommandError("DEGREE TOO LOW")

799 first, first_modulus, second, second_modulus = [self.lookup(name) for name in arguments[1:]]
800 require_divisor(first_modulus)
801 require_divisor(second_modulus)
802 if degree(first_modulus) + degree(second_modulus) > MAX_DEGREE:
803 raise CommandError("DEGREE LIMIT")
804 cofactor = bezout_coefficient(first_modulus, second_modulus)
805 first_residue = modulo(first, first_modulus)
806 second_residue = modulo(second, second_modulus)
807 difference = subtract(second_residue, first_residue)
808 correction = modulo(multiply(difference, cofactor), second_modulus)
809 self.store(target, add(first_residue, multiply(first_modulus, correction)))

816 raise CommandError("DEGREE LIMIT")
817 self.store(target, modular_power(base, int(exponent), modulus))

959 def execute(self, line):
960 if line.startswith("DEF "):
961 words = line.split(" ", 2)
962 else:
963 words = line.split(" ")
964 command, arguments = words[0], words[1:]
965 try:
966 if command in self.operations:
967 self.combine(self.operations[command], arguments)
968 else:
969 self.handlers[command](arguments)
970 except CommandError as error:
971 self.emit(f"ERROR {error}")
