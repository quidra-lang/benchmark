38notation <- function(a, variable = "x") {
39 terms <- character()
40 for (i in rev(seq_along(a))) {
41 value <- a[i]
42 if (value == ZERO) next
43 exponent <- i - 1L
44 coefficient <- if (exponent > 0L && abs(value) == ONE) "" else fraction(abs(value))
45 suffix <- if (exponent == 0L) "" else if (exponent == 1L) variable else paste0(variable, "^", exponent)
46 prefix <- if (!length(terms)) {
47 if (value < ZERO) "-" else ""
48 } else if (value < ZERO) " - " else " + "
49 terms <- c(terms, paste0(prefix, coefficient, suffix))
50 }
51 if (length(terms)) paste(terms, collapse = "") else "0"
52}

82divide <- function(a, b) {
83 quotient <- rep(ZERO, max(0L, length(a) - length(b) + 1L))
84 remainder <- a
85 while (length(remainder) > 0L && length(remainder) >= length(b)) {
86 offset <- length(remainder) - length(b)
87 factor <- leading(remainder) / leading(b)
88 quotient[offset + 1L] <- factor
89 for (j in seq_along(b)) remainder[offset + j] <- remainder[offset + j] - factor * b[j]
90 remainder <- trim(remainder)
91 }
92 list(quotient = trim(quotient), remainder = remainder)
93}

122extended_gcd <- function(a, b) {
123 s0 <- ONE
124 s1 <- EMPTY
125 t0 <- EMPTY
126 t1 <- ONE
127 while (length(b)) {
128 division <- divide(a, b)
129 next_s <- subtract(s0, multiply(division$quotient, s1))
130 next_t <- subtract(t0, multiply(division$quotient, t1))
131 a <- b
132 b <- division$remainder
133 s0 <- s1
134 s1 <- next_s
135 t0 <- t1
136 t1 <- next_t
137 }
138 if (length(a)) {
139 divisor <- leading(a)
140 a <- divide_scalar(a, divisor)
141 s0 <- divide_scalar(s0, divisor)
142 t0 <- divide_scalar(t0, divisor)
143 }
144 list(a, s0, t0)

146content <- function(a) {
147 multiple <- as.bigz(1L)
148 common <- as.bigz(0L)
149 for (i in seq_along(a)) {
150 d <- denominator(a[i])
151 multiple <- (multiple * d) %/% gcd.bigz(multiple, d)
152 common <- gcd.bigz(common, abs(numerator(a[i])))
153 }
154 if (leading(a) < ZERO) common <- -common
155 as.bigq(common, multiple)
156}

164rational_roots <- function(a) {
165 remaining <- primitive(a)
166 zeros <- 0L
167 while (remaining[1L] == ZERO) {
168 remaining <- remaining[-1L]
169 zeros <- zeros + 1L
170 }
171 numerator_limit <- as.integer(abs(remaining[1L]))
172 denominator_limit <- as.integer(leading(remaining))
173 keys <- integer()
174 for (s in divisors(numerator_limit)) {
175 for (t in divisors(denominator_limit)) {
176 if (gcd.bigz(as.bigz(s), as.bigz(t)) != 1L) next
177 key <- s * (denominator_limit %/% t)
178 keys <- c(keys, -key, key)
179 }
180 }
181 roots <- list()
182 for (key in sort(keys, method = "radix")) {
183 value <- as.bigq(key, denominator_limit)
184 factor <- as.bigq(c(-numerator(value), denominator(value)))
185 multiplicity <- 0L
186 while (degree(remaining) >= 1L && evaluate(remaining, value) == ZERO) {
187 remaining <- divide(remaining, factor)$quotient
188 multiplicity <- multiplicity + 1L
189 }
190 if (multiplicity > 0L) roots[[length(roots) + 1L]] <- list(value = value, multiplicity = multiplicity, key = key)
191 }
192 if (zeros > 0L) roots[[length(roots) + 1L]] <- list(value = ZERO, multiplicity = zeros, key = 0L)
193 if (length(roots)) roots <- roots[order(vapply(roots, function(root) root$key, integer(1L)), method = "radix")]
194 list(roots = roots, remaining = remaining)
195}

211squarefree_factors <- function(a) {
212 result <- fraction(content(a))
213 if (degree(a) == 0L) return(result)
214 normalized <- monic(a)
215 common <- polynomial_gcd(normalized, derivative(normalized))
216 remaining <- divide(normalized, common)$quotient
217 multiplicity <- 1L
218 while (degree(remaining) >= 1L) {
219 shared <- polynomial_gcd(remaining, common)
220 factor <- divide(remaining, shared)$quotient
221 if (degree(factor) >= 1L) {
222 text <- notation(primitive(factor))
223 if (text != "x") text <- paste0("(", text, ")")
224 result <- paste0(result, factor_text(text, multiplicity))
225 }
226 common <- divide(common, shared)$quotient
227 remaining <- shared
228 multiplicity <- multiplicity + 1L
229 }
230 result

269isolate <- function(a, precision) {
270 sequence <- sturm_sequence(a)
271 first <- sequence[[1L]]
272 beta <- if (degree(first) == 0L) ONE else ONE + max(abs(first[-length(first)] / leading(first)))
273 width <- as.bigq(1L, bitwShiftL(1L, precision))
274 intervals <- character()
275 count_roots <- function(low, high) variations(sequence, bound(low)) - variations(sequence, bound(high))
276 examine <- function(low, high) {
277 count <- count_roots(low, high)
278 if (count >= 2L) {
279 middle <- (low + high) / 2L
280 examine(low, middle)
281 examine(middle, high)
282 } else if (count == 1L) {
283 while (high - low > width) {
284 middle <- (low + high) / 2L
285 if (count_roots(low, middle) >= 1L) high <- middle else low <- middle
286 }
287 intervals <<- c(intervals, paste0("(", fraction(low), ", ", fraction(high), "]"))
288 }
289 }
290 examine(-beta, beta)
291 if (length(intervals)) paste(intervals, collapse = ", ") else "none"

293resultant <- function(a, b) {
294 result <- ONE
295 while (degree(b) >= 1L) {
296 remainder <- modulo(a, b)
297 if (!length(remainder)) return(ZERO)
298 result <- result * fraction_power(leading(b), degree(a) - degree(remainder))
299 if ((degree(a) * degree(b)) %% 2L) result <- -result
300 a <- b
301 b <- remainder
302 }
303 result * fraction_power(b[1L], degree(a))

305power_sums <- function(a, count) {
306 d <- degree(a)
307 normalized <- monic(a)
308 sums <- rep(ZERO, count)
309 for (j in seq_len(count)) {
310 value <- if (j <= d) j * normalized[d - j + 1L] else ZERO
311 for (i in seq_len(min(j - 1L, d))) value <- value + normalized[d - i + 1L] * sums[j - i]
312 sums[j] <- -value
313 }
314 paste(vapply(seq_along(sums), function(i) fraction(sums[i]), character(1L)), collapse = ", ")

316interpolate <- function(values) {
317 count <- length(values) %/% 2L
318 x <- values[seq.int(1L, length(values), by = 2L)]
319 differences <- values[seq.int(2L, length(values), by = 2L)]
320 for (j in seq_len(count)) {
321 if (any(x[j] == x[seq_len(j - 1L)])) fail(paste("DUPLICATE X", fraction(x[j])))
322 }
323 for (j in seq_len(count - 1L)) {
324 for (i in rev(seq.int(j + 1L, count))) {
325 differences[i] <- (differences[i] - differences[i - 1L]) / (x[i] - x[i - j])
326 }
327 }
328 result <- EMPTY
329 for (i in rev(seq_len(count))) result <- add(multiply(result, linear(-x[i])), constant(differences[i]))
330 result
331}
332special <- function(kind, count) {
333 if (!kind %in% c("chebyshev", "chebyshev2", "legendre", "hermite", "laguerre")) fail(paste("UNKNOWN KIND", kind))
334 degree_limit(count > 30L)
335 previous <- ONE
336 if (count == 0L) return(previous)
337 current <- switch(kind, chebyshev = X, chebyshev2 = scale(X, as.bigq(2L)), legendre = X,
338 hermite = scale(X, as.bigq(2L)), laguerre = as.bigq(c(1L, -1L)))
339 for (j in seq_len(count - 1L)) {
340 if (kind %in% c("chebyshev", "chebyshev2", "hermite")) {
341 factor <- if (kind == "hermite") 2L * j else 1L
342 next_value <- subtract(multiply(scale(X, as.bigq(2L)), current), scale(previous, as.bigq(factor)))
343 } else {
344 factor <- if (kind == "legendre") linear(ZERO, as.bigq(2L * j + 1L)) else linear(as.bigq(2L * j + 1L), -ONE)
345 next_value <- scale(subtract(multiply(factor, current), scale(previous, as.bigq(j))), as.bigq(1L, j + 1L))
346 }
347 previous <- current
348 current <- next_value
349 }
350 current
351}

354parse_definition <- function(source) {
355 positions <- gregexpr("[0-9]+|[A-Z][A-Z0-9]*|[^ ]", source, perl = TRUE)[[1L]]
356 words <- regmatches(source, list(positions))[[1L]]
357 position <- 1L
358 peek <- function() if (position <= length(words)) words[position] else ""
359 syntax_error <- function() fail(paste("SYNTAX AT", if (position <= length(words)) positions[position] else "END"))
360 take <- function() {
361 word <- peek()
362 position <<- position + 1L
363 word
364 }
365 expect <- function(word) {
366 if (peek() != word) syntax_error()
367 take()
368 }
369 parse_primary <- function() {
370 word <- peek()
371 if (grepl("^[0-9]+$", word)) return(node("number", value = as.integer(take())))
372 if (word == "x") {
373 take()
374 return(node("x"))
375 }
376 if (grepl("^[A-Z]", word)) {
377 name <- take()
378 count <- 0L
379 while (peek() == "'") {
380 take()
381 count <- count + 1L
382 }
383 result <- node("name", name = name, count = count)
384 if (peek() == "(") {
385 take()
386 argument <- parse_sum()
387 expect(")")
388 result <- node("call", left = result, right = argument)
389 }
390 return(result)
391 }
392 if (word == "(") {
393 take()
394 result <- parse_sum()
395 expect(")")
396 return(result)
397 }
398 syntax_error()
399 }
400 parse_factor <- function() {
401 result <- parse_primary()
402 if (peek() == "^") {
403 take()
404 if (!grepl("^[0-9]+$", peek())) syntax_error()
405 result <- node("power", left = result, exponent = as.integer(take()))
406 }
407 result
408 }
409 parse_term <- function() {
410 result <- parse_factor()
411 repeat {
412 word <- peek()
413 if (word %in% c("*", "/")) {
414 take()
415 } else if (word == "x" || word == "(" || grepl("^[A-Z]", word)) {
416 word <- "*"
417 } else break
418 right <- parse_factor()
419 result <- node(word, left = result, right = right)
420 }
421 result
422 }
423 parse_sum <- function() {
424 negative <- peek() == "-"
425 if (negative) take()
426 result <- parse_term()
427 if (negative) result <- node("negate", left = result)
428 while (peek() %in% c("+", "-")) {
429 operation <- take()
430 right <- parse_term()
431 result <- node(operation, left = result, right = right)
432 }
433 result
434 }
435 result <- parse_sum()
436 if (position <= length(words)) syntax_error()
437 result
438}

445evaluate_node <- function(tree) {
446 if (tree$kind == "number") return(constant(as.bigq(tree$value)))
447 if (tree$kind == "x") return(X)
448 if (tree$kind == "name") return(derivative(lookup(tree$name), tree$count))
449 left <- evaluate_node(tree$left)
450 if (tree$kind == "negate") return(-left)
451 if (tree$kind == "power") {
452 degree_limit(length(left) > 0L && degree(left) * tree$exponent > 30L)
453 return(power(left, tree$exponent))
454 }
455 right <- evaluate_node(tree$right)
456 if (tree$kind == "+") return(add(left, right))
457 if (tree$kind == "-") return(subtract(left, right))
458 if (tree$kind == "*") {
459 product_limit(left, right)
460 return(multiply(left, right))
461 }
462 if (tree$kind == "call") {
463 composition_limit(left, right)
464 return(compose(left, right))
465 }
466 if (!length(right)) fail("DIVISION BY ZERO")
467 if (degree(right) >= 1L) fail("NONCONSTANT DIVISOR")
468 divide_scalar(left, right[1L])
469}

479execute <- function(source) {
480 words <- strsplit(source, " ", fixed = TRUE)[[1L]]
481 command <- words[1L]
482 name <- words[2L]
483 pending <- list()
484 output <- character()
485 store <- function(destination, value) {
486 pending[[length(pending) + 1L]] <<- list(name = destination, value = value)
487 output <<- c(output, paste0(destination, " = ", notation(value)))
488 }
489 if (command == "DEF") {
490 syntax <- parse_definition(substr(source, nchar(name) + 6L, nchar(source)))
491 store(name, evaluate_node(syntax))
492 } else {
493 if ((command == "DIV" && words[2L] == words[3L]) ||
494 (command == "XGCD" && anyDuplicated(words[2L:4L]))) fail("SAME NAME")
495 operands <- lapply(REFERENCES[[command]], function(index) lookup(words[index]))
496 a <- if (length(operands) >= 1L) operands[[1L]] else EMPTY
497 b <- if (length(operands) >= 2L) operands[[2L]] else EMPTY
498 c <- if (length(operands) >= 3L) operands[[3L]] else EMPTY
499 d <- if (length(operands) >= 4L) operands[[4L]] else EMPTY
500 literal_words <- switch(command, AREA = words[3L:4L], STURM = words[3L:4L], TANGENT = words[4L],
501 TAYLOR = words[3L], EVAL = words[-(1L:2L)], INTERP = words[-(1L:2L)], character())
502 if (command == "STURM") {
503 endpoints <- lapply(literal_words, parse_bound)
504 } else values <- if (length(literal_words)) c_bigq(lapply(literal_words, literal)) else EMPTY
505 switch(command,
506 PRINT = { output <- paste0(name, " = ", notation(a)) },
507 DELETE = {
508 rm(list = name, envir = polynomials)
509 output <- paste("deleted", name)
510 },
511 LIST = {
512 names <- sort(ls(polynomials, all.names = TRUE), method = "radix")
513 output <- if (length(names)) vapply(names, function(key) paste0(key, " = ", notation(lookup(key))), character(1L)) else "no polynomials"
514 },
515 DEGREE = { output <- paste0("deg(", name, ") = ", if (length(a)) degree(a) else "-inf") },
516 ADD = { store(name, add(a, b)) },
517 SUB = { store(name, subtract(a, b)) },
518 MUL = { product_limit(a, b); store(name, multiply(a, b)) },
519 POW = {
520 exponent <- as.integer(words[4L])
521 degree_limit(length(a) > 0L && degree(a) * exponent > 30L)
522 store(name, power(a, exponent))
523 },
524 DIV = {
525 if (!length(b)) fail("DIVISION BY ZERO")
526 division <- divide(a, b)
527 store(name, division$quotient)
528 store(words[3L], division$remainder)
529 },
530 GCD = { store(name, polynomial_gcd(a, b)) },
531 XGCD = {
532 result <- extended_gcd(a, b)
533 for (i in 1L:3L) store(words[i + 1L], result[[i]])
534 },
535 LCM = {
536 if (!length(a) || !length(b)) {
537 store(name, EMPTY)
538 } else {
539 common <- polynomial_gcd(a, b)
540 degree_limit(degree(a) + degree(b) - degree(common) > 30L)
541 store(name, monic(multiply(divide(a, common)$quotient, b)))
542 }
543 },
544 INVMOD = {
545 if (!length(b)) fail("DIVISION BY ZERO")
546 if (degree(b) == 0L) fail("DEGREE TOO LOW")
547 result <- extended_gcd(a, b)
548 if (!is_one(result[[1L]])) fail("NOT COPRIME")
549 store(name, modulo(result[[2L]], b))
550 },
551 CRT = {
552 if (!length(b) || !length(d)) fail("DIVISION BY ZERO")
553 degree_limit(degree(b) + degree(d) > 30L)
554 result <- extended_gcd(b, d)
555 if (!is_one(result[[1L]])) fail("NOT COPRIME")
556 first <- modulo(a, b)
557 second <- modulo(c, d)
558 correction <- modulo(multiply(subtract(second, first), result[[2L]]), d)
559 store(name, add(first, multiply(b, correction)))
560 },
561 MODPOW = {
562 if (!length(b)) fail("DIVISION BY ZERO")
563 degree_limit(2L * degree(b) - 2L > 30L)
564 result <- modulo(ONE, b)
565 base <- modulo(a, b)
566 exponent <- as.integer(words[4L])
567 while (exponent > 0L) {
568 if (exponent %% 2L) result <- modulo(multiply(result, base), b)
569 exponent <- exponent %/% 2L
570 if (exponent > 0L) base <- modulo(multiply(base, base), b)
571 }
572 store(name, result)
573 },
574 DERIV = { store(name, derivative(a, as.integer(words[4L]))) },
575 INTEG = {
576 degree_limit(length(a) > 0L && degree(a) + 1L > 30L)
577 store(name, integral(a))
578 },
579 AREA = {
580 antiderivative <- integral(a)
581 value <- evaluate(antiderivative, values[2L]) - evaluate(antiderivative, values[1L])
582 output <- paste0("integral(", name, ", ", fraction(values[1L]), ", ", fraction(values[2L]), ") = ", fraction(value))
583 },
584 TANGENT = {
585 point <- values[1L]
586 value <- evaluate(a, point)
587 slope <- evaluate(derivative(a), point)
588 store(name, linear(value - slope * point, slope))
589 },
590 TAYLOR = {
591 point <- values[1L]
592 variable <- if (point == ZERO) "x" else if (point > ZERO) paste0("(x - ", fraction(point), ")") else paste0("(x + ", fraction(-point), ")")
593 output <- paste0("taylor(", name, ", ", fraction(point), ") = ", notation(compose(a, linear(point)), variable))
594 },
595 COMPOSE = { composition_limit(a, b); store(name, compose(a, b)) },
596 EVAL = {
597 output <- vapply(seq_along(values), function(i) paste0(name, "(", fraction(values[i]), ") = ", fraction(evaluate(a, values[i]))), character(1L))
598 },
599 ROOTS = {
600 nonzero(a)
601 roots <- rational_roots(a)$roots
602 text <- if (length(roots)) paste(vapply(roots, function(root) paste0(fraction(root$value), " (", root$multiplicity, ")"), character(1L)), collapse = ", ") else "none"
603 output <- paste0("roots(", name, ") = ", text)
604 },
605 FACTOR = {
606 nonzero(a)
607 found <- rational_roots(a)
608 output <- paste0("factor(", name, ") = ", factorization(a, found))
609 },
610 SQFREE = { nonzero(a); output <- paste0("sqfree(", name, ") = ", squarefree_factors(a)) },
611 DESCARTES = {
612 nonzero(a)
613 signs <- vapply(seq_along(a), function(i) sign_of(a[i]), integer(1L))
614 positive <- sign_changes(rev(signs))
615 signs[seq_along(signs) %% 2L == 0L] <- -signs[seq_along(signs) %% 2L == 0L]
616 output <- paste0("descartes(", name, ") = positive ", positive, ", negative ", sign_changes(rev(signs)))
617 },
618 STURM = {
619 nonzero(a)
620 if (bound_compare(endpoints[[1L]], endpoints[[2L]]) >= 0L) fail("BAD INTERVAL")
621 sequence <- sturm_sequence(a)
622 count <- variations(sequence, endpoints[[1L]]) - variations(sequence, endpoints[[2L]])
623 output <- paste0("sturm(", name, ", ", bound_text(endpoints[[1L]]), ", ", bound_text(endpoints[[2L]]), ") = ", count)
624 },
625 ISOLATE = {
626 nonzero(a)
627 precision <- as.integer(words[3L])
628 output <- paste0("isolate(", name, ", ", precision, ") = ", isolate(a, precision))
629 },
630 RESULTANT = {
631 nonzero(a)
632 nonzero(b)
633 output <- paste0("res(", name, ", ", words[3L], ") = ", fraction(resultant(a, b)))
634 },
635 DISCRIM = {
636 nonzero(a)
637 if (degree(a) == 0L) fail("DEGREE TOO LOW")
638 value <- resultant(a, derivative(a)) / leading(a)
639 if (((degree(a) * (degree(a) - 1L)) %/% 2L) %% 2L) value <- -value
640 output <- paste0("disc(", name, ") = ", fraction(value))
641 },
642 POWERSUM = {
643 nonzero(a)
644 if (degree(a) == 0L) fail("DEGREE TOO LOW")
645 output <- paste0("powersum(", name, ") = ", power_sums(a, as.integer(words[3L])))
646 },
647 INTERP = { store(name, interpolate(values)) },
648 SPECIAL = { store(name, special(words[3L], as.integer(words[4L]))) }
649 )
650 }
