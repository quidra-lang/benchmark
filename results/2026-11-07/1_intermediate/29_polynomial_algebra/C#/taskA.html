9readonly record struct Fraction : IComparable<Fraction>
10{
11 public long Numerator { get; }
12 public long Denominator { get; }
13 public static Fraction Zero => new(0);
14 public static Fraction One => new(1);
15 public bool IsZero => Numerator == 0;
16 public int Sign => Math.Sign(Numerator);
17
18 public Fraction(long numerator, long denominator = 1)
19 {
20 if (denominator < 0) { numerator = -numerator; denominator = -denominator; }
21 long divisor = Gcd(Math.Abs(numerator), denominator);
22 Numerator = numerator / divisor;
23 Denominator = denominator / divisor;
24 }
25
26 public static long Gcd(long a, long b)
27 {
28 while (b != 0) (a, b) = (b, a % b);
29 return a;
30 }
31
32 public static Fraction Parse(string text)
33 {
34 string[] parts = text.Split('/');
35 long numerator = long.Parse(parts[0]);
36 long denominator = parts.Length == 1 ? 1 : long.Parse(parts[1]);
37 if (denominator == 0) throw new CommandError("ERROR ZERO DENOMINATOR");
38 return new(numerator, denominator);
39 }
40
41 public static Fraction operator +(Fraction a, Fraction b) =>
42 new(a.Numerator * b.Denominator + b.Numerator * a.Denominator, a.Denominator * b.Denominator);
43 public static Fraction operator -(Fraction a, Fraction b) =>
44 new(a.Numerator * b.Denominator - b.Numerator * a.Denominator, a.Denominator * b.Denominator);
45 public static Fraction operator *(Fraction a, Fraction b) =>
46 new(a.Numerator * b.Numerator, a.Denominator * b.Denominator);
47 public static Fraction operator /(Fraction a, Fraction b) =>
48 new(a.Numerator * b.Denominator, a.Denominator * b.Numerator);
49 public static Fraction operator -(Fraction a) => new(-a.Numerator, a.Denominator);
50 public Fraction Abs() => new(Math.Abs(Numerator), Denominator);
51 public int CompareTo(Fraction other) =>
52 (Numerator * other.Denominator).CompareTo(other.Numerator * Denominator);
53 public Fraction Power(int exponent)
54 {
55 Fraction result = One;
56 for (int i = 0; i < exponent; i++) result *= this;
57 return result;
58 }
59 public override string ToString() => Denominator == 1 ? $"{Numerator}" : $"{Numerator}/{Denominator}";
60}

62sealed class Polynomial
63{
64 private readonly Fraction[] coefficients;
65 public static Polynomial Zero { get; } = new();
66 public static Polynomial One { get; } = new(Fraction.One);
67 public static Polynomial X { get; } = new(Fraction.Zero, Fraction.One);
68 public int Degree => coefficients.Length - 1;
69 public bool IsZero => coefficients.Length == 0;
70 public Fraction Leading => coefficients[^1];
71 public Fraction this[int degree] => degree < coefficients.Length ? coefficients[degree] : Fraction.Zero;
72
73 public Polynomial(params Fraction[] values)
74 {
75 int length = values.Length;
76 while (length > 0 && values[length - 1].IsZero) length--;
77 coefficients = values[..length];
78 }
79
80 private static Fraction[] Zeros(int count) => Enumerable.Repeat(Fraction.Zero, count).ToArray();
81 public static Polynomial operator +(Polynomial a, Polynomial b)
82 {
83 Fraction[] result = Zeros(Math.Max(a.Degree, b.Degree) + 1);
84 for (int i = 0; i < result.Length; i++) result[i] = a[i] + b[i];
85 return new(result);
86 }
87 public static Polynomial operator -(Polynomial a, Polynomial b)
88 {
89 Fraction[] result = Zeros(Math.Max(a.Degree, b.Degree) + 1);
90 for (int i = 0; i < result.Length; i++) result[i] = a[i] - b[i];
91 return new(result);
92 }
93 public static Polynomial operator -(Polynomial value) => new(value.coefficients.Select(c => -c).ToArray());
94 public static Polynomial operator *(Polynomial a, Polynomial b)
95 {
96 if (a.IsZero || b.IsZero) return Zero;
97 Fraction[] result = Zeros(a.Degree + b.Degree + 1);
98 for (int i = 0; i <= a.Degree; i++)
99 for (int j = 0; j <= b.Degree; j++) result[i + j] += a[i] * b[j];
100 return new(result);
101 }
102 public Polynomial Scale(Fraction factor) => new(coefficients.Select(c => c * factor).ToArray());
103 public Polynomial Divide(Fraction divisor) => new(coefficients.Select(c => c / divisor).ToArray());
104 public Polynomial Monic() => IsZero ? Zero : Divide(Leading);
105 public Polynomial Power(int exponent)
106 {
107 if (exponent == 0) return One;
108 if (IsZero) return Zero;
109 if (Degree == 0 && Leading.Denominator == 1 && Math.Abs(Leading.Numerator) == 1)
110 return Leading.Numerator == 1 || exponent % 2 == 0 ? One : this;
111 Polynomial result = One;
112 for (int i = 0; i < exponent; i++) result *= this;
113 return result;
114 }
115 public Fraction Evaluate(Fraction point)
116 {
117 Fraction result = Fraction.Zero;
118 for (int i = Degree; i >= 0; i--) result = result * point + this[i];
119 return result;
120 }
121 public Polynomial Compose(Polynomial inner)
122 {
123 Polynomial result = Zero;
124 for (int i = Degree; i >= 0; i--) result = result * inner + new Polynomial(this[i]);
125 return result;
126 }
127 public Polynomial Derivative(int order = 1)
128 {
129 Polynomial result = this;
130 for (int step = 0; step < order; step++) {
131 Fraction[] next = Zeros(Math.Max(0, result.Degree));
132 for (int i = 1; i <= result.Degree; i++) next[i - 1] = new Fraction(i) * result[i];
133 result = new(next);
134 }
135 return result;
136 }
137 public Polynomial Integral()
138 {
139 if (IsZero) return Zero;
140 Fraction[] result = Zeros(coefficients.Length + 1);
141 for (int i = 0; i <= Degree; i++) result[i + 1] = this[i] / new Fraction(i + 1);
142 return new(result);
143 }
144 public (Polynomial Quotient, Polynomial Remainder) Divide(Polynomial divisor)
145 {
146 Fraction[] quotient = Zeros(Math.Max(0, Degree - divisor.Degree + 1));
147 Fraction[] remainder = (Fraction[])coefficients.Clone();
148 int degree = Degree;
149 while (degree >= divisor.Degree && degree >= 0) {
150 Fraction factor = remainder[degree] / divisor.Leading;
151 int shift = degree - divisor.Degree;
152 quotient[shift] = factor;
153 for (int j = 0; j <= divisor.Degree; j++) remainder[shift + j] -= factor * divisor[j];
154 while (degree >= 0 && remainder[degree].IsZero) degree--;
155 }
156 return (new(quotient), new(remainder));
157 }
158 public static Polynomial Gcd(Polynomial a, Polynomial b)
159 {
160 while (!b.IsZero) (a, b) = (b, a.Divide(b).Remainder);
161 return a.Monic();
162 }
163 public static (Polynomial G, Polynomial S, Polynomial T) ExtendedGcd(Polynomial a, Polynomial b)
164 {
165 Polynomial r0 = a, r1 = b, s0 = One, s1 = Zero, t0 = Zero, t1 = One;
166 while (!r1.IsZero) {
167 var (quotient, remainder) = r0.Divide(r1);
168 (r0, r1) = (r1, remainder);
169 (s0, s1) = (s1, s0 - quotient * s1);
170 (t0, t1) = (t1, t0 - quotient * t1);
171 }
172 return r0.IsZero ? (r0, s0, t0) : (r0.Monic(), s0.Divide(r0.Leading), t0.Divide(r0.Leading));
173 }
174 public Fraction Content()
175 {
176 long denominator = 1, numerator = 0;
177 foreach (Fraction coefficient in coefficients) {
178 denominator = denominator * coefficient.Denominator / Fraction.Gcd(denominator, coefficient.Denominator);
179 numerator = Fraction.Gcd(numerator, Math.Abs(coefficient.Numerator));
180 }
181 return new(Leading.Sign < 0 ? -numerator : numerator, denominator);
182 }
183 public Polynomial Primitive() => Divide(Content());
184 public Polynomial SquareFreePart() => Degree < 1 ? this : Divide(Gcd(this, Derivative())).Quotient;
185 public Polynomial WithoutConstant() => new(coefficients[1..]);
186
187 public string Format(string variable = "x")
188 {
189 if (IsZero) return "0";
190 StringBuilder result = new();
191 for (int i = Degree; i >= 0; i--) {
192 Fraction coefficient = this[i];
193 if (coefficient.IsZero) continue;
194 if (result.Length == 0) { if (coefficient.Sign < 0) result.Append('-'); }
195 else result.Append(coefficient.Sign < 0 ? " - " : " + ");
196 Fraction magnitude = coefficient.Abs();
197 if (i == 0 || magnitude != Fraction.One) result.Append(magnitude);
198 if (i > 0) result.Append(variable);
199 if (i > 1) result.Append('^').Append(i);
200 }
201 return result.ToString();
202 }
203}

237sealed record LiteralExpression(Polynomial Value) : Expression
238{
239 public override Polynomial Evaluate(IReadOnlyDictionary<string, Polynomial> storage) => Value;
240}
241sealed record NameExpression(string Name, int Order) : Expression
242{
243 public override Polynomial Evaluate(IReadOnlyDictionary<string, Polynomial> storage) =>
244 storage.TryGetValue(Name, out Polynomial? value) ? value.Derivative(Order) : throw new CommandError($"ERROR UNKNOWN {Name}");
245}
246sealed record NegativeExpression(Expression Operand) : Expression
247{
248 public override Polynomial Evaluate(IReadOnlyDictionary<string, Polynomial> storage) => -Operand.Evaluate(storage);
249}
250sealed record BinaryExpression(char Operation, Expression Left, Expression Right) : Expression
251{
252 public override Polynomial Evaluate(IReadOnlyDictionary<string, Polynomial> storage)
253 {
254 Polynomial a = Left.Evaluate(storage), b = Right.Evaluate(storage);
255 switch (Operation) {
256 case '+': return a + b;
257 case '-': return a - b;
258 case '*': Checks.Product(a, b); return a * b;
259 default:
260 Checks.Divisor(b);
261 if (b.Degree >= 1) throw new CommandError("ERROR NONCONSTANT DIVISOR");
262 return a.Divide(b[0]);
263 }
264 }
265}
266sealed record PowerExpression(Expression Operand, int Exponent) : Expression
267{
268 public override Polynomial Evaluate(IReadOnlyDictionary<string, Polynomial> storage)
269 {
270 Polynomial value = Operand.Evaluate(storage);
271 Checks.Power(value, Exponent);
272 return value.Power(Exponent);
273 }
274}
275sealed record CallExpression(Expression Function, Expression Argument) : Expression
276{
277 public override Polynomial Evaluate(IReadOnlyDictionary<string, Polynomial> storage)
278 {
279 Polynomial function = Function.Evaluate(storage), argument = Argument.Evaluate(storage);
280 Checks.Composition(function, argument);
281 return function.Compose(argument);
282 }
283}

285enum TokenKind { Number, Name, Symbol, End }
286readonly record struct Token(TokenKind Kind, string Text, int Position);
287
288sealed class Parser
289{
290 private readonly List<Token> tokens = [];
291 private int position;
292 private Token Current => tokens[position];
293 private bool At(char symbol) => Current.Kind == TokenKind.Symbol && Current.Text[0] == symbol;
294
295 public Parser(string source)
296 {
297 for (int i = 0; i < source.Length;) {
298 if (source[i] == ' ') { i++; continue; }
299 int start = i++;
300 TokenKind kind;
301 if (char.IsAsciiDigit(source[start])) {
302 kind = TokenKind.Number;
303 while (i < source.Length && char.IsAsciiDigit(source[i])) i++;
304 } else if (source[start] is >= 'A' and <= 'Z') {
305 kind = TokenKind.Name;
306 while (i < source.Length && (source[i] is >= 'A' and <= 'Z' || char.IsAsciiDigit(source[i]))) i++;
307 } else kind = TokenKind.Symbol;
308 tokens.Add(new(kind, source[start..i], start + 1));
309 }
310 tokens.Add(new(TokenKind.End, "", source.Length + 1));
311 }
312 private void Syntax() => throw new CommandError(Current.Kind == TokenKind.End ? "ERROR SYNTAX AT END" : $"ERROR SYNTAX AT {Current.Position}");
313 private void Expect(char symbol)
314 {
315 if (!At(symbol)) Syntax();
316 position++;
317 }
318 public Expression Parse()
319 {
320 Expression result = ParseExpression();
321 if (Current.Kind != TokenKind.End) Syntax();
322 return result;
323 }
324 private Expression ParseExpression()
325 {
326 bool negative = At('-');
327 if (negative) position++;
328 Expression result = ParseTerm();
329 if (negative) result = new NegativeExpression(result);
330 while (At('+') || At('-')) {
331 char operation = Current.Text[0];
332 position++;
333 result = new BinaryExpression(operation, result, ParseTerm());
334 }
335 return result;
336 }
337 private Expression ParseTerm()
338 {
339 Expression result = ParseFactor();
340 while (true) {
341 char operation;
342 if (At('*') || At('/')) { operation = Current.Text[0]; position++; }
343 else if (At('x') || At('(') || Current.Kind == TokenKind.Name) operation = '*';
344 else break;
345 result = new BinaryExpression(operation, result, ParseFactor());
346 }
347 return result;
348 }
349 private Expression ParseFactor()
350 {
351 Expression result = ParsePrimary();
352 if (At('^')) {
353 position++;
354 if (Current.Kind != TokenKind.Number) Syntax();
355 int exponent = int.Parse(Current.Text);
356 position++;
357 result = new PowerExpression(result, exponent);
358 }
359 return result;
360 }
361 private Expression ParsePrimary()
362 {
363 if (Current.Kind == TokenKind.Number) {
364 Polynomial value = new(new Fraction(long.Parse(Current.Text)));
365 position++;
366 return new LiteralExpression(value);
367 }
368 if (At('x')) { position++; return new LiteralExpression(Polynomial.X); }
369 if (Current.Kind == TokenKind.Name) {
370 string name = Current.Text;
371 position++;
372 int order = 0;
373 while (At('\'')) { order++; position++; }
374 Expression reference = new NameExpression(name, order);
375 if (!At('(')) return reference;
376 position++;
377 Expression argument = ParseExpression();
378 Expect(')');
379 return new CallExpression(reference, argument);
380 }
381 if (At('(')) {
382 position++;
383 Expression result = ParseExpression();
384 Expect(')');
385 return result;
386 }
387 Syntax();
388 throw new InvalidOperationException();
389 }
390}

401static class Algebra
402{
403 public static (Fraction Content, Polynomial Remaining, List<(Fraction Root, int Count)> Roots) RationalRoots(Polynomial value)
404 {
405 Fraction content = value.Content();
406 Polynomial remaining = value.Divide(content);
407 int zeros = 0;
408 while (remaining[0].IsZero) { remaining = remaining.WithoutConstant(); zeros++; }
409 static List<long> Divisors(long number)
410 {
411 List<long> result = [];
412 for (long i = 1; i <= number; i++) if (number % i == 0) result.Add(i);
413 return result;
414 }
415 List<long> numerators = Divisors(Math.Abs(remaining[0].Numerator));
416 List<long> denominators = Divisors(remaining.Leading.Numerator);
417 List<Fraction> candidates = [];
418 foreach (long numerator in numerators)
419 foreach (long denominator in denominators)
420 if (Fraction.Gcd(numerator, denominator) == 1) {
421 candidates.Add(new(numerator, denominator));
422 candidates.Add(new(-numerator, denominator));
423 }
424 candidates.Sort();
425 List<(Fraction Root, int Count)> roots = [];
426 foreach (Fraction candidate in candidates) {
427 int count = 0;
428 Polynomial divisor = new(new Fraction(-candidate.Numerator), new Fraction(candidate.Denominator));
429 while (remaining.Degree >= 1 && remaining.Evaluate(candidate).IsZero) {
430 remaining = remaining.Divide(divisor).Quotient;
431 count++;
432 }
433 if (count > 0) roots.Add((candidate, count));
434 }
435 if (zeros > 0) roots.Add((Fraction.Zero, zeros));
436 roots.Sort((left, right) => left.Root.CompareTo(right.Root));
437 return (content, remaining, roots);
438 }

440 public static List<(Polynomial Factor, int Count)> SquareFreeFactors(Polynomial value)
441 {
442 List<(Polynomial Factor, int Count)> result = [];
443 if (value.Degree < 1) return result;
444 Polynomial monic = value.Monic();
445 Polynomial c = Polynomial.Gcd(monic, monic.Derivative());
446 Polynomial w = monic.Divide(c).Quotient;
447 for (int i = 1; w.Degree >= 1; i++) {
448 Polynomial y = Polynomial.Gcd(w, c);
449 Polynomial factor = w.Divide(y).Quotient;
450 if (factor.Degree >= 1) result.Add((factor.Primitive(), i));
451 c = c.Divide(y).Quotient;
452 w = y;
453 }
454 return result;
455 }

457 public static List<Polynomial> SturmSequence(Polynomial value)
458 {
459 Polynomial previous = value.SquareFreePart(), current = previous.Derivative();
460 List<Polynomial> result = [previous];
461 while (!current.IsZero) {
462 result.Add(current);
463 Polynomial next = -previous.Divide(current).Remainder;
464 previous = current;
465 current = next;
466 }
467 return result;
468 }
469 public static int Variations(List<Polynomial> sequence, Endpoint point)
470 {
471 int previous = 0, changes = 0;
472 foreach (Polynomial value in sequence) {
473 int sign = point.Infinity == 0 ? value.Evaluate(point.Value).Sign : value.Leading.Sign;
474 if (point.Infinity < 0 && value.Degree % 2 != 0) sign = -sign;
475 if (sign == 0) continue;
476 if (previous != 0 && sign != previous) changes++;
477 previous = sign;
478 }
479 return changes;
480 }
481 public static int RootCount(List<Polynomial> sequence, Endpoint left, Endpoint right) => Variations(sequence, left) - Variations(sequence, right);
482 public static List<(Fraction Left, Fraction Right)> Isolate(Polynomial value, int precision)
483 {
484 List<Polynomial> sequence = SturmSequence(value);
485 Polynomial first = sequence[0];
486 Fraction maximum = Fraction.Zero;
487 for (int i = 0; i < first.Degree; i++) {
488 Fraction candidate = (first[i] / first.Leading).Abs();
489 if (candidate.CompareTo(maximum) > 0) maximum = candidate;
490 }
491 Fraction beta = Fraction.One + maximum, width = new(1, 1L << precision);
492 List<(Fraction Left, Fraction Right)> result = [];
493 int Count(Fraction left, Fraction right) => RootCount(sequence, new(0, left), new(0, right));
494 void Examine(Fraction left, Fraction right)
495 {
496 int count = Count(left, right);
497 if (count >= 2) {
498 Fraction middle = (left + right) / new Fraction(2);
499 Examine(left, middle);
500 Examine(middle, right);
501 } else if (count == 1) {
502 while ((right - left).CompareTo(width) > 0) {
503 Fraction middle = (left + right) / new Fraction(2);
504 if (Count(left, middle) >= 1) right = middle;
505 else left = middle;
506 }
507 result.Add((left, right));
508 }
509 }
510 Examine(-beta, beta);
511 return result;
512 }
513 public static Fraction Resultant(Polynomial p, Polynomial q)
514 {
515 Fraction result = Fraction.One;
516 while (q.Degree >= 1) {
517 Polynomial remainder = p.Divide(q).Remainder;
518 if (remainder.IsZero) return Fraction.Zero;
519 result *= q.Leading.Power(p.Degree - remainder.Degree);
520 if (p.Degree * q.Degree % 2 != 0) result = -result;
521 (p, q) = (q, remainder);
522 }
523 return result * q[0].Power(p.Degree);
524 }
525 public static int Descartes(Polynomial value, bool negative)
526 {
527 int previous = 0, count = 0;
528 for (int i = value.Degree; i >= 0; i--) {
529 int sign = value[i].Sign;
530 if (negative && i % 2 != 0) sign = -sign;
531 if (sign == 0) continue;
532 if (previous != 0 && previous != sign) count++;
533 previous = sign;
534 }
535 return count;
536 }
537 public static Polynomial Interpolate(Fraction[] values)
538 {
539 int count = values.Length / 2;
540 Fraction[] x = new Fraction[count], differences = new Fraction[count];
541 for (int i = 0; i < count; i++) { x[i] = values[2 * i]; differences[i] = values[2 * i + 1]; }
542 for (int j = 1; j < count; j++)
543 for (int i = 0; i < j; i++)
544 if (x[j].CompareTo(x[i]) == 0) throw new CommandError($"ERROR DUPLICATE X {x[j]}");
545 for (int j = 1; j < count; j++)
546 for (int i = count - 1; i >= j; i--) differences[i] = (differences[i] - differences[i - 1]) / (x[i] - x[i - j]);
547 Polynomial result = Polynomial.Zero;
548 for (int i = count - 1; i >= 0; i--) result = result * new Polynomial(-x[i], Fraction.One) + new Polynomial(differences[i]);
549 return result;
550 }
551 public static Polynomial Special(string kind, int degree)
552 {
553 if (kind is not ("chebyshev" or "chebyshev2" or "legendre" or "hermite" or "laguerre"))
554 throw new CommandError($"ERROR UNKNOWN KIND {kind}");
555 if (degree > 30) throw new CommandError("ERROR DEGREE LIMIT");
556 Polynomial previous = Polynomial.One;
557 Polynomial current = kind switch {
558 "chebyshev2" or "hermite" => new(Fraction.Zero, new Fraction(2)),
559 "laguerre" => new(Fraction.One, new Fraction(-1)), _ => Polynomial.X,
560 };
561 if (degree == 0) return previous;
562 for (int j = 1; j < degree; j++) {
563 Polynomial factor = kind switch {
564 "legendre" => new(Fraction.Zero, new Fraction(2 * j + 1)),
565 "laguerre" => new(new Fraction(2 * j + 1), new Fraction(-1)),
566 _ => new(Fraction.Zero, new Fraction(2)),
567 };
568 int scale = kind is "legendre" or "laguerre" ? j : kind == "hermite" ? 2 * j : 1;
569 Polynomial next = factor * current - previous.Scale(new Fraction(scale));
570 if (kind is "legendre" or "laguerre") next = next.Scale(new Fraction(1, j + 1));
571 previous = current;
572 current = next;
573 }
574 return current;
575 }

593 public List<string> Execute(string line)
594 {
595 List<string> output = [];
596 List<(string Name, Polynomial Value)> pending = [];
597 string? deletion = null;
598 void Save(string name, Polynomial value)
599 {
600 pending.Add((name, value));
601 output.Add($"{name} = {value.Format()}");
602 }
603
604 if (line.StartsWith("DEF ", StringComparison.Ordinal)) {
605 string[] definition = line.Split(' ', 3);
606 Expression expression = new Parser(definition[2]).Parse();
607 Save(definition[1], expression.Evaluate(storage));
608 } else {
609 string[] words = line.Split(' ');
610 string command = words[0];
611 if (command == "DIV" && words[1] == words[2] || command == "XGCD" &&
612 (words[1] == words[2] || words[1] == words[3] || words[2] == words[3])) {
613 throw new CommandError("ERROR SAME NAME");
614 }
615 Polynomial[] operands = NameArguments(command).Select(index => Lookup(words[index])).ToArray();
616 int[] literalArguments = command switch {
617 "AREA" => [2, 3], "TANGENT" => [3], "TAYLOR" => [2],
618 "EVAL" or "INTERP" => Enumerable.Range(2, words.Length - 2).ToArray(), _ => [],
619 };
620 Fraction[] literals = literalArguments.Select(index => Fraction.Parse(words[index])).ToArray();
621 Endpoint left = default, right = default;
622 if (command == "STURM") { left = Endpoint.Parse(words[2]); right = Endpoint.Parse(words[3]); }
623 Polynomial a = operands.Length > 0 ? operands[0] : Polynomial.Zero;
624 Polynomial b = operands.Length > 1 ? operands[1] : Polynomial.Zero;
625 switch (command) {
626 case "PRINT": output.Add($"{words[1]} = {a.Format()}"); break;
627 case "DELETE": deletion = words[1]; output.Add($"deleted {words[1]}"); break;
628 case "LIST":
629 if (storage.Count == 0) output.Add("no polynomials");
630 else foreach (string name in storage.Keys.OrderBy(name => name, StringComparer.Ordinal))
631 output.Add($"{name} = {storage[name].Format()}");
632 break;
633 case "DEGREE": output.Add($"deg({words[1]}) = {(a.IsZero ? "-inf" : a.Degree.ToString())}"); break;
634 case "ADD": Save(words[1], a + b); break;
635 case "SUB": Save(words[1], a - b); break;
636 case "MUL": Checks.Product(a, b); Save(words[1], a * b); break;
637 case "POW": {
638 int exponent = int.Parse(words[3]);
639 Checks.Power(a, exponent);
640 Save(words[1], a.Power(exponent));
641 break;
642 }
643 case "DIV": {
644 Checks.Divisor(b);
645 var (quotient, remainder) = a.Divide(b);
646 Save(words[1], quotient);
647 Save(words[2], remainder);
648 break;
649 }
650 case "GCD": Save(words[1], Polynomial.Gcd(a, b)); break;
651 case "XGCD": {
652 var (g, s, t) = Polynomial.ExtendedGcd(a, b);
653 Save(words[1], g); Save(words[2], s); Save(words[3], t);
654 break;
655 }
656 case "LCM": {
657 if (a.IsZero || b.IsZero) { Save(words[1], Polynomial.Zero); break; }
658 Polynomial gcd = Polynomial.Gcd(a, b);
659 if (a.Degree + b.Degree - gcd.Degree > 30) throw new CommandError("ERROR DEGREE LIMIT");
660 Save(words[1], (a.Divide(gcd).Quotient * b).Monic());
661 break;
662 }
663 case "INVMOD": {
664 Checks.Divisor(b); Checks.PositiveDegree(b);
665 var (g, s, _) = Polynomial.ExtendedGcd(a, b);
666 if (g.Degree != 0 || g[0] != Fraction.One) throw new CommandError("ERROR NOT COPRIME");
667 Save(words[1], s.Divide(b).Remainder);
668 break;
669 }
670 case "CRT": {
671 Polynomial a2 = operands[2], m2 = operands[3];
672 Checks.Divisor(b); Checks.Divisor(m2);
673 if (b.Degree + m2.Degree > 30) throw new CommandError("ERROR DEGREE LIMIT");
674 var (g, s, _) = Polynomial.ExtendedGcd(b, m2);
675 if (g.Degree != 0 || g[0] != Fraction.One) throw new CommandError("ERROR NOT COPRIME");
676 Polynomial r1 = a.Divide(b).Remainder, r2 = a2.Divide(m2).Remainder;
677 Polynomial u = ((r2 - r1) * s).Divide(m2).Remainder;
678 Save(words[1], r1 + b * u);
679 break;
680 }
681 case "MODPOW": {
682 Checks.Divisor(b);
683 if (2 * b.Degree - 2 > 30) throw new CommandError("ERROR DEGREE LIMIT");
684 Polynomial x = Polynomial.One.Divide(b).Remainder, y = a.Divide(b).Remainder;
685 int exponent = int.Parse(words[3]);
686 while (exponent > 0) {
687 if (exponent % 2 != 0) x = (x * y).Divide(b).Remainder;
688 exponent /= 2;
689 if (exponent != 0) y = (y * y).Divide(b).Remainder;
690 }
691 Save(words[1], x);
692 break;
693 }
694 case "DERIV": Save(words[1], a.Derivative(int.Parse(words[3]))); break;
695 case "INTEG":
696 if (a.Degree == 30) throw new CommandError("ERROR DEGREE LIMIT");
697 Save(words[1], a.Integral());
698 break;
699 case "AREA": {
700 Polynomial integral = a.Integral();
701 Fraction value = integral.Evaluate(literals[1]) - integral.Evaluate(literals[0]);
702 output.Add($"integral({words[1]}, {literals[0]}, {literals[1]}) = {value}");
703 break;
704 }
705 case "TANGENT": {
706 Fraction point = literals[0], value = a.Evaluate(point), slope = a.Derivative().Evaluate(point);
707 Save(words[1], new Polynomial(value - slope * point, slope));
708 break;
709 }
710 case "TAYLOR": {
711 Fraction point = literals[0];
712 Polynomial shifted = a.Compose(new Polynomial(point, Fraction.One));
713 string variable = point.IsZero ? "x" : point.Sign > 0 ? $"(x - {point})" : $"(x + {-point})";
714 output.Add($"taylor({words[1]}, {point}) = {shifted.Format(variable)}");
715 break;
716 }
717 case "COMPOSE": Checks.Composition(a, b); Save(words[1], a.Compose(b)); break;
718 case "EVAL":
719 foreach (Fraction point in literals) output.Add($"{words[1]}({point}) = {a.Evaluate(point)}");
720 break;
721 case "ROOTS": {
722 Checks.Nonzero(a);
723 var (_, _, roots) = Algebra.RationalRoots(a);
724 string result = roots.Count == 0 ? "none" : string.Join(", ", roots.Select(root => $"{root.Root} ({root.Count})"));
725 output.Add($"roots({words[1]}) = {result}");
726 break;
727 }
728 case "FACTOR": {
729 Checks.Nonzero(a);
730 var (content, remaining, roots) = Algebra.RationalRoots(a);
731 StringBuilder result = new($"factor({words[1]}) = {content}");
732 foreach (var (root, count) in roots) {
733 string factor = root.IsZero ? "x" : $"({new Polynomial(new Fraction(-root.Numerator), new Fraction(root.Denominator)).Format()})";
734 result.Append(" * ").Append(factor);
735 if (count >= 2) result.Append('^').Append(count);
736 }
737 if (remaining.Degree >= 1) result.Append(" * (").Append(remaining.Format()).Append(')');
738 output.Add(result.ToString());
739 break;
740 }
741 case "SQFREE": {
742 Checks.Nonzero(a);
743 List<(Polynomial Factor, int Count)> factors = Algebra.SquareFreeFactors(a);
744 StringBuilder result = new($"sqfree({words[1]}) = {a.Content()}");
745 foreach (var (factor, count) in factors) {
746 bool isX = factor.Degree == 1 && factor[0].IsZero && factor[1] == Fraction.One;
747 result.Append(" * ").Append(isX ? "x" : $"({factor.Format()})");
748 if (count >= 2) result.Append('^').Append(count);
749 }
750 output.Add(result.ToString());
751 break;
752 }
753 case "DESCARTES":
754 Checks.Nonzero(a);
755 output.Add($"descartes({words[1]}) = positive {Algebra.Descartes(a, false)}, negative {Algebra.Descartes(a, true)}");
756 break;
757 case "STURM": {
758 Checks.Nonzero(a);
759 if (left.CompareTo(right) >= 0) throw new CommandError("ERROR BAD INTERVAL");
760 List<Polynomial> sequence = Algebra.SturmSequence(a);
761 output.Add($"sturm({words[1]}, {left}, {right}) = {Algebra.RootCount(sequence, left, right)}");
762 break;
763 }
764 case "ISOLATE": {
765 Checks.Nonzero(a);
766 int precision = int.Parse(words[2]);
767 List<(Fraction Left, Fraction Right)> intervals = Algebra.Isolate(a, precision);
768 string result = intervals.Count == 0 ? "none" : string.Join(", ", intervals.Select(interval => $"({interval.Left}, {interval.Right}]"));
769 output.Add($"isolate({words[1]}, {precision}) = {result}");
770 break;
771 }
772 case "RESULTANT":
773 Checks.Nonzero(a); Checks.Nonzero(b);
774 output.Add($"res({words[1]}, {words[2]}) = {Algebra.Resultant(a, b)}");
775 break;
776 case "DISCRIM": {
777 Checks.Nonzero(a); Checks.PositiveDegree(a);
778 Fraction value = Algebra.Resultant(a, a.Derivative()) / a.Leading;
779 if (a.Degree * (a.Degree - 1) / 2 % 2 != 0) value = -value;
780 output.Add($"disc({words[1]}) = {value}");
781 break;
782 }
783 case "POWERSUM": {
784 Checks.Nonzero(a); Checks.PositiveDegree(a);
785 int count = int.Parse(words[2]);
786 Polynomial monic = a.Monic();
787 Fraction[] sums = Enumerable.Repeat(Fraction.Zero, count + 1).ToArray();
788 for (int j = 1; j <= count; j++) {
789 Fraction value = j <= a.Degree ? new Fraction(j) * monic[a.Degree - j] : Fraction.Zero;
790 for (int i = 1; i <= Math.Min(j - 1, a.Degree); i++) value += monic[a.Degree - i] * sums[j - i];
791 sums[j] = -value;
792 }
793 output.Add($"powersum({words[1]}) = {string.Join(", ", sums.Skip(1))}");
794 break;
795 }
796 case "INTERP": Save(words[1], Algebra.Interpolate(literals)); break;
797 case "SPECIAL": Save(words[1], Algebra.Special(words[2], int.Parse(words[3]))); break;
798 default: throw new InvalidOperationException();
799 }
800 }
801 if (deletion is not null) storage.Remove(deletion);
802 foreach (var (name, value) in pending) storage[name] = value;
803 return output;
804 }
805}
