11class Fraction {
12 constructor(numerator, denominator = 1n) {
13 if (denominator < 0n) { numerator = -numerator; denominator = -denominator; }
14 const divisor = gcd(numerator, denominator);
15 this.n = numerator / divisor;
16 this.d = denominator / divisor;
17 }
18 static parse(word) {
19 const [numerator, denominator = '1'] = word.split('/');
20 if (BigInt(denominator) === 0n) fail('ZERO DENOMINATOR');
21 return new Fraction(BigInt(numerator), BigInt(denominator));
22 }
23 add(other) { return new Fraction(this.n * other.d + other.n * this.d, this.d * other.d); }
24 sub(other) { return new Fraction(this.n * other.d - other.n * this.d, this.d * other.d); }
25 mul(other) { return new Fraction(this.n * other.n, this.d * other.d); }
26 div(other) { return new Fraction(this.n * other.d, this.d * other.n); }
27 neg() { return new Fraction(-this.n, this.d); }
28 abs() { return this.n < 0n ? this.neg() : this; }
29 compare(other) {
30 const left = this.n * other.d, right = other.n * this.d;
31 return left < right ? -1 : left > right ? 1 : 0;
32 }
33 pow(exponent) {
34 let result = ONE;
35 for (let i = 0; i < exponent; i++) result = result.mul(this);
36 return result;
37 }
38 toString() { return this.d === 1n ? String(this.n) : `${this.n}/${this.d}`; }
39}

44class Polynomial {
45 constructor(coefficients = []) {
46 this.coefficients = coefficients.slice();
47 while (this.coefficients.length && this.coefficients.at(-1).n === 0n) this.coefficients.pop();
48 }
49 static constant(value) { return new Polynomial([value]); }
50 get degree() { return this.coefficients.length - 1; }
51 get zero() { return this.degree < 0; }
52 get leading() { return this.coefficients.at(-1); }
53 coefficient(index) { return this.coefficients[index] ?? ZERO; }
54 add(other) {
55 return new Polynomial(Array.from({ length: Math.max(this.degree, other.degree) + 1 },
56 (_, i) => this.coefficient(i).add(other.coefficient(i))));
57 }
58 sub(other) {
59 return new Polynomial(Array.from({ length: Math.max(this.degree, other.degree) + 1 },
60 (_, i) => this.coefficient(i).sub(other.coefficient(i))));
61 }
62 scale(value) { return new Polynomial(this.coefficients.map(coefficient => coefficient.mul(value))); }
63 divideScalar(value) { return new Polynomial(this.coefficients.map(coefficient => coefficient.div(value))); }
64 neg() { return new Polynomial(this.coefficients.map(coefficient => coefficient.neg())); }
65 mul(other) {
66 if (this.zero || other.zero) return new Polynomial();
67 const result = Array(this.degree + other.degree + 1).fill(ZERO);
68 for (let i = 0; i <= this.degree; i++) {
69 for (let j = 0; j <= other.degree; j++) {
70 result[i + j] = result[i + j].add(this.coefficients[i].mul(other.coefficients[j]));
71 }
72 }
73 return new Polynomial(result);
74 }
75 pow(exponent) {
76 if (exponent === 0) return UNIT;
77 if (this.zero) return this;
78 if (this.degree === 0 && this.leading.d === 1n && abs(this.leading.n) === 1n) {
79 return Polynomial.constant(this.leading.n < 0n && exponent % 2 ? integer(-1) : ONE);
80 }
81 let result = UNIT;
82 for (let i = 0; i < exponent; i++) result = result.mul(this);
83 return result;
84 }
85 divide(other) {
86 const remainder = this.coefficients.slice();
87 const quotient = Array(Math.max(0, this.degree - other.degree + 1)).fill(ZERO);
88 while (remainder.length && remainder.length >= other.coefficients.length) {
89 const value = remainder.at(-1).div(other.leading);
90 const shift = remainder.length - other.coefficients.length;
91 quotient[shift] = value;
92 for (let j = 0; j <= other.degree; j++) {
93 remainder[shift + j] = remainder[shift + j].sub(value.mul(other.coefficients[j]));
94 }
95 while (remainder.length && remainder.at(-1).n === 0n) remainder.pop();
96 }
97 return [new Polynomial(quotient), new Polynomial(remainder)];
98 }
99 mod(other) { return this.divide(other)[1]; }
100 monic() { return this.zero ? this : this.divideScalar(this.leading); }
101 derivative(count = 1) {
102 let result = this;
103 for (let repetition = 0; repetition < count && !result.zero; repetition++) {
104 result = new Polynomial(result.coefficients.slice(1).map((coefficient, i) => integer(i + 1).mul(coefficient)));
105 }
106 return result;
107 }
108 integral() {
109 return new Polynomial([ZERO, ...this.coefficients.map((coefficient, i) => coefficient.div(integer(i + 1)))]);
110 }
111 evaluate(value) {
112 let result = ZERO;
113 for (let i = this.degree; i >= 0; i--) result = result.mul(value).add(this.coefficients[i]);
114 return result;
115 }
116 compose(other) {
117 let result = new Polynomial();
118 for (let i = this.degree; i >= 0; i--) result = result.mul(other).add(Polynomial.constant(this.coefficients[i]));
119 return result;
120 }
121 content() {
122 let multiple = 1n, divisor = 0n;
123 for (const coefficient of this.coefficients) {
124 multiple = multiple * coefficient.d / gcd(multiple, coefficient.d);
125 divisor = gcd(divisor, coefficient.n);
126 }
127 return new Fraction(this.leading.n < 0n ? -divisor : divisor, multiple);
128 }
129 primitive() { return this.divideScalar(this.content()); }
130 notation(variable = 'x') {
131 let result = '';
132 for (let i = this.degree; i >= 0; i--) {
133 const coefficient = this.coefficients[i];
134 if (coefficient.n === 0n) continue;
135 result += result === '' ? (coefficient.n < 0n ? '-' : '') : (coefficient.n < 0n ? ' - ' : ' + ');
136 const magnitude = coefficient.abs();
137 if (i === 0 || magnitude.compare(ONE) !== 0) result += magnitude;
138 if (i > 0) result += variable + (i === 1 ? '' : `^${i}`);
139 }
140 return result || '0';
141 }
142}

146const degreeLimit = condition => { if (condition) fail('DEGREE LIMIT'); };
147const productLimit = (a, b) => degreeLimit(!a.zero && !b.zero && a.degree + b.degree > 30);
148const compositionLimit = (a, b) => degreeLimit(a.degree >= 1 && b.degree >= 1 && a.degree * b.degree > 30);

152function polynomialGcd(a, b) {
153 while (!b.zero) [a, b] = [b, a.mod(b)];
154 return a.monic();

157function extendedGcd(a, b) {
158 let s0 = UNIT, s1 = new Polynomial(), t0 = new Polynomial(), t1 = UNIT;
159 while (!b.zero) {
160 const [quotient, remainder] = a.divide(b);
161 [a, b] = [b, remainder];
162 [s0, s1] = [s1, s0.sub(quotient.mul(s1))];
163 [t0, t1] = [t1, t0.sub(quotient.mul(t1))];
164 }
165 return a.zero ? [a, s0, t0] : [a.monic(), s0.divideScalar(a.leading), t0.divideScalar(a.leading)];
166}

168function divisors(number) {
169 const result = [];
170 for (let divisor = 1n; divisor * divisor <= number; divisor++) {
171 if (number % divisor !== 0n) continue;
172 result.push(divisor);
173 if (divisor * divisor !== number) result.push(number / divisor);
174 }
175 return result;
176}

178function rationalRoots(a) {
179 let remaining = a.primitive(), zeros = 0;
180 while (remaining.coefficient(0).n === 0n) {
181 remaining = new Polynomial(remaining.coefficients.slice(1));
182 zeros++;
183 }
184 const candidates = [];
185 for (const numerator of divisors(abs(remaining.coefficient(0).n))) {
186 for (const denominator of divisors(remaining.leading.n)) {
187 if (gcd(numerator, denominator) === 1n) {
188 const root = new Fraction(numerator, denominator);
189 candidates.push(root, root.neg());
190 }
191 }
192 }
193 candidates.sort((a, b) => a.compare(b));
194 const roots = [];
195 for (const root of candidates) {
196 let multiplicity = 0;
197 const factor = linear(new Fraction(-root.n), new Fraction(root.d));
198 while (remaining.degree >= 1 && remaining.evaluate(root).n === 0n) {
199 remaining = remaining.divide(factor)[0];
200 multiplicity++;
201 }
202 if (multiplicity) roots.push({ root, multiplicity });
203 }
204 if (zeros) roots.push({ root: ZERO, multiplicity: zeros });
205 roots.sort((a, b) => a.root.compare(b.root));
206 return { roots, remaining };
207}

213function sturmSequence(a) {
214 const first = squareFreePart(a), sequence = [first];
215 let previous = first, current = first.derivative();
216 while (!current.zero) {
217 sequence.push(current);
218 [previous, current] = [current, previous.mod(current).neg()];

223function signChanges(signs) {
224 let previous = 0, changes = 0;
225 for (const sign of signs) {
226 if (sign === 0) continue;
227 if (previous !== 0 && previous !== sign) changes++;
228 previous = sign;
229 }

248function isolate(a, precision) {
249 const sequence = sturmSequence(a), first = sequence[0];
250 let maximum = ZERO;
251 for (let i = 0; i < first.degree; i++) {
252 const ratio = first.coefficients[i].div(first.leading).abs();
253 if (ratio.compare(maximum) > 0) maximum = ratio;
254 }
255 const bound = ONE.add(maximum), width = new Fraction(1n, 1n << BigInt(precision)), intervals = [];
256 const count = (left, right) => variations(sequence, left) - variations(sequence, right);
257 function examine(left, right) {
258 const roots = count(left, right);
259 if (roots >= 2) {
260 const middle = left.add(right).div(TWO);
261 examine(left, middle);
262 examine(middle, right);
263 } else if (roots === 1) {
264 while (right.sub(left).compare(width) > 0) {
265 const middle = left.add(right).div(TWO);
266 if (count(left, middle) >= 1) right = middle;
267 else left = middle;
268 }
269 intervals.push(`(${left}, ${right}]`);
270 }
271 }
272 examine(bound.neg(), bound);
273 return intervals;
274}

276function resultant(a, b) {
277 let result = ONE;
278 while (b.degree >= 1) {
279 const remainder = a.mod(b);
280 if (remainder.zero) return ZERO;
281 result = result.mul(b.leading.pow(a.degree - remainder.degree));
282 if (a.degree * b.degree % 2) result = result.neg();
283 [a, b] = [b, remainder];
284 }
285 return result.mul(b.coefficient(0).pow(a.degree));
286}

288function interpolate(values) {
289 const xs = values.filter((_, i) => i % 2 === 0), differences = values.filter((_, i) => i % 2 === 1);
290 for (let j = 1; j < xs.length; j++) {
291 if (xs.slice(0, j).some(x => x.compare(xs[j]) === 0)) fail(`DUPLICATE X ${xs[j]}`);
292 }
293 for (let j = 1; j < xs.length; j++) {
294 for (let i = xs.length - 1; i >= j; i--) {
295 differences[i] = differences[i].sub(differences[i - 1]).div(xs[i].sub(xs[i - j]));
296 }
297 }
298 let result = new Polynomial();
299 for (let i = xs.length - 1; i >= 0; i--) result = result.mul(linear(xs[i].neg())).add(Polynomial.constant(differences[i]));
300 return result;
301}

303function special(kind, order) {
304 if (!['chebyshev', 'chebyshev2', 'legendre', 'hermite', 'laguerre'].includes(kind)) fail(`UNKNOWN KIND ${kind}`);
305 degreeLimit(order > 30);
306 if (order === 0) return UNIT;
307 let previous = UNIT;
308 let current = kind === 'laguerre' ? linear(ONE, integer(-1))
309 : kind === 'chebyshev2' || kind === 'hermite' ? X.scale(TWO) : X;
310 for (let j = 1; j < order; j++) {
311 let next;
312 if (kind === 'chebyshev' || kind === 'chebyshev2') next = X.scale(TWO).mul(current).sub(previous);
313 else if (kind === 'hermite') next = X.scale(TWO).mul(current).sub(previous.scale(integer(2 * j)));
314 else {
315 const multiplier = kind === 'legendre' ? X.scale(integer(2 * j + 1)) : linear(integer(2 * j + 1), integer(-1));
316 next = multiplier.mul(current).sub(previous.scale(integer(j))).scale(new Fraction(1n, BigInt(j + 1)));
317 }
318 [previous, current] = [current, next];
319 }
320 return current;
321}

323class ExpressionParser {
324 constructor(source) {
325 this.tokens = [];
326 const pattern = /[0-9]+|[A-Z][A-Z0-9]*|[^ ]/g;
327 for (const match of source.matchAll(pattern)) {
328 const word = match[0];
329 const kind = /^[0-9]/.test(word) ? 'number' : /^[A-Z]/.test(word) ? 'name' : word;
330 this.tokens.push({ kind, word, position: match.index + 1 });
331 }
332 this.tokens.push({ kind: 'end' });
333 this.position = 0;
334 }
335 get token() { return this.tokens[this.position]; }
336 take(kind) {
337 if (this.token.kind !== kind) return false;
338 this.position++;
339 return true;
340 }
341 syntaxError() { fail(`SYNTAX AT ${this.token.kind === 'end' ? 'END' : this.token.position}`); }
342 expect(kind) { if (!this.take(kind)) this.syntaxError(); }
343 parse() {
344 const result = this.expression();
345 if (this.token.kind !== 'end') this.syntaxError();
346 return result;
347 }
348 expression() {
349 const negative = this.take('-');
350 let result = this.term();
351 if (negative) result = { kind: 'neg', operand: result };
352 while (this.token.kind === '+' || this.token.kind === '-') {
353 const kind = this.token.kind;
354 this.position++;
355 result = { kind, left: result, right: this.term() };
356 }
357 return result;
358 }
359 term() {
360 let result = this.factor();
361 while (['*', '/', 'x', 'name', '('].includes(this.token.kind)) {
362 let kind = '*';
363 if (this.token.kind === '*' || this.token.kind === '/') {
364 kind = this.token.kind;
365 this.position++;
366 }
367 result = { kind, left: result, right: this.factor() };
368 }
369 return result;
370 }
371 factor() {
372 let result = this.primary();
373 if (this.take('^')) {
374 if (this.token.kind !== 'number') this.syntaxError();
375 const exponent = Number(this.token.word);
376 this.position++;
377 result = { kind: 'pow', operand: result, exponent };
378 }
379 return result;
380 }
381 primary() {
382 const token = this.token;
383 if (this.take('number')) return { kind: 'number', value: new Fraction(BigInt(token.word)) };
384 if (this.take('x')) return { kind: 'x' };
385 if (this.take('name')) {
386 let derivatives = 0;
387 while (this.take("'")) derivatives++;
388 const reference = { kind: 'name', name: token.word, derivatives };
389 if (!this.take('(')) return reference;
390 const argument = this.expression();
391 this.expect(')');
392 return { kind: 'call', left: reference, right: argument };
393 }
394 if (this.take('(')) {
395 const result = this.expression();
396 this.expect(')');
397 return result;
398 }
399 return this.syntaxError();
400 }
401}

403function evaluateExpression(node, lookup) {
404 if (node.kind === 'number') return Polynomial.constant(node.value);
405 if (node.kind === 'x') return X;
406 if (node.kind === 'name') return lookup(node.name).derivative(node.derivatives);
407 if (node.kind === 'neg') return evaluateExpression(node.operand, lookup).neg();
408 if (node.kind === 'pow') {
409 const value = evaluateExpression(node.operand, lookup);
410 degreeLimit(!value.zero && value.degree * node.exponent > 30);
411 return value.pow(node.exponent);
412 }
413 const left = evaluateExpression(node.left, lookup), right = evaluateExpression(node.right, lookup);
414 if (node.kind === '+') return left.add(right);
415 if (node.kind === '-') return left.sub(right);
416 if (node.kind === '*') { productLimit(left, right); return left.mul(right); }
417 if (node.kind === 'call') { compositionLimit(left, right); return left.compose(right); }
418 if (right.zero) fail('DIVISION BY ZERO');
419 if (right.degree > 0) fail('NONCONSTANT DIVISOR');
420 return left.divideScalar(right.leading);
421}

431class Calculator {
432 polynomials = new Map();
433
434 lookup(name) {
435 const polynomial = this.polynomials.get(name);
436 if (polynomial === undefined) fail(`UNKNOWN ${name}`);
437 return polynomial;
438 }
439
440 execute(source) {
441 const words = source.split(' '), command = words[0], name = words[1];
442 const pending = [], output = [];
443 const store = (name, polynomial) => {
444 pending.push([name, polynomial]);
445 output.push(`${name} = ${polynomial.notation()}`);
446 };
447 if (command === 'DEF') {
448 const expression = source.slice(command.length + name.length + 2);
449 const syntax = new ExpressionParser(expression).parse();
450 store(name, evaluateExpression(syntax, name => this.lookup(name)));
451 } else {
452 if (command === 'DIV' && words[1] === words[2]
453 || command === 'XGCD' && new Set(words.slice(1, 4)).size < 3) fail('SAME NAME');
454 const [a, b, c, d] = (REFERENCES[command] ?? []).map(index => this.lookup(words[index]));
455 let literalWords = [];
456 if (command === 'AREA' || command === 'STURM') literalWords = words.slice(2, 4);
457 else if (command === 'TANGENT') literalWords = words.slice(3, 4);
458 else if (command === 'TAYLOR') literalWords = words.slice(2, 3);
459 else if (command === 'EVAL' || command === 'INTERP') literalWords = words.slice(2);
460 const values = literalWords.map(word => word === '-inf' || word === 'inf' ? word : Fraction.parse(word));
461 switch (command) {
462 case 'PRINT': output.push(`${name} = ${a.notation()}`); break;
463 case 'DELETE': this.polynomials.delete(name); output.push(`deleted ${name}`); break;
464 case 'LIST':
465 for (const name of [...this.polynomials.keys()].sort()) output.push(`${name} = ${this.lookup(name).notation()}`);
466 if (output.length === 0) output.push('no polynomials');
467 break;
468 case 'DEGREE': output.push(`deg(${name}) = ${a.zero ? '-inf' : a.degree}`); break;
469 case 'ADD': store(name, a.add(b)); break;
470 case 'SUB': store(name, a.sub(b)); break;
471 case 'MUL': productLimit(a, b); store(name, a.mul(b)); break;
472 case 'POW': {
473 const exponent = Number(words[3]);
474 degreeLimit(!a.zero && a.degree * exponent > 30);
475 store(name, a.pow(exponent));
476 break;
477 }
478 case 'DIV': {
479 if (b.zero) fail('DIVISION BY ZERO');
480 const [quotient, remainder] = a.divide(b);
481 store(name, quotient); store(words[2], remainder);
482 break;
483 }
484 case 'GCD': store(name, polynomialGcd(a, b)); break;
485 case 'XGCD': extendedGcd(a, b).forEach((value, i) => store(words[i + 1], value)); break;
486 case 'LCM': {
487 if (a.zero || b.zero) store(name, new Polynomial());
488 else {
489 const common = polynomialGcd(a, b);
490 degreeLimit(a.degree + b.degree - common.degree > 30);
491 store(name, a.divide(common)[0].mul(b).monic());
492 }
493 break;
494 }
495 case 'INVMOD': {
496 if (b.zero) fail('DIVISION BY ZERO');
497 if (b.degree === 0) fail('DEGREE TOO LOW');
498 const [common, inverse] = extendedGcd(a, b);
499 if (!isOne(common)) fail('NOT COPRIME');
500 store(name, inverse.mod(b));
501 break;
502 }
503 case 'CRT': {
504 if (b.zero || d.zero) fail('DIVISION BY ZERO');
505 degreeLimit(b.degree + d.degree > 30);
506 const [common, inverse] = extendedGcd(b, d);
507 if (!isOne(common)) fail('NOT COPRIME');
508 const first = a.mod(b), second = c.mod(d), correction = second.sub(first).mul(inverse).mod(d);
509 store(name, first.add(b.mul(correction)));
510 break;
511 }
512 case 'MODPOW': {
513 if (b.zero) fail('DIVISION BY ZERO');
514 degreeLimit(2 * b.degree - 2 > 30);
515 let result = UNIT.mod(b), base = a.mod(b), exponent = Number(words[3]);
516 while (exponent > 0) {
517 if (exponent % 2) result = result.mul(base).mod(b);
518 exponent = Math.floor(exponent / 2);
519 if (exponent) base = base.mul(base).mod(b);
520 }
521 store(name, result);
522 break;
523 }
524 case 'DERIV': store(name, a.derivative(Number(words[3]))); break;
525 case 'INTEG': degreeLimit(!a.zero && a.degree + 1 > 30); store(name, a.integral()); break;
526 case 'AREA': {
527 const integral = a.integral(), [u, v] = values;
528 output.push(`integral(${name}, ${u}, ${v}) = ${integral.evaluate(v).sub(integral.evaluate(u))}`);
529 break;
530 }
531 case 'TANGENT': {
532 const point = values[0], value = a.evaluate(point), slope = a.derivative().evaluate(point);
533 store(name, linear(value.sub(slope.mul(point)), slope));
534 break;
535 }
536 case 'TAYLOR': {
537 const point = values[0];
538 const variable = point.n === 0n ? 'x' : point.n > 0n ? `(x - ${point})` : `(x + ${point.neg()})`;
539 output.push(`taylor(${name}, ${point}) = ${a.compose(linear(point)).notation(variable)}`);
540 break;
541 }
542 case 'COMPOSE': compositionLimit(a, b); store(name, a.compose(b)); break;
543 case 'EVAL': for (const point of values) output.push(`${name}(${point}) = ${a.evaluate(point)}`); break;
544 case 'ROOTS': case 'FACTOR': {
545 requireNonzero(a);
546 const { roots, remaining } = rationalRoots(a);
547 if (command === 'ROOTS') {
548 output.push(`roots(${name}) = ` + (roots.map(({ root, multiplicity }) => `${root} (${multiplicity})`).join(', ') || 'none'));
549 } else {
550 let result = `factor(${name}) = ${a.content()}`;
551 for (const { root, multiplicity } of roots) {
552 const factor = root.n === 0n ? 'x' : `(${linear(new Fraction(-root.n), new Fraction(root.d)).notation()})`;
553 result += ` * ${factor}` + (multiplicity > 1 ? `^${multiplicity}` : '');
554 }
555 if (remaining.degree >= 1) result += ` * (${remaining.notation()})`;
556 output.push(result);
557 }
558 break;
559 }
560 case 'SQFREE': {
561 requireNonzero(a);
562 let result = `sqfree(${name}) = ${a.content()}`;
563 if (a.degree >= 1) {
564 const monic = a.monic();
565 let common = polynomialGcd(monic, monic.derivative()), remaining = monic.divide(common)[0];
566 for (let multiplicity = 1; remaining.degree >= 1; multiplicity++) {
567 const shared = polynomialGcd(remaining, common), factor = remaining.divide(shared)[0];
568 if (factor.degree >= 1) {
569 const notation = factor.primitive().notation();
570 result += ' * ' + (notation === 'x' ? 'x' : `(${notation})`) + (multiplicity > 1 ? `^${multiplicity}` : '');
571 }
572 common = common.divide(shared)[0];
573 remaining = shared;
574 }
575 }
576 output.push(result);
577 break;
578 }
579 case 'DESCARTES': {
580 requireNonzero(a);
581 const signs = a.coefficients.map(coefficient => coefficient.compare(ZERO));
582 const positive = signChanges(signs.toReversed());
583 const negative = signChanges(signs.map((sign, i) => i % 2 ? -sign : sign).reverse());
584 output.push(`descartes(${name}) = positive ${positive}, negative ${negative}`);
585 break;
586 }
587 case 'STURM': {
588 requireNonzero(a);
589 const [u, v] = values;
590 if (compareEndpoints(u, v) >= 0) fail('BAD INTERVAL');
591 const sequence = sturmSequence(a);
592 output.push(`sturm(${name}, ${u}, ${v}) = ${variations(sequence, u) - variations(sequence, v)}`);
593 break;
594 }
595 case 'ISOLATE': {
596 requireNonzero(a);
597 const precision = Number(words[2]), intervals = isolate(a, precision);
598 output.push(`isolate(${name}, ${precision}) = ${intervals.join(', ') || 'none'}`);
599 break;
600 }
601 case 'RESULTANT':
602 requireNonzero(a); requireNonzero(b);
603 output.push(`res(${name}, ${words[2]}) = ${resultant(a, b)}`);
604 break;
605 case 'DISCRIM': {
606 requireNonzero(a);
607 if (a.degree === 0) fail('DEGREE TOO LOW');
608 let discriminant = resultant(a, a.derivative()).div(a.leading);
609 if (a.degree * (a.degree - 1) / 2 % 2) discriminant = discriminant.neg();
610 output.push(`disc(${name}) = ${discriminant}`);
611 break;
612 }
613 case 'POWERSUM': {
614 requireNonzero(a);
615 if (a.degree === 0) fail('DEGREE TOO LOW');
616 const monic = a.monic(), sums = [ZERO], count = Number(words[2]);
617 for (let j = 1; j <= count; j++) {
618 let sum = j <= a.degree ? integer(j).mul(monic.coefficient(a.degree - j)) : ZERO;
619 for (let i = 1; i <= Math.min(j - 1, a.degree); i++) sum = sum.add(monic.coefficient(a.degree - i).mul(sums[j - i]));
620 sums.push(sum.neg());
621 }
622 output.push(`powersum(${name}) = ${sums.slice(1).join(', ')}`);
623 break;
624 }
625 case 'INTERP': store(name, interpolate(values)); break;
626 case 'SPECIAL': store(name, special(words[2], Number(words[3]))); break;
627 }
628 }
629 for (const [name, polynomial] of pending) this.polynomials.set(name, polynomial);
630 return output;
631 }
632}
