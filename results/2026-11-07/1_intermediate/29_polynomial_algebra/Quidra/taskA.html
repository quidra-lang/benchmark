15Ratio ratio_parts(real value)
16 text := string(value)
17 fields := text.split("/")
18 if #fields == 2
19 return Ratio(parse_integer(fields[0]), parse_integer(fields[1]))
20 fields = text.split(".")
21 if #fields == 1
22 return Ratio(parse_integer(text), 1)

24 numerator := parse_integer(fields[0]) * denominator + (if text.starts_with("-") then -1 else 1) * parse_integer(fields[1])

86Polynomial product(const Polynomial left, const Polynomial right)

115Division divide_polynomials(const Polynomial dividend, const Polynomial divisor)
116 quotient := array(nat(math.max(0, degree(dividend) - degree(divisor) + 1)), fill = 0.0)
117 remainder := dividend
118 while degree(remainder) >= 0 and degree(remainder) >= degree(divisor)
119 factor := leading(remainder) / leading(divisor)
120 offset := degree(remainder) - degree(divisor)
121 quotient[offset] = factor
122 for index := divisor.coefficients.indices()[]
123 remainder.coefficients[offset + int(index)] -= factor * divisor.coefficients[index]
124 remainder = normalized(remainder.coefficients)
125 return Division(normalized(quotient), remainder)

165Bezout extended_gcd(Polynomial first, Polynomial second)
166 s0 := constant(1.0)
167 s1 := Polynomial()
168 t0 := Polynomial()
169 t1 := constant(1.0)
170 while degree(second) >= 0
171 division := divide_polynomials(first, second)
172 following_s := subtract(s0, product(division.quotient, s1))
173 following_t := subtract(t0, product(division.quotient, t1))
174 first = second
175 second = division.remainder
176 s0 = s1
177 s1 = following_s
178 t0 = t1
179 t1 = following_t
180 if degree(first) >= 0
181 lead := leading(first)
182 first = divided(first, lead)
183 s0 = divided(s0, lead)
184 t0 = divided(t0, lead)
185 return Bezout(first, s0, t0)

187real content(const Polynomial polynomial)
188 denominator := 1
189 numerator := 0
190 for value := polynomial.coefficients[]
191 parts := ratio_parts(value)
192 denominator = denominator * parts.denominator / gcd_integer(denominator, parts.denominator)
193 numerator = gcd_integer(numerator, parts.numerator)
194 result := real(numerator) / real(denominator)
195 return if leading(polynomial) < 0.0 then -result else result

386Polynomial | error evaluate_node(const Algebra algebra, const Node[] nodes, int index)
387 item := nodes[index]
388 if item.kind == Operation.Literal
389 return constant(real(item.number))
390 if item.kind == Operation.Variable
391 return Polynomial([0.0, 1.0])
392 if item.kind == Operation.Reference
393 referenced := try lookup(algebra, item.name)
394 return differentiated(referenced, item.number)
395 left := try evaluate_node(algebra, nodes, item.left)
396 if item.kind == Operation.Negative
397 return negated(left)
398 if item.kind == Operation.Power
399 return checked_power(left, item.number)
400 right := try evaluate_node(algebra, nodes, item.right)
401 if item.kind == Operation.Add
402 return add(left, right)
403 if item.kind == Operation.Subtract
404 return subtract(left, right)
405 if item.kind == Operation.Multiply
406 return checked_product(left, right)
407 if item.kind == Operation.Compose
408 return checked_composition(left, right)
409 if degree(right) < 0
410 return error("DIVISION BY ZERO")
411 if degree(right) > 0
412 return error("NONCONSTANT DIVISOR")
413 return divided(left, right.coefficients[0])

422RationalRoots rational_roots(const Polynomial polynomial)
423 remaining := primitive(polynomial)
424 zeros := 0
425 while remaining.coefficients[0] == 0.0
426 remaining = Polynomial(remaining.coefficients[1:])
427 zeros += 1
428 common_denominator := int(leading(remaining))
429 int[] candidate_keys := []
430 for numerator := divisors(math.abs(int(remaining.coefficients[0])))[]
431 for denominator := divisors(common_denominator)[]
432 if gcd_integer(numerator, denominator) == 1
433 key := numerator * (common_denominator / denominator)
434 candidate_keys = candidate_keys.append(key).append(-key)
435 Root[] roots := []
436 for key := candidate_keys.sorted()[]
437 candidate := real(key) / real(common_denominator)
438 parts := ratio_parts(candidate)
439 factor := Polynomial([real(-parts.numerator), real(parts.denominator)])
440 multiplicity := 0
441 while degree(remaining) >= 1 and evaluate(remaining, candidate) == 0.0
442 remaining = divide_polynomials(remaining, factor).quotient
443 multiplicity += 1
444 if multiplicity > 0
445 roots = roots.append(Root(candidate, multiplicity))
446 if zeros > 0
447 insertion := 0
448 while insertion < int(#roots) and roots[insertion].value < 0.0
449 insertion += 1
450 roots = roots[0:<insertion].append(Root(0.0, zeros)).concat(roots[insertion:])
451 return RationalRoots(roots, remaining)

583 xs := [for index := (0:<int(#values):2)[] then values[index]]
584 differences := [for index := (1:<int(#values))[] if index % 2 == 1 then values[index]]
585 count := int(#xs)
586 for: 1 <= index < count
587 if includes(xs[0:<index], xs[index])
588 return error("DUPLICATE X " + fraction_text(xs[index]))
589 for: 1 <= order < count

599 return error("UNKNOWN KIND " + kind)
600 if order > 30
601 return error("DEGREE LIMIT")
602 previous := constant(1.0)
603 if order == 0
604 return previous
605 current := Polynomial([0.0, 1.0])
606 if kind == "chebyshev2" or kind == "hermite"
607 current = Polynomial([0.0, 2.0])
608 elif kind == "laguerre"
609 current = Polynomial([1.0, -1.0])
610 for: 1 <= index < order
611 factor := Polynomial([0.0, 2.0])

616 elif kind == "hermite"
617 multiplier = real(2 * index)
618 elif kind == "laguerre"
619 factor = Polynomial([real(2 * index + 1), -1.0])
620 multiplier = real(index)
621 following := subtract(product(factor, current), scaled(previous, multiplier))
622 if kind == "legendre" or kind == "laguerre"
623 following = scaled(following, 1.0 / real(index + 1))
624 previous = current
625 current = following
626 return current

670 elif command == "TANGENT"

678 low = try bound_literal(args[2])
679 high = try bound_literal(args[3])
680 first := if #operands > 0 then operands[0] else Polynomial()
681 second := if #operands > 1 then operands[1] else Polynomial()
682 name := if #args > 1 then args[1] else ""
683 if command == "PRINT"
684 emit(name + " = " + notation(first))
685 elif command == "DELETE"
686 .algebra.polynomials.remove(name)
687 emit("deleted " + name)
688 elif command == "LIST"
689 names := algebra.polynomials.keys().sorted()
690 if #names == 0
691 emit("no polynomials")
692 for key := names[]
693 stored := try lookup(algebra, key)
694 emit(key + " = " + notation(stored))
695 elif command == "DEGREE"
696 amount := if degree(first) < 0 then "-inf" else string(degree(first))
697 emit("deg({name}) = {amount}")
698 elif command == "ADD"
699 save(&algebra, name, add(first, second))
700 elif command == "SUB"
701 save(&algebra, name, subtract(first, second))
702 elif command == "MUL"
703 save(&algebra, name, try checked_product(first, second))
704 elif command == "POW"
705 save(&algebra, name, try checked_power(first, parse_integer(args[3])))
706 elif command == "DIV"
707 if degree(second) < 0
708 return error("DIVISION BY ZERO")
709 division := divide_polynomials(first, second)
710 save(&algebra, name, division.quotient)
711 save(&algebra, args[2], division.remainder)
712 elif command == "GCD"
713 save(&algebra, name, gcd_polynomials(first, second))
714 elif command == "XGCD"
715 result := extended_gcd(first, second)
716 save(&algebra, name, result.common)
717 save(&algebra, args[2], result.left)
718 save(&algebra, args[3], result.right)
719 elif command == "LCM"
720 result := Polynomial()
721 if degree(first) >= 0 and degree(second) >= 0
722 common := gcd_polynomials(first, second)
723 if degree(first) + degree(second) - degree(common) > 30
724 return error("DEGREE LIMIT")
725 result = monic(product(divide_polynomials(first, common).quotient, second))
726 save(&algebra, name, result)
727 elif command == "INVMOD"
728 if degree(second) < 0
729 return error("DIVISION BY ZERO")
730 if degree(second) == 0
731 return error("DEGREE TOO LOW")
732 result := extended_gcd(first, second)
733 if result.common != constant(1.0)
734 return error("NOT COPRIME")
735 save(&algebra, name, divide_polynomials(result.left, second).remainder)
736 elif command == "CRT"
737 m1 := operands[1]
738 a2 := operands[2]
739 m2 := operands[3]
740 if degree(m1) < 0 or degree(m2) < 0
741 return error("DIVISION BY ZERO")
742 if degree(m1) + degree(m2) > 30
743 return error("DEGREE LIMIT")
744 result := extended_gcd(m1, m2)
745 if result.common != constant(1.0)
746 return error("NOT COPRIME")
747 r1 := divide_polynomials(first, m1).remainder
748 r2 := divide_polynomials(a2, m2).remainder
749 correction := divide_polynomials(product(subtract(r2, r1), result.left), m2).remainder
750 save(&algebra, name, add(r1, product(m1, correction)))
751 elif command == "MODPOW"
752 if degree(second) < 0
753 return error("DIVISION BY ZERO")
754 if 2 * degree(second) - 2 > 30
755 return error("DEGREE LIMIT")
756 result := divide_polynomials(constant(1.0), second).remainder
757 base := divide_polynomials(first, second).remainder
758 exponent := parse_integer(args[3])
759 while exponent > 0
760 if exponent % 2 != 0
761 result = divide_polynomials(product(result, base), second).remainder
762 exponent /= 2
763 if exponent > 0
764 base = divide_polynomials(product(base, base), second).remainder
765 save(&algebra, name, result)
766 elif command == "DERIV"
767 save(&algebra, name, differentiated(first, parse_integer(args[3])))
768 elif command == "INTEG"
769 if degree(first) + 1 > 30
770 return error("DEGREE LIMIT")
771 save(&algebra, name, integral(first))
772 elif command == "AREA"
773 antiderivative := integral(first)
774 value := evaluate(antiderivative, values[1]) - evaluate(antiderivative, values[0])
775 emit("integral({name}, {fraction_text(values[0])}, {fraction_text(values[1])}) = {fraction_text(value)}")
776 elif command == "TANGENT"
777 height := evaluate(first, values[0])
778 slope := evaluate(derivative(first), values[0])
779 save(&algebra, name, normalized([height - slope * values[0], slope]))
780 elif command == "TAYLOR"
781 center := values[0]
782 variable := "x"
783 if center > 0.0
784 variable = "(x - {fraction_text(center)})"
785 elif center < 0.0
786 variable = "(x + {fraction_text(-center)})"
787 emit("taylor({name}, {fraction_text(center)}) = {notation(compose(first, Polynomial([center, 1.0])), variable)}")
788 elif command == "COMPOSE"
789 save(&algebra, name, try checked_composition(first, second))
790 elif command == "EVAL"
791 for point := values[]
792 emit("{name}({fraction_text(point)}) = {fraction_text(evaluate(first, point))}")
793 elif command == "INTERP"
794 save(&algebra, name, try interpolate(values))
795 elif command == "SPECIAL"
796 save(&algebra, name, try special(args[2], parse_integer(args[3])))
797 else
798 if degree(first) < 0
799 return error("ZERO POLYNOMIAL")
800 if command == "ROOTS"
801 roots := rational_roots(first)
802 terms := [for entry := roots.roots[] then "{fraction_text(entry.value)} ({entry.multiplicity})"]
803 text := if #terms == 0 then "none" else terms.join(", ")
804 emit("roots({name}) = {text}")
805 elif command == "FACTOR"
806 emit("factor({name}) = {factorization(first)}")
807 elif command == "SQFREE"
808 emit("sqfree({name}) = {squarefree_factors(first)}")
809 elif command == "DESCARTES"
810 positive := sign_changes([for index := (int(degree(first)):>-1)[] then sign_of(first.coefficients[index])])
811 negative := sign_changes([for index := (int(degree(first)):>-1)[] then sign_of(first.coefficients[index]) * (if index % 2 == 0 then 1 else -1)])
812 emit("descartes({name}) = positive {positive}, negative {negative}")
813 elif command == "STURM"
814 if not before(low, high)
815 return error("BAD INTERVAL")
816 count := root_count(sturm_sequence(first), low, high)
817 emit("sturm({name}, {bound_text(low)}, {bound_text(high)}) = {count}")
818 elif command == "ISOLATE"
819 precision := parse_integer(args[2])
820 intervals := isolate(first, precision)
821 terms := [for interval := intervals[] then "({fraction_text(interval.low)}, {fraction_text(interval.high)}]"]
822 text := if #terms == 0 then "none" else terms.join(", ")
823 emit("isolate({name}, {precision}) = {text}")
824 elif command == "RESULTANT"
825 if degree(second) < 0
826 return error("ZERO POLYNOMIAL")
827 emit("res({name}, {args[2]}) = {fraction_text(resultant(first, second))}")
828 elif command == "DISCRIM" or command == "POWERSUM"
829 order := degree(first)
830 if order == 0
831 return error("DEGREE TOO LOW")
832 if command == "DISCRIM"
833 value := resultant(first, derivative(first)) / leading(first)
834 if order * (order - 1) / 2 % 2 != 0
835 value = -value
836 emit("disc({name}) = {fraction_text(value)}")
837 else
838 unit := monic(first)
839 sums := [0.0]
840 for: 1 <= exponent <= parse_integer(args[2])
841 total := if exponent <= order then real(exponent) * unit.coefficients[order - exponent] else 0.0
842 for: 1 <= index <= math.min(exponent - 1, order)
843 total += unit.coefficients[order - index] * sums[exponent - index]
844 sums = sums.append(-total)
845 terms := [for value := sums[1:][] then fraction_text(value)]
846 emit("powersum({name}) = " + terms.join(", "))
847 return void
