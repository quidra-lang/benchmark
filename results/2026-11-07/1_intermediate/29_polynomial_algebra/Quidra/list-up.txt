class Ratio(int numerator, int denominator)
int parse_integer(string text)
    return int(text)
int gcd_integer(int left, int right)
    left = math.abs(left)
    right = math.abs(right)
    while right != 0
        left, right = right, left % right
    return left
Ratio ratio_parts(real value)
    text := string(value)
    fields := text.split("/")
    if #fields == 2
        return Ratio(parse_integer(fields[0]), parse_integer(fields[1]))
    fields = text.split(".")
    if #fields == 1
        return Ratio(parse_integer(text), 1)
    denominator := 10 ^ int(#fields[1])
    numerator := parse_integer(fields[0]) * denominator + (if text.starts_with("-") then -1 else 1) * parse_integer(fields[1])
    divisor := gcd_integer(numerator, denominator)
    return Ratio(numerator / divisor, denominator / divisor)
string fraction_text(real value)
    parts := ratio_parts(value)
    return if parts.denominator == 1 then string(parts.numerator) else "{parts.numerator}/{parts.denominator}"
real | error rational(string text)
    fields := text.split("/")
    numerator := parse_integer(fields[0])
    denominator := if #fields == 2 then parse_integer(fields[1]) else 1
    if denominator == 0
        return error("ZERO DENOMINATOR")
    return real(numerator) / real(denominator)
bool includes<T>(const T[] values, T sought)
    for item := values[]
        if item == sought
            return true
void emit(string text)
    println(text)
class Polynomial(real[] coefficients := [])
Polynomial normalized(const real[] coefficients)
    count := int(#coefficients)
    while count > 0 and coefficients[count - 1] == 0.0
        count -= 1
    return Polynomial(coefficients[0:<count])
Polynomial constant(real value)
    return if value == 0.0 then Polynomial() else Polynomial([value])
int degree(const Polynomial polynomial)
    return int(#polynomial.coefficients) - 1
real coefficient(const Polynomial polynomial, int index)
    return if index <= degree(polynomial) then polynomial.coefficients[index] else 0.0
Polynomial add(const Polynomial left, const Polynomial right)
    count := math.max(degree(left), degree(right)) + 1
    return normalized([for index := (0:<int(count))[] then coefficient(left, index) + coefficient(right, index)])
Polynomial product(const Polynomial left, const Polynomial right)
    if degree(left) < 0 or degree(right) < 0
        return Polynomial()
    coefficients := array(nat(degree(left) + degree(right) + 1), fill = 0.0)
    for first := left.coefficients.indices()[]
        for second := right.coefficients.indices()[]
            coefficients[first + second] += left.coefficients[first] * right.coefficients[second]
    return normalized(coefficients)
string notation(const Polynomial polynomial, variable := "x")
    result := ""
    for: degree(polynomial) >= index >= 0 step -1
        value := polynomial.coefficients[index]
        if value == 0.0
            continue
        term := ""
        if index == 0 or math.abs(value) != 1.0
            term = fraction_text(math.abs(value))
        if index > 0
            term += variable
        if index > 1
            term += "^{index}"
        if result == ""
            if value < 0.0
                result = "-"
        else
            result += if value < 0.0 then " - " else " + "
        result += term
    return if result == "" then "0" else result
class Algebra()
    polynomials := map<string, Polynomial>()
void save(Algebra &algebra, string name, const Polynomial value)
    .algebra.polynomials.set(name, value)
    emit(name + " = " + notation(value))
Polynomial | error interpolate(const real[] values)
    xs := [for index := (0:<int(#values):2)[] then values[index]]
    differences := [for index := (1:<int(#values))[] if index % 2 == 1 then values[index]]
    count := int(#xs)
    for: 1 <= index < count
        if includes(xs[0:<index], xs[index])
            return error("DUPLICATE X " + fraction_text(xs[index]))
    for: 1 <= order < count
        for: count - 1 >= index >= order step -1
            differences[index] = (differences[index] - differences[index - 1]) / (xs[index] - xs[index - order])
    result := Polynomial()
    for: count - 1 >= index >= 0 step -1
        result = add(product(result, Polynomial([-xs[index], 1.0])), constant(differences[index]))
    return result
void | error execute(Algebra &algebra, string line)
    args := line.split(" ")
    command := args[0]
    real[] values := []
    literal_start := -1
    if includes(["AREA", "TAYLOR", "EVAL", "INTERP"], command)
        literal_start = 2
    if literal_start >= 0
        for: literal_start <= index < int(#args)
            values = values.append(try rational(args[index]))
    name := if #args > 1 then args[1] else ""
    elif command == "INTERP"
        save(&algebra, name, try interpolate(values))
