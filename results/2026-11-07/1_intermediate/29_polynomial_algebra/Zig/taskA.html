54fn equal(a: []const u8, b: []const u8) bool {

57fn integer(text: []const u8) i64 {
58 return std.fmt.parseInt(i64, text, 10) catch unreachable;
59}
60fn natural(text: []const u8) usize {
61 return @intCast(integer(text));
62}
63fn line(out: *Writer, comptime fmt: []const u8, args: anytype) Writer.Error!void {
64 try out.print(fmt ++ "\n", args);

66fn absolute(value: i64) i64 {
67 return if (value < 0) -value else value;
68}
69fn gcd(a: i64, b: i64) i64 {
70 var x = absolute(a);
71 var y = absolute(b);
72 while (y != 0) {
73 const remainder = @mod(x, y);
74 x = y;
75 y = remainder;
76 }
77 return x;
78}

80const Fraction = struct {
81 n: i64 = 0,
82 d: i64 = 1,
83 const zero: Fraction = .{};
84 const one: Fraction = .{ .n = 1 };
85
86 fn init(n: i64, d: i64) Fraction {
87 const divisor = gcd(n, d) * @as(i64, if (d < 0) -1 else 1);
88 return .{ .n = @divTrunc(n, divisor), .d = @divTrunc(d, divisor) };
89 }
90 fn whole(n: i64) Fraction {
91 return .{ .n = n };
92 }
93 fn parse(text: []const u8) SemanticError!Fraction {
94 const slash = std.mem.findScalar(u8, text, '/');
95 const n = integer(if (slash) |i| text[0..i] else text);
96 const d = if (slash) |i| integer(text[i + 1 ..]) else 1;
97 if (d == 0) return error.ZeroDenominator;
98 return init(n, d);
99 }
100 fn add(a: Fraction, b: Fraction) Fraction {
101 return init(a.n * b.d + b.n * a.d, a.d * b.d);
102 }
103 fn sub(a: Fraction, b: Fraction) Fraction {
104 return init(a.n * b.d - b.n * a.d, a.d * b.d);
105 }
106 fn mul(a: Fraction, b: Fraction) Fraction {
107 return init(a.n * b.n, a.d * b.d);
108 }
109 fn div(a: Fraction, b: Fraction) Fraction {
110 return init(a.n * b.d, a.d * b.n);
111 }
112 fn neg(a: Fraction) Fraction {
113 return .{ .n = -a.n, .d = a.d };
114 }
115 fn abs(a: Fraction) Fraction {
116 return .{ .n = absolute(a.n), .d = a.d };
117 }
118 fn isZero(a: Fraction) bool {
119 return a.n == 0;
120 }
121 fn sign(a: Fraction) i32 {
122 return if (a.n < 0) -1 else if (a.n > 0) 1 else 0;
123 }
124 fn eq(a: Fraction, b: Fraction) bool {
125 return a.n == b.n and a.d == b.d;
126 }
127 fn less(a: Fraction, b: Fraction) bool {
128 return a.n * b.d < b.n * a.d;
129 }
130 fn lessThan(_: void, a: Fraction, b: Fraction) bool {
131 return a.less(b);
132 }
133 fn power(a: Fraction, exponent: usize) Fraction {
134 var result = one;
135 for (0..exponent) |_| result = result.mul(a);
136 return result;
137 }
138 pub fn format(value: Fraction, out: *Writer) Writer.Error!void {
139 try out.print("{d}", .{value.n});
140 if (value.d != 1) try out.print("/{d}", .{value.d});
141 }

144const Polynomial = struct {
145 coefficients: []const Fraction = &.{},
146 const zero: Polynomial = .{};
147 const one: Polynomial = .{ .coefficients = &.{Fraction.one} };
148 const variable: Polynomial = .{ .coefficients = &.{ Fraction.zero, Fraction.one } };
149 const Division = struct { quotient: Polynomial, remainder: Polynomial };
150 const Extended = struct { g: Polynomial, s: Polynomial, t: Polynomial };
151
152 fn fromOwned(values: []const Fraction) Polynomial {
153 var length = values.len;
154 while (length > 0 and values[length - 1].isZero()) length -= 1;
155 return .{ .coefficients = values[0..length] };
156 }
157 fn init(gpa: Allocator, values: []const Fraction) Allocator.Error!Polynomial {
158 return fromOwned(try gpa.dupe(Fraction, values));
159 }
160 fn constant(gpa: Allocator, value: Fraction) Allocator.Error!Polynomial {
161 return if (value.isZero()) zero else init(gpa, &.{value});
162 }
163 fn degree(value: Polynomial) i32 {
164 return @as(i32, @intCast(value.coefficients.len)) - 1;
165 }
166 fn isZero(value: Polynomial) bool {
167 return value.coefficients.len == 0;
168 }
169 fn leading(value: Polynomial) Fraction {
170 return value.coefficients[value.coefficients.len - 1];
171 }
172 fn at(value: Polynomial, index: usize) Fraction {
173 return if (index < value.coefficients.len) value.coefficients[index] else Fraction.zero;
174 }
175 fn add(a: Polynomial, gpa: Allocator, b: Polynomial) Allocator.Error!Polynomial {
176 const result = try gpa.alloc(Fraction, @max(a.coefficients.len, b.coefficients.len));
177 for (result, 0..) |*value, i| value.* = a.at(i).add(b.at(i));
178 return fromOwned(result);
179 }
180 fn sub(a: Polynomial, gpa: Allocator, b: Polynomial) Allocator.Error!Polynomial {
181 const result = try gpa.alloc(Fraction, @max(a.coefficients.len, b.coefficients.len));
182 for (result, 0..) |*value, i| value.* = a.at(i).sub(b.at(i));
183 return fromOwned(result);
184 }
185 fn neg(a: Polynomial, gpa: Allocator) Allocator.Error!Polynomial {
186 const result = try gpa.alloc(Fraction, a.coefficients.len);
187 for (result, a.coefficients) |*value, coefficient| value.* = coefficient.neg();
188 return fromOwned(result);
189 }
190 fn mul(a: Polynomial, gpa: Allocator, b: Polynomial) Allocator.Error!Polynomial {
191 if (a.isZero() or b.isZero()) return zero;
192 const result = try gpa.alloc(Fraction, a.coefficients.len + b.coefficients.len - 1);
193 @memset(result, Fraction.zero);
194 for (a.coefficients, 0..) |left, i| {
195 for (b.coefficients, 0..) |right, j| result[i + j] = result[i + j].add(left.mul(right));
196 }
197 return fromOwned(result);
198 }
199 fn scale(a: Polynomial, gpa: Allocator, factor: Fraction) Allocator.Error!Polynomial {
200 const result = try gpa.alloc(Fraction, a.coefficients.len);
201 for (result, a.coefficients) |*value, coefficient| value.* = coefficient.mul(factor);
202 return fromOwned(result);
203 }
204 fn divideScalar(a: Polynomial, gpa: Allocator, divisor: Fraction) Allocator.Error!Polynomial {
205 const result = try gpa.alloc(Fraction, a.coefficients.len);
206 for (result, a.coefficients) |*value, coefficient| value.* = coefficient.div(divisor);
207 return fromOwned(result);
208 }
209 fn monic(a: Polynomial, gpa: Allocator) Allocator.Error!Polynomial {
210 return if (a.isZero()) zero else a.divideScalar(gpa, a.leading());
211 }
212 fn power(a: Polynomial, gpa: Allocator, exponent: usize) Allocator.Error!Polynomial {
213 if (exponent == 0) return one;
214 if (a.isZero()) return zero;
215 if (a.degree() == 0 and a.leading().d == 1 and absolute(a.leading().n) == 1) {
216 return if (a.leading().n == 1 or exponent % 2 == 0) one else a;
217 }
218 var result = one;
219 for (0..exponent) |_| result = try result.mul(gpa, a);
220 return result;
221 }
222 fn evaluate(a: Polynomial, point: Fraction) Fraction {
223 var result = Fraction.zero;
224 var i = a.coefficients.len;
225 while (i > 0) {
226 i -= 1;
227 result = result.mul(point).add(a.at(i));
228 }
229 return result;
230 }
231 fn compose(a: Polynomial, gpa: Allocator, inner: Polynomial) Allocator.Error!Polynomial {
232 var result = zero;
233 var i = a.coefficients.len;
234 while (i > 0) {
235 i -= 1;
236 result = try (try result.mul(gpa, inner)).add(gpa, try constant(gpa, a.at(i)));
237 }
238 return result;
239 }
240 fn derivative(a: Polynomial, gpa: Allocator, order: usize) Allocator.Error!Polynomial {
241 var result = a;
242 for (0..order) |_| {
243 const values = try gpa.alloc(Fraction, result.coefficients.len -| 1);
244 for (values, 1..) |*value, i| value.* = Fraction.whole(@intCast(i)).mul(result.at(i));
245 result = fromOwned(values);
246 }
247 return result;
248 }
249 fn integral(a: Polynomial, gpa: Allocator) Allocator.Error!Polynomial {
250 if (a.isZero()) return zero;
251 const result = try gpa.alloc(Fraction, a.coefficients.len + 1);
252 result[0] = Fraction.zero;
253 for (a.coefficients, 1..) |value, i| result[i] = value.div(Fraction.whole(@intCast(i)));
254 return fromOwned(result);
255 }
256 fn divide(a: Polynomial, gpa: Allocator, b: Polynomial) Allocator.Error!Division {
257 const length = if (a.coefficients.len >= b.coefficients.len) a.coefficients.len - b.coefficients.len + 1 else 0;
258 const quotient = try gpa.alloc(Fraction, length);
259 @memset(quotient, Fraction.zero);
260 const remainder = try gpa.dupe(Fraction, a.coefficients);
261 var remaining = remainder.len;
262 while (remaining > 0 and remaining >= b.coefficients.len) {
263 const factor = remainder[remaining - 1].div(b.leading());
264 const shift = remaining - b.coefficients.len;
265 quotient[shift] = factor;
266 for (b.coefficients, 0..) |value, j| remainder[shift + j] = remainder[shift + j].sub(factor.mul(value));
267 while (remaining > 0 and remainder[remaining - 1].isZero()) remaining -= 1;
268 }
269 return .{ .quotient = fromOwned(quotient), .remainder = fromOwned(remainder[0..remaining]) };
270 }
271 fn greatestCommonDivisor(gpa: Allocator, left: Polynomial, right: Polynomial) Allocator.Error!Polynomial {
272 var a = left;
273 var b = right;
274 while (!b.isZero()) {
275 const remainder = (try a.divide(gpa, b)).remainder;
276 a = b;
277 b = remainder;
278 }
279 return a.monic(gpa);
280 }
281 fn extendedGcd(gpa: Allocator, a: Polynomial, b: Polynomial) Allocator.Error!Extended {
282 var r0 = a;
283 var r1 = b;
284 var s0 = one;
285 var s1 = zero;
286 var t0 = zero;
287 var t1 = one;
288 while (!r1.isZero()) {
289 const division = try r0.divide(gpa, r1);
290 r0 = r1;
291 r1 = division.remainder;
292 const next_s = try s0.sub(gpa, try division.quotient.mul(gpa, s1));
293 s0 = s1;
294 s1 = next_s;
295 const next_t = try t0.sub(gpa, try division.quotient.mul(gpa, t1));
296 t0 = t1;
297 t1 = next_t;
298 }
299 if (r0.isZero()) return .{ .g = r0, .s = s0, .t = t0 };
300 return .{ .g = try r0.monic(gpa), .s = try s0.divideScalar(gpa, r0.leading()), .t = try t0.divideScalar(gpa, r0.leading()) };
301 }
302 fn content(a: Polynomial) Fraction {
303 var denominator: i64 = 1;
304 var numerator: i64 = 0;
305 for (a.coefficients) |value| {
306 denominator = @divTrunc(denominator * value.d, gcd(denominator, value.d));
307 numerator = gcd(numerator, value.n);
308 }
309 return Fraction.init(if (a.leading().n < 0) -numerator else numerator, denominator);
310 }
311 fn primitive(a: Polynomial, gpa: Allocator) Allocator.Error!Polynomial {
312 return a.divideScalar(gpa, a.content());
313 }
314 fn squareFreePart(a: Polynomial, gpa: Allocator) Allocator.Error!Polynomial {
315 if (a.degree() < 1) return a;
316 const common = try greatestCommonDivisor(gpa, a, try a.derivative(gpa, 1));
317 return (try a.divide(gpa, common)).quotient;
318 }

320 return .{ .value = a, .variable = variable_text };
321 }

323 try a.notation("x").format(out);

326const Notation = struct {
327 value: Polynomial,
328 variable: []const u8,
329 pub fn format(notation: Notation, out: *Writer) Writer.Error!void {
330 if (notation.value.isZero()) return out.writeByte('0');
331 var first = true;
332 var i = notation.value.coefficients.len;
333 while (i > 0) {
334 i -= 1;
335 const coefficient = notation.value.at(i);
336 if (coefficient.isZero()) continue;
337 if (first) {
338 if (coefficient.n < 0) try out.writeByte('-');
339 } else try out.writeAll(if (coefficient.n < 0) " - " else " + ");
340 const magnitude = coefficient.abs();
341 if (i == 0 or !magnitude.eq(Fraction.one)) try out.print("{f}", .{magnitude});
342 if (i > 0) try out.writeAll(notation.variable);
343 if (i > 1) try out.print("^{d}", .{i});
344 first = false;
345 }
346 }

349fn checkProduct(a: Polynomial, b: Polynomial) SemanticError!void {
350 if (!a.isZero() and !b.isZero() and a.degree() + b.degree() > 30) return error.DegreeLimit;
351}
352fn checkPower(a: Polynomial, exponent: usize) SemanticError!void {
353 if (!a.isZero() and @as(usize, @intCast(a.degree())) * exponent > 30) return error.DegreeLimit;
354}
355fn checkComposition(a: Polynomial, b: Polynomial) SemanticError!void {
356 if (a.degree() >= 1 and b.degree() >= 1 and a.degree() * b.degree() > 30) return error.DegreeLimit;
357}
358fn checkNonzero(a: Polynomial) SemanticError!void {
359 if (a.isZero()) return error.ZeroPolynomial;
360}
361fn checkDivisor(a: Polynomial) SemanticError!void {
362 if (a.isZero()) return error.DivisionByZero;
363}

365 if (a.degree() == 0) return error.DegreeTooLow;

368const Expression = union(enum) {
369 number: i64,
370 variable,
371 name: struct { text: []const u8, order: usize },
372 negative: *const Expression,
373 power: struct { operand: *const Expression, exponent: usize },
374 binary: struct { operation: u8, left: *const Expression, right: *const Expression },
375 call: struct { function: *const Expression, argument: *const Expression },
376
377 fn evaluate(expression: *const Expression, session: *Session) Error!Polynomial {
378 const gpa = session.gpa;
379 switch (expression.*) {
380 .number => |value| return Polynomial.constant(gpa, Fraction.whole(value)),
381 .variable => return Polynomial.variable,
382 .name => |name| return (try session.lookup(name.text)).derivative(gpa, name.order),
383 .negative => |operand| return (try operand.evaluate(session)).neg(gpa),
384 .power => |power| {
385 const value = try power.operand.evaluate(session);
386 try checkPower(value, power.exponent);
387 return value.power(gpa, power.exponent);
388 },
389 .binary => |binary| {
390 const a = try binary.left.evaluate(session);
391 const b = try binary.right.evaluate(session);
392 switch (binary.operation) {
393 '+' => return a.add(gpa, b),
394 '-' => return a.sub(gpa, b),
395 '*' => {
396 try checkProduct(a, b);
397 return a.mul(gpa, b);
398 },
399 else => {
400 try checkDivisor(b);
401 if (b.degree() >= 1) return error.NonconstantDivisor;
402 return a.divideScalar(gpa, b.at(0));
403 },
404 }
405 },
406 .call => |call| {
407 const outer = try call.function.evaluate(session);
408 const inner = try call.argument.evaluate(session);
409 try checkComposition(outer, inner);
410 return outer.compose(gpa, inner);
411 },
412 }
413 }

415const TokenKind = enum { number, name, symbol, end };
416const Token = struct { kind: TokenKind, text: []const u8, position: usize };
417const Parser = struct {
418 session: *Session,
419 tokens: [101]Token = undefined,
420 count: usize = 0,
421 position: usize = 0,
422
423 fn digit(ch: u8) bool {
424 return ch >= '0' and ch <= '9';
425 }
426 fn upper(ch: u8) bool {
427 return ch >= 'A' and ch <= 'Z';
428 }
429 fn init(session: *Session, source: []const u8) Parser {
430 var parser: Parser = .{ .session = session };
431 var i: usize = 0;
432 while (i < source.len) {
433 if (source[i] == ' ') {
434 i += 1;
435 continue;
436 }
437 const start = i;
438 i += 1;
439 const kind: TokenKind = if (digit(source[start])) .number else if (upper(source[start])) .name else .symbol;
440 if (kind == .number) {
441 while (i < source.len and digit(source[i])) i += 1;
442 } else if (kind == .name) {
443 while (i < source.len and (upper(source[i]) or digit(source[i]))) i += 1;
444 }
445 parser.tokens[parser.count] = .{ .kind = kind, .text = source[start..i], .position = start + 1 };
446 parser.count += 1;
447 }
448 parser.tokens[parser.count] = .{ .kind = .end, .text = "", .position = source.len + 1 };
449 return parser;
450 }
451 fn current(parser: *const Parser) Token {
452 return parser.tokens[parser.position];
453 }
454 fn at(parser: *const Parser, symbol: []const u8) bool {
455 return equal(parser.current().text, symbol);
456 }
457 fn syntax(parser: *Parser) Error {
458 if (parser.current().kind == .end) return parser.session.fail("SYNTAX AT END", .{});
459 return parser.session.fail("SYNTAX AT {d}", .{parser.current().position});
460 }
461 fn expect(parser: *Parser, symbol: []const u8) Error!void {
462 if (!parser.at(symbol)) return parser.syntax();
463 parser.position += 1;
464 }
465 fn node(parser: *Parser, tree: Expression) Allocator.Error!*const Expression {
466 const result = try parser.session.gpa.create(Expression);
467 result.* = tree;
468 return result;
469 }
470 fn parse(parser: *Parser) Error!*const Expression {
471 const result = try parser.expression();
472 if (parser.current().kind != .end) return parser.syntax();
473 return result;
474 }
475 fn expression(parser: *Parser) Error!*const Expression {
476 const negative = parser.at("-");
477 if (negative) parser.position += 1;
478 var result = try parser.term();
479 if (negative) result = try parser.node(.{ .negative = result });
480 while (parser.at("+") or parser.at("-")) {
481 const operation = parser.current().text[0];
482 parser.position += 1;
483 const right = try parser.term();
484 result = try parser.node(.{ .binary = .{ .operation = operation, .left = result, .right = right } });
485 }
486 return result;
487 }
488 fn term(parser: *Parser) Error!*const Expression {
489 var result = try parser.factor();
490 while (true) {
491 var operation: u8 = '*';
492 if (parser.at("*") or parser.at("/")) {
493 operation = parser.current().text[0];
494 parser.position += 1;
495 } else if (!parser.at("x") and !parser.at("(") and parser.current().kind != .name) break;
496 const right = try parser.factor();
497 result = try parser.node(.{ .binary = .{ .operation = operation, .left = result, .right = right } });
498 }
499 return result;
500 }
501 fn factor(parser: *Parser) Error!*const Expression {
502 var result = try parser.primary();
503 if (parser.at("^")) {
504 parser.position += 1;
505 if (parser.current().kind != .number) return parser.syntax();
506 const exponent = natural(parser.current().text);
507 parser.position += 1;
508 result = try parser.node(.{ .power = .{ .operand = result, .exponent = exponent } });
509 }
510 return result;
511 }
512 fn primary(parser: *Parser) Error!*const Expression {
513 const token = parser.current();
514 if (token.kind == .number) {
515 parser.position += 1;
516 return parser.node(.{ .number = integer(token.text) });
517 }
518 if (parser.at("x")) {
519 parser.position += 1;
520 return parser.node(.variable);
521 }
522 if (token.kind == .name) {
523 parser.position += 1;
524 var order: usize = 0;
525 while (parser.at("'")) {
526 parser.position += 1;
527 order += 1;
528 }
529 const reference = try parser.node(.{ .name = .{ .text = token.text, .order = order } });
530 if (!parser.at("(")) return reference;
531 parser.position += 1;
532 const argument = try parser.expression();
533 try parser.expect(")");
534 return parser.node(.{ .call = .{ .function = reference, .argument = argument } });
535 }
536 if (parser.at("(")) {
537 parser.position += 1;
538 const result = try parser.expression();
539 try parser.expect(")");
540 return result;
541 }
542 return parser.syntax();
543 }

572fn rationalRoots(gpa: Allocator, value: Polynomial) Allocator.Error!Roots {
573 const content = value.content();
574 var remaining = try value.divideScalar(gpa, content);
575 var zeros: usize = 0;
576 while (remaining.at(0).isZero()) {
577 remaining = .{ .coefficients = remaining.coefficients[1..] };
578 zeros += 1;
579 }
580 const a = absolute(remaining.at(0).n);
581 const b = remaining.leading().n;
582 var candidates: std.ArrayList(Fraction) = .empty;
583 var s: i64 = 1;
584 while (s <= a) : (s += 1) {
585 if (@mod(a, s) != 0) continue;
586 var t: i64 = 1;
587 while (t <= b) : (t += 1) {
588 if (@mod(b, t) != 0 or gcd(s, t) != 1) continue;
589 try candidates.append(gpa, Fraction.init(s, t));
590 try candidates.append(gpa, Fraction.init(-s, t));
591 }
592 }
593 std.sort.heap(Fraction, candidates.items, {}, Fraction.lessThan);
594 var roots: std.ArrayList(Root) = .empty;
595 for (candidates.items) |candidate| {
596 const divisor = try Polynomial.init(gpa, &.{ Fraction.whole(-candidate.n), Fraction.whole(candidate.d) });
597 var count: usize = 0;
598 while (remaining.degree() >= 1 and remaining.evaluate(candidate).isZero()) {
599 remaining = (try remaining.divide(gpa, divisor)).quotient;
600 count += 1;
601 }
602 if (count > 0) try roots.append(gpa, .{ .value = candidate, .multiplicity = count });
603 }
604 if (zeros > 0) try roots.append(gpa, .{ .value = Fraction.zero, .multiplicity = zeros });
605 std.sort.heap(Root, roots.items, {}, Root.lessThan);
606 return .{ .content = content, .remaining = remaining, .roots = roots.items };
607}

609fn squareFreeFactors(gpa: Allocator, value: Polynomial) Allocator.Error![]const Factor {
610 var result: std.ArrayList(Factor) = .empty;
611 if (value.degree() < 1) return result.items;
612 const monic = try value.monic(gpa);
613 var c = try Polynomial.greatestCommonDivisor(gpa, monic, try monic.derivative(gpa, 1));
614 var w = (try monic.divide(gpa, c)).quotient;
615 var i: usize = 1;
616 while (w.degree() >= 1) : (i += 1) {
617 const y = try Polynomial.greatestCommonDivisor(gpa, w, c);
618 const factor = (try w.divide(gpa, y)).quotient;
619 if (factor.degree() >= 1) try result.append(gpa, .{ .value = try factor.primitive(gpa), .multiplicity = i });
620 c = (try c.divide(gpa, y)).quotient;
621 w = y;
622 }
623 return result.items;
624}

626fn sturmSequence(gpa: Allocator, value: Polynomial) Allocator.Error![]const Polynomial {
627 var previous = try value.squareFreePart(gpa);
628 var current = try previous.derivative(gpa, 1);
629 var result: std.ArrayList(Polynomial) = .empty;
630 try result.append(gpa, previous);
631 while (!current.isZero()) {
632 try result.append(gpa, current);
633 const next = try (try previous.divide(gpa, current)).remainder.neg(gpa);
634 previous = current;
635 current = next;
636 }
637 return result.items;
638}
639fn variations(sequence: []const Polynomial, point: Endpoint) i32 {
640 var previous: i32 = 0;
641 var count: i32 = 0;
642 for (sequence) |value| {
643 var sign = if (point.infinity == 0) value.evaluate(point.value).sign() else value.leading().sign();
644 if (point.infinity < 0 and @mod(value.degree(), 2) != 0) sign = -sign;
645 if (sign == 0) continue;
646 if (previous != 0 and sign != previous) count += 1;
647 previous = sign;
648 }
649 return count;
650}
651fn rootCount(sequence: []const Polynomial, left: Endpoint, right: Endpoint) i32 {
652 const left_count = variations(sequence, left);
653 const right_count = variations(sequence, right);
654 return left_count - right_count;
655}
656const Isolation = struct {
657 gpa: Allocator,
658 sequence: []const Polynomial,
659 width: Fraction,
660 intervals: std.ArrayList(Interval) = .empty,
661 fn count(isolation: *const Isolation, left: Fraction, right: Fraction) i32 {
662 return rootCount(isolation.sequence, .{ .value = left }, .{ .value = right });
663 }
664 fn examine(isolation: *Isolation, low: Fraction, high: Fraction) Allocator.Error!void {
665 const root_count = isolation.count(low, high);
666 if (root_count >= 2) {
667 const middle = low.add(high).div(Fraction.whole(2));
668 try isolation.examine(low, middle);
669 try isolation.examine(middle, high);
670 } else if (root_count == 1) {
671 var left = low;
672 var right = high;
673 while (isolation.width.less(right.sub(left))) {
674 const middle = left.add(right).div(Fraction.whole(2));
675 if (isolation.count(left, middle) >= 1) right = middle else left = middle;
676 }
677 try isolation.intervals.append(isolation.gpa, .{ .left = left, .right = right });
678 }
679 }
680};
681fn isolate(gpa: Allocator, value: Polynomial, precision: usize) Allocator.Error![]const Interval {
682 const sequence = try sturmSequence(gpa, value);
683 const first = sequence[0];
684 var maximum = Fraction.zero;
685 for (first.coefficients[0 .. first.coefficients.len - 1]) |coefficient| {
686 const candidate = coefficient.div(first.leading()).abs();
687 if (maximum.less(candidate)) maximum = candidate;
688 }
689 const beta = Fraction.one.add(maximum);
690 var isolation: Isolation = .{ .gpa = gpa, .sequence = sequence, .width = Fraction.init(1, @as(i64, 1) << @intCast(precision)) };
691 try isolation.examine(beta.neg(), beta);
692 return isolation.intervals.items;
693}
694fn resultant(gpa: Allocator, a: Polynomial, b: Polynomial) Allocator.Error!Fraction {
695 var p = a;
696 var q = b;
697 var result = Fraction.one;
698 while (q.degree() >= 1) {
699 const remainder = (try p.divide(gpa, q)).remainder;
700 if (remainder.isZero()) return Fraction.zero;
701 result = result.mul(q.leading().power(@intCast(p.degree() - remainder.degree())));
702 if (@mod(p.degree() * q.degree(), 2) != 0) result = result.neg();
703 p = q;
704 q = remainder;
705 }
706 return result.mul(q.at(0).power(@intCast(p.degree())));
707}
708fn descartes(value: Polynomial, negative: bool) usize {
709 var previous: i32 = 0;
710 var count: usize = 0;
711 var i = value.coefficients.len;
712 while (i > 0) {
713 i -= 1;
714 var sign = value.at(i).sign();
715 if (negative and i % 2 != 0) sign = -sign;
716 if (sign == 0) continue;
717 if (previous != 0 and sign != previous) count += 1;
718 previous = sign;
719 }
720 return count;
721}

723fn interpolate(session: *Session, values: []const Fraction) Error!Polynomial {
724 const gpa = session.gpa;
725 const count = values.len / 2;
726 const points = try gpa.alloc(Fraction, count);
727 const differences = try gpa.alloc(Fraction, count);
728 for (0..count) |i| {
729 points[i] = values[2 * i];
730 differences[i] = values[2 * i + 1];
731 }
732 for (1..count) |j| {
733 for (0..j) |i| if (points[j].eq(points[i])) return session.fail("DUPLICATE X {f}", .{points[j]});
734 }
735 for (1..count) |j| {
736 var i = count;
737 while (i > j) {
738 i -= 1;
739 differences[i] = differences[i].sub(differences[i - 1]).div(points[i].sub(points[i - j]));
740 }
741 }
742 var result = Polynomial.zero;
743 var i = count;
744 while (i > 0) {
745 i -= 1;
746 const factor = try Polynomial.init(gpa, &.{ points[i].neg(), Fraction.one });
747 result = try (try result.mul(gpa, factor)).add(gpa, try Polynomial.constant(gpa, differences[i]));
748 }
749 return result;
750}
751const SpecialKind = enum { chebyshev, chebyshev2, legendre, hermite, laguerre };
752fn special(session: *Session, text: []const u8, degree: usize) Error!Polynomial {
753 const kind = std.meta.stringToEnum(SpecialKind, text) orelse return session.fail("UNKNOWN KIND {s}", .{text});
754 if (degree > 30) return error.DegreeLimit;
755 const gpa = session.gpa;
756 var previous = Polynomial.one;
757 if (degree == 0) return previous;
758 var current = switch (kind) {
759 .chebyshev2, .hermite => try Polynomial.init(gpa, &.{ Fraction.zero, Fraction.whole(2) }),
760 .laguerre => try Polynomial.init(gpa, &.{ Fraction.one, Fraction.whole(-1) }),
761 else => Polynomial.variable,
762 };
763 for (1..degree) |step| {
764 const j: i64 = @intCast(step);
765 const factor = switch (kind) {
766 .legendre => try Polynomial.init(gpa, &.{ Fraction.zero, Fraction.whole(2 * j + 1) }),
767 .laguerre => try Polynomial.init(gpa, &.{ Fraction.whole(2 * j + 1), Fraction.whole(-1) }),
768 else => try Polynomial.init(gpa, &.{ Fraction.zero, Fraction.whole(2) }),
769 };
770 const scale = switch (kind) {
771 .legendre, .laguerre => j,
772 .hermite => 2 * j,
773 else => 1,
774 };
775 var next = try (try factor.mul(gpa, current)).sub(gpa, try previous.scale(gpa, Fraction.whole(scale)));
776 if (kind == .legendre or kind == .laguerre) next = try next.scale(gpa, Fraction.init(1, j + 1));
777 previous = current;
778 current = next;
779 }
780 return current;
781}

783const Name = struct {

795 try out.writeAll(name.bytes[0..name.len]);

799 name: Name,
800 value: Polynomial,

803 }

805const Storage = struct {
806 gpa: Allocator,
807 entries: std.ArrayList(Stored) = .empty,
808 fn find(storage: *const Storage, text: []const u8) ?usize {
809 for (storage.entries.items, 0..) |entry, i| if (entry.name.matches(text)) return i;

813 const saved: Stored = .{ .name = entry.name, .value = try Polynomial.init(storage.gpa, entry.value.coefficients) };
814 if (storage.find(entry.name.bytes[0..entry.name.len])) |i| storage.entries.items[i] = saved else try storage.entries.append(storage.gpa, saved);
815 }

817fn nameArguments(command: Command) []const usize {
818 return switch (command) {
819 .PRINT, .DELETE, .DEGREE, .AREA, .TAYLOR, .EVAL, .ROOTS, .FACTOR, .SQFREE, .DESCARTES, .STURM, .ISOLATE, .DISCRIM, .POWERSUM => &.{1},
820 .ADD, .SUB, .MUL, .GCD, .LCM, .INVMOD, .COMPOSE => &.{ 2, 3 },
821 .POW, .DERIV, .INTEG, .TANGENT => &.{2},
822 .DIV => &.{ 3, 4 },
823 .XGCD => &.{ 4, 5 },
824 .CRT => &.{ 2, 3, 4, 5 },
825 .MODPOW => &.{ 2, 4 },
826 .RESULTANT => &.{ 1, 2 },
827 else => &.{},
828 };

830const Session = struct {
831 gpa: Allocator,
832 storage: *Storage,
833 out: *Writer,
834 problem: ?[]const u8 = null,
835 pending: [3]Stored = undefined,
836 pending_count: usize = 0,
837 deletion: ?usize = null,
838
839 fn fail(session: *Session, comptime fmt: []const u8, args: anytype) Error {
840 session.problem = std.fmt.allocPrint(session.gpa, fmt, args) catch return error.OutOfMemory;
841 return error.BadCommand;
842 }
843 fn lookup(session: *Session, text: []const u8) Error!Polynomial {
844 const i = session.storage.find(text) orelse return session.fail("UNKNOWN {s}", .{text});
845 return session.storage.entries.items[i].value;
846 }
847 fn save(session: *Session, name: []const u8, value: Polynomial) Writer.Error!void {
848 session.pending[session.pending_count] = .{ .name = Name.init(name), .value = value };
849 session.pending_count += 1;
850 try line(session.out, "{s} = {f}", .{ name, value });
851 }
852 fn commit(session: *Session) Allocator.Error!void {
853 if (session.deletion) |i| _ = session.storage.entries.orderedRemove(i);
854 for (session.pending[0..session.pending_count]) |entry| try session.storage.set(entry);
855 }
856 fn process(session: *Session, text: []const u8) Error!void {
857 const gpa = session.gpa;
858 const out = session.out;
859 if (std.mem.startsWith(u8, text, "DEF ")) {
860 const separator = 4 + std.mem.findScalar(u8, text[4..], ' ').?;
861 var parser = Parser.init(session, text[separator + 1 ..]);
862 const expression = try parser.parse();
863 return session.save(text[4..separator], try expression.evaluate(session));
864 }
865 var words_buffer: [64][]const u8 = undefined;
866 var word_count: usize = 0;
867 var iterator = std.mem.splitScalar(u8, text, ' ');
868 while (iterator.next()) |word| {
869 words_buffer[word_count] = word;
870 word_count += 1;
871 }
872 const words = words_buffer[0..word_count];
873 const command = std.meta.stringToEnum(Command, words[0]).?;
874 if (command == .DIV and equal(words[1], words[2]) or command == .XGCD and
875 (equal(words[1], words[2]) or equal(words[1], words[3]) or equal(words[2], words[3]))) return error.SameName;
876 var operands: [4]Polynomial = @splat(Polynomial.zero);
877 for (nameArguments(command), 0..) |index, i| operands[i] = try session.lookup(words[index]);
878 var literals: [62]Fraction = undefined;
879 var literal_count: usize = 0;
880 const literal_indices: []const usize = switch (command) {
881 .AREA => &.{ 2, 3 },
882 .TANGENT => &.{3},
883 .TAYLOR => &.{2},
884 else => &.{},
885 };
886 for (literal_indices) |index| {
887 literals[literal_count] = try Fraction.parse(words[index]);
888 literal_count += 1;
889 }
890 if (command == .EVAL or command == .INTERP) {
891 for (words[2..]) |word| {
892 literals[literal_count] = try Fraction.parse(word);
893 literal_count += 1;
894 }
895 }
896 var left: Endpoint = .{};
897 var right: Endpoint = .{};
898 if (command == .STURM) {
899 left = try Endpoint.parse(words[2]);
900 right = try Endpoint.parse(words[3]);
901 }
902 const a = operands[0];
903 const b = operands[1];
904 switch (command) {
905 .DEF => unreachable,
906 .PRINT => try line(out, "{s} = {f}", .{ words[1], a }),
907 .DELETE => {
908 session.deletion = session.storage.find(words[1]).?;
909 try line(out, "deleted {s}", .{words[1]});
910 },
911 .LIST => {
912 if (session.storage.entries.items.len == 0) return line(out, "no polynomials", .{});
913 const entries = try gpa.dupe(Stored, session.storage.entries.items);
914 std.sort.heap(Stored, entries, {}, Stored.lessThan);
915 for (entries) |entry| try line(out, "{f} = {f}", .{ entry.name, entry.value });
916 },
917 .DEGREE => {
918 if (a.isZero()) try line(out, "deg({s}) = -inf", .{words[1]}) else try line(out, "deg({s}) = {d}", .{ words[1], a.degree() });
919 },
920 .ADD => try session.save(words[1], try a.add(gpa, b)),
921 .SUB => try session.save(words[1], try a.sub(gpa, b)),
922 .MUL => {
923 try checkProduct(a, b);
924 try session.save(words[1], try a.mul(gpa, b));
925 },
926 .POW => {
927 const exponent = natural(words[3]);
928 try checkPower(a, exponent);
929 try session.save(words[1], try a.power(gpa, exponent));
930 },
931 .DIV => {
932 try checkDivisor(b);
933 const division = try a.divide(gpa, b);
934 try session.save(words[1], division.quotient);
935 try session.save(words[2], division.remainder);
936 },
937 .GCD => try session.save(words[1], try Polynomial.greatestCommonDivisor(gpa, a, b)),
938 .XGCD => {
939 const result = try Polynomial.extendedGcd(gpa, a, b);
940 try session.save(words[1], result.g);
941 try session.save(words[2], result.s);
942 try session.save(words[3], result.t);
943 },
944 .LCM => {
945 var result = Polynomial.zero;
946 if (!a.isZero() and !b.isZero()) {
947 const common = try Polynomial.greatestCommonDivisor(gpa, a, b);
948 if (a.degree() + b.degree() - common.degree() > 30) return error.DegreeLimit;
949 result = try (try (try a.divide(gpa, common)).quotient.mul(gpa, b)).monic(gpa);
950 }
951 try session.save(words[1], result);
952 },
953 .INVMOD => {
954 try checkDivisor(b);
955 try checkPositiveDegree(b);
956 const result = try Polynomial.extendedGcd(gpa, a, b);
957 if (result.g.degree() != 0 or !result.g.at(0).eq(Fraction.one)) return error.NotCoprime;
958 try session.save(words[1], (try result.s.divide(gpa, b)).remainder);
959 },
960 .CRT => {
961 const a2 = operands[2];
962 const m2 = operands[3];
963 try checkDivisor(b);
964 try checkDivisor(m2);
965 if (b.degree() + m2.degree() > 30) return error.DegreeLimit;
966 const result = try Polynomial.extendedGcd(gpa, b, m2);
967 if (result.g.degree() != 0 or !result.g.at(0).eq(Fraction.one)) return error.NotCoprime;
968 const r1 = (try a.divide(gpa, b)).remainder;
969 const r2 = (try a2.divide(gpa, m2)).remainder;
970 const u = (try (try (try r2.sub(gpa, r1)).mul(gpa, result.s)).divide(gpa, m2)).remainder;
971 try session.save(words[1], try r1.add(gpa, try b.mul(gpa, u)));
972 },
973 .MODPOW => {
974 try checkDivisor(b);
975 if (2 * b.degree() - 2 > 30) return error.DegreeLimit;
976 var x = (try Polynomial.one.divide(gpa, b)).remainder;
977 var y = (try a.divide(gpa, b)).remainder;
978 var exponent = natural(words[3]);
979 while (exponent > 0) {
980 if (exponent % 2 != 0) x = (try (try x.mul(gpa, y)).divide(gpa, b)).remainder;
981 exponent /= 2;
982 if (exponent != 0) y = (try (try y.mul(gpa, y)).divide(gpa, b)).remainder;
983 }
984 try session.save(words[1], x);
985 },
986 .DERIV => try session.save(words[1], try a.derivative(gpa, natural(words[3]))),
987 .INTEG => {
988 if (a.degree() == 30) return error.DegreeLimit;
989 try session.save(words[1], try a.integral(gpa));
990 },
991 .AREA => {
992 const antiderivative = try a.integral(gpa);
993 const upper = antiderivative.evaluate(literals[1]);
994 const lower = antiderivative.evaluate(literals[0]);
995 try line(out, "integral({s}, {f}, {f}) = {f}", .{ words[1], literals[0], literals[1], upper.sub(lower) });
996 },
997 .TANGENT => {
998 const point = literals[0];
999 const height = a.evaluate(point);
1000 const slope = (try a.derivative(gpa, 1)).evaluate(point);
1001 try session.save(words[1], try Polynomial.init(gpa, &.{ height.sub(slope.mul(point)), slope }));
1002 },
1003 .TAYLOR => {
1004 const point = literals[0];
1005 const shifted = try a.compose(gpa, try Polynomial.init(gpa, &.{ point, Fraction.one }));
1006 const variable_text = if (point.isZero()) "x" else if (point.n > 0) try std.fmt.allocPrint(gpa, "(x - {f})", .{point}) else try std.fmt.allocPrint(gpa, "(x + {f})", .{point.neg()});
1007 try line(out, "taylor({s}, {f}) = {f}", .{ words[1], point, shifted.notation(variable_text) });
1008 },
1009 .COMPOSE => {
1010 try checkComposition(a, b);
1011 try session.save(words[1], try a.compose(gpa, b));
1012 },
1013 .EVAL => for (literals[0..literal_count]) |point| try line(out, "{s}({f}) = {f}", .{ words[1], point, a.evaluate(point) }),
1014 .ROOTS, .FACTOR => {
1015 try checkNonzero(a);
1016 const result = try rationalRoots(gpa, a);
1017 if (command == .ROOTS) {
1018 try out.print("roots({s}) = ", .{words[1]});
1019 if (result.roots.len == 0) try out.writeAll("none");
1020 for (result.roots, 0..) |root, i| {
1021 if (i > 0) try out.writeAll(", ");
1022 try out.print("{f} ({d})", .{ root.value, root.multiplicity });
1023 }
1024 } else {
1025 try out.print("factor({s}) = {f}", .{ words[1], result.content });
1026 for (result.roots) |root| {
1027 if (root.value.isZero()) try out.writeAll(" * x") else {
1028 const factor = try Polynomial.init(gpa, &.{ Fraction.whole(-root.value.n), Fraction.whole(root.value.d) });
1029 try out.print(" * ({f})", .{factor});
1030 }
1031 if (root.multiplicity >= 2) try out.print("^{d}", .{root.multiplicity});
1032 }
1033 if (result.remaining.degree() >= 1) try out.print(" * ({f})", .{result.remaining});
1034 }
1035 try out.writeByte('\n');
1036 },
1037 .SQFREE => {
1038 try checkNonzero(a);
1039 const factors = try squareFreeFactors(gpa, a);
1040 try out.print("sqfree({s}) = {f}", .{ words[1], a.content() });
1041 for (factors) |factor| {
1042 const is_x = factor.value.degree() == 1 and factor.value.at(0).isZero() and factor.value.at(1).eq(Fraction.one);
1043 if (is_x) try out.writeAll(" * x") else try out.print(" * ({f})", .{factor.value});
1044 if (factor.multiplicity >= 2) try out.print("^{d}", .{factor.multiplicity});
1045 }
1046 try out.writeByte('\n');
1047 },
1048 .DESCARTES => {
1049 try checkNonzero(a);
1050 try line(out, "descartes({s}) = positive {d}, negative {d}", .{ words[1], descartes(a, false), descartes(a, true) });
1051 },
1052 .STURM => {
1053 try checkNonzero(a);
1054 if (!left.less(right)) return error.BadInterval;
1055 const sequence = try sturmSequence(gpa, a);
1056 try line(out, "sturm({s}, {f}, {f}) = {d}", .{ words[1], left, right, rootCount(sequence, left, right) });
1057 },
1058 .ISOLATE => {
1059 try checkNonzero(a);
1060 const precision = natural(words[2]);
1061 const intervals = try isolate(gpa, a, precision);
1062 try out.print("isolate({s}, {d}) = ", .{ words[1], precision });
1063 if (intervals.len == 0) try out.writeAll("none");
1064 for (intervals, 0..) |interval, i| {
1065 if (i > 0) try out.writeAll(", ");
1066 try out.print("({f}, {f}]", .{ interval.left, interval.right });
1067 }
1068 try out.writeByte('\n');
1069 },
1070 .RESULTANT => {
1071 try checkNonzero(a);
1072 try checkNonzero(b);
1073 try line(out, "res({s}, {s}) = {f}", .{ words[1], words[2], try resultant(gpa, a, b) });
1074 },
1075 .DISCRIM => {
1076 try checkNonzero(a);
1077 try checkPositiveDegree(a);
1078 var value = (try resultant(gpa, a, try a.derivative(gpa, 1))).div(a.leading());
1079 if (@mod(@divTrunc(a.degree() * (a.degree() - 1), 2), 2) != 0) value = value.neg();
1080 try line(out, "disc({s}) = {f}", .{ words[1], value });
1081 },
1082 .POWERSUM => {
1083 try checkNonzero(a);
1084 try checkPositiveDegree(a);
1085 const count = natural(words[2]);
1086 const monic = try a.monic(gpa);
1087 const degree: usize = @intCast(a.degree());
1088 var sums: [31]Fraction = @splat(Fraction.zero);
1089 for (1..count + 1) |j| {
1090 var value = if (j <= degree) Fraction.whole(@intCast(j)).mul(monic.at(degree - j)) else Fraction.zero;
1091 for (1..@min(j - 1, degree) + 1) |i| value = value.add(monic.at(degree - i).mul(sums[j - i]));
1092 sums[j] = value.neg();
1093 }
1094 try out.print("powersum({s}) = ", .{words[1]});
1095 for (sums[1 .. count + 1], 0..) |value, i| {
1096 if (i > 0) try out.writeAll(", ");
1097 try out.print("{f}", .{value});
1098 }
1099 try out.writeByte('\n');
1100 },
1101 .INTERP => try session.save(words[1], try interpolate(session, literals[0..literal_count])),
1102 .SPECIAL => try session.save(words[1], try special(session, words[2], natural(words[3]))),
1103 }
1104 }
