12function parse_fraction(text::AbstractString)
13 parts = split(text, '/')
14 n = parse(Int64, parts[1]; base = 10)
15 d = length(parts) == 1 ? Int64(1) : parse(Int64, parts[2]; base = 10)
16 d == 0 && fail("ZERO DENOMINATOR")
17 n // d
18end

20struct Polynomial
21 coefficients::Vector{Fraction}
22 function Polynomial(values::Vector{Fraction})
23 size = length(values)
24 while size > 0 && iszero(values[size])
25 size -= 1
26 end
27 new(values[1:size])
28 end
29end
30Polynomial() = Polynomial(Fraction[])
31Polynomial(value::Fraction) = Polynomial(Fraction[value])
32const PZERO = Polynomial()
33const PONE = Polynomial(FONE)
34const PX = Polynomial(Fraction[0, 1])
35linear(value::Fraction, slope::Fraction = FONE) = Polynomial(Fraction[value, slope])
36degree(a::Polynomial) = length(a.coefficients) - 1
37leading(a::Polynomial) = last(a.coefficients)
38coefficient(a::Polynomial, i::Int) = i <= degree(a) ? a.coefficients[i + 1] : FZERO
39Base.iszero(a::Polynomial) = isempty(a.coefficients)
40Base.isone(a::Polynomial) = degree(a) == 0 && leading(a) == FONE
41require_nonzero(a::Polynomial) = iszero(a) && fail("ZERO POLYNOMIAL")
42require_divisor(a::Polynomial) = iszero(a) && fail("DIVISION BY ZERO")
43product_limit(a::Polynomial, b::Polynomial) = degree_limit(!iszero(a) && !iszero(b) && degree(a) + degree(b) > 30)
44composition_limit(a::Polynomial, b::Polynomial) = degree_limit(degree(a) >= 1 && degree(b) >= 1 && degree(a) * degree(b) > 30)
45
46Base.:+(a::Polynomial, b::Polynomial) = Polynomial(Fraction[coefficient(a, i) + coefficient(b, i) for i in 0:max(degree(a), degree(b))])
47Base.:-(a::Polynomial, b::Polynomial) = Polynomial(Fraction[coefficient(a, i) - coefficient(b, i) for i in 0:max(degree(a), degree(b))])
48Base.:-(a::Polynomial) = Polynomial(-a.coefficients)
49Base.:*(a::Polynomial, value::Fraction) = Polynomial(a.coefficients .* value)
50Base.:/(a::Polynomial, value::Fraction) = Polynomial(a.coefficients ./ value)
51function Base.:*(a::Polynomial, b::Polynomial)
52 (iszero(a) || iszero(b)) && return PZERO
53 result = zeros(Fraction, degree(a) + degree(b) + 1)
54 for i in eachindex(a.coefficients), j in eachindex(b.coefficients)
55 result[i + j - 1] += a.coefficients[i] * b.coefficients[j]
56 end
57 Polynomial(result)
58end
59function Base.:^(a::Polynomial, exponent::Int)
60 exponent == 0 && return PONE
61 iszero(a) && return PZERO
62 if degree(a) == 0 && abs(leading(a)) == FONE
63 return isodd(exponent) ? a : PONE
64 end
65 result = PONE
66 for _ in 1:exponent
67 result = result * a
68 end
69 result
70end
71function scalar_power(value::Fraction, exponent::Int)
72 result = FONE
73 for _ in 1:exponent
74 result *= value
75 end
76 result
77end
78function Base.divrem(a::Polynomial, b::Polynomial)
79 quotient = zeros(Fraction, max(0, degree(a) - degree(b) + 1))
80 remainder = copy(a.coefficients)
81 while !isempty(remainder) && length(remainder) >= length(b.coefficients)
82 offset = length(remainder) - length(b.coefficients)
83 factor = last(remainder) / leading(b)
84 quotient[offset + 1] = factor
85 for j in eachindex(b.coefficients)
86 remainder[offset + j] -= factor * b.coefficients[j]
87 end
88 while !isempty(remainder) && iszero(last(remainder))
89 pop!(remainder)
90 end
91 end
92 Polynomial(quotient), Polynomial(remainder)
93end
94quotient(a::Polynomial, b::Polynomial) = first(divrem(a, b))
95Base.rem(a::Polynomial, b::Polynomial) = last(divrem(a, b))
96monic(a::Polynomial) = iszero(a) ? a : a / leading(a)
97function derivative(a::Polynomial, count::Int = 1)
98 for _ in 1:count
99 degree(a) <= 0 && return PZERO
100 a = Polynomial(Fraction[i * coefficient(a, i) for i in 1:degree(a)])
101 end
102 a
103end
104integral(a::Polynomial) = Polynomial(vcat(FZERO, Fraction[coefficient(a, i) / (i + 1) for i in 0:degree(a)]))
105function evaluate(a::Polynomial, point::Fraction)
106 value = FZERO
107 for i in degree(a):-1:0
108 value = value * point + coefficient(a, i)
109 end
110 value

112function compose(a::Polynomial, b::Polynomial)
113 result = PZERO
114 for i in degree(a):-1:0
115 result = result * b + Polynomial(coefficient(a, i))
116 end
117 result

119function polynomial_gcd(a::Polynomial, b::Polynomial)
120 while !iszero(b)
121 a, b = b, rem(a, b)
122 end
123 monic(a)
124end
125function extended_gcd(a::Polynomial, b::Polynomial)
126 s0, s1, t0, t1 = PONE, PZERO, PZERO, PONE
127 while !iszero(b)
128 q, r = divrem(a, b)
129 next_s = s0 - q * s1
130 next_t = t0 - q * t1
131 a, b = b, r
132 s0, s1 = s1, next_s
133 t0, t1 = t1, next_t
134 end
135 iszero(a) && return (a, s0, t0)
136 divisor = leading(a)
137 a / divisor, s0 / divisor, t0 / divisor
138end
139function content(a::Polynomial)
140 common, multiple = Int64(0), Int64(1)
141 for value in a.coefficients
142 d = denominator(value)
143 multiple = (multiple * d) ÷ gcd(multiple, d)
144 common = gcd(common, abs(numerator(value)))
145 end
146 fraction(leading(a) < 0 ? -common : common, multiple)
147end
148primitive_part(a::Polynomial) = a / content(a)
149squarefree(a::Polynomial) = degree(a) >= 1 ? quotient(a, polynomial_gcd(a, derivative(a))) : a

151function notation(a::Polynomial, variable::String = "x")
152 result = IOBuffer()
153 first_term = true
154 for i in degree(a):-1:0
155 value = coefficient(a, i)
156 iszero(value) && continue
157 print(result, first_term ? (value < 0 ? "-" : "") : (value < 0 ? " - " : " + "))
158 if i == 0 || abs(value) != FONE
159 print(result, fraction_text(abs(value)))
160 end
161 i >= 1 && print(result, variable)
162 i >= 2 && print(result, '^', i)
163 first_term = false
164 end
165 first_term ? "0" : String(take!(result))
166end

168struct Root
169 value::Fraction
170 multiplicity::Int
171end
172function rational_roots(a::Polynomial)
173 remaining = primitive_part(a)
174 zeros = 0
175 while iszero(first(remaining.coefficients))
176 remaining = Polynomial(remaining.coefficients[2:end])
177 zeros += 1
178 end
179 divisors(n::Int64) = Int64[i for i in 1:n if n % i == 0]
180 candidates = Fraction[]
181 for n in divisors(abs(numerator(first(remaining.coefficients))))
182 for d in divisors(numerator(leading(remaining)))
183 gcd(n, d) == 1 || continue
184 push!(candidates, -n // d, n // d)
185 end
186 end
187 roots = Root[]
188 for value in sort!(candidates)
189 factor = linear(fraction(-numerator(value)), fraction(denominator(value)))
190 multiplicity = 0
191 while degree(remaining) >= 1 && iszero(evaluate(remaining, value))
192 remaining = quotient(remaining, factor)
193 multiplicity += 1
194 end
195 multiplicity > 0 && push!(roots, Root(value, multiplicity))
196 end
197 zeros > 0 && push!(roots, Root(FZERO, zeros))
198 sort!(roots; by = root -> root.value)
199 roots, remaining
200end
201factor_text(text::String, multiplicity::Int) = " * " * text * (multiplicity > 1 ? "^$multiplicity" : "")
202function factorization(a::Polynomial, roots::Vector{Root}, remaining::Polynomial)
203 result = fraction_text(content(a))
204 for root in roots
205 value = root.value
206 text = iszero(value) ? "x" : "(" * notation(linear(fraction(-numerator(value)), fraction(denominator(value)))) * ")"
207 result *= factor_text(text, root.multiplicity)
208 end
209 degree(remaining) >= 1 && (result *= " * (" * notation(remaining) * ")")
210 result
211end
212function squarefree_factors(a::Polynomial)
213 result = fraction_text(content(a))
214 degree(a) == 0 && return result
215 normalized = monic(a)
216 common = polynomial_gcd(normalized, derivative(normalized))
217 remaining = quotient(normalized, common)
218 multiplicity = 1
219 while degree(remaining) >= 1
220 shared = polynomial_gcd(remaining, common)
221 factor = quotient(remaining, shared)
222 if degree(factor) >= 1
223 text = notation(primitive_part(factor))
224 result *= factor_text(text == "x" ? text : "($text)", multiplicity)
225 end
226 common, remaining = quotient(common, shared), shared
227 multiplicity += 1
228 end
229 result
230end
231function sign_changes(signs::Vector{Int})
232 signs = filter(!iszero, signs)
233 count(i -> signs[i] != signs[i - 1], 2:length(signs))
234end
235struct Bound

237 value::Fraction

239Bound(value::Fraction) = Bound(0, value)
240parse_bound(text::String) = text == "-inf" ? Bound(-1, FZERO) : text == "inf" ? Bound(1, FZERO) : Bound(parse_fraction(text))
241bound_text(value::Bound) = value.kind == 0 ? fraction_text(value.value) : value.kind < 0 ? "-inf" : "inf"
242function compare_bounds(a::Bound, b::Bound)
243 a.kind != b.kind && return cmp(a.kind, b.kind)
244 a.kind == 0 ? cmp(a.value, b.value) : 0

246function sturm_sequence(a::Polynomial)
247 first = squarefree(a)
248 second = derivative(first)
249 sequence = Polynomial[first]
250 while !iszero(second)
251 push!(sequence, second)
252 first, second = second, -rem(first, second)
253 end
254 sequence
255end
256function variations(sequence::Vector{Polynomial}, point::Bound)
257 signs = Int[]
258 for a in sequence
259 value = point.kind == 0 ? sign(numerator(evaluate(a, point.value))) : sign(numerator(leading(a)))
260 point.kind < 0 && isodd(degree(a)) && (value = -value)
261 push!(signs, value)
262 end
263 sign_changes(signs)
264end
265root_count(sequence::Vector{Polynomial}, low::Bound, high::Bound) = variations(sequence, low) - variations(sequence, high)
266function isolate(a::Polynomial, precision::Int)
267 sequence = sturm_sequence(a)
268 first = sequence[1]
269 beta = degree(first) == 0 ? FONE : FONE + maximum(abs.(first.coefficients[1:end-1] ./ leading(first)))
270 width = fraction(1, Int64(1) << precision)
271 intervals = Tuple{Fraction,Fraction}[]
272 function examine(low::Fraction, high::Fraction)
273 count = root_count(sequence, Bound(low), Bound(high))
274 if count >= 2
275 middle = (low + high) / 2
276 examine(low, middle)
277 examine(middle, high)
278 elseif count == 1
279 while high - low > width
280 middle = (low + high) / 2
281 if root_count(sequence, Bound(low), Bound(middle)) >= 1
282 high = middle
283 else
284 low = middle
285 end
286 end
287 push!(intervals, (low, high))
288 end
289 end
290 examine(-beta, beta)
291 isempty(intervals) ? "none" : join(("($(fraction_text(low)), $(fraction_text(high))]" for (low, high) in intervals), ", ")
292end
293function resultant(a::Polynomial, b::Polynomial)
294 result = FONE
295 while degree(b) >= 1
296 remainder = rem(a, b)
297 iszero(remainder) && return FZERO
298 result *= scalar_power(leading(b), degree(a) - degree(remainder))
299 isodd(degree(a) * degree(b)) && (result = -result)
300 a, b = b, remainder
301 end
302 result * scalar_power(coefficient(b, 0), degree(a))
303end
304function power_sums(a::Polynomial, count::Int)
305 d = degree(a)
306 normalized = monic(a)
307 sums = zeros(Fraction, count)
308 for j in 1:count
309 value = j <= d ? j * coefficient(normalized, d - j) : FZERO
310 for i in 1:min(j - 1, d)
311 value += coefficient(normalized, d - i) * sums[j - i]
312 end
313 sums[j] = -value
314 end
315 join(fraction_text.(sums), ", ")
316end
317function interpolate(values::Vector{Fraction})
318 xs = values[1:2:end]
319 differences = values[2:2:end]
320 count = length(xs)
321 for j in 2:count
322 any(==(xs[j]), xs[1:j-1]) && fail("DUPLICATE X " * fraction_text(xs[j]))
323 end
324 for j in 1:count-1, i in count:-1:j+1
325 differences[i] = (differences[i] - differences[i - 1]) / (xs[i] - xs[i - j])
326 end
327 result = PZERO
328 for i in count:-1:1
329 result = result * linear(-xs[i]) + Polynomial(differences[i])
330 end
331 result
332end
333function special(kind::String, count::Int)
334 kind in ("chebyshev", "chebyshev2", "legendre", "hermite", "laguerre") || fail("UNKNOWN KIND $kind")
335 degree_limit(count > 30)
336 previous = PONE
337 count == 0 && return previous
338 current = kind in ("chebyshev2", "hermite") ? PX * fraction(2) : kind == "laguerre" ? linear(FONE, -FONE) : PX
339 for j in 1:count-1
340 factor = kind == "legendre" ? PX * fraction(2j + 1) : kind == "laguerre" ? linear(fraction(2j + 1), -FONE) : PX * fraction(2)
341 multiplier = kind in ("legendre", "laguerre") ? j : kind == "hermite" ? 2j : 1
342 next_value = factor * current - previous * fraction(multiplier)
343 kind in ("legendre", "laguerre") && (next_value = next_value * fraction(1, j + 1))
344 previous, current = current, next_value
345 end
346 current
347end

384function Parser(source::String)
385 tokens = Token[]
386 for match in eachmatch(r"[0-9]+|[A-Z][A-Z0-9]*|[^ ]", source)
387 word = String(match.match)
388 kind = isdigit(first(word)) ? NumberToken : 'A' <= first(word) <= 'Z' ? NameToken : SymbolToken
389 push!(tokens, Token(kind, word, match.offset))
390 end
391 push!(tokens, Token(EndToken, "", length(source) + 1))
392 Parser(tokens, 1)
393end
394current(parser::Parser) = parser.tokens[parser.position]
395syntax_error(parser::Parser) = fail("SYNTAX AT " * (current(parser).kind == EndToken ? "END" : string(current(parser).position)))
396function accept!(parser::Parser, word::String)
397 if current(parser).kind == SymbolToken && current(parser).text == word
398 parser.position += 1
399 return true
400 end
401 false
402end
403expect!(parser::Parser, word::String) = accept!(parser, word) || syntax_error(parser)
404function primary!(parser::Parser)::Expression
405 token = current(parser)
406 if token.kind == NumberToken
407 parser.position += 1
408 return Literal(fraction(parse(Int64, token.text; base = 10)))
409 end
410 accept!(parser, "x") && return Variable()
411 if token.kind == NameToken
412 parser.position += 1
413 order = 0
414 while accept!(parser, "'")
415 order += 1
416 end
417 reference = Reference(token.text, order)
418 accept!(parser, "(") || return reference
419 argument = expression!(parser)
420 expect!(parser, ")")
421 return Call(reference, argument)
422 end
423 if accept!(parser, "(")
424 result = expression!(parser)
425 expect!(parser, ")")
426 return result
427 end
428 syntax_error(parser)
429end
430function factor!(parser::Parser)::Expression
431 result = primary!(parser)
432 if accept!(parser, "^")
433 current(parser).kind == NumberToken || syntax_error(parser)
434 exponent = parse(Int, current(parser).text; base = 10)
435 parser.position += 1
436 result = Power(result, exponent)
437 end
438 result
439end
440function term!(parser::Parser)::Expression
441 result = factor!(parser)
442 while true
443 if accept!(parser, "*")
444 operation = '*'
445 elseif accept!(parser, "/")
446 operation = '/'
447 elseif current(parser).kind == NameToken || current(parser).text in ("x", "(")
448 operation = '*'
449 else
450 break
451 end
452 result = Binary(operation, result, factor!(parser))
453 end
454 result
455end
456function expression!(parser::Parser)::Expression
457 negative = accept!(parser, "-")
458 result = term!(parser)
459 negative && (result = Negative(result))
460 while true
461 if accept!(parser, "+")
462 operation = '+'
463 elseif accept!(parser, "-")
464 operation = '-'
465 else
466 break
467 end
468 result = Binary(operation, result, term!(parser))
469 end
470 result
471end
472function parse_definition(source::String)
473 parser = Parser(source)
474 result = expression!(parser)
475 current(parser).kind == EndToken || syntax_error(parser)
476 result
477end

500 a = evaluate(node.left, storage)
501 b = evaluate(node.right, storage)

503 node.operation == '-' && return a - b
504 if node.operation == '*'
505 product_limit(a, b)
506 return a * b
507 end
508 require_divisor(b)
509 degree(b) >= 1 && fail("NONCONSTANT DIVISOR")
510 a / leading(b)

513const REFERENCES = Dict(
514 "PRINT" => [2], "DELETE" => [2], "DEGREE" => [2], "ADD" => [3, 4], "SUB" => [3, 4], "MUL" => [3, 4],
515 "POW" => [3], "DIV" => [4, 5], "GCD" => [3, 4], "XGCD" => [5, 6], "LCM" => [3, 4], "INVMOD" => [3, 4],
516 "CRT" => [3, 4, 5, 6], "MODPOW" => [3, 5], "DERIV" => [3], "INTEG" => [3], "AREA" => [2], "TANGENT" => [3],
517 "TAYLOR" => [2], "COMPOSE" => [3, 4], "EVAL" => [2], "ROOTS" => [2], "FACTOR" => [2], "SQFREE" => [2],
518 "DESCARTES" => [2], "STURM" => [2], "ISOLATE" => [2], "RESULTANT" => [2, 3], "DISCRIM" => [2], "POWERSUM" => [2]
519)
520function execute!(storage::Storage, source::String)
521 words = String.(split(source, ' '))
522 command = words[1]
523 name = length(words) >= 2 ? words[2] : ""
524 pending = Pair{String,Polynomial}[]
525 output = String[]
526 deletion = nothing
527 function save!(destination::String, value::Polynomial)
528 push!(pending, destination => value)
529 push!(output, "$destination = $(notation(value))")
530 end
531 if command == "DEF"
532 tree = parse_definition(source[length(name)+6:end])
533 save!(name, evaluate(tree, storage))
534 else
535 ((command == "DIV" && words[2] == words[3]) || (command == "XGCD" && length(unique(words[2:4])) < 3)) && fail("SAME NAME")
536 operands = Polynomial[lookup(storage, words[i]) for i in get(REFERENCES, command, Int[])]
537 a = length(operands) >= 1 ? operands[1] : PZERO
538 b = length(operands) >= 2 ? operands[2] : PZERO
539 literal_words = command in ("AREA", "STURM") ? words[3:4] : command == "TANGENT" ? words[4:4] :
540 command == "TAYLOR" ? words[3:3] : command in ("EVAL", "INTERP") ? words[3:end] : String[]
541 bounds = command == "STURM" ? parse_bound.(literal_words) : Bound[]
542 values = command == "STURM" ? Fraction[] : parse_fraction.(literal_words)
543 if command == "PRINT"
544 push!(output, "$name = $(notation(a))")
545 elseif command == "DELETE"
546 deletion = name
547 push!(output, "deleted $name")
548 elseif command == "LIST"
549 for key in sort!(collect(keys(storage)))
550 push!(output, "$key = $(notation(storage[key]))")
551 end
552 isempty(output) && push!(output, "no polynomials")
553 elseif command == "DEGREE"
554 push!(output, "deg($name) = $(iszero(a) ? "-inf" : degree(a))")
555 elseif command in ("ADD", "SUB", "MUL")
556 command == "MUL" && product_limit(a, b)
557 save!(name, command == "ADD" ? a + b : command == "SUB" ? a - b : a * b)
558 elseif command == "POW"
559 exponent = parse(Int, words[4])
560 degree_limit(!iszero(a) && degree(a) * exponent > 30)
561 save!(name, a ^ exponent)
562 elseif command == "DIV"
563 require_divisor(b)
564 q, r = divrem(a, b)
565 save!(name, q)
566 save!(words[3], r)
567 elseif command == "GCD"
568 save!(name, polynomial_gcd(a, b))
569 elseif command == "XGCD"
570 result = extended_gcd(a, b)
571 for i in 1:3
572 save!(words[i + 1], result[i])
573 end
574 elseif command == "LCM"
575 result = PZERO
576 if !iszero(a) && !iszero(b)
577 common = polynomial_gcd(a, b)
578 degree_limit(degree(a) + degree(b) - degree(common) > 30)
579 result = monic(quotient(a, common) * b)
580 end
581 save!(name, result)
582 elseif command == "INVMOD"
583 require_divisor(b)
584 degree(b) == 0 && fail("DEGREE TOO LOW")
585 common, inverse, _ = extended_gcd(a, b)
586 isone(common) || fail("NOT COPRIME")
587 save!(name, rem(inverse, b))
588 elseif command == "CRT"
589 c, d = operands[3:4]
590 require_divisor(b)
591 require_divisor(d)
592 degree_limit(degree(b) + degree(d) > 30)
593 common, inverse, _ = extended_gcd(b, d)
594 isone(common) || fail("NOT COPRIME")
595 first = rem(a, b)
596 second = rem(c, d)
597 correction = rem((second - first) * inverse, d)
598 save!(name, first + b * correction)
599 elseif command == "MODPOW"
600 require_divisor(b)
601 degree_limit(2 * degree(b) - 2 > 30)
602 result, base = rem(PONE, b), rem(a, b)
603 exponent = parse(Int, words[4])
604 while exponent > 0
605 isodd(exponent) && (result = rem(result * base, b))
606 exponent ÷= 2
607 exponent > 0 && (base = rem(base * base, b))
608 end
609 save!(name, result)
610 elseif command == "DERIV"
611 save!(name, derivative(a, parse(Int, words[4])))
612 elseif command == "INTEG"
613 degree_limit(!iszero(a) && degree(a) + 1 > 30)
614 save!(name, integral(a))
615 elseif command == "AREA"
616 antiderivative = integral(a)
617 value = evaluate(antiderivative, values[2]) - evaluate(antiderivative, values[1])
618 push!(output, "integral($name, $(fraction_text(values[1])), $(fraction_text(values[2]))) = $(fraction_text(value))")
619 elseif command == "TANGENT"
620 point = values[1]
621 value = evaluate(a, point)
622 slope = evaluate(derivative(a), point)
623 save!(name, linear(value - slope * point, slope))
624 elseif command == "TAYLOR"
625 point = values[1]
626 variable = iszero(point) ? "x" : point > 0 ? "(x - $(fraction_text(point)))" : "(x + $(fraction_text(-point)))"
627 push!(output, "taylor($name, $(fraction_text(point))) = $(notation(compose(a, linear(point)), variable))")
628 elseif command == "COMPOSE"
629 composition_limit(a, b)
630 save!(name, compose(a, b))
631 elseif command == "EVAL"
632 for point in values
633 push!(output, "$name($(fraction_text(point))) = $(fraction_text(evaluate(a, point)))")
634 end
635 elseif command in ("ROOTS", "FACTOR")
636 require_nonzero(a)
637 roots, remaining = rational_roots(a)
638 if command == "ROOTS"
639 text = isempty(roots) ? "none" : join(("$(fraction_text(root.value)) ($(root.multiplicity))" for root in roots), ", ")
640 push!(output, "roots($name) = $text")
641 else
642 push!(output, "factor($name) = $(factorization(a, roots, remaining))")
643 end
644 elseif command == "SQFREE"
645 require_nonzero(a)
646 push!(output, "sqfree($name) = $(squarefree_factors(a))")
647 elseif command == "DESCARTES"
648 require_nonzero(a)
649 signs = Int[sign(numerator(value)) for value in a.coefficients]
650 positive = sign_changes(reverse(signs))
651 signs[2:2:end] .*= -1
652 push!(output, "descartes($name) = positive $positive, negative $(sign_changes(reverse(signs)))")
653 elseif command == "STURM"
654 require_nonzero(a)
655 compare_bounds(bounds[1], bounds[2]) >= 0 && fail("BAD INTERVAL")
656 sequence = sturm_sequence(a)
657 push!(output, "sturm($name, $(bound_text(bounds[1])), $(bound_text(bounds[2]))) = $(root_count(sequence, bounds[1], bounds[2]))")
658 elseif command == "ISOLATE"
659 require_nonzero(a)
660 precision = parse(Int, words[3])
661 push!(output, "isolate($name, $precision) = $(isolate(a, precision))")
662 elseif command == "RESULTANT"
663 require_nonzero(a)
664 require_nonzero(b)
665 push!(output, "res($name, $(words[3])) = $(fraction_text(resultant(a, b)))")
666 elseif command in ("DISCRIM", "POWERSUM")
667 require_nonzero(a)
668 degree(a) == 0 && fail("DEGREE TOO LOW")
669 if command == "DISCRIM"
670 value = resultant(a, derivative(a)) / leading(a)
671 isodd((degree(a) * (degree(a) - 1)) ÷ 2) && (value = -value)
672 push!(output, "disc($name) = $(fraction_text(value))")
673 else
674 push!(output, "powersum($name) = $(power_sums(a, parse(Int, words[3])))")
675 end
676 elseif command == "INTERP"
677 save!(name, interpolate(values))
678 elseif command == "SPECIAL"
679 save!(name, special(words[3], parse(Int, words[4])))
680 end
681 end
682 deletion !== nothing && delete!(storage, deletion)
683 for (destination, value) in pending
684 storage[destination] = value
685 end
686 output
687end
