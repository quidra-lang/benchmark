4enum Operation: string
5{
6 case SUM = 'SUM'; case MIN = 'MIN'; case MAX = 'MAX'; case GCD = 'GCD';
7 public function combine(int $a, int $b): int
8 {
9 return match ($this) { self::SUM => $a + $b, self::MIN => min($a, $b), self::MAX => max($a, $b), self::GCD => gcd($a, $b) };
10 }

15function gcd(int $a, int $b): int
16{
17 $a = abs($a); $b = abs($b);
18 while ($b !== 0) [$a, $b] = [$b, $a % $b];
19 return $a;

22{
23 $result = 1;

29 $result = 0;

43final readonly class Layout
44{
45 public array $ranges;
46 public int $count, $maxNode;
47 public function __construct(int $n)
48 {
49 $ranges = array_fill(0, 4 * $n, null);
50 $maxNode = 0;
51 $build = function (int $node, int $left, int $right) use (&$build, &$ranges, &$maxNode): void {
52 $ranges[$node] = new Interval($left, $right);
53 $maxNode = max($maxNode, $node);
54 if ($left === $right) return;
55 $middle = intdiv($left + $right, 2);
56 $build(2 * $node, $left, $middle);
57 $build(2 * $node + 1, $middle + 1, $right);
58 };
59 $build(1, 0, $n - 1);
60 $this->ranges = $ranges;
61 $this->count = 2 * $n - 1;
62 $this->maxNode = $maxNode;
63 }
64 public function nodes(): iterable
65 {
66 foreach ($this->ranges as $node => $range) if ($range !== null) yield $node => $range;
67 }

76final class SegmentTree
77{
78 public readonly Layout $shape;
79 private array $values;
80 public function __construct(array $contents, public readonly Operation $operation)
81 {
82 $this->shape = new Layout(count($contents));
83 $this->values = array_fill(0, 4 * count($contents), 0);
84 $this->build(1, $contents);
85 }
86 private function build(int $node, array $contents): void
87 {
88 $range = $this->shape->ranges[$node];
89 if ($range->leaf()) {
90 $this->values[$node] = $this->operation === Operation::GCD ? abs($contents[$range->left]) : $contents[$range->left];
91 } else {
92 $this->build(2 * $node, $contents);
93 $this->build(2 * $node + 1, $contents);
94 $this->pull($node);
95 }
96 }
97 private function pull(int $node): void { $this->values[$node] = $this->operation->combine($this->values[2 * $node], $this->values[2 * $node + 1]); }
98 public function root(): int { return $this->values[1]; }
99 public function query(int $left, int $right, int $node = 1): ScalarAnswer
100 {
101 $range = $this->shape->ranges[$node];
102 if ($range->covered($left, $right)) return new ScalarAnswer($this->values[$node], 1);
103 if ($right <= $range->middle() || $left > $range->middle()) {
104 $child = $this->query($left, $right, $right <= $range->middle() ? 2 * $node : 2 * $node + 1);
105 return new ScalarAnswer($child->value, $child->visited + 1);
106 }
107 $a = $this->query($left, $right, 2 * $node);
108 $b = $this->query($left, $right, 2 * $node + 1);
109 return new ScalarAnswer($this->operation->combine($a->value, $b->value), $a->visited + $b->visited + 1);
110 }
111 public function update(int $index, int $value, int $node = 1): int
112 {
113 $range = $this->shape->ranges[$node];
114 if ($range->leaf()) {
115 $this->values[$node] = $this->operation === Operation::GCD ? abs($value) : $value;
116 return 1;
117 }
118 $visited = $this->update($index, $value, $index <= $range->middle() ? 2 * $node : 2 * $node + 1);
119 $this->pull($node);
120 return $visited + 1;
121 }
122 public function describe(ScalarAnswer $answer): string { return strtolower($this->operation->value) . "={$answer->value} visited={$answer->visited}"; }
123 public function dump(): array
124 {
125 $rows = ["SEG DUMP {$this->operation->value} nodes={$this->shape->count}"];
126 foreach ($this->shape->nodes() as $node => $range) $rows[] = "NODE $node {$range->left} {$range->right} {$this->values[$node]}";
127 return $rows;
128 }

148final class LazyTree
149{
150 public readonly Layout $shape;
151 private array $values, $assign, $add;
152 public function __construct(array $contents)
153 {
154 $this->shape = new Layout(count($contents));
155 $this->values = $this->assign = array_fill(0, 4 * count($contents), null);
156 $this->add = array_fill(0, 4 * count($contents), 0);
157 $this->build(1, $contents);
158 }
159 private function build(int $node, array $contents): void
160 {
161 $range = $this->shape->ranges[$node];
162 if ($range->leaf()) {
163 $value = $contents[$range->left];
164 $this->values[$node] = new Aggregate($value, $value, $value);
165 } else {
166 $this->build(2 * $node, $contents);
167 $this->build(2 * $node + 1, $contents);
168 $this->pull($node);
169 }
170 }
171 private function pull(int $node): void { $this->values[$node] = Aggregate::merge($this->values[2 * $node], $this->values[2 * $node + 1]); }
172 public function root(): Aggregate { return $this->values[1]; }
173 private function apply(int $node, int $value, bool $assignment): void
174 {
175 $range = $this->shape->ranges[$node];
176 $old = $this->values[$node];
177 $this->values[$node] = $assignment ? new Aggregate($value * $range->length(), $value, $value)
178 : new Aggregate($old->sum + $value * $range->length(), $old->min + $value, $old->max + $value);
179 if ($range->leaf()) return;
180 if ($assignment) { $this->assign[$node] = $value; $this->add[$node] = 0; }
181 else $this->add[$node] += $value;
182 }
183 private function push(int $node): int
184 {
185 if ($this->assign[$node] === null && $this->add[$node] === 0) return 0;
186 for ($child = 2 * $node; $child <= 2 * $node + 1; $child++) {
187 if ($this->assign[$node] !== null) $this->apply($child, $this->assign[$node], true);
188 if ($this->add[$node] !== 0) $this->apply($child, $this->add[$node], false);
189 }
190 $this->assign[$node] = null;
191 $this->add[$node] = 0;
192 return 1;
193 }
194 public function update(int $left, int $right, int $value, bool $assignment): Work
195 {
196 $work = new Work();
197 $visit = function (int $node) use (&$visit, $left, $right, $value, $assignment, $work): void {
198 $work->visited++;
199 $range = $this->shape->ranges[$node];
200 if ($range->covered($left, $right)) {
201 $work->covered++;
202 $this->apply($node, $value, $assignment);
203 return;
204 }
205 $work->pushes += $this->push($node);
206 if ($left <= $range->middle()) $visit(2 * $node);
207 if ($right > $range->middle()) $visit(2 * $node + 1);
208 $this->pull($node);
209 };
210 $visit(1);
211 return $work;
212 }
213 public function query(int $left, int $right, int $node = 1): LazyAnswer
214 {
215 $range = $this->shape->ranges[$node];
216 if ($range->covered($left, $right)) return new LazyAnswer($this->values[$node], 1, 0);
217 $pushes = $this->push($node);
218 if ($right <= $range->middle() || $left > $range->middle()) {
219 $child = $this->query($left, $right, $right <= $range->middle() ? 2 * $node : 2 * $node + 1);
220 return new LazyAnswer($child->value, $child->visited + 1, $child->pushes + $pushes);
221 }
222 $a = $this->query($left, $right, 2 * $node);
223 $b = $this->query($left, $right, 2 * $node + 1);
224 return new LazyAnswer(Aggregate::merge($a->value, $b->value), $a->visited + $b->visited + 1, $a->pushes + $b->pushes + $pushes);
225 }
226 public function dump(): array
227 {
228 $rows = ["LAZY DUMP nodes={$this->shape->count}"];
229 foreach ($this->shape->nodes() as $node => $range) {
230 $assignment = $this->assign[$node] ?? '-';
231 $rows[] = "NODE $node {$range->left} {$range->right} {$this->values[$node]} assign=$assignment add={$this->add[$node]}";
232 }
233 return $rows;
234 }
235}

328{
329 private array $minimum, $maximum;
330 public readonly int $levels, $entries;
331 public function __construct(private readonly array $contents)
332 {
333 $n = count($contents);
334 $this->levels = floorLog2($n) + 1;
335 $this->minimum = $this->maximum = [range(0, $n - 1)];
336 $entries = $n;
337 for ($level = 1; $level < $this->levels; $level++) {
338 $width = 1 << $level;
339 $small = $large = [];
340 for ($i = 0; $i + $width <= $n; $i++) {
341 $small[] = $this->choose($this->minimum[$level - 1][$i], $this->minimum[$level - 1][$i + intdiv($width, 2)], true);
342 $large[] = $this->choose($this->maximum[$level - 1][$i], $this->maximum[$level - 1][$i + intdiv($width, 2)], false);
343 }
344 $this->minimum[] = $small;
345 $this->maximum[] = $large;
346 $entries += count($small);
347 }
348 $this->entries = $entries;
349 }
350 private function choose(int $a, int $b, bool $minimum): int
351 {
352 return ($minimum ? $this->contents[$b] < $this->contents[$a] : $this->contents[$b] > $this->contents[$a]) ? $b : $a;
353 }
354 public function query(int $left, int $right): SparseAnswer
355 {
356 $level = floorLog2($right - $left + 1);
357 $end = $right + 1 - (1 << $level);
358 $small = $this->choose($this->minimum[$level][$left], $this->minimum[$level][$end], true);
359 $large = $this->choose($this->maximum[$level][$left], $this->maximum[$level][$end], false);
360 return new SparseAnswer($this->contents[$small], $small, $this->contents[$large], $large, $level);
361 }

430final readonly class PersistentNode
431{
432 public function __construct(public int $value, public ?self $left = null, public ?self $right = null) {}
433 public static function build(array $contents, int $left, int $right): self
434 {
435 if ($left === $right) return new self($contents[$left]);
436 $middle = intdiv($left + $right, 2);
437 $a = self::build($contents, $left, $middle);
438 $b = self::build($contents, $middle + 1, $right);
439 return new self($a->value + $b->value, $a, $b);
440 }
441 public function change(int $left, int $right, int $index, int $value, bool $increment): array
442 {
443 if ($left === $right) return [new self($increment ? $this->value + $value : $value), 1];
444 $middle = intdiv($left + $right, 2);
445 $a = $this->left;
446 $b = $this->right;
447 if ($index <= $middle) [$a, $created] = $a->change($left, $middle, $index, $value, $increment);
448 else [$b, $created] = $b->change($middle + 1, $right, $index, $value, $increment);
449 return [new self($a->value + $b->value, $a, $b), $created + 1];
450 }
451 public function query(int $start, int $end, int $left, int $right): ScalarAnswer
452 {
453 if ($left <= $start && $end <= $right) return new ScalarAnswer($this->value, 1);
454 $middle = intdiv($start + $end, 2);
455 if ($right <= $middle || $left > $middle) {
456 $answer = $right <= $middle ? $this->left->query($start, $middle, $left, $right) : $this->right->query($middle + 1, $end, $left, $right);
457 return new ScalarAnswer($answer->value, $answer->visited + 1);
458 }
459 $a = $this->left->query($start, $middle, $left, $right);
460 $b = $this->right->query($middle + 1, $end, $left, $right);
461 return new ScalarAnswer($a->value + $b->value, $a->visited + $b->visited + 1);
462 }

465final class PersistentTree
466{
467 private array $versions, $valueRoots, $values;
468 public int $positionNodes, $valueNodes;
469 public readonly int $n, $distinct;
470 public function __construct(array $contents)
471 {
472 $this->n = count($contents);
473 $this->versions = [PersistentNode::build($contents, 0, $this->n - 1)];
474 $this->positionNodes = 2 * $this->n - 1;
475 $sorted = $contents;
476 sort($sorted, SORT_NUMERIC);
477 $this->values = [];
478 foreach ($sorted as $value) if ($this->values === [] || $this->values[count($this->values) - 1] !== $value) $this->values[] = $value;
479 $this->distinct = count($this->values);
480 $ranks = array_flip($this->values);
481 $this->valueRoots = [PersistentNode::build(array_fill(0, $this->distinct, 0), 0, $this->distinct - 1)];
482 $this->valueNodes = 2 * $this->distinct - 1;
483 foreach ($contents as $i => $value) {
484 [$root, $created] = $this->valueRoots[$i]->change(0, $this->distinct - 1, $ranks[$value], 1, true);
485 $this->valueRoots[] = $root;
486 $this->valueNodes += $created;
487 }
488 }
489 public function versionCount(): int { return count($this->versions); }
490 public function setValue(int $version, int $index, int $value): int
491 {
492 [$root, $created] = $this->versions[$version]->change(0, $this->n - 1, $index, $value, false);
493 $this->versions[] = $root;
494 $this->positionNodes += $created;
495 return $created;
496 }
497 public function query(int $version, int $left, int $right): ScalarAnswer { return $this->versions[$version]->query(0, $this->n - 1, $left, $right); }
498 public function kth(int $left, int $right, int $k): ScalarAnswer
499 {
500 $u = $this->valueRoots[$right + 1]; $w = $this->valueRoots[$left];
501 $first = 0; $last = $this->distinct - 1; $visited = 1;
502 while ($first < $last) {
503 $middle = intdiv($first + $last, 2);
504 $difference = $u->left->value - $w->left->value;
505 if ($k <= $difference) { $u = $u->left; $w = $w->left; $last = $middle; }
506 else { $k -= $difference; $u = $u->right; $w = $w->right; $first = $middle + 1; }
507 $visited++;
508 }
509 return new ScalarAnswer($this->values[$first], $visited);
510 }

513final class GridFenwick
514{
515 private array $tree;
516 public function __construct(public readonly int $height, public readonly int $width, array $contents)
517 {
518 $this->tree = array_fill(0, $height + 1, array_fill(0, $width + 1, 0));
519 foreach ($contents as $index => $value) $this->add(intdiv($index, $width), $index % $width, $value);
520 }
521 public function add(int $row, int $column, int $value): int
522 {
523 $visited = 0;
524 for ($y = $row + 1; $y <= $this->height; $y += $y & -$y)
525 for ($z = $column + 1; $z <= $this->width; $z += $z & -$z) { $this->tree[$y][$z] += $value; $visited++; }
526 return $visited;
527 }
528 private function prefix(int $row, int $column): ScalarAnswer
529 {
530 $sum = $visited = 0;
531 for ($y = $row; $y > 0; $y -= $y & -$y)
532 for ($z = $column; $z > 0; $z -= $z & -$z) { $sum += $this->tree[$y][$z]; $visited++; }
533 return new ScalarAnswer($sum, $visited);
534 }
535 public function query(int $top, int $left, int $bottom, int $right): ScalarAnswer
536 {
537 $a = $this->prefix($bottom + 1, $right + 1);
538 $b = $this->prefix($top, $right + 1);
539 $c = $this->prefix($bottom + 1, $left);
540 $d = $this->prefix($top, $left);
541 return new ScalarAnswer($a->value - $b->value - $c->value + $d->value, $a->visited + $b->visited + $c->visited + $d->visited);
542 }
543 public function dump(): array
544 {
545 $rows = ["GRID DUMP rows={$this->height} cols={$this->width}"];
546 for ($row = 1; $row <= $this->height; $row++) $rows[] = "ROW $row " . implode(' ', array_slice($this->tree[$row], 1));
547 return $rows;
548 }
549}

556final class Laboratory
557{
558 private array $a, $original;
559 private ?SegmentTree $segment = null;
560 private ?LazyTree $lazy = null;
561 private ?PointFenwick $fenwick = null;
562 private ?RangeFenwick $rangeFenwick = null;
563 private ?SparseTable $sparse = null;
564 private ?SqrtBlocks $sqrt = null;
565 private ?PersistentTree $persistent = null;
566 private ?GridFenwick $grid = null;
567 public function __construct(array $contents) { $this->a = $this->original = $contents; }
568 private function index(int $index): void
569 {
570 if ($index < 0 || $index >= count($this->a)) throw new CommandError("BAD INDEX $index");
571 }
572 private function range(int $left, int $right): void
573 {
574 $this->index($left); $this->index($right);
575 if ($left > $right) throw new CommandError("BAD RANGE $left $right");
576 }
577 private function cell(int $row, int $column): void
578 {
579 if ($row < 0 || $row >= $this->grid->height || $column < 0 || $column >= $this->grid->width) throw new CommandError("BAD CELL $row $column");
580 }
581 private function version(int $version): void
582 {
583 if ($version < 0 || $version >= $this->persistent->versionCount()) throw new CommandError("BAD VERSION $version");
584 }
585 private function arrayCount(): int
586 {
587 return count(array_filter([$this->segment, $this->lazy, $this->fenwick, $this->rangeFenwick, $this->sparse, $this->sqrt, $this->persistent], static fn($tree) => $tree !== null));
588 }
589 private function updateArray(int $left, int $right, int $value, bool $assignment = false): void
590 {
591 for ($i = $left; $i <= $right; $i++) $this->a[$i] = $assignment ? $value : $this->a[$i] + $value;
592 }
593 private function reset(): array
594 {
595 $this->a = $this->original;
596 if ($this->segment !== null) $this->segment = new SegmentTree($this->a, $this->segment->operation);
597 if ($this->lazy !== null) $this->lazy = new LazyTree($this->a);
598 if ($this->fenwick !== null) $this->fenwick = new PointFenwick($this->a);
599 if ($this->rangeFenwick !== null) $this->rangeFenwick = new RangeFenwick($this->a);
600 if ($this->sparse !== null) $this->sparse = new SparseTable($this->a);
601 if ($this->sqrt !== null) $this->sqrt = new SqrtBlocks($this->a, $this->sqrt->block);
602 if ($this->persistent !== null) $this->persistent = new PersistentTree($this->a);
603 return ['RESET rebuilt=' . $this->arrayCount()];
604 }
605 private function build(string $kind, array $words): array
606 {
607 $n = count($this->a);
608 switch ($kind) {
609 case 'SEG':
610 $operation = Operation::tryFrom($words[2]) ?? throw new CommandError("BAD OP {$words[2]}");
611 $tree = new SegmentTree($this->a, $operation);
612 $this->segment = $tree;
613 return ["SEG BUILD {$operation->value} nodes={$tree->shape->count} maxnode={$tree->shape->maxNode} root=" . $tree->root()];
614 case 'LAZY':
615 $tree = new LazyTree($this->a); $this->lazy = $tree;
616 return ["LAZY BUILD nodes={$tree->shape->count} " . $tree->root()];
617 case 'FEN':
618 $this->fenwick = new PointFenwick($this->a);
619 return ["FEN BUILD n=$n total=" . array_sum($this->a)];
620 case 'RFEN':
621 $this->rangeFenwick = new RangeFenwick($this->a);
622 return ["RFEN BUILD n=$n total=" . array_sum($this->a)];
623 case 'SPARSE':
624 $tree = new SparseTable($this->a); $this->sparse = $tree;
625 return ["SPARSE BUILD levels={$tree->levels} entries={$tree->entries}"];
626 case 'SQRT':
627 $block = isset($words[2]) ? (int)$words[2] : integerSqrt($n);
628 if ($block > $n) throw new CommandError("BAD BLOCK $block");
629 $tree = new SqrtBlocks($this->a, $block); $this->sqrt = $tree;
630 return ["SQRT BUILD block=$block blocks={$tree->blocks}"];
631 case 'PERSIST':
632 $tree = new PersistentTree($this->a); $this->persistent = $tree;
633 return ["PERSIST BUILD nodes={$tree->positionNodes} distinct={$tree->distinct} valuenodes={$tree->valueNodes}"];
634 case 'GRID':
635 $height = (int)$words[2]; $width = (int)$words[3];
636 $values = array_map('intval', array_slice($words, 4));
637 $count = count($values); $expected = $height * $width;
638 if ($count !== $expected) throw new CommandError("BAD COUNT $count $expected");
639 $this->grid = new GridFenwick($height, $width, $values);
640 return ["GRID BUILD rows=$height cols=$width total=" . array_sum($values)];
641 }
642 throw new LogicException('Unknown structure');
643 }
644 private function check(int $left, int $right): array
645 {
646 $this->range($left, $right);
647 $sum = $gcd = 0; $min = $max = $this->a[$left];
648 for ($i = $left; $i <= $right; $i++) {
649 $sum += $this->a[$i]; $min = min($min, $this->a[$i]); $max = max($max, $this->a[$i]); $gcd = gcd($gcd, $this->a[$i]);
650 }
651 $rows = ["CHECK $left $right sum=$sum min=$min max=$max gcd=$gcd"];
652 $ok = $diff = 0;
653 $append = static function (string $kind, bool $same, string $items) use (&$rows, &$ok, &$diff): void {
654 if ($same) $ok++; else $diff++;
655 $rows[] = "$kind " . ($same ? 'OK' : 'DIFF') . " $items";
656 };
657 if ($this->segment !== null) {
658 $answer = $this->segment->query($left, $right);
659 $expected = match ($this->segment->operation) { Operation::SUM => $sum, Operation::MIN => $min, Operation::MAX => $max, Operation::GCD => $gcd };
660 $append('SEG', $answer->value === $expected, $this->segment->describe($answer));
661 }
662 if ($this->lazy !== null) {
663 $answer = $this->lazy->query($left, $right);
664 $append('LAZY', $answer->value->sum === $sum && $answer->value->min === $min && $answer->value->max === $max, (string)$answer);
665 }
666 if ($this->fenwick !== null) {
667 $answer = $this->fenwick->query($left, $right); $append('FEN', $answer->value === $sum, (string)$answer);
668 }
669 if ($this->rangeFenwick !== null) {
670 $answer = $this->rangeFenwick->query($left, $right); $append('RFEN', $answer->value === $sum, (string)$answer);
671 }
672 if ($this->sparse !== null) {
673 $answer = $this->sparse->query($left, $right); $append('SPARSE', $answer->min === $min && $answer->max === $max, (string)$answer);
674 }
675 if ($this->sqrt !== null) {
676 $answer = $this->sqrt->query($left, $right); $append('SQRT', $answer->value === $sum, (string)$answer);
677 }
678 if ($this->persistent !== null) {
679 $version = $this->persistent->versionCount() - 1;
680 $answer = $this->persistent->query($version, $left, $right); $append('PERSIST', $answer->value === $sum, "version=$version $answer");
681 }
682 $rows[] = "RESULT ok=$ok diff=$diff";
683 return $rows;
684 }
685 private function mo(array $words): array
686 {
687 $queries = [];
688 for ($i = 1; $i < count($words); $i += 2) {
689 $left = (int)$words[$i]; $right = (int)$words[$i + 1];
690 $this->range($left, $right);
691 $queries[] = new MoQuery($left, $right, count($queries) + 1);
692 }
693 $block = integerSqrt(count($this->a));
694 $ordered = $queries;
695 usort($ordered, static function (MoQuery $a, MoQuery $b) use ($block): int {
696 $first = intdiv($a->left, $block); $second = intdiv($b->left, $block);
697 return ($first <=> $second) ?: (($first % 2 === 0 ? $a->right <=> $b->right : $b->right <=> $a->right) ?: $a->number <=> $b->number);
698 });
699 $frequency = $answers = []; $distinct = $moves = $left = 0; $right = -1;
700 $add = function (int $index) use (&$frequency, &$distinct, &$moves): void {
701 $value = $this->a[$index];
702 $old = $frequency[$value] ?? 0;
703 if ($old === 0) $distinct++;
704 $frequency[$value] = $old + 1; $moves++;
705 };
706 $remove = function (int $index) use (&$frequency, &$distinct, &$moves): void {
707 $value = $this->a[$index];
708 if (--$frequency[$value] === 0) $distinct--;
709 $moves++;
710 };
711 foreach ($ordered as $query) {
712 while ($left > $query->left) $add(--$left);
713 while ($right < $query->right) $add(++$right);
714 while ($left < $query->left) $remove($left++);
715 while ($right > $query->right) $remove($right--);
716 $answers[$query->number] = $distinct;
717 }
718 $rows = ['MO queries=' . count($queries) . " block=$block moves=$moves", 'ORDER ' . implode(' ', array_map(static fn(MoQuery $q) => $q->number, $ordered))];
719 foreach ($queries as $query) $rows[] = "ANSWER {$query->number} {$query->left} {$query->right} distinct={$answers[$query->number]}";
720 return $rows;
721 }
722 public function execute(array $words): array
723 {
724 $kind = $words[0];
725 if ($kind === 'LOAD') {
726 $removed = $this->arrayCount();
727 $this->a = $this->original = array_map('intval', array_slice($words, 1));
728 $this->segment = $this->lazy = $this->fenwick = $this->rangeFenwick = $this->sparse = $this->sqrt = $this->persistent = null;
729 return ['LOAD n=' . count($this->a) . " removed=$removed"];
730 }
731 if ($kind === 'RESET') return $this->reset();
732 if ($kind === 'MO') return $this->mo($words);
733 if ($kind === 'CHECK') return $this->check((int)$words[1], (int)$words[2]);
734 if ($kind === 'ARRAY') {
735 $left = (int)$words[1]; $right = (int)$words[2]; $this->range($left, $right);
736 return ["ARRAY $left $right " . implode(' ', array_slice($this->a, $left, $right - $left + 1))];
737 }
738 $action = $words[1];
739 if ($action === 'BUILD') return $this->build($kind, $words);
740 $tree = match ($kind) {
741 'SEG' => $this->segment, 'LAZY' => $this->lazy, 'FEN' => $this->fenwick, 'RFEN' => $this->rangeFenwick,
742 'SPARSE' => $this->sparse, 'SQRT' => $this->sqrt, 'PERSIST' => $this->persistent, 'GRID' => $this->grid
743 };
744 if ($tree === null) throw new CommandError("NOT BUILT $kind");
745 $n = count($this->a);
746 if ($action === 'DUMP') {
747 if ($kind === 'FEN') return ["FEN DUMP n=$n", $tree->bits->dump('T')];
748 if ($kind === 'RFEN') return ["RFEN DUMP n=$n", $tree->b1->dump('B1'), $tree->b2->dump('B2')];
749 return $tree->dump();
750 }
751 if ($kind === 'PERSIST') {
752 if ($action === 'NODES') return ['PERSIST NODES versions=' . $tree->versionCount() . " position={$tree->positionNodes} value={$tree->valueNodes}"];
753 if ($action === 'KTH') {
754 $left = (int)$words[2]; $right = (int)$words[3]; $k = (int)$words[4]; $this->range($left, $right);
755 if ($k < 1 || $k > $right - $left + 1) throw new CommandError("BAD K $k");
756 $answer = $tree->kth($left, $right, $k);
757 return ["PERSIST KTH $left $right $k value={$answer->value} visited={$answer->visited}"];
758 }
759 $version = (int)$words[2]; $this->version($version);
760 $index = (int)$words[3];
761 if ($action === 'SET') {
762 $this->index($index); $value = (int)$words[4]; $newVersion = $tree->versionCount();
763 $created = $tree->setValue($version, $index, $value);
764 return ["PERSIST SET $version $index $value version=$newVersion created=$created"];
765 }
766 $right = (int)$words[4]; $this->range($index, $right);
767 return ["PERSIST QUERY $version $index $right " . $tree->query($version, $index, $right)];
768 }
769 if ($kind === 'GRID') {
770 $row = (int)$words[2]; $column = (int)$words[3]; $this->cell($row, $column);
771 if ($action === 'ADD') {
772 $value = (int)$words[4]; $visited = $tree->add($row, $column, $value);
773 return ["GRID ADD $row $column $value visited=$visited"];
774 }
775 $bottom = (int)$words[4]; $right = (int)$words[5]; $this->cell($bottom, $right);
776 if ($row > $bottom || $column > $right) throw new CommandError("BAD RECT $row $column $bottom $right");
777 return ["GRID SUM $row $column $bottom $right " . $tree->query($row, $column, $bottom, $right)];
778 }
779 if ($kind === 'FEN' && $action === 'LOWER') {
780 foreach ($tree->contents as $i => $value) if ($value < 0) throw new CommandError("NEGATIVE $i");
781 $k = (int)$words[2]; [$index, $visited] = $tree->lower($k); $text = $index ?? 'none';
782 return ["FEN LOWER $k index=$text visited=$visited"];
783 }
784 $left = (int)$words[2];
785 if ($kind === 'SEG' && $action === 'UPDATE') {
786 $this->index($left); $value = (int)$words[3]; $visited = $tree->update($left, $value); $this->a[$left] = $value;
787 return ["SEG UPDATE $left $value root=" . $tree->root() . " visited=$visited"];
788 }
789 if ($kind === 'FEN' && ($action === 'ADD' || $action === 'PREFIX')) {
790 $this->index($left);
791 if ($action === 'PREFIX') return ["FEN PREFIX $left " . $tree->bits->prefix($left + 1)];
792 $value = (int)$words[3]; $visited = $tree->add($left, $value); $this->a[$left] += $value;
793 return ["FEN ADD $left $value visited=$visited"];
794 }
795 $right = (int)$words[3]; $this->range($left, $right);
796 if ($action === 'ADD' || $action === 'ASSIGN') {
797 $value = (int)$words[4];
798 if ($kind === 'LAZY') $items = (string)$tree->update($left, $right, $value, $action === 'ASSIGN');
799 elseif ($kind === 'RFEN') $items = 'visited=' . $tree->add($left, $right, $value);
800 else $items = (string)$tree->add($left, $right, $value);
801 $this->updateArray($left, $right, $value, $action === 'ASSIGN');
802 return ["$kind $action $left $right $value $items"];
803 }
804 $answer = $tree->query($left, $right);
805 $items = $kind === 'SEG' ? $tree->describe($answer) : (string)$answer;
806 return ["$kind $action $left $right $items"];
807 }
808}
