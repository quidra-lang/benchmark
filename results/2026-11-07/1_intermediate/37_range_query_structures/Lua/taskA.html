49local function layout(n)
50 local shape = { ranges = {}, count = 2 * n - 1, maxNode = 0 }
51 local function build(node, left, right)
52 shape.ranges[node] = { left = left, right = right, middle = (left + right) // 2, length = right - left + 1 }
53 shape.maxNode = math.max(shape.maxNode, node)
54 if left < right then
55 local middle = (left + right) // 2
56 build(2 * node, left, middle)
57 build(2 * node + 1, middle + 1, right)
58 end
59 end
60 build(1, 0, n - 1)
61 return shape
62end
63local function covered(range, left, right) return left <= range.left and range.right <= right end

65local Segment = {}
66Segment.__index = Segment
67function Segment.new(contents, operation)
68 local self = setmetatable({ shape = layout(#contents), values = {}, operation = operation }, Segment)
69 local function build(node)
70 local range = self.shape.ranges[node]
71 if range.left == range.right then
72 local value = contents[range.left + 1]
73 self.values[node] = operation == "GCD" and math.abs(value) or value
74 else
75 build(2 * node)
76 build(2 * node + 1)
77 self:pull(node)
78 end
79 end
80 build(1)
81 return self
82end
83function Segment:pull(node) self.values[node] = operations[self.operation](self.values[2 * node], self.values[2 * node + 1]) end
84function Segment:query(left, right, node)
85 node = node or 1
86 local range = self.shape.ranges[node]
87 if covered(range, left, right) then return scalar(self.values[node], 1) end
88 if right <= range.middle or left > range.middle then
89 local child = self:query(left, right, right <= range.middle and 2 * node or 2 * node + 1)
90 return scalar(child.value, child.visited + 1)
91 end
92 local a = self:query(left, right, 2 * node)
93 local b = self:query(left, right, 2 * node + 1)
94 return scalar(operations[self.operation](a.value, b.value), a.visited + b.visited + 1)
95end
96function Segment:update(index, value, node)
97 node = node or 1
98 local range = self.shape.ranges[node]
99 if range.left == range.right then
100 self.values[node] = self.operation == "GCD" and math.abs(value) or value
101 return 1
102 end
103 local visited = self:update(index, value, index <= range.middle and 2 * node or 2 * node + 1)
104 self:pull(node)
105 return visited + 1
106end
107function Segment:describe(answer) return format("%s=%d visited=%d", self.operation:lower(), answer.value, answer.visited) end
108function Segment:dump()
109 local rows = { format("SEG DUMP %s nodes=%d", self.operation, self.shape.count) }
110 for node = 1, self.shape.maxNode do
111 local range = self.shape.ranges[node]
112 if range then rows[#rows + 1] = format("NODE %d %d %d %d", node, range.left, range.right, self.values[node]) end
113 end
114 return rows
115end

117local Lazy = {}
118Lazy.__index = Lazy
119function Lazy.new(contents)
120 local self = setmetatable({ shape = layout(#contents), values = {}, assign = {}, add = zeros(4 * #contents) }, Lazy)
121 local function build(node)
122 local range = self.shape.ranges[node]
123 if range.left == range.right then
124 local value = contents[range.left + 1]
125 self.values[node] = aggregate(value, value, value)
126 else
127 build(2 * node)
128 build(2 * node + 1)
129 self:pull(node)
130 end
131 end
132 build(1)
133 return self
134end
135function Lazy:pull(node) self.values[node] = merge(self.values[2 * node], self.values[2 * node + 1]) end
136function Lazy:apply(node, value, assignment)
137 local range, old = self.shape.ranges[node], self.values[node]
138 self.values[node] = assignment and aggregate(value * range.length, value, value)
139 or aggregate(old.sum + value * range.length, old.min + value, old.max + value)
140 if range.left == range.right then return end
141 if assignment then self.assign[node], self.add[node] = value, 0
142 else self.add[node] = self.add[node] + value end
143end
144function Lazy:push(node)
145 local assignment, addition = self.assign[node], self.add[node]
146 if assignment == nil and addition == 0 then return 0 end
147 for child = 2 * node, 2 * node + 1 do
148 if assignment ~= nil then self:apply(child, assignment, true) end
149 if addition ~= 0 then self:apply(child, addition, false) end
150 end
151 self.assign[node], self.add[node] = nil, 0
152 return 1
153end
154function Lazy:update(left, right, value, assignment)
155 local work = { visited = 0, covered = 0, pushes = 0 }
156 local function visit(node)
157 work.visited = work.visited + 1
158 local range = self.shape.ranges[node]
159 if covered(range, left, right) then
160 work.covered = work.covered + 1
161 self:apply(node, value, assignment)
162 return
163 end
164 work.pushes = work.pushes + self:push(node)
165 if left <= range.middle then visit(2 * node) end
166 if right > range.middle then visit(2 * node + 1) end
167 self:pull(node)
168 end
169 visit(1)
170 return work
171end
172function Lazy:query(left, right, node)
173 node = node or 1
174 local range = self.shape.ranges[node]
175 if covered(range, left, right) then return { value = self.values[node], visited = 1, pushes = 0 } end
176 local pushes = self:push(node)
177 if right <= range.middle or left > range.middle then
178 local child = self:query(left, right, right <= range.middle and 2 * node or 2 * node + 1)
179 return { value = child.value, visited = child.visited + 1, pushes = child.pushes + pushes }
180 end
181 local a = self:query(left, right, 2 * node)
182 local b = self:query(left, right, 2 * node + 1)
183 return { value = merge(a.value, b.value), visited = a.visited + b.visited + 1, pushes = a.pushes + b.pushes + pushes }
184end
185function Lazy:dump()
186 local rows = { format("LAZY DUMP nodes=%d", self.shape.count) }
187 for node = 1, self.shape.maxNode do
188 local range = self.shape.ranges[node]
189 if range then
190 local assignment = self.assign[node] == nil and "-" or tostring(self.assign[node])
191 rows[#rows + 1] = format("NODE %d %d %d %s assign=%s add=%d", node, range.left, range.right, aggregateText(self.values[node]), assignment, self.add[node])
192 end
193 end
194 return rows
195end

234function PointFenwick:lower(rest)
235 local position, visited, step = 0, 0, 1 << log2(#self.contents)
236 while step >= 1 do
237 if position + step <= #self.contents then
238 visited = visited + 1
239 local value = self.bits.tree[position + step]
240 if value < rest then rest, position = rest - value, position + step end
241 end
242 step = step // 2
243 end
244 return position == #self.contents and "none" or tostring(position), visited
245end

284function Sparse.new(contents)
285 local initial = {}
286 for i = 1, #contents do initial[i] = i - 1 end
287 local self = setmetatable({ contents = copy(contents), minimum = { initial }, maximum = { copy(initial) }, entries = #contents }, Sparse)
288 local width = 2
289 while width <= #contents do
290 local previous = #self.minimum
291 local small, large = {}, {}
292 for i = 1, #contents - width + 1 do
293 small[i] = self:choose(self.minimum[previous][i], self.minimum[previous][i + width // 2], true)
294 large[i] = self:choose(self.maximum[previous][i], self.maximum[previous][i + width // 2], false)
295 end
296 self.minimum[previous + 1], self.maximum[previous + 1] = small, large
297 self.entries = self.entries + #small
298 width = width * 2
299 end
300 return self
301end
302function Sparse:query(left, right)
303 local level = log2(right - left + 1)
304 local last = right + 1 - (1 << level)
305 local small = self:choose(self.minimum[level + 1][left + 1], self.minimum[level + 1][last + 1], true)
306 local large = self:choose(self.maximum[level + 1][left + 1], self.maximum[level + 1][last + 1], false)
307 return { min = self.contents[small + 1], minpos = small, max = self.contents[large + 1], maxpos = large, level = level }
308end

313 rows[#rows + 1] = format("MAX %d %s", level - 1, concat(self.maximum[level], " "))

320function SqrtBlocks.new(contents, block)
321 local count = (#contents + block - 1) // block
322 local self = setmetatable({ raw = copy(contents), block = block, tags = zeros(count), sums = zeros(count) }, SqrtBlocks)
323 for i, value in ipairs(contents) do

328end
329function SqrtBlocks:split(left, right)
330 local first, last = left // self.block, right // self.block
331 local parts = { cells = {}, whole = {} }
332 if first == last then
333 for i = left, right do parts.cells[#parts.cells + 1] = i end
334 else
335 for i = left, (first + 1) * self.block - 1 do parts.cells[#parts.cells + 1] = i end
336 for i = last * self.block, right do parts.cells[#parts.cells + 1] = i end
337 for group = first + 1, last - 1 do parts.whole[#parts.whole + 1] = group end
338 end
339 return parts
340end
341function SqrtBlocks:add(left, right, value)
342 local parts = self:split(left, right)
343 for _, i in ipairs(parts.cells) do
344 local group = i // self.block + 1
345 self.raw[i + 1] = self.raw[i + 1] + value
346 self.sums[group] = self.sums[group] + value
347 end
348 for _, group in ipairs(parts.whole) do
349 self.tags[group + 1] = self.tags[group + 1] + value
350 self.sums[group + 1] = self.sums[group + 1] + value * self.block
351 end
352 return parts
353end
354function SqrtBlocks:query(left, right)
355 local parts, sum = self:split(left, right), 0
356 for _, i in ipairs(parts.cells) do sum = sum + (self.raw[i + 1] + self.tags[i // self.block + 1]) end
357 for _, group in ipairs(parts.whole) do sum = sum + self.sums[group + 1] end
358 return { value = sum, parts = parts }
359end

362 for group = 1, #self.tags do
363 local first, last = (group - 1) * self.block, math.min(#self.raw, group * self.block) - 1
364 rows[#rows + 1] = format("BLOCK %d %d %d add=%d sum=%d %s", group - 1, first, last, self.tags[group], self.sums[group], concat(self.raw, " ", first + 1, last + 1))
365 end
366 return rows
367end

369local function node(value, left, right) return { value = value, left = left, right = right } end
370local function buildNodes(contents, left, right)
371 if left == right then return node(contents[left + 1]) end
372 local middle = (left + right) // 2
373 local a = buildNodes(contents, left, middle)
374 local b = buildNodes(contents, middle + 1, right)
375 return node(a.value + b.value, a, b)
376end
377local function changeNode(source, left, right, index, value, increment)
378 if left == right then return node(increment and source.value + value or value), 1 end
379 local middle = (left + right) // 2
380 local a, b, created = source.left, source.right, 0
381 if index <= middle then a, created = changeNode(a, left, middle, index, value, increment)
382 else b, created = changeNode(b, middle + 1, right, index, value, increment) end
383 return node(a.value + b.value, a, b), created + 1
384end
385local function queryNode(root, start, last, left, right)
386 if left <= start and last <= right then return scalar(root.value, 1) end
387 local middle = (start + last) // 2
388 if right <= middle or left > middle then
389 local answer
390 if right <= middle then answer = queryNode(root.left, start, middle, left, right)
391 else answer = queryNode(root.right, middle + 1, last, left, right) end
392 return scalar(answer.value, answer.visited + 1)
393 end
394 local a = queryNode(root.left, start, middle, left, right)
395 local b = queryNode(root.right, middle + 1, last, left, right)
396 return scalar(a.value + b.value, a.visited + b.visited + 1)
397end
398local Persistent = {}
399Persistent.__index = Persistent
400function Persistent.new(contents)
401 local sorted, values, ranks = copy(contents), {}, {}
402 table.sort(sorted)
403 for _, value in ipairs(sorted) do
404 if #values == 0 or values[#values] ~= value then values[#values + 1] = value; ranks[value] = #values - 1 end
405 end
406 local self = setmetatable({ n = #contents, values = values, versions = { buildNodes(contents, 0, #contents - 1) },
407 positionNodes = 2 * #contents - 1, valueRoots = { buildNodes(zeros(#values), 0, #values - 1) }, valueNodes = 2 * #values - 1 }, Persistent)
408 for i, value in ipairs(contents) do
409 local root, created = changeNode(self.valueRoots[i], 0, #values - 1, ranks[value], 1, true)
410 self.valueRoots[i + 1] = root
411 self.valueNodes = self.valueNodes + created
412 end
413 return self
414end
415function Persistent:setValue(version, index, value)
416 local root, created = changeNode(self.versions[version + 1], 0, self.n - 1, index, value, false)
417 self.versions[#self.versions + 1] = root
418 self.positionNodes = self.positionNodes + created
419 return created
420end
421function Persistent:query(version, left, right) return queryNode(self.versions[version + 1], 0, self.n - 1, left, right) end
422function Persistent:kth(left, right, k)
423 local u, w = self.valueRoots[right + 2], self.valueRoots[left + 1]
424 local first, last, visited = 0, #self.values - 1, 1
425 while first < last do
426 local middle = (first + last) // 2
427 local difference = u.left.value - w.left.value
428 if k <= difference then u, w, last = u.left, w.left, middle
429 else k, u, w, first = k - difference, u.right, w.right, middle + 1 end
430 visited = visited + 1
431 end
432 return scalar(self.values[first + 1], visited)
433end

542function Laboratory:check(left, right)

544 local sum, minimum, maximum, common = 0, self.a[left + 1], self.a[left + 1], 0
545 for i = left + 1, right + 1 do

548 local rows = { format("CHECK %d %d sum=%d min=%d max=%d gcd=%d", left, right, sum, minimum, maximum, common) }
549 local ok, different = 0, 0
550 local function append(kind, same, items)
551 if same then ok = ok + 1 else different = different + 1 end
552 rows[#rows + 1] = kind .. (same and " OK " or " DIFF ") .. items
553 end
554 for _, kind in ipairs(kinds) do
555 local tree = self.structures[kind]
556 if tree then
557 if kind == "PERSIST" then
558 local version = #tree.versions - 1
559 local answer = tree:query(version, left, right)
560 append(kind, answer.value == sum, format("version=%d %s", version, scalarText(answer)))
561 else
562 local answer = tree:query(left, right)
563 if kind == "SEG" then
564 local expected = ({ SUM = sum, MIN = minimum, MAX = maximum, GCD = common })[tree.operation]
565 append(kind, answer.value == expected, tree:describe(answer))
566 elseif kind == "LAZY" then
567 append(kind, answer.value.sum == sum and answer.value.min == minimum and answer.value.max == maximum, lazyText(answer))
568 elseif kind == "SPARSE" then
569 append(kind, answer.min == minimum and answer.max == maximum, sparseText(answer))
570 elseif kind == "SQRT" then append(kind, answer.value == sum, sqrtText(answer))
571 else append(kind, answer.value == sum, scalarText(answer)) end
572 end
573 end
574 end
575 rows[#rows + 1] = format("RESULT ok=%d diff=%d", ok, different)
576 return rows
577end
578function Laboratory:mo(words)
579 local queries = {}
580 for i = 2, #words, 2 do
581 local left, right = tonumber(words[i]), tonumber(words[i + 1])
582 self:range(left, right)
583 queries[#queries + 1] = { left = left, right = right, number = #queries + 1 }
584 end
585 local block, ordered = isqrt(#self.a), copy(queries)
586 table.sort(ordered, function(a, b)
587 local first, second = a.left // block, b.left // block
588 if first ~= second then return first < second end
589 if a.right ~= b.right then
590 if first % 2 == 0 then return a.right < b.right end
591 return a.right > b.right
592 end
593 return a.number < b.number
594 end)
595 local frequency, answers, order = {}, {}, {}
596 local distinct, moves, left, right = 0, 0, 0, -1
597 local function add(index)
598 local value = self.a[index + 1]
599 local old = frequency[value] or 0
600 if old == 0 then distinct = distinct + 1 end
601 frequency[value], moves = old + 1, moves + 1
602 end
603 local function remove(index)
604 local value = self.a[index + 1]
605 frequency[value], moves = frequency[value] - 1, moves + 1
606 if frequency[value] == 0 then distinct = distinct - 1 end
607 end
608 for _, query in ipairs(ordered) do
609 while left > query.left do left = left - 1; add(left) end
610 while right < query.right do right = right + 1; add(right) end
611 while left < query.left do remove(left); left = left + 1 end
612 while right > query.right do remove(right); right = right - 1 end
613 answers[query.number] = distinct
614 order[#order + 1] = query.number
615 end
616 local rows = { format("MO queries=%d block=%d moves=%d", #queries, block, moves), "ORDER " .. concat(order, " ") }
617 for _, query in ipairs(queries) do rows[#rows + 1] = format("ANSWER %d %d %d distinct=%d", query.number, query.left, query.right, answers[query.number]) end
618 return rows
619end
620local function numericTail(words, start)
621 local values = {}
622 for i = start, #words do values[#values + 1] = tonumber(words[i]) end
623 return values
624end
625function Laboratory:execute(words, line)
626 local kind = words[1]
627 if kind == "LOAD" then
628 local removed = self:arrayCount()
629 self.a = numericTail(words, 2)
630 self.original, self.structures = copy(self.a), {}
631 return { format("LOAD n=%d removed=%d", #self.a, removed) }
632 end
633 if kind == "RESET" then return self:reset() end
634 if kind == "MO" then return self:mo(words) end
635 if kind == "CHECK" then return self:check(tonumber(words[2]), tonumber(words[3])) end
636 if kind == "ARRAY" then
637 local left, right = tonumber(words[2]), tonumber(words[3])
638 self:range(left, right)
639 return { line .. " " .. concat(self.a, " ", left + 1, right + 1) }
640 end
641 local action = words[2]
642 if action == "BUILD" then
643 local parameter
644 if kind == "SEG" then
645 parameter = words[3]
646 if not operations[parameter] then fail("BAD OP " .. parameter) end
647 elseif kind == "SQRT" then
648 parameter = words[3] and tonumber(words[3]) or isqrt(#self.a)
649 if parameter > #self.a then fail(format("BAD BLOCK %d", parameter)) end
650 elseif kind == "GRID" then
651 local height, width = tonumber(words[3]), tonumber(words[4])
652 local values = numericTail(words, 5)
653 if #values ~= height * width then fail(format("BAD COUNT %d %d", #values, height * width)) end
654 self.grid = Grid.new(height, width, values)
655 return { format("GRID BUILD rows=%d cols=%d total=%d", height, width, total(values)) }
656 end
657 return { self:buildArray(kind, parameter) }
658 end
659 local tree = kind == "GRID" and self.grid or self.structures[kind]
660 if not tree then fail("NOT BUILT " .. kind) end
661 if action == "DUMP" then return tree:dump() end
662 local args = numericTail(words, 3)
663 if kind == "PERSIST" then
664 if action == "NODES" then return { format("PERSIST NODES versions=%d position=%d value=%d", #tree.versions, tree.positionNodes, tree.valueNodes) } end
665 if action == "KTH" then
666 local left, right, k = args[1], args[2], args[3]
667 self:range(left, right)
668 if k < 1 or k > right - left + 1 then fail(format("BAD K %d", k)) end
669 local answer = tree:kth(left, right, k)
670 return { format("%s value=%d visited=%d", line, answer.value, answer.visited) }
671 end
672 local version = args[1]
673 self:version(version)
674 if action == "SET" then
675 self:index(args[2])
676 local newVersion = #tree.versions
677 local created = tree:setValue(version, args[2], args[3])
678 return { format("%s version=%d created=%d", line, newVersion, created) }
679 end
680 self:range(args[2], args[3])
681 return { line .. " " .. scalarText(tree:query(version, args[2], args[3])) }
682 end
683 if kind == "GRID" then
684 local row, column = args[1], args[2]
685 self:cell(row, column)
686 if action == "ADD" then return { format("%s visited=%d", line, tree:add(row, column, args[3])) } end
687 self:cell(args[3], args[4])
688 if row > args[3] or column > args[4] then fail("BAD RECT " .. concat(args, " ")) end
689 return { line .. " " .. scalarText(tree:query(row, column, args[3], args[4])) }
690 end
691 if kind == "FEN" and action == "LOWER" then
692 for i, value in ipairs(tree.contents) do if value < 0 then fail(format("NEGATIVE %d", i - 1)) end end
693 local index, visited = tree:lower(args[1])
694 return { format("%s index=%s visited=%d", line, index, visited) }
695 end
696 local left = args[1]
697 if kind == "SEG" and action == "UPDATE" then
698 self:index(left)
699 local visited = tree:update(left, args[2])
700 self.a[left + 1] = args[2]
701 return { format("%s root=%d visited=%d", line, tree.values[1], visited) }
702 end
703 if kind == "FEN" and (action == "ADD" or action == "PREFIX") then
704 self:index(left)
705 if action == "PREFIX" then return { line .. " " .. scalarText(tree.bits:prefix(left + 1)) } end
706 local visited = tree:add(left, args[2])
707 self.a[left + 1] = self.a[left + 1] + args[2]
708 return { format("%s visited=%d", line, visited) }
709 end
710 local right = args[2]
711 self:range(left, right)
712 if action == "ADD" or action == "ASSIGN" then
713 local value, assignment, items = args[3], action == "ASSIGN"
714 if kind == "LAZY" then items = workText(tree:update(left, right, value, assignment))
715 elseif kind == "RFEN" then items = format("visited=%d", tree:add(left, right, value))
716 else items = partitionText(tree:add(left, right, value)) end
717 for i = left + 1, right + 1 do self.a[i] = assignment and value or self.a[i] + value end
718 return { line .. " " .. items }
719 end
720 local answer, items = tree:query(left, right)
721 if kind == "SEG" then items = tree:describe(answer)
722 elseif kind == "LAZY" then items = lazyText(answer)
723 elseif kind == "SPARSE" then items = sparseText(answer)
724 elseif kind == "SQRT" then items = sqrtText(answer)
725 else items = scalarText(answer) end
726 return { line .. " " .. items }
727end

729local function wordsOf(line)
730 local words = {}
731 for word in line:gmatch("%S+") do words[#words + 1] = word end
732 return words
733end
734local first = io.read("*l")
735if not first then return end
736local laboratory = Laboratory.new(numericTail(wordsOf(io.read("*l")), 1))
737local commands = tonumber(io.read("*l"))
738for _ = 1, commands do
739 local words = wordsOf(io.read("*l"))
740 local ok, rows = pcall(function() return laboratory:execute(words, concat(words, " ")) end)
741 if ok then io.write(concat(rows, "\n"), "\n")
742 elseif type(rows) == "table" and rows.message then io.write("ERROR ", rows.message, "\n")
743 else error(rows, 0) end
744 io.flush()
745end
