58 record Range(int l, int r) {
59 int length() { return r - l + 1; }
60 int middle() { return (l + r) / 2; }
61 boolean leaf() { return l == r; }
62 boolean covers(Range other) { return l <= other.l && other.r <= r; }
63 Range left() { return new Range(l, middle()); }
64 Range right() { return new Range(middle() + 1, r); }
65 }
66 static Range whole(int n) { return new Range(0, n - 1); }
67 record Answer(long value, int visited) {
68 Report report() { return new Report(fmt("sum=%d visited=%d", value, visited), new long[]{value}); }
69 }
70 record Report(String text, long[] values) {}
71 record Aggregate(long sum, long min, long max) {
72 static Aggregate single(long value) { return new Aggregate(value, value, value); }
73 Aggregate merge(Aggregate other) { return new Aggregate(sum + other.sum, Math.min(min, other.min), Math.max(max, other.max)); }
74 Aggregate add(long value, int length) { return new Aggregate(sum + value * length, min + value, max + value); }
75 String text() { return fmt("sum=%d min=%d max=%d", sum, min, max); }
76 }

80 Layout(int n) { ranges = new Range[4 * n]; build(1, whole(n)); }

84 if (!range.leaf()) {

92 static final class Segment {
93 final Operation op;
94 final Layout shape;
95 final long[] tree;
96 Segment(long[] values, Operation op) {
97 this.op = op; shape = new Layout(values.length); tree = new long[4 * values.length];
98 build(1, values);
99 }
100 void build(int node, long[] values) {
101 Range range = shape.ranges[node];
102 if (range.leaf()) tree[node] = op.leaf(values[range.l]);
103 else { build(2 * node, values); build(2 * node + 1, values); pull(node); }
104 }
105 void pull(int node) { tree[node] = op.combine(tree[2 * node], tree[2 * node + 1]); }
106 Answer queryAt(int node, Range wanted) {
107 Range range = shape.ranges[node];
108 if (wanted.covers(range)) return new Answer(tree[node], 1);
109 Answer answer;
110 if (wanted.r <= range.middle()) answer = queryAt(2 * node, wanted);
111 else if (wanted.l > range.middle()) answer = queryAt(2 * node + 1, wanted);
112 else {
113 Answer left = queryAt(2 * node, wanted), right = queryAt(2 * node + 1, wanted);
114 answer = new Answer(op.combine(left.value, right.value), left.visited + right.visited);
115 }
116 return new Answer(answer.value, answer.visited + 1);
117 }
118 Report query(Range wanted) {
119 Answer answer = queryAt(1, wanted);
120 return new Report(fmt("%s=%d visited=%d", op.label(), answer.value, answer.visited), new long[]{answer.value});
121 }
122 int update(int node, int index, long value) {
123 Range range = shape.ranges[node];
124 if (range.leaf()) { tree[node] = op.leaf(value); return 1; }
125 int visited = update(index <= range.middle() ? 2 * node : 2 * node + 1, index, value);
126 pull(node);
127 return visited + 1;
128 }
129 List<String> dump() {
130 var rows = new ArrayList<String>();
131 rows.add(fmt("SEG DUMP %s nodes=%d", op, shape.count()));
132 for (int node = 1; node <= shape.maxNode; node++) {
133 Range range = shape.ranges[node];
134 if (range != null) rows.add(fmt("NODE %d %d %d %d", node, range.l, range.r, tree[node]));
135 }
136 return rows;
137 }

147 static final class Lazy {
148 final Layout shape;
149 final LazyNode[] nodes;
150 Lazy(long[] values) {
151 shape = new Layout(values.length); nodes = new LazyNode[4 * values.length]; build(1, values);
152 }
153 void build(int node, long[] values) {
154 Range range = shape.ranges[node];
155 nodes[node] = new LazyNode();
156 if (range.leaf()) nodes[node].value = Aggregate.single(values[range.l]);
157 else { build(2 * node, values); build(2 * node + 1, values); pull(node); }
158 }
159 void pull(int node) { nodes[node].value = nodes[2 * node].value.merge(nodes[2 * node + 1].value); }
160 void apply(int node, long value, boolean assign) {
161 Range range = shape.ranges[node];
162 LazyNode target = nodes[node];
163 target.value = assign ? new Aggregate(value * range.length(), value, value) : target.value.add(value, range.length());
164 if (!range.leaf()) {
165 if (assign) { target.assign = value; target.add = 0; }
166 else target.add += value;
167 }
168 }
169 int push(int node) {
170 LazyNode source = nodes[node];
171 if (source.assign == null && source.add == 0) return 0;
172 for (int child = 2 * node; child <= 2 * node + 1; child++) {
173 if (source.assign != null) apply(child, source.assign, true);
174 if (source.add != 0) apply(child, source.add, false);
175 }
176 source.assign = null; source.add = 0;
177 return 1;
178 }
179 void updateAt(int node, Range wanted, long value, boolean assign, Counts counts) {
180 counts.visited++;
181 Range range = shape.ranges[node];
182 if (wanted.covers(range)) { counts.covered++; apply(node, value, assign); return; }
183 counts.pushes += push(node);
184 if (wanted.l <= range.middle()) updateAt(2 * node, wanted, value, assign, counts);
185 if (wanted.r > range.middle()) updateAt(2 * node + 1, wanted, value, assign, counts);
186 pull(node);
187 }
188 Counts update(Range wanted, long value, boolean assign) {
189 var counts = new Counts();
190 updateAt(1, wanted, value, assign, counts);
191 return counts;
192 }
193 LazyAnswer queryAt(int node, Range wanted) {
194 Range range = shape.ranges[node];
195 if (wanted.covers(range)) return new LazyAnswer(nodes[node].value, 1, 0);
196 int pushes = push(node);
197 LazyAnswer answer;
198 if (wanted.r <= range.middle()) answer = queryAt(2 * node, wanted);
199 else if (wanted.l > range.middle()) answer = queryAt(2 * node + 1, wanted);
200 else {
201 LazyAnswer left = queryAt(2 * node, wanted), right = queryAt(2 * node + 1, wanted);
202 answer = new LazyAnswer(left.value.merge(right.value), left.visited + right.visited, left.pushes + right.pushes);
203 }
204 return new LazyAnswer(answer.value, answer.visited + 1, answer.pushes + pushes);
205 }
206 Report query(Range wanted) {
207 LazyAnswer answer = queryAt(1, wanted);
208 Aggregate value = answer.value;
209 return new Report(fmt("%s visited=%d pushes=%d", value.text(), answer.visited, answer.pushes), new long[]{value.sum, value.min, value.max});
210 }
211 List<String> dump() {
212 var rows = new ArrayList<String>();
213 rows.add(fmt("LAZY DUMP nodes=%d", shape.count()));
214 for (int node = 1; node <= shape.maxNode; node++) {
215 Range range = shape.ranges[node];
216 if (range == null) continue;
217 LazyNode item = nodes[node];
218 rows.add(fmt("NODE %d %d %d %s assign=%s add=%d", node, range.l, range.r, item.value.text(), item.assign == null ? "-" : item.assign.toString(), item.add));
219 }
220 return rows;
221 }

224 static final class Bit {
225 final long[] tree;
226 Bit(int n) { tree = new long[n + 1]; }
227 Bit(long[] values) { this(values.length); for (int i = 0; i < values.length; i++) add(i + 1, values[i]); }
228 int add(int position, long value) {
229 int visited = 0;
230 for (; position < tree.length; position += position & -position) { tree[position] += value; visited++; }
231 return visited;
232 }
233 Answer prefix(int position) {
234 long sum = 0; int visited = 0;
235 for (; position > 0; position -= position & -position) { sum += tree[position]; visited++; }
236 return new Answer(sum, visited);
237 }
238 Answer query(Range range) {
239 Answer a = prefix(range.r + 1), b = prefix(range.l);
240 return new Answer(a.value - b.value, a.visited + b.visited);
241 }
242 Answer lower(long rest) {
243 int position = 0, visited = 0, n = tree.length - 1;
244 for (int step = Integer.highestOneBit(n); step > 0; step /= 2) {
245 if (position + step <= n) {
246 long value = tree[position + step]; visited++;
247 if (value < rest) { rest -= value; position += step; }
248 }
249 }
250 return new Answer(position < n ? position : -1, visited);
251 }
252 String dump(String name) { return name + " " + joined(tree, 1, tree.length); }
253 }

259 static final class RangeFenwick {
260 final Bit first, second;
261 RangeFenwick(long[] values) {
262 first = new Bit(values.length); second = new Bit(values.length);
263 for (int i = 0; i < values.length; i++) {
264 long difference = values[i] - (i == 0 ? 0 : values[i - 1]);
265 first.add(i + 1, difference); second.add(i + 1, difference * i);
266 }
267 }
268 int add(Range range, long value) {
269 int visited = first.add(range.l + 1, value);
270 visited += first.add(range.r + 2, -value);
271 visited += second.add(range.l + 1, value * range.l);
272 visited += second.add(range.r + 2, -value * (range.r + 1));
273 return visited;
274 }
275 Answer prefix(int position) {
276 Answer a = first.prefix(position), b = second.prefix(position);
277 return new Answer(position * a.value - b.value, a.visited + b.visited);
278 }
279 Answer query(Range range) {
280 Answer a = prefix(range.r + 1), b = prefix(range.l);
281 return new Answer(a.value - b.value, a.visited + b.visited);
282 }

285 static final class Sparse {
286 final long[] values;
287 final List<int[]> minimum = new ArrayList<>(), maximum = new ArrayList<>();
288 int entries;
289 Sparse(long[] values) {
290 this.values = values.clone();
291 int[] positions = new int[values.length];
292 for (int i = 0; i < positions.length; i++) positions[i] = i;
293 minimum.add(positions); maximum.add(positions.clone()); entries = values.length;
294 for (int level = 1; (1 << level) <= values.length; level++) {
295 int length = values.length - (1 << level) + 1, half = 1 << (level - 1);
296 int[] low = new int[length], high = new int[length];
297 for (int i = 0; i < length; i++) {
298 low[i] = select(minimum.get(level - 1)[i], minimum.get(level - 1)[i + half], true);
299 high[i] = select(maximum.get(level - 1)[i], maximum.get(level - 1)[i + half], false);
300 }
301 minimum.add(low); maximum.add(high); entries += length;
302 }
303 }
304 int select(int a, int b, boolean min) { return (min ? values[a] <= values[b] : values[a] >= values[b]) ? a : b; }
305 Report query(Range range) {
306 int level = 31 - Integer.numberOfLeadingZeros(range.length()), end = range.r - (1 << level) + 1;
307 int low = select(minimum.get(level)[range.l], minimum.get(level)[end], true);
308 int high = select(maximum.get(level)[range.l], maximum.get(level)[end], false);
309 return new Report(fmt("min=%d minpos=%d max=%d maxpos=%d level=%d", values[low], low, values[high], high, level), new long[]{values[low], values[high]});
310 }
311 List<String> dump() {
312 var rows = new ArrayList<String>();
313 rows.add(fmt("SPARSE DUMP levels=%d", minimum.size()));
314 for (int level = 0; level < minimum.size(); level++) {
315 rows.add(fmt("MIN %d %s", level, joined(minimum.get(level))));
316 rows.add(fmt("MAX %d %s", level, joined(maximum.get(level))));
317 }
318 return rows;
319 }

325 static final class Sqrt {
326 final int block;
327 final long[] raw, tags, sums;
328 Sqrt(long[] values, int block) {
329 this.block = block; raw = values.clone();
330 tags = new long[(values.length + block - 1) / block]; sums = new long[tags.length];
331 for (int i = 0; i < raw.length; i++) sums[i / block] += raw[i];
332 }
333 Partition split(Range range) {
334 var cells = new ArrayList<Integer>();
335 var blocks = new ArrayList<Integer>();
336 int first = range.l / block, last = range.r / block;
337 for (int i = range.l; i <= range.r; i++) if (i / block == first || i / block == last) cells.add(i);
338 for (int b = first + 1; b < last; b++) blocks.add(b);
339 return new Partition(cells, blocks);
340 }
341 Partition add(Range range, long value) {
342 Partition part = split(range);
343 for (int i : part.cells) { raw[i] += value; sums[i / block] += value; }
344 for (int b : part.blocks) { tags[b] += value; sums[b] += value * block; }
345 return part;
346 }
347 Report query(Range range) {
348 Partition part = split(range); long sum = 0;
349 for (int i : part.cells) sum += raw[i] + tags[i / block];
350 for (int b : part.blocks) sum += sums[b];
351 return new Report(fmt("sum=%d %s", sum, part.text()), new long[]{sum});
352 }
353 List<String> dump() {
354 var rows = new ArrayList<String>();
355 rows.add(fmt("SQRT DUMP block=%d blocks=%d", block, tags.length));
356 for (int b = 0; b < tags.length; b++) {
357 int first = b * block, end = Math.min((b + 1) * block, raw.length);
358 rows.add(fmt("BLOCK %d %d %d add=%d sum=%d %s", b, first, end - 1, tags[b], sums[b], joined(raw, first, end)));
359 }
360 return rows;
361 }

364 record PersistentNode(long value, PersistentNode left, PersistentNode right) {}
365 record ChangedNode(PersistentNode node, int created) {}
366 static PersistentNode buildNode(long[] values, Range range) {
367 if (range.leaf()) return new PersistentNode(values == null ? 0 : values[range.l], null, null);
368 PersistentNode left = buildNode(values, range.left()), right = buildNode(values, range.right());
369 return new PersistentNode(left.value + right.value, left, right);
370 }
371 static ChangedNode changeNode(PersistentNode source, Range range, int index, long value, boolean increment) {
372 if (range.leaf()) return new ChangedNode(new PersistentNode(increment ? source.value + value : value, null, null), 1);
373 PersistentNode left = source.left, right = source.right;
374 ChangedNode changed;
375 if (index <= range.middle()) {
376 changed = changeNode(left, range.left(), index, value, increment); left = changed.node;
377 } else {
378 changed = changeNode(right, range.right(), index, value, increment); right = changed.node;
379 }
380 return new ChangedNode(new PersistentNode(left.value + right.value, left, right), changed.created + 1);
381 }
382 static Answer queryNode(PersistentNode node, Range range, Range wanted) {
383 if (wanted.covers(range)) return new Answer(node.value, 1);
384 Answer answer;
385 if (wanted.r <= range.middle()) answer = queryNode(node.left, range.left(), wanted);
386 else if (wanted.l > range.middle()) answer = queryNode(node.right, range.right(), wanted);
387 else {
388 Answer left = queryNode(node.left, range.left(), wanted), right = queryNode(node.right, range.right(), wanted);
389 answer = new Answer(left.value + right.value, left.visited + right.visited);
390 }
391 return new Answer(answer.value, answer.visited + 1);
392 }
393 static final class Persistent {
394 final int n;
395 final long[] values;
396 final List<PersistentNode> roots = new ArrayList<>(), prefixes = new ArrayList<>();
397 int positionNodes, valueNodes;
398 Persistent(long[] contents) {
399 n = contents.length; values = Arrays.stream(contents).distinct().sorted().toArray();
400 roots.add(buildNode(contents, whole(n))); positionNodes = 2 * n - 1;
401 prefixes.add(buildNode(null, whole(values.length))); valueNodes = 2 * values.length - 1;
402 for (long value : contents) {
403 int rank = Arrays.binarySearch(values, value);
404 ChangedNode changed = changeNode(prefixes.getLast(), whole(values.length), rank, 1, true);
405 prefixes.add(changed.node); valueNodes += changed.created;
406 }
407 }
408 int set(int version, int index, long value) {
409 ChangedNode changed = changeNode(roots.get(version), whole(n), index, value, false);
410 roots.add(changed.node); positionNodes += changed.created;
411 return changed.created;
412 }
413 Report query(int version, Range range) { return queryNode(roots.get(version), whole(n), range).report(); }
414 Answer kth(Range wanted, long k) {
415 PersistentNode u = prefixes.get(wanted.r + 1), w = prefixes.get(wanted.l);
416 Range range = whole(values.length); int visited = 1;
417 while (!range.leaf()) {
418 long count = u.left.value - w.left.value;
419 if (k <= count) { u = u.left; w = w.left; range = range.left(); }
420 else { k -= count; u = u.right; w = w.right; range = range.right(); }
421 visited++;
422 }
423 return new Answer(values[range.l], visited);
424 }
425 }

430 Grid(int height, int width, long[] values) {
431 this.height = height; this.width = width; tree = new long[height + 1][width + 1];
432 for (int i = 0; i < values.length; i++) add(i / width, i % width, values[i]);

434 void cell(int row, int column) {
435 if (row < 0 || row >= height || column < 0 || column >= width) fail("BAD CELL %d %d", row, column);
436 }
437 int add(int row, int column, long value) {
438 int visited = 0;
439 for (int y = row + 1; y <= height; y += y & -y) {
440 for (int x = column + 1; x <= width; x += x & -x) { tree[y][x] += value; visited++; }
441 }
442 return visited;
443 }
444 Answer prefix(int row, int column) {
445 long sum = 0; int visited = 0;
446 for (int y = row; y > 0; y -= y & -y) {
447 for (int x = column; x > 0; x -= x & -x) { sum += tree[y][x]; visited++; }
448 }
449 return new Answer(sum, visited);
450 }
451 Answer query(int row, int column, int bottom, int right) {
452 Answer a = prefix(bottom + 1, right + 1), b = prefix(row, right + 1);
453 Answer c = prefix(bottom + 1, column), d = prefix(row, column);
454 return new Answer(a.value - b.value - c.value + d.value, a.visited + b.visited + c.visited + d.visited);
455 }

460 return rows;
461 }

529 List<String> check(Range range) {
530 Aggregate expected = new Aggregate(0, values[range.l], values[range.l]); long common = 0;
531 for (int i = range.l; i <= range.r; i++) { expected = expected.merge(Aggregate.single(values[i])); common = gcd(common, values[i]); }
532 var rows = new ArrayList<String>();
533 rows.add(fmt("CHECK %d %d %s gcd=%d", range.l, range.r, expected.text(), common));
534 int ok = 0, diff = 0;
535 for (Kind kind : Kind.values()) {
536 if (kind == Kind.GRID || !has(kind)) continue;
537 long[] wanted = switch (kind) {
538 case SEG -> new long[]{switch (seg.op) { case SUM -> expected.sum; case MIN -> expected.min; case MAX -> expected.max; case GCD -> common; }};
539 case LAZY -> new long[]{expected.sum, expected.min, expected.max};
540 case SPARSE -> new long[]{expected.min, expected.max};
541 default -> new long[]{expected.sum};
542 };
543 int version = kind == Kind.PERSIST ? persist.roots.size() - 1 : 0;
544 Report answer = query(kind, version, range);
545 boolean equal = Arrays.equals(answer.values, wanted);
546 if (equal) ok++; else diff++;
547 rows.add(fmt("%s %s %s%s", kind, equal ? "OK" : "DIFF", kind == Kind.PERSIST ? fmt("version=%d ", version) : "", answer.text));
548 }
549 rows.add(fmt("RESULT ok=%d diff=%d", ok, diff));
550 return rows;
551 }
552 List<String> mo(List<Range> queries) {
553 int block = isqrt(values.length);
554 var order = new ArrayList<Integer>();
555 for (int i = 0; i < queries.size(); i++) order.add(i);
556 order.sort(Comparator.comparingInt((Integer i) -> queries.get(i).l / block)
557 .thenComparingInt(i -> (queries.get(i).l / block % 2 == 0 ? 1 : -1) * queries.get(i).r).thenComparingInt(i -> i));
558 long[] unique = Arrays.stream(values).distinct().sorted().toArray();
559 int[] ranks = new int[values.length], frequencies = new int[unique.length], answers = new int[queries.size()];
560 for (int i = 0; i < values.length; i++) ranks[i] = Arrays.binarySearch(unique, values[i]);
561 int left = 0, right = 0, moves = 0, distinct = 0;
562 for (int index : order) {
563 Range range = queries.get(index); int end = range.r + 1;
564 moves += Math.abs(range.l - left) + Math.abs(end - right);
565 while (left > range.l) if (frequencies[ranks[--left]]++ == 0) distinct++;
566 while (right < end) if (frequencies[ranks[right++]]++ == 0) distinct++;
567 while (left < range.l) if (--frequencies[ranks[left++]] == 0) distinct--;
568 while (right > end) if (--frequencies[ranks[--right]] == 0) distinct--;
569 answers[index] = distinct;
570 }
571 var rows = new ArrayList<String>();
572 rows.add(fmt("MO queries=%d block=%d moves=%d", queries.size(), block, moves));
573 rows.add("ORDER " + joined(order.stream().mapToInt(i -> i + 1).toArray()));
574 for (int i = 0; i < queries.size(); i++) rows.add(fmt("ANSWER %d %d %d distinct=%d", i + 1, queries.get(i).l, queries.get(i).r, answers[i]));
575 return rows;
576 }

578 List<String> execute(String line) {
579 String[] words = line.split(" ");
580 switch (words[0]) {
581 case "LOAD": {
582 int removed = 0;
583 for (Kind kind : Kind.values()) if (kind != Kind.GRID && has(kind)) removed++;
584 values = numbers(words, 1); original = values.clone();
585 seg = null; lazy = null; fen = null; rfen = null; sparse = null; sqrt = null; persist = null;
586 return List.of(fmt("LOAD n=%d removed=%d", values.length, removed));
587 }
588 case "RESET": {
589 values = original.clone();
590 Operation op = seg == null ? Operation.SUM : seg.op;
591 int block = sqrt == null ? 0 : sqrt.block, rebuilt = 0;
592 for (Kind kind : Kind.values()) if (kind != Kind.GRID && has(kind)) { build(kind, op, block); rebuilt++; }
593 return List.of(fmt("RESET rebuilt=%d", rebuilt));
594 }
595 case "ARRAY", "CHECK": {
596 Range range = parsedRange(words, 1); range(range);
597 return words[0].equals("CHECK") ? check(range) : List.of(fmt("ARRAY %d %d %s", range.l, range.r, joined(values, range.l, range.r + 1)));
598 }
599 case "MO": {
600 var queries = new ArrayList<Range>();
601 for (int i = 1; i < words.length; i += 2) { Range range = parsedRange(words, i); range(range); queries.add(range); }
602 return mo(queries);
603 }
604 }
605 Kind kind = Kind.valueOf(words[0]); String action = words[1];
606 if (action.equals("BUILD")) {
607 Operation op = Operation.SUM; int block = 0;
608 if (kind == Kind.SEG) {
609 try { op = Operation.valueOf(words[2]); }
610 catch (IllegalArgumentException error) { throw new CommandError("BAD OP " + words[2]); }
611 }
612 if (kind == Kind.SQRT) {
613 block = words.length == 3 ? integer(words, 2) : isqrt(values.length);
614 if (block > values.length) fail("BAD BLOCK %d", block);
615 }
616 if (kind == Kind.GRID) {
617 int height = integer(words, 2), width = integer(words, 3), count = words.length - 4;
618 if (count != height * width) fail("BAD COUNT %d %d", count, height * width);
619 long[] contents = numbers(words, 4);
620 grid = new Grid(height, width, contents);
621 return List.of(fmt("GRID BUILD rows=%d cols=%d total=%d", height, width, total(contents)));
622 }
623 return List.of(build(kind, op, block));
624 }
625 if (!has(kind)) fail("NOT BUILT %s", kind);
626 if (action.equals("DUMP")) return dump(kind);
627 if (kind == Kind.GRID) {
628 int row = integer(words, 2), column = integer(words, 3); grid.cell(row, column);
629 if (action.equals("ADD")) {
630 long value = number(words, 4); int visited = grid.add(row, column, value);
631 return List.of(fmt("GRID ADD %d %d %d visited=%d", row, column, value, visited));
632 }
633 int bottom = integer(words, 4), right = integer(words, 5); grid.cell(bottom, right);
634 if (row > bottom || column > right) fail("BAD RECT %d %d %d %d", row, column, bottom, right);
635 return List.of(fmt("GRID SUM %d %d %d %d %s", row, column, bottom, right, grid.query(row, column, bottom, right).report().text));
636 }
637 if (kind == Kind.PERSIST) {
638 if (action.equals("NODES")) return List.of(fmt("PERSIST NODES versions=%d position=%d value=%d", persist.roots.size(), persist.positionNodes, persist.valueNodes));
639 if (action.equals("KTH")) {
640 Range range = parsedRange(words, 2); range(range); long k = number(words, 4);
641 if (k < 1 || k > range.length()) fail("BAD K %d", k);
642 Answer answer = persist.kth(range, k);
643 return List.of(fmt("PERSIST KTH %d %d %d value=%d visited=%d", range.l, range.r, k, answer.value, answer.visited));
644 }
645 int version = integer(words, 2); version(version);
646 int index = integer(words, 3);
647 if (action.equals("SET")) {
648 index(index); long value = number(words, 4); int next = persist.roots.size();
649 int created = persist.set(version, index, value);
650 return List.of(fmt("PERSIST SET %d %d %d version=%d created=%d", version, index, value, next, created));
651 }
652 Range range = new Range(index, integer(words, 4)); range(range);
653 return List.of(fmt("PERSIST QUERY %d %d %d %s", version, range.l, range.r, persist.query(version, range).text));
654 }
655 if (kind == Kind.FEN && action.equals("LOWER")) {
656 for (int i = 0; i < fen.values.length; i++) if (fen.values[i] < 0) fail("NEGATIVE %d", i);
657 long k = number(words, 2); Answer answer = fen.bit.lower(k);
658 return List.of(fmt("FEN LOWER %d index=%s visited=%d", k, answer.value < 0 ? "none" : Long.toString(answer.value), answer.visited));
659 }
660 if (kind == Kind.SEG && action.equals("UPDATE")) {
661 int index = integer(words, 2); index(index); long value = number(words, 3);
662 int visited = seg.update(1, index, value); values[index] = value;
663 return List.of(fmt("SEG UPDATE %d %d root=%d visited=%d", index, value, seg.tree[1], visited));
664 }
665 if (kind == Kind.FEN && (action.equals("ADD") || action.equals("PREFIX"))) {
666 int index = integer(words, 2); index(index);
667 if (action.equals("PREFIX")) return List.of(fmt("FEN PREFIX %d %s", index, fen.bit.prefix(index + 1).report().text));
668 long value = number(words, 3); values[index] += value; fen.values[index] += value;
669 return List.of(fmt("FEN ADD %d %d visited=%d", index, value, fen.bit.add(index + 1, value)));
670 }
671 Range range = parsedRange(words, 2); range(range);
672 if (action.equals("ADD") || action.equals("ASSIGN")) {
673 long value = number(words, 4); boolean assign = action.equals("ASSIGN");
674 String details = switch (kind) {
675 case LAZY -> {
676 Counts counts = lazy.update(range, value, assign);
677 yield fmt("visited=%d covered=%d pushes=%d", counts.visited, counts.covered, counts.pushes);
678 }
679 case RFEN -> fmt("visited=%d", rfen.add(range, value));
680 case SQRT -> sqrt.add(range, value).text();
681 default -> throw new IllegalStateException();
682 };
683 for (int i = range.l; i <= range.r; i++) values[i] = assign ? value : values[i] + value;
684 return List.of(fmt("%s %s %d %d %d %s", kind, action, range.l, range.r, value, details));
685 }
686 return List.of(fmt("%s %s %d %d %s", kind, action, range.l, range.r, query(kind, 0, range).text));
687 }
688 }

690 static long number(String[] words, int index) { return Long.parseLong(words[index]); }
691 static int integer(String[] words, int index) { return Integer.parseInt(words[index]); }
692 static Range parsedRange(String[] words, int index) { return new Range(integer(words, index), integer(words, index + 1)); }
693 static long[] numbers(String[] words, int from) {
694 long[] values = new long[words.length - from];
695 for (int i = 0; i < values.length; i++) values[i] = number(words, i + from);
696 return values;
697 }
698 public static void main(String[] args) throws Exception {
699 var input = new BufferedReader(new InputStreamReader(System.in));

707 output.flush();
708 }
