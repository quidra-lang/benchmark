    var middle: Int {
        (left + right) / 2
    }
    var leaf: Bool {
        left == right
    }
    var length: Int {
        right - left + 1
    }
    func covers(_ other: Span) -> Bool {
        left <= other.left && other.right <= right
    }
    func merged(with other: Aggregate) -> Aggregate {
        Aggregate(sum: sum + other.sum, minimum: min(minimum, other.minimum), maximum: max(maximum, other.maximum))
    }
struct Counts { var visited = 0, covered = 0, pushes = 0 }
    private func apply(_ index: Int, value: Int, assignment: Bool) {
        var node = nodes[index]!
        if assignment {
            node.value = Aggregate(sum: value * node.span.length, minimum: value, maximum: value)
        } else {
            node.value = Aggregate(
                sum: node.value.sum + value * node.span.length,
                minimum: node.value.minimum + value,
                maximum: node.value.maximum + value
            )
        }
        if !node.span.leaf {
            if assignment {
                node.assignment = value
                node.addition = 0
            } else {
                node.addition += value
            }
        }
        nodes[index] = node
    }
    private func push(_ index: Int) -> Int {
        let node = nodes[index]!
        if node.assignment == nil && node.addition == 0 {
            return 0
        }
        for child in [2 * index, 2 * index + 1] {
            if let value = node.assignment {
                apply(child, value: value, assignment: true)
            }
            if node.addition != 0 {
                apply(child, value: node.addition, assignment: false)
            }
        }
        nodes[index]!.assignment = nil
        nodes[index]!.addition = 0
        return 1
    }
    func update(_ range: Span, value: Int, assignment: Bool) -> Counts {
        var counts = Counts()
        func visit(_ index: Int) {
            counts.visited += 1
            let span = nodes[index]!.span
            if range.covers(span) {
                counts.covered += 1
                apply(index, value: value, assignment: assignment)
                return
            }
    func query(_ range: Span) -> Report {
        var counts = Counts()
        func visit(_ index: Int) -> Aggregate {
            counts.visited += 1
            let span = nodes[index]!.span
            if range.covers(span) {
                return nodes[index]!.value
            }
            counts.pushes += push(index)
            if range.right <= span.middle {
                return visit(2 * index)
            }
            if range.left > span.middle {
                return visit(2 * index + 1)
            }
            let left = visit(2 * index), right = visit(2 * index + 1)
            return left.merged(with: right)
        }
        let value = visit(1)
    }
