9enum Operation: String {
10 case sum = "SUM", min = "MIN", max = "MAX", gcd = "GCD"
11 func combine(_ a: Int, _ b: Int) -> Int {
12 switch self {
13 case .sum:
14 return a + b
15 case .min:
16 return Swift.min(a, b)
17 case .max:
18 return Swift.max(a, b)
19 case .gcd:
20 return greatestCommonDivisor(a, b)
21 }
22 }
23}
24struct Span {
25 let left: Int, right: Int
26 var middle: Int {
27 (left + right) / 2
28 }
29 var leaf: Bool {
30 left == right
31 }
32 var length: Int {
33 right - left + 1
34 }
35 var leftChild: Span {
36 Span(left: left, right: middle)
37 }
38 var rightChild: Span {
39 Span(left: middle + 1, right: right)
40 }
41 func covers(_ other: Span) -> Bool {
42 left <= other.left && other.right <= right
43 }
44}
45struct Aggregate {
46 let sum: Int, minimum: Int, maximum: Int
47 var text: String {
48 "sum=\(sum) min=\(minimum) max=\(maximum)"
49 }
50 var values: [Int] {

52 }
53 func merged(with other: Aggregate) -> Aggregate {
54 Aggregate(sum: sum + other.sum, minimum: min(minimum, other.minimum), maximum: max(maximum, other.maximum))
55 }
56}

76func greatestCommonDivisor(_ a: Int, _ b: Int) -> Int {
77 var a = abs(a), b = abs(b)
78 while b != 0 {
79 (a, b) = (b, a % b)
80 }
81 return a
82}
83func lowbit(_ value: Int) -> Int {

85}
86func floorLog(_ value: Int) -> Int {
87 Int.bitWidth - 1 - value.leadingZeroBitCount
88}
89func integerSqrt(_ value: Int) -> Int {
90 var root = 1
91 while (root + 1) * (root + 1) <= value {
92 root += 1
93 }
94 return root
95}

103final class SegmentTree {
104 struct Node { let span: Span
105 var value: Int
106 }
107 let operation: Operation, count: Int
108 private(set) var nodes: [Node?]
109 private var maxNode = 0
110 init(_ values: [Int], operation: Operation) {
111 self.operation = operation
112 count = values.count
113 nodes = Array(repeating: nil, count: 4 * count)
114 build(values, index: 1, span: fullSpan(count))
115 }
116 private func build(_ values: [Int], index: Int, span: Span) {
117 maxNode = max(maxNode, index)
118 let value: Int
119 if span.leaf {
120 value = operation == .gcd ? abs(values[span.left]) : values[span.left]
121 } else {
122 build(values, index: 2 * index, span: span.leftChild)
123 build(values, index: 2 * index + 1, span: span.rightChild)
124 value = operation.combine(nodes[2 * index]!.value, nodes[2 * index + 1]!.value)
125 }
126 nodes[index] = Node(span: span, value: value)
127 }
128 func query(_ range: Span, index: Int = 1) -> Answer {
129 let node = nodes[index]!
130 if range.covers(node.span) {
131 return Answer(value: node.value, visited: 1)
132 }
133 let answer: Answer
134 if range.right <= node.span.middle {
135 answer = query(range, index: 2 * index)
136 } else if range.left > node.span.middle {
137 answer = query(range, index: 2 * index + 1)
138 } else {
139 let left = query(range, index: 2 * index), right = query(range, index: 2 * index + 1)
140 answer = Answer(value: operation.combine(left.value, right.value), visited: left.visited + right.visited)
141 }
142 return Answer(value: answer.value, visited: answer.visited + 1)
143 }
144 func update(_ position: Int, value: Int, index: Int = 1) -> Int {
145 let span = nodes[index]!.span
146 if span.leaf {
147 nodes[index]!.value = operation == .gcd ? abs(value) : value
148 return 1
149 }
150 let visited = update(position, value: value, index: position <= span.middle ? 2 * index : 2 * index + 1)
151 nodes[index]!.value = operation.combine(nodes[2 * index]!.value, nodes[2 * index + 1]!.value)
152 return visited + 1
153 }
154 var root: Int {
155 nodes[1]!.value
156 }
157 var buildLine: String {
158 "SEG BUILD \(operation.rawValue) nodes=\(2 * count - 1) maxnode=\(maxNode) root=\(root)"
159 }
160 func dump() -> [String] {
161 var rows = ["SEG DUMP \(operation.rawValue) nodes=\(2 * count - 1)"]
162 for (index, entry) in nodes.enumerated() {
163 if let node = entry {
164 rows.append("NODE \(index) \(node.span.left) \(node.span.right) \(node.value)")
165 }
166 }
167 return rows
168 }
169}
170final class LazyTree {
171 struct Node { let span: Span
172 var value: Aggregate
173 var assignment: Int? = nil
174 var addition = 0
175 }
176 let count: Int
177 private var nodes: [Node?]
178 init(_ values: [Int]) {
179 count = values.count
180 nodes = Array(repeating: nil, count: 4 * count)
181 build(values, index: 1, span: fullSpan(count))
182 }
183 private func build(_ values: [Int], index: Int, span: Span) {
184 let value: Aggregate
185 if span
186 .leaf
187 {
188 value = Aggregate(sum: values[span.left], minimum: values[span.left], maximum: values[span.left])
189 } else {
190 build(values, index: 2 * index, span: span.leftChild)
191 build(values, index: 2 * index + 1, span: span.rightChild)
192 value = nodes[2 * index]!.value.merged(with: nodes[2 * index + 1]!.value)
193 }
194 nodes[index] = Node(span: span, value: value)
195 }
196 private func pull(_ index: Int) {
197 nodes[index]!.value = nodes[2 * index]!.value.merged(with: nodes[2 * index + 1]!.value)
198 }
199 private func apply(_ index: Int, value: Int, assignment: Bool) {
200 var node = nodes[index]!
201 if assignment {
202 node.value = Aggregate(sum: value * node.span.length, minimum: value, maximum: value)
203 } else {
204 node.value = Aggregate(
205 sum: node.value.sum + value * node.span.length,
206 minimum: node.value.minimum + value,
207 maximum: node.value.maximum + value
208 )
209 }
210 if !node.span.leaf {
211 if assignment {
212 node.assignment = value
213 node.addition = 0
214 } else {
215 node.addition += value
216 }
217 }
218 nodes[index] = node
219 }
220 private func push(_ index: Int) -> Int {
221 let node = nodes[index]!
222 if node.assignment == nil && node.addition == 0 {
223 return 0
224 }
225 for child in [2 * index, 2 * index + 1] {
226 if let value = node.assignment {
227 apply(child, value: value, assignment: true)
228 }
229 if node.addition != 0 {
230 apply(child, value: node.addition, assignment: false)
231 }
232 }
233 nodes[index]!.assignment = nil
234 nodes[index]!.addition = 0
235 return 1
236 }
237 func update(_ range: Span, value: Int, assignment: Bool) -> Counts {
238 var counts = Counts()
239 func visit(_ index: Int) {
240 counts.visited += 1
241 let span = nodes[index]!.span
242 if range.covers(span) {
243 counts.covered += 1
244 apply(index, value: value, assignment: assignment)
245 return
246 }
247 counts.pushes += push(index)
248 if range.left <= span.middle {
249 visit(2 * index)
250 }
251 if range.right > span.middle {
252 visit(2 * index + 1)
253 }
254 pull(index)
255 }
256 visit(1)
257 return counts
258 }
259 func query(_ range: Span) -> Report {
260 var counts = Counts()
261 func visit(_ index: Int) -> Aggregate {
262 counts.visited += 1
263 let span = nodes[index]!.span
264 if range.covers(span) {
265 return nodes[index]!.value
266 }
267 counts.pushes += push(index)
268 if range.right <= span.middle {
269 return visit(2 * index)
270 }
271 if range.left > span.middle {
272 return visit(2 * index + 1)
273 }
274 let left = visit(2 * index), right = visit(2 * index + 1)
275 return left.merged(with: right)
276 }
277 let value = visit(1)
278 return Report(text: "\(value.text) visited=\(counts.visited) pushes=\(counts.pushes)", values: value.values)
279 }
280 var buildLine: String {
281 "LAZY BUILD nodes=\(2 * count - 1) \(nodes[1]!.value.text)"
282 }
283 func dump() -> [String] {
284 var rows = ["LAZY DUMP nodes=\(2 * count - 1)"]
285 for (index, entry) in nodes.enumerated() {
286 if let node = entry {
287 rows
288 .append(
289 "NODE \(index) \(node.span.left) \(node.span.right) \(node.value.text) assign=\(node.assignment.map(String.init) ?? "-") add=\(node.addition)"
290 )
291 }
292 }
293 return rows
294 }
295}

297final class Bit {
298 private(set) var tree: [Int]
299 init(_ values: [Int]) {
300 tree = [0] + values
301 for position in 1..<tree.count {
302 let parent = position + lowbit(position)
303 if parent < tree.count {
304 tree[parent] += tree[position]
305 }
306 }
307 }
308 func add(_ position: Int, value: Int) -> Int {
309 var position = position, visited = 0
310 while position < tree.count {
311 tree[position] += value
312 visited += 1
313 position += lowbit(position)
314 }
315 return visited
316 }
317 func prefix(_ position: Int) -> Answer {
318 var position = position, sum = 0, visited = 0
319 while position > 0 {
320 sum += tree[position]
321 visited += 1
322 position -= lowbit(position)
323 }
324 return Answer(value: sum, visited: visited)
325 }
326 func query(_ range: Span) -> Answer {
327 let right = prefix(range.right + 1), left = prefix(range.left)
328 return Answer(value: right.value - left.value, visited: right.visited + left.visited)
329 }
330 func lower(_ target: Int) -> Answer {
331 let count = tree.count - 1
332 var step = 1 << floorLog(count), position = 0, rest = target, visited = 0
333 while step > 0 {
334 if position + step <= count {
335 visited += 1
336 if tree[position + step] < rest {
337 rest -= tree[position + step]
338 position += step
339 }
340 }
341 step /= 2
342 }
343 return Answer(value: position < count ? position : -1, visited: visited)
344 }
345 var text: String {
346 words(tree.dropFirst())
347 }
348}

367final class RangeFenwick {
368 let first: Bit, second: Bit
369 init(_ values: [Int]) {
370 let differences = values.enumerated().map { index, value in value - (index == 0 ? 0 : values[index - 1]) }
371 first = Bit(differences)

379 return visited
380 }

396final class SparseTable {
397 let values: [Int]
398 private var minima: [[Int]], maxima: [[Int]]
399 init(_ values: [Int]) {
400 self.values = values
401 minima = [Array(values.indices)]
402 maxima = [Array(values.indices)]
403 var width = 2
404 while width <= values.count {
405 var lows: [Int] = [], highs: [Int] = []
406 for index in 0...(values.count - width) {
407 let a = minima.last![index], b = minima.last![index + width / 2]
408 let c = maxima.last![index], d = maxima.last![index + width / 2]
409 lows.append(values[a] <= values[b] ? a : b)
410 highs.append(values[c] >= values[d] ? c : d)
411 }
412 minima.append(lows)
413 maxima.append(highs)
414 width *= 2
415 }
416 }
417 func query(_ range: Span) -> Report {
418 let level = floorLog(range.length), end = range.right - (1 << level) + 1
419 let a = minima[level][range.left], b = minima[level][end], c = maxima[level][range.left], d = maxima[level][end]
420 let low = values[a] <= values[b] ? a : b, high = values[c] >= values[d] ? c : d
421 return Report(
422 text: "min=\(values[low]) minpos=\(low) max=\(values[high]) maxpos=\(high) level=\(level)",
423 values: [values[low], values[high]]
424 )
425 }
426 var buildLine: String {
427 "SPARSE BUILD levels=\(minima.count) entries=\(minima.reduce(0) { $0 + $1.count })"
428 }
429 func dump() -> [String] {
430 var rows = ["SPARSE DUMP levels=\(minima.count)"]
431 for level in minima.indices {
432 rows.append("MIN \(level) \(words(minima[level]))")
433 rows.append("MAX \(level) \(words(maxima[level]))")
434 }
435 return rows
436 }

438final class SquareRoot {
439 let block: Int
440 private var raw: [Int], tags: [Int], sums: [Int]
441 init(_ values: [Int], block: Int) {
442 self.block = block
443 raw = values
444 tags = Array(repeating: 0, count: (values.count + block - 1) / block)
445 sums = tags
446 for index in values.indices {
447 sums[index / block] += values[index]
448 }
449 }
450 private func split(_ range: Span) -> (cells: [Int], whole: [Int]) {
451 let first = range.left / block, last = range.right / block
452 if first == last {
453 return (Array(range.left...range.right), [])
454 }
455 return (
456 Array(range.left..<((first + 1) * block)) + Array((last * block)...range.right),
457 Array((first + 1)..<last)
458 )
459 }
460 func add(_ range: Span, value: Int) -> BlockAnswer {
461 let parts = split(range)
462 for index in parts.cells {
463 raw[index] += value
464 sums[index / block] += value
465 }
466 for index in parts.whole {
467 tags[index] += value
468 sums[index] += value * block
469 }
470 return BlockAnswer(sum: 0, whole: parts.whole.count, cells: parts.cells.count)
471 }
472 func query(_ range: Span) -> BlockAnswer {
473 let parts = split(range)
474 let total = parts.cells.reduce(0) { $0 + raw[$1] + tags[$1 / block] } + parts.whole.reduce(0) { $0 + sums[$1] }
475 return BlockAnswer(sum: total, whole: parts.whole.count, cells: parts.cells.count)
476 }
477 var buildLine: String {
478 "SQRT BUILD block=\(block) blocks=\(tags.count)"
479 }
480 func dump() -> [String] {
481 var rows = ["SQRT DUMP block=\(block) blocks=\(tags.count)"]
482 for index in tags.indices {
483 let first = index * block, end = min(first + block, raw.count)
484 rows
485 .append(
486 "BLOCK \(index) \(first) \(end - 1) add=\(tags[index]) sum=\(sums[index]) \(words(raw[first..<end]))"
487 )
488 }
489 return rows
490 }

502final class PersistentTree {
503 let count: Int
504 private(set) var created = 0
505 private(set) var roots: [PersistentNode] = []
506 init(_ values: [Int]) {
507 count = values.count
508 roots.append(build(values, span: fullSpan(count)))
509 }
510 private func node(_ sum: Int, left: PersistentNode? = nil, right: PersistentNode? = nil) -> PersistentNode {
511 created += 1
512 return PersistentNode(sum, left: left, right: right)
513 }
514 private func build(_ values: [Int], span: Span) -> PersistentNode {
515 if span.leaf {
516 return node(values[span.left])
517 }
518 let left = build(values, span: span.leftChild), right = build(values, span: span.rightChild)
519 return node(left.sum + right.sum, left: left, right: right)
520 }
521 private func copyPath(_ previous: PersistentNode, span: Span, index: Int, value: Int,
522 addition: Bool) -> PersistentNode
523 {
524 if span.leaf {
525 return node(addition ? previous.sum + value : value)
526 }
527 var left = previous.left!, right = previous.right!
528 if index <= span.middle {
529 left = copyPath(
530 left,
531 span: span.leftChild,
532 index: index,
533 value: value,
534 addition: addition
535 )
536 } else {
537 right = copyPath(right, span: span.rightChild, index: index, value: value, addition: addition)
538 }
539 return node(left.sum + right.sum, left: left, right: right)
540 }
541 func fork(_ version: Int, index: Int, value: Int, addition: Bool = false) -> Int {
542 let before = created
543 let root = copyPath(roots[version], span: fullSpan(count), index: index, value: value, addition: addition)
544 roots.append(root)
545 return created - before
546 }
547 func query(_ version: Int, range: Span) -> Answer {
548 func visit(_ node: PersistentNode, span: Span) -> Answer {
549 if range.covers(span) {
550 return Answer(value: node.sum, visited: 1)
551 }
552 let answer: Answer
553 if range.right <= span.middle {
554 answer = visit(node.left!, span: span.leftChild)
555 } else if range.left > span.middle {
556 answer = visit(node.right!, span: span.rightChild)
557 } else {
558 let left = visit(node.left!, span: span.leftChild), right = visit(node.right!, span: span.rightChild)
559 answer = Answer(value: left.value + right.value, visited: left.visited + right.visited)
560 }
561 return Answer(value: answer.value, visited: answer.visited + 1)
562 }
563 return visit(roots[version], span: fullSpan(count))
564 }
565}
566final class Persistence {
567 let positions: PersistentTree
568 private let prefixes: PersistentTree, domain: [Int]
569 init(_ values: [Int]) {
570 positions = PersistentTree(values)
571 domain = Array(Set(values)).sorted()
572 prefixes = PersistentTree(Array(repeating: 0, count: domain.count))
573 let ranks = Dictionary(uniqueKeysWithValues: domain.enumerated().map { ($0.element, $0.offset) })
574 for (index, value) in values.enumerated() {
575 _ = prefixes.fork(
576 index,
577 index: ranks[value]!,
578 value: 1,
579 addition: true
580 )
581 }
582 }
583 func kth(_ range: Span, k: Int) -> Answer {
584 var upper = prefixes.roots[range.right + 1], lower = prefixes.roots[range.left]
585 var left = 0, right = domain.count - 1, k = k, visited = 1
586 while left < right {
587 let middle = (left + right) / 2, count = upper.left!.sum - lower.left!.sum
588 if k <= count {
589 upper = upper.left!
590 lower = lower.left!
591 right = middle
592 } else {
593 k -= count
594 upper = upper.right!
595 lower = lower.right!
596 left = middle + 1
597 }
598 visited += 1
599 }
600 return Answer(value: domain[left], visited: visited)
601 }
602 var buildLine: String {
603 "PERSIST BUILD nodes=\(positions.created) distinct=\(domain.count) valuenodes=\(prefixes.created)"
604 }
605 var nodeLine: String {
606 "PERSIST NODES versions=\(positions.roots.count) position=\(positions.created) value=\(prefixes.created)"
607 }
608}
609final class Grid {
610 let height: Int, width: Int
611 private var tree: [[Int]]
612 init(height: Int, width: Int, values: [Int]) {
613 self.height = height
614 self.width = width
615 tree = Array(repeating: Array(repeating: 0, count: width + 1), count: height + 1)
616 for (index, value) in values.enumerated() {
617 _ = add(index / width, column: index % width, value: value)
618 }
619 }
620 func check(_ row: Int, _ column: Int) throws {
621 if row < 0 || row >= height || column < 0 || column >=
622 width
623 {
624 throw CommandError.invalid("BAD CELL \(row) \(column)")
625 }
626 }
627 func add(_ row: Int, column: Int, value: Int) -> Int {
628 var y = row + 1, visited = 0
629 while y <= height {
630 var x = column + 1
631 while x <= width {
632 tree[y][x] += value
633 visited += 1
634 x += lowbit(x)
635 }
636 y += lowbit(y)
637 }
638 return visited
639 }
640 private func prefix(_ row: Int, _ column: Int) -> Answer {
641 var y = row, sum = 0, visited = 0
642 while y > 0 {
643 var x = column
644 while x > 0 {
645 sum += tree[y][x]
646 visited += 1
647 x -= lowbit(x)
648 }
649 y -= lowbit(y)
650 }
651 return Answer(value: sum, visited: visited)
652 }
653 func query(_ row: Int, _ column: Int, _ bottom: Int, _ right: Int) -> Answer {

655 row,
656 column
657 )
658 return Answer(
659 value: a.value - b.value - c.value + d.value,
660 visited: a.visited + b.visited + c.visited + d.visited
661 )
662 }

664 ["GRID DUMP rows=\(height) cols=\(width)"] + (1...height).map { "ROW \($0) \(words(tree[$0].dropFirst()))" }
665 }

667func mo(_ values: [Int], ranges: [Span]) -> [String] {
668 let block = integerSqrt(values.count)
669 let order = ranges.indices.sorted { a, b in
670 let first = ranges[a].left / block, second = ranges[b].left / block
671 if first != second {
672 return first < second
673 }
674 if ranges[a].right != ranges[b]
675 .right
676 {
677 return first % 2 == 0 ? ranges[a].right < ranges[b].right : ranges[a].right > ranges[b].right
678 }
679 return a < b
680 }
681 var frequency: [Int: Int] = [:], answers = Array(repeating: 0, count: ranges.count)
682 var left = 0, right = -1, moves = 0
683 func add(_ index: Int) {
684 frequency[values[index], default: 0] += 1
685 moves += 1
686 }
687 func remove(_ index: Int) {
688 let value = values[index], count = frequency[value]! - 1
689 if count == 0 {
690 frequency.removeValue(forKey: value)
691 } else {
692 frequency[value] = count
693 }
694 moves += 1
695 }
696 for index in order {
697 let range = ranges[index]
698 while left > range.left {
699 left -= 1
700 add(left)
701 }
702 while right < range.right {
703 right += 1
704 add(right)
705 }
706 while left < range.left {
707 remove(left)
708 left += 1
709 }
710 while right > range.right {
711 remove(right)
712 right -= 1
713 }
714 answers[index] = frequency.count
715 }
716 return ["MO queries=\(ranges.count) block=\(block) moves=\(moves)", "ORDER \(words(order.map { $0 + 1 }))"]
717 + ranges.indices.map { "ANSWER \($0 + 1) \(ranges[$0].left) \(ranges[$0].right) distinct=\(answers[$0])" }
718}

720enum Structure {
721 case seg(SegmentTree), lazy(LazyTree), fen(FenwickArray), rfen(RangeFenwick)
722 case sparse(SparseTable), sqrt(SquareRoot), persist(Persistence)
723 var buildLine: String {
724 switch self {
725 case .seg(let tree):
726 return tree.buildLine
727 case .lazy(let tree):
728 return tree.buildLine
729 case .fen(let tree):
730 return tree.buildLine
731 case .rfen(let tree):
732 return tree.buildLine
733 case .sparse(let tree):
734 return tree.buildLine
735 case .sqrt(let tree):
736 return tree.buildLine
737 case .persist(let tree):
738 return tree.buildLine
739 }
740 }
741 func query(_ span: Span, version: Int? = nil) -> Report {
742 switch self {
743 case .seg(let tree):
744 let answer = tree.query(span)
745 return Report(
746 text: "\(tree.operation.rawValue.lowercased())=\(answer.value) visited=\(answer.visited)",
747 values: [answer.value]
748 )
749 case .lazy(let tree):
750 return tree.query(span)
751 case .fen(let tree):
752 return tree.bit.query(span).report
753 case .rfen(let tree):
754 return tree.query(span).report
755 case .sparse(let tree):
756 return tree.query(span)
757 case .sqrt(let tree):
758 let answer = tree.query(span)
759 return Report(text: "sum=\(answer.sum) \(answer.partition)", values: [answer.sum])
760 case .persist(let tree):
761 return tree.positions.query(version ?? tree.positions.roots.count - 1, range: span)
762 .report
763 }
764 }
765 func dump() -> [String] {
766 switch self {
767 case .seg(let tree):
768 return tree.dump()
769 case .lazy(let tree):
770 return tree.dump()
771 case .fen(let tree):
772 return tree.dump()
773 case .rfen(let tree):
774 return tree.dump()
775 case .sparse(let tree):
776 return tree.dump()
777 case .sqrt(let tree):
778 return tree.dump()
779 case .persist:
780 preconditionFailure("PERSIST has no DUMP command")
781 }
782 }

784final class Laboratory {
785 private var values: [Int], original: [Int]
786 private var structures: [Kind: Structure] = [:]
787 private var grid: Grid?
788 init(_ values: [Int]) {
789 self.values = values
790 original = values
791 }
792 private func checkIndex(_ index: Int) throws {
793 if index < 0 || index >= values.count {
794 throw CommandError.invalid("BAD INDEX \(index)")
795 }
796 }
797 private func checkedRange(_ left: Int, _ right: Int) throws -> Span {
798 try checkIndex(left)
799 try checkIndex(right)
800 if left > right {
801 throw CommandError.invalid("BAD RANGE \(left) \(right)")
802 }
803 return Span(left: left, right: right)
804 }
805 private func checkVersion(_ version: Int, tree: Persistence) throws {
806 if version < 0 || version >= tree.positions.roots.count {
807 throw CommandError.invalid("BAD VERSION \(version)")
808 }
809 }
810 private func create(_ kind: Kind, operation: Operation = .sum, block: Int = 1) -> Structure {
811 switch kind {
812 case .seg:
813 return .seg(SegmentTree(values, operation: operation))
814 case .lazy:
815 return .lazy(LazyTree(values))
816 case .fen:
817 return .fen(FenwickArray(values))
818 case .rfen:
819 return .rfen(RangeFenwick(values))
820 case .sparse:
821 return .sparse(SparseTable(values))
822 case .sqrt:
823 return .sqrt(SquareRoot(values, block: block))
824 case .persist:
825 return .persist(Persistence(values))
826 }
827 }
828 private func change(_ range: Span, value: Int, assignment: Bool = false) {
829 for index in range.left...range.right {
830 values[index] = assignment ? value : values[index] + value
831 }
832 }
833 private func check(_ range: Span) -> [String] {
834 let cells = values[range.left...range.right]
835 let direct = Aggregate(sum: cells.reduce(0, +), minimum: cells.min()!, maximum: cells.max()!)
836 let common = cells.reduce(0, greatestCommonDivisor)
837 var rows = ["CHECK \(range.left) \(range.right) \(direct.text) gcd=\(common)"], ok = 0, different = 0
838 for kind in Kind.allCases {
839 guard let structure = structures[kind] else {
840 continue
841 }
842 var expected = [direct.sum], prefix = ""
843 switch structure {
844 case .seg(let tree):
845 switch tree.operation {
846 case .sum:
847 expected = [direct.sum]
848 case .min:
849 expected = [direct.minimum]
850 case .max:
851 expected = [direct.maximum]
852 case .gcd:
853 expected = [common]
854 }
855 case .lazy:
856 expected = direct.values
857 case .sparse:
858 expected = [direct.minimum, direct.maximum]
859 case .persist(let tree):
860 prefix = "version=\(tree.positions.roots.count - 1) "
861 default:
862 break
863 }
864 let answer = structure.query(range), matches = answer.values == expected
865 if matches {
866 ok += 1
867 } else {
868 different += 1
869 }
870 rows.append("\(kind.rawValue) \(matches ? "OK" : "DIFF") \(prefix)\(answer.text)")
871 }
872 rows.append("RESULT ok=\(ok) diff=\(different)")
873 return rows
874 }
875 func execute(_ line: String) throws -> [String] {
876 let parts = line.split(separator: " ").map(String.init), command = parts[0]
877 if command == "LOAD" {
878 let removed = structures.count
879 values = parts.dropFirst().map { Int($0)! }
880 original = values
881 structures.removeAll()
882 return ["LOAD n=\(values.count) removed=\(removed)"]
883 }
884 if command == "RESET" {
885 values = original
886 var operation = Operation.sum, block = 1, rebuilt = 0
887 if case .seg(let tree)? = structures[.seg] {
888 operation = tree.operation
889 }
890 if case .sqrt(let tree)? = structures[.sqrt] {
891 block = tree.block
892 }
893 for kind in Kind.allCases where structures[kind] != nil {
894 structures[kind] = create(
895 kind,
896 operation: operation,
897 block: block
898 )
899 rebuilt += 1
900 }
901 return ["RESET rebuilt=\(rebuilt)"]
902 }
903 if command == "ARRAY" || command == "CHECK" {
904 let range = try checkedRange(Int(parts[1])!, Int(parts[2])!)
905 return command == "CHECK" ? check(range) : ["\(line) \(words(values[range.left...range.right]))"]
906 }
907 if command == "MO" {
908 var ranges: [Span] = []
909 for index in stride(from: 1, to: parts.count, by: 2) {
910 ranges.append(try checkedRange(
911 Int(parts[index])!,
912 Int(parts[index + 1])!
913 ))
914 }
915 return mo(values, ranges: ranges)
916 }
917 let action = parts[1]
918 if command == "GRID" {
919 let args = parts.dropFirst(2).map { Int($0)! }
920 if action == "BUILD" {
921 let height = args[0], width = args[1], cells = Array(args.dropFirst(2))
922 if cells
923 .count != height *
924 width
925 {
926 throw CommandError.invalid("BAD COUNT \(cells.count) \(height * width)")
927 }
928 grid = Grid(height: height, width: width, values: cells)
929 return ["GRID BUILD rows=\(height) cols=\(width) total=\(cells.reduce(0, +))"]
930 }
931 guard let grid else {
932 throw CommandError.invalid("NOT BUILT GRID")
933 }
934 if action == "DUMP" {
935 return grid.dump()
936 }
937 let row = args[0], column = args[1]
938 try grid.check(row, column)
939 if action == "ADD" {
940 return ["\(line) visited=\(grid.add(row, column: column, value: args[2]))"]
941 }
942 let bottom = args[2], right = args[3]
943 try grid.check(bottom, right)
944 if row > bottom || column > right {
945 throw CommandError.invalid("BAD RECT \(words(args))")
946 }
947 return ["\(line) \(grid.query(row, column, bottom, right).text)"]
948 }
949 let kind = Kind(rawValue: command)!
950 if action == "BUILD" {
951 var operation = Operation.sum, block = 1
952 if kind == .seg {
953 guard let selected = Operation(rawValue: parts[2])
954 else {
955 throw CommandError.invalid("BAD OP \(parts[2])")
956 }
957 operation = selected
958 } else if kind == .sqrt {
959 block = parts.count == 3 ? Int(parts[2])! : integerSqrt(values.count)
960 if block > values.count {
961 throw CommandError.invalid("BAD BLOCK \(block)")
962 }
963 }
964 let state = create(kind, operation: operation, block: block)
965 structures[kind] = state
966 return [state.buildLine]
967 }
968 guard let state = structures[kind] else {
969 throw CommandError.invalid("NOT BUILT \(command)")
970 }
971 if action == "DUMP" {
972 return state.dump()
973 }
974 let args = parts.dropFirst(2).map { Int($0)! }
975 switch state {
976 case .seg(let tree) where action == "UPDATE":
977 let index = args[0], value = args[1]
978 try checkIndex(index)
979 let visited = tree.update(index, value: value)
980 values[index] = value
981 return ["\(line) root=\(tree.root) visited=\(visited)"]
982 case .fen(let tree) where action == "LOWER":
983 if let index = tree.values.firstIndex(where: { $0 < 0 }) {
984 throw CommandError.invalid("NEGATIVE \(index)")
985 }
986 let answer = tree.bit.lower(args[0])
987 return ["\(line) index=\(answer.value < 0 ? "none" : String(answer.value)) visited=\(answer.visited)"]
988 case .fen(let tree) where action == "ADD" || action == "PREFIX":
989 let index = args[0]
990 try checkIndex(index)
991 if action == "PREFIX" {
992 return ["\(line) \(tree.bit.prefix(index + 1).text)"]
993 }
994 let visited = tree.add(index, value: args[1])
995 values[index] += args[1]
996 return ["\(line) visited=\(visited)"]
997 case .persist(let tree):
998 if action == "NODES" {
999 return [tree.nodeLine]
1000 }
1001 if action == "KTH" {
1002 let range = try checkedRange(args[0], args[1]), k = args[2]
1003 if k < 1 || k > range.length {
1004 throw CommandError.invalid("BAD K \(k)")
1005 }
1006 let answer = tree.kth(range, k: k)
1007 return ["\(line) value=\(answer.value) visited=\(answer.visited)"]
1008 }
1009 let version = args[0]
1010 try checkVersion(version, tree: tree)
1011 if action == "SET" {
1012 let index = args[1]
1013 try checkIndex(index)
1014 let next = tree.positions.roots.count, created = tree.positions.fork(
1015 version,
1016 index: index,
1017 value: args[2]
1018 )
1019 return ["\(line) version=\(next) created=\(created)"]
1020 }
1021 let range = try checkedRange(args[1], args[2])
1022 return ["\(line) \(state.query(range, version: version).text)"]
1023 default:
1024 break
1025 }
1026 let range = try checkedRange(args[0], args[1])
1027 if action == "ADD" || action == "ASSIGN" {
1028 let value = args[2], assignment = action == "ASSIGN", detail: String
1029 switch state {
1030 case .lazy(let tree):
1031 let counts = tree.update(range, value: value, assignment: assignment)
1032 detail = "visited=\(counts.visited) covered=\(counts.covered) pushes=\(counts.pushes)"
1033 case .rfen(let tree):
1034 detail = "visited=\(tree.add(range, value: value))"
1035 case .sqrt(let tree):
1036 detail = tree.add(range, value: value).partition
1037 default:
1038 preconditionFailure("Invalid range update")
1039 }
1040 change(range, value: value, assignment: assignment)
1041 return ["\(line) \(detail)"]
1042 }
1043 return ["\(line) \(state.query(range).text)"]
1044 }
1045}

1048let laboratory = Laboratory(readLine()!.split(separator: " ").map { Int($0)! })
1049let commandCount = Int(readLine()!)!
1050for _ in 0..<commandCount {
1051 let line = readLine()!
1052 do { for row in try laboratory.execute(line) {
1053 print(row)
1054 } } catch CommandError.invalid(let message) { print("ERROR \(message)") }
1055 catch { preconditionFailure("Unexpected error: \(error)") }
1056}
