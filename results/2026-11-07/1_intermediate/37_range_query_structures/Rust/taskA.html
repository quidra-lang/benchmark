5#[derive(Clone, Copy, PartialEq, Eq)]
6enum Kind { Seg, Lazy, Fen, Rfen, Sparse, Sqrt, Persist, Grid }
7impl Kind {
8 const ARRAYS: [Self; 7] = [Self::Seg, Self::Lazy, Self::Fen, Self::Rfen, Self::Sparse, Self::Sqrt, Self::Persist];
9 fn name(self) -> &'static str {
10 match self { Self::Seg => "SEG", Self::Lazy => "LAZY", Self::Fen => "FEN", Self::Rfen => "RFEN",
11 Self::Sparse => "SPARSE", Self::Sqrt => "SQRT", Self::Persist => "PERSIST", Self::Grid => "GRID" }
12 }
13 fn parse(word: &str) -> Self {
14 Self::ARRAYS.iter().copied().chain(std::iter::once(Self::Grid)).find(|kind| kind.name() == word).unwrap()
15 }
16}
17#[derive(Clone, Copy)]
18enum Operation { Sum, Min, Max, Gcd }
19impl Operation {
20 fn parse(word: &str) -> Result<Self, String> {
21 match word { "SUM" => Ok(Self::Sum), "MIN" => Ok(Self::Min), "MAX" => Ok(Self::Max), "GCD" => Ok(Self::Gcd),
22 _ => Err(format!("BAD OP {word}")) }
23 }
24 fn name(self) -> &'static str { match self { Self::Sum => "SUM", Self::Min => "MIN", Self::Max => "MAX", Self::Gcd => "GCD" } }
25 fn label(self) -> &'static str { match self { Self::Sum => "sum", Self::Min => "min", Self::Max => "max", Self::Gcd => "gcd" } }
26 fn combine(self, a: i64, b: i64) -> i64 {
27 match self { Self::Sum => a + b, Self::Min => a.min(b), Self::Max => a.max(b), Self::Gcd => gcd(a, b) }
28 }
29 fn leaf(self, value: i64) -> i64 { if matches!(self, Self::Gcd) { value.abs() } else { value } }
30}
31fn gcd(a: i64, b: i64) -> i64 {
32 let (mut a, mut b) = (a.abs(), b.abs());
33 while b != 0 { (a, b) = (b, a % b); }
34 a
35}
36fn isqrt(n: usize) -> usize {
37 let mut result = 1;
38 while (result + 1) * (result + 1) <= n { result += 1; }
39 result
40}
41fn joined<T: Display>(values: impl IntoIterator<Item = T>) -> String {
42 values.into_iter().map(|value| value.to_string()).collect::<Vec<_>>().join(" ")
43}
44#[derive(Clone, Copy)]
45struct Span { left: usize, right: usize }
46impl Span {
47 fn whole(n: usize) -> Self { Self { left: 0, right: n - 1 } }
48 fn len(self) -> usize { self.right - self.left + 1 }
49 fn leaf(self) -> bool { self.left == self.right }
50 fn covers(self, other: Self) -> bool { self.left <= other.left && other.right <= self.right }
51 fn split(self) -> (Self, Self) {
52 let mid = (self.left + self.right) / 2;
53 (Self { left: self.left, right: mid }, Self { left: mid + 1, right: self.right })
54 }
55}
56#[derive(Clone, Copy, Default, PartialEq, Eq)]
57struct Aggregate { sum: i64, min: i64, max: i64 }
58impl Aggregate {
59 fn single(value: i64) -> Self { Self { sum: value, min: value, max: value } }
60 fn combine(self, other: Self) -> Self {
61 Self { sum: self.sum + other.sum, min: self.min.min(other.min), max: self.max.max(other.max) }
62 }
63 fn text(self) -> String { format!("sum={} min={} max={}", self.sum, self.min, self.max) }
64}
65#[derive(Default)]
66struct Counts { visited: usize, covered: usize, pushes: usize }
67#[derive(Default)]
68struct Answer { value: i64, visited: usize }
69#[derive(PartialEq, Eq)]
70enum QueryValue { Scalar(i64), Triple(Aggregate), Extrema(i64, i64) }
71struct Report { text: String, value: QueryValue }
72impl Report {
73 fn sum(answer: Answer) -> Self {
74 Self { text: format!("sum={} visited={}", answer.value, answer.visited), value: QueryValue::Scalar(answer.value) }
75 }
76}
77fn shape(n: usize) -> Vec<(usize, Span)> {
78 fn visit(node: usize, span: Span, nodes: &mut Vec<(usize, Span)>) {
79 nodes.push((node, span));
80 if !span.leaf() {
81 let (left, right) = span.split();
82 visit(2 * node, left, nodes);
83 visit(2 * node + 1, right, nodes);
84 }
85 }
86 let mut nodes = Vec::new();
87 visit(1, Span::whole(n), &mut nodes);
88 nodes.sort_by_key(|&(node, _)| node);
89 nodes
90}

92struct Segment { op: Operation, n: usize, tree: Vec<i64>, max_node: usize }
93impl Segment {
94 fn new(values: &[i64], op: Operation) -> Self {
95 let mut result = Self { op, n: values.len(), tree: vec![0; 4 * values.len()], max_node: 0 };
96 result.build(1, Span::whole(values.len()), values);
97 result
98 }
99 fn build(&mut self, node: usize, span: Span, values: &[i64]) {
100 self.max_node = self.max_node.max(node);
101 if span.leaf() { self.tree[node] = self.op.leaf(values[span.left]); }
102 else {
103 let (left, right) = span.split();
104 self.build(2 * node, left, values);
105 self.build(2 * node + 1, right, values);
106 self.tree[node] = self.op.combine(self.tree[2 * node], self.tree[2 * node + 1]);
107 }
108 }
109 fn query_at(&self, node: usize, span: Span, wanted: Span) -> Answer {
110 if wanted.covers(span) { return Answer { value: self.tree[node], visited: 1 }; }
111 let (left, right) = span.split();
112 let mut answer = if wanted.right <= left.right { self.query_at(2 * node, left, wanted) }
113 else if wanted.left >= right.left { self.query_at(2 * node + 1, right, wanted) }
114 else {
115 let a = self.query_at(2 * node, left, wanted);
116 let b = self.query_at(2 * node + 1, right, wanted);
117 Answer { value: self.op.combine(a.value, b.value), visited: a.visited + b.visited }
118 };
119 answer.visited += 1;
120 answer
121 }
122 fn query(&self, span: Span) -> Report {
123 let answer = self.query_at(1, Span::whole(self.n), span);
124 Report { text: format!("{}={} visited={}", self.op.label(), answer.value, answer.visited), value: QueryValue::Scalar(answer.value) }
125 }
126 fn update_at(&mut self, node: usize, span: Span, index: usize, value: i64) -> usize {
127 if span.leaf() { self.tree[node] = self.op.leaf(value); return 1; }
128 let (left, right) = span.split();
129 let visited = if index <= left.right { self.update_at(2 * node, left, index, value) }
130 else { self.update_at(2 * node + 1, right, index, value) };
131 self.tree[node] = self.op.combine(self.tree[2 * node], self.tree[2 * node + 1]);
132 visited + 1
133 }
134 fn dump(&self) -> Vec<String> {
135 let mut rows = vec![format!("SEG DUMP {} nodes={}", self.op.name(), 2 * self.n - 1)];
136 rows.extend(shape(self.n).into_iter().map(|(node, span)| format!("NODE {node} {} {} {}", span.left, span.right, self.tree[node])));
137 rows
138 }
139}
140#[derive(Clone, Copy, Default)]
141struct LazyNode { value: Aggregate, assign: Option<i64>, add: i64 }
142struct Lazy { n: usize, nodes: Vec<LazyNode> }
143impl Lazy {
144 fn new(values: &[i64]) -> Self {
145 let mut result = Self { n: values.len(), nodes: vec![LazyNode::default(); 4 * values.len()] };
146 result.build(1, Span::whole(values.len()), values);
147 result
148 }
149 fn build(&mut self, node: usize, span: Span, values: &[i64]) {
150 if span.leaf() { self.nodes[node].value = Aggregate::single(values[span.left]); }
151 else {
152 let (left, right) = span.split();
153 self.build(2 * node, left, values);
154 self.build(2 * node + 1, right, values);
155 self.pull(node);
156 }
157 }
158 fn pull(&mut self, node: usize) { self.nodes[node].value = self.nodes[2 * node].value.combine(self.nodes[2 * node + 1].value); }
159 fn apply(&mut self, node: usize, span: Span, value: i64, assign: bool) {
160 let current = &mut self.nodes[node];
161 if assign { current.value = Aggregate { sum: value * span.len() as i64, min: value, max: value }; }
162 else { current.value.sum += value * span.len() as i64; current.value.min += value; current.value.max += value; }
163 if !span.leaf() {
164 if assign { current.assign = Some(value); current.add = 0; } else { current.add += value; }
165 }
166 }
167 fn push(&mut self, node: usize, span: Span, counts: &mut Counts) {
168 let current = self.nodes[node];
169 if current.assign.is_none() && current.add == 0 { return; }
170 counts.pushes += 1;
171 let (left, right) = span.split();
172 for (child, range) in [(2 * node, left), (2 * node + 1, right)] {
173 if let Some(value) = current.assign { self.apply(child, range, value, true); }
174 if current.add != 0 { self.apply(child, range, current.add, false); }
175 }
176 self.nodes[node].assign = None;
177 self.nodes[node].add = 0;
178 }
179 fn update_at(&mut self, node: usize, span: Span, wanted: Span, value: i64, assign: bool, counts: &mut Counts) {
180 counts.visited += 1;
181 if wanted.covers(span) { counts.covered += 1; self.apply(node, span, value, assign); return; }
182 self.push(node, span, counts);
183 let (left, right) = span.split();
184 if wanted.left <= left.right { self.update_at(2 * node, left, wanted, value, assign, counts); }
185 if wanted.right >= right.left { self.update_at(2 * node + 1, right, wanted, value, assign, counts); }
186 self.pull(node);
187 }
188 fn update(&mut self, span: Span, value: i64, assign: bool) -> Counts {
189 let mut counts = Counts::default();
190 self.update_at(1, Span::whole(self.n), span, value, assign, &mut counts);
191 counts
192 }
193 fn query_at(&mut self, node: usize, span: Span, wanted: Span, counts: &mut Counts) -> Aggregate {
194 counts.visited += 1;
195 if wanted.covers(span) { return self.nodes[node].value; }
196 self.push(node, span, counts);
197 let (left, right) = span.split();
198 if wanted.right <= left.right { return self.query_at(2 * node, left, wanted, counts); }
199 if wanted.left >= right.left { return self.query_at(2 * node + 1, right, wanted, counts); }
200 let a = self.query_at(2 * node, left, wanted, counts);
201 let b = self.query_at(2 * node + 1, right, wanted, counts);
202 a.combine(b)
203 }
204 fn query(&mut self, span: Span) -> Report {
205 let mut counts = Counts::default();
206 let answer = self.query_at(1, Span::whole(self.n), span, &mut counts);
207 Report { text: format!("{} visited={} pushes={}", answer.text(), counts.visited, counts.pushes), value: QueryValue::Triple(answer) }
208 }
209 fn dump(&self) -> Vec<String> {
210 let mut rows = vec![format!("LAZY DUMP nodes={}", 2 * self.n - 1)];
211 for (node, span) in shape(self.n) {
212 let current = self.nodes[node];
213 let assign = current.assign.map_or_else(|| "-".into(), |value| value.to_string());
214 rows.push(format!("NODE {node} {} {} {} assign={assign} add={}", span.left, span.right, current.value.text(), current.add));
215 }
216 rows
217 }
218}
219struct Bit { tree: Vec<i64> }
220impl Bit {
221 fn empty(n: usize) -> Self { Self { tree: vec![0; n + 1] } }
222 fn new(values: &[i64]) -> Self {
223 let mut bit = Self::empty(values.len());
224 for (i, &value) in values.iter().enumerate() { bit.add(i + 1, value); }
225 bit
226 }
227 fn add(&mut self, mut position: usize, value: i64) -> usize {
228 let mut visited = 0;
229 while position < self.tree.len() { self.tree[position] += value; visited += 1; position += position & position.wrapping_neg(); }
230 visited
231 }
232 fn prefix(&self, mut position: usize) -> Answer {
233 let mut answer = Answer::default();
234 while position > 0 { answer.value += self.tree[position]; answer.visited += 1; position -= position & position.wrapping_neg(); }
235 answer
236 }
237 fn range(&self, span: Span) -> Answer {
238 let a = self.prefix(span.right + 1);
239 let b = self.prefix(span.left);
240 Answer { value: a.value - b.value, visited: a.visited + b.visited }
241 }
242 fn lower(&self, mut rest: i64) -> (Option<usize>, usize) {
243 let n = self.tree.len() - 1;
244 let (mut step, mut position, mut visited) = (1, 0, 0);
245 while 2 * step <= n { step *= 2; }
246 while step > 0 {
247 if position + step <= n {
248 visited += 1;
249 if self.tree[position + step] < rest { rest -= self.tree[position + step]; position += step; }
250 }
251 step /= 2;
252 }
253 ((position < n).then_some(position), visited)
254 }
255}
256struct Fenwick { values: Vec<i64>, bit: Bit }
257impl Fenwick { fn new(values: &[i64]) -> Self { Self { values: values.to_vec(), bit: Bit::new(values) } } }
258struct RangeFenwick { first: Bit, second: Bit }
259impl RangeFenwick {
260 fn new(values: &[i64]) -> Self {
261 let mut result = Self { first: Bit::empty(values.len()), second: Bit::empty(values.len()) };
262 for (i, &value) in values.iter().enumerate() {
263 let difference = value - if i == 0 { 0 } else { values[i - 1] };
264 result.first.add(i + 1, difference);
265 result.second.add(i + 1, difference * i as i64);
266 }
267 result
268 }
269 fn add(&mut self, span: Span, value: i64) -> usize {
270 let mut visited = self.first.add(span.left + 1, value);
271 visited += self.first.add(span.right + 2, -value);
272 visited += self.second.add(span.left + 1, value * span.left as i64);
273 visited += self.second.add(span.right + 2, -value * (span.right + 1) as i64);
274 visited
275 }
276 fn prefix(&self, position: usize) -> Answer {
277 let a = self.first.prefix(position);
278 let b = self.second.prefix(position);
279 Answer { value: position as i64 * a.value - b.value, visited: a.visited + b.visited }
280 }
281 fn range(&self, span: Span) -> Answer {
282 let a = self.prefix(span.right + 1);
283 let b = self.prefix(span.left);
284 Answer { value: a.value - b.value, visited: a.visited + b.visited }
285 }
286}
287struct Sparse { values: Vec<i64>, min: Vec<Vec<usize>>, max: Vec<Vec<usize>> }
288impl Sparse {
289 fn new(values: &[i64]) -> Self {
290 let n = values.len();
291 let mut result = Self { values: values.to_vec(), min: vec![(0..n).collect()], max: vec![(0..n).collect()] };
292 let mut length = 2;
293 while length <= n {
294 let level = result.min.len();
295 let (mut lows, mut highs) = (Vec::new(), Vec::new());
296 for i in 0..=n - length {
297 let (a, b) = (result.min[level - 1][i], result.min[level - 1][i + length / 2]);
298 lows.push(if values[a] <= values[b] { a } else { b });
299 let (a, b) = (result.max[level - 1][i], result.max[level - 1][i + length / 2]);
300 highs.push(if values[a] >= values[b] { a } else { b });
301 }
302 result.min.push(lows); result.max.push(highs);
303 length *= 2;
304 }
305 result
306 }
307 fn query(&self, span: Span) -> Report {
308 let (mut level, mut length) = (0, 1);
309 while 2 * length <= span.len() { length *= 2; level += 1; }
310 let end = span.right + 1 - length;
311 let (a, b) = (self.min[level][span.left], self.min[level][end]);
312 let low = if self.values[a] <= self.values[b] { a } else { b };
313 let (a, b) = (self.max[level][span.left], self.max[level][end]);
314 let high = if self.values[a] >= self.values[b] { a } else { b };
315 Report { text: format!("min={} minpos={low} max={} maxpos={high} level={level}", self.values[low], self.values[high]),
316 value: QueryValue::Extrema(self.values[low], self.values[high]) }
317 }
318 fn dump(&self) -> Vec<String> {
319 let mut rows = vec![format!("SPARSE DUMP levels={}", self.min.len())];
320 for level in 0..self.min.len() {
321 rows.push(format!("MIN {level} {}", joined(&self.min[level])));
322 rows.push(format!("MAX {level} {}", joined(&self.max[level])));
323 }
324 rows
325 }
326}
327struct Sqrt { block: usize, values: Vec<i64>, tags: Vec<i64>, sums: Vec<i64> }
328impl Sqrt {
329 fn new(values: &[i64], block: usize) -> Self {
330 let blocks = (values.len() + block - 1) / block;
331 let mut result = Self { block, values: values.to_vec(), tags: vec![0; blocks], sums: vec![0; blocks] };
332 for (i, &value) in values.iter().enumerate() { result.sums[i / block] += value; }
333 result
334 }
335 fn split(&self, span: Span) -> (Vec<usize>, Vec<usize>) {
336 let (first, last) = (span.left / self.block, span.right / self.block);
337 let cells = (span.left..=span.right).filter(|&i| i / self.block == first || i / self.block == last).collect();
338 (cells, (first + 1..last).collect())
339 }
340 fn add(&mut self, span: Span, value: i64) -> (usize, usize) {
341 let (cells, blocks) = self.split(span);
342 for &i in &cells { self.values[i] += value; self.sums[i / self.block] += value; }
343 for &block in &blocks { self.tags[block] += value; self.sums[block] += value * self.block as i64; }
344 (blocks.len(), cells.len())
345 }
346 fn query(&self, span: Span) -> Report {
347 let (cells, blocks) = self.split(span);
348 let mut sum: i64 = cells.iter().map(|&i| self.values[i] + self.tags[i / self.block]).sum();
349 for &block in &blocks { sum += self.sums[block]; }
350 Report { text: format!("sum={sum} whole={} cells={}", blocks.len(), cells.len()), value: QueryValue::Scalar(sum) }
351 }
352 fn dump(&self) -> Vec<String> {
353 let mut rows = vec![format!("SQRT DUMP block={} blocks={}", self.block, self.tags.len())];
354 for block in 0..self.tags.len() {
355 let first = block * self.block;
356 let end = ((block + 1) * self.block).min(self.values.len());
357 rows.push(format!("BLOCK {block} {first} {} add={} sum={} {}", end - 1, self.tags[block], self.sums[block], joined(&self.values[first..end])));
358 }
359 rows
360 }
361}

363struct PersistentNode { value: i64, left: Option<Rc<PersistentNode>>, right: Option<Rc<PersistentNode>> }
364impl PersistentNode {
365 fn build(values: Option<&[i64]>, span: Span, created: &mut usize) -> Rc<Self> {
366 *created += 1;
367 if span.leaf() { return Rc::new(Self { value: values.map_or(0, |items| items[span.left]), left: None, right: None }); }
368 let (left, right) = span.split();
369 let left = Self::build(values, left, created);
370 let right = Self::build(values, right, created);
371 Rc::new(Self { value: left.value + right.value, left: Some(left), right: Some(right) })
372 }
373 fn set(source: &Rc<Self>, span: Span, index: usize, value: i64, increment: bool, created: &mut usize) -> Rc<Self> {
374 *created += 1;
375 if span.leaf() { return Rc::new(Self { value: if increment { source.value + value } else { value }, left: None, right: None }); }
376 let (left_span, right_span) = span.split();
377 let mut left = Rc::clone(source.left.as_ref().unwrap());
378 let mut right = Rc::clone(source.right.as_ref().unwrap());
379 if index <= left_span.right { left = Self::set(&left, left_span, index, value, increment, created); }
380 else { right = Self::set(&right, right_span, index, value, increment, created); }
381 Rc::new(Self { value: left.value + right.value, left: Some(left), right: Some(right) })
382 }
383 fn query(&self, span: Span, wanted: Span) -> Answer {
384 if wanted.covers(span) { return Answer { value: self.value, visited: 1 }; }
385 let (left, right) = span.split();
386 let mut answer = if wanted.right <= left.right { self.left.as_ref().unwrap().query(left, wanted) }
387 else if wanted.left >= right.left { self.right.as_ref().unwrap().query(right, wanted) }
388 else {
389 let a = self.left.as_ref().unwrap().query(left, wanted);
390 let b = self.right.as_ref().unwrap().query(right, wanted);
391 Answer { value: a.value + b.value, visited: a.visited + b.visited }
392 };
393 answer.visited += 1;
394 answer
395 }
396}
397struct Persistent { n: usize, roots: Vec<Rc<PersistentNode>>, prefixes: Vec<Rc<PersistentNode>>, values: Vec<i64>, position_nodes: usize, value_nodes: usize }
398impl Persistent {
399 fn new(values: &[i64]) -> Self {
400 let mut result = Self { n: values.len(), roots: Vec::new(), prefixes: Vec::new(), values: values.to_vec(), position_nodes: 0, value_nodes: 0 };
401 result.roots.push(PersistentNode::build(Some(values), Span::whole(values.len()), &mut result.position_nodes));
402 result.values.sort_unstable(); result.values.dedup();
403 let span = Span::whole(result.values.len());
404 result.prefixes.push(PersistentNode::build(None, span, &mut result.value_nodes));
405 for value in values {
406 let rank = result.values.binary_search(value).unwrap();
407 let root = PersistentNode::set(result.prefixes.last().unwrap(), span, rank, 1, true, &mut result.value_nodes);
408 result.prefixes.push(root);
409 }
410 result
411 }
412 fn set(&mut self, version: usize, index: usize, value: i64) -> (usize, usize) {
413 let before = self.position_nodes;
414 let version_number = self.roots.len();
415 let root = PersistentNode::set(&self.roots[version], Span::whole(self.n), index, value, false, &mut self.position_nodes);
416 self.roots.push(root);
417 (version_number, self.position_nodes - before)
418 }
419 fn query(&self, version: usize, span: Span) -> Report { Report::sum(self.roots[version].query(Span::whole(self.n), span)) }
420 fn kth(&self, span: Span, mut k: i64) -> Answer {
421 let (mut u, mut w) = (&self.prefixes[span.right + 1], &self.prefixes[span.left]);
422 let mut range = Span::whole(self.values.len());
423 let mut visited = 1;
424 while !range.leaf() {
425 let (left, right) = range.split();
426 let count = u.left.as_ref().unwrap().value - w.left.as_ref().unwrap().value;
427 if k <= count { u = u.left.as_ref().unwrap(); w = w.left.as_ref().unwrap(); range = left; }
428 else { k -= count; u = u.right.as_ref().unwrap(); w = w.right.as_ref().unwrap(); range = right; }
429 visited += 1;
430 }
431 Answer { value: self.values[range.left], visited }
432 }
433}
434struct Grid { height: usize, width: usize, tree: Vec<Vec<i64>> }
435impl Grid {
436 fn new(height: usize, width: usize, values: &[i64]) -> Self {
437 let mut result = Self { height, width, tree: vec![vec![0; width + 1]; height + 1] };
438 for (i, &value) in values.iter().enumerate() { result.add(i / width, i % width, value); }
439 result
440 }
441 fn cell(&self, row: i64, col: i64) -> Result<(usize, usize), String> {
442 if row < 0 || row >= self.height as i64 || col < 0 || col >= self.width as i64 { Err(format!("BAD CELL {row} {col}")) }
443 else { Ok((row as usize, col as usize)) }
444 }
445 fn add(&mut self, row: usize, col: usize, value: i64) -> usize {
446 let (mut y, mut visited) = (row + 1, 0);
447 while y <= self.height {
448 let mut x = col + 1;
449 while x <= self.width { self.tree[y][x] += value; visited += 1; x += x & x.wrapping_neg(); }
450 y += y & y.wrapping_neg();
451 }
452 visited
453 }
454 fn prefix(&self, mut row: usize, col: usize) -> Answer {
455 let mut answer = Answer::default();
456 while row > 0 {
457 let mut x = col;
458 while x > 0 { answer.value += self.tree[row][x]; answer.visited += 1; x -= x & x.wrapping_neg(); }
459 row -= row & row.wrapping_neg();
460 }
461 answer
462 }
463 fn query(&self, row: usize, col: usize, bottom: usize, right: usize) -> Answer {
464 let a = self.prefix(bottom + 1, right + 1);
465 let b = self.prefix(row, right + 1);
466 let c = self.prefix(bottom + 1, col);
467 let d = self.prefix(row, col);
468 Answer { value: a.value - b.value - c.value + d.value, visited: a.visited + b.visited + c.visited + d.visited }
469 }
470 fn dump(&self) -> Vec<String> {
471 let mut rows = vec![format!("GRID DUMP rows={} cols={}", self.height, self.width)];
472 rows.extend((1..=self.height).map(|row| format!("ROW {row} {}", joined(&self.tree[row][1..]))));
473 rows
474 }
475}

477#[derive(Default)]
478struct Lab {
479 values: Vec<i64>, original: Vec<i64>,
480 seg: Option<Segment>, lazy: Option<Lazy>, fen: Option<Fenwick>, rfen: Option<RangeFenwick>,
481 sparse: Option<Sparse>, sqrt: Option<Sqrt>, persist: Option<Persistent>, grid: Option<Grid>,
482}
483impl Lab {
484 fn new(values: Vec<i64>) -> Self { Self { original: values.clone(), values, ..Self::default() } }
485 fn has(&self, kind: Kind) -> bool {
486 match kind { Kind::Seg => self.seg.is_some(), Kind::Lazy => self.lazy.is_some(), Kind::Fen => self.fen.is_some(),
487 Kind::Rfen => self.rfen.is_some(), Kind::Sparse => self.sparse.is_some(), Kind::Sqrt => self.sqrt.is_some(),
488 Kind::Persist => self.persist.is_some(), Kind::Grid => self.grid.is_some() }
489 }
490 fn ready(&self, kind: Kind) -> Result<(), String> {
491 if self.has(kind) { Ok(()) } else { Err(format!("NOT BUILT {}", kind.name())) }
492 }
493 fn index(&self, index: i64) -> Result<usize, String> {
494 if index < 0 || index >= self.values.len() as i64 { Err(format!("BAD INDEX {index}")) } else { Ok(index as usize) }
495 }
496 fn span(&self, left: i64, right: i64) -> Result<Span, String> {
497 let start = self.index(left)?;
498 let end = self.index(right)?;
499 if start > end { Err(format!("BAD RANGE {left} {right}")) } else { Ok(Span { left: start, right: end }) }
500 }
501 fn version(&self, version: i64) -> Result<usize, String> {
502 if version < 0 || version >= self.persist.as_ref().unwrap().roots.len() as i64 { Err(format!("BAD VERSION {version}")) }
503 else { Ok(version as usize) }
504 }
505 fn build(&mut self, kind: Kind, operation: Option<Operation>, block: Option<usize>) -> String {
506 let n = self.values.len();
507 match kind {
508 Kind::Seg => {
509 let tree = Segment::new(&self.values, operation.unwrap());
510 let row = format!("SEG BUILD {} nodes={} maxnode={} root={}", tree.op.name(), 2 * n - 1, tree.max_node, tree.tree[1]);
511 self.seg = Some(tree); row
512 }
513 Kind::Lazy => {
514 let tree = Lazy::new(&self.values);
515 let row = format!("LAZY BUILD nodes={} {}", 2 * n - 1, tree.nodes[1].value.text());
516 self.lazy = Some(tree); row
517 }
518 Kind::Fen => {
519 self.fen = Some(Fenwick::new(&self.values));
520 format!("FEN BUILD n={n} total={}", self.values.iter().sum::<i64>())
521 }
522 Kind::Rfen => {
523 self.rfen = Some(RangeFenwick::new(&self.values));
524 format!("RFEN BUILD n={n} total={}", self.values.iter().sum::<i64>())
525 }
526 Kind::Sparse => {
527 let tree = Sparse::new(&self.values);
528 let row = format!("SPARSE BUILD levels={} entries={}", tree.min.len(), tree.min.iter().map(Vec::len).sum::<usize>());
529 self.sparse = Some(tree); row
530 }
531 Kind::Sqrt => {
532 let tree = Sqrt::new(&self.values, block.unwrap());
533 let row = format!("SQRT BUILD block={} blocks={}", tree.block, tree.tags.len());
534 self.sqrt = Some(tree); row
535 }
536 Kind::Persist => {
537 let tree = Persistent::new(&self.values);
538 let row = format!("PERSIST BUILD nodes={} distinct={} valuenodes={}", tree.position_nodes, tree.values.len(), tree.value_nodes);
539 self.persist = Some(tree); row
540 }
541 Kind::Grid => unreachable!(),
542 }
543 }
544 fn query(&mut self, kind: Kind, version: usize, span: Span) -> Report {
545 match kind {
546 Kind::Seg => self.seg.as_ref().unwrap().query(span),
547 Kind::Lazy => self.lazy.as_mut().unwrap().query(span),
548 Kind::Fen => Report::sum(self.fen.as_ref().unwrap().bit.range(span)),
549 Kind::Rfen => Report::sum(self.rfen.as_ref().unwrap().range(span)),
550 Kind::Sparse => self.sparse.as_ref().unwrap().query(span),
551 Kind::Sqrt => self.sqrt.as_ref().unwrap().query(span),
552 Kind::Persist => self.persist.as_ref().unwrap().query(version, span),
553 Kind::Grid => unreachable!(),
554 }
555 }
556 fn dump(&self, kind: Kind) -> Vec<String> {
557 match kind {
558 Kind::Seg => self.seg.as_ref().unwrap().dump(),
559 Kind::Lazy => self.lazy.as_ref().unwrap().dump(),
560 Kind::Fen => vec![format!("FEN DUMP n={}", self.values.len()), format!("T {}", joined(&self.fen.as_ref().unwrap().bit.tree[1..]))],
561 Kind::Rfen => {
562 let tree = self.rfen.as_ref().unwrap();
563 vec![format!("RFEN DUMP n={}", self.values.len()), format!("B1 {}", joined(&tree.first.tree[1..])), format!("B2 {}", joined(&tree.second.tree[1..]))]
564 }
565 Kind::Sparse => self.sparse.as_ref().unwrap().dump(),
566 Kind::Sqrt => self.sqrt.as_ref().unwrap().dump(),
567 Kind::Grid => self.grid.as_ref().unwrap().dump(),
568 Kind::Persist => unreachable!(),
569 }
570 }
571 fn check(&mut self, span: Span) -> Vec<String> {
572 let values = &self.values[span.left..=span.right];
573 let aggregate = Aggregate { sum: values.iter().sum(), min: *values.iter().min().unwrap(), max: *values.iter().max().unwrap() };
574 let common = values.iter().fold(0, |value, &next| gcd(value, next));
575 let mut rows = vec![format!("CHECK {} {} {} gcd={common}", span.left, span.right, aggregate.text())];
576 let (mut ok, mut diff) = (0, 0);
577 for kind in Kind::ARRAYS {
578 if !self.has(kind) { continue; }
579 let expected = match kind {
580 Kind::Seg => QueryValue::Scalar(match self.seg.as_ref().unwrap().op {
581 Operation::Sum => aggregate.sum, Operation::Min => aggregate.min, Operation::Max => aggregate.max, Operation::Gcd => common }),
582 Kind::Lazy => QueryValue::Triple(aggregate),
583 Kind::Sparse => QueryValue::Extrema(aggregate.min, aggregate.max),
584 _ => QueryValue::Scalar(aggregate.sum),
585 };
586 let version = self.persist.as_ref().map_or(0, |tree| tree.roots.len() - 1);
587 let result = self.query(kind, version, span);
588 let verdict = if result.value == expected { ok += 1; "OK" } else { diff += 1; "DIFF" };
589 let prefix = if kind == Kind::Persist { format!("version={version} ") } else { String::new() };
590 rows.push(format!("{} {verdict} {prefix}{}", kind.name(), result.text));
591 }
592 rows.push(format!("RESULT ok={ok} diff={diff}"));
593 rows
594 }
595 fn mo(&self, queries: &[Span]) -> Vec<String> {
596 let block = isqrt(self.values.len());
597 let mut order: Vec<usize> = (0..queries.len()).collect();
598 order.sort_by_key(|&i| {
599 let group = queries[i].left / block;
600 (group, if group % 2 == 0 { queries[i].right as i64 } else { -(queries[i].right as i64) }, i)
601 });
602 let mut unique = self.values.clone(); unique.sort_unstable(); unique.dedup();
603 let ranks: Vec<usize> = self.values.iter().map(|value| unique.binary_search(value).unwrap()).collect();
604 let mut frequencies = vec![0usize; unique.len()];
605 let mut answers = vec![0; queries.len()];
606 let (mut left, mut right, mut moves, mut distinct) = (0usize, 0usize, 0usize, 0usize);
607 for &i in &order {
608 let span = queries[i];
609 let end = span.right + 1;
610 moves += left.abs_diff(span.left) + right.abs_diff(end);
611 while left > span.left { left -= 1; let rank = ranks[left]; if frequencies[rank] == 0 { distinct += 1; } frequencies[rank] += 1; }
612 while right < end { let rank = ranks[right]; if frequencies[rank] == 0 { distinct += 1; } frequencies[rank] += 1; right += 1; }
613 while left < span.left { let rank = ranks[left]; frequencies[rank] -= 1; if frequencies[rank] == 0 { distinct -= 1; } left += 1; }
614 while right > end { right -= 1; let rank = ranks[right]; frequencies[rank] -= 1; if frequencies[rank] == 0 { distinct -= 1; } }
615 answers[i] = distinct;
616 }
617 let mut rows = vec![format!("MO queries={} block={block} moves={moves}", queries.len()), format!("ORDER {}", joined(order.iter().map(|i| i + 1)))];
618 rows.extend(queries.iter().enumerate().map(|(i, span)| format!("ANSWER {} {} {} distinct={}", i + 1, span.left, span.right, answers[i])));
619 rows
620 }
621 fn execute(&mut self, line: &str) -> Result<Vec<String>, String> {
622 let words: Vec<&str> = line.split_whitespace().collect();
623 let number = |index: usize| words[index].parse::<i64>().unwrap();
624 match words[0] {
625 "LOAD" => {
626 let values = words[1..].iter().map(|word| word.parse::<i64>().unwrap()).collect();
627 let removed = Kind::ARRAYS.iter().filter(|&&kind| self.has(kind)).count();
628 let grid = self.grid.take();
629 *self = Self { grid, ..Self::new(values) };
630 return Ok(vec![format!("LOAD n={} removed={removed}", self.values.len())]);
631 }
632 "RESET" => {
633 self.values.clone_from(&self.original);
634 let operation = self.seg.as_ref().map(|tree| tree.op);
635 let block = self.sqrt.as_ref().map(|tree| tree.block);
636 let mut rebuilt = 0;
637 for kind in Kind::ARRAYS { if self.has(kind) { self.build(kind, operation, block); rebuilt += 1; } }
638 return Ok(vec![format!("RESET rebuilt={rebuilt}")]);
639 }
640 "ARRAY" | "CHECK" => {
641 let span = self.span(number(1), number(2))?;
642 return Ok(if words[0] == "CHECK" { self.check(span) }
643 else { vec![format!("ARRAY {} {} {}", span.left, span.right, joined(&self.values[span.left..=span.right]))] });
644 }
645 "MO" => {
646 let mut queries = Vec::new();
647 for i in (1..words.len()).step_by(2) { queries.push(self.span(number(i), number(i + 1))?); }
648 return Ok(self.mo(&queries));
649 }
650 _ => {}
651 }
652 let kind = Kind::parse(words[0]);
653 let action = words[1];
654 if action == "BUILD" {
655 let operation = if kind == Kind::Seg { Some(Operation::parse(words[2])?) } else { None };
656 let block = if kind == Kind::Sqrt {
657 let value = if words.len() == 3 { number(2) as usize } else { isqrt(self.values.len()) };
658 if value > self.values.len() { return Err(format!("BAD BLOCK {value}")); }
659 Some(value)
660 } else { None };
661 if kind == Kind::Grid {
662 let (height, width) = (number(2) as usize, number(3) as usize);
663 if words.len() - 4 != height * width { return Err(format!("BAD COUNT {} {}", words.len() - 4, height * width)); }
664 let values: Vec<i64> = (4..words.len()).map(number).collect();
665 self.grid = Some(Grid::new(height, width, &values));
666 return Ok(vec![format!("GRID BUILD rows={height} cols={width} total={}", values.iter().sum::<i64>())]);
667 }
668 return Ok(vec![self.build(kind, operation, block)]);
669 }
670 self.ready(kind)?;
671 if action == "DUMP" { return Ok(self.dump(kind)); }
672 if kind == Kind::Grid {
673 let grid = self.grid.as_mut().unwrap();
674 let (row, col) = grid.cell(number(2), number(3))?;
675 if action == "ADD" {
676 let value = number(4);
677 let visited = grid.add(row, col, value);
678 return Ok(vec![format!("GRID ADD {row} {col} {value} visited={visited}")]);
679 }
680 let (bottom, right) = grid.cell(number(4), number(5))?;
681 if row > bottom || col > right { return Err(format!("BAD RECT {row} {col} {bottom} {right}")); }
682 let answer = grid.query(row, col, bottom, right);
683 return Ok(vec![format!("GRID SUM {row} {col} {bottom} {right} sum={} visited={}", answer.value, answer.visited)]);
684 }
685 if kind == Kind::Persist {
686 if action == "NODES" {
687 let tree = self.persist.as_ref().unwrap();
688 return Ok(vec![format!("PERSIST NODES versions={} position={} value={}", tree.roots.len(), tree.position_nodes, tree.value_nodes)]);
689 }
690 if action == "KTH" {
691 let span = self.span(number(2), number(3))?;
692 let k = number(4);
693 if k < 1 || k > span.len() as i64 { return Err(format!("BAD K {k}")); }
694 let answer = self.persist.as_ref().unwrap().kth(span, k);
695 return Ok(vec![format!("PERSIST KTH {} {} {k} value={} visited={}", span.left, span.right, answer.value, answer.visited)]);
696 }
697 let version = self.version(number(2))?;
698 if action == "SET" {
699 let index = self.index(number(3))?;
700 let value = number(4);
701 let (next, created) = self.persist.as_mut().unwrap().set(version, index, value);
702 return Ok(vec![format!("PERSIST SET {version} {index} {value} version={next} created={created}")]);
703 }
704 let span = self.span(number(3), number(4))?;
705 let result = self.query(kind, version, span);
706 return Ok(vec![format!("PERSIST QUERY {version} {} {} {}", span.left, span.right, result.text)]);
707 }
708 if kind == Kind::Fen && action == "LOWER" {
709 let tree = self.fen.as_ref().unwrap();
710 if let Some(index) = tree.values.iter().position(|&value| value < 0) { return Err(format!("NEGATIVE {index}")); }
711 let k = number(2);
712 let (index, visited) = tree.bit.lower(k);
713 let index = index.map_or_else(|| "none".into(), |value| value.to_string());
714 return Ok(vec![format!("FEN LOWER {k} index={index} visited={visited}")]);
715 }
716 if kind == Kind::Seg && action == "UPDATE" {
717 let index = self.index(number(2))?;
718 let value = number(3);
719 let tree = self.seg.as_mut().unwrap();
720 let visited = tree.update_at(1, Span::whole(tree.n), index, value);
721 self.values[index] = value;
722 return Ok(vec![format!("SEG UPDATE {index} {value} root={} visited={visited}", tree.tree[1])]);
723 }
724 if kind == Kind::Fen && (action == "ADD" || action == "PREFIX") {
725 let index = self.index(number(2))?;
726 let tree = self.fen.as_mut().unwrap();
727 if action == "ADD" {
728 let value = number(3);
729 tree.values[index] += value;
730 self.values[index] += value;
731 let visited = tree.bit.add(index + 1, value);
732 return Ok(vec![format!("FEN ADD {index} {value} visited={visited}")]);
733 }
734 let answer = tree.bit.prefix(index + 1);
735 return Ok(vec![format!("FEN PREFIX {index} sum={} visited={}", answer.value, answer.visited)]);
736 }
737 let span = self.span(number(2), number(3))?;
738 if action == "ADD" || action == "ASSIGN" {
739 let value = number(4);
740 let assign = action == "ASSIGN";
741 let details = match kind {
742 Kind::Lazy => {
743 let counts = self.lazy.as_mut().unwrap().update(span, value, assign);
744 format!("visited={} covered={} pushes={}", counts.visited, counts.covered, counts.pushes)
745 }
746 Kind::Rfen => format!("visited={}", self.rfen.as_mut().unwrap().add(span, value)),
747 Kind::Sqrt => {
748 let (whole, cells) = self.sqrt.as_mut().unwrap().add(span, value);
749 format!("whole={whole} cells={cells}")
750 }
751 _ => unreachable!(),
752 };
753 for item in &mut self.values[span.left..=span.right] { if assign { *item = value; } else { *item += value; } }
754 Ok(vec![format!("{} {action} {} {} {value} {details}", kind.name(), span.left, span.right)])
755 } else {
756 let result = self.query(kind, 0, span);
757 Ok(vec![format!("{} {action} {} {} {}", kind.name(), span.left, span.right, result.text)])
758 }
759 }
760}

762fn main() -> io::Result<()> {
763 let input = io::stdin();
764 let mut lines = input.lock().lines();
765 let Some(first) = lines.next() else { return Ok(()); };
766 let n: usize = first?.parse().unwrap();
767 let values = lines.next().unwrap()?.split_whitespace().take(n).map(|word| word.parse::<i64>().unwrap()).collect();
768 let mut lab = Lab::new(values);
769 let commands: usize = lines.next().unwrap()?.parse().unwrap();
770 let mut output = io::BufWriter::new(io::stdout().lock());
771 for _ in 0..commands {
772 let line = lines.next().unwrap()?;
773 match lab.execute(&line) {
774 Ok(rows) => for row in rows { writeln!(output, "{row}")?; },
775 Err(reason) => writeln!(output, "ERROR {reason}")?,
776 }
777 }
778 output.flush()
779}
