22 fun combine(a: Long, b: Long): Long =
23 when (this) {
24 SUM -> a + b
25 MIN -> min(a, b)
26 MAX -> max(a, b)
27 GCD -> gcd(a, b)

37private fun gcd(a: Long, b: Long): Long {
38 var x = abs(a)
39 var y = abs(b)
40 while (y != 0L) {
41 val rest = x % y
42 x = y
43 y = rest
44 }
45 return x
46}

48private fun isqrt(n: Int): Int {
49 var root = 1
50 while ((root + 1) * (root + 1) <= n) root++
51 return root

54private data class Span(val l: Int, val r: Int) {
55 val length
56 get() = r - l + 1
57
58 val middle
59 get() = (l + r) / 2
60
61 val leaf
62 get() = l == r
63
64 fun covers(other: Span) = l <= other.l && other.r <= r
65
66 fun left() = Span(l, middle)
67
68 fun right() = Span(middle + 1, r)
69}

92private class Layout(n: Int) {
93 val spans = arrayOfNulls<Span>(4 * n)
94 val count = 2 * n - 1
95 var maxNode = 0
96 private set
97
98 init {
99 build(1, whole(n))
100 }

102 private fun build(node: Int, span: Span) {
103 spans[node] = span
104 maxNode = max(maxNode, node)
105 if (!span.leaf) {
106 build(2 * node, span.left())
107 build(2 * node + 1, span.right())
108 }
109 }

111 operator fun get(node: Int) = spans[node]!!

114private class Segment(values: LongArray, val op: Operation) {
115 val shape = Layout(values.size)
116 val tree = LongArray(4 * values.size)
117
118 init {
119 build(1, values)
120 }
121
122 private fun build(node: Int, values: LongArray) {
123 val span = shape[node]
124 if (span.leaf) tree[node] = op.leaf(values[span.l])
125 else {
126 build(2 * node, values)
127 build(2 * node + 1, values)
128 pull(node)
129 }
130 }
131
132 private fun pull(node: Int) {
133 tree[node] = op.combine(tree[2 * node], tree[2 * node + 1])
134 }
135
136 private fun queryAt(node: Int, wanted: Span): Answer {
137 val span = shape[node]
138 if (wanted.covers(span)) return Answer(tree[node], 1)
139 val answer =
140 when {
141 wanted.r <= span.middle -> queryAt(2 * node, wanted)
142 wanted.l > span.middle -> queryAt(2 * node + 1, wanted)
143 else -> {
144 val left = queryAt(2 * node, wanted)
145 val right = queryAt(2 * node + 1, wanted)
146 Answer(op.combine(left.value, right.value), left.visited + right.visited)
147 }
148 }
149 return answer.copy(visited = answer.visited + 1)
150 }
151
152 fun query(wanted: Span): Report {
153 val answer = queryAt(1, wanted)
154 return Report(
155 "${op.name.lowercase()}=${answer.value} visited=${answer.visited}",
156 longArrayOf(answer.value),
157 )
158 }
159
160 fun update(index: Int, value: Long, node: Int = 1): Int {
161 val span = shape[node]
162 if (span.leaf) {
163 tree[node] = op.leaf(value)
164 return 1
165 }
166 val visited = update(index, value, if (index <= span.middle) 2 * node else 2 * node + 1)
167 pull(node)
168 return visited + 1
169 }
170
171 fun dump(): List<String> = buildList {
172 add("SEG DUMP $op nodes=${shape.count}")
173 for (node in 1..shape.maxNode) {
174 val span = shape.spans[node] ?: continue
175 add("NODE $node ${span.l} ${span.r} ${tree[node]}")
176 }
177 }
178}

192private class Lazy(values: LongArray) {
193 val shape = Layout(values.size)
194 val nodes = Array(4 * values.size) { LazyNode() }
195
196 init {
197 build(1, values)
198 }
199
200 private fun build(node: Int, values: LongArray) {
201 val span = shape[node]
202 if (span.leaf) nodes[node].value = Aggregate.single(values[span.l])
203 else {
204 build(2 * node, values)
205 build(2 * node + 1, values)
206 pull(node)
207 }
208 }
209
210 private fun pull(node: Int) {
211 nodes[node].value = nodes[2 * node].value.merge(nodes[2 * node + 1].value)
212 }
213
214 private fun apply(node: Int, value: Long, assignment: Boolean) {
215 val span = shape[node]
216 val target = nodes[node]
217 target.value =
218 if (assignment) Aggregate(value * span.length, value, value)
219 else target.value.add(value, span.length)
220 if (!span.leaf) {
221 if (assignment) {
222 target.assign = value
223 target.add = 0
224 } else target.add += value
225 }
226 }
227
228 private fun push(node: Int): Int {
229 val source = nodes[node]
230 val assignment = source.assign
231 if (assignment == null && source.add == 0L) return 0
232 for (child in 2 * node..2 * node + 1) {
233 if (assignment != null) apply(child, assignment, true)
234 if (source.add != 0L) apply(child, source.add, false)
235 }
236 source.assign = null
237 source.add = 0
238 return 1
239 }
240
241 fun update(wanted: Span, value: Long, assignment: Boolean): Counts {
242 val counts = Counts()
243 fun visit(node: Int) {
244 counts.visited++
245 val span = shape[node]
246 if (wanted.covers(span)) {
247 counts.covered++
248 apply(node, value, assignment)
249 return
250 }
251 counts.pushes += push(node)
252 if (wanted.l <= span.middle) visit(2 * node)
253 if (wanted.r > span.middle) visit(2 * node + 1)
254 pull(node)
255 }
256 visit(1)
257 return counts
258 }
259
260 private fun queryAt(node: Int, wanted: Span): LazyAnswer {
261 val span = shape[node]
262 if (wanted.covers(span)) return LazyAnswer(nodes[node].value, 1, 0)
263 val pushes = push(node)
264 val answer =
265 when {
266 wanted.r <= span.middle -> queryAt(2 * node, wanted)
267 wanted.l > span.middle -> queryAt(2 * node + 1, wanted)
268 else -> {
269 val left = queryAt(2 * node, wanted)
270 val right = queryAt(2 * node + 1, wanted)
271 LazyAnswer(
272 left.value.merge(right.value),
273 left.visited + right.visited,
274 left.pushes + right.pushes,
275 )
276 }
277 }
278 return answer.copy(visited = answer.visited + 1, pushes = answer.pushes + pushes)
279 }
280
281 fun query(wanted: Span): Report {
282 val answer = queryAt(1, wanted)
283 val value = answer.value
284 return Report(
285 "${value.text()} visited=${answer.visited} pushes=${answer.pushes}",
286 longArrayOf(value.sum, value.min, value.max),
287 )
288 }

290 fun dump(): List<String> = buildList {
291 add("LAZY DUMP nodes=${shape.count}")
292 for (node in 1..shape.maxNode) {
293 val span = shape.spans[node] ?: continue
294 val item = nodes[node]
295 add(
296 "NODE $node ${span.l} ${span.r} ${item.value.text()} assign=${item.assign ?: "-"} add=${item.add}",
297 )

302private class Bit(n: Int) {
303 val tree = LongArray(n + 1)
304
305 constructor(values: LongArray) : this(values.size) {
306 values.forEachIndexed { i, value -> add(i + 1, value) }
307 }
308
309 fun add(position: Int, value: Long): Int {
310 var p = position
311 var visited = 0
312 while (p < tree.size) {
313 tree[p] += value
314 visited++
315 p += p and -p
316 }
317 return visited
318 }
319
320 fun prefix(position: Int): Answer {
321 var p = position
322 var sum = 0L
323 var visited = 0
324 while (p > 0) {
325 sum += tree[p]
326 visited++
327 p -= p and -p
328 }
329 return Answer(sum, visited)
330 }
331
332 fun query(span: Span): Answer {
333 val a = prefix(span.r + 1)
334 val b = prefix(span.l)
335 return Answer(a.value - b.value, a.visited + b.visited)
336 }
337
338 fun lower(target: Long): Answer {
339 val n = tree.size - 1
340 var position = 0
341 var visited = 0
342 var rest = target
343 var step = Integer.highestOneBit(n)
344 while (step > 0) {
345 if (position + step <= n) {
346 val value = tree[position + step]
347 visited++
348 if (value < rest) {
349 rest -= value
350 position += step
351 }
352 }
353 step /= 2
354 }
355 return Answer(if (position < n) position.toLong() else -1, visited)
356 }
357
358 fun dump(name: String) = "$name ${tree.drop(1).joinToString(" ")}"
359}

366private class RangeFenwick(values: LongArray) {
367 val first = Bit(values.size)
368 val second = Bit(values.size)
369
370 init {
371 for (i in values.indices) {
372 val difference = values[i] - if (i == 0) 0 else values[i - 1]
373 first.add(i + 1, difference)
374 second.add(i + 1, difference * i)
375 }
376 }
377
378 fun add(span: Span, value: Long): Int {
379 var visited = first.add(span.l + 1, value)
380 visited += first.add(span.r + 2, -value)
381 visited += second.add(span.l + 1, value * span.l)
382 visited += second.add(span.r + 2, -value * (span.r + 1))
383 return visited
384 }
385
386 private fun prefix(position: Int): Answer {
387 val a = first.prefix(position)
388 val b = second.prefix(position)
389 return Answer(position * a.value - b.value, a.visited + b.visited)
390 }

392 fun query(span: Span): Answer {
393 val a = prefix(span.r + 1)
394 val b = prefix(span.l)
395 return Answer(a.value - b.value, a.visited + b.visited)

399private class Sparse(contents: LongArray) {
400 val values = contents.copyOf()
401 val minimum = mutableListOf(IntArray(values.size) { it })
402 val maximum = mutableListOf(IntArray(values.size) { it })
403 var entries = values.size
404 private set
405
406 init {
407 var level = 1
408 while ((1 shl level) <= values.size) {
409 val length = values.size - (1 shl level) + 1
410 val half = 1 shl (level - 1)
411 val low =
412 IntArray(length) {
413 select(minimum[level - 1][it], minimum[level - 1][it + half], true)
414 }
415 val high =
416 IntArray(length) {
417 select(maximum[level - 1][it], maximum[level - 1][it + half], false)
418 }
419 minimum.add(low)
420 maximum.add(high)
421 entries += length
422 level++
423 }
424 }
425
426 private fun select(a: Int, b: Int, minimum: Boolean) =
427 if (if (minimum) values[a] <= values[b] else values[a] >= values[b]) a else b
428
429 fun query(span: Span): Report {
430 val level = 31 - Integer.numberOfLeadingZeros(span.length)
431 val end = span.r - (1 shl level) + 1
432 val low = select(minimum[level][span.l], minimum[level][end], true)
433 val high = select(maximum[level][span.l], maximum[level][end], false)
434 return Report(
435 "min=${values[low]} minpos=$low max=${values[high]} maxpos=$high level=$level",
436 longArrayOf(values[low], values[high]),
437 )
438 }
439
440 fun dump(): List<String> = buildList {
441 add("SPARSE DUMP levels=${minimum.size}")
442 for (level in minimum.indices) {
443 add("MIN $level ${minimum[level].joinToString(" ")}")
444 add("MAX $level ${maximum[level].joinToString(" ")}")
445 }
446 }
447}

453private class Sqrt(values: LongArray, val block: Int) {
454 val raw = values.copyOf()
455 val tags = LongArray((values.size + block - 1) / block)
456 val sums = LongArray(tags.size)
457
458 init {
459 raw.forEachIndexed { i, value -> sums[i / block] += value }
460 }
461
462 private fun split(span: Span): Partition {
463 val first = span.l / block
464 val last = span.r / block
465 val cells = (span.l..span.r).filter { it / block == first || it / block == last }
466 return Partition(cells, (first + 1 until last).toList())
467 }
468
469 fun add(span: Span, value: Long): Partition {
470 val part = split(span)
471 for (i in part.cells) {
472 raw[i] += value
473 sums[i / block] += value
474 }
475 for (b in part.blocks) {
476 tags[b] += value
477 sums[b] += value * block
478 }
479 return part
480 }
481
482 fun query(span: Span): Report {
483 val part = split(span)
484 var sum = 0L
485 for (i in part.cells) sum += raw[i] + tags[i / block]
486 for (b in part.blocks) sum += sums[b]
487 return Report("sum=$sum ${part.text()}", longArrayOf(sum))
488 }
489
490 fun dump(): List<String> = buildList {
491 add("SQRT DUMP block=$block blocks=${tags.size}")
492 for (b in tags.indices) {
493 val first = b * block
494 val end = min((b + 1) * block, raw.size)
495 add(
496 "BLOCK $b $first ${end - 1} add=${tags[b]} sum=${sums[b]} ${raw.sliceArray(first until end).joinToString(" ")}",
497 )
498 }
499 }
500}

505 val right: PersistentNode? = null,

508private data class ChangedNode(val node: PersistentNode, val created: Int)

510private fun buildNode(values: LongArray?, span: Span): PersistentNode {
511 if (span.leaf) return PersistentNode(values?.get(span.l) ?: 0)
512 val left = buildNode(values, span.left())
513 val right = buildNode(values, span.right())
514 return PersistentNode(left.value + right.value, left, right)
515}
516
517private fun changeNode(
518 source: PersistentNode,
519 span: Span,
520 index: Int,
521 value: Long,
522 increment: Boolean,
523): ChangedNode {
524 if (span.leaf)
525 return ChangedNode(PersistentNode(if (increment) source.value + value else value), 1)
526 var left = source.left!!
527 var right = source.right!!
528 val changed =
529 if (index <= span.middle) {
530 changeNode(left, span.left(), index, value, increment).also { left = it.node }
531 } else {
532 changeNode(right, span.right(), index, value, increment).also { right = it.node }
533 }
534 return ChangedNode(PersistentNode(left.value + right.value, left, right), changed.created + 1)
535}
536
537private fun queryNode(node: PersistentNode, span: Span, wanted: Span): Answer {
538 if (wanted.covers(span)) return Answer(node.value, 1)
539 val answer =
540 when {
541 wanted.r <= span.middle -> queryNode(node.left!!, span.left(), wanted)
542 wanted.l > span.middle -> queryNode(node.right!!, span.right(), wanted)
543 else -> {
544 val left = queryNode(node.left!!, span.left(), wanted)
545 val right = queryNode(node.right!!, span.right(), wanted)
546 Answer(left.value + right.value, left.visited + right.visited)
547 }
548 }
549 return answer.copy(visited = answer.visited + 1)
550}

552private class Persistent(contents: LongArray) {
553 val n = contents.size
554 val values = contents.distinct().sorted().toLongArray()
555 val roots = mutableListOf(buildNode(contents, whole(n)))
556 val prefixes = mutableListOf(buildNode(null, whole(values.size)))
557 var positionNodes = 2 * n - 1
558 private set
559
560 var valueNodes = 2 * values.size - 1
561 private set
562
563 init {
564 for (value in contents) {
565 val changed =
566 changeNode(prefixes.last(), whole(values.size), values.binarySearch(value), 1, true)
567 prefixes.add(changed.node)
568 valueNodes += changed.created
569 }
570 }
571
572 fun set(version: Int, index: Int, value: Long): Int {
573 val changed = changeNode(roots[version], whole(n), index, value, false)
574 roots.add(changed.node)
575 positionNodes += changed.created
576 return changed.created
577 }
578
579 fun query(version: Int, span: Span) = queryNode(roots[version], whole(n), span).report()
580
581 fun kth(wanted: Span, target: Long): Answer {
582 var u = prefixes[wanted.r + 1]
583 var w = prefixes[wanted.l]
584 var span = whole(values.size)
585 var k = target
586 var visited = 1
587 while (!span.leaf) {
588 val count = u.left!!.value - w.left!!.value
589 if (k <= count) {
590 u = u.left!!
591 w = w.left!!
592 span = span.left()
593 } else {
594 k -= count
595 u = u.right!!
596 w = w.right!!
597 span = span.right()
598 }
599 visited++
600 }
601 return Answer(values[span.l], visited)
602 }
603}

605private class Grid(val height: Int, val width: Int, values: LongArray) {
606 val tree = Array(height + 1) { LongArray(width + 1) }
607
608 init {
609 values.forEachIndexed { i, value -> add(i / width, i % width, value) }
610 }
611
612 fun checkCell(row: Int, column: Int) {
613 if (row < 0 || row >= height || column < 0 || column >= width) fail("BAD CELL $row $column")
614 }
615
616 fun add(row: Int, column: Int, value: Long): Int {
617 var visited = 0
618 var y = row + 1
619 while (y <= height) {
620 var x = column + 1
621 while (x <= width) {
622 tree[y][x] += value
623 visited++
624 x += x and -x
625 }
626 y += y and -y
627 }
628 return visited
629 }

631 private fun prefix(row: Int, column: Int): Answer {
632 var sum = 0L
633 var visited = 0
634 var y = row
635 while (y > 0) {
636 var x = column
637 while (x > 0) {
638 sum += tree[y][x]
639 visited++
640 x -= x and -x
641 }
642 y -= y and -y
643 }
644 return Answer(sum, visited)
645 }

647 fun query(row: Int, column: Int, bottom: Int, right: Int): Answer {
648 val a = prefix(bottom + 1, right + 1)
649 val b = prefix(row, right + 1)
650 val c = prefix(bottom + 1, column)
651 val d = prefix(row, column)
652 return Answer(
653 a.value - b.value - c.value + d.value,
654 a.visited + b.visited + c.visited + d.visited,
655 )
656 }

769 var expected = Aggregate(0, values[span.l], values[span.l])
770 var common = 0L
771 for (i in span.l..span.r) {
772 expected = expected.merge(Aggregate.single(values[i]))
773 common = gcd(common, values[i])
774 }
775 val rows = mutableListOf("CHECK ${span.l} ${span.r} ${expected.text()} gcd=$common")
776 var ok = 0
777 var diff = 0
778 for (kind in Kind.entries) {
779 if (kind == Kind.GRID || !has(kind)) continue
780 val wanted =
781 when (kind) {
782 Kind.SEG ->
783 longArrayOf(
784 when (seg!!.op) {
785 Operation.SUM -> expected.sum
786 Operation.MIN -> expected.min
787 Operation.MAX -> expected.max
788 Operation.GCD -> common
789 },
790 )
791 Kind.LAZY -> longArrayOf(expected.sum, expected.min, expected.max)
792 Kind.SPARSE -> longArrayOf(expected.min, expected.max)
793 else -> longArrayOf(expected.sum)
794 }
795 val version = if (kind == Kind.PERSIST) persist!!.roots.lastIndex else 0
796 val answer = query(kind, version, span)
797 val equal = answer.values.contentEquals(wanted)
798 if (equal) ok++ else diff++
799 val prefix = if (kind == Kind.PERSIST) "version=$version " else ""
800 rows.add("$kind ${if (equal) "OK" else "DIFF"} $prefix${answer.text}")
801 }
802 rows.add("RESULT ok=$ok diff=$diff")
803 return rows
804 }

806 fun mo(queries: List<Span>): List<String> {
807 val block = isqrt(values.size)
808 val order =
809 queries.indices.sortedWith(
810 compareBy<Int> { queries[it].l / block }
811 .thenBy { (if (queries[it].l / block % 2 == 0) 1 else -1) * queries[it].r }
812 .thenBy { it },
813 )
814 val unique = values.distinct().sorted().toLongArray()
815 val ranks = IntArray(values.size) { unique.binarySearch(values[it]) }
816 val frequencies = IntArray(unique.size)
817 val answers = IntArray(queries.size)
818 var left = 0
819 var right = 0
820 var moves = 0
821 var distinct = 0
822 for (index in order) {
823 val span = queries[index]
824 val end = span.r + 1
825 moves += abs(span.l - left) + abs(end - right)
826 while (left > span.l) {
827 left--
828 val rank = ranks[left]
829 if (frequencies[rank] == 0) distinct++
830 frequencies[rank]++
831 }
832 while (right < end) {
833 val rank = ranks[right++]
834 if (frequencies[rank] == 0) distinct++
835 frequencies[rank]++
836 }
837 while (left < span.l) {
838 val rank = ranks[left++]
839 frequencies[rank]--
840 if (frequencies[rank] == 0) distinct--
841 }
842 while (right > end) {
843 right--
844 val rank = ranks[right]
845 frequencies[rank]--
846 if (frequencies[rank] == 0) distinct--
847 }
848 answers[index] = distinct
849 }
850 return buildList {
851 add("MO queries=${queries.size} block=$block moves=$moves")
852 add("ORDER ${order.joinToString(" ") { (it + 1).toString() }}")
853 queries.forEachIndexed { i, span ->
854 add("ANSWER ${i + 1} ${span.l} ${span.r} distinct=${answers[i]}")
855 }
856 }
857 }

859 fun execute(line: String): List<String> {
860 val words = line.split(' ')
861 fun number(i: Int) = words[i].toLong()
862 fun index(i: Int) = words[i].toInt()
863 fun span(i: Int) = Span(index(i), index(i + 1))
864 when (words[0]) {
865 "LOAD" -> {
866 val removed = Kind.entries.count { it != Kind.GRID && has(it) }
867 values = words.drop(1).map(String::toLong).toLongArray()
868 original = values.copyOf()
869 seg = null
870 lazy = null
871 fen = null
872 rfen = null
873 sparse = null
874 sqrt = null
875 persist = null
876 return listOf("LOAD n=${values.size} removed=$removed")
877 }
878 "RESET" -> {
879 values = original.copyOf()
880 val op = seg?.op ?: Operation.SUM
881 val block = sqrt?.block ?: 0
882 var rebuilt = 0
883 for (kind in Kind.entries) if (kind != Kind.GRID && has(kind)) {
884 build(kind, op, block)
885 rebuilt++
886 }
887 return listOf("RESET rebuilt=$rebuilt")
888 }
889 "ARRAY",
890 "CHECK" -> {
891 val range = span(1)
892 checkRange(range)
893 return if (words[0] == "CHECK") check(range)
894 else
895 listOf(
896 "ARRAY ${range.l} ${range.r} ${values.sliceArray(range.l..range.r).joinToString(" ")}",
897 )
898 }
899 "MO" -> {
900 val queries = (1 until words.size step 2).map { span(it).also(::checkRange) }
901 return mo(queries)
902 }
903 }
904 val kind = Kind.valueOf(words[0])
905 val action = words[1]
906 if (action == "BUILD") {
907 val op =
908 if (kind == Kind.SEG)
909 Operation.entries.find { it.name == words[2] } ?: fail("BAD OP ${words[2]}")
910 else Operation.SUM
911 var block = 0
912 if (kind == Kind.SQRT) {
913 block = if (words.size == 3) index(2) else isqrt(values.size)
914 if (block > values.size) fail("BAD BLOCK $block")
915 }
916 if (kind == Kind.GRID) {
917 val height = index(2)
918 val width = index(3)
919 val count = words.size - 4
920 if (count != height * width) fail("BAD COUNT $count ${height * width}")
921 val contents = words.drop(4).map(String::toLong).toLongArray()
922 grid = Grid(height, width, contents)
923 return listOf("GRID BUILD rows=$height cols=$width total=${contents.sum()}")
924 }
925 return listOf(build(kind, op, block))
926 }
927 if (!has(kind)) fail("NOT BUILT $kind")
928 if (action == "DUMP") return dump(kind)
929 if (kind == Kind.GRID) {
930 val tree = grid!!
931 val row = index(2)
932 val column = index(3)
933 tree.checkCell(row, column)
934 if (action == "ADD") {
935 val value = number(4)
936 return listOf(
937 "GRID ADD $row $column $value visited=${tree.add(row, column, value)}",
938 )
939 }
940 val bottom = index(4)
941 val right = index(5)
942 tree.checkCell(bottom, right)
943 if (row > bottom || column > right) fail("BAD RECT $row $column $bottom $right")
944 return listOf(
945 "GRID SUM $row $column $bottom $right ${tree.query(row, column, bottom, right).report().text}",
946 )
947 }
948 if (kind == Kind.PERSIST) {
949 val tree = persist!!
950 if (action == "NODES")
951 return listOf(
952 "PERSIST NODES versions=${tree.roots.size} position=${tree.positionNodes} value=${tree.valueNodes}",
953 )
954 if (action == "KTH") {
955 val range = span(2)
956 checkRange(range)
957 val k = number(4)
958 if (k < 1 || k > range.length) fail("BAD K $k")
959 val answer = tree.kth(range, k)
960 return listOf(
961 "PERSIST KTH ${range.l} ${range.r} $k value=${answer.value} visited=${answer.visited}",
962 )
963 }
964 val version = index(2)
965 checkVersion(version)
966 val position = index(3)
967 if (action == "SET") {
968 checkIndex(position)
969 val value = number(4)
970 val next = tree.roots.size
971 val created = tree.set(version, position, value)
972 return listOf(
973 "PERSIST SET $version $position $value version=$next created=$created",
974 )
975 }
976 val range = Span(position, index(4))
977 checkRange(range)
978 return listOf(
979 "PERSIST QUERY $version ${range.l} ${range.r} ${tree.query(version, range).text}",
980 )
981 }
982 if (kind == Kind.FEN && action == "LOWER") {
983 val tree = fen!!
984 for (i in tree.values.indices) if (tree.values[i] < 0) fail("NEGATIVE $i")
985 val k = number(2)
986 val answer = tree.bit.lower(k)
987 val position = if (answer.value < 0) "none" else answer.value.toString()
988 return listOf("FEN LOWER $k index=$position visited=${answer.visited}")
989 }
990 if (kind == Kind.SEG && action == "UPDATE") {
991 val position = index(2)
992 checkIndex(position)
993 val value = number(3)
994 val tree = seg!!
995 val visited = tree.update(position, value)
996 values[position] = value
997 return listOf("SEG UPDATE $position $value root=${tree.tree[1]} visited=$visited")
998 }
999 if (kind == Kind.FEN && (action == "ADD" || action == "PREFIX")) {
1000 val position = index(2)
1001 checkIndex(position)
1002 val tree = fen!!
1003 if (action == "PREFIX")
1004 return listOf("FEN PREFIX $position ${tree.bit.prefix(position + 1).report().text}")
1005 val value = number(3)
1006 values[position] += value
1007 tree.values[position] += value
1008 return listOf("FEN ADD $position $value visited=${tree.bit.add(position + 1, value)}")
1009 }
1010 val range = span(2)
1011 checkRange(range)
1012 if (action == "ADD" || action == "ASSIGN") {
1013 val value = number(4)
1014 val assignment = action == "ASSIGN"
1015 val details =
1016 when (kind) {
1017 Kind.LAZY -> lazy!!.update(range, value, assignment).toString()
1018 Kind.RFEN -> "visited=${rfen!!.add(range, value)}"
1019 Kind.SQRT -> sqrt!!.add(range, value).text()
1020 else -> error("Not a range update")
1021 }
1022 for (i in range.l..range.r) values[i] = if (assignment) value else values[i] + value
1023 return listOf("$kind $action ${range.l} ${range.r} $value $details")
1024 }
1025 return listOf("$kind $action ${range.l} ${range.r} ${query(kind, 0, range).text}")
1026 }
1027}

1029fun main() {
1030 val input = System.`in`.bufferedReader()
1031 if (input.readLine() == null) return
1032 val lab = Lab(input.readLine().split(' ').map(String::toLong).toLongArray())
1033 val count = input.readLine().toInt()
1034 val output = System.out.bufferedWriter()
1035 repeat(count) {
1036 try {
1037 for (row in lab.execute(input.readLine())) output.append(row).append('\n')
1038 } catch (error: CommandError) {
1039 output.append("ERROR ").append(error.message!!).append('\n')
1040 }
1041 output.flush()
1042 }
