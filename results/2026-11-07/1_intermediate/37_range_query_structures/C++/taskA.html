57Integer combine(Operation operation, Integer a, Integer b) {
58 switch (operation) {
59 case Operation::Sum: return a + b;
60 case Operation::Min: return min(a, b);
61 case Operation::Max: return max(a, b);
62 case Operation::Gcd: return gcd(a, b);
63 }
64 throw logic_error("Invalid operation");
65}

73class SegmentTree {
74 struct Node { Span span; Integer value = 0; };
75 int n, maxNode = 0;
76 vector<Node> nodes;
77 void build(const Values& values, int index, Span span) {
78 maxNode = max(maxNode, index);
79 nodes[index].span = span;
80 if (span.leaf()) nodes[index].value = operation == Operation::Gcd ? abs(values[span.left]) : values[span.left];
81 else {
82 build(values, 2 * index, {span.left, span.middle()});
83 build(values, 2 * index + 1, {span.middle() + 1, span.right});
84 pull(index);
85 }
86 }
87 void pull(int index) { nodes[index].value = combine(operation, nodes[2 * index].value, nodes[2 * index + 1].value); }
88public:
89 Operation operation;
90 SegmentTree(const Values& values, Operation op) : n(static_cast<int>(values.size())), nodes(4 * n), operation(op) { build(values, 1, {0, n - 1}); }
91 Integer root() const { return nodes[1].value; }
92 Answer query(Span range, int index = 1) const {
93 const Node& node = nodes[index];
94 if (range.covers(node.span)) return {node.value, 1};
95 Answer answer;
96 if (range.right <= node.span.middle()) answer = query(range, 2 * index);
97 else if (range.left > node.span.middle()) answer = query(range, 2 * index + 1);
98 else {
99 Answer left = query(range, 2 * index), right = query(range, 2 * index + 1);
100 answer = {combine(operation, left.value, right.value), left.visited + right.visited};
101 }
102 ++answer.visited;
103 return answer;
104 }
105 int update(int position, Integer value, int index = 1) {
106 Node& node = nodes[index];
107 if (node.span.leaf()) { node.value = operation == Operation::Gcd ? abs(value) : value; return 1; }
108 int visited = update(position, value, position <= node.span.middle() ? 2 * index : 2 * index + 1);
109 pull(index);
110 return visited + 1;
111 }
112 string buildLine() const { return "SEG BUILD " + operations[static_cast<int>(operation)] + " nodes=" + to_string(2 * n - 1) + " maxnode=" + to_string(maxNode) + " root=" + to_string(root()); }
113 Lines dump() const {
114 Lines rows{"SEG DUMP " + operations[static_cast<int>(operation)] + " nodes=" + to_string(2 * n - 1)};
115 for (int index = 1; index <= maxNode; ++index) {
116 const Node& node = nodes[index];
117 if (node.span.right >= 0) rows.push_back("NODE " + to_string(index) + " " + to_string(node.span.left) + " " + to_string(node.span.right) + " " + to_string(node.value));
118 }
119 return rows;
120 }

123class LazyTree {
124 struct Node { Span span; Aggregate value; optional<Integer> assignment; Integer addition = 0; };
125 int n;
126 vector<Node> nodes;
127 void pull(int index) { nodes[index].value = merge(nodes[2 * index].value, nodes[2 * index + 1].value); }
128 void build(const Values& values, int index, Span span) {
129 nodes[index].span = span;
130 if (span.leaf()) nodes[index].value = {values[span.left], values[span.left], values[span.left]};
131 else {
132 build(values, 2 * index, {span.left, span.middle()});
133 build(values, 2 * index + 1, {span.middle() + 1, span.right});
134 pull(index);
135 }
136 }
137 void apply(int index, Integer value, bool assignment) {
138 Node& node = nodes[index];
139 Integer length = node.span.right - node.span.left + 1;
140 if (assignment) node.value = {value * length, value, value};
141 else { node.value.sum += value * length; node.value.minimum += value; node.value.maximum += value; }
142 if (!node.span.leaf()) {
143 if (assignment) { node.assignment = value; node.addition = 0; }
144 else node.addition += value;
145 }
146 }
147 int push(int index) {
148 Node& node = nodes[index];
149 if (!node.assignment && node.addition == 0) return 0;
150 for (int child : {2 * index, 2 * index + 1}) {
151 if (node.assignment) apply(child, *node.assignment, true);
152 if (node.addition != 0) apply(child, node.addition, false);
153 }
154 node.assignment.reset(); node.addition = 0;
155 return 1;
156 }
157 void updateAt(Span range, Integer value, bool assignment, int index, Counts& counts) {
158 ++counts.visited;
159 Span span = nodes[index].span;
160 if (range.covers(span)) { ++counts.covered; apply(index, value, assignment); return; }
161 counts.pushes += push(index);
162 if (range.left <= span.middle()) updateAt(range, value, assignment, 2 * index, counts);
163 if (range.right > span.middle()) updateAt(range, value, assignment, 2 * index + 1, counts);
164 pull(index);
165 }
166 Aggregate queryAt(Span range, int index, Counts& counts) {
167 ++counts.visited;
168 Span span = nodes[index].span;
169 if (range.covers(span)) return nodes[index].value;
170 counts.pushes += push(index);
171 if (range.right <= span.middle()) return queryAt(range, 2 * index, counts);
172 if (range.left > span.middle()) return queryAt(range, 2 * index + 1, counts);
173 Aggregate left = queryAt(range, 2 * index, counts), right = queryAt(range, 2 * index + 1, counts);
174 return merge(left, right);
175 }
176public:
177 explicit LazyTree(const Values& values) : n(static_cast<int>(values.size())), nodes(4 * n) { build(values, 1, {0, n - 1}); }
178 Counts update(Span range, Integer value, bool assignment) { Counts counts; updateAt(range, value, assignment, 1, counts); return counts; }
179 LazyAnswer query(Span range) { Counts counts; Aggregate value = queryAt(range, 1, counts); return {value, counts}; }
180 string buildLine() const { return "LAZY BUILD nodes=" + to_string(2 * n - 1) + " " + nodes[1].value.text(); }
181 Lines dump() const {
182 Lines rows{"LAZY DUMP nodes=" + to_string(2 * n - 1)};
183 for (size_t index = 1; index < nodes.size(); ++index) {
184 const Node& node = nodes[index];
185 if (node.span.right < 0) continue;
186 rows.push_back("NODE " + to_string(index) + " " + to_string(node.span.left) + " " + to_string(node.span.right) + " " + node.value.text()
187 + " assign=" + (node.assignment ? to_string(*node.assignment) : "-") + " add=" + to_string(node.addition));
188 }
189 return rows;
190 }

193class Fenwick {
194public:
195 Values tree;
196 explicit Fenwick(const Values& values = {}) : tree(values.size() + 1) {
197 copy(values.begin(), values.end(), tree.begin() + 1);
198 for (int position = 1; position < static_cast<int>(tree.size()); ++position) {
199 int parent = position + lowbit(position);
200 if (parent < static_cast<int>(tree.size())) tree[parent] += tree[position];
201 }
202 }
203 int add(int position, Integer value) {
204 int visited = 0;
205 for (; position < static_cast<int>(tree.size()); position += lowbit(position)) { tree[position] += value; ++visited; }
206 return visited;
207 }
208 Answer prefix(int position) const {
209 Answer answer;
210 for (; position >= 1; position -= lowbit(position)) { answer.value += tree[position]; ++answer.visited; }
211 return answer;
212 }
213 Answer range(Span span) const {
214 Answer right = prefix(span.right + 1), left = prefix(span.left);
215 return {right.value - left.value, right.visited + left.visited};
216 }
217 Answer lower(Integer rest) const {
218 int size = static_cast<int>(tree.size()) - 1, position = 0, visited = 0;
219 for (int step = 1 << floorLog(size); step >= 1; step /= 2) {
220 if (position + step <= size) {
221 ++visited;
222 if (tree[position + step] < rest) { rest -= tree[position + step]; position += step; }
223 }
224 }
225 return {position < size ? position : -1, visited};
226 }
227 string text() const { return joined(slice(tree, 1, static_cast<int>(tree.size()))); }

237class RangeFenwick {
238 int n;
239 Fenwick first, second;
240 Answer prefix(int position) const {
241 Answer a = first.prefix(position), b = second.prefix(position);
242 return {position * a.value - b.value, a.visited + b.visited};
243 }
244public:
245 explicit RangeFenwick(const Values& values) : n(static_cast<int>(values.size())) {
246 Values differences(n), weighted(n);
247 for (int index = 0; index < n; ++index) {
248 differences[index] = values[index] - (index == 0 ? 0 : values[index - 1]);
249 weighted[index] = differences[index] * index;
250 }
251 first = Fenwick(differences); second = Fenwick(weighted);
252 }
253 int add(Span span, Integer value) {
254 int visited = first.add(span.left + 1, value);
255 visited += first.add(span.right + 2, -value);
256 visited += second.add(span.left + 1, value * span.left);
257 visited += second.add(span.right + 2, -value * (span.right + 1));
258 return visited;
259 }
260 Answer query(Span span) const {
261 Answer right = prefix(span.right + 1), left = prefix(span.left);
262 return {right.value - left.value, right.visited + left.visited};
263 }
264 string buildLine() const { return "RFEN BUILD n=" + to_string(n) + " total=" + to_string(prefix(n).value); }
265 Lines dump() const { return {"RFEN DUMP n=" + to_string(n), "B1 " + first.text(), "B2 " + second.text()}; }

276 explicit SparseTable(const Values& source) : values(source) {

279 minima.push_back(indices); maxima.push_back(indices);

289 }
290 }

293 int a = minima[level][span.left], b = minima[level][end], c = maxima[level][span.left], d = maxima[level][end];

311class SquareRoot {
312 int n;
313 Values raw, tags, sums;
314 struct Split { vector<int> cells, whole; };
315 Split split(Span span) const {
316 int left = span.left / block, right = span.right / block;
317 Split result;
318 if (left == right) {
319 for (int index = span.left; index <= span.right; ++index) result.cells.push_back(index);
320 } else {
321 for (int index = span.left; index < (left + 1) * block; ++index) result.cells.push_back(index);
322 for (int index = right * block; index <= span.right; ++index) result.cells.push_back(index);
323 for (int index = left + 1; index < right; ++index) result.whole.push_back(index);
324 }
325 return result;
326 }
327public:
328 int block;
329 SquareRoot(const Values& values, int width) : n(static_cast<int>(values.size())), raw(values), tags((n + width - 1) / width), sums(tags.size()), block(width) {
330 for (int index = 0; index < n; ++index) sums[index / block] += raw[index];
331 }
332 BlockAnswer add(Span span, Integer value) {
333 Split parts = split(span);
334 for (int index : parts.cells) { raw[index] += value; sums[index / block] += value; }
335 for (int index : parts.whole) { tags[index] += value; sums[index] += value * block; }
336 return {0, static_cast<int>(parts.whole.size()), static_cast<int>(parts.cells.size())};
337 }
338 BlockAnswer query(Span span) const {
339 Split parts = split(span);
340 Integer value = 0;
341 for (int index : parts.cells) value += raw[index] + tags[index / block];
342 for (int index : parts.whole) value += sums[index];
343 return {value, static_cast<int>(parts.whole.size()), static_cast<int>(parts.cells.size())};
344 }
345 string buildLine() const { return "SQRT BUILD block=" + to_string(block) + " blocks=" + to_string(tags.size()); }
346 Lines dump() const {
347 Lines rows{"SQRT DUMP block=" + to_string(block) + " blocks=" + to_string(tags.size())};
348 for (int index = 0; index < static_cast<int>(tags.size()); ++index) {
349 int first = index * block, end = min(first + block, n);
350 rows.push_back("BLOCK " + to_string(index) + " " + to_string(first) + " " + to_string(end - 1)
351 + " add=" + to_string(tags[index]) + " sum=" + to_string(sums[index]) + " " + joined(slice(raw, first, end)));
352 }
353 return rows;
354 }

362class PersistentTree {
363 int n;
364 Root node(Integer value, Root left = nullptr, Root right = nullptr) {
365 ++created;
366 return make_shared<const PersistentNode>(PersistentNode{value, move(left), move(right)});
367 }
368 Root build(const Values& values, Span span) {
369 if (span.leaf()) return node(values[span.left]);
370 Root left = build(values, {span.left, span.middle()}), right = build(values, {span.middle() + 1, span.right});
371 return node(left->sum + right->sum, left, right);
372 }
373 Root copyPath(Root previous, Span span, int index, Integer value, bool addition) {
374 if (span.leaf()) return node(addition ? previous->sum + value : value);
375 Root left = previous->left, right = previous->right;
376 if (index <= span.middle()) left = copyPath(left, {span.left, span.middle()}, index, value, addition);
377 else right = copyPath(right, {span.middle() + 1, span.right}, index, value, addition);
378 return node(left->sum + right->sum, left, right);
379 }
380 Answer queryAt(Root root, Span span, Span range) const {
381 if (range.covers(span)) return {root->sum, 1};
382 Answer answer;
383 if (range.right <= span.middle()) answer = queryAt(root->left, {span.left, span.middle()}, range);
384 else if (range.left > span.middle()) answer = queryAt(root->right, {span.middle() + 1, span.right}, range);
385 else {
386 Answer left = queryAt(root->left, {span.left, span.middle()}, range);
387 Answer right = queryAt(root->right, {span.middle() + 1, span.right}, range);
388 answer = {left.value + right.value, left.visited + right.visited};
389 }
390 ++answer.visited;
391 return answer;
392 }
393public:
394 int created = 0;
395 vector<Root> roots;
396 explicit PersistentTree(const Values& values) : n(static_cast<int>(values.size())) { roots.push_back(build(values, {0, n - 1})); }
397 int fork(int version, int index, Integer value, bool addition = false) {
398 int before = created;
399 roots.push_back(copyPath(roots[version], {0, n - 1}, index, value, addition));
400 return created - before;
401 }
402 Answer query(int version, Span span) const { return queryAt(roots[version], {0, n - 1}, span); }

404class Persistence {
405 static Values distinct(Values values) {
406 sort(values.begin(), values.end());
407 values.erase(unique(values.begin(), values.end()), values.end());
408 return values;
409 }
410 Values domain;
411 PersistentTree prefixes;
412public:
413 PersistentTree positions;
414 explicit Persistence(const Values& values) : domain(distinct(values)), prefixes(Values(domain.size())), positions(values) {
415 for (int index = 0; index < static_cast<int>(values.size()); ++index) {
416 int rank = static_cast<int>(lower_bound(domain.begin(), domain.end(), values[index]) - domain.begin());
417 prefixes.fork(index, rank, 1, true);
418 }
419 }
420 Answer kth(Span span, int k) const {
421 Root upper = prefixes.roots[span.right + 1], lower = prefixes.roots[span.left];
422 int left = 0, right = static_cast<int>(domain.size()) - 1, visited = 1;
423 while (left < right) {
424 int middle = (left + right) / 2;
425 Integer count = upper->left->sum - lower->left->sum;
426 if (k <= count) { upper = upper->left; lower = lower->left; right = middle; }
427 else { k -= static_cast<int>(count); upper = upper->right; lower = lower->right; left = middle + 1; }
428 ++visited;
429 }
430 return {domain[left], visited};
431 }
432 string buildLine() const { return "PERSIST BUILD nodes=" + to_string(positions.created) + " distinct=" + to_string(domain.size()) + " valuenodes=" + to_string(prefixes.created); }
433 string nodeLine() const { return "PERSIST NODES versions=" + to_string(positions.roots.size()) + " position=" + to_string(positions.created) + " value=" + to_string(prefixes.created); }

467Lines mo(const Values& values, const vector<Span>& ranges) {
468 int block = integerSqrt(static_cast<int>(values.size()));
469 vector<int> order(ranges.size()), answers(ranges.size());
470 iota(order.begin(), order.end(), 0);
471 sort(order.begin(), order.end(), [&](int a, int b) {
472 int first = ranges[a].left / block, second = ranges[b].left / block;
473 if (first != second) return first < second;
474 if (ranges[a].right != ranges[b].right) return first % 2 == 0 ? ranges[a].right < ranges[b].right : ranges[a].right > ranges[b].right;
475 return a < b;
476 });
477 unordered_map<Integer, int> frequency;
478 int left = 0, right = -1, moves = 0;
479 auto add = [&](int index) { ++frequency[values[index]]; ++moves; };
480 auto remove = [&](int index) { auto entry = frequency.find(values[index]); if (--entry->second == 0) frequency.erase(entry); ++moves; };
481 for (int index : order) {
482 Span range = ranges[index];
483 while (left > range.left) add(--left);
484 while (right < range.right) add(++right);
485 while (left < range.left) remove(left++);
486 while (right > range.right) remove(right--);
487 answers[index] = static_cast<int>(frequency.size());
488 }
489 for (int& index : order) ++index;
490 Lines rows{"MO queries=" + to_string(ranges.size()) + " block=" + to_string(block) + " moves=" + to_string(moves), "ORDER " + joined(order)};
491 for (int index = 0; index < static_cast<int>(ranges.size()); ++index) {
492 rows.push_back("ANSWER " + to_string(index + 1) + " " + to_string(ranges[index].left) + " " + to_string(ranges[index].right) + " distinct=" + to_string(answers[index]));
493 }
494 return rows;
495}

497class Laboratory {
498 Values values, original;
499 optional<SegmentTree> segment;
500 optional<LazyTree> lazy;
501 optional<FenwickArray> fen;
502 optional<RangeFenwick> rfen;
503 optional<SparseTable> sparse;
504 optional<SquareRoot> sqrt;
505 optional<Persistence> persist;
506 optional<Grid> grid;
507 bool built(Kind kind) const {
508 switch (kind) {
509 case Kind::Seg: return segment.has_value();
510 case Kind::Lazy: return lazy.has_value();
511 case Kind::Fen: return fen.has_value();
512 case Kind::Rfen: return rfen.has_value();
513 case Kind::Sparse: return sparse.has_value();
514 case Kind::Sqrt: return sqrt.has_value();
515 case Kind::Persist: return persist.has_value();
516 }
517 return false;
518 }
519 int checkIndex(Integer index) const {
520 if (index < 0 || index >= static_cast<Integer>(values.size())) throw CommandError("BAD INDEX " + to_string(index));
521 return static_cast<int>(index);
522 }
523 Span checkRange(Integer left, Integer right) const {
524 int l = checkIndex(left), r = checkIndex(right);
525 if (l > r) throw CommandError("BAD RANGE " + to_string(l) + " " + to_string(r));
526 return {l, r};
527 }
528 int checkVersion(Integer version) const {
529 if (version < 0 || version >= static_cast<Integer>(persist->positions.roots.size())) throw CommandError("BAD VERSION " + to_string(version));
530 return static_cast<int>(version);
531 }
532 void checkCell(int row, int column) const {
533 if (row < 0 || row >= grid->height || column < 0 || column >= grid->width) throw CommandError("BAD CELL " + to_string(row) + " " + to_string(column));
534 }
535 string build(Kind kind, Operation operation = Operation::Sum, int block = 1) {
536 switch (kind) {
537 case Kind::Seg: segment.emplace(values, operation); return segment->buildLine();
538 case Kind::Lazy: lazy.emplace(values); return lazy->buildLine();
539 case Kind::Fen: fen.emplace(values); return fen->buildLine();
540 case Kind::Rfen: rfen.emplace(values); return rfen->buildLine();
541 case Kind::Sparse: sparse.emplace(values); return sparse->buildLine();
542 case Kind::Sqrt: sqrt.emplace(values, block); return sqrt->buildLine();
543 case Kind::Persist: persist.emplace(values); return persist->buildLine();
544 }
545 throw logic_error("Invalid structure");
546 }
547 Report describe(Kind kind, Span span, int version = -1) {
548 switch (kind) {
549 case Kind::Seg: {
550 Answer answer = segment->query(span);
551 return {labels[static_cast<int>(segment->operation)] + "=" + to_string(answer.value) + " visited=" + to_string(answer.visited), {answer.value}};
552 }
553 case Kind::Lazy: {
554 LazyAnswer answer = lazy->query(span);
555 return {answer.value.text() + " visited=" + to_string(answer.counts.visited) + " pushes=" + to_string(answer.counts.pushes), {answer.value.sum, answer.value.minimum, answer.value.maximum}};
556 }
557 case Kind::Fen: { Answer answer = fen->bit.range(span); return {measured(answer), {answer.value}}; }
558 case Kind::Rfen: { Answer answer = rfen->query(span); return {measured(answer), {answer.value}}; }
559 case Kind::Sparse: {
560 SparseTable::Result answer = sparse->query(span);
561 return {answer.text(), {answer.minimum, answer.maximum}};
562 }
563 case Kind::Sqrt: {
564 BlockAnswer answer = sqrt->query(span);
565 return {"sum=" + to_string(answer.sum) + " whole=" + to_string(answer.whole) + " cells=" + to_string(answer.cells), {answer.sum}};
566 }
567 case Kind::Persist: {
568 if (version < 0) version = static_cast<int>(persist->positions.roots.size()) - 1;
569 Answer answer = persist->positions.query(version, span);
570 return {measured(answer), {answer.value}};
571 }
572 }
573 throw logic_error("Invalid structure");
574 }
575 Lines dump(Kind kind) const {
576 switch (kind) {
577 case Kind::Seg: return segment->dump();
578 case Kind::Lazy: return lazy->dump();
579 case Kind::Fen: return fen->dump();
580 case Kind::Rfen: return rfen->dump();
581 case Kind::Sparse: return sparse->dump();
582 case Kind::Sqrt: return sqrt->dump();
583 case Kind::Persist: throw logic_error("PERSIST has no DUMP");
584 }
585 throw logic_error("Invalid structure");
586 }
587 Lines check(Span span) {
588 Aggregate expected = direct(values, span);
589 Integer common = 0;
590 for (int index = span.left; index <= span.right; ++index) common = gcd(common, values[index]);
591 Lines rows{"CHECK " + to_string(span.left) + " " + to_string(span.right) + " " + expected.text() + " gcd=" + to_string(common)};
592 int ok = 0, different = 0;
593 for (int index = 0; index < 7; ++index) {
594 Kind kind = static_cast<Kind>(index);
595 if (!built(kind)) continue;
596 Values comparison{expected.sum};
597 if (kind == Kind::Seg) comparison = {array<Integer, 4>{expected.sum, expected.minimum, expected.maximum, common}[static_cast<int>(segment->operation)]};
598 else if (kind == Kind::Lazy) comparison = {expected.sum, expected.minimum, expected.maximum};
599 else if (kind == Kind::Sparse) comparison = {expected.minimum, expected.maximum};
600 Report report = describe(kind, span);
601 bool matches = report.values == comparison;
602 if (matches) ++ok; else ++different;
603 string prefix = kind == Kind::Persist ? "version=" + to_string(persist->positions.roots.size() - 1) + " " : "";
604 rows.push_back(names[index] + (matches ? " OK " : " DIFF ") + prefix + report.text);
605 }
606 rows.push_back("RESULT ok=" + to_string(ok) + " diff=" + to_string(different));
607 return rows;
608 }
609 void change(Span span, Integer value, bool assignment = false) {
610 for (int index = span.left; index <= span.right; ++index) values[index] = assignment ? value : values[index] + value;
611 }
612 static Values arguments(const vector<string>& words, size_t start) {
613 Values result;
614 for (size_t index = start; index < words.size(); ++index) result.push_back(stoll(words[index]));
615 return result;
616 }
617public:
618 explicit Laboratory(Values source) : values(source), original(move(source)) {}
619 Lines execute(const string& line) {
620 istringstream input(line);
621 vector<string> words;
622 for (string word; input >> word;) words.push_back(move(word));
623 const string& command = words[0];
624 if (command == "LOAD") {
625 int removed = 0;
626 for (int index = 0; index < 7; ++index) removed += built(static_cast<Kind>(index));
627 values = arguments(words, 1); original = values;
628 segment.reset(); lazy.reset(); fen.reset(); rfen.reset(); sparse.reset(); sqrt.reset(); persist.reset();
629 return {"LOAD n=" + to_string(values.size()) + " removed=" + to_string(removed)};
630 }
631 if (command == "RESET") {
632 values = original;
633 Operation operation = segment ? segment->operation : Operation::Sum;
634 int block = sqrt ? sqrt->block : 1, rebuilt = 0;
635 for (int index = 0; index < 7; ++index) {
636 Kind kind = static_cast<Kind>(index);
637 if (built(kind)) { build(kind, operation, block); ++rebuilt; }
638 }
639 return {"RESET rebuilt=" + to_string(rebuilt)};
640 }
641 if (command == "ARRAY" || command == "CHECK") {
642 Span span = checkRange(stoll(words[1]), stoll(words[2]));
643 return command == "CHECK" ? check(span) : Lines{line + " " + joined(slice(values, span.left, span.right + 1))};
644 }
645 if (command == "MO") {
646 vector<Span> ranges;
647 for (size_t index = 1; index < words.size(); index += 2) ranges.push_back(checkRange(stoll(words[index]), stoll(words[index + 1])));
648 return mo(values, ranges);
649 }
650 const string& action = words[1];
651 if (command == "GRID") {
652 Values args = arguments(words, 2);
653 if (action == "BUILD") {
654 int h = static_cast<int>(args[0]), w = static_cast<int>(args[1]), count = static_cast<int>(args.size()) - 2;
655 if (count != h * w) throw CommandError("BAD COUNT " + to_string(count) + " " + to_string(h * w));
656 Values cells = slice(args, 2, static_cast<int>(args.size()));
657 grid.emplace(h, w, cells);
658 return {"GRID BUILD rows=" + to_string(h) + " cols=" + to_string(w) + " total=" + to_string(accumulate(cells.begin(), cells.end(), Integer{0}))};
659 }
660 if (!grid) throw CommandError("NOT BUILT GRID");
661 if (action == "DUMP") return grid->dump();
662 int row = static_cast<int>(args[0]), column = static_cast<int>(args[1]);
663 checkCell(row, column);
664 if (action == "ADD") return {line + " visited=" + to_string(grid->add(row, column, args[2]))};
665 int bottom = static_cast<int>(args[2]), right = static_cast<int>(args[3]);
666 checkCell(bottom, right);
667 if (row > bottom || column > right) throw CommandError("BAD RECT " + joined(args));
668 return {line + " " + measured(grid->query(row, column, bottom, right))};
669 }
670 auto found = find(names.begin(), names.end(), command);
671 Kind kind = static_cast<Kind>(found - names.begin());
672 if (action == "BUILD") {
673 Operation operation = Operation::Sum;
674 int block = 1;
675 if (kind == Kind::Seg) {
676 auto match = find(operations.begin(), operations.end(), words[2]);
677 if (match == operations.end()) throw CommandError("BAD OP " + words[2]);
678 operation = static_cast<Operation>(match - operations.begin());
679 } else if (kind == Kind::Sqrt) {
680 block = words.size() == 3 ? stoi(words[2]) : integerSqrt(static_cast<int>(values.size()));
681 if (block > static_cast<int>(values.size())) throw CommandError("BAD BLOCK " + to_string(block));
682 }
683 return {build(kind, operation, block)};
684 }
685 if (!built(kind)) throw CommandError("NOT BUILT " + command);
686 if (action == "DUMP") return dump(kind);
687 Values args = arguments(words, 2);
688 if (kind == Kind::Persist) {
689 if (action == "NODES") return {persist->nodeLine()};
690 if (action == "KTH") {
691 Span span = checkRange(args[0], args[1]);
692 Integer k = args[2];
693 if (k < 1 || k > span.right - span.left + 1) throw CommandError("BAD K " + to_string(k));
694 Answer answer = persist->kth(span, static_cast<int>(k));
695 return {line + " value=" + to_string(answer.value) + " visited=" + to_string(answer.visited)};
696 }
697 int version = checkVersion(args[0]);
698 if (action == "SET") {
699 int index = checkIndex(args[1]), next = static_cast<int>(persist->positions.roots.size());
700 int created = persist->positions.fork(version, index, args[2]);
701 return {line + " version=" + to_string(next) + " created=" + to_string(created)};
702 }
703 Span span = checkRange(args[1], args[2]);
704 return {line + " " + describe(kind, span, version).text};
705 }
706 if (kind == Kind::Fen && action == "LOWER") {
707 auto negative = find_if(fen->values.begin(), fen->values.end(), [](Integer value) { return value < 0; });
708 if (negative != fen->values.end()) throw CommandError("NEGATIVE " + to_string(negative - fen->values.begin()));
709 Answer answer = fen->bit.lower(args[0]);
710 return {line + " index=" + (answer.value < 0 ? "none" : to_string(answer.value)) + " visited=" + to_string(answer.visited)};
711 }
712 if (kind == Kind::Seg && action == "UPDATE") {
713 int index = checkIndex(args[0]), visited = segment->update(index, args[1]);
714 values[index] = args[1];
715 return {line + " root=" + to_string(segment->root()) + " visited=" + to_string(visited)};
716 }
717 if (kind == Kind::Fen && (action == "ADD" || action == "PREFIX")) {
718 int index = checkIndex(args[0]);
719 if (action == "PREFIX") return {line + " " + measured(fen->bit.prefix(index + 1))};
720 int visited = fen->add(index, args[1]);
721 values[index] += args[1];
722 return {line + " visited=" + to_string(visited)};
723 }
724 Span span = checkRange(args[0], args[1]);
725 if (action == "ADD" || action == "ASSIGN") {
726 Integer value = args[2];
727 bool assignment = action == "ASSIGN";
728 string details;
729 if (kind == Kind::Lazy) {
730 Counts counts = lazy->update(span, value, assignment);
731 details = "visited=" + to_string(counts.visited) + " covered=" + to_string(counts.covered) + " pushes=" + to_string(counts.pushes);
732 } else if (kind == Kind::Rfen) details = "visited=" + to_string(rfen->add(span, value));
733 else {
734 BlockAnswer answer = sqrt->add(span, value);
735 details = "whole=" + to_string(answer.whole) + " cells=" + to_string(answer.cells);
736 }
737 change(span, value, assignment);
738 return {line + " " + details};
739 }
740 return {line + " " + describe(kind, span).text};
741 }
742};
743
744int main() {
745 ios::sync_with_stdio(false);
746 cin.tie(nullptr);
747 int size, commands;
748 cin >> size;
749 Values values(size);
750 for (Integer& value : values) cin >> value;
751 cin >> commands;
752 Laboratory laboratory(move(values));
753 string line;
754 getline(cin, line);
755 for (int index = 0; index < commands; ++index) {
756 getline(cin, line);
757 if (!line.empty() && line.back() == '\r') line.pop_back();
758 try { for (const string& row : laboratory.execute(line)) cout << row << '\n'; }
759 catch (const CommandError& error) { cout << "ERROR " << error.what() << '\n'; }
760 }
761}
