17function gcd(a: number, b: number): number {
18 a = Math.abs(a); b = Math.abs(b);

22function combine(op: Operation, a: number, b: number): number {
23 switch (op) {
24 case 'SUM': return a + b;
25 case 'MIN': return Math.min(a, b);
26 case 'MAX': return Math.max(a, b);
27 case 'GCD': return gcd(a, b);

30class Span {
31 constructor(readonly left: number, readonly right: number) {}
32 get length(): number { return this.right - this.left + 1; }
33 get middle(): number { return Math.floor((this.left + this.right) / 2); }
34 get leaf(): boolean { return this.left === this.right; }
35 covers(other: Span): boolean { return this.left <= other.left && other.right <= this.right; }
36 leftChild(): Span { return new Span(this.left, this.middle); }
37 rightChild(): Span { return new Span(this.middle + 1, this.right); }
38}
39const whole = (n: number): Span => new Span(0, n - 1);
40interface Aggregate { readonly sum: number; readonly min: number; readonly max: number }
41interface Answer { readonly value: number; readonly visited: number }
42interface Report { readonly text: string; readonly values: readonly number[] }
43const aggregate = (values: readonly number[]): Aggregate => ({ sum: sum(values), min: Math.min(...values), max: Math.max(...values) });

50class Layout {
51 readonly ranges: (Span | undefined)[];
52 readonly nodes: NumberedSpan[] = [];
53 constructor(readonly n: number) {
54 this.ranges = Array<Span | undefined>(4 * n);
55 const build = (node: number, span: Span): void => {
56 this.ranges[node] = span;
57 this.nodes.push({ node, span });
58 if (!span.leaf) { build(2 * node, span.leftChild()); build(2 * node + 1, span.rightChild()); }
59 };
60 build(1, whole(n));
61 this.nodes.sort((a, b) => a.node - b.node);
62 }
63 at(node: number): Span { return this.ranges[node]!; }
64}

66class Segment {
67 readonly shape: Layout;
68 readonly tree: number[];
69 constructor(values: readonly number[], readonly op: Operation) {
70 this.shape = new Layout(values.length);
71 this.tree = Array<number>(4 * values.length).fill(0);
72 const build = (node: number): void => {
73 const span = this.shape.at(node);
74 if (span.leaf) this.tree[node] = this.leafValue(values[span.left]);
75 else { build(2 * node); build(2 * node + 1); this.pull(node); }
76 };
77 build(1);
78 }
79 private leafValue(value: number): number { return this.op === 'GCD' ? Math.abs(value) : value; }
80 private pull(node: number): void { this.tree[node] = combine(this.op, this.tree[2 * node], this.tree[2 * node + 1]); }
81 private queryAt(node: number, wanted: Span): Answer {
82 const span = this.shape.at(node);
83 if (wanted.covers(span)) return { value: this.tree[node], visited: 1 };
84 let answer: Answer;
85 if (wanted.right <= span.middle) answer = this.queryAt(2 * node, wanted);
86 else if (wanted.left > span.middle) answer = this.queryAt(2 * node + 1, wanted);
87 else {
88 const a = this.queryAt(2 * node, wanted), b = this.queryAt(2 * node + 1, wanted);
89 answer = { value: combine(this.op, a.value, b.value), visited: a.visited + b.visited };
90 }
91 return { value: answer.value, visited: answer.visited + 1 };
92 }
93 query(span: Span): Report {
94 const answer = this.queryAt(1, span);
95 return { text: `${this.op.toLowerCase()}=${answer.value} visited=${answer.visited}`, values: [answer.value] };
96 }
97 update(index: number, value: number, node = 1): number {
98 const span = this.shape.at(node);
99 if (span.leaf) { this.tree[node] = this.leafValue(value); return 1; }
100 const visited = this.update(index, value, index <= span.middle ? 2 * node : 2 * node + 1);
101 this.pull(node);
102 return visited + 1;
103 }
104 dump(): string[] {
105 return [`SEG DUMP ${this.op} nodes=${this.shape.nodes.length}`,
106 ...this.shape.nodes.map(({ node, span }) => `NODE ${node} ${span.left} ${span.right} ${this.tree[node]}`)];
107 }

109interface LazyNode { value: Aggregate; assign: number | null; add: number }
110class LazyTree {
111 readonly shape: Layout;
112 readonly nodes: LazyNode[];
113 constructor(values: readonly number[]) {
114 this.shape = new Layout(values.length);
115 this.nodes = Array.from({ length: 4 * values.length }, () => ({ value: { sum: 0, min: 0, max: 0 }, assign: null, add: 0 }));
116 const build = (node: number): void => {
117 const span = this.shape.at(node);
118 if (span.leaf) this.nodes[node].value = { sum: values[span.left], min: values[span.left], max: values[span.left] };
119 else { build(2 * node); build(2 * node + 1); this.pull(node); }
120 };
121 build(1);
122 }
123 private pull(node: number): void { this.nodes[node].value = merge(this.nodes[2 * node].value, this.nodes[2 * node + 1].value); }
124 private apply(node: number, value: number, assign: boolean): void {
125 const span = this.shape.at(node), target = this.nodes[node], old = target.value;
126 target.value = assign ? { sum: value * span.length, min: value, max: value } :
127 { sum: old.sum + value * span.length, min: old.min + value, max: old.max + value };
128 if (!span.leaf) {
129 if (assign) { target.assign = value; target.add = 0; }
130 else target.add += value;
131 }
132 }
133 private push(node: number, counts: Counts): void {
134 const source = this.nodes[node];
135 if (source.assign === null && source.add === 0) return;
136 for (const child of [2 * node, 2 * node + 1]) {
137 if (source.assign !== null) this.apply(child, source.assign, true);
138 if (source.add !== 0) this.apply(child, source.add, false);
139 }
140 source.assign = null; source.add = 0; counts.pushes++;
141 }
142 update(wanted: Span, value: number, assign: boolean): Counts {
143 const counts = new Counts();
144 const visit = (node: number): void => {
145 counts.visited++;
146 const span = this.shape.at(node);
147 if (wanted.covers(span)) { counts.covered++; this.apply(node, value, assign); return; }
148 this.push(node, counts);
149 if (wanted.left <= span.middle) visit(2 * node);
150 if (wanted.right > span.middle) visit(2 * node + 1);
151 this.pull(node);
152 };
153 visit(1);
154 return counts;
155 }
156 query(wanted: Span): Report {
157 const counts = new Counts();
158 const visit = (node: number): Aggregate => {
159 counts.visited++;
160 const span = this.shape.at(node);
161 if (wanted.covers(span)) return this.nodes[node].value;
162 this.push(node, counts);
163 if (wanted.right <= span.middle) return visit(2 * node);
164 if (wanted.left > span.middle) return visit(2 * node + 1);
165 const a = visit(2 * node), b = visit(2 * node + 1);
166 return merge(a, b);
167 };
168 const value = visit(1);
169 return { text: `${describe(value)} visited=${counts.visited} pushes=${counts.pushes}`, values: [value.sum, value.min, value.max] };
170 }
171 dump(): string[] {
172 return [`LAZY DUMP nodes=${this.shape.nodes.length}`, ...this.shape.nodes.map(({ node, span }) => {
173 const item = this.nodes[node];
174 return `NODE ${node} ${span.left} ${span.right} ${describe(item.value)} assign=${item.assign ?? '-'} add=${item.add}`;
175 })];
176 }
177}

179class Bit {
180 readonly tree: number[];
181 constructor(n: number, values?: readonly number[]) {
182 this.tree = Array<number>(n + 1).fill(0);
183 values?.forEach((value, index) => this.add(index + 1, value));
184 }
185 add(position: number, value: number): number {
186 let visited = 0;
187 for (; position < this.tree.length; position += lowbit(position)) { this.tree[position] += value; visited++; }
188 return visited;
189 }
190 prefix(position: number): Answer {
191 let value = 0, visited = 0;
192 for (; position > 0; position -= lowbit(position)) { value += this.tree[position]; visited++; }
193 return { value, visited };
194 }
195 query(span: Span): Answer {
196 const a = this.prefix(span.right + 1), b = this.prefix(span.left);
197 return { value: a.value - b.value, visited: a.visited + b.visited };
198 }
199 lower(rest: number): Answer {
200 const n = this.tree.length - 1;
201 let step = 1, position = 0, visited = 0;
202 while (2 * step <= n) step *= 2;
203 for (; step > 0; step = Math.floor(step / 2)) {
204 if (position + step <= n) {
205 const value = this.tree[position + step]; visited++;
206 if (value < rest) { rest -= value; position += step; }
207 }
208 }
209 return { value: position < n ? position : -1, visited };
210 }
211 dump(name: string): string { return `${name} ${words(this.tree.slice(1))}`; }
212}

220class RangeFenwick {
221 readonly first: Bit;
222 readonly second: Bit;
223 constructor(values: readonly number[]) {
224 this.first = new Bit(values.length); this.second = new Bit(values.length);
225 values.forEach((value, index) => {
226 const difference = value - (index === 0 ? 0 : values[index - 1]);
227 this.first.add(index + 1, difference); this.second.add(index + 1, difference * index);
228 });
229 }
230 add(span: Span, value: number): number {
231 let visited = this.first.add(span.left + 1, value);
232 visited += this.first.add(span.right + 2, -value);
233 visited += this.second.add(span.left + 1, value * span.left);
234 visited += this.second.add(span.right + 2, -value * (span.right + 1));
235 return visited;
236 }
237 private prefix(position: number): Answer {
238 const a = this.first.prefix(position), b = this.second.prefix(position);
239 return { value: position * a.value - b.value, visited: a.visited + b.visited };
240 }
241 query(span: Span): Report {
242 const a = this.prefix(span.right + 1), b = this.prefix(span.left);
243 return report({ value: a.value - b.value, visited: a.visited + b.visited });
244 }
245 dump(): string[] { return [`RFEN DUMP n=${this.first.tree.length - 1}`, this.first.dump('B1'), this.second.dump('B2')]; }

248class SparseTable {
249 readonly values: readonly number[];
250 readonly minima: number[][];
251 readonly maxima: number[][];
252 constructor(values: readonly number[]) {
253 this.values = values.slice();
254 this.minima = [Array.from({ length: values.length }, (_, i) => i)];
255 this.maxima = [this.minima[0].slice()];
256 for (let width = 2; width <= values.length; width *= 2) {
257 const previousMin = this.minima.at(-1)!, previousMax = this.maxima.at(-1)!;
258 const minimum: number[] = [], maximum: number[] = [];
259 for (let index = 0; index + width <= values.length; index++) {
260 minimum.push(this.select(previousMin[index], previousMin[index + width / 2], true));
261 maximum.push(this.select(previousMax[index], previousMax[index + width / 2], false));
262 }
263 this.minima.push(minimum); this.maxima.push(maximum);
264 }
265 }
266 private select(a: number, b: number, minimum: boolean): number {
267 return (minimum ? this.values[a] <= this.values[b] : this.values[a] >= this.values[b]) ? a : b;
268 }
269 query(span: Span): Report {
270 let level = 0, width = 1;
271 while (2 * width <= span.length) { level++; width *= 2; }
272 const end = span.right - width + 1;
273 const low = this.select(this.minima[level][span.left], this.minima[level][end], true);
274 const high = this.select(this.maxima[level][span.left], this.maxima[level][end], false);
275 return { text: `min=${this.values[low]} minpos=${low} max=${this.values[high]} maxpos=${high} level=${level}`, values: [this.values[low], this.values[high]] };
276 }
277 dump(): string[] {
278 const rows = [`SPARSE DUMP levels=${this.minima.length}`];
279 this.minima.forEach((row, level) => rows.push(`MIN ${level} ${words(row)}`, `MAX ${level} ${words(this.maxima[level])}`));
280 return rows;
281 }

283interface Partition { readonly cells: readonly number[]; readonly blocks: readonly number[] }
284const partitionText = (part: Partition): string => `whole=${part.blocks.length} cells=${part.cells.length}`;
285class SqrtBlocks {
286 readonly raw: number[];
287 readonly tags: number[];
288 readonly sums: number[];
289 constructor(values: readonly number[], readonly block: number) {
290 this.raw = values.slice(); this.tags = Array<number>(Math.ceil(values.length / block)).fill(0); this.sums = this.tags.slice();
291 values.forEach((value, index) => { this.sums[Math.floor(index / block)] += value; });
292 }
293 private partition(span: Span): Partition {
294 const first = Math.floor(span.left / this.block), last = Math.floor(span.right / this.block);
295 const cells: number[] = [], blocks: number[] = [];
296 for (let index = span.left; index <= span.right; index++) {
297 const block = Math.floor(index / this.block);
298 if (block === first || block === last) cells.push(index);
299 }
300 for (let block = first + 1; block < last; block++) blocks.push(block);
301 return { cells, blocks };
302 }
303 add(span: Span, value: number): Partition {
304 const part = this.partition(span);
305 for (const index of part.cells) { this.raw[index] += value; this.sums[Math.floor(index / this.block)] += value; }
306 for (const block of part.blocks) { this.tags[block] += value; this.sums[block] += value * this.block; }
307 return part;
308 }
309 query(span: Span): Report {
310 const part = this.partition(span);
311 let value = 0;
312 for (const index of part.cells) value += this.raw[index] + this.tags[Math.floor(index / this.block)];
313 for (const block of part.blocks) value += this.sums[block];
314 return { text: `sum=${value} ${partitionText(part)}`, values: [value] };
315 }
316 dump(): string[] {
317 return [`SQRT DUMP block=${this.block} blocks=${this.tags.length}`, ...this.tags.map((tag, block) => {
318 const first = block * this.block, end = Math.min(first + this.block, this.raw.length);
319 return `BLOCK ${block} ${first} ${end - 1} add=${tag} sum=${this.sums[block]} ${words(this.raw.slice(first, end))}`;
320 })];
321 }

324type PersistentNode = { readonly value: number } & (
325 | { readonly kind: 'leaf' }
326 | { readonly kind: 'branch'; readonly left: PersistentNode; readonly right: PersistentNode }
327);
328interface ChangedNode { readonly node: PersistentNode; readonly created: number }
329function buildNode(values: readonly number[] | null, span: Span): PersistentNode {
330 if (span.leaf) return { kind: 'leaf', value: values?.[span.left] ?? 0 };
331 const left = buildNode(values, span.leftChild()), right = buildNode(values, span.rightChild());
332 return { kind: 'branch', value: left.value + right.value, left, right };
333}
334function changeNode(source: PersistentNode, span: Span, index: number, value: number, increment: boolean): ChangedNode {
335 if (source.kind === 'leaf') return { node: { kind: 'leaf', value: increment ? source.value + value : value }, created: 1 };
336 let left = source.left, right = source.right;
337 let changed: ChangedNode;
338 if (index <= span.middle) { changed = changeNode(left, span.leftChild(), index, value, increment); left = changed.node; }
339 else { changed = changeNode(right, span.rightChild(), index, value, increment); right = changed.node; }
340 return { node: { kind: 'branch', value: left.value + right.value, left, right }, created: changed.created + 1 };
341}
342function queryNode(node: PersistentNode, span: Span, wanted: Span): Answer {
343 if (node.kind === 'leaf' || wanted.covers(span)) return { value: node.value, visited: 1 };
344 let answer: Answer;
345 if (wanted.right <= span.middle) answer = queryNode(node.left, span.leftChild(), wanted);
346 else if (wanted.left > span.middle) answer = queryNode(node.right, span.rightChild(), wanted);
347 else {
348 const a = queryNode(node.left, span.leftChild(), wanted), b = queryNode(node.right, span.rightChild(), wanted);
349 answer = { value: a.value + b.value, visited: a.visited + b.visited };
350 }
351 return { value: answer.value, visited: answer.visited + 1 };
352}
353class Persistent {
354 readonly n: number;
355 readonly domain: readonly number[];
356 readonly roots: PersistentNode[];
357 readonly prefixes: PersistentNode[];
358 positionNodes: number;
359 valueNodes: number;
360 constructor(values: readonly number[]) {
361 this.n = values.length;
362 this.domain = [...new Set(values)].sort((a, b) => a - b);
363 this.roots = [buildNode(values, whole(this.n))]; this.positionNodes = 2 * this.n - 1;
364 this.prefixes = [buildNode(null, whole(this.domain.length))]; this.valueNodes = 2 * this.domain.length - 1;
365 const ranks = new Map(this.domain.map((value, index) => [value, index]));
366 for (const value of values) {
367 const changed = changeNode(this.prefixes.at(-1)!, whole(this.domain.length), ranks.get(value)!, 1, true);
368 this.prefixes.push(changed.node); this.valueNodes += changed.created;
369 }
370 }
371 set(version: number, index: number, value: number): number {
372 const changed = changeNode(this.roots[version], whole(this.n), index, value, false);
373 this.roots.push(changed.node); this.positionNodes += changed.created;
374 return changed.created;
375 }
376 query(version: number, span: Span): Report { return report(queryNode(this.roots[version], whole(this.n), span)); }
377 kth(wanted: Span, k: number): Answer {
378 let upper = this.prefixes[wanted.right + 1], lower = this.prefixes[wanted.left];
379 let span = whole(this.domain.length), visited = 1;
380 while (upper.kind === 'branch' && lower.kind === 'branch') {
381 const count = upper.left.value - lower.left.value;
382 if (k <= count) { upper = upper.left; lower = lower.left; span = span.leftChild(); }
383 else { k -= count; upper = upper.right; lower = lower.right; span = span.rightChild(); }
384 visited++;
385 }
386 return { value: this.domain[span.left], visited };
387 }
388}
389class Grid {
390 readonly tree: number[][];
391 constructor(readonly height: number, readonly width: number, values: readonly number[]) {
392 this.tree = Array.from({ length: height + 1 }, () => Array<number>(width + 1).fill(0));

399 let visited = 0;

406 let value = 0, visited = 0;

409 }

411 }

414 const c = this.prefix(bottom + 1, column), d = this.prefix(row, column);

446function parseCommand(line: string): Command {
447 const tokens = line.split(' '), name = tokens[0];
448 const number = (index: number): number => Number(tokens[index]);
449 const span = (index: number): Span => new Span(number(index), number(index + 1));
450 const cell = (index: number): Cell => ({ row: number(index), column: number(index + 1) });
451 if (name === 'LOAD') return { type: 'LOAD', values: tokens.slice(1).map(Number) };
452 if (name === 'RESET') return { type: 'RESET' };
453 if (name === 'ARRAY' || name === 'CHECK') return { type: name, span: span(1) };
454 if (name === 'MO') {
455 const queries: Span[] = [];
456 for (let index = 1; index < tokens.length; index += 2) queries.push(span(index));
457 return { type: 'MO', queries };
458 }
459 const kind = name as Kind, action = tokens[1];
460 if (action === 'BUILD') {
461 if (kind === 'SEG') return { type: 'SEG BUILD', op: tokens[2] };
462 if (kind === 'SQRT') return { type: 'SQRT BUILD', block: tokens.length === 3 ? number(2) : undefined };
463 if (kind === 'GRID') return { type: 'GRID BUILD', height: number(2), width: number(3), values: tokens.slice(4).map(Number) };
464 return { type: 'BUILD', kind };
465 }
466 if (action === 'DUMP' && kind !== 'PERSIST') return { type: 'DUMP', kind };
467 if (kind === 'GRID') return action === 'ADD' ?
468 { type: 'GRID ADD', row: number(2), column: number(3), value: number(4) } :
469 { type: 'GRID SUM', first: cell(2), second: cell(4) };
470 if (kind === 'PERSIST') {
471 if (action === 'NODES') return { type: 'PERSIST NODES' };
472 if (action === 'SET') return { type: 'PERSIST SET', version: number(2), index: number(3), value: number(4) };
473 if (action === 'KTH') return { type: 'PERSIST KTH', span: span(2), k: number(4) };
474 return { type: 'PERSIST QUERY', version: number(2), span: span(3) };
475 }
476 if (kind === 'SEG' && action === 'UPDATE') return { type: 'SEG UPDATE', index: number(2), value: number(3) };
477 if (kind === 'FEN') {
478 if (action === 'ADD') return { type: 'FEN ADD', index: number(2), value: number(3) };
479 if (action === 'PREFIX') return { type: 'FEN PREFIX', index: number(2) };
480 if (action === 'LOWER') return { type: 'FEN LOWER', k: number(2) };
481 }
482 if (kind === 'LAZY' && (action === 'ADD' || action === 'ASSIGN')) return {
483 type: 'LAZY UPDATE', span: span(2), value: number(4), assign: action === 'ASSIGN',
484 };
485 if ((kind === 'RFEN' || kind === 'SQRT') && action === 'ADD') return { type: 'RANGE ADD', kind, span: span(2), value: number(4) };
486 return { type: 'QUERY', kind, span: span(2) };
487}
488
489interface StructureMap {
490 SEG: Segment; LAZY: LazyTree; FEN: Fenwick; RFEN: RangeFenwick;
491 SPARSE: SparseTable; SQRT: SqrtBlocks; PERSIST: Persistent; GRID: Grid;
492}
493class Laboratory {
494 private values: number[];
495 private original: readonly number[];
496 private structures: Partial<StructureMap> = {};
497 constructor(values: readonly number[]) { this.values = values.slice(); this.original = values.slice(); }
498 private require<K extends Kind>(kind: K): StructureMap[K] {
499 const tree = this.structures[kind];
500 if (tree === undefined) fail(`NOT BUILT ${kind}`);
501 return tree;
502 }
503 private checkIndex(index: number): void { if (index < 0 || index >= this.values.length) fail(`BAD INDEX ${index}`); }
504 private checkRange(span: Span): void {
505 this.checkIndex(span.left); this.checkIndex(span.right);
506 if (span.left > span.right) fail(`BAD RANGE ${span.left} ${span.right}`);
507 }
508 private checkVersion(tree: Persistent, version: number): void {
509 if (version < 0 || version >= tree.roots.length) fail(`BAD VERSION ${version}`);
510 }
511 private build(kind: ArrayKind, op: Operation = 'SUM', block = isqrt(this.values.length)): string {
512 const n = this.values.length;
513 switch (kind) {
514 case 'SEG': {
515 const tree = new Segment(this.values, op); this.structures.SEG = tree;
516 return `SEG BUILD ${op} nodes=${tree.shape.nodes.length} maxnode=${tree.shape.nodes.at(-1)!.node} root=${tree.tree[1]}`;
517 }
518 case 'LAZY': {
519 const tree = new LazyTree(this.values); this.structures.LAZY = tree;
520 return `LAZY BUILD nodes=${tree.shape.nodes.length} ${describe(tree.nodes[1].value)}`;
521 }
522 case 'FEN': this.structures.FEN = new Fenwick(this.values); return `FEN BUILD n=${n} total=${sum(this.values)}`;
523 case 'RFEN': this.structures.RFEN = new RangeFenwick(this.values); return `RFEN BUILD n=${n} total=${sum(this.values)}`;
524 case 'SPARSE': {
525 const tree = new SparseTable(this.values); this.structures.SPARSE = tree;
526 return `SPARSE BUILD levels=${tree.minima.length} entries=${sum(tree.minima.map(row => row.length))}`;
527 }
528 case 'SQRT': {
529 const tree = new SqrtBlocks(this.values, block); this.structures.SQRT = tree;
530 return `SQRT BUILD block=${block} blocks=${tree.tags.length}`;
531 }
532 case 'PERSIST': {
533 const tree = new Persistent(this.values); this.structures.PERSIST = tree;
534 return `PERSIST BUILD nodes=${tree.positionNodes} distinct=${tree.domain.length} valuenodes=${tree.valueNodes}`;
535 }
536 }
537 }
538 private updateArray(span: Span, value: number, assign = false): void {
539 for (let index = span.left; index <= span.right; index++) this.values[index] = assign ? value : this.values[index] + value;
540 }
541 private check(span: Span): string[] {
542 const values = this.values.slice(span.left, span.right + 1), direct = aggregate(values), common = values.reduce(gcd, 0);
543 const rows = [`CHECK ${span.left} ${span.right} ${describe(direct)} gcd=${common}`];
544 let ok = 0, different = 0;
545 for (const kind of arrayKinds) {
546 if (this.structures[kind] === undefined) continue;
547 let expected: readonly number[] = [direct.sum];
548 if (kind === 'SEG') expected = [{ SUM: direct.sum, MIN: direct.min, MAX: direct.max, GCD: common }[this.require('SEG').op]];
549 else if (kind === 'LAZY') expected = [direct.sum, direct.min, direct.max];
550 else if (kind === 'SPARSE') expected = [direct.min, direct.max];
551 let answer: Report, prefix = '';
552 if (kind === 'PERSIST') {
553 const tree = this.require(kind), version = tree.roots.length - 1;
554 answer = tree.query(version, span); prefix = `version=${version} `;
555 } else answer = this.require(kind).query(span);
556 const same = answer.values.every((value, index) => value === expected[index]);
557 if (same) ok++; else different++;
558 rows.push(`${kind} ${same ? 'OK' : 'DIFF'} ${prefix}${answer.text}`);
559 }
560 rows.push(`RESULT ok=${ok} diff=${different}`);
561 return rows;
562 }
563 private mo(queries: readonly Span[]): string[] {
564 const blockSize = isqrt(this.values.length);
565 const order = queries.map((_, index) => index).sort((a, b) => {
566 const first = queries[a], second = queries[b], block = Math.floor(first.left / blockSize);
567 return block - Math.floor(second.left / blockSize) ||
568 (block % 2 === 0 ? first.right - second.right : second.right - first.right) || a - b;
569 });
570 const frequencies = new Map<number, number>(), answers = Array<number>(queries.length).fill(0);
571 let left = 0, right = -1, moves = 0, distinct = 0;
572 const add = (index: number): void => {
573 const value = this.values[index], count = frequencies.get(value) ?? 0;
574 if (count === 0) distinct++;
575 frequencies.set(value, count + 1);
576 };
577 const remove = (index: number): void => {
578 const value = this.values[index], count = frequencies.get(value)! - 1;
579 frequencies.set(value, count);
580 if (count === 0) distinct--;
581 };
582 for (const index of order) {
583 const target = queries[index]; moves += Math.abs(target.left - left) + Math.abs(target.right - right);
584 while (left > target.left) { left--; add(left); }
585 while (right < target.right) { right++; add(right); }
586 while (left < target.left) { remove(left); left++; }
587 while (right > target.right) { remove(right); right--; }
588 answers[index] = distinct;
589 }
590 return [`MO queries=${queries.length} block=${blockSize} moves=${moves}`, `ORDER ${words(order.map(index => index + 1))}`,
591 ...queries.map((span, index) => `ANSWER ${index + 1} ${span.left} ${span.right} distinct=${answers[index]}`)];
592 }
593 execute(command: Command, line: string): string[] {
594 switch (command.type) {
595 case 'LOAD': {
596 const removed = arrayKinds.filter(kind => this.structures[kind] !== undefined).length;
597 const grid = this.structures.GRID;
598 this.values = command.values.slice(); this.original = command.values.slice(); this.structures = {};
599 if (grid !== undefined) this.structures.GRID = grid;
600 return [`LOAD n=${this.values.length} removed=${removed}`];
601 }
602 case 'RESET': {
603 this.values = this.original.slice();
604 const op = this.structures.SEG?.op ?? 'SUM', block = this.structures.SQRT?.block ?? 1;
605 let rebuilt = 0;
606 for (const kind of arrayKinds) if (this.structures[kind] !== undefined) { this.build(kind, op, block); rebuilt++; }
607 return [`RESET rebuilt=${rebuilt}`];
608 }
609 case 'ARRAY':
610 this.checkRange(command.span);
611 return [`${line} ${words(this.values.slice(command.span.left, command.span.right + 1))}`];
612 case 'CHECK': this.checkRange(command.span); return this.check(command.span);
613 case 'MO':
614 for (const span of command.queries) this.checkRange(span);
615 return this.mo(command.queries);
616 case 'SEG BUILD':
617 if (!isOperation(command.op)) fail(`BAD OP ${command.op}`);
618 return [this.build('SEG', command.op)];
619 case 'SQRT BUILD': {
620 const block = command.block ?? isqrt(this.values.length);
621 if (block > this.values.length) fail(`BAD BLOCK ${block}`);
622 return [this.build('SQRT', 'SUM', block)];
623 }
624 case 'BUILD': return [this.build(command.kind)];
625 case 'GRID BUILD': {
626 const { height, width, values } = command;
627 if (values.length !== height * width) fail(`BAD COUNT ${values.length} ${height * width}`);
628 this.structures.GRID = new Grid(height, width, values);
629 return [`GRID BUILD rows=${height} cols=${width} total=${sum(values)}`];
630 }
631 case 'DUMP': return this.require(command.kind).dump();
632 case 'QUERY': {
633 const tree = this.require(command.kind);
634 this.checkRange(command.span);
635 return [`${line} ${tree.query(command.span).text}`];
636 }
637 case 'SEG UPDATE': {
638 const tree = this.require('SEG'); this.checkIndex(command.index);
639 const visited = tree.update(command.index, command.value);
640 this.values[command.index] = command.value;
641 return [`${line} root=${tree.tree[1]} visited=${visited}`];
642 }
643 case 'LAZY UPDATE': {
644 const tree = this.require('LAZY'); this.checkRange(command.span);
645 const counts = tree.update(command.span, command.value, command.assign);
646 this.updateArray(command.span, command.value, command.assign);
647 return [`${line} visited=${counts.visited} covered=${counts.covered} pushes=${counts.pushes}`];
648 }
649 case 'FEN ADD': {
650 const tree = this.require('FEN'); this.checkIndex(command.index);
651 const visited = tree.bit.add(command.index + 1, command.value);
652 tree.values[command.index] += command.value; this.values[command.index] += command.value;
653 return [`${line} visited=${visited}`];
654 }
655 case 'FEN PREFIX': {
656 const tree = this.require('FEN'); this.checkIndex(command.index);
657 return [`${line} ${report(tree.bit.prefix(command.index + 1)).text}`];
658 }
659 case 'FEN LOWER': {
660 const tree = this.require('FEN'), negative = tree.values.findIndex(value => value < 0);
661 if (negative >= 0) fail(`NEGATIVE ${negative}`);
662 const answer = tree.bit.lower(command.k);
663 return [`${line} index=${answer.value < 0 ? 'none' : answer.value} visited=${answer.visited}`];
664 }
665 case 'RANGE ADD': {
666 const tree = this.require(command.kind); this.checkRange(command.span);
667 const result = tree.add(command.span, command.value);
668 this.updateArray(command.span, command.value);
669 return [`${line} ${typeof result === 'number' ? `visited=${result}` : partitionText(result)}`];
670 }
671 case 'PERSIST SET': {
672 const tree = this.require('PERSIST'); this.checkVersion(tree, command.version); this.checkIndex(command.index);
673 const version = tree.roots.length, created = tree.set(command.version, command.index, command.value);
674 return [`${line} version=${version} created=${created}`];
675 }
676 case 'PERSIST QUERY': {
677 const tree = this.require('PERSIST'); this.checkVersion(tree, command.version); this.checkRange(command.span);
678 return [`${line} ${tree.query(command.version, command.span).text}`];
679 }
680 case 'PERSIST KTH': {
681 const tree = this.require('PERSIST'); this.checkRange(command.span);
682 if (command.k < 1 || command.k > command.span.length) fail(`BAD K ${command.k}`);
683 const answer = tree.kth(command.span, command.k);
684 return [`${line} value=${answer.value} visited=${answer.visited}`];
685 }
686 case 'PERSIST NODES': {
687 const tree = this.require('PERSIST');
688 return [`PERSIST NODES versions=${tree.roots.length} position=${tree.positionNodes} value=${tree.valueNodes}`];
689 }
690 case 'GRID ADD': {
691 const tree = this.require('GRID'); tree.checkCell(command.row, command.column);
692 return [`${line} visited=${tree.add(command.row, command.column, command.value)}`];
693 }
694 case 'GRID SUM': {
695 const tree = this.require('GRID'), { first, second } = command;
696 tree.checkCell(first.row, first.column); tree.checkCell(second.row, second.column);
697 if (first.row > second.row || first.column > second.column) fail(`BAD RECT ${first.row} ${first.column} ${second.row} ${second.column}`);
698 return [`${line} ${report(tree.query(first.row, first.column, second.row, second.column)).text}`];
699 }
700 }
701 }
702}

704const input = readFileSync(0, 'utf8').trimEnd().split(/\r?\n/);
705const laboratory = new Laboratory(input[1].split(' ').map(Number));
706const commandCount = Number(input[2]);
707for (let index = 0; index < commandCount; index++) {
708 const line = input[index + 3];
709 let rows: string[];
710 try { rows = laboratory.execute(parseCommand(line), line); }
711 catch (error) {
712 if (!(error instanceof CommandError)) throw error;
713 rows = [`ERROR ${error.message}`];
714 }
715 process.stdout.write(`${rows.join('\n')}\n`);
