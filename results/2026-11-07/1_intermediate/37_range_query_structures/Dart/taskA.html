46final class Interval {
47 final int left, right;
48 const Interval(this.left, this.right);
49 bool get leaf => left == right;
50 int get middle => (left + right) ~/ 2;
51 int get length => right - left + 1;
52 bool covered(int l, int r) => l <= left && right <= r;
53}

55final class Layout {
56 final List<Interval?> ranges;
57 final int count;
58 int maxNode = 0;
59 Layout(int n) : ranges = List.filled(4 * n, null), count = 2 * n - 1 {
60 _build(1, 0, n - 1);
61 }
62 Interval operator [](int node) => ranges[node]!;
63 void _build(int node, int left, int right) {
64 ranges[node] = Interval(left, right);
65 maxNode = math.max(maxNode, node);
66 if (left == right) return;
67 final middle = (left + right) ~/ 2;
68 _build(2 * node, left, middle);
69 _build(2 * node + 1, middle + 1, right);
70 }
71 Iterable<(int, Interval)> get nodes sync* {
72 for (var node = 1; node <= maxNode; node++) {
73 final range = ranges[node];
74 if (range != null) yield (node, range);
75 }
76 }
77}

79final class SegmentTree {
80 final Operation operation;
81 final Layout shape;
82 final List<int> values;
83 int get root => values[1];
84 SegmentTree(List<int> contents, this.operation)
85 : shape = Layout(contents.length), values = List.filled(4 * contents.length, 0) {
86 _build(1, contents);
87 }
88
89 void _build(int node, List<int> contents) {
90 final range = shape[node];
91 if (range.leaf) {
92 values[node] = operation == Operation.gcd ? contents[range.left].abs() : contents[range.left];
93 } else {
94 _build(2 * node, contents);
95 _build(2 * node + 1, contents);
96 _pull(node);
97 }
98 }
99 void _pull(int node) => values[node] = combine(operation, values[2 * node], values[2 * node + 1]);
100
101 ScalarAnswer query(int left, int right, [int node = 1]) {
102 final range = shape[node];
103 if (range.covered(left, right)) return (value: values[node], visited: 1);
104 if (right <= range.middle || left > range.middle) {
105 final child = query(left, right, right <= range.middle ? 2 * node : 2 * node + 1);
106 return (value: child.value, visited: child.visited + 1);
107 }
108 final a = query(left, right, 2 * node);
109 final b = query(left, right, 2 * node + 1);
110 return (value: combine(operation, a.value, b.value), visited: a.visited + b.visited + 1);
111 }
112
113 int update(int index, int value, [int node = 1]) {
114 final range = shape[node];
115 if (range.leaf) {
116 values[node] = operation == Operation.gcd ? value.abs() : value;
117 return 1;
118 }
119 final visited = update(index, value, index <= range.middle ? 2 * node : 2 * node + 1);
120 _pull(node);
121 return visited + 1;
122 }
123
124 String describe(ScalarAnswer answer) => '${operation.name}=${answer.value} visited=${answer.visited}';
125 Iterable<String> dump() sync* {
126 yield 'SEG DUMP ${operation.name.toUpperCase()} nodes=${shape.count}';
127 for (final (node, range) in shape.nodes) {
128 yield 'NODE $node ${range.left} ${range.right} ${values[node]}';
129 }
130 }
131}

139final class LazyTree {
140 final Layout shape;
141 final List<Aggregate> values;
142 final List<int?> assign;
143 final List<int> add;
144 Aggregate get root => values[1];
145 LazyTree(List<int> contents)
146 : shape = Layout(contents.length),
147 values = List.filled(4 * contents.length, (sum: 0, min: 0, max: 0)),
148 assign = List.filled(4 * contents.length, null),
149 add = List.filled(4 * contents.length, 0) {
150 _build(1, contents);
151 }
152
153 void _build(int node, List<int> contents) {
154 final range = shape[node];
155 if (range.leaf) {
156 final value = contents[range.left];
157 values[node] = (sum: value, min: value, max: value);
158 } else {
159 _build(2 * node, contents);
160 _build(2 * node + 1, contents);
161 _pull(node);
162 }
163 }
164 void _pull(int node) => values[node] = merge(values[2 * node], values[2 * node + 1]);
165
166 void _apply(int node, int value, bool assignment) {
167 final range = shape[node];
168 final old = values[node];
169 values[node] = assignment
170 ? (sum: value * range.length, min: value, max: value)
171 : (sum: old.sum + value * range.length, min: old.min + value, max: old.max + value);
172 if (range.leaf) return;
173 if (assignment) {
174 assign[node] = value;
175 add[node] = 0;
176 } else {
177 add[node] += value;
178 }
179 }
180
181 int _push(int node) {
182 final assignment = assign[node];
183 if (assignment == null && add[node] == 0) return 0;
184 for (var child = 2 * node; child <= 2 * node + 1; child++) {
185 if (assignment != null) _apply(child, assignment, true);
186 if (add[node] != 0) _apply(child, add[node], false);
187 }
188 assign[node] = null;
189 add[node] = 0;
190 return 1;
191 }
192
193 Work update(int left, int right, int value, bool assignment) {
194 final work = Work();
195 void visit(int node) {
196 work.visited++;
197 final range = shape[node];
198 if (range.covered(left, right)) {
199 work.covered++;
200 _apply(node, value, assignment);
201 return;
202 }
203 work.pushes += _push(node);
204 if (left <= range.middle) visit(2 * node);
205 if (right > range.middle) visit(2 * node + 1);
206 _pull(node);
207 }
208 visit(1);
209 return work;
210 }
211
212 LazyAnswer query(int left, int right, [int node = 1]) {
213 final range = shape[node];
214 if (range.covered(left, right)) return (value: values[node], visited: 1, pushes: 0);
215 final pushes = _push(node);
216 if (right <= range.middle || left > range.middle) {
217 final child = query(left, right, right <= range.middle ? 2 * node : 2 * node + 1);
218 return (value: child.value, visited: child.visited + 1, pushes: child.pushes + pushes);
219 }
220 final a = query(left, right, 2 * node);
221 final b = query(left, right, 2 * node + 1);
222 return (value: merge(a.value, b.value), visited: a.visited + b.visited + 1, pushes: a.pushes + b.pushes + pushes);
223 }
224
225 Iterable<String> dump() sync* {
226 yield 'LAZY DUMP nodes=${shape.count}';
227 for (final (node, range) in shape.nodes) {
228 yield 'NODE $node ${range.left} ${range.right} ${aggregateText(values[node])} assign=${assign[node] ?? '-'} add=${add[node]}';
229 }
230 }
231}

270 ({int? index, int visited}) lower(int rest) {
271 var position = 0, visited = 0;
272 for (var step = 1 << log2(contents.length); step >= 1; step ~/= 2) {
273 if (position + step > contents.length) continue;
274 visited++;

277 rest -= value;
278 position += step;
279 }
280 }

311final class SparseTable {
312 final List<int> contents;
313 final List<List<int>> minimum = [], maximum = [];
314 int get entries => minimum.fold(0, (sum, row) => sum + row.length);
315 SparseTable(List<int> values) : contents = List.of(values) {
316 minimum.add(List.generate(values.length, (i) => i));
317 maximum.add(List.generate(values.length, (i) => i));
318 for (var level = 1; (1 << level) <= values.length; level++) {
319 final length = values.length - (1 << level) + 1, half = 1 << (level - 1);
320 final minima = List<int>.filled(length, 0), maxima = List<int>.filled(length, 0);
321 for (var i = 0; i < length; i++) {
322 minima[i] = _select(minimum[level - 1][i], minimum[level - 1][i + half], true);
323 maxima[i] = _select(maximum[level - 1][i], maximum[level - 1][i + half], false);
324 }
325 minimum.add(minima);
326 maximum.add(maxima);
327 }
328 }
329 int _select(int a, int b, bool isMinimum) => (isMinimum ? contents[a] <= contents[b] : contents[a] >= contents[b]) ? a : b;
330 SparseAnswer query(int left, int right) {
331 final level = log2(right - left + 1);
332 final end = right - (1 << level) + 1;
333 final a = _select(minimum[level][left], minimum[level][end], true);
334 final z = _select(maximum[level][left], maximum[level][end], false);
335 return (min: contents[a], minpos: a, max: contents[z], maxpos: z, level: level);
336 }
337 Iterable<String> dump() sync* {
338 yield 'SPARSE DUMP levels=${minimum.length}';
339 for (var level = 0; level < minimum.length; level++) {
340 yield 'MIN $level ${minimum[level].join(' ')}';
341 yield 'MAX $level ${maximum[level].join(' ')}';
342 }
343 }

346final class SqrtBlocks {
347 final int blockSize;
348 final List<int> raw, tag, sums;
349 int get count => tag.length;
350 SqrtBlocks(List<int> contents, this.blockSize)
351 : raw = List.of(contents),
352 tag = List.filled((contents.length + blockSize - 1) ~/ blockSize, 0),
353 sums = List.filled((contents.length + blockSize - 1) ~/ blockSize, 0) {
354 for (var i = 0; i < raw.length; i++) sums[i ~/ blockSize] += raw[i];
355 }
356 Partition _partition(int left, int right) {
357 final cells = <int>[], whole = <int>[];
358 final first = left ~/ blockSize, last = right ~/ blockSize;
359 if (first == last) {
360 for (var i = left; i <= right; i++) cells.add(i);
361 } else {
362 for (var i = left; i < (first + 1) * blockSize; i++) cells.add(i);
363 for (var i = last * blockSize; i <= right; i++) cells.add(i);
364 for (var block = first + 1; block < last; block++) whole.add(block);
365 }
366 return (cells: cells, whole: whole);
367 }
368 Partition add(int left, int right, int value) {
369 final split = _partition(left, right);
370 for (final i in split.cells) {
371 raw[i] += value;
372 sums[i ~/ blockSize] += value;
373 }
374 for (final block in split.whole) {
375 tag[block] += value;
376 sums[block] += value * blockSize;
377 }
378 return split;
379 }
380 SqrtAnswer query(int left, int right) {
381 final split = _partition(left, right);
382 var sum = 0;
383 for (final i in split.cells) sum += raw[i] + tag[i ~/ blockSize];
384 for (final block in split.whole) sum += sums[block];
385 return (sum: sum, partition: split);
386 }
387 Iterable<String> dump() sync* {
388 yield 'SQRT DUMP block=$blockSize blocks=$count';
389 for (var block = 0; block < count; block++) {
390 final first = block * blockSize, end = math.min((block + 1) * blockSize, raw.length);
391 yield 'BLOCK $block $first ${end - 1} add=${tag[block]} sum=${sums[block]} ${raw.sublist(first, end).join(' ')}';
392 }
393 }
394}

402final class PersistentTree {
403 final int n;
404 final List<int> values;
405 final List<PersistentNode> versions = [], valueRoots = [];
406 int positionNodes, valueNodes = 0;
407 int get distinct => values.length;
408 PersistentTree(List<int> contents)
409 : n = contents.length,
410 values = contents.toSet().toList()..sort(),
411 positionNodes = 2 * contents.length - 1 {
412 versions.add(_build(contents, 0, n - 1));
413 valueRoots.add(_build(List.filled(values.length, 0), 0, values.length - 1));
414 valueNodes = 2 * values.length - 1;
415 final ranks = {for (var i = 0; i < values.length; i++) values[i]: i};
416 for (final value in contents) {
417 final changed = _change(valueRoots.last, 0, values.length - 1, ranks[value]!, 1, true);
418 valueRoots.add(changed.node);
419 valueNodes += changed.created;
420 }
421 }
422 static PersistentNode _build(List<int> contents, int left, int right) {
423 if (left == right) return PersistentNode(contents[left]);
424 final middle = (left + right) ~/ 2;
425 final a = _build(contents, left, middle);
426 final b = _build(contents, middle + 1, right);
427 return PersistentNode(a.value + b.value, a, b);
428 }
429 static ({PersistentNode node, int created}) _change(PersistentNode node, int left, int right, int index, int value, bool increment) {
430 if (left == right) return (node: PersistentNode(increment ? node.value + value : value), created: 1);
431 final middle = (left + right) ~/ 2;
432 var a = node.left!, b = node.right!;
433 final changed = index <= middle
434 ? _change(a, left, middle, index, value, increment)
435 : _change(b, middle + 1, right, index, value, increment);
436 if (index <= middle) a = changed.node; else b = changed.node;
437 return (node: PersistentNode(a.value + b.value, a, b), created: changed.created + 1);
438 }
439 int setValue(int version, int index, int value) {
440 final changed = _change(versions[version], 0, n - 1, index, value, false);
441 versions.add(changed.node);
442 positionNodes += changed.created;
443 return changed.created;
444 }
445 static ScalarAnswer _sum(PersistentNode node, int nl, int nr, int left, int right) {
446 if (left <= nl && nr <= right) return (value: node.value, visited: 1);
447 final middle = (nl + nr) ~/ 2;
448 if (right <= middle || left > middle) {
449 final child = right <= middle ? _sum(node.left!, nl, middle, left, right) : _sum(node.right!, middle + 1, nr, left, right);
450 return (value: child.value, visited: child.visited + 1);
451 }
452 final a = _sum(node.left!, nl, middle, left, right);
453 final b = _sum(node.right!, middle + 1, nr, left, right);
454 return (value: a.value + b.value, visited: a.visited + b.visited + 1);
455 }
456 ScalarAnswer query(int version, int left, int right) => _sum(versions[version], 0, n - 1, left, right);
457 ScalarAnswer kth(int left, int right, int k) {
458 var u = valueRoots[right + 1], w = valueRoots[left];
459 var p = 0, q = values.length - 1, visited = 1;
460 while (p < q) {
461 final middle = (p + q) ~/ 2;
462 final difference = u.left!.value - w.left!.value;
463 if (k <= difference) {
464 u = u.left!;
465 w = w.left!;
466 q = middle;
467 } else {
468 k -= difference;
469 u = u.right!;
470 w = w.right!;
471 p = middle + 1;
472 }
473 visited++;
474 }
475 return (value: values[p], visited: visited);
476 }

603 final version = versions.versions.length - 1;
604 final answer = versions.query(version, left, right);
605 report('PERSIST', answer.value == sum, 'version=$version ${sumText(answer)}');
606 }
607 rows.add('RESULT ok=$ok diff=$diff');
608 return rows;
609 }

611 List<String> _mo(List<String> words) {
612 final queries = <MoQuery>[];
613 for (var i = 1; i < words.length; i += 2) {
614 final left = int.parse(words[i]), right = int.parse(words[i + 1]);
615 _checkRange(left, right);
616 queries.add((left: left, right: right, number: queries.length + 1));
617 }
618 final blockSize = integerSqrt(array.length);
619 final order = List<MoQuery>.of(queries)..sort((a, b) {
620 final blockA = a.left ~/ blockSize, blockB = b.left ~/ blockSize;
621 if (blockA != blockB) return blockA.compareTo(blockB);
622 if (a.right != b.right) return blockA.isEven ? a.right.compareTo(b.right) : b.right.compareTo(a.right);
623 return a.number.compareTo(b.number);
624 });
625 final frequency = <int, int>{};
626 final answers = List<int>.filled(queries.length, 0);
627 var leftEdge = 0, rightEdge = -1, moves = 0;
628 void add(int i) {
629 frequency[array[i]] = (frequency[array[i]] ?? 0) + 1;
630 moves++;
631 }
632 void remove(int i) {
633 final count = frequency[array[i]]! - 1;
634 if (count == 0) frequency.remove(array[i]); else frequency[array[i]] = count;
635 moves++;
636 }
637 for (final query in order) {
638 while (leftEdge > query.left) add(--leftEdge);
639 while (rightEdge < query.right) add(++rightEdge);
640 while (leftEdge < query.left) remove(leftEdge++);
641 while (rightEdge > query.right) remove(rightEdge--);
642 answers[query.number - 1] = frequency.length;
643 }
644 return [
645 'MO queries=${queries.length} block=$blockSize moves=$moves',
646 'ORDER ${order.map((q) => q.number).join(' ')}',
647 for (final query in queries) 'ANSWER ${query.number} ${query.left} ${query.right} distinct=${answers[query.number - 1]}',
648 ];
649 }

651 List<String> execute(String line) {
652 final words = line.split(' ');
653 int number(int index) => int.parse(words[index]);
654 final kind = words[0];
655 if (kind == 'ARRAY') {
656 final left = number(1), right = number(2);
657 _checkRange(left, right);
658 return ['ARRAY $left $right ${array.sublist(left, right + 1).join(' ')}'];
659 }
660 if (kind == 'LOAD') {
661 final removed = structureCount;
662 original = words.skip(1).map(int.parse).toList();
663 array = List.of(original);
664 segment = null; lazy = null; fenwick = null; rangeFenwick = null; sparse = null; sqrt = null; persistent = null;
665 return ['LOAD n=${array.length} removed=$removed'];
666 }
667 if (kind == 'RESET') {
668 final rebuilt = structureCount;
669 array = List.of(original);
670 if (segment != null) segment = SegmentTree(array, segment!.operation);
671 if (lazy != null) lazy = LazyTree(array);
672 if (fenwick != null) fenwick = PointFenwick(array);
673 if (rangeFenwick != null) rangeFenwick = RangeFenwick(array);
674 if (sparse != null) sparse = SparseTable(array);
675 if (sqrt != null) sqrt = SqrtBlocks(array, sqrt!.blockSize);
676 if (persistent != null) persistent = PersistentTree(array);
677 return ['RESET rebuilt=$rebuilt'];
678 }
679 if (kind == 'CHECK') return _check(number(1), number(2));
680 if (kind == 'MO') return _mo(words);
681 final action = words[1];
682 switch (kind) {
683 case 'SEG':
684 if (action == 'BUILD') {
685 final matches = Operation.values.where((op) => op.name.toUpperCase() == words[2]);
686 if (matches.isEmpty) throw CommandError('BAD OP ${words[2]}');
687 final tree = SegmentTree(array, matches.first);
688 segment = tree;
689 return ['SEG BUILD ${words[2]} nodes=${tree.shape.count} maxnode=${tree.shape.maxNode} root=${tree.root}'];
690 }
691 final tree = _need(segment, kind);
692 if (action == 'DUMP') return tree.dump().toList();
693 final left = number(2);
694 if (action == 'UPDATE') {
695 _checkIndex(left);
696 final value = number(3);
697 final visited = tree.update(left, value);
698 array[left] = value;
699 return ['SEG UPDATE $left $value root=${tree.root} visited=$visited'];
700 }
701 final right = number(3);
702 _checkRange(left, right);
703 return ['SEG QUERY $left $right ${tree.describe(tree.query(left, right))}'];
704 case 'LAZY':
705 if (action == 'BUILD') {
706 final tree = LazyTree(array);
707 lazy = tree;
708 return ['LAZY BUILD nodes=${tree.shape.count} ${aggregateText(tree.root)}'];
709 }
710 final tree = _need(lazy, kind);
711 if (action == 'DUMP') return tree.dump().toList();
712 final left = number(2), right = number(3);
713 _checkRange(left, right);
714 if (action == 'QUERY') return ['LAZY QUERY $left $right ${lazyText(tree.query(left, right))}'];
715 final value = number(4);
716 final assignment = action == 'ASSIGN';
717 final work = tree.update(left, right, value, assignment);
718 _changeArray(left, right, value, assignment);
719 return ['LAZY $action $left $right $value $work'];
720 case 'FEN':
721 if (action == 'BUILD') {
722 final tree = PointFenwick(array);
723 fenwick = tree;
724 return ['FEN BUILD n=${array.length} total=${tree.bits.prefix(array.length).value}'];
725 }
726 final tree = _need(fenwick, kind);
727 if (action == 'DUMP') return ['FEN DUMP n=${array.length}', tree.bits.dump('T')];
728 if (action == 'LOWER') {
729 for (var i = 0; i < array.length; i++) {
730 if (tree.contents[i] < 0) throw CommandError('NEGATIVE $i');
731 }
732 final target = number(2);
733 final answer = tree.lower(target);
734 return ['FEN LOWER $target index=${answer.index ?? 'none'} visited=${answer.visited}'];
735 }
736 final left = number(2);
737 if (action == 'RANGE') {
738 final right = number(3);
739 _checkRange(left, right);
740 return ['FEN RANGE $left $right ${sumText(tree.query(left, right))}'];
741 }
742 _checkIndex(left);
743 if (action == 'PREFIX') return ['FEN PREFIX $left ${sumText(tree.bits.prefix(left + 1))}'];
744 final value = number(3);
745 final visited = tree.add(left, value);
746 array[left] += value;
747 return ['FEN ADD $left $value visited=$visited'];
748 case 'RFEN':
749 if (action == 'BUILD') {
750 final tree = RangeFenwick(array);
751 rangeFenwick = tree;
752 return ['RFEN BUILD n=${array.length} total=${tree.query(0, array.length - 1).value}'];
753 }
754 final tree = _need(rangeFenwick, kind);
755 if (action == 'DUMP') return ['RFEN DUMP n=${array.length}', tree.b1.dump('B1'), tree.b2.dump('B2')];
756 final left = number(2), right = number(3);
757 _checkRange(left, right);
758 if (action == 'SUM') return ['RFEN SUM $left $right ${sumText(tree.query(left, right))}'];
759 final value = number(4);
760 final visited = tree.add(left, right, value);
761 _changeArray(left, right, value);
762 return ['RFEN ADD $left $right $value visited=$visited'];
763 case 'SPARSE':
764 if (action == 'BUILD') {
765 final tree = SparseTable(array);
766 sparse = tree;
767 return ['SPARSE BUILD levels=${tree.minimum.length} entries=${tree.entries}'];
768 }
769 final tree = _need(sparse, kind);
770 if (action == 'DUMP') return tree.dump().toList();
771 final left = number(2), right = number(3);
772 _checkRange(left, right);
773 return ['SPARSE QUERY $left $right ${sparseText(tree.query(left, right))}'];
774 case 'SQRT':
775 if (action == 'BUILD') {
776 final size = words.length == 3 ? number(2) : integerSqrt(array.length);
777 if (size > array.length) throw CommandError('BAD BLOCK $size');
778 final tree = SqrtBlocks(array, size);
779 sqrt = tree;
780 return ['SQRT BUILD block=$size blocks=${tree.count}'];
781 }
782 final tree = _need(sqrt, kind);
783 if (action == 'DUMP') return tree.dump().toList();
784 final left = number(2), right = number(3);
785 _checkRange(left, right);
786 if (action == 'SUM') return ['SQRT SUM $left $right ${sqrtText(tree.query(left, right))}'];
787 final value = number(4);
788 final split = tree.add(left, right, value);
789 _changeArray(left, right, value);
790 return ['SQRT ADD $left $right $value ${partitionText(split)}'];
791 case 'PERSIST':
792 if (action == 'BUILD') {
793 final tree = PersistentTree(array);
794 persistent = tree;
795 return ['PERSIST BUILD nodes=${tree.positionNodes} distinct=${tree.distinct} valuenodes=${tree.valueNodes}'];
796 }
797 final tree = _need(persistent, kind);
798 if (action == 'NODES') return ['PERSIST NODES versions=${tree.versions.length} position=${tree.positionNodes} value=${tree.valueNodes}'];
799 if (action == 'KTH') {
800 final left = number(2), right = number(3), k = number(4);
801 _checkRange(left, right);
802 if (k < 1 || k > right - left + 1) throw CommandError('BAD K $k');
803 final answer = tree.kth(left, right, k);
804 return ['PERSIST KTH $left $right $k value=${answer.value} visited=${answer.visited}'];
805 }
806 final version = number(2);
807 _checkVersion(tree, version);
808 final index = number(3);
809 if (action == 'SET') {
810 _checkIndex(index);
811 final value = number(4);
812 final created = tree.setValue(version, index, value);
813 return ['PERSIST SET $version $index $value version=${tree.versions.length - 1} created=$created'];
814 }
815 final end = number(4);
816 _checkRange(index, end);
817 return ['PERSIST QUERY $version $index $end ${sumText(tree.query(version, index, end))}'];
818 case 'GRID':
819 if (action == 'BUILD') {
820 final height = number(2), width = number(3), count = words.length - 4;
821 if (count != height * width) throw CommandError('BAD COUNT $count ${height * width}');
822 final values = words.skip(4).map(int.parse).toList();
823 grid = GridFenwick(height, width, values);
824 return ['GRID BUILD rows=$height cols=$width total=${values.fold<int>(0, (a, b) => a + b)}'];
825 }
826 final tree = _need(grid, kind);
827 if (action == 'DUMP') return tree.dump().toList();
828 final row = number(2), column = number(3);
829 _checkCell(tree, row, column);
830 if (action == 'ADD') {
831 final value = number(4);
832 return ['GRID ADD $row $column $value visited=${tree.add(row, column, value)}'];
833 }
834 final bottom = number(4), right = number(5);
835 _checkCell(tree, bottom, right);
836 if (row > bottom || column > right) throw CommandError('BAD RECT $row $column $bottom $right');
837 return ['GRID SUM $row $column $bottom $right ${sumText(tree.query(row, column, bottom, right))}'];
838 }
839 throw StateError('Unknown command');
840 }
